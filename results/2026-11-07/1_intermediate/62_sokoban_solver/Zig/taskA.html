27fn isNumber(text: []const u8) bool {
28 if (text.len == 0 or text.len > 6 or (text.len > 1 and text[0] == '0')) return false;
29 for (text) |letter| if (!std.ascii.isDigit(letter)) return false;
30 return true;
31}

33fn decode(gpa: Allocator, text: []const u8, allowed: []const u8) Allocator.Error!?[]const u8 {
34 var result: std.ArrayList(u8) = .empty;
35 var index: usize = 0;
36 while (index < text.len) {
37 const start = index;
38 while (index < text.len and std.ascii.isDigit(text[index])) : (index += 1) {}
39 const digits = text[start..index];
40 if (digits.len > 2 or (digits.len > 0 and digits[0] == '0') or index == text.len) return null;
41 const letter = text[index];
42 if (!includes(allowed, letter)) return null;
43 const count = if (digits.len == 0) 1 else number(digits);
44 try result.appendNTimes(gpa, letter, count);
45 index += 1;
46 }
47 return result.items;
48}

50fn encode(gpa: Allocator, text: []const u8) Allocator.Error![]const u8 {
51 var result: std.ArrayList(u8) = .empty;
52 var index: usize = 0;
53 while (index < text.len) {
54 var end = index + 1;
55 while (end < text.len and text[end] == text[index]) : (end += 1) {}
56 if (end - index > 1) {
57 const digits = try std.fmt.allocPrint(gpa, "{d}", .{end - index});
58 try result.appendSlice(gpa, digits);
59 }
60 try result.append(gpa, text[index]);
61 index = end;
62 }
63 return result.items;
64}

104 var result = self;

158const Level = struct {
159 height: usize = 0,
160 width: usize = 0,
161 walls: Flags = @splat(false),
162 goals: Flags = @splat(false),
163 interior: Flags = @splat(false),
164 dead: Flags = @splat(false),
165 goal_cells: Boxes = .{},
166 distances: [4][100]i16 = @splat(@splat(-1)),
167 reason: ?[]const u8 = null,
168
169 fn size(self: *const Level) usize {
170 return self.height * self.width;
171 }
172
173 fn neighbor(self: *const Level, cell: u8, direction: usize) ?u8 {
174 const row = cell / self.width;
175 const column = cell % self.width;
176 const destination: usize = switch (direction) {
177 0 => if (row > 0) cell - self.width else return null,
178 1 => if (row + 1 < self.height) cell + self.width else return null,
179 2 => if (column > 0) cell - 1 else return null,
180 3 => if (column + 1 < self.width) cell + 1 else return null,
181 else => unreachable,
182 };
183 return @intCast(destination);
184 }
185
186 fn cellText(self: *const Level, gpa: Allocator, cell: u8) Allocator.Error![]const u8 {
187 return std.fmt.allocPrint(gpa, "{d} {d}", .{ cell / self.width + 1, cell % self.width + 1 });
188 }

190 fn analyze(self: *Level) void {
191 self.distances = @splat(@splat(-1));
192 self.dead = @splat(false);
193 for (self.goal_cells.slice(), 0..) |goal, goal_index| {
194 const distance = &self.distances[goal_index];
195 distance[goal] = 0;
196 var queue: [100]u8 = undefined;
197 queue[0] = goal;
198 var head: usize = 0;
199 var tail: usize = 1;
200 while (head < tail) : (head += 1) {
201 const cell = queue[head];
202 for (0..4) |direction| {
203 const box = self.neighbor(cell, opposite[direction]) orelse continue;
204 const support = self.neighbor(box, opposite[direction]) orelse continue;
205 if (self.interior[box] and self.interior[support] and distance[box] == -1) {
206 distance[box] = distance[cell] + 1;
207 queue[tail] = box;
208 tail += 1;
209 }
210 }
211 }
212 }
213 for (0..self.size()) |cell| {
214 self.dead[cell] = self.interior[cell] and self.goalDistance(@intCast(cell)) == null;
215 }
216 }
217
218 fn goalDistance(self: *const Level, cell: u8) ?i16 {
219 var best: ?i16 = null;
220 for (0..self.goal_cells.len) |index| {
221 const distance = self.distances[index][cell];
222 if (distance >= 0 and (best == null or distance < best.?)) best = distance;
223 }
224 return best;
225 }

227 fn onGoals(self: *const Level, boxes: Boxes) usize {
228 var count: usize = 0;
229 for (boxes.slice()) |box| {
230 if (self.goals[box]) count += 1;
231 }
232 return count;
233 }
234
235 fn solved(self: *const Level, boxes: Boxes) bool {
236 return self.onGoals(boxes) == boxes.len;
237 }

239 fn reachable(self: *const Level, player: u8, boxes: Boxes) Flags {
240 var seen: Flags = @splat(false);
241 seen[player] = true;
242 var queue: [100]u8 = undefined;
243 queue[0] = player;
244 var head: usize = 0;
245 var tail: usize = 1;
246 while (head < tail) : (head += 1) {
247 for (0..4) |direction| {
248 const cell = self.neighbor(queue[head], direction) orelse continue;
249 if (self.interior[cell] and !seen[cell] and !boxes.contains(cell)) {
250 seen[cell] = true;
251 queue[tail] = cell;
252 tail += 1;
253 }
254 }
255 }
256 return seen;
257 }

259 fn representative(self: *const Level, player: u8, boxes: Boxes) u8 {
260 return firstFlag(self.reachable(player, boxes));
261 }

263 fn walkPath(self: *const Level, gpa: Allocator, start: u8, target: u8, boxes: Boxes) Allocator.Error![]const u8 {
264 var seen: Flags = @splat(false);
265 var previous: [100]u8 = undefined;
266 var came_direction: [100]u8 = undefined;
267 var queue: [100]u8 = undefined;
268 seen[start] = true;
269 queue[0] = start;
270 var head: usize = 0;
271 var tail: usize = 1;
272 while (head < tail) : (head += 1) {
273 const cell = queue[head];
274 for (0..4) |direction| {
275 const neighbor_cell = self.neighbor(cell, direction) orelse continue;
276 if (self.interior[neighbor_cell] and !seen[neighbor_cell] and !boxes.contains(neighbor_cell)) {
277 seen[neighbor_cell] = true;
278 previous[neighbor_cell] = cell;
279 came_direction[neighbor_cell] = directions[direction];
280 queue[tail] = neighbor_cell;
281 tail += 1;
282 }
283 }
284 }
285 var letters: [100]u8 = undefined;
286 var len: usize = 0;
287 var cell = target;
288 while (cell != start) {
289 letters[len] = came_direction[cell];
290 len += 1;
291 cell = previous[cell];
292 }
293 std.mem.reverse(u8, letters[0..len]);
294 return gpa.dupe(u8, letters[0..len]);
295 }

297 fn step(self: *const Level, position: Position, direction: usize) Step {
298 const target = self.neighbor(position.player, direction) orelse
299 return .{ .position = position, .failure = "wall" };
300 if (self.walls[target]) return .{ .position = position, .failure = "wall" };
301 if (!position.boxes.contains(target)) {
302 return .{ .position = .{ .player = target, .boxes = position.boxes }, .letter = directions[direction] };
303 }
304 const beyond = self.neighbor(target, direction) orelse
305 return .{ .position = position, .failure = "blocked" };
306 if (self.walls[beyond] or position.boxes.contains(beyond)) {
307 return .{ .position = position, .failure = "blocked" };
308 }
309 return .{
310 .position = .{ .player = target, .boxes = position.boxes.moved(target, beyond) },
311 .letter = std.ascii.toUpper(directions[direction]),
312 };
313 }

315 fn pushable(self: *const Level, box: u8, direction: usize, boxes: Boxes, reach: Flags) bool {
316 const destination = self.neighbor(box, direction) orelse return false;
317 const support = self.neighbor(box, opposite[direction]) orelse return false;
318 return reach[support] and self.interior[destination] and !boxes.contains(destination);
319 }
320
321 fn sideBlocked(self: *const Level, box: u8, a: u8, b: u8, boxes: Boxes, assumed: Flags) bool {
322 if (self.walls[a] or self.walls[b] or assumed[a] or assumed[b]) return true;
323 if (self.dead[a] and self.dead[b]) return true;
324 var extended = assumed;
325 extended[box] = true;
326 return (boxes.contains(a) and self.frozen(a, boxes, extended)) or
327 (boxes.contains(b) and self.frozen(b, boxes, extended));
328 }
329
330 fn frozen(self: *const Level, box: u8, boxes: Boxes, assumed: Flags) bool {
331 const left = self.neighbor(box, 2).?;
332 const right = self.neighbor(box, 3).?;
333 const up = self.neighbor(box, 0).?;
334 const down = self.neighbor(box, 1).?;
335 return self.sideBlocked(box, left, right, boxes, assumed) and
336 self.sideBlocked(box, up, down, boxes, assumed);
337 }
338
339 fn hasFrozenBox(self: *const Level, boxes: Boxes) bool {
340 for (boxes.slice()) |box| {
341 if (!self.goals[box] and self.frozen(box, boxes, @splat(false))) return true;
342 }
343 return false;
344 }

346 fn symbol(self: *const Level, cell: u8, position: Position, dead_map: bool) u8 {
347 if (self.walls[cell]) return '#';
348 if (dead_map) return if (self.goals[cell]) '.' else if (self.dead[cell]) 'x' else '-';
349 if (position.player == cell) return if (self.goals[cell]) '+' else '@';
350 if (position.boxes.contains(cell)) return if (self.goals[cell]) '*' else '$';
351 return if (self.goals[cell]) '.' else '-';
352 }
353
354 fn mapCell(self: *const Level, kind: Transform, cell: u8) u8 {
355 const row = cell / self.width;
356 const column = cell % self.width;
357 return @intCast(switch (kind) {
358 .ROT90 => column * self.height + self.height - 1 - row,
359 .ROT180 => (self.height - 1 - row) * self.width + self.width - 1 - column,
360 .ROT270 => (self.width - 1 - column) * self.height + row,
361 .FLIPH => row * self.width + self.width - 1 - column,
362 .FLIPV => (self.height - 1 - row) * self.width + column,
363 });
364 }
365
366 fn mapPosition(self: *const Level, kind: Transform, position: Position) Position {
367 var result = position;
368 result.player = self.mapCell(kind, position.player);
369 for (result.boxes.cells[0..result.boxes.len]) |*box| box.* = self.mapCell(kind, box.*);
370 std.mem.sort(u8, result.boxes.cells[0..result.boxes.len], {}, before);
371 return result;
372 }
373
374 fn transformed(self: *const Level, kind: Transform) Level {
375 const rotate = kind == .ROT90 or kind == .ROT270;
376 var result: Level = .{
377 .height = if (rotate) self.width else self.height,
378 .width = if (rotate) self.height else self.width,
379 };
380 for (0..self.size()) |cell| {
381 const moved = self.mapCell(kind, @intCast(cell));
382 result.walls[moved] = self.walls[cell];
383 result.goals[moved] = self.goals[cell];
384 result.interior[moved] = self.interior[cell];
385 }
386 for (0..result.size()) |cell| {
387 if (result.goals[cell]) {
388 result.goal_cells.cells[result.goal_cells.len] = @intCast(cell);
389 result.goal_cells.len += 1;
390 }
391 }
392 result.analyze();
393 return result;
394 }

397const Loaded = struct { level: Level, position: Position };

399fn buildLevel(gpa: Allocator, rows: []const []const u8) Allocator.Error!Loaded {
400 var level: Level = .{ .height = rows.len, .width = rows[0].len };
401 var position: Position = .{};
402 var players: usize = 0;
403 for (rows, 0..) |row, row_index| {
404 for (row, 0..) |symbol, column| {
405 const cell: u8 = @intCast(row_index * level.width + column);
406 level.walls[cell] = symbol == '#';
407 level.goals[cell] = includes(".*+", symbol);
408 if (level.goals[cell]) {
409 level.goal_cells.cells[level.goal_cells.len] = cell;
410 level.goal_cells.len += 1;
411 }
412 if (includes("$*", symbol)) {
413 position.boxes.cells[position.boxes.len] = cell;
414 position.boxes.len += 1;
415 }
416 if (includes("@+", symbol)) {
417 if (players == 0) position.player = cell;
418 players += 1;
419 }
420 }
421 }
422 if (players == 1) {
423 var queue: [100]u8 = undefined;
424 queue[0] = position.player;
425 level.interior[position.player] = true;
426 var head: usize = 0;
427 var tail: usize = 1;
428 while (head < tail) : (head += 1) {
429 for (0..4) |direction| {
430 const cell = level.neighbor(queue[head], direction) orelse continue;
431 if (!level.walls[cell] and !level.interior[cell]) {
432 level.interior[cell] = true;
433 queue[tail] = cell;
434 tail += 1;
435 }
436 }
437 }
438 }
439 if (players != 1) {
440 level.reason = try std.fmt.allocPrint(gpa, "players {d}", .{players});
441 } else if (position.boxes.len == 0) {
442 level.reason = "no boxes";
443 } else if (position.boxes.len != level.goal_cells.len) {
444 level.reason = try std.fmt.allocPrint(gpa, "boxes {d} goals {d}", .{ position.boxes.len, level.goal_cells.len });
445 } else {
446 for (0..level.size()) |cell| {
447 const row = cell / level.width;
448 const column = cell % level.width;
449 if (level.interior[cell] and (row == 0 or row + 1 == level.height or column == 0 or column + 1 == level.width)) {
450 level.reason = try std.fmt.allocPrint(gpa, "open {d} {d}", .{ row + 1, column + 1 });
451 break;
452 }
453 }
454 if (level.reason == null) {
455 for (0..level.size()) |cell| {
456 if ((level.goals[cell] or position.boxes.contains(@intCast(cell))) and !level.interior[cell]) {
457 level.reason = try std.fmt.allocPrint(gpa, "outside {d} {d}", .{ cell / level.width + 1, cell % level.width + 1 });
458 break;
459 }
460 }
461 }
462 }
463 if (level.reason == null) level.analyze();
464 return .{ .level = level, .position = position };
465}

467const SearchNode = struct {
468 boxes: Boxes,
469 representative: u8,
470 parent: ?usize = null,
471 box: u8 = 0,
472 direction: usize = 0,
473};
474const SearchKey = struct { boxes: [4]u8, representative: u8 };
475const Outcome = enum { found, no_solution, limit };
476const SearchResult = struct { outcome: Outcome, expanded: usize, states: usize, notation: []const u8 = "" };
477
478fn solutionNotation(gpa: Allocator, level: *const Level, start: Position, nodes: []const SearchNode) Allocator.Error![]const u8 {
479 var pushes: std.ArrayList(SearchNode) = .empty;
480 var index = nodes.len - 1;
481 while (nodes[index].parent) |parent| {
482 try pushes.append(gpa, nodes[index]);
483 index = parent;
484 }
485 var notation: std.ArrayList(u8) = .empty;
486 var position = start;
487 var remaining = pushes.items.len;
488 while (remaining > 0) {
489 remaining -= 1;
490 const push = pushes.items[remaining];
491 const support = level.neighbor(push.box, opposite[push.direction]).?;
492 const path = try level.walkPath(gpa, position.player, support, position.boxes);
493 try notation.appendSlice(gpa, path);
494 try notation.append(gpa, std.ascii.toUpper(directions[push.direction]));
495 position = .{ .player = push.box, .boxes = position.boxes.moved(push.box, level.neighbor(push.box, push.direction).?) };
496 }
497 return notation.items;
498}

500fn pushSearch(gpa: Allocator, level: *const Level, start: Position, fixed: Boxes, target: ?u8, pruning: bool, limit: ?usize) Allocator.Error!SearchResult {
501 const representative = level.representative(start.player, start.boxes);
502 var nodes: std.ArrayList(SearchNode) = .empty;
503 var known: std.AutoHashMapUnmanaged(SearchKey, void) = .empty;
504 try nodes.append(gpa, .{ .boxes = start.boxes, .representative = representative });
505 try known.put(gpa, .{ .boxes = start.boxes.cells, .representative = representative }, {});
506 var expanded: usize = 0;
507 while (expanded < nodes.items.len) {
508 if (limit != null and expanded == limit.?) return .{ .outcome = .limit, .expanded = expanded, .states = nodes.items.len };
509 const parent = expanded;
510 const node = nodes.items[parent];
511 expanded += 1;
512 const reach = level.reachable(node.representative, node.boxes);
513 for (node.boxes.slice()) |box| {
514 if (fixed.contains(box)) continue;
515 for (0..4) |direction| {
516 if (!level.pushable(box, direction, node.boxes, reach)) continue;
517 const destination = level.neighbor(box, direction).?;
518 const boxes = node.boxes.moved(box, destination);
519 if (pruning and (level.dead[destination] or level.hasFrozenBox(boxes))) continue;
520 const next_rep = level.representative(box, boxes);
521 const entry = try known.getOrPut(gpa, .{ .boxes = boxes.cells, .representative = next_rep });
522 if (entry.found_existing) continue;
523 try nodes.append(gpa, .{ .boxes = boxes, .representative = next_rep, .parent = parent, .box = box, .direction = direction });
524 const reached = if (target) |goal| destination == goal else level.solved(boxes);
525 if (reached) return .{
526 .outcome = .found,
527 .expanded = expanded,
528 .states = nodes.items.len,
529 .notation = try solutionNotation(gpa, level, start, nodes.items),
530 };
531 }
532 }
533 }
534 return .{ .outcome = .no_solution, .expanded = expanded, .states = nodes.items.len };
535}

581fn validArgument(gpa: Allocator, kind: Kind, text: []const u8) Allocator.Error!bool {
582 switch (kind) {
583 .number => return isNumber(text),
584 .steps => return (try decode(gpa, text, step_characters)) != null,
585 .direction => return text.len == 1 and includes(directions, text[0]),
586 .name => {
587 if (text.len == 0 or text.len > 8) return false;
588 for (text) |letter| if (letter < 'a' or letter > 'z') return false;
589 return true;
590 },
591 .transform => return std.meta.stringToEnum(Transform, text) != null,
592 .text => return true,
593 }
594}

604const Assignment = struct { order: [4]usize = @splat(0), total: ?usize = null };
605
606fn assignGoals(level: *const Level, boxes: Boxes, depth: usize, used: [4]bool, order: [4]usize, total: usize, best: *Assignment) void {
607 if (depth == boxes.len) {
608 if (best.total == null or total < best.total.?) best.* = .{ .order = order, .total = total };
609 return;
610 }
611 for (0..level.goal_cells.len) |goal| {
612 const distance = level.distances[goal][boxes.cells[depth]];
613 if (used[goal] or distance < 0) continue;
614 var next_used = used;
615 var next_order = order;
616 next_used[goal] = true;
617 next_order[depth] = goal;
618 assignGoals(level, boxes, depth + 1, next_used, next_order, total + @as(usize, @intCast(distance)), best);
619 }
620}

630 redo_list: std.ArrayList(u8) = .empty,
631 snapshots: std.StringHashMapUnmanaged(Snapshot) = .empty,
632 solution: ?Solution = null,
633 problem: []const u8 = "",
634
635 fn fail(self: *Game, comptime pattern: []const u8, args: anytype) Error {
636 self.problem = std.fmt.allocPrint(self.scratch, pattern, args) catch return error.OutOfMemory;
637 return error.Invalid;
638 }
639
640 fn load(self: *Game, rows: []const []const u8) Allocator.Error!void {
641 const loaded = try buildLevel(self.gpa, rows);
642 self.level = loaded.level;
643 self.initial = loaded.position;
644 self.current = loaded.position;
645 self.history.clearRetainingCapacity();
646 self.redo_list.clearRetainingCapacity();
647 self.snapshots.clearRetainingCapacity();
648 self.solution = null;
649 }
650
651 fn cellArgument(self: *const Game, arguments: []const []const u8, index: usize) u8 {
652 return @intCast((number(arguments[index]) - 1) * self.level.width + number(arguments[index + 1]) - 1);
653 }
654
655 fn simulate(self: *const Game, letters: []const u8) Allocator.Error!Simulation {
656 var position = self.current;
657 var notation: std.ArrayList(u8) = .empty;
658 for (letters, 0..) |letter, index| {
659 const direction = std.mem.findScalar(u8, directions, std.ascii.toLower(letter)).?;
660 const result = self.level.step(position, direction);
661 if (result.failure) |failure| return .{
662 .position = position,
663 .letters = notation.items,
664 .failed_step = index + 1,
665 .failure = failure,
666 };
667 position = result.position;
668 try notation.append(self.scratch, result.letter);
669 }
670 return .{ .position = position, .letters = notation.items };
671 }
672
673 fn takeSteps(self: *Game, letters: []const u8) Allocator.Error!Simulation {
674 const result = try self.simulate(letters);
675 self.current = result.position;
676 try self.history.appendSlice(self.gpa, result.letters);
677 if (result.letters.len > 0) self.redo_list.clearRetainingCapacity();
678 return result;
679 }

681 fn progress(self: *const Game, boxes: Boxes) Writer.Error!void {
682 const on_goals = self.level.onGoals(boxes);
683 try self.out.print("goals {d}/{d}", .{ on_goals, boxes.len });
684 if (on_goals == boxes.len) try self.out.writeAll(" solved");
685 try self.out.writeByte('\n');
686 }

688 fn counted(self: *const Game, label: []const u8, letters: []const u8) Writer.Error!void {
689 try self.out.print("{s} {d}", .{ label, letters.len });
690 if (letters.len > 0) try self.out.print(" {s}", .{letters});
691 try self.out.writeByte('\n');
692 }

694 fn show(self: *const Game, dead_map: bool, export_level: bool) Error!void {
695 if (export_level) try self.out.writeAll("level ");
696 for (0..self.level.height) |row| {
697 var symbols: [10]u8 = undefined;
698 for (0..self.level.width) |column| {
699 symbols[column] = self.level.symbol(@intCast(row * self.level.width + column), self.current, dead_map);
700 }
701 const text = symbols[0..self.level.width];
702 if (export_level) {
703 if (row > 0) try self.out.writeByte('|');
704 try self.out.writeAll(try encode(self.scratch, text));
705 } else try printLine(self.out, "{s}", .{text});
706 }
707 if (export_level) try self.out.writeByte('\n');
708 }

710 fn solveOrHint(self: *Game, command: Command, limit: usize) Error!void {
711 if (command == .SOLVE) self.solution = null;
712 if (self.level.solved(self.current.boxes)) {
713 try self.out.writeAll("already solved\n");
714 return;
715 }
716 const result = try pushSearch(self.scratch, &self.level, self.current, .{}, null, true, limit);
717 if (result.outcome != .found) {
718 try printLine(self.out, "{s} expanded {d} states {d}", .{
719 if (result.outcome == .limit) @as([]const u8, "limit") else "no solution", result.expanded, result.states,
720 });
721 return;
722 }
723 const notation = result.notation;
724 if (command == .SOLVE) {
725 try printLine(self.out, "solution pushes {d} moves {d} expanded {d} states {d}", .{
726 pushCount(notation), notation.len, result.expanded, result.states,
727 });
728 try printLine(self.out, "{s}", .{notation});
729 self.solution = .{ .position = self.current, .notation = try self.gpa.dupe(u8, notation) };
730 } else {
731 var end: usize = 0;
732 while (!std.ascii.isUpper(notation[end])) : (end += 1) {}
733 const part = notation[0 .. end + 1];
734 _ = try self.takeSteps(part);
735 try printLine(self.out, "hint {s}", .{part});
736 }
737 }
738
739 fn process(self: *Game, command: Command, a: []const []const u8) Error!void {
740 switch (command) {
741 .IMPORT => {
742 var rows: std.ArrayList([]const u8) = .empty;
743 var parts = std.mem.splitScalar(u8, a[0], '|');
744 while (parts.next()) |part| {
745 const row = (try decode(self.scratch, part, level_characters)) orelse return self.fail("bad level", .{});
746 try rows.append(self.scratch, row);
747 }
748 if (rows.items.len > 10 or rows.items[0].len == 0 or rows.items[0].len > 10) return self.fail("bad level", .{});
749 for (rows.items) |row| if (row.len != rows.items[0].len) return self.fail("bad level", .{});
750 try self.load(rows.items);
751 try printLine(self.out, "loaded {d}x{d}", .{ self.level.height, self.level.width });
752 },
753 .VALID => {
754 if (self.level.reason) |reason| try printLine(self.out, "invalid {s}", .{reason}) else try self.out.writeAll("valid\n");
755 },
756 .INFO => try printLine(self.out, "size {d}x{d} walls {d} cells {d} boxes {d} dead {d}", .{
757 self.level.height, self.level.width, countFlags(self.level.walls), countFlags(self.level.interior), self.current.boxes.len, countFlags(self.level.dead),
758 }),
759 .SHOW => try self.show(false, false),
760 .EXPORT => try self.show(false, true),
761 .MOVE => {
762 const result = try self.takeSteps((try decode(self.scratch, a[0], step_characters)).?);
763 if (result.letters.len > 0) try printLine(self.out, "moved {s}", .{result.letters});
764 if (result.failure) |failure| return self.fail("step {d} {s}", .{ result.failed_step, failure });
765 },
766 .PUSH => {
767 const box = self.cellArgument(a, 0);
768 const direction = std.mem.findScalar(u8, directions, a[2][0]).?;
769 if (!self.current.boxes.contains(box)) return self.fail("no box", .{});
770 const destination = self.level.neighbor(box, direction) orelse return self.fail("blocked", .{});
771 if (!self.level.interior[destination] or self.current.boxes.contains(destination)) return self.fail("blocked", .{});
772 const support = self.level.neighbor(box, opposite[direction]) orelse return self.fail("unreachable", .{});
773 const reach = self.level.reachable(self.current.player, self.current.boxes);
774 if (!reach[support]) return self.fail("unreachable", .{});
775 const path = try self.level.walkPath(self.scratch, self.current.player, support, self.current.boxes);
776 const sequence = try std.mem.concat(self.scratch, u8, &.{ path, a[2] });
777 const result = try self.takeSteps(sequence);
778 try printLine(self.out, "pushed {s}", .{result.letters});
779 },
780 .BRING => {
781 const source = self.cellArgument(a, 0);
782 const target = self.cellArgument(a, 2);
783 if (!self.current.boxes.contains(source)) return self.fail("no box", .{});
784 if (source == target) {
785 try self.out.writeAll("brought 0\n");
786 return;
787 }
788 var fixed: Boxes = .{};
789 for (self.current.boxes.slice()) |box| {
790 if (box == source) continue;
791 fixed.cells[fixed.len] = box;
792 fixed.len += 1;
793 }
794 const result = try pushSearch(self.scratch, &self.level, self.current, fixed, target, false, null);
795 if (result.outcome != .found) return self.fail("no route", .{});
796 _ = try self.takeSteps(result.notation);
797 try printLine(self.out, "brought {d} {s}", .{ pushCount(result.notation), result.notation });
798 },
799 .PATH => {
800 const target = self.cellArgument(a, 0);
801 const reach = self.level.reachable(self.current.player, self.current.boxes);
802 if (!reach[target]) return self.fail("unreachable", .{});
803 const path = try self.level.walkPath(self.scratch, self.current.player, target, self.current.boxes);
804 const result = try self.takeSteps(path);
805 try self.counted("path", result.letters);
806 },
807 .UNDO => {
808 if (self.history.items.len == 0) return self.fail("nothing to undo", .{});
809 const count = @min(number(a[0]), self.history.items.len);
810 for (0..count) |_| {
811 const letter = self.history.items[self.history.items.len - 1];
812 self.history.items.len -= 1;
813 const lower = std.ascii.toLower(letter);
814 const direction = std.mem.findScalar(u8, directions, lower).?;
815 const player = self.current.player;
816 if (std.ascii.isUpper(letter)) {
817 self.current.boxes = self.current.boxes.moved(self.level.neighbor(player, direction).?, player);
818 }
819 self.current.player = self.level.neighbor(player, opposite[direction]).?;
820 try self.redo_list.append(self.gpa, lower);
821 }
822 try printLine(self.out, "undone {d}", .{count});
823 },
824 .REDO => {
825 if (self.redo_list.items.len == 0) return self.fail("nothing to redo", .{});
826 const count = @min(number(a[0]), self.redo_list.items.len);
827 for (0..count) |_| {
828 const letter = self.redo_list.items[self.redo_list.items.len - 1];
829 self.redo_list.items.len -= 1;
830 const result = self.level.step(self.current, std.mem.findScalar(u8, directions, letter).?);
831 self.current = result.position;
832 try self.history.append(self.gpa, result.letter);
833 }
834 try printLine(self.out, "redone {d}", .{count});
835 },
836 .RESET => {
837 self.current = self.initial;
838 self.history.clearRetainingCapacity();
839 self.redo_list.clearRetainingCapacity();
840 try self.out.writeAll("reset\n");
841 },
842 .STATUS => {
843 try self.out.print("moves {d} pushes {d} ", .{ self.history.items.len, pushCount(self.history.items) });
844 try self.progress(self.current.boxes);
845 },
846 .HISTORY => try self.counted("history", self.history.items),
847 .BOXES => {
848 try printLine(self.out, "boxes {d}", .{self.current.boxes.len});
849 const reach = self.level.reachable(self.current.player, self.current.boxes);
850 for (self.current.boxes.slice()) |box| {
851 try self.out.print("box {s} ", .{try self.level.cellText(self.scratch, box)});
852 if (self.level.goalDistance(box)) |distance| try self.out.print("{d} ", .{distance}) else try self.out.writeAll("dead ");
853 var letters: [4]u8 = undefined;
854 var len: usize = 0;
855 for (0..4) |direction| {
856 if (self.level.pushable(box, direction, self.current.boxes, reach)) {
857 letters[len] = directions[direction];
858 len += 1;
859 }
860 }
861 try printLine(self.out, "{s}", .{if (len == 0) @as([]const u8, "-") else letters[0..len]});
862 }
863 },
864 .REACH => {
865 const reach = self.level.reachable(self.current.player, self.current.boxes);
866 try printLine(self.out, "reach {d} first {s}", .{ countFlags(reach), try self.level.cellText(self.scratch, firstFlag(reach)) });
867 },
868 .DEADSQUARES => {
869 try printLine(self.out, "dead {d}", .{countFlags(self.level.dead)});
870 try self.show(true, false);
871 },
872 .DEADLOCK => {
873 for (self.current.boxes.slice()) |box| {
874 if (self.level.dead[box]) {
875 try printLine(self.out, "deadlock dead {s}", .{try self.level.cellText(self.scratch, box)});
876 return;
877 }
878 }
879 for (self.current.boxes.slice()) |box| {
880 if (!self.level.goals[box] and self.level.frozen(box, self.current.boxes, @splat(false))) {
881 try printLine(self.out, "deadlock frozen {s}", .{try self.level.cellText(self.scratch, box)});
882 return;
883 }
884 }
885 try self.out.writeAll("no deadlock\n");
886 },
887 .BOUND => {
888 var best: Assignment = .{};
889 assignGoals(&self.level, self.current.boxes, 0, @splat(false), @splat(0), 0, &best);
890 const total = best.total orelse {
891 try self.out.writeAll("bound none\n");
892 return;
893 };
894 try printLine(self.out, "bound {d}", .{total});
895 for (self.current.boxes.slice(), 0..) |box, index| {
896 const goal = best.order[index];
897 try printLine(self.out, "box {s} goal {s} pushes {d}", .{
898 try self.level.cellText(self.scratch, box), try self.level.cellText(self.scratch, self.level.goal_cells.cells[goal]), self.level.distances[goal][box],
899 });
900 }
901 },
902 .SAVE => {
903 const entry = try self.snapshots.getOrPut(self.gpa, a[0]);
904 if (!entry.found_existing) entry.key_ptr.* = try self.gpa.dupe(u8, a[0]);
905 entry.value_ptr.* = .{ .position = self.current, .history = try self.gpa.dupe(u8, self.history.items) };
906 try printLine(self.out, "saved {s} moves {d}", .{ a[0], self.history.items.len });
907 },
908 .RESTORE => {
909 const snapshot = self.snapshots.get(a[0]) orelse return self.fail("no snapshot {s}", .{a[0]});
910 self.current = snapshot.position;
911 self.history.clearRetainingCapacity();
912 try self.history.appendSlice(self.gpa, snapshot.history);
913 self.redo_list.clearRetainingCapacity();
914 try printLine(self.out, "restored {s} moves {d}", .{ a[0], self.history.items.len });
915 },
916 .CHECK => {
917 const result = try self.simulate((try decode(self.scratch, a[0], step_characters)).?);
918 if (result.failure) |failure| {
919 try printLine(self.out, "check fail step {d} {s}", .{ result.failed_step, failure });
920 } else {
921 try self.out.print("check ok moves {d} pushes {d} ", .{ result.letters.len, pushCount(result.letters) });
922 try self.progress(result.position.boxes);
923 }
924 },
925 .SOLVE, .HINT => try self.solveOrHint(command, number(a[0])),
926 .PLAY => {
927 const solution = self.solution orelse return self.fail("no solution", .{});
928 if (!solution.position.same(self.current)) return self.fail("no solution", .{});
929 _ = try self.takeSteps(solution.notation);
930 try printLine(self.out, "played pushes {d} moves {d}", .{ pushCount(solution.notation), solution.notation.len });
931 },
932 .TRANSFORM => {
933 const kind = std.meta.stringToEnum(Transform, a[0]).?;
934 const old = self.level;
935 self.level = old.transformed(kind);
936 self.initial = old.mapPosition(kind, self.initial);
937 self.current = old.mapPosition(kind, self.current);
938 for (self.history.items) |*letter| letter.* = kind.turn(letter.*);
939 for (self.redo_list.items) |*letter| letter.* = kind.turn(letter.*);
940 var snapshots = self.snapshots.valueIterator();
941 while (snapshots.next()) |snapshot| {
942 snapshot.position = old.mapPosition(kind, snapshot.position);
943 snapshot.history = try kind.turnLetters(self.gpa, snapshot.history);
944 }
945 if (self.solution) |*solution| {
946 solution.position = old.mapPosition(kind, solution.position);
947 solution.notation = try kind.turnLetters(self.gpa, solution.notation);
948 }
949 try printLine(self.out, "transformed {d}x{d}", .{ self.level.height, self.level.width });
950 },
951 }
952 }

954 fn execute(self: *Game, words: []const []const u8, rows: []const []const u8) Error!void {
955 if (equal(words[0], "LOAD")) {
956 const width = number(words[2]);
957 for (rows, 0..) |row, index| {
958 if (row.len != width) return self.fail("bad row {d}", .{index + 1});
959 for (row) |symbol| if (!includes(level_characters, symbol)) return self.fail("bad row {d}", .{index + 1});
960 }
961 try self.load(rows);
962 try printLine(self.out, "loaded {d}x{d}", .{ rows.len, width });
963 return;
964 }
965 const command = std.meta.stringToEnum(Command, words[0]) orelse return self.fail("unknown command {s}", .{words[0]});
966 const arguments = words[1..];
967 const kinds = command.kinds();
968 if (arguments.len != kinds.len) return self.fail("bad arguments", .{});
969 for (kinds, arguments) |kind, text| if (!try validArgument(self.scratch, kind, text)) return self.fail("bad arguments", .{});
970 if (command != .IMPORT and command != .VALID) {
971 if (self.level.reason != null) return self.fail("invalid level", .{});
972 const cells: usize = if (command == .BRING) 2 else if (command == .PATH or command == .PUSH) 1 else 0;
973 for (0..cells) |index| {
974 const row = number(arguments[2 * index]);
975 const column = number(arguments[2 * index + 1]);
976 if (row < 1 or row > self.level.height or column < 1 or column > self.level.width) return self.fail("bad value", .{});
977 }
978 if ((command == .UNDO or command == .REDO) and number(arguments[0]) < 1) return self.fail("bad value", .{});
979 if ((command == .SOLVE or command == .HINT) and (number(arguments[0]) < 1 or number(arguments[0]) > 100000)) return self.fail("bad value", .{});
980 }
981 try self.process(command, arguments);
982 }

985fn splitWords(gpa: Allocator, line: []const u8) Allocator.Error![]const []const u8 {
986 var words: std.ArrayList([]const u8) = .empty;
987 var pieces = std.mem.splitScalar(u8, line, ' ');
988 while (pieces.next()) |piece| try words.append(gpa, piece);
989 return words.items;
990}

992pub fn main(init: std.process.Init) !void {
993 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
994 defer storage.deinit();
995 const gpa = storage.allocator();
996 var input_buffer: [4096]u8 = undefined;
997 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
998 var output_buffer: [4096]u8 = undefined;
999 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1000 const text = try input.interface.allocRemaining(gpa, .unlimited);
1001 var lines: std.ArrayList([]const u8) = .empty;
1002 var pieces = std.mem.splitScalar(u8, text, '\n');
1003 while (pieces.next()) |piece| try lines.append(gpa, piece);
1004 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1005 defer scratch.deinit();
1006 var game: Game = .{ .gpa = gpa, .scratch = scratch.allocator(), .out = &output.interface };
1007 const header = try splitWords(gpa, lines.items[0]);
1008 const height = number(header[0]);
1009 try game.load(lines.items[1 .. height + 1]);
1010 const count = number(lines.items[height + 1]);
1011 var index = height + 2;
1012 for (0..count) |_| {
1013 _ = scratch.reset(.retain_capacity);
1014 const words = try splitWords(game.scratch, lines.items[index]);
1015 index += 1;
1016 var rows: []const []const u8 = &.{};
1017 if (equal(words[0], "LOAD")) {
1018 const load_height = number(words[1]);
1019 rows = lines.items[index .. index + load_height];
1020 index += load_height;
1021 }
1022 game.execute(words, rows) catch |err| {
1023 if (err != error.Invalid) return err;
1024 try printLine(&output.interface, "error: {s}", .{game.problem});
1025 };
1026 }
1027 try output.interface.flush();
