157std::vector<std::string> split(const std::string& text, char separator) {
158 std::vector<std::string> parts;
159 std::size_t start = 0;
160 while (true) {
161 std::size_t end = text.find(separator, start);
162 if (end == std::string::npos) {
163 parts.push_back(text.substr(start));
164 return parts;
165 }
166 parts.push_back(text.substr(start, end - start));
167 start = end + 1;
168 }
169}

171std::optional<std::string> decode(const std::string& text, const std::string& allowed) {
172 std::string result;
173 std::size_t index = 0;
174 while (index < text.size()) {
175 std::size_t start = index;
176 while (index < text.size() && std::isdigit(static_cast<unsigned char>(text[index]))) {
177 ++index;
178 }
179 std::string count = text.substr(start, index - start);
180 if (count.size() > 2 || (!count.empty() && count[0] == '0') || index == text.size()) {
181 return std::nullopt;
182 }
183 if (allowed.find(text[index]) == std::string::npos) {
184 return std::nullopt;
185 }
186 result.append(count.empty() ? 1 : std::stoi(count), text[index]);
187 ++index;
188 }
189 return result;
190}

192std::string encode(const std::string& text) {
193 std::string result;
194 std::size_t start = 0;
195 while (start < text.size()) {
196 std::size_t end = start;
197 while (end < text.size() && text[end] == text[start]) {
198 ++end;
199 }
200 std::size_t run = end - start;
201 if (run > 1) {
202 result += std::to_string(run);
203 }
204 result += text[start];
205 start = end;
206 }
207 return result;
208}

214bool validArgument(Kind kind, const std::string& text) {
215 switch (kind) {
216 case Kind::Number:

218 case Kind::Steps:

220 case Kind::Direction:

222 case Kind::Name:
223 return std::regex_match(text, NAME_WORD);
224 case Kind::Transform:
225 return TURNS.count(text) > 0;
226 case Kind::Text:
227 break;
228 }
229 return true;
230}

259std::vector<bool> findInterior(int height, int width, const std::vector<bool>& walls, int start) {
260 std::vector<bool> interior(height * width, false);
261 interior[start] = true;
262 std::vector<int> stack = {start};
263 while (!stack.empty()) {
264 int cell = stack.back();
265 stack.pop_back();
266 int row = cell / width;
267 int column = cell % width;
268 for (const auto& [dRow, dColumn] : DELTAS) {
269 int nextRow = row + dRow;
270 int nextColumn = column + dColumn;
271 if (nextRow >= 0 && nextRow < height && nextColumn >= 0 && nextColumn < width) {
272 int neighbor = nextRow * width + nextColumn;
273 if (!walls[neighbor] && !interior[neighbor]) {
274 interior[neighbor] = true;
275 stack.push_back(neighbor);
276 }
277 }
278 }
279 }
280 return interior;
281}

334 std::optional<std::string> invalidity(int players, const Boxes& boxes) const {
335 if (players != 1) {
336 return "players " + std::to_string(players);
337 }
338 if (boxes.empty()) {
339 return "no boxes";
340 }
341 if (boxes.size() != goalCells.size()) {
342 return "boxes " + std::to_string(boxes.size()) + " goals " +
343 std::to_string(goalCells.size());
344 }
345 for (int cell = 0; cell < size; ++cell) {
346 if (interior[cell] && onBorder(cell)) {
347 return "open " + cellText(cell);
348 }
349 }
350 for (int cell = 0; cell < size; ++cell) {
351 if ((goals[cell] || includes(boxes, cell)) && !interior[cell]) {
352 return "outside " + cellText(cell);
353 }
354 }

358 void analyze() {
359 distances.clear();
360 for (int goal : goalCells) {
361 distances.push_back(pushDistances(goal));
362 }
363 for (int cell = 0; cell < size; ++cell) {
364 dead[cell] = interior[cell] && !goalDistance(cell);
365 }
366 }

368 std::vector<int> pushDistances(int goal) const {
369 std::vector<int> distance(size, UNDEFINED);
370 distance[goal] = 0;
371 std::queue<int> queue;
372 queue.push(goal);
373 while (!queue.empty()) {
374 int cell = queue.front();
375 queue.pop();
376 for (int offset : offsets) {
377 int box = cell - offset;
378 if (interior[box] && interior[box - offset] && distance[box] == UNDEFINED) {
379 distance[box] = distance[cell] + 1;
380 queue.push(box);
381 }
382 }
383 }
384 return distance;
385 }
386
387 std::optional<int> goalDistance(int cell) const {
388 std::optional<int> best;
389 for (const std::vector<int>& distance : distances) {
390 if (distance[cell] != UNDEFINED && (!best || distance[cell] < *best)) {
391 best = distance[cell];
392 }
393 }
394 return best;
395 }

406 std::vector<bool> reachable(int player, const Boxes& boxes) const {
407 std::vector<bool> seen(size, false);
408 seen[player] = true;
409 std::vector<int> stack = {player};
410 while (!stack.empty()) {
411 int cell = stack.back();
412 stack.pop_back();
413 for (int offset : offsets) {
414 int neighbor = cell + offset;
415 if (interior[neighbor] && !seen[neighbor] && !includes(boxes, neighbor)) {
416 seen[neighbor] = true;
417 stack.push_back(neighbor);
418 }
419 }
420 }

428 std::string walkPath(int start, int target, const Boxes& boxes) const {
429 std::vector<std::optional<std::pair<int, char>>> cameFrom(size);
430 std::queue<int> queue;
431 queue.push(start);
432 while (!queue.empty()) {
433 int cell = queue.front();
434 queue.pop();
435 for (char direction : DIRECTIONS) {
436 int neighbor = cell + offsetOf(direction);
437 if (interior[neighbor] && !includes(boxes, neighbor) && neighbor != start &&
438 !cameFrom[neighbor]) {
439 cameFrom[neighbor] = std::make_pair(cell, direction);
440 queue.push(neighbor);
441 }
442 }
443 }
444 std::string directions;
445 int cell = target;
446 while (cell != start) {
447 auto [previous, direction] = *cameFrom[cell];
448 directions += direction;
449 cell = previous;
450 }
451 std::reverse(directions.begin(), directions.end());
452 return directions;
453 }

455 std::pair<Position, char> step(const Position& position, char direction) const {
456 int offset = offsetOf(direction);
457 int target = position.player + offset;
458 if (walls[target]) {
459 throw StepFailure("wall");
460 }
461 if (!includes(position.boxes, target)) {
462 return {Position{target, position.boxes}, direction};
463 }
464 int beyond = target + offset;
465 if (walls[beyond] || includes(position.boxes, beyond)) {
466 throw StepFailure("blocked");
467 }
468 return {Position{target, moveBox(position.boxes, target, beyond)}, toUpper(direction)};
469 }

476 !includes(boxes, destination);

479 bool sideBlocked(int box, const std::array<int, 2>& sides, const Boxes& boxes,
480 const std::vector<int>& assumed) const {
481 if (std::any_of(sides.begin(), sides.end(),
482 [&](int side) { return walls[side] || includes(assumed, side); })) {
483 return true;
484 }
485 if (dead[sides[0]] && dead[sides[1]]) {
486 return true;
487 }
488 std::vector<int> extended = assumed;
489 extended.push_back(box);
490 return std::any_of(sides.begin(), sides.end(), [&](int side) {
491 return includes(boxes, side) && frozen(side, boxes, extended);
492 });
493 }

499 sideBlocked(box, vertical, boxes, assumed);

554 int mapCell(const std::string& kind, int cell) const {
555 int row = cell / width;
556 int column = cell % width;
557 int lastRow = height - 1;
558 int lastColumn = width - 1;
559 if (kind == "ROT90") {
560 return column * height + lastRow - row;
561 }
562 if (kind == "ROT180") {
563 return (lastRow - row) * width + lastColumn - column;
564 }
565 if (kind == "ROT270") {
566 return (lastColumn - column) * height + row;
567 }
568 if (kind == "FLIPH") {
569 return row * width + lastColumn - column;
570 }
571 return (lastRow - row) * width + column;
572 }

601std::pair<Level, Position> buildLevel(const std::vector<std::string>& rows) {
602 int height = static_cast<int>(rows.size());
603 int width = static_cast<int>(rows[0].size());
604 std::string cells;
605 for (const std::string& row : rows) {
606 cells += row;
607 }
608 std::vector<bool> walls;
609 std::vector<bool> goals;
610 std::vector<int> players;
611 Boxes boxes;
612 for (int cell = 0; cell < static_cast<int>(cells.size()); ++cell) {
613 char symbol = cells[cell];
614 walls.push_back(symbol == '#');
615 goals.push_back(symbol == '.' || symbol == '*' || symbol == '+');
616 if (symbol == '@' || symbol == '+') {
617 players.push_back(cell);
618 }
619 if (symbol == '$' || symbol == '*') {
620 boxes.push_back(cell);
621 }
622 }
623 std::vector<bool> interior = players.size() == 1
624 ? findInterior(height, width, walls, players[0])
625 : std::vector<bool>(cells.size(), false);
626 Level level(height, width, walls, goals, interior);
627 level.reason = level.invalidity(static_cast<int>(players.size()), boxes);
628 if (!level.reason) {
629 level.analyze();
630 }
631 int player = players.empty() ? -1 : players[0];
632 return {level, Position{player, boxes}};
633}

635std::vector<Push> solutionPushes(const std::vector<SearchNode>& nodes) {
636 std::vector<Push> pushes;
637 for (const SearchNode* node = &nodes.back(); node->parent; node = &nodes[*node->parent]) {

644std::string solutionNotation(const Level& level, Position position,
645 const std::vector<Push>& pushes) {
646 std::string notation;
647 for (const Push& push : pushes) {
648 int offset = level.offsetOf(push.direction);
649 notation += level.walkPath(position.player, push.box - offset, position.boxes);
650 notation += toUpper(push.direction);
651 position = Position{push.box, moveBox(position.boxes, push.box, push.box + offset)};
652 }
653 return notation;
654}

656SearchResult pushSearch(const Level& level, const Position& start, const Boxes& fixed,
657 std::optional<int> target, bool pruning,
658 std::optional<std::size_t> limit) {
659 int startRepresentative = level.representative(start.player, start.boxes);
660 std::vector<SearchNode> nodes = {{start.boxes, startRepresentative, std::nullopt, Push{}}};
661 std::set<std::pair<Boxes, int>> known = {{start.boxes, startRepresentative}};
662 std::size_t expanded = 0;
663 while (expanded < nodes.size()) {
664 if (limit && expanded == *limit) {
665 return {"limit", expanded, nodes.size(), ""};
666 }
667 SearchNode node = nodes[expanded];
668 ++expanded;
669 std::vector<bool> reachable = level.reachable(node.representative, node.boxes);
670 for (int box : node.boxes) {
671 if (includes(fixed, box)) {
672 continue;
673 }
674 for (char direction : DIRECTIONS) {
675 if (!level.pushable(box, direction, node.boxes, reachable)) {
676 continue;
677 }
678 int destination = box + level.offsetOf(direction);
679 Boxes boxes = moveBox(node.boxes, box, destination);
680 if (pruning && (level.dead[destination] || level.hasFrozenBox(boxes))) {
681 continue;
682 }
683 int representative = level.representative(box, boxes);
684 if (!known.insert({boxes, representative}).second) {
685 continue;
686 }
687 nodes.push_back({boxes, representative, expanded - 1, Push{box, direction}});
688 bool reached = target ? destination == *target : level.solved(boxes);
689 if (reached) {
690 std::string notation = solutionNotation(level, start, solutionPushes(nodes));
691 return {"found", expanded, nodes.size(), notation};
692 }
693 }
694 }
695 }
696 return {"no solution", expanded, nodes.size(), ""};
697}

751 Simulation simulate(const std::string& directions) const {
752 Position position = current;
753 std::string letters;
754 try {
755 for (char direction : directions) {
756 auto [next, letter] = level.step(position, direction);
757 position = next;
758 letters += letter;
759 }
760 } catch (const StepFailure& failure) {
761 std::string reason =
762 "step " + std::to_string(letters.size() + 1) + " " + failure.what();
763 return {position, letters, reason};
764 }
765 return {position, letters, std::nullopt};
766 }
767
768 Simulation takeSteps(const std::string& directions) {
769 Simulation simulation = simulate(directions);
770 current = simulation.position;
771 history += simulation.letters;
772 if (!simulation.letters.empty()) {
773 redoList.clear();
774 }
775 return simulation;
776 }

778 std::string progress(const Boxes& boxes) const {
779 int onGoals = level.boxesOnGoals(boxes);
780 std::string text = "goals " + std::to_string(onGoals) + "/" + std::to_string(boxes.size());
781 return onGoals == static_cast<int>(boxes.size()) ? text + " solved" : text;

784 std::pair<std::string, std::optional<std::string>> searchSolution(int limit) const {
785 if (level.solved(current.boxes)) {
786 return {"already solved", std::nullopt};
787 }
788 SearchResult result = pushSearch(level, current, {}, std::nullopt, true, limit);
789 std::string counts = "expanded " + std::to_string(result.expanded) + " states " +
790 std::to_string(result.states);
791 if (result.outcome != "found") {
792 return {result.outcome + " " + counts, std::nullopt};
793 }
794 const std::string& notation = result.notation;
795 std::string summary = "solution pushes " + std::to_string(pushCount(notation)) +
796 " moves " + std::to_string(notation.size()) + " " + counts;
797 return {summary, notation};
798 }

816 if (!row) {

875 void push(const Arguments& arguments) {
876 int box = cellArgument(arguments, 0);
877 char direction = arguments[2][0];
878 int offset = level.offsetOf(direction);
879 if (!includes(current.boxes, box)) {
880 throw CommandError("no box");
881 }
882 int destination = box + offset;
883 if (!level.interior[destination] || includes(current.boxes, destination)) {
884 throw CommandError("blocked");
885 }
886 if (!level.reachable(current.player, current.boxes)[box - offset]) {
887 throw CommandError("unreachable");
888 }
889 std::string path = level.walkPath(current.player, box - offset, current.boxes);
890 emit("pushed", takeSteps(path + direction).letters);
891 }
892
893 void bring(const Arguments& arguments) {
894 int source = cellArgument(arguments, 0);
895 int target = cellArgument(arguments, 2);
896 if (!includes(current.boxes, source)) {
897 throw CommandError("no box");
898 }
899 if (source == target) {
900 emit("brought", 0);
901 return;
902 }
903 Boxes fixed;
904 std::copy_if(current.boxes.begin(), current.boxes.end(), std::back_inserter(fixed),
905 [source](int box) { return box != source; });
906 SearchResult result = pushSearch(level, current, fixed, target, false, std::nullopt);
907 if (result.outcome != "found") {
908 throw CommandError("no route");
909 }
910 takeSteps(lowered(result.notation));
911 emit("brought", pushCount(result.notation), result.notation);
912 }

917 throw CommandError("unreachable");
918 }
919 Simulation simulation = takeSteps(level.walkPath(current.player, target, current.boxes));
920 emit(counted("path", simulation.letters));
921 }

923 void undo(const Arguments& arguments) {
924 if (history.empty()) {
925 throw CommandError("nothing to undo");
926 }
927 std::size_t count = std::min<std::size_t>(std::stoi(arguments[0]), history.size());
928 for (std::size_t index = 0; index < count; ++index) {
929 char letter = history.back();
930 history.pop_back();
931 char direction = toLower(letter);
932 int offset = level.offsetOf(direction);
933 int player = current.player;
934 Boxes boxes = current.boxes;
935 if (isPush(letter)) {
936 boxes = moveBox(boxes, player + offset, player);
937 }
938 current = Position{player - offset, boxes};
939 redoList += direction;
940 }
941 emit("undone", count);
942 }

944 void redo(const Arguments& arguments) {
945 if (redoList.empty()) {
946 throw CommandError("nothing to redo");
947 }
948 std::size_t count = std::min<std::size_t>(std::stoi(arguments[0]), redoList.size());
949 for (std::size_t index = 0; index < count; ++index) {
950 char direction = redoList.back();
951 redoList.pop_back();
952 auto [position, letter] = level.step(current, direction);
953 current = position;
954 history += letter;
955 }
956 emit("redone", count);
957 }

1002 void deadlock(const Arguments&) {

1007 }

1018 void bound(const Arguments&) {
1019 const Boxes& boxes = current.boxes;
1020 std::vector<int> order(level.goalCells.size());
1021 std::iota(order.begin(), order.end(), 0);
1022 std::optional<Assignment> best;
1023 do {
1024 std::vector<int> pushes;
1025 for (std::size_t index = 0; index < boxes.size(); ++index) {
1026 pushes.push_back(level.distances[order[index]][boxes[index]]);
1027 }
1028 if (includes(pushes, UNDEFINED)) {
1029 continue;
1030 }
1031 int total = std::accumulate(pushes.begin(), pushes.end(), 0);
1032 if (!best || total < best->total) {
1033 best = Assignment{order, pushes, total};
1034 }
1035 } while (std::next_permutation(order.begin(), order.end()));
1036 if (!best) {
1037 emit("bound", "none");
1038 return;
1039 }
1040 emit("bound", best->total);
1041 for (std::size_t index = 0; index < boxes.size(); ++index) {
1042 int goal = level.goalCells[best->order[index]];
1043 emit("box", level.cellText(boxes[index]), "goal", level.cellText(goal), "pushes",
1044 best->pushes[index]);
1045 }
1046 }

1109 Level old = level;
1110 level = old.transformed(kind);
1111 initial = old.mapPosition(kind, initial);
1112 current = old.mapPosition(kind, current);

1114 redoList = turnLetters(redoList, kind);
1115 for (auto& entry : snapshots) {

1117 snapshot = Snapshot{old.mapPosition(kind, snapshot.position),
1118 turnLetters(snapshot.history, kind)};

1121 solution = Solution{turnLetters(solution->notation, kind),
1122 old.mapPosition(kind, solution->position)};

1146 void runCommand(const std::string& name, const Arguments& arguments) {
1147 static const std::map<std::string, Handler> handlers = {
1148 {"IMPORT", &Sokoban::importLevel},
1149 {"VALID", &Sokoban::valid},
1150 {"INFO", &Sokoban::info},
1151 {"SHOW", &Sokoban::show},
1152 {"EXPORT", &Sokoban::exportLevel},
1153 {"MOVE", &Sokoban::move},
1154 {"PUSH", &Sokoban::push},
1155 {"BRING", &Sokoban::bring},
1156 {"PATH", &Sokoban::path},
1157 {"UNDO", &Sokoban::undo},
1158 {"REDO", &Sokoban::redo},
1159 {"RESET", &Sokoban::reset},
1160 {"STATUS", &Sokoban::status},
1161 {"HISTORY", &Sokoban::showHistory},
1162 {"BOXES", &Sokoban::showBoxes},
1163 {"REACH", &Sokoban::reach},
1164 {"DEADSQUARES", &Sokoban::deadSquares},
1165 {"DEADLOCK", &Sokoban::deadlock},
1166 {"BOUND", &Sokoban::bound},
1167 {"SAVE", &Sokoban::save},
1168 {"RESTORE", &Sokoban::restore},
1169 {"CHECK", &Sokoban::check},
1170 {"SOLVE", &Sokoban::solve},
1171 {"HINT", &Sokoban::hint},
1172 {"PLAY", &Sokoban::play},
1173 {"TRANSFORM", &Sokoban::transform},
1174 };
1175 auto command = COMMANDS.find(name);
1176 if (command == COMMANDS.end()) {
1177 throw CommandError("unknown command " + name);
1178 }
1179 const std::vector<Kind>& kinds = command->second;
1180 if (arguments.size() != kinds.size() ||
1181 !std::equal(kinds.begin(), kinds.end(), arguments.begin(), validArgument)) {
1182 throw CommandError("bad arguments");
1183 }
1184 if (LEVEL_FREE_COMMANDS.count(name) == 0) {
1185 if (level.reason) {
1186 throw CommandError("invalid level");
1187 }
1188 checkValues(name, arguments);
1189 }
1190 (this->*handlers.at(name))(arguments);
1191 }
