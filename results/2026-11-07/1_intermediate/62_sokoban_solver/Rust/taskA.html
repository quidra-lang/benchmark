18fn moved(boxes: &[i32], from: i32, to: i32) -> Vec<i32> {
19 let mut result: Vec<i32> = boxes.iter().map(|&b| if b == from {

22 b

28fn decode(word: &str, allowed: &[u8]) -> Option<Vec<u8>> {
29 let bytes = word.as_bytes();
30 let mut result = Vec::new();
31 let mut i = 0;
32 while i < bytes.len() {
33 let start = i;
34 let mut n = 0;
35 while i < bytes.len() && bytes[i].is_ascii_digit() {
36 if i - start < 2 {
37 n = n * 10 + (bytes[i] - b'0') as usize;
38 }
39 i += 1;
40 }
41 if i - start > 2 || (i > start && bytes[start] == b'0') || i == bytes.len() || !allowed.contains(&bytes[i]) {
42 return None;
43 }
44 result.extend(std::iter::repeat_n(bytes[i], if i == start {
45 1
46 } else {
47 n
48 }));
49 i += 1;
50 }
51 Some(result)
52}

54fn encode(row: &str) -> String {
55 let bytes = row.as_bytes();
56 let mut result = String::new();
57 let mut i = 0;
58 while i < bytes.len() {
59 let mut j = i + 1;
60 while j < bytes.len() && bytes[j] == bytes[i] {
61 j += 1;
62 }
63 if j - i > 1 {
64 result.push_str(&(j - i).to_string());
65 }
66 result.push(bytes[i] as char);
67 i = j;
68 }
69 result
70}

72fn turns(kind: &str) -> Option<&'static [u8; 4]> {
73 match kind {
74 "ROT90" => Some(b"rlud"),
75 "ROT180" => Some(b"durl"),
76 "ROT270" => Some(b"lrdu"),
77 "FLIPH" => Some(b"udrl"),
78 "FLIPV" => Some(b"dulr"),
79 _ => None
80 }

83fn turn(letters: &[u8], kind: &str) -> Vec<u8> {
84 letters.iter().map(|c| {
85 let d = DIRS.iter().position(|&d| d == c.to_ascii_lowercase()).unwrap();
86 let next = turns(kind).unwrap()[d];
87 if c.is_ascii_uppercase() {
88 next.to_ascii_uppercase()
89 } else {
90 next
91 }
92 }).collect()
93}

95fn kinds(command: &str) -> Option<&'static str> {
96 match command {
97 "IMPORT" => Some("t"),
98 "MOVE" | "CHECK" => Some("s"),
99 "PUSH" => Some("nnd"),
100 "BRING" => Some("nnnn"),
101 "PATH" => Some("nn"),
102 "UNDO" | "REDO" | "SOLVE" | "HINT" => Some("n"),
103 "SAVE" | "RESTORE" => Some("a"),
104 "TRANSFORM" => Some("f"),
105 "VALID" | "INFO" | "SHOW" | "EXPORT" | "RESET" | "STATUS" | "HISTORY"
106 | "BOXES" | "REACH" | "DEADSQUARES" | "DEADLOCK" | "BOUND" | "PLAY" => Some(""),
107 _ => None,
108 }

111fn argument(kind: u8, word: &str) -> bool {
112 let b = word.as_bytes();
113 match kind {
114 b'n' => !b.is_empty() && b.len() <= 6 && b.iter().all(u8::is_ascii_digit) && (b.len() == 1 || b[0] != b'0'),
115 b's' => decode(word, b"udlrUDLR").is_some(),
116 b'd' => b.len() == 1 && DIRS.contains(&b[0]),
117 b'a' => !b.is_empty() && b.len() <= 8 && b.iter().all(u8::is_ascii_lowercase),
118 b'f' => turns(word).is_some(),
119 _ => true,
120 }
121}

129#[derive(Clone)]
130struct Level {
131 h: usize,
132 w: usize,
133 walls: Vec<bool>,
134 goals: Vec<bool>,
135 inside: Vec<bool>,
136 dead: Vec<bool>,
137 goal_cells: Vec<i32>,
138 distances: Vec<Vec<i32>>,
139 reason: Option<String>,
140}
141
142impl Level {
143 fn size(&self) -> usize {
144 self.h * self.w
145 }
146
147 fn offsets(&self) -> [i32; 4] {
148 [-(self.w as i32), self.w as i32, -1, 1]
149 }
150
151 fn offset(&self, direction: u8) -> i32 {
152 self.offsets()[DIRS.iter().position(|&d| d == direction.to_ascii_lowercase()).unwrap()]
153 }
154
155 fn interior(&self, cell: i32) -> bool {
156 cell >= 0 && self.inside.get(cell as usize).copied().unwrap_or(false)
157 }
158
159 fn cell(&self, cell: i32) -> String {
160 format!("{} {}", cell as usize / self.w + 1, cell as usize % self.w + 1)
161 }
162
163 fn build(rows: &[String]) -> (Self, Position) {
164 let h = rows.len();
165 let w = rows[0].len();
166 let bytes: Vec<u8> = rows.iter().flat_map(|r| r.bytes()).collect();
167 let walls: Vec<bool> = bytes.iter().map(|&c| c == b'#').collect();
168 let goals: Vec<bool> = bytes.iter().map(|c| b".*+".contains(c)).collect();
169 let players: Vec<i32> = (0..bytes.len()).filter(|&i| b"@+".contains(&bytes[i])).map(|i| i as i32).collect();
170 let boxes: Vec<i32> = (0..bytes.len()).filter(|&i| b"$*".contains(&bytes[i])).map(|i| i as i32).collect();
171 let goal_cells = (0..bytes.len()).filter(|&i| goals[i]).map(|i| i as i32).collect();
172 let mut level = Self {
173 h,
174 w,
175 walls,
176 goals,
177 inside: vec![false; h * w],
178 dead: vec![false; h * w],
179 goal_cells,
180 distances: Vec::new(),
181 reason: None
182 };
183 if players.len() == 1 {
184 let mut queue = vec![players[0] as usize];
185 level.inside[queue[0]] = true;
186 let mut head = 0;
187 while head < queue.len() {
188 let cell = queue[head];
189 head += 1;
190 let r = cell / w;
191 let c = cell % w;
192 for (dr, dc) in [(-1, 0), (1, 0), (0, -1), (0, 1)] {
193 let nr = r as i32 + dr;
194 let nc = c as i32 + dc;
195 if nr >= 0 && nr < h as i32 && nc >= 0 && nc < w as i32 {
196 let p = nr as usize * w + nc as usize;
197 if !level.walls[p] && !level.inside[p] {
198 level.inside[p] = true;
199 queue.push(p);
200 }
201 }
202 }
203 }
204 }
205 level.reason = if players.len() != 1 {
206 Some(format!("players {}", players.len()))
207 } else if boxes.is_empty() {
208 Some("no boxes".into())
209 } else if boxes.len() != level.goal_cells.len() {
210 Some(format!("boxes {} goals {}", boxes.len(), level.goal_cells.len()))
211 } else if let Some(p) = (0..h * w).find(|&p| level.inside[p] && (p / w == 0 || p / w == h - 1 || p % w == 0 || p % w == w - 1)) {
212 Some(format!("open {}", level.cell(p as i32)))
213 } else if let Some(p) = (0..h * w).find(|&p| (level.goals[p] || boxes.contains(&(p as i32))) && !level.inside[p]) {
214 Some(format!("outside {}", level.cell(p as i32)))
215 } else {
216 None
217 };
218 if level.reason.is_none() {
219 level.analyze();
220 }
221 (level, Position {
222 player: players.first().copied().unwrap_or(0),
223 boxes
224 })
225 }
226
227 fn analyze(&mut self) {
228 self.goal_cells = (0..self.size()).filter(|&p| self.goals[p]).map(|p| p as i32).collect();
229 self.distances = self.goal_cells.iter().map(|&goal| {
230 let mut dist = vec![-1; self.size()];
231 dist[goal as usize] = 0;
232 let mut queue = vec![goal];
233 let mut head = 0;
234 while head < queue.len() {
235 let q = queue[head];
236 head += 1;
237 for delta in self.offsets() {
238 let b = q - delta;
239 if self.interior(b) && self.interior(b - delta) && dist[b as usize] < 0 {
240 dist[b as usize] = dist[q as usize] + 1;
241 queue.push(b);
242 }
243 }
244 }
245 dist
246 }).collect();
247 self.dead = (0..self.size()).map(|p| self.inside[p] && self.distances.iter().all(|d| d[p] < 0)).collect();
248 }
249
250 fn reach(&self, player: i32, boxes: &[i32]) -> Vec<bool> {
251 let mut seen = vec![false; self.size()];
252 seen[player as usize] = true;
253 let mut queue = vec![player];
254 let mut head = 0;
255 while head < queue.len() {
256 let q = queue[head];
257 head += 1;
258 for delta in self.offsets() {
259 let p = q + delta;
260 if self.interior(p) && !seen[p as usize] && !boxes.contains(&p) {
261 seen[p as usize] = true;
262 queue.push(p);
263 }
264 }
265 }
266 seen
267 }
268
269 fn representative(&self, player: i32, boxes: &[i32]) -> i32 {
270 self.reach(player, boxes).iter().position(|&v| v).unwrap() as i32
271 }
272
273 fn path(&self, player: i32, target: i32, boxes: &[i32]) -> Vec<u8> {
274 let mut from = vec![None; self.size()];
275 let mut queue = vec![player];
276 let mut head = 0;
277 while head < queue.len() {
278 let q = queue[head];
279 head += 1;
280 for (d, delta) in self.offsets().iter().enumerate() {
281 let p = q + delta;
282 if self.interior(p) && p != player && !boxes.contains(&p) && from[p as usize].is_none() {
283 from[p as usize] = Some((q, DIRS[d]));
284 queue.push(p);
285 }
286 }
287 }
288 let mut result = Vec::new();
289 let mut p = target;
290 while p != player {
291 let (q, d) = from[p as usize].unwrap();
292 result.push(d);
293 p = q;
294 }
295 result.reverse();
296 result
297 }

299 fn step(&self, position: &Position, direction: u8) -> Result<(Position, u8), &'static str> {
300 let d = direction.to_ascii_lowercase();
301 let delta = self.offset(d);
302 let p = position.player + delta;
303 if self.walls[p as usize] {
304 return Err("wall");
305 }
306 let mut next = Position {
307 player: p,
308 boxes: position.boxes.clone()
309 };
310 if position.boxes.contains(&p) {
311 let dest = p + delta;
312 if self.walls[dest as usize] || position.boxes.contains(&dest) {
313 return Err("blocked");
314 }
315 next.boxes = moved(&next.boxes, p, dest);
316 Ok((next, d.to_ascii_uppercase()))
317 } else {
318 Ok((next, d))
319 }
320 }
321
322 fn pushable(&self, b: i32, d: usize, boxes: &[i32], reach: &[bool]) -> bool {
323 let delta = self.offsets()[d];
324 reach[(b - delta) as usize] && self.interior(b + delta) && !boxes.contains(&(b + delta))
325 }
326
327 fn frozen(&self, b: i32, boxes: &[i32], assumed: &[i32]) -> bool {
328 let axis = |sides: [i32;2]| {
329 if sides.iter().any(|&p| self.walls[p as usize] || assumed.contains(&p)) || sides.iter().all(|&p| self.dead[p as usize]) {
330 return true;
331 }
332 let mut extended = assumed.to_vec();
333 extended.push(b);
334 sides.iter().any(|&p| boxes.contains(&p) && self.frozen(p, boxes, &extended))
335 };
336 axis([b - 1, b + 1]) && axis([b - self.w as i32, b + self.w as i32])
337 }
338
339 fn on_goals(&self, boxes: &[i32]) -> usize {
340 boxes.iter().filter(|&&b| self.goals[b as usize]).count()
341 }
342
343 fn solved(&self, boxes: &[i32]) -> bool {
344 self.on_goals(boxes) == boxes.len()
345 }
346
347 fn render(&self, pos: &Position, dead_map: bool) -> Vec<String> {
348 let cells: Vec<u8> = (0..self.size()).map(|p| {
349 if self.walls[p] {
350 b'#'
351 } else if dead_map {
352 if self.goals[p] {
353 b'.'
354 } else if self.dead[p] {
355 b'x'
356 } else {
357 b'-'
358 }
359 } else if pos.player == p as i32 {
360 if self.goals[p] {
361 b'+'
362 } else {
363 b'@'
364 }
365 } else if pos.boxes.contains(&(p as i32)) {
366 if self.goals[p] {
367 b'*'
368 } else {
369 b'$'
370 }
371 } else if self.goals[p] {
372 b'.'
373 } else {
374 b'-'
375 }
376 }).collect();
377 cells.chunks(self.w).map(text).collect()
378 }
379
380 fn map_cell(&self, p: i32, kind: &str) -> i32 {
381 let r = p as usize / self.w;
382 let c = p as usize % self.w;
383 (match kind {
384 "ROT90" => c * self.h + self.h - 1 - r,
385 "ROT180" => (self.h - 1 - r) * self.w + self.w - 1 - c,
386 "ROT270" => (self.w - 1 - c) * self.h + r,
387 "FLIPH" => r * self.w + self.w - 1 - c,
388 _ => (self.h - 1 - r) * self.w + c
389 }) as i32
390 }
391
392 fn map_position(&self, p: &Position, kind: &str) -> Position {
393 let mut boxes: Vec<i32> = p.boxes.iter().map(|&b| self.map_cell(b, kind)).collect();
394 boxes.sort_unstable();
395 Position {
396 player: self.map_cell(p.player, kind),
397 boxes
398 }
399 }
400
401 fn transformed(&self, kind: &str) -> Self {
402 let mut result = self.clone();
403 if kind == "ROT90" || kind == "ROT270" {
404 result.h = self.w;
405 result.w = self.h;
406 }
407 for p in 0..self.size() {
408 let q = self.map_cell(p as i32, kind) as usize;
409 result.walls[q] = self.walls[p];
410 result.goals[q] = self.goals[p];
411 result.inside[q] = self.inside[p];
412 }
413 result.analyze();
414 result
415 }
416}

430struct Search {
431 outcome: &'static str,
432 expanded: usize,
433 states: usize,
434 notation: Vec<u8>
435}

437fn search(level: &Level, start: &Position, fixed: &[i32], target: Option<i32>, pruning: bool, limit: Option<usize>) -> Search {
438 let first = State {
439 boxes: start.boxes.clone(),
440 representative: level.representative(start.player, &start.boxes)
441 };
442 let mut known = HashSet::from([first.clone()]);
443 let mut nodes = vec![Node {
444 state: first,
445 parent: None,
446 push: (0, 0)
447 }];
448 let mut expanded = 0;
449 while expanded < nodes.len() {
450 if Some(expanded) == limit {
451 return Search {
452 outcome: "limit",
453 expanded,
454 states: nodes.len(),
455 notation: Vec::new()
456 };
457 }
458 let parent = expanded;
459 let state = nodes[parent].state.clone();
460 expanded += 1;
461 let reach = level.reach(state.representative, &state.boxes);
462 for &b in &state.boxes {
463 if fixed.contains(&b) {
464 continue;
465 }
466 for d in 0..4 {
467 if !level.pushable(b, d, &state.boxes, &reach) {
468 continue;
469 }
470 let dest = b + level.offsets()[d];
471 let boxes = moved(&state.boxes, b, dest);
472 if pruning && (level.dead[dest as usize] || boxes.iter().any(|&p| !level.goals[p as usize] && level.frozen(p, &boxes, &[]))) {
473 continue;
474 }
475 let next = State {
476 representative: level.representative(b, &boxes),
477 boxes
478 };
479 if !known.insert(next.clone()) {
480 continue;
481 }
482 let found = match target {
483 Some(t) => dest == t,
484 None => level.solved(&next.boxes)
485 };
486 nodes.push(Node {
487 state: next,
488 parent: Some(parent),
489 push: (b, d)
490 });
491 if found {
492 let mut steps = Vec::new();
493 let mut i = nodes.len() - 1;
494 while let Some(p) = nodes[i].parent {
495 steps.push(nodes[i].push);
496 i = p;
497 }
498 steps.reverse();
499 let mut position = start.clone();
500 let mut notation = Vec::new();
501 for (b, d) in steps {
502 let delta = level.offsets()[d];
503 notation.extend(level.path(position.player, b - delta, &position.boxes));
504 notation.push(DIRS[d].to_ascii_uppercase());
505 position = Position {
506 player: b,
507 boxes: moved(&position.boxes, b, b + delta)
508 };
509 }
510 return Search {
511 outcome: "found",
512 expanded,
513 states: nodes.len(),
514 notation
515 };
516 }
517 }
518 }
519 }
520 Search {
521 outcome: "no solution",
522 expanded,
523 states: nodes.len(),
524 notation: Vec::new()
525 }
526}

590 fn simulate(&self, directions: &[u8]) -> (Position, Vec<u8>, Option<String>) {
591 let mut position = self.current.clone();
592 let mut letters = Vec::new();
593 for &d in directions {
594 match self.level.step(&position, d) {
595 Ok((next, c)) => {
596 position = next;
597 letters.push(c);
598 },
599 Err(reason) => {
600 let error = format!("step {} {}", letters.len() + 1, reason);
601 return (position, letters, Some(error));
602 }
603 }
604 }
605 (position, letters, None)
606 }
607
608 fn take(&mut self, directions: &[u8]) -> (Vec<u8>, Option<String>) {
609 let (position, letters, failure) = self.simulate(directions);
610 self.current = position;
611 self.history.extend(&letters);
612 if !letters.is_empty() {
613 self.redo.clear();
614 }
615 (letters, failure)
616 }

618 fn cell_argument(&self, a: &[&str], i: usize) -> i32 {
619 (a[i].parse::<i32>().unwrap() - 1) * self.level.w as i32 + a[i + 1].parse::<i32>().unwrap() - 1
620 }
621
622 fn execute(&mut self, words: &[&str], rows: &[String]) -> Result<(), String> {
623 let name = words[0];
624 let a = &words[1..];
625 if name == "LOAD" {
626 let w = a[1].parse::<usize>().unwrap();
627 if let Some(i) = rows.iter().position(|r| r.len() != w || !r.bytes().all(|c| LEVEL.contains(&c))) {
628 return Err(format!("bad row {}", i + 1));
629 }
630 self.load(rows);
631 self.emit(format!("loaded {}x{}", rows.len(), w));
632 return Ok(());
633 }
634 let schema = kinds(name).ok_or_else(|| format!("unknown command {}", name))?;
635 if a.len() != schema.len() || !schema.bytes().zip(a.iter()).all(|(k, w)| argument(k, w)) {
636 return Err("bad arguments".into());
637 }
638 if name == "IMPORT" {
639 let rows: Option<Vec<Vec<u8>>> = a[0].split('|').map(|s| decode(s, LEVEL)).collect();
640 let rows = rows.ok_or("bad level")?;
641 if rows.len() > 10 || rows[0].is_empty() || rows[0].len() > 10 || rows.iter().any(|r| r.len() != rows[0].len()) {
642 return Err("bad level".into());
643 }
644 let rows: Vec<String> = rows.iter().map(|r| text(r)).collect();
645 self.load(&rows);
646 self.emit(format!("loaded {}x{}", self.level.h, self.level.w));
647 return Ok(());
648 }
649 if name == "VALID" {
650 self.emit(self.level.reason.as_ref().map_or("valid".into(), |r| format!("invalid {}", r)));
651 return Ok(());
652 }
653 if self.level.reason.is_some() {
654 return Err("invalid level".into());
655 }
656 let cells = match name {
657 "PATH" | "PUSH" => 1,
658 "BRING" => 2,
659 _ => 0
660 };
661 for i in 0..cells {
662 let r = a[2 * i].parse::<usize>().unwrap();
663 let c = a[2 * i + 1].parse::<usize>().unwrap();
664 if r < 1 || r > self.level.h || c < 1 || c > self.level.w {
665 return Err("bad value".into());
666 }
667 }
668 if matches!(name, "UNDO" | "REDO" | "SOLVE" | "HINT") {
669 let n = a[0].parse::<usize>().unwrap();
670 if n < 1 || (matches!(name, "SOLVE" | "HINT") && n > 100000) {
671 return Err("bad value".into());
672 }
673 }
674 match name {
675 "INFO" => self.emit(format!(
676 "size {}x{} walls {} cells {} boxes {} dead {}",
677 self.level.h, self.level.w, count(&self.level.walls), count(&self.level.inside), self.current.boxes.len(), count(&self.level.dead))),
678 "SHOW" => {
679 self.output.extend(self.level.render(&self.current, false));
680 },
681 "EXPORT" => self.emit(format!(
682 "level {}",
683 self.level.render(&self.current, false).iter().map(|r| encode(r)).collect::<Vec<_>>().join("|"))),
684 "MOVE" => {
685 let steps = decode(a[0], b"udlrUDLR").unwrap();
686 let (letters, failure) = self.take(&steps);
687 if !letters.is_empty() {
688 self.emit(format!("moved {}", text(&letters)));
689 }
690 if let Some(error) = failure {
691 return Err(error);
692 }
693 },
694 "PUSH" => {
695 let b = self.cell_argument(a, 0);
696 let d = a[2].as_bytes()[0];
697 let delta = self.level.offset(d);
698 if !self.current.boxes.contains(&b) {
699 return Err("no box".into());
700 }
701 if !self.level.interior(b + delta) || self.current.boxes.contains(&(b + delta)) {
702 return Err("blocked".into());
703 }
704 if !self.level.reach(self.current.player, &self.current.boxes)[(b - delta) as usize] {
705 return Err("unreachable".into());
706 }
707 let mut path = self.level.path(self.current.player, b - delta, &self.current.boxes);
708 path.push(d);
709 let (letters, _) = self.take(&path);
710 self.emit(format!("pushed {}", text(&letters)));
711 },
712 "BRING" => {
713 let b = self.cell_argument(a, 0);
714 let target = self.cell_argument(a, 2);
715 if !self.current.boxes.contains(&b) {
716 return Err("no box".into());
717 }
718 if b == target {
719 self.emit("brought 0".into());
720 } else {
721 let fixed: Vec<i32> = self.current.boxes.iter().copied().filter(|&p| p != b).collect();
722 let result = search(&self.level, &self.current, &fixed, Some(target), false, None);
723 if result.outcome != "found" {
724 return Err("no route".into());
725 }
726 self.take(&result.notation);
727 self.emit(format!("brought {} {}", pushes(&result.notation), text(&result.notation)));
728 }
729 },
730 "PATH" => {
731 let target = self.cell_argument(a, 0);
732 if !self.level.reach(self.current.player, &self.current.boxes)[target as usize] {
733 return Err("unreachable".into());
734 }
735 let path = self.level.path(self.current.player, target, &self.current.boxes);
736 let (letters, _) = self.take(&path);
737 self.emit(if letters.is_empty() {
738 "path 0".into()
739 } else {
740 format!("path {} {}", letters.len(), text(&letters))
741 });
742 },
743 "UNDO" => {
744 if self.history.is_empty() {
745 return Err("nothing to undo".into());
746 }
747 let n = a[0].parse::<usize>().unwrap().min(self.history.len());
748 for _ in 0..n {
749 let letter = self.history.pop().unwrap();
750 let d = letter.to_ascii_lowercase();
751 let delta = self.level.offset(d);
752 let p = self.current.player;
753 if letter.is_ascii_uppercase() {
754 self.current.boxes = moved(&self.current.boxes, p + delta, p);
755 }
756 self.current.player = p - delta;
757 self.redo.push(d);
758 }
759 self.emit(format!("undone {}", n));
760 },
761 "REDO" => {
762 if self.redo.is_empty() {
763 return Err("nothing to redo".into());
764 }
765 let n = a[0].parse::<usize>().unwrap().min(self.redo.len());
766 for _ in 0..n {
767 let d = self.redo.pop().unwrap();
768 let (p, c) = self.level.step(&self.current, d).unwrap();
769 self.current = p;
770 self.history.push(c);
771 }
772 self.emit(format!("redone {}", n));
773 },
774 "RESET" => {
775 self.current = self.initial.clone();
776 self.history.clear();
777 self.redo.clear();
778 self.emit("reset".into());
779 },
780 "STATUS" => self.emit(format!("moves {} pushes {} {}", self.history.len(), pushes(&self.history), self.progress(&self.current.boxes))),
781 "HISTORY" => self.emit(if self.history.is_empty() {
782 "history 0".into()
783 } else {
784 format!("history {} {}", self.history.len(), text(&self.history))
785 }),
786 "BOXES" => {
787 self.emit(format!("boxes {}", self.current.boxes.len()));
788 let reach = self.level.reach(self.current.player, &self.current.boxes);
789 for b in self.current.boxes.clone() {
790 let distance = self.level.distances.iter().map(|d| d[b as usize]).filter(|&d| d >= 0).min().map_or("dead".into(), |d| d.to_string());
791 let dirs: Vec<u8> = (0..4).filter(|&d| self.level.pushable(b, d, &self.current.boxes, &reach)).map(|d| DIRS[d]).collect();
792 self.emit(format!("box {} {} {}", self.level.cell(b), distance, if dirs.is_empty() {
793 "-".into()
794 } else {
795 text(&dirs)
796 }));
797 }
798 },
799 "REACH" => {
800 let reach = self.level.reach(self.current.player, &self.current.boxes);
801 self.emit(format!("reach {} first {}", count(&reach), self.level.cell(reach.iter().position(|&v| v).unwrap() as i32)));
802 },
803 "DEADSQUARES" => {
804 self.emit(format!("dead {}", count(&self.level.dead)));
805 self.output.extend(self.level.render(&self.current, true));
806 },
807 "DEADLOCK" => {
808 let dead = self.current.boxes.iter().copied().find(|&b| self.level.dead[b as usize]);
809 let frozen = self.current.boxes.iter().copied().find(|&b| !self.level.goals[b as usize] && self.level.frozen(b, &self.current.boxes, &[]));
810 self.emit(if let Some(b) = dead {
811 format!("deadlock dead {}", self.level.cell(b))
812 } else if let Some(b) = frozen {
813 format!("deadlock frozen {}", self.level.cell(b))
814 } else {
815 "no deadlock".into()
816 });
817 },
818 "BOUND" => {
819 let mut best = None;
820 self.assignment(&mut Vec::new(), 0, 0, &mut best);
821 if let Some((total, order)) = best {
822 self.emit(format!("bound {}", total));
823 for (i, g) in order.iter().enumerate() {
824 let b = self.current.boxes[i];
825 self.emit(format!("box {} goal {} pushes {}", self.level.cell(b), self.level.cell(self.level.goal_cells[*g]), self.level.distances[*g][b as usize]));
826 }
827 } else {
828 self.emit("bound none".into());
829 }
830 },
831 "SAVE" => {
832 self.snapshots.insert(a[0].into(), Snapshot {
833 position: self.current.clone(),
834 history: self.history.clone()
835 });
836 self.emit(format!("saved {} moves {}", a[0], self.history.len()));
837 },
838 "RESTORE" => {
839 let snapshot = self.snapshots.get(a[0]).ok_or_else(|| format!("no snapshot {}", a[0]))?.clone();
840 self.current = snapshot.position;
841 self.history = snapshot.history;
842 self.redo.clear();
843 self.emit(format!("restored {} moves {}", a[0], self.history.len()));
844 },
845 "CHECK" => {
846 let (p, letters, failure) = self.simulate(&decode(a[0], b"udlrUDLR").unwrap());
847 self.emit(if let Some(error) = failure {
848 format!("check fail {}", error)
849 } else {
850 format!("check ok moves {} pushes {} {}", letters.len(), pushes(&letters), self.progress(&p.boxes))
851 });
852 },
853 "SOLVE" | "HINT" => {
854 let (summary, notation) = self.search_solution(a[0].parse().unwrap());
855 if name == "SOLVE" {
856 self.solution = notation.as_ref().map(|n| Solution {
857 position: self.current.clone(),
858 notation: n.clone()
859 });
860 self.emit(summary);
861 if let Some(n) = notation {
862 self.emit(text(&n));
863 }
864 } else if let Some(n) = notation {
865 let end = n.iter().position(u8::is_ascii_uppercase).unwrap() + 1;
866 self.take(&n[..end]);
867 self.emit(format!("hint {}", text(&n[..end])));
868 } else {
869 self.emit(summary);
870 }
871 },
872 "PLAY" => {
873 let solution = self.solution.as_ref().filter(|s| s.position == self.current).ok_or("no solution")?.clone();
874 self.take(&solution.notation);
875 self.emit(format!("played pushes {} moves {}", pushes(&solution.notation), solution.notation.len()));
876 },
877 "TRANSFORM" => {
878 let kind = a[0];
879 let old = self.level.clone();
880 self.level = old.transformed(kind);
881 self.initial = old.map_position(&self.initial, kind);
882 self.current = old.map_position(&self.current, kind);
883 self.history = turn(&self.history, kind);
884 self.redo = turn(&self.redo, kind);
885 for snapshot in self.snapshots.values_mut() {
886 snapshot.position = old.map_position(&snapshot.position, kind);
887 snapshot.history = turn(&snapshot.history, kind);
888 }
889 if let Some(s) = &mut self.solution {
890 s.position = old.map_position(&s.position, kind);
891 s.notation = turn(&s.notation, kind);
892 }
893 self.emit(format!("transformed {}x{}", self.level.h, self.level.w));
894 },
895 _ => unreachable!(),
896 }
897 Ok(())
898 }
899
900 fn assignment(&self, order: &mut Vec<usize>, used: u32, total: i32, best: &mut Option<(i32, Vec<usize>)>) {
901 let i = order.len();
902 if i == self.current.boxes.len() {
903 if best.as_ref().is_none_or(|(cost, _)|total < *cost) {
904 *best = Some((total, order.clone()));
905 }
906 return;
907 }
908 for g in 0..self.level.goal_cells.len() {
909 let d = self.level.distances[g][self.current.boxes[i] as usize];
910 if used & (1 << g) == 0 && d >= 0 {
911 order.push(g);
912 self.assignment(order, used | (1 << g), total + d, best);
913 order.pop();
914 }
915 }
916 }
917
918 fn search_solution(&self, limit: usize) -> (String, Option<Vec<u8>>) {
919 if self.level.solved(&self.current.boxes) {
920 return ("already solved".into(), None);
921 }
922 let result = search(&self.level, &self.current, &[], None, true, Some(limit));
923 let counts = format!("expanded {} states {}", result.expanded, result.states);
924 if result.outcome != "found" {
925 return (format!("{} {}", result.outcome, counts), None);
926 }
927 (format!("solution pushes {} moves {} {}", pushes(&result.notation), result.notation.len(), counts), Some(result.notation))
928 }
