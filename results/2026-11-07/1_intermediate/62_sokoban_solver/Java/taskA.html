118 static Optional<String> decode(String text, String allowed) {
119 StringBuilder result = new StringBuilder();
120 int index = 0;
121 while (index < text.length()) {
122 int start = index;
123 while (index < text.length() && Character.isDigit(text.charAt(index))) {
124 index++;
125 }
126 String count = text.substring(start, index);
127 if (count.length() > 2 || count.startsWith("0") || index == text.length()) {
128 return Optional.empty();
129 }
130 char symbol = text.charAt(index);
131 if (allowed.indexOf(symbol) < 0) {
132 return Optional.empty();
133 }
134 result.repeat(symbol, count.isEmpty() ? 1 : Integer.parseInt(count));
135 index++;
136 }
137 return Optional.of(result.toString());
138 }

140 static String encode(String text) {
141 StringBuilder result = new StringBuilder();
142 int start = 0;
143 while (start < text.length()) {
144 int end = start;
145 while (end < text.length() && text.charAt(end) == text.charAt(start)) {
146 end++;
147 }
148 int run = end - start;
149 if (run > 1) {
150 result.append(run);
151 }
152 result.append(text.charAt(start));
153 start = end;
154 }
155 return result.toString();
156 }

215 static List<List<Integer>> permutations(List<Integer> items) {
216 if (items.isEmpty()) {
217 return List.of(List.of());
218 }
219 List<List<Integer>> result = new ArrayList<>();
220 for (Integer item : items) {
221 List<Integer> rest = new ArrayList<>(items);
222 rest.remove(item);
223 for (List<Integer> tail : permutations(rest)) {
224 List<Integer> permutation = new ArrayList<>(List.of(item));
225 permutation.addAll(tail);
226 result.add(permutation);
227 }
228 }
229 return result;
230 }

232 static boolean[] findInterior(int height, int width, boolean[] walls, int start) {
233 boolean[] interior = new boolean[height * width];
234 interior[start] = true;
235 Deque<Integer> stack = new ArrayDeque<>();
236 stack.push(start);
237 while (!stack.isEmpty()) {
238 int cell = stack.pop();
239 int row = cell / width;
240 int column = cell % width;
241 for (int[] delta : DELTAS) {
242 int nextRow = row + delta[0];
243 int nextColumn = column + delta[1];
244 if (nextRow >= 0 && nextRow < height && nextColumn >= 0 && nextColumn < width) {
245 int neighbor = nextRow * width + nextColumn;
246 if (!walls[neighbor] && !interior[neighbor]) {
247 interior[neighbor] = true;
248 stack.push(neighbor);
249 }
250 }
251 }
252 }
253 return interior;
254 }

329 distances = goalCells.stream().map(this::pushDistances).toList();

335 int[] pushDistances(int goal) {
336 int[] distance = new int[size];
337 Arrays.fill(distance, UNDEFINED);
338 distance[goal] = 0;
339 Deque<Integer> queue = new ArrayDeque<>();
340 queue.add(goal);
341 while (!queue.isEmpty()) {
342 int cell = queue.poll();
343 for (int offset : offsets) {
344 int box = cell - offset;
345 if (interior[box] && interior[box - offset] && distance[box] == UNDEFINED) {
346 distance[box] = distance[cell] + 1;
347 queue.add(box);
348 }
349 }
350 }
351 return distance;
352 }

354 OptionalInt goalDistance(int cell) {
355 return distances.stream()
356 .mapToInt(distance -> distance[cell])
357 .filter(distance -> distance != UNDEFINED)
358 .min();
359 }

391 String walkPath(int start, int target, List<Integer> boxes) {
392 CameFrom[] cameFrom = new CameFrom[size];
393 Deque<Integer> queue = new ArrayDeque<>();
394 queue.add(start);
395 while (!queue.isEmpty()) {
396 int cell = queue.poll();
397 for (char direction : DIRECTIONS.toCharArray()) {
398 int neighbor = cell + offsetOf(direction);
399 if (interior[neighbor] && !boxes.contains(neighbor) && neighbor != start
400 && cameFrom[neighbor] == null) {
401 cameFrom[neighbor] = new CameFrom(cell, direction);
402 queue.add(neighbor);
403 }
404 }
405 }
406 StringBuilder directions = new StringBuilder();
407 for (int cell = target; cell != start; cell = cameFrom[cell].cell()) {
408 directions.append(cameFrom[cell].direction());
409 }
410 return directions.reverse().toString();
411 }

413 Step step(Position position, char direction) {
414 int offset = offsetOf(direction);
415 int target = position.player() + offset;
416 if (walls[target]) {
417 throw new StepException("wall");
418 }
419 if (!position.boxes().contains(target)) {
420 return new Step(new Position(target, position.boxes()), direction);
421 }
422 int beyond = target + offset;
423 if (walls[beyond] || position.boxes().contains(beyond)) {
424 throw new StepException("blocked");
425 }
426 List<Integer> boxes = moveBox(position.boxes(), target, beyond);
427 return new Step(new Position(target, boxes), Character.toUpperCase(direction));
428 }

430 boolean pushable(int box, char direction, List<Integer> boxes, boolean[] reachable) {
431 int offset = offsetOf(direction);
432 int destination = box + offset;
433 return reachable[box - offset] && interior[destination] && !boxes.contains(destination);
434 }

436 boolean sideBlocked(int box, int[] sides, List<Integer> boxes, List<Integer> assumed) {
437 if (Arrays.stream(sides).anyMatch(side -> walls[side] || assumed.contains(side))) {
438 return true;
439 }
440 if (Arrays.stream(sides).allMatch(side -> dead[side])) {
441 return true;
442 }
443 List<Integer> extended = new ArrayList<>(assumed);
444 extended.add(box);
445 return Arrays.stream(sides)
446 .anyMatch(side -> boxes.contains(side) && frozen(side, boxes, extended));
447 }

450 int[] horizontal = {box + offsetOf('l'), box + offsetOf('r')};
451 int[] vertical = {box + offsetOf('u'), box + offsetOf('d')};
452 return sideBlocked(box, horizontal, boxes, assumed)
453 && sideBlocked(box, vertical, boxes, assumed);
454 }

507 int mapCell(String kind, int cell) {
508 int row = cell / width;
509 int column = cell % width;
510 int lastRow = height - 1;
511 int lastColumn = width - 1;
512 return switch (kind) {
513 case "ROT90" -> column * height + lastRow - row;
514 case "ROT180" -> (lastRow - row) * width + lastColumn - column;
515 case "ROT270" -> (lastColumn - column) * height + row;
516 case "FLIPH" -> row * width + lastColumn - column;
517 default -> (lastRow - row) * width + column;
518 };
519 }

547 static LoadedLevel buildLevel(List<String> rows) {
548 int height = rows.size();
549 int width = rows.get(0).length();
550 String cells = String.join("", rows);
551 boolean[] walls = new boolean[cells.length()];
552 boolean[] goals = new boolean[cells.length()];
553 List<Integer> players = new ArrayList<>();
554 List<Integer> boxes = new ArrayList<>();
555 for (int cell = 0; cell < cells.length(); cell++) {
556 char symbol = cells.charAt(cell);
557 walls[cell] = symbol == '#';
558 goals[cell] = ".*+".indexOf(symbol) >= 0;
559 if ("@+".indexOf(symbol) >= 0) {
560 players.add(cell);
561 }
562 if ("$*".indexOf(symbol) >= 0) {
563 boxes.add(cell);
564 }
565 }
566 boolean[] interior = players.size() == 1
567 ? findInterior(height, width, walls, players.get(0))
568 : new boolean[cells.length()];
569 Level level = new Level(height, width, walls, goals, interior);
570 level.reason = level.invalidity(players.size(), boxes);
571 if (level.reason == null) {
572 level.analyze();
573 }
574 int player = players.isEmpty() ? -1 : players.get(0);
575 return new LoadedLevel(level, new Position(player, List.copyOf(boxes)));
576 }

587 static String solutionNotation(Level level, Position start, List<Push> pushes) {
588 Position position = start;
589 StringBuilder notation = new StringBuilder();
590 for (Push push : pushes) {
591 int offset = level.offsetOf(push.direction());

593 notation.append(level.walkPath(position.player(), behind, position.boxes()));
594 notation.append(Character.toUpperCase(push.direction()));

596 position = new Position(push.box(), boxes);
597 }
598 return notation.toString();

601 static SearchResult pushSearch(Level level, Position start, List<Integer> fixed,
602 OptionalInt target, boolean pruning, OptionalInt limit) {
603 int startRepresentative = level.representative(start.player(), start.boxes());
604 List<SearchNode> nodes = new ArrayList<>();
605 nodes.add(new SearchNode(start.boxes(), startRepresentative, NO_PARENT, null));
606 Set<SearchState> known = new HashSet<>();
607 known.add(new SearchState(start.boxes(), startRepresentative));
608 int expanded = 0;
609 while (expanded < nodes.size()) {
610 if (limit.isPresent() && expanded == limit.getAsInt()) {
611 return new SearchResult("limit", expanded, nodes.size(), "");
612 }
613 SearchNode node = nodes.get(expanded);
614 expanded++;
615 boolean[] reachable = level.reachable(node.representative(), node.boxes());
616 for (int box : node.boxes()) {
617 if (fixed.contains(box)) {
618 continue;
619 }
620 for (char direction : DIRECTIONS.toCharArray()) {
621 if (!level.pushable(box, direction, node.boxes(), reachable)) {
622 continue;
623 }
624 int destination = box + level.offsetOf(direction);
625 List<Integer> boxes = moveBox(node.boxes(), box, destination);
626 if (pruning && (level.dead[destination] || level.hasFrozenBox(boxes))) {
627 continue;
628 }
629 int representative = level.representative(box, boxes);
630 if (!known.add(new SearchState(boxes, representative))) {
631 continue;
632 }
633 Push push = new Push(box, direction);
634 nodes.add(new SearchNode(boxes, representative, expanded - 1, push));
635 boolean reached = target.isPresent()
636 ? destination == target.getAsInt()
637 : level.solved(boxes);
638 if (reached) {
639 String notation = solutionNotation(level, start, solutionPushes(nodes));
640 return new SearchResult("found", expanded, nodes.size(), notation);
641 }
642 }
643 }
644 }
645 return new SearchResult("no solution", expanded, nodes.size(), "");
646 }

747 private void importLevel(String text) {
748 List<String> rows = new ArrayList<>();
749 for (String part : text.split("\\|", -1)) {
750 Optional<String> row = decode(part, LEVEL_CHARACTERS);
751 if (row.isEmpty()) {
752 throw new CommandException("bad level");
753 }
754 rows.add(row.get());
755 }
756 if (rows.size() > MAX_SIDE) {
757 throw new CommandException("bad level");

761 if (uneven || width < 1 || width > MAX_SIDE) {
762 throw new CommandException("bad level");
763 }
764 load(rows);
765 emit("loaded", rows.size() + "x" + width);
766 }

803 private void push(int box, char direction) {
804 int offset = level.offsetOf(direction);
805 List<Integer> boxes = current.boxes();
806 if (!boxes.contains(box)) {
807 throw new CommandException("no box");
808 }
809 int destination = box + offset;
810 if (!level.interior[destination] || boxes.contains(destination)) {
811 throw new CommandException("blocked");
812 }
813 if (!level.reachable(current.player(), boxes)[box - offset]) {
814 throw new CommandException("unreachable");
815 }
816 String path = level.walkPath(current.player(), box - offset, boxes);
817 emit("pushed", takeSteps(path + direction).letters());
818 }

820 private void bring(int source, int target) {
821 List<Integer> boxes = current.boxes();
822 if (!boxes.contains(source)) {
823 throw new CommandException("no box");
824 }
825 if (source == target) {
826 emit("brought", 0);
827 return;
828 }
829 List<Integer> fixed = boxes.stream().filter(box -> box != source).toList();
830 SearchResult result = pushSearch(level, current, fixed, OptionalInt.of(target), false,
831 OptionalInt.empty());
832 if (!result.outcome().equals("found")) {
833 throw new CommandException("no route");
834 }
835 takeSteps(result.notation().toLowerCase());
836 emit("brought", pushCount(result.notation()), result.notation());
837 }

841 throw new CommandException("unreachable");
842 }
843 String path = level.walkPath(current.player(), target, current.boxes());

847 private void undo(int requested) {
848 if (history.isEmpty()) {
849 throw new CommandException("nothing to undo");
850 }
851 int count = Math.min(requested, history.length());
852 for (int index = 0; index < count; index++) {
853 char letter = history.charAt(history.length() - 1);
854 history.deleteCharAt(history.length() - 1);
855 char direction = Character.toLowerCase(letter);
856 int offset = level.offsetOf(direction);
857 int player = current.player();
858 List<Integer> boxes = Character.isUpperCase(letter)
859 ? moveBox(current.boxes(), player + offset, player)
860 : current.boxes();
861 current = new Position(player - offset, boxes);
862 redoList.append(direction);
863 }
864 emit("undone", count);
865 }
866
867 private void redo(int requested) {
868 if (redoList.isEmpty()) {
869 throw new CommandException("nothing to redo");
870 }
871 int count = Math.min(requested, redoList.length());
872 for (int index = 0; index < count; index++) {
873 char direction = redoList.charAt(redoList.length() - 1);
874 redoList.deleteCharAt(redoList.length() - 1);
875 Step step = level.step(current, direction);
876 current = step.position();
877 history.append(step.letter());
878 }
879 emit("redone", count);
880 }

928 private void deadlock() {
929 List<Integer> boxes = current.boxes();
930 for (int box : boxes) {
931 if (level.dead[box]) {
932 emit("deadlock", "dead", level.cellText(box));
933 return;
934 }
935 }
936 for (int box : boxes) {
937 if (!level.goals[box] && level.frozen(box, boxes, List.of())) {
938 emit("deadlock", "frozen", level.cellText(box));
939 return;
940 }
941 }
942 emit("no deadlock");
943 }

945 private void bound() {
946 List<Integer> boxes = current.boxes();
947 List<Integer> goalIndices = IntStream.range(0, level.goalCells.size()).boxed().toList();
948 Assignment best = null;
949 for (List<Integer> order : permutations(goalIndices)) {
950 List<Integer> pushes = new ArrayList<>();
951 for (int index = 0; index < boxes.size(); index++) {
952 pushes.add(level.distances.get(order.get(index))[boxes.get(index)]);
953 }
954 if (pushes.contains(UNDEFINED)) {
955 continue;
956 }
957 int total = pushes.stream().mapToInt(Integer::intValue).sum();
958 if (best == null || total < best.total()) {
959 best = new Assignment(order, pushes, total);
960 }
961 }
962 if (best == null) {
963 emit("bound", "none");
964 return;
965 }
966 emit("bound", best.total());
967 for (int index = 0; index < boxes.size(); index++) {
968 int goal = level.goalCells.get(best.order().get(index));
969 emit("box", level.cellText(boxes.get(index)), "goal", level.cellText(goal),
970 "pushes", best.pushes().get(index));
971 }
972 }

1053 private void checkValues(Command command, List<String> arguments) {
1054 int cellCount = switch (command) {
1055 case PATH, PUSH -> 1;
1056 case BRING -> 2;
1057 default -> 0;
1058 };
1059 for (int index = 0; index < 2 * cellCount; index += 2) {
1060 int row = Integer.parseInt(arguments.get(index));
1061 int column = Integer.parseInt(arguments.get(index + 1));
1062 if (!level.contains(row, column)) {
1063 throw new CommandException("bad value");
1064 }
1065 }
1066 if ((command == Command.UNDO || command == Command.REDO)
1067 && Integer.parseInt(arguments.get(0)) < 1) {
1068 throw new CommandException("bad value");
1069 }
1070 if (command == Command.SOLVE || command == Command.HINT) {
1071 int limit = Integer.parseInt(arguments.get(0));
1072 if (limit < 1 || limit > MAX_LIMIT) {
1073 throw new CommandException("bad value");
1074 }
1075 }
1076 }

1078 private void runCommand(String name, List<String> arguments) {
1079 Command command = Command.named(name)
1080 .orElseThrow(() -> new CommandException("unknown command " + name));
1081 List<Kind> kinds = command.kinds;
1082 if (arguments.size() != kinds.size() || !IntStream.range(0, kinds.size())
1083 .allMatch(index -> validArgument(kinds.get(index), arguments.get(index)))) {
1084 throw new CommandException("bad arguments");
1085 }
1086 if (command != Command.IMPORT && command != Command.VALID) {
1087 if (level.reason != null) {
1088 throw new CommandException("invalid level");
1089 }
1090 checkValues(command, arguments);
1091 }
1092 switch (command) {
1093 case IMPORT -> importLevel(arguments.get(0));
1094 case VALID -> valid();
1095 case INFO -> info();
1096 case SHOW -> show();
1097 case EXPORT -> exportLevel();
1098 case MOVE -> move(arguments.get(0));
1099 case PUSH -> push(cellArgument(arguments, 0), arguments.get(2).charAt(0));
1100 case BRING -> bring(cellArgument(arguments, 0), cellArgument(arguments, 2));
1101 case PATH -> path(cellArgument(arguments, 0));
1102 case UNDO -> undo(Integer.parseInt(arguments.get(0)));
1103 case REDO -> redo(Integer.parseInt(arguments.get(0)));
1104 case RESET -> reset();
1105 case STATUS -> status();
1106 case HISTORY -> showHistory();
1107 case BOXES -> showBoxes();
1108 case REACH -> reach();
1109 case DEADSQUARES -> deadSquares();
1110 case DEADLOCK -> deadlock();
1111 case BOUND -> bound();
1112 case SAVE -> save(arguments.get(0));
1113 case RESTORE -> restore(arguments.get(0));
1114 case CHECK -> check(arguments.get(0));
1115 case SOLVE -> solve(Integer.parseInt(arguments.get(0)));
1116 case HINT -> hint(Integer.parseInt(arguments.get(0)));
1117 case PLAY -> play();
1118 case TRANSFORM -> transform(arguments.get(0));
1119 }
1120 }
