50function decode(text: string, allowed: string): string | null {
51 let result = '';
52 for (let index = 0; index < text.length;) {
53 const start = index;
54 while (index < text.length && text[index] >= '0' && text[index] <= '9') index++;
55 const count = text.slice(start, index);
56 if (count.length > 2 || count.startsWith('0') || index === text.length) return null;
57 const symbol = text[index++];
58 if (!allowed.includes(symbol)) return null;
59 result += symbol.repeat(count ? Number(count) : 1);
60 }
61 return result;
62}

64function encode(text: string): string {
65 let result = '';
66 for (let start = 0; start < text.length;) {
67 let end = start + 1;
68 while (end < text.length && text[end] === text[start]) end++;
69 const count = end - start;
70 result += (count > 1 ? String(count) : '') + text[start];
71 start = end;
72 }
73 return result;
74}

76function validArgument(kind: ArgumentKind, text: string): boolean {
77 switch (kind) {
78 case 'number': return /^(0|[1-9][0-9]{0,5})$/.test(text);
79 case 'steps': return decode(text, STEP_LETTERS) !== null;
80 case 'direction': return DIRECTIONS.some(direction => direction === text);
81 case 'name': return /^[a-z]{1,8}$/.test(text);
82 case 'transform': return TURNS.has(text);
83 default: return true;
84 }

108function findInterior(height: number, width: number, walls: readonly boolean[], start: number): boolean[] {
109 const seen = Array<boolean>(height * width).fill(false);
110 const queue = [start];
111 seen[start] = true;
112 for (let head = 0; head < queue.length; head++) {
113 const cell = queue[head];
114 const row = Math.floor(cell / width), column = cell % width;
115 for (const [dr, dc] of DELTAS) {
116 const r = row + dr, c = column + dc;
117 if (r < 0 || r >= height || c < 0 || c >= width) continue;
118 const neighbor = r * width + c;
119 if (!walls[neighbor] && !seen[neighbor]) {
120 seen[neighbor] = true;
121 queue.push(neighbor);
122 }
123 }
124 }
125 return seen;
126}

128class Level {
129 readonly height: number;
130 readonly width: number;
131 readonly size: number;
132 readonly walls: readonly boolean[];
133 readonly goals: readonly boolean[];
134 readonly interior: readonly boolean[];
135 readonly offsets: readonly number[];
136 readonly goalCells: readonly number[];
137 reason: string | null;
138 distances: number[][];
139 dead: boolean[];
140 constructor(height: number, width: number, walls: readonly boolean[], goals: readonly boolean[], interior: readonly boolean[]) {
141 this.height = height;
142 this.width = width;
143 this.size = height * width;
144 this.walls = walls;
145 this.goals = goals;
146 this.interior = interior;
147 this.offsets = [-width, width, -1, 1];
148 this.goalCells = Array.from({length: this.size}, (_, cell) => cell).filter(cell => goals[cell]);
149 this.reason = null;
150 this.distances = [];
151 this.dead = Array<boolean>(this.size).fill(false);
152 }
153
154 offsetOf(direction: Direction): number { return this.offsets[DIRECTIONS.indexOf(direction)]; }
155 cellText(cell: number): string { return `${Math.floor(cell / this.width) + 1} ${cell % this.width + 1}`; }
156 cellAt(row: number, column: number): number { return (row - 1) * this.width + column - 1; }
157 contains(row: number, column: number): boolean { return row >= 1 && row <= this.height && column >= 1 && column <= this.width; }
158 onBorder(cell: number): boolean {
159 const row = Math.floor(cell / this.width), column = cell % this.width;
160 return row === 0 || row === this.height - 1 || column === 0 || column === this.width - 1;
161 }
162
163 invalidity(players: number, boxes: readonly number[]): string | null {
164 if (players !== 1) return `players ${players}`;
165 if (!boxes.length) return 'no boxes';
166 if (boxes.length !== this.goalCells.length) return `boxes ${boxes.length} goals ${this.goalCells.length}`;
167 for (let cell = 0; cell < this.size; cell++) {
168 if (this.interior[cell] && this.onBorder(cell)) return `open ${this.cellText(cell)}`;
169 }
170 for (let cell = 0; cell < this.size; cell++) {
171 if ((this.goals[cell] || boxes.includes(cell)) && !this.interior[cell]) return `outside ${this.cellText(cell)}`;
172 }
173 return null;
174 }
175
176 analyze(): void {
177 this.distances = this.goalCells.map(goal => this.pushDistances(goal));
178 this.dead = this.interior.map((inside, cell) => inside && this.distances.every(distance => distance[cell] === -1));
179 }
180
181 pushDistances(goal: number): number[] {
182 const distance = Array<number>(this.size).fill(-1);
183 const queue = [goal];
184 distance[goal] = 0;
185 for (let head = 0; head < queue.length; head++) {
186 const cell = queue[head];
187 for (const offset of this.offsets) {
188 const box = cell - offset;
189 if (this.interior[box] && this.interior[box - offset] && distance[box] === -1) {
190 distance[box] = distance[cell] + 1;
191 queue.push(box);
192 }
193 }
194 }
195 return distance;
196 }
197
198 goalDistance(cell: number): number | null {
199 const defined = this.distances.map(distance => distance[cell]).filter(distance => distance >= 0);
200 return defined.length ? Math.min(...defined) : null;
201 }
202 boxesOnGoals(boxes: readonly number[]): number { return boxes.filter(box => this.goals[box]).length; }
203 solved(boxes: readonly number[]): boolean { return this.boxesOnGoals(boxes) === boxes.length; }
204
205 reachable(player: number, boxes: readonly number[]): boolean[] {
206 const seen = Array<boolean>(this.size).fill(false);
207 const queue = [player];
208 seen[player] = true;
209 for (let head = 0; head < queue.length; head++) {
210 for (const offset of this.offsets) {
211 const neighbor = queue[head] + offset;
212 if (this.interior[neighbor] && !seen[neighbor] && !boxes.includes(neighbor)) {
213 seen[neighbor] = true;
214 queue.push(neighbor);
215 }
216 }
217 }
218 return seen;
219 }
220 representative(player: number, boxes: readonly number[]): number { return this.reachable(player, boxes).indexOf(true); }
221
222 walkPath(start: number, target: number, boxes: readonly number[]): string {
223 const cameFrom = Array<CameFrom | null>(this.size).fill(null);
224 const queue = [start];
225 for (let head = 0; head < queue.length; head++) {
226 const cell = queue[head];
227 for (const direction of DIRECTIONS) {
228 const neighbor = cell + this.offsetOf(direction);
229 if (this.interior[neighbor] && !boxes.includes(neighbor)
230 && neighbor !== start && cameFrom[neighbor] === null) {
231 cameFrom[neighbor] = {cell, direction};
232 queue.push(neighbor);
233 }
234 }
235 }
236 const path: Direction[] = [];
237 for (let cell = target; cell !== start;) {
238 const from = cameFrom[cell]!;
239 path.push(from.direction);
240 cell = from.cell;
241 }
242 return path.reverse().join('');
243 }
244
245 step(position: Position, direction: Direction): Step {
246 const offset = this.offsetOf(direction);
247 const target = position.player + offset;
248 if (this.walls[target]) throw new StepFailure('wall');
249 if (!position.boxes.includes(target)) {
250 return {position: {player: target, boxes: position.boxes}, letter: direction};
251 }
252 const beyond = target + offset;
253 if (this.walls[beyond] || position.boxes.includes(beyond)) throw new StepFailure('blocked');
254 return {position: {player: target, boxes: moveBox(position.boxes, target, beyond)}, letter: direction.toUpperCase()};
255 }
256
257 pushable(box: number, direction: Direction, boxes: readonly number[], reachable: readonly boolean[]): boolean {
258 const offset = this.offsetOf(direction), destination = box + offset;
259 return !!reachable[box - offset] && !!this.interior[destination] && !boxes.includes(destination);
260 }
261 sideBlocked(box: number, sides: readonly number[], boxes: readonly number[], assumed: readonly number[]): boolean {
262 if (sides.some(side => this.walls[side] || assumed.includes(side))) return true;
263 if (sides.every(side => this.dead[side])) return true;
264 const extended = [...assumed, box];
265 return sides.some(side => boxes.includes(side) && this.frozen(side, boxes, extended));
266 }
267 frozen(box: number, boxes: readonly number[], assumed: readonly number[] = []): boolean {
268 return this.sideBlocked(box, [box - 1, box + 1], boxes, assumed)
269 && this.sideBlocked(box, [box - this.width, box + this.width], boxes, assumed);
270 }
271 hasFrozenBox(boxes: readonly number[]): boolean { return boxes.some(box => !this.goals[box] && this.frozen(box, boxes)); }
272
273 splitRows(symbols: string): string[] {
274 const rows: string[] = [];
275 for (let start = 0; start < this.size; start += this.width) rows.push(symbols.slice(start, start + this.width));
276 return rows;
277 }
278 render(position: Position): string[] {
279 let symbols = '';
280 for (let cell = 0; cell < this.size; cell++) {
281 if (this.walls[cell]) symbols += '#';
282 else if (cell === position.player) symbols += this.goals[cell] ? '+' : '@';
283 else if (position.boxes.includes(cell)) symbols += this.goals[cell] ? '*' : '$';
284 else symbols += this.goals[cell] ? '.' : '-';
285 }
286 return this.splitRows(symbols);
287 }
288 deadMap(): string[] {
289 let symbols = '';
290 for (let cell = 0; cell < this.size; cell++) {
291 symbols += this.walls[cell] ? '#' : this.goals[cell] ? '.' : this.dead[cell] ? 'x' : '-';
292 }
293 return this.splitRows(symbols);
294 }

296 mapCell(kind: Transform, cell: number): number {
297 const row = Math.floor(cell / this.width), column = cell % this.width;
298 switch (kind) {
299 case 'ROT90': return column * this.height + this.height - 1 - row;
300 case 'ROT180': return (this.height - 1 - row) * this.width + this.width - 1 - column;
301 case 'ROT270': return (this.width - 1 - column) * this.height + row;
302 case 'FLIPH': return row * this.width + this.width - 1 - column;
303 default: return (this.height - 1 - row) * this.width + column;
304 }
305 }
306 mapPosition(kind: Transform, position: Position): Position {
307 return {player: this.mapCell(kind, position.player), boxes: position.boxes.map(box => this.mapCell(kind, box)).sort((a, b) => a - b)};
308 }
309 transformed(kind: Transform): Level {
310 const rotated = kind === 'ROT90' || kind === 'ROT270';
311 const walls = Array<boolean>(this.size).fill(false), goals = Array<boolean>(this.size).fill(false), interior = Array<boolean>(this.size).fill(false);
312 for (let cell = 0; cell < this.size; cell++) {
313 const moved = this.mapCell(kind, cell);
314 walls[moved] = this.walls[cell];
315 goals[moved] = this.goals[cell];
316 interior[moved] = this.interior[cell];
317 }
318 const level = new Level(rotated ? this.width : this.height, rotated ? this.height : this.width, walls, goals, interior);
319 level.analyze();
320 return level;
321 }
322}

324function buildLevel(rows: readonly string[]): LoadedLevel {
325 const height = rows.length, width = rows[0].length;
326 const cells = [...rows.join('')];
327 const walls = cells.map(symbol => symbol === '#'), goals = cells.map(symbol => '.*+'.includes(symbol));
328 const players: number[] = [], boxes: number[] = [];
329 cells.forEach((symbol, cell) => {
330 if ('@+'.includes(symbol)) players.push(cell);
331 if ('$*'.includes(symbol)) boxes.push(cell);
332 });
333 const interior = players.length === 1 ? findInterior(height, width, walls, players[0]) : Array<boolean>(cells.length).fill(false);
334 const level = new Level(height, width, walls, goals, interior);
335 level.reason = level.invalidity(players.length, boxes);
336 if (level.reason === null) level.analyze();
337 return {level, position: {player: players.length ? players[0] : -1, boxes}};
338}

340function stateKey(boxes: readonly number[], representative: number): string { return `${boxes.join(',')}:${representative}`; }
341function solutionNotation(level: Level, start: Position, nodes: readonly SearchNode[]): string {
342 const pushes: Push[] = [];
343 for (let node = nodes[nodes.length - 1]; node.parent !== -1; node = nodes[node.parent]) {
344 if (node.push !== null) pushes.push(node.push);
345 }
346 let position = start, notation = '';
347 for (const {box, direction} of pushes.reverse()) {
348 const offset = level.offsetOf(direction);
349 notation += level.walkPath(position.player, box - offset, position.boxes) + direction.toUpperCase();
350 position = {player: box, boxes: moveBox(position.boxes, box, box + offset)};
351 }
352 return notation;
353}

355function pushSearch(level: Level, start: Position, fixed: readonly number[], target: number | null, pruning: boolean, limit: number | null): SearchResult {
356 const representative = level.representative(start.player, start.boxes);
357 const nodes: SearchNode[] = [{boxes: start.boxes, representative, parent: -1, push: null}];
358 const known = new Set([stateKey(start.boxes, representative)]);
359 let expanded = 0;
360 while (expanded < nodes.length) {
361 if (limit !== null && expanded === limit) return {outcome: 'limit', expanded, states: nodes.length, notation: ''};
362 const parent = expanded, node = nodes[expanded++];
363 const reachable = level.reachable(node.representative, node.boxes);
364 for (const box of node.boxes) {
365 if (fixed.includes(box)) continue;
366 for (const direction of DIRECTIONS) {
367 if (!level.pushable(box, direction, node.boxes, reachable)) continue;
368 const destination = box + level.offsetOf(direction);
369 const boxes = moveBox(node.boxes, box, destination);
370 if (pruning && (level.dead[destination] || level.hasFrozenBox(boxes))) continue;
371 const nextRepresentative = level.representative(box, boxes);
372 const key = stateKey(boxes, nextRepresentative);
373 if (known.has(key)) continue;
374 known.add(key);
375 nodes.push({boxes, representative: nextRepresentative, parent, push: {box, direction}});
376 if (target === null ? level.solved(boxes) : destination === target) {
377 return {outcome: 'found', expanded, states: nodes.length, notation: solutionNotation(level, start, nodes)};
378 }
379 }
380 }
381 }
382 return {outcome: 'no solution', expanded, states: nodes.length, notation: ''};
383}

385class Sokoban {
386 readonly output: string[];
387 private level!: Level;
388 private initial!: Position;
389 private current!: Position;
390 private history = '';
391 private redoList = '';
392 private snapshots = new Map<string, Snapshot>();
393 private solution: Solution | null = null;
394 constructor(rows: readonly string[]) {
395 this.output = [];
396 this.load(rows);
397 }
398 emit(...parts: (string | number)[]): void { this.output.push(parts.join(' ')); }
399 load(rows: readonly string[]): void {
400 const loaded = buildLevel(rows);
401 this.level = loaded.level;
402 this.initial = loaded.position;
403 this.current = this.initial;
404 this.history = '';
405 this.redoList = '';
406 this.snapshots = new Map<string, Snapshot>();
407 this.solution = null;
408 }
409 cellArgument(args: readonly string[], index: number): number { return this.level.cellAt(Number(args[index]), Number(args[index + 1])); }
410
411 simulate(directions: string): Simulation {
412 let position = this.current, letters = '';
413 for (const letter of directions) {
414 try {
415 const step = this.level.step(position, directionOf(letter));
416 position = step.position;
417 letters += step.letter;
418 } catch (error) {
419 if (!(error instanceof StepFailure)) throw error;
420 return {position, letters, failure: `step ${letters.length + 1} ${error.message}`};
421 }
422 }
423 return {position, letters, failure: null};
424 }
425 takeSteps(directions: string): Simulation {
426 const result = this.simulate(directions);
427 this.current = result.position;
428 this.history += result.letters;
429 if (result.letters.length) this.redoList = '';
430 return result;
431 }
432 progress(boxes: readonly number[]): string {
433 const count = this.level.boxesOnGoals(boxes);
434 return `goals ${count}/${boxes.length}` + (count === boxes.length ? ' solved' : '');
435 }
436 searchSolution(limit: number): Summary {
437 if (this.level.solved(this.current.boxes)) return {text: 'already solved', notation: null};
438 const result = pushSearch(this.level, this.current, [], null, true, limit);
439 const counts = `expanded ${result.expanded} states ${result.states}`;
440 if (result.outcome !== 'found') return {text: `${result.outcome} ${counts}`, notation: null};
441 const notation = result.notation;
442 return {text: `solution pushes ${pushCount(notation)} moves ${notation.length} ${counts}`, notation};
443 }
444
445 loadRows(width: number, rows: readonly string[]): void {
446 const bad = rows.findIndex(row => row.length !== width || [...row].some(symbol => !LEVEL_CHARACTERS.includes(symbol)));
447 if (bad !== -1) throw new CommandError(`bad row ${bad + 1}`);
448 this.load(rows);
449 this.emit('loaded', `${rows.length}x${width}`);
450 }
451 importLevel(text: string): void {
452 const rows: string[] = [];
453 for (const part of text.split('|')) {
454 const row = decode(part, LEVEL_CHARACTERS);
455 if (row === null) throw new CommandError('bad level');
456 rows.push(row);
457 }
458 const width = rows[0].length;
459 if (rows.length > 10 || width < 1 || width > 10 || rows.some(row => row.length !== width)) throw new CommandError('bad level');
460 this.load(rows);
461 this.emit('loaded', `${rows.length}x${width}`);
462 }
463 move(word: string): void {
464 const result = this.takeSteps(decode(word, STEP_LETTERS)!);
465 if (result.letters.length) this.emit('moved', result.letters);
466 if (result.failure !== null) throw new CommandError(result.failure);
467 }
468 push(box: number, direction: Direction): void {
469 const level = this.level, boxes = this.current.boxes;
470 if (!boxes.includes(box)) throw new CommandError('no box');
471 const offset = level.offsetOf(direction), destination = box + offset;
472 if (!level.interior[destination] || boxes.includes(destination)) throw new CommandError('blocked');
473 if (!level.reachable(this.current.player, boxes)[box - offset]) throw new CommandError('unreachable');
474 const path = level.walkPath(this.current.player, box - offset, boxes);
475 this.emit('pushed', this.takeSteps(path + direction).letters);
476 }
477 bring(source: number, target: number): void {
478 const boxes = this.current.boxes;
479 if (!boxes.includes(source)) throw new CommandError('no box');
480 if (source === target) { this.emit('brought', 0); return; }
481 const result = pushSearch(this.level, this.current, boxes.filter(box => box !== source), target, false, null);
482 if (result.outcome !== 'found') throw new CommandError('no route');
483 this.takeSteps(result.notation);
484 this.emit('brought', pushCount(result.notation), result.notation);
485 }
486 path(target: number): void {
487 if (!this.level.reachable(this.current.player, this.current.boxes)[target]) throw new CommandError('unreachable');
488 const path = this.level.walkPath(this.current.player, target, this.current.boxes);
489 this.emit(counted('path', this.takeSteps(path).letters));
490 }
491 undo(requested: number): void {
492 if (!this.history.length) throw new CommandError('nothing to undo');
493 const count = Math.min(requested, this.history.length);
494 for (let index = 0; index < count; index++) {
495 const letter = this.history[this.history.length - 1], direction = directionOf(letter);
496 this.history = this.history.slice(0, -1);
497 const offset = this.level.offsetOf(direction), player = this.current.player;
498 const boxes = letter === direction ? this.current.boxes : moveBox(this.current.boxes, player + offset, player);
499 this.current = {player: player - offset, boxes};
500 this.redoList += direction;
501 }
502 this.emit('undone', count);
503 }
504 redo(requested: number): void {
505 if (!this.redoList.length) throw new CommandError('nothing to redo');
506 const count = Math.min(requested, this.redoList.length);
507 for (let index = 0; index < count; index++) {
508 const direction = directionOf(this.redoList[this.redoList.length - 1]);
509 this.redoList = this.redoList.slice(0, -1);
510 const step = this.level.step(this.current, direction);
511 this.current = step.position;
512 this.history += step.letter;
513 }
514 this.emit('redone', count);
515 }
516 showBoxes(): void {
517 const boxes = this.current.boxes, level = this.level;
518 const reachable = level.reachable(this.current.player, boxes);
519 this.emit('boxes', boxes.length);
520 for (const box of boxes) {
521 const distance = level.goalDistance(box);
522 const directions = DIRECTIONS.filter(direction => level.pushable(box, direction, boxes, reachable)).join('');
523 this.emit('box', level.cellText(box), distance === null ? 'dead' : distance, directions || '-');
524 }
525 }
526 deadlock(): void {
527 const level = this.level, boxes = this.current.boxes;
528 const dead = boxes.find(box => level.dead[box]);
529 if (dead !== undefined) { this.emit('deadlock dead', level.cellText(dead)); return; }
530 const frozen = boxes.find(box => !level.goals[box] && level.frozen(box, boxes));
531 if (frozen !== undefined) { this.emit('deadlock frozen', level.cellText(frozen)); return; }
532 this.emit('no deadlock');
533 }
534 bound(): void {
535 const boxes = this.current.boxes, level = this.level;
536 const best: {value: Assignment | null} = {value: null};
537 const assign = (order: number[], pushes: number[], used: ReadonlySet<number>, total: number): void => {
538 if (order.length === boxes.length) {
539 if (best.value === null || total < best.value.total) best.value = {order, pushes, total};
540 return;
541 }
542 for (let goal = 0; goal < level.goalCells.length; goal++) {
543 const distance = level.distances[goal][boxes[order.length]];
544 if (!used.has(goal) && distance >= 0) {
545 assign([...order, goal], [...pushes, distance], new Set([...used, goal]), total + distance);
546 }
547 }
548 };
549 assign([], [], new Set<number>(), 0);
550 const chosen = best.value;
551 if (chosen === null) { this.emit('bound none'); return; }
552 this.emit('bound', chosen.total);
553 boxes.forEach((box, index) => this.emit('box', level.cellText(box), 'goal', level.cellText(level.goalCells[chosen.order[index]]), 'pushes', chosen.pushes[index]));
554 }
555 save(name: string): void {
556 this.snapshots.set(name, {position: this.current, history: this.history});
557 this.emit('saved', name, 'moves', this.history.length);
558 }
559 restore(name: string): void {
560 const snapshot = this.snapshots.get(name);
561 if (snapshot === undefined) throw new CommandError(`no snapshot ${name}`);
562 this.current = snapshot.position;
563 this.history = snapshot.history;
564 this.redoList = '';
565 this.emit('restored', name, 'moves', this.history.length);
566 }
567 check(word: string): void {
568 const result = this.simulate(decode(word, STEP_LETTERS)!);
569 if (result.failure !== null) this.emit('check fail', result.failure);
570 else this.emit('check ok moves', result.letters.length, 'pushes', pushCount(result.letters), this.progress(result.position.boxes));
571 }
572 solve(limit: number): void {
573 const summary = this.searchSolution(limit);
574 this.emit(summary.text);
575 this.solution = null;
576 if (summary.notation !== null) {
577 this.emit(summary.notation);
578 this.solution = {notation: summary.notation, position: this.current};
579 }
580 }
581 hint(limit: number): void {
582 const summary = this.searchSolution(limit);
583 if (summary.notation === null) { this.emit(summary.text); return; }
584 const end = [...summary.notation].findIndex(letter => letter !== letter.toLowerCase());
585 const part = summary.notation.slice(0, end + 1);
586 this.takeSteps(part);
587 this.emit('hint', part);
588 }
589 play(): void {
590 if (this.solution === null || !samePosition(this.solution.position, this.current)) throw new CommandError('no solution');
591 const notation = this.solution.notation;
592 this.takeSteps(notation);
593 this.emit('played pushes', pushCount(notation), 'moves', notation.length);
594 }
595 transform(kind: Transform): void {
596 const old = this.level;
597 this.level = old.transformed(kind);
598 this.initial = old.mapPosition(kind, this.initial);
599 this.current = old.mapPosition(kind, this.current);
600 this.history = turnLetters(this.history, kind);
601 this.redoList = turnLetters(this.redoList, kind);
602 for (const [name, snapshot] of this.snapshots) {
603 this.snapshots.set(name, {position: old.mapPosition(kind, snapshot.position), history: turnLetters(snapshot.history, kind)});
604 }
605 if (this.solution !== null) this.solution = {notation: turnLetters(this.solution.notation, kind), position: old.mapPosition(kind, this.solution.position)};
606 this.emit('transformed', `${this.level.height}x${this.level.width}`);
607 }

609 checkValues(name: string, args: readonly string[]): void {
610 const cells = name === 'BRING' ? 2 : name === 'PATH' || name === 'PUSH' ? 1 : 0;
611 for (let index = 0; index < cells * 2; index += 2) {
612 if (!this.level.contains(Number(args[index]), Number(args[index + 1]))) throw new CommandError('bad value');
613 }
614 if ((name === 'UNDO' || name === 'REDO') && Number(args[0]) < 1) throw new CommandError('bad value');
615 if ((name === 'SOLVE' || name === 'HINT') && (Number(args[0]) < 1 || Number(args[0]) > 100000)) throw new CommandError('bad value');
616 }
617 runCommand(name: string, args: readonly string[]): void {
618 const kinds = COMMANDS.get(name);
619 if (kinds === undefined) throw new CommandError(`unknown command ${name}`);
620 if (args.length !== kinds.length || kinds.some((kind, index) => !validArgument(kind, args[index]))) throw new CommandError('bad arguments');
621 if (name === 'IMPORT') { this.importLevel(args[0]); return; }
622 if (name === 'VALID') { this.emit(this.level.reason === null ? 'valid' : `invalid ${this.level.reason}`); return; }
623 if (this.level.reason !== null) throw new CommandError('invalid level');
624 this.checkValues(name, args);
625 const level = this.level;
626 switch (name) {
627 case 'INFO': this.emit('size', `${level.height}x${level.width}`, 'walls', countTrue(level.walls), 'cells', countTrue(level.interior), 'boxes', this.current.boxes.length, 'dead', countTrue(level.dead)); break;
628 case 'SHOW': level.render(this.current).forEach(row => this.emit(row)); break;
629 case 'EXPORT': this.emit('level', level.render(this.current).map(encode).join('|')); break;
630 case 'MOVE': this.move(args[0]); break;
631 case 'PUSH': this.push(this.cellArgument(args, 0), directionOf(args[2])); break;
632 case 'BRING': this.bring(this.cellArgument(args, 0), this.cellArgument(args, 2)); break;
633 case 'PATH': this.path(this.cellArgument(args, 0)); break;
634 case 'UNDO': this.undo(Number(args[0])); break;
635 case 'REDO': this.redo(Number(args[0])); break;
636 case 'RESET': this.current = this.initial; this.history = ''; this.redoList = ''; this.emit('reset'); break;
637 case 'STATUS': this.emit('moves', this.history.length, 'pushes', pushCount(this.history), this.progress(this.current.boxes)); break;
638 case 'HISTORY': this.emit(counted('history', this.history)); break;
639 case 'BOXES': this.showBoxes(); break;
640 case 'REACH': {
641 const reach = level.reachable(this.current.player, this.current.boxes);
642 this.emit('reach', countTrue(reach), 'first', level.cellText(reach.indexOf(true)));
643 break;
644 }
645 case 'DEADSQUARES': this.emit('dead', countTrue(level.dead)); level.deadMap().forEach(row => this.emit(row)); break;
646 case 'DEADLOCK': this.deadlock(); break;
647 case 'BOUND': this.bound(); break;
648 case 'SAVE': this.save(args[0]); break;
649 case 'RESTORE': this.restore(args[0]); break;
650 case 'CHECK': this.check(args[0]); break;
651 case 'SOLVE': this.solve(Number(args[0])); break;
652 case 'HINT': this.hint(Number(args[0])); break;
653 case 'PLAY': this.play(); break;
654 case 'TRANSFORM': this.transform(args[0] as Transform); break;
655 }
656 }
657 execute(words: readonly string[], rows: readonly string[]): void {
658 try {
659 if (words[0] === 'LOAD') this.loadRows(Number(words[2]), rows);
660 else this.runCommand(words[0], words.slice(1));
661 } catch (error) {
662 if (!(error instanceof CommandError)) throw error;
663 this.emit('error:', error.message);
664 }
665 }
