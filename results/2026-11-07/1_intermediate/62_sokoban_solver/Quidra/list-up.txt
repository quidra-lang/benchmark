nat pushes(string text)
    nat count := 0
    for: i < #text
        if "UDLR".contains(text[i])
            count += 1
    return count
    Simulation simulate(Position start, string steps)
        position := start
        letters := ""
        for: i < #steps
            d := direction_index(steps[i])
            delta := offsets()[d]
            target := position.player + delta
            push := has(position.boxes, target)
            if push
                position.boxes = moved_box(position.boxes, target, target + delta)
            position.player = target
            letters += if push then "UDLR"[d] else "udlr"[d]
        return Simulation(position, letters, "")
