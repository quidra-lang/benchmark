15time_text(minutes::Int) = @sprintf("%02d:%02d", div(minutes, 60), mod(minutes, 60))
16rate_text(count::Int, capacity::Int) = capacity == 0 ? "n/a" :

18course_word(text::String) = occursin(r"^[A-Z]{2,4}[0-9]{3}$", text)

21struct Expression
22 kind::Symbol
23 word::String
24 children::Vector{Expression}
25 references::Set{String}
26end
27
28true_expression() = Expression(:TRUE, "", Expression[], Set{String}())
29
30function leaf(word::String)
31 refs = year_word(word) ? Set{String}() : Set([word])
32 Expression(:WORD, word, Expression[], refs)
33end
34
35function combine(kind::Symbol, children::Vector{Expression})
36 flat = Expression[]
37 for child in children
38 if child.kind == kind
39 append!(flat, child.children)
40 else
41 push!(flat, child)
42 end
43 end
44 length(flat) == 1 && return flat[1]
45 refs = Set{String}()
46 for child in flat
47 union!(refs, child.references)
48 end
49 Expression(kind, "", flat, refs)
50end

52mutable struct Parser
53 tokens::Vector{String}
54 position::Int
55end
56
57peek(parser::Parser) =
58 parser.position > length(parser.tokens) ? "" : parser.tokens[parser.position]
59
60function parse_factor!(parser::Parser)
61 token = peek(parser)
62 if token == "" || token in ("&", "|", ")")
63 throw(ParseError())
64 end
65 parser.position += 1
66 if token == "("
67 node = parse_expression!(parser)
68 peek(parser) == ")" || throw(ParseError())
69 parser.position += 1
70 return node
71 end
72 leaf(token)
73end
74
75function parse_term!(parser::Parser)
76 children = Expression[parse_factor!(parser)]
77 while peek(parser) == "&"
78 parser.position += 1
79 push!(children, parse_factor!(parser))
80 end
81 combine(:AND, children)
82end
83
84function parse_expression!(parser::Parser)
85 children = Expression[parse_term!(parser)]
86 while peek(parser) == "|"
87 parser.position += 1
88 push!(children, parse_term!(parser))
89 end
90 combine(:OR, children)
91end

93function prerequisite(code::String, text::String)
94 text == "-" && return true_expression()
95 tokens = String[String(match.match) for match in eachmatch(r"[A-Z0-9]+|[&|()]", text)]
96 try
97 for token in tokens
98 if !(token in ("&", "|", "(", ")") || course_word(token) || year_word(token))
99 throw(ParseError())
100 end
101 end
102 parser = Parser(tokens, 1)
103 result = parse_expression!(parser)
104 parser.position == length(tokens) + 1 || throw(ParseError())
105 code in result.references && fail("self prereq " * code)
106 return result
107 catch error
108 error isa ParseError || rethrow()
109 fail("invalid prereq " * code)
110 end
111end
112
113function canonical(expr::Expression)
114 expr.kind == :TRUE && return "-"
115 expr.kind == :WORD && return expr.word
116 separator = expr.kind == :AND ? "&" : "|"
117 parts = String[]
118 for child in expr.children
119 text = canonical(child)
120 if expr.kind == :AND && child.kind == :OR
121 text = "(" * text * ")"
122 end
123 push!(parts, text)
124 end
125 join(parts, separator)
126end

128struct Slot
129 day::Int
130 start::Int
131 finish::Int
132 original::String
133end
134
135function make_slot(text::String)
136 day = get(DAY_INDEX, text[1:2], 0)
137 sh = parse(Int, text[3:4])
138 sm = parse(Int, text[5:6])
139 eh = parse(Int, text[8:9])
140 em = parse(Int, text[10:11])
141 first, last = sh * 60 + sm, eh * 60 + em
142 if day == 0 || sh > 23 || sm > 59 || eh > 23 || em > 59 || first >= last
143 fail("invalid slot " * text)
144 end
145 Slot(day, first, last, text)
146end
147
148overlaps(a::Slot, b::Slot) =
149 a.day == b.day && a.start < b.finish && b.start < a.finish
150slot_text(slot::Slot) = time_text(slot.start) * "-" * time_text(slot.finish)

180function Student(id::String, year::Int, limit::Int,
181 records::Dict{String, Tuple{Char, Int}})
182 passed = Dict{String, Char}(code => grade for (code, (grade, _)) in records
183 if grade in ('A', 'B', 'C'))
184 Student(id, year, limit, records, passed, Set{String}(),
185 Set{String}(), 0, Tuple{Int, String}[])
186end
187
188earned(student::Student) =
189 sum(grade == 'F' ? 0 : credits for (grade, credits) in values(student.records); init=0)
190
191function gpa(student::Student)
192 isempty(student.records) && return "n/a"
193 points = sum(GRADE_POINTS[grade] * credits
194 for (grade, credits) in values(student.records))
195 credits = sum(value for (_, value) in values(student.records))
196 rounded = div(200 * points + credits, 2 * credits)
197 @sprintf("%d.%02d", div(rounded, 100), mod(rounded, 100))
198end
199
200function satisfied(expr::Expression, student::Student)
201 expr.kind == :TRUE && return true
202 if expr.kind == :WORD
203 return year_word(expr.word) ?
204 student.year >= parse(Int, expr.word[2:end]) :
205 haskey(student.passed, expr.word)
206 elseif expr.kind == :AND
207 return all(satisfied(child, student) for child in expr.children)
208 end
209 any(satisfied(child, student) for child in expr.children)
210end

245function conflict(student::Student, course::Course, excluded::String="")
246 options = [code for code in student.courses
247 if code != excluded && code in course.conflicts]
248 isempty(options) ? nothing : minimum(options)
249end
250
251function eligibility(registrar::Registrar, student::Student, course::Course,
252 excluded::String="")
253 !course.open && return decision("course closed")
254 course.code in student.courses && return decision("already enrolled")
255 course.code in student.waiting && return decision("already waitlisted")
256 haskey(student.passed, course.code) &&
257 return decision("already completed", student.passed[course.code])
258 !satisfied(course.required, student) && return decision("prereq not met")
259 occupied = conflict(student, course, excluded)
260 occupied !== nothing && return decision("time conflict", occupied)
261 credits = student.credits + course.credits
262 if excluded != ""
263 credits -= registrar.courses[excluded].credits
264 end
265 credits > student.limit && return decision("credit limit", credits, student.limit)
266 decision("OK")
267end

269wait_position(registrar::Registrar, student::Student, course::Course) =
270 1 + count(id -> registrar.students[id].year >= student.year, course.waitlist)
271
272function request(registrar::Registrar, student::Student, course::Course)
273 result = eligibility(registrar, student, course)
274 result.kind != "OK" && return result
275 length(course.enrolled) < course.capacity && return decision("enroll")
276 length(student.waiting) >= 3 && return decision("waitlist limit")
277 decision("waitlist", wait_position(registrar, student, course))
278end
279
280function fail_request(result::Decision, student::Student, course::Course)
281 fail(join(vcat([result.kind, student.id, course.code], result.extra), " "))
282end

284function enroll!(student::Student, course::Course)
285 push!(course.enrolled, student.id)

290function withdraw!(student::Student, course::Course)
291 delete!(course.enrolled, student.id)

296function unlist!(student::Student, course::Course)
297 position = findfirst(==(student.id), course.waitlist)
298 position !== nothing && deleteat!(course.waitlist, position)
299 delete!(student.waiting, course.code)
300end

302function promotion!(registrar::Registrar)
303 for code in sort(collect(keys(registrar.courses)))
304 course = registrar.courses[code]
305 course.open || continue
306 for id in copy(course.waitlist)
307 length(course.enrolled) < course.capacity || break
308 student = registrar.students[id]
309 if conflict(student, course) !== nothing ||
310 student.credits + course.credits > student.limit
311 continue
312 end
313 unlist!(student, course)
314 enroll!(student, course)
315 record!(registrar, student, "promote " * code)
316 emit!(registrar, "PROMOTE", id, code, "enrolled",
317 "$(length(course.enrolled))/$(course.capacity)",
318 "credits", "$(student.credits)/$(student.limit)"; indent=2)
319 end
320 end
321end

323function cmd_course!(r::Registrar, args::Vector{String})
324 code, credits, capacity, timetable, text = args
325 haskey(r.courses, code) && fail("duplicate course " * code)
326 slots = timetable == "-" ? Slot[] :
327 [make_slot(String(value)) for value in split(timetable, ',')]
328 for i in 1:length(slots), j in (i+1):length(slots)
329 if overlaps(slots[i], slots[j])
330 fail("overlapping slots $(slots[i].original) $(slots[j].original)")
331 end
332 end
333 required = prerequisite(code, text)
334 course = Course(code, parse(Int, credits), parse(Int, capacity), slots, required,
335 true, Set{String}(), String[], Set{String}())
336 for (other_code, other) in r.courses
337 if any(overlaps(a, b) for a in slots for b in other.slots)
338 push!(course.conflicts, other_code)
339 push!(other.conflicts, code)
340 end
341 end
342 r.courses[code] = course
343 emit!(r, "COURSE", code, "credits", credits, "capacity", capacity,
344 "meetings", length(slots), "weekly", time_text(weekly(course)),
345 "prereq", canonical(required))
346end

348function cmd_student!(r::Registrar, args::Vector{String})
349 id, year, limit, text = args
350 haskey(r.students, id) && fail("duplicate student " * id)
351 records = Dict{String, Tuple{Char, Int}}()
352 if text != "-"
353 for entry in split(text, ',')
354 fields = split(entry, ':')
355 code = String(fields[1])
356 grade = only(fields[2])
357 haskey(GRADE_POINTS, grade) || fail("invalid grade $id $code $grade")
358 haskey(records, code) && fail("duplicate record $id $code")
359 records[code] = (grade, parse(Int, fields[3]))
360 end
361 end
362 student = Student(id, parse(Int, year), parse(Int, limit), records)
363 r.students[id] = student
364 emit!(r, "STUDENT", id, "year", year, "limit", limit,
365 "records", length(records), "earned", earned(student), "gpa", gpa(student))
366end

421function cmd_swap!(r::Registrar, args::Vector{String})
422 s = require_student(r, args[1])
423 source = require_course(r, args[2])
424 destination = require_course(r, args[3])
425 source.code in s.courses || fail("not enrolled $(s.id) $(source.code)")
426 source.code != destination.code || fail("same course " * source.code)
427 result = eligibility(r, s, destination, source.code)
428 result.kind == "OK" || fail_request(result, s, destination)
429 length(destination.enrolled) < destination.capacity ||
430 fail("course full " * destination.code)
431 withdraw!(s, source)
432 enroll!(s, destination)
433 record!(r, s, "swap $(source.code) $(destination.code)")
434 emit!(r, "SWAP", s.id, source.code, "->", destination.code, "credits",
435 "$(s.credits)/$(s.limit)")
436 promotion!(r)
437end

539function cmd_schedule!(r::Registrar, args::Vector{String})
540 s = require_student(r, args[1])
541 minutes = sum(weekly(r.courses[code]) for code in s.courses; init=0)
542 emit!(r, "SCHEDULE", s.id, "courses", length(s.courses),
543 "credits", s.credits, "weekly", time_text(minutes))
544 for (day, name) in enumerate(DAYS)
545 slots = student_slots(r, s, day)
546 isempty(slots) && continue
547 parts = String[name]
548 for (first, last, code) in slots
549 append!(parts, [time_text(first) * "-" * time_text(last), code])
550 end
551 emit!(r, join(parts, " "); indent=2)
552 end
553 undated = sort([code for code in s.courses if isempty(r.courses[code].slots)])
554 if !isempty(undated)
555 emit!(r, "TBA", join(undated, " "); indent=2)
556 end
557end

559function cmd_free!(r::Registrar, args::Vector{String})
560 s = require_student(r, args[1])
561 day = get(DAY_INDEX, args[2], 0)
562 day != 0 || fail("invalid day " * args[2])
563 occupied = Tuple{Int, Int}[]
564 for (first, last, _) in student_slots(r, s, day)
565 start, finish = max(480, first), min(1320, last)
566 if start < finish
567 push!(occupied, (start, finish))
568 end
569 end
570 free = Tuple{Int, Int}[]
571 cursor = 480
572 for (first, last) in occupied
573 if cursor < first
574 push!(free, (cursor, first))
575 end
576 cursor = max(cursor, last)
577 end
578 if cursor < 1320
579 push!(free, (cursor, 1320))
580 end
581 total = sum(last - first for (first, last) in free; init=0)
582 parts = ["FREE", s.id, args[2], "total", time_text(total)]
583 append!(parts, [time_text(a) * "-" * time_text(b) for (a, b) in free])
584 emit!(r, join(parts, " "))
585end

646function cmd_report!(r::Registrar, args::Vector{String})
647 all_courses = collect(values(r.courses))
648 emit!(r, "REPORT", "courses", length(r.courses), group_counts(all_courses))
649 departments = Dict{String, Vector{Course}}()
650 for code in sort(collect(keys(r.courses)))
651 c = r.courses[code]
652 emit!(r, code, status(c), "$(length(c.enrolled))/$(c.capacity)",
653 rate_text(length(c.enrolled), c.capacity),
654 "waitlist", length(c.waitlist); indent=2)
655 push!(get!(departments, department(c), Course[]), c)
656 end
657 for dept in sort(collect(keys(departments)))
658 selected = departments[dept]
659 emit!(r, "dept", dept, "courses", length(selected), group_counts(selected); indent=2)
660 end
661end

663const HANDLERS = Dict{String, Function}(
664 "COURSE" => cmd_course!, "STUDENT" => cmd_student!,
665 "ENROLL" => cmd_enroll!, "CHECK" => cmd_check!,
666 "OPTIONS" => cmd_options!, "DROP" => cmd_drop!,
667 "SWAP" => cmd_swap!, "CLOSE" => cmd_close!,
668 "OPEN" => cmd_open!, "CAPACITY" => cmd_capacity!,
669 "LIMIT" => cmd_limit!, "PREREQ" => cmd_prereq!,
670 "CANCEL" => cmd_cancel!, "ROSTER" => cmd_roster!,
671 "SCHEDULE" => cmd_schedule!, "FREE" => cmd_free!,

673 "HISTORY" => cmd_history!, "STUDENTS" => cmd_students!,
674 "REPORT" => cmd_report!
675)

677function execute!(registrar::Registrar, text::String)
678 registrar.number += 1
679 fields = String.(split(text))
680 command = first(fields)
681 try
682 HANDLERS[command](registrar, fields[2:end])
683 catch error
684 if error isa CommandError
685 emit!(registrar, "ERROR", error.reason)
686 else
687 rethrow()
688 end
689 end
690end
