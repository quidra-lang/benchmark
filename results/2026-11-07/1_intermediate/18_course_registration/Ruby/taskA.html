1DAYS = %w[MO TU WE TH FR SA SU].freeze
2POINTS = {'A' => 4, 'B' => 3, 'C' => 2, 'D' => 1, 'F' => 0}.freeze
3Slot = Struct.new(:day, :start, :finish, :raw)
4Grade = Struct.new(:code, :grade, :credits)
5Student = Struct.new(:id, :year, :limit, :grades, :taking, :history)
6Course = Struct.new(:code, :credits, :capacity, :slots, :prerequisite, :open, :enrolled, :waiting)
7class CommandError < StandardError; end
8
9def time_text(minutes)
10 format('%02d:%02d', minutes / 60, minutes % 60)
11end
12
13def overlap?(a, b)
14 a.day == b.day && a.start < b.finish && b.start < a.finish
15end
16
17def weekly(c)
18 c.slots.sum { |s| s.finish - s.start }
19end
20
21def earned(s)
22 s.grades.reject { |r| r.grade == 'F' }.sum(&:credits)
23end
24
25def passed(s, code)
26 s.grades.find { |r| r.code == code && %w[A B C].include?(r.grade) }
27end
28
29def rate(k, m)
30 m.positive? ? "#{k * 100 / m}%" : 'n/a'
31end
32
33def gpa(s)
34 return 'n/a' if s.grades.empty?
35 p = s.grades.sum { |r| POINTS[r.grade] * r.credits }
36 c = s.grades.sum(&:credits)
37 v = (200 * p + c) / (2 * c)
38 format('%d.%02d', v / 100, v % 100)
39end
40
41def combine(kind, children)
42 flat = children.flat_map { |c| c[:kind] == kind ? c[:children] : [c] }
43 flat.length == 1 ? flat.first : {kind: kind, children: flat}
44end
45
46def canonical(node)
47 return '-' if node.nil?
48 return node[:word] if node[:kind] == :leaf
49 node[:children].map do |c|
50 node[:kind] == :and && c[:kind] == :or ? "(#{canonical(c)})" : canonical(c)
51 end.join(node[:kind] == :and ? '&' : '|')
52end
53
54def contains?(node, code)
55 return false if node.nil?
56 node[:kind] == :leaf ? node[:word] == code : node[:children].any? { |c| contains?(c, code) }
57end
58
59def satisfied?(node, s)
60 return true if node.nil?
61 if node[:kind] == :leaf
62 return /\AY[1-9]\z/.match?(node[:word]) ? s.year >= node[:word][1].to_i : !passed(s, node[:word]).nil?
63 end
64 node[:kind] == :and ? node[:children].all? { |c| satisfied?(c, s) } : node[:children].any? { |c| satisfied?(c, s) }
65end

67class PrerequisiteParser
68 def initialize(code, text)
69 @code = code
70 @tokens = text.scan(/[A-Z0-9]+|[&|()]/)
71 @position = 0
72 @tokens.each do |token|
73 invalid unless %w[& | ( )].include?(token) || /\A(?:[A-Z]{2,4}[0-9]{3}|Y[1-9])\z/.match?(token)
74 end
75 end
76
77 def invalid
78 raise CommandError, "invalid prereq #{@code}"
79 end
80
81 def factor
82 token = @tokens[@position]
83 @position += 1
84 if token == '('
85 node = expression
86 invalid unless @tokens[@position] == ')'
87 @position += 1
88 node
89 else
90 invalid if token.nil? || %w[& | )].include?(token)
91 {kind: :leaf, word: token}
92 end
93 end
94
95 def term
96 nodes = [factor]
97 while @tokens[@position] == '&'
98 @position += 1
99 nodes << factor
100 end
101 combine(:and, nodes)
102 end
103
104 def expression
105 nodes = [term]
106 while @tokens[@position] == '|'
107 @position += 1
108 nodes << term
109 end
110 combine(:or, nodes)
111 end
112
113 def parse
114 node = expression
115 invalid unless @position == @tokens.length
116 raise CommandError, "self prereq #{@code}" if contains?(node, @code)
117 node
118 end
119end
120
121def prerequisite(code, text)
122 text == '-' ? nil : PrerequisiteParser.new(code, text).parse
123end
124
125def slots(text)
126 return [] if text == '-'
127 result = text.split(',').map do |raw|
128 day = raw[0, 2]
129 sh, sm, eh, em = [raw[2, 2], raw[4, 2], raw[7, 2], raw[9, 2]].map(&:to_i)
130 start = sh * 60 + sm
131 finish = eh * 60 + em
132 unless DAYS.include?(day) && sh <= 23 && sm <= 59 && eh <= 23 && em <= 59 && start < finish
133 raise CommandError, "invalid slot #{raw}"
134 end
135 Slot.new(day, start, finish, raw)
136 end
137 result.each_index do |i|
138 ((i + 1)...result.length).each do |j|
139 raise CommandError, "overlapping slots #{result[i].raw} #{result[j].raw}" if overlap?(result[i], result[j])
140 end
141 end
142 result
143end

145class Registrar
146 attr_reader :output
147
148 def initialize
149 @courses = {}
150 @students = {}
151 @command_number = 0
152 @output = []
153 end
154
155 def emit(indent, *parts)
156 @output << ' ' * indent + parts.join(' ')
157 end
158
159 def student(id)
160 raise CommandError, "unknown student #{id}" unless @students.key?(id)
161 @students[id]
162 end
163
164 def course(code)
165 raise CommandError, "unknown course #{code}" unless @courses.key?(code)
166 @courses[code]
167 end
168
169 def sorted_courses
170 @courses.values.sort_by(&:code)
171 end
172
173 def taking(s, except = nil)
174 s.taking.keys.reject { |code| code == except }.sort.map { |code| @courses[code] }
175 end
176
177 def credits(s, except = nil)
178 taking(s, except).sum(&:credits)
179 end
180
181 def waitlists(s)
182 sorted_courses.select { |c| c.waiting.include?(s.id) }
183 end
184
185 def conflict(s, c, except = nil)
186 taking(s, except).find { |other| other.slots.any? { |a| c.slots.any? { |b| overlap?(a, b) } } }
187 end
188
189 def eligible(s, c, except = nil)
190 return ['course closed', []] unless c.open
191 return ['already enrolled', []] if c.enrolled.key?(s.id)
192 return ['already waitlisted', []] if c.waiting.include?(s.id)
193 grade = passed(s, c.code)
194 return ['already completed', [grade.grade]] if grade
195 return ['prereq not met', []] unless satisfied?(c.prerequisite, s)
196 conflicting = conflict(s, c, except)
197 return ['time conflict', [conflicting.code]] if conflicting
198 total = credits(s, except) + c.credits
199 return ['credit limit', [total, s.limit]] if total > s.limit
200 ['OK', []]
201 end
202
203 def judgment(s, c)
204 result = eligible(s, c)
205 return result unless result[0] == 'OK'
206 return ['enroll', []] if c.enrolled.length < c.capacity
207 return ['waitlist limit', []] if waitlists(s).length >= 3
208 ['waitlist', [1 + c.waiting.count { |id| @students[id].year >= s.year }]]
209 end
210
211 def reject(s, c, result)
212 raise CommandError, [result[0], s.id, c.code, *result[1]].join(' ')
213 end
214
215 def history(s, event)
216 s.history << [@command_number, event]
217 end
218
219 def enroll(s, c)
220 c.enrolled[s.id] = true
221 s.taking[c.code] = true
222 end
223
224 def remove(s, c)
225 c.enrolled.delete(s.id)
226 s.taking.delete(c.code)
227 end
228
229 def enrollment_line(label, s, c, indent = 0)
230 emit(indent, label, s.id, c.code, 'enrolled', "#{c.enrolled.length}/#{c.capacity}", 'credits', "#{credits(s)}/#{s.limit}")
231 end
232
233 def promote
234 sorted_courses.each do |c|
235 next unless c.open
236 while c.enrolled.length < c.capacity
237 index = c.waiting.index do |id|
238 s = @students[id]
239 conflict(s, c).nil? && credits(s) + c.credits <= s.limit
240 end
241 break if index.nil?
242 s = @students[c.waiting.delete_at(index)]
243 enroll(s, c)
244 history(s, "promote #{c.code}")
245 enrollment_line('PROMOTE', s, c, 2)
246 end
247 end
248 end
249
250 def schedule(s)
251 taking = taking(s)
252 emit(0, 'SCHEDULE', s.id, 'courses', taking.length, 'credits', credits(s), 'weekly', time_text(taking.sum { |c| weekly(c) }))
253 DAYS.each do |day|
254 meetings = taking.flat_map { |c| c.slots.select { |slot| slot.day == day }.map { |slot| [c.code, slot] } }
255 meetings.sort_by! { |_, slot| slot.start }
256 emit(2, day, *meetings.flat_map { |code, slot| ["#{time_text(slot.start)}-#{time_text(slot.finish)}", code] }) unless meetings.empty?
257 end
258 tba = taking.select { |c| c.slots.empty? }.map(&:code)
259 emit(2, 'TBA', *tba) unless tba.empty?
260 end
261
262 def command(cmd, a)
263 case cmd
264 when 'COURSE'
265 raise CommandError, "duplicate course #{a[0]}" if @courses.key?(a[0])
266 meetings = slots(a[3])
267 tree = prerequisite(a[0], a[4])
268 c = Course.new(a[0], a[1].to_i, a[2].to_i, meetings, tree, true, {}, [])
269 @courses[c.code] = c
270 emit(0, 'COURSE', c.code, 'credits', c.credits, 'capacity', c.capacity,
271 'meetings', meetings.length, 'weekly', time_text(weekly(c)), 'prereq', canonical(tree))
272 when 'STUDENT'
273 raise CommandError, "duplicate student #{a[0]}" if @students.key?(a[0])
274 grades = []
275 seen = {}
276 (a[3] == '-' ? [] : a[3].split(',')).each do |token|
277 code, grade, credits = token.split(':')
278 raise CommandError, "invalid grade #{a[0]} #{code} #{grade}" unless POINTS.key?(grade)
279 raise CommandError, "duplicate record #{a[0]} #{code}" if seen.key?(code)
280 seen[code] = true
281 grades << Grade.new(code, grade, credits.to_i)
282 end
283 s = Student.new(a[0], a[1].to_i, a[2].to_i, grades, {}, [])
284 @students[s.id] = s
285 emit(0, 'STUDENT', s.id, 'year', s.year, 'limit', s.limit, 'records', grades.length, 'earned', earned(s), 'gpa', gpa(s))
286 when 'ENROLL', 'CHECK'
287 s = student(a[0])
288 c = course(a[1])
289 result = judgment(s, c)
290 if cmd == 'CHECK'
291 emit(0, 'CHECK', s.id, c.code, result[0], *result[1])
292 elsif result[0] == 'enroll'
293 enroll(s, c)
294 history(s, "enroll #{c.code}")
295 enrollment_line('ENROLL', s, c)
296 elsif result[0] == 'waitlist'
297 position = result[1][0]
298 c.waiting.insert(position - 1, s.id)
299 history(s, "waitlist #{c.code} #{position}")
300 emit(0, 'ENROLL', s.id, c.code, 'waitlist', position)
301 else
302 reject(s, c, result)
303 end
304 when 'OPTIONS'
305 s = student(a[0])
306 emit(0, 'OPTIONS', s.id, 'courses', @courses.length)
307 sorted_courses.each do |c|
308 result = judgment(s, c)
309 emit(2, c.code, result[0], *result[1])
310 end
311 when 'DROP'
312 s = student(a[0])
313 c = course(a[1])
314 position = c.waiting.index(s.id)
315 if c.enrolled.key?(s.id)
316 remove(s, c)
317 history(s, "drop #{c.code}")
318 enrollment_line('DROP', s, c)
319 elsif position
320 c.waiting.delete_at(position)
321 history(s, "drop #{c.code} waitlist")
322 emit(0, 'DROP', s.id, c.code, 'waitlist', 'removed')
323 else
324 raise CommandError, "not registered #{s.id} #{c.code}"
325 end
326 promote
327 when 'SWAP'
328 s = student(a[0])
329 old = course(a[1])
330 following = course(a[2])
331 raise CommandError, "not enrolled #{s.id} #{old.code}" unless old.enrolled.key?(s.id)
332 raise CommandError, "same course #{old.code}" if old.equal?(following)
333 result = eligible(s, following, old.code)
334 reject(s, following, result) unless result[0] == 'OK'
335 raise CommandError, "course full #{following.code}" if following.enrolled.length >= following.capacity
336 remove(s, old)
337 enroll(s, following)
338 history(s, "swap #{old.code} #{following.code}")
339 emit(0, 'SWAP', s.id, old.code, '->', following.code, 'credits', "#{credits(s)}/#{s.limit}")
340 promote
341 when 'CLOSE', 'OPEN'
342 c = course(a[0])
343 if c.open == (cmd == 'OPEN')
344 raise CommandError, "already #{cmd == 'CLOSE' ? 'closed' : 'open'} #{c.code}"
345 end
346 c.open = cmd == 'OPEN'
347 emit(0, cmd, c.code, 'enrolled', "#{c.enrolled.length}/#{c.capacity}", 'waitlist', c.waiting.length)
348 promote if cmd == 'OPEN'
349 when 'CAPACITY'
350 c = course(a[0])
351 capacity = a[1].to_i
352 raise CommandError, "capacity below enrollment #{c.code} #{c.enrolled.length}" if capacity < c.enrolled.length
353 old = c.capacity
354 c.capacity = capacity
355 emit(0, 'CAPACITY', c.code, old, '->', capacity)
356 promote
357 when 'LIMIT'
358 s = student(a[0])
359 limit = a[1].to_i
360 total = credits(s)
361 raise CommandError, "limit below enrolled #{s.id} #{total}" if limit < total
362 old = s.limit
363 s.limit = limit
364 emit(0, 'LIMIT', s.id, old, '->', limit)
365 promote
366 when 'PREREQ'
367 c = course(a[0])
368 tree = prerequisite(c.code, a[1])
369 c.prerequisite = tree
370 emit(0, 'PREREQ', c.code, canonical(tree))
371 c.waiting = c.waiting.select do |id|
372 s = @students[id]
373 if satisfied?(tree, s)
374 true
375 else
376 history(s, "unlist #{c.code}")
377 emit(2, 'UNLIST', s.id, c.code)
378 false
379 end
380 end
381 when 'CANCEL'
382 c = course(a[0])
383 emit(0, 'CANCEL', c.code, 'enrolled', c.enrolled.length, 'waitlist', c.waiting.length)
384 c.enrolled.each_key do |id|
385 s = @students[id]
386 s.taking.delete(c.code)
387 history(s, "cancel #{c.code}")
388 end
389 c.waiting.each { |id| history(@students[id], "cancel #{c.code} waitlist") }
390 @courses.delete(c.code)
391 promote
392 when 'ROSTER'
393 c = course(a[0])
394 emit(0, 'ROSTER', c.code, c.open ? 'open' : 'closed', 'enrolled', "#{c.enrolled.length}/#{c.capacity}", 'waitlist', c.waiting.length)
395 c.enrolled.keys.sort.each { |id| emit(2, 'enrolled', id, 'year', @students[id].year) }
396 c.waiting.each_with_index { |id, i| emit(2, 'waitlist', i + 1, id, 'year', @students[id].year) }
397 when 'SCHEDULE'
398 schedule(student(a[0]))
399 when 'FREE'
400 s = student(a[0])
401 raise CommandError, "invalid day #{a[1]}" unless DAYS.include?(a[1])
402 meetings = taking(s).flat_map(&:slots).select { |t| t.day == a[1] && t.start < 1320 && t.finish > 480 }.sort_by(&:start)
403 ranges = []
404 cursor = 480
405 meetings.each do |slot|
406 start = [slot.start, 480].max
407 ranges << [cursor, start] if start > cursor
408 cursor = [cursor, [slot.finish, 1320].min].max
409 end
410 ranges << [cursor, 1320] if cursor < 1320
411 emit(0, 'FREE', s.id, a[1], 'total', time_text(ranges.sum { |x, y| y - x }),
412 *ranges.map { |x, y| "#{time_text(x)}-#{time_text(y)}" })
413 when 'INFO'
414 c = course(a[0])
415 emit(0, 'INFO', c.code, c.open ? 'open' : 'closed', 'credits', c.credits, 'capacity', c.capacity, 'prereq', canonical(c.prerequisite))
416 c.slots.sort_by { |t| [DAYS.index(t.day), t.start] }.each do |slot|
417 emit(2, 'meet', slot.day, "#{time_text(slot.start)}-#{time_text(slot.finish)}")
418 end
419 emit(2, 'meet', 'TBA') if c.slots.empty?
420 required = sorted_courses.select { |other| contains?(other.prerequisite, c.code) }.map(&:code)
421 emit(2, 'required', 'by', *(required.empty? ? ['none'] : required))
422 when 'AUDIT'
423 s = student(a[0])
424 emit(0, 'AUDIT', s.id, 'year', s.year, 'limit', s.limit)
425 s.grades.sort_by(&:code).each { |r| emit(2, 'record', r.code, r.grade, r.credits) }
426 emit(2, 'earned', earned(s), 'gpa', gpa(s))
427 taking(s).each { |c| emit(2, 'enrolled', c.code, c.credits) }
428 emit(2, 'credits', credits(s), 'remaining', s.limit - credits(s))
429 waitlists(s).each { |c| emit(2, 'waitlist', c.code, c.waiting.index(s.id) + 1) }
430 when 'HISTORY'
431 s = student(a[0])
432 emit(0, 'HISTORY', s.id, 'events', s.history.length)
433 s.history.each { |command, event| emit(2, command, event) }
434 when 'STUDENTS'
435 emit(0, 'STUDENTS', @students.length)
436 @students.values.sort_by(&:id).each do |s|
437 emit(2, s.id, 'year', s.year, 'credits', "#{credits(s)}/#{s.limit}", 'courses', s.taking.length, 'waitlists', waitlists(s).length)
438 end
439 when 'REPORT'
440 courses = sorted_courses
441 enrolled = courses.sum { |c| c.enrolled.length }
442 capacity = courses.sum(&:capacity)
443 waiting = courses.sum { |c| c.waiting.length }
444 emit(0, 'REPORT', 'courses', courses.length, 'enrolled', "#{enrolled}/#{capacity}", 'rate', rate(enrolled, capacity), 'waitlisted', waiting)
445 courses.each do |c|
446 emit(2, c.code, c.open ? 'open' : 'closed', "#{c.enrolled.length}/#{c.capacity}", rate(c.enrolled.length, c.capacity), 'waitlist', c.waiting.length)
447 end
448 courses.group_by { |c| c.code[0...-3] }.sort.each do |department, group|
449 k = group.sum { |c| c.enrolled.length }
450 m = group.sum(&:capacity)
451 emit(2, 'dept', department, 'courses', group.length, 'enrolled', "#{k}/#{m}", 'rate', rate(k, m),
452 'waitlisted', group.sum { |c| c.waiting.length })
453 end
454 end
455 end
456
457 def execute(line)
458 @command_number += 1
459 cmd, *args = line.split
460 begin
461 command(cmd, args)
462 rescue CommandError => error
463 emit(0, 'ERROR', error.message)
464 end
465 end
466end

468registrar = Registrar.new
469STDIN.gets.to_i.times { registrar.execute(STDIN.gets.chomp) }
470STDOUT.write(registrar.output.join("\n") + "\n")
