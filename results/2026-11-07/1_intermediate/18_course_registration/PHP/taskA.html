23 && $first['start'] < $second['end']
24 && $second['start'] < $first['end'];

36function parseSlots(string $timetable): array
37{
38 if ($timetable === '-') {
39 return [];
40 }
41
42 $slots = [];
43 foreach (explode(',', $timetable) as $raw) {
44 $day = substr($raw, 0, 2);
45 $startHour = (int) substr($raw, 2, 2);
46 $startMinute = (int) substr($raw, 4, 2);
47 $endHour = (int) substr($raw, 7, 2);
48 $endMinute = (int) substr($raw, 9, 2);
49 $start = $startHour * 60 + $startMinute;
50 $end = $endHour * 60 + $endMinute;
51 if (!in_array($day, DAYS, true)
52 || $startHour > 23 || $startMinute > 59
53 || $endHour > 23 || $endMinute > 59 || $start >= $end) {
54 throw new CommandError("invalid slot $raw");
55 }
56 $slots[] = ['day' => $day, 'start' => $start, 'end' => $end, 'raw' => $raw];
57 }
58
59 for ($i = 0; $i < count($slots); ++$i) {
60 for ($j = $i + 1; $j < count($slots); ++$j) {
61 if (overlapping($slots[$i], $slots[$j])) {
62 throw new CommandError(
63 'overlapping slots ' . $slots[$i]['raw'] . ' ' . $slots[$j]['raw']
64 );
65 }
66 }
67 }
68 return $slots;
69}

71function combineNode(string $kind, array $nodes): array
72{
73 $children = [];
74 foreach ($nodes as $node) {
75 if ($node['kind'] === $kind) {
76 foreach ($node['children'] as $child) {
77 $children[] = $child;
78 }
79 } else {
80 $children[] = $node;
81 }
82 }
83 return count($children) === 1
84 ? $children[0]
85 : ['kind' => $kind, 'children' => $children, 'word' => ''];
86}

88function canonical(?array $node): string
89{
90 if ($node === null) {
91 return '-';
92 }
93 if ($node['kind'] === 'leaf') {
94 return $node['word'];
95 }
96 $parts = [];
97 foreach ($node['children'] as $child) {
98 $value = canonical($child);
99 if ($node['kind'] === 'and' && $child['kind'] === 'or') {
100 $value = '(' . $value . ')';
101 }
102 $parts[] = $value;
103 }
104 return implode($node['kind'] === 'and' ? '&' : '|', $parts);
105}
106
107function containsCondition(?array $node, string $code): bool
108{
109 if ($node === null) {
110 return false;
111 }
112 if ($node['kind'] === 'leaf') {
113 return $node['word'] === $code;
114 }
115 foreach ($node['children'] as $child) {
116 if (containsCondition($child, $code)) {
117 return true;
118 }
119 }
120 return false;
121}

123final class PrerequisiteParser
124{
125 private array $tokens;
126 private int $position = 0;
127 private string $course;
128
129 public function __construct(string $course, string $text)
130 {
131 $this->course = $course;
132 preg_match_all('/[A-Z0-9]+|[&|()]/', $text, $matches);
133 $this->tokens = $matches[0];
134 if (implode('', $this->tokens) !== $text) {
135 $this->invalid();
136 }
137 foreach ($this->tokens as $token) {
138 if (in_array($token, ['&', '|', '(', ')'], true)) {
139 continue;
140 }
141 if (!preg_match('/^(?:[A-Z]{2,4}[0-9]{3}|Y[1-9])$/D', $token)) {
142 $this->invalid();
143 }
144 }
145 }
146
147 private function invalid(): never
148 {
149 throw new CommandError('invalid prereq ' . $this->course);
150 }
151
152 private function peek(): ?string
153 {
154 return $this->tokens[$this->position] ?? null;
155 }
156
157 private function factor(): array
158 {
159 $token = $this->peek();
160 ++$this->position;
161 if ($token === '(') {
162 $inner = $this->expression();
163 if ($this->peek() !== ')') {
164 $this->invalid();
165 }
166 ++$this->position;
167 return $inner;
168 }
169 if ($token === null || in_array($token, ['&', '|', ')'], true)) {
170 $this->invalid();
171 }
172 return ['kind' => 'leaf', 'word' => $token, 'children' => []];
173 }
174
175 private function term(): array
176 {
177 $nodes = [$this->factor()];
178 while ($this->peek() === '&') {
179 ++$this->position;
180 $nodes[] = $this->factor();
181 }
182 return combineNode('and', $nodes);
183 }
184
185 private function expression(): array
186 {
187 $nodes = [$this->term()];
188 while ($this->peek() === '|') {
189 ++$this->position;
190 $nodes[] = $this->term();
191 }
192 return combineNode('or', $nodes);
193 }
194
195 public function parse(): array
196 {
197 $node = $this->expression();
198 if ($this->position !== count($this->tokens)) {
199 $this->invalid();
200 }
201 if (containsCondition($node, $this->course)) {
202 throw new CommandError('self prereq ' . $this->course);
203 }
204 return $node;
205 }
206}
207
208function prerequisite(string $course, string $text): ?array
209{
210 return $text === '-' ? null : (new PrerequisiteParser($course, $text))->parse();
211}

238final class Student
239{
240 public array $taking = [];
241 public array $history = [];
242
243 public function __construct(
244 public string $id,
245 public int $year,
246 public int $limit,
247 public array $grades
248 ) {}
249
250 public function passed(string $code): ?string
251 {
252 foreach ($this->grades as $record) {
253 if ($record['code'] === $code && in_array($record['grade'], ['A', 'B', 'C'], true)) {
254 return $record['grade'];
255 }
256 }
257 return null;
258 }
259
260 public function earned(): int
261 {
262 $credits = 0;
263 foreach ($this->grades as $record) {
264 if ($record['grade'] !== 'F') {
265 $credits += $record['credits'];
266 }
267 }
268 return $credits;
269 }
270
271 public function gpa(): string
272 {
273 if (!$this->grades) {
274 return 'n/a';
275 }
276 $points = 0;
277 $credits = 0;
278 foreach ($this->grades as $record) {
279 $points += GRADE_POINTS[$record['grade']] * $record['credits'];
280 $credits += $record['credits'];
281 }
282 $hundredths = intdiv(200 * $points + $credits, 2 * $credits);
283 return sprintf('%d.%02d', intdiv($hundredths, 100), $hundredths % 100);
284 }
285}
286
287function satisfies(?array $node, Student $student): bool
288{
289 if ($node === null) {
290 return true;
291 }
292 if ($node['kind'] === 'leaf') {
293 $word = $node['word'];
294 if (preg_match('/^Y[1-9]$/D', $word)) {
295 return $student->year >= (int) substr($word, 1);
296 }
297 return $student->passed($word) !== null;
298 }
299 if ($node['kind'] === 'and') {
300 foreach ($node['children'] as $child) {
301 if (!satisfies($child, $student)) {
302 return false;
303 }
304 }
305 return true;
306 }
307 foreach ($node['children'] as $child) {
308 if (satisfies($child, $student)) {
309 return true;
310 }
311 }
312 return false;
313}

316{

318 private array $students = [];
319 private array $output = [];

353 sort($codes, SORT_STRING);
354 $result = [];
355 foreach ($codes as $code) {

357 $result[] = $this->courses[$code];

365 $result = 0;
366 foreach ($this->taking($student, $excluded) as $course) {
367 $result += $course->credits;

373 {
374 $result = [];
375 foreach ($this->sortedCourses() as $course) {
376 if (in_array($student->id, $course->waiting, true)) {
377 $result[] = $course;
378 }
379 }
380 return $result;
381 }

383 private function conflict(Student $student, Course $target, ?string $excluded = null): ?Course
384 {
385 foreach ($this->taking($student, $excluded) as $course) {
386 foreach ($course->slots as $first) {
387 foreach ($target->slots as $second) {
388 if (overlapping($first, $second)) {
389 return $course;
390 }
391 }
392 }
393 }
394 return null;
395 }
396
397 private function eligible(Student $student, Course $course, ?string $excluded = null): array
398 {
399 if (!$course->open) {
400 return ['course closed', []];
401 }
402 if (isset($course->enrolled[$student->id])) {
403 return ['already enrolled', []];
404 }
405 if (in_array($student->id, $course->waiting, true)) {
406 return ['already waitlisted', []];
407 }
408 $grade = $student->passed($course->code);
409 if ($grade !== null) {
410 return ['already completed', [$grade]];
411 }
412 if (!satisfies($course->prerequisite, $student)) {
413 return ['prereq not met', []];
414 }
415 $conflicting = $this->conflict($student, $course, $excluded);
416 if ($conflicting !== null) {
417 return ['time conflict', [$conflicting->code]];
418 }
419 $credits = $this->credits($student, $excluded) + $course->credits;
420 if ($credits > $student->limit) {
421 return ['credit limit', [$credits, $student->limit]];
422 }
423 return ['OK', []];
424 }
425
426 private function waitPosition(Student $student, Course $course): int
427 {
428 $count = 0;
429 foreach ($course->waiting as $id) {
430 if ($this->students[$id]->year >= $student->year) {
431 ++$count;
432 }
433 }
434 return $count + 1;
435 }

437 private function judgment(Student $student, Course $course): array
438 {
439 [$word, $extra] = $this->eligible($student, $course);
440 if ($word !== 'OK') {
441 return [$word, $extra];
442 }
443 if (count($course->enrolled) < $course->capacity) {
444 return ['enroll', []];
445 }
446 if (count($this->waitlists($student)) >= 3) {
447 return ['waitlist limit', []];
448 }
449 return ['waitlist', [$this->waitPosition($student, $course)]];
450 }

459 [$word, $extra] = $judgment;

461 throw new CommandError(implode(' ', $parts));

475 private function enrollmentLine(string $label, Student $student, Course $course, int $indent = 0): void
476 {
477 $this->emit($indent, $label, $student->id, $course->code,
478 'enrolled', count($course->enrolled) . '/' . $course->capacity,
479 'credits', $this->credits($student) . '/' . $student->limit);
480 }

482 private function promote(): void
483 {
484 foreach ($this->sortedCourses() as $course) {
485 if (!$course->open) {
486 continue;
487 }
488 while (count($course->enrolled) < $course->capacity) {
489 $selected = null;
490 foreach ($course->waiting as $index => $id) {
491 $student = $this->students[$id];
492 if ($this->conflict($student, $course) === null
493 && $this->credits($student) + $course->credits <= $student->limit) {
494 $selected = $index;
495 break;
496 }
497 }
498 if ($selected === null) {
499 break;
500 }
501 $id = $course->waiting[$selected];
502 array_splice($course->waiting, $selected, 1);
503 $student = $this->students[$id];
504 $this->enroll($student, $course);
505 $this->record($student, 'promote ' . $course->code);
506 $this->enrollmentLine('PROMOTE', $student, $course, 2);
507 }
508 }
509 }

511 private function schedule(Student $student): void
512 {
513 $taking = $this->taking($student);
514 $weekly = 0;
515 foreach ($taking as $course) {
516 $weekly += weeklyTime($course);
517 }
518 $this->emit(0, 'SCHEDULE', $student->id, 'courses', count($taking),
519 'credits', $this->credits($student), 'weekly', timeText($weekly));
520
521 foreach (DAYS as $day) {
522 $meetings = [];
523 foreach ($taking as $course) {
524 foreach ($course->slots as $slot) {
525 if ($slot['day'] === $day) {
526 $meetings[] = ['slot' => $slot, 'code' => $course->code];
527 }
528 }
529 }
530 usort($meetings, static fn(array $a, array $b): int =>
531 $a['slot']['start'] <=> $b['slot']['start']);
532 if (!$meetings) {
533 continue;
534 }
535 $parts = [$day];
536 foreach ($meetings as $meeting) {
537 $slot = $meeting['slot'];
538 $parts[] = timeText($slot['start']) . '-' . timeText($slot['end']);
539 $parts[] = $meeting['code'];
540 }
541 $this->emit(2, ...$parts);
542 }
543 $tba = [];
544 foreach ($taking as $course) {
545 if (!$course->slots) {
546 $tba[] = $course->code;
547 }
548 }
549 if ($tba) {
550 $this->emit(2, 'TBA', ...$tba);
551 }
552 }

554 private function command(string $command, array $a): void
555 {
556 switch ($command) {
557 case 'COURSE':
558 if (isset($this->courses[$a[0]])) {
559 throw new CommandError('duplicate course ' . $a[0]);
560 }
561 $slots = parseSlots($a[3]);
562 $tree = prerequisite($a[0], $a[4]);
563 $course = new Course($a[0], (int) $a[1], (int) $a[2], $slots, $tree);
564 $this->courses[$course->code] = $course;
565 $this->emit(0, 'COURSE', $course->code, 'credits', $course->credits,
566 'capacity', $course->capacity, 'meetings', count($slots),
567 'weekly', timeText(weeklyTime($course)), 'prereq', canonical($tree));
568 return;
569
570 case 'STUDENT':
571 if (isset($this->students[$a[0]])) {
572 throw new CommandError('duplicate student ' . $a[0]);
573 }
574 $grades = [];
575 $seen = [];
576 foreach ($a[3] === '-' ? [] : explode(',', $a[3]) as $token) {
577 [$code, $grade, $credits] = explode(':', $token);
578 if (!array_key_exists($grade, GRADE_POINTS)) {
579 throw new CommandError("invalid grade $a[0] $code $grade");
580 }
581 if (isset($seen[$code])) {
582 throw new CommandError("duplicate record $a[0] $code");
583 }
584 $seen[$code] = true;
585 $grades[] = ['code' => $code, 'grade' => $grade, 'credits' => (int) $credits];
586 }
587 $student = new Student($a[0], (int) $a[1], (int) $a[2], $grades);
588 $this->students[$student->id] = $student;
589 $this->emit(0, 'STUDENT', $student->id, 'year', $student->year,
590 'limit', $student->limit, 'records', count($grades),
591 'earned', $student->earned(), 'gpa', $student->gpa());
592 return;
593
594 case 'ENROLL':
595 case 'CHECK':
596 $student = $this->student($a[0]);
597 $course = $this->course($a[1]);
598 $decision = $this->judgment($student, $course);
599 [$word, $extra] = $decision;
600 if ($command === 'CHECK') {
601 $this->emit(0, 'CHECK', $student->id, $course->code, $word, ...$extra);
602 } elseif ($word === 'enroll') {
603 $this->enroll($student, $course);
604 $this->record($student, 'enroll ' . $course->code);
605 $this->enrollmentLine('ENROLL', $student, $course);
606 } elseif ($word === 'waitlist') {
607 $position = (int) $extra[0];
608 array_splice($course->waiting, $position - 1, 0, [$student->id]);
609 $this->record($student, 'waitlist ' . $course->code . ' ' . $position);
610 $this->emit(0, 'ENROLL', $student->id, $course->code, 'waitlist', $position);
611 } else {
612 $this->reject($student, $course, $decision);
613 }
614 return;
615
616 case 'OPTIONS':
617 $student = $this->student($a[0]);
618 $this->emit(0, 'OPTIONS', $student->id, 'courses', count($this->courses));
619 foreach ($this->sortedCourses() as $course) {
620 [$word, $extra] = $this->judgment($student, $course);
621 $this->emit(2, $course->code, $word, ...$extra);
622 }
623 return;
624
625 case 'DROP':
626 $student = $this->student($a[0]);
627 $course = $this->course($a[1]);
628 $position = array_search($student->id, $course->waiting, true);
629 if (isset($course->enrolled[$student->id])) {
630 $this->withdraw($student, $course);
631 $this->record($student, 'drop ' . $course->code);
632 $this->enrollmentLine('DROP', $student, $course);
633 } elseif ($position !== false) {
634 array_splice($course->waiting, $position, 1);
635 $this->record($student, 'drop ' . $course->code . ' waitlist');
636 $this->emit(0, 'DROP', $student->id, $course->code, 'waitlist', 'removed');
637 } else {
638 throw new CommandError("not registered $student->id $course->code");
639 }
640 $this->promote();
641 return;
642
643 case 'SWAP':
644 $student = $this->student($a[0]);
645 $old = $this->course($a[1]);
646 $next = $this->course($a[2]);
647 if (!isset($old->enrolled[$student->id])) {
648 throw new CommandError("not enrolled $student->id $old->code");
649 }
650 if ($old === $next) {
651 throw new CommandError('same course ' . $old->code);
652 }
653 $decision = $this->eligible($student, $next, $old->code);
654 if ($decision[0] !== 'OK') {
655 $this->reject($student, $next, $decision);
656 }
657 if (count($next->enrolled) >= $next->capacity) {
658 throw new CommandError('course full ' . $next->code);
659 }
660 $this->withdraw($student, $old);
661 $this->enroll($student, $next);
662 $this->record($student, 'swap ' . $old->code . ' ' . $next->code);
663 $this->emit(0, 'SWAP', $student->id, $old->code, '->', $next->code,
664 'credits', $this->credits($student) . '/' . $student->limit);
665 $this->promote();
666 return;
667
668 case 'CLOSE':
669 case 'OPEN':
670 $course = $this->course($a[0]);
671 if ($course->open === ($command === 'OPEN')) {
672 throw new CommandError('already ' . ($command === 'OPEN' ? 'open ' : 'closed ') . $course->code);
673 }
674 $course->open = $command === 'OPEN';
675 $this->emit(0, $command, $course->code, 'enrolled',
676 count($course->enrolled) . '/' . $course->capacity,
677 'waitlist', count($course->waiting));
678 if ($command === 'OPEN') {
679 $this->promote();
680 }
681 return;
682
683 case 'CAPACITY':
684 $course = $this->course($a[0]);
685 $capacity = (int) $a[1];
686 if ($capacity < count($course->enrolled)) {
687 throw new CommandError(
688 'capacity below enrollment ' . $course->code . ' ' . count($course->enrolled)
689 );
690 }
691 $previous = $course->capacity;
692 $course->capacity = $capacity;
693 $this->emit(0, 'CAPACITY', $course->code, $previous, '->', $capacity);
694 $this->promote();
695 return;
696
697 case 'LIMIT':
698 $student = $this->student($a[0]);
699 $limit = (int) $a[1];
700 $credits = $this->credits($student);
701 if ($limit < $credits) {
702 throw new CommandError("limit below enrolled $student->id $credits");
703 }
704 $previous = $student->limit;
705 $student->limit = $limit;
706 $this->emit(0, 'LIMIT', $student->id, $previous, '->', $limit);
707 $this->promote();
708 return;
709
710 case 'PREREQ':
711 $course = $this->course($a[0]);
712 $tree = prerequisite($course->code, $a[1]);
713 $course->prerequisite = $tree;
714 $this->emit(0, 'PREREQ', $course->code, canonical($tree));
715 $remaining = [];
716 foreach ($course->waiting as $id) {
717 $student = $this->students[$id];
718 if (satisfies($tree, $student)) {
719 $remaining[] = $id;
720 } else {
721 $this->record($student, 'unlist ' . $course->code);
722 $this->emit(2, 'UNLIST', $student->id, $course->code);
723 }
724 }
725 $course->waiting = $remaining;
726 return;
727
728 case 'CANCEL':
729 $course = $this->course($a[0]);
730 $this->emit(0, 'CANCEL', $course->code,
731 'enrolled', count($course->enrolled), 'waitlist', count($course->waiting));
732 foreach (array_keys($course->enrolled) as $id) {
733 $student = $this->students[$id];
734 unset($student->taking[$course->code]);
735 $this->record($student, 'cancel ' . $course->code);
736 }
737 foreach ($course->waiting as $id) {
738 $this->record($this->students[$id], 'cancel ' . $course->code . ' waitlist');
739 }
740 unset($this->courses[$course->code]);
741 $this->promote();
742 return;
743
744 case 'ROSTER':
745 $course = $this->course($a[0]);
746 $this->emit(0, 'ROSTER', $course->code, $course->status(),
747 'enrolled', count($course->enrolled) . '/' . $course->capacity,
748 'waitlist', count($course->waiting));
749 $ids = array_keys($course->enrolled);
750 sort($ids, SORT_STRING);
751 foreach ($ids as $id) {
752 $this->emit(2, 'enrolled', $id, 'year', $this->students[$id]->year);
753 }
754 foreach ($course->waiting as $position => $id) {
755 $this->emit(2, 'waitlist', $position + 1, $id, 'year', $this->students[$id]->year);
756 }
757 return;
758
759 case 'SCHEDULE':
760 $this->schedule($this->student($a[0]));
761 return;
762
763 case 'FREE':
764 $student = $this->student($a[0]);
765 $day = $a[1];
766 if (!in_array($day, DAYS, true)) {
767 throw new CommandError('invalid day ' . $day);
768 }
769 $meetings = [];
770 foreach ($this->taking($student) as $course) {
771 foreach ($course->slots as $slot) {
772 if ($slot['day'] === $day && $slot['start'] < 1320 && $slot['end'] > 480) {
773 $meetings[] = $slot;
774 }
775 }
776 }
777 usort($meetings, static fn(array $a, array $b): int =>
778 $a['start'] <=> $b['start']);
779 $ranges = [];
780 $cursor = 480;
781 foreach ($meetings as $slot) {
782 $start = max($slot['start'], 480);
783 if ($start > $cursor) {
784 $ranges[] = [$cursor, $start];
785 }
786 $cursor = max($cursor, min($slot['end'], 1320));
787 }
788 if ($cursor < 1320) {
789 $ranges[] = [$cursor, 1320];
790 }
791 $minutes = 0;
792 $parts = ['FREE', $student->id, $day, 'total'];
793 foreach ($ranges as [$first, $last]) {
794 $minutes += $last - $first;
795 }
796 $parts[] = timeText($minutes);
797 foreach ($ranges as [$first, $last]) {
798 $parts[] = timeText($first) . '-' . timeText($last);
799 }
800 $this->emit(0, ...$parts);
801 return;
802
803 case 'INFO':
804 $course = $this->course($a[0]);
805 $this->emit(0, 'INFO', $course->code, $course->status(),
806 'credits', $course->credits, 'capacity', $course->capacity,
807 'prereq', canonical($course->prerequisite));
808 $slots = $course->slots;
809 usort($slots, static fn(array $a, array $b): int =>
810 (array_search($a['day'], DAYS, true) <=> array_search($b['day'], DAYS, true))
811 ?: ($a['start'] <=> $b['start']));
812 if (!$slots) {
813 $this->emit(2, 'meet', 'TBA');
814 } else {
815 foreach ($slots as $slot) {
816 $this->emit(2, 'meet', $slot['day'],
817 timeText($slot['start']) . '-' . timeText($slot['end']));
818 }
819 }
820 $required = [];
821 foreach ($this->sortedCourses() as $other) {
822 if (containsCondition($other->prerequisite, $course->code)) {
823 $required[] = $other->code;
824 }
825 }
826 $this->emit(2, 'required', 'by', ...($required ?: ['none']));
827 return;
828
829 case 'AUDIT':
830 $student = $this->student($a[0]);
831 $this->emit(0, 'AUDIT', $student->id, 'year', $student->year, 'limit', $student->limit);
832 $grades = $student->grades;
833 usort($grades, static fn(array $a, array $b): int => strcmp($a['code'], $b['code']));
834 foreach ($grades as $grade) {
835 $this->emit(2, 'record', $grade['code'], $grade['grade'], $grade['credits']);
836 }
837 $this->emit(2, 'earned', $student->earned(), 'gpa', $student->gpa());
838 foreach ($this->taking($student) as $course) {
839 $this->emit(2, 'enrolled', $course->code, $course->credits);
840 }
841 $credits = $this->credits($student);
842 $this->emit(2, 'credits', $credits, 'remaining', $student->limit - $credits);
843 foreach ($this->waitlists($student) as $course) {
844 $position = array_search($student->id, $course->waiting, true) + 1;
845 $this->emit(2, 'waitlist', $course->code, $position);
846 }
847 return;
848
849 case 'HISTORY':
850 $student = $this->student($a[0]);
851 $this->emit(0, 'HISTORY', $student->id, 'events', count($student->history));
852 foreach ($student->history as [$number, $event]) {
853 $this->emit(2, $number, $event);
854 }
855 return;
856
857 case 'STUDENTS':
858 $students = $this->students;
859 ksort($students, SORT_STRING);
860 $this->emit(0, 'STUDENTS', count($students));
861 foreach ($students as $student) {
862 $this->emit(2, $student->id, 'year', $student->year,
863 'credits', $this->credits($student) . '/' . $student->limit,
864 'courses', count($student->taking),
865 'waitlists', count($this->waitlists($student)));
866 }
867 return;
868
869 case 'REPORT':
870 $courses = $this->sortedCourses();
871 $enrolled = 0;
872 $capacity = 0;
873 $waiting = 0;
874 $departments = [];
875 foreach ($courses as $course) {
876 $enrolled += count($course->enrolled);
877 $capacity += $course->capacity;
878 $waiting += count($course->waiting);
879 $departments[$course->department()][] = $course;
880 }
881 $this->emit(0, 'REPORT', 'courses', count($courses), 'enrolled',
882 $enrolled . '/' . $capacity, 'rate', rateText($enrolled, $capacity),
883 'waitlisted', $waiting);
884 foreach ($courses as $course) {
885 $count = count($course->enrolled);
886 $this->emit(2, $course->code, $course->status(),
887 $count . '/' . $course->capacity, rateText($count, $course->capacity),
888 'waitlist', count($course->waiting));
889 }
890 ksort($departments, SORT_STRING);
891 foreach ($departments as $department => $group) {
892 $count = 0;
893 $seats = 0;
894 $queued = 0;
895 foreach ($group as $course) {
896 $count += count($course->enrolled);
897 $seats += $course->capacity;
898 $queued += count($course->waiting);
899 }
900 $this->emit(2, 'dept', $department, 'courses', count($group),
901 'enrolled', $count . '/' . $seats,
902 'rate', rateText($count, $seats), 'waitlisted', $queued);
903 }
904 return;
905 }
906 }

908 public function execute(string $line): void
909 {
910 ++$this->commandNumber;
911 $tokens = explode(' ', $line);
912 $command = array_shift($tokens);
913 try {
914 $this->command($command, $tokens);
915 } catch (CommandError $error) {
916 $this->emit(0, 'ERROR', $error->getMessage());
917 }
918 }
