37 private record Slot(String day, int start, int end, String raw) {
38 boolean overlaps(Slot other) {
39 return day.equals(other.day) && start < other.end && other.start < end;
40 }
41 }

43 private static List<Slot> parseSlots(String timetable) {
44 if (timetable.equals("-")) {
45 return new ArrayList<>();
46 }
47 List<Slot> slots = new ArrayList<>();
48 for (String raw : timetable.split(",")) {
49 String day = raw.substring(0, 2);
50 int sh = Integer.parseInt(raw.substring(2, 4));
51 int sm = Integer.parseInt(raw.substring(4, 6));
52 int eh = Integer.parseInt(raw.substring(7, 9));
53 int em = Integer.parseInt(raw.substring(9, 11));
54 int start = 60 * sh + sm;
55 int end = 60 * eh + em;
56 if (!DAYS.contains(day) || sh > 23 || sm > 59 ||
57 eh > 23 || em > 59 || start >= end) {
58 throw new CommandError("invalid slot " + raw);
59 }
60 slots.add(new Slot(day, start, end, raw));
61 }
62 for (int i = 0; i < slots.size(); ++i) {
63 for (int j = i + 1; j < slots.size(); ++j) {
64 if (slots.get(i).overlaps(slots.get(j))) {
65 throw new CommandError(
66 "overlapping slots " + slots.get(i).raw() + " " + slots.get(j).raw()
67 );
68 }
69 }
70 }
71 return slots;
72 }

74 private static final class Expression {
75 final String kind;
76 final String word;
77 final List<Expression> children;
78
79 Expression(String kind, String word, List<Expression> children) {
80 this.kind = kind;
81 this.word = word;
82 this.children = children;
83 }
84
85 static Expression word(String word) {
86 return new Expression("word", word, List.of());
87 }
88
89 static Expression combine(String kind, List<Expression> nodes) {
90 List<Expression> children = new ArrayList<>();
91 for (Expression node : nodes) {
92 if (node.kind.equals(kind)) {
93 children.addAll(node.children);
94 } else {
95 children.add(node);
96 }
97 }
98 return children.size() == 1 ? children.get(0)
99 : new Expression(kind, "", children);
100 }
101
102 String canonical() {
103 if (kind.equals("word")) {
104 return word;
105 }
106 StringJoiner joiner = new StringJoiner(kind.equals("and") ? "&" : "|");
107 for (Expression child : children) {
108 String text = child.canonical();
109 if (kind.equals("and") && child.kind.equals("or")) {
110 text = "(" + text + ")";
111 }
112 joiner.add(text);
113 }
114 return joiner.toString();
115 }
116
117 boolean contains(String code) {
118 if (kind.equals("word")) {
119 return word.equals(code);
120 }
121 for (Expression child : children) {
122 if (child.contains(code)) {
123 return true;
124 }
125 }
126 return false;
127 }
128
129 boolean satisfied(Student student) {
130 if (kind.equals("word")) {
131 return word.length() == 2 && word.charAt(0) == 'Y'
132 ? student.year >= word.charAt(1) - '0'
133 : student.passed(word) != null;
134 }
135 if (kind.equals("and")) {
136 for (Expression child : children) {
137 if (!child.satisfied(student)) {
138 return false;
139 }
140 }
141 return true;
142 }
143 for (Expression child : children) {
144 if (child.satisfied(student)) {
145 return true;
146 }
147 }
148 return false;
149 }
150 }

152 private static final class PrerequisiteParser {
153 private static final Pattern TOKEN = Pattern.compile("[A-Z0-9]+|[&|()]");
154 private static final Pattern WORD =
155 Pattern.compile("(?:[A-Z]{2,4}[0-9]{3}|Y[1-9])");
156
157 private final String course;
158 private final List<String> tokens = new ArrayList<>();
159 private int position;
160
161 PrerequisiteParser(String course, String text) {
162 this.course = course;
163 Matcher matcher = TOKEN.matcher(text);
164 while (matcher.find()) {
165 tokens.add(matcher.group());
166 }
167 if (!String.join("", tokens).equals(text)) {
168 invalid();
169 }
170 for (String token : tokens) {
171 if (List.of("&", "|", "(", ")").contains(token)) {
172 continue;
173 }
174 if (!WORD.matcher(token).matches()) {
175 invalid();
176 }
177 }
178 }
179
180 private void invalid() {
181 throw new CommandError("invalid prereq " + course);
182 }
183
184 private String peek() {
185 return position < tokens.size() ? tokens.get(position) : null;
186 }
187
188 private Expression factor() {
189 String token = peek();
190 ++position;
191 if ("(".equals(token)) {
192 Expression node = expression();
193 if (!")".equals(peek())) {
194 invalid();
195 }
196 ++position;
197 return node;
198 }
199 if (token == null || List.of("&", "|", ")").contains(token)) {
200 invalid();
201 }
202 return Expression.word(token);
203 }
204
205 private Expression term() {
206 List<Expression> nodes = new ArrayList<>();
207 nodes.add(factor());
208 while ("&".equals(peek())) {
209 ++position;
210 nodes.add(factor());
211 }
212 return Expression.combine("and", nodes);
213 }
214
215 private Expression expression() {
216 List<Expression> nodes = new ArrayList<>();
217 nodes.add(term());
218 while ("|".equals(peek())) {
219 ++position;
220 nodes.add(term());
221 }
222 return Expression.combine("or", nodes);
223 }
224
225 Expression parse() {
226 Expression node = expression();
227 if (position != tokens.size()) {
228 invalid();
229 }
230 if (node.contains(course)) {
231 throw new CommandError("self prereq " + course);
232 }
233 return node;
234 }
235
236 static Expression from(String course, String text) {
237 return text.equals("-") ? null : new PrerequisiteParser(course, text).parse();
238 }
239 }

298 String passed(String code) {
299 for (Grade grade : grades) {
300 if (grade.course().equals(code) &&
301 List.of("A", "B", "C").contains(grade.grade())) {
302 return grade.grade();
303 }
304 }
305 return null;
306 }
307
308 int earned() {
309 int total = 0;
310 for (Grade grade : grades) {
311 if (!grade.grade().equals("F")) {
312 total += grade.credits();
313 }
314 }
315 return total;
316 }
317
318 String gpa() {
319 if (grades.isEmpty()) {
320 return "n/a";
321 }
322 int points = 0;
323 int credits = 0;
324 for (Grade grade : grades) {
325 points += GRADE_POINTS.get(grade.grade()) * grade.credits();
326 credits += grade.credits();
327 }
328 int hundredths = (200 * points + credits) / (2 * credits);
329 return String.format("%d.%02d", hundredths / 100, hundredths % 100);
330 }

407 private Course conflict(Student student, Course target, String excluded) {
408 for (Course course : taking(student, excluded)) {
409 for (Slot a : course.slots) {
410 for (Slot b : target.slots) {

412 return course;

420 private Decision eligible(Student student, Course course, String excluded) {
421 if (!course.open) return new Decision("course closed");
422 if (course.enrolled.contains(student.id)) return new Decision("already enrolled");
423 if (course.waiting.contains(student.id)) return new Decision("already waitlisted");
424
425 String grade = student.passed(course.code);
426 if (grade != null) return new Decision("already completed", grade);
427 if (course.prerequisite != null && !course.prerequisite.satisfied(student))
428 return new Decision("prereq not met");
429
430 Course conflict = conflict(student, course, excluded);
431 if (conflict != null) return new Decision("time conflict", conflict.code);
432
433 int requested = credits(student, excluded) + course.credits;
434 if (requested > student.limit)
435 return new Decision(
436 "credit limit", Integer.toString(requested), Integer.toString(student.limit)
437 );
438 return new Decision("OK");
439 }

441 private int waitPosition(Student student, Course course) {
442 int position = 1;
443 for (String id : course.waiting) {
444 if (students.get(id).year >= student.year) {
445 ++position;
446 }
447 }
448 return position;

451 private Decision judgment(Student student, Course course) {
452 Decision decision = eligible(student, course, null);
453 if (!decision.word.equals("OK")) return decision;
454 if (course.enrolled.size() < course.capacity) return new Decision("enroll");
455 if (waitlists(student).size() >= 3) return new Decision("waitlist limit");
456 return new Decision("waitlist", Integer.toString(waitPosition(student, course)));
457 }

486 private void promote() {
487 for (Course course : courses.values()) {
488 if (!course.open) continue;
489 while (course.enrolled.size() < course.capacity) {
490 int selected = -1;
491 for (int i = 0; i < course.waiting.size(); ++i) {
492 Student student = students.get(course.waiting.get(i));
493 if (conflict(student, course, null) == null &&
494 credits(student, null) + course.credits <= student.limit) {
495 selected = i;
496 break;
497 }
498 }
499 if (selected < 0) break;
500 Student student = students.get(course.waiting.remove(selected));
501 enroll(student, course);
502 record(student, "promote " + course.code);
503 enrollmentLine("PROMOTE", student, course, 2);
504 }
505 }

508 private void schedule(Student student) {
509 List<Course> taking = taking(student, null);
510 int weekly = taking.stream().mapToInt(Course::weekly).sum();
511 emit(0, "SCHEDULE", student.id, "courses", taking.size(), "credits",
512 credits(student, null), "weekly", time(weekly));
513
514 for (String day : DAYS) {
515 List<Meeting> meetings = new ArrayList<>();
516 for (Course course : taking) {
517 for (Slot slot : course.slots) {
518 if (slot.day().equals(day)) {
519 meetings.add(new Meeting(course.code, slot));
520 }
521 }
522 }
523 meetings.sort(Comparator.comparingInt(m -> m.slot().start()));
524 if (!meetings.isEmpty()) {
525 List<Object> line = new ArrayList<>();
526 line.add(day);
527 for (Meeting meeting : meetings) {
528 line.add(time(meeting.slot().start()) + "-" + time(meeting.slot().end()));
529 line.add(meeting.code());
530 }
531 emit(2, line.toArray());
532 }
533 }
534
535 List<Object> tba = new ArrayList<>();
536 tba.add("TBA");
537 for (Course course : taking) {
538 if (course.slots.isEmpty()) {
539 tba.add(course.code);
540 }
541 }
542 if (tba.size() > 1) {
543 emit(2, tba.toArray());
544 }
545 }

547 private void command(String command, String[] a) {
548 switch (command) {
549 case "COURSE" -> {
550 if (courses.containsKey(a[0]))
551 throw new CommandError("duplicate course " + a[0]);
552 List<Slot> slots = parseSlots(a[3]);
553 Expression prerequisite = PrerequisiteParser.from(a[0], a[4]);
554 Course course = new Course(
555 a[0], Integer.parseInt(a[1]), Integer.parseInt(a[2]), slots, prerequisite
556 );
557 courses.put(course.code, course);
558 emit(0, "COURSE", course.code, "credits", course.credits,
559 "capacity", course.capacity, "meetings", slots.size(),
560 "weekly", time(course.weekly()), "prereq", course.prerequisiteText());
561 }
562
563 case "STUDENT" -> {
564 if (students.containsKey(a[0]))
565 throw new CommandError("duplicate student " + a[0]);
566 List<Grade> grades = new ArrayList<>();
567 Set<String> seen = new HashSet<>();
568 if (!a[3].equals("-")) {
569 for (String raw : a[3].split(",")) {
570 String[] entry = raw.split(":");
571 String code = entry[0], grade = entry[1];
572 if (!GRADE_POINTS.containsKey(grade))
573 throw new CommandError(
574 "invalid grade " + a[0] + " " + code + " " + grade
575 );
576 if (!seen.add(code))
577 throw new CommandError("duplicate record " + a[0] + " " + code);
578 grades.add(new Grade(code, grade, Integer.parseInt(entry[2])));
579 }
580 }
581 Student student = new Student(
582 a[0], Integer.parseInt(a[1]), Integer.parseInt(a[2]), grades
583 );
584 students.put(student.id, student);
585 emit(0, "STUDENT", student.id, "year", student.year,
586 "limit", student.limit, "records", grades.size(),
587 "earned", student.earned(), "gpa", student.gpa());
588 }
589
590 case "ENROLL", "CHECK" -> {
591 Student student = student(a[0]);
592 Course course = course(a[1]);
593 Decision decision = judgment(student, course);
594 if (command.equals("CHECK")) {
595 emit(0, "CHECK", student.id, course.code, decision.text());
596 } else if (decision.word.equals("enroll")) {
597 enroll(student, course);
598 record(student, "enroll " + course.code);
599 enrollmentLine("ENROLL", student, course, 0);
600 } else if (decision.word.equals("waitlist")) {
601 int position = Integer.parseInt(decision.extra.get(0));
602 course.waiting.add(position - 1, student.id);
603 record(student, "waitlist " + course.code + " " + position);
604 emit(0, "ENROLL", student.id, course.code, "waitlist", position);
605 } else {
606 reject(student, course, decision);
607 }
608 }
609
610 case "OPTIONS" -> {
611 Student student = student(a[0]);
612 emit(0, "OPTIONS", student.id, "courses", courses.size());
613 for (Course course : courses.values()) {
614 emit(2, course.code, judgment(student, course).text());
615 }
616 }
617
618 case "DROP" -> {
619 Student student = student(a[0]);
620 Course course = course(a[1]);
621 if (course.enrolled.contains(student.id)) {
622 withdraw(student, course);
623 record(student, "drop " + course.code);
624 enrollmentLine("DROP", student, course, 0);
625 } else if (course.waiting.remove(student.id)) {
626 record(student, "drop " + course.code + " waitlist");
627 emit(0, "DROP", student.id, course.code, "waitlist", "removed");
628 } else {
629 throw new CommandError("not registered " + student.id + " " + course.code);
630 }
631 promote();
632 }
633
634 case "SWAP" -> {
635 Student student = student(a[0]);
636 Course old = course(a[1]);
637 Course target = course(a[2]);
638 if (!old.enrolled.contains(student.id))
639 throw new CommandError("not enrolled " + student.id + " " + old.code);
640 if (old == target)
641 throw new CommandError("same course " + old.code);
642 Decision decision = eligible(student, target, old.code);
643 if (!decision.word.equals("OK")) reject(student, target, decision);
644 if (target.enrolled.size() >= target.capacity)
645 throw new CommandError("course full " + target.code);
646 withdraw(student, old);
647 enroll(student, target);
648 record(student, "swap " + old.code + " " + target.code);
649 emit(0, "SWAP", student.id, old.code, "->", target.code,
650 "credits", credits(student, null) + "/" + student.limit);
651 promote();
652 }
653
654 case "CLOSE", "OPEN" -> {
655 Course course = course(a[0]);
656 boolean opening = command.equals("OPEN");
657 if (course.open == opening)
658 throw new CommandError(
659 "already " + (opening ? "open " : "closed ") + course.code
660 );
661 course.open = opening;
662 emit(0, command, course.code, "enrolled",
663 course.enrolled.size() + "/" + course.capacity,
664 "waitlist", course.waiting.size());
665 if (opening) promote();
666 }
667
668 case "CAPACITY" -> {
669 Course course = course(a[0]);
670 int capacity = Integer.parseInt(a[1]);
671 if (capacity < course.enrolled.size())
672 throw new CommandError(
673 "capacity below enrollment " + course.code + " " + course.enrolled.size()
674 );
675 int previous = course.capacity;
676 course.capacity = capacity;
677 emit(0, "CAPACITY", course.code, previous, "->", capacity);
678 promote();
679 }
680
681 case "LIMIT" -> {
682 Student student = student(a[0]);
683 int limit = Integer.parseInt(a[1]);
684 int enrolled = credits(student, null);
685 if (limit < enrolled)
686 throw new CommandError("limit below enrolled " + student.id + " " + enrolled);
687 int previous = student.limit;
688 student.limit = limit;
689 emit(0, "LIMIT", student.id, previous, "->", limit);
690 promote();
691 }
692
693 case "PREREQ" -> {
694 Course course = course(a[0]);
695 Expression prerequisite = PrerequisiteParser.from(course.code, a[1]);
696 course.prerequisite = prerequisite;
697 emit(0, "PREREQ", course.code, course.prerequisiteText());
698 List<String> remaining = new ArrayList<>();
699 for (String id : course.waiting) {
700 Student student = students.get(id);
701 if (prerequisite == null || prerequisite.satisfied(student)) {
702 remaining.add(id);
703 } else {
704 record(student, "unlist " + course.code);
705 emit(2, "UNLIST", student.id, course.code);
706 }
707 }
708 course.waiting.clear();
709 course.waiting.addAll(remaining);
710 }
711
712 case "CANCEL" -> {
713 Course course = course(a[0]);
714 emit(0, "CANCEL", course.code, "enrolled",
715 course.enrolled.size(), "waitlist", course.waiting.size());
716 for (String id : course.enrolled) {
717 Student student = students.get(id);
718 student.taking.remove(course.code);
719 record(student, "cancel " + course.code);
720 }
721 for (String id : course.waiting)
722 record(students.get(id), "cancel " + course.code + " waitlist");
723 courses.remove(course.code);
724 promote();
725 }
726
727 case "ROSTER" -> {
728 Course course = course(a[0]);
729 emit(0, "ROSTER", course.code, course.status(), "enrolled",
730 course.enrolled.size() + "/" + course.capacity,
731 "waitlist", course.waiting.size());
732 for (String id : course.enrolled)
733 emit(2, "enrolled", id, "year", students.get(id).year);
734 for (int position = 0; position < course.waiting.size(); ++position) {
735 String id = course.waiting.get(position);
736 emit(2, "waitlist", position + 1, id, "year", students.get(id).year);
737 }
738 }
739
740 case "SCHEDULE" -> schedule(student(a[0]));
741
742 case "FREE" -> {
743 Student student = student(a[0]);
744 String day = a[1];
745 if (!DAYS.contains(day)) throw new CommandError("invalid day " + day);
746 List<Slot> meetings = new ArrayList<>();
747 for (Course course : taking(student, null))
748 for (Slot slot : course.slots)
749 if (slot.day().equals(day) && slot.start() < 1320 && slot.end() > 480)
750 meetings.add(slot);
751 meetings.sort(Comparator.comparingInt(Slot::start));
752 List<int[]> ranges = new ArrayList<>();
753 int cursor = 480;
754 for (Slot slot : meetings) {
755 int start = Math.max(slot.start(), 480);
756 if (start > cursor) ranges.add(new int[] { cursor, start });
757 cursor = Math.max(cursor, Math.min(slot.end(), 1320));
758 }
759 if (cursor < 1320) ranges.add(new int[] { cursor, 1320 });
760 int free = ranges.stream().mapToInt(interval -> interval[1] - interval[0]).sum();
761 List<Object> line = new ArrayList<>();
762 line.addAll(Arrays.asList("FREE", student.id, day, "total", time(free)));
763 for (int[] interval : ranges)
764 line.add(time(interval[0]) + "-" + time(interval[1]));
765 emit(0, line.toArray());
766 }
767
768 case "INFO" -> {
769 Course course = course(a[0]);
770 emit(0, "INFO", course.code, course.status(), "credits", course.credits,
771 "capacity", course.capacity, "prereq", course.prerequisiteText());
772 List<Slot> slots = new ArrayList<>(course.slots);
773 slots.sort(Comparator.comparingInt((Slot slot) -> DAYS.indexOf(slot.day()))
774 .thenComparingInt(Slot::start));
775 if (slots.isEmpty()) emit(2, "meet", "TBA");
776 for (Slot slot : slots)
777 emit(2, "meet", slot.day(),
778 time(slot.start()) + "-" + time(slot.end()));
779 List<Object> required = new ArrayList<>();
780 required.add("required");
781 required.add("by");
782 for (Course other : courses.values()) {
783 if (other.prerequisite != null &&
784 other.prerequisite.contains(course.code)) {
785 required.add(other.code);
786 }
787 }
788 if (required.size() == 2) required.add("none");
789 emit(2, required.toArray());
790 }
791
792 case "AUDIT" -> {
793 Student student = student(a[0]);
794 emit(0, "AUDIT", student.id, "year", student.year, "limit", student.limit);
795 List<Grade> grades = new ArrayList<>(student.grades);
796 grades.sort(Comparator.comparing(Grade::course));
797 for (Grade grade : grades)
798 emit(2, "record", grade.course(), grade.grade(), grade.credits());
799 emit(2, "earned", student.earned(), "gpa", student.gpa());
800 for (Course course : taking(student, null))
801 emit(2, "enrolled", course.code, course.credits);
802 int credits = credits(student, null);
803 emit(2, "credits", credits, "remaining", student.limit - credits);
804 for (Course course : waitlists(student))
805 emit(2, "waitlist", course.code, course.waiting.indexOf(student.id) + 1);
806 }
807
808 case "HISTORY" -> {
809 Student student = student(a[0]);
810 emit(0, "HISTORY", student.id, "events", student.history.size());
811 for (History history : student.history)
812 emit(2, history.number(), history.event());
813 }
814
815 case "STUDENTS" -> {
816 emit(0, "STUDENTS", students.size());
817 for (Student student : students.values())
818 emit(2, student.id, "year", student.year,
819 "credits", credits(student, null) + "/" + student.limit,
820 "courses", student.taking.size(), "waitlists", waitlists(student).size());
821 }
822
823 case "REPORT" -> {
824 int enrolled = 0;
825 int capacity = 0;
826 int waiting = 0;
827 Map<String, List<Course>> departments = new TreeMap<>();
828 for (Course course : courses.values()) {
829 enrolled += course.enrolled.size();
830 capacity += course.capacity;
831 waiting += course.waiting.size();
832 departments.computeIfAbsent(course.department(), key -> new ArrayList<>())
833 .add(course);
834 }
835 emit(0, "REPORT", "courses", courses.size(), "enrolled",
836 enrolled + "/" + capacity, "rate", rate(enrolled, capacity),
837 "waitlisted", waiting);
838 for (Course course : courses.values()) {
839 emit(2, course.code, course.status(),
840 course.enrolled.size() + "/" + course.capacity,
841 rate(course.enrolled.size(), course.capacity),
842 "waitlist", course.waiting.size());
843 }
844 for (var entry : departments.entrySet()) {
845 List<Course> group = entry.getValue();
846 int count = group.stream().mapToInt(c -> c.enrolled.size()).sum();
847 int seats = group.stream().mapToInt(c -> c.capacity).sum();
848 int queue = group.stream().mapToInt(c -> c.waiting.size()).sum();
849 emit(2, "dept", entry.getKey(), "courses", group.size(),
850 "enrolled", count + "/" + seats, "rate", rate(count, seats),
851 "waitlisted", queue);
852 }
853 }
854 }

857 void execute(String line) {
858 ++commandNumber;
859 String[] tokens = line.split(" ");
860 try {
861 command(tokens[0], Arrays.copyOfRange(tokens, 1, tokens.length));
862 } catch (CommandError error) {
863 emit(0, "ERROR", error.getMessage());
864 }
865 }

867 String output() {
868 return output.isEmpty() ? "" : String.join("\n", output) + "\n";
869 }
