4const DAYS: [&str; 7] = ["MO", "TU", "WE", "TH", "FR", "SA", "SU"];
5
6fn number(text: &str) -> i32 { text.parse().unwrap() }
7
8fn time(minutes: i32) -> String { format!("{:02}:{:02}", minutes / 60, minutes % 60) }
9
10fn rate(enrolled: usize, capacity: usize) -> String {
11 if capacity == 0 { "n/a".into() } else { format!("{}%", enrolled * 100 / capacity) }
12}

14#[derive(Clone)]
15struct Slot { day: usize, start: i32, end: i32, text: String }
16
17impl Slot {
18 fn parse(text: &str) -> Result<Self, String> {
19 let invalid = || format!("invalid slot {text}");
20 let day = DAYS.iter().position(|day| *day == &text[..2]).ok_or_else(invalid)?;
21 let (start_hour, start_minute) = (number(&text[2..4]), number(&text[4..6]));
22 let (end_hour, end_minute) = (number(&text[7..9]), number(&text[9..11]));
23 let (start, end) = (start_hour * 60 + start_minute, end_hour * 60 + end_minute);
24 if start_hour > 23 || end_hour > 23 || start_minute > 59 || end_minute > 59 || start >= end {
25 return Err(invalid());
26 }
27 Ok(Self { day, start, end, text: text.into() })
28 }
29
30 fn overlaps(&self, other: &Self) -> bool {
31 self.day == other.day && self.start < other.end && other.start < self.end
32 }
33}

35enum Prerequisite {
36 None,
37 Course(String),
38 Year(i32),
39 And(Vec<Prerequisite>),
40 Or(Vec<Prerequisite>),
41}
42
43impl Prerequisite {
44 fn combine(children: Vec<Self>, conjunction: bool) -> Self {
45 let mut flattened = Vec::new();
46 for child in children {
47 match child {
48 Self::And(nested) if conjunction => flattened.extend(nested),
49 Self::Or(nested) if !conjunction => flattened.extend(nested),
50 other => flattened.push(other),
51 }
52 }
53 if flattened.len() == 1 { flattened.pop().unwrap() }
54 else if conjunction { Self::And(flattened) } else { Self::Or(flattened) }
55 }
56
57 fn parse(code: &str, text: &str) -> Result<Self, String> {
58 if text == "-" { return Ok(Self::None); }
59 let mut parser = Parser { text: text.as_bytes(), position: 0 };
60 let expression = parser.expression().map_err(|_| format!("invalid prereq {code}"))?;
61 if parser.position != text.len() { return Err(format!("invalid prereq {code}")); }
62 if expression.contains(code) { return Err(format!("self prereq {code}")); }
63 Ok(expression)
64 }
65
66 fn contains(&self, course: &str) -> bool {
67 match self {
68 Self::Course(code) => code == course,
69 Self::And(children) | Self::Or(children) => children.iter().any(|child| child.contains(course)),
70 _ => false,
71 }
72 }
73
74 fn satisfied(&self, student: &Student) -> bool {
75 match self {
76 Self::None => true,
77 Self::Course(course) => student.passed(course).is_some(),
78 Self::Year(year) => student.year >= *year,
79 Self::And(children) => children.iter().all(|child| child.satisfied(student)),
80 Self::Or(children) => children.iter().any(|child| child.satisfied(student)),
81 }
82 }
83
84 fn canonical(&self) -> String {
85 match self {
86 Self::None => "-".into(),
87 Self::Course(course) => course.clone(),
88 Self::Year(year) => format!("Y{year}"),
89 Self::And(children) => children.iter().map(|child| {
90 let text = child.canonical();
91 if matches!(child, Self::Or(_)) { format!("({text})") } else { text }
92 }).collect::<Vec<_>>().join("&"),
93 Self::Or(children) => children.iter().map(Self::canonical).collect::<Vec<_>>().join("|"),
94 }
95 }
96}

98struct Parser<'a> { text: &'a [u8], position: usize }
99
100impl Parser<'_> {
101 fn take(&mut self, symbol: u8) -> bool {
102 if self.text.get(self.position) == Some(&symbol) { self.position += 1; true } else { false }
103 }
104
105 fn expression(&mut self) -> Result<Prerequisite, ()> {
106 let mut children = vec![self.term()?];
107 while self.take(b'|') { children.push(self.term()?); }
108 Ok(Prerequisite::combine(children, false))
109 }
110
111 fn term(&mut self) -> Result<Prerequisite, ()> {
112 let mut children = vec![self.factor()?];
113 while self.take(b'&') { children.push(self.factor()?); }
114 Ok(Prerequisite::combine(children, true))
115 }
116
117 fn factor(&mut self) -> Result<Prerequisite, ()> {
118 if self.take(b'(') {
119 let expression = self.expression()?;
120 if !self.take(b')') { return Err(()); }
121 return Ok(expression);
122 }
123 let start = self.position;
124 while self.text.get(self.position).is_some_and(|byte| byte.is_ascii_alphanumeric()) { self.position += 1; }
125 let word = &self.text[start..self.position];
126 if word.len() == 2 && word[0] == b'Y' && (b'1'..=b'9').contains(&word[1]) {
127 return Ok(Prerequisite::Year((word[1] - b'0') as i32));
128 }
129 let letters = word.iter().take_while(|byte| byte.is_ascii_uppercase()).count();
130 if (2..=4).contains(&letters) && word.len() == letters + 3 && word[letters..].iter().all(u8::is_ascii_digit) {
131 Ok(Prerequisite::Course(String::from_utf8(word.to_vec()).unwrap()))
132 } else { Err(()) }
133 }
134}

136struct Grade { letter: char, credits: i32 }
137
138impl Grade {
139 fn points(&self) -> i32 {
140 match self.letter { 'A' => 4, 'B' => 3, 'C' => 2, 'D' => 1, _ => 0 }
141 }
142}

144struct Student {
145 year: i32,
146 limit: i32,
147 grades: BTreeMap<String, Grade>,
148 enrolled: BTreeSet<String>,
149 history: Vec<(usize, String)>,
150}
151
152impl Student {
153 fn passed(&self, course: &str) -> Option<char> {
154 self.grades.get(course).filter(|grade| matches!(grade.letter, 'A' | 'B' | 'C')).map(|grade| grade.letter)
155 }
156
157 fn earned(&self) -> i32 {
158 self.grades.values().filter(|grade| grade.letter != 'F').map(|grade| grade.credits).sum()
159 }
160
161 fn gpa(&self) -> String {
162 if self.grades.is_empty() { return "n/a".into(); }
163 let credits: i32 = self.grades.values().map(|grade| grade.credits).sum();
164 let points: i32 = self.grades.values().map(|grade| grade.points() * grade.credits).sum();
165 let value = (200 * points + credits) / (2 * credits);
166 format!("{}.{:02}", value / 100, value % 100)
167 }
168}

170struct Course {
171 credits: i32,
172 capacity: usize,
173 slots: Vec<Slot>,
174 prereq: Prerequisite,
175 open: bool,
176 enrolled: BTreeSet<String>,
177 waitlist: Vec<String>,
178}
179
180impl Course {
181 fn weekly(&self) -> i32 { self.slots.iter().map(|slot| slot.end - slot.start).sum() }
182
183 fn status(&self) -> &'static str { if self.open { "open" } else { "closed" } }
184
185 fn free_seat(&self) -> bool { self.enrolled.len() < self.capacity }
186
187 fn conflicts(&self, other: &Self) -> bool {
188 self.slots.iter().any(|slot| other.slots.iter().any(|other| slot.overlaps(other)))
189 }
190}

192struct Reason { word: &'static str, extra: String }
193
194impl Reason {
195 fn new(word: &'static str) -> Self { Self { word, extra: String::new() } }
196
197 fn text(&self) -> String { format!("{}{}", self.word, self.extra) }
198
199 fn error(&self, student: &str, course: &str) -> String { format!("{} {student} {course}{}", self.word, self.extra) }
200}

202enum Judgment { Enroll, Waitlist(usize), Reject(Reason) }
203
204impl Judgment {
205 fn text(&self) -> String {
206 match self {
207 Self::Enroll => "enroll".into(),
208 Self::Waitlist(position) => format!("waitlist {position}"),
209 Self::Reject(reason) => reason.text(),
210 }
211 }

214#[derive(Default)]
215struct Registrar {
216 courses: BTreeMap<String, Course>,
217 students: BTreeMap<String, Student>,
218 command_number: usize,
219}
220
221impl Registrar {
222 fn student(&self, id: &str) -> Result<&Student, String> {
223 self.students.get(id).ok_or_else(|| format!("unknown student {id}"))
224 }
225
226 fn course(&self, code: &str) -> Result<&Course, String> {
227 self.courses.get(code).ok_or_else(|| format!("unknown course {code}"))
228 }
229
230 fn credits(&self, student: &str, excluded: Option<&str>) -> i32 {
231 self.students[student].enrolled.iter().filter(|code| Some(code.as_str()) != excluded)
232 .map(|code| self.courses[code].credits).sum()
233 }
234
235 fn waitlists(&self, student: &str) -> usize {
236 self.courses.values().filter(|course| course.waitlist.iter().any(|id| id == student)).count()
237 }
238
239 fn conflict(&self, student: &str, target: &str, excluded: Option<&str>) -> Option<&str> {
240 self.students[student].enrolled.iter().filter(|code| Some(code.as_str()) != excluded)
241 .find(|code| self.courses[*code].conflicts(&self.courses[target])).map(String::as_str)
242 }
243
244 fn eligible(&self, id: &str, code: &str, excluded: Option<&str>) -> Result<(), Reason> {
245 let student = &self.students[id];
246 let course = &self.courses[code];
247 if !course.open { return Err(Reason::new("course closed")); }
248 if course.enrolled.contains(id) { return Err(Reason::new("already enrolled")); }
249 if course.waitlist.iter().any(|waiting| waiting == id) { return Err(Reason::new("already waitlisted")); }
250 if let Some(grade) = student.passed(code) { return Err(Reason { word: "already completed", extra: format!(" {grade}") }); }
251 if !course.prereq.satisfied(student) { return Err(Reason::new("prereq not met")); }
252 if let Some(conflict) = self.conflict(id, code, excluded) { return Err(Reason { word: "time conflict", extra: format!(" {conflict}") }); }
253 let total = self.credits(id, excluded) + course.credits;
254 if total > student.limit { return Err(Reason { word: "credit limit", extra: format!(" {total} {}", student.limit) }); }
255 Ok(())
256 }
257
258 fn judge(&self, student: &str, course: &str) -> Judgment {
259 if let Err(reason) = self.eligible(student, course, None) { return Judgment::Reject(reason); }
260 if self.courses[course].free_seat() { return Judgment::Enroll; }
261 if self.waitlists(student) >= 3 { return Judgment::Reject(Reason::new("waitlist limit")); }
262 let year = self.students[student].year;
263 Judgment::Waitlist(1 + self.courses[course].waitlist.iter().filter(|id| self.students[*id].year >= year).count())
264 }
265
266 fn history(&mut self, student: &str, event: String) {
267 self.students.get_mut(student).unwrap().history.push((self.command_number, event));
268 }
269
270 fn enroll(&mut self, student: &str, course: &str) {
271 self.students.get_mut(student).unwrap().enrolled.insert(course.into());
272 self.courses.get_mut(course).unwrap().enrolled.insert(student.into());
273 }
274
275 fn remove_enrollment(&mut self, student: &str, course: &str) {
276 self.students.get_mut(student).unwrap().enrolled.remove(course);
277 self.courses.get_mut(course).unwrap().enrolled.remove(student);
278 }
279
280 fn enrollment_line(&self, prefix: &str, student: &str, code: &str) {
281 let course = &self.courses[code];
282 println!("{prefix} {student} {code} enrolled {}/{} credits {}/{}", course.enrolled.len(), course.capacity,
283 self.credits(student, None), self.students[student].limit);
284 }
285
286 fn promote(&mut self) {
287 let courses: Vec<String> = self.courses.keys().cloned().collect();
288 for code in courses {
289 loop {
290 let course = &self.courses[&code];
291 if !course.open || !course.free_seat() { break; }
292 let candidate = course.waitlist.iter().position(|id| self.conflict(id, &code, None).is_none()
293 && self.credits(id, None) + course.credits <= self.students[id].limit);
294 let Some(position) = candidate else { break; };
295 let student = self.courses.get_mut(&code).unwrap().waitlist.remove(position);
296 self.enroll(&student, &code);
297 self.history(&student, format!("promote {code}"));
298 self.enrollment_line(" PROMOTE", &student, &code);
299 }
300 }
301 }

303 fn meetings(&self, student: &str) -> Vec<(usize, i32, i32, &str)> {
304 let mut meetings = Vec::new();
305 for code in &self.students[student].enrolled {
306 for slot in &self.courses[code].slots { meetings.push((slot.day, slot.start, slot.end, code.as_str())); }
307 }
308 meetings.sort_by_key(|meeting| (meeting.0, meeting.1));
309 meetings
310 }

312 fn schedule(&self, id: &str) {
313 let student = &self.students[id];
314 let weekly = student.enrolled.iter().map(|code| self.courses[code].weekly()).sum();
315 println!("SCHEDULE {id} courses {} credits {} weekly {}", student.enrolled.len(), self.credits(id, None), time(weekly));
316 let meetings = self.meetings(id);
317 for (day, name) in DAYS.iter().enumerate() {
318 let slots: Vec<String> = meetings.iter().filter(|meeting| meeting.0 == day)
319 .map(|(_, start, end, code)| format!("{}-{} {code}", time(*start), time(*end))).collect();
320 if !slots.is_empty() { println!(" {name} {}", slots.join(" ")); }
321 }
322 let tba: Vec<&str> = student.enrolled.iter().filter(|code| self.courses[*code].slots.is_empty()).map(String::as_str).collect();
323 if !tba.is_empty() { println!(" TBA {}", tba.join(" ")); }
324 }

326 fn free(&self, id: &str, day: usize) {
327 let mut busy = [false; 840];
328 for (meeting_day, start, end, _) in self.meetings(id) {
329 if meeting_day == day {
330 for minute in start.max(480)..end.min(1320) { busy[(minute - 480) as usize] = true; }
331 }
332 }
333 let mut position = 0;
334 let mut total = 0;
335 let mut ranges = Vec::new();
336 while position < busy.len() {
337 if busy[position] { position += 1; continue; }
338 let start = position;
339 while position < busy.len() && !busy[position] { position += 1; }
340 total += (position - start) as i32;
341 ranges.push(format!("{}-{}", time(start as i32 + 480), time(position as i32 + 480)));
342 }
343 println!("FREE {id} {} total {}{}", DAYS[day], time(total),
344 if ranges.is_empty() { String::new() } else { format!(" {}", ranges.join(" ")) });
345 }

355 println!(" required by {}", if dependents.is_empty() { "none".into() } else { dependents.join(" ") });
356 }

372 let mut departments: BTreeMap<&str, [usize; 4]> = BTreeMap::new();
373 let mut total = [0; 4];

381 for (code, course) in &self.courses {
382 println!(" {code} {} {}/{} {} waitlist {}", course.status(), course.enrolled.len(), course.capacity,
383 rate(course.enrolled.len(), course.capacity), course.waitlist.len());
384 }
385 for (department, counts) in departments {
386 println!(" dept {department} courses {} enrolled {}/{} rate {} waitlisted {}", counts[0], counts[1], counts[2],
387 rate(counts[1], counts[2]), counts[3]);
388 }

391 fn command(&mut self, tokens: &[&str]) -> Result<(), String> {
392 let a = &tokens[1..];
393 match tokens[0] {
394 "COURSE" => {
395 if self.courses.contains_key(a[0]) { return Err(format!("duplicate course {}", a[0])); }
396 let slots: Vec<Slot> = if a[3] == "-" { Vec::new() }
397 else { a[3].split(',').map(Slot::parse).collect::<Result<_, _>>()? };
398 for first in 0..slots.len() {
399 for second in first + 1..slots.len() {
400 if slots[first].overlaps(&slots[second]) {
401 return Err(format!("overlapping slots {} {}", slots[first].text, slots[second].text));
402 }
403 }
404 }
405 let prereq = Prerequisite::parse(a[0], a[4])?;
406 let course = Course { credits: number(a[1]), capacity: number(a[2]) as usize,
407 slots, prereq, open: true, enrolled: BTreeSet::new(), waitlist: Vec::new() };
408 println!("COURSE {} credits {} capacity {} meetings {} weekly {} prereq {}", a[0], course.credits,
409 course.capacity, course.slots.len(), time(course.weekly()), course.prereq.canonical());
410 self.courses.insert(a[0].into(), course);
411 }
412 "STUDENT" => {
413 if self.students.contains_key(a[0]) { return Err(format!("duplicate student {}", a[0])); }
414 let mut grades = BTreeMap::new();
415 if a[3] != "-" {
416 for record in a[3].split(',') {
417 let parts: Vec<&str> = record.split(':').collect();
418 let letter = parts[1].chars().next().unwrap();
419 if !['A', 'B', 'C', 'D', 'F'].contains(&letter) {
420 return Err(format!("invalid grade {} {} {letter}", a[0], parts[0]));
421 }
422 if grades.contains_key(parts[0]) { return Err(format!("duplicate record {} {}", a[0], parts[0])); }
423 grades.insert(parts[0].to_string(), Grade { letter, credits: number(parts[2]) });
424 }
425 }
426 let student = Student { year: number(a[1]), limit: number(a[2]), grades,
427 enrolled: BTreeSet::new(), history: Vec::new() };
428 println!("STUDENT {} year {} limit {} records {} earned {} gpa {}", a[0], student.year,
429 student.limit, student.grades.len(), student.earned(), student.gpa());
430 self.students.insert(a[0].into(), student);
431 }
432 "ENROLL" | "CHECK" => {
433 self.student(a[0])?;
434 self.course(a[1])?;
435 let judgment = self.judge(a[0], a[1]);
436 if tokens[0] == "CHECK" { println!("CHECK {} {} {}", a[0], a[1], judgment.text()); }
437 else {
438 match judgment {
439 Judgment::Reject(reason) => return Err(reason.error(a[0], a[1])),
440 Judgment::Enroll => {
441 self.enroll(a[0], a[1]);
442 self.history(a[0], format!("enroll {}", a[1]));
443 self.enrollment_line("ENROLL", a[0], a[1]);
444 }
445 Judgment::Waitlist(position) => {
446 self.courses.get_mut(a[1]).unwrap().waitlist.insert(position - 1, a[0].into());
447 self.history(a[0], format!("waitlist {} {position}", a[1]));
448 println!("ENROLL {} {} waitlist {position}", a[0], a[1]);
449 }
450 }
451 }
452 }
453 "OPTIONS" => {
454 self.student(a[0])?;
455 println!("OPTIONS {} courses {}", a[0], self.courses.len());
456 for code in self.courses.keys() { println!(" {code} {}", self.judge(a[0], code).text()); }
457 }
458 "DROP" => {
459 self.student(a[0])?;
460 let course = self.course(a[1])?;
461 if course.enrolled.contains(a[0]) {
462 self.remove_enrollment(a[0], a[1]);
463 self.history(a[0], format!("drop {}", a[1]));
464 self.enrollment_line("DROP", a[0], a[1]);
465 } else if let Some(position) = course.waitlist.iter().position(|id| id == a[0]) {
466 self.courses.get_mut(a[1]).unwrap().waitlist.remove(position);
467 self.history(a[0], format!("drop {} waitlist", a[1]));
468 println!("DROP {} {} waitlist removed", a[0], a[1]);
469 } else { return Err(format!("not registered {} {}", a[0], a[1])); }
470 self.promote();
471 }
472 "SWAP" => {
473 self.student(a[0])?;
474 self.course(a[1])?;
475 self.course(a[2])?;
476 if !self.courses[a[1]].enrolled.contains(a[0]) { return Err(format!("not enrolled {} {}", a[0], a[1])); }
477 if a[1] == a[2] { return Err(format!("same course {}", a[1])); }
478 self.eligible(a[0], a[2], Some(a[1])).map_err(|reason| reason.error(a[0], a[2]))?;
479 if !self.courses[a[2]].free_seat() { return Err(format!("course full {}", a[2])); }
480 self.remove_enrollment(a[0], a[1]);
481 self.enroll(a[0], a[2]);
482 self.history(a[0], format!("swap {} {}", a[1], a[2]));
483 println!("SWAP {} {} -> {} credits {}/{}", a[0], a[1], a[2], self.credits(a[0], None), self.students[a[0]].limit);
484 self.promote();
485 }
486 "CLOSE" | "OPEN" => {
487 let course = self.course(a[0])?;
488 let open = tokens[0] == "OPEN";
489 if course.open == open { return Err(format!("already {} {}", course.status(), a[0])); }
490 let course = self.courses.get_mut(a[0]).unwrap();
491 course.open = open;
492 println!("{} {} enrolled {}/{} waitlist {}", tokens[0], a[0], course.enrolled.len(), course.capacity, course.waitlist.len());
493 if open { self.promote(); }
494 }
495 "CAPACITY" => {
496 let course = self.course(a[0])?;
497 let capacity = number(a[1]) as usize;
498 if capacity < course.enrolled.len() { return Err(format!("capacity below enrollment {} {}", a[0], course.enrolled.len())); }
499 let course = self.courses.get_mut(a[0]).unwrap();
500 let old = course.capacity;
501 course.capacity = capacity;
502 println!("CAPACITY {} {old} -> {capacity}", a[0]);
503 self.promote();
504 }
505 "LIMIT" => {
506 self.student(a[0])?;
507 let limit = number(a[1]);
508 let credits = self.credits(a[0], None);
509 if limit < credits { return Err(format!("limit below enrolled {} {credits}", a[0])); }
510 let student = self.students.get_mut(a[0]).unwrap();
511 let old = student.limit;
512 student.limit = limit;
513 println!("LIMIT {} {old} -> {limit}", a[0]);
514 self.promote();
515 }
516 "PREREQ" => {
517 self.course(a[0])?;
518 let prereq = Prerequisite::parse(a[0], a[1])?;
519 let removed: Vec<String> = self.courses[a[0]].waitlist.iter()
520 .filter(|id| !prereq.satisfied(&self.students[*id])).cloned().collect();
521 println!("PREREQ {} {}", a[0], prereq.canonical());
522 let course = self.courses.get_mut(a[0]).unwrap();
523 course.prereq = prereq;
524 course.waitlist.retain(|student| !removed.contains(student));
525 for student in removed {
526 self.history(&student, format!("unlist {}", a[0]));
527 println!(" UNLIST {student} {}", a[0]);
528 }
529 }
530 "CANCEL" => {
531 self.course(a[0])?;
532 let course = self.courses.remove(a[0]).unwrap();
533 println!("CANCEL {} enrolled {} waitlist {}", a[0], course.enrolled.len(), course.waitlist.len());
534 for student in course.enrolled {
535 self.students.get_mut(&student).unwrap().enrolled.remove(a[0]);
536 self.history(&student, format!("cancel {}", a[0]));
537 }
538 for student in course.waitlist { self.history(&student, format!("cancel {} waitlist", a[0])); }
539 self.promote();
540 }
541 "ROSTER" => {
542 let course = self.course(a[0])?;
543 println!("ROSTER {} {} enrolled {}/{} waitlist {}", a[0], course.status(), course.enrolled.len(), course.capacity, course.waitlist.len());
544 for id in &course.enrolled { println!(" enrolled {id} year {}", self.students[id].year); }
545 for (index, id) in course.waitlist.iter().enumerate() { println!(" waitlist {} {id} year {}", index + 1, self.students[id].year); }
546 }
547 "SCHEDULE" => { self.student(a[0])?; self.schedule(a[0]); }
548 "FREE" => {
549 self.student(a[0])?;
550 let day = DAYS.iter().position(|day| *day == a[1]).ok_or_else(|| format!("invalid day {}", a[1]))?;
551 self.free(a[0], day);
552 }
553 "INFO" => { self.course(a[0])?; self.info(a[0]); }
554 "AUDIT" => { self.student(a[0])?; self.audit(a[0]); }
555 "HISTORY" => {
556 let student = self.student(a[0])?;
557 println!("HISTORY {} events {}", a[0], student.history.len());
558 for (command, event) in &student.history { println!(" {command} {event}"); }
559 }
560 "STUDENTS" => {
561 println!("STUDENTS {}", self.students.len());
562 for (id, student) in &self.students {
563 println!(" {id} year {} credits {}/{} courses {} waitlists {}", student.year,
564 self.credits(id, None), student.limit, student.enrolled.len(), self.waitlists(id));
565 }
566 }
567 "REPORT" => self.report(),
568 _ => unreachable!(),
569 }
570 Ok(())
571 }
572}

574fn main() {
575 let mut input = String::new();
576 io::stdin().read_to_string(&mut input).unwrap();
577 let mut lines = input.lines();
578 let count: usize = lines.next().unwrap().parse().unwrap();
579 let mut registrar = Registrar::default();
580 for (index, line) in lines.take(count).enumerate() {
581 registrar.command_number = index + 1;
582 let tokens: Vec<&str> = line.split_ascii_whitespace().collect();
583 if let Err(error) = registrar.command(&tokens) { println!("ERROR {error}"); }
584 }
585}
