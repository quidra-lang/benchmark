22private func courseWord(_ word: String) -> Bool {
23 let characters = Array(word.utf8)
24 guard (5...7).contains(characters.count) else { return false }
25 let prefix = characters.count - 3
26 return (2...4).contains(prefix)
27 && characters[..<prefix].allSatisfy { (65...90).contains(Int($0)) }
28 && characters[prefix...].allSatisfy { (48...57).contains(Int($0)) }
29}
30
31private func yearWord(_ word: String) -> Bool {
32 let characters = Array(word.utf8)
33 return characters.count == 2 && characters[0] == 89 && (49...57).contains(Int(characters[1]))
34}

36private indirect enum Expression {
37 case unconditional
38 case word(String)
39 case and([Expression])
40 case or([Expression])
41
42 var canonical: String {
43 switch self {
44 case .unconditional:
45 return "-"
46 case .word(let value):
47 return value
48 case .and(let children):
49 return children.map { child in
50 if case .or = child {
51 return "(\(child.canonical))"
52 }
53 return child.canonical
54 }.joined(separator: "&")
55 case .or(let children):
56 return children.map(\.canonical).joined(separator: "|")
57 }
58 }
59
60 var references: Set<String> {
61 switch self {
62 case .unconditional:
63 return []
64 case .word(let value):
65 return yearWord(value) ? [] : [value]
66 case .and(let children), .or(let children):
67 return children.reduce(into: Set<String>()) { $0.formUnion($1.references) }
68 }
69 }
70
71 func evaluates(_ student: Student) -> Bool {
72 switch self {
73 case .unconditional:
74 return true
75 case .word(let value):
76 if yearWord(value) {
77 return student.year >= Int(String(value.suffix(1)))!
78 }
79 return student.passed[value] != nil
80 case .and(let children):
81 return children.allSatisfy { $0.evaluates(student) }
82 case .or(let children):
83 return children.contains { $0.evaluates(student) }
84 }
85 }
86
87 static func combined(_ conjunction: Bool, _ children: [Expression]) -> Expression {
88 var flattened: [Expression] = []
89 for child in children {
90 switch child {
91 case .and(let values) where conjunction:
92 flattened.append(contentsOf: values)
93 case .or(let values) where !conjunction:
94 flattened.append(contentsOf: values)
95 default:
96 flattened.append(child)
97 }
98 }
99 if flattened.count == 1 { return flattened[0] }
100 return conjunction ? .and(flattened) : .or(flattened)
101 }
102}

104private final class PrerequisiteParser {
105 private var tokens: [String] = []
106 private var position = 0
107
108 init(_ source: String) throws {
109 let bytes = Array(source.utf8)
110 var index = 0
111 while index < bytes.count {
112 let character = bytes[index]
113 if character == 38 || character == 124 || character == 40 || character == 41 {
114 tokens.append(String(UnicodeScalar(character)))
115 index += 1
116 continue
117 }
118 let start = index
119 while index < bytes.count
120 && ((48...57).contains(Int(bytes[index])) || (65...90).contains(Int(bytes[index]))) {
121 index += 1
122 }
123 guard index > start else { try fail("parse") }
124 let word = String(decoding: bytes[start..<index], as: UTF8.self)
125 guard courseWord(word) || yearWord(word) else { try fail("parse") }
126 tokens.append(word)
127 }
128 }
129
130 private var next: String? {
131 position < tokens.count ? tokens[position] : nil
132 }
133
134 private func factor() throws -> Expression {
135 guard let token = next, !["&", "|", ")"].contains(token) else {
136 try fail("parse")
137 }
138 position += 1
139 if token == "(" {
140 let value = try expression()
141 guard next == ")" else { try fail("parse") }
142 position += 1
143 return value
144 }
145 return .word(token)
146 }
147
148 private func term() throws -> Expression {
149 var children = [try factor()]
150 while next == "&" {
151 position += 1
152 children.append(try factor())
153 }
154 return .combined(true, children)
155 }
156
157 private func expression() throws -> Expression {
158 var children = [try term()]
159 while next == "|" {
160 position += 1
161 children.append(try term())
162 }
163 return .combined(false, children)
164 }
165
166 func parse() throws -> Expression {
167 let result = try expression()
168 guard position == tokens.count else { try fail("parse") }
169 return result
170 }
171}
172
173private func prerequisite(_ code: String, _ source: String) throws -> Expression {
174 if source == "-" { return .unconditional }
175 let result: Expression
176 do {
177 result = try PrerequisiteParser(source).parse()
178 } catch {
179 try fail("invalid prereq \(code)")
180 }
181 guard !result.references.contains(code) else { try fail("self prereq \(code)") }
182 return result
183}

185private struct Slot {
186 let day: Int
187 let start: Int
188 let end: Int
189 let original: String
190
191 init(_ source: String) throws {
192 let bytes = Array(source.utf8)
193 guard bytes.count == 11 else { try fail("invalid slot \(source)") }
194 let dayText = String(decoding: bytes[0...1], as: UTF8.self)
195 guard let day = days.firstIndex(of: dayText) else { try fail("invalid slot \(source)") }
196 let startHour = Int(String(decoding: bytes[2...3], as: UTF8.self))!
197 let startMinute = Int(String(decoding: bytes[4...5], as: UTF8.self))!
198 let endHour = Int(String(decoding: bytes[7...8], as: UTF8.self))!
199 let endMinute = Int(String(decoding: bytes[9...10], as: UTF8.self))!
200 guard startHour <= 23, endHour <= 23, startMinute <= 59, endMinute <= 59 else {
201 try fail("invalid slot \(source)")
202 }
203 let start = startHour * 60 + startMinute
204 let end = endHour * 60 + endMinute
205 guard start < end else { try fail("invalid slot \(source)") }
206 self.day = day
207 self.start = start
208 self.end = end
209 self.original = source
210 }
211
212 func overlaps(_ other: Slot) -> Bool {
213 day == other.day && start < other.end && other.start < end
214 }
215
216 var display: String {
217 "\(clock(start))-\(clock(end))"
218 }
219}

221private final class Course {
222 let code: String
223 let credits: Int
224 var capacity: Int
225 let slots: [Slot]
226 var prerequisite: Expression
227 var isOpen = true
228 var enrolled: Set<String> = []
229 var waitlist: [String] = []
230 var conflicts: Set<String> = []
231
232 init(_ code: String, _ credits: Int, _ capacity: Int,
233 _ slots: [Slot], _ prerequisite: Expression) {
234 self.code = code
235 self.credits = credits
236 self.capacity = capacity
237 self.slots = slots
238 self.prerequisite = prerequisite
239 }
240
241 var status: String { isOpen ? "open" : "closed" }
242 var weekly: Int { slots.reduce(0) { $0 + $1.end - $1.start } }
243 var department: String { String(code.dropLast(3)) }
244}

246private final class Student {
247 let identifier: String
248 let year: Int
249 var limit: Int
250 let records: [String: (grade: String, credits: Int)]
251 let passed: [String: String]
252 var enrolled: Set<String> = []
253 var waiting: Set<String> = []
254 var credits = 0
255 var history: [(number: Int, event: String)] = []
256
257 init(_ identifier: String, _ year: Int, _ limit: Int,
258 _ records: [String: (grade: String, credits: Int)]) {
259 self.identifier = identifier
260 self.year = year
261 self.limit = limit
262 self.records = records
263 var passed: [String: String] = [:]
264 for (code, record) in records where ["A", "B", "C"].contains(record.grade) {
265 passed[code] = record.grade
266 }
267 self.passed = passed
268 }
269
270 var earned: Int {
271 records.values.reduce(0) { $0 + ($1.grade == "F" ? 0 : $1.credits) }
272 }
273
274 var gpa: String {
275 guard !records.isEmpty else { return "n/a" }
276 let weighted = records.values.reduce(0) { $0 + gradePoints[$1.grade]! * $1.credits }
277 let credits = records.values.reduce(0) { $0 + $1.credits }
278 let value = (200 * weighted + credits) / (2 * credits)
279 return "\(value / 100).\(String(format: "%02d", value % 100))"
280 }
281}

320 private func conflict(_ student: Student, _ course: Course, excluding: String? = nil) -> String? {
321 student.enrolled.intersection(course.conflicts)
322 .filter { $0 != excluding }.sorted().first
323 }

325 private func eligibility(_ student: Student, _ course: Course,
326 excluding: String? = nil) -> Decision? {
327 let code = course.code
328 if !course.isOpen { return Decision(kind: "course closed", extra: []) }
329 if student.enrolled.contains(code) { return Decision(kind: "already enrolled", extra: []) }
330 if student.waiting.contains(code) { return Decision(kind: "already waitlisted", extra: []) }
331 if let grade = student.passed[code] {
332 return Decision(kind: "already completed", extra: [grade])
333 }
334 if !course.prerequisite.evaluates(student) {
335 return Decision(kind: "prereq not met", extra: [])
336 }
337 if let conflict = conflict(student, course, excluding: excluding) {
338 return Decision(kind: "time conflict", extra: [conflict])
339 }
340 let excludedCredits = excluding.flatMap { courses[$0]?.credits } ?? 0
341 let total = student.credits - excludedCredits + course.credits
342 if total > student.limit {
343 return Decision(kind: "credit limit", extra: [String(total), String(student.limit)])
344 }
345 return nil
346 }
347
348 private func waitPosition(_ student: Student, _ course: Course) -> Int {
349 1 + course.waitlist.filter { students[$0]!.year >= student.year }.count
350 }
351
352 private func request(_ student: Student, _ course: Course) -> Decision {
353 if let denied = eligibility(student, course) { return denied }
354 if course.enrolled.count < course.capacity {
355 return Decision(kind: "enroll", extra: [])
356 }
357 if student.waiting.count >= 3 {
358 return Decision(kind: "waitlist limit", extra: [])
359 }
360 return Decision(kind: "waitlist", extra: [String(waitPosition(student, course))])
361 }

363 private func failRequest(_ decision: Decision, _ student: Student,
364 _ course: Course) throws -> Never {
365 try fail((["\(decision.kind) \(student.identifier) \(course.code)"] + decision.extra)
366 .joined(separator: " "))
367 }

386 private func promotion() {
387 for code in courses.keys.sorted() {
388 guard let course = courses[code], course.isOpen else { continue }
389 for identifier in course.waitlist {
390 if course.enrolled.count >= course.capacity { break }
391 let student = students[identifier]!
392 if conflict(student, course) != nil
393 || student.credits + course.credits > student.limit { continue }
394 unlist(student, course)
395 enroll(student, course)
396 record(student, "promote \(code)")
397 emit("PROMOTE \(identifier) \(code) enrolled \(course.enrolled.count)/\(course.capacity) credits \(student.credits)/\(student.limit)", indented: true)
398 }
399 }

402 private func cmdCourse(_ a: [String]) throws {
403 let code = a[0]
404 if courses[code] != nil { try fail("duplicate course \(code)") }
405 let slots = try a[3] == "-" ? [] : a[3].split(separator: ",").map { try Slot(String($0)) }
406 for i in slots.indices {
407 for j in (i + 1)..<slots.count where slots[i].overlaps(slots[j]) {
408 try fail("overlapping slots \(slots[i].original) \(slots[j].original)")
409 }
410 }
411 let expression = try prerequisite(code, a[4])
412 let course = Course(code, Int(a[1])!, Int(a[2])!, slots, expression)
413 for other in courses.values {
414 if slots.contains(where: { first in other.slots.contains { first.overlaps($0) } }) {
415 course.conflicts.insert(other.code)
416 other.conflicts.insert(code)
417 }
418 }
419 courses[code] = course
420 emit("COURSE \(code) credits \(a[1]) capacity \(a[2]) meetings \(slots.count) weekly \(clock(course.weekly)) prereq \(expression.canonical)")
421 }

423 private func cmdStudent(_ a: [String]) throws {
424 let identifier = a[0]
425 if students[identifier] != nil { try fail("duplicate student \(identifier)") }
426 var records: [String: (grade: String, credits: Int)] = [:]
427 if a[3] != "-" {
428 for raw in a[3].split(separator: ",") {
429 let pieces = String(raw).split(separator: ":").map(String.init)
430 let code = pieces[0], grade = pieces[1]
431 if gradePoints[grade] == nil {
432 try fail("invalid grade \(identifier) \(code) \(grade)")
433 }
434 if records[code] != nil {
435 try fail("duplicate record \(identifier) \(code)")
436 }
437 records[code] = (grade, Int(pieces[2])!)
438 }
439 }
440 let student = Student(identifier, Int(a[1])!, Int(a[2])!, records)
441 students[identifier] = student
442 emit("STUDENT \(identifier) year \(a[1]) limit \(a[2]) records \(records.count) earned \(student.earned) gpa \(student.gpa)")
443 }

445 private func cmdEnroll(_ a: [String]) throws {
446 let student = try student(a[0])
447 let course = try course(a[1])
448 let decision = request(student, course)
449 if decision.kind != "enroll" && decision.kind != "waitlist" {
450 try failRequest(decision, student, course)
451 }
452 if decision.kind == "enroll" {
453 enroll(student, course)
454 record(student, "enroll \(course.code)")
455 emit("ENROLL \(student.identifier) \(course.code) enrolled \(course.enrolled.count)/\(course.capacity) credits \(student.credits)/\(student.limit)")
456 } else {
457 let position = Int(decision.extra[0])!
458 course.waitlist.insert(student.identifier, at: position - 1)
459 student.waiting.insert(course.code)
460 record(student, "waitlist \(course.code) \(position)")
461 emit("ENROLL \(student.identifier) \(course.code) waitlist \(position)")
462 }

625 private func cmdFree(_ a: [String]) throws {
626 let student = try student(a[0])
627 guard let day = days.firstIndex(of: a[1]) else { try fail("invalid day \(a[1])") }
628 let slots = studentSlots(student, day: day)
629 var free: [(Int, Int)] = []
630 var cursor = 480
631 for slot in slots {
632 let start = max(480, slot.start), end = min(1320, slot.end)
633 if start < end {
634 if cursor < start { free.append((cursor, start)) }
635 cursor = max(cursor, end)
636 }
637 }
638 if cursor < 1320 { free.append((cursor, 1320)) }
639 let total = free.reduce(0) { $0 + $1.1 - $1.0 }
640 let values = free.map { "\(clock($0.0))-\(clock($0.1))" }
641 emit((["FREE \(a[0]) \(a[1]) total \(clock(total))"] + values).joined(separator: " "))
642 }

673 for code in student.waiting.sorted() {
674 let position = courses[code]!.waitlist.firstIndex(of: student.identifier)! + 1
675 emit("waitlist \(code) \(position)", indented: true)
676 }
677 }

679 private func cmdHistory(_ a: [String]) throws {
680 let student = try student(a[0])
681 emit("HISTORY \(a[0]) events \(student.history.count)")
682 for event in student.history {
683 emit("\(event.number) \(event.event)", indented: true)
684 }
685 }

687 private func cmdStudents() {
688 emit("STUDENTS \(students.count)")
689 for identifier in students.keys.sorted() {
690 let student = students[identifier]!
691 emit("\(identifier) year \(student.year) credits \(student.credits)/\(student.limit) courses \(student.enrolled.count) waitlists \(student.waiting.count)", indented: true)
692 }
693 }
694
695 private func groupCounts(_ courses: [Course]) -> String {
696 let enrolled = courses.reduce(0) { $0 + $1.enrolled.count }
697 let capacity = courses.reduce(0) { $0 + $1.capacity }
698 let waitlisted = courses.reduce(0) { $0 + $1.waitlist.count }
699 return "enrolled \(enrolled)/\(capacity) rate \(rate(enrolled, capacity)) waitlisted \(waitlisted)"
700 }
701
702 private func cmdReport() {
703 emit("REPORT courses \(courses.count) \(groupCounts(Array(courses.values)))")
704 var departments: [String: [Course]] = [:]
705 for code in courses.keys.sorted() {
706 let course = courses[code]!
707 emit("\(code) \(course.status) \(course.enrolled.count)/\(course.capacity) \(rate(course.enrolled.count, course.capacity)) waitlist \(course.waitlist.count)", indented: true)
708 departments[course.department, default: []].append(course)
709 }
710 for department in departments.keys.sorted() {
711 let group = departments[department]!
712 emit("dept \(department) courses \(group.count) \(groupCounts(group))", indented: true)
713 }
714 }

716 func execute(_ line: String) {
717 commandNumber += 1
718 let parts = line.split(separator: " ").map(String.init)
719 guard let command = parts.first else { return }
720 let a = Array(parts.dropFirst())
721 do {
722 switch command {
723 case "COURSE": try cmdCourse(a)
724 case "STUDENT": try cmdStudent(a)
725 case "ENROLL": try cmdEnroll(a)
726 case "CHECK": try cmdCheck(a)
727 case "OPTIONS": try cmdOptions(a)
728 case "DROP": try cmdDrop(a)
729 case "SWAP": try cmdSwap(a)
730 case "CLOSE": try cmdClose(a)
731 case "OPEN": try cmdOpen(a)
732 case "CAPACITY": try cmdCapacity(a)
733 case "LIMIT": try cmdLimit(a)
734 case "PREREQ": try cmdPrereq(a)
735 case "CANCEL": try cmdCancel(a)
736 case "ROSTER": try cmdRoster(a)
737 case "SCHEDULE": try cmdSchedule(a)
738 case "FREE": try cmdFree(a)
739 case "INFO": try cmdInfo(a)
740 case "AUDIT": try cmdAudit(a)
741 case "HISTORY": try cmdHistory(a)
742 case "STUDENTS": cmdStudents()
743 case "REPORT": cmdReport()
744 default: break
745 }
746 } catch let error as CommandError {
747 emit("ERROR \(error.message)")
748 } catch {
749 fatalError("Unexpected error: \(error)")
750 }
751 }
752}

758}
