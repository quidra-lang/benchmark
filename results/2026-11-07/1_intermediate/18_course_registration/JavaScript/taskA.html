4const DAYS = ["MO", "TU", "WE", "TH", "FR", "SA", "SU"];
5const POINTS = {A: 4, B: 3, C: 2, D: 1, F: 0};
6const lex = (a, b) => a < b ? -1 : a > b ? 1 : 0;
7const time = (m) => String(Math.floor(m / 60)).padStart(2, "0") + ":" + String(m % 60).padStart(2, "0");
8const overlap = (a, b) => a.day === b.day && a.start < b.end && b.start < a.end;
9const weekly = (c) => c.slots.reduce((v, s) => v + s.end - s.start, 0);
10const earned = (s) => s.grades.filter(r => r.grade !== "F").reduce((v, r) => v + r.credits, 0);
11const passed = (s, code) => s.grades.find(r => r.code === code && ["A", "B", "C"].includes(r.grade));
12const rate = (k, m) => m ? Math.floor(k * 100 / m) + "%" : "n/a";
13class CommandError extends Error {}

15function gpa(s) {
16 if (!s.grades.length) return "n/a";
17 const p = s.grades.reduce((v, r) => v + POINTS[r.grade] * r.credits, 0);
18 const c = s.grades.reduce((v, r) => v + r.credits, 0);
19 const v = Math.floor((200 * p + c) / (2 * c));
20 return Math.floor(v / 100) + "." + String(v % 100).padStart(2, "0");
21}

23function combine(kind, children) {
24 const flat = children.flatMap(c => c.kind === kind ? c.children : [c]);
25 return flat.length === 1 ? flat[0] : {kind, word: "", children: flat};
26}
27
28function canonical(node) {
29 if (node === null) return "-";
30 if (node.kind === "leaf") return node.word;
31 return node.children.map(c => node.kind === "and" && c.kind === "or" ? "(" + canonical(c) + ")" : canonical(c)).join(node.kind === "and" ? "&" : "|");
32}
33
34function contains(node, code) {
35 return node !== null && (node.kind === "leaf" ? node.word === code : node.children.some(c => contains(c, code)));
36}
37
38function satisfied(node, s) {
39 if (node === null) return true;
40 if (node.kind === "leaf") return /^Y[1-9]$/.test(node.word) ? s.year >= Number(node.word[1]) : passed(s, node.word) !== undefined;
41 return node.kind === "and" ? node.children.every(c => satisfied(c, s)) : node.children.some(c => satisfied(c, s));
42}

44function prerequisite(code, text) {
45 if (text === "-") return null;
46 const tokens = text.match(/[A-Z0-9]+|[&|()]/g) ?? [];
47 const invalid = () => { throw new CommandError("invalid prereq " + code); };
48 for (const t of tokens) {
49 if (!["&", "|", "(", ")"].includes(t) && !/^(?:[A-Z]{2,4}[0-9]{3}|Y[1-9])$/.test(t)) invalid();
50 }
51 let position = 0;
52 const factor = () => {
53 const token = tokens[position++];
54 if (token === "(") {
55 const node = expression();
56 if (tokens[position++] !== ")") invalid();
57 return node;
58 }
59 if (token === undefined || ["&", "|", ")"].includes(token)) return invalid();
60 return {kind: "leaf", word: token, children: []};
61 };
62 const term = () => {
63 const nodes = [factor()];
64 while (tokens[position] === "&") {
65 position++;
66 nodes.push(factor());
67 }
68 return combine("and", nodes);
69 };
70 const expression = () => {
71 const nodes = [term()];
72 while (tokens[position] === "|") {
73 position++;
74 nodes.push(term());
75 }
76 return combine("or", nodes);
77 };
78 const node = expression();
79 if (position !== tokens.length) invalid();
80 if (contains(node, code)) throw new CommandError("self prereq " + code);
81 return node;
82}

84function slots(text) {
85 if (text === "-") return [];
86 const result = text.split(",").map(raw => {
87 const day = raw.slice(0, 2);
88 const sh = Number(raw.slice(2, 4));
89 const sm = Number(raw.slice(4, 6));
90 const eh = Number(raw.slice(7, 9));
91 const em = Number(raw.slice(9, 11));
92 const start = sh * 60 + sm;
93 const end = eh * 60 + em;
94 if (!DAYS.includes(day) || sh > 23 || sm > 59 || eh > 23 || em > 59 || start >= end) throw new CommandError("invalid slot " + raw);
95 return {day, start, end, raw};
96 });
97 for (let i = 0; i < result.length; i++) {
98 for (let j = i + 1; j < result.length; j++) {
99 if (overlap(result[i], result[j])) throw new CommandError("overlapping slots " + result[i].raw + " " + result[j].raw);
100 }
101 }
102 return result;
103}

105class Registrar {
106 courses = new Map();
107 students = new Map();
108 commandNumber = 0;
109 output = [];
110
111 emit(indent, ...parts) {
112 this.output.push(" ".repeat(indent) + parts.join(" "));
113 }
114
115 student(id) {
116 const s = this.students.get(id);
117 if (!s) throw new CommandError("unknown student " + id);
118 return s;
119 }
120
121 course(code) {
122 const c = this.courses.get(code);
123 if (!c) throw new CommandError("unknown course " + code);
124 return c;
125 }
126
127 sortedCourses() {
128 return [...this.courses.values()].sort((a, b) => lex(a.code, b.code));
129 }
130
131 taking(s, except = null) {
132 return [...s.taking].filter(c => c !== except).sort().map(code => this.courses.get(code));
133 }
134
135 credits(s, except = null) {
136 return this.taking(s, except).reduce((v, c) => v + c.credits, 0);
137 }
138
139 waitlists(s) {
140 return this.sortedCourses().filter(c => c.waiting.includes(s.id));
141 }
142
143 conflict(s, c, except = null) {
144 return this.taking(s, except).find(other => other.slots.some(a => c.slots.some(b => overlap(a, b))));
145 }
146
147 eligible(s, c, except = null) {
148 if (!c.open) return {word: "course closed", extra: []};
149 if (c.enrolled.has(s.id)) return {word: "already enrolled", extra: []};
150 if (c.waiting.includes(s.id)) return {word: "already waitlisted", extra: []};
151 const grade = passed(s, c.code);
152 if (grade) return {word: "already completed", extra: [grade.grade]};
153 if (!satisfied(c.prerequisite, s)) return {word: "prereq not met", extra: []};
154 const conflicting = this.conflict(s, c, except);
155 if (conflicting) return {word: "time conflict", extra: [conflicting.code]};
156 const credits = this.credits(s, except) + c.credits;
157 if (credits > s.limit) return {word: "credit limit", extra: [credits, s.limit]};
158 return {word: "OK", extra: []};
159 }
160
161 judgment(s, c) {
162 const result = this.eligible(s, c);
163 if (result.word !== "OK") return result;
164 if (c.enrolled.size < c.capacity) return {word: "enroll", extra: []};
165 if (this.waitlists(s).length >= 3) return {word: "waitlist limit", extra: []};
166 return {word: "waitlist", extra: [1 + c.waiting.filter(id => this.students.get(id).year >= s.year).length]};
167 }
168
169 reject(s, c, result) {
170 throw new CommandError([result.word, s.id, c.code, ...result.extra].join(" "));
171 }
172
173 history(s, event) {
174 s.history.push({command: this.commandNumber, event});
175 }
176
177 enroll(s, c) {
178 c.enrolled.add(s.id);
179 s.taking.add(c.code);
180 }
181
182 remove(s, c) {
183 c.enrolled.delete(s.id);
184 s.taking.delete(c.code);
185 }
186
187 promote() {
188 for (const c of this.sortedCourses()) {
189 if (!c.open) continue;
190 while (c.enrolled.size < c.capacity) {
191 const index = c.waiting.findIndex(id => {
192 const s = this.students.get(id);
193 return !this.conflict(s, c) && this.credits(s) + c.credits <= s.limit;
194 });
195 if (index < 0) break;
196 const [id] = c.waiting.splice(index, 1);
197 const s = this.students.get(id);
198 this.enroll(s, c);
199 this.history(s, "promote " + c.code);
200 this.enrollmentLine("PROMOTE", s, c, 2);
201 }
202 }
203 }
204
205 enrollmentLine(label, s, c, indent = 0) {
206 this.emit(indent, label, s.id, c.code, "enrolled", c.enrolled.size + "/" + c.capacity, "credits", this.credits(s) + "/" + s.limit);
207 }
208
209 schedule(s) {
210 const taking = this.taking(s);
211 this.emit(0, "SCHEDULE", s.id, "courses", taking.length, "credits", this.credits(s), "weekly", time(taking.reduce((v, c) => v + weekly(c), 0)));
212 for (const day of DAYS) {
213 const meetings = taking.flatMap(c => c.slots.filter(slot => slot.day === day).map(slot => ({code: c.code, slot})));
214 meetings.sort((a, b) => a.slot.start - b.slot.start);
215 if (meetings.length) this.emit(2, day, ...meetings.flatMap(m => [time(m.slot.start) + "-" + time(m.slot.end), m.code]));
216 }
217 const tba = taking.filter(c => !c.slots.length).map(c => c.code);
218 if (tba.length) this.emit(2, "TBA", ...tba);
219 }

221 command(cmd, a) {
222 switch (cmd) {
223 case "COURSE": {
224 if (this.courses.has(a[0])) throw new CommandError("duplicate course " + a[0]);
225 const meetings = slots(a[3]);
226 const tree = prerequisite(a[0], a[4]);
227 const c = {code: a[0], credits: Number(a[1]), capacity: Number(a[2]), slots: meetings,
228 prerequisite: tree, open: true, enrolled: new Set(), waiting: []};
229 this.courses.set(c.code, c);
230 this.emit(0, "COURSE", c.code, "credits", c.credits, "capacity", c.capacity, "meetings", meetings.length, "weekly", time(weekly(c)), "prereq", canonical(tree));
231 break;
232 }
233 case "STUDENT": {
234 if (this.students.has(a[0])) throw new CommandError("duplicate student " + a[0]);
235 const grades = [];
236 const seen = new Set();
237 for (const token of a[3] === "-" ? [] : a[3].split(",")) {
238 const [code, grade, credits] = token.split(":");
239 if (!Object.hasOwn(POINTS, grade)) throw new CommandError("invalid grade " + a[0] + " " + code + " " + grade);
240 if (seen.has(code)) throw new CommandError("duplicate record " + a[0] + " " + code);
241 seen.add(code);
242 grades.push({code, grade, credits: Number(credits)});
243 }
244 const s = {id: a[0], year: Number(a[1]), limit: Number(a[2]), grades, taking: new Set(), history: []};
245 this.students.set(s.id, s);
246 this.emit(0, "STUDENT", s.id, "year", s.year, "limit", s.limit, "records", grades.length, "earned", earned(s), "gpa", gpa(s));
247 break;
248 }
249 case "ENROLL":
250 case "CHECK": {
251 const s = this.student(a[0]);
252 const c = this.course(a[1]);
253 const result = this.judgment(s, c);
254 if (cmd === "CHECK") this.emit(0, "CHECK", s.id, c.code, result.word, ...result.extra);
255 else if (result.word === "enroll") {
256 this.enroll(s, c);
257 this.history(s, "enroll " + c.code);
258 this.enrollmentLine("ENROLL", s, c);
259 } else if (result.word === "waitlist") {
260 const position = Number(result.extra[0]);
261 c.waiting.splice(position - 1, 0, s.id);
262 this.history(s, "waitlist " + c.code + " " + position);
263 this.emit(0, "ENROLL", s.id, c.code, "waitlist", position);
264 } else this.reject(s, c, result);
265 break;
266 }
267 case "OPTIONS": {
268 const s = this.student(a[0]);
269 this.emit(0, "OPTIONS", s.id, "courses", this.courses.size);
270 for (const c of this.sortedCourses()) {
271 const result = this.judgment(s, c);
272 this.emit(2, c.code, result.word, ...result.extra);
273 }
274 break;
275 }
276 case "DROP": {
277 const s = this.student(a[0]);
278 const c = this.course(a[1]);
279 const position = c.waiting.indexOf(s.id);
280 if (c.enrolled.has(s.id)) {
281 this.remove(s, c);
282 this.history(s, "drop " + c.code);
283 this.enrollmentLine("DROP", s, c);
284 } else if (position >= 0) {
285 c.waiting.splice(position, 1);
286 this.history(s, "drop " + c.code + " waitlist");
287 this.emit(0, "DROP", s.id, c.code, "waitlist", "removed");
288 } else throw new CommandError("not registered " + s.id + " " + c.code);
289 this.promote();
290 break;
291 }
292 case "SWAP": {
293 const s = this.student(a[0]);
294 const old = this.course(a[1]);
295 const next = this.course(a[2]);
296 if (!old.enrolled.has(s.id)) throw new CommandError("not enrolled " + s.id + " " + old.code);
297 if (old === next) throw new CommandError("same course " + old.code);
298 const result = this.eligible(s, next, old.code);
299 if (result.word !== "OK") this.reject(s, next, result);
300 if (next.enrolled.size >= next.capacity) throw new CommandError("course full " + next.code);
301 this.remove(s, old);
302 this.enroll(s, next);
303 this.history(s, "swap " + old.code + " " + next.code);
304 this.emit(0, "SWAP", s.id, old.code, "->", next.code, "credits", this.credits(s) + "/" + s.limit);
305 this.promote();
306 break;
307 }
308 case "CLOSE":
309 case "OPEN": {
310 const c = this.course(a[0]);
311 if (c.open === (cmd === "OPEN")) throw new CommandError("already " + cmd.toLowerCase() + (cmd === "CLOSE" ? "d " : " ") + c.code);
312 c.open = cmd === "OPEN";
313 this.emit(0, cmd, c.code, "enrolled", c.enrolled.size + "/" + c.capacity, "waitlist", c.waiting.length);
314 if (cmd === "OPEN") this.promote();
315 break;
316 }
317 case "CAPACITY": {
318 const c = this.course(a[0]);
319 const capacity = Number(a[1]);
320 if (capacity < c.enrolled.size) throw new CommandError("capacity below enrollment " + c.code + " " + c.enrolled.size);
321 const old = c.capacity;
322 c.capacity = capacity;
323 this.emit(0, "CAPACITY", c.code, old, "->", capacity);
324 this.promote();
325 break;
326 }
327 case "LIMIT": {
328 const s = this.student(a[0]);
329 const limit = Number(a[1]);
330 const credits = this.credits(s);
331 if (limit < credits) throw new CommandError("limit below enrolled " + s.id + " " + credits);
332 const old = s.limit;
333 s.limit = limit;
334 this.emit(0, "LIMIT", s.id, old, "->", limit);
335 this.promote();
336 break;
337 }
338 case "PREREQ": {
339 const c = this.course(a[0]);
340 const tree = prerequisite(c.code, a[1]);
341 c.prerequisite = tree;
342 this.emit(0, "PREREQ", c.code, canonical(tree));
343 c.waiting = c.waiting.filter(id => {
344 const s = this.students.get(id);
345 if (satisfied(tree, s)) return true;
346 this.history(s, "unlist " + c.code);
347 this.emit(2, "UNLIST", s.id, c.code);
348 return false;
349 });
350 break;
351 }
352 case "CANCEL": {
353 const c = this.course(a[0]);
354 this.emit(0, "CANCEL", c.code, "enrolled", c.enrolled.size, "waitlist", c.waiting.length);
355 for (const id of c.enrolled) {
356 const s = this.students.get(id);
357 s.taking.delete(c.code);
358 this.history(s, "cancel " + c.code);
359 }
360 for (const id of c.waiting) this.history(this.students.get(id), "cancel " + c.code + " waitlist");
361 this.courses.delete(c.code);
362 this.promote();
363 break;
364 }
365 case "ROSTER": {
366 const c = this.course(a[0]);
367 this.emit(0, "ROSTER", c.code, c.open ? "open" : "closed", "enrolled", c.enrolled.size + "/" + c.capacity, "waitlist", c.waiting.length);
368 for (const id of [...c.enrolled].sort()) this.emit(2, "enrolled", id, "year", this.students.get(id).year);
369 c.waiting.forEach((id, i) => this.emit(2, "waitlist", i + 1, id, "year", this.students.get(id).year));
370 break;
371 }
372 case "SCHEDULE":
373 this.schedule(this.student(a[0]));
374 break;
375 case "FREE": {
376 const s = this.student(a[0]);
377 if (!DAYS.includes(a[1])) throw new CommandError("invalid day " + a[1]);
378 const meetings = this.taking(s).flatMap(c => c.slots).filter(t => t.day === a[1] && t.start < 1320 && t.end > 480).sort((a, b) => a.start - b.start);
379 const ranges = [];
380 let cursor = 480;
381 for (const slot of meetings) {
382 const start = Math.max(slot.start, 480);
383 if (start > cursor) ranges.push([cursor, start]);
384 cursor = Math.max(cursor, Math.min(slot.end, 1320));
385 }
386 if (cursor < 1320) ranges.push([cursor, 1320]);
387 this.emit(0, "FREE", s.id, a[1], "total", time(ranges.reduce((v, [a, b]) => v + b - a, 0)),
388 ...ranges.map(([a, b]) => time(a) + "-" + time(b)));
389 break;
390 }
391 case "INFO": {
392 const c = this.course(a[0]);
393 this.emit(0, "INFO", c.code, c.open ? "open" : "closed", "credits", c.credits, "capacity", c.capacity, "prereq", canonical(c.prerequisite));
394 for (const slot of c.slots.toSorted((a, b) => DAYS.indexOf(a.day) - DAYS.indexOf(b.day) || a.start - b.start)) this.emit(2, "meet", slot.day, time(slot.start) + "-" + time(slot.end));
395 if (!c.slots.length) this.emit(2, "meet", "TBA");
396 const required = this.sortedCourses().filter(other => contains(other.prerequisite, c.code)).map(other => other.code);
397 this.emit(2, "required", "by", ...(required.length ? required : ["none"]));
398 break;
399 }
400 case "AUDIT": {
401 const s = this.student(a[0]);
402 this.emit(0, "AUDIT", s.id, "year", s.year, "limit", s.limit);
403 for (const r of s.grades.toSorted((a, b) => lex(a.code, b.code))) this.emit(2, "record", r.code, r.grade, r.credits);
404 this.emit(2, "earned", earned(s), "gpa", gpa(s));
405 for (const c of this.taking(s)) this.emit(2, "enrolled", c.code, c.credits);
406 this.emit(2, "credits", this.credits(s), "remaining", s.limit - this.credits(s));
407 for (const c of this.waitlists(s)) this.emit(2, "waitlist", c.code, c.waiting.indexOf(s.id) + 1);
408 break;
409 }
410 case "HISTORY": {
411 const s = this.student(a[0]);
412 this.emit(0, "HISTORY", s.id, "events", s.history.length);
413 for (const row of s.history) this.emit(2, row.command, row.event);
414 break;
415 }
416 case "STUDENTS":
417 this.emit(0, "STUDENTS", this.students.size);
418 for (const s of [...this.students.values()].sort((a, b) => lex(a.id, b.id))) this.emit(2, s.id, "year", s.year,
419 "credits", this.credits(s) + "/" + s.limit, "courses", s.taking.size, "waitlists", this.waitlists(s).length);
420 break;
421 case "REPORT": {
422 const courses = this.sortedCourses();
423 const enrolled = courses.reduce((v, c) => v + c.enrolled.size, 0);
424 const capacity = courses.reduce((v, c) => v + c.capacity, 0);
425 const waiting = courses.reduce((v, c) => v + c.waiting.length, 0);
426 this.emit(0, "REPORT", "courses", courses.length, "enrolled", enrolled + "/" + capacity, "rate", rate(enrolled, capacity), "waitlisted", waiting);
427 for (const c of courses) this.emit(2, c.code, c.open ? "open" : "closed", c.enrolled.size + "/" + c.capacity, rate(c.enrolled.size, c.capacity), "waitlist", c.waiting.length);
428 const departments = Map.groupBy(courses, c => c.code.slice(0, -3));
429 for (const department of [...departments.keys()].sort()) {
430 const group = departments.get(department);
431 const k = group.reduce((v, c) => v + c.enrolled.size, 0);
432 const m = group.reduce((v, c) => v + c.capacity, 0);
433 this.emit(2, "dept", department, "courses", group.length, "enrolled", k + "/" + m,
434 "rate", rate(k, m), "waitlisted", group.reduce((v, c) => v + c.waiting.length, 0));
435 }
436 break;
437 }
438 }
439 }

441 execute(line) {
442 this.commandNumber++;
443 const [cmd, ...args] = line.split(" ");
444 try {
445 this.command(cmd, args);
446 } catch (error) {
447 if (!(error instanceof CommandError)) throw error;
448 this.emit(0, "ERROR", error.message);
449 }
450 }
