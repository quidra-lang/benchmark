33class Expression:
34 def __init__(self, kind, children=(), word=None):
35 self.kind = kind
36 self.children = children
37 self.word = word
38 self.references = {word} if kind == "WORD" and not year_word(word) else set()
39 for child in children:
40 self.references.update(child.references)
41
42 def canonical(self):
43 if self.kind == "TRUE":
44 return "-"
45 if self.kind == "WORD":
46 return self.word
47 parts = []
48 for child in self.children:
49 text = child.canonical()
50 if self.kind == "AND" and child.kind == "OR":
51 text = "(" + text + ")"
52 parts.append(text)
53 return ("&" if self.kind == "AND" else "|").join(parts)
54
55 def evaluate(self, student):
56 if self.kind == "TRUE":
57 return True
58 if self.kind == "WORD":
59 return student.year >= int(self.word[1]) if year_word(self.word) else self.word in student.passed
60 if self.kind == "AND":
61 return all(child.evaluate(student) for child in self.children)
62 return any(child.evaluate(student) for child in self.children)
63
64
65def combine(kind, children):
66 flattened = []
67 for child in children:
68 if child.kind == kind:
69 flattened.extend(child.children)
70 else:
71 flattened.append(child)
72 return flattened[0] if len(flattened) == 1 else Expression(kind, flattened)

75class PrerequisiteParser:
76 def __init__(self, text):
77 self.tokens = re.findall(r"[A-Z0-9]+|[&|()]", text)
78 self.position = 0
79 for token in self.tokens:
80 if token not in "&|()" and not (course_word(token) or year_word(token)):
81 raise ParseError()
82
83 def peek(self):
84 return self.tokens[self.position] if self.position < len(self.tokens) else None
85
86 def factor(self):
87 token = self.peek()
88 if token is None or token in ("&", "|", ")"):
89 raise ParseError()
90 self.position += 1
91 if token == "(":
92 expression = self.expression()
93 if self.peek() != ")":
94 raise ParseError()
95 self.position += 1
96 return expression
97 return Expression("WORD", word=token)
98
99 def term(self):
100 children = [self.factor()]
101 while self.peek() == "&":
102 self.position += 1
103 children.append(self.factor())
104 return combine("AND", children)
105
106 def expression(self):
107 children = [self.term()]
108 while self.peek() == "|":
109 self.position += 1
110 children.append(self.term())
111 return combine("OR", children)
112
113 def parse(self):
114 result = self.expression()
115 if self.position != len(self.tokens):
116 raise ParseError()
117 return result

120def prerequisite(code, text):
121 if text == "-":
122 return Expression("TRUE")
123 try:
124 expression = PrerequisiteParser(text).parse()
125 except ParseError:
126 raise CommandError(f"invalid prereq {code}") from None
127 if code in expression.references:
128 raise CommandError(f"self prereq {code}")
129 return expression

132class Slot:
133 def __init__(self, text):
134 day = text[:2]
135 start_hour, start_minute = int(text[2:4]), int(text[4:6])
136 end_hour, end_minute = int(text[7:9]), int(text[9:11])
137 if day not in DAYS or start_hour > 23 or end_hour > 23 or start_minute > 59 or end_minute > 59:
138 raise CommandError(f"invalid slot {text}")
139 self.start = start_hour * 60 + start_minute
140 self.end = end_hour * 60 + end_minute
141 if self.start >= self.end:
142 raise CommandError(f"invalid slot {text}")
143 self.day = DAYS.index(day)
144 self.original = text
145
146 def overlaps(self, other):
147 return self.day == other.day and self.start < other.end and other.start < self.end
148
149 def text(self):
150 return time_text(self.start) + "-" + time_text(self.end)

175class Student:
176 def __init__(self, identifier, year, limit, records):
177 self.identifier = identifier
178 self.year = year
179 self.limit = limit
180 self.records = records
181 self.passed = {code: grade for code, (grade, _) in records.items() if grade in ("A", "B", "C")}
182 self.courses = set()
183 self.waiting = set()
184 self.credits = 0
185 self.history = []
186
187 def earned(self):
188 return sum(credits for grade, credits in self.records.values() if grade != "F")
189
190 def gpa(self):
191 if not self.records:
192 return "n/a"
193 points = sum(GRADE_POINTS[grade] * credits for grade, credits in self.records.values())
194 credits = sum(credits for _, credits in self.records.values())
195 value = (200 * points + credits) // (2 * credits)
196 return f"{value // 100}.{value % 100:02d}"

222 def conflict(self, student, course, excluded=None):
223 codes = student.courses & course.conflicts
224 codes.discard(excluded)
225 return min(codes) if codes else None

227 def eligibility(self, student, course, excluded=None):
228 if not course.open:
229 return "course closed", ()
230 if course.code in student.courses:
231 return "already enrolled", ()
232 if course.code in student.waiting:
233 return "already waitlisted", ()
234 if course.code in student.passed:
235 return "already completed", (student.passed[course.code],)
236 if not course.prerequisite.evaluate(student):
237 return "prereq not met", ()
238 conflict = self.conflict(student, course, excluded)
239 if conflict is not None:
240 return "time conflict", (conflict,)
241 total = student.credits + course.credits
242 if excluded is not None:
243 total -= self.courses[excluded].credits
244 if total > student.limit:
245 return "credit limit", (total, student.limit)
246 return None

248 def wait_position(self, student, course):
249 return 1 + sum(other.year >= student.year for other in course.waitlist)

251 def request(self, student, course):
252 failure = self.eligibility(student, course)
253 if failure is not None:
254 return failure
255 if len(course.enrolled) < course.capacity:
256 return "enroll", ()
257 if len(student.waiting) >= 3:
258 return "waitlist limit", ()
259 return "waitlist", (self.wait_position(student, course),)

261 def fail_request(self, decision, student, course):

263 raise CommandError(" ".join(str(part) for part in (reason, student.identifier, course.code, *extra)))

279 def promotion(self):
280 for code in sorted(self.courses):
281 course = self.courses[code]
282 if not course.open:
283 continue
284 for student in course.waitlist.copy():
285 if len(course.enrolled) >= course.capacity:
286 break
287 if self.conflict(student, course) is not None or student.credits + course.credits > student.limit:
288 continue
289 self.unlist(student, course)
290 self.enroll(student, course)
291 self.record(student, "promote " + code)
292 self.emit(
293 "PROMOTE", student.identifier, code, "enrolled", f"{len(course.enrolled)}/{course.capacity}",
294 "credits", f"{student.credits}/{student.limit}", indent=2,
295 )

297 def cmd_course(self, code, credits, capacity, timetable, text):
298 if code in self.courses:
299 raise CommandError(f"duplicate course {code}")
300 slots = [] if timetable == "-" else [Slot(value) for value in timetable.split(",")]
301 for first in range(len(slots)):
302 for second in range(first + 1, len(slots)):
303 if slots[first].overlaps(slots[second]):
304 raise CommandError(f"overlapping slots {slots[first].original} {slots[second].original}")
305 expression = prerequisite(code, text)
306 course = Course(code, int(credits), int(capacity), slots, expression)
307 for other in self.courses.values():
308 if any(first.overlaps(second) for first in slots for second in other.slots):
309 course.conflicts.add(other.code)
310 other.conflicts.add(code)
311 self.courses[code] = course
312 self.emit(
313 "COURSE", code, "credits", credits, "capacity", capacity, "meetings", len(slots),
314 "weekly", time_text(course.weekly()), "prereq", expression.canonical(),
315 )

317 def cmd_student(self, identifier, year, limit, text):
318 if identifier in self.students:
319 raise CommandError(f"duplicate student {identifier}")
320 records = {}
321 if text != "-":
322 for value in text.split(","):
323 code, grade, credits = value.split(":")
324 if grade not in GRADE_POINTS:
325 raise CommandError(f"invalid grade {identifier} {code} {grade}")
326 if code in records:
327 raise CommandError(f"duplicate record {identifier} {code}")
328 records[code] = grade, int(credits)
329 student = Student(identifier, int(year), int(limit), records)
330 self.students[identifier] = student
331 self.emit(
332 "STUDENT", identifier, "year", year, "limit", limit, "records", len(records),
333 "earned", student.earned(), "gpa", student.gpa(),
334 )

336 def cmd_enroll(self, identifier, code):
337 student = self.student(identifier)
338 course = self.course(code)
339 decision = self.request(student, course)
340 kind, extra = decision
341 if kind not in ("enroll", "waitlist"):
342 self.fail_request(decision, student, course)
343 if kind == "enroll":
344 self.enroll(student, course)
345 self.record(student, "enroll " + code)
346 self.emit(
347 "ENROLL", identifier, code, "enrolled", f"{len(course.enrolled)}/{course.capacity}",
348 "credits", f"{student.credits}/{student.limit}",
349 )
350 else:
351 position = extra[0]
352 course.waitlist.insert(position - 1, student)
353 student.waiting.add(code)
354 self.record(student, f"waitlist {code} {position}")
355 self.emit("ENROLL", identifier, code, "waitlist", position)

388 def cmd_swap(self, identifier, first, second):
389 student = self.student(identifier)
390 source = self.course(first)
391 destination = self.course(second)
392 if first not in student.courses:
393 raise CommandError(f"not enrolled {identifier} {first}")
394 if first == second:
395 raise CommandError(f"same course {first}")
396 failure = self.eligibility(student, destination, first)
397 if failure is not None:
398 self.fail_request(failure, student, destination)
399 if len(destination.enrolled) >= destination.capacity:
400 raise CommandError(f"course full {second}")
401 self.withdraw(student, source)
402 self.enroll(student, destination)
403 self.record(student, f"swap {first} {second}")
404 self.emit("SWAP", identifier, first, "->", second, "credits", f"{student.credits}/{student.limit}")
405 self.promotion()

485 def cmd_schedule(self, identifier):
486 student = self.student(identifier)
487 weekly = sum(self.courses[code].weekly() for code in student.courses)

490 slots = self.student_slots(student, day)

492 parts = [name]

500 def cmd_free(self, identifier, day):
501 student = self.student(identifier)
502 if day not in DAYS:
503 raise CommandError(f"invalid day {day}")
504 occupied = []
505 for start, end, _ in self.student_slots(student, DAYS.index(day)):
506 start, end = max(480, start), min(1320, end)
507 if start < end:
508 occupied.append((start, end))
509 free = []
510 cursor = 480
511 for start, end in occupied:
512 if cursor < start:
513 free.append((cursor, start))
514 cursor = max(cursor, end)
515 if cursor < 1320:
516 free.append((cursor, 1320))
517 total = sum(end - start for start, end in free)
518 self.emit("FREE", identifier, day, "total", time_text(total), *(time_text(start) + "-" + time_text(end) for start, end in free))

582 def execute(self, text):
583 self.command += 1
584 command, *arguments = text.split()
585 try:
586 getattr(self, "cmd_" + command.lower())(*arguments)
587 except CommandError as error:
588 self.emit("ERROR", str(error))
