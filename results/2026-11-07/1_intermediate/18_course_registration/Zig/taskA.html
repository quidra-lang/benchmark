17const Node = union(enum) {
18 word: []const u8,
19 all: []const *Node,
20 any: []const *Node,
21 fn contains(node: *const Node, code: []const u8) bool {
22 switch (node.*) {
23 .word => |word| return equal(word, code),
24 .all, .any => |children| {
25 for (children) |child| if (child.contains(code)) return true;
26 return false;
27 },
28 }
29 }
30 fn satisfied(node: *const Node, s: *const Student) bool {
31 switch (node.*) {
32 .word => |word| return if (word.len == 2) s.year >= integer(word[1..]) else s.passed(word) != null,
33 .all => |children| {
34 for (children) |child| if (!child.satisfied(s)) return false;
35 return true;
36 },
37 .any => |children| {
38 for (children) |child| if (child.satisfied(s)) return true;
39 return false;
40 },
41 }
42 }
43 fn write(node: *const Node, out: *std.Io.Writer) std.Io.Writer.Error!void {
44 switch (node.*) {
45 .word => |word| try out.writeAll(word),
46 .all, .any => |children| {
47 const conjunction = node.* == .all;
48 for (children, 0..) |child, i| {
49 if (i > 0) try out.writeAll(if (conjunction) "&" else "|");
50 const parens = conjunction and child.* == .any;
51 if (parens) try out.writeAll("(");
52 try child.write(out);
53 if (parens) try out.writeAll(")");
54 }
55 },
56 }
57 }

59const Parser = struct {
60 gpa: Allocator,
61 text: []const u8,
62 position: usize = 0,
63 fn factor(self: *Parser) ParseError!*Node {
64 if (self.position == self.text.len) return error.InvalidPrereq;
65 if (self.text[self.position] == '(') {
66 self.position += 1;
67 const node = try self.expression();
68 if (self.position == self.text.len or self.text[self.position] != ')') return error.InvalidPrereq;
69 self.position += 1;
70 return node;
71 }
72 const start = self.position;
73 while (self.position < self.text.len and std.ascii.isAlphanumeric(self.text[self.position])) self.position += 1;
74 const word = self.text[start..self.position];
75 var letters: usize = 0;
76 while (letters < word.len and std.ascii.isUpper(word[letters])) letters += 1;
77 var course = letters >= 2 and letters <= 4 and word.len == letters + 3;
78 if (course) for (word[letters..]) |ch| {
79 if (!std.ascii.isDigit(ch)) course = false;
80 };
81 const year = word.len == 2 and word[0] == 'Y' and word[1] >= '1' and word[1] <= '9';
82 if (!course and !year) return error.InvalidPrereq;
83 const node = try self.gpa.create(Node);
84 node.* = .{ .word = word };
85 return node;
86 }
87 fn append(self: *Parser, children: *std.ArrayList(*Node), child: *Node, conjunction: bool) Allocator.Error!void {
88 if (conjunction and child.* == .all) {
89 try children.appendSlice(self.gpa, child.all);
90 } else if (!conjunction and child.* == .any) {
91 try children.appendSlice(self.gpa, child.any);
92 } else try children.append(self.gpa, child);
93 }
94 fn group(self: *Parser, conjunction: bool) ParseError!*Node {
95 var children: std.ArrayList(*Node) = .empty;
96 try self.append(&children, if (conjunction) try self.factor() else try self.term(), conjunction);
97 const separator: u8 = if (conjunction) '&' else '|';
98 while (self.position < self.text.len and self.text[self.position] == separator) {
99 self.position += 1;
100 try self.append(&children, if (conjunction) try self.factor() else try self.term(), conjunction);
101 }
102 if (children.items.len == 1) return children.items[0];
103 const node = try self.gpa.create(Node);
104 node.* = if (conjunction) .{ .all = children.items } else .{ .any = children.items };
105 return node;
106 }
107 fn term(self: *Parser) ParseError!*Node {
108 return self.group(true);
109 }
110 fn expression(self: *Parser) ParseError!*Node {
111 return self.group(false);
112 }

124const Slot = struct {
125 text: []const u8,
126 day: usize,
127 start: usize,
128 end: usize,
129 fn overlaps(a: Slot, b: Slot) bool {
130 return a.day == b.day and a.start < b.end and b.start < a.end;

205 fn clock(self: *Registration, minutes: usize) Allocator.Error![]const u8 {

208 fn gpaText(self: *Registration, s: *Student) Allocator.Error![]const u8 {
209 if (s.grades.len == 0) return "n/a";
210 var points: i64 = 0;
211 var credits: i64 = 0;
212 for (s.grades) |r| {
213 credits += r.credits;
214 const p: i64 = if (r.grade == 'F') 0 else 'E' - @as(i64, r.grade);
215 points += p * r.credits;
216 }
217 const value: u64 = @intCast(@divTrunc(200 * points + credits, 2 * credits));
218 return std.fmt.allocPrint(self.gpa, "{d}.{d:0>2}", .{ value / 100, value % 100 });
219 }
220 fn rate(self: *Registration, enrolled: usize, capacity: usize) Allocator.Error![]const u8 {
221 return if (capacity == 0) "n/a" else std.fmt.allocPrint(self.gpa, "{d}%", .{enrolled * 100 / capacity});
222 }
223 fn ordered(self: *Registration, comptime T: type, values: []const *T) Allocator.Error![]*T {
224 const result = try self.gpa.dupe(*T, values);

226 fn less(_: void, a: *T, b: *T) bool {
227 return std.mem.order(u8, a.name, b.name) == .lt;
228 }
229 }.less);
230 return result;
231 }

238 fn prerequisite(self: *Registration, name: []const u8, text: []const u8) Error!Prerequisite {
239 if (equal(text, "-")) return .{};
240 var parser = Parser{ .gpa = self.gpa, .text = text };
241 const node = parser.expression() catch |err| {
242 if (err == error.InvalidPrereq) return self.fail("invalid prereq {s}", .{name});
243 return err;
244 };
245 if (parser.position != text.len) return self.fail("invalid prereq {s}", .{name});
246 if (node.contains(name)) return self.fail("self prereq {s}", .{name});
247 var out: std.Io.Writer.Allocating = .init(self.gpa);
248 try node.write(&out.writer);
249 return .{ .root = node, .text = out.writer.buffered() };
250 }
251 fn slots(self: *Registration, token: []const u8) Error![]const Slot {
252 var result: std.ArrayList(Slot) = .empty;
253 if (equal(token, "-")) return result.items;
254 var it = std.mem.splitScalar(u8, token, ',');
255 while (it.next()) |part| {
256 const day = dayIndex(part[0..2]) orelse return self.fail("invalid slot {s}", .{part});
257 const h1 = integer(part[2..4]);
258 const m1 = integer(part[4..6]);
259 const h2 = integer(part[7..9]);
260 const m2 = integer(part[9..11]);
261 if (h1 > 23 or h2 > 23 or m1 > 59 or m2 > 59 or h1 * 60 + m1 >= h2 * 60 + m2) return self.fail("invalid slot {s}", .{part});
262 try result.append(self.gpa, .{ .text = part, .day = day, .start = @intCast(h1 * 60 + m1), .end = @intCast(h2 * 60 + m2) });
263 }
264 for (result.items, 0..) |left, i| for (result.items[i + 1 ..]) |right| {
265 if (left.overlaps(right)) return self.fail("overlapping slots {s} {s}", .{ left.text, right.text });
266 };
267 return result.items;
268 }
269 fn grades(self: *Registration, name: []const u8, token: []const u8) Error![]const Grade {
270 var result: std.ArrayList(Grade) = .empty;
271 if (equal(token, "-")) return result.items;
272 var it = std.mem.splitScalar(u8, token, ',');
273 while (it.next()) |part| {
274 const colon = std.mem.findScalar(u8, part, ':').?;
275 const code = part[0..colon];
276 const grade = part[colon + 1];
277 if (std.mem.findScalar(u8, "ABCDF", grade) == null) return self.fail("invalid grade {s} {s} {c}", .{ name, code, grade });
278 for (result.items) |r| if (equal(r.code, code)) {
279 return self.fail("duplicate record {s} {s}", .{ name, code });
280 };
281 try result.append(self.gpa, .{ .code = code, .grade = grade, .credits = integer(part[colon + 3 ..]) });
282 }
283 std.mem.sort(Grade, result.items, {}, struct {
284 fn less(_: void, a: Grade, b: Grade) bool {
285 return std.mem.order(u8, a.code, b.code) == .lt;
286 }
287 }.less);
288 return result.items;
289 }

291 var count: usize = 0;

297 fn conflict(s: *Student, c: *Course, excluded: ?*Course) ?*Course {
298 var first: ?*Course = null;
299 for (s.courses.items) |other| {
300 if (other == excluded or !c.conflicts(other)) continue;
301 if (first == null or std.mem.order(u8, other.name, first.?.name) == .lt) first = other;
302 }
303 return first;
304 }
305 fn eligible(self: *Registration, s: *Student, c: *Course, excluded: ?*Course) Allocator.Error!Judgment {
306 if (!c.open) return .{ .word = "course closed" };
307 if (std.mem.findScalar(*Student, c.enrolled.items, s) != null) return .{ .word = "already enrolled" };
308 if (std.mem.findScalar(*Student, c.waiting.items, s) != null) return .{ .word = "already waitlisted" };
309 if (s.passed(c.name)) |grade| return .{ .word = "already completed", .extra = try std.fmt.allocPrint(self.gpa, "{c}", .{grade}) };
310 if (!c.prereq.met(s)) return .{ .word = "prereq not met" };
311 if (conflict(s, c, excluded)) |other| return .{ .word = "time conflict", .extra = other.name };
312 const credits = s.credits(excluded) + c.credits;
313 if (credits > s.limit) return .{ .word = "credit limit", .extra = try std.fmt.allocPrint(self.gpa, "{d} {d}", .{ credits, s.limit }) };
314 return .{ .action = .ok };
315 }
316 fn judge(self: *Registration, s: *Student, c: *Course) Allocator.Error!Judgment {
317 const result = try self.eligible(s, c, null);
318 if (result.action == .reason) return result;
319 if (c.enrolled.items.len < c.capacity) return .{ .action = .enroll };
320 if (self.waitCount(s) >= 3) return .{ .word = "waitlist limit" };
321 return .{ .action = .waitlist, .position = c.position(s) + 1 };
322 }
323 fn judgmentText(self: *Registration, result: Judgment) Allocator.Error![]const u8 {
324 return switch (result.action) {
325 .ok => "OK",
326 .enroll => "enroll",
327 .waitlist => std.fmt.allocPrint(self.gpa, "waitlist {d}", .{result.position}),
328 .reason => if (result.extra.len == 0) result.word else std.fmt.allocPrint(self.gpa, "{s} {s}", .{ result.word, result.extra }),
329 };
330 }

333 }

344 fn promote(self: *Registration) Error!void {
345 for (try self.ordered(Course, self.courses.values())) |c| {
346 if (!c.open) continue;
347 while (c.enrolled.items.len < c.capacity) {
348 var selected: ?usize = null;
349 for (c.waiting.items, 0..) |s, i| {
350 if (conflict(s, c, null) == null and s.credits(null) + c.credits <= s.limit) {
351 selected = i;
352 break;
353 }
354 }
355 const s = c.waiting.orderedRemove(selected orelse break);
356 try self.enroll(s, c);
357 try self.history(s, "promote {s}", .{c.name});
358 try self.enrollmentLine(" PROMOTE", s, c);
359 }
360 }

362 fn schedule(self: *Registration, s: *Student) Error!void {
363 const Meeting = struct { slot: Slot, code: []const u8 };
364 var meetings: std.ArrayList(Meeting) = .empty;
365 var weekly: usize = 0;
366 const courses = try self.ordered(Course, s.courses.items);
367 for (courses) |c| {
368 weekly += c.weekly();
369 for (c.slots) |slot| try meetings.append(self.gpa, .{ .slot = slot, .code = c.name });
370 }
371 std.mem.sort(Meeting, meetings.items, {}, struct {
372 fn less(_: void, a: Meeting, b: Meeting) bool {
373 return Slot.less({}, a.slot, b.slot);
374 }
375 }.less);
376 try self.out.print("SCHEDULE {s} courses {d} credits {d} weekly {s}\n", .{ s.name, s.courses.items.len, s.credits(null), try self.clock(weekly) });
377 for (days, 0..) |day, i| {
378 var found = false;
379 for (meetings.items) |m| {
380 if (m.slot.day != i) continue;
381 if (!found) try self.out.print(" {s}", .{day});
382 found = true;
383 try self.out.print(" {s}-{s} {s}", .{ try self.clock(m.slot.start), try self.clock(m.slot.end), m.code });
384 }
385 if (found) try self.out.writeAll("\n");
386 }
387 var tba = false;
388 for (courses) |c| {
389 if (c.slots.len > 0) continue;
390 if (!tba) try self.out.writeAll(" TBA");
391 tba = true;
392 try self.out.print(" {s}", .{c.name});
393 }
394 if (tba) try self.out.writeAll("\n");
395 }
396 fn freeTime(self: *Registration, s: *Student, token: []const u8) Error!void {
397 const day = dayIndex(token) orelse return self.fail("invalid day {s}", .{token});
398 var free: [840]bool = @splat(true);
399 for (s.courses.items) |c| for (c.slots) |slot| {
400 if (slot.day == day and slot.end > 480 and slot.start < 1320) @memset(free[@max(slot.start, 480) - 480 .. @min(slot.end, 1320) - 480], false);
401 };
402 var total: usize = 0;
403 for (free) |available| if (available) {
404 total += 1;
405 };
406 try self.out.print("FREE {s} {s} total {s}", .{ s.name, token, try self.clock(total) });
407 var i: usize = 0;
408 while (i < free.len) {
409 if (!free[i]) {
410 i += 1;
411 continue;
412 }
413 const start = i;
414 while (i < free.len and free[i]) i += 1;
415 try self.out.print(" {s}-{s}", .{ try self.clock(start + 480), try self.clock(i + 480) });
416 }
417 try self.out.writeAll("\n");
418 }

424 for (courses) |c| {
425 const name = c.name[0 .. c.name.len - 3];
426 const entry = try departments.getOrPut(self.gpa, name);

428 const d = entry.value_ptr;
429 d.courses += 1;
430 d.enrolled += c.enrolled.items.len;
431 d.capacity += c.capacity;
432 d.waiting += c.waiting.items.len;
433 total.enrolled += c.enrolled.items.len;
434 total.capacity += c.capacity;
435 total.waiting += c.waiting.items.len;
436 }
437 try self.out.print("REPORT courses {d} enrolled {d}/{d} rate {s} waitlisted {d}\n", .{ courses.len, total.enrolled, total.capacity, try self.rate(total.enrolled, total.capacity), total.waiting });
438 for (courses) |c| try self.out.print(" {s} {s} {d}/{d} {s} waitlist {d}\n", .{ c.name, if (c.open) "open" else "closed", c.enrolled.items.len, c.capacity, try self.rate(c.enrolled.items.len, c.capacity), c.waiting.items.len });
439 const ordered_departments = try self.gpa.dupe(Department, departments.values());
440 std.mem.sort(Department, ordered_departments, {}, struct {
441 fn less(_: void, a: Department, b: Department) bool {
442 return std.mem.order(u8, a.name, b.name) == .lt;
443 }
444 }.less);
445 for (ordered_departments) |d| try self.out.print(" dept {s} courses {d} enrolled {d}/{d} rate {s} waitlisted {d}\n", .{ d.name, d.courses, d.enrolled, d.capacity, try self.rate(d.enrolled, d.capacity), d.waiting });
446 }
447 fn execute(self: *Registration, a: []const []const u8) Error!void {
448 const op = std.meta.stringToEnum(Command, a[0]).?;
449 const id: []const u8 = if (a.len > 1) a[1] else "";
450 switch (op) {
451 .COURSE => {
452 if (self.courses.contains(id)) return self.fail("duplicate course {s}", .{id});
453 const meetings = try self.slots(a[4]);
454 const prereq = try self.prerequisite(id, a[5]);
455 const c = try self.gpa.create(Course);
456 c.* = .{ .name = id, .credits = integer(a[2]), .capacity = @intCast(integer(a[3])), .slots = meetings, .prereq = prereq };
457 try self.courses.put(self.gpa, id, c);
458 try self.out.print("COURSE {s} credits {d} capacity {d} meetings {d} weekly {s} prereq {s}\n", .{ id, c.credits, c.capacity, meetings.len, try self.clock(c.weekly()), prereq.text });
459 },
460 .STUDENT => {
461 if (self.students.contains(id)) return self.fail("duplicate student {s}", .{id});
462 const records = try self.grades(id, a[4]);
463 const s = try self.gpa.create(Student);
464 s.* = .{ .name = id, .year = integer(a[2]), .limit = integer(a[3]), .grades = records };
465 try self.students.put(self.gpa, id, s);
466 try self.out.print("STUDENT {s} year {d} limit {d} records {d} earned {d} gpa {s}\n", .{ id, s.year, s.limit, records.len, s.earned(), try self.gpaText(s) });
467 },
468 .ENROLL, .CHECK, .DROP, .SWAP, .OPTIONS, .LIMIT, .SCHEDULE, .FREE, .AUDIT, .HISTORY => {
469 const s = try self.student(id);
470 switch (op) {
471 .ENROLL, .CHECK => {
472 const c = try self.course(a[2]);
473 const result = try self.judge(s, c);
474 if (op == .CHECK) {
475 try self.out.print("CHECK {s} {s} {s}\n", .{ id, c.name, try self.judgmentText(result) });
476 } else if (result.action == .reason) {
477 return self.reject(result, s, c);
478 } else if (result.action == .enroll) {
479 try self.enroll(s, c);
480 try self.history(s, "enroll {s}", .{c.name});
481 try self.enrollmentLine("ENROLL", s, c);
482 } else {
483 try c.waiting.insert(self.gpa, result.position - 1, s);
484 try self.history(s, "waitlist {s} {d}", .{ c.name, result.position });
485 try self.out.print("ENROLL {s} {s} waitlist {d}\n", .{ id, c.name, result.position });
486 }
487 },
488 .DROP => {
489 const c = try self.course(a[2]);
490 if (std.mem.findScalar(*Student, c.enrolled.items, s) != null) {
491 remove(Student, &c.enrolled, s);
492 remove(Course, &s.courses, c);
493 try self.history(s, "drop {s}", .{c.name});
494 try self.enrollmentLine("DROP", s, c);
495 } else if (std.mem.findScalar(*Student, c.waiting.items, s) != null) {
496 remove(Student, &c.waiting, s);
497 try self.history(s, "drop {s} waitlist", .{c.name});
498 try self.out.print("DROP {s} {s} waitlist removed\n", .{ id, c.name });
499 } else return self.fail("not registered {s} {s}", .{ id, c.name });
500 try self.promote();
501 },
502 .SWAP => {
503 const from = try self.course(a[2]);
504 const to = try self.course(a[3]);
505 if (std.mem.findScalar(*Student, from.enrolled.items, s) == null) return self.fail("not enrolled {s} {s}", .{ id, from.name });
506 if (from == to) return self.fail("same course {s}", .{from.name});
507 const result = try self.eligible(s, to, from);
508 if (result.action == .reason) return self.reject(result, s, to);
509 if (to.enrolled.items.len == to.capacity) return self.fail("course full {s}", .{to.name});
510 remove(Student, &from.enrolled, s);
511 remove(Course, &s.courses, from);
512 try self.enroll(s, to);
513 try self.history(s, "swap {s} {s}", .{ from.name, to.name });
514 try self.out.print("SWAP {s} {s} -> {s} credits {d}/{d}\n", .{ id, from.name, to.name, s.credits(null), s.limit });
515 try self.promote();
516 },
517 .OPTIONS => {
518 try self.out.print("OPTIONS {s} courses {d}\n", .{ id, self.courses.count() });
519 for (try self.ordered(Course, self.courses.values())) |c| try self.out.print(" {s} {s}\n", .{ c.name, try self.judgmentText(try self.judge(s, c)) });
520 },
521 .LIMIT => {
522 const limit = integer(a[2]);
523 if (limit < s.credits(null)) return self.fail("limit below enrolled {s} {d}", .{ id, s.credits(null) });
524 const old = s.limit;
525 s.limit = limit;
526 try self.out.print("LIMIT {s} {d} -> {d}\n", .{ id, old, limit });
527 try self.promote();
528 },
529 .SCHEDULE => try self.schedule(s),
530 .FREE => try self.freeTime(s, a[2]),
531 .AUDIT => {
532 try self.out.print("AUDIT {s} year {d} limit {d}\n", .{ id, s.year, s.limit });
533 for (s.grades) |r| try self.out.print(" record {s} {c} {d}\n", .{ r.code, r.grade, r.credits });
534 try self.out.print(" earned {d} gpa {s}\n", .{ s.earned(), try self.gpaText(s) });
535 for (try self.ordered(Course, s.courses.items)) |c| try self.out.print(" enrolled {s} {d}\n", .{ c.name, c.credits });
536 try self.out.print(" credits {d} remaining {d}\n", .{ s.credits(null), s.limit - s.credits(null) });
537 for (try self.ordered(Course, self.courses.values())) |c| {
538 if (std.mem.findScalar(*Student, c.waiting.items, s)) |i| try self.out.print(" waitlist {s} {d}\n", .{ c.name, i + 1 });
539 }
540 },
541 .HISTORY => {
542 try self.out.print("HISTORY {s} events {d}\n", .{ id, s.history.items.len });
543 for (s.history.items) |h| try self.out.print(" {d} {s}\n", .{ h.command, h.event });
544 },
545 else => unreachable,
546 }
547 },
548 .CLOSE, .OPEN, .CAPACITY, .PREREQ, .CANCEL, .ROSTER, .INFO => {
549 const c = try self.course(id);
550 switch (op) {
551 .CLOSE, .OPEN => {
552 const open = op == .OPEN;
553 if (c.open == open) return self.fail("already {s} {s}", .{ if (open) "open" else "closed", id });
554 c.open = open;
555 try self.out.print("{s} {s} enrolled {d}/{d} waitlist {d}\n", .{ a[0], id, c.enrolled.items.len, c.capacity, c.waiting.items.len });
556 if (open) try self.promote();
557 },
558 .CAPACITY => {
559 const capacity: usize = @intCast(integer(a[2]));
560 if (capacity < c.enrolled.items.len) return self.fail("capacity below enrollment {s} {d}", .{ id, c.enrolled.items.len });
561 const old = c.capacity;
562 c.capacity = capacity;
563 try self.out.print("CAPACITY {s} {d} -> {d}\n", .{ id, old, capacity });
564 try self.promote();
565 },
566 .PREREQ => {
567 const prereq = try self.prerequisite(id, a[2]);
568 c.prereq = prereq;
569 try self.out.print("PREREQ {s} {s}\n", .{ id, prereq.text });
570 var i: usize = 0;
571 while (i < c.waiting.items.len) {
572 const s = c.waiting.items[i];
573 if (prereq.met(s)) {
574 i += 1;
575 continue;
576 }
577 _ = c.waiting.orderedRemove(i);
578 try self.history(s, "unlist {s}", .{id});
579 try self.out.print(" UNLIST {s} {s}\n", .{ s.name, id });
580 }
581 },
582 .CANCEL => {
583 try self.out.print("CANCEL {s} enrolled {d} waitlist {d}\n", .{ id, c.enrolled.items.len, c.waiting.items.len });
584 for (c.enrolled.items) |s| {
585 remove(Course, &s.courses, c);
586 try self.history(s, "cancel {s}", .{id});
587 }
588 for (c.waiting.items) |s| try self.history(s, "cancel {s} waitlist", .{id});
589 _ = self.courses.swapRemove(id);
590 try self.promote();
591 },
592 .ROSTER => {
593 try self.out.print("ROSTER {s} {s} enrolled {d}/{d} waitlist {d}\n", .{ id, if (c.open) "open" else "closed", c.enrolled.items.len, c.capacity, c.waiting.items.len });
594 for (try self.ordered(Student, c.enrolled.items)) |s| try self.out.print(" enrolled {s} year {d}\n", .{ s.name, s.year });
595 for (c.waiting.items, 0..) |s, i| try self.out.print(" waitlist {d} {s} year {d}\n", .{ i + 1, s.name, s.year });
596 },
597 .INFO => {
598 try self.out.print("INFO {s} {s} credits {d} capacity {d} prereq {s}\n", .{ id, if (c.open) "open" else "closed", c.credits, c.capacity, c.prereq.text });
599 const meetings = try self.gpa.dupe(Slot, c.slots);
600 std.mem.sort(Slot, meetings, {}, Slot.less);
601 for (meetings) |slot| try self.out.print(" meet {s} {s}-{s}\n", .{ days[slot.day], try self.clock(slot.start), try self.clock(slot.end) });
602 if (meetings.len == 0) try self.out.writeAll(" meet TBA\n");
603 try self.out.writeAll(" required by");
604 var found = false;
605 for (try self.ordered(Course, self.courses.values())) |other| {
606 if (!other.prereq.contains(id)) continue;
607 found = true;
608 try self.out.print(" {s}", .{other.name});
609 }
610 if (!found) try self.out.writeAll(" none");
611 try self.out.writeAll("\n");
612 },
613 else => unreachable,
614 }
615 },
616 .STUDENTS => {
617 try self.out.print("STUDENTS {d}\n", .{self.students.count()});
618 for (try self.ordered(Student, self.students.values())) |s| try self.out.print(" {s} year {d} credits {d}/{d} courses {d} waitlists {d}\n", .{ s.name, s.year, s.credits(null), s.limit, s.courses.items.len, self.waitCount(s) });
619 },
620 .REPORT => try self.report(),
621 }
622 }
623};
624pub fn main(init: std.process.Init) !void {
625 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
626 defer arena.deinit();
627 const gpa = arena.allocator();
628 var input_buffer: [4096]u8 = undefined;
629 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
630 var output_buffer: [8192]u8 = undefined;
631 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
632 var registration = Registration{ .gpa = gpa, .out = &output.interface };
633 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
634 for (0..n) |i| {
635 registration.command = i + 1;
636 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
637 var tokens: std.ArrayList([]const u8) = .empty;
638 var it = std.mem.splitScalar(u8, line, ' ');
639 while (it.next()) |token| try tokens.append(gpa, token);
640 registration.execute(tokens.items) catch |err| {
641 if (err != error.BadCommand) return err;
642 try output.interface.print("ERROR {s}\n", .{registration.error_message});
643 };
644 }
645 try output.interface.flush();
646}
