74struct Expression {
75 enum class Kind { Always, Word, And, Or };
76 Kind kind;
77 string word;
78 vector<shared_ptr<Expression>> children;
79 set<string> references;
80
81 explicit Expression(Kind k) : kind(k) {}
82
83 static shared_ptr<Expression> leaf(const string& value) {
84 auto expression = make_shared<Expression>(Kind::Word);
85 expression->word = value;
86 if (!year_word(value)) expression->references.insert(value);
87 return expression;
88 }
89
90 static shared_ptr<Expression> combine(Kind kind,
91 const vector<shared_ptr<Expression>>& items) {
92 vector<shared_ptr<Expression>> flat;
93 for (const auto& item : items) {
94 if (item->kind == kind) {
95 flat.insert(flat.end(), item->children.begin(), item->children.end());
96 } else {
97 flat.push_back(item);
98 }
99 }
100 if (flat.size() == 1) return flat.front();
101 auto result = make_shared<Expression>(kind);
102 result->children = move(flat);
103 for (const auto& child : result->children) {
104 result->references.insert(child->references.begin(), child->references.end());
105 }
106 return result;
107 }
108
109 string canonical() const {
110 if (kind == Kind::Always) return "-";
111 if (kind == Kind::Word) return word;
112 string result;
113 for (const auto& child : children) {
114 if (!result.empty()) result += kind == Kind::And ? "&" : "|";
115 string text = child->canonical();
116 if (kind == Kind::And && child->kind == Kind::Or) text = "(" + text + ")";
117 result += text;
118 }
119 return result;
120 }
121
122 bool evaluate(const Student& student) const;
123};

125class PrerequisiteParser {
126 vector<string> tokens;
127 size_t position = 0;
128
129 string peek() const {
130 return position < tokens.size() ? tokens[position] : "";
131 }
132
133 shared_ptr<Expression> factor() {
134 string token = peek();
135 if (token.empty() || token == "&" || token == "|" || token == ")")
136 throw ParseError();
137 ++position;
138 if (token == "(") {
139 auto inner = expression();
140 if (peek() != ")") throw ParseError();
141 ++position;
142 return inner;
143 }
144 return Expression::leaf(token);
145 }
146
147 shared_ptr<Expression> term() {
148 vector<shared_ptr<Expression>> parts{factor()};
149 while (peek() == "&") {
150 ++position;
151 parts.push_back(factor());
152 }
153 return Expression::combine(Expression::Kind::And, parts);
154 }
155
156 shared_ptr<Expression> expression() {
157 vector<shared_ptr<Expression>> parts{term()};
158 while (peek() == "|") {
159 ++position;
160 parts.push_back(term());
161 }
162 return Expression::combine(Expression::Kind::Or, parts);
163 }
164
165public:
166 explicit PrerequisiteParser(const string& text) {
167 size_t index = 0;
168 while (index < text.size()) {
169 char ch = text[index];
170 if (ch == '&' || ch == '|' || ch == '(' || ch == ')') {
171 tokens.emplace_back(1, ch);
172 ++index;
173 continue;
174 }
175 size_t first = index;
176 while (index < text.size() && isalnum(static_cast<unsigned char>(text[index])))
177 ++index;
178 if (first == index) throw ParseError();
179 string value = text.substr(first, index - first);
180 if (!course_word(value) && !year_word(value)) throw ParseError();
181 tokens.push_back(move(value));
182 }
183 }
184
185 shared_ptr<Expression> parse() {
186 auto result = expression();
187 if (position != tokens.size()) throw ParseError();
188 return result;
189 }
190};
191
192shared_ptr<Expression> prerequisite(const string& code, const string& text) {
193 if (text == "-") return make_shared<Expression>(Expression::Kind::Always);
194 shared_ptr<Expression> expression;
195 try {
196 expression = PrerequisiteParser(text).parse();
197 } catch (const ParseError&) {
198 throw CommandError("invalid prereq " + code);
199 }
200 if (expression->references.count(code))
201 throw CommandError("self prereq " + code);
202 return expression;
203}

205struct Slot {
206 int day;
207 int start;
208 int end;
209 string original;
210
211 explicit Slot(const string& text) : day(-1), start(0), end(0), original(text) {
212 string prefix = text.substr(0, 2);
213 for (int i = 0; i < static_cast<int>(DAYS.size()); ++i) {
214 if (DAYS[i] == prefix) day = i;
215 }
216 int start_hour = stoi(text.substr(2, 2));
217 int start_minute = stoi(text.substr(4, 2));
218 int end_hour = stoi(text.substr(7, 2));
219 int end_minute = stoi(text.substr(9, 2));
220 start = 60 * start_hour + start_minute;
221 end = 60 * end_hour + end_minute;
222 if (day < 0 || start_hour > 23 || end_hour > 23 ||
223 start_minute > 59 || end_minute > 59 || start >= end)
224 throw CommandError("invalid slot " + text);
225 }
226
227 bool overlaps(const Slot& other) const {
228 return day == other.day && start < other.end && other.start < end;
229 }
230
231 string text() const { return time_text(start) + "-" + time_text(end); }
232};

261struct Student {
262 string id;
263 int year;
264 int limit;
265 map<string, Record> records;
266 map<string, char> passed;
267 set<string> courses;
268 set<string> waiting;

271
272 Student(string id, int year, int limit, map<string, Record> records)
273 : id(move(id)), year(year), limit(limit), records(move(records)) {
274 for (const auto& [code, record] : this->records) {
275 if (record.grade == 'A' || record.grade == 'B' || record.grade == 'C')
276 passed[code] = record.grade;
277 }
278 }
279
280 int earned() const {
281 int result = 0;
282 for (const auto& [code, record] : records) {
283 if (record.grade != 'F') result += record.credits;
284 }
285 return result;
286 }
287
288 string gpa() const {
289 if (records.empty()) return "n/a";
290 int points = 0, credits_sum = 0;
291 for (const auto& [code, record] : records) {
292 points += GRADE_POINTS.at(record.grade) * record.credits;
293 credits_sum += record.credits;
294 }
295 int rounded = (200 * points + credits_sum) / (2 * credits_sum);
296 ostringstream out;
297 out << rounded / 100 << '.' << setfill('0') << setw(2) << rounded % 100;
298 return out.str();
299 }

302bool Expression::evaluate(const Student& student) const {
303 if (kind == Kind::Always) return true;
304 if (kind == Kind::Word) {
305 return year_word(word) ? student.year >= word[1] - '0' :
306 student.passed.count(word) != 0;
307 }
308 if (kind == Kind::And) {
309 for (const auto& child : children) {
310 if (!child->evaluate(student)) return false;
311 }
312 return true;
313 }
314 for (const auto& child : children) {
315 if (child->evaluate(student)) return true;
316 }
317 return false;
318}

351 string conflict(const Student& s, const Course& c,
352 const string& excluded = "") const {
353 for (const string& code : s.courses) {
354 if (code != excluded && c.conflicts.count(code)) return code;
355 }

358
359 Decision eligibility(const Student& s, const Course& c,
360 const string& excluded = "") const {
361 if (!c.open) return {"course closed", {}};
362 if (s.courses.count(c.code)) return {"already enrolled", {}};
363 if (s.waiting.count(c.code)) return {"already waitlisted", {}};
364 auto past = s.passed.find(c.code);
365 if (past != s.passed.end())
366 return {"already completed", {string(1, past->second)}};
367 if (!c.required->evaluate(s)) return {"prereq not met", {}};
368 string conflicted = conflict(s, c, excluded);
369 if (!conflicted.empty()) return {"time conflict", {conflicted}};
370 int credits = s.credits + c.credits;
371 if (!excluded.empty()) credits -= courses.at(excluded).credits;
372 if (credits > s.limit)
373 return {"credit limit", {to_string(credits), to_string(s.limit)}};
374 return {"OK", {}};
375 }

377 int wait_position(const Student& s, const Course& c) const {
378 int position = 1;
379 for (const string& id : c.waitlist) {
380 if (students.at(id).year >= s.year) ++position;
381 }
382 return position;
383 }

389 return {"enroll", {}};
390 if (s.waiting.size() >= 3) return {"waitlist limit", {}};
391 return {"waitlist", {to_string(wait_position(s, c))}};

425 void promotion() {
426 for (auto& [code, c] : courses) {
427 if (!c.open) continue;
428 vector<string> waiting = c.waitlist;
429 for (const string& id : waiting) {
430 if (c.enrolled.size() >= static_cast<size_t>(c.capacity)) break;
431 Student& s = students.at(id);
432 if (!conflict(s, c).empty() || s.credits + c.credits > s.limit) continue;
433 unlist(s, c);
434 enroll(s, c);
435 record(s, "promote " + code);
436 emit(words("PROMOTE", s.id, code, "enrolled",
437 to_string(c.enrolled.size()) + "/" + to_string(c.capacity),
438 "credits", to_string(s.credits) + "/" + to_string(s.limit)), true);
439 }

443 void cmd_course(const vector<string>& a) {
444 const string& code = a[0];
445 if (courses.count(code)) throw CommandError("duplicate course " + code);
446 vector<Slot> slots;
447 if (a[3] != "-") {
448 for (const string& token : split(a[3], ',')) slots.emplace_back(token);
449 }
450 for (size_t i = 0; i < slots.size(); ++i) {
451 for (size_t j = i + 1; j < slots.size(); ++j) {
452 if (slots[i].overlaps(slots[j]))
453 throw CommandError(words("overlapping slots",
454 slots[i].original, slots[j].original));
455 }
456 }
457 auto required = prerequisite(code, a[4]);
458 Course new_course(code, stoi(a[1]), stoi(a[2]), move(slots), required);
459 for (auto& [other_code, other] : courses) {
460 bool overlaps = false;
461 for (const Slot& first : new_course.slots) {
462 for (const Slot& second : other.slots) {
463 if (first.overlaps(second)) overlaps = true;
464 }
465 }
466 if (overlaps) {
467 new_course.conflicts.insert(other_code);
468 other.conflicts.insert(code);
469 }
470 }
471 int weekly = new_course.weekly();
472 int meetings = static_cast<int>(new_course.slots.size());
473 courses.emplace(code, move(new_course));
474 emit(words("COURSE", code, "credits", a[1], "capacity", a[2],
475 "meetings", meetings, "weekly", time_text(weekly),
476 "prereq", required->canonical()));
477 }

479 void cmd_student(const vector<string>& a) {
480 const string& id = a[0];
481 if (students.count(id)) throw CommandError("duplicate student " + id);
482 map<string, Record> records;
483 if (a[3] != "-") {
484 for (const string& text : split(a[3], ',')) {
485 vector<string> fields = split(text, ':');
486 const string& code = fields[0];
487 char grade = fields[1][0];
488 if (!GRADE_POINTS.count(grade))
489 throw CommandError(words("invalid grade", id, code, grade));
490 if (records.count(code))
491 throw CommandError(words("duplicate record", id, code));
492 records[code] = {grade, stoi(fields[2])};
493 }
494 }
495 int count = static_cast<int>(records.size());
496 Student s(id, stoi(a[1]), stoi(a[2]), move(records));
497 emit(words("STUDENT", id, "year", a[1], "limit", a[2],
498 "records", count, "earned", s.earned(), "gpa", s.gpa()));
499 students.emplace(id, move(s));
500 }

555 void cmd_swap(const vector<string>& a) {
556 Student& s = student(a[0]);
557 Course& from = course(a[1]);
558 Course& to = course(a[2]);
559 if (!s.courses.count(from.code))
560 throw CommandError(words("not enrolled", s.id, from.code));
561 if (from.code == to.code) throw CommandError("same course " + from.code);
562 Decision eligibility_result = eligibility(s, to, from.code);
563 if (eligibility_result.kind != "OK")
564 fail_request(eligibility_result, s, to);
565 if (to.enrolled.size() >= static_cast<size_t>(to.capacity))
566 throw CommandError("course full " + to.code);
567 withdraw(s, from);
568 enroll(s, to);
569 record(s, words("swap", from.code, to.code));
570 emit(words("SWAP", s.id, from.code, "->", to.code, "credits",
571 to_string(s.credits) + "/" + to_string(s.limit)));
572 promotion();
573 }

680 void cmd_schedule(const vector<string>& a) {
681 Student& s = student(a[0]);
682 int weekly = 0;
683 for (const string& code : s.courses) weekly += courses.at(code).weekly();
684 emit(words("SCHEDULE", s.id, "courses", s.courses.size(), "credits",
685 s.credits, "weekly", time_text(weekly)));
686 for (int day = 0; day < static_cast<int>(DAYS.size()); ++day) {
687 vector<tuple<int, int, string>> slots = student_slots(s, day);
688 if (slots.empty()) continue;
689 string text = DAYS[day];
690 for (const auto& [start, end, code] : slots)
691 text += " " + time_text(start) + "-" + time_text(end) + " " + code;
692 emit(text, true);
693 }
694 string pending = "TBA";
695 bool found = false;
696 for (const string& code : s.courses) {
697 if (courses.at(code).slots.empty()) {
698 pending += " " + code;
699 found = true;
700 }
701 }
702 if (found) emit(pending, true);
703 }
704
705 void cmd_free(const vector<string>& a) {
706 Student& s = student(a[0]);
707 int day = -1;
708 for (int i = 0; i < static_cast<int>(DAYS.size()); ++i) {
709 if (DAYS[i] == a[1]) day = i;
710 }
711 if (day < 0) throw CommandError("invalid day " + a[1]);
712 vector<pair<int, int>> occupied;
713 for (const auto& [start, end, code] : student_slots(s, day)) {
714 int first = max(480, start), last = min(1320, end);
715 if (first < last) occupied.emplace_back(first, last);
716 }
717 vector<pair<int, int>> free_ranges;
718 int cursor = 480;
719 for (const auto& [start, end] : occupied) {
720 if (cursor < start) free_ranges.emplace_back(cursor, start);
721 cursor = max(cursor, end);
722 }
723 if (cursor < 1320) free_ranges.emplace_back(cursor, 1320);
724 int minutes = 0;
725 for (const auto& [start, end] : free_ranges) minutes += end - start;
726 string result = words("FREE", s.id, a[1], "total", time_text(minutes));
727 for (const auto& [start, end] : free_ranges)
728 result += " " + time_text(start) + "-" + time_text(end);
729 emit(result);
730 }
731
732 void cmd_info(const vector<string>& a) {
733 Course& c = course(a[0]);
734 emit(words("INFO", c.code, c.status(), "credits", c.credits,
735 "capacity", c.capacity, "prereq", c.required->canonical()));
736 vector<Slot> slots = c.slots;
737 sort(slots.begin(), slots.end(), [](const Slot& a, const Slot& b) {
738 return tie(a.day, a.start) < tie(b.day, b.start);
739 });
740 if (slots.empty()) {
741 emit("meet TBA", true);
742 } else {
743 for (const Slot& slot : slots)
744 emit(words("meet", DAYS[slot.day], slot.text()), true);
745 }
746 string result = "required by";
747 bool found = false;
748 for (const auto& [code, other] : courses) {
749 if (other.required->references.count(c.code)) {
750 result += " " + code;
751 found = true;
752 }
753 }
754 if (!found) result += " none";
755 emit(result, true);
756 }

758 void cmd_audit(const vector<string>& a) {
759 Student& s = student(a[0]);
760 emit(words("AUDIT", s.id, "year", s.year, "limit", s.limit));
761 for (const auto& [code, record] : s.records)
762 emit(words("record", code, record.grade, record.credits), true);
763 emit(words("earned", s.earned(), "gpa", s.gpa()), true);
764 for (const string& code : s.courses)
765 emit(words("enrolled", code, courses.at(code).credits), true);
766 emit(words("credits", s.credits, "remaining", s.limit - s.credits), true);
767 for (const string& code : s.waiting) {
768 const auto& waiting = courses.at(code).waitlist;
769 auto it = find(waiting.begin(), waiting.end(), s.id);
770 int position = static_cast<int>(distance(waiting.begin(), it)) + 1;
771 emit(words("waitlist", code, position), true);
772 }
773 }

791 string group_counts(const vector<const Course*>& subset) const {
792 int enrolled = 0, capacity = 0, waitlisted = 0;
793 for (const Course* c : subset) {
794 enrolled += static_cast<int>(c->enrolled.size());
795 capacity += c->capacity;
796 waitlisted += static_cast<int>(c->waitlist.size());
797 }
798 return words("enrolled", to_string(enrolled) + "/" + to_string(capacity),
799 "rate", rate_text(enrolled, capacity), "waitlisted", waitlisted);
800 }

802 void cmd_report(const vector<string>&) {
803 vector<const Course*> all_courses;
804 map<string, vector<const Course*>> departments;
805 for (const auto& [code, c] : courses) {
806 all_courses.push_back(&c);
807 departments[c.department()].push_back(&c);
808 }
809 emit(words("REPORT", "courses", courses.size(), group_counts(all_courses)));
810 for (const auto& [code, c] : courses) {
811 emit(words(code, c.status(),
812 to_string(c.enrolled.size()) + "/" + to_string(c.capacity),
813 rate_text(static_cast<int>(c.enrolled.size()), c.capacity),
814 "waitlist", c.waitlist.size()), true);
815 }
816 for (const auto& [department, subset] : departments)
817 emit(words("dept", department, "courses", subset.size(),
818 group_counts(subset)), true);
819 }

821public:
822 void execute(const string& text) {
823 ++command_number;
824 istringstream input(text);
825 string command, argument;
826 input >> command;
827 vector<string> args;
828 while (input >> argument) args.push_back(argument);
829 try {
830 if (command == "COURSE") cmd_course(args);
831 else if (command == "STUDENT") cmd_student(args);
832 else if (command == "ENROLL") cmd_enroll(args);
833 else if (command == "CHECK") cmd_check(args);
834 else if (command == "OPTIONS") cmd_options(args);
835 else if (command == "DROP") cmd_drop(args);
836 else if (command == "SWAP") cmd_swap(args);
837 else if (command == "CLOSE") cmd_close(args);
838 else if (command == "OPEN") cmd_open(args);
839 else if (command == "CAPACITY") cmd_capacity(args);
840 else if (command == "LIMIT") cmd_limit(args);
841 else if (command == "PREREQ") cmd_prereq(args);
842 else if (command == "CANCEL") cmd_cancel(args);
843 else if (command == "ROSTER") cmd_roster(args);
844 else if (command == "SCHEDULE") cmd_schedule(args);
845 else if (command == "FREE") cmd_free(args);
846 else if (command == "INFO") cmd_info(args);
847 else if (command == "AUDIT") cmd_audit(args);
848 else if (command == "HISTORY") cmd_history(args);
849 else if (command == "STUDENTS") cmd_students(args);
850 else if (command == "REPORT") cmd_report(args);
851 } catch (const CommandError& error) {
852 emit("ERROR " + string(error.what()));
853 }
854 }

857 for (const string& line : output) cout << line << '\n';
858 }
