41local function separated(text, separator)
42 local result = {}
43 local start = 1
44 while true do
45 local first, last = text:find(separator, start, true)
46 if not first then
47 result[#result + 1] = text:sub(start)
48 break
49 end
50 result[#result + 1] = text:sub(start, first - 1)
51 start = last + 1
52 end
53 return result
54end

66 return department ~= nil and #department >= 2 and #department <= 4 and #digits == 3

73local function expression(kind, children, word)
74 local references = {}
75 if kind == "WORD" and not year_word(word) then
76 references[word] = true
77 end
78 for _, child in ipairs(children or {}) do
79 for code in pairs(child.references) do
80 references[code] = true
81 end
82 end
83 return {kind = kind, children = children or {}, word = word, references = references}
84end

86local function combine(kind, children)
87 local flattened = {}
88 for _, child in ipairs(children) do
89 if child.kind == kind then
90 for _, nested in ipairs(child.children) do
91 flattened[#flattened + 1] = nested
92 end
93 else
94 flattened[#flattened + 1] = child
95 end
96 end
97 if #flattened == 1 then return flattened[1] end
98 return expression(kind, flattened)
99end

101local function prerequisite(code, text)
102 if text == "-" then return expression("TRUE") end
103 local tokens = {}
104 local position = 1
105 while position <= #text do
106 local symbol = text:sub(position, position)
107 if symbol == "&" or symbol == "|" or symbol == "(" or symbol == ")" then
108 tokens[#tokens + 1] = symbol
109 position = position + 1
110 else
111 local word = text:sub(position):match("^([A-Z0-9]+)")
112 if not word or not (course_word(word) or year_word(word)) then
113 fail("invalid prereq " .. code)
114 end
115 tokens[#tokens + 1] = word
116 position = position + #word
117 end
118 end
119
120 local next_token = 1
121 local function peek() return tokens[next_token] end
122 local parse_expression
123
124 local function factor()
125 local token = peek()
126 if not token or token == "&" or token == "|" or token == ")" then
127 fail("invalid prereq " .. code)
128 end
129 next_token = next_token + 1
130 if token == "(" then
131 local child = parse_expression()
132 if peek() ~= ")" then fail("invalid prereq " .. code) end
133 next_token = next_token + 1
134 return child
135 end
136 return expression("WORD", nil, token)
137 end
138
139 local function term()
140 local children = {factor()}
141 while peek() == "&" do
142 next_token = next_token + 1
143 children[#children + 1] = factor()
144 end
145 return combine("AND", children)
146 end
147
148 parse_expression = function()
149 local children = {term()}
150 while peek() == "|" do
151 next_token = next_token + 1
152 children[#children + 1] = term()
153 end
154 return combine("OR", children)
155 end
156
157 local result = parse_expression()
158 if peek() then fail("invalid prereq " .. code) end
159 if result.references[code] then fail("self prereq " .. code) end
160 return result
161end

163local function canonical(node)
164 if node.kind == "TRUE" then return "-" end
165 if node.kind == "WORD" then return node.word end
166 local parts = {}
167 for _, child in ipairs(node.children) do
168 local text = canonical(child)
169 if node.kind == "AND" and child.kind == "OR" then
170 text = "(" .. text .. ")"
171 end
172 parts[#parts + 1] = text
173 end
174 return table.concat(parts, node.kind == "AND" and "&" or "|")
175end

177local function satisfied(node, student)
178 if node.kind == "TRUE" then return true end
179 if node.kind == "WORD" then
180 if year_word(node.word) then
181 return student.year >= tonumber(node.word:sub(2))
182 end
183 return student.passed[node.word] ~= nil
184 end
185 if node.kind == "AND" then
186 for _, child in ipairs(node.children) do
187 if not satisfied(child, student) then return false end
188 end
189 return true
190 end
191 for _, child in ipairs(node.children) do
192 if satisfied(child, student) then return true end
193 end
194 return false
195end

197local function parse_slot(text)
198 local day = text:sub(1, 2)
199 local sh = tonumber(text:sub(3, 4))
200 local sm = tonumber(text:sub(5, 6))
201 local eh = tonumber(text:sub(8, 9))
202 local em = tonumber(text:sub(10, 11))
203 local first = sh * 60 + sm
204 local last = eh * 60 + em
205 if not DAY_INDEX[day] or sh > 23 or sm > 59 or eh > 23 or em > 59 or
206 first >= last then
207 fail("invalid slot " .. text)
208 end
209 return {day = DAY_INDEX[day], start = first, finish = last, original = text}
210end

244local function gpa(student)
245 if count_keys(student.records) == 0 then return "n/a" end
246 local points, credits = 0, 0
247 for _, record in pairs(student.records) do
248 points = points + GRADE_POINTS[record.grade] * record.credits
249 credits = credits + record.credits
250 end
251 local value = math.floor((200 * points + credits) / (2 * credits))
252 return string.format("%d.%02d", math.floor(value / 100), value % 100)
253end

292local function eligibility(student, course, excluded)
293 if not course.open then return decision("course closed") end
294 if student.courses[course.code] then return decision("already enrolled") end
295 if student.waiting[course.code] then return decision("already waitlisted") end
296 if student.passed[course.code] then
297 return decision("already completed", student.passed[course.code])
298 end
299 if not satisfied(course.prerequisite, student) then
300 return decision("prereq not met")
301 end
302 local occupied = conflict(student, course, excluded)
303 if occupied then return decision("time conflict", occupied) end
304 local credits = student.credits + course.credits
305 if excluded then credits = credits - courses[excluded].credits end
306 if credits > student.limit then
307 return decision("credit limit", credits, student.limit)
308 end
309 return decision("OK")
310end

359local function promotion()
360 for _, code in ipairs(sorted_keys(courses)) do
361 local course = courses[code]
362 if course.open then
363 local waiting = {table.unpack(course.waitlist)}
364 for _, id in ipairs(waiting) do
365 if count_keys(course.enrolled) >= course.capacity then break end
366 local student = students[id]
367 if not conflict(student, course) and
368 student.credits + course.credits <= student.limit then
369 unlist_student(student, course)
370 enroll(student, course)
371 record(student, "promote " .. code)
372 emit_indented("PROMOTE", id, code, "enrolled",
373 count_keys(course.enrolled) .. "/" .. course.capacity,
374 "credits", student.credits .. "/" .. student.limit)
375 end
376 end

381local function cmd_course(args)
382 local code, credits, capacity, timetable, text = table.unpack(args)
383 if courses[code] then fail("duplicate course " .. code) end
384 local slots = {}
385 if timetable ~= "-" then
386 for _, value in ipairs(separated(timetable, ",")) do
387 slots[#slots + 1] = parse_slot(value)
388 end
389 end
390 for i = 1, #slots do
391 for j = i + 1, #slots do
392 if overlap(slots[i], slots[j]) then
393 fail("overlapping slots " .. slots[i].original .. " " .. slots[j].original)
394 end
395 end
396 end
397 local required = prerequisite(code, text)
398 local course = {
399 code = code, credits = tonumber(credits), capacity = tonumber(capacity),
400 slots = slots, prerequisite = required, open = true,
401 enrolled = {}, waitlist = {}, conflicts = {}
402 }
403 for other_code, other in pairs(courses) do
404 local overlapping = false
405 for _, first in ipairs(slots) do
406 for _, second in ipairs(other.slots) do
407 if overlap(first, second) then overlapping = true end
408 end
409 end
410 if overlapping then
411 course.conflicts[other_code] = true
412 other.conflicts[code] = true
413 end
414 end
415 courses[code] = course
416 emit("COURSE", code, "credits", credits, "capacity", capacity,
417 "meetings", #slots, "weekly", time_text(weekly(course)),
418 "prereq", canonical(required))
419end

421local function cmd_student(args)
422 local id, year, limit, text = table.unpack(args)
423 if students[id] then fail("duplicate student " .. id) end
424 local records = {}
425 if text ~= "-" then
426 for _, entry in ipairs(separated(text, ",")) do
427 local parts = separated(entry, ":")
428 local code, grade, credits = table.unpack(parts)
429 if not GRADE_POINTS[grade] then
430 fail("invalid grade " .. id .. " " .. code .. " " .. grade)
431 end
432 if records[code] then fail("duplicate record " .. id .. " " .. code) end
433 records[code] = {grade = grade, credits = tonumber(credits)}
434 end
435 end
436 local student = {
437 id = id, year = tonumber(year), limit = tonumber(limit),
438 records = records, passed = {}, courses = {}, waiting = {},
439 credits = 0, history = {}
440 }
441 for code, entry in pairs(records) do
442 if entry.grade == "A" or entry.grade == "B" or entry.grade == "C" then
443 student.passed[code] = entry.grade
444 end
445 end
446 students[id] = student
447 emit("STUDENT", id, "year", year, "limit", limit,
448 "records", count_keys(records), "earned", earned(student), "gpa", gpa(student))
449end

623local function student_slots(student, day)
624 local slots = {}
625 for code in pairs(student.courses) do
626 for _, slot in ipairs(courses[code].slots) do
627 if slot.day == day then
628 slots[#slots + 1] = {
629 start = slot.start, finish = slot.finish, code = code
630 }
631 end
632 end
633 end
634 table.sort(slots, function(a, b)
635 if a.start ~= b.start then return a.start < b.start end
636 if a.finish ~= b.finish then return a.finish < b.finish end
637 return a.code < b.code
638 end)
639 return slots
640end

669local function cmd_free(args)
670 local student = require_student(args[1])
671 local day = DAY_INDEX[args[2]]
672 if not day then fail("invalid day " .. args[2]) end
673 local occupied = {}
674 for _, slot in ipairs(student_slots(student, day)) do
675 local first, last = math.max(480, slot.start), math.min(1320, slot.finish)
676 if first < last then occupied[#occupied + 1] = {first, last} end
677 end
678 local free, cursor = {}, 480
679 for _, interval in ipairs(occupied) do
680 if cursor < interval[1] then
681 free[#free + 1] = {cursor, interval[1]}
682 end
683 cursor = math.max(cursor, interval[2])
684 end
685 if cursor < 1320 then free[#free + 1] = {cursor, 1320} end
686 local minutes = 0
687 for _, interval in ipairs(free) do
688 minutes = minutes + interval[2] - interval[1]
689 end
690 local parts = {"FREE", student.id, args[2], "total", time_text(minutes)}
691 for _, interval in ipairs(free) do
692 parts[#parts + 1] = time_text(interval[1]) .. "-" .. time_text(interval[2])
693 end
694 emit(table.unpack(parts))
695end

730 end
731 emit_indented("earned", earned(student), "gpa", gpa(student))
732 for _, code in ipairs(sorted_keys(student.courses)) do
733 emit_indented("enrolled", code, courses[code].credits)
734 end
735 emit_indented("credits", student.credits, "remaining", student.limit - student.credits)
736 for _, code in ipairs(sorted_keys(student.waiting)) do
737 for position, id in ipairs(courses[code].waitlist) do
738 if id == student.id then
739 emit_indented("waitlist", code, position)
740 break
741 end
742 end
743 end
744end

765local function group_counts(group)
766 local enrolled, capacity, waitlisted = 0, 0, 0
767 for _, course in ipairs(group) do
768 enrolled = enrolled + count_keys(course.enrolled)
769 capacity = capacity + course.capacity
770 waitlisted = waitlisted + #course.waitlist
771 end
772 return words("enrolled", enrolled .. "/" .. capacity,
773 "rate", rate_text(enrolled, capacity), "waitlisted", waitlisted)
774end

776local function cmd_report()
777 local all, departments = {}, {}
778 for _, code in ipairs(sorted_keys(courses)) do
779 local course = courses[code]
780 all[#all + 1] = course
781 local dept = department(code)

783 departments[dept][#departments[dept] + 1] = course
784 end

786 for _, code in ipairs(sorted_keys(courses)) do
787 local course = courses[code]
788 emit_indented(code, status(course),
789 count_keys(course.enrolled) .. "/" .. course.capacity,
790 rate_text(count_keys(course.enrolled), course.capacity),
791 "waitlist", #course.waitlist)
792 end
793 for _, dept in ipairs(sorted_keys(departments)) do
794 emit_indented("dept", dept, "courses", #departments[dept],
795 group_counts(departments[dept]))
