134static Text gpa(const Student *s) {
135 Text result;
136 if (!s->grade_count) { strcpy(result.text, "n/a"); return result; }
137 int points = 0, total = 0;
138 for (int i = 0; i < s->grade_count; ++i) {
139 Grade r = s->grades[i];
140 points += (r.grade == 'F' ? 0 : 'E' - r.grade) * r.credits;
141 total += r.credits;
142 }
143 int value = (200 * points + total) / (2 * total);
144 snprintf(result.text, sizeof result.text, "%d.%02d", value / 100, value % 100);
145 return result;
146}

157static Node *parse_factor(Parser *p) {
158 if (p->text[p->position] == '(') {
159 ++p->position;
160 Node *node = parse_expression(p);
161 if (!node || p->text[p->position] != ')') { free_node(node); return NULL; }
162 ++p->position;
163 return node;
164 }
165 int start = p->position;
166 while (upper(p->text[p->position]) || digit(p->text[p->position])) ++p->position;
167 int length = p->position - start, letters = 0;
168 const char *word = p->text + start;
169 while (letters < length && upper(word[letters])) ++letters;
170 bool code = letters >= 2 && letters <= 4 && length == letters + 3;
171 for (int i = letters; i < length; ++i) if (!digit(word[i])) code = false;
172 bool year = length == 2 && word[0] == 'Y' && word[1] >= '1' && word[1] <= '9';
173 if (!code && !year) return NULL;
174 Node *node = allocate(sizeof *node);
175 node->kind = WORD;
176 memcpy(node->word, word, (size_t)length);
177 return node;
178}
179
180static Node *parse_term(Parser *p) {
181 Node *left = parse_factor(p);
182 if (!left) return NULL;
183 while (p->text[p->position] == '&') {
184 ++p->position;
185 Node *right = parse_factor(p);
186 if (!right) { free_node(left); return NULL; }
187 Node *parent = allocate(sizeof *parent);
188 parent->kind = AND;
189 parent->left = left;
190 parent->right = right;
191 left = parent;
192 }
193 return left;
194}
195
196static Node *parse_expression(Parser *p) {
197 Node *left = parse_term(p);
198 if (!left) return NULL;
199 while (p->text[p->position] == '|') {
200 ++p->position;
201 Node *right = parse_term(p);
202 if (!right) { free_node(left); return NULL; }
203 Node *parent = allocate(sizeof *parent);
204 parent->kind = OR;
205 parent->left = left;
206 parent->right = right;
207 left = parent;
208 }
209 return left;
210}

213 if (!node) return false;

215 return contains(node->left, code) || contains(node->right, code);

218static bool satisfied(const Node *node, const Student *s) {
219 if (!node) return true;
220 if (node->kind == AND) return satisfied(node->left, s) && satisfied(node->right, s);
221 if (node->kind == OR) return satisfied(node->left, s) || satisfied(node->right, s);
222 return strlen(node->word) == 2 ? s->year >= node->word[1] - '0' : passed(s, node->word) != '\0';
223}

225static void write_node(const Node *node, bool within_and, char *text, int *length) {
226 bool parens = within_and && node->kind == OR;
227 if (parens) text[(*length)++] = '(';
228 if (node->kind == WORD) {
229 int size = (int)strlen(node->word);
230 memcpy(text + *length, node->word, (size_t)size);
231 *length += size;
232 } else {
233 write_node(node->left, node->kind == AND, text, length);
234 text[(*length)++] = node->kind == AND ? '&' : '|';
235 write_node(node->right, node->kind == AND, text, length);
236 }
237 if (parens) text[(*length)++] = ')';
238}

240static bool prerequisite(const char *code, const char *text, Node **root, char *canonical) {
241 if (equal(text, "-")) { *root = NULL; strcpy(canonical, "-"); return true; }
242 Parser parser = { .text = text };
243 Node *node = parse_expression(&parser);
244 if (!node || text[parser.position]) { free_node(node); return fail("invalid prereq %s", code); }
245 if (contains(node, code)) { free_node(node); return fail("self prereq %s", code); }
246 int length = 0;
247 write_node(node, false, canonical, &length);
248 canonical[length] = '\0';
249 *root = node;
250 return true;
251}

258static bool overlaps(Slot a, Slot b) { return a.day == b.day && a.start < b.end && b.start < a.end; }

260static bool parse_slots(const char *text, Slot *slots, int *count) {
261 *count = 0;
262 if (equal(text, "-")) return true;
263 char copy[72];
264 strcpy(copy, text);
265 for (char *part = strtok(copy, ","); part; part = strtok(NULL, ",")) {
266 Slot slot = { 0 };
267 strcpy(slot.text, part);
268 char day[3] = { part[0], part[1], '\0' };
269 slot.day = day_index(day);
270 int h1 = (part[2] - '0') * 10 + part[3] - '0', m1 = (part[4] - '0') * 10 + part[5] - '0';
271 int h2 = (part[7] - '0') * 10 + part[8] - '0', m2 = (part[9] - '0') * 10 + part[10] - '0';
272 slot.start = h1 * 60 + m1;
273 slot.end = h2 * 60 + m2;
274 if (slot.day < 0 || h1 > 23 || h2 > 23 || m1 > 59 || m2 > 59 || slot.start >= slot.end)
275 return fail("invalid slot %s", part);
276 slots[(*count)++] = slot;
277 }
278 for (int i = 0; i < *count; ++i)
279 for (int j = i + 1; j < *count; ++j)
280 if (overlaps(slots[i], slots[j])) return fail("overlapping slots %s %s", slots[i].text, slots[j].text);
281 return true;
282}

284static bool parse_grades(const char *id, const char *text, Grade *grades, int *count) {
285 *count = 0;
286 if (equal(text, "-")) return true;
287 char copy[240];
288 strcpy(copy, text);
289 for (char *part = strtok(copy, ","); part; part = strtok(NULL, ",")) {
290 char *colon = strchr(part, ':');
291 Grade grade = { .grade = colon[1], .credits = colon[3] - '0' };
292 *colon = '\0';
293 strcpy(grade.code, part);
294 if (!strchr("ABCDF", grade.grade)) return fail("invalid grade %s %s %c", id, grade.code, grade.grade);
295 for (int i = 0; i < *count; ++i)
296 if (equal(grades[i].code, grade.code)) return fail("duplicate record %s %s", id, grade.code);
297 grades[(*count)++] = grade;
298 }
299 qsort(grades, (size_t)*count, sizeof *grades, grade_order);
300 return true;
301}

337static Judgment eligible(Student *s, Course *c, Course *excluded) {
338 Judgment j = { .decision = REASON };
339 if (!c->open) j.reason = "course closed";
340 else if (enrolled_index(s, c) >= 0) j.reason = "already enrolled";
341 else if (waiting_index(c, s) >= 0) j.reason = "already waitlisted";
342 else if (passed(s, c->code)) {
343 j.reason = "already completed";
344 snprintf(j.extra, sizeof j.extra, "%c", passed(s, c->code));
345 } else if (!satisfied(c->prereq, s)) j.reason = "prereq not met";
346 else {
347 Course *other = conflict(s, c, excluded);
348 int attempted = credits(s, excluded) + c->credits;
349 if (other) { j.reason = "time conflict"; strcpy(j.extra, other->code); }
350 else if (attempted > s->limit) {
351 j.reason = "credit limit";
352 snprintf(j.extra, sizeof j.extra, "%d %d", attempted, s->limit);
353 } else j.decision = ELIGIBLE;
354 }
355 return j;
356}

358static Judgment judge(Student *s, Course *c) {
359 Judgment j = eligible(s, c, NULL);
360 if (j.decision == REASON) return j;
361 if (c->enrolled_count < c->capacity) j.decision = ENROLL;
362 else if (wait_count(s) >= 3) { j.decision = REASON; j.reason = "waitlist limit"; }
363 else {
364 j.decision = WAITLIST;
365 j.position = 1;
366 for (int i = 0; i < c->waiting_count; ++i) j.position += c->waiting[i]->year >= s->year;
367 }
368 return j;
369}

371static Text judgment_text(Judgment j) {
372 Text text;
373 if (j.decision == ENROLL) strcpy(text.text, "enroll");
374 else if (j.decision == WAITLIST) snprintf(text.text, sizeof text.text, "waitlist %d", j.position);
375 else if (j.decision == ELIGIBLE) strcpy(text.text, "OK");
376 else snprintf(text.text, sizeof text.text, "%s%s%s", j.reason, j.extra[0] ? " " : "", j.extra);
377 return text;
378}

428static void promote(void) {
429 for (int i = 0; i < course_count; ++i) {
430 Course *c = courses[i];
431 if (!c->open) continue;
432 while (c->enrolled_count < c->capacity) {
433 int j = 0;
434 while (j < c->waiting_count) {
435 Student *s = c->waiting[j];
436 if (!conflict(s, c, NULL) && credits(s, NULL) + c->credits <= s->limit) break;
437 ++j;
438 }
439 if (j == c->waiting_count) break;
440 Student *s = remove_waiting(c, j);
441 enroll(s, c);
442 history(s, "promote %s", c->code);
443 enrollment_line(" PROMOTE", s, c);
444 }
445 }
446}

458static void schedule(Student *s) {
459 Meeting meetings[300];
460 int count = 0, minutes = 0;
461 for (int i = 0; i < s->course_count; ++i) {
462 Course *c = s->courses[i];
463 minutes += weekly(c);
464 for (int j = 0; j < c->slot_count; ++j) meetings[count++] = (Meeting){ c->slots[j], c->code };
465 }
466 qsort(meetings, (size_t)count, sizeof *meetings, meeting_order);
467 printf("SCHEDULE %s courses %d credits %d weekly %s\n", s->id, s->course_count, credits(s, NULL), time_text(minutes).text);
468 for (int day = 0; day < 7; ++day) {
469 bool found = false;
470 for (int i = 0; i < count; ++i) {
471 Slot slot = meetings[i].slot;
472 if (slot.day != day) continue;
473 if (!found) printf(" %s", day_names[day]);
474 found = true;
475 printf(" %s-%s %s", time_text(slot.start).text, time_text(slot.end).text, meetings[i].code);
476 }
477 if (found) putchar('\n');
478 }
479 bool tba = false;
480 for (int i = 0; i < s->course_count; ++i) {
481 if (s->courses[i]->slot_count) continue;
482 if (!tba) fputs(" TBA", stdout);
483 tba = true;
484 printf(" %s", s->courses[i]->code);
485 }
486 if (tba) putchar('\n');
487}

489static void free_time(Student *s, const char *name) {
490 int day = day_index(name);
491 if (day < 0) { fail("invalid day %s", name); return; }
492 bool available[840];
493 for (int i = 0; i < 840; ++i) available[i] = true;
494 for (int i = 0; i < s->course_count; ++i) {
495 Course *c = s->courses[i];
496 for (int j = 0; j < c->slot_count; ++j) {
497 Slot slot = c->slots[j];
498 if (slot.day != day) continue;
499 for (int minute = maximum(480, slot.start); minute < minimum(1320, slot.end); ++minute)
500 available[minute - 480] = false;
501 }
502 }
503 int total = 0;
504 for (int i = 0; i < 840; ++i) total += available[i];
505 printf("FREE %s %s total %s", s->id, name, time_text(total).text);
506 int i = 0;
507 while (i < 840) {
508 if (!available[i]) { ++i; continue; }
509 int first = i;
510 while (i < 840 && available[i]) ++i;
511 printf(" %s-%s", time_text(first + 480).text, time_text(i + 480).text);
512 }
513 putchar('\n');
514}

550static void execute(char **a) {
551 const char *op = a[0];
552 if (equal(op, "COURSE")) {
553 if (find_course(a[1])) { fail("duplicate course %s", a[1]); return; }
554 Course data = { .credits = atoi(a[2]), .capacity = atoi(a[3]), .open = true };
555 strcpy(data.code, a[1]);
556 if (!parse_slots(a[4], data.slots, &data.slot_count)) return;
557 if (!prerequisite(a[1], a[5], &data.prereq, data.prereq_text)) return;
558 Course *c = allocate(sizeof *c);
559 *c = data;
560 courses[course_count++] = c;
561 qsort(courses, (size_t)course_count, sizeof *courses, course_order);
562 printf("COURSE %s credits %d capacity %d meetings %d weekly %s prereq %s\n", c->code, c->credits, c->capacity, c->slot_count, time_text(weekly(c)).text, c->prereq_text);
563 } else if (equal(op, "STUDENT")) {
564 if (find_student(a[1])) { fail("duplicate student %s", a[1]); return; }
565 Student data = { .year = atoi(a[2]), .limit = atoi(a[3]) };
566 strcpy(data.id, a[1]);
567 if (!parse_grades(a[1], a[4], data.grades, &data.grade_count)) return;
568 Student *s = allocate(sizeof *s);
569 *s = data;
570 students[student_count++] = s;
571 qsort(students, (size_t)student_count, sizeof *students, student_order);
572 printf("STUDENT %s year %d limit %d records %d earned %d gpa %s\n", s->id, s->year, s->limit, s->grade_count, earned(s), gpa(s).text);
573 } else if (equal(op, "STUDENTS")) {
574 printf("STUDENTS %d\n", student_count);
575 for (int i = 0; i < student_count; ++i) {
576 Student *s = students[i];
577 printf(" %s year %d credits %d/%d courses %d waitlists %d\n", s->id, s->year, credits(s, NULL), s->limit, s->course_count, wait_count(s));
578 }
579 } else if (equal(op, "REPORT")) {
580 report();
581 } else if (equal(op, "ENROLL") || equal(op, "CHECK") || equal(op, "OPTIONS") || equal(op, "DROP") || equal(op, "SWAP") ||
582 equal(op, "LIMIT") || equal(op, "SCHEDULE") || equal(op, "FREE") || equal(op, "AUDIT") || equal(op, "HISTORY")) {
583 Student *s = student(a[1]);
584 if (!s) return;
585 if (equal(op, "ENROLL") || equal(op, "CHECK")) {
586 Course *c = course(a[2]);
587 if (!c) return;
588 Judgment j = judge(s, c);
589 if (equal(op, "CHECK")) printf("CHECK %s %s %s\n", s->id, c->code, judgment_text(j).text);
590 else if (j.decision == REASON) reject(j, s, c);
591 else if (j.decision == ENROLL) {
592 enroll(s, c);
593 history(s, "enroll %s", c->code);
594 enrollment_line("ENROLL", s, c);
595 } else {
596 int i = j.position - 1;
597 memmove(c->waiting + i + 1, c->waiting + i, (size_t)(c->waiting_count - i) * sizeof *c->waiting);
598 c->waiting[i] = s;
599 ++c->waiting_count;
600 history(s, "waitlist %s %d", c->code, j.position);
601 printf("ENROLL %s %s waitlist %d\n", s->id, c->code, j.position);
602 }
603 } else if (equal(op, "OPTIONS")) {
604 printf("OPTIONS %s courses %d\n", s->id, course_count);
605 for (int i = 0; i < course_count; ++i) printf(" %s %s\n", courses[i]->code, judgment_text(judge(s, courses[i])).text);
606 } else if (equal(op, "DROP")) {
607 Course *c = course(a[2]);
608 if (!c) return;
609 int waiting = waiting_index(c, s);
610 if (enrolled_index(s, c) >= 0) {
611 remove_enrollment(s, c);
612 history(s, "drop %s", c->code);
613 enrollment_line("DROP", s, c);
614 } else if (waiting >= 0) {
615 remove_waiting(c, waiting);
616 history(s, "drop %s waitlist", c->code);
617 printf("DROP %s %s waitlist removed\n", s->id, c->code);
618 } else { fail("not registered %s %s", s->id, c->code); return; }
619 promote();
620 } else if (equal(op, "SWAP")) {
621 Course *from = course(a[2]);
622 if (!from) return;
623 Course *to = course(a[3]);
624 if (!to) return;
625 if (enrolled_index(s, from) < 0) { fail("not enrolled %s %s", s->id, from->code); return; }
626 if (from == to) { fail("same course %s", from->code); return; }
627 Judgment j = eligible(s, to, from);
628 if (j.decision == REASON) { reject(j, s, to); return; }
629 if (to->enrolled_count == to->capacity) { fail("course full %s", to->code); return; }
630 remove_enrollment(s, from);
631 enroll(s, to);
632 history(s, "swap %s %s", from->code, to->code);
633 printf("SWAP %s %s -> %s credits %d/%d\n", s->id, from->code, to->code, credits(s, NULL), s->limit);
634 promote();
635 } else if (equal(op, "LIMIT")) {
636 int limit = atoi(a[2]);
637 if (limit < credits(s, NULL)) { fail("limit below enrolled %s %d", s->id, credits(s, NULL)); return; }
638 int old = s->limit;
639 s->limit = limit;
640 printf("LIMIT %s %d -> %d\n", s->id, old, limit);
641 promote();
642 } else if (equal(op, "SCHEDULE")) schedule(s);
643 else if (equal(op, "FREE")) free_time(s, a[2]);
644 else if (equal(op, "AUDIT")) {
645 printf("AUDIT %s year %d limit %d\n", s->id, s->year, s->limit);
646 for (int i = 0; i < s->grade_count; ++i) printf(" record %s %c %d\n", s->grades[i].code, s->grades[i].grade, s->grades[i].credits);
647 printf(" earned %d gpa %s\n", earned(s), gpa(s).text);
648 for (int i = 0; i < s->course_count; ++i) printf(" enrolled %s %d\n", s->courses[i]->code, s->courses[i]->credits);
649 printf(" credits %d remaining %d\n", credits(s, NULL), s->limit - credits(s, NULL));
650 for (int i = 0; i < course_count; ++i) {
651 int position = waiting_index(courses[i], s);
652 if (position >= 0) printf(" waitlist %s %d\n", courses[i]->code, position + 1);
653 }
654 } else if (equal(op, "HISTORY")) {
655 printf("HISTORY %s events %d\n", s->id, s->history_count);
656 for (int i = 0; i < s->history_count; ++i) printf(" %d %s\n", s->history[i].command, s->history[i].event);
657 }
658 } else {
659 Course *c = course(a[1]);
660 if (!c) return;
661 if (equal(op, "CLOSE") || equal(op, "OPEN")) {
662 bool open = equal(op, "OPEN");
663 if (c->open == open) { fail("already %s %s", open ? "open" : "closed", c->code); return; }
664 c->open = open;
665 printf("%s %s enrolled %d/%d waitlist %d\n", op, c->code, c->enrolled_count, c->capacity, c->waiting_count);
666 if (open) promote();
667 } else if (equal(op, "CAPACITY")) {
668 int capacity = atoi(a[2]);
669 if (capacity < c->enrolled_count) { fail("capacity below enrollment %s %d", c->code, c->enrolled_count); return; }
670 int old = c->capacity;
671 c->capacity = capacity;
672 printf("CAPACITY %s %d -> %d\n", c->code, old, capacity);
673 promote();
674 } else if (equal(op, "PREREQ")) {
675 Node *root;
676 char canonical[61];
677 if (!prerequisite(c->code, a[2], &root, canonical)) return;
678 free_node(c->prereq);
679 c->prereq = root;
680 strcpy(c->prereq_text, canonical);
681 printf("PREREQ %s %s\n", c->code, canonical);
682 int i = 0;
683 while (i < c->waiting_count) {
684 Student *s = c->waiting[i];
685 if (satisfied(root, s)) { ++i; continue; }
686 remove_waiting(c, i);
687 history(s, "unlist %s", c->code);
688 printf(" UNLIST %s %s\n", s->id, c->code);
689 }
690 } else if (equal(op, "CANCEL")) {
691 printf("CANCEL %s enrolled %d waitlist %d\n", c->code, c->enrolled_count, c->waiting_count);
692 while (c->enrolled_count) {
693 Student *s = c->enrolled[0];
694 history(s, "cancel %s", c->code);
695 remove_enrollment(s, c);
696 }
697 for (int i = 0; i < c->waiting_count; ++i) history(c->waiting[i], "cancel %s waitlist", c->code);
698 int i = 0;
699 while (courses[i] != c) ++i;
700 --course_count;
701 memmove(courses + i, courses + i + 1, (size_t)(course_count - i) * sizeof *courses);
702 free_node(c->prereq);
703 free(c);
704 promote();
705 } else if (equal(op, "ROSTER")) {
706 printf("ROSTER %s %s enrolled %d/%d waitlist %d\n", c->code, c->open ? "open" : "closed", c->enrolled_count, c->capacity, c->waiting_count);
707 for (int i = 0; i < c->enrolled_count; ++i) printf(" enrolled %s year %d\n", c->enrolled[i]->id, c->enrolled[i]->year);
708 for (int i = 0; i < c->waiting_count; ++i) printf(" waitlist %d %s year %d\n", i + 1, c->waiting[i]->id, c->waiting[i]->year);
709 } else if (equal(op, "INFO")) {
710 printf("INFO %s %s credits %d capacity %d prereq %s\n", c->code, c->open ? "open" : "closed", c->credits, c->capacity, c->prereq_text);
711 Slot slots[6];
712 memcpy(slots, c->slots, (size_t)c->slot_count * sizeof *slots);
713 qsort(slots, (size_t)c->slot_count, sizeof *slots, slot_order);
714 for (int i = 0; i < c->slot_count; ++i) printf(" meet %s %s-%s\n", day_names[slots[i].day], time_text(slots[i].start).text, time_text(slots[i].end).text);
715 if (!c->slot_count) puts(" meet TBA");
716 fputs(" required by", stdout);
717 bool found = false;
718 for (int i = 0; i < course_count; ++i) {
719 if (!contains(courses[i]->prereq, c->code)) continue;
720 found = true;
721 printf(" %s", courses[i]->code);
722 }
723 if (!found) fputs(" none", stdout);
724 putchar('\n');
725 }
726 }
727}
