28Slot[] | error parse_slots(string text)
29 Slot[] result := []
30 if text == "-"
31 return result
32 for token := text.split(",")[]
33 day := day_index(token[0:<2])
34 start_hour := parse_integer(token[2:<4])
35 start_minute := parse_integer(token[4:<6])
36 end_hour := parse_integer(token[7:<9])
37 end_minute := parse_integer(token[9:<11])
38 start := 60 * start_hour + start_minute
39 ending := 60 * end_hour + end_minute
40 if day < 0 or start_hour > 23 or end_hour > 23 or start_minute > 59 or end_minute > 59 or start >= ending
41 return error("invalid slot " + token)
42 result = result.append(Slot(day, start, ending, token))
43 for: 0 <= left < int(#result)
44 for: left + 1 <= right < int(#result)
45 if overlaps(result[left], result[right])
46 return error("overlapping slots {result[left].text} {result[right].text}")
47 return result

60bool course_word(string word)
61 letters := 0
62 while letters < int(#word) and "ABCDEFGHIJKLMNOPQRSTUVWXYZ".contains(word[letters])
63 letters += 1
64 if letters < 2 or letters > 4 or int(#word) != letters + 3
65 return false
66 for: letters <= index < int(#word)

71int | error parse_expression(Parser &parser, int minimum)
72 if parser.position >= int(#parser.source)
73 return error("invalid prereq " + parser.code)
74 int left
75 if parser.source[parser.position] == "("
76 parser.position += 1
77 left = try parse_expression(&parser, 1)
78 if parser.position == int(#parser.source) or parser.source[parser.position] != ")"
79 return error("invalid prereq " + parser.code)
80 parser.position += 1
81 else
82 first := parser.position
83 while parser.position < int(#parser.source) and not "&|()".contains(parser.source[parser.position])
84 parser.position += 1
85 word := parser.source[first:<parser.position]
86 if #word == 2 and word[0] == "Y" and "123456789".contains(word[1])
87 left = append_node(&parser, Node("YEAR", word, parse_integer(word[1]), -1, -1, word))
88 elif course_word(word)
89 left = append_node(&parser, Node("COURSE", word, 0, -1, -1, word))
90 else
91 return error("invalid prereq " + parser.code)
92 while parser.position < int(#parser.source)
93 operation := parser.source[parser.position]
94 precedence := if operation == "|" then 1 elif operation == "&" then 2 else 0
95 if precedence < minimum
96 break
97 parser.position += 1
98 right := try parse_expression(&parser, precedence + 1)
99 first := parser.nodes[left]
100 second := parser.nodes[right]
101 first_text := if operation == "&" and first.kind == "|" then "(" + first.canonical + ")" else first.canonical
102 second_text := if operation == "&" and second.kind == "|" then "(" + second.canonical + ")" else second.canonical
103 left = append_node(&parser, Node(operation, "", 0, left, right, first_text + operation + second_text))
104 return left

112Prerequisite | error parse_prerequisite(string code, string text)
113 if text == "-"
114 return Prerequisite([Node("NONE", "", 0, -1, -1, "-")], 0)
115 parser := Parser(code, text)
116 root := try parse_expression(&parser, 1)
117 if parser.position != int(#text)
118 return error("invalid prereq " + code)
119 result := Prerequisite(parser.nodes, root)
120 if mentions(result, code)
121 return error("self prereq " + code)
122 return result

190bool satisfies(const Student student, const Prerequisite prerequisite)
191 values := array(#prerequisite.nodes, fill = false)
192 for: 0 <= index < int(#prerequisite.nodes)
193 node := prerequisite.nodes[index]
194 if node.kind == "NONE"
195 values[index] = true
196 elif node.kind == "YEAR"
197 values[index] = student.year >= node.year
198 elif node.kind == "COURSE"
199 values[index] = passed_grade(student, node.word) != ""
200 elif node.kind == "&"
201 values[index] = values[node.left] and values[node.right]
202 else
203 values[index] = values[node.left] or values[node.right]
204 return values[prerequisite.root]

267Judgment eligibility(const State state, int student_index, int course_index, excluded := -1)
268 student := state.students[student_index]
269 course := state.courses[course_index]
270 if not course.open
271 return Judgment("course closed")
272 if course.enrolled.has(student_index)
273 return Judgment("already enrolled")
274 if wait_position(course, student_index) > 0
275 return Judgment("already waitlisted")
276 grade := passed_grade(student, course.code)
277 if grade != ""
278 return Judgment("already completed", grade)
279 if not satisfies(student, course.prerequisite)
280 return Judgment("prereq not met")
281 conflict := first_conflict(state, student_index, course_index, excluded)
282 if conflict != ""
283 return Judgment("time conflict", conflict)
284 credits := enrolled_credits(state, student_index, excluded) + course.credits
285 if credits > student.limit
286 return Judgment("credit limit", "{credits} {student.limit}")
287 return Judgment("OK")

289Judgment request(const State state, int student_index, int course_index)

294 if int(course.enrolled.size()) < course.capacity
295 return Judgment("enroll")
296 if wait_count(state, student_index) >= 3
297 return Judgment("waitlist limit")
298 return Judgment("waitlist", string(joining_position(state, student_index, course_index)))

322void promote(State &state)
323 for course_index := course_order(state)[]
324 if not state.courses[course_index].open
325 continue
326 while int(state.courses[course_index].enrolled.size()) < state.courses[course_index].capacity
327 chosen := -1
328 for student_index := state.courses[course_index].waitlist[]
329 if first_conflict(state, student_index, course_index) == "" and enrolled_credits(state, student_index) + state.courses[course_index].credits <= state.students[student_index].limit
330 chosen = student_index
331 break
332 if chosen < 0
333 break
334 state.courses[course_index].waitlist = [for index := state.courses[course_index].waitlist[] if index != chosen then index]
335 add_enrollment(&state, chosen, course_index)
336 history_event(&state, chosen, "promote " + state.courses[course_index].code)
337 enrollment_notice(state, " PROMOTE", chosen, course_index)

433void | error command(State &state, const string[] words)
434 verb := words[0]
435 if verb == "COURSE"
436 code := words[1]
437 if course_id(state, code) >= 0
438 return error("duplicate course " + code)
439 slots := try parse_slots(words[4])
440 prerequisite := try parse_prerequisite(code, words[5])
441 course := Course(code, parse_integer(words[2]), parse_integer(words[3]), slots, prerequisite)
442 state.courses = state.courses.append(course)
443 emit("COURSE {code} credits {course.credits} capacity {course.capacity} meetings {#slots} weekly {clock_text(weekly_minutes(course))} prereq {canonical(prerequisite)}")
444 elif verb == "STUDENT"
445 id := words[1]
446 if student_id(state, id) >= 0
447 return error("duplicate student " + id)
448 grades := try parse_grades(id, words[4])
449 student := Student(id, parse_integer(words[2]), parse_integer(words[3]), grades)
450 state.students = state.students.append(student)
451 emit("STUDENT {id} year {student.year} limit {student.limit} records {#grades} earned {earned_credits(student)} gpa {gpa(student)}")
452 elif verb == "ENROLL" or verb == "CHECK" or verb == "DROP" or verb == "SWAP"
453 student_index := try require_student(state, words[1])
454 course_index := try require_course(state, words[2])
455 course := state.courses[course_index]
456 if verb == "ENROLL" or verb == "CHECK"
457 judgment := request(state, student_index, course_index)
458 if verb == "CHECK"
459 emit("CHECK {words[1]} {words[2]} {judgment_text(judgment)}")
460 elif judgment.word == "enroll"
461 add_enrollment(&state, student_index, course_index)
462 history_event(&state, student_index, "enroll " + course.code)
463 enrollment_notice(state, "ENROLL", student_index, course_index)
464 elif judgment.word == "waitlist"
465 position := parse_integer(judgment.extra) - 1
466 state.courses[course_index].waitlist = course.waitlist[0:<position].append(student_index).concat(course.waitlist[position:])
467 history_event(&state, student_index, "waitlist {course.code} {position + 1}")
468 emit("ENROLL {words[1]} {course.code} waitlist {position + 1}")
469 else
470 return request_error(judgment, words[1], course.code)
471 elif verb == "DROP"
472 if course.enrolled.has(student_index)
473 remove_enrollment(&state, student_index, course_index)
474 history_event(&state, student_index, "drop " + course.code)
475 enrollment_notice(state, "DROP", student_index, course_index)
476 elif wait_position(course, student_index) > 0
477 state.courses[course_index].waitlist = [for index := course.waitlist[] if index != student_index then index]
478 history_event(&state, student_index, "drop " + course.code + " waitlist")
479 emit("DROP {words[1]} {course.code} waitlist removed")
480 else
481 return error("not registered {words[1]} {course.code}")
482 promote(&state)
483 else
484 destination := try require_course(state, words[3])
485 if not course.enrolled.has(student_index)
486 return error("not enrolled {words[1]} {course.code}")
487 if course_index == destination
488 return error("same course " + course.code)
489 judgment := eligibility(state, student_index, destination, course_index)
490 if judgment.word != "OK"
491 return request_error(judgment, words[1], words[3])
492 if int(state.courses[destination].enrolled.size()) >= state.courses[destination].capacity
493 return error("course full " + words[3])
494 remove_enrollment(&state, student_index, course_index)
495 add_enrollment(&state, student_index, destination)
496 history_event(&state, student_index, "swap {words[2]} {words[3]}")
497 emit("SWAP {words[1]} {words[2]} -> {words[3]} credits {enrolled_credits(state, student_index)}/{state.students[student_index].limit}")
498 promote(&state)
499 elif verb == "CLOSE" or verb == "OPEN" or verb == "CAPACITY" or verb == "PREREQ" or verb == "CANCEL" or verb == "ROSTER" or verb == "INFO"
500 course_index := try require_course(state, words[1])
501 course := state.courses[course_index]
502 status := if course.open then "open" else "closed"
503 if verb == "CLOSE" or verb == "OPEN"
504 opening := verb == "OPEN"
505 if opening == course.open
506 return error("already {status} {course.code}")
507 state.courses[course_index].open = opening
508 emit("{verb} {course.code} enrolled {course.enrolled.size()}/{course.capacity} waitlist {#course.waitlist}")
509 if opening
510 promote(&state)
511 elif verb == "CAPACITY"
512 capacity := parse_integer(words[2])
513 if capacity < int(course.enrolled.size())
514 return error("capacity below enrollment {course.code} {course.enrolled.size()}")
515 state.courses[course_index].capacity = capacity
516 emit("CAPACITY {course.code} {course.capacity} -> {capacity}")
517 promote(&state)
518 elif verb == "PREREQ"
519 prerequisite := try parse_prerequisite(course.code, words[2])
520 state.courses[course_index].prerequisite = prerequisite
521 emit("PREREQ {course.code} {canonical(prerequisite)}")
522 int[] retained := []
523 for student_index := course.waitlist[]
524 if satisfies(state.students[student_index], prerequisite)
525 retained = retained.append(student_index)
526 else
527 history_event(&state, student_index, "unlist " + course.code)
528 emit(" UNLIST {state.students[student_index].id} {course.code}")
529 state.courses[course_index].waitlist = retained
530 elif verb == "CANCEL"
531 emit("CANCEL {course.code} enrolled {course.enrolled.size()} waitlist {#course.waitlist}")
532 for student_index := course.enrolled.values()[]
533 .state.students[student_index].enrolled.remove(course_index)
534 history_event(&state, student_index, "cancel " + course.code)
535 for student_index := course.waitlist[]
536 history_event(&state, student_index, "cancel " + course.code + " waitlist")
537 state.courses[course_index].registered = false
538 state.courses[course_index].enrolled = distinct<int>()
539 state.courses[course_index].waitlist = []
540 promote(&state)
541 elif verb == "ROSTER"
542 emit("ROSTER {course.code} {status} enrolled {course.enrolled.size()}/{course.capacity} waitlist {#course.waitlist}")
543 ids := [for index := course.enrolled.values()[] then state.students[index].id]
544 for id := ids.sorted()[]
545 student := state.students[student_id(state, id)]
546 emit(" enrolled {id} year {student.year}")
547 for: 0 <= position < int(#course.waitlist)
548 student := state.students[course.waitlist[position]]
549 emit(" waitlist {position + 1} {student.id} year {student.year}")
550 else
551 emit("INFO {course.code} {status} credits {course.credits} capacity {course.capacity} prereq {canonical(course.prerequisite)}")
552 meetings := [for slot := course.slots[] then Meeting(slot.day, slot.start, slot.end, course.code)]
