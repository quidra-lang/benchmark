3private val DAYS = listOf("MO", "TU", "WE", "TH", "FR", "SA", "SU")
4private val POINTS = mapOf('A' to 4, 'B' to 3, 'C' to 2, 'D' to 1, 'F' to 0)
5private val COURSE_WORD = Regex("^[A-Z]{2,4}[0-9]{3}$")
6private val YEAR_WORD = Regex("^Y[1-9]$")
7private val TOKENS = Regex("[A-Z0-9]+|[&|()]")

9private class CommandError(message: String) : RuntimeException(message)
10private class ParseError : RuntimeException()

12private fun fail(reason: String): Nothing = throw CommandError(reason)
13private fun timeText(minutes: Int) =
14 String.format(Locale.ROOT, "%02d:%02d", minutes / 60, minutes % 60)
15private fun rateText(count: Int, capacity: Int) =
16 if (capacity == 0) "n/a" else (count * 100 / capacity).toString() + "%"
17private fun isYear(word: String) = YEAR_WORD.matches(word)

19private enum class Kind { TRUE, WORD, AND, OR }
20
21private class Expr(
22 val kind: Kind,
23 val word: String = "",
24 val children: List<Expr> = emptyList()
25) {
26 val references: Set<String> =
27 if (kind == Kind.WORD && !isYear(word)) setOf(word)
28 else children.flatMap { it.references }.toSet()
29
30 fun canonical(): String = when (kind) {
31 Kind.TRUE -> "-"
32 Kind.WORD -> word
33 Kind.AND, Kind.OR -> children.joinToString(if (kind == Kind.AND) "&" else "|") { child ->
34 val text = child.canonical()
35 if (kind == Kind.AND && child.kind == Kind.OR) "(" + text + ")" else text
36 }
37 }
38
39 fun evaluate(student: Student): Boolean = when (kind) {
40 Kind.TRUE -> true
41 Kind.WORD -> if (isYear(word)) student.year >= word[1].digitToInt()
42 else student.passedGrade(word) != null
43 Kind.AND -> children.all { it.evaluate(student) }
44 Kind.OR -> children.any { it.evaluate(student) }
45 }
46}
47
48private fun combine(kind: Kind, children: List<Expr>): Expr {
49 val flat = children.flatMap { if (it.kind == kind) it.children else listOf(it) }
50 return if (flat.size == 1) flat[0] else Expr(kind, children = flat)
51}

53private class Parser(text: String) {
54 private val tokens = TOKENS.findAll(text).map { it.value }.toList()
55 private var position = 0
56
57 init {
58 if (tokens.isEmpty() ||
59 tokens.any { it !in listOf("&", "|", "(", ")") &&
60 !COURSE_WORD.matches(it) && !YEAR_WORD.matches(it) }) {
61 throw ParseError()
62 }
63 }
64
65 private fun peek() = tokens.getOrNull(position)
66
67 private fun factor(): Expr {
68 val token = peek() ?: throw ParseError()
69 if (token == "&" || token == "|" || token == ")") throw ParseError()
70 position++
71 if (token == "(") {
72 val node = expression()
73 if (peek() != ")") throw ParseError()
74 position++
75 return node
76 }
77 return Expr(Kind.WORD, token)
78 }
79
80 private fun term(): Expr {
81 val children = mutableListOf(factor())
82 while (peek() == "&") {
83 position++
84 children.add(factor())
85 }
86 return combine(Kind.AND, children)
87 }
88
89 private fun expression(): Expr {
90 val children = mutableListOf(term())
91 while (peek() == "|") {
92 position++
93 children.add(term())
94 }
95 return combine(Kind.OR, children)
96 }
97
98 fun parse(): Expr {
99 val node = expression()
100 if (position != tokens.size) throw ParseError()
101 return node
102 }
103}
104
105private fun prerequisite(code: String, text: String): Expr {
106 if (text == "-") return Expr(Kind.TRUE)
107 val expr = try {
108 Parser(text).parse()
109 } catch (_: ParseError) {
110 fail("invalid prereq " + code)
111 }
112 if (code in expr.references) fail("self prereq " + code)
113 return expr
114}

116private data class Slot(val day: Int, val start: Int, val end: Int, val original: String) {
117 fun overlaps(other: Slot) =
118 day == other.day && start < other.end && other.start < end
119 fun display() = timeText(start) + "-" + timeText(end)
120}
121
122private fun slot(text: String): Slot {
123 val day = DAYS.indexOf(text.substring(0, 2))
124 val sh = text.substring(2, 4).toInt()
125 val sm = text.substring(4, 6).toInt()
126 val eh = text.substring(7, 9).toInt()
127 val em = text.substring(9, 11).toInt()
128 val start = sh * 60 + sm
129 val end = eh * 60 + em
130 if (day == -1 || sh > 23 || sm > 59 || eh > 23 || em > 59 || start >= end) {
131 fail("invalid slot " + text)
132 }
133 return Slot(day, start, end, text)
134}

155private class Student(
156 val id: String,
157 val year: Int,
158 var limit: Int,
159 val records: Map<String, GradeRecord>
160) {
161 val courses = mutableSetOf<String>()
162 val waiting = mutableSetOf<String>()
163 var credits = 0
164 val history = mutableListOf<Pair<Int, String>>()
165
166 fun passedGrade(code: String): Char? =
167 records[code]?.grade?.takeIf { it == 'A' || it == 'B' || it == 'C' }
168
169 fun earned() = records.values.sumOf { if (it.grade == 'F') 0 else it.credits }
170
171 fun gpa(): String {
172 if (records.isEmpty()) return "n/a"
173 val points = records.values.sumOf { POINTS.getValue(it.grade) * it.credits }
174 val total = records.values.sumOf { it.credits }
175 val rounded = (200 * points + total) / (2 * total)
176 return String.format(Locale.ROOT, "%d.%02d", rounded / 100, rounded % 100)
177 }
178}

204 private fun conflict(student: Student, course: Course, excluded: String? = null) =
205 student.courses.filter { it != excluded && it in course.conflicts }.minOrNull()
206
207 private fun eligibility(student: Student, course: Course, excluded: String? = null): Decision {
208 if (!course.open) return Decision("course closed")
209 if (course.code in student.courses) return Decision("already enrolled")
210 if (course.code in student.waiting) return Decision("already waitlisted")
211 val grade = student.passedGrade(course.code)
212 if (grade != null) return Decision("already completed", listOf(grade.toString()))
213 if (!course.prerequisite.evaluate(student)) return Decision("prereq not met")
214 val occupied = conflict(student, course, excluded)
215 if (occupied != null) return Decision("time conflict", listOf(occupied))
216 val credits = student.credits + course.credits -
217 (if (excluded == null) 0 else courses.getValue(excluded).credits)
218 if (credits > student.limit) {
219 return Decision("credit limit", listOf(credits.toString(), student.limit.toString()))
220 }
221 return Decision("OK")
222 }
223
224 private fun waitPosition(student: Student, course: Course) =
225 1 + course.waitlist.count { students.getValue(it).year >= student.year }
226
227 private fun request(student: Student, course: Course): Decision {
228 val result = eligibility(student, course)
229 if (result.kind != "OK") return result
230 if (course.enrolled.size < course.capacity) return Decision("enroll")
231 if (student.waiting.size >= 3) return Decision("waitlist limit")
232 return Decision("waitlist", listOf(waitPosition(student, course).toString()))
233 }

235 private fun failRequest(result: Decision, student: Student, course: Course): Nothing =
236 fail((listOf(result.kind, student.id, course.code) + result.extra).joinToString(" "))

255 private fun promotion() {
256 for (code in courses.keys.sorted()) {
257 val course = courses.getValue(code)
258 if (!course.open) continue
259 for (id in course.waitlist.toList()) {
260 if (course.enrolled.size >= course.capacity) break
261 val student = students.getValue(id)
262 if (conflict(student, course) != null ||
263 student.credits + course.credits > student.limit) continue
264 unlist(student, course)
265 enroll(student, course)
266 record(student, "promote " + code)
267 emit("PROMOTE", id, code, "enrolled",
268 course.enrolled.size.toString() + "/" + course.capacity,
269 "credits", student.credits.toString() + "/" + student.limit,
270 indented = true)
271 }

275 private fun cmdCourse(a: List<String>) {
276 val code = a[0]
277 if (courses.containsKey(code)) fail("duplicate course " + code)
278 val slots = if (a[3] == "-") emptyList() else a[3].split(',').map(::slot)
279 for (i in slots.indices) {
280 for (j in i + 1 until slots.size) {
281 if (slots[i].overlaps(slots[j])) {
282 fail("overlapping slots " + slots[i].original + " " + slots[j].original)
283 }
284 }
285 }
286 val required = prerequisite(code, a[4])
287 val course = Course(code, a[1].toInt(), a[2].toInt(), slots, required)
288 for ((otherCode, other) in courses) {
289 if (slots.any { first -> other.slots.any { second -> first.overlaps(second) } }) {
290 course.conflicts.add(otherCode)
291 other.conflicts.add(code)
292 }
293 }
294 courses[code] = course
295 emit("COURSE", code, "credits", a[1], "capacity", a[2],
296 "meetings", slots.size, "weekly", timeText(course.weekly()),
297 "prereq", required.canonical())
298 }

300 private fun cmdStudent(a: List<String>) {
301 val id = a[0]
302 if (students.containsKey(id)) fail("duplicate student " + id)
303 val records = linkedMapOf<String, GradeRecord>()
304 if (a[3] != "-") {
305 for (entry in a[3].split(',')) {
306 val (code, gradeText, creditsText) = entry.split(':')
307 val grade = gradeText.single()
308 if (grade !in POINTS) fail("invalid grade " + id + " " + code + " " + grade)
309 if (code in records) fail("duplicate record " + id + " " + code)
310 records[code] = GradeRecord(grade, creditsText.toInt())
311 }
312 }
313 val student = Student(id, a[1].toInt(), a[2].toInt(), records)
314 students[id] = student
315 emit("STUDENT", id, "year", a[1], "limit", a[2],
316 "records", records.size, "earned", student.earned(), "gpa", student.gpa())
317 }

377 val student = student(a[0])
378 val source = course(a[1])
379 val destination = course(a[2])
380 if (source.code !in student.courses) {
381 fail("not enrolled " + student.id + " " + source.code)

387 fail("course full " + destination.code)
388 }
389 withdraw(student, source)
390 enroll(student, destination)
391 record(student, "swap " + source.code + " " + destination.code)
392 emit("SWAP", student.id, source.code, "->", destination.code,
393 "credits", student.credits.toString() + "/" + student.limit)

515 if (tba.isNotEmpty()) emit("TBA " + tba.joinToString(" "), indented = true)

518 private fun cmdFree(a: List<String>) {
519 val student = student(a[0])
520 val day = DAYS.indexOf(a[1])
521 if (day < 0) fail("invalid day " + a[1])
522 val occupied = studentSlots(student, day).mapNotNull { (start, end, _) ->
523 val first = maxOf(480, start)
524 val last = minOf(1320, end)
525 if (first < last) first to last else null
526 }
527 val free = mutableListOf<Pair<Int, Int>>()
528 var cursor = 480
529 for ((start, end) in occupied) {
530 if (cursor < start) free.add(cursor to start)
531 cursor = maxOf(cursor, end)
532 }
533 if (cursor < 1320) free.add(cursor to 1320)
534 val minutes = free.sumOf { it.second - it.first }
535 val parts = free.map { timeText(it.first) + "-" + timeText(it.second) }
536 emit((listOf("FREE", student.id, a[1], "total", timeText(minutes)) +
537 parts).joinToString(" "))
538 }

620 fun execute(line: String) {
621 commandNumber++
622 val tokens = line.split(' ')
623 val command = tokens[0]
624 val a = tokens.drop(1)
625 try {
626 when (command) {
627 "COURSE" -> cmdCourse(a)
628 "STUDENT" -> cmdStudent(a)
629 "ENROLL" -> cmdEnroll(a)
630 "CHECK" -> cmdCheck(a)
631 "OPTIONS" -> cmdOptions(a)
632 "DROP" -> cmdDrop(a)
633 "SWAP" -> cmdSwap(a)
634 "CLOSE" -> cmdClose(a)
635 "OPEN" -> cmdOpen(a)
636 "CAPACITY" -> cmdCapacity(a)
637 "LIMIT" -> cmdLimit(a)
638 "PREREQ" -> cmdPrereq(a)
639 "CANCEL" -> cmdCancel(a)
640 "ROSTER" -> cmdRoster(a)
641 "SCHEDULE" -> cmdSchedule(a)
