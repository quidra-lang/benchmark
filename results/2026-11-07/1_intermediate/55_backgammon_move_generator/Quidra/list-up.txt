Move[] sorted_moves(Move[] moves)
    for: 1 <= a < int(#moves)
        b := a
        while b > 0 and (moves[b].source > moves[b - 1].source or (moves[b].source == moves[b - 1].source and moves[b].target > moves[b - 1].target))
            temporary := moves[b]
            moves[b] = moves[b - 1]
            moves[b - 1] = temporary
            b -= 1
    return moves
