19def notation(moves, hits)
20 key = moves.map(&:dup).sort.reverse
21 loop do
22 index = key.index { |s,t| t > 0 && !hits.include?(t) && key.any? { |a,_| a == t } }
23 break unless index
24 a = key.delete_at(index)
25 other = key.index { |s,_| s == a[1] }
26 b = key.delete_at(other)
27 key << [a[0],b[1]]
28 key.sort!.reverse!
29 end
30 parts, marked, i = [],[],0
31 while i < key.size
32 s,t = key[i]
33 j = i+1
34 j += 1 while j < key.size && key[j] == key[i]
35 text = "#{s == 25 ? 'bar' : s}/#{t == 0 ? 'off' : t}"
36 if hits.include?(t) && !marked.include?(t)
37 text += '*'
38 marked << t
39 end
40 text += "(#{j-i})" if j-i > 1
41 parts << text
42 i = j
43 end
44 [key, parts.empty? ? 'none' : parts.join(' ')]
45end

47def enumerate(pos, side, d1, d2)
48 all = []
49 dice = d1 == d2 ? Array.new(4,d1) : [d1,d2]
50 visit = lambda do |board, remaining, moves, hits, used|
51 key,text = notation(moves,hits)
52 all << {pos:board,key:key,text:text,count:moves.size,used:used,id:position_key(board)}
53 own, opponent = board[side],board[1-side]
54 remaining.uniq.each do |d|
55 sources = own[25] > 0 ? [25] : (1..24).select { |p| own[p] > 0 }
56 sources.each do |s|
57 t = [0,s-d].max
58 if t > 0
59 next if opponent[25-t] >= 2
60 else
61 next if own[25] > 0 || (7..24).any? { |p| own[p] > 0 }
62 next if s < d && ((s+1)..6).any? { |p| own[p] > 0 }
63 end
64 next_board = board.map(&:dup)
65 next_board[side][s] -= 1
66 next_board[side][t] += 1
67 next_hits = hits.dup
68 if t > 0 && opponent[25-t] == 1
69 next_board[1-side][25-t] = 0
70 next_board[1-side][25] += 1
71 next_hits << t unless next_hits.include?(t)
72 end
73 rest = remaining.dup
74 rest.delete_at(rest.index(d))
75 visit.call(next_board,rest,moves+[[s,t]],next_hits,used+[d])
76 end
77 end
78 end
79 visit.call(pos,dice,[],[],[])
80 maximum = all.map { |p| p[:count] }.max
81 candidates = all.select { |p| p[:count] == maximum }
82 if maximum == 1 && d1 != d2 && candidates.any? { |p| p[:used][0] == [d1,d2].max }
83 candidates.select! { |p| p[:used][0] == [d1,d2].max }
84 end
85 unique = {}
86 candidates.each { |p| unique[p[:id]] ||= p }
87 legal = unique.values.sort { |a,b| b[:key] <=> a[:key] }
88 [legal,all,maximum]
89end

91def pips(own)
92 (1..25).sum { |p| p*own[p] }
93end

95def evaluation(pos, side)
96 own, opponent = pos[side],pos[1-side]
97 p = pips(opponent)-pips(own)
98 s = 0
99 (1..24).each do |point|
100 next unless own[point] == 1
101 distances = (0...point).select { |q| opponent[25-q] > 0 }.map { |q| point-q }
102 (1..6).each do |a|
103 (1..6).each do |b|
104 if distances.include?(a) || distances.include?(b) || distances.include?(a+b) || (a == b && (distances.include?(3*a) || distances.include?(4*a)))
105 s += 1
106 end
107 end
108 end
109 end
110 home = (1..6).count { |i| own[i] >= 2 }
111 anchors = (19..24).count { |i| own[i] >= 2 }
112 run, prime = 0,0
113 (1..24).each { |i| run = own[i] >= 2 ? run+1 : 0; prime = [prime,run].max }
114 bar = opponent[25]-own[25]
115 [2*p-s+4*home+3*prime+3*anchors+6*bar,p,s,home,prime,anchors,bar]
116end

118def keith(own)
119 pips(own)+2*[own[1]-1,0].max+[own[2]-1,0].max+[own[3]-3,0].max+(4..6).count { |i| own[i] == 0 }
120end

122def contact(pos)
123 w = pos[0][25] > 0 ? 25 : ((1..24).select { |p| pos[0][p] > 0 }.max || 0)
124 b = pos[1][25] > 0 ? 0 : (25-((1..24).select { |p| pos[1][p] > 0 }.max || 0))
125 w > b
126end

128def cube_decision(state, forbidden = false)
129 x = state[:side]
130 return 'unavailable' if state[:owner] == 1-x || state[:cube] == 64 || forbidden
131 if contact(state[:pos])
132 value = evaluation(state[:pos],x)[0]
133 return 'no double' if value < 60
134 value <= 100 ? 'double take' : 'double drop'
135 else
136 value = 8*keith(state[:pos][x])-7*keith(state[:pos][1-x])
137 limit = state[:owner] == -1 ? 28 : 21
138 return 'no double' if value > limit
139 value >= 14 ? 'double take' : 'double drop'
140 end
141end

143def end_game(state,winner,kind = nil)
144 unless kind
145 opponent = state[:pos][1-winner]
146 kind = if opponent[0] > 0 then 'single'
147 elsif opponent[25] > 0 || (19..24).any? { |p| opponent[p] > 0 } then 'backgammon'
148 else 'gammon'
149 end
150 end
151 multiplier = {'single'=>1,'gammon'=>2,'backgammon'=>3,'drop'=>1}[kind]
152 points = state[:cube]*multiplier
153 state[:pending], state[:over] = false,true
154 state[:score][winner] += points
155 [winner,kind,points]
156end

158def apply_play(state,play)
159 state[:pos] = copy(play[:pos])
160 if state[:pos][state[:side]][0] == 15
161 end_game(state,state[:side])
162 else
163 state[:side] = 1-state[:side]
164 nil
165 end
166end

168def ranked(legal,side)
169 legal.each_with_index.map { |p,i| [evaluation(p[:pos],side)[0],i,p] }.sort_by { |e,i,_| [-e,i] }
170end

172def parse_play(tokens)
173 return [] if tokens == ['none']
174 moves = []
175 tokens.each do |token|
176 match = token.match(/\A(.+?)(?:\(([234])\))?\z/)
177 raise 'syntax' unless match
178 path = match[1].split('/',-1)
179 raise 'syntax' if path.size < 2
180 values = path.each_with_index.map do |word,i|
181 if i == 0
182 if word == 'bar' then 25
183 elsif word.match?(/\A(?:[1-9]|1[0-9]|2[0-4])\z/) then word.to_i
184 else raise 'syntax'
185 end
186 else
187 if word == 'off' then 0
188 elsif word.match?(/\A(?:[1-9]|1[0-9]|2[0-4])\*?\z/) then word.to_i
189 else raise 'syntax'
190 end
191 end
192 end
193 raise 'syntax' unless values.each_cons(2).all? { |s,t| s > t }
194 values.each_cons(2) { |pair| (match[2] || '1').to_i.times { moves << pair } }
195 end
196 moves
197end

199class RandomStream
200 def initialize(seed)
201 @value = seed
202 end
203 def next_value
204 @value = @value*48271%2147483647
205 end
206 def die
207 next_value%6+1
208 end
209end

211def simulate(state,rng,strategies,opening = false,target = nil,scores = nil,crawford = false,verbose = false)
212 state = copy(state)
213 dice = nil
214 if opening
215 loop do
216 dice = [rng.die,rng.die]
217 break if dice[0] != dice[1]
218 end
219 state[:side] = dice[0] > dice[1] ? 0 : 1
220 end
221 first, turn, plays = state[:side],1,0
222 loop do
223 x = state[:side]
224 unless opening && turn == 1
225 forbidden = target && (crawford || scores[x]+state[:cube] >= target)
226 decision = cube_decision(state,forbidden)
227 if decision.start_with?('double')
228 puts "#{turn} #{color(x)} doubles #{state[:cube]*2}" if verbose
229 if decision == 'double drop'
230 puts "#{turn} #{color(1-x)} drops" if verbose
231 return [end_game(state,x,'drop'),plays,first]
232 end
233 puts "#{turn} #{color(1-x)} takes" if verbose
234 state[:cube] *= 2
235 state[:owner] = 1-x
236 end
237 dice = [rng.die,rng.die]
238 end
239 legal = enumerate(state[:pos],x,*dice)[0]
240 play = case strategies[x]
241 when 'FIRST' then legal[0]
242 when 'RANDOM' then legal[rng.next_value%legal.size]
243 else ranked(legal,x)[0][2]
244 end
245 puts "#{turn} #{color(x)} #{dice[0]}-#{dice[1]} #{play[:text]}" if verbose
246 result = apply_play(state,play)
247 plays += 1
248 return [result,plays,first] if result
249 turn += 1
250 end
251end
