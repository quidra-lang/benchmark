17fn parsedNumber(word: Text) ?I {
18 if (word.len == 0 or (word.len > 1 and word[0] == '0')) return null;
19 for (word) |c| if (!std.ascii.isDigit(c)) return null;
20 return std.fmt.parseInt(I, word, 10) catch null;
21}
22fn inRange(word: Text, low: I, high: I) bool {
23 const value = parsedNumber(word) orelse return false;
24 return value >= low and value <= high;

36fn positionsEqual(a: Position, b: Position) bool {
37 return std.mem.eql(I, &a[0], &b[0]) and std.mem.eql(I, &a[1], &b[1]);

39fn standardPosition() Position {
40 var pos: Position = .{ @splat(0), @splat(0) };
41 for (0..2) |side| {
42 pos[side][24] = 2;
43 pos[side][13] = 5;
44 pos[side][8] = 3;
45 pos[side][6] = 5;
46 }
47 return pos;
48}
49fn pips(pos: Position, side: usize) I {
50 var result: I = 0;
51 for (1..26) |point| result += @as(I, @intCast(point)) * pos[side][point];
52 return result;
53}
54fn keith(pos: Position, side: usize) I {
55 const own = pos[side];
56 var result = pips(pos, side) + 2 * @max(own[1] - 1, 0) + @max(own[2] - 1, 0) + @max(own[3] - 3, 0);
57 for (4..7) |point| if (own[point] == 0) {
58 result += 1;
59 };
60 return result;
61}
62fn contact(pos: Position) bool {
63 var white: I = 0;
64 var black: I = 25;
65 for (1..26) |point| {
66 if (pos[0][point] > 0) white = @intCast(point);
67 if (pos[1][point] > 0) black = 25 - @as(I, @intCast(point));
68 }
69 return white > black;
70}
71const Evaluation = struct { value: I, pip_delta: I, blots: I, home: I, prime: I, anchors: I, bar_delta: I };
72fn evaluate(pos: Position, side: usize) Evaluation {
73 const other = 1 - side;
74 const own = pos[side];
75 var s: I = 0;
76 var h: I = 0;
77 var r: I = 0;
78 var run: I = 0;
79 var anchors_count: I = 0;
80 for (1..25) |point| {
81 if (own[point] >= 2) {
82 run += 1;
83 r = @max(r, run);
84 if (point <= 6) h += 1;
85 if (point >= 19) anchors_count += 1;
86 } else run = 0;
87 if (own[point] == 1) {
88 var distances: [25]bool = @splat(false);
89 for (0..point) |q| if (pos[other][25 - q] > 0) {
90 distances[point - q] = true;
91 };
92 for (1..7) |a| for (1..7) |b| {
93 if (distances[a] or distances[b] or distances[a + b] or (a == b and (distances[3 * a] or distances[4 * a]))) s += 1;
94 };
95 }
96 }
97 const p = pips(pos, other) - pips(pos, side);
98 const g = pos[other][25] - own[25];
99 return .{ .value = 2 * p - s + 4 * h + 3 * r + 3 * anchors_count + 6 * g, .pip_delta = p, .blots = s, .home = h, .prime = r, .anchors = anchors_count, .bar_delta = g };
100}
101const Move = struct { source: usize, target: usize };
102const Sequence = struct {
103 values: [4]Move = undefined,
104 len: usize = 0,
105 fn add(s: *Sequence, pair: Move) void {
106 s.values[s.len] = pair;
107 s.len += 1;
108 }
109 fn erase(s: *Sequence, at: usize) void {
110 var j = at;
111 while (j + 1 < s.len) : (j += 1) s.values[j] = s.values[j + 1];
112 s.len -= 1;
113 }
114 fn sort(s: *Sequence) void {
115 std.mem.sort(Move, s.values[0..s.len], {}, struct {
116 fn less(_: void, a: Move, b: Move) bool {
117 return a.source > b.source or (a.source == b.source and a.target > b.target);
118 }
119 }.less);
120 }
121};
122fn canonicalKey(sequence: Sequence, hits: [26]bool) Sequence {
123 var result = sequence;
124 result.sort();
125 while (true) {
126 var first: ?usize = null;
127 var second: usize = 0;
128 for (result.values[0..result.len], 0..) |pair, j| {
129 if (pair.target == 0 or hits[pair.target]) continue;
130 for (result.values[0..result.len], 0..) |candidate, k| {
131 if (candidate.source == pair.target) {
132 first = j;
133 second = k;
134 break;
135 }
136 }
137 if (first != null) break;
138 }
139 const a = first orelse return result;
140 const merged: Move = .{ .source = result.values[a].source, .target = result.values[second].target };
141 result.erase(@max(a, second));
142 result.erase(@min(a, second));
143 result.add(merged);
144 result.sort();
145 }
146}
147const Play = struct {
148 final_position: Position,
149 key: Sequence,
150 hits: [26]bool,
151 used: usize,
152 first_die: usize,
153 fn notation(p: Play, h: Host) !Text {
154 if (p.key.len == 0) return "none";
155 var b: std.Io.Writer.Allocating = .init(h.a);
156 var printed_hit: [26]bool = @splat(false);
157 var at: usize = 0;
158 while (at < p.key.len) {
159 const pair = p.key.values[at];
160 var count: usize = 1;
161 while (at + count < p.key.len and (p.key.values[at + count].source == pair.source and p.key.values[at + count].target == pair.target)) count += 1;
162 if (at > 0) try b.writer.writeByte(' ');
163 if (pair.source == 25) try b.writer.writeAll("bar") else try b.writer.print("{d}", .{pair.source});
164 try b.writer.writeByte('/');
165 if (pair.target == 0) try b.writer.writeAll("off") else try b.writer.print("{d}", .{pair.target});
166 if (p.hits[pair.target] and !printed_hit[pair.target]) {
167 try b.writer.writeByte('*');
168 printed_hit[pair.target] = true;
169 }
170 if (count > 1) try b.writer.print("({d})", .{count});
171 at += count;
172 }
173 return b.toOwnedSlice();
174 }
175};
176fn before(_: void, a: Play, b: Play) bool {
177 for (0..@min(a.key.len, b.key.len)) |j| {
178 if (a.key.values[j].source != b.key.values[j].source) return a.key.values[j].source > b.key.values[j].source;
179 if (a.key.values[j].target != b.key.values[j].target) return a.key.values[j].target > b.key.values[j].target;
180 }
181 return a.key.len > b.key.len;
182}
183fn enumeratePlays(a: std.mem.Allocator, pos: Position, side: usize, remaining: [7]usize, sequence: Sequence, hits: [26]bool, first_die: usize, results: *std.ArrayList(Play)) anyerror!void {
184 try results.append(a, .{ .final_position = pos, .key = canonicalKey(sequence, hits), .hits = hits, .used = sequence.len, .first_die = first_die });
185 const own = pos[side];
186 var home_only = true;
187 for (7..26) |point| if (own[point] > 0) {
188 home_only = false;
189 };
190 for (1..7) |die| {
191 if (remaining[die] == 0) continue;
192 var source: usize = 25;
193 while (source >= 1) : (source -= 1) {
194 if (own[source] == 0 or (own[25] > 0 and source != 25)) continue;
195 const signed_target = @as(I, @intCast(source)) - @as(I, @intCast(die));
196 var target: usize = 0;
197 if (signed_target > 0) {
198 target = ix(signed_target);
199 if (pos[1 - side][25 - target] >= 2) continue;
200 } else {
201 if (!home_only) continue;
202 var farther = false;
203 if (source < 6) for (source + 1..7) |point| {
204 farther = farther or own[point] > 0;
205 };
206 if (signed_target < 0 and farther) continue;
207 }
208 var next_pos = pos;
209 var next_hits = hits;
210 next_pos[side][source] -= 1;
211 next_pos[side][target] += 1;
212 if (target > 0 and next_pos[1 - side][25 - target] == 1) {
213 next_pos[1 - side][25 - target] = 0;
214 next_pos[1 - side][25] += 1;
215 next_hits[target] = true;
216 }
217 var next_remaining = remaining;
218 next_remaining[die] -= 1;
219 var next_sequence = sequence;
220 next_sequence.add(.{ .source = source, .target = target });
221 try enumeratePlays(a, next_pos, side, next_remaining, next_sequence, next_hits, if (sequence.len == 0) die else first_die, results);
222 }
223 }
224}
225const Generated = struct { possible: std.ArrayList(Play), legal: std.ArrayList(Play), maximum_used: usize };
226fn generate(a: std.mem.Allocator, pos: Position, side: usize, d1: usize, d2: usize) !Generated {
227 var remaining: [7]usize = @splat(0);
228 remaining[d1] += if (d1 == d2) 4 else 1;
229 if (d1 != d2) remaining[d2] += 1;
230 var possible: std.ArrayList(Play) = .empty;
231 try enumeratePlays(a, pos, side, remaining, .{}, @splat(false), 0, &possible);
232 var maximum_used: usize = 0;
233 var larger = false;
234 for (possible.items) |play| {
235 maximum_used = @max(maximum_used, play.used);
236 larger = larger or (play.used == 1 and play.first_die == @max(d1, d2));
237 }
238 var legal: std.ArrayList(Play) = .empty;
239 for (possible.items) |play| {
240 if (play.used != maximum_used or (maximum_used == 1 and d1 != d2 and larger and play.first_die != @max(d1, d2))) continue;
241 var duplicate = false;
242 for (legal.items) |prior| if (positionsEqual(prior.final_position, play.final_position)) {
243 duplicate = true;
244 break;
245 };
246 if (!duplicate) try legal.append(a, play);
247 }
248 std.mem.sort(Play, legal.items, {}, before);
249 return .{ .possible = possible, .legal = legal, .maximum_used = maximum_used };
250}
251fn bestPlay(plays: []const Play, side: usize) usize {
252 var chosen: usize = 0;
253 var best_value = evaluate(plays[0].final_position, side).value;
254 for (plays[1..], 1..) |play, j| {
255 const value = evaluate(play.final_position, side).value;
256 if (value > best_value) {
257 chosen = j;
258 best_value = value;
259 }
260 }
261 return chosen;
262}
263const State = struct {
264 pos: Position = standardPosition(),
265 side: usize = 0,
266 cube: I = 1,
267 owner: ?usize = null,
268 pending: bool = false,
269 over: bool = false,
270 score: [2]I = .{ 0, 0 },
271 winner: usize = 0,
272 kind: Text = "",
273 points: I = 0,
274 fn ownerText(s: State) Text {
275 return if (s.owner) |owner| color(owner) else "center";
276 }
277 fn cubeDecision(s: State, prohibited: bool) Text {
278 if (s.owner == 1 - s.side or s.cube == 64 or prohibited) return "unavailable";
279 if (contact(s.pos)) {
280 const value = evaluate(s.pos, s.side).value;
281 return if (value < 60) "no double" else if (value <= 100) "double take" else "double drop";
282 }
283 const value = 8 * keith(s.pos, s.side) - 7 * keith(s.pos, 1 - s.side);
284 const limit: I = if (s.owner == null) 28 else 21;
285 return if (value > limit) "no double" else if (value >= 14) "double take" else "double drop";
286 }
287 fn finish(s: *State, winning_side: usize, result_kind: Text) void {
288 s.over = true;
289 s.pending = false;
290 s.winner = winning_side;
291 s.kind = result_kind;
292 s.points = multiplier(result_kind) * s.cube;
293 s.score[winning_side] += s.points;
294 }
295 fn applyPlay(s: *State, play: Play) void {
296 s.pos = play.final_position;
297 if (s.pos[s.side][0] == 15) {
298 const other = 1 - s.side;
299 var result_kind: Text = "single";
300 if (s.pos[other][0] == 0) {
301 result_kind = "gammon";
302 var backgammon = s.pos[other][25] > 0;
303 for (19..25) |point| backgammon = backgammon or s.pos[other][point] > 0;
304 if (backgammon) result_kind = "backgammon";
305 }
306 s.finish(s.side, result_kind);
307 } else s.side = 1 - s.side;
308 }
309 fn resultLine(s: State, h: Host) !Text {
310 return h.text("result {s} {s} {d}", .{ color(s.winner), s.kind, s.points });
311 }

323const GameResult = struct { state_value: State, first_player: usize, turns: usize };
324fn simulate(h: Host, start_state: State, strategies: [2]Text, generator: *Random, opening: bool, match_limit: I, match_scores: [2]I, crawford: bool, verbose: bool) !GameResult {
325 var current = start_state;
326 var d1: usize = 0;
327 var d2: usize = 0;
328 if (opening) {
329 while (d1 == d2) {
330 d1 = generator.die();
331 d2 = generator.die();
332 }
333 current.side = if (d1 > d2) 0 else 1;
334 }
335 const first_player = current.side;
336 var turn_number: usize = 1;
337 var plays_made: usize = 0;
338 while (!current.over) : (turn_number += 1) {
339 const side = current.side;
340 const other = 1 - side;
341 if (!(opening and turn_number == 1)) {
342 const prohibited = crawford or (match_limit > 0 and match_scores[side] + current.cube >= match_limit);
343 const decision = current.cubeDecision(prohibited);
344 if (eq(decision, "double take") or eq(decision, "double drop")) {
345 if (verbose) {
346 try h.line("{d} {s} doubles {d}", .{ turn_number, color(side), current.cube * 2 });
347 try h.line("{d} {s} {s}", .{ turn_number, color(other), if (eq(decision, "double take")) @as(Text, "takes") else "drops" });
348 }
349 if (eq(decision, "double drop")) {
350 current.finish(side, "drop");
351 break;
352 }
353 current.cube *= 2;
354 current.owner = other;
355 }
356 d1 = generator.die();
357 d2 = generator.die();
358 }
359 var scratch: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
360 defer scratch.deinit();
361 const generated = try generate(scratch.allocator(), current.pos, side, d1, d2);
362 const plays = generated.legal.items;
363 const strategy = strategies[side];
364 const choice = if (eq(strategy, "BEST")) bestPlay(plays, side) else if (eq(strategy, "RANDOM")) ix(@mod(generator.nextValue(), @as(I, @intCast(plays.len)))) else 0;
365 if (verbose) try h.line("{d} {s} {d}-{d} {s}", .{ turn_number, color(side), d1, d2, try plays[choice].notation(h) });
366 current.applyPlay(plays[choice]);
367 plays_made += 1;
368 }
369 if (verbose) try h.line("{s}", .{try current.resultLine(h)});
370 return .{ .state_value = current, .first_player = first_player, .turns = plays_made };
371}
372fn parsePlay(a: std.mem.Allocator, elements: []const Text) ![]Move {
373 if (elements.len == 1 and eq(elements[0], "none")) return &.{};
374 var result: std.ArrayList(Move) = .empty;
375 for (elements) |original| {
376 var element = original;
377 var copies: usize = 1;
378 if (std.mem.endsWith(u8, element, ")")) {
379 const size = element.len;
380 if (size < 3 or element[size - 3] != '(' or element[size - 2] < '2' or element[size - 2] > '4') return error.Syntax;
381 copies = element[size - 2] - '0';
382 element = element[0 .. size - 3];
383 }
384 if (std.mem.indexOfAny(u8, element, "()") != null) return error.Syntax;
385 var parts = std.mem.splitScalar(u8, element, '/');
386 const first = parts.next().?;
387 const source_value: I = if (eq(first, "bar")) 25 else parsedNumber(first) orelse return error.Syntax;
388 if (source_value < 1 or source_value > 25 or (source_value == 25 and !eq(first, "bar"))) return error.Syntax;
389 var source = ix(source_value);
390 var count_parts: usize = 1;
391 while (parts.next()) |original_part| {
392 var part = original_part;
393 if (std.mem.endsWith(u8, part, "*")) {
394 part = part[0 .. part.len - 1];
395 if (eq(part, "off")) return error.Syntax;
396 }
397 const target_value: I = if (eq(part, "off")) 0 else parsedNumber(part) orelse return error.Syntax;
398 if (target_value < 0 or target_value > 24 or (target_value == 0 and !eq(part, "off")) or ix(target_value) >= source) return error.Syntax;
399 const target = ix(target_value);
400 for (0..copies) |_| try result.append(a, .{ .source = source, .target = target });
401 source = target;
402 count_parts += 1;
403 }
404 if (count_parts < 2) return error.Syntax;
405 }
406 return result.toOwnedSlice(a);
407}

423 fn boardRow(s: *const Referee, upper: bool, row: I) !void {
424 for (0..12) |j| {
425 const point = if (upper) 13 + j else 12 - j;
426 if (j == 6) try s.h.out.writeAll(" |");
427 var cell = try s.h.text("{d}", .{point});
428 if (row > 0) {
429 const white = s.state_value.pos[0][point];
430 const black = s.state_value.pos[1][25 - point];
431 const count_men = @max(white, black);
432 cell = if (count_men < row) "." else if (row == 5 and count_men > 5) try s.h.text("{d}", .{count_men}) else if (white > 0) "W" else "B";
433 }
434 try s.h.out.splatByteAll(' ', 3 - cell.len);
435 try s.h.out.writeAll(cell);
436 }
437 try s.h.out.writeByte('\n');
438 }
439 fn command(s: *Referee, words: []const Text) !void {
440 const op = words[0];
441 const no_args = [_]Text{ "NEW", "SHOW", "BOARD", "PIPS", "RACE", "EVAL", "SHOTS", "CUBE", "DOUBLE", "TAKE", "DROP", "SCORE", "HISTORY" };
442 var known = false;
443 var valid = false;
444 for (no_args) |candidate| {
445 if (eq(candidate, op)) {
446 known = true;
447 valid = words.len == 1;
448 }
449 }
450 if (eq(op, "POS")) {
451 if (words.len != 5 or !inRange(words[2], 0, 15) or !inRange(words[3], 0, 15) or (!eq(words[4], "W") and !eq(words[4], "B"))) return s.h.line("ERROR bad position", .{});
452 var next_state: State = .{};
453 next_state.pos = .{ @splat(0), @splat(0) };
454 next_state.score = s.state_value.score;
455 next_state.side = if (eq(words[4], "W")) 0 else 1;
456 var totals: [2]I = .{ parsedNumber(words[2]).?, parsedNumber(words[3]).? };
457 next_state.pos[0][25] = totals[0];
458 next_state.pos[1][25] = totals[1];
459 var points = std.mem.splitScalar(u8, words[1], ',');
460 var point: usize = 1;
461 while (points.next()) |word| : (point += 1) {
462 if (point > 24) return s.h.line("ERROR bad position", .{});
463 const value = std.fmt.parseInt(I, word, 10) catch return s.h.line("ERROR bad position", .{});
464 if (value > 0) {
465 next_state.pos[0][point] = value;
466 totals[0] += value;
467 } else if (value < 0) {
468 next_state.pos[1][25 - point] = -value;
469 totals[1] -= value;
470 }
471 }
472 if (point != 25 or totals[0] < 1 or totals[0] > 15 or totals[1] < 1 or totals[1] > 15) return s.h.line("ERROR bad position", .{});
473 next_state.pos[0][0] = 15 - totals[0];
474 next_state.pos[1][0] = 15 - totals[1];
475 s.state_value = next_state;
476 s.history = .empty;
477 s.redo_list = .empty;
478 return s.h.line("OK", .{});
479 }
480 if (eq(op, "MOVES") or eq(op, "PLAY") or eq(op, "AUTO") or eq(op, "HINT")) {
481 known = true;
482 valid = if (eq(op, "PLAY")) words.len >= 4 else if (eq(op, "HINT")) words.len == 4 else words.len == 3;
483 if (valid) valid = inRange(words[1], 1, 6) and inRange(words[2], 1, 6) and (!eq(op, "HINT") or inRange(words[3], 1, 1000));
484 } else if (eq(op, "UNDO") or eq(op, "REDO")) {
485 known = true;
486 valid = words.len == 2 and inRange(words[1], 1, 1000);
487 } else if (eq(op, "RESIGN")) {
488 known = true;
489 valid = words.len == 2 and (eq(words[1], "single") or eq(words[1], "gammon") or eq(words[1], "backgammon"));
490 } else if (eq(op, "GAME")) {
491 known = true;
492 valid = words.len == 4;
493 if (valid) {
494 valid = inRange(words[1], 1, 2147483646);
495 for (words[2..4]) |strategy| valid = valid and (eq(strategy, "BEST") or eq(strategy, "FIRST") or eq(strategy, "RANDOM"));
496 }
497 } else if (eq(op, "MATCH") or eq(op, "STATS")) {
498 known = true;
499 valid = words.len == 3 and inRange(words[1], 1, if (eq(op, "MATCH")) 15 else 50) and inRange(words[2], 1, 2147483646);
500 }
501 if (!known) return s.h.line("ERROR unknown command", .{});
502 if (!valid) return s.h.line("ERROR bad arguments", .{});
503 const disallow_over = eq(op, "RACE") or eq(op, "MOVES") or eq(op, "PLAY") or eq(op, "AUTO") or eq(op, "HINT") or eq(op, "EVAL") or eq(op, "SHOTS") or eq(op, "CUBE") or eq(op, "DOUBLE") or eq(op, "TAKE") or eq(op, "DROP") or eq(op, "RESIGN") or eq(op, "GAME");
504 if (disallow_over and s.state_value.over) return s.h.line("ERROR game over", .{});
505 const disallow_pending = eq(op, "PLAY") or eq(op, "AUTO") or eq(op, "CUBE") or eq(op, "DOUBLE") or eq(op, "RESIGN") or eq(op, "GAME");
506 if (disallow_pending and s.state_value.pending) return s.h.line("ERROR double pending", .{});
507 const side = s.state_value.side;
508 const pos = s.state_value.pos;
509 if (eq(op, "NEW")) {
510 const old_score = s.state_value.score;
511 s.state_value = .{ .score = old_score };
512 s.history = .empty;
513 s.redo_list = .empty;
514 return s.h.line("OK", .{});
515 }
516 if (eq(op, "SHOW")) {
517 try s.h.out.writeAll("POS ");
518 for (1..25) |point| {
519 if (point > 1) try s.h.out.writeByte(',');
520 try s.h.out.print("{d}", .{pos[0][point] - pos[1][25 - point]});
521 }
522 return s.h.line(" {d} {d} {s}", .{ pos[0][25], pos[1][25], color(side) });
523 }
524 if (eq(op, "BOARD")) {
525 try s.boardRow(true, 0);
526 for (1..6) |row| try s.boardRow(true, @intCast(row));
527 const status = if (s.state_value.over) "game over" else try s.h.text("{s} {s}", .{ color(side), if (s.state_value.pending) @as(Text, "doubles") else "to move" });
528 try s.h.line("bar W{d} B{d} | off W{d} B{d} | cube {d} {s} | {s}", .{ pos[0][25], pos[1][25], pos[0][0], pos[1][0], s.state_value.cube, s.state_value.ownerText(), status });
529 var row: I = 5;
530 while (row >= 1) : (row -= 1) try s.boardRow(false, row);
531 return s.boardRow(false, 0);
532 }
533 if (eq(op, "PIPS")) return s.h.line("pips W {d} B {d}", .{ pips(pos, 0), pips(pos, 1) });
534 if (eq(op, "RACE")) {
535 if (contact(pos)) return s.h.line("contact", .{});
536 return s.h.line("race W {d} B {d}", .{ keith(pos, 0), keith(pos, 1) });
537 }
538 if (eq(op, "EVAL")) {
539 const ev = evaluate(pos, side);
540 return s.h.line("eval {d} pips {d} blots {d} home {d} prime {d} anchors {d} bar {d}", .{ ev.value, ev.pip_delta, ev.blots, ev.home, ev.prime, ev.anchors, ev.bar_delta });
541 }
542 if (eq(op, "CUBE")) return s.h.line("cube {s}", .{s.state_value.cubeDecision(false)});
543 if (eq(op, "SCORE")) return s.h.line("score W {d} B {d}", .{ s.state_value.score[0], s.state_value.score[1] });
544 if (eq(op, "HISTORY")) {
545 try s.h.line("history {d}", .{s.history.items.len});
546 for (s.history.items, 1..) |entry, j| try s.h.line("{d} {s}", .{ j, entry.record_text });
547 return;
548 }
549 if (eq(op, "UNDO") or eq(op, "REDO")) {
550 const count = ix(parsedNumber(words[1]).?);
551 if (eq(op, "UNDO")) {
552 if (count > s.history.items.len) return s.h.line("ERROR cannot undo", .{});
553 const at = s.history.items.len - count;
554 s.state_value = s.history.items[at].before_state;
555 var new_redo: std.ArrayList(Entry) = .empty;
556 try new_redo.appendSlice(s.h.a, s.history.items[at..]);
557 try new_redo.appendSlice(s.h.a, s.redo_list.items);
558 s.redo_list = new_redo;
559 s.history.shrinkRetainingCapacity(at);
560 } else {
561 if (count > s.redo_list.items.len) return s.h.line("ERROR cannot redo", .{});
562 s.state_value = s.redo_list.items[count - 1].after_state;
563 try s.history.appendSlice(s.h.a, s.redo_list.items[0..count]);
564 var new_redo: std.ArrayList(Entry) = .empty;
565 try new_redo.appendSlice(s.h.a, s.redo_list.items[count..]);
566 s.redo_list = new_redo;
567 }
568 return s.h.line("OK", .{});
569 }
570 if (eq(op, "DOUBLE")) {
571 if (s.state_value.owner == 1 - side or s.state_value.cube == 64) return s.h.line("ERROR cube unavailable", .{});
572 s.state_value.pending = true;
573 s.redo_list = .empty;
574 return s.h.line("OK double {d}", .{s.state_value.cube * 2});
575 }
576 if (eq(op, "TAKE") or eq(op, "DROP")) {
577 if (!s.state_value.pending) return s.h.line("ERROR no double", .{});
578 s.state_value.pending = false;
579 s.redo_list = .empty;
580 if (eq(op, "TAKE")) {
581 s.state_value.cube *= 2;
582 s.state_value.owner = 1 - side;
583 return s.h.line("OK take {d} {s}", .{ s.state_value.cube, s.state_value.ownerText() });
584 }
585 s.state_value.finish(side, "drop");
586 try s.h.line("OK drop", .{});
587 return s.h.line("{s}", .{try s.state_value.resultLine(s.h)});
588 }
589 if (eq(op, "RESIGN")) {
590 s.state_value.finish(1 - side, words[1]);
591 s.redo_list = .empty;
592 try s.h.line("OK resign", .{});
593 return s.h.line("{s}", .{try s.state_value.resultLine(s.h)});
594 }
595 if (eq(op, "GAME")) {
596 var generator: Random = .{ .seed = parsedNumber(words[1]).? };
597 _ = try simulate(s.h, s.state_value, .{ words[2], words[3] }, &generator, false, 0, .{ 0, 0 }, false, true);
598 return;
599 }
600 if (eq(op, "MATCH") or eq(op, "STATS")) {
601 const count = parsedNumber(words[1]).?;
602 var generator: Random = .{ .seed = parsedNumber(words[2]).? };
603 var scores: [2]I = .{ 0, 0 };
604 var wins: [4][2]usize = .{ @splat(0), @splat(0), @splat(0), @splat(0) };
605 const kinds = [_]Text{ "single", "gammon", "backgammon", "drop" };
606 var total_turns: usize = 0;
607 var game_number: I = 0;
608 var crawford_used = false;
609 while ((eq(op, "STATS") and game_number < count) or (eq(op, "MATCH") and @max(scores[0], scores[1]) < count)) {
610 const crawford = eq(op, "MATCH") and !crawford_used and @max(scores[0], scores[1]) == count - 1;
611 crawford_used = crawford_used or crawford;
612 const result = try simulate(s.h, .{}, .{ "BEST", "BEST" }, &generator, true, if (eq(op, "MATCH")) count else 0, scores, crawford, false);
613 const ended = result.state_value;
614 scores[ended.winner] += ended.points;
615 total_turns += result.turns;
616 game_number += 1;
617 for (kinds, 0..) |kind, j| if (eq(kind, ended.kind)) {
618 wins[j][ended.winner] += 1;
619 };
620 if (eq(op, "MATCH")) try s.h.line("game {d}{s} start {s} turns {d} {s} score {d}-{d}", .{ game_number, if (crawford) @as(Text, " crawford") else "", color(result.first_player), result.turns, try ended.resultLine(s.h), scores[0], scores[1] });
621 }
622 if (eq(op, "MATCH")) return s.h.line("match {s} {d}-{d}", .{ color(if (scores[0] >= count) 0 else 1), scores[0], scores[1] });
623 for (kinds, wins) |kind, counts| try s.h.line("{s} W {d} B {d}", .{ kind, counts[0], counts[1] });
624 try s.h.line("points W {d} B {d}", .{ scores[0], scores[1] });
625 return s.h.line("turns {d}", .{total_turns});
626 }
627 if (eq(op, "SHOTS")) {
628 var counts: [26]I = @splat(0);
629 var total: I = 0;
630 var rolls: std.Io.Writer.Allocating = .init(s.h.a);
631 var a: usize = 6;
632 while (a >= 1) : (a -= 1) {
633 var b = a;
634 while (b >= 1) : (b -= 1) {
635 var scratch: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
636 defer scratch.deinit();
637 const generated = try generate(scratch.allocator(), pos, 1 - side, a, b);
638 var hit_points: [26]bool = @splat(false);
639 for (generated.legal.items) |play| for (1..25) |point| {
640 if (play.hits[point]) hit_points[25 - point] = true;
641 };
642 var any_hit = false;
643 const weight: I = if (a == b) 1 else 2;
644 for (1..25) |point| {
645 if (pos[side][point] == 1 and hit_points[point]) {
646 counts[point] += weight;
647 any_hit = true;
648 }
649 }
650 if (any_hit) {
651 total += weight;
652 try rolls.writer.print(" {d}-{d}", .{ a, b });
653 }
654 }
655 }
656 try s.h.line("shots {d}/36{s}", .{ total, rolls.written() });
657 var point: usize = 24;
658 while (point >= 1) : (point -= 1) if (pos[side][point] == 1) {
659 try s.h.line("blot {d} {d}/36", .{ point, counts[point] });
660 };
661 return;
662 }
663 const d1 = ix(parsedNumber(words[1]).?);
664 const d2 = ix(parsedNumber(words[2]).?);
665 var scratch: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
666 defer scratch.deinit();
667 const generated = try generate(scratch.allocator(), pos, side, d1, d2);
668 if (eq(op, "MOVES")) {
669 for (generated.legal.items) |play| try s.h.line("{s}", .{try play.notation(s.h)});
670 return s.h.line("total {d}", .{generated.legal.items.len});
671 }
672 if (eq(op, "HINT")) {
673 const ranked = generated.legal.items;
674 var j: usize = 1;
675 while (j < ranked.len) : (j += 1) {
676 var k = j;
677 while (k > 0 and evaluate(ranked[k].final_position, side).value > evaluate(ranked[k - 1].final_position, side).value) : (k -= 1) std.mem.swap(Play, &ranked[k], &ranked[k - 1]);
678 }
679 for (ranked[0..@min(ix(parsedNumber(words[3]).?), ranked.len)]) |play| try s.h.line("{d} {s}", .{ evaluate(play.final_position, side).value, try play.notation(s.h) });
680 return;
681 }
682 if (eq(op, "AUTO")) {
683 const chosen = generated.legal.items[bestPlay(generated.legal.items, side)];
684 return s.makePlay(chosen, d1, d2, try s.h.text("OK {s} eval {d}", .{ try chosen.notation(s.h), evaluate(chosen.final_position, side).value }));
685 }
686 const moves = parsePlay(scratch.allocator(), words[3..]) catch |err| {
687 if (err == error.Syntax) return s.h.line("ILLEGAL syntax", .{});
688 return err;
689 };
690 var final_position = pos;
691 var destinations: [26]bool = @splat(false);
692 var nonbar = false;
693 var bearing = false;
694 var blocked = false;
695 for (moves) |pair| {
696 final_position[side][pair.source] -= 1;
697 final_position[side][pair.target] += 1;
698 destinations[pair.target] = true;
699 nonbar = nonbar or pair.source != 25;
700 bearing = bearing or pair.target == 0;
701 blocked = blocked or (pair.target > 0 and pos[1 - side][25 - pair.target] >= 2);
702 }
703 for (1..25) |point| if (destinations[point] and pos[1 - side][25 - point] == 1) {
704 final_position[1 - side][25 - point] = 0;
705 final_position[1 - side][25] += 1;
706 };
707 for (1..26) |point| if (final_position[side][point] < 0) return s.h.line("ILLEGAL no checker", .{});
708 if (final_position[side][25] > 0 and nonbar) return s.h.line("ILLEGAL bar", .{});
709 if (blocked) return s.h.line("ILLEGAL blocked", .{});
710 if (bearing) for (7..26) |point| {
711 if (final_position[side][point] > 0) return s.h.line("ILLEGAL bear-off", .{});
712 };
713 for (generated.legal.items) |play| if (positionsEqual(play.final_position, final_position)) {
714 return s.makePlay(play, d1, d2, try s.h.text("OK {s}", .{try play.notation(s.h)}));
715 };
716 var possible = false;
717 var too_few = false;
718 for (generated.possible.items) |play| if (positionsEqual(play.final_position, final_position)) {
719 possible = true;
720 too_few = too_few or play.used < generated.maximum_used;
721 };
722 return s.h.line("ILLEGAL {s}", .{if (too_few) @as(Text, "too few dice") else if (possible) "larger die required" else "impossible"});
723 }
724};
725fn readLine(r: *std.Io.Reader, a: std.mem.Allocator) !?Text {
726 var b: std.Io.Writer.Allocating = .init(a);
727 defer b.deinit();
728 _ = try r.streamDelimiterEnding(&b.writer, '\n');
729 if (r.bufferedLen() == 0 and b.written().len == 0) return null;
730 if (r.bufferedLen() > 0) r.toss(1);
731 return b.toOwnedSlice();
732}
733pub fn main(init: std.process.Init) !void {
734 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
735 defer arena.deinit();
736 const a = arena.allocator();
737 var ib: [4096]u8 = undefined;
738 var ob: [4096]u8 = undefined;
739 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
740 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
741 var referee: Referee = .{ .h = .{ .a = a, .out = &output.interface } };
742 const count = parsedNumber(std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r")).?;
743 for (0..ix(count)) |_| {
744 const line = std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r");
745 var words: std.ArrayList(Text) = .empty;
746 var it = std.mem.splitScalar(u8, line, ' ');
747 while (it.next()) |word| try words.append(a, word);
748 try referee.command(words.items);
749 try output.interface.flush();
750 }
751 try output.interface.flush();
752}
