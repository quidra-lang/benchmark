24class Position {
25 final List<List<int>> men;
26 Position([List<List<int>>? values])
27 : men = values ?? List.generate(2, (_) => List.filled(26, 0));
28 factory Position.initial() {
29 final p = Position();
30 for (int c = 0; c < 2; c++) {
31 p.men[c][24] = 2;
32 p.men[c][13] = 5;
33 p.men[c][8] = 3;
34 p.men[c][6] = 5;
35 }
36 return p;
37 }
38 Position copy() => Position(men.map((v) => List<int>.of(v)).toList());
39 String key() => '${men[0].join(',')};${men[1].join(',')}';
40 int board(int p) => men[0][p] - men[1][25 - p];
41 int opposing(int c, int p) => men[1 - c][25 - p];
42 int pips(int c) {
43 int total = 0;
44 for (int p = 1; p <= 25; p++) {
45 total += p * men[c][p];
46 }
47 return total;
48 }
49
50 int keith(int c) {
51 final v = men[c];
52 int extra(int n) => n > 0 ? n : 0;
53 int total =
54 pips(c) + 2 * extra(v[1] - 1) + extra(v[2] - 1) + extra(v[3] - 3);
55 for (int p = 4; p <= 6; p++) {
56 if (v[p] == 0) total++;
57 }
58 return total;
59 }
60
61 int rear(int c) {
62 for (int p = 25; p > 0; p--) {
63 if (men[c][p] > 0) return p;
64 }
65 return 0;
66 }
67
68 bool contact() => rear(0) + rear(1) > 25;
69 bool home(int c) {
70 for (int p = 7; p <= 25; p++) {
71 if (men[c][p] > 0) return false;
72 }
73 return true;
74 }
75
76 List<int> starts(int c) => men[c][25] > 0
77 ? [25]
78 : [
79 for (int p = 24; p > 0; p--)
80 if (men[c][p] > 0) p,
81 ];
82 Outcome? move(int c, int source, int die) {
83 int target = source - die;
84 bool hit = false;
85 if (target >= 1) {
86 if (opposing(c, target) >= 2) return null;
87 hit = opposing(c, target) == 1;
88 } else {
89 if (!home(c)) return null;
90 if (target < 0) {
91 for (int p = source + 1; p <= 6; p++) {
92 if (men[c][p] > 0) return null;
93 }
94 }
95 target = 0;
96 }
97 final after = copy();
98 after.men[c][source]--;
99 after.men[c][target]++;
100 if (hit) {
101 after.men[1 - c][25 - target]--;
102 after.men[1 - c][25]++;
103 }
104 return Outcome(after, target, hit);
105 }

135class Play {
136 final Position position;
137 final Set<int> hits;
138 final List<Move> key;
139 late final String notation;
140 Play(Sequence s)
141 : position = s.position,
142 hits = s.hits,
143 key = List.of(s.moves) {
144 key.sort(moveOrder);
145 while (true) {
146 int first = -1, second = -1;
147 search:
148 for (int i = 0; i < key.length; i++) {
149 final middle = key[i].$2;
150 if (middle < 1 || hits.contains(middle)) continue;
151 for (int j = 0; j < key.length; j++) {
152 if (key[j].$1 == middle) {
153 first = i;
154 second = j;
155 break search;
156 }
157 }
158 }
159 if (first < 0) break;
160 final merged = (key[first].$1, key[second].$2);
161 key.removeAt(first > second ? first : second);
162 key.removeAt(first < second ? first : second);
163 key.add(merged);
164 key.sort(moveOrder);
165 }
166 final List<String> elements = [];
167 final Set<int> starred = {};
168 int i = 0;
169 while (i < key.length) {
170 final m = key[i];
171 int end = i + 1;
172 while (end < key.length && key[end] == m) {
173 end++;
174 }
175 String text = '${point(m.$1)}/${point(m.$2)}';
176 if (hits.contains(m.$2) && starred.add(m.$2)) text += '*';
177 if (end - i >= 2) text += '(${end - i})';
178 elements.add(text);
179 i = end;
180 }
181 notation = elements.isEmpty ? 'none' : elements.join(' ');
182 }
183}

185int comparePlays(Play a, Play b) {
186 for (int i = 0; i < a.key.length && i < b.key.length; i++) {
187 final c = moveOrder(a.key[i], b.key[i]);
188 if (c != 0) return c;
189 }
190 return b.key.length.compareTo(a.key.length);
191}

194 final int color;
195 final List<Sequence> sequences = [];

199 Generator(Position position, this.color, int first, int second) {
200 explore(
201 position,
202 first == second ? List.filled(4, first) : [first, second],
203 [],
204 [],
205 {},
206 );
207 maximum = sequences
208 .map((s) => s.moves.length)
209 .reduce((a, b) => a > b ? a : b);
210 final larger = first > second ? first : second;
211 final largerRequired =
212 maximum == 1 &&
213 first != second &&
214 sequences.any((s) => s.moves.length == 1 && s.used[0] == larger);
215 final Set<String> seen = {};

218 (s) =>
219 s.moves.length == maximum &&
220 (!largerRequired || s.used[0] == larger) &&
221 seen.add(s.position.key()),
222 )
223 .map(Play.new)

227 void explore(
228 Position p,
229 List<int> remaining,
230 List<Move> moves,
231 List<int> used,
232 Set<int> hits,
233 ) {
234 final key = p.key(), previous = shortest[p.key()];
235 if (previous == null || moves.length < previous)
236 shortest[key] = moves.length;
237 sequences.add(Sequence(p, moves, used, hits));
238 final Set<int> tried = {};
239 for (int i = 0; i < remaining.length; i++) {
240 final die = remaining[i];
241 if (!tried.add(die)) continue;
242 final rest = List<int>.of(remaining)..removeAt(i);
243 for (final source in p.starts(color)) {
244 final outcome = p.move(color, source, die);
245 if (outcome == null) continue;
246 explore(
247 outcome.position,
248 rest,
249 [...moves, (source, outcome.target)],
250 [...used, die],
251 {...hits, if (outcome.hit) outcome.target},
252 );
253 }
254 }
255 }

258List<int> terms(Position p, int c) {
259 final pip = p.pips(1 - c) - p.pips(c);
260 int blots = 0, home = 0, prime = 0, run = 0, anchors = 0;
261 for (int point = 1; point <= 24; point++) {
262 if (p.men[c][point] == 1) {
263 final Set<int> distances = {
264 for (int q = 0; q < point; q++)
265 if (p.opposing(c, q) > 0) point - q,
266 };
267 for (int a = 1; a <= 6; a++) {
268 for (int b = 1; b <= 6; b++) {
269 if (distances.contains(a) ||
270 distances.contains(b) ||
271 distances.contains(a + b) ||
272 a == b &&
273 (distances.contains(3 * a) || distances.contains(4 * a)))
274 blots++;
275 }
276 }
277 }
278 run = p.men[c][point] >= 2 ? run + 1 : 0;
279 if (run > prime) prime = run;
280 if (p.men[c][point] >= 2 && point <= 6) home++;
281 if (p.men[c][point] >= 2 && point >= 19) anchors++;
282 }
283 return [pip, blots, home, prime, anchors, p.men[1 - c][25] - p.men[c][25]];
284}

326 final c = s.side;
327 if (s.owner == 1 - c || s.cube == 64 || prohibited) return 'unavailable';

376List<Move>? parsePlay(List<String> elements) {
377 if (elements.length == 1 && elements[0] == 'none') return [];
378 final List<Move> moves = [];
379 for (final element in elements) {
380 final match = RegExp(r'^(.*)\(([234])\)$').firstMatch(element);
381 final count = match == null ? 1 : int.parse(match.group(2)!);
382 final path = match == null ? element : match.group(1)!;
383 final parts = path.split('/');
384 if (parts.length < 2) return null;
385 final List<int> points = [];
386 for (int i = 0; i < parts.length; i++) {
387 String part = parts[i];
388 final int point;
389 if (i == 0 && part == 'bar') {
390 point = 25;
391 } else if (i > 0 && part == 'off') {
392 point = 0;
393 } else {
394 if (i > 0 && part.endsWith('*'))
395 part = part.substring(0, part.length - 1);
396 if (!RegExp(r'^[1-9][0-9]?$').hasMatch(part)) return null;
397 point = int.parse(part);
398 if (point > 24) return null;
399 }
400 points.add(point);
401 }
402 for (int i = 1; i < points.length; i++) {
403 if (points[i - 1] <= points[i]) return null;
404 for (int j = 0; j < count; j++) {
405 moves.add((points[i - 1], points[i]));
406 }
407 }
408 }
409 return moves;
410}

412void play(int first, int second, List<String> elements) {
413 final moves = parsePlay(elements);
414 if (moves == null) {
415 emit('ILLEGAL syntax');
416 return;
417 }
418 final c = state.side,
419 original = state.position,
420 after = state.position.copy();
421 final Set<int> destinations = {};
422 for (final m in moves) {
423 after.men[c][m.$1]--;
424 after.men[c][m.$2]++;
425 if (m.$2 > 0) destinations.add(m.$2);
426 }
427 for (final p in destinations) {
428 if (original.opposing(c, p) == 1) {
429 after.men[1 - c][25 - p]--;
430 after.men[1 - c][25]++;
431 }
432 }
433 String? error;
434 if (after.men[c].any((n) => n < 0)) {
435 error = 'no checker';
436 } else if (after.men[c][25] > 0 && moves.any((m) => m.$1 != 25)) {
437 error = 'bar';
438 } else if (moves.any((m) => m.$2 > 0 && original.opposing(c, m.$2) >= 2)) {
439 error = 'blocked';
440 } else if (moves.any((m) => m.$2 == 0) && !after.home(c)) {
441 error = 'bear-off';
442 }
443 if (error != null) {
444 emit('ILLEGAL $error');
445 return;
446 }
447 final g = Generator(original, c, first, second), key = after.key();
448 for (final p in g.plays) {
449 if (p.position.key() == key) {
450 emit('OK ${p.notation}');
451 commit(p, first, second);
452 return;
453 }
454 }
455 final shortest = g.shortest[key];
456 emit(
457 'ILLEGAL ${shortest == null
458 ? 'impossible'
459 : shortest < g.maximum
460 ? 'too few dice'
461 : 'larger die required'}',
462 );
463}

499Game simulate(
500 State s,
501 List<String> strategies,
502 Dice dice, {
503 List<int>? opening,
504 int target = 0,
505 List<int> matchPoints = const [0, 0],
506 bool crawford = false,
507}) {
508 final game = Game();
509 for (int turn = 1; ; turn++) {
510 final side = s.side, other = 1 - s.side;
511 final int first, second;
512 if (turn == 1 && opening != null) {
513 first = opening[0];
514 second = opening[1];
515 } else {
516 final decision = cube(
517 s,
518 target > 0 && (crawford || matchPoints[side] + s.cube >= target),
519 );
520 if (decision.startsWith('double'))
521 game.lines.add('$turn ${colors[side]} doubles ${2 * s.cube}');
522 if (decision == 'double drop') {
523 game.lines.add('$turn ${colors[other]} drops');
524 game.finish(side, 'drop', s.cube);
525 return game;
526 }
527 if (decision == 'double take') {
528 game.lines.add('$turn ${colors[other]} takes');
529 s.cube *= 2;
530 s.owner = other;
531 }
532 first = dice.roll();
533 second = dice.roll();
534 }
535 final plays = Generator(s.position, side, first, second).plays;
536 final chosen = strategies[side] == 'BEST'
537 ? ranked(plays, side).first
538 : strategies[side] == 'FIRST'
539 ? plays.first
540 : plays[dice.next() % plays.length];
541 s.position = chosen.position;
542 game.plays++;
543 game.lines.add('$turn ${colors[side]} $first-$second ${chosen.notation}');
544 if (s.position.men[side][0] == 15) {
545 final r = result(s.position, side);
546 game.finish(side, r, multiplier(r) * s.cube);
547 return game;
548 }
549 s.side = other;
550 }

696 } else if (c == 'GAME') {

700 } else if (c == 'MATCH' || c == 'STATS') {
701 requireValue(a.length == 2);

707 if ([

709 'MOVES',
710 'PLAY',
711 'AUTO',
712 'HINT',
713 'EVAL',
714 'SHOTS',

716 'DOUBLE',

719 'RESIGN',

725 state.pending)

727 switch (c) {

730 final score = state.score;
731 state = State();
732 state.score = score;
733 history.clear();
734 redo.clear();
735 emit('OK');
736 }
737 case 'SHOW':
738 {
739 emit(positionText());
740 }
741 case 'BOARD':
742 {
743 board();
744 }
745 case 'PIPS':
746 {
747 emit('pips W ${state.position.pips(0)} B ${state.position.pips(1)}');
748 }
749 case 'RACE':
750 {
751 emit(
752 state.position.contact()
753 ? 'contact'
754 : 'race W ${state.position.keith(0)} B ${state.position.keith(1)}',
755 );
756 }
757 case 'MOVES':
758 {
759 final plays = Generator(
760 state.position,
761 state.side,
762 first,
763 second,
764 ).plays;
765 for (final p in plays) {
766 emit(p.notation);
767 }
768 emit('total ${plays.length}');
769 }
770 case 'PLAY':
771 {
772 play(first, second, a.skip(2).toList());
773 }
774 case 'AUTO':
775 case 'HINT':
776 {
777 final plays = ranked(
778 Generator(state.position, state.side, first, second).plays,
779 state.side,
780 );
781 if (c == 'AUTO') {
782 final p = plays.first;
783 emit('OK ${p.notation} eval ${evaluate(p.position, state.side)}');
784 commit(p, first, second);
785 } else {
786 for (final p in plays.take(amount)) {
787 emit('${evaluate(p.position, state.side)} ${p.notation}');
788 }
789 }
790 }
791 case 'EVAL':
792 {
793 final t = terms(state.position, state.side);
794 emit(
795 'eval ${evaluate(state.position, state.side)} pips ${t[0]} blots ${t[1]} home ${t[2]} prime ${t[3]} anchors ${t[4]} bar ${t[5]}',
796 );
797 }
798 case 'SHOTS':
799 {
800 shots();
801 }
802 case 'CUBE':
803 {
804 emit('cube ${cube(state, false)}');
805 }
806 case 'DOUBLE':
807 {
808 if (state.owner == 1 - state.side || state.cube == 64)
809 fail('cube unavailable');
810 state.pending = true;
811 redo.clear();
812 emit('OK double ${2 * state.cube}');
813 }
814 case 'TAKE':
815 {
816 if (!state.pending) fail('no double');
817 state.pending = false;
818 state.cube *= 2;
819 state.owner = 1 - state.side;
820 redo.clear();
821 emit('OK take ${state.cube} ${state.ownerName()}');
822 }
823 case 'DROP':
824 {
825 if (!state.pending) fail('no double');
826 state.pending = false;
827 redo.clear();
828 emit('OK drop');
829 end(state.side, 'drop', state.cube);
830 }
831 case 'RESIGN':
832 {
833 redo.clear();
834 emit('OK resign');
835 end(1 - state.side, a[0], multiplier(a[0]) * state.cube);
836 }
837 case 'SCORE':
838 {
839 emit('score W ${state.score[0]} B ${state.score[1]}');
840 }
841 case 'HISTORY':
842 {
843 emit('history ${history.length}');
844 for (int i = 0; i < history.length; i++) {
845 emit('${i + 1} ${history[i].record}');
846 }
847 }
848 case 'UNDO':
849 {
850 if (amount > history.length) fail('cannot undo');
851 final index = history.length - amount;
852 final removed = history.sublist(index);
853 state = removed.first.before.copy();
854 history.removeRange(index, history.length);
855 redo.insertAll(0, removed);
856 emit('OK');
857 }
858 case 'REDO':
859 {
860 if (amount > redo.length) fail('cannot redo');
861 final entries = redo.take(amount).toList();
862 redo.removeRange(0, amount);
863 history.addAll(entries);
864 state = entries.last.after.copy();
865 emit('OK');
866 }
867 case 'GAME':
868 {
869 final g = simulate(state.copy(), [a[1], a[2]], Dice(seed));
870 for (final line in g.lines) {
871 emit(line);
872 }
873 emit('result ${colors[g.winner]} ${g.result} ${g.points}');
874 }
875 case 'MATCH':
876 case 'STATS':
877 {
878 final dice = Dice(seed),
879 points = [0, 0],
880 counts = List.generate(4, (_) => [0, 0]);
881 int turns = 0, games = 0;
882 bool crawfordPlayed = false;
883 while (c == 'MATCH'
884 ? (points[0] > points[1] ? points[0] : points[1]) < amount
885 : games < amount) {
886 final maxPoints = points[0] > points[1] ? points[0] : points[1];
887 final crawford =
888 c == 'MATCH' && !crawfordPlayed && maxPoints == amount - 1;
889 int white, black;
890 do {
891 white = dice.roll();
892 black = dice.roll();
893 } while (white == black);
894 final start = State();
895 start.side = white > black ? 0 : 1;
896 final firstPlayer = start.side;
897 final g = simulate(
898 start,
899 ['BEST', 'BEST'],
900 dice,
901 opening: [white, black],
902 target: c == 'MATCH' ? amount : 0,
903 matchPoints: points,
904 crawford: crawford,
905 );
906 games++;
907 turns += g.plays;
908 points[g.winner] += g.points;
909 counts[results.indexOf(g.result)][g.winner]++;
910 if (crawford) crawfordPlayed = true;
911 if (c == 'MATCH')
912 emit(
913 'game $games${crawford ? ' crawford' : ''} start ${colors[firstPlayer]} turns ${g.plays} result ${colors[g.winner]} ${g.result} ${g.points} score ${points[0]}-${points[1]}',
914 );
915 }
916 if (c == 'MATCH') {
917 emit(
918 'match ${colors[points[0] >= amount ? 0 : 1]} ${points[0]}-${points[1]}',
919 );
920 } else {
921 for (int i = 0; i < 4; i++) {
922 emit('${results[i]} W ${counts[i][0]} B ${counts[i][1]}');
923 }
924 emit('points W ${points[0]} B ${points[1]}');
925 emit('turns $turns');
926 }
927 }
928 }
929}
930
931Future<void> main() async {
932 final n = int.parse(stdin.readLineSync()!);
933 for (int i = 0; i < n; i++) {
934 try {
935 execute(stdin.readLineSync()!.split(' '));
936 } on Failure catch (error) {
937 emit('ERROR ${error.message}');
938 }
939 stdout.write(output);
940 await stdout.flush();
941 output.clear();
942 }
943}
