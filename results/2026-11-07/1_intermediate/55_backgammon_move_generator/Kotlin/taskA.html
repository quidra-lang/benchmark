20private class Position(val men: Array<IntArray> = Array(2) { IntArray(26) }) {
21 companion object {
22 fun initial(): Position {
23 val p = Position()
24 for (c in 0..1) {
25 p.men[c][24] = 2
26 p.men[c][13] = 5
27 p.men[c][8] = 3
28 p.men[c][6] = 5
29 }
30 return p
31 }
32 }
33
34 fun duplicate() = Position(Array(2) { men[it].clone() })
35
36 fun key() = men[0].joinToString(",") + ";" + men[1].joinToString(",")
37
38 fun board(p: Int) = men[0][p] - men[1][25 - p]
39
40 fun opposing(c: Int, p: Int) = men[1 - c][25 - p]
41
42 fun pips(c: Int) = (1..25).sumOf { it * men[c][it] }
43
44 fun keith(c: Int): Int {
45 val v = men[c]
46 return pips(c) +
47 2 * maxOf(v[1] - 1, 0) +
48 maxOf(v[2] - 1, 0) +
49 maxOf(v[3] - 3, 0) +
50 (4..6).count { v[it] == 0 }
51 }
52
53 fun rear(c: Int) = (25 downTo 1).find { men[c][it] > 0 } ?: 0
54
55 fun contact() = rear(0) + rear(1) > 25
56
57 fun home(c: Int) = (7..25).all { men[c][it] == 0 }
58
59 fun starts(c: Int) = if (men[c][25] > 0) listOf(25) else (24 downTo 1).filter { men[c][it] > 0 }
60
61 fun move(c: Int, source: Int, die: Int): Outcome? {
62 var target = source - die
63 var hit = false
64 if (target >= 1) {
65 if (opposing(c, target) >= 2) return null
66 hit = opposing(c, target) == 1
67 } else {
68 if (!home(c)) return null
69 if (target < 0 && (source + 1..6).any { men[c][it] > 0 }) return null
70 target = 0
71 }
72 val after = duplicate()
73 after.men[c][source]--
74 after.men[c][target]++
75 if (hit) {
76 after.men[1 - c][25 - target]--
77 after.men[1 - c][25]++
78 }
79 return Outcome(after, target, hit)
80 }
81}

103private class Play(s: Sequence) {
104 val position = s.position
105 val hits = s.hits
106 val key = s.moves.sortedWith(moveOrder).toMutableList()
107 val notation: String
108
109 init {
110 while (true) {
111 var first = -1
112 var second = -1
113 search@ for (i in key.indices) {
114 val middle = key[i].target
115 if (middle < 1 || middle in hits) continue
116 for (j in key.indices) if (key[j].source == middle) {
117 first = i
118 second = j
119 break@search
120 }
121 }
122 if (first < 0) break
123 val merged = Move(key[first].source, key[second].target)
124 key.removeAt(maxOf(first, second))
125 key.removeAt(minOf(first, second))
126 key.add(merged)
127 key.sortWith(moveOrder)
128 }
129 val elements = mutableListOf<String>()
130 val starred = mutableSetOf<Int>()
131 var i = 0
132 while (i < key.size) {
133 val m = key[i]
134 var end = i + 1
135 while (end < key.size && key[end] == m) end++
136 var text = "${point(m.source)}/${point(m.target)}"
137 if (m.target in hits && starred.add(m.target)) text += "*"
138 if (end - i >= 2) text += "(${end - i})"
139 elements.add(text)
140 i = end
141 }
142 notation = if (elements.isEmpty()) "none" else elements.joinToString(" ")
143 }
144}

146private fun comparePlays(a: Play, b: Play): Int {
147 for (i in 0 until minOf(a.key.size, b.key.size)) {
148 val result = moveOrder.compare(a.key[i], b.key[i])
149 if (result != 0) return result
150 }
151 return b.key.size.compareTo(a.key.size)
152}

154private class Generator(position: Position, val color: Int, first: Int, second: Int) {
155 val sequences = mutableListOf<Sequence>()
156 val shortest = mutableMapOf<String, Int>()
157 val maximum: Int
158 val plays: List<Play>
159
160 init {
161 explore(
162 position,
163 if (first == second) List(4) { first } else listOf(first, second),
164 emptyList(),
165 emptyList(),
166 emptySet(),
167 )
168 maximum = sequences.maxOf { it.moves.size }
169 val largerRequired =
170 maximum == 1 &&
171 first != second &&
172 sequences.any { it.moves.size == 1 && it.used[0] == maxOf(first, second) }
173 val seen = mutableSetOf<String>()
174 plays =
175 sequences
176 .filter {
177 it.moves.size == maximum &&
178 (!largerRequired || it.used[0] == maxOf(first, second)) &&
179 seen.add(it.position.key())
180 }
181 .map { Play(it) }
182 .sortedWith(::comparePlays)
183 }
184
185 fun explore(
186 p: Position,
187 remaining: List<Int>,
188 moves: List<Move>,
189 used: List<Int>,
190 hits: Set<Int>,
191 ) {
192 val key = p.key()
193 shortest[key] = minOf(shortest[key] ?: Int.MAX_VALUE, moves.size)
194 sequences.add(Sequence(p, moves, used, hits))
195 val tried = mutableSetOf<Int>()
196 for (i in remaining.indices) {
197 val die = remaining[i]
198 if (!tried.add(die)) continue
199 val rest = remaining.toMutableList().also { it.removeAt(i) }
200 for (source in p.starts(color)) {
201 val outcome = p.move(color, source, die) ?: continue
202 explore(
203 outcome.position,
204 rest,
205 moves + Move(source, outcome.target),
206 used + die,
207 if (outcome.hit) hits + outcome.target else hits,
208 )
209 }
210 }
211 }
212}

214private fun terms(p: Position, c: Int): IntArray {
215 val pip = p.pips(1 - c) - p.pips(c)
216 var blots = 0
217 var home = 0
218 var prime = 0
219 var run = 0
220 var anchors = 0
221 for (point in 1..24) {
222 if (p.men[c][point] == 1) {
223 val distances =
224 (0 until point).filter { p.opposing(c, it) > 0 }.map { point - it }.toSet()
225 for (a in 1..6) for (b in 1..6) if (
226 a in distances ||
227 b in distances ||
228 a + b in distances ||
229 a == b && (3 * a in distances || 4 * a in distances)
230 )
231 blots++
232 }
233 run = if (p.men[c][point] >= 2) run + 1 else 0
234 prime = maxOf(prime, run)
235 if (p.men[c][point] >= 2 && point <= 6) home++
236 if (p.men[c][point] >= 2 && point >= 19) anchors++
237 }
238 return intArrayOf(pip, blots, home, prime, anchors, p.men[1 - c][25] - p.men[c][25])
239}

241private fun evaluate(p: Position, c: Int): Int {
242 val t = terms(p, c)
243 return 2 * t[0] - t[1] + 4 * t[2] + 3 * t[3] + 3 * t[4] + 6 * t[5]
244}

246private fun result(p: Position, winner: Int): String {
247 val loser = p.men[1 - winner]
248 if (loser[0] > 0) return "single"
249 if (loser[25] > 0 || (19..24).any { loser[it] > 0 }) return "backgammon"
250 return "gammon"
251}

284private fun cube(s: State, prohibited: Boolean): String {
285 val c = s.side
286 if (s.owner == 1 - c || s.cube == 64 || prohibited) return "unavailable"
287 if (s.position.contact()) {
288 val e = evaluate(s.position, c)
289 return if (e < 60) "no double" else if (e <= 100) "double take" else "double drop"
290 }
291 val v = 8 * s.position.keith(c) - 7 * s.position.keith(1 - c)
292 val limit = if (s.owner == -1) 28 else 21
293 return if (v > limit) "no double" else if (v >= 14) "double take" else "double drop"
294}

325private fun parsePlay(elements: List<String>): List<Move>? {
326 if (elements == listOf("none")) return emptyList()
327 val moves = mutableListOf<Move>()
328 for (element in elements) {
329 val match = Regex("(.*)\\(([234])\\)").matchEntire(element)
330 val count = match?.groupValues?.get(2)?.toInt() ?: 1
331 val path = match?.groupValues?.get(1) ?: element
332 val parts = path.split('/')
333 if (parts.size < 2) return null
334 val points = mutableListOf<Int>()
335 for (i in parts.indices) {
336 var part = parts[i]
337 val point: Int
338 if (i == 0 && part == "bar") point = 25
339 else if (i > 0 && part == "off") point = 0
340 else {
341 if (i > 0 && part.endsWith('*')) part = part.dropLast(1)
342 if (!part.matches(Regex("[1-9][0-9]?"))) return null
343 point = part.toInt()
344 if (point > 24) return null
345 }
346 points.add(point)
347 }
348 for (i in 1 until points.size) {
349 if (points[i - 1] <= points[i]) return null
350 repeat(count) { moves.add(Move(points[i - 1], points[i])) }
351 }
352 }
353 return moves
354}

356private fun play(first: Int, second: Int, elements: List<String>) {
357 val moves = parsePlay(elements)
358 if (moves == null) {
359 emit("ILLEGAL syntax")
360 return
361 }
362 val c = state.side
363 val original = state.position
364 val after = original.duplicate()
365 val destinations = mutableSetOf<Int>()
366 for (m in moves) {
367 after.men[c][m.source]--
368 after.men[c][m.target]++
369 if (m.target > 0) destinations.add(m.target)
370 }
371 for (p in destinations) if (original.opposing(c, p) == 1) {
372 after.men[1 - c][25 - p]--
373 after.men[1 - c][25]++
374 }
375 val error =
376 when {
377 after.men[c].any { it < 0 } -> "no checker"
378 after.men[c][25] > 0 && moves.any { it.source != 25 } -> "bar"
379 moves.any { it.target > 0 && original.opposing(c, it.target) >= 2 } -> "blocked"
380 moves.any { it.target == 0 } && !after.home(c) -> "bear-off"
381 else -> null
382 }
383 if (error != null) {
384 emit("ILLEGAL $error")
385 return
386 }
387 val g = Generator(original, c, first, second)
388 val key = after.key()
389 val found = g.plays.find { it.position.key() == key }
390 if (found != null) {
391 emit("OK ${found.notation}")
392 commit(found, first, second)
393 return
394 }
395 val shortest = g.shortest[key]
396 emit(
397 "ILLEGAL " +
398 if (shortest == null) "impossible"
399 else if (shortest < g.maximum) "too few dice" else "larger die required",
400 )
401}

403private fun ranked(plays: List<Play>, c: Int): List<Play> {
404 val scores = plays.associateWith { evaluate(it.position, c) }
405 return plays.sortedWith { a, b ->
406 val comparison = scores.getValue(b).compareTo(scores.getValue(a))
407 if (comparison == 0) comparePlays(a, b) else comparison
408 }

436private fun simulate(
437 s: State,
438 strategies: List<String>,
439 dice: Dice,
440 opening: List<Int>? = null,
441 target: Int = 0,
442 matchPoints: IntArray = IntArray(2),
443 crawford: Boolean = false,
444): Game {
445 val game = Game()
446 var turn = 1
447 while (true) {
448 val side = s.side
449 val other = 1 - side
450 val first: Int
451 val second: Int
452 if (turn == 1 && opening != null) {
453 first = opening[0]
454 second = opening[1]
455 } else {
456 val decision = cube(s, target > 0 && (crawford || matchPoints[side] + s.cube >= target))
457 if (decision.startsWith("double"))
458 game.lines.add("$turn ${colors[side]} doubles ${2 * s.cube}")
459 if (decision == "double drop") {
460 game.lines.add("$turn ${colors[other]} drops")
461 game.finish(side, "drop", s.cube)
462 return game
463 }
464 if (decision == "double take") {
465 game.lines.add("$turn ${colors[other]} takes")
466 s.cube *= 2
467 s.owner = other
468 }
469 first = dice.roll()
470 second = dice.roll()
471 }
472 val plays = Generator(s.position, side, first, second).plays
473 val chosen =
474 when (strategies[side]) {
475 "BEST" -> ranked(plays, side).first()
476 "FIRST" -> plays.first()
477 else -> plays[(dice.next() % plays.size).toInt()]
478 }
479 s.position = chosen.position
480 game.plays++
481 game.lines.add("$turn ${colors[side]} $first-$second ${chosen.notation}")
482 if (s.position.men[side][0] == 15) {
483 val r = result(s.position, side)
484 game.finish(side, r, multiplier(r) * s.cube)
485 return game
486 }
487 s.side = other
488 turn++
489 }
490}

505private fun board() {

507 boardLine(

528private fun shots() {
529 val c = state.side
530 val blots = (24 downTo 1).filter { state.position.men[c][it] == 1 }
531 val counts = blots.associateWith { 0 }.toMutableMap()
532 var total = 0
533 val rolls = mutableListOf<String>()
534 for (a in 6 downTo 1) for (b in a downTo 1) {
535 val plays = Generator(state.position, 1 - c, a, b).plays
536 val hit = blots.filter { blot -> plays.any { 25 - blot in it.hits } }
537 val weight = if (a == b) 1 else 2
538 if (hit.isNotEmpty()) {
539 total += weight
540 rolls.add("$a-$b")
541 }
542 for (p in hit) counts[p] = counts.getValue(p) + weight
543 }
544 emit("shots $total/36" + if (rolls.isEmpty()) "" else " ${rolls.joinToString(" ")}")
545 for (p in blots) emit("blot $p ${counts.getValue(p)}/36")
546}

548private fun execute(words: List<String>) {
549 val c = words[0]
550 val a = words.drop(1)
551 val known =
552 setOf(
553 "NEW",
554 "POS",
555 "SHOW",
556 "BOARD",
557 "PIPS",
558 "RACE",
559 "MOVES",
560 "PLAY",
561 "AUTO",
562 "HINT",
563 "EVAL",
564 "SHOTS",
565 "CUBE",
566 "DOUBLE",
567 "TAKE",
568 "DROP",
569 "RESIGN",
570 "SCORE",
571 "HISTORY",
572 "UNDO",
573 "REDO",
574 "GAME",
575 "MATCH",
576 "STATS",
577 )
578 if (c !in known) fail("unknown command")
579 if (c == "POS") {
580 requireValue(a.size == 4, "bad position")
581 val points = a[0].split(',')
582 requireValue(points.size == 24, "bad position")
583 val p = Position()
584 p.men[0][25] = number(a[1], 0, 15, "bad position")
585 p.men[1][25] = number(a[2], 0, 15, "bad position")
586 requireValue(a[3] in colors, "bad position")
587 for (i in points.indices) {
588 requireValue(points[i].matches(Regex("-?[0-9]+")), "bad position")
589 val v = points[i].toIntOrNull() ?: fail("bad position")
590 requireValue(v in -15..15, "bad position")
591 if (v > 0) p.men[0][i + 1] = v else p.men[1][24 - i] = -v
592 }
593 for (side in 0..1) {
594 p.men[side][0] = 15 - p.men[side].sum()
595 requireValue(p.men[side][0] in 0..14, "bad position")
596 }
597 val score = state.score
598 state = State()
599 state.position = p
600 state.side = colors.indexOf(a[3])
601 state.score = score
602 history.clear()
603 redo.clear()
604 emit("OK")
605 return
606 }
607 var first = 0
608 var second = 0
609 var amount = 0
610 var seed = 0
611 when (c) {
612 "MOVES",
613 "PLAY",
614 "AUTO",
615 "HINT" -> {
616 requireValue(if (c == "PLAY") a.size >= 3 else a.size == if (c == "HINT") 3 else 2)
617 first = number(a[0], 1, 6)
618 second = number(a[1], 1, 6)
619 if (c == "HINT") amount = number(a[2], 1, 1000)
620 }
621 "UNDO",
622 "REDO" -> {
623 requireValue(a.size == 1)
624 amount = number(a[0], 1, 1000)
625 }
626 "RESIGN" -> requireValue(a.size == 1 && a[0] in results.take(3))
627 "GAME" -> {
628 requireValue(a.size == 3)
629 seed = number(a[0], 1, 2147483646)
630 requireValue(a[1] in strategies && a[2] in strategies)
631 }
632 "MATCH",
633 "STATS" -> {
634 requireValue(a.size == 2)
635 amount = number(a[0], 1, if (c == "MATCH") 15 else 50)
636 seed = number(a[1], 1, 2147483646)
637 }
638 else -> requireValue(a.isEmpty())
639 }
640 if (
641 c in
642 setOf(
643 "RACE",
644 "MOVES",
645 "PLAY",
646 "AUTO",
647 "HINT",
648 "EVAL",
649 "SHOTS",
650 "CUBE",
651 "DOUBLE",
652 "TAKE",
653 "DROP",
654 "RESIGN",
655 "GAME",
656 ) && state.over
657 )
658 fail("game over")
659 if (c in setOf("PLAY", "AUTO", "CUBE", "DOUBLE", "RESIGN", "GAME") && state.pending)
660 fail("double pending")
661 when (c) {
662 "NEW" -> {
663 val score = state.score
664 state = State()
665 state.score = score
666 history.clear()
667 redo.clear()
668 emit("OK")
669 }
670 "SHOW" -> emit(positionText())
671 "BOARD" -> board()
672 "PIPS" -> emit("pips W ${state.position.pips(0)} B ${state.position.pips(1)}")
673 "RACE" ->
674 emit(
675 if (state.position.contact()) "contact"
676 else "race W ${state.position.keith(0)} B ${state.position.keith(1)}",
677 )
678 "MOVES" -> {
679 val plays = Generator(state.position, state.side, first, second).plays
680 plays.forEach { emit(it.notation) }
681 emit("total ${plays.size}")
682 }
683 "PLAY" -> play(first, second, a.drop(2))
684 "AUTO",
685 "HINT" -> {
686 val plays =
687 ranked(Generator(state.position, state.side, first, second).plays, state.side)
688 if (c == "AUTO") {
689 val p = plays.first()
690 emit("OK ${p.notation} eval ${evaluate(p.position, state.side)}")
691 commit(p, first, second)
692 } else
693 plays.take(amount).forEach {
694 emit("${evaluate(it.position, state.side)} ${it.notation}")
695 }
696 }
697 "EVAL" -> {
698 val t = terms(state.position, state.side)
699 emit(
700 "eval ${evaluate(state.position, state.side)} pips ${t[0]} blots ${t[1]} home ${t[2]} prime ${t[3]} anchors ${t[4]} bar ${t[5]}",
701 )
702 }
703 "SHOTS" -> shots()
704 "CUBE" -> emit("cube ${cube(state, false)}")
705 "DOUBLE" -> {
706 if (state.owner == 1 - state.side || state.cube == 64) fail("cube unavailable")
707 state.pending = true
708 redo.clear()
709 emit("OK double ${2 * state.cube}")
710 }
711 "TAKE" -> {
712 if (!state.pending) fail("no double")
713 state.pending = false
714 state.cube *= 2
715 state.owner = 1 - state.side
716 redo.clear()
717 emit("OK take ${state.cube} ${state.ownerName()}")
718 }
719 "DROP" -> {
720 if (!state.pending) fail("no double")
721 state.pending = false
722 redo.clear()
723 emit("OK drop")
724 end(state.side, "drop", state.cube)
725 }
726 "RESIGN" -> {
727 redo.clear()
728 emit("OK resign")
729 end(1 - state.side, a[0], multiplier(a[0]) * state.cube)
730 }
731 "SCORE" -> emit("score W ${state.score[0]} B ${state.score[1]}")
732 "HISTORY" -> {
733 emit("history ${history.size}")
734 history.forEachIndexed { i, e -> emit("${i + 1} ${e.record}") }
735 }
736 "UNDO" -> {
737 if (amount > history.size) fail("cannot undo")
738 val index = history.size - amount
739 val removed = history.subList(index, history.size).toList()
740 state = removed.first().before.duplicate()
741 history.subList(index, history.size).clear()
742 redo.addAll(0, removed)
743 emit("OK")
744 }
745 "REDO" -> {
746 if (amount > redo.size) fail("cannot redo")
747 val entries = redo.take(amount)
748 redo.subList(0, amount).clear()
749 history.addAll(entries)
750 state = entries.last().after.duplicate()
751 emit("OK")
752 }
753 "GAME" -> {
754 val g = simulate(state.duplicate(), listOf(a[1], a[2]), Dice(seed))
755 g.lines.forEach { emit(it) }
756 emit("result ${colors[g.winner]} ${g.result} ${g.points}")
757 }
758 "MATCH",
759 "STATS" -> {
760 val dice = Dice(seed)
761 val points = IntArray(2)
762 val counts = Array(4) { IntArray(2) }
763 var turns = 0
764 var games = 0
765 var crawfordPlayed = false
766 while (if (c == "MATCH") maxOf(points[0], points[1]) < amount else games < amount) {
767 val crawford =
768 c == "MATCH" && !crawfordPlayed && maxOf(points[0], points[1]) == amount - 1
769 var white: Int
770 var black: Int
771 do {
772 white = dice.roll()
773 black = dice.roll()
774 } while (white == black)
775 val start = State()
776 start.side = if (white > black) 0 else 1
777 val firstPlayer = start.side
778 val g =
779 simulate(
780 start,
781 listOf("BEST", "BEST"),
782 dice,
783 listOf(white, black),
784 if (c == "MATCH") amount else 0,
785 points,
786 crawford,
787 )
788 games++
789 turns += g.plays
790 points[g.winner] += g.points
791 counts[results.indexOf(g.result)][g.winner]++
792 if (crawford) crawfordPlayed = true
793 if (c == "MATCH")
794 emit(
795 "game $games" +
796 (if (crawford) " crawford" else "") +
797 " start ${colors[firstPlayer]} turns ${g.plays} result ${colors[g.winner]} ${g.result} ${g.points} score ${points[0]}-${points[1]}",
798 )
799 }
800 if (c == "MATCH")
801 emit("match ${colors[if (points[0] >= amount) 0 else 1]} ${points[0]}-${points[1]}")
802 else {
803 for (i in 0..3) emit("${results[i]} W ${counts[i][0]} B ${counts[i][1]}")
804 emit("points W ${points[0]} B ${points[1]}")
805 emit("turns $turns")
806 }
807 }
808 }
809}
