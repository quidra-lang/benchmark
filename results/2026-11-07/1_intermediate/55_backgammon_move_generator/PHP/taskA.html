51function notation(array $moves, array $hits): array
52{
53 $key = $moves;
54 usort($key, "pairCompare");
55 while (true) {
56 $ai = null;
57 $bi = null;
58 foreach ($key as $i => $a) {
59 if ($a[1] === 0 || isset($hits[$a[1]])) {
60 continue;
61 }
62 foreach ($key as $j => $b) {
63 if ($b[0] === $a[1]) {
64 $ai = $i;
65 $bi = $j;
66 break;
67 }
68 }
69 if ($ai !== null) {
70 break;
71 }
72 }
73 if ($ai === null) {
74 break;
75 }
76 $a = $key[$ai];
77 $b = $key[$bi];
78 unset($key[$ai], $key[$bi]);
79 $key = array_values($key);
80 $key[] = [$a[0], $b[1]];
81 usort($key, "pairCompare");
82 }

85 for ($i = 0; $i < count($key); $i = $j) {

101 return [$key, $parts ? implode(" ", $parts) : "none"];

212function evaluation(array $pos, int $side): array
213{
214 $own = $pos[$side];
215 $opp = $pos[1 - $side];

217 $s = 0;
218 $h = 0;
219 $r = 0;
220 $a = 0;
221 $run = 0;
222 for ($point = 1; $point <= 24; ++$point) {
223 if ($own[$point] === 1) {
224 $distances = [];
225 for ($q = 0; $q < $point; ++$q) {
226 if ($opp[25 - $q] > 0) {
227 $distances[$point - $q] = true;
228 }
229 }
230 for ($d1 = 1; $d1 <= 6; ++$d1) {
231 for ($d2 = 1; $d2 <= 6; ++$d2) {
232 if (
233 isset($distances[$d1]) ||
234 isset($distances[$d2]) ||
235 isset($distances[$d1 + $d2]) ||
236 ($d1 === $d2 && (isset($distances[3 * $d1]) || isset($distances[4 * $d1])))
237 ) {
238 ++$s;
239 }
240 };
241 }
242 }
243 if ($own[$point] >= 2) {
244 if ($point <= 6) {
245 ++$h;
246 }
247 if ($point >= 19) {
248 ++$a;
249 }
250 $r = max($r, ++$run);
251 } else {
252 $run = 0;
253 }
254 }
255 $g = $opp[25] - $own[25];
256 return [2 * $p - $s + 4 * $h + 3 * $r + 3 * $a + 6 * $g, $p, $s, $h, $r, $a, $g];

288function cubeDecision(array $state, bool $forbidden = false): string
289{
290 $x = $state["side"];
291 if ($state["owner"] === 1 - $x || $state["cube"] === 64 || $forbidden) {
292 return "unavailable";
293 }
294 if (contact($state["pos"])) {
295 $e = evaluation($state["pos"], $x)[0];
296 if ($e < 60) {
297 return "no double";
298 }
299 return $e <= 100 ? "double take" : "double drop";
300 }
301 $v = 8 * keith($state["pos"][$x]) - 7 * keith($state["pos"][1 - $x]);
302 $limit = $state["owner"] === -1 ? 28 : 21;
303 if ($v > $limit) {
304 return "no double";
305 }
306 return $v >= 14 ? "double take" : "double drop";
307}
308function endGame(array &$state, int $winner, ?string $kind = null): array
309{

323 $state["pending"] = false;

710 if ($p["id"] === $key) {
711 $play = $p;
712 break;
713 }
714 }
715 if ($play === null) {
716 $possible = false;
717 $fewer = false;
718 foreach ($all as $p) {
719 if ($p["id"] === $key) {
720 $possible = true;
721 if ($p["count"] < $maximum) {
722 $fewer = true;
723 }
724 }
725 }
726 ensure($possible, "ILLEGAL impossible");
727 throw new RuntimeException(
728 $fewer ? "ILLEGAL too few dice" : "ILLEGAL larger die required",
729 );
730 }
731 }
732 $before = $state;
733 $result = applyPlay($state, $play);
734 out("OK " . $play["text"] . ($cmd === "AUTO" ? " eval $value" : ""));
735 if ($result !== null) {
736 out(resultLine($result));
737 }
738 $history[] = [color($side) . " $d1-$d2 {$play["text"]}", $before, $state];
739 $redo = [];
740 }
741 break;
742 case "SHOTS":
743 $blots = [];
744 $counts = [];
745 for ($p = 24; $p >= 1; --$p) {
746 if ($state["pos"][$side][$p] === 1) {
747 $blots[] = $p;
748 $counts[$p] = 0;
749 }
750 }
751 $total = 0;
752 $rolls = [];
753 for ($d1 = 6; $d1 >= 1; --$d1) {
754 for ($d2 = $d1; $d2 >= 1; --$d2) {
755 $legal = enumerate($state["pos"], 1 - $side, $d1, $d2)[0];
756 $hit = [];
757 foreach ($blots as $p) {
758 foreach ($legal as $play) {
759 if ($play["pos"][$side][$p] === 0) {
760 $hit[] = $p;
761 break;
762 }
763 };
764 }
765 if ($hit) {
766 $weight = $d1 === $d2 ? 1 : 2;
767 $total += $weight;
768 $rolls[] = "$d1-$d2";
769 foreach ($hit as $p) {
770 $counts[$p] += $weight;
771 }
772 }
773 };
774 }
775 out("shots $total/36" . ($rolls ? " " . implode(" ", $rolls) : ""));
776 foreach ($blots as $p) {
777 out("blot $p {$counts[$p]}/36");
778 }
779 break;
780 case "CUBE":
781 out("cube " . cubeDecision($state));
782 break;
783 case "DOUBLE":
784 ensure(
785 $state["owner"] !== 1 - $side && $state["cube"] !== 64,
786 "ERROR cube unavailable",
787 );
788 $state["pending"] = true;
789 $redo = [];
790 out("OK double " . $state["cube"] * 2);
791 break;
792 case "TAKE":
793 case "DROP":
794 ensure($state["pending"], "ERROR no double");
795 $state["pending"] = false;
796 $redo = [];
797 if ($cmd === "TAKE") {
798 $state["cube"] *= 2;
799 $state["owner"] = 1 - $side;
800 out("OK take {$state["cube"]} " . color($state["owner"]));
801 } else {
802 out("OK drop");
803 out(resultLine(endGame($state, $side, "drop")));
804 }
805 break;
806 case "RESIGN":
807 $redo = [];
808 out("OK resign");
809 out(resultLine(endGame($state, 1 - $side, $a[0])));
810 break;
811 case "SCORE":
812 out("score W {$state["score"][0]} B {$state["score"][1]}");
813 break;
814 case "HISTORY":
815 out("history " . count($history));
816 foreach ($history as $i => $e) {
817 out($i + 1 . " " . $e[0]);
818 }
819 break;
820 case "UNDO":
821 case "REDO":
822 $count = (int) $a[0];
823 if ($cmd === "UNDO") {
824 ensure($count <= count($history), "ERROR cannot undo");
825 $entries = array_splice($history, count($history) - $count);
826 $state = $entries[0][1];
827 $redo = array_merge($entries, $redo);
828 } else {
829 ensure($count <= count($redo), "ERROR cannot redo");
830 $entries = array_splice($redo, 0, $count);
831 $state = $entries[count($entries) - 1][2];
832 $history = array_merge($history, $entries);
833 }
834 out("OK");
835 break;
836 case "GAME":
837 [$result] = simulate(
838 $state,
839 new RandomStream((int) $a[0]),
840 [$a[1], $a[2]],
841 false,
842 null,
843 null,
844 false,
845 true,
846 );
847 out(resultLine($result));
848 break;
849 case "MATCH":
850 case "STATS":
851 $count = (int) $a[0];
852 $rng = new RandomStream((int) $a[1]);
853 $points = [0, 0];
854 $turns = 0;
855 $game = 0;
856 $crawfordUsed = false;
857 $outcomes = [
858 "single" => [0, 0],
859 "gammon" => [0, 0],
860 "backgammon" => [0, 0],
861 "drop" => [0, 0],
862 ];
863 while ($cmd === "STATS" ? $game < $count : max($points) < $count) {
864 ++$game;
865 $crawford = $cmd === "MATCH" && !$crawfordUsed && max($points) === $count - 1;
866 $crawfordUsed = $crawfordUsed || $crawford;
867 [$result, $played, $first] = simulate(
868 initial(),
869 $rng,
870 ["BEST", "BEST"],
871 true,
872 $cmd === "MATCH" ? $count : null,
873 $points,
874 $crawford,
875 );
876 [$winner, $kind, $value] = $result;
877 $points[$winner] += $value;
878 ++$outcomes[$kind][$winner];
879 $turns += $played;
880 if ($cmd === "MATCH") {
881 out(
882 "game $game" .
883 ($crawford ? " crawford" : "") .
884 " start " .
885 color($first) .
886 " turns $played " .
887 resultLine($result) .
888 " score {$points[0]}-{$points[1]}",
889 );
890 }
891 }
892 if ($cmd === "MATCH") {
893 out(
894 "match " .
895 color($points[0] >= $count ? 0 : 1) .
896 " {$points[0]}-{$points[1]}",
897 );
898 } else {
899 foreach ($outcomes as $kind => $values) {
900 out("$kind W {$values[0]} B {$values[1]}");
901 }
902 out("points W {$points[0]} B {$points[1]}");
903 out("turns $turns");
904 }
905 break;
906 }
907 } catch (RuntimeException $e) {
908 out($e->getMessage());
909 }
