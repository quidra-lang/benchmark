675 selfPlay(state, random, a.slice(1), false, true, () => false);
676 break;
677 }
678 const scores = { W: 0, B: 0 },
679 counts = new Map();
680 for (const kind of ["single", "gammon", "backgammon", "drop"])
681 counts.set(kind, { W: 0, B: 0 });
682 let game = 0,
683 turns = 0,
684 crawfordUsed = false;
685 while (cmd === "MATCH" ? scores.W < n[0] && scores.B < n[0] : game < n[0]) {
686 game++;
687 const crawford =
688 cmd === "MATCH" && !crawfordUsed && Math.max(scores.W, scores.B) === n[0] - 1;
689 if (crawford) crawfordUsed = true;
690 const played = selfPlay(
691 initial(),
692 random,
693 ["BEST", "BEST"],
694 true,
695 false,
696 (s) => cmd === "MATCH" && (crawford || scores[s.side] + s.cube >= n[0]),
697 );
698 const r = played.result;
699 scores[r.winner] += r.points;
700 counts.get(r.kind)[r.winner]++;
701 turns += played.turns;
702 if (cmd === "MATCH")
703 emit(
704 `game ${game}${crawford ? " crawford" : ""} start ${played.first} turns ${played.turns} ${resultLine(r)} score ${scores.W}-${scores.B}`,
705 );
706 }
707 if (cmd === "MATCH")
708 emit(`match ${scores.W >= n[0] ? "W" : "B"} ${scores.W}-${scores.B}`);
709 else {
710 for (const [kind, c] of counts) emit(`${kind} W ${c.W} B ${c.B}`);
711 emit(`points W ${scores.W} B ${scores.B}`);
712 emit("turns " + turns);
713 }
714 break;
715 }
716 }
717}
