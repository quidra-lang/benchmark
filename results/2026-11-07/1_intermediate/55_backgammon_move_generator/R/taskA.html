25contact <- function(pos) {
26 white <- if (pos[1, 26] > 0) 25 else 0
27 black <- if (pos[2, 26] > 0) 0 else 25
28 for (p in 1:24) {
29 if (pos[1, p + 1L] > 0) white <- max(white, p)
30 if (pos[2, p + 1L] > 0) black <- min(black, 25 - p)
31 }
32 white > black
33}
34evaluate <- function(pos, side) {
35 other <- 3L - side
36 P <- pips(pos, other) - pips(pos, side)
37 G <- pos[other, 26] - pos[side, 26]
38 S <- H <- R <- A <- run <- 0
39 for (p in 1:24) {
40 if (pos[side, p + 1L] >= 2) {
41 run <- run + 1L
42 R <- max(R, run)
43 H <- H + (p <= 6)
44 A <- A + (p >= 19)
45 } else {
46 run <- 0L
47 }
48 if (pos[side, p + 1L] == 1) {
49 q <- 0:(p - 1L)
50 distances <- p - q[pos[other, 26L - q] > 0]
51 for (a in 1:6) {
52 for (b in 1:6) {
53 hit <- a %in% distances || b %in% distances || (a + b) %in% distances || (a == b && ((3 * a) %in% distances || (4 * a) %in% distances))
54 S <- S + hit
55 }
56 }
57 }
58 }
59 list(E = 2 * P - S + 4 * H + 3 * R + 3 * A + 6 * G, P = P, S = S, H = H, R = R, A = A, G = G)
60}
61cube_decision <- function(s, prohibited = FALSE) {
62 if (s$owner == 3L - s$side || s$cube == 64 || prohibited) return("unavailable")
63 if (contact(s$position)) {
64 E <- evaluate(s$position, s$side)$E
65 if (E < 60) return("no double")
66 return(if (E <= 100) "double take" else "double drop")
67 }
68 V <- 8 * keith(s$position, s$side) - 7 * keith(s$position, 3L - s$side)
69 L <- if (!s$owner) 28 else 21
70 if (V > L) return("no double")
71 if (V >= 14) "double take" else "double drop"
72}
73submoves <- function(pos, side, die) {
74 sources <- if (pos[side, 26] > 0) 25L else (24:1)[pos[side, 25:2] > 0]
75 home <- pos[side, 26] == 0 && all(pos[side, 8:25] == 0)
76 moves <- list()
77 for (from in sources) {
78 to <- from - die
79 if (to >= 1) {
80 if (pos[3L - side, 26L - to] < 2) moves[[length(moves) + 1L]] <- c(from, to)
81 } else if (home) {
82 allowed <- TRUE
83 if (to < 0 && from < 6) allowed <- all(pos[side, seq.int(from + 2L, 7L)] == 0)
84 if (allowed) moves[[length(moves) + 1L]] <- c(from, 0)
85 }
86 }
87 moves

105canonical <- function(moves, hits) {
106 key <- sort_pairs(moves)
107 repeat {
108 first <- second <- 0L
109 for (i in seq_along(key)) {
110 to <- key[[i]][2]
111 if (!to || to %in% hits) next
112 matches <- which(vapply(key, function(b) b[1] == to, logical(1)))
113 if (length(matches)) {
114 first <- i
115 second <- matches[1L]
116 break
117 }
118 }
119 if (!first) break
120 combined <- c(key[[first]][1], key[[second]][2])
121 key <- key[-c(first, second)]
122 key[[length(key) + 1L]] <- combined
123 key <- sort_pairs(key)
124 }
125 parts <- character()
126 marked <- numeric()
127 i <- 1L
128 while (i <= length(key)) {
129 pair <- key[[i]]
130 j <- i + 1L
131 while (j <= length(key) && all(key[[j]] == pair)) j <- j + 1L
132 from <- if (pair[1] == 25) "bar" else as.character(pair[1])
133 to <- if (!pair[2]) "off" else as.character(pair[2])
134 part <- paste0(from, "/", to)
135 if (pair[2] %in% hits && !(pair[2] %in% marked)) {
136 part <- paste0(part, "*")
137 marked <- c(marked, pair[2])
138 }
139 if (j - i >= 2L) part <- paste0(part, "(", j - i, ")")
140 parts <- c(parts, part)
141 i <- j
142 }
143 list(key = key, notation = if (length(parts)) paste(parts, collapse = " ") else "none")
144}
145key_less <- function(a, b) {
146 for (i in seq_len(min(length(a), length(b)))) {
147 if (a[[i]][1] != b[[i]][1]) return(a[[i]][1] > b[[i]][1])
148 if (a[[i]][2] != b[[i]][2]) return(a[[i]][2] > b[[i]][2])
149 }
150 length(a) > length(b)
151}
152sort_plays <- function(plays) {
153 if (length(plays) < 2L) return(plays)
154 merge <- function(a, b) {
155 result <- list()
156 i <- j <- 1L
157 while (i <= length(a) && j <= length(b)) {
158 if (key_less(b[[j]]$key, a[[i]]$key)) {
159 result[[length(result) + 1L]] <- b[[j]]

161 } else {
162 result[[length(result) + 1L]] <- a[[i]]
163 i <- i + 1L
164 }
165 }
166 if (i <= length(a)) result <- c(result, a[seq.int(i, length(a))])
167 if (j <= length(b)) result <- c(result, b[seq.int(j, length(b))])

173enumerate_plays <- function(pos, side, d1, d2) {
174 dice <- if (d1 == d2) rep(d1, 4L) else c(d1, d2)
175 possible <- list()
176 maximum <- 0L
177 search <- function(current, left, path, hits, first) {
178 possible[[length(possible) + 1L]] <<- list(position = current, moves = path, hits = hits, die = first)
179 maximum <<- max(maximum, length(path))
180 seen <- numeric()
181 for (i in seq_along(left)) {
182 die <- left[i]
183 if (die %in% seen) next
184 seen <- c(seen, die)
185 rest <- left[-i]
186 for (move in submoves(current, side, die)) {
187 next_state <- apply_submove(current, side, move)
188 next_hits <- if (next_state$hit) union(hits, move[2]) else hits
189 search(next_state$position, rest, c(path, list(move)), next_hits, if (length(path)) first else die)
190 }
191 }
192 }
193 search(pos, dice, list(), numeric(), 0)
194 larger <- maximum == 1L && d1 != d2 && any(vapply(possible, function(p) length(p$moves) == 1L && p$die == max(d1, d2), logical(1)))
195 legal <- list()
196 positions <- new.env(hash = TRUE, parent = emptyenv())
197 for (p in possible) {
198 if (length(p$moves) != maximum || (larger && p$die != max(d1, d2))) next
199 key <- paste(as.vector(p$position), collapse = ",")
200 if (exists(key, envir = positions, inherits = FALSE)) next
201 normalized <- canonical(p$moves, p$hits)
202 p$key <- normalized$key
203 p$notation <- normalized$notation
204 legal[[length(legal) + 1L]] <- p
205 positions[[key]] <- TRUE
206 }
207 list(legal = sort_plays(legal), possible = possible, maximum = maximum)

261parse_play <- function(words) {
262 if (identical(words, "none")) return(list())
263 moves <- list()
264 pattern <- "^(bar|[1-9]|1[0-9]|2[0-4])(?:/(?:[1-9]|1[0-9]|2[0-4])\\*?)*(?:/off|/(?:[1-9]|1[0-9]|2[0-4])\\*?)(?:\\([234]\\))?$"
265 for (word in words) {
266 if (!grepl(pattern, word, perl = TRUE)) return(NULL)
267 count <- 1L
268 if (endsWith(word, ")")) {
269 count <- as.integer(substring(word, nchar(word) - 1L, nchar(word) - 1L))
270 word <- substring(word, 1L, nchar(word) - 3L)
271 }
272 parts <- strsplit(word, "/", fixed = TRUE)[[1]]
273 points <- vapply(parts, function(part) {
274 if (part == "bar") 25 else if (part == "off") 0 else as.numeric(sub("\\*$", "", part))
275 }, numeric(1))
276 for (i in seq_len(length(points) - 1L)) {
277 if (points[i] <= points[i + 1L]) return(NULL)
278 for (j in seq_len(count)) moves[[length(moves) + 1L]] <- points[c(i, i + 1L)]
279 }
280 }
281 moves
282}
283validate_play <- function(pos, side, moves, generated) {
284 final <- pos
285 destinations <- numeric()
286 nonbar <- bear <- FALSE
287 for (move in moves) {
288 from <- move[1]
289 to <- move[2]
290 final[side, from + 1L] <- final[side, from + 1L] - 1L
291 final[side, to + 1L] <- final[side, to + 1L] + 1L
292 destinations <- union(destinations, to)
293 nonbar <- nonbar || from != 25
294 bear <- bear || to == 0
295 }
296 for (to in destinations) {
297 if (to > 0 && pos[3L - side, 26L - to] == 1) {
298 final[3L - side, 26L - to] <- 0
299 final[3L - side, 26] <- final[3L - side, 26] + 1L
300 }
301 }
302 reject <- function(reason) list(play = NULL, reason = reason)
303 if (any(final[side, 2:26] < 0)) return(reject("no checker"))
304 if (final[side, 26] > 0 && nonbar) return(reject("bar"))
305 if (any(vapply(destinations, function(to) to > 0 && pos[3L - side, 26L - to] >= 2, logical(1)))) return(reject("blocked"))
306 if (bear && (final[side, 26] > 0 || any(final[side, 8:25] > 0))) return(reject("bear-off"))
307 for (p in generated$legal) {
308 if (all(p$position == final)) return(list(play = p, reason = ""))
309 }
310 matched <- Filter(function(p) all(p$position == final), generated$possible)
311 if (any(vapply(matched, function(p) length(p$moves) < generated$maximum, logical(1)))) return(reject("too few dice"))
312 if (length(matched)) return(reject("larger die required"))
313 reject("impossible")

315simulation <- function(s, rng, strategies, opening = FALSE, target = 0, score = c(0, 0), crawford = FALSE, verbose = FALSE) {
316 first_dice <- c(0, 0)
317 if (opening) {
318 while (first_dice[1] == first_dice[2]) first_dice <- c(die(rng), die(rng))
319 s$side <- if (first_dice[1] > first_dice[2]) 1L else 2L
320 }
321 start <- s$side
322 turns <- 0L
323 t <- 1L
324 repeat {
325 side <- s$side
326 other <- 3L - side
327 if (opening && t == 1L) {
328 d1 <- first_dice[1]
329 d2 <- first_dice[2]
330 } else {
331 prohibited <- target > 0 && (crawford || score[side] + s$cube >= target)
332 decision <- cube_decision(s, prohibited)
333 if (decision %in% c("double take", "double drop")) {
334 if (verbose) line(t, " ", color(side), " doubles ", s$cube * 2)
335 if (decision == "double drop") {
336 if (verbose) line(t, " ", color(other), " drops")
337 r <- finish(s, side, "drop")
338 r$turns <- turns
339 r$start <- start
340 return(r)
341 }
342 s$cube <- s$cube * 2
343 s$owner <- other
344 if (verbose) line(t, " ", color(other), " takes")
345 }
346 d1 <- die(rng)
347 d2 <- die(rng)
348 }
349 plays <- enumerate_plays(s$position, side, d1, d2)$legal
350 chosen <- plays[[1L]]
351 if (strategies[side] == "BEST") chosen <- best(plays, side)$play
352 if (strategies[side] == "RANDOM") chosen <- plays[[random(rng) %% length(plays) + 1L]]
353 s$position <- chosen$position
354 turns <- turns + 1L
355 if (verbose) line(t, " ", color(side), " ", d1, "-", d2, " ", chosen$notation)
356 if (s$position[side, 1] == 15) {
357 r <- bear_result(s, side)
358 r$turns <- turns
359 r$start <- start
360 return(r)
361 }
362 s$side <- other
363 t <- t + 1L
364 }

531 } else {
532 engine$state$pending <- TRUE

540 s$pending <- FALSE

542 s$owner <- 3L - s$side

547 } else if (cmd == "DROP") {
548 if (!s$pending) {

552 engine$state <- result$state

556 }
557 } else if (cmd == "RESIGN") {
558 result <- finish(s, 3L - s$side, a[1])
559 engine$state <- result$state
560 engine$redo <- list()
561 line("OK resign")

563 } else if (cmd == "SCORE") {
564 line("score W ", s$score[1], " B ", s$score[2])
565 } else if (cmd == "HISTORY") {

568 } else if (cmd == "UNDO") {

571 line("ERROR cannot undo")

574 engine$state <- engine$history[[index]]$before
575 engine$redo <- c(engine$history[seq.int(index, length(engine$history))], engine$redo)
576 engine$history <- engine$history[seq_len(index - 1L)]
577 line("OK")
578 }
579 } else if (cmd == "REDO") {
580 n <- as.integer(a[1])
581 if (n > length(engine$redo)) {
582 line("ERROR cannot redo")
583 } else {
584 engine$state <- engine$redo[[n]]$after
585 engine$history <- c(engine$history, engine$redo[seq_len(n)])
586 engine$redo <- engine$redo[-seq_len(n)]
587 line("OK")
588 }
589 } else if (cmd == "GAME") {
590 result <- simulation(s, random_source(as.numeric(a[1])), a[2:3], verbose = TRUE)
591 line(result_line(result))
592 } else if (cmd == "MATCH") {

594 rng <- random_source(as.numeric(a[2]))
595 score <- c(0, 0)
596 used <- FALSE
597 game <- 0L

599 game <- game + 1L
600 crawford <- !used && max(score) == target - 1L
601 used <- used || crawford
602 r <- simulation(initial(), rng, c("BEST", "BEST"), opening = TRUE, target = target, score = score, crawford = crawford)
603 score[r$winner] <- score[r$winner] + r$points
604 label <- if (crawford) " crawford" else ""
605 line("game ", game, label, " start ", color(r$start), " turns ", r$turns, " result ", color(r$winner), " ", r$kind, " ", r$points, " score ", score[1], "-", score[2])
606 }
607 winner <- if (score[1] >= target) 1L else 2L
608 line("match ", color(winner), " ", score[1], "-", score[2])
609 } else if (cmd == "STATS") {
610 rng <- random_source(as.numeric(a[2]))
611 kinds <- c("single", "gammon", "backgammon", "drop")
612 counts <- setNames(rep(list(c(0, 0)), 4L), kinds)
613 points <- c(0, 0)
614 turns <- 0
615 for (i in seq_len(as.integer(a[1]))) {
616 r <- simulation(initial(), rng, c("BEST", "BEST"), opening = TRUE)
617 counts[[r$kind]][r$winner] <- counts[[r$kind]][r$winner] + 1L
618 points[r$winner] <- points[r$winner] + r$points
619 turns <- turns + r$turns
620 }
621 for (kind in kinds) line(kind, " W ", counts[[kind]][1], " B ", counts[[kind]][2])
622 line("points W ", points[1], " B ", points[2])
623 line("turns ", turns)
624 }
625}
