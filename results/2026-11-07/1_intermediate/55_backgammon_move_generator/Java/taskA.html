129 static class Play {
130 Position position;
131 List<Move> key;
132 Set<Integer> hits;
133 String notation;
134 Play(Sequence s) {
135 position = s.position;
136 hits = s.hits;
137 key = new ArrayList<>(s.moves);
138 key.sort(MOVE_ORDER);
139 while (true) {
140 int a = -1, b = -1;
141 outer:
142 for (int i = 0; i < key.size(); i++) {
143 int middle = key.get(i).target;
144 if (middle < 1 || hits.contains(middle))
145 continue;
146 for (int j = 0; j < key.size(); j++)
147 if (key.get(j).source == middle) {
148 a = i;
149 b = j;
150 break outer;
151 }
152 }
153 if (a < 0)
154 break;
155 Move merged = new Move(key.get(a).source, key.get(b).target);
156 key.remove(Math.max(a, b));
157 key.remove(Math.min(a, b));
158 key.add(merged);
159 key.sort(MOVE_ORDER);
160 }
161 List<String> elements = new ArrayList<>();
162 Set<Integer> starred = new HashSet<>();
163 for (int i = 0; i < key.size();) {
164 Move m = key.get(i);
165 int end = i + 1;
166 while (end < key.size() && key.get(end).equals(m))
167 end++;
168 String text = point(m.source) + "/" + point(m.target);
169 if (hits.contains(m.target) && starred.add(m.target))
170 text += "*";
171 if (end - i >= 2)
172 text += "(" + (end - i) + ")";
173 elements.add(text);
174 i = end;
175 }
176 notation = elements.isEmpty() ? "none" : String.join(" ", elements);
177 }
178 }
179 static int comparePlays(Play a, Play b) {
180 for (int i = 0; i < Math.min(a.key.size(), b.key.size()); i++) {
181 int c = MOVE_ORDER.compare(a.key.get(i), b.key.get(i));
182 if (c != 0)
183 return c;
184 }
185 return Integer.compare(b.key.size(), a.key.size());
186 }

188 static class Generator {
189 int color, maximum;
190 List<Sequence> sequences = new ArrayList<>();
191 Map<String, Integer> shortest = new HashMap<>();
192 List<Play> plays = new ArrayList<>();
193 Generator(Position p, int color, int first, int second) {
194 this.color = color;
195 explore(p,
196 first == second ? List.of(first, first, first, first) : List.of(first, second),
197 List.of(), List.of(), Set.of());
198 for (Sequence s : sequences)
199 maximum = Math.max(maximum, s.moves.size());
200 boolean largerRequired =
201 maximum == 1 && first != second &&
202 sequences.stream().anyMatch(
203 s -> s.moves.size() == 1 && s.used.get(0) == Math.max(first, second));
204 Set<String> seen = new HashSet<>();
205 for (Sequence s : sequences)
206 if (s.moves.size() == maximum &&
207 (!largerRequired || s.used.get(0) == Math.max(first, second)) &&
208 seen.add(s.position.key()))
209 plays.add(new Play(s));
210 plays.sort(Main::comparePlays);
211 }
212 void explore(Position p, List<Integer> remaining, List<Move> moves, List<Integer> used,
213 Set<Integer> hits) {
214 shortest.merge(p.key(), moves.size(), Math::min);
215 sequences.add(new Sequence(p, moves, used, hits));
216 Set<Integer> tried = new HashSet<>();
217 for (int i = 0; i < remaining.size(); i++) {
218 int die = remaining.get(i);
219 if (!tried.add(die))
220 continue;
221 List<Integer> rest = new ArrayList<>(remaining);
222 rest.remove(i);
223 for (int source : p.starts(color)) {
224 Outcome outcome = p.move(color, source, die);
225 if (outcome == null)
226 continue;
227 List<Move> nextMoves = new ArrayList<>(moves);
228 nextMoves.add(new Move(source, outcome.target));
229 List<Integer> nextUsed = new ArrayList<>(used);
230 nextUsed.add(die);
231 Set<Integer> nextHits = new HashSet<>(hits);
232 if (outcome.hit)
233 nextHits.add(outcome.target);
234 explore(outcome.position, rest, nextMoves, nextUsed, nextHits);
235 }
236 }

239 static int[] terms(Position p, int c) {
240 int pip = p.pips(1 - c) - p.pips(c), blots = 0, home = 0, prime = 0, run = 0, anchors = 0;
241 for (int point = 1; point <= 24; point++) {
242 if (p.men[c][point] == 1) {
243 Set<Integer> distances = new HashSet<>();
244 for (int q = 0; q < point; q++)
245 if (p.opposing(c, q) > 0)
246 distances.add(point - q);
247 for (int a = 1; a <= 6; a++)
248 for (int b = 1; b <= 6; b++)
249 if (distances.contains(a) || distances.contains(b) ||
250 distances.contains(a + b) ||
251 a == b && (distances.contains(3 * a) || distances.contains(4 * a)))
252 blots++;
253 }
254 run = p.men[c][point] >= 2 ? run + 1 : 0;
255 prime = Math.max(prime, run);
256 if (p.men[c][point] >= 2 && point <= 6)
257 home++;
258 if (p.men[c][point] >= 2 && point >= 19)
259 anchors++;
260 }
261 return new int[] {pip, blots, home, prime, anchors, p.men[1 - c][25] - p.men[c][25]};
262 }
263 static int evaluate(Position p, int c) {
264 int[] t = terms(p, c);
265 return 2 * t[0] - t[1] + 4 * t[2] + 3 * t[3] + 3 * t[4] + 6 * t[5];
266 }

342 static List<Move> parsePlay(List<String> elements) {
343 if (elements.equals(List.of("none")))
344 return List.of();
345 List<Move> moves = new ArrayList<>();
346 for (String element : elements) {
347 Matcher match = Pattern.compile("(.*)\\(([234])\\)").matcher(element);
348 int count = 1;
349 String path = element;
350 if (match.matches()) {
351 path = match.group(1);
352 count = Integer.parseInt(match.group(2));
353 }
354 String[] parts = path.split("/", -1);
355 if (parts.length < 2)
356 return null;
357 List<Integer> points = new ArrayList<>();
358 for (int i = 0; i < parts.length; i++) {
359 String part = parts[i];
360 int point;
361 if (i == 0 && part.equals("bar"))
362 point = 25;
363 else if (i > 0 && part.equals("off"))
364 point = 0;
365 else {
366 if (i > 0 && part.endsWith("*"))
367 part = part.substring(0, part.length() - 1);
368 if (!part.matches("[1-9][0-9]?"))
369 return null;
370 point = Integer.parseInt(part);
371 if (point > 24)
372 return null;
373 }
374 points.add(point);
375 }
376 for (int i = 1; i < points.size(); i++) {
377 if (points.get(i - 1) <= points.get(i))
378 return null;
379 for (int j = 0; j < count; j++)
380 moves.add(new Move(points.get(i - 1), points.get(i)));
381 }
382 }
383 return moves;
384 }
385 static void play(int first, int second, List<String> elements) {
386 List<Move> moves = parsePlay(elements);
387 if (moves == null) {
388 emit("ILLEGAL syntax");
389 return;
390 }
391 int c = state.side;
392 Position original = state.position, after = original.copy();
393 Set<Integer> destinations = new HashSet<>();
394 for (Move m : moves) {
395 after.men[c][m.source]--;
396 after.men[c][m.target]++;
397 if (m.target > 0)
398 destinations.add(m.target);
399 }
400 for (int p : destinations)
401 if (original.opposing(c, p) == 1) {
402 after.men[1 - c][25 - p]--;
403 after.men[1 - c][25]++;
404 }
405 String error = null;
406 if (Arrays.stream(after.men[c]).anyMatch(n -> n < 0))
407 error = "no checker";
408 else if (after.men[c][25] > 0 && moves.stream().anyMatch(m -> m.source != 25))
409 error = "bar";
410 else if (moves.stream().anyMatch(m -> m.target > 0 && original.opposing(c, m.target) >= 2))
411 error = "blocked";
412 else if (moves.stream().anyMatch(m -> m.target == 0) && !after.home(c))
413 error = "bear-off";
414 if (error != null) {
415 emit("ILLEGAL " + error);
416 return;
417 }
418 Generator g = new Generator(original, c, first, second);
419 String key = after.key();
420 for (Play p : g.plays)
421 if (p.position.key().equals(key)) {
422 emit("OK " + p.notation);
423 commit(p, first, second);
424 return;
425 }
426 Integer shortest = g.shortest.get(key);
427 emit("ILLEGAL " + (shortest == null ? "impossible"
428 : shortest < g.maximum ? "too few dice"
429 : "larger die required"));
430 }

465 static Game simulate(State s, String[] strategies, Dice dice, int[] opening, int target,
466 int[] matchPoints, boolean crawford) {
467 Game game = new Game();
468 for (int turn = 1;; turn++) {
469 int side = s.side, other = 1 - side, first, second;
470 if (turn == 1 && opening != null) {
471 first = opening[0];
472 second = opening[1];
473 } else {
474 String decision =
475 cube(s, target > 0 && (crawford || matchPoints[side] + s.cube >= target));
476 if (decision.startsWith("double"))
477 game.lines.add(turn + " " + COLORS[side] + " doubles " + 2 * s.cube);
478 if (decision.equals("double drop")) {
479 game.lines.add(turn + " " + COLORS[other] + " drops");
480 game.finish(side, "drop", s.cube);
481 return game;
482 }
483 if (decision.equals("double take")) {
484 game.lines.add(turn + " " + COLORS[other] + " takes");
485 s.cube *= 2;
486 s.owner = other;
487 }
488 first = dice.roll();
489 second = dice.roll();
490 }
491 List<Play> plays = new Generator(s.position, side, first, second).plays;
492 String strategy = strategies[side];
493 Play chosen = strategy.equals("BEST") ? ranked(plays, side).get(0)
494 : strategy.equals("FIRST") ? plays.get(0)
495 : plays.get((int)(dice.next() % plays.size()));
496 s.position = chosen.position;
497 game.plays++;
498 game.lines.add(turn + " " + COLORS[side] + " " + first + "-" + second + " " +
499 chosen.notation);
500 if (s.position.men[side][0] == 15) {
501 String r = result(s.position, side);
502 game.finish(side, r, multiplier(r) * s.cube);
503 return game;
504 }
505 s.side = other;
506 }

609 }

613 }

617 state.side = a.get(3).equals("W") ? 0 : 1;
618 state.score = score;
619 history.clear();
620 redo.clear();

622 return;
623 }
624 int first = 0, second = 0, amount = 0, seed = 0;
625 if (Set.of("MOVES", "PLAY", "AUTO", "HINT").contains(c)) {
626 require(c.equals("PLAY") ? a.size() >= 3 : a.size() == (c.equals("HINT") ? 3 : 2),

629 second = number(a.get(1), 1, 6, "bad arguments");
630 if (c.equals("HINT"))

632 } else if (c.equals("UNDO") || c.equals("REDO")) {
633 require(a.size() == 1, "bad arguments");
634 amount = number(a.get(0), 1, 1000, "bad arguments");
635 } else if (c.equals("RESIGN"))
636 require(a.size() == 1 && RESULTS.subList(0, 3).contains(a.get(0)), "bad arguments");
637 else if (c.equals("GAME")) {
638 require(a.size() == 3, "bad arguments");

640 require(STRATEGIES.contains(a.get(1)) && STRATEGIES.contains(a.get(2)),

642 } else if (c.equals("MATCH") || c.equals("STATS")) {
643 require(a.size() == 2, "bad arguments");
644 amount = number(a.get(0), 1, c.equals("MATCH") ? 15 : 50, "bad arguments");
645 seed = number(a.get(1), 1, 2147483646, "bad arguments");
646 } else
647 require(a.isEmpty(), "bad arguments");
648 if (Set.of("RACE", "MOVES", "PLAY", "AUTO", "HINT", "EVAL", "SHOTS", "CUBE", "DOUBLE",
649 "TAKE", "DROP", "RESIGN", "GAME")
650 .contains(c) &&
651 state.over)
652 fail("game over");
653 if (Set.of("PLAY", "AUTO", "CUBE", "DOUBLE", "RESIGN", "GAME").contains(c) && state.pending)
654 fail("double pending");
655 switch (c) {
656 case "NEW" -> {
657 int[] score = state.score;
658 state = new State();
659 state.score = score;
660 history.clear();
661 redo.clear();
662 emit("OK");
663 }
664 case "SHOW" -> emit(positionText());
665 case "BOARD" -> board();
666 case "PIPS" -> emit("pips W " + state.position.pips(0) + " B " + state.position.pips(1));
667 case "RACE" ->
668 emit(state.position.contact()
669 ? "contact"
670 : "race W " + state.position.keith(0) + " B " + state.position.keith(1));
671 case "MOVES" -> {
672 List<Play> plays = new Generator(state.position, state.side, first, second).plays;
673 for (Play p : plays)
674 emit(p.notation);
675 emit("total " + plays.size());
676 }
677 case "PLAY" -> play(first, second, a.subList(2, a.size()));
678 case "AUTO", "HINT" -> {
679 List<Play> plays =
680 ranked(new Generator(state.position, state.side, first, second).plays, state.side);
681 if (c.equals("AUTO")) {
682 Play p = plays.get(0);
683 emit("OK " + p.notation + " eval " + evaluate(p.position, state.side));
684 commit(p, first, second);
685 } else
686 for (Play p : plays.subList(0, Math.min(amount, plays.size())))
687 emit(evaluate(p.position, state.side) + " " + p.notation);
688 }
689 case "EVAL" -> {
690 int[] t = terms(state.position, state.side);
691 emit("eval " + evaluate(state.position, state.side) + " pips " + t[0] + " blots " +
692 t[1] + " home " + t[2] + " prime " + t[3] + " anchors " + t[4] + " bar " + t[5]);
693 }
694 case "SHOTS" -> shots();
695 case "CUBE" -> emit("cube " + cube(state, false));
696 case "DOUBLE" -> {
697 if (state.owner == 1 - state.side || state.cube == 64)
698 fail("cube unavailable");
699 state.pending = true;
700 redo.clear();
701 emit("OK double " + 2 * state.cube);
702 }
703 case "TAKE" -> {
704 if (!state.pending)
705 fail("no double");
706 state.pending = false;
707 state.cube *= 2;
708 state.owner = 1 - state.side;
709 redo.clear();
710 emit("OK take " + state.cube + " " + state.ownerName());
711 }
712 case "DROP" -> {
713 if (!state.pending)
714 fail("no double");
715 state.pending = false;
716 redo.clear();
717 emit("OK drop");
718 end(state.side, "drop", state.cube);
719 }
720 case "RESIGN" -> {
721 redo.clear();
722 emit("OK resign");
723 end(1 - state.side, a.get(0), multiplier(a.get(0)) * state.cube);
724 }
725 case "SCORE" -> emit("score W " + state.score[0] + " B " + state.score[1]);
726 case "HISTORY" -> {
727 emit("history " + history.size());
728 for (int i = 0; i < history.size(); i++)
729 emit((i + 1) + " " + history.get(i).record);
730 }
731 case "UNDO" -> {
732 if (amount > history.size())
733 fail("cannot undo");
734 int index = history.size() - amount;
735 List<Entry> removed = new ArrayList<>(history.subList(index, history.size()));
736 state = removed.get(0).before.copy();
737 history.subList(index, history.size()).clear();
738 redo.addAll(0, removed);
739 emit("OK");
740 }
741 case "REDO" -> {
742 if (amount > redo.size())
743 fail("cannot redo");
744 List<Entry> entries = new ArrayList<>(redo.subList(0, amount));
745 redo.subList(0, amount).clear();
746 history.addAll(entries);
747 state = entries.get(entries.size() - 1).after.copy();
748 emit("OK");
749 }
750 case "GAME" -> {
751 Game g = simulate(state.copy(), new String[] {a.get(1), a.get(2)}, new Dice(seed), null,
752 0, new int[2], false);
753 g.lines.forEach(Main::emit);
754 emit("result " + COLORS[g.winner] + " " + g.result + " " + g.points);
755 }
756 case "MATCH", "STATS" -> {
757 Dice dice = new Dice(seed);
758 int[] points = new int[2];
759 int[][] counts = new int[4][2];
760 int turns = 0, games = 0;
761 boolean crawfordPlayed = false;
762 while (c.equals("MATCH") ? Math.max(points[0], points[1]) < amount : games < amount) {
763 boolean crawford = c.equals("MATCH") && !crawfordPlayed &&
764 Math.max(points[0], points[1]) == amount - 1;
765 int white, black;
766 do {
767 white = dice.roll();
768 black = dice.roll();
769 } while (white == black);
770 State start = new State();
771 start.side = white > black ? 0 : 1;
772 int firstPlayer = start.side;
773 Game g =
774 simulate(start, new String[] {"BEST", "BEST"}, dice, new int[] {white, black},
775 c.equals("MATCH") ? amount : 0, points, crawford);
776 games++;
777 turns += g.plays;
778 points[g.winner] += g.points;
779 counts[RESULTS.indexOf(g.result)][g.winner]++;
780 if (crawford)
781 crawfordPlayed = true;
782 if (c.equals("MATCH"))
783 emit("game " + games + (crawford ? " crawford" : "") + " start " +
784 COLORS[firstPlayer] + " turns " + g.plays + " result " + COLORS[g.winner] +
785 " " + g.result + " " + g.points + " score " + points[0] + "-" + points[1]);
786 }
787 if (c.equals("MATCH"))
788 emit("match " + COLORS[points[0] >= amount ? 0 : 1] + " " + points[0] + "-" +
789 points[1]);
790 else {
791 for (int i = 0; i < 4; i++)
792 emit(RESULTS.get(i) + " W " + counts[i][0] + " B " + counts[i][1]);
793 emit("points W " + points[0] + " B " + points[1]);
794 emit("turns " + turns);
795 }
796 }
797 }
798 }
799 public static void main(String[] args) throws IOException {
800 BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
801 int n = Integer.parseInt(reader.readLine());
802 for (int i = 0; i < n; i++) {
803 try {
804 execute(Arrays.asList(reader.readLine().split(" ")));
805 } catch (Failure e) {
806 emit("ERROR " + e.getMessage());
807 }
808 System.out.print(output);
809 System.out.flush();
810 output.setLength(0);
811 }
812 }
