44std::vector<std::string> split(const std::string &text, char delimiter) {
45 std::vector<std::string> result;
46 std::size_t start = 0;
47 while (true) {
48 const std::size_t end = text.find(delimiter, start);
49 result.push_back(text.substr(start, end == std::string::npos ? end : end - start));
50 if (end == std::string::npos)
51 return result;
52 start = end + 1;
53 }

71struct Position {
72 std::array<std::array<int, 26>, 2> checkers{};
73
74 static Position starting() {
75 Position position;
76 for (int color = 0; color < 2; ++color) {
77 position.checkers[color][24] = 2;
78 position.checkers[color][13] = 5;
79 position.checkers[color][8] = 3;
80 position.checkers[color][6] = 5;
81 }
82 return position;
83 }
84
85 bool operator<(const Position &other) const {
86 return checkers < other.checkers;
87 }
88 bool operator==(const Position &other) const {
89 return checkers == other.checkers;
90 }
91 int opposing(int color, int point) const {
92 return checkers[1 - color][25 - point];
93 }
94 int boardValue(int point) const {
95 return checkers[0][point] - checkers[1][25 - point];
96 }
97 bool finished(int color) const {
98 return checkers[color][Off] == 15;
99 }
100
101 bool allHome(int color) const {
102 for (int point = 7; point <= Bar; ++point)
103 if (checkers[color][point] != 0)
104 return false;
105 return true;
106 }
107
108 int pips(int color) const {
109 int count = 0;
110 for (int point = 1; point <= Bar; ++point)
111 count += point * checkers[color][point];
112 return count;
113 }
114
115 int keith(int color) const {
116 const std::array<int, 26> &mine = checkers[color];
117 int count = pips(color) + 2 * std::max(mine[1] - 1, 0) + std::max(mine[2] - 1, 0) +
118 std::max(mine[3] - 3, 0);
119 for (int point = 4; point <= 6; ++point)
120 if (mine[point] == 0)
121 ++count;
122 return count;
123 }
124
125 int rearmost(int color) const {
126 for (int point = Bar; point >= 1; --point)
127 if (checkers[color][point] != 0)
128 return point;
129 return 0;
130 }
131
132 bool contact() const {
133 return rearmost(0) + rearmost(1) > 25;
134 }
135};

151struct Evaluation {
152 int pips = 0, blots = 0, home = 0, prime = 0, anchors = 0, bar = 0;
153 int value() const {
154 return 2 * pips - blots + 4 * home + 3 * prime + 3 * anchors + 6 * bar;
155 }
156};
157
158Evaluation evaluate(const Position &position, int color) {
159 Evaluation terms;
160 const std::array<int, 26> &mine = position.checkers[color];
161 terms.pips = position.pips(1 - color) - position.pips(color);
162 for (int point = 1; point <= 24; ++point) {
163 if (mine[point] != 1)
164 continue;
165 std::array<bool, 26> distance{};
166 for (int target = 0; target < point; ++target) {
167 if (position.opposing(color, target) > 0)
168 distance[point - target] = true;
169 }
170 for (int first = 1; first <= 6; ++first) {
171 for (int second = 1; second <= 6; ++second) {
172 if (distance[first] || distance[second] || distance[first + second] ||
173 (first == second && (distance[3 * first] || distance[4 * first])))
174 ++terms.blots;
175 }
176 }
177 }
178 int run = 0;
179 for (int point = 1; point <= 24; ++point) {
180 if (mine[point] >= 2) {
181 ++run;
182 terms.prime = std::max(terms.prime, run);
183 if (point <= 6)
184 ++terms.home;
185 if (point >= 19)
186 ++terms.anchors;
187 } else
188 run = 0;
189 }
190 terms.bar = position.checkers[1 - color][Bar] - mine[Bar];
191 return terms;
192}

194std::vector<Move> canonicalKey(std::vector<Move> pairs, const std::set<int> &hits) {
195 const auto sortPairs = [&pairs]() {
196 std::sort(pairs.begin(), pairs.end(), std::greater<Move>());
197 };
198 sortPairs();
199 while (true) {
200 bool merged = false;
201 for (std::size_t a = 0; a < pairs.size() && !merged; ++a) {
202 const int middle = pairs[a].second;
203 if (middle < 1 || hits.count(middle) != 0)
204 continue;
205 for (std::size_t b = 0; b < pairs.size(); ++b) {
206 if (pairs[b].first != middle)
207 continue;
208 const Move combined{pairs[a].first, pairs[b].second};
209 pairs.erase(pairs.begin() + std::max(a, b));
210 pairs.erase(pairs.begin() + std::min(a, b));
211 pairs.push_back(combined);
212 sortPairs();
213 merged = true;
214 break;
215 }
216 }
217 if (!merged)
218 return pairs;
219 }
220}

222std::string notation(const std::vector<Move> &pairs, const std::set<int> &hits) {
223 if (pairs.empty())
224 return "none";
225 std::string text;
226 std::set<int> starred;
227 for (std::size_t i = 0; i < pairs.size();) {
228 std::size_t end = i + 1;
229 while (end < pairs.size() && pairs[end] == pairs[i])
230 ++end;
231 if (!text.empty())
232 text += ' ';
233 text += pointName(pairs[i].first) + '/' + pointName(pairs[i].second);
234 if (hits.count(pairs[i].second) != 0 && starred.insert(pairs[i].second).second)
235 text += '*';
236 if (end - i >= 2)
237 text += '(' + std::to_string(end - i) + ')';
238 i = end;
239 }
240 return text;
241}

260 void explore(const Position &position, std::array<int, 7> &remaining, std::vector<Move> &moves,
261 const std::set<int> &hits, int firstDie) {
262 const auto found = shortest.find(position);
263 if (found == shortest.end() || moves.size() < found->second)
264 shortest[position] = moves.size();
265 if (moves.size() > maximum) {
266 maximum = moves.size();
267 candidates.clear();
268 }
269 if (moves.size() == maximum)
270 candidates.push_back({position, moves, hits, firstDie});
271 for (int die = 1; die <= 6; ++die) {
272 if (remaining[die] == 0)
273 continue;
274 --remaining[die];
275 for (int start = Bar; start >= 1; --start) {
276 if (position.checkers[color][start] == 0)
277 continue;
278 if (position.checkers[color][Bar] > 0 && start != Bar)
279 continue;
280 int destination = start - die;
281 bool hit = false;
282 if (destination >= 1) {
283 if (position.opposing(color, destination) >= 2)
284 continue;
285 hit = position.opposing(color, destination) == 1;
286 } else {
287 if (!position.allHome(color))
288 continue;
289 bool farther = false;
290 if (destination < 0) {
291 for (int point = start + 1; point <= 6; ++point)
292 farther = farther || position.checkers[color][point] > 0;
293 }
294 if (farther)
295 continue;
296 destination = Off;
297 }
298 Position after = position;
299 --after.checkers[color][start];
300 ++after.checkers[color][destination];
301 std::set<int> nextHits = hits;
302 if (hit) {
303 --after.checkers[1 - color][25 - destination];
304 ++after.checkers[1 - color][Bar];
305 nextHits.insert(destination);
306 }
307 moves.push_back({start, destination});
308 explore(after, remaining, moves, nextHits, firstDie == 0 ? die : firstDie);
309 moves.pop_back();
310 }
311 ++remaining[die];
312 }
313 }
314
315 public:
316 std::size_t maximum = 0;
317 std::map<Position, std::size_t> shortest;
318 std::vector<Play> plays;
319
320 Generator(const Position &position, int side, int first, int second) : color(side) {
321 std::array<int, 7> remaining{};
322 if (first == second)
323 remaining[first] = 4;
324 else {
325 remaining[first] = 1;
326 remaining[second] = 1;
327 }
328 std::vector<Move> moves;
329 explore(position, remaining, moves, {}, 0);
330 bool largerRequired = false;
331 if (maximum == 1 && first != second) {
332 for (const Candidate &candidate : candidates)
333 if (candidate.firstDie == std::max(first, second))
334 largerRequired = true;
335 }
336 std::set<Position> seen;
337 for (const Candidate &candidate : candidates) {
338 if (largerRequired && candidate.firstDie != std::max(first, second))
339 continue;
340 if (!seen.insert(candidate.position).second)
341 continue;
342 const std::vector<Move> key = canonicalKey(candidate.moves, candidate.hits);
343 plays.push_back({candidate.position, key, notation(key, candidate.hits)});
344 }
345 std::sort(plays.begin(), plays.end(),
346 [](const Play &a, const Play &b) { return a.key > b.key; });
347 }
348};

350int bestIndex(const std::vector<Play> &plays, int side) {
351 int best = 0;
352 int value = evaluate(plays[0].position, side).value();
353 for (std::size_t i = 1; i < plays.size(); ++i) {
354 const int next = evaluate(plays[i].position, side).value();
355 if (next > value) {
356 best = static_cast<int>(i);
357 value = next;
358 }
359 }
360 return best;
361}

363std::optional<std::vector<Move>> parsePlay(const std::vector<std::string> &elements) {
364 if (elements.size() == 1 && elements[0] == "none")
365 return std::vector<Move>{};
366 std::vector<Move> moves;
367 for (std::string element : elements) {
368 int repetitions = 1;
369 if (element.size() >= 3 && element[element.size() - 3] == '(' && element.back() == ')' &&
370 element[element.size() - 2] >= '2' && element[element.size() - 2] <= '4') {
371 repetitions = element[element.size() - 2] - '0';
372 element.resize(element.size() - 3);
373 }
374 const std::vector<std::string> path = split(element, '/');
375 if (path.size() < 2)
376 return std::nullopt;
377 std::vector<int> points;
378 for (std::size_t i = 0; i < path.size(); ++i) {
379 std::string point = path[i];
380 int value;
381 if (i == 0 && point == "bar")
382 value = Bar;
383 else if (i > 0 && point == "off")
384 value = Off;
385 else {
386 if (i > 0 && !point.empty() && point.back() == '*')
387 point.pop_back();
388 if (point.empty() || point.size() > 2 || point[0] < '1' || point[0] > '9' ||
389 !std::all_of(point.begin(), point.end(),
390 [](char ch) { return ch >= '0' && ch <= '9'; }))
391 return std::nullopt;
392 value = std::stoi(point);
393 if (value > 24)
394 return std::nullopt;
395 }
396 if (!points.empty() && points.back() <= value)
397 return std::nullopt;
398 points.push_back(value);
399 }
400 for (std::size_t i = 1; i < points.size(); ++i) {
401 for (int repeat = 0; repeat < repetitions; ++repeat)
402 moves.push_back({points[i - 1], points[i]});
403 }
404 }
405 return moves;
406}

408const Play &validatePlay(const Position &position, int color, const std::vector<Move> &moves,
409 const Generator &generator) {
410 Position final = position;
411 std::set<int> destinations;
412 for (Move move : moves) {
413 --final.checkers[color][move.first];
414 ++final.checkers[color][move.second];
415 if (move.second >= 1)
416 destinations.insert(move.second);
417 }
418 for (int destination : destinations) {
419 if (position.opposing(color, destination) == 1) {
420 --final.checkers[1 - color][25 - destination];
421 ++final.checkers[1 - color][Bar];
422 }
423 }
424 for (int point = 1; point <= Bar; ++point)
425 if (final.checkers[color][point] < 0)
426 illegal("no checker");
427 if (final.checkers[color][Bar] > 0) {
428 for (Move move : moves)
429 if (move.first != Bar)
430 illegal("bar");
431 }
432 for (Move move : moves)
433 if (move.second >= 1 && position.opposing(color, move.second) >= 2)
434 illegal("blocked");
435 for (Move move : moves)
436 if (move.second == Off && !final.allHome(color))
437 illegal("bear-off");
438 for (const Play &play : generator.plays)
439 if (play.position == final)
440 return play;
441 const auto possible = generator.shortest.find(final);
442 if (possible != generator.shortest.end())
443 illegal(possible->second < generator.maximum ? "too few dice" : "larger die required");
444 illegal("impossible");
445}

453std::string naturalResult(const Position &position, int winner) {
454 const std::array<int, 26> &loser = position.checkers[1 - winner];
455 if (loser[Off] > 0)
456 return "single";
457 if (loser[Bar] > 0)
458 return "backgammon";
459 for (int point = 19; point <= 24; ++point)
460 if (loser[point] > 0)
461 return "backgammon";
462 return "gammon";
463}

465std::string cubeDecision(const State &state, bool prohibited = false) {
466 const int side = state.side, other = 1 - side;
467 if (state.owner == other || state.cube == 64 || prohibited)
468 return "unavailable";
469 if (state.position.contact()) {
470 const int value = evaluate(state.position, side).value();
471 if (value < 60)
472 return "no double";
473 return value <= 100 ? "double take" : "double drop";
474 }
475 const int value = 8 * state.position.keith(side) - 7 * state.position.keith(other);
476 const int limit = state.owner < 0 ? 28 : 21;
477 if (value > limit)
478 return "no double";
479 return value >= 14 ? "double take" : "double drop";
480}

539struct GameRecord {
540 Result result;
541 int plays;
542 std::vector<std::string> lines;
543};
544
545GameRecord playGame(State state, const std::array<std::string, 2> &strategies, Dice &dice,
546 std::optional<std::pair<int, int>> opening = std::nullopt, int target = 0,
547 std::array<int, 2> points = {}, bool crawford = false) {
548 GameRecord record{{0, "", 0}, 0, {}};
549 for (int turn = 1;; ++turn) {
550 const int side = state.side, other = 1 - side;
551 int first, second;
552 const std::string prefix = std::to_string(turn) + ' ';
553 if (turn == 1 && opening.has_value()) {
554 first = opening->first;
555 second = opening->second;
556 } else {
557 const bool prohibited = target > 0 && (crawford || points[side] + state.cube >= target);
558 const std::string decision = cubeDecision(state, prohibited);
559 if (decision == "double take" || decision == "double drop")
560 record.lines.push_back(prefix + colors[side] + " doubles " +
561 std::to_string(2 * state.cube));
562 if (decision == "double drop") {
563 record.lines.push_back(prefix + colors[other] + " drops");
564 record.result = {side, "drop", state.cube};
565 return record;
566 }
567 if (decision == "double take") {
568 record.lines.push_back(prefix + colors[other] + " takes");
569 state.cube *= 2;
570 state.owner = other;
571 }
572 first = dice.roll();
573 second = dice.roll();
574 }
575 const Generator generator(state.position, side, first, second);
576 std::size_t index = 0;
577 if (strategies[side] == "BEST")
578 index = bestIndex(generator.plays, side);
579 else if (strategies[side] == "RANDOM")
580 index = dice.next() % generator.plays.size();
581 const Play &play = generator.plays[index];
582 state.position = play.position;
583 ++record.plays;
584 record.lines.push_back(prefix + colors[side] + ' ' + std::to_string(first) + '-' +
585 std::to_string(second) + ' ' + play.text);
586 if (state.position.finished(side)) {
587 const std::string kind = naturalResult(state.position, side);
588 record.result = {side, kind, multiplier(kind) * state.cube};
589 return record;
590 }
591 state.side = other;
592 }
593}

595std::pair<State, std::pair<int, int>> openingGame(Dice &dice) {
596 int white, black;
597 do {
598 white = dice.roll();
599 black = dice.roll();
600 } while (white == black);
601 State state;
602 state.side = white > black ? 0 : 1;
603 return {state, {white, black}};
604}

606State parsePosition(const std::vector<std::string> &args, std::array<int, 2> score) {
607 require(args.size() == 4, "bad position");
608 const std::vector<std::string> values = split(args[0], ',');
609 require(values.size() == 24 && (args[3] == "W" || args[3] == "B"), "bad position");
610 State state;
611 state.position = Position{};
612 state.side = args[3] == "W" ? 0 : 1;
613 state.score = score;
614 for (int point = 1; point <= 24; ++point) {
615 const std::string &text = values[point - 1];
616 const bool negative = !text.empty() && text[0] == '-';
617 const int value =
618 static_cast<int>(argument(negative ? text.substr(1) : text, 0, 15, "bad position"));
619 if (negative)
620 state.position.checkers[1][25 - point] = value;
621 else
622 state.position.checkers[0][point] = value;
623 }
624 state.position.checkers[0][Bar] = static_cast<int>(argument(args[1], 0, 15, "bad position"));
625 state.position.checkers[1][Bar] = static_cast<int>(argument(args[2], 0, 15, "bad position"));
626 for (int color = 0; color < 2; ++color) {
627 int total = 0;
628 for (int point = 1; point <= Bar; ++point)
629 total += state.position.checkers[color][point];
630 require(total >= 1 && total <= 15, "bad position");
631 state.position.checkers[color][Off] = 15 - total;
632 }
633 return state;
634}

688void shots(const State &state) {
689 std::vector<int> blots;
690 std::array<int, 26> weights{};
691 for (int point = 24; point >= 1; --point)
692 if (state.position.checkers[state.side][point] == 1)
693 blots.push_back(point);
694 int total = 0;
695 std::string rolls;
696 for (int first = 6; first >= 1; --first) {
697 for (int second = first; second >= 1; --second) {
698 const Generator generator(state.position, 1 - state.side, first, second);
699 std::set<int> hit;
700 for (const Play &play : generator.plays) {
701 for (int point : blots)
702 if (play.position.checkers[state.side][point] == 0)
703 hit.insert(point);
704 }
705 if (hit.empty())
706 continue;
707 const int weight = first == second ? 1 : 2;
708 total += weight;
709 rolls += ' ' + std::to_string(first) + '-' + std::to_string(second);
710 for (int point : hit)
711 weights[point] += weight;
712 }
713 }
714 std::cout << "shots " << total << "/36" << rolls << '\n';
715 for (int point : blots)
716 std::cout << "blot " << point << ' ' << weights[point] << "/36\n";
717}

720 return names.count(name) != 0;

723void validateArguments(const std::string &op, const std::vector<std::string> &args) {
724 const std::set<std::string> noArgs = {"NEW", "SHOW", "BOARD", "PIPS", "RACE",
725 "EVAL", "SHOTS", "CUBE", "DOUBLE", "TAKE",
726 "DROP", "SCORE", "HISTORY"};
727 if (contains(noArgs, op)) {
728 require(args.empty());
729 return;
730 }
731 if (op == "MOVES" || op == "AUTO" || op == "HINT" || op == "PLAY") {
732 if (op == "HINT")
733 require(args.size() == 3);
734 else if (op == "PLAY")
735 require(args.size() >= 3);
736 else
737 require(args.size() == 2);
738 argument(args[0], 1, 6);
739 argument(args[1], 1, 6);
740 if (op == "HINT")
741 argument(args[2], 1, 1000);
742 } else if (op == "UNDO" || op == "REDO") {
743 require(args.size() == 1);
744 argument(args[0], 1, 1000);
745 } else if (op == "RESIGN") {
746 require(args.size() == 1);
747 require(args[0] == "single" || args[0] == "gammon" || args[0] == "backgammon");
748 } else if (op == "GAME") {
749 require(args.size() == 3);
750 argument(args[0], 1, 2147483646);
751 const std::set<std::string> strategies = {"BEST", "FIRST", "RANDOM"};
752 require(contains(strategies, args[1]) && contains(strategies, args[2]));
753 } else if (op == "MATCH" || op == "STATS") {
754 require(args.size() == 2);
755 argument(args[0], 1, op == "MATCH" ? 15 : 50);
756 argument(args[1], 1, 2147483646);
757 }
758}

760int main() {
761 std::ios::sync_with_stdio(false);
762 std::cin.tie(nullptr);
763 std::string line;
764 if (!std::getline(std::cin, line))
765 return 0;
766 const int commands = std::stoi(line);
767 State state;
768 std::vector<HistoryEntry> history, redo;
769 const std::set<std::string> known = {"NEW", "POS", "SHOW", "BOARD", "PIPS", "RACE",
770 "MOVES", "PLAY", "AUTO", "HINT", "EVAL", "SHOTS",
771 "CUBE", "DOUBLE", "TAKE", "DROP", "RESIGN", "SCORE",
772 "HISTORY", "UNDO", "REDO", "GAME", "MATCH", "STATS"};
773 const std::set<std::string> needsGame = {"RACE", "MOVES", "PLAY", "AUTO", "HINT",
774 "EVAL", "SHOTS", "CUBE", "DOUBLE", "TAKE",
775 "DROP", "RESIGN", "GAME"};
776 const std::set<std::string> noPending = {"PLAY", "AUTO", "CUBE", "DOUBLE", "RESIGN", "GAME"};
777 for (int i = 0; i < commands && std::getline(std::cin, line); ++i) {
778 const std::vector<std::string> tokens = words(line);
779 const std::string &op = tokens[0];
780 const std::vector<std::string> args(tokens.begin() + 1, tokens.end());
781 try {
782 require(contains(known, op), "unknown command");
783 if (op == "POS") {
784 State replacement = parsePosition(args, state.score);
785 state = std::move(replacement);
786 history.clear();
787 redo.clear();
788 std::cout << "OK\n";
789 std::cout.flush();
790 continue;
791 }
792 validateArguments(op, args);
793 require(!contains(needsGame, op) || !state.over, "game over");
794 require(!contains(noPending, op) || !state.pending, "double pending");
795 if (op == "NEW") {
796 const std::array<int, 2> score = state.score;
797 state = State{};
798 state.score = score;
799 history.clear();
800 redo.clear();
801 std::cout << "OK\n";
802 } else if (op == "SHOW")
803 show(state);
804 else if (op == "BOARD")
805 board(state);
806 else if (op == "PIPS")
807 std::cout << "pips W " << state.position.pips(0) << " B " << state.position.pips(1)
808 << '\n';
809 else if (op == "RACE") {
810 if (state.position.contact())
811 std::cout << "contact\n";
812 else
813 std::cout << "race W " << state.position.keith(0) << " B "
814 << state.position.keith(1) << '\n';
815 } else if (op == "MOVES" || op == "PLAY" || op == "AUTO" || op == "HINT") {
816 const int first = std::stoi(args[0]), second = std::stoi(args[1]);
817 std::optional<std::vector<Move>> requested;
818 if (op == "PLAY") {
819 requested = parsePlay(std::vector<std::string>(args.begin() + 2, args.end()));
820 if (!requested.has_value())
821 illegal("syntax");
822 }
823 const Generator generator(state.position, state.side, first, second);
824 if (op == "MOVES") {
825 for (const Play &play : generator.plays)
826 std::cout << play.text << '\n';
827 std::cout << "total " << generator.plays.size() << '\n';
828 } else if (op == "HINT") {
829 std::vector<std::pair<int, std::size_t>> ranked;
830 for (std::size_t index = 0; index < generator.plays.size(); ++index)
831 ranked.push_back(
832 {evaluate(generator.plays[index].position, state.side).value(), index});
833 std::stable_sort(
834 ranked.begin(), ranked.end(),
835 [](const auto &a, const auto &b) { return a.first > b.first; });
836 const std::size_t count =
837 std::min(static_cast<std::size_t>(std::stoi(args[2])), ranked.size());
838 for (std::size_t index = 0; index < count; ++index)
839 std::cout << ranked[index].first << ' '
840 << generator.plays[ranked[index].second].text << '\n';
841 } else {
842 const Play &play =
843 op == "PLAY"
844 ? validatePlay(state.position, state.side, *requested, generator)
845 : generator.plays[bestIndex(generator.plays, state.side)];
846 const int value = evaluate(play.position, state.side).value();
847 const std::optional<Result> result =
848 commitPlay(state, play, first, second, history, redo);
849 std::cout << "OK " << play.text;
850 if (op == "AUTO")
851 std::cout << " eval " << value;
852 std::cout << '\n';
853 if (result.has_value())
854 printResult(*result);
855 }
856 } else if (op == "EVAL") {
857 const Evaluation terms = evaluate(state.position, state.side);
858 std::cout << "eval " << terms.value() << " pips " << terms.pips << " blots "
859 << terms.blots << " home " << terms.home << " prime " << terms.prime
860 << " anchors " << terms.anchors << " bar " << terms.bar << '\n';
861 } else if (op == "SHOTS")
862 shots(state);
863 else if (op == "CUBE")
864 std::cout << "cube " << cubeDecision(state) << '\n';
865 else if (op == "DOUBLE") {
866 require(state.owner != 1 - state.side && state.cube != 64, "cube unavailable");
867 state.pending = true;
868 redo.clear();
869 std::cout << "OK double " << 2 * state.cube << '\n';
870 } else if (op == "TAKE") {
871 require(state.pending, "no double");
872 state.pending = false;
873 state.cube *= 2;
874 state.owner = 1 - state.side;
875 redo.clear();
876 std::cout << "OK take " << state.cube << ' ' << state.ownerName() << '\n';
877 } else if (op == "DROP" || op == "RESIGN") {
878 if (op == "DROP")
879 require(state.pending, "no double");
880 const int winner = op == "DROP" ? state.side : 1 - state.side;
881 const Result result = finish(state, winner, op == "DROP" ? "drop" : args[0]);
882 redo.clear();
883 std::cout << (op == "DROP" ? "OK drop\n" : "OK resign\n");
884 printResult(result);
885 } else if (op == "SCORE")
886 std::cout << "score W " << state.score[0] << " B " << state.score[1] << '\n';
887 else if (op == "HISTORY") {
888 std::cout << "history " << history.size() << '\n';
889 for (std::size_t index = 0; index < history.size(); ++index)
890 std::cout << index + 1 << ' ' << history[index].record << '\n';
891 } else if (op == "UNDO") {
892 const std::size_t count = std::stoi(args[0]);
893 require(count <= history.size(), "cannot undo");
894 const std::size_t start = history.size() - count;
895 state = history[start].before;
896 redo.insert(redo.begin(), history.begin() + start, history.end());
897 history.erase(history.begin() + start, history.end());
898 std::cout << "OK\n";
899 } else if (op == "REDO") {
900 const std::size_t count = std::stoi(args[0]);
901 require(count <= redo.size(), "cannot redo");
902 state = redo[count - 1].after;
903 history.insert(history.end(), redo.begin(), redo.begin() + count);
904 redo.erase(redo.begin(), redo.begin() + count);
905 std::cout << "OK\n";
906 } else if (op == "GAME") {
907 Dice dice(std::stoll(args[0]));
908 const GameRecord game = playGame(state, {args[1], args[2]}, dice);
909 for (const std::string &output : game.lines)
910 std::cout << output << '\n';
911 printResult(game.result);
912 } else if (op == "MATCH") {
913 const int target = std::stoi(args[0]);
914 Dice dice(std::stoll(args[1]));
915 std::array<int, 2> points{};
916 bool crawfordPlayed = false;
917 int gameNumber = 0;
918 while (points[0] < target && points[1] < target) {
919 ++gameNumber;
920 const bool crawford =
921 !crawfordPlayed && std::max(points[0], points[1]) == target - 1;
922 if (crawford)
923 crawfordPlayed = true;
924 const auto opening = openingGame(dice);
925 const GameRecord game = playGame(opening.first, {"BEST", "BEST"}, dice,
926 opening.second, target, points, crawford);
927 points[game.result.winner] += game.result.points;
928 std::cout << "game " << gameNumber;
929 if (crawford)
930 std::cout << " crawford";
931 std::cout << " start " << colors[opening.first.side] << " turns " << game.plays
932 << " result " << colors[game.result.winner] << ' ' << game.result.kind
933 << ' ' << game.result.points << " score " << points[0] << '-'
934 << points[1] << '\n';
935 }
936 std::cout << "match " << colors[points[0] >= target ? 0 : 1] << ' ' << points[0]
937 << '-' << points[1] << '\n';
938 } else if (op == "STATS") {
939 const int count = std::stoi(args[0]);
940 Dice dice(std::stoll(args[1]));
941 std::array<std::array<int, 2>, 4> counts{};
942 std::array<int, 2> points{};
943 int turns = 0;
944 for (int gameNumber = 0; gameNumber < count; ++gameNumber) {
945 const auto opening = openingGame(dice);
946 const GameRecord game =
947 playGame(opening.first, {"BEST", "BEST"}, dice, opening.second);
948 const std::size_t kind =
949 std::find(resultNames.begin(), resultNames.end(), game.result.kind) -
950 resultNames.begin();
951 ++counts[kind][game.result.winner];
952 points[game.result.winner] += game.result.points;
953 turns += game.plays;
954 }
955 for (std::size_t kind = 0; kind < counts.size(); ++kind)
956 std::cout << resultNames[kind] << " W " << counts[kind][0] << " B "
957 << counts[kind][1] << '\n';
958 std::cout << "points W " << points[0] << " B " << points[1] << "\nturns " << turns
959 << '\n';
960 }
961 } catch (const CommandError &error) {
962 std::cout << error.what() << '\n';
963 }
964 std::cout.flush();
965 }
