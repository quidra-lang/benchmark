56function notation(pairs: Pair[], hits: number[]): { key: Pair[]; notation: string } {
57 let key = pairs.map((p) => [p[0], p[1]] as Pair).sort(pairOrder);
58 while (true) {
59 const a = key.findIndex(
60 (p) => p[1] > 0 && !hits.includes(p[1]) && key.some((q) => q[0] === p[1]),
61 );
62 if (a < 0) break;
63 const b = key.findIndex((p) => p[0] === key[a][1]);
64 const pair: Pair = [key[a][0], key[b][1]];
65 key = key
66 .filter((_, i) => i !== a && i !== b)
67 .concat([pair])
68 .sort(pairOrder);
69 }
70 const elements: string[] = [],
71 marked: number[] = [];
72 for (let i = 0; i < key.length;) {
73 const [s, t] = key[i];
74 let j = i + 1;
75 while (j < key.length && key[j][0] === s && key[j][1] === t) j++;
76 const star = hits.includes(t) && !marked.includes(t);
77 if (star) marked.push(t);
78 elements.push(
79 `${s === 25 ? "bar" : s}/${t === 0 ? "off" : t}${star ? "*" : ""}${j - i > 1 ? "(" + (j - i) + ")" : ""}`,
80 );
81 i = j;
82 }
83 return { key, notation: elements.length ? elements.join(" ") : "none" };

85function canonical(a: Play, b: Play): number {

88 if (d) return d;
89 }

156function evaluate(p: Position, side: Color): number[] {

161 H = 0,
162 R = 0,
163 A = 0,
164 run = 0;
165 for (let point = 1; point <= 24; point++) {
166 if (own[point] >= 2) {
167 run++;
168 R = Math.max(R, run);
169 if (point <= 6) H++;
170 if (point >= 19) A++;

172 if (own[point] !== 1) continue;

177 for (let b = 1; b <= 6; b++) {

179 distances.has(a) ||
180 distances.has(b) ||
181 distances.has(a + b) ||

184 S++;

210function cube(s: State, prohibited = false): string {

255function parsePlay(words: string[]): Pair[] | undefined {
256 if (words.length === 1 && words[0] === "none") return [];

258 for (const element of words) {
259 const match = /^(.*?)(?:\(([234])\))?$/.exec(element)!;
260 const points = match[1].split("/"),
261 repeat = match[2] ? Number(match[2]) : 1;
262 if (points.length < 2 || !/^(bar|[1-9]|1[0-9]|2[0-4])$/.test(points[0])) return undefined;
263 const nums = [points[0] === "bar" ? 25 : Number(points[0])];
264 for (let i = 1; i < points.length; i++) {
265 if (!/^(off|(?:[1-9]|1[0-9]|2[0-4])\*?)$/.test(points[i])) return undefined;
266 nums.push(points[i] === "off" ? 0 : Number(points[i].replace("*", "")));
267 if (nums[i] >= nums[i - 1]) return undefined;
268 }
269 for (let i = 1; i < nums.length; i++)
270 for (let k = 0; k < repeat; k++) pairs.push([nums[i - 1], nums[i]]);
271 }

290 start: State,

293 opening: boolean,

298 let openingDice: number[] | undefined;
299 if (opening) {
300 let w = 0,
301 b = 0;
302 while (w === b) {
303 w = (random() % 6) + 1;
304 b = (random() % 6) + 1;

659 emit(`OK take ${state.cube} ${state.owner}`);
660 }
661 break;
662 case "DROP":
663 if (!state.pending) error("no double");
664 else {
665 const r = end(state, state.side, "drop");
666 redo = [];
667 emit("OK drop");
668 emit(resultLine(r));
669 }
670 break;
671 case "RESIGN": {
672 const r = end(state, opposite(state.side), a[0]);
673 redo = [];
674 emit("OK resign");
675 emit(resultLine(r));
676 break;
677 }
678 case "SCORE":
679 emit(`score W ${state.score.W} B ${state.score.B}`);
680 break;
681 case "HISTORY":
682 emit("history " + history.length);
683 history.forEach((e, i) => emit(`${i + 1} ${e.record}`));
684 break;
685 case "UNDO":
686 if (n[0] > history.length) error("cannot undo");
687 else {
688 const entries = history.splice(history.length - n[0]);
689 state = clone(entries[0].before);
690 redo = entries.concat(redo);
691 emit("OK");
692 }
693 break;
694 case "REDO":
695 if (n[0] > redo.length) error("cannot redo");
696 else {
697 const entries = redo.splice(0, n[0]);
698 history = history.concat(entries);
699 state = clone(entries[entries.length - 1].after);
700 emit("OK");
701 }
702 break;
703 case "GAME":
704 case "MATCH":
705 case "STATS": {
706 let seed = cmd === "GAME" ? n[0] : n[1];
707 const random = (): number => {
708 seed = (48271 * seed) % 2147483647;
709 return seed;
710 };
711 if (cmd === "GAME") {
712 selfPlay(state, random, a.slice(1), false, true, () => false);
713 break;
714 }
715 const scores = { W: 0, B: 0 },
716 counts = new Map<string, { W: number; B: number }>();
717 for (const kind of ["single", "gammon", "backgammon", "drop"])
718 counts.set(kind, { W: 0, B: 0 });
719 let game = 0,
720 turns = 0,
721 crawfordUsed = false;
722 while (cmd === "MATCH" ? scores.W < n[0] && scores.B < n[0] : game < n[0]) {
723 game++;
724 const crawford =
725 cmd === "MATCH" && !crawfordUsed && Math.max(scores.W, scores.B) === n[0] - 1;
726 if (crawford) crawfordUsed = true;
727 const played = selfPlay(
728 initial(),
729 random,
730 ["BEST", "BEST"],
731 true,
732 false,
733 (s) => cmd === "MATCH" && (crawford || scores[s.side] + s.cube >= n[0]),
734 );
735 const r = played.result;
736 scores[r.winner] += r.points;
737 counts.get(r.kind)![r.winner]++;
738 turns += played.turns;
739 if (cmd === "MATCH")
740 emit(
741 `game ${game}${crawford ? " crawford" : ""} start ${played.first} turns ${played.turns} ${resultLine(r)} score ${scores.W}-${scores.B}`,
742 );
743 }
744 if (cmd === "MATCH")
745 emit(`match ${scores.W >= n[0] ? "W" : "B"} ${scores.W}-${scores.B}`);
746 else {
747 for (const [kind, c] of counts) emit(`${kind} W ${c.W} B ${c.B}`);
748 emit(`points W ${scores.W} B ${scores.B}`);
749 emit("turns " + turns);
750 }
751 break;
752 }
753 }
754}
755const lines = readFileSync(0, "utf8").trimEnd().split("\n");
756for (let i = 1; i <= Number(lines[0]); i++) {
757 execute(lines[i].trimEnd().split(" "));
758 writeFileSync(1, output.join("\n") + "\n");
759 output.length = 0;
