64 static string PositionKey(int[,] pos) {
65 List<int> counts = new List<int>(52);
66 for (int x = 0; x < 2; ++x)
67 for (int p = 0; p <= 25; ++p)
68 counts.Add(pos[x, p]);
69 return string.Join(',', counts);

71 static int PairCompare(Move a, Move b) {
72 int comparison = b.From.CompareTo(a.From);
73 return comparison != 0 ? comparison : b.To.CompareTo(a.To);
74 }
75 static int KeyCompare(List<Move> a, List<Move> b) {
76 for (int i = 0; i < Math.Min(a.Count, b.Count); ++i) {
77 int comparison = PairCompare(a[i], b[i]);
78 if (comparison != 0)
79 return comparison;
80 }
81 return b.Count.CompareTo(a.Count);
82 }
83 static (List<Move>, string) Notation(List<Move> moves, HashSet<int> hits) {
84 List<Move> key = new List<Move>(moves);
85 key.Sort(PairCompare);
86 while (true) {
87 int ai = -1, bi = -1;
88 for (int i = 0; i < key.Count && ai < 0; ++i) {
89 int t = key[i].To;
90 if (t == 0 || hits.Contains(t))
91 continue;
92 int j = key.FindIndex(m => m.From == t);
93 if (j >= 0) {
94 ai = i;
95 bi = j;
96 }
97 }
98 if (ai < 0)
99 break;
100 Move merged = new Move(key[ai].From, key[bi].To);
101 key.RemoveAt(Math.Max(ai, bi));
102 key.RemoveAt(Math.Min(ai, bi));
103 key.Add(merged);
104 key.Sort(PairCompare);
105 }
106 List<string> parts = new List<string>();
107 HashSet<int> marked = new HashSet<int>();
108 for (int i = 0; i < key.Count;) {
109 Move move = key[i];
110 int j = i + 1;
111 while (j < key.Count && key[j] == move)
112 ++j;
113 string text = (move.From == 25 ? "bar" : move.From.ToString()) + "/" +
114 (move.To == 0 ? "off" : move.To.ToString());
115 if (hits.Contains(move.To) && marked.Add(move.To))
116 text += "*";
117 if (j - i > 1)
118 text += $"({j - i})";
119 parts.Add(text);
120 i = j;
121 }
122 return (key, parts.Count == 0 ? "none" : string.Join(' ', parts));
123 }
124 static Plays Enumerate(int[,] pos, int side, int d1, int d2) {
125 Plays plays = new Plays();
126 List<int> dice = d1 == d2 ? new List<int> { d1, d1, d1, d1 } : new List<int> { d1, d2 };
127 void Visit(int[,] board, List<int> remaining, List<Move> moves, HashSet<int> hits,
128 int firstDie) {
129 (List<Move> key, string text) = Notation(moves, hits);
130 plays.All.Add(new Play { Pos = board, Key = key, Text = text, Id = PositionKey(board),
131 Count = moves.Count, FirstDie = firstDie });
132 plays.Maximum = Math.Max(plays.Maximum, moves.Count);
133 HashSet<int> tried = new HashSet<int>();
134 for (int di = 0; di < remaining.Count; ++di) {
135 int die = remaining[di];
136 if (!tried.Add(die))
137 continue;
138 for (int s = 1; s <= 25; ++s) {
139 if (board[side, s] == 0 || (board[side, 25] > 0 && s != 25))
140 continue;
141 int t = Math.Max(0, s - die);
142 bool valid = true;
143 if (t > 0)
144 valid = board[1 - side, 25 - t] < 2;
145 else {
146 if (board[side, 25] > 0)
147 valid = false;
148 for (int p = 7; p <= 24; ++p)
149 if (board[side, p] > 0)
150 valid = false;
151 if (s < die)
152 for (int p = s + 1; p <= 6; ++p)
153 if (board[side, p] > 0)
154 valid = false;
155 }
156 if (!valid)
157 continue;
158 int[,] next = (int[,])board.Clone();
159 --next[side, s];
160 ++next[side, t];
161 HashSet<int> nextHits = new HashSet<int>(hits);
162 if (t > 0 && board[1 - side, 25 - t] == 1) {
163 next[1 - side, 25 - t] = 0;
164 ++next[1 - side, 25];
165 nextHits.Add(t);
166 }
167 List<int> rest = new List<int>(remaining);
168 rest.RemoveAt(di);
169 List<Move> path = new List<Move>(moves) { new Move(s, t) };
170 Visit(next, rest, path, nextHits, moves.Count == 0 ? die : firstDie);
171 }
172 }
173 }
174 Visit(pos, dice, new List<Move>(), new HashSet<int>(), 0);
175 bool larger = plays.Maximum == 1 && d1 != d2 &&
176 plays.All.Any(p => p.Count == 1 && p.FirstDie == Math.Max(d1, d2));
177 HashSet<string> seen = new HashSet<string>();
178 foreach (Play play in plays.All) {
179 if (play.Count == plays.Maximum && (!larger || play.FirstDie == Math.Max(d1, d2)) &&
180 seen.Add(play.Id))
181 plays.Legal.Add(play);
182 }
183 plays.Legal.Sort((a, b) => KeyCompare(a.Key, b.Key));
184 return plays;
185 }
186 static int Pips(int[,] pos, int x) {
187 int n = 0;
188 for (int p = 1; p <= 25; ++p)
189 n += p * pos[x, p];
190 return n;
191 }
192 static int[] Evaluation(int[,] pos, int side) {
193 int opponent = 1 - side;
194 int p = Pips(pos, opponent) - Pips(pos, side), s = 0, home = 0, prime = 0, anchors = 0,
195 run = 0;
196 for (int point = 1; point <= 24; ++point) {
197 if (pos[side, point] == 1) {
198 HashSet<int> distances = new HashSet<int>();
199 for (int q = 0; q < point; ++q)
200 if (pos[opponent, 25 - q] > 0)
201 distances.Add(point - q);
202 for (int a = 1; a <= 6; ++a)
203 for (int b = 1; b <= 6; ++b) {
204 if (distances.Contains(a) || distances.Contains(b) ||
205 distances.Contains(a + b) ||
206 (a == b && (distances.Contains(3 * a) || distances.Contains(4 * a))))
207 ++s;
208 }
209 }
210 if (pos[side, point] >= 2) {
211 if (point <= 6)
212 ++home;
213 if (point >= 19)
214 ++anchors;
215 prime = Math.Max(prime, ++run);
216 } else
217 run = 0;
218 }
219 int bar = pos[opponent, 25] - pos[side, 25];
220 return new[] { 2 * p - s + 4 * home + 3 * prime + 3 * anchors + 6 * bar,
221 p,
222 s,
223 home,
224 prime,
225 anchors,
226 bar };
227 }
228 static int Keith(int[,] pos, int x) {
229 int n = Pips(pos, x) + 2 * Math.Max(pos[x, 1] - 1, 0) + Math.Max(pos[x, 2] - 1, 0) +
230 Math.Max(pos[x, 3] - 3, 0);
231 for (int p = 4; p <= 6; ++p)
232 if (pos[x, p] == 0)
233 ++n;
234 return n;
235 }
236 static bool Contact(int[,] pos) {
237 int w = 0, b = 25;
238 for (int p = 1; p <= 24; ++p) {
239 if (pos[0, p] > 0)
240 w = p;
241 if (pos[1, p] > 0)
242 b = 25 - p;
243 }
244 if (pos[0, 25] > 0)
245 w = 25;
246 if (pos[1, 25] > 0)
247 b = 0;
248 return w > b;
249 }
250 static string CubeDecision(State state, bool forbidden = false) {
251 int x = state.Side;
252 if (state.Owner == 1 - x || state.Cube == 64 || forbidden)
253 return "unavailable";
254 if (Contact(state.Pos)) {
255 int value = Evaluation(state.Pos, x)[0];
256 if (value < 60)
257 return "no double";
258 return value <= 100 ? "double take" : "double drop";
259 }
260 int v = 8 * Keith(state.Pos, x) - 7 * Keith(state.Pos, 1 - x);
261 int limit = state.Owner == -1 ? 28 : 21;
262 if (v > limit)
263 return "no double";
264 return v >= 14 ? "double take" : "double drop";
265 }
266 static Result EndGame(State state, int winner, string? kind = null) {
267 if (kind == null) {
268 int opponent = 1 - winner;
269 bool back = state.Pos[opponent, 25] > 0;
270 for (int p = 19; p <= 24; ++p)
271 if (state.Pos[opponent, p] > 0)
272 back = true;
273 kind = state.Pos[opponent, 0] > 0 ? "single" : (back ? "backgammon" : "gammon");
274 }
275 int multiplier = kind == "gammon" ? 2 : (kind == "backgammon" ? 3 : 1);
276 int points = state.Cube * multiplier;
277 state.Pending = false;
278 state.Over = true;
279 state.Score[winner] += points;
280 return new Result(winner, kind, points);
281 }
282 static Result? ApplyPlay(State state, Play play) {
283 state.Pos = (int[,])play.Pos.Clone();
284 if (state.Pos[state.Side, 0] == 15)
285 return EndGame(state, state.Side);
286 state.Side = 1 - state.Side;
287 return null;
288 }
289 static List<(int Value, int Index, Play Play)> Ranked(List<Play> legal, int side) {
290 return legal.Select((p, i) => (Value: Evaluation(p.Pos, side)[0], Index: i, Play: p))
291 .OrderByDescending(x => x.Value)
292 .ThenBy(x => x.Index)
293 .ToList();
294 }
295 static List<Move> ParsePlay(string[] tokens) {
296 if (tokens.Length == 1 && tokens[0] == "none")
297 return new List<Move>();
298 List<Move> moves = new List<Move>();
299 foreach (string token in tokens) {
300 Match match = Regex.Match(token, @"\A(.+?)(?:\(([234])\))?\z");
301 Require(match.Success, "ILLEGAL syntax");
302 string[] path = match.Groups[1].Value.Split('/');
303 int repeat = match.Groups[2].Success ? int.Parse(match.Groups[2].Value) : 1;
304 Require(path.Length >= 2, "ILLEGAL syntax");
305 int previous = -1;
306 for (int i = 0; i < path.Length; ++i) {
307 string word = path[i];
308 int point;
309 if (i == 0 && word == "bar")
310 point = 25;
311 else if (i > 0 && word == "off")
312 point = 0;
313 else {
314 string pattern =
315 i == 0 ? @"\A(?:[1-9]|1[0-9]|2[0-4])\z" : @"\A(?:[1-9]|1[0-9]|2[0-4])\*?\z";
316 Require(Regex.IsMatch(word, pattern), "ILLEGAL syntax");
317 point = int.Parse(word.TrimEnd('*'));
318 }
319 if (i > 0) {
320 Require(previous > point, "ILLEGAL syntax");
321 for (int k = 0; k < repeat; ++k)
322 moves.Add(new Move(previous, point));
323 }
324 previous = point;
325 }
326 }
327 return moves;
328 }
329 static Play ChoosePlay(State state, string[] tokens, Plays plays) {
330 List<Move> moves = ParsePlay(tokens);
331 int side = state.Side;
332 int[,] final = (int[,])state.Pos.Clone();
333 HashSet<int> destinations = new HashSet<int>();
334 foreach (Move move in moves) {
335 --final[side, move.From];
336 ++final[side, move.To];
337 if (move.To > 0)
338 destinations.Add(move.To);
339 }
340 foreach (int t in destinations)
341 if (state.Pos[1 - side, 25 - t] == 1) {
342 final[1 - side, 25 - t] = 0;
343 ++final[1 - side, 25];
344 }
345 for (int p = 1; p <= 25; ++p)
346 Require(final[side, p] >= 0, "ILLEGAL no checker");
347 Require(final[side, 25] == 0 || moves.All(m => m.From == 25), "ILLEGAL bar");
348 foreach (int t in destinations)
349 Require(state.Pos[1 - side, 25 - t] < 2, "ILLEGAL blocked");
350 if (moves.Any(m => m.To == 0)) {
351 Require(final[side, 25] == 0, "ILLEGAL bear-off");
352 for (int p = 7; p <= 24; ++p)
353 Require(final[side, p] == 0, "ILLEGAL bear-off");
354 }
355 string id = PositionKey(final);
356 Play? legal = plays.Legal.Find(p => p.Id == id);
357 if (legal != null)
358 return legal;
359 List<Play> possible = plays.All.Where(p => p.Id == id).ToList();
360 Require(possible.Count > 0, "ILLEGAL impossible");
361 throw new InvalidOperationException(possible.Any(p => p.Count < plays.Maximum)
362 ? "ILLEGAL too few dice"
363 : "ILLEGAL larger die required");
364 }

367 static Simulation Simulate(State original, RandomStream rng, string[] strategies,
368 bool opening = false, int target = 0, int[]? scores = null,
369 bool crawford = false, bool verbose = false) {
370 State state = original.Copy();
371 int d1 = 0, d2 = 0;
372 if (opening) {
373 do {
374 d1 = rng.Die();
375 d2 = rng.Die();
376 } while (d1 == d2);
377 state.Side = d1 > d2 ? 0 : 1;
378 }
379 int first = state.Side, plays = 0;
380 for (int turn = 1;; ++turn) {
381 int x = state.Side;
382 if (!opening || turn != 1) {
383 bool forbidden = target > 0 && (crawford || scores![x] + state.Cube >= target);
384 string decision = CubeDecision(state, forbidden);
385 if (decision.StartsWith("double", StringComparison.Ordinal)) {
386 if (verbose)
387 Console.WriteLine($"{turn} {Color(x)} doubles {state.Cube * 2}");
388 if (decision == "double drop") {
389 if (verbose)
390 Console.WriteLine($"{turn} {Color(1 - x)} drops");
391 return new Simulation(EndGame(state, x, "drop"), plays, first);
392 }
393 if (verbose)
394 Console.WriteLine($"{turn} {Color(1 - x)} takes");
395 state.Cube *= 2;
396 state.Owner = 1 - x;
397 }
398 d1 = rng.Die();
399 d2 = rng.Die();
400 }
401 List<Play> legal = Enumerate(state.Pos, x, d1, d2).Legal;
402 Play play = strategies[x] switch { "FIRST" => legal[0],
403 "RANDOM" => legal[(int)(rng.Next() % legal.Count)],
404 _ => Ranked(legal, x)[0].Play };
405 if (verbose)
406 Console.WriteLine($"{turn} {Color(x)} {d1}-{d2} {play.Text}");
407 Result? result = ApplyPlay(state, play);
408 ++plays;
409 if (result.HasValue)
410 return new Simulation(result.Value, plays, first);
411 }
412 }
413 static bool Numeric(string? word, int low, int high) {
414 return word != null && word.Length > 0 && word.All(c => c >= '0' && c <= '9') &&
415 long.TryParse(word, out long n) && n >= low && n <= high;
416 }
417 static int [,]? ParsePosition(string[] args) {
418 if (args.Length != 4 || !Numeric(args[1], 0, 15) || !Numeric(args[2], 0, 15) ||
419 (args[3] != "W" && args[3] != "B"))
420 return null;
421 string[] points = args[0].Split(',');
422 if (points.Length != 24)
423 return null;
424 int[,] pos = new int[2, 26];
425 for (int i = 0; i < 24; ++i) {
426 if (!int.TryParse(points[i], out int n) || n < -15 || n > 15)
427 return null;
428 if (n > 0)
429 pos[0, i + 1] = n;
430 else
431 pos[1, 24 - i] = -n;
432 }
433 pos[0, 25] = int.Parse(args[1]);
434 pos[1, 25] = int.Parse(args[2]);
435 for (int x = 0; x < 2; ++x) {
436 int count = 0;
437 for (int p = 1; p <= 25; ++p)
438 count += pos[x, p];
439 pos[x, 0] = 15 - count;
440 if (pos[x, 0] < 0 || pos[x, 0] > 14)
441 return null;
442 }
443 return pos;
444 }
445 static void BoardLine(State state, bool top, int row = 0) {

450 if (row > 0) {

453 }
454 line.Append(value.PadLeft(3));

457 }

468 static bool ArgumentsValid(string command, string[] a) {
469 if (NoArguments.Contains(command))
470 return a.Length == 0;
471 string ? At(int i) => i < a.Length ? a[i] : null;
472 if (command == "POS")
473 return ParsePosition(a) != null;
474 if (new[] { "MOVES", "PLAY", "AUTO", "HINT" }.Contains(command)) {
475 if (!Numeric(At(0), 1, 6) || !Numeric(At(1), 1, 6))
476 return false;
477 if (command == "PLAY")
478 return a.Length >= 3;
479 if (command == "HINT")
480 return a.Length == 3 && Numeric(At(2), 1, 1000);
481 return a.Length == 2;
482 }
483 if (command == "RESIGN")
484 return a.Length == 1 && new[] { "single", "gammon", "backgammon" }.Contains(a[0]);
485 if (command == "UNDO" || command == "REDO")
486 return a.Length == 1 && Numeric(At(0), 1, 1000);
487 if (command == "GAME")
488 return a.Length == 3 && Numeric(At(0), 1, 2147483646) &&
489 a.Skip(1).All(s => new[] { "BEST", "FIRST", "RANDOM" }.Contains(s));
490 if (command == "MATCH" || command == "STATS")
491 return a.Length == 2 && Numeric(At(0), 1, command == "MATCH" ? 15 : 50) &&
492 Numeric(At(1), 1, 2147483646);
493 return false;
494 }

599 List<string> rolls = new List<string>();
600 for (int d1 = 6; d1 >= 1; --d1)
601 for (int d2 = d1; d2 >= 1; --d2) {
602 List<Play> legal = Enumerate(state.Pos, 1 - side, d1, d2).Legal;
603 List<int> hit =
604 blots.Where(p => legal.Any(play => play.Pos[side, p] == 0))
605 .ToList();
606 if (hit.Count > 0) {
607 int weight = d1 == d2 ? 1 : 2;
608 total += weight;
609 rolls.Add($"{d1}-{d2}");
610 foreach (int p in hit)
611 counts[p] += weight;
612 }
613 }
614 Console.WriteLine($"shots {total}/36" +
615 (rolls.Count == 0 ? "" : " " + string.Join(' ', rolls)));
616 foreach (int p in blots)
617 Console.WriteLine($"blot {p} {counts[p]}/36");
618 break;
619 }
620 case "CUBE":
621 Console.WriteLine("cube " + CubeDecision(state));
622 break;
623 case "DOUBLE":
624 Require(state.Owner != 1 - side && state.Cube != 64, "ERROR cube unavailable");
625 state.Pending = true;
626 redo.Clear();
627 Console.WriteLine($"OK double {state.Cube * 2}");
628 break;
629 case "TAKE":
630 case "DROP":
631 Require(state.Pending, "ERROR no double");
632 state.Pending = false;
633 redo.Clear();
634 if (command == "TAKE") {
635 state.Cube *= 2;
636 state.Owner = 1 - side;
637 Console.WriteLine($"OK take {state.Cube} {Color(state.Owner)}");
638 } else {
639 Console.WriteLine("OK drop");
640 Console.WriteLine(ResultLine(EndGame(state, side, "drop")));
641 }
642 break;
643 case "RESIGN":
644 redo.Clear();
645 Console.WriteLine("OK resign");
646 Console.WriteLine(ResultLine(EndGame(state, 1 - side, a[0])));
647 break;
648 case "SCORE":
649 Console.WriteLine($"score W {state.Score[0]} B {state.Score[1]}");
650 break;
651 case "HISTORY":
652 Console.WriteLine($"history {history.Count}");
653 for (int i = 0; i < history.Count; ++i)
654 Console.WriteLine($"{i + 1} {history[i].Record}");
655 break;
656 case "UNDO":
657 case "REDO": {
658 int n = int.Parse(a[0]);
659 if (command == "UNDO") {
660 Require(n <= history.Count, "ERROR cannot undo");
661 List<Entry> entries = history.GetRange(history.Count - n, n);
662 state = entries[0].Before.Copy();
663 history.RemoveRange(history.Count - n, n);
664 redo.InsertRange(0, entries);
665 } else {
666 Require(n <= redo.Count, "ERROR cannot redo");
667 List<Entry> entries = redo.GetRange(0, n);
668 state = entries[n - 1].After.Copy();
669 redo.RemoveRange(0, n);
670 history.AddRange(entries);
671 }
672 Console.WriteLine("OK");
673 break;
674 }
675 case "GAME": {
676 Simulation simulation = Simulate(state, new RandomStream(long.Parse(a[0])),
677 a.Skip(1).ToArray(), verbose: true);
678 Console.WriteLine(ResultLine(simulation.Result));
679 break;
680 }
681 case "MATCH":
682 case "STATS": {
683 int n = int.Parse(a[0]);
684 RandomStream rng = new RandomStream(long.Parse(a[1]));
685 int[] points = { 0, 0 };
686 int turns = 0, game = 0;
687 bool crawfordUsed = false;
688 string[] kinds = { "single", "gammon", "backgammon", "drop" };
689 Dictionary<string, int[]> outcomes =
690 kinds.ToDictionary(k => k, k => new int[2]);
691 while (command == "STATS" ? game < n : points.Max() < n) {
692 ++game;
693 bool crawford =
694 command == "MATCH" && !crawfordUsed && points.Max() == n - 1;
695 crawfordUsed |= crawford;
696 Simulation simulation =
697 Simulate(State.Initial(), rng, new[] { "BEST", "BEST" }, true,
698 command == "MATCH" ? n : 0, points, crawford);
699 Result result = simulation.Result;
700 points[result.Winner] += result.Points;
701 ++outcomes[result.Kind][result.Winner];
702 turns += simulation.Turns;
703 if (command == "MATCH")
704 Console.WriteLine(
705 $"game {game}{(crawford ? " crawford" : "")} start {Color(simulation.First)} turns {simulation.Turns} {ResultLine(result)} score {points[0]}-{points[1]}");
706 }
707 if (command == "MATCH")
708 Console.WriteLine(
709 $"match {Color(points[0] >= n ? 0 : 1)} {points[0]}-{points[1]}");
710 else {
711 foreach (string kind in kinds)
712 Console.WriteLine(
713 $"{kind} W {outcomes[kind][0]} B {outcomes[kind][1]}");
714 Console.WriteLine($"points W {points[0]} B {points[1]}");
715 Console.WriteLine($"turns {turns}");
716 }
717 break;
718 }
719 }
720 } catch (InvalidOperationException error) {
721 Console.WriteLine(error.Message);
722 }
723 }
724 }
