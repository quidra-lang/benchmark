69fn notation(moves: &[Move], hits: u32) -> (Vec<Move>, String) {
70 let mut key = moves.to_vec();
71 key.sort_by(|a, b| b.cmp(a));
72 loop {
73 let mut pair = None;
74 for (i, &(_, t)) in key.iter().enumerate() {
75 if t == 0 || hits & (1u32 << t) != 0 { continue; }
76 if let Some(j) = key.iter().position(|&(s, _)| s == t) {
77 pair = Some((i, j));
78 break;
79 }
80 }
81 let Some((ai, bi)) = pair else { break; };
82 let merged = (key[ai].0, key[bi].1);
83 key.remove(ai.max(bi));
84 key.remove(ai.min(bi));
85 key.push(merged);
86 key.sort_by(|a, b| b.cmp(a));
87 }
88 let mut parts = Vec::new();
89 let mut marked = 0u32;
90 let mut i = 0;
91 while i < key.len() {
92 let (s, t) = key[i];
93 let mut j = i + 1;
94 while j < key.len() && key[j] == key[i] { j += 1; }
95 let from = if s == 25 { "bar".to_owned() } else { s.to_string() };
96 let to = if t == 0 { "off".to_owned() } else { t.to_string() };
97 let mut text = format!("{from}/{to}");
98 if hits & (1u32 << t) != 0 && marked & (1u32 << t) == 0 {
99 text.push('*');
100 marked |= 1u32 << t;
101 }
102 if j - i > 1 { text.push_str(&format!("({})", j - i)); }
103 parts.push(text);
104 i = j;
105 }
106 let text = if parts.is_empty() { "none".to_owned() } else { parts.join(" ") };
107 (key, text)
108}
109fn visit(pos: Position, side: usize, dice: &[i32], moves: &mut Vec<Move>, hits: u32,
110 first_die: i32, all: &mut Vec<Play>) {
111 let (key, text) = notation(moves, hits);
112 all.push(Play { pos, key, text, count: moves.len(), first_die });
113 let mut tried = [false; 7];
114 for (di, &die) in dice.iter().enumerate() {
115 if tried[die as usize] { continue; }
116 tried[die as usize] = true;
117 for s in 1..=25 {
118 if pos[side][s] == 0 || (pos[side][25] > 0 && s != 25) { continue; }
119 let t = (s as i32 - die).max(0) as usize;
120 if t > 0 {
121 if pos[1 - side][25 - t] >= 2 { continue; }
122 } else {
123 if pos[side][25] > 0 || (7..=24).any(|p| pos[side][p] > 0) { continue; }
124 if (s as i32) < die && (s + 1..=6).any(|p| pos[side][p] > 0) { continue; }
125 }
126 let mut next = pos;
127 next[side][s] -= 1;
128 next[side][t] += 1;
129 let mut next_hits = hits;
130 if t > 0 && pos[1 - side][25 - t] == 1 {
131 next[1 - side][25 - t] = 0;
132 next[1 - side][25] += 1;
133 next_hits |= 1u32 << t;
134 }
135 let mut rest = dice.to_vec();
136 rest.remove(di);
137 let first = if moves.is_empty() { die } else { first_die };
138 moves.push((s, t));
139 visit(next, side, &rest, moves, next_hits, first, all);
140 moves.pop();
141 }
142 }
143}
144fn enumerate(pos: Position, side: usize, d1: i32, d2: i32) -> Plays {
145 let dice = if d1 == d2 { vec![d1; 4] } else { vec![d1, d2] };
146 let mut all = Vec::new();
147 visit(pos, side, &dice, &mut Vec::new(), 0, 0, &mut all);
148 let maximum = all.iter().map(|p| p.count).max().unwrap();
149 let larger = maximum == 1 && d1 != d2 && all.iter().any(|p| p.count == 1 && p.first_die == d1.max(d2));
150 let mut seen = HashSet::new();
151 let mut legal: Vec<Play> = all.iter().filter(|p| p.count == maximum && (!larger || p.first_die == d1.max(d2)))
152 .filter(|p| seen.insert(p.pos)).cloned().collect();
153 legal.sort_by(|a, b| b.key.cmp(&a.key));
154 Plays { all, legal, maximum }
155}
156fn pips(own: &[i32; 26]) -> i32 {
157 (1..=25).map(|p| p as i32 * own[p]).sum()
158}
159fn evaluation(pos: Position, side: usize) -> [i32; 7] {
160 let own = pos[side];
161 let opponent = pos[1 - side];
162 let p = pips(&opponent) - pips(&own);
163 let (mut s, mut home, mut prime, mut anchors, mut run) = (0i32, 0i32, 0i32, 0i32, 0i32);
164 for point in 1..=24 {
165 if own[point] == 1 {
166 let mut distances = [false; 25];
167 for q in 0..point {
168 if opponent[25 - q] > 0 { distances[point - q] = true; }
169 }
170 for a in 1..=6 {
171 for b in 1..=6 {
172 if distances[a] || distances[b] || distances[a + b] || (a == b && (distances[3 * a] || distances[4 * a])) {
173 s += 1;
174 }
175 }
176 }
177 }
178 if own[point] >= 2 {
179 if point <= 6 { home += 1; }
180 if point >= 19 { anchors += 1; }
181 run += 1;
182 prime = prime.max(run);
183 } else { run = 0; }
184 }
185 let bar = opponent[25] - own[25];
186 [2 * p - s + 4 * home + 3 * prime + 3 * anchors + 6 * bar, p, s, home, prime, anchors, bar]
187}
188fn keith(own: &[i32; 26]) -> i32 {
189 pips(own) + 2 * (own[1] - 1).max(0) + (own[2] - 1).max(0) + (own[3] - 3).max(0)
190 + (4..=6).filter(|&p| own[p] == 0).count() as i32
191}
192fn contact(pos: Position) -> bool {
193 let w = if pos[0][25] > 0 { 25 } else { (1..=24).filter(|&p| pos[0][p] > 0).max().unwrap_or(0) };
194 let b = if pos[1][25] > 0 { 0 } else { 25 - (1..=24).filter(|&p| pos[1][p] > 0).max().unwrap_or(0) };
195 w > b
196}
197fn cube_decision(state: &State, forbidden: bool) -> &'static str {
198 let x = state.side;
199 if state.owner == Some(1 - x) || state.cube == 64 || forbidden { return "unavailable"; }
200 if contact(state.pos) {
201 let value = evaluation(state.pos, x)[0];
202 if value < 60 { return "no double"; }
203 return if value <= 100 { "double take" } else { "double drop" };
204 }
205 let value = 8 * keith(&state.pos[x]) - 7 * keith(&state.pos[1 - x]);
206 let limit = if state.owner.is_none() { 28 } else { 21 };
207 if value > limit { return "no double"; }
208 if value >= 14 { "double take" } else { "double drop" }
209}
210fn end_game(state: &mut State, winner: usize, kind: Option<&'static str>) -> Outcome {
211 let kind = kind.unwrap_or_else(|| {
212 let opponent = state.pos[1 - winner];
213 if opponent[0] > 0 { "single" }
214 else if opponent[25] > 0 || (19..=24).any(|p| opponent[p] > 0) { "backgammon" }
215 else { "gammon" }
216 });
217 let multiplier = match kind { "gammon" => 2, "backgammon" => 3, _ => 1 };
218 let points = state.cube * multiplier;
219 state.over = true;
220 state.pending = false;
221 state.score[winner] += points;
222 Outcome { winner, kind, points }
223}
224fn apply_play(state: &mut State, play: &Play) -> Option<Outcome> {
225 state.pos = play.pos;
226 if state.pos[state.side][0] == 15 {
227 let winner = state.side;
228 return Some(end_game(state, winner, None));
229 }
230 state.side = 1 - state.side;
231 None
232}
233fn ranked(legal: &[Play], side: usize) -> Vec<(i32, usize)> {
234 let mut values: Vec<_> = legal.iter().enumerate().map(|(i, p)| (evaluation(p.pos, side)[0], i)).collect();
235 values.sort_by(|a, b| b.0.cmp(&a.0).then_with(|| a.1.cmp(&b.1)));
236 values
237}
238fn parse_point(word: &str, first: bool) -> Result<usize, String> {
239 if first && word == "bar" { return Ok(25); }
240 if !first && word == "off" { return Ok(0); }
241 let word = if !first { word.strip_suffix('*').unwrap_or(word) } else { word };
242 let point = word.parse::<usize>().ok();
243 require(!word.is_empty() && !word.starts_with('0') && word.bytes().all(|b| b.is_ascii_digit()) && point.is_some_and(|n| (1..=24).contains(&n)), "ILLEGAL syntax")?;
244 Ok(point.unwrap())
245}
246fn parse_play(tokens: &[&str]) -> Result<Vec<Move>, String> {
247 if tokens == ["none"] { return Ok(Vec::new()); }
248 let mut moves = Vec::new();
249 for &token in tokens {
250 let (path, repeat) = if token.ends_with(')') {
251 let bytes = token.as_bytes();
252 require(bytes.len() >= 3 && bytes[bytes.len() - 3] == b'(' && (b'2'..=b'4').contains(&bytes[bytes.len() - 2]), "ILLEGAL syntax")?;
253 (&token[..token.len() - 3], (bytes[bytes.len() - 2] - b'0') as usize)
254 } else { (token, 1) };
255 let fields: Vec<_> = path.split('/').collect();
256 require(fields.len() >= 2, "ILLEGAL syntax")?;
257 let mut previous = parse_point(fields[0], true)?;
258 for word in &fields[1..] {
259 let point = parse_point(word, false)?;
260 require(previous > point, "ILLEGAL syntax")?;
261 for _ in 0..repeat { moves.push((previous, point)); }
262 previous = point;
263 }
264 }
265 Ok(moves)
266}
267fn choose_play(state: &State, tokens: &[&str], plays: &Plays) -> Result<Play, String> {
268 let moves = parse_play(tokens)?;
269 let side = state.side;
270 let mut final_pos = state.pos;
271 let mut destinations = [false; 25];
272 for &(s, t) in &moves {
273 final_pos[side][s] -= 1;
274 final_pos[side][t] += 1;
275 if t > 0 { destinations[t] = true; }
276 }
277 for t in 1..=24 {
278 if destinations[t] && state.pos[1 - side][25 - t] == 1 {
279 final_pos[1 - side][25 - t] = 0;
280 final_pos[1 - side][25] += 1;
281 }
282 }
283 require((1..=25).all(|p| final_pos[side][p] >= 0), "ILLEGAL no checker")?;
284 require(final_pos[side][25] == 0 || moves.iter().all(|&(s, _)| s == 25), "ILLEGAL bar")?;
285 require((1..=24).all(|t| !destinations[t] || state.pos[1 - side][25 - t] < 2), "ILLEGAL blocked")?;
286 if moves.iter().any(|&(_, t)| t == 0) {
287 require(final_pos[side][25] == 0 && (7..=24).all(|p| final_pos[side][p] == 0), "ILLEGAL bear-off")?;
288 }
289 if let Some(play) = plays.legal.iter().find(|p| p.pos == final_pos) { return Ok(play.clone()); }
290 let possible: Vec<_> = plays.all.iter().filter(|p| p.pos == final_pos).collect();
291 require(!possible.is_empty(), "ILLEGAL impossible")?;
292 let message = if possible.iter().any(|p| p.count < plays.maximum) { "ILLEGAL too few dice" } else { "ILLEGAL larger die required" };
293 Err(message.to_owned())
294}
295fn result_line(result: &Outcome) -> String {
296 format!("result {} {} {}", color(result.winner), result.kind, result.points)
297}
298fn simulate(mut state: State, rng: &mut RandomStream, strategies: [&str; 2], opening: bool,
299 target: i32, scores: [i32; 2], crawford: bool, verbose: bool) -> Simulation {
300 let (mut d1, mut d2) = (0, 0);
301 if opening {
302 loop {
303 d1 = rng.die(); d2 = rng.die();
304 if d1 != d2 { break; }
305 }
306 state.side = if d1 > d2 { 0 } else { 1 };
307 }
308 let first = state.side;
309 let mut plays = 0;
310 for turn in 1.. {
311 let x = state.side;
312 if !opening || turn != 1 {
313 let forbidden = target > 0 && (crawford || scores[x] + state.cube >= target);
314 let decision = cube_decision(&state, forbidden);
315 if decision.starts_with("double") {
316 if verbose { println!("{turn} {} doubles {}", color(x), state.cube * 2); }
317 if decision == "double drop" {
318 if verbose { println!("{turn} {} drops", color(1 - x)); }
319 return Simulation { outcome: end_game(&mut state, x, Some("drop")), turns: plays, first };
320 }
321 if verbose { println!("{turn} {} takes", color(1 - x)); }
322 state.cube *= 2;
323 state.owner = Some(1 - x);
324 }
325 d1 = rng.die(); d2 = rng.die();
326 }
327 let legal = enumerate(state.pos, x, d1, d2).legal;
328 let index = match strategies[x] {
329 "FIRST" => 0,
330 "RANDOM" => (rng.next() % legal.len() as i64) as usize,
331 _ => ranked(&legal, x)[0].1,
332 };
333 let play = &legal[index];
334 if verbose { println!("{turn} {} {d1}-{d2} {}", color(x), play.text); }
335 let outcome = apply_play(&mut state, play);
336 plays += 1;
337 if let Some(outcome) = outcome { return Simulation { outcome, turns: plays, first }; }
338 }
339 unreachable!()
340}
341fn numeric(word: Option<&&str>, low: i64, high: i64) -> bool {
342 word.is_some_and(|s| !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()) && s.parse::<i64>().is_ok_and(|n| n >= low && n <= high))
343}
344fn parse_position(args: &[&str]) -> Option<Position> {
345 if args.len() != 4 || !numeric(args.get(1), 0, 15) || !numeric(args.get(2), 0, 15) || !["W", "B"].contains(&args[3]) { return None; }
346 let points: Option<Vec<i32>> = args[0].split(',').map(|s| s.parse().ok()).collect();
347 let points = points?;
348 if points.len() != 24 { return None; }
349 let mut pos = [[0; 26]; 2];
350 for (i, &n) in points.iter().enumerate() {
351 if !(-15..=15).contains(&n) { return None; }
352 if n > 0 { pos[0][i + 1] = n; } else { pos[1][24 - i] = -n; }
353 }
354 pos[0][25] = args[1].parse().ok()?;
355 pos[1][25] = args[2].parse().ok()?;
356 for own in &mut pos {
357 own[0] = 15 - own.iter().sum::<i32>();
358 if !(0..=14).contains(&own[0]) { return None; }
359 }
360 Some(pos)
361}
362const NO_ARGUMENTS: &[&str] = &["NEW", "SHOW", "BOARD", "PIPS", "RACE", "EVAL", "SHOTS", "CUBE", "DOUBLE", "TAKE", "DROP", "SCORE", "HISTORY"];
363const OTHER: &[&str] = &["POS", "MOVES", "PLAY", "AUTO", "HINT", "RESIGN", "UNDO", "REDO", "GAME", "MATCH", "STATS"];
364const LIMITED: &[&str] = &["RACE", "MOVES", "PLAY", "AUTO", "HINT", "EVAL", "SHOTS", "CUBE", "DOUBLE", "TAKE", "DROP", "RESIGN", "GAME"];
365const PENDING_LIMITED: &[&str] = &["PLAY", "AUTO", "CUBE", "DOUBLE", "RESIGN", "GAME"];
366fn arguments_valid(command: &str, a: &[&str]) -> bool {
367 if NO_ARGUMENTS.contains(&command) { return a.is_empty(); }
368 match command {
369 "POS" => parse_position(a).is_some(),
370 "MOVES" | "PLAY" | "AUTO" | "HINT" => {
371 numeric(a.first(), 1, 6) && numeric(a.get(1), 1, 6) && match command {
372 "PLAY" => a.len() >= 3,
373 "HINT" => a.len() == 3 && numeric(a.get(2), 1, 1000),
374 _ => a.len() == 2,
375 }
376 }
377 "RESIGN" => a.len() == 1 && ["single", "gammon", "backgammon"].contains(&a[0]),
378 "UNDO" | "REDO" => a.len() == 1 && numeric(a.first(), 1, 1000),
379 "GAME" => a.len() == 3 && numeric(a.first(), 1, 2147483646) && a[1..].iter().all(|s| ["BEST", "FIRST", "RANDOM"].contains(s)),
380 "MATCH" | "STATS" => a.len() == 2 && numeric(a.first(), 1, if command == "MATCH" { 15 } else { 50 }) && numeric(a.get(1), 1, 2147483646),
381 _ => false,
382 }
383}
384fn board_line(pos: Position, top: bool, row: i32) {
385 let mut line = String::new();
386 for i in 1..=12 {
387 let p = if top { i + 12 } else { 13 - i };
388 let value = if row == 0 { p.to_string() } else {
389 let (w, b) = (pos[0][p], pos[1][25 - p]);
390 let n = w.max(b);
391 if row == 5 && n > 5 { n.to_string() }
392 else if n >= row { (if w > 0 { "W" } else { "B" }).to_owned() }
393 else { ".".to_owned() }
394 };
395 line.push_str(&format!("{value:>3}"));
396 if i == 6 { line.push_str(" |"); }
397 }
398 println!("{line}");
399}
400struct Engine { state: State, history: Vec<Entry>, redo: Vec<Entry> }
401impl Engine {
402 fn run(&mut self, line: &str) -> Result<(), String> {
403 let words: Vec<_> = line.split_whitespace().collect();
404 let command = words[0];
405 let a = &words[1..];
406 require(NO_ARGUMENTS.contains(&command) || OTHER.contains(&command), "ERROR unknown command")?;
407 require(arguments_valid(command, a), if command == "POS" { "ERROR bad position" } else { "ERROR bad arguments" })?;
408 require(!self.state.over || !LIMITED.contains(&command), "ERROR game over")?;
409 require(!self.state.pending || !PENDING_LIMITED.contains(&command), "ERROR double pending")?;
410 let side = self.state.side;
411 match command {
412 "NEW" | "POS" => {
413 self.state = State::initial(self.state.score);
414 if command == "POS" {
415 self.state.pos = parse_position(a).unwrap();
416 self.state.side = if a[3] == "W" { 0 } else { 1 };
417 }
418 self.history.clear(); self.redo.clear(); println!("OK");
419 }
420 "SHOW" => {
421 let points: Vec<_> = (1..=24).map(|p| (self.state.pos[0][p] - self.state.pos[1][25 - p]).to_string()).collect();
422 println!("POS {} {} {} {}", points.join(","), self.state.pos[0][25], self.state.pos[1][25], color(side));
423 }
424 "BOARD" => {
425 board_line(self.state.pos, true, 0);
426 for row in 1..=5 { board_line(self.state.pos, true, row); }
427 let owner = self.state.owner.map(color).unwrap_or("center");
428 let status = if self.state.over { "game over".to_owned() } else { format!("{} {}", color(side), if self.state.pending { "doubles" } else { "to move" }) };
429 println!("bar W{} B{} | off W{} B{} | cube {} {} | {}", self.state.pos[0][25], self.state.pos[1][25], self.state.pos[0][0], self.state.pos[1][0], self.state.cube, owner, status);
430 for row in (1..=5).rev() { board_line(self.state.pos, false, row); }
431 board_line(self.state.pos, false, 0);
432 }
433 "PIPS" => println!("pips W {} B {}", pips(&self.state.pos[0]), pips(&self.state.pos[1])),
434 "RACE" => {
435 if contact(self.state.pos) { println!("contact"); }
436 else { println!("race W {} B {}", keith(&self.state.pos[0]), keith(&self.state.pos[1])); }
437 }
438 "EVAL" => {
439 let values = evaluation(self.state.pos, side);
440 let labels = ["eval", "pips", "blots", "home", "prime", "anchors", "bar"];
441 println!("{}", labels.iter().zip(values).map(|(s, v)| format!("{s} {v}")).collect::<Vec<_>>().join(" "));
442 }
443 "MOVES" | "PLAY" | "AUTO" | "HINT" => {
444 let d1 = a[0].parse().unwrap(); let d2 = a[1].parse().unwrap();
445 let plays = enumerate(self.state.pos, side, d1, d2);
446 if command == "MOVES" {
447 for play in &plays.legal { println!("{}", play.text); }
448 println!("total {}", plays.legal.len());
449 } else if command == "HINT" {
450 let count = a[2].parse::<usize>().unwrap();
451 for (value, index) in ranked(&plays.legal, side).into_iter().take(count) { println!("{value} {}", plays.legal[index].text); }
452 } else {
453 let play = if command == "AUTO" { plays.legal[ranked(&plays.legal, side)[0].1].clone() } else { choose_play(&self.state, &a[2..], &plays)? };
454 let value = evaluation(play.pos, side)[0];
455 let before = self.state.clone();
456 let result = apply_play(&mut self.state, &play);
457 let suffix = if command == "AUTO" { format!(" eval {value}") } else { String::new() };
458 println!("OK {}{suffix}", play.text);
459 if let Some(result) = result { println!("{}", result_line(&result)); }
460 self.history.push(Entry { record: format!("{} {d1}-{d2} {}", color(side), play.text), before, after: self.state.clone() });
461 self.redo.clear();
462 }
463 }
464 "SHOTS" => {
465 let blots: Vec<_> = (1..=24).rev().filter(|&p| self.state.pos[side][p] == 1).collect();
466 let mut counts = [0; 25];
467 let mut total = 0;
468 let mut rolls = Vec::new();
469 for d1 in (1..=6).rev() {
470 for d2 in (1..=d1).rev() {
471 let legal = enumerate(self.state.pos, 1 - side, d1, d2).legal;
472 let hit: Vec<_> = blots.iter().copied().filter(|&p| legal.iter().any(|play| play.pos[side][p] == 0)).collect();
473 if !hit.is_empty() {
474 let weight = if d1 == d2 { 1 } else { 2 };
475 total += weight;
476 rolls.push(format!("{d1}-{d2}"));
477 for p in hit { counts[p] += weight; }
478 }
479 }
480 }
481 let suffix = if rolls.is_empty() { String::new() } else { format!(" {}", rolls.join(" ")) };
482 println!("shots {total}/36{suffix}");
483 for p in blots { println!("blot {p} {}/36", counts[p]); }
484 }
485 "CUBE" => println!("cube {}", cube_decision(&self.state, false)),
486 "DOUBLE" => {
487 require(self.state.owner != Some(1 - side) && self.state.cube != 64, "ERROR cube unavailable")?;
488 self.state.pending = true;
489 self.redo.clear();
490 println!("OK double {}", self.state.cube * 2);
491 }
492 "TAKE" | "DROP" => {
493 require(self.state.pending, "ERROR no double")?;
494 self.state.pending = false;
495 self.redo.clear();
496 if command == "TAKE" {
497 self.state.cube *= 2; self.state.owner = Some(1 - side);
498 println!("OK take {} {}", self.state.cube, color(1 - side));
499 } else {
500 println!("OK drop");
501 let result = end_game(&mut self.state, side, Some("drop"));
502 println!("{}", result_line(&result));
503 }
504 }
505 "RESIGN" => {
506 let kind = match a[0] { "single" => "single", "gammon" => "gammon", _ => "backgammon" };
507 self.redo.clear(); println!("OK resign");
508 let result = end_game(&mut self.state, 1 - side, Some(kind));
509 println!("{}", result_line(&result));
510 }
511 "SCORE" => println!("score W {} B {}", self.state.score[0], self.state.score[1]),
512 "HISTORY" => {
513 println!("history {}", self.history.len());
514 for (i, entry) in self.history.iter().enumerate() { println!("{} {}", i + 1, entry.record); }
515 }
516 "UNDO" | "REDO" => {
517 let count = a[0].parse::<usize>().unwrap();
518 if command == "UNDO" {
519 require(count <= self.history.len(), "ERROR cannot undo")?;
520 let entries = self.history.split_off(self.history.len() - count);
521 self.state = entries[0].before.clone();
522 self.redo.splice(0..0, entries);
523 } else {
524 require(count <= self.redo.len(), "ERROR cannot redo")?;
525 let entries: Vec<_> = self.redo.drain(..count).collect();
526 self.state = entries[count - 1].after.clone();
527 self.history.extend(entries);
528 }
529 println!("OK");
530 }
531 "GAME" => {
532 let mut rng = RandomStream(a[0].parse().unwrap());
533 let result = simulate(self.state.clone(), &mut rng, [a[1], a[2]], false, 0, [0, 0], false, true);
534 println!("{}", result_line(&result.outcome));
535 }
536 "MATCH" | "STATS" => {
537 let n = a[0].parse::<i32>().unwrap();
538 let mut rng = RandomStream(a[1].parse().unwrap());
539 let mut points = [0i32, 0i32];
540 let (mut turns, mut game, mut crawford_used) = (0, 0, false);
541 let kinds = ["single", "gammon", "backgammon", "drop"];
542 let mut outcomes = [[0; 2]; 4];
543 while if command == "STATS" { game < n } else { points[0].max(points[1]) < n } {
544 game += 1;
545 let crawford = command == "MATCH" && !crawford_used && points[0].max(points[1]) == n - 1;
546 crawford_used |= crawford;
547 let result = simulate(State::initial([0, 0]), &mut rng, ["BEST", "BEST"], true, if command == "MATCH" { n } else { 0 }, points, crawford, false);
548 points[result.outcome.winner] += result.outcome.points;
549 let kind = kinds.iter().position(|&k| k == result.outcome.kind).unwrap();
550 outcomes[kind][result.outcome.winner] += 1;
551 turns += result.turns;
552 if command == "MATCH" {
553 println!("game {game}{} start {} turns {} {} score {}-{}", if crawford { " crawford" } else { "" }, color(result.first), result.turns, result_line(&result.outcome), points[0], points[1]);
554 }
555 }
556 if command == "MATCH" {
557 println!("match {} {}-{}", color(if points[0] >= n { 0 } else { 1 }), points[0], points[1]);
558 } else {
559 for (i, kind) in kinds.iter().enumerate() { println!("{kind} W {} B {}", outcomes[i][0], outcomes[i][1]); }
560 println!("points W {} B {}", points[0], points[1]);
561 println!("turns {turns}");
562 }
563 }
564 _ => unreachable!(),
565 }
566 Ok(())
567 }
568}
569fn main() {
570 let mut input = String::new();
571 io::stdin().read_to_string(&mut input).unwrap();
572 let mut lines = input.lines();
573 let count = lines.next().unwrap().parse::<usize>().unwrap();
574 let mut engine = Engine { state: State::initial([0, 0]), history: Vec::new(), redo: Vec::new() };
575 for line in lines.take(count) {
576 if let Err(message) = engine.run(line) { println!("{message}"); }
577 }
578}
