101func pips(pos Position, side int) int {
102 total := 0
103 for p := 1; p <= 25; p++ {
104 total += p * pos[side][p]
105 }
106 return total
107}

109func keith(pos Position, side int) int {
110 c := pos[side]
111 total := pips(pos, side) + 2*max(c[1]-1, 0) + max(c[2]-1, 0) + max(c[3]-3, 0)
112 for p := 4; p <= 6; p++ {
113 if c[p] == 0 {
114 total++
115 }
116 }
117 return total
118}

120func contact(pos Position) bool {
121 white, black := 0, 25
122 if pos[0][25] > 0 {
123 white = 25
124 }
125 if pos[1][25] > 0 {
126 black = 0
127 }
128 for p := 1; p <= 24; p++ {
129 if pos[0][p] > 0 {
130 white = max(white, p)
131 }
132 if pos[1][p] > 0 {
133 black = min(black, 25-p)
134 }
135 }
136 return white > black
137}

139func evaluate(pos Position, side int) Evaluation {
140 other := 1 - side
141 value := Evaluation{P: pips(pos, other) - pips(pos, side), G: pos[other][25] - pos[side][25]}
142 run := 0
143 for p := 1; p <= 24; p++ {
144 if pos[side][p] >= 2 {
145 run++
146 value.R = max(value.R, run)
147 if p <= 6 {
148 value.H++
149 }
150 if p >= 19 {
151 value.A++
152 }
153 } else {
154 run = 0
155 }
156 if pos[side][p] == 1 {
157 distances := [25]bool{}
158 for q := 0; q < p; q++ {
159 if pos[other][25-q] > 0 {
160 distances[p-q] = true
161 }
162 }
163 for a := 1; a <= 6; a++ {
164 for b := 1; b <= 6; b++ {
165 if distances[a] || distances[b] || distances[a+b] || a == b && (distances[3*a] || distances[4*a]) {
166 value.S++
167 }
168 }
169 }
170 }
171 }
172 value.E = 2*value.P - value.S + 4*value.H + 3*value.R + 3*value.A + 6*value.G
173 return value
174}

176func cubeDecision(s State, prohibited bool) string {
177 if s.Owner == 1-s.Side || s.Cube == 64 || prohibited {
178 return "unavailable"
179 }
180 if contact(s.Position) {
181 e := evaluate(s.Position, s.Side).E
182 if e < 60 {
183 return "no double"
184 }
185 if e <= 100 {
186 return "double take"
187 }
188 return "double drop"
189 }
190 v := 8*keith(s.Position, s.Side) - 7*keith(s.Position, 1-s.Side)
191 limit := 21
192 if s.Owner < 0 {
193 limit = 28
194 }
195 if v > limit {
196 return "no double"
197 }
198 if v >= 14 {
199 return "double take"
200 }
201 return "double drop"
202}

204func submoves(pos Position, side, die int) []Move {
205 sources := []int{}
206 if pos[side][25] > 0 {
207 sources = append(sources, 25)
208 } else {
209 for p := 24; p >= 1; p-- {
210 if pos[side][p] > 0 {
211 sources = append(sources, p)
212 }
213 }
214 }
215 home := pos[side][25] == 0
216 for p := 7; p <= 24; p++ {
217 if pos[side][p] > 0 {
218 home = false
219 }
220 }
221 moves := []Move{}
222 for _, from := range sources {
223 to := from - die
224 if to >= 1 {
225 if pos[1-side][25-to] < 2 {
226 moves = append(moves, Move{from, to})
227 }
228 } else if home {
229 allowed := true
230 if to < 0 {
231 for p := from + 1; p <= 6; p++ {
232 if pos[side][p] > 0 {
233 allowed = false
234 }
235 }
236 }
237 if allowed {
238 moves = append(moves, Move{from, 0})
239 }
240 }
241 }
242 return moves
243}

245func applySubmove(pos Position, side int, move Move) (Position, bool) {
246 pos[side][move.From]--
247 pos[side][move.To]++
248 hit := move.To > 0 && pos[1-side][25-move.To] == 1
249 if hit {
250 pos[1-side][25-move.To] = 0
251 pos[1-side][25]++
252 }
253 return pos, hit
254}

256func moveLess(a, b Move) bool {
257 if a.From != b.From {
258 return a.From > b.From
259 }
260 return a.To > b.To
261}

263func canonical(moves []Move, hits uint32) ([]Move, string) {
264 key := append([]Move{}, moves...)
265 sort.Slice(key, func(i, j int) bool {
266 return moveLess(key[i], key[j])
267 })
268 for {
269 first, second := -1, -1
270 for i, a := range key {
271 if a.To == 0 || hits&(1<<a.To) != 0 {
272 continue
273 }
274 for j, b := range key {
275 if b.From == a.To {
276 first = i
277 second = j
278 break
279 }
280 }
281 if first >= 0 {
282 break
283 }
284 }
285 if first < 0 {
286 break
287 }
288 combined := Move{key[first].From, key[second].To}
289 next := []Move{}
290 for i, pair := range key {
291 if i != first && i != second {
292 next = append(next, pair)
293 }
294 }
295 key = append(next, combined)
296 sort.Slice(key, func(i, j int) bool {
297 return moveLess(key[i], key[j])
298 })
299 }
300 parts := []string{}
301 marked := map[int]bool{}
302 for i := 0; i < len(key); {
303 pair := key[i]
304 j := i + 1
305 for j < len(key) && key[j] == pair {
306 j++
307 }
308 from, to := strconv.Itoa(pair.From), strconv.Itoa(pair.To)
309 if pair.From == 25 {
310 from = "bar"
311 }
312 if pair.To == 0 {
313 to = "off"
314 }
315 part := from + "/" + to
316 if hits&(1<<pair.To) != 0 && !marked[pair.To] {
317 part += "*"
318 marked[pair.To] = true
319 }
320 if j-i >= 2 {
321 part += fmt.Sprintf("(%d)", j-i)
322 }
323 parts = append(parts, part)
324 i = j
325 }
326 if len(parts) == 0 {
327 return key, "none"
328 }
329 return key, strings.Join(parts, " ")
330}

332func keyLess(a, b []Move) bool {
333 for i := 0; i < len(a) && i < len(b); i++ {
334 if a[i] != b[i] {
335 return moveLess(a[i], b[i])
336 }
337 }
338 return len(a) > len(b)
339}

341func enumerate(pos Position, side, d1, d2 int) ([]Play, []Play, int) {
342 dice := []int{d1, d2}
343 if d1 == d2 {
344 dice = []int{d1, d1, d1, d1}
345 }
346 possible := []Play{}
347 maximum := 0
348 var search func(Position, []int, []Move, uint32, int)
349 search = func(current Position, left []int, path []Move, hits uint32, first int) {
350 copied := append([]Move{}, path...)
351 possible = append(possible, Play{Position: current, Moves: copied, Hits: hits, Die: first})
352 maximum = max(maximum, len(path))
353 seenDie := map[int]bool{}
354 for i, die := range left {
355 if seenDie[die] {
356 continue
357 }
358 seenDie[die] = true
359 rest := append([]int{}, left[:i]...)
360 rest = append(rest, left[i+1:]...)
361 for _, move := range submoves(current, side, die) {
362 next, hit := applySubmove(current, side, move)
363 mask := hits
364 if hit {
365 mask |= 1 << move.To
366 }
367 used := first
368 if len(path) == 0 {
369 used = die
370 }
371 search(next, rest, append(copied, move), mask, used)
372 }
373 }
374 }
375 search(pos, dice, nil, 0, 0)
376 larger := false
377 if maximum == 1 && d1 != d2 {
378 for _, play := range possible {
379 if len(play.Moves) == 1 && play.Die == max(d1, d2) {
380 larger = true
381 }
382 }
383 }
384 legal := []Play{}
385 positions := map[Position]bool{}
386 for _, play := range possible {
387 if len(play.Moves) != maximum || larger && play.Die != max(d1, d2) || positions[play.Position] {
388 continue
389 }
390 play.Key, play.Notation = canonical(play.Moves, play.Hits)
391 legal = append(legal, play)
392 positions[play.Position] = true
393 }
394 sort.Slice(legal, func(i, j int) bool {
395 return keyLess(legal[i].Key, legal[j].Key)
396 })
397 return legal, possible, maximum
398}

400func best(plays []Play, side int) (Play, int) {
401 chosen := plays[0]
402 score := evaluate(chosen.Position, side).E
403 for _, play := range plays[1:] {
404 value := evaluate(play.Position, side).E
405 if value > score {
406 chosen, score = play, value
407 }
408 }
409 return chosen, score
410}

430func bearResult(s *State, winner int) Result {
431 other := 1 - winner
432 kind := "single"
433 if s.Position[other][0] == 0 {
434 kind = "gammon"
435 back := s.Position[other][25] > 0
436 for p := 19; p <= 24; p++ {
437 back = back || s.Position[other][p] > 0
438 }
439 if back {
440 kind = "backgammon"
441 }
442 }
443 return finish(s, winner, kind)
444}

469func parsePlay(words []string) ([]Move, bool) {
470 if len(words) == 1 && words[0] == "none" {
471 return nil, true
472 }
473 moves := []Move{}
474 for _, word := range words {
475 if !playElement.MatchString(word) {
476 return nil, false
477 }
478 count := 1
479 if strings.HasSuffix(word, ")") {
480 count = number(word[len(word)-2 : len(word)-1])
481 word = word[:len(word)-3]
482 }
483 parts := strings.Split(word, "/")
484 points := []int{}
485 for _, part := range parts {
486 point := 0
487 if part == "bar" {
488 point = 25
489 } else if part != "off" {
490 point = number(strings.TrimSuffix(part, "*"))
491 }
492 points = append(points, point)
493 }
494 for i := 0; i+1 < len(points); i++ {
495 if points[i] <= points[i+1] {
496 return nil, false
497 }
498 for j := 0; j < count; j++ {
499 moves = append(moves, Move{points[i], points[i+1]})
500 }
501 }
502 }
503 return moves, true
504}

506func validatePlay(pos Position, side int, moves []Move, legal, possible []Play, maximum int) (Play, string) {
507 final := pos
508 destinations := map[int]bool{}
509 nonbar, bear := false, false
510 for _, move := range moves {
511 final[side][move.From]--
512 final[side][move.To]++
513 destinations[move.To] = true
514 nonbar = nonbar || move.From != 25
515 bear = bear || move.To == 0
516 }
517 for to := range destinations {
518 if to > 0 && pos[1-side][25-to] == 1 {
519 final[1-side][25-to] = 0
520 final[1-side][25]++
521 }
522 }
523 for p := 1; p <= 25; p++ {
524 if final[side][p] < 0 {
525 return Play{}, "no checker"
526 }
527 }
528 if final[side][25] > 0 && nonbar {
529 return Play{}, "bar"
530 }
531 for to := range destinations {
532 if to > 0 && pos[1-side][25-to] >= 2 {
533 return Play{}, "blocked"
534 }
535 }
536 if bear {
537 if final[side][25] > 0 {
538 return Play{}, "bear-off"
539 }
540 for p := 7; p <= 24; p++ {
541 if final[side][p] > 0 {
542 return Play{}, "bear-off"
543 }
544 }
545 }
546 for _, play := range legal {
547 if play.Position == final {
548 return play, ""
549 }
550 }
551 matched, short := false, false
552 for _, play := range possible {
553 if play.Position == final {
554 matched = true
555 short = short || len(play.Moves) < maximum
556 }
557 }
558 if short {
559 return Play{}, "too few dice"
560 }
561 if matched {
562 return Play{}, "larger die required"
563 }
564 return Play{}, "impossible"
565}

567func simulation(s State, rng *Random, strategies [2]string, opening bool, target int, score [2]int, crawford, verbose bool) Result {
568 firstDice := [2]int{}
569 if opening {
570 for {
571 firstDice = [2]int{rng.die(), rng.die()}
572 if firstDice[0] != firstDice[1] {
573 break
574 }
575 }
576 s.Side = 0
577 if firstDice[1] > firstDice[0] {
578 s.Side = 1
579 }
580 }
581 start, turns := s.Side, 0
582 for t := 1; ; t++ {
583 side, other := s.Side, 1-s.Side
584 d1, d2 := 0, 0
585 if opening && t == 1 {
586 d1, d2 = firstDice[0], firstDice[1]
587 } else {
588 prohibited := target > 0 && (crawford || score[side]+s.Cube >= target)
589 decision := cubeDecision(s, prohibited)
590 if decision == "double take" || decision == "double drop" {
591 if verbose {
592 emit(fmt.Sprintf("%d %s doubles %d", t, color(side), s.Cube*2))
593 }
594 if decision == "double drop" {
595 if verbose {
596 emit(fmt.Sprintf("%d %s drops", t, color(other)))
597 }
598 result := finish(&s, side, "drop")
599 result.Turns = turns
600 result.Start = start
601 return result
602 }
603 s.Cube *= 2
604 s.Owner = other
605 if verbose {
606 emit(fmt.Sprintf("%d %s takes", t, color(other)))
607 }
608 }
609 d1, d2 = rng.die(), rng.die()
610 }
611 plays, _, _ := enumerate(s.Position, side, d1, d2)
612 chosen := plays[0]
613 switch strategies[side] {
614 case "BEST":
615 chosen, _ = best(plays, side)
616 case "RANDOM":
617 chosen = plays[int(rng.next()%int64(len(plays)))]
618 }
619 s.Position = chosen.Position
620 turns++
621 if verbose {
622 emit(fmt.Sprintf("%d %s %d-%d %s", t, color(side), d1, d2, chosen.Notation))
623 }
624 if s.Position[side][0] == 15 {
625 result := bearResult(&s, side)
626 result.Turns = turns
627 result.Start = start
628 return result
629 }
630 s.Side = other
631 }
632}

634func (e *Engine) board() {
635 s := e.State
636 format := func(values []string) string {
637 line := ""
638 for i, value := range values {
639 if i == 6 {
640 line += " |"
641 }
642 line += fmt.Sprintf("%3s", value)
643 }
644 return line
645 }
646 for half := 0; half < 2; half++ {
647 points := []int{}
648 for i := 0; i < 12; i++ {
649 if half == 0 {
650 points = append(points, 13+i)
651 } else {
652 points = append(points, 12-i)
653 }
654 }
655 labels := []string{}
656 for _, p := range points {
657 labels = append(labels, strconv.Itoa(p))
658 }
659 if half == 0 {
660 emit(format(labels))
661 }
662 for row := 1; row <= 5; row++ {
663 level := row
664 if half == 1 {
665 level = 6 - row
666 }
667 fields := []string{}
668 for _, p := range points {
669 count, letter := s.Position[0][p], "W"
670 if count == 0 {
671 count = s.Position[1][25-p]
672 letter = "B"
673 }
674 value := "."
675 if count >= level {
676 value = letter
677 }
678 if level == 5 && count > 5 {
679 value = strconv.Itoa(count)
680 }
681 fields = append(fields, value)
682 }
683 emit(format(fields))
684 }
685 if half == 0 {
686 status := color(s.Side) + " to move"
687 if s.Over {
688 status = "game over"
689 } else if s.Pending {
690 status = color(s.Side) + " doubles"
691 }
692 emit(fmt.Sprintf("bar W%d B%d | off W%d B%d | cube %d %s | %s", s.Position[0][25], s.Position[1][25], s.Position[0][0], s.Position[1][0], s.Cube, owner(s), status))
693 } else {
694 emit(format(labels))
695 }
696 }
697}

699func (e *Engine) execute(words []string) {
700 cmd, a := words[0], words[1:]
701 noargs := " NEW SHOW BOARD PIPS RACE EVAL SHOTS CUBE DOUBLE TAKE DROP SCORE HISTORY "
702 known := strings.Contains(noargs, " "+cmd+" ") || strings.Contains(" POS MOVES PLAY AUTO HINT RESIGN UNDO REDO GAME MATCH STATS ", " "+cmd+" ")
703 if !known {
704 emit("ERROR unknown command")
705 return
706 }
707 valid := true
708 rangeArg := func(at, low, high int) bool {
709 return at < len(a) && number(a[at]) >= low && number(a[at]) <= high
710 }
711 if strings.Contains(noargs, " "+cmd+" ") {
712 valid = len(a) == 0
713 }
714 switch cmd {
715 case "POS":
716 valid = len(a) == 4
717 case "MOVES", "AUTO":
718 valid = len(a) == 2 && rangeArg(0, 1, 6) && rangeArg(1, 1, 6)
719 case "PLAY":
720 valid = len(a) >= 3 && rangeArg(0, 1, 6) && rangeArg(1, 1, 6)
721 case "HINT":
722 valid = len(a) == 3 && rangeArg(0, 1, 6) && rangeArg(1, 1, 6) && rangeArg(2, 1, 1000)
723 case "RESIGN":
724 valid = len(a) == 1 && (a[0] == "single" || a[0] == "gammon" || a[0] == "backgammon")
725 case "UNDO", "REDO":
726 valid = len(a) == 1 && rangeArg(0, 1, 1000)
727 case "GAME":
728 valid = len(a) == 3 && rangeArg(0, 1, 2147483646)
729 if valid {
730 for _, strategy := range a[1:] {
731 valid = valid && (strategy == "BEST" || strategy == "FIRST" || strategy == "RANDOM")
732 }
733 }
734 case "MATCH":
735 valid = len(a) == 2 && rangeArg(0, 1, 15) && rangeArg(1, 1, 2147483646)
736 case "STATS":
737 valid = len(a) == 2 && rangeArg(0, 1, 50) && rangeArg(1, 1, 2147483646)
738 }
739 if cmd == "POS" && valid {
740 points := strings.Split(a[0], ",")
741 valid = len(points) == 24 && rangeArg(1, 0, 15) && rangeArg(2, 0, 15) && (a[3] == "W" || a[3] == "B")
742 s := initial()
743 s.Position = Position{}
744 s.Score = e.State.Score
745 if valid {
746 s.Position[0][25] = number(a[1])
747 s.Position[1][25] = number(a[2])
748 for i, text := range points {
749 value := number(text)
750 if value > 0 {
751 s.Position[0][i+1] = value
752 } else if value < 0 {
753 s.Position[1][24-i] = -value
754 }
755 }
756 for side := 0; side < 2; side++ {
757 total := 0
758 for p := 1; p <= 25; p++ {
759 total += s.Position[side][p]
760 }
761 s.Position[side][0] = 15 - total
762 valid = valid && s.Position[side][0] >= 0 && s.Position[side][0] <= 14
763 }
764 if a[3] == "B" {
765 s.Side = 1
766 }
767 }
768 if valid {
769 e.State = s
770 e.History = nil
771 e.Redo = nil
772 emit("OK")
773 return
774 }
775 }
776 if !valid {
777 if cmd == "POS" {
778 emit("ERROR bad position")
779 } else {
780 emit("ERROR bad arguments")
781 }
782 return
783 }
784 after := " RACE MOVES PLAY AUTO HINT EVAL SHOTS CUBE DOUBLE TAKE DROP RESIGN GAME "
785 pending := " PLAY AUTO CUBE DOUBLE RESIGN GAME "
786 if e.State.Over && strings.Contains(after, " "+cmd+" ") {
787 emit("ERROR game over")
788 return
789 }
790 if e.State.Pending && strings.Contains(pending, " "+cmd+" ") {
791 emit("ERROR double pending")
792 return
793 }
794 s := &e.State
795 switch cmd {
796 case "NEW":
797 score := s.Score
798 e.State = initial()
799 e.State.Score = score
800 e.History = nil
801 e.Redo = nil
802 emit("OK")
803 case "SHOW":
804 points := []string{}
805 for p := 1; p <= 24; p++ {
806 points = append(points, strconv.Itoa(s.Position[0][p]-s.Position[1][25-p]))
807 }
808 emit(fmt.Sprintf("POS %s %d %d %s", strings.Join(points, ","), s.Position[0][25], s.Position[1][25], color(s.Side)))
809 case "BOARD":
810 e.board()
811 case "PIPS":
812 emit(fmt.Sprintf("pips W %d B %d", pips(s.Position, 0), pips(s.Position, 1)))
813 case "RACE":
814 if contact(s.Position) {
815 emit("contact")
816 } else {
817 emit(fmt.Sprintf("race W %d B %d", keith(s.Position, 0), keith(s.Position, 1)))
818 }
819 case "EVAL":
820 v := evaluate(s.Position, s.Side)
821 emit(fmt.Sprintf("eval %d pips %d blots %d home %d prime %d anchors %d bar %d", v.E, v.P, v.S, v.H, v.R, v.A, v.G))
822 case "MOVES", "PLAY", "AUTO", "HINT":
823 d1, d2 := number(a[0]), number(a[1])
824 if cmd == "PLAY" {
825 moves, ok := parsePlay(a[2:])
826 if !ok {
827 emit("ILLEGAL syntax")
828 return
829 }
830 legal, possible, maximum := enumerate(s.Position, s.Side, d1, d2)
831 play, reason := validatePlay(s.Position, s.Side, moves, legal, possible, maximum)
832 if reason != "" {
833 emit("ILLEGAL " + reason)
834 return
835 }
836 emit("OK " + play.Notation)
837 e.makePlay(play, d1, d2)
838 return
839 }
840 legal, _, _ := enumerate(s.Position, s.Side, d1, d2)
841 if cmd == "MOVES" {
842 for _, play := range legal {
843 emit(play.Notation)
844 }
845 emit(fmt.Sprintf("total %d", len(legal)))
846 } else if cmd == "AUTO" {
847 play, value := best(legal, s.Side)
848 emit(fmt.Sprintf("OK %s eval %d", play.Notation, value))
849 e.makePlay(play, d1, d2)
850 } else {
851 scores := make([]int, len(legal))
852 for i, play := range legal {
853 scores[i] = evaluate(play.Position, s.Side).E
854 }
855 order := make([]int, len(legal))
856 for i := range order {
857 order[i] = i
858 }
859 sort.SliceStable(order, func(i, j int) bool {
860 return scores[order[i]] > scores[order[j]]
861 })
862 for _, i := range order[:min(number(a[2]), len(order))] {
863 emit(fmt.Sprintf("%d %s", scores[i], legal[i].Notation))
864 }
865 }
866 case "SHOTS":
867 blots := []int{}
868 for p := 24; p >= 1; p-- {
869 if s.Position[s.Side][p] == 1 {
870 blots = append(blots, p)
871 }
872 }
873 counts := map[int]int{}
874 total := 0
875 rolls := []string{}
876 for a := 6; a >= 1; a-- {
877 for b := a; b >= 1; b-- {
878 legal, _, _ := enumerate(s.Position, 1-s.Side, a, b)
879 hit := map[int]bool{}
880 for _, play := range legal {
881 for _, p := range blots {
882 if play.Hits&(1<<(25-p)) != 0 {
883 hit[p] = true
884 }
885 }
886 }
887 weight := 2
888 if a == b {
889 weight = 1
890 }
891 if len(hit) > 0 {
892 total += weight
893 rolls = append(rolls, fmt.Sprintf("%d-%d", a, b))
894 }
895 for p := range hit {
896 counts[p] += weight
897 }
898 }
899 }
900 line := fmt.Sprintf("shots %d/36", total)
901 if len(rolls) > 0 {
902 line += " " + strings.Join(rolls, " ")
903 }
904 emit(line)
905 for _, p := range blots {
906 emit(fmt.Sprintf("blot %d %d/36", p, counts[p]))
907 }
908 case "CUBE":
909 emit("cube " + cubeDecision(*s, false))
910 case "DOUBLE":
911 if s.Owner == 1-s.Side || s.Cube == 64 {
912 emit("ERROR cube unavailable")
913 } else {
914 s.Pending = true
915 e.Redo = nil
916 emit(fmt.Sprintf("OK double %d", s.Cube*2))
917 }
918 case "TAKE":
919 if !s.Pending {
920 emit("ERROR no double")
921 } else {
922 s.Pending = false
923 s.Cube *= 2
924 s.Owner = 1 - s.Side
925 e.Redo = nil
926 emit(fmt.Sprintf("OK take %d %s", s.Cube, owner(*s)))
927 }
928 case "DROP":
929 if !s.Pending {
930 emit("ERROR no double")
931 } else {
932 result := finish(s, s.Side, "drop")
933 e.Redo = nil
934 emit("OK drop")
935 emit(resultLine(result))
936 }
937 case "RESIGN":
938 result := finish(s, 1-s.Side, a[0])
939 e.Redo = nil
940 emit("OK resign")
941 emit(resultLine(result))
942 case "SCORE":
943 emit(fmt.Sprintf("score W %d B %d", s.Score[0], s.Score[1]))
944 case "HISTORY":
945 emit(fmt.Sprintf("history %d", len(e.History)))
946 for i, entry := range e.History {
947 emit(fmt.Sprintf("%d %s", i+1, entry.Record))
948 }
949 case "UNDO":
950 n := number(a[0])
951 index := len(e.History) - n
952 if index < 0 {
953 emit("ERROR cannot undo")
954 } else {
955 e.State = e.History[index].Before
956 undone := append([]Entry{}, e.History[index:]...)
957 e.Redo = append(undone, e.Redo...)
958 e.History = e.History[:index]
959 emit("OK")
960 }
961 case "REDO":
962 n := number(a[0])
963 if n > len(e.Redo) {
964 emit("ERROR cannot redo")
965 } else {
966 e.State = e.Redo[n-1].After
967 e.History = append(e.History, e.Redo[:n]...)
968 e.Redo = e.Redo[n:]
969 emit("OK")
970 }
971 case "GAME":
972 rng := Random{int64(number(a[0]))}
973 result := simulation(*s, &rng, [2]string{a[1], a[2]}, false, 0, [2]int{}, false, true)
974 emit(resultLine(result))
975 case "MATCH":
976 target := number(a[0])
977 rng := Random{int64(number(a[1]))}
978 score := [2]int{}
979 used := false
980 game := 0
981 for score[0] < target && score[1] < target {
982 game++
983 crawford := !used && max(score[0], score[1]) == target-1
984 used = used || crawford
985 result := simulation(initial(), &rng, [2]string{"BEST", "BEST"}, true, target, score, crawford, false)
986 score[result.Winner] += result.Points
987 label := ""
988 if crawford {
989 label = " crawford"
990 }
991 emit(fmt.Sprintf("game %d%s start %s turns %d result %s %s %d score %d-%d", game, label, color(result.Start), result.Turns, color(result.Winner), result.Kind, result.Points, score[0], score[1]))
992 }
993 winner := 0
994 if score[1] >= target {
995 winner = 1
996 }
997 emit(fmt.Sprintf("match %s %d-%d", color(winner), score[0], score[1]))
998 case "STATS":
999 rng := Random{int64(number(a[1]))}
1000 counts := map[string][2]int{}
1001 points := [2]int{}
1002 turns := 0
1003 for i := 0; i < number(a[0]); i++ {
1004 result := simulation(initial(), &rng, [2]string{"BEST", "BEST"}, true, 0, [2]int{}, false, false)
1005 c := counts[result.Kind]
1006 c[result.Winner]++
1007 counts[result.Kind] = c
1008 points[result.Winner] += result.Points
1009 turns += result.Turns
1010 }
1011 for _, kind := range []string{"single", "gammon", "backgammon", "drop"} {
1012 c := counts[kind]
1013 emit(fmt.Sprintf("%s W %d B %d", kind, c[0], c[1]))
1014 }
1015 emit(fmt.Sprintf("points W %d B %d", points[0], points[1]))
1016 emit(fmt.Sprintf("turns %d", turns))
1017 }
1018}
