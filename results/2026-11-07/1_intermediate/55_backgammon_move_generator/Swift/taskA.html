75func pairLess(_ a: Move, _ b: Move) -> Bool {
76 a.from != b.from ? a.from > b.from : a.to > b.to
77}
78func keyLess(_ a: [Move], _ b: [Move]) -> Bool {
79 for i in 0..<min(a.count, b.count) {
80 if a[i] != b[i] {
81 return pairLess(a[i], b[i])
82 }
83 }
84 return a.count > b.count
85}
86func notation(_ moves: [Move], _ hits: Set<Int>) -> ([Move], String) {
87 var key = moves.sorted(by: pairLess)
88 while true {
89 var pair: (Int, Int)?
90 for i in key.indices {
91 let t = key[i].to
92 if t == 0 || hits.contains(t) {
93 continue
94 }
95 if let j = key.firstIndex(where: { $0.from == t }) {
96 pair = (i, j)
97 break
98 }
99 }
100 guard let (ai, bi) = pair else {
101 break
102 }
103 let merged = Move(from: key[ai].from, to: key[bi].to)
104 key.remove(at: max(ai, bi))
105 key.remove(at: min(ai, bi))
106 key.append(merged)
107 key.sort(by: pairLess)
108 }
109 var parts: [String] = []
110 var marked = Set<Int>()
111 var i = 0
112 while i < key.count {
113 let move = key[i]
114 var j = i + 1
115 while j < key.count && key[j] == move {
116 j += 1
117 }
118 let from = move.from == 25 ? "bar" : String(move.from)
119 let to = move.to == 0 ? "off" : String(move.to)
120 var text = "\(from)/\(to)"
121 if hits.contains(move.to) && marked.insert(move.to).inserted {
122 text += "*"
123 }
124 if j - i > 1 {
125 text += "(\(j - i))"
126 }
127 parts.append(text)
128 i = j
129 }
130 return (key, parts.isEmpty ? "none" : parts.joined(separator: " "))
131}
132func enumerate(_ pos: Position, _ side: Int, _ d1: Int, _ d2: Int) -> Plays {
133 var all: [Play] = []
134 let dice = d1 == d2 ? Array(repeating: d1, count: 4) : [d1, d2]
135 func visit(_ board: Position, _ remaining: [Int], _ moves: [Move], _ hits: Set<Int>, _ firstDie: Int) {
136 let (key, text) = notation(moves, hits)
137 all.append(Play(pos: board, key: key, text: text, count: moves.count, firstDie: firstDie))
138 var tried = Set<Int>()
139 for (di, die) in remaining.enumerated() {
140 if !tried.insert(die).inserted {
141 continue
142 }
143 for s in 1...25 {
144 if board[side][s] == 0 || (board[side][25] > 0 && s != 25) {
145 continue
146 }
147 let t = max(0, s - die)
148 if t > 0 {
149 if board[1 - side][25 - t] >= 2 {
150 continue
151 }
152 } else {
153 if board[side][25] > 0 || (7...24).contains(where: { board[side][$0] > 0 }) {
154 continue
155 }
156 if s < die && (s + 1...6).contains(where: { board[side][$0] > 0 }) {
157 continue
158 }
159 }
160 var next = board
161 next[side][s] -= 1
162 next[side][t] += 1
163 var nextHits = hits
164 if t > 0 && board[1 - side][25 - t] == 1 {
165 next[1 - side][25 - t] = 0
166 next[1 - side][25] += 1
167 nextHits.insert(t)
168 }
169 var rest = remaining
170 rest.remove(at: di)
171 visit(next, rest, moves + [Move(from: s, to: t)], nextHits, moves.isEmpty ? die : firstDie)
172 }
173 }
174 }
175 visit(pos, dice, [], [], 0)
176 let maximum = all.map(\.count).max()!
177 let larger = maximum == 1 && d1 != d2 && all.contains(where: { $0.count == 1 && $0.firstDie == max(d1, d2) })
178 var seen = Set<Position>()
179 let legal = all.filter { play in
180 play.count == maximum && (!larger || play.firstDie == max(d1, d2)) && seen.insert(play.pos).inserted
181 }.sorted { keyLess($0.key, $1.key) }
182 return Plays(all: all, legal: legal, maximum: maximum)
183}
184func pips(_ own: [Int]) -> Int {
185 (1...25).reduce(0) { $0 + $1 * own[$1] }
186}
187func evaluation(_ pos: Position, _ side: Int) -> [Int] {
188 let own = pos[side], opponent = pos[1 - side]
189 let p = pips(opponent) - pips(own)
190 var s = 0, home = 0, prime = 0, anchors = 0, run = 0
191 for point in 1...24 {
192 if own[point] == 1 {
193 var distances = Set<Int>()
194 for q in 0..<point where opponent[25 - q] > 0 {
195 distances.insert(point - q)
196 }
197 for a in 1...6 {
198 for b in 1...6 {
199 if distances.contains(a) || distances.contains(b) || distances
200 .contains(a + b) || (a == b && (distances.contains(3 * a) || distances.contains(4 * a)))
201 {
202 s += 1
203 }
204 }
205 }
206 }
207 if own[point] >= 2 {
208 if point <= 6 {
209 home += 1
210 }
211 if point >= 19 {
212 anchors += 1
213 }
214 run += 1
215 prime = max(prime, run)
216 } else {
217 run = 0
218 }
219 }
220 let bar = opponent[25] - own[25]
221 return [2 * p - s + 4 * home + 3 * prime + 3 * anchors + 6 * bar, p, s, home, prime, anchors, bar]
222}
223func keith(_ own: [Int]) -> Int {
224 pips(own) + 2 * max(own[1] - 1, 0) + max(own[2] - 1, 0) + max(own[3] - 3, 0)
225 + (4...6).filter { own[$0] == 0 }.count
226}
227func contact(_ pos: Position) -> Bool {
228 let w = pos[0][25] > 0 ? 25 : ((1...24).filter { pos[0][$0] > 0 }.max() ?? 0)
229 let b = pos[1][25] > 0 ? 0 : 25 - ((1...24).filter { pos[1][$0] > 0 }.max() ?? 0)
230 return w > b
231}
232func cubeDecision(_ state: State, forbidden: Bool = false) -> String {
233 let x = state.side
234 if state.owner == 1 - x || state.cube == 64 || forbidden {
235 return "unavailable"
236 }
237 if contact(state.pos) {
238 let value = evaluation(state.pos, x)[0]
239 if value < 60 {
240 return "no double"
241 }
242 return value <= 100 ? "double take" : "double drop"
243 }
244 let value = 8 * keith(state.pos[x]) - 7 * keith(state.pos[1 - x])
245 let limit = state.owner == nil ? 28 : 21
246 if value > limit {
247 return "no double"
248 }
249 return value >= 14 ? "double take" : "double drop"
250}
251func endGame(_ state: inout State, winner: Int, kind suppliedKind: String? = nil) -> Outcome {
252 let kind: String
253 if let suppliedKind {
254 kind = suppliedKind
255 } else {
256 let opponent = state.pos[1 - winner]
257 if opponent[0] > 0 {
258 kind = "single"
259 } else if opponent[25] > 0 || (19...24).contains(where: { opponent[$0] > 0 }) {
260 kind = "backgammon"
261 } else {
262 kind = "gammon"
263 }
264 }
265 let multiplier = kind == "gammon" ? 2 : (kind == "backgammon" ? 3 : 1)
266 let points = state.cube * multiplier
267 state.pending = false
268 state.over = true
269 state.score[winner] += points
270 return Outcome(winner: winner, kind: kind, points: points)
271}
272func applyPlay(_ state: inout State, _ play: Play) -> Outcome? {
273 state.pos = play.pos
274 if state.pos[state.side][0] == 15 {
275 let winner = state.side
276 return endGame(&state, winner: winner)
277 }
278 state.side = 1 - state.side
279 return nil
280}
281func ranked(_ legal: [Play], _ side: Int) -> [(value: Int, index: Int)] {
282 legal.enumerated().map { (value: evaluation($0.element.pos, side)[0], index: $0.offset) }
283 .sorted { $0.value != $1.value ? $0.value > $1.value : $0.index < $1.index }
284}
285func parsePoint(_ original: String, first: Bool) throws -> Int {
286 if first && original == "bar" {
287 return 25
288 }
289 if !first && original == "off" {
290 return 0
291 }
292 var word = original
293 if !first && word.hasSuffix("*") {
294 word.removeLast()
295 }
296 guard !word.isEmpty, word.first != "0", word.utf8.allSatisfy({ $0 >= 48 && $0 <= 57 }),
297 let point = Int(word), (1...24).contains(point)
298 else {
299 throw EngineError.message("ILLEGAL syntax")
300 }
301 return point
302}
303func parsePlay(_ tokens: [String]) throws -> [Move] {
304 if tokens == ["none"] {
305 return []
306 }
307 var moves: [Move] = []
308 for token in tokens {
309 var path = token, repeatCount = 1
310 if token.hasSuffix(")") {
311 let chars = Array(token)
312 try require(chars.count >= 3, "ILLEGAL syntax")
313 try require(
314 chars[chars.count - 3] == "(" && ["2", "3", "4"].contains(String(chars[chars.count - 2])),
315 "ILLEGAL syntax"
316 )
317 repeatCount = Int(String(chars[chars.count - 2]))!
318 path = String(chars.dropLast(3))
319 }
320 let fields = path.split(separator: "/", omittingEmptySubsequences: false).map(String.init)
321 try require(fields.count >= 2, "ILLEGAL syntax")
322 var previous = try parsePoint(fields[0], first: true)
323 for field in fields.dropFirst() {
324 let point = try parsePoint(field, first: false)
325 try require(previous > point, "ILLEGAL syntax")
326 for _ in 0..<repeatCount {
327 moves.append(Move(from: previous, to: point))
328 }
329 previous = point
330 }
331 }
332 return moves
333}
334func choosePlay(_ state: State, tokens: [String], plays: Plays) throws -> Play {
335 let moves = try parsePlay(tokens)
336 let side = state.side
337 var final = state.pos
338 var destinations = Set<Int>()
339 for move in moves {
340 final[side][move.from] -= 1
341 final[side][move.to] += 1
342 if move.to > 0 {
343 destinations.insert(move.to)
344 }
345 }
346 for t in destinations where state.pos[1 - side][25 - t] == 1 {
347 final[1 - side][25 - t] = 0
348 final[1 - side][25] += 1
349 }
350 try require((1...25).allSatisfy { final[side][$0] >= 0 }, "ILLEGAL no checker")
351 try require(final[side][25] == 0 || moves.allSatisfy { $0.from == 25 }, "ILLEGAL bar")
352 try require(destinations.allSatisfy { state.pos[1 - side][25 - $0] < 2 }, "ILLEGAL blocked")
353 if moves.contains(where: { $0.to == 0 }) {
354 try require(final[side][25] == 0 && (7...24).allSatisfy { final[side][$0] == 0 }, "ILLEGAL bear-off")
355 }
356 if let legal = plays.legal.first(where: { $0.pos == final }) {
357 return legal
358 }
359 let possible = plays.all.filter { $0.pos == final }
360 try require(!possible.isEmpty, "ILLEGAL impossible")
361 throw EngineError
362 .message(possible
363 .contains(where: { $0.count < plays.maximum }) ? "ILLEGAL too few dice" : "ILLEGAL larger die required")
364}

368func simulate(_ original: State, rng: RandomStream, strategies: [String], opening: Bool = false,
369 target: Int = 0, scores: [Int] = [0, 0], crawford: Bool = false, verbose: Bool = false) -> Simulation
370{
371 var state = original
372 var d1 = 0, d2 = 0
373 if opening {
374 repeat {
375 d1 = rng.die()
376 d2 = rng.die()
377 } while d1 == d2
378 state.side = d1 > d2 ? 0 : 1
379 }
380 let first = state.side
381 var turn = 1, plays = 0
382 while true {
383 let x = state.side
384 if !opening || turn != 1 {
385 let forbidden = target > 0 && (crawford || scores[x] + state.cube >= target)
386 let decision = cubeDecision(state, forbidden: forbidden)
387 if decision.hasPrefix("double") {
388 if verbose {
389 print("\(turn) \(color(x)) doubles \(state.cube * 2)")
390 }
391 if decision == "double drop" {
392 if verbose {
393 print("\(turn) \(color(1 - x)) drops")
394 }
395 return Simulation(outcome: endGame(&state, winner: x, kind: "drop"), turns: plays, first: first)
396 }
397 if verbose {
398 print("\(turn) \(color(1 - x)) takes")
399 }
400 state.cube *= 2
401 state.owner = 1 - x
402 }
403 d1 = rng.die()
404 d2 = rng.die()
405 }
406 let legal = enumerate(state.pos, x, d1, d2).legal
407 let index: Int
408 switch strategies[x] {
409 case "FIRST":
410 index = 0
411 case "RANDOM":
412 index = Int(rng.next() % Int64(legal.count))
413 default:
414 index = ranked(legal, x)[0].index
415 }
416 let play = legal[index]
417 if verbose {
418 print("\(turn) \(color(x)) \(d1)-\(d2) \(play.text)")
419 }
420 let outcome = applyPlay(&state, play)
421 plays += 1
422 if let outcome {
423 return Simulation(outcome: outcome, turns: plays, first: first)
424 }
425 turn += 1
426 }
427}
428func numeric(_ word: String?, _ low: Int, _ high: Int) -> Bool {
429 guard let word, !word.isEmpty, word.utf8.allSatisfy({ $0 >= 48 && $0 <= 57 }),
430 let value = Int(word)
431 else {
432 return false
433 }
434 return value >= low && value <= high
435}
436func parsePosition(_ args: [String]) -> Position? {
437 guard args.count == 4, numeric(args[1], 0, 15), numeric(args[2], 0, 15),
438 ["W", "B"].contains(args[3])
439 else {
440 return nil
441 }
442 let points = args[0].split(separator: ",", omittingEmptySubsequences: false)
443 guard points.count == 24 else {
444 return nil
445 }
446 var pos = Array(repeating: Array(repeating: 0, count: 26), count: 2)
447 for (i, word) in points.enumerated() {
448 guard let n = Int(word), (-15...15).contains(n) else {
449 return nil
450 }
451 if n > 0 {
452 pos[0][i + 1] = n
453 } else {
454 pos[1][24 - i] = -n
455 }
456 }
457 pos[0][25] = Int(args[1])!
458 pos[1][25] = Int(args[2])!
459 for x in 0..<2 {
460 pos[x][0] = 15 - pos[x].reduce(0, +)
461 if !(0...14).contains(pos[x][0]) {
462 return nil
463 }
464 }
465 return pos
466}
467let noArguments = Set("NEW SHOW BOARD PIPS RACE EVAL SHOTS CUBE DOUBLE TAKE DROP SCORE HISTORY".split(separator: " ")
468 .map(String.init))
469let known = noArguments
470 .union("POS MOVES PLAY AUTO HINT RESIGN UNDO REDO GAME MATCH STATS".split(separator: " ").map(String.init))
471let limited = Set("RACE MOVES PLAY AUTO HINT EVAL SHOTS CUBE DOUBLE TAKE DROP RESIGN GAME".split(separator: " ")
472 .map(String.init))
473let pendingLimited = Set("PLAY AUTO CUBE DOUBLE RESIGN GAME".split(separator: " ").map(String.init))
474func argumentsValid(_ command: String, _ a: [String]) -> Bool {
475 if noArguments.contains(command) {
476 return a.isEmpty
477 }
478 func at(_ i: Int) -> String? {
479 i < a.count ? a[i] : nil
480 }
481 switch command {
482 case "POS":
483 return parsePosition(a) != nil
484 case "MOVES", "PLAY", "AUTO", "HINT":
485 guard numeric(at(0), 1, 6), numeric(at(1), 1, 6) else {
486 return false
487 }
488 if command == "PLAY" {
489 return a.count >= 3
490 }
491 if command == "HINT" {
492 return a.count == 3 && numeric(at(2), 1, 1000)
493 }
494 return a.count == 2
495 case "RESIGN":
496 return a.count == 1 && ["single", "gammon", "backgammon"].contains(a[0])
497 case "UNDO", "REDO":
498 return a.count == 1 && numeric(at(0), 1, 1000)
499 case "GAME":
500 return a.count == 3 && numeric(at(0), 1, 2147483646) && a.dropFirst().allSatisfy { [
501 "BEST",
502 "FIRST",
503 "RANDOM"
504 ].contains($0) }
505 case "MATCH", "STATS":
506 return a.count == 2 && numeric(at(0), 1, command == "MATCH" ? 15 : 50) && numeric(
507 at(1),
508 1,
509 2147483646
510 )
511 default:
512 return false
513 }

515func boardLine(_ pos: Position, top: Bool, row: Int = 0) {
516 var line = ""
517 for i in 1...12 {
518 let p = top ? i + 12 : 13 - i
519 let value: String
520 if row == 0 {
521 value = String(p)
522 } else {
523 let w = pos[0][p], b = pos[1][25 - p], n = max(w, b)
524 value = row == 5 && n > 5 ? String(n) : (n >= row ? (w > 0 ? "W" : "B") : ".")
525 }
526 line += String(repeating: " ", count: max(0, 3 - value.count)) + value
527 if i == 6 {
528 line += " |"
529 }
530 }
531 print(line)

533struct Engine {
534 var state = State.initial()
535 var history: [Entry] = []
536 var redo: [Entry] = []
537
538 mutating func run(_ line: String) throws {
539 let words = line.split(separator: " ").map(String.init)
540 let command = words[0]
541 let a = Array(words.dropFirst())
542 try require(known.contains(command), "ERROR unknown command")
543 try require(argumentsValid(command, a), command == "POS" ? "ERROR bad position" : "ERROR bad arguments")
544 try require(!state.over || !limited.contains(command), "ERROR game over")
545 try require(!state.pending || !pendingLimited.contains(command), "ERROR double pending")
546 let side = state.side
547 switch command {
548 case "NEW", "POS":
549 state = State.initial(score: state.score)
550 if command == "POS" {
551 state.pos = parsePosition(a)!
552 state.side = a[3] == "W" ? 0 : 1
553 }
554 history.removeAll()
555 redo.removeAll()
556 print("OK")
557 case "SHOW":
558 let points = (1...24).map { String(state.pos[0][$0] - state.pos[1][25 - $0]) }
559 print("POS \(points.joined(separator: ",")) \(state.pos[0][25]) \(state.pos[1][25]) \(color(side))")
560 case "BOARD":
561 boardLine(state.pos, top: true)
562 for row in 1...5 {
563 boardLine(state.pos, top: true, row: row)
564 }
565 let owner = state.owner.map(color) ?? "center"
566 let status = state.over ? "game over" : color(side) + (state.pending ? " doubles" : " to move")
567 print(
568 "bar W\(state.pos[0][25]) B\(state.pos[1][25]) | off W\(state.pos[0][0]) B\(state.pos[1][0]) | cube \(state.cube) \(owner) | \(status)"
569 )
570 for row in stride(from: 5, through: 1, by: -1) {
571 boardLine(state.pos, top: false, row: row)
572 }
573 boardLine(state.pos, top: false)
574 case "PIPS":
575 print("pips W \(pips(state.pos[0])) B \(pips(state.pos[1]))")
576 case "RACE":
577 print(contact(state.pos) ? "contact" : "race W \(keith(state.pos[0])) B \(keith(state.pos[1]))")
578 case "EVAL":
579 let labels = ["eval", "pips", "blots", "home", "prime", "anchors", "bar"]
580 print(zip(labels, evaluation(state.pos, side)).map { "\($0.0) \($0.1)" }.joined(separator: " "))
581 case "MOVES", "PLAY", "AUTO", "HINT":
582 let d1 = Int(a[0])!, d2 = Int(a[1])!
583 let plays = enumerate(state.pos, side, d1, d2)
584 if command == "MOVES" {
585 for play in plays.legal {
586 print(play.text)
587 }
588 print("total \(plays.legal.count)")
589 } else if command == "HINT" {
590 for item in ranked(plays.legal, side)
591 .prefix(Int(a[2])!)
592 {
593 print("\(item.value) \(plays.legal[item.index].text)")
594 }
595 } else {
596 let play: Play
597 if command == "AUTO" {
598 play = plays.legal[ranked(plays.legal, side)[0].index]
599 } else {
600 play = try choosePlay(state, tokens: Array(a.dropFirst(2)), plays: plays)
601 }
602 let value = evaluation(play.pos, side)[0]
603 let before = state
604 let outcome = applyPlay(&state, play)
605 print("OK " + play.text + (command == "AUTO" ? " eval \(value)" : ""))
606 if let outcome {
607 print(resultLine(outcome))
608 }
609 history.append(Entry(record: "\(color(side)) \(d1)-\(d2) \(play.text)", before: before, after: state))
610 redo.removeAll()
611 }
612 case "SHOTS":
613 let blots = (1...24).reversed().filter { state.pos[side][$0] == 1 }
614 var counts = Array(repeating: 0, count: 25)
615 var total = 0
616 var rolls: [String] = []
617 for d1 in stride(from: 6, through: 1, by: -1) {
618 for d2 in stride(from: d1, through: 1, by: -1) {
619 let legal = enumerate(state.pos, 1 - side, d1, d2).legal
620 let hit = blots.filter { p in legal.contains { $0.pos[side][p] == 0 } }
621 if !hit.isEmpty {
622 let weight = d1 == d2 ? 1 : 2
623 total += weight
624 rolls.append("\(d1)-\(d2)")
625 for p in hit {
626 counts[p] += weight
627 }
628 }
629 }
630 }
631 print("shots \(total)/36" + (rolls.isEmpty ? "" : " " + rolls.joined(separator: " ")))
632 for p in blots {
633 print("blot \(p) \(counts[p])/36")
634 }
635 case "CUBE":
636 print("cube " + cubeDecision(state))
637 case "DOUBLE":
638 try require(state.owner != 1 - side && state.cube != 64, "ERROR cube unavailable")
639 state.pending = true
640 redo.removeAll()
641 print("OK double \(state.cube * 2)")
642 case "TAKE", "DROP":
643 try require(state.pending, "ERROR no double")
644 state.pending = false
645 redo.removeAll()
646 if command == "TAKE" {
647 state.cube *= 2
648 state.owner = 1 - side
649 print("OK take \(state.cube) \(color(1 - side))")
650 } else {
651 print("OK drop")
652 print(resultLine(endGame(&state, winner: side, kind: "drop")))
653 }
654 case "RESIGN":
655 redo.removeAll()
656 print("OK resign")
657 print(resultLine(endGame(&state, winner: 1 - side, kind: a[0])))
658 case "SCORE":
659 print("score W \(state.score[0]) B \(state.score[1])")
660 case "HISTORY":
661 print("history \(history.count)")
662 for (i, entry) in history.enumerated() {
663 print("\(i + 1) \(entry.record)")
664 }
665 case "UNDO", "REDO":
666 let count = Int(a[0])!
667 if command == "UNDO" {
668 try require(count <= history.count, "ERROR cannot undo")
669 let entries = Array(history.suffix(count))
670 state = entries[0].before
671 history.removeLast(count)
672 redo.insert(contentsOf: entries, at: 0)
673 } else {
674 try require(count <= redo.count, "ERROR cannot redo")
675 let entries = Array(redo.prefix(count))
676 state = entries[count - 1].after
677 redo.removeFirst(count)
678 history.append(contentsOf: entries)
679 }
680 print("OK")
681 case "GAME":
682 let simulation = simulate(
683 state,
684 rng: RandomStream(Int64(a[0])!),
685 strategies: Array(a.dropFirst()),
686 verbose: true
687 )
688 print(resultLine(simulation.outcome))
689 case "MATCH", "STATS":
690 let n = Int(a[0])!, rng = RandomStream(Int64(a[1])!)
691 var points = [0, 0], turns = 0, game = 0
692 var crawfordUsed = false
693 let kinds = ["single", "gammon", "backgammon", "drop"]
694 var outcomes = Dictionary(uniqueKeysWithValues: kinds.map { ($0, [0, 0]) })
695 while command == "STATS" ? game < n : points.max()! < n {
696 game += 1
697 let crawford = command == "MATCH" && !crawfordUsed && points.max()! == n - 1
698 crawfordUsed = crawfordUsed || crawford
699 let simulation = simulate(State.initial(), rng: rng, strategies: ["BEST", "BEST"], opening: true,
700 target: command == "MATCH" ? n : 0, scores: points, crawford: crawford)
701 let result = simulation.outcome
702 points[result.winner] += result.points
703 outcomes[result.kind]![result.winner] += 1
704 turns += simulation.turns
705 if command == "MATCH" {
706 print(
707 "game \(game)\(crawford ? " crawford" : "") start \(color(simulation.first)) turns \(simulation.turns) \(resultLine(result)) score \(points[0])-\(points[1])"
708 )
709 }
710 }
711 if command == "MATCH" {
712 print("match \(color(points[0] >= n ? 0 : 1)) \(points[0])-\(points[1])")
713 } else {
714 for kind in kinds {
715 print("\(kind) W \(outcomes[kind]![0]) B \(outcomes[kind]![1])")
716 }
717 print("points W \(points[0]) B \(points[1])")
718 print("turns \(turns)")
719 }
720 default:
721 break
722 }
723 }
