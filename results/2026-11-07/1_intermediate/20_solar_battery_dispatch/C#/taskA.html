48 {

103 static List<Strategy> Strategies(string[] words)
104 {
105 var strategies = new List<Strategy>();
106 for (int i = 0; i < words.Length; i++)
107 {
108 string name = words[i];
109 var kind = Named<StrategyKind>(name, "unknown strategy");
110 var strategy = new Strategy(kind);
111 if (strategy.HasParameter)
112 {
113 if (++i == words.Length) throw new CommandError($"missing parameter for {name}");
114 long parameter = Integer(words[i]);
115 long low = kind == StrategyKind.Tou ? 1 : 0;
116 long high = kind == StrategyKind.PeakShave ? 100000 : 100;
117 if (parameter < low || parameter > high)
118 throw new CommandError($"invalid parameter for {name}");
119 strategy = new Strategy(kind, parameter);
120 }
121 strategies.Add(strategy);
122 }
123 return strategies;
124 }

126 static Strategy OneStrategy(string[] words)
127 {
128 var parsed = Strategies(words);
129 if (parsed.Count != 1) throw new CommandError("wrong number of arguments");
130 return parsed[0];

136 static (int Low, int High) Extremes(IReadOnlyList<long> prices)
137 {
138 int low = 0, high = 0;
139 for (int p = 1; p < prices.Count; p++)
140 {
141 if (prices[p] < prices[low]) low = p;
142 if (prices[p] > prices[high]) high = p;
143 }
144 return (low, high);
145 }

200 public Result(Settings settings, IReadOnlyList<long> prices, Strategy strategy,
201 HourRecord[] records, long starting, long reserve)
202 {
203 Settings = settings;
204 Prices = prices.ToArray();
205 Strategy = strategy;
206 Records = records;
207 Starting = starting;
208 Reserve = reserve;
209 Periods = Enumerable.Range(0, prices.Count).Select(_ => new Metrics()).ToArray();
210 PeriodHours = new int[prices.Count];
211 foreach (var record in records)
212 {
213 Total.Add(record.Values);
214 Periods[record.Period].Add(record.Values);
215 PeriodHours[record.Period]++;
216 Peak = Math.Max(Peak, record.Values.Imported);
217 for (int i = 0; i < Flows.Length; i++) Flows[i] += record.Flows[i];
218 }
219 DemandCharge = Peak * settings.Demand / 1000;
220 Bill = Total.Net + settings.Fixed + DemandCharge;
221 }

233 static Result Simulate(Data data, Settings settings, IReadOnlyList<long> prices,
234 Strategy strategy, Outage? outage = null)
235 {
236 long capacity = settings.Capacity;
237 long reservePercent = strategy.Kind == StrategyKind.Backup
238 ? Math.Max(settings.Reserve, strategy.Parameter) : settings.Reserve;
239 long reserve = capacity * reservePercent / 100;
240 long starting = capacity * settings.Initial / 100;
241 var periods = Extremes(prices);
242 bool arbitrage = strategy.Kind == StrategyKind.ExportPeak && settings.Feedin > prices[periods.Low];
243 long soc = starting;
244 var records = new HourRecord[data.Hours.Length];
245 for (int i = 0; i < records.Length; i++)
246 {
247 var input = data.Hours[i];
248 int p = input.Period;
249 long pv = input.Pv * settings.Pvscale / 100;
250 long direct = Math.Min(input.Load, pv);
251 long surplus = pv - direct, deficit = input.Load - direct;
252 long x1 = Min3(surplus, settings.Charge, CeilDiv(100 * (capacity - soc), settings.Ceff));
253 long stored1 = x1 * settings.Ceff / 100;
254 soc += stored1;
255 surplus -= x1;
256 bool off = outage != null && i >= outage.First && i <= outage.Last;
257 long target = off ? deficit : strategy.Kind switch
258 {
259 StrategyKind.Self or StrategyKind.Backup => deficit,
260 StrategyKind.Tou => p == periods.High ? deficit : 0,
261 StrategyKind.PeakShave => Math.Max(0, deficit - strategy.Parameter),
262 StrategyKind.ExportPeak => arbitrage && p == periods.Low && p != periods.High ? 0 : deficit,
263 _ => throw new InvalidOperationException()
264 };
265 long lower = off ? 0 : reserve;
266 long y1 = Min3(CeilDiv(100 * target, settings.Deff), settings.Discharge, Math.Max(0, soc - lower));
267 long d1 = y1 * settings.Deff / 100;
268 soc -= y1;
269 deficit -= d1;
270 long gridLoad = off ? 0 : deficit;
271 long e1 = off ? 0 : Math.Min(surplus, settings.Export);
272 long x2 = 0, y2 = 0, d2 = 0;
273 if (!off)
274 {
275 long headroom = settings.Charge - x1;
276 long? targetSoc = null;
277 switch (strategy.Kind)
278 {
279 case StrategyKind.Tou:
280 if (p == periods.Low && p != periods.High) targetSoc = capacity * strategy.Parameter / 100;
281 break;
282 case StrategyKind.PeakShave:
283 if (p == periods.Low)
284 {
285 targetSoc = capacity;
286 headroom = Math.Min(headroom, Math.Max(0, strategy.Parameter - gridLoad));
287 }
288 break;
289 case StrategyKind.Backup:
290 if (p == periods.Low) targetSoc = reserve;
291 break;
292 case StrategyKind.ExportPeak:
293 if (arbitrage && p == periods.Low && p != periods.High) targetSoc = capacity;
294 break;
295 }
296 if (targetSoc is long goal)
297 {
298 x2 = Math.Min(headroom, CeilDiv(100 * Math.Max(0, goal - soc), settings.Ceff));
299 soc += x2 * settings.Ceff / 100;
300 }
301 if (strategy.Kind == StrategyKind.ExportPeak && arbitrage && p == periods.High && x1 == 0)
302 {
303 y2 = Min3(CeilDiv(100 * (settings.Export - e1), settings.Deff),
304 settings.Discharge - y1, Math.Max(0, soc - reserve));
305 d2 = y2 * settings.Deff / 100;
306 soc -= y2;
307 }
308 }
309 long imported = gridLoad + x2, exported = e1 + d2, curtailed = surplus - e1;
310 var values = new Metrics
311 {
312 Load = input.Load, Pv = pv, Charge = x1 + x2, Discharge = d1 + d2,
313 Drawn = y1 + y2, Soc = soc, Imported = imported, Exported = exported,
314 Curtailed = curtailed, Shortfall = off ? deficit : 0,
315 Loss = x1 - stored1 + x2 - x2 * settings.Ceff / 100 + y1 - d1 + y2 - d2,
316 ImportCost = imported * prices[p] / 100000,
317 ExportIncome = exported * settings.Feedin / 100000
318 };
319 records[i] = new HourRecord(p, values,
320 new[] { direct, x1, e1, curtailed, d1, d2, gridLoad, x2 });
321 }
322 return new Result(settings, prices, strategy, records, starting, reserve);
323 }

398 void Size(string[] args)
399 {
400 Arity(args, 3);
401 long low = Integer(args[0]), high = Integer(args[1]), step = Integer(args[2]);
402 ValidateRange(low, high, step, Item.Capacity);
403 var baseline = Baseline(data, settings, prices);
404 (long Capacity, long Cost, long Saving)? best = null;
405 for (long capacity = low; capacity <= high; capacity += step)
406 {
407 var trial = Simulate(data, settings with { Capacity = capacity }, prices, new Strategy(StrategyKind.Self));
408 long saving = baseline.Bill - trial.Bill;
409 long cost = capacity * settings.Cost / 1000;
410 string payback = "never";
411 if (saving > 0)
412 {
413 long value = 10 * cost / saving;
414 payback = $"{value / 10}.{value % 10}";
415 if (best is null || cost * best.Value.Saving < best.Value.Cost * saving)
416 best = (capacity, cost, saving);
417 }
418 Emit($"SIZE {capacity} bill {Money(trial.Bill)} saving {Money(saving)} cost {Money(cost)} payback {payback}");
419 }
420 Emit($"BEST {(best is { } chosen ? chosen.Capacity.ToString(CultureInfo.InvariantCulture) : "none")}");
421 }

427 long low = Integer(args[1]), high = Integer(args[2]), step = Integer(args[3]);
428 ValidateRange(low, high, step, item);

432 {
433 var changed = settings.With(item, value);
434 var trial = Simulate(data, changed, prices, strategy);
435 var baseline = Baseline(data, changed, prices);
436 Emit($"SWEEP {args[0]} {value} bill {Money(trial.Bill)} " +
437 $"saving {Money(baseline.Bill - trial.Bill)} sufficiency {trial.Sufficiency}%");
438 if (best is null || trial.Bill < best.Value.Bill) best = (value, trial.Bill);
439 }
440 Emit($"BEST {best!.Value.SettingValue}");
441 }

572 void Usage(Result r)
573 {
574 int charging = 0, discharging = 0, idle = 0, full = 0, low = 0;
575 int streak = 0, longest = 0, first = 0;
576 for (int i = 0; i < r.Records.Length; i++)
577 {
578 var v = r.Records[i].Values;
579 if (v.Charge > 0) charging++;
580 if (v.Drawn > 0) discharging++;
581 if (v.Charge == 0 && v.Drawn == 0) idle++;
582 if (v.Soc == r.Settings.Capacity) full++;
583 if (v.Soc <= r.Reserve)
584 {
585 low++;
586 streak++;
587 if (streak > longest) { longest = streak; first = i - streak + 1; }
588 }
589 else streak = 0;
590 }
591 Emit($"USAGE charging {charging} discharging {discharging} idle {idle} full {full} low {low} longest {longest}" +
592 (longest > 0 ? $" from day {first / 24 + 1} hour {first % 24}" : ""));
593 }

595 void Report(string command, Result r, int argument)
596 {
597 switch (command)
598 {
599 case "HOURLY": Hourly(r, argument); break;
600 case "DAILY": Daily(r); break;
601 case "BILL": Bill(r); break;
602 case "SAVINGS": Savings(r); break;
603 case "DAYTYPE": DayType(r); break;
604 case "PEAKS": Peaks(r, argument); break;
605 case "USAGE": Usage(r); break;
606 case "BYPERIOD":
607 Emit($"BYPERIOD strategy {r.Strategy.Notation}");
608 for (int p = 0; p < data.Periods.Length; p++)
609 {
610 var v = r.Periods[p];
611 Emit($"{data.Periods[p]} hours {r.PeriodHours[p]} load {v.Load} pv {v.Pv} " +
612 $"charge {v.Charge} discharge {v.Discharge} import {v.Imported} export {v.Exported}");
613 }
614 break;
615 case "PROFILE":
616 Emit($"PROFILE strategy {r.Strategy.Notation} days {data.Days}");
617 Emit("hour load pv import export soc");
618 for (int hour = 0; hour < 24; hour++)
619 {
620 var v = r.Sum(Enumerable.Range(0, data.Days).Select(day => day * 24 + hour));
621 Table(hour, v.Load / data.Days, v.Pv / data.Days, v.Imported / data.Days, v.Exported / data.Days, v.Soc / data.Days);
622 }
623 break;
624 case "FLOWS":
625 Emit($"FLOWS strategy {r.Strategy.Notation}");
626 for (int i = 0; i < FlowNames.Length; i++) Emit($"{FlowNames[i]} {r.Flows[i]}");
627 Emit($"LOSSES {r.Total.Loss}");
628 Emit($"STORED {r.EndSoc - r.Starting}");
629 break;
630 case "SELFSUFF":
631 Emit($"SELFSUFF sufficiency {r.Sufficiency}% consumption {Percentage(r.Flows[0] + r.Flows[1], r.Total.Pv)}%");
632 break;
633 case "CYCLES":
634 {
635 long drawn = r.Total.Drawn, capacity = r.Settings.Capacity;
636 long divisor = Gcd(drawn, capacity), value = 1000 * drawn / capacity;
637 Emit($"CYCLES {drawn / divisor}/{capacity / divisor} {value / 1000}.{value % 1000:000} drawn {drawn} losses {r.Total.Loss}");
638 break;
639 }
640 case "HISTOGRAM":
641 {
642 Emit($"HISTOGRAM strategy {r.Strategy.Notation}");
643 var counts = new int[10];
644 foreach (var record in r.Records)
645 counts[(int)Math.Min(9, 10 * record.Values.Soc / r.Settings.Capacity)]++;
646 for (int i = 0; i < counts.Length; i++)
647 Emit($"{i * 10,3}% {Bar(20L * counts[i] / r.Records.Length)} {counts[i]}");
648 break;
649 }
650 case "CHART":
651 Emit($"CHART day {argument + 1} {data.DayName(argument)} strategy {r.Strategy.Notation}");
652 for (int hour = 0; hour < 24; hour++)
653 {
654 long soc = r.Records[argument * 24 + hour].Values.Soc, capacity = r.Settings.Capacity;
655 Emit($"{hour,2} {Bar(20 * soc / capacity)} {Percentage(soc, capacity)}%");
656 }
657 break;
658 }
659 }

661 public bool Execute(string line)
662 {
663 string[] words = line.Split(' '), args = words[1..];
664 string command = words[0];
665 try
666 {
667 if (ReportArities.TryGetValue(command, out int count))
668 {
669 Arity(args, count);
670 int argument = 0;
671 if (command is "HOURLY" or "CHART") argument = DayArgument(args[0]);
672 if (command == "PEAKS")
673 {
674 long value = Integer(args[0]);
675 if (value < 1) throw new CommandError("invalid count");
676 argument = (int)value;
677 }
678 Report(command, Current, argument);
679 return false;
680 }
681 switch (command)
682 {
683 case "END": Arity(args, 0); return true;
684 case "RUN":
685 {
686 var strategy = OneStrategy(args);
687 var trial = Simulate(data, settings, prices, strategy);
688 current = trial;
689 Emit($"RUN {strategy.Notation} bill {Money(trial.Bill)} import {trial.Total.Imported} " +
690 $"export {trial.Total.Exported} soc {trial.EndSoc}");
691 break;
692 }
693 case "SET":
694 {
695 Arity(args, 2);
696 var item = Named<Item>(args[0], "unknown parameter");
697 long value = Integer(args[1]);
698 var bounds = Bounds(item);
699 if (value < bounds.Minimum || value > bounds.Maximum) throw new CommandError($"invalid value for {args[0]}");
700 long previous = settings[item];
701 settings = settings.With(item, value);
702 Emit($"SET {args[0]} {previous} -> {value}");
703 break;
704 }
705 case "PRICE":
706 {
707 Arity(args, 2);
708 int p = Array.IndexOf(data.Periods, args[0]);
709 if (p < 0) throw new CommandError($"unknown period {args[0]}");
710 long value = Integer(args[1]);
711 if (value < 0 || value > 10000) throw new CommandError($"invalid value for {args[0]}");
712 long previous = prices[p];
713 prices[p] = value;
714 var bounds = Extremes(prices);
715 Emit($"PRICE {args[0]} {previous} -> {value} cheapest {data.Periods[bounds.Low]} peak {data.Periods[bounds.High]}");
716 break;
717 }
718 case "RESET":
719 Arity(args, 0);
720 settings = initialSettings;
721 prices = initialPrices.ToArray();
722 Emit("RESET");
723 break;
724 case "CONFIG": Arity(args, 0); Config(); break;
725 case "NOBATTERY":
726 {
727 Arity(args, 0);
728 var baseline = Baseline(data, settings, prices);
729 Emit($"NOBATTERY bill {Money(baseline.Bill)} import {baseline.Total.Imported} " +
730 $"export {baseline.Total.Exported} curtailed {baseline.Total.Curtailed} peak {baseline.Peak}");
731 break;
732 }
733 case "COMPARE": Compare(args); break;
734 case "SIZE": Size(args); break;
735 case "SWEEP": Sweep(args); break;
736 case "OUTAGE": EvaluateOutage(args); break;
737 case "DIFF": Difference(args); break;
738 default: throw new CommandError($"unknown command {command}");
739 }
740 }
741 catch (CommandError error) { Emit($"ERROR {error.Message}"); }
742 return false;
743 }
744 }

746 static Dispatch ReadInput()
747 {
748 static string[] Words() => Console.ReadLine()!.Split(' ');
749 long[] battery = Words().Skip(1).Select(Integer).ToArray();
750 long[] grid = Words().Skip(1).Select(Integer).ToArray();
751 var settings = new Settings(battery[0], battery[1], battery[2], battery[3],
752 battery[4], battery[5], battery[6], battery[7], grid[0], grid[1], grid[2], grid[3]);
753 int periodCount = (int)Integer(Words()[1]);
754 var names = new string[periodCount];
755 var prices = new long[periodCount];
756 for (int i = 0; i < periodCount; i++)
757 {
758 var line = Words();
759 names[i] = line[0];
760 prices[i] = Integer(line[1]);
761 }
762 var divisions = new int[2][];
763 for (int type = 0; type < 2; type++)
764 {
765 var line = Words();
766 divisions[type] = new int[24];
767 for (int i = 1; i < line.Length; i += 2)
768 {
769 int start = (int)Integer(line[i]);
770 int end = i + 2 < line.Length ? (int)Integer(line[i + 2]) : 24;
771 int period = Array.IndexOf(names, line[i + 1]);
772 Array.Fill(divisions[type], period, start, end - start);
773 }
774 }
775 int profileCount = (int)Integer(Words()[1]);
776 var profiles = new Dictionary<(string Kind, string Name), long[]>();
777 for (int i = 0; i < profileCount; i++)
778 {
779 var line = Words();
780 profiles[(line[0], line[1])] = line.Skip(2).Select(Integer).ToArray();
781 }
782 int days = (int)Integer(Words()[1]);
783 var hours = new HourInput[days * 24];
784 for (int day = 0; day < days; day++)
785 {
786 var line = Words();
787 var load = profiles[("LOAD", line[0])];
788 var pv = profiles[("PV", line[2])];
789 long loadPercent = Integer(line[1]), pvPercent = Integer(line[3]);
790 for (int hour = 0; hour < 24; hour++)
791 hours[day * 24 + hour] = new HourInput(load[hour] * loadPercent / 100, pv[hour] * pvPercent / 100, 0);
792 }
793 var holidays = Words().Skip(2).Select(word => (int)Integer(word) - 1).ToHashSet();
794 var data = new Data(names, hours, holidays);
795 for (int i = 0; i < hours.Length; i++)
796 hours[i] = hours[i] with { Period = divisions[data.Weekend(i / 24) ? 1 : 0][i % 24] };
797 int overrides = (int)Integer(Words()[1]);
798 for (int i = 0; i < overrides; i++)
799 {
800 var line = Words().Select(Integer).ToArray();
801 int index = ((int)line[0] - 1) * 24 + (int)line[1];
802 hours[index] = hours[index] with { Load = line[2], Pv = line[3] };
803 }
804 return new Dispatch(data, settings, prices);
805 }
