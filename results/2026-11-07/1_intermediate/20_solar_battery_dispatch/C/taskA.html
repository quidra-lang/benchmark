86static Money value(const char *text) { return strtoll(text, NULL, 10); }
87static Money min(Money a, Money b) { return a < b ? a : b; }
88static Money max(Money a, Money b) { return a > b ? a : b; }
89static Money min3(Money a, Money b, Money c) { return min(min(a, b), c); }
90static Money ceiling(Money a, Money b) { return (a + b - 1) / b; }
91static Money gcd(Money a, Money b) {
92 while (b) { Money t = b; b = a % b; a = t; }
93 return a;

96static const char *money(Money amount) {
97 static char buffers[16][36];
98 static int next = 0;
99 char *buffer = buffers[next++ % 16];
100 Money positive = amount < 0 ? -amount : amount;
101 snprintf(buffer, 36, "%s%lld.%02lld", amount < 0 ? "-" : "",
102 (long long)(positive / 100), (long long)(positive % 100));
103 return buffer;
104}
105static const char *percentage(Money numerator, Money denominator) {
106 static char buffers[16][36];
107 static int next = 0;
108 char *buffer = buffers[next++ % 16];
109 Money fraction = denominator ? numerator * 10000 / denominator : 0;
110 snprintf(buffer, 36, "%lld.%02lld", (long long)(fraction / 100),
111 (long long)(fraction % 100));
112 return buffer;
113}

123static int tokenize(char *line, char *tokens[MAX_ARGS]) {
124 int count = 0;
125 for (char *token = strtok(line, " \t\r\n"); token && count < MAX_ARGS;
126 token = strtok(NULL, " \t\r\n")) tokens[count++] = token;
127 return count;
128}
129static void inputTokens(char *tokens[MAX_ARGS], int *count) {
130 static char line[4096];
131 if (!fgets(line, sizeof(line), stdin)) exit(1);
132 *count = tokenize(line, tokens);
133}

144static void periodHours(char *tokens[MAX_ARGS], int n, int hours[24]) {
145 for (int i = 1; i < n; i += 2) {
146 int start = (int)value(tokens[i]);
147 int end = i + 2 < n ? (int)value(tokens[i + 2]) : 24;
148 int period = periodIndex(tokens[i + 1]);
149 for (int h = start; h < end; ++h) hours[h] = period;
150 }
151}

153static void readData(void) {
154 char *tokens[MAX_ARGS];
155 int n;
156 inputTokens(tokens, &n);
157 for (int i = 0; i < 8; ++i) data.initial.values[i] = value(tokens[i + 1]);
158 inputTokens(tokens, &n);
159 for (int i = 0; i < 4; ++i) data.initial.values[i + 8] = value(tokens[i + 1]);
160 data.initial.values[PVSCALE] = 100;
161 inputTokens(tokens, &n);
162 data.priceCount = (int)value(tokens[1]);
163 for (int i = 0; i < data.priceCount; ++i) {
164 inputTokens(tokens, &n);
165 strcpy(data.prices[i].name, tokens[0]);
166 data.prices[i].price = value(tokens[1]);
167 }
168 int weekday[24], weekend[24];
169 inputTokens(tokens, &n);
170 periodHours(tokens, n, weekday);
171 inputTokens(tokens, &n);
172 periodHours(tokens, n, weekend);
173 inputTokens(tokens, &n);
174 int profileCount = (int)value(tokens[1]);
175 Profile profiles[MAX_PROFILES];
176 for (int i = 0; i < profileCount; ++i) {
177 inputTokens(tokens, &n);
178 profiles[i].kind = !strcmp(tokens[0], "LOAD") ? 0 : 1;
179 strcpy(profiles[i].name, tokens[1]);
180 for (int h = 0; h < 24; ++h) profiles[i].values[h] = value(tokens[h + 2]);
181 }
182 inputTokens(tokens, &n);
183 data.dayCount = (int)value(tokens[1]);
184 for (int day = 0; day < data.dayCount; ++day) {
185 inputTokens(tokens, &n);
186 Profile *load = NULL, *pv = NULL;
187 for (int i = 0; i < profileCount; ++i) {
188 if (!profiles[i].kind && !strcmp(profiles[i].name, tokens[0])) load = &profiles[i];
189 if (profiles[i].kind && !strcmp(profiles[i].name, tokens[2])) pv = &profiles[i];
190 }
191 Money loadPct = value(tokens[1]), pvPct = value(tokens[3]);
192 for (int h = 0; h < 24; ++h) {
193 data.loads[24 * day + h] = load->values[h] * loadPct / 100;
194 data.pv[24 * day + h] = pv->values[h] * pvPct / 100;
195 }
196 }
197 int holidays[MAX_DAYS + 1] = {0};
198 inputTokens(tokens, &n);
199 for (int i = 2; i < n; ++i) holidays[(int)value(tokens[i])] = 1;
200 inputTokens(tokens, &n);
201 int overrides = (int)value(tokens[1]);
202 for (int i = 0; i < overrides; ++i) {
203 inputTokens(tokens, &n);
204 int index = ((int)value(tokens[0]) - 1) * 24 + (int)value(tokens[1]);
205 data.loads[index] = value(tokens[2]);
206 data.pv[index] = value(tokens[3]);
207 }
208 for (int day = 1; day <= data.dayCount; ++day) {
209 int holiday = holidays[day];
210 int isWeekend = holiday || (day - 1) % 7 >= 5;
211 data.weekend[day - 1] = isWeekend;
212 strcpy(data.days[day - 1], holiday ? "HOL" : dayNames[(day - 1) % 7]);
213 for (int hour = 0; hour < 24; ++hour)
214 data.periods[(day - 1) * 24 + hour] = isWeekend ? weekend[hour] : weekday[hour];
215 }
216 settings = data.initial;
217 for (int i = 0; i < data.priceCount; ++i) prices[i] = data.prices[i].price;
218}

220static void tariffExtremes(const Money price[MAX_PERIODS], int *cheapest, int *peak) {
221 *cheapest = *peak = 0;
222 for (int i = 0; i < data.priceCount; ++i) {
223 if (price[i] < price[*cheapest]) *cheapest = i;
224 if (price[i] > price[*peak]) *peak = i;
225 }

227static const char *strategyName(enum Method method) {
228 static const char *names[] = {"SELF", "TOU", "PEAKSHAVE", "BACKUP", "EXPORTPEAK"};
229 return names[method];
230}
231static const char *strategyText(Strategy strategy) {
232 static char buffer[50];
233 if (strategy.method == SELF || strategy.method == EXPORTPEAK)
234 snprintf(buffer, sizeof(buffer), "%s", strategyName(strategy.method));
235 else snprintf(buffer, sizeof(buffer), "%s %lld", strategyName(strategy.method),
236 (long long)strategy.parameter);
237 return buffer;
238}
239static bool parseStrategies(char *tokens[], int n, Strategy results[],
240 int *count, char *errorMessage) {
241 *count = 0;
242 for (int i = 0; i < n;) {
243 char *name = tokens[i++];
244 Strategy strategy = {0};
245 if (!strcmp(name, "SELF")) strategy.method = SELF;
246 else if (!strcmp(name, "EXPORTPEAK")) strategy.method = EXPORTPEAK;
247 else {
248 Money low = 0, high = 0;
249 if (!strcmp(name, "TOU")) {
250 strategy.method = TOU; low = 1; high = 100;
251 } else if (!strcmp(name, "PEAKSHAVE")) {
252 strategy.method = PEAKSHAVE; low = 0; high = 100000;
253 } else if (!strcmp(name, "BACKUP")) {
254 strategy.method = BACKUP; low = 0; high = 100;
255 } else {
256 snprintf(errorMessage, 100, "unknown strategy %s", name);
257 return false;
258 }
259 if (i == n) {
260 snprintf(errorMessage, 100, "missing parameter for %s", name);
261 return false;
262 }
263 strategy.parameter = value(tokens[i++]);
264 if (strategy.parameter < low || strategy.parameter > high) {
265 snprintf(errorMessage, 100, "invalid parameter for %s", name);
266 return false;
267 }
268 }
269 results[(*count)++] = strategy;
270 }
271 return true;
272}
273static bool oneStrategy(char *tokens[], int n, Strategy *strategy) {
274 char message[100];
275 Strategy results[MAX_ARGS];
276 int count;
277 if (!parseStrategies(tokens, n, results, &count, message)) { error(message); return false; }
278 if (count != 1) { requireArgumentError(); return false; }
279 *strategy = results[0];
280 return true;
281}

283static void addHour(Totals *total, const Hour *hour) {
284 ++total->count;
285 total->load += hour->load; total->pv += hour->pv;
286 total->charge += hour->charge; total->discharge += hour->discharge;
287 total->imported += hour->imported; total->exported += hour->exported;
288 total->curtailed += hour->curtailed; total->drawn += hour->drawn;
289 total->shortfall += hour->shortfall; total->loss += hour->loss;
290 total->importCost += hour->importCost; total->exportIncome += hour->exportIncome;
291 total->netCost += hour->netCost;
292 for (int i = 0; i < 8; ++i) total->flows[i] += hour->flows[i];
293}

295static void simulate(Strategy strategy, const Settings *options,
296 const Money tariff[MAX_PERIODS], Result *result,
297 int outageFirst, int outageLast) {
298 memset(result, 0, sizeof(*result));
299 result->strategy = strategy;
300 result->settings = *options;
301 memcpy(result->prices, tariff, sizeof(result->prices));
302 Money capacity = options->values[CAPACITY];
303 Money ceff = options->values[CEFF], deff = options->values[DEFF];
304 Money reservePct = options->values[RESERVE];
305 if (strategy.method == BACKUP) reservePct = max(reservePct, strategy.parameter);
306 result->reserve = capacity * reservePct / 100;
307 result->startSoc = capacity * options->values[INITIAL] / 100;
308 Money soc = result->startSoc;
309 int cheapest, peak;
310 tariffExtremes(tariff, &cheapest, &peak);
311 int arbitrage = strategy.method == EXPORTPEAK &&
312 options->values[FEEDIN] > tariff[cheapest];
313 int count = data.dayCount * 24;
314 for (int index = 0; index < count; ++index) {
315 Hour *r = &result->hours[index];
316 r->index = index;
317 r->day = index / 24 + 1;
318 r->hour = index % 24;
319 r->period = data.periods[index];
320 r->load = data.loads[index];
321 r->pv = data.pv[index] * options->values[PVSCALE] / 100;
322 Money direct = min(r->load, r->pv);
323 Money surplus = r->pv - direct;
324 Money deficit = r->load - direct;
325 Money pvCharge = min3(surplus, options->values[CHARGE],
326 ceiling(100 * (capacity - soc), ceff));
327 soc += pvCharge * ceff / 100;
328 surplus -= pvCharge;
329 bool outage = index >= outageFirst && index <= outageLast;
330 Money target = deficit;
331 if (!outage) {
332 if (strategy.method == TOU) target = r->period == peak ? deficit : 0;
333 else if (strategy.method == PEAKSHAVE) target = max(0, deficit - strategy.parameter);
334 else if (arbitrage && r->period == cheapest && r->period != peak) target = 0;
335 }
336 Money lowerBound = outage ? 0 : result->reserve;
337 Money loadDrawn = min3(ceiling(100 * target, deff), options->values[DISCHARGE],
338 max(0, soc - lowerBound));
339 Money loadDischarge = loadDrawn * deff / 100;
340 soc -= loadDrawn;
341 deficit -= loadDischarge;
342 Money gridLoad = 0, shortfall = 0, pvExport = 0, curtailed = 0, gridCharge = 0;
343 Money exportDrawn = 0, batteryExport = 0;
344 if (outage) { shortfall = deficit; curtailed = surplus; }
345 else {
346 gridLoad = deficit;
347 pvExport = min(surplus, options->values[EXPORT]);
348 curtailed = surplus - pvExport;
349 Money headroom = options->values[CHARGE] - pvCharge;
350 Money chargeTarget = -1;
351 if (strategy.method == TOU && r->period == cheapest && r->period != peak)
352 chargeTarget = capacity * strategy.parameter / 100;
353 else if (strategy.method == PEAKSHAVE && r->period == cheapest) {
354 chargeTarget = capacity;
355 headroom = min(headroom, max(0, strategy.parameter - gridLoad));
356 } else if (strategy.method == BACKUP && r->period == cheapest)
357 chargeTarget = result->reserve;
358 else if (arbitrage && r->period == cheapest && r->period != peak)
359 chargeTarget = capacity;
360 if (chargeTarget >= 0) {
361 gridCharge = min(headroom, ceiling(100 * max(0, chargeTarget - soc), ceff));
362 soc += gridCharge * ceff / 100;
363 }
364 if (arbitrage && r->period == peak && pvCharge == 0) {
365 exportDrawn = min3(ceiling(100 * (options->values[EXPORT] - pvExport), deff),
366 options->values[DISCHARGE] - loadDrawn,
367 max(0, soc - result->reserve));
368 batteryExport = exportDrawn * deff / 100;
369 soc -= exportDrawn;
370 }
371 }
372 r->soc = soc;
373 r->flows[0] = direct; r->flows[1] = pvCharge;
374 r->flows[2] = pvExport; r->flows[3] = curtailed;
375 r->flows[4] = loadDischarge; r->flows[5] = batteryExport;
376 r->flows[6] = gridLoad; r->flows[7] = gridCharge;
377 r->charge = pvCharge + gridCharge;
378 r->discharge = loadDischarge + batteryExport;
379 r->imported = gridLoad + gridCharge;
380 r->exported = pvExport + batteryExport;
381 r->curtailed = curtailed;
382 r->drawn = loadDrawn + exportDrawn;
383 r->shortfall = shortfall;
384 r->loss = pvCharge - pvCharge * ceff / 100 + gridCharge - gridCharge * ceff / 100 +
385 loadDrawn - loadDischarge + exportDrawn - batteryExport;
386 r->importCost = r->imported * tariff[r->period] / 100000;
387 r->exportIncome = r->exported * options->values[FEEDIN] / 100000;
388 r->netCost = r->importCost - r->exportIncome;
389 addHour(&result->total, r);
390 addHour(&result->daily[r->day - 1], r);
391 addHour(&result->byPeriod[r->period], r);
392 result->peak = max(result->peak, r->imported);
393 }
394 result->demandCharge = result->peak * options->values[DEMAND] / 1000;
395 result->bill = result->total.netCost + options->values[FIXED] + result->demandCharge;
396 result->endSoc = soc;
397}

432static void execute(char *tokens[MAX_ARGS], int n) {
433 if (!n) return;
434 char *command = tokens[0];
435 char **args = tokens + 1;
436 int argc = n - 1;
437
438 if (!strcmp(command, "RUN")) {
439 Strategy strategy;
440 if (!oneStrategy(args, argc, &strategy)) return;
441 simulate(strategy, &settings, prices, &current, 1, 0);
442 hasCurrent = true;
443 printf("RUN %s bill %s import %lld export %lld soc %lld\n", strategyText(strategy),
444 money(current.bill), (long long)current.total.imported,
445 (long long)current.total.exported, (long long)current.endSoc);
446 } else if (!strcmp(command, "SET")) {
447 if (!checkCount(argc, 2)) return;
448 int which = settingIndex(args[0]);
449 if (which < 0) { printf("ERROR unknown parameter %s\n", args[0]); return; }
450 Money number = value(args[1]);
451 if (number < settingMinimum[which] || number > settingMaximum[which]) {
452 printf("ERROR invalid value for %s\n", args[0]); return;
453 }
454 printf("SET %s %lld -> %lld\n", args[0], (long long)settings.values[which], (long long)number);
455 settings.values[which] = number;
456 } else if (!strcmp(command, "PRICE")) {
457 if (!checkCount(argc, 2)) return;
458 int which = periodIndex(args[0]);
459 if (which < 0) { printf("ERROR unknown period %s\n", args[0]); return; }
460 Money number = value(args[1]);
461 if (number < 0 || number > 10000) {
462 printf("ERROR invalid value for %s\n", args[0]); return;
463 }
464 Money previous = prices[which];
465 prices[which] = number;
466 int cheapest, peak;
467 tariffExtremes(prices, &cheapest, &peak);
468 printf("PRICE %s %lld -> %lld cheapest %s peak %s\n", args[0],
469 (long long)previous, (long long)number,
470 data.prices[cheapest].name, data.prices[peak].name);
471 } else if (!strcmp(command, "RESET")) {
472 if (!checkCount(argc, 0)) return;
473 settings = data.initial;
474 for (int i = 0; i < data.priceCount; ++i) prices[i] = data.prices[i].price;
475 puts("RESET");
476 } else if (!strcmp(command, "CONFIG")) {
477 if (!checkCount(argc, 0)) return;
478 printf("BATTERY");
479 for (int i = 0; i < 8; ++i) {
480 char name[16];
481 strcpy(name, settingNames[i]);
482 for (char *p = name; *p; ++p) if (*p >= 'A' && *p <= 'Z') *p += 'a' - 'A';
483 printf(" %s %lld", name, (long long)settings.values[i]);
484 }
485 putchar('\n');
486 printf("GRID");
487 for (int i = 8; i < 12; ++i) {
488 char name[16];
489 strcpy(name, settingNames[i]);
490 for (char *p = name; *p; ++p) if (*p >= 'A' && *p <= 'Z') *p += 'a' - 'A';
491 printf(" %s %lld", name, (long long)settings.values[i]);
492 }
493 printf("\nSOLAR pvscale %lld\n", (long long)settings.values[PVSCALE]);
494 printf("TARIFF");
495 for (int i = 0; i < data.priceCount; ++i)
496 printf(" %s %lld", data.prices[i].name, (long long)prices[i]);
497 int cheapest, peak;
498 tariffExtremes(prices, &cheapest, &peak);
499 printf(" cheapest %s peak %s\n", data.prices[cheapest].name, data.prices[peak].name);
500 } else if (!strcmp(command, "NOBATTERY")) {
501 if (!checkCount(argc, 0)) return;
502 baseline(&settings, prices, &reference);
503 printf("NOBATTERY bill %s import %lld export %lld curtailed %lld peak %lld\n",
504 money(reference.bill), (long long)reference.total.imported,
505 (long long)reference.total.exported, (long long)reference.total.curtailed,
506 (long long)reference.peak);
507 } else if (!strcmp(command, "COMPARE")) {
508 Strategy strategies[MAX_ARGS];
509 char message[100];
510 int count;
511 if (!parseStrategies(args, argc, strategies, &count, message)) { error(message); return; }
512 if (!count) { requireArgumentError(); return; }
513 baseline(&settings, prices, &reference);
514 struct CompareRow {
515 Strategy strategy;
516 Money bill;
517 char sufficiency[36];
518 } rows[10];
519 for (int i = 0; i < count; ++i) {
520 simulate(strategies[i], &settings, prices, &trial, 1, 0);
521 rows[i].strategy = strategies[i];
522 rows[i].bill = trial.bill;
523 strcpy(rows[i].sufficiency,
524 percentage(max(0, trial.total.load - trial.total.imported), trial.total.load));
525 }
526 for (int i = 1; i < count; ++i) {
527 struct CompareRow row = rows[i];
528 int j = i;
529 while (j > 0 && rows[j - 1].bill > row.bill) {
530 rows[j] = rows[j - 1]; --j;
531 }
532 rows[j] = row;
533 }
534 printf("COMPARE baseline %s\n", money(reference.bill));
535 for (int i = 0; i < count; ++i) {
536 printf("%d. %s bill %s saving %s sufficiency %s%%\n", i + 1,
537 strategyText(rows[i].strategy), money(rows[i].bill),
538 money(reference.bill - rows[i].bill), rows[i].sufficiency);
539 }
540 } else if (!strcmp(command, "SIZE")) {
541 if (!checkCount(argc, 3)) return;
542 Money low = value(args[0]), high = value(args[1]), step = value(args[2]);
543 if (!checkedRange(low, high, step, settingMinimum[CAPACITY], settingMaximum[CAPACITY])) return;
544 baseline(&settings, prices, &reference);
545 Money bestCapacity = 0, bestCost = 0, bestSaving = 0;
546 for (Money capacity = low; capacity <= high; capacity += step) {
547 Settings options = settings;
548 options.values[CAPACITY] = capacity;
549 simulate((Strategy){SELF, 0}, &options, prices, &trial, 1, 0);
550 Money saving = reference.bill - trial.bill;
551 Money cost = capacity * options.values[COST] / 1000;
552 char payback[40] = "never";
