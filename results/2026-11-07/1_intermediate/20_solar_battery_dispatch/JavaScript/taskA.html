3const SETTING_RANGES = {
4 CAPACITY: [1, 100000], INITIAL: [0, 100], CHARGE: [1, 20000],

23const sum = numbers => numbers.reduce((total, value) => total + value, 0);
24const quotient = (a, b) => Math.floor(a / b);
25const ceiling = (a, b) => Math.floor((a + b - 1) / b);
26const money = amount => (amount < 0 ? '-' : '') + quotient(Math.abs(amount), 100) +
27 '.' + String(Math.abs(amount) % 100).padStart(2, '0');
28const percentage = (numerator, denominator) => {
29 const value = denominator ? quotient(10000 * numerator, denominator) : 0;
30 return quotient(value, 100) + '.' + String(value % 100).padStart(2, '0');
31};
32const tableRow = (values, widths) => values.map((value, i) => String(value).padStart(widths[i])).join('');
33const bar = width => '|' + '#'.repeat(width) + '.'.repeat(20 - width) + '|';
34const strategyText = ([name, parameter]) => parameter === null ? name : `${name} ${parameter}`;

36function tariffExtremes(prices) {
37 const names = Object.keys(prices);
38 let cheapest = names[0], peak = names[0];
39 for (const name of names) {
40 if (prices[name] < prices[cheapest]) cheapest = name;
41 if (prices[name] > prices[peak]) peak = name;
42 }
43 return [cheapest, peak];
44}

46function readStrategies(tokens) {
47 const strategies = [];
48 for (let index = 0; index < tokens.length;) {
49 const name = tokens[index++];
50 if (name === 'SELF' || name === 'EXPORTPEAK') {
51 strategies.push([name, null]);
52 continue;
53 }
54 if (!Object.hasOwn(STRATEGY_RANGES, name)) throw new CommandError('unknown strategy ' + name);
55 if (index === tokens.length) throw new CommandError('missing parameter for ' + name);
56 const parameter = Number(tokens[index++]);
57 const [low, high] = STRATEGY_RANGES[name];
58 if (parameter < low || parameter > high) throw new CommandError('invalid parameter for ' + name);
59 strategies.push([name, parameter]);
60 }
61 return strategies;
62}

64function requireCount(values, expected) {
65 if (values.length !== expected) throw new CommandError('wrong number of arguments');
66}
67function oneStrategy(values) {
68 const result = readStrategies(values);
69 requireCount(result, 1);
70 return result[0];

72function checkedRange(low, high, step, [minimum, maximum]) {
73 if (low < minimum || high > maximum || high < low || step < 1 ||
74 quotient(high - low, step) + 1 > 20) throw new CommandError('invalid range');
75 const values = [];
76 for (let value = low; value <= high; value += step) values.push(value);
77 return values;
78}
79function gcd(a, b) {
80 while (b !== 0) [a, b] = [b, a % b];
81 return a;

84class InputData {
85 constructor(lines) {
86 let position = 0;
87 const read = () => lines[position++].split(' ');
88 const battery = read().slice(1).map(Number);
89 const grid = read().slice(1).map(Number);
90 this.initialSettings = Object.fromEntries(
91 Object.keys(SETTING_RANGES).map((key, i) => [key, [...battery, ...grid, 100][i]])
92 );
93 const periodCount = Number(read()[1]);
94 this.initialPrices = {};
95 for (let i = 0; i < periodCount; ++i) {
96 const [name, price] = read();
97 this.initialPrices[name] = Number(price);
98 }
99 const weekday = InputData.readPeriodHours(read().slice(1));
100 const weekend = InputData.readPeriodHours(read().slice(1));
101 const profileCount = Number(read()[1]);
102 const profiles = new Map();
103 for (let i = 0; i < profileCount; ++i) {
104 const [kind, name, ...values] = read();
105 profiles.set(kind + '/' + name, values.map(Number));
106 }
107 this.dayCount = Number(read()[1]);
108 this.loads = [];
109 this.basePV = [];
110 for (let day = 0; day < this.dayCount; ++day) {
111 const [loadName, loadPct, pvName, pvPct] = read();
112 this.loads.push(...profiles.get('LOAD/' + loadName).map(v => quotient(v * Number(loadPct), 100)));
113 this.basePV.push(...profiles.get('PV/' + pvName).map(v => quotient(v * Number(pvPct), 100)));
114 }
115 const holidays = new Set(read().slice(2).map(Number));
116 const overrideCount = Number(read()[1]);
117 for (let i = 0; i < overrideCount; ++i) {
118 const [day, hour, load, pv] = read().map(Number);
119 const index = (day - 1) * 24 + hour;
120 this.loads[index] = load;
121 this.basePV[index] = pv;
122 }
123 this.dayNames = [];
124 this.weekendDays = [];
125 this.periods = [];
126 for (let day = 1; day <= this.dayCount; ++day) {
127 const holiday = holidays.has(day);
128 const weekendDay = holiday || (day - 1) % 7 >= 5;
129 this.dayNames.push(holiday ? 'HOL' : DAY_NAMES[(day - 1) % 7]);
130 this.weekendDays.push(weekendDay);
131 this.periods.push(...(weekendDay ? weekend : weekday));
132 }
133 this.commands = lines.slice(position);
134 }

136 static readPeriodHours(tokens) {
137 const hours = [];
138 for (let i = 0; i < tokens.length; i += 2) {
139 const start = Number(tokens[i]);
140 const end = i + 2 < tokens.length ? Number(tokens[i + 2]) : 24;
141 for (let hour = start; hour < end; ++hour) hours.push(tokens[i + 1]);
142 }
143 return hours;
144 }

147class HourRecord {
148 constructor(index, period, load, pv, soc, flows, drawn, shortfall, loss, price, feedin) {
149 this.index = index;
150 this.day = quotient(index, 24) + 1;
151 this.hour = index % 24;
152 this.period = period;
153 this.load = load;
154 this.pv = pv;
155 this.soc = soc;
156 this.flows = flows;
157 this.charge = flows[1] + flows[7];
158 this.discharge = flows[4] + flows[5];
159 this.imported = flows[6] + flows[7];
160 this.exported = flows[2] + flows[5];
161 this.curtailed = flows[3];
162 this.drawn = drawn;
163 this.shortfall = shortfall;
164 this.loss = loss;
165 this.import_cost = quotient(this.imported * price, 100000);
166 this.export_income = quotient(this.exported * feedin, 100000);
167 this.net_cost = this.import_cost - this.export_income;
168 }
169}

171class Totals {
172 constructor(records) {
173 this.count = records.length;
174 for (const field of TOTAL_FIELDS) this[field] = sum(records.map(row => row[field]));
175 this.flows = FLOW_NAMES.map((_, i) => sum(records.map(row => row.flows[i])));
176 this.net_cost = this.import_cost - this.export_income;
177 }
178}

180class SimulationResult {
181 constructor(strategy, settings, prices, startSOC, reserve, records) {
182 this.strategy = strategy;
183 this.settings = {...settings};
184 this.prices = {...prices};
185 this.startSOC = startSOC;
186 this.reserve = reserve;
187 this.hours = records;
188 this.days = [];
189 for (let i = 0; i < records.length; i += 24) this.days.push(records.slice(i, i + 24));
190 this.daily = this.days.map(day => new Totals(day));
191 this.byPeriod = {};
192 for (const period of Object.keys(prices)) {
193 this.byPeriod[period] = new Totals(records.filter(record => record.period === period));
194 }
195 this.total = new Totals(records);
196 this.peak = Math.max(...records.map(record => record.imported));
197 this.demandCharge = quotient(this.peak * settings.DEMAND, 1000);
198 this.bill = this.total.net_cost + settings.FIXED + this.demandCharge;
199 this.endSOC = records.at(-1).soc;
200 this.sufficiency = percentage(Math.max(0, this.total.load - this.total.imported), this.total.load);
201 }
202}

204class Simulator {
205 constructor(data, strategy, settings, prices, outage = null) {
206 this.data = data;
207 this.strategy = strategy;
208 [this.name, this.parameter] = strategy;
209 this.settings = {...settings};
210 this.prices = {...prices};
211 this.outage = outage;
212 this.capacity = settings.CAPACITY;
213 this.ceff = settings.CEFF;
214 this.deff = settings.DEFF;
215 const reservePercent = this.name === 'BACKUP' ? Math.max(settings.RESERVE, this.parameter) : settings.RESERVE;
216 this.reserve = quotient(this.capacity * reservePercent, 100);
217 this.startSOC = quotient(this.capacity * settings.INITIAL, 100);
218 this.soc = this.startSOC;
219 [this.cheapest, this.peak] = tariffExtremes(prices);
220 this.arbitrage = this.name === 'EXPORTPEAK' && settings.FEEDIN > prices[this.cheapest];
221 }
222
223 dischargeTarget(deficit, period) {
224 if (this.name === 'TOU') return period === this.peak ? deficit : 0;
225 if (this.name === 'PEAKSHAVE') return Math.max(0, deficit - this.parameter);
226 if (this.arbitrage && period === this.cheapest && period !== this.peak) return 0;
227 return deficit;
228 }
229
230 gridCharge(period, pvCharge, gridLoad) {
231 let headroom = this.settings.CHARGE - pvCharge;
232 let target = null;
233 if (this.name === 'TOU' && period === this.cheapest && period !== this.peak) {
234 target = quotient(this.capacity * this.parameter, 100);
235 } else if (this.name === 'PEAKSHAVE' && period === this.cheapest) {
236 target = this.capacity;
237 headroom = Math.min(headroom, Math.max(0, this.parameter - gridLoad));
238 } else if (this.name === 'BACKUP' && period === this.cheapest) {
239 target = this.reserve;
240 } else if (this.arbitrage && period === this.cheapest && period !== this.peak) {
241 target = this.capacity;
242 }
243 if (target === null) return 0;
244 const amount = Math.min(headroom, ceiling(100 * Math.max(0, target - this.soc), this.ceff));
245 this.soc += quotient(amount * this.ceff, 100);
246 return amount;
247 }
248
249 runHour(index) {
250 const settings = this.settings;
251 const load = this.data.loads[index];
252 const pv = quotient(this.data.basePV[index] * settings.PVSCALE, 100);
253 const period = this.data.periods[index];
254 const direct = Math.min(load, pv);
255 let surplus = pv - direct;
256 let deficit = load - direct;
257 const pvCharge = Math.min(surplus, settings.CHARGE,
258 ceiling(100 * (this.capacity - this.soc), this.ceff));
259 this.soc += quotient(pvCharge * this.ceff, 100);
260 surplus -= pvCharge;
261 const isOutage = this.outage !== null && this.outage[0] <= index && index <= this.outage[1];
262 const target = isOutage ? deficit : this.dischargeTarget(deficit, period);
263 const lowerBound = isOutage ? 0 : this.reserve;
264 const loadDrawn = Math.min(ceiling(100 * target, this.deff), settings.DISCHARGE,
265 Math.max(0, this.soc - lowerBound));
266 const loadDischarge = quotient(loadDrawn * this.deff, 100);
267 this.soc -= loadDrawn;
268 deficit -= loadDischarge;
269 let gridLoad, shortfall, pvExport, curtailed, gridCharge, exportDrawn = 0, batteryExport = 0;
270 if (isOutage) {
271 gridLoad = 0;
272 shortfall = deficit;
273 pvExport = 0;
274 curtailed = surplus;
275 gridCharge = 0;
276 } else {
277 gridLoad = deficit;
278 shortfall = 0;
279 pvExport = Math.min(surplus, settings.EXPORT);
280 curtailed = surplus - pvExport;
281 gridCharge = this.gridCharge(period, pvCharge, gridLoad);
282 if (this.arbitrage && period === this.peak && pvCharge === 0) {
283 exportDrawn = Math.min(
284 ceiling(100 * (settings.EXPORT - pvExport), this.deff),
285 settings.DISCHARGE - loadDrawn,
286 Math.max(0, this.soc - this.reserve)
287 );
288 batteryExport = quotient(exportDrawn * this.deff, 100);
289 this.soc -= exportDrawn;
290 }
291 }
292 const loss = pvCharge - quotient(pvCharge * this.ceff, 100) +
293 gridCharge - quotient(gridCharge * this.ceff, 100) +
294 loadDrawn - loadDischarge + exportDrawn - batteryExport;
295 const flows = [direct, pvCharge, pvExport, curtailed, loadDischarge,
296 batteryExport, gridLoad, gridCharge];
297 return new HourRecord(index, period, load, pv, this.soc, flows,
298 loadDrawn + exportDrawn, shortfall, loss, this.prices[period], settings.FEEDIN);
299 }
300
301 run() {
302 const records = this.data.loads.map((_, index) => this.runHour(index));
303 return new SimulationResult(this.strategy, this.settings, this.prices,
304 this.startSOC, this.reserve, records);
305 }
306}

308class DispatchDesk {
309 constructor(data) {
310 this.data = data;
311 this.settings = {...data.initialSettings};
312 this.prices = {...data.initialPrices};
313 this.current = null;
314 this.handlers = {
315 RUN: args => this.run(args), SET: args => this.setValue(args),
316 PRICE: args => this.price(args), RESET: args => this.reset(args),
317 CONFIG: args => this.config(args), NOBATTERY: args => this.noBattery(args),
318 COMPARE: args => this.compare(args), SIZE: args => this.size(args),
319 SWEEP: args => this.sweep(args), OUTAGE: args => this.outage(args),
320 DIFF: args => this.diff(args), HOURLY: args => this.hourly(args),
321 DAILY: args => this.daily(args), BILL: args => this.bill(args),
322 SAVINGS: args => this.savings(args), BYPERIOD: args => this.byPeriod(args),
323 DAYTYPE: args => this.dayType(args), PEAKS: args => this.peaks(args),
324 PROFILE: args => this.profile(args), FLOWS: args => this.flows(args),
325 SELFSUFF: args => this.selfSuff(args), CYCLES: args => this.cycles(args),
326 USAGE: args => this.usage(args), HISTOGRAM: args => this.histogram(args),
327 CHART: args => this.chart(args)
328 };
329 }

331 execute(tokens) {
332 const [command, ...args] = tokens;
333 if (!Object.hasOwn(this.handlers, command)) return [`ERROR unknown command ${command}`];
334 try {
335 return this.handlers[command](args);
336 } catch (error) {
337 if (!(error instanceof CommandError)) throw error;
338 return [`ERROR ${error.message}`];
339 }

381 setValue(args) {
382 requireCount(args, 2);
383 const [item, valueText] = args;
384 if (!Object.hasOwn(SETTING_RANGES, item)) throw new CommandError(`unknown parameter ${item}`);

393 price(args) {
394 requireCount(args, 2);
395 const [period, valueText] = args;
396 if (!Object.hasOwn(this.prices, period)) throw new CommandError(`unknown period ${period}`);
397 const value = Number(valueText);
398 if (value < 0 || value > 10000) throw new CommandError(`invalid value for ${period}`);
399 const previous = this.prices[period];
400 this.prices[period] = value;
401 const [cheapest, peak] = tariffExtremes(this.prices);
402 return [`PRICE ${period} ${previous} -> ${value} cheapest ${cheapest} peak ${peak}`];
403 }

432 compare(args) {
433 const strategies = readStrategies(args);
434 if (!strategies.length) throw new CommandError('wrong number of arguments');
435 const baseline = this.baseline(this.settings, this.prices);
436 const trials = strategies.map(strategy => this.simulate(strategy));
437 trials.sort((a, b) => a.bill - b.bill);
438 const output = [`COMPARE baseline ${money(baseline.bill)}`];
439 trials.forEach((result, index) => {
440 output.push(`${index + 1}. ${strategyText(result.strategy)} bill ${money(result.bill)} ` +
441 `saving ${money(baseline.bill - result.bill)} sufficiency ${result.sufficiency}%`);
442 });
443 return output;
444 }
445
446 size(args) {
447 requireCount(args, 3);
448 const capacities = checkedRange(...args.map(Number), SETTING_RANGES.CAPACITY);
449 const baseline = this.baseline(this.settings, this.prices);
450 const output = [];
451 let best = null;
452 for (const capacity of capacities) {
453 const settings = {...this.settings, CAPACITY: capacity};
454 const result = this.simulate(['SELF', null], settings);
455 const saving = baseline.bill - result.bill;
456 const cost = quotient(capacity * settings.COST, 1000);
457 let payback = 'never';
458 if (saving > 0) {
459 const tenths = quotient(10 * cost, saving);
460 payback = `${quotient(tenths, 10)}.${tenths % 10}`;
461 if (best === null || cost * best.saving < best.cost * saving) {
462 best = {capacity, cost, saving};
463 }
464 }
465 output.push(`SIZE ${capacity} bill ${money(result.bill)} saving ${money(saving)} ` +
466 `cost ${money(cost)} payback ${payback}`);
467 }
468 output.push(`BEST ${best === null ? 'none' : best.capacity}`);
469 return output;
470 }
471
472 sweep(args) {
473 if (args.length < 5) throw new CommandError('wrong number of arguments');
474 const item = args[0];
475 if (!Object.hasOwn(SETTING_RANGES, item)) throw new CommandError(`unknown parameter ${item}`);
476 const values = checkedRange(...args.slice(1, 4).map(Number), SETTING_RANGES[item]);
477 const strategy = oneStrategy(args.slice(4));
478 const output = [];
479 let best = null;
480 for (const value of values) {
481 const settings = {...this.settings, [item]: value};
482 const result = this.simulate(strategy, settings);
483 const baseline = this.baseline(settings, this.prices);
484 if (best === null || result.bill < best.bill) best = {value, bill: result.bill};
485 output.push(`SWEEP ${item} ${value} bill ${money(result.bill)} ` +
486 `saving ${money(baseline.bill - result.bill)} sufficiency ${result.sufficiency}%`);
487 }
488 output.push(`BEST ${best.value}`);
489 return output;
490 }
491
492 outage(args) {
493 requireCount(args, 3);
494 const day = this.checkedDay(args[0]);
495 const [start, end] = args.slice(1).map(Number);
496 if (start < 0 || end > 23 || start > end) throw new CommandError('invalid hours');
497 const current = this.requireResult();
498 const first = (day - 1) * 24 + start;
499 const last = (day - 1) * 24 + end;
500 const result = this.simulate(current.strategy, current.settings, current.prices, [first, last]);
501 const before = first ? result.hours[first - 1].soc : result.startSOC;
502 const output = [`OUTAGE day ${day} hours ${start}-${end} ` +
503 `strategy ${strategyText(result.strategy)} soc ${before}`,
504 'hour load pv disch soc short curt'];
505 const records = result.hours.slice(first, last + 1);
506 for (const record of records) {
507 output.push(tableRow([record.hour, record.load, record.pv, record.discharge,
508 record.soc, record.shortfall, record.curtailed], [4, 7, 7, 7, 7, 7, 7]));
509 }
510 const total = new Totals(records);
511 const served = total.load ? percentage(total.load - total.shortfall, total.load) : '100.00';
512 output.push(`TOTAL load ${total.load} unserved ${total.shortfall} ` +
513 `served ${served}% bill ${money(result.bill)}`);
514 return output;
515 }
516
517 diff(args) {
518 if (args.length < 2) throw new CommandError('wrong number of arguments');
519 const day = this.checkedDay(args[0]);
520 const strategy = oneStrategy(args.slice(1));
521 const current = this.requireResult();
522 const comparison = this.simulate(strategy, current.settings, current.prices);
523 const output = [`DIFF day ${day} ${this.data.dayNames[day - 1]} ` +
524 `${strategyText(current.strategy)} vs ${strategyText(strategy)}`,
525 'hour imp1 imp2 exp1 exp2 soc1 soc2'];
526 const before = current.days[day - 1];
527 const after = comparison.days[day - 1];
528 for (let i = 0; i < before.length; ++i) {
529 const first = before[i], second = after[i];
530 output.push(tableRow([first.hour, first.imported, second.imported, first.exported,
531 second.exported, first.soc, second.soc], [4, 7, 7, 7, 7, 7, 7]));
532 }
533 const cost1 = current.daily[day - 1].net_cost;
534 const cost2 = comparison.daily[day - 1].net_cost;
535 output.push(`TOTAL cost1 ${money(cost1)} cost2 ${money(cost2)} ` +
536 `difference ${money(cost2 - cost1)}`);
537 return output;
538 }

540 hourly(args) {
541 const [day, result] = this.resultForDay(args);
542 const output = [`HOURLY day ${day} ${this.data.dayNames[day - 1]} ` +
543 `strategy ${strategyText(result.strategy)}`,
544 'hour load pv charge disch soc import export curt'];
545 const widths = [4, ...Array(8).fill(7)];

548 record.discharge, record.soc, record.imported, record.exported, record.curtailed], widths));
549 }

555
556 daily(args) {
557 const result = this.resultWithoutArguments(args);
558 const output = [`DAILY strategy ${strategyText(result.strategy)}`,
559 'day name load pv import export curt soc cost'];
560 result.daily.forEach((total, index) => {
561 output.push(tableRow([index + 1, this.data.dayNames[index], total.load, total.pv,
562 total.imported, total.exported, total.curtailed, result.days[index].at(-1).soc,
563 money(total.net_cost)], [3, 5, 8, 8, 8, 8, 8, 8, 10]));
564 });
565 const total = result.total;
566 output.push(`TOTAL load ${total.load} pv ${total.pv} import ${total.imported} ` +
567 `export ${total.exported} curtailed ${total.curtailed} cost ${money(total.net_cost)}`);
568 return output;
569 }

592 ['DEMAND', baseline.demandCharge, result.demandCharge],
593 ['TOTAL', baseline.bill, result.bill]);
594 const output = [`SAVINGS strategy ${strategyText(result.strategy)}`];

617 .filter(index => index !== -1);

619 const total = new Totals(records);

621 output.push(`${name} days ${count} load ${quotient(total.load, count)} ` +

629 peaks(args) {
630 requireCount(args, 1);
631 const count = Number(args[0]);
632 if (count < 1) throw new CommandError('invalid count');
633 const result = this.requireResult();
634 const records = result.hours.filter(record => record.imported > 0)
635 .sort((a, b) => b.imported - a.imported || a.index - b.index);
636 const output = [`PEAKS strategy ${strategyText(result.strategy)}`];
637 records.slice(0, count).forEach((record, index) => {
638 output.push(`${index + 1}. day ${record.day} hour ${record.hour} ${record.period} ` +
639 `import ${record.imported} load ${record.load} soc ${record.soc}`);
640 });
641 if (!records.length) output.push('none');
642 return output;

673 cycles(args) {
674 const result = this.resultWithoutArguments(args);
675 const drawn = result.total.drawn;
676 const capacity = result.settings.CAPACITY;
677 const common = gcd(drawn, capacity);
678 const numerator = quotient(drawn, common), denominator = quotient(capacity, common);
679 const thousandths = quotient(1000 * drawn, capacity);
680 const decimal = `${quotient(thousandths, 1000)}.${String(thousandths % 1000).padStart(3, '0')}`;
681 return [`CYCLES ${numerator}/${denominator} ${decimal} ` +
682 `drawn ${drawn} losses ${result.total.loss}`];
683 }
684
685 usage(args) {
686 const result = this.resultWithoutArguments(args);
687 const hours = result.hours;
688 const charging = hours.filter(record => record.charge > 0).length;
689 const discharging = hours.filter(record => record.drawn > 0).length;
690 const idle = hours.filter(record => record.charge === 0 && record.drawn === 0).length;
691 const full = hours.filter(record => record.soc === result.settings.CAPACITY).length;
692 let low = 0, runLength = 0, longest = 0, first = null;
693 for (const record of hours) {
694 if (record.soc <= result.reserve) {
695 ++low;
696 ++runLength;
697 if (runLength > longest) {
698 longest = runLength;
699 first = hours[record.index - runLength + 1];
700 }
701 } else {
702 runLength = 0;
703 }
704 }
705 let line = `USAGE charging ${charging} discharging ${discharging} idle ${idle} ` +
706 `full ${full} low ${low} longest ${longest}`;
707 if (first !== null) line += ` from day ${first.day} hour ${first.hour}`;
708 return [line];
709 }

711 histogram(args) {
712 const result = this.resultWithoutArguments(args);
713 const counts = Array(10).fill(0);
714 for (const record of result.hours) {
715 ++counts[Math.min(9, quotient(10 * record.soc, result.settings.CAPACITY))];
716 }
717 const output = [`HISTOGRAM strategy ${strategyText(result.strategy)}`];
718 counts.forEach((count, index) => {
719 const width = quotient(20 * count, result.hours.length);
720 output.push(`${String(10 * index).padStart(3)}% ${bar(width)} ${count}`);
721 });
722 return output;
723 }

725 chart(args) {
726 const [day, result] = this.resultForDay(args);
727 const capacity = result.settings.CAPACITY;
728 const output = [`CHART day ${day} ${this.data.dayNames[day - 1]} ` +
729 `strategy ${strategyText(result.strategy)}`];
730 for (const record of result.days[day - 1]) {
731 const width = quotient(20 * record.soc, capacity);
732 output.push(`${String(record.hour).padStart(2)} ${bar(width)} ` +
733 `${percentage(record.soc, capacity)}%`);
734 }
735 return output;
736 }

739function main() {
