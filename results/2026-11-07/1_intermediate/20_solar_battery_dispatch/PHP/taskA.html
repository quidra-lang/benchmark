29function money(int $value): string
30{
31 $amount = abs($value);
32 return ($value < 0 ? '-' : '') . intdiv($amount, 100) . '.' .
33 str_pad((string) ($amount % 100), 2, '0', STR_PAD_LEFT);
34}

36function percentage(int $numerator, int $denominator): string
37{
38 $value = $denominator === 0 ? 0 : intdiv(10000 * $numerator, $denominator);
39 return intdiv($value, 100) . '.' . str_pad((string) ($value % 100), 2, '0', STR_PAD_LEFT);
40}

42function ceiling(int $number, int $denominator): int
43{
44 return intdiv($number + $denominator - 1, $denominator);
45}

61function tariffExtremes(array $prices): array
62{
63 $cheapest = $peak = array_key_first($prices);
64 foreach ($prices as $name => $value) {
65 if ($value < $prices[$cheapest]) $cheapest = $name;
66 if ($value > $prices[$peak]) $peak = $name;
67 }
68 return [$cheapest, $peak];
69}

71function strategyText(array $strategy): string
72{
73 return $strategy[0] . ($strategy[1] === null ? '' : ' ' . $strategy[1]);
74}
75
76function readStrategies(array $tokens): array
77{
78 $strategies = [];
79 for ($index = 0; $index < count($tokens);) {
80 $name = $tokens[$index++];
81 if ($name === 'SELF' || $name === 'EXPORTPEAK') {
82 $strategies[] = [$name, null];
83 continue;
84 }
85 if (!isset(STRATEGY_RANGES[$name])) throw new CommandError('unknown strategy ' . $name);
86 if ($index === count($tokens)) throw new CommandError('missing parameter for ' . $name);
87 $parameter = (int) $tokens[$index++];
88 [$low, $high] = STRATEGY_RANGES[$name];
89 if ($parameter < $low || $parameter > $high) {
90 throw new CommandError('invalid parameter for ' . $name);
91 }
92 $strategies[] = [$name, $parameter];
93 }
94 return $strategies;
95}
96
97function requireCount(array $values, int $count): void
98{
99 if (count($values) !== $count) throw new CommandError('wrong number of arguments');
100}
101
102function oneStrategy(array $arguments): array
103{
104 $strategies = readStrategies($arguments);
105 requireCount($strategies, 1);
106 return $strategies[0];
107}
108
109function checkedRange(int $low, int $high, int $step, array $bounds): array
110{
111 [$minimum, $maximum] = $bounds;
112 if ($low < $minimum || $high > $maximum || $high < $low || $step < 1 ||
113 intdiv(max(0, $high - $low), max(1, $step)) + 1 > 20) {
114 throw new CommandError('invalid range');
115 }
116 $values = [];
117 for ($value = $low; $value <= $high; $value += $step) $values[] = $value;
118 return $values;
119}

121final class InputData
122{
123 public array $initialSettings;
124 public array $initialPrices;
125 public int $dayCount;
126 public array $loads = [];
127 public array $basePV = [];
128 public array $dayNames = [];
129 public array $weekendDays = [];
130 public array $periods = [];
131 public array $commands;
132 private array $lines;
133 private int $position = 0;
134
135 public function __construct(array $lines)
136 {
137 $this->lines = $lines;
138 $battery = array_map('intval', array_slice($this->read(), 1));
139 $grid = array_map('intval', array_slice($this->read(), 1));
140 $this->initialSettings = array_combine(array_keys(SETTING_RANGES),
141 array_merge($battery, $grid, [100]));
142 $periodCount = (int) $this->read()[1];
143 $this->initialPrices = [];
144 for ($i = 0; $i < $periodCount; ++$i) {
145 [$name, $price] = $this->read();
146 $this->initialPrices[$name] = (int) $price;
147 }
148 $weekday = $this->readPeriodHours(array_slice($this->read(), 1));
149 $weekend = $this->readPeriodHours(array_slice($this->read(), 1));
150 $profileCount = (int) $this->read()[1];
151 $profiles = [];
152 for ($i = 0; $i < $profileCount; ++$i) {
153 [$kind, $name, $values] = $this->splitProfile($this->read());
154 $profiles[$kind . '/' . $name] = $values;
155 }
156 $this->dayCount = (int) $this->read()[1];
157 for ($i = 0; $i < $this->dayCount; ++$i) {
158 [$loadName, $loadPct, $pvName, $pvPct] = $this->read();
159 foreach ($profiles['LOAD/' . $loadName] as $value) {
160 $this->loads[] = intdiv($value * (int) $loadPct, 100);
161 }
162 foreach ($profiles['PV/' . $pvName] as $value) {
163 $this->basePV[] = intdiv($value * (int) $pvPct, 100);
164 }
165 }
166 $holidays = array_fill_keys(array_map('intval', array_slice($this->read(), 2)), true);
167 $overrideCount = (int) $this->read()[1];
168 for ($i = 0; $i < $overrideCount; ++$i) {
169 [$day, $hour, $load, $pv] = array_map('intval', $this->read());
170 $index = ($day - 1) * 24 + $hour;
171 $this->loads[$index] = $load;
172 $this->basePV[$index] = $pv;
173 }
174 for ($day = 1; $day <= $this->dayCount; ++$day) {
175 $holiday = isset($holidays[$day]);
176 $weekendDay = $holiday || ($day - 1) % 7 >= 5;
177 $this->dayNames[] = $holiday ? 'HOL' : DAY_NAMES[($day - 1) % 7];
178 $this->weekendDays[] = $weekendDay;
179 array_push($this->periods, ...($weekendDay ? $weekend : $weekday));
180 }
181 $this->commands = array_slice($this->lines, $this->position);
182 }
183
184 private function read(): array
185 {
186 return explode(' ', $this->lines[$this->position++]);
187 }
188
189 private function splitProfile(array $tokens): array
190 {
191 return [$tokens[0], $tokens[1], array_map('intval', array_slice($tokens, 2))];
192 }
193
194 private function readPeriodHours(array $tokens): array
195 {
196 $hours = [];
197 for ($i = 0; $i < count($tokens); $i += 2) {
198 $start = (int) $tokens[$i];
199 $end = isset($tokens[$i + 2]) ? (int) $tokens[$i + 2] : 24;
200 for ($hour = $start; $hour < $end; ++$hour) $hours[] = $tokens[$i + 1];
201 }
202 return $hours;
203 }
204}

206final class HourRecord
207{
208 public int $day;
209 public int $hour;
210 public int $charge;
211 public int $discharge;
212 public int $imported;
213 public int $exported;
214 public int $curtailed;
215 public int $importCost;
216 public int $exportIncome;
217 public int $netCost;
218
219 public function __construct(
220 public int $index, public string $period, public int $load, public int $pv,
221 public int $soc, public array $flows, public int $drawn, public int $shortfall,
222 public int $loss, int $price, int $feedin
223 ) {
224 $this->day = intdiv($index, 24) + 1;
225 $this->hour = $index % 24;
226 $this->charge = $flows[1] + $flows[7];
227 $this->discharge = $flows[4] + $flows[5];
228 $this->imported = $flows[6] + $flows[7];
229 $this->exported = $flows[2] + $flows[5];
230 $this->curtailed = $flows[3];
231 $this->importCost = intdiv($this->imported * $price, 100000);
232 $this->exportIncome = intdiv($this->exported * $feedin, 100000);
233 $this->netCost = $this->importCost - $this->exportIncome;
234 }
235}

237final class Totals
238{
239 public int $count;
240 public int $load = 0;
241 public int $pv = 0;
242 public int $charge = 0;
243 public int $discharge = 0;
244 public int $imported = 0;
245 public int $exported = 0;
246 public int $curtailed = 0;
247 public int $drawn = 0;
248 public int $loss = 0;
249 public int $shortfall = 0;
250 public int $importCost = 0;
251 public int $exportIncome = 0;
252 public int $netCost;
253 public array $flows;
254
255 public function __construct(array $records)
256 {
257 $this->count = count($records);
258 $this->flows = array_fill(0, count(FLOW_NAMES), 0);
259 foreach ($records as $record) {
260 foreach (TOTAL_FIELDS as $field) $this->$field += $record->$field;
261 foreach ($record->flows as $index => $value) $this->flows[$index] += $value;
262 }
263 $this->netCost = $this->importCost - $this->exportIncome;
264 }
265}

267final class SimulationResult
268{
269 public array $days = [];
270 public array $daily = [];
271 public array $byPeriod = [];
272 public Totals $total;
273 public int $peak;
274 public int $demandCharge;
275 public int $bill;
276 public int $endSoc;
277 public string $sufficiency;
278
279 public function __construct(
280 public array $strategy, public array $settings, public array $prices,
281 public int $startSoc, public int $reserve, public array $hours
282 ) {
283 $this->days = array_chunk($hours, 24);
284 foreach ($this->days as $day) $this->daily[] = new Totals($day);
285 foreach ($prices as $period => $_) {
286 $this->byPeriod[$period] = new Totals(array_values(array_filter(
287 $hours, static fn(HourRecord $record): bool => $record->period === $period
288 )));
289 }
290 $this->total = new Totals($hours);
291 $this->peak = max(array_map(static fn(HourRecord $record): int => $record->imported, $hours));
292 $this->demandCharge = intdiv($this->peak * $settings['DEMAND'], 1000);
293 $this->bill = $this->total->netCost + $settings['FIXED'] + $this->demandCharge;
294 $this->endSoc = end($hours)->soc;
295 $this->sufficiency = percentage(max(0, $this->total->load - $this->total->imported),
296 $this->total->load);
297 }
298}

300final class Simulator
301{
302 private string $name;
303 private ?int $parameter;
304 private int $capacity;
305 private int $ceff;
306 private int $deff;
307 private int $reserve;
308 private int $startSoc;
309 private int $soc;
310 private string $cheapest;
311 private string $peak;
312 private bool $arbitrage;
313
314 public function __construct(
315 private InputData $data, private array $strategy,
316 private array $settings, private array $prices, private ?array $outage = null
317 ) {
318 [$this->name, $this->parameter] = $strategy;
319 $this->capacity = $settings['CAPACITY'];
320 $this->ceff = $settings['CEFF'];
321 $this->deff = $settings['DEFF'];
322 $reservePercent = $this->name === 'BACKUP'
323 ? max($settings['RESERVE'], $this->parameter) : $settings['RESERVE'];
324 $this->reserve = intdiv($this->capacity * $reservePercent, 100);
325 $this->startSoc = intdiv($this->capacity * $settings['INITIAL'], 100);
326 $this->soc = $this->startSoc;
327 [$this->cheapest, $this->peak] = tariffExtremes($prices);
328 $this->arbitrage = $this->name === 'EXPORTPEAK'
329 && $settings['FEEDIN'] > $prices[$this->cheapest];
330 }
331
332 private function dischargeTarget(int $deficit, string $period): int
333 {
334 if ($this->name === 'TOU') return $period === $this->peak ? $deficit : 0;
335 if ($this->name === 'PEAKSHAVE') return max(0, $deficit - $this->parameter);
336 if ($this->arbitrage && $period === $this->cheapest && $period !== $this->peak) return 0;
337 return $deficit;
338 }
339
340 private function gridCharge(string $period, int $pvCharge, int $gridLoad): int
341 {
342 $headroom = $this->settings['CHARGE'] - $pvCharge;
343 $target = null;
344 if ($this->name === 'TOU' && $period === $this->cheapest && $period !== $this->peak) {
345 $target = intdiv($this->capacity * $this->parameter, 100);
346 } elseif ($this->name === 'PEAKSHAVE' && $period === $this->cheapest) {
347 $target = $this->capacity;
348 $headroom = min($headroom, max(0, $this->parameter - $gridLoad));
349 } elseif ($this->name === 'BACKUP' && $period === $this->cheapest) {
350 $target = $this->reserve;
351 } elseif ($this->arbitrage && $period === $this->cheapest && $period !== $this->peak) {
352 $target = $this->capacity;
353 }
354 if ($target === null) return 0;
355 $amount = min($headroom, ceiling(100 * max(0, $target - $this->soc), $this->ceff));
356 $this->soc += intdiv($amount * $this->ceff, 100);
357 return $amount;
358 }
359
360 private function runHour(int $index): HourRecord
361 {
362 $settings = $this->settings;
363 $load = $this->data->loads[$index];
364 $pv = intdiv($this->data->basePV[$index] * $settings['PVSCALE'], 100);
365 $period = $this->data->periods[$index];
366 $direct = min($load, $pv);
367 $surplus = $pv - $direct;
368 $deficit = $load - $direct;
369 $pvCharge = min($surplus, $settings['CHARGE'],
370 ceiling(100 * ($this->capacity - $this->soc), $this->ceff));
371 $this->soc += intdiv($pvCharge * $this->ceff, 100);
372 $surplus -= $pvCharge;
373 $isOutage = $this->outage !== null
374 && $this->outage[0] <= $index && $index <= $this->outage[1];
375 $target = $isOutage ? $deficit : $this->dischargeTarget($deficit, $period);
376 $lowerBound = $isOutage ? 0 : $this->reserve;
377 $loadDrawn = min(ceiling(100 * $target, $this->deff), $settings['DISCHARGE'],
378 max(0, $this->soc - $lowerBound));
379 $loadDischarge = intdiv($loadDrawn * $this->deff, 100);
380 $this->soc -= $loadDrawn;
381 $deficit -= $loadDischarge;
382 $exportDrawn = $batteryExport = 0;
383 if ($isOutage) {
384 $gridLoad = 0;
385 $shortfall = $deficit;
386 $pvExport = 0;
387 $curtailed = $surplus;
388 $gridCharge = 0;
389 } else {
390 $gridLoad = $deficit;
391 $shortfall = 0;
392 $pvExport = min($surplus, $settings['EXPORT']);
393 $curtailed = $surplus - $pvExport;
394 $gridCharge = $this->gridCharge($period, $pvCharge, $gridLoad);
395 if ($this->arbitrage && $period === $this->peak && $pvCharge === 0) {
396 $exportDrawn = min(ceiling(100 * ($settings['EXPORT'] - $pvExport), $this->deff),
397 $settings['DISCHARGE'] - $loadDrawn,
398 max(0, $this->soc - $this->reserve));
399 $batteryExport = intdiv($exportDrawn * $this->deff, 100);
400 $this->soc -= $exportDrawn;
401 }
402 }
403 $loss = $pvCharge - intdiv($pvCharge * $this->ceff, 100)
404 + $gridCharge - intdiv($gridCharge * $this->ceff, 100)
405 + $loadDrawn - $loadDischarge + $exportDrawn - $batteryExport;
406 $flows = [$direct, $pvCharge, $pvExport, $curtailed, $loadDischarge,
407 $batteryExport, $gridLoad, $gridCharge];
408 return new HourRecord($index, $period, $load, $pv, $this->soc, $flows,
409 $loadDrawn + $exportDrawn, $shortfall, $loss,
410 $this->prices[$period], $settings['FEEDIN']);
411 }
412
413 public function run(): SimulationResult
414 {
415 $records = [];
416 foreach ($this->data->loads as $index => $_) $records[] = $this->runHour($index);
417 return new SimulationResult($this->strategy, $this->settings, $this->prices,
418 $this->startSoc, $this->reserve, $records);
419 }
420}

422final class DispatchDesk
423{
424 private array $settings;
425 private array $prices;
426 private ?SimulationResult $current = null;
427
428 private const COMMANDS = [
429 'RUN' => 'cmdRun', 'SET' => 'cmdSet', 'PRICE' => 'cmdPrice',
430 'RESET' => 'cmdReset', 'CONFIG' => 'cmdConfig',
431 'NOBATTERY' => 'cmdNoBattery', 'COMPARE' => 'cmdCompare',
432 'SIZE' => 'cmdSize', 'SWEEP' => 'cmdSweep', 'OUTAGE' => 'cmdOutage',
433 'DIFF' => 'cmdDiff', 'HOURLY' => 'cmdHourly', 'DAILY' => 'cmdDaily',
434 'BILL' => 'cmdBill', 'SAVINGS' => 'cmdSavings',
435 'BYPERIOD' => 'cmdByPeriod', 'DAYTYPE' => 'cmdDayType',
436 'PEAKS' => 'cmdPeaks', 'PROFILE' => 'cmdProfile',
437 'FLOWS' => 'cmdFlows', 'SELFSUFF' => 'cmdSelfSuff',
438 'CYCLES' => 'cmdCycles', 'USAGE' => 'cmdUsage',
439 'HISTOGRAM' => 'cmdHistogram', 'CHART' => 'cmdChart',

441
442 public function __construct(private InputData $data)
443 {
444 $this->settings = $data->initialSettings;
445 $this->prices = $data->initialPrices;
446 }
447
448 public function execute(array $tokens): array
449 {
450 $command = array_shift($tokens);
451 if (!isset(self::COMMANDS[$command])) return ['ERROR unknown command ' . $command];
452 try {
453 return $this->{self::COMMANDS[$command]}($tokens);
454 } catch (CommandError $error) {
455 return ['ERROR ' . $error->getMessage()];
456 }
457 }

459 private function requireResult(): SimulationResult

462 return $this->current;
463 }

465 private function checkedDay(string $text): int
466 {
467 $day = (int) $text;
468 if ($day < 1 || $day > $this->data->dayCount) throw new CommandError('invalid day');
469 return $day;

476 }
477

485 private function simulate(array $strategy, ?array $settings = null,

492 private function baseline(array $settings, array $prices): SimulationResult

494 $settings['CAPACITY'] = 0;

498 private function cmdRun(array $args): array
499 {
500 $strategy = oneStrategy($args);
501 $result = $this->simulate($strategy);
502 $this->current = $result;
503 $total = $result->total;
504 return ['RUN ' . strategyText($strategy) . ' bill ' . money($result->bill) .
505 ' import ' . $total->imported . ' export ' . $total->exported .
506 ' soc ' . $result->endSoc];
507 }
508
509 private function cmdSet(array $args): array
510 {
511 requireCount($args, 2);
512 [$item, $text] = $args;
513 if (!isset(SETTING_RANGES[$item])) throw new CommandError('unknown parameter ' . $item);
514 $value = (int) $text;
515 [$low, $high] = SETTING_RANGES[$item];
516 if ($value < $low || $value > $high) throw new CommandError('invalid value for ' . $item);
517 $previous = $this->settings[$item];
518 $this->settings[$item] = $value;
519 return ["SET $item $previous -> $value"];
520 }
521
522 private function cmdPrice(array $args): array
523 {
524 requireCount($args, 2);
525 [$period, $text] = $args;
526 if (!array_key_exists($period, $this->prices)) {
527 throw new CommandError('unknown period ' . $period);
528 }
529 $value = (int) $text;
530 if ($value < 0 || $value > 10000) throw new CommandError('invalid value for ' . $period);
531 $previous = $this->prices[$period];
532 $this->prices[$period] = $value;
533 [$cheapest, $peak] = tariffExtremes($this->prices);
534 return ["PRICE $period $previous -> $value cheapest $cheapest peak $peak"];
535 }
536
537 private function cmdReset(array $args): array
538 {
539 requireCount($args, 0);
540 $this->settings = $this->data->initialSettings;
541 $this->prices = $this->data->initialPrices;
542 return ['RESET'];
543 }
544
545 private function cmdConfig(array $args): array
546 {
547 requireCount($args, 0);
548 $names = array_keys(SETTING_RANGES);
549 $settings = $this->settings;
550 $battery = implode(' ', array_map(static fn(string $item): string =>
551 strtolower($item) . ' ' . $settings[$item], array_slice($names, 0, 8)));
552 $grid = implode(' ', array_map(static fn(string $item): string =>
553 strtolower($item) . ' ' . $settings[$item], array_slice($names, 8, 4)));
554 $tariff = [];
555 foreach ($this->prices as $period => $price) $tariff[] = "$period $price";
556 [$cheapest, $peak] = tariffExtremes($this->prices);
557 return ['BATTERY ' . $battery, 'GRID ' . $grid,
558 'SOLAR pvscale ' . $settings['PVSCALE'],
559 'TARIFF ' . implode(' ', $tariff) . " cheapest $cheapest peak $peak"];
560 }

562 private function cmdNoBattery(array $args): array
563 {
564 requireCount($args, 0);
565 $result = $this->baseline($this->settings, $this->prices);
566 $total = $result->total;
567 return ['NOBATTERY bill ' . money($result->bill) . ' import ' . $total->imported .
568 ' export ' . $total->exported . ' curtailed ' . $total->curtailed .
569 ' peak ' . $result->peak];
570 }

572 private function cmdCompare(array $args): array
573 {
574 $strategies = readStrategies($args);
575 if (!$strategies) throw new CommandError('wrong number of arguments');
576 $baseline = $this->baseline($this->settings, $this->prices);
577 $trials = array_map(fn(array $strategy): SimulationResult => $this->simulate($strategy),
578 $strategies);
579 usort($trials, static fn(SimulationResult $a, SimulationResult $b): int =>
580 $a->bill <=> $b->bill);
581 $output = ['COMPARE baseline ' . money($baseline->bill)];
582 foreach ($trials as $i => $result) {
583 $output[] = ($i + 1) . '. ' . strategyText($result->strategy) .
584 ' bill ' . money($result->bill) .
585 ' saving ' . money($baseline->bill - $result->bill) .
586 ' sufficiency ' . $result->sufficiency . '%';
587 }
588 return $output;
589 }

591 private function cmdSize(array $args): array
592 {
593 requireCount($args, 3);
594 $capacities = checkedRange((int) $args[0], (int) $args[1], (int) $args[2],
595 SETTING_RANGES['CAPACITY']);
596 $baseline = $this->baseline($this->settings, $this->prices);
597 $output = [];
598 $best = null;
599 foreach ($capacities as $capacity) {
600 $settings = $this->settings;
601 $settings['CAPACITY'] = $capacity;
602 $result = $this->simulate(['SELF', null], $settings);
603 $saving = $baseline->bill - $result->bill;
604 $cost = intdiv($capacity * $settings['COST'], 1000);
605 $payback = 'never';
606 if ($saving > 0) {
607 $tenths = intdiv(10 * $cost, $saving);
608 $payback = intdiv($tenths, 10) . '.' . ($tenths % 10);
609 if ($best === null || $cost * $best[2] < $best[1] * $saving) {
610 $best = [$capacity, $cost, $saving];
611 }
612 }
613 $output[] = "SIZE $capacity bill " . money($result->bill) .
614 ' saving ' . money($saving) . ' cost ' . money($cost) .
615 ' payback ' . $payback;
616 }
617 $output[] = $best === null ? 'BEST none' : 'BEST ' . $best[0];
618 return $output;
619 }

881 private function cmdUsage(array $args): array
882 {
883 $result = $this->resultWithoutArguments($args);
884 $charging = $discharging = $idle = $full = $low = 0;
885 $runLength = $longest = 0;
886 $first = null;
887 foreach ($result->hours as $record) {
888 if ($record->charge > 0) ++$charging;
889 if ($record->drawn > 0) ++$discharging;
890 if ($record->charge === 0 && $record->drawn === 0) ++$idle;
891 if ($record->soc === $result->settings['CAPACITY']) ++$full;
892 if ($record->soc <= $result->reserve) {
893 ++$low;
894 ++$runLength;
895 if ($runLength > $longest) {
896 $longest = $runLength;
897 $first = $result->hours[$record->index - $runLength + 1];
898 }
899 } else {
900 $runLength = 0;
901 }
902 }
903 $line = "USAGE charging $charging discharging $discharging idle $idle " .
904 "full $full low $low longest $longest";
905 if ($first !== null) $line .= " from day $first->day hour $first->hour";
906 return [$line];
