181fn simulate(data: Data, s: Settings, prices: [6]i64, strategy: Strategy, outage: ?Outage) Allocator.Error!Result {
182 const records = try std.heap.page_allocator.alloc(Record, data.hours.len);
183 const reserve_pct = if (strategy.kind == .BACKUP) @max(s.reserve, strategy.parameter) else s.reserve;
184 const reserve = @divFloor(s.capacity * reserve_pct, 100);
185 var soc = @divFloor(s.capacity * s.initial, 100);
186 var result: Result = .{ .records = records, .settings = s, .prices = prices, .strategy = strategy, .starting = soc, .reserve = reserve };
187 const periods = data.extremes(prices);
188 const arbitrage = strategy.kind == .EXPORTPEAK and s.feedin > prices[periods.low];
189 for (data.hours, records, 0..) |input, *record, i| {
190 const p = input.period;
191 const pv = @divFloor(input.pv * s.pvscale, 100);
192 const u = @min(input.load, pv);
193 var surplus = pv - u;
194 var deficit = input.load - u;
195 const x1 = @min(surplus, s.charge, ceilDiv(100 * (s.capacity - soc), s.ceff));
196 const stored1 = @divFloor(x1 * s.ceff, 100);
197 soc += stored1;
198 surplus -= x1;
199 const off = if (outage) |o| i >= o.first and i <= o.last else false;
200 const target = if (off) deficit else switch (strategy.kind) {
201 .SELF, .BACKUP => deficit,
202 .TOU => if (p == periods.high) deficit else 0,
203 .PEAKSHAVE => @max(0, deficit - strategy.parameter),
204 .EXPORTPEAK => if (arbitrage and p == periods.low and p != periods.high) 0 else deficit,
205 };
206 const lower = if (off) 0 else reserve;
207 const y1 = @min(ceilDiv(100 * target, s.deff), s.discharge, @max(0, soc - lower));
208 const d1 = @divFloor(y1 * s.deff, 100);
209 soc -= y1;
210 deficit -= d1;
211 const g = if (off) 0 else deficit;
212 const e1 = if (off) 0 else @min(surplus, s.export_limit);
213 var x2: i64 = 0;
214 var y2: i64 = 0;
215 var d2: i64 = 0;
216 if (!off) {
217 var headroom = s.charge - x1;
218 const charge_target: ?i64 = switch (strategy.kind) {
219 .TOU => if (p == periods.low and p != periods.high) @divFloor(s.capacity * strategy.parameter, 100) else null,
220 .PEAKSHAVE => if (p == periods.low) blk: {
221 headroom = @min(headroom, @max(0, strategy.parameter - g));
222 break :blk s.capacity;
223 } else null,
224 .BACKUP => if (p == periods.low) reserve else null,
225 .EXPORTPEAK => if (arbitrage and p == periods.low and p != periods.high) s.capacity else null,
226 .SELF => null,
227 };
228 if (charge_target) |v| {
229 x2 = @min(headroom, ceilDiv(100 * @max(0, v - soc), s.ceff));
230 soc += @divFloor(x2 * s.ceff, 100);
231 }
232 if (strategy.kind == .EXPORTPEAK and arbitrage and p == periods.high and x1 == 0) {
233 y2 = @min(ceilDiv(100 * (s.export_limit - e1), s.deff), s.discharge - y1, @max(0, soc - reserve));
234 d2 = @divFloor(y2 * s.deff, 100);
235 soc -= y2;
236 }
237 }
238 const curtailed = surplus - e1;
239 record.* = .{
240 .values = .{ .load = input.load, .pv = pv, .charge = x1 + x2, .discharge = d1 + d2, .drawn = y1 + y2, .soc = soc, .imported = g + x2, .exported = e1 + d2, .curtailed = curtailed, .shortfall = if (off) deficit else 0, .loss = x1 - stored1 + x2 - @divFloor(x2 * s.ceff, 100) + y1 - d1 + y2 - d2, .import_cost = @divFloor((g + x2) * prices[p], 100000), .export_income = @divFloor((e1 + d2) * s.feedin, 100000) },
241 .period = p,
242 .flows = .{ u, x1, e1, curtailed, d1, d2, g, x2 },
243 };
244 result.total.add(record.values);
245 result.periods[p].add(record.values);
246 result.period_hours[p] += 1;
247 result.peak = @max(result.peak, record.values.imported);
248 for (&result.flows, record.flows) |*total, flow| total.* += flow;
249 }
250 result.demand_charge = @divFloor(result.peak * s.demand, 1000);
251 result.bill = result.total.net() + s.fixed + result.demand_charge;
252 return result;
253}
254fn baseline(data: Data, settings: Settings, prices: [6]i64) Allocator.Error!Result {
255 var s = settings;
256 s.capacity = 0;
257 return simulate(data, s, prices, .{ .kind = .SELF }, null);
258}

287 fn fail(app: *App, gpa: Allocator, comptime format: []const u8, args: anytype) Error {
288 app.error_message = std.fmt.allocPrint(gpa, format, args) catch return error.OutOfMemory;

291 fn arity(app: *App, gpa: Allocator, a: []const []const u8, count: usize) Error!void {
292 if (a.len != count) return app.fail(gpa, "wrong number of arguments", .{});
293 }
294 fn dayArgument(app: *App, gpa: Allocator, text: []const u8) Error!usize {
295 const day = integer(text);
296 if (day < 1 or day > @as(i64, @intCast(app.data.days()))) return app.fail(gpa, "invalid day", .{});
297 return ix(day - 1);
298 }
299 fn itemArgument(app: *App, gpa: Allocator, text: []const u8) Error!Item {
300 return std.meta.stringToEnum(Item, text) orelse app.fail(gpa, "unknown parameter {s}", .{text});
301 }
302 fn range(app: *App, gpa: Allocator, low: i64, high: i64, step: i64, minimum: i64, maximum: i64) Error!void {
303 if (low < minimum or high > maximum or high < low or step < 1 or @divFloor(high - low, step) + 1 > 20) return app.fail(gpa, "invalid range", .{});
304 }
305 fn strategies(app: *App, gpa: Allocator, a: []const []const u8) Error![]Strategy {
306 var parsed: std.ArrayList(Strategy) = .empty;
307 var i: usize = 0;
308 while (i < a.len) {
309 const name = a[i];
310 i += 1;
311 var strategy: Strategy = .{ .kind = std.meta.stringToEnum(Kind, name) orelse return app.fail(gpa, "unknown strategy {s}", .{name}) };
312 if (strategy.hasParameter()) {
313 if (i == a.len) return app.fail(gpa, "missing parameter for {s}", .{name});
314 strategy.parameter = integer(a[i]);
315 i += 1;
316 const low: i64 = if (strategy.kind == .TOU) 1 else 0;
317 const high: i64 = if (strategy.kind == .PEAKSHAVE) 100000 else 100;
318 if (strategy.parameter < low or strategy.parameter > high) return app.fail(gpa, "invalid parameter for {s}", .{name});
319 }
320 try parsed.append(gpa, strategy);
321 }
322 return parsed.items;
323 }
324 fn oneStrategy(app: *App, gpa: Allocator, a: []const []const u8) Error!Strategy {
325 const parsed = try app.strategies(gpa, a);
326 if (parsed.len != 1) return app.fail(gpa, "wrong number of arguments", .{});
327 return parsed[0];
328 }
329 fn result(app: *App, gpa: Allocator) Error!Result {
330 return app.current orelse app.fail(gpa, "no run", .{});
331 }
332 fn config(app: *App, out: *std.Io.Writer) Error!void {
333 const s = app.settings;
334 const extremes = app.data.extremes(app.prices);
335 try out.print("BATTERY capacity {d} initial {d} charge {d} discharge {d} ceff {d} deff {d} reserve {d} cost {d}\nGRID export {d} feedin {d} fixed {d} demand {d}\nSOLAR pvscale {d}\nTARIFF", .{ s.capacity, s.initial, s.charge, s.discharge, s.ceff, s.deff, s.reserve, s.cost, s.export_limit, s.feedin, s.fixed, s.demand, s.pvscale });
336 for (app.data.period_names, 0..) |name, p| try out.print(" {s} {d}", .{ name, app.prices[p] });
337 try out.print(" cheapest {s} peak {s}\n", .{ app.data.period_names[extremes.low], app.data.period_names[extremes.high] });
338 }
339 fn execute(app: *App, gpa: Allocator, out: *std.Io.Writer, words: []const []const u8) Error!bool {
340 const cmd = std.meta.stringToEnum(Command, words[0]) orelse return app.fail(gpa, "unknown command {s}", .{words[0]});
341 const a = words[1..];
342 switch (cmd) {
343 .END => {
344 try app.arity(gpa, a, 0);
345 return true;
346 },
347 .RUN => {
348 const strategy = try app.oneStrategy(gpa, a);
349 var trial = try simulate(app.data, app.settings, app.prices, strategy, null);
350 errdefer trial.deinit();
351 try out.print("RUN {s} bill {s} import {d} export {d} soc {d}\n", .{ try strategy.text(gpa), try money(gpa, trial.bill), trial.total.imported, trial.total.exported, trial.endSoc() });
352 if (app.current) |*old| old.deinit();
353 app.current = trial;
354 },
355 .SET => {
356 try app.arity(gpa, a, 2);
357 const item = try app.itemArgument(gpa, a[0]);
358 const value = integer(a[1]);
359 if (value < item.minimum() or value > item.maximum()) return app.fail(gpa, "invalid value for {s}", .{a[0]});
360 const old = item.get(app.settings);
361 item.set(&app.settings, value);
362 try out.print("SET {s} {d} -> {d}\n", .{ a[0], old, value });
363 },
364 .PRICE => {
365 try app.arity(gpa, a, 2);
366 const p = app.data.periodIndex(a[0]) orelse return app.fail(gpa, "unknown period {s}", .{a[0]});
367 const value = integer(a[1]);
368 if (value < 0 or value > 10000) return app.fail(gpa, "invalid value for {s}", .{a[0]});
369 const old = app.prices[p];
370 app.prices[p] = value;
371 const extremes = app.data.extremes(app.prices);
372 try out.print("PRICE {s} {d} -> {d} cheapest {s} peak {s}\n", .{ a[0], old, value, app.data.period_names[extremes.low], app.data.period_names[extremes.high] });
373 },
374 .RESET => {
375 try app.arity(gpa, a, 0);
376 app.settings = app.initial_settings;
377 app.prices = app.initial_prices;
378 try out.writeAll("RESET\n");
379 },
380 .CONFIG => {
381 try app.arity(gpa, a, 0);
382 try app.config(out);
383 },
384 .NOBATTERY => {
385 try app.arity(gpa, a, 0);
386 var base = try baseline(app.data, app.settings, app.prices);
387 defer base.deinit();
388 try out.print("NOBATTERY bill {s} import {d} export {d} curtailed {d} peak {d}\n", .{ try money(gpa, base.bill), base.total.imported, base.total.exported, base.total.curtailed, base.peak });
389 },
390 .COMPARE => {
391 const parsed = try app.strategies(gpa, a);
392 if (parsed.len == 0) return app.fail(gpa, "wrong number of arguments", .{});
393 var base = try baseline(app.data, app.settings, app.prices);
394 defer base.deinit();
395 var rows: std.ArrayList(Comparison) = .empty;
396 for (parsed, 0..) |strategy, order| {
397 var trial = try simulate(app.data, app.settings, app.prices, strategy, null);
398 defer trial.deinit();
399 try rows.append(gpa, .{ .strategy = strategy, .bill = trial.bill, .load = trial.total.load, .imported = trial.total.imported, .order = order });
400 }
401 std.mem.sort(Comparison, rows.items, {}, Comparison.less);
402 try out.print("COMPARE baseline {s}\n", .{try money(gpa, base.bill)});
403 for (rows.items, 1..) |row, rank| try out.print("{d}. {s} bill {s} saving {s} sufficiency {s}%\n", .{ rank, try row.strategy.text(gpa), try money(gpa, row.bill), try money(gpa, base.bill - row.bill), try percentage(gpa, @max(0, row.load - row.imported), row.load) });
404 },
405 .SIZE => {
406 try app.arity(gpa, a, 3);
407 const low = integer(a[0]);
408 const high = integer(a[1]);
409 const step = integer(a[2]);
410 try app.range(gpa, low, high, step, 1, 100000);
411 var base = try baseline(app.data, app.settings, app.prices);
412 defer base.deinit();
413 var best: ?struct { capacity: i64, cost: i64, saving: i64 } = null;
414 var capacity = low;
415 while (capacity <= high) : (capacity += step) {
416 var s = app.settings;
417 s.capacity = capacity;
418 var trial = try simulate(app.data, s, app.prices, .{ .kind = .SELF }, null);
419 defer trial.deinit();
420 const saving = base.bill - trial.bill;
421 const cost = @divFloor(capacity * s.cost, 1000);
422 try out.print("SIZE {d} bill {s} saving {s} cost {s} payback ", .{ capacity, try money(gpa, trial.bill), try money(gpa, saving), try money(gpa, cost) });
423 if (saving > 0) {
424 const w = @divFloor(10 * cost, saving);
425 try out.print("{d}.{d}\n", .{ @divFloor(w, 10), @mod(w, 10) });
426 if (best == null or cost * best.?.saving < best.?.cost * saving) best = .{ .capacity = capacity, .cost = cost, .saving = saving };
427 } else try out.writeAll("never\n");
428 }
429 if (best) |b| try out.print("BEST {d}\n", .{b.capacity}) else try out.writeAll("BEST none\n");
430 },
431 .SWEEP => {
432 if (a.len < 5) return app.fail(gpa, "wrong number of arguments", .{});
433 const item = try app.itemArgument(gpa, a[0]);
434 const low = integer(a[1]);
435 const high = integer(a[2]);
436 const step = integer(a[3]);
437 try app.range(gpa, low, high, step, item.minimum(), item.maximum());
438 const strategy = try app.oneStrategy(gpa, a[4..]);
439 var best: ?struct { value: i64, bill: i64 } = null;
440 var value = low;
441 while (value <= high) : (value += step) {
442 var s = app.settings;
443 item.set(&s, value);
444 var trial = try simulate(app.data, s, app.prices, strategy, null);
445 defer trial.deinit();
446 var base = try baseline(app.data, s, app.prices);
447 defer base.deinit();
448 try out.print("SWEEP {s} {d} bill {s} saving {s} sufficiency {s}%\n", .{ a[0], value, try money(gpa, trial.bill), try money(gpa, base.bill - trial.bill), try trial.sufficiency(gpa) });
449 if (best == null or trial.bill < best.?.bill) best = .{ .value = value, .bill = trial.bill };
450 }
451 try out.print("BEST {d}\n", .{best.?.value});
452 },
453 .OUTAGE => {
454 try app.arity(gpa, a, 3);
455 const day = try app.dayArgument(gpa, a[0]);
456 const start = integer(a[1]);
457 const end = integer(a[2]);
458 if (start < 0 or start > end or end > 23) return app.fail(gpa, "invalid hours", .{});
459 const r = try app.result(gpa);
460 const first = day * 24 + ix(start);
461 const last = day * 24 + ix(end);
462 var trial = try simulate(app.data, r.settings, r.prices, r.strategy, .{ .first = first, .last = last });
463 defer trial.deinit();
464 const before = if (first == 0) trial.starting else trial.records[first - 1].values.soc;
465 try out.print("OUTAGE day {d} hours {d}-{d} strategy {s} soc {d}\nhour load pv disch soc short curt\n", .{ day + 1, start, end, try r.strategy.text(gpa), before });
466 for (trial.records[first .. last + 1], ix(start)..) |record, h| {
467 const v = record.values;
468 try out.print("{d:>4}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}\n", .{ h, v.load, v.pv, v.discharge, v.soc, v.shortfall, v.curtailed });
469 }
470 const v = trial.sum(first, last + 1);
471 const served = if (v.load == 0) "100.00" else try percentage(gpa, v.load - v.shortfall, v.load);
472 try out.print("TOTAL load {d} unserved {d} served {s}% bill {s}\n", .{ v.load, v.shortfall, served, try money(gpa, trial.bill) });
473 },
474 .DIFF => {
475 if (a.len < 2) return app.fail(gpa, "wrong number of arguments", .{});
476 const day = try app.dayArgument(gpa, a[0]);
477 const strategy = try app.oneStrategy(gpa, a[1..]);
478 const r = try app.result(gpa);
479 var trial = try simulate(app.data, r.settings, r.prices, strategy, null);
480 defer trial.deinit();
481 try out.print("DIFF day {d} {s} {s} vs {s}\nhour imp1 imp2 exp1 exp2 soc1 soc2\n", .{ day + 1, app.data.dayName(day), try r.strategy.text(gpa), try strategy.text(gpa) });
482 const first = day * 24;
483 const last = first + 24;
484 for (r.records[first..last], trial.records[first..last], 0..) |left, right, h| try out.print("{d:>4}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}\n", .{ h, left.values.imported, right.values.imported, left.values.exported, right.values.exported, left.values.soc, right.values.soc });
485 const cost1 = r.sum(first, last).net();
486 const cost2 = trial.sum(first, last).net();
487 try out.print("TOTAL cost1 {s} cost2 {s} difference {s}\n", .{ try money(gpa, cost1), try money(gpa, cost2), try money(gpa, cost2 - cost1) });
488 },
489 .HOURLY, .CHART => {
490 try app.arity(gpa, a, 1);
491 const day = try app.dayArgument(gpa, a[0]);
492 try app.report(gpa, out, cmd, day, 0);
493 },
494 .PEAKS => {
495 try app.arity(gpa, a, 1);
496 const count = integer(a[0]);
497 if (count < 1) return app.fail(gpa, "invalid count", .{});
498 try app.report(gpa, out, cmd, 0, ix(count));
499 },
500 else => {
501 try app.arity(gpa, a, 0);
502 try app.report(gpa, out, cmd, 0, 0);
503 },
504 }
505 return false;
506 }
507 fn report(app: *App, gpa: Allocator, out: *std.Io.Writer, cmd: Command, day: usize, count: usize) Error!void {
508 const r = try app.result(gpa);
509 const label = try r.strategy.text(gpa);
510 const total = r.total;
511 switch (cmd) {
512 .HOURLY => {
513 try out.print("HOURLY day {d} {s} strategy {s}\nhour load pv charge disch soc import export curt\n", .{ day + 1, app.data.dayName(day), label });
514 for (r.records[day * 24 .. (day + 1) * 24], 0..) |record, h| {
515 const v = record.values;
516 try out.print("{d:>4}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}\n", .{ h, v.load, v.pv, v.charge, v.discharge, v.soc, v.imported, v.exported, v.curtailed });
517 }
518 const v = r.sum(day * 24, (day + 1) * 24);
519 try out.print(" sum{d:>7}{d:>7}{d:>7}{d:>7} {d:>7}{d:>7}{d:>7}\n", .{ v.load, v.pv, v.charge, v.discharge, v.imported, v.exported, v.curtailed });
520 },
521 .DAILY => {
522 try out.print("DAILY strategy {s}\nday name load pv import export curt soc cost\n", .{label});
523 for (0..app.data.days()) |d| {
524 const v = r.sum(d * 24, (d + 1) * 24);
525 try out.print("{d:>3}{s:>5}{d:>8}{d:>8}{d:>8}{d:>8}{d:>8}{d:>8}{s:>10}\n", .{ d + 1, app.data.dayName(d), v.load, v.pv, v.imported, v.exported, v.curtailed, r.records[d * 24 + 23].values.soc, try money(gpa, v.net()) });
526 }
527 try out.print("TOTAL load {d} pv {d} import {d} export {d} curtailed {d} cost {s}\n", .{ total.load, total.pv, total.imported, total.exported, total.curtailed, try money(gpa, total.net()) });
528 },
529 .BILL => {
530 try out.print("BILL strategy {s} days {d}\n", .{ label, app.data.days() });
531 for (app.data.period_names, 0..) |name, p| try out.print("IMPORT {s} {d} Wh {s}\n", .{ name, r.periods[p].imported, try money(gpa, r.periods[p].import_cost) });
532 try out.print("EXPORT {d} Wh {s}\nFIXED {s}\nDEMAND {d} Wh {s}\nTOTAL {s}\n", .{ total.exported, try money(gpa, -total.export_income), try money(gpa, r.settings.fixed), r.peak, try money(gpa, r.demand_charge), try money(gpa, r.bill) });
533 },
534 .SAVINGS => {
535 var base = try baseline(app.data, r.settings, r.prices);
536 defer base.deinit();
537 try out.print("SAVINGS strategy {s}\n", .{label});
538 for (app.data.period_names, 0..) |name, p| try savingsRow(gpa, out, name, base.periods[p].import_cost, r.periods[p].import_cost);
539 try savingsRow(gpa, out, "EXPORT", -base.total.export_income, -total.export_income);
540 try savingsRow(gpa, out, "DEMAND", base.demand_charge, r.demand_charge);
541 try savingsRow(gpa, out, "TOTAL", base.bill, r.bill);
542 },
543 .BYPERIOD => {
544 try out.print("BYPERIOD strategy {s}\n", .{label});
545 for (app.data.period_names, 0..) |name, p| {
546 const v = r.periods[p];
547 try out.print("{s} hours {d} load {d} pv {d} charge {d} discharge {d} import {d} export {d}\n", .{ name, r.period_hours[p], v.load, v.pv, v.charge, v.discharge, v.imported, v.exported });
548 }
549 },
550 .DAYTYPE => {
551 try out.print("DAYTYPE strategy {s}\n", .{label});
552 for ([_][]const u8{ "WEEKDAY", "WEEKEND" }, 0..) |name, kind| {
553 var v: Metrics = .{};
554 var days: i64 = 0;
555 for (0..app.data.days()) |d| {
556 if (app.data.weekend(d) != (kind == 1)) continue;
557 days += 1;
558 v.add(r.sum(d * 24, (d + 1) * 24));
559 }
560 try out.print("{s} days {d} load {d} pv {d} import {d} export {d} cost {s}\n", .{ name, days, @divFloor(v.load, days), @divFloor(v.pv, days), @divFloor(v.imported, days), @divFloor(v.exported, days), try money(gpa, @divTrunc(v.net(), days)) });
561 }
562 },
563 .PEAKS => {
564 try out.print("PEAKS strategy {s}\n", .{label});
565 var peaks: std.ArrayList(Peak) = .empty;
566 for (r.records, 0..) |record, i| if (record.values.imported > 0) try peaks.append(gpa, .{ .index = i, .imported = record.values.imported });
567 std.mem.sort(Peak, peaks.items, {}, Peak.less);
568 if (peaks.items.len == 0) try out.writeAll("none\n");
569 for (peaks.items[0..@min(count, peaks.items.len)], 1..) |peak, rank| {
570 const record = r.records[peak.index];
571 try out.print("{d}. day {d} hour {d} {s} import {d} load {d} soc {d}\n", .{ rank, peak.index / 24 + 1, peak.index % 24, app.data.period_names[record.period], record.values.imported, record.values.load, record.values.soc });
572 }
573 },
574 .PROFILE => {
575 try out.print("PROFILE strategy {s} days {d}\nhour load pv import export soc\n", .{ label, app.data.days() });
576 const days: i64 = @intCast(app.data.days());
577 for (0..24) |h| {
578 var v: Metrics = .{};
579 for (0..app.data.days()) |d| v.add(r.records[d * 24 + h].values);
580 try out.print("{d:>4}{d:>7}{d:>7}{d:>7}{d:>7}{d:>7}\n", .{ h, @divFloor(v.load, days), @divFloor(v.pv, days), @divFloor(v.imported, days), @divFloor(v.exported, days), @divFloor(v.soc, days) });
581 }
582 },
583 .FLOWS => {
584 try out.print("FLOWS strategy {s}\n", .{label});
585 for (flow_names, r.flows) |name, flow| try out.print("{s} {d}\n", .{ name, flow });
586 try out.print("LOSSES {d}\nSTORED {d}\n", .{ total.loss, r.endSoc() - r.starting });
587 },
588 .SELFSUFF => try out.print("SELFSUFF sufficiency {s}% consumption {s}%\n", .{ try r.sufficiency(gpa), try percentage(gpa, r.flows[0] + r.flows[1], total.pv) }),
589 .CYCLES => {
590 var a = total.drawn;
591 var b = r.settings.capacity;
592 while (b != 0) {
593 const remainder = @mod(a, b);
594 a = b;
595 b = remainder;
596 }
597 const cycles = @divFloor(1000 * total.drawn, r.settings.capacity);
598 try out.print("CYCLES {d}/{d} {d}.{d:0>3} drawn {d} losses {d}\n", .{ @divExact(total.drawn, a), @divExact(r.settings.capacity, a), @divFloor(cycles, 1000), @mod(cycles, 1000), total.drawn, total.loss });
599 },
600 .USAGE => {
601 var charging: usize = 0;
602 var discharging: usize = 0;
603 var idle: usize = 0;
604 var full: usize = 0;
605 var low: usize = 0;
606 var streak: usize = 0;
607 var longest: usize = 0;
608 var first: usize = 0;
609 for (r.records, 0..) |record, i| {
610 const v = record.values;
611 charging += @intFromBool(v.charge > 0);
612 discharging += @intFromBool(v.drawn > 0);
613 idle += @intFromBool(v.charge == 0 and v.drawn == 0);
614 full += @intFromBool(v.soc == r.settings.capacity);
615 if (v.soc <= r.reserve) {
616 low += 1;
617 streak += 1;
618 if (streak > longest) {
619 longest = streak;
620 first = i + 1 - streak;
621 }
622 } else streak = 0;
623 }
624 try out.print("USAGE charging {d} discharging {d} idle {d} full {d} low {d} longest {d}", .{ charging, discharging, idle, full, low, longest });
625 if (longest > 0) try out.print(" from day {d} hour {d}", .{ first / 24 + 1, first % 24 });
626 try out.writeAll("\n");
627 },
628 .HISTOGRAM => {
629 try out.print("HISTOGRAM strategy {s}\n", .{label});
630 var bands: [10]i64 = @splat(0);
631 for (r.records) |record| bands[ix(@min(9, @divFloor(10 * record.values.soc, r.settings.capacity)))] += 1;
632 for (bands, 0..) |frequency, band| {
633 try out.print("{d:>3}% ", .{band * 10});
634 try bar(out, @divFloor(20 * frequency, @as(i64, @intCast(r.records.len))));
635 try out.print(" {d}\n", .{frequency});
636 }
637 },
638 .CHART => {
639 try out.print("CHART day {d} {s} strategy {s}\n", .{ day + 1, app.data.dayName(day), label });
640 for (r.records[day * 24 .. (day + 1) * 24], 0..) |record, h| {
641 const soc = record.values.soc;
642 try out.print("{d:>2} ", .{h});
643 try bar(out, @divFloor(20 * soc, r.settings.capacity));
644 try out.print(" {s}%\n", .{try percentage(gpa, soc, r.settings.capacity)});
645 }
646 },
647 else => unreachable,
648 }
649 }
650};

656fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
657 var buffer: std.Io.Writer.Allocating = .init(gpa);
658 defer buffer.deinit();
659 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
660 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
661 const line = try buffer.toOwnedSlice();
662 var words: std.ArrayList([]const u8) = .empty;
663 var split = std.mem.splitScalar(u8, line, ' ');
664 while (split.next()) |word| try words.append(gpa, word);
665 return words.items;
666}
667fn readInput(gpa: Allocator, input: *std.Io.Reader) !App {
668 const battery = try readWords(gpa, input);
669 const grid = try readWords(gpa, input);
670 const settings: Settings = .{ .capacity = integer(battery[1]), .initial = integer(battery[2]), .charge = integer(battery[3]), .discharge = integer(battery[4]), .ceff = integer(battery[5]), .deff = integer(battery[6]), .reserve = integer(battery[7]), .cost = integer(battery[8]), .export_limit = integer(grid[1]), .feedin = integer(grid[2]), .fixed = integer(grid[3]), .demand = integer(grid[4]) };
671 const period_count = ix(integer((try readWords(gpa, input))[1]));
672 const names = try gpa.alloc([]const u8, period_count);
673 var prices: [6]i64 = @splat(0);
674 for (names, 0..) |*name, p| {
675 const words = try readWords(gpa, input);
676 name.* = words[0];
677 prices[p] = integer(words[1]);
678 }
679 var divisions: [2][24]usize = undefined;
680 for (&divisions) |*division| {
681 const words = try readWords(gpa, input);
682 var j: usize = 1;
683 while (j < words.len) : (j += 2) {
684 const first = ix(integer(words[j]));
685 const last = if (j + 2 < words.len) ix(integer(words[j + 2])) else 24;
686 var p: usize = 0;
687 while (!equal(names[p], words[j + 1])) p += 1;
688 @memset(division[first..last], p);
689 }
690 }
691 const Profile = struct { kind: []const u8, name: []const u8, values: [24]i64 };
692 const profile_count = ix(integer((try readWords(gpa, input))[1]));
693 const profiles = try gpa.alloc(Profile, profile_count);
694 for (profiles) |*profile| {
695 const words = try readWords(gpa, input);
696 profile.* = .{ .kind = words[0], .name = words[1], .values = undefined };
697 for (&profile.values, words[2..]) |*value, word| value.* = integer(word);
698 }
699 const days = ix(integer((try readWords(gpa, input))[1]));
700 const hours = try gpa.alloc(HourInput, days * 24);
701 for (0..days) |day| {
702 const words = try readWords(gpa, input);
703 var load: *const Profile = undefined;
704 var pv: *const Profile = undefined;
705 for (profiles) |*profile| {
706 if (equal(profile.kind, "LOAD") and equal(profile.name, words[0])) load = profile;
707 if (equal(profile.kind, "PV") and equal(profile.name, words[2])) pv = profile;
708 }
709 for (0..24) |h| hours[day * 24 + h] = .{ .load = @divFloor(load.values[h] * integer(words[1]), 100), .pv = @divFloor(pv.values[h] * integer(words[3]), 100) };
710 }
711 const holiday = try gpa.alloc(bool, days);
712 @memset(holiday, false);
713 const holiday_words = try readWords(gpa, input);
714 for (holiday_words[2..]) |word| holiday[ix(integer(word) - 1)] = true;
715 const data: Data = .{ .period_names = names, .hours = hours, .holiday = holiday };
716 for (hours, 0..) |*hour, i| hour.period = divisions[@intFromBool(data.weekend(i / 24))][i % 24];
717 const overrides = ix(integer((try readWords(gpa, input))[1]));
718 for (0..overrides) |_| {
719 const words = try readWords(gpa, input);
720 const i = ix(integer(words[0]) - 1) * 24 + ix(integer(words[1]));
721 hours[i].load = integer(words[2]);
722 hours[i].pv = integer(words[3]);
723 }
724 return .{ .data = data, .settings = settings, .prices = prices, .initial_settings = settings, .initial_prices = prices };
725}

727pub fn main(init: std.process.Init) !void {
728 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
729 defer arena.deinit();
730 var input_buffer: [8192]u8 = undefined;
731 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
732 var output_buffer: [8192]u8 = undefined;
733 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
734 var app = try readInput(arena.allocator(), &input.interface);
735 defer if (app.current) |*r| r.deinit();
736 while (true) {
737 var command_arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
738 defer command_arena.deinit();
739 const gpa = command_arena.allocator();
740 const words = try readWords(gpa, &input.interface);
741 var response: std.Io.Writer.Allocating = .init(gpa);
742 const done = app.execute(gpa, &response.writer, words) catch |err| {
743 if (err != error.BadCommand) return err;
744 try output.interface.print("ERROR {s}\n", .{app.error_message});
745 continue;
746 };
747 try output.interface.writeAll(response.written());
748 if (done) break;
749 }
750 try output.interface.flush();
751}
