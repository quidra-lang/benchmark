1local SETTING_NAMES = {

3 "EXPORT", "FEEDIN", "FIXED", "DEMAND", "PVSCALE"

8 RESERVE = {0, 100}, COST = {0, 100000}, EXPORT = {0, 100000},

41local function money(amount)
42 local absolute = math.abs(amount)
43 return (amount < 0 and "-" or "") .. tostring(absolute // 100)
44 .. "." .. string.format("%02d", absolute % 100)
45end

47local function percentage(numerator, denominator)
48 local value = denominator ~= 0 and floor(10000 * numerator, denominator) or 0
49 return tostring(value // 100) .. "." .. string.format("%02d", value % 100)
50end

87local function tariff_extremes(prices, names)
88 local cheapest, peak = names[1], names[1]
89 for _, name in ipairs(names) do
90 if prices[name] < prices[cheapest] then cheapest = name end
91 if prices[name] > prices[peak] then peak = name end
92 end
93 return cheapest, peak
94end

96local function parse_strategies(args)
97 local strategies = {}
98 local index = 1
99 while index <= #args do
100 local name = args[index]
101 index = index + 1
102 if name == "SELF" or name == "EXPORTPEAK" then
103 strategies[#strategies + 1] = {name = name}
104 else
105 local range = STRATEGY_RANGES[name]
106 if not range then fail("unknown strategy " .. name) end
107 if index > #args then fail("missing parameter for " .. name) end
108 local parameter = number(args[index])
109 index = index + 1
110 if parameter < range[1] or parameter > range[2] then
111 fail("invalid parameter for " .. name)
112 end
113 strategies[#strategies + 1] = {name = name, parameter = parameter}
114 end
115 end
116 return strategies
117end

119local function one_strategy(args)
120 local strategies = parse_strategies(args)
121 require_count(strategies, 1)
122 return strategies[1]
123end

125local function checked_range(low, high, step, bounds)
126 if low < bounds[1] or high > bounds[2] or high < low or step < 1
127 or floor(high - low, step) + 1 > 20 then
128 fail("invalid range")
129 end
130 local values = {}
131 for value = low, high, step do values[#values + 1] = value end
132 return values
133end

135local function read_period_hours(args)
136 local hours = {}
137 for i = 1, #args, 2 do
138 local start = number(args[i])
139 local finish = i + 2 <= #args and number(args[i + 2]) or 24
140 for hour = start, finish - 1 do hours[#hours + 1] = args[i + 1] end

145local function read_input()
146 local function line() return split(io.read("*l")) end
147 local data = {}
148 local battery = line()
149 local grid = line()
150 local settings = {}
151 for i = 1, 8 do settings[SETTING_NAMES[i]] = number(battery[i + 1]) end
152 for i = 9, 12 do settings[SETTING_NAMES[i]] = number(grid[i - 7]) end
153 settings.PVSCALE = 100
154 data.initial_settings = settings
155
156 local period_count = number(line()[2])
157 data.period_names = {}
158 data.initial_prices = {}
159 for _ = 1, period_count do
160 local args = line()
161 local name = args[1]
162 data.period_names[#data.period_names + 1] = name
163 data.initial_prices[name] = number(args[2])
164 end
165 local weekday = line()
166 local weekend = line()
167 table.remove(weekday, 1)
168 table.remove(weekend, 1)
169 weekday = read_period_hours(weekday)
170 weekend = read_period_hours(weekend)
171
172 local profile_count = number(line()[2])
173 local profiles = {}
174 for _ = 1, profile_count do
175 local args = line()
176 local values = {}
177 for i = 3, #args do values[#values + 1] = number(args[i]) end
178 profiles[args[1] .. "/" .. args[2]] = values
179 end
180 data.day_count = number(line()[2])
181 data.loads = {}
182 data.base_pv = {}
183 for _ = 1, data.day_count do
184 local args = line()
185 local load_values = profiles["LOAD/" .. args[1]]
186 local pv_values = profiles["PV/" .. args[3]]
187 local load_percent, pv_percent = number(args[2]), number(args[4])
188 for hour = 1, 24 do
189 data.loads[#data.loads + 1] = floor(load_values[hour] * load_percent, 100)
190 data.base_pv[#data.base_pv + 1] = floor(pv_values[hour] * pv_percent, 100)
191 end
192 end
193 local holiday_tokens = line()
194 local holidays = {}
195 for i = 3, #holiday_tokens do holidays[number(holiday_tokens[i])] = true end
196 local overrides = number(line()[2])
197 for _ = 1, overrides do
198 local args = line()
199 local index = (number(args[1]) - 1) * 24 + number(args[2]) + 1
200 data.loads[index] = number(args[3])
201 data.base_pv[index] = number(args[4])
202 end
203 data.day_names = {}
204 data.weekend_days = {}
205 data.periods = {}
206 for day = 1, data.day_count do
207 local holiday = not not holidays[day]
208 local weekend_day = holiday or (day - 1) % 7 >= 5
209 data.day_names[day] = holiday and "HOL" or DAY_NAMES[(day - 1) % 7 + 1]
210 data.weekend_days[day] = weekend_day
211 local schedule = weekend_day and weekend or weekday
212 for hour = 1, 24 do data.periods[#data.periods + 1] = schedule[hour] end
213 end
214 return data
215end

217local function hour_record(index, period, load, pv, soc, flows, drawn, shortfall, loss, price, feedin)
218 local record = {
219 index = index - 1, day = floor(index - 1, 24) + 1, hour = (index - 1) % 24,
220 period = period, load = load, pv = pv, soc = soc, flows = flows,
221 charge = flows[2] + flows[8], discharge = flows[5] + flows[6],
222 imported = flows[7] + flows[8], exported = flows[3] + flows[6],
223 curtailed = flows[4], drawn = drawn, shortfall = shortfall, loss = loss
224 }
225 record.import_cost = floor(record.imported * price, 100000)
226 record.export_income = floor(record.exported * feedin, 100000)
227 record.net_cost = record.import_cost - record.export_income
228 return record
229end

231local function totals(records)
232 local result = {count = #records, flows = {}}
233 for _, field in ipairs(TOTAL_FIELDS) do result[field] = 0 end
234 for index = 1, #FLOW_NAMES do result.flows[index] = 0 end
235 for _, record in ipairs(records) do
236 for _, field in ipairs(TOTAL_FIELDS) do result[field] = result[field] + record[field] end
237 for index, value in ipairs(record.flows) do
238 result.flows[index] = result.flows[index] + value
239 end
240 end
241 result.net_cost = result.import_cost - result.export_income
242 return result
243end

245local function simulation_result(strategy, settings, prices, data, start_soc, reserve, records)
246 local result = {
247 strategy = strategy, settings = copy(settings), prices = copy(prices),
248 start_soc = start_soc, reserve = reserve, hours = records,
249 days = {}, daily = {}, by_period = {}
250 }
251 for i = 1, #records, 24 do
252 local daily = {}
253 for j = i, i + 23 do daily[#daily + 1] = records[j] end
254 result.days[#result.days + 1] = daily
255 result.daily[#result.daily + 1] = totals(daily)
256 end
257 for _, period in ipairs(data.period_names) do
258 local subset = {}
259 for _, record in ipairs(records) do
260 if record.period == period then subset[#subset + 1] = record end
261 end
262 result.by_period[period] = totals(subset)
263 end
264 result.total = totals(records)
265 result.peak = 0
266 for _, record in ipairs(records) do result.peak = max(result.peak, record.imported) end
267 result.demand_charge = floor(result.peak * settings.DEMAND, 1000)
268 result.bill = result.total.net_cost + settings.FIXED + result.demand_charge
269 result.end_soc = records[#records].soc
270 result.sufficiency = percentage(max(0, result.total.load - result.total.imported), result.total.load)
271 return result
272end

274local function simulate(data, strategy, settings, prices, outage)
275 local name, parameter = strategy.name, strategy.parameter
276 local capacity, ceff, deff = settings.CAPACITY, settings.CEFF, settings.DEFF
277 local reserve_percent = name == "BACKUP" and max(settings.RESERVE, parameter) or settings.RESERVE
278 local reserve = floor(capacity * reserve_percent, 100)
279 local start_soc = floor(capacity * settings.INITIAL, 100)
280 local soc = start_soc
281 local cheapest, peak = tariff_extremes(prices, data.period_names)
282 local arbitrage = name == "EXPORTPEAK" and settings.FEEDIN > prices[cheapest]
283 local records = {}
284
285 local function discharge_target(deficit, period)
286 if name == "TOU" then return period == peak and deficit or 0 end
287 if name == "PEAKSHAVE" then return max(0, deficit - parameter) end
288 if arbitrage and period == cheapest and period ~= peak then return 0 end
289 return deficit
290 end
291
292 local function grid_charge(period, pv_charge, grid_load)
293 local headroom = settings.CHARGE - pv_charge
294 local target = nil
295 if name == "TOU" and period == cheapest and period ~= peak then
296 target = floor(capacity * parameter, 100)
297 elseif name == "PEAKSHAVE" and period == cheapest then
298 target = capacity
299 headroom = min(headroom, max(0, parameter - grid_load))
300 elseif name == "BACKUP" and period == cheapest then
301 target = reserve
302 elseif arbitrage and period == cheapest and period ~= peak then
303 target = capacity
304 end
305 if target == nil then return 0 end
306 local amount = min(headroom, ceiling(100 * max(0, target - soc), ceff))
307 soc = soc + floor(amount * ceff, 100)
308 return amount
309 end
310
311 for index = 1, #data.loads do
312 local load = data.loads[index]
313 local pv = floor(data.base_pv[index] * settings.PVSCALE, 100)
314 local period = data.periods[index]
315 local direct = min(load, pv)
316 local surplus, deficit = pv - direct, load - direct
317 local pv_charge = min(surplus, settings.CHARGE, ceiling(100 * (capacity - soc), ceff))
318 soc = soc + floor(pv_charge * ceff, 100)
319 surplus = surplus - pv_charge
320 local is_outage = outage ~= nil and outage[1] <= index and index <= outage[2]
321 local target = is_outage and deficit or discharge_target(deficit, period)
322 local lower_bound = is_outage and 0 or reserve
323 local load_drawn = min(ceiling(100 * target, deff), settings.DISCHARGE, max(0, soc - lower_bound))
324 local load_discharge = floor(load_drawn * deff, 100)
325 soc = soc - load_drawn
326 deficit = deficit - load_discharge
327 local grid_load, shortfall, pv_export, curtailed, additional_charge
328 local export_drawn, battery_export = 0, 0
329 if is_outage then
330 grid_load, shortfall, pv_export, curtailed, additional_charge = 0, deficit, 0, surplus, 0
331 else
332 grid_load, shortfall = deficit, 0
333 pv_export = min(surplus, settings.EXPORT)
334 curtailed = surplus - pv_export
335 additional_charge = grid_charge(period, pv_charge, grid_load)
336 if arbitrage and period == peak and pv_charge == 0 then
337 export_drawn = min(ceiling(100 * (settings.EXPORT - pv_export), deff),
338 settings.DISCHARGE - load_drawn, max(0, soc - reserve))
339 battery_export = floor(export_drawn * deff, 100)
340 soc = soc - export_drawn
341 end
342 end
343 local loss = pv_charge - floor(pv_charge * ceff, 100)
344 + additional_charge - floor(additional_charge * ceff, 100)
345 + load_drawn - load_discharge + export_drawn - battery_export
346 local flows = {direct, pv_charge, pv_export, curtailed, load_discharge,
347 battery_export, grid_load, additional_charge}
348 records[#records + 1] = hour_record(index, period, load, pv, soc, flows,
349 load_drawn + export_drawn, shortfall, loss, prices[period], settings.FEEDIN)
350 end
351 return simulation_result(strategy, settings, prices, data, start_soc, reserve, records)
352end

467function Desk:compare(args)
468 local strategies = parse_strategies(args)
469 if #strategies == 0 then fail("wrong number of arguments") end
470 local baseline = self:baseline(self.settings, self.prices)
471 local trials = {}
472 for i, strategy in ipairs(strategies) do
473 trials[#trials + 1] = {result = self:simulate(strategy), index = i}
474 end
475 table.sort(trials, function(a, b)
476 if a.result.bill ~= b.result.bill then return a.result.bill < b.result.bill end
477 return a.index < b.index
478 end)
479 local output = {"COMPARE baseline " .. money(baseline.bill)}
480 for index, item in ipairs(trials) do
481 local result = item.result
482 output[#output + 1] = index .. ". " .. strategy_text(result.strategy)
483 .. " bill " .. money(result.bill) .. " saving " .. money(baseline.bill - result.bill)
484 .. " sufficiency " .. result.sufficiency .. "%"
485 end
486 return output
487end

489function Desk:size(args)
490 require_count(args, 3)
491 local capacities = checked_range(number(args[1]), number(args[2]), number(args[3]), SETTING_RANGES.CAPACITY)
492 local baseline = self:baseline(self.settings, self.prices)
493 local output, best = {}, nil
494 for _, capacity in ipairs(capacities) do
495 local settings = copy(self.settings)
496 settings.CAPACITY = capacity
497 local result = self:simulate({name = "SELF"}, settings)
498 local saving = baseline.bill - result.bill
499 local cost = floor(capacity * settings.COST, 1000)
500 local payback = "never"
501 if saving > 0 then
502 local tenths = floor(10 * cost, saving)
503 payback = tostring(tenths // 10) .. "." .. tostring(tenths % 10)
504 if best == nil or cost * best.saving < best.cost * saving then
505 best = {capacity = capacity, cost = cost, saving = saving}
506 end
507 end
508 output[#output + 1] = "SIZE " .. capacity .. " bill " .. money(result.bill)
509 .. " saving " .. money(saving) .. " cost " .. money(cost) .. " payback " .. payback
510 end
511 output[#output + 1] = "BEST " .. (best and best.capacity or "none")
512 return output
513end

515function Desk:sweep(args)
516 if #args < 5 then fail("wrong number of arguments") end
517 local item = args[1]
518 local bounds = SETTING_RANGES[item]
519 if not bounds then fail("unknown parameter " .. item) end
520 local values = checked_range(number(args[2]), number(args[3]), number(args[4]), bounds)
521 local rest = {}
522 for i = 5, #args do rest[#rest + 1] = args[i] end
523 local strategy = one_strategy(rest)
524 local output, best = {}, nil
525 for _, value in ipairs(values) do
526 local settings = copy(self.settings)
527 settings[item] = value
528 local result = self:simulate(strategy, settings)
529 local baseline = self:baseline(settings, self.prices)
530 if best == nil or result.bill < best.bill then best = {value = value, bill = result.bill} end
531 output[#output + 1] = "SWEEP " .. item .. " " .. value .. " bill " .. money(result.bill)
532 .. " saving " .. money(baseline.bill - result.bill)
533 .. " sufficiency " .. result.sufficiency .. "%"
534 end
535 output[#output + 1] = "BEST " .. best.value
536 return output
537end

539function Desk:outage(args)
540 require_count(args, 3)
541 local day = self:checked_day(args[1])
542 local start_hour, end_hour = number(args[2]), number(args[3])
543 if start_hour < 0 or end_hour > 23 or start_hour > end_hour then fail("invalid hours") end
544 local current = self:require_result()
545 local first = (day - 1) * 24 + start_hour + 1
546 local last = (day - 1) * 24 + end_hour + 1
547 local result = self:simulate(current.strategy, current.settings, current.prices, {first, last})
548 local before = first > 1 and result.hours[first - 1].soc or result.start_soc
549 local output = {"OUTAGE day " .. day .. " hours " .. start_hour .. "-" .. end_hour
550 .. " strategy " .. strategy_text(result.strategy) .. " soc " .. before,
551 "hour load pv disch soc short curt"}
552 local records = {}
553 for i = first, last do
554 local record = result.hours[i]
555 records[#records + 1] = record
556 output[#output + 1] = row({record.hour, record.load, record.pv, record.discharge,
557 record.soc, record.shortfall, record.curtailed}, {4, 7, 7, 7, 7, 7, 7})
558 end
559 local total = totals(records)
560 local served = total.load > 0 and percentage(total.load - total.shortfall, total.load) or "100.00"
561 output[#output + 1] = "TOTAL load " .. total.load .. " unserved " .. total.shortfall
562 .. " served " .. served .. "% bill " .. money(result.bill)
563 return output
564end

566function Desk:diff(args)
567 if #args < 2 then fail("wrong number of arguments") end
568 local day = self:checked_day(args[1])
569 local remaining = {}
570 for i = 2, #args do remaining[#remaining + 1] = args[i] end
571 local strategy = one_strategy(remaining)
572 local current = self:require_result()
573 local comparison = self:simulate(strategy, current.settings, current.prices)
574 local output = {"DIFF day " .. day .. " " .. self.data.day_names[day] .. " "
575 .. strategy_text(current.strategy) .. " vs " .. strategy_text(strategy),
576 "hour imp1 imp2 exp1 exp2 soc1 soc2"}
577 local before, after = current.days[day], comparison.days[day]
578 for i = 1, 24 do
579 local a, b = before[i], after[i]
580 output[#output + 1] = row({a.hour, a.imported, b.imported, a.exported,
581 b.exported, a.soc, b.soc}, {4, 7, 7, 7, 7, 7, 7})
582 end
583 local cost1, cost2 = current.daily[day].net_cost, comparison.daily[day].net_cost
584 output[#output + 1] = "TOTAL cost1 " .. money(cost1) .. " cost2 " .. money(cost2)
585 .. " difference " .. money(cost2 - cost1)
586 return output
587end

589function Desk:hourly(args)
590 local day, result = self:result_for_day(args)
591 local output = {"HOURLY day " .. day .. " " .. self.data.day_names[day]

694function Desk:peaks(args)
695 require_count(args, 1)
696 local amount = number(args[1])
697 if amount < 1 then fail("invalid count") end
698 local result = self:require_result()
699 local records = {}
700 for _, record in ipairs(result.hours) do
701 if record.imported > 0 then records[#records + 1] = record end
702 end
703 table.sort(records, function(a, b)
704 if a.imported ~= b.imported then return a.imported > b.imported end
705 return a.index < b.index
706 end)
707 local output = {"PEAKS strategy " .. strategy_text(result.strategy)}
708 for index = 1, min(amount, #records) do
709 local record = records[index]
710 output[#output + 1] = index .. ". day " .. record.day .. " hour " .. record.hour
711 .. " " .. record.period .. " import " .. record.imported
712 .. " load " .. record.load .. " soc " .. record.soc
713 end
714 if #records == 0 then output[#output + 1] = "none" end
715 return output
716end

766function Desk:usage(args)
767 local result = self:result_without_arguments(args)
768 local hours = result.hours
769 local charging, discharging, idle, full = 0, 0, 0, 0
770 local low, run_length, longest, first = 0, 0, 0, nil
771 for i, record in ipairs(hours) do
772 if record.charge > 0 then charging = charging + 1 end
773 if record.drawn > 0 then discharging = discharging + 1 end
774 if record.charge == 0 and record.drawn == 0 then idle = idle + 1 end
775 if record.soc == result.settings.CAPACITY then full = full + 1 end
776 if record.soc <= result.reserve then
777 low = low + 1
778 run_length = run_length + 1
779 if run_length > longest then
780 longest = run_length
781 first = hours[i - run_length + 1]
782 end
783 else
784 run_length = 0
785 end
786 end
787 local text = "USAGE charging " .. charging .. " discharging " .. discharging
788 .. " idle " .. idle .. " full " .. full .. " low " .. low .. " longest " .. longest
789 if first then text = text .. " from day " .. first.day .. " hour " .. first.hour end
790 return {text}
791end

833function Desk:execute(tokens)
834 local command = tokens[1]
835 local handler = HANDLERS[command]
836 if handler == nil then return {"ERROR unknown command " .. command} end
837 local args = {}
838 for i = 2, #tokens do args[#args + 1] = tokens[i] end
839 local success, output = pcall(self[handler], self, args)
840 if success then return output end
841 if type(output) ~= "table" or output.reason == nil then error(output, 0) end
842 return {"ERROR " .. output.reason}
843end

849 local line = io.read("*l")
850 if not line then break end
851 local tokens = split(line)

853 local lines = desk:execute(tokens)
854 for _, result in ipairs(lines) do output[#output + 1] = result end
855end
856if #output > 0 then io.write(table.concat(output, "\n"), "\n") end
