28const sum = (numbers: number[]): number => numbers.reduce((total, value) => total + value, 0);
29const quotient = (a: number, b: number): number => Math.floor(a / b);
30const ceiling = (a: number, b: number): number => Math.floor((a + b - 1) / b);
31const money = (amount: number): string => (amount < 0 ? '-' : '') + quotient(Math.abs(amount), 100) +
32 '.' + String(Math.abs(amount) % 100).padStart(2, '0');
33const percentage = (numerator: number, denominator: number): string => {
34 const value = denominator ? quotient(10000 * numerator, denominator) : 0;
35 return quotient(value, 100) + '.' + String(value % 100).padStart(2, '0');
36};
37const tableRow = (values: (number | string)[], widths: number[]): string => values.map((value, i) => String(value).padStart(widths[i])).join('');
38const bar = (width: number): string => '|' + '#'.repeat(width) + '.'.repeat(20 - width) + '|';
39const strategyText = ([name, parameter]: Strategy): string => parameter === null ? name : `${name} ${parameter}`;

41function tariffExtremes(prices: Prices): [string, string] {
42 const names = Object.keys(prices);
43 let cheapest = names[0], peak = names[0];
44 for (const name of names) {
45 if (prices[name] < prices[cheapest]) cheapest = name;
46 if (prices[name] > prices[peak]) peak = name;
47 }
48 return [cheapest, peak];
49}

51function readStrategies(tokens: string[]): Strategy[] {
52 const strategies: Strategy[] = [];
53 for (let index = 0; index < tokens.length;) {
54 const name = tokens[index++];
55 if (name === 'SELF' || name === 'EXPORTPEAK') {
56 strategies.push([name, null]);
57 continue;
58 }
59 if (!Object.hasOwn(STRATEGY_RANGES, name)) throw new CommandError('unknown strategy ' + name);
60 if (index === tokens.length) throw new CommandError('missing parameter for ' + name);
61 const parameter = Number(tokens[index++]);
62 const [low, high] = STRATEGY_RANGES[name];
63 if (parameter < low || parameter > high) throw new CommandError('invalid parameter for ' + name);
64 strategies.push([name, parameter]);
65 }
66 return strategies;
67}
68
69function requireCount(values: unknown[], expected: number): void {
70 if (values.length !== expected) throw new CommandError('wrong number of arguments');
71}
72function oneStrategy(values: string[]): Strategy {
73 const result = readStrategies(values);
74 requireCount(result, 1);
75 return result[0];
76}
77function checkedRange(low: number, high: number, step: number, [minimum, maximum]: [number, number]): number[] {
78 if (low < minimum || high > maximum || high < low || step < 1 ||
79 quotient(high - low, step) + 1 > 20) throw new CommandError('invalid range');
80 const values: number[] = [];
81 for (let value = low; value <= high; value += step) values.push(value);
82 return values;
83}
84function gcd(a: number, b: number): number {
85 while (b !== 0) [a, b] = [b, a % b];
86 return a;
87}

89class InputData {
90 initialSettings: Settings;
91 initialPrices: Prices;
92 dayCount: number;
93 loads: number[];
94 basePV: number[];
95 dayNames: string[];
96 weekendDays: boolean[];
97 periods: string[];
98 commands: string[];
99
100 constructor(lines: string[]) {
101 let position = 0;
102 const read = (): string[] => lines[position++].split(' ');
103 const battery = read().slice(1).map(Number);
104 const grid = read().slice(1).map(Number);
105 this.initialSettings = Object.fromEntries(
106 Object.keys(SETTING_RANGES).map((key, i) => [key, [...battery, ...grid, 100][i]])
107 );
108 const periodCount = Number(read()[1]);
109 this.initialPrices = {};
110 for (let i = 0; i < periodCount; ++i) {
111 const [name, price] = read();
112 this.initialPrices[name] = Number(price);
113 }
114 const weekday = InputData.readPeriodHours(read().slice(1));
115 const weekend = InputData.readPeriodHours(read().slice(1));
116 const profileCount = Number(read()[1]);
117 const profiles = new Map<string, number[]>();
118 for (let i = 0; i < profileCount; ++i) {
119 const [kind, name, ...values] = read();
120 profiles.set(kind + '/' + name, values.map(Number));
121 }
122 this.dayCount = Number(read()[1]);
123 this.loads = [];
124 this.basePV = [];
125 for (let day = 0; day < this.dayCount; ++day) {
126 const [loadName, loadPct, pvName, pvPct] = read();
127 this.loads.push(...profiles.get('LOAD/' + loadName)!.map(v => quotient(v * Number(loadPct), 100)));
128 this.basePV.push(...profiles.get('PV/' + pvName)!.map(v => quotient(v * Number(pvPct), 100)));
129 }
130 const holidays = new Set(read().slice(2).map(Number));
131 const overrideCount = Number(read()[1]);
132 for (let i = 0; i < overrideCount; ++i) {
133 const [day, hour, load, pv] = read().map(Number);
134 const index = (day - 1) * 24 + hour;
135 this.loads[index] = load;
136 this.basePV[index] = pv;
137 }
138 this.dayNames = [];
139 this.weekendDays = [];
140 this.periods = [];
141 for (let day = 1; day <= this.dayCount; ++day) {
142 const holiday = holidays.has(day);
143 const weekendDay = holiday || (day - 1) % 7 >= 5;
144 this.dayNames.push(holiday ? 'HOL' : DAY_NAMES[(day - 1) % 7]);
145 this.weekendDays.push(weekendDay);
146 this.periods.push(...(weekendDay ? weekend : weekday));
147 }
148 this.commands = lines.slice(position);
149 }
150
151 static readPeriodHours(tokens: string[]): string[] {
152 const hours: string[] = [];
153 for (let i = 0; i < tokens.length; i += 2) {
154 const start = Number(tokens[i]);
155 const end = i + 2 < tokens.length ? Number(tokens[i + 2]) : 24;
156 for (let hour = start; hour < end; ++hour) hours.push(tokens[i + 1]);
157 }
158 return hours;
159 }
160}

162class HourRecord {
163 index: number;
164 day: number;
165 hour: number;
166 period: string;
167 load: number;
168 pv: number;
169 soc: number;
170 flows: number[];
171 charge: number;
172 discharge: number;
173 imported: number;
174 exported: number;
175 curtailed: number;
176 drawn: number;
177 shortfall: number;
178 loss: number;
179 import_cost: number;
180 export_income: number;
181 net_cost: number;
182
183 constructor(index: number, period: string, load: number, pv: number, soc: number,
184 flows: number[], drawn: number, shortfall: number, loss: number,
185 price: number, feedin: number) {
186 this.index = index;
187 this.day = quotient(index, 24) + 1;
188 this.hour = index % 24;
189 this.period = period;
190 this.load = load;
191 this.pv = pv;
192 this.soc = soc;
193 this.flows = flows;
194 this.charge = flows[1] + flows[7];
195 this.discharge = flows[4] + flows[5];
196 this.imported = flows[6] + flows[7];
197 this.exported = flows[2] + flows[5];
198 this.curtailed = flows[3];
199 this.drawn = drawn;
200 this.shortfall = shortfall;
201 this.loss = loss;
202 this.import_cost = quotient(this.imported * price, 100000);
203 this.export_income = quotient(this.exported * feedin, 100000);
204 this.net_cost = this.import_cost - this.export_income;
205 }
206}

208class Totals {
209 count: number;
210 load = 0;
211 pv = 0;
212 charge = 0;
213 discharge = 0;
214 imported = 0;
215 exported = 0;
216 curtailed = 0;
217 drawn = 0;
218 loss = 0;
219 shortfall = 0;
220 import_cost = 0;
221 export_income = 0;
222 flows: number[];
223 net_cost: number;
224
225 constructor(records: HourRecord[]) {
226 this.count = records.length;
227 for (const field of TOTAL_FIELDS) this[field] = sum(records.map(row => row[field]));
228 this.flows = FLOW_NAMES.map((_, i) => sum(records.map(row => row.flows[i])));
229 this.net_cost = this.import_cost - this.export_income;
230 }

233class SimulationResult {
234 strategy: Strategy;
235 settings: Settings;
236 prices: Prices;
237 startSOC: number;
238 reserve: number;
239 hours: HourRecord[];
240 days: HourRecord[][];
241 daily: Totals[];
242 byPeriod: Record<string, Totals>;
243 total: Totals;
244 peak: number;
245 demandCharge: number;
246 bill: number;
247 endSOC: number;
248 sufficiency: string;
249
250 constructor(strategy: Strategy, settings: Settings, prices: Prices,
251 startSOC: number, reserve: number, records: HourRecord[]) {
252 this.strategy = strategy;
253 this.settings = {...settings};
254 this.prices = {...prices};
255 this.startSOC = startSOC;
256 this.reserve = reserve;
257 this.hours = records;
258 this.days = [];
259 for (let i = 0; i < records.length; i += 24) this.days.push(records.slice(i, i + 24));
260 this.daily = this.days.map(day => new Totals(day));
261 this.byPeriod = {};
262 for (const period of Object.keys(prices)) {
263 this.byPeriod[period] = new Totals(records.filter(record => record.period === period));
264 }
265 this.total = new Totals(records);
266 this.peak = Math.max(...records.map(record => record.imported));
267 this.demandCharge = quotient(this.peak * settings.DEMAND, 1000);
268 this.bill = this.total.net_cost + settings.FIXED + this.demandCharge;
269 this.endSOC = records.at(-1)!.soc;
270 this.sufficiency = percentage(Math.max(0, this.total.load - this.total.imported), this.total.load);
271 }
272}

274class Simulator {
275 data: InputData;
276 strategy: Strategy;
277 name: string;
278 parameter: number | null;
279 settings: Settings;
280 prices: Prices;
281 outage: Outage;
282 capacity: number;
283 ceff: number;
284 deff: number;
285 reserve: number;
286 startSOC: number;
287 soc: number;
288 cheapest: string;
289 peak: string;
290 arbitrage: boolean;
291
292 constructor(data: InputData, strategy: Strategy, settings: Settings,
293 prices: Prices, outage: Outage = null) {
294 this.data = data;
295 this.strategy = strategy;
296 [this.name, this.parameter] = strategy;
297 this.settings = {...settings};
298 this.prices = {...prices};
299 this.outage = outage;
300 this.capacity = settings.CAPACITY;
301 this.ceff = settings.CEFF;
302 this.deff = settings.DEFF;
303 const reservePercent = this.name === 'BACKUP' ? Math.max(settings.RESERVE, this.parameter ?? 0) : settings.RESERVE;
304 this.reserve = quotient(this.capacity * reservePercent, 100);
305 this.startSOC = quotient(this.capacity * settings.INITIAL, 100);
306 this.soc = this.startSOC;
307 [this.cheapest, this.peak] = tariffExtremes(prices);
308 this.arbitrage = this.name === 'EXPORTPEAK' && settings.FEEDIN > prices[this.cheapest];
309 }
310
311 dischargeTarget(deficit: number, period: string): number {
312 if (this.name === 'TOU') return period === this.peak ? deficit : 0;
313 if (this.name === 'PEAKSHAVE') return Math.max(0, deficit - this.parameter!);
314 if (this.arbitrage && period === this.cheapest && period !== this.peak) return 0;
315 return deficit;
316 }
317
318 gridCharge(period: string, pvCharge: number, gridLoad: number): number {
319 let headroom = this.settings.CHARGE - pvCharge;
320 let target = null;
321 if (this.name === 'TOU' && period === this.cheapest && period !== this.peak) {
322 target = quotient(this.capacity * this.parameter!, 100);
323 } else if (this.name === 'PEAKSHAVE' && period === this.cheapest) {
324 target = this.capacity;
325 headroom = Math.min(headroom, Math.max(0, this.parameter! - gridLoad));
326 } else if (this.name === 'BACKUP' && period === this.cheapest) {
327 target = this.reserve;
328 } else if (this.arbitrage && period === this.cheapest && period !== this.peak) {
329 target = this.capacity;
330 }
331 if (target === null) return 0;
332 const amount = Math.min(headroom, ceiling(100 * Math.max(0, target - this.soc), this.ceff));
333 this.soc += quotient(amount * this.ceff, 100);
334 return amount;
335 }
336
337 runHour(index: number): HourRecord {
338 const settings = this.settings;
339 const load = this.data.loads[index];
340 const pv = quotient(this.data.basePV[index] * settings.PVSCALE, 100);
341 const period = this.data.periods[index];
342 const direct = Math.min(load, pv);
343 let surplus = pv - direct;
344 let deficit = load - direct;
345 const pvCharge = Math.min(surplus, settings.CHARGE,
346 ceiling(100 * (this.capacity - this.soc), this.ceff));
347 this.soc += quotient(pvCharge * this.ceff, 100);
348 surplus -= pvCharge;
349 const isOutage = this.outage !== null && this.outage[0] <= index && index <= this.outage[1];
350 const target = isOutage ? deficit : this.dischargeTarget(deficit, period);
351 const lowerBound = isOutage ? 0 : this.reserve;
352 const loadDrawn = Math.min(ceiling(100 * target, this.deff), settings.DISCHARGE,
353 Math.max(0, this.soc - lowerBound));
354 const loadDischarge = quotient(loadDrawn * this.deff, 100);
355 this.soc -= loadDrawn;
356 deficit -= loadDischarge;
357 let gridLoad, shortfall, pvExport, curtailed, gridCharge, exportDrawn = 0, batteryExport = 0;
358 if (isOutage) {
359 gridLoad = 0;
360 shortfall = deficit;
361 pvExport = 0;
362 curtailed = surplus;
363 gridCharge = 0;
364 } else {
365 gridLoad = deficit;
366 shortfall = 0;
367 pvExport = Math.min(surplus, settings.EXPORT);
368 curtailed = surplus - pvExport;
369 gridCharge = this.gridCharge(period, pvCharge, gridLoad);
370 if (this.arbitrage && period === this.peak && pvCharge === 0) {
371 exportDrawn = Math.min(
372 ceiling(100 * (settings.EXPORT - pvExport), this.deff),
373 settings.DISCHARGE - loadDrawn,
374 Math.max(0, this.soc - this.reserve)
375 );
376 batteryExport = quotient(exportDrawn * this.deff, 100);
377 this.soc -= exportDrawn;
378 }
379 }
380 const loss = pvCharge - quotient(pvCharge * this.ceff, 100) +
381 gridCharge - quotient(gridCharge * this.ceff, 100) +
382 loadDrawn - loadDischarge + exportDrawn - batteryExport;
383 const flows = [direct, pvCharge, pvExport, curtailed, loadDischarge,
384 batteryExport, gridLoad, gridCharge];
385 return new HourRecord(index, period, load, pv, this.soc, flows,
386 loadDrawn + exportDrawn, shortfall, loss, this.prices[period], settings.FEEDIN);
387 }
388
389 run(): SimulationResult {
390 const records = this.data.loads.map((_, index) => this.runHour(index));
391 return new SimulationResult(this.strategy, this.settings, this.prices,
392 this.startSOC, this.reserve, records);
393 }
394}

396class DispatchDesk {
397 data: InputData;
398 settings: Settings;
399 prices: Prices;
400 current: SimulationResult | null;
401 handlers: Record<string, (args: string[]) => string[]>;
402
403 constructor(data: InputData) {
404 this.data = data;
405 this.settings = {...data.initialSettings};
406 this.prices = {...data.initialPrices};
407 this.current = null;
408 this.handlers = {
409 RUN: args => this.run(args), SET: args => this.setValue(args),
410 PRICE: args => this.price(args), RESET: args => this.reset(args),
411 CONFIG: args => this.config(args), NOBATTERY: args => this.noBattery(args),
412 COMPARE: args => this.compare(args), SIZE: args => this.size(args),
413 SWEEP: args => this.sweep(args), OUTAGE: args => this.outage(args),
414 DIFF: args => this.diff(args), HOURLY: args => this.hourly(args),
415 DAILY: args => this.daily(args), BILL: args => this.bill(args),

417 DAYTYPE: args => this.dayType(args), PEAKS: args => this.peaks(args),

425 execute(tokens: string[]): string[] {
426 const [command, ...args] = tokens;
427 if (!Object.hasOwn(this.handlers, command)) return [`ERROR unknown command ${command}`];
428 try {
429 return this.handlers[command](args);
430 } catch (error) {
431 if (!(error instanceof CommandError)) throw error;
432 return [`ERROR ${error.message}`];
433 }
434 }

466 run(args: string[]): string[] {
467 const strategy = oneStrategy(args);
468 const result = this.simulate(strategy);
469 this.current = result;
470 const total = result.total;
471 return [`RUN ${strategyText(strategy)} bill ${money(result.bill)} ` +

474
475 setValue(args: string[]): string[] {
476 requireCount(args, 2);
477 const [item, valueText] = args;
478 if (!Object.hasOwn(SETTING_RANGES, item)) throw new CommandError(`unknown parameter ${item}`);
479 const value = Number(valueText);
480 const [min, max] = SETTING_RANGES[item];
481 if (value < min || value > max) throw new CommandError(`invalid value for ${item}`);
482 const previous = this.settings[item];
483 this.settings[item] = value;
484 return [`SET ${item} ${previous} -> ${value}`];
485 }
486
487 price(args: string[]): string[] {
488 requireCount(args, 2);
489 const [period, valueText] = args;
490 if (!Object.hasOwn(this.prices, period)) throw new CommandError(`unknown period ${period}`);
491 const value = Number(valueText);
492 if (value < 0 || value > 10000) throw new CommandError(`invalid value for ${period}`);
493 const previous = this.prices[period];
494 this.prices[period] = value;
495 const [cheapest, peak] = tariffExtremes(this.prices);
496 return [`PRICE ${period} ${previous} -> ${value} cheapest ${cheapest} peak ${peak}`];
497 }

500 requireCount(args, 0);
501 this.settings = {...this.data.initialSettings};
502 this.prices = {...this.data.initialPrices};

506 config(args: string[]): string[] {
507 requireCount(args, 0);
508 const names = Object.keys(SETTING_RANGES);
509 const settings = this.settings;
510 const battery = names.slice(0, 8).map(item => `${item.toLowerCase()} ${settings[item]}`).join(' ');
511 const grid = names.slice(8, 12).map(item => `${item.toLowerCase()} ${settings[item]}`).join(' ');
512 const tariff = Object.entries(this.prices).map(([item, value]) => `${item} ${value}`).join(' ');
513 const [cheapest, peak] = tariffExtremes(this.prices);
514 return [`BATTERY ${battery}`, `GRID ${grid}`, `SOLAR pvscale ${settings.PVSCALE}`,
515 `TARIFF ${tariff} cheapest ${cheapest} peak ${peak}`];

520 const result = this.baseline(this.settings, this.prices);
521 const total = result.total;
522 return [`NOBATTERY bill ${money(result.bill)} import ${total.imported} ` +

525
526 compare(args: string[]): string[] {
527 const strategies = readStrategies(args);
528 if (!strategies.length) throw new CommandError('wrong number of arguments');
529 const baseline = this.baseline(this.settings, this.prices);
530 const trials = strategies.map(strategy => this.simulate(strategy));
531 trials.sort((a, b) => a.bill - b.bill);
532 const output = [`COMPARE baseline ${money(baseline.bill)}`];
533 trials.forEach((result, index) => {
534 output.push(`${index + 1}. ${strategyText(result.strategy)} bill ${money(result.bill)} ` +
535 `saving ${money(baseline.bill - result.bill)} sufficiency ${result.sufficiency}%`);
536 });
537 return output;
538 }
539
540 size(args: string[]): string[] {
541 requireCount(args, 3);
542 const capacities = checkedRange(Number(args[0]), Number(args[1]), Number(args[2]), SETTING_RANGES.CAPACITY);
543 const baseline = this.baseline(this.settings, this.prices);
544 const output: string[] = [];
545 let best: {capacity: number; cost: number; saving: number} | null = null;
546 for (const capacity of capacities) {
547 const settings: Settings = {...this.settings, CAPACITY: capacity};
548 const result = this.simulate(['SELF', null], settings);
549 const saving = baseline.bill - result.bill;
550 const cost = quotient(capacity * settings.COST, 1000);
551 let payback = 'never';
552 if (saving > 0) {
553 const tenths = quotient(10 * cost, saving);
554 payback = `${quotient(tenths, 10)}.${tenths % 10}`;
555 if (best === null || cost * best.saving < best.cost * saving) {
556 best = {capacity, cost, saving};
557 }
558 }
559 output.push(`SIZE ${capacity} bill ${money(result.bill)} saving ${money(saving)} ` +
560 `cost ${money(cost)} payback ${payback}`);
561 }
562 output.push(`BEST ${best === null ? 'none' : best.capacity}`);
563 return output;
564 }
565
566 sweep(args: string[]): string[] {
567 if (args.length < 5) throw new CommandError('wrong number of arguments');
568 const item = args[0];
569 if (!Object.hasOwn(SETTING_RANGES, item)) throw new CommandError(`unknown parameter ${item}`);
570 const values = checkedRange(Number(args[1]), Number(args[2]), Number(args[3]), SETTING_RANGES[item]);
571 const strategy = oneStrategy(args.slice(4));
572 const output: string[] = [];
573 let best: {value: number; bill: number} | null = null;
574 for (const value of values) {
575 const settings = {...this.settings, [item]: value};
576 const result = this.simulate(strategy, settings);
577 const baseline = this.baseline(settings, this.prices);
578 if (best === null || result.bill < best.bill) best = {value, bill: result.bill};
579 output.push(`SWEEP ${item} ${value} bill ${money(result.bill)} ` +
580 `saving ${money(baseline.bill - result.bill)} sufficiency ${result.sufficiency}%`);
581 }
582 output.push(`BEST ${best!.value}`);
583 return output;
584 }
585
586 outage(args: string[]): string[] {
587 requireCount(args, 3);
588 const day = this.checkedDay(args[0]);
589 const [start, end] = args.slice(1).map(Number);
590 if (start < 0 || end > 23 || start > end) throw new CommandError('invalid hours');
591 const current = this.requireResult();
592 const first = (day - 1) * 24 + start;
593 const last = (day - 1) * 24 + end;
594 const result = this.simulate(current.strategy, current.settings, current.prices, [first, last]);
595 const before = first ? result.hours[first - 1].soc : result.startSOC;
596 const output = [`OUTAGE day ${day} hours ${start}-${end} ` +
597 `strategy ${strategyText(result.strategy)} soc ${before}`,
598 'hour load pv disch soc short curt'];
599 const records = result.hours.slice(first, last + 1);
600 for (const record of records) {
601 output.push(tableRow([record.hour, record.load, record.pv, record.discharge,
602 record.soc, record.shortfall, record.curtailed], [4, 7, 7, 7, 7, 7, 7]));
603 }
604 const total = new Totals(records);
605 const served = total.load ? percentage(total.load - total.shortfall, total.load) : '100.00';
606 output.push(`TOTAL load ${total.load} unserved ${total.shortfall} ` +
607 `served ${served}% bill ${money(result.bill)}`);
608 return output;
609 }

611 diff(args: string[]): string[] {
612 if (args.length < 2) throw new CommandError('wrong number of arguments');
613 const day = this.checkedDay(args[0]);
614 const strategy = oneStrategy(args.slice(1));
615 const current = this.requireResult();
616 const comparison = this.simulate(strategy, current.settings, current.prices);
617 const output = [`DIFF day ${day} ${this.data.dayNames[day - 1]} ` +
618 `${strategyText(current.strategy)} vs ${strategyText(strategy)}`,
619 'hour imp1 imp2 exp1 exp2 soc1 soc2'];
620 const before = current.days[day - 1];
621 const after = comparison.days[day - 1];
622 for (let i = 0; i < before.length; ++i) {
623 const first = before[i], second = after[i];
624 output.push(tableRow([first.hour, first.imported, second.imported, first.exported,
625 second.exported, first.soc, second.soc], [4, 7, 7, 7, 7, 7, 7]));
626 }
627 const cost1 = current.daily[day - 1].net_cost;
628 const cost2 = comparison.daily[day - 1].net_cost;
629 output.push(`TOTAL cost1 ${money(cost1)} cost2 ${money(cost2)} ` +
630 `difference ${money(cost2 - cost1)}`);
631 return output;
632 }

767 cycles(args: string[]): string[] {
768 const result = this.resultWithoutArguments(args);
769 const drawn = result.total.drawn;
770 const capacity = result.settings.CAPACITY;
771 const common = gcd(drawn, capacity);

773 const thousandths = quotient(1000 * drawn, capacity);
774 const decimal = `${quotient(thousandths, 1000)}.${String(thousandths % 1000).padStart(3, '0')}`;
775 return [`CYCLES ${numerator}/${denominator} ${decimal} ` +
776 `drawn ${drawn} losses ${result.total.loss}`];

779 usage(args: string[]): string[] {
780 const result = this.resultWithoutArguments(args);
781 const hours = result.hours;
782 const charging = hours.filter(record => record.charge > 0).length;
783 const discharging = hours.filter(record => record.drawn > 0).length;
784 const idle = hours.filter(record => record.charge === 0 && record.drawn === 0).length;
785 const full = hours.filter(record => record.soc === result.settings.CAPACITY).length;
786 let low = 0, runLength = 0, longest = 0;
787 let first: HourRecord | null = null;
788 for (const record of hours) {
789 if (record.soc <= result.reserve) {
790 ++low;
791 ++runLength;
792 if (runLength > longest) {
793 longest = runLength;
794 first = hours[record.index - runLength + 1];
795 }
796 } else {
797 runLength = 0;
798 }
799 }
800 let line = `USAGE charging ${charging} discharging ${discharging} idle ${idle} ` +
801 `full ${full} low ${low} longest ${longest}`;
802 if (first !== null) line += ` from day ${first.day} hour ${first.hour}`;
803 return [line];
804 }
