71values_range <- function(args, bounds) {
72 values <- as.numeric(args)
73 low <- values[1]
74 high <- values[2]
75 step <- values[3]
76 if (low < bounds[1] || high > bounds[2] || high < low || step < 1 ||

78 seq(low, high, by = step)

81strategies <- function(tokens) {
82 result <- list()
83 index <- 1L
84 while (index <= length(tokens)) {
85 name <- tokens[index]
86 index <- index + 1L
87 parameter <- 0
88 if (name %in% names(strategy_ranges)) {
89 if (index > length(tokens)) fail(paste("missing parameter for", name))
90 parameter <- as.numeric(tokens[index])
91 index <- index + 1L
92 bounds <- strategy_ranges[[name]]
93 if (parameter < bounds[1] || parameter > bounds[2]) {
94 fail(paste("invalid parameter for", name))
95 }
96 } else if (!name %in% c("SELF", "EXPORTPEAK")) {
97 fail(paste("unknown strategy", name))
98 }
99 result[[length(result) + 1L]] <- list(name = name, parameter = parameter)
100 }
101 result
102}

116read_data <- function(input) {
117 tokens <- function() strsplit(readLines(input, n = 1L, warn = FALSE), " ", fixed = TRUE)[[1]]
118 battery <- as.numeric(tokens()[-1])
119 grid <- as.numeric(tokens()[-1])
120 settings <- setNames(c(battery, grid, 100), setting_names)
121 count <- as.numeric(tokens()[2])
122 periods <- character(count)
123 prices <- numeric(count)
124 for (i in seq_len(count)) {
125 fields <- tokens()
126 periods[i] <- fields[1]
127 prices[i] <- as.numeric(fields[2])
128 }
129 divisions <- matrix(0, nrow = 2L, ncol = 24L)
130 for (type in 1:2) {
131 fields <- tokens()
132 for (i in seq.int(2L, length(fields), by = 2L)) {
133 first <- as.numeric(fields[i])
134 last <- if (i + 2L <= length(fields)) as.numeric(fields[i + 2L]) - 1 else 23
135 divisions[type, seq.int(first + 1, last + 1)] <- match(fields[i + 1L], periods)
136 }
137 }
138 count <- as.numeric(tokens()[2])
139 profiles <- list()
140 for (i in seq_len(count)) {
141 fields <- tokens()
142 profiles[[paste(fields[1], fields[2])]] <- as.numeric(fields[-c(1, 2)])
143 }
144 days <- as.numeric(tokens()[2])
145 loads <- generation <- matrix(0, nrow = days, ncol = 24L)
146 for (day in seq_len(days)) {
147 fields <- tokens()
148 loads[day, ] <- (profiles[[paste("LOAD", fields[1])]] * as.numeric(fields[2])) %/% 100
149 generation[day, ] <- (profiles[[paste("PV", fields[3])]] * as.numeric(fields[4])) %/% 100
150 }
151 holidays <- as.numeric(tokens()[-c(1, 2)])
152 count <- as.numeric(tokens()[2])
153 for (i in seq_len(count)) {
154 values <- as.numeric(tokens())
155 loads[values[1], values[2] + 1] <- values[3]
156 generation[values[1], values[2] + 1] <- values[4]
157 }
158 weekdays <- c("MON", "TUE", "WED", "THU", "FRI", "SAT", "SUN")
159 day_numbers <- seq_len(days)
160 weekend <- day_numbers %in% holidays | (day_numbers - 1) %% 7 >= 5
161 day_names <- ifelse(day_numbers %in% holidays, "HOL", weekdays[(day_numbers - 1) %% 7 + 1])
162 period_map <- divisions[ifelse(weekend, 2L, 1L), , drop = FALSE]
163 hours <- cbind(day = rep(day_numbers, each = 24L), hour = rep(0:23, days),
164 period = as.vector(t(period_map)), load = as.vector(t(loads)),
165 pv = as.vector(t(generation)))
166 list(data = list(days = days, periods = periods, hours = hours,
167 weekend = weekend, day_names = day_names),
168 settings = settings, prices = prices)
169}

179simulate <- function(data, strategy, settings, prices, outage = NULL) {
180 capacity <- settings[["CAPACITY"]]
181 ceff <- settings[["CEFF"]]
182 deff <- settings[["DEFF"]]
183 reserve_pct <- settings[["RESERVE"]]
184 if (strategy$name == "BACKUP") reserve_pct <- max(reserve_pct, strategy$parameter)
185 reserve <- (capacity * reserve_pct) %/% 100
186 starting <- soc <- (capacity * settings[["INITIAL"]]) %/% 100
187 low <- which.min(prices)
188 high <- which.max(prices)
189 arbitrage <- strategy$name == "EXPORTPEAK" && settings[["FEEDIN"]] > prices[low]
190 fields <- c("day", "hour", "period", "load", "pv", "charge", "discharge", "drawn",
191 "soc", "import", "export", "curtailed", "shortfall", "loss", flow_names,
192 "cost", "income")
193 records <- matrix(0, nrow = nrow(data$hours), ncol = length(fields),
194 dimnames = list(NULL, fields))
195 for (i in seq_len(nrow(data$hours))) {
196 input <- data$hours[i, ]
197 pv <- (input[["pv"]] * settings[["PVSCALE"]]) %/% 100
198 direct <- min(input[["load"]], pv)
199 surplus <- pv - direct
200 deficit <- input[["load"]] - direct
201 x1 <- min(surplus, settings[["CHARGE"]], ceil_div(100 * (capacity - soc), ceff))
202 stored1 <- (x1 * ceff) %/% 100
203 soc <- soc + stored1
204 surplus <- surplus - x1
205 disconnected <- !is.null(outage) && input[["day"]] == outage[1] &&
206 input[["hour"]] >= outage[2] && input[["hour"]] <= outage[3]
207 target <- deficit
208 if (!disconnected) {
209 if (strategy$name == "TOU") {
210 target <- if (input[["period"]] == high) deficit else 0
211 } else if (strategy$name == "PEAKSHAVE") {
212 target <- max(0, deficit - strategy$parameter)
213 } else if (strategy$name == "EXPORTPEAK" && arbitrage &&
214 input[["period"]] == low && low != high) {
215 target <- 0
216 }
217 }
218 lower <- if (disconnected) 0 else reserve
219 y1 <- min(ceil_div(100 * target, deff), settings[["DISCHARGE"]], max(0, soc - lower))
220 d1 <- (y1 * deff) %/% 100
221 soc <- soc - y1
222 deficit <- deficit - d1
223 x2 <- y2 <- d2 <- stored2 <- 0
224 if (disconnected) {
225 grid <- export_pv <- 0
226 shortfall <- deficit
227 curtailed <- surplus
228 } else {
229 grid <- deficit
230 shortfall <- 0
231 export_pv <- min(surplus, settings[["EXPORT"]])
232 curtailed <- surplus - export_pv
233 headroom <- settings[["CHARGE"]] - x1
234 charge_target <- NULL
235 if (strategy$name == "TOU" && input[["period"]] == low && low != high) {
236 charge_target <- (capacity * strategy$parameter) %/% 100
237 } else if (strategy$name == "PEAKSHAVE" && input[["period"]] == low) {
238 charge_target <- capacity
239 headroom <- min(headroom, max(0, strategy$parameter - grid))
240 } else if (strategy$name == "BACKUP" && input[["period"]] == low) {
241 charge_target <- reserve
242 } else if (strategy$name == "EXPORTPEAK" && arbitrage &&
243 input[["period"]] == low && low != high) {
244 charge_target <- capacity
245 }
246 if (!is.null(charge_target)) {
247 x2 <- min(headroom, ceil_div(100 * max(0, charge_target - soc), ceff))
248 stored2 <- (x2 * ceff) %/% 100
249 soc <- soc + stored2
250 }
251 if (strategy$name == "EXPORTPEAK" && arbitrage && input[["period"]] == high && x1 == 0) {
252 y2 <- min(ceil_div(100 * (settings[["EXPORT"]] - export_pv), deff),
253 settings[["DISCHARGE"]] - y1, max(0, soc - reserve))
254 d2 <- (y2 * deff) %/% 100
255 soc <- soc - y2
256 }
257 }
258 imported <- grid + x2
259 exported <- export_pv + d2
260 loss <- x1 - stored1 + x2 - stored2 + y1 - d1 + y2 - d2
261 flows <- c(direct, x1, export_pv, curtailed, d1, d2, grid, x2)
262 records[i, ] <- c(input[1:3], input[["load"]], pv, x1 + x2, d1 + d2, y1 + y2,
263 soc, imported, exported, curtailed, shortfall, loss, flows,
264 (imported * prices[input[["period"]]]) %/% 100000,
265 (exported * settings[["FEEDIN"]]) %/% 100000)
266 }
267 peak <- max(records[, "import"])
268 demand <- (peak * settings[["DEMAND"]]) %/% 1000
269 list(strategy = strategy, settings = settings, prices = prices, records = records,
270 starting = starting, reserve = reserve, peak = peak, demand = demand,
271 bill = net_cost(records) + settings[["FIXED"]] + demand)
272}

351grouped_report <- function(command, data, result) {
352 emit(command, "strategy", notation(result$strategy))
353 records <- result$records
354 if (command == "BYPERIOD") {
355 for (i in seq_along(data$periods)) {
356 selected <- records[records[, "period"] == i, , drop = FALSE]
357 totals <- colSums(selected)
358 emit(data$periods[i], "hours", nrow(selected), "load", totals[["load"]],
359 "pv", totals[["pv"]], "charge", totals[["charge"]], "discharge", totals[["discharge"]],
360 "import", totals[["import"]], "export", totals[["export"]])
361 }
362 } else {
363 for (weekend in c(FALSE, TRUE)) {
364 days <- sum(data$weekend == weekend)
365 selected <- records[data$weekend[records[, "day"]] == weekend, , drop = FALSE]
366 totals <- colSums(selected)
367 cost <- net_cost(selected)
368 emit(if (weekend) "WEEKEND" else "WEEKDAY", "days", days,
369 "load", totals[["load"]] %/% days, "pv", totals[["pv"]] %/% days,
370 "import", totals[["import"]] %/% days, "export", totals[["export"]] %/% days,
371 "cost", money(sign(cost) * (abs(cost) %/% days)))
372 }
373 }

376usage <- function(result) {
377 records <- result$records
378 low <- records[, "soc"] <= result$reserve
379 runs <- rle(low)
380 lengths <- ifelse(runs$values, runs$lengths, 0L)
381 longest <- max(lengths)
382 text <- c("USAGE charging", sum(records[, "charge"] > 0),
383 "discharging", sum(records[, "drawn"] > 0),
384 "idle", sum(records[, "charge"] == 0 & records[, "drawn"] == 0),
385 "full", sum(records[, "soc"] == result$settings[["CAPACITY"]]),
386 "low", sum(low), "longest", longest)
387 if (longest > 0) {
388 run <- which.max(lengths)
389 start <- sum(runs$lengths[seq_len(run - 1L)]) + 1L
390 text <- c(text, "from day", records[start, "day"], "hour", records[start, "hour"])
391 }
392 emit(text)
393}

395report <- function(command, data, result) {
396 records <- result$records
397 strategy <- notation(result$strategy)
398 if (command == "DAILY") {
399 daily(data, result)
400 } else if (command == "BILL") {
401 billing(data, result)
402 } else if (command == "SAVINGS") {
403 savings(data, result)
404 } else if (command %in% c("BYPERIOD", "DAYTYPE")) {
405 grouped_report(command, data, result)
406 } else if (command == "PROFILE") {
407 emit("PROFILE strategy", strategy, "days", data$days)
408 emit("hour load pv import export soc")
409 fields <- c("load", "pv", "import", "export", "soc")
410 for (hour in 0:23) {
411 selected <- records[records[, "hour"] == hour, fields, drop = FALSE]
412 table_row(c(hour, colSums(selected) %/% data$days), c(4, rep(7, 5)))
413 }
414 } else if (command == "FLOWS") {
415 emit("FLOWS strategy", strategy)
416 for (name in flow_names) emit(name, sum(records[, name]))
417 emit("LOSSES", sum(records[, "loss"]))
418 emit("STORED", records[nrow(records), "soc"] - result$starting)
419 } else if (command == "SELFSUFF") {
420 consumed <- sum(records[, "PV>LOAD"]) + sum(records[, "PV>BATTERY"])
421 emit("SELFSUFF sufficiency", paste0(sufficiency(result), "%"),
422 "consumption", paste0(percentage(consumed, sum(records[, "pv"])), "%"))
423 } else if (command == "CYCLES") {
424 drawn <- sum(records[, "drawn"])
425 capacity <- result$settings[["CAPACITY"]]
426 a <- drawn
427 b <- capacity
428 while (b != 0) {
429 remainder <- a %% b
430 a <- b
431 b <- remainder
432 }
433 fraction <- paste0(text_value(drawn %/% a), "/", text_value(capacity %/% a))
434 emit("CYCLES", fraction, decimal(drawn, capacity, 3),
435 "drawn", drawn, "losses", sum(records[, "loss"]))
436 } else if (command == "USAGE") {
437 usage(result)
438 } else if (command == "HISTOGRAM") {
439 emit("HISTOGRAM strategy", strategy)
440 bands <- pmin(9, (10 * records[, "soc"]) %/% result$settings[["CAPACITY"]])
441 counts <- tabulate(bands + 1L, nbins = 10L)
442 for (i in 0:9) {
443 emit(paste0(sprintf("%3.0f", 10 * i), "%"),
444 bar((20 * counts[i + 1L]) %/% nrow(records)), counts[i + 1L])
445 }
446 }
447}

449outage_report <- function(data, result, day, first, last) {
450 alternate <- simulate(data, result$strategy, result$settings, result$prices, c(day, first, last))
451 start <- (day - 1) * 24 + first + 1
452 before <- if (start == 1) alternate$starting else alternate$records[start - 1, "soc"]
453 emit("OUTAGE day", day, "hours", paste0(text_value(first), "-", text_value(last)),
454 "strategy", notation(result$strategy), "soc", before)
455 emit("hour load pv disch soc short curt")
456 records <- alternate$records[seq.int(start, (day - 1) * 24 + last + 1), , drop = FALSE]
457 fields <- c("hour", "load", "pv", "discharge", "soc", "shortfall", "curtailed")
458 for (i in seq_len(nrow(records))) table_row(records[i, fields], c(4, rep(7, 6)))
459 load <- sum(records[, "load"])
460 shortfall <- sum(records[, "shortfall"])
461 served <- if (load == 0) "100.00" else percentage(load - shortfall, load)
462 emit("TOTAL load", load, "unserved", shortfall, "served", paste0(served, "%"),
463 "bill", money(alternate$bill))
464}

482comparisons <- function(command, args, data, settings, prices) {
483 if (command == "COMPARE") {
484 choices <- strategies(args)
485 if (length(choices) == 0L) fail("wrong number of arguments")
486 base <- baseline(data, settings, prices)$bill
487 results <- lapply(choices, function(strategy) simulate(data, strategy, settings, prices))
488 bills <- vapply(results, function(result) result$bill, numeric(1))
489 ordering <- order(bills, seq_along(results))
490 emit("COMPARE baseline", money(base))
491 for (rank in seq_along(ordering)) {
492 result <- results[[ordering[rank]]]
493 emit(paste0(rank, "."), notation(result$strategy), "bill", money(result$bill),
494 "saving", money(base - result$bill), "sufficiency", paste0(sufficiency(result), "%"))
495 }
496 } else if (command == "SIZE") {
497 arity(args, 3L)
498 values <- values_range(args, c(1, 100000))
499 base <- baseline(data, settings, prices)$bill
500 best <- NULL
501 for (capacity in values) {
502 changed <- settings
503 changed[["CAPACITY"]] <- capacity
504 result <- simulate(data, list(name = "SELF", parameter = 0), changed, prices)
505 saving <- base - result$bill
506 cost <- (capacity * settings[["COST"]]) %/% 1000
507 payback <- if (saving > 0) decimal(cost, saving, 1) else "never"
508 emit("SIZE", capacity, "bill", money(result$bill), "saving", money(saving),
509 "cost", money(cost), "payback", payback)
510 if (saving > 0 && (is.null(best) || cost * best$saving < best$cost * saving)) {
511 best <- list(capacity = capacity, cost = cost, saving = saving)
512 }
513 }
514 emit("BEST", if (is.null(best)) "none" else best$capacity)
515 } else {
516 if (length(args) < 5L) fail("wrong number of arguments")
517 item <- args[1]
518 bounds <- parameter_range(item)
519 values <- values_range(args[2:4], bounds)
520 strategy <- one_strategy(args[5:length(args)])
521 best <- NULL
522 for (value in values) {
523 changed <- settings
524 changed[[item]] <- value
525 result <- simulate(data, strategy, changed, prices)
526 base <- baseline(data, changed, prices)$bill
527 emit("SWEEP", item, value, "bill", money(result$bill), "saving", money(base - result$bill),
528 "sufficiency", paste0(sufficiency(result), "%"))
529 if (is.null(best) || result$bill < best$bill) best <- list(value = value, bill = result$bill)
530 }
531 emit("BEST", best$value)
532 }
533}

536 input <- file("stdin", open = "r")
537 on.exit(close(input))
538 initial <- read_data(input)
539 data <- initial$data
540 settings <- initial$settings
541 prices <- initial$prices
542 result <- NULL
543 repeat {
544 line <- readLines(input, n = 1L, warn = FALSE)
545 if (length(line) == 0L) break
546 fields <- strsplit(line, " ", fixed = TRUE)[[1]]
547 command <- fields[1]
548 args <- fields[-1]
549 if (command == "END") break
550 tryCatch({
551 if (command == "RUN") {
552 strategy <- one_strategy(args)
553 result <- simulate(data, strategy, settings, prices)
554 records <- result$records
555 emit("RUN", notation(strategy), "bill", money(result$bill),
556 "import", sum(records[, "import"]), "export", sum(records[, "export"]),
557 "soc", records[nrow(records), "soc"])
558 } else if (command == "SET") {
559 arity(args, 2L)
560 item <- args[1]
561 bounds <- parameter_range(item)
562 value <- as.numeric(args[2])
563 if (value < bounds[1] || value > bounds[2]) fail(paste("invalid value for", item))
564 previous <- settings[[item]]
565 settings[[item]] <- value
566 emit("SET", item, previous, "->", value)
567 } else if (command == "PRICE") {
568 arity(args, 2L)
569 index <- match(args[1], data$periods)
570 if (is.na(index)) fail(paste("unknown period", args[1]))
571 value <- as.numeric(args[2])
572 if (value < 0 || value > 10000) fail(paste("invalid value for", args[1]))
573 previous <- prices[index]
574 prices[index] <- value
575 emit("PRICE", args[1], previous, "->", value,
576 "cheapest", data$periods[which.min(prices)], "peak", data$periods[which.max(prices)])
577 } else if (command == "RESET") {
578 arity(args, 0L)
579 settings <- initial$settings
580 prices <- initial$prices
581 emit("RESET")
582 } else if (command == "CONFIG") {
583 arity(args, 0L)
584 configure(data, settings, prices)
585 } else if (command == "NOBATTERY") {
586 arity(args, 0L)
587 base <- baseline(data, settings, prices)
588 totals <- colSums(base$records)
589 emit("NOBATTERY bill", money(base$bill), "import", totals[["import"]],
590 "export", totals[["export"]], "curtailed", totals[["curtailed"]], "peak", base$peak)
591 } else if (command %in% c("COMPARE", "SIZE", "SWEEP")) {
592 comparisons(command, args, data, settings, prices)
593 } else if (command == "OUTAGE") {
594 arity(args, 3L)
595 day <- valid_day(args[1], data)
596 hours <- as.numeric(args[2:3])
597 if (hours[1] < 0 || hours[1] > hours[2] || hours[2] > 23) fail("invalid hours")
598 if (is.null(result)) fail("no run")
599 outage_report(data, result, day, hours[1], hours[2])
600 } else if (command == "DIFF") {
601 if (length(args) < 2L) fail("wrong number of arguments")
602 day <- valid_day(args[1], data)
603 strategy <- one_strategy(args[-1])
604 if (is.null(result)) fail("no run")
605 difference(data, result, day, strategy)
606 } else if (command %in% c("HOURLY", "CHART")) {
607 arity(args, 1L)
608 day <- valid_day(args[1], data)
609 if (is.null(result)) fail("no run")
610 if (command == "HOURLY") {
611 hourly(data, result, day)
612 } else {
613 emit("CHART day", day, data$day_names[day], "strategy", notation(result$strategy))
614 capacity <- result$settings[["CAPACITY"]]
615 records <- day_records(result, day)
616 for (i in seq_len(nrow(records))) {
617 soc <- records[i, "soc"]
618 emit(sprintf("%2.0f", records[i, "hour"]), bar((20 * soc) %/% capacity),
619 paste0(percentage(soc, capacity), "%"))
620 }
621 }
622 } else if (command == "PEAKS") {
623 arity(args, 1L)
624 count <- as.numeric(args[1])
625 if (count < 1) fail("invalid count")
626 if (is.null(result)) fail("no run")
627 emit("PEAKS strategy", notation(result$strategy))
628 records <- result$records[result$records[, "import"] > 0, , drop = FALSE]
629 ordering <- order(-records[, "import"], records[, "day"], records[, "hour"])
630 for (rank in seq_len(min(count, length(ordering)))) {
631 record <- records[ordering[rank], ]
632 emit(paste0(rank, "."), "day", record[["day"]], "hour", record[["hour"]],
633 data$periods[record[["period"]]], "import", record[["import"]],
634 "load", record[["load"]], "soc", record[["soc"]])
635 }
636 if (nrow(records) == 0L) emit("none")
637 } else if (command %in% report_names) {
638 arity(args, 0L)
639 if (is.null(result)) fail("no run")
640 report(command, data, result)
641 } else {
642 fail(paste("unknown command", command))
643 }
644 }, command_error = function(error) emit("ERROR", conditionMessage(error)))
645 }
646}
647
648main()
