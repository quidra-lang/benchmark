34func money(_ amount: Int) -> String {

55func gcd(_ a: Int, _ b: Int) -> Int {
56 var x = a, y = b
57 while y != 0 {
58 (x, y) = (y, x % y)
59 }

62func tariffExtremes(_ prices: Prices, _ names: [String]) -> (String, String) {
63 var cheapest = names[0], peak = names[0]
64 for name in names {
65 if at(prices, name) < at(prices, cheapest) {
66 cheapest = name
67 }
68 if at(prices, name) > at(prices, peak) {
69 peak = name
70 }
71 }
72 return (cheapest, peak)
73}
74struct Strategy {
75 let name: String
76 let parameter: Int?
77 var text: String {
78 parameter.map { "\(name) \($0)" } ?? name
79 }
80}
81func parseStrategies(_ tokens: [String]) throws -> [Strategy] {
82 var result: [Strategy] = []
83 var i = 0
84 while i < tokens.count {
85 let name = tokens[i]
86 i += 1
87 if name == "SELF" || name == "EXPORTPEAK" {
88 result.append(Strategy(name: name, parameter: nil))
89 continue
90 }
91 guard let range = strategyRanges[name] else {
92 throw CommandError("unknown strategy \(name)")
93 }
94 guard i < tokens.count else {
95 throw CommandError("missing parameter for \(name)")
96 }
97 let parameter = Int(tokens[i])!
98 i += 1
99 guard range.contains(parameter) else {
100 throw CommandError("invalid parameter for \(name)")
101 }
102 result.append(Strategy(name: name, parameter: parameter))
103 }
104 return result
105}
106func requireCount(_ tokens: [String], _ count: Int) throws {
107 if tokens.count != count {
108 throw CommandError("wrong number of arguments")
109 }
110}
111func oneStrategy(_ tokens: [String]) throws -> Strategy {
112 let strategies = try parseStrategies(tokens)
113 if strategies.count != 1 {
114 throw CommandError("wrong number of arguments")
115 }
116 return strategies[0]
117}
118func checkedRange(_ low: Int, _ high: Int, _ step: Int, _ bounds: ClosedRange<Int>) throws -> [Int] {
119 guard step >= 1, low >= bounds.lowerBound, high <= bounds.upperBound, high >= low,
120 (high - low) / step + 1 <= 20
121 else {
122 throw CommandError("invalid range")
123 }
124 return Array(stride(from: low, through: high, by: step))
125}
126func strings(_ a: [Int]) -> [String] {
127 a.map(String.init)

130struct InputData {
131 let initialSettings: Settings
132 let initialPrices: Prices
133 let periodNames: [String]
134 let dayCount: Int
135 let loads: [Int]
136 let basePV: [Int]
137 let dayNames: [String]
138 let weekendDays: [Bool]
139 let periods: [String]
140 let commands: [String]
141
142 init(_ lines: [String]) {
143 var position = 0
144 func read() -> [String] {
145 let words = lines[position].split(separator: " ").map(String.init)
146 position += 1
147 return words
148 }
149 let battery = read().dropFirst().map { Int($0)! }
150 let grid = read().dropFirst().map { Int($0)! }
151 self.initialSettings = Dictionary(uniqueKeysWithValues: zip(settingNames, battery + grid + [100]))
152 let periodCount = Int(read()[1])!
153 var priceList: [String] = []
154 var initialPrices: Prices = [:]
155 for _ in 0..<periodCount {
156 let pair = read()
157 priceList.append(pair[0])
158 initialPrices[pair[0]] = Int(pair[1])!
159 }
160 self.initialPrices = initialPrices
161 self.periodNames = priceList
162 func periodHours(_ words: [String]) -> [String] {
163 var hours: [String] = []
164 for i in stride(from: 0, to: words.count, by: 2) {
165 let start = Int(words[i])!
166 let end = i + 2 < words.count ? Int(words[i + 2])! : 24
167 hours += Array(repeating: words[i + 1], count: end - start)
168 }
169 return hours
170 }
171 let weekday = periodHours(Array(read().dropFirst()))
172 let weekend = periodHours(Array(read().dropFirst()))
173 let profileCount = Int(read()[1])!
174 var profiles: [String: [Int]] = [:]
175 for _ in 0..<profileCount {
176 let values = read()
177 profiles[values[0] + "/" + values[1]] = values.dropFirst(2).map { Int($0)! }
178 }
179 let days = Int(read()[1])!
180 self.dayCount = days
181 var loads: [Int] = [], basePV: [Int] = []
182 for _ in 0..<days {
183 let row = read()
184 let lpct = Int(row[1])!, ppct = Int(row[3])!
185 loads += profiles["LOAD/" + row[0]]!.map { $0 * lpct / 100 }
186 basePV += profiles["PV/" + row[2]]!.map { $0 * ppct / 100 }
187 }
188 let holidays = Set(read().dropFirst(2).map { Int($0)! })
189 let overrideCount = Int(read()[1])!
190 for _ in 0..<overrideCount {
191 let row = read().map { Int($0)! }
192 let index = (row[0] - 1) * 24 + row[1]
193 loads[index] = row[2]
194 basePV[index] = row[3]
195 }
196 self.loads = loads
197 self.basePV = basePV
198 var dnames: [String] = [], weekends: [Bool] = [], pnames: [String] = []
199 for day in 1...days {
200 let holiday = holidays.contains(day)
201 let weekendDay = holiday || (day - 1) % 7 >= 5
202 dnames.append(holiday ? "HOL" : SwiftDayName(day))
203 weekends.append(weekendDay)
204 pnames += weekendDay ? weekend : weekday
205 }
206 self.dayNames = dnames
207 self.weekendDays = weekends
208 self.periods = pnames
209 self.commands = Array(lines.dropFirst(position))
210 }
211}
212func SwiftDayName(_ day: Int) -> String {
213 dayNames[(day - 1) % 7]
214}

216struct HourRecord {
217 let index: Int
218 let period: String
219 let load: Int
220 let pv: Int
221 let soc: Int
222 let flows: [Int]
223 let drawn: Int
224 let shortfall: Int
225 let loss: Int
226 let importCost: Int
227 let exportIncome: Int
228 var day: Int {
229 index / 24 + 1
230 }
231 var hour: Int {
232 index % 24
233 }
234 var charge: Int {
235 flows[1] + flows[7]
236 }
237 var discharge: Int {
238 flows[4] + flows[5]
239 }
240 var imported: Int {
241 flows[6] + flows[7]
242 }
243 var exported: Int {
244 flows[2] + flows[5]
245 }
246 var curtailed: Int {
247 flows[3]
248 }
249 var netCost: Int {
250 importCost - exportIncome
251 }
252 init(_ index: Int, _ period: String, _ load: Int, _ pv: Int, _ soc: Int,
253 _ flows: [Int], _ drawn: Int, _ shortfall: Int, _ loss: Int,
254 _ price: Int, _ feedin: Int)
255 {
256 self.index = index
257 self.period = period
258 self.load = load
259 self.pv = pv
260 self.soc = soc
261 self.flows = flows
262 self.drawn = drawn
263 self.shortfall = shortfall
264 self.loss = loss
265 self.importCost = (flows[6] + flows[7]) * price / 100000
266 self.exportIncome = (flows[2] + flows[5]) * feedin / 100000
267 }
268}
269struct Totals {
270 let count: Int
271 let load, pv, charge, discharge, imported, exported, curtailed: Int
272 let drawn, loss, shortfall, importCost, exportIncome: Int
273 let flows: [Int]
274 var netCost: Int {
275 importCost - exportIncome
276 }
277 init(_ records: [HourRecord]) {
278 count = records.count
279 load = records.reduce(0) { $0 + $1.load }
280 pv = records.reduce(0) { $0 + $1.pv }
281 charge = records.reduce(0) { $0 + $1.charge }
282 discharge = records.reduce(0) { $0 + $1.discharge }
283 imported = records.reduce(0) { $0 + $1.imported }
284 exported = records.reduce(0) { $0 + $1.exported }
285 curtailed = records.reduce(0) { $0 + $1.curtailed }
286 drawn = records.reduce(0) { $0 + $1.drawn }
287 loss = records.reduce(0) { $0 + $1.loss }
288 shortfall = records.reduce(0) { $0 + $1.shortfall }
289 importCost = records.reduce(0) { $0 + $1.importCost }
290 exportIncome = records.reduce(0) { $0 + $1.exportIncome }
291 flows = (0..<8).map { i in records.reduce(0) { $0 + $1.flows[i] } }
292 }
293}
294struct SimulationResult {
295 let strategy: Strategy
296 let settings: Settings
297 let prices: Prices
298 let startSOC: Int
299 let reserve: Int
300 let hours: [HourRecord]
301 let days: [[HourRecord]]
302 let daily: [Totals]
303 let byPeriod: [String: Totals]
304 let total: Totals
305 let peak: Int
306 let demandCharge: Int
307 let bill: Int
308 let endSOC: Int
309 let sufficiency: String
310 init(_ strategy: Strategy, _ settings: Settings, _ prices: Prices,
311 _ startSOC: Int, _ reserve: Int, _ records: [HourRecord])
312 {
313 self.strategy = strategy
314 self.settings = settings
315 self.prices = prices
316 self.startSOC = startSOC
317 self.reserve = reserve
318 hours = records
319 days = stride(from: 0, to: records.count, by: 24).map { i in Array(records[i..<min(i + 24, records.count)]) }
320 daily = days.map(Totals.init)
321 var byPeriod: [String: Totals] = [:]
322 for name in prices.keys {
323 byPeriod[name] = Totals(records.filter { $0.period == name })
324 }
325 self.byPeriod = byPeriod
326 total = Totals(records)
327 peak = records.map(\.imported).max()!
328 demandCharge = peak * at(settings, "DEMAND") / 1000
329 bill = total.netCost + at(settings, "FIXED") + demandCharge
330 endSOC = records.last!.soc
331 sufficiency = percentage(max(0, total.load - total.imported), total.load)
332 }
333}

335final class Simulator {
336 let data: InputData
337 let strategy: Strategy
338 let settings: Settings
339 let prices: Prices
340 let outage: ClosedRange<Int>?
341 let capacity, ceff, deff, reserve, startSOC: Int
342 let cheapest, peak: String
343 let arbitrage: Bool
344 var soc: Int
345 init(_ data: InputData, _ strategy: Strategy, _ settings: Settings, _ prices: Prices,
346 _ outage: ClosedRange<Int>? = nil)
347 {
348 self.data = data
349 self.strategy = strategy
350 self.settings = settings
351 self.prices = prices
352 self.outage = outage
353 capacity = at(settings, "CAPACITY")
354 ceff = at(settings, "CEFF")
355 deff = at(settings, "DEFF")
356 let reservePct = strategy.name == "BACKUP" ? max(at(settings, "RESERVE"), strategy.parameter!) : at(
357 settings,
358 "RESERVE"
359 )
360 reserve = capacity * reservePct / 100
361 startSOC = capacity * at(settings, "INITIAL") / 100
362 soc = startSOC
363 (cheapest, peak) = tariffExtremes(prices, data.periodNames)
364 arbitrage = strategy.name == "EXPORTPEAK" && at(settings, "FEEDIN") > at(prices, cheapest)
365 }
366 func dischargeTarget(_ deficit: Int, _ period: String) -> Int {
367 if strategy.name == "TOU" {
368 return period == peak ? deficit : 0
369 }
370 if strategy.name == "PEAKSHAVE" {
371 return max(0, deficit - strategy.parameter!)
372 }
373 if arbitrage && period == cheapest && period != peak {
374 return 0
375 }
376 return deficit
377 }
378 func gridCharge(_ period: String, _ pvCharge: Int, _ gridLoad: Int) -> Int {
379 var headroom = at(settings, "CHARGE") - pvCharge
380 var target: Int? = nil
381 if strategy.name == "TOU" && period == cheapest && period != peak {
382 target = capacity * strategy.parameter! / 100
383 } else if strategy.name == "PEAKSHAVE" && period == cheapest {
384 target = capacity
385 headroom = min(headroom, max(0, strategy.parameter! - gridLoad))
386 } else if strategy.name == "BACKUP" && period == cheapest {
387 target = reserve
388 } else if arbitrage && period == cheapest && period != peak {
389 target = capacity
390 }
391 guard let target = target else {
392 return 0
393 }
394 let amount = min(headroom, ceilDiv(100 * max(0, target - soc), ceff))
395 soc += amount * ceff / 100
396 return amount
397 }
398 func runHour(_ index: Int) -> HourRecord {
399 let load = data.loads[index]
400 let pv = data.basePV[index] * at(settings, "PVSCALE") / 100
401 let period = data.periods[index]
402 let direct = min(load, pv)
403 var surplus = pv - direct, deficit = load - direct
404 let pvCharge = min(surplus, at(settings, "CHARGE"), ceilDiv(100 * (capacity - soc), ceff))
405 soc += pvCharge * ceff / 100
406 surplus -= pvCharge
407 let isOutage = outage?.contains(index) ?? false
408 let target = isOutage ? deficit : dischargeTarget(deficit, period)
409 let lowerBound = isOutage ? 0 : reserve
410 let loadDrawn = min(ceilDiv(100 * target, deff), at(settings, "DISCHARGE"), max(0, soc - lowerBound))
411 let loadDischarge = loadDrawn * deff / 100
412 soc -= loadDrawn
413 deficit -= loadDischarge
414 var gridLoad = 0, shortfall = 0, pvExport = 0, curtailed = 0, gridCharged = 0
415 var exportDrawn = 0, batteryExport = 0
416 if isOutage {
417 shortfall = deficit
418 curtailed = surplus
419 } else {
420 gridLoad = deficit
421 pvExport = min(surplus, at(settings, "EXPORT"))
422 curtailed = surplus - pvExport
423 gridCharged = gridCharge(period, pvCharge, gridLoad)
424 if arbitrage && period == peak && pvCharge == 0 {
425 exportDrawn = min(ceilDiv(100 * (at(settings, "EXPORT") - pvExport), deff),
426 at(settings, "DISCHARGE") - loadDrawn, max(0, soc - reserve))
427 batteryExport = exportDrawn * deff / 100
428 soc -= exportDrawn
429 }
430 }
431 let loss = pvCharge - pvCharge * ceff / 100 + gridCharged - gridCharged * ceff / 100
432 + loadDrawn - loadDischarge + exportDrawn - batteryExport
433 let flows = [direct, pvCharge, pvExport, curtailed, loadDischarge, batteryExport, gridLoad, gridCharged]
434 return HourRecord(index, period, load, pv, soc, flows, loadDrawn + exportDrawn, shortfall, loss,
435 at(prices, period), at(settings, "FEEDIN"))
436 }
437 func run() -> SimulationResult {
438 let records = (0..<data.loads.count).map(runHour)
439 return SimulationResult(strategy, settings, prices, startSOC, reserve, records)
440 }
441}

443final class DispatchDesk {
444 let data: InputData
445 var settings: Settings
446 var prices: Prices
447 var current: SimulationResult? = nil
448 init(_ data: InputData) {
449 self.data = data
450 settings = data.initialSettings
451 prices = data.initialPrices
452 }
453 func execute(_ tokens: [String]) -> [String] {
454 let name = tokens[0], args = Array(tokens.dropFirst())
455 let names: Set<String> = ["RUN", "SET", "PRICE", "RESET", "CONFIG", "NOBATTERY", "COMPARE",
456 "SIZE", "SWEEP", "OUTAGE", "DIFF", "HOURLY", "DAILY", "BILL", "SAVINGS",
457 "BYPERIOD", "DAYTYPE", "PEAKS", "PROFILE", "FLOWS", "SELFSUFF", "CYCLES",
458 "USAGE", "HISTOGRAM", "CHART"]
459 if !names.contains(name) {
460 return ["ERROR unknown command \(name)"]
461 }
462 do { return try command(name, args) }
463 catch let error as CommandError { return ["ERROR \(error.message)"] }
464 catch { fatalError("Unexpected execution failure: \(error)") }
465 }
466 func requireResult() throws -> SimulationResult {
467 guard let result = current else {
468 throw CommandError("no run")
469 }
470 return result
471 }
472 func checkedDay(_ text: String) throws -> Int {
473 let day = Int(text)!
474 if day < 1 || day > data.dayCount {
475 throw CommandError("invalid day")
476 }
477 return day
478 }
479 func resultWithoutArguments(_ args: [String]) throws -> SimulationResult {
480 try requireCount(args, 0)
481 return try requireResult()
482 }
483 func resultForDay(_ args: [String]) throws -> (Int, SimulationResult) {
484 try requireCount(args, 1)
485 let day = try checkedDay(args[0])
486 return (day, try requireResult())
487 }
488 func simulate(_ strategy: Strategy, _ settings: Settings? = nil,
489 _ prices: Prices? = nil, _ outage: ClosedRange<Int>? = nil) -> SimulationResult
490 {
491 Simulator(data, strategy, settings ?? self.settings, prices ?? self.prices, outage).run()
492 }
493 func baseline(_ settings: Settings, _ prices: Prices) -> SimulationResult {
494 var settingCopy = settings
495 settingCopy["CAPACITY"] = 0
496 return simulate(Strategy(name: "SELF", parameter: nil), settingCopy, prices)
497 }
498 func command(_ name: String, _ args: [String]) throws -> [String] {
499 switch name {
500 case "RUN":
501 let strategy = try oneStrategy(args)
502 let result = simulate(strategy)
503 current = result
504 let total = result.total
505 return [
506 "RUN \(strategy.text) bill \(money(result.bill)) import \(total.imported) export \(total.exported) soc \(result.endSOC)"
507 ]
508 case "SET":
509 try requireCount(args, 2)
510 let item = args[0]
511 guard let range = settingRanges[item] else {
512 throw CommandError("unknown parameter \(item)")
513 }
514 let value = Int(args[1])!
515 guard range.contains(value) else {
516 throw CommandError("invalid value for \(item)")
517 }
518 let old = at(settings, item)
519 settings[item] = value
520 return ["SET \(item) \(old) -> \(value)"]
521 case "PRICE":
522 try requireCount(args, 2)
523 let item = args[0], value = Int(args[1])!
524 guard let old = prices[item] else {
525 throw CommandError("unknown period \(item)")
526 }
527 guard (0...10000).contains(value) else {
528 throw CommandError("invalid value for \(item)")
529 }
530 prices[item] = value
531 let (low, high) = tariffExtremes(prices, data.periodNames)
532 return ["PRICE \(item) \(old) -> \(value) cheapest \(low) peak \(high)"]
533 case "RESET":
534 try requireCount(args, 0)
535 settings = data.initialSettings
536 prices = data.initialPrices
537 return ["RESET"]
538 case "CONFIG":
539 try requireCount(args, 0)
540 let battery = settingNames.prefix(8).map { "\($0.lowercased()) \(at(settings, $0))" }.joined(separator: " ")
541 let grid = settingNames.dropFirst(8).prefix(4).map { "\($0.lowercased()) \(at(settings, $0))" }
542 .joined(separator: " ")
543 let tariff = data.periodNames.map { "\($0) \(at(prices, $0))" }.joined(separator: " ")
544 let (low, high) = tariffExtremes(prices, data.periodNames)
545 return ["BATTERY \(battery)", "GRID \(grid)", "SOLAR pvscale \(at(settings, "PVSCALE"))",
546 "TARIFF \(tariff) cheapest \(low) peak \(high)"]
547 case "NOBATTERY":
548 try requireCount(args, 0)
549 let result = baseline(settings, prices), total = result.total
550 return [
551 "NOBATTERY bill \(money(result.bill)) import \(total.imported) export \(total.exported) curtailed \(total.curtailed) peak \(result.peak)"
552 ]
553 case "COMPARE":
554 let strategies = try parseStrategies(args)
555 if strategies.isEmpty {
556 throw CommandError("wrong number of arguments")
557 }
558 let base = baseline(settings, prices)
559 var trials: [(Int, SimulationResult)] = []
560 for (index, strategy) in strategies.enumerated() {
561 trials.append((index, simulate(strategy)))
562 }
563 trials.sort { lhs, rhs in
564 if lhs.1.bill == rhs.1.bill {
565 return lhs.0 < rhs.0
566 }
567 return lhs.1.bill < rhs.1.bill
568 }
569 var output = ["COMPARE baseline \(money(base.bill))"]
570 for (index, trial) in trials.enumerated() {
571 let result = trial.1
572 output
573 .append(
574 "\(index + 1). \(result.strategy.text) bill \(money(result.bill)) saving \(money(base.bill - result.bill)) sufficiency \(result.sufficiency)%"
575 )
576 }
577 return output
578 case "SIZE":
579 try requireCount(args, 3)
580 let a = args.map { Int($0)! }
581 let capacities = try checkedRange(a[0], a[1], a[2], settingRanges["CAPACITY"]!)
582 let base = baseline(settings, prices)
583 var output: [String] = []
584 var best: (capacity: Int, cost: Int, saving: Int)? = nil
585 for capacity in capacities {
586 var s = settings
587 s["CAPACITY"] = capacity
588 let r = simulate(Strategy(name: "SELF", parameter: nil), s)
589 let saving = base.bill - r.bill
590 let cost = capacity * at(s, "COST") / 1000
591 var payback = "never"
592 if saving > 0 {
593 let tenths = 10 * cost / saving
594 payback = "\(tenths / 10).\(tenths % 10)"
595 if best == nil || cost * best!.saving < best!.cost * saving {
596 best = (capacity, cost, saving)
597 }
598 }
599 output
600 .append(
601 "SIZE \(capacity) bill \(money(r.bill)) saving \(money(saving)) cost \(money(cost)) payback \(payback)"
602 )
603 }
604 output.append("BEST \(best.map { String($0.capacity) } ?? "none")")
605 return output
606 case "SWEEP":
607 guard args.count >= 5 else {
608 throw CommandError("wrong number of arguments")
609 }
610 let item = args[0]
611 guard let range = settingRanges[item] else {
612 throw CommandError("unknown parameter \(item)")
613 }
614 let a = args[1...3].map { Int($0)! }
615 let values = try checkedRange(a[0], a[1], a[2], range)
616 let strategy = try oneStrategy(Array(args.dropFirst(4)))
617 var output: [String] = []
618 var best: (value: Int, bill: Int)? = nil
619 for value in values {
620 var s = settings
621 s[item] = value
622 let result = simulate(strategy, s)
623 let base = baseline(s, prices)
624 if best == nil || result.bill < best!.bill {
625 best = (value, result.bill)
626 }
627 output
628 .append(
629 "SWEEP \(item) \(value) bill \(money(result.bill)) saving \(money(base.bill - result.bill)) sufficiency \(result.sufficiency)%"
630 )
631 }
632 output.append("BEST \(best!.value)")
633 return output
634 case "OUTAGE":
635 try requireCount(args, 3)
636 let day = try checkedDay(args[0])
637 let start = Int(args[1])!, end = Int(args[2])!
638 guard start >= 0 && end <= 23 && start <= end else {
639 throw CommandError("invalid hours")
640 }
641 let prior = try requireResult()
642 let first = (day - 1) * 24 + start, last = (day - 1) * 24 + end
643 let result = simulate(prior.strategy, prior.settings, prior.prices, first...last)
644 let soc = first == 0 ? result.startSOC : result.hours[first - 1].soc
645 var output = ["OUTAGE day \(day) hours \(start)-\(end) strategy \(result.strategy.text) soc \(soc)",
646 "hour load pv disch soc short curt"]
647 let records = Array(result.hours[first...last])
648 for r in records {
649 output.append(tableRow(
650 strings([r.hour, r.load, r.pv, r.discharge, r.soc, r.shortfall, r.curtailed]),
651 Array(repeating: 7, count: 7).enumerated().map { $0.offset == 0 ? 4 : $0.element }
652 ))
653 }
654 let t = Totals(records)
655 let served = t.load != 0 ? percentage(t.load - t.shortfall, t.load) : "100.00"
656 output.append("TOTAL load \(t.load) unserved \(t.shortfall) served \(served)% bill \(money(result.bill))")
657 return output
658 case "DIFF":
659 guard args.count >= 2 else {
660 throw CommandError("wrong number of arguments")
661 }
662 let day = try checkedDay(args[0])
663 let strategy = try oneStrategy(Array(args.dropFirst()))
664 let prior = try requireResult()
665 let second = simulate(strategy, prior.settings, prior.prices)
666 var output = ["DIFF day \(day) \(data.dayNames[day - 1]) \(prior.strategy.text) vs \(strategy.text)",
667 "hour imp1 imp2 exp1 exp2 soc1 soc2"]
668 for i in 0..<24 {
669 let a = prior.days[day - 1][i], b = second.days[day - 1][i]
670 output.append(tableRow(
671 strings([a.hour, a.imported, b.imported, a.exported, b.exported, a.soc, b.soc]),
672 [4, 7, 7, 7, 7, 7, 7]
673 ))
674 }
675 let cost1 = prior.daily[day - 1].netCost, cost2 = second.daily[day - 1].netCost
676 output.append("TOTAL cost1 \(money(cost1)) cost2 \(money(cost2)) difference \(money(cost2 - cost1))")
677 return output
678 case "HOURLY":
679 let (day, result) = try resultForDay(args)
680 var output = ["HOURLY day \(day) \(data.dayNames[day - 1]) strategy \(result.strategy.text)",
681 "hour load pv charge disch soc import export curt"]
682 let widths = [4] + Array(repeating: 7, count: 8)
683 for r in result.days[day - 1] {
684 output.append(tableRow(
685 strings([r.hour, r.load, r.pv, r.charge, r.discharge, r.soc, r.imported, r.exported, r.curtailed]),
686 widths
687 ))
688 }
689 let t = result.daily[day - 1]
690 output.append(tableRow(
691 ["sum", String(t.load), String(t.pv), String(t.charge), String(t.discharge), "", String(t.imported),
692 String(t.exported), String(t.curtailed)],
693 widths
694 ))
695 return output
696 case "DAILY":
697 let result = try resultWithoutArguments(args)
698 var output = ["DAILY strategy \(result.strategy.text)",
699 "day name load pv import export curt soc cost"]
700 for (index, t) in result.daily.enumerated() {
701 let values = [String(index + 1), data.dayNames[index], String(t.load), String(t.pv), String(t.imported),
702 String(t.exported), String(t.curtailed), String(result.days[index].last!.soc),
703 money(t.netCost)]
704 output.append(tableRow(values, [3, 5, 8, 8, 8, 8, 8, 8, 10]))
705 }
706 let t = result.total
707 output
708 .append(
709 "TOTAL load \(t.load) pv \(t.pv) import \(t.imported) export \(t.exported) curtailed \(t.curtailed) cost \(money(t.netCost))"
710 )
711 return output
712 default:
713 return try additionalCommand(name, args)
714 }

716 func additionalCommand(_ name: String, _ args: [String]) throws -> [String] {
717 switch name {
718 case "BILL":
719 let result = try resultWithoutArguments(args)
720 var output = ["BILL strategy \(result.strategy.text) days \(data.dayCount)"]
721 for period in data.periodNames {
722 let total = result.byPeriod[period]!
723 output.append("IMPORT \(period) \(total.imported) Wh \(money(total.importCost))")
724 }
725 let total = result.total
726 output += ["EXPORT \(total.exported) Wh \(money(-total.exportIncome))",
727 "FIXED \(money(at(result.settings, "FIXED")))",
728 "DEMAND \(result.peak) Wh \(money(result.demandCharge))",
729 "TOTAL \(money(result.bill))"]
730 return output
731 case "SAVINGS":
732 let result = try resultWithoutArguments(args)
733 let base = baseline(result.settings, result.prices)
734 var entries: [(String, Int, Int)] = []
735 for period in data.periodNames {
736 entries.append((period, base.byPeriod[period]!.importCost, result.byPeriod[period]!.importCost))
737 }
738 entries += [("EXPORT", -base.total.exportIncome, -result.total.exportIncome),
739 ("DEMAND", base.demandCharge, result.demandCharge), ("TOTAL", base.bill, result.bill)]
740 var output = ["SAVINGS strategy \(result.strategy.text)"]
741 for (label, before, after) in entries {
742 output.append("\(label) \(money(before)) -> \(money(after)) saving \(money(before - after))")
743 }
744 return output
745 case "BYPERIOD":
746 let result = try resultWithoutArguments(args)
747 var output = ["BYPERIOD strategy \(result.strategy.text)"]
748 for period in data.periodNames {
749 let t = result.byPeriod[period]!
750 output
751 .append(
752 "\(period) hours \(t.count) load \(t.load) pv \(t.pv) charge \(t.charge) discharge \(t.discharge) import \(t.imported) export \(t.exported)"
753 )
754 }
755 return output
756 case "DAYTYPE":
757 let result = try resultWithoutArguments(args)
758 var output = ["DAYTYPE strategy \(result.strategy.text)"]
759 for (label, weekend) in [("WEEKDAY", false), ("WEEKEND", true)] {
760 let indices = data.weekendDays.indices.filter { data.weekendDays[$0] == weekend }
761 let rows = indices.flatMap { result.days[$0] }
762 let t = Totals(rows), count = indices.count
763 output
764 .append(
765 "\(label) days \(count) load \(t.load / count) pv \(t.pv / count) import \(t.imported / count) export \(t.exported / count) cost \(money(t.netCost / count))"
766 )
767 }
768 return output
769 case "PEAKS":
770 try requireCount(args, 1)
771 let count = Int(args[0])!
772 if count < 1 {
773 throw CommandError("invalid count")
774 }
775 let result = try requireResult()
776 let rows = result.hours.filter { $0.imported > 0 }.sorted {
777 $0.imported == $1.imported ? $0.index < $1.index : $0.imported > $1.imported
778 }
779 var output = ["PEAKS strategy \(result.strategy.text)"]
780 for (i, r) in rows.prefix(count).enumerated() {
781 output
782 .append(
783 "\(i + 1). day \(r.day) hour \(r.hour) \(r.period) import \(r.imported) load \(r.load) soc \(r.soc)"
784 )
785 }
786 if rows.isEmpty {
787 output.append("none")
788 }
789 return output
790 case "PROFILE":
791 let result = try resultWithoutArguments(args)
792 var output = ["PROFILE strategy \(result.strategy.text) days \(data.dayCount)",
793 "hour load pv import export soc"]
794 for hour in 0..<24 {
795 let records = result.hours.filter { $0.hour == hour }
796 let count = data.dayCount
797 let values = [hour, records.reduce(0) { $0 + $1.load } / count,
798 records.reduce(0) { $0 + $1.pv } / count,
799 records.reduce(0) { $0 + $1.imported } / count,
800 records.reduce(0) { $0 + $1.exported } / count,
801 records.reduce(0) { $0 + $1.soc } / count]
802 output.append(tableRow(strings(values), [4, 7, 7, 7, 7, 7]))
803 }
804 return output
805 case "FLOWS":
806 let result = try resultWithoutArguments(args)
807 var output = ["FLOWS strategy \(result.strategy.text)"]
808 for (i, name) in flowNames.enumerated() {
809 output.append("\(name) \(result.total.flows[i])")
810 }
811 output.append("LOSSES \(result.total.loss)")
812 output.append("STORED \(result.endSOC - result.startSOC)")
813 return output
814 case "SELFSUFF":
815 let result = try resultWithoutArguments(args)
816 let flows = result.total.flows
817 return [
818 "SELFSUFF sufficiency \(result.sufficiency)% consumption \(percentage(flows[0] + flows[1], result.total.pv))%"
819 ]
820 case "CYCLES":
821 let result = try resultWithoutArguments(args)
822 let drawn = result.total.drawn, capacity = at(result.settings, "CAPACITY")
823 let common = gcd(drawn, capacity)
824 let thousandths = 1000 * drawn / capacity
825 let decimal = "\(thousandths / 1000).\(String(thousandths % 1000).leftPadded(3, by: "0"))"
826 return [
827 "CYCLES \(drawn / common)/\(capacity / common) \(decimal) drawn \(drawn) losses \(result.total.loss)"
828 ]
829 case "USAGE":
830 let result = try resultWithoutArguments(args), hours = result.hours
831 let charging = hours.filter { $0.charge > 0 }.count
832 let discharging = hours.filter { $0.drawn > 0 }.count
833 let idle = hours.filter { $0.charge == 0 && $0.drawn == 0 }.count
834 let full = hours.filter { $0.soc == at(result.settings, "CAPACITY") }.count
835 var low = 0, runLength = 0, longest = 0, first: HourRecord? = nil
836 for row in hours {
837 if row.soc <= result.reserve {
838 low += 1
839 runLength += 1
840 if runLength > longest {
841 longest = runLength
842 first = hours[row.index - runLength + 1]
843 }
844 } else {
845 runLength = 0
846 }
847 }
848 var line = "USAGE charging \(charging) discharging \(discharging) idle \(idle) full \(full) low \(low) longest \(longest)"
849 if let first = first {
850 line += " from day \(first.day) hour \(first.hour)"
851 }
852 return [line]
853 case "HISTOGRAM":
854 let result = try resultWithoutArguments(args)
855 let capacity = at(result.settings, "CAPACITY")
856 var counts = Array(repeating: 0, count: 10)
857 for r in result.hours {
858 counts[min(9, 10 * r.soc / capacity)] += 1
859 }
860 var output = ["HISTOGRAM strategy \(result.strategy.text)"]
861 for (i, count) in counts.enumerated() {
862 let width = 20 * count / result.hours.count
863 output.append("\(String(10 * i).leftPadded(3))% \(bar(width)) \(count)")
864 }
865 return output
866 case "CHART":
867 let (day, result) = try resultForDay(args)
868 let capacity = at(result.settings, "CAPACITY")
869 var output = ["CHART day \(day) \(data.dayNames[day - 1]) strategy \(result.strategy.text)"]
870 for r in result.days[day - 1] {
871 let width = 20 * r.soc / capacity
872 output.append("\(String(r.hour).leftPadded(2)) \(bar(width)) \(percentage(r.soc, capacity))%")
873 }
874 return output
875 default:
876 fatalError("Unknown command case")
877 }
878 }
