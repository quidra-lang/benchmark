6private val settingNames = listOf(
7 "CAPACITY", "INITIAL", "CHARGE", "DISCHARGE", "CEFF", "DEFF", "RESERVE", "COST",
8 "EXPORT", "FEEDIN", "FIXED", "DEMAND", "PVSCALE"
9)
10private val settingRanges = mapOf(
11 "CAPACITY" to (1L..100000L), "INITIAL" to (0L..100L),
12 "CHARGE" to (1L..20000L), "DISCHARGE" to (1L..20000L),
13 "CEFF" to (1L..100L), "DEFF" to (1L..100L),
14 "RESERVE" to (0L..100L), "COST" to (0L..100000L),
15 "EXPORT" to (0L..100000L), "FEEDIN" to (0L..10000L),
16 "FIXED" to (0L..100000L), "DEMAND" to (0L..100000L),
17 "PVSCALE" to (0L..300L)
18)
19private val strategyRanges = mapOf(
20 "TOU" to (1L..100L), "PEAKSHAVE" to (0L..100000L), "BACKUP" to (0L..100L)
21)
22private val weekdays = listOf("MON", "TUE", "WED", "THU", "FRI", "SAT", "SUN")
23private val flowNames = listOf(
24 "PV>LOAD", "PV>BATTERY", "PV>GRID", "PV>CURTAIL",
25 "BATTERY>LOAD", "BATTERY>GRID", "GRID>LOAD", "GRID>BATTERY"
26)

28private class CommandError(message: String) : Exception(message)
29private fun fail(message: String): Nothing = throw CommandError(message)
30private fun ceil(numerator: Long, denominator: Long): Long =
31 (numerator + denominator - 1) / denominator
32private fun money(cents: Long): String {
33 val absolute = if (cents < 0) -cents else cents
34 return "${if (cents < 0) "-" else ""}${absolute / 100}.${(absolute % 100).toString().padStart(2, '0')}"
35}
36private fun percentage(numerator: Long, denominator: Long): String {
37 val n = if (denominator == 0L) 0 else 10000 * numerator / denominator
38 return "${n / 100}.${(n % 100).toString().padStart(2, '0')}"
39}
40private fun rightRow(values: List<Any>, widths: List<Int>): String =
41 values.zip(widths).joinToString("") { (value, width) -> value.toString().padStart(width) }
42private fun bar(width: Int): String = "|${"#".repeat(width)}${".".repeat(20 - width)}|"
43private fun gcd(left: Long, right: Long): Long {
44 var a = left
45 var b = right
46 while (b != 0L) { val old = b; b = a % b; a = old }
47 return a
48}

50private data class Strategy(val name: String, val parameter: Long? = null) {
51 override fun toString(): String = if (parameter == null) name else "$name $parameter"
52}
53
54private fun readStrategies(words: List<String>): List<Strategy> {
55 val strategies = mutableListOf<Strategy>()
56 var i = 0
57 while (i < words.size) {
58 val name = words[i++]
59 if (name == "SELF" || name == "EXPORTPEAK") {
60 strategies.add(Strategy(name))
61 continue
62 }
63 val valid = strategyRanges[name] ?: fail("unknown strategy $name")
64 if (i >= words.size) fail("missing parameter for $name")
65 val parameter = words[i++].toLong()
66 if (parameter !in valid) fail("invalid parameter for $name")
67 strategies.add(Strategy(name, parameter))
68 }
69 return strategies
70}
71private fun requireCount(args: List<String>, expected: Int) {
72 if (args.size != expected) fail("wrong number of arguments")
73}
74private fun oneStrategy(args: List<String>): Strategy {
75 val strategies = readStrategies(args)
76 if (strategies.size != 1) fail("wrong number of arguments")
77 return strategies.single()
78}
79private fun checkedRange(low: Long, high: Long, step: Long, range: LongRange): List<Long> {
80 if (low < range.first || high > range.last || high < low || step < 1L ||
81 (high - low) / step + 1 > 20) fail("invalid range")
82 return generateSequence(low) { it + step }.takeWhile { it <= high }.toList()
83}

85private class InputData(lines: List<String>) {
86 val initialSettings: Map<String, Long>
87 val initialPrices: Map<String, Long>
88 val priceOrder: List<String>
89 val dayCount: Int
90 val loads: List<Long>
91 val basePV: List<Long>
92 val dayNames: List<String>
93 val weekendDays: List<Boolean>
94 val periods: List<String>
95 val commands: List<String>
96
97 init {
98 var position = 0
99 fun read(): List<String> = lines[position++].split(' ')
100 val settings = (read().drop(1) + read().drop(1) + "100").map(String::toLong)
101 initialSettings = settingNames.zip(settings).toMap()
102 val prices = linkedMapOf<String, Long>()
103 repeat(read()[1].toInt()) {
104 val tokens = read()
105 prices[tokens[0]] = tokens[1].toLong()
106 }
107 initialPrices = prices
108 priceOrder = prices.keys.toList()
109 val weekdayHours = periodHours(read().drop(1))
110 val weekendHours = periodHours(read().drop(1))
111 val profiles = mutableMapOf<Pair<String, String>, List<Long>>()
112 repeat(read()[1].toInt()) {
113 val tokens = read()
114 profiles[tokens[0] to tokens[1]] = tokens.drop(2).map(String::toLong)
115 }
116 dayCount = read()[1].toInt()
117 val hourlyLoads = mutableListOf<Long>()
118 val hourlyPV = mutableListOf<Long>()
119 repeat(dayCount) {
120 val tokens = read()
121 hourlyLoads.addAll(profiles.getValue("LOAD" to tokens[0]).map { it * tokens[1].toLong() / 100 })
122 hourlyPV.addAll(profiles.getValue("PV" to tokens[2]).map { it * tokens[3].toLong() / 100 })
123 }
124 val holidays = read().drop(2).map(String::toInt).toSet()
125 repeat(read()[1].toInt()) {
126 val tokens = read().map(String::toLong)
127 val hour = ((tokens[0] - 1) * 24 + tokens[1]).toInt()
128 hourlyLoads[hour] = tokens[2]
129 hourlyPV[hour] = tokens[3]
130 }
131 loads = hourlyLoads
132 basePV = hourlyPV
133 dayNames = (1..dayCount).map { if (it in holidays) "HOL" else weekdays[(it - 1) % 7] }
134 weekendDays = (1..dayCount).map { it in holidays || (it - 1) % 7 >= 5 }
135 periods = weekendDays.flatMap { if (it) weekendHours else weekdayHours }
136 commands = lines.drop(position)
137 }
138
139 private fun periodHours(tokens: List<String>): List<String> = buildList {
140 for (i in tokens.indices step 2) {
141 val start = tokens[i].toInt()
142 val end = if (i + 2 < tokens.size) tokens[i + 2].toInt() else 24
143 repeat(end - start) { add(tokens[i + 1]) }
144 }
145 }
146}

148private class HourRecord(
149 val index: Int, val period: String, val load: Long, val pv: Long,
150 val soc: Long, val flows: List<Long>, val drawn: Long, val shortfall: Long,
151 val loss: Long, price: Long, feedin: Long
152) {
153 val day = index / 24 + 1
154 val hour = index % 24
155 val charge = flows[1] + flows[7]
156 val discharge = flows[4] + flows[5]
157 val imported = flows[6] + flows[7]
158 val exported = flows[2] + flows[5]
159 val curtailed = flows[3]
160 val importCost = imported * price / 100000
161 val exportIncome = exported * feedin / 100000
162 val netCost = importCost - exportIncome
163}
164
165private class Totals(records: List<HourRecord>) {
166 val count = records.size
167 val load = records.sumOf { it.load }
168 val pv = records.sumOf { it.pv }
169 val charge = records.sumOf { it.charge }
170 val discharge = records.sumOf { it.discharge }
171 val imported = records.sumOf { it.imported }
172 val exported = records.sumOf { it.exported }
173 val curtailed = records.sumOf { it.curtailed }
174 val drawn = records.sumOf { it.drawn }
175 val loss = records.sumOf { it.loss }
176 val shortfall = records.sumOf { it.shortfall }
177 val importCost = records.sumOf { it.importCost }
178 val exportIncome = records.sumOf { it.exportIncome }
179 val netCost = importCost - exportIncome
180 val flows = flowNames.indices.map { i -> records.sumOf { it.flows[i] } }
181}

183private class SimulationResult(
184 val strategy: Strategy, originalSettings: Map<String, Long>, originalPrices: Map<String, Long>,
185 val startSOC: Long, val reserve: Long, val hours: List<HourRecord>, priceOrder: List<String>
186) {
187 val settings = originalSettings.toMap()
188 val prices = originalPrices.toMap()
189 val days = hours.chunked(24)
190 val daily = days.map(::Totals)
191 val byPeriod = priceOrder.associateWith { name -> Totals(hours.filter { it.period == name }) }
192 val total = Totals(hours)
193 val peak = hours.maxOf { it.imported }
194 val demandCharge = peak * settings.getValue("DEMAND") / 1000
195 val bill = total.netCost + settings.getValue("FIXED") + demandCharge
196 val endSOC = hours.last().soc
197 val sufficiency = percentage(max(0L, total.load - total.imported), total.load)
198}

200private class Simulator(
201 private val data: InputData, private val strategy: Strategy,
202 originalSettings: Map<String, Long>, originalPrices: Map<String, Long>,
203 private val outage: IntRange? = null
204) {
205 private val settings = originalSettings.toMap()
206 private val prices = originalPrices.toMap()
207 private val name = strategy.name
208 private val parameter = strategy.parameter ?: 0L
209 private val capacity = settings.getValue("CAPACITY")
210 private val ceff = settings.getValue("CEFF")
211 private val deff = settings.getValue("DEFF")
212 private val reservePercent = if (name == "BACKUP") max(settings.getValue("RESERVE"), parameter)
213 else settings.getValue("RESERVE")
214 private val reserve = capacity * reservePercent / 100
215 private val startSOC = capacity * settings.getValue("INITIAL") / 100
216 private var soc = startSOC
217 private val cheapest = data.priceOrder.minBy { prices.getValue(it) }
218 private val peak = data.priceOrder.maxBy { prices.getValue(it) }
219 private val arbitrage = name == "EXPORTPEAK" && settings.getValue("FEEDIN") > prices.getValue(cheapest)
220
221 private fun dischargeTarget(deficit: Long, period: String): Long = when {
222 name == "TOU" -> if (period == peak) deficit else 0L
223 name == "PEAKSHAVE" -> max(0L, deficit - parameter)
224 arbitrage && period == cheapest && period != peak -> 0L
225 else -> deficit
226 }
227
228 private fun gridCharge(period: String, pvCharge: Long, gridLoad: Long): Long {
229 var headroom = settings.getValue("CHARGE") - pvCharge
230 val target = when {
231 name == "TOU" && period == cheapest && period != peak -> capacity * parameter / 100
232 name == "PEAKSHAVE" && period == cheapest -> {
233 headroom = min(headroom, max(0L, parameter - gridLoad))
234 capacity
235 }
236 name == "BACKUP" && period == cheapest -> reserve
237 arbitrage && period == cheapest && period != peak -> capacity
238 else -> return 0L
239 }
240 val amount = min(headroom, ceil(100 * max(0L, target - soc), ceff))
241 soc += amount * ceff / 100
242 return amount
243 }
244
245 private fun hour(index: Int): HourRecord {
246 val load = data.loads[index]
247 val pv = data.basePV[index] * settings.getValue("PVSCALE") / 100
248 val period = data.periods[index]
249 val direct = min(load, pv)
250 var surplus = pv - direct
251 var deficit = load - direct
252 val pvCharge = minOf(surplus, settings.getValue("CHARGE"), ceil(100 * (capacity - soc), ceff))
253 soc += pvCharge * ceff / 100
254 surplus -= pvCharge
255 val isOutage = outage?.contains(index) == true
256 val target = if (isOutage) deficit else dischargeTarget(deficit, period)
257 val lowerBound = if (isOutage) 0L else reserve
258 val loadDrawn = minOf(ceil(100 * target, deff), settings.getValue("DISCHARGE"),
259 max(0L, soc - lowerBound))
260 val loadDischarge = loadDrawn * deff / 100
261 soc -= loadDrawn
262 deficit -= loadDischarge
263 var gridLoad = 0L
264 var shortfall = 0L
265 var pvExport = 0L
266 var curtailed = 0L
267 var gridCharge = 0L
268 var exportDrawn = 0L
269 var batteryExport = 0L
270 if (isOutage) {
271 shortfall = deficit
272 curtailed = surplus
273 } else {
274 gridLoad = deficit
275 pvExport = min(surplus, settings.getValue("EXPORT"))
276 curtailed = surplus - pvExport
277 gridCharge = gridCharge(period, pvCharge, gridLoad)
278 if (arbitrage && period == peak && pvCharge == 0L) {
279 exportDrawn = minOf(ceil(100 * (settings.getValue("EXPORT") - pvExport), deff),
280 settings.getValue("DISCHARGE") - loadDrawn, max(0L, soc - reserve))
281 batteryExport = exportDrawn * deff / 100
282 soc -= exportDrawn
283 }
284 }
285 val loss = pvCharge - pvCharge * ceff / 100 + gridCharge - gridCharge * ceff / 100 +
286 loadDrawn - loadDischarge + exportDrawn - batteryExport
287 val flows = listOf(direct, pvCharge, pvExport, curtailed, loadDischarge,
288 batteryExport, gridLoad, gridCharge)
289 return HourRecord(index, period, load, pv, soc, flows, loadDrawn + exportDrawn,
290 shortfall, loss, prices.getValue(period), settings.getValue("FEEDIN"))
291 }
292
293 fun run(): SimulationResult = SimulationResult(strategy, settings, prices, startSOC, reserve,
294 data.loads.indices.map(::hour), data.priceOrder)
295}

297private class DispatchDesk(private val data: InputData) {
298 private var settings = data.initialSettings.toMutableMap()
299 private var prices = data.initialPrices.toMutableMap()
300 private var current: SimulationResult? = null
301
302 fun execute(tokens: List<String>): List<String> = try {
303 handle(tokens[0], tokens.drop(1))
304 } catch (error: CommandError) {
305 listOf("ERROR ${error.message}")
306 }
307
308 private fun simulate(
309 strategy: Strategy, settings: Map<String, Long> = this.settings,
310 prices: Map<String, Long> = this.prices, outage: IntRange? = null
311 ) = Simulator(data, strategy, settings, prices, outage).run()
312
313 private fun baseline(settings: Map<String, Long>, prices: Map<String, Long>) =
314 simulate(Strategy("SELF"), settings + ("CAPACITY" to 0L), prices)
315
316 private fun needResult() = current ?: fail("no run")
317 private fun checkedDay(token: String): Int {
318 val day = token.toInt()
319 if (day !in 1..data.dayCount) fail("invalid day")
320 return day
321 }
322 private fun resultNoArgs(args: List<String>): SimulationResult {
323 requireCount(args, 0)
324 return needResult()
325 }
326 private fun resultForDay(args: List<String>): Pair<Int, SimulationResult> {
327 requireCount(args, 1)
328 val day = checkedDay(args.single())
329 return day to needResult()
330 }
331
332 private fun handle(command: String, args: List<String>): List<String> = when (command) {
333 "RUN" -> {
334 val strategy = oneStrategy(args)
335 val result = simulate(strategy)
336 current = result
337 listOf("RUN $strategy bill ${money(result.bill)} import ${result.total.imported} " +
338 "export ${result.total.exported} soc ${result.endSOC}")
339 }
340 "SET" -> {
341 requireCount(args, 2)
342 val name = args[0]
343 val allowed = settingRanges[name] ?: fail("unknown parameter $name")
344 val number = args[1].toLong()
345 if (number !in allowed) fail("invalid value for $name")
346 val old = settings.getValue(name)
347 settings[name] = number
348 listOf("SET $name $old -> $number")
349 }
350 "PRICE" -> {
351 requireCount(args, 2)
352 val name = args[0]
353 val old = prices[name] ?: fail("unknown period $name")
354 val number = args[1].toLong()
355 if (number !in 0L..10000L) fail("invalid value for $name")
356 prices[name] = number
357 val cheap = data.priceOrder.minBy { prices.getValue(it) }
358 val expensive = data.priceOrder.maxBy { prices.getValue(it) }
359 listOf("PRICE $name $old -> $number cheapest $cheap peak $expensive")
360 }
361 "RESET" -> {
362 requireCount(args, 0)
363 settings = data.initialSettings.toMutableMap()
364 prices = data.initialPrices.toMutableMap()
365 listOf("RESET")
366 }
367 "CONFIG" -> {
368 requireCount(args, 0)
369 val battery = settingNames.take(8).joinToString(" ") { "${it.lowercase()} ${settings.getValue(it)}" }
370 val grid = settingNames.drop(8).take(4).joinToString(" ") { "${it.lowercase()} ${settings.getValue(it)}" }
371 val tariff = data.priceOrder.joinToString(" ") { "$it ${prices.getValue(it)}" }
372 val cheap = data.priceOrder.minBy { prices.getValue(it) }
373 val expensive = data.priceOrder.maxBy { prices.getValue(it) }
374 listOf("BATTERY $battery", "GRID $grid", "SOLAR pvscale ${settings.getValue("PVSCALE")}",
375 "TARIFF $tariff cheapest $cheap peak $expensive")
376 }
377 "NOBATTERY" -> {
378 requireCount(args, 0)
379 val result = baseline(settings, prices)
380 listOf("NOBATTERY bill ${money(result.bill)} import ${result.total.imported} " +
381 "export ${result.total.exported} curtailed ${result.total.curtailed} peak ${result.peak}")
382 }
383 "COMPARE" -> {
384 val strategies = readStrategies(args)
385 if (strategies.isEmpty()) fail("wrong number of arguments")
386 val reference = baseline(settings, prices)
387 val trials = strategies.map { simulate(it) }.sortedBy { it.bill }
388 buildList {
389 add("COMPARE baseline ${money(reference.bill)}")
390 trials.forEachIndexed { i, result ->
391 add("${i + 1}. ${result.strategy} bill ${money(result.bill)} " +
392 "saving ${money(reference.bill - result.bill)} sufficiency ${result.sufficiency}%")
393 }
394 }
395 }
396 "SIZE" -> {
397 requireCount(args, 3)
398 val limits = settingRanges.getValue("CAPACITY")
399 val capacities = checkedRange(args[0].toLong(), args[1].toLong(), args[2].toLong(), limits)
400 val reference = baseline(settings, prices)
401 var bestCapacity: Long? = null
402 var bestCost = 0L
403 var bestSaving = 0L
404 buildList {
405 for (capacity in capacities) {
406 val trialSettings = settings + ("CAPACITY" to capacity)
407 val result = simulate(Strategy("SELF"), trialSettings)
408 val saving = reference.bill - result.bill
409 val cost = capacity * trialSettings.getValue("COST") / 1000
410 var payback = "never"
411 if (saving > 0) {
412 val tenth = 10 * cost / saving
413 payback = "${tenth / 10}.${tenth % 10}"
414 if (bestCapacity == null || cost * bestSaving < bestCost * saving) {
415 bestCapacity = capacity
416 bestCost = cost
417 bestSaving = saving
418 }
419 }
420 add("SIZE $capacity bill ${money(result.bill)} saving ${money(saving)} " +
421 "cost ${money(cost)} payback $payback")
422 }
423 add("BEST ${bestCapacity ?: "none"}")
424 }
425 }
426 "SWEEP" -> {
427 if (args.size < 5) fail("wrong number of arguments")
428 val name = args[0]
429 val limits = settingRanges[name] ?: fail("unknown parameter $name")
430 val values = checkedRange(args[1].toLong(), args[2].toLong(), args[3].toLong(), limits)
431 val strategy = oneStrategy(args.drop(4))
432 var bestValue = values.first()
433 var bestBill: Long? = null
434 buildList {
435 for (number in values) {
436 val newSettings = settings + (name to number)
437 val result = simulate(strategy, newSettings)
438 val reference = baseline(newSettings, prices)
439 if (bestBill == null || result.bill < bestBill!!) {
440 bestValue = number
441 bestBill = result.bill
442 }
443 add("SWEEP $name $number bill ${money(result.bill)} " +
444 "saving ${money(reference.bill - result.bill)} sufficiency ${result.sufficiency}%")
445 }
446 add("BEST $bestValue")
447 }
448 }
449 "OUTAGE" -> {
450 requireCount(args, 3)
451 val day = checkedDay(args[0])
452 val start = args[1].toInt()
453 val finish = args[2].toInt()
454 if (start < 0 || start > finish || finish > 23) fail("invalid hours")
455 val previous = needResult()
456 val first = (day - 1) * 24 + start
457 val last = (day - 1) * 24 + finish
458 val result = simulate(previous.strategy, previous.settings, previous.prices, first..last)
459 val before = if (first == 0) result.startSOC else result.hours[first - 1].soc
460 val subset = result.hours.subList(first, last + 1)
461 val total = Totals(subset)
462 val served = if (total.load == 0L) "100.00" else percentage(total.load - total.shortfall, total.load)
463 buildList {
464 add("OUTAGE day $day hours $start-$finish strategy ${result.strategy} soc $before")
465 add("hour load pv disch soc short curt")
466 subset.forEach { record -> add(rightRow(listOf(record.hour, record.load, record.pv,
467 record.discharge, record.soc, record.shortfall, record.curtailed),
468 listOf(4, 7, 7, 7, 7, 7, 7))) }
469 add("TOTAL load ${total.load} unserved ${total.shortfall} served $served% bill ${money(result.bill)}")
470 }
471 }
472 "DIFF" -> {
473 if (args.size < 2) fail("wrong number of arguments")
474 val day = checkedDay(args[0])
475 val strategy = oneStrategy(args.drop(1))
476 val previous = needResult()
477 val comparison = simulate(strategy, previous.settings, previous.prices)
478 val firstDay = previous.days[day - 1]
479 val secondDay = comparison.days[day - 1]
480 val cost1 = previous.daily[day - 1].netCost
481 val cost2 = comparison.daily[day - 1].netCost
482 buildList {
483 add("DIFF day $day ${data.dayNames[day - 1]} ${previous.strategy} vs $strategy")
484 add("hour imp1 imp2 exp1 exp2 soc1 soc2")
485 firstDay.zip(secondDay).forEach { (a, b) ->
486 add(rightRow(listOf(a.hour, a.imported, b.imported, a.exported, b.exported,
487 a.soc, b.soc), listOf(4, 7, 7, 7, 7, 7, 7)))
488 }
489 add("TOTAL cost1 ${money(cost1)} cost2 ${money(cost2)} difference ${money(cost2 - cost1)}")
490 }
491 }
492 "HOURLY" -> {
493 val (day, result) = resultForDay(args)
494 val widths = listOf(4, 7, 7, 7, 7, 7, 7, 7, 7)
495 val total = result.daily[day - 1]
496 buildList {
497 add("HOURLY day $day ${data.dayNames[day - 1]} strategy ${result.strategy}")
498 add("hour load pv charge disch soc import export curt")
499 result.days[day - 1].forEach { r ->
500 add(rightRow(listOf(r.hour, r.load, r.pv, r.charge, r.discharge,
501 r.soc, r.imported, r.exported, r.curtailed), widths))
502 }
503 add(rightRow(listOf("sum", total.load, total.pv, total.charge,
504 total.discharge, "", total.imported, total.exported, total.curtailed), widths))
505 }
506 }
507 "DAILY" -> {
508 val result = resultNoArgs(args)
509 val widths = listOf(3, 5, 8, 8, 8, 8, 8, 8, 10)
510 val total = result.total
511 buildList {
512 add("DAILY strategy ${result.strategy}")
513 add("day name load pv import export curt soc cost")
514 result.daily.forEachIndexed { i, item ->
515 add(rightRow(listOf(i + 1, data.dayNames[i], item.load, item.pv, item.imported,
516 item.exported, item.curtailed, result.days[i].last().soc,
517 money(item.netCost)), widths))
518 }
519 add("TOTAL load ${total.load} pv ${total.pv} import ${total.imported} " +
520 "export ${total.exported} curtailed ${total.curtailed} cost ${money(total.netCost)}")
521 }
522 }
523 "BILL" -> {
524 val result = resultNoArgs(args)
525 buildList {
526 add("BILL strategy ${result.strategy} days ${data.dayCount}")
527 data.priceOrder.forEach { name ->
528 val total = result.byPeriod.getValue(name)
529 add("IMPORT $name ${total.imported} Wh ${money(total.importCost)}")
530 }
531 add("EXPORT ${result.total.exported} Wh ${money(-result.total.exportIncome)}")
532 add("FIXED ${money(result.settings.getValue("FIXED"))}")
533 add("DEMAND ${result.peak} Wh ${money(result.demandCharge)}")
534 add("TOTAL ${money(result.bill)}")
535 }
536 }
537 "SAVINGS" -> {
538 val result = resultNoArgs(args)
539 val reference = baseline(result.settings, result.prices)
540 buildList {
541 add("SAVINGS strategy ${result.strategy}")
542 data.priceOrder.forEach { name ->
543 val before = reference.byPeriod.getValue(name).importCost
544 val after = result.byPeriod.getValue(name).importCost
545 add("$name ${money(before)} -> ${money(after)} saving ${money(before - after)}")
546 }
547 val pairs = listOf(
548 Triple("EXPORT", -reference.total.exportIncome, -result.total.exportIncome),
549 Triple("DEMAND", reference.demandCharge, result.demandCharge),
550 Triple("TOTAL", reference.bill, result.bill)
551 )
552 pairs.forEach { (name, before, after) ->
553 add("$name ${money(before)} -> ${money(after)} saving ${money(before - after)}")
554 }
555 }
556 }
557 "BYPERIOD" -> {
558 val result = resultNoArgs(args)
559 buildList {
560 add("BYPERIOD strategy ${result.strategy}")
561 data.priceOrder.forEach { name ->
562 val t = result.byPeriod.getValue(name)
563 add("$name hours ${t.count} load ${t.load} pv ${t.pv} charge ${t.charge} " +
564 "discharge ${t.discharge} import ${t.imported} export ${t.exported}")
565 }
566 }
567 }
568 "DAYTYPE" -> {
569 val result = resultNoArgs(args)
570 buildList {
571 add("DAYTYPE strategy ${result.strategy}")
572 listOf("WEEKDAY" to false, "WEEKEND" to true).forEach { (name, isWeekend) ->
573 val days = data.weekendDays.indices.filter { data.weekendDays[it] == isWeekend }
574 val total = Totals(days.flatMap { result.days[it] })
575 val count = days.size.toLong()
576 add("$name days $count load ${total.load / count} pv ${total.pv / count} " +
577 "import ${total.imported / count} export ${total.exported / count} " +
578 "cost ${money(total.netCost / count)}")
579 }
580 }
581 }
582 "PEAKS" -> {
583 requireCount(args, 1)
584 val count = args[0].toInt()
585 if (count < 1) fail("invalid count")
586 val result = needResult()
587 val records = result.hours.filter { it.imported > 0 }
588 .sortedWith(compareByDescending<HourRecord> { it.imported }.thenBy { it.index })
589 buildList {
590 add("PEAKS strategy ${result.strategy}")
591 records.take(count).forEachIndexed { i, r ->
592 add("${i + 1}. day ${r.day} hour ${r.hour} ${r.period} " +
593 "import ${r.imported} load ${r.load} soc ${r.soc}")
594 }
595 if (records.isEmpty()) add("none")
596 }
597 }
598 "PROFILE" -> {
599 val result = resultNoArgs(args)
600 buildList {
601 add("PROFILE strategy ${result.strategy} days ${data.dayCount}")
602 add("hour load pv import export soc")
603 repeat(24) { hour ->
604 val records = result.hours.filter { it.hour == hour }
605 val count = data.dayCount.toLong()
606 add(rightRow(listOf(hour, records.sumOf { it.load } / count,
607 records.sumOf { it.pv } / count, records.sumOf { it.imported } / count,
608 records.sumOf { it.exported } / count, records.sumOf { it.soc } / count),
609 listOf(4, 7, 7, 7, 7, 7)))
610 }
611 }
612 }
613 "FLOWS" -> {
614 val result = resultNoArgs(args)
615 buildList {
616 add("FLOWS strategy ${result.strategy}")
617 flowNames.forEachIndexed { i, name -> add("$name ${result.total.flows[i]}") }
618 add("LOSSES ${result.total.loss}")
619 add("STORED ${result.endSOC - result.startSOC}")
620 }
621 }
622 "SELFSUFF" -> {
623 val result = resultNoArgs(args)
624 val flow = result.total.flows
625 listOf("SELFSUFF sufficiency ${result.sufficiency}% " +
626 "consumption ${percentage(flow[0] + flow[1], result.total.pv)}%")
627 }
628 "CYCLES" -> {
629 val result = resultNoArgs(args)
630 val drawn = result.total.drawn
631 val capacity = result.settings.getValue("CAPACITY")
632 val common = gcd(drawn, capacity)
633 val thousandths = 1000 * drawn / capacity
634 listOf("CYCLES ${drawn / common}/${capacity / common} " +
635 "${thousandths / 1000}.${(thousandths % 1000).toString().padStart(3, '0')} " +
636 "drawn $drawn losses ${result.total.loss}")
637 }
638 "USAGE" -> {
639 val result = resultNoArgs(args)
640 var low = 0
641 var length = 0
642 var longest = 0
643 var first: HourRecord? = null
644 result.hours.forEach { hour ->
645 if (hour.soc <= result.reserve) {
646 low++
647 length++
648 if (length > longest) {
649 longest = length
650 first = result.hours[hour.index - length + 1]
651 }
652 } else length = 0
653 }
654 val charging = result.hours.count { it.charge > 0 }
655 val discharging = result.hours.count { it.drawn > 0 }
656 val idle = result.hours.count { it.charge == 0L && it.drawn == 0L }
657 val full = result.hours.count { it.soc == result.settings.getValue("CAPACITY") }
658 val suffix = if (first == null) "" else " from day ${first!!.day} hour ${first!!.hour}"
659 listOf("USAGE charging $charging discharging $discharging idle $idle " +
660 "full $full low $low longest $longest$suffix")
661 }
662 "HISTOGRAM" -> {
663 val result = resultNoArgs(args)
664 val capacity = result.settings.getValue("CAPACITY")
665 val bands = IntArray(10)
666 result.hours.forEach { bands[min(9L, 10 * it.soc / capacity).toInt()]++ }
667 buildList {
668 add("HISTOGRAM strategy ${result.strategy}")
669 bands.forEachIndexed { index, count ->
670 add("${(10 * index).toString().padStart(3)}% " +
671 "${bar(20 * count / result.hours.size)} $count")
672 }
673 }
674 }
675 "CHART" -> {
676 val (day, result) = resultForDay(args)
677 val capacity = result.settings.getValue("CAPACITY")
678 buildList {
679 add("CHART day $day ${data.dayNames[day - 1]} strategy ${result.strategy}")
680 result.days[day - 1].forEach { r ->
681 add("${r.hour.toString().padStart(2)} ${bar((20 * r.soc / capacity).toInt())} " +
682 "${percentage(r.soc, capacity)}%")
683 }
684 }
685 }
686 else -> fail("unknown command $command")
687 }
688}

690fun main() {
691 val reader = BufferedReader(InputStreamReader(System.`in`))
692 val lines = generateSequence { reader.readLine() }.toList()
693 if (lines.isEmpty()) return
694 val data = InputData(lines)
695 val desk = DispatchDesk(data)
696 val output = mutableListOf<String>()
697 for (line in data.commands) {
698 val tokens = line.split(' ')
699 if (tokens.first() == "END") break
700 output.addAll(desk.execute(tokens))
701 }
702 if (output.isNotEmpty()) println(output.joinToString("\n"))
703}
