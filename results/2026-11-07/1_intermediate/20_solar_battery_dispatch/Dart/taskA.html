59class Strategy {
60 final Kind kind;
61 final int parameter;
62 const Strategy(this.kind, [this.parameter = 0]);
63
64 bool get hasParameter => const {Kind.tou, Kind.peakshave, Kind.backup}.contains(kind);
65 String get text => '${kind.name.toUpperCase()}${hasParameter ? ' $parameter' : ''}';
66}

68List<Strategy> strategies(List<String> words) {
69 final parsed = <Strategy>[];
70 for (var i = 0; i < words.length; i++) {
71 final name = words[i];
72 Kind? kind;
73 for (final candidate in Kind.values) {
74 if (candidate.name.toUpperCase() == name) kind = candidate;
75 }
76 if (kind == null) fail('unknown strategy $name');
77 var strategy = Strategy(kind);
78 if (strategy.hasParameter) {
79 if (++i == words.length) fail('missing parameter for $name');
80 final parameter = int.parse(words[i]);
81 final low = kind == Kind.tou ? 1 : 0;
82 final high = kind == Kind.peakshave ? 100000 : 100;
83 if (parameter < low || parameter > high) fail('invalid parameter for $name');
84 strategy = Strategy(kind, parameter);
85 }
86 parsed.add(strategy);
87 }
88 return parsed;
89}
90
91Strategy oneStrategy(List<String> words) {
92 final parsed = strategies(words);
93 if (parsed.length != 1) fail('wrong number of arguments');
94 return parsed.single;
95}

101Extremes extremes(List<int> prices) {
102 var low = 0;
103 var high = 0;
104 for (var p = 1; p < prices.length; p++) {
105 if (prices[p] < prices[low]) low = p;
106 if (prices[p] > prices[high]) high = p;
107 }
108 return (low: low, high: high);
109}

172 Result(this.settings, List<int> prices, this.strategy, this.records,

174 : prices = List.unmodifiable(prices),

176 periodHours = List.filled(prices.length, 0) {
177 for (final record in records) {
178 total.add(record.values);
179 periods[record.period].add(record.values);
180 periodHours[record.period]++;
181 peak = math.max(peak, record.values.imported);
182 for (var i = 0; i < flows.length; i++) {
183 flows[i] += record.flows[i];
184 }
185 }
186 demandCharge = peak * settings[Setting.demand] ~/ 1000;
187 bill = total.net + settings[Setting.fixed] + demandCharge;
188 }
189
190 int get endSoc => records.last.values.soc;
191 String get sufficiency => percentage(math.max(0, total.load - total.imported), total.load);

193 Metrics sum(Iterable<int> indices) {
194 final sum = Metrics();
195 for (final index in indices) {
196 sum.add(records[index].values);
197 }
198 return sum;
199 }

201 Metrics day(int day) => sum(List.generate(24, (hour) => day * 24 + hour));

204Result simulate(Data data, Settings settings, List<int> prices, Strategy strategy,
205 [Outage? outage]) {
206 final capacity = settings[Setting.capacity];
207 final ceff = settings[Setting.ceff];
208 final deff = settings[Setting.deff];
209 final chargeMax = settings[Setting.charge];
210 final dischargeMax = settings[Setting.discharge];
211 final exportMax = settings[Setting.exportLimit];
212 final reservePercent = strategy.kind == Kind.backup
213 ? math.max(settings[Setting.reserve], strategy.parameter)
214 : settings[Setting.reserve];
215 final reserve = capacity * reservePercent ~/ 100;
216 final starting = capacity * settings[Setting.initial] ~/ 100;
217 final periods = extremes(prices);
218 final arbitrage = strategy.kind == Kind.exportpeak &&
219 settings[Setting.feedin] > prices[periods.low];
220 var soc = starting;
221 final records = <HourRecord>[];
222 for (var i = 0; i < data.hours.length; i++) {
223 final input = data.hours[i];
224 final p = input.period;
225 final pv = input.pv * settings[Setting.pvscale] ~/ 100;
226 final direct = math.min(input.load, pv);
227 var surplus = pv - direct;
228 var deficit = input.load - direct;
229 final x1 = min3(surplus, chargeMax, ceilDiv(100 * (capacity - soc), ceff));
230 final stored1 = x1 * ceff ~/ 100;
231 soc += stored1;
232 surplus -= x1;
233 final off = outage != null && i >= outage.first && i <= outage.last;
234 final target = off ? deficit : switch (strategy.kind) {
235 Kind.self || Kind.backup => deficit,
236 Kind.tou => p == periods.high ? deficit : 0,
237 Kind.peakshave => math.max(0, deficit - strategy.parameter),
238 Kind.exportpeak => arbitrage && p == periods.low && p != periods.high ? 0 : deficit,
239 };
240 final lower = off ? 0 : reserve;
241 final y1 = min3(ceilDiv(100 * target, deff), dischargeMax, math.max(0, soc - lower));
242 final d1 = y1 * deff ~/ 100;
243 soc -= y1;
244 deficit -= d1;
245 final gridLoad = off ? 0 : deficit;
246 final e1 = off ? 0 : math.min(surplus, exportMax);
247 var x2 = 0;
248 var y2 = 0;
249 var d2 = 0;
250 if (!off) {
251 var headroom = chargeMax - x1;
252 int? targetSoc;
253 switch (strategy.kind) {
254 case Kind.tou:
255 if (p == periods.low && p != periods.high) {
256 targetSoc = capacity * strategy.parameter ~/ 100;
257 }
258 break;
259 case Kind.peakshave:
260 if (p == periods.low) {
261 targetSoc = capacity;
262 headroom = math.min(headroom, math.max(0, strategy.parameter - gridLoad));
263 }
264 break;
265 case Kind.backup:
266 if (p == periods.low) targetSoc = reserve;
267 break;
268 case Kind.exportpeak:
269 if (arbitrage && p == periods.low && p != periods.high) targetSoc = capacity;
270 break;
271 case Kind.self: break;
272 }
273 if (targetSoc != null) {
274 x2 = math.min(headroom, ceilDiv(100 * math.max(0, targetSoc - soc), ceff));
275 soc += x2 * ceff ~/ 100;
276 }
277 if (strategy.kind == Kind.exportpeak && arbitrage && p == periods.high && x1 == 0) {
278 y2 = min3(ceilDiv(100 * (exportMax - e1), deff), dischargeMax - y1,
279 math.max(0, soc - reserve));
280 d2 = y2 * deff ~/ 100;
281 soc -= y2;
282 }
283 }
284 final imported = gridLoad + x2;
285 final exported = e1 + d2;
286 final curtailed = surplus - e1;
287 final values = Metrics(load: input.load, pv: pv, charge: x1 + x2,
288 discharge: d1 + d2, drawn: y1 + y2, soc: soc, imported: imported,
289 exported: exported, curtailed: curtailed, shortfall: off ? deficit : 0,
290 loss: x1 - stored1 + x2 - x2 * ceff ~/ 100 + y1 - d1 + y2 - d2,
291 importCost: imported * prices[p] ~/ 100000,
292 exportIncome: exported * settings[Setting.feedin] ~/ 100000);
293 records.add(HourRecord(p, values,
294 [direct, x1, e1, curtailed, d1, d2, gridLoad, x2]));
295 }
296 return Result(settings, prices, strategy, records, starting, reserve);
297}

374 void size(List<String> args) {
375 arity(args, 3);
376 final low = int.parse(args[0]);
377 final high = int.parse(args[1]);
378 final step = int.parse(args[2]);
379 validateRange(low, high, step, Setting.capacity);
380 final base = baseline(data, settings, prices);
381 ({int capacity, int cost, int saving})? best;
382 for (var capacity = low; capacity <= high; capacity += step) {
383 final changed = settings.withValue(Setting.capacity, capacity);
384 final trial = simulate(data, changed, prices, const Strategy(Kind.self));
385 final saving = base.bill - trial.bill;
386 final cost = capacity * settings[Setting.cost] ~/ 1000;
387 var payback = 'never';
388 if (saving > 0) {
389 final value = 10 * cost ~/ saving;
390 payback = '${value ~/ 10}.${value % 10}';
391 if (best == null || cost * best.saving < best.cost * saving) {
392 best = (capacity: capacity, cost: cost, saving: saving);
393 }
394 }
395 emit('SIZE $capacity bill ${money(trial.bill)} saving ${money(saving)} '
396 'cost ${money(cost)} payback $payback');
397 }
398 emit('BEST ${best?.capacity ?? 'none'}');
399 }

403 final item = itemArgument(args[0]);

406 final step = int.parse(args[3]);

413 final base = baseline(data, changed, prices);
414 emit('SWEEP ${item.label} $value bill ${money(trial.bill)} '

416 if (best == null || trial.bill < best.bill) best = (value: value, bill: trial.bill);
417 }
418 emit('BEST ${best!.value}');
419 }

421 void outage(List<String> args) {
422 arity(args, 3);
423 final day = dayArgument(args[0]);
424 final start = int.parse(args[1]);
425 final end = int.parse(args[2]);

433 emit('OUTAGE day ${day + 1} hours $start-$end strategy ${saved.strategy.text} soc $before');

442 emit('TOTAL load ${total.load} unserved ${total.shortfall} served $served% bill ${money(trial.bill)}');
443 }

546 void usage(Result r) {
547 var charging = 0;
548 var discharging = 0;
549 var idle = 0;
550 var full = 0;
551 var low = 0;
552 var streak = 0;
553 var longest = 0;
554 var first = 0;
555 for (var i = 0; i < r.records.length; i++) {
556 final v = r.records[i].values;
557 if (v.charge > 0) charging++;
558 if (v.drawn > 0) discharging++;
559 if (v.charge == 0 && v.drawn == 0) idle++;
560 if (v.soc == r.settings[Setting.capacity]) full++;
561 if (v.soc <= r.reserve) {
562 low++;
563 streak++;
564 if (streak > longest) {
565 longest = streak;
566 first = i - streak + 1;
567 }
568 } else {
569 streak = 0;
570 }
571 }
572 emit('USAGE charging $charging discharging $discharging idle $idle full $full '
573 'low $low longest $longest${longest == 0 ? '' : ' from day ${first ~/ 24 + 1} hour ${first % 24}'}');
574 }

576 void report(String command, Result r, int argument) {
577 switch (command) {
578 case 'HOURLY': hourly(r, argument); break;
579 case 'DAILY': daily(r); break;
580 case 'BILL': bill(r); break;
581 case 'SAVINGS': savings(r); break;
582 case 'DAYTYPE': dayType(r); break;
583 case 'PEAKS': peaks(r, argument); break;
584 case 'USAGE': usage(r); break;
585 case 'BYPERIOD':
586 emit('BYPERIOD strategy ${r.strategy.text}');
587 for (var p = 0; p < data.periods.length; p++) {
588 final v = r.periods[p];
589 emit('${data.periods[p]} hours ${r.periodHours[p]} load ${v.load} pv ${v.pv} '
590 'charge ${v.charge} discharge ${v.discharge} import ${v.imported} export ${v.exported}');
591 }
592 break;
593 case 'PROFILE':
594 emit('PROFILE strategy ${r.strategy.text} days ${data.days}');
595 emit('hour load pv import export soc');
596 for (var hour = 0; hour < 24; hour++) {
597 final v = r.sum(List.generate(data.days, (day) => day * 24 + hour));
598 table([hour, v.load ~/ data.days, v.pv ~/ data.days, v.imported ~/ data.days,
599 v.exported ~/ data.days, v.soc ~/ data.days]);
600 }
601 break;
602 case 'FLOWS':
603 emit('FLOWS strategy ${r.strategy.text}');
604 for (var i = 0; i < flowNames.length; i++) {
605 emit('${flowNames[i]} ${r.flows[i]}');
606 }
607 emit('LOSSES ${r.total.loss}');
608 emit('STORED ${r.endSoc - r.starting}');
609 break;
610 case 'SELFSUFF':
611 emit('SELFSUFF sufficiency ${r.sufficiency}% '
612 'consumption ${percentage(r.flows[0] + r.flows[1], r.total.pv)}%');
613 break;
614 case 'CYCLES': {
615 final drawn = r.total.drawn;
616 final capacity = r.settings[Setting.capacity];
617 final divisor = drawn.gcd(capacity);
618 final value = 1000 * drawn ~/ capacity;
619 emit('CYCLES ${drawn ~/ divisor}/${capacity ~/ divisor} '
620 '${value ~/ 1000}.${(value % 1000).toString().padLeft(3, '0')} '
621 'drawn $drawn losses ${r.total.loss}');
622 break;
623 }
624 case 'HISTOGRAM': {
625 emit('HISTOGRAM strategy ${r.strategy.text}');
626 final counts = List<int>.filled(10, 0);
627 for (final record in r.records) {
628 counts[math.min(9, 10 * record.values.soc ~/ r.settings[Setting.capacity])]++;
629 }
630 for (var i = 0; i < counts.length; i++) {
631 emit('${field(i * 10, 3)}% ${bar(20 * counts[i] ~/ r.records.length)} ${counts[i]}');
632 }
633 break;
634 }
635 case 'CHART':
636 emit('CHART day ${argument + 1} ${data.dayName(argument)} strategy ${r.strategy.text}');
637 for (var hour = 0; hour < 24; hour++) {
638 final soc = r.records[argument * 24 + hour].values.soc;
639 final capacity = r.settings[Setting.capacity];
640 emit('${field(hour, 2)} ${bar(20 * soc ~/ capacity)} ${percentage(soc, capacity)}%');
641 }
642 break;
643 }
644 }
645
646 bool execute(String line) {
647 final words = line.split(' ');
648 final command = words[0];
649 final args = words.sublist(1);
650 try {
651 final count = reportArities[command];
652 if (count != null) {
653 arity(args, count);
654 var argument = 0;
655 if (command == 'HOURLY' || command == 'CHART') argument = dayArgument(args[0]);
656 if (command == 'PEAKS') {
657 argument = int.parse(args[0]);
658 if (argument < 1) fail('invalid count');
659 }
660 report(command, result, argument);
661 return false;
662 }
663 switch (command) {
664 case 'END': arity(args, 0); return true;
665 case 'RUN': {
666 final strategy = oneStrategy(args);
667 final trial = simulate(data, settings, prices, strategy);
668 current = trial;
669 emit('RUN ${strategy.text} bill ${money(trial.bill)} import ${trial.total.imported} '
670 'export ${trial.total.exported} soc ${trial.endSoc}');
671 break;
672 }
673 case 'SET': {
674 arity(args, 2);
675 final item = itemArgument(args[0]);
676 final value = int.parse(args[1]);
677 if (value < item.minimum || value > item.maximum) fail('invalid value for ${item.label}');
678 final previous = settings[item];
679 settings = settings.withValue(item, value);
680 emit('SET ${item.label} $previous -> $value');
681 break;
682 }
683 case 'PRICE': {
684 arity(args, 2);
685 final p = data.periods.indexOf(args[0]);
686 if (p < 0) fail('unknown period ${args[0]}');
687 final value = int.parse(args[1]);
688 if (value < 0 || value > 10000) fail('invalid value for ${args[0]}');
689 final previous = prices[p];
690 prices[p] = value;
691 final bounds = extremes(prices);
692 emit('PRICE ${args[0]} $previous -> $value cheapest ${data.periods[bounds.low]} '
693 'peak ${data.periods[bounds.high]}');
694 break;
695 }
696 case 'RESET':
697 arity(args, 0);
698 settings = initialSettings;
699 prices = initialPrices.toList();
700 emit('RESET');
701 break;
702 case 'CONFIG': arity(args, 0); config(); break;
703 case 'NOBATTERY': {
704 arity(args, 0);
705 final base = baseline(data, settings, prices);
706 emit('NOBATTERY bill ${money(base.bill)} import ${base.total.imported} '
707 'export ${base.total.exported} curtailed ${base.total.curtailed} peak ${base.peak}');
708 break;
709 }
710 case 'COMPARE': compare(args); break;
711 case 'SIZE': size(args); break;
712 case 'SWEEP': sweep(args); break;
713 case 'OUTAGE': outage(args); break;
714 case 'DIFF': difference(args); break;
715 default: fail('unknown command $command');
716 }
717 } on CommandError catch (error) {
718 emit('ERROR ${error.message}');
719 }
720 return false;
721 }
722}

724Dispatch readInput() {
725 List<String> words() => stdin.readLineSync()!.split(' ');
726 final battery = words().skip(1).map(int.parse).toList();
727 final grid = words().skip(1).map(int.parse).toList();
728 final settings = Settings([...battery, ...grid, 100]);
729 final periodCount = int.parse(words()[1]);
730 final names = <String>[];
731 final prices = <int>[];
732 for (var i = 0; i < periodCount; i++) {
733 final line = words();
734 names.add(line[0]);
735 prices.add(int.parse(line[1]));
736 }
737 final divisions = <List<int>>[];
738 for (var type = 0; type < 2; type++) {
739 final line = words();
740 final hours = List<int>.filled(24, 0);
741 for (var i = 1; i < line.length; i += 2) {
742 final start = int.parse(line[i]);
743 final end = i + 2 < line.length ? int.parse(line[i + 2]) : 24;
744 final period = names.indexOf(line[i + 1]);
745 hours.fillRange(start, end, period);
746 }
747 divisions.add(hours);
748 }
749 final profileCount = int.parse(words()[1]);
750 final profiles = <(String, String), List<int>>{};
751 for (var i = 0; i < profileCount; i++) {
752 final line = words();
753 profiles[(line[0], line[1])] = line.skip(2).map(int.parse).toList();
754 }
755 final days = int.parse(words()[1]);
756 final hours = <HourInput>[];
757 for (var day = 0; day < days; day++) {
758 final line = words();
759 final load = profiles[('LOAD', line[0])]!;
760 final pv = profiles[('PV', line[2])]!;
761 final loadPercent = int.parse(line[1]);
762 final pvPercent = int.parse(line[3]);
763 for (var hour = 0; hour < 24; hour++) {
764 hours.add((load: load[hour] * loadPercent ~/ 100,
765 pv: pv[hour] * pvPercent ~/ 100, period: 0));
766 }
767 }
768 final holidays = words().skip(2).map((word) => int.parse(word) - 1).toSet();
769 final data = Data(names, hours, holidays);
770 for (var i = 0; i < hours.length; i++) {
771 final input = hours[i];
772 hours[i] = (load: input.load, pv: input.pv,
773 period: divisions[data.weekend(i ~/ 24) ? 1 : 0][i % 24]);
774 }
775 final overrides = int.parse(words()[1]);
776 for (var i = 0; i < overrides; i++) {
777 final line = words().map(int.parse).toList();
778 final index = (line[0] - 1) * 24 + line[1];
779 hours[index] = (load: line[2], pv: line[3], period: hours[index].period);
780 }
781 return Dispatch(data, settings, prices);
782}

784void main() {
785 final dispatch = readInput();
786 while (true) {
787 final line = stdin.readLineSync();
788 if (line == null || dispatch.execute(line)) break;
789 }
790 stdout.write(dispatch.output);
791}
