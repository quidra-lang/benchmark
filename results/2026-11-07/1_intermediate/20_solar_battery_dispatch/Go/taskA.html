105func tariffExtremes(prices map[string]int, names []string) (string, string) {
106 cheapest, peak := names[0], names[0]
107 for _, name := range names {
108 if prices[name] < prices[cheapest] {
109 cheapest = name
110 }
111 if prices[name] > prices[peak] {
112 peak = name
113 }
114 }
115 return cheapest, peak
116}

118func readStrategies(words []string) ([]Strategy, error) {
119 strategies := make([]Strategy, 0)
120 for i := 0; i < len(words); {
121 name := words[i]
122 i++
123 if name == "SELF" || name == "EXPORTPEAK" {
124 strategies = append(strategies, Strategy{Name: name})
125 continue
126 }
127 limits, ok := strategyRanges[name]
128 if !ok {
129 return nil, fmt.Errorf("unknown strategy %s", name)
130 }
131 if i >= len(words) {
132 return nil, fmt.Errorf("missing parameter for %s", name)
133 }
134 parameter := value(words[i])
135 i++
136 if parameter < limits[0] || parameter > limits[1] {
137 return nil, fmt.Errorf("invalid parameter for %s", name)
138 }
139 strategies = append(strategies, Strategy{Name: name, Parameter: parameter, HasParam: true})
140 }
141 return strategies, nil
142}

144func oneStrategy(words []string) (Strategy, error) {
145 strategies, err := readStrategies(words)
146 if err != nil {
147 return Strategy{}, err
148 }
149 if len(strategies) != 1 {
150 return Strategy{}, fmt.Errorf("wrong number of arguments")
151 }
152 return strategies[0], nil
153}

162func checkedRange(low, high, step int, limits [2]int) ([]int, error) {
163 if low < limits[0] || high > limits[1] || high < low || step < 1 ||
164 (high-low)/step+1 > 20 {
165 return nil, fmt.Errorf("invalid range")
166 }
167 values := make([]int, 0, 20)
168 for number := low; number <= high; number += step {
169 values = append(values, number)
170 }
171 return values, nil
172}

194func readPeriodHours(tokens []string) []string {
195 hours := make([]string, 0, 24)
196 for i := 0; i < len(tokens); i += 2 {
197 end := 24
198 if i+2 < len(tokens) {
199 end = value(tokens[i+2])
200 }
201 for hour := value(tokens[i]); hour < end; hour++ {
202 hours = append(hours, tokens[i+1])
203 }
204 }
205 return hours
206}

208func readInput(lines []string) *InputData {
209 index := 0
210 read := func() []string {
211 fields := strings.Fields(lines[index])
212 index++
213 return fields
214 }
215 battery := read()[1:]
216 grid := read()[1:]
217 fields := append(append(append([]string{}, battery...), grid...), "100")
218 settings := make(map[string]int, len(settingNames))
219 for i, name := range settingNames {
220 settings[name] = value(fields[i])
221 }
222 count := value(read()[1])
223 prices := make(map[string]int, count)
224 names := make([]string, 0, count)
225 for i := 0; i < count; i++ {
226 pair := read()
227 names = append(names, pair[0])
228 prices[pair[0]] = value(pair[1])
229 }
230 weekday, weekend := readPeriodHours(read()[1:]), readPeriodHours(read()[1:])
231 profileCount := value(read()[1])
232 profiles := make(map[string][]int, profileCount)
233 for i := 0; i < profileCount; i++ {
234 spec := read()
235 vals := make([]int, 0, 24)
236 for _, token := range spec[2:] {
237 vals = append(vals, value(token))
238 }
239 profiles[spec[0]+"/"+spec[1]] = vals
240 }
241 days := value(read()[1])
242 result := &InputData{
243 InitialSettings: settings, InitialPrices: prices, PriceOrder: names,
244 DayCount: days, Loads: make([]int, 0, 24*days), BasePV: make([]int, 0, 24*days),
245 }
246 for day := 0; day < days; day++ {
247 args := read()
248 for _, load := range profiles["LOAD/"+args[0]] {
249 result.Loads = append(result.Loads, load*value(args[1])/100)
250 }
251 for _, pv := range profiles["PV/"+args[2]] {
252 result.BasePV = append(result.BasePV, pv*value(args[3])/100)
253 }
254 }
255 holidayArgs := read()
256 holidays := make(map[int]bool)
257 for _, name := range holidayArgs[2:] {
258 holidays[value(name)] = true
259 }
260 overrides := value(read()[1])
261 for i := 0; i < overrides; i++ {
262 args := read()
263 offset := (value(args[0])-1)*24 + value(args[1])
264 result.Loads[offset] = value(args[2])
265 result.BasePV[offset] = value(args[3])
266 }
267 for day := 1; day <= days; day++ {
268 holiday := holidays[day]
269 isWeekend := holiday || (day-1)%7 >= 5
270 name := dayNames[(day-1)%7]
271 if holiday {
272 name = "HOL"
273 }
274 result.DayNames = append(result.DayNames, name)
275 result.WeekendDays = append(result.WeekendDays, isWeekend)
276 if isWeekend {
277 result.Periods = append(result.Periods, weekend...)
278 } else {
279 result.Periods = append(result.Periods, weekday...)
280 }
281 }
282 result.Commands = lines[index:]
283 return result
284}

286type HourRecord struct {
287 Index, Day, Hour int
288 Period string
289 Load, PV, SOC int
290 Flows [8]int
291 Charge, Discharge, Imported, Exported, Curtailed int
292 Drawn, Shortfall, Loss, ImportCost, ExportIncome, NetCost int
293}

295func newHourRecord(index int, period string, load, pv, soc int, flows [8]int,
296 drawn, shortfall, loss, price, feedin int) HourRecord {
297 record := HourRecord{
298 Index: index, Day: index/24 + 1, Hour: index % 24, Period: period,
299 Load: load, PV: pv, SOC: soc, Flows: flows, Drawn: drawn,
300 Shortfall: shortfall, Loss: loss,
301 }
302 record.Charge = flows[1] + flows[7]
303 record.Discharge = flows[4] + flows[5]
304 record.Imported = flows[6] + flows[7]
305 record.Exported = flows[2] + flows[5]
306 record.Curtailed = flows[3]
307 record.ImportCost = record.Imported * price / 100000
308 record.ExportIncome = record.Exported * feedin / 100000
309 record.NetCost = record.ImportCost - record.ExportIncome
310 return record
311}

313type Totals struct {
314 Count int
315 Load, PV, Charge, Discharge, Imported, Exported, Curtailed int
316 Drawn, Loss, Shortfall, ImportCost, ExportIncome, NetCost int
317 Flows [8]int
318}
319
320func totalRecords(records []HourRecord) Totals {
321 total := Totals{Count: len(records)}
322 for _, record := range records {
323 total.Load += record.Load
324 total.PV += record.PV
325 total.Charge += record.Charge
326 total.Discharge += record.Discharge
327 total.Imported += record.Imported
328 total.Exported += record.Exported
329 total.Curtailed += record.Curtailed
330 total.Drawn += record.Drawn
331 total.Loss += record.Loss
332 total.Shortfall += record.Shortfall
333 total.ImportCost += record.ImportCost
334 total.ExportIncome += record.ExportIncome
335 for i, amount := range record.Flows {
336 total.Flows[i] += amount
337 }
338 }
339 total.NetCost = total.ImportCost - total.ExportIncome
340 return total
341}

343type SimulationResult struct {
344 Strategy Strategy
345 Settings, Prices map[string]int
346 StartSOC, Reserve int
347 Hours []HourRecord
348 Days [][]HourRecord
349 Daily []Totals
350 ByPeriod map[string]Totals
351 Total Totals
352 Peak, DemandCharge, Bill, EndSOC int
353 Sufficiency string
354}
355
356func newSimulationResult(strategy Strategy, settings, prices map[string]int,
357 names []string, startSOC, reserve int, records []HourRecord) *SimulationResult {
358 result := &SimulationResult{
359 Strategy: strategy, Settings: clone(settings), Prices: clone(prices),
360 StartSOC: startSOC, Reserve: reserve, Hours: records, ByPeriod: make(map[string]Totals),
361 }
362 for start := 0; start < len(records); start += 24 {
363 day := records[start : start+24]
364 result.Days = append(result.Days, day)
365 result.Daily = append(result.Daily, totalRecords(day))
366 }
367 for _, name := range names {
368 belonging := make([]HourRecord, 0)
369 for _, record := range records {
370 if record.Period == name {
371 belonging = append(belonging, record)
372 }
373 }
374 result.ByPeriod[name] = totalRecords(belonging)
375 }
376 result.Total = totalRecords(records)
377 for _, record := range records {
378 result.Peak = max(result.Peak, record.Imported)
379 }
380 result.DemandCharge = result.Peak * settings["DEMAND"] / 1000
381 result.Bill = result.Total.NetCost + settings["FIXED"] + result.DemandCharge
382 result.EndSOC = records[len(records)-1].SOC
383 result.Sufficiency = percentage(max(0, result.Total.Load-result.Total.Imported), result.Total.Load)
384 return result
385}

387type Simulator struct {
388 Data *InputData
389 Strategy Strategy
390 Settings, Prices map[string]int
391 Name string
392 Parameter, Capacity, CEff, DEff, Reserve, StartSOC, SOC int
393 Cheapest, Peak string
394 Arbitrage bool
395 Outage *[2]int
396}

398func newSimulator(data *InputData, strategy Strategy, settings, prices map[string]int, outage *[2]int) *Simulator {
399 capacity := settings["CAPACITY"]
400 reservePct := settings["RESERVE"]
401 if strategy.Name == "BACKUP" {
402 reservePct = max(reservePct, strategy.Parameter)
403 }
404 cheapest, peak := tariffExtremes(prices, data.PriceOrder)
405 initial := capacity * settings["INITIAL"] / 100
406 return &Simulator{
407 Data: data, Strategy: strategy, Settings: clone(settings), Prices: clone(prices),
408 Name: strategy.Name, Parameter: strategy.Parameter,
409 Capacity: capacity, CEff: settings["CEFF"], DEff: settings["DEFF"],
410 Reserve: capacity * reservePct / 100, StartSOC: initial, SOC: initial,
411 Cheapest: cheapest, Peak: peak,
412 Arbitrage: strategy.Name == "EXPORTPEAK" && settings["FEEDIN"] > prices[cheapest], Outage: outage,
413 }
414}

416func (s *Simulator) dischargeTarget(deficit int, period string) int {
417 switch {
418 case s.Name == "TOU":
419 if period == s.Peak {
420 return deficit
421 }
422 return 0
423 case s.Name == "PEAKSHAVE":
424 return max(0, deficit-s.Parameter)
425 case s.Arbitrage && period == s.Cheapest && period != s.Peak:
426 return 0
427 default:
428 return deficit
429 }
430}
431
432func (s *Simulator) gridCharge(period string, pvCharge, gridLoad int) int {
433 headroom := s.Settings["CHARGE"] - pvCharge
434 target := -1
435 switch {
436 case s.Name == "TOU" && period == s.Cheapest && period != s.Peak:
437 target = s.Capacity * s.Parameter / 100
438 case s.Name == "PEAKSHAVE" && period == s.Cheapest:
439 target = s.Capacity
440 headroom = min(headroom, max(0, s.Parameter-gridLoad))
441 case s.Name == "BACKUP" && period == s.Cheapest:
442 target = s.Reserve
443 case s.Arbitrage && period == s.Cheapest && period != s.Peak:
444 target = s.Capacity
445 }
446 if target < 0 {
447 return 0
448 }
449 amount := min(headroom, ceiling(100*max(0, target-s.SOC), s.CEff))
450 s.SOC += amount * s.CEff / 100
451 return amount
452}

454func (s *Simulator) runHour(index int) HourRecord {
455 settings := s.Settings
456 load := s.Data.Loads[index]
457 pv := s.Data.BasePV[index] * settings["PVSCALE"] / 100
458 period := s.Data.Periods[index]
459 direct := min(load, pv)
460 surplus := pv - direct
461 deficit := load - direct
462 pvCharge := min3(surplus, settings["CHARGE"], ceiling(100*(s.Capacity-s.SOC), s.CEff))
463 s.SOC += pvCharge * s.CEff / 100
464 surplus -= pvCharge
465 isOutage := s.Outage != nil && s.Outage[0] <= index && index <= s.Outage[1]
466 target := s.dischargeTarget(deficit, period)
467 lowerBound := s.Reserve
468 if isOutage {
469 target = deficit
470 lowerBound = 0
471 }
472 loadDrawn := min3(ceiling(100*target, s.DEff), settings["DISCHARGE"], max(0, s.SOC-lowerBound))
473 loadDischarge := loadDrawn * s.DEff / 100
474 s.SOC -= loadDrawn
475 deficit -= loadDischarge
476 gridLoad, shortfall, pvExport, curtailed, gridCharge, exportDrawn, batteryExport := 0, 0, 0, 0, 0, 0, 0
477 if isOutage {
478 shortfall, curtailed = deficit, surplus
479 } else {
480 gridLoad = deficit
481 pvExport = min(surplus, settings["EXPORT"])
482 curtailed = surplus - pvExport
483 gridCharge = s.gridCharge(period, pvCharge, gridLoad)
484 if s.Arbitrage && period == s.Peak && pvCharge == 0 {
485 exportDrawn = min3(ceiling(100*(settings["EXPORT"]-pvExport), s.DEff),
486 settings["DISCHARGE"]-loadDrawn, max(0, s.SOC-s.Reserve))
487 batteryExport = exportDrawn * s.DEff / 100
488 s.SOC -= exportDrawn
489 }
490 }
491 loss := pvCharge - pvCharge*s.CEff/100 + gridCharge - gridCharge*s.CEff/100 +
492 loadDrawn - loadDischarge + exportDrawn - batteryExport
493 flows := [8]int{direct, pvCharge, pvExport, curtailed, loadDischarge, batteryExport, gridLoad, gridCharge}
494 return newHourRecord(index, period, load, pv, s.SOC, flows, loadDrawn+exportDrawn,
495 shortfall, loss, s.Prices[period], settings["FEEDIN"])
496}

498func (s *Simulator) run() *SimulationResult {
499 records := make([]HourRecord, 0, len(s.Data.Loads))
500 for i := range s.Data.Loads {
501 records = append(records, s.runHour(i))
502 }
503 return newSimulationResult(s.Strategy, s.Settings, s.Prices, s.Data.PriceOrder, s.StartSOC, s.Reserve, records)
504}

568func (d *Desk) handle(command string, args []string) ([]string, error) {
569 switch command {
570 case "RUN":
571 strategy, err := oneStrategy(args)
572 if err != nil {
573 return nil, err
574 }
575 result := d.simulate(strategy, d.settings, d.prices, nil)
576 d.current = result
577 return []string{fmt.Sprintf("RUN %s bill %s import %d export %d soc %d",
578 strategy, money(result.Bill), result.Total.Imported, result.Total.Exported, result.EndSOC)}, nil
579
580 case "SET":
581 if err := requireCount(args, 2); err != nil {
582 return nil, err
583 }
584 name := args[0]
585 limits, ok := settingRanges[name]
586 if !ok {
587 return nil, fmt.Errorf("unknown parameter %s", name)
588 }
589 v := value(args[1])
590 if v < limits[0] || v > limits[1] {
591 return nil, fmt.Errorf("invalid value for %s", name)
592 }
593 previous := d.settings[name]
594 d.settings[name] = v
595 return []string{fmt.Sprintf("SET %s %d -> %d", name, previous, v)}, nil
596
597 case "PRICE":
598 if err := requireCount(args, 2); err != nil {
599 return nil, err
600 }
601 name := args[0]
602 previous, ok := d.prices[name]
603 if !ok {
604 return nil, fmt.Errorf("unknown period %s", name)
605 }
606 v := value(args[1])
607 if v < 0 || v > 10000 {
608 return nil, fmt.Errorf("invalid value for %s", name)
609 }
610 d.prices[name] = v
611 cheapest, peak := tariffExtremes(d.prices, d.data.PriceOrder)
612 return []string{fmt.Sprintf("PRICE %s %d -> %d cheapest %s peak %s",
613 name, previous, v, cheapest, peak)}, nil
614
615 case "RESET":
616 if err := requireCount(args, 0); err != nil {
617 return nil, err
618 }
619 d.settings = clone(d.data.InitialSettings)
620 d.prices = clone(d.data.InitialPrices)
621 return []string{"RESET"}, nil
622
623 case "CONFIG":
624 if err := requireCount(args, 0); err != nil {
625 return nil, err
626 }
627 pieces := []string{}
628 for _, name := range settingNames[:8] {
629 pieces = append(pieces, strings.ToLower(name), strconv.Itoa(d.settings[name]))
630 }
631 battery := "BATTERY " + strings.Join(pieces, " ")
632 pieces = nil
633 for _, name := range settingNames[8:12] {
634 pieces = append(pieces, strings.ToLower(name), strconv.Itoa(d.settings[name]))
635 }
636 grid := "GRID " + strings.Join(pieces, " ")
637 pieces = nil
638 for _, name := range d.data.PriceOrder {
639 pieces = append(pieces, name, strconv.Itoa(d.prices[name]))
640 }
641 cheapest, peak := tariffExtremes(d.prices, d.data.PriceOrder)
642 tariff := fmt.Sprintf("TARIFF %s cheapest %s peak %s", strings.Join(pieces, " "), cheapest, peak)
643 return []string{battery, grid, fmt.Sprintf("SOLAR pvscale %d", d.settings["PVSCALE"]), tariff}, nil
644
645 case "NOBATTERY":
646 if err := requireCount(args, 0); err != nil {
647 return nil, err
648 }
649 result := d.baseline(d.settings, d.prices)
650 return []string{fmt.Sprintf("NOBATTERY bill %s import %d export %d curtailed %d peak %d",
651 money(result.Bill), result.Total.Imported, result.Total.Exported, result.Total.Curtailed, result.Peak)}, nil
652
653 case "COMPARE":
654 strategies, err := readStrategies(args)
655 if err != nil {
656 return nil, err
657 }
658 if len(strategies) == 0 {
659 return nil, fmt.Errorf("wrong number of arguments")
660 }
661 reference := d.baseline(d.settings, d.prices)
662 results := make([]*SimulationResult, 0, len(strategies))
663 for _, strategy := range strategies {
664 results = append(results, d.simulate(strategy, d.settings, d.prices, nil))
665 }
666 sort.SliceStable(results, func(i, j int) bool { return results[i].Bill < results[j].Bill })
667 output := []string{fmt.Sprintf("COMPARE baseline %s", money(reference.Bill))}
668 for i, result := range results {
669 output = append(output, fmt.Sprintf("%d. %s bill %s saving %s sufficiency %s%%",
670 i+1, result.Strategy, money(result.Bill), money(reference.Bill-result.Bill), result.Sufficiency))
671 }
672 return output, nil
673
674 case "SIZE":
675 if err := requireCount(args, 3); err != nil {
676 return nil, err
677 }
678 capacities, err := checkedRange(value(args[0]), value(args[1]), value(args[2]), settingRanges["CAPACITY"])
679 if err != nil {
680 return nil, err
681 }
682 reference := d.baseline(d.settings, d.prices)
683 output := []string{}
684 bestCapacity, bestCost, bestSaving := 0, 0, 0
685 for _, capacity := range capacities {
686 settings := clone(d.settings)
687 settings["CAPACITY"] = capacity
688 result := d.simulate(Strategy{Name: "SELF"}, settings, d.prices, nil)
689 saving := reference.Bill - result.Bill
690 cost := capacity * settings["COST"] / 1000
691 payback := "never"
692 if saving > 0 {
693 tenths := 10 * cost / saving
694 payback = fmt.Sprintf("%d.%d", tenths/10, tenths%10)

1107 }
