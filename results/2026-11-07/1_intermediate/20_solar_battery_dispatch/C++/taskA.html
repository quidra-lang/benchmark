51std::vector<std::string> words(const std::string& line) {
52 std::istringstream input(line);
53 std::vector<std::string> result;
54 for (std::string token; input >> token;) result.push_back(token);
55 return result;
56}

58i64 number(const std::string& s) { return std::stoll(s); }
59std::string str(i64 n) { return std::to_string(n); }
60
61i64 value(const Settings& settings, const std::string& key) {
62 return settings.at(key);
63}
64i64 priceOf(const Prices& prices, const std::string& name) {
65 for (const auto& [key, price] : prices) if (key == name) return price;
66 throw std::logic_error("undefined tariff period");
67}

69std::string money(i64 amount) {
70 i64 cents = std::llabs(amount);
71 std::ostringstream out;
72 if (amount < 0) out << '-';
73 out << cents / 100 << '.' << std::setw(2) << std::setfill('0') << cents % 100;
74 return out.str();
75}
76std::string percentage(i64 numerator, i64 denominator) {
77 i64 percent = denominator ? (10000 * numerator / denominator) : 0;
78 std::ostringstream out;
79 out << percent / 100 << '.' << std::setw(2) << std::setfill('0') << percent % 100;
80 return out.str();
81}
82i64 ceiling(i64 numerator, i64 denominator) {
83 return (numerator + denominator - 1) / denominator;
84}
85std::string padded(const std::string& field, int width) {
86 return field.size() < static_cast<std::size_t>(width)
87 ? std::string(width - field.size(), ' ') + field : field;
88}
89std::string tableRow(const std::vector<std::string>& values,
90 const std::vector<int>& widths) {
91 std::string line;
92 for (std::size_t i = 0; i < values.size(); ++i) line += padded(values[i], widths[i]);
93 return line;
94}
95std::string bar(i64 width) {
96 return "|" + std::string(static_cast<std::size_t>(width), '#')
97 + std::string(static_cast<std::size_t>(20 - width), '.') + "|";
98}
99std::pair<std::string, std::string> tariffExtremes(const Prices& prices) {
100 std::string cheapest = prices.front().first;
101 std::string peak = cheapest;
102 for (const auto& [name, cost] : prices) {
103 if (cost < priceOf(prices, cheapest)) cheapest = name;
104 if (cost > priceOf(prices, peak)) peak = name;
105 }
106 return {cheapest, peak};
107}

109struct Strategy {
110 std::string name;
111 std::optional<i64> parameter;
112};
113std::string strategyText(const Strategy& strategy) {
114 return strategy.name + (strategy.parameter ? " " + str(*strategy.parameter) : "");
115}
116void requireCount(const std::vector<std::string>& values, std::size_t expected) {
117 if (values.size() != expected) throw CommandError("wrong number of arguments");
118}
119std::vector<Strategy> readStrategies(const std::vector<std::string>& tokens) {
120 std::vector<Strategy> result;
121 for (std::size_t i = 0; i < tokens.size();) {
122 std::string name = tokens[i++];
123 if (name == "SELF" || name == "EXPORTPEAK") {
124 result.push_back({name, std::nullopt});
125 continue;
126 }
127 auto it = STRATEGY_RANGES.find(name);
128 if (it == STRATEGY_RANGES.end()) throw CommandError("unknown strategy " + name);
129 if (i == tokens.size()) throw CommandError("missing parameter for " + name);
130 i64 parameter = number(tokens[i++]);
131 if (parameter < it->second.first || parameter > it->second.second)
132 throw CommandError("invalid parameter for " + name);
133 result.push_back({name, parameter});
134 }
135 return result;
136}
137Strategy oneStrategy(const std::vector<std::string>& tokens) {
138 auto result = readStrategies(tokens);
139 if (result.size() != 1) throw CommandError("wrong number of arguments");
140 return result.front();
141}
142std::vector<i64> checkedRange(i64 low, i64 high, i64 step,
143 std::pair<i64, i64> limits) {
144 if (low < limits.first || high > limits.second || high < low || step < 1 ||
145 (high - low) / step + 1 > 20) throw CommandError("invalid range");
146 std::vector<i64> values;
147 for (i64 i = low; i <= high; i += step) values.push_back(i);
148 return values;
149}

151struct InputData {
152 Settings initialSettings;
153 Prices initialPrices;
154 int dayCount = 0;
155 std::vector<i64> loads, basePV;
156 std::vector<std::string> dayNames, periods;
157 std::vector<bool> weekendDays;
158 std::vector<std::string> commands;
159
160 static std::vector<std::string> readPeriodHours(const std::vector<std::string>& tokens) {
161 std::vector<std::string> hours;
162 for (std::size_t i = 0; i < tokens.size(); i += 2) {
163 i64 start = number(tokens[i]);
164 i64 end = i + 2 < tokens.size() ? number(tokens[i + 2]) : 24;
165 for (i64 h = start; h < end; ++h) hours.push_back(tokens[i + 1]);
166 }
167 return hours;
168 }
169
170 explicit InputData(std::istream& stream) {
171 auto read = [&]() {
172 std::string line;
173 std::getline(stream, line);
174 return words(line);
175 };
176 auto battery = read();
177 auto grid = read();
178 std::vector<i64> settings;
179 for (std::size_t i = 1; i < battery.size(); ++i) settings.push_back(number(battery[i]));
180 for (std::size_t i = 1; i < grid.size(); ++i) settings.push_back(number(grid[i]));
181 settings.push_back(100);
182 for (std::size_t i = 0; i < SETTING_NAMES.size(); ++i)
183 initialSettings[SETTING_NAMES[i]] = settings[i];
184
185 int periodCount = static_cast<int>(number(read()[1]));
186 for (int i = 0; i < periodCount; ++i) {
187 auto row = read();
188 initialPrices.emplace_back(row[0], number(row[1]));
189 }
190 auto wdTokens = read();
191 auto weTokens = read();
192 auto weekday = readPeriodHours({wdTokens.begin() + 1, wdTokens.end()});
193 auto weekend = readPeriodHours({weTokens.begin() + 1, weTokens.end()});
194
195 int profileCount = static_cast<int>(number(read()[1]));
196 std::unordered_map<std::string, std::vector<i64>> profiles;
197 for (int i = 0; i < profileCount; ++i) {
198 auto row = read();
199 std::vector<i64> numbers;
200 for (std::size_t j = 2; j < row.size(); ++j) numbers.push_back(number(row[j]));
201 profiles.emplace(row[0] + "/" + row[1], std::move(numbers));
202 }
203 dayCount = static_cast<int>(number(read()[1]));
204 for (int day = 0; day < dayCount; ++day) {
205 auto row = read();
206 const auto& load = profiles.at("LOAD/" + row[0]);
207 const auto& pv = profiles.at("PV/" + row[2]);
208 for (i64 item : load) loads.push_back(item * number(row[1]) / 100);
209 for (i64 item : pv) basePV.push_back(item * number(row[3]) / 100);
210 }
211 auto holidayLine = read();
212 std::unordered_set<int> holidays;
213 for (std::size_t i = 2; i < holidayLine.size(); ++i)
214 holidays.insert(static_cast<int>(number(holidayLine[i])));
215 int overrideCount = static_cast<int>(number(read()[1]));
216 for (int i = 0; i < overrideCount; ++i) {
217 auto row = read();
218 auto index = static_cast<std::size_t>((number(row[0]) - 1) * 24 + number(row[1]));
219 loads[index] = number(row[2]);
220 basePV[index] = number(row[3]);
221 }
222 for (int day = 1; day <= dayCount; ++day) {
223 bool holiday = holidays.count(day) != 0;
224 bool weekendDay = holiday || (day - 1) % 7 >= 5;
225 weekendDays.push_back(weekendDay);
226 dayNames.push_back(holiday ? "HOL" : DAY_NAMES[(day - 1) % 7]);
227 const auto& schedule = weekendDay ? weekend : weekday;
228 periods.insert(periods.end(), schedule.begin(), schedule.end());
229 }
230 for (std::string line; std::getline(stream, line);) {
231 if (!line.empty()) commands.push_back(line);
232 }
233 }
234};

236struct HourRecord {
237 i64 index = 0, day = 0, hour = 0;
238 std::string period;
239 i64 load = 0, pv = 0, soc = 0;
240 std::array<i64, 8> flows{};
241 i64 charge = 0, discharge = 0, imported = 0, exported = 0, curtailed = 0;
242 i64 drawn = 0, shortfall = 0, loss = 0;
243 i64 importCost = 0, exportIncome = 0, netCost = 0;
244
245 HourRecord(i64 index_, std::string period_, i64 load_, i64 pv_, i64 soc_,
246 std::array<i64, 8> flows_, i64 drawn_, i64 shortfall_, i64 loss_,
247 i64 price, i64 feedin)
248 : index(index_), day(index_ / 24 + 1), hour(index_ % 24),
249 period(std::move(period_)), load(load_), pv(pv_), soc(soc_),
250 flows(flows_), drawn(drawn_), shortfall(shortfall_), loss(loss_) {
251 charge = flows[1] + flows[7];
252 discharge = flows[4] + flows[5];
253 imported = flows[6] + flows[7];
254 exported = flows[2] + flows[5];
255 curtailed = flows[3];
256 importCost = imported * price / 100000;
257 exportIncome = exported * feedin / 100000;
258 netCost = importCost - exportIncome;
259 }
260};
261
262struct Totals {
263 i64 count = 0, load = 0, pv = 0, charge = 0, discharge = 0;
264 i64 imported = 0, exported = 0, curtailed = 0, drawn = 0;
265 i64 loss = 0, shortfall = 0, importCost = 0, exportIncome = 0, netCost = 0;
266 std::array<i64, 8> flows{};
267
268 void add(const HourRecord& record) {
269 ++count;
270 load += record.load; pv += record.pv;
271 charge += record.charge; discharge += record.discharge;
272 imported += record.imported; exported += record.exported;
273 curtailed += record.curtailed; drawn += record.drawn;
274 loss += record.loss; shortfall += record.shortfall;
275 importCost += record.importCost; exportIncome += record.exportIncome;
276 netCost += record.netCost;
277 for (std::size_t i = 0; i < flows.size(); ++i) flows[i] += record.flows[i];
278 }
279};

281struct SimulationResult {
282 Strategy strategy;
283 Settings settings;
284 Prices prices;
285 i64 startSOC, reserve, endSOC = 0, peak = 0, demandCharge = 0, bill = 0;
286 std::string sufficiency;
287 std::vector<HourRecord> hours;
288 std::vector<Totals> daily, byPeriod;
289 Totals total;
290
291 SimulationResult(Strategy strategy_, Settings settings_, Prices prices_,
292 i64 start, i64 reserve_, std::vector<HourRecord> records)
293 : strategy(std::move(strategy_)), settings(std::move(settings_)),
294 prices(std::move(prices_)), startSOC(start), reserve(reserve_),
295 hours(std::move(records)) {
296 for (std::size_t i = 0; i < hours.size(); ++i) {
297 if (i % 24 == 0) daily.emplace_back();
298 const auto& record = hours[i];
299 daily.back().add(record);
300 total.add(record);
301 peak = std::max(peak, record.imported);
302 }
303 byPeriod.resize(prices.size());
304 for (const auto& record : hours) {
305 for (std::size_t i = 0; i < prices.size(); ++i) {
306 if (prices[i].first == record.period) {
307 byPeriod[i].add(record);
308 break;
309 }
310 }
311 }
312 endSOC = hours.back().soc;
313 demandCharge = peak * value(settings, "DEMAND") / 1000;
314 bill = total.netCost + value(settings, "FIXED") + demandCharge;
315 sufficiency = percentage(std::max<i64>(0, total.load - total.imported), total.load);
316 }
317};

319struct Simulator {
320 const InputData& data;
321 Strategy strategy;
322 Settings settings;
323 Prices prices;
324 std::optional<std::pair<i64, i64>> outage;
325 i64 capacity, ceff, deff, reserve, startSOC, soc;
326 std::string cheapest, peak;
327 bool arbitrage;
328
329 Simulator(const InputData& data_, Strategy strategy_, Settings settings_, Prices prices_,
330 std::optional<std::pair<i64, i64>> outage_ = std::nullopt)
331 : data(data_), strategy(std::move(strategy_)), settings(std::move(settings_)),
332 prices(std::move(prices_)), outage(outage_),
333 capacity(value(settings, "CAPACITY")), ceff(value(settings, "CEFF")),
334 deff(value(settings, "DEFF")), reserve(0), startSOC(0), soc(0),
335 arbitrage(false) {
336 i64 reservePercent = value(settings, "RESERVE");
337 if (strategy.name == "BACKUP") reservePercent = std::max(reservePercent, *strategy.parameter);
338 reserve = capacity * reservePercent / 100;
339 startSOC = capacity * value(settings, "INITIAL") / 100;
340 soc = startSOC;
341 auto extremes = tariffExtremes(prices);
342 cheapest = extremes.first; peak = extremes.second;
343 arbitrage = strategy.name == "EXPORTPEAK" &&
344 value(settings, "FEEDIN") > priceOf(prices, cheapest);
345 }
346
347 i64 dischargeTarget(i64 deficit, const std::string& period) const {
348 if (strategy.name == "TOU") return period == peak ? deficit : 0;
349 if (strategy.name == "PEAKSHAVE") return std::max<i64>(0, deficit - *strategy.parameter);
350 if (arbitrage && period == cheapest && period != peak) return 0;
351 return deficit;
352 }
353
354 i64 gridCharge(const std::string& period, i64 pvCharge, i64 gridLoad) {
355 i64 headroom = value(settings, "CHARGE") - pvCharge;
356 std::optional<i64> target;
357 if (strategy.name == "TOU" && period == cheapest && period != peak) {
358 target = capacity * *strategy.parameter / 100;
359 } else if (strategy.name == "PEAKSHAVE" && period == cheapest) {
360 target = capacity;
361 headroom = std::min(headroom, std::max<i64>(0, *strategy.parameter - gridLoad));
362 } else if (strategy.name == "BACKUP" && period == cheapest) {
363 target = reserve;
364 } else if (arbitrage && period == cheapest && period != peak) {
365 target = capacity;
366 }
367 if (!target) return 0;
368 i64 amount = std::min(headroom,
369 ceiling(100 * std::max<i64>(0, *target - soc), ceff));
370 soc += amount * ceff / 100;
371 return amount;
372 }
373
374 HourRecord runHour(i64 index) {
375 i64 load = data.loads[index];
376 i64 pv = data.basePV[index] * value(settings, "PVSCALE") / 100;
377 const auto& period = data.periods[index];
378 i64 direct = std::min(load, pv);
379 i64 surplus = pv - direct;
380 i64 deficit = load - direct;
381 i64 pvCharge = std::min({surplus, value(settings, "CHARGE"),
382 ceiling(100 * (capacity - soc), ceff)});
383 soc += pvCharge * ceff / 100;
384 surplus -= pvCharge;
385 bool isOutage = outage && outage->first <= index && index <= outage->second;
386 i64 target = isOutage ? deficit : dischargeTarget(deficit, period);
387 i64 lowerBound = isOutage ? 0 : reserve;
388 i64 loadDrawn = std::min({ceiling(100 * target, deff), value(settings, "DISCHARGE"),
389 std::max<i64>(0, soc - lowerBound)});
390 i64 loadDischarge = loadDrawn * deff / 100;
391 soc -= loadDrawn;
392 deficit -= loadDischarge;
393 i64 gridLoad = 0, shortfall = 0, pvExport = 0, curtailed = 0;
394 i64 gridCharged = 0, exportDrawn = 0, batteryExport = 0;
395 if (isOutage) {
396 shortfall = deficit;
397 curtailed = surplus;
398 } else {
399 gridLoad = deficit;
400 pvExport = std::min(surplus, value(settings, "EXPORT"));
401 curtailed = surplus - pvExport;
402 gridCharged = gridCharge(period, pvCharge, gridLoad);
403 if (arbitrage && period == peak && pvCharge == 0) {
404 exportDrawn = std::min({
405 ceiling(100 * (value(settings, "EXPORT") - pvExport), deff),
406 value(settings, "DISCHARGE") - loadDrawn,
407 std::max<i64>(0, soc - reserve)
408 });
409 batteryExport = exportDrawn * deff / 100;
410 soc -= exportDrawn;
411 }
412 }
413 i64 loss = pvCharge - pvCharge * ceff / 100 +
414 gridCharged - gridCharged * ceff / 100 +
415 loadDrawn - loadDischarge + exportDrawn - batteryExport;
416 std::array<i64, 8> flows = {
417 direct, pvCharge, pvExport, curtailed, loadDischarge,
418 batteryExport, gridLoad, gridCharged
419 };
420 return HourRecord(index, period, load, pv, soc, flows, loadDrawn + exportDrawn,
421 shortfall, loss, priceOf(prices, period), value(settings, "FEEDIN"));
422 }
423
424 SimulationResult run() {
425 std::vector<HourRecord> records;
426 records.reserve(data.loads.size());
427 for (std::size_t i = 0; i < data.loads.size(); ++i) records.push_back(runHour(i));
428 return SimulationResult(strategy, settings, prices, startSOC, reserve, std::move(records));
429 }
430};

432class DispatchDesk {
433 const InputData& data;
434 Settings settings;
435 Prices prices;
436 std::optional<SimulationResult> current;
437
438 const SimulationResult& requireResult() const {
439 if (!current) throw CommandError("no run");
440 return *current;
441 }
442 int checkedDay(const std::string& text) const {
443 i64 day = number(text);
444 if (day < 1 || day > data.dayCount) throw CommandError("invalid day");
445 return static_cast<int>(day);
446 }
447 const SimulationResult& resultWithoutArguments(const std::vector<std::string>& args) const {
448 requireCount(args, 0);
449 return requireResult();
450 }
451 std::pair<int, const SimulationResult&> resultForDay(
452 const std::vector<std::string>& args) const {
453 requireCount(args, 1);
454 int day = checkedDay(args[0]);
455 return {day, requireResult()};
456 }
457 SimulationResult simulate(Strategy strategy, Settings settings_, Prices prices_,
458 std::optional<std::pair<i64, i64>> outage = std::nullopt) const {
459 return Simulator(data, std::move(strategy), std::move(settings_),
460 std::move(prices_), outage).run();
461 }
462 SimulationResult simulate(Strategy strategy) const {
463 return simulate(std::move(strategy), settings, prices);
464 }
465 SimulationResult baseline(Settings settings_, Prices prices_) const {
466 settings_["CAPACITY"] = 0;
467 return simulate({"SELF", std::nullopt}, std::move(settings_), std::move(prices_));
468 }
469
470 std::vector<std::string> run(const std::vector<std::string>& args) {
471 Strategy strategy = oneStrategy(args);
472 current = simulate(strategy);
473 const auto& result = *current;
474 const auto& total = result.total;
475 return {"RUN " + strategyText(strategy) + " bill " + money(result.bill) +
476 " import " + str(total.imported) + " export " + str(total.exported) +
477 " soc " + str(result.endSOC)};
478 }

480 std::vector<std::string> setValue(const std::vector<std::string>& args) {
481 requireCount(args, 2);
482 auto it = SETTING_RANGES.find(args[0]);
483 if (it == SETTING_RANGES.end()) throw CommandError("unknown parameter " + args[0]);
484 i64 newValue = number(args[1]);
485 if (newValue < it->second.first || newValue > it->second.second)
486 throw CommandError("invalid value for " + args[0]);
487 i64 oldValue = settings.at(args[0]);
488 settings[args[0]] = newValue;
489 return {"SET " + args[0] + " " + str(oldValue) + " -> " + str(newValue)};
490 }
491
492 std::vector<std::string> changePrice(const std::vector<std::string>& args) {
493 requireCount(args, 2);
494 auto it = std::find_if(prices.begin(), prices.end(), [&](const auto& p) {
495 return p.first == args[0];
496 });
497 if (it == prices.end()) throw CommandError("unknown period " + args[0]);
498 i64 newValue = number(args[1]);
499 if (newValue < 0 || newValue > 10000)
500 throw CommandError("invalid value for " + args[0]);
501 i64 oldValue = it->second;
502 it->second = newValue;
503 auto [cheapest, peak] = tariffExtremes(prices);
504 return {"PRICE " + args[0] + " " + str(oldValue) + " -> " + str(newValue) +
505 " cheapest " + cheapest + " peak " + peak};
506 }

508 std::vector<std::string> reset(const std::vector<std::string>& args) {
509 requireCount(args, 0);
510 settings = data.initialSettings;
511 prices = data.initialPrices;
512 return {"RESET"};
513 }

515 std::vector<std::string> config(const std::vector<std::string>& args) const {
516 requireCount(args, 0);
517 std::string battery = "BATTERY", grid = "GRID";
518 for (std::size_t i = 0; i < 12; ++i) {
519 std::string name = SETTING_NAMES[i];
520 std::transform(name.begin(), name.end(), name.begin(), [](char c) {
521 return static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
522 });
523 std::string part = " " + name + " " + str(value(settings, SETTING_NAMES[i]));
524 (i < 8 ? battery : grid) += part;
525 }
526 std::string tariff = "TARIFF";
527 for (const auto& [name, cost] : prices) tariff += " " + name + " " + str(cost);
528 auto [cheapest, peak] = tariffExtremes(prices);
529 return {battery, grid, "SOLAR pvscale " + str(value(settings, "PVSCALE")),
530 tariff + " cheapest " + cheapest + " peak " + peak};
531 }

533 std::vector<std::string> noBattery(const std::vector<std::string>& args) const {
534 requireCount(args, 0);
535 auto result = baseline(settings, prices);
536 const auto& total = result.total;
537 return {"NOBATTERY bill " + money(result.bill) +
538 " import " + str(total.imported) + " export " + str(total.exported) +
539 " curtailed " + str(total.curtailed) + " peak " + str(result.peak)};
540 }

542 std::vector<std::string> compare(const std::vector<std::string>& args) const {
543 auto strategies = readStrategies(args);
544 if (strategies.empty()) throw CommandError("wrong number of arguments");
545 auto noBattery = baseline(settings, prices);
546 std::vector<SimulationResult> trials;
547 trials.reserve(strategies.size());
548 for (const auto& strategy : strategies) trials.push_back(simulate(strategy));
549 std::stable_sort(trials.begin(), trials.end(), [](const auto& a, const auto& b) {
550 return a.bill < b.bill;
551 });
552 std::vector<std::string> result = {"COMPARE baseline " + money(noBattery.bill)};
553 for (std::size_t i = 0; i < trials.size(); ++i) {
554 const auto& trial = trials[i];
555 result.push_back(str(i + 1) + ". " + strategyText(trial.strategy) +
556 " bill " + money(trial.bill) +
557 " saving " + money(noBattery.bill - trial.bill) +
558 " sufficiency " + trial.sufficiency + "%");
559 }
560 return result;
561 }

563 std::vector<std::string> size(const std::vector<std::string>& args) const {
564 requireCount(args, 3);
565 auto capacities = checkedRange(number(args[0]), number(args[1]), number(args[2]),
566 SETTING_RANGES.at("CAPACITY"));
567 auto noBattery = baseline(settings, prices);
568 std::vector<std::string> lines;
569 std::optional<i64> bestCapacity;
570 i64 bestCost = 0, bestSaving = 0;
571 for (i64 capacity : capacities) {
572 Settings trialSettings = settings;
573 trialSettings["CAPACITY"] = capacity;
574 auto trial = simulate({"SELF", std::nullopt}, trialSettings, prices);
575 i64 saving = noBattery.bill - trial.bill;
576 i64 cost = capacity * value(trialSettings, "COST") / 1000;
577 std::string payback = "never";
578 if (saving > 0) {
579 i64 tenths = 10 * cost / saving;
580 payback = str(tenths / 10) + "." + str(tenths % 10);
581 if (!bestCapacity || cost * bestSaving < bestCost * saving) {
582 bestCapacity = capacity;
583 bestCost = cost;
584 bestSaving = saving;
585 }
586 }
587 lines.push_back("SIZE " + str(capacity) + " bill " + money(trial.bill) +
588 " saving " + money(saving) + " cost " + money(cost) +
589 " payback " + payback);
590 }
591 lines.push_back("BEST " + (bestCapacity ? str(*bestCapacity) : "none"));
592 return lines;
593 }

595 std::vector<std::string> sweep(const std::vector<std::string>& args) const {
596 if (args.size() < 5) throw CommandError("wrong number of arguments");
597 auto it = SETTING_RANGES.find(args[0]);
598 if (it == SETTING_RANGES.end()) throw CommandError("unknown parameter " + args[0]);
599 auto values = checkedRange(number(args[1]), number(args[2]), number(args[3]), it->second);
600 Strategy strategy = oneStrategy({args.begin() + 4, args.end()});
601 std::vector<std::string> lines;
602 i64 bestBill = 0, bestValue = 0;
603 bool found = false;
604 for (i64 settingValue : values) {
605 Settings trialSettings = settings;
606 trialSettings[args[0]] = settingValue;
607 auto trial = simulate(strategy, trialSettings, prices);
608 auto noBattery = baseline(trialSettings, prices);
609 if (!found || trial.bill < bestBill) {
610 found = true;
611 bestBill = trial.bill;
612 bestValue = settingValue;
613 }
614 lines.push_back("SWEEP " + args[0] + " " + str(settingValue) +
615 " bill " + money(trial.bill) +
616 " saving " + money(noBattery.bill - trial.bill) +
617 " sufficiency " + trial.sufficiency + "%");
618 }
619 lines.push_back("BEST " + str(bestValue));
620 return lines;
621 }

623 std::vector<std::string> outage(const std::vector<std::string>& args) const {
624 requireCount(args, 3);
625 int day = checkedDay(args[0]);
626 i64 start = number(args[1]), end = number(args[2]);
627 if (start < 0 || start > end || end > 23) throw CommandError("invalid hours");
628 const auto& current = requireResult();
629 i64 first = (day - 1) * 24 + start;
630 i64 last = (day - 1) * 24 + end;
631 auto result = simulate(current.strategy, current.settings, current.prices,
632 std::make_pair(first, last));
633 i64 before = first > 0 ? result.hours[first - 1].soc : result.startSOC;
634 std::vector<std::string> lines = {
635 "OUTAGE day " + str(day) + " hours " + str(start) + "-" + str(end) +
636 " strategy " + strategyText(result.strategy) + " soc " + str(before),
637 "hour load pv disch soc short curt"
638 };
639 Totals total;
640 for (i64 i = first; i <= last; ++i) {
641 const auto& r = result.hours[i];
642 total.add(r);
643 lines.push_back(tableRow({str(r.hour), str(r.load), str(r.pv),
644 str(r.discharge), str(r.soc), str(r.shortfall),
645 str(r.curtailed)}, {4, 7, 7, 7, 7, 7, 7}));
646 }
647 std::string served = total.load
648 ? percentage(total.load - total.shortfall, total.load) : "100.00";
649 lines.push_back("TOTAL load " + str(total.load) +
650 " unserved " + str(total.shortfall) +
651 " served " + served + "% bill " + money(result.bill));
652 return lines;
653 }

655 std::vector<std::string> diff(const std::vector<std::string>& args) const {
656 if (args.size() < 2) throw CommandError("wrong number of arguments");
657 int day = checkedDay(args[0]);
658 Strategy strategy = oneStrategy({args.begin() + 1, args.end()});
659 const auto& current = requireResult();
660 auto comparison = simulate(strategy, current.settings, current.prices);
661 std::vector<std::string> lines = {
662 "DIFF day " + str(day) + " " + data.dayNames[day - 1] + " " +
663 strategyText(current.strategy) + " vs " + strategyText(strategy),
664 "hour imp1 imp2 exp1 exp2 soc1 soc2"
665 };
666 for (int hour = 0; hour < 24; ++hour) {
667 const auto& r1 = current.hours[(day - 1) * 24 + hour];
668 const auto& r2 = comparison.hours[(day - 1) * 24 + hour];
669 lines.push_back(tableRow({str(r1.hour), str(r1.imported), str(r2.imported),
670 str(r1.exported), str(r2.exported),
671 str(r1.soc), str(r2.soc)}, {4, 7, 7, 7, 7, 7, 7}));
672 }
673 i64 cost1 = current.daily[day - 1].netCost;
674 i64 cost2 = comparison.daily[day - 1].netCost;
675 lines.push_back("TOTAL cost1 " + money(cost1) + " cost2 " + money(cost2) +
676 " difference " + money(cost2 - cost1));
677 return lines;
678 }

741 std::vector<std::string> savings(const std::vector<std::string>& args) const {
742 const auto& result = resultWithoutArguments(args);
743 auto noBattery = baseline(result.settings, result.prices);
744 std::vector<std::string> lines = {
745 "SAVINGS strategy " + strategyText(result.strategy)
746 };
747 auto add = [&](const std::string& label, i64 before, i64 after) {
748 lines.push_back(label + " " + money(before) + " -> " + money(after) +
749 " saving " + money(before - after));
750 };
751 for (std::size_t i = 0; i < result.prices.size(); ++i) {
752 add(result.prices[i].first, noBattery.byPeriod[i].importCost,
753 result.byPeriod[i].importCost);
754 }
755 add("EXPORT", -noBattery.total.exportIncome, -result.total.exportIncome);
756 add("DEMAND", noBattery.demandCharge, result.demandCharge);
757 add("TOTAL", noBattery.bill, result.bill);
758 return lines;

799 std::vector<std::string> peaks(const std::vector<std::string>& args) const {
800 requireCount(args, 1);
801 i64 count = number(args[0]);
802 if (count < 1) throw CommandError("invalid count");
803 const auto& result = requireResult();
804 std::vector<const HourRecord*> records;
805 for (const auto& record : result.hours) {
806 if (record.imported > 0) records.push_back(&record);
807 }
808 std::sort(records.begin(), records.end(), [](const auto* a, const auto* b) {
809 return a->imported != b->imported
810 ? a->imported > b->imported : a->index < b->index;
811 });
812 std::vector<std::string> lines = {
813 "PEAKS strategy " + strategyText(result.strategy)
814 };
815 for (std::size_t i = 0; i < records.size() && i < static_cast<std::size_t>(count); ++i) {
816 const auto& r = *records[i];
817 lines.push_back(str(i + 1) + ". day " + str(r.day) +
818 " hour " + str(r.hour) + " " + r.period +
819 " import " + str(r.imported) + " load " + str(r.load) +
820 " soc " + str(r.soc));
821 }
822 if (records.empty()) lines.push_back("none");
823 return lines;
824 }

867 std::vector<std::string> cycles(const std::vector<std::string>& args) const {
868 const auto& result = resultWithoutArguments(args);
869 i64 drawn = result.total.drawn;
870 i64 capacity = value(result.settings, "CAPACITY");
871 i64 common = std::gcd(drawn, capacity);
872 i64 numerator = drawn / common, denominator = capacity / common;
873 i64 thousandths = 1000 * drawn / capacity;
874 std::string decimal = str(thousandths / 1000) + "." +
875 padded(str(thousandths % 1000), 3);
876 std::replace(decimal.begin(), decimal.end(), ' ', '0');
877 return {"CYCLES " + str(numerator) + "/" + str(denominator) +
878 " " + decimal + " drawn " + str(drawn) +
879 " losses " + str(result.total.loss)};
880 }

882 std::vector<std::string> usage(const std::vector<std::string>& args) const {
883 const auto& result = resultWithoutArguments(args);
884 i64 charging = 0, discharging = 0, idle = 0, full = 0, low = 0;
885 i64 run = 0, longest = 0;
886 const HourRecord* first = nullptr;
887 for (const auto& record : result.hours) {
888 charging += record.charge > 0;
889 discharging += record.drawn > 0;
890 idle += record.charge == 0 && record.drawn == 0;
891 full += record.soc == value(result.settings, "CAPACITY");
892 if (record.soc <= result.reserve) {
893 ++low;
894 ++run;
895 if (run > longest) {
896 longest = run;
897 first = &result.hours[record.index - run + 1];
898 }
899 } else {
900 run = 0;
901 }
902 }
903 std::string line = "USAGE charging " + str(charging) +
904 " discharging " + str(discharging) + " idle " + str(idle) +
905 " full " + str(full) + " low " + str(low) +
906 " longest " + str(longest);
907 if (first) line += " from day " + str(first->day) + " hour " + str(first->hour);
908 return {line};
909 }

949 std::vector<std::string> execute(const std::vector<std::string>& tokens) {
950 const std::string& cmd = tokens[0];
951 std::vector<std::string> args(tokens.begin() + 1, tokens.end());
952 try {
953 if (cmd == "RUN") return run(args);
954 if (cmd == "SET") return setValue(args);
955 if (cmd == "PRICE") return changePrice(args);
956 if (cmd == "RESET") return reset(args);
957 if (cmd == "CONFIG") return config(args);
958 if (cmd == "NOBATTERY") return noBattery(args);
959 if (cmd == "COMPARE") return compare(args);
960 if (cmd == "SIZE") return size(args);
961 if (cmd == "SWEEP") return sweep(args);
962 if (cmd == "OUTAGE") return outage(args);
963 if (cmd == "DIFF") return diff(args);
964 if (cmd == "HOURLY") return hourly(args);
965 if (cmd == "DAILY") return daily(args);
966 if (cmd == "BILL") return bill(args);
967 if (cmd == "SAVINGS") return savings(args);
968 if (cmd == "BYPERIOD") return byPeriod(args);
969 if (cmd == "DAYTYPE") return dayType(args);
970 if (cmd == "PEAKS") return peaks(args);
971 if (cmd == "PROFILE") return profile(args);
972 if (cmd == "FLOWS") return flows(args);
973 if (cmd == "SELFSUFF") return selfSuff(args);
974 if (cmd == "CYCLES") return cycles(args);
975 if (cmd == "USAGE") return usage(args);
976 if (cmd == "HISTOGRAM") return histogram(args);
977 if (cmd == "CHART") return chart(args);
978 return {"ERROR unknown command " + cmd};
979 } catch (const CommandError& error) {
980 return {"ERROR " + std::string(error.what())};
981 }
982 }
