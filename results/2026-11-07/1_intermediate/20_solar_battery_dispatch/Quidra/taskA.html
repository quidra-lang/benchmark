82Strategy[] | error parse_strategies(const string[] arguments)
83 Strategy[] strategies := []
84 position := 0
85 while position < int(#arguments)
86 name := arguments[position]
87 position += 1
88 if name == "SELF" or name == "EXPORTPEAK"
89 strategies = strategies.append(Strategy(name))
90 continue
91 if name != "TOU" and name != "PEAKSHAVE" and name != "BACKUP"
92 return error("unknown strategy " + name)
93 if position == int(#arguments)
94 return error("missing parameter for " + name)
95 value := parse_integer(arguments[position])
96 position += 1
97 if value < (if name == "TOU" then 1 else 0) or value > (if name == "PEAKSHAVE" then 100000 else 100)
98 return error("invalid parameter for " + name)
99 strategies = strategies.append(Strategy(name, value))
100 return strategies

111House read_house()
112 battery := read_tokens()
113 grid := read_tokens()
114 values := [for token := battery[1:][] then parse_integer(token)]
115 values = values.concat([for token := grid[1:][] then parse_integer(token)]).append(100)
116 header := read_tokens()
117 string[] periods := []
118 int[] prices := []
119 for: 0 <= index < parse_integer(header[1])
120 fields := read_tokens()
121 periods = periods.append(fields[0])
122 prices = prices.append(parse_integer(fields[1]))
123 int[][] divisions := []
124 for: 0 <= kind < 2
125 fields := read_tokens()
126 hours := array(24, fill = 0)
127 position := 1
128 while position < int(#fields)
129 first := parse_integer(fields[position])
130 last := if position + 2 < int(#fields) then parse_integer(fields[position + 2]) else 24
131 period := find_index(periods, fields[position + 1])
132 for: first <= hour < last
133 hours[hour] = period
134 position += 2
135 divisions = divisions.append(hours)
136 string[] profile_names := []
137 int[][] profiles := []
138 header = read_tokens()
139 for: 0 <= index < parse_integer(header[1])
140 fields := read_tokens()
141 profile := [for token := fields[2:][] then parse_integer(token)]
142 profile_names = profile_names.append(fields[0] + " " + fields[1])
143 profiles = profiles.append(profile)
144 header = read_tokens()
145 days := parse_integer(header[1])
146 int[] loads := []
147 int[] bases := []
148 string[] day_names := []
149 bool[] weekends := []
150 weekdays := "MON TUE WED THU FRI SAT SUN".split(" ")
151 for: 0 <= day < days
152 fields := read_tokens()
153 load_profile := profiles[find_index(profile_names, "LOAD " + fields[0])]
154 pv_profile := profiles[find_index(profile_names, "PV " + fields[2])]
155 loads = loads.concat(load_profile * parse_integer(fields[1]) / 100)
156 bases = bases.concat(pv_profile * parse_integer(fields[3]) / 100)
157 day_names = day_names.append(weekdays[day % 7])
158 weekends = weekends.append(day % 7 >= 5)
159 holidays := read_tokens()
160 for token := holidays[2:][]
161 day := parse_integer(token) - 1
162 day_names[day] = "HOL"
163 weekends[day] = true
164 int[] hourly_periods := []
165 for weekend := weekends[]
166 hourly_periods = hourly_periods.concat(divisions[if weekend then 1 else 0])
167 header = read_tokens()
168 for: 0 <= index < parse_integer(header[1])
169 fields := read_tokens()
170 hour := 24 * (parse_integer(fields[0]) - 1) + parse_integer(fields[1])
171 loads[hour] = parse_integer(fields[2])
172 bases[hour] = parse_integer(fields[3])
173 return House(settings_from(values), prices, periods, loads, bases, hourly_periods, day_names, weekends)

192class Result(Strategy strategy, Settings settings, int[] prices, Record[] records, int starting, int reserve)
193 totals := Totals(records)
194 int peak := [for entry := records[] then entry.imported].max()
195 int demand_charge := peak * settings.demand / 1000
196 int bill := totals.net + settings.fixed + demand_charge
197 int final_soc := records[-1].soc
198 int[] period_imports := [for index := prices.indices()[] then [for entry := records[] if entry.period == int(index) then entry.imported].sum()]
199 int[] period_costs := [for index := prices.indices()[] then [for entry := records[] if entry.period == int(index) then entry.import_cost].sum()]

201Result simulate(const House house, const Settings settings, const int[] prices, const Strategy strategy, outage_first := -1, outage_last := -1)
202 capacity := settings.capacity
203 reserve := capacity * (if strategy.name == "BACKUP" then math.max(settings.reserve, strategy.parameter) else settings.reserve) / 100
204 soc := capacity * settings.initial / 100
205 starting := soc
206 cheapest := extreme_period(prices, false)
207 peak := extreme_period(prices, true)
208 arbitrage := strategy.name == "EXPORTPEAK" and settings.feedin > prices[cheapest]
209 Record[] records := []
210 for: 0 <= hour < int(#house.loads)
211 load := house.loads[hour]
212 pv := house.bases[hour] * settings.pvscale / 100
213 period := house.hourly_periods[hour]
214 outage := outage_first <= hour <= outage_last
215 direct := math.min(load, pv)
216 surplus := pv - direct
217 deficit := load - direct
218 x1 := math.min(surplus, math.min(settings.charge, ceil_div(100 * (capacity - soc), settings.ceff)))
219 stored1 := x1 * settings.ceff / 100
220 soc += stored1
221 surplus -= x1
222 target := deficit
223 if not outage
224 if strategy.name == "TOU" and period != peak
225 target = 0
226 elif strategy.name == "PEAKSHAVE"
227 target = math.max(0, deficit - strategy.parameter)
228 elif arbitrage and period == cheapest and period != peak
229 target = 0
230 lower := if outage then 0 else reserve
231 y1 := math.min(ceil_div(100 * target, settings.deff), math.min(settings.discharge, math.max(0, soc - lower)))
232 d1 := y1 * settings.deff / 100
233 soc -= y1
234 deficit -= d1
235 grid_load := if outage then 0 else deficit
236 shortfall := if outage then deficit else 0
237 e1 := if outage then 0 else math.min(surplus, settings.export_limit)
238 curtailed := surplus - e1
239 x2 := 0
240 y2 := 0
241 d2 := 0
242 stored2 := 0
243 if not outage
244 headroom := settings.charge - x1
245 desired := -1
246 if strategy.name == "TOU" and period == cheapest and period != peak
247 desired = capacity * strategy.parameter / 100
248 elif strategy.name == "PEAKSHAVE" and period == cheapest
249 desired = capacity
250 headroom = math.min(headroom, math.max(0, strategy.parameter - grid_load))
251 elif strategy.name == "BACKUP" and period == cheapest
252 desired = reserve
253 elif arbitrage and period == cheapest and period != peak
254 desired = capacity
255 if desired >= 0
256 x2 = math.min(headroom, ceil_div(100 * math.max(0, desired - soc), settings.ceff))
257 stored2 = x2 * settings.ceff / 100
258 soc += stored2
259 if arbitrage and period == peak and x1 == 0
260 y2 = math.min(ceil_div(100 * (settings.export_limit - e1), settings.deff), math.min(settings.discharge - y1, math.max(0, soc - reserve)))
261 d2 = y2 * settings.deff / 100
262 soc -= y2
263 imported := grid_load + x2
264 exported := e1 + d2
265 loss := x1 - stored1 + x2 - stored2 + y1 - d1 + y2 - d2
266 flows := [direct, x1, e1, curtailed, d1, d2, grid_load, x2]
267 records = records.append(Record(load, pv, period, x1 + x2, d1 + d2, y1 + y2, soc, imported, exported, curtailed, shortfall, loss, flows, imported * prices[period] / 100000, exported * settings.feedin / 100000))
268 return Result(strategy, settings, prices, records, starting, reserve)

279int[] ranked_indices(const int[] values, descending := false)
280 map<int, int[]> groups
281 for index := values.indices()[]
282 int[] positions := []
283 match groups.get(values[index])
284 int[] previous
285 positions = previous
286 none
287 positions = []
288 .groups.set(values[index], positions.append(int(index)))

291 match groups.get(priority)
292 int[] positions
293 indices = indices.concat(positions)

466void | error command(State &state, const string[] words)
467 verb := words[0]
468 arguments := words[1:]
469 known := "RUN SET PRICE RESET CONFIG NOBATTERY COMPARE SIZE SWEEP OUTAGE DIFF HOURLY DAILY BILL SAVINGS BYPERIOD DAYTYPE PEAKS PROFILE FLOWS SELFSUFF CYCLES USAGE HISTOGRAM CHART".split(" ")
470 if find_index(known, verb) < 0
471 return error("unknown command " + verb)
472 count := int(#arguments)
473 if verb == "RUN" or verb == "COMPARE"
474 strategies := try parse_strategies(arguments)
475 if #strategies == 0 or (verb == "RUN" and #strategies != 1)
476 return error("wrong number of arguments")
477 results := [for strategy := strategies[] then simulate(state.house, state.settings, state.prices, strategy)]
478 if verb == "RUN"
479 result := results[0]
480 state.current = result
481 emit("RUN {result.strategy.notation} bill {money(result.bill)} import {result.totals.imported} export {result.totals.exported} soc {result.final_soc}")
482 else
483 original := baseline(state.house, state.settings, state.prices).bill
484 emit("COMPARE baseline {money(original)}")
485 bills := [for result := results[] then result.bill]
486 rank := 0
487 for index := ranked_indices(bills)[]
488 result := results[index]
489 rank += 1
490 emit("{rank}. {result.strategy.notation} bill {money(result.bill)} saving {money(original - result.bill)} sufficiency {sufficiency(result)}%")
491 return void
492 if verb == "SET" or verb == "PRICE"
493 if count != 2
494 return error("wrong number of arguments")
495 name := arguments[0]
496 if verb == "SET"
497 index := try parameter_index(name)
498 value := parse_integer(arguments[1])
499 minimums := item_minimums()
500 maximums := item_maximums()
501 if value < minimums[index] or value > maximums[index]
502 return error("invalid value for " + name)
503 values := setting_values(state.settings)
504 state.settings = changed_setting(state.settings, index, value)
505 emit("SET {name} {values[index]} -> {value}")
506 else
507 index := find_index(state.house.periods, name)
