3const SETTING_NAMES = split("CAPACITY INITIAL CHARGE DISCHARGE CEFF DEFF RESERVE COST EXPORT FEEDIN FIXED DEMAND PVSCALE")
4const SETTING_RANGES = Dict(zip(SETTING_NAMES, [(1, 100000), (0, 100), (1, 20000),
5 (1, 20000), (1, 100), (1, 100), (0, 100), (0, 100000), (0, 100000),
6 (0, 10000), (0, 100000), (0, 100000), (0, 300)]))
7const FLOW_NAMES = split("PV>LOAD PV>BATTERY PV>GRID PV>CURTAIL BATTERY>LOAD BATTERY>GRID GRID>LOAD GRID>BATTERY")
8const STRATEGY_RANGES = Dict("TOU" => (1, 100), "PEAKSHAVE" => (0, 100000), "BACKUP" => (0, 100))
9const REPORTS = Set(split("DAILY BILL SAVINGS BYPERIOD DAYTYPE PROFILE FLOWS SELFSUFF CYCLES USAGE HISTOGRAM"))

11struct CommandError <: Exception
12 message::String
13end
14fail(message) = throw(CommandError(message))
15number(text) = parse(Int, text)
16money(value) = @sprintf("%s%d.%02d", value < 0 ? "-" : "", div(abs(value), 100), mod(abs(value), 100))
17percentage(a, b) = money(b == 0 ? 0 : div(10000 * a, b))
18decimal(a, b, places) = string(div(a * 10^places, b) ÷ 10^places, ".",
19 lpad(mod(div(a * 10^places, b), 10^places), places, '0'))
20bar(width) = "|" * repeat("#", width) * repeat(".", 20 - width) * "|"
21row(values, widths) = println(join(lpad(string(value), width) for (value, width) in zip(values, widths)))

76total(records, field) = sum((getfield(record, field) for record in records); init=0)
77netcost(records) = total(records, :cost) - total(records, :income)
78sufficiency(result) = percentage(max(0, total(result.records, :load) - total(result.records, :imported)), total(result.records, :load))
79dayrecords(result, day) = @view result.records[(day - 1) * 24 + 1:day * 24]

81function read_data()
82 tokens() = split(readline())
83 battery = number.(tokens()[2:end])
84 grid = number.(tokens()[2:end])
85 settings = Dict{String, Int}(zip(SETTING_NAMES, vcat(battery, grid, 100)))
86 count = number(tokens()[2])
87 periods = String[]
88 prices = Int[]
89 for _ in 1:count
90 fields = tokens()
91 push!(periods, fields[1])
92 push!(prices, number(fields[2]))
93 end
94 period_index = Dict(name => i for (i, name) in enumerate(periods))
95 divisions = Vector{Int}[]
96 for _ in 1:2
97 fields = tokens()
98 hours = zeros(Int, 24)
99 for i in 2:2:length(fields)
100 first = number(fields[i])
101 last = i + 2 <= length(fields) ? number(fields[i + 2]) - 1 : 23
102 hours[first + 1:last + 1] .= period_index[fields[i + 1]]
103 end
104 push!(divisions, hours)
105 end
106 count = number(tokens()[2])
107 profiles = Dict{Tuple{String, String}, Vector{Int}}()
108 for _ in 1:count
109 fields = tokens()
110 profiles[(fields[1], fields[2])] = number.(fields[3:end])
111 end
112 days = number(tokens()[2])
113 loads = zeros(Int, 24, days)
114 generation = zeros(Int, 24, days)
115 for day in 1:days
116 fields = tokens()
117 loads[:, day] = div.(profiles[("LOAD", fields[1])] .* number(fields[2]), 100)
118 generation[:, day] = div.(profiles[("PV", fields[3])] .* number(fields[4]), 100)
119 end
120 holidays = Set(number.(tokens()[3:end]))
121 count = number(tokens()[2])
122 for _ in 1:count
123 day, hour, load, pv = number.(tokens())
124 loads[hour + 1, day] = load
125 generation[hour + 1, day] = pv
126 end
127 weekdays = split("MON TUE WED THU FRI SAT SUN")
128 daynames = [day in holidays ? "HOL" : String(weekdays[mod(day - 1, 7) + 1]) for day in 1:days]
129 weekends = BitVector([day in holidays || mod(day - 1, 7) >= 5 for day in 1:days])
130 hours = [InputHour(day, hour, divisions[weekends[day] ? 2 : 1][hour + 1],
131 loads[hour + 1, day], generation[hour + 1, day])
132 for day in 1:days for hour in 0:23]
133 Dataset(periods, hours, daynames, weekends), settings, prices
134end

136function simulate(data, strategy, settings, prices; outage=nothing)
137 capacity = settings["CAPACITY"]
138 ceff, deff = settings["CEFF"], settings["DEFF"]
139 reserve_pct = strategy.name == "BACKUP" ? max(settings["RESERVE"], strategy.parameter) : settings["RESERVE"]
140 reserve = div(capacity * reserve_pct, 100)
141 starting = soc = div(capacity * settings["INITIAL"], 100)
142 low, high = argmin(prices), argmax(prices)
143 arbitrage = strategy.name == "EXPORTPEAK" && settings["FEEDIN"] > prices[low]
144 records = HourRecord[]
145 for input in data.hours
146 pv = div(input.pv * settings["PVSCALE"], 100)
147 direct = min(input.load, pv)
148 surplus, deficit = pv - direct, input.load - direct
149 x1 = min(surplus, settings["CHARGE"], cld(100 * (capacity - soc), ceff))
150 stored1 = div(x1 * ceff, 100)
151 soc += stored1
152 surplus -= x1
153 disconnected = outage !== nothing && input.day == outage[1] && outage[2] <= input.hour <= outage[3]
154 target = deficit
155 if !disconnected
156 if strategy.name == "TOU"
157 target = input.period == high ? deficit : 0
158 elseif strategy.name == "PEAKSHAVE"
159 target = max(0, deficit - strategy.parameter)
160 elseif strategy.name == "EXPORTPEAK" && arbitrage && input.period == low && low != high
161 target = 0
162 end
163 end
164 lower = disconnected ? 0 : reserve
165 y1 = min(cld(100 * target, deff), settings["DISCHARGE"], max(0, soc - lower))
166 d1 = div(y1 * deff, 100)
167 soc -= y1
168 deficit -= d1
169 x2 = y2 = d2 = stored2 = 0
170 if disconnected
171 grid = export_pv = 0
172 shortfall, curtailed = deficit, surplus
173 else
174 grid, shortfall = deficit, 0
175 export_pv = min(surplus, settings["EXPORT"])
176 curtailed = surplus - export_pv
177 headroom = settings["CHARGE"] - x1
178 charge_target = nothing
179 if strategy.name == "TOU" && input.period == low && low != high
180 charge_target = div(capacity * strategy.parameter, 100)
181 elseif strategy.name == "PEAKSHAVE" && input.period == low
182 charge_target = capacity
183 headroom = min(headroom, max(0, strategy.parameter - grid))
184 elseif strategy.name == "BACKUP" && input.period == low
185 charge_target = reserve
186 elseif strategy.name == "EXPORTPEAK" && arbitrage && input.period == low && low != high
187 charge_target = capacity
188 end
189 if charge_target !== nothing
190 x2 = min(headroom, cld(100 * max(0, charge_target - soc), ceff))
191 stored2 = div(x2 * ceff, 100)
192 soc += stored2
193 end
194 if strategy.name == "EXPORTPEAK" && arbitrage && input.period == high && x1 == 0
195 y2 = min(cld(100 * (settings["EXPORT"] - export_pv), deff),
196 settings["DISCHARGE"] - y1, max(0, soc - reserve))
197 d2 = div(y2 * deff, 100)
198 soc -= y2
199 end
200 end
201 imported, exported = grid + x2, export_pv + d2
202 loss = x1 - stored1 + x2 - stored2 + y1 - d1 + y2 - d2
203 flows = (direct, x1, export_pv, curtailed, d1, d2, grid, x2)
204 push!(records, HourRecord(input.day, input.hour, input.period, input.load, pv,
205 x1 + x2, d1 + d2, y1 + y2, soc, imported, exported, curtailed, shortfall,
206 loss, flows, div(imported * prices[input.period], 100000),
207 div(exported * settings["FEEDIN"], 100000)))
208 end
209 peak = maximum(record.imported for record in records)
210 demand = div(peak * settings["DEMAND"], 1000)
211 bill = netcost(records) + settings["FIXED"] + demand
212 Result(strategy, copy(settings), copy(prices), records, starting, reserve, peak, demand, bill)
213end

215baseline(data, settings, prices) = simulate(data, Strategy("SELF", 0), merge(settings, Dict("CAPACITY" => 0)), prices)

217function strategies(tokens)
218 result = Strategy[]
219 i = 1
220 while i <= length(tokens)
221 name = tokens[i]
222 i += 1
223 parameter = 0
224 if haskey(STRATEGY_RANGES, name)
225 i <= length(tokens) || fail("missing parameter for $name")
226 parameter = number(tokens[i])
227 i += 1
228 low, high = STRATEGY_RANGES[name]
229 low <= parameter <= high || fail("invalid parameter for $name")
230 elseif name != "SELF" && name != "EXPORTPEAK"
231 fail("unknown strategy $name")
232 end
233 push!(result, Strategy(String(name), parameter))
234 end
235 result
236end

238function one_strategy(tokens)
239 choices = strategies(tokens)
240 length(choices) == 1 || fail("wrong number of arguments")
241 only(choices)
242end
243
244arguments(args, count) = length(args) == count || fail("wrong number of arguments")
245function valid_day(text, data)
246 day = number(text)
247 1 <= day <= length(data.daynames) || fail("invalid day")
248 day
249end
250function parameter_range(name)
251 haskey(SETTING_RANGES, name) || fail("unknown parameter $name")
252 SETTING_RANGES[name]
253end
254function values_range(args, bounds)
255 low, high, step = number.(args)
256 bounds[1] <= low <= high <= bounds[2] && step >= 1 && div(high - low, step) + 1 <= 20 || fail("invalid range")
257 low:step:high
258end

260function configure(data, settings, prices)
261 println("BATTERY capacity $(settings["CAPACITY"]) initial $(settings["INITIAL"]) charge $(settings["CHARGE"]) discharge $(settings["DISCHARGE"]) ceff $(settings["CEFF"]) deff $(settings["DEFF"]) reserve $(settings["RESERVE"]) cost $(settings["COST"])")
262 println("GRID export $(settings["EXPORT"]) feedin $(settings["FEEDIN"]) fixed $(settings["FIXED"]) demand $(settings["DEMAND"])")
263 println("SOLAR pvscale $(settings["PVSCALE"])")
264 tariff = join(("$name $price" for (name, price) in zip(data.periods, prices)), " ")
265 println("TARIFF $tariff cheapest $(data.periods[argmin(prices)]) peak $(data.periods[argmax(prices)])")
266end

268function hourly(data, result, day)
269 println("HOURLY day $day $(data.daynames[day]) strategy $(notation(result.strategy))")
270 println("hour load pv charge disch soc import export curt")
271 records = dayrecords(result, day)
272 for r in records
273 row((r.hour, r.load, r.pv, r.charge, r.discharge, r.soc, r.imported, r.exported, r.curtailed), (4, 7, 7, 7, 7, 7, 7, 7, 7))
274 end
275 row(("sum", total(records, :load), total(records, :pv), total(records, :charge), total(records, :discharge), "",
276 total(records, :imported), total(records, :exported), total(records, :curtailed)), (4, 7, 7, 7, 7, 7, 7, 7, 7))
277end

279function daily(data, result)
280 println("DAILY strategy $(notation(result.strategy))")
281 println("day name load pv import export curt soc cost")
282 for day in eachindex(data.daynames)
283 records = dayrecords(result, day)
284 row((day, data.daynames[day], total(records, :load), total(records, :pv),
285 total(records, :imported), total(records, :exported), total(records, :curtailed),
286 last(records).soc, money(netcost(records))), (3, 5, 8, 8, 8, 8, 8, 8, 10))
287 end
288 r = result.records
289 println("TOTAL load $(total(r, :load)) pv $(total(r, :pv)) import $(total(r, :imported)) export $(total(r, :exported)) curtailed $(total(r, :curtailed)) cost $(money(netcost(r)))")
290end

292function bill(data, result)
293 println("BILL strategy $(notation(result.strategy)) days $(length(data.daynames))")
294 for (i, name) in enumerate(data.periods)
295 records = filter(r -> r.period == i, result.records)
296 println("IMPORT $name $(total(records, :imported)) Wh $(money(total(records, :cost)))")
297 end
298 println("EXPORT $(total(result.records, :exported)) Wh $(money(-total(result.records, :income)))")
299 println("FIXED $(money(result.settings["FIXED"]))")
300 println("DEMAND $(result.peak) Wh $(money(result.demand))")
301 println("TOTAL $(money(result.bill))")
302end

304function savings(data, result)
305 original = baseline(data, result.settings, result.prices)
306 println("SAVINGS strategy $(notation(result.strategy))")
307 entries = [(name, total(filter(r -> r.period == i, original.records), :cost),
308 total(filter(r -> r.period == i, result.records), :cost)) for (i, name) in enumerate(data.periods)]
309 append!(entries, [("EXPORT", -total(original.records, :income), -total(result.records, :income)),
310 ("DEMAND", original.demand, result.demand), ("TOTAL", original.bill, result.bill)])
311 for (name, before, after) in entries
312 println("$name $(money(before)) -> $(money(after)) saving $(money(before - after))")
313 end
314end

316function grouped_reports(command, data, result)
317 println("$command strategy $(notation(result.strategy))")
318 if command == "BYPERIOD"
319 for (i, name) in enumerate(data.periods)
320 r = filter(r -> r.period == i, result.records)
321 println("$name hours $(length(r)) load $(total(r, :load)) pv $(total(r, :pv)) charge $(total(r, :charge)) discharge $(total(r, :discharge)) import $(total(r, :imported)) export $(total(r, :exported))")
322 end
323 else
324 for (name, weekend) in (("WEEKDAY", false), ("WEEKEND", true))
325 days = count(==(weekend), data.weekends)
326 r = filter(r -> data.weekends[r.day] == weekend, result.records)
327 println("$name days $days load $(div(total(r, :load), days)) pv $(div(total(r, :pv), days)) import $(div(total(r, :imported), days)) export $(div(total(r, :exported), days)) cost $(money(div(netcost(r), days)))")
328 end
329 end

332function usage(result)
333 r = result.records
334 charging = count(x -> x.charge > 0, r)
335 discharging = count(x -> x.drawn > 0, r)
336 idle = count(x -> x.charge == 0 && x.drawn == 0, r)
337 full = count(x -> x.soc == result.settings["CAPACITY"], r)
338 low = count(x -> x.soc <= result.reserve, r)
339 current = longest = start = 0
340 for (i, record) in enumerate(r)
341 current = record.soc <= result.reserve ? current + 1 : 0
342 if current > longest
343 longest, start = current, i - current + 1
344 end
345 end
346 print("USAGE charging $charging discharging $discharging idle $idle full $full low $low longest $longest")
347 longest > 0 && print(" from day $(r[start].day) hour $(r[start].hour)")
348 println()
349end

351function report(command, data, result)
352 r = result.records
353 strategy = notation(result.strategy)
354 if command == "DAILY"
355 daily(data, result)
356 elseif command == "BILL"
357 bill(data, result)
358 elseif command == "SAVINGS"
359 savings(data, result)
360 elseif command == "BYPERIOD" || command == "DAYTYPE"
361 grouped_reports(command, data, result)
362 elseif command == "PROFILE"
363 days = length(data.daynames)
364 println("PROFILE strategy $strategy days $days")
365 println("hour load pv import export soc")
366 for hour in 0:23
367 records = @view r[hour + 1:24:end]
368 row((hour, (div(total(records, field), days) for field in (:load, :pv, :imported, :exported, :soc))...), (4, 7, 7, 7, 7, 7))
369 end
370 elseif command == "FLOWS"
371 println("FLOWS strategy $strategy")
372 for (i, name) in enumerate(FLOW_NAMES)
373 println("$name $(sum(record.flows[i] for record in r))")
374 end
375 println("LOSSES $(total(r, :loss))")
376 println("STORED $(last(r).soc - result.starting)")
377 elseif command == "SELFSUFF"
378 consumed = sum(record.flows[1] + record.flows[2] for record in r)
379 println("SELFSUFF sufficiency $(sufficiency(result))% consumption $(percentage(consumed, total(r, :pv)))%")
380 elseif command == "CYCLES"
381 drawn, capacity = total(r, :drawn), result.settings["CAPACITY"]
382 divisor = gcd(drawn, capacity)
383 println("CYCLES $(div(drawn, divisor))/$(div(capacity, divisor)) $(decimal(drawn, capacity, 3)) drawn $drawn losses $(total(r, :loss))")
384 elseif command == "USAGE"
385 usage(result)
386 elseif command == "HISTOGRAM"
387 println("HISTOGRAM strategy $strategy")
388 bands = zeros(Int, 10)
389 for record in r
390 bands[min(9, div(10 * record.soc, result.settings["CAPACITY"])) + 1] += 1
391 end
392 for i in 0:9
393 println(lpad(10 * i, 3), "% ", bar(div(20 * bands[i + 1], length(r))), " ", bands[i + 1])
394 end
395 end
396end

398function outage_report(data, result, day, first, last_hour)
399 alternate = simulate(data, result.strategy, result.settings, result.prices; outage=(day, first, last_hour))
400 begin_index = (day - 1) * 24 + first + 1
401 before = begin_index == 1 ? alternate.starting : alternate.records[begin_index - 1].soc
402 println("OUTAGE day $day hours $first-$last_hour strategy $(notation(result.strategy)) soc $before")
403 println("hour load pv disch soc short curt")
404 records = @view alternate.records[begin_index:(day - 1) * 24 + last_hour + 1]
405 for r in records
406 row((r.hour, r.load, r.pv, r.discharge, r.soc, r.shortfall, r.curtailed), (4, 7, 7, 7, 7, 7, 7))
407 end
408 load, shortfall = total(records, :load), total(records, :shortfall)
409 served = load == 0 ? "100.00" : percentage(load - shortfall, load)
410 println("TOTAL load $load unserved $shortfall served $served% bill $(money(alternate.bill))")
411end

413function difference(data, result, day, strategy)
414 alternate = simulate(data, strategy, result.settings, result.prices)
415 println("DIFF day $day $(data.daynames[day]) $(notation(result.strategy)) vs $(notation(strategy))")
416 println("hour imp1 imp2 exp1 exp2 soc1 soc2")
417 left, right = dayrecords(result, day), dayrecords(alternate, day)
418 for (a, b) in zip(left, right)
419 row((a.hour, a.imported, b.imported, a.exported, b.exported, a.soc, b.soc), (4, 7, 7, 7, 7, 7, 7))
420 end
421 cost1, cost2 = netcost(left), netcost(right)
422 println("TOTAL cost1 $(money(cost1)) cost2 $(money(cost2)) difference $(money(cost2 - cost1))")
423end

425function comparisons(command, args, data, settings, prices)
426 if command == "COMPARE"
427 choices = strategies(args)
428 isempty(choices) && fail("wrong number of arguments")
429 base = baseline(data, settings, prices).bill
430 results = [simulate(data, strategy, settings, prices) for strategy in choices]
431 println("COMPARE baseline $(money(base))")
432 order = sortperm(eachindex(results); by=i -> (results[i].bill, i))
433 for (rank, i) in enumerate(order)
434 result = results[i]
435 println("$rank. $(notation(result.strategy)) bill $(money(result.bill)) saving $(money(base - result.bill)) sufficiency $(sufficiency(result))%")
436 end
437 elseif command == "SIZE"
438 arguments(args, 3)
439 values = values_range(args, (1, 100000))
440 base = baseline(data, settings, prices).bill
441 best = nothing
442 for capacity in values
443 result = simulate(data, Strategy("SELF", 0), merge(settings, Dict("CAPACITY" => capacity)), prices)
444 saving = base - result.bill
445 cost = div(capacity * settings["COST"], 1000)
446 payback = saving > 0 ? decimal(cost, saving, 1) : "never"
447 println("SIZE $capacity bill $(money(result.bill)) saving $(money(saving)) cost $(money(cost)) payback $payback")
448 if saving > 0 && (best === nothing || cost * best[3] < best[2] * saving)
449 best = (capacity, cost, saving)
450 end
451 end
452 println("BEST ", best === nothing ? "none" : string(best[1]))
453 else
454 length(args) >= 5 || fail("wrong number of arguments")
455 item = args[1]
456 bounds = parameter_range(item)
457 values = values_range(args[2:4], bounds)
458 strategy = one_strategy(args[5:end])
459 best = nothing
460 for value in values
461 changed = merge(settings, Dict(String(item) => value))
462 result = simulate(data, strategy, changed, prices)
463 base = baseline(data, changed, prices).bill
464 println("SWEEP $item $value bill $(money(result.bill)) saving $(money(base - result.bill)) sufficiency $(sufficiency(result))%")
465 if best === nothing || result.bill < best[2]
466 best = (value, result.bill)
467 end
468 end
469 println("BEST $(best[1])")
470 end
471end

473function main()
474 data, settings, prices = read_data()
475 initial_settings, initial_prices = copy(settings), copy(prices)
476 result = nothing
477 for line in eachline()
478 fields = split(line)
479 command, args = fields[1], fields[2:end]
480 command == "END" && break
481 try
482 if command == "RUN"
483 strategy = one_strategy(args)
484 result = simulate(data, strategy, settings, prices)
485 println("RUN $(notation(strategy)) bill $(money(result.bill)) import $(total(result.records, :imported)) export $(total(result.records, :exported)) soc $(last(result.records).soc)")
486 elseif command == "SET"
487 arguments(args, 2)
488 item = args[1]
489 low, high = parameter_range(item)
490 value = number(args[2])
491 low <= value <= high || fail("invalid value for $item")
492 previous = settings[item]
493 settings[item] = value
494 println("SET $item $previous -> $value")
495 elseif command == "PRICE"
496 arguments(args, 2)
497 index = findfirst(==(args[1]), data.periods)
498 index !== nothing || fail("unknown period $(args[1])")
499 value = number(args[2])
500 0 <= value <= 10000 || fail("invalid value for $(args[1])")
501 previous = prices[index]
502 prices[index] = value
503 println("PRICE $(args[1]) $previous -> $value cheapest $(data.periods[argmin(prices)]) peak $(data.periods[argmax(prices)])")
504 elseif command == "RESET"
505 arguments(args, 0)
506 settings, prices = copy(initial_settings), copy(initial_prices)
507 println("RESET")
508 elseif command == "CONFIG"
509 arguments(args, 0)
510 configure(data, settings, prices)
511 elseif command == "NOBATTERY"
512 arguments(args, 0)
513 base = baseline(data, settings, prices)
514 r = base.records
515 println("NOBATTERY bill $(money(base.bill)) import $(total(r, :imported)) export $(total(r, :exported)) curtailed $(total(r, :curtailed)) peak $(base.peak)")
516 elseif command in ("COMPARE", "SIZE", "SWEEP")
517 comparisons(command, args, data, settings, prices)
518 elseif command == "OUTAGE"
519 arguments(args, 3)
520 day = valid_day(args[1], data)
521 first, last_hour = number.(args[2:3])
522 0 <= first <= last_hour <= 23 || fail("invalid hours")
523 result !== nothing || fail("no run")
524 outage_report(data, result, day, first, last_hour)
525 elseif command == "DIFF"
526 length(args) >= 2 || fail("wrong number of arguments")
527 day = valid_day(args[1], data)
528 strategy = one_strategy(args[2:end])
529 result !== nothing || fail("no run")
530 difference(data, result, day, strategy)
531 elseif command == "HOURLY" || command == "CHART"
532 arguments(args, 1)
533 day = valid_day(args[1], data)
534 result !== nothing || fail("no run")
535 if command == "HOURLY"
536 hourly(data, result, day)
537 else
538 println("CHART day $day $(data.daynames[day]) strategy $(notation(result.strategy))")
539 capacity = result.settings["CAPACITY"]
540 for r in dayrecords(result, day)
541 println(lpad(r.hour, 2), " ", bar(div(20 * r.soc, capacity)), " ", percentage(r.soc, capacity), "%")
542 end
543 end
544 elseif command == "PEAKS"
545 arguments(args, 1)
546 count = number(args[1])
547 count >= 1 || fail("invalid count")
548 result !== nothing || fail("no run")
549 println("PEAKS strategy $(notation(result.strategy))")
550 selected = filter(r -> r.imported > 0, result.records)
551 sort!(selected; by=r -> (-r.imported, r.day, r.hour))
552 for (rank, r) in enumerate(Iterators.take(selected, count))
553 println("$rank. day $(r.day) hour $(r.hour) $(data.periods[r.period]) import $(r.imported) load $(r.load) soc $(r.soc)")
554 end
555 isempty(selected) && println("none")
556 elseif command in REPORTS
557 arguments(args, 0)
558 result !== nothing || fail("no run")
559 report(command, data, result)
560 else
561 fail("unknown command $command")
562 end
563 catch error
564 error isa CommandError || rethrow()
565 println("ERROR ", error.message)
566 end
567 end
568end

570main()
