79def read_strategies(tokens):
80 strategies = []
81 index = 0
82 while index < len(tokens):
83 name = tokens[index]
84 index += 1
85 if name in ("SELF", "EXPORTPEAK"):
86 strategies.append((name, None))
87 continue
88 if name not in STRATEGY_RANGES:
89 raise CommandError(f"unknown strategy {name}")
90 if index == len(tokens):
91 raise CommandError(f"missing parameter for {name}")
92 parameter = int(tokens[index])
93 index += 1
94 low, high = STRATEGY_RANGES[name]
95 if not low <= parameter <= high:
96 raise CommandError(f"invalid parameter for {name}")
97 strategies.append((name, parameter))
98 return strategies

101def one_strategy(tokens):
102 strategies = read_strategies(tokens)
103 require_count(strategies, 1)
104 return strategies[0]

112def checked_range(low, high, step, bounds):
113 minimum, maximum = bounds
114 if low < minimum or high > maximum or high < low or step < 1:
115 raise CommandError("invalid range")
116 if (high - low) // step + 1 > 20:
117 raise CommandError("invalid range")
118 return range(low, high + 1, step)

121class InputData:
122 def __init__(self, source):
123 battery = list(map(int, source.readline().split()[1:]))
124 grid = list(map(int, source.readline().split()[1:]))
125 self.initial_settings = dict(zip(SETTING_RANGES, battery + grid + [100]))
126 period_count = int(source.readline().split()[1])
127 self.initial_prices = {}
128 for _ in range(period_count):
129 name, price = source.readline().split()
130 self.initial_prices[name] = int(price)
131 weekday = self.read_period_hours(source.readline().split()[1:])
132 weekend = self.read_period_hours(source.readline().split()[1:])
133 profile_count = int(source.readline().split()[1])
134 profiles = {}
135 for _ in range(profile_count):
136 kind, name, *values = source.readline().split()
137 profiles[kind, name] = tuple(map(int, values))
138 self.day_count = int(source.readline().split()[1])
139 loads = []
140 base_pv = []
141 for _ in range(self.day_count):
142 load_name, load_pct, pv_name, pv_pct = source.readline().split()
143 loads.extend(value * int(load_pct) // 100
144 for value in profiles["LOAD", load_name])
145 base_pv.extend(value * int(pv_pct) // 100
146 for value in profiles["PV", pv_name])
147 holidays = set(map(int, source.readline().split()[2:]))
148 override_count = int(source.readline().split()[1])
149 for _ in range(override_count):
150 day, hour, load, pv = map(int, source.readline().split())
151 index = (day - 1) * 24 + hour
152 loads[index] = load
153 base_pv[index] = pv
154 self.loads = tuple(loads)
155 self.base_pv = tuple(base_pv)
156 self.day_names = tuple(
157 "HOL" if day in holidays else DAY_NAMES[(day - 1) % 7]
158 for day in range(1, self.day_count + 1)
159 )
160 self.weekend_days = tuple(
161 day in holidays or (day - 1) % 7 >= 5
162 for day in range(1, self.day_count + 1)
163 )
164 self.periods = tuple(
165 period
166 for is_weekend in self.weekend_days
167 for period in (weekend if is_weekend else weekday)
168 )

170 @staticmethod
171 def read_period_hours(tokens):
172 hours = []
173 for index in range(0, len(tokens), 2):
174 start = int(tokens[index])
175 end = int(tokens[index + 2]) if index + 2 < len(tokens) else 24
176 hours.extend([tokens[index + 1]] * (end - start))
177 return tuple(hours)

180class HourRecord:
181 def __init__(self, index, period, load, pv, soc, flows, drawn,
182 shortfall, loss, price, feedin):
183 self.index = index
184 self.day = index // 24 + 1
185 self.hour = index % 24
186 self.period = period
187 self.load = load
188 self.pv = pv
189 self.soc = soc
190 self.flows = flows
191 self.charge = flows[1] + flows[7]
192 self.discharge = flows[4] + flows[5]
193 self.imported = flows[6] + flows[7]
194 self.exported = flows[2] + flows[5]
195 self.curtailed = flows[3]
196 self.drawn = drawn
197 self.shortfall = shortfall
198 self.loss = loss
199 self.import_cost = self.imported * price // 100000
200 self.export_income = self.exported * feedin // 100000
201 self.net_cost = self.import_cost - self.export_income

204class Totals:
205 def __init__(self, records):
206 self.count = len(records)
207 for field in TOTAL_FIELDS:
208 setattr(self, field, sum(getattr(record, field) for record in records))
209 self.flows = tuple(sum(record.flows[index] for record in records)
210 for index in range(len(FLOW_NAMES)))
211 self.net_cost = self.import_cost - self.export_income

214class SimulationResult:
215 def __init__(self, strategy, settings, prices, start_soc, reserve, records):
216 self.strategy = strategy
217 self.settings = settings.copy()
218 self.prices = prices.copy()
219 self.start_soc = start_soc
220 self.reserve = reserve
221 self.hours = tuple(records)
222 self.days = tuple(self.hours[start:start + 24]
223 for start in range(0, len(records), 24))
224 self.daily = tuple(Totals(day) for day in self.days)
225 self.by_period = {
226 period: Totals(tuple(record for record in records if record.period == period))
227 for period in prices
228 }
229 self.total = Totals(self.hours)
230 self.peak = max(record.imported for record in self.hours)
231 self.demand_charge = self.peak * settings["DEMAND"] // 1000
232 self.bill = self.total.net_cost + settings["FIXED"] + self.demand_charge
233 self.end_soc = self.hours[-1].soc
234 self.sufficiency = percentage(
235 max(0, self.total.load - self.total.imported), self.total.load
236 )

239class Simulator:
240 def __init__(self, data, strategy, settings, prices, outage=None):
241 self.data = data
242 self.strategy = strategy
243 self.name, self.parameter = strategy
244 self.settings = settings.copy()
245 self.prices = prices.copy()
246 self.outage = outage
247 self.capacity = settings["CAPACITY"]
248 self.ceff = settings["CEFF"]
249 self.deff = settings["DEFF"]
250 reserve_pct = settings["RESERVE"]
251 if self.name == "BACKUP":
252 reserve_pct = max(reserve_pct, self.parameter)
253 self.reserve = self.capacity * reserve_pct // 100
254 self.start_soc = self.capacity * settings["INITIAL"] // 100
255 self.soc = self.start_soc
256 self.cheapest, self.peak = tariff_extremes(prices)
257 self.arbitrage = (self.name == "EXPORTPEAK"
258 and settings["FEEDIN"] > prices[self.cheapest])
259
260 def discharge_target(self, deficit, period):
261 if self.name == "TOU":
262 return deficit if period == self.peak else 0
263 if self.name == "PEAKSHAVE":
264 return max(0, deficit - self.parameter)
265 if (self.arbitrage and period == self.cheapest
266 and period != self.peak):
267 return 0
268 return deficit
269
270 def grid_charge(self, period, pv_charge, grid_load):
271 headroom = self.settings["CHARGE"] - pv_charge
272 target = None
273 if self.name == "TOU" and period == self.cheapest and period != self.peak:
274 target = self.capacity * self.parameter // 100
275 elif self.name == "PEAKSHAVE" and period == self.cheapest:
276 target = self.capacity
277 headroom = min(headroom, max(0, self.parameter - grid_load))
278 elif self.name == "BACKUP" and period == self.cheapest:
279 target = self.reserve
280 elif self.arbitrage and period == self.cheapest and period != self.peak:
281 target = self.capacity
282 if target is None:
283 return 0
284 amount = min(headroom, ceiling(100 * max(0, target - self.soc), self.ceff))
285 self.soc += amount * self.ceff // 100
286 return amount

288 def run_hour(self, index):
289 settings = self.settings
290 load = self.data.loads[index]
291 pv = self.data.base_pv[index] * settings["PVSCALE"] // 100
292 period = self.data.periods[index]
293 direct = min(load, pv)
294 surplus = pv - direct
295 deficit = load - direct
296 pv_charge = min(surplus, settings["CHARGE"],
297 ceiling(100 * (self.capacity - self.soc), self.ceff))
298 self.soc += pv_charge * self.ceff // 100
299 surplus -= pv_charge
300 is_outage = (self.outage is not None
301 and self.outage[0] <= index <= self.outage[1])
302 target = deficit if is_outage else self.discharge_target(deficit, period)
303 lower_bound = 0 if is_outage else self.reserve
304 load_drawn = min(ceiling(100 * target, self.deff), settings["DISCHARGE"],
305 max(0, self.soc - lower_bound))
306 load_discharge = load_drawn * self.deff // 100
307 self.soc -= load_drawn
308 deficit -= load_discharge
309 if is_outage:
310 grid_load = 0
311 shortfall = deficit
312 pv_export = 0
313 curtailed = surplus
314 grid_charge = 0
315 export_drawn = 0
316 battery_export = 0
317 else:
318 grid_load = deficit
319 shortfall = 0
320 pv_export = min(surplus, settings["EXPORT"])
321 curtailed = surplus - pv_export
322 grid_charge = self.grid_charge(period, pv_charge, grid_load)
323 export_drawn = 0
324 battery_export = 0
325 if self.arbitrage and period == self.peak and pv_charge == 0:
326 export_drawn = min(
327 ceiling(100 * (settings["EXPORT"] - pv_export), self.deff),
328 settings["DISCHARGE"] - load_drawn,
329 max(0, self.soc - self.reserve),
330 )
331 battery_export = export_drawn * self.deff // 100
332 self.soc -= export_drawn
333 loss = (pv_charge - pv_charge * self.ceff // 100
334 + grid_charge - grid_charge * self.ceff // 100
335 + load_drawn - load_discharge + export_drawn - battery_export)
336 flows = (direct, pv_charge, pv_export, curtailed, load_discharge,
337 battery_export, grid_load, grid_charge)
338 return HourRecord(index, period, load, pv, self.soc, flows,
339 load_drawn + export_drawn, shortfall, loss,
340 self.prices[period], settings["FEEDIN"])

342 def run(self):
343 records = [self.run_hour(index) for index in range(len(self.data.loads))]
344 return SimulationResult(self.strategy, self.settings, self.prices,
345 self.start_soc, self.reserve, records)

350 self.data = data
351 self.settings = data.initial_settings.copy()
352 self.prices = data.initial_prices.copy()
353 self.current = None

357 "COMPARE": self.compare, "SIZE": self.size, "SWEEP": self.sweep,

366 def execute(self, tokens):
367 command, *arguments = tokens
368 if command not in self.handlers:
369 return [f"ERROR unknown command {command}"]
370 try:
371 return self.handlers[command](arguments)
372 except CommandError as error:
373 return [f"ERROR {error}"]

467 def compare(self, arguments):
468 strategies = read_strategies(arguments)
469 if not strategies:
470 raise CommandError("wrong number of arguments")
471 baseline = self.baseline(self.settings, self.prices)
472 trials = [self.simulate(strategy) for strategy in strategies]
473 trials.sort(key=lambda result: result.bill)
474 output = [f"COMPARE baseline {money(baseline.bill)}"]
475 for rank, result in enumerate(trials, 1):
476 output.append(f"{rank}. {strategy_text(result.strategy)} "
477 f"bill {money(result.bill)} saving {money(baseline.bill - result.bill)} "
478 f"sufficiency {result.sufficiency}%")
479 return output

481 def size(self, arguments):
482 require_count(arguments, 3)
483 capacities = checked_range(*map(int, arguments), SETTING_RANGES["CAPACITY"])
484 baseline = self.baseline(self.settings, self.prices)
485 output = []
486 best = None
487 for capacity in capacities:
488 settings = self.settings.copy()
489 settings["CAPACITY"] = capacity
490 result = self.simulate(("SELF", None), settings)
491 saving = baseline.bill - result.bill
492 cost = capacity * settings["COST"] // 1000
493 payback = "never"
494 if saving > 0:
495 tenths = 10 * cost // saving
496 payback = f"{tenths // 10}.{tenths % 10}"
497 if best is None or cost * best[2] < best[1] * saving:
498 best = (capacity, cost, saving)
499 output.append(f"SIZE {capacity} bill {money(result.bill)} saving {money(saving)} "
500 f"cost {money(cost)} payback {payback}")
501 output.append(f"BEST {best[0]}" if best is not None else "BEST none")
502 return output

504 def sweep(self, arguments):
505 if len(arguments) < 5:
506 raise CommandError("wrong number of arguments")
507 item = arguments[0]
508 if item not in SETTING_RANGES:
509 raise CommandError(f"unknown parameter {item}")
510 values = checked_range(*map(int, arguments[1:4]), SETTING_RANGES[item])
511 strategy = one_strategy(arguments[4:])
512 output = []
513 best = None
514 for value in values:
515 settings = self.settings.copy()
516 settings[item] = value
517 result = self.simulate(strategy, settings)
518 baseline = self.baseline(settings, self.prices)
519 if best is None or result.bill < best[1]:
520 best = (value, result.bill)
521 output.append(f"SWEEP {item} {value} bill {money(result.bill)} "
522 f"saving {money(baseline.bill - result.bill)} "
523 f"sufficiency {result.sufficiency}%")
524 output.append(f"BEST {best[0]}")
525 return output

527 def outage(self, arguments):
528 require_count(arguments, 3)
529 day = self.checked_day(arguments[0])
530 start, end = map(int, arguments[1:])
531 if not 0 <= start <= end <= 23:
532 raise CommandError("invalid hours")
533 current = self.require_result()
534 first = (day - 1) * 24 + start
535 last = (day - 1) * 24 + end
536 result = self.simulate(current.strategy, current.settings, current.prices,
537 (first, last))
538 before = result.hours[first - 1].soc if first else result.start_soc
539 output = [f"OUTAGE day {day} hours {start}-{end} "
540 f"strategy {strategy_text(result.strategy)} soc {before}",
541 "hour load pv disch soc short curt"]
542 records = result.hours[first:last + 1]
543 for record in records:
544 output.append(table_row(
545 (record.hour, record.load, record.pv, record.discharge,
546 record.soc, record.shortfall, record.curtailed), (4,) + (7,) * 6
547 ))
548 total = Totals(records)
549 served = percentage(total.load - total.shortfall, total.load) if total.load else "100.00"
550 output.append(f"TOTAL load {total.load} unserved {total.shortfall} "
551 f"served {served}% bill {money(result.bill)}")
552 return output

554 def diff(self, arguments):
555 if len(arguments) < 2:
556 raise CommandError("wrong number of arguments")
557 day = self.checked_day(arguments[0])
558 strategy = one_strategy(arguments[1:])
559 current = self.require_result()
560 comparison = self.simulate(strategy, current.settings, current.prices)
561 output = [f"DIFF day {day} {self.data.day_names[day - 1]} "
562 f"{strategy_text(current.strategy)} vs {strategy_text(strategy)}",
563 "hour imp1 imp2 exp1 exp2 soc1 soc2"]
564 for first, second in zip(current.days[day - 1], comparison.days[day - 1]):
565 output.append(table_row(
566 (first.hour, first.imported, second.imported, first.exported,
567 second.exported, first.soc, second.soc), (4,) + (7,) * 6
568 ))
569 cost1 = current.daily[day - 1].net_cost
570 cost2 = comparison.daily[day - 1].net_cost
571 output.append(f"TOTAL cost1 {money(cost1)} cost2 {money(cost2)} "
572 f"difference {money(cost2 - cost1)}")
573 return output

610 def bill(self, arguments):
611 result = self.result_without_arguments(arguments)
612 output = [f"BILL strategy {strategy_text(result.strategy)} days {self.data.day_count}"]
613 for period, total in result.by_period.items():
614 output.append(f"IMPORT {period} {total.imported} Wh {money(total.import_cost)}")
615 output.extend([
616 f"EXPORT {result.total.exported} Wh {money(-result.total.export_income)}",
617 f"FIXED {money(result.settings['FIXED'])}",
618 f"DEMAND {result.peak} Wh {money(result.demand_charge)}",

704 def cycles(self, arguments):
705 result = self.result_without_arguments(arguments)
706 drawn = result.total.drawn
707 capacity = result.settings["CAPACITY"]
708 common = gcd(drawn, capacity)
709 numerator, denominator = drawn // common, capacity // common
710 thousandths = 1000 * drawn // capacity
711 decimal = f"{thousandths // 1000}.{thousandths % 1000:03d}"
712 return [f"CYCLES {numerator}/{denominator} {decimal} "
713 f"drawn {drawn} losses {result.total.loss}"]

715 def usage(self, arguments):
716 result = self.result_without_arguments(arguments)
717 charging = sum(record.charge > 0 for record in result.hours)
718 discharging = sum(record.drawn > 0 for record in result.hours)
719 idle = sum(record.charge == 0 and record.drawn == 0 for record in result.hours)
720 full = sum(record.soc == result.settings["CAPACITY"] for record in result.hours)
721 low_runs = [list(records) for is_low, records in groupby(
722 result.hours, key=lambda record: record.soc <= result.reserve
723 ) if is_low]
724 low = sum(map(len, low_runs))
725 longest = max(low_runs, key=len, default=[])
726 line = (f"USAGE charging {charging} discharging {discharging} idle {idle} "
727 f"full {full} low {low} longest {len(longest)}")
728 if longest:
729 line += f" from day {longest[0].day} hour {longest[0].hour}"
730 return [line]
