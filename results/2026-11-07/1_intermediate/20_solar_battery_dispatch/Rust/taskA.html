13fn number(text: &str) -> i64 { text.parse().unwrap() }
14
15fn money(value: i64) -> String {
16 format!("{}{}.{:02}", if value < 0 { "-" } else { "" }, value.abs() / 100, value.abs() % 100)
17}
18
19fn percentage(numerator: i64, denominator: i64) -> String {
20 money(if denominator == 0 { 0 } else { 10000 * numerator / denominator })
21}
22
23fn ceil_div(numerator: i64, denominator: i64) -> i64 { (numerator + denominator - 1) / denominator }
24
25fn bar(width: i64) -> String { format!("|{}{}|", "#".repeat(width as usize), ".".repeat((20 - width) as usize)) }
26
27fn arity(args: &[&str], expected: usize) -> Result<(), String> {
28 if args.len() == expected { Ok(()) } else { Err("wrong number of arguments".into()) }
29}
30
31fn bounds(item: &str) -> Result<(i64, i64), String> {
32 PARAMETERS.iter().find(|(name, _, _)| *name == item).map(|(_, low, high)| (*low, *high))
33 .ok_or_else(|| format!("unknown parameter {item}"))
34}
35
36fn check_range(low: i64, high: i64, step: i64, minimum: i64, maximum: i64) -> Result<(), String> {
37 if low < minimum || high > maximum || high < low || step < 1 || (high - low) / step + 1 > 20 {
38 Err("invalid range".into())
39 } else { Ok(()) }
40}

74#[derive(Clone)]
75struct Tariff { names: Vec<String>, prices: Vec<i64> }
76
77impl Tariff {
78 fn extremes(&self) -> (usize, usize) {
79 let (mut cheapest, mut peak) = (0, 0);
80 for index in 1..self.prices.len() {
81 if self.prices[index] < self.prices[cheapest] { cheapest = index; }
82 if self.prices[index] > self.prices[peak] { peak = index; }
83 }
84 (cheapest, peak)
85 }
86}

88#[derive(Clone, Copy)]
89enum Strategy { SelfConsumption, Tou(i64), PeakShave(i64), Backup(i64), ExportPeak }
90
91impl Strategy {
92 fn text(self) -> String {
93 match self {
94 Self::SelfConsumption => "SELF".into(), Self::Tou(target) => format!("TOU {target}"),
95 Self::PeakShave(limit) => format!("PEAKSHAVE {limit}"), Self::Backup(reserve) => format!("BACKUP {reserve}"),
96 Self::ExportPeak => "EXPORTPEAK".into(),
97 }
98 }
99
100 fn parse_sequence(tokens: &[&str]) -> Result<Vec<Self>, String> {
101 let mut strategies = Vec::new();
102 let mut position = 0;
103 while position < tokens.len() {
104 let name = tokens[position];
105 position += 1;
106 let strategy = match name {
107 "SELF" => Self::SelfConsumption,
108 "EXPORTPEAK" => Self::ExportPeak,
109 "TOU" | "PEAKSHAVE" | "BACKUP" => {
110 if position == tokens.len() { return Err(format!("missing parameter for {name}")); }
111 let value = number(tokens[position]);
112 position += 1;
113 let (low, high) = match name { "TOU" => (1, 100), "BACKUP" => (0, 100), _ => (0, 100000) };
114 if value < low || value > high { return Err(format!("invalid parameter for {name}")); }
115 match name { "TOU" => Self::Tou(value), "BACKUP" => Self::Backup(value), _ => Self::PeakShave(value) }
116 }
117 _ => return Err(format!("unknown strategy {name}")),
118 };
119 strategies.push(strategy);
120 }
121 Ok(strategies)
122 }
123
124 fn parse_one(tokens: &[&str]) -> Result<Self, String> {
125 let strategies = Self::parse_sequence(tokens)?;
126 if strategies.len() != 1 { Err("wrong number of arguments".into()) } else { Ok(strategies[0]) }
127 }
128}

176impl Totals {
177 fn of<'a>(records: impl IntoIterator<Item = &'a Record>) -> Self {
178 let mut total = Self::default();
179 for record in records {
180 total.hours += 1;
181 total.load += record.load;
182 total.pv += record.pv;
183 total.charge += record.charge;
184 total.discharge += record.discharge;
185 total.drawn += record.drawn;
186 total.soc += record.soc;
187 total.import += record.import;
188 total.export += record.export;
189 total.curtailed += record.curtailed;
190 total.shortfall += record.shortfall;
191 total.loss += record.loss;
192 total.import_cost += record.import_cost;
193 total.income += record.income;
194 for (value, flow) in total.flows.iter_mut().zip(record.flows) { *value += flow; }
195 }
196 total
197 }

199 fn net_cost(&self) -> i64 { self.import_cost - self.income }
200
201 fn sufficiency(&self) -> String { percentage((self.load - self.import).max(0), self.load) }

204struct Simulation {
205 strategy: Strategy,
206 settings: Settings,
207 tariff: Tariff,
208 records: Vec<Record>,
209 starting: i64,
210 reserve: i64,
211 totals: Totals,
212 peak: i64,
213 demand_charge: i64,
214 bill: i64,
215}
216
217impl Simulation {
218 fn run(data: &Data, settings: Settings, tariff: &Tariff, strategy: Strategy, outage: Option<(usize, usize, usize)>) -> Self {
219 let (cheapest, peak_period) = tariff.extremes();
220 let reserve_pct = match strategy { Strategy::Backup(value) => settings.reserve.max(value), _ => settings.reserve };
221 let reserve = settings.capacity * reserve_pct / 100;
222 let starting = settings.capacity * settings.initial / 100;
223 let mut soc = starting;
224 let arbitrage = matches!(strategy, Strategy::ExportPeak) && settings.feedin > tariff.prices[cheapest];
225 let mut records = Vec::new();
226 for index in 0..data.load.len() {
227 let (load, pv, period) = (data.load[index], data.pv[index] * settings.pvscale / 100, data.periods[index]);
228 let interrupted = outage.is_some_and(|(day, start, end)| index / 24 + 1 == day && (start..=end).contains(&(index % 24)));
229 let direct = load.min(pv);
230 let (mut surplus, mut deficit) = (pv - direct, load - direct);
231 let pv_charge = surplus.min(settings.charge).min(ceil_div(100 * (settings.capacity - soc), settings.ceff));
232 let pv_stored = pv_charge * settings.ceff / 100;
233 soc += pv_stored;
234 surplus -= pv_charge;
235 let target = if interrupted { deficit } else {
236 match strategy {
237 Strategy::SelfConsumption | Strategy::Backup(_) => deficit,
238 Strategy::Tou(_) => if period == peak_period { deficit } else { 0 },
239 Strategy::PeakShave(limit) => (deficit - limit).max(0),
240 Strategy::ExportPeak => if arbitrage && period == cheapest && period != peak_period { 0 } else { deficit },
241 }
242 };
243 let lower = if interrupted { 0 } else { reserve };
244 let load_drawn = ceil_div(100 * target, settings.deff).min(settings.discharge).min((soc - lower).max(0));
245 let battery_load = load_drawn * settings.deff / 100;
246 soc -= load_drawn;
247 deficit -= battery_load;
248 let (grid_load, pv_export, curtailed, shortfall, grid_charge, export_drawn, battery_export);
249 if interrupted {
250 grid_load = 0;
251 pv_export = 0;
252 curtailed = surplus;
253 shortfall = deficit;
254 grid_charge = 0;
255 export_drawn = 0;
256 battery_export = 0;
257 } else {
258 grid_load = deficit;
259 pv_export = surplus.min(settings.export);
260 curtailed = surplus - pv_export;
261 shortfall = 0;
262 let mut headroom = settings.charge - pv_charge;
263 let charge_target = match strategy {
264 Strategy::Tou(target) if period == cheapest && period != peak_period => Some(settings.capacity * target / 100),
265 Strategy::PeakShave(limit) if period == cheapest => {
266 headroom = headroom.min((limit - grid_load).max(0));
267 Some(settings.capacity)
268 }
269 Strategy::Backup(_) if period == cheapest => Some(reserve),
270 Strategy::ExportPeak if arbitrage && period == cheapest && period != peak_period => Some(settings.capacity),
271 _ => None,
272 };
273 grid_charge = charge_target.map(|target| headroom.min(ceil_div(100 * (target - soc).max(0), settings.ceff))).unwrap_or(0);
274 soc += grid_charge * settings.ceff / 100;
275 export_drawn = if arbitrage && period == peak_period && pv_charge == 0 {
276 ceil_div(100 * (settings.export - pv_export), settings.deff)
277 .min(settings.discharge - load_drawn).min((soc - reserve).max(0))
278 } else { 0 };
279 battery_export = export_drawn * settings.deff / 100;
280 soc -= export_drawn;
281 }
282 let import = grid_load + grid_charge;
283 let export = pv_export + battery_export;
284 records.push(Record {
285 period, load, pv, charge: pv_charge + grid_charge, discharge: battery_load + battery_export,
286 drawn: load_drawn + export_drawn, soc, import, export, curtailed, shortfall,
287 loss: pv_charge - pv_stored + grid_charge - grid_charge * settings.ceff / 100
288 + load_drawn - battery_load + export_drawn - battery_export,
289 flows: [direct, pv_charge, pv_export, curtailed, battery_load, battery_export, grid_load, grid_charge],
290 import_cost: import * tariff.prices[period] / 100000, income: export * settings.feedin / 100000,
291 });
292 }
293 let totals = Totals::of(&records);
294 let peak = records.iter().map(|record| record.import).max().unwrap();
295 let demand_charge = peak * settings.demand / 1000;
296 let bill = totals.net_cost() + settings.fixed + demand_charge;
297 Self { strategy, settings, tariff: tariff.clone(), records, starting, reserve, totals, peak, demand_charge, bill }
298 }

454 fn usage(&self, result: &Simulation) {
455 let (mut charging, mut discharging, mut idle, mut full, mut low) = (0, 0, 0, 0, 0);
456 let (mut run, mut longest, mut start) = (0, 0, 0);
457 for (index, record) in result.records.iter().enumerate() {
458 if record.charge > 0 { charging += 1; }
459 if record.drawn > 0 { discharging += 1; }
460 if record.charge == 0 && record.drawn == 0 { idle += 1; }
461 if record.soc == result.settings.capacity { full += 1; }
462 if record.soc <= result.reserve {
463 low += 1;
464 run += 1;
465 if run > longest { longest = run; start = index + 1 - run; }
466 } else { run = 0; }
467 }
468 println!("USAGE charging {charging} discharging {discharging} idle {idle} full {full} low {low} longest {longest}{}",
469 if longest == 0 { String::new() } else { format!(" from day {} hour {}", start / 24 + 1, start % 24) });
470 }

517 fn command(&mut self, tokens: &[&str]) -> Result<(), String> {
518 let a = &tokens[1..];
519 match tokens[0] {
520 "RUN" => {
521 let strategy = Strategy::parse_one(a)?;
522 let result = Simulation::run(&self.data, self.settings, &self.tariff, strategy, None);
523 println!("RUN {} bill {} import {} export {} soc {}", strategy.text(), money(result.bill),
524 result.totals.import, result.totals.export, result.last_soc());
525 self.current = Some(result);
526 }
527 "SET" => {
528 arity(a, 2)?;
529 let (low, high) = bounds(a[0])?;
530 let value = number(a[1]);
531 if value < low || value > high { return Err(format!("invalid value for {}", a[0])); }
532 let old = self.settings.set(a[0], value);
533 println!("SET {} {old} -> {value}", a[0]);
534 }
535 "PRICE" => {
536 arity(a, 2)?;
537 let index = self.tariff.names.iter().position(|name| name == a[0]).ok_or_else(|| format!("unknown period {}", a[0]))?;
538 let price = number(a[1]);
539 if !(0..=10000).contains(&price) { return Err(format!("invalid value for {}", a[0])); }
540 let old = std::mem::replace(&mut self.tariff.prices[index], price);
541 let (cheapest, peak) = self.tariff.extremes();
542 println!("PRICE {} {old} -> {price} cheapest {} peak {}", a[0], self.tariff.names[cheapest], self.tariff.names[peak]);
543 }
544 "RESET" => {
545 arity(a, 0)?;
546 self.settings = self.initial_settings;
547 self.tariff = self.initial_tariff.clone();
548 println!("RESET");
549 }
550 "CONFIG" => { arity(a, 0)?; self.config(); }
551 "NOBATTERY" => {
552 arity(a, 0)?;
553 let result = Simulation::baseline(&self.data, self.settings, &self.tariff);
554 println!("NOBATTERY bill {} import {} export {} curtailed {} peak {}", money(result.bill), result.totals.import,
555 result.totals.export, result.totals.curtailed, result.peak);
556 }
557 "COMPARE" => {
558 let strategies = Strategy::parse_sequence(a)?;
559 if strategies.is_empty() { return Err("wrong number of arguments".into()); }
560 let baseline = Simulation::baseline(&self.data, self.settings, &self.tariff).bill;
561 let mut outcomes = Vec::new();
562 for (index, strategy) in strategies.iter().enumerate() {
563 let result = Simulation::run(&self.data, self.settings, &self.tariff, *strategy, None);
564 outcomes.push((result.bill, index, result.totals.sufficiency()));
565 }
566 outcomes.sort_by_key(|(bill, index, _)| (*bill, *index));
567 println!("COMPARE baseline {}", money(baseline));
568 for (rank, (bill, index, sufficiency)) in outcomes.into_iter().enumerate() {
569 println!("{}. {} bill {} saving {} sufficiency {sufficiency}%", rank + 1, strategies[index].text(), money(bill), money(baseline - bill));
570 }
571 }
572 "SIZE" => {
573 arity(a, 3)?;
574 let (low, high, step) = (number(a[0]), number(a[1]), number(a[2]));
575 check_range(low, high, step, 1, 100000)?;
576 let baseline = Simulation::baseline(&self.data, self.settings, &self.tariff).bill;
577 let mut best = None;
578 let mut capacity = low;
579 while capacity <= high {
580 let settings = Settings { capacity, ..self.settings };
581 let result = Simulation::run(&self.data, settings, &self.tariff, Strategy::SelfConsumption, None);
582 let saving = baseline - result.bill;
583 let cost = capacity * settings.cost / 1000;
584 let payback = if saving > 0 {
585 let tenths = 10 * cost / saving;
586 if best.is_none_or(|(old_cost, old_saving, _)| cost * old_saving < old_cost * saving) {
587 best = Some((cost, saving, capacity));
588 }
589 format!("{}.{:01}", tenths / 10, tenths % 10)
590 } else { "never".into() };
591 println!("SIZE {capacity} bill {} saving {} cost {} payback {payback}", money(result.bill), money(saving), money(cost));
592 capacity += step;
593 }
594 println!("BEST {}", best.map(|(_, _, capacity)| capacity.to_string()).unwrap_or_else(|| "none".into()));
595 }
596 "SWEEP" => {
597 if a.len() < 5 { return Err("wrong number of arguments".into()); }
598 let (minimum, maximum) = bounds(a[0])?;
599 let (low, high, step) = (number(a[1]), number(a[2]), number(a[3]));
600 check_range(low, high, step, minimum, maximum)?;
601 let strategy = Strategy::parse_one(&a[4..])?;
602 let mut best = None;
603 let mut value = low;
604 while value <= high {
605 let mut settings = self.settings;
606 settings.set(a[0], value);
607 let result = Simulation::run(&self.data, settings, &self.tariff, strategy, None);
608 let baseline = Simulation::baseline(&self.data, settings, &self.tariff).bill;
609 println!("SWEEP {} {value} bill {} saving {} sufficiency {}%", a[0], money(result.bill), money(baseline - result.bill), result.totals.sufficiency());
610 if best.is_none_or(|(bill, _)| result.bill < bill) { best = Some((result.bill, value)); }
611 value += step;
612 }
613 println!("BEST {}", best.unwrap().1);
614 }
615 "OUTAGE" => {
616 arity(a, 3)?;
617 let day = self.check_day(a[0])?;
618 let (start, end) = (number(a[1]), number(a[2]));
619 if start < 0 || start > end || end > 23 { return Err("invalid hours".into()); }
620 self.outage(self.result()?, day, start as usize, end as usize);
621 }
622 "DIFF" => {
623 if a.len() < 2 { return Err("wrong number of arguments".into()); }
624 let day = self.check_day(a[0])?;
625 let strategy = Strategy::parse_one(&a[1..])?;
626 self.diff(self.result()?, day, strategy);
627 }
628 "HOURLY" | "CHART" => {
629 arity(a, 1)?;
630 let day = self.check_day(a[0])?;
631 let result = self.result()?;
632 if tokens[0] == "HOURLY" { self.hourly(result, day); } else { self.chart(result, day); }
633 }
634 "PEAKS" => {
635 arity(a, 1)?;
636 let count = number(a[0]);
637 if count < 1 { return Err("invalid count".into()); }
638 self.peaks(self.result()?, count as usize);
639 }
640 "DAILY" | "BILL" | "SAVINGS" | "BYPERIOD" | "DAYTYPE" | "PROFILE" | "FLOWS" | "SELFSUFF" | "CYCLES" | "USAGE" | "HISTOGRAM" => {
641 arity(a, 0)?;
642 let result = self.result()?;
643 match tokens[0] {
644 "DAILY" => self.daily(result), "BILL" => self.bill(result), "SAVINGS" => self.savings(result),
645 "BYPERIOD" => self.by_period(result), "DAYTYPE" => self.day_type(result), "PROFILE" => self.profile(result),
646 "FLOWS" => self.flows(result), "CYCLES" => self.cycles(result), "USAGE" => self.usage(result),
647 "HISTOGRAM" => self.histogram(result),
648 "SELFSUFF" => println!("SELFSUFF sufficiency {}% consumption {}%", result.totals.sufficiency(),
649 percentage(result.totals.flows[0] + result.totals.flows[1], result.totals.pv)),
650 _ => unreachable!(),
651 }
652 }
653 _ => return Err(format!("unknown command {}", tokens[0])),
654 }
655 Ok(())
656 }
657}

659fn tokens<'a>(lines: &mut std::str::Lines<'a>) -> Vec<&'a str> {
660 lines.next().unwrap().split_ascii_whitespace().collect()
661}
662
663fn main() {
664 let mut input = String::new();
665 io::stdin().read_to_string(&mut input).unwrap();
666 let mut lines = input.lines();
667 let battery = tokens(&mut lines);
668 let grid = tokens(&mut lines);
669 let settings = Settings {
670 capacity: number(battery[1]), initial: number(battery[2]), charge: number(battery[3]), discharge: number(battery[4]),
671 ceff: number(battery[5]), deff: number(battery[6]), reserve: number(battery[7]), cost: number(battery[8]),
672 export: number(grid[1]), feedin: number(grid[2]), fixed: number(grid[3]), demand: number(grid[4]), pvscale: 100,
673 };
674 let count = number(tokens(&mut lines)[1]) as usize;
675 let mut tariff = Tariff { names: Vec::new(), prices: Vec::new() };
676 for _ in 0..count {
677 let parts = tokens(&mut lines);
678 tariff.names.push(parts[0].into());
679 tariff.prices.push(number(parts[1]));
680 }
681 let mut divisions = [[0; 24]; 2];
682 for division in &mut divisions {
683 let parts = tokens(&mut lines);
684 for position in (1..parts.len()).step_by(2) {
685 let start = number(parts[position]) as usize;
686 let end = if position + 2 < parts.len() { number(parts[position + 2]) as usize } else { 24 };
687 let period = tariff.names.iter().position(|name| name == parts[position + 1]).unwrap();
688 division[start..end].fill(period);
689 }
690 }
691 let count = number(tokens(&mut lines)[1]) as usize;
692 let mut loads = BTreeMap::new();
693 let mut generation = BTreeMap::new();
694 for _ in 0..count {
695 let parts = tokens(&mut lines);
696 let values: Vec<i64> = parts[2..].iter().map(|text| number(text)).collect();
697 let profiles = if parts[0] == "LOAD" { &mut loads } else { &mut generation };
698 profiles.insert(parts[1].to_string(), values);
699 }
700 let days = number(tokens(&mut lines)[1]) as usize;
701 let mut load = Vec::new();
702 let mut pv = Vec::new();
703 for _ in 0..days {
704 let parts = tokens(&mut lines);
705 let (load_pct, pv_pct) = (number(parts[1]), number(parts[3]));
706 for hour in 0..24 {
707 load.push(loads[parts[0]][hour] * load_pct / 100);
708 pv.push(generation[parts[2]][hour] * pv_pct / 100);
709 }
710 }
711 let parts = tokens(&mut lines);
712 let mut holidays = vec![false; days];
713 for day in &parts[2..] { holidays[number(day) as usize - 1] = true; }
714 let mut data = Data { days, load, pv, periods: Vec::new(), weekend: Vec::new(), day_names: Vec::new() };
715 for (day, holiday) in holidays.iter().enumerate() {
716 let weekend = *holiday || day % 7 >= 5;
717 data.weekend.push(weekend);
718 data.day_names.push(if *holiday { "HOL" } else { DAY_NAMES[day % 7] });
719 data.periods.extend_from_slice(&divisions[usize::from(weekend)]);
720 }
721 let count = number(tokens(&mut lines)[1]) as usize;
722 for _ in 0..count {
723 let parts = tokens(&mut lines);
724 let index = (number(parts[0]) as usize - 1) * 24 + number(parts[1]) as usize;
725 data.load[index] = number(parts[2]);
726 data.pv[index] = number(parts[3]);
727 }
728 let mut app = App { data, settings, initial_settings: settings, initial_tariff: tariff.clone(), tariff, current: None };
729 for line in lines {
730 let parts: Vec<&str> = line.split_ascii_whitespace().collect();
731 if parts[0] == "END" { break; }
732 if let Err(error) = app.command(&parts) { println!("ERROR {error}"); }
733 }
734}
