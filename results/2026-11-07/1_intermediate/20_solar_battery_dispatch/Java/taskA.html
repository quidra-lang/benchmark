88 private static String[] tariffExtremes(Map<String, Long> prices) {
89 String cheapest = prices.keySet().iterator().next();
90 String peak = cheapest;
91 for (Map.Entry<String, Long> item : prices.entrySet()) {
92 if (item.getValue() < prices.get(cheapest)) cheapest = item.getKey();
93 if (item.getValue() > prices.get(peak)) peak = item.getKey();
94 }
95 return new String[]{cheapest, peak};
96 }

98 private record Strategy(String name, Long parameter) {
99 String text() { return parameter == null ? name : name + " " + parameter; }
100 }
101 private static List<Strategy> strategies(List<String> tokens) {
102 List<Strategy> result = new ArrayList<>();
103 for (int i = 0; i < tokens.size();) {
104 String name = tokens.get(i++);
105 if (name.equals("SELF") || name.equals("EXPORTPEAK")) {
106 result.add(new Strategy(name, null));
107 continue;
108 }
109 long[] limits = STRATEGY_RANGES.get(name);
110 if (limits == null) throw new CommandError("unknown strategy " + name);
111 if (i == tokens.size()) throw new CommandError("missing parameter for " + name);
112 long parameter = number(tokens.get(i++));
113 if (parameter < limits[0] || parameter > limits[1])
114 throw new CommandError("invalid parameter for " + name);
115 result.add(new Strategy(name, parameter));
116 }
117 return result;
118 }
119 private static Strategy oneStrategy(List<String> args) {
120 List<Strategy> parsed = strategies(args);
121 if (parsed.size() != 1) throw new CommandError("wrong number of arguments");
122 return parsed.get(0);
123 }
124 private static List<Long> checkedRange(long low, long high, long step, long[] limits) {
125 if (low < limits[0] || high > limits[1] || high < low || step < 1 ||
126 (high - low) / step + 1 > 20) throw new CommandError("invalid range");
127 List<Long> result = new ArrayList<>();
128 for (long value = low; value <= high; value += step) result.add(value);
129 return result;
130 }

143 private static List<String> periodHours(String[] tokens) {
144 List<String> result = new ArrayList<>();
145 for (int i = 1; i < tokens.length; i += 2) {
146 int start = (int) number(tokens[i]);
147 int end = i + 2 < tokens.length ? (int) number(tokens[i + 2]) : 24;
148 for (int hour = start; hour < end; ++hour) result.add(tokens[i + 1]);
149 }
150 return result;

153 InputData(BufferedReader reader) throws IOException {
154 String[] battery = words(reader.readLine());
155 String[] grid = words(reader.readLine());
156 List<Long> settings = new ArrayList<>();
157 for (int i = 1; i < battery.length; ++i) settings.add(number(battery[i]));
158 for (int i = 1; i < grid.length; ++i) settings.add(number(grid[i]));
159 settings.add(100L);
160 for (int i = 0; i < SETTING_NAMES.length; ++i)
161 initialSettings.put(SETTING_NAMES[i], settings.get(i));
162 int periodCount = (int) number(words(reader.readLine())[1]);
163 for (int i = 0; i < periodCount; ++i) {
164 String[] line = words(reader.readLine());
165 initialPrices.put(line[0], number(line[1]));
166 }
167 List<String> weekday = periodHours(words(reader.readLine()));
168 List<String> weekend = periodHours(words(reader.readLine()));
169 int profileCount = (int) number(words(reader.readLine())[1]);
170 Map<String, long[]> profiles = new HashMap<>();
171 for (int i = 0; i < profileCount; ++i) {
172 String[] line = words(reader.readLine());
173 long[] profile = new long[line.length - 2];
174 for (int j = 2; j < line.length; ++j) profile[j - 2] = number(line[j]);
175 profiles.put(line[0] + "/" + line[1], profile);
176 }
177 dayCount = (int) number(words(reader.readLine())[1]);
178 for (int day = 0; day < dayCount; ++day) {
179 String[] line = words(reader.readLine());
180 long[] load = profiles.get("LOAD/" + line[0]);
181 long[] pv = profiles.get("PV/" + line[2]);
182 for (long v : load) loads.add(v * number(line[1]) / 100);
183 for (long v : pv) basePV.add(v * number(line[3]) / 100);
184 }
185 String[] holidayLine = words(reader.readLine());
186 Set<Integer> holidays = new HashSet<>();
187 for (int i = 2; i < holidayLine.length; ++i)
188 holidays.add((int) number(holidayLine[i]));
189 int overrideCount = (int) number(words(reader.readLine())[1]);
190 for (int i = 0; i < overrideCount; ++i) {
191 String[] line = words(reader.readLine());
192 int index = (int) ((number(line[0]) - 1) * 24 + number(line[1]));
193 loads.set(index, number(line[2]));
194 basePV.set(index, number(line[3]));
195 }
196 for (int day = 1; day <= dayCount; ++day) {
197 boolean holiday = holidays.contains(day);
198 boolean weekendDay = holiday || (day - 1) % 7 >= 5;
199 weekendDays.add(weekendDay);
200 dayNames.add(holiday ? "HOL" : DAY_NAMES[(day - 1) % 7]);
201 periods.addAll(weekendDay ? weekend : weekday);
202 }
203 for (String line; (line = reader.readLine()) != null;)
204 if (!line.isEmpty()) commands.add(line);
205 }
206 }

208 private static final class HourRecord {
209 final int index, day, hour;
210 final String period;
211 final long load, pv, soc;
212 final long[] flows;
213 final long charge, discharge, imported, exported, curtailed;
214 final long drawn, shortfall, loss, importCost, exportIncome, netCost;
215
216 HourRecord(int index, String period, long load, long pv, long soc, long[] flows,
217 long drawn, long shortfall, long loss, long price, long feedin) {
218 this.index = index;
219 this.day = index / 24 + 1;
220 this.hour = index % 24;
221 this.period = period;
222 this.load = load;
223 this.pv = pv;
224 this.soc = soc;
225 this.flows = flows;
226 this.charge = flows[1] + flows[7];
227 this.discharge = flows[4] + flows[5];
228 this.imported = flows[6] + flows[7];
229 this.exported = flows[2] + flows[5];
230 this.curtailed = flows[3];
231 this.drawn = drawn;
232 this.shortfall = shortfall;
233 this.loss = loss;
234 this.importCost = imported * price / 100000;
235 this.exportIncome = exported * feedin / 100000;
236 this.netCost = importCost - exportIncome;
237 }
238 }

240 private static final class Totals {
241 long count, load, pv, charge, discharge, imported, exported, curtailed;
242 long drawn, loss, shortfall, importCost, exportIncome, netCost;
243 final long[] flows = new long[8];
244
245 void add(HourRecord row) {
246 ++count;
247 load += row.load; pv += row.pv;
248 charge += row.charge; discharge += row.discharge;
249 imported += row.imported; exported += row.exported;
250 curtailed += row.curtailed; drawn += row.drawn;
251 loss += row.loss; shortfall += row.shortfall;
252 importCost += row.importCost; exportIncome += row.exportIncome;
253 netCost += row.netCost;
254 for (int i = 0; i < flows.length; ++i) flows[i] += row.flows[i];
255 }

258 private static final class SimulationResult {
259 final Strategy strategy;
260 final Map<String, Long> settings;
261 final LinkedHashMap<String, Long> prices;
262 final long startSOC, reserve, endSOC, peak, demandCharge, bill;
263 final String sufficiency;
264 final List<HourRecord> hours;
265 final List<Totals> daily = new ArrayList<>();
266 final LinkedHashMap<String, Totals> byPeriod = new LinkedHashMap<>();
267 final Totals total = new Totals();
268
269 SimulationResult(Strategy strategy, Map<String, Long> settings,
270 LinkedHashMap<String, Long> prices, long startSOC,
271 long reserve, List<HourRecord> hours) {
272 this.strategy = strategy;
273 this.settings = new HashMap<>(settings);
274 this.prices = new LinkedHashMap<>(prices);
275 this.startSOC = startSOC;
276 this.reserve = reserve;
277 this.hours = hours;
278 for (String name : prices.keySet()) byPeriod.put(name, new Totals());
279 long maximum = 0;
280 for (int i = 0; i < hours.size(); ++i) {
281 if (i % 24 == 0) daily.add(new Totals());
282 HourRecord row = hours.get(i);
283 daily.get(daily.size() - 1).add(row);
284 byPeriod.get(row.period).add(row);
285 total.add(row);
286 maximum = Math.max(maximum, row.imported);
287 }
288 peak = maximum;
289 endSOC = hours.get(hours.size() - 1).soc;
290 demandCharge = peak * value(settings, "DEMAND") / 1000;
291 bill = total.netCost + value(settings, "FIXED") + demandCharge;
292 sufficiency = percentage(Math.max(0, total.load - total.imported), total.load);
293 }
294 }

296 private static final class Simulator {
297 final InputData data;
298 final Strategy strategy;
299 final Map<String, Long> settings;
300 final LinkedHashMap<String, Long> prices;
301 final int[] outage;
302 final long capacity, ceff, deff, reserve, startSOC;
303 final String cheapest, peak;
304 final boolean arbitrage;
305 long soc;
306
307 Simulator(InputData data, Strategy strategy, Map<String, Long> settings,
308 LinkedHashMap<String, Long> prices, int[] outage) {
309 this.data = data;
310 this.strategy = strategy;
311 this.settings = new HashMap<>(settings);
312 this.prices = new LinkedHashMap<>(prices);
313 this.outage = outage;
314 capacity = value(settings, "CAPACITY");
315 ceff = value(settings, "CEFF");
316 deff = value(settings, "DEFF");
317 long reservePercent = value(settings, "RESERVE");
318 if (strategy.name.equals("BACKUP"))
319 reservePercent = Math.max(reservePercent, strategy.parameter);
320 reserve = capacity * reservePercent / 100;
321 startSOC = capacity * value(settings, "INITIAL") / 100;
322 soc = startSOC;
323 String[] extremes = tariffExtremes(prices);
324 cheapest = extremes[0];
325 peak = extremes[1];
326 arbitrage = strategy.name.equals("EXPORTPEAK") &&
327 value(settings, "FEEDIN") > prices.get(cheapest);
328 }

330 private long dischargeTarget(long deficit, String period) {
331 if (strategy.name.equals("TOU")) return period.equals(peak) ? deficit : 0;
332 if (strategy.name.equals("PEAKSHAVE"))
333 return Math.max(0, deficit - strategy.parameter);
334 if (arbitrage && period.equals(cheapest) && !period.equals(peak)) return 0;
335 return deficit;
336 }
337
338 private long gridCharge(String period, long pvCharge, long gridLoad) {
339 long headroom = value(settings, "CHARGE") - pvCharge;
340 Long target = null;
341 if (strategy.name.equals("TOU") && period.equals(cheapest) && !period.equals(peak)) {
342 target = capacity * strategy.parameter / 100;
343 } else if (strategy.name.equals("PEAKSHAVE") && period.equals(cheapest)) {
344 target = capacity;
345 headroom = Math.min(headroom, Math.max(0, strategy.parameter - gridLoad));
346 } else if (strategy.name.equals("BACKUP") && period.equals(cheapest)) {
347 target = reserve;
348 } else if (arbitrage && period.equals(cheapest) && !period.equals(peak)) {
349 target = capacity;
350 }
351 if (target == null) return 0;
352 long amount = Math.min(headroom, ceiling(100 * Math.max(0, target - soc), ceff));
353 soc += amount * ceff / 100;
354 return amount;
355 }
356
357 private HourRecord runHour(int index) {
358 long load = data.loads.get(index);
359 long pv = data.basePV.get(index) * value(settings, "PVSCALE") / 100;
360 String period = data.periods.get(index);
361 long direct = Math.min(load, pv);
362 long surplus = pv - direct;
363 long deficit = load - direct;
364 long pvCharge = Math.min(Math.min(surplus, value(settings, "CHARGE")),
365 ceiling(100 * (capacity - soc), ceff));
366 soc += pvCharge * ceff / 100;
367 surplus -= pvCharge;
368 boolean isOutage = outage != null && outage[0] <= index && index <= outage[1];
369 long target = isOutage ? deficit : dischargeTarget(deficit, period);
370 long lowerBound = isOutage ? 0 : reserve;
371 long loadDrawn = Math.min(Math.min(ceiling(100 * target, deff),
372 value(settings, "DISCHARGE")), Math.max(0, soc - lowerBound));
373 long loadDischarge = loadDrawn * deff / 100;
374 soc -= loadDrawn;
375 deficit -= loadDischarge;
376 long gridLoad = 0, shortfall = 0, pvExport = 0, curtailed = 0;
377 long gridCharge = 0, exportDrawn = 0, batteryExport = 0;
378 if (isOutage) {
379 shortfall = deficit;
380 curtailed = surplus;
381 } else {
382 gridLoad = deficit;
383 pvExport = Math.min(surplus, value(settings, "EXPORT"));
384 curtailed = surplus - pvExport;
385 gridCharge = gridCharge(period, pvCharge, gridLoad);
386 if (arbitrage && period.equals(peak) && pvCharge == 0) {
387 exportDrawn = Math.min(Math.min(
388 ceiling(100 * (value(settings, "EXPORT") - pvExport), deff),
389 value(settings, "DISCHARGE") - loadDrawn),
390 Math.max(0, soc - reserve));
391 batteryExport = exportDrawn * deff / 100;
392 soc -= exportDrawn;
393 }
394 }
395 long loss = pvCharge - pvCharge * ceff / 100 +
396 gridCharge - gridCharge * ceff / 100 +
397 loadDrawn - loadDischarge + exportDrawn - batteryExport;
398 long[] flows = {direct, pvCharge, pvExport, curtailed,
399 loadDischarge, batteryExport, gridLoad, gridCharge};
400 return new HourRecord(index, period, load, pv, soc, flows,
401 loadDrawn + exportDrawn, shortfall, loss, prices.get(period),
402 value(settings, "FEEDIN"));
403 }
404
405 SimulationResult run() {
406 List<HourRecord> records = new ArrayList<>(data.loads.size());
407 for (int i = 0; i < data.loads.size(); ++i) records.add(runHour(i));
408 return new SimulationResult(strategy, settings, prices, startSOC, reserve, records);
409 }
410 }

518 private List<String> compare(List<String> args) {
519 List<Strategy> options = strategies(args);
520 if (options.isEmpty()) throw new CommandError("wrong number of arguments");
521 SimulationResult noBattery = baseline(settings, prices);
522 List<SimulationResult> trials = new ArrayList<>();
523 for (Strategy option : options) trials.add(simulate(option));
524 trials.sort(Comparator.comparingLong(r -> r.bill));
525 List<String> output = new ArrayList<>();
526 output.add("COMPARE baseline " + money(noBattery.bill));
527 for (int i = 0; i < trials.size(); ++i) {
528 SimulationResult r = trials.get(i);
529 output.add((i + 1) + ". " + r.strategy.text() +
530 " bill " + money(r.bill) + " saving " + money(noBattery.bill - r.bill) +
531 " sufficiency " + r.sufficiency + "%");
532 }
533 return output;
534 }
535 private List<String> size(List<String> args) {
536 requireCount(args, 3);
537 List<Long> capacities = checkedRange(number(args.get(0)), number(args.get(1)),
538 number(args.get(2)), SETTING_RANGES.get("CAPACITY"));
539 SimulationResult noBattery = baseline(settings, prices);
540 List<String> output = new ArrayList<>();
541 Long bestCapacity = null;
542 long bestCost = 0, bestSaving = 0;
543 for (long capacity : capacities) {
544 Map<String, Long> trialSettings = new HashMap<>(settings);
545 trialSettings.put("CAPACITY", capacity);
546 SimulationResult trial = simulate(new Strategy("SELF", null),
547 trialSettings, prices, null);
548 long saving = noBattery.bill - trial.bill;
549 long cost = capacity * value(trialSettings, "COST") / 1000;
550 String payback = "never";
551 if (saving > 0) {
552 long tenths = 10 * cost / saving;
553 payback = tenths / 10 + "." + tenths % 10;
554 if (bestCapacity == null || cost * bestSaving < bestCost * saving) {
555 bestCapacity = capacity;
556 bestCost = cost;
557 bestSaving = saving;
558 }
559 }
560 output.add("SIZE " + capacity + " bill " + money(trial.bill) +
561 " saving " + money(saving) + " cost " + money(cost) +
562 " payback " + payback);
563 }
564 output.add("BEST " + (bestCapacity == null ? "none" : bestCapacity));
565 return output;
566 }
567 private List<String> sweep(List<String> args) {
568 if (args.size() < 5) throw new CommandError("wrong number of arguments");
569 String name = args.get(0);
570 long[] limits = SETTING_RANGES.get(name);
571 if (limits == null) throw new CommandError("unknown parameter " + name);
572 List<Long> values = checkedRange(number(args.get(1)), number(args.get(2)),
573 number(args.get(3)), limits);
574 Strategy option = oneStrategy(args.subList(4, args.size()));
575 List<String> output = new ArrayList<>();
576 Long bestValue = null;
577 long bestBill = 0;
578 for (long v : values) {
579 Map<String, Long> trialSettings = new HashMap<>(settings);
580 trialSettings.put(name, v);
581 SimulationResult trial = simulate(option, trialSettings, prices, null);
582 SimulationResult noBattery = baseline(trialSettings, prices);
583 if (bestValue == null || trial.bill < bestBill) {
584 bestValue = v;
585 bestBill = trial.bill;
586 }
587 output.add("SWEEP " + name + " " + v + " bill " + money(trial.bill) +
588 " saving " + money(noBattery.bill - trial.bill) +
589 " sufficiency " + trial.sufficiency + "%");
590 }
591 output.add("BEST " + bestValue);
592 return output;
593 }

597 long start = number(args.get(1)), end = number(args.get(2));

600 SimulationResult previous = requireResult();

602 int last = (int) ((day - 1) * 24 + end);
603 SimulationResult result = simulate(previous.strategy, previous.settings,
604 previous.prices, new int[]{first, last});
605 long before = first > 0 ? result.hours.get(first - 1).soc : result.startSOC;
606 List<String> output = new ArrayList<>();
607 output.add("OUTAGE day " + day + " hours " + start + "-" + end +
608 " strategy " + result.strategy.text() + " soc " + before);
609 output.add("hour load pv disch soc short curt");
610 Totals total = new Totals();
611 for (int i = first; i <= last; ++i) {
612 HourRecord r = result.hours.get(i);
613 total.add(r);
614 output.add(row(new Object[]{r.hour, r.load, r.pv, r.discharge,
615 r.soc, r.shortfall, r.curtailed}, 4, 7, 7, 7, 7, 7, 7));
616 }
617 String served = total.load == 0 ? "100.00" :
618 percentage(total.load - total.shortfall, total.load);
619 output.add("TOTAL load " + total.load + " unserved " + total.shortfall +
620 " served " + served + "% bill " + money(result.bill));
621 return output;
622 }

750 private List<String> peaks(List<String> args) {

754 SimulationResult r = requireResult();

756 for (HourRecord h : r.hours) if (h.imported > 0) records.add(h);
757 records.sort(Comparator.comparingLong((HourRecord h) -> -h.imported)
758 .thenComparingInt(h -> h.index));
759 List<String> output = new ArrayList<>();
760 output.add("PEAKS strategy " + r.strategy.text());
761 for (int i = 0; i < records.size() && i < count; ++i) {
762 HourRecord h = records.get(i);

807 private List<String> cycles(List<String> args) {
808 SimulationResult r = withoutArgs(args);
809 long drawn = r.total.drawn, capacity = value(r.settings, "CAPACITY");
810 long common = gcd(drawn, capacity);
811 long numerator = drawn / common, denominator = capacity / common;
812 long thousandths = 1000 * drawn / capacity;
813 String decimal = thousandths / 1000 + "." + String.format("%03d", thousandths % 1000);
814 return List.of("CYCLES " + numerator + "/" + denominator + " " + decimal +
815 " drawn " + drawn + " losses " + r.total.loss);
816 }
817 private List<String> usage(List<String> args) {
818 SimulationResult r = withoutArgs(args);
819 long charging = 0, discharging = 0, idle = 0, full = 0, low = 0;
820 long run = 0, longest = 0;
821 HourRecord first = null;
822 for (HourRecord h : r.hours) {
823 if (h.charge > 0) ++charging;
824 if (h.drawn > 0) ++discharging;
825 if (h.charge == 0 && h.drawn == 0) ++idle;
826 if (h.soc == value(r.settings, "CAPACITY")) ++full;
827 if (h.soc <= r.reserve) {
828 ++low;
829 ++run;
830 if (run > longest) {
831 longest = run;
832 first = r.hours.get((int) (h.index - run + 1));
833 }
834 } else {
835 run = 0;
836 }
837 }
838 String line = "USAGE charging " + charging + " discharging " + discharging +
839 " idle " + idle + " full " + full + " low " + low +
840 " longest " + longest;
841 if (first != null) line += " from day " + first.day + " hour " + first.hour;
842 return List.of(line);
843 }

871 List<String> execute(String[] tokens) {
872 String command = tokens[0];
873 List<String> args = Arrays.asList(tokens).subList(1, tokens.length);
874 try {
875 return switch (command) {
876 case "RUN" -> run(args);
877 case "SET" -> setValue(args);
878 case "PRICE" -> changePrice(args);
879 case "RESET" -> reset(args);
880 case "CONFIG" -> config(args);
881 case "NOBATTERY" -> noBattery(args);
882 case "COMPARE" -> compare(args);
883 case "SIZE" -> size(args);
884 case "SWEEP" -> sweep(args);
885 case "OUTAGE" -> outage(args);
886 case "DIFF" -> diff(args);
887 case "HOURLY" -> hourly(args);
888 case "DAILY" -> daily(args);
889 case "BILL" -> bill(args);
890 case "SAVINGS" -> savings(args);
891 case "BYPERIOD" -> byPeriod(args);
892 case "DAYTYPE" -> dayType(args);
893 case "PEAKS" -> peaks(args);
894 case "PROFILE" -> profile(args);
895 case "FLOWS" -> flows(args);
896 case "SELFSUFF" -> selfSuff(args);
897 case "CYCLES" -> cycles(args);
898 case "USAGE" -> usage(args);
899 case "HISTOGRAM" -> histogram(args);
900 case "CHART" -> chart(args);
901 default -> List.of("ERROR unknown command " + command);
902 };
903 } catch (CommandError error) {
904 return List.of("ERROR " + error.getMessage());
905 }
906 }
