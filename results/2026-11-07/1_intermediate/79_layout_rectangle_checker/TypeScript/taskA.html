95function relation(a: Geometry, b: Geometry): Relation {
96 const common = {
97 x1: Math.max(a.x1, b.x1),
98 y1: Math.max(a.y1, b.y1),
99 x2: Math.min(a.x2, b.x2),
100 y2: Math.min(a.y2, b.y2),
101 };
102 const ox = common.x2 - common.x1,
103 oy = common.y2 - common.y1;
104 const kind =
105 ox < 0 || oy < 0
106 ? "separated"
107 : ox === 0 && oy === 0
108 ? "corner"
109 : ox === 0 || oy === 0
110 ? "edge"
111 : "overlap";
112 return {
113 common,
114 ox,
115 oy,
116 kind,
117 gap: Math.max(0, -ox, -oy),
118 direction: ox < 0 && oy < 0 ? "diagonal" : ox < 0 ? "x" : "y",
119 };
120}
121function connected(a: Geometry, b: Geometry): boolean {
122 const kind = relation(a, b).kind;
123 return kind === "overlap" || kind === "edge";
124}

129 return [first, second];
130}

135class Coverage {
136 x: number[];
137 y: number[];
138 nx: number;
139 ny: number;
140 counts: number[][];
141 prefix: number[][];

143 constructor(rects: Geometry[]) {
144 this.x = [...new Set(rects.flatMap((rect) => [rect.x1, rect.x2]))].sort((a, b) => a - b);
145 this.y = [...new Set(rects.flatMap((rect) => [rect.y1, rect.y2]))].sort((a, b) => a - b);
146 this.nx = this.x.length - 1;
147 this.ny = this.y.length - 1;
148 const xi = new Map<number, number>(this.x.map((value, i) => [value, i])),
149 yi = new Map<number, number>(this.y.map((value, i) => [value, i]));
150 const diff: number[][] = Array.from({ length: this.y.length }, () =>
151 Array(this.x.length).fill(0),
152 );
153 for (const rect of rects) {
154 const a = xi.get(rect.x1)!,
155 b = xi.get(rect.x2)!,
156 c = yi.get(rect.y1)!,
157 d = yi.get(rect.y2)!;
158 diff[c][a]++;
159 diff[c][b]--;
160 diff[d][a]--;
161 diff[d][b]++;
162 }
163 for (let j = 0; j < this.y.length; j++)
164 for (let i = 0; i < this.x.length; i++)
165 diff[j][i] +=
166 (j ? diff[j - 1][i] : 0) +
167 (i ? diff[j][i - 1] : 0) -
168 (i && j ? diff[j - 1][i - 1] : 0);
169 this.counts = diff.slice(0, this.ny).map((row) => row.slice(0, this.nx));
170 this.prefix = Array.from({ length: this.ny + 1 }, () => Array(this.nx + 1).fill(0));
171 for (let j = 0; j < this.ny; j++)
172 for (let i = 0; i < this.nx; i++)
173 this.prefix[j + 1][i + 1] =
174 this.prefix[j][i + 1] +
175 this.prefix[j + 1][i] -
176 this.prefix[j][i] +
177 (this.counts[j][i] > 0
178 ? (this.x[i + 1] - this.x[i]) * (this.y[j + 1] - this.y[j])
179 : 0);
180 }
181 covered(i: number, j: number): boolean {
182 return i >= 0 && i < this.nx && j >= 0 && j < this.ny && this.counts[j][i] > 0;
183 }
184 area(threshold = 1): number {
185 let total = 0;
186 for (let j = 0; j < this.ny; j++)
187 for (let i = 0; i < this.nx; i++)
188 if (this.counts[j][i] >= threshold)
189 total += (this.x[i + 1] - this.x[i]) * (this.y[j + 1] - this.y[j]);
190 return total;
191 }
192 perimeter(): number {
193 let total = 0;
194 for (let j = 0; j < this.ny; j++)
195 for (let i = 0; i < this.nx; i++)
196 if (this.covered(i, j)) {
197 if (!this.covered(i - 1, j)) total += this.y[j + 1] - this.y[j];
198 if (!this.covered(i + 1, j)) total += this.y[j + 1] - this.y[j];
199 if (!this.covered(i, j - 1)) total += this.x[i + 1] - this.x[i];
200 if (!this.covered(i, j + 1)) total += this.x[i + 1] - this.x[i];
201 }
202 return total;
203 }
204 coordinate(values: number[], point: number): number {
205 let low = 0,
206 high = values.length;
207 while (low < high) {
208 const middle = Math.floor((low + high) / 2);
209 if (values[middle] <= point) low = middle + 1;
210 else high = middle;
211 }
212 return low - 1;
213 }
214 integral(x: number, y: number): number {
215 x = Math.max(this.x[0], Math.min(x, this.x.at(-1)!));
216 y = Math.max(this.y[0], Math.min(y, this.y.at(-1)!));
217 const i = this.coordinate(this.x, x),
218 j = this.coordinate(this.y, y),
219 dx = x - this.x[i],
220 dy = y - this.y[j];
221 let value = this.prefix[j][i];
222 if (dx > 0)
223 value +=
224 ((this.prefix[j][i + 1] - this.prefix[j][i]) / (this.x[i + 1] - this.x[i])) * dx;
225 if (dy > 0)
226 value +=
227 ((this.prefix[j + 1][i] - this.prefix[j][i]) / (this.y[j + 1] - this.y[j])) * dy;
228 if (dx > 0 && dy > 0 && this.covered(i, j)) value += dx * dy;
229 return value;
230 }
231 window(rect: Geometry): number {
232 return (
233 this.integral(rect.x2, rect.y2) -
234 this.integral(rect.x1, rect.y2) -
235 this.integral(rect.x2, rect.y1) +
236 this.integral(rect.x1, rect.y1)
237 );
238 }
239 empty(): Geometry | null {
240 let best: Geometry | null = null;
241 let bestArea = 0;
242 for (let left = 0; left < this.nx; left++) {
243 const blocked: boolean[] = Array(this.ny).fill(false);
244 for (let right = left + 1; right <= this.nx; right++) {
245 for (let j = 0; j < this.ny; j++) blocked[j] ||= this.covered(right - 1, j);
246 for (let bottom = 0; bottom < this.ny;) {
247 if (blocked[bottom]) {
248 bottom++;
249 continue;
250 }
251 let top = bottom + 1;
252 while (top < this.ny && !blocked[top]) top++;
253 const candidate = {
254 x1: this.x[left],
255 y1: this.y[bottom],
256 x2: this.x[right],
257 y2: this.y[top],
258 },
259 size = area(candidate);
260 if (
261 size > bestArea ||
262 (size === bestArea &&
263 (best === null ||
264 candidate.x1 < best.x1 ||
265 (candidate.x1 === best.x1 &&
266 (candidate.y1 < best.y1 ||
267 (candidate.y1 === best.y1 &&
268 candidate.x2 - candidate.x1 < best.x2 - best.x1)))))
269 ) {
270 best = candidate;
271 bestArea = size;
272 }
273 bottom = top;
274 }
275 }
276 }
277 return best;
278 }
279 holes(): Hole[] {
280 const visited: boolean[][] = Array.from({ length: this.ny }, () =>
281 Array(this.nx).fill(false),
282 );
283 const result: Hole[] = [];
284 for (let j = 0; j < this.ny; j++)
285 for (let i = 0; i < this.nx; i++)
286 if (!visited[j][i] && !this.covered(i, j)) {
287 const queue: Point[] = [[i, j]];
288 const cells: Geometry[] = [];
289 let boundary = false,
290 total = 0;
291 visited[j][i] = true;
292 for (let at = 0; at < queue.length; at++) {
293 const [x, y] = queue[at];
294 cells.push({
295 x1: this.x[x],
296 y1: this.y[y],
297 x2: this.x[x + 1],
298 y2: this.y[y + 1],
299 });
300 total += area(cells.at(-1)!);
301 if (x === 0 || y === 0 || x + 1 === this.nx || y + 1 === this.ny)
302 boundary = true;
303 for (const [a, b] of [
304 [x - 1, y],
305 [x + 1, y],
306 [x, y - 1],
307 [x, y + 1],
308 ])
309 if (
310 a >= 0 &&
311 a < this.nx &&
312 b >= 0 &&
313 b < this.ny &&
314 !visited[b][a] &&
315 !this.covered(a, b)
316 ) {
317 visited[b][a] = true;
318 queue.push([a, b]);
319 }
320 }
321 if (!boundary)
322 result.push({
323 box: bounds(cells),
324 area: total,
325 x: this.x[i],
326 y: this.y[j],
327 });
328 }
329 return result.sort((a, b) => a.y - b.y || a.x - b.x);
330 }
331 pieces(): Geometry[] {
332 let active = new Map<string, Geometry>();
333 const result: Geometry[] = [];
334 for (let j = 0; j < this.ny; j++) {
335 const next = new Map<string, Geometry>();
336 for (let i = 0; i < this.nx;) {
337 if (!this.covered(i, j)) {
338 i++;
339 continue;
340 }
341 const left = i;
342 while (i < this.nx && this.covered(i, j)) i++;
343 const key = this.x[left] + ":" + this.x[i],
344 old = active.get(key);
345 const rect = old ?? {
346 x1: this.x[left],
347 y1: this.y[j],
348 x2: this.x[i],
349 y2: this.y[j],
350 };
351 rect.y2 = this.y[j + 1];
352 next.set(key, rect);
353 }
354 for (const [key, rect] of active) if (!next.has(key)) result.push(rect);
355 active = next;
356 }
357 result.push(...active.values());
358 return result.sort((a, b) => a.y1 - b.y1 || a.x1 - b.x1);
359 }

361function nets(rects: Rectangle[]): Rectangle[][] {
362 const seen = new Set<string>();
363 const result: Rectangle[][] = [];
364 for (const first of rects)
365 if (!seen.has(first.name)) {
366 const queue = [first];
367 seen.add(first.name);
368 for (let i = 0; i < queue.length; i++)
369 for (const next of rects)
370 if (!seen.has(next.name) && connected(queue[i], next)) {
371 seen.add(next.name);
372 queue.push(next);
373 }
374 result.push(queue.sort((a, b) => compare(a.name, b.name)));
375 }
376 return result;
377}
378function violations(rects: Rectangle[], spacing: number): Violation[] {
379 const result: Violation[] = [];
380 for (let i = 0; i < rects.length; i++)
381 for (let j = i + 1; j < rects.length; j++) {
382 const rel = relation(rects[i], rects[j]);
383 if (rel.kind === "corner" || (rel.kind === "separated" && rel.gap < spacing))
384 result.push({
385 a: rects[i].name,
386 b: rects[j].name,
387 gap: rel.gap,
388 direction: rel.kind === "corner" ? "corner" : rel.direction,
389 });
390 }
391 return result.sort((a, b) => a.gap - b.gap || compare(a.a, b.a) || compare(a.b, b.b));
392}
393function violationLines(items: Violation[]): string[] {
394 return items.map((item) => `${item.a} ${item.b} gap ${item.gap} ${item.direction}`);
395}
396function execute(line: string): void {
397 const [command, ...args] = line.split(" "),
398 a = args[0],
399 b = args[1];
400 if (["ADD", "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE"].includes(command)) {
401 const before = new Map(rectangles);
402 if (command === "ADD") {
403 if (rectangles.has(a)) fail("duplicate rectangle " + a);
404 const [x1, y1, x2, y2] = args.slice(2).map(Number),
405 rect = { name: a, layer: b, x1, y1, x2, y2 };
406 if (x1 >= x2 || y1 >= y2) fail("invalid rectangle");
407 checkBounds(rect);
408 rectangles.set(a, rect);
409 output.push(`added ${a} ${b} ${notation(rect)}`);
410 } else {
411 const old = rectangle(a);
412 if (command === "COPY" || command === "MOVE") {
413 if (command === "COPY" && rectangles.has(b)) fail("duplicate rectangle " + b);
414 const offset = command === "COPY" ? 2 : 1,
415 dx = Number(args[offset]),
416 dy = Number(args[offset + 1]);
417 const rect = {
418 ...old,
419 name: command === "COPY" ? b : a,
420 x1: old.x1 + dx,
421 y1: old.y1 + dy,
422 x2: old.x2 + dx,
423 y2: old.y2 + dy,
424 };
425 checkBounds(rect);
426 rectangles.set(rect.name, rect);
427 output.push(
428 command === "COPY"
429 ? `copied ${a} to ${b} ${notation(rect)}`
430 : `moved ${a} to ${notation(rect)}`,
431 );
432 } else if (command === "RESIZE") {
433 const w = Number(b),
434 h = Number(args[2]);
435 if (w <= 0 || h <= 0) fail("invalid size");
436 const rect = { ...old, x2: old.x1 + w, y2: old.y1 + h };
437 checkBounds(rect);
438 rectangles.set(a, rect);
439 output.push(`resized ${a} to ${notation(rect)}`);
440 } else if (command === "SPLIT") {
441 const position = Number(args[2]),
442 name = args[3];
443 if (rectangles.has(name)) fail("duplicate rectangle " + name);
444 if (
445 b === "x"
446 ? position <= old.x1 || position >= old.x2
447 : position <= old.y1 || position >= old.y2
448 )
449 fail("cut outside " + a);
450 const first = { ...old, ...(b === "x" ? { x2: position } : { y2: position }) },
451 second = { ...old, name, ...(b === "x" ? { x1: position } : { y1: position }) };
452 rectangles.set(a, first);
453 rectangles.set(name, second);
454 output.push(
455 `split ${a} at ${b}=${position}: ${a} ${notation(first)}, ${name} ${notation(second)}`,
456 );
457 } else if (command === "LAYER") {
458 if (old.layer === b) fail("already on " + b);
459 rectangles.set(a, { ...old, layer: b });
460 output.push(`layer ${a} ${old.layer} -> ${b}`);
461 } else {
462 rectangles.delete(a);
463 output.push("deleted " + a);
464 }
465 }
466 history.push({ rectangles: before, line });
467 return;
468 }
469 if (command === "UNDO") {
470 if (!history.length) fail("nothing to undo");
471 const previous = history.pop()!;
472 rectangles = previous.rectangles;
473 output.push("undo: " + previous.line);
474 return;
475 }
476 if (command === "LIST") {
477 output.push("rectangles: " + rectangles.size);
478 for (const rect of ordered())
479 output.push(
480 `${rect.name} ${rect.layer} ${notation(rect)} ${rect.x2 - rect.x1}x${rect.y2 - rect.y1}`,
481 );
482 return;
483 }
484 if (command === "INFO") {
485 const rect = rectangle(a),
486 others = ordered().filter((other) => other.name !== a),
487 same = others.filter((other) => other.layer === rect.layer);
488 output.push(
489 `${a} ${rect.layer} ${notation(rect)} ${rect.x2 - rect.x1}x${rect.y2 - rect.y1} area ${area(rect)}`,
490 );
491 for (const [label, list] of [
492 ["overlaps", same.filter((other) => relation(rect, other).kind === "overlap")],
493 ["touches", same.filter((other) => relation(rect, other).kind === "edge")],
494 ["corners", same.filter((other) => relation(rect, other).kind === "corner")],
495 ["encloses", others.filter((other) => contains(rect, other))],
496 ["enclosed by", others.filter((other) => contains(other, rect))],
497 ] as [string, Rectangle[]][])
498 output.push(label + ": " + (list.map((other) => other.name).join(" ") || "none"));
499 return;
500 }
501 if (command === "LAYERS") {
502 const layers = [...new Set(ordered().map((rect) => rect.layer))].sort();
503 output.push("layers: " + layers.length);
504 for (const layer of layers) {
505 const rects = onLayer(layer);
506 output.push(
507 `${layer}: ${rects.length} rectangles, area ${new Coverage(rects).area()}, nets ${nets(rects).length}`,
508 );
509 }
510 return;
511 }
512 if (command === "OVERLAPS") {
513 const layers =
514 a === undefined ? [...new Set(ordered().map((rect) => rect.layer))].sort() : [a];
515 const found: string[] = [];
516 if (a !== undefined) onLayer(a);
517 for (const layer of layers) {
518 const rects = onLayer(layer);
519 for (let i = 0; i < rects.length; i++)
520 for (let j = i + 1; j < rects.length; j++) {
521 const rel = relation(rects[i], rects[j]);
522 if (rel.kind === "overlap")
523 found.push(
524 `${a === undefined ? layer + " " : ""}${rects[i].name} ${rects[j].name} ${notation(rel.common)} area ${area(rel.common)}`,
525 );
526 }
527 }
528 output.push(`overlaps ${a ?? "all layers"}: ${found.length}`, ...found);
529 return;
530 }
531 if (["INTERSECT", "ADJACENT", "ENCLOSE", "PATH"].includes(command)) {
532 const [first, second] = pair(a, b),
533 rel = relation(first, second),
534 heading = command.toLowerCase() + " " + a + " " + b + ": ";
535 if (command === "INTERSECT")
536 output.push(
537 heading +
538 (rel.kind === "overlap"
539 ? `${notation(rel.common)} area ${area(rel.common)}`
540 : rel.kind === "edge"
541 ? `segment ${notation(rel.common)} length ${rel.ox + rel.oy}`
542 : rel.kind === "corner"
543 ? "point " + point(rel.common.x1, rel.common.y1)
544 : "none, gap " + rel.gap),
545 );
546 else if (command === "ADJACENT") {
547 const direction =
548 rel.ox === 0
549 ? first.x2 === rel.common.x1
550 ? "west"
551 : "east"
552 : first.y2 === rel.common.y1
553 ? "south"
554 : "north";
555 output.push(
556 heading +
557 (rel.kind === "overlap"
558 ? "overlapping"
559 : rel.kind === "edge"
560 ? `${a} is ${direction} of ${b}, edge ${notation(rel.common)}, length ${rel.ox + rel.oy}`
561 : rel.kind === "corner"
562 ? "corner only at " + point(rel.common.x1, rel.common.y1)
563 : "not adjacent"),
564 );
565 } else if (command === "ENCLOSE") {
566 const values = margins(first, second),
567 yes = values.every((value) => value >= 0);
568 output.push(
569 heading +
570 `${yes ? "yes" : "no"}, margins left ${values[0]} bottom ${values[1]} right ${values[2]} top ${values[3]}${yes ? ", min " + Math.min(...values) : ""}`,
571 );
572 } else {
573 if (first.layer !== second.layer) fail("different layers");
574 const rects = onLayer(first.layer),
575 queue = [first],
576 parents = new Map<string, string | null>([[a, null]]);
577 for (let i = 0; i < queue.length; i++)
578 for (const next of rects)
579 if (!parents.has(next.name) && connected(queue[i], next)) {
580 parents.set(next.name, queue[i].name);
581 queue.push(next);
582 }
583 if (!parents.has(b)) output.push(heading + "none");
584 else {
585 const path: string[] = [];
586 for (let name: string | null = b; name !== null; name = parents.get(name)!)
587 path.push(name);
588 path.reverse();
589 output.push(heading + `${path.join(" -> ")} (${path.length - 1} steps)`);
590 }
591 }
592 return;
593 }
594 if (command === "AT") {
595 const x = Number(a),
596 y = Number(b),
597 list = ordered()
598 .filter((rect) => rect.x1 <= x && x <= rect.x2 && rect.y1 <= y && y <= rect.y2)
599 .map((rect) => {
600 const horizontal = x === rect.x1 || x === rect.x2,
601 vertical = y === rect.y1 || y === rect.y2;
602 return `${rect.name} ${rect.layer} ${horizontal && vertical ? "corner" : horizontal || vertical ? "edge" : "inside"}`;
603 });
604 output.push(`at ${point(x, y)}: ${list.join(", ") || "none"}`);
605 return;
606 }
607 if (command === "NEAREST") {
608 const rect = rectangle(a),
609 others = onLayer(rect.layer)
610 .filter((other) => other.name !== a)
611 .sort(
612 (x, y) =>
613 relation(rect, x).gap - relation(rect, y).gap || compare(x.name, y.name),
614 );
615 output.push(
616 `nearest ${a}: ${others.length ? others[0].name + " gap " + relation(rect, others[0]).gap : "none"}`,
617 );
618 return;
619 }
620 if (command === "RULE") {
621 const kind = a.toLowerCase();
622 if (kind === "enclose" && b === args[2]) fail("same layer " + b);
623 const rule = {
624 kind,
625 layer: b,
626 outer: kind === "enclose" ? args[2] : "",
627 value: Number(args[kind === "enclose" ? 3 : 2]),
628 };
629 rules.set([kind, rule.layer, rule.outer].join(" "), rule);
630 output.push(
631 `rule ${kind} ${rule.layer}${kind === "enclose" ? " " + rule.outer : ""} ${rule.value}`,
632 );
633 return;
634 }
635 if (command === "CHECK") {
636 if (!rules.size) fail("no rules");
637 let total = 0;
638 for (const rule of [...rules.values()].sort(
639 (a, b) =>
640 compare(a.kind, b.kind) || compare(a.layer, b.layer) || compare(a.outer, b.outer),
641 )) {
642 const rects = onLayer(rule.layer, false);
643 const found: string[] = [];
644 if (rule.kind === "spacing")
645 found.push(...violationLines(violations(rects, rule.value)));
646 else if (rule.kind === "width") {
647 for (const rect of rects) {
648 const width = Math.min(rect.x2 - rect.x1, rect.y2 - rect.y1);
649 if (width < rule.value) found.push(`${rect.name} width ${width}`);
650 }
651 } else {
652 const outers = onLayer(rule.outer, false);
653 for (const rect of rects) {
654 const best = outers.length
655 ? Math.max(...outers.map((outer) => Math.min(...margins(outer, rect))))
656 : null;
657 if (best === null || best < rule.value)
658 found.push(`${rect.name} best margin ${best ?? "none"}`);
659 }
660 }
661 output.push(
662 `${rule.kind} ${rule.layer}${rule.kind === "enclose" ? " " + rule.outer : ""} ${rule.value}: ${found.length} violations`,
663 ...found,
664 );
665 total += found.length;
666 }
667 output.push(`check: ${total} violations in ${rules.size} rules`);
668 return;
669 }

672 grid = new Coverage(rects),
673 box = bounds(rects);
674 if (command === "AREA") output.push(`area ${layer}: ${grid.area()}`);
675 else if (command === "COVER") output.push(`cover ${layer} >= ${a}: ${grid.area(Number(a))}`);
676 else if (command === "PERIMETER") output.push(`perimeter ${layer}: ${grid.perimeter()}`);
677 else if (command === "NETS") {
678 const groups = nets(rects);
679 output.push(`nets ${layer}: ${groups.length}`);

683 ),
684 );
685 } else if (command === "DRC") {
686 const items = violations(rects, Number(b));
687 output.push(
688 `drc ${layer} spacing ${b}: ${items.length} violations`,
689 ...violationLines(items),
690 );
691 } else if (command === "BBOX")
692 output.push(
693 `bbox ${layer}: ${notation(box)} ${box.x2 - box.x1}x${box.y2 - box.y1}, ${rects.length} rectangles, density ${decimal(100 * grid.area(), area(box))}%`,
694 );
695 else if (command === "DENSITY") {
696 const w = Number(b),
697 columns = Math.ceil((box.x2 - box.x1) / w),
698 rows = Math.ceil((box.y2 - box.y1) / w);
699 if (columns > 20 || rows > 20) fail("too many windows");
700 const values = Array.from({ length: rows }, (_, j) =>
701 Array.from({ length: columns }, (_, i) =>
702 grid.window({
703 x1: box.x1 + i * w,
704 y1: box.y1 + j * w,
705 x2: box.x1 + (i + 1) * w,
706 y2: box.y1 + (j + 1) * w,
707 }),
708 ),
709 );
710 output.push(`density ${layer} window ${w}: ${columns}x${rows}`);
711 for (let j = rows - 1; j >= 0; j--)
712 output.push(values[j].map((value) => Math.floor((100 * value) / (w * w))).join(" "));
713 let min: Point = [0, 0],
714 max: Point = [0, 0];
715 for (let j = 0; j < rows; j++)
716 for (let i = 0; i < columns; i++) {
717 if (values[j][i] < values[min[1]][min[0]]) min = [i, j];
718 if (values[j][i] > values[max[1]][max[0]]) max = [i, j];
719 }
720 for (const [label, at] of [
721 ["min", min],
722 ["max", max],
723 ] as [string, Point][])
724 output.push(
725 `${label} ${decimal(100 * values[at[1]][at[0]], w * w)}% at ${point(box.x1 + at[0] * w, box.y1 + at[1] * w)}`,
726 );
727 } else if (command === "RENDER") {
728 const [x1, y1, x2, y2] = args.slice(1).map(Number),
729 window = { x1, y1, x2, y2 };
730 if (x1 >= x2 || y1 >= y2) fail("invalid window");
731 if (x2 - x1 > 60 || y2 - y1 > 40) fail("window too large");
732 output.push(`render ${layer} ${notation(window)}`);

738 ).length;
739 row += count === 0 ? "." : count >= 10 ? "*" : String(count);
740 }
741 output.push(row);
742 }
743 } else if (command === "EMPTY") {
744 const empty = grid.empty();
745 output.push(
746 `empty ${layer}: ${empty === null ? "none" : notation(empty) + " area " + area(empty)}`,
747 );

750 output.push(`holes ${layer}: ${holes.length}`);

753 );
754 } else if (command === "MERGE") {

757 }
