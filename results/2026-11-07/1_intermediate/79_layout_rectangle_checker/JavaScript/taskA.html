10function compare(a, b) {

26function notation(rect) {

32function area(rect) {
33 return (rect.x2 - rect.x1) * (rect.y2 - rect.y1);

35function bounds(rects) {
36 return {
37 x1: Math.min(...rects.map((rect) => rect.x1)),
38 y1: Math.min(...rects.map((rect) => rect.y1)),
39 x2: Math.max(...rects.map((rect) => rect.x2)),
40 y2: Math.max(...rects.map((rect) => rect.y2)),
41 };
42}
43function checkBounds(rect) {
44 if ([rect.x1, rect.y1, rect.x2, rect.y2].some((value) => Math.abs(value) > 10000))
45 fail("out of bounds");
46}
47function margins(outer, inner) {
48 return [inner.x1 - outer.x1, inner.y1 - outer.y1, outer.x2 - inner.x2, outer.y2 - inner.y2];
49}
50function contains(outer, inner) {
51 return margins(outer, inner).every((value) => value >= 0);
52}
53function relation(a, b) {
54 const common = {
55 x1: Math.max(a.x1, b.x1),
56 y1: Math.max(a.y1, b.y1),
57 x2: Math.min(a.x2, b.x2),
58 y2: Math.min(a.y2, b.y2),
59 };
60 const ox = common.x2 - common.x1,
61 oy = common.y2 - common.y1;
62 const kind =
63 ox < 0 || oy < 0
64 ? "separated"
65 : ox === 0 && oy === 0
66 ? "corner"
67 : ox === 0 || oy === 0
68 ? "edge"
69 : "overlap";
70 return {
71 common,
72 ox,
73 oy,
74 kind,
75 gap: Math.max(0, -ox, -oy),
76 direction: ox < 0 && oy < 0 ? "diagonal" : ox < 0 ? "x" : "y",
77 };
78}
79function connected(a, b) {
80 const kind = relation(a, b).kind;
81 return kind === "overlap" || kind === "edge";
82}
83function pair(a, b) {
84 const first = rectangle(a),
85 second = rectangle(b);
86 if (a === b) fail("same rectangle " + a);
87 return [first, second];
88}
89function decimal(numerator, denominator) {
90 const value = Math.floor((100 * numerator) / denominator);
91 return Math.floor(value / 100) + "." + String(value % 100).padStart(2, "0");

93class Coverage {
94 constructor(rects) {
95 this.x = [...new Set(rects.flatMap((rect) => [rect.x1, rect.x2]))].sort((a, b) => a - b);
96 this.y = [...new Set(rects.flatMap((rect) => [rect.y1, rect.y2]))].sort((a, b) => a - b);
97 this.nx = this.x.length - 1;
98 this.ny = this.y.length - 1;
99 const xi = new Map(this.x.map((value, i) => [value, i])),
100 yi = new Map(this.y.map((value, i) => [value, i]));
101 const diff = Array.from({ length: this.y.length }, () => Array(this.x.length).fill(0));
102 for (const rect of rects) {
103 const a = xi.get(rect.x1),
104 b = xi.get(rect.x2),
105 c = yi.get(rect.y1),
106 d = yi.get(rect.y2);
107 diff[c][a]++;
108 diff[c][b]--;
109 diff[d][a]--;
110 diff[d][b]++;
111 }
112 for (let j = 0; j < this.y.length; j++)
113 for (let i = 0; i < this.x.length; i++)
114 diff[j][i] +=
115 (j ? diff[j - 1][i] : 0) +
116 (i ? diff[j][i - 1] : 0) -
117 (i && j ? diff[j - 1][i - 1] : 0);
118 this.counts = diff.slice(0, this.ny).map((row) => row.slice(0, this.nx));
119 this.prefix = Array.from({ length: this.ny + 1 }, () => Array(this.nx + 1).fill(0));
120 for (let j = 0; j < this.ny; j++)
121 for (let i = 0; i < this.nx; i++)
122 this.prefix[j + 1][i + 1] =
123 this.prefix[j][i + 1] +
124 this.prefix[j + 1][i] -
125 this.prefix[j][i] +
126 (this.counts[j][i] > 0
127 ? (this.x[i + 1] - this.x[i]) * (this.y[j + 1] - this.y[j])
128 : 0);
129 }
130 covered(i, j) {
131 return i >= 0 && i < this.nx && j >= 0 && j < this.ny && this.counts[j][i] > 0;
132 }
133 area(threshold = 1) {
134 let total = 0;
135 for (let j = 0; j < this.ny; j++)
136 for (let i = 0; i < this.nx; i++)
137 if (this.counts[j][i] >= threshold)
138 total += (this.x[i + 1] - this.x[i]) * (this.y[j + 1] - this.y[j]);
139 return total;
140 }
141 perimeter() {
142 let total = 0;
143 for (let j = 0; j < this.ny; j++)
144 for (let i = 0; i < this.nx; i++)
145 if (this.covered(i, j)) {
146 if (!this.covered(i - 1, j)) total += this.y[j + 1] - this.y[j];
147 if (!this.covered(i + 1, j)) total += this.y[j + 1] - this.y[j];
148 if (!this.covered(i, j - 1)) total += this.x[i + 1] - this.x[i];
149 if (!this.covered(i, j + 1)) total += this.x[i + 1] - this.x[i];
150 }
151 return total;
152 }
153 coordinate(values, point) {
154 let low = 0,
155 high = values.length;
156 while (low < high) {
157 const middle = Math.floor((low + high) / 2);
158 if (values[middle] <= point) low = middle + 1;
159 else high = middle;
160 }
161 return low - 1;
162 }
163 integral(x, y) {
164 x = Math.max(this.x[0], Math.min(x, this.x.at(-1)));
165 y = Math.max(this.y[0], Math.min(y, this.y.at(-1)));
166 const i = this.coordinate(this.x, x),
167 j = this.coordinate(this.y, y),
168 dx = x - this.x[i],
169 dy = y - this.y[j];
170 let value = this.prefix[j][i];
171 if (dx > 0)
172 value +=
173 ((this.prefix[j][i + 1] - this.prefix[j][i]) / (this.x[i + 1] - this.x[i])) * dx;
174 if (dy > 0)
175 value +=
176 ((this.prefix[j + 1][i] - this.prefix[j][i]) / (this.y[j + 1] - this.y[j])) * dy;
177 if (dx > 0 && dy > 0 && this.covered(i, j)) value += dx * dy;
178 return value;
179 }
180 window(rect) {
181 return (
182 this.integral(rect.x2, rect.y2) -
183 this.integral(rect.x1, rect.y2) -
184 this.integral(rect.x2, rect.y1) +
185 this.integral(rect.x1, rect.y1)
186 );
187 }
188 empty() {
189 let best = null,
190 bestArea = 0;
191 for (let left = 0; left < this.nx; left++) {
192 const blocked = Array(this.ny).fill(false);
193 for (let right = left + 1; right <= this.nx; right++) {
194 for (let j = 0; j < this.ny; j++) blocked[j] ||= this.covered(right - 1, j);
195 for (let bottom = 0; bottom < this.ny;) {
196 if (blocked[bottom]) {
197 bottom++;
198 continue;
199 }
200 let top = bottom + 1;
201 while (top < this.ny && !blocked[top]) top++;
202 const candidate = {
203 x1: this.x[left],
204 y1: this.y[bottom],
205 x2: this.x[right],
206 y2: this.y[top],
207 },
208 size = area(candidate);
209 if (
210 size > bestArea ||
211 (size === bestArea &&
212 (best === null ||
213 candidate.x1 < best.x1 ||
214 (candidate.x1 === best.x1 &&
215 (candidate.y1 < best.y1 ||
216 (candidate.y1 === best.y1 &&
217 candidate.x2 - candidate.x1 < best.x2 - best.x1)))))
218 ) {
219 best = candidate;
220 bestArea = size;
221 }
222 bottom = top;
223 }
224 }
225 }
226 return best;
227 }
228 holes() {
229 const visited = Array.from({ length: this.ny }, () => Array(this.nx).fill(false)),
230 result = [];
231 for (let j = 0; j < this.ny; j++)
232 for (let i = 0; i < this.nx; i++)
233 if (!visited[j][i] && !this.covered(i, j)) {
234 const queue = [[i, j]],
235 cells = [];
236 let boundary = false,
237 total = 0;
238 visited[j][i] = true;
239 for (let at = 0; at < queue.length; at++) {
240 const [x, y] = queue[at];
241 cells.push({
242 x1: this.x[x],
243 y1: this.y[y],
244 x2: this.x[x + 1],
245 y2: this.y[y + 1],
246 });
247 total += area(cells.at(-1));
248 if (x === 0 || y === 0 || x + 1 === this.nx || y + 1 === this.ny)
249 boundary = true;
250 for (const [a, b] of [
251 [x - 1, y],
252 [x + 1, y],
253 [x, y - 1],
254 [x, y + 1],
255 ])
256 if (
257 a >= 0 &&
258 a < this.nx &&
259 b >= 0 &&
260 b < this.ny &&
261 !visited[b][a] &&
262 !this.covered(a, b)
263 ) {
264 visited[b][a] = true;
265 queue.push([a, b]);
266 }
267 }
268 if (!boundary)
269 result.push({
270 box: bounds(cells),
271 area: total,
272 x: this.x[i],
273 y: this.y[j],
274 });
275 }
276 return result.sort((a, b) => a.y - b.y || a.x - b.x);
277 }
278 pieces() {
279 let active = new Map();
280 const result = [];
281 for (let j = 0; j < this.ny; j++) {
282 const next = new Map();
283 for (let i = 0; i < this.nx;) {
284 if (!this.covered(i, j)) {
285 i++;
286 continue;
287 }
288 const left = i;
289 while (i < this.nx && this.covered(i, j)) i++;
290 const key = this.x[left] + ":" + this.x[i],
291 old = active.get(key);
292 const rect = old ?? {
293 x1: this.x[left],
294 y1: this.y[j],
295 x2: this.x[i],
296 y2: this.y[j],
297 };
298 rect.y2 = this.y[j + 1];
299 next.set(key, rect);
300 }
301 for (const [key, rect] of active) if (!next.has(key)) result.push(rect);
302 active = next;
303 }
304 result.push(...active.values());
305 return result.sort((a, b) => a.y1 - b.y1 || a.x1 - b.x1);
306 }
307}
308function nets(rects) {
309 const seen = new Set(),
310 result = [];
311 for (const first of rects)
312 if (!seen.has(first.name)) {
313 const queue = [first];
314 seen.add(first.name);
315 for (let i = 0; i < queue.length; i++)
316 for (const next of rects)
317 if (!seen.has(next.name) && connected(queue[i], next)) {
318 seen.add(next.name);
319 queue.push(next);
320 }
321 result.push(queue.sort((a, b) => compare(a.name, b.name)));
322 }
323 return result;
324}
325function violations(rects, spacing) {
326 const result = [];
327 for (let i = 0; i < rects.length; i++)
328 for (let j = i + 1; j < rects.length; j++) {
329 const rel = relation(rects[i], rects[j]);
330 if (rel.kind === "corner" || (rel.kind === "separated" && rel.gap < spacing))
331 result.push({
332 a: rects[i].name,
333 b: rects[j].name,
334 gap: rel.gap,
335 direction: rel.kind === "corner" ? "corner" : rel.direction,
336 });
337 }
338 return result.sort((a, b) => a.gap - b.gap || compare(a.a, b.a) || compare(a.b, b.b));
339}
340function violationLines(items) {
341 return items.map((item) => `${item.a} ${item.b} gap ${item.gap} ${item.direction}`);
342}
343function execute(line) {
344 const [command, ...args] = line.split(" "),
345 a = args[0],
346 b = args[1];
347 if (["ADD", "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE"].includes(command)) {
348 const before = new Map(rectangles);
349 if (command === "ADD") {
350 if (rectangles.has(a)) fail("duplicate rectangle " + a);
351 const [x1, y1, x2, y2] = args.slice(2).map(Number),
352 rect = { name: a, layer: b, x1, y1, x2, y2 };
353 if (x1 >= x2 || y1 >= y2) fail("invalid rectangle");
354 checkBounds(rect);
355 rectangles.set(a, rect);
356 output.push(`added ${a} ${b} ${notation(rect)}`);
357 } else {
358 const old = rectangle(a);
359 if (command === "COPY" || command === "MOVE") {
360 if (command === "COPY" && rectangles.has(b)) fail("duplicate rectangle " + b);
361 const offset = command === "COPY" ? 2 : 1,
362 dx = Number(args[offset]),
363 dy = Number(args[offset + 1]);
364 const rect = {
365 ...old,
366 name: command === "COPY" ? b : a,
367 x1: old.x1 + dx,
368 y1: old.y1 + dy,
369 x2: old.x2 + dx,
370 y2: old.y2 + dy,
371 };
372 checkBounds(rect);
373 rectangles.set(rect.name, rect);
374 output.push(
375 command === "COPY"
376 ? `copied ${a} to ${b} ${notation(rect)}`
377 : `moved ${a} to ${notation(rect)}`,
378 );
379 } else if (command === "RESIZE") {
380 const w = Number(b),
381 h = Number(args[2]);
382 if (w <= 0 || h <= 0) fail("invalid size");
383 const rect = { ...old, x2: old.x1 + w, y2: old.y1 + h };
384 checkBounds(rect);
385 rectangles.set(a, rect);
386 output.push(`resized ${a} to ${notation(rect)}`);
387 } else if (command === "SPLIT") {
388 const position = Number(args[2]),
389 name = args[3];
390 if (rectangles.has(name)) fail("duplicate rectangle " + name);
391 if (
392 b === "x"
393 ? position <= old.x1 || position >= old.x2
394 : position <= old.y1 || position >= old.y2
395 )
396 fail("cut outside " + a);
397 const first = { ...old, ...(b === "x" ? { x2: position } : { y2: position }) },
398 second = { ...old, name, ...(b === "x" ? { x1: position } : { y1: position }) };
399 rectangles.set(a, first);
400 rectangles.set(name, second);
401 output.push(
402 `split ${a} at ${b}=${position}: ${a} ${notation(first)}, ${name} ${notation(second)}`,
403 );
404 } else if (command === "LAYER") {
405 if (old.layer === b) fail("already on " + b);
406 rectangles.set(a, { ...old, layer: b });
407 output.push(`layer ${a} ${old.layer} -> ${b}`);
408 } else {
409 rectangles.delete(a);
410 output.push("deleted " + a);
411 }
412 }
413 history.push({ rectangles: before, line });
414 return;
415 }
416 if (command === "UNDO") {
417 if (!history.length) fail("nothing to undo");
418 const previous = history.pop();
419 rectangles = previous.rectangles;
420 output.push("undo: " + previous.line);
421 return;
422 }
423 if (command === "LIST") {
424 output.push("rectangles: " + rectangles.size);
425 for (const rect of ordered())
426 output.push(
427 `${rect.name} ${rect.layer} ${notation(rect)} ${rect.x2 - rect.x1}x${rect.y2 - rect.y1}`,
428 );
429 return;
430 }
431 if (command === "INFO") {
432 const rect = rectangle(a),
433 others = ordered().filter((other) => other.name !== a),
434 same = others.filter((other) => other.layer === rect.layer);
435 output.push(
436 `${a} ${rect.layer} ${notation(rect)} ${rect.x2 - rect.x1}x${rect.y2 - rect.y1} area ${area(rect)}`,
437 );
438 for (const [label, list] of [
439 ["overlaps", same.filter((other) => relation(rect, other).kind === "overlap")],
440 ["touches", same.filter((other) => relation(rect, other).kind === "edge")],
441 ["corners", same.filter((other) => relation(rect, other).kind === "corner")],
442 ["encloses", others.filter((other) => contains(rect, other))],
443 ["enclosed by", others.filter((other) => contains(other, rect))],
444 ])
445 output.push(label + ": " + (list.map((other) => other.name).join(" ") || "none"));
446 return;
447 }
448 if (command === "LAYERS") {
449 const layers = [...new Set(ordered().map((rect) => rect.layer))].sort();
450 output.push("layers: " + layers.length);
451 for (const layer of layers) {
452 const rects = onLayer(layer);
453 output.push(
454 `${layer}: ${rects.length} rectangles, area ${new Coverage(rects).area()}, nets ${nets(rects).length}`,
455 );
456 }
457 return;
458 }
459 if (command === "OVERLAPS") {
460 const layers =
461 a === undefined ? [...new Set(ordered().map((rect) => rect.layer))].sort() : [a],
462 found = [];
463 if (a !== undefined) onLayer(a);
464 for (const layer of layers) {
465 const rects = onLayer(layer);
466 for (let i = 0; i < rects.length; i++)
467 for (let j = i + 1; j < rects.length; j++) {
468 const rel = relation(rects[i], rects[j]);
469 if (rel.kind === "overlap")
470 found.push(
471 `${a === undefined ? layer + " " : ""}${rects[i].name} ${rects[j].name} ${notation(rel.common)} area ${area(rel.common)}`,
472 );
473 }
474 }
475 output.push(`overlaps ${a ?? "all layers"}: ${found.length}`, ...found);
476 return;
477 }
478 if (["INTERSECT", "ADJACENT", "ENCLOSE", "PATH"].includes(command)) {
479 const [first, second] = pair(a, b),
480 rel = relation(first, second),
481 heading = command.toLowerCase() + " " + a + " " + b + ": ";
482 if (command === "INTERSECT")
483 output.push(
484 heading +
485 (rel.kind === "overlap"
486 ? `${notation(rel.common)} area ${area(rel.common)}`
487 : rel.kind === "edge"
488 ? `segment ${notation(rel.common)} length ${rel.ox + rel.oy}`
489 : rel.kind === "corner"
490 ? "point " + point(rel.common.x1, rel.common.y1)
491 : "none, gap " + rel.gap),
492 );
493 else if (command === "ADJACENT") {
494 const direction =
495 rel.ox === 0
496 ? first.x2 === rel.common.x1
497 ? "west"
498 : "east"
499 : first.y2 === rel.common.y1
500 ? "south"
501 : "north";
502 output.push(
503 heading +
504 (rel.kind === "overlap"
505 ? "overlapping"
506 : rel.kind === "edge"
507 ? `${a} is ${direction} of ${b}, edge ${notation(rel.common)}, length ${rel.ox + rel.oy}`
508 : rel.kind === "corner"
509 ? "corner only at " + point(rel.common.x1, rel.common.y1)
510 : "not adjacent"),
511 );
512 } else if (command === "ENCLOSE") {
513 const values = margins(first, second),
514 yes = values.every((value) => value >= 0);
515 output.push(
516 heading +
517 `${yes ? "yes" : "no"}, margins left ${values[0]} bottom ${values[1]} right ${values[2]} top ${values[3]}${yes ? ", min " + Math.min(...values) : ""}`,
518 );
519 } else {
520 if (first.layer !== second.layer) fail("different layers");
521 const rects = onLayer(first.layer),
522 queue = [first],
523 parents = new Map([[a, null]]);
524 for (let i = 0; i < queue.length; i++)
525 for (const next of rects)
526 if (!parents.has(next.name) && connected(queue[i], next)) {
527 parents.set(next.name, queue[i].name);
528 queue.push(next);
529 }
530 if (!parents.has(b)) output.push(heading + "none");
531 else {
532 const path = [];
533 for (let name = b; name !== null; name = parents.get(name)) path.push(name);
534 path.reverse();
535 output.push(heading + `${path.join(" -> ")} (${path.length - 1} steps)`);
536 }
537 }
538 return;
539 }
540 if (command === "AT") {
541 const x = Number(a),
542 y = Number(b),
543 list = ordered()
544 .filter((rect) => rect.x1 <= x && x <= rect.x2 && rect.y1 <= y && y <= rect.y2)
545 .map((rect) => {
546 const horizontal = x === rect.x1 || x === rect.x2,
547 vertical = y === rect.y1 || y === rect.y2;
548 return `${rect.name} ${rect.layer} ${horizontal && vertical ? "corner" : horizontal || vertical ? "edge" : "inside"}`;
549 });
550 output.push(`at ${point(x, y)}: ${list.join(", ") || "none"}`);
551 return;
552 }
553 if (command === "NEAREST") {
554 const rect = rectangle(a),
555 others = onLayer(rect.layer)
556 .filter((other) => other.name !== a)
557 .sort(
558 (x, y) =>
559 relation(rect, x).gap - relation(rect, y).gap || compare(x.name, y.name),
560 );
561 output.push(
562 `nearest ${a}: ${others.length ? others[0].name + " gap " + relation(rect, others[0]).gap : "none"}`,
563 );
564 return;
565 }
566 if (command === "RULE") {
567 const kind = a.toLowerCase();
568 if (kind === "enclose" && b === args[2]) fail("same layer " + b);
569 const rule = {
570 kind,
571 layer: b,
572 outer: kind === "enclose" ? args[2] : "",
573 value: Number(args[kind === "enclose" ? 3 : 2]),
574 };
575 rules.set([kind, rule.layer, rule.outer].join(" "), rule);
576 output.push(
577 `rule ${kind} ${rule.layer}${kind === "enclose" ? " " + rule.outer : ""} ${rule.value}`,
578 );
579 return;
580 }
581 if (command === "CHECK") {
582 if (!rules.size) fail("no rules");
583 let total = 0;
584 for (const rule of [...rules.values()].sort(
585 (a, b) =>
586 compare(a.kind, b.kind) || compare(a.layer, b.layer) || compare(a.outer, b.outer),
587 )) {
588 const rects = onLayer(rule.layer, false),
589 found = [];
590 if (rule.kind === "spacing")
591 found.push(...violationLines(violations(rects, rule.value)));
592 else if (rule.kind === "width") {
593 for (const rect of rects) {
594 const width = Math.min(rect.x2 - rect.x1, rect.y2 - rect.y1);
595 if (width < rule.value) found.push(`${rect.name} width ${width}`);
596 }
597 } else {
598 const outers = onLayer(rule.outer, false);
599 for (const rect of rects) {
600 const best = outers.length
601 ? Math.max(...outers.map((outer) => Math.min(...margins(outer, rect))))
602 : null;
603 if (best === null || best < rule.value)
604 found.push(`${rect.name} best margin ${best ?? "none"}`);
605 }
606 }
607 output.push(
608 `${rule.kind} ${rule.layer}${rule.kind === "enclose" ? " " + rule.outer : ""} ${rule.value}: ${found.length} violations`,
609 ...found,
610 );
611 total += found.length;
612 }
613 output.push(`check: ${total} violations in ${rules.size} rules`);
614 return;
615 }
616 const layer = command === "COVER" ? b : a,
617 rects = onLayer(layer),
618 grid = new Coverage(rects),
619 box = bounds(rects);
620 if (command === "AREA") output.push(`area ${layer}: ${grid.area()}`);
621 else if (command === "COVER") output.push(`cover ${layer} >= ${a}: ${grid.area(Number(a))}`);
622 else if (command === "PERIMETER") output.push(`perimeter ${layer}: ${grid.perimeter()}`);

629 ),

640 );
641 else if (command === "DENSITY") {
642 const w = Number(b),
643 columns = Math.ceil((box.x2 - box.x1) / w),
644 rows = Math.ceil((box.y2 - box.y1) / w);

647 Array.from({ length: columns }, (_, i) =>

649 x1: box.x1 + i * w,
650 y1: box.y1 + j * w,
651 x2: box.x1 + (i + 1) * w,
652 y2: box.y1 + (j + 1) * w,
653 }),
654 ),
655 );
656 output.push(`density ${layer} window ${w}: ${columns}x${rows}`);
657 for (let j = rows - 1; j >= 0; j--)
658 output.push(values[j].map((value) => Math.floor((100 * value) / (w * w))).join(" "));
659 let min = [0, 0],
660 max = [0, 0];
661 for (let j = 0; j < rows; j++)
662 for (let i = 0; i < columns; i++) {
663 if (values[j][i] < values[min[1]][min[0]]) min = [i, j];
664 if (values[j][i] > values[max[1]][max[0]]) max = [i, j];
665 }

667 ["min", min],
668 ["max", max],
669 ])

672 );
673 } else if (command === "RENDER") {

684 ).length;
685 row += count === 0 ? "." : count >= 10 ? "*" : String(count);
686 }

688 }
689 } else if (command === "EMPTY") {
690 const empty = grid.empty();
691 output.push(
692 `empty ${layer}: ${empty === null ? "none" : notation(empty) + " area " + area(empty)}`,
693 );
694 } else if (command === "HOLES") {

696 output.push(`holes ${layer}: ${holes.length}`);
697 holes.forEach((hole, i) =>

699 );
700 } else if (command === "MERGE") {
701 const pieces = grid.pieces();

703 }
704}
