59 static class Relation {
60 Box common;
61 int ox, oy, gap, kind;
62 String direction;
63
64 Relation(Rectangle a, Rectangle b) {
65 Box x = a.box, y = b.box;
66 common = new Box(Math.max(x.x1, y.x1), Math.max(x.y1, y.y1), Math.min(x.x2, y.x2), Math.min(x.y2, y.y2));
67 ox = common.width();
68 oy = common.height();
69 gap = Math.max(0, Math.max(-ox, -oy));
70 kind = ox < 0 || oy < 0 ? 3 : ox > 0 && oy > 0 ? 0 : ox == 0 && oy == 0 ? 2 : 1;
71 direction = kind == 2 ? "corner" : ox < 0 && oy < 0 ? "diagonal" : ox < 0 ? "x" : "y";
72 }
73
74 boolean connected() {
75 return kind <= 1;
76 }
77 }

83 Grid(List<Rectangle> rectangles) {
84 TreeSet<Integer> x = new TreeSet<>(), y = new TreeSet<>();
85 for (Rectangle r : rectangles) {
86 x.add(r.box.x1);
87 x.add(r.box.x2);
88 y.add(r.box.y1);
89 y.add(r.box.y2);
90 }
91 xs = x.stream().mapToInt(Integer::intValue).toArray();
92 ys = y.stream().mapToInt(Integer::intValue).toArray();
93 counts = new int[ys.length - 1][xs.length - 1];
94 for (Rectangle r : rectangles) for (int j = 0; j < counts.length; j++) if (ys[j] >= r.box.y1 && ys[j + 1] <= r.box.y2) for (int i = 0; i < counts[j].length; i++) if (xs[i] >= r.box.x1 && xs[i + 1] <= r.box.x2) counts[j][i]++;
95 }
96
97 Box bounds() {
98 return new Box(xs[0], ys[0], xs[xs.length - 1], ys[ys.length - 1]);
99 }
100
101 long area(int threshold) {
102 long result = 0;
103 for (int j = 0; j < counts.length; j++) for (int i = 0; i < counts[j].length; i++) if (counts[j][i] >= threshold) result += (long)(xs[i + 1] - xs[i]) * (ys[j + 1] - ys[j]);
104 return result;
105 }
106
107 long clipped(Box b) {
108 long result = 0;
109 for (int j = 0; j < counts.length; j++) for (int i = 0; i < counts[j].length; i++) if (counts[j][i] > 0) result += (long)Math.max(0, Math.min(xs[i + 1], b.x2) - Math.max(xs[i], b.x1)) * Math.max(0, Math.min(ys[j + 1], b.y2) - Math.max(ys[j], b.y1));
110 return result;
111 }
112
113 long perimeter() {
114 long result = 0;
115 for (int j = 0; j < counts.length; j++) for (int i = 0; i < counts[j].length; i++) if (counts[j][i] > 0) {
116 if (i == 0 || counts[j][i - 1] == 0) result += ys[j + 1] - ys[j];
117 if (i + 1 == counts[j].length || counts[j][i + 1] == 0) result += ys[j + 1] - ys[j];
118 if (j == 0 || counts[j - 1][i] == 0) result += xs[i + 1] - xs[i];
119 if (j + 1 == counts.length || counts[j + 1][i] == 0) result += xs[i + 1] - xs[i];
120 }
121 return result;
122 }
123
124 Box empty() {
125 Box best = null;
126 for (int left = 0; left + 1 < xs.length; left++) {
127 boolean[] blocked = new boolean[counts.length];
128 for (int right = left + 1; right < xs.length; right++) {
129 for (int j = 0; j < counts.length; j++) blocked[j] |= counts[j][right - 1] > 0;
130 int bottom = 0;
131 while (bottom < counts.length) {
132 if (blocked[bottom]) {
133 bottom++;
134 continue;
135 }
136 int top = bottom + 1;
137 while (top < counts.length && !blocked[top]) top++;
138 Box candidate = new Box(xs[left], ys[bottom], xs[right], ys[top]);
139 if (best == null || candidate.area() > best.area() || candidate.area() == best.area() && (candidate.x1 < best.x1 || candidate.x1 == best.x1 && (candidate.y1 < best.y1 || candidate.y1 == best.y1 && candidate.width() < best.width()))) best = candidate;
140 bottom = top;
141 }
142 }
143 }
144 return best;
145 }

147 List<Hole> holes() {
148 boolean[][] seen = new boolean[counts.length][xs.length - 1];
149 List<Hole> result = new ArrayList<>();
150 int[] dx = {-1, 1, 0, 0}, dy = {0, 0, -1, 1};
151 for (int j = 0; j < counts.length; j++) for (int i = 0; i < counts[j].length; i++) if (counts[j][i] == 0 && !seen[j][i]) {
152 ArrayDeque<int[]> queue = new ArrayDeque<>();
153 queue.add(new int[]{i, j});
154 seen[j][i] = true;
155 boolean boundary = false;
156 long area = 0;
157 int x1 = xs[i], y1 = ys[j], x2 = xs[i + 1], y2 = ys[j + 1], ax = xs[i], ay = ys[j];
158 while (!queue.isEmpty()) {
159 int[] cell = queue.remove();
160 int x = cell[0], y = cell[1];
161 boundary |= x == 0 || y == 0 || x + 1 == xs.length - 1 || y + 1 == counts.length;
162 area += (long)(xs[x + 1] - xs[x]) * (ys[y + 1] - ys[y]);
163 x1 = Math.min(x1, xs[x]);
164 y1 = Math.min(y1, ys[y]);
165 x2 = Math.max(x2, xs[x + 1]);
166 y2 = Math.max(y2, ys[y + 1]);
167 if (ys[y] < ay || ys[y] == ay && xs[x] < ax) {
168 ax = xs[x];
169 ay = ys[y];
170 }
171 for (int d = 0; d < 4; d++) {
172 int xx = x + dx[d], yy = y + dy[d];
173 if (xx >= 0 && xx + 1 < xs.length && yy >= 0 && yy < counts.length && counts[yy][xx] == 0 && !seen[yy][xx]) {
174 seen[yy][xx] = true;
175 queue.add(new int[]{xx, yy});
176 }
177 }
178 }
179 if (!boundary) result.add(new Hole(area, new Box(x1, y1, x2, y2), ax, ay));
180 }
181 result.sort(Comparator.comparingInt(Hole::anchorY).thenComparingInt(Hole::anchorX));
182 return result;
183 }

185 List<Box> pieces() {
186 Map<Interval, Box> active = new HashMap<>();
187 List<Box> result = new ArrayList<>();
188 for (int j = 0; j < counts.length; j++) {
189 Map<Interval, Box> next = new HashMap<>();
190 int i = 0;
191 while (i + 1 < xs.length) {
192 if (counts[j][i] == 0) {
193 i++;
194 continue;
195 }
196 int right = i + 1;
197 while (right + 1 < xs.length && counts[j][right] > 0) right++;
198 Interval key = new Interval(xs[i], xs[right]);
199 Box old = active.remove(key);
200 next.put(key, new Box(xs[i], old == null ? ys[j] : old.y1, xs[right], ys[j + 1]));
201 i = right;
202 }
203 result.addAll(active.values());
204 active = next;
205 }
206 result.addAll(active.values());
207 result.sort(Comparator.comparingInt(Box::y1).thenComparingInt(Box::x1));
208 return result;
209 }

255 static List<List<Rectangle>> nets(List<Rectangle> list) {
256 boolean[] seen = new boolean[list.size()];
257 List<List<Rectangle>> result = new ArrayList<>();
258 for (int start = 0; start < list.size(); start++) if (!seen[start]) {
259 List<Rectangle> net = new ArrayList<>();
260 ArrayDeque<Integer> queue = new ArrayDeque<>();
261 queue.add(start);
262 seen[start] = true;
263 while (!queue.isEmpty()) {
264 int i = queue.remove();
265 net.add(list.get(i));
266 for (int j = 0; j < list.size(); j++) if (!seen[j] && new Relation(list.get(i), list.get(j)).connected()) {
267 seen[j] = true;
268 queue.add(j);
269 }
270 }
271 net.sort(Comparator.comparing(Rectangle::name));
272 result.add(net);
273 }
274 return result;
275 }

277 static List<Violation> spacing(List<Rectangle> list, int s) {
278 List<Violation> result = new ArrayList<>();
279 for (int i = 0; i < list.size(); i++) for (int j = i + 1; j < list.size(); j++) {
280 Relation relation = new Relation(list.get(i), list.get(j));
281 if (relation.kind == 2 || relation.kind == 3 && relation.gap < s) result.add(new Violation(list.get(i), list.get(j), relation.gap, relation.direction));
282 }
283 result.sort(Comparator.comparingInt(Violation::gap).thenComparing(v -> v.a.name).thenComparing(v -> v.b.name));
284 return result;
285 }

295 List<String> command(String line) {
296 String[] a = line.split(" ");
297 String c = a[0];
298 List<String> out = new ArrayList<>();
299 switch (c) {
300 case "ADD" -> {
301 if (rectangles.containsKey(a[1])) fail("duplicate rectangle " + a[1]);
302 Box b = new Box(Integer.parseInt(a[3]), Integer.parseInt(a[4]), Integer.parseInt(a[5]), Integer.parseInt(a[6]));
303 if (b.x1 >= b.x2 || b.y1 >= b.y2) fail("invalid rectangle");
304 bounds(b);
305 save(line);
306 rectangles.put(a[1], new Rectangle(a[1], a[2], b));
307 out.add("added " + a[1] + " " + a[2] + " " + b);
308 }
309 case "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE" -> {
310 Rectangle r = get(a[1]);
311 Box b = r.box;
312 String layer = r.layer;
313 Rectangle extra = null;
314 switch (c) {
315 case "COPY" -> {
316 if (rectangles.containsKey(a[2])) fail("duplicate rectangle " + a[2]);
317 b = b.shift(Integer.parseInt(a[3]), Integer.parseInt(a[4]));
318 bounds(b);
319 extra = new Rectangle(a[2], layer, b);
320 out.add("copied " + r.name + " to " + a[2] + " " + b);
321 }
322 case "MOVE" -> {
323 b = b.shift(Integer.parseInt(a[2]), Integer.parseInt(a[3]));
324 bounds(b);
325 out.add("moved " + r.name + " to " + b);
326 }
327 case "RESIZE" -> {
328 int w = Integer.parseInt(a[2]), h = Integer.parseInt(a[3]);
329 if (w <= 0 || h <= 0) fail("invalid size");
330 b = new Box(b.x1, b.y1, b.x1 + w, b.y1 + h);
331 bounds(b);
332 out.add("resized " + r.name + " to " + b);
333 }
334 case "SPLIT" -> {
335 if (rectangles.containsKey(a[4])) fail("duplicate rectangle " + a[4]);
336 int pos = Integer.parseInt(a[3]);
337 boolean x = a[2].equals("x");
338 if (pos <= (x ? b.x1 : b.y1) || pos >= (x ? b.x2 : b.y2)) fail("cut outside " + r.name);
339 Box second = x ? new Box(pos, b.y1, b.x2, b.y2) : new Box(b.x1, pos, b.x2, b.y2);
340 b = x ? new Box(b.x1, b.y1, pos, b.y2) : new Box(b.x1, b.y1, b.x2, pos);
341 extra = new Rectangle(a[4], layer, second);
342 out.add("split " + r.name + " at " + a[2] + "=" + pos + ": " + r.name + " " + b + ", " + a[4] + " " + second);
343 }
344 case "LAYER" -> {
345 if (layer.equals(a[2])) fail("already on " + layer);
346 out.add("layer " + r.name + " " + layer + " -> " + a[2]);
347 layer = a[2];
348 }
349 case "DELETE" -> out.add("deleted " + r.name);
350 default -> throw new IllegalStateException();
351 }
352 save(line);
353 if (c.equals("DELETE")) rectangles.remove(r.name);
354 else if (!c.equals("COPY")) rectangles.put(r.name, new Rectangle(r.name, layer, b));
355 if (extra != null) rectangles.put(extra.name, extra);
356 }
357 case "UNDO" -> {
358 if (history.isEmpty()) fail("nothing to undo");
359 Saved saved = history.removeLast();
360 rectangles = saved.rectangles;
361 out.add("undo: " + saved.command);
362 }
363 case "LIST" -> {
364 out.add("rectangles: " + rectangles.size());
365 for (Rectangle r : rectangles.values()) out.add(info(r));
366 }
367 case "INFO" -> {
368 Rectangle r = get(a[1]);
369 out.add(info(r) + " area " + r.box.area());
370 String[] labels = {"overlaps", "touches", "corners", "encloses", "enclosed by"};
371 for (int k = 0; k < labels.length; k++) {
372 List<Rectangle> list = new ArrayList<>();
373 for (Rectangle other : rectangles.values()) if (!r.name.equals(other.name)) {
374 Relation relation = new Relation(r, other);
375 if (k < 3 ? r.layer.equals(other.layer) && relation.kind == k : k == 3 ? r.box.contains(other.box) : other.box.contains(r.box)) list.add(other);
376 }
377 out.add(labels[k] + ": " + names(list));
378 }
379 }
380 case "LAYERS" -> {
381 TreeSet<String> layers = new TreeSet<>();
382 for (Rectangle r : rectangles.values()) layers.add(r.layer);
383 out.add("layers: " + layers.size());
384 for (String name : layers) {
385 List<Rectangle> list = layer(name, true);
386 out.add(name + ": " + list.size() + " rectangles, area " + new Grid(list).area(1) + ", nets " + nets(list).size());
387 }
388 }
389 case "OVERLAPS" -> {
390 List<String> details = new ArrayList<>();
391 TreeSet<String> layers = new TreeSet<>();
392 if (a.length == 2) {
393 layer(a[1], true);
394 layers.add(a[1]);
395 } else for (Rectangle r : rectangles.values()) layers.add(r.layer);
396 for (String name : layers) {
397 List<Rectangle> list = layer(name, true);
398 for (int i = 0; i < list.size(); i++) for (int j = i + 1; j < list.size(); j++) {
399 Rectangle x = list.get(i), y = list.get(j);
400 Relation relation = new Relation(x, y);
401 if (relation.kind == 0) details.add((a.length == 1 ? name + " " : "") + x.name + " " + y.name + " " + relation.common + " area " + relation.common.area());
402 }
403 }
404 out.add("overlaps " + (a.length == 1 ? "all layers" : a[1]) + ": " + details.size());
405 out.addAll(details);
406 }
407 case "INTERSECT", "ADJACENT", "ENCLOSE", "PATH" -> {
408 Rectangle x = get(a[1]), y = get(a[2]);
409 if (x.name.equals(y.name)) fail("same rectangle " + x.name);
410 Relation relation = new Relation(x, y);
411 String prefix = c.toLowerCase(Locale.ROOT) + " " + x.name + " " + y.name + ": ";
412 if (c.equals("PATH")) {
413 if (!x.layer.equals(y.layer)) fail("different layers");
414 List<Rectangle> list = layer(x.layer, true);
415 int start = list.indexOf(x), goal = list.indexOf(y);
416 int[] previous = new int[list.size()];
417 Arrays.fill(previous, -1);
418 previous[start] = start;
419 ArrayDeque<Integer> queue = new ArrayDeque<>();
420 queue.add(start);
421 while (!queue.isEmpty()) {
422 int i = queue.remove();
423 for (int j = 0; j < list.size(); j++) if (previous[j] < 0 && new Relation(list.get(i), list.get(j)).connected()) {
424 previous[j] = i;
425 queue.add(j);
426 }
427 }
428 List<String> path = new ArrayList<>();
429 if (previous[goal] >= 0) for (int at = goal;; at = previous[at]) {
430 path.add(0, list.get(at).name);
431 if (at == start) break;
432 }
433 out.add(prefix + (path.isEmpty() ? "none" : String.join(" -> ", path) + " (" + (path.size() - 1) + " steps)"));
434 } else if (c.equals("ENCLOSE")) {
435 int left = y.box.x1 - x.box.x1, bottom = y.box.y1 - x.box.y1, right = x.box.x2 - y.box.x2, top = x.box.y2 - y.box.y2, min = margin(x, y);
436 out.add(prefix + (min >= 0 ? "yes" : "no") + ", margins left " + left + " bottom " + bottom + " right " + right + " top " + top + (min >= 0 ? ", min " + min : ""));
437 } else if (c.equals("INTERSECT")) out.add(prefix + switch (relation.kind) {
438 case 0 -> relation.common + " area " + relation.common.area(); case 1 -> "segment " + relation.common + " length " + (relation.ox + relation.oy); case 2 -> "point (" + relation.common.x1 + "," + relation.common.y1 + ")"; default -> "none, gap " + relation.gap;
439 });
440 else out.add(prefix + switch (relation.kind) {
441 case 0 -> "overlapping"; case 1 -> x.name + " is " + (relation.ox == 0 ? x.box.x2 == relation.common.x1 ? "west" : "east" : x.box.y2 == relation.common.y1 ? "south" : "north") + " of " + y.name + ", edge " + relation.common + ", length " + (relation.ox + relation.oy); case 2 -> "corner only at (" + relation.common.x1 + "," + relation.common.y1 + ")"; default -> "not adjacent";
442 });
443 }
444 case "AT" -> {
445 int x = Integer.parseInt(a[1]), y = Integer.parseInt(a[2]);
446 List<String> found = new ArrayList<>();
447 for (Rectangle r : rectangles.values()) if (x >= r.box.x1 && x <= r.box.x2 && y >= r.box.y1 && y <= r.box.y2) {
448 boolean xx = x == r.box.x1 || x == r.box.x2, yy = y == r.box.y1 || y == r.box.y2;
449 found.add(r.name + " " + r.layer + " " + (xx && yy ? "corner" : xx || yy ? "edge" : "inside"));
450 }
451 out.add("at (" + x + "," + y + "): " + (found.isEmpty() ? "none" : String.join(", ", found)));
452 }
453 case "NEAREST" -> {
454 Rectangle r = get(a[1]), best = null;
455 int gap = Integer.MAX_VALUE;
456 for (Rectangle other : layer(r.layer, true)) if (!r.name.equals(other.name)) {
457 int g = new Relation(r, other).gap;
458 if (g < gap) {
459 gap = g;
460 best = other;
461 }
462 }
463 out.add("nearest " + r.name + ": " + (best == null ? "none" : best.name + " gap " + gap));
464 }
465 case "RULE" -> {
466 String kind = a[1].toLowerCase(Locale.ROOT), inner = a[2], outer = kind.equals("enclose") ? a[3] : "";
467 int value = Integer.parseInt(a[kind.equals("enclose") ? 4 : 3]);
468 if (kind.equals("enclose") && inner.equals(outer)) fail("same layer " + inner);
469 rules.put(kind + " " + inner + " " + outer, new Rule(kind, inner, outer, value));
470 out.add("rule " + kind + " " + inner + (outer.isEmpty() ? "" : " " + outer) + " " + value);
471 }
472 case "CHECK" -> {
473 if (rules.isEmpty()) fail("no rules");
474 int total = 0;
475 for (Rule rule : rules.values()) {
476 List<String> violations = new ArrayList<>();
477 List<Rectangle> list = layer(rule.inner, false);
478 if (rule.kind.equals("spacing")) for (Violation v : spacing(list, rule.value)) violations.add(violation(v));
479 else if (rule.kind.equals("width")) for (Rectangle r : list) {
480 int width = Math.min(r.box.width(), r.box.height());
481 if (width < rule.value) violations.add(r.name + " width " + width);
482 } else {
483 List<Rectangle> outer = layer(rule.outer, false);
484 for (Rectangle r : list) {
485 Integer best = null;
486 for (Rectangle o : outer) best = best == null ? margin(o, r) : Math.max(best, margin(o, r));
487 if (best == null || best < rule.value) violations.add(r.name + " best margin " + (best == null ? "none" : best));
488 }
489 }
490 out.add(rule.kind + " " + rule.inner + (rule.outer.isEmpty() ? "" : " " + rule.outer) + " " + rule.value + ": " + violations.size() + " violations");
491 out.addAll(violations);
492 total += violations.size();
493 }
494 out.add("check: " + total + " violations in " + rules.size() + " rules");
495 }
496 default -> {
497 String name = c.equals("COVER") ? a[2] : a[1];
498 List<Rectangle> list = layer(name, true);
499 Grid grid = new Grid(list);
500 Box b = grid.bounds();
501 switch (c) {
502 case "AREA" -> out.add("area " + name + ": " + grid.area(1));
503 case "COVER" -> out.add("cover " + name + " >= " + a[1] + ": " + grid.area(Integer.parseInt(a[1])));
504 case "PERIMETER" -> out.add("perimeter " + name + ": " + grid.perimeter());
505 case "NETS" -> {
506 List<List<Rectangle>> groups = nets(list);
507 out.add("nets " + name + ": " + groups.size());
508 for (int i = 0; i < groups.size(); i++) out.add("net " + (i + 1) + ": " + names(groups.get(i)) + " area " + new Grid(groups.get(i)).area(1));
509 }
510 case "DRC" -> {
511 int s = Integer.parseInt(a[2]);
512 List<Violation> violations = spacing(list, s);
513 out.add("drc " + name + " spacing " + s + ": " + violations.size() + " violations");
514 for (Violation v : violations) out.add(violation(v));
515 }
516 case "BBOX" -> out.add("bbox " + name + ": " + b + " " + b.width() + "x" + b.height() + ", " + list.size() + " rectangles, density " + density(grid.area(1), b.area()) + "%");
517 case "DENSITY" -> {
518 int w = Integer.parseInt(a[2]), columns = (b.width() + w - 1) / w, rows = (b.height() + w - 1) / w;
519 if (columns > 20 || rows > 20) fail("too many windows");
520 long[][] values = new long[rows][columns];
521 long min = Long.MAX_VALUE, max = -1;
522 int minX = 0, minY = 0, maxX = 0, maxY = 0;
523 for (int j = 0; j < rows; j++) for (int i = 0; i < columns; i++) {
524 values[j][i] = grid.clipped(new Box(b.x1 + i * w, b.y1 + j * w, b.x1 + (i + 1) * w, b.y1 + (j + 1) * w));
525 if (values[j][i] < min) {
526 min = values[j][i];
527 minX = i;
528 minY = j;
529 }
530 if (values[j][i] > max) {
531 max = values[j][i];
532 maxX = i;
533 maxY = j;
534 }
535 }
536 out.add("density " + name + " window " + w + ": " + columns + "x" + rows);
537 for (int j = rows - 1; j >= 0; j--) {
538 List<String> row = new ArrayList<>();
539 for (int i = 0; i < columns; i++) row.add(Long.toString(100L * values[j][i] / ((long)w * w)));
540 out.add(String.join(" ", row));
541 }
542 out.add("min " + density(min, (long)w * w) + "% at (" + (b.x1 + minX * w) + "," + (b.y1 + minY * w) + ")");
543 out.add("max " + density(max, (long)w * w) + "% at (" + (b.x1 + maxX * w) + "," + (b.y1 + maxY * w) + ")");
544 }
545 case "RENDER" -> {
546 Box window = new Box(Integer.parseInt(a[2]), Integer.parseInt(a[3]), Integer.parseInt(a[4]), Integer.parseInt(a[5]));
547 if (window.x1 >= window.x2 || window.y1 >= window.y2) fail("invalid window");
548 if (window.width() > 60 || window.height() > 40) fail("window too large");
549 out.add("render " + name + " " + window);
550 for (int y = window.y2 - 1; y >= window.y1; y--) {
551 StringBuilder row = new StringBuilder();
552 for (int x = window.x1; x < window.x2; x++) {
553 int count = 0;
554 for (Rectangle r : list) if (r.box.x1 <= x && x < r.box.x2 && r.box.y1 <= y && y < r.box.y2) count++;
555 row.append(count == 0 ? '.' : count < 10 ? (char)('0' + count) : '*');
556 }
557 out.add(row.toString());
558 }
559 }
560 case "EMPTY" -> {
561 Box empty = grid.empty();
562 out.add("empty " + name + ": " + (empty == null ? "none" : empty + " area " + empty.area()));
563 }
564 case "HOLES" -> {
565 List<Hole> holes = grid.holes();
566 out.add("holes " + name + ": " + holes.size());
567 for (int i = 0; i < holes.size(); i++) out.add("hole " + (i + 1) + ": area " + holes.get(i).area + ", bbox " + holes.get(i).box);
568 }
569 case "MERGE" -> {
570 List<Box> pieces = grid.pieces();
571 out.add("merge " + name + ": " + pieces.size() + " pieces");
572 for (Box piece : pieces) out.add(piece.toString());
573 }
574 default -> throw new IllegalStateException();
575 }
576 }
577 }
578 return out;
579 }
