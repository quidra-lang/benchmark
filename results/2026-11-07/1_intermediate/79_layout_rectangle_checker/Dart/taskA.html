72String notation(Geometry rect) => '(${rect.x1},${rect.y1})-(${rect.x2},${rect.y2})';
73String point(int x, int y) => '($x,$y)';
74int area(Geometry rect) => (rect.x2 - rect.x1) * (rect.y2 - rect.y1);
75Geometry bounds(List<Geometry> rects) => Geometry(
76 rects.map((rect) => rect.x1).reduce(math.min),
77 rects.map((rect) => rect.y1).reduce(math.min),
78 rects.map((rect) => rect.x2).reduce(math.max),
79 rects.map((rect) => rect.y2).reduce(math.max));
80void checkBounds(Geometry rect) {
81 if ([rect.x1, rect.y1, rect.x2, rect.y2].any((value) => value.abs() > 10000)) {
82 fail('out of bounds');
83 }

85List<int> margins(Geometry outer, Geometry inner) =>
86 [inner.x1 - outer.x1, inner.y1 - outer.y1, outer.x2 - inner.x2, outer.y2 - inner.y2];
87bool contains(Geometry outer, Geometry inner) => margins(outer, inner).every((value) => value >= 0);
88Relation relation(Geometry a, Geometry b) {
89 final common = Geometry(math.max(a.x1, b.x1), math.max(a.y1, b.y1),
90 math.min(a.x2, b.x2), math.min(a.y2, b.y2));
91 final ox = common.x2 - common.x1, oy = common.y2 - common.y1;
92 final kind = ox < 0 || oy < 0 ? 'separated'
93 : ox == 0 && oy == 0 ? 'corner' : ox == 0 || oy == 0 ? 'edge' : 'overlap';
94 return Relation(common, ox, oy, kind, math.max(0, math.max(-ox, -oy)),
95 ox < 0 && oy < 0 ? 'diagonal' : ox < 0 ? 'x' : 'y');
96}
97bool connected(Geometry a, Geometry b) {
98 final kind = relation(a, b).kind;
99 return kind == 'overlap' || kind == 'edge';

111class Coverage {
112 late final List<int> x, y;
113 late final int nx, ny;
114 late final List<List<int>> counts, prefix;
115 Coverage(List<Geometry> rects) {
116 x = {for (final rect in rects) ...[rect.x1, rect.x2]}.toList()..sort();
117 y = {for (final rect in rects) ...[rect.y1, rect.y2]}.toList()..sort();
118 nx = x.length - 1;
119 ny = y.length - 1;
120 final xi = {for (int i = 0; i < x.length; i++) x[i]: i};
121 final yi = {for (int j = 0; j < y.length; j++) y[j]: j};
122 final diff = List.generate(y.length, (_) => List.filled(x.length, 0));
123 for (final rect in rects) {
124 final a = xi[rect.x1]!, b = xi[rect.x2]!, c = yi[rect.y1]!, d = yi[rect.y2]!;
125 diff[c][a]++;
126 diff[c][b]--;
127 diff[d][a]--;
128 diff[d][b]++;
129 }
130 for (int j = 0; j < y.length; j++) {
131 for (int i = 0; i < x.length; i++) {
132 diff[j][i] += (j > 0 ? diff[j - 1][i] : 0) + (i > 0 ? diff[j][i - 1] : 0)
133 - (i > 0 && j > 0 ? diff[j - 1][i - 1] : 0);
134 }
135 }
136 counts = [for (int j = 0; j < ny; j++) diff[j].sublist(0, nx)];
137 prefix = List.generate(ny + 1, (_) => List.filled(nx + 1, 0));
138 for (int j = 0; j < ny; j++) {
139 for (int i = 0; i < nx; i++) {
140 prefix[j + 1][i + 1] = prefix[j][i + 1] + prefix[j + 1][i] - prefix[j][i]
141 + (counts[j][i] > 0 ? (x[i + 1] - x[i]) * (y[j + 1] - y[j]) : 0);
142 }
143 }
144 }
145 bool covered(int i, int j) => i >= 0 && i < nx && j >= 0 && j < ny && counts[j][i] > 0;
146 int coveredArea([int threshold = 1]) {
147 int total = 0;
148 for (int j = 0; j < ny; j++) {
149 for (int i = 0; i < nx; i++) {
150 if (counts[j][i] >= threshold) total += (x[i + 1] - x[i]) * (y[j + 1] - y[j]);
151 }
152 }
153 return total;
154 }
155 int perimeter() {
156 int total = 0;
157 for (int j = 0; j < ny; j++) {
158 for (int i = 0; i < nx; i++) {
159 if (!covered(i, j)) continue;
160 if (!covered(i - 1, j)) total += y[j + 1] - y[j];
161 if (!covered(i + 1, j)) total += y[j + 1] - y[j];
162 if (!covered(i, j - 1)) total += x[i + 1] - x[i];
163 if (!covered(i, j + 1)) total += x[i + 1] - x[i];
164 }
165 }
166 return total;
167 }
168 int coordinate(List<int> values, int point) {
169 int low = 0, high = values.length;
170 while (low < high) {
171 final middle = (low + high) ~/ 2;
172 if (values[middle] <= point) { low = middle + 1; } else { high = middle; }
173 }
174 return low - 1;
175 }
176 int integral(int px, int py) {
177 px = math.max(x.first, math.min(px, x.last));
178 py = math.max(y.first, math.min(py, y.last));
179 final i = coordinate(x, px), j = coordinate(y, py);
180 final dx = px - x[i], dy = py - y[j];
181 int value = prefix[j][i];
182 if (dx > 0) value += (prefix[j][i + 1] - prefix[j][i]) ~/ (x[i + 1] - x[i]) * dx;
183 if (dy > 0) value += (prefix[j + 1][i] - prefix[j][i]) ~/ (y[j + 1] - y[j]) * dy;
184 if (dx > 0 && dy > 0 && covered(i, j)) value += dx * dy;
185 return value;
186 }
187 int window(Geometry rect) => integral(rect.x2, rect.y2) - integral(rect.x1, rect.y2)
188 - integral(rect.x2, rect.y1) + integral(rect.x1, rect.y1);
189 Geometry? empty() {
190 Geometry? best;
191 int bestArea = 0;
192 for (int left = 0; left < nx; left++) {
193 final blocked = List.filled(ny, false);
194 for (int right = left + 1; right <= nx; right++) {
195 for (int j = 0; j < ny; j++) { blocked[j] = blocked[j] || covered(right - 1, j); }
196 for (int bottom = 0; bottom < ny;) {
197 if (blocked[bottom]) { bottom++; continue; }
198 int top = bottom + 1;
199 while (top < ny && !blocked[top]) { top++; }
200 final candidate = Geometry(x[left], y[bottom], x[right], y[top]);
201 final size = area(candidate);
202 if (size > bestArea || size == bestArea && (best == null || candidate.x1 < best.x1
203 || candidate.x1 == best.x1 && (candidate.y1 < best.y1
204 || candidate.y1 == best.y1 && candidate.x2 - candidate.x1 < best.x2 - best.x1))) {
205 best = candidate;
206 bestArea = size;
207 }
208 bottom = top;
209 }
210 }
211 }
212 return best;
213 }
214 List<Hole> holes() {
215 final visited = List.generate(ny, (_) => List.filled(nx, false));
216 final result = <Hole>[];
217 for (int j = 0; j < ny; j++) {
218 for (int i = 0; i < nx; i++) {
219 if (visited[j][i] || covered(i, j)) continue;
220 final queue = <Point>[(i, j)], cells = <Geometry>[];
221 bool boundary = false;
222 int total = 0;
223 visited[j][i] = true;
224 for (int at = 0; at < queue.length; at++) {
225 final (cx, cy) = queue[at];
226 cells.add(Geometry(x[cx], y[cy], x[cx + 1], y[cy + 1]));
227 total += area(cells.last);
228 if (cx == 0 || cy == 0 || cx + 1 == nx || cy + 1 == ny) boundary = true;
229 for (final (a, b) in [(cx - 1, cy), (cx + 1, cy), (cx, cy - 1), (cx, cy + 1)]) {
230 if (a >= 0 && a < nx && b >= 0 && b < ny && !visited[b][a] && !covered(a, b)) {
231 visited[b][a] = true;
232 queue.add((a, b));
233 }
234 }
235 }
236 if (!boundary) result.add(Hole(bounds(cells), total, x[i], y[j]));
237 }
238 }
239 return result..sort((a, b) => a.y != b.y ? a.y.compareTo(b.y) : a.x.compareTo(b.x));
240 }
241 List<Geometry> pieces() {
242 var active = <String, Geometry>{};
243 final result = <Geometry>[];
244 for (int j = 0; j < ny; j++) {
245 final next = <String, Geometry>{};
246 for (int i = 0; i < nx;) {
247 if (!covered(i, j)) { i++; continue; }
248 final left = i;
249 while (i < nx && covered(i, j)) { i++; }
250 final key = '${x[left]}:${x[i]}';
251 final rect = active[key] ?? Geometry(x[left], y[j], x[i], y[j]);
252 rect.y2 = y[j + 1];
253 next[key] = rect;
254 }
255 for (final entry in active.entries) { if (!next.containsKey(entry.key)) result.add(entry.value); }
256 active = next;
257 }
258 result.addAll(active.values);
259 return result..sort((a, b) => a.y1 != b.y1 ? a.y1.compareTo(b.y1) : a.x1.compareTo(b.x1));
260 }
261}

263List<List<Rectangle>> nets(List<Rectangle> rects) {
264 final seen = <String>{}, result = <List<Rectangle>>[];
265 for (final first in rects) {
266 if (seen.contains(first.name)) continue;
267 final queue = [first];
268 seen.add(first.name);
269 for (int i = 0; i < queue.length; i++) {
270 for (final next in rects) {
271 if (!seen.contains(next.name) && connected(queue[i], next)) {
272 seen.add(next.name);
273 queue.add(next);
274 }
275 }
276 }
277 result.add(queue..sort((a, b) => a.name.compareTo(b.name)));
278 }
279 return result;
280}
281List<Violation> violations(List<Rectangle> rects, int spacing) {
282 final result = <Violation>[];
283 for (int i = 0; i < rects.length; i++) {
284 for (int j = i + 1; j < rects.length; j++) {
285 final rel = relation(rects[i], rects[j]);
286 if (rel.kind == 'corner' || rel.kind == 'separated' && rel.gap < spacing) {
287 result.add(Violation(rects[i].name, rects[j].name, rel.gap,
288 rel.kind == 'corner' ? 'corner' : rel.direction));
289 }
290 }
291 }
292 return result..sort((a, b) => a.gap != b.gap ? a.gap.compareTo(b.gap)
293 : a.a != b.a ? a.a.compareTo(b.a) : a.b.compareTo(b.b));
294}

296void execute(String line) {
297 final words = line.split(' '), command = words[0], args = words.sublist(1);
298 final a = args.isEmpty ? '' : args[0], b = args.length < 2 ? '' : args[1];
299 if (['ADD', 'COPY', 'MOVE', 'RESIZE', 'SPLIT', 'LAYER', 'DELETE'].contains(command)) {
300 final before = Map<String, Rectangle>.of(rectangles);
301 if (command == 'ADD') {
302 if (rectangles.containsKey(a)) fail('duplicate rectangle $a');
303 final values = args.sublist(2).map(int.parse).toList();
304 final rect = Rectangle(a, b, values[0], values[1], values[2], values[3]);
305 if (rect.x1 >= rect.x2 || rect.y1 >= rect.y2) fail('invalid rectangle');
306 checkBounds(rect);
307 rectangles[a] = rect;
308 output.add('added $a $b ${notation(rect)}');
309 } else {
310 final old = rectangle(a);
311 if (command == 'COPY' || command == 'MOVE') {
312 if (command == 'COPY' && rectangles.containsKey(b)) fail('duplicate rectangle $b');
313 final offset = command == 'COPY' ? 2 : 1;
314 final dx = int.parse(args[offset]), dy = int.parse(args[offset + 1]);
315 final rect = old.copy(name: command == 'COPY' ? b : a,
316 x1: old.x1 + dx, y1: old.y1 + dy, x2: old.x2 + dx, y2: old.y2 + dy);
317 checkBounds(rect);
318 rectangles[rect.name] = rect;
319 output.add(command == 'COPY' ? 'copied $a to $b ${notation(rect)}' : 'moved $a to ${notation(rect)}');
320 } else if (command == 'RESIZE') {
321 final w = int.parse(b), h = int.parse(args[2]);
322 if (w <= 0 || h <= 0) fail('invalid size');
323 final rect = old.copy(x2: old.x1 + w, y2: old.y1 + h);
324 checkBounds(rect);
325 rectangles[a] = rect;
326 output.add('resized $a to ${notation(rect)}');
327 } else if (command == 'SPLIT') {
328 final position = int.parse(args[2]), name = args[3];
329 if (rectangles.containsKey(name)) fail('duplicate rectangle $name');
330 if (b == 'x' ? position <= old.x1 || position >= old.x2 : position <= old.y1 || position >= old.y2) {
331 fail('cut outside $a');
332 }
333 final first = b == 'x' ? old.copy(x2: position) : old.copy(y2: position);
334 final second = b == 'x' ? old.copy(name: name, x1: position) : old.copy(name: name, y1: position);
335 rectangles[a] = first;
336 rectangles[name] = second;
337 output.add('split $a at $b=$position: $a ${notation(first)}, $name ${notation(second)}');
338 } else if (command == 'LAYER') {
339 if (old.layer == b) fail('already on $b');
340 rectangles[a] = old.copy(layer: b);
341 output.add('layer $a ${old.layer} -> $b');
342 } else {
343 rectangles.remove(a);
344 output.add('deleted $a');
345 }
346 }
347 history.add(UndoEntry(before, line));
348 return;
349 }
350 if (command == 'UNDO') {
351 if (history.isEmpty) fail('nothing to undo');
352 final previous = history.removeLast();
353 rectangles = previous.rectangles;
354 output.add('undo: ${previous.line}');
355 return;
356 }
357 if (command == 'LIST') {
358 output.add('rectangles: ${rectangles.length}');
359 for (final rect in ordered()) { output.add('${rect.name} ${rect.layer} ${notation(rect)} ${rect.x2 - rect.x1}x${rect.y2 - rect.y1}'); }
360 return;
361 }
362 if (command == 'INFO') {
363 final rect = rectangle(a), others = ordered().where((other) => other.name != a).toList();
364 final same = others.where((other) => other.layer == rect.layer).toList();
365 output.add('$a ${rect.layer} ${notation(rect)} ${rect.x2 - rect.x1}x${rect.y2 - rect.y1} area ${area(rect)}');
366 final groups = <(String, Iterable<Rectangle>)>[
367 ('overlaps', same.where((other) => relation(rect, other).kind == 'overlap')),
368 ('touches', same.where((other) => relation(rect, other).kind == 'edge')),
369 ('corners', same.where((other) => relation(rect, other).kind == 'corner')),
370 ('encloses', others.where((other) => contains(rect, other))),
371 ('enclosed by', others.where((other) => contains(other, rect)))
372 ];
373 for (final (label, list) in groups) {
374 final names = list.map((other) => other.name).join(' ');
375 output.add('$label: ${names.isEmpty ? 'none' : names}');
376 }
377 return;
378 }
379 if (command == 'LAYERS') {
380 final layers = ordered().map((rect) => rect.layer).toSet().toList()..sort();
381 output.add('layers: ${layers.length}');
382 for (final layer in layers) {
383 final rects = onLayer(layer);
384 output.add('$layer: ${rects.length} rectangles, area ${Coverage(rects).coveredArea()}, nets ${nets(rects).length}');
385 }
386 return;
387 }
388 if (command == 'OVERLAPS') {
389 final layers = args.isEmpty ? (ordered().map((rect) => rect.layer).toSet().toList()..sort()) : [a];
390 final found = <String>[];
391 if (args.isNotEmpty) onLayer(a);
392 for (final layer in layers) {
393 final rects = onLayer(layer);
394 for (int i = 0; i < rects.length; i++) {
395 for (int j = i + 1; j < rects.length; j++) {
396 final rel = relation(rects[i], rects[j]);
397 if (rel.kind == 'overlap') found.add('${args.isEmpty ? '$layer ' : ''}${rects[i].name} ${rects[j].name} ${notation(rel.common)} area ${area(rel.common)}');
398 }
399 }
400 }
401 output.add('overlaps ${args.isEmpty ? 'all layers' : a}: ${found.length}');
402 output.addAll(found);
403 return;
404 }
405 if (['INTERSECT', 'ADJACENT', 'ENCLOSE', 'PATH'].contains(command)) {
406 final (first, second) = pair(a, b);
407 final rel = relation(first, second), heading = '${command.toLowerCase()} $a $b: ';
408 if (command == 'INTERSECT') {
409 output.add(heading + (rel.kind == 'overlap' ? '${notation(rel.common)} area ${area(rel.common)}'
410 : rel.kind == 'edge' ? 'segment ${notation(rel.common)} length ${rel.ox + rel.oy}'
411 : rel.kind == 'corner' ? 'point ${point(rel.common.x1, rel.common.y1)}' : 'none, gap ${rel.gap}'));
412 } else if (command == 'ADJACENT') {
413 final direction = rel.ox == 0 ? (first.x2 == rel.common.x1 ? 'west' : 'east')
414 : first.y2 == rel.common.y1 ? 'south' : 'north';
415 output.add(heading + (rel.kind == 'overlap' ? 'overlapping'
416 : rel.kind == 'edge' ? '$a is $direction of $b, edge ${notation(rel.common)}, length ${rel.ox + rel.oy}'
417 : rel.kind == 'corner' ? 'corner only at ${point(rel.common.x1, rel.common.y1)}' : 'not adjacent'));
418 } else if (command == 'ENCLOSE') {
419 final values = margins(first, second), yes = values.every((value) => value >= 0);
420 output.add('$heading${yes ? 'yes' : 'no'}, margins left ${values[0]} bottom ${values[1]} right ${values[2]} top ${values[3]}${yes ? ', min ${values.reduce(math.min)}' : ''}');
421 } else {
422 if (first.layer != second.layer) fail('different layers');
423 final rects = onLayer(first.layer), queue = [first];
424 final parents = <String, String?>{a: null};
425 for (int i = 0; i < queue.length; i++) {
426 for (final next in rects) {
427 if (!parents.containsKey(next.name) && connected(queue[i], next)) {
428 parents[next.name] = queue[i].name;
429 queue.add(next);
430 }
431 }
432 }
433 if (!parents.containsKey(b)) { output.add('${heading}none'); }
434 else {
435 final path = <String>[];
436 for (String? name = b; name != null; name = parents[name]) { path.add(name); }
437 output.add('$heading${path.reversed.join(' -> ')} (${path.length - 1} steps)');
438 }
439 }
440 return;
441 }
442 if (command == 'AT') {
443 final x = int.parse(a), y = int.parse(b), list = <String>[];
444 for (final rect in ordered()) {
445 if (!(rect.x1 <= x && x <= rect.x2 && rect.y1 <= y && y <= rect.y2)) continue;
446 final horizontal = x == rect.x1 || x == rect.x2, vertical = y == rect.y1 || y == rect.y2;
447 list.add('${rect.name} ${rect.layer} ${horizontal && vertical ? 'corner' : horizontal || vertical ? 'edge' : 'inside'}');
448 }
449 output.add('at ${point(x, y)}: ${list.isEmpty ? 'none' : list.join(', ')}');
450 return;
451 }
452 if (command == 'NEAREST') {
453 final rect = rectangle(a);
454 final others = onLayer(rect.layer).where((other) => other.name != a).toList();
455 others.sort((x, y) {
456 final order = relation(rect, x).gap.compareTo(relation(rect, y).gap);
457 return order != 0 ? order : x.name.compareTo(y.name);
458 });
459 output.add('nearest $a: ${others.isEmpty ? 'none' : '${others.first.name} gap ${relation(rect, others.first).gap}'}');
460 return;
461 }
462 if (command == 'RULE') {
463 final kind = a.toLowerCase();
464 if (kind == 'enclose' && b == args[2]) fail('same layer $b');
465 final rule = Rule(kind, b, kind == 'enclose' ? args[2] : '', int.parse(args[kind == 'enclose' ? 3 : 2]));
466 rules['$kind ${rule.layer} ${rule.outer}'] = rule;
467 output.add('rule $kind ${rule.layer}${kind == 'enclose' ? ' ${rule.outer}' : ''} ${rule.value}');
468 return;
469 }
470 if (command == 'CHECK') {
471 if (rules.isEmpty) fail('no rules');
472 int total = 0;
473 final all = rules.values.toList()..sort((a, b) => a.kind != b.kind ? a.kind.compareTo(b.kind)
474 : a.layer != b.layer ? a.layer.compareTo(b.layer) : a.outer.compareTo(b.outer));
475 for (final rule in all) {
476 final rects = onLayer(rule.layer, false), found = <String>[];
477 if (rule.kind == 'spacing') { found.addAll(violations(rects, rule.value).map((item) => item.line())); }
478 else if (rule.kind == 'width') {
479 for (final rect in rects) {
480 final width = math.min(rect.x2 - rect.x1, rect.y2 - rect.y1);
481 if (width < rule.value) found.add('${rect.name} width $width');
482 }
483 } else {
484 final outers = onLayer(rule.outer, false);
485 for (final rect in rects) {
486 final best = outers.isEmpty ? null : outers.map((outer) => margins(outer, rect).reduce(math.min)).reduce(math.max);
487 if (best == null || best < rule.value) found.add('${rect.name} best margin ${best ?? 'none'}');
488 }
489 }
490 output.add('${rule.kind} ${rule.layer}${rule.kind == 'enclose' ? ' ${rule.outer}' : ''} ${rule.value}: ${found.length} violations');
491 output.addAll(found);
492 total += found.length;
493 }
494 output.add('check: $total violations in ${rules.length} rules');
495 return;
496 }
497 final layer = command == 'COVER' ? b : a, rects = onLayer(command == 'COVER' ? b : a);
498 final grid = Coverage(rects), box = bounds(rects);
499 if (command == 'AREA') { output.add('area $layer: ${grid.coveredArea()}'); }
500 else if (command == 'COVER') { output.add('cover $layer >= $a: ${grid.coveredArea(int.parse(a))}'); }
501 else if (command == 'PERIMETER') { output.add('perimeter $layer: ${grid.perimeter()}'); }
502 else if (command == 'NETS') {
503 final groups = nets(rects);
504 output.add('nets $layer: ${groups.length}');
505 for (int i = 0; i < groups.length; i++) {
506 final group = groups[i];
507 output.add('net ${i + 1}: ${group.map((rect) => rect.name).join(' ')} area ${Coverage(group).coveredArea()}');
508 }
509 } else if (command == 'DRC') {
510 final items = violations(rects, int.parse(b));
511 output.add('drc $layer spacing $b: ${items.length} violations');
512 output.addAll(items.map((item) => item.line()));
513 } else if (command == 'BBOX') {
514 output.add('bbox $layer: ${notation(box)} ${box.x2 - box.x1}x${box.y2 - box.y1}, ${rects.length} rectangles, density ${decimal(100 * grid.coveredArea(), area(box))}%');
515 } else if (command == 'DENSITY') {
516 final w = int.parse(b);
517 final columns = (box.x2 - box.x1 + w - 1) ~/ w;
518 final rows = (box.y2 - box.y1 + w - 1) ~/ w;
519 if (columns > 20 || rows > 20) fail('too many windows');
520 final values = List.generate(rows, (j) => List.generate(columns, (i) => grid.window(
521 Geometry(box.x1 + i * w, box.y1 + j * w, box.x1 + (i + 1) * w, box.y1 + (j + 1) * w))));
522 output.add('density $layer window $w: ${columns}x$rows');
523 for (int j = rows - 1; j >= 0; j--) { output.add(values[j].map((value) => 100 * value ~/ (w * w)).join(' ')); }
524 Point minimum = (0, 0), maximum = (0, 0);
525 for (int j = 0; j < rows; j++) {
526 for (int i = 0; i < columns; i++) {
527 if (values[j][i] < values[minimum.$2][minimum.$1]) minimum = (i, j);
528 if (values[j][i] > values[maximum.$2][maximum.$1]) maximum = (i, j);
529 }
530 }
531 for (final (label, at) in [('min', minimum), ('max', maximum)]) {
532 output.add('$label ${decimal(100 * values[at.$2][at.$1], w * w)}% at ${point(box.x1 + at.$1 * w, box.y1 + at.$2 * w)}');
533 }
534 } else if (command == 'RENDER') {
535 final values = args.sublist(1).map(int.parse).toList();
536 final window = Geometry(values[0], values[1], values[2], values[3]);
537 if (window.x1 >= window.x2 || window.y1 >= window.y2) fail('invalid window');
538 if (window.x2 - window.x1 > 60 || window.y2 - window.y1 > 40) fail('window too large');
539 output.add('render $layer ${notation(window)}');
540 for (int y = window.y2 - 1; y >= window.y1; y--) {
541 final row = StringBuffer();
542 for (int x = window.x1; x < window.x2; x++) {
543 final count = rects.where((rect) => rect.x1 <= x && x < rect.x2 && rect.y1 <= y && y < rect.y2).length;
544 row.write(count == 0 ? '.' : count >= 10 ? '*' : '$count');
545 }
546 output.add(row.toString());
547 }
548 } else if (command == 'EMPTY') {
549 final empty = grid.empty();
550 output.add('empty $layer: ${empty == null ? 'none' : '${notation(empty)} area ${area(empty)}'}');
551 } else if (command == 'HOLES') {
552 final holes = grid.holes();
553 output.add('holes $layer: ${holes.length}');
554 for (int i = 0; i < holes.length; i++) { output.add('hole ${i + 1}: area ${holes[i].area}, bbox ${notation(holes[i].box)}'); }
555 } else if (command == 'MERGE') {
556 final pieces = grid.pieces();
557 output.add('merge $layer: ${pieces.length} pieces');
558 output.addAll(pieces.map(notation));
559 }
560}

562void main() {
563 final count = int.parse(stdin.readLineSync()!);
564 for (int i = 0; i < count; i++) {
565 try { execute(stdin.readLineSync()!); }
566 on Problem catch (error) { output.add(error.message); }
567 }
568 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
569}
