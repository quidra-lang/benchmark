1struct CommandError <: Exception
2 message::String
3end
4fail(s) = throw(CommandError(s))
5check(b, s) = b ? nothing : fail(s)
6struct Rectangle
7 name::String
8 layer::String
9 box::NTuple{4,Int}
10end
11boxtext(b) = "($(b[1]),$(b[2]))-($(b[3]),$(b[4]))"
12pointtext(x, y) = "($x,$y)"
13rectwidth(r) = r.box[3]-r.box[1]
14rectheight(r) = r.box[4]-r.box[2]
15rectarea(r) = rectwidth(r)*rectheight(r)
16namelist(rects) = isempty(rects) ? "none" : join((r.name for r in rects), ' ')
17function margins(a, b)
18 x, y, u, v=a.box
19 p, q, r, s=b.box
20 (p-x, q-y, u-r, v-s)
21end
22containsrect(a, b) = minimum(margins(a, b))>=0
23function relation(a, b)
24 x=max(a.box[1], b.box[1])
25 y=max(a.box[2], b.box[2])
26 u=min(a.box[3], b.box[3])
27 v=min(a.box[4], b.box[4])
28 ox=u-x
29 oy=v-y
30 kind=ox>0 && oy>0 ? 0 : ox<0 || oy<0 ? 3 : ox==0 && oy==0 ? 2 : 1
31 (
32 kind = kind,
33 box = (x, y, u, v),
34 ox = ox,
35 oy = oy,
36 gap = max(0, -ox, -oy),
37 direction = ox<0 && oy<0 ? "diagonal" : ox<0 ? "x" : "y",
38 )
39end
40connected(a, b) = relation(a, b).kind in (0, 1)
41function bounding(rects)
42 (
43 minimum(r.box[1] for r in rects),
44 minimum(r.box[2] for r in rects),
45 maximum(r.box[3] for r in rects),
46 maximum(r.box[4] for r in rects),
47 )
48end
49function spacinglines(rects, s)
50 found=Tuple{Int,String,String,String}[]
51 for i in 1:(length(rects)-1), j in (i+1):length(rects)
52 a=rects[i]
53 b=rects[j]
54 r=relation(a, b)
55 r.kind==2 && push!(found, (0, a.name, b.name, "corner"))
56 r.kind==3 && r.gap<s && push!(found, (r.gap, a.name, b.name, r.direction))
57 end
58 ["$a $b gap $gap $direction" for (gap, a, b, direction) in sort!(found)]
59end
60function overlaplines(rects)
61 result=String[]
62 for i in 1:(length(rects)-1), j in (i+1):length(rects)
63 a=rects[i]
64 b=rects[j]
65 r=relation(a, b)
66 r.kind==0 && push!(result, "$(a.name) $(b.name) $(boxtext(r.box)) area $(r.ox*r.oy)")
67 end
68 result
69end
70function nets(rects)
71 seen=falses(length(rects))
72 groups=Vector{Rectangle}[]
73 for start in eachindex(rects)
74 seen[start] && continue
75 pending=[start]
76 seen[start]=true
77 group=Rectangle[]
78 while !isempty(pending)
79 i=pop!(pending)
80 push!(group, rects[i])
81 for j in eachindex(rects)
82 if !seen[j] && connected(rects[i], rects[j])
83 seen[j]=true
84 push!(pending, j)
85 end
86 end
87 end
88 push!(groups, sort!(group; by = r->r.name))
89 end
90 groups
91end
92struct Coverage
93 xs::Vector{Int}
94 ys::Vector{Int}
95 counts::Matrix{Int}
96end
97function Coverage(rects, extraxs = Int[], extrays = Int[])
98 xs=sort!(unique([extraxs; [v for r in rects for v in (r.box[1], r.box[3])]]))
99 ys=sort!(unique([extrays; [v for r in rects for v in (r.box[2], r.box[4])]]))
100 counts=zeros(Int, length(xs)-1, length(ys)-1)
101 for r in rects
102 a=searchsortedfirst(xs, r.box[1])
103 b=searchsortedfirst(xs, r.box[3])
104 c=searchsortedfirst(ys, r.box[2])
105 f=searchsortedfirst(ys, r.box[4])
106 for i in a:(b-1), j in c:(f-1)
107 counts[i, j]+=1
108 end
109 end
110 Coverage(xs, ys, counts)
111end
112cellarea(c, i, j) = (c.xs[i+1]-c.xs[i])*(c.ys[j+1]-c.ys[j])
113covered(c, i, j) = 1<=i<size(c.xs, 1) && 1<=j<size(c.ys, 1) && c.counts[i, j]>0
114coveragearea(c, k = 1) = sum(
115 (cellarea(c, i, j) for i in axes(c.counts, 1) for j in axes(c.counts, 2) if c.counts[i, j]>=k);
116 init = 0,
117)
118function perimeter(c)
119 result=0
120 nx, ny=size(c.counts)
121 for i in 1:(nx+1), j in 1:ny
122 covered(c, i-1, j)!=covered(c, i, j) && (result+=c.ys[j+1]-c.ys[j])
123 end
124 for j in 1:(ny+1), i in 1:nx
125 covered(c, i, j-1)!=covered(c, i, j) && (result+=c.xs[i+1]-c.xs[i])
126 end
127 result
128end
129function runs(flags, coordinates)
130 result=Tuple{Int,Int}[]
131 i=1
132 while i<=length(flags)
133 if !flags[i]
134 i+=1
135 continue
136 end
137 start=i
138 while i<=length(flags) && flags[i]
139 i+=1
140 end
141 push!(result, (coordinates[start], coordinates[i]))
142 end
143 result
144end
145function holes(c)
146 nx, ny=size(c.counts)
147 seen=falses(nx, ny)
148 result=Tuple{Int,NTuple{4,Int}}[]
149 for j in 1:ny, i in 1:nx
150 (seen[i, j] || c.counts[i, j]>0) && continue
151 pending=[(i, j)]
152 seen[i, j]=true
153 boundary=false
154 area=0
155 x=c.xs[i]
156 y=c.ys[j]
157 u=c.xs[i+1]
158 v=c.ys[j+1]
159 while !isempty(pending)
160 a, b=pop!(pending)
161 boundary|=a in (1, nx) || b in (1, ny)
162 area+=cellarea(c, a, b)
163 x=min(x, c.xs[a])
164 y=min(y, c.ys[b])
165 u=max(u, c.xs[a+1])
166 v=max(v, c.ys[b+1])
167 for (p, q) in ((a-1, b), (a+1, b), (a, b-1), (a, b+1))
168 if 1<=p<=nx && 1<=q<=ny && !seen[p, q] && c.counts[p, q]==0
169 seen[p, q]=true
170 push!(pending, (p, q))
171 end
172 end
173 end
174 !boundary && push!(result, (area, (x, y, u, v)))
175 end
176 result
177end
178function pieces(c)
179 nx, ny=size(c.counts)
180 rows=[runs(c.counts[:, j] .> 0, c.xs) for j in 1:ny]
181 result=NTuple{4,Int}[]
182 for j in 1:ny, (left, right) in rows[j]
183 j>1 && (left, right) in rows[j-1] && continue
184 top=j+1
185 while top<=ny && (left, right) in rows[top]
186 top+=1
187 end
188 push!(result, (left, c.ys[j], right, c.ys[top]))
189 end
190 result
191end
192function largestempty(c)
193 nx, ny=size(c.counts)
194 best=nothing
195 box=nothing
196 for a in 1:nx
197 free=trues(ny)
198 for b in (a+1):(nx+1)
199 free .&= c.counts[b-1, :] .== 0
200 width=c.xs[b]-c.xs[a]
201 for (bottom, top) in runs(free, c.ys)
202 candidate=(-width*(top-bottom), c.xs[a], bottom, width)
203 if best===nothing || candidate<best
204 best=candidate
205 box=(c.xs[a], bottom, c.xs[b], top)
206 end
207 end
208 end
209 end
210 box
211end
212function percentage(area, denominator)
213 h=div(10000area, denominator)
214 "$(div(h,100)).$(lpad(string(h%100),2,'0'))"
215end

226sortedrects(w) = sort!(collect(values(w.rectangles)); by = r->r.name)
227onlayer(w, layer) = [r for r in sortedrects(w) if r.layer==layer]
228function requirelayer(w, layer)
229 result=onlayer(w, layer)
230 check(!isempty(result), "no rectangles on $layer")
231 result
232end
233function getrect(w, name)
234 check(haskey(w.rectangles, name), "unknown rectangle $name")
235 w.rectangles[name]
236end
237unused(w, name) = check(!haskey(w.rectangles, name), "duplicate rectangle $name")
238function pair(w, a, b)
239 firstrect=getrect(w, a)
240 secondrect=getrect(w, b)
241 check(a!=b, "same rectangle $a")
242 firstrect, secondrect
243end
244function place!(w, r)
245 check(all(x->abs(x)<=10000, r.box), "out of bounds")
246 w.rectangles[r.name]=r
247end
248function report(heading, lines)
249 println(heading)
250 foreach(println, lines)
251end
252function ruleviolations(w, kind, layer, other, value)
253 kind=="spacing" && return spacinglines(onlayer(w, layer), value)
254 result=String[]
255 for r in onlayer(w, layer)
256 if kind=="width"
257 size=min(rectwidth(r), rectheight(r))
258 size<value && push!(result, "$(r.name) width $size")
259 else
260 outers=onlayer(w, other)
261 best=isempty(outers) ? nothing : maximum(minimum(margins(u, r)) for u in outers)
262 (best===nothing || best<value) &&
263 push!(result, "$(r.name) best margin $(best===nothing ? "none" : string(best))")
264 end
265 end
266 result
267end
268function execute!(w, line)
269 parts=String.(split(line, ' '))
270 command=parts[1]
271 args=parts[2:end]
272 before=copy(w.rectangles)
273 if command=="ADD"
274 name, layer=args[1:2]
275 x, y, u, v=parse.(Int, args[3:6])
276 unused(w, name)
277 check(x<u && y<v, "invalid rectangle")
278 r=Rectangle(name, layer, (x, y, u, v))
279 place!(w, r)
280 println("added $name $layer $(boxtext(r.box))")
281 elseif command in ("COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE")
282 name=args[1]
283 r=getrect(w, name)
284 x, y, u, v=r.box
285 if command=="COPY"
286 new=args[2]
287 unused(w, new)
288 dx, dy=parse.(Int, args[3:4])
289 copied=Rectangle(new, r.layer, (x+dx, y+dy, u+dx, v+dy))
290 place!(w, copied)
291 println("copied $name to $new $(boxtext(copied.box))")
292 elseif command=="MOVE"
293 dx, dy=parse.(Int, args[2:3])
294 moved=Rectangle(name, r.layer, (x+dx, y+dy, u+dx, v+dy))
295 place!(w, moved)
296 println("moved $name to $(boxtext(moved.box))")
297 elseif command=="RESIZE"
298 width, height=parse.(Int, args[2:3])
299 check(width>0 && height>0, "invalid size")
300 resized=Rectangle(name, r.layer, (x, y, x+width, y+height))
301 place!(w, resized)
302 println("resized $name to $(boxtext(resized.box))")
303 elseif command=="SPLIT"
304 axis=args[2]
305 pos=parse(Int, args[3])
306 new=args[4]
307 unused(w, new)
308 check(axis=="x" ? x<pos<u : y<pos<v, "cut outside $name")
309 firstbox=axis=="x" ? (x, y, pos, v) : (x, y, u, pos)
310 secondbox=axis=="x" ? (pos, y, u, v) : (x, pos, u, v)
311 w.rectangles[name]=Rectangle(name, r.layer, firstbox)
312 w.rectangles[new]=Rectangle(new, r.layer, secondbox)
313 println(
314 "split $name at $axis=$pos: $name $(boxtext(firstbox)), $new $(boxtext(secondbox))",
315 )
316 elseif command=="LAYER"
317 layer=args[2]
318 check(layer!=r.layer, "already on $layer")
319 w.rectangles[name]=Rectangle(name, layer, r.box)
320 println("layer $name $(r.layer) -> $layer")
321 else
322 delete!(w.rectangles, name)
323 println("deleted $name")
324 end
325 elseif command=="UNDO"
326 check(!isempty(w.history), "nothing to undo")
327 w.rectangles, original=pop!(w.history)
328 println("undo: $original")
329 elseif command=="LIST"
330 report(
331 "rectangles: $(length(w.rectangles))",
332 [
333 "$(r.name) $(r.layer) $(boxtext(r.box)) $(rectwidth(r))x$(rectheight(r))" for
334 r in sortedrects(w)
335 ],
336 )
337 elseif command=="INFO"
338 name=args[1]
339 r=getrect(w, name)
340 others=[s for s in sortedrects(w) if s.name!=name]
341 println(
342 "$name $(r.layer) $(boxtext(r.box)) $(rectwidth(r))x$(rectheight(r)) area $(rectarea(r))",
343 )
344 for (label, kind) in (("overlaps", 0), ("touches", 1), ("corners", 2))
345 println(
346 "$label: $(namelist([s for s in others if s.layer==r.layer && relation(r,s).kind==kind]))",
347 )
348 end
349 println(
350 "encloses: $(namelist([s for s in others if containsrect(r,s)]))\nenclosed by: $(namelist([s for s in others if containsrect(s,r)]))",
351 )
352 elseif command=="LAYERS"
353 layers=sort!(unique([r.layer for r in values(w.rectangles)]))
354 println("layers: $(length(layers))")
355 for layer in layers
356 rects=onlayer(w, layer)
357 println(
358 "$layer: $(length(rects)) rectangles, area $(coveragearea(Coverage(rects))), nets $(length(nets(rects)))",
359 )
360 end
361 elseif command=="OVERLAPS"
362 if isempty(args)
363 lines=String[]
364 for layer in sort!(unique([r.layer for r in values(w.rectangles)]))
365 append!(lines, ["$layer $s" for s in overlaplines(onlayer(w, layer))])
366 end
367 report("overlaps all layers: $(length(lines))", lines)
368 else
369 lines=overlaplines(requirelayer(w, args[1]))
370 report("overlaps $(args[1]): $(length(lines))", lines)
371 end
372 elseif command in ("INTERSECT", "ADJACENT", "ENCLOSE", "PATH")
373 a, b=args
374 firstrect, secondrect=pair(w, a, b)
375 r=relation(firstrect, secondrect)
376 if command=="INTERSECT"
377 detail=r.kind==0 ? "$(boxtext(r.box)) area $(r.ox*r.oy)" :
378 r.kind==1 ? "segment $(boxtext(r.box)) length $(r.ox+r.oy)" :
379 r.kind==2 ? "point $(pointtext(r.box[1],r.box[2]))" : "none, gap $(r.gap)"
380 println("intersect $a $b: $detail")
381 elseif command=="ADJACENT"
382 side=r.ox==0 ? (firstrect.box[3]==r.box[1] ? "west" : "east") :
383 (firstrect.box[4]==r.box[2] ? "south" : "north")
384 detail=r.kind==0 ? "overlapping" :
385 r.kind==1 ? "$a is $side of $b, edge $(boxtext(r.box)), length $(r.ox+r.oy)" :
386 r.kind==2 ? "corner only at $(pointtext(r.box[1],r.box[2]))" : "not adjacent"
387 println("adjacent $a $b: $detail")
388 elseif command=="ENCLOSE"
389 p, q, r0, s=margins(firstrect, secondrect)
390 minimumvalue=min(p, q, r0, s)
391 text="margins left $p bottom $q right $r0 top $s"
392 println(
393 "enclose $a $b: ",
394 minimumvalue>=0 ? "yes, $text, min $minimumvalue" : "no, $text",
395 )
396 else
397 check(firstrect.layer==secondrect.layer, "different layers")
398 rects=onlayer(w, firstrect.layer)
399 distances=Dict(b=>0)
400 queue=[b]
401 at=1
402 while at<=length(queue)
403 current=queue[at]
404 at+=1
405 for other in rects
406 if !haskey(distances, other.name) && connected(w.rectangles[current], other)
407 distances[other.name]=distances[current]+1
408 push!(queue, other.name)
409 end
410 end
411 end
412 if !haskey(distances, a)
413 println("path $a $b: none")
414 else
415 path=[a]
416 while path[end]!=b
417 push!(
418 path,
419 first(
420 s.name for
421 s in rects if get(distances, s.name, -1)==distances[path[end]]-1 &&
422 connected(w.rectangles[path[end]], s)
423 ),
424 )
425 end
426 println("path $a $b: $(join(path," -> ")) ($(length(path)-1) steps)")
427 end
428 end
429 elseif command=="AT"
430 x, y=parse.(Int, args)
431 found=String[]
432 for r in sortedrects(w)
433 a, b, c, f=r.box
434 if a<=x<=c && b<=y<=f
435 sides=(x in (a, c))+(y in (b, f))
436 push!(found, "$(r.name) $(r.layer) $(("inside","edge","corner")[sides+1])")
437 end
438 end
439 println("at $(pointtext(x,y)): ", isempty(found) ? "none" : join(found, ", "))
440 elseif command=="NEAREST"
441 r=getrect(w, args[1])
442 others=[s for s in onlayer(w, r.layer) if s.name!=r.name]
443 if isempty(others)
444 println("nearest $(r.name): none")
445 else
446 gap, name=minimum((relation(r, s).gap, s.name) for s in others)
447 println("nearest $(r.name): $name gap $gap")
448 end
449 elseif command=="RULE"
450 kind=lowercase(args[1])
451 layer=args[2]
452 other=kind=="enclose" ? args[3] : ""
453 value=parse(Int, args[end])
454 check(kind!="enclose" || layer!=other, "same layer $layer")
455 w.rules[(kind, layer, other)]=value
456 println("rule ", join([kind; args[2:end]], ' '))
457 elseif command=="CHECK"
458 check(!isempty(w.rules), "no rules")
459 total=0
460 for (kind, layer, other) in sort!(collect(keys(w.rules)))
461 value=w.rules[(kind, layer, other)]
462 lines=ruleviolations(w, kind, layer, other, value)
463 heading=kind=="enclose" ? "$kind $layer $other $value" : "$kind $layer $value"
464 report("$heading: $(length(lines)) violations", lines)
465 total+=length(lines)
466 end
467 println("check: $total violations in $(length(w.rules)) rules")
468 else
469 minimumcount=command=="COVER" ? parse(Int, args[1]) : 1
470 layer=args[command=="COVER" ? 2 : 1]
471 rects=requirelayer(w, layer)
472 if command=="DRC"
473 s=parse(Int, args[2])
474 lines=spacinglines(rects, s)
475 report("drc $layer spacing $s: $(length(lines)) violations", lines)
476 elseif command=="NETS"
477 groups=nets(rects)
478 println("nets $layer: $(length(groups))")
479 for (i, group) in enumerate(groups)
480 println("net $i: $(namelist(group)) area $(coveragearea(Coverage(group)))")
481 end
482 elseif command=="RENDER"
483 x, y, u, v=parse.(Int, args[2:5])
484 check(x<u && y<v, "invalid window")
485 check(u-x<=60 && v-y<=40, "window too large")
486 println("render $layer $(boxtext((x,y,u,v)))")
487 for j in (v-1):-1:y
488 println(
489 join([
490 begin
491 count=sum(r.box[1]<=i<r.box[3] && r.box[2]<=j<r.box[4] for r in rects)
492 count==0 ? "." : count>=10 ? "*" : string(count)
493 end for i in x:(u-1)
494 ]),
495 )
496 end
497 elseif command=="BBOX"
498 bounds=bounding(rects)
499 width=bounds[3]-bounds[1]
500 height=bounds[4]-bounds[2]
501 println(
502 "bbox $layer: $(boxtext(bounds)) $(width)x$height, $(length(rects)) rectangles, density $(percentage(coveragearea(Coverage(rects)),width*height))%",
503 )
504 elseif command=="DENSITY"
505 window=parse(Int, args[2])
506 x, y, u, v=bounding(rects)
507 nc=cld(u-x, window)
508 nr=cld(v-y, window)
509 check(nc<=20 && nr<=20, "too many windows")
510 coverage=Coverage(rects, [x+i*window for i in 0:nc], [y+j*window for j in 0:nr])
511 tiles=zeros(Int, nc, nr)
512 for i in axes(coverage.counts, 1), j in axes(coverage.counts, 2)
513 if coverage.counts[i, j]>0
514 a=div(coverage.xs[i]-x, window)+1
515 b=div(coverage.ys[j]-y, window)+1
516 tiles[a, b]+=cellarea(coverage, i, j)
517 end
518 end
519 println("density $layer window $window: $(nc)x$nr")
520 for j in nr:-1:1
521 println(join([div(100tiles[i, j], window*window) for i in 1:nc], ' '))
522 end
523 for label in ("min", "max")
524 best=label=="min" ? minimum(tiles) : maximum(tiles)
525 chosen=first((i, j) for j in 1:nr for i in 1:nc if tiles[i, j]==best)
526 a, b=chosen
527 println(
528 "$label $(percentage(best,window*window))% at $(pointtext(x+(a-1)*window,y+(b-1)*window))",
529 )
530 end
531 else
532 c=Coverage(rects)
533 if command=="AREA"
534 println("area $layer: $(coveragearea(c))")
535 elseif command=="COVER"
536 println("cover $layer >= $minimumcount: $(coveragearea(c,minimumcount))")
537 elseif command=="PERIMETER"
538 println("perimeter $layer: $(perimeter(c))")
539 elseif command=="EMPTY"
540 box=largestempty(c)
541 println(
542 "empty $layer: ",
543 box===nothing ? "none" :
544 "$(boxtext(box)) area $((box[3]-box[1])*(box[4]-box[2]))",
545 )
546 elseif command=="HOLES"
547 found=holes(c)
548 println("holes $layer: $(length(found))")
549 for (i, (area, box)) in enumerate(found)
550 println("hole $i: area $area, bbox $(boxtext(box))")
551 end
552 elseif command=="MERGE"
553 found=pieces(c)
554 report("merge $layer: $(length(found)) pieces", boxtext.(found))
555 end
556 end
557 end
558 command in ("ADD", "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE") &&
559 push!(w.history, (before, line))
560end

566 execute!(workbench, readline())
567 catch e
568 e isa CommandError || rethrow()
569 println("error: $(e.message)")
570 end
571 end
572end
