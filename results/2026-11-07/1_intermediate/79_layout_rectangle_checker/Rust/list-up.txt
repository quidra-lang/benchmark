    fn shape(x1: i64, y1: i64, x2: i64, y2: i64) -> Self {
        Self {
            x1,
            y1,
            x2,
            y2,
        }
    fn width(&self) -> i64 {
        self.x2 - self.x1
    }

    fn height(&self) -> i64 {
        self.y2 - self.y1
    }
impl Relation {
    fn direction(&self) -> &str {
        if self.kind == 2 {
            "corner"
        } else if self.ox < 0 && self.oy < 0 {
            "diagonal"
        } else if self.ox < 0 {
            "x"
        } else {
            "y"
        }
    }
fn relation(a: &Rect, b: &Rect) -> Relation {
    let common = Rect::shape(
        a.x1.max(b.x1),
        a.y1.max(b.y1),
        a.x2.min(b.x2),
        a.y2.min(b.y2),
    );
    let ox = common.width();
    let oy = common.height();
    let kind = if ox > 0 && oy > 0 {
        0
    } else if ox >= 0 && oy >= 0 && ox + oy > 0 {
        1
    } else if ox == 0 && oy == 0 {
        2
    } else {
        3
    };
    Relation {
        common,
        ox,
        oy,
        kind,
        gap: 0.max(-ox).max(-oy),
    }
}
fn spacing(rs: &[Rect], s: i64) -> Vec<(String, String, i64, String)> {
    let mut found = Vec::new();
    for i in 0..rs.len() {
        for j in i + 1..rs.len() {
            let r = relation(&rs[i], &rs[j]);
            if r.kind == 2 || r.kind == 3 && r.gap < s {
                found.push((
                    rs[i].name.clone(),
                    rs[j].name.clone(),
                    r.gap,
                    r.direction().into(),
                ));
            }
        }
    }
    found.sort_by(|a, b| (a.2, &a.0, &a.1).cmp(&(b.2, &b.0, &b.1)));
    found
}

    format!("{} {} gap {} {}", v.0, v.1, v.2, v.3)
