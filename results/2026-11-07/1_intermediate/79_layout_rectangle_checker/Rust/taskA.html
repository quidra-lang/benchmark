84 if [self.x1, self.y1, self.x2, self.y2]

88 Err(error("out of bounds"))

95struct Relation {
96 common: Rect,
97 ox: i64,
98 oy: i64,
99 kind: usize,
100 gap: i64,
101}
102
103impl Relation {
104 fn direction(&self) -> &str {
105 if self.kind == 2 {
106 "corner"
107 } else if self.ox < 0 && self.oy < 0 {
108 "diagonal"
109 } else if self.ox < 0 {
110 "x"
111 } else {
112 "y"
113 }
114 }
115}
116
117fn relation(a: &Rect, b: &Rect) -> Relation {
118 let common = Rect::shape(
119 a.x1.max(b.x1),
120 a.y1.max(b.y1),
121 a.x2.min(b.x2),
122 a.y2.min(b.y2),
123 );
124 let ox = common.width();
125 let oy = common.height();
126 let kind = if ox > 0 && oy > 0 {
127 0
128 } else if ox >= 0 && oy >= 0 && ox + oy > 0 {
129 1
130 } else if ox == 0 && oy == 0 {
131 2
132 } else {
133 3
134 };
135 Relation {
136 common,
137 ox,
138 oy,
139 kind,
140 gap: 0.max(-ox).max(-oy),
141 }
142}
143
144fn connected(a: &Rect, b: &Rect) -> bool {
145 relation(a, b).kind <= 1
146}

151 rs.iter().map(|r| r.y1).min().unwrap(),
152 rs.iter().map(|r| r.x2).max().unwrap(),
153 rs.iter().map(|r| r.y2).max().unwrap(),
154 )
155}

157struct Grid {
158 xs: Vec<i64>,
159 ys: Vec<i64>,
160 count: Vec<Vec<usize>>,
161}
162
163impl Grid {
164 fn new(rs: &[Rect]) -> Self {
165 let xs: Vec<i64> = rs
166 .iter()
167 .flat_map(|r| [r.x1, r.x2])
168 .collect::<BTreeSet<_>>()
169 .into_iter()
170 .collect();
171 let ys: Vec<i64> = rs
172 .iter()
173 .flat_map(|r| [r.y1, r.y2])
174 .collect::<BTreeSet<_>>()
175 .into_iter()
176 .collect();
177 let count = (0..ys.len() - 1)
178 .map(|y| {
179 (0..xs.len() - 1)
180 .map(|x| {
181 rs.iter()
182 .filter(|r| {
183 r.x1 <= xs[x] && r.x2 > xs[x] && r.y1 <= ys[y] && r.y2 > ys[y]
184 })
185 .count()
186 })
187 .collect()
188 })
189 .collect();
190 Self { xs, ys, count }
191 }
192
193 fn nx(&self) -> usize {
194 self.xs.len() - 1
195 }
196
197 fn ny(&self) -> usize {
198 self.ys.len() - 1
199 }
200
201 fn area(&self, k: usize, window: Option<&Rect>) -> i64 {
202 let mut sum = 0;
203 for y in 0..self.ny() {
204 for x in 0..self.nx() {
205 if self.count[y][x] >= k {
206 let x1 = self.xs[x].max(window.map_or(self.xs[x], |r| r.x1));
207 let x2 = self.xs[x + 1].min(window.map_or(self.xs[x + 1], |r| r.x2));
208 let y1 = self.ys[y].max(window.map_or(self.ys[y], |r| r.y1));
209 let y2 = self.ys[y + 1].min(window.map_or(self.ys[y + 1], |r| r.y2));
210 sum += (x2 - x1).max(0) * (y2 - y1).max(0);
211 }
212 }
213 }
214 sum
215 }
216
217 fn perimeter(&self) -> i64 {
218 let mut sum = 0;
219 for y in 0..self.ny() {
220 for x in 0..self.nx() {
221 if self.count[y][x] > 0 {
222 if x == 0 || self.count[y][x - 1] == 0 {
223 sum += self.ys[y + 1] - self.ys[y];
224 }
225 if x + 1 == self.nx() || self.count[y][x + 1] == 0 {
226 sum += self.ys[y + 1] - self.ys[y];
227 }
228 if y == 0 || self.count[y - 1][x] == 0 {
229 sum += self.xs[x + 1] - self.xs[x];
230 }
231 if y + 1 == self.ny() || self.count[y + 1][x] == 0 {
232 sum += self.xs[x + 1] - self.xs[x];
233 }
234 }
235 }
236 }
237 sum
238 }
239
240 fn empty(&self) -> Option<Rect> {
241 let mut best: Option<Rect> = None;
242 for bottom in 0..self.ny() {
243 let mut free = vec![true; self.nx()];
244 for top in bottom..self.ny() {
245 for x in 0..self.nx() {
246 free[x] &= self.count[top][x] == 0;
247 }
248 let mut x = 0;
249 while x < self.nx() {
250 if !free[x] {
251 x += 1;
252 continue;
253 }
254 let start = x;
255 while x < self.nx() && free[x] {
256 x += 1;
257 }
258 let r = Rect::shape(
259 self.xs[start],
260 self.ys[bottom],
261 self.xs[x],
262 self.ys[top + 1],
263 );
264 if best.as_ref().is_none_or(|old| {
265 (-r.area(), r.x1, r.y1, r.width())
266 < (-old.area(), old.x1, old.y1, old.width())
267 }) {
268 best = Some(r);
269 }
270 }
271 }
272 }
273 best
274 }

276 fn holes(&self) -> Vec<(i64, Rect)> {
277 let mut seen = vec![vec![false; self.nx()]; self.ny()];
278 let mut result = Vec::new();
279 for y in 0..self.ny() {
280 for x in 0..self.nx() {
281 if self.count[y][x] > 0 || seen[y][x] {
282 continue;
283 }
284 let mut queue = VecDeque::from(vec![(x, y)]);
285 seen[y][x] = true;
286 let mut boundary = false;
287 let mut area = 0;
288 let mut r = Rect::shape(self.xs[x], self.ys[y], self.xs[x + 1], self.ys[y + 1]);
289 let mut anchor = (self.ys[y], self.xs[x]);
290 while let Some((cx, cy)) = queue.pop_front() {
291 boundary |= cx == 0 || cy == 0 || cx + 1 == self.nx() || cy + 1 == self.ny();
292 area += (self.xs[cx + 1] - self.xs[cx]) * (self.ys[cy + 1] - self.ys[cy]);
293 r.x1 = r.x1.min(self.xs[cx]);
294 r.y1 = r.y1.min(self.ys[cy]);
295 r.x2 = r.x2.max(self.xs[cx + 1]);
296 r.y2 = r.y2.max(self.ys[cy + 1]);
297 anchor = anchor.min((self.ys[cy], self.xs[cx]));
298 for (dx, dy) in [(-1isize, 0isize), (1, 0), (0, -1), (0, 1)].iter().copied() {
299 let xx = cx as isize + dx;
300 let yy = cy as isize + dy;
301 if xx >= 0
302 && yy >= 0
303 && (xx as usize) < self.nx()
304 && (yy as usize) < self.ny()
305 {
306 let (xx, yy) = (xx as usize, yy as usize);
307 if !seen[yy][xx] && self.count[yy][xx] == 0 {
308 seen[yy][xx] = true;
309 queue.push_back((xx, yy));
310 }
311 }
312 }
313 }
314 if !boundary {
315 result.push((anchor, area, r));
316 }
317 }
318 }
319 result.sort_by_key(|r| r.0);
320 result.into_iter().map(|(_, area, r)| (area, r)).collect()
321 }

323 fn pieces(&self) -> Vec<Rect> {
324 let mut active: BTreeMap<(i64, i64), Rect> = BTreeMap::new();
325 let mut result = Vec::new();
326 for y in 0..self.ny() {
327 let mut next = BTreeMap::new();
328 let mut x = 0;
329 while x < self.nx() {
330 if self.count[y][x] == 0 {
331 x += 1;
332 continue;
333 }
334 let start = x;
335 while x < self.nx() && self.count[y][x] > 0 {
336 x += 1;
337 }
338 let key = (self.xs[start], self.xs[x]);
339 let mut r = active
340 .remove(&key)
341 .unwrap_or_else(|| Rect::shape(key.0, self.ys[y], key.1, self.ys[y + 1]));
342 r.y2 = self.ys[y + 1];
343 next.insert(key, r);
344 }
345 result.extend(active.into_values());
346 active = next;
347 }
348 result.extend(active.into_values());
349 result.sort_by_key(|r| (r.y1, r.x1));
350 result
351 }
352}

354fn spacing(rs: &[Rect], s: i64) -> Vec<(String, String, i64, String)> {
355 let mut found = Vec::new();
356 for i in 0..rs.len() {
357 for j in i + 1..rs.len() {
358 let r = relation(&rs[i], &rs[j]);
359 if r.kind == 2 || r.kind == 3 && r.gap < s {
360 found.push((
361 rs[i].name.clone(),
362 rs[j].name.clone(),
363 r.gap,
364 r.direction().into(),
365 ));
366 }
367 }
368 }
369 found.sort_by(|a, b| (a.2, &a.0, &a.1).cmp(&(b.2, &b.0, &b.1)));
370 found
371}

377fn nets(rs: &[Rect]) -> Vec<Vec<Rect>> {
378 let mut seen = BTreeSet::new();
379 let mut groups = Vec::new();
380 for r in rs {
381 if !seen.insert(r.name.clone()) {
382 continue;
383 }
384 let mut queue = vec![r.clone()];
385 let mut at = 0;
386 while at < queue.len() {
387 let a = queue[at].clone();
388 at += 1;
389 for b in rs {
390 if !seen.contains(&b.name) && connected(&a, b) {
391 seen.insert(b.name.clone());
392 queue.push(b.clone());
393 }
394 }
395 }
396 queue.sort_by(|a, b| a.name.cmp(&b.name));
397 groups.push(queue);
398 }
399 groups
400}

417 self.rectangles
418 .values()
419 .filter(|r| r.layer == name)

421 .collect()
422 }

427 Err(error(format!("no rectangles on {}", name)))

429 Ok(rs)
430 }

433 fn command(&mut self, line: &str) -> Answer {
434 let w: Vec<&str> = line.split(' ').collect();
435 let c = w[0];
436 let n = |i: usize| w[i].parse::<i64>().unwrap();
437 let mut out = Vec::new();
438 if ["ADD", "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE"].contains(&c) {
439 let mut next = self.rectangles.clone();
440 let name = w[1];
441 match c {
442 "ADD" => {
443 if self.rectangles.contains_key(name) {
444 return Err(error(format!("duplicate rectangle {}", name)));
445 }
446 let mut r = Rect::shape(n(3), n(4), n(5), n(6));
447 r.name = name.into();
448 r.layer = w[2].into();
449 if r.width() <= 0 || r.height() <= 0 {
450 return Err(error("invalid rectangle"));
451 }
452 r.check_bounds()?;
453 out.push(format!("added {} {} {}", name, r.layer, r.text()));
454 next.insert(name.into(), r);
455 }
456 "COPY" => {
457 let old = self.get(name)?;
458 if self.rectangles.contains_key(w[2]) {
459 return Err(error(format!("duplicate rectangle {}", w[2])));
460 }
461 let mut r = old.moved(n(3), n(4));
462 r.name = w[2].into();
463 r.check_bounds()?;
464 out.push(format!("copied {} to {} {}", name, r.name, r.text()));
465 next.insert(r.name.clone(), r);
466 }
467 "MOVE" => {
468 let r = self.get(name)?.moved(n(2), n(3));
469 r.check_bounds()?;
470 out.push(format!("moved {} to {}", name, r.text()));
471 next.insert(name.into(), r);
472 }
473 "RESIZE" => {
474 let mut r = self.get(name)?;
475 if n(2) <= 0 || n(3) <= 0 {
476 return Err(error("invalid size"));
477 }
478 r.x2 = r.x1 + n(2);
479 r.y2 = r.y1 + n(3);
480 r.check_bounds()?;
481 out.push(format!("resized {} to {}", name, r.text()));
482 next.insert(name.into(), r);
483 }
484 "SPLIT" => {
485 let old = self.get(name)?;
486 if self.rectangles.contains_key(w[4]) {
487 return Err(error(format!("duplicate rectangle {}", w[4])));
488 }
489 let p = n(3);
490 let x = w[2] == "x";
491 if p <= (if x { old.x1 } else { old.y1 })
492 || p >= (if x { old.x2 } else { old.y2 })
493 {
494 return Err(error(format!("cut outside {}", name)));
495 }
496 let mut a = old.clone();
497 let mut b = old;
498 b.name = w[4].into();
499 if x {
500 a.x2 = p;
501 b.x1 = p;
502 } else {
503 a.y2 = p;
504 b.y1 = p;
505 }
506 out.push(format!(
507 "split {} at {}={}: {} {}, {} {}",
508 name,
509 w[2],
510 p,
511 name,
512 a.text(),
513 b.name,
514 b.text()
515 ));
516 next.insert(name.into(), a);
517 next.insert(b.name.clone(), b);
518 }
519 "LAYER" => {
520 let mut r = self.get(name)?;
521 if r.layer == w[2] {
522 return Err(error(format!("already on {}", w[2])));
523 }
524 out.push(format!("layer {} {} -> {}", name, r.layer, w[2]));
525 r.layer = w[2].into();
526 next.insert(name.into(), r);
527 }
528 _ => {
529 self.get(name)?;
530 next.remove(name);
531 out.push(format!("deleted {}", name));
532 }
533 }
534 self.history
535 .push((std::mem::replace(&mut self.rectangles, next), line.into()));
536 return Ok(out);
537 }
538 match c {
539 "UNDO" => {
540 let (old, text) = self.history.pop().ok_or_else(|| error("nothing to undo"))?;
541 self.rectangles = old;
542 return Ok(vec![format!("undo: {}", text)]);
543 }
544 "LIST" => {
545 out.push(format!("rectangles: {}", self.rectangles.len()));
546 out.extend(self.rectangles.values().map(|r| {
547 format!(
548 "{} {} {} {}x{}",
549 r.name,
550 r.layer,
551 r.text(),
552 r.width(),
553 r.height()
554 )
555 }));
556 return Ok(out);
557 }
558 "INFO" => {
559 let a = self.get(w[1])?;
560 let others: Vec<&Rect> = self
561 .rectangles
562 .values()
563 .filter(|r| r.name != a.name)
564 .collect();
565 out.push(format!(
566 "{} {} {} {}x{} area {}",
567 a.name,
568 a.layer,
569 a.text(),
570 a.width(),
571 a.height(),
572 a.area()
573 ));
574 for (label, kind) in [("overlaps", 0), ("touches", 1), ("corners", 2)]
575 .iter()
576 .copied()
577 {
578 out.push(format!(
579 "{}: {}",
580 label,
581 names(
582 others
583 .iter()
584 .filter(|r| r.layer == a.layer && relation(&a, r).kind == kind)
585 .map(|r| r.name.clone())
586 .collect()
587 )
588 ));
589 }
590 out.push(format!(
591 "encloses: {}",
592 names(
593 others
594 .iter()
595 .filter(|r| a.contains(r))
596 .map(|r| r.name.clone())
597 .collect()
598 )
599 ));
600 out.push(format!(
601 "enclosed by: {}",
602 names(
603 others
604 .iter()
605 .filter(|r| r.contains(&a))
606 .map(|r| r.name.clone())
607 .collect()
608 )
609 ));
610 return Ok(out);
611 }
612 "LAYERS" => {
613 let layers = self
614 .rectangles
615 .values()
616 .map(|r| r.layer.clone())
617 .collect::<BTreeSet<_>>();
618 out.push(format!("layers: {}", layers.len()));
619 for l in layers {
620 let rs = self.layer(&l);
621 out.push(format!(
622 "{}: {} rectangles, area {}, nets {}",
623 l,
624 rs.len(),
625 Grid::new(&rs).area(1, None),
626 nets(&rs).len()
627 ));
628 }
629 return Ok(out);
630 }
631 "INTERSECT" | "ADJACENT" | "ENCLOSE" | "PATH" => {
632 let a = self.get(w[1])?;
633 let b = self.get(w[2])?;
634 if a.name == b.name {
635 return Err(error(format!("same rectangle {}", a.name)));
636 }
637 let r = relation(&a, &b);
638 let q = &r.common;
639 if c == "INTERSECT" {
640 return Ok(vec![format!(
641 "intersect {} {}: {}",
642 a.name,
643 b.name,
644 match r.kind {
645 0 => format!("{} area {}", q.text(), q.area()),
646 1 => format!("segment {} length {}", q.text(), r.ox + r.oy),
647 2 => format!("point ({},{})", q.x1, q.y1),
648 _ => format!("none, gap {}", r.gap),
649 }
650 )]);
651 }
652 if c == "ADJACENT" {
653 let d = if r.ox == 0 {
654 if a.x2 == q.x1 { "west" } else { "east" }
655 } else if a.y2 == q.y1 {
656 "south"
657 } else {
658 "north"
659 };
660 return Ok(vec![format!(
661 "adjacent {} {}: {}",
662 a.name,
663 b.name,
664 match r.kind {
665 0 => "overlapping".into(),
666 1 => format!(
667 "{} is {} of {}, edge {}, length {}",
668 a.name,
669 d,
670 b.name,
671 q.text(),
672 r.ox + r.oy
673 ),
674 2 => format!("corner only at ({},{})", q.x1, q.y1),
675 _ => "not adjacent".into(),
676 }
677 )]);
678 }
679 if c == "ENCLOSE" {
680 let m = a.margins(&b);
681 return Ok(vec![format!(
682 "enclose {} {}: {}, margins left {} bottom {} right {} top {}{}",
683 a.name,
684 b.name,
685 if a.contains(&b) { "yes" } else { "no" },
686 m[0],
687 m[1],
688 m[2],
689 m[3],
690 if a.contains(&b) {
691 format!(", min {}", m.iter().min().unwrap())
692 } else {
693 String::new()
694 }
695 )]);
696 }
697 if a.layer != b.layer {
698 return Err(error("different layers"));
699 }
700 let rs = self.layer(&a.layer);
701 let mut seen = BTreeSet::from([a.name.clone()]);
702 let mut previous = BTreeMap::new();
703 let mut queue = VecDeque::from(vec![a.clone()]);
704 while let Some(p) = queue.pop_front() {
705 for q in &rs {
706 if !seen.contains(&q.name) && connected(&p, q) {
707 seen.insert(q.name.clone());
708 previous.insert(q.name.clone(), p.name.clone());
709 queue.push_back(q.clone());
710 }
711 }
712 }
713 if !seen.contains(&b.name) {
714 return Ok(vec![format!("path {} {}: none", a.name, b.name)]);
715 }
716 let mut path = vec![b.name.clone()];
717 while path.last().unwrap() != &a.name {
718 path.push(previous[path.last().unwrap()].clone());
719 }
720 path.reverse();
721 return Ok(vec![format!(
722 "path {} {}: {} ({} steps)",
723 a.name,
724 b.name,
725 path.join(" -> "),
726 path.len() - 1
727 )]);
728 }
729 "AT" => {
730 let (x, y) = (n(1), n(2));
731 let found: Vec<String> = self
732 .rectangles
733 .values()
734 .filter(|r| r.x1 <= x && x <= r.x2 && r.y1 <= y && y <= r.y2)
735 .map(|r| {
736 let xx = x == r.x1 || x == r.x2;
737 let yy = y == r.y1 || y == r.y2;
738 format!(
739 "{} {} {}",
740 r.name,
741 r.layer,
742 if xx && yy {
743 "corner"
744 } else if xx || yy {
745 "edge"
746 } else {
747 "inside"
748 }
749 )
750 })
751 .collect();
752 return Ok(vec![format!(
753 "at ({},{}): {}",
754 x,
755 y,
756 if found.is_empty() {
757 "none".into()
758 } else {
759 found.join(", ")
760 }
761 )]);
762 }
763 "NEAREST" => {
764 let a = self.get(w[1])?;
765 let rs = self.layer(&a.layer);
766 let b = rs
767 .iter()
768 .filter(|r| r.name != a.name)
769 .min_by_key(|r| (relation(&a, r).gap, r.name.clone()));
770 return Ok(vec![format!(
771 "nearest {}: {}",
772 a.name,
773 b.map_or("none".into(), |r| format!(
774 "{} gap {}",
775 r.name,
776 relation(&a, r).gap
777 ))
778 )]);
779 }
780 "OVERLAPS" => {
781 let rs = if w.len() == 2 {
782 self.checked_layer(w[1])?
783 } else {
784 self.rectangles.values().cloned().collect()
785 };
786 let mut found = Vec::new();
787 for i in 0..rs.len() {
788 for j in i + 1..rs.len() {
789 if rs[i].layer == rs[j].layer {
790 let r = relation(&rs[i], &rs[j]);
791 if r.kind == 0 {
792 found.push((&rs[i], &rs[j], r.common));
793 }
794 }
795 }
796 }
797 found.sort_by(|a, b| {
798 (&a.0.layer, &a.0.name, &a.1.name).cmp(&(&b.0.layer, &b.0.name, &b.1.name))
799 });
800 out.push(format!(
801 "overlaps {}: {}",
802 if w.len() == 1 { "all layers" } else { w[1] },
803 found.len()
804 ));
805 for (a, b, r) in found {
806 out.push(format!(
807 "{}{} {} {} area {}",
808 if w.len() == 1 {
809 format!("{} ", a.layer)
810 } else {
811 String::new()
812 },
813 a.name,
814 b.name,
815 r.text(),
816 r.area()
817 ));
818 }
819 return Ok(out);
820 }
821 "RULE" => {
822 let kind = w[1].to_lowercase();
823 if kind == "enclose" && w[2] == w[3] {
824 return Err(error(format!("same layer {}", w[2])));
825 }
826 let outer = if kind == "enclose" { w[3] } else { "" };
827 let v = n(if kind == "enclose" { 4 } else { 3 });
828 self.rules
829 .insert((kind.clone(), w[2].into(), outer.into()), v);
830 return Ok(vec![format!(
831 "rule {} {}{} {}",
832 kind,
833 w[2],
834 if outer.is_empty() {
835 String::new()
836 } else {
837 format!(" {}", outer)
838 },
839 v
840 )]);
841 }
842 "CHECK" => {
843 if self.rules.is_empty() {
844 return Err(error("no rules"));
845 }
846 let mut total = 0;
847 for ((kind, inner, outer), &v) in &self.rules {
848 let rs = self.layer(inner);
849 let found: Vec<String> = match kind.as_str() {
850 "spacing" => spacing(&rs, v).iter().map(violation).collect(),
851 "width" => rs
852 .iter()
853 .filter(|r| r.width().min(r.height()) < v)
854 .map(|r| format!("{} width {}", r.name, r.width().min(r.height())))
855 .collect(),
856 _ => {
857 let outer = self.layer(outer);
858 rs.iter()
859 .filter_map(|r| {
860 let best = outer
861 .iter()
862 .map(|a| *a.margins(r).iter().min().unwrap())
863 .max();
864 if best.is_none_or(|b| b < v) {
865 Some(format!(
866 "{} best margin {}",
867 r.name,
868 best.map_or("none".into(), |b| b.to_string())
869 ))
870 } else {
871 None
872 }
873 })
874 .collect()
875 }
876 };
877 total += found.len();
878 out.push(format!(
879 "{} {}{} {}: {} violations",
880 kind,
881 inner,
882 if kind == "enclose" {
883 format!(" {}", outer)
884 } else {
885 String::new()
886 },
887 v,
888 found.len()
889 ));
890 out.extend(found);
891 }
892 out.push(format!(
893 "check: {} violations in {} rules",
894 total,
895 self.rules.len()
896 ));
897 return Ok(out);
898 }
899 _ => {}
900 }
901 let rs = self.checked_layer(if c == "COVER" { w[2] } else { w[1] })?;
902 let l = &rs[0].layer;
903 let g = Grid::new(&rs);
904 let b = bbox(&rs);
905 match c {
906 "AREA" => Ok(vec![format!("area {}: {}", l, g.area(1, None))]),
907 "COVER" => Ok(vec![format!(
908 "cover {} >= {}: {}",
909 l,
910 w[1],
911 g.area(n(1) as usize, None)
912 )]),
913 "PERIMETER" => Ok(vec![format!("perimeter {}: {}", l, g.perimeter())]),
914 "NETS" => {
915 let groups = nets(&rs);
916 out.push(format!("nets {}: {}", l, groups.len()));
917 for (i, group) in groups.iter().enumerate() {
918 out.push(format!(
919 "net {}: {} area {}",
920 i + 1,
921 names(group.iter().map(|r| r.name.clone()).collect()),
922 Grid::new(group).area(1, None)
923 ));
924 }
925 Ok(out)
926 }
927 "DRC" => {
928 let violations = spacing(&rs, n(2));
929 out.push(format!(
930 "drc {} spacing {}: {} violations",
931 l,
932 w[2],
933 violations.len()
934 ));
935 out.extend(violations.iter().map(violation));
936 Ok(out)
937 }
938 "BBOX" => Ok(vec![format!(
939 "bbox {}: {} {}x{}, {} rectangles, density {}%",
940 l,
941 b.text(),
942 b.width(),
943 b.height(),
944 rs.len(),
945 decimal(100 * g.area(1, None), b.area())
946 )]),
947 "DENSITY" => {
948 let size = n(2);
949 let cols = (b.width() + size - 1) / size;
950 let rows = (b.height() + size - 1) / size;
951 if cols > 20 || rows > 20 {
952 return Err(error("too many windows"));
953 }
954 let values: Vec<Vec<i64>> = (0..rows)
955 .map(|y| {
956 (0..cols)
957 .map(|x| {
958 let xx = b.x1 + x * size;
959 let yy = b.y1 + y * size;
960 g.area(1, Some(&Rect::shape(xx, yy, xx + size, yy + size)))
961 })
962 .collect()
963 })
964 .collect();
965 let denominator = size * size;
966 out.push(format!("density {} window {}: {}x{}", l, size, cols, rows));
967 for row in values.iter().rev() {
968 out.push(
969 row.iter()
970 .map(|v| (100 * v / denominator).to_string())
971 .collect::<Vec<_>>()
972 .join(" "),
973 );
974 }
975 let (mut min_x, mut min_y, mut max_x, mut max_y) = (0, 0, 0, 0);
976 for y in 0..rows as usize {
977 for x in 0..cols as usize {
978 if values[y][x] < values[min_y][min_x] {
979 min_x = x;
980 min_y = y;
981 }
982 if values[y][x] > values[max_y][max_x] {
983 max_x = x;
984 max_y = y;
985 }
986 }
987 }
988 out.push(format!(
989 "min {}% at ({},{})",
990 decimal(100 * values[min_y][min_x], denominator),
991 b.x1 + min_x as i64 * size,
992 b.y1 + min_y as i64 * size
993 ));
994 out.push(format!(
995 "max {}% at ({},{})",
996 decimal(100 * values[max_y][max_x], denominator),
997 b.x1 + max_x as i64 * size,
998 b.y1 + max_y as i64 * size
999 ));
1000 Ok(out)
1001 }
1002 "RENDER" => {
1003 let r = Rect::shape(n(2), n(3), n(4), n(5));
1004 if r.width() <= 0 || r.height() <= 0 {
1005 return Err(error("invalid window"));
1006 }
1007 if r.width() > 60 || r.height() > 40 {
1008 return Err(error("window too large"));
1009 }
1010 out.push(format!("render {} {}", l, r.text()));
1011 for y in (r.y1..r.y2).rev() {
1012 out.push(
1013 (r.x1..r.x2)
1014 .map(|x| {
1015 let count = rs
1016 .iter()
1017 .filter(|a| a.x1 <= x && x < a.x2 && a.y1 <= y && y < a.y2)
1018 .count();
1019 if count == 0 {
1020 '.'
1021 } else if count >= 10 {
1022 '*'
1023 } else {
1024 (b'0' + count as u8) as char
1025 }
1026 })
1027 .collect(),
1028 );
1029 }
1030 Ok(out)
1031 }
1032 "EMPTY" => Ok(vec![format!(
1033 "empty {}: {}",
1034 l,
1035 g.empty()
1036 .map_or("none".into(), |r| format!("{} area {}", r.text(), r.area()))
1037 )]),
1038 "HOLES" => {
1039 let holes = g.holes();
1040 out.push(format!("holes {}: {}", l, holes.len()));
1041 out.extend(holes.iter().enumerate().map(|(i, (area, r))| {
1042 format!("hole {}: area {}, bbox {}", i + 1, area, r.text())
1043 }));
1044 Ok(out)
1045 }
1046 "MERGE" => {
1047 let pieces = g.pieces();
1048 out.push(format!("merge {}: {} pieces", l, pieces.len()));
1049 out.extend(pieces.iter().map(Rect::text));
1050 Ok(out)
1051 }
1052 _ => unreachable!(),
1053 }
1054 }
1055}

1058 let mut input = String::new();
1059 io::stdin().read_to_string(&mut input).unwrap();
1060 let mut lines = input.lines();

1062 let mut layout = Layout {

1064 history: Vec::new(),
1065 rules: BTreeMap::new(),
1066 };
1067 for line in lines.take(count) {
1068 match layout.command(line) {
1069 Ok(lines) => {
1070 for line in lines {
1071 println!("{}", line);
1072 }
1073 }
1074 Err(e) => println!("{}", e),
1075 }
1076 }
