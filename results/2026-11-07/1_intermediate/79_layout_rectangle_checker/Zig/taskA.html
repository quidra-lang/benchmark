35fn words(gpa: Allocator, text: Text) Allocator.Error![]Text {
36 var result: std.ArrayList(Text) = .empty;
37 var iterator = std.mem.tokenizeScalar(u8, text, ' ');
38 while (iterator.next()) |word| try result.append(gpa, word);
39 return result.items;
40}
41fn join(gpa: Allocator, list: []const Text, separator: Text) Allocator.Error!Text {
42 var out: Builder = .{ .gpa = gpa };
43 for (list, 0..) |item, i| {
44 if (i > 0) try out.add(separator);
45 try out.add(item);
46 }
47 return out.bytes.items;
48}
49const Geometry = struct { x1: Int, y1: Int, x2: Int, y2: Int };
50const Rectangle = struct { name: Text, layer: Text, r: Geometry };
51const RelationKind = enum { overlap, edge, corner, separate };
52const Relation = struct { r: Geometry, ox: Int, oy: Int, gap: Int, kind: RelationKind, direction: Text = "" };
53fn notation(gpa: Allocator, r: Geometry) Allocator.Error!Text {
54 return std.fmt.allocPrint(gpa, "({d},{d})-({d},{d})", .{ r.x1, r.y1, r.x2, r.y2 });
55}
56fn point(gpa: Allocator, x: Int, y: Int) Allocator.Error!Text {
57 return std.fmt.allocPrint(gpa, "({d},{d})", .{ x, y });
58}
59fn area(r: Geometry) Int {
60 return (r.x2 - r.x1) * (r.y2 - r.y1);
61}
62fn encloses(a: Geometry, b: Geometry) bool {
63 return a.x1 <= b.x1 and a.y1 <= b.y1 and a.x2 >= b.x2 and a.y2 >= b.y2;
64}
65fn compare(a: Geometry, b: Geometry) Relation {
66 const r: Geometry = .{ .x1 = @max(a.x1, b.x1), .y1 = @max(a.y1, b.y1), .x2 = @min(a.x2, b.x2), .y2 = @min(a.y2, b.y2) };
67 const ox = r.x2 - r.x1;
68 const oy = r.y2 - r.y1;
69 const kind: RelationKind = if (ox > 0 and oy > 0) .overlap else if (ox == 0 and oy == 0) .corner else if (ox >= 0 and oy >= 0) .edge else .separate;
70 return .{ .r = r, .ox = ox, .oy = oy, .gap = @max(0, @max(-ox, -oy)), .kind = kind, .direction = if (ox < 0 and oy < 0) "diagonal" else if (ox < 0) "x" else "y" };
71}
72fn connected(a: Rectangle, b: Rectangle) bool {
73 const r = compare(a.r, b.r);
74 return equal(a.layer, b.layer) and (r.kind == .overlap or r.kind == .edge);
75}
76fn margins(a: Geometry, b: Geometry) [4]Int {
77 return .{ b.x1 - a.x1, b.y1 - a.y1, a.x2 - b.x2, a.y2 - b.y2 };
78}
79fn minimum(values: [4]Int) Int {
80 var result = values[0];
81 for (values[1..]) |value| result = @min(result, value);
82 return result;
83}
84fn decimal(gpa: Allocator, numerator: Int, denominator: Int) Allocator.Error!Text {
85 const h = @divTrunc(numerator * 100, denominator);
86 return std.fmt.allocPrint(gpa, "{d}.{d:0>2}", .{ @divTrunc(h, 100), @mod(h, 100) });
87}
88fn bounds(list: []const Rectangle) Geometry {
89 var r = list[0].r;
90 for (list[1..]) |item| {
91 r.x1 = @min(r.x1, item.r.x1);
92 r.y1 = @min(r.y1, item.r.y1);
93 r.x2 = @max(r.x2, item.r.x2);
94 r.y2 = @max(r.y2, item.r.y2);
95 }
96 return r;
97}
98fn rectangleLess(_: void, a: Rectangle, b: Rectangle) bool {
99 return std.mem.order(u8, a.name, b.name) == .lt;
100}
101fn intLess(_: void, a: Int, b: Int) bool {
102 return a < b;
103}
104fn distinct(gpa: Allocator, list: []Int) Allocator.Error![]Int {
105 std.mem.sort(Int, list, {}, intLess);
106 var result: std.ArrayList(Int) = .empty;
107 for (list) |value| if (result.items.len == 0 or value != result.items[result.items.len - 1]) try result.append(gpa, value);
108 return result.items;
109}
110const Cell = struct { x: usize, y: usize };
111const Hole = struct { r: Geometry, area: Int = 0, ax: Int, ay: Int };
112fn holeLess(_: void, a: Hole, b: Hole) bool {
113 return a.ay < b.ay or a.ay == b.ay and a.ax < b.ax;
114}
115fn pieceLess(_: void, a: Geometry, b: Geometry) bool {
116 return a.y1 < b.y1 or a.y1 == b.y1 and a.x1 < b.x1;
117}
118const Grid = struct {
119 xs: []const Int,
120 ys: []const Int,
121 counts: []i32,
122 fn w(self: Grid) usize {
123 return if (self.xs.len == 0) 0 else self.xs.len - 1;
124 }
125 fn h(self: Grid) usize {
126 return if (self.ys.len == 0) 0 else self.ys.len - 1;
127 }
128 fn count(self: Grid, x: usize, y: usize) i32 {
129 return self.counts[y * self.xs.len + x];
130 }
131 fn build(gpa: Allocator, list: []const Rectangle) Allocator.Error!Grid {
132 var xs: std.ArrayList(Int) = .empty;
133 var ys: std.ArrayList(Int) = .empty;
134 for (list) |item| {
135 try xs.appendSlice(gpa, &.{ item.r.x1, item.r.x2 });
136 try ys.appendSlice(gpa, &.{ item.r.y1, item.r.y2 });
137 }
138 const xx = try distinct(gpa, xs.items);
139 const yy = try distinct(gpa, ys.items);
140 const counts = try gpa.alloc(i32, xx.len * yy.len);
141 @memset(counts, 0);
142 for (list) |item| {
143 const a = std.mem.findScalar(Int, xx, item.r.x1).?;
144 const b = std.mem.findScalar(Int, yy, item.r.y1).?;
145 const c = std.mem.findScalar(Int, xx, item.r.x2).?;
146 const d = std.mem.findScalar(Int, yy, item.r.y2).?;
147 counts[b * xx.len + a] += 1;
148 counts[b * xx.len + c] -= 1;
149 counts[d * xx.len + a] -= 1;
150 counts[d * xx.len + c] += 1;
151 }
152 for (0..yy.len) |y| {
153 for (0..xx.len) |x| {
154 const index = y * xx.len + x;
155 if (y > 0) counts[index] += counts[index - xx.len];
156 if (x > 0) counts[index] += counts[index - 1];
157 if (y > 0 and x > 0) counts[index] -= counts[index - xx.len - 1];
158 }
159 }
160 return .{ .xs = xx, .ys = yy, .counts = counts };
161 }
162 fn covered(self: Grid, k: i32, window: ?Geometry) Int {
163 var total: Int = 0;
164 for (0..self.h()) |y| {
165 for (0..self.w()) |x| {
166 if (self.count(x, y) < k) continue;
167 var r: Geometry = .{ .x1 = self.xs[x], .y1 = self.ys[y], .x2 = self.xs[x + 1], .y2 = self.ys[y + 1] };
168 if (window) |clip| r = compare(r, clip).r;
169 if (r.x1 < r.x2 and r.y1 < r.y2) total += area(r);
170 }
171 }
172 return total;
173 }
174 fn perimeter(self: Grid) Int {
175 var total: Int = 0;
176 for (0..self.h()) |y| {
177 for (0..self.w()) |x| {
178 if (self.count(x, y) == 0) continue;
179 const width = self.xs[x + 1] - self.xs[x];
180 const height = self.ys[y + 1] - self.ys[y];
181 if (x == 0 or self.count(x - 1, y) == 0) total += height;
182 if (x + 1 == self.w() or self.count(x + 1, y) == 0) total += height;
183 if (y == 0 or self.count(x, y - 1) == 0) total += width;
184 if (y + 1 == self.h() or self.count(x, y + 1) == 0) total += width;
185 }
186 }
187 return total;
188 }
189 fn empty(self: Grid, gpa: Allocator) Allocator.Error!?Geometry {
190 var best: ?Geometry = null;
191 const blocked = try gpa.alloc(bool, self.h());
192 for (0..self.w()) |left| {
193 @memset(blocked, false);
194 for (left..self.w()) |right| {
195 for (blocked, 0..) |*value, y| value.* = value.* or self.count(right, y) > 0;
196 var y: usize = 0;
197 while (y < blocked.len) {
198 if (blocked[y]) {
199 y += 1;
200 continue;
201 }
202 const start = y;
203 while (y < blocked.len and !blocked[y]) y += 1;
204 const r: Geometry = .{ .x1 = self.xs[left], .y1 = self.ys[start], .x2 = self.xs[right + 1], .y2 = self.ys[y] };
205 if (best == null or area(r) > area(best.?) or area(r) == area(best.?) and (r.x1 < best.?.x1 or r.x1 == best.?.x1 and (r.y1 < best.?.y1 or r.y1 == best.?.y1 and r.x2 - r.x1 < best.?.x2 - best.?.x1))) best = r;
206 }
207 }
208 }
209 return best;
210 }
211 fn holes(self: Grid, gpa: Allocator) Allocator.Error![]Hole {
212 var result: std.ArrayList(Hole) = .empty;
213 const seen = try gpa.alloc(bool, self.counts.len);
214 @memset(seen, false);
215 for (0..self.h()) |y| {
216 for (0..self.w()) |x| {
217 const first = y * self.xs.len + x;
218 if (self.count(x, y) > 0 or seen[first]) continue;
219 var queue: std.ArrayList(Cell) = .empty;
220 try queue.append(gpa, .{ .x = x, .y = y });
221 seen[first] = true;
222 var outside = false;
223 var hole: Hole = .{ .r = .{ .x1 = self.xs[x], .y1 = self.ys[y], .x2 = self.xs[x + 1], .y2 = self.ys[y + 1] }, .ax = self.xs[x], .ay = self.ys[y] };
224 var i: usize = 0;
225 while (i < queue.items.len) : (i += 1) {
226 const p = queue.items[i];
227 if (p.x == 0 or p.y == 0 or p.x + 1 == self.w() or p.y + 1 == self.h()) outside = true;
228 hole.area += (self.xs[p.x + 1] - self.xs[p.x]) * (self.ys[p.y + 1] - self.ys[p.y]);
229 hole.r.x1 = @min(hole.r.x1, self.xs[p.x]);
230 hole.r.y1 = @min(hole.r.y1, self.ys[p.y]);
231 hole.r.x2 = @max(hole.r.x2, self.xs[p.x + 1]);
232 hole.r.y2 = @max(hole.r.y2, self.ys[p.y + 1]);
233 if (self.ys[p.y] < hole.ay or self.ys[p.y] == hole.ay and self.xs[p.x] < hole.ax) {
234 hole.ax = self.xs[p.x];
235 hole.ay = self.ys[p.y];
236 }
237 for ([_][2]Int{ .{ 1, 0 }, .{ -1, 0 }, .{ 0, 1 }, .{ 0, -1 } }) |delta| {
238 const a = si(p.x) + delta[0];
239 const b = si(p.y) + delta[1];
240 if (a < 0 or a >= si(self.w()) or b < 0 or b >= si(self.h())) continue;
241 const index = u(b) * self.xs.len + u(a);
242 if (!seen[index] and self.count(u(a), u(b)) == 0) {
243 seen[index] = true;
244 try queue.append(gpa, .{ .x = u(a), .y = u(b) });
245 }
246 }
247 }
248 if (!outside) try result.append(gpa, hole);
249 }
250 }
251 std.mem.sort(Hole, result.items, {}, holeLess);
252 return result.items;
253 }
254 fn pieces(self: Grid, gpa: Allocator) Allocator.Error![]Geometry {
255 var active: std.ArrayList(Geometry) = .empty;
256 var result: std.ArrayList(Geometry) = .empty;
257 for (0..self.h()) |y| {
258 var current: std.ArrayList(Geometry) = .empty;
259 const matched = try gpa.alloc(bool, active.items.len);
260 @memset(matched, false);
261 var x: usize = 0;
262 while (x < self.w()) {
263 if (self.count(x, y) == 0) {
264 x += 1;
265 continue;
266 }
267 const start = x;
268 while (x < self.w() and self.count(x, y) > 0) x += 1;
269 var r: Geometry = .{ .x1 = self.xs[start], .y1 = self.ys[y], .x2 = self.xs[x], .y2 = self.ys[y + 1] };
270 for (active.items, 0..) |old, i| if (old.x1 == r.x1 and old.x2 == r.x2) {
271 r.y1 = old.y1;
272 matched[i] = true;
273 break;
274 };
275 try current.append(gpa, r);
276 }
277 for (active.items, 0..) |r, i| if (!matched[i]) try result.append(gpa, r);
278 active = current;
279 }
280 try result.appendSlice(gpa, active.items);
281 std.mem.sort(Geometry, result.items, {}, pieceLess);
282 return result.items;
283 }
284};
285fn names(gpa: Allocator, list: []const Rectangle) Allocator.Error!Text {
286 var result: std.ArrayList(Text) = .empty;
287 for (list) |item| try result.append(gpa, item.name);
288 return if (result.items.len == 0) "none" else try join(gpa, result.items, " ");
289}
290fn nets(gpa: Allocator, list: []const Rectangle) Allocator.Error![][]Rectangle {
291 var result: std.ArrayList([]Rectangle) = .empty;
292 const seen = try gpa.alloc(bool, list.len);
293 @memset(seen, false);
294 for (list, 0..) |item, index| {
295 if (seen[index]) continue;
296 var group: std.ArrayList(Rectangle) = .empty;
297 try group.append(gpa, item);
298 seen[index] = true;
299 var i: usize = 0;
300 while (i < group.items.len) : (i += 1) {
301 for (list, 0..) |candidate, j| if (!seen[j] and connected(group.items[i], candidate)) {
302 seen[j] = true;
303 try group.append(gpa, candidate);
304 };
305 }
306 std.mem.sort(Rectangle, group.items, {}, rectangleLess);
307 try result.append(gpa, group.items);
308 }
309 return result.items;
310}
311const Violation = struct { a: Text, b: Text, gap: Int, direction: Text };
312fn violationLess(_: void, a: Violation, b: Violation) bool {
313 const first = std.mem.order(u8, a.a, b.a);
314 return a.gap < b.gap or a.gap == b.gap and (first == .lt or first == .eq and std.mem.order(u8, a.b, b.b) == .lt);
315}
316fn spacing(gpa: Allocator, list: []const Rectangle, s: Int) Allocator.Error![]Violation {
317 var result: std.ArrayList(Violation) = .empty;
318 for (list, 0..) |a, i| {
319 for (list[i + 1 ..]) |b| {
320 const r = compare(a.r, b.r);
321 if (r.kind == .corner or r.kind == .separate and r.gap < s) try result.append(gpa, .{ .a = a.name, .b = b.name, .gap = r.gap, .direction = if (r.kind == .corner) "corner" else r.direction });
322 }
323 }
324 std.mem.sort(Violation, result.items, {}, violationLess);
325 return result.items;
326}
327const RuleKind = enum { enclose, spacing, width };
328const Rule = struct { kind: RuleKind, inner: Text, outer: Text = "", value: Int };
329fn ruleLess(_: void, a: Rule, b: Rule) bool {
330 if (a.kind != b.kind) return @backingInt(a.kind) < @backingInt(b.kind);
331 const inner = std.mem.order(u8, a.inner, b.inner);
332 return inner == .lt or inner == .eq and std.mem.order(u8, a.outer, b.outer) == .lt;
333}
334const Snapshot = struct { rectangles: []const Rectangle, command: Text };
335const Checker = struct {
336 gpa: Allocator,
337 scratch: Allocator,
338 out: *Builder,

342 fn fail(self: *Checker, comptime format: Text, args: anytype) Error {
343 failure = std.fmt.allocPrint(self.gpa, "error: " ++ format, args) catch return error.OutOfMemory;
344 return error.Invalid;

346 fn get(self: *Checker, name: Text) Error!Rectangle {
347 return self.rectangles.get(name) orelse return self.fail("unknown rectangle {s}", .{name});
348 }

350 var result: std.ArrayList(Rectangle) = .empty;
351 var iterator = self.rectangles.valueIterator();
352 while (iterator.next()) |item| if (layer == null or equal(item.layer, layer.?)) try result.append(self.scratch, item.*);
353 std.mem.sort(Rectangle, result.items, {}, rectangleLess);
354 return result.items;
355 }
356 fn require(self: *Checker, layer: Text) Error![]Rectangle {
357 const result = try self.list(layer);
358 if (result.len == 0) return self.fail("no rectangles on {s}", .{layer});
359 return result;

361 fn remember(self: *Checker, line: Text) Error!void {
362 try self.history.append(self.gpa, .{ .rectangles = try self.gpa.dupe(Rectangle, try self.list(null)), .command = line });
363 }
364 fn checkBounds(self: *Checker, r: Geometry) Error!void {
365 if (abs(r.x1) > 10000 or abs(r.y1) > 10000 or abs(r.x2) > 10000 or abs(r.y2) > 10000) return self.fail("out of bounds", .{});
366 }
367 fn violationLines(self: *Checker, violations: []const Violation) Error!void {
368 for (violations) |v| try self.out.line("{s} {s} gap {d} {s}", .{ v.a, v.b, v.gap, v.direction });
369 }
370 fn command(self: *Checker, line: Text) Error!void {
371 const tokens = try words(self.scratch, line);
372 const commandName = tokens[0];
373 const args = tokens[1..];
374 if (equal(commandName, "ADD")) {
375 if (self.rectangles.contains(args[0])) return self.fail("duplicate rectangle {s}", .{args[0]});
376 const item: Rectangle = .{ .name = args[0], .layer = args[1], .r = .{ .x1 = number(args[2]), .y1 = number(args[3]), .x2 = number(args[4]), .y2 = number(args[5]) } };
377 if (item.r.x1 >= item.r.x2 or item.r.y1 >= item.r.y2) return self.fail("invalid rectangle", .{});
378 try self.checkBounds(item.r);
379 try self.remember(line);
380 try self.rectangles.put(self.gpa, item.name, item);
381 try self.out.line("added {s} {s} {s}", .{ item.name, item.layer, try notation(self.scratch, item.r) });
382 return;
383 }
384 if (oneOf(commandName, &.{ "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE" })) {
385 const original = try self.get(args[0]);
386 var item = original;
387 if (equal(commandName, "COPY")) {
388 if (self.rectangles.contains(args[1])) return self.fail("duplicate rectangle {s}", .{args[1]});
389 item.name = args[1];
390 const dx = number(args[2]);
391 const dy = number(args[3]);
392 item.r.x1 += dx;
393 item.r.x2 += dx;
394 item.r.y1 += dy;
395 item.r.y2 += dy;
396 try self.checkBounds(item.r);
397 try self.remember(line);
398 try self.rectangles.put(self.gpa, item.name, item);
399 try self.out.line("copied {s} to {s} {s}", .{ original.name, item.name, try notation(self.scratch, item.r) });
400 } else if (equal(commandName, "MOVE")) {
401 const dx = number(args[1]);
402 const dy = number(args[2]);
403 item.r.x1 += dx;
404 item.r.x2 += dx;
405 item.r.y1 += dy;
406 item.r.y2 += dy;
407 try self.checkBounds(item.r);
408 try self.remember(line);
409 try self.rectangles.put(self.gpa, item.name, item);
410 try self.out.line("moved {s} to {s}", .{ item.name, try notation(self.scratch, item.r) });
411 } else if (equal(commandName, "RESIZE")) {
412 const width = number(args[1]);
413 const height = number(args[2]);
414 if (width <= 0 or height <= 0) return self.fail("invalid size", .{});
415 item.r.x2 = item.r.x1 + width;
416 item.r.y2 = item.r.y1 + height;
417 try self.checkBounds(item.r);
418 try self.remember(line);
419 try self.rectangles.put(self.gpa, item.name, item);
420 try self.out.line("resized {s} to {s}", .{ item.name, try notation(self.scratch, item.r) });
421 } else if (equal(commandName, "SPLIT")) {
422 if (self.rectangles.contains(args[3])) return self.fail("duplicate rectangle {s}", .{args[3]});
423 const pos = number(args[2]);
424 const horizontal = equal(args[1], "x");
425 if (horizontal and (pos <= item.r.x1 or pos >= item.r.x2) or !horizontal and (pos <= item.r.y1 or pos >= item.r.y2)) return self.fail("cut outside {s}", .{item.name});
426 var other = item;
427 other.name = args[3];
428 if (horizontal) {
429 item.r.x2 = pos;
430 other.r.x1 = pos;
431 } else {
432 item.r.y2 = pos;
433 other.r.y1 = pos;
434 }
435 try self.remember(line);
436 try self.rectangles.put(self.gpa, item.name, item);
437 try self.rectangles.put(self.gpa, other.name, other);
438 try self.out.line("split {s} at {s}={d}: {s} {s}, {s} {s}", .{ item.name, args[1], pos, item.name, try notation(self.scratch, item.r), other.name, try notation(self.scratch, other.r) });
439 } else if (equal(commandName, "LAYER")) {
440 if (equal(item.layer, args[1])) return self.fail("already on {s}", .{item.layer});
441 item.layer = args[1];
442 try self.remember(line);
443 try self.rectangles.put(self.gpa, item.name, item);
444 try self.out.line("layer {s} {s} -> {s}", .{ item.name, original.layer, item.layer });
445 } else {
446 try self.remember(line);
447 _ = self.rectangles.remove(item.name);
448 try self.out.line("deleted {s}", .{item.name});
449 }
450 return;
451 }
452 if (equal(commandName, "UNDO")) {
453 const previous = self.history.pop() orelse return self.fail("nothing to undo", .{});
454 self.rectangles.clearRetainingCapacity();
455 for (previous.rectangles) |item| try self.rectangles.put(self.gpa, item.name, item);
456 try self.out.line("undo: {s}", .{previous.command});
457 return;
458 }
459 if (equal(commandName, "LIST")) {
460 try self.out.line("rectangles: {d}", .{self.rectangles.count()});
461 for (try self.list(null)) |item| try self.out.line("{s} {s} {s} {d}x{d}", .{ item.name, item.layer, try notation(self.scratch, item.r), item.r.x2 - item.r.x1, item.r.y2 - item.r.y1 });
462 return;
463 }
464 if (equal(commandName, "INFO")) {
465 const item = try self.get(args[0]);
466 try self.out.line("{s} {s} {s} {d}x{d} area {d}", .{ item.name, item.layer, try notation(self.scratch, item.r), item.r.x2 - item.r.x1, item.r.y2 - item.r.y1, area(item.r) });
467 for ([_]Text{ "overlaps", "touches", "corners", "encloses", "enclosed by" }, 0..) |heading, index| {
468 var selected: std.ArrayList(Rectangle) = .empty;
469 for (try self.list(null)) |other| {
470 if (equal(other.name, item.name)) continue;
471 const r = compare(item.r, other.r);
472 const matched = if (index == 3) encloses(item.r, other.r) else if (index == 4) encloses(other.r, item.r) else equal(item.layer, other.layer) and r.kind == @as(RelationKind, @fromBackingInt(@intCast(index)));
473 if (matched) try selected.append(self.scratch, other);
474 }
475 try self.out.line("{s}: {s}", .{ heading, try names(self.scratch, selected.items) });
476 }
477 return;
478 }
479 if (equal(commandName, "LAYERS")) {
480 var layers: std.StringHashMapUnmanaged(void) = .empty;
481 for (try self.list(null)) |item| try layers.put(self.scratch, item.layer, {});
482 const layerNames = try sortedKeys(self.scratch, layers);
483 try self.out.line("layers: {d}", .{layerNames.len});
484 for (layerNames) |layer| {
485 const rectangles = try self.list(layer);
486 const g = try Grid.build(self.scratch, rectangles);
487 try self.out.line("{s}: {d} rectangles, area {d}, nets {d}", .{ layer, rectangles.len, g.covered(1, null), (try nets(self.scratch, rectangles)).len });
488 }
489 return;
490 }
491 if (equal(commandName, "OVERLAPS")) {
492 try self.overlapsCommand(args);
493 return;
494 }
495 if (oneOf(commandName, &.{ "INTERSECT", "ADJACENT", "ENCLOSE", "PATH" })) {
496 const a = try self.get(args[0]);
497 const b = try self.get(args[1]);
498 if (equal(a.name, b.name)) return self.fail("same rectangle {s}", .{a.name});
499 if (equal(commandName, "PATH")) {
500 if (!equal(a.layer, b.layer)) return self.fail("different layers", .{});
501 try self.path(a, b);
502 return;
503 }
504 const r = compare(a.r, b.r);
505 if (equal(commandName, "INTERSECT")) {
506 switch (r.kind) {
507 .overlap => try self.out.line("intersect {s} {s}: {s} area {d}", .{ a.name, b.name, try notation(self.scratch, r.r), r.ox * r.oy }),
508 .edge => try self.out.line("intersect {s} {s}: segment {s} length {d}", .{ a.name, b.name, try notation(self.scratch, r.r), r.ox + r.oy }),
509 .corner => try self.out.line("intersect {s} {s}: point {s}", .{ a.name, b.name, try point(self.scratch, r.r.x1, r.r.y1) }),
510 .separate => try self.out.line("intersect {s} {s}: none, gap {d}", .{ a.name, b.name, r.gap }),
511 }
512 } else if (equal(commandName, "ADJACENT")) {
513 switch (r.kind) {
514 .overlap => try self.out.line("adjacent {s} {s}: overlapping", .{ a.name, b.name }),
515 .corner => try self.out.line("adjacent {s} {s}: corner only at {s}", .{ a.name, b.name, try point(self.scratch, r.r.x1, r.r.y1) }),
516 .separate => try self.out.line("adjacent {s} {s}: not adjacent", .{ a.name, b.name }),
517 .edge => {
518 const direction = if (r.ox == 0) (if (a.r.x2 == r.r.x1) "west" else "east") else (if (a.r.y2 == r.r.y1) "south" else "north");
519 try self.out.line("adjacent {s} {s}: {s} is {s} of {s}, edge {s}, length {d}", .{ a.name, b.name, a.name, direction, b.name, try notation(self.scratch, r.r), r.ox + r.oy });
520 },
521 }
522 } else {
523 const m = margins(a.r, b.r);
524 try self.out.print("enclose {s} {s}: {s}, margins left {d} bottom {d} right {d} top {d}", .{ a.name, b.name, if (minimum(m) >= 0) "yes" else "no", m[0], m[1], m[2], m[3] });
525 if (minimum(m) >= 0) try self.out.print(", min {d}", .{minimum(m)});
526 try self.out.add("\n");
527 }
528 return;
529 }
530 if (equal(commandName, "AT")) {
531 const x = number(args[0]);
532 const y = number(args[1]);
533 var parts: std.ArrayList(Text) = .empty;
534 for (try self.list(null)) |item| {
535 const r = item.r;
536 if (x < r.x1 or x > r.x2 or y < r.y1 or y > r.y2) continue;
537 const vertical = x == r.x1 or x == r.x2;
538 const horizontal = y == r.y1 or y == r.y2;
539 const pos = if (vertical and horizontal) "corner" else if (vertical or horizontal) "edge" else "inside";
540 try parts.append(self.scratch, try std.fmt.allocPrint(self.scratch, "{s} {s} {s}", .{ item.name, item.layer, pos }));
541 }
542 try self.out.line("at {s}: {s}", .{ try point(self.scratch, x, y), if (parts.items.len == 0) "none" else try join(self.scratch, parts.items, ", ") });
543 return;
544 }
545 if (equal(commandName, "NEAREST")) {
546 const item = try self.get(args[0]);
547 var best: ?Rectangle = null;
548 var gap: Int = 0;
549 for (try self.list(item.layer)) |other| {
550 if (equal(item.name, other.name)) continue;
551 const value = compare(item.r, other.r).gap;
552 if (best == null or value < gap) {
553 best = other;
554 gap = value;
555 }
556 }
557 if (best) |other| {
558 try self.out.line("nearest {s}: {s} gap {d}", .{ item.name, other.name, gap });
559 } else try self.out.line("nearest {s}: none", .{item.name});
560 return;
561 }
562 if (equal(commandName, "RULE")) {
563 const kind: RuleKind = if (equal(args[0], "SPACING")) .spacing else if (equal(args[0], "WIDTH")) .width else .enclose;
564 const r: Rule = .{ .kind = kind, .inner = args[1], .outer = if (kind == .enclose) args[2] else "", .value = number(args[if (kind == .enclose) @as(usize, 3) else 2]) };
565 if (kind == .enclose and equal(r.inner, r.outer)) return self.fail("same layer {s}", .{r.inner});
566 const key = try std.fmt.allocPrint(self.gpa, "{s}:{s}:{s}", .{ @tagName(kind), r.inner, r.outer });
567 try self.rules.put(self.gpa, key, r);
568 if (kind == .enclose) {
569 try self.out.line("rule enclose {s} {s} {d}", .{ r.inner, r.outer, r.value });
570 } else try self.out.line("rule {s} {s} {d}", .{ @tagName(kind), r.inner, r.value });
571 return;
572 }
573 if (equal(commandName, "CHECK")) {
574 try self.check();
575 return;
576 }
577 const layer = args[if (equal(commandName, "COVER")) @as(usize, 1) else 0];
578 const rectangles = try self.require(layer);
579 const g = try Grid.build(self.scratch, rectangles);
580 if (equal(commandName, "AREA")) {
581 try self.out.line("area {s}: {d}", .{ layer, g.covered(1, null) });
582 } else if (equal(commandName, "COVER")) {
583 const k = number(args[0]);
584 try self.out.line("cover {s} >= {d}: {d}", .{ layer, k, g.covered(@intCast(k), null) });
585 } else if (equal(commandName, "PERIMETER")) {
586 try self.out.line("perimeter {s}: {d}", .{ layer, g.perimeter() });
587 } else if (equal(commandName, "NETS")) {
588 const groups = try nets(self.scratch, rectangles);
589 try self.out.line("nets {s}: {d}", .{ layer, groups.len });
590 for (groups, 0..) |group, i| {
591 const unionGrid = try Grid.build(self.scratch, group);
592 try self.out.line("net {d}: {s} area {d}", .{ i + 1, try names(self.scratch, group), unionGrid.covered(1, null) });
593 }
594 } else if (equal(commandName, "DRC")) {
595 const s = number(args[1]);
596 const violations = try spacing(self.scratch, rectangles, s);
597 try self.out.line("drc {s} spacing {d}: {d} violations", .{ layer, s, violations.len });
598 try self.violationLines(violations);
599 } else if (equal(commandName, "BBOX")) {
600 const r = bounds(rectangles);
601 try self.out.line("bbox {s}: {s} {d}x{d}, {d} rectangles, density {s}%", .{ layer, try notation(self.scratch, r), r.x2 - r.x1, r.y2 - r.y1, rectangles.len, try decimal(self.scratch, 100 * g.covered(1, null), area(r)) });
602 } else if (equal(commandName, "DENSITY")) {
603 try self.density(layer, rectangles, g, number(args[1]));
604 } else if (equal(commandName, "RENDER")) {
605 const r: Geometry = .{ .x1 = number(args[1]), .y1 = number(args[2]), .x2 = number(args[3]), .y2 = number(args[4]) };
606 if (r.x1 >= r.x2 or r.y1 >= r.y2) return self.fail("invalid window", .{});
607 if (r.x2 - r.x1 > 60 or r.y2 - r.y1 > 40) return self.fail("window too large", .{});
608 try self.out.line("render {s} {s}", .{ layer, try notation(self.scratch, r) });
609 var y = r.y2;
610 while (y > r.y1) {
611 y -= 1;
612 var x = r.x1;
613 while (x < r.x2) : (x += 1) {
614 var count: usize = 0;
615 for (rectangles) |item| if (x >= item.r.x1 and x < item.r.x2 and y >= item.r.y1 and y < item.r.y2) {
616 count += 1;
617 };
618 try self.out.bytes.append(self.gpa, if (count == 0) '.' else if (count >= 10) '*' else @as(u8, '0') + @as(u8, @intCast(count)));
619 }
620 try self.out.add("\n");
621 }
622 } else if (equal(commandName, "EMPTY")) {
623 if (try g.empty(self.scratch)) |r| {
624 try self.out.line("empty {s}: {s} area {d}", .{ layer, try notation(self.scratch, r), area(r) });
625 } else try self.out.line("empty {s}: none", .{layer});
626 } else if (equal(commandName, "HOLES")) {
627 const holes = try g.holes(self.scratch);
628 try self.out.line("holes {s}: {d}", .{ layer, holes.len });
629 for (holes, 0..) |hole, i| try self.out.line("hole {d}: area {d}, bbox {s}", .{ i + 1, hole.area, try notation(self.scratch, hole.r) });
630 } else if (equal(commandName, "MERGE")) {
631 const pieces = try g.pieces(self.scratch);
632 try self.out.line("merge {s}: {d} pieces", .{ layer, pieces.len });
633 for (pieces) |r| try self.out.line("{s}", .{try notation(self.scratch, r)});
634 }
635 }
636 fn overlapsCommand(self: *Checker, args: []const Text) Error!void {
637 const rectangles = if (args.len > 0) try self.require(args[0]) else try self.list(null);
638 const Pair = struct { a: Rectangle, b: Rectangle, r: Relation };
639 var pairs: std.ArrayList(Pair) = .empty;
640 for (rectangles, 0..) |a, i| {
641 for (rectangles[i + 1 ..]) |b| {
642 const r = compare(a.r, b.r);
643 if (equal(a.layer, b.layer) and r.kind == .overlap) try pairs.append(self.scratch, .{ .a = a, .b = b, .r = r });
644 }
645 }
646 std.mem.sort(Pair, pairs.items, {}, struct {
647 fn less(_: void, a: Pair, b: Pair) bool {
648 const layer = std.mem.order(u8, a.a.layer, b.a.layer);
649 const first = std.mem.order(u8, a.a.name, b.a.name);
650 return layer == .lt or layer == .eq and (first == .lt or first == .eq and std.mem.order(u8, a.b.name, b.b.name) == .lt);
651 }
652 }.less);
653 if (args.len == 0) {
654 try self.out.line("overlaps all layers: {d}", .{pairs.items.len});
655 } else try self.out.line("overlaps {s}: {d}", .{ args[0], pairs.items.len });
656 for (pairs.items) |pair| {
657 if (args.len == 0) try self.out.print("{s} ", .{pair.a.layer});
658 try self.out.line("{s} {s} {s} area {d}", .{ pair.a.name, pair.b.name, try notation(self.scratch, pair.r.r), pair.r.ox * pair.r.oy });
659 }
660 }
661 fn path(self: *Checker, a: Rectangle, b: Rectangle) Error!void {
662 var queue: std.ArrayList(Rectangle) = .empty;
663 try queue.append(self.scratch, a);
664 var parents: std.StringHashMapUnmanaged(Text) = .empty;
665 try parents.put(self.scratch, a.name, "");
666 const rectangles = try self.list(a.layer);
667 var i: usize = 0;
668 while (i < queue.items.len) : (i += 1) {
669 for (rectangles) |item| if (!parents.contains(item.name) and connected(queue.items[i], item)) {
670 try parents.put(self.scratch, item.name, queue.items[i].name);
671 try queue.append(self.scratch, item);
672 };
673 }
674 if (!parents.contains(b.name)) {
675 try self.out.line("path {s} {s}: none", .{ a.name, b.name });
676 return;
677 }
678 var parts: std.ArrayList(Text) = .empty;
679 var name = b.name;
680 while (name.len > 0) {
681 try parts.append(self.scratch, name);
682 name = parents.get(name).?;
683 }
684 std.mem.reverse(Text, parts.items);
685 try self.out.line("path {s} {s}: {s} ({d} steps)", .{ a.name, b.name, try join(self.scratch, parts.items, " -> "), parts.items.len - 1 });
686 }
687 fn density(self: *Checker, layer: Text, rectangles: []const Rectangle, g: Grid, width: Int) Error!void {
688 const r = bounds(rectangles);
689 const columns = @divTrunc(r.x2 - r.x1 + width - 1, width);
690 const rows = @divTrunc(r.y2 - r.y1 + width - 1, width);
691 if (columns > 20 or rows > 20) return self.fail("too many windows", .{});
692 const values = try self.scratch.alloc(Int, u(columns * rows));
693 var lowest: Int = 0;
694 var highest: Int = 0;
695 var lowX: Int = 0;
696 var lowY: Int = 0;
697 var highX: Int = 0;
698 var highY: Int = 0;
699 var y: Int = 0;
700 while (y < rows) : (y += 1) {
701 var x: Int = 0;
702 while (x < columns) : (x += 1) {
703 const tile: Geometry = .{ .x1 = r.x1 + x * width, .y1 = r.y1 + y * width, .x2 = r.x1 + (x + 1) * width, .y2 = r.y1 + (y + 1) * width };
704 const value = g.covered(1, tile);
705 values[u(y * columns + x)] = value;
706 if (x == 0 and y == 0 or value < lowest) {
707 lowest = value;
708 lowX = x;
709 lowY = y;
710 }
711 if (x == 0 and y == 0 or value > highest) {
712 highest = value;
713 highX = x;
714 highY = y;
715 }
716 }
717 }
718 try self.out.line("density {s} window {d}: {d}x{d}", .{ layer, width, columns, rows });
719 y = rows;
720 while (y > 0) {
721 y -= 1;
722 var x: Int = 0;
723 while (x < columns) : (x += 1) {
724 if (x > 0) try self.out.add(" ");
725 try self.out.print("{d}", .{@divTrunc(100 * values[u(y * columns + x)], width * width)});
726 }
727 try self.out.add("\n");
728 }
729 try self.out.line("min {s}% at {s}", .{ try decimal(self.scratch, 100 * lowest, width * width), try point(self.scratch, r.x1 + lowX * width, r.y1 + lowY * width) });
730 try self.out.line("max {s}% at {s}", .{ try decimal(self.scratch, 100 * highest, width * width), try point(self.scratch, r.x1 + highX * width, r.y1 + highY * width) });
731 }
732 fn check(self: *Checker) Error!void {
733 if (self.rules.count() == 0) return self.fail("no rules", .{});
734 var rules: std.ArrayList(Rule) = .empty;
735 var iterator = self.rules.valueIterator();
736 while (iterator.next()) |r| try rules.append(self.scratch, r.*);
737 std.mem.sort(Rule, rules.items, {}, ruleLess);
738 var total: usize = 0;
739 for (rules.items) |r| {
740 const rectangles = try self.list(r.inner);
741 var lines: std.ArrayList(Text) = .empty;
742 switch (r.kind) {
743 .spacing => for (try spacing(self.scratch, rectangles, r.value)) |v| {
744 try lines.append(self.scratch, try std.fmt.allocPrint(self.scratch, "{s} {s} gap {d} {s}", .{ v.a, v.b, v.gap, v.direction }));
745 },
746 .width => for (rectangles) |item| {
747 const width = @min(item.r.x2 - item.r.x1, item.r.y2 - item.r.y1);
748 if (width < r.value) try lines.append(self.scratch, try std.fmt.allocPrint(self.scratch, "{s} width {d}", .{ item.name, width }));
749 },
750 .enclose => {
751 const outer = try self.list(r.outer);
752 for (rectangles) |item| {
753 var best: ?Int = null;
754 for (outer) |other| {
755 const value = minimum(margins(other.r, item.r));
756 best = if (best) |old| @max(old, value) else value;
757 }
758 if (best == null) {
759 try lines.append(self.scratch, try std.fmt.allocPrint(self.scratch, "{s} best margin none", .{item.name}));
760 } else if (best.? < r.value) {
761 try lines.append(self.scratch, try std.fmt.allocPrint(self.scratch, "{s} best margin {d}", .{ item.name, best.? }));
762 }
763 }
764 },
765 }
766 if (r.kind == .enclose) {
767 try self.out.line("enclose {s} {s} {d}: {d} violations", .{ r.inner, r.outer, r.value, lines.items.len });
768 } else try self.out.line("{s} {s} {d}: {d} violations", .{ @tagName(r.kind), r.inner, r.value, lines.items.len });
769 for (lines.items) |line| try self.out.line("{s}", .{line});
770 total += lines.items.len;
771 }
772 try self.out.line("check: {d} violations in {d} rules", .{ total, rules.items.len });
773 }
774};
775fn oneOf(s: Text, choices: []const Text) bool {
776 for (choices) |choice| if (equal(s, choice)) return true;
777 return false;
778}
779fn sortedKeys(gpa: Allocator, map: anytype) Allocator.Error![]Text {
780 var result: std.ArrayList(Text) = .empty;
781 var iterator = map.keyIterator();
782 while (iterator.next()) |key| try result.append(gpa, key.*);
783 std.mem.sort(Text, result.items, {}, struct {
784 fn less(_: void, a: Text, b: Text) bool {
785 return std.mem.order(u8, a, b) == .lt;
786 }
787 }.less);
788 return result.items;
789}
790pub fn main(init: std.process.Init) !void {
791 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
792 defer arena.deinit();
793 const gpa = arena.allocator();
794 var inputBuffer: [8192]u8 = undefined;
795 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
796 const text = try input.interface.allocRemaining(gpa, .unlimited);
797 var lines = std.mem.splitScalar(u8, text, '\n');
798 const count = u(number(std.mem.trimEnd(u8, lines.next().?, "\r")));
799 var out: Builder = .{ .gpa = gpa };
800 var c: Checker = .{ .gpa = gpa, .scratch = gpa, .out = &out };
801 for (0..count) |_| {
802 const line = std.mem.trimEnd(u8, lines.next().?, "\r");
803 var temporary = std.heap.ArenaAllocator.init(std.heap.page_allocator);
804 defer temporary.deinit();
805 c.scratch = temporary.allocator();
806 c.command(line) catch |err| {
807 if (err != error.Invalid) return err;
808 try out.line("{s}", .{failure});
809 };
810 }
811 var outputBuffer: [8192]u8 = undefined;
812 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
813 try output.interface.writeAll(out.bytes.items);
814 try output.interface.flush();
815}
