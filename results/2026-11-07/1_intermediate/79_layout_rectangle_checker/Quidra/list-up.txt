void emit(string text)
    println(text)

int minimum(int a, int b)
    return if a < b then a else b

int maximum(int a, int b)
    return if a > b then a else b
bool has_text(string[] choices, string value)
    for choice := choices[]
        if choice == value
            return true
    return false
class Box(int x1, int y1, int x2, int y2)
class Rectangle(string name, string layer, Box box)
class Relation(nat kind, Box common, int ox, int oy, int gap, string direction)
Relation relation(Rectangle a, Rectangle b)
    common := Box(maximum(a.box.x1, b.box.x1), maximum(a.box.y1, b.box.y1), minimum(a.box.x2, b.box.x2), minimum(a.box.y2, b.box.y2))
    ox := common.x2 - common.x1
    oy := common.y2 - common.y1
    nat kind := if ox > 0 and oy > 0 then 0 else (if ox < 0 or oy < 0 then 3 else (if ox == 0 and oy == 0 then 2 else 1))
    direction := if ox < 0 and oy < 0 then "diagonal" else (if ox < 0 then "x" else "y")
    return Relation(kind, common, ox, oy, maximum(0, maximum(-ox, -oy)), direction)
class Violation(int gap, string first, string second, string direction)
string[] spacing_lines(Rectangle[] rects, int spacing)
    Violation[] found := []
    for: i < #rects
        for: i + 1 <= j < #rects
            current := relation(rects[i], rects[j])
            if current.kind != 2 and not (current.kind == 3 and current.gap < spacing)
                continue
            entry := Violation(current.gap, rects[i].name, rects[j].name, if current.kind == 2 then "corner" else current.direction)
            nat at := 0
            while at < #found and found[at].gap <= entry.gap
                at += 1
            found = found[:<at].append(entry).concat(found[at:])
    return [for entry := found[] then entry.first + " " + entry.second + " gap {entry.gap} " + entry.direction]
void report(string heading, string[] lines)
    emit(heading)
    for line := lines[]
        emit(line)

        Rectangle[] result := []
        for name := rect_table.keys().sorted()[]
            match rect_table.get(name)
                Rectangle value
                    result = result.append(value)
        return result
    Rectangle[] on_layer(string layer)
        Rectangle[] result := []
        for rect := all_rects()[]
            if rect.layer == layer
                result = result.append(rect)
        return result

    Rectangle[] | error require_layer(string layer)
        Rectangle[] result := on_layer(layer)
        if #result == 0
            return error("no rectangles on " + layer)
        return result
