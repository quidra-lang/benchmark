137 Hole[] result := []

158 for next := [Cell(x - 1, y), Cell(x + 1, y), Cell(x, y - 1), Cell(x, y + 1)][]

181 Box | none largest_empty()
182 found := false
183 best_area := 0
184 best := Box(0, 0, 0, 0)
185 for: int a < column_count()
186 free := [for j := (0:<int(row_count()))[] then true]
187 for: a + 1 <= int b <= column_count()
188 for: int j < row_count()
189 free[j] = free[j] and counts[b - 1][j] == 0

193 better := not found or amount > best_area
194 if found and amount == best_area

196 if better
