28class CoverageGrid
29{
30 public array $xs;
31 public array $ys;
32 public array $cells;
33
34 public function __construct(array $rectangles)
35 {
36 $xs = $ys = [];
37 foreach ($rectangles as $r) {
38 $xs[] = $r->x1;
39 $xs[] = $r->x2;
40 $ys[] = $r->y1;
41 $ys[] = $r->y2;
42 }
43 $this->xs = array_values(array_unique($xs, SORT_NUMERIC));
44 $this->ys = array_values(array_unique($ys, SORT_NUMERIC));
45 sort($this->xs, SORT_NUMERIC);
46 sort($this->ys, SORT_NUMERIC);
47 $this->cells = array_fill(0, count($this->ys) - 1, array_fill(0, count($this->xs) - 1, 0));
48 foreach ($rectangles as $r) {
49 $left = array_search($r->x1, $this->xs, true);
50 $right = array_search($r->x2, $this->xs, true);
51 $bottom = array_search($r->y1, $this->ys, true);
52 $top = array_search($r->y2, $this->ys, true);
53 for ($j = $bottom; $j < $top; $j++) {
54 for ($i = $left; $i < $right; $i++) {
55 $this->cells[$j][$i]++;
56 }
57 }
58 }
59 }
60
61 public function area(int $k = 1, ?array $window = null): int
62 {
63 $result = 0;
64 foreach ($this->cells as $j => $row) {
65 foreach ($row as $i => $count) {
66 if ($count < $k) {
67 continue;
68 }
69 $dx = $window === null ? $this->xs[$i + 1] - $this->xs[$i] : max(0, min($this->xs[$i + 1], $window[2]) - max($this->xs[$i], $window[0]));
70 $dy = $window === null ? $this->ys[$j + 1] - $this->ys[$j] : max(0, min($this->ys[$j + 1], $window[3]) - max($this->ys[$j], $window[1]));
71 $result += $dx * $dy;
72 }
73 }
74 return $result;
75 }
76
77 public function perimeter(): int
78 {
79 $result = 0;
80 foreach ($this->cells as $j => $row) {
81 foreach ($row as $i => $count) {
82 if ($count === 0) {
83 continue;
84 }
85 $dy = $this->ys[$j + 1] - $this->ys[$j];
86 $dx = $this->xs[$i + 1] - $this->xs[$i];
87 if ($i === 0 || $row[$i - 1] === 0) {
88 $result += $dy;
89 }
90 if ($i + 1 === count($row) || $row[$i + 1] === 0) {
91 $result += $dy;
92 }
93 if ($j === 0 || $this->cells[$j - 1][$i] === 0) {
94 $result += $dx;
95 }
96 if ($j + 1 === count($this->cells) || $this->cells[$j + 1][$i] === 0) {
97 $result += $dx;
98 }
99 }
100 }
101 return $result;
102 }
103
104 public function largestEmpty(): ?array
105 {
106 $best = null;
107 for ($left = 0; $left + 1 < count($this->xs); $left++) {
108 $blocked = array_fill(0, count($this->ys) - 1, false);
109 for ($right = $left; $right + 1 < count($this->xs); $right++) {
110 foreach ($blocked as $j => $value) {
111 $blocked[$j] = $value || $this->cells[$j][$right] > 0;
112 }
113 for ($j = 0; $j < count($blocked);) {
114 if ($blocked[$j]) {
115 $j++;
116 continue;
117 }
118 $start = $j;
119 while ($j < count($blocked) && !$blocked[$j]) {
120 $j++;
121 }
122 $width = $this->xs[$right + 1] - $this->xs[$left];
123 $area = $width * ($this->ys[$j] - $this->ys[$start]);
124 $choice = [-$area, $this->xs[$left], $this->ys[$start], $width, $this->xs[$right + 1], $this->ys[$j]];
125 if ($best === null || $choice < $best) {
126 $best = $choice;
127 }
128 }
129 }
130 }
131 return $best === null ? null : [[$best[1], $best[2], $best[4], $best[5]], -$best[0]];
132 }
133
134 public function holes(): array
135 {
136 $visited = $result = [];
137 $nx = count($this->xs) - 1;
138 $ny = count($this->ys) - 1;
139 for ($j = 0; $j < $ny; $j++) {
140 for ($i = 0; $i < $nx; $i++) {
141 if ($this->cells[$j][$i] > 0 || isset($visited["$i,$j"])) {
142 continue;
143 }
144 $queue = [[$i, $j]];
145 $visited["$i,$j"] = true;
146 $area = $head = 0;
147 $boundary = false;
148 $x1 = $x2 = $this->xs[$i];
149 $y1 = $y2 = $this->ys[$j];
150 $anchor = [$y1, $x1];
151 while ($head < count($queue)) {
152 [$x, $y] = $queue[$head++];
153 $area += ($this->xs[$x + 1] - $this->xs[$x]) * ($this->ys[$y + 1] - $this->ys[$y]);
154 $x1 = min($x1, $this->xs[$x]);
155 $x2 = max($x2, $this->xs[$x + 1]);
156 $y1 = min($y1, $this->ys[$y]);
157 $y2 = max($y2, $this->ys[$y + 1]);
158 $point = [$this->ys[$y], $this->xs[$x]];
159 if ($point < $anchor) {
160 $anchor = $point;
161 }
162 $boundary = $boundary || $x === 0 || $y === 0 || $x + 1 === $nx || $y + 1 === $ny;
163 foreach ([[$x - 1, $y], [$x + 1, $y], [$x, $y - 1], [$x, $y + 1]] as [$a, $b]) {
164 if ($a < 0 || $b < 0 || $a >= $nx || $b >= $ny || $this->cells[$b][$a] > 0 || isset($visited["$a,$b"])) {
165 continue;
166 }
167 $visited["$a,$b"] = true;
168 $queue[] = [$a, $b];
169 }
170 }
171 if (!$boundary) {
172 $result[] = [$anchor, $area, [$x1, $y1, $x2, $y2]];
173 }
174 }
175 }
176 usort($result, fn($a, $b) => $a[0] <=> $b[0]);
177 return $result;
178 }
179
180 public function pieces(): array
181 {
182 $active = $result = [];
183 foreach ($this->cells as $j => $row) {
184 $current = [];
185 for ($i = 0; $i < count($row);) {
186 if ($row[$i] === 0) {
187 $i++;
188 continue;
189 }
190 $start = $i;
191 while ($i < count($row) && $row[$i] > 0) {
192 $i++;
193 }
194 $left = $this->xs[$start];
195 $right = $this->xs[$i];
196 $key = "$left,$right";
197 $current[$key] = [$left, $active[$key][1] ?? $this->ys[$j], $right];
198 unset($active[$key]);
199 }
200 foreach ($active as [$left, $bottom, $right]) {
201 $result[] = [$left, $bottom, $right, $this->ys[$j]];
202 }
203 $active = $current;
204 }
205 foreach ($active as [$left, $bottom, $right]) {
206 $result[] = [$left, $bottom, $right, $this->ys[count($this->ys) - 1]];
207 }
208 usort($result, fn($a, $b) => [$a[1], $a[0]] <=> [$b[1], $b[0]]);
209 return $result;
210 }

219 private function notation(array $box): string
220 {

231 $value = intdiv(10000 * $area, $denominator);
232 return sprintf('%d.%02d', intdiv($value, 100), $value % 100);

235 private function sorted(): array
236 {
237 $result = array_values($this->rectangles);

258 if (!isset($this->rectangles[$name])) {

264 private function bounds(array $box): void
265 {
266 foreach ($box as $coordinate) {
267 if (abs($coordinate) > 10000) {
268 throw new LayoutError('out of bounds');
269 }
270 }
271 }
272
273 private function remember(string $line): void
274 {
275 $this->history[] = [array_map(fn($r) => clone $r, $this->rectangles), $line];
276 }
277
278 private function relation(Rectangle $a, Rectangle $b): array
279 {
280 $box = [max($a->x1, $b->x1), max($a->y1, $b->y1), min($a->x2, $b->x2), min($a->y2, $b->y2)];
281 $ox = $box[2] - $box[0];
282 $oy = $box[3] - $box[1];
283 $kind = $ox < 0 || $oy < 0 ? 'separated' : ($ox === 0 && $oy === 0 ? 'corner' : ($ox === 0 || $oy === 0 ? 'edge' : 'overlap'));
284 $direction = $ox < 0 && $oy < 0 ? 'diagonal' : ($ox < 0 ? 'x' : 'y');
285 return [$kind, $box, $ox, $oy, max(0, -$ox, -$oy), $direction];
286 }
287
288 private function contains(Rectangle $a, Rectangle $b): bool
289 {
290 return $a->x1 <= $b->x1 && $a->y1 <= $b->y1 && $a->x2 >= $b->x2 && $a->y2 >= $b->y2;
291 }
292
293 private function connected(Rectangle $a, Rectangle $b): bool
294 {
295 return in_array($this->relation($a, $b)[0], ['overlap', 'edge'], true);
296 }
297
298 private function nets(array $rectangles): array
299 {
300 $remaining = $rectangles;
301 $result = [];
302 while ($remaining !== []) {
303 $group = [array_shift($remaining)];
304 for ($i = 0; $i < count($group); $i++) {
305 $next = [];
306 foreach ($remaining as $r) {
307 if ($this->connected($group[$i], $r)) {
308 $group[] = $r;
309 } else {
310 $next[] = $r;
311 }
312 }
313 $remaining = $next;
314 }
315 usort($group, fn($a, $b) => strcmp($a->name, $b->name));
316 $result[] = $group;
317 }
318 return $result;
319 }
320
321 private function spacing(array $rectangles, int $minimum): array
322 {
323 $result = [];
324 foreach ($rectangles as $i => $a) {
325 for ($j = $i + 1; $j < count($rectangles); $j++) {
326 $b = $rectangles[$j];
327 [$kind, , , , $gap, $direction] = $this->relation($a, $b);
328 if ($kind === 'corner' || ($kind === 'separated' && $gap < $minimum)) {
329 $result[] = [$gap, $a->name, $b->name, $kind === 'corner' ? 'corner' : $direction];
330 }
331 }
332 }
333 usort($result, fn($a, $b) => $a <=> $b);
334 return $result;
335 }

337 private function spacingLines(array $violations): void
338 {
339 foreach ($violations as [$gap, $a, $b, $direction]) {
340 echo "$a $b gap $gap $direction\n";
341 }

344 private function bounding(array $rectangles): array
345 {
346 return [min(array_map(fn($r) => $r->x1, $rectangles)), min(array_map(fn($r) => $r->y1, $rectangles)), max(array_map(fn($r) => $r->x2, $rectangles)), max(array_map(fn($r) => $r->y2, $rectangles))];
347 }

349 public function execute(string $line): void
350 {
351 $args = explode(' ', $line);
352 $command = array_shift($args);
353 try {
354 if ($command === 'ADD') {
355 [$name, $layer] = array_slice($args, 0, 2);
356 if (isset($this->rectangles[$name])) {
357 throw new LayoutError("duplicate rectangle $name");
358 }
359 [$x1, $y1, $x2, $y2] = array_map('intval', array_slice($args, 2));
360 if ($x1 >= $x2 || $y1 >= $y2) {
361 throw new LayoutError('invalid rectangle');
362 }
363 $this->bounds([$x1, $y1, $x2, $y2]);
364 $this->remember($line);
365 $this->rectangles[$name] = new Rectangle($name, $layer, $x1, $y1, $x2, $y2);
366 echo "added $name $layer ", $this->notation([$x1, $y1, $x2, $y2]), "\n";
367 } elseif (in_array($command, ['COPY', 'MOVE', 'RESIZE', 'SPLIT', 'LAYER', 'DELETE'], true)) {
368 $name = $args[0];
369 $original = $this->rectangle($name);
370 $changed = clone $original;
371 if ($command === 'COPY' || $command === 'MOVE') {
372 $newName = $command === 'COPY' ? $args[1] : $name;
373 if ($command === 'COPY' && isset($this->rectangles[$newName])) {
374 throw new LayoutError("duplicate rectangle $newName");
375 }
376 [$dx, $dy] = array_map('intval', array_slice($args, $command === 'COPY' ? 2 : 1));
377 $changed->name = $newName;
378 $changed->x1 += $dx;
379 $changed->x2 += $dx;
380 $changed->y1 += $dy;
381 $changed->y2 += $dy;
382 $this->bounds($changed->box());
383 $message = $command === 'COPY' ? "copied $name to $newName " . $this->notation($changed->box()) : "moved $name to " . $this->notation($changed->box());
384 } elseif ($command === 'RESIZE') {
385 [$width, $height] = array_map('intval', array_slice($args, 1));
386 if ($width <= 0 || $height <= 0) {
387 throw new LayoutError('invalid size');
388 }
389 $changed->x2 = $changed->x1 + $width;
390 $changed->y2 = $changed->y1 + $height;
391 $this->bounds($changed->box());
392 $message = "resized $name to " . $this->notation($changed->box());
393 } elseif ($command === 'SPLIT') {
394 [$axis, $position, $newName] = array_slice($args, 1);
395 $position = (int)$position;
396 if (isset($this->rectangles[$newName])) {
397 throw new LayoutError("duplicate rectangle $newName");
398 }
399 $valid = $axis === 'x' ? $original->x1 < $position && $position < $original->x2 : $original->y1 < $position && $position < $original->y2;
400 if (!$valid) {
401 throw new LayoutError("cut outside $name");
402 }
403 $added = clone $original;
404 $added->name = $newName;
405 if ($axis === 'x') {
406 $changed->x2 = $added->x1 = $position;
407 } else {
408 $changed->y2 = $added->y1 = $position;
409 }
410 $message = "split $name at $axis=$position: $name " . $this->notation($changed->box()) . ", $newName " . $this->notation($added->box());
411 } elseif ($command === 'LAYER') {
412 if ($original->layer === $args[1]) {
413 throw new LayoutError("already on {$args[1]}");
414 }
415 $changed->layer = $args[1];
416 $message = "layer $name {$original->layer} -> {$changed->layer}";
417 } else {
418 $message = "deleted $name";
419 }
420 $this->remember($line);
421 if ($command === 'DELETE') {
422 unset($this->rectangles[$name]);
423 } else {
424 $this->rectangles[$changed->name] = $changed;
425 if ($command === 'SPLIT') {
426 $this->rectangles[$added->name] = $added;
427 }
428 }
429 echo $message, "\n";
430 } elseif ($command === 'UNDO') {
431 if ($this->history === []) {
432 throw new LayoutError('nothing to undo');
433 }
434 [$this->rectangles, $previous] = array_pop($this->history);
435 echo "undo: $previous\n";
436 } elseif ($command === 'LIST') {
437 echo 'rectangles: ', count($this->rectangles), "\n";
438 foreach ($this->sorted() as $r) {
439 echo "{$r->name} {$r->layer} ", $this->notation($r->box()), ' ', $r->width(), 'x', $r->height(), "\n";
440 }
441 } elseif ($command === 'INFO') {
442 $a = $this->rectangle($args[0]);
443 echo "{$a->name} {$a->layer} ", $this->notation($a->box()), ' ', $a->width(), 'x', $a->height(), ' area ', $a->width() * $a->height(), "\n";
444 foreach (['overlaps', 'touches', 'corners', 'encloses', 'enclosed by'] as $i => $label) {
445 $names = [];
446 foreach ($this->sorted() as $b) {
447 if ($a->name === $b->name) {
448 continue;
449 }
450 $matches = $i < 3 ? $b->layer === $a->layer && $this->relation($a, $b)[0] === ['overlap', 'edge', 'corner'][$i] : ($i === 3 ? $this->contains($a, $b) : $this->contains($b, $a));
451 if ($matches) {
452 $names[] = $b->name;
453 }
454 }
455 echo "$label: ", $names === [] ? 'none' : implode(' ', $names), "\n";
456 }
457 } elseif ($command === 'LAYERS') {
458 $layers = array_values(array_unique(array_map(fn($r) => $r->layer, $this->sorted()), SORT_STRING));
459 sort($layers, SORT_STRING);
460 echo 'layers: ', count($layers), "\n";
461 foreach ($layers as $layer) {
462 $rectangles = $this->onLayer($layer);
463 echo "$layer: ", count($rectangles), ' rectangles, area ', (new CoverageGrid($rectangles))->area(), ', nets ', count($this->nets($rectangles)), "\n";
464 }
465 } elseif ($command === 'OVERLAPS') {
466 $rectangles = $args === [] ? $this->sorted() : $this->requireLayer($args[0]);
467 $pairs = [];
468 foreach ($rectangles as $i => $a) {
469 for ($j = $i + 1; $j < count($rectangles); $j++) {
470 $b = $rectangles[$j];
471 [$kind, $box, $ox, $oy] = $this->relation($a, $b);
472 if ($a->layer === $b->layer && $kind === 'overlap') {
473 $pairs[] = [$a->layer, $a->name, $b->name, $box, $ox * $oy];
474 }
475 }
476 }
477 usort($pairs, fn($a, $b) => array_slice($a, 0, 3) <=> array_slice($b, 0, 3));
478 echo 'overlaps ', $args === [] ? 'all layers' : $args[0], ': ', count($pairs), "\n";
479 foreach ($pairs as [$layer, $a, $b, $box, $area]) {
480 echo $args === [] ? "$layer " : '', "$a $b ", $this->notation($box), " area $area\n";
481 }
482 } elseif (in_array($command, ['AREA', 'COVER', 'PERIMETER', 'BBOX', 'DENSITY', 'RENDER', 'EMPTY', 'HOLES', 'MERGE', 'NETS', 'DRC'], true)) {
483 $layer = $command === 'COVER' ? $args[1] : $args[0];
484 $rectangles = $this->requireLayer($layer);
485 $grid = new CoverageGrid($rectangles);
486 $box = $this->bounding($rectangles);
487 if ($command === 'AREA') {
488 echo "area $layer: ", $grid->area(), "\n";
489 } elseif ($command === 'COVER') {
490 echo "cover $layer >= {$args[0]}: ", $grid->area((int)$args[0]), "\n";
491 } elseif ($command === 'PERIMETER') {
492 echo "perimeter $layer: ", $grid->perimeter(), "\n";
493 } elseif ($command === 'BBOX') {
494 $width = $box[2] - $box[0];
495 $height = $box[3] - $box[1];
496 echo "bbox $layer: ", $this->notation($box), " {$width}x$height, ", count($rectangles), ' rectangles, density ', $this->percent($grid->area(), $width * $height), "%\n";
497 } elseif ($command === 'DENSITY') {
498 $width = (int)$args[1];
499 $columns = intdiv($box[2] - $box[0] + $width - 1, $width);
500 $rows = intdiv($box[3] - $box[1] + $width - 1, $width);
501 if ($columns > 20 || $rows > 20) {
502 throw new LayoutError('too many windows');
503 }
504 $values = [];
505 for ($j = 0; $j < $rows; $j++) {
506 for ($i = 0; $i < $columns; $i++) {
507 $values[$j][$i] = $grid->area(1, [$box[0] + $i * $width, $box[1] + $j * $width, $box[0] + ($i + 1) * $width, $box[1] + ($j + 1) * $width]);
508 }
509 }
510 echo "density $layer window $width: {$columns}x$rows\n";
511 for ($j = $rows - 1; $j >= 0; $j--) {
512 echo implode(' ', array_map(fn($area) => intdiv(100 * $area, $width * $width), $values[$j])), "\n";
513 }
514 $flat = array_merge(...$values);
515 foreach ([['min', min($flat)], ['max', max($flat)]] as [$label, $value]) {
516 $index = array_search($value, $flat, true);
517 echo $label, ' ', $this->percent($value, $width * $width), '% at ', $this->point($box[0] + ($index % $columns) * $width, $box[1] + intdiv($index, $columns) * $width), "\n";
518 }
519 } elseif ($command === 'RENDER') {
520 [$x1, $y1, $x2, $y2] = array_map('intval', array_slice($args, 1));
521 if ($x1 >= $x2 || $y1 >= $y2) {
522 throw new LayoutError('invalid window');
523 }
524 if ($x2 - $x1 > 60 || $y2 - $y1 > 40) {
525 throw new LayoutError('window too large');
526 }
527 echo "render $layer ", $this->notation([$x1, $y1, $x2, $y2]), "\n";
528 for ($y = $y2 - 1; $y >= $y1; $y--) {
529 $row = '';
530 for ($x = $x1; $x < $x2; $x++) {
531 $count = 0;
532 foreach ($rectangles as $r) {
533 if ($r->x1 <= $x && $x < $r->x2 && $r->y1 <= $y && $y < $r->y2) {
534 $count++;
535 }
536 }
537 $row .= $count === 0 ? '.' : ($count >= 10 ? '*' : (string)$count);
538 }
539 echo $row, "\n";
540 }
541 } elseif ($command === 'EMPTY') {
542 $empty = $grid->largestEmpty();
543 echo "empty $layer: ", $empty === null ? 'none' : $this->notation($empty[0]) . ' area ' . $empty[1], "\n";
544 } elseif ($command === 'HOLES') {
545 $holes = $grid->holes();
546 echo "holes $layer: ", count($holes), "\n";
547 foreach ($holes as $i => [, $area, $bounds]) {
548 echo 'hole ', $i + 1, ": area $area, bbox ", $this->notation($bounds), "\n";
549 }
550 } elseif ($command === 'MERGE') {
551 $pieces = $grid->pieces();
552 echo "merge $layer: ", count($pieces), " pieces\n";
553 foreach ($pieces as $piece) {
554 echo $this->notation($piece), "\n";
555 }
556 } elseif ($command === 'NETS') {
557 $groups = $this->nets($rectangles);
558 echo "nets $layer: ", count($groups), "\n";
559 foreach ($groups as $i => $group) {
560 echo 'net ', $i + 1, ': ', implode(' ', array_map(fn($r) => $r->name, $group)), ' area ', (new CoverageGrid($group))->area(), "\n";
561 }
562 } else {
563 $violations = $this->spacing($rectangles, (int)$args[1]);
564 echo "drc $layer spacing {$args[1]}: ", count($violations), " violations\n";
565 $this->spacingLines($violations);
566 }
567 } elseif (in_array($command, ['INTERSECT', 'ADJACENT', 'ENCLOSE', 'PATH'], true)) {
568 $a = $this->rectangle($args[0]);
569 $b = $this->rectangle($args[1]);
570 if ($a->name === $b->name) {
571 throw new LayoutError("same rectangle {$a->name}");
572 }
573 [$kind, $box, $ox, $oy, $gap] = $this->relation($a, $b);
574 $prefix = strtolower($command) . " {$a->name} {$b->name}: ";
575 if ($command === 'ENCLOSE') {
576 $margins = [$b->x1 - $a->x1, $b->y1 - $a->y1, $a->x2 - $b->x2, $a->y2 - $b->y2];
577 echo $prefix, min($margins) >= 0 ? 'yes, ' : 'no, ', "margins left {$margins[0]} bottom {$margins[1]} right {$margins[2]} top {$margins[3]}", min($margins) >= 0 ? ', min ' . min($margins) : '', "\n";
578 } elseif ($command === 'PATH') {
579 if ($a->layer !== $b->layer) {
580 throw new LayoutError('different layers');
581 }
582 $queue = [[$a->name]];
583 $seen = [$a->name => true];
584 $head = 0;
585 $path = null;
586 while ($head < count($queue)) {
587 $route = $queue[$head++];
588 $last = $route[count($route) - 1];
589 if ($last === $b->name) {
590 $path = $route;
591 break;
592 }
593 foreach ($this->onLayer($a->layer) as $r) {
594 if (!isset($seen[$r->name]) && $this->connected($this->rectangles[$last], $r)) {
595 $seen[$r->name] = true;
596 $queue[] = array_merge($route, [$r->name]);
597 }
598 }
599 }
600 echo $prefix, $path === null ? 'none' : implode(' -> ', $path) . ' (' . (count($path) - 1) . ' steps)', "\n";
601 } elseif ($command === 'INTERSECT') {
602 $result = match ($kind) {
603 'overlap' => $this->notation($box) . ' area ' . ($ox * $oy),
604 'edge' => 'segment ' . $this->notation($box) . ' length ' . ($ox + $oy),
605 'corner' => 'point ' . $this->point($box[0], $box[1]),
606 default => "none, gap $gap"
607 };
608 echo $prefix, $result, "\n";
609 } else {
610 if ($kind === 'edge') {
611 $direction = $ox === 0 ? ($a->x2 === $box[0] ? 'west' : 'east') : ($a->y2 === $box[1] ? 'south' : 'north');
612 $result = "{$a->name} is $direction of {$b->name}, edge " . $this->notation($box) . ', length ' . ($ox + $oy);
613 } else {
614 $result = $kind === 'overlap' ? 'overlapping' : ($kind === 'corner' ? 'corner only at ' . $this->point($box[0], $box[1]) : 'not adjacent');
615 }
616 echo $prefix, $result, "\n";
617 }
618 } elseif ($command === 'AT') {
619 [$x, $y] = array_map('intval', $args);
620 $found = [];
621 foreach ($this->sorted() as $r) {
622 if ($r->x1 <= $x && $x <= $r->x2 && $r->y1 <= $y && $y <= $r->y2) {
623 $horizontal = $x === $r->x1 || $x === $r->x2;
624 $vertical = $y === $r->y1 || $y === $r->y2;
625 $found[] = "{$r->name} {$r->layer} " . ($horizontal && $vertical ? 'corner' : ($horizontal || $vertical ? 'edge' : 'inside'));
626 }
627 }
628 echo 'at ', $this->point($x, $y), ': ', $found === [] ? 'none' : implode(', ', $found), "\n";
629 } elseif ($command === 'NEAREST') {
630 $a = $this->rectangle($args[0]);
631 $nearest = null;
632 foreach ($this->onLayer($a->layer) as $r) {
633 if ($a->name !== $r->name) {
634 $choice = [$this->relation($a, $r)[4], $r->name];
635 if ($nearest === null || $choice < $nearest) {
636 $nearest = $choice;
637 }
638 }
639 }
640 echo "nearest {$a->name}: ", $nearest === null ? 'none' : "$nearest[1] gap $nearest[0]", "\n";
641 } elseif ($command === 'RULE') {
642 $kind = strtolower($args[0]);
643 if ($kind === 'enclose' && $args[1] === $args[2]) {
644 throw new LayoutError("same layer {$args[1]}");
645 }
646 $key = $kind . ' ' . $args[1] . ($kind === 'enclose' ? ' ' . $args[2] : '');
647 $this->rules[$key] = [$kind, $args[1], $kind === 'enclose' ? $args[2] : null, (int)$args[$kind === 'enclose' ? 3 : 2]];
648 echo 'rule ', $kind, ' ', implode(' ', array_slice($args, 1)), "\n";
649 } elseif ($command === 'CHECK') {
650 if ($this->rules === []) {
651 throw new LayoutError('no rules');
652 }
653 $rules = $this->rules;
654 ksort($rules, SORT_STRING);
655 $total = 0;
656 foreach ($rules as [$kind, $layer, $outer, $value]) {
657 $rectangles = $this->onLayer($layer);
658 if ($kind === 'spacing') {
659 $violations = $this->spacing($rectangles, $value);
660 echo "spacing $layer $value: ", count($violations), " violations\n";
661 $this->spacingLines($violations);
662 } elseif ($kind === 'width') {
663 $violations = array_values(array_filter($rectangles, fn($r) => min($r->width(), $r->height()) < $value));
664 echo "width $layer $value: ", count($violations), " violations\n";
665 foreach ($violations as $r) {
666 echo "{$r->name} width ", min($r->width(), $r->height()), "\n";
667 }
668 } else {
669 $enclosures = $this->onLayer($outer);
670 $violations = [];
671 foreach ($rectangles as $r) {
672 $best = null;
673 foreach ($enclosures as $u) {
674 $margin = min($r->x1 - $u->x1, $r->y1 - $u->y1, $u->x2 - $r->x2, $u->y2 - $r->y2);
675 $best = $best === null ? $margin : max($best, $margin);
676 }
677 if ($best === null || $best < $value) {
678 $violations[] = [$r->name, $best];
679 }
680 }
681 echo "enclose $layer $outer $value: ", count($violations), " violations\n";
682 foreach ($violations as [$name, $best]) {
683 echo "$name best margin ", $best === null ? 'none' : $best, "\n";
684 }
685 }
686 $total += count($violations);
687 }
688 echo "check: $total violations in ", count($this->rules), " rules\n";
689 }
690 } catch (LayoutError $error) {
691 echo 'error: ', $error->getMessage(), "\n";
692 }
693 }
694}
