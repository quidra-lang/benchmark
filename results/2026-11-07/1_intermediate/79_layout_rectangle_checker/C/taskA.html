92static char *read_line(void) {
93 size_t count = 0, capacity = 128;
94 char *line = malloc(capacity);
95 int ch;
96 if (!line) exit(EXIT_FAILURE);
97 while ((ch = getchar()) != EOF && ch != '\n') {
98 if (count + 1 == capacity) {
99 capacity *= 2;
100 char *next = realloc(line, capacity);
101 if (!next) exit(EXIT_FAILURE);
102 line = next;
103 }
104 line[count++] = (char)ch;
105 }
106 line[count] = '\0';
107 return line;
108}

189static Relation relation(const Rectangle *a, const Rectangle *b) {
190 Box x = a->box, y = b->box;
191 Relation r = {0};
192 r.common = (Box){maximum(x.x1, y.x1), maximum(x.y1, y.y1), minimum(x.x2, y.x2), minimum(x.y2, y.y2)};
193 r.ox = width(r.common);
194 r.oy = height(r.common);
195 r.gap = maximum(0, maximum(-r.ox, -r.oy));
196 r.kind = r.ox < 0 || r.oy < 0 ? 3 : r.ox > 0 && r.oy > 0 ? 0 : r.ox == 0 && r.oy == 0 ? 2 : 1;
197 r.direction = r.kind == 2 ? "corner" : r.ox < 0 && r.oy < 0 ? "diagonal" : r.ox < 0 ? "x" : "y";
198 return r;
199}

205static Grid make_grid(const Rectangles *rectangles) {
206 Grid grid = {0};
207 for (int i = 0; i < rectangles->count; i++) {
208 Box b = rectangles->items[i]->box;
209 grid.xs[grid.nx++] = b.x1;
210 grid.xs[grid.nx++] = b.x2;
211 grid.ys[grid.ny++] = b.y1;
212 grid.ys[grid.ny++] = b.y2;
213 }
214 qsort(grid.xs, (size_t)grid.nx, sizeof *grid.xs, integer_compare);
215 qsort(grid.ys, (size_t)grid.ny, sizeof *grid.ys, integer_compare);
216 int n = 0;
217 for (int i = 0; i < grid.nx; i++) if (!n || grid.xs[i] != grid.xs[n - 1]) grid.xs[n++] = grid.xs[i];
218 grid.nx = n;
219 n = 0;
220 for (int i = 0; i < grid.ny; i++) if (!n || grid.ys[i] != grid.ys[n - 1]) grid.ys[n++] = grid.ys[i];
221 grid.ny = n;
222 for (int r = 0; r < rectangles->count; r++) {
223 Box b = rectangles->items[r]->box;
224 for (int j = 0; j + 1 < grid.ny; j++) if (grid.ys[j] >= b.y1 && grid.ys[j + 1] <= b.y2) for (int i = 0; i + 1 < grid.nx; i++) if (grid.xs[i] >= b.x1 && grid.xs[i + 1] <= b.x2) grid.counts[j][i]++;
225 }
226 return grid;
227}

239static long long clipped(const Grid *grid, Box box) {
240 long long result = 0;
241 for (int j = 0; j + 1 < grid->ny; j++) for (int i = 0; i + 1 < grid->nx; i++) if (grid->counts[j][i] > 0) result += (long long)maximum(0, minimum(grid->xs[i + 1], box.x2) - maximum(grid->xs[i], box.x1)) * maximum(0, minimum(grid->ys[j + 1], box.y2) - maximum(grid->ys[j], box.y1));
242 return result;

245static long long perimeter(const Grid *grid) {
246 long long result = 0;
247 for (int j = 0; j + 1 < grid->ny; j++) for (int i = 0; i + 1 < grid->nx; i++) if (grid->counts[j][i] > 0) {
248 if (!i || !grid->counts[j][i - 1]) result += grid->ys[j + 1] - grid->ys[j];
249 if (i + 2 == grid->nx || !grid->counts[j][i + 1]) result += grid->ys[j + 1] - grid->ys[j];
250 if (!j || !grid->counts[j - 1][i]) result += grid->xs[i + 1] - grid->xs[i];
251 if (j + 2 == grid->ny || !grid->counts[j + 1][i]) result += grid->xs[i + 1] - grid->xs[i];
252 }
253 return result;
254}

256static bool empty_rectangle(const Grid *grid, Box *best) {
257 bool found = false;
258 for (int left = 0; left + 1 < grid->nx; left++) {
259 bool blocked[MAX_CELLS] = {false};
260 for (int right = left + 1; right < grid->nx; right++) {
261 for (int j = 0; j + 1 < grid->ny; j++) blocked[j] |= grid->counts[j][right - 1] > 0;
262 int bottom = 0;
263 while (bottom + 1 < grid->ny) {
264 if (blocked[bottom]) {
265 bottom++;
266 continue;
267 }
268 int top = bottom + 1;
269 while (top + 1 < grid->ny && !blocked[top]) top++;
270 Box candidate = {grid->xs[left], grid->ys[bottom], grid->xs[right], grid->ys[top]};
271 if (!found || area(candidate) > area(*best) || (area(candidate) == area(*best) && (candidate.x1 < best->x1 || (candidate.x1 == best->x1 && (candidate.y1 < best->y1 || (candidate.y1 == best->y1 && width(candidate) < width(*best))))))) {
272 *best = candidate;
273 found = true;
274 }
275 bottom = top;
276 }
277 }
278 }
279 return found;
280}

297static Boxes pieces(const Grid *grid) {
298 Boxes result = {0};
299 Box active[MAX_RECTANGLES];
300 int active_count = 0;
301 for (int j = 0; j + 1 < grid->ny; j++) {
302 Box next[MAX_RECTANGLES];
303 int count = 0;
304 bool used[MAX_RECTANGLES] = {false};
305 int i = 0;
306 while (i + 1 < grid->nx) {
307 if (!grid->counts[j][i]) {
308 i++;
309 continue;
310 }
311 int right = i + 1;
312 while (right + 1 < grid->nx && grid->counts[j][right]) right++;
313 int bottom = grid->ys[j];
314 for (int k = 0; k < active_count; k++) if (active[k].x1 == grid->xs[i] && active[k].x2 == grid->xs[right]) {
315 bottom = active[k].y1;
316 used[k] = true;
317 break;
318 }
319 next[count++] = (Box){grid->xs[i], bottom, grid->xs[right], grid->ys[j + 1]};
320 i = right;
321 }
322 for (int k = 0; k < active_count; k++) if (!used[k]) add_box(&result, active[k]);
323 memcpy(active, next, (size_t)count * sizeof *active);
324 active_count = count;
325 }
326 for (int k = 0; k < active_count; k++) add_box(&result, active[k]);
327 if (result.count > 1) qsort(result.items, (size_t)result.count, sizeof *result.items, box_compare);
328 return result;
329}

336static void print_holes(const Grid *grid, const char *layer_name) {
337 bool seen[MAX_CELLS][MAX_CELLS] = {
338 {false}
339 };
340 int queue[MAX_CELLS * MAX_CELLS];
341 Hole *holes = NULL;
342 int count = 0, capacity = 0;
343 int dx[] = {-1, 1, 0, 0}, dy[] = {0, 0, -1, 1};
344 for (int j = 0; j + 1 < grid->ny; j++) for (int i = 0; i + 1 < grid->nx; i++) if (!grid->counts[j][i] && !seen[j][i]) {
345 Hole hole = {
346 0, {grid->xs[i], grid->ys[j], grid->xs[i + 1], grid->ys[j + 1]}, grid->xs[i], grid->ys[j]
347 };
348 bool boundary = false;
349 int head = 0, tail = 0;
350 queue[tail++] = j * MAX_CELLS + i;
351 seen[j][i] = true;
352 while (head < tail) {
353 int cell = queue[head++], x = cell % MAX_CELLS, y = cell / MAX_CELLS;
354 boundary |= !x || !y || x + 2 == grid->nx || y + 2 == grid->ny;
355 hole.area += (long long)(grid->xs[x + 1] - grid->xs[x]) * (grid->ys[y + 1] - grid->ys[y]);
356 hole.box.x1 = minimum(hole.box.x1, grid->xs[x]);
357 hole.box.y1 = minimum(hole.box.y1, grid->ys[y]);
358 hole.box.x2 = maximum(hole.box.x2, grid->xs[x + 1]);
359 hole.box.y2 = maximum(hole.box.y2, grid->ys[y + 1]);
360 if (grid->ys[y] < hole.anchor_y || (grid->ys[y] == hole.anchor_y && grid->xs[x] < hole.anchor_x)) {
361 hole.anchor_x = grid->xs[x];
362 hole.anchor_y = grid->ys[y];
363 }
364 for (int d = 0; d < 4; d++) {
365 int xx = x + dx[d], yy = y + dy[d];
366 if (xx >= 0 && xx + 1 < grid->nx && yy >= 0 && yy + 1 < grid->ny && !grid->counts[yy][xx] && !seen[yy][xx]) {
367 seen[yy][xx] = true;
368 queue[tail++] = yy * MAX_CELLS + xx;
369 }
370 }
371 }
372 if (!boundary) {
373 if (count == capacity) {
374 capacity = capacity ? capacity * 2 : 8;
375 Hole *next = realloc(holes, (size_t)capacity * sizeof *next);
376 if (!next) exit(EXIT_FAILURE);
377 holes = next;
378 }
379 holes[count++] = hole;
380 }
381 }
382 if (count > 1) qsort(holes, (size_t)count, sizeof *holes, hole_compare);
383 printf("holes %s: %d\n", layer_name, count);
384 for (int i = 0; i < count; i++) {
385 printf("hole %d: area %lld, bbox ", i + 1, holes[i].area);
386 print_box(holes[i].box);
387 putchar('\n');
388 }
389 free(holes);
390}

392static int nets(const Rectangles *rectangles, Rectangles result[MAX_RECTANGLES]) {
393 bool seen[MAX_RECTANGLES] = {false};
394 int count = 0;
395 for (int start = 0; start < rectangles->count; start++) if (!seen[start]) {
396 Rectangles group = {0};
397 int queue[MAX_RECTANGLES], head = 0, tail = 0;
398 queue[tail++] = start;
399 seen[start] = true;
400 while (head < tail) {
401 int i = queue[head++];
402 group.items[group.count++] = rectangles->items[i];
403 for (int j = 0; j < rectangles->count; j++) if (!seen[j] && relation(rectangles->items[i], rectangles->items[j]).kind <= 1) {
404 seen[j] = true;
405 queue[tail++] = j;
406 }
407 }
408 qsort(group.items, (size_t)group.count, sizeof *group.items, pointer_compare);
409 result[count++] = group;
410 }
411 return count;
412}

421static int spacing(const Rectangles *rectangles, int space, Pair result[MAX_PAIRS]) {
422 int count = 0;
423 for (int i = 0; i < rectangles->count; i++) for (int j = i + 1; j < rectangles->count; j++) {
424 Rectangle *a = rectangles->items[i], *b = rectangles->items[j];
425 Relation r = relation(a, b);
426 if (r.kind == 2 || (r.kind == 3 && r.gap < space)) result[count++] = (Pair){a, b, r};
427 }
428 if (count > 1) qsort(result, (size_t)count, sizeof *result, spacing_compare);
429 return count;
430}

462static void check(void) {
463 if (!rule_count) fail("no rules");
464 int total = 0;
465 for (int i = 0; i < rule_count; i++) {
466 Rule *rule = &rules[i];
467 Rectangles list = layer(rule->inner, false);
468 Pair violations[MAX_PAIRS];
469 int count = 0, best[MAX_RECTANGLES], widths[MAX_RECTANGLES];
470 bool marked[MAX_RECTANGLES] = {false};
471 Rectangles outer = layer(rule->outer, false);
472 if (rule->kind == 1) count = spacing(&list, rule->value, violations);
473 else for (int j = 0; j < list.count; j++) {
474 Rectangle *r = list.items[j];
475 if (rule->kind == 2) {
476 widths[j] = minimum(width(r->box), height(r->box));
477 marked[j] = widths[j] < rule->value;
478 } else {
479 best[j] = INT_MIN;
480 for (int k = 0; k < outer.count; k++) best[j] = maximum(best[j], margin(outer.items[k], r));
481 marked[j] = !outer.count || best[j] < rule->value;
482 }
483 count += marked[j];
484 }
485 printf("%s %s", rule_kinds[rule->kind], rule->inner);
486 if (rule->kind == 0) printf(" %s", rule->outer);
487 printf(" %d: %d violations\n", rule->value, count);
488 if (rule->kind == 1) for (int j = 0; j < count; j++) print_spacing(&violations[j]);
489 else for (int j = 0; j < list.count; j++) if (marked[j]) {
490 printf("%s %s ", list.items[j]->name, rule->kind == 2 ? "width" : "best margin");
491 if (rule->kind == 2) printf("%d\n", widths[j]);
492 else if (outer.count) printf("%d\n", best[j]);
493 else puts("none");
494 }
495 total += count;
496 }
497 printf("check: %d violations in %d rules\n", total, rule_count);
498}

500static void command(const char *line, char *parsed) {
501 char *a[8];
502 int argc = 0;
503 for (char *p = strtok(parsed, " "); p; p = strtok(NULL, " ")) a[argc++] = p;
504 const char *c = a[0];
505 if (!strcmp(c, "ADD")) {
506 if (exists(a[1])) fail("duplicate rectangle %s", a[1]);
507 Box b = {atoi(a[3]), atoi(a[4]), atoi(a[5]), atoi(a[6])};
508 if (b.x1 >= b.x2 || b.y1 >= b.y2) fail("invalid rectangle");
509 bounds(b);
510 Rectangle r = {0};
511 strcpy(r.name, a[1]);
512 strcpy(r.layer, a[2]);
513 r.box = b;
514 save(line);
515 add(r);
516 printf("added %s %s ", r.name, r.layer);
517 print_box(b);
518 putchar('\n');
519 } else if (!strcmp(c, "COPY") || !strcmp(c, "MOVE") || !strcmp(c, "RESIZE") || !strcmp(c, "SPLIT") || !strcmp(c, "LAYER") || !strcmp(c, "DELETE")) {
520 int index = get(a[1]);
521 Rectangle old = state.rectangles[index], changed = old, extra = old;
522 bool add_extra = false;
523 if (!strcmp(c, "COPY")) {
524 if (exists(a[2])) fail("duplicate rectangle %s", a[2]);
525 strcpy(extra.name, a[2]);
526 extra.box = shift(old.box, atoi(a[3]), atoi(a[4]));
527 bounds(extra.box);
528 add_extra = true;
529 } else if (!strcmp(c, "MOVE")) {
530 changed.box = shift(old.box, atoi(a[2]), atoi(a[3]));
531 bounds(changed.box);
532 } else if (!strcmp(c, "RESIZE")) {
533 int w = atoi(a[2]), h = atoi(a[3]);
534 if (w <= 0 || h <= 0) fail("invalid size");
535 changed.box.x2 = old.box.x1 + w;
536 changed.box.y2 = old.box.y1 + h;
537 bounds(changed.box);
538 } else if (!strcmp(c, "SPLIT")) {
539 if (exists(a[4])) fail("duplicate rectangle %s", a[4]);
540 int pos = atoi(a[3]);
541 bool x = !strcmp(a[2], "x");
542 if (pos <= (x ? old.box.x1 : old.box.y1) || pos >= (x ? old.box.x2 : old.box.y2)) fail("cut outside %s", old.name);
543 if (x) {
544 changed.box.x2 = pos;
545 extra.box.x1 = pos;
546 } else {
547 changed.box.y2 = pos;
548 extra.box.y1 = pos;
549 }
550 strcpy(extra.name, a[4]);
551 add_extra = true;
552 } else if (!strcmp(c, "LAYER")) {
553 if (!strcmp(old.layer, a[2])) fail("already on %s", old.layer);
554 strcpy(changed.layer, a[2]);
555 }
556 save(line);
557 if (!strcmp(c, "DELETE")) {
558 memmove(state.rectangles + index, state.rectangles + index + 1, (size_t)(state.count - index - 1) * sizeof *state.rectangles);
559 state.count--;
560 printf("deleted %s\n", old.name);
561 } else if (!strcmp(c, "COPY")) {
562 add(extra);
563 printf("copied %s to %s ", old.name, extra.name);
564 print_box(extra.box);
565 putchar('\n');
566 } else {
567 state.rectangles[index] = changed;
568 if (add_extra) add(extra);
569 if (!strcmp(c, "LAYER")) printf("layer %s %s -> %s\n", old.name, old.layer, changed.layer);
570 else if (!strcmp(c, "SPLIT")) {
571 printf("split %s at %s=%s: %s ", old.name, a[2], a[3], old.name);
572 print_box(changed.box);
573 printf(", %s ", extra.name);
574 print_box(extra.box);
575 putchar('\n');
576 } else {
577 printf("%s %s to ", !strcmp(c, "MOVE") ? "moved" : "resized", old.name);
578 print_box(changed.box);
579 putchar('\n');
580 }
581 }
582 } else if (!strcmp(c, "UNDO")) {
583 if (!history_count) fail("nothing to undo");
584 Saved saved = history[--history_count];
585 state = saved.state;
586 printf("undo: %s\n", saved.command);
587 free(saved.command);
588 } else if (!strcmp(c, "LIST")) {
589 printf("rectangles: %d\n", state.count);
590 for (int i = 0; i < state.count; i++) {
591 print_info(&state.rectangles[i]);
592 putchar('\n');
593 }
594 } else if (!strcmp(c, "INFO")) {
595 Rectangle *r = &state.rectangles[get(a[1])];
596 print_info(r);
597 printf(" area %lld\n", area(r->box));
598 const char *labels[] = {"overlaps", "touches", "corners", "encloses", "enclosed by"};
599 for (int k = 0; k < 5; k++) {
600 Rectangles list = {0};
601 for (int i = 0; i < state.count; i++) {
602 Rectangle *other = &state.rectangles[i];
603 if (other == r) continue;
604 Relation rel = relation(r, other);
605 if (k < 3 ? !strcmp(r->layer, other->layer) && rel.kind == k : k == 3 ? contains(r->box, other->box) : contains(other->box, r->box)) list.items[list.count++] = other;
606 }
607 printf("%s: ", labels[k]);
608 names(&list);
609 putchar('\n');
610 }
611 } else if (!strcmp(c, "LAYERS")) {
612 char layers[MAX_RECTANGLES][9];
613 int count = 0;
614 for (int i = 0; i < state.count; i++) {
615 bool found = false;
616 for (int j = 0; j < count; j++) if (!strcmp(layers[j], state.rectangles[i].layer)) found = true;
617 if (!found) strcpy(layers[count++], state.rectangles[i].layer);
618 }
619 for (int i = 1; i < count; i++) {
620 char name[9];
621 strcpy(name, layers[i]);
622 int j = i;
623 while (j > 0 && strcmp(layers[j - 1], name) > 0) {
624 strcpy(layers[j], layers[j - 1]);
625 j--;
626 }
627 strcpy(layers[j], name);
628 }
629 printf("layers: %d\n", count);
630 for (int i = 0; i < count; i++) {
631 Rectangles list = layer(layers[i], true), groups[MAX_RECTANGLES];
632 Grid grid = make_grid(&list);
633 int k = nets(&list, groups);
634 printf("%s: %d rectangles, area %lld, nets %d\n", layers[i], list.count, grid_area(&grid, 1), k);
635 }
636 } else if (!strcmp(c, "OVERLAPS")) {
637 if (argc == 2) layer(a[1], true);
638 Pair pairs[MAX_PAIRS];
639 int count = 0;
640 for (int i = 0; i < state.count; i++) for (int j = i + 1; j < state.count; j++) {
641 Rectangle *x = &state.rectangles[i], *y = &state.rectangles[j];
642 if (strcmp(x->layer, y->layer) || (argc == 2 && strcmp(x->layer, a[1]))) continue;
643 Relation r = relation(x, y);
644 if (r.kind == 0) pairs[count++] = (Pair){x, y, r};
645 }
646 if (count > 1) qsort(pairs, (size_t)count, sizeof *pairs, overlap_compare);
647 printf("overlaps %s: %d\n", argc == 1 ? "all layers" : a[1], count);
648 for (int i = 0; i < count; i++) {
649 if (argc == 1) printf("%s ", pairs[i].a->layer);
650 printf("%s %s ", pairs[i].a->name, pairs[i].b->name);
651 print_box(pairs[i].relation.common);
652 printf(" area %lld\n", area(pairs[i].relation.common));
653 }
654 } else if (!strcmp(c, "INTERSECT") || !strcmp(c, "ADJACENT") || !strcmp(c, "ENCLOSE") || !strcmp(c, "PATH")) {
655 int x_index = get(a[1]);
656 int y_index = get(a[2]);
657 if (x_index == y_index) fail("same rectangle %s", a[1]);
658 Rectangle *x = &state.rectangles[x_index], *y = &state.rectangles[y_index];
659 Relation r = relation(x, y);
660 if (!strcmp(c, "PATH")) {
661 if (strcmp(x->layer, y->layer)) fail("different layers");
662 int previous[MAX_RECTANGLES], queue[MAX_RECTANGLES], head = 0, tail = 0;
663 for (int i = 0; i < state.count; i++) previous[i] = -1;
664 previous[x_index] = x_index;
665 queue[tail++] = x_index;
666 while (head < tail) {
667 int at = queue[head++];
668 for (int j = 0; j < state.count; j++) if (previous[j] < 0 && !strcmp(x->layer, state.rectangles[j].layer) && relation(&state.rectangles[at], &state.rectangles[j]).kind <= 1) {
669 previous[j] = at;
670 queue[tail++] = j;
671 }
672 }
673 printf("path %s %s: ", x->name, y->name);
674 if (previous[y_index] < 0) puts("none");
675 else {
676 int path[MAX_RECTANGLES], count = 0;
677 for (int at = y_index;; at = previous[at]) {
678 path[count++] = at;
679 if (at == x_index) break;
680 }
681 for (int i = count - 1; i >= 0; i--) printf("%s%s", i == count - 1 ? "" : " -> ", state.rectangles[path[i]].name);
682 printf(" (%d steps)\n", count - 1);
683 }
684 } else if (!strcmp(c, "ENCLOSE")) {
685 int m = margin(x, y);
686 printf("enclose %s %s: %s, margins left %d bottom %d right %d top %d", x->name, y->name, m >= 0 ? "yes" : "no", y->box.x1 - x->box.x1, y->box.y1 - x->box.y1, x->box.x2 - y->box.x2, x->box.y2 - y->box.y2);
687 if (m >= 0) printf(", min %d", m);
688 putchar('\n');
689 } else if (!strcmp(c, "INTERSECT")) {
690 printf("intersect %s %s: ", x->name, y->name);
691 if (r.kind == 0) {
692 print_box(r.common);
693 printf(" area %lld\n", area(r.common));
694 } else if (r.kind == 1) {
695 fputs("segment ", stdout);
696 print_box(r.common);
697 printf(" length %d\n", r.ox + r.oy);
698 } else if (r.kind == 2) printf("point (%d,%d)\n", r.common.x1, r.common.y1);
699 else printf("none, gap %d\n", r.gap);
700 } else {
701 printf("adjacent %s %s: ", x->name, y->name);
702 if (r.kind == 0) puts("overlapping");
703 else if (r.kind == 1) {
704 const char *direction = !r.ox ? x->box.x2 == r.common.x1 ? "west" : "east" : x->box.y2 == r.common.y1 ? "south" : "north";
705 printf("%s is %s of %s, edge ", x->name, direction, y->name);
706 print_box(r.common);
707 printf(", length %d\n", r.ox + r.oy);
708 } else if (r.kind == 2) printf("corner only at (%d,%d)\n", r.common.x1, r.common.y1);
709 else puts("not adjacent");
710 }
711 } else if (!strcmp(c, "AT")) {
712 int x = atoi(a[1]), y = atoi(a[2]);
713 bool found = false;
714 printf("at (%d,%d): ", x, y);
715 for (int i = 0; i < state.count; i++) {
716 Rectangle *r = &state.rectangles[i];
717 Box b = r->box;
718 if (x >= b.x1 && x <= b.x2 && y >= b.y1 && y <= b.y2) {
719 bool xx = x == b.x1 || x == b.x2, yy = y == b.y1 || y == b.y2;
720 printf("%s%s %s %s", found ? ", " : "", r->name, r->layer, xx && yy ? "corner" : xx || yy ? "edge" : "inside");
721 found = true;
722 }
723 }
724 if (!found) printf("none");
725 putchar('\n');
726 } else if (!strcmp(c, "NEAREST")) {
727 Rectangle *r = &state.rectangles[get(a[1])], *best = NULL;
728 int gap = INT_MAX;
729 for (int i = 0; i < state.count; i++) {
730 Rectangle *other = &state.rectangles[i];
731 if (r != other && !strcmp(r->layer, other->layer)) {
732 int g = relation(r, other).gap;
733 if (g < gap) {
734 gap = g;
735 best = other;
736 }
737 }
738 }
739 printf("nearest %s: ", r->name);
740 if (best) printf("%s gap %d\n", best->name, gap);
741 else puts("none");
742 } else if (!strcmp(c, "RULE")) {
743 int kind = !strcmp(a[1], "ENCLOSE") ? 0 : !strcmp(a[1], "SPACING") ? 1 : 2;
744 if (kind == 0 && !strcmp(a[2], a[3])) fail("same layer %s", a[2]);
745 Rule rule = {0};
746 rule.kind = kind;
747 strcpy(rule.inner, a[2]);
748 if (!kind) strcpy(rule.outer, a[3]);
749 rule.value = atoi(a[kind ? 3 : 4]);
750 int i = 0;
751 while (i < rule_count && rule_compare(&rules[i], &rule)) i++;
752 if (i == rule_count) rule_count++;
753 rules[i] = rule;
754 qsort(rules, (size_t)rule_count, sizeof *rules, rule_compare);
755 printf("rule %s %s", rule_kinds[kind], rule.inner);
756 if (!kind) printf(" %s", rule.outer);
757 printf(" %d\n", rule.value);
758 } else if (!strcmp(c, "CHECK")) check();
759 else {
760 const char *layer_name = !strcmp(c, "COVER") ? a[2] : a[1];
761 Rectangles list = layer(layer_name, true);
762 Grid grid = make_grid(&list);
763 Box box = grid_bounds(&grid);
764 if (!strcmp(c, "AREA")) printf("area %s: %lld\n", layer_name, grid_area(&grid, 1));
765 else if (!strcmp(c, "COVER")) printf("cover %s >= %s: %lld\n", layer_name, a[1], grid_area(&grid, atoi(a[1])));
766 else if (!strcmp(c, "PERIMETER")) printf("perimeter %s: %lld\n", layer_name, perimeter(&grid));
767 else if (!strcmp(c, "NETS")) {
768 Rectangles groups[MAX_RECTANGLES];
769 int count = nets(&list, groups);
770 printf("nets %s: %d\n", layer_name, count);
771 for (int i = 0; i < count; i++) {
772 Grid net_grid = make_grid(&groups[i]);
773 printf("net %d: ", i + 1);
774 names(&groups[i]);
775 printf(" area %lld\n", grid_area(&net_grid, 1));
776 }
777 } else if (!strcmp(c, "DRC")) {
778 Pair violations[MAX_PAIRS];
779 int s = atoi(a[2]), count = spacing(&list, s, violations);
780 printf("drc %s spacing %d: %d violations\n", layer_name, s, count);
781 for (int i = 0; i < count; i++) print_spacing(&violations[i]);
782 } else if (!strcmp(c, "BBOX")) {
783 printf("bbox %s: ", layer_name);
784 print_box(box);
785 printf(" %dx%d, %d rectangles, density ", width(box), height(box), list.count);
786 density(grid_area(&grid, 1), area(box));
787 puts("%");
788 } else if (!strcmp(c, "DENSITY")) {
789 int w = atoi(a[2]), columns = (width(box) + w - 1) / w, rows = (height(box) + w - 1) / w;
790 if (columns > 20 || rows > 20) fail("too many windows");
791 long long values[20][20], low = LLONG_MAX, high = -1;
792 int low_x = 0, low_y = 0, high_x = 0, high_y = 0;
793 for (int j = 0; j < rows; j++) for (int i = 0; i < columns; i++) {
794 values[j][i] = clipped(&grid, (Box){box.x1 + i * w, box.y1 + j * w, box.x1 + (i + 1) * w, box.y1 + (j + 1) * w});
795 if (values[j][i] < low) {
796 low = values[j][i];
797 low_x = i;
798 low_y = j;
799 }
800 if (values[j][i] > high) {
801 high = values[j][i];
802 high_x = i;
803 high_y = j;
804 }
805 }
806 printf("density %s window %d: %dx%d\n", layer_name, w, columns, rows);
807 for (int j = rows - 1; j >= 0; j--) {
808 for (int i = 0; i < columns; i++) printf("%s%lld", i ? " " : "", 100LL * values[j][i] / ((long long)w * w));
809 putchar('\n');
810 }
811 fputs("min ", stdout);
812 density(low, (long long)w * w);
813 printf("%% at (%d,%d)\n", box.x1 + low_x * w, box.y1 + low_y * w);
814 fputs("max ", stdout);
815 density(high, (long long)w * w);
816 printf("%% at (%d,%d)\n", box.x1 + high_x * w, box.y1 + high_y * w);
817 } else if (!strcmp(c, "RENDER")) {
818 Box window = {atoi(a[2]), atoi(a[3]), atoi(a[4]), atoi(a[5])};
819 if (window.x1 >= window.x2 || window.y1 >= window.y2) fail("invalid window");
820 if (width(window) > 60 || height(window) > 40) fail("window too large");
821 printf("render %s ", layer_name);
822 print_box(window);
823 putchar('\n');
824 for (int y = window.y2 - 1; y >= window.y1; y--) {
825 for (int x = window.x1; x < window.x2; x++) {
826 int count = 0;
827 for (int i = 0; i < list.count; i++) {
828 Box b = list.items[i]->box;
829 if (x >= b.x1 && x < b.x2 && y >= b.y1 && y < b.y2) count++;
830 }
831 putchar(count == 0 ? '.' : count < 10 ? '0' + count : '*');
832 }
833 putchar('\n');
834 }
835 } else if (!strcmp(c, "EMPTY")) {
836 Box empty;
837 printf("empty %s: ", layer_name);
838 if (empty_rectangle(&grid, &empty)) {
839 print_box(empty);
840 printf(" area %lld\n", area(empty));
841 } else puts("none");
842 } else if (!strcmp(c, "HOLES")) print_holes(&grid, layer_name);
843 else if (!strcmp(c, "MERGE")) {
844 Boxes result = pieces(&grid);
845 printf("merge %s: %d pieces\n", layer_name, result.count);
846 for (int i = 0; i < result.count; i++) {
847 print_box(result.items[i]);
848 putchar('\n');
849 }
850 free(result.items);
851 }
852 }
