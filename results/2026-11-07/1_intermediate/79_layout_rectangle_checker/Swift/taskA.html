20struct Rect {
21 var name: String
22 var layer: String
23 var x1: Int
24 var y1: Int
25 var x2: Int
26 var y2: Int
27
28 init(_ name: String, _ layer: String, _ x1: Int, _ y1: Int, _ x2: Int, _ y2: Int) {
29 self.name = name
30 self.layer = layer
31 self.x1 = x1
32 self.y1 = y1
33 self.x2 = x2
34 self.y2 = y2
35 }
36 var width: Int {
37 x2 - x1
38 }
39 var height: Int {
40 y2 - y1
41 }
42 var area: Int {
43 width * height
44 }
45 var text: String {
46 "(\(x1),\(y1))-(\(x2),\(y2))"
47 }
48
49 func contains(_ b: Rect) -> Bool {
50 x1 <= b.x1 && y1 <= b.y1 && x2 >= b.x2 && y2 >= b.y2
51 }
52
53 func margins(_ b: Rect) -> [Int] {
54 [b.x1 - x1, b.y1 - y1, x2 - b.x2, y2 - b.y2]
55 }
56
57 func moved(_ dx: Int, _ dy: Int) -> Rect {
58 Rect(name, layer, x1 + dx, y1 + dy, x2 + dx, y2 + dy)
59 }
60
61 func checkBounds() throws {
62 if [x1, y1, x2, y2].contains(where: {
63 abs($0) > 10000
64 }) {
65 try fail("out of bounds")
66 }
67 }
68}

70struct Relation {
71 let common: Rect
72 let ox: Int
73 let oy: Int
74 let kind: Int
75 let gap: Int
76
77 init(_ a: Rect, _ b: Rect) {
78 let overlap = Rect("", "", max(a.x1, b.x1), max(a.y1, b.y1), min(a.x2, b.x2), min(a.y2, b.y2))
79 common = overlap
80 let x = overlap.width
81 let y = overlap.height
82 ox = x
83 oy = y
84 kind = x > 0 && y > 0 ? 0 : x >= 0 && y >= 0 && x + y > 0 ? 1 : x == 0 && y == 0 ? 2 : 3
85 gap = max(0, max(-x, -y))
86 }
87 var direction: String {
88 kind == 2 ? "corner" : ox < 0 && oy < 0 ? "diagonal" : ox < 0 ? "x" : "y"
89 }
90}
91
92func connected(_ a: Rect, _ b: Rect) -> Bool {
93 Relation(a, b).kind <= 1
94}

96func bbox(_ rs: [Rect]) -> Rect {
97 Rect("", "", rs.map(\.x1).min()!, rs.map(\.y1).min()!, rs.map(\.x2).max()!, rs.map(\.y2).max()!)
98}

105final class Grid {
106 let xs: [Int]
107 let ys: [Int]
108 let count: [[Int]]
109 var nx: Int {
110 xs.count - 1
111 }
112 var ny: Int {
113 ys.count - 1
114 }
115
116 init(_ rs: [Rect]) {
117 let xs = Set(rs.flatMap {
118 [$0.x1, $0.x2]
119 }).sorted()
120 let ys = Set(rs.flatMap {
121 [$0.y1, $0.y2]
122 }).sorted()
123 self.xs = xs
124 self.ys = ys
125 count = (0..<(ys.count - 1)).map {
126 y in (0..<(xs.count - 1)).map {
127 x in rs.filter {
128 $0.x1 <= xs[x] && $0.x2 > xs[x] && $0.y1 <= ys[y] && $0.y2 > ys[y]
129 }.count
130 }
131 }
132 }
133
134 func area(_ k: Int = 1, _ window: Rect? = nil) -> Int {
135 var sum = 0
136 for y in 0..<ny {
137 for x in 0..<nx where count[y][x] >= k {
138 let w = max(0, min(xs[x + 1], window?.x2 ?? xs[x + 1]) - max(xs[x], window?.x1 ?? xs[x]))
139 let h = max(0, min(ys[y + 1], window?.y2 ?? ys[y + 1]) - max(ys[y], window?.y1 ?? ys[y]))
140 sum += w * h
141 }
142 }
143 return sum
144 }
145
146 func perimeter() -> Int {
147 var sum = 0
148 for y in 0..<ny {
149 for x in 0..<nx where count[y][x] > 0 {
150 if x == 0 || count[y][x - 1] == 0 {
151 sum += ys[y + 1] - ys[y]
152 }
153 if x + 1 == nx || count[y][x + 1] == 0 {
154 sum += ys[y + 1] - ys[y]
155 }
156 if y == 0 || count[y - 1][x] == 0 {
157 sum += xs[x + 1] - xs[x]
158 }
159 if y + 1 == ny || count[y + 1][x] == 0 {
160 sum += xs[x + 1] - xs[x]
161 }
162 }
163 }
164 return sum
165 }
166
167 func empty() -> Rect? {
168 var best: Rect?
169 for bottom in 0..<ny {
170 var free = Array(repeating: true, count: nx)
171 for top in bottom..<ny {
172 for x in 0..<nx {
173 free[x] = free[x] && count[top][x] == 0
174 }
175 var x = 0
176 while x < nx {
177 if !free[x] {
178 x += 1
179 continue
180 }
181 let start = x
182 while x < nx && free[x] {
183 x += 1
184 }
185 let r = Rect("", "", xs[start], ys[bottom], xs[x], ys[top + 1])
186 if best == nil || (-r.area, r.x1, r.y1, r.width) < (-best!.area, best!.x1, best!.y1, best!.width) {
187 best = r
188 }
189 }
190 }
191 }
192 return best
193 }

195 func holes() -> [(Int, Rect)] {
196 var seen = Array(repeating: Array(repeating: false, count: nx), count: ny)
197 var result = [(Int, Int, Int, Rect)]()
198 for y in 0..<ny {
199 for x in 0..<nx {
200 if count[y][x] > 0 || seen[y][x] {
201 continue
202 }
203 var queue = [(x, y)]
204 var at = 0
205 seen[y][x] = true
206 var boundary = false
207 var area = 0
208 var r = Rect("", "", xs[x], ys[y], xs[x + 1], ys[y + 1])
209 var anchor = (ys[y], xs[x])
210 while at < queue.count {
211 let (cx, cy) = queue[at]
212 at += 1
213 boundary = boundary || cx == 0 || cy == 0 || cx + 1 == nx || cy + 1 == ny
214 area += (xs[cx + 1] - xs[cx]) * (ys[cy + 1] - ys[cy])
215 r.x1 = min(r.x1, xs[cx])
216 r.y1 = min(r.y1, ys[cy])
217 r.x2 = max(r.x2, xs[cx + 1])
218 r.y2 = max(r.y2, ys[cy + 1])
219 if (ys[cy], xs[cx]) < anchor {
220 anchor = (ys[cy], xs[cx])
221 }
222 for (dx, dy) in [(-1, 0), (1, 0), (0, -1), (0, 1)] {
223 let xx = cx + dx
224 let yy = cy + dy
225 if xx >= 0 && xx < nx && yy >= 0 && yy < ny && !seen[yy][xx] && count[yy][xx] == 0 {
226 seen[yy][xx] = true
227 queue.append((xx, yy))
228 }
229 }
230 }
231 if !boundary {
232 result.append((anchor.0, anchor.1, area, r))
233 }
234 }
235 }
236 return result.sorted {
237 ($0.0, $0.1) < ($1.0, $1.1)
238 }.map {
239 ($0.2, $0.3)
240 }
241 }

243 func pieces() -> [Rect] {
244 var active = [Interval: Rect]()
245 var result = [Rect]()
246 for y in 0..<ny {
247 var next = [Interval: Rect]()
248 var x = 0
249 while x < nx {
250 if count[y][x] == 0 {
251 x += 1
252 continue
253 }
254 let start = x
255 while x < nx && count[y][x] > 0 {
256 x += 1
257 }
258 let key = Interval(left: xs[start], right: xs[x])
259 var r = active[key] ?? Rect("", "", xs[start], ys[y], xs[x], ys[y + 1])
260 r.y2 = ys[y + 1]
261 next[key] = r
262 }
263 for (key, r) in active where next[key] == nil {
264 result.append(r)
265 }
266 active = next
267 }
268 result.append(contentsOf: active.values)
269 return result.sorted {
270 ($0.y1, $0.x1) < ($1.y1, $1.x1)
271 }
272 }

275struct Violation {
276 let a: Rect
277 let b: Rect
278 let relation: Relation
279 var text: String {
280 "\(a.name) \(b.name) gap \(relation.gap) \(relation.direction)"
281 }
282}

284func spacing(_ rs: [Rect], _ s: Int) -> [Violation] {
285 var result = [Violation]()
286 for i in rs.indices {
287 for j in (i + 1)..<rs.count {
288 let r = Relation(rs[i], rs[j])
289 if r.kind == 2 || r.kind == 3 && r.gap < s {
290 result.append(Violation(a: rs[i], b: rs[j], relation: r))
291 }
292 }
293 }
294 return result.sorted {
295 ($0.relation.gap, $0.a.name, $0.b.name) < ($1.relation.gap, $1.a.name, $1.b.name)
296 }
297}

299func nets(_ rs: [Rect]) -> [[Rect]] {
300 var seen = Set<String>()
301 var groups = [[Rect]]()
302 for r in rs {
303 if !seen.insert(r.name).inserted {
304 continue
305 }
306 var queue = [r]
307 var at = 0
308 while at < queue.count {
309 let a = queue[at]
310 at += 1
311 for b in rs where !seen.contains(b.name) && connected(a, b) {
312 seen.insert(b.name)
313 queue.append(b)
314 }
315 }
316 groups.append(queue.sorted {
317 $0.name < $1.name
318 })
319 }
320 return groups
321}

332final class Layout {
333 var rectangles = [String: Rect]()
334 var history = [([String: Rect], String)]()
335 var rules = [RuleKey: Int]()
336 var all: [Rect] {
337 rectangles.values.sorted {
338 $0.name < $1.name

343 guard let r = rectangles[name] else {

363 func command(_ line: String) throws -> [String] {
364 let w = line.components(separatedBy: " ")
365 let c = w[0]
366
367 func n(_ i: Int) -> Int {
368 Int(w[i])!
369 }
370 var out = [String]()
371 if ["ADD", "COPY", "MOVE", "RESIZE", "SPLIT", "LAYER", "DELETE"].contains(c) {
372 var next = rectangles
373 let name = w[1]
374 switch c {
375 case "ADD":
376 if rectangles[name] != nil {
377 try fail("duplicate rectangle \(name)")
378 }
379 let r = Rect(name, w[2], n(3), n(4), n(5), n(6))
380 if r.width <= 0 || r.height <= 0 {
381 try fail("invalid rectangle")
382 }
383 try r.checkBounds()
384 next[name] = r
385 out.append("added \(name) \(r.layer) \(r.text)")
386 case "COPY":
387 let old = try get(name)
388 if rectangles[w[2]] != nil {
389 try fail("duplicate rectangle \(w[2])")
390 }
391 var r = old.moved(n(3), n(4))
392 r.name = w[2]
393 try r.checkBounds()
394 next[r.name] = r
395 out.append("copied \(name) to \(r.name) \(r.text)")
396 case "MOVE":
397 let r = try get(name).moved(n(2), n(3))
398 try r.checkBounds()
399 next[name] = r
400 out.append("moved \(name) to \(r.text)")
401 case "RESIZE":
402 var r = try get(name)
403 if n(2) <= 0 || n(3) <= 0 {
404 try fail("invalid size")
405 }
406 r.x2 = r.x1 + n(2)
407 r.y2 = r.y1 + n(3)
408 try r.checkBounds()
409 next[name] = r
410 out.append("resized \(name) to \(r.text)")
411 case "SPLIT":
412 let old = try get(name)
413 if rectangles[w[4]] != nil {
414 try fail("duplicate rectangle \(w[4])")
415 }
416 let p = n(3)
417 let x = w[2] == "x"
418 if p <= (x ? old.x1 : old.y1) || p >= (x ? old.x2 : old.y2) {
419 try fail("cut outside \(name)")
420 }
421 var a = old
422 var b = old
423 b.name = w[4]
424 if x {
425 a.x2 = p
426 b.x1 = p
427 } else {
428 a.y2 = p
429 b.y1 = p
430 }
431 next[name] = a
432 next[b.name] = b
433 out.append("split \(name) at \(w[2])=\(p): \(name) \(a.text), \(b.name) \(b.text)")
434 case "LAYER":
435 var r = try get(name)
436 if r.layer == w[2] {
437 try fail("already on \(w[2])")
438 }
439 out.append("layer \(name) \(r.layer) -> \(w[2])")
440 r.layer = w[2]
441 next[name] = r
442 default:
443 _ = try get(name)
444 next.removeValue(forKey: name)
445 out.append("deleted \(name)")
446 }
447 history.append((rectangles, line))
448 rectangles = next
449 return out
450 }
451 switch c {
452 case "UNDO":
453 guard let (old, text) = history.popLast() else {
454 try fail("nothing to undo")
455 }
456 rectangles = old
457 return ["undo: \(text)"]
458 case "LIST":
459 return ["rectangles: \(rectangles.count)"] + all.map {
460 "\($0.name) \($0.layer) \($0.text) \($0.width)x\($0.height)"
461 }
462 case "INFO":
463 let a = try get(w[1])
464 let others = all.filter {
465 $0.name != a.name
466 }
467 out.append("\(a.name) \(a.layer) \(a.text) \(a.width)x\(a.height) area \(a.area)")
468 for (label, kind) in [("overlaps", 0), ("touches", 1), ("corners", 2)] {
469 out.append(label + ": " + names(others.filter {
470 $0.layer == a.layer && Relation(a, $0).kind == kind
471 }.map(\.name)))
472 }
473 out.append("encloses: " + names(others.filter {
474 a.contains($0)
475 }.map(\.name)))
476 out.append("enclosed by: " + names(others.filter {
477 $0.contains(a)
478 }.map(\.name)))
479 return out
480 case "LAYERS":
481 let layers = Set(all.map(\.layer)).sorted()
482 out.append("layers: \(layers.count)")
483 for l in layers {
484 let rs = layer(l)
485 out.append("\(l): \(rs.count) rectangles, area \(Grid(rs).area()), nets \(nets(rs).count)")
486 }
487 return out
488 case "INTERSECT", "ADJACENT", "ENCLOSE", "PATH":
489 let a = try get(w[1])
490 let b = try get(w[2])
491 if a.name == b.name {
492 try fail("same rectangle \(a.name)")
493 }
494 let r = Relation(a, b)
495 let q = r.common
496 if c == "INTERSECT" {
497 let text: String
498 switch r.kind {
499 case 0:
500 text = "\(q.text) area \(q.area)"
501 case 1:
502 text = "segment \(q.text) length \(r.ox + r.oy)"
503 case 2:
504 text = "point (\(q.x1),\(q.y1))"
505 default:
506 text = "none, gap \(r.gap)"
507 }
508 return ["intersect \(a.name) \(b.name): \(text)"]
509 }
510 if c == "ADJACENT" {
511 let d = r.ox == 0 ? (a.x2 == q.x1 ? "west" : "east") : (a.y2 == q.y1 ? "south" : "north")
512 let text: String
513 switch r.kind {
514 case 0:
515 text = "overlapping"
516 case 1:
517 text = "\(a.name) is \(d) of \(b.name), edge \(q.text), length \(r.ox + r.oy)"
518 case 2:
519 text = "corner only at (\(q.x1),\(q.y1))"
520 default:
521 text = "not adjacent"
522 }
523 return ["adjacent \(a.name) \(b.name): \(text)"]
524 }
525 if c == "ENCLOSE" {
526 let m = a.margins(b)
527 return ["enclose \(a.name) \(b.name): " + (a.contains(b) ? "yes" : "no") + ", margins left \(m[0]) bottom \(m[1]) right \(m[2]) top \(m[3])" + (a.contains(b) ? ", min \(m.min()!)" : "")]
528 }
529 if a.layer != b.layer {
530 try fail("different layers")
531 }
532 let rs = layer(a.layer)
533 var previous = [String: String]()
534 var seen: Set<String> = [a.name]
535 var queue = [a]
536 var at = 0
537 while at < queue.count {
538 let p = queue[at]
539 at += 1
540 for q in rs where !seen.contains(q.name) && connected(p, q) {
541 seen.insert(q.name)
542 previous[q.name] = p.name
543 queue.append(q)
544 }
545 }
546 if !seen.contains(b.name) {
547 return ["path \(a.name) \(b.name): none"]
548 }
549 var path = [b.name]
550 while path.last! != a.name {
551 path.append(previous[path.last!]!)
552 }
553 path.reverse()
554 return ["path \(a.name) \(b.name): \(path.joined(separator: " -> ")) (\(path.count - 1) steps)"]
555 case "AT":
556 let x = n(1)
557 let y = n(2)
558 let found = all.filter {
559 $0.x1 <= x && x <= $0.x2 && $0.y1 <= y && y <= $0.y2
560 }.map {
561 r -> String in let xx = x == r.x1 || x == r.x2
562 let yy = y == r.y1 || y == r.y2
563 return "\(r.name) \(r.layer) " + (xx && yy ? "corner" : xx || yy ? "edge" : "inside")
564 }
565 return ["at (\(x),\(y)): " + (found.isEmpty ? "none" : found.joined(separator: ", "))]
566 case "NEAREST":
567 let a = try get(w[1])
568 let b = layer(a.layer).filter {
569 $0.name != a.name
570 }.min {
571 (Relation(a, $0).gap, $0.name) < (Relation(a, $1).gap, $1.name)
572 }
573 return ["nearest \(a.name): " + (b.map {
574 "\($0.name) gap \(Relation(a, $0).gap)"
575 } ?? "none")]
576 case "OVERLAPS":
577 let rs: [Rect]
578 if w.count == 2 {
579 rs = try checkedLayer(w[1])
580 } else {
581 rs = all
582 }
583 var found = [(Rect, Rect, Rect)]()
584 for i in rs.indices {
585 for j in (i + 1)..<rs.count where rs[i].layer == rs[j].layer {
586 let r = Relation(rs[i], rs[j])
587 if r.kind == 0 {
588 found.append((rs[i], rs[j], r.common))
589 }
590 }
591 }
592 found.sort {
593 ($0.0.layer, $0.0.name, $0.1.name) < ($1.0.layer, $1.0.name, $1.1.name)
594 }
595 out.append("overlaps " + (w.count == 1 ? "all layers" : w[1]) + ": \(found.count)")
596 for (a, b, r) in found {
597 out.append((w.count == 1 ? "\(a.layer) " : "") + "\(a.name) \(b.name) \(r.text) area \(r.area)")
598 }
599 return out
600 case "RULE":
601 let kind = w[1].lowercased()
602 if kind == "enclose" && w[2] == w[3] {
603 try fail("same layer \(w[2])")
604 }
605 let outer = kind == "enclose" ? w[3] : ""
606 let v = n(kind == "enclose" ? 4 : 3)
607 rules[RuleKey(kind: kind, inner: w[2], outer: outer)] = v
608 return ["rule \(kind) \(w[2])" + (outer.isEmpty ? "" : " \(outer)") + " \(v)"]
609 case "CHECK":
610 if rules.isEmpty {
611 try fail("no rules")
612 }
613 var total = 0
614 for rule in rules.keys.sorted() {
615 let v = rules[rule]!
616 let rs = layer(rule.inner)
617 var found = [String]()
618 if rule.kind == "spacing" {
619 found = spacing(rs, v).map(\.text)
620 } else if rule.kind == "width" {
621 found = rs.filter {
622 min($0.width, $0.height) < v
623 }.map {
624 "\($0.name) width \(min($0.width, $0.height))"
625 }
626 } else {
627 let outer = layer(rule.outer)
628 for r in rs {
629 let best = outer.map {
630 $0.margins(r).min()!
631 }.max()
632 if best == nil || best! < v {
633 found.append("\(r.name) best margin " + (best.map(String.init) ?? "none"))
634 }
635 }
636 }
637 total += found.count
638 out.append("\(rule.kind) \(rule.inner)" + (rule.kind == "enclose" ? " \(rule.outer)" : "") + " \(v): \(found.count) violations")
639 out += found
640 }
641 out.append("check: \(total) violations in \(rules.count) rules")
642 return out
643 default:
644 break
645 }
646 let rs = try checkedLayer(c == "COVER" ? w[2] : w[1])
647 let l = rs[0].layer
648 let g = Grid(rs)
649 let b = bbox(rs)
650 switch c {
651 case "AREA":
652 return ["area \(l): \(g.area())"]
653 case "COVER":
654 return ["cover \(l) >= \(w[1]): \(g.area(n(1)))"]
655 case "PERIMETER":
656 return ["perimeter \(l): \(g.perimeter())"]
657 case "NETS":
658 let groups = nets(rs)
659 out.append("nets \(l): \(groups.count)")
660 for (i, group) in groups.enumerated() {
661 out.append("net \(i + 1): \(names(group.map(\.name))) area \(Grid(group).area())")
662 }
663 return out
664 case "DRC":
665 let violations = spacing(rs, n(2))
666 return ["drc \(l) spacing \(w[2]): \(violations.count) violations"] + violations.map(\.text)
667 case "BBOX":
668 return ["bbox \(l): \(b.text) \(b.width)x\(b.height), \(rs.count) rectangles, density \(decimal(100 * g.area(), b.area))%"]
669 case "DENSITY":
670 let size = n(2)
671 let cols = (b.width + size - 1) / size
672 let rows = (b.height + size - 1) / size
673 if cols > 20 || rows > 20 {
674 try fail("too many windows")
675 }
676 let values = (0..<rows).map {
677 y in (0..<cols).map {
678 x -> Int in let xx = b.x1 + x * size
679 let yy = b.y1 + y * size
680 return g.area(1, Rect("", "", xx, yy, xx + size, yy + size))
681 }
682 }
683 let denominator = size * size
684 out.append("density \(l) window \(size): \(cols)x\(rows)")
685 for row in values.reversed() {
686 out.append(row.map {
687 String(100 * $0 / denominator)
688 }.joined(separator: " "))
689 }
690 var minX = 0
691 var minY = 0
692 var maxX = 0
693 var maxY = 0
694 for y in 0..<rows {
695 for x in 0..<cols {
696 if values[y][x] < values[minY][minX] {
697 minX = x
698 minY = y
699 }
