95class Relation:
96 def __init__(self, a, b):
97 self.x1 = max(a.x1, b.x1)
98 self.y1 = max(a.y1, b.y1)
99 self.x2 = min(a.x2, b.x2)
100 self.y2 = min(a.y2, b.y2)
101 self.ox = self.x2 - self.x1
102 self.oy = self.y2 - self.y1
103 if self.ox > 0 and self.oy > 0:
104 self.kind = Kind.OVERLAP
105 elif self.ox < 0 or self.oy < 0:
106 self.kind = Kind.SEPARATE
107 elif self.ox == 0 and self.oy == 0:
108 self.kind = Kind.CORNER
109 else:
110 self.kind = Kind.EDGE
111 self.gap = max(0, -self.ox, -self.oy)
112
113 def common(self):
114 return box(self.x1, self.y1, self.x2, self.y2)

116 def corner(self):
117 return point(self.x1, self.y1)

119 def direction(self):
120 if self.ox < 0 and self.oy < 0:
121 return "diagonal"
122 return "x" if self.ox < 0 else "y"

144def spacing_violations(rects, spacing):
145 found = []
146 for a, b in combinations(rects, 2):
147 relation = Relation(a, b)
148 if relation.kind == Kind.CORNER:
149 found.append((0, a.name, b.name, "corner"))
150 elif relation.kind == Kind.SEPARATE and relation.gap < spacing:
151 found.append((relation.gap, a.name, b.name, relation.direction()))
152 found.sort()
153 return [f"{a} {b} gap {gap} {where}" for gap, a, b, where in found]

156def find_nets(rects):
157 seen = set()
158 groups = []
159 for start in rects:
160 if start.name in seen:
161 continue
162 seen.add(start.name)
163 group = [start]
164 stack = [start]
165 while stack:
166 current = stack.pop()
167 for other in rects:
168 if other.name not in seen and connected(current, other):
169 seen.add(other.name)
170 group.append(other)
171 stack.append(other)
172 groups.append(sorted(group, key=lambda rect: rect.name))
173 return groups

176def runs(flags, coordinates):
177 found = []
178 start = 0
179 for flag, group in groupby(flags):
180 end = start + len(list(group))
181 if flag:
182 found.append((coordinates[start], coordinates[end]))
183 start = end
184 return found

187class Coverage:
188
189 def __init__(self, rects, extra_xs=(), extra_ys=()):
190 self.xs = sorted({x for rect in rects for x in (rect.x1, rect.x2)}.union(extra_xs))
191 self.ys = sorted({y for rect in rects for y in (rect.y1, rect.y2)}.union(extra_ys))
192 self.columns = len(self.xs) - 1
193 self.rows = len(self.ys) - 1
194 x_index = {x: i for i, x in enumerate(self.xs)}
195 y_index = {y: j for j, y in enumerate(self.ys)}
196 self.counts = [[0] * self.rows for _ in range(self.columns)]
197 for rect in rects:
198 for i in range(x_index[rect.x1], x_index[rect.x2]):
199 column = self.counts[i]
200 for j in range(y_index[rect.y1], y_index[rect.y2]):
201 column[j] += 1
202
203 def cells(self):
204 for i in range(self.columns):
205 for j in range(self.rows):
206 yield i, j
207
208 def covered(self, i, j):
209 return 0 <= i < self.columns and 0 <= j < self.rows and self.counts[i][j] > 0
210
211 def cell_area(self, i, j):
212 return (self.xs[i + 1] - self.xs[i]) * (self.ys[j + 1] - self.ys[j])
213
214 def area(self, minimum=1):
215 return sum(self.cell_area(i, j) for i, j in self.cells()
216 if self.counts[i][j] >= minimum)
217
218 def perimeter(self):
219 total = 0
220 for i in range(self.columns + 1):
221 for j in range(self.rows):
222 if self.covered(i - 1, j) != self.covered(i, j):
223 total += self.ys[j + 1] - self.ys[j]
224 for j in range(self.rows + 1):
225 for i in range(self.columns):
226 if self.covered(i, j - 1) != self.covered(i, j):
227 total += self.xs[i + 1] - self.xs[i]
228 return total
229
230 def flood(self, start):
231 component = {start}
232 stack = [start]
233 while stack:
234 i, j = stack.pop()
235 for neighbor in ((i - 1, j), (i + 1, j), (i, j - 1), (i, j + 1)):
236 ni, nj = neighbor
237 if (0 <= ni < self.columns and 0 <= nj < self.rows
238 and self.counts[ni][nj] == 0 and neighbor not in component):
239 component.add(neighbor)
240 stack.append(neighbor)
241 return component
242
243 def holes(self):
244 seen = set()
245 found = []
246 for j in range(self.rows):
247 for i in range(self.columns):
248 if (i, j) in seen or self.counts[i][j] > 0:
249 continue
250 component = self.flood((i, j))
251 seen.update(component)
252 if any(ci in (0, self.columns - 1) or cj in (0, self.rows - 1)
253 for ci, cj in component):
254 continue
255 area = sum(self.cell_area(ci, cj) for ci, cj in component)
256 bounds = (min(self.xs[ci] for ci, _ in component),
257 min(self.ys[cj] for _, cj in component),
258 max(self.xs[ci + 1] for ci, _ in component),
259 max(self.ys[cj + 1] for _, cj in component))
260 found.append((area, bounds))
261 return found
262
263 def pieces(self):
264 row_runs = [runs([self.counts[i][j] > 0 for i in range(self.columns)], self.xs)
265 for j in range(self.rows)]
266 found = []
267 for j, current in enumerate(row_runs):
268 for left, right in current:
269 if j > 0 and (left, right) in row_runs[j - 1]:
270 continue
271 top = j + 1
272 while top < self.rows and (left, right) in row_runs[top]:
273 top += 1
274 found.append((left, self.ys[j], right, self.ys[top]))
275 return found
276
277 def largest_empty(self):
278 candidates = []
279 for a in range(self.columns):
280 free = [True] * self.rows
281 for b in range(a + 1, self.columns + 1):
282 free = [is_free and self.counts[b - 1][j] == 0
283 for j, is_free in enumerate(free)]
284 width = self.xs[b] - self.xs[a]
285 for bottom, top in runs(free, self.ys):
286 candidates.append((-width * (top - bottom), self.xs[a], bottom, width, top))
287 if not candidates:
288 return None
289 _, left, bottom, width, top = min(candidates)
290 return left, bottom, left + width, top

500 def adjacent(self, a, b):
501 first, second = self.pair(a, b)

504 case Kind.OVERLAP:

543 def path(self, a, b):
544 first, second = self.pair(a, b)
545 if first.layer != second.layer:
546 raise CommandError("different layers")
547 rects = self.on_layer(first.layer)
548 links = {rect.name: [other.name for other in rects
549 if other.name != rect.name and connected(rect, other)]
550 for rect in rects}
551 distance = {b: 0}
552 queue = deque([b])
553 while queue:
554 current = queue.popleft()
555 for neighbor in links[current]:
556 if neighbor not in distance:
557 distance[neighbor] = distance[current] + 1
558 queue.append(neighbor)
559 if a not in distance:
560 print(f"path {a} {b}: none")
561 return
562 route = [a]
563 while route[-1] != b:
564 remaining = distance[route[-1]] - 1
565 route.append(min(name for name in links[route[-1]]
566 if distance.get(name) == remaining))
567 print(f"path {a} {b}: {' -> '.join(route)} ({len(route) - 1} steps)")

591 def density(self, layer, window):
592 rects = self.require_layer(layer)
593 x1, y1, x2, y2 = bounding_box(rects)
594 columns = -(-(x2 - x1) // window)
595 rows = -(-(y2 - y1) // window)
596 if columns > MAX_TILES or rows > MAX_TILES:
597 raise CommandError("too many windows")
598 coverage = Coverage(rects, [x1 + i * window for i in range(columns + 1)],
599 [y1 + j * window for j in range(rows + 1)])
600 covered = {(i, j): 0 for j in range(rows) for i in range(columns)}
601 for i, j in coverage.cells():
602 if coverage.counts[i][j] > 0:
603 tile = ((coverage.xs[i] - x1) // window, (coverage.ys[j] - y1) // window)
604 covered[tile] += coverage.cell_area(i, j)
605 tile_area = window * window
606 print(f"density {layer} window {window}: {columns}x{rows}")
607 for j in reversed(range(rows)):
608 print(" ".join(str(100 * covered[i, j] // tile_area) for i in range(columns)))
609 for label, choose in (("min", min), ("max", max)):
610 i, j = choose(covered, key=covered.get)
611 percent = two_decimals(100 * covered[i, j], tile_area)
612 print(f"{label} {percent}% at {point(x1 + i * window, y1 + j * window)}")

614 def render(self, layer, x1, y1, x2, y2):

619 raise CommandError("window too large")

622 counts = (sum(r.x1 <= i < r.x2 and r.y1 <= j < r.y2 for r in rects)
623 for i in range(x1, x2))
624 lines.append("".join(symbol(count) for count in counts))
625 emit(f"render {layer} {box(x1, y1, x2, y2)}", lines)

641 def merge(self, layer):
642 pieces = Coverage(self.require_layer(layer)).pieces()
643 emit(f"merge {layer}: {len(pieces)} pieces", [box(*piece) for piece in pieces])
644
645 def rule(self, kind, *values):
646 kind = kind.lower()
647 *layers, value = values
648 if kind == "enclose" and layers[0] == layers[1]:
649 raise CommandError(f"same layer {layers[0]}")
650 self.rules[kind, tuple(layers)] = value
651 print(f"rule {kind} {' '.join(layers)} {value}")
652
653 def enclosure_violations(self, inner, outer, margin):
654 outers = self.on_layer(outer)
655 lines = []
656 for rect in self.on_layer(inner):
657 if not outers:
658 lines.append(f"{rect.name} best margin none")
659 continue
660 best = max(min(other.margins(rect)) for other in outers)
661 if best < margin:
662 lines.append(f"{rect.name} best margin {best}")
663 return lines
664
665 def width_violations(self, layer, minimum):
666 lines = []
667 for rect in self.on_layer(layer):
668 size = min(rect.width, rect.height)
669 if size < minimum:
670 lines.append(f"{rect.name} width {size}")
671 return lines
672
673 def check(self):
674 if not self.rules:
675 raise CommandError("no rules")
676 total = 0
677 for (kind, layers), value in sorted(self.rules.items()):
678 if kind == "enclose":
679 lines = self.enclosure_violations(*layers, value)
680 elif kind == "spacing":
681 lines = spacing_violations(self.on_layer(layers[0]), value)
682 else:
683 lines = self.width_violations(layers[0], value)
684 emit(f"{kind} {' '.join(layers)} {value}: {len(lines)} violations", lines)
685 total += len(lines)
686 print(f"check: {total} violations in {len(self.rules)} rules")
