18struct Rect {
19 string name, layer;
20 int x1, y1, x2, y2;
21 int width() const {
22 return x2 - x1;
23 }
24 int height() const {
25 return y2 - y1;
26 }
27 i64 area() const {
28 return i64(width()) * height();
29 }
30 string text() const {
31 return "(" + to_string(x1) + "," + to_string(y1) + ")-(" +
32 to_string(x2) + "," + to_string(y2) + ")";
33 }
34};

36string point(int x, int y) {
37 return "(" + to_string(x) + "," + to_string(y) + ")";
38}
39string joined(const vector<string>& values, const string& separator = " ") {
40 string result;
41 for (const auto& value : values) {
42 if (!result.empty()) result += separator;
43 result += value;
44 }
45 return result;

47string decimal(i64 numerator, i64 denominator) {
48 i64 hundredths = 100 * numerator / denominator;
49 return to_string(hundredths / 100) + "." +
50 (hundredths % 100 < 10 ? "0" : "") + to_string(hundredths % 100);
51}
52void require(bool condition, const string& message) {
53 if (!condition) throw runtime_error(message);
54}
55void bounded(const Rect& r) {
56 require(r.x1 >= -10000 && r.y1 >= -10000 && r.x2 <= 10000 && r.y2 <= 10000,
57 "out of bounds");
58}
59Rect common(const Rect& a, const Rect& b) {
60 return {"", "", max(a.x1, b.x1), max(a.y1, b.y1), min(a.x2, b.x2), min(a.y2, b.y2)};
61}
62int relation(const Rect& a, const Rect& b) {
63 Rect r = common(a, b);
64 if (r.width() < 0 || r.height() < 0) return 3;
65 if (r.width() == 0 && r.height() == 0) return 2;
66 if (r.width() == 0 || r.height() == 0) return 1;
67 return 0;
68}
69int gap(const Rect& a, const Rect& b) {
70 Rect r = common(a, b);
71 return max({0, -r.width(), -r.height()});
72}
73array<int, 4> margins(const Rect& outer, const Rect& inner) {
74 return {inner.x1 - outer.x1, inner.y1 - outer.y1,
75 outer.x2 - inner.x2, outer.y2 - inner.y2};
76}
77int min_margin(const Rect& outer, const Rect& inner) {
78 auto values = margins(outer, inner);
79 return *min_element(values.begin(), values.end());
80}

82struct Grid {
83 vector<int> xs, ys;
84 vector<vector<int>> counts;
85 explicit Grid(const vector<Rect>& rectangles) {
86 for (const Rect& r : rectangles) {
87 xs.push_back(r.x1);
88 xs.push_back(r.x2);
89 ys.push_back(r.y1);
90 ys.push_back(r.y2);
91 }
92 for (auto* axis : {&xs, &ys}) {
93 sort(axis->begin(), axis->end());
94 axis->erase(unique(axis->begin(), axis->end()), axis->end());
95 }
96 if (rectangles.empty()) return;
97 counts.assign(ys.size() - 1, vector<int>(xs.size() - 1));
98 for (const Rect& r : rectangles) {
99 int l = lower_bound(xs.begin(), xs.end(), r.x1) - xs.begin();
100 int e = lower_bound(xs.begin(), xs.end(), r.x2) - xs.begin();
101 int b = lower_bound(ys.begin(), ys.end(), r.y1) - ys.begin();
102 int t = lower_bound(ys.begin(), ys.end(), r.y2) - ys.begin();
103 for (int j = b; j < t; ++j)
104 for (int i = l; i < e; ++i) ++counts[j][i];
105 }
106 }
107 int rows() const {
108 return counts.size();
109 }
110 int columns() const {
111 return counts.empty() ? 0 : counts[0].size();
112 }
113 Rect bounds() const {
114 return {"", "", xs.front(), ys.front(), xs.back(), ys.back()};
115 }
116 i64 area(int threshold = 1) const {
117 i64 result = 0;
118 for (int j = 0; j < rows(); ++j)
119 for (int i = 0; i < columns(); ++i)
120 if (counts[j][i] >= threshold)
121 result += i64(xs[i + 1] - xs[i]) * (ys[j + 1] - ys[j]);
122 return result;
123 }
124 i64 clipped_area(const Rect& window) const {
125 i64 result = 0;
126 for (int j = 0; j < rows(); ++j)
127 for (int i = 0; i < columns(); ++i) if (counts[j][i]) {
128 int w = min(xs[i + 1], window.x2) - max(xs[i], window.x1);
129 int h = min(ys[j + 1], window.y2) - max(ys[j], window.y1);
130 if (w > 0 && h > 0) result += i64(w) * h;
131 }
132 return result;
133 }
134 i64 perimeter() const {
135 i64 result = 0;
136 for (int j = 0; j < rows(); ++j)
137 for (int i = 0; i < columns(); ++i) if (counts[j][i]) {
138 if (i == 0 || !counts[j][i - 1]) result += ys[j + 1] - ys[j];
139 if (i + 1 == columns() || !counts[j][i + 1]) result += ys[j + 1] - ys[j];
140 if (j == 0 || !counts[j - 1][i]) result += xs[i + 1] - xs[i];
141 if (j + 1 == rows() || !counts[j + 1][i]) result += xs[i + 1] - xs[i];
142 }
143 return result;
144 }
145 void empty_rectangle(const string& layer) const {
146 Rect best{};
147 i64 best_area = 0;
148 for (int l = 0; l < columns(); ++l) {
149 vector<bool> blocked(rows());
150 for (int e = l + 1; e <= columns(); ++e) {
151 for (int j = 0; j < rows(); ++j) blocked[j] = blocked[j] || counts[j][e - 1];
152 for (int j = 0; j < rows();) {
153 if (blocked[j]) {
154 ++j;
155 continue;
156 }
157 int b = j;
158 while (j < rows() && !blocked[j]) ++j;
159 Rect candidate{"", "", xs[l], ys[b], xs[e], ys[j]};
160 if (candidate.area() > best_area ||
161 (candidate.area() == best_area &&
162 make_tuple(candidate.x1, candidate.y1, candidate.width()) <
163 make_tuple(best.x1, best.y1, best.width()))) {
164 best = candidate;
165 best_area = best.area();
166 }
167 }
168 }
169 }
170 cout << "empty " << layer << ": ";
171 if (!best_area) cout << "none\n";
172 else cout << best.text() << " area " << best_area << '\n';
173 }
174 void holes(const string& layer) const {
175 struct Hole {
176 int anchor_x, anchor_y;
177 i64 area;
178 Rect box;
179 };
180 vector<Hole> holes;
181 vector<vector<bool>> seen(rows(), vector<bool>(columns()));
182 constexpr int dx[] = {1, -1, 0, 0}, dy[] = {0, 0, 1, -1};
183 for (int j = 0; j < rows(); ++j) for (int i = 0; i < columns(); ++i) {
184 if (counts[j][i] || seen[j][i]) continue;
185 Hole hole{xs[i], ys[j], 0, {"", "", xs[i], ys[j], xs[i + 1], ys[j + 1]}};
186 bool outside = false;
187 queue<pair<int, int>> pending;
188 pending.push({i, j});
189 seen[j][i] = true;
190 while (!pending.empty()) {
191 auto [x, y] = pending.front();
192 pending.pop();
193 outside = outside || x == 0 || y == 0 || x + 1 == columns() || y + 1 == rows();
194 hole.area += i64(xs[x + 1] - xs[x]) * (ys[y + 1] - ys[y]);
195 hole.box.x1 = min(hole.box.x1, xs[x]);
196 hole.box.y1 = min(hole.box.y1, ys[y]);
197 hole.box.x2 = max(hole.box.x2, xs[x + 1]);
198 hole.box.y2 = max(hole.box.y2, ys[y + 1]);
199 if (make_pair(ys[y], xs[x]) < make_pair(hole.anchor_y, hole.anchor_x)) {
200 hole.anchor_x = xs[x];
201 hole.anchor_y = ys[y];
202 }
203 for (int d = 0; d < 4; ++d) {
204 int nx = x + dx[d], ny = y + dy[d];
205 if (nx >= 0 && nx < columns() && ny >= 0 && ny < rows() &&
206 !counts[ny][nx] && !seen[ny][nx]) {
207 seen[ny][nx] = true;
208 pending.push({nx, ny});
209 }
210 }
211 }
212 if (!outside) holes.push_back(hole);
213 }
214 sort(holes.begin(), holes.end(), [](const Hole& a, const Hole& b) {
215 return tie(a.anchor_y, a.anchor_x) < tie(b.anchor_y, b.anchor_x);
216 });
217 cout << "holes " << layer << ": " << holes.size() << '\n';
218 for (size_t i = 0; i < holes.size(); ++i)
219 cout << "hole " << i + 1 << ": area " << holes[i].area << ", bbox " << holes[i].box.text() << '\n';
220 }
221 void merge(const string& layer) const {
222 map<pair<int, int>, int> active;
223 vector<Rect> pieces;
224 for (int j = 0; j <= rows(); ++j) {
225 set<pair<int, int>> runs;
226 if (j < rows()) for (int i = 0; i < columns();) {
227 if (!counts[j][i]) {
228 ++i;
229 continue;
230 }
231 int l = i;
232 while (i < columns() && counts[j][i]) ++i;
233 runs.insert({xs[l], xs[i]});
234 }
235 for (auto it = active.begin(); it != active.end();) {
236 if (!runs.count(it->first)) {
237 pieces.push_back({"", "", it->first.first, it->second, it->first.second, ys[j]});
238 it = active.erase(it);
239 } else ++it;
240 }
241 for (const auto& run : runs) if (!active.count(run)) active[run] = ys[j];
242 }
243 sort(pieces.begin(), pieces.end(), [](const Rect& a, const Rect& b) {
244 return tie(a.y1, a.x1) < tie(b.y1, b.x1);
245 });
246 cout << "merge " << layer << ": " << pieces.size() << " pieces\n";
247 for (const auto& r : pieces) cout << r.text() << '\n';
248 }
249};

251vector<vector<Rect>> nets(const vector<Rect>& rectangles) {
252 vector<vector<Rect>> result;
253 vector<bool> seen(rectangles.size());
254 for (size_t start = 0; start < rectangles.size(); ++start) if (!seen[start]) {
255 vector<Rect> component;
256 queue<size_t> pending;
257 pending.push(start);
258 seen[start] = true;
259 while (!pending.empty()) {
260 size_t u = pending.front();
261 pending.pop();
262 component.push_back(rectangles[u]);
263 for (size_t v = 0; v < rectangles.size(); ++v)
264 if (!seen[v] && relation(rectangles[u], rectangles[v]) <= 1) {
265 seen[v] = true;
266 pending.push(v);
267 }
268 }
269 sort(component.begin(), component.end(), [](const Rect& a, const Rect& b) {
270 return a.name < b.name;
271 });
272 result.push_back(component);
273 }
274 return result;
275}
276vector<string> spacing(const vector<Rect>& rectangles, int distance) {
277 vector<tuple<int, string, string, string>> violations;
278 for (size_t i = 0; i < rectangles.size(); ++i) for (size_t j = i + 1; j < rectangles.size(); ++j) {
279 const auto& a = rectangles[i];
280 const auto& b = rectangles[j];
281 int kind = relation(a, b), g = gap(a, b);
282 if (kind != 2 && !(kind == 3 && g < distance)) continue;
283 Rect r = common(a, b);
284 string direction = kind == 2 ? "corner" : r.width() < 0 && r.height() < 0 ?
285 "diagonal" : r.width() < 0 ? "x" : "y";
286 violations.emplace_back(g, a.name, b.name, direction);
287 }
288 sort(violations.begin(), violations.end());
289 vector<string> result;
290 for (const auto& [g, a, b, d] : violations) result.push_back(a + " " + b + " gap " + to_string(g) + " " + d);
291 return result;
292}

294class Layout {
295 map<string, Rect> rectangles;
296 vector<pair<map<string, Rect>, string>> history;
297 map<tuple<int, string, string>, int> rules;
298
299 Rect get(const string& name) const {
300 auto it = rectangles.find(name);
301 require(it != rectangles.end(), "unknown rectangle " + name);
302 return it->second;
303 }
304 vector<Rect> on_layer(const string& layer, bool needed = true) const {
305 vector<Rect> result;
306 for (const auto& [name, r] : rectangles) if (r.layer == layer) result.push_back(r);
307 require(!needed || !result.empty(), "no rectangles on " + layer);
308 return result;
309 }
310 void remember(const string& line) {
311 history.push_back({rectangles, line});
312 }
313 static string describe(const Rect& r) {
314 return r.name + " " + r.layer + " " + r.text() + " " + to_string(r.width()) + "x" + to_string(r.height());
315 }
316public:
317 void command(const string& line) {
318 istringstream input(line);
319 string op;
320 input >> op;
321 if (op == "ADD") {
322 Rect r;
323 input >> r.name >> r.layer >> r.x1 >> r.y1 >> r.x2 >> r.y2;
324 require(!rectangles.count(r.name), "duplicate rectangle " + r.name);
325 require(r.x1 < r.x2 && r.y1 < r.y2, "invalid rectangle");
326 bounded(r);
327 remember(line);
328 rectangles[r.name] = r;
329 cout << "added " << r.name << ' ' << r.layer << ' ' << r.text() << '\n';
330 } else if (op == "COPY" || op == "MOVE" || op == "RESIZE" || op == "SPLIT" || op == "LAYER" || op == "DELETE") {
331 string name;
332 input >> name;
333 Rect r = get(name);
334 if (op == "COPY") {
335 string target;
336 int dx, dy;
337 input >> target >> dx >> dy;
338 require(!rectangles.count(target), "duplicate rectangle " + target);
339 r.name = target;
340 r.x1 += dx;
341 r.x2 += dx;
342 r.y1 += dy;
343 r.y2 += dy;
344 bounded(r);
345 remember(line);
346 rectangles[target] = r;
347 cout << "copied " << name << " to " << target << ' ' << r.text() << '\n';
348 } else if (op == "MOVE" || op == "RESIZE") {
349 int a, b;
350 input >> a >> b;
351 if (op == "MOVE") {
352 r.x1 += a;
353 r.x2 += a;
354 r.y1 += b;
355 r.y2 += b;
356 }
357 else {
358 require(a > 0 && b > 0, "invalid size");
359 r.x2 = r.x1 + a;
360 r.y2 = r.y1 + b;
361 }
362 bounded(r);
363 remember(line);
364 rectangles[name] = r;
365 cout << (op == "MOVE" ? "moved " : "resized ") << name << " to " << r.text() << '\n';
366 } else if (op == "SPLIT") {
367 string axis, target;
368 int position;
369 input >> axis >> position >> target;
370 require(!rectangles.count(target), "duplicate rectangle " + target);
371 require(axis == "x" ? r.x1 < position && position < r.x2 : r.y1 < position && position < r.y2,
372 "cut outside " + name);
373 Rect other = r;
374 other.name = target;
375 if (axis == "x") {
376 r.x2 = position;
377 other.x1 = position;
378 }
379 else {
380 r.y2 = position;
381 other.y1 = position;
382 }
383 remember(line);
384 rectangles[name] = r;
385 rectangles[target] = other;
386 cout << "split " << name << " at " << axis << '=' << position << ": " << name << ' ' << r.text()
387 << ", " << target << ' ' << other.text() << '\n';
388 } else if (op == "LAYER") {
389 string layer;
390 input >> layer;
391 require(r.layer != layer, "already on " + layer);
392 string old = r.layer;
393 r.layer = layer;
394 remember(line);
395 rectangles[name] = r;
396 cout << "layer " << name << ' ' << old << " -> " << layer << '\n';
397 } else {
398 remember(line);
399 rectangles.erase(name);
400 cout << "deleted " << name << '\n';
401 }
402 } else if (op == "UNDO") {
403 require(!history.empty(), "nothing to undo");
404 auto previous = move(history.back());
405 history.pop_back();
406 rectangles = move(previous.first);
407 cout << "undo: " << previous.second << '\n';
408 } else if (op == "LIST") {
409 cout << "rectangles: " << rectangles.size() << '\n';
410 for (const auto& [name, r] : rectangles) cout << describe(r) << '\n';
411 } else if (op == "INFO") {
412 string name;
413 input >> name;
414 Rect a = get(name);
415 cout << describe(a) << " area " << a.area() << '\n';
416 array<vector<string>, 5> lists;
417 for (const auto& [other, b] : rectangles) if (other != name) {
418 int kind = relation(a, b);
419 if (a.layer == b.layer && kind < 3) lists[kind].push_back(other);
420 if (min_margin(a, b) >= 0) lists[3].push_back(other);
421 if (min_margin(b, a) >= 0) lists[4].push_back(other);
422 }
423 const string labels[] = {"overlaps", "touches", "corners", "encloses", "enclosed by"};
424 for (int i = 0; i < 5; ++i) cout << labels[i] << ": " << (lists[i].empty() ? "none" : joined(lists[i])) << '\n';
425 } else if (op == "LAYERS") {
426 set<string> layers;
427 for (const auto& [name, r] : rectangles) layers.insert(r.layer);
428 cout << "layers: " << layers.size() << '\n';
429 for (const auto& layer : layers) {
430 auto values = on_layer(layer);
431 cout << layer << ": " << values.size() << " rectangles, area " << Grid(values).area()
432 << ", nets " << nets(values).size() << '\n';
433 }
434 } else if (op == "OVERLAPS") {
435 string layer;
436 bool filtered = bool(input >> layer);
437 if (filtered) on_layer(layer);
438 vector<tuple<string, string, string, Rect>> pairs;
439 for (auto a = rectangles.begin(); a != rectangles.end(); ++a)
440 for (auto b = next(a); b != rectangles.end(); ++b)
441 if (a->second.layer == b->second.layer && (!filtered || a->second.layer == layer) &&
442 relation(a->second, b->second) == 0)
443 pairs.emplace_back(a->second.layer, a->first, b->first, common(a->second, b->second));
444 sort(pairs.begin(), pairs.end(), [](const auto& a, const auto& b) {
445 return tie(std::get<0>(a), std::get<1>(a), std::get<2>(a)) < tie(std::get<0>(b), std::get<1>(b), std::get<2>(b));
446 });
447 cout << "overlaps " << (filtered ? layer : "all layers") << ": " << pairs.size() << '\n';
448 for (const auto& [l, a, b, r] : pairs) {
449 if (!filtered) cout << l << ' ';
450 cout << a << ' ' << b << ' ' << r.text() << " area " << r.area() << '\n';
451 }
452 } else if (op == "INTERSECT" || op == "ADJACENT" || op == "ENCLOSE" || op == "PATH") {
453 string first, second;
454 input >> first >> second;
455 Rect a = get(first), b = get(second);
456 require(first != second, "same rectangle " + first);
457 if (op == "PATH") require(a.layer == b.layer, "different layers");
458 Rect r = common(a, b);
459 int kind = relation(a, b);
460 string heading = op == "INTERSECT" ? "intersect" : op == "ADJACENT" ? "adjacent" : op == "ENCLOSE" ? "enclose" : "path";
461 cout << heading << ' ' << first << ' ' << second << ": ";
462 if (op == "INTERSECT") {
463 if (kind == 0) cout << r.text() << " area " << r.area();
464 else if (kind == 1) cout << "segment " << r.text() << " length " << r.width() + r.height();
465 else if (kind == 2) cout << "point " << point(r.x1, r.y1);
466 else cout << "none, gap " << gap(a, b);
467 } else if (op == "ADJACENT") {
468 if (kind == 0) cout << "overlapping";
469 else if (kind == 1) {
470 string direction = r.width() == 0 ? (a.x2 == r.x1 ? "west" : "east") : (a.y2 == r.y1 ? "south" : "north");
471 cout << first << " is " << direction << " of " << second << ", edge " << r.text() << ", length " << r.width() + r.height();
472 } else if (kind == 2) cout << "corner only at " << point(r.x1, r.y1);
473 else cout << "not adjacent";
474 } else if (op == "ENCLOSE") {
475 auto m = margins(a, b);
476 int least = *min_element(m.begin(), m.end());
477 cout << (least >= 0 ? "yes" : "no") << ", margins left " << m[0] << " bottom " << m[1] << " right " << m[2] << " top " << m[3];
478 if (least >= 0) cout << ", min " << least;
479 } else {
480 map<string, string> previous{{first, ""}};
481 queue<string> pending;
482 pending.push(first);
483 auto values = on_layer(a.layer);
484 while (!pending.empty() && !previous.count(second)) {
485 string current = pending.front();
486 pending.pop();
487 for (const Rect& neighbor : values)
488 if (!previous.count(neighbor.name) && relation(get(current), neighbor) <= 1) {
489 previous[neighbor.name] = current;
490 pending.push(neighbor.name);
491 }
492 }
493 if (!previous.count(second)) cout << "none";
494 else {
495 vector<string> path;
496 for (string p = second; !p.empty(); p = previous[p]) path.push_back(p);
497 reverse(path.begin(), path.end());
498 cout << joined(path, " -> ") << " (" << path.size() - 1 << " steps)";
499 }
500 }
501 cout << '\n';
502 } else if (op == "AT") {
503 int x, y;
504 input >> x >> y;
505 vector<string> found;
506 for (const auto& [name, r] : rectangles) if (r.x1 <= x && x <= r.x2 && r.y1 <= y && y <= r.y2) {
507 int borders = int(x == r.x1 || x == r.x2) + int(y == r.y1 || y == r.y2);
508 found.push_back(name + " " + r.layer + " " + (borders == 2 ? "corner" : borders == 1 ? "edge" : "inside"));
509 }
510 cout << "at " << point(x, y) << ": " << (found.empty() ? "none" : joined(found, ", ")) << '\n';
511 } else if (op == "NEAREST") {
512 string name;
513 input >> name;
514 Rect a = get(name);
515 string best;
516 int distance = 0;
517 for (const auto& [other, b] : rectangles) if (other != name && b.layer == a.layer) {
518 int d = gap(a, b);
519 if (best.empty() || d < distance) {
520 best = other;
521 distance = d;
522 }
523 }
524 cout << "nearest " << name << ": " << (best.empty() ? "none" : best + " gap " + to_string(distance)) << '\n';
525 } else if (op == "RULE") {
526 string kind, inner, outer;
527 int value;
528 input >> kind >> inner;
529 if (kind == "ENCLOSE") {
530 input >> outer >> value;
531 require(inner != outer, "same layer " + inner);
532 rules[{0, inner, outer}] = value;
533 cout << "rule enclose " << inner << ' ' << outer << ' ' << value << '\n';
534 } else {
535 input >> value;
536 rules[{kind == "SPACING" ? 1 : 2, inner, ""}] = value;
537 cout << "rule " << (kind == "SPACING" ? "spacing" : "width") << ' ' << inner << ' ' << value << '\n';
538 }
539 } else if (op == "CHECK") {
540 require(!rules.empty(), "no rules");
541 size_t total = 0;
542 for (const auto& [key, value] : rules) {
543 const auto& [kind, inner, outer] = key;
544 vector<string> violations;
545 auto values = on_layer(inner, false);
546 string heading;
547 if (kind == 0) {
548 auto containers = on_layer(outer, false);
549 for (const auto& r : values) {
550 int best = -100000;
551 for (const auto& container : containers) best = max(best, min_margin(container, r));
552 if (containers.empty() || best < value)
553 violations.push_back(r.name + " best margin " + (containers.empty() ? "none" : to_string(best)));
554 }
555 heading = "enclose " + inner + " " + outer;
556 } else if (kind == 1) {
557 violations = spacing(values, value);
558 heading = "spacing " + inner;
559 } else {
560 for (const auto& r : values) if (min(r.width(), r.height()) < value)
561 violations.push_back(r.name + " width " + to_string(min(r.width(), r.height())));
562 heading = "width " + inner;
563 }
564 cout << heading << ' ' << value << ": " << violations.size() << " violations\n";
565 for (const auto& violation : violations) cout << violation << '\n';
566 total += violations.size();
567 }
568 cout << "check: " << total << " violations in " << rules.size() << " rules\n";
569 } else {
570 int threshold = 1;
571 string layer;
572 if (op == "COVER") input >> threshold;
573 input >> layer;
574 auto values = on_layer(layer);
575 Grid grid(values);
576 if (op == "AREA" || op == "COVER" || op == "PERIMETER") {
577 if (op == "AREA") cout << "area " << layer << ": " << grid.area() << '\n';
578 else if (op == "COVER") cout << "cover " << layer << " >= " << threshold << ": " << grid.area(threshold) << '\n';
579 else cout << "perimeter " << layer << ": " << grid.perimeter() << '\n';
580 } else if (op == "NETS") {
581 auto components = nets(values);
582 cout << "nets " << layer << ": " << components.size() << '\n';
583 for (size_t i = 0; i < components.size(); ++i) {
584 vector<string> names;
585 for (const auto& r : components[i]) names.push_back(r.name);
586 cout << "net " << i + 1 << ": " << joined(names) << " area " << Grid(components[i]).area() << '\n';
587 }
588 } else if (op == "DRC") {
589 int distance;
590 input >> distance;
591 auto violations = spacing(values, distance);
592 cout << "drc " << layer << " spacing " << distance << ": " << violations.size() << " violations\n";
593 for (const auto& violation : violations) cout << violation << '\n';
594 } else if (op == "BBOX") {
595 Rect b = grid.bounds();
596 cout << "bbox " << layer << ": " << b.text() << ' ' << b.width() << 'x' << b.height()
597 << ", " << values.size() << " rectangles, density " << decimal(100 * grid.area(), b.area()) << "%\n";
598 } else if (op == "DENSITY") {
599 int w;
600 input >> w;
601 Rect b = grid.bounds();
602 int c = (b.width() + w - 1) / w, r = (b.height() + w - 1) / w;
603 require(c <= 20 && r <= 20, "too many windows");
604 vector<vector<i64>> areas(r, vector<i64>(c));
605 pair<int, int> least{0, 0}, most{0, 0};
606 for (int j = 0; j < r; ++j) for (int i = 0; i < c; ++i) {
607 Rect tile{"", "", b.x1 + i * w, b.y1 + j * w, b.x1 + (i + 1) * w, b.y1 + (j + 1) * w};
608 areas[j][i] = grid.clipped_area(tile);
609 if (areas[j][i] < areas[least.second][least.first]) least = {i, j};
610 if (areas[j][i] > areas[most.second][most.first]) most = {i, j};
611 }
612 cout << "density " << layer << " window " << w << ": " << c << 'x' << r << '\n';
613 for (int j = r - 1; j >= 0; --j) {
614 for (int i = 0; i < c; ++i) cout << (i ? " " : "") << 100 * areas[j][i] / (i64(w) * w);
615 cout << '\n';
616 }
617 for (const auto& [label, location] : vector<pair<string, pair<int, int>>>{{"min", least}, {"max", most}}) {
618 auto [i, j] = location;
619 cout << label << ' ' << decimal(100 * areas[j][i], i64(w) * w) << "% at " << point(b.x1 + i * w, b.y1 + j * w) << '\n';
620 }
621 } else if (op == "RENDER") {
622 Rect window;
623 input >> window.x1 >> window.y1 >> window.x2 >> window.y2;
624 require(window.x1 < window.x2 && window.y1 < window.y2, "invalid window");
625 require(window.width() <= 60 && window.height() <= 40, "window too large");
626 cout << "render " << layer << ' ' << window.text() << '\n';
627 for (int y = window.y2 - 1; y >= window.y1; --y) {
628 for (int x = window.x1; x < window.x2; ++x) {
629 int count = 0;
630 for (const auto& r : values) count += r.x1 <= x && x < r.x2 && r.y1 <= y && y < r.y2;
631 cout << (count == 0 ? '.' : count < 10 ? char('0' + count) : '*');
632 }
633 cout << '\n';
634 }
635 } else if (op == "EMPTY") grid.empty_rectangle(layer);
636 else if (op == "HOLES") grid.holes(layer);
637 else if (op == "MERGE") grid.merge(layer);
638 }
639 }
640};
