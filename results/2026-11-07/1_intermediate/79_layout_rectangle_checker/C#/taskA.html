7record Box(int X1, int Y1, int X2, int Y2)
8{
9 public int Width => X2 - X1;
10 public int Height => Y2 - Y1;
11 public long Area => (long)Width * Height;
12
13 public override string ToString() => $"({X1},{Y1})-({X2},{Y2})";
14
15 public Box Shift(int dx, int dy) => new(X1 + dx, Y1 + dy, X2 + dx, Y2 + dy);
16
17 public bool Contains(Box other) => X1 <= other.X1 && Y1 <= other.Y1 && X2 >= other.X2 && Y2 >= other.Y2;
18}

25class Relation
26{
27 public Box Common;
28 public int Ox, Oy, Gap, Kind;
29 public string Direction;
30
31 public Relation(Rectangle a, Rectangle b)
32 {
33 Box x = a.Box, y = b.Box;
34 Common = new(Math.Max(x.X1, y.X1), Math.Max(x.Y1, y.Y1), Math.Min(x.X2, y.X2), Math.Min(x.Y2, y.Y2));
35 Ox = Common.Width;
36 Oy = Common.Height;
37 Gap = Math.Max(0, Math.Max(-Ox, -Oy));
38 Kind = Ox < 0 || Oy < 0 ? 3 : Ox > 0 && Oy > 0 ? 0 : Ox == 0 && Oy == 0 ? 2 : 1;
39 Direction = Kind == 2 ? "corner" : Ox < 0 && Oy < 0 ? "diagonal" : Ox < 0 ? "x" : "y";
40 }
41 public bool Connected => Kind <= 1;
42}
43class Grid
44{
45 public int[] Xs, Ys;
46 readonly int[,] counts;
47 int Rows => Ys.Length - 1;
48 int Columns => Xs.Length - 1;
49
50 public Grid(List<Rectangle> rectangles)
51 {
52 Xs = rectangles.SelectMany(r => new[] { r.Box.X1, r.Box.X2 }).Distinct().Order().ToArray();
53 Ys = rectangles.SelectMany(r => new[] { r.Box.Y1, r.Box.Y2 }).Distinct().Order().ToArray();
54 counts = new int[Rows, Columns];
55 foreach (Rectangle r in rectangles) for (int j = 0; j < Rows; j++) if (Ys[j] >= r.Box.Y1 && Ys[j + 1] <= r.Box.Y2) for (int i = 0; i < Columns; i++) if (Xs[i] >= r.Box.X1 && Xs[i + 1] <= r.Box.X2) counts[j, i]++;
56 }
57 public Box Bounds => new(Xs[0], Ys[0], Xs[^1], Ys[^1]);
58
59 public long Area(int threshold)
60 {
61 long result = 0;
62 for (int j = 0; j < Rows; j++) for (int i = 0; i < Columns; i++) if (counts[j, i] >= threshold) result += (long)(Xs[i + 1] - Xs[i]) * (Ys[j + 1] - Ys[j]);
63 return result;
64 }
65
66 public long Clipped(Box box)
67 {
68 long result = 0;
69 for (int j = 0; j < Rows; j++) for (int i = 0; i < Columns; i++) if (counts[j, i] > 0) result += (long)Math.Max(0, Math.Min(Xs[i + 1], box.X2) - Math.Max(Xs[i], box.X1)) * Math.Max(0, Math.Min(Ys[j + 1], box.Y2) - Math.Max(Ys[j], box.Y1));
70 return result;
71 }
72
73 public long Perimeter()
74 {
75 long result = 0;
76 for (int j = 0; j < Rows; j++) for (int i = 0; i < Columns; i++) if (counts[j, i] > 0)
77 {
78 if (i == 0 || counts[j, i - 1] == 0) result += Ys[j + 1] - Ys[j];
79 if (i + 1 == Columns || counts[j, i + 1] == 0) result += Ys[j + 1] - Ys[j];
80 if (j == 0 || counts[j - 1, i] == 0) result += Xs[i + 1] - Xs[i];
81 if (j + 1 == Rows || counts[j + 1, i] == 0) result += Xs[i + 1] - Xs[i];
82 }
83 return result;
84 }
85
86 public Box Empty()
87 {
88 Box best = null;
89 for (int left = 0; left < Columns; left++)
90 {
91 bool[] blocked = new bool[Rows];
92 for (int right = left + 1; right <= Columns; right++)
93 {
94 for (int j = 0; j < Rows; j++) blocked[j] |= counts[j, right - 1] > 0;
95 int bottom = 0;
96 while (bottom < Rows)
97 {
98 if (blocked[bottom])
99 {
100 bottom++;
101 continue;
102 }
103 int top = bottom + 1;
104 while (top < Rows && !blocked[top]) top++;
105 Box candidate = new(Xs[left], Ys[bottom], Xs[right], Ys[top]);
106 if (best == null || candidate.Area > best.Area || candidate.Area == best.Area && (candidate.X1 < best.X1 || candidate.X1 == best.X1 && (candidate.Y1 < best.Y1 || candidate.Y1 == best.Y1 && candidate.Width < best.Width))) best = candidate;
107 bottom = top;
108 }
109 }
110 }
111 return best;
112 }
113
114 public List<Hole> Holes()
115 {
116 bool[,] seen = new bool[Rows, Columns];
117 List<Hole> result = new();
118 int[] dx = { -1, 1, 0, 0 }, dy = { 0, 0, -1, 1 };
119 for (int j = 0; j < Rows; j++) for (int i = 0; i < Columns; i++) if (counts[j, i] == 0 && !seen[j, i])
120 {
121 Queue<(int X, int Y)> queue = new();
122 queue.Enqueue((i, j));
123 seen[j, i] = true;
124 bool boundary = false;
125 long area = 0;
126 int x1 = Xs[i], y1 = Ys[j], x2 = Xs[i + 1], y2 = Ys[j + 1], ax = Xs[i], ay = Ys[j];
127 while (queue.Count > 0)
128 {
129 var (x, y) = queue.Dequeue();
130 boundary |= x == 0 || y == 0 || x + 1 == Columns || y + 1 == Rows;
131 area += (long)(Xs[x + 1] - Xs[x]) * (Ys[y + 1] - Ys[y]);
132 x1 = Math.Min(x1, Xs[x]);
133 y1 = Math.Min(y1, Ys[y]);
134 x2 = Math.Max(x2, Xs[x + 1]);
135 y2 = Math.Max(y2, Ys[y + 1]);
136 if (Ys[y] < ay || Ys[y] == ay && Xs[x] < ax)
137 {
138 ax = Xs[x];
139 ay = Ys[y];
140 }
141 for (int d = 0; d < 4; d++)
142 {
143 int xx = x + dx[d], yy = y + dy[d];
144 if (xx >= 0 && xx < Columns && yy >= 0 && yy < Rows && counts[yy, xx] == 0 && !seen[yy, xx])
145 {
146 seen[yy, xx] = true;
147 queue.Enqueue((xx, yy));
148 }
149 }
150 }
151 if (!boundary) result.Add(new(area, new(x1, y1, x2, y2), ax, ay));
152 }
153 return result.OrderBy(h => h.AnchorY).ThenBy(h => h.AnchorX).ToList();
154 }
155
156 public List<Box> Pieces()
157 {
158 Dictionary<Interval, Box> active = new();
159 List<Box> result = new();
160 for (int j = 0; j < Rows; j++)
161 {
162 Dictionary<Interval, Box> next = new();
163 int i = 0;
164 while (i < Columns)
165 {
166 if (counts[j, i] == 0)
167 {
168 i++;
169 continue;
170 }
171 int right = i + 1;
172 while (right < Columns && counts[j, right] > 0) right++;
173 Interval key = new(Xs[i], Xs[right]);
174 active.Remove(key, out Box old);
175 next[key] = new(Xs[i], old == null ? Ys[j] : old.Y1, Xs[right], Ys[j + 1]);
176 i = right;
177 }
178 result.AddRange(active.Values);
179 active = next;
180 }
181 result.AddRange(active.Values);
182 return result.OrderBy(b => b.Y1).ThenBy(b => b.X1).ToList();
183 }

221 static List<List<Rectangle>> Nets(List<Rectangle> list)
222 {
223 bool[] seen = new bool[list.Count];
224 List<List<Rectangle>> result = new();
225 for (int start = 0; start < list.Count; start++) if (!seen[start])
226 {
227 List<Rectangle> net = new();
228 Queue<int> queue = new();
229 queue.Enqueue(start);
230 seen[start] = true;
231 while (queue.Count > 0)
232 {
233 int i = queue.Dequeue();
234 net.Add(list[i]);
235 for (int j = 0; j < list.Count; j++) if (!seen[j] && new Relation(list[i], list[j]).Connected)
236 {
237 seen[j] = true;
238 queue.Enqueue(j);
239 }
240 }
241 net.Sort((a, b) => StringComparer.Ordinal.Compare(a.Name, b.Name));
242 result.Add(net);
243 }
244 return result;
245 }

247 static List<Violation> Spacing(List<Rectangle> list, int space)
248 {
249 List<Violation> result = new();
250 for (int i = 0; i < list.Count; i++) for (int j = i + 1; j < list.Count; j++)
251 {
252 Relation relation = new(list[i], list[j]);
253 if (relation.Kind == 2 || relation.Kind == 3 && relation.Gap < space) result.Add(new(list[i], list[j], relation.Gap, relation.Direction));
254 }
255 return result.OrderBy(v => v.Gap).ThenBy(v => v.A.Name, StringComparer.Ordinal).ThenBy(v => v.B.Name, StringComparer.Ordinal).ToList();
256 }

262 List<string> Command(string line)
263 {
264 string[] a = line.Split(' ');
265 string c = a[0];
266 List<string> output = new();
267 switch (c)
268 {
269 case "ADD":
270 {
271 if (rectangles.ContainsKey(a[1])) Fail("duplicate rectangle " + a[1]);
272 Box box = new(int.Parse(a[3]), int.Parse(a[4]), int.Parse(a[5]), int.Parse(a[6]));
273 if (box.X1 >= box.X2 || box.Y1 >= box.Y2) Fail("invalid rectangle");
274 Bounds(box);
275 Save(line);
276 rectangles[a[1]] = new(a[1], a[2], box);
277 output.Add($"added {a[1]} {a[2]} {box}");
278 break;
279 }
280 case "COPY":
281 case "MOVE":
282 case "RESIZE":
283 case "SPLIT":
284 case "LAYER":
285 case "DELETE":
286 {
287 Rectangle rectangle = Get(a[1]);
288 Box box = rectangle.Box;
289 string layer = rectangle.Layer;
290 Rectangle extra = null;
291 switch (c)
292 {
293 case "COPY":
294 if (rectangles.ContainsKey(a[2])) Fail("duplicate rectangle " + a[2]);
295 box = box.Shift(int.Parse(a[3]), int.Parse(a[4]));
296 Bounds(box);
297 extra = new(a[2], layer, box);
298 output.Add($"copied {rectangle.Name} to {a[2]} {box}");
299 break;
300 case "MOVE":
301 box = box.Shift(int.Parse(a[2]), int.Parse(a[3]));
302 Bounds(box);
303 output.Add($"moved {rectangle.Name} to {box}");
304 break;
305 case "RESIZE":
306 {
307 int w = int.Parse(a[2]), h = int.Parse(a[3]);
308 if (w <= 0 || h <= 0) Fail("invalid size");
309 box = new(box.X1, box.Y1, box.X1 + w, box.Y1 + h);
310 Bounds(box);
311 output.Add($"resized {rectangle.Name} to {box}");
312 break;
313 }
314 case "SPLIT":
315 {
316 if (rectangles.ContainsKey(a[4])) Fail("duplicate rectangle " + a[4]);
317 int pos = int.Parse(a[3]);
318 bool x = a[2] == "x";
319 if (pos <= (x ? box.X1 : box.Y1) || pos >= (x ? box.X2 : box.Y2)) Fail("cut outside " + rectangle.Name);
320 Box second = x ? new(pos, box.Y1, box.X2, box.Y2) : new(box.X1, pos, box.X2, box.Y2);
321 box = x ? new(box.X1, box.Y1, pos, box.Y2) : new(box.X1, box.Y1, box.X2, pos);
322 extra = new(a[4], layer, second);
323 output.Add($"split {rectangle.Name} at {a[2]}={pos}: {rectangle.Name} {box}, {a[4]} {second}");
324 break;
325 }
326 case "LAYER":
327 if (layer == a[2]) Fail("already on " + layer);
328 output.Add($"layer {rectangle.Name} {layer} -> {a[2]}");
329 layer = a[2];
330 break;
331 case "DELETE":
332 output.Add("deleted " + rectangle.Name);
333 break;
334 }
335 Save(line);
336 if (c == "DELETE") rectangles.Remove(rectangle.Name);
337 else if (c != "COPY") rectangles[rectangle.Name] = new(rectangle.Name, layer, box);
338 if (extra != null) rectangles[extra.Name] = extra;
339 break;
340 }
341 case "UNDO":
342 if (history.Count == 0) Fail("nothing to undo");
343 Saved saved = history[^1];
344 history.RemoveAt(history.Count - 1);
345 rectangles = saved.Rectangles;
346 output.Add("undo: " + saved.Command);
347 break;
348 case "LIST":
349 output.Add("rectangles: " + rectangles.Count);
350 foreach (Rectangle rectangle in rectangles.Values) output.Add(Info(rectangle));
351 break;
352 case "INFO":
353 {
354 Rectangle rectangle = Get(a[1]);
355 output.Add(Info(rectangle) + " area " + rectangle.Box.Area);
356 string[] labels = { "overlaps", "touches", "corners", "encloses", "enclosed by" };
357 for (int k = 0; k < labels.Length; k++)
358 {
359 List<Rectangle> list = new();
360 foreach (Rectangle other in rectangles.Values) if (rectangle.Name != other.Name)
361 {
362 Relation relation = new(rectangle, other);
363 if (k < 3 ? rectangle.Layer == other.Layer && relation.Kind == k : k == 3 ? rectangle.Box.Contains(other.Box) : other.Box.Contains(rectangle.Box)) list.Add(other);
364 }
365 output.Add(labels[k] + ": " + Names(list));
366 }
367 break;
368 }
369 case "LAYERS":
370 {
371 List<string> layers = rectangles.Values.Select(r => r.Layer).Distinct().Order(StringComparer.Ordinal).ToList();
372 output.Add("layers: " + layers.Count);
373 foreach (string name in layers)
374 {
375 List<Rectangle> list = Layer(name, true);
376 output.Add($"{name}: {list.Count} rectangles, area {new Grid(list).Area(1)}, nets {Nets(list).Count}");
377 }
378 break;
379 }
380 case "OVERLAPS":
381 {
382 List<string> details = new();
383 List<string> layers;
384 if (a.Length == 2)
385 {
386 Layer(a[1], true);
387 layers = new() { a[1] };
388 }
389 else layers = rectangles.Values.Select(r => r.Layer).Distinct().Order(StringComparer.Ordinal).ToList();
390 foreach (string name in layers)
391 {
392 List<Rectangle> list = Layer(name, true);
393 for (int i = 0; i < list.Count; i++) for (int j = i + 1; j < list.Count; j++)
394 {
395 Rectangle x = list[i], y = list[j];
396 Relation relation = new(x, y);
397 if (relation.Kind == 0) details.Add((a.Length == 1 ? name + " " : "") + $"{x.Name} {y.Name} {relation.Common} area {relation.Common.Area}");
398 }
399 }
400 output.Add($"overlaps {(a.Length == 1 ? "all layers" : a[1])}: {details.Count}");
401 output.AddRange(details);
402 break;
403 }
404 case "INTERSECT":
405 case "ADJACENT":
406 case "ENCLOSE":
407 case "PATH":
408 {
409 Rectangle x = Get(a[1]), y = Get(a[2]);
410 if (x.Name == y.Name) Fail("same rectangle " + x.Name);
411 Relation relation = new(x, y);
412 string prefix = $"{c.ToLowerInvariant()} {x.Name} {y.Name}: ";
413 if (c == "PATH")
414 {
415 if (x.Layer != y.Layer) Fail("different layers");
416 List<Rectangle> list = Layer(x.Layer, true);
417 int start = list.IndexOf(x), goal = list.IndexOf(y);
418 int[] previous = Enumerable.Repeat(-1, list.Count).ToArray();
419 previous[start] = start;
420 Queue<int> queue = new();
421 queue.Enqueue(start);
422 while (queue.Count > 0)
423 {
424 int i = queue.Dequeue();
425 for (int j = 0; j < list.Count; j++) if (previous[j] < 0 && new Relation(list[i], list[j]).Connected)
426 {
427 previous[j] = i;
428 queue.Enqueue(j);
429 }
430 }
431 List<string> path = new();
432 if (previous[goal] >= 0) for (int at = goal;; at = previous[at])
433 {
434 path.Insert(0, list[at].Name);
435 if (at == start) break;
436 }
437 output.Add(prefix + (path.Count == 0 ? "none" : string.Join(" -> ", path) + $" ({path.Count - 1} steps)"));
438 }
439 else if (c == "ENCLOSE")
440 {
441 int left = y.Box.X1 - x.Box.X1, bottom = y.Box.Y1 - x.Box.Y1, right = x.Box.X2 - y.Box.X2, top = x.Box.Y2 - y.Box.Y2, min = Margin(x, y);
442 output.Add(prefix + (min >= 0 ? "yes" : "no") + $", margins left {left} bottom {bottom} right {right} top {top}" + (min >= 0 ? $", min {min}" : ""));
443 }
444 else if (c == "INTERSECT") output.Add(prefix + (relation.Kind switch
445 {
446 0 => $"{relation.Common} area {relation.Common.Area}",
447 1 => $"segment {relation.Common} length {relation.Ox + relation.Oy}",
448 2 => $"point ({relation.Common.X1},{relation.Common.Y1})",
449 _ => $"none, gap {relation.Gap}"
450 }));
451 else output.Add(prefix + (relation.Kind switch
452 {
453 0 => "overlapping",
454 1 => $"{x.Name} is {(relation.Ox == 0 ? x.Box.X2 == relation.Common.X1 ? "west" : "east" : x.Box.Y2 == relation.Common.Y1 ? "south" : "north")} of {y.Name}, edge {relation.Common}, length {relation.Ox + relation.Oy}",
455 2 => $"corner only at ({relation.Common.X1},{relation.Common.Y1})",
456 _ => "not adjacent"
457 }));
458 break;
459 }
460 case "AT":
461 {
462 int x = int.Parse(a[1]), y = int.Parse(a[2]);
463 List<string> found = new();
464 foreach (Rectangle rectangle in rectangles.Values) if (x >= rectangle.Box.X1 && x <= rectangle.Box.X2 && y >= rectangle.Box.Y1 && y <= rectangle.Box.Y2)
465 {
466 bool xx = x == rectangle.Box.X1 || x == rectangle.Box.X2, yy = y == rectangle.Box.Y1 || y == rectangle.Box.Y2;
467 found.Add($"{rectangle.Name} {rectangle.Layer} " + (xx && yy ? "corner" : xx || yy ? "edge" : "inside"));
468 }
469 output.Add($"at ({x},{y}): " + (found.Count == 0 ? "none" : string.Join(", ", found)));
470 break;
471 }
472 case "NEAREST":
473 {
474 Rectangle rectangle = Get(a[1]), best = null;
475 int gap = int.MaxValue;
476 foreach (Rectangle other in Layer(rectangle.Layer, true)) if (rectangle.Name != other.Name)
477 {
478 int g = new Relation(rectangle, other).Gap;
479 if (g < gap)
480 {
481 gap = g;
482 best = other;
483 }
484 }
485 output.Add($"nearest {rectangle.Name}: " + (best == null ? "none" : $"{best.Name} gap {gap}"));
486 break;
487 }
488 case "RULE":
489 {
490 string kind = a[1].ToLowerInvariant(), inner = a[2], outer = kind == "enclose" ? a[3] : "";
491 int value = int.Parse(a[kind == "enclose" ? 4 : 3]);
492 if (kind == "enclose" && inner == outer) Fail("same layer " + inner);
493 rules[kind + " " + inner + " " + outer] = new(kind, inner, outer, value);
494 output.Add($"rule {kind} {inner}" + (outer.Length == 0 ? "" : " " + outer) + $" {value}");
495 break;
496 }
497 case "CHECK":
498 {
499 if (rules.Count == 0) Fail("no rules");
500 int total = 0;
501 foreach (Rule rule in rules.Values)
502 {
503 List<string> violations = new();
504 List<Rectangle> list = Layer(rule.Inner, false);
505 if (rule.Kind == "spacing") foreach (Violation violation in Spacing(list, rule.Value)) violations.Add(ViolationText(violation));
506 else if (rule.Kind == "width") foreach (Rectangle rectangle in list)
507 {
508 int width = Math.Min(rectangle.Box.Width, rectangle.Box.Height);
509 if (width < rule.Value) violations.Add($"{rectangle.Name} width {width}");
510 }
511 else
512 {
513 List<Rectangle> outer = Layer(rule.Outer, false);
514 foreach (Rectangle rectangle in list)
515 {
516 int? best = null;
517 foreach (Rectangle o in outer) best = !best.HasValue ? Margin(o, rectangle) : Math.Max(best.Value, Margin(o, rectangle));
518 if (!best.HasValue || best.Value < rule.Value) violations.Add($"{rectangle.Name} best margin " + (best.HasValue ? best.Value.ToString() : "none"));
519 }
520 }
521 output.Add($"{rule.Kind} {rule.Inner}" + (rule.Outer.Length == 0 ? "" : " " + rule.Outer) + $" {rule.Value}: {violations.Count} violations");
522 output.AddRange(violations);
523 total += violations.Count;
524 }
525 output.Add($"check: {total} violations in {rules.Count} rules");
526 break;
527 }
528 default:
529 {
530 string name = c == "COVER" ? a[2] : a[1];
531 List<Rectangle> list = Layer(name, true);
532 Grid grid = new(list);
533 Box box = grid.Bounds;
534 switch (c)
535 {
536 case "AREA":
537 output.Add($"area {name}: {grid.Area(1)}");
538 break;
539 case "COVER":
540 output.Add($"cover {name} >= {a[1]}: {grid.Area(int.Parse(a[1]))}");
541 break;
542 case "PERIMETER":
543 output.Add($"perimeter {name}: {grid.Perimeter()}");
544 break;
545 case "NETS":
546 {
547 List<List<Rectangle>> groups = Nets(list);
548 output.Add($"nets {name}: {groups.Count}");
549 for (int i = 0; i < groups.Count; i++) output.Add($"net {i + 1}: {Names(groups[i])} area {new Grid(groups[i]).Area(1)}");
550 break;
551 }
552 case "DRC":
553 {
554 int s = int.Parse(a[2]);
555 List<Violation> violations = Spacing(list, s);
556 output.Add($"drc {name} spacing {s}: {violations.Count} violations");
557 foreach (Violation violation in violations) output.Add(ViolationText(violation));
558 break;
559 }
560 case "BBOX":
561 output.Add($"bbox {name}: {box} {box.Width}x{box.Height}, {list.Count} rectangles, density {Density(grid.Area(1), box.Area)}%");
562 break;
563 case "DENSITY":
564 {
565 int w = int.Parse(a[2]), columns = (box.Width + w - 1) / w, rows = (box.Height + w - 1) / w;
566 if (columns > 20 || rows > 20) Fail("too many windows");
567 long[,] values = new long[rows, columns];
568 long min = long.MaxValue, max = -1;
569 int minX = 0, minY = 0, maxX = 0, maxY = 0;
570 for (int j = 0; j < rows; j++) for (int i = 0; i < columns; i++)
571 {
572 values[j, i] = grid.Clipped(new(box.X1 + i * w, box.Y1 + j * w, box.X1 + (i + 1) * w, box.Y1 + (j + 1) * w));
573 if (values[j, i] < min)
574 {
575 min = values[j, i];
576 minX = i;
577 minY = j;
578 }
579 if (values[j, i] > max)
580 {
581 max = values[j, i];
582 maxX = i;
583 maxY = j;
584 }
585 }
586 output.Add($"density {name} window {w}: {columns}x{rows}");
587 for (int j = rows - 1; j >= 0; j--)
588 {
589 List<string> row = new();
590 for (int i = 0; i < columns; i++) row.Add((100L * values[j, i] / ((long)w * w)).ToString());
591 output.Add(string.Join(" ", row));
592 }
593 output.Add($"min {Density(min, (long)w * w)}% at ({box.X1 + minX * w},{box.Y1 + minY * w})");
594 output.Add($"max {Density(max, (long)w * w)}% at ({box.X1 + maxX * w},{box.Y1 + maxY * w})");
595 break;
596 }
597 case "RENDER":
598 {
599 Box window = new(int.Parse(a[2]), int.Parse(a[3]), int.Parse(a[4]), int.Parse(a[5]));
600 if (window.X1 >= window.X2 || window.Y1 >= window.Y2) Fail("invalid window");
601 if (window.Width > 60 || window.Height > 40) Fail("window too large");
602 output.Add($"render {name} {window}");
603 for (int y = window.Y2 - 1; y >= window.Y1; y--)
604 {
605 char[] row = new char[window.Width];
606 for (int x = window.X1; x < window.X2; x++)
607 {
608 int count = list.Count(r => r.Box.X1 <= x && x < r.Box.X2 && r.Box.Y1 <= y && y < r.Box.Y2);
609 row[x - window.X1] = count == 0 ? '.' : count < 10 ? (char)('0' + count) : '*';
610 }
611 output.Add(new string(row));
612 }
613 break;
614 }
615 case "EMPTY":
616 {
617 Box empty = grid.Empty();
618 output.Add($"empty {name}: " + (empty == null ? "none" : $"{empty} area {empty.Area}"));
619 break;
620 }
621 case "HOLES":
622 {
623 List<Hole> holes = grid.Holes();
624 output.Add($"holes {name}: {holes.Count}");
625 for (int i = 0; i < holes.Count; i++) output.Add($"hole {i + 1}: area {holes[i].Area}, bbox {holes[i].Box}");
626 break;
627 }
628 case "MERGE":
629 {
630 List<Box> pieces = grid.Pieces();
631 output.Add($"merge {name}: {pieces.Count} pieces");
632 foreach (Box piece in pieces) output.Add(piece.ToString());
633 break;
634 }
635 }
636 break;
637 }
638 }
639 return output;
640 }
