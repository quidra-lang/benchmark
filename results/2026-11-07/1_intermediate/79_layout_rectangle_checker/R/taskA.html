19coverage_grid <- function(rectangles) {
20 xs <- sort(unique(unlist(lapply(rectangles, function(r) r$box[c(1, 3)]))))
21 ys <- sort(unique(unlist(lapply(rectangles, function(r) r$box[c(2, 4)]))))
22 nx <- length(xs) - 1L
23 ny <- length(ys) - 1L
24 cells <- matrix(0, nrow = ny, ncol = nx)
25 for (r in rectangles) {
26 columns <- seq.int(match(r$box[1], xs), match(r$box[3], xs) - 1L)
27 rows <- seq.int(match(r$box[2], ys), match(r$box[4], ys) - 1L)
28 cells[rows, columns] <- cells[rows, columns, drop = FALSE] + 1
29 }
30 area <- function(k = 1, window = NULL) {
31 result <- 0
32 for (j in seq_len(ny)) {
33 for (i in seq_len(nx)) {
34 if (cells[j, i] < k) next
35 dx <- if (is.null(window)) xs[i + 1L] - xs[i] else max(0, min(xs[i + 1L], window[3]) - max(xs[i], window[1]))
36 dy <- if (is.null(window)) ys[j + 1L] - ys[j] else max(0, min(ys[j + 1L], window[4]) - max(ys[j], window[2]))
37 result <- result + dx * dy
38 }
39 }
40 result
41 }
42 perimeter <- function() {
43 result <- 0
44 for (j in seq_len(ny)) {
45 for (i in seq_len(nx)) {
46 if (cells[j, i] == 0) next
47 dy <- ys[j + 1L] - ys[j]
48 dx <- xs[i + 1L] - xs[i]
49 if (i == 1 || cells[j, i - 1L] == 0) result <- result + dy
50 if (i == nx || cells[j, i + 1L] == 0) result <- result + dy
51 if (j == 1 || cells[j - 1L, i] == 0) result <- result + dx
52 if (j == ny || cells[j + 1L, i] == 0) result <- result + dx
53 }
54 }
55 result
56 }
57 largest_empty <- function() {
58 best <- NULL
59 for (left in seq_len(nx)) {
60 blocked <- rep(FALSE, ny)
61 for (right in seq.int(left, nx)) {
62 blocked <- blocked | cells[, right] > 0
63 j <- 1L
64 while (j <= ny) {
65 if (blocked[j]) {
66 j <- j + 1L
67 next
68 }
69 start <- j
70 while (j <= ny && !blocked[j]) j <- j + 1L
71 width <- xs[right + 1L] - xs[left]
72 value <- width * (ys[j] - ys[start])
73 choice <- c(-value, xs[left], ys[start], width, xs[right + 1L], ys[j])
74 if (lex_less(choice, best)) best <- choice
75 }
76 }
77 }
78 if (is.null(best)) NULL else list(box = best[c(2, 3, 5, 6)], area = -best[1])
79 }
80 holes <- function() {
81 visited <- matrix(FALSE, nrow = ny, ncol = nx)
82 result <- list()
83 for (j in seq_len(ny)) {
84 for (i in seq_len(nx)) {
85 if (cells[j, i] > 0 || visited[j, i]) next
86 queue <- list(c(i, j))
87 visited[j, i] <- TRUE
88 head <- 1L
89 area <- 0
90 boundary <- FALSE
91 x1 <- x2 <- xs[i]
92 y1 <- y2 <- ys[j]
93 anchor <- c(ys[j], xs[i])
94 while (head <= length(queue)) {
95 cell <- queue[[head]]
96 x <- cell[1]
97 y <- cell[2]
98 head <- head + 1L
99 area <- area + (xs[x + 1L] - xs[x]) * (ys[y + 1L] - ys[y])
100 x1 <- min(x1, xs[x])
101 x2 <- max(x2, xs[x + 1L])
102 y1 <- min(y1, ys[y])
103 y2 <- max(y2, ys[y + 1L])
104 candidate <- c(ys[y], xs[x])
105 if (lex_less(candidate, anchor)) anchor <- candidate
106 boundary <- boundary || x == 1 || y == 1 || x == nx || y == ny
107 for (neighbor in list(c(x - 1L, y), c(x + 1L, y), c(x, y - 1L), c(x, y + 1L))) {
108 a <- neighbor[1]
109 b <- neighbor[2]
110 if (a < 1 || b < 1 || a > nx || b > ny || cells[b, a] > 0 || visited[b, a]) next
111 visited[b, a] <- TRUE
112 queue[[length(queue) + 1L]] <- neighbor
113 }
114 }
115 if (!boundary) result[[length(result) + 1L]] <- list(anchor = anchor, area = area, box = c(x1, y1, x2, y2))
116 }
117 }
118 if (length(result) > 0) result <- result[order(vapply(result, function(h) h$anchor[1], numeric(1)), vapply(result, function(h) h$anchor[2], numeric(1)))]
119 result
120 }
121 pieces <- function() {
122 active <- result <- list()
123 for (j in seq_len(ny)) {
124 current <- list()
125 i <- 1L
126 while (i <= nx) {
127 if (cells[j, i] == 0) {
128 i <- i + 1L
129 next
130 }
131 start <- i
132 while (i <= nx && cells[j, i] > 0) i <- i + 1L
133 left <- xs[start]
134 right <- xs[i]
135 key <- paste(left, right, sep = ",")
136 bottom <- if (is.null(active[[key]])) ys[j] else active[[key]][2]
137 current[[key]] <- c(left, bottom, right)
138 active[[key]] <- NULL
139 }
140 for (run in active) result[[length(result) + 1L]] <- c(run, ys[j])
141 active <- current
142 }
143 for (run in active) result[[length(result) + 1L]] <- c(run, tail(ys, 1))
144 if (length(result) > 0) result <- result[order(vapply(result, function(r) r[2], numeric(1)), vapply(result, function(r) r[1], numeric(1)))]
145 result
146 }
147 list(area = area, perimeter = perimeter, largest_empty = largest_empty, holes = holes, pieces = pieces)
148}

169 relation <- function(a, b) {
170 box <- c(max(a$box[1], b$box[1]), max(a$box[2], b$box[2]), min(a$box[3], b$box[3]), min(a$box[4], b$box[4]))
171 ox <- box[3] - box[1]
172 oy <- box[4] - box[2]
173 kind <- if (ox < 0 || oy < 0) "separated" else if (ox == 0 && oy == 0) "corner" else if (ox == 0 || oy == 0) "edge" else "overlap"
174 direction <- if (ox < 0 && oy < 0) "diagonal" else if (ox < 0) "x" else "y"
175 list(kind = kind, box = box, ox = ox, oy = oy, gap = max(0, -ox, -oy), direction = direction)
176 }
177 contains <- function(a, b) all(c(a$box[1:2] <= b$box[1:2], a$box[3:4] >= b$box[3:4]))
178 connected <- function(a, b) relation(a, b)$kind %in% c("overlap", "edge")
179 nets <- function(rectangles) {
180 remaining <- rectangles
181 result <- list()
182 while (length(remaining) > 0) {
183 group <- remaining[1]
184 remaining <- remaining[-1]
185 i <- 1L
186 while (i <= length(group)) {

188 group <- c(group, remaining[selected])
189 remaining <- remaining[!selected]
190 i <- i + 1L
191 }
192 names <- vapply(group, function(r) r$name, character(1))
193 result[[length(result) + 1L]] <- group[order(names, method = "radix")]
194 }
195 result
196 }
