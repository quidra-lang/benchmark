84fn leap(calendar: &str, year: i64) -> bool {
85 match calendar {
86 "GREGORIAN" =>
87 year.rem_euclid(4) == 0 &&
88 (year.rem_euclid(100) != 0 || year.rem_euclid(400) == 0),
89 "JULIAN" => year.rem_euclid(4) == 0,
90 "ISLAMIC" => matches!(
91 year.rem_euclid(30), 2 | 5 | 7 | 10 | 13 | 16 | 18 | 21 | 24 | 26 | 29
92 ),
93 "HEBREW" => (7 * year + 1).rem_euclid(19) < 7,
94 "PERSIAN" => {
95 let b = (year - 474).rem_euclid(2820) + 474;
96 ((b + 38) * 682).rem_euclid(2816) < 682
97 }
98 "COPTIC" | "ETHIOPIC" => year.rem_euclid(4) == 3,
99 "FRENCH" => {
100 year.rem_euclid(4) == 0 &&
101 !matches!(year.rem_euclid(400), 100 | 200 | 300) &&
102 year.rem_euclid(4000) != 0
103 }
104 _ => (start_fixed("ISO", year + 1) - start_fixed("ISO", year)) / 7 == 53,
105 }

119fn molad(year: i64) -> Molad {
120 let quantity = 5604 + 765433 * months_before(year);
121 let (day, parts) = (quantity.div_euclid(25920), quantity.rem_euclid(25920));
122 let weekday = (day - 1373427).rem_euclid(7);
123 let mut postponed = day;
124 let mut rules = Vec::new();
125 if parts >= 19440 {
126 postponed += 1;
127 rules.push("ZAKEN");
128 } else if !leap("HEBREW", year) && weekday == 2 && parts >= 9924 {
129 postponed += 2;
130 rules.push("GATARAD");
131 } else if leap("HEBREW", year - 1) && weekday == 1 && parts >= 16789 {
132 postponed += 1;
133 rules.push("BETUTAKPAT");

135 if matches!((postponed - 1373427).rem_euclid(7), 0 | 3 | 5) {
136 postponed += 1;

138 }
139 Molad { day, parts, fixed: postponed - 1373427, rules }

142fn start_fixed(calendar: &str, year: i64) -> i64 {
143 let previous = year - 1;
144 match calendar {
145 "GREGORIAN" => {
146 365 * previous + previous.div_euclid(4) - previous.div_euclid(100) +
147 previous.div_euclid(400) + 1
148 }
149 "JULIAN" => 365 * previous + previous.div_euclid(4) - 1,
150 "ISLAMIC" => 227015 + 354 * previous + (3 + 11 * year).div_euclid(30),
151 "HEBREW" => molad(year).fixed,
152 "PERSIAN" => {
153 let a = year - 474;
154 let b = a.rem_euclid(2820) + 474;
155 226896 + 1029983 * a.div_euclid(2820) + 365 * (b - 1) +
156 (682 * b - 110).div_euclid(2816)
157 }
158 "COPTIC" => 103605 + 365 * previous + year.div_euclid(4),
159 "ETHIOPIC" => 2796 + 365 * previous + year.div_euclid(4),
160 "FRENCH" => {
161 654415 + 365 * previous + previous.div_euclid(4) -
162 previous.div_euclid(100) + previous.div_euclid(400) -
163 previous.div_euclid(4000)
164 }
165 _ => {
166 let january_four = start_fixed("GREGORIAN", year) + 3;
167 january_four - (january_four - 1).rem_euclid(7)
168 }
169 }

235fn fixed_date(calendar: &str, year: i64, month: i64, day: i64) -> i64 {
236 if calendar == "ISO" {
237 return start_fixed(calendar, year) + 7 * (month - 1) + day - 1;
238 }
239 let days = month_lengths(calendar, year);
240 start_fixed(calendar, year) + days[..(month as usize - 1)].iter().sum::<i64>() + day - 1
241}

243fn year_of(calendar: &str, fixed: i64) -> i64 {
244 if fixed < start_fixed(calendar, 1) {
245 return 0;
246 }
247 let mut low = 1;
248 let mut high = 100000;
249 while low + 1 < high {
250 let middle = (low + high) / 2;
251 if start_fixed(calendar, middle) <= fixed {
252 low = middle;
253 } else {
254 high = middle;
255 }
256 }
257 low
258}

260fn parts_of(calendar: &str, fixed: i64) -> Option<(i64, i64, i64)> {
261 let year = year_of(calendar, fixed);
262 if year == 0 {
263 return None;
264 }
265 let mut offset = fixed - start_fixed(calendar, year);
266 if calendar == "ISO" {
267 return Some((year, offset / 7 + 1, offset % 7 + 1));
268 }
269 for (month, length) in month_lengths(calendar, year).iter().enumerate() {
270 if offset < *length {
271 return Some((year, month as i64 + 1, offset + 1));
272 }
273 offset -= length;
274 }
275 unreachable!("inconsistent calendar year length");
276}

278fn format_date(calendar: &str, fixed: i64) -> String {
279 if !(1..=MAX_FIXED).contains(&fixed) {
280 return "OUT OF RANGE".into();
281 }
282 if year_calendar(calendar) {
283 let Some((year, month, day)) = parts_of(calendar, fixed) else {
284 return "OUT OF RANGE".into();
285 };
286 if calendar == "ISO" {
287 return format!("{year:04}-W{month:02}-{day}");
288 }
289 return format!("{year:04}-{month:02}-{day:02}");
290 }
291 if calendar == "JD" {
292 return (fixed + 1721425).to_string();
293 }
294 if calendar == "FIXED" {
295 return fixed.to_string();
296 }
297 let count = fixed + 1137142;
298 let long_count = [
299 count.div_euclid(144000),
300 count.div_euclid(7200).rem_euclid(20),
301 count.div_euclid(360).rem_euclid(20),
302 count.div_euclid(20).rem_euclid(18),
303 count.rem_euclid(20),
304 ];
305 let tz_number = (count + 3).rem_euclid(13) + 1;
306 let tz_name = TZOLKIN[((count + 19).rem_euclid(20)) as usize];
307 let haab = (count + 348).rem_euclid(365);
308 format!(
309 "{} {} {} {} {}",
310 long_count.iter().map(i64::to_string).collect::<Vec<_>>().join("."),
311 tz_number, tz_name, haab.rem_euclid(20), HAAB[(haab / 20) as usize]
312 )
313}

315fn decimal_digits(text: &str, low: usize, high: usize) -> bool {
316 (low..=high).contains(&text.len()) && text.bytes().all(|b| b.is_ascii_digit())

319fn parse_date(calendar: &str, text: &str) -> Result<i64> {
320 let fixed;
321 if ymd(calendar) {
322 let fields: Vec<&str> = text.split('-').collect();
323 if fields.len() != 3 || !decimal_digits(fields[0], 1, 5) ||
324 !decimal_digits(fields[1], 1, 2) || !decimal_digits(fields[2], 1, 2) {
325 return Err(Problem::Date("FORMAT"));
326 }
327 let year = fields[0].parse::<i64>().unwrap();
328 let month = fields[1].parse::<i64>().unwrap();
329 let day = fields[2].parse::<i64>().unwrap();
330 if year == 0 { return Err(Problem::Date("YEAR")); }
331 let days = month_lengths(calendar, year);
332 if month < 1 || month > days.len() as i64 {
333 return Err(Problem::Date("MONTH"));
334 }
335 if day < 1 || day > days[month as usize - 1] {
336 return Err(Problem::Date("DAY"));
337 }
338 fixed = fixed_date(calendar, year, month, day);
339 } else if calendar == "ISO" {
340 let fields: Vec<&str> = text.split('-').collect();
341 if fields.len() != 3 || !decimal_digits(fields[0], 1, 5) ||
342 !fields[1].starts_with('W') || !decimal_digits(&fields[1][1..], 1, 2) ||
343 !decimal_digits(fields[2], 1, 1) {
344 return Err(Problem::Date("FORMAT"));
345 }
346 let year = fields[0].parse::<i64>().unwrap();
347 let week = fields[1][1..].parse::<i64>().unwrap();
348 let day = fields[2].parse::<i64>().unwrap();
349 if year == 0 { return Err(Problem::Date("YEAR")); }
350 let weeks = (start_fixed(calendar, year + 1) - start_fixed(calendar, year)) / 7;
351 if week < 1 || week > weeks { return Err(Problem::Date("WEEK")); }
352 if !(1..=7).contains(&day) { return Err(Problem::Date("DAY")); }
353 fixed = fixed_date(calendar, year, week, day);
354 } else if calendar == "MAYAN" {
355 let fields: Vec<&str> = text.split('.').collect();
356 if fields.len() != 5 || fields.iter().any(|s| !decimal_digits(s, 1, 2)) {
357 return Err(Problem::Date("FORMAT"));
358 }
359 let parts: Vec<i64> = fields.iter().map(|s| s.parse().unwrap()).collect();
360 let (b, k, t, u, n) = (parts[0], parts[1], parts[2], parts[3], parts[4]);
361 if k > 19 || t > 19 || u > 17 || n > 19 {
362 return Err(Problem::Date("FIELD"));
363 }
364 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142;
365 } else {
366 let digits = text.strip_prefix('-').unwrap_or(text);
367 if !decimal_digits(digits, 1, 8) {
368 return Err(Problem::Date("FORMAT"));
369 }
370 fixed = text.parse::<i64>().unwrap() - if calendar == "JD" { 1721425 } else { 0 };
371 }
372 if !(1..=MAX_FIXED).contains(&fixed) {
373 return Err(Problem::Date("RANGE"));
374 }
375 Ok(fixed)
376}

387fn shift_date(calendar: &str, fixed: i64, years: i64, months: i64) -> Option<i64> {
388 let (year, mut month, day) = parts_of(calendar, fixed)?;
389 let target_year = year + years;
390 if calendar == "ISO" {
391 if !(1..=99999).contains(&target_year) {
392 return None;
393 }
394 let weeks = (start_fixed(calendar, target_year + 1) -
395 start_fixed(calendar, target_year)) / 7;
396 return Some(fixed_date(calendar, target_year, month.min(weeks), day));
397 }
398 if calendar == "HEBREW" {
399 let old_names = month_names(calendar, year);
400 let mut old_name = old_names[month as usize - 1];
401 let new_names = month_names(calendar, target_year);
402 if !new_names.contains(&old_name) {
403 old_name = if old_name == "Adar" { "AdarII" } else { "Adar" };
404 }
405 month = new_names.iter().position(|name| *name == old_name)? as i64 + 1;
406 }
407 let absolute = month_index(calendar, target_year) + month - 1 + months;
408 let final_year;
409 if calendar == "HEBREW" {
410 let mut candidate = (19 * absolute + 234).div_euclid(235);
411 while month_index(calendar, candidate) > absolute {
412 candidate -= 1;
413 }
414 while month_index(calendar, candidate + 1) <= absolute {
415 candidate += 1;
416 }
417 final_year = candidate;
418 } else {
419 let count = month_names(calendar, target_year).len() as i64;
420 final_year = absolute.div_euclid(count) + 1;
421 }
422 if !(1..=99999).contains(&final_year) {
423 return None;
424 }
425 let final_month = absolute - month_index(calendar, final_year) + 1;
426 let final_day = day.min(month_lengths(calendar, final_year)[final_month as usize - 1]);
427 Some(fixed_date(calendar, final_year, final_month, final_day))
428}

430fn month_bounds(calendar: &str, year: i64, month: i64) -> Result<(i64, i64)> {
431 let lengths = month_lengths(calendar, year);

434 }

440 let mut year = year_of(calendar, first).max(1);

445 while start_fixed(calendar, year) <= last {

459fn easter(year: i64, calendar: &str) -> Easter {
460 let golden = year.rem_euclid(19);
461 let mut epact = if calendar == "GREGORIAN" {
462 let century = year / 100 + 1;
463 (14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25).rem_euclid(30)
464 } else {
465 (14 + 11 * golden).rem_euclid(30)
466 };
467 if calendar == "GREGORIAN" && (epact == 0 || (epact == 1 && golden > 10)) {
468 epact += 1;
469 }
470 let moon = fixed_date(calendar, year, 4, 19) - epact;
471 let sunday = moon + 7 - moon.rem_euclid(7);
472 let index = (-start_fixed(calendar, year)).rem_euclid(7);
473 let alphabet = b"ABCDEFG";
474 let mut letters = (alphabet[index as usize] as char).to_string();
475 if leap(calendar, year) {
476 letters.push(alphabet[(index - 1).rem_euclid(7) as usize] as char);
477 }
478 Easter { epact, letters, moon, sunday }
479}

481fn holiday_lines(year: i64) -> Vec<String> {
482 let first = start_fixed("GREGORIAN", year);
483 let last = start_fixed("GREGORIAN", year + 1) - 1;
484 let west = easter(year, "GREGORIAN").sunday;
485 let east = easter(year, "JULIAN").sunday;
486 let movable = [
487 ("AshWednesday", -46), ("GoodFriday", -2), ("Easter", 0),
488 ("Ascension", 39), ("Pentecost", 49),
489 ];
490 let mut entries: Vec<(i64, usize, &str, &str)> = Vec::new();
491 for (order, (name, offset)) in movable.into_iter().enumerate() {
492 entries.push((west + offset, order, name, "GREGORIAN"));
493 }
494 entries.push((east, 5, "OrthodoxEaster", "JULIAN"));
495 for (index, (name, calendar, month_name, day)) in HOLIDAYS.iter().enumerate() {
496 let mut candidate_year = year_of(calendar, first).max(1);
497 while start_fixed(calendar, candidate_year) <= last {
498 let name_in_year = if *calendar == "HEBREW" && *month_name == "Adar" &&
499 leap(calendar, candidate_year) {
500 "AdarII"
501 } else {
502 month_name
503 };
504 let names = month_names(calendar, candidate_year);
505 let month = names.iter().position(|s| *s == name_in_year).unwrap() as i64 + 1;
506 if *day <= month_lengths(calendar, candidate_year)[month as usize - 1] {
507 let fixed = fixed_date(calendar, candidate_year, month, *day);
508 if first <= fixed && fixed <= last {
509 entries.push((fixed, index + 6, *name, *calendar));
510 }
511 }
512 candidate_year += 1;
513 }
514 }
515 entries.sort();
516 entries.into_iter().map(|(day, _, name, calendar)| {
517 format!(
518 "{} {} {} {} {}",
519 format_date("GREGORIAN", day), WEEKDAYS[day.rem_euclid(7) as usize],
520 name, calendar, format_date(calendar, day)
521 )
522 }).collect()
523}

542fn validate_arguments(command: &str, args: &[&str]) -> Result<()> {
543 if matches!(command, "EASTER" | "COMPUTUS") {
544 return checked(valid_year(args[0]) && matches!(args[1], "GREGORIAN" | "JULIAN"));
545 }
546 if matches!(command, "PASSOVER" | "HOLIDAYS" | "MOLAD") {
547 return checked(valid_year(args[0]));
548 }
549 let calendar = args[0];
550 let valid_calendar = if matches!(command, "INFO" | "MONTHS" | "MONTHCAL" | "NTH" | "SPAN") {
551 ymd(calendar)
552 } else if matches!(command, "LEAP" | "NEWYEAR") {
553 year_calendar(calendar)
554 } else {
555 CALENDARS.contains(&calendar)
556 };
557 checked(valid_calendar)?;
558 match command {
559 "CONVERT" => checked(args[2] == "TO" && CALENDARS.contains(&args[3])),
560 "DIFF" => checked(CALENDARS.contains(&args[2])),
561 "LEAP" | "MONTHS" | "MONTHCAL" | "NTH" | "NEWYEAR" => {
562 checked(valid_year(args[1]))?;
563 if matches!(command, "MONTHCAL" | "NTH") {
564 checked(decimal_digits(args[2], 1, 2))?;
565 }
566 if command == "NTH" {
567 checked(matches!(args[3], "1" | "2" | "3" | "4" | "5" | "LAST"))?;
568 checked(WEEKDAYS.contains(&args[4]))?;
569 }
570 Ok(())
571 }
572 "COUNT" => checked(WEEKDAYS.contains(&args[3])),
573 "ADD" => {
574 let digits = args[2].strip_prefix('-').unwrap_or(args[2]);
575 checked(decimal_digits(digits, 1, 7))?;
576 checked(matches!(args[3], "DAYS" | "WEEKS" | "MONTHS" | "YEARS"))?;
577 if args[3] == "MONTHS" {
578 checked(ymd(calendar))?;
579 }
580 if args[3] == "YEARS" {
581 checked(year_calendar(calendar))?;
582 }
583 Ok(())
584 }
585 "ROUND" => {
586 checked(decimal_digits(args[2], 1, 2))?;
587 let number: i64 = args[2].parse().unwrap();
588 checked((1..=13).contains(&number) && TZOLKIN.contains(&args[3]))?;
589 checked(decimal_digits(args[4], 1, 2))?;
590 checked(HAAB.contains(&args[5]))?;
591 let day: i64 = args[4].parse().unwrap();
592 checked(day <= (if args[5] == "Uayeb" { 4 } else { 19 }))
593 }
594 _ => Ok(()),
595 }
596}

598fn calendar_lines(calendar: &str, year: i64, month: i64) -> Result<Vec<String>> {
599 let (first, last) = month_bounds(calendar, year, month)?;
600 if !(1..=MAX_FIXED).contains(&first) || !(1..=MAX_FIXED).contains(&last) {
601 return Ok(vec!["OUT OF RANGE".to_string()]);
602 }
603 let title = format!(
604 "{} {} ({} to {})",
605 month_names(calendar, year)[month as usize - 1],
606 year, format_date("GREGORIAN", first), format_date("GREGORIAN", last)
607 );
608 let mut lines = vec![title, "Mo Tu We Th Fr Sa Su".to_string()];
609 let mut cursor = first - (first - 1).rem_euclid(7);
610 while cursor <= last {
611 let fields: Vec<String> = (cursor..cursor + 7).map(|day| {
612 if first <= day && day <= last {
613 format!("{:>2}", day - first + 1)
614 } else {
615 " ".into()
616 }
617 }).collect();
618 lines.push(fields.join(" ").trim_end().to_string());
619 cursor += 7;
620 }
621 Ok(lines)
622}

624fn span_text(calendar: &str, first: i64, last: i64) -> String {
625 let negative = first > last;
626 let (first, last) = if negative { (last, first) } else { (first, last) };
627 let start_year = parts_of(calendar, first).unwrap().0;
628 let end_year = parts_of(calendar, last).unwrap().0;
629 let mut low = 0;
630 let mut high = end_year - start_year + 2;
631 while low + 1 < high {
632 let middle = (low + high) / 2;
633 if shift_date(calendar, first, middle, 0).is_some_and(|day| day <= last) {
634 low = middle;
635 } else {
636 high = middle;
637 }
638 }
639 let years = low;
640 low = 0;
641 high = 32;
642 while low + 1 < high {
643 let middle = (low + high) / 2;
644 if shift_date(calendar, first, years, middle).is_some_and(|day| day <= last) {
645 low = middle;
646 } else {
647 high = middle;
648 }
649 }
650 let months = low;
651 let days = last - shift_date(calendar, first, years, months).unwrap();
652 format!("{}{}y {}m {}d", if negative { "-" } else { "" }, years, months, days)
653}

655fn round_date(calendar: &str, fixed: i64, number: i64, tz_name: &str,
656 day: i64, month: &str) -> String {
657 let target_tz = TZOLKIN.iter().position(|name| *name == tz_name).unwrap() as i64;
658 let target_haab = HAAB.iter().position(|name| *name == month).unwrap() as i64 * 20 + day;
659 let mut candidate = fixed;
660 loop {
661 let count = candidate + 1137142;
662 if (count + 3).rem_euclid(13) + 1 == number &&
663 (count + 19).rem_euclid(20) == target_tz {
664 break;
665 }
666 candidate += 1;
667 }
668 while candidate <= fixed + 18979 {
669 if (candidate + 1137142 + 348).rem_euclid(365) == target_haab {
670 return format_date(calendar, candidate);
671 }
672 candidate += 260;
673 }
674 "NONE".to_string()
675}

677fn execute(command: &str, args: &[&str]) -> Result<Vec<String>> {
678 validate_arguments(command, args)?;
679 if matches!(command, "EASTER" | "COMPUTUS") {
680 let year: i64 = args[0].parse().unwrap();
681 if year > 9999 {
682 return Ok(vec!["OUT OF RANGE".into()]);
683 }
684 let kind = args[1];
685 let Easter { epact, letters, moon, sunday } = easter(year, kind);
686 if command == "EASTER" {
687 return Ok(vec![format_date("GREGORIAN", sunday)]);
688 }
689 return Ok(vec![format!(
690 "golden={} epact={} letters={} moon={} easter={}",
691 year.rem_euclid(19) + 1, epact, letters,
692 format_date(kind, moon), format_date(kind, sunday)
693 )]);
694 }
695 if matches!(command, "PASSOVER" | "HOLIDAYS" | "MOLAD") {
696 let year: i64 = args[0].parse().unwrap();
697 if command == "MOLAD" {
698 let entry = molad(year);
699 if !(1..=MAX_FIXED).contains(&entry.fixed) {
700 return Ok(vec!["OUT OF RANGE".into()]);
701 }
702 let applied = if entry.rules.is_empty() {
703 "NONE".into()
704 } else {
705 entry.rules.join("+")
706 };
707 return Ok(vec![format!(
708 "{} {}h {}p {} {} {}",
709 WEEKDAYS[(entry.day - 1373427).rem_euclid(7) as usize],
710 entry.parts / 1080, entry.parts % 1080, applied,
711 WEEKDAYS[entry.fixed.rem_euclid(7) as usize],
712 format_date("GREGORIAN", entry.fixed)
713 )]);
714 }
715 if year > 9999 {
716 return Ok(vec!["OUT OF RANGE".into()]);
717 }
718 if command == "HOLIDAYS" {
719 return Ok(holiday_lines(year));
720 }
721 let hebrew_year = year + 3760;
722 let month = month_names("HEBREW", hebrew_year)
723 .iter().position(|name| *name == "Nisan").unwrap() as i64 + 1;
724 let fixed = fixed_date("HEBREW", hebrew_year, month, 15);
725 return Ok(vec![format!(
726 "{} {}", format_date("GREGORIAN", fixed),
727 WEEKDAYS[fixed.rem_euclid(7) as usize]
728 )]);
729 }
730
731 let calendar = args[0];
732 if matches!(command, "LEAP" | "MONTHS" | "MONTHCAL" | "NTH" | "NEWYEAR") {
733 let year: i64 = args[1].parse().unwrap();
734 if command == "LEAP" {
735 let mut result = format!(
736 "{} {}", if leap(calendar, year) { "YES" } else { "NO" },
737 start_fixed(calendar, year + 1) - start_fixed(calendar, year)
738 );
739 if calendar == "HEBREW" {
740 result.push(' ');
741 result.push_str(year_type(year));
742 }
743 return Ok(vec![result]);
744 }
745 if command == "MONTHS" {
746 let names = month_names(calendar, year);
747 let lengths = month_lengths(calendar, year);
748 return Ok(vec![names.iter().zip(lengths.iter())
749 .map(|(name, days)| format!("{name}={days}"))
750 .collect::<Vec<_>>().join(" ")]);
751 }
752 if command == "NEWYEAR" {
753 if year > 9999 {
754 return Ok(vec!["OUT OF RANGE".into()]);
755 }
756 let first = start_fixed("GREGORIAN", year);
757 let last = start_fixed("GREGORIAN", year + 1) - 1;
758 let entries = new_years(calendar, first, last);
759 return Ok(vec![if entries.is_empty() { "NONE".into() } else { entries.join(" ") }]);
760 }
761 let month: i64 = args[2].parse().unwrap();
762 if command == "MONTHCAL" {
763 return calendar_lines(calendar, year, month);
764 }
765 let (first, last) = month_bounds(calendar, year, month)?;
766 let weekday = WEEKDAYS.iter().position(|name| *name == args[4]).unwrap() as i64;
767 let fixed = if args[3] == "LAST" {
768 last - (last - weekday).rem_euclid(7)
769 } else {
770 first + (weekday - first).rem_euclid(7) +
771 7 * (args[3].parse::<i64>().unwrap() - 1)
772 };
773 return Ok(vec![if fixed > last {
774 "NONE".to_string()
775 } else {
776 format_date(calendar, fixed)
777 }]);
778 }
779
780 if command == "VALID" {
781 return Ok(vec![match parse_date(calendar, args[1]) {
782 Ok(day) => format!("YES {}", format_date(calendar, day)),
783 Err(Problem::Date(reason)) => format!("NO {reason}"),
784 Err(Problem::Command) => unreachable!(),
785 }]);
786 }
787 let first = parse_date(calendar, args[1])?;
788 match command {
789 "DIFF" => {
790 let last = parse_date(args[2], args[3])?;
791 Ok(vec![(last - first).to_string()])
792 }
793 "COUNT" | "SPAN" => {
794 let last = parse_date(calendar, args[2])?;
795 if command == "SPAN" {
796 return Ok(vec![span_text(calendar, first, last)]);
797 }
798 let (low, high) = if first <= last { (first, last) } else { (last, first) };
799 let weekday = WEEKDAYS.iter().position(|name| *name == args[3]).unwrap() as i64;
800 let occurrence = low + (weekday - low).rem_euclid(7);
801 Ok(vec![if occurrence > high {
802 "0".to_string()
803 } else {
804 ((high - occurrence) / 7 + 1).to_string()
805 }])
806 }
807 "CONVERT" => Ok(vec![format_date(args[3], first)]),
808 "WEEKDAY" => Ok(vec![WEEKDAYS[first.rem_euclid(7) as usize].into()]),
809 "ALL" => {
810 let mut rows = Vec::new();
811 for target in CALENDARS {
812 let mut text = format_date(target, first);
813 if ymd(target) && text != "OUT OF RANGE" {
814 let (year, month, _) = parts_of(target, first).unwrap();
815 text.push(' ');
816 text.push_str(month_names(target, year)[month as usize - 1]);
817 }
818 rows.push(format!("{target} {text}"));
819 }
820 Ok(rows)
821 }
822 "INFO" => {
823 let (year, month, day) = parts_of(calendar, first).unwrap();
824 let weekday = if calendar == "FRENCH" {
825 if month <= 12 {
826 DECADES[(day - 1).rem_euclid(10) as usize]
827 } else {
828 FESTIVALS[(day - 1) as usize]
829 }
830 } else {
831 WEEKDAYS[first.rem_euclid(7) as usize]
832 };
833 let ordinal = first - start_fixed(calendar, year) + 1;
834 let length = start_fixed(calendar, year + 1) - start_fixed(calendar, year);
835 Ok(vec![format!(
836 "{} {} {} {} day={}/{}",
837 weekday, day, month_names(calendar, year)[month as usize - 1],
838 year, ordinal, length
839 )])
840 }
841 "ROUND" => {
842 let number = args[2].parse::<i64>().unwrap();
843 let day = args[4].parse::<i64>().unwrap();
844 Ok(vec![round_date(calendar, first, number, args[3], day, args[5])])
845 }
846 "ADD" => {
847 let amount = args[2].parse::<i64>().unwrap();
848 let result = match args[3] {
849 "DAYS" => Some(first + amount),
850 "WEEKS" => Some(first + amount * 7),
851 "YEARS" => shift_date(calendar, first, amount, 0),
852 _ => shift_date(calendar, first, 0, amount),
853 };
854 Ok(vec![result.map(|day| format_date(calendar, day))
855 .unwrap_or_else(|| "OUT OF RANGE".into())])
856 }
857 _ => unreachable!(),
858 }
859}

861fn main() {
862 let stdin = io::stdin();
863 let mut lines = stdin.lock().lines();
864 let count: usize = lines.next().unwrap().unwrap().parse().unwrap();
865 let mut output: Vec<String> = Vec::new();
866 for _ in 0..count {
867 let line = lines.next().unwrap().unwrap();
868 let words: Vec<&str> = line.split(' ').collect();
869 let command = words[0];
870 let arguments = &words[1..];
871 if arity(command) != Some(arguments.len()) {
872 output.push("INVALID COMMAND".into());
873 continue;
874 }
875 match execute(command, arguments) {
876 Ok(rows) => output.extend(rows),
877 Err(Problem::Command) => output.push("INVALID COMMAND".into()),
878 Err(Problem::Date(reason)) => output.push(
879 if reason == "RANGE" { "OUT OF RANGE" } else { "INVALID DATE" }.into()
880 ),
881 }
882 }
883 println!("{}", output.join("\n"));
884}
