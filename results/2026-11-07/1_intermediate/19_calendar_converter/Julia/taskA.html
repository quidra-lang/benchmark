50function is_leap(calendar::String, year::Int)
51 if calendar == "GREGORIAN"
52 return mod(year, 4) == 0 && (mod(year, 100) != 0 || mod(year, 400) == 0)
53 elseif calendar == "JULIAN"
54 return mod(year, 4) == 0
55 elseif calendar == "ISLAMIC"
56 return mod(year, 30) in (2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29)
57 elseif calendar == "HEBREW"
58 return mod(7 * year + 1, 19) < 7
59 elseif calendar == "PERSIAN"
60 b = mod(year - 474, 2820) + 474
61 return mod((b + 38) * 682, 2816) < 682
62 elseif calendar in ("COPTIC", "ETHIOPIC")
63 return mod(year, 4) == 3
64 elseif calendar == "FRENCH"
65 return mod(year, 4) == 0 &&
66 !(mod(year, 400) in (100, 200, 300)) && mod(year, 4000) != 0
67 end
68 div(start_fixed("ISO", year + 1) - start_fixed("ISO", year), 7) == 53
69end

73function molad(year::Int)
74 quantity = 5604 + 765433 * months_before(year)
75 day, parts = fldmod(quantity, 25920)
76 weekday = mod(day - 1373427, 7)
77 postponed = day
78 rules = String[]
79 if parts >= 19440
80 postponed += 1
81 push!(rules, "ZAKEN")
82 elseif !is_leap("HEBREW", year) && weekday == 2 && parts >= 9924
83 postponed += 2
84 push!(rules, "GATARAD")
85 elseif is_leap("HEBREW", year - 1) && weekday == 1 && parts >= 16789
86 postponed += 1
87 push!(rules, "BETUTAKPAT")
88 end
89 if mod(postponed - 1373427, 7) in (0, 3, 5)
90 postponed += 1
91 push!(rules, "ADU")
92 end
93 return (day=day, parts=parts, fixed=postponed - 1373427, rules=rules)
94end

96function start_fixed(calendar::String, year::Int)
97 previous = year - 1
98 if calendar == "GREGORIAN"
99 return 365 * previous + fld(previous, 4) - fld(previous, 100) +
100 fld(previous, 400) + 1
101 elseif calendar == "JULIAN"
102 return 365 * previous + fld(previous, 4) - 1
103 elseif calendar == "ISLAMIC"
104 return 227015 + 354 * previous + fld(3 + 11 * year, 30)
105 elseif calendar == "HEBREW"
106 return molad(year).fixed
107 elseif calendar == "PERSIAN"
108 a = year - 474
109 b = mod(a, 2820) + 474
110 return 226896 + 1029983 * fld(a, 2820) + 365 * (b - 1) +
111 fld(682 * b - 110, 2816)
112 elseif calendar == "COPTIC"
113 return 103605 + 365 * previous + fld(year, 4)
114 elseif calendar == "ETHIOPIC"
115 return 2796 + 365 * previous + fld(year, 4)
116 elseif calendar == "FRENCH"
117 return 654415 + 365 * previous + fld(previous, 4) -
118 fld(previous, 100) + fld(previous, 400) - fld(previous, 4000)
119 end
120 january_four = start_fixed("GREGORIAN", year) + 3
121 january_four - mod(january_four - 1, 7)
122end

125 length = start_fixed("HEBREW", year + 1) - start_fixed("HEBREW", year)
126 if length in (353, 383)
127 return "DEFICIENT"

131 "COMPLETE"

141function month_lengths(calendar::String, year::Int)
142 extra = Int(is_leap(calendar, year))
143 if calendar in ("GREGORIAN", "JULIAN")
144 return [31, 28 + extra, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
145 elseif calendar == "ISLAMIC"
146 return vcat(repeat([30, 29], 5), [30, 29 + extra])
147 elseif calendar == "PERSIAN"
148 return vcat(fill(31, 6), fill(30, 5), [29 + extra])
149 elseif calendar in ("COPTIC", "ETHIOPIC", "FRENCH")
150 return vcat(fill(30, 12), [5 + extra])
151 end
152 kind = year_type(year)
153 heshvan = kind == "COMPLETE" ? 30 : 29
154 kislev = kind == "DEFICIENT" ? 29 : 30
155 if extra == 1
156 return [30, heshvan, kislev, 29, 30, 30, 29, 30, 29, 30, 29, 30, 29]
157 end
158 [30, heshvan, kislev, 29, 30, 29, 30, 29, 30, 29, 30, 29]
159end

161function fixed_date(calendar::String, year::Int, month::Int, day::Int)
162 if calendar == "ISO"
163 return start_fixed(calendar, year) + 7 * (month - 1) + day - 1
164 end
165 lengths = month_lengths(calendar, year)
166 start_fixed(calendar, year) + sum(lengths[1:month-1]; init=0) + day - 1
167end

169function year_of(calendar::String, fixed::Int)
170 fixed < start_fixed(calendar, 1) && return 0
171 low, high = 1, 100000
172 while low + 1 < high
173 middle = fld(low + high, 2)
174 if start_fixed(calendar, middle) <= fixed
175 low = middle
176 else
177 high = middle
178 end
179 end
180 low
181end

183function parts_of(calendar::String, fixed::Int)
184 year = year_of(calendar, fixed)
185 year == 0 && return nothing
186 offset = fixed - start_fixed(calendar, year)
187 if calendar == "ISO"
188 return (year, fld(offset, 7) + 1, mod(offset, 7) + 1)
189 end
190 for (month, days) in enumerate(month_lengths(calendar, year))
191 if offset < days
192 return (year, month, offset + 1)
193 end
194 offset -= days
195 end
196 error("Calendar year has an inconsistent length")
197end

199function format_date(calendar::String, fixed::Int)
200 in_range(fixed) || return "OUT OF RANGE"
201 if calendar in YEAR_CALENDARS
202 fields = parts_of(calendar, fixed)
203 fields === nothing && return "OUT OF RANGE"
204 year, month, day = fields
205 if calendar == "ISO"
206 return @sprintf("%04d-W%02d-%d", year, month, day)
207 end
208 return @sprintf("%04d-%02d-%02d", year, month, day)
209 elseif calendar == "JD"
210 return string(fixed + 1721425)
211 elseif calendar == "FIXED"
212 return string(fixed)
213 end
214 count = fixed + 1137142
215 long_count = [
216 fld(count, 144000), mod(fld(count, 7200), 20),
217 mod(fld(count, 360), 20), mod(fld(count, 20), 18), mod(count, 20)
218 ]
219 tz_number = mod(count + 3, 13) + 1
220 tz_name = TZOLKIN[mod(count + 19, 20) + 1]
221 haab = mod(count + 348, 365)
222 join(long_count, ".") * " $tz_number $tz_name " *
223 string(mod(haab, 20)) * " " * HAAB[fld(haab, 20) + 1]
224end

226function parse_date(calendar::String, text::String)
227 fixed = 0
228 if calendar in YMD
229 occursin(r"^[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}$", text) ||
230 throw(DateError("FORMAT"))
231 year, month, day = parse.(Int, split(text, '-'))
232 year != 0 || throw(DateError("YEAR"))
233 lengths = month_lengths(calendar, year)
234 1 <= month <= length(lengths) || throw(DateError("MONTH"))
235 1 <= day <= lengths[month] || throw(DateError("DAY"))
236 fixed = fixed_date(calendar, year, month, day)
237 elseif calendar == "ISO"
238 occursin(r"^[0-9]{1,5}-W[0-9]{1,2}-[0-9]$", text) ||
239 throw(DateError("FORMAT"))
240 year_text, week_text, day_text = split(text, '-')
241 year = parse(Int, year_text)
242 week = parse(Int, week_text[2:end])
243 day = parse(Int, day_text)
244 year != 0 || throw(DateError("YEAR"))
245 weeks = div(start_fixed(calendar, year + 1) - start_fixed(calendar, year), 7)
246 1 <= week <= weeks || throw(DateError("WEEK"))
247 1 <= day <= 7 || throw(DateError("DAY"))
248 fixed = fixed_date(calendar, year, week, day)
249 elseif calendar == "MAYAN"
250 occursin(r"^[0-9]{1,2}(\.[0-9]{1,2}){4}$", text) ||
251 throw(DateError("FORMAT"))
252 b, k, t, u, n = parse.(Int, split(text, '.'))
253 k <= 19 && t <= 19 && u <= 17 && n <= 19 || throw(DateError("FIELD"))
254 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142
255 else
256 occursin(r"^-?[0-9]{1,8}$", text) || throw(DateError("FORMAT"))
257 fixed = parse(Int, text) - (calendar == "JD" ? 1721425 : 0)
258 end
259 in_range(fixed) || throw(DateError("RANGE"))
260 fixed
261end

264 if calendar == "HEBREW"
265 return months_before(year)
266 end
267 (calendar in ("COPTIC", "ETHIOPIC", "FRENCH") ? 13 : 12) * (year - 1)

270function shift_date(calendar::String, fixed::Int, years::Int, months::Int)
271 parts = parts_of(calendar, fixed)
272 parts === nothing && return nothing
273 year, month, day = parts
274 target_year = year + years
275 if calendar == "ISO"
276 1 <= target_year <= 99999 || return nothing
277 weeks = div(start_fixed(calendar, target_year + 1) -
278 start_fixed(calendar, target_year), 7)
279 return fixed_date(calendar, target_year, min(month, weeks), day)
280 end
281 if calendar == "HEBREW"
282 old_name = names_for(calendar, year)[month]
283 new_names = names_for(calendar, target_year)
284 if !(old_name in new_names)
285 old_name = old_name == "Adar" ? "AdarII" : "Adar"
286 end
287 month = findfirst(==(old_name), new_names)
288 end
289 absolute = month_index(calendar, target_year) + month - 1 + months
290 if calendar == "HEBREW"
291 final_year = fld(19 * absolute + 234, 235)
292 while month_index(calendar, final_year) > absolute
293 final_year -= 1
294 end
295 while month_index(calendar, final_year + 1) <= absolute
296 final_year += 1
297 end
298 else
299 count = length(names_for(calendar, target_year))
300 final_year = fld(absolute, count) + 1
301 end
302 1 <= final_year <= 99999 || return nothing
303 final_month = absolute - month_index(calendar, final_year) + 1
304 final_day = min(day, month_lengths(calendar, final_year)[final_month])
305 fixed_date(calendar, final_year, final_month, final_day)
306end

308function month_bounds(calendar::String, year::Int, month::Int)
309 lengths = month_lengths(calendar, year)
310 1 <= month <= length(lengths) || throw(DateError("MONTH"))
311 first = fixed_date(calendar, year, month, 1)
312 (first, first + lengths[month] - 1)
313end

315function new_years(calendar::String, first::Int, last::Int)
316 year = max(1, year_of(calendar, first))
317 while start_fixed(calendar, year) < first
318 year += 1
319 end

321 while start_fixed(calendar, year) <= last

323 year += 1
324 end

328function easter(year::Int, calendar::String)
329 golden = mod(year, 19)
330 if calendar == "GREGORIAN"
331 century = fld(year, 100) + 1
332 epact = mod(14 + 11 * golden - fld(3 * century, 4) +
333 fld(5 + 8 * century, 25), 30)
334 if epact == 0 || (epact == 1 && golden > 10)
335 epact += 1
336 end
337 else
338 epact = mod(14 + 11 * golden, 30)
339 end
340 moon = fixed_date(calendar, year, 4, 19) - epact
341 sunday = moon + 7 - mod(moon, 7)
342 index = mod(-start_fixed(calendar, year), 7)
343 letters = string("ABCDEFG"[index + 1])
344 if is_leap(calendar, year)
345 letters *= string("ABCDEFG"[mod(index - 1, 7) + 1])
346 end
347 (epact, letters, moon, sunday)
348end

378function holiday_lines(year::Int)
379 first = start_fixed("GREGORIAN", year)
380 last = start_fixed("GREGORIAN", year + 1) - 1
381 western_easter = easter(year, "GREGORIAN")[4]
382 orthodox_easter = easter(year, "JULIAN")[4]
383 entries = Tuple{Int, Int, String, String}[]
384 movable = [
385 ("AshWednesday", -46), ("GoodFriday", -2), ("Easter", 0),
386 ("Ascension", 39), ("Pentecost", 49)
387 ]
388 for (order, (name, offset)) in enumerate(movable)
389 push!(entries, (western_easter + offset, order - 1, name, "GREGORIAN"))
390 end
391 push!(entries, (orthodox_easter, 5, "OrthodoxEaster", "JULIAN"))
392 for (order, (name, calendar, month_name, day)) in enumerate(HOLIDAYS)
393 candidate_year = max(1, year_of(calendar, first))
394 while start_fixed(calendar, candidate_year) <= last
395 name_in_year = calendar == "HEBREW" && month_name == "Adar" &&
396 is_leap(calendar, candidate_year) ? "AdarII" : month_name
397 month = findfirst(==(name_in_year), names_for(calendar, candidate_year))
398 if day <= month_lengths(calendar, candidate_year)[month]
399 fixed = fixed_date(calendar, candidate_year, month, day)
400 if first <= fixed <= last
401 push!(entries, (fixed, order + 5, name, calendar))
402 end
403 end
404 candidate_year += 1
405 end
406 end
407 sort!(entries)
408 [
409 format_date("GREGORIAN", fixed) * " " * WEEKDAYS[mod(fixed, 7) + 1] *
410 " " * name * " " * calendar * " " * format_date(calendar, fixed)
411 for (fixed, _, name, calendar) in entries
412 ]
413end

417function validate_arguments(command::String, args::Vector{String})
418 if command in ("EASTER", "COMPUTUS")
419 check(valid_year(args[1]) && args[2] in ("GREGORIAN", "JULIAN"))
420 return
421 elseif command in ("PASSOVER", "HOLIDAYS", "MOLAD")
422 check(valid_year(args[1]))
423 return
424 end
425 calendar = args[1]
426 allowed = if command in ("INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN")
427 YMD
428 elseif command in ("LEAP", "NEWYEAR")
429 YEAR_CALENDARS
430 else
431 Set(CALENDARS)
432 end
433 check(calendar in allowed)
434 if command == "CONVERT"
435 check(args[3] == "TO" && args[4] in CALENDARS)
436 elseif command == "DIFF"
437 check(args[3] in CALENDARS)
438 elseif command in ("LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR")
439 check(valid_year(args[2]))
440 if command in ("MONTHCAL", "NTH")
441 check(occursin(r"^[0-9]{1,2}$", args[3]))
442 end
443 if command == "NTH"
444 check(args[4] in ("1", "2", "3", "4", "5", "LAST"))
445 check(args[5] in WEEKDAYS)
446 end
447 elseif command == "COUNT"
448 check(args[4] in WEEKDAYS)
449 elseif command == "ADD"
450 check(occursin(r"^-?[0-9]{1,7}$", args[3]))
451 check(args[4] in ("DAYS", "WEEKS", "MONTHS", "YEARS"))
452 if args[4] == "MONTHS"
453 check(calendar in YMD)
454 elseif args[4] == "YEARS"
455 check(calendar in YEAR_CALENDARS)
456 end
457 elseif command == "ROUND"
458 check(occursin(r"^[0-9]{1,2}$", args[3]))
459 check(1 <= parse(Int, args[3]) <= 13 && args[4] in TZOLKIN)
460 check(occursin(r"^[0-9]{1,2}$", args[5]))
461 check(args[6] in HAAB)
462 check(parse(Int, args[5]) <= (args[6] == "Uayeb" ? 4 : 19))
463 end
464end

466function calendar_lines(calendar::String, year::Int, month::Int)
467 first, last = month_bounds(calendar, year, month)
468 if !in_range(first) || !in_range(last)
469 return ["OUT OF RANGE"]
470 end
471 title = string(names_for(calendar, year)[month], " ", year, " (",
472 format_date("GREGORIAN", first), " to ",
473 format_date("GREGORIAN", last), ")")
474 result = [title, "Mo Tu We Th Fr Sa Su"]
475 cursor = first - mod(first - 1, 7)
476 while cursor <= last
477 fields = [first <= fixed <= last ? @sprintf("%2d", fixed - first + 1) : " "
478 for fixed in cursor:(cursor + 6)]
479 push!(result, rstrip(join(fields, " ")))
480 cursor += 7
481 end
482 result
483end

485function span_text(calendar::String, first::Int, last::Int)
486 negative = first > last
487 if negative
488 first, last = last, first
489 end
490 start_year = parts_of(calendar, first)[1]
491 end_year = parts_of(calendar, last)[1]
492 low = 0
493 high = end_year - start_year + 2
494 while low + 1 < high
495 middle = fld(low + high, 2)
496 shifted = shift_date(calendar, first, middle, 0)
497 if shifted !== nothing && shifted <= last
498 low = middle
499 else
500 high = middle
501 end
502 end
503 years = low
504 low = 0
505 high = 32
506 while low + 1 < high
507 middle = fld(low + high, 2)
508 shifted = shift_date(calendar, first, years, middle)
509 if shifted !== nothing && shifted <= last
510 low = middle
511 else
512 high = middle
513 end
514 end
515 months = low
516 days = last - shift_date(calendar, first, years, months)
517 string(negative ? "-" : "", years, "y ", months, "m ", days, "d")
518end

520function round_date(calendar::String, fixed::Int, number::Int,
521 tz_name::String, day::Int, month::String)
522 target_tz = findfirst(==(tz_name), TZOLKIN) - 1
523 target_haab = (findfirst(==(month), HAAB) - 1) * 20 + day
524 candidate = fixed
525 while true
526 count = candidate + 1137142
527 if mod(count + 3, 13) + 1 == number && mod(count + 19, 20) == target_tz
528 break
529 end
530 candidate += 1
531 end
532 while candidate <= fixed + 18979
533 if mod(candidate + 1137142 + 348, 365) == target_haab
534 return format_date(calendar, candidate)
535 end
536 candidate += 260
537 end
538 "NONE"
539end

541function execute(command::String, args::Vector{String})
542 validate_arguments(command, args)
543 if command in ("EASTER", "COMPUTUS")
544 year = parse(Int, args[1])
545 year > 9999 && return ["OUT OF RANGE"]
546 calendar = args[2]
547 epact, letters, moon, sunday = easter(year, calendar)
548 if command == "EASTER"
549 return [format_date("GREGORIAN", sunday)]
550 end
551 return [
552 "golden=$(mod(year, 19) + 1) epact=$epact letters=$letters " *
553 "moon=$(format_date(calendar, moon)) easter=$(format_date(calendar, sunday))"
554 ]
555 end
556
557 if command in ("PASSOVER", "HOLIDAYS", "MOLAD")
558 year = parse(Int, args[1])
559 if command == "MOLAD"
560 record = molad(year)
561 in_range(record.fixed) || return ["OUT OF RANGE"]
562 applied = isempty(record.rules) ? "NONE" : join(record.rules, "+")
563 return [
564 string(WEEKDAYS[mod(record.day - 1373427, 7) + 1], " ",
565 fld(record.parts, 1080), "h ", mod(record.parts, 1080), "p ",
566 applied, " ", WEEKDAYS[mod(record.fixed, 7) + 1], " ",
567 format_date("GREGORIAN", record.fixed))
568 ]
569 end
570 year > 9999 && return ["OUT OF RANGE"]
571 command == "HOLIDAYS" && return holiday_lines(year)
572 hebrew_year = year + 3760
573 month = findfirst(==("Nisan"), names_for("HEBREW", hebrew_year))
574 fixed = fixed_date("HEBREW", hebrew_year, month, 15)
575 return [format_date("GREGORIAN", fixed) * " " * WEEKDAYS[mod(fixed, 7) + 1]]
576 end

578 calendar = args[1]
579 if command in ("LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR")
580 year = parse(Int, args[2])
581 if command == "LEAP"
582 result = string(is_leap(calendar, year) ? "YES" : "NO", " ",
583 start_fixed(calendar, year + 1) - start_fixed(calendar, year))
584 calendar == "HEBREW" && (result *= " " * year_type(year))
585 return [result]
586 elseif command == "MONTHS"
587 return [join(["$name=$days"
588 for (name, days) in zip(names_for(calendar, year),
589 month_lengths(calendar, year))], " ")]
590 elseif command == "NEWYEAR"
591 year > 9999 && return ["OUT OF RANGE"]
592 first = start_fixed("GREGORIAN", year)
593 last = start_fixed("GREGORIAN", year + 1) - 1
594 entries = new_years(calendar, first, last)
595 return [isempty(entries) ? "NONE" : join(entries, " ")]
596 end
597 month = parse(Int, args[3])
598 command == "MONTHCAL" && return calendar_lines(calendar, year, month)
599 first, last = month_bounds(calendar, year, month)
600 weekday = findfirst(==(args[5]), WEEKDAYS) - 1
601 fixed = if args[4] == "LAST"
602 last - mod(last - weekday, 7)
603 else
604 first + mod(weekday - first, 7) + 7 * (parse(Int, args[4]) - 1)
605 end
606 return [fixed > last ? "NONE" : format_date(calendar, fixed)]
607 end
608
609 if command == "VALID"
610 try
611 fixed = parse_date(calendar, args[2])
612 return ["YES " * format_date(calendar, fixed)]
613 catch e
614 if e isa DateError
615 return ["NO " * e.reason]
616 end
617 rethrow()
618 end
619 end
620
621 first = parse_date(calendar, args[2])
622 if command == "DIFF"
623 last = parse_date(args[3], args[4])
624 return [string(last - first)]
625 elseif command in ("COUNT", "SPAN")
626 last = parse_date(calendar, args[3])
627 command == "SPAN" && return [span_text(calendar, first, last)]
628 low, high = minmax(first, last)
629 occurrence = low + mod(findfirst(==(args[4]), WEEKDAYS) - 1 - low, 7)
630 return [string(occurrence > high ? 0 : fld(high - occurrence, 7) + 1)]
631 elseif command == "CONVERT"
632 return [format_date(args[4], first)]
633 elseif command == "WEEKDAY"
634 return [WEEKDAYS[mod(first, 7) + 1]]
635 elseif command == "ALL"
636 result = String[]
637 for target in CALENDARS
638 text = format_date(target, first)
639 if target in YMD && text != "OUT OF RANGE"
640 year, month, _ = parts_of(target, first)
641 text *= " " * names_for(target, year)[month]
642 end
643 push!(result, target * " " * text)
644 end
645 return result
646 elseif command == "INFO"
647 year, month, day = parts_of(calendar, first)
648 weekday = WEEKDAYS[mod(first, 7) + 1]
649 if calendar == "FRENCH"
650 weekday = month <= 12 ? DECADES[mod(day - 1, 10) + 1] : FESTIVALS[day]
651 end
652 ordinal = first - start_fixed(calendar, year) + 1
653 length = start_fixed(calendar, year + 1) - start_fixed(calendar, year)
654 return [string(weekday, " ", day, " ", names_for(calendar, year)[month],
655 " ", year, " day=", ordinal, "/", length)]
656 elseif command == "ROUND"
657 return [round_date(calendar, first, parse(Int, args[3]), args[4],
658 parse(Int, args[5]), args[6])]
659 end
660
661 amount = parse(Int, args[3])
662 unit = args[4]
663 result = if unit == "DAYS"
664 first + amount
665 elseif unit == "WEEKS"
666 first + 7 * amount
667 elseif unit == "YEARS"
668 shift_date(calendar, first, amount, 0)
669 else
670 shift_date(calendar, first, 0, amount)
671 end
672 [result === nothing ? "OUT OF RANGE" : format_date(calendar, result)]
673end

675function main()
676 count = parse(Int, readline(stdin))
677 output = String[]
678 for _ in 1:count
679 fields = String.(split(readline(stdin), ' '; keepempty=true))
680 command = first(fields)
681 arguments = fields[2:end]
682 if !haskey(ARITY, command) || length(arguments) != ARITY[command]
683 push!(output, "INVALID COMMAND")
684 continue
685 end
686 try
687 append!(output, execute(command, arguments))
688 catch error
689 if error isa CommandError
690 push!(output, "INVALID COMMAND")
691 elseif error isa DateError
692 push!(output, error.reason == "RANGE" ? "OUT OF RANGE" : "INVALID DATE")
693 else
694 rethrow()
695 end
696 end
697 end
698 println(join(output, "\n"))
699end
