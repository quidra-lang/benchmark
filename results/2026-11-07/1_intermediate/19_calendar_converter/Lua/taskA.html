63local function leap(calendar, year)
64 if calendar == "GREGORIAN" then
65 return year % 4 == 0 and (year % 100 ~= 0 or year % 400 == 0)
66 elseif calendar == "JULIAN" then
67 return year % 4 == 0
68 elseif calendar == "ISLAMIC" then
69 return contains({2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29}, year % 30)
70 elseif calendar == "HEBREW" then
71 return (7 * year + 1) % 19 < 7
72 elseif calendar == "PERSIAN" then
73 local b = (year - 474) % 2820 + 474
74 return (b + 38) * 682 % 2816 < 682
75 elseif calendar == "COPTIC" or calendar == "ETHIOPIC" then
76 return year % 4 == 3
77 elseif calendar == "FRENCH" then
78 return year % 4 == 0 and not contains({100, 200, 300}, year % 400) and year % 4000 ~= 0
79 end
80 return (start("ISO", year + 1) - start("ISO", year)) // 7 == 53
81end

87local function molad(year)
88 local quantity = 5604 + 765433 * months_before(year)
89 local day, parts = quantity // 25920, quantity % 25920
90 local weekday = (day - 1373427) % 7
91 local rules = {}
92 local postponed = day
93 if parts >= 19440 then
94 postponed = postponed + 1
95 rules[#rules + 1] = "ZAKEN"
96 elseif not leap("HEBREW", year) and weekday == 2 and parts >= 9924 then
97 postponed = postponed + 2
98 rules[#rules + 1] = "GATARAD"
99 elseif leap("HEBREW", year - 1) and weekday == 1 and parts >= 16789 then
100 postponed = postponed + 1
101 rules[#rules + 1] = "BETUTAKPAT"
102 end
103 if contains({0, 3, 5}, (postponed - 1373427) % 7) then
104 postponed = postponed + 1
105 rules[#rules + 1] = "ADU"
106 end
107 return day, parts, postponed - 1373427, rules
108end

110function start(calendar, year)
111 local previous = year - 1
112 if calendar == "GREGORIAN" then
113 return 365 * previous + previous // 4 - previous // 100 + previous // 400 + 1
114 elseif calendar == "JULIAN" then
115 return 365 * previous + previous // 4 - 1
116 elseif calendar == "ISLAMIC" then
117 return 227015 + 354 * previous + (3 + 11 * year) // 30
118 elseif calendar == "HEBREW" then
119 local _, _, fixed = molad(year)
120 return fixed
121 elseif calendar == "PERSIAN" then
122 local a = year - 474
123 local b = a % 2820 + 474
124 return 226896 + 1029983 * (a // 2820) + 365 * (b - 1) + (682 * b - 110) // 2816
125 elseif calendar == "COPTIC" then
126 return 103605 + 365 * previous + year // 4
127 elseif calendar == "ETHIOPIC" then
128 return 2796 + 365 * previous + year // 4
129 elseif calendar == "FRENCH" then
130 return 654415 + 365 * previous + previous // 4 - previous // 100 + previous // 400 - previous // 4000
131 end
132 local january_four = start("GREGORIAN", year) + 3
133 return january_four - (january_four - 1) % 7
134end

173local function fixed_date(calendar, year, month, day)
174 if calendar == "ISO" then
175 return start(calendar, year) + 7 * (month - 1) + day - 1
176 end
177 local days = lengths(calendar, year)
178 local before = 0
179 for i = 1, month - 1 do
180 before = before + days[i]
181 end
182 return start(calendar, year) + before + day - 1
183end

185local function year_of(calendar, fixed)
186 if fixed < start(calendar, 1) then
187 return 0
188 end
189 local low, high = 1, 100000
190 while low + 1 < high do
191 local middle = (low + high) // 2
192 if start(calendar, middle) <= fixed then
193 low = middle
194 else
195 high = middle
196 end
197 end
198 return low
199end

201local function parts_of(calendar, fixed)
202 local year = year_of(calendar, fixed)
203 if year == 0 then
204 return nil
205 end
206 local offset = fixed - start(calendar, year)
207 if calendar == "ISO" then
208 return year, offset // 7 + 1, offset % 7 + 1
209 end
210 for month, days in ipairs(lengths(calendar, year)) do
211 if offset < days then
212 return year, month, offset + 1
213 end
214 offset = offset - days
215 end
216 error("calendar year length")
217end

219local function format_date(calendar, fixed)
220 if not in_range(fixed) then
221 return "OUT OF RANGE"
222 end
223 if contains(YEAR_CALENDARS, calendar) then
224 local year, month, day = parts_of(calendar, fixed)
225 if year == nil then
226 return "OUT OF RANGE"
227 end
228 if calendar == "ISO" then
229 return string.format("%04d-W%02d-%d", year, month, day)
230 end
231 return string.format("%04d-%02d-%02d", year, month, day)
232 elseif calendar == "JD" then
233 return tostring(fixed + 1721425)
234 elseif calendar == "FIXED" then
235 return tostring(fixed)
236 end
237 local count = fixed + 1137142
238 local long_count = {count // 144000, count // 7200 % 20, count // 360 % 20, count // 20 % 18, count % 20}
239 local tz_number = (count + 3) % 13 + 1
240 local tz_name = TZ[(count + 19) % 20 + 1]
241 local haab = (count + 348) % 365
242 return string.format("%s %d %s %d %s", table.concat(long_count, "."), tz_number, tz_name, haab % 20, HB[haab // 20 + 1])
243end

245local function parse_date(calendar, text)
246 local fixed
247 if contains(YMD, calendar) then
248 local year_text, month_text, day_text = text:match("^(%d+)%-(%d+)%-(%d+)$")
249 if not (year_text and #year_text <= 5 and #month_text <= 2 and #day_text <= 2) then
250 date_error("FORMAT")
251 end
252 local year, month, day = tonumber(year_text), tonumber(month_text), tonumber(day_text)
253 if year == 0 then
254 date_error("YEAR")
255 end
256 local days = lengths(calendar, year)
257 if month < 1 or month > #days then
258 date_error("MONTH")
259 end
260 if day < 1 or day > days[month] then
261 date_error("DAY")
262 end
263 fixed = fixed_date(calendar, year, month, day)
264 elseif calendar == "ISO" then
265 local year_text, week_text, day_text = text:match("^(%d+)%-W(%d+)%-(%d)$")
266 if not (year_text and #year_text <= 5 and #week_text <= 2) then
267 date_error("FORMAT")
268 end
269 local year, week, day = tonumber(year_text), tonumber(week_text), tonumber(day_text)
270 if year == 0 then
271 date_error("YEAR")
272 end
273 local weeks = (start(calendar, year + 1) - start(calendar, year)) // 7
274 if week < 1 or week > weeks then
275 date_error("WEEK")
276 end
277 if day < 1 or day > 7 then
278 date_error("DAY")
279 end
280 fixed = fixed_date(calendar, year, week, day)
281 elseif calendar == "MAYAN" then
282 local fields = {text:match("^(%d+)%.(%d+)%.(%d+)%.(%d+)%.(%d+)$")}
283 if #fields ~= 5 then
284 date_error("FORMAT")
285 end
286 for i, field in ipairs(fields) do
287 if #field > 2 then
288 date_error("FORMAT")
289 end
290 fields[i] = tonumber(field)
291 end
292 local b, k, t, u, n = table.unpack(fields)
293 if k > 19 or t > 19 or u > 17 or n > 19 then
294 date_error("FIELD")
295 end
296 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142
297 else
298 local digits = text:match("^%-?(%d+)$")
299 if not (digits and #digits <= 8) then
300 date_error("FORMAT")
301 end
302 fixed = tonumber(text) - (calendar == "JD" and 1721425 or 0)
303 end
304 if not in_range(fixed) then
305 date_error("RANGE")
306 end
307 return fixed
308end

318local function shift(calendar, fixed, years, months)
319 local year, month, day = parts_of(calendar, fixed)
320 local target_year = year + years
321 if calendar == "ISO" then
322 if target_year < 1 or target_year > 99999 then
323 return nil
324 end
325 local weeks = (start(calendar, target_year + 1) - start(calendar, target_year)) // 7
326 return fixed_date(calendar, target_year, math.min(month, weeks), day)
327 end
328 if calendar == "HEBREW" then
329 local old_name = names(calendar, year)[month]
330 local new_names = names(calendar, target_year)
331 if not contains(new_names, old_name) then
332 old_name = old_name == "Adar" and "AdarII" or "Adar"
333 end
334 month = index_of(new_names, old_name)
335 end
336 local absolute = month_index(calendar, target_year) + month - 1 + months
337 local final_year
338 if calendar == "HEBREW" then
339 final_year = (19 * absolute + 234) // 235
340 while month_index(calendar, final_year) > absolute do
341 final_year = final_year - 1
342 end
343 while month_index(calendar, final_year + 1) <= absolute do
344 final_year = final_year + 1
345 end
346 else
347 final_year = absolute // #names(calendar, target_year) + 1
348 end
349 if final_year < 1 or final_year > 99999 then
350 return nil
351 end
352 local final_month = absolute - month_index(calendar, final_year) + 1
353 local final_day = math.min(day, lengths(calendar, final_year)[final_month])
354 return fixed_date(calendar, final_year, final_month, final_day)
355end

379local function easter(year, calendar)
380 local golden = year % 19
381 local epact
382 if calendar == "GREGORIAN" then
383 local century = year // 100 + 1
384 epact = (14 + 11 * golden - 3 * century // 4 + (5 + 8 * century) // 25) % 30
385 if epact == 0 or (epact == 1 and golden > 10) then
386 epact = epact + 1
387 end
388 else
389 epact = (14 + 11 * golden) % 30
390 end
391 local moon = fixed_date(calendar, year, 4, 19) - epact
392 local sunday = moon + 7 - moon % 7
393 local index = -start(calendar, year) % 7
394 local letters = ("ABCDEFG"):sub(index + 1, index + 1)
395 if leap(calendar, year) then
396 local second = (index - 1) % 7
397 letters = letters .. ("ABCDEFG"):sub(second + 1, second + 1)
398 end
399 return epact, letters, moon, sunday
400end

430local function holiday_lines(year)
431 local first = start("GREGORIAN", year)
432 local last = start("GREGORIAN", year + 1) - 1
433 local gregorian_easter = select(4, easter(year, "GREGORIAN"))
434 local orthodox_easter = select(4, easter(year, "JULIAN"))
435 local entries = {}
436 local movable = {{"AshWednesday", -46}, {"GoodFriday", -2}, {"Easter", 0}, {"Ascension", 39}, {"Pentecost", 49}}
437 for order, item in ipairs(movable) do
438 entries[#entries + 1] = {fixed = gregorian_easter + item[2], order = order, name = item[1], calendar = "GREGORIAN"}
439 end
440 entries[#entries + 1] = {fixed = orthodox_easter, order = 6, name = "OrthodoxEaster", calendar = "JULIAN"}
441 for index, holiday in ipairs(HOLIDAYS) do
442 local name, calendar, month_name, day = table.unpack(holiday)
443 local candidate_year = math.max(1, year_of(calendar, first))
444 while start(calendar, candidate_year) <= last do
445 local actual_name = month_name
446 if calendar == "HEBREW" and month_name == "Adar" and leap(calendar, candidate_year) then
447 actual_name = "AdarII"
448 end
449 local month = index_of(names(calendar, candidate_year), actual_name)
450 if day <= lengths(calendar, candidate_year)[month] then
451 local fixed = fixed_date(calendar, candidate_year, month, day)
452 if first <= fixed and fixed <= last then
453 entries[#entries + 1] = {fixed = fixed, order = index + 6, name = name, calendar = calendar}
454 end
455 end
456 candidate_year = candidate_year + 1
457 end
458 end
459 table.sort(entries, function(a, b)
460 if a.fixed ~= b.fixed then
461 return a.fixed < b.fixed
462 end
463 return a.order < b.order
464 end)
465 local lines = {}
466 for i, entry in ipairs(entries) do
467 lines[i] = string.format("%s %s %s %s %s", format_date("GREGORIAN", entry.fixed), WEEKDAYS[entry.fixed % 7 + 1],
468 entry.name, entry.calendar, format_date(entry.calendar, entry.fixed))
469 end
470 return lines
471end

488local function validate_arguments(command, args)
489 if command == "EASTER" or command == "COMPUTUS" then
490 require_valid(valid_year(args[1]) and (args[2] == "GREGORIAN" or args[2] == "JULIAN"))
491 return
492 end
493 if contains({"PASSOVER", "HOLIDAYS", "MOLAD"}, command) then
494 require_valid(valid_year(args[1]))
495 return
496 end
497 local calendar = args[1]
498 local allowed = CALENDARS
499 if contains({"INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN"}, command) then
500 allowed = YMD
501 elseif command == "LEAP" or command == "NEWYEAR" then
502 allowed = YEAR_CALENDARS
503 end
504 require_valid(contains(allowed, calendar))
505 if command == "CONVERT" then
506 require_valid(args[3] == "TO" and contains(CALENDARS, args[4]))
507 elseif command == "DIFF" then
508 require_valid(contains(CALENDARS, args[3]))
509 elseif contains({"LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR"}, command) then
510 require_valid(valid_year(args[2]))
511 if command == "MONTHCAL" or command == "NTH" then
512 require_valid(digit_string(args[3], 2))
513 end
514 if command == "NTH" then
515 require_valid(contains({"1", "2", "3", "4", "5", "LAST"}, args[4]))
516 require_valid(contains(WEEKDAYS, args[5]))
517 end
518 elseif command == "COUNT" then
519 require_valid(contains(WEEKDAYS, args[4]))
520 elseif command == "ADD" then
521 local digits = args[3]:match("^%-?(%d+)$")
522 require_valid(digits ~= nil and #digits <= 7)
523 require_valid(contains({"DAYS", "WEEKS", "MONTHS", "YEARS"}, args[4]))
524 if args[4] == "MONTHS" then
525 require_valid(contains(YMD, calendar))
526 end
527 if args[4] == "YEARS" then
528 require_valid(contains(YEAR_CALENDARS, calendar))
529 end
530 elseif command == "ROUND" then
531 require_valid(digit_string(args[3], 2))
532 require_valid(tonumber(args[3]) >= 1 and tonumber(args[3]) <= 13 and contains(TZ, args[4]))
533 require_valid(digit_string(args[5], 2))
534 require_valid(contains(HB, args[6]))
535 require_valid(tonumber(args[5]) <= (args[6] == "Uayeb" and 4 or 19))
536 end

539local function calendar_lines(calendar, year, month)
540 local first, last = month_bounds(calendar, year, month)
541 if not in_range(first) or not in_range(last) then
542 return {"OUT OF RANGE"}
543 end
544 local title = string.format("%s %d (%s to %s)", names(calendar, year)[month], year,
545 format_date("GREGORIAN", first), format_date("GREGORIAN", last))
546 local result = {title, "Mo Tu We Th Fr Sa Su"}
547 for cursor = first - (first - 1) % 7, last, 7 do
548 local fields = {}
549 for fixed = cursor, cursor + 6 do
550 if first <= fixed and fixed <= last then
551 fields[#fields + 1] = string.format("%2d", fixed - first + 1)
552 else
553 fields[#fields + 1] = " "
554 end
555 end
556 result[#result + 1] = (table.concat(fields, " "):gsub(" +$", ""))
557 end
558 return result

561local function span_text(calendar, first, last)
562 local negative = first > last
563 if negative then
564 first, last = last, first
565 end
566 local start_year = parts_of(calendar, first)
567 local end_year = parts_of(calendar, last)
568 local low, high = 0, end_year - start_year + 2
569 while low + 1 < high do
570 local middle = (low + high) // 2
571 local shifted = shift(calendar, first, middle, 0)
572 if shifted ~= nil and shifted <= last then
573 low = middle
574 else
575 high = middle
576 end
577 end
578 local years = low
579 low, high = 0, 32
580 while low + 1 < high do
581 local middle = (low + high) // 2
582 local shifted = shift(calendar, first, years, middle)
583 if shifted ~= nil and shifted <= last then
584 low = middle
585 else
586 high = middle
587 end
588 end
589 local months = low
590 local days = last - shift(calendar, first, years, months)
591 return string.format("%s%dy %dm %dd", negative and "-" or "", years, months, days)
592end

594local function round_date(calendar, fixed, number, tz_name, day, month)
595 local target_tz = index_of(TZ, tz_name) - 1
596 local target_haab = (index_of(HB, month) - 1) * 20 + day
597 local candidate = fixed
598 while true do
599 local count = candidate + 1137142
600 if (count + 3) % 13 + 1 == number and (count + 19) % 20 == target_tz then
601 break
602 end
603 candidate = candidate + 1
604 end
605 while candidate <= fixed + 18979 do
606 if (candidate + 1137142 + 348) % 365 == target_haab then
607 return format_date(calendar, candidate)
608 end
609 candidate = candidate + 260
610 end
611 return "NONE"
612end

614local function execute(command, args)
615 validate_arguments(command, args)
616 if command == "EASTER" or command == "COMPUTUS" then
617 local year = tonumber(args[1])
618 if year > 9999 then
619 return {"OUT OF RANGE"}
620 end
621 local kind = args[2]
622 local epact, letters, moon, sunday = easter(year, kind)
623 if command == "EASTER" then
624 return {format_date("GREGORIAN", sunday)}
625 end
626 return {string.format("golden=%d epact=%d letters=%s moon=%s easter=%s", year % 19 + 1, epact, letters,
627 format_date(kind, moon), format_date(kind, sunday))}
628 end
629 if contains({"PASSOVER", "HOLIDAYS", "MOLAD"}, command) then
630 local year = tonumber(args[1])
631 if command == "MOLAD" then
632 local day, parts, fixed, rules = molad(year)
633 if not in_range(fixed) then
634 return {"OUT OF RANGE"}
635 end
636 local applied = #rules > 0 and table.concat(rules, "+") or "NONE"
637 return {string.format("%s %dh %dp %s %s %s", WEEKDAYS[(day - 1373427) % 7 + 1], parts // 1080, parts % 1080,
638 applied, WEEKDAYS[fixed % 7 + 1], format_date("GREGORIAN", fixed))}
639 end
640 if year > 9999 then
641 return {"OUT OF RANGE"}
642 end
643 if command == "HOLIDAYS" then
644 return holiday_lines(year)
645 end
646 local hebrew_year = year + 3760
647 local month = index_of(names("HEBREW", hebrew_year), "Nisan")
648 local fixed = fixed_date("HEBREW", hebrew_year, month, 15)
649 return {format_date("GREGORIAN", fixed) .. " " .. WEEKDAYS[fixed % 7 + 1]}
650 end
651 local calendar = args[1]
652 if contains({"LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR"}, command) then
653 local year = tonumber(args[2])
654 if command == "LEAP" then
655 local text = (leap(calendar, year) and "YES" or "NO") .. " " .. (start(calendar, year + 1) - start(calendar, year))
656 if calendar == "HEBREW" then
657 text = text .. " " .. year_type(year)
658 end
659 return {text}
660 elseif command == "MONTHS" then
661 local days = lengths(calendar, year)
662 local items = {}
663 for i, name in ipairs(names(calendar, year)) do
664 items[i] = name .. "=" .. days[i]
665 end
666 return {table.concat(items, " ")}
667 elseif command == "NEWYEAR" then
668 if year > 9999 then
669 return {"OUT OF RANGE"}
670 end
671 local items = {}
672 for i, entry in ipairs(new_years(calendar, start("GREGORIAN", year), start("GREGORIAN", year + 1) - 1)) do
673 items[i] = entry[1] .. ":" .. format_date("GREGORIAN", entry[2])
674 end
675 return {#items > 0 and table.concat(items, " ") or "NONE"}
676 end
677 local month = tonumber(args[3])
678 if command == "MONTHCAL" then
679 return calendar_lines(calendar, year, month)
680 end
681 local first, last = month_bounds(calendar, year, month)
682 local weekday = index_of(WEEKDAYS, args[5]) - 1
683 local fixed
684 if args[4] == "LAST" then
685 fixed = last - (last - weekday) % 7
686 else
687 fixed = first + (weekday - first) % 7 + 7 * (tonumber(args[4]) - 1)
688 end
689 return {fixed > last and "NONE" or format_date(calendar, fixed)}
690 end
691 if command == "VALID" then
692 local ok, result = pcall(parse_date, calendar, args[2])
693 if ok then
694 return {"YES " .. format_date(calendar, result)}
695 elseif is_error(result, "date") then
696 return {"NO " .. result.reason}
697 end
698 error(result, 0)
699 end
700 local first = parse_date(calendar, args[2])
701 if command == "DIFF" then
702 local last = parse_date(args[3], args[4])
703 return {tostring(last - first)}
704 end
705 if command == "COUNT" or command == "SPAN" then
706 local last = parse_date(calendar, args[3])
707 if command == "SPAN" then
708 return {span_text(calendar, first, last)}
709 end
710 local low, high = math.min(first, last), math.max(first, last)
711 local occurrence = low + (index_of(WEEKDAYS, args[4]) - 1 - low) % 7
712 return {tostring(occurrence > high and 0 or (high - occurrence) // 7 + 1)}
713 end
714 if command == "CONVERT" then
715 return {format_date(args[4], first)}
716 elseif command == "WEEKDAY" then
717 return {WEEKDAYS[first % 7 + 1]}
718 elseif command == "ALL" then
719 local result = {}
720 for i, target in ipairs(CALENDARS) do
721 local text = format_date(target, first)
722 if contains(YMD, target) and text ~= "OUT OF RANGE" then
723 local year, month = parts_of(target, first)
724 text = text .. " " .. names(target, year)[month]
725 end
726 result[i] = target .. " " .. text
727 end
728 return result
729 elseif command == "INFO" then
730 local year, month, day = parts_of(calendar, first)
731 local weekday = WEEKDAYS[first % 7 + 1]
732 if calendar == "FRENCH" then
733 weekday = month <= 12 and DC[(day - 1) % 10 + 1] or FS[day]
734 end
735 local ordinal = first - start(calendar, year) + 1
736 local length = start(calendar, year + 1) - start(calendar, year)
737 return {string.format("%s %d %s %d day=%d/%d", weekday, day, names(calendar, year)[month], year, ordinal, length)}
738 elseif command == "ROUND" then
739 return {round_date(calendar, first, tonumber(args[3]), args[4], tonumber(args[5]), args[6])}
740 end
741 local amount = tonumber(args[3])
742 local unit = args[4]
743 local result
744 if unit == "DAYS" then
745 result = first + amount
746 elseif unit == "WEEKS" then
747 result = first + 7 * amount
748 elseif unit == "YEARS" then
749 result = shift(calendar, first, amount, 0)
750 else
751 result = shift(calendar, first, 0, amount)
752 end
753 return {result == nil and "OUT OF RANGE" or format_date(calendar, result)}
754end
755
756local count = tonumber(io.read("l"))
757local output = {}
758for _ = 1, count do
759 local args = words(io.read("l"))
760 local command = table.remove(args, 1)
761 if ARITY[command] == nil or #args ~= ARITY[command] then
762 output[#output + 1] = "INVALID COMMAND"
763 else
764 local ok, result = pcall(execute, command, args)
765 if ok then
766 table.move(result, 1, #result, #output + 1, output)
767 elseif is_error(result, "command") then
768 output[#output + 1] = "INVALID COMMAND"
769 elseif is_error(result, "date") then
770 output[#output + 1] = result.reason == "RANGE" and "OUT OF RANGE" or "INVALID DATE"
771 else
772 error(result, 0)
773 end
774 end
775end
776io.write(table.concat(output, "\n"), "\n")
