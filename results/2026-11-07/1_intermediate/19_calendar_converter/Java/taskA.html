85 static boolean leap(Calendar calendar, long year) {
86 return switch (calendar) {
87 case GREGORIAN -> year % 4 == 0 && (year % 100 != 0 || year % 400 == 0);
88 case JULIAN -> year % 4 == 0;
89 case ISLAMIC -> Set.of(2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29).contains(Math.floorMod(year, 30));
90 case HEBREW -> Math.floorMod(7 * year + 1, 19) < 7;
91 case PERSIAN -> (Math.floorMod(year - 474, 2820) + 474 + 38) * 682 % 2816 < 682;
92 case COPTIC, ETHIOPIC -> Math.floorMod(year, 4) == 3;
93 case FRENCH -> year % 4 == 0 && !Set.of(100, 200, 300).contains(Math.floorMod(year, 400)) && year % 4000 != 0;
94 case ISO -> weeks(year) == 53;
95 default -> throw new IllegalArgumentException("no years in " + calendar);
96 };
97 }

104 static Molad molad(long year) {

122 postponed += 1;
123 rules.add("ADU");

127 static long start(Calendar calendar, long year) {
128 long previous = year - 1;
129 return switch (calendar) {
130 case GREGORIAN -> 365 * previous + previous / 4 - previous / 100 + previous / 400 + 1;
131 case JULIAN -> 365 * previous + previous / 4 - 1;
132 case ISLAMIC -> 227015 + 354 * previous + (3 + 11 * year) / 30;
133 case HEBREW -> molad(year).start();
134 case PERSIAN -> {
135 long a = year - 474;
136 long b = Math.floorMod(a, 2820) + 474;
137 yield 226896 + 1029983 * Math.floorDiv(a, 2820) + 365 * (b - 1) + (682 * b - 110) / 2816;
138 }
139 case COPTIC -> 103605 + 365 * previous + year / 4;
140 case ETHIOPIC -> 2796 + 365 * previous + year / 4;
141 case FRENCH -> 654415 + 365 * previous + previous / 4 - previous / 100 + previous / 400 - previous / 4000;
142 case ISO -> {
143 long januaryFour = start(Calendar.GREGORIAN, year) + 3;
144 yield januaryFour - (januaryFour - 1) % 7;
145 }
146 default -> throw new IllegalArgumentException("no years in " + calendar);
147 };

176 static long fixedDate(Calendar calendar, long year, int month, int day) {
177 if (calendar == Calendar.ISO) return start(calendar, year) + 7 * (month - 1) + day - 1;
178 return start(calendar, year) + Arrays.stream(lengths(calendar, year), 0, month - 1).sum() + day - 1;

180 static long yearOf(Calendar calendar, long fixed) {
181 if (fixed < start(calendar, 1)) return 0;
182 long low = 1, high = 100000;
183 while (low + 1 < high) {
184 long middle = (low + high) / 2;
185 if (start(calendar, middle) <= fixed) low = middle;
186 else high = middle;
187 }
188 return low;
189 }
190 static Parts partsOf(Calendar calendar, long fixed) {
191 long year = yearOf(calendar, fixed);
192 if (year == 0) return null;
193 int offset = (int) (fixed - start(calendar, year));
194 if (calendar == Calendar.ISO) return new Parts(year, offset / 7 + 1, offset % 7 + 1);
195 int[] days = lengths(calendar, year);
196 int month = 0;
197 while (offset >= days[month]) {
198 offset -= days[month];
199 month++;
200 }
201 return new Parts(year, month + 1, offset + 1);
202 }
203 static String formatDate(Calendar calendar, long fixed) {
204 if (!inRange(fixed)) return "OUT OF RANGE";
205 if (YEAR_CALENDARS.contains(calendar)) {
206 Parts parts = partsOf(calendar, fixed);
207 if (parts == null) return "OUT OF RANGE";
208 String pattern = calendar == Calendar.ISO ? "%04d-W%02d-%d" : "%04d-%02d-%02d";
209 return String.format(Locale.ROOT, pattern, parts.year(), parts.month(), parts.day());
210 }
211 if (calendar == Calendar.JD) return Long.toString(fixed + 1721425);
212 if (calendar == Calendar.FIXED) return Long.toString(fixed);
213 long count = fixed + 1137142;
214 String longCount = LongStream.of(count / 144000, count / 7200 % 20, count / 360 % 20, count / 20 % 18, count % 20)
215 .mapToObj(Long::toString).collect(Collectors.joining("."));
216 long haab = (count + 348) % 365;
217 return longCount + " " + ((count + 3) % 13 + 1) + " " + TZ.get((int) ((count + 19) % 20)) + " " +
218 haab % 20 + " " + HB.get((int) (haab / 20));
219 }
220 static long parseDate(Calendar calendar, String text) {
221 long fixed;
222 if (YMD.contains(calendar)) {
223 if (!text.matches("[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}")) throw new DateError("FORMAT");
224 String[] fields = text.split("-");
225 long year = Long.parseLong(fields[0]);
226 int month = Integer.parseInt(fields[1]), day = Integer.parseInt(fields[2]);
227 if (year == 0) throw new DateError("YEAR");
228 int[] days = lengths(calendar, year);
229 if (month < 1 || month > days.length) throw new DateError("MONTH");
230 if (day < 1 || day > days[month - 1]) throw new DateError("DAY");
231 fixed = fixedDate(calendar, year, month, day);
232 } else if (calendar == Calendar.ISO) {
233 if (!text.matches("[0-9]{1,5}-W[0-9]{1,2}-[0-9]")) throw new DateError("FORMAT");
234 String[] fields = text.split("-");
235 long year = Long.parseLong(fields[0]);
236 int week = Integer.parseInt(fields[1].substring(1)), day = Integer.parseInt(fields[2]);
237 if (year == 0) throw new DateError("YEAR");
238 if (week < 1 || week > weeks(year)) throw new DateError("WEEK");
239 if (day < 1 || day > 7) throw new DateError("DAY");
240 fixed = fixedDate(calendar, year, week, day);
241 } else if (calendar == Calendar.MAYAN) {
242 if (!text.matches("[0-9]{1,2}(?:\\.[0-9]{1,2}){4}")) throw new DateError("FORMAT");
243 int[] v = Arrays.stream(text.split("\\.")).mapToInt(Integer::parseInt).toArray();
244 if (v[1] > 19 || v[2] > 19 || v[3] > 17 || v[4] > 19) throw new DateError("FIELD");
245 fixed = (((v[0] * 20L + v[1]) * 20 + v[2]) * 18 + v[3]) * 20 + v[4] - 1137142;
246 } else {
247 if (!text.matches("-?[0-9]{1,8}")) throw new DateError("FORMAT");
248 fixed = Long.parseLong(text) - (calendar == Calendar.JD ? 1721425 : 0);
249 }
250 if (!inRange(fixed)) throw new DateError("RANGE");
251 return fixed;
252 }

257 static OptionalLong shift(Calendar calendar, long fixed, long years, long months) {
258 Parts parts = partsOf(calendar, fixed);
259 long targetYear = parts.year() + years;
260 if (calendar == Calendar.ISO) {
261 if (targetYear < 1 || targetYear > 99999) return OptionalLong.empty();
262 return OptionalLong.of(fixedDate(calendar, targetYear, Math.min(parts.month(), weeks(targetYear)), parts.day()));
263 }
264 int month = parts.month();
265 if (calendar == Calendar.HEBREW) {
266 String oldName = names(calendar, parts.year()).get(month - 1);
267 List<String> newNames = names(calendar, targetYear);
268 if (!newNames.contains(oldName)) oldName = oldName.equals("Adar") ? "AdarII" : "Adar";
269 month = newNames.indexOf(oldName) + 1;
270 }
271 long absolute = monthIndex(calendar, targetYear) + month - 1 + months;
272 long finalYear;
273 if (calendar == Calendar.HEBREW) {
274 finalYear = Math.floorDiv(19 * absolute + 234, 235);
275 while (monthIndex(calendar, finalYear) > absolute) finalYear--;
276 while (monthIndex(calendar, finalYear + 1) <= absolute) finalYear++;
277 } else {
278 finalYear = Math.floorDiv(absolute, MONTH_NAMES.get(calendar).size()) + 1;
279 }
280 if (finalYear < 1 || finalYear > 99999) return OptionalLong.empty();
281 int finalMonth = (int) (absolute - monthIndex(calendar, finalYear) + 1);
282 int finalDay = Math.min(parts.day(), lengths(calendar, finalYear)[finalMonth - 1]);
283 return OptionalLong.of(fixedDate(calendar, finalYear, finalMonth, finalDay));
284 }

289 return new Period(first, first + days[month - 1] - 1);

298 static Easter easter(int year, Calendar calendar) {
299 int golden = year % 19;
300 int epact;
301 if (calendar == Calendar.GREGORIAN) {
302 int century = year / 100 + 1;
303 epact = Math.floorMod(14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25, 30);
304 if (epact == 0 || epact == 1 && golden > 10) epact++;
305 } else {
306 epact = (14 + 11 * golden) % 30;
307 }
308 long moon = fixedDate(calendar, year, 4, 19) - epact;
309 long sunday = moon + 7 - moon % 7;
310 int index = Math.floorMod(-start(calendar, year), 7);
311 String letters = String.valueOf("ABCDEFG".charAt(index));
312 if (leap(calendar, year)) letters += "ABCDEFG".charAt(Math.floorMod(index - 1, 7));
313 return new Easter(epact, letters, moon, sunday);
314 }

349 static void validateArguments(String command, String[] args) {
350 if (command.equals("EASTER") || command.equals("COMPUTUS")) {
351 require(validYear(args[0]) && (args[1].equals("GREGORIAN") || args[1].equals("JULIAN")));
352 return;
353 }
354 if (List.of("PASSOVER", "HOLIDAYS", "MOLAD").contains(command)) {
355 require(validYear(args[0]));
356 return;
357 }
358 Set<Calendar> allowed = CALENDARS;
359 if (List.of("INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN").contains(command)) allowed = YMD;
360 else if (command.equals("LEAP") || command.equals("NEWYEAR")) allowed = YEAR_CALENDARS;
361 require(isCalendar(args[0], allowed));
362 switch (command) {
363 case "CONVERT" -> require(args[2].equals("TO") && isCalendar(args[3], CALENDARS));
364 case "DIFF" -> require(isCalendar(args[2], CALENDARS));
365 case "LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR" -> {
366 require(validYear(args[1]));
367 if (command.equals("MONTHCAL") || command.equals("NTH")) require(args[2].matches("[0-9]{1,2}"));
368 if (command.equals("NTH")) {
369 require(List.of("1", "2", "3", "4", "5", "LAST").contains(args[3]));
370 require(WEEKDAYS.contains(args[4]));
371 }
372 }
373 case "COUNT" -> require(WEEKDAYS.contains(args[3]));
374 case "ADD" -> {
375 require(args[2].matches("-?[0-9]{1,7}"));
376 require(List.of("DAYS", "WEEKS", "MONTHS", "YEARS").contains(args[3]));
377 Calendar calendar = Calendar.valueOf(args[0]);
378 if (args[3].equals("MONTHS")) require(YMD.contains(calendar));
379 if (args[3].equals("YEARS")) require(YEAR_CALENDARS.contains(calendar));
380 }
381 case "ROUND" -> {
382 require(args[2].matches("[0-9]{1,2}"));
383 int number = Integer.parseInt(args[2]);
384 require(number >= 1 && number <= 13 && TZ.contains(args[3]));
385 require(args[4].matches("[0-9]{1,2}"));
386 require(HB.contains(args[5]));
387 require(Integer.parseInt(args[4]) <= (args[5].equals("Uayeb") ? 4 : 19));
388 }
389 default -> {}
390 }
391 }
392 static List<String> calendarLines(Calendar calendar, long year, int month) {
393 Period bounds = monthBounds(calendar, year, month);
394 long first = bounds.first(), last = bounds.last();
395 if (!inRange(first) || !inRange(last)) return List.of("OUT OF RANGE");
396 List<String> result = new ArrayList<>();
397 result.add(names(calendar, year).get(month - 1) + " " + year + " (" + formatDate(Calendar.GREGORIAN, first) + " to " +
398 formatDate(Calendar.GREGORIAN, last) + ")");
399 result.add("Mo Tu We Th Fr Sa Su");
400 for (long cursor = first - (first - 1) % 7; cursor <= last; cursor += 7) {
401 result.add(LongStream.range(cursor, cursor + 7)
402 .mapToObj(fixed -> first <= fixed && fixed <= last ? String.format(Locale.ROOT, "%2d", fixed - first + 1) : " ")
403 .collect(Collectors.joining(" ")).stripTrailing());
404 }
405 return result;
406 }
407 static String spanText(Calendar calendar, long first, long last) {
408 long from = Math.min(first, last), to = Math.max(first, last);
409 long low = 0, high = partsOf(calendar, to).year() - partsOf(calendar, from).year() + 2;
410 while (low + 1 < high) {
411 long middle = (low + high) / 2;
412 OptionalLong shifted = shift(calendar, from, middle, 0);
413 if (shifted.isPresent() && shifted.getAsLong() <= to) low = middle;
414 else high = middle;
415 }
416 long years = low;
417 low = 0;
418 high = 32;
419 while (low + 1 < high) {
420 long middle = (low + high) / 2;
421 OptionalLong shifted = shift(calendar, from, years, middle);
422 if (shifted.isPresent() && shifted.getAsLong() <= to) low = middle;
423 else high = middle;
424 }
425 long months = low;
426 long days = to - shift(calendar, from, years, months).getAsLong();
427 return (first > last ? "-" : "") + years + "y " + months + "m " + days + "d";
428 }
429 static String roundDate(Calendar calendar, long fixed, int number, String tzName, int day, String month) {
430 int targetTz = TZ.indexOf(tzName);
431 int targetHaab = HB.indexOf(month) * 20 + day;
432 long candidate = fixed;
433 while ((candidate + 1137142 + 3) % 13 + 1 != number || (candidate + 1137142 + 19) % 20 != targetTz) candidate++;
434 for (; candidate <= fixed + 18979; candidate += 260) {
435 if ((candidate + 1137142 + 348) % 365 == targetHaab) return formatDate(calendar, candidate);
436 }
437 return "NONE";
438 }
439 static List<String> execute(String command, String[] args) {
440 validateArguments(command, args);
441 if (command.equals("EASTER") || command.equals("COMPUTUS")) {
442 int year = Integer.parseInt(args[0]);
443 if (year > 9999) return List.of("OUT OF RANGE");
444 Calendar kind = Calendar.valueOf(args[1]);
445 Easter easter = easter(year, kind);
446 if (command.equals("EASTER")) return List.of(formatDate(Calendar.GREGORIAN, easter.sunday()));
447 return List.of("golden=" + (year % 19 + 1) + " epact=" + easter.epact() + " letters=" + easter.letters() +
448 " moon=" + formatDate(kind, easter.moon()) + " easter=" + formatDate(kind, easter.sunday()));
449 }
450 if (List.of("PASSOVER", "HOLIDAYS", "MOLAD").contains(command)) {
451 int year = Integer.parseInt(args[0]);
452 if (command.equals("MOLAD")) {
453 Molad molad = molad(year);
454 if (!inRange(molad.start())) return List.of("OUT OF RANGE");
455 String applied = molad.rules().isEmpty() ? "NONE" : String.join("+", molad.rules());
456 return List.of(weekdayName(molad.day() - 1373427) + " " + molad.parts() / 1080 + "h " + molad.parts() % 1080 + "p " +
457 applied + " " + weekdayName(molad.start()) + " " + formatDate(Calendar.GREGORIAN, molad.start()));
458 }
459 if (year > 9999) return List.of("OUT OF RANGE");
460 if (command.equals("HOLIDAYS")) return holidayLines(year);
461 long hebrewYear = year + 3760;
462 int month = names(Calendar.HEBREW, hebrewYear).indexOf("Nisan") + 1;
463 long fixed = fixedDate(Calendar.HEBREW, hebrewYear, month, 15);
464 return List.of(formatDate(Calendar.GREGORIAN, fixed) + " " + weekdayName(fixed));
465 }
466 Calendar calendar = Calendar.valueOf(args[0]);
467 if (List.of("LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR").contains(command)) {
468 long year = Long.parseLong(args[1]);
469 return switch (command) {
470 case "LEAP" -> {
471 String text = (leap(calendar, year) ? "YES " : "NO ") + (start(calendar, year + 1) - start(calendar, year));
472 yield List.of(calendar == Calendar.HEBREW ? text + " " + yearType(year) : text);
473 }
474 case "MONTHS" -> {
475 List<String> monthNames = names(calendar, year);
476 int[] days = lengths(calendar, year);
477 List<String> items = new ArrayList<>();
478 for (int i = 0; i < days.length; i++) items.add(monthNames.get(i) + "=" + days[i]);
479 yield List.of(String.join(" ", items));
480 }
481 case "NEWYEAR" -> {
482 if (year > 9999) yield List.of("OUT OF RANGE");
483 List<YearStart> entries = newYears(calendar, start(Calendar.GREGORIAN, year), start(Calendar.GREGORIAN, year + 1) - 1);
484 if (entries.isEmpty()) yield List.of("NONE");
485 yield List.of(entries.stream().map(entry -> entry.year() + ":" + formatDate(Calendar.GREGORIAN, entry.fixed()))
486 .collect(Collectors.joining(" ")));
487 }
488 case "MONTHCAL" -> calendarLines(calendar, year, Integer.parseInt(args[2]));
489 default -> {
490 Period bounds = monthBounds(calendar, year, Integer.parseInt(args[2]));
491 int weekday = WEEKDAYS.indexOf(args[4]);
492 long fixed;
493 if (args[3].equals("LAST")) fixed = bounds.last() - Math.floorMod(bounds.last() - weekday, 7);
494 else fixed = bounds.first() + Math.floorMod(weekday - bounds.first(), 7) + 7L * (Integer.parseInt(args[3]) - 1);
495 yield List.of(fixed > bounds.last() ? "NONE" : formatDate(calendar, fixed));
496 }
497 };
498 }
499 if (command.equals("VALID")) {
500 try {
501 return List.of("YES " + formatDate(calendar, parseDate(calendar, args[1])));
502 } catch (DateError error) {
503 return List.of("NO " + error.getMessage());
504 }
505 }
506 long first = parseDate(calendar, args[1]);
507 return switch (command) {
508 case "DIFF" -> List.of(Long.toString(parseDate(Calendar.valueOf(args[2]), args[3]) - first));
509 case "COUNT" -> {
510 long last = parseDate(calendar, args[2]);
511 long low = Math.min(first, last), high = Math.max(first, last);
512 long occurrence = low + Math.floorMod(WEEKDAYS.indexOf(args[3]) - low, 7);
513 yield List.of(Long.toString(occurrence > high ? 0 : (high - occurrence) / 7 + 1));
514 }
515 case "SPAN" -> List.of(spanText(calendar, first, parseDate(calendar, args[2])));
516 case "CONVERT" -> List.of(formatDate(Calendar.valueOf(args[3]), first));
517 case "WEEKDAY" -> List.of(weekdayName(first));
518 case "ALL" -> {
519 List<String> result = new ArrayList<>();
520 for (Calendar target : Calendar.values()) {
521 String text = formatDate(target, first);
522 if (YMD.contains(target) && !text.equals("OUT OF RANGE")) {
523 Parts parts = partsOf(target, first);
524 text += " " + names(target, parts.year()).get(parts.month() - 1);
525 }
526 result.add(target + " " + text);
527 }
528 yield result;
529 }
530 case "INFO" -> {
531 Parts parts = partsOf(calendar, first);
532 String weekday = weekdayName(first);
533 if (calendar == Calendar.FRENCH) weekday = parts.month() <= 12 ? DC.get((parts.day() - 1) % 10) : FS.get(parts.day() - 1);
534 long ordinal = first - start(calendar, parts.year()) + 1;
535 long length = start(calendar, parts.year() + 1) - start(calendar, parts.year());
536 yield List.of(weekday + " " + parts.day() + " " + names(calendar, parts.year()).get(parts.month() - 1) + " " +
537 parts.year() + " day=" + ordinal + "/" + length);
538 }
539 case "ROUND" -> List.of(roundDate(calendar, first, Integer.parseInt(args[2]), args[3], Integer.parseInt(args[4]), args[5]));
540 default -> {
541 int amount = Integer.parseInt(args[2]);
542 OptionalLong result = switch (args[3]) {
543 case "DAYS" -> OptionalLong.of(first + amount);
544 case "WEEKS" -> OptionalLong.of(first + 7L * amount);
545 case "YEARS" -> shift(calendar, first, amount, 0);
546 default -> shift(calendar, first, 0, amount);
547 };
548 yield List.of(result.isPresent() ? formatDate(calendar, result.getAsLong()) : "OUT OF RANGE");
549 }
550 };
551 }
552 public static void main(String[] args) throws IOException {
553 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
554 PrintWriter output = new PrintWriter(new BufferedWriter(new OutputStreamWriter(System.out)));
555 int count = Integer.parseInt(input.readLine());
556 for (int i = 0; i < count; i++) {
557 String[] words = input.readLine().split(" ");
558 String command = words[0];
559 String[] arguments = Arrays.copyOfRange(words, 1, words.length);
560 List<String> lines;
561 if (ARITY.getOrDefault(command, -1) != arguments.length) {
562 lines = List.of("INVALID COMMAND");
563 } else {
564 try {
565 lines = execute(command, arguments);
566 } catch (CommandError error) {
567 lines = List.of("INVALID COMMAND");
568 } catch (DateError error) {
569 lines = List.of(error.getMessage().equals("RANGE") ? "OUT OF RANGE" : "INVALID DATE");
570 }
571 }
572 lines.forEach(output::println);
573 }
574 output.flush();
575 }
