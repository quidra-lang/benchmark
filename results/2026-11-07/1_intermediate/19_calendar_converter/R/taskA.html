47leap <- function(calendar, year) {
48 if (calendar == "GREGORIAN")
49 return(year %% 4 == 0 && (year %% 100 != 0 || year %% 400 == 0))
50 if (calendar == "JULIAN") return(year %% 4 == 0)
51 if (calendar == "ISLAMIC")
52 return(year %% 30 %in% c(2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29))
53 if (calendar == "HEBREW") return((7 * year + 1) %% 19 < 7)
54 if (calendar == "PERSIAN") {
55 b <- (year - 474) %% 2820 + 474
56 return(((b + 38) * 682) %% 2816 < 682)
57 }
58 if (calendar %in% c("COPTIC", "ETHIOPIC")) return(year %% 4 == 3)
59 if (calendar == "FRENCH")
60 return(year %% 4 == 0 && !(year %% 400 %in% c(100, 200, 300)) && year %% 4000 != 0)
61 (start_fixed("ISO", year + 1) - start_fixed("ISO", year)) / 7 == 53
62}

66molad <- function(year) {
67 quantity <- 5604 + 765433 * months_before(year)
68 day <- floor(quantity / 25920)
69 parts <- quantity %% 25920
70 weekday <- (day - 1373427) %% 7
71 postponed <- day
72 rules <- character()
73 if (parts >= 19440) {
74 postponed <- postponed + 1
75 rules <- c(rules, "ZAKEN")
76 } else if (!leap("HEBREW", year) && weekday == 2 && parts >= 9924) {
77 postponed <- postponed + 2
78 rules <- c(rules, "GATARAD")
79 } else if (leap("HEBREW", year - 1) && weekday == 1 && parts >= 16789) {
80 postponed <- postponed + 1
81 rules <- c(rules, "BETUTAKPAT")
82 }
83 if ((postponed - 1373427) %% 7 %in% c(0, 3, 5)) {
84 postponed <- postponed + 1
85 rules <- c(rules, "ADU")
86 }
87 list(day = day, parts = parts, fixed = postponed - 1373427, rules = rules)
88}

90start_fixed <- function(calendar, year) {
91 previous <- year - 1
92 if (calendar == "GREGORIAN")
93 return(365 * previous + floor(previous / 4) - floor(previous / 100) +
94 floor(previous / 400) + 1)
95 if (calendar == "JULIAN") return(365 * previous + floor(previous / 4) - 1)
96 if (calendar == "ISLAMIC")
97 return(227015 + 354 * previous + floor((3 + 11 * year) / 30))
98 if (calendar == "HEBREW") return(molad(year)$fixed)
99 if (calendar == "PERSIAN") {
100 a <- year - 474
101 b <- a %% 2820 + 474
102 return(226896 + 1029983 * floor(a / 2820) + 365 * (b - 1) +
103 floor((682 * b - 110) / 2816))
104 }
105 if (calendar == "COPTIC") return(103605 + 365 * previous + floor(year / 4))
106 if (calendar == "ETHIOPIC") return(2796 + 365 * previous + floor(year / 4))
107 if (calendar == "FRENCH")
108 return(654415 + 365 * previous + floor(previous / 4) -
109 floor(previous / 100) + floor(previous / 400) - floor(previous / 4000))
110 january_four <- start_fixed("GREGORIAN", year) + 3
111 january_four - (january_four - 1) %% 7
112}

147year_of <- function(calendar, fixed) {
148 if (fixed < start_fixed(calendar, 1)) return(0)
149 low <- 1
150 high <- 100000
151 while (low + 1 < high) {
152 middle <- floor((low + high) / 2)
153 if (start_fixed(calendar, middle) <= fixed) low <- middle else high <- middle
154 }

158parts_of <- function(calendar, fixed) {
159 year <- year_of(calendar, fixed)
160 if (year == 0) return(NULL)
161 offset <- fixed - start_fixed(calendar, year)
162 if (calendar == "ISO")
163 return(c(year, floor(offset / 7) + 1, offset %% 7 + 1))
164 lengths <- month_lengths(calendar, year)
165 for (month in seq_along(lengths)) {
166 if (offset < lengths[month]) return(c(year, month, offset + 1))
167 offset <- offset - lengths[month]
168 }
169 stop("Inconsistent calendar year", call. = FALSE)
170}

172format_date <- function(calendar, fixed) {
173 if (!supported(fixed)) return("OUT OF RANGE")
174 if (calendar %in% year_calendars) {
175 fields <- parts_of(calendar, fixed)
176 if (is.null(fields)) return("OUT OF RANGE")
177 if (calendar == "ISO")
178 return(sprintf("%04d-W%02d-%d", fields[1], fields[2], fields[3]))
179 return(sprintf("%04d-%02d-%02d", fields[1], fields[2], fields[3]))
180 }
181 if (calendar == "JD") return(sprintf("%.0f", fixed + 1721425))
182 if (calendar == "FIXED") return(sprintf("%.0f", fixed))
183 count <- fixed + 1137142
184 digits <- c(
185 floor(count / 144000),
186 floor(count / 7200) %% 20,
187 floor(count / 360) %% 20,
188 floor(count / 20) %% 18,
189 count %% 20
190 )
191 tz_number <- (count + 3) %% 13 + 1
192 tz_name <- tzolkin[(count + 19) %% 20 + 1]
193 haab_count <- (count + 348) %% 365
194 paste(
195 paste(digits, collapse = "."), tz_number, tz_name,
196 haab_count %% 20, haab[floor(haab_count / 20) + 1]
197 )
198}

200parse_date <- function(calendar, text) {
201 if (calendar %in% ymd) {
202 if (!grepl("^[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}$", text))
203 date_error("FORMAT")
204 components <- as.numeric(strsplit(text, "-", fixed = TRUE)[[1]])
205 year <- components[1]
206 month <- components[2]
207 day <- components[3]
208 if (year == 0) date_error("YEAR")
209 lengths <- month_lengths(calendar, year)
210 if (month < 1 || month > length(lengths)) date_error("MONTH")
211 if (day < 1 || day > lengths[month]) date_error("DAY")
212 fixed <- fixed_date(calendar, year, month, day)
213 } else if (calendar == "ISO") {
214 if (!grepl("^[0-9]{1,5}-W[0-9]{1,2}-[0-9]$", text)) date_error("FORMAT")
215 components <- strsplit(text, "-", fixed = TRUE)[[1]]
216 year <- as.numeric(components[1])
217 week <- as.numeric(substring(components[2], 2))
218 day <- as.numeric(components[3])
219 if (year == 0) date_error("YEAR")
220 weeks <- (start_fixed(calendar, year + 1) - start_fixed(calendar, year)) / 7
221 if (week < 1 || week > weeks) date_error("WEEK")
222 if (day < 1 || day > 7) date_error("DAY")
223 fixed <- fixed_date(calendar, year, week, day)
224 } else if (calendar == "MAYAN") {
225 if (!grepl("^[0-9]{1,2}(\\.[0-9]{1,2}){4}$", text))
226 date_error("FORMAT")
227 values <- as.numeric(strsplit(text, ".", fixed = TRUE)[[1]])
228 if (values[2] > 19 || values[3] > 19 || values[4] > 17 || values[5] > 19)
229 date_error("FIELD")
230 fixed <- (((values[1] * 20 + values[2]) * 20 + values[3]) *
231 18 + values[4]) * 20 + values[5] - 1137142
232 } else {
233 if (!grepl("^-?[0-9]{1,8}$", text)) date_error("FORMAT")
234 fixed <- as.numeric(text)
235 if (calendar == "JD") fixed <- fixed - 1721425
236 }
237 if (!supported(fixed)) date_error("RANGE")
238 fixed
239}

247shift_date <- function(calendar, fixed, years, months) {
248 fields <- parts_of(calendar, fixed)
249 year <- fields[1]
250 month <- fields[2]
251 day <- fields[3]
252 target_year <- year + years
253 if (calendar == "ISO") {
254 if (target_year < 1 || target_year > 99999) return(NA_real_)
255 weeks <- (start_fixed(calendar, target_year + 1) - start_fixed(calendar, target_year)) / 7
256 return(fixed_date(calendar, target_year, min(month, weeks), day))
257 }
258 if (calendar == "HEBREW") {
259 old_name <- names_for(calendar, year)[month]
260 new_names <- names_for(calendar, target_year)
261 if (!(old_name %in% new_names))
262 old_name <- if (old_name == "Adar") "AdarII" else "Adar"
263 month <- match(old_name, new_names)
264 }
265 absolute <- month_index(calendar, target_year) + month - 1 + months
266 if (calendar == "HEBREW") {
267 final_year <- floor((19 * absolute + 234) / 235)
268 while (month_index(calendar, final_year) > absolute)
269 final_year <- final_year - 1
270 while (month_index(calendar, final_year + 1) <= absolute)
271 final_year <- final_year + 1
272 } else {
273 count <- length(names_for(calendar, target_year))
274 final_year <- floor(absolute / count) + 1
275 }
276 if (final_year < 1 || final_year > 99999) return(NA_real_)
277 final_month <- absolute - month_index(calendar, final_year) + 1
278 final_day <- min(day, month_lengths(calendar, final_year)[final_month])
279 fixed_date(calendar, final_year, final_month, final_day)
280}

300easter <- function(year, calendar) {
301 golden <- year %% 19
302 if (calendar == "GREGORIAN") {
303 century <- floor(year / 100) + 1
304 epact <- (14 + 11 * golden - floor(3 * century / 4) +
305 floor((5 + 8 * century) / 25)) %% 30
306 if (epact == 0 || (epact == 1 && golden > 10)) epact <- epact + 1
307 } else {
308 epact <- (14 + 11 * golden) %% 30
309 }
310 moon <- fixed_date(calendar, year, 4, 19) - epact
311 sunday <- moon + 7 - moon %% 7
312 index <- (-start_fixed(calendar, year)) %% 7
313 letters <- substring("ABCDEFG", index + 1, index + 1)
314 if (leap(calendar, year)) {
315 another <- (index - 1) %% 7
316 letters <- paste0(letters, substring("ABCDEFG", another + 1, another + 1))
317 }
318 list(epact = epact, letters = letters, moon = moon, sunday = sunday)
319}

349holiday_lines <- function(year) {
350 first <- start_fixed("GREGORIAN", year)
351 last <- start_fixed("GREGORIAN", year + 1) - 1
352 western <- easter(year, "GREGORIAN")$sunday
353 orthodox <- easter(year, "JULIAN")$sunday
354 fixed <- numeric()
355 priority <- integer()
356 name <- character()
357 calendar <- character()
358 add <- function(day, order, label, kind) {
359 fixed <<- c(fixed, day)
360 priority <<- c(priority, order)
361 name <<- c(name, label)
362 calendar <<- c(calendar, kind)
363 }
364 movable <- c(AshWednesday = -46, GoodFriday = -2, Easter = 0,
365 Ascension = 39, Pentecost = 49)
366 for (i in seq_along(movable))
367 add(western + movable[i], i - 1, names(movable)[i], "GREGORIAN")
368 add(orthodox, 5, "OrthodoxEaster", "JULIAN")
369 for (i in seq_len(nrow(holidays))) {
370 entry <- holidays[i, ]
371 kind <- entry[2]
372 candidate <- max(1, year_of(kind, first))
373 while (start_fixed(kind, candidate) <= last) {
374 months <- names_for(kind, candidate)
375 month_name <- entry[3]
376 if (kind == "HEBREW" && month_name == "Adar" && leap(kind, candidate))
377 month_name <- "AdarII"
378 month <- match(month_name, months)
379 day <- as.numeric(entry[4])
380 if (day <= month_lengths(kind, candidate)[month]) {
381 current <- fixed_date(kind, candidate, month, day)
382 if (first <= current && current <= last)
383 add(current, 5 + i, entry[1], kind)
384 }
385 candidate <- candidate + 1
386 }
387 }
388 sorted <- order(fixed, priority)
389 result <- character()
390 for (i in sorted) {
391 day <- fixed[i]
392 result <- c(result, paste(
393 format_date("GREGORIAN", day), weekdays[day %% 7 + 1],
394 name[i], calendar[i], format_date(calendar[i], day)
395 ))
396 }
397 result
398}

400validate_arguments <- function(command, args) {
401 if (command %in% c("EASTER", "COMPUTUS")) {
402 require_command(year_valid(args[1]) && args[2] %in% c("GREGORIAN", "JULIAN"))
403 return(invisible(NULL))
404 }
405 if (command %in% c("PASSOVER", "HOLIDAYS", "MOLAD")) {
406 require_command(year_valid(args[1]))
407 return(invisible(NULL))
408 }
409 calendar <- args[1]
410 allowed <- calendars
411 if (command %in% c("INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN")) {
412 allowed <- ymd
413 } else if (command %in% c("LEAP", "NEWYEAR")) {
414 allowed <- year_calendars
415 }
416 require_command(calendar %in% allowed)
417 if (command == "CONVERT") {
418 require_command(args[3] == "TO" && args[4] %in% calendars)
419 } else if (command == "DIFF") {
420 require_command(args[3] %in% calendars)
421 } else if (command %in% c("LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR")) {
422 require_command(year_valid(args[2]))
423 if (command %in% c("MONTHCAL", "NTH")) require_command(digits(args[3], 2))
424 if (command == "NTH") {
425 require_command(args[4] %in% c("1", "2", "3", "4", "5", "LAST"))
426 require_command(args[5] %in% weekdays)
427 }
428 } else if (command == "COUNT") {
429 require_command(args[4] %in% weekdays)
430 } else if (command == "ADD") {
431 require_command(grepl("^-?[0-9]{1,7}$", args[3]))
432 require_command(args[4] %in% c("DAYS", "WEEKS", "MONTHS", "YEARS"))
433 if (args[4] == "MONTHS") require_command(calendar %in% ymd)
434 if (args[4] == "YEARS") require_command(calendar %in% year_calendars)
435 } else if (command == "ROUND") {
436 require_command(digits(args[3], 2))
437 require_command(as.numeric(args[3]) >= 1 && as.numeric(args[3]) <= 13)
438 require_command(args[4] %in% tzolkin)
439 require_command(digits(args[5], 2))
440 require_command(args[6] %in% haab)
441 maximum_day <- if (args[6] == "Uayeb") 4 else 19
442 require_command(as.numeric(args[5]) <= maximum_day)
443 }
444}

446calendar_lines <- function(calendar, year, month) {
447 bounds <- month_bounds(calendar, year, month)
448 first <- bounds[1]
449 last <- bounds[2]
450 if (!supported(first) || !supported(last)) return("OUT OF RANGE")
451 title <- paste0(
452 names_for(calendar, year)[month], " ", year, " (",
453 format_date("GREGORIAN", first), " to ", format_date("GREGORIAN", last), ")"
454 )
455 result <- c(title, "Mo Tu We Th Fr Sa Su")
456 cursor <- first - (first - 1) %% 7
457 while (cursor <= last) {
458 fields <- character()
459 for (day in cursor:(cursor + 6)) {
460 if (first <= day && day <= last) {
461 fields <- c(fields, sprintf("%2d", day - first + 1))
462 } else {
463 fields <- c(fields, " ")
464 }
465 }
466 result <- c(result, sub("[[:space:]]+$", "", paste(fields, collapse = " ")))
467 cursor <- cursor + 7
468 }
469 result
470}

472span_text <- function(calendar, first, last) {
473 negative <- first > last
474 if (negative) {
475 temporary <- first
476 first <- last
477 last <- temporary
478 }
479 first_year <- parts_of(calendar, first)[1]
480 last_year <- parts_of(calendar, last)[1]
481 low <- 0
482 high <- last_year - first_year + 2
483 while (low + 1 < high) {
484 middle <- floor((low + high) / 2)
485 shifted <- shift_date(calendar, first, middle, 0)
486 if (!is.na(shifted) && shifted <= last) low <- middle else high <- middle
487 }
488 years <- low
489 low <- 0
490 high <- 32
491 while (low + 1 < high) {
492 middle <- floor((low + high) / 2)
493 shifted <- shift_date(calendar, first, years, middle)
494 if (!is.na(shifted) && shifted <= last) low <- middle else high <- middle
495 }
496 months <- low
497 days <- last - shift_date(calendar, first, years, months)
498 paste0(if (negative) "-" else "", years, "y ", months, "m ", days, "d")
499}

501round_date <- function(calendar, fixed, number, tz_name, day, month) {
502 target_tz <- match(tz_name, tzolkin) - 1
503 target_haab <- (match(month, haab) - 1) * 20 + day
504 candidate <- fixed

506 count <- candidate + 1137142

511 if ((candidate + 1137142 + 348) %% 365 == target_haab)
512 return(format_date(calendar, candidate))
513 candidate <- candidate + 260
514 }
515 "NONE"
516}

518execute <- function(command, args) {
519 validate_arguments(command, args)
520 if (command %in% c("EASTER", "COMPUTUS")) {
521 year <- as.numeric(args[1])
522 if (year > 9999) return("OUT OF RANGE")
523 kind <- args[2]
524 result <- easter(year, kind)
525 if (command == "EASTER") return(format_date("GREGORIAN", result$sunday))
526 return(paste0(
527 "golden=", year %% 19 + 1, " epact=", result$epact, " letters=", result$letters,
