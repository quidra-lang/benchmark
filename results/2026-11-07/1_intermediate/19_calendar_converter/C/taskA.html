129static bool leap(Calendar calendar, long long year) {
130 switch (calendar) {
131 case GREGORIAN:
132 return mod(year, 4) == 0 && (mod(year, 100) != 0 || mod(year, 400) == 0);
133 case JULIAN:
134 return mod(year, 4) == 0;
135 case ISLAMIC: {
136 static const int leap_years[] = {2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29};
137 for (int i = 0; i < 11; ++i) {
138 if (mod(year, 30) == leap_years[i]) return true;
139 }
140 return false;
141 }
142 case HEBREW:
143 return mod(7 * year + 1, 19) < 7;
144 case PERSIAN: {
145 long long b = mod(year - 474, 2820) + 474;
146 return mod((b + 38) * 682, 2816) < 682;
147 }
148 case COPTIC:
149 case ETHIOPIC:
150 return mod(year, 4) == 3;
151 case FRENCH: {
152 long long century = mod(year, 400);
153 return mod(year, 4) == 0 && century != 100 && century != 200 && century != 300 && mod(year, 4000) != 0;
154 }
155 default:
156 return (start(ISO, year + 1) - start(ISO, year)) / 7 == 53;
157 }
158}

167static Molad molad(long long year) {
168 long long quantity = 5604 + 765433 * months_before(year);
169 Molad result = {floor_div(quantity, 25920), mod(quantity, 25920), 0, ""};
170 long long weekday = mod(result.day - 1373427, 7);
171 long long postponed = result.day;
172 if (result.parts >= 19440) {
173 postponed += 1;
174 add_rule(&result, "ZAKEN");
175 } else if (!leap(HEBREW, year) && weekday == 2 && result.parts >= 9924) {
176 postponed += 2;
177 add_rule(&result, "GATARAD");
178 } else if (leap(HEBREW, year - 1) && weekday == 1 && result.parts >= 16789) {
179 postponed += 1;
180 add_rule(&result, "BETUTAKPAT");
181 }
182 long long final_weekday = mod(postponed - 1373427, 7);
183 if (final_weekday == 0 || final_weekday == 3 || final_weekday == 5) {
184 postponed += 1;
185 add_rule(&result, "ADU");
186 }
187 result.start = postponed - 1373427;
188 return result;
189}

191static long long start(Calendar calendar, long long year) {
192 long long previous = year - 1;
193 switch (calendar) {
194 case GREGORIAN:
195 return 365 * previous + floor_div(previous, 4) - floor_div(previous, 100) + floor_div(previous, 400) + 1;
196 case JULIAN:
197 return 365 * previous + floor_div(previous, 4) - 1;
198 case ISLAMIC:
199 return 227015 + 354 * previous + floor_div(3 + 11 * year, 30);
200 case HEBREW:
201 return molad(year).start;
202 case PERSIAN: {
203 long long a = year - 474;
204 long long b = mod(a, 2820) + 474;
205 return 226896 + 1029983 * floor_div(a, 2820) + 365 * (b - 1) + floor_div(682 * b - 110, 2816);
206 }
207 case COPTIC:
208 return 103605 + 365 * previous + floor_div(year, 4);
209 case ETHIOPIC:
210 return 2796 + 365 * previous + floor_div(year, 4);
211 case FRENCH:
212 return 654415 + 365 * previous + floor_div(previous, 4) - floor_div(previous, 100)
213 + floor_div(previous, 400) - floor_div(previous, 4000);
214 default: {
215 long long january_four = start(GREGORIAN, year) + 3;
216 return january_four - mod(january_four - 1, 7);
217 }
218 }
219}

264static long long fixed_date(Calendar calendar, long long year, int month, int day) {
265 if (calendar == ISO) return start(calendar, year) + 7 * (month - 1) + day - 1;
266 Months months = lengths(calendar, year);
267 long long fixed = start(calendar, year) + day - 1;
268 for (int i = 0; i < month - 1; ++i) fixed += months.days[i];
269 return fixed;
270}

272static long long year_of(Calendar calendar, long long fixed) {
273 if (fixed < start(calendar, 1)) return 0;
274 long long low = 1, high = 100000;
275 while (low + 1 < high) {
276 long long middle = (low + high) / 2;
277 if (start(calendar, middle) <= fixed) low = middle;
278 else high = middle;
279 }
280 return low;
281}

283static Date parts_of(Calendar calendar, long long fixed) {
284 Date date = {year_of(calendar, fixed), 0, 0};
285 if (date.year == 0) return date;
286 long long offset = fixed - start(calendar, date.year);
287 if (calendar == ISO) {
288 date.month = (int)(offset / 7) + 1;
289 date.day = (int)(offset % 7) + 1;
290 return date;
291 }
292 Months months = lengths(calendar, date.year);
293 date.month = 1;
294 while (offset >= months.days[date.month - 1]) {
295 offset -= months.days[date.month - 1];
296 ++date.month;
297 }
298 date.day = (int)offset + 1;
299 return date;
300}

302static char *format_date(Calendar calendar, long long fixed, char *text) {
303 if (!in_range(fixed)) {
304 snprintf(text, TEXT_SIZE, "OUT OF RANGE");
305 } else if (has_years(calendar)) {
306 Date date = parts_of(calendar, fixed);
307 if (date.year == 0) snprintf(text, TEXT_SIZE, "OUT OF RANGE");
308 else if (calendar == ISO) snprintf(text, TEXT_SIZE, "%04lld-W%02d-%d", date.year, date.month, date.day);
309 else snprintf(text, TEXT_SIZE, "%04lld-%02d-%02d", date.year, date.month, date.day);
310 } else if (calendar == JD) {
311 snprintf(text, TEXT_SIZE, "%lld", fixed + 1721425);
312 } else if (calendar == FIXED) {
313 snprintf(text, TEXT_SIZE, "%lld", fixed);
314 } else {
315 long long count = fixed + 1137142;
316 long long haab = (count + 348) % 365;
317 snprintf(text, TEXT_SIZE, "%lld.%lld.%lld.%lld.%lld %lld %s %lld %s",
318 count / 144000, count / 7200 % 20, count / 360 % 20, count / 20 % 18, count % 20,
319 (count + 3) % 13 + 1, tzolkin_names[(count + 19) % 20], haab % 20, haab_names[haab / 20]);
320 }
321 return text;
322}

324static Reason parse_date(Calendar calendar, const char *text, long long *fixed) {
325 int length = 0;
326 if (is_ymd(calendar)) {
327 char year_text[6], month_text[3], day_text[3];
328 int matched = sscanf(text, "%5[0-9]-%2[0-9]-%2[0-9]%n", year_text, month_text, day_text, &length);
329 if (matched != 3 || text[length] != '\0') return FORMAT;
330 long long year = atoll(year_text);
331 int month = atoi(month_text), day = atoi(day_text);
332 if (year == 0) return YEAR;
333 Months months = lengths(calendar, year);
334 if (month < 1 || month > months.count) return MONTH;
335 if (day < 1 || day > months.days[month - 1]) return DAY;
336 *fixed = fixed_date(calendar, year, month, day);
337 } else if (calendar == ISO) {
338 char year_text[6], week_text[3], day_text[2];
339 int matched = sscanf(text, "%5[0-9]-W%2[0-9]-%1[0-9]%n", year_text, week_text, day_text, &length);
340 if (matched != 3 || text[length] != '\0') return FORMAT;
341 long long year = atoll(year_text);
342 int week = atoi(week_text), day = atoi(day_text);
343 if (year == 0) return YEAR;
344 long long weeks = (start(calendar, year + 1) - start(calendar, year)) / 7;
345 if (week < 1 || week > weeks) return WEEK;
346 if (day < 1 || day > 7) return DAY;
347 *fixed = fixed_date(calendar, year, week, day);
348 } else if (calendar == MAYAN) {
349 char fields[5][3];
350 int matched = sscanf(text, "%2[0-9].%2[0-9].%2[0-9].%2[0-9].%2[0-9]%n",
351 fields[0], fields[1], fields[2], fields[3], fields[4], &length);
352 if (matched != 5 || text[length] != '\0') return FORMAT;
353 long long b = atoll(fields[0]), k = atoll(fields[1]), t = atoll(fields[2]);
354 long long u = atoll(fields[3]), n = atoll(fields[4]);
355 if (k > 19 || t > 19 || u > 17 || n > 19) return FIELD;
356 *fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142;
357 } else {
358 if (!digits(text[0] == '-' ? text + 1 : text, 1, 8)) return FORMAT;
359 *fixed = atoll(text) - (calendar == JD ? 1721425 : 0);
360 }
361 return in_range(*fixed) ? ACCEPTED : RANGE;
362}

373static bool shift(Calendar calendar, long long fixed, long long years, long long months, long long *result) {
374 Date date = parts_of(calendar, fixed);
375 long long target_year = date.year + years;
376 if (calendar == ISO) {
377 if (target_year < 1 || target_year > 99999) return false;
378 long long weeks = (start(calendar, target_year + 1) - start(calendar, target_year)) / 7;
379 *result = fixed_date(calendar, target_year, (int)smaller(date.month, weeks), date.day);
380 return true;
381 }
382 int month = date.month;
383 if (calendar == HEBREW) {
384 const char *name = names(calendar, date.year)[month - 1];
385 month = month_number(calendar, target_year, name);
386 if (month == 0) month = month_number(calendar, target_year, strcmp(name, "Adar") == 0 ? "AdarII" : "Adar");
387 }
388 long long absolute = month_index(calendar, target_year) + month - 1 + months;
389 long long final_year;
390 if (calendar == HEBREW) {
391 final_year = floor_div(19 * absolute + 234, 235);
392 while (month_index(calendar, final_year) > absolute) --final_year;
393 while (month_index(calendar, final_year + 1) <= absolute) ++final_year;
394 } else {
395 final_year = floor_div(absolute, months_per_year(calendar)) + 1;
396 }
397 if (final_year < 1 || final_year > 99999) return false;
398 int final_month = (int)(absolute - month_index(calendar, final_year)) + 1;
399 int final_day = (int)smaller(date.day, lengths(calendar, final_year).days[final_month - 1]);
400 *result = fixed_date(calendar, final_year, final_month, final_day);
401 return true;
402}

404static bool month_bounds(Calendar calendar, long long year, int month, long long *first, long long *last) {
405 Months months = lengths(calendar, year);
406 if (month < 1 || month > months.count) return false;
407 *first = fixed_date(calendar, year, month, 1);
408 *last = *first + months.days[month - 1] - 1;
409 return true;
410}

412static int new_years(Calendar calendar, long long first, long long last, NewYear *found) {
413 long long year = larger(1, year_of(calendar, first));
414 while (start(calendar, year) < first) ++year;
415 int count = 0;
416 for (; start(calendar, year) <= last; ++year) found[count++] = (NewYear){year, start(calendar, year)};
417 return count;
418}

420static Easter easter(long long year, Calendar calendar) {
421 long long golden = year % 19;
422 Easter result = {0, "", 0, 0};
423 if (calendar == GREGORIAN) {
424 long long century = year / 100 + 1;
425 result.epact = mod(14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25, 30);
426 if (result.epact == 0 || (result.epact == 1 && golden > 10)) ++result.epact;
427 } else {
428 result.epact = (14 + 11 * golden) % 30;
429 }
430 result.moon = fixed_date(calendar, year, 4, 19) - result.epact;
431 result.sunday = result.moon + 7 - mod(result.moon, 7);
432 long long index = mod(-start(calendar, year), 7);
433 result.letters[0] = "ABCDEFG"[index];
434 if (leap(calendar, year)) result.letters[1] = "ABCDEFG"[mod(index - 1, 7)];
435 return result;
436}

438static int compare_entries(const void *left, const void *right) {
439 const Entry *a = left, *b = right;
440 if (a->fixed != b->fixed) return a->fixed < b->fixed ? -1 : 1;
441 return a->order - b->order;
442}

444static void print_holidays(long long year) {
445 static const struct { const char *name; int offset; } movable[] = {
446 {"AshWednesday", -46}, {"GoodFriday", -2}, {"Easter", 0}, {"Ascension", 39}, {"Pentecost", 49},
447 };
448 long long first = start(GREGORIAN, year), last = start(GREGORIAN, year + 1) - 1;
449 long long gregorian_easter = easter(year, GREGORIAN).sunday;
450 Entry entries[64];
451 int count = 0;
452 for (int order = 0; order < 5; ++order) {
453 entries[count++] = (Entry){gregorian_easter + movable[order].offset, order, movable[order].name, GREGORIAN};
454 }
455 entries[count++] = (Entry){easter(year, JULIAN).sunday, 5, "OrthodoxEaster", JULIAN};
456 int holiday_count = (int)(sizeof holiday_table / sizeof *holiday_table);
457 for (int i = 0; i < holiday_count; ++i) {
458 const Holiday *holiday = &holiday_table[i];
459 Calendar calendar = holiday->calendar;
460 for (long long candidate = larger(1, year_of(calendar, first)); start(calendar, candidate) <= last; ++candidate) {
461 const char *month_name = holiday->month;
462 if (calendar == HEBREW && strcmp(month_name, "Adar") == 0 && leap(calendar, candidate)) month_name = "AdarII";
463 int month = month_number(calendar, candidate, month_name);
464 if (holiday->day > lengths(calendar, candidate).days[month - 1]) continue;
465 long long fixed = fixed_date(calendar, candidate, month, holiday->day);
466 if (fixed >= first && fixed <= last) entries[count++] = (Entry){fixed, i + 6, holiday->name, calendar};
467 }
468 }
469 qsort(entries, (size_t)count, sizeof *entries, compare_entries);
470 for (int i = 0; i < count; ++i) {
471 char gregorian[TEXT_SIZE], native[TEXT_SIZE];
472 printf("%s %s %s %s %s\n", format_date(GREGORIAN, entries[i].fixed, gregorian), weekday_names[entries[i].fixed % 7],
473 entries[i].name, calendar_names[entries[i].calendar], format_date(entries[i].calendar, entries[i].fixed, native));
474 }

477static bool valid_year(const char *text) { return digits(text, 1, 5) && atol(text) >= 1; }
478
479static bool valid_arguments(Command command, char **args) {
480 if (command == EASTER || command == COMPUTUS) {
481 int kind = calendar_id(args[1]);
482 return valid_year(args[0]) && (kind == GREGORIAN || kind == JULIAN);
483 }
484 if (command == PASSOVER || command == HOLIDAYS || command == MOLAD) return valid_year(args[0]);
485 int calendar = calendar_id(args[0]);
486 if (calendar < 0) return false;
487 if ((command == INFO || command == MONTHS || command == MONTHCAL || command == NTH || command == SPAN) && !is_ymd(calendar)) return false;
488 if ((command == LEAP || command == NEWYEAR) && !has_years(calendar)) return false;
489 switch (command) {
490 case CONVERT:
491 return strcmp(args[2], "TO") == 0 && calendar_id(args[3]) >= 0;
492 case DIFF:
493 return calendar_id(args[2]) >= 0;
494 case LEAP:
495 case MONTHS:
496 case NEWYEAR:
497 return valid_year(args[1]);
498 case MONTHCAL:
499 return valid_year(args[1]) && digits(args[2], 1, 2);
500 case NTH: {
501 static const char *const ordinals[] = {"1", "2", "3", "4", "5", "LAST"};
502 return valid_year(args[1]) && digits(args[2], 1, 2) && lookup(ordinals, 6, args[3]) >= 0 && weekday_id(args[4]) >= 0;
503 }
504 case COUNT:
505 return weekday_id(args[3]) >= 0;
506 case ADD: {
507 static const char *const units[] = {"DAYS", "WEEKS", "MONTHS", "YEARS"};
508 if (!digits(args[2][0] == '-' ? args[2] + 1 : args[2], 1, 7) || lookup(units, 4, args[3]) < 0) return false;
509 if (strcmp(args[3], "MONTHS") == 0) return is_ymd(calendar);
510 if (strcmp(args[3], "YEARS") == 0) return has_years(calendar);
511 return true;
512 }
513 case ROUND:
514 if (!digits(args[2], 1, 2) || atoi(args[2]) < 1 || atoi(args[2]) > 13 || lookup(tzolkin_names, 20, args[3]) < 0) return false;
515 if (!digits(args[4], 1, 2) || lookup(haab_names, 19, args[5]) < 0) return false;
516 return atoi(args[4]) <= (strcmp(args[5], "Uayeb") == 0 ? 4 : 19);
517 default:
518 return true;
519 }
520}

522static void print_month_calendar(Calendar calendar, long long year, int month) {
523 long long first, last;
524 if (!month_bounds(calendar, year, month, &first, &last)) {
525 puts("INVALID DATE");
526 return;
527 }
528 if (!in_range(first) || !in_range(last)) {
529 puts("OUT OF RANGE");
530 return;
531 }
532 char from[TEXT_SIZE], to[TEXT_SIZE];
533 printf("%s %lld (%s to %s)\n", names(calendar, year)[month - 1], year,
534 format_date(GREGORIAN, first, from), format_date(GREGORIAN, last, to));
535 puts("Mo Tu We Th Fr Sa Su");
536 for (long long cursor = first - (first - 1) % 7; cursor <= last; cursor += 7) {
537 char row[32];
538 int length = 0;
539 for (long long fixed = cursor; fixed < cursor + 7; ++fixed) {
540 if (fixed > cursor) row[length++] = ' ';
541 if (fixed >= first && fixed <= last) length += sprintf(row + length, "%2lld", fixed - first + 1);
542 else length += sprintf(row + length, " ");
543 }
544 while (length > 0 && row[length - 1] == ' ') --length;
545 row[length] = '\0';
546 puts(row);
547 }
548}

550static char *span_text(Calendar calendar, long long first, long long last, char *text) {
551 bool negative = first > last;
552 if (negative) {
553 long long swap = first;
554 first = last;
555 last = swap;
556 }
557 long long start_year = parts_of(calendar, first).year;
558 long long end_year = parts_of(calendar, last).year;
559 long long shifted;
560 long long low = 0, high = end_year - start_year + 2;
561 while (low + 1 < high) {
562 long long middle = (low + high) / 2;
563 if (shift(calendar, first, middle, 0, &shifted) && shifted <= last) low = middle;
564 else high = middle;
565 }
566 long long years = low;
567 low = 0;
568 high = 32;
569 while (low + 1 < high) {
570 long long middle = (low + high) / 2;
571 if (shift(calendar, first, years, middle, &shifted) && shifted <= last) low = middle;
572 else high = middle;
573 }
574 long long months = low;
575 shift(calendar, first, years, months, &shifted);
576 snprintf(text, TEXT_SIZE, "%s%lldy %lldm %lldd", negative ? "-" : "", years, months, last - shifted);
577 return text;
578}

580static char *round_date(Calendar calendar, long long fixed, int number, const char *tz_name, int day, const char *month, char *text) {
581 int target_tz = lookup(tzolkin_names, 20, tz_name);
582 int target_haab = lookup(haab_names, 19, month) * 20 + day;
583 long long candidate = fixed;
584 for (;;) {
585 long long count = candidate + 1137142;
586 if ((count + 3) % 13 + 1 == number && (count + 19) % 20 == target_tz) break;
587 ++candidate;
588 }
589 for (; candidate <= fixed + 18979; candidate += 260) {
590 if ((candidate + 1137142 + 348) % 365 == target_haab) return format_date(calendar, candidate, text);
591 }
592 snprintf(text, TEXT_SIZE, "NONE");
593 return text;
594}

596static bool date_argument(Calendar calendar, const char *text, long long *fixed) {
597 Reason reason = parse_date(calendar, text, fixed);
598 if (reason != ACCEPTED) puts(reason == RANGE ? "OUT OF RANGE" : "INVALID DATE");
599 return reason == ACCEPTED;

602static void execute(Command command, char **args) {
603 char text[TEXT_SIZE], other[TEXT_SIZE];
604 if (!valid_arguments(command, args)) {
605 puts("INVALID COMMAND");
606 return;
607 }
608 if (command == EASTER || command == COMPUTUS) {
609 long long year = atoll(args[0]);
610 if (year > 9999) {
611 puts("OUT OF RANGE");
612 return;
613 }
614 Calendar kind = calendar_id(args[1]);
615 Easter result = easter(year, kind);
616 if (command == EASTER) puts(format_date(GREGORIAN, result.sunday, text));
617 else printf("golden=%lld epact=%lld letters=%s moon=%s easter=%s\n", year % 19 + 1, result.epact, result.letters,
618 format_date(kind, result.moon, text), format_date(kind, result.sunday, other));
619 return;
620 }
621 if (command == PASSOVER || command == HOLIDAYS || command == MOLAD) {
622 long long year = atoll(args[0]);
623 if (command == MOLAD) {
624 Molad result = molad(year);
625 if (!in_range(result.start)) {
626 puts("OUT OF RANGE");
627 return;
628 }
629 printf("%s %lldh %lldp %s %s %s\n", weekday_names[mod(result.day - 1373427, 7)], result.parts / 1080,
630 result.parts % 1080, result.rules[0] != '\0' ? result.rules : "NONE", weekday_names[result.start % 7],
631 format_date(GREGORIAN, result.start, text));
632 return;
633 }
634 if (year > 9999) {
635 puts("OUT OF RANGE");
636 return;
637 }
638 if (command == HOLIDAYS) {
639 print_holidays(year);
640 return;
641 }
642 long long hebrew_year = year + 3760;
643 long long fixed = fixed_date(HEBREW, hebrew_year, month_number(HEBREW, hebrew_year, "Nisan"), 15);
644 printf("%s %s\n", format_date(GREGORIAN, fixed, text), weekday_names[fixed % 7]);
645 return;
646 }
647 Calendar calendar = calendar_id(args[0]);
648 if (command == LEAP || command == MONTHS || command == MONTHCAL || command == NTH || command == NEWYEAR) {
649 long long year = atoll(args[1]);
650 if (command == LEAP) {
651 printf("%s %lld", leap(calendar, year) ? "YES" : "NO", start(calendar, year + 1) - start(calendar, year));
652 if (calendar == HEBREW) printf(" %s", year_type(year));
653 putchar('\n');
654 } else if (command == MONTHS) {
655 Months months = lengths(calendar, year);
656 const char *const *list = names(calendar, year);
657 for (int i = 0; i < months.count; ++i) printf("%s%s=%d", i > 0 ? " " : "", list[i], months.days[i]);
658 putchar('\n');
659 } else if (command == NEWYEAR) {
660 if (year > 9999) {
661 puts("OUT OF RANGE");
662 return;
663 }
664 NewYear found[4];
665 int count = new_years(calendar, start(GREGORIAN, year), start(GREGORIAN, year + 1) - 1, found);
666 if (count == 0) {
667 puts("NONE");
668 return;
669 }
670 for (int i = 0; i < count; ++i) printf("%s%lld:%s", i > 0 ? " " : "", found[i].year, format_date(GREGORIAN, found[i].fixed, text));
671 putchar('\n');
672 } else if (command == MONTHCAL) {
673 print_month_calendar(calendar, year, atoi(args[2]));
674 } else {
675 long long first, last;
676 if (!month_bounds(calendar, year, atoi(args[2]), &first, &last)) {
677 puts("INVALID DATE");
678 return;
679 }
680 int weekday = weekday_id(args[4]);
681 long long fixed;
682 if (strcmp(args[3], "LAST") == 0) fixed = last - mod(last - weekday, 7);
683 else fixed = first + mod(weekday - first, 7) + 7 * (atoi(args[3]) - 1);
684 puts(fixed > last ? "NONE" : format_date(calendar, fixed, text));
685 }
686 return;
687 }
688 if (command == VALID) {
689 long long fixed;
690 Reason reason = parse_date(calendar, args[1], &fixed);
691 if (reason == ACCEPTED) printf("YES %s\n", format_date(calendar, fixed, text));
692 else printf("NO %s\n", reason_names[reason]);
693 return;
694 }
695 long long first, last;
696 if (!date_argument(calendar, args[1], &first)) return;
697 switch (command) {
698 case DIFF:
699 if (date_argument(calendar_id(args[2]), args[3], &last)) printf("%lld\n", last - first);
700 break;
701 case COUNT: {
702 if (!date_argument(calendar, args[2], &last)) break;
703 long long low = smaller(first, last), high = larger(first, last);
704 long long occurrence = low + mod(weekday_id(args[3]) - low, 7);
705 printf("%lld\n", occurrence > high ? 0 : (high - occurrence) / 7 + 1);
706 break;
707 }
708 case SPAN:
709 if (date_argument(calendar, args[2], &last)) puts(span_text(calendar, first, last, text));
710 break;
711 case CONVERT:
712 puts(format_date(calendar_id(args[3]), first, text));
713 break;
714 case WEEKDAY:
715 puts(weekday_names[first % 7]);
716 break;
717 case ALL:
718 for (int target = 0; target < CALENDAR_COUNT; ++target) {
719 format_date(target, first, text);
720 if (is_ymd(target) && strcmp(text, "OUT OF RANGE") != 0) {
721 Date date = parts_of(target, first);
722 printf("%s %s %s\n", calendar_names[target], text, names(target, date.year)[date.month - 1]);
723 } else {
724 printf("%s %s\n", calendar_names[target], text);
725 }
726 }
727 break;
728 case INFO: {
729 Date date = parts_of(calendar, first);
730 const char *weekday = weekday_names[first % 7];
731 if (calendar == FRENCH) weekday = date.month <= 12 ? decade_names[(date.day - 1) % 10] : festival_names[date.day - 1];
732 long long ordinal = first - start(calendar, date.year) + 1;
733 long long length = start(calendar, date.year + 1) - start(calendar, date.year);
734 printf("%s %d %s %lld day=%lld/%lld\n", weekday, date.day, names(calendar, date.year)[date.month - 1], date.year, ordinal, length);
735 break;
736 }
737 case ROUND:
738 puts(round_date(calendar, first, atoi(args[2]), args[3], atoi(args[4]), args[5], text));
739 break;
740 default: {
741 long long amount = atoll(args[2]), result = 0;
742 bool shifted = true;
743 if (strcmp(args[3], "DAYS") == 0) result = first + amount;
744 else if (strcmp(args[3], "WEEKS") == 0) result = first + 7 * amount;
745 else if (strcmp(args[3], "YEARS") == 0) shifted = shift(calendar, first, amount, 0, &result);
746 else shifted = shift(calendar, first, 0, amount, &result);
747 puts(shifted ? format_date(calendar, result, text) : "OUT OF RANGE");
748 }
749 }
750}

752int main(void) {
753 char line[256];
754 if (!fgets(line, sizeof line, stdin)) return 0;
755 int count = atoi(line);
756 for (int i = 0; i < count && fgets(line, sizeof line, stdin); ++i) {
757 line[strcspn(line, "\n")] = '\0';
758 char *words[128];
759 int length = 0;
760 for (char *word = strtok(line, " "); word != NULL; word = strtok(NULL, " ")) words[length++] = word;
761 int command = length > 0 ? lookup(command_names, COMMAND_COUNT, words[0]) : -1;
762 if (command < 0 || length - 1 != arity[command]) puts("INVALID COMMAND");
763 else execute(command, words + 1);
764 }
765 return 0;
