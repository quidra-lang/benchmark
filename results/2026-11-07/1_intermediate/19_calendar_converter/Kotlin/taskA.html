77private fun leap(calendar: Calendar, year: Long): Boolean = when (calendar) {
78 Calendar.GREGORIAN -> year.mod(4) == 0 && (year.mod(100) != 0 || year.mod(400) == 0)
79 Calendar.JULIAN -> year.mod(4) == 0
80 Calendar.ISLAMIC -> year.mod(30) in setOf(2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29)
81 Calendar.HEBREW -> (7 * year + 1).mod(19) < 7
82 Calendar.PERSIAN -> ((year - 474).mod(2820) + 474 + 38) * 682 % 2816 < 682
83 Calendar.COPTIC, Calendar.ETHIOPIC -> year.mod(4) == 3
84 Calendar.FRENCH -> year.mod(4) == 0 && year.mod(400) !in setOf(100, 200, 300) && year.mod(4000) != 0
85 Calendar.ISO -> weeks(year) == 53
86 else -> throw IllegalArgumentException("no years in $calendar")
87}

93private fun molad(year: Long): Molad {
94 val quantity = 5604 + 765433 * monthsBefore(year)
95 val day = quantity.floorDiv(25920)
96 val parts = quantity.mod(25920)
97 val weekday = (day - 1373427).mod(7)
98 val rules = mutableListOf<String>()
99 var postponed = day
100 if (parts >= 19440) {
101 postponed += 1
102 rules.add("ZAKEN")
103 } else if (!leap(Calendar.HEBREW, year) && weekday == 2 && parts >= 9924) {
104 postponed += 2
105 rules.add("GATARAD")
106 } else if (leap(Calendar.HEBREW, year - 1) && weekday == 1 && parts >= 16789) {
107 postponed += 1
108 rules.add("BETUTAKPAT")
109 }
110 if ((postponed - 1373427).mod(7) in setOf(0, 3, 5)) {
111 postponed += 1
112 rules.add("ADU")
113 }
114 return Molad(day, parts, postponed - 1373427, rules)
115}

117private fun start(calendar: Calendar, year: Long): Long {
118 val previous = year - 1
119 return when (calendar) {
120 Calendar.GREGORIAN -> 365 * previous + previous / 4 - previous / 100 + previous / 400 + 1
121 Calendar.JULIAN -> 365 * previous + previous / 4 - 1
122 Calendar.ISLAMIC -> 227015 + 354 * previous + (3 + 11 * year) / 30
123 Calendar.HEBREW -> molad(year).start
124 Calendar.PERSIAN -> {
125 val a = year - 474
126 val b = a.mod(2820) + 474
127 226896 + 1029983 * a.floorDiv(2820) + 365 * (b - 1) + (682 * b - 110) / 2816
128 }
129 Calendar.COPTIC -> 103605 + 365 * previous + year / 4
130 Calendar.ETHIOPIC -> 2796 + 365 * previous + year / 4
131 Calendar.FRENCH -> 654415 + 365 * previous + previous / 4 - previous / 100 + previous / 400 - previous / 4000
132 Calendar.ISO -> {
133 val januaryFour = start(Calendar.GREGORIAN, year) + 3
134 januaryFour - (januaryFour - 1) % 7
135 }
136 else -> throw IllegalArgumentException("no years in $calendar")
137 }
138}

167private fun fixedDate(calendar: Calendar, year: Long, month: Int, day: Int): Long {
168 if (calendar == Calendar.ISO) return start(calendar, year) + 7 * (month - 1) + day - 1
169 return start(calendar, year) + lengths(calendar, year).take(month - 1).sum() + day - 1
170}

172private fun yearOf(calendar: Calendar, fixed: Long): Long {
173 if (fixed < start(calendar, 1)) return 0
174 var low = 1L
175 var high = 100000L
176 while (low + 1 < high) {
177 val middle = (low + high) / 2
178 if (start(calendar, middle) <= fixed) low = middle else high = middle
179 }
180 return low
181}

183private fun partsOf(calendar: Calendar, fixed: Long): Parts? {
184 val year = yearOf(calendar, fixed)
185 if (year == 0L) return null
186 var offset = (fixed - start(calendar, year)).toInt()
187 if (calendar == Calendar.ISO) return Parts(year, offset / 7 + 1, offset % 7 + 1)
188 for ((index, days) in lengths(calendar, year).withIndex()) {
189 if (offset < days) return Parts(year, index + 1, offset + 1)
190 offset -= days
191 }
192 error("calendar year length")
193}

195private fun formatDate(calendar: Calendar, fixed: Long): String {
196 if (!inRange(fixed)) return "OUT OF RANGE"
197 if (calendar in YEAR_CALENDARS) {
198 val (year, month, day) = partsOf(calendar, fixed) ?: return "OUT OF RANGE"
199 val pattern = if (calendar == Calendar.ISO) "%04d-W%02d-%d" else "%04d-%02d-%02d"
200 return pattern.format(Locale.ROOT, year, month, day)
201 }
202 if (calendar == Calendar.JD) return (fixed + 1721425).toString()
203 if (calendar == Calendar.FIXED) return fixed.toString()
204 val count = fixed + 1137142
205 val longCount = listOf(count / 144000, count / 7200 % 20, count / 360 % 20, count / 20 % 18, count % 20).joinToString(".")
206 val haab = (count + 348).mod(365)
207 return "$longCount ${(count + 3) % 13 + 1} ${TZ[(count + 19).mod(20)]} ${haab % 20} ${HB[haab / 20]}"
208}

210private fun parseDate(calendar: Calendar, text: String): Long {
211 val fixed = when (calendar) {
212 in YMD -> {
213 if (!Regex("[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}").matches(text)) throw DateError("FORMAT")
214 val (yearText, monthText, dayText) = text.split("-")
215 val year = yearText.toLong()
216 val month = monthText.toInt()
217 val day = dayText.toInt()
218 if (year == 0L) throw DateError("YEAR")
219 val days = lengths(calendar, year)
220 if (month !in 1..days.size) throw DateError("MONTH")
221 if (day !in 1..days[month - 1]) throw DateError("DAY")
222 fixedDate(calendar, year, month, day)
223 }
224 Calendar.ISO -> {
225 if (!Regex("[0-9]{1,5}-W[0-9]{1,2}-[0-9]").matches(text)) throw DateError("FORMAT")
226 val (yearText, weekText, dayText) = text.split("-")
227 val year = yearText.toLong()
228 val week = weekText.drop(1).toInt()
229 val day = dayText.toInt()
230 if (year == 0L) throw DateError("YEAR")
231 if (week !in 1..weeks(year)) throw DateError("WEEK")
232 if (day !in 1..7) throw DateError("DAY")
233 fixedDate(calendar, year, week, day)
234 }
235 Calendar.MAYAN -> {
236 if (!Regex("[0-9]{1,2}(?:\\.[0-9]{1,2}){4}").matches(text)) throw DateError("FORMAT")
237 val (b, k, t, u, n) = text.split(".").map { it.toLong() }
238 if (k > 19 || t > 19 || u > 17 || n > 19) throw DateError("FIELD")
239 (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142
240 }
241 else -> {
242 if (!Regex("-?[0-9]{1,8}").matches(text)) throw DateError("FORMAT")
243 text.toLong() - (if (calendar == Calendar.JD) 1721425 else 0)
244 }
245 }
246 if (!inRange(fixed)) throw DateError("RANGE")
247 return fixed
248}

253private fun shift(calendar: Calendar, fixed: Long, years: Int, months: Int): Long? {
254 val (year, month, day) = partsOf(calendar, fixed)!!
255 val targetYear = year + years
256 if (calendar == Calendar.ISO) {
257 if (targetYear !in 1L..99999L) return null
258 return fixedDate(calendar, targetYear, minOf(month, weeks(targetYear)), day)
259 }
260 var targetMonth = month
261 if (calendar == Calendar.HEBREW) {
262 var name = names(calendar, year)[month - 1]
263 val newNames = names(calendar, targetYear)
264 if (name !in newNames) name = if (name == "Adar") "AdarII" else "Adar"
265 targetMonth = newNames.indexOf(name) + 1
266 }
267 val absolute = monthIndex(calendar, targetYear) + targetMonth - 1 + months
268 var finalYear: Long
269 if (calendar == Calendar.HEBREW) {
270 finalYear = (19 * absolute + 234).floorDiv(235)
271 while (monthIndex(calendar, finalYear) > absolute) finalYear--
272 while (monthIndex(calendar, finalYear + 1) <= absolute) finalYear++
273 } else {
274 finalYear = absolute.floorDiv(MONTH_NAMES.getValue(calendar).size) + 1
275 }
276 if (finalYear !in 1L..99999L) return null
277 val finalMonth = (absolute - monthIndex(calendar, finalYear) + 1).toInt()
278 val finalDay = minOf(day, lengths(calendar, finalYear)[finalMonth - 1])
279 return fixedDate(calendar, finalYear, finalMonth, finalDay)
280}

282private fun monthBounds(calendar: Calendar, year: Long, month: Int): LongRange {
283 val days = lengths(calendar, year)
284 if (month !in 1..days.size) throw DateError("MONTH")
285 val first = fixedDate(calendar, year, month, 1)
286 return first..first + days[month - 1] - 1

289private fun newYears(calendar: Calendar, first: Long, last: Long): List<Pair<Long, Long>> {
290 var year = maxOf(1, yearOf(calendar, first))
291 while (start(calendar, year) < first) year++
292 val result = mutableListOf<Pair<Long, Long>>()
293 while (start(calendar, year) <= last) {
294 result.add(year to start(calendar, year))
295 year++
296 }
297 return result
298}

300private fun easter(year: Long, calendar: Calendar): Easter {
301 val golden = year.mod(19)
302 var epact: Int
303 if (calendar == Calendar.GREGORIAN) {
304 val century = year / 100 + 1
305 epact = (14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25).mod(30)
306 if (epact == 0 || epact == 1 && golden > 10) epact++
307 } else {
308 epact = (14 + 11 * golden) % 30
309 }
310 val moon = fixedDate(calendar, year, 4, 19) - epact
311 val sunday = moon + 7 - moon % 7
312 val index = (-start(calendar, year)).mod(7)
313 var letters = "ABCDEFG"[index].toString()
314 if (leap(calendar, year)) letters += "ABCDEFG"[(index - 1).mod(7)]
315 return Easter(epact, letters, moon, sunday)
316}

318private fun holidayLines(year: Long): List<String> {
319 val first = start(Calendar.GREGORIAN, year)
320 val last = start(Calendar.GREGORIAN, year + 1) - 1
321 val gregorianEaster = easter(year, Calendar.GREGORIAN).sunday
322 val orthodoxEaster = easter(year, Calendar.JULIAN).sunday
323 val movable = listOf("AshWednesday" to -46, "GoodFriday" to -2, "Easter" to 0, "Ascension" to 39, "Pentecost" to 49)
324 val entries = movable.mapIndexed { order, (name, offset) ->
325 Entry(gregorianEaster + offset, order, name, Calendar.GREGORIAN)
326 }.toMutableList()
327 entries.add(Entry(orthodoxEaster, 5, "OrthodoxEaster", Calendar.JULIAN))
328 for ((index, holiday) in HOLIDAYS.withIndex()) {
329 val calendar = holiday.calendar
330 var candidateYear = maxOf(1, yearOf(calendar, first))

332 var actualName = holiday.month

334 val month = names(calendar, candidateYear).indexOf(actualName) + 1
335 if (holiday.day <= lengths(calendar, candidateYear)[month - 1]) {
336 val fixed = fixedDate(calendar, candidateYear, month, holiday.day)
337 if (fixed in first..last) entries.add(Entry(fixed, index + 6, holiday.name, calendar))
338 }
339 candidateYear++
340 }
341 }
342 return entries.sortedWith(compareBy({ it.fixed }, { it.order })).map {
343 "${formatDate(Calendar.GREGORIAN, it.fixed)} ${weekdayName(it.fixed)} ${it.name} ${it.calendar} ${formatDate(it.calendar, it.fixed)}"

351private fun validateArguments(command: String, args: List<String>) {
352 if (command == "EASTER" || command == "COMPUTUS") {
353 ensure(validYear(args[0]) && args[1] in setOf("GREGORIAN", "JULIAN"))
354 return
355 }
356 if (command in setOf("PASSOVER", "HOLIDAYS", "MOLAD")) {
357 ensure(validYear(args[0]))
358 return
359 }
360 val allowed = when (command) {
361 "INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN" -> YMD
362 "LEAP", "NEWYEAR" -> YEAR_CALENDARS
363 else -> CALENDARS
364 }
365 ensure(isCalendar(args[0], allowed))
366 when (command) {
367 "CONVERT" -> ensure(args[2] == "TO" && isCalendar(args[3], CALENDARS))
368 "DIFF" -> ensure(isCalendar(args[2], CALENDARS))
369 "LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR" -> {
370 ensure(validYear(args[1]))
371 if (command == "MONTHCAL" || command == "NTH") ensure(Regex("[0-9]{1,2}").matches(args[2]))
372 if (command == "NTH") {
373 ensure(args[3] in setOf("1", "2", "3", "4", "5", "LAST"))
374 ensure(args[4] in WEEKDAYS)
375 }
376 }
377 "COUNT" -> ensure(args[3] in WEEKDAYS)
378 "ADD" -> {
379 ensure(Regex("-?[0-9]{1,7}").matches(args[2]))
380 ensure(args[3] in setOf("DAYS", "WEEKS", "MONTHS", "YEARS"))
381 val calendar = Calendar.valueOf(args[0])
382 if (args[3] == "MONTHS") ensure(calendar in YMD)
383 if (args[3] == "YEARS") ensure(calendar in YEAR_CALENDARS)
384 }
385 "ROUND" -> {
386 ensure(Regex("[0-9]{1,2}").matches(args[2]))
387 ensure(args[2].toInt() in 1..13 && args[3] in TZ)
388 ensure(Regex("[0-9]{1,2}").matches(args[4]))
389 ensure(args[5] in HB)
390 ensure(args[4].toInt() <= (if (args[5] == "Uayeb") 4 else 19))
391 }
392 }
393}

395private fun calendarLines(calendar: Calendar, year: Long, month: Int): List<String> {
396 val bounds = monthBounds(calendar, year, month)
397 if (!inRange(bounds.first) || !inRange(bounds.last)) return listOf("OUT OF RANGE")
398 val result = mutableListOf(
399 "${names(calendar, year)[month - 1]} $year " +
400 "(${formatDate(Calendar.GREGORIAN, bounds.first)} to ${formatDate(Calendar.GREGORIAN, bounds.last)})",
401 "Mo Tu We Th Fr Sa Su",
402 )
403 var cursor = bounds.first - (bounds.first - 1) % 7
404 while (cursor <= bounds.last) {
405 val fields = (cursor until cursor + 7).map { if (it in bounds) (it - bounds.first + 1).toString().padStart(2) else " " }
406 result.add(fields.joinToString(" ").trimEnd())
407 cursor += 7
408 }
409 return result
410}

412private fun spanText(calendar: Calendar, first: Long, last: Long): String {
413 val from = minOf(first, last)
414 val to = maxOf(first, last)
415 var low = 0
416 var high = (partsOf(calendar, to)!!.year - partsOf(calendar, from)!!.year + 2).toInt()
417 while (low + 1 < high) {
418 val middle = (low + high) / 2
419 val shifted = shift(calendar, from, middle, 0)
420 if (shifted != null && shifted <= to) low = middle else high = middle
421 }
422 val years = low
423 low = 0
424 high = 32
425 while (low + 1 < high) {
426 val middle = (low + high) / 2
427 val shifted = shift(calendar, from, years, middle)
428 if (shifted != null && shifted <= to) low = middle else high = middle
429 }
430 val months = low
431 val days = to - shift(calendar, from, years, months)!!
432 val sign = if (first > last) "-" else ""
433 return "$sign${years}y ${months}m ${days}d"
434}

436private fun roundDate(calendar: Calendar, fixed: Long, number: Int, tzName: String, day: Int, month: String): String {
437 val targetTz = TZ.indexOf(tzName)
438 val targetHaab = HB.indexOf(month) * 20 + day
439 var candidate = fixed
440 while ((candidate + 1137142 + 3).mod(13) + 1 != number || (candidate + 1137142 + 19).mod(20) != targetTz) candidate++
441 while (candidate <= fixed + 18979) {
442 if ((candidate + 1137142 + 348).mod(365) == targetHaab) return formatDate(calendar, candidate)
443 candidate += 260
444 }
445 return "NONE"
446}

448private fun execute(command: String, args: List<String>): List<String> {
449 validateArguments(command, args)
450 if (command == "EASTER" || command == "COMPUTUS") {
451 val year = args[0].toLong()
452 if (year > 9999) return listOf("OUT OF RANGE")
453 val kind = Calendar.valueOf(args[1])
454 val (epact, letters, moon, sunday) = easter(year, kind)
455 if (command == "EASTER") return listOf(formatDate(Calendar.GREGORIAN, sunday))
456 return listOf(
457 "golden=${year % 19 + 1} epact=$epact letters=$letters " +
458 "moon=${formatDate(kind, moon)} easter=${formatDate(kind, sunday)}"
459 )
460 }
461 if (command in setOf("PASSOVER", "HOLIDAYS", "MOLAD")) {
462 val year = args[0].toLong()
463 if (command == "MOLAD") {
464 val (day, parts, fixed, rules) = molad(year)
465 if (!inRange(fixed)) return listOf("OUT OF RANGE")
466 val applied = if (rules.isEmpty()) "NONE" else rules.joinToString("+")
467 return listOf(
468 "${weekdayName(day - 1373427)} ${parts / 1080}h ${parts % 1080}p $applied " +
469 "${weekdayName(fixed)} ${formatDate(Calendar.GREGORIAN, fixed)}"
470 )
471 }
472 if (year > 9999) return listOf("OUT OF RANGE")
473 if (command == "HOLIDAYS") return holidayLines(year)
474 val hebrewYear = year + 3760
475 val month = names(Calendar.HEBREW, hebrewYear).indexOf("Nisan") + 1
476 val fixed = fixedDate(Calendar.HEBREW, hebrewYear, month, 15)
477 return listOf("${formatDate(Calendar.GREGORIAN, fixed)} ${weekdayName(fixed)}")
478 }
479 val calendar = Calendar.valueOf(args[0])
480 if (command in setOf("LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR")) {
481 val year = args[1].toLong()
482 return when (command) {
483 "LEAP" -> {
484 var text = "${if (leap(calendar, year)) "YES" else "NO"} ${start(calendar, year + 1) - start(calendar, year)}"
485 if (calendar == Calendar.HEBREW) text += " ${yearType(year)}"
486 listOf(text)
487 }
488 "MONTHS" -> listOf(names(calendar, year).zip(lengths(calendar, year)).joinToString(" ") { (name, days) -> "$name=$days" })
489 "NEWYEAR" -> {
490 if (year > 9999) return listOf("OUT OF RANGE")
491 val entries = newYears(calendar, start(Calendar.GREGORIAN, year), start(Calendar.GREGORIAN, year + 1) - 1)
492 listOf(entries.joinToString(" ") { (y, f) -> "$y:${formatDate(Calendar.GREGORIAN, f)}" }.ifEmpty { "NONE" })
493 }
494 "MONTHCAL" -> calendarLines(calendar, year, args[2].toInt())
495 else -> {
496 val bounds = monthBounds(calendar, year, args[2].toInt())
497 val weekday = WEEKDAYS.indexOf(args[4])
498 val fixed = if (args[3] == "LAST") bounds.last - (bounds.last - weekday).mod(7)
499 else bounds.first + (weekday - bounds.first).mod(7) + 7 * (args[3].toInt() - 1)
500 listOf(if (fixed > bounds.last) "NONE" else formatDate(calendar, fixed))
501 }
502 }
503 }
504 if (command == "VALID") {
505 return try {
506 listOf("YES " + formatDate(calendar, parseDate(calendar, args[1])))
507 } catch (error: DateError) {
508 listOf("NO " + error.reason)
509 }
510 }
511 val first = parseDate(calendar, args[1])
512 return when (command) {
513 "DIFF" -> listOf((parseDate(Calendar.valueOf(args[2]), args[3]) - first).toString())
514 "COUNT" -> {
515 val last = parseDate(calendar, args[2])
516 val low = minOf(first, last)
517 val high = maxOf(first, last)
518 val occurrence = low + (WEEKDAYS.indexOf(args[3]) - low).mod(7)
519 listOf((if (occurrence > high) 0 else (high - occurrence) / 7 + 1).toString())
520 }
521 "SPAN" -> listOf(spanText(calendar, first, parseDate(calendar, args[2])))
522 "CONVERT" -> listOf(formatDate(Calendar.valueOf(args[3]), first))
523 "WEEKDAY" -> listOf(weekdayName(first))
524 "ALL" -> Calendar.entries.map { target ->
525 var text = formatDate(target, first)
526 if (target in YMD && text != "OUT OF RANGE") {
527 val parts = partsOf(target, first)!!
528 text += " " + names(target, parts.year)[parts.month - 1]
529 }
530 "$target $text"
531 }
532 "INFO" -> {
533 val (year, month, day) = partsOf(calendar, first)!!
534 var weekday = weekdayName(first)
535 if (calendar == Calendar.FRENCH) weekday = if (month <= 12) DC[(day - 1) % 10] else FS[day - 1]
536 val ordinal = first - start(calendar, year) + 1
537 val length = start(calendar, year + 1) - start(calendar, year)
538 listOf("$weekday $day ${names(calendar, year)[month - 1]} $year day=$ordinal/$length")
539 }
540 "ROUND" -> listOf(roundDate(calendar, first, args[2].toInt(), args[3], args[4].toInt(), args[5]))
541 else -> {
542 val amount = args[2].toInt()
543 val result = when (args[3]) {
544 "DAYS" -> first + amount
545 "WEEKS" -> first + 7L * amount
546 "YEARS" -> shift(calendar, first, amount, 0)
547 else -> shift(calendar, first, 0, amount)
548 }
549 listOf(if (result == null) "OUT OF RANGE" else formatDate(calendar, result))
550 }
551 }
552}

554fun main() {
555 val input = System.`in`.bufferedReader()
556 val output = System.out.bufferedWriter()
557 repeat(input.readLine().toInt()) {
558 val words = input.readLine().split(" ")
559 val command = words[0]
560 val args = words.drop(1)
561 val lines = if (ARITY[command] != args.size) {
562 listOf("INVALID COMMAND")
563 } else {
564 try {
565 execute(command, args)
566 } catch (error: CommandError) {
567 listOf("INVALID COMMAND")
568 } catch (error: DateError) {
569 listOf(if (error.reason == "RANGE") "OUT OF RANGE" else "INVALID DATE")
570 }
571 }
572 for (line in lines) {
573 output.write(line)
574 output.newLine()
575 }
576 }
577 output.flush()
578}
