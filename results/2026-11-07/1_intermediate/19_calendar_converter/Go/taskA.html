118func leap(calendar Calendar, year int64) bool {
119 switch calendar {
120 case Gregorian:
121 return mod(year, 4) == 0 && (mod(year, 100) != 0 || mod(year, 400) == 0)
122 case Julian:
123 return mod(year, 4) == 0
124 case Islamic:
125 return slices.Contains(islamicLeapYears, mod(year, 30))
126 case Hebrew:
127 return mod(7*year+1, 19) < 7
128 case Persian:
129 b := mod(year-474, 2820) + 474
130 return mod((b+38)*682, 2816) < 682
131 case Coptic, Ethiopic:
132 return mod(year, 4) == 3
133 case French:
134 century := mod(year, 400)
135 return mod(year, 4) == 0 && century != 100 && century != 200 && century != 300 && mod(year, 4000) != 0
136 }
137 return (start(ISO, year+1)-start(ISO, year))/7 == 53
138}

142func molad(year int64) (day, parts, fixed int64, rules []string) {
143 quantity := 5604 + 765433*monthsBefore(year)
144 day, parts = floorDiv(quantity, 25920), mod(quantity, 25920)
145 weekday := mod(day-1373427, 7)
146 postponed := day
147 switch {
148 case parts >= 19440:
149 postponed++
150 rules = append(rules, "ZAKEN")
151 case !leap(Hebrew, year) && weekday == 2 && parts >= 9924:
152 postponed += 2
153 rules = append(rules, "GATARAD")
154 case leap(Hebrew, year-1) && weekday == 1 && parts >= 16789:
155 postponed++
156 rules = append(rules, "BETUTAKPAT")
157 }
158 if postponedWeekday := mod(postponed-1373427, 7); postponedWeekday == 0 || postponedWeekday == 3 || postponedWeekday == 5 {
159 postponed++
160 rules = append(rules, "ADU")
161 }
162 return day, parts, postponed - 1373427, rules
163}

165func start(calendar Calendar, year int64) int64 {
166 previous := year - 1
167 switch calendar {
168 case Gregorian:
169 return 365*previous + floorDiv(previous, 4) - floorDiv(previous, 100) + floorDiv(previous, 400) + 1
170 case Julian:
171 return 365*previous + floorDiv(previous, 4) - 1
172 case Islamic:
173 return 227015 + 354*previous + floorDiv(3+11*year, 30)
174 case Hebrew:
175 _, _, fixed, _ := molad(year)
176 return fixed
177 case Persian:
178 a := year - 474
179 b := mod(a, 2820) + 474
180 return 226896 + 1029983*floorDiv(a, 2820) + 365*(b-1) + floorDiv(682*b-110, 2816)
181 case Coptic:
182 return 103605 + 365*previous + floorDiv(year, 4)
183 case Ethiopic:
184 return 2796 + 365*previous + floorDiv(year, 4)
185 case French:
186 return 654415 + 365*previous + floorDiv(previous, 4) - floorDiv(previous, 100) + floorDiv(previous, 400) - floorDiv(previous, 4000)
187 }
188 januaryFour := start(Gregorian, year) + 3
189 return januaryFour - mod(januaryFour-1, 7)
190}

238func fixedDate(calendar Calendar, year int64, month, day int) int64 {
239 if calendar == ISO {
240 return start(calendar, year) + int64(7*(month-1)+day-1)
241 }
242 fixed := start(calendar, year) + int64(day-1)
243 for _, days := range lengths(calendar, year)[:month-1] {
244 fixed += int64(days)
245 }
246 return fixed
247}

249func yearOf(calendar Calendar, fixed int64) int64 {
250 if fixed < start(calendar, 1) {
251 return 0
252 }
253 low, high := int64(1), int64(100000)
254 for low+1 < high {
255 middle := (low + high) / 2
256 if start(calendar, middle) <= fixed {
257 low = middle
258 } else {
259 high = middle
260 }
261 }
262 return low
263}

265func partsOf(calendar Calendar, fixed int64) (year int64, month, day int, ok bool) {
266 year = yearOf(calendar, fixed)
267 if year == 0 {
268 return 0, 0, 0, false
269 }
270 offset := fixed - start(calendar, year)
271 if calendar == ISO {
272 return year, int(offset/7) + 1, int(offset%7) + 1, true
273 }
274 for index, days := range lengths(calendar, year) {
275 if offset < int64(days) {
276 return year, index + 1, int(offset) + 1, true
277 }
278 offset -= int64(days)
279 }
280 panic("calendar year length")
281}

283func formatDate(calendar Calendar, fixed int64) string {
284 if !inRange(fixed) {
285 return "OUT OF RANGE"
286 }
287 if hasYears(calendar) {
288 year, month, day, ok := partsOf(calendar, fixed)
289 if !ok {
290 return "OUT OF RANGE"
291 }
292 if calendar == ISO {
293 return fmt.Sprintf("%04d-W%02d-%d", year, month, day)
294 }
295 return fmt.Sprintf("%04d-%02d-%02d", year, month, day)
296 }
297 switch calendar {
298 case JD:
299 return strconv.FormatInt(fixed+1721425, 10)
300 case Fixed:
301 return strconv.FormatInt(fixed, 10)
302 }
303 count := fixed + 1137142
304 haabDay := (count + 348) % 365
305 return fmt.Sprintf("%d.%d.%d.%d.%d %d %s %d %s", count/144000, count/7200%20, count/360%20, count/20%18, count%20,
306 (count+3)%13+1, tzolkin[(count+19)%20], haabDay%20, haab[haabDay/20])
307}

309func parseDate(calendar Calendar, text string) (int64, error) {
310 var fixed int64
311 switch {
312 case isYMD(calendar):
313 match := ymdPattern.FindStringSubmatch(text)
314 if match == nil {
315 return 0, dateError("FORMAT")
316 }
317 year, month, day := int64(number(match[1])), number(match[2]), number(match[3])
318 if year == 0 {
319 return 0, dateError("YEAR")
320 }
321 days := lengths(calendar, year)
322 if month < 1 || month > len(days) {
323 return 0, dateError("MONTH")
324 }
325 if day < 1 || day > days[month-1] {
326 return 0, dateError("DAY")
327 }
328 fixed = fixedDate(calendar, year, month, day)
329 case calendar == ISO:
330 match := isoPattern.FindStringSubmatch(text)
331 if match == nil {
332 return 0, dateError("FORMAT")
333 }
334 year, week, day := int64(number(match[1])), number(match[2]), number(match[3])
335 if year == 0 {
336 return 0, dateError("YEAR")
337 }
338 weeks := int((start(calendar, year+1) - start(calendar, year)) / 7)
339 if week < 1 || week > weeks {
340 return 0, dateError("WEEK")
341 }
342 if day < 1 || day > 7 {
343 return 0, dateError("DAY")
344 }
345 fixed = fixedDate(calendar, year, week, day)
346 case calendar == Mayan:
347 match := mayanPattern.FindStringSubmatch(text)
348 if match == nil {
349 return 0, dateError("FORMAT")
350 }
351 b, k, t, u, n := int64(number(match[1])), int64(number(match[2])), int64(number(match[3])), int64(number(match[4])), int64(number(match[5]))
352 if k > 19 || t > 19 || u > 17 || n > 19 {
353 return 0, dateError("FIELD")
354 }
355 fixed = (((b*20+k)*20+t)*18+u)*20 + n - 1137142
356 default:
357 if !integerPattern.MatchString(text) {
358 return 0, dateError("FORMAT")
359 }
360 fixed = int64(number(text))
361 if calendar == JD {
362 fixed -= 1721425
363 }
364 }
365 if !inRange(fixed) {
366 return 0, dateError("RANGE")
367 }
368 return fixed, nil
369}

372 if calendar == Hebrew {

377 count = 13

382func shift(calendar Calendar, fixed, years, months int64) (int64, bool) {
383 year, month, day, _ := partsOf(calendar, fixed)
384 targetYear := year + years
385 if calendar == ISO {
386 if targetYear < 1 || targetYear > 99999 {
387 return 0, false
388 }
389 weeks := int((start(calendar, targetYear+1) - start(calendar, targetYear)) / 7)
390 return fixedDate(calendar, targetYear, min(month, weeks), day), true
391 }
392 if calendar == Hebrew {
393 oldName := names(calendar, year)[month-1]
394 newNames := names(calendar, targetYear)
395 if !slices.Contains(newNames, oldName) {
396 if oldName == "Adar" {
397 oldName = "AdarII"
398 } else {
399 oldName = "Adar"
400 }
401 }
402 month = slices.Index(newNames, oldName) + 1
403 }
404 absolute := monthIndex(calendar, targetYear) + int64(month-1) + months
405 var finalYear int64
406 if calendar == Hebrew {
407 finalYear = floorDiv(19*absolute+234, 235)
408 for monthIndex(calendar, finalYear) > absolute {
409 finalYear--
410 }
411 for monthIndex(calendar, finalYear+1) <= absolute {
412 finalYear++
413 }
414 } else {
415 finalYear = floorDiv(absolute, int64(len(names(calendar, targetYear)))) + 1
416 }
417 if finalYear < 1 || finalYear > 99999 {
418 return 0, false
419 }
420 finalMonth := int(absolute-monthIndex(calendar, finalYear)) + 1
421 finalDay := min(day, lengths(calendar, finalYear)[finalMonth-1])
422 return fixedDate(calendar, finalYear, finalMonth, finalDay), true
423}

425func monthBounds(calendar Calendar, year int64, month int) (int64, int64, error) {
426 days := lengths(calendar, year)
427 if month < 1 || month > len(days) {
428 return 0, 0, dateError("MONTH")
429 }
430 first := fixedDate(calendar, year, month, 1)
431 return first, first + int64(days[month-1]) - 1, nil
432}

446func easter(year int64, calendar Calendar) (epact int64, letters string, moon, sunday int64) {
447 golden := year % 19
448 if calendar == Gregorian {
449 century := year/100 + 1
450 epact = mod(14+11*golden-3*century/4+(5+8*century)/25, 30)
451 if epact == 0 || (epact == 1 && golden > 10) {
452 epact++
453 }
454 } else {
455 epact = (14 + 11*golden) % 30
456 }
457 moon = fixedDate(calendar, year, 4, 19) - epact
458 sunday = moon + 7 - mod(moon, 7)
459 const alphabet = "ABCDEFG"
460 index := mod(-start(calendar, year), 7)
461 letters = alphabet[index : index+1]
462 if leap(calendar, year) {
463 previous := mod(index-1, 7)
464 letters += alphabet[previous : previous+1]
465 }
466 return epact, letters, moon, sunday
467}

497func holidayLines(year int64) []string {
498 first := start(Gregorian, year)
499 last := start(Gregorian, year+1) - 1
500 _, _, _, gregorianEaster := easter(year, Gregorian)
501 _, _, _, orthodoxEaster := easter(year, Julian)
502 movable := []struct {
503 name string
504 offset int64
505 }{
506 {"AshWednesday", -46}, {"GoodFriday", -2}, {"Easter", 0}, {"Ascension", 39}, {"Pentecost", 49},
507 }
508 var entries []entry
509 for order, feast := range movable {
510 entries = append(entries, entry{gregorianEaster + feast.offset, order, feast.name, Gregorian})
511 }
512 entries = append(entries, entry{orthodoxEaster, 5, "OrthodoxEaster", Julian})
513 for index, h := range holidays {
514 for candidate := max(1, yearOf(h.calendar, first)); start(h.calendar, candidate) <= last; candidate++ {
515 actualName := h.month
516 if h.calendar == Hebrew && actualName == "Adar" && leap(h.calendar, candidate) {
517 actualName = "AdarII"
518 }
519 month := slices.Index(names(h.calendar, candidate), actualName) + 1
520 if h.day > lengths(h.calendar, candidate)[month-1] {
521 continue
522 }
523 fixed := fixedDate(h.calendar, candidate, month, h.day)
524 if first <= fixed && fixed <= last {
525 entries = append(entries, entry{fixed, index + 6, h.name, h.calendar})
526 }
527 }
528 }
529 slices.SortFunc(entries, func(a, b entry) int {
530 return cmp.Or(cmp.Compare(a.fixed, b.fixed), cmp.Compare(a.order, b.order))
531 })
532 lines := make([]string, 0, len(entries))
533 for _, e := range entries {
534 lines = append(lines, fmt.Sprintf("%s %s %s %s %s", formatDate(Gregorian, e.fixed), weekdays[e.fixed%7],
535 e.name, calendarNames[e.calendar], formatDate(e.calendar, e.fixed)))
536 }
537 return lines
538}

549func validArguments(command string, args []string) bool {
550 switch command {
551 case "EASTER", "COMPUTUS":
552 return validYear(args[0]) && (args[1] == "GREGORIAN" || args[1] == "JULIAN")
553 case "PASSOVER", "HOLIDAYS", "MOLAD":
554 return validYear(args[0])
555 }
556 if !slices.Contains(calendarNames, args[0]) {
557 return false
558 }
559 calendar := calendarOf(args[0])
560 switch command {
561 case "INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN":
562 if !isYMD(calendar) {
563 return false
564 }
565 case "LEAP", "NEWYEAR":
566 if !hasYears(calendar) {
567 return false
568 }
569 }
570 switch command {
571 case "CONVERT":
572 return args[2] == "TO" && slices.Contains(calendarNames, args[3])
573 case "DIFF":
574 return slices.Contains(calendarNames, args[2])
575 case "LEAP", "MONTHS", "NEWYEAR":
576 return validYear(args[1])
577 case "MONTHCAL":
578 return validYear(args[1]) && twoDigitPattern.MatchString(args[2])
579 case "NTH":
580 return validYear(args[1]) && twoDigitPattern.MatchString(args[2]) &&
581 slices.Contains([]string{"1", "2", "3", "4", "5", "LAST"}, args[3]) && slices.Contains(weekdays, args[4])
582 case "COUNT":
583 return slices.Contains(weekdays, args[3])
584 case "ADD":
585 if !amountPattern.MatchString(args[2]) || !slices.Contains([]string{"DAYS", "WEEKS", "MONTHS", "YEARS"}, args[3]) {
586 return false
587 }
588 switch args[3] {
589 case "MONTHS":
590 return isYMD(calendar)
591 case "YEARS":
592 return hasYears(calendar)
593 }
594 case "ROUND":
595 if !twoDigitPattern.MatchString(args[2]) || !twoDigitPattern.MatchString(args[4]) {
596 return false
597 }
598 limit := 19
599 if args[5] == "Uayeb" {
600 limit = 4
601 }
602 value := number(args[2])
603 return 1 <= value && value <= 13 && slices.Contains(tzolkin, args[3]) && slices.Contains(haab, args[5]) && number(args[4]) <= limit
604 }
605 return true
606}

608func calendarLines(calendar Calendar, year int64, month int) ([]string, error) {
609 first, last, err := monthBounds(calendar, year, month)
610 if err != nil {
611 return nil, err
612 }
613 if !inRange(first) || !inRange(last) {
614 return []string{"OUT OF RANGE"}, nil
615 }
616 lines := []string{
617 fmt.Sprintf("%s %d (%s to %s)", names(calendar, year)[month-1], year, formatDate(Gregorian, first), formatDate(Gregorian, last)),
618 "Mo Tu We Th Fr Sa Su",
619 }
620 for cursor := first - (first-1)%7; cursor <= last; cursor += 7 {
621 fields := make([]string, 0, 7)
622 for fixed := cursor; fixed < cursor+7; fixed++ {
623 if first <= fixed && fixed <= last {
624 fields = append(fields, fmt.Sprintf("%2d", fixed-first+1))
625 } else {
626 fields = append(fields, " ")
627 }
628 }
629 lines = append(lines, strings.TrimRight(strings.Join(fields, " "), " "))
630 }
631 return lines, nil
632}

634func spanText(calendar Calendar, first, last int64) string {
635 negative := first > last
636 if negative {
637 first, last = last, first
638 }
639 startYear, _, _, _ := partsOf(calendar, first)
640 endYear, _, _, _ := partsOf(calendar, last)
641 low, high := int64(0), endYear-startYear+2
642 for low+1 < high {
643 middle := (low + high) / 2
644 if shifted, ok := shift(calendar, first, middle, 0); ok && shifted <= last {
645 low = middle
646 } else {
647 high = middle
648 }
649 }
650 years := low
651 low, high = 0, 32
652 for low+1 < high {
653 middle := (low + high) / 2
654 if shifted, ok := shift(calendar, first, years, middle); ok && shifted <= last {
655 low = middle
656 } else {
657 high = middle
658 }
659 }
660 months := low
661 shifted, _ := shift(calendar, first, years, months)
662 sign := ""
663 if negative {
664 sign = "-"
665 }
666 return fmt.Sprintf("%s%dy %dm %dd", sign, years, months, last-shifted)
667}

669func roundDate(calendar Calendar, fixed, tzNumber int64, tzName string, day int64, month string) string {
670 targetTz := int64(slices.Index(tzolkin, tzName))
671 targetHaab := int64(slices.Index(haab, month))*20 + day
672 candidate := fixed
673 for {
674 count := candidate + 1137142
675 if (count+3)%13+1 == tzNumber && (count+19)%20 == targetTz {
676 break
677 }
678 candidate++
679 }
680 for ; candidate <= fixed+18979; candidate += 260 {
681 if (candidate+1137142+348)%365 == targetHaab {
682 return formatDate(calendar, candidate)
683 }
684 }
685 return "NONE"
686}

688func execute(command string, args []string) ([]string, error) {
689 if !validArguments(command, args) {
690 return nil, errCommand
691 }
692 switch command {
693 case "EASTER", "COMPUTUS":
694 year := int64(number(args[0]))
695 if year > 9999 {
696 return []string{"OUT OF RANGE"}, nil
697 }
698 kind := calendarOf(args[1])
699 epact, letters, moon, sunday := easter(year, kind)
700 if command == "EASTER" {
701 return []string{formatDate(Gregorian, sunday)}, nil
702 }
703 return []string{fmt.Sprintf("golden=%d epact=%d letters=%s moon=%s easter=%s",
704 year%19+1, epact, letters, formatDate(kind, moon), formatDate(kind, sunday))}, nil
705 case "MOLAD":
706 day, parts, fixed, rules := molad(int64(number(args[0])))
707 if !inRange(fixed) {
708 return []string{"OUT OF RANGE"}, nil
709 }
710 applied := "NONE"
711 if len(rules) > 0 {
712 applied = strings.Join(rules, "+")
713 }
714 return []string{fmt.Sprintf("%s %dh %dp %s %s %s", weekdays[mod(day-1373427, 7)], parts/1080, parts%1080,
715 applied, weekdays[fixed%7], formatDate(Gregorian, fixed))}, nil
716 case "PASSOVER", "HOLIDAYS":
717 year := int64(number(args[0]))
718 if year > 9999 {
719 return []string{"OUT OF RANGE"}, nil
720 }
721 if command == "HOLIDAYS" {
722 return holidayLines(year), nil
723 }
724 hebrewYear := year + 3760
725 month := slices.Index(names(Hebrew, hebrewYear), "Nisan") + 1
726 fixed := fixedDate(Hebrew, hebrewYear, month, 15)
727 return []string{formatDate(Gregorian, fixed) + " " + weekdays[fixed%7]}, nil
728 }
729 calendar := calendarOf(args[0])
730 switch command {
731 case "LEAP":
732 year := int64(number(args[1]))
733 answer := "NO"
734 if leap(calendar, year) {
735 answer = "YES"
736 }
737 line := fmt.Sprintf("%s %d", answer, start(calendar, year+1)-start(calendar, year))
738 if calendar == Hebrew {
739 line += " " + yearType(year)
740 }
741 return []string{line}, nil
742 case "MONTHS":
743 year := int64(number(args[1]))
744 monthList := names(calendar, year)
745 var items []string
746 for index, days := range lengths(calendar, year) {
747 items = append(items, fmt.Sprintf("%s=%d", monthList[index], days))
748 }
749 return []string{strings.Join(items, " ")}, nil
750 case "NEWYEAR":
751 year := int64(number(args[1]))
752 if year > 9999 {
753 return []string{"OUT OF RANGE"}, nil
754 }
755 var items []string
756 for _, found := range newYears(calendar, start(Gregorian, year), start(Gregorian, year+1)-1) {
757 items = append(items, fmt.Sprintf("%d:%s", found.year, formatDate(Gregorian, found.fixed)))
758 }
759 if len(items) == 0 {
760 return []string{"NONE"}, nil
761 }
762 return []string{strings.Join(items, " ")}, nil
763 case "MONTHCAL":
764 return calendarLines(calendar, int64(number(args[1])), number(args[2]))
765 case "NTH":
766 first, last, err := monthBounds(calendar, int64(number(args[1])), number(args[2]))
767 if err != nil {
768 return nil, err
769 }
770 weekday := int64(slices.Index(weekdays, args[4]))
771 var fixed int64
772 if args[3] == "LAST" {
773 fixed = last - mod(last-weekday, 7)
774 } else {
775 fixed = first + mod(weekday-first, 7) + 7*int64(number(args[3])-1)
776 }
777 if fixed > last {
778 return []string{"NONE"}, nil
779 }
780 return []string{formatDate(calendar, fixed)}, nil
781 case "VALID":
782 fixed, err := parseDate(calendar, args[1])
783 if err != nil {
784 return []string{"NO " + err.Error()}, nil
785 }
786 return []string{"YES " + formatDate(calendar, fixed)}, nil
787 }
788 first, err := parseDate(calendar, args[1])
789 if err != nil {
790 return nil, err
791 }
792 switch command {
793 case "DIFF":
794 last, err := parseDate(calendarOf(args[2]), args[3])
795 if err != nil {
796 return nil, err
797 }
798 return []string{strconv.FormatInt(last-first, 10)}, nil
799 case "COUNT", "SPAN":
800 last, err := parseDate(calendar, args[2])
801 if err != nil {
802 return nil, err
803 }
804 if command == "SPAN" {
805 return []string{spanText(calendar, first, last)}, nil
806 }
807 low, high := min(first, last), max(first, last)
808 occurrence := low + mod(int64(slices.Index(weekdays, args[3]))-low, 7)
809 total := int64(0)
810 if occurrence <= high {
811 total = (high-occurrence)/7 + 1
812 }
813 return []string{strconv.FormatInt(total, 10)}, nil
814 case "CONVERT":
815 return []string{formatDate(calendarOf(args[3]), first)}, nil
816 case "WEEKDAY":
817 return []string{weekdays[first%7]}, nil
818 case "ALL":
819 lines := make([]string, 0, len(calendarNames))
820 for index, name := range calendarNames {
821 target := Calendar(index)
822 text := formatDate(target, first)
823 if isYMD(target) && text != "OUT OF RANGE" {
824 year, month, _, _ := partsOf(target, first)
825 text += " " + names(target, year)[month-1]
826 }
827 lines = append(lines, name+" "+text)
828 }
829 return lines, nil
830 case "INFO":
831 year, month, day, _ := partsOf(calendar, first)
832 weekday := weekdays[first%7]
833 if calendar == French {
834 if month <= 12 {
835 weekday = decades[(day-1)%10]
836 } else {
837 weekday = festivals[day-1]
838 }
839 }
840 ordinal := first - start(calendar, year) + 1
841 length := start(calendar, year+1) - start(calendar, year)
842 return []string{fmt.Sprintf("%s %d %s %d day=%d/%d", weekday, day, names(calendar, year)[month-1], year, ordinal, length)}, nil
843 case "ROUND":
844 return []string{roundDate(calendar, first, int64(number(args[2])), args[3], int64(number(args[4])), args[5])}, nil
845 }
846 amount := int64(number(args[2]))
847 var result int64
848 ok := true
849 switch args[3] {
850 case "DAYS":
851 result = first + amount
852 case "WEEKS":
853 result = first + 7*amount
854 case "YEARS":
855 result, ok = shift(calendar, first, amount, 0)
856 default:
857 result, ok = shift(calendar, first, 0, amount)
858 }
859 if !ok {
860 return []string{"OUT OF RANGE"}, nil
861 }
862 return []string{formatDate(calendar, result)}, nil
863}

865func main() {
866 scanner := bufio.NewScanner(os.Stdin)
867 out := bufio.NewWriter(os.Stdout)
868 defer out.Flush()
869 scanner.Scan()
870 count := number(scanner.Text())
871 for i := 0; i < count && scanner.Scan(); i++ {
872 words := strings.Split(scanner.Text(), " ")
873 command, args := words[0], words[1:]
874 if expected, known := arity[command]; !known || len(args) != expected {
875 fmt.Fprintln(out, "INVALID COMMAND")
876 continue
877 }
878 lines, err := execute(command, args)
879 var reason dateError
880 switch {
881 case errors.As(err, &reason) && reason == "RANGE":
882 lines = []string{"OUT OF RANGE"}
883 case errors.As(err, &reason):
884 lines = []string{"INVALID DATE"}
885 case err != nil:
886 lines = []string{"INVALID COMMAND"}
887 }
888 for _, line := range lines {
889 fmt.Fprintln(out, line)
890 }
891 }
892}
