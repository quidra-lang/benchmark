93function leap(string $calendar, int $year): bool
94{
95 if ($calendar === 'GREGORIAN') {
96 return $year % 4 === 0 && ($year % 100 !== 0 || $year % 400 === 0);
97 }
98 if ($calendar === 'JULIAN') {
99 return $year % 4 === 0;
100 }
101 if ($calendar === 'ISLAMIC') {
102 return in_array(positive_mod($year, 30), [2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29], true);
103 }
104 if ($calendar === 'HEBREW') {
105 return positive_mod(7 * $year + 1, 19) < 7;
106 }
107 if ($calendar === 'PERSIAN') {
108 $b = positive_mod($year - 474, 2820) + 474;
109 return ($b + 38) * 682 % 2816 < 682;
110 }
111 if ($calendar === 'COPTIC' || $calendar === 'ETHIOPIC') {
112 return positive_mod($year, 4) === 3;
113 }
114 if ($calendar === 'FRENCH') {
115 return $year % 4 === 0 && !in_array(positive_mod($year, 400), [100, 200, 300], true) && $year % 4000 !== 0;
116 }
117 return intdiv(start('ISO', $year + 1) - start('ISO', $year), 7) === 53;
118}

150function start(string $calendar, int $year): int
151{
152 $previous = $year - 1;
153 if ($calendar === 'GREGORIAN') {
154 return 365 * $previous + floor_div($previous, 4) - floor_div($previous, 100) + floor_div($previous, 400) + 1;
155 }
156 if ($calendar === 'JULIAN') {
157 return 365 * $previous + floor_div($previous, 4) - 1;
158 }
159 if ($calendar === 'ISLAMIC') {
160 return 227015 + 354 * $previous + floor_div(3 + 11 * $year, 30);
161 }
162 if ($calendar === 'HEBREW') {
163 return molad($year)[2];
164 }
165 if ($calendar === 'PERSIAN') {
166 $a = $year - 474;
167 $b = positive_mod($a, 2820) + 474;
168 return 226896 + 1029983 * floor_div($a, 2820) + 365 * ($b - 1) + floor_div(682 * $b - 110, 2816);
169 }
170 if ($calendar === 'COPTIC') {
171 return 103605 + 365 * $previous + floor_div($year, 4);
172 }
173 if ($calendar === 'ETHIOPIC') {
174 return 2796 + 365 * $previous + floor_div($year, 4);
175 }
176 if ($calendar === 'FRENCH') {
177 return 654415 + 365 * $previous + floor_div($previous, 4) - floor_div($previous, 100)
178 + floor_div($previous, 400) - floor_div($previous, 4000);
179 }
180 $januaryFour = start('GREGORIAN', $year) + 3;
181 return $januaryFour - positive_mod($januaryFour - 1, 7);
182}

226{
227 if ($calendar === 'ISO') {
228 return start($calendar, $year) + 7 * ($month - 1) + $day - 1;

233function year_of(string $calendar, int $fixed): int
234{
235 if ($fixed < start($calendar, 1)) {
236 return 0;
237 }
238 $low = 1;
239 $high = 100000;
240 while ($low + 1 < $high) {
241 $middle = intdiv($low + $high, 2);
242 if (start($calendar, $middle) <= $fixed) {
243 $low = $middle;
244 } else {
245 $high = $middle;
246 }
247 }
248 return $low;
249}

251function parts_of(string $calendar, int $fixed): ?array
252{
253 $year = year_of($calendar, $fixed);
254 if ($year === 0) {
255 return null;
256 }
257 $offset = $fixed - start($calendar, $year);
258 if ($calendar === 'ISO') {
259 return [$year, intdiv($offset, 7) + 1, $offset % 7 + 1];
260 }
261 foreach (lengths($calendar, $year) as $index => $days) {
262 if ($offset < $days) {
263 return [$year, $index + 1, $offset + 1];
264 }
265 $offset -= $days;
266 }
267 throw new LogicException('calendar year length');
268}

270function format_date(string $calendar, int $fixed): string
271{
272 if (!in_range($fixed)) {
273 return 'OUT OF RANGE';
274 }
275 if (in_array($calendar, YEAR_CALENDARS, true)) {
276 $parts = parts_of($calendar, $fixed);
277 if ($parts === null) {
278 return 'OUT OF RANGE';
279 }
280 if ($calendar === 'ISO') {
281 return sprintf('%04d-W%02d-%d', ...$parts);
282 }
283 return sprintf('%04d-%02d-%02d', ...$parts);
284 }
285 if ($calendar === 'JD') {
286 return (string) ($fixed + 1721425);
287 }
288 if ($calendar === 'FIXED') {
289 return (string) $fixed;
290 }
291 $count = $fixed + 1137142;
292 $longCount = [
293 intdiv($count, 144000),
294 intdiv($count, 7200) % 20,
295 intdiv($count, 360) % 20,
296 intdiv($count, 20) % 18,
297 $count % 20
298 ];
299 $tzNumber = ($count + 3) % 13 + 1;
300 $tzName = TZ[($count + 19) % 20];
301 $haab = ($count + 348) % 365;
302 return sprintf('%s %d %s %d %s', implode('.', $longCount), $tzNumber, $tzName, $haab % 20, HB[intdiv($haab, 20)]);
303}

305function parse_date(string $calendar, string $text): int
306{
307 if (in_array($calendar, YMD, true)) {
308 if (preg_match('/^[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}$/D', $text) !== 1) {
309 throw new DateValidationError('FORMAT');
310 }
311 [$year, $month, $day] = array_map('intval', explode('-', $text));
312 if ($year === 0) {
313 throw new DateValidationError('YEAR');
314 }
315 $days = lengths($calendar, $year);
316 if ($month < 1 || $month > count($days)) {
317 throw new DateValidationError('MONTH');
318 }
319 if ($day < 1 || $day > $days[$month - 1]) {
320 throw new DateValidationError('DAY');
321 }
322 $fixed = fixed_date($calendar, $year, $month, $day);
323 } elseif ($calendar === 'ISO') {
324 if (preg_match('/^[0-9]{1,5}-W[0-9]{1,2}-[0-9]$/D', $text) !== 1) {
325 throw new DateValidationError('FORMAT');
326 }
327 [$yearText, $weekText, $dayText] = explode('-', $text);
328 $year = (int) $yearText;
329 $week = (int) substr($weekText, 1);
330 $day = (int) $dayText;
331 if ($year === 0) {
332 throw new DateValidationError('YEAR');
333 }
334 $weeks = intdiv(start($calendar, $year + 1) - start($calendar, $year), 7);
335 if ($week < 1 || $week > $weeks) {
336 throw new DateValidationError('WEEK');
337 }
338 if ($day < 1 || $day > 7) {
339 throw new DateValidationError('DAY');
340 }
341 $fixed = fixed_date($calendar, $year, $week, $day);
342 } elseif ($calendar === 'MAYAN') {
343 if (preg_match('/^[0-9]{1,2}(?:\.[0-9]{1,2}){4}$/D', $text) !== 1) {
344 throw new DateValidationError('FORMAT');
345 }
346 [$b, $k, $t, $u, $n] = array_map('intval', explode('.', $text));
347 if ($k > 19 || $t > 19 || $u > 17 || $n > 19) {
348 throw new DateValidationError('FIELD');
349 }
350 $fixed = ((($b * 20 + $k) * 20 + $t) * 18 + $u) * 20 + $n - 1137142;
351 } else {
352 if (preg_match('/^-?[0-9]{1,8}$/D', $text) !== 1) {
353 throw new DateValidationError('FORMAT');
354 }
355 $fixed = (int) $text - ($calendar === 'JD' ? 1721425 : 0);
356 }
357 if (!in_range($fixed)) {
358 throw new DateValidationError('RANGE');
359 }
360 return $fixed;
361}

372function shift(string $calendar, int $fixed, int $years, int $months): ?int
373{
374 [$year, $month, $day] = parts_of($calendar, $fixed);
375 $targetYear = $year + $years;
376 if ($calendar === 'ISO') {
377 if ($targetYear < 1 || $targetYear > 99999) {
378 return null;
379 }
380 $weeks = intdiv(start($calendar, $targetYear + 1) - start($calendar, $targetYear), 7);
381 return fixed_date($calendar, $targetYear, min($month, $weeks), $day);
382 }
383 if ($calendar === 'HEBREW') {
384 $oldName = names($calendar, $year)[$month - 1];
385 $newNames = names($calendar, $targetYear);
386 if (!in_array($oldName, $newNames, true)) {
387 $oldName = $oldName === 'Adar' ? 'AdarII' : 'Adar';
388 }
389 $month = array_search($oldName, $newNames, true) + 1;
390 }
391 $absolute = month_index($calendar, $targetYear) + $month - 1 + $months;
392 if ($calendar === 'HEBREW') {
393 $finalYear = floor_div(19 * $absolute + 234, 235);
394 while (month_index($calendar, $finalYear) > $absolute) {
395 --$finalYear;
396 }
397 while (month_index($calendar, $finalYear + 1) <= $absolute) {
398 ++$finalYear;
399 }
400 } else {
401 $finalYear = floor_div($absolute, count(names($calendar, $targetYear))) + 1;
402 }
403 if ($finalYear < 1 || $finalYear > 99999) {
404 return null;
405 }
406 $finalMonth = $absolute - month_index($calendar, $finalYear) + 1;
407 $finalDay = min($day, lengths($calendar, $finalYear)[$finalMonth - 1]);
408 return fixed_date($calendar, $finalYear, $finalMonth, $finalDay);
409}

422{

430 ++$year;

435function easter(int $year, string $calendar): array
436{
437 $golden = $year % 19;
438 if ($calendar === 'GREGORIAN') {
439 $century = intdiv($year, 100) + 1;
440 $epact = positive_mod(14 + 11 * $golden - intdiv(3 * $century, 4) + intdiv(5 + 8 * $century, 25), 30);
441 if ($epact === 0 || ($epact === 1 && $golden > 10)) {
442 ++$epact;
443 }
444 } else {
445 $epact = (14 + 11 * $golden) % 30;
446 }
447 $moon = fixed_date($calendar, $year, 4, 19) - $epact;
448 $sunday = $moon + 7 - positive_mod($moon, 7);
449 $index = positive_mod(-start($calendar, $year), 7);
450 $letters = 'ABCDEFG'[$index];
451 if (leap($calendar, $year)) {
452 $letters .= 'ABCDEFG'[positive_mod($index - 1, 7)];
453 }
454 return [$epact, $letters, $moon, $sunday];
455}

485function holiday_lines(int $year): array
486{
487 $first = start('GREGORIAN', $year);
488 $last = start('GREGORIAN', $year + 1) - 1;
489 $gregorianEaster = easter($year, 'GREGORIAN')[3];
490 $orthodoxEaster = easter($year, 'JULIAN')[3];
491 $entries = [];
492 $movable = [['AshWednesday', -46], ['GoodFriday', -2], ['Easter', 0], ['Ascension', 39], ['Pentecost', 49]];
493 foreach ($movable as $order => [$name, $offset]) {

495 }
496 $entries[] = [$orthodoxEaster, 5, 'OrthodoxEaster', 'JULIAN'];
497 foreach (HOLIDAYS as $index => [$name, $calendar, $monthName, $day]) {
498 $candidateYear = max(1, year_of($calendar, $first));
499 while (start($calendar, $candidateYear) <= $last) {
500 $actualName = $monthName;
501 if ($calendar === 'HEBREW' && $monthName === 'Adar' && leap($calendar, $candidateYear)) {
502 $actualName = 'AdarII';
503 }
504 $month = array_search($actualName, names($calendar, $candidateYear), true) + 1;
505 if ($day <= lengths($calendar, $candidateYear)[$month - 1]) {
506 $fixed = fixed_date($calendar, $candidateYear, $month, $day);
507 if ($first <= $fixed && $fixed <= $last) {
508 $entries[] = [$fixed, $index + 6, $name, $calendar];
509 }
510 }
511 ++$candidateYear;
512 }
513 }
514 usort($entries, fn($a, $b) => [$a[0], $a[1]] <=> [$b[0], $b[1]]);
515 $lines = [];
516 foreach ($entries as [$fixed, , $name, $calendar]) {
517 $lines[] = sprintf('%s %s %s %s %s', format_date('GREGORIAN', $fixed), WEEKDAYS[$fixed % 7], $name, $calendar, format_date($calendar, $fixed));
518 }
519 return $lines;
520}

534function validate_arguments(string $command, array $args): void
535{
536 if ($command === 'EASTER' || $command === 'COMPUTUS') {
537 require_valid(valid_year($args[0]) && in_array($args[1], ['GREGORIAN', 'JULIAN'], true));
538 return;
539 }
540 if (in_array($command, ['PASSOVER', 'HOLIDAYS', 'MOLAD'], true)) {
541 require_valid(valid_year($args[0]));
542 return;
543 }
544 $calendar = $args[0];
545 $allowed = match ($command) {
546 'INFO', 'MONTHS', 'MONTHCAL', 'NTH', 'SPAN' => YMD,
547 'LEAP', 'NEWYEAR' => YEAR_CALENDARS,
548 default => CALENDARS,
549 };
550 require_valid(in_array($calendar, $allowed, true));
551 if ($command === 'CONVERT') {
552 require_valid($args[2] === 'TO' && in_array($args[3], CALENDARS, true));
553 } elseif ($command === 'DIFF') {
554 require_valid(in_array($args[2], CALENDARS, true));
555 } elseif (in_array($command, ['LEAP', 'MONTHS', 'MONTHCAL', 'NTH', 'NEWYEAR'], true)) {
556 require_valid(valid_year($args[1]));
557 if ($command === 'MONTHCAL' || $command === 'NTH') {
558 require_valid(preg_match('/^[0-9]{1,2}$/D', $args[2]) === 1);
559 }
560 if ($command === 'NTH') {
561 require_valid(in_array($args[3], ['1', '2', '3', '4', '5', 'LAST'], true));
562 require_valid(in_array($args[4], WEEKDAYS, true));
563 }
564 } elseif ($command === 'COUNT') {
565 require_valid(in_array($args[3], WEEKDAYS, true));
566 } elseif ($command === 'ADD') {
567 require_valid(preg_match('/^-?[0-9]{1,7}$/D', $args[2]) === 1);
568 require_valid(in_array($args[3], ['DAYS', 'WEEKS', 'MONTHS', 'YEARS'], true));
569 if ($args[3] === 'MONTHS') {
570 require_valid(in_array($calendar, YMD, true));
571 }
572 if ($args[3] === 'YEARS') {
573 require_valid(in_array($calendar, YEAR_CALENDARS, true));
574 }
575 } elseif ($command === 'ROUND') {
576 require_valid(preg_match('/^[0-9]{1,2}$/D', $args[2]) === 1);
577 require_valid((int) $args[2] >= 1 && (int) $args[2] <= 13 && in_array($args[3], TZ, true));
578 require_valid(preg_match('/^[0-9]{1,2}$/D', $args[4]) === 1);
579 require_valid(in_array($args[5], HB, true));
580 require_valid((int) $args[4] <= ($args[5] === 'Uayeb' ? 4 : 19));
581 }
582}

584function calendar_lines(string $calendar, int $year, int $month): array
585{
586 [$first, $last] = month_bounds($calendar, $year, $month);
587 if (!in_range($first) || !in_range($last)) {
588 return ['OUT OF RANGE'];
589 }
590 $title = sprintf('%s %d (%s to %s)', names($calendar, $year)[$month - 1], $year,
591 format_date('GREGORIAN', $first), format_date('GREGORIAN', $last));
592 $result = [$title, 'Mo Tu We Th Fr Sa Su'];
593 for ($cursor = $first - ($first - 1) % 7; $cursor <= $last; $cursor += 7) {
594 $fields = [];
595 for ($fixed = $cursor; $fixed < $cursor + 7; ++$fixed) {
596 $fields[] = $first <= $fixed && $fixed <= $last ? sprintf('%2d', $fixed - $first + 1) : ' ';
597 }
598 $result[] = rtrim(implode(' ', $fields));
599 }
600 return $result;
601}

603function span_text(string $calendar, int $first, int $last): string
604{
605 $negative = $first > $last;
606 if ($negative) {
607 [$first, $last] = [$last, $first];
608 }
609 $startYear = parts_of($calendar, $first)[0];
610 $endYear = parts_of($calendar, $last)[0];
611 $low = 0;
612 $high = $endYear - $startYear + 2;
613 while ($low + 1 < $high) {
614 $middle = intdiv($low + $high, 2);
615 $shifted = shift($calendar, $first, $middle, 0);
616 if ($shifted !== null && $shifted <= $last) {
617 $low = $middle;
618 } else {
619 $high = $middle;
620 }
621 }
622 $years = $low;
623 $low = 0;
624 $high = 32;
625 while ($low + 1 < $high) {
626 $middle = intdiv($low + $high, 2);
627 $shifted = shift($calendar, $first, $years, $middle);
628 if ($shifted !== null && $shifted <= $last) {
629 $low = $middle;
630 } else {
631 $high = $middle;
632 }
633 }
634 $months = $low;
635 $days = $last - shift($calendar, $first, $years, $months);
636 return sprintf('%s%dy %dm %dd', $negative ? '-' : '', $years, $months, $days);
637}

639function round_date(string $calendar, int $fixed, int $number, string $tzName, int $day, string $month): string
640{
641 $targetTz = array_search($tzName, TZ, true);
642 $targetHaab = array_search($month, HB, true) * 20 + $day;
643 $candidate = $fixed;
644 while (true) {
645 $count = $candidate + 1137142;
646 if (($count + 3) % 13 + 1 === $number && ($count + 19) % 20 === $targetTz) {
647 break;
648 }
649 ++$candidate;
650 }
651 while ($candidate <= $fixed + 18979) {
652 if (($candidate + 1137142 + 348) % 365 === $targetHaab) {
653 return format_date($calendar, $candidate);
654 }
655 $candidate += 260;
656 }
657 return 'NONE';
658}

660function execute(string $command, array $args): array
661{
662 validate_arguments($command, $args);
663 if ($command === 'EASTER' || $command === 'COMPUTUS') {
664 $year = (int) $args[0];
665 if ($year > 9999) {
666 return ['OUT OF RANGE'];
667 }
668 $kind = $args[1];
669 [$epact, $letters, $moon, $sunday] = easter($year, $kind);
670 if ($command === 'EASTER') {
671 return [format_date('GREGORIAN', $sunday)];
672 }
673 return [sprintf('golden=%d epact=%d letters=%s moon=%s easter=%s', $year % 19 + 1, $epact, $letters,
674 format_date($kind, $moon), format_date($kind, $sunday))];
675 }
676 if (in_array($command, ['PASSOVER', 'HOLIDAYS', 'MOLAD'], true)) {
677 $year = (int) $args[0];
678 if ($command === 'MOLAD') {
679 [$day, $parts, $fixed, $rules] = molad($year);
680 if (!in_range($fixed)) {
681 return ['OUT OF RANGE'];
682 }
683 $applied = $rules === [] ? 'NONE' : implode('+', $rules);
684 return [sprintf('%s %dh %dp %s %s %s', WEEKDAYS[positive_mod($day - 1373427, 7)], intdiv($parts, 1080),
685 $parts % 1080, $applied, WEEKDAYS[$fixed % 7], format_date('GREGORIAN', $fixed))];
686 }
687 if ($year > 9999) {
688 return ['OUT OF RANGE'];
689 }
690 if ($command === 'HOLIDAYS') {
691 return holiday_lines($year);
692 }
693 $hebrewYear = $year + 3760;
694 $month = array_search('Nisan', names('HEBREW', $hebrewYear), true) + 1;
695 $fixed = fixed_date('HEBREW', $hebrewYear, $month, 15);
696 return [format_date('GREGORIAN', $fixed) . ' ' . WEEKDAYS[$fixed % 7]];
697 }
698 $calendar = $args[0];
699 if (in_array($command, ['LEAP', 'MONTHS', 'MONTHCAL', 'NTH', 'NEWYEAR'], true)) {
700 $year = (int) $args[1];
701 if ($command === 'LEAP') {
702 $text = (leap($calendar, $year) ? 'YES' : 'NO') . ' ' . (start($calendar, $year + 1) - start($calendar, $year));
703 if ($calendar === 'HEBREW') {
704 $text .= ' ' . year_type($year);
705 }
706 return [$text];
707 }
708 if ($command === 'MONTHS') {
709 $items = array_map(fn($name, $days) => "$name=$days", names($calendar, $year), lengths($calendar, $year));
710 return [implode(' ', $items)];
711 }
712 if ($command === 'NEWYEAR') {
713 if ($year > 9999) {
714 return ['OUT OF RANGE'];
715 }
716 $entries = new_years($calendar, start('GREGORIAN', $year), start('GREGORIAN', $year + 1) - 1);
717 if ($entries === []) {
718 return ['NONE'];
719 }
720 return [implode(' ', array_map(fn($entry) => $entry[0] . ':' . format_date('GREGORIAN', $entry[1]), $entries))];
721 }
722 $month = (int) $args[2];
723 if ($command === 'MONTHCAL') {
724 return calendar_lines($calendar, $year, $month);
725 }
726 [$first, $last] = month_bounds($calendar, $year, $month);
727 $weekday = array_search($args[4], WEEKDAYS, true);
728 if ($args[3] === 'LAST') {
729 $fixed = $last - positive_mod($last - $weekday, 7);
730 } else {
731 $fixed = $first + positive_mod($weekday - $first, 7) + 7 * ((int) $args[3] - 1);
732 }
733 return [$fixed > $last ? 'NONE' : format_date($calendar, $fixed)];
734 }
735 if ($command === 'VALID') {
736 try {
737 $fixed = parse_date($calendar, $args[1]);
738 } catch (DateValidationError $error) {
739 return ['NO ' . $error->getMessage()];
740 }
741 return ['YES ' . format_date($calendar, $fixed)];
742 }
743 $first = parse_date($calendar, $args[1]);
744 if ($command === 'DIFF') {
745 $last = parse_date($args[2], $args[3]);
746 return [(string) ($last - $first)];
747 }
748 if ($command === 'COUNT' || $command === 'SPAN') {
749 $last = parse_date($calendar, $args[2]);
750 if ($command === 'SPAN') {
751 return [span_text($calendar, $first, $last)];
752 }
753 $low = min($first, $last);
754 $high = max($first, $last);
755 $occurrence = $low + positive_mod(array_search($args[3], WEEKDAYS, true) - $low, 7);
756 return [(string) ($occurrence > $high ? 0 : intdiv($high - $occurrence, 7) + 1)];
757 }
758 if ($command === 'CONVERT') {
759 return [format_date($args[3], $first)];
760 }
761 if ($command === 'WEEKDAY') {
762 return [WEEKDAYS[$first % 7]];
763 }
764 if ($command === 'ALL') {
765 $result = [];
766 foreach (CALENDARS as $target) {
767 $text = format_date($target, $first);
768 if (in_array($target, YMD, true) && $text !== 'OUT OF RANGE') {
769 [$year, $month] = parts_of($target, $first);
770 $text .= ' ' . names($target, $year)[$month - 1];
771 }
772 $result[] = "$target $text";
773 }
774 return $result;
775 }
776 if ($command === 'INFO') {
777 [$year, $month, $day] = parts_of($calendar, $first);
778 $weekday = WEEKDAYS[$first % 7];
779 if ($calendar === 'FRENCH') {
780 $weekday = $month <= 12 ? DC[($day - 1) % 10] : FS[$day - 1];
781 }
782 $ordinal = $first - start($calendar, $year) + 1;
783 $length = start($calendar, $year + 1) - start($calendar, $year);
784 return [sprintf('%s %d %s %d day=%d/%d', $weekday, $day, names($calendar, $year)[$month - 1], $year, $ordinal, $length)];
785 }
786 if ($command === 'ROUND') {
787 return [round_date($calendar, $first, (int) $args[2], $args[3], (int) $args[4], $args[5])];
788 }
789 $amount = (int) $args[2];
790 $result = match ($args[3]) {
791 'DAYS' => $first + $amount,
792 'WEEKS' => $first + 7 * $amount,
793 'YEARS' => shift($calendar, $first, $amount, 0),
794 default => shift($calendar, $first, 0, $amount),
795 };
796 return [$result === null ? 'OUT OF RANGE' : format_date($calendar, $result)];
797}

799$count = (int) fgets(STDIN);
800$output = [];
801for ($index = 0; $index < $count; ++$index) {
802 $args = explode(' ', rtrim(fgets(STDIN), "\r\n"));
803 $command = array_shift($args);
804 if (!array_key_exists($command, ARITY) || count($args) !== ARITY[$command]) {
805 $output[] = 'INVALID COMMAND';
806 continue;
807 }
808 try {
809 array_push($output, ...execute($command, $args));
810 } catch (CommandError) {
811 $output[] = 'INVALID COMMAND';
812 } catch (DateValidationError $error) {
813 $output[] = $error->getMessage() === 'RANGE' ? 'OUT OF RANGE' : 'INVALID DATE';
814 }
815}
816echo implode("\n", $output), "\n";
