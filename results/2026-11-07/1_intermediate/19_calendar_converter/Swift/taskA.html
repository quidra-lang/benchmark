89func leap(_ calendar: Calendar, _ year: Int) -> Bool {
90 switch calendar {
91 case .gregorian: return year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
92 case .julian: return year % 4 == 0
93 case .islamic: return [2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29].contains(mod(year, 30))
94 case .hebrew: return mod(7 * year + 1, 19) < 7
95 case .persian: return (mod(year - 474, 2820) + 474 + 38) * 682 % 2816 < 682
96 case .coptic, .ethiopic: return mod(year, 4) == 3
97 case .french: return year % 4 == 0 && ![100, 200, 300].contains(mod(year, 400)) && year % 4000 != 0
98 case .iso: return weeks(year) == 53
99 case .mayan, .jd, .fixed: preconditionFailure("no years in \(calendar.rawValue)")
100 }
101}

108func molad(_ year: Int) -> (day: Int, parts: Int, start: Int, rules: [String]) {
109 let quantity = 5604 + 765433 * monthsBefore(year)
110 let day = floorDiv(quantity, 25920), parts = mod(quantity, 25920)
111 let weekday = mod(day - 1373427, 7)
112 var rules: [String] = []
113 var postponed = day
114 if parts >= 19440 {
115 postponed += 1
116 rules.append("ZAKEN")
117 } else if !leap(.hebrew, year) && weekday == 2 && parts >= 9924 {
118 postponed += 2
119 rules.append("GATARAD")
120 } else if leap(.hebrew, year - 1) && weekday == 1 && parts >= 16789 {
121 postponed += 1
122 rules.append("BETUTAKPAT")
123 }
124 if [0, 3, 5].contains(mod(postponed - 1373427, 7)) {
125 postponed += 1
126 rules.append("ADU")
127 }
128 return (day, parts, postponed - 1373427, rules)

130func start(_ calendar: Calendar, _ year: Int) -> Int {
131 let previous = year - 1
132 switch calendar {
133 case .gregorian: return 365 * previous + previous / 4 - previous / 100 + previous / 400 + 1
134 case .julian: return 365 * previous + previous / 4 - 1
135 case .islamic: return 227015 + 354 * previous + (3 + 11 * year) / 30
136 case .hebrew: return molad(year).start
137 case .persian:
138 let a = year - 474
139 let b = mod(a, 2820) + 474
140 return 226896 + 1029983 * floorDiv(a, 2820) + 365 * (b - 1) + (682 * b - 110) / 2816
141 case .coptic: return 103605 + 365 * previous + year / 4
142 case .ethiopic: return 2796 + 365 * previous + year / 4
143 case .french: return 654415 + 365 * previous + previous / 4 - previous / 100 + previous / 400 - previous / 4000
144 case .iso:
145 let januaryFour = start(.gregorian, year) + 3
146 return januaryFour - (januaryFour - 1) % 7
147 case .mayan, .jd, .fixed: preconditionFailure("no years in \(calendar.rawValue)")
148 }

176func fixedDate(_ calendar: Calendar, _ year: Int, _ month: Int, _ day: Int) -> Int {
177 if calendar == .iso { return start(calendar, year) + 7 * (month - 1) + day - 1 }
178 return start(calendar, year) + lengths(calendar, year).prefix(month - 1).reduce(0, +) + day - 1

180func yearOf(_ calendar: Calendar, _ fixed: Int) -> Int {
181 if fixed < start(calendar, 1) { return 0 }
182 var low = 1, high = 100000
183 while low + 1 < high {
184 let middle = (low + high) / 2
185 if start(calendar, middle) <= fixed { low = middle } else { high = middle }
186 }
187 return low
188}
189func partsOf(_ calendar: Calendar, _ fixed: Int) -> Parts? {
190 let year = yearOf(calendar, fixed)
191 if year == 0 { return nil }
192 var offset = fixed - start(calendar, year)
193 if calendar == .iso { return (year, offset / 7 + 1, offset % 7 + 1) }
194 for (index, days) in lengths(calendar, year).enumerated() {
195 if offset < days { return (year, index + 1, offset + 1) }
196 offset -= days
197 }
198 preconditionFailure("calendar year length")
199}
200func formatDate(_ calendar: Calendar, _ fixed: Int) -> String {
201 if !inRange(fixed) { return "OUT OF RANGE" }
202 if yearCalendars.contains(calendar) {
203 guard let parts = partsOf(calendar, fixed) else { return "OUT OF RANGE" }
204 if calendar == .iso { return "\(padded(parts.year, 4))-W\(padded(parts.month, 2))-\(parts.day)" }
205 return "\(padded(parts.year, 4))-\(padded(parts.month, 2))-\(padded(parts.day, 2))"
206 }
207 if calendar == .jd { return String(fixed + 1721425) }
208 if calendar == .fixed { return String(fixed) }
209 let count = fixed + 1137142
210 let longCount = [count / 144000, count / 7200 % 20, count / 360 % 20, count / 20 % 18, count % 20]
211 let haab = (count + 348) % 365
212 return longCount.map(String.init).joined(separator: ".") +
213 " \((count + 3) % 13 + 1) \(tzolkinNames[(count + 19) % 20]) \(haab % 20) \(haabNames[haab / 20])"
214}
215func parseDate(_ calendar: Calendar, _ text: String) throws -> Int {
216 let fixed: Int
217 if ymd.contains(calendar) {
218 guard text.wholeMatch(of: #/[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}/#) != nil else { throw CalendarError.invalidDate("FORMAT") }
219 let fields = text.split(separator: "-").map { Int($0)! }
220 let year = fields[0], month = fields[1], day = fields[2]
221 if year == 0 { throw CalendarError.invalidDate("YEAR") }
222 let days = lengths(calendar, year)
223 guard (1...days.count).contains(month) else { throw CalendarError.invalidDate("MONTH") }
224 guard (1...days[month - 1]).contains(day) else { throw CalendarError.invalidDate("DAY") }
225 fixed = fixedDate(calendar, year, month, day)
226 } else if calendar == .iso {
227 guard text.wholeMatch(of: #/[0-9]{1,5}-W[0-9]{1,2}-[0-9]/#) != nil else { throw CalendarError.invalidDate("FORMAT") }
228 let fields = text.split(separator: "-")
229 let year = Int(fields[0])!, week = Int(fields[1].dropFirst())!, day = Int(fields[2])!
230 if year == 0 { throw CalendarError.invalidDate("YEAR") }
231 guard (1...weeks(year)).contains(week) else { throw CalendarError.invalidDate("WEEK") }
232 guard (1...7).contains(day) else { throw CalendarError.invalidDate("DAY") }
233 fixed = fixedDate(calendar, year, week, day)
234 } else if calendar == .mayan {
235 guard text.wholeMatch(of: #/[0-9]{1,2}(?:\.[0-9]{1,2}){4}/#) != nil else { throw CalendarError.invalidDate("FORMAT") }
236 let v = text.split(separator: ".").map { Int($0)! }
237 if v[1] > 19 || v[2] > 19 || v[3] > 17 || v[4] > 19 { throw CalendarError.invalidDate("FIELD") }
238 fixed = (((v[0] * 20 + v[1]) * 20 + v[2]) * 18 + v[3]) * 20 + v[4] - 1137142
239 } else {
240 guard text.wholeMatch(of: #/-?[0-9]{1,8}/#) != nil else { throw CalendarError.invalidDate("FORMAT") }
241 fixed = Int(text)! - (calendar == .jd ? 1721425 : 0)
242 }
243 guard inRange(fixed) else { throw CalendarError.invalidDate("RANGE") }
244 return fixed
245}
246func monthIndex(_ calendar: Calendar, _ year: Int) -> Int {
247 calendar == .hebrew ? monthsBefore(year) : monthNames[calendar]!.count * (year - 1)

249func shift(_ calendar: Calendar, _ fixed: Int, _ years: Int, _ months: Int) -> Int? {
250 let parts = partsOf(calendar, fixed)!
251 let targetYear = parts.year + years
252 if calendar == .iso {
253 guard (1...99999).contains(targetYear) else { return nil }
254 return fixedDate(calendar, targetYear, min(parts.month, weeks(targetYear)), parts.day)
255 }
256 var month = parts.month
257 if calendar == .hebrew {
258 var name = names(calendar, parts.year)[month - 1]
259 let newNames = names(calendar, targetYear)
260 if !newNames.contains(name) { name = name == "Adar" ? "AdarII" : "Adar" }
261 month = newNames.firstIndex(of: name)! + 1
262 }
263 let absolute = monthIndex(calendar, targetYear) + month - 1 + months
264 var finalYear: Int
265 if calendar == .hebrew {
266 finalYear = floorDiv(19 * absolute + 234, 235)
267 while monthIndex(calendar, finalYear) > absolute { finalYear -= 1 }
268 while monthIndex(calendar, finalYear + 1) <= absolute { finalYear += 1 }
269 } else {
270 finalYear = floorDiv(absolute, monthNames[calendar]!.count) + 1
271 }
272 guard (1...99999).contains(finalYear) else { return nil }
273 let finalMonth = absolute - monthIndex(calendar, finalYear) + 1
274 let finalDay = min(parts.day, lengths(calendar, finalYear)[finalMonth - 1])
275 return fixedDate(calendar, finalYear, finalMonth, finalDay)
276}
277func monthBounds(_ calendar: Calendar, _ year: Int, _ month: Int) throws -> (first: Int, last: Int) {
278 let days = lengths(calendar, year)
279 guard (1...days.count).contains(month) else { throw CalendarError.invalidDate("MONTH") }
280 let first = fixedDate(calendar, year, month, 1)
281 return (first, first + days[month - 1] - 1)
282}
283func newYears(_ calendar: Calendar, _ first: Int, _ last: Int) -> [(year: Int, start: Int)] {
284 var year = max(1, yearOf(calendar, first))
285 while start(calendar, year) < first { year += 1 }
286 var result: [(year: Int, start: Int)] = []
287 while start(calendar, year) <= last {

293func easter(_ year: Int, _ calendar: Calendar) -> (epact: Int, letters: String, moon: Int, sunday: Int) {
294 let golden = year % 19
295 var epact: Int
296 if calendar == .gregorian {
297 let century = year / 100 + 1
298 epact = mod(14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25, 30)
299 if epact == 0 || epact == 1 && golden > 10 { epact += 1 }
300 } else {
301 epact = (14 + 11 * golden) % 30
302 }
303 let moon = fixedDate(calendar, year, 4, 19) - epact
304 let sunday = moon + 7 - moon % 7
305 let alphabet = Array("ABCDEFG")
306 let index = mod(-start(calendar, year), 7)
307 var letters = String(alphabet[index])
308 if leap(calendar, year) { letters.append(alphabet[mod(index - 1, 7)]) }
309 return (epact, letters, moon, sunday)
310}
311func holidayLines(_ year: Int) -> [String] {
312 let first = start(.gregorian, year), last = start(.gregorian, year + 1) - 1
313 let gregorianEaster = easter(year, .gregorian).sunday
314 let orthodoxEaster = easter(year, .julian).sunday
315 let movable: [(name: String, offset: Int)] = [("AshWednesday", -46), ("GoodFriday", -2), ("Easter", 0), ("Ascension", 39), ("Pentecost", 49)]
316 var entries: [Entry] = movable.enumerated().map { order, feast in
317 (gregorianEaster + feast.offset, order, feast.name, .gregorian)
318 }
319 entries.append((orthodoxEaster, 5, "OrthodoxEaster", .julian))
320 for (index, holiday) in holidays.enumerated() {
321 let calendar = holiday.calendar
322 var candidateYear = max(1, yearOf(calendar, first))
323 while start(calendar, candidateYear) <= last {
324 var actualName = holiday.month
325 if calendar == .hebrew && actualName == "Adar" && leap(calendar, candidateYear) { actualName = "AdarII" }
326 let month = names(calendar, candidateYear).firstIndex(of: actualName)! + 1
327 if holiday.day <= lengths(calendar, candidateYear)[month - 1] {
328 let fixed = fixedDate(calendar, candidateYear, month, holiday.day)
329 if (first...last).contains(fixed) { entries.append((fixed, index + 6, holiday.name, calendar)) }
330 }
331 candidateYear += 1
332 }
333 }
334 entries.sort { ($0.fixed, $0.order) < ($1.fixed, $1.order) }
335 return entries.map {
336 "\(formatDate(.gregorian, $0.fixed)) \(weekdayName($0.fixed)) \($0.name) \($0.calendar.rawValue) \(formatDate($0.calendar, $0.fixed))"

345func validateArguments(_ command: String, _ args: [String]) throws {
346 if command == "EASTER" || command == "COMPUTUS" {
347 try ensure(validYear(args[0]) && ["GREGORIAN", "JULIAN"].contains(args[1]))
348 return
349 }
350 if ["PASSOVER", "HOLIDAYS", "MOLAD"].contains(command) {
351 try ensure(validYear(args[0]))
352 return
353 }
354 let allCalendars = Set(Calendar.allCases)
355 var allowed = allCalendars
356 if ["INFO", "MONTHS", "MONTHCAL", "NTH", "SPAN"].contains(command) {
357 allowed = ymd
358 } else if command == "LEAP" || command == "NEWYEAR" {
359 allowed = yearCalendars
360 }
361 try ensure(isCalendar(args[0], allowed))
362 switch command {
363 case "CONVERT":
364 try ensure(args[2] == "TO" && isCalendar(args[3], allCalendars))
365 case "DIFF":
366 try ensure(isCalendar(args[2], allCalendars))
367 case "LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR":
368 try ensure(validYear(args[1]))
369 if command == "MONTHCAL" || command == "NTH" { try ensure(args[2].wholeMatch(of: #/[0-9]{1,2}/#) != nil) }
370 if command == "NTH" {
371 try ensure(["1", "2", "3", "4", "5", "LAST"].contains(args[3]))
372 try ensure(weekdays.contains(args[4]))
373 }
374 case "COUNT":
375 try ensure(weekdays.contains(args[3]))
376 case "ADD":
377 try ensure(args[2].wholeMatch(of: #/-?[0-9]{1,7}/#) != nil)
378 try ensure(["DAYS", "WEEKS", "MONTHS", "YEARS"].contains(args[3]))
379 let calendar = Calendar(rawValue: args[0])!
380 if args[3] == "MONTHS" { try ensure(ymd.contains(calendar)) }
381 if args[3] == "YEARS" { try ensure(yearCalendars.contains(calendar)) }
382 case "ROUND":
383 try ensure(args[2].wholeMatch(of: #/[0-9]{1,2}/#) != nil)
384 try ensure((1...13).contains(Int(args[2])!) && tzolkinNames.contains(args[3]))
385 try ensure(args[4].wholeMatch(of: #/[0-9]{1,2}/#) != nil)
386 try ensure(haabNames.contains(args[5]))
387 try ensure(Int(args[4])! <= (args[5] == "Uayeb" ? 4 : 19))
388 default:
389 break
390 }
391}
392func calendarLines(_ calendar: Calendar, _ year: Int, _ month: Int) throws -> [String] {
393 let (first, last) = try monthBounds(calendar, year, month)
394 if !inRange(first) || !inRange(last) { return ["OUT OF RANGE"] }
395 var result = [
396 "\(names(calendar, year)[month - 1]) \(year) (\(formatDate(.gregorian, first)) to \(formatDate(.gregorian, last)))",
397 "Mo Tu We Th Fr Sa Su",
398 ]
399 for cursor in stride(from: first - (first - 1) % 7, through: last, by: 7) {
400 let fields = (cursor..<cursor + 7).map { day in
401 (first...last).contains(day) ? padded(day - first + 1, 2, with: " ") : " "
402 }
403 var line = fields.joined(separator: " ")
404 while line.hasSuffix(" ") { line.removeLast() }
405 result.append(line)
406 }
407 return result
408}
409func spanText(_ calendar: Calendar, _ first: Int, _ last: Int) -> String {
410 let from = min(first, last), to = max(first, last)
411 var low = 0, high = partsOf(calendar, to)!.year - partsOf(calendar, from)!.year + 2
412 while low + 1 < high {
413 let middle = (low + high) / 2
414 if let shifted = shift(calendar, from, middle, 0), shifted <= to { low = middle } else { high = middle }
415 }
416 let years = low
417 low = 0
418 high = 32
419 while low + 1 < high {
420 let middle = (low + high) / 2
421 if let shifted = shift(calendar, from, years, middle), shifted <= to { low = middle } else { high = middle }
422 }
423 let months = low
424 let days = to - shift(calendar, from, years, months)!
425 return "\(first > last ? "-" : "")\(years)y \(months)m \(days)d"
426}
427func roundDate(_ calendar: Calendar, _ fixed: Int, _ number: Int, _ tzName: String, _ day: Int, _ month: String) -> String {
428 let targetTz = tzolkinNames.firstIndex(of: tzName)!
429 let targetHaab = haabNames.firstIndex(of: month)! * 20 + day
430 var candidate = fixed
431 while (candidate + 1137142 + 3) % 13 + 1 != number || (candidate + 1137142 + 19) % 20 != targetTz { candidate += 1 }
432 while candidate <= fixed + 18979 {
433 if (candidate + 1137142 + 348) % 365 == targetHaab { return formatDate(calendar, candidate) }
434 candidate += 260
435 }
436 return "NONE"
437}
438func execute(_ command: String, _ args: [String]) throws -> [String] {
439 try validateArguments(command, args)
440 if command == "EASTER" || command == "COMPUTUS" {
441 let year = Int(args[0])!
442 if year > 9999 { return ["OUT OF RANGE"] }
443 let kind = Calendar(rawValue: args[1])!
444 let (epact, letters, moon, sunday) = easter(year, kind)
445 if command == "EASTER" { return [formatDate(.gregorian, sunday)] }
446 return ["golden=\(year % 19 + 1) epact=\(epact) letters=\(letters) moon=\(formatDate(kind, moon)) easter=\(formatDate(kind, sunday))"]
447 }
448 if ["PASSOVER", "HOLIDAYS", "MOLAD"].contains(command) {
449 let year = Int(args[0])!
450 if command == "MOLAD" {
451 let (day, parts, fixed, rules) = molad(year)
452 if !inRange(fixed) { return ["OUT OF RANGE"] }
453 let applied = rules.isEmpty ? "NONE" : rules.joined(separator: "+")
454 return ["\(weekdayName(day - 1373427)) \(parts / 1080)h \(parts % 1080)p \(applied) \(weekdayName(fixed)) \(formatDate(.gregorian, fixed))"]
455 }
456 if year > 9999 { return ["OUT OF RANGE"] }
457 if command == "HOLIDAYS" { return holidayLines(year) }
458 let hebrewYear = year + 3760
459 let month = names(.hebrew, hebrewYear).firstIndex(of: "Nisan")! + 1
460 let fixed = fixedDate(.hebrew, hebrewYear, month, 15)
461 return ["\(formatDate(.gregorian, fixed)) \(weekdayName(fixed))"]
462 }
463 let calendar = Calendar(rawValue: args[0])!
464 if ["LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR"].contains(command) {
465 let year = Int(args[1])!
466 switch command {
467 case "LEAP":
468 var text = "\(leap(calendar, year) ? "YES" : "NO") \(start(calendar, year + 1) - start(calendar, year))"
469 if calendar == .hebrew { text += " \(yearType(year))" }
470 return [text]
471 case "MONTHS":
472 return [zip(names(calendar, year), lengths(calendar, year)).map { "\($0)=\($1)" }.joined(separator: " ")]
473 case "NEWYEAR":
474 if year > 9999 { return ["OUT OF RANGE"] }
475 let entries = newYears(calendar, start(.gregorian, year), start(.gregorian, year + 1) - 1)
476 if entries.isEmpty { return ["NONE"] }
477 return [entries.map { "\($0.year):\(formatDate(.gregorian, $0.start))" }.joined(separator: " ")]
478 case "MONTHCAL":
479 return try calendarLines(calendar, year, Int(args[2])!)
480 default:
481 let (first, last) = try monthBounds(calendar, year, Int(args[2])!)
482 let weekday = weekdays.firstIndex(of: args[4])!
483 let fixed = args[3] == "LAST" ? last - mod(last - weekday, 7) : first + mod(weekday - first, 7) + 7 * (Int(args[3])! - 1)
484 return [fixed > last ? "NONE" : formatDate(calendar, fixed)]
485 }
486 }
487 if command == "VALID" {
488 do {
489 return ["YES " + formatDate(calendar, try parseDate(calendar, args[1]))]
490 } catch CalendarError.invalidDate(let reason) {
491 return ["NO " + reason]
492 }
493 }
494 let first = try parseDate(calendar, args[1])
495 switch command {
496 case "DIFF":
497 return [String(try parseDate(Calendar(rawValue: args[2])!, args[3]) - first)]
498 case "COUNT":
499 let last = try parseDate(calendar, args[2])
500 let low = min(first, last), high = max(first, last)
501 let occurrence = low + mod(weekdays.firstIndex(of: args[3])! - low, 7)
502 return [String(occurrence > high ? 0 : (high - occurrence) / 7 + 1)]
503 case "SPAN":
504 return [spanText(calendar, first, try parseDate(calendar, args[2]))]
505 case "CONVERT":
506 return [formatDate(Calendar(rawValue: args[3])!, first)]
507 case "WEEKDAY":
508 return [weekdayName(first)]
509 case "ALL":
510 return Calendar.allCases.map { target in
511 var text = formatDate(target, first)
512 if ymd.contains(target) && text != "OUT OF RANGE" {
513 let parts = partsOf(target, first)!
514 text += " " + names(target, parts.year)[parts.month - 1]
515 }
516 return "\(target.rawValue) \(text)"
517 }
518 case "INFO":
519 let (year, month, day) = partsOf(calendar, first)!
520 var weekday = weekdayName(first)
521 if calendar == .french { weekday = month <= 12 ? decadeNames[(day - 1) % 10] : festivalNames[day - 1] }
522 let ordinal = first - start(calendar, year) + 1
523 let length = start(calendar, year + 1) - start(calendar, year)
524 return ["\(weekday) \(day) \(names(calendar, year)[month - 1]) \(year) day=\(ordinal)/\(length)"]
525 case "ROUND":
526 return [roundDate(calendar, first, Int(args[2])!, args[3], Int(args[4])!, args[5])]
527 default:
528 let amount = Int(args[2])!
529 let result: Int?
530 switch args[3] {
531 case "DAYS": result = first + amount
532 case "WEEKS": result = first + 7 * amount
533 case "YEARS": result = shift(calendar, first, amount, 0)
534 default: result = shift(calendar, first, 0, amount)
535 }
536 guard let result else { return ["OUT OF RANGE"] }
537 return [formatDate(calendar, result)]
538 }
