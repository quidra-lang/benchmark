119bool leap(Calendar calendar, long long year) {
120 switch (calendar) {
121 case GREGORIAN:
122 return mod(year, 4) == 0 && (mod(year, 100) != 0 || mod(year, 400) == 0);
123 case JULIAN:
124 return mod(year, 4) == 0;
125 case ISLAMIC: {
126 const std::array<long long, 11> leapYears = {2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29};
127 return std::find(leapYears.begin(), leapYears.end(), mod(year, 30)) != leapYears.end();
128 }
129 case HEBREW:
130 return mod(7 * year + 1, 19) < 7;
131 case PERSIAN: {
132 long long b = mod(year - 474, 2820) + 474;
133 return mod((b + 38) * 682, 2816) < 682;
134 }
135 case COPTIC:
136 case ETHIOPIC:
137 return mod(year, 4) == 3;
138 case FRENCH: {
139 long long century = mod(year, 400);
140 return mod(year, 4) == 0 && century != 100 && century != 200 && century != 300 && mod(year, 4000) != 0;
141 }
142 default:
143 return (start(ISO, year + 1) - start(ISO, year)) / 7 == 53;
144 }
145}

149Molad molad(long long year) {
150 long long quantity = 5604 + 765433 * monthsBefore(year);
151 Molad result{floorDiv(quantity, 25920), mod(quantity, 25920), 0, {}};
152 long long weekday = mod(result.day - 1373427, 7);
153 long long postponed = result.day;
154 if (result.parts >= 19440) {
155 postponed += 1;
156 result.rules.push_back("ZAKEN");
157 } else if (!leap(HEBREW, year) && weekday == 2 && result.parts >= 9924) {
158 postponed += 2;
159 result.rules.push_back("GATARAD");
160 } else if (leap(HEBREW, year - 1) && weekday == 1 && result.parts >= 16789) {
161 postponed += 1;
162 result.rules.push_back("BETUTAKPAT");
163 }
164 long long finalWeekday = mod(postponed - 1373427, 7);
165 if (finalWeekday == 0 || finalWeekday == 3 || finalWeekday == 5) {
166 postponed += 1;
167 result.rules.push_back("ADU");
168 }
169 result.start = postponed - 1373427;
170 return result;
171}

173long long start(Calendar calendar, long long year) {
174 long long previous = year - 1;
175 switch (calendar) {
176 case GREGORIAN:
177 return 365 * previous + floorDiv(previous, 4) - floorDiv(previous, 100) + floorDiv(previous, 400) + 1;
178 case JULIAN:
179 return 365 * previous + floorDiv(previous, 4) - 1;
180 case ISLAMIC:
181 return 227015 + 354 * previous + floorDiv(3 + 11 * year, 30);
182 case HEBREW:
183 return molad(year).start;
184 case PERSIAN: {
185 long long a = year - 474;
186 long long b = mod(a, 2820) + 474;
187 return 226896 + 1029983 * floorDiv(a, 2820) + 365 * (b - 1) + floorDiv(682 * b - 110, 2816);
188 }
189 case COPTIC:
190 return 103605 + 365 * previous + floorDiv(year, 4);
191 case ETHIOPIC:
192 return 2796 + 365 * previous + floorDiv(year, 4);
193 case FRENCH:
194 return 654415 + 365 * previous + floorDiv(previous, 4) - floorDiv(previous, 100) + floorDiv(previous, 400) - floorDiv(previous, 4000);
195 default: {
196 long long januaryFour = start(GREGORIAN, year) + 3;
197 return januaryFour - mod(januaryFour - 1, 7);
198 }
199 }

238long long fixedDate(Calendar calendar, long long year, int month, int day) {
239 if (calendar == ISO) return start(calendar, year) + 7 * (month - 1) + day - 1;
240 vector<int> days = lengths(calendar, year);

244long long yearOf(Calendar calendar, long long fixed) {
245 if (fixed < start(calendar, 1)) return 0;
246 long long low = 1, high = 100000;
247 while (low + 1 < high) {
248 long long middle = (low + high) / 2;
249 if (start(calendar, middle) <= fixed) low = middle;
250 else high = middle;
251 }
252 return low;
253}

255std::optional<Date> partsOf(Calendar calendar, long long fixed) {
256 long long year = yearOf(calendar, fixed);
257 if (year == 0) return std::nullopt;
258 long long offset = fixed - start(calendar, year);
259 if (calendar == ISO) return Date{year, int(offset / 7) + 1, int(offset % 7) + 1};
260 vector<int> days = lengths(calendar, year);
261 for (int month = 1; month <= int(days.size()); ++month) {
262 if (offset < days[month - 1]) return Date{year, month, int(offset) + 1};
263 offset -= days[month - 1];
264 }
265 throw std::logic_error("calendar year length");
266}

268string formatDate(Calendar calendar, long long fixed) {
269 if (!inRange(fixed)) return "OUT OF RANGE";
270 if (hasYears(calendar)) {
271 std::optional<Date> date = partsOf(calendar, fixed);
272 if (!date) return "OUT OF RANGE";
273 if (calendar == ISO) return text(padded(date->year, 4), "-W", padded(date->month, 2), "-", date->day);
274 return text(padded(date->year, 4), "-", padded(date->month, 2), "-", padded(date->day, 2));
275 }
276 if (calendar == JD) return std::to_string(fixed + 1721425);
277 if (calendar == FIXED) return std::to_string(fixed);
278 long long count = fixed + 1137142;
279 long long haabDay = (count + 348) % 365;
280 return text(count / 144000, ".", count / 7200 % 20, ".", count / 360 % 20, ".", count / 20 % 18, ".", count % 20, " ",
281 (count + 3) % 13 + 1, " ", tzolkin[(count + 19) % 20], " ", haabDay % 20, " ", haab[haabDay / 20]);
282}

284long long parseDate(Calendar calendar, const string &value) {
285 std::smatch match;
286 long long fixed;
287 if (isYmd(calendar)) {
288 if (!std::regex_match(value, match, ymdPattern)) throw DateError("FORMAT");
289 long long year = std::stoll(match.str(1));
290 int month = std::stoi(match.str(2)), day = std::stoi(match.str(3));
291 if (year == 0) throw DateError("YEAR");
292 vector<int> days = lengths(calendar, year);
293 if (month < 1 || month > int(days.size())) throw DateError("MONTH");
294 if (day < 1 || day > days[month - 1]) throw DateError("DAY");
295 fixed = fixedDate(calendar, year, month, day);
296 } else if (calendar == ISO) {
297 if (!std::regex_match(value, match, isoPattern)) throw DateError("FORMAT");
298 long long year = std::stoll(match.str(1));
299 int week = std::stoi(match.str(2)), day = std::stoi(match.str(3));
300 if (year == 0) throw DateError("YEAR");
301 long long weeks = (start(calendar, year + 1) - start(calendar, year)) / 7;
302 if (week < 1 || week > weeks) throw DateError("WEEK");
303 if (day < 1 || day > 7) throw DateError("DAY");
304 fixed = fixedDate(calendar, year, week, day);
305 } else if (calendar == MAYAN) {
306 if (!std::regex_match(value, match, mayanPattern)) throw DateError("FORMAT");
307 long long b = std::stoll(match.str(1)), k = std::stoll(match.str(2)), t = std::stoll(match.str(3));
308 long long u = std::stoll(match.str(4)), n = std::stoll(match.str(5));
309 if (k > 19 || t > 19 || u > 17 || n > 19) throw DateError("FIELD");
310 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142;
311 } else {
312 if (!std::regex_match(value, integerPattern)) throw DateError("FORMAT");
313 fixed = std::stoll(value) - (calendar == JD ? 1721425 : 0);
314 }
315 if (!inRange(fixed)) throw DateError("RANGE");
316 return fixed;
317}

325std::optional<long long> shift(Calendar calendar, long long fixed, long long years, long long months) {
326 Date date = *partsOf(calendar, fixed);
327 long long targetYear = date.year + years;
328 if (calendar == ISO) {
329 if (targetYear < 1 || targetYear > 99999) return std::nullopt;
330 long long weeks = (start(calendar, targetYear + 1) - start(calendar, targetYear)) / 7;
331 return fixedDate(calendar, targetYear, int(std::min<long long>(date.month, weeks)), date.day);
332 }
333 int month = date.month;
334 if (calendar == HEBREW) {
335 string oldName = names(calendar, date.year)[month - 1];
336 const vector<string> &newNames = names(calendar, targetYear);
337 if (!contains(newNames, oldName)) oldName = oldName == "Adar" ? "AdarII" : "Adar";
338 month = indexOf(newNames, oldName) + 1;
339 }
340 long long absolute = monthIndex(calendar, targetYear) + month - 1 + months;
341 long long finalYear;
342 if (calendar == HEBREW) {
343 finalYear = floorDiv(19 * absolute + 234, 235);
344 while (monthIndex(calendar, finalYear) > absolute) --finalYear;
345 while (monthIndex(calendar, finalYear + 1) <= absolute) ++finalYear;
346 } else {
347 long long count = names(calendar, targetYear).size();
348 finalYear = floorDiv(absolute, count) + 1;
349 }
350 if (finalYear < 1 || finalYear > 99999) return std::nullopt;
351 int finalMonth = int(absolute - monthIndex(calendar, finalYear)) + 1;
352 int finalDay = std::min(date.day, lengths(calendar, finalYear)[finalMonth - 1]);
353 return fixedDate(calendar, finalYear, finalMonth, finalDay);
354}

371Easter easter(long long year, Calendar calendar) {
372 long long golden = year % 19;
373 long long epact;
374 if (calendar == GREGORIAN) {
375 long long century = year / 100 + 1;
376 epact = mod(14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25, 30);
377 if (epact == 0 || (epact == 1 && golden > 10)) ++epact;
378 } else {
379 epact = (14 + 11 * golden) % 30;
380 }
381 long long moon = fixedDate(calendar, year, 4, 19) - epact;
382 long long sunday = moon + 7 - mod(moon, 7);
383 long long index = mod(-start(calendar, year), 7);
384 const string alphabet = "ABCDEFG";
385 string letters(1, alphabet[index]);
386 if (leap(calendar, year)) letters += alphabet[mod(index - 1, 7)];
387 return {epact, letters, moon, sunday};
388}

418vector<string> holidayLines(long long year) {
419 long long first = start(GREGORIAN, year);
420 long long last = start(GREGORIAN, year + 1) - 1;
421 long long gregorianEaster = easter(year, GREGORIAN).sunday;

427 for (int order = 0; order < int(movable.size()); ++order) {

432 const Holiday &holiday = holidays[i];
433 Calendar calendar = holiday.calendar;
434 for (long long candidate = std::max(1LL, yearOf(calendar, first)); start(calendar, candidate) <= last; ++candidate) {
435 string actualName = holiday.month;
436 if (calendar == HEBREW && actualName == "Adar" && leap(calendar, candidate)) actualName = "AdarII";
437 int month = indexOf(names(calendar, candidate), actualName) + 1;
438 if (holiday.day > lengths(calendar, candidate)[month - 1]) continue;
439 long long fixed = fixedDate(calendar, candidate, month, holiday.day);
440 if (first <= fixed && fixed <= last) entries.push_back({fixed, i + 6, holiday.name, calendar});
441 }
442 }
443 std::sort(entries.begin(), entries.end(), [](const Entry &a, const Entry &b) {
444 return std::tie(a.fixed, a.order) < std::tie(b.fixed, b.order);
445 });
446 vector<string> result;
447 for (const Entry &entry : entries) {
448 result.push_back(text(formatDate(GREGORIAN, entry.fixed), " ", weekdays[entry.fixed % 7], " ", entry.name, " ",
449 calendarNames[entry.calendar], " ", formatDate(entry.calendar, entry.fixed)));
450 }
451 return result;
452}

463void validateArguments(const string &command, const vector<string> &args) {
464 if (command == "EASTER" || command == "COMPUTUS") {
465 require(validYear(args[0]) && (args[1] == "GREGORIAN" || args[1] == "JULIAN"));
466 return;
467 }
468 if (command == "PASSOVER" || command == "HOLIDAYS" || command == "MOLAD") {
469 require(validYear(args[0]));
470 return;
471 }
472 require(contains(calendarNames, args[0]));
473 Calendar calendar = calendarOf(args[0]);
474 if (command == "INFO" || command == "MONTHS" || command == "MONTHCAL" || command == "NTH" || command == "SPAN") {
475 require(isYmd(calendar));
476 } else if (command == "LEAP" || command == "NEWYEAR") {
477 require(hasYears(calendar));
478 }
479 if (command == "CONVERT") {
480 require(args[2] == "TO" && contains(calendarNames, args[3]));
481 } else if (command == "DIFF") {
482 require(contains(calendarNames, args[2]));
483 } else if (command == "LEAP" || command == "MONTHS" || command == "MONTHCAL" || command == "NTH" || command == "NEWYEAR") {
484 require(validYear(args[1]));
485 if (command == "MONTHCAL" || command == "NTH") require(std::regex_match(args[2], twoDigitPattern));
486 if (command == "NTH") {
487 require(contains({"1", "2", "3", "4", "5", "LAST"}, args[3]));
488 require(contains(weekdays, args[4]));
489 }
490 } else if (command == "COUNT") {
491 require(contains(weekdays, args[3]));
492 } else if (command == "ADD") {
493 require(std::regex_match(args[2], amountPattern));
494 require(contains({"DAYS", "WEEKS", "MONTHS", "YEARS"}, args[3]));
495 if (args[3] == "MONTHS") require(isYmd(calendar));
496 if (args[3] == "YEARS") require(hasYears(calendar));
497 } else if (command == "ROUND") {
498 require(std::regex_match(args[2], twoDigitPattern));
499 int number = std::stoi(args[2]);
500 require(1 <= number && number <= 13 && contains(tzolkin, args[3]));
501 require(std::regex_match(args[4], twoDigitPattern));
502 require(contains(haab, args[5]));
503 require(std::stoi(args[4]) <= (args[5] == "Uayeb" ? 4 : 19));
504 }
505}

507vector<string> calendarLines(Calendar calendar, long long year, int month) {
508 auto [first, last] = monthBounds(calendar, year, month);
509 if (!inRange(first) || !inRange(last)) return {"OUT OF RANGE"};
510 vector<string> result = {
511 text(names(calendar, year)[month - 1], " ", year, " (", formatDate(GREGORIAN, first), " to ", formatDate(GREGORIAN, last), ")"),
512 "Mo Tu We Th Fr Sa Su",
513 };
514 for (long long cursor = first - (first - 1) % 7; cursor <= last; cursor += 7) {
515 vector<string> fields;
516 for (long long fixed = cursor; fixed < cursor + 7; ++fixed) {
517 fields.push_back(first <= fixed && fixed <= last ? padded(fixed - first + 1, 2, ' ') : " ");
518 }
519 string row = join(fields, " ");
520 row.erase(row.find_last_not_of(' ') + 1);
521 result.push_back(row);
522 }
523 return result;
524}

526string spanText(Calendar calendar, long long first, long long last) {
527 bool negative = first > last;
528 if (negative) std::swap(first, last);
529 long long startYear = partsOf(calendar, first)->year;
530 long long endYear = partsOf(calendar, last)->year;
531 long long low = 0, high = endYear - startYear + 2;
532 while (low + 1 < high) {
533 long long middle = (low + high) / 2;
534 std::optional<long long> shifted = shift(calendar, first, middle, 0);
535 if (shifted && *shifted <= last) low = middle;
536 else high = middle;
537 }
538 long long years = low;
539 low = 0;
540 high = 32;
541 while (low + 1 < high) {
542 long long middle = (low + high) / 2;
543 std::optional<long long> shifted = shift(calendar, first, years, middle);
544 if (shifted && *shifted <= last) low = middle;
545 else high = middle;
546 }
547 long long months = low;
548 long long days = last - *shift(calendar, first, years, months);
549 return text(negative ? "-" : "", years, "y ", months, "m ", days, "d");
550}

552string roundDate(Calendar calendar, long long fixed, int number, const string &tzName, int day, const string &month) {
553 int targetTz = indexOf(tzolkin, tzName);
554 int targetHaab = indexOf(haab, month) * 20 + day;
555 long long candidate = fixed;
556 while (true) {
557 long long count = candidate + 1137142;
558 if ((count + 3) % 13 + 1 == number && (count + 19) % 20 == targetTz) break;
559 ++candidate;
560 }
561 for (; candidate <= fixed + 18979; candidate += 260) {
562 if ((candidate + 1137142 + 348) % 365 == targetHaab) return formatDate(calendar, candidate);
563 }
564 return "NONE";
565}

567vector<string> execute(const string &command, const vector<string> &args) {
568 validateArguments(command, args);
569 if (command == "EASTER" || command == "COMPUTUS") {
570 long long year = std::stoll(args[0]);
571 if (year > 9999) return {"OUT OF RANGE"};
572 Calendar kind = calendarOf(args[1]);
573 Easter result = easter(year, kind);
574 if (command == "EASTER") return {formatDate(GREGORIAN, result.sunday)};
575 return {text("golden=", year % 19 + 1, " epact=", result.epact, " letters=", result.letters,
576 " moon=", formatDate(kind, result.moon), " easter=", formatDate(kind, result.sunday))};
577 }
578 if (command == "PASSOVER" || command == "HOLIDAYS" || command == "MOLAD") {
579 long long year = std::stoll(args[0]);
580 if (command == "MOLAD") {
581 Molad result = molad(year);
582 if (!inRange(result.start)) return {"OUT OF RANGE"};
583 string applied = result.rules.empty() ? "NONE" : join(result.rules, "+");
584 return {text(weekdays[mod(result.day - 1373427, 7)], " ", result.parts / 1080, "h ", result.parts % 1080, "p ",
585 applied, " ", weekdays[result.start % 7], " ", formatDate(GREGORIAN, result.start))};
586 }
587 if (year > 9999) return {"OUT OF RANGE"};
588 if (command == "HOLIDAYS") return holidayLines(year);
589 long long hebrewYear = year + 3760;
590 int month = indexOf(names(HEBREW, hebrewYear), "Nisan") + 1;
591 long long fixed = fixedDate(HEBREW, hebrewYear, month, 15);
592 return {formatDate(GREGORIAN, fixed) + " " + weekdays[fixed % 7]};
593 }
594 Calendar calendar = calendarOf(args[0]);
595 if (command == "LEAP" || command == "MONTHS" || command == "MONTHCAL" || command == "NTH" || command == "NEWYEAR") {
596 long long year = std::stoll(args[1]);
597 if (command == "LEAP") {
598 string line = text(leap(calendar, year) ? "YES" : "NO", " ", start(calendar, year + 1) - start(calendar, year));
599 if (calendar == HEBREW) line += " " + yearType(year);
600 return {line};
601 }
602 if (command == "MONTHS") {
603 const vector<string> &monthList = names(calendar, year);
604 vector<int> days = lengths(calendar, year);
605 vector<string> items;
606 for (std::size_t i = 0; i < days.size(); ++i) items.push_back(text(monthList[i], "=", days[i]));
607 return {join(items, " ")};
608 }
609 if (command == "NEWYEAR") {
610 if (year > 9999) return {"OUT OF RANGE"};
611 vector<string> items;
612 for (const auto &[newYear, fixed] : newYears(calendar, start(GREGORIAN, year), start(GREGORIAN, year + 1) - 1)) {
613 items.push_back(text(newYear, ":", formatDate(GREGORIAN, fixed)));
614 }
615 return {items.empty() ? "NONE" : join(items, " ")};
616 }
617 int month = std::stoi(args[2]);
618 if (command == "MONTHCAL") return calendarLines(calendar, year, month);
619 auto [first, last] = monthBounds(calendar, year, month);
620 int weekday = indexOf(weekdays, args[4]);
621 long long fixed;
622 if (args[3] == "LAST") fixed = last - mod(last - weekday, 7);
623 else fixed = first + mod(weekday - first, 7) + 7 * (std::stoi(args[3]) - 1);
624 return {fixed > last ? "NONE" : formatDate(calendar, fixed)};
625 }
626 if (command == "VALID") {
627 try {
628 return {"YES " + formatDate(calendar, parseDate(calendar, args[1]))};
629 } catch (const DateError &error) {
630 return {string("NO ") + error.what()};
631 }
632 }
633 long long first = parseDate(calendar, args[1]);
634 if (command == "DIFF") return {std::to_string(parseDate(calendarOf(args[2]), args[3]) - first)};
635 if (command == "COUNT" || command == "SPAN") {
636 long long last = parseDate(calendar, args[2]);
637 if (command == "SPAN") return {spanText(calendar, first, last)};
638 long long low = std::min(first, last), high = std::max(first, last);
639 long long occurrence = low + mod(indexOf(weekdays, args[3]) - low, 7);
640 return {std::to_string(occurrence > high ? 0 : (high - occurrence) / 7 + 1)};
641 }
642 if (command == "CONVERT") return {formatDate(calendarOf(args[3]), first)};
643 if (command == "WEEKDAY") return {weekdays[first % 7]};
644 if (command == "ALL") {
645 vector<string> result;
646 for (std::size_t index = 0; index < calendarNames.size(); ++index) {
647 Calendar target = Calendar(index);
648 string line = formatDate(target, first);
649 if (isYmd(target) && line != "OUT OF RANGE") {
650 Date date = *partsOf(target, first);
651 line += " " + names(target, date.year)[date.month - 1];
652 }
653 result.push_back(calendarNames[target] + " " + line);
654 }
655 return result;
656 }
657 if (command == "INFO") {
658 Date date = *partsOf(calendar, first);
659 string weekday = weekdays[first % 7];
660 if (calendar == FRENCH) weekday = date.month <= 12 ? decades[(date.day - 1) % 10] : festivals[date.day - 1];
661 long long ordinal = first - start(calendar, date.year) + 1;
662 long long length = start(calendar, date.year + 1) - start(calendar, date.year);
663 return {text(weekday, " ", date.day, " ", names(calendar, date.year)[date.month - 1], " ", date.year, " day=", ordinal, "/", length)};
664 }
665 if (command == "ROUND") return {roundDate(calendar, first, std::stoi(args[2]), args[3], std::stoi(args[4]), args[5])};
666 long long amount = std::stoll(args[2]);
667 const string &unit = args[3];
668 std::optional<long long> result;
669 if (unit == "DAYS") result = first + amount;
670 else if (unit == "WEEKS") result = first + 7 * amount;
671 else if (unit == "YEARS") result = shift(calendar, first, amount, 0);
672 else result = shift(calendar, first, 0, amount);
673 return {result ? formatDate(calendar, *result) : "OUT OF RANGE"};
674}
