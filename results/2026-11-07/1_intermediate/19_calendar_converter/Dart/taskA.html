90bool leap(Calendar calendar, int year) => switch (calendar) {

92 Calendar.julian => year % 4 == 0,
93 Calendar.islamic => const {2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29}.contains(year % 30),

106Molad molad(int year) {
107 final quantity = 5604 + 765433 * monthsBefore(year);
108 final day = floorDiv(quantity, 25920), parts = quantity % 25920;
109 final weekday = (day - 1373427) % 7;
110 final rules = <String>[];
111 var postponed = day;
112 if (parts >= 19440) {
113 postponed += 1;
114 rules.add('ZAKEN');
115 } else if (!leap(Calendar.hebrew, year) && weekday == 2 && parts >= 9924) {
116 postponed += 2;
117 rules.add('GATARAD');
118 } else if (leap(Calendar.hebrew, year - 1) && weekday == 1 && parts >= 16789) {
119 postponed += 1;

123 postponed += 1;
124 rules.add('ADU');

129int start(Calendar calendar, int year) {
130 final previous = year - 1;
131 switch (calendar) {
132 case Calendar.gregorian:
133 return 365 * previous + previous ~/ 4 - previous ~/ 100 + previous ~/ 400 + 1;
134 case Calendar.julian:
135 return 365 * previous + previous ~/ 4 - 1;
136 case Calendar.islamic:
137 return 227015 + 354 * previous + (3 + 11 * year) ~/ 30;
138 case Calendar.hebrew:
139 return molad(year).start;
140 case Calendar.persian:
141 final a = year - 474;
142 final b = a % 2820 + 474;
143 return 226896 + 1029983 * floorDiv(a, 2820) + 365 * (b - 1) + (682 * b - 110) ~/ 2816;
144 case Calendar.coptic:
145 return 103605 + 365 * previous + year ~/ 4;
146 case Calendar.ethiopic:
147 return 2796 + 365 * previous + year ~/ 4;
148 case Calendar.french:
149 return 654415 + 365 * previous + previous ~/ 4 - previous ~/ 100 + previous ~/ 400 - previous ~/ 4000;
150 case Calendar.iso:
151 final januaryFour = start(Calendar.gregorian, year) + 3;
152 return januaryFour - (januaryFour - 1) % 7;
153 default:
154 throw ArgumentError('no years in ${calendar.label}');
155 }

193int yearOf(Calendar calendar, int fixed) {
194 if (fixed < start(calendar, 1)) return 0;
195 var low = 1, high = 100000;
196 while (low + 1 < high) {
197 final middle = (low + high) ~/ 2;
198 if (start(calendar, middle) <= fixed) {
199 low = middle;
200 } else {
201 high = middle;
202 }
203 }
204 return low;
205}

207Parts? partsOf(Calendar calendar, int fixed) {
208 final year = yearOf(calendar, fixed);
209 if (year == 0) return null;
210 var offset = fixed - start(calendar, year);
211 if (calendar == Calendar.iso) return (year: year, month: offset ~/ 7 + 1, day: offset % 7 + 1);
212 final days = lengths(calendar, year);
213 var month = 0;
214 while (offset >= days[month]) {
215 offset -= days[month];
216 month++;
217 }
218 return (year: year, month: month + 1, day: offset + 1);

221String formatDate(Calendar calendar, int fixed) {
222 if (!inRange(fixed)) return 'OUT OF RANGE';
223 if (yearCalendars.contains(calendar)) {
224 final parts = partsOf(calendar, fixed);
225 if (parts == null) return 'OUT OF RANGE';
226 final (:year, :month, :day) = parts;
227 if (calendar == Calendar.iso) return '${padded(year, 4)}-W${padded(month, 2)}-$day';
228 return '${padded(year, 4)}-${padded(month, 2)}-${padded(day, 2)}';
229 }
230 if (calendar == Calendar.jd) return '${fixed + 1721425}';
231 if (calendar == Calendar.fixed) return '$fixed';
232 final count = fixed + 1137142;
233 final longCount = [count ~/ 144000, count ~/ 7200 % 20, count ~/ 360 % 20, count ~/ 20 % 18, count % 20].join('.');
234 final haab = (count + 348) % 365;
235 return '$longCount ${(count + 3) % 13 + 1} ${tzolkinNames[(count + 19) % 20]} ${haab % 20} ${haabNames[haab ~/ 20]}';
236}

238int parseDate(Calendar calendar, String text) {
239 final int fixed;
240 if (ymd.contains(calendar)) {
241 if (!RegExp(r'^[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}$').hasMatch(text)) throw DateError('FORMAT');
242 final [year, month, day] = text.split('-').map(int.parse).toList();
243 if (year == 0) throw DateError('YEAR');
244 final days = lengths(calendar, year);
245 if (month < 1 || month > days.length) throw DateError('MONTH');
246 if (day < 1 || day > days[month - 1]) throw DateError('DAY');
247 fixed = fixedDate(calendar, year, month, day);
248 } else if (calendar == Calendar.iso) {
249 if (!RegExp(r'^[0-9]{1,5}-W[0-9]{1,2}-[0-9]$').hasMatch(text)) throw DateError('FORMAT');
250 final [yearText, weekText, dayText] = text.split('-');
251 final year = int.parse(yearText), week = int.parse(weekText.substring(1)), day = int.parse(dayText);
252 if (year == 0) throw DateError('YEAR');
253 if (week < 1 || week > weeks(year)) throw DateError('WEEK');
254 if (day < 1 || day > 7) throw DateError('DAY');
255 fixed = fixedDate(calendar, year, week, day);
256 } else if (calendar == Calendar.mayan) {
257 if (!RegExp(r'^[0-9]{1,2}(?:\.[0-9]{1,2}){4}$').hasMatch(text)) throw DateError('FORMAT');
258 final [b, k, t, u, n] = text.split('.').map(int.parse).toList();
259 if (k > 19 || t > 19 || u > 17 || n > 19) throw DateError('FIELD');
260 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142;
261 } else {
262 if (!RegExp(r'^-?[0-9]{1,8}$').hasMatch(text)) throw DateError('FORMAT');
263 fixed = int.parse(text) - (calendar == Calendar.jd ? 1721425 : 0);
264 }
265 if (!inRange(fixed)) throw DateError('RANGE');
266 return fixed;
267}

272int? shift(Calendar calendar, int fixed, int years, int months) {
273 final (:year, :month, :day) = partsOf(calendar, fixed)!;
274 final targetYear = year + years;
275 if (calendar == Calendar.iso) {
276 if (targetYear < 1 || targetYear > 99999) return null;
277 return fixedDate(calendar, targetYear, math.min(month, weeks(targetYear)), day);
278 }
279 var targetMonth = month;
280 if (calendar == Calendar.hebrew) {
281 var name = names(calendar, year)[month - 1];
282 final newNames = names(calendar, targetYear);
283 if (!newNames.contains(name)) name = name == 'Adar' ? 'AdarII' : 'Adar';
284 targetMonth = newNames.indexOf(name) + 1;
285 }
286 final absolute = monthIndex(calendar, targetYear) + targetMonth - 1 + months;
287 int finalYear;
288 if (calendar == Calendar.hebrew) {
289 finalYear = floorDiv(19 * absolute + 234, 235);
290 while (monthIndex(calendar, finalYear) > absolute) {
291 finalYear--;
292 }
293 while (monthIndex(calendar, finalYear + 1) <= absolute) {
294 finalYear++;
295 }
296 } else {
297 finalYear = floorDiv(absolute, monthNames[calendar]!.length) + 1;
298 }
299 if (finalYear < 1 || finalYear > 99999) return null;
300 final finalMonth = absolute - monthIndex(calendar, finalYear) + 1;
301 final finalDay = math.min(day, lengths(calendar, finalYear)[finalMonth - 1]);
302 return fixedDate(calendar, finalYear, finalMonth, finalDay);
303}

324Easter easter(int year, Calendar calendar) {
325 final golden = year % 19;
326 int epact;
327 if (calendar == Calendar.gregorian) {
328 final century = year ~/ 100 + 1;
329 epact = (14 + 11 * golden - 3 * century ~/ 4 + (5 + 8 * century) ~/ 25) % 30;
330 if (epact == 0 || epact == 1 && golden > 10) epact++;
331 } else {
332 epact = (14 + 11 * golden) % 30;
333 }
334 final moon = fixedDate(calendar, year, 4, 19) - epact;
335 final sunday = moon + 7 - moon % 7;
336 final index = -start(calendar, year) % 7;
337 var letters = 'ABCDEFG'[index];
338 if (leap(calendar, year)) letters += 'ABCDEFG'[(index - 1) % 7];
339 return (epact: epact, letters: letters, moon: moon, sunday: sunday);
340}

375void validateArguments(String command, List<String> args) {
376 if (command == 'EASTER' || command == 'COMPUTUS') {
377 ensure(validYear(args[0]) && (args[1] == 'GREGORIAN' || args[1] == 'JULIAN'));
378 return;
379 }
380 if (const {'PASSOVER', 'HOLIDAYS', 'MOLAD'}.contains(command)) {
381 ensure(validYear(args[0]));
382 return;
383 }
384 final allowed = switch (command) {
385 'INFO' || 'MONTHS' || 'MONTHCAL' || 'NTH' || 'SPAN' => ymd,
386 'LEAP' || 'NEWYEAR' => yearCalendars,
387 _ => Calendar.values,
388 };
389 ensure(isCalendar(args[0], allowed));
390 switch (command) {
391 case 'CONVERT':
392 ensure(args[2] == 'TO' && isCalendar(args[3], Calendar.values));
393 case 'DIFF':
394 ensure(isCalendar(args[2], Calendar.values));
395 case 'LEAP' || 'MONTHS' || 'MONTHCAL' || 'NTH' || 'NEWYEAR':
396 ensure(validYear(args[1]));
397 if (command == 'MONTHCAL' || command == 'NTH') ensure(RegExp(r'^[0-9]{1,2}$').hasMatch(args[2]));
398 if (command == 'NTH') {
399 ensure(const {'1', '2', '3', '4', '5', 'LAST'}.contains(args[3]));
400 ensure(weekdays.contains(args[4]));
401 }
402 case 'COUNT':
403 ensure(weekdays.contains(args[3]));
404 case 'ADD':
405 ensure(RegExp(r'^-?[0-9]{1,7}$').hasMatch(args[2]));
406 ensure(const {'DAYS', 'WEEKS', 'MONTHS', 'YEARS'}.contains(args[3]));
407 final calendar = calendarNamed(args[0]);
408 if (args[3] == 'MONTHS') ensure(ymd.contains(calendar));
409 if (args[3] == 'YEARS') ensure(yearCalendars.contains(calendar));
410 case 'ROUND':
411 ensure(RegExp(r'^[0-9]{1,2}$').hasMatch(args[2]));
412 final number = int.parse(args[2]);
413 ensure(number >= 1 && number <= 13 && tzolkinNames.contains(args[3]));
414 ensure(RegExp(r'^[0-9]{1,2}$').hasMatch(args[4]));
415 ensure(haabNames.contains(args[5]));
416 ensure(int.parse(args[4]) <= (args[5] == 'Uayeb' ? 4 : 19));
417 }
418}

438String spanText(Calendar calendar, int first, int last) {
439 final from = math.min(first, last), to = math.max(first, last);
440 var low = 0, high = partsOf(calendar, to)!.year - partsOf(calendar, from)!.year + 2;
441 while (low + 1 < high) {
442 final middle = (low + high) ~/ 2;
443 final shifted = shift(calendar, from, middle, 0);
444 if (shifted != null && shifted <= to) {
445 low = middle;
446 } else {
447 high = middle;
448 }
449 }
450 final years = low;
451 low = 0;
452 high = 32;
453 while (low + 1 < high) {
454 final middle = (low + high) ~/ 2;
455 final shifted = shift(calendar, from, years, middle);
456 if (shifted != null && shifted <= to) {
457 low = middle;
458 } else {
459 high = middle;
460 }
461 }
462 final months = low;
463 final days = to - shift(calendar, from, years, months)!;
464 return '${first > last ? '-' : ''}${years}y ${months}m ${days}d';
465}

467String roundDate(Calendar calendar, int fixed, int number, String tzName, int day, String month) {
468 final targetTz = tzolkinNames.indexOf(tzName);
469 final targetHaab = haabNames.indexOf(month) * 20 + day;
470 var candidate = fixed;
471 while ((candidate + 1137142 + 3) % 13 + 1 != number || (candidate + 1137142 + 19) % 20 != targetTz) {
472 candidate++;
473 }
474 for (; candidate <= fixed + 18979; candidate += 260) {
475 if ((candidate + 1137142 + 348) % 365 == targetHaab) return formatDate(calendar, candidate);
476 }
477 return 'NONE';
478}

480List<String> execute(String command, List<String> args) {
481 validateArguments(command, args);
482 if (command == 'EASTER' || command == 'COMPUTUS') {
483 final year = int.parse(args[0]);
484 if (year > 9999) return ['OUT OF RANGE'];
485 final kind = calendarNamed(args[1]);
486 final (:epact, :letters, :moon, :sunday) = easter(year, kind);
487 if (command == 'EASTER') return [formatDate(Calendar.gregorian, sunday)];
488 return [
489 'golden=${year % 19 + 1} epact=$epact letters=$letters '
490 'moon=${formatDate(kind, moon)} easter=${formatDate(kind, sunday)}',
491 ];
492 }
493 if (const {'PASSOVER', 'HOLIDAYS', 'MOLAD'}.contains(command)) {
494 final year = int.parse(args[0]);
495 if (command == 'MOLAD') {
496 final (:day, :parts, :start, :rules) = molad(year);
497 if (!inRange(start)) return ['OUT OF RANGE'];
498 final applied = rules.isEmpty ? 'NONE' : rules.join('+');
499 return [
500 '${weekdayName(day - 1373427)} ${parts ~/ 1080}h ${parts % 1080}p $applied '
501 '${weekdayName(start)} ${formatDate(Calendar.gregorian, start)}',
502 ];
503 }
504 if (year > 9999) return ['OUT OF RANGE'];
505 if (command == 'HOLIDAYS') return holidayLines(year);
506 final hebrewYear = year + 3760;
507 final month = names(Calendar.hebrew, hebrewYear).indexOf('Nisan') + 1;
508 final fixed = fixedDate(Calendar.hebrew, hebrewYear, month, 15);
509 return ['${formatDate(Calendar.gregorian, fixed)} ${weekdayName(fixed)}'];
510 }
511 final calendar = calendarNamed(args[0]);
512 if (const {'LEAP', 'MONTHS', 'MONTHCAL', 'NTH', 'NEWYEAR'}.contains(command)) {
513 final year = int.parse(args[1]);
514 switch (command) {
515 case 'LEAP':
516 var text = '${leap(calendar, year) ? 'YES' : 'NO'} ${start(calendar, year + 1) - start(calendar, year)}';
517 if (calendar == Calendar.hebrew) text += ' ${yearType(year)}';
518 return [text];
519 case 'MONTHS':
520 final days = lengths(calendar, year);
521 return [[for (final (index, name) in names(calendar, year).indexed) '$name=${days[index]}'].join(' ')];
522 case 'NEWYEAR':
523 if (year > 9999) return ['OUT OF RANGE'];
524 final entries = newYears(calendar, start(Calendar.gregorian, year), start(Calendar.gregorian, year + 1) - 1);
525 if (entries.isEmpty) return ['NONE'];
526 return [entries.map((entry) => '${entry.$1}:${formatDate(Calendar.gregorian, entry.$2)}').join(' ')];
527 case 'MONTHCAL':
528 return calendarLines(calendar, year, int.parse(args[2]));
529 default:
530 final (:first, :last) = monthBounds(calendar, year, int.parse(args[2]));
531 final weekday = weekdays.indexOf(args[4]);
532 final fixed = args[3] == 'LAST'
533 ? last - (last - weekday) % 7
534 : first + (weekday - first) % 7 + 7 * (int.parse(args[3]) - 1);
535 return [fixed > last ? 'NONE' : formatDate(calendar, fixed)];
536 }
537 }
538 if (command == 'VALID') {
539 try {
540 return ['YES ${formatDate(calendar, parseDate(calendar, args[1]))}'];
541 } on DateError catch (error) {
542 return ['NO ${error.reason}'];
543 }
544 }
545 final first = parseDate(calendar, args[1]);
546 switch (command) {
547 case 'DIFF':
548 return ['${parseDate(calendarNamed(args[2]), args[3]) - first}'];
549 case 'COUNT':
550 final last = parseDate(calendar, args[2]);
551 final low = math.min(first, last), high = math.max(first, last);
552 final occurrence = low + (weekdays.indexOf(args[3]) - low) % 7;
553 return ['${occurrence > high ? 0 : (high - occurrence) ~/ 7 + 1}'];
554 case 'SPAN':
555 return [spanText(calendar, first, parseDate(calendar, args[2]))];
556 case 'CONVERT':
557 return [formatDate(calendarNamed(args[3]), first)];
558 case 'WEEKDAY':
559 return [weekdayName(first)];
560 case 'ALL':
561 final result = <String>[];
562 for (final target in Calendar.values) {
563 var text = formatDate(target, first);
564 if (ymd.contains(target) && text != 'OUT OF RANGE') {
565 final parts = partsOf(target, first)!;
566 text += ' ${names(target, parts.year)[parts.month - 1]}';
567 }
568 result.add('${target.label} $text');
569 }
570 return result;
571 case 'INFO':
572 final (:year, :month, :day) = partsOf(calendar, first)!;
573 var weekday = weekdayName(first);
574 if (calendar == Calendar.french) weekday = month <= 12 ? decadeNames[(day - 1) % 10] : festivalNames[day - 1];
575 final ordinal = first - start(calendar, year) + 1;
576 final length = start(calendar, year + 1) - start(calendar, year);
577 return ['$weekday $day ${names(calendar, year)[month - 1]} $year day=$ordinal/$length'];
578 case 'ROUND':
579 return [roundDate(calendar, first, int.parse(args[2]), args[3], int.parse(args[4]), args[5])];
580 default:
581 final amount = int.parse(args[2]);
582 final result = switch (args[3]) {
583 'DAYS' => first + amount,
584 'WEEKS' => first + 7 * amount,
585 'YEARS' => shift(calendar, first, amount, 0),
586 _ => shift(calendar, first, 0, amount),
587 };
588 return [result == null ? 'OUT OF RANGE' : formatDate(calendar, result)];
589 }
590}

592void main() {
593 final count = int.parse(stdin.readLineSync()!);
594 final output = StringBuffer();
595 for (var i = 0; i < count; i++) {
596 final words = stdin.readLineSync()!.split(' ');
597 final command = words.first, args = words.sublist(1);
598 List<String> lines;
599 if (arity[command] != args.length) {
600 lines = ['INVALID COMMAND'];
601 } else {
602 try {
603 lines = execute(command, args);
604 } on CommandError {
605 lines = ['INVALID COMMAND'];
606 } on DateError catch (error) {
607 lines = [error.reason == 'RANGE' ? 'OUT OF RANGE' : 'INVALID DATE'];
608 }
609 }
610 for (final line in lines) {
611 output.writeln(line);
612 }
613 }
614 stdout.write(output);
