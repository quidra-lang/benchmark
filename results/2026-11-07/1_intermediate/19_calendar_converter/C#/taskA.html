78 static bool Leap(Calendar calendar, long year) => calendar switch {
79 Calendar.GREGORIAN => year % 4 == 0 && (year % 100 != 0 || year % 400 == 0),
80 Calendar.JULIAN => year % 4 == 0,
81 Calendar.ISLAMIC => Mod(year, 30) is 2 or 5 or 7 or 10 or 13 or 16 or 18 or 21 or 24 or 26 or 29,
82 Calendar.HEBREW => Mod(7 * year + 1, 19) < 7,
83 Calendar.PERSIAN => (Mod(year - 474, 2820) + 474 + 38) * 682 % 2816 < 682,
84 Calendar.COPTIC or Calendar.ETHIOPIC => Mod(year, 4) == 3,
85 Calendar.FRENCH => year % 4 == 0 && Mod(year, 400) is not (100 or 200 or 300) && year % 4000 != 0,
86 Calendar.ISO => Weeks(year) == 53,
87 _ => throw new ArgumentException($"no years in {calendar}"),

91 static Molad MoladOf(long year) {

94 int parts = Mod(quantity, 25920);
95 int weekday = Mod(day - 1373427, 7);

97 long postponed = day;
98 if (parts >= 19440) {
99 postponed += 1;
100 rules.Add("ZAKEN");

102 postponed += 2;
103 rules.Add("GATARAD");

105 postponed += 1;
106 rules.Add("BETUTAKPAT");
107 }
108 if (Mod(postponed - 1373427, 7) is 0 or 3 or 5) {
109 postponed += 1;
110 rules.Add("ADU");
111 }
112 return new Molad(day, parts, postponed - 1373427, rules);

114 static long Start(Calendar calendar, long year) {
115 long previous = year - 1;
116 switch (calendar) {
117 case Calendar.GREGORIAN: return 365 * previous + previous / 4 - previous / 100 + previous / 400 + 1;
118 case Calendar.JULIAN: return 365 * previous + previous / 4 - 1;
119 case Calendar.ISLAMIC: return 227015 + 354 * previous + (3 + 11 * year) / 30;
120 case Calendar.HEBREW: return MoladOf(year).Start;
121 case Calendar.PERSIAN: {
122 long a = year - 474;
123 long b = Mod(a, 2820) + 474;
124 return 226896 + 1029983 * FloorDiv(a, 2820) + 365 * (b - 1) + (682 * b - 110) / 2816;
125 }
126 case Calendar.COPTIC: return 103605 + 365 * previous + year / 4;
127 case Calendar.ETHIOPIC: return 2796 + 365 * previous + year / 4;
128 case Calendar.FRENCH: return 654415 + 365 * previous + previous / 4 - previous / 100 + previous / 400 - previous / 4000;
129 case Calendar.ISO: {
130 long januaryFour = Start(Calendar.GREGORIAN, year) + 3;
131 return januaryFour - (januaryFour - 1) % 7;
132 }
133 default: throw new ArgumentException($"no years in {calendar}");
134 }

166 static long YearOf(Calendar calendar, long fixedDay) {
167 if (fixedDay < Start(calendar, 1)) return 0;
168 long low = 1, high = 100000;
169 while (low + 1 < high) {
170 long middle = (low + high) / 2;
171 if (Start(calendar, middle) <= fixedDay) low = middle;
172 else high = middle;
173 }
174 return low;
175 }
176 static DateParts? PartsOf(Calendar calendar, long fixedDay) {
177 long year = YearOf(calendar, fixedDay);
178 if (year == 0) return null;
179 int offset = (int)(fixedDay - Start(calendar, year));
180 if (calendar == Calendar.ISO) return new DateParts(year, offset / 7 + 1, offset % 7 + 1);
181 int[] days = Lengths(calendar, year);
182 int month = 0;
183 while (offset >= days[month]) {
184 offset -= days[month];
185 month++;
186 }
187 return new DateParts(year, month + 1, offset + 1);
188 }
189 static string FormatDate(Calendar calendar, long fixedDay) {
190 if (!InRange(fixedDay)) return "OUT OF RANGE";
191 if (YEAR_CALENDARS.Contains(calendar)) {
192 DateParts? parts = PartsOf(calendar, fixedDay);
193 if (parts == null) return "OUT OF RANGE";
194 if (calendar == Calendar.ISO) return $"{parts.Year:D4}-W{parts.Month:D2}-{parts.Day}";
195 return $"{parts.Year:D4}-{parts.Month:D2}-{parts.Day:D2}";
196 }
197 if (calendar == Calendar.JD) return (fixedDay + 1721425).ToString();
198 if (calendar == Calendar.FIXED) return fixedDay.ToString();
199 long count = fixedDay + 1137142;
200 long[] longCount = [count / 144000, count / 7200 % 20, count / 360 % 20, count / 20 % 18, count % 20];
201 int haab = Mod(count + 348, 365);
202 return $"{string.Join(".", longCount)} {(count + 3) % 13 + 1} {TZ[Mod(count + 19, 20)]} {haab % 20} {HB[haab / 20]}";
203 }
204 static long ParseDate(Calendar calendar, string text) {
205 long fixedDay;
206 if (YMD.Contains(calendar)) {
207 if (!FullMatch(text, "[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}")) throw new DateError("FORMAT");
208 string[] fields = text.Split('-');
209 long year = long.Parse(fields[0]);
210 int month = int.Parse(fields[1]), day = int.Parse(fields[2]);
211 if (year == 0) throw new DateError("YEAR");
212 int[] days = Lengths(calendar, year);
213 if (month < 1 || month > days.Length) throw new DateError("MONTH");
214 if (day < 1 || day > days[month - 1]) throw new DateError("DAY");
215 fixedDay = FixedDate(calendar, year, month, day);
216 } else if (calendar == Calendar.ISO) {
217 if (!FullMatch(text, "[0-9]{1,5}-W[0-9]{1,2}-[0-9]")) throw new DateError("FORMAT");
218 string[] fields = text.Split('-');
219 long year = long.Parse(fields[0]);
220 int week = int.Parse(fields[1][1..]), day = int.Parse(fields[2]);
221 if (year == 0) throw new DateError("YEAR");
222 if (week < 1 || week > Weeks(year)) throw new DateError("WEEK");
223 if (day < 1 || day > 7) throw new DateError("DAY");
224 fixedDay = FixedDate(calendar, year, week, day);
225 } else if (calendar == Calendar.MAYAN) {
226 if (!FullMatch(text, @"[0-9]{1,2}(?:\.[0-9]{1,2}){4}")) throw new DateError("FORMAT");
227 long[] v = [.. text.Split('.').Select(long.Parse)];
228 if (v[1] > 19 || v[2] > 19 || v[3] > 17 || v[4] > 19) throw new DateError("FIELD");
229 fixedDay = (((v[0] * 20 + v[1]) * 20 + v[2]) * 18 + v[3]) * 20 + v[4] - 1137142;
230 } else {
231 if (!FullMatch(text, "-?[0-9]{1,8}")) throw new DateError("FORMAT");
232 fixedDay = long.Parse(text) - (calendar == Calendar.JD ? 1721425 : 0);
233 }
234 if (!InRange(fixedDay)) throw new DateError("RANGE");
235 return fixedDay;
236 }
237 static long MonthIndex(Calendar calendar, long year) {
238 if (calendar == Calendar.HEBREW) return MonthsBefore(year);
239 return MONTH_NAMES[calendar].Count * (year - 1);

241 static long? Shift(Calendar calendar, long fixedDay, long years, long months) {
242 DateParts parts = PartsOf(calendar, fixedDay)!;
243 long targetYear = parts.Year + years;
244 if (calendar == Calendar.ISO) {
245 if (targetYear < 1 || targetYear > 99999) return null;
246 return FixedDate(calendar, targetYear, Math.Min(parts.Month, Weeks(targetYear)), parts.Day);
247 }
248 int month = parts.Month;
249 if (calendar == Calendar.HEBREW) {
250 string oldName = Names(calendar, parts.Year)[month - 1];
251 List<string> newNames = Names(calendar, targetYear);
252 if (!newNames.Contains(oldName)) oldName = oldName == "Adar" ? "AdarII" : "Adar";
253 month = newNames.IndexOf(oldName) + 1;
254 }
255 long absolute = MonthIndex(calendar, targetYear) + month - 1 + months;
256 long finalYear;
257 if (calendar == Calendar.HEBREW) {
258 finalYear = FloorDiv(19 * absolute + 234, 235);
259 while (MonthIndex(calendar, finalYear) > absolute) finalYear--;
260 while (MonthIndex(calendar, finalYear + 1) <= absolute) finalYear++;
261 } else {
262 finalYear = FloorDiv(absolute, MONTH_NAMES[calendar].Count) + 1;
263 }
264 if (finalYear < 1 || finalYear > 99999) return null;
265 int finalMonth = (int)(absolute - MonthIndex(calendar, finalYear) + 1);
266 int finalDay = Math.Min(parts.Day, Lengths(calendar, finalYear)[finalMonth - 1]);
267 return FixedDate(calendar, finalYear, finalMonth, finalDay);
268 }
269 static (long First, long Last) MonthBounds(Calendar calendar, long year, int month) {
270 int[] days = Lengths(calendar, year);
271 if (month < 1 || month > days.Length) throw new DateError("MONTH");
272 long first = FixedDate(calendar, year, month, 1);
273 return (first, first + days[month - 1] - 1);
274 }

282 static Easter EasterOf(int year, Calendar calendar) {
283 int golden = year % 19;
284 int epact;
285 if (calendar == Calendar.GREGORIAN) {
286 int century = year / 100 + 1;
287 epact = Mod(14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25, 30);
288 if (epact == 0 || epact == 1 && golden > 10) epact++;
289 } else {
290 epact = (14 + 11 * golden) % 30;
291 }
292 long moon = FixedDate(calendar, year, 4, 19) - epact;
293 long sunday = moon + 7 - moon % 7;
294 int index = Mod(-Start(calendar, year), 7);
295 string letters = "ABCDEFG"[index].ToString();
296 if (Leap(calendar, year)) letters += "ABCDEFG"[Mod(index - 1, 7)];
297 return new Easter(epact, letters, moon, sunday);
298 }

322 static bool ValidYear(string text) => FullMatch(text, "[0-9]{1,5}") && int.Parse(text) >= 1;
323 static bool IsCalendar(string text, IEnumerable<Calendar> allowed) => allowed.Any(calendar => calendar.ToString() == text);
324 static void ValidateArguments(string command, string[] args) {
325 if (command is "EASTER" or "COMPUTUS") {
326 Require(ValidYear(args[0]) && (args[1] is "GREGORIAN" or "JULIAN"));
327 return;
328 }
329 if (command is "PASSOVER" or "HOLIDAYS" or "MOLAD") {
330 Require(ValidYear(args[0]));
331 return;
332 }
333 IEnumerable<Calendar> allowed = command switch {
334 "INFO" or "MONTHS" or "MONTHCAL" or "NTH" or "SPAN" => YMD,
335 "LEAP" or "NEWYEAR" => YEAR_CALENDARS,
336 _ => CALENDARS,
337 };
338 Require(IsCalendar(args[0], allowed));
339 switch (command) {
340 case "CONVERT":
341 Require(args[2] == "TO" && IsCalendar(args[3], CALENDARS));
342 break;
343 case "DIFF":
344 Require(IsCalendar(args[2], CALENDARS));
345 break;
346 case "LEAP" or "MONTHS" or "MONTHCAL" or "NTH" or "NEWYEAR":
347 Require(ValidYear(args[1]));
348 if (command is "MONTHCAL" or "NTH") Require(FullMatch(args[2], "[0-9]{1,2}"));
349 if (command == "NTH") {
350 Require(args[3] is "1" or "2" or "3" or "4" or "5" or "LAST");
351 Require(WEEKDAYS.Contains(args[4]));
352 }
353 break;
354 case "COUNT":
355 Require(WEEKDAYS.Contains(args[3]));
356 break;
357 case "ADD":
358 Require(FullMatch(args[2], "-?[0-9]{1,7}"));
359 Require(args[3] is "DAYS" or "WEEKS" or "MONTHS" or "YEARS");
360 Calendar calendar = Enum.Parse<Calendar>(args[0]);
361 if (args[3] == "MONTHS") Require(YMD.Contains(calendar));
362 if (args[3] == "YEARS") Require(YEAR_CALENDARS.Contains(calendar));
363 break;
364 case "ROUND":
365 Require(FullMatch(args[2], "[0-9]{1,2}"));
366 Require(int.Parse(args[2]) is >= 1 and <= 13 && TZ.Contains(args[3]));
367 Require(FullMatch(args[4], "[0-9]{1,2}"));
368 Require(HB.Contains(args[5]));
369 Require(int.Parse(args[4]) <= (args[5] == "Uayeb" ? 4 : 19));
370 break;
371 }

385 }

388 static string SpanText(Calendar calendar, long first, long last) {
389 long from = Math.Min(first, last), to = Math.Max(first, last);
390 long low = 0, high = PartsOf(calendar, to)!.Year - PartsOf(calendar, from)!.Year + 2;
391 while (low + 1 < high) {
392 long middle = (low + high) / 2;
393 long? shifted = Shift(calendar, from, middle, 0);
394 if (shifted <= to) low = middle;
395 else high = middle;
396 }
397 long years = low;
398 low = 0;
399 high = 32;
400 while (low + 1 < high) {
401 long middle = (low + high) / 2;
402 long? shifted = Shift(calendar, from, years, middle);
403 if (shifted <= to) low = middle;
404 else high = middle;
405 }
406 long months = low;
407 long days = to - Shift(calendar, from, years, months)!.Value;
408 return $"{(first > last ? "-" : "")}{years}y {months}m {days}d";
409 }
410 static string RoundDate(Calendar calendar, long fixedDay, int number, string tzName, int day, string month) {
411 int targetTz = TZ.IndexOf(tzName);
412 int targetHaab = HB.IndexOf(month) * 20 + day;
413 long candidate = fixedDay;
414 while ((candidate + 1137142 + 3) % 13 + 1 != number || (candidate + 1137142 + 19) % 20 != targetTz) candidate++;
415 for (; candidate <= fixedDay + 18979; candidate += 260) {
416 if ((candidate + 1137142 + 348) % 365 == targetHaab) return FormatDate(calendar, candidate);
417 }
418 return "NONE";
419 }
420 static List<string> Execute(string command, string[] args) {
421 ValidateArguments(command, args);
422 if (command is "EASTER" or "COMPUTUS") {
423 int year = int.Parse(args[0]);
424 if (year > 9999) return ["OUT OF RANGE"];
425 Calendar kind = Enum.Parse<Calendar>(args[1]);
426 Easter easter = EasterOf(year, kind);
427 if (command == "EASTER") return [FormatDate(Calendar.GREGORIAN, easter.Sunday)];
428 return [$"golden={year % 19 + 1} epact={easter.Epact} letters={easter.Letters} " +
429 $"moon={FormatDate(kind, easter.Moon)} easter={FormatDate(kind, easter.Sunday)}"];
430 }
431 if (command is "PASSOVER" or "HOLIDAYS" or "MOLAD") {
432 int year = int.Parse(args[0]);
433 if (command == "MOLAD") {
434 Molad molad = MoladOf(year);
435 if (!InRange(molad.Start)) return ["OUT OF RANGE"];
436 string applied = molad.Rules.Count == 0 ? "NONE" : string.Join("+", molad.Rules);
437 return [$"{WeekdayName(molad.Day - 1373427)} {molad.Parts / 1080}h {molad.Parts % 1080}p {applied} " +
438 $"{WeekdayName(molad.Start)} {FormatDate(Calendar.GREGORIAN, molad.Start)}"];
439 }
440 if (year > 9999) return ["OUT OF RANGE"];
441 if (command == "HOLIDAYS") return HolidayLines(year);
442 long hebrewYear = year + 3760;
443 int month = Names(Calendar.HEBREW, hebrewYear).IndexOf("Nisan") + 1;
444 long fixedDay = FixedDate(Calendar.HEBREW, hebrewYear, month, 15);
445 return [$"{FormatDate(Calendar.GREGORIAN, fixedDay)} {WeekdayName(fixedDay)}"];
446 }
447 Calendar calendar = Enum.Parse<Calendar>(args[0]);
448 if (command is "LEAP" or "MONTHS" or "MONTHCAL" or "NTH" or "NEWYEAR") {
449 long year = long.Parse(args[1]);
450 switch (command) {
451 case "LEAP": {
452 string text = $"{(Leap(calendar, year) ? "YES" : "NO")} {Start(calendar, year + 1) - Start(calendar, year)}";
453 if (calendar == Calendar.HEBREW) text += " " + YearType(year);
454 return [text];
455 }
456 case "MONTHS":
457 return [string.Join(" ", Names(calendar, year).Zip(Lengths(calendar, year), (name, days) => $"{name}={days}"))];
458 case "NEWYEAR": {
459 if (year > 9999) return ["OUT OF RANGE"];
460 List<(long Year, long Start)> entries = NewYears(calendar, Start(Calendar.GREGORIAN, year), Start(Calendar.GREGORIAN, year + 1) - 1);
461 if (entries.Count == 0) return ["NONE"];
462 return [string.Join(" ", entries.Select(entry => $"{entry.Year}:{FormatDate(Calendar.GREGORIAN, entry.Start)}"))];
463 }
464 case "MONTHCAL":
465 return CalendarLines(calendar, year, int.Parse(args[2]));
466 default: {
467 (long first, long last) = MonthBounds(calendar, year, int.Parse(args[2]));
468 int weekday = WEEKDAYS.IndexOf(args[4]);
469 long fixedDay = args[3] == "LAST" ? last - Mod(last - weekday, 7)
470 : first + Mod(weekday - first, 7) + 7 * (int.Parse(args[3]) - 1);
471 return [fixedDay > last ? "NONE" : FormatDate(calendar, fixedDay)];
472 }
473 }
474 }
475 if (command == "VALID") {
476 try {
477 return ["YES " + FormatDate(calendar, ParseDate(calendar, args[1]))];
478 } catch (DateError error) {
479 return ["NO " + error.Message];
480 }
481 }
482 long date = ParseDate(calendar, args[1]);
483 switch (command) {
484 case "DIFF":
485 return [(ParseDate(Enum.Parse<Calendar>(args[2]), args[3]) - date).ToString()];
486 case "COUNT": {
487 long other = ParseDate(calendar, args[2]);
488 long low = Math.Min(date, other), high = Math.Max(date, other);
489 long occurrence = low + Mod(WEEKDAYS.IndexOf(args[3]) - low, 7);
490 return [(occurrence > high ? 0 : (high - occurrence) / 7 + 1).ToString()];
491 }
492 case "SPAN":
493 return [SpanText(calendar, date, ParseDate(calendar, args[2]))];
494 case "CONVERT":
495 return [FormatDate(Enum.Parse<Calendar>(args[3]), date)];
496 case "WEEKDAY":
497 return [WeekdayName(date)];
498 case "ALL": {
499 List<string> result = [];
500 foreach (Calendar target in CALENDARS) {
501 string text = FormatDate(target, date);
502 if (YMD.Contains(target) && text != "OUT OF RANGE") {
503 DateParts parts = PartsOf(target, date)!;
504 text += " " + Names(target, parts.Year)[parts.Month - 1];
505 }
506 result.Add($"{target} {text}");
507 }
508 return result;
509 }
510 case "INFO": {
511 DateParts parts = PartsOf(calendar, date)!;
512 string weekday = WeekdayName(date);
513 if (calendar == Calendar.FRENCH) weekday = parts.Month <= 12 ? DC[(parts.Day - 1) % 10] : FS[parts.Day - 1];
514 long ordinal = date - Start(calendar, parts.Year) + 1;
515 long length = Start(calendar, parts.Year + 1) - Start(calendar, parts.Year);
516 return [$"{weekday} {parts.Day} {Names(calendar, parts.Year)[parts.Month - 1]} {parts.Year} day={ordinal}/{length}"];
517 }
518 case "ROUND":
519 return [RoundDate(calendar, date, int.Parse(args[2]), args[3], int.Parse(args[4]), args[5])];
520 default: {
521 int amount = int.Parse(args[2]);
522 long? result = args[3] switch {
523 "DAYS" => date + amount,
524 "WEEKS" => date + 7L * amount,
525 "YEARS" => Shift(calendar, date, amount, 0),
526 _ => Shift(calendar, date, 0, amount),
527 };
528 return [result is long day ? FormatDate(calendar, day) : "OUT OF RANGE"];
529 }
530 }
531 }
532 static void Main() {
533 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
534 int count = int.Parse(Console.ReadLine()!);
535 for (int i = 0; i < count; i++) {
536 string[] words = Console.ReadLine()!.Split(' ');
537 string command = words[0];
538 string[] args = words[1..];
539 List<string> lines;
540 if (!ARITY.TryGetValue(command, out int arity) || arity != args.Length) {
541 lines = ["INVALID COMMAND"];

544 lines = Execute(command, args);

546 lines = ["INVALID COMMAND"];
547 } catch (DateError error) {
548 lines = [error.Message == "RANGE" ? "OUT OF RANGE" : "INVALID DATE"];
549 }
550 }
551 foreach (string line in lines) Console.WriteLine(line);
552 }
