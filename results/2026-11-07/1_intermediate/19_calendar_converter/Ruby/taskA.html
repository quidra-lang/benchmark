34def leap?(calendar, year)
35 case calendar
36 when 'GREGORIAN' then (year % 4).zero? && (year % 100 != 0 || (year % 400).zero?)
37 when 'JULIAN' then (year % 4).zero?
38 when 'ISLAMIC' then [2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29].include?(year % 30)
39 when 'HEBREW' then (7 * year + 1) % 19 < 7
40 when 'PERSIAN'
41 b = (year - 474) % 2820 + 474
42 (b + 38) * 682 % 2816 < 682
43 when 'COPTIC', 'ETHIOPIC' then year % 4 == 3
44 when 'FRENCH' then (year % 4).zero? && ![100, 200, 300].include?(year % 400) && year % 4000 != 0
45 else (start('ISO', year + 1) - start('ISO', year)) / 7 == 53
46 end
47end

49def months_before(year)
50 (235 * year - 234) / 19
51end

53def molad(year)
54 quantity = 5604 + 765433 * months_before(year)
55 day, parts = quantity.divmod(25920)
56 weekday = (day - 1373427) % 7
57 rules = []
58 postponed = day
59 if parts >= 19440
60 postponed += 1
61 rules << 'ZAKEN'
62 elsif !leap?('HEBREW', year) && weekday == 2 && parts >= 9924
63 postponed += 2
64 rules << 'GATARAD'
65 elsif leap?('HEBREW', year - 1) && weekday == 1 && parts >= 16789
66 postponed += 1
67 rules << 'BETUTAKPAT'
68 end
69 if [0, 3, 5].include?((postponed - 1373427) % 7)
70 postponed += 1
71 rules << 'ADU'
72 end
73 [day, parts, postponed - 1373427, rules]
74end

76def start(calendar, year)
77 previous = year - 1
78 case calendar
79 when 'GREGORIAN' then 365 * previous + previous / 4 - previous / 100 + previous / 400 + 1
80 when 'JULIAN' then 365 * previous + previous / 4 - 1
81 when 'ISLAMIC' then 227015 + 354 * previous + (3 + 11 * year) / 30
82 when 'HEBREW' then molad(year)[2]
83 when 'PERSIAN'
84 a = year - 474
85 b = a % 2820 + 474
86 226896 + 1029983 * (a / 2820) + 365 * (b - 1) + (682 * b - 110) / 2816
87 when 'COPTIC' then 103605 + 365 * previous + year / 4
88 when 'ETHIOPIC' then 2796 + 365 * previous + year / 4
89 when 'FRENCH'
90 654415 + 365 * previous + previous / 4 - previous / 100 + previous / 400 - previous / 4000
91 else
92 january_four = start('GREGORIAN', year) + 3
93 january_four - (january_four - 1) % 7
94 end
95end

129def fixed_date(calendar, year, month, day)
130 return start(calendar, year) + 7 * (month - 1) + day - 1 if calendar == 'ISO'
131 start(calendar, year) + lengths(calendar, year).take(month - 1).sum + day - 1
132end

134def year_of(calendar, fixed)
135 return 0 if fixed < start(calendar, 1)
136 (1..100000).bsearch { |year| start(calendar, year) > fixed } - 1
137end

139def parts_of(calendar, fixed)
140 year = year_of(calendar, fixed)
141 return nil if year.zero?
142 offset = fixed - start(calendar, year)
143 return [year, offset / 7 + 1, offset % 7 + 1] if calendar == 'ISO'
144 lengths(calendar, year).each.with_index(1) do |days, month|
145 return [year, month, offset + 1] if offset < days
146 offset -= days
147 end
148 raise 'calendar year length'
149end

151def format_date(calendar, fixed)
152 return 'OUT OF RANGE' unless in_range?(fixed)
153 if YEAR_CALENDARS.include?(calendar)
154 parts = parts_of(calendar, fixed)
155 return 'OUT OF RANGE' if parts.nil?
156 return format('%04d-W%02d-%d', *parts) if calendar == 'ISO'
157 return format('%04d-%02d-%02d', *parts)
158 end
159 return (fixed + 1721425).to_s if calendar == 'JD'
160 return fixed.to_s if calendar == 'FIXED'
161 count = fixed + 1137142
162 long_count = [count / 144000, count / 7200 % 20, count / 360 % 20, count / 20 % 18, count % 20]
163 tz_number = (count + 3) % 13 + 1
164 tz_name = TZ[(count + 19) % 20]
165 haab = (count + 348) % 365
166 "#{long_count.join('.')} #{tz_number} #{tz_name} #{haab % 20} #{HB[haab / 20]}"
167end

169def parse_date(calendar, text)
170 if YMD.include?(calendar)
171 raise DateError, 'FORMAT' unless text.match?(/\A[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}\z/)
172 year, month, day = text.split('-').map(&:to_i)
173 raise DateError, 'YEAR' if year.zero?
174 days = lengths(calendar, year)
175 raise DateError, 'MONTH' unless month.between?(1, days.length)
176 raise DateError, 'DAY' unless day.between?(1, days[month - 1])
177 fixed = fixed_date(calendar, year, month, day)
178 elsif calendar == 'ISO'
179 raise DateError, 'FORMAT' unless text.match?(/\A[0-9]{1,5}-W[0-9]{1,2}-[0-9]\z/)
180 year_text, week_text, day_text = text.split('-')
181 year = year_text.to_i
182 week = week_text[1..].to_i
183 day = day_text.to_i
184 raise DateError, 'YEAR' if year.zero?
185 weeks = (start(calendar, year + 1) - start(calendar, year)) / 7
186 raise DateError, 'WEEK' unless week.between?(1, weeks)
187 raise DateError, 'DAY' unless day.between?(1, 7)
188 fixed = fixed_date(calendar, year, week, day)
189 elsif calendar == 'MAYAN'
190 raise DateError, 'FORMAT' unless text.match?(/\A[0-9]{1,2}(?:\.[0-9]{1,2}){4}\z/)
191 b, k, t, u, n = text.split('.').map(&:to_i)
192 raise DateError, 'FIELD' if k > 19 || t > 19 || u > 17 || n > 19
193 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142
194 else
195 raise DateError, 'FORMAT' unless text.match?(/\A-?[0-9]{1,8}\z/)
196 fixed = text.to_i - (calendar == 'JD' ? 1721425 : 0)
197 end
198 raise DateError, 'RANGE' unless in_range?(fixed)
199 fixed
200end

202def month_index(calendar, year)
203 return months_before(year) if calendar == 'HEBREW'
204 count = %w[COPTIC ETHIOPIC FRENCH].include?(calendar) ? 13 : 12
205 count * (year - 1)

208def shift(calendar, fixed, years, months)
209 year, month, day = parts_of(calendar, fixed)
210 target_year = year + years
211 if calendar == 'ISO'
212 return nil unless target_year.between?(1, 99999)
213 weeks = (start(calendar, target_year + 1) - start(calendar, target_year)) / 7
214 return fixed_date(calendar, target_year, [month, weeks].min, day)
215 end
216 if calendar == 'HEBREW'
217 old_name = names(calendar, year)[month - 1]
218 new_names = names(calendar, target_year)
219 old_name = old_name == 'Adar' ? 'AdarII' : 'Adar' unless new_names.include?(old_name)
220 month = new_names.index(old_name) + 1
221 end
222 absolute = month_index(calendar, target_year) + month - 1 + months
223 if calendar == 'HEBREW'
224 final_year = (19 * absolute + 234) / 235
225 final_year -= 1 while month_index(calendar, final_year) > absolute
226 final_year += 1 while month_index(calendar, final_year + 1) <= absolute
227 else
228 final_year = absolute / names(calendar, target_year).length + 1
229 end
230 return nil unless final_year.between?(1, 99999)
231 final_month = absolute - month_index(calendar, final_year) + 1
232 final_day = [day, lengths(calendar, final_year)[final_month - 1]].min
233 fixed_date(calendar, final_year, final_month, final_day)
234end

236def month_bounds(calendar, year, month)
237 days = lengths(calendar, year)
238 raise DateError, 'MONTH' unless month.between?(1, days.length)
239 first = fixed_date(calendar, year, month, 1)
240 [first, first + days[month - 1] - 1]
241end

243def new_years(calendar, first, last)
244 year = [1, year_of(calendar, first)].max
245 year += 1 while start(calendar, year) < first
246 result = []
247 while start(calendar, year) <= last
248 result << [year, start(calendar, year)]
249 year += 1
250 end
251 result
252end

254def easter(year, calendar)
255 golden = year % 19
256 if calendar == 'GREGORIAN'
257 century = year / 100 + 1
258 epact = (14 + 11 * golden - 3 * century / 4 + (5 + 8 * century) / 25) % 30
259 epact += 1 if epact.zero? || (epact == 1 && golden > 10)
260 else
261 epact = (14 + 11 * golden) % 30
262 end
263 moon = fixed_date(calendar, year, 4, 19) - epact
264 sunday = moon + 7 - moon % 7
265 index = -start(calendar, year) % 7
266 letters = 'ABCDEFG'[index]
267 letters += 'ABCDEFG'[(index - 1) % 7] if leap?(calendar, year)
268 [epact, letters, moon, sunday]
269end

299def holiday_lines(year)
300 first = start('GREGORIAN', year)
301 last = start('GREGORIAN', year + 1) - 1
302 gregorian_easter = easter(year, 'GREGORIAN')[3]
303 orthodox_easter = easter(year, 'JULIAN')[3]
304 movable = [['AshWednesday', -46], ['GoodFriday', -2], ['Easter', 0], ['Ascension', 39], ['Pentecost', 49]]
305 entries = movable.each_with_index.map do |(name, offset), order|

308 entries << [orthodox_easter, 5, 'OrthodoxEaster', 'JULIAN']
309 HOLIDAYS.each.with_index(6) do |(name, calendar, month_name, day), order|
310 candidate_year = [1, year_of(calendar, first)].max
311 while start(calendar, candidate_year) <= last

315 if day <= lengths(calendar, candidate_year)[month - 1]

317 entries << [fixed, order, name, calendar] if fixed.between?(first, last)
318 end
319 candidate_year += 1
320 end
321 end
322 entries.sort_by { |fixed, order| [fixed, order] }.map do |fixed, _, name, calendar|
323 "#{format_date('GREGORIAN', fixed)} #{WEEKDAYS[fixed % 7]} #{name} #{calendar} #{format_date(calendar, fixed)}"
324 end

338def validate_arguments(command, args)
339 case command
340 when 'EASTER', 'COMPUTUS'
341 validate(valid_year?(args[0]) && %w[GREGORIAN JULIAN].include?(args[1]))
342 return
343 when 'PASSOVER', 'HOLIDAYS', 'MOLAD'
344 validate(valid_year?(args[0]))
345 return
346 end
347 calendar = args[0]
348 allowed = case command
349 when 'INFO', 'MONTHS', 'MONTHCAL', 'NTH', 'SPAN' then YMD
350 when 'LEAP', 'NEWYEAR' then YEAR_CALENDARS
351 else CALENDARS
352 end
353 validate(allowed.include?(calendar))
354 case command
355 when 'CONVERT'
356 validate(args[2] == 'TO' && CALENDARS.include?(args[3]))
357 when 'DIFF'
358 validate(CALENDARS.include?(args[2]))
359 when 'LEAP', 'MONTHS', 'MONTHCAL', 'NTH', 'NEWYEAR'
360 validate(valid_year?(args[1]))
361 validate(args[2].match?(/\A[0-9]{1,2}\z/)) if %w[MONTHCAL NTH].include?(command)
362 if command == 'NTH'
363 validate(%w[1 2 3 4 5 LAST].include?(args[3]))
364 validate(WEEKDAYS.include?(args[4]))
365 end
366 when 'COUNT'
367 validate(WEEKDAYS.include?(args[3]))
368 when 'ADD'
369 validate(args[2].match?(/\A-?[0-9]{1,7}\z/))
370 validate(%w[DAYS WEEKS MONTHS YEARS].include?(args[3]))
371 validate(YMD.include?(calendar)) if args[3] == 'MONTHS'
372 validate(YEAR_CALENDARS.include?(calendar)) if args[3] == 'YEARS'
373 when 'ROUND'
374 validate(args[2].match?(/\A[0-9]{1,2}\z/))
375 validate(args[2].to_i.between?(1, 13) && TZ.include?(args[3]))
376 validate(args[4].match?(/\A[0-9]{1,2}\z/))
377 validate(HB.include?(args[5]))
378 validate(args[4].to_i <= (args[5] == 'Uayeb' ? 4 : 19))
379 end
380end

382def calendar_lines(calendar, year, month)
383 first, last = month_bounds(calendar, year, month)
384 return ['OUT OF RANGE'] unless in_range?(first) && in_range?(last)
385 title = "#{names(calendar, year)[month - 1]} #{year} " \
386 "(#{format_date('GREGORIAN', first)} to #{format_date('GREGORIAN', last)})"
387 result = [title, 'Mo Tu We Th Fr Sa Su']
388 (first - (first - 1) % 7).step(last, 7) do |cursor|
389 fields = (cursor...cursor + 7).map do |fixed|
390 fixed.between?(first, last) ? format('%2d', fixed - first + 1) : ' '
391 end
392 result << fields.join(' ').rstrip
393 end
394 result
395end

397def span_text(calendar, first, last)
398 negative = first > last
399 first, last = last, first if negative
400 start_year = parts_of(calendar, first)[0]
401 end_year = parts_of(calendar, last)[0]
402 beyond = lambda do |years, months|
403 shifted = shift(calendar, first, years, months)
404 shifted.nil? || shifted > last
405 end
406 years = (1..end_year - start_year + 1).bsearch { |count| beyond.call(count, 0) } - 1
407 months = (1..31).bsearch { |count| beyond.call(years, count) } - 1
408 days = last - shift(calendar, first, years, months)
409 "#{negative ? '-' : ''}#{years}y #{months}m #{days}d"
410end

412def round_date(calendar, fixed, number, tz_name, day, month)
413 target_tz = TZ.index(tz_name)
414 target_haab = HB.index(month) * 20 + day
415 first_match = (fixed..).find do |candidate|
416 count = candidate + 1137142
417 (count + 3) % 13 + 1 == number && (count + 19) % 20 == target_tz
418 end
419 first_match.step(fixed + 18979, 260) do |candidate|
420 return format_date(calendar, candidate) if (candidate + 1137142 + 348) % 365 == target_haab
421 end
422 'NONE'
423end

425def execute(command, args)
426 validate_arguments(command, args)
427 if %w[EASTER COMPUTUS].include?(command)
428 year = args[0].to_i
429 return ['OUT OF RANGE'] if year > 9999
430 kind = args[1]
431 epact, letters, moon, sunday = easter(year, kind)
432 return [format_date('GREGORIAN', sunday)] if command == 'EASTER'
433 return ["golden=#{year % 19 + 1} epact=#{epact} letters=#{letters} " \
434 "moon=#{format_date(kind, moon)} easter=#{format_date(kind, sunday)}"]
435 end
436 if %w[PASSOVER HOLIDAYS MOLAD].include?(command)
437 year = args[0].to_i
438 if command == 'MOLAD'
439 day, parts, fixed, rules = molad(year)
440 return ['OUT OF RANGE'] unless in_range?(fixed)
441 applied = rules.empty? ? 'NONE' : rules.join('+')
442 return ["#{WEEKDAYS[(day - 1373427) % 7]} #{parts / 1080}h #{parts % 1080}p #{applied} " \
443 "#{WEEKDAYS[fixed % 7]} #{format_date('GREGORIAN', fixed)}"]
444 end
445 return ['OUT OF RANGE'] if year > 9999
446 return holiday_lines(year) if command == 'HOLIDAYS'
447 hebrew_year = year + 3760
448 month = names('HEBREW', hebrew_year).index('Nisan') + 1
449 fixed = fixed_date('HEBREW', hebrew_year, month, 15)
450 return ["#{format_date('GREGORIAN', fixed)} #{WEEKDAYS[fixed % 7]}"]
451 end
452 calendar = args[0]
453 if %w[LEAP MONTHS MONTHCAL NTH NEWYEAR].include?(command)
454 year = args[1].to_i
455 case command
456 when 'LEAP'
457 text = "#{leap?(calendar, year) ? 'YES' : 'NO'} #{start(calendar, year + 1) - start(calendar, year)}"
458 text += " #{year_type(year)}" if calendar == 'HEBREW'
459 return [text]
460 when 'MONTHS'
461 return [names(calendar, year).zip(lengths(calendar, year)).map { |name, days| "#{name}=#{days}" }.join(' ')]
462 when 'NEWYEAR'
463 return ['OUT OF RANGE'] if year > 9999
464 entries = new_years(calendar, start('GREGORIAN', year), start('GREGORIAN', year + 1) - 1)
465 return ['NONE'] if entries.empty?
466 return [entries.map { |new_year, fixed| "#{new_year}:#{format_date('GREGORIAN', fixed)}" }.join(' ')]
467 end
468 month = args[2].to_i
469 return calendar_lines(calendar, year, month) if command == 'MONTHCAL'
470 first, last = month_bounds(calendar, year, month)
471 weekday = WEEKDAYS.index(args[4])
472 fixed = if args[3] == 'LAST'
473 last - (last - weekday) % 7
474 else
475 first + (weekday - first) % 7 + 7 * (args[3].to_i - 1)
476 end
477 return [fixed > last ? 'NONE' : format_date(calendar, fixed)]
478 end
479 if command == 'VALID'
480 begin
481 fixed = parse_date(calendar, args[1])
482 rescue DateError => error
483 return ["NO #{error.message}"]
484 end
485 return ["YES #{format_date(calendar, fixed)}"]
486 end
487 first = parse_date(calendar, args[1])
488 case command
489 when 'DIFF'
490 [(parse_date(args[2], args[3]) - first).to_s]
491 when 'COUNT', 'SPAN'
492 last = parse_date(calendar, args[2])
493 return [span_text(calendar, first, last)] if command == 'SPAN'
494 low, high = [first, last].minmax
495 occurrence = low + (WEEKDAYS.index(args[3]) - low) % 7
496 [(occurrence > high ? 0 : (high - occurrence) / 7 + 1).to_s]
497 when 'CONVERT'
498 [format_date(args[3], first)]
499 when 'WEEKDAY'
500 [WEEKDAYS[first % 7]]
501 when 'ALL'
502 CALENDARS.map do |target|
503 text = format_date(target, first)
504 if YMD.include?(target) && text != 'OUT OF RANGE'
505 year, month = parts_of(target, first)
506 text += " #{names(target, year)[month - 1]}"
507 end
508 "#{target} #{text}"
509 end
510 when 'INFO'
511 year, month, day = parts_of(calendar, first)
512 weekday = WEEKDAYS[first % 7]
513 weekday = month <= 12 ? DC[(day - 1) % 10] : FS[day - 1] if calendar == 'FRENCH'
514 ordinal = first - start(calendar, year) + 1
515 length = start(calendar, year + 1) - start(calendar, year)
516 ["#{weekday} #{day} #{names(calendar, year)[month - 1]} #{year} day=#{ordinal}/#{length}"]
517 when 'ROUND'
518 [round_date(calendar, first, args[2].to_i, args[3], args[4].to_i, args[5])]

524 when 'YEARS' then shift(calendar, first, amount, 0)

528 end
