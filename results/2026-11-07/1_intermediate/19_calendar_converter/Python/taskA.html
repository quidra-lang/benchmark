47def leap(calendar, year):
48 if calendar == "GREGORIAN":
49 return year % 4 == 0 and (year % 100 != 0 or year % 400 == 0)
50 if calendar == "JULIAN":
51 return year % 4 == 0
52 if calendar == "ISLAMIC":
53 return year % 30 in (2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29)
54 if calendar == "HEBREW":
55 return (7 * year + 1) % 19 < 7
56 if calendar == "PERSIAN":
57 b = (year - 474) % 2820 + 474
58 return ((b + 38) * 682) % 2816 < 682
59 if calendar in ("COPTIC", "ETHIOPIC"):
60 return year % 4 == 3
61 if calendar == "FRENCH":
62 return year % 4 == 0 and year % 400 not in (100, 200, 300) and year % 4000 != 0
63 return (start("ISO", year + 1) - start("ISO", year)) // 7 == 53

70def molad(year):
71 quantity = 5604 + 765433 * months_before(year)
72 day, parts = divmod(quantity, 25920)

77 postponed += 1
78 rules.append("ZAKEN")

80 postponed += 2

82 elif leap("HEBREW", year - 1) and weekday == 1 and parts >= 16789:
83 postponed += 1
84 rules.append("BETUTAKPAT")
85 if (postponed - 1373427) % 7 in (0, 3, 5):
86 postponed += 1
87 rules.append("ADU")

91def start(calendar, year):
92 previous = year - 1
93 if calendar == "GREGORIAN":
94 return 365 * previous + previous // 4 - previous // 100 + previous // 400 + 1
95 if calendar == "JULIAN":
96 return 365 * previous + previous // 4 - 1
97 if calendar == "ISLAMIC":
98 return 227015 + 354 * previous + (3 + 11 * year) // 30
99 if calendar == "HEBREW":
100 return molad(year)[2]
101 if calendar == "PERSIAN":
102 a = year - 474
103 b = a % 2820 + 474
104 return 226896 + 1029983 * (a // 2820) + 365 * (b - 1) + (682 * b - 110) // 2816
105 if calendar == "COPTIC":
106 return 103605 + 365 * previous + year // 4
107 if calendar == "ETHIOPIC":
108 return 2796 + 365 * previous + year // 4
109 if calendar == "FRENCH":
110 return 654415 + 365 * previous + previous // 4 - previous // 100 + previous // 400 - previous // 4000
111 january_four = start("GREGORIAN", year) + 3
112 return january_four - (january_four - 1) % 7

154def year_of(calendar, fixed):
155 if fixed < start(calendar, 1):
156 return 0
157 low = 1
158 high = 100000
159 while low + 1 < high:
160 middle = (low + high) // 2
161 if start(calendar, middle) <= fixed:
162 low = middle
163 else:
164 high = middle
165 return low

168def parts_of(calendar, fixed):
169 year = year_of(calendar, fixed)
170 if year == 0:
171 return None
172 offset = fixed - start(calendar, year)
173 if calendar == "ISO":
174 return year, offset // 7 + 1, offset % 7 + 1
175 for month, days in enumerate(lengths(calendar, year), 1):
176 if offset < days:
177 return year, month, offset + 1
178 offset -= days
179 raise ValueError("calendar year length")

182def format_date(calendar, fixed):
183 if not in_range(fixed):
184 return "OUT OF RANGE"
185 if calendar in YEAR_CALENDARS:
186 parts = parts_of(calendar, fixed)
187 if parts is None:
188 return "OUT OF RANGE"

190 if calendar == "ISO":
191 return f"{year:04d}-W{month:02d}-{day}"
192 return f"{year:04d}-{month:02d}-{day:02d}"
193 if calendar == "JD":
194 return str(fixed + 1721425)
195 if calendar == "FIXED":
196 return str(fixed)
197 count = fixed + 1137142
198 long_count = [
199 count // 144000,
200 count // 7200 % 20,
201 count // 360 % 20,
202 count // 20 % 18,
203 count % 20,
204 ]
205 tz_number = (count + 3) % 13 + 1
206 tz_name = TZ[(count + 19) % 20]
207 haab = (count + 348) % 365
208 return ".".join(map(str, long_count)) + f" {tz_number} {tz_name} {haab % 20} {HB[haab // 20]}"

211def parse_date(calendar, text):
212 if calendar in YMD:
213 if not re.fullmatch(r"[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}", text):
214 raise DateError("FORMAT")
215 year, month, day = map(int, text.split("-"))
216 if year == 0:
217 raise DateError("YEAR")
218 days = lengths(calendar, year)
219 if not 1 <= month <= len(days):
220 raise DateError("MONTH")
221 if not 1 <= day <= days[month - 1]:
222 raise DateError("DAY")
223 fixed = fixed_date(calendar, year, month, day)
224 elif calendar == "ISO":
225 if not re.fullmatch(r"[0-9]{1,5}-W[0-9]{1,2}-[0-9]", text):
226 raise DateError("FORMAT")
227 year_text, week_text, day_text = text.split("-")
228 year, week, day = int(year_text), int(week_text[1:]), int(day_text)
229 if year == 0:
230 raise DateError("YEAR")
231 weeks = (start(calendar, year + 1) - start(calendar, year)) // 7
232 if not 1 <= week <= weeks:
233 raise DateError("WEEK")
234 if not 1 <= day <= 7:
235 raise DateError("DAY")
236 fixed = fixed_date(calendar, year, week, day)
237 elif calendar == "MAYAN":
238 if not re.fullmatch(r"[0-9]{1,2}(?:\.[0-9]{1,2}){4}", text):
239 raise DateError("FORMAT")
240 b, k, t, u, n = map(int, text.split("."))
241 if k > 19 or t > 19 or u > 17 or n > 19:
242 raise DateError("FIELD")
243 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142
244 else:
245 if not re.fullmatch(r"-?[0-9]{1,8}", text):
246 raise DateError("FORMAT")
247 fixed = int(text) - (1721425 if calendar == "JD" else 0)
248 if not in_range(fixed):
249 raise DateError("RANGE")
250 return fixed

260def shift(calendar, fixed, years, months):
261 year, month, day = parts_of(calendar, fixed)
262 target_year = year + years
263 if calendar == "ISO":
264 if not 1 <= target_year <= 99999:
265 return None
266 weeks = (start(calendar, target_year + 1) - start(calendar, target_year)) // 7
267 return fixed_date(calendar, target_year, min(month, weeks), day)
268 if calendar == "HEBREW":
269 old_name = names(calendar, year)[month - 1]
270 new_names = names(calendar, target_year)
271 if old_name not in new_names:
272 old_name = "AdarII" if old_name == "Adar" else "Adar"
273 month = new_names.index(old_name) + 1
274 absolute = month_index(calendar, target_year) + month - 1 + months
275 if calendar == "HEBREW":
276 final_year = (19 * absolute + 234) // 235
277 while month_index(calendar, final_year) > absolute:
278 final_year -= 1
279 while month_index(calendar, final_year + 1) <= absolute:
280 final_year += 1
281 else:
282 count = len(names(calendar, target_year))
283 final_year = absolute // count + 1
284 if not 1 <= final_year <= 99999:
285 return None
286 final_month = absolute - month_index(calendar, final_year) + 1
287 final_day = min(day, lengths(calendar, final_year)[final_month - 1])
288 return fixed_date(calendar, final_year, final_month, final_day)

300 year = max(1, year_of(calendar, first))
301 while start(calendar, year) < first:
302 year += 1
303 result = []

310def easter(year, calendar):
311 golden = year % 19
312 if calendar == "GREGORIAN":
313 century = year // 100 + 1
314 epact = (14 + 11 * golden - 3 * century // 4 + (5 + 8 * century) // 25) % 30
315 if epact == 0 or (epact == 1 and golden > 10):
316 epact += 1
317 else:
318 epact = (14 + 11 * golden) % 30
319 moon = fixed_date(calendar, year, 4, 19) - epact
320 sunday = moon + 7 - moon % 7
321 index = -start(calendar, year) % 7
322 letters = "ABCDEFG"[index]
323 if leap(calendar, year):
324 letters += "ABCDEFG"[(index - 1) % 7]
325 return epact, letters, moon, sunday

445def calendar_lines(calendar, year, month):
446 first, last = month_bounds(calendar, year, month)
447 if not in_range(first) or not in_range(last):
448 return ["OUT OF RANGE"]
449 title = f"{names(calendar, year)[month - 1]} {year} "
450 title += f"({format_date('GREGORIAN', first)} to {format_date('GREGORIAN', last)})"
451 result = [title, "Mo Tu We Th Fr Sa Su"]
452 cursor = first - (first - 1) % 7
453 while cursor <= last:
454 fields = [f"{fixed - first + 1:2d}" if first <= fixed <= last else " "
455 for fixed in range(cursor, cursor + 7)]
456 result.append(" ".join(fields).rstrip())
457 cursor += 7
458 return result

461def span_text(calendar, first, last):
462 negative = first > last
463 if negative:
464 first, last = last, first
465 start_year = parts_of(calendar, first)[0]
466 end_year = parts_of(calendar, last)[0]
467 low = 0
468 high = end_year - start_year + 2
469 while low + 1 < high:
470 middle = (low + high) // 2
471 shifted = shift(calendar, first, middle, 0)
472 if shifted is not None and shifted <= last:
473 low = middle
474 else:
475 high = middle
476 years = low
477 low = 0
478 high = 32
479 while low + 1 < high:
480 middle = (low + high) // 2
481 shifted = shift(calendar, first, years, middle)
482 if shifted is not None and shifted <= last:
483 low = middle
484 else:
485 high = middle
486 months = low
487 days = last - shift(calendar, first, years, months)
488 sign = "-" if negative else ""
489 return f"{sign}{years}y {months}m {days}d"

492def round_date(calendar, fixed, number, tz_name, day, month):
493 target_tz = TZ.index(tz_name)
494 target_haab = HB.index(month) * 20 + day
495 candidate = fixed
496 while True:
497 count = candidate + 1137142

500 candidate += 1
501 while candidate <= fixed + 18979:
502 if (candidate + 1137142 + 348) % 365 == target_haab:
503 return format_date(calendar, candidate)
504 candidate += 260
505 return "NONE"

508def execute(command, args):
509 validate_arguments(command, args)
510 if command in ("EASTER", "COMPUTUS"):
511 year = int(args[0])
512 if year > 9999:
513 return ["OUT OF RANGE"]
514 kind = args[1]
515 epact, letters, moon, sunday = easter(year, kind)
516 if command == "EASTER":
517 return [format_date("GREGORIAN", sunday)]
518 return [
519 f"golden={year % 19 + 1} epact={epact} letters={letters} "
520 f"moon={format_date(kind, moon)} easter={format_date(kind, sunday)}"
521 ]
522 if command in ("PASSOVER", "HOLIDAYS", "MOLAD"):
523 year = int(args[0])
524 if command == "MOLAD":
525 day, parts, fixed, rules = molad(year)
526 if not in_range(fixed):
527 return ["OUT OF RANGE"]
528 applied = "+".join(rules) if rules else "NONE"
529 return [
530 f"{WEEKDAYS[(day - 1373427) % 7]} {parts // 1080}h "
531 f"{parts % 1080}p {applied} {WEEKDAYS[fixed % 7]} "
532 f"{format_date('GREGORIAN', fixed)}"
533 ]
534 if year > 9999:
535 return ["OUT OF RANGE"]
536 if command == "HOLIDAYS":
537 return holiday_lines(year)
538 hebrew_year = year + 3760
539 month = names("HEBREW", hebrew_year).index("Nisan") + 1
540 fixed = fixed_date("HEBREW", hebrew_year, month, 15)
541 return [f"{format_date('GREGORIAN', fixed)} {WEEKDAYS[fixed % 7]}"]
542 calendar = args[0]
543 if command in ("LEAP", "MONTHS", "MONTHCAL", "NTH", "NEWYEAR"):
544 year = int(args[1])
545 if command == "LEAP":
546 text = f"{'YES' if leap(calendar, year) else 'NO'} "
547 text += str(start(calendar, year + 1) - start(calendar, year))
548 if calendar == "HEBREW":
549 text += " " + year_type(year)
550 return [text]
551 if command == "MONTHS":
552 return [" ".join(
553 f"{name}={days}" for name, days in zip(names(calendar, year), lengths(calendar, year))
554 )]
555 if command == "NEWYEAR":
556 if year > 9999:
557 return ["OUT OF RANGE"]
558 first = start("GREGORIAN", year)
559 last = start("GREGORIAN", year + 1) - 1
560 entries = new_years(calendar, first, last)
561 return [" ".join(f"{y}:{format_date('GREGORIAN', f)}" for y, f in entries) or "NONE"]
562 month = int(args[2])
563 if command == "MONTHCAL":
564 return calendar_lines(calendar, year, month)
565 first, last = month_bounds(calendar, year, month)
566 weekday = WEEKDAYS.index(args[4])
567 if args[3] == "LAST":
568 fixed = last - (last - weekday) % 7
569 else:
570 fixed = first + (weekday - first) % 7 + 7 * (int(args[3]) - 1)
571 return ["NONE" if fixed > last else format_date(calendar, fixed)]
572 if command == "VALID":
573 try:
574 fixed = parse_date(calendar, args[1])
575 except DateError as error:
576 return ["NO " + str(error)]
577 return ["YES " + format_date(calendar, fixed)]
578 first = parse_date(calendar, args[1])
579 if command == "DIFF":
580 last = parse_date(args[2], args[3])
581 return [str(last - first)]
582 if command in ("COUNT", "SPAN"):
583 last = parse_date(calendar, args[2])
584 if command == "SPAN":
585 return [span_text(calendar, first, last)]
586 low, high = sorted((first, last))
587 occurrence = low + (WEEKDAYS.index(args[3]) - low) % 7
588 return [str(0 if occurrence > high else (high - occurrence) // 7 + 1)]
589 if command == "CONVERT":
590 return [format_date(args[3], first)]
591 if command == "WEEKDAY":
592 return [WEEKDAYS[first % 7]]
593 if command == "ALL":
594 result = []
595 for target in CALENDARS:
596 text = format_date(target, first)
597 if target in YMD and text != "OUT OF RANGE":
598 year, month, _ = parts_of(target, first)
599 text += " " + names(target, year)[month - 1]
600 result.append(target + " " + text)
601 return result
602 if command == "INFO":
603 year, month, day = parts_of(calendar, first)
604 weekday = WEEKDAYS[first % 7]
605 if calendar == "FRENCH":
606 weekday = DC[(day - 1) % 10] if month <= 12 else FS[day - 1]
607 ordinal = first - start(calendar, year) + 1
608 length = start(calendar, year + 1) - start(calendar, year)

610 if command == "ROUND":
611 return [round_date(calendar, first, int(args[2]), args[3], int(args[4]), args[5])]
612 amount = int(args[2])

614 if unit == "DAYS":

618 elif unit == "YEARS":
619 result = shift(calendar, first, amount, 0)
