51function leap(calendar, year) {
52 if (calendar === "GREGORIAN") return mod(year, 4) === 0 && (mod(year, 100) !== 0 || mod(year, 400) === 0);
53 if (calendar === "JULIAN") return mod(year, 4) === 0;
54 if (calendar === "ISLAMIC") return [2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29].includes(mod(year, 30));
55 if (calendar === "HEBREW") return mod(7 * year + 1, 19) < 7;
56 if (calendar === "PERSIAN") {
57 const b = mod(year - 474, 2820) + 474;
58 return mod((b + 38) * 682, 2816) < 682;
59 }
60 if (calendar === "COPTIC" || calendar === "ETHIOPIC") return mod(year, 4) === 3;
61 if (calendar === "FRENCH") {
62 return mod(year, 4) === 0 && ![100, 200, 300].includes(mod(year, 400)) && mod(year, 4000) !== 0;
63 }
64 return (start("ISO", year + 1) - start("ISO", year)) / 7 === 53;
65}

95function start(calendar, year) {
96 const previous = year - 1;
97 if (calendar === "GREGORIAN") {
98 return 365 * previous + floorDiv(previous, 4) - floorDiv(previous, 100) + floorDiv(previous, 400) + 1;
99 }
100 if (calendar === "JULIAN") return 365 * previous + floorDiv(previous, 4) - 1;
101 if (calendar === "ISLAMIC") return 227015 + 354 * previous + floorDiv(3 + 11 * year, 30);
102 if (calendar === "HEBREW") return molad(year).fixed;
103 if (calendar === "PERSIAN") {
104 const a = year - 474;
105 const b = mod(a, 2820) + 474;
106 return 226896 + 1029983 * floorDiv(a, 2820) + 365 * (b - 1) + floorDiv(682 * b - 110, 2816);
107 }
108 if (calendar === "COPTIC") return 103605 + 365 * previous + floorDiv(year, 4);
109 if (calendar === "ETHIOPIC") return 2796 + 365 * previous + floorDiv(year, 4);
110 if (calendar === "FRENCH") {

112 floorDiv(previous, 400) - floorDiv(previous, 4000);

114 const januaryFour = start("GREGORIAN", year) + 3;
115 return januaryFour - mod(januaryFour - 1, 7);

161function partsOf(calendar, fixed) {
162 const year = yearOf(calendar, fixed);
163 if (year === 0) return null;
164 let offset = fixed - start(calendar, year);
165 if (calendar === "ISO") return [year, floorDiv(offset, 7) + 1, mod(offset, 7) + 1];
166 for (const [index, days] of lengths(calendar, year).entries()) {
167 if (offset < days) return [year, index + 1, offset + 1];
168 offset -= days;
169 }
170 throw new Error("calendar year length");
171}

198function parseDate(calendar, text) {
199 let fixed;
200 if (YMD.includes(calendar)) {
201 if (!/^[0-9]{1,5}-[0-9]{1,2}-[0-9]{1,2}$/.test(text)) throw new DateError("FORMAT");
202 const [year, month, day] = text.split("-").map(Number);
203 if (year === 0) throw new DateError("YEAR");
204 const days = lengths(calendar, year);
205 if (month < 1 || month > days.length) throw new DateError("MONTH");
206 if (day < 1 || day > days[month - 1]) throw new DateError("DAY");
207 fixed = fixedDate(calendar, year, month, day);
208 } else if (calendar === "ISO") {
209 if (!/^[0-9]{1,5}-W[0-9]{1,2}-[0-9]$/.test(text)) throw new DateError("FORMAT");
210 const [yearText, weekText, dayText] = text.split("-");
211 const year = Number(yearText);
212 const week = Number(weekText.slice(1));
213 const day = Number(dayText);
214 if (year === 0) throw new DateError("YEAR");
215 const weeks = (start(calendar, year + 1) - start(calendar, year)) / 7;
216 if (week < 1 || week > weeks) throw new DateError("WEEK");
217 if (day < 1 || day > 7) throw new DateError("DAY");
218 fixed = fixedDate(calendar, year, week, day);
219 } else if (calendar === "MAYAN") {
220 if (!/^[0-9]{1,2}(?:\.[0-9]{1,2}){4}$/.test(text)) throw new DateError("FORMAT");
221 const [b, k, t, u, n] = text.split(".").map(Number);
222 if (k > 19 || t > 19 || u > 17 || n > 19) throw new DateError("FIELD");
223 fixed = (((b * 20 + k) * 20 + t) * 18 + u) * 20 + n - 1137142;
224 } else {
225 if (!/^-?[0-9]{1,8}$/.test(text)) throw new DateError("FORMAT");
226 fixed = Number(text) - (calendar === "JD" ? 1721425 : 0);
227 }
228 if (!inRange(fixed)) throw new DateError("RANGE");
229 return fixed;
230}

238function shift(calendar, fixed, years, months) {
239 let [year, month, day] = partsOf(calendar, fixed);
240 const targetYear = year + years;
241 if (calendar === "ISO") {
242 if (targetYear < 1 || targetYear > 99999) return null;
243 const weeks = (start(calendar, targetYear + 1) - start(calendar, targetYear)) / 7;
244 return fixedDate(calendar, targetYear, Math.min(month, weeks), day);
245 }
246 if (calendar === "HEBREW") {
247 let oldName = names(calendar, year)[month - 1];
248 const newNames = names(calendar, targetYear);
249 if (!newNames.includes(oldName)) oldName = oldName === "Adar" ? "AdarII" : "Adar";
250 month = newNames.indexOf(oldName) + 1;
251 }
252 const absolute = monthIndex(calendar, targetYear) + month - 1 + months;
253 let finalYear;
254 if (calendar === "HEBREW") {
255 finalYear = floorDiv(19 * absolute + 234, 235);
256 while (monthIndex(calendar, finalYear) > absolute) finalYear--;
257 while (monthIndex(calendar, finalYear + 1) <= absolute) finalYear++;
258 } else {
259 finalYear = floorDiv(absolute, names(calendar, targetYear).length) + 1;
260 }
261 if (finalYear < 1 || finalYear > 99999) return null;
262 const finalMonth = absolute - monthIndex(calendar, finalYear) + 1;
263 const finalDay = Math.min(day, lengths(calendar, finalYear)[finalMonth - 1]);
264 return fixedDate(calendar, finalYear, finalMonth, finalDay);
265}

285function easter(year, calendar) {
286 const golden = mod(year, 19);
287 let epact;
288 if (calendar === "GREGORIAN") {
289 const century = floorDiv(year, 100) + 1;
290 epact = mod(14 + 11 * golden - floorDiv(3 * century, 4) + floorDiv(5 + 8 * century, 25), 30);
291 if (epact === 0 || (epact === 1 && golden > 10)) epact += 1;
292 } else {
293 epact = mod(14 + 11 * golden, 30);
294 }
295 const moon = fixedDate(calendar, year, 4, 19) - epact;
296 const sunday = moon + 7 - mod(moon, 7);
297 const index = mod(-start(calendar, year), 7);
298 let letters = "ABCDEFG"[index];
299 if (leap(calendar, year)) letters += "ABCDEFG"[mod(index - 1, 7)];
300 return {epact, letters, moon, sunday};
301}

423function spanText(calendar, first, last) {
424 const negative = first > last;
425 if (negative) [first, last] = [last, first];
426 const startYear = partsOf(calendar, first)[0];
427 const endYear = partsOf(calendar, last)[0];
428 let low = 0;
429 let high = endYear - startYear + 2;
430 while (low + 1 < high) {
431 const middle = floorDiv(low + high, 2);
432 const shifted = shift(calendar, first, middle, 0);
433 if (shifted !== null && shifted <= last) low = middle;
434 else high = middle;
435 }
436 const years = low;
437 low = 0;
438 high = 32;
439 while (low + 1 < high) {
440 const middle = floorDiv(low + high, 2);
441 const shifted = shift(calendar, first, years, middle);
442 if (shifted !== null && shifted <= last) low = middle;
443 else high = middle;
444 }
445 const months = low;
446 const days = last - shift(calendar, first, years, months);
447 return (negative ? "-" : "") + years + "y " + months + "m " + days + "d";
448}

495 const year = Number(args[1]);
496 if (command === "LEAP") {
497 let text = (leap(calendar, year) ? "YES" : "NO") + " " + (start(calendar, year + 1) - start(calendar, year));
498 if (calendar === "HEBREW") text += " " + yearType(year);
499 return [text];
500 }
501 if (command === "MONTHS") {
502 const days = lengths(calendar, year);
503 return [names(calendar, year).map((name, i) => name + "=" + days[i]).join(" ")];
504 }
505 if (command === "NEWYEAR") {
506 if (year > 9999) return ["OUT OF RANGE"];
507 const first = start("GREGORIAN", year);
508 const last = start("GREGORIAN", year + 1) - 1;
509 const entries = newYears(calendar, first, last);
510 return [entries.map(([y, f]) => y + ":" + formatDate("GREGORIAN", f)).join(" ") || "NONE"];
511 }
512 const month = Number(args[2]);
513 if (command === "MONTHCAL") return calendarLines(calendar, year, month);
514 const [first, last] = monthBounds(calendar, year, month);
515 const weekday = WEEKDAYS.indexOf(args[4]);
516 let fixed;
517 if (args[3] === "LAST") fixed = last - mod(last - weekday, 7);
518 else fixed = first + mod(weekday - first, 7) + 7 * (Number(args[3]) - 1);
519 return [fixed > last ? "NONE" : formatDate(calendar, fixed)];
520 }
521 if (command === "VALID") {
522 try {
523 return ["YES " + formatDate(calendar, parseDate(calendar, args[1]))];
524 } catch (error) {
525 if (!(error instanceof DateError)) throw error;
526 return ["NO " + error.message];
527 }
528 }
529 const first = parseDate(calendar, args[1]);
530 if (command === "DIFF") {
531 const last = parseDate(args[2], args[3]);
532 return [String(last - first)];
533 }
534 if (command === "COUNT" || command === "SPAN") {
535 const last = parseDate(calendar, args[2]);
536 if (command === "SPAN") return [spanText(calendar, first, last)];
537 const low = Math.min(first, last);
538 const high = Math.max(first, last);
539 const occurrence = low + mod(WEEKDAYS.indexOf(args[3]) - low, 7);
540 return [String(occurrence > high ? 0 : floorDiv(high - occurrence, 7) + 1)];
541 }
542 if (command === "CONVERT") return [formatDate(args[3], first)];
543 if (command === "WEEKDAY") return [WEEKDAYS[mod(first, 7)]];
544 if (command === "ALL") {
545 return CALENDARS.map(target => {
546 let text = formatDate(target, first);
547 if (YMD.includes(target) && text !== "OUT OF RANGE") {
548 const [year, month] = partsOf(target, first);
549 text += " " + names(target, year)[month - 1];
550 }
551 return target + " " + text;
552 });
553 }
554 if (command === "INFO") {
555 const [year, month, day] = partsOf(calendar, first);
556 let weekday = WEEKDAYS[mod(first, 7)];
557 if (calendar === "FRENCH") weekday = month <= 12 ? DC[mod(day - 1, 10)] : FS[day - 1];
558 const ordinal = first - start(calendar, year) + 1;
559 const length = start(calendar, year + 1) - start(calendar, year);
560 return [weekday + " " + day + " " + names(calendar, year)[month - 1] + " " + year + " day=" + ordinal + "/" + length];
561 }
562 if (command === "ROUND") {
563 return [roundDate(calendar, first, Number(args[2]), args[3], Number(args[4]), args[5])];
564 }
565 const amount = Number(args[2]);
566 const unit = args[3];
567 let result;
568 if (unit === "DAYS") result = first + amount;
569 else if (unit === "WEEKS") result = first + 7 * amount;
570 else if (unit === "YEARS") result = shift(calendar, first, amount, 0);
571 else result = shift(calendar, first, 0, amount);
572 return [result === null ? "OUT OF RANGE" : formatDate(calendar, result)];
573}

575const lines = readFileSync(0, "utf8").split("\n");
576const output = [];
577for (let i = 1; i <= Number(lines[0]); i++) {
578 const [command, ...args] = lines[i].split(" ");
579 if (!ARITY.has(command) || args.length !== ARITY.get(command)) {
580 output.push("INVALID COMMAND");
581 continue;
582 }
583 try {
584 output.push(...execute(command, args));
585 } catch (error) {
586 if (error instanceof CommandError) output.push("INVALID COMMAND");
587 else if (error instanceof DateError) output.push(error.message === "RANGE" ? "OUT OF RANGE" : "INVALID DATE");
588 else throw error;

590}
591process.stdout.write(output.join("\n") + "\n");
