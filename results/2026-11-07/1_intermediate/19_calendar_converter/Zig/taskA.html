25fn number(text: []const u8, digits: usize, signed: bool) ?i64 {

29 for (body) |ch| if (!std.ascii.isDigit(ch)) return null;
30 return std.fmt.parseInt(i64, text, 10) catch null;

48fn molad(y: i64) Molad {
49 const q = 5604 + 765433 * @divFloor(235 * y - 234, 19);
50 const d = @divFloor(q, 25920);
51 const t = @mod(q, 25920);
52 const wd = @mod(d - 1373427, 7);
53 var r = d;
54 var rule: ?[]const u8 = null;
55 if (t >= 19440) {
56 r += 1;
57 rule = "ZAKEN";
58 } else if (!Calendar.HEBREW.leap(y) and wd == 2 and t >= 9924) {
59 r += 2;
60 rule = "GATARAD";
61 } else if (Calendar.HEBREW.leap(y - 1) and wd == 1 and t >= 16789) {
62 r += 1;
63 rule = "BETUTAKPAT";
64 }
65 const rw = @mod(r - 1373427, 7);
66 const adu = rw == 0 or rw == 3 or rw == 5;
67 return .{ .day = d, .parts = t, .fixed = r + @intFromBool(adu) - 1373427, .first_rule = rule, .adu = adu };
68}

90 fn leap(c: Calendar, y: i64) bool {
91 return switch (c) {
92 .GREGORIAN => @mod(y, 4) == 0 and (@mod(y, 100) != 0 or @mod(y, 400) == 0),
93 .JULIAN => @mod(y, 4) == 0,
94 .ISO => c.start(y + 1) - c.start(y) == 371,
95 .ISLAMIC => switch (@mod(y, 30)) {
96 2, 5, 7, 10, 13, 16, 18, 21, 24, 26, 29 => true,
97 else => false,
98 },
99 .HEBREW => @mod(7 * y + 1, 19) < 7,
100 .PERSIAN => blk: {
101 const b = @mod(y - 474, 2820) + 474;
102 break :blk @mod((b + 38) * 682, 2816) < 682;
103 },
104 .COPTIC, .ETHIOPIC => @mod(y, 4) == 3,
105 .FRENCH => @mod(y, 4) == 0 and @mod(y, 400) != 100 and @mod(y, 400) != 200 and @mod(y, 400) != 300 and @mod(y, 4000) != 0,
106 else => unreachable,
107 };
108 }
109 fn start(c: Calendar, y: i64) i64 {

112 .GREGORIAN => 365 * p + @divFloor(p, 4) - @divFloor(p, 100) + @divFloor(p, 400) + 1,
113 .JULIAN => 365 * p + @divFloor(p, 4) - 1,
114 .ISO => blk: {
115 const jan4 = Calendar.GREGORIAN.start(y) + 3;

120 .PERSIAN => blk: {
121 const a = y - 474;
122 const b = @mod(a, 2820) + 474;
123 break :blk 226896 + 1029983 * @divFloor(a, 2820) + 365 * (b - 1) + @divFloor(682 * b - 110, 2816);
124 },
125 .COPTIC => 103605 + 365 * p + @divFloor(y, 4),
126 .ETHIOPIC => 2796 + 365 * p + @divFloor(y, 4),
127 .FRENCH => 654415 + 365 * p + @divFloor(p, 4) - @divFloor(p, 100) + @divFloor(p, 400) - @divFloor(p, 4000),
128 else => unreachable,
129 };

153 fn monthDays(c: Calendar, y: i64, m: i64) i64 {
154 const extra: i64 = @intFromBool(c.leap(y));
155 return switch (c) {
156 .GREGORIAN, .JULIAN => if (m == 2) 28 + extra else if (m == 4 or m == 6 or m == 9 or m == 11) 30 else 31,
157 .ISLAMIC => if (@mod(m, 2) == 1) 30 else if (m == 12) 29 + extra else 29,
158 .PERSIAN => if (m <= 6) 31 else if (m <= 11) 30 else 29 + extra,
159 .COPTIC, .ETHIOPIC, .FRENCH => if (m <= 12) 30 else 5 + extra,
160 .HEBREW => blk: {
161 const length = c.start(y + 1) - c.start(y);
162 if (m == 2) break :blk if (@mod(length, 10) == 5) 30 else 29;
163 if (m == 3) break :blk if (@mod(length, 10) == 3) 29 else 30;
164 const common = [_]i64{ 30, 29, 30, 29, 30, 29, 30, 29, 30, 29, 30, 29 };
165 const intercalary = [_]i64{ 30, 29, 30, 29, 30, 30, 29, 30, 29, 30, 29, 30, 29 };
166 break :blk if (extra == 1) intercalary[ix(m - 1)] else common[ix(m - 1)];
167 },
168 else => unreachable,
169 };

171 fn fixed(c: Calendar, d: Date) i64 {
172 if (c == .ISO) return c.start(d.year) + 7 * (d.month - 1) + d.day - 1;
173 var f = c.start(d.year) + d.day - 1;
174 var m: i64 = 1;
175 while (m < d.month) : (m += 1) f += c.monthDays(d.year, m);
176 return f;
177 }
178 fn yearOf(c: Calendar, f: i64) i64 {
179 if (f < c.start(1)) return 0;
180 var low: i64 = 1;
181 var high: i64 = 100000;
182 while (low + 1 < high) {
183 const middle = @divFloor(low + high, 2);
184 if (c.start(middle) <= f) low = middle else high = middle;
185 }
186 return low;
187 }
188 fn parts(c: Calendar, f: i64) ?Date {
189 const y = c.yearOf(f);
190 if (y == 0) return null;
191 var offset = f - c.start(y);
192 if (c == .ISO) return .{ .year = y, .month = @divFloor(offset, 7) + 1, .day = @mod(offset, 7) + 1 };
193 var m: i64 = 1;
194 while (offset >= c.monthDays(y, m)) : (m += 1) offset -= c.monthDays(y, m);
195 return .{ .year = y, .month = m, .day = offset + 1 };
196 }
197 fn monthsBefore(c: Calendar, y: i64) i64 {
198 return if (c == .HEBREW) @divFloor(235 * y - 234, 19) else c.monthCount(y) * (y - 1);
199 }
200 fn shift(c: Calendar, f: i64, years: i64, months: i64) ?i64 {
201 const d = c.parts(f).?;
202 const next_year = d.year + years;
203 if (c == .ISO) {
204 if (next_year < 1 or next_year > 99999) return null;
205 const weeks = @divFloor(c.start(next_year + 1) - c.start(next_year), 7);
206 return c.fixed(.{ .year = next_year, .month = @min(d.month, weeks), .day = d.day });
207 }
208 var m = d.month;
209 if (c == .HEBREW) {
210 var name = c.names(d.year)[ix(m - 1)];
211 if (indexOf(c.names(next_year), name) == null) name = if (equal(name, "Adar")) "AdarII" else "Adar";
212 m = indexOf(c.names(next_year), name).? + 1;
213 }
214 const absolute = c.monthsBefore(next_year) + m - 1 + months;
215 var y: i64 = undefined;
216 if (c == .HEBREW) {
217 y = @divFloor(19 * absolute + 234, 235);
218 while (c.monthsBefore(y) > absolute) y -= 1;
219 while (c.monthsBefore(y + 1) <= absolute) y += 1;
220 } else y = @divFloor(absolute, c.monthCount(next_year)) + 1;
221 if (y < 1 or y > 99999) return null;
222 m = absolute - c.monthsBefore(y) + 1;
223 return c.fixed(.{ .year = y, .month = m, .day = @min(d.day, c.monthDays(y, m)) });
224 }

227fn parseDate(c: Calendar, text: []const u8) DateError!i64 {
228 var f: i64 = undefined;
229 if (c.isYear()) {
230 var split = std.mem.splitScalar(u8, text, '-');
231 const a = split.next() orelse return error.FORMAT;
232 var b = split.next() orelse return error.FORMAT;
233 const d = split.next() orelse return error.FORMAT;
234 if (split.next() != null) return error.FORMAT;
235 const y = number(a, 5, false) orelse return error.FORMAT;
236 if (c == .ISO) {
237 if (b.len == 0 or b[0] != 'W') return error.FORMAT;
238 b = b[1..];
239 }
240 const m = number(b, 2, false) orelse return error.FORMAT;
241 const day = number(d, if (c == .ISO) 1 else 2, false) orelse return error.FORMAT;
242 if (y == 0) return error.YEAR;
243 if (c == .ISO) {
244 if (m < 1 or m > @divFloor(c.start(y + 1) - c.start(y), 7)) return error.WEEK;
245 if (day < 1 or day > 7) return error.DAY;
246 } else {
247 if (m < 1 or m > c.monthCount(y)) return error.MONTH;
248 if (day < 1 or day > c.monthDays(y, m)) return error.DAY;
249 }
250 f = c.fixed(.{ .year = y, .month = m, .day = day });
251 } else if (c == .MAYAN) {
252 var split = std.mem.splitScalar(u8, text, '.');
253 var values: [5]i64 = undefined;
254 for (&values) |*v| v.* = number(split.next() orelse return error.FORMAT, 2, false) orelse return error.FORMAT;
255 if (split.next() != null) return error.FORMAT;
256 if (values[1] > 19 or values[2] > 19 or values[3] > 17 or values[4] > 19) return error.FIELD;
257 f = (((values[0] * 20 + values[1]) * 20 + values[2]) * 18 + values[3]) * 20 + values[4] - 1137142;
258 } else {
259 f = number(text, 8, true) orelse return error.FORMAT;
260 if (c == .JD) f -= 1721425;
261 }
262 if (!inRange(f)) return error.RANGE;
263 return f;
264}
265fn checkedDate(c: Calendar, text: []const u8) Error!i64 {
266 return parseDate(c, text) catch |err| return if (err == error.RANGE) error.OutOfRange else error.InvalidDate;
267}
268fn writeDate(out: *std.Io.Writer, c: Calendar, f: i64) Error!void {
269 if (!inRange(f)) return error.OutOfRange;
270 if (c.isYear()) {
271 const d = c.parts(f) orelse return error.OutOfRange;
272 if (c == .ISO) try out.print("{d:0>4}-W{d:0>2}-{d}", .{ d.year, d.month, d.day }) else try out.print("{d:0>4}-{d:0>2}-{d:0>2}", .{ d.year, d.month, d.day });
273 } else if (c == .MAYAN) {
274 const count = f + 1137142;
275 const h = @mod(count + 348, 365);
276 try out.print("{d}.{d}.{d}.{d}.{d} {d} {s} {d} {s}", .{
277 @divFloor(count, 144000), @mod(@divFloor(count, 7200), 20), @mod(@divFloor(count, 360), 20), @mod(@divFloor(count, 20), 18), @mod(count, 20),
278 @mod(count + 3, 13) + 1, tz[ix(@mod(count + 19, 20))], @mod(h, 20), hb[ix(@divFloor(h, 20))],
279 });
280 } else try out.print("{d}", .{f + @as(i64, if (c == .JD) 1721425 else 0)});
281}

284fn easter(y: i64, c: Calendar) Easter {
285 const g = @mod(y, 19);
286 var e = @mod(14 + 11 * g, 30);
287 if (c == .GREGORIAN) {
288 const century = @divFloor(y, 100) + 1;
289 e = @mod(14 + 11 * g - @divFloor(3 * century, 4) + @divFloor(5 + 8 * century, 25), 30);
290 if (e == 0 or (e == 1 and g > 10)) e += 1;
291 }
292 const p = c.fixed(.{ .year = y, .month = 4, .day = 19 }) - e;
293 const s = @mod(-c.start(y), 7);
294 return .{ .epact = e, .letters = .{ "ABCDEFG"[ix(s)], "ABCDEFG"[ix(@mod(s - 1, 7))] }, .letter_count = if (c.leap(y)) 2 else 1, .moon = p, .sunday = p + 7 - @mod(p, 7) };
295}

333fn writeHolidays(gpa: Allocator, out: *std.Io.Writer, y: i64) Error!void {
334 const first = Calendar.GREGORIAN.start(y);
335 const last = Calendar.GREGORIAN.start(y + 1) - 1;
336 const sunday = easter(y, .GREGORIAN).sunday;
337 var entries: std.ArrayList(Occurrence) = .empty;
338 const movable = [_][]const u8{ "AshWednesday", "GoodFriday", "Easter", "Ascension", "Pentecost" };
339 for (movable, [_]i64{ -46, -2, 0, 39, 49 }, 0..) |name, offset, i| try entries.append(gpa, .{ .fixed = sunday + offset, .order = i, .name = name, .calendar = .GREGORIAN });
340 try entries.append(gpa, .{ .fixed = easter(y, .JULIAN).sunday, .order = 5, .name = "OrthodoxEaster", .calendar = .JULIAN });
341 for (holidays, 6..) |holiday, order| {
342 const c = holiday.calendar;
343 var candidate = @max(1, c.yearOf(first));
344 while (c.start(candidate) <= last) : (candidate += 1) {

358 }

366fn validate(cmd: Command, a: []const []const u8) Error!void {
367 const count: usize = switch (cmd) {
368 .CONVERT, .DIFF, .COUNT, .ADD => 4,
369 .MONTHCAL, .SPAN => 3,
370 .NTH => 5,
371 .PASSOVER, .HOLIDAYS, .MOLAD => 1,
372 .ROUND => 6,
373 else => 2,
374 };
375 if (a.len != count) return error.InvalidCommand;
376 switch (cmd) {
377 .EASTER, .COMPUTUS => {
378 if (yearNumber(a[0]) == null or (!equal(a[1], "GREGORIAN") and !equal(a[1], "JULIAN"))) return error.InvalidCommand;
379 return;
380 },
381 .PASSOVER, .HOLIDAYS, .MOLAD => {
382 if (yearNumber(a[0]) == null) return error.InvalidCommand;
383 return;
384 },
385 else => {},
386 }
387 const c = calendar(a[0]) orelse return error.InvalidCommand;
388 switch (cmd) {
389 .INFO, .MONTHS, .MONTHCAL, .NTH, .SPAN => if (!c.isYmd()) return error.InvalidCommand,
390 .LEAP, .NEWYEAR => if (!c.isYear()) return error.InvalidCommand,
391 else => {},
392 }
393 switch (cmd) {
394 .CONVERT => if (!equal(a[2], "TO") or calendar(a[3]) == null) return error.InvalidCommand,
395 .DIFF => if (calendar(a[2]) == null) return error.InvalidCommand,
396 .LEAP, .MONTHS, .MONTHCAL, .NTH, .NEWYEAR => {
397 if (yearNumber(a[1]) == null) return error.InvalidCommand;
398 if (cmd == .MONTHCAL or cmd == .NTH) if (number(a[2], 2, false) == null) return error.InvalidCommand;
399 if (cmd == .NTH) {
400 if (indexOf(&.{ "1", "2", "3", "4", "5", "LAST" }, a[3]) == null or indexOf(&weekdays, a[4]) == null) return error.InvalidCommand;
401 }
402 },
403 .COUNT => if (indexOf(&weekdays, a[3]) == null) return error.InvalidCommand,
404 .ADD => {
405 if (number(a[2], 7, true) == null) return error.InvalidCommand;
406 const unit = std.meta.stringToEnum(Unit, a[3]) orelse return error.InvalidCommand;
407 if ((unit == .MONTHS and !c.isYmd()) or (unit == .YEARS and !c.isYear())) return error.InvalidCommand;
408 },
409 .ROUND => {
410 const n = number(a[2], 2, false) orelse return error.InvalidCommand;
411 const d = number(a[4], 2, false) orelse return error.InvalidCommand;
412 const month = indexOf(&hb, a[5]) orelse return error.InvalidCommand;
413 if (n < 1 or n > 13 or indexOf(&tz, a[3]) == null or d > @as(i64, if (month == 18) 4 else 19)) return error.InvalidCommand;
414 },
415 else => {},
416 }
417}

419fn writeMonth(out: *std.Io.Writer, c: Calendar, y: i64, m: i64, first: i64, last: i64) Error!void {
420 if (!inRange(first) or !inRange(last)) return error.OutOfRange;
421 try out.print("{s} {d} (", .{ c.names(y)[ix(m - 1)], y });
422 try writeDate(out, .GREGORIAN, first);
423 try out.writeAll(" to ");
424 try writeDate(out, .GREGORIAN, last);
425 try out.writeAll(")\nMo Tu We Th Fr Sa Su\n");
426 var cursor = first - @mod(first - 1, 7);
427 while (cursor <= last) : (cursor += 7) {
428 var row: [20]u8 = @splat(' ');
429 for (0..7) |i| {
430 const f = cursor + @as(i64, @intCast(i));
431 if (f < first or f > last) continue;
432 const d = f - first + 1;
433 if (d >= 10) row[i * 3] = '0' + @as(u8, @intCast(@divFloor(d, 10)));
434 row[i * 3 + 1] = '0' + @as(u8, @intCast(@mod(d, 10)));
435 }
436 try out.print("{s}\n", .{std.mem.trimEnd(u8, &row, " ")});
437 }
438}
439fn writeSpan(out: *std.Io.Writer, c: Calendar, f1: i64, f2: i64) Error!void {
440 const first = @min(f1, f2);
441 const last = @max(f1, f2);
442 var low: i64 = 0;
443 var high = c.parts(last).?.year - c.parts(first).?.year + 2;
444 while (low + 1 < high) {
445 const middle = @divFloor(low + high, 2);
446 if ((c.shift(first, middle, 0) orelse max_fixed + 1) <= last) low = middle else high = middle;
447 }
448 var months: i64 = 0;
449 while (true) {
450 const shifted = c.shift(first, low, months + 1) orelse break;
451 if (shifted > last) break;
452 months += 1;
453 }
454 try out.print("{s}{d}y {d}m {d}d", .{ if (f1 > f2) "-" else "", low, months, last - c.shift(first, low, months).? });
455}

457fn execute(gpa: Allocator, out: *std.Io.Writer, words: []const []const u8) Error!void {
458 const cmd = std.meta.stringToEnum(Command, words[0]) orelse return error.InvalidCommand;
459 const a = words[1..];
460 try validate(cmd, a);
461 switch (cmd) {
462 .EASTER, .COMPUTUS => {
463 const y = integer(a[0]);
464 if (y > 9999) return error.OutOfRange;
465 const c = calendar(a[1]).?;
466 const e = easter(y, c);
467 if (cmd == .EASTER) try writeDate(out, .GREGORIAN, e.sunday) else {
468 try out.print("golden={d} epact={d} letters={s} moon=", .{ @mod(y, 19) + 1, e.epact, e.letters[0..e.letter_count] });
469 try writeDate(out, c, e.moon);
470 try out.writeAll(" easter=");
471 try writeDate(out, c, e.sunday);
472 }
473 },
474 .PASSOVER, .HOLIDAYS, .MOLAD => {
475 const y = integer(a[0]);
476 if (cmd == .MOLAD) {
477 const m = molad(y);
478 if (!inRange(m.fixed)) return error.OutOfRange;
479 try out.print("{s} {d}h {d}p ", .{ weekdays[ix(@mod(m.day - 1373427, 7))], @divFloor(m.parts, 1080), @mod(m.parts, 1080) });
480 if (m.first_rule) |rule| try out.writeAll(rule);
481 if (m.adu) try out.writeAll(if (m.first_rule == null) "ADU" else "+ADU");
482 if (m.first_rule == null and !m.adu) try out.writeAll("NONE");
483 try out.print(" {s} ", .{weekdays[ix(@mod(m.fixed, 7))]});
484 try writeDate(out, .GREGORIAN, m.fixed);
485 } else {
486 if (y > 9999) return error.OutOfRange;
487 if (cmd == .HOLIDAYS) return writeHolidays(gpa, out, y);
488 const hy = y + 3760;
489 const m = indexOf(Calendar.HEBREW.names(hy), "Nisan").? + 1;
490 const f = Calendar.HEBREW.fixed(.{ .year = hy, .month = m, .day = 15 });
491 try writeDate(out, .GREGORIAN, f);
492 try out.print(" {s}", .{weekdays[ix(@mod(f, 7))]});
493 }
494 },
495 .LEAP, .MONTHS, .MONTHCAL, .NTH, .NEWYEAR => {
496 const c = calendar(a[0]).?;
497 const y = integer(a[1]);
498 switch (cmd) {
499 .LEAP => {
500 try out.print("{s} {d}", .{ if (c.leap(y)) "YES" else "NO", c.start(y + 1) - c.start(y) });
501 if (c == .HEBREW) try out.print(" {s}", .{c.yearType(y)});
502 },
503 .MONTHS => for (c.names(y), 1..) |name, m| {
504 if (m > 1) try out.writeAll(" ");
505 try out.print("{s}={d}", .{ name, c.monthDays(y, @intCast(m)) });
506 },
507 .NEWYEAR => {
508 if (y > 9999) return error.OutOfRange;
509 const first = Calendar.GREGORIAN.start(y);
510 const last = Calendar.GREGORIAN.start(y + 1) - 1;
511 var candidate = @max(1, c.yearOf(first));
512 var any = false;
513 while (c.start(candidate) <= last) : (candidate += 1) {
514 const f = c.start(candidate);
515 if (f < first) continue;
516 if (any) try out.writeAll(" ");
517 try out.print("{d}:", .{candidate});
518 try writeDate(out, .GREGORIAN, f);
519 any = true;
520 }
521 if (!any) try out.writeAll("NONE");
522 },
523 .MONTHCAL, .NTH => {
524 const m = integer(a[2]);
525 if (m < 1 or m > c.monthCount(y)) return error.InvalidDate;
526 const first = c.fixed(.{ .year = y, .month = m, .day = 1 });
527 const last = first + c.monthDays(y, m) - 1;
528 if (cmd == .MONTHCAL) return writeMonth(out, c, y, m, first, last);
529 const wd = indexOf(&weekdays, a[4]).?;
530 const f = if (equal(a[3], "LAST")) last - @mod(last - wd, 7) else first + @mod(wd - first, 7) + 7 * (integer(a[3]) - 1);
531 if (f > last) try out.writeAll("NONE") else try writeDate(out, c, f);
532 },
533 else => unreachable,
534 }
535 },
536 else => {
537 const c = calendar(a[0]).?;
538 if (cmd == .VALID) {
539 const f = parseDate(c, a[1]) catch |err| {
540 try out.print("NO {s}\n", .{@errorName(err)});
541 return;
542 };
543 try out.writeAll("YES ");
544 try writeDate(out, c, f);
545 try out.writeAll("\n");
546 return;
547 }
548 const f = try checkedDate(c, a[1]);
549 switch (cmd) {
550 .CONVERT => try writeDate(out, calendar(a[3]).?, f),
551 .WEEKDAY => try out.writeAll(weekdays[ix(@mod(f, 7))]),
552 .ALL => {
553 for (std.enums.values(Calendar)) |target| {
554 try out.print("{s} ", .{@tagName(target)});
555 if (target.isYmd() and target.parts(f) == null) {
556 try out.writeAll("OUT OF RANGE\n");
557 continue;
558 }
559 try writeDate(out, target, f);
560 if (target.isYmd()) {
561 const d = target.parts(f).?;
562 try out.print(" {s}", .{target.names(d.year)[ix(d.month - 1)]});
563 }
564 try out.writeAll("\n");
565 }
566 return;
567 },
568 .INFO => {
569 const d = c.parts(f).?;
570 const name = if (c != .FRENCH) weekdays[ix(@mod(f, 7))] else if (d.month <= 12) decade[ix(@mod(d.day - 1, 10))] else festivals[ix(d.day - 1)];
571 try out.print("{s} {d} {s} {d} day={d}/{d}", .{ name, d.day, c.names(d.year)[ix(d.month - 1)], d.year, f - c.start(d.year) + 1, c.start(d.year + 1) - c.start(d.year) });
572 },
573 .DIFF => try out.print("{d}", .{try checkedDate(calendar(a[2]).?, a[3]) - f}),
574 .COUNT, .SPAN => {
575 const last = try checkedDate(c, a[2]);
576 if (cmd == .SPAN) try writeSpan(out, c, f, last) else {
577 const lo = @min(f, last);
578 const hi = @max(f, last);
579 const occurrence = lo + @mod(indexOf(&weekdays, a[3]).? - lo, 7);
580 try out.print("{d}", .{if (occurrence > hi) @as(i64, 0) else @divFloor(hi - occurrence, 7) + 1});
581 }
582 },
583 .ADD => {
584 const amount = integer(a[2]);
585 const unit = std.meta.stringToEnum(Unit, a[3]).?;
586 const result = switch (unit) {
587 .DAYS => f + amount,
588 .WEEKS => f + 7 * amount,
589 .YEARS => c.shift(f, amount, 0) orelse return error.OutOfRange,
590 .MONTHS => c.shift(f, 0, amount) orelse return error.OutOfRange,
591 };
592 try writeDate(out, c, result);
593 },
594 .ROUND => {
595 const num = integer(a[2]);
596 const tz_name = indexOf(&tz, a[3]).?;
597 const haab = 20 * indexOf(&hb, a[5]).? + integer(a[4]);
598 var candidate = f;
599 while (@mod(candidate + 1137142 + 3, 13) + 1 != num or @mod(candidate + 1137142 + 19, 20) != tz_name) candidate += 1;
600 while (candidate <= f + 18979) : (candidate += 260) {
601 if (@mod(candidate + 1137142 + 348, 365) == haab) {
602 try writeDate(out, c, candidate);
603 try out.writeAll("\n");
604 return;
605 }
606 }
607 try out.writeAll("NONE");
608 },
609 else => unreachable,
610 }
611 },
612 }
613 try out.writeAll("\n");
614}

616pub fn main(init: std.process.Init) !void {
617 var input_buffer: [4096]u8 = undefined;
618 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
619 var output_buffer: [8192]u8 = undefined;
620 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
621 const count = integer((try input.interface.takeDelimiter('\n')).?);
622 var n: i64 = 0;
623 while (n < count) : (n += 1) {
624 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
625 defer arena.deinit();
626 const gpa = arena.allocator();
627 const line = (try input.interface.takeDelimiter('\n')).?;
628 var words: std.ArrayList([]const u8) = .empty;
629 var split = std.mem.splitScalar(u8, line, ' ');
630 while (split.next()) |word| try words.append(gpa, word);
631 var result: std.Io.Writer.Allocating = .init(gpa);
632 execute(gpa, &result.writer, words.items) catch |err| {
633 const message = switch (err) {
634 error.InvalidCommand => "INVALID COMMAND\n",
635 error.InvalidDate => "INVALID DATE\n",
636 error.OutOfRange => "OUT OF RANGE\n",
637 else => return err,
638 };
639 try output.interface.writeAll(message);
640 continue;
641 };
642 try output.interface.writeAll(result.written());
643 }
644 try output.interface.flush();
645}
