37func integerWord(_ word: String) -> Bool {
38 if word == "0" {
39 return true
40 }
41 let positive = word.first == "-" ? String(word.dropFirst()) : word
42 return positive.count >= 1 && positive.count <= 7 &&
43 positive.first != "0" && digits(positive)
44}

90 func neighborhood(_ x: Int, _ y: Int) -> [Int] {
91 var result = [Int]()
92 result.reserveCapacity(9)
93 for dy in -1...1 {
94 for dx in -1...1 {
95 result.append(get(clamp(x + dx, 0, width - 1),

110func parsePGM(_ lines: [String]) throws -> Image {
111 var words = [String]()
112 for line in lines {
113 let uncommented = line.split(separator: "#", maxSplits: 1,
114 omittingEmptySubsequences: false).first ?? ""
115 words.append(contentsOf: uncommented.split(separator: " ").map(String.init))
116 }
117
118 try require(!words.isEmpty && words[0] == "P2", "bad magic")
119
120 func header(_ index: Int, _ maximum: Int, _ reason: String) throws -> Int {
121 try require(index < words.count, "truncated header")
122 try require(digits(words[index]), "bad number")
123 let number = Int(words[index])!
124 try require(number >= 1 && number <= maximum, reason)
125 return number
126 }
127
128 let width = try header(1, 64, "bad size")
129 let height = try header(2, 64, "bad size")
130 let maxval = try header(3, 255, "bad maxval")
131 let count = words.count - 4
132 let required = width * height
133 var pixels = [Int]()
134 pixels.reserveCapacity(required)
135
136 for i in 0..<required {
137 let x = i % width, y = i / width
138 try require(i < count, "too few pixels (\(count) of \(required))")
139 let word = words[i + 4]
140 try require(digits(word), "bad pixel at \(x),\(y)")
141 let value = Int(word)!
142 try require(value <= maxval, "pixel out of range at \(x),\(y)")
143 pixels.append(value)
144 }
145
146 try require(count <= required, "too many pixels (\(count) of \(required))")
147 return Image(width: width, height: height, maxval: maxval, pixels: pixels)
148}

177func convolution(_ image: Image, _ kernel: [Int], _ divisor: Int) -> Image {
178 transformed(image, image.width, image.height) { x, y in
179 let q = image.neighborhood(x, y)
180 var total = 0
181 for i in 0..<9 {
182 total += q[i] * kernel[i]
183 }
184 return image.clamped(floorDivide(total, divisor))
185 }

194func sobel(_ image: Image) -> Image {
195 transformed(image, image.width, image.height) { x, y in
196 let q = image.neighborhood(x, y)
197 let gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6]
198 let gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2]
199 return image.clamped(abs(gx) + abs(gy))
200 }

203func linePoints(_ x0: Int, _ y0: Int, _ x1: Int, _ y1: Int) -> [Point] {
204 let dx = abs(x1 - x0), dy = -abs(y1 - y0)
205 let sx = x0 < x1 ? 1 : -1, sy = y0 < y1 ? 1 : -1
206 var error = dx + dy
207 var x = x0, y = y0
208 var points = [Point]()
209 while true {
210 points.append(Point(x: x, y: y))
211 if x == x1 && y == y1 {
212 break
213 }
214 let twice = 2 * error
215 if twice >= dy {
216 error += dy
217 x += sx
218 }
219 if twice <= dx {
220 error += dx
221 y += sy
222 }
223 }
224 return points
225}

227func rectanglePoints(
228 _ image: Image, _ left: Int, _ top: Int,
229 _ width: Int, _ height: Int, _ fill: Bool
230) -> [Point] {
231 let right = left + width - 1, bottom = top + height - 1
232 var points = [Point]()
233 let firstY = max(top, 0), lastY = min(bottom, image.height - 1)
234 let firstX = max(left, 0), lastX = min(right, image.width - 1)
235 if firstX > lastX || firstY > lastY {
236 return points
237 }
238 for y in firstY...lastY {
239 for x in firstX...lastX {
240 if fill || x == left || x == right || y == top || y == bottom {
241 points.append(Point(x: x, y: y))
242 }
243 }
244 }
245 return points
246}

248func circlePoints(_ cx: Int, _ cy: Int, _ radius: Int) -> [Point] {
249 var a = radius, b = 0, error = 1 - radius
250 var points = [Point]()
251 while a >= b {
252 points.append(contentsOf: [
253 Point(x: cx + a, y: cy + b), Point(x: cx + b, y: cy + a),
254 Point(x: cx - b, y: cy + a), Point(x: cx - a, y: cy + b),
255 Point(x: cx - a, y: cy - b), Point(x: cx - b, y: cy - a),
256 Point(x: cx + b, y: cy - a), Point(x: cx + a, y: cy - b)
257 ])
258 b += 1
259 if error < 0 {
260 error += 2 * b + 1
261 } else {
262 a -= 1
263 error += 2 * (b - a) + 1
264 }
265 }
266 return points
267}

269func discPoints(_ image: Image, _ outline: [Point]) -> [Point] {
270 var rows = [Int: (left: Int, right: Int)]()
271 for p in outline {
272 if let limits = rows[p.y] {
273 rows[p.y] = (min(limits.left, p.x), max(limits.right, p.x))
274 } else {
275 rows[p.y] = (p.x, p.x)
276 }
277 }
278 var points = [Point]()
279 for (y, limits) in rows {
280 if y < 0 || y >= image.height {
281 continue
282 }
283 let first = max(0, limits.left)
284 let last = min(image.width - 1, limits.right)
285 if first <= last {
286 for x in first...last {
287 points.append(Point(x: x, y: y))
288 }
289 }
290 }
291 return points
292}

294func floodPoints(_ image: Image, _ x: Int, _ y: Int) -> [Point] {
295 let target = image.get(x, y)
296 var stack = [Point(x: x, y: y)]
297 var seen: Set<Int> = [y * image.width + x]
298 var result = [Point]()
299 while let p = stack.popLast() {
300 result.append(p)
301 for d in neighbors4 {
302 let nx = p.x + d.x, ny = p.y + d.y
303 if !image.contains(nx, ny) || image.get(nx, ny) != target {
304 continue
305 }
306 let index = ny * image.width + nx
307 if seen.insert(index).inserted {
308 stack.append(Point(x: nx, y: ny))
309 }
310 }
311 }
312 return result
313}

315func equalized(_ image: Image) -> Image {
316 var counts = [Int](repeating: 0, count: image.maxval + 1)
317 for pixel in image.pixels {
318 counts[pixel] += 1
319 }
320 var accumulated = 0
321 let cdf = counts.map { count -> Int in
322 accumulated += count
323 return accumulated
324 }
325 let size = image.pixels.count
326 let minimum = cdf[image.pixels.min()!]
327 if minimum == size {
328 return image.withPixels(image.pixels)
329 }
330 let span = size - minimum
331 return image.mapped {
332 (2 * (cdf[$0] - minimum) * image.maxval + span) / (2 * span)
333 }
334}

336func components(_ image: Image, _ threshold: Int) -> [[Point]] {
337 var visited = [Bool](repeating: false, count: image.pixels.count)
338 var groups = [[Point]]()
339 for start in image.pixels.indices {
340 if visited[start] || image.pixels[start] < threshold {
341 continue
342 }
343 visited[start] = true
344 var stack = [Point(x: start % image.width, y: start / image.width)]
345 var group = [Point]()
346 while let p = stack.popLast() {
347 group.append(p)
348 for d in neighbors8 {
349 let nx = p.x + d.x, ny = p.y + d.y
350 if !image.contains(nx, ny) {
351 continue
352 }
353 let index = ny * image.width + nx
354 if !visited[index] && image.pixels[index] >= threshold {
355 visited[index] = true
356 stack.append(Point(x: nx, y: ny))
357 }
358 }
359 }
360 groups.append(group)
361 }
362 return groups
363}

365func componentText(_ image: Image, _ threshold: Int, _ group: [Point]) -> String {
366 var minX = group[0].x, maxX = group[0].x
367 var minY = group[0].y, maxY = group[0].y
368 var sumX = 0, sumY = 0, perimeter = 0
369 for p in group {
370 minX = min(minX, p.x)
371 maxX = max(maxX, p.x)
372 minY = min(minY, p.y)
373 maxY = max(maxY, p.y)
374 sumX += p.x
375 sumY += p.y
376 if neighbors4.contains(where: {
377 !image.contains(p.x + $0.x, p.y + $0.y) ||
378 image.get(p.x + $0.x, p.y + $0.y) < threshold
379 }) {
380 perimeter += 1
381 }
382 }
383 let area = group.count
384 return "area \(area) perimeter \(perimeter) box \(minX) \(minY) \(maxX) \(maxY) " +
385 "center \(sumX / area) \(sumY / area)"
386}

388func distanceMap(_ image: Image, _ threshold: Int) -> Image {
389 if image.pixels.allSatisfy({ $0 >= threshold }) {
390 return image.withPixels([Int](repeating: image.maxval, count: image.pixels.count))
391 }
392 let w = image.width, h = image.height, far = w + h
393 var distances = image.pixels.map { $0 < threshold ? 0 : far }
394 for y in 0..<h {
395 for x in 0..<w {
396 let index = y * w + x
397 if x > 0 {
398 distances[index] = min(distances[index], distances[index - 1] + 1)
399 }
400 if y > 0 {
401 distances[index] = min(distances[index], distances[index - w] + 1)
402 }
403 }
404 }
405 for y in stride(from: h - 1, through: 0, by: -1) {
406 for x in stride(from: w - 1, through: 0, by: -1) {
407 let index = y * w + x
408 if x < w - 1 {
409 distances[index] = min(distances[index], distances[index + 1] + 1)
410 }
411 if y < h - 1 {
412 distances[index] = min(distances[index], distances[index + w] + 1)
413 }
414 }
415 }
416 return image.withPixels(distances.map { min($0, image.maxval) })
417}

419func statistics(_ pixels: [Int], _ maxval: Int) -> String {
420 let sorted = pixels.sorted()
421 var counts = [Int](repeating: 0, count: maxval + 1)
422 var total = 0
423 for p in pixels {
424 counts[p] += 1
425 total += p
426 }
427 var mode = 0
428 for p in 1...maxval {
429 if counts[p] > counts[mode] {
430 mode = p
431 }
432 }
433 return "min \(sorted.first!) max \(sorted.last!) mean \(total / pixels.count) " +
434 "median \(sorted[(pixels.count - 1) / 2]) mode \(mode)"
435}

437func dithered(_ image: Image) -> Image {
438 var errors = image.pixels
439 var pixels = [Int](repeating: 0, count: errors.count)
440 let offsets = [(1, 0, 7), (-1, 1, 3), (0, 1, 5), (1, 1, 1)]
441 for y in 0..<image.height {
442 for x in 0..<image.width {
443 let index = y * image.width + x
444 let old = errors[index]
445 let value = 2 * old >= image.maxval ? image.maxval : 0
446 pixels[index] = value
447 let error = old - value
448 for (dx, dy, weight) in offsets {
449 if image.contains(x + dx, y + dy) {
450 errors[(y + dy) * image.width + x + dx] += floorDivide(error * weight, 16)
451 }
452 }
453 }
454 }
455 return image.withPixels(pixels)
456}

497 private func parse(_ words: [String]) throws -> Command {
498 let name = words[0]
499 guard let pattern = Self.patterns[name] else {
500 throw CommandError(reason: "unknown command " + name)
501 }
502 var given = Array(words.dropFirst())
503 let fill = (name == "RECT" || name == "CIRCLE") &&
504 given.count == pattern.count + 1 && given.last == "FILL"
505 if fill {
506 given.removeLast()
507 }
508 try require(given.count == pattern.count, "bad arguments")
509
510 let types = Array(pattern)
511 var numbers = [Int]()
512 var slot = ""
513 for (index, word) in given.enumerated() {
514 if types[index] == "i" {
515 try require(integerWord(word), "bad arguments")
516 numbers.append(Int(word)!)
517 } else {
518 try require(slotWord(word), "bad arguments")
519 slot = word
520 }
521 }
522 return Command(name: name, numbers: numbers, slot: slot, fill: fill)
523 }

525 mutating func execute(_ words: [String]) {
526 do {
527 let command = try parse(words)
528 if !Self.imageless.contains(command.name) {
529 try require(current != nil, "no image")
530 }
531 if Self.needsSlot.contains(command.name) {
532 try require(slots[command.slot] != nil, "no slot " + command.slot)
533 }
534 try run(command)
535 } catch let e as CommandError {
536 emit("error: " + e.reason)
537 } catch {
538 fatalError("Unexpected error: \(error)")
539 }
540 }

587 private mutating func run(_ command: Command) throws {
588 let a = command.numbers
589 let name = command.name, slot = command.slot, fill = command.fill
590
591 switch name {
592 case "NEW":
593 try require(a[0] >= 1 && a[0] <= 64 && a[1] >= 1 && a[1] <= 64)
594 try require(a[2] >= 1 && a[2] <= 255 && a[3] >= 0 && a[3] <= a[2])
595 replace(Image(width: a[0], height: a[1], maxval: a[2],
596 pixels: [Int](repeating: a[3], count: a[0] * a[1])), "created")
597 return
598 case "RECALL":
599 replace(slots[slot]!, "recalled " + slot)
600 return
601 case "DROP":
602 slots.removeValue(forKey: slot)
603 emit("dropped " + slot)
604 return
605 case "SLOTS":
606 if slots.isEmpty {
607 emit("no slots")
608 }
609 for key in slots.keys.sorted() {
610 emit(key + " " + slots[key]!.summary())
611 }
612 return
613 case "INFO":
614 emit("image " + (current?.summary() ?? "none") +
615 " undo \(undo.count) redo \(redo.count) slots \(slots.count)")
616 return
617 case "UNDO":
618 try require(!undo.isEmpty, "nothing to undo")
619 redo.append(current!)
620 current = undo.removeLast()
621 emit("undo " + current!.summary())
622 return
623 case "REDO":
624 try require(!redo.isEmpty, "nothing to redo")
625 undo.append(current!)
626 current = redo.removeLast()
627 emit("redo " + current!.summary())
628 return
629 default:
630 break
631 }
632
633 let image = current!
634 let w = image.width, h = image.height, m = image.maxval
635 switch name {
636 case "OUTPUT":
637 emit("P2")
638 emit("\(w) \(h)")
639 emit(String(m))
640 for y in 0..<h {
641 emit((0..<w).map { String(image.get($0, y)) }.joined(separator: " "))
642 }
643 case "INVERT":
644 commit(image.mapped { m - $0 })
645 case "BRIGHTNESS":
646 try require((-255...255).contains(a[0]))
647 commit(image.mapped { image.clamped($0 + a[0]) })
648 case "CONTRAST":
649 try require((0...100).contains(a[0]) && (1...100).contains(a[1]))
650 let center = (m + 1) / 2
651 commit(image.mapped { image.clamped(center + floorDivide(($0 - center) * a[0], a[1])) })
652 case "THRESHOLD":
653 try require((0...m).contains(a[0]))
654 commit(image.mapped { $0 >= a[0] ? m : 0 })
655 case "LEVELS":
656 try require(a[0] >= 0 && a[0] < a[1] && a[1] <= m)
657 commit(image.mapped { image.clamped(floorDivide(($0 - a[0]) * m, a[1] - a[0])) })
658 case "POSTERIZE":
659 try require((2...(m + 1)).contains(a[0]))
660 commit(image.mapped { (($0 * a[0]) / (m + 1)) * m / (a[0] - 1) })
661 case "MAXVAL":
662 try require((1...255).contains(a[0]))
663 commit(Image(width: w, height: h, maxval: a[0],
664 pixels: image.pixels.map { (2 * $0 * a[0] + m) / (2 * m) }))
665 case "FLIPH":
666 commit(transformed(image, w, h) { x, y in image.get(w - 1 - x, y) })
667 case "FLIPV":
668 commit(transformed(image, w, h) { x, y in image.get(x, h - 1 - y) })
669 case "ROTATE":
670 try require([90, 180, 270].contains(a[0]))
671 switch a[0] {
672 case 90:
