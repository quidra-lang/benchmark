44int clamp(int value, int low, int high) => math.min(math.max(value, low), high);
45int floorDiv(int x, int n) => x >= 0 ? x ~/ n : -((-x + n - 1) ~/ n);
46bool withinLimit(Iterable<int> values) => values.every((v) => v >= -1000 && v <= 1000);
47int sum(Iterable<int> values) => values.fold(0, (total, value) => total + value);

49class Image {
50 final int width, height, maxval;
51 final List<int> pixels;
52 Image(this.width, this.height, this.maxval, this.pixels);
53
54 int get(int x, int y) => pixels[y * width + x];
55 bool contains(int x, int y) => x >= 0 && y >= 0 && x < width && y < height;
56 int clampPixel(int value) => clamp(value, 0, maxval);
57 Image withPixels(List<int> values) => Image(width, height, maxval, values);
58 Image mapPixels(int Function(int) change) => withPixels(pixels.map(change).toList());
59 List<int> row(int y) => pixels.sublist(y * width, (y + 1) * width);

61 List<int> neighborhood(int x, int y) {
62 final values = <int>[];
63 for (var dy = -1; dy <= 1; dy++) {
64 for (var dx = -1; dx <= 1; dx++) {
65 values.add(get(clamp(x + dx, 0, width - 1), clamp(y + dy, 0, height - 1)));
66 }
67 }
68 return values;
69 }
70
71 String summary() {
72 var checksum = 0;
73 for (var i = 0; i < pixels.length; i++) checksum += (i + 1) * pixels[i];
74 return '${width}x$height maxval $maxval checksum $checksum';
75 }
76}

78Image loadPgm(List<String> lines) {
79 final words = <String>[];
80 for (final line in lines) {
81 final content = line.split('#').first.trim();
82 if (content.isNotEmpty) words.addAll(content.split(RegExp(r' +')));
83 }
84 check(words.isNotEmpty && words.first == 'P2', 'bad magic');
85
86 int header(int index, int limit, String reason) {
87 check(index < words.length, 'truncated header');
88 check(RegExp(r'^[0-9]+$').hasMatch(words[index]), 'bad number');
89 final number = int.parse(words[index]);
90 check(number >= 1 && number <= limit, reason);
91 return number;
92 }
93
94 final width = header(1, 64, 'bad size');
95 final height = header(2, 64, 'bad size');
96 final maxval = header(3, 255, 'bad maxval');
97 final count = words.length - 4;
98 final pixels = <int>[];
99 for (var i = 0; i < width * height; i++) {
100 final x = i % width, y = i ~/ width;
101 check(i < count, 'too few pixels ($count of ${width * height})');
102 check(RegExp(r'^[0-9]+$').hasMatch(words[i + 4]), 'bad pixel at $x,$y');
103 final value = int.parse(words[i + 4]);
104 check(value <= maxval, 'pixel out of range at $x,$y');
105 pixels.add(value);
106 }
107 check(count <= width * height, 'too many pixels ($count of ${width * height})');
108 return Image(width, height, maxval, pixels);
109}

126Image convolution(Image image, List<int> weights, int divisor) =>
127 transformed(image, image.width, image.height, (x, y) {
128 final q = image.neighborhood(x, y);
129 var value = 0;
130 for (var i = 0; i < 9; i++) value += q[i] * weights[i];
131 return image.clampPixel(floorDiv(value, divisor));

140Image sobel(Image image) =>
141 transformed(image, image.width, image.height, (x, y) {
142 final q = image.neighborhood(x, y);
143 final gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6];
144 final gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
145 return image.clampPixel(gx.abs() + gy.abs());
146 });

148List<Point> linePoints(int x0, int y0, int x1, int y1) {
149 final dx = (x1 - x0).abs(), dy = -(y1 - y0).abs();
150 final sx = x0 < x1 ? 1 : -1, sy = y0 < y1 ? 1 : -1;
151 var error = dx + dy, x = x0, y = y0;
152 final points = <Point>[];
153 while (true) {
154 points.add((x, y));
155 if (x == x1 && y == y1) return points;
156 final twice = 2 * error;
157 if (twice >= dy) { error += dy; x += sx; }
158 if (twice <= dx) { error += dx; y += sy; }
159 }
160}

162List<Point> rectanglePoints(Image image, int left, int top, int width, int height, bool fill) {
163 final right = left + width - 1, bottom = top + height - 1;
164 final points = <Point>[];
165 for (var y = math.max(0, top); y <= math.min(bottom, image.height - 1); y++) {
166 for (var x = math.max(0, left); x <= math.min(right, image.width - 1); x++) {
167 if (fill || x == left || x == right || y == top || y == bottom) points.add((x, y));
168 }
169 }
170 return points;
171}

173List<Point> circlePoints(int cx, int cy, int radius) {
174 var a = radius, b = 0, error = 1 - radius;
175 final points = <Point>[];
176 while (a >= b) {
177 points.addAll([
178 (cx + a, cy + b), (cx + b, cy + a), (cx - b, cy + a), (cx - a, cy + b),
179 (cx - a, cy - b), (cx - b, cy - a), (cx + b, cy - a), (cx + a, cy - b)
180 ]);
181 b++;
182 if (error < 0) { error += 2 * b + 1; }
183 else { a--; error += 2 * (b - a) + 1; }
184 }
185 return points;
186}

188List<Point> discPoints(Image image, List<Point> outline) {
189 final rows = <int, (int, int)>{};
190 for (final point in outline) {
191 final x = point.$1, y = point.$2;
192 final previous = rows[y];
193 rows[y] = previous == null
194 ? (x, x) : (math.min(previous.$1, x), math.max(previous.$2, x));
195 }
196 final points = <Point>[];
197 for (final entry in rows.entries) {
198 final y = entry.key;
199 if (y < 0 || y >= image.height) continue;
200 for (var x = math.max(0, entry.value.$1); x <= math.min(image.width - 1, entry.value.$2); x++) {
201 points.add((x, y));
202 }
203 }
204 return points;
205}

207List<Point> floodPoints(Image image, int x, int y) {
208 final wanted = image.get(x, y);
209 final stack = <Point>[(x, y)];
210 final visited = <int>{y * image.width + x};
211 final points = <Point>[];
212 while (stack.isNotEmpty) {
213 final point = stack.removeLast();
214 final cx = point.$1, cy = point.$2;
215 points.add(point);
216 for (final delta in adjacent4) {
217 final nx = cx + delta.$1, ny = cy + delta.$2;
218 if (!image.contains(nx, ny) || image.get(nx, ny) != wanted) continue;
219 final index = ny * image.width + nx;
220 if (visited.add(index)) stack.add((nx, ny));
221 }
222 }
223 return points;
224}

226Image equalized(Image image) {
227 final counts = List<int>.filled(image.maxval + 1, 0);
228 for (final pixel in image.pixels) counts[pixel]++;
229 final cdf = <int>[];
230 var running = 0;
231 for (final count in counts) {
232 running += count;
233 cdf.add(running);
234 }
235 final size = image.pixels.length;
236 final first = cdf[image.pixels.reduce(math.min)];
237 if (first == size) return image.withPixels(List<int>.of(image.pixels));
238 final span = size - first;
239 return image.mapPixels((pixel) =>
240 (2 * (cdf[pixel] - first) * image.maxval + span) ~/ (2 * span));
241}

243List<List<Point>> components(Image image, int threshold) {
244 final visited = <int>{};
245 final groups = <List<Point>>[];
246 for (var start = 0; start < image.pixels.length; start++) {
247 if (visited.contains(start) || image.pixels[start] < threshold) continue;
248 visited.add(start);
249 final stack = <Point>[(start % image.width, start ~/ image.width)];
250 final group = <Point>[];
251 while (stack.isNotEmpty) {
252 final point = stack.removeLast();
253 group.add(point);
254 for (final delta in adjacent8) {
255 final x = point.$1 + delta.$1, y = point.$2 + delta.$2;
256 if (!image.contains(x, y)) continue;
257 final index = y * image.width + x;
258 if (image.pixels[index] >= threshold && visited.add(index)) stack.add((x, y));
259 }
260 }
261 groups.add(group);
262 }
263 return groups;
264}

266String componentText(Image image, int threshold, List<Point> group) {
267 var minX = image.width, minY = image.height, maxX = -1, maxY = -1;
268 var sumX = 0, sumY = 0, perimeter = 0;
269 for (final point in group) {
270 final x = point.$1, y = point.$2;
271 minX = math.min(minX, x);
272 minY = math.min(minY, y);
273 maxX = math.max(maxX, x);
274 maxY = math.max(maxY, y);
275 sumX += x;
276 sumY += y;
277 if (adjacent4.any((delta) {
278 final nx = x + delta.$1, ny = y + delta.$2;
279 return !image.contains(nx, ny) || image.get(nx, ny) < threshold;
280 })) perimeter++;
281 }
282 final area = group.length;
283 return 'area $area perimeter $perimeter box $minX $minY $maxX $maxY '
284 'center ${sumX ~/ area} ${sumY ~/ area}';
285}

287Image distanceMap(Image image, int threshold) {
288 if (image.pixels.every((pixel) => pixel >= threshold)) {
289 return image.withPixels(List<int>.filled(image.pixels.length, image.maxval));
290 }
291 final w = image.width, h = image.height;
292 final distances = image.pixels.map((pixel) => pixel < threshold ? 0 : w + h).toList();
293 for (var y = 0; y < h; y++) {
294 for (var x = 0; x < w; x++) {
295 final index = y * w + x;
296 if (x > 0) distances[index] = math.min(distances[index], distances[index - 1] + 1);
297 if (y > 0) distances[index] = math.min(distances[index], distances[index - w] + 1);
298 }
299 }
300 for (var y = h - 1; y >= 0; y--) {
301 for (var x = w - 1; x >= 0; x--) {
302 final index = y * w + x;
303 if (x < w - 1) distances[index] = math.min(distances[index], distances[index + 1] + 1);
304 if (y < h - 1) distances[index] = math.min(distances[index], distances[index + w] + 1);
305 }
306 }
307 return image.withPixels(distances.map((value) => math.min(value, image.maxval)).toList());
308}

310String statistics(List<int> values, int maxval) {
311 final sorted = List<int>.of(values)..sort();
312 final counts = List<int>.filled(maxval + 1, 0);
313 for (final value in values) counts[value]++;
314 var mode = 0;
315 for (var value = 1; value <= maxval; value++) {
316 if (counts[value] > counts[mode]) mode = value;
317 }
318 return 'min ${sorted.first} max ${sorted.last} mean ${sum(values) ~/ values.length} '
319 'median ${sorted[(values.length - 1) ~/ 2]} mode $mode';
320}

322Image dithered(Image image) {
323 final working = List<int>.of(image.pixels);
324 final pixels = List<int>.filled(working.length, 0);
325 const neighbors = <(int, int, int)>[(1, 0, 7), (-1, 1, 3), (0, 1, 5), (1, 1, 1)];
326 for (var y = 0; y < image.height; y++) {
327 for (var x = 0; x < image.width; x++) {
328 final index = y * image.width + x;
329 final old = working[index];
330 final value = 2 * old >= image.maxval ? image.maxval : 0;
331 pixels[index] = value;
332 final error = old - value;
333 for (final delta in neighbors) {
334 final nx = x + delta.$1, ny = y + delta.$2;
335 if (image.contains(nx, ny)) {
336 working[ny * image.width + nx] += floorDiv(error * delta.$3, 16);
337 }
338 }
339 }
340 }
341 return image.withPixels(pixels);
342}

351class Editor {
352 Image? current;

354 final slots = <String, Image>{};
355 final output = <String>[];
356
357 void emit(String line) => output.add(line);
358
359 Command parseCommand(List<String> words) {
360 final name = words.first;
361 check(patterns.containsKey(name), 'unknown command $name');
362 final pattern = patterns[name]!;
363 final given = words.sublist(1);
364 final fill = (name == 'RECT' || name == 'CIRCLE') &&
365 given.length == pattern.length + 1 && given.last == 'FILL';
366 if (fill) given.removeLast();
367 check(given.length == pattern.length, 'bad arguments');
368 final numbers = <int>[];
369 var slot = '';
370 for (var i = 0; i < given.length; i++) {
371 if (pattern[i] == 'i') {
372 check(RegExp(r'^(?:0|-?[1-9][0-9]{0,6})$').hasMatch(given[i]), 'bad arguments');
373 numbers.add(int.parse(given[i]));
374 } else {
375 check(RegExp(r'^[a-z]{1,8}$').hasMatch(given[i]), 'bad arguments');
376 slot = given[i];
377 }
378 }
379 return Command(name, numbers, slot, fill);
380 }

382 void execute(List<String> words) {
383 try {
384 final command = parseCommand(words);
385 if (!imageless.contains(command.name)) check(current != null, 'no image');
386 if (needsSlot.contains(command.name)) {
387 check(slots.containsKey(command.slot), 'no slot ${command.slot}');
388 }
389 run(command);
390 } on CommandError catch (error) {
391 emit('error: ${error.reason}');
392 }
393 }

446 void run(Command cmd) {
447 final name = cmd.name, a = cmd.numbers, slot = cmd.slot;
448 switch (name) {
449 case 'NEW':
450 check(a[0] >= 1 && a[0] <= 64 && a[1] >= 1 && a[1] <= 64);
451 check(a[2] >= 1 && a[2] <= 255 && a[3] >= 0 && a[3] <= a[2]);
452 replace(Image(a[0], a[1], a[2], List<int>.filled(a[0] * a[1], a[3])), 'created');
453 return;
454 case 'RECALL':
455 replace(slots[slot]!, 'recalled $slot');
456 return;
457 case 'DROP':
458 slots.remove(slot);
459 emit('dropped $slot');
460 return;
461 case 'SLOTS':
462 if (slots.isEmpty) emit('no slots');
463 final names = slots.keys.toList()..sort();
464 for (final key in names) emit('$key ${slots[key]!.summary()}');
465 return;
466 case 'INFO':
467 emit('image ${current == null ? 'none' : current!.summary()} '
468 'undo ${undo.length} redo ${redo.length} slots ${slots.length}');
469 return;
470 case 'UNDO':
471 check(undo.isNotEmpty, 'nothing to undo');
472 redo.add(current!);
473 current = undo.removeLast();
474 emit('undo ${current!.summary()}');
475 return;
476 case 'REDO':
477 check(redo.isNotEmpty, 'nothing to redo');
478 undo.add(current!);
479 current = redo.removeLast();
480 emit('redo ${current!.summary()}');
481 return;
482 }
483 runWithImage(current!, cmd);
484 }

486 void runWithImage(Image image, Command cmd) {
487 final name = cmd.name, a = cmd.numbers, slot = cmd.slot, fill = cmd.fill;
488 final w = image.width, h = image.height, m = image.maxval;
489 switch (name) {
490 case 'OUTPUT':
491 emit('P2');
492 emit('$w $h');
493 emit('$m');
494 for (var y = 0; y < h; y++) emit(image.row(y).join(' '));
495 break;
496 case 'INVERT':
497 commit(image.mapPixels((p) => m - p));
498 break;
499 case 'BRIGHTNESS':
500 check(a[0] >= -255 && a[0] <= 255);
501 commit(image.mapPixels((p) => image.clampPixel(p + a[0])));
502 break;
503 case 'CONTRAST':
504 check(a[0] >= 0 && a[0] <= 100 && a[1] >= 1 && a[1] <= 100);
505 final center = (m + 1) ~/ 2;
506 commit(image.mapPixels((p) =>
507 image.clampPixel(center + floorDiv((p - center) * a[0], a[1]))));
508 break;
509 case 'THRESHOLD':
510 check(a[0] >= 0 && a[0] <= m);
511 commit(image.mapPixels((p) => p >= a[0] ? m : 0));
512 break;
513 case 'LEVELS':
514 check(a[0] >= 0 && a[0] < a[1] && a[1] <= m);
515 commit(image.mapPixels((p) =>
516 image.clampPixel(floorDiv((p - a[0]) * m, a[1] - a[0]))));
517 break;
518 case 'POSTERIZE':
519 check(a[0] >= 2 && a[0] <= m + 1);
520 commit(image.mapPixels((p) => (p * a[0] ~/ (m + 1) * m) ~/ (a[0] - 1)));
521 break;
522 case 'MAXVAL':
523 check(a[0] >= 1 && a[0] <= 255);
524 commit(Image(w, h, a[0],
525 image.pixels.map((p) => (2 * p * a[0] + m) ~/ (2 * m)).toList()));
526 break;
527 case 'FLIPH':
528 commit(transformed(image, w, h, (x, y) => image.get(w - 1 - x, y)));
529 break;
530 case 'FLIPV':
531 commit(transformed(image, w, h, (x, y) => image.get(x, h - 1 - y)));
532 break;
533 case 'ROTATE':
534 check(a[0] == 90 || a[0] == 180 || a[0] == 270);
535 if (a[0] == 180) {
536 commit(transformed(image, w, h, (x, y) => image.get(w - 1 - x, h - 1 - y)));
537 } else if (a[0] == 90) {
538 commit(transformed(image, h, w, (x, y) => image.get(y, h - 1 - x)));
539 } else {
540 commit(transformed(image, h, w, (x, y) => image.get(w - 1 - y, x)));
541 }
542 break;
543 case 'CROP':
544 checkedCrop(a);
545 commit(cropped(image, a[0], a[1], a[2], a[3]));
546 break;
547 case 'PAD':
548 check(a[0] >= 1 && a[1] >= 0 && a[1] <= m);
549 check(w + 2 * a[0] <= 64 && h + 2 * a[0] <= 64, 'too large');
550 commit(translated(image, w + 2 * a[0], h + 2 * a[0], a[0], a[0], a[1]));
551 break;
552 case 'SCALE':
553 check(a[0] >= 1 && a[0] <= 16 && a[1] >= 1 && a[1] <= 16);
554 final width = w * a[0] ~/ a[1], height = h * a[0] ~/ a[1];
555 check(width > 0 && height > 0, 'too small');
556 check(width <= 64 && height <= 64, 'too large');
557 commit(transformed(image, width, height, (x, y) => image.get(
558 (2 * x + 1) * a[1] ~/ (2 * a[0]), (2 * y + 1) * a[1] ~/ (2 * a[0]))));
559 break;
560 case 'SHIFT':
561 check(withinLimit(a.take(2)) && a[2] >= 0 && a[2] <= m);
562 commit(translated(image, w, h, a[0], a[1], a[2]));
563 break;
564 case 'TILE':
565 check(a[0] >= 1 && a[1] >= 1);
566 check(w * a[0] <= 64 && h * a[1] <= 64, 'too large');
567 commit(transformed(image, w * a[0], h * a[1], (x, y) => image.get(x % w, y % h)));
568 break;
569 case 'BLUR':
570 case 'GAUSS':
571 case 'SHARPEN':
572 final (weights, divisor) = kernels[name]!;
573 commit(convolution(image, weights, divisor));
574 break;
575 case 'CONVOLVE':
576 check(a.take(9).every((v) => v >= -16 && v <= 16) && a[9] >= 1 && a[9] <= 256);
577 commit(convolution(image, a.take(9).toList(), a[9]));
578 break;
579 case 'SOBEL':
580 commit(sobel(image));
581 break;
582 case 'MEDIAN':
583 commit(rankFiltered(image, 4));
584 break;
585 case 'ERODE':
586 commit(rankFiltered(image, 0));
587 break;
588 case 'DILATE':
589 commit(rankFiltered(image, 8));
590 break;
591 case 'OPEN':
592 commit(rankFiltered(rankFiltered(image, 0), 8));
593 break;
594 case 'CLOSE':
595 commit(rankFiltered(rankFiltered(image, 8), 0));
596 break;
597 case 'GRADIENT':
598 final high = rankFiltered(image, 8).pixels, low = rankFiltered(image, 0).pixels;
599 commit(image.withPixels(List<int>.generate(high.length, (i) => high[i] - low[i])));
600 break;
601 case 'LINE':
602 check(withinLimit(a.take(4)) && a[4] >= 0 && a[4] <= m);
603 paint(linePoints(a[0], a[1], a[2], a[3]), a[4]);
604 break;
605 case 'RECT':
606 check(withinLimit(a.take(2)) && a[2] >= 1 && a[2] <= 1000 &&
607 a[3] >= 1 && a[3] <= 1000 && a[4] >= 0 && a[4] <= m);
608 paint(rectanglePoints(image, a[0], a[1], a[2], a[3], fill), a[4]);
609 break;
610 case 'CIRCLE':
611 check(withinLimit(a.take(2)) && a[2] >= 0 && a[2] <= 1000 &&
612 a[3] >= 0 && a[3] <= m);
613 final outline = circlePoints(a[0], a[1], a[2]);
614 paint(fill ? discPoints(image, outline) : outline, a[3]);
615 break;
616 case 'FLOOD':
617 check(image.contains(a[0], a[1]) && a[2] >= 0 && a[2] <= m);
618 paint(floodPoints(image, a[0], a[1]), a[2]);
619 break;
620 case 'HISTOGRAM':
621 final lows = List<int?>.filled(16, null);
622 final highs = List<int>.filled(16, 0), counts = List<int>.filled(16, 0);
623 for (var value = 0; value <= m; value++) {
624 final bin = value * 16 ~/ (m + 1);
625 lows[bin] ??= value;
626 highs[bin] = value;
627 }
628 for (final pixel in image.pixels) counts[pixel * 16 ~/ (m + 1)]++;
629 final top = counts.reduce(math.max);
630 for (var bin = 0; bin < 16; bin++) {
631 if (lows[bin] == null) continue;
632 emit('${lows[bin].toString().padLeft(3)}-${highs[bin].toString().padLeft(3)} '
633 '${counts[bin].toString().padLeft(4)} |'
634 '${List.filled(counts[bin] * 32 ~/ top, '#').join()}');
635 }
636 break;
637 case 'EQUALIZE':
638 commit(equalized(image));
639 break;
640 case 'LABEL':
641 check(a[0] >= 0 && a[0] <= m);
642 final groups = components(image, a[0]);
643 emit('labels ${groups.length}');
644 for (var i = 0; i < groups.length; i++) {
645 emit('label ${i + 1} ${componentText(image, a[0], groups[i])}');
646 }
647 break;
648 case 'DISTANCE':
649 check(a[0] >= 0 && a[0] <= m);
650 commit(distanceMap(image, a[0]));
651 break;
652 case 'STATS':
653 emit(statistics(image.pixels, m));
654 break;
655 case 'REGION':
656 checkedCrop(a);
657 emit(statistics(cropped(image, a[0], a[1], a[2], a[3]).pixels, m));
658 break;
659 case 'DITHER':
660 commit(dithered(image));
661 break;
662 case 'ASCIIART':
663 for (var y = 0; y < h; y++) {
664 emit(image.row(y).map((p) => shades[p * 10 ~/ (m + 1)]).join());
665 }
666 break;
667 case 'STORE':
668 slots[slot] = image;
669 emit('stored $slot ${image.summary()}');
670 break;
671 case 'BLEND':
672 check(a[0] >= 0 && a[0] <= a[1] && a[1] >= 1 && a[1] <= 100);
673 final other = matchingSlot(slot);
674 commit(image.withPixels(List<int>.generate(image.pixels.length, (i) =>
675 (image.pixels[i] * (a[1] - a[0]) + other.pixels[i] * a[0]) ~/ a[1])));
676 break;
677 case 'DIFF':
678 final other = matchingSlot(slot);
679 commit(image.withPixels(List<int>.generate(image.pixels.length, (i) =>
680 (image.pixels[i] - other.pixels[i]).abs())));
681 break;
682 case 'PASTE':
683 check(withinLimit(a));
684 final other = slots[slot]!;
685 check(other.maxval == m, 'maxval mismatch');
686 final pixels = List<int>.of(image.pixels);
687 var painted = 0;
688 for (var y = 0; y < other.height; y++) {
689 for (var x = 0; x < other.width; x++) {
690 if (image.contains(a[0] + x, a[1] + y)) {
691 pixels[(a[1] + y) * w + a[0] + x] = other.get(x, y);
692 painted++;
693 }
694 }
695 }
696 commit(image.withPixels(pixels), painted);
697 break;
698 }
699 }
700}
