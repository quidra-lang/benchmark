53class Image:
54 def __init__(self, width, height, maxval, pixels):
55 self.width = width
56 self.height = height
57 self.maxval = maxval
58 self.pixels = pixels
59
60 def get(self, x, y):
61 return self.pixels[y * self.width + x]
62
63 def row(self, y):
64 return self.pixels[y * self.width:(y + 1) * self.width]
65
66 def contains(self, x, y):
67 return 0 <= x < self.width and 0 <= y < self.height
68
69 def clamp(self, value):
70 return min(max(value, 0), self.maxval)
71
72 def with_pixels(self, pixels):
73 return Image(self.width, self.height, self.maxval, pixels)
74
75 def neighborhood(self, x, y):
76 values = []
77 for dx, dy in NEIGHBORHOOD:
78 nx = min(max(x + dx, 0), self.width - 1)
79 ny = min(max(y + dy, 0), self.height - 1)
80 values.append(self.get(nx, ny))
81 return values

83 def checksum(self):
84 return sum((index + 1) * value for index, value in enumerate(self.pixels))

86 def summary(self):
87 return f"{self.width}x{self.height} maxval {self.maxval} checksum {self.checksum()}"

98def header_value(words, position, limit, reason):
99 require(position < len(words), "truncated header")
100 word = words[position]
101 require(DIGIT_WORD.fullmatch(word) is not None, "bad number")
102 value = int(word)
103 require(1 <= value <= limit, reason)
104 return value

107def read_pgm(lines):
108 words = []
109 for line in lines:
110 words.extend(line.partition("#")[0].split())
111 require(words and words[0] == "P2", "bad magic")
112 width = header_value(words, 1, MAX_SIDE, "bad size")
113 height = header_value(words, 2, MAX_SIDE, "bad size")
114 maxval = header_value(words, 3, MAX_MAXVAL, "bad maxval")
115 values = words[4:]
116 total = width * height
117 pixels = []
118 for index in range(total):
119 x, y = index % width, index // width
120 require(index < len(values), f"too few pixels ({len(values)} of {total})")
121 word = values[index]
122 require(DIGIT_WORD.fullmatch(word) is not None, f"bad pixel at {x},{y}")
123 require(int(word) <= maxval, f"pixel out of range at {x},{y}")
124 pixels.append(int(word))
125 require(len(values) <= total, f"too many pixels ({len(values)} of {total})")
126 return Image(width, height, maxval, pixels)

187def sobel_edges(image):
188 pixels = []
189 for y in range(image.height):
190 for x in range(image.width):
191 q = image.neighborhood(x, y)
192 gx = (q[2] + 2 * q[5] + q[8]) - (q[0] + 2 * q[3] + q[6])
193 gy = (q[6] + 2 * q[7] + q[8]) - (q[0] + 2 * q[1] + q[2])
194 pixels.append(image.clamp(abs(gx) + abs(gy)))
195 return image.with_pixels(pixels)

198def line_points(x0, y0, x1, y1):
199 dx = abs(x1 - x0)
200 dy = -abs(y1 - y0)
201 step_x = 1 if x0 < x1 else -1
202 step_y = 1 if y0 < y1 else -1
203 error = dx + dy
204 x, y = x0, y0
205 points = []
206 while True:
207 points.append((x, y))
208 if x == x1 and y == y1:
209 return points
210 doubled = 2 * error
211 if doubled >= dy:
212 error += dy
213 x += step_x
214 if doubled <= dx:
215 error += dx
216 y += step_y

220 right = left + width - 1
221 bottom = top + height - 1
222 points = []
223 for y in range(max(top, 0), min(bottom, image.height - 1) + 1):

230def circle_points(cx, cy, radius):
231 a, b = radius, 0
232 error = 1 - radius
233 points = []
234 while a >= b:
235 points.extend((
236 (cx + a, cy + b), (cx + b, cy + a), (cx - b, cy + a), (cx - a, cy + b),
237 (cx - a, cy - b), (cx - b, cy - a), (cx + b, cy - a), (cx + a, cy - b),
238 ))
239 b += 1
240 if error < 0:
241 error += 2 * b + 1
242 else:
243 a -= 1
244 error += 2 * (b - a) + 1
245 return points

264def flood_points(image, x, y):
265 target = image.get(x, y)
266 visited = [False] * len(image.pixels)
267 visited[y * image.width + x] = True
268 stack = [(x, y)]
269 points = []
270 while stack:
271 cx, cy = stack.pop()
272 points.append((cx, cy))
273 for dx, dy in FOUR_NEIGHBORS:
274 nx, ny = cx + dx, cy + dy
275 if not image.contains(nx, ny) or image.get(nx, ny) != target:
276 continue
277 index = ny * image.width + nx
278 if not visited[index]:
279 visited[index] = True
280 stack.append((nx, ny))
281 return points

284def equalized(image):
285 total = len(image.pixels)
286 counts = [0] * (image.maxval + 1)
287 for pixel in image.pixels:
288 counts[pixel] += 1
289 cdf = list(accumulate(counts))
290 lowest = cdf[min(image.pixels)]
291 if lowest == total:
292 return image
293 span = total - lowest
294 pixels = [
295 (2 * (cdf[pixel] - lowest) * image.maxval + span) // (2 * span)
296 for pixel in image.pixels
297 ]
298 return image.with_pixels(pixels)

301def components(image, threshold):
302 width = image.width
303 foreground = [pixel >= threshold for pixel in image.pixels]
304 visited = [False] * len(image.pixels)
305 result = []
306 for start in range(len(image.pixels)):
307 if not foreground[start] or visited[start]:
308 continue
309 visited[start] = True
310 stack = [(start % width, start // width)]
311 members = []
312 while stack:
313 x, y = stack.pop()
314 members.append((x, y))
315 for dx, dy in EIGHT_NEIGHBORS:
316 nx, ny = x + dx, y + dy
317 if not image.contains(nx, ny):
318 continue
319 index = ny * width + nx
320 if foreground[index] and not visited[index]:
321 visited[index] = True
322 stack.append((nx, ny))
323 result.append(members)
324 return result

346def distance_map(image, threshold):
347 width, height = image.width, image.height
348 if all(pixel >= threshold for pixel in image.pixels):
349 return image.with_pixels([image.maxval] * len(image.pixels))
350 far = width + height
351 distances = [0 if pixel < threshold else far for pixel in image.pixels]
352 for y in range(height):
353 for x in range(width):
354 index = y * width + x
355 if x > 0:
356 distances[index] = min(distances[index], distances[index - 1] + 1)
357 if y > 0:
358 distances[index] = min(distances[index], distances[index - width] + 1)
359 for y in range(height - 1, -1, -1):
360 for x in range(width - 1, -1, -1):
361 index = y * width + x
362 if x < width - 1:
363 distances[index] = min(distances[index], distances[index + 1] + 1)
364 if y < height - 1:
365 distances[index] = min(distances[index], distances[index + width] + 1)
366 return image.with_pixels([min(distance, image.maxval) for distance in distances])

382def dithered(image):
383 width, maxval = image.width, image.maxval
384 errors = list(image.pixels)
385 pixels = [0] * len(errors)
386 for y in range(image.height):
387 for x in range(width):
388 old = errors[y * width + x]
389 new = maxval if 2 * old >= maxval else 0
390 pixels[y * width + x] = new
391 error = old - new
392 for dx, dy, weight in DIFFUSION:
393 if image.contains(x + dx, y + dy):
394 errors[(y + dy) * width + x + dx] += error * weight // 16
395 return image.with_pixels(pixels)

461 def parse(self, words):
462 name, arguments = words[0], words[1:]
463 if name not in self.handlers:
464 raise CommandError("unknown command " + name)
465 pattern = self.handlers[name][0]
466 fill = (
467 name in FILL_COMMANDS
468 and len(arguments) == len(pattern) + 1
469 and arguments[-1] == "FILL"
470 )
471 if fill:
472 arguments = arguments[:-1]
473 require(len(arguments) == len(pattern), "bad arguments")
474 numbers = []
475 slot = None
476 for kind, word in zip(pattern, arguments):
477 if kind == "i":
478 require(INTEGER_WORD.fullmatch(word) is not None, "bad arguments")
479 numbers.append(int(word))
480 else:
481 require(NAME_WORD.fullmatch(word) is not None, "bad arguments")
482 slot = word
483 return Command(name, numbers, slot, fill)

485 def execute(self, words):
486 try:
487 command = self.parse(words)
488 if command.name not in IMAGELESS_COMMANDS:
489 require(self.current is not None, "no image")
490 if command.name in SLOT_COMMANDS:
491 require(command.slot in self.slots, "no slot " + command.slot)
492 self.handlers[command.name][1](command)
493 except CommandError as error:
494 self.emit(f"error: {error}")

725 def histogram(self, command):
726 image = self.current
727 levels = image.maxval + 1
728 lows = [None] * HISTOGRAM_BINS
729 highs = [0] * HISTOGRAM_BINS
730 counts = [0] * HISTOGRAM_BINS
731 for value in range(levels):
732 number = value * HISTOGRAM_BINS // levels
733 if lows[number] is None:
734 lows[number] = value
735 highs[number] = value
736 for pixel in image.pixels:
737 counts[pixel * HISTOGRAM_BINS // levels] += 1
738 used = [number for number in range(HISTOGRAM_BINS) if lows[number] is not None]
739 top = max(counts[number] for number in used)
740 for number in used:
741 bar = "#" * (counts[number] * BAR_LENGTH // top)
742 self.emit(f"{lows[number]:3d}-{highs[number]:3d} {counts[number]:4d} |{bar}")

807 def paste(self, command):
808 left, top = command.numbers
809 require(within_limit(left, top))
810 other = self.slots[command.slot]
811 image = self.current
812 require(other.maxval == image.maxval, "maxval mismatch")
813 pixels = list(image.pixels)
814 painted = 0
815 for j in range(other.height):
816 for i in range(other.width):
817 if image.contains(left + i, top + j):
818 pixels[(top + j) * image.width + left + i] = other.get(i, j)
819 painted += 1
820 self.commit(image.with_pixels(pixels), painted)
