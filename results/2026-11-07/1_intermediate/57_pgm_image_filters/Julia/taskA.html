38struct GrayImage
39 width::Int
40 height::Int
41 maxval::Int
42 pixels::Vector{Int}

45pixel(image::GrayImage, x::Int, y::Int) = image.pixels[y*image.width + x + 1]
46contains(image::GrayImage, x::Int, y::Int) =
47 0 <= x < image.width && 0 <= y < image.height
48row(image::GrayImage, y::Int) =
49 image.pixels[(y*image.width + 1):((y + 1)*image.width)]
50with_pixels(image::GrayImage, pixels::Vector{Int}) =
51 GrayImage(image.width, image.height, image.maxval, pixels)
52limit(image::GrayImage, value::Int) = clamp(value, 0, image.maxval)

54function neighborhood(image::GrayImage, x::Int, y::Int)
55 [pixel(image, clamp(x + dx, 0, image.width - 1),
56 clamp(y + dy, 0, image.height - 1))
57 for dy in -1:1 for dx in -1:1]
58end

60function summary(image::GrayImage)
61 checksum = sum(i*p for (i, p) in enumerate(image.pixels))
62 "$(image.width)x$(image.height) maxval $(image.maxval) checksum $checksum"
63end

65function read_pgm(lines::Vector{String})
66 words = String[]
67 for line in lines
68 append!(words, split(strip(first(split(line, "#"; limit=2)))))
69 end
70 check(!isempty(words) && words[1] == "P2", "bad magic")
71 function header(i, upper, message)
72 check(i <= length(words), "truncated header")
73 check(occursin(r"^[0-9]+$", words[i]), "bad number")
74 number = parse(Int, words[i])
75 check(1 <= number <= upper, message)
76 number
77 end
78 width = header(2, 64, "bad size")
79 height = header(3, 64, "bad size")
80 maxval = header(4, 255, "bad maxval")
81 values = words[5:end]
82 pixels = Int[]
83 needed = width * height
84 for i in 0:(needed - 1)
85 x, y = i % width, i ÷ width
86 check(i + 1 <= length(values),
87 "too few pixels ($(length(values)) of $needed)")
88 word = values[i + 1]
89 check(occursin(r"^[0-9]+$", word), "bad pixel at $x,$y")
90 value = parse(Int, word)
91 check(value <= maxval, "pixel out of range at $x,$y")
92 push!(pixels, value)
93 end
94 check(length(values) == needed,
95 "too many pixels ($(length(values)) of $needed)")
96 GrayImage(width, height, maxval, pixels)
97end

99function transform(image::GrayImage, width::Int, height::Int, f)
100 values = [f(x, y) for y in 0:(height - 1) for x in 0:(width - 1)]
101 GrayImage(width, height, image.maxval, values)
102end
103transform(image::GrayImage, f) = transform(image, image.width, image.height, f)
104transform(f, image::GrayImage) = transform(image, f)
105transform(f, image::GrayImage, width::Int, height::Int) = transform(image, width, height, f)

107function convolve(image::GrayImage, kernel::Vector{Int}, divisor::Int)
108 transform(image) do x, y
109 q = neighborhood(image, x, y)
110 limit(image, fld(sum(k*v for (k, v) in zip(kernel, q)), divisor))
111 end
112end
113
114function rank_filter(image::GrayImage, rank::Int)
115 transform(image) do x, y
116 sort(neighborhood(image, x, y))[rank + 1]
117 end
118end

120function sobel(image::GrayImage)
121 transform(image) do x, y
122 q = neighborhood(image, x, y)
123 gx = (q[3] + 2q[6] + q[9]) - (q[1] + 2q[4] + q[7])
124 gy = (q[7] + 2q[8] + q[9]) - (q[1] + 2q[2] + q[3])
125 limit(image, abs(gx) + abs(gy))
126 end
127end

129function line_points(x::Int, y::Int, dest_x::Int, dest_y::Int)
130 dx = abs(dest_x - x)
131 dy = -abs(dest_y - y)
132 sx = x < dest_x ? 1 : -1
133 sy = y < dest_y ? 1 : -1
134 error = dx + dy
135 points = Point[]
136 while true
137 push!(points, (x, y))
138 x == dest_x && y == dest_y && break
139 twice = 2error
140 if twice >= dy
141 error += dy
142 x += sx
143 end
144 if twice <= dx
145 error += dx
146 y += sy
147 end
148 end
149 points
150end

152function rectangle_points(image::GrayImage, x::Int, y::Int, width::Int, height::Int, fill::Bool)
153 right, bottom = x + width - 1, y + height - 1
154 points = Point[]
155 for j in max(0, y):min(bottom, image.height - 1)
156 for i in max(0, x):min(right, image.width - 1)
157 (fill || i == x || i == right || j == y || j == bottom) &&
158 push!(points, (i, j))
159 end
160 end
161 points
162end

164function circle_points(cx::Int, cy::Int, radius::Int)
165 a, b, error = radius, 0, 1 - radius
166 points = Point[]
167 while a >= b
168 append!(points, [
169 (cx + a, cy + b), (cx + b, cy + a),
170 (cx - b, cy + a), (cx - a, cy + b),
171 (cx - a, cy - b), (cx - b, cy - a),
172 (cx + b, cy - a), (cx + a, cy - b)
173 ])
174 b += 1
175 if error < 0
176 error += 2b + 1
177 else
178 a -= 1
179 error += 2(b - a) + 1
180 end
181 end
182 points
183end

185function filled_circle(image::GrayImage, points::Vector{Point})
186 rows = Dict{Int,Tuple{Int,Int}}()
187 for (x, y) in points
188 if haskey(rows, y)
189 lo, hi = rows[y]
190 rows[y] = (min(lo, x), max(hi, x))
191 else
192 rows[y] = (x, x)
193 end
194 end
195 result = Point[]
196 for (y, (lo, hi)) in rows
197 0 <= y < image.height || continue
198 append!(result, [(x, y) for x in max(0, lo):min(hi, image.width - 1)])
199 end
200 result
201end

203function flood_points(image::GrayImage, x::Int, y::Int)
204 target = pixel(image, x, y)
205 stack = Point[(x, y)]
206 visited = Set([y*image.width + x])
207 points = Point[]
208 while !isempty(stack)
209 cx, cy = pop!(stack)
210 push!(points, (cx, cy))
211 for (dx, dy) in NEIGHBORS4
212 nx, ny = cx + dx, cy + dy
213 contains(image, nx, ny) && pixel(image, nx, ny) == target || continue
214 index = ny*image.width + nx
215 if !(index in visited)
216 push!(visited, index)
217 push!(stack, (nx, ny))
218 end
219 end
220 end
221 points
222end

224function equalize(image::GrayImage)
225 counts = zeros(Int, image.maxval + 1)
226 for p in image.pixels
227 counts[p + 1] += 1
228 end
229 cdf = cumsum(counts)
230 cmin = cdf[minimum(image.pixels) + 1]
231 size = length(image.pixels)
232 cmin == size && return with_pixels(image, copy(image.pixels))
233 span = size - cmin
234 with_pixels(image, [fld(2(cdf[p + 1] - cmin)*image.maxval + span, 2span)
235 for p in image.pixels])
236end

238function components(image::GrayImage, threshold::Int)
239 visited = falses(length(image.pixels))
240 groups = Vector{Point}[]
241 for start in eachindex(image.pixels)
242 (visited[start] || image.pixels[start] < threshold) && continue
243 visited[start] = true
244 stack = Point[((start - 1) % image.width, (start - 1) ÷ image.width)]
245 group = Point[]
246 while !isempty(stack)
247 x, y = pop!(stack)
248 push!(group, (x, y))
249 for (dx, dy) in NEIGHBORS8
250 nx, ny = x + dx, y + dy
251 contains(image, nx, ny) || continue
252 index = ny*image.width + nx + 1
253 if !visited[index] && image.pixels[index] >= threshold
254 visited[index] = true
255 push!(stack, (nx, ny))
256 end
257 end
258 end
259 push!(groups, group)
260 end
261 groups
262end

264function component_text(image::GrayImage, threshold::Int, group::Vector{Point})
265 xs = first.(group)
266 ys = last.(group)
267 perimeter = 0
268 for (x, y) in group
269 if any(!contains(image, x + dx, y + dy) ||
270 pixel(image, x + dx, y + dy) < threshold
271 for (dx, dy) in NEIGHBORS4)
272 perimeter += 1
273 end
274 end
275 area = length(group)
276 "area $area perimeter $perimeter box $(minimum(xs)) $(minimum(ys)) " *
277 "$(maximum(xs)) $(maximum(ys)) center $(sum(xs) ÷ area) $(sum(ys) ÷ area)"
278end

280function distance_map(image::GrayImage, threshold::Int)
281 w, h = image.width, image.height
282 any(p -> p < threshold, image.pixels) ||
283 return with_pixels(image, fill(image.maxval, w*h))
284 distances = [p < threshold ? 0 : w + h for p in image.pixels]
285 for y in 0:(h - 1), x in 0:(w - 1)
286 i = y*w + x + 1
287 x > 0 && (distances[i] = min(distances[i], distances[i - 1] + 1))
288 y > 0 && (distances[i] = min(distances[i], distances[i - w] + 1))
289 end
290 for y in (h - 1):-1:0, x in (w - 1):-1:0
291 i = y*w + x + 1
292 x < w - 1 && (distances[i] = min(distances[i], distances[i + 1] + 1))
293 y < h - 1 && (distances[i] = min(distances[i], distances[i + w] + 1))
294 end
295 with_pixels(image, [min(n, image.maxval) for n in distances])
296end

298function statistics(pixels::Vector{Int}, maxval::Int)
299 sorted = sort(pixels)
300 counts = zeros(Int, maxval + 1)
301 for pixel in pixels
302 counts[pixel + 1] += 1
303 end
304 size = length(pixels)
305 "min $(first(sorted)) max $(last(sorted)) mean $(sum(pixels) ÷ size) " *
306 "median $(sorted[fld(size - 1, 2) + 1]) mode $(argmax(counts) - 1)"
307end

309function dither(image::GrayImage)
310 errors = copy(image.pixels)
311 pixels = zeros(Int, length(errors))
312 diffusion = ((1, 0, 7), (-1, 1, 3), (0, 1, 5), (1, 1, 1))
313 for y in 0:(image.height - 1), x in 0:(image.width - 1)
314 index = y*image.width + x + 1
315 old = errors[index]
316 new_pixel = 2old >= image.maxval ? image.maxval : 0
317 pixels[index] = new_pixel
318 error = old - new_pixel
319 for (dx, dy, weight) in diffusion
320 nx, ny = x + dx, y + dy
321 if contains(image, nx, ny)
322 errors[ny*image.width + nx + 1] += fld(error*weight, 16)
323 end
324 end
325 end
326 with_pixels(image, pixels)
327end

340function parse_command(words::Vector{String})
341 name = words[1]
342 check(haskey(PATTERNS, name), "unknown command $name")
343 pattern = PATTERNS[name]
344 args = words[2:end]
345 fill = name in ("RECT", "CIRCLE") && length(args) == length(pattern) + 1 &&
346 last(args) == "FILL"
347 fill && pop!(args)
348 check(length(args) == length(pattern), "bad arguments")
349 numbers = Int[]
350 slot = ""
351 for (kind, word) in zip(pattern, args)
352 if kind == 'i'
353 check(occursin(r"^(?:0|-?[1-9][0-9]{0,6})$", word), "bad arguments")
354 push!(numbers, parse(Int, word))
355 else
356 check(occursin(r"^[a-z]{1,8}$", word), "bad arguments")
357 slot = word
358 end
359 end
360 (name, numbers, slot, fill)
361end

363function replace!(editor::ImageEditor, image::GrayImage, label::String)
364 editor.current = image
365 empty!(editor.undo)
366 empty!(editor.redo)
367 emit!(editor, "$label $(summary(image))")

370function commit!(editor::ImageEditor, image::GrayImage; painted=nothing)
371 push!(editor.undo, editor.current)
372 empty!(editor.redo)
373 editor.current = image
374 suffix = isnothing(painted) ? "" : " painted $painted"
375 emit!(editor, "ok $(summary(image))$suffix")
376end

378function paint!(editor::ImageEditor, points::Vector{Point}, value::Int)
379 image = editor.current
380 positions = Set{Int}()
381 for (x, y) in points
382 contains(image, x, y) && push!(positions, y*image.width + x + 1)
383 end
384 pixels = copy(image.pixels)
385 for index in positions
386 pixels[index] = value
387 end
388 commit!(editor, with_pixels(image, pixels); painted=length(positions))
389end

407function run!(editor::ImageEditor, name::String, a::Vector{Int}, slot::String, fill_shape::Bool)
408 image = editor.current
409 w = isnothing(image) ? 0 : image.width
410 h = isnothing(image) ? 0 : image.height
411 m = isnothing(image) ? 0 : image.maxval
412 if name == "NEW"
413 width, height, maxval, value = a
414 check(1 <= width <= 64 && 1 <= height <= 64)
415 check(1 <= maxval <= 255 && 0 <= value <= maxval)
416 replace!(editor, GrayImage(width, height, maxval,
417 fill(value, width*height)), "created")
418 elseif name == "OUTPUT"
419 emit!(editor, "P2")
420 emit!(editor, "$w $h")
421 emit!(editor, string(m))
422 for y in 0:(h - 1)
423 emit!(editor, join(row(image, y), " "))
424 end
425 elseif name == "INVERT"
426 commit!(editor, with_pixels(image, [m - p for p in image.pixels]))
427 elseif name == "BRIGHTNESS"
428 check(-255 <= a[1] <= 255)
429 commit!(editor, with_pixels(image, [limit(image, p + a[1])
430 for p in image.pixels]))
431 elseif name == "CONTRAST"
432 num, den = a
433 check(0 <= num <= 100 && 1 <= den <= 100)
434 center = (m + 1) ÷ 2
435 commit!(editor, with_pixels(image, [
436 limit(image, center + fld((p - center)*num, den)) for p in image.pixels
437 ]))
438 elseif name == "THRESHOLD"
439 check(0 <= a[1] <= m)
440 commit!(editor, with_pixels(image, [p >= a[1] ? m : 0
441 for p in image.pixels]))
442 elseif name == "LEVELS"
443 lo, hi = a
444 check(0 <= lo < hi <= m)
445 commit!(editor, with_pixels(image, [
446 limit(image, fld((p - lo)*m, hi - lo)) for p in image.pixels
447 ]))
448 elseif name == "POSTERIZE"
449 levels = a[1]
450 check(2 <= levels <= m + 1)
451 commit!(editor, with_pixels(image, [
452 fld(fld(p*levels, m + 1)*m, levels - 1) for p in image.pixels
453 ]))
454 elseif name == "MAXVAL"
455 new_max = a[1]
456 check(1 <= new_max <= 255)
457 values = [fld(2p*new_max + m, 2m) for p in image.pixels]
458 commit!(editor, GrayImage(w, h, new_max, values))
459 elseif name == "FLIPH"
460 commit!(editor, transform(image, (x, y) -> pixel(image, w - 1 - x, y)))
461 elseif name == "FLIPV"
462 commit!(editor, transform(image, (x, y) -> pixel(image, x, h - 1 - y)))
463 elseif name == "ROTATE"
464 angle = a[1]
465 check(angle in (90, 180, 270))
466 if angle == 90
467 result = transform(image, h, w, (x, y) -> pixel(image, y, h - 1 - x))
468 elseif angle == 180
469 result = transform(image, (x, y) -> pixel(image, w - 1 - x, h - 1 - y))
470 else
471 result = transform(image, h, w, (x, y) -> pixel(image, w - 1 - y, x))
472 end
473 commit!(editor, result)
474 elseif name == "CROP"
475 x, y, width, height = crop_area(image, a)
476 commit!(editor, transform(image, width, height,
477 (i, j) -> pixel(image, x + i, y + j)))
478 elseif name == "PAD"
479 n, value = a
480 check(n >= 1 && 0 <= value <= m)
481 check(w + 2n <= 64 && h + 2n <= 64, "too large")
482 commit!(editor, transform(image, w + 2n, h + 2n) do x, y
483 contains(image, x - n, y - n) ? pixel(image, x - n, y - n) : value
484 end)
485 elseif name == "SCALE"
486 num, den = a
487 check(1 <= num <= 16 && 1 <= den <= 16)
488 width, height = fld(w*num, den), fld(h*num, den)
489 check(width >= 1 && height >= 1, "too small")
490 check(width <= 64 && height <= 64, "too large")
491 commit!(editor, transform(image, width, height, (x, y) ->
492 pixel(image, fld((2x + 1)*den, 2num), fld((2y + 1)*den, 2num))))
493 elseif name == "SHIFT"
494 dx, dy, value = a
495 check(-1000 <= dx <= 1000 && -1000 <= dy <= 1000 && 0 <= value <= m)
496 commit!(editor, transform(image, (x, y) ->
497 contains(image, x - dx, y - dy) ? pixel(image, x - dx, y - dy) : value))
498 elseif name == "TILE"
499 nx, ny = a
500 check(nx >= 1 && ny >= 1)
501 check(w*nx <= 64 && h*ny <= 64, "too large")
502 commit!(editor, transform(image, w*nx, h*ny,
503 (x, y) -> pixel(image, x % w, y % h)))
504 elseif haskey(KERNELS, name)
505 kernel, divisor = KERNELS[name]
506 commit!(editor, convolve(image, kernel, divisor))
507 elseif name == "CONVOLVE"
508 check(all(n -> -16 <= n <= 16, a[1:9]) && 1 <= a[10] <= 256)
509 commit!(editor, convolve(image, a[1:9], a[10]))
510 elseif name == "SOBEL"
511 commit!(editor, sobel(image))
512 elseif name == "MEDIAN"
513 commit!(editor, rank_filter(image, 4))
514 elseif name == "ERODE"
515 commit!(editor, rank_filter(image, 0))
516 elseif name == "DILATE"
517 commit!(editor, rank_filter(image, 8))
518 elseif name == "OPEN"
519 commit!(editor, rank_filter(rank_filter(image, 0), 8))
520 elseif name == "CLOSE"
521 commit!(editor, rank_filter(rank_filter(image, 8), 0))
522 elseif name == "GRADIENT"
523 low = rank_filter(image, 0).pixels
524 high = rank_filter(image, 8).pixels
525 commit!(editor, with_pixels(image, high .- low))
526 elseif name == "LINE"
527 x0, y0, x1, y1, value = a
528 check(all(n -> -1000 <= n <= 1000, a[1:4]) && 0 <= value <= m)
529 paint!(editor, line_points(x0, y0, x1, y1), value)
530 elseif name == "RECT"
531 x, y, width, height, value = a
532 check(-1000 <= x <= 1000 && -1000 <= y <= 1000)
533 check(1 <= width <= 1000 && 1 <= height <= 1000 && 0 <= value <= m)
534 paint!(editor, rectangle_points(image, x, y, width, height, fill_shape), value)
535 elseif name == "CIRCLE"
536 cx, cy, radius, value = a
537 check(-1000 <= cx <= 1000 && -1000 <= cy <= 1000)
538 check(0 <= radius <= 1000 && 0 <= value <= m)
539 outline = circle_points(cx, cy, radius)
540 paint!(editor, fill_shape ? filled_circle(image, outline) : outline, value)
541 elseif name == "FLOOD"
542 x, y, value = a
543 check(contains(image, x, y) && 0 <= value <= m)
544 paint!(editor, flood_points(image, x, y), value)
545 elseif name == "HISTOGRAM"
546 lows = Union{Nothing,Int}[nothing for _ in 1:16]
547 highs = zeros(Int, 16)
548 counts = zeros(Int, 16)
549 for value in 0:m
550 index = fld(value*16, m + 1) + 1
551 isnothing(lows[index]) && (lows[index] = value)
552 highs[index] = value
553 end
554 for p in image.pixels
555 counts[fld(p*16, m + 1) + 1] += 1
556 end
557 top = maximum(counts)
558 for i in 1:16
559 isnothing(lows[i]) && continue
560 emit!(editor, @sprintf("%3d-%3d %4d |%s", lows[i], highs[i], counts[i],
561 repeat("#", fld(counts[i]*32, top))))
562 end
563 elseif name == "EQUALIZE"
564 commit!(editor, equalize(image))
565 elseif name == "LABEL"
566 check(0 <= a[1] <= m)
567 groups = components(image, a[1])
568 emit!(editor, "labels $(length(groups))")
569 for (index, group) in enumerate(groups)
570 emit!(editor, "label $index $(component_text(image, a[1], group))")
571 end
572 elseif name == "DISTANCE"
573 check(0 <= a[1] <= m)
574 commit!(editor, distance_map(image, a[1]))
575 elseif name == "STATS"
576 emit!(editor, statistics(image.pixels, m))
577 elseif name == "REGION"
578 x, y, width, height = crop_area(image, a)
579 region = transform(image, width, height, (i, j) -> pixel(image, x + i, y + j))
580 emit!(editor, statistics(region.pixels, m))
581 elseif name == "DITHER"
582 commit!(editor, dither(image))
583 elseif name == "ASCIIART"
584 for y in 0:(h - 1)
585 emit!(editor, join([SHADES[fld(p*10, m + 1) + 1] for p in row(image, y)]))
586 end
587 elseif name == "STORE"
588 editor.slots[slot] = image
589 emit!(editor, "stored $slot $(summary(image))")
590 elseif name == "RECALL"
591 replace!(editor, editor.slots[slot], "recalled $slot")
592 elseif name == "DROP"
593 delete!(editor.slots, slot)
594 emit!(editor, "dropped $slot")
595 elseif name == "BLEND"
596 num, den = a
597 check(0 <= num <= den && 1 <= den <= 100)
598 other = matching_slot(editor, slot)
599 pixels = [fld(p*(den - num) + q*num, den)
600 for (p, q) in zip(image.pixels, other.pixels)]
601 commit!(editor, with_pixels(image, pixels))
602 elseif name == "DIFF"
603 other = matching_slot(editor, slot)
604 commit!(editor, with_pixels(image, abs.(image.pixels .- other.pixels)))
605 elseif name == "PASTE"
606 x, y = a
607 check(-1000 <= x <= 1000 && -1000 <= y <= 1000)
608 other = editor.slots[slot]
609 check(other.maxval == m, "maxval mismatch")
610 pixels = copy(image.pixels)
611 painted = 0
612 for j in 0:(other.height - 1), i in 0:(other.width - 1)
613 if contains(image, x + i, y + j)
614 pixels[(y + j)*w + x + i + 1] = pixel(other, i, j)
615 painted += 1
616 end
617 end
618 commit!(editor, with_pixels(image, pixels); painted=painted)
619 elseif name == "SLOTS"
620 isempty(editor.slots) && emit!(editor, "no slots")
621 for key in sort(collect(keys(editor.slots)))
622 emit!(editor, "$key $(summary(editor.slots[key]))")
623 end
624 elseif name == "INFO"
625 description = isnothing(image) ? "none" : summary(image)
626 emit!(editor, "image $description undo $(length(editor.undo)) " *
627 "redo $(length(editor.redo)) slots $(length(editor.slots))")
628 elseif name == "UNDO"
629 check(!isempty(editor.undo), "nothing to undo")
630 push!(editor.redo, image)
631 editor.current = pop!(editor.undo)
632 emit!(editor, "undo $(summary(editor.current))")
633 elseif name == "REDO"
634 check(!isempty(editor.redo), "nothing to redo")
635 push!(editor.undo, image)
636 editor.current = pop!(editor.redo)
637 emit!(editor, "redo $(summary(editor.current))")
638 end
639end
640
641function execute!(editor::ImageEditor, words::Vector{String})
642 try
643 name, a, slot, fill_shape = parse_command(words)
644 name in IMAGELESS || check(!isnothing(editor.current), "no image")
645 name in NEED_SLOT && check(haskey(editor.slots, slot), "no slot $slot")
646 run!(editor, name, a, slot, fill_shape)
647 catch e
648 e isa CommandError || rethrow()
649 emit!(editor, "error: $(e.reason)")
650 end
651end
652
653function load!(editor::ImageEditor, lines::Vector{String})
654 try
655 replace!(editor, read_pgm(lines), "loaded")
656 catch e
657 e isa CommandError || rethrow()
658 emit!(editor, "error: load: $(e.reason)")
659 end
660end
661
662function main()
663 lines = readlines(stdin)
664 count = parse(Int, lines[1])
665 index = 2
666 editor = ImageEditor()
667 for _ in 1:count
668 words = String.(split(lines[index], ' '))
669 index += 1
670 if words[1] == "LOAD"
671 amount = parse(Int, words[2])
672 load!(editor, lines[index:(index + amount - 1)])
673 index += amount
674 else
675 execute!(editor, words)
676 end
677 end
678 for line in editor.output
679 println(line)
680 end
681end
