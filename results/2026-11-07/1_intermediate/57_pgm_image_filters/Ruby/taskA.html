3PATTERNS = {
4 "NEW" => "iiii", "OUTPUT" => "", "INVERT" => "", "BRIGHTNESS" => "i",
5 "CONTRAST" => "ii", "THRESHOLD" => "i", "LEVELS" => "ii", "POSTERIZE" => "i",
6 "MAXVAL" => "i", "FLIPH" => "", "FLIPV" => "", "ROTATE" => "i",
7 "CROP" => "iiii", "PAD" => "ii", "SCALE" => "ii", "SHIFT" => "iii",
8 "TILE" => "ii", "BLUR" => "", "GAUSS" => "", "SHARPEN" => "",
9 "CONVOLVE" => "iiiiiiiiii", "SOBEL" => "", "MEDIAN" => "", "ERODE" => "",
10 "DILATE" => "", "OPEN" => "", "CLOSE" => "", "GRADIENT" => "",
11 "LINE" => "iiiii", "RECT" => "iiiii", "CIRCLE" => "iiii", "FLOOD" => "iii",
12 "HISTOGRAM" => "", "EQUALIZE" => "", "LABEL" => "i", "DISTANCE" => "i",
13 "STATS" => "", "REGION" => "iiii", "DITHER" => "", "ASCIIART" => "",
14 "STORE" => "n", "RECALL" => "n", "DROP" => "n", "BLEND" => "nii",
15 "DIFF" => "n", "PASTE" => "nii", "SLOTS" => "", "INFO" => "",
16 "UNDO" => "", "REDO" => ""
17}.freeze
18IMAGELESS = %w[NEW RECALL DROP SLOTS INFO UNDO REDO].freeze
19SLOT_COMMANDS = %w[RECALL DROP BLEND DIFF PASTE].freeze
20NEIGHBORS4 = [[0, -1], [-1, 0], [1, 0], [0, 1]].freeze
21NEIGHBORS8 = (-1..1).flat_map { |dy| (-1..1).filter_map { |dx| [dx, dy] unless dx.zero? && dy.zero? } }.freeze
22SHADES = "@#*+=~-:,.".freeze
23KERNELS = {
24 "BLUR" => [[1, 1, 1, 1, 1, 1, 1, 1, 1], 9],
25 "GAUSS" => [[1, 2, 1, 2, 4, 2, 1, 2, 1], 16],
26 "SHARPEN" => [[0, -1, 0, -1, 5, -1, 0, -1, 0], 1]
27}.freeze

40class Image
41 attr_reader :width, :height, :maxval, :pixels
42
43 def initialize(width, height, maxval, pixels)
44 @width, @height, @maxval, @pixels = width, height, maxval, pixels
45 end
46
47 def get(x, y)
48 @pixels[y * @width + x]
49 end
50
51 def contains?(x, y)
52 x >= 0 && y >= 0 && x < @width && y < @height
53 end
54
55 def row(y)
56 @pixels[y * @width, @width]
57 end
58
59 def with_pixels(pixels)
60 Image.new(@width, @height, @maxval, pixels)
61 end
62
63 def mapped
64 with_pixels(@pixels.map { |value| yield value })
65 end
66
67 def clamp(value)
68 [[value, 0].max, @maxval].min
69 end
70
71 def neighborhood(x, y)
72 (-1..1).flat_map do |dy|
73 (-1..1).map do |dx|
74 get([[x + dx, 0].max, @width - 1].min,
75 [[y + dy, 0].max, @height - 1].min)
76 end
77 end
78 end
79
80 def summary
81 checksum = @pixels.each_with_index.sum { |value, index| value * (index + 1) }
82 "#{@width}x#{@height} maxval #{@maxval} checksum #{checksum}"
83 end
84end

86def read_pgm(lines)
87 words = lines.flat_map { |line| line.split("#", 2).first.split }
88 check(!words.empty? && words[0] == "P2", "bad magic")
89 parse_header = lambda do |index, maxval, reason|
90 check(index < words.length, "truncated header")
91 check(/\A[0-9]+\z/.match?(words[index]), "bad number")
92 number = words[index].to_i
93 check(number.between?(1, maxval), reason)
94 number
95 end
96 width = parse_header.call(1, 64, "bad size")
97 height = parse_header.call(2, 64, "bad size")
98 maximum = parse_header.call(3, 255, "bad maxval")
99 values = words.drop(4)
100 count = width * height
101 pixels = []
102 count.times do |index|
103 x, y = index % width, index / width
104 check(index < values.length, "too few pixels (#{values.length} of #{count})")
105 check(/\A[0-9]+\z/.match?(values[index]), "bad pixel at #{x},#{y}")
106 number = values[index].to_i
107 check(number <= maximum, "pixel out of range at #{x},#{y}")
108 pixels << number
109 end
110 check(values.length <= count, "too many pixels (#{values.length} of #{count})")
111 Image.new(width, height, maximum, pixels)
112end

114def transformed(image, width, height)
115 pixels = (0...height).flat_map do |y|
116 (0...width).map { |x| yield x, y }
117 end
118 Image.new(width, height, image.maxval, pixels)
119end

121def cropped(image, x, y, width, height)
122 transformed(image, width, height) { |i, j| image.get(x + i, y + j) }
123end
124
125def translated(image, width, height, dx, dy, fill)
126 transformed(image, width, height) do |x, y|
127 image.contains?(x - dx, y - dy) ? image.get(x - dx, y - dy) : fill
128 end
129end

131def convolution(image, coefficients, divisor)
132 transformed(image, image.width, image.height) do |x, y|
133 value = image.neighborhood(x, y).zip(coefficients).sum { |pixel, weight| pixel * weight }
134 image.clamp(value.div(divisor))
135 end
136end
137
138def rank_filtered(image, rank)
139 transformed(image, image.width, image.height) do |x, y|
140 image.neighborhood(x, y).sort[rank]
141 end
142end
143
144def sobel(image)
145 transformed(image, image.width, image.height) do |x, y|
146 q = image.neighborhood(x, y)
147 gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6]
148 gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2]
149 image.clamp(gx.abs + gy.abs)
150 end
151end

153def line_points(x0, y0, x1, y1)
154 dx, dy = (x1 - x0).abs, -(y1 - y0).abs
155 sx, sy = x0 < x1 ? 1 : -1, y0 < y1 ? 1 : -1
156 error, x, y = dx + dy, x0, y0
157 points = []
158 loop do
159 points << [x, y]
160 break if x == x1 && y == y1
161 twice = 2 * error
162 if twice >= dy
163 error += dy
164 x += sx
165 end
166 if twice <= dx
167 error += dx
168 y += sy
169 end
170 end
171 points
172end
173
174def rectangle_points(image, left, top, width, height, fill)
175 right, bottom = left + width - 1, top + height - 1
176 points = []
177 ([top, 0].max..[bottom, image.height - 1].min).each do |y|
178 ([left, 0].max..[right, image.width - 1].min).each do |x|
179 points << [x, y] if fill || x == left || x == right || y == top || y == bottom
180 end
181 end
182 points
183end
184
185def circle_points(cx, cy, radius)
186 a, b, error = radius, 0, 1 - radius
187 points = []
188 while a >= b
189 points.concat([
190 [cx + a, cy + b], [cx + b, cy + a], [cx - b, cy + a], [cx - a, cy + b],
191 [cx - a, cy - b], [cx - b, cy - a], [cx + b, cy - a], [cx + a, cy - b]
192 ])
193 b += 1
194 if error < 0
195 error += 2 * b + 1
196 else
197 a -= 1
198 error += 2 * (b - a) + 1
199 end
200 end
201 points
202end

204def disc_points(image, outline)
205 rows = {}
206 outline.each do |x, y|
207 if rows.key?(y)
208 left, right = rows[y]
209 rows[y] = [[left, x].min, [right, x].max]
210 else
211 rows[y] = [x, x]
212 end
213 end
214 rows.flat_map do |y, (left, right)|
215 next [] unless y.between?(0, image.height - 1)
216 ([left, 0].max..[right, image.width - 1].min).map { |x| [x, y] }
217 end
218end

220def flood_points(image, x, y)
221 wanted = image.get(x, y)
222 seen = Set[[x, y]]
223 stack = [[x, y]]
224 points = []
225 until stack.empty?
226 cx, cy = stack.pop
227 points << [cx, cy]
228 NEIGHBORS4.each do |dx, dy|
229 nx, ny = cx + dx, cy + dy
230 next unless image.contains?(nx, ny) && image.get(nx, ny) == wanted
231 next unless seen.add?([nx, ny])
232 stack << [nx, ny]
233 end
234 end
235 points
236end

238def equalized(image)
239 counts = Array.new(image.maxval + 1, 0)
240 image.pixels.each { |pixel| counts[pixel] += 1 }
241 cdf = counts.each_with_object([]) { |count, values| values << count + (values.last || 0) }
242 n = image.pixels.length
243 first = cdf[image.pixels.min]
244 return image.with_pixels(image.pixels.dup) if first == n
245 image.mapped { |pixel| (2 * (cdf[pixel] - first) * image.maxval + n - first).div(2 * (n - first)) }
246end

248def components(image, threshold)
249 width = image.width
250 visited = Array.new(image.pixels.length, false)
251 groups = []
252 image.pixels.each_index do |start|
253 next if visited[start] || image.pixels[start] < threshold
254 visited[start] = true
255 stack = [[start % width, start / width]]
256 group = []
257 until stack.empty?
258 x, y = stack.pop
259 group << [x, y]
260 NEIGHBORS8.each do |dx, dy|
261 nx, ny = x + dx, y + dy
262 next unless image.contains?(nx, ny)
263 index = ny * width + nx
264 next if visited[index] || image.pixels[index] < threshold
265 visited[index] = true
266 stack << [nx, ny]
267 end
268 end
269 groups << group
270 end
271 groups
272end

274def component_text(image, threshold, group)
275 xx, yy = group.transpose
276 perimeter = group.count do |x, y|
277 NEIGHBORS4.any? do |dx, dy|
278 !image.contains?(x + dx, y + dy) || image.get(x + dx, y + dy) < threshold
279 end
280 end
281 "area #{group.length} perimeter #{perimeter} box #{xx.min} #{yy.min} #{xx.max} #{yy.max} " \
282 "center #{xx.sum.div(group.length)} #{yy.sum.div(group.length)}"
283end

285def distance_map(image, threshold)
286 if image.pixels.all? { |pixel| pixel >= threshold }
287 return image.with_pixels(Array.new(image.pixels.length, image.maxval))
288 end
289 w, h = image.width, image.height
290 far = w + h
291 distances = image.pixels.map { |pixel| pixel < threshold ? 0 : far }
292 (0...h).each do |y|
293 (0...w).each do |x|
294 index = y * w + x
295 distances[index] = [distances[index], distances[index - 1] + 1].min if x > 0
296 distances[index] = [distances[index], distances[index - w] + 1].min if y > 0
297 end
298 end
299 (h - 1).downto(0) do |y|
300 (w - 1).downto(0) do |x|
301 index = y * w + x
302 distances[index] = [distances[index], distances[index + 1] + 1].min if x < w - 1
303 distances[index] = [distances[index], distances[index + w] + 1].min if y < h - 1
304 end
305 end
306 image.with_pixels(distances.map { |value| [value, image.maxval].min })
307end

309def statistics(pixels, maxval)
310 values = pixels.sort
311 counts = Array.new(maxval + 1, 0)
312 pixels.each { |value| counts[value] += 1 }
313 "min #{values.first} max #{values.last} mean #{pixels.sum.div(pixels.length)} " \
314 "median #{values[(pixels.length - 1) / 2]} mode #{counts.index(counts.max)}"
315end

317def dithered(image)
318 errors = image.pixels.dup
319 pixels = Array.new(errors.length, 0)
320 neighbors = [[1, 0, 7], [-1, 1, 3], [0, 1, 5], [1, 1, 1]]
321 (0...image.height).each do |y|
322 (0...image.width).each do |x|
323 index = y * image.width + x
324 old = errors[index]
325 value = 2 * old >= image.maxval ? image.maxval : 0
326 pixels[index] = value
327 difference = old - value
328 neighbors.each do |dx, dy, weight|
329 next unless image.contains?(x + dx, y + dy)
330 errors[(y + dy) * image.width + x + dx] += (difference * weight).div(16)
331 end
332 end
333 end
334 image.with_pixels(pixels)
335end

337class Editor
338 attr_reader :output
339
340 def initialize
341 @current = nil
342 @undo = []
343 @redo = []
344 @slots = {}
345 @output = []
346 end
347
348 def emit(line)
349 @output << line
350 end
351
352 def load(lines)
353 replace(read_pgm(lines), "loaded")
354 rescue CommandError => error
355 emit("error: load: #{error.message}")
356 end
357
358 def replace(image, label)
359 @current = image
360 @undo.clear
361 @redo.clear
362 emit("#{label} #{image.summary}")
363 end
364
365 def commit(image, painted = nil)
366 @undo << @current
367 @redo.clear
368 @current = image
369 emit("ok #{image.summary}#{painted.nil? ? '' : " painted #{painted}"}")
370 end

372 def paint(points, value)
373 image = @current
374 touched = Set.new
375 points.each do |x, y|
376 touched.add(y * image.width + x) if image.contains?(x, y)
377 end
378 pixels = image.pixels.each_with_index.map { |pixel, index| touched.include?(index) ? value : pixel }
379 commit(image.with_pixels(pixels), touched.length)
380 end

382 def checked_crop(args)
383 x, y, w, h = args
384 image = @current
385 check(x >= 0 && y >= 0 && w >= 1 && h >= 1)
386 check(x + w <= image.width && y + h <= image.height)
387 args
388 end

390 def matching_slot(name)
391 other = @slots.fetch(name)
392 check(other.width == @current.width && other.height == @current.height, "size mismatch")
393 check(other.maxval == @current.maxval, "maxval mismatch")
394 other
395 end

397 def execute(words)
398 name = words.first
399 check(PATTERNS.key?(name), "unknown command #{name}")
400 pattern = PATTERNS[name]
401 given = words.drop(1)
402 fill = %w[RECT CIRCLE].include?(name) && given.length == pattern.length + 1 && given[-1] == "FILL"
403 given = given[0...-1] if fill
404 check(given.length == pattern.length, "bad arguments")
405 values = []
406 slot = nil
407 given.each_with_index do |word, index|
408 if pattern[index] == "i"
409 check(/\A(?:0|-?[1-9][0-9]{0,6})\z/.match?(word), "bad arguments")
410 values << word.to_i
411 else
412 check(/\A[a-z]{1,8}\z/.match?(word), "bad arguments")
413 slot = word
414 end
415 end
416 check(!@current.nil?, "no image") unless IMAGELESS.include?(name)
417 check(@slots.key?(slot), "no slot #{slot}") if SLOT_COMMANDS.include?(name)
418 run(name, values, slot, fill)
419 rescue CommandError => error
420 emit("error: #{error.message}")
421 end
422
423 def run(name, a, slot, fill)
424 image = @current
425 m = image&.maxval
426 case name
427 when "NEW"
428 w, h, maximum, value = a
429 check(w.between?(1, 64) && h.between?(1, 64))
430 check(maximum.between?(1, 255) && value.between?(0, maximum))
431 replace(Image.new(w, h, maximum, Array.new(w * h, value)), "created")
432 when "OUTPUT"
433 emit("P2")
434 emit("#{image.width} #{image.height}")
435 emit(m.to_s)
436 image.height.times { |y| emit(image.row(y).join(" ")) }
437 when "INVERT"
438 commit(image.mapped { |pixel| m - pixel })
439 when "BRIGHTNESS"
440 check(a[0].between?(-255, 255))
441 commit(image.mapped { |pixel| image.clamp(pixel + a[0]) })
442 when "CONTRAST"
443 num, den = a
444 check(num.between?(0, 100) && den.between?(1, 100))
445 center = (m + 1) / 2
446 commit(image.mapped { |pixel| image.clamp(center + ((pixel - center) * num).div(den)) })
447 when "THRESHOLD"
448 check(a[0].between?(0, m))
449 commit(image.mapped { |pixel| pixel >= a[0] ? m : 0 })
450 when "LEVELS"
451 low, high = a
452 check(low >= 0 && low < high && high <= m)
453 commit(image.mapped { |pixel| image.clamp(((pixel - low) * m).div(high - low)) })
454 when "POSTERIZE"
455 levels = a[0]
456 check(levels.between?(2, m + 1))
457 commit(image.mapped { |pixel| ((pixel * levels).div(m + 1) * m).div(levels - 1) })
458 when "MAXVAL"
459 maximum = a[0]
460 check(maximum.between?(1, 255))
461 pixels = image.pixels.map { |pixel| (2 * pixel * maximum + m).div(2 * m) }
462 commit(Image.new(image.width, image.height, maximum, pixels))
463 when "FLIPH"
464 commit(transformed(image, image.width, image.height) { |x, y| image.get(image.width - 1 - x, y) })
465 when "FLIPV"
466 commit(transformed(image, image.width, image.height) { |x, y| image.get(x, image.height - 1 - y) })
467 when "ROTATE"
468 angle = a[0]
469 check([90, 180, 270].include?(angle))
470 w, h = image.width, image.height
471 result = if angle == 180
472 transformed(image, w, h) { |x, y| image.get(w - 1 - x, h - 1 - y) }
473 elsif angle == 90
474 transformed(image, h, w) { |x, y| image.get(y, h - 1 - x) }
475 else
476 transformed(image, h, w) { |x, y| image.get(w - 1 - y, x) }
477 end
478 commit(result)
479 when "CROP"
480 commit(cropped(image, *checked_crop(a)))
481 when "PAD"
482 border, fill_value = a
483 check(border >= 1 && fill_value.between?(0, m))
484 w, h = image.width + 2 * border, image.height + 2 * border
485 check(w <= 64 && h <= 64, "too large")
486 commit(translated(image, w, h, border, border, fill_value))
487 when "SCALE"
488 num, den = a
489 check(num.between?(1, 16) && den.between?(1, 16))
490 w, h = image.width * num / den, image.height * num / den
491 check(w > 0 && h > 0, "too small")
492 check(w <= 64 && h <= 64, "too large")
493 commit(transformed(image, w, h) do |x, y|
494 image.get(((2 * x + 1) * den).div(2 * num), ((2 * y + 1) * den).div(2 * num))
495 end)
496 when "SHIFT"
497 dx, dy, fill_value = a
498 check(within_limit?(dx, dy) && fill_value.between?(0, m))
499 commit(translated(image, image.width, image.height, dx, dy, fill_value))
500 when "TILE"
501 nx, ny = a
502 check(nx >= 1 && ny >= 1)
503 w, h = image.width * nx, image.height * ny
504 check(w <= 64 && h <= 64, "too large")
505 commit(transformed(image, w, h) { |x, y| image.get(x % image.width, y % image.height) })
506 when "BLUR", "GAUSS", "SHARPEN"
507 commit(convolution(image, *KERNELS.fetch(name)))
508 when "CONVOLVE"
509 check(a.first(9).all? { |v| v.between?(-16, 16) } && a[9].between?(1, 256))
510 commit(convolution(image, a.first(9), a[9]))
511 when "SOBEL"
512 commit(sobel(image))
513 when "MEDIAN", "ERODE", "DILATE"
514 commit(rank_filtered(image, { "MEDIAN" => 4, "ERODE" => 0, "DILATE" => 8 }.fetch(name)))
515 when "OPEN"
516 commit(rank_filtered(rank_filtered(image, 0), 8))
517 when "CLOSE"
518 commit(rank_filtered(rank_filtered(image, 8), 0))
519 when "GRADIENT"
520 high = rank_filtered(image, 8).pixels
521 low = rank_filtered(image, 0).pixels
522 commit(image.with_pixels(high.zip(low).map { |a1, b1| a1 - b1 }))
523 when "LINE"
524 check(within_limit?(*a.first(4)) && a[4].between?(0, m))
525 paint(line_points(*a.first(4)), a[4])
526 when "RECT"
527 x, y, w, h, value = a
528 check(within_limit?(x, y) && w.between?(1, 1000) && h.between?(1, 1000) && value.between?(0, m))
529 paint(rectangle_points(image, x, y, w, h, fill), value)
530 when "CIRCLE"
531 cx, cy, radius, value = a
532 check(within_limit?(cx, cy) && radius.between?(0, 1000) && value.between?(0, m))
533 points = circle_points(cx, cy, radius)
534 paint(fill ? disc_points(image, points) : points, value)
535 when "FLOOD"
536 x, y, value = a
537 check(image.contains?(x, y) && value.between?(0, m))
538 paint(flood_points(image, x, y), value)
539 when "HISTOGRAM"
540 lows = Array.new(16)
541 highs = Array.new(16, 0)
542 counts = Array.new(16, 0)
543 (0..m).each do |value|
544 index = value * 16 / (m + 1)
545 lows[index] ||= value
546 highs[index] = value
547 end
548 image.pixels.each { |value| counts[value * 16 / (m + 1)] += 1 }
549 top = counts.max
550 16.times do |index|
551 next if lows[index].nil?
552 emit("#{lows[index].to_s.rjust(3)}-#{highs[index].to_s.rjust(3)} " \
553 "#{counts[index].to_s.rjust(4)} |#{'#' * (counts[index] * 32 / top)}")
554 end
555 when "EQUALIZE"
556 commit(equalized(image))
557 when "LABEL"
558 check(a[0].between?(0, m))
559 groups = components(image, a[0])
560 emit("labels #{groups.length}")
561 groups.each_with_index do |group, index|
562 emit("label #{index + 1} #{component_text(image, a[0], group)}")
563 end
564 when "DISTANCE"
565 check(a[0].between?(0, m))
566 commit(distance_map(image, a[0]))
567 when "STATS"
568 emit(statistics(image.pixels, m))
569 when "REGION"
570 emit(statistics(cropped(image, *checked_crop(a)).pixels, m))
571 when "DITHER"
572 commit(dithered(image))
573 when "ASCIIART"
574 image.height.times do |y|
575 emit(image.row(y).map { |pixel| SHADES[pixel * 10 / (m + 1)] }.join)
576 end
577 when "STORE"
578 @slots[slot] = image
579 emit("stored #{slot} #{image.summary}")
580 when "RECALL"
581 replace(@slots.fetch(slot), "recalled #{slot}")
582 when "DROP"
583 @slots.delete(slot)
584 emit("dropped #{slot}")
585 when "BLEND"
586 num, den = a
587 check(num >= 0 && num <= den && den.between?(1, 100))
588 other = matching_slot(slot)
589 pixels = image.pixels.each_with_index.map do |pixel, index|
590 (pixel * (den - num) + other.pixels[index] * num).div(den)
591 end
592 commit(image.with_pixels(pixels))
593 when "DIFF"
594 other = matching_slot(slot)
595 commit(image.with_pixels(image.pixels.each_with_index.map { |pixel, i| (pixel - other.pixels[i]).abs }))
596 when "PASTE"
597 x, y = a
598 check(within_limit?(x, y))
599 other = @slots.fetch(slot)
600 check(other.maxval == m, "maxval mismatch")
601 pixels = image.pixels.dup
602 painted = 0
603 other.height.times do |j|
604 other.width.times do |i|
605 next unless image.contains?(x + i, y + j)
606 pixels[(y + j) * image.width + x + i] = other.get(i, j)
607 painted += 1
608 end
609 end
