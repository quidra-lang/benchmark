101class Image {

103 this.width = width;
104 this.height = height;
105 this.maxval = maxval;
106 this.pixels = pixels;
107 }
108
109 get(x, y) {
110 return this.pixels[y * this.width + x];
111 }
112
113 contains(x, y) {
114 return x >= 0 && y >= 0 && x < this.width && y < this.height;
115 }
116
117 row(y) {
118 return this.pixels.slice(y * this.width, (y + 1) * this.width);
119 }
120
121 withPixels(pixels) {
122 return new Image(this.width, this.height, this.maxval, pixels);

128
129 neighborhood(x, y) {
130 const values = [];
131 for (let dy = -1; dy <= 1; ++dy) {
132 for (let dx = -1; dx <= 1; ++dx) {
133 values.push(
134 this.get(clamp(x + dx, 0, this.width - 1), clamp(y + dy, 0, this.height - 1)),
135 );
136 }
137 }
138 return values;
139 }

142 const checksum = this.pixels.reduce(
143 (total, value, index) => total + (index + 1) * value,
144 0,
145 );
146 return this.width + "x" + this.height + " maxval " + this.maxval + " checksum " + checksum;

150function readPGM(lines) {
151 const words = lines.flatMap((line) =>
152 line.split("#", 1)[0].trim().split(/\s+/).filter(Boolean),
153 );
154 requireValue(words.length > 0 && words[0] === "P2", "bad magic");
155 const header = (position, limit, reason) => {
156 requireValue(position < words.length, "truncated header");
157 requireValue(DIGITS.test(words[position]), "bad number");
158 const value = Number(words[position]);
159 requireValue(value >= 1 && value <= limit, reason);
160 return value;
161 };
162 const width = header(1, 64, "bad size");
163 const height = header(2, 64, "bad size");
164 const maxval = header(3, 255, "bad maxval");
165 const values = words.slice(4);
166 const pixels = [];
167 const needed = width * height;
168 for (let i = 0; i < needed; ++i) {
169 const x = i % width;
170 const y = Math.floor(i / width);
171 requireValue(i < values.length, "too few pixels (" + values.length + " of " + needed + ")");
172 requireValue(DIGITS.test(values[i]), "bad pixel at " + x + "," + y);
173 const pixel = Number(values[i]);
174 requireValue(pixel <= maxval, "pixel out of range at " + x + "," + y);
175 pixels.push(pixel);
176 }
177 requireValue(
178 values.length <= needed,
179 "too many pixels (" + values.length + " of " + needed + ")",
180 );
181 return new Image(width, height, maxval, pixels);
182}

192function rotated(image, angle) {
193 const w = image.width,
194 h = image.height;

239function sobel(image) {
240 return transformed(image, image.width, image.height, (x, y) => {
241 const q = image.neighborhood(x, y);
242 const gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6];
243 const gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
244 return image.clamp(Math.abs(gx) + Math.abs(gy));
245 });

248function linePoints(x0, y0, x1, y1) {
249 const dx = Math.abs(x1 - x0),
250 dy = -Math.abs(y1 - y0);
251 const sx = x0 < x1 ? 1 : -1,
252 sy = y0 < y1 ? 1 : -1;
253 let error = dx + dy,
254 x = x0,
255 y = y0;
256 const points = [];
257 while (true) {
258 points.push([x, y]);
259 if (x === x1 && y === y1) return points;
260 const twice = 2 * error;
261 if (twice >= dy) {
262 error += dy;
263 x += sx;
264 }
265 if (twice <= dx) {
266 error += dx;
267 y += sy;
268 }
269 }
270}

272function rectanglePoints(image, left, top, width, height, fill) {
273 const right = left + width - 1,
274 bottom = top + height - 1;
275 const points = [];
276 for (let y = Math.max(0, top); y <= Math.min(bottom, image.height - 1); ++y) {
277 for (let x = Math.max(0, left); x <= Math.min(right, image.width - 1); ++x) {
278 if (fill || x === left || x === right || y === top || y === bottom) points.push([x, y]);
279 }
280 }
281 return points;

284function circlePoints(cx, cy, radius) {
285 let a = radius,
286 b = 0,
287 error = 1 - radius;
288 const points = [];
289 while (a >= b) {
290 points.push(
291 [cx + a, cy + b],
292 [cx + b, cy + a],
293 [cx - b, cy + a],
294 [cx - a, cy + b],
295 [cx - a, cy - b],
296 [cx - b, cy - a],
297 [cx + b, cy - a],
298 [cx + a, cy - b],
299 );
300 ++b;
301 if (error < 0) error += 2 * b + 1;
302 else {
303 --a;
304 error += 2 * (b - a) + 1;
305 }
306 }
307 return points;
308}

310function discPoints(image, outline) {
311 const rows = new Map();
312 for (const [x, y] of outline) {
313 if (rows.has(y)) {
314 const [left, right] = rows.get(y);
315 rows.set(y, [Math.min(left, x), Math.max(right, x)]);
316 } else rows.set(y, [x, x]);
317 }
318 const result = [];
319 for (const [y, [left, right]] of rows) {
320 if (y < 0 || y >= image.height) continue;
321 for (let x = Math.max(0, left); x <= Math.min(image.width - 1, right); ++x) {
322 result.push([x, y]);
323 }
324 }
325 return result;
326}

328function floodPoints(image, x, y) {
329 const target = image.get(x, y);
330 const stack = [[x, y]];
331 const seen = new Set([y * image.width + x]);
332 const result = [];
333 while (stack.length > 0) {
334 const [cx, cy] = stack.pop();
335 result.push([cx, cy]);
336 for (const [dx, dy] of NEIGHBORS4) {
337 const nx = cx + dx,
338 ny = cy + dy;
339 if (!image.contains(nx, ny) || image.get(nx, ny) !== target) continue;
340 const index = ny * image.width + nx;
341 if (!seen.has(index)) {
342 seen.add(index);
343 stack.push([nx, ny]);
344 }
345 }
346 }
347 return result;
348}

350function equalized(image) {
351 const n = image.pixels.length;
352 const counts = Array(image.maxval + 1).fill(0);
353 for (const value of image.pixels) ++counts[value];
354 const cdf = [];
355 let accumulated = 0;
356 for (const count of counts) {
357 accumulated += count;
358 cdf.push(accumulated);
359 }
360 const minimumCDF = cdf[Math.min(...image.pixels)];
361 if (minimumCDF === n) return image.withPixels([...image.pixels]);
362 const span = n - minimumCDF;
363 return image.withPixels(
364 image.pixels.map((pixel) =>
365 Math.floor((2 * (cdf[pixel] - minimumCDF) * image.maxval + span) / (2 * span)),
366 ),
367 );
368}

370function components(image, threshold) {
371 const visited = new Set(),
372 groups = [];
373 for (let start = 0; start < image.pixels.length; ++start) {
374 if (visited.has(start) || image.pixels[start] < threshold) continue;
375 visited.add(start);
376 const stack = [[start % image.width, Math.floor(start / image.width)]];
377 const group = [];
378 while (stack.length > 0) {
379 const [x, y] = stack.pop();
380 group.push([x, y]);
381 for (const [dx, dy] of NEIGHBORS8) {
382 const nx = x + dx,
383 ny = y + dy;
384 if (!image.contains(nx, ny)) continue;
385 const index = ny * image.width + nx;
386 if (image.pixels[index] >= threshold && !visited.has(index)) {
387 visited.add(index);
388 stack.push([nx, ny]);
389 }
390 }
391 }
392 groups.push(group);
393 }
394 return groups;
395}

397function componentText(image, threshold, group) {
398 let minX = Infinity,
399 minY = Infinity,
400 maxX = -Infinity,
401 maxY = -Infinity;
402 let sumX = 0,
403 sumY = 0,
404 perimeter = 0;
405 for (const [x, y] of group) {
406 minX = Math.min(minX, x);
407 minY = Math.min(minY, y);
408 maxX = Math.max(maxX, x);
409 maxY = Math.max(maxY, y);
410 sumX += x;
411 sumY += y;
412 if (
413 NEIGHBORS4.some(
414 ([dx, dy]) =>
415 !image.contains(x + dx, y + dy) || image.get(x + dx, y + dy) < threshold,
416 )
417 )
418 ++perimeter;
419 }
420 const area = group.length;
421 return (
422 "area " +
423 area +
424 " perimeter " +
425 perimeter +
426 " box " +
427 [minX, minY, maxX, maxY].join(" ") +
428 " center " +
429 Math.floor(sumX / area) +
430 " " +
431 Math.floor(sumY / area)
432 );
433}

435function distanceMap(image, threshold) {
436 if (image.pixels.every((pixel) => pixel >= threshold)) {
437 return image.withPixels(Array(image.pixels.length).fill(image.maxval));
438 }
439 const w = image.width,
440 h = image.height;
441 const far = w + h;
442 const distances = image.pixels.map((pixel) => (pixel < threshold ? 0 : far));
443 for (let y = 0; y < h; ++y) {
444 for (let x = 0; x < w; ++x) {
445 const index = y * w + x;
446 if (x > 0) distances[index] = Math.min(distances[index], distances[index - 1] + 1);
447 if (y > 0) distances[index] = Math.min(distances[index], distances[index - w] + 1);
448 }
449 }
450 for (let y = h - 1; y >= 0; --y) {
451 for (let x = w - 1; x >= 0; --x) {
452 const index = y * w + x;
453 if (x < w - 1) distances[index] = Math.min(distances[index], distances[index + 1] + 1);
454 if (y < h - 1) distances[index] = Math.min(distances[index], distances[index + w] + 1);
455 }
456 }
457 return image.withPixels(distances.map((value) => Math.min(value, image.maxval)));
458}

460function statistics(values, maxval) {
461 const sorted = [...values].sort((a, b) => a - b);
462 const counts = Array(maxval + 1).fill(0);
463 for (const value of values) ++counts[value];
464 let mode = 0;
465 for (let value = 1; value <= maxval; ++value) {
466 if (counts[value] > counts[mode]) mode = value;
467 }
468 const n = sorted.length;
469 return (
470 "min " +
471 sorted[0] +
472 " max " +
473 sorted[n - 1] +
474 " mean " +
475 Math.floor(sum(values) / n) +
476 " median " +
477 sorted[Math.floor((n - 1) / 2)] +
478 " mode " +
479 mode
480 );
481}

483function dithered(image) {
484 const errors = [...image.pixels];
485 const pixels = Array(errors.length).fill(0);
486 const diffusion = [
487 [1, 0, 7],
488 [-1, 1, 3],
489 [0, 1, 5],
490 [1, 1, 1],
491 ];
492 for (let y = 0; y < image.height; ++y) {
493 for (let x = 0; x < image.width; ++x) {
494 const index = y * image.width + x;
495 const old = errors[index];
496 const next = 2 * old >= image.maxval ? image.maxval : 0;
497 pixels[index] = next;
498 const error = old - next;
499 for (const [dx, dy, weight] of diffusion) {
500 if (image.contains(x + dx, y + dy)) {
501 errors[(y + dy) * image.width + x + dx] += Math.floor((error * weight) / 16);
502 }
503 }
504 }
505 }
506 return image.withPixels(pixels);
507}

509class ImageEditor {
510 constructor() {
511 this.current = null;
512 this.undoHistory = [];
513 this.redoHistory = [];
514 this.slots = new Map();
515 this.output = [];
516 }
517
518 emit(text) {
519 this.output.push(text);
520 }
521
522 parse(words) {
523 const name = words[0],
524 given = words.slice(1);
525 requireValue(Object.hasOwn(PATTERNS, name), "unknown command " + name);
526 const pattern = PATTERNS[name];
527 const fill =
528 (name === "RECT" || name === "CIRCLE") &&
529 given.length === pattern.length + 1 &&
530 given.at(-1) === "FILL";
531 const args = fill ? given.slice(0, -1) : given;
532 requireValue(args.length === pattern.length, "bad arguments");
533 const numbers = [];
534 let slot = null;
535 for (let i = 0; i < args.length; ++i) {
536 if (pattern[i] === "i") {
537 requireValue(INTEGER_WORD.test(args[i]), "bad arguments");
538 numbers.push(Number(args[i]));
539 } else {
540 requireValue(NAME_WORD.test(args[i]), "bad arguments");
541 slot = args[i];
542 }
543 }
544 return { name, numbers, slot, fill };
545 }
546
547 execute(words) {
548 try {
549 const command = this.parse(words);
550 if (!IMAGELESS.has(command.name)) requireValue(this.current !== null, "no image");
551 if (NEED_SLOT.has(command.name)) {
552 requireValue(this.slots.has(command.slot), "no slot " + command.slot);
553 }
554 this.run(command);
555 } catch (error) {
556 if (!(error instanceof CommandError)) throw error;
557 this.emit("error: " + error.message);
558 }
559 }
560
561 load(lines) {
562 try {
563 this.replace(readPGM(lines), "loaded");
564 } catch (error) {
565 if (!(error instanceof CommandError)) throw error;
566 this.emit("error: load: " + error.message);
567 }
568 }
569
570 replace(image, label) {
571 this.current = image;
572 this.undoHistory = [];
573 this.redoHistory = [];
574 this.emit(label + " " + image.summary());
575 }
576
577 commit(image, painted = null) {
578 this.undoHistory.push(this.current);
579 this.redoHistory = [];
580 this.current = image;
581 this.emit("ok " + image.summary() + (painted === null ? "" : " painted " + painted));
582 }
583
584 paint(points, value) {
585 const image = this.current;
586 const marked = new Set();
587 for (const [x, y] of points) {
588 if (image.contains(x, y)) marked.add(y * image.width + x);
589 }
590 this.commit(
591 image.withPixels(image.pixels.map((p, i) => (marked.has(i) ? value : p))),
592 marked.size,
593 );
594 }
595
596 cropArea([x, y, w, h]) {
597 const image = this.current;
598 requireValue(x >= 0 && y >= 0 && w >= 1 && h >= 1);
599 requireValue(x + w <= image.width && y + h <= image.height);
600 return [x, y, w, h];
601 }
602
603 matchingSlot(name) {
604 const other = this.slots.get(name),
605 image = this.current;
606 requireValue(other.width === image.width && other.height === image.height, "size mismatch");
607 requireValue(other.maxval === image.maxval, "maxval mismatch");
608 return other;
609 }
610
611 run({ name, numbers: a, slot, fill }) {
612 const image = this.current;
613 const m = image?.maxval;
614 switch (name) {
615 case "NEW": {
616 const [w, h, maxval, value] = a;
617 requireValue(w >= 1 && w <= 64 && h >= 1 && h <= 64);
618 requireValue(maxval >= 1 && maxval <= 255 && value >= 0 && value <= maxval);
619 this.replace(new Image(w, h, maxval, Array(w * h).fill(value)), "created");
620 break;
621 }
622 case "OUTPUT":
623 this.emit("P2");
624 this.emit(image.width + " " + image.height);
625 this.emit(String(m));
626 for (let y = 0; y < image.height; ++y) this.emit(image.row(y).join(" "));
627 break;
628 case "INVERT":
629 this.commit(image.withPixels(image.pixels.map((p) => m - p)));
630 break;
631 case "BRIGHTNESS":
632 requireValue(a[0] >= -255 && a[0] <= 255);
633 this.commit(image.withPixels(image.pixels.map((p) => image.clamp(p + a[0]))));
634 break;
635 case "CONTRAST": {
636 const [num, den] = a;
637 requireValue(num >= 0 && num <= 100 && den >= 1 && den <= 100);
638 const center = Math.floor((m + 1) / 2);
639 this.commit(
640 image.withPixels(
641 image.pixels.map((p) =>
642 image.clamp(center + Math.floor(((p - center) * num) / den)),
643 ),
644 ),
645 );
646 break;
647 }
648 case "THRESHOLD":
649 requireValue(a[0] >= 0 && a[0] <= m);
650 this.commit(image.withPixels(image.pixels.map((p) => (p >= a[0] ? m : 0))));
651 break;
652 case "LEVELS": {
653 const [low, high] = a;
654 requireValue(low >= 0 && low < high && high <= m);
655 this.commit(
656 image.withPixels(
657 image.pixels.map((p) =>
658 image.clamp(Math.floor(((p - low) * m) / (high - low))),
659 ),
660 ),
661 );
662 break;
663 }
664 case "POSTERIZE": {
665 const count = a[0];
666 requireValue(count >= 2 && count <= m + 1);
667 this.commit(
668 image.withPixels(
669 image.pixels.map((p) =>
670 Math.floor((Math.floor((p * count) / (m + 1)) * m) / (count - 1)),
671 ),
672 ),
673 );
674 break;
675 }
676 case "MAXVAL": {
677 const maxval = a[0];
678 requireValue(maxval >= 1 && maxval <= 255);
679 const pixels = image.pixels.map((p) => Math.floor((2 * p * maxval + m) / (2 * m)));
680 this.commit(new Image(image.width, image.height, maxval, pixels));
681 break;
682 }
683 case "FLIPH":
684 this.commit(
685 transformed(image, image.width, image.height, (x, y) =>
686 image.get(image.width - 1 - x, y),
687 ),
688 );
689 break;
690 case "FLIPV":
691 this.commit(
692 transformed(image, image.width, image.height, (x, y) =>
693 image.get(x, image.height - 1 - y),
694 ),
695 );
696 break;
697 case "ROTATE":
698 requireValue([90, 180, 270].includes(a[0]));
699 this.commit(rotated(image, a[0]));
700 break;
701 case "CROP":
702 this.commit(cropped(image, ...this.cropArea(a)));
703 break;
704 case "PAD": {
705 const [n, value] = a;
706 requireValue(n >= 1 && value >= 0 && value <= m);
707 const w = image.width + 2 * n,
708 h = image.height + 2 * n;
709 requireValue(w <= 64 && h <= 64, "too large");
710 this.commit(translated(image, w, h, n, n, value));
711 break;
712 }
713 case "SCALE": {
714 const [num, den] = a;
715 requireValue(num >= 1 && num <= 16 && den >= 1 && den <= 16);
716 const w = Math.floor((image.width * num) / den);
717 const h = Math.floor((image.height * num) / den);
718 requireValue(w > 0 && h > 0, "too small");
719 requireValue(w <= 64 && h <= 64, "too large");
720 this.commit(scaled(image, w, h, num, den));
721 break;
722 }
723 case "SHIFT":
724 requireValue(withinLimit(a[0], a[1]) && a[2] >= 0 && a[2] <= m);
725 this.commit(translated(image, image.width, image.height, ...a));
726 break;
727 case "TILE": {
728 const [nx, ny] = a;
729 requireValue(nx >= 1 && ny >= 1);
730 const w = image.width * nx,
731 h = image.height * ny;
732 requireValue(w <= 64 && h <= 64, "too large");
733 this.commit(
734 transformed(image, w, h, (x, y) =>
735 image.get(x % image.width, y % image.height),
736 ),
737 );
738 break;
739 }
740 case "BLUR":
741 case "GAUSS":
742 case "SHARPEN":
743 this.commit(convolution(image, ...KERNELS[name]));
744 break;
745 case "CONVOLVE":
746 requireValue(
747 a.slice(0, 9).every((k) => k >= -16 && k <= 16) && a[9] >= 1 && a[9] <= 256,
748 );
749 this.commit(convolution(image, a.slice(0, 9), a[9]));
750 break;
751 case "SOBEL":
752 this.commit(sobel(image));
753 break;
754 case "MEDIAN":
755 case "ERODE":
756 case "DILATE":
757 this.commit(rankFiltered(image, { MEDIAN: 4, ERODE: 0, DILATE: 8 }[name]));
758 break;
759 case "OPEN":
760 this.commit(rankFiltered(rankFiltered(image, 0), 8));
761 break;
762 case "CLOSE":
763 this.commit(rankFiltered(rankFiltered(image, 8), 0));
764 break;
765 case "GRADIENT": {
766 const high = rankFiltered(image, 8).pixels;
767 const low = rankFiltered(image, 0).pixels;
768 this.commit(image.withPixels(high.map((p, i) => p - low[i])));
769 break;
770 }
771 case "LINE":
772 requireValue(withinLimit(...a.slice(0, 4)) && a[4] >= 0 && a[4] <= m);
773 this.paint(linePoints(...a.slice(0, 4)), a[4]);
774 break;
775 case "RECT":
776 requireValue(
777 withinLimit(a[0], a[1]) &&
778 a[2] >= 1 &&
779 a[2] <= 1000 &&
780 a[3] >= 1 &&
781 a[3] <= 1000 &&
782 a[4] >= 0 &&
783 a[4] <= m,
784 );
785 this.paint(rectanglePoints(image, ...a.slice(0, 4), fill), a[4]);
786 break;
787 case "CIRCLE": {
788 const [cx, cy, radius, value] = a;
789 requireValue(
790 withinLimit(cx, cy) &&
791 radius >= 0 &&
792 radius <= 1000 &&
793 value >= 0 &&
794 value <= m,
795 );
796 const outline = circlePoints(cx, cy, radius);
797 this.paint(fill ? discPoints(image, outline) : outline, value);
798 break;
799 }
800 case "FLOOD": {
801 const [x, y, value] = a;
802 requireValue(image.contains(x, y) && value >= 0 && value <= m);
803 this.paint(floodPoints(image, x, y), value);
804 break;
805 }
806 case "HISTOGRAM": {
807 const lows = Array(16).fill(null),
808 highs = Array(16).fill(0);
809 const counts = Array(16).fill(0);
810 for (let v = 0; v <= m; ++v) {
811 const bin = Math.floor((v * 16) / (m + 1));
812 if (lows[bin] === null) lows[bin] = v;
813 highs[bin] = v;
814 }
815 for (const p of image.pixels) ++counts[Math.floor((p * 16) / (m + 1))];
816 const top = Math.max(...counts);
817 for (let bin = 0; bin < 16; ++bin) {
818 if (lows[bin] === null) continue;
819 this.emit(
820 String(lows[bin]).padStart(3) +
821 "-" +
822 String(highs[bin]).padStart(3) +
823 " " +
824 String(counts[bin]).padStart(4) +
825 " |" +
826 "#".repeat(Math.floor((counts[bin] * 32) / top)),
827 );
828 }
829 break;
830 }
831 case "EQUALIZE":
832 this.commit(equalized(image));
833 break;
834 case "LABEL": {
835 requireValue(a[0] >= 0 && a[0] <= m);
836 const groups = components(image, a[0]);
837 this.emit("labels " + groups.length);
838 groups.forEach((group, index) =>
839 this.emit("label " + (index + 1) + " " + componentText(image, a[0], group)),
840 );
841 break;
842 }
843 case "DISTANCE":
844 requireValue(a[0] >= 0 && a[0] <= m);
845 this.commit(distanceMap(image, a[0]));
846 break;
847 case "STATS":
848 this.emit(statistics(image.pixels, m));
849 break;
850 case "REGION":
851 this.emit(statistics(cropped(image, ...this.cropArea(a)).pixels, m));
852 break;
853 case "DITHER":
854 this.commit(dithered(image));
855 break;
856 case "ASCIIART":
857 for (let y = 0; y < image.height; ++y) {
858 this.emit(
859 image
860 .row(y)
861 .map((p) => SHADES[Math.floor((p * SHADES.length) / (m + 1))])
862 .join(""),
863 );
864 }
865 break;
866 case "STORE":
867 this.slots.set(slot, image);
868 this.emit("stored " + slot + " " + image.summary());
869 break;
870 case "RECALL":
871 this.replace(this.slots.get(slot), "recalled " + slot);
872 break;
873 case "DROP":
874 this.slots.delete(slot);
875 this.emit("dropped " + slot);
876 break;
877 case "BLEND": {
878 const [num, den] = a;
879 requireValue(num >= 0 && num <= den && den >= 1 && den <= 100);
880 const other = this.matchingSlot(slot);
881 this.commit(
882 image.withPixels(
883 image.pixels.map((p, i) =>
884 Math.floor((p * (den - num) + other.pixels[i] * num) / den),
885 ),
886 ),
887 );
888 break;
889 }
890 case "DIFF": {
891 const other = this.matchingSlot(slot);
892 this.commit(
893 image.withPixels(image.pixels.map((p, i) => Math.abs(p - other.pixels[i]))),
894 );
895 break;
896 }
897 case "PASTE": {
898 const [x, y] = a;
899 requireValue(withinLimit(x, y));
900 const other = this.slots.get(slot);
901 requireValue(other.maxval === m, "maxval mismatch");
902 const pixels = [...image.pixels];
903 let painted = 0;
904 for (let j = 0; j < other.height; ++j) {
905 for (let i = 0; i < other.width; ++i) {
906 if (image.contains(x + i, y + j)) {
907 pixels[(y + j) * image.width + x + i] = other.get(i, j);
908 ++painted;
909 }
910 }
911 }
912 this.commit(image.withPixels(pixels), painted);
913 break;
914 }
915 case "SLOTS":
916 if (this.slots.size === 0) this.emit("no slots");
917 for (const name of [...this.slots.keys()].sort()) {
918 this.emit(name + " " + this.slots.get(name).summary());
919 }
920 break;
921 case "INFO":
922 this.emit(
923 "image " +
924 (image === null ? "none" : image.summary()) +
925 " undo " +
926 this.undoHistory.length +
927 " redo " +
928 this.redoHistory.length +
929 " slots " +
930 this.slots.size,
931 );
932 break;
933 case "UNDO":
934 requireValue(this.undoHistory.length > 0, "nothing to undo");
935 this.redoHistory.push(image);
936 this.current = this.undoHistory.pop();
937 this.emit("undo " + this.current.summary());
938 break;
939 case "REDO":
940 requireValue(this.redoHistory.length > 0, "nothing to redo");
941 this.undoHistory.push(image);
942 this.current = this.redoHistory.pop();
943 this.emit("redo " + this.current.summary());
944 break;
945 }
946 }
947}

949function main() {
950 const lines = readFileSync(0, "utf8").split(/\r?\n/);
951 let position = 0;
952 const commands = Number(lines[position++]);
953 const editor = new ImageEditor();
954 for (let i = 0; i < commands; ++i) {
955 const words = lines[position++].split(" ");
956 if (words[0] === "LOAD") {
957 const count = Number(words[1]);
958 editor.load(lines.slice(position, position + count));
959 position += count;
960 } else {
961 editor.execute(words);
962 }
963 }
964 process.stdout.write(editor.output.join("\n") + "\n");
965}
