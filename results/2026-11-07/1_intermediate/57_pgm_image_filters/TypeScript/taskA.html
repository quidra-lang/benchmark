154function readPGM(lines: string[]): Image {
155 const words = lines.flatMap((line) =>
156 line.split("#", 1)[0].trim().split(/\s+/).filter(Boolean),
157 );
158 requireValue(words.length > 0 && words[0] === "P2", "bad magic");
159 const header = (position: number, limit: number, reason: string): number => {
160 requireValue(position < words.length, "truncated header");
161 requireValue(DIGITS.test(words[position]), "bad number");
162 const value = Number(words[position]);
163 requireValue(value >= 1 && value <= limit, reason);
164 return value;
165 };
166 const width = header(1, 64, "bad size");
167 const height = header(2, 64, "bad size");
168 const maxval = header(3, 255, "bad maxval");
169 const values = words.slice(4);
170 const pixels = [];
171 const needed = width * height;
172 for (let i = 0; i < needed; ++i) {
173 const x = i % width;
174 const y = Math.floor(i / width);
175 requireValue(i < values.length, "too few pixels (" + values.length + " of " + needed + ")");
176 requireValue(DIGITS.test(values[i]), "bad pixel at " + x + "," + y);
177 const pixel = Number(values[i]);
178 requireValue(pixel <= maxval, "pixel out of range at " + x + "," + y);
179 pixels.push(pixel);
180 }
181 requireValue(
182 values.length <= needed,
183 "too many pixels (" + values.length + " of " + needed + ")",
184 );
185 return new Image(width, height, maxval, pixels);
186}

261function sobel(image: Image): Image {
262 return transformed(image, image.width, image.height, (x, y) => {
263 const q = image.neighborhood(x, y);
264 const gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6];
265 const gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
266 return image.clamp(Math.abs(gx) + Math.abs(gy));
267 });

270function linePoints(x0: number, y0: number, x1: number, y1: number): Point[] {
271 const dx = Math.abs(x1 - x0),
272 dy = -Math.abs(y1 - y0);
273 const sx = x0 < x1 ? 1 : -1,
274 sy = y0 < y1 ? 1 : -1;
275 let error = dx + dy,
276 x = x0,
277 y = y0;
278 const points: Point[] = [];
279 while (true) {
280 points.push([x, y]);
281 if (x === x1 && y === y1) return points;
282 const twice = 2 * error;
283 if (twice >= dy) {
284 error += dy;
285 x += sx;
286 }
287 if (twice <= dx) {
288 error += dx;
289 y += sy;
290 }
291 }
292}

307 if (fill || x === left || x === right || y === top || y === bottom) points.push([x, y]);
308 }

313function circlePoints(cx: number, cy: number, radius: number): Point[] {
314 let a = radius,
315 b = 0,
316 error = 1 - radius;
317 const points: Point[] = [];
318 while (a >= b) {
319 points.push(
320 [cx + a, cy + b],
321 [cx + b, cy + a],
322 [cx - b, cy + a],
323 [cx - a, cy + b],
324 [cx - a, cy - b],
325 [cx - b, cy - a],
326 [cx + b, cy - a],
327 [cx + a, cy - b],
328 );
329 ++b;
330 if (error < 0) error += 2 * b + 1;
331 else {
332 --a;
333 error += 2 * (b - a) + 1;
334 }
335 }
336 return points;
337}

339function discPoints(image: Image, outline: Point[]): Point[] {
340 const rows = new Map<number, Point>();
341 for (const [x, y] of outline) {
342 if (rows.has(y)) {
343 const [left, right] = rows.get(y)!;
344 rows.set(y, [Math.min(left, x), Math.max(right, x)]);
345 } else rows.set(y, [x, x]);
346 }
347 const result: Point[] = [];
348 for (const [y, [left, right]] of rows) {
349 if (y < 0 || y >= image.height) continue;
350 for (let x = Math.max(0, left); x <= Math.min(image.width - 1, right); ++x) {
351 result.push([x, y]);
352 }
353 }
354 return result;
355}

357function floodPoints(image: Image, x: number, y: number): Point[] {
358 const target = image.get(x, y);
359 const stack: Point[] = [[x, y]];
360 const seen = new Set<number>([y * image.width + x]);
361 const result: Point[] = [];
362 while (stack.length > 0) {
363 const [cx, cy] = stack.pop()!;
364 result.push([cx, cy]);
365 for (const [dx, dy] of NEIGHBORS4) {
366 const nx = cx + dx,
367 ny = cy + dy;
368 if (!image.contains(nx, ny) || image.get(nx, ny) !== target) continue;
369 const index = ny * image.width + nx;
370 if (!seen.has(index)) {
371 seen.add(index);
372 stack.push([nx, ny]);
373 }
374 }
375 }
376 return result;
377}

379function equalized(image: Image): Image {
380 const n = image.pixels.length;
381 const counts = Array(image.maxval + 1).fill(0);
382 for (const value of image.pixels) ++counts[value];
383 const cdf = [];
384 let accumulated = 0;
385 for (const count of counts) {
386 accumulated += count;
387 cdf.push(accumulated);
388 }
389 const minimumCDF = cdf[Math.min(...image.pixels)];
390 if (minimumCDF === n) return image.withPixels([...image.pixels]);
391 const span = n - minimumCDF;
392 return image.withPixels(
393 image.pixels.map((pixel) =>
394 Math.floor((2 * (cdf[pixel] - minimumCDF) * image.maxval + span) / (2 * span)),
395 ),
396 );
397}

399function components(image: Image, threshold: number): Point[][] {
400 const visited = new Set<number>(),
401 groups: Point[][] = [];
402 for (let start = 0; start < image.pixels.length; ++start) {
403 if (visited.has(start) || image.pixels[start] < threshold) continue;
404 visited.add(start);
405 const stack: Point[] = [[start % image.width, Math.floor(start / image.width)]];
406 const group: Point[] = [];
407 while (stack.length > 0) {
408 const [x, y] = stack.pop()!;
409 group.push([x, y]);
410 for (const [dx, dy] of NEIGHBORS8) {
411 const nx = x + dx,
412 ny = y + dy;
413 if (!image.contains(nx, ny)) continue;
414 const index = ny * image.width + nx;
415 if (image.pixels[index] >= threshold && !visited.has(index)) {
416 visited.add(index);
417 stack.push([nx, ny]);
418 }
419 }
420 }
421 groups.push(group);
422 }
423 return groups;
424}

426function componentText(image: Image, threshold: number, group: Point[]): string {
427 let minX = Infinity,
428 minY = Infinity,
429 maxX = -Infinity,
430 maxY = -Infinity;
431 let sumX = 0,
432 sumY = 0,
433 perimeter = 0;
434 for (const [x, y] of group) {
435 minX = Math.min(minX, x);
436 minY = Math.min(minY, y);
437 maxX = Math.max(maxX, x);
438 maxY = Math.max(maxY, y);
439 sumX += x;
440 sumY += y;
441 if (
442 NEIGHBORS4.some(
443 ([dx, dy]) =>
444 !image.contains(x + dx, y + dy) || image.get(x + dx, y + dy) < threshold,
445 )
446 )
447 ++perimeter;
448 }
449 const area = group.length;
450 return (
451 "area " +
452 area +
453 " perimeter " +
454 perimeter +
455 " box " +
456 [minX, minY, maxX, maxY].join(" ") +
457 " center " +
458 Math.floor(sumX / area) +
459 " " +
460 Math.floor(sumY / area)
461 );
462}

464function distanceMap(image: Image, threshold: number): Image {
465 if (image.pixels.every((pixel) => pixel >= threshold)) {
466 return image.withPixels(Array(image.pixels.length).fill(image.maxval));
467 }
468 const w = image.width,
469 h = image.height;
470 const far = w + h;
471 const distances = image.pixels.map((pixel) => (pixel < threshold ? 0 : far));
472 for (let y = 0; y < h; ++y) {
473 for (let x = 0; x < w; ++x) {
474 const index = y * w + x;
475 if (x > 0) distances[index] = Math.min(distances[index], distances[index - 1] + 1);
476 if (y > 0) distances[index] = Math.min(distances[index], distances[index - w] + 1);
477 }
478 }
479 for (let y = h - 1; y >= 0; --y) {
480 for (let x = w - 1; x >= 0; --x) {
481 const index = y * w + x;
482 if (x < w - 1) distances[index] = Math.min(distances[index], distances[index + 1] + 1);
483 if (y < h - 1) distances[index] = Math.min(distances[index], distances[index + w] + 1);
484 }
485 }
486 return image.withPixels(distances.map((value) => Math.min(value, image.maxval)));
487}

489function statistics(values: number[], maxval: number): string {
490 const sorted = [...values].sort((a, b) => a - b);
491 const counts = Array(maxval + 1).fill(0);
492 for (const value of values) ++counts[value];
493 let mode = 0;
494 for (let value = 1; value <= maxval; ++value) {
495 if (counts[value] > counts[mode]) mode = value;
496 }
497 const n = sorted.length;
498 return (
499 "min " +
500 sorted[0] +
501 " max " +
502 sorted[n - 1] +
503 " mean " +
504 Math.floor(sum(values) / n) +
505 " median " +
506 sorted[Math.floor((n - 1) / 2)] +
507 " mode " +
508 mode
509 );
510}

512function dithered(image: Image): Image {
513 const errors = [...image.pixels];
514 const pixels = Array(errors.length).fill(0);
515 const diffusion = [
516 [1, 0, 7],
517 [-1, 1, 3],
518 [0, 1, 5],
519 [1, 1, 1],
520 ];
521 for (let y = 0; y < image.height; ++y) {
522 for (let x = 0; x < image.width; ++x) {
523 const index = y * image.width + x;
524 const old = errors[index];
525 const next = 2 * old >= image.maxval ? image.maxval : 0;
526 pixels[index] = next;
527 const error = old - next;
528 for (const [dx, dy, weight] of diffusion) {
529 if (image.contains(x + dx, y + dy)) {
530 errors[(y + dy) * image.width + x + dx] += Math.floor((error * weight) / 16);
531 }
532 }
533 }
534 }
535 return image.withPixels(pixels);
536}

538class ImageEditor {
539 current: Image | null = null;
540 undoHistory: Image[] = [];
541 redoHistory: Image[] = [];
542 slots = new Map<string, Image>();
543 output: string[] = [];
544
545 emit(text: string): void {
546 this.output.push(text);
547 }
548
549 parse(words: string[]): ParsedCommand {
550 const name = words[0],
551 given = words.slice(1);
552 requireValue(Object.hasOwn(PATTERNS, name), "unknown command " + name);
553 const pattern = PATTERNS[name];
554 const fill =
555 (name === "RECT" || name === "CIRCLE") &&
556 given.length === pattern.length + 1 &&
557 given.at(-1) === "FILL";
558 const args = fill ? given.slice(0, -1) : given;
559 requireValue(args.length === pattern.length, "bad arguments");
560 const numbers: number[] = [];
561 let slot = "";
562 for (let i = 0; i < args.length; ++i) {
563 if (pattern[i] === "i") {
564 requireValue(INTEGER_WORD.test(args[i]), "bad arguments");
565 numbers.push(Number(args[i]));
566 } else {
567 requireValue(NAME_WORD.test(args[i]), "bad arguments");
568 slot = args[i];
569 }
570 }
571 return { name, numbers, slot, fill };
572 }
573
574 execute(words: string[]): void {
575 try {
576 const command = this.parse(words);
577 if (!IMAGELESS.has(command.name)) requireValue(this.current !== null, "no image");
578 if (NEED_SLOT.has(command.name)) {
579 requireValue(this.slots.has(command.slot), "no slot " + command.slot);
580 }
581 this.run(command);
582 } catch (error) {
583 if (!(error instanceof CommandError)) throw error;
584 this.emit("error: " + error.message);
585 }
586 }
587
588 load(lines: string[]): void {
589 try {
590 this.replace(readPGM(lines), "loaded");
591 } catch (error) {
592 if (!(error instanceof CommandError)) throw error;
593 this.emit("error: load: " + error.message);
594 }
595 }
596
597 replace(image: Image, label: string): void {
598 this.current = image;
599 this.undoHistory = [];
600 this.redoHistory = [];
601 this.emit(label + " " + image.summary());
602 }

604 commit(image: Image, painted: number | null = null): void {
605 this.undoHistory.push(this.current!);
606 this.redoHistory = [];
607 this.current = image;
608 this.emit("ok " + image.summary() + (painted === null ? "" : " painted " + painted));
609 }

611 paint(points: Point[], value: number): void {
612 const image = this.current!;
613 const marked = new Set<number>();
614 for (const [x, y] of points) {
615 if (image.contains(x, y)) marked.add(y * image.width + x);
616 }
617 this.commit(
618 image.withPixels(image.pixels.map((p, i) => (marked.has(i) ? value : p))),
619 marked.size,
620 );
621 }

623 cropArea([x, y, w, h]: number[]): Rectangle {
624 const image = this.current!;

626 requireValue(x + w <= image.width && y + h <= image.height);
627 return [x, y, w, h];

630 matchingSlot(name: string): Image {
631 const other = this.slots.get(name)!,
632 image = this.current!;
633 requireValue(other.width === image.width && other.height === image.height, "size mismatch");
634 requireValue(other.maxval === image.maxval, "maxval mismatch");
635 return other;
636 }
637
638 run({ name, numbers: a, slot, fill }: ParsedCommand): void {
639 const image = this.current!;
640 const m = image?.maxval ?? 0;
641 switch (name) {
642 case "NEW": {
643 const [w, h, maxval, value] = a;
644 requireValue(w >= 1 && w <= 64 && h >= 1 && h <= 64);
645 requireValue(maxval >= 1 && maxval <= 255 && value >= 0 && value <= maxval);
646 this.replace(new Image(w, h, maxval, Array(w * h).fill(value)), "created");
647 break;
648 }
649 case "OUTPUT":
650 this.emit("P2");
651 this.emit(image.width + " " + image.height);
652 this.emit(String(m));
653 for (let y = 0; y < image.height; ++y) this.emit(image.row(y).join(" "));
654 break;
655 case "INVERT":
656 this.commit(image.withPixels(image.pixels.map((p) => m - p)));
657 break;
658 case "BRIGHTNESS":
659 requireValue(a[0] >= -255 && a[0] <= 255);
660 this.commit(image.withPixels(image.pixels.map((p) => image.clamp(p + a[0]))));
661 break;
662 case "CONTRAST": {
663 const [num, den] = a;
664 requireValue(num >= 0 && num <= 100 && den >= 1 && den <= 100);
665 const center = Math.floor((m + 1) / 2);
666 this.commit(
667 image.withPixels(
668 image.pixels.map((p) =>
669 image.clamp(center + Math.floor(((p - center) * num) / den)),
670 ),
671 ),
672 );
673 break;
674 }
675 case "THRESHOLD":
676 requireValue(a[0] >= 0 && a[0] <= m);
677 this.commit(image.withPixels(image.pixels.map((p) => (p >= a[0] ? m : 0))));
678 break;
679 case "LEVELS": {
680 const [low, high] = a;
681 requireValue(low >= 0 && low < high && high <= m);
682 this.commit(
683 image.withPixels(
684 image.pixels.map((p) =>
685 image.clamp(Math.floor(((p - low) * m) / (high - low))),
686 ),
687 ),
688 );
689 break;
690 }
691 case "POSTERIZE": {
692 const count = a[0];
693 requireValue(count >= 2 && count <= m + 1);
694 this.commit(
695 image.withPixels(
696 image.pixels.map((p) =>
697 Math.floor((Math.floor((p * count) / (m + 1)) * m) / (count - 1)),
698 ),
699 ),
700 );
701 break;
702 }
703 case "MAXVAL": {
704 const maxval = a[0];
705 requireValue(maxval >= 1 && maxval <= 255);
706 const pixels = image.pixels.map((p) => Math.floor((2 * p * maxval + m) / (2 * m)));
707 this.commit(new Image(image.width, image.height, maxval, pixels));
708 break;
709 }
710 case "FLIPH":
711 this.commit(
712 transformed(image, image.width, image.height, (x, y) =>
713 image.get(image.width - 1 - x, y),
714 ),
715 );
716 break;
717 case "FLIPV":
718 this.commit(
719 transformed(image, image.width, image.height, (x, y) =>
720 image.get(x, image.height - 1 - y),
721 ),
722 );
723 break;
724 case "ROTATE":
725 requireValue([90, 180, 270].includes(a[0]));
726 this.commit(rotated(image, a[0]));
727 break;
728 case "CROP":
729 this.commit(cropped(image, ...this.cropArea(a)));
730 break;
731 case "PAD": {
732 const [n, value] = a;
733 requireValue(n >= 1 && value >= 0 && value <= m);
734 const w = image.width + 2 * n,
735 h = image.height + 2 * n;
736 requireValue(w <= 64 && h <= 64, "too large");
737 this.commit(translated(image, w, h, n, n, value));
738 break;
739 }
740 case "SCALE": {
741 const [num, den] = a;
742 requireValue(num >= 1 && num <= 16 && den >= 1 && den <= 16);
743 const w = Math.floor((image.width * num) / den);
744 const h = Math.floor((image.height * num) / den);
745 requireValue(w > 0 && h > 0, "too small");
746 requireValue(w <= 64 && h <= 64, "too large");
747 this.commit(scaled(image, w, h, num, den));
748 break;
749 }
750 case "SHIFT":
751 requireValue(withinLimit(a[0], a[1]) && a[2] >= 0 && a[2] <= m);
752 this.commit(translated(image, image.width, image.height, ...a));
753 break;
754 case "TILE": {
755 const [nx, ny] = a;
756 requireValue(nx >= 1 && ny >= 1);
757 const w = image.width * nx,
758 h = image.height * ny;
759 requireValue(w <= 64 && h <= 64, "too large");
760 this.commit(
761 transformed(image, w, h, (x, y) =>
762 image.get(x % image.width, y % image.height),
763 ),
764 );
765 break;
766 }
767 case "BLUR":
768 case "GAUSS":
769 case "SHARPEN":
770 this.commit(convolution(image, ...KERNELS[name]));
771 break;
772 case "CONVOLVE":
773 requireValue(
774 a.slice(0, 9).every((k) => k >= -16 && k <= 16) && a[9] >= 1 && a[9] <= 256,
775 );
776 this.commit(convolution(image, a.slice(0, 9), a[9]));
777 break;
778 case "SOBEL":
779 this.commit(sobel(image));
780 break;
781 case "MEDIAN":
782 case "ERODE":
783 case "DILATE":
784 this.commit(rankFiltered(image, { MEDIAN: 4, ERODE: 0, DILATE: 8 }[name]));
785 break;
786 case "OPEN":
787 this.commit(rankFiltered(rankFiltered(image, 0), 8));
788 break;
789 case "CLOSE":
790 this.commit(rankFiltered(rankFiltered(image, 8), 0));
791 break;
792 case "GRADIENT": {
793 const high = rankFiltered(image, 8).pixels;
794 const low = rankFiltered(image, 0).pixels;
795 this.commit(image.withPixels(high.map((p, i) => p - low[i])));
796 break;
797 }
798 case "LINE":
799 requireValue(withinLimit(...a.slice(0, 4)) && a[4] >= 0 && a[4] <= m);
800 this.paint(linePoints(...a.slice(0, 4)), a[4]);
801 break;
802 case "RECT":
803 requireValue(
804 withinLimit(a[0], a[1]) &&
805 a[2] >= 1 &&
806 a[2] <= 1000 &&
807 a[3] >= 1 &&
808 a[3] <= 1000 &&
809 a[4] >= 0 &&
810 a[4] <= m,
811 );
812 this.paint(rectanglePoints(image, ...a.slice(0, 4), fill), a[4]);
813 break;
814 case "CIRCLE": {
815 const [cx, cy, radius, value] = a;
816 requireValue(
817 withinLimit(cx, cy) &&
818 radius >= 0 &&
819 radius <= 1000 &&
820 value >= 0 &&
821 value <= m,
822 );
823 const outline = circlePoints(cx, cy, radius);
824 this.paint(fill ? discPoints(image, outline) : outline, value);
825 break;
826 }
827 case "FLOOD": {
828 const [x, y, value] = a;
829 requireValue(image.contains(x, y) && value >= 0 && value <= m);
830 this.paint(floodPoints(image, x, y), value);
831 break;
832 }
833 case "HISTOGRAM": {
834 const lows = Array(16).fill(null),
835 highs = Array(16).fill(0);
836 const counts = Array(16).fill(0);
837 for (let v = 0; v <= m; ++v) {
838 const bin = Math.floor((v * 16) / (m + 1));
839 if (lows[bin] === null) lows[bin] = v;
840 highs[bin] = v;
841 }
842 for (const p of image.pixels) ++counts[Math.floor((p * 16) / (m + 1))];
843 const top = Math.max(...counts);
844 for (let bin = 0; bin < 16; ++bin) {
845 if (lows[bin] === null) continue;
846 this.emit(
847 String(lows[bin]).padStart(3) +
848 "-" +
849 String(highs[bin]).padStart(3) +
850 " " +
851 String(counts[bin]).padStart(4) +
852 " |" +
853 "#".repeat(Math.floor((counts[bin] * 32) / top)),
854 );
855 }
856 break;
857 }
858 case "EQUALIZE":
859 this.commit(equalized(image));
860 break;
861 case "LABEL": {
862 requireValue(a[0] >= 0 && a[0] <= m);
863 const groups = components(image, a[0]);
864 this.emit("labels " + groups.length);
865 groups.forEach((group, index) =>
866 this.emit("label " + (index + 1) + " " + componentText(image, a[0], group)),
867 );
868 break;
869 }
870 case "DISTANCE":
871 requireValue(a[0] >= 0 && a[0] <= m);
872 this.commit(distanceMap(image, a[0]));
873 break;
874 case "STATS":
875 this.emit(statistics(image.pixels, m));
876 break;
877 case "REGION":
878 this.emit(statistics(cropped(image, ...this.cropArea(a)).pixels, m));
879 break;
880 case "DITHER":
881 this.commit(dithered(image));
882 break;
883 case "ASCIIART":
884 for (let y = 0; y < image.height; ++y) {
885 this.emit(
886 image
887 .row(y)
888 .map((p) => SHADES[Math.floor((p * SHADES.length) / (m + 1))])
889 .join(""),
890 );
891 }
892 break;
893 case "STORE":
894 this.slots.set(slot, image);
895 this.emit("stored " + slot + " " + image.summary());
896 break;
897 case "RECALL":
898 this.replace(this.slots.get(slot)!, "recalled " + slot);
899 break;
900 case "DROP":
901 this.slots.delete(slot);
902 this.emit("dropped " + slot);
903 break;
904 case "BLEND": {
905 const [num, den] = a;
906 requireValue(num >= 0 && num <= den && den >= 1 && den <= 100);
907 const other = this.matchingSlot(slot);
908 this.commit(
909 image.withPixels(
910 image.pixels.map((p, i) =>
911 Math.floor((p * (den - num) + other.pixels[i] * num) / den),
912 ),
913 ),
914 );
915 break;
916 }
917 case "DIFF": {
918 const other = this.matchingSlot(slot);
919 this.commit(
920 image.withPixels(image.pixels.map((p, i) => Math.abs(p - other.pixels[i]))),
921 );
922 break;
923 }
924 case "PASTE": {
925 const [x, y] = a;
926 requireValue(withinLimit(x, y));
927 const other = this.slots.get(slot)!;
928 requireValue(other.maxval === m, "maxval mismatch");
929 const pixels = [...image.pixels];
930 let painted = 0;
931 for (let j = 0; j < other.height; ++j) {
932 for (let i = 0; i < other.width; ++i) {
933 if (image.contains(x + i, y + j)) {
934 pixels[(y + j) * image.width + x + i] = other.get(i, j);
935 ++painted;
936 }
937 }
938 }
939 this.commit(image.withPixels(pixels), painted);
940 break;
941 }
942 case "SLOTS":
943 if (this.slots.size === 0) this.emit("no slots");
944 for (const name of [...this.slots.keys()].sort()) {
945 this.emit(name + " " + this.slots.get(name)!.summary());
946 }
947 break;
948 case "INFO":
949 this.emit(
950 "image " +
951 (image === null ? "none" : image.summary()) +
952 " undo " +
953 this.undoHistory.length +
954 " redo " +
955 this.redoHistory.length +
956 " slots " +
957 this.slots.size,
958 );
959 break;
960 case "UNDO":
961 requireValue(this.undoHistory.length > 0, "nothing to undo");
962 this.redoHistory.push(image);
963 this.current = this.undoHistory.pop()!;
964 this.emit("undo " + this.current.summary());
965 break;
966 case "REDO":
967 requireValue(this.redoHistory.length > 0, "nothing to redo");
968 this.undoHistory.push(image);
969 this.current = this.redoHistory.pop()!;
970 this.emit("redo " + this.current.summary());
971 break;
972 }
973 }
974}

976function main(): void {
977 const lines = readFileSync(0, "utf8").split(/\r?\n/);
978 let position = 0;
979 const commands = Number(lines[position++]);
980 const editor = new ImageEditor();
981 for (let i = 0; i < commands; ++i) {
982 const words = lines[position++].split(" ");
983 if (words[0] === "LOAD") {
984 const count = Number(words[1]);
985 editor.load(lines.slice(position, position + count));
986 position += count;
987 } else {
988 editor.execute(words);
989 }
990 }
991 process.stdout.write(editor.output.join("\n") + "\n");
992}
