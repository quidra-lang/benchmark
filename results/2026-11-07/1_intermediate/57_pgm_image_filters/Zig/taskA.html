138fn allDigits(word: []const u8) bool {
139 if (word.len == 0) return false;
140 for (word) |c| {

146fn isIntegerWord(word: []const u8) bool {
147 if (std.mem.eql(u8, word, "0")) return true;
148 const digits = if (std.mem.startsWith(u8, word, "-")) word[1..] else word;
149 return allDigits(digits) and digits.len <= 7 and digits[0] != '0';
150}

297fn convolution(gpa: Allocator, image: Image, kernel: [9]i32, divisor: i32) Allocator.Error!Image {
298 const result = try image.alike(gpa);
299 for (0..image.height) |y| {
300 for (0..image.width) |x| {
301 var total: i32 = 0;
302 for (kernel, image.neighborhood(x, y)) |weight, value| total += weight * value;
303 result.set(x, y, image.clamp(@divFloor(total, divisor)));
304 }
305 }
306 return result;

321fn sobelEdges(gpa: Allocator, image: Image) Allocator.Error!Image {
322 const result = try image.alike(gpa);
323 for (0..image.height) |y| {
324 for (0..image.width) |x| {
325 const q = image.neighborhood(x, y);
326 const gx = (q[2] + 2 * q[5] + q[8]) - (q[0] + 2 * q[3] + q[6]);
327 const gy = (q[6] + 2 * q[7] + q[8]) - (q[0] + 2 * q[1] + q[2]);
328 result.set(x, y, image.clamp(absolute(gx) + absolute(gy)));
329 }
330 }
331 return result;
332}

334fn linePoints(gpa: Allocator, x0: i32, y0: i32, x1: i32, y1: i32) Allocator.Error![]const Point {
335 const dx = absolute(x1 - x0);
336 const dy = -absolute(y1 - y0);
337 const step_x: i32 = if (x0 < x1) 1 else -1;
338 const step_y: i32 = if (y0 < y1) 1 else -1;
339 var err = dx + dy;
340 var x = x0;
341 var y = y0;
342 var points: std.ArrayList(Point) = .empty;
343 while (true) {
344 try points.append(gpa, .{ .x = x, .y = y });
345 if (x == x1 and y == y1) return points.items;
346 const doubled = 2 * err;
347 if (doubled >= dy) {
348 err += dy;
349 x += step_x;
350 }
351 if (doubled <= dx) {
352 err += dx;
353 y += step_y;
354 }
355 }
356}

358fn rectanglePoints(
359 gpa: Allocator,
360 image: Image,
361 left: i32,
362 top: i32,
363 width: i32,
364 height: i32,
365 filled: bool,
366) Allocator.Error![]const Point {
367 const right = left + width - 1;
368 const bottom = top + height - 1;
369 var points: std.ArrayList(Point) = .empty;
370 var y: i32 = @max(top, 0);
371 while (y <= @min(bottom, signed(image.height) - 1)) : (y += 1) {
372 var x: i32 = @max(left, 0);
373 while (x <= @min(right, signed(image.width) - 1)) : (x += 1) {
374 if (filled or x == left or x == right or y == top or y == bottom) {
375 try points.append(gpa, .{ .x = x, .y = y });
376 }
377 }
378 }
379 return points.items;
380}

382fn circlePoints(gpa: Allocator, cx: i32, cy: i32, radius: i32) Allocator.Error![]const Point {
383 var a = radius;
384 var b: i32 = 0;
385 var err = 1 - radius;
386 var points: std.ArrayList(Point) = .empty;
387 while (a >= b) {
388 try points.appendSlice(gpa, &[_]Point{
389 .{ .x = cx + a, .y = cy + b }, .{ .x = cx + b, .y = cy + a },
390 .{ .x = cx - b, .y = cy + a }, .{ .x = cx - a, .y = cy + b },
391 .{ .x = cx - a, .y = cy - b }, .{ .x = cx - b, .y = cy - a },
392 .{ .x = cx + b, .y = cy - a }, .{ .x = cx + a, .y = cy - b },
393 });
394 b += 1;
395 if (err < 0) {
396 err += 2 * b + 1;
397 } else {
398 a -= 1;
399 err += 2 * (b - a) + 1;
400 }
401 }
402 return points.items;
403}

405fn discPoints(gpa: Allocator, image: Image, outline: []const Point) Allocator.Error![]const Point {
406 const Span = struct { low: i32, high: i32 };
407 const spans = try gpa.alloc(?Span, image.height);
408 @memset(spans, null);
409 for (outline) |point| {
410 if (point.y < 0 or point.y >= signed(image.height)) continue;
411 const span = &spans[@intCast(point.y)];
412 span.* = if (span.*) |known|
413 .{ .low = @min(known.low, point.x), .high = @max(known.high, point.x) }
414 else
415 .{ .low = point.x, .high = point.x };
416 }
417 var points: std.ArrayList(Point) = .empty;
418 for (spans, 0..) |maybe_span, y| {
419 const span = maybe_span orelse continue;
420 var x: i32 = @max(span.low, 0);
421 while (x <= @min(span.high, signed(image.width) - 1)) : (x += 1) {
422 try points.append(gpa, .{ .x = x, .y = signed(y) });
423 }
424 }
425 return points.items;
426}

428fn floodPoints(gpa: Allocator, image: Image, x: i32, y: i32) Allocator.Error![]const Point {
429 const target = image.at(x, y);
430 const visited = try gpa.alloc(bool, image.pixels.len);
431 @memset(visited, false);
432 visited[image.index(x, y)] = true;
433 var stack: std.ArrayList(Point) = .empty;
434 try stack.append(gpa, .{ .x = x, .y = y });
435 var points: std.ArrayList(Point) = .empty;
436 while (stack.pop()) |point| {
437 try points.append(gpa, point);
438 for (four_neighbors) |offset| {
439 const nx = point.x + offset[0];
440 const ny = point.y + offset[1];
441 if (!image.contains(nx, ny) or image.at(nx, ny) != target) continue;
442 const index = image.index(nx, ny);
443 if (!visited[index]) {
444 visited[index] = true;
445 try stack.append(gpa, .{ .x = nx, .y = ny });
446 }
447 }
448 }
449 return points.items;
450}

452fn equalized(gpa: Allocator, image: Image) Allocator.Error!Image {
453 const total = signed(image.pixels.len);
454 const cdf = try gpa.alloc(i32, @intCast(image.maxval + 1));
455 @memset(cdf, 0);
456 for (image.pixels) |pixel| cdf[@intCast(pixel)] += 1;
457 for (1..cdf.len) |value| cdf[value] += cdf[value - 1];
458 const lowest = cdf[@intCast(std.mem.min(i32, image.pixels))];
459 if (lowest == total) return image;
460 const span = total - lowest;
461 const result = try image.alike(gpa);
462 for (result.pixels, image.pixels) |*target, pixel| {
463 target.* = @divFloor(2 * (cdf[@intCast(pixel)] - lowest) * image.maxval + span, 2 * span);
464 }
465 return result;
466}

468fn components(gpa: Allocator, image: Image, threshold: i32) Allocator.Error![]const []const Point {
469 const visited = try gpa.alloc(bool, image.pixels.len);
470 @memset(visited, false);
471 var result: std.ArrayList([]const Point) = .empty;
472 for (image.pixels, 0..) |pixel, start| {
473 if (pixel < threshold or visited[start]) continue;
474 visited[start] = true;
475 var stack: std.ArrayList(Point) = .empty;
476 try stack.append(gpa, .{
477 .x = signed(start % image.width),
478 .y = signed(start / image.width),
479 });
480 var members: std.ArrayList(Point) = .empty;
481 while (stack.pop()) |point| {
482 try members.append(gpa, point);
483 for (eight_neighbors) |offset| {
484 const nx = point.x + offset[0];
485 const ny = point.y + offset[1];
486 if (!image.contains(nx, ny)) continue;
487 const index = image.index(nx, ny);
488 if (image.pixels[index] >= threshold and !visited[index]) {
489 visited[index] = true;
490 try stack.append(gpa, .{ .x = nx, .y = ny });
491 }
492 }
493 }
494 try result.append(gpa, members.items);
495 }
496 return result.items;
497}

499fn onBoundary(image: Image, point: Point, threshold: i32) bool {
500 for (four_neighbors) |offset| {
501 const x = point.x + offset[0];
502 const y = point.y + offset[1];
503 if (!image.contains(x, y) or image.at(x, y) < threshold) return true;
504 }
505 return false;
506}

508fn writeComponent(
509 out: *Writer,
510 image: Image,
511 threshold: i32,
512 number: usize,
513 members: []const Point,
514) Writer.Error!void {
515 var low = members[0];
516 var high = members[0];
517 var sum: Point = .{ .x = 0, .y = 0 };
518 var perimeter: usize = 0;
519 for (members) |point| {
520 low = .{ .x = @min(low.x, point.x), .y = @min(low.y, point.y) };
521 high = .{ .x = @max(high.x, point.x), .y = @max(high.y, point.y) };
522 sum = .{ .x = sum.x + point.x, .y = sum.y + point.y };
523 if (onBoundary(image, point, threshold)) perimeter += 1;
524 }
525 const area = signed(members.len);
526 try out.print("label {d} area {d} perimeter {d} box {d} {d} {d} {d} center {d} {d}\n", .{
527 number, area, perimeter, low.x, low.y, high.x, high.y,
528 @divFloor(sum.x, area), @divFloor(sum.y, area),
529 });
530}

532fn distanceMap(gpa: Allocator, image: Image, threshold: i32) Allocator.Error!Image {
533 const result = try image.alike(gpa);
534 if (std.mem.min(i32, image.pixels) >= threshold) {
535 @memset(result.pixels, image.maxval);
536 return result;
537 }
538 const width = image.width;
539 const far = signed(width + image.height);
540 const distances = result.pixels;
541 for (distances, image.pixels) |*distance, pixel| distance.* = if (pixel < threshold) 0 else far;
542 for (0..distances.len) |index| {
543 if (index % width > 0) {
544 distances[index] = @min(distances[index], distances[index - 1] + 1);
545 }
546 if (index >= width) {
547 distances[index] = @min(distances[index], distances[index - width] + 1);
548 }
549 }
550 var index = distances.len;
551 while (index > 0) {
552 index -= 1;
553 if (index % width < width - 1) {
554 distances[index] = @min(distances[index], distances[index + 1] + 1);
555 }
556 if (index + width < distances.len) {
557 distances[index] = @min(distances[index], distances[index + width] + 1);
558 }
559 }
560 for (distances) |*distance| distance.* = @min(distance.*, image.maxval);
561 return result;
562}

564fn dithered(gpa: Allocator, image: Image) Allocator.Error!Image {
565 const errors = try gpa.dupe(i32, image.pixels);
566 const result = try image.alike(gpa);
567 for (0..image.height) |y| {
568 for (0..image.width) |x| {
569 const old = errors[y * image.width + x];
570 const new = if (2 * old >= image.maxval) image.maxval else 0;
571 result.set(x, y, new);
572 const err = old - new;
573 for (diffusion) |spread| {
574 const nx = signed(x) + spread[0];
575 const ny = signed(y) + spread[1];
576 if (image.contains(nx, ny)) {
577 errors[image.index(nx, ny)] += @divFloor(err * spread[2], 16);
578 }
579 }
580 }
581 }
582 return result;
583}

610 fn headerValue(
611 editor: *Editor,
612 words: []const []const u8,
613 position: usize,
614 limit: usize,
615 reason: []const u8,
616 ) Error!usize {
617 try editor.require(position < words.len, "truncated header");
618 const word = words[position];
619 try editor.require(allDigits(word), "bad number");
620 const value = std.fmt.parseInt(usize, word, 10) catch unreachable;
621 try editor.require(1 <= value and value <= limit, reason);
622 return value;
623 }

625 fn readPgm(editor: *Editor, lines: []const []const u8) Error!Image {
626 var words: std.ArrayList([]const u8) = .empty;
627 for (lines) |text| {
628 const content = text[0 .. std.mem.findScalar(u8, text, '#') orelse text.len];
629 var tokens = std.mem.tokenizeScalar(u8, content, ' ');
630 while (tokens.next()) |word| try words.append(editor.gpa, word);
631 }
632 const all = words.items;
633 try editor.require(all.len > 0 and std.mem.eql(u8, all[0], "P2"), "bad magic");
634 const width = try editor.headerValue(all, 1, max_side, "bad size");
635 const height = try editor.headerValue(all, 2, max_side, "bad size");
636 const maxval: i32 = @intCast(try editor.headerValue(all, 3, max_maxval, "bad maxval"));
637 const values = all[4..];
638 const total = width * height;
639 const pixels = try editor.gpa.alloc(i32, total);
640 for (pixels, 0..) |*pixel, index| {
641 const x = index % width;
642 const y = index / width;
643 if (index >= values.len) {
644 return editor.fail("too few pixels ({d} of {d})", .{ values.len, total });
645 }
646 const word = values[index];
647 if (!allDigits(word)) return editor.fail("bad pixel at {d},{d}", .{ x, y });
648 pixel.* = std.fmt.parseInt(i32, word, 10) catch unreachable;
649 if (pixel.* > maxval) return editor.fail("pixel out of range at {d},{d}", .{ x, y });
650 }
651 if (values.len > total) {
652 return editor.fail("too many pixels ({d} of {d})", .{ values.len, total });
653 }
654 return .{ .width = width, .height = height, .maxval = maxval, .pixels = pixels };
655 }

657 fn parse(editor: *Editor, words: []const []const u8) Error!Command {
658 const name = std.meta.stringToEnum(Name, words[0]) orelse
659 return editor.fail("unknown command {s}", .{words[0]});
660 const kinds = pattern(name);
661 var arguments = words[1..];
662 const fill = (name == .RECT or name == .CIRCLE) and arguments.len == kinds.len + 1 and
663 std.mem.eql(u8, arguments[arguments.len - 1], "FILL");
664 if (fill) arguments = arguments[0 .. arguments.len - 1];
665 try editor.require(arguments.len == kinds.len, "bad arguments");
666 var command: Command = .{ .name = name, .fill = fill };
667 var count: usize = 0;
668 for (kinds, arguments) |kind, word| {
669 if (kind == 'i') {
670 try editor.require(isIntegerWord(word), "bad arguments");
671 command.numbers[count] = std.fmt.parseInt(i32, word, 10) catch unreachable;
672 count += 1;
673 } else {
674 try editor.require(isNameWord(word), "bad arguments");
675 command.slot = word;
676 }
677 }
678 return command;
679 }

683 if (err != error.Invalid) return err;
684 try editor.out.print("error: {s}\n", .{editor.reason});
685 };

692 return editor.fail("no slot {s}", .{command.slot});

695 switch (command.name) {
696 .NEW => try editor.newImage(command),
697 .OUTPUT => try editor.writeImage(),
698 .INVERT => try editor.invert(),
699 .BRIGHTNESS => try editor.brightness(command),
700 .CONTRAST => try editor.contrast(command),
701 .THRESHOLD => try editor.thresholdImage(command),
702 .LEVELS => try editor.adjustLevels(command),
703 .POSTERIZE => try editor.posterize(command),
704 .MAXVAL => try editor.changeMaxval(command),
705 .FLIPH, .FLIPV => try editor.flip(command.name),
706 .ROTATE => try editor.rotate(command),
707 .CROP => try editor.crop(command),
708 .PAD => try editor.pad(command),
709 .SCALE => try editor.scale(command),
710 .SHIFT => try editor.shift(command),
711 .TILE => try editor.tile(command),
712 .BLUR, .GAUSS, .SHARPEN => try editor.kernelFilter(command.name),
713 .CONVOLVE => try editor.convolve(command),
714 .SOBEL => try editor.commit(try sobelEdges(gpa, editor.current.?)),
715 .MEDIAN, .ERODE, .DILATE => try editor.rankFilter(command.name),
716 .OPEN, .CLOSE, .GRADIENT => try editor.morphology(command.name),
717 .LINE => try editor.drawLine(command),
718 .RECT => try editor.drawRect(command),
719 .CIRCLE => try editor.drawCircle(command),
720 .FLOOD => try editor.floodFill(command),
721 .HISTOGRAM => try editor.histogram(),
722 .EQUALIZE => try editor.commit(try equalized(gpa, editor.current.?)),
723 .LABEL => try editor.label(command),
724 .DISTANCE => try editor.distance(command),
725 .STATS => try editor.writeStatistics(editor.current.?.pixels, editor.current.?.maxval),
726 .REGION => try editor.region(command),
727 .DITHER => try editor.commit(try dithered(gpa, editor.current.?)),
728 .ASCIIART => try editor.asciiArt(),
729 .STORE => try editor.store(command.slot),
730 .RECALL => try editor.recall(command.slot),
731 .DROP => try editor.drop(command.slot),

735 .SLOTS => try editor.listSlots(),
736 .INFO => try editor.info(),
737 .UNDO => try editor.undo(),
738 .REDO => try editor.redo(),

1069 fn histogram(editor: *Editor) Error!void {
1070 const image = editor.current.?;
1071 const levels: usize = @intCast(image.maxval + 1);
1072 var lows: [histogram_bins]?usize = @splat(null);

1074 var counts: [histogram_bins]usize = @splat(0);
1075 for (0..levels) |value| {
1076 const number = value * histogram_bins / levels;
1077 if (lows[number] == null) lows[number] = value;
1078 highs[number] = value;
1079 }
1080 for (image.pixels) |pixel| {
1081 counts[@as(usize, @intCast(pixel)) * histogram_bins / levels] += 1;
1082 }
1083 var top: usize = 0;
1084 for (lows, counts) |low, count| {
1085 if (low != null) top = @max(top, count);
1086 }
1087 for (lows, highs, counts) |maybe_low, high, count| {
1088 const low = maybe_low orelse continue;
1089 try editor.out.print("{d:>3}-{d:>3} {d:>4} |", .{ low, high, count });
1090 try editor.out.splatByteAll('#', count * bar_length / top);
1091 try editor.out.writeByte('\n');
1092 }

1247pub fn main(init: std.process.Init) !void {
1248 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1249 defer storage.deinit();
1250 const gpa = storage.allocator();
1251 var input_buffer: [4096]u8 = undefined;
1252 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
1253 var output_buffer: [4096]u8 = undefined;
1254 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1255 const text = try input.interface.allocRemaining(gpa, .unlimited);
1256 var lines = std.mem.splitScalar(u8, text, '\n');
1257 var editor: Editor = .{ .gpa = gpa, .out = &output.interface };
1258 const count = try std.fmt.parseInt(usize, lines.next().?, 10);
1259 for (0..count) |_| {
1260 var words: std.ArrayList([]const u8) = .empty;
1261 var tokens = std.mem.splitScalar(u8, lines.next().?, ' ');
1262 while (tokens.next()) |word| try words.append(gpa, word);
1263 if (std.mem.eql(u8, words.items[0], "LOAD")) {
1264 const data = try gpa.alloc([]const u8, try std.fmt.parseInt(usize, words.items[1], 10));
1265 for (data) |*line| line.* = lines.next().?;
1266 try editor.load(data);
1267 } else {
1268 try editor.execute(words.items);
1269 }
1270 }
1271 try output.interface.flush();
1272}
