10fn floor_divide(value: i32, divisor: i32) -> i32 {
11 let quotient = value / divisor;
12 if value < 0 && value % divisor != 0 { quotient - 1 } else { quotient }
13}

20struct Point {

29const FOUR: [Point; 4] = [
30 Point { x: 0, y: -1 }, Point { x: -1, y: 0 },
31 Point { x: 1, y: 0 }, Point { x: 0, y: 1 },
32];
33
34const EIGHT: [Point; 8] = [
35 Point { x: -1, y: -1 }, Point { x: 0, y: -1 },
36 Point { x: 1, y: -1 }, Point { x: -1, y: 0 },
37 Point { x: 1, y: 0 }, Point { x: -1, y: 1 },
38 Point { x: 0, y: 1 }, Point { x: 1, y: 1 },
39];

41#[derive(Clone)]
42struct Image {
43 width: i32,
44 height: i32,
45 maxval: i32,
46 pixels: Vec<i32>,
47}
48
49impl Image {
50 fn at(&self, x: i32, y: i32) -> i32 {
51 self.pixels[(y * self.width + x) as usize]
52 }
53
54 fn contains(&self, x: i32, y: i32) -> bool {
55 x >= 0 && y >= 0 && x < self.width && y < self.height
56 }
57
58 fn clamped(&self, value: i32) -> i32 {
59 value.clamp(0, self.maxval)
60 }
61
62 fn with_pixels(&self, pixels: Vec<i32>) -> Self {
63 Self { width: self.width, height: self.height, maxval: self.maxval, pixels }
64 }
65
66 fn map(&self, change: impl Fn(i32) -> i32) -> Self {
67 self.with_pixels(self.pixels.iter().map(|&pixel| change(pixel)).collect())
68 }
69
70 fn neighborhood(&self, x: i32, y: i32) -> [i32; 9] {
71 let mut result = [0; 9];
72 let mut index = 0;
73 for dy in -1..=1 {
74 for dx in -1..=1 {
75 result[index] = self.at(
76 (x + dx).clamp(0, self.width - 1),
77 (y + dy).clamp(0, self.height - 1),
78 );
79 index += 1;
80 }
81 }
82 result
83 }
84
85 fn summary(&self) -> String {
86 let checksum: i64 = self.pixels.iter().enumerate()
87 .map(|(index, &value)| (index as i64 + 1) * value as i64)
88 .sum();
89 format!("{}x{} maxval {} checksum {}", self.width, self.height, self.maxval, checksum)
90 }
91}

93fn transformed(
94 image: &Image, width: i32, height: i32,
95 pixel: impl Fn(i32, i32) -> i32,
96) -> Image {
97 let mut pixels = Vec::with_capacity((width * height) as usize);
98 for y in 0..height {
99 for x in 0..width { pixels.push(pixel(x, y)); }
100 }
101 Image { width, height, maxval: image.maxval, pixels }
102}

104fn cropped(image: &Image, x: i32, y: i32, width: i32, height: i32) -> Image {
105 transformed(image, width, height, |i, j| image.at(x + i, y + j))
106}

108fn translated(
109 image: &Image, width: i32, height: i32,
110 dx: i32, dy: i32, fill: i32,
111) -> Image {
112 transformed(image, width, height, |x, y| {
113 if image.contains(x - dx, y - dy) { image.at(x - dx, y - dy) } else { fill }
114 })
115}

117fn convolved(image: &Image, kernel: &[i32; 9], divisor: i32) -> Image {
118 transformed(image, image.width, image.height, |x, y| {
119 let q = image.neighborhood(x, y);
120 let total: i32 = q.iter().zip(kernel).map(|(&pixel, &weight)| pixel * weight).sum();
121 image.clamped(floor_divide(total, divisor))
122 })
123}

125fn rank_filtered(image: &Image, rank: usize) -> Image {
126 transformed(image, image.width, image.height, |x, y| {
127 let mut values = image.neighborhood(x, y);
128 values.sort_unstable();
129 values[rank]
130 })
131}

133fn sobel(image: &Image) -> Image {
134 transformed(image, image.width, image.height, |x, y| {
135 let q = image.neighborhood(x, y);
136 let gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6];
137 let gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
138 image.clamped(gx.abs() + gy.abs())
139 })
140}

142fn line_points(x0: i32, y0: i32, x1: i32, y1: i32) -> Vec<Point> {
143 let dx = (x1 - x0).abs();
144 let dy = -(y1 - y0).abs();
145 let sx = if x0 < x1 { 1 } else { -1 };
146 let sy = if y0 < y1 { 1 } else { -1 };
147 let mut error = dx + dy;
148 let (mut x, mut y) = (x0, y0);
149 let mut points = Vec::new();
150 loop {
151 points.push(Point::new(x, y));
152 if x == x1 && y == y1 { break; }
153 let twice = 2 * error;
154 if twice >= dy { error += dy; x += sx; }
155 if twice <= dx { error += dx; y += sy; }
156 }
157 points
158}

160fn rectangle_points(
161 image: &Image, left: i32, top: i32,
162 width: i32, height: i32, fill: bool,
163) -> Vec<Point> {
164 let right = left + width - 1;
165 let bottom = top + height - 1;
166 let first_x = left.max(0);
167 let last_x = right.min(image.width - 1);
168 let first_y = top.max(0);
169 let last_y = bottom.min(image.height - 1);
170 let mut points = Vec::new();
171 if first_x <= last_x && first_y <= last_y {
172 for y in first_y..=last_y {
173 for x in first_x..=last_x {
174 if fill || x == left || x == right || y == top || y == bottom {
175 points.push(Point::new(x, y));
176 }
177 }
178 }
179 }
180 points
181}

183fn circle_points(cx: i32, cy: i32, radius: i32) -> Vec<Point> {
184 let (mut a, mut b, mut error) = (radius, 0, 1 - radius);
185 let mut points = Vec::new();
186 while a >= b {
187 points.extend_from_slice(&[
188 Point::new(cx + a, cy + b), Point::new(cx + b, cy + a),
189 Point::new(cx - b, cy + a), Point::new(cx - a, cy + b),
190 Point::new(cx - a, cy - b), Point::new(cx - b, cy - a),
191 Point::new(cx + b, cy - a), Point::new(cx + a, cy - b),
192 ]);
193 b += 1;
194 if error < 0 { error += 2 * b + 1; }
195 else { a -= 1; error += 2 * (b - a) + 1; }
196 }
197 points
198}

200fn disc_points(image: &Image, outline: &[Point]) -> Vec<Point> {
201 let mut rows = BTreeMap::<i32, (i32, i32)>::new();
202 for point in outline {
203 let bounds = rows.entry(point.y).or_insert((point.x, point.x));
204 bounds.0 = bounds.0.min(point.x);
205 bounds.1 = bounds.1.max(point.x);
206 }
207 let mut points = Vec::new();
208 for (y, (left, right)) in rows {
209 if y < 0 || y >= image.height { continue; }
210 let start = left.max(0);
211 let end = right.min(image.width - 1);
212 if start <= end {
213 for x in start..=end { points.push(Point::new(x, y)); }
214 }
215 }
216 points
217}

219fn flood_points(image: &Image, x: i32, y: i32) -> Vec<Point> {
220 let target = image.at(x, y);
221 let mut stack = vec![Point::new(x, y)];
222 let mut seen = HashSet::from([(y * image.width + x) as usize]);
223 let mut points = Vec::new();
224 while let Some(p) = stack.pop() {
225 points.push(p);
226 for d in FOUR {
227 let (nx, ny) = (p.x + d.x, p.y + d.y);
228 if !image.contains(nx, ny) || image.at(nx, ny) != target { continue; }
229 if seen.insert((ny * image.width + nx) as usize) {
230 stack.push(Point::new(nx, ny));
231 }
232 }
233 }
234 points
235}

237fn equalized(image: &Image) -> Image {
238 let mut counts = vec![0; (image.maxval + 1) as usize];
239 for &pixel in &image.pixels { counts[pixel as usize] += 1; }
240 let mut running = 0;
241 let cdf: Vec<i32> = counts.into_iter().map(|v| { running += v; running }).collect();
242 let first = cdf[*image.pixels.iter().min().unwrap() as usize];
243 let size = image.pixels.len() as i32;
244 if first == size { return image.clone(); }
245 let span = size - first;
246 image.map(|pixel| (2 * (cdf[pixel as usize] - first) * image.maxval + span) / (2 * span))
247}

249fn components(image: &Image, threshold: i32) -> Vec<Vec<Point>> {
250 let mut visited = vec![false; image.pixels.len()];
251 let mut groups = Vec::new();
252 for start in 0..image.pixels.len() {
253 if visited[start] || image.pixels[start] < threshold { continue; }
254 visited[start] = true;
255 let mut stack = vec![Point::new(start as i32 % image.width, start as i32 / image.width)];
256 let mut group = Vec::new();
257 while let Some(p) = stack.pop() {
258 group.push(p);
259 for d in EIGHT {
260 let (nx, ny) = (p.x + d.x, p.y + d.y);
261 if !image.contains(nx, ny) { continue; }
262 let index = (ny * image.width + nx) as usize;
263 if !visited[index] && image.pixels[index] >= threshold {
264 visited[index] = true;
265 stack.push(Point::new(nx, ny));
266 }
267 }
268 }
269 groups.push(group);
270 }
271 groups
272}

274fn component_text(image: &Image, threshold: i32, group: &[Point]) -> String {
275 let mut left = group[0].x;
276 let mut right = group[0].x;
277 let mut top = group[0].y;
278 let mut bottom = group[0].y;
279 let (mut sum_x, mut sum_y, mut perimeter) = (0, 0, 0);
280 for p in group {
281 left = left.min(p.x);
282 right = right.max(p.x);
283 top = top.min(p.y);
284 bottom = bottom.max(p.y);
285 sum_x += p.x;
286 sum_y += p.y;
287 if FOUR.iter().any(|d| {
288 !image.contains(p.x + d.x, p.y + d.y) ||
289 image.at(p.x + d.x, p.y + d.y) < threshold
290 }) { perimeter += 1; }
291 }
292 let area = group.len() as i32;
293 format!("area {area} perimeter {perimeter} box {left} {top} {right} {bottom} \
294 center {} {}", sum_x / area, sum_y / area)
295}

297fn distance_map(image: &Image, threshold: i32) -> Image {
298 if image.pixels.iter().all(|&p| p >= threshold) {
299 return image.with_pixels(vec![image.maxval; image.pixels.len()]);
300 }
301 let (w, h) = (image.width as usize, image.height as usize);
302 let far = (w + h) as i32;
303 let mut distances: Vec<i32> = image.pixels.iter()
304 .map(|&p| if p < threshold { 0 } else { far }).collect();
305 for y in 0..h {
306 for x in 0..w {
307 let i = y * w + x;
308 if x > 0 { distances[i] = distances[i].min(distances[i - 1] + 1); }
309 if y > 0 { distances[i] = distances[i].min(distances[i - w] + 1); }
310 }
311 }
312 for y in (0..h).rev() {
313 for x in (0..w).rev() {
314 let i = y * w + x;
315 if x + 1 < w { distances[i] = distances[i].min(distances[i + 1] + 1); }
316 if y + 1 < h { distances[i] = distances[i].min(distances[i + w] + 1); }
317 }
318 }
319 image.with_pixels(distances.into_iter().map(|v| v.min(image.maxval)).collect())
320}

322fn statistics(values: &[i32], maxval: i32) -> String {
323 let mut sorted = values.to_vec();
324 sorted.sort_unstable();
325 let mut counts = vec![0; (maxval + 1) as usize];
326 let mut total: i64 = 0;
327 for &value in values {
328 counts[value as usize] += 1;
329 total += value as i64;
330 }
331 let mut mode = 0usize;
332 for i in 1..counts.len() {
333 if counts[i] > counts[mode] { mode = i; }
334 }
335 format!("min {} max {} mean {} median {} mode {}",
336 sorted[0], sorted[sorted.len() - 1], total / values.len() as i64,
337 sorted[(sorted.len() - 1) / 2], mode)
338}

340fn dithered(image: &Image) -> Image {
341 let mut errors = image.pixels.clone();
342 let mut pixels = vec![0; errors.len()];
343 let offsets = [(1, 0, 7), (-1, 1, 3), (0, 1, 5), (1, 1, 1)];
344 for y in 0..image.height {
345 for x in 0..image.width {
346 let i = (y * image.width + x) as usize;
347 let old = errors[i];
348 let value = if 2 * old >= image.maxval { image.maxval } else { 0 };
349 pixels[i] = value;
350 let error = old - value;
351 for &(dx, dy, weight) in &offsets {
352 let nx = x + dx;
353 let ny = y + dy;
354 if image.contains(nx, ny) {
355 errors[(ny * image.width + nx) as usize] += floor_divide(error * weight, 16);
356 }
357 }
358 }
359 }
360 image.with_pixels(pixels)
361}

363fn pgm_header(words: &[&str], index: usize, maximum: i32, reason: &str) -> Outcome<i32> {
364 check(index < words.len(), "truncated header")?;
365 check(!words[index].is_empty() && words[index].bytes().all(|b| b.is_ascii_digit()),
366 "bad number")?;
367 let value: i32 = words[index].parse().unwrap();
368 check(value >= 1 && value <= maximum, reason)?;
369 Ok(value)
370}

372fn parse_pgm(lines: &[&str]) -> Outcome<Image> {
373 let words: Vec<&str> = lines.iter()
374 .flat_map(|line| line.split('#').next().unwrap_or("").split_whitespace())
375 .collect();
376 check(words.first() == Some(&"P2"), "bad magic")?;
377 let width = pgm_header(&words, 1, 64, "bad size")?;
378 let height = pgm_header(&words, 2, 64, "bad size")?;
379 let maxval = pgm_header(&words, 3, 255, "bad maxval")?;
380 let count = words.len().saturating_sub(4);
381 let required = (width * height) as usize;
382 let mut pixels = Vec::with_capacity(required);
383 for i in 0..required {
384 let x = i as i32 % width;
385 let y = i as i32 / width;
386 check(i < count, &format!("too few pixels ({count} of {required})"))?;
387 let word = words[i + 4];
388 check(!word.is_empty() && word.bytes().all(|b| b.is_ascii_digit()),
389 &format!("bad pixel at {x},{y}"))?;
390 let value: i32 = word.parse().unwrap();
391 check(value <= maxval, &format!("pixel out of range at {x},{y}"))?;
392 pixels.push(value);
393 }
394 check(count <= required, &format!("too many pixels ({count} of {required})"))?;
395 Ok(Image { width, height, maxval, pixels })
396}

398fn command_pattern(name: &str) -> Option<&'static str> {
399 match name {
400 "NEW" => Some("iiii"),
401 "OUTPUT" | "INVERT" | "FLIPH" | "FLIPV" | "BLUR" | "GAUSS" |
402 "SHARPEN" | "SOBEL" | "MEDIAN" | "ERODE" | "DILATE" | "OPEN" |
403 "CLOSE" | "GRADIENT" | "HISTOGRAM" | "EQUALIZE" | "STATS" |
404 "DITHER" | "ASCIIART" | "SLOTS" | "INFO" | "UNDO" | "REDO" =>
405 Some(""),
406 "BRIGHTNESS" | "THRESHOLD" | "POSTERIZE" | "MAXVAL" | "ROTATE" |
407 "LABEL" | "DISTANCE" => Some("i"),
408 "CONTRAST" | "LEVELS" | "PAD" | "SCALE" | "TILE" => Some("ii"),
409 "SHIFT" | "FLOOD" => Some("iii"),
410 "CROP" | "REGION" | "CIRCLE" => Some("iiii"),
411 "RECT" | "LINE" => Some("iiiii"),
412 "CONVOLVE" => Some("iiiiiiiiii"),
413 "STORE" | "RECALL" | "DROP" => Some("n"),
414 "BLEND" => Some("nii"),
415 "DIFF" => Some("n"),
416 "PASTE" => Some("nii"),
417 _ => None,
418 }
419}

421fn integer_word(word: &str) -> bool {
422 if word == "0" { return true; }
423 let digits = word.strip_prefix('-').unwrap_or(word);
424 !digits.is_empty() && digits.len() <= 7 && !digits.starts_with('0') &&
425 digits.bytes().all(|b| b.is_ascii_digit())
426}
427
428fn name_word(word: &str) -> bool {
429 !word.is_empty() && word.len() <= 8 &&
430 word.bytes().all(|b| b.is_ascii_lowercase())
431}

433struct Command {
434 name: String,
435 args: Vec<i32>,
436 slot: String,
437 fill: bool,
438}
439
440fn parse_command(words: &[&str]) -> Outcome<Command> {
441 let name = words[0];
442 let pattern = command_pattern(name)
443 .ok_or_else(|| format!("unknown command {name}"))?;
444 let mut given = words[1..].to_vec();
445 let fill = (name == "RECT" || name == "CIRCLE") &&
446 given.len() == pattern.len() + 1 && given.last() == Some(&"FILL");
447 if fill { given.pop(); }
448 check(given.len() == pattern.len(), "bad arguments")?;
449 let mut args = Vec::new();
450 let mut slot = String::new();
451 for (kind, word) in pattern.bytes().zip(given) {
452 if kind == b'i' {
453 check(integer_word(word), "bad arguments")?;
454 args.push(word.parse().unwrap());
455 } else {
456 check(name_word(word), "bad arguments")?;
457 slot = word.to_string();
458 }
459 }
460 Ok(Command { name: name.to_string(), args, slot, fill })
461}

494 let summary = self.current.as_ref().unwrap().summary();
495 let suffix = painted.map(|count| format!(" painted {count}")).unwrap_or_default();
496 self.emit(format!("ok {summary}{suffix}"));
497 }

499 fn paint(&mut self, image: &Image, points: Vec<Point>, value: i32) {
500 let mut touched = HashSet::new();
501 for point in points {
502 if image.contains(point.x, point.y) {
503 touched.insert((point.y * image.width + point.x) as usize);
504 }
505 }
506 let mut pixels = image.pixels.clone();
507 for &index in &touched { pixels[index] = value; }
508 self.commit(image.with_pixels(pixels), Some(touched.len()));
509 }

519 let other = &self.slots[slot];

525 fn execute(&mut self, words: &[&str]) {
526 let result = (|| {
527 let command = parse_command(words)?;
528 let name = command.name.as_str();
529 let imageless = matches!(
530 name, "NEW" | "RECALL" | "DROP" | "SLOTS" | "INFO" | "UNDO" | "REDO"
531 );
532 check(imageless || self.current.is_some(), "no image")?;
533 if matches!(name, "RECALL" | "DROP" | "BLEND" | "DIFF" | "PASTE") {
534 check(self.slots.contains_key(&command.slot),
535 &format!("no slot {}", command.slot))?;
536 }
537 self.run(&command)
538 })();
539 if let Err(reason) = result { self.emit(format!("error: {reason}")); }
540 }

549 fn run(&mut self, command: &Command) -> Outcome<()> {
550 let name = command.name.as_str();
551 let a = &command.args;
552 let slot = command.slot.as_str();
553 let fill = command.fill;
554
555 match name {
556 "NEW" => {
557 check((1..=64).contains(&a[0]) && (1..=64).contains(&a[1]), "bad value")?;
558 check((1..=255).contains(&a[2]) && a[3] >= 0 && a[3] <= a[2],
559 "bad value")?;
560 self.replace(Image {
561 width: a[0], height: a[1], maxval: a[2],
562 pixels: vec![a[3]; (a[0] * a[1]) as usize],
563 }, "created");
564 return Ok(());
565 }
566 "RECALL" => {
567 self.replace(self.slots[slot].clone(), &format!("recalled {slot}"));
568 return Ok(());
569 }
570 "DROP" => {
571 self.slots.remove(slot);
572 self.emit(format!("dropped {slot}"));
573 return Ok(());
574 }
575 "SLOTS" => {
576 if self.slots.is_empty() { self.emit("no slots".to_string()); }
577 let lines: Vec<String> = self.slots.iter()
578 .map(|(name, image)| format!("{name} {}", image.summary())).collect();
579 self.output.extend(lines);
580 return Ok(());
581 }
582 "INFO" => {
583 let summary = self.current.as_ref()
584 .map(Image::summary).unwrap_or_else(|| "none".to_string());
585 self.emit(format!(
586 "image {summary} undo {} redo {} slots {}",
587 self.undo.len(), self.redo.len(), self.slots.len()
588 ));
589 return Ok(());
590 }
591 "UNDO" => {
592 check(!self.undo.is_empty(), "nothing to undo")?;
593 self.redo.push(self.current.take().unwrap());
594 self.current = self.undo.pop();
595 self.emit(format!("undo {}", self.current.as_ref().unwrap().summary()));
596 return Ok(());
597 }
598 "REDO" => {
599 check(!self.redo.is_empty(), "nothing to redo")?;
600 self.undo.push(self.current.take().unwrap());
601 self.current = self.redo.pop();
602 self.emit(format!("redo {}", self.current.as_ref().unwrap().summary()));
603 return Ok(());
604 }
605 _ => (),
606 }
607
608 let image = self.current.as_ref().unwrap().clone();
609 let (w, h, m) = (image.width, image.height, image.maxval);
610
611 match name {
612 "OUTPUT" => {
613 self.emit("P2".to_string());
614 self.emit(format!("{w} {h}"));
615 self.emit(m.to_string());
616 for y in 0..h {
617 let row = (0..w).map(|x| image.at(x, y).to_string())
618 .collect::<Vec<_>>().join(" ");
619 self.emit(row);
620 }
621 }
622 "INVERT" => self.commit(image.map(|p| m - p), None),
623 "BRIGHTNESS" => {
624 check((-255..=255).contains(&a[0]), "bad value")?;
625 self.commit(image.map(|p| image.clamped(p + a[0])), None);
626 }
627 "CONTRAST" => {
628 check((0..=100).contains(&a[0]) && (1..=100).contains(&a[1]), "bad value")?;
629 let center = (m + 1) / 2;
630 self.commit(image.map(|p| {
631 image.clamped(center + floor_divide((p - center) * a[0], a[1]))
632 }), None);
633 }
634 "THRESHOLD" => {
635 check((0..=m).contains(&a[0]), "bad value")?;
636 self.commit(image.map(|p| if p >= a[0] { m } else { 0 }), None);
637 }
638 "LEVELS" => {
639 check(a[0] >= 0 && a[0] < a[1] && a[1] <= m, "bad value")?;
640 self.commit(image.map(|p| {
641 image.clamped(floor_divide((p - a[0]) * m, a[1] - a[0]))
642 }), None);
643 }
644 "POSTERIZE" => {
645 check((2..=m + 1).contains(&a[0]), "bad value")?;
646 self.commit(image.map(|p| (p * a[0] / (m + 1)) * m / (a[0] - 1)), None);
647 }
648 "MAXVAL" => {
649 check((1..=255).contains(&a[0]), "bad value")?;
650 self.commit(Image {
651 width: w, height: h, maxval: a[0],
652 pixels: image.pixels.iter()
653 .map(|&p| (2 * p * a[0] + m) / (2 * m)).collect(),
654 }, None);
655 }
656 "FLIPH" => self.commit(
657 transformed(&image, w, h, |x, y| image.at(w - 1 - x, y)), None
658 ),
659 "FLIPV" => self.commit(
660 transformed(&image, w, h, |x, y| image.at(x, h - 1 - y)), None
661 ),
662 "ROTATE" => {
663 check([90, 180, 270].contains(&a[0]), "bad value")?;
664 let result = match a[0] {
665 90 => transformed(&image, h, w, |x, y| image.at(y, h - 1 - x)),
666 180 => transformed(&image, w, h, |x, y| image.at(w - 1 - x, h - 1 - y)),
667 _ => transformed(&image, h, w, |x, y| image.at(w - 1 - y, x)),
668 };
669 self.commit(result, None);
670 }
671 "CROP" => {
672 self.crop_area(&image, a)?;
673 self.commit(cropped(&image, a[0], a[1], a[2], a[3]), None);
674 }
675 "PAD" => {
676 check(a[0] >= 1 && a[1] >= 0 && a[1] <= m, "bad value")?;
677 check(w + 2 * a[0] <= 64 && h + 2 * a[0] <= 64, "too large")?;
678 self.commit(translated(&image, w + 2 * a[0], h + 2 * a[0],
679 a[0], a[0], a[1]), None);
680 }
681 "SCALE" => {
682 check((1..=16).contains(&a[0]) && (1..=16).contains(&a[1]), "bad value")?;
683 let nw = w * a[0] / a[1];
684 let nh = h * a[0] / a[1];
685 check(nw >= 1 && nh >= 1, "too small")?;
686 check(nw <= 64 && nh <= 64, "too large")?;
687 self.commit(transformed(&image, nw, nh, |x, y| {
688 image.at((2 * x + 1) * a[1] / (2 * a[0]),
689 (2 * y + 1) * a[1] / (2 * a[0]))
690 }), None);
691 }
692 "SHIFT" => {
693 check(within_limits(&a[..2]) && a[2] >= 0 && a[2] <= m, "bad value")?;
694 self.commit(translated(&image, w, h, a[0], a[1], a[2]), None);
695 }
696 "TILE" => {
697 check(a[0] >= 1 && a[1] >= 1, "bad value")?;
698 check(w * a[0] <= 64 && h * a[1] <= 64, "too large")?;
699 self.commit(transformed(&image, w * a[0], h * a[1], |x, y| {
700 image.at(x % w, y % h)
701 }), None);
702 }
703 "BLUR" | "GAUSS" | "SHARPEN" => {
704 let (kernel, divisor) = match name {
705 "BLUR" => ([1, 1, 1, 1, 1, 1, 1, 1, 1], 9),
706 "GAUSS" => ([1, 2, 1, 2, 4, 2, 1, 2, 1], 16),
707 _ => ([0, -1, 0, -1, 5, -1, 0, -1, 0], 1),
708 };
709 self.commit(convolved(&image, &kernel, divisor), None);
710 }
711 "CONVOLVE" => {
712 check(a[..9].iter().all(|&k| (-16..=16).contains(&k)) &&
713 (1..=256).contains(&a[9]), "bad value")?;
714 let kernel: [i32; 9] = a[..9].try_into().unwrap();
715 self.commit(convolved(&image, &kernel, a[9]), None);
716 }
717 "SOBEL" => self.commit(sobel(&image), None),
718 "MEDIAN" => self.commit(rank_filtered(&image, 4), None),
719 "ERODE" => self.commit(rank_filtered(&image, 0), None),
720 "DILATE" => self.commit(rank_filtered(&image, 8), None),
721 "OPEN" => self.commit(rank_filtered(&rank_filtered(&image, 0), 8), None),
722 "CLOSE" => self.commit(rank_filtered(&rank_filtered(&image, 8), 0), None),
723 "GRADIENT" => {
724 let high = rank_filtered(&image, 8);
725 let low = rank_filtered(&image, 0);
726 let result = image.pixels.iter().enumerate()
727 .map(|(i, _)| high.pixels[i] - low.pixels[i]).collect();
728 self.commit(image.with_pixels(result), None);
729 }
730 "LINE" => {
731 check(within_limits(&a[..4]) && (0..=m).contains(&a[4]), "bad value")?;
732 self.paint(&image, line_points(a[0], a[1], a[2], a[3]), a[4]);
733 }
734 "RECT" => {
735 check(within_limits(&a[..2]) && (1..=1000).contains(&a[2]) &&
736 (1..=1000).contains(&a[3]) && (0..=m).contains(&a[4]),
737 "bad value")?;
738 self.paint(&image, rectangle_points(&image, a[0], a[1], a[2], a[3], fill), a[4]);
739 }
740 "CIRCLE" => {
741 check(within_limits(&a[..2]) && (0..=1000).contains(&a[2]) &&
742 (0..=m).contains(&a[3]), "bad value")?;
743 let outline = circle_points(a[0], a[1], a[2]);
744 let points = if fill { disc_points(&image, &outline) } else { outline };
745 self.paint(&image, points, a[3]);
746 }
747 "FLOOD" => {
748 check(image.contains(a[0], a[1]) && (0..=m).contains(&a[2]), "bad value")?;
749 self.paint(&image, flood_points(&image, a[0], a[1]), a[2]);
750 }
751 "HISTOGRAM" => {
752 let mut lows = [None; 16];
753 let mut highs = [0; 16];
754 let mut counts = [0; 16];
755 for value in 0..=m {
756 let bin = (value * 16 / (m + 1)) as usize;
757 if lows[bin].is_none() { lows[bin] = Some(value); }
758 highs[bin] = value;
759 }
760 for &p in &image.pixels { counts[(p * 16 / (m + 1)) as usize] += 1; }
761 let top = *counts.iter().max().unwrap();
762 for bin in 0..16 {
763 if let Some(low) = lows[bin] {
764 self.emit(format!("{low:>3}-{hi:>3} {count:>4} |{bar}",
765 hi = highs[bin], count = counts[bin],
766 bar = "#".repeat((counts[bin] * 32 / top) as usize)));
767 }
768 }
769 }
770 "EQUALIZE" => self.commit(equalized(&image), None),
771 "LABEL" => {
772 check((0..=m).contains(&a[0]), "bad value")?;
773 let groups = components(&image, a[0]);
774 self.emit(format!("labels {}", groups.len()));
775 for (index, group) in groups.iter().enumerate() {
776 self.emit(format!("label {} {}", index + 1,
777 component_text(&image, a[0], group)));
778 }
779 }
780 "DISTANCE" => {
781 check((0..=m).contains(&a[0]), "bad value")?;
782 self.commit(distance_map(&image, a[0]), None);
783 }
784 "STATS" => self.emit(statistics(&image.pixels, m)),
785 "REGION" => {
786 self.crop_area(&image, a)?;
787 self.emit(statistics(&cropped(&image, a[0], a[1], a[2], a[3]).pixels, m));
788 }
789 "DITHER" => self.commit(dithered(&image), None),
790 "ASCIIART" => {
791 let shades = b"@#*+=~-:,.";
792 for y in 0..h {
793 let row: String = (0..w).map(|x| {
794 shades[(image.at(x, y) * 10 / (m + 1)) as usize] as char
795 }).collect();
796 self.emit(row);
797 }
798 }
799 "STORE" => {
800 self.slots.insert(slot.to_string(), image.clone());
801 self.emit(format!("stored {slot} {}", image.summary()));
802 }
803 "BLEND" => {
804 check(a[0] >= 0 && a[0] <= a[1] && (1..=100).contains(&a[1]),
805 "bad value")?;
806 let saved = self.matching_slot(&image, slot)?;
807 let pixels = image.pixels.iter().enumerate().map(|(i, &p)| {
808 (p * (a[1] - a[0]) + saved.pixels[i] * a[0]) / a[1]
809 }).collect();
810 self.commit(image.with_pixels(pixels), None);
811 }
812 "DIFF" => {
813 let saved = self.matching_slot(&image, slot)?;
814 let pixels = image.pixels.iter().enumerate().map(|(i, &p)| {
815 (p - saved.pixels[i]).abs()
816 }).collect();
817 self.commit(image.with_pixels(pixels), None);
818 }
819 "PASTE" => {
820 check(within_limits(a), "bad value")?;
821 let saved = &self.slots[slot];
822 check(saved.maxval == m, "maxval mismatch")?;
823 let mut pixels = image.pixels.clone();
824 let mut painted = 0;
825 for y in 0..saved.height {
826 for x in 0..saved.width {
827 if image.contains(a[0] + x, a[1] + y) {
828 pixels[((a[1] + y) * w + a[0] + x) as usize] = saved.at(x, y);
829 painted += 1;
830 }
831 }
832 }
833 self.commit(image.with_pixels(pixels), Some(painted));
834 }
835 _ => unreachable!("all commands are dispatched"),
836 }
837 Ok(())
838 }
839}
840
841fn main() {
842 let mut input = String::new();
843 io::stdin().read_to_string(&mut input).unwrap();
844 let lines: Vec<&str> = input.lines().collect();
845 let command_count: usize = lines[0].parse().unwrap();
846 let mut position = 1;
847 let mut editor = Editor::new();
848 for _ in 0..command_count {
849 let words: Vec<&str> = lines[position].split_whitespace().collect();
850 position += 1;
851 if words[0] == "LOAD" {
852 let count: usize = words[1].parse().unwrap();
853 editor.load(&lines[position..position + count]);
854 position += count;
855 } else {
856 editor.execute(&words);
857 }
858 }
859 if !editor.output.is_empty() {
860 println!("{}", editor.output.join("\n"));
861 }
862}
