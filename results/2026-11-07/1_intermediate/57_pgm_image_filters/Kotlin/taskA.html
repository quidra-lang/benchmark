21private class Image(
22 val width: Int, val height: Int, val maxval: Int, val pixels: IntArray
23) {
24 fun get(x: Int, y: Int): Int = pixels[y * width + x]
25 fun contains(x: Int, y: Int): Boolean = x in 0 until width && y in 0 until height
26 fun clamp(value: Int): Int = value.coerceIn(0, maxval)
27 fun withPixels(values: IntArray): Image = Image(width, height, maxval, values)
28 fun map(change: (Int) -> Int): Image = withPixels(pixels.map(change).toIntArray())

30 fun neighborhood(x: Int, y: Int): IntArray {
31 val result = IntArray(9)
32 var index = 0
33 for (dy in -1..1) {
34 for (dx in -1..1) {
35 result[index++] = get(
36 (x + dx).coerceIn(0, width - 1),
37 (y + dy).coerceIn(0, height - 1)
38 )
39 }
40 }
41 return result
42 }
43
44 fun summary(): String {
45 var checksum = 0L
46 for (i in pixels.indices) checksum += (i + 1).toLong() * pixels[i]
47 return width.toString() + "x" + height + " maxval " + maxval +
48 " checksum " + checksum
49 }

52private fun loadPgm(lines: List<String>): Image {
53 val words = lines.flatMap { line ->
54 line.substringBefore('#').trim().split(Regex(" +")).filter { it.isNotEmpty() }
55 }
56 check(words.isNotEmpty() && words[0] == "P2", "bad magic")
57
58 fun header(index: Int, limit: Int, reason: String): Int {
59 check(index < words.size, "truncated header")
60 check(words[index].all { it in '0'..'9' }, "bad number")
61 val number = words[index].toInt()
62 check(number in 1..limit, reason)
63 return number
64 }
65
66 val width = header(1, 64, "bad size")
67 val height = header(2, 64, "bad size")
68 val maxval = header(3, 255, "bad maxval")
69 val count = words.size - 4
70 val pixels = IntArray(width * height)
71 for (index in pixels.indices) {
72 val x = index % width
73 val y = index / width
74 check(index < count, "too few pixels (" + count + " of " + pixels.size + ")")
75 val word = words[index + 4]
76 check(word.all { it in '0'..'9' }, "bad pixel at " + x + "," + y)
77 val value = word.toInt()
78 check(value <= maxval, "pixel out of range at " + x + "," + y)
79 pixels[index] = value
80 }
81 check(count <= pixels.size, "too many pixels (" + count + " of " + pixels.size + ")")
82 return Image(width, height, maxval, pixels)
83}

101private fun convolution(image: Image, weights: IntArray, denominator: Int) =
102 transformed(image) { x, y ->
103 val q = image.neighborhood(x, y)
104 var total = 0
105 for (i in 0 until 9) total += q[i] * weights[i]
106 image.clamp(Math.floorDiv(total, denominator))
107 }

112private fun sobel(image: Image) = transformed(image) { x, y ->
113 val q = image.neighborhood(x, y)
114 val gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6]
115 val gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2]
116 image.clamp(kotlin.math.abs(gx) + kotlin.math.abs(gy))
117}

119private fun linePoints(x0: Int, y0: Int, x1: Int, y1: Int): List<Point> {
120 val dx = kotlin.math.abs(x1 - x0)
121 val dy = -kotlin.math.abs(y1 - y0)
122 val sx = if (x0 < x1) 1 else -1
123 val sy = if (y0 < y1) 1 else -1
124 var error = dx + dy
125 var x = x0
126 var y = y0
127 val points = mutableListOf<Point>()
128 while (true) {
129 points.add(Point(x, y))
130 if (x == x1 && y == y1) break
131 val twice = 2 * error
132 if (twice >= dy) { error += dy; x += sx }
133 if (twice <= dx) { error += dx; y += sy }
134 }
135 return points
136}

138private fun rectanglePoints(
139 image: Image, left: Int, top: Int, width: Int, height: Int, fill: Boolean
140): List<Point> {
141 val right = left + width - 1
142 val bottom = top + height - 1
143 val points = mutableListOf<Point>()
144 for (y in maxOf(0, top)..minOf(bottom, image.height - 1)) {
145 for (x in maxOf(0, left)..minOf(right, image.width - 1)) {
146 if (fill || x == left || x == right || y == top || y == bottom)
147 points.add(Point(x, y))
148 }
149 }
150 return points

153private fun circlePoints(cx: Int, cy: Int, radius: Int): List<Point> {
154 var a = radius
155 var b = 0
156 var error = 1 - radius
157 val points = mutableListOf<Point>()
158 while (a >= b) {
159 points.addAll(listOf(
160 Point(cx + a, cy + b), Point(cx + b, cy + a),
161 Point(cx - b, cy + a), Point(cx - a, cy + b),
162 Point(cx - a, cy - b), Point(cx - b, cy - a),
163 Point(cx + b, cy - a), Point(cx + a, cy - b)
164 ))
165 b++
166 if (error < 0) error += 2 * b + 1
167 else { a--; error += 2 * (b - a) + 1 }
168 }
169 return points
170}

172private fun discPoints(image: Image, outline: List<Point>): List<Point> {
173 val rows = mutableMapOf<Int, Pair<Int, Int>>()
174 for ((x, y) in outline) {
175 val previous = rows[y]
176 rows[y] = if (previous == null) x to x
177 else minOf(previous.first, x) to maxOf(previous.second, x)
178 }
179 return rows.flatMap { (y, bounds) ->
180 if (y !in 0 until image.height) emptyList()
181 else (maxOf(0, bounds.first)..minOf(bounds.second, image.width - 1))
182 .map { x -> Point(x, y) }
183 }
184}

186private fun floodPoints(image: Image, x: Int, y: Int): List<Point> {
187 val target = image.get(x, y)
188 val stack = ArrayDeque<Point>()
189 val visited = mutableSetOf(y * image.width + x)
190 val points = mutableListOf<Point>()
191 stack.addLast(Point(x, y))
192 while (stack.isNotEmpty()) {
193 val position = stack.removeLast()
194 points.add(position)
195 for (direction in directions4) {
196 val nx = position.x + direction.x
197 val ny = position.y + direction.y
198 if (!image.contains(nx, ny) || image.get(nx, ny) != target) continue
199 if (visited.add(ny * image.width + nx)) stack.addLast(Point(nx, ny))
200 }
201 }
202 return points
203}

205private fun equalized(image: Image): Image {
206 val counts = IntArray(image.maxval + 1)
207 for (pixel in image.pixels) counts[pixel]++
208 var accumulated = 0
209 val cumulative = counts.map { accumulated += it; accumulated }
210 val size = image.pixels.size
211 val first = cumulative[image.pixels.min()]
212 if (first == size) return image.withPixels(image.pixels.copyOf())
213 val span = size - first
214 return image.map { pixel ->
215 (2 * (cumulative[pixel] - first) * image.maxval + span) / (2 * span)
216 }
217}

219private fun components(image: Image, threshold: Int): List<List<Point>> {
220 val visited = BooleanArray(image.pixels.size)
221 val groups = mutableListOf<List<Point>>()
222 for (start in image.pixels.indices) {
223 if (visited[start] || image.pixels[start] < threshold) continue
224 visited[start] = true
225 val stack = ArrayDeque<Point>()
226 stack.addLast(Point(start % image.width, start / image.width))
227 val group = mutableListOf<Point>()
228 while (stack.isNotEmpty()) {
229 val position = stack.removeLast()
230 group.add(position)
231 for (direction in directions8) {
232 val nx = position.x + direction.x
233 val ny = position.y + direction.y
234 if (!image.contains(nx, ny)) continue
235 val index = ny * image.width + nx
236 if (!visited[index] && image.pixels[index] >= threshold) {
237 visited[index] = true
238 stack.addLast(Point(nx, ny))
239 }
240 }
241 }
242 groups.add(group)
243 }
244 return groups
245}

247private fun componentText(image: Image, threshold: Int, group: List<Point>): String {
248 val x = group.map { it.x }
249 val y = group.map { it.y }
250 val perimeter = group.count { p ->
251 directions4.any { d ->
252 !image.contains(p.x + d.x, p.y + d.y) ||
253 image.get(p.x + d.x, p.y + d.y) < threshold
254 }
255 }
256 return "area " + group.size + " perimeter " + perimeter +
257 " box " + x.min() + " " + y.min() + " " + x.max() + " " + y.max() +
258 " center " + (x.sum() / group.size) + " " + (y.sum() / group.size)
259}

261private fun distanceMap(image: Image, threshold: Int): Image {
262 if (image.pixels.all { it >= threshold })
263 return image.withPixels(IntArray(image.pixels.size) { image.maxval })
264 val w = image.width
265 val h = image.height
266 val distances = IntArray(image.pixels.size) {
267 if (image.pixels[it] < threshold) 0 else w + h
268 }
269 for (y in 0 until h) {
270 for (x in 0 until w) {
271 val index = y * w + x
272 if (x > 0) distances[index] = minOf(distances[index], distances[index - 1] + 1)
273 if (y > 0) distances[index] = minOf(distances[index], distances[index - w] + 1)
274 }
275 }
276 for (y in h - 1 downTo 0) {
277 for (x in w - 1 downTo 0) {
278 val index = y * w + x
279 if (x < w - 1) distances[index] = minOf(distances[index], distances[index + 1] + 1)
280 if (y < h - 1) distances[index] = minOf(distances[index], distances[index + w] + 1)
281 }
282 }
283 return image.withPixels(distances.map { minOf(it, image.maxval) }.toIntArray())
284}

286private fun statistics(pixels: IntArray, maxval: Int): String {
287 val sorted = pixels.sorted()
288 val counts = IntArray(maxval + 1)
289 for (pixel in pixels) counts[pixel]++
290 var mode = 0
291 for (value in 1..maxval) if (counts[value] > counts[mode]) mode = value
292 return "min " + sorted.first() + " max " + sorted.last() +
293 " mean " + (pixels.sum() / pixels.size) +
294 " median " + sorted[(pixels.size - 1) / 2] + " mode " + mode
295}

297private fun dithered(image: Image): Image {
298 val errors = image.pixels.copyOf()
299 val pixels = IntArray(errors.size)
300 val diffusion = listOf(Point(1, 0) to 7, Point(-1, 1) to 3,
301 Point(0, 1) to 5, Point(1, 1) to 1)
302 for (y in 0 until image.height) {
303 for (x in 0 until image.width) {
304 val index = y * image.width + x
305 val old = errors[index]
306 val value = if (2 * old >= image.maxval) image.maxval else 0
307 pixels[index] = value
308 val error = old - value
309 for ((delta, weight) in diffusion) {
310 val nx = x + delta.x
311 val ny = y + delta.y
312 if (image.contains(nx, ny))
313 errors[ny * image.width + nx] += Math.floorDiv(error * weight, 16)
314 }
315 }
316 }
317 return image.withPixels(pixels)
318}

322 "NEW" to "iiii", "OUTPUT" to "", "INVERT" to "", "BRIGHTNESS" to "i",
323 "CONTRAST" to "ii", "THRESHOLD" to "i", "LEVELS" to "ii",
324 "POSTERIZE" to "i", "MAXVAL" to "i", "FLIPH" to "", "FLIPV" to "",
325 "ROTATE" to "i", "CROP" to "iiii", "PAD" to "ii", "SCALE" to "ii",
326 "SHIFT" to "iii", "TILE" to "ii", "BLUR" to "", "GAUSS" to "",
327 "SHARPEN" to "", "CONVOLVE" to "iiiiiiiiii", "SOBEL" to "",
328 "MEDIAN" to "", "ERODE" to "", "DILATE" to "", "OPEN" to "",
329 "CLOSE" to "", "GRADIENT" to "", "LINE" to "iiiii",
330 "RECT" to "iiiii", "CIRCLE" to "iiii", "FLOOD" to "iii",

338 private val needsSlot = setOf("RECALL", "DROP", "BLEND", "DIFF", "PASTE")
339 private val kernels = mapOf(
340 "BLUR" to (intArrayOf(1, 1, 1, 1, 1, 1, 1, 1, 1) to 9),
341 "GAUSS" to (intArrayOf(1, 2, 1, 2, 4, 2, 1, 2, 1) to 16),
342 "SHARPEN" to (intArrayOf(0, -1, 0, -1, 5, -1, 0, -1, 0) to 1)

352 private fun parse(words: List<String>): Command {
353 val name = words.first()
354 val pattern = patterns[name] ?: throw CommandError("unknown command " + name)
355 var given = words.drop(1)
356 val fill = name in setOf("RECT", "CIRCLE") &&
357 given.size == pattern.length + 1 && given.last() == "FILL"
358 if (fill) given = given.dropLast(1)
359 check(given.size == pattern.length, "bad arguments")
360 val numbers = mutableListOf<Int>()
361 var slot = ""
362 for ((i, word) in given.withIndex()) {
363 if (pattern[i] == 'i') {
364 check(Regex("""(?:0|-?[1-9][0-9]{0,6})""").matches(word), "bad arguments")
365 numbers.add(word.toInt())
366 } else {
367 check(Regex("""[a-z]{1,8}""").matches(word), "bad arguments")
368 slot = word
369 }
370 }
371 return Command(name, numbers, slot, fill)
372 }

375 try {
376 val cmd = parse(words)
377 if (cmd.name !in imageless) check(current != null, "no image")
378 if (cmd.name in needsSlot) check(slots.containsKey(cmd.slot), "no slot " + cmd.slot)
379 run(cmd)
380 } catch (error: CommandError) {
381 emit("error: " + error.message)
382 }

429 private fun run(cmd: Command) {
430 val a = cmd.args
431 val name = cmd.name
432 val slot = cmd.slot
433 val fill = cmd.fill
434 when (name) {
435 "NEW" -> {
436 val (w, h, maxval, value) = a
437 check(w in 1..64 && h in 1..64)
438 check(maxval in 1..255 && value in 0..maxval)
439 replace(Image(w, h, maxval, IntArray(w * h) { value }), "created")
440 return
441 }
442 "RECALL" -> { replace(slots.getValue(slot), "recalled " + slot); return }
443 "DROP" -> { slots.remove(slot); emit("dropped " + slot); return }
444 "SLOTS" -> {
445 if (slots.isEmpty()) emit("no slots")
446 for (key in slots.keys.sorted()) emit(key + " " + slots.getValue(key).summary())
447 return
448 }
449 "INFO" -> {
450 emit("image " + (current?.summary() ?: "none") +
451 " undo " + undo.size + " redo " + redo.size + " slots " + slots.size)
452 return
453 }
454 "UNDO" -> {
455 check(undo.isNotEmpty(), "nothing to undo")
456 redo.add(current!!)
457 current = undo.removeAt(undo.lastIndex)
458 emit("undo " + current!!.summary())
459 return
460 }
461 "REDO" -> {
462 check(redo.isNotEmpty(), "nothing to redo")
463 undo.add(current!!)
464 current = redo.removeAt(redo.lastIndex)
465 emit("redo " + current!!.summary())
466 return
467 }
468 }
469
470 val image = current!!
471 val m = image.maxval
472 when (name) {
473 "OUTPUT" -> {
474 emit("P2")
475 emit(image.width.toString() + " " + image.height)
476 emit(m.toString())
477 for (y in 0 until image.height)
478 emit((0 until image.width).joinToString(" ") { x -> image.get(x, y).toString() })
479 }
480 "INVERT" -> commit(image.map { m - it })
481 "BRIGHTNESS" -> {
482 check(a[0] in -255..255)
483 commit(image.map { image.clamp(it + a[0]) })
484 }
485 "CONTRAST" -> {
486 val (num, den) = a
487 check(num in 0..100 && den in 1..100)
488 val center = (m + 1) / 2
489 commit(image.map { image.clamp(center + Math.floorDiv((it - center) * num, den)) })
490 }
491 "THRESHOLD" -> {
492 check(a[0] in 0..m)
493 commit(image.map { if (it >= a[0]) m else 0 })
494 }
495 "LEVELS" -> {
496 val (low, high) = a
497 check(low >= 0 && low < high && high <= m)
498 commit(image.map { image.clamp(Math.floorDiv((it - low) * m, high - low)) })
499 }
500 "POSTERIZE" -> {
501 val levels = a[0]
502 check(levels in 2..(m + 1))
503 commit(image.map { ((it * levels) / (m + 1)) * m / (levels - 1) })
504 }
505 "MAXVAL" -> {
506 val maximum = a[0]
507 check(maximum in 1..255)
508 commit(Image(image.width, image.height, maximum,
509 image.pixels.map { (2 * it * maximum + m) / (2 * m) }.toIntArray()))
510 }
511 "FLIPH" -> commit(transformed(image) { x, y -> image.get(image.width - 1 - x, y) })
512 "FLIPV" -> commit(transformed(image) { x, y -> image.get(x, image.height - 1 - y) })
513 "ROTATE" -> {
514 val angle = a[0]
515 check(angle in listOf(90, 180, 270))
516 val w = image.width
517 val h = image.height
518 commit(when (angle) {
519 90 -> transformed(image, h, w) { x, y -> image.get(y, h - 1 - x) }
520 180 -> transformed(image, w, h) { x, y -> image.get(w - 1 - x, h - 1 - y) }
521 else -> transformed(image, h, w) { x, y -> image.get(w - 1 - y, x) }
522 })
523 }
524 "CROP" -> {
525 checkedCrop(image, a)
526 commit(cropped(image, a[0], a[1], a[2], a[3]))
527 }
528 "PAD" -> {
529 val (n, fillValue) = a
530 check(n >= 1 && fillValue in 0..m)
531 val w = image.width + 2 * n
532 val h = image.height + 2 * n
533 check(w <= 64 && h <= 64, "too large")
534 commit(translated(image, w, h, n, n, fillValue))
535 }
536 "SCALE" -> {
537 val (num, den) = a
538 check(num in 1..16 && den in 1..16)
539 val w = image.width * num / den
540 val h = image.height * num / den
541 check(w >= 1 && h >= 1, "too small")
542 check(w <= 64 && h <= 64, "too large")
543 commit(transformed(image, w, h) { x, y ->
544 image.get((2 * x + 1) * den / (2 * num),
545 (2 * y + 1) * den / (2 * num))
546 })
547 }
548 "SHIFT" -> {
549 val (dx, dy, fillValue) = a
550 check(withinLimit(dx, dy) && fillValue in 0..m)
551 commit(translated(image, image.width, image.height, dx, dy, fillValue))
552 }
553 "TILE" -> {
554 val (nx, ny) = a
555 check(nx >= 1 && ny >= 1)
556 val w = image.width * nx
557 val h = image.height * ny
558 check(w <= 64 && h <= 64, "too large")
559 commit(transformed(image, w, h) { x, y ->
560 image.get(x % image.width, y % image.height)
561 })
562 }
563 "BLUR", "GAUSS", "SHARPEN" -> {
564 val (weights, den) = kernels.getValue(name)
565 commit(convolution(image, weights, den))
566 }
567 "CONVOLVE" -> {
568 check(a.take(9).all { it in -16..16 } && a[9] in 1..256)
569 commit(convolution(image, a.take(9).toIntArray(), a[9]))
570 }
571 "SOBEL" -> commit(sobel(image))
572 "MEDIAN" -> commit(rankFiltered(image, 4))
573 "ERODE" -> commit(rankFiltered(image, 0))
574 "DILATE" -> commit(rankFiltered(image, 8))
575 "OPEN" -> commit(rankFiltered(rankFiltered(image, 0), 8))
576 "CLOSE" -> commit(rankFiltered(rankFiltered(image, 8), 0))
577 "GRADIENT" -> {
578 val high = rankFiltered(image, 8).pixels
579 val low = rankFiltered(image, 0).pixels
580 commit(image.withPixels(IntArray(high.size) { high[it] - low[it] }))
581 }
582 "LINE" -> {
583 check(withinLimit(*a.take(4).toIntArray()) && a[4] in 0..m)
584 paint(image, linePoints(a[0], a[1], a[2], a[3]), a[4])
585 }
586 "RECT" -> {
587 val (x, y, w, h, value) = a
588 check(withinLimit(x, y) && w in 1..1000 && h in 1..1000 && value in 0..m)
589 paint(image, rectanglePoints(image, x, y, w, h, fill), value)
590 }
591 "CIRCLE" -> {
592 val (x, y, radius, value) = a
593 check(withinLimit(x, y) && radius in 0..1000 && value in 0..m)
594 val outline = circlePoints(x, y, radius)
595 paint(image, if (fill) discPoints(image, outline) else outline, value)
596 }
597 "FLOOD" -> {
598 val (x, y, value) = a
599 check(image.contains(x, y) && value in 0..m)
600 paint(image, floodPoints(image, x, y), value)
601 }
602 "HISTOGRAM" -> {
603 val low = IntArray(16) { -1 }
604 val high = IntArray(16)
605 val counts = IntArray(16)
606 for (value in 0..m) {
607 val index = value * 16 / (m + 1)
608 if (low[index] < 0) low[index] = value
609 high[index] = value
610 }
611 for (value in image.pixels) counts[value * 16 / (m + 1)]++
612 val top = counts.max()
613 for (index in low.indices) {
614 if (low[index] < 0) continue
615 emit(low[index].toString().padStart(3) + "-" +
616 high[index].toString().padStart(3) + " " +
617 counts[index].toString().padStart(4) + " |" +
618 "#".repeat(counts[index] * 32 / top))
619 }
620 }
621 "EQUALIZE" -> commit(equalized(image))
622 "LABEL" -> {
623 check(a[0] in 0..m)
624 val groups = components(image, a[0])
625 emit("labels " + groups.size)
626 for ((index, group) in groups.withIndex())
627 emit("label " + (index + 1) + " " + componentText(image, a[0], group))
628 }
629 "DISTANCE" -> {
630 check(a[0] in 0..m)
631 commit(distanceMap(image, a[0]))
632 }
633 "STATS" -> emit(statistics(image.pixels, m))
634 "REGION" -> {
635 checkedCrop(image, a)
636 emit(statistics(cropped(image, a[0], a[1], a[2], a[3]).pixels, m))
637 }
638 "DITHER" -> commit(dithered(image))
639 "ASCIIART" -> {
640 for (y in 0 until image.height)
641 emit((0 until image.width).map { x ->
642 shades[image.get(x, y) * 10 / (m + 1)]
643 }.joinToString(""))
644 }
645 "STORE" -> {
646 slots[slot] = image
647 emit("stored " + slot + " " + image.summary())
648 }
649 "BLEND" -> {
650 val (num, den) = a
651 check(num >= 0 && num <= den && den in 1..100)
652 val other = matchingSlot(image, slot)
653 commit(image.withPixels(IntArray(image.pixels.size) { i ->
654 (image.pixels[i] * (den - num) + other.pixels[i] * num) / den
655 }))
656 }
657 "DIFF" -> {
658 val other = matchingSlot(image, slot)
659 commit(image.withPixels(IntArray(image.pixels.size) { i ->
660 kotlin.math.abs(image.pixels[i] - other.pixels[i])
661 }))
662 }
663 "PASTE" -> {
664 val (x, y) = a
665 check(withinLimit(x, y))
666 val other = slots.getValue(slot)
667 check(other.maxval == m, "maxval mismatch")
668 val pixels = image.pixels.copyOf()
669 var painted = 0
670 for (j in 0 until other.height) for (i in 0 until other.width) {
671 if (image.contains(x + i, y + j)) {
672 pixels[(y + j) * image.width + x + i] = other.get(i, j)
673 painted++
674 }
675 }
676 commit(image.withPixels(pixels), painted)
677 }
678 }
679 }

681 private fun withinLimit(vararg values: Int): Boolean = values.all { it in -1000..1000 }
682}
