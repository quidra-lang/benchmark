20 public Image(int width, int height, int maxval, int[] pixels) {

22 Height = height;
23 Maxval = maxval;
24 Pixels = pixels;

26
27 public int Get(int x, int y) => Pixels[y * Width + x];
28 public bool Contains(int x, int y) => x >= 0 && y >= 0 && x < Width && y < Height;
29 public int Clamp(int value) => Math.Clamp(value, 0, Maxval);
30 public Image WithPixels(int[] pixels) => new(Width, Height, Maxval, pixels);
31
32 public Image Map(Func<int, int> change) => WithPixels(Pixels.Select(change).ToArray());

34 public int[] Neighborhood(int x, int y) {
35 int[] values = new int[9];
36 int index = 0;
37 for (int dy = -1; dy <= 1; ++dy) {
38 for (int dx = -1; dx <= 1; ++dx) {
39 values[index++] =
40 Get(Math.Clamp(x + dx, 0, Width - 1), Math.Clamp(y + dy, 0, Height - 1));
41 }
42 }
43 return values;
44 }
45
46 public string Summary() {
47 long checksum = 0;
48 for (int i = 0; i < Pixels.Length; ++i)
49 checksum += (long)(i + 1) * Pixels[i];
50 return $"{Width}x{Height} maxval {Maxval} checksum {checksum}";
51 }

61 int quotient = dividend / divisor;
62 return dividend % divisor < 0 ? quotient - 1 : quotient;

65 public static Image Load(IReadOnlyList<string> lines) {
66 List<string> words = new();
67 foreach (string line in lines) {
68 string uncommented = line.Split('#')[0];
69 words.AddRange(uncommented.Split(' ', StringSplitOptions.RemoveEmptyEntries));
70 }
71 Check(words.Count > 0 && words[0] == "P2", "bad magic");
72
73 int Header(int index, int limit, string reason) {
74 Check(index < words.Count, "truncated header");
75 Check(Regex.IsMatch(words[index], @"\A[0-9]+\z"), "bad number");
76 int value = int.Parse(words[index]);
77 Check(value >= 1 && value <= limit, reason);
78 return value;
79 }
80
81 int width = Header(1, 64, "bad size");
82 int height = Header(2, 64, "bad size");
83 int maxval = Header(3, 255, "bad maxval");
84 int[] pixels = new int[width * height];
85 int count = words.Count - 4;
86 for (int i = 0; i < pixels.Length; ++i) {
87 int x = i % width;
88 int y = i / width;
89 Check(i < count, $"too few pixels ({count} of {pixels.Length})");
90 string word = words[i + 4];
91 Check(Regex.IsMatch(word, @"\A[0-9]+\z"), $"bad pixel at {x},{y}");
92 int value = int.Parse(word);
93 Check(value <= maxval, $"pixel out of range at {x},{y}");
94 pixels[i] = value;
95 }
96 Check(count <= pixels.Length, $"too many pixels ({count} of {pixels.Length})");
97 return new Image(width, height, maxval, pixels);
98 }

100 public static Image Transform(Image source, int width, int height, Func<int, int, int> pixel) {
101 int[] values = new int[width * height];
102 for (int y = 0; y < height; ++y)
103 for (int x = 0; x < width; ++x)
104 values[y * width + x] = pixel(x, y);
105 return new Image(width, height, source.Maxval, values);
106 }

116 public static Image Convolve(Image image, int[] weights, int divisor) =>
117 Transform(image, image.Width, image.Height, (x, y) => {
118 int[] q = image.Neighborhood(x, y);
119 int value = 0;
120 for (int i = 0; i < 9; ++i)
121 value += q[i] * weights[i];
122 return image.Clamp(FloorDivide(value, divisor));
123 });

125 public static Image Rank(Image image,
126 int index) => Transform(image, image.Width, image.Height, (x, y) => {
127 int[] q = image.Neighborhood(x, y);
128 Array.Sort(q);
129 return q[index];
130 });

132 public static Image
133 Sobel(Image image) => Transform(image, image.Width, image.Height, (x, y) => {
134 int[] q = image.Neighborhood(x, y);
135 int gx = q[2] + 2 * q[5] + q[8] - q[0] - 2 * q[3] - q[6];
136 int gy = q[6] + 2 * q[7] + q[8] - q[0] - 2 * q[1] - q[2];
137 return image.Clamp(Math.Abs(gx) + Math.Abs(gy));
138 });

140 public static List<Point> Line(int x0, int y0, int x1, int y1) {
141 int dx = Math.Abs(x1 - x0);
142 int dy = -Math.Abs(y1 - y0);
143 int sx = x0 < x1 ? 1 : -1;
144 int sy = y0 < y1 ? 1 : -1;
145 int error = dx + dy;
146 int x = x0, y = y0;
147 List<Point> points = new();
148 while (true) {
149 points.Add(new Point(x, y));
150 if (x == x1 && y == y1)
151 break;
152 int twice = 2 * error;
153 if (twice >= dy) {
154 error += dy;
155 x += sx;
156 }
157 if (twice <= dx) {
158 error += dx;
159 y += sy;
160 }
161 }
162 return points;
163 }

165 public static List<Point> Rectangle(Image image, int left, int top, int width, int height,
166 bool fill) {
167 int right = left + width - 1;
168 int bottom = top + height - 1;
169 List<Point> points = new();
170 for (int y = Math.Max(top, 0); y <= Math.Min(bottom, image.Height - 1); ++y) {
171 for (int x = Math.Max(left, 0); x <= Math.Min(right, image.Width - 1); ++x) {
172 if (fill || x == left || x == right || y == top || y == bottom)
173 points.Add(new Point(x, y));
174 }
175 }
176 return points;
177 }

179 public static List<Point> Circle(int cx, int cy, int radius) {
180 int a = radius, b = 0, error = 1 - radius;
181 List<Point> points = new();
182 while (a >= b) {
183 points.AddRange(new[] { new Point(cx + a, cy + b), new Point(cx + b, cy + a),
184 new Point(cx - b, cy + a), new Point(cx - a, cy + b),
185 new Point(cx - a, cy - b), new Point(cx - b, cy - a),
186 new Point(cx + b, cy - a), new Point(cx + a, cy - b) });
187 ++b;
188 if (error < 0)
189 error += 2 * b + 1;
190 else {
191 --a;
192 error += 2 * (b - a) + 1;
193 }
194 }
195 return points;
196 }

198 public static List<Point> Disc(Image image, IEnumerable<Point> outline) {
199 Dictionary<int, (int Left, int Right)> rows = new();
200 foreach (Point point in outline) {
201 if (rows.TryGetValue(point.Y, out var limits))
202 rows[point.Y] = (Math.Min(limits.Left, point.X), Math.Max(limits.Right, point.X));
203 else
204 rows[point.Y] = (point.X, point.X);
205 }
206
207 List<Point> points = new();
208 foreach (var row in rows) {
209 int y = row.Key;
210 if (y < 0 || y >= image.Height)
211 continue;
212 for (int x = Math.Max(row.Value.Left, 0);
213 x <= Math.Min(row.Value.Right, image.Width - 1); ++x)
214 points.Add(new Point(x, y));
215 }
216 return points;
217 }

222 new(1, 0), new(-1, 1), new(0, 1), new(1, 1) };

224 public static List<Point> Flood(Image image, int x, int y) {
225 int target = image.Get(x, y);
226 Stack<Point> stack = new();
227 HashSet<int> seen = new();
228 stack.Push(new Point(x, y));
229 seen.Add(y * image.Width + x);
230 List<Point> points = new();
231 while (stack.Count > 0) {
232 Point p = stack.Pop();
233 points.Add(p);
234 foreach (Point delta in Neighbors4) {
235 int nx = p.X + delta.X, ny = p.Y + delta.Y;
236 if (!image.Contains(nx, ny) || image.Get(nx, ny) != target)
237 continue;
238 if (seen.Add(ny * image.Width + nx))
239 stack.Push(new Point(nx, ny));
240 }
241 }
242 return points;
243 }

245 public static Image Equalize(Image image) {
246 int[] counts = new int[image.Maxval + 1];
247 foreach (int pixel in image.Pixels)
248 ++counts[pixel];
249 int[] cdf = new int[counts.Length];
250 int running = 0;
251 for (int i = 0; i < counts.Length; ++i) {
252 running += counts[i];
253 cdf[i] = running;
254 }
255 int n = image.Pixels.Length;
256 int first = cdf[image.Pixels.Min()];
257 if (first == n)
258 return image.WithPixels((int[])image.Pixels.Clone());
259 int span = n - first;
260 return image.Map(p => (2 * (cdf[p] - first) * image.Maxval + span) / (2 * span));
261 }

263 public static List<List<Point>> Components(Image image, int threshold) {
264 HashSet<int> visited = new();
265 List<List<Point>> groups = new();
266 for (int start = 0; start < image.Pixels.Length; ++start) {
267 if (visited.Contains(start) || image.Pixels[start] < threshold)
268 continue;
269 visited.Add(start);
270 Stack<Point> stack = new();
271 stack.Push(new Point(start % image.Width, start / image.Width));
272 List<Point> group = new();
273 while (stack.Count > 0) {
274 Point p = stack.Pop();
275 group.Add(p);
276 foreach (Point delta in Neighbors8) {
277 int nx = p.X + delta.X, ny = p.Y + delta.Y;
278 if (!image.Contains(nx, ny))
279 continue;
280 int index = ny * image.Width + nx;
281 if (image.Pixels[index] >= threshold && visited.Add(index))
282 stack.Push(new Point(nx, ny));
283 }
284 }
285 groups.Add(group);
286 }
287 return groups;
288 }

290 public static string ComponentDescription(Image image, int threshold, List<Point> group) {
291 int left = group.Min(p => p.X), top = group.Min(p => p.Y);
292 int right = group.Max(p => p.X), bottom = group.Max(p => p.Y);
293 int perimeter = 0;
294 foreach (Point point in group) {
295 if (Neighbors4.Any(delta =>
296 !image.Contains(point.X + delta.X, point.Y + delta.Y) ||
297 image.Get(point.X + delta.X, point.Y + delta.Y) < threshold))
298 ++perimeter;
299 }
300 int count = group.Count;
301 return $"area {count} perimeter {perimeter} box {left} {top} {right} {bottom} " +
302 $"center {group.Sum(p => p.X) / count} {group.Sum(p => p.Y) / count}";
303 }

305 public static Image Distance(Image image, int threshold) {
306 if (image.Pixels.All(pixel => pixel >= threshold))
307 return image.WithPixels(Enumerable.Repeat(image.Maxval, image.Pixels.Length).ToArray());
308 int width = image.Width, height = image.Height, far = width + height;
309 int[] distances = image.Pixels.Select(p => p < threshold ? 0 : far).ToArray();
310 for (int y = 0; y < height; ++y) {
311 for (int x = 0; x < width; ++x) {
312 int index = y * width + x;
313 if (x > 0)
314 distances[index] = Math.Min(distances[index], distances[index - 1] + 1);
315 if (y > 0)
316 distances[index] = Math.Min(distances[index], distances[index - width] + 1);
317 }
318 }
319 for (int y = height - 1; y >= 0; --y) {
320 for (int x = width - 1; x >= 0; --x) {
321 int index = y * width + x;
322 if (x < width - 1)
323 distances[index] = Math.Min(distances[index], distances[index + 1] + 1);
324 if (y < height - 1)
325 distances[index] = Math.Min(distances[index], distances[index + width] + 1);
326 }
327 }
328 return image.WithPixels(distances.Select(v => Math.Min(v, image.Maxval)).ToArray());
329 }

331 public static string Statistics(int[] pixels, int maxval) {
332 int[] sorted = pixels.OrderBy(v => v).ToArray();
333 int[] counts = new int[maxval + 1];
334 foreach (int pixel in pixels)
335 ++counts[pixel];
336 int mode = 0;
337 for (int i = 1; i <= maxval; ++i)
338 if (counts[i] > counts[mode])
339 mode = i;
340 return $"min {sorted[0]} max {sorted[^1]} mean {pixels.Sum() / pixels.Length} " +
341 $"median {sorted[(pixels.Length - 1) / 2]} mode {mode}";
342 }

344 public static Image Dither(Image image) {
345 int width = image.Width, height = image.Height;
346 int[] errors = (int[])image.Pixels.Clone();
347 int[] pixels = new int[errors.Length];
348 var diffusion =
349 new(int Dx, int Dy, int Weight)[] { (1, 0, 7), (-1, 1, 3), (0, 1, 5), (1, 1, 1) };
350 for (int y = 0; y < height; ++y) {
351 for (int x = 0; x < width; ++x) {
352 int index = y * width + x;
353 int old = errors[index];
354 int value = 2 * old >= image.Maxval ? image.Maxval : 0;
355 pixels[index] = value;
356 int error = old - value;
357 foreach (var (dx, dy, weight) in diffusion) {
358 if (image.Contains(x + dx, y + dy))
359 errors[(y + dy) * width + x + dx] += FloorDivide(error * weight, 16);
360 }
361 }
362 }
363 return image.WithPixels(pixels);
364 }

406 private static Command Parse(string[] words) {
407 string name = words[0];
408 ImageOperations.Check(Patterns.TryGetValue(name, out string? pattern), $"unknown command {name}");
409 string[] given = words.Skip(1).ToArray();
410 bool fill = (name == "RECT" || name == "CIRCLE") && given.Length == pattern!.Length + 1 &&
411 given[^1] == "FILL";
412 if (fill)
413 given = given[..^ 1];
414 ImageOperations.Check(given.Length == pattern!.Length, "bad arguments");
415 List<int> numbers = new();
416 string slot = "";
417 for (int i = 0; i < given.Length; ++i) {
418 if (pattern[i] == 'i') {
419 ImageOperations.Check(Regex.IsMatch(given[i], @"\A(?:0|-?[1-9][0-9]{0,6})\z"),
420 "bad arguments");
421 numbers.Add(int.Parse(given[i]));
422 } else {
423 ImageOperations.Check(Regex.IsMatch(given[i], @"\A[a-z]{1,8}\z"), "bad arguments");
424 slot = given[i];
425 }
426 }
427 return new Command(name, numbers.ToArray(), slot, fill);
428 }

495 private void Run(Command command) {
496 string name = command.Name, slot = command.Slot;
497 int[] a = command.Numbers;
498 bool fill = command.Fill;
499 Image image = current!;
500 int width = image?.Width ?? 0, height = image?.Height ?? 0, m = image?.Maxval ?? 0;
501
502 switch (name) {
503 case "NEW": {
504 var (w, h, maxval, v) = (a[0], a[1], a[2], a[3]);
505 ImageOperations.Check(w >= 1 && w <= 64 && h >= 1 && h <= 64);
506 ImageOperations.Check(maxval >= 1 && maxval <= 255 && v >= 0 && v <= maxval);
507 Replace(new Image(w, h, maxval, Enumerable.Repeat(v, w * h).ToArray()), "created");
508 break;
509 }
510 case "OUTPUT":
511 Emit("P2");
512 Emit($"{width} {height}");
513 Emit(m.ToString());
514 for (int y = 0; y < height; ++y)
515 Emit(string.Join(" ", Enumerable.Range(0, width).Select(x => image.Get(x, y))));
516 break;
517 case "INVERT":
518 Commit(image.Map(p => m - p));
519 break;
520 case "BRIGHTNESS":
521 ImageOperations.Check(a[0] >= -255 && a[0] <= 255);
522 Commit(image.Map(p => image.Clamp(p + a[0])));
523 break;
524 case "CONTRAST": {
525 var (num, den) = (a[0], a[1]);
526 ImageOperations.Check(num >= 0 && num <= 100 && den >= 1 && den <= 100);
527 int center = (m + 1) / 2;
528 Commit(image.Map(
529 p => image.Clamp(center + ImageOperations.FloorDivide((p - center) * num, den))));
530 break;
531 }
532 case "THRESHOLD":
533 ImageOperations.Check(a[0] >= 0 && a[0] <= m);
534 Commit(image.Map(p => p >= a[0] ? m : 0));
535 break;
536 case "LEVELS": {
537 var (low, high) = (a[0], a[1]);
538 ImageOperations.Check(low >= 0 && low < high && high <= m);
539 Commit(image.Map(
540 p => image.Clamp(ImageOperations.FloorDivide((p - low) * m, high - low))));
541 break;
542 }
543 case "POSTERIZE": {
544 int levels = a[0];
545 ImageOperations.Check(levels >= 2 && levels <= m + 1);
546 Commit(image.Map(p => ((p * levels) / (m + 1)) * m / (levels - 1)));
547 break;
548 }
549 case "MAXVAL": {
550 int maxval = a[0];
551 ImageOperations.Check(maxval >= 1 && maxval <= 255);
552 Commit(new Image(width, height, maxval,
553 image.Pixels.Select(p => (2 * p * maxval + m) / (2 * m)).ToArray()));
554 break;
555 }
556 case "FLIPH":
557 Commit(ImageOperations.Transform(image, width, height,
558 (x, y) => image.Get(width - 1 - x, y)));
559 break;
560 case "FLIPV":
561 Commit(ImageOperations.Transform(image, width, height,
562 (x, y) => image.Get(x, height - 1 - y)));
563 break;
564 case "ROTATE": {
565 int angle = a[0];
566 ImageOperations.Check(angle == 90 || angle == 180 || angle == 270);
567 if (angle == 180)
568 Commit(ImageOperations.Transform(
569 image, width, height, (x, y) => image.Get(width - 1 - x, height - 1 - y)));
570 else if (angle == 90)
571 Commit(ImageOperations.Transform(image, height, width,
572 (x, y) => image.Get(y, height - 1 - x)));
573 else
574 Commit(ImageOperations.Transform(image, height, width,
575 (x, y) => image.Get(width - 1 - y, x)));
576 break;
577 }
578 case "CROP": {
579 int[] area = CropArea(a);
580 Commit(ImageOperations.Crop(image, area[0], area[1], area[2], area[3]));
581 break;
582 }
583 case "PAD": {
584 var (border, value) = (a[0], a[1]);
585 ImageOperations.Check(border >= 1 && value >= 0 && value <= m);
586 ImageOperations.Check(width + 2 * border <= 64 && height + 2 * border <= 64,
587 "too large");
588 Commit(ImageOperations.Translate(image, width + 2 * border, height + 2 * border, border,
589 border, value));
590 break;
591 }
592 case "SCALE": {
593 var (num, den) = (a[0], a[1]);
594 ImageOperations.Check(num >= 1 && num <= 16 && den >= 1 && den <= 16);
595 int w = width * num / den, h = height * num / den;
596 ImageOperations.Check(w > 0 && h > 0, "too small");
597 ImageOperations.Check(w <= 64 && h <= 64, "too large");
598 Commit(ImageOperations.Transform(image, w, h,
599 (x, y) => image.Get(((2 * x + 1) * den) / (2 * num),
600 ((2 * y + 1) * den) / (2 * num))));
601 break;
602 }
603 case "SHIFT":
604 ImageOperations.Check(WithinLimit(a[0], a[1]) && a[2] >= 0 && a[2] <= m);
605 Commit(ImageOperations.Translate(image, width, height, a[0], a[1], a[2]));
606 break;
607 case "TILE": {
608 var (nx, ny) = (a[0], a[1]);
609 ImageOperations.Check(nx >= 1 && ny >= 1);
610 ImageOperations.Check(width * nx <= 64 && height * ny <= 64, "too large");
611 Commit(ImageOperations.Transform(image, width * nx, height * ny,
612 (x, y) => image.Get(x % width, y % height)));
613 break;
614 }
615 case "BLUR":
616 case "GAUSS":
617 case "SHARPEN": {
618 var kernel = Kernels[name];
619 Commit(ImageOperations.Convolve(image, kernel.Weights, kernel.Divisor));
620 break;
621 }
622 case "CONVOLVE":
623 ImageOperations.Check(a.Take(9).All(v => v >= -16 && v <= 16) && a[9] >= 1 &&
624 a[9] <= 256);
625 Commit(ImageOperations.Convolve(image, a[..9], a[9]));
626 break;
627 case "SOBEL":
628 Commit(ImageOperations.Sobel(image));
629 break;
630 case "MEDIAN":
631 Commit(ImageOperations.Rank(image, 4));
632 break;
633 case "ERODE":
634 Commit(ImageOperations.Rank(image, 0));
635 break;
636 case "DILATE":
637 Commit(ImageOperations.Rank(image, 8));
638 break;
639 case "OPEN":
640 Commit(ImageOperations.Rank(ImageOperations.Rank(image, 0), 8));
641 break;
642 case "CLOSE":
643 Commit(ImageOperations.Rank(ImageOperations.Rank(image, 8), 0));
644 break;
645 case "GRADIENT": {
646 int[] high = ImageOperations.Rank(image, 8).Pixels;
647 int[] low = ImageOperations.Rank(image, 0).Pixels;
648 Commit(image.WithPixels(high.Select((p, i) => p - low[i]).ToArray()));
649 break;
650 }
651 case "LINE":
652 ImageOperations.Check(WithinLimit(a[..4]) && a[4] >= 0 && a[4] <= m);
653 Paint(ImageOperations.Line(a[0], a[1], a[2], a[3]), a[4]);
654 break;
655 case "RECT":
656 ImageOperations.Check(WithinLimit(a[0], a[1]) && a[2] >= 1 && a[2] <= 1000 &&
657 a[3] >= 1 && a[3] <= 1000 && a[4] >= 0 && a[4] <= m);
658 Paint(ImageOperations.Rectangle(image, a[0], a[1], a[2], a[3], fill), a[4]);
659 break;
660 case "CIRCLE":
661 ImageOperations.Check(WithinLimit(a[0], a[1]) && a[2] >= 0 && a[2] <= 1000 &&
662 a[3] >= 0 && a[3] <= m);
663 List<Point> outline = ImageOperations.Circle(a[0], a[1], a[2]);
664 Paint(fill ? ImageOperations.Disc(image, outline) : outline, a[3]);
665 break;
666 case "FLOOD":
667 ImageOperations.Check(image.Contains(a[0], a[1]) && a[2] >= 0 && a[2] <= m);
668 Paint(ImageOperations.Flood(image, a[0], a[1]), a[2]);
669 break;
670 case "HISTOGRAM": {
671 int?[] lows = new int?[16];
672 int[] highs = new int[16], counts = new int[16];
673 for (int value = 0; value <= m; ++value) {
674 int bin = value * 16 / (m + 1);
675 lows[bin] ??= value;
676 highs[bin] = value;
677 }
678 foreach (int pixel in image.Pixels)
679 ++counts[pixel * 16 / (m + 1)];
680 int top = counts.Max();
681 for (int bin = 0; bin < 16; ++bin) {
682 if (lows[bin] == null)
683 continue;
684 Emit($"{lows[bin],3}-{highs[bin],3} {counts[bin],4} |" +
685 new string('#', counts[bin] * 32 / top));
686 }
687 break;
688 }
689 case "EQUALIZE":
690 Commit(ImageOperations.Equalize(image));
691 break;
692 case "LABEL": {
693 ImageOperations.Check(a[0] >= 0 && a[0] <= m);
694 var groups = ImageOperations.Components(image, a[0]);
695 Emit("labels " + groups.Count);
696 for (int i = 0; i < groups.Count; ++i)
697 Emit($"label {i + 1} " +
698 ImageOperations.ComponentDescription(image, a[0], groups[i]));
699 break;
700 }
701 case "DISTANCE":
702 ImageOperations.Check(a[0] >= 0 && a[0] <= m);
703 Commit(ImageOperations.Distance(image, a[0]));
704 break;
705 case "STATS":
706 Emit(ImageOperations.Statistics(image.Pixels, m));
707 break;
708 case "REGION": {
709 int[] area = CropArea(a);
710 Emit(ImageOperations.Statistics(
711 ImageOperations.Crop(image, area[0], area[1], area[2], area[3]).Pixels, m));
712 break;
713 }
714 case "DITHER":
715 Commit(ImageOperations.Dither(image));
716 break;
717 case "ASCIIART": {
718 const string shades = "@#*+=~-:,.";
719 for (int y = 0; y < height; ++y) {
720 char[] row = new char[width];
721 for (int x = 0; x < width; ++x)
722 row[x] = shades[image.Get(x, y) * 10 / (m + 1)];
723 Emit(new string(row));
724 }
725 break;
726 }
727 case "STORE":
728 slots[slot] = image;
729 Emit($"stored {slot} {image.Summary()}");
730 break;
731 case "RECALL":
732 Replace(slots[slot], "recalled " + slot);
733 break;
734 case "DROP":
735 slots.Remove(slot);
736 Emit("dropped " + slot);
737 break;
738 case "BLEND": {
739 var (num, den) = (a[0], a[1]);
740 ImageOperations.Check(num >= 0 && num <= den && den >= 1 && den <= 100);
741 Image saved = MatchingSlot(slot);
742 Commit(image.WithPixels(
743 image.Pixels.Select((p, i) => (p * (den - num) + saved.Pixels[i] * num) / den)
744 .ToArray()));
745 break;
746 }
747 case "DIFF": {
748 Image saved = MatchingSlot(slot);
749 Commit(image.WithPixels(
750 image.Pixels.Select((p, i) => Math.Abs(p - saved.Pixels[i])).ToArray()));
751 break;
752 }
753 case "PASTE": {
754 var (x, y) = (a[0], a[1]);
755 ImageOperations.Check(WithinLimit(x, y));
756 Image saved = slots[slot];
757 ImageOperations.Check(saved.Maxval == m, "maxval mismatch");
758 int[] pixels = (int[])image.Pixels.Clone();
759 int painted = 0;
760 for (int j = 0; j < saved.Height; ++j) {
761 for (int i = 0; i < saved.Width; ++i) {
762 if (!image.Contains(x + i, y + j))
763 continue;
764 pixels[(y + j) * width + x + i] = saved.Get(i, j);
765 ++painted;
766 }
767 }
768 Commit(image.WithPixels(pixels), painted);
769 break;
770 }
771 case "SLOTS":
772 if (slots.Count == 0)
773 Emit("no slots");
774 foreach (string key in slots.Keys.OrderBy(v => v, StringComparer.Ordinal))
775 Emit($"{key} {slots[key].Summary()}");
776 break;
777 case "INFO":
778 Emit($"image {(image == null ? "none" : image.Summary())} undo {undo.Count} " +
779 $"redo {redo.Count} slots {slots.Count}");
780 break;
781 case "UNDO":
782 ImageOperations.Check(undo.Count > 0, "nothing to undo");
783 redo.Push(current!);
784 current = undo.Pop();
785 Emit("undo " + current.Summary());
786 break;
787 case "REDO":
788 ImageOperations.Check(redo.Count > 0, "nothing to redo");
789 undo.Push(current!);
790 current = redo.Pop();
791 Emit("redo " + current.Summary());
792 break;
793 }
794 }
795}

810 editor.Execute(words);
811 }
812 }
