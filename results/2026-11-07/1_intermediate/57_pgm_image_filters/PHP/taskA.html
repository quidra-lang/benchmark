86 {
87 $result = [];
88 for ($dy = -1; $dy <= 1; ++$dy) {

109function readPGM(array $lines): GrayImage
110{
111 $words = [];
112 foreach ($lines as $line) {
113 $text = trim(explode('#', $line, 2)[0]);
114 if ($text !== '') {
115 array_push($words, ...preg_split('/\s+/', $text));
116 }
117 }
118 requireValue($words && $words[0] === 'P2', 'bad magic');
119 $header = static function (int $position, int $limit, string $error) use ($words): int {
120 requireValue(isset($words[$position]), 'truncated header');
121 requireValue((bool) preg_match('/^[0-9]+$/D', $words[$position]), 'bad number');
122 $value = (int) $words[$position];
123 requireValue($value >= 1 && $value <= $limit, $error);
124 return $value;
125 };
126 $width = $header(1, 64, 'bad size');
127 $height = $header(2, 64, 'bad size');
128 $maxval = $header(3, 255, 'bad maxval');
129 $values = array_slice($words, 4);
130 $needed = $width * $height;
131 $pixels = [];
132 for ($i = 0; $i < $needed; ++$i) {
133 $x = $i % $width;
134 $y = intdiv($i, $width);
135 requireValue(isset($values[$i]), 'too few pixels (' . count($values) . " of $needed)");
136 requireValue((bool) preg_match('/^[0-9]+$/D', $values[$i]), "bad pixel at $x,$y");
137 $pixel = (int) $values[$i];
138 requireValue($pixel <= $maxval, "pixel out of range at $x,$y");
139 $pixels[] = $pixel;
140 }
141 requireValue(count($values) === $needed, 'too many pixels (' . count($values) . " of $needed)");
142 return new GrayImage($width, $height, $maxval, $pixels);
143}

145function transform(GrayImage $image, int $width, int $height, callable $valueAt): GrayImage
146{
147 $pixels = [];
148 for ($y = 0; $y < $height; ++$y) {
149 for ($x = 0; $x < $width; ++$x) {
150 $pixels[] = $valueAt($x, $y);
151 }
152 }
153 return new GrayImage($width, $height, $image->maxval, $pixels);

156function convolve(GrayImage $image, array $kernel, int $divisor): GrayImage
157{
158 return transform($image, $image->width, $image->height,
159 static function (int $x, int $y) use ($image, $kernel, $divisor): int {
160 $near = $image->neighborhood($x, $y);
161 $sum = 0;
162 foreach ($kernel as $index => $coefficient) {
163 $sum += $coefficient * $near[$index];
164 }
165 return $image->limit(floorDiv($sum, $divisor));
166 }
167 );
168}

181function sobel(GrayImage $image): GrayImage
182{
183 return transform($image, $image->width, $image->height,
184 static function (int $x, int $y) use ($image): int {
185 $q = $image->neighborhood($x, $y);
186 $gx = $q[2] + 2 * $q[5] + $q[8] - $q[0] - 2 * $q[3] - $q[6];
187 $gy = $q[6] + 2 * $q[7] + $q[8] - $q[0] - 2 * $q[1] - $q[2];
188 return $image->limit(abs($gx) + abs($gy));
189 }
190 );

193function linePoints(int $x, int $y, int $toX, int $toY): array
194{
195 $dx = abs($toX - $x);
196 $dy = -abs($toY - $y);
197 $sx = $x < $toX ? 1 : -1;
198 $sy = $y < $toY ? 1 : -1;
199 $error = $dx + $dy;
200 $points = [];
201 while (true) {
202 $points[] = [$x, $y];
203 if ($x === $toX && $y === $toY) {
204 break;
205 }
206 $twice = 2 * $error;
207 if ($twice >= $dy) {
208 $error += $dy;
209 $x += $sx;
210 }
211 if ($twice <= $dx) {
212 $error += $dx;
213 $y += $sy;
214 }
215 }
216 return $points;
217}

219function rectanglePoints(GrayImage $image, int $x, int $y, int $width, int $height, bool $fill): array
220{
221 $right = $x + $width - 1;
222 $bottom = $y + $height - 1;
223 $points = [];
224 for ($j = max($y, 0); $j <= min($bottom, $image->height - 1); ++$j) {
225 for ($i = max($x, 0); $i <= min($right, $image->width - 1); ++$i) {
226 if ($fill || $i === $x || $i === $right || $j === $y || $j === $bottom) {
227 $points[] = [$i, $j];
228 }
229 }
230 }
231 return $points;

234function circlePoints(int $cx, int $cy, int $radius): array
235{
236 $a = $radius;
237 $b = 0;
238 $error = 1 - $radius;
239 $points = [];
240 while ($a >= $b) {
241 $points[] = [$cx + $a, $cy + $b];
242 $points[] = [$cx + $b, $cy + $a];
243 $points[] = [$cx - $b, $cy + $a];
244 $points[] = [$cx - $a, $cy + $b];
245 $points[] = [$cx - $a, $cy - $b];
246 $points[] = [$cx - $b, $cy - $a];
247 $points[] = [$cx + $b, $cy - $a];
248 $points[] = [$cx + $a, $cy - $b];
249 ++$b;
250 if ($error < 0) {
251 $error += 2 * $b + 1;
252 } else {
253 --$a;
254 $error += 2 * ($b - $a) + 1;
255 }
256 }
257 return $points;
258}

260function filledCircle(GrayImage $image, array $outline): array
261{
262 $rows = [];
263 foreach ($outline as [$x, $y]) {
264 if (isset($rows[$y])) {
265 $rows[$y][0] = min($rows[$y][0], $x);
266 $rows[$y][1] = max($rows[$y][1], $x);
267 } else {
268 $rows[$y] = [$x, $x];
269 }

271 $points = [];
272 foreach ($rows as $y => [$lo, $hi]) {
273 if ($y < 0 || $y >= $image->height) {
274 continue;
275 }
276 for ($x = max(0, $lo); $x <= min($hi, $image->width - 1); ++$x) {
277 $points[] = [$x, $y];
278 }
279 }
280 return $points;

283function floodPoints(GrayImage $image, int $x, int $y): array
284{
285 $target = $image->get($x, $y);
286 $stack = [[$x, $y]];
287 $visited = [$y * $image->width + $x => true];
288 $points = [];
289 while ($stack) {
290 [$cx, $cy] = array_pop($stack);
291 $points[] = [$cx, $cy];
292 foreach (NEIGHBORS4 as [$dx, $dy]) {
293 $nx = $cx + $dx;
294 $ny = $cy + $dy;
295 if (!$image->contains($nx, $ny) || $image->get($nx, $ny) !== $target) {
296 continue;
297 }
298 $index = $ny * $image->width + $nx;
299 if (!isset($visited[$index])) {
300 $visited[$index] = true;
301 $stack[] = [$nx, $ny];
302 }
303 }
304 }
305 return $points;
306}

308function equalize(GrayImage $image): GrayImage
309{
310 $counts = array_fill(0, $image->maxval + 1, 0);
311 foreach ($image->pixels as $pixel) {
312 ++$counts[$pixel];
313 }
314 $cdf = [];
315 $total = 0;
316 foreach ($counts as $count) {
317 $total += $count;
318 $cdf[] = $total;
319 }
320 $minimum = $cdf[min($image->pixels)];
321 $size = count($image->pixels);
322 if ($minimum === $size) {
323 return $image->withPixels($image->pixels);
324 }
325 $span = $size - $minimum;
326 return $image->withPixels(array_map(
327 static fn(int $p): int => intdiv(2 * ($cdf[$p] - $minimum) * $image->maxval + $span, 2 * $span),
328 $image->pixels
329 ));
330}

332function components(GrayImage $image, int $threshold): array
333{
334 $visited = array_fill(0, count($image->pixels), false);
335 $groups = [];
336 foreach ($image->pixels as $start => $value) {
337 if ($visited[$start] || $value < $threshold) {
338 continue;
339 }
340 $visited[$start] = true;
341 $stack = [[$start % $image->width, intdiv($start, $image->width)]];
342 $group = [];
343 while ($stack) {
344 [$x, $y] = array_pop($stack);
345 $group[] = [$x, $y];
346 for ($dy = -1; $dy <= 1; ++$dy) {
347 for ($dx = -1; $dx <= 1; ++$dx) {
348 if ($dx === 0 && $dy === 0) {
349 continue;
350 }
351 $nx = $x + $dx;
352 $ny = $y + $dy;
353 if (!$image->contains($nx, $ny)) {
354 continue;
355 }
356 $index = $ny * $image->width + $nx;
357 if (!$visited[$index] && $image->pixels[$index] >= $threshold) {
358 $visited[$index] = true;
359 $stack[] = [$nx, $ny];
360 }
361 }
362 }
363 }
364 $groups[] = $group;
365 }
366 return $groups;
367}

369function componentText(GrayImage $image, int $threshold, array $group): string
370{
371 $minX = $image->width;
372 $minY = $image->height;
373 $maxX = -1;
374 $maxY = -1;
375 $sumX = 0;
376 $sumY = 0;
377 $perimeter = 0;
378 foreach ($group as [$x, $y]) {
379 $minX = min($minX, $x);
380 $minY = min($minY, $y);
381 $maxX = max($maxX, $x);
382 $maxY = max($maxY, $y);
383 $sumX += $x;
384 $sumY += $y;
385 foreach (NEIGHBORS4 as [$dx, $dy]) {
386 if (!$image->contains($x + $dx, $y + $dy) ||
387 $image->get($x + $dx, $y + $dy) < $threshold) {
388 ++$perimeter;
389 break;
390 }
391 }
392 }
393 $area = count($group);
394 return "area $area perimeter $perimeter box $minX $minY $maxX $maxY center " .
395 intdiv($sumX, $area) . ' ' . intdiv($sumY, $area);
396}

398function distanceMap(GrayImage $image, int $threshold): GrayImage
399{
400 $w = $image->width;
401 $h = $image->height;
402 $far = $w + $h;
403 $distances = [];
404 $hasBackground = false;
405 foreach ($image->pixels as $pixel) {
406 if ($pixel < $threshold) {
407 $hasBackground = true;
408 $distances[] = 0;
409 } else {
410 $distances[] = $far;
411 }
412 }
413 if (!$hasBackground) {
414 return $image->withPixels(array_fill(0, $w * $h, $image->maxval));
415 }
416 for ($y = 0; $y < $h; ++$y) {
417 for ($x = 0; $x < $w; ++$x) {
418 $i = $y * $w + $x;
419 if ($x > 0) $distances[$i] = min($distances[$i], $distances[$i - 1] + 1);
420 if ($y > 0) $distances[$i] = min($distances[$i], $distances[$i - $w] + 1);
421 }
422 }
423 for ($y = $h - 1; $y >= 0; --$y) {
424 for ($x = $w - 1; $x >= 0; --$x) {
425 $i = $y * $w + $x;
426 if ($x < $w - 1) $distances[$i] = min($distances[$i], $distances[$i + 1] + 1);
427 if ($y < $h - 1) $distances[$i] = min($distances[$i], $distances[$i + $w] + 1);
428 }
429 }
430 return $image->withPixels(array_map(static fn(int $n): int => min($n, $image->maxval), $distances));
431}

433function statistics(array $pixels, int $maxval): string
434{
435 $sorted = $pixels;
436 sort($sorted, SORT_NUMERIC);
437 $counts = array_fill(0, $maxval + 1, 0);
438 foreach ($pixels as $pixel) ++$counts[$pixel];
439 $mode = 0;
440 for ($i = 1; $i <= $maxval; ++$i) {
441 if ($counts[$i] > $counts[$mode]) $mode = $i;
442 }
443 $size = count($pixels);
444 return 'min ' . $sorted[0] . ' max ' . $sorted[$size - 1] .
445 ' mean ' . intdiv(array_sum($pixels), $size) .
446 ' median ' . $sorted[intdiv($size - 1, 2)] . " mode $mode";
447}

449function dither(GrayImage $image): GrayImage
450{
451 $errors = $image->pixels;
452 $pixels = array_fill(0, count($errors), 0);
453 $diffusion = [[1, 0, 7], [-1, 1, 3], [0, 1, 5], [1, 1, 1]];
454 for ($y = 0; $y < $image->height; ++$y) {
455 for ($x = 0; $x < $image->width; ++$x) {
456 $index = $y * $image->width + $x;
457 $old = $errors[$index];
458 $new = 2 * $old >= $image->maxval ? $image->maxval : 0;
459 $pixels[$index] = $new;
460 $difference = $old - $new;
461 foreach ($diffusion as [$dx, $dy, $weight]) {
462 if ($image->contains($x + $dx, $y + $dy)) {
463 $neighbor = ($y + $dy) * $image->width + $x + $dx;
464 $errors[$neighbor] += floorDiv($difference * $weight, 16);
465 }
466 }
467 }
468 }
469 return $image->withPixels($pixels);
470}

472final class ImageEditor
473{
474 private ?GrayImage $current = null;
475 private array $undo = [];
476 private array $redo = [];
477 private array $slots = [];
478 public array $output = [];
479
480 private function emit(string $line): void
481 {
482 $this->output[] = $line;
483 }
484
485 private function parse(array $words): array
486 {
487 $name = array_shift($words);
488 requireValue(array_key_exists($name, PATTERNS), "unknown command $name");
489 $pattern = PATTERNS[$name];
490 $fill = in_array($name, ['RECT', 'CIRCLE'], true) &&
491 count($words) === strlen($pattern) + 1 && end($words) === 'FILL';
492 if ($fill) array_pop($words);
493 requireValue(count($words) === strlen($pattern), 'bad arguments');
494 $values = [];
495 $slot = null;
496 foreach (str_split($pattern) as $i => $kind) {
497 if ($kind === 'i') {
498 requireValue((bool) preg_match('/^(?:0|-?[1-9][0-9]{0,6})$/D', $words[$i]), 'bad arguments');
499 $values[] = (int) $words[$i];
500 } else {
501 requireValue((bool) preg_match('/^[a-z]{1,8}$/D', $words[$i]), 'bad arguments');
502 $slot = $words[$i];
503 }
504 }
505 return [$name, $values, $slot, $fill];
506 }
507
508 public function execute(array $words): void
509 {
510 try {
511 [$name, $args, $slot, $fill] = $this->parse($words);
512 if (!in_array($name, IMAGELESS, true)) {
513 requireValue($this->current !== null, 'no image');
514 }
515 if (in_array($name, NEED_SLOT, true)) {
516 requireValue(isset($this->slots[$slot]), "no slot $slot");
517 }
518 $this->run($name, $args, $slot, $fill);
519 } catch (CommandError $error) {
520 $this->emit('error: ' . $error->getMessage());
521 }
522 }
523
524 public function load(array $lines): void
525 {
526 try {
527 $this->replace(readPGM($lines), 'loaded');
528 } catch (CommandError $error) {
529 $this->emit('error: load: ' . $error->getMessage());
530 }
531 }

535 $this->current = $image;
536 $this->undo = [];
537 $this->redo = [];

541 private function commit(GrayImage $image, ?int $painted = null): void

548 }

550 private function paint(array $points, int $value): void
551 {
552 $image = $this->current;
553 $marked = [];
554 foreach ($points as [$x, $y]) {
555 if ($image->contains($x, $y)) $marked[$y * $image->width + $x] = true;
556 }
557 $pixels = $image->pixels;
558 foreach ($marked as $index => $_) $pixels[$index] = $value;
559 $this->commit($image->withPixels($pixels), count($marked));
560 }

570 private function matchingSlot(string $name): GrayImage
571 {
572 $other = $this->slots[$name];
573 $image = $this->current;
574 requireValue($other->width === $image->width && $other->height === $image->height, 'size mismatch');
575 requireValue($other->maxval === $image->maxval, 'maxval mismatch');
576 return $other;

578
579 private function run(string $name, array $a, ?string $slot, bool $fill): void
580 {
581 $image = $this->current;
582 $w = $image?->width;
583 $h = $image?->height;
584 $m = $image?->maxval;
585 switch ($name) {
586 case 'NEW':
587 [$width, $height, $maxval, $value] = $a;
588 requireValue($width >= 1 && $width <= 64 && $height >= 1 && $height <= 64);
589 requireValue($maxval >= 1 && $maxval <= 255 && $value >= 0 && $value <= $maxval);
590 $this->replace(new GrayImage($width, $height, $maxval,
591 array_fill(0, $width * $height, $value)), 'created');
592 break;
593 case 'OUTPUT':
594 $this->emit('P2');
595 $this->emit("$w $h");
596 $this->emit((string) $m);
597 for ($y = 0; $y < $h; ++$y) $this->emit(implode(' ', $image->row($y)));
598 break;
599 case 'INVERT':
600 $this->commit($image->withPixels(array_map(static fn(int $p): int => $m - $p, $image->pixels)));
601 break;
602 case 'BRIGHTNESS':
603 requireValue($a[0] >= -255 && $a[0] <= 255);
604 $this->commit($image->withPixels(array_map(
605 static fn(int $p): int => $image->limit($p + $a[0]), $image->pixels
606 )));
607 break;
608 case 'CONTRAST':
609 [$num, $den] = $a;
610 requireValue($num >= 0 && $num <= 100 && $den >= 1 && $den <= 100);
611 $center = intdiv($m + 1, 2);
612 $this->commit($image->withPixels(array_map(
613 static fn(int $p): int => $image->limit($center + floorDiv(($p - $center) * $num, $den)),
614 $image->pixels
615 )));
616 break;
617 case 'THRESHOLD':
618 requireValue($a[0] >= 0 && $a[0] <= $m);
619 $this->commit($image->withPixels(array_map(
620 static fn(int $p): int => $p >= $a[0] ? $m : 0, $image->pixels
621 )));
622 break;
623 case 'LEVELS':
624 [$lo, $hi] = $a;
625 requireValue(0 <= $lo && $lo < $hi && $hi <= $m);
626 $this->commit($image->withPixels(array_map(
627 static fn(int $p): int => $image->limit(floorDiv(($p - $lo) * $m, $hi - $lo)),
628 $image->pixels
629 )));
630 break;
631 case 'POSTERIZE':
632 $levels = $a[0];
633 requireValue($levels >= 2 && $levels <= $m + 1);
634 $this->commit($image->withPixels(array_map(
635 static fn(int $p): int => intdiv(intdiv($p * $levels, $m + 1) * $m, $levels - 1),
636 $image->pixels
637 )));
638 break;
639 case 'MAXVAL':
640 $newMax = $a[0];
641 requireValue($newMax >= 1 && $newMax <= 255);
642 $pixels = array_map(static fn(int $p): int => intdiv(2 * $p * $newMax + $m, 2 * $m),
643 $image->pixels);
644 $this->commit(new GrayImage($w, $h, $newMax, $pixels));
645 break;
646 case 'FLIPH':
647 $this->commit(transform($image, $w, $h,
648 static fn(int $x, int $y): int => $image->get($w - $x - 1, $y)));
649 break;
650 case 'FLIPV':
651 $this->commit(transform($image, $w, $h,
652 static fn(int $x, int $y): int => $image->get($x, $h - $y - 1)));
653 break;
654 case 'ROTATE':
655 $angle = $a[0];
656 requireValue(in_array($angle, [90, 180, 270], true));
657 if ($angle === 90) {
658 $result = transform($image, $h, $w,
659 static fn(int $x, int $y): int => $image->get($y, $h - 1 - $x));
660 } elseif ($angle === 180) {
661 $result = transform($image, $w, $h,
662 static fn(int $x, int $y): int => $image->get($w - 1 - $x, $h - 1 - $y));
663 } else {
664 $result = transform($image, $h, $w,
665 static fn(int $x, int $y): int => $image->get($w - 1 - $y, $x));
666 }
667 $this->commit($result);
668 break;
669 case 'CROP':
670 [$x, $y, $width, $height] = $this->cropArea($a);
671 $this->commit(transform($image, $width, $height,
672 static fn(int $i, int $j): int => $image->get($x + $i, $y + $j)));
673 break;
674 case 'PAD':
675 [$n, $value] = $a;
676 requireValue($n >= 1 && $value >= 0 && $value <= $m);
677 requireValue($w + 2 * $n <= 64 && $h + 2 * $n <= 64, 'too large');
678 $this->commit(transform($image, $w + 2 * $n, $h + 2 * $n,
679 static fn(int $x, int $y): int =>
680 $image->contains($x - $n, $y - $n) ? $image->get($x - $n, $y - $n) : $value));
681 break;
682 case 'SCALE':
683 [$num, $den] = $a;
684 requireValue($num >= 1 && $num <= 16 && $den >= 1 && $den <= 16);
685 $width = intdiv($w * $num, $den);
686 $height = intdiv($h * $num, $den);
687 requireValue($width >= 1 && $height >= 1, 'too small');
688 requireValue($width <= 64 && $height <= 64, 'too large');
689 $this->commit(transform($image, $width, $height,
690 static fn(int $x, int $y): int =>
691 $image->get(intdiv((2 * $x + 1) * $den, 2 * $num),
692 intdiv((2 * $y + 1) * $den, 2 * $num))));
693 break;
694 case 'SHIFT':
695 [$dx, $dy, $value] = $a;
696 requireValue($dx >= -1000 && $dx <= 1000 && $dy >= -1000 && $dy <= 1000);
697 requireValue($value >= 0 && $value <= $m);
698 $this->commit(transform($image, $w, $h,
699 static fn(int $x, int $y): int =>
700 $image->contains($x - $dx, $y - $dy) ?
701 $image->get($x - $dx, $y - $dy) : $value));
702 break;
703 case 'TILE':
704 [$nx, $ny] = $a;
705 requireValue($nx >= 1 && $ny >= 1);
706 requireValue($w * $nx <= 64 && $h * $ny <= 64, 'too large');
707 $this->commit(transform($image, $w * $nx, $h * $ny,
708 static fn(int $x, int $y): int => $image->get($x % $w, $y % $h)));
709 break;
710 case 'BLUR':
711 case 'GAUSS':
712 case 'SHARPEN':
713 [$kernel, $divisor] = KERNELS[$name];
714 $this->commit(convolve($image, $kernel, $divisor));
715 break;
716 case 'CONVOLVE':
717 requireValue(!array_filter(array_slice($a, 0, 9), static fn(int $n): bool => $n < -16 || $n > 16)
718 && $a[9] >= 1 && $a[9] <= 256);
719 $this->commit(convolve($image, array_slice($a, 0, 9), $a[9]));
720 break;
721 case 'SOBEL':
722 $this->commit(sobel($image));
723 break;
724 case 'MEDIAN':
725 $this->commit(rankFilter($image, 4));
726 break;
727 case 'ERODE':
728 $this->commit(rankFilter($image, 0));
729 break;
730 case 'DILATE':
731 $this->commit(rankFilter($image, 8));
732 break;
733 case 'OPEN':
734 $this->commit(rankFilter(rankFilter($image, 0), 8));
735 break;
736 case 'CLOSE':
737 $this->commit(rankFilter(rankFilter($image, 8), 0));
738 break;
739 case 'GRADIENT':
740 $lower = rankFilter($image, 0)->pixels;
741 $upper = rankFilter($image, 8)->pixels;
742 $pixels = array_map(static fn(int $hi, int $lo): int => $hi - $lo, $upper, $lower);
743 $this->commit($image->withPixels($pixels));
744 break;
745 case 'LINE':
746 [$x0, $y0, $x1, $y1, $value] = $a;
747 requireValue(!array_filter(array_slice($a, 0, 4),
748 static fn(int $n): bool => $n < -1000 || $n > 1000) && $value >= 0 && $value <= $m);
749 $this->paint(linePoints($x0, $y0, $x1, $y1), $value);
750 break;
751 case 'RECT':
752 [$x, $y, $width, $height, $value] = $a;
753 requireValue($x >= -1000 && $x <= 1000 && $y >= -1000 && $y <= 1000);
754 requireValue($width >= 1 && $width <= 1000 && $height >= 1 && $height <= 1000);
755 requireValue($value >= 0 && $value <= $m);
756 $this->paint(rectanglePoints($image, $x, $y, $width, $height, $fill), $value);
757 break;
758 case 'CIRCLE':
759 [$cx, $cy, $radius, $value] = $a;
760 requireValue($cx >= -1000 && $cx <= 1000 && $cy >= -1000 && $cy <= 1000);
761 requireValue($radius >= 0 && $radius <= 1000 && $value >= 0 && $value <= $m);
762 $points = circlePoints($cx, $cy, $radius);
763 $this->paint($fill ? filledCircle($image, $points) : $points, $value);
764 break;
765 case 'FLOOD':
766 [$x, $y, $value] = $a;
767 requireValue($image->contains($x, $y) && $value >= 0 && $value <= $m);
768 $this->paint(floodPoints($image, $x, $y), $value);
769 break;
770 case 'HISTOGRAM':
771 $lower = array_fill(0, 16, null);
772 $upper = array_fill(0, 16, 0);
773 $counts = array_fill(0, 16, 0);
774 for ($value = 0; $value <= $m; ++$value) {
775 $bin = intdiv($value * 16, $m + 1);
776 if ($lower[$bin] === null) $lower[$bin] = $value;
777 $upper[$bin] = $value;
778 }
779 foreach ($image->pixels as $pixel) ++$counts[intdiv($pixel * 16, $m + 1)];
780 $top = max($counts);
781 for ($bin = 0; $bin < 16; ++$bin) {
782 if ($lower[$bin] === null) continue;
783 $this->emit(sprintf('%3d-%3d %4d |%s', $lower[$bin], $upper[$bin],
784 $counts[$bin], str_repeat('#', intdiv($counts[$bin] * 32, $top))));
785 }
786 break;
787 case 'EQUALIZE':
788 $this->commit(equalize($image));
789 break;
790 case 'LABEL':
791 requireValue($a[0] >= 0 && $a[0] <= $m);
792 $groups = components($image, $a[0]);
793 $this->emit('labels ' . count($groups));
794 foreach ($groups as $i => $group) {
795 $this->emit('label ' . ($i + 1) . ' ' . componentText($image, $a[0], $group));
796 }
797 break;
798 case 'DISTANCE':
799 requireValue($a[0] >= 0 && $a[0] <= $m);
800 $this->commit(distanceMap($image, $a[0]));
801 break;
802 case 'STATS':
803 $this->emit(statistics($image->pixels, $m));
804 break;
805 case 'REGION':
806 [$x, $y, $width, $height] = $this->cropArea($a);
807 $region = transform($image, $width, $height,
808 static fn(int $i, int $j): int => $image->get($x + $i, $y + $j));
809 $this->emit(statistics($region->pixels, $m));
810 break;
811 case 'DITHER':
812 $this->commit(dither($image));
813 break;
814 case 'ASCIIART':
815 for ($y = 0; $y < $h; ++$y) {
816 $this->emit(implode('', array_map(
817 static fn(int $p): string => SHADES[intdiv($p * 10, $m + 1)], $image->row($y))));
818 }
819 break;
820 case 'STORE':
821 $this->slots[$slot] = $image;
822 $this->emit("stored $slot " . $image->summary());
823 break;
824 case 'RECALL':
825 $this->replace($this->slots[$slot], "recalled $slot");
826 break;
827 case 'DROP':
828 unset($this->slots[$slot]);
829 $this->emit("dropped $slot");
830 break;
831 case 'BLEND':
832 [$num, $den] = $a;
833 requireValue($num >= 0 && $den >= 1 && $den <= 100 && $num <= $den);
834 $other = $this->matchingSlot($slot);
835 $pixels = array_map(static fn(int $p, int $q): int =>
836 intdiv($p * ($den - $num) + $q * $num, $den), $image->pixels, $other->pixels);
837 $this->commit($image->withPixels($pixels));
838 break;
839 case 'DIFF':
840 $other = $this->matchingSlot($slot);
841 $pixels = array_map(static fn(int $p, int $q): int => abs($p - $q),
842 $image->pixels, $other->pixels);
843 $this->commit($image->withPixels($pixels));
844 break;
845 case 'PASTE':
846 [$x, $y] = $a;
847 requireValue($x >= -1000 && $x <= 1000 && $y >= -1000 && $y <= 1000);
848 $other = $this->slots[$slot];
849 requireValue($other->maxval === $m, 'maxval mismatch');
850 $pixels = $image->pixels;
851 $painted = 0;
852 for ($j = 0; $j < $other->height; ++$j) {
853 for ($i = 0; $i < $other->width; ++$i) {
854 if ($image->contains($x + $i, $y + $j)) {
855 $pixels[($y + $j) * $w + $x + $i] = $other->get($i, $j);
856 ++$painted;
857 }
858 }
859 }
860 $this->commit($image->withPixels($pixels), $painted);
861 break;
862 case 'SLOTS':
863 if (!$this->slots) $this->emit('no slots');
864 $sorted = $this->slots;
865 ksort($sorted, SORT_STRING);
866 foreach ($sorted as $key => $stored) $this->emit("$key " . $stored->summary());
867 break;
868 case 'INFO':
869 $summary = $image === null ? 'none' : $image->summary();
870 $this->emit("image $summary undo " . count($this->undo) .
871 ' redo ' . count($this->redo) . ' slots ' . count($this->slots));
872 break;
873 case 'UNDO':
874 requireValue((bool) $this->undo, 'nothing to undo');
875 $this->redo[] = $image;
876 $this->current = array_pop($this->undo);
877 $this->emit('undo ' . $this->current->summary());
878 break;
879 case 'REDO':
880 requireValue((bool) $this->redo, 'nothing to redo');
881 $this->undo[] = $image;
882 $this->current = array_pop($this->redo);
883 $this->emit('redo ' . $this->current->summary());
884 break;
885 }
886 }
887}
