95int seatValue(String word) {
96 if (word.length != 1 || !seatLetters.contains(word)) fail('bad seat $word');
97 return seatLetters.indexOf(word);

100int cardValue(String word) {
101 if (word.length != 2 ||
102 !rankLetters.contains(word[0]) ||
103 !suitLetters.contains(word[1])) {
104 fail('bad card $word');
105 }
106 return suitLetters.indexOf(word[1]) * 13 + rankLetters.indexOf(word[0]);
107}

114int bounded(String word, int low, int high) {
115 final value = int.tryParse(word);
116 if (value == null || value < low || value > high) fail('bad value $word');
117 return value;

120String choice(String word, List<String> allowed) =>

123bool switchValue(String word) => choice(word, const ['ON', 'OFF']) == 'ON';

125String passValue(String word) =>
126 RegExp(r'^[LRAH]{1,8}$').hasMatch(word) ? word : fail('bad value $word');

128String onOff(bool on) => on ? 'ON' : 'OFF';

130int byRank(int a, int b) =>
131 a % 13 != b % 13 ? a % 13 - b % 13 : a ~/ 13 - b ~/ 13;
132
133int byRankDescending(int a, int b) =>
134 a % 13 != b % 13 ? b % 13 - a % 13 : a ~/ 13 - b ~/ 13;
135
136int lowest(Iterable<int> cards) =>
137 cards.reduce((best, card) => byRank(card, best) < 0 ? card : best);
138
139int highest(Iterable<int> cards) => cards.reduce(
140 (best, card) => byRankDescending(card, best) < 0 ? card : best,
141);

143bool isPenalty(int card) => card == queenOfSpades || card ~/ 13 == 3;

157void checkDistinct(List<int> cards) {
158 final seen = <int>{};
159 for (final card in cards) {
160 if (!seen.add(card)) fail('duplicate card ${cardText(card)}');

164String averageText(int total, int games) {
165 final numerator = 100 * total;
166 final cents = (numerator - numerator % games) ~/ games;
167 final sign = cents < 0 ? '-' : '';
168 final fraction = (cents.abs() % 100).toString().padLeft(2, '0');
169 return '$sign${cents.abs() ~/ 100}.$fraction';
170}

172int trickWinner(List<Play> plays) {
173 final led = plays.first.card ~/ 13;
174 return plays
175 .where((play) => play.card ~/ 13 == led)
176 .reduce((best, play) => play.card % 13 > best.card % 13 ? play : best)
177 .seat;
178}

241 Rules withRule(String name, String word) => switch (name) {

337 int point(int card) {
338 if (card == queenOfSpades) return 13;
339 if (card ~/ 13 == 3) return 1;
340 return card == jackOfDiamonds && rules.jack ? -10 : 0;

353 void deal() {
354 number++;
355 direction = Direction.fromCode(
356 rules.pass[(number - 1) % rules.pass.length],
357 );
358 round = Round();
359 final used = {for (final cards in stacked.values) ...cards};
360 final deck = [
361 for (var card = 0; card < 52; card++)
362 if (!used.contains(card)) card,
363 ];
364 generator.shuffle(deck);
365 final free = [
366 for (var seat = 0; seat < 4; seat++)
367 if (!stacked.containsKey(seat)) seat,
368 ];
369 stacked.forEach((seat, cards) => round.hands[seat].addAll(cards));
370 for (var i = 0; i < deck.length; i++) {
371 round.hands[free[i % free.length]].add(deck[i]);
372 }
373 stacked.clear();
374 passes.clear();
375 undoStack.clear();
376 write(1, 'hand $number pass ${direction.label}');
377 for (var seat = 0; seat < 4; seat++) {
378 write(2, '${seatName(seat)}: ${cardsText(round.hands[seat])}');
379 }
380 if (direction == Direction.hold) {
381 beginPlay();
382 } else {
383 state = State.passing;
384 }
385 }

399 round.hands[seat].removeAll(cards);

403 passes.forEach(

408 }

413 Set<int> legal() {
414 final hand = round.hands[actor];
415 if (round.current.isEmpty) {
416 if (round.tricks.isEmpty) return {twoOfClubs};
417 final nonHearts = hand.where((card) => card ~/ 13 != 3).toSet();
418 return !round.broken && nonHearts.isNotEmpty ? nonHearts : {...hand};
419 }
420 final suit = round.current.first.card ~/ 13;
421 final followers = hand.where((card) => card ~/ 13 == suit).toSet();
422 if (followers.isNotEmpty) return followers;
423 if (round.tricks.isEmpty && !rules.blood) {
424 final safe = hand.where((card) => !isPenalty(card)).toSet();
425 if (safe.isNotEmpty) return safe;
426 }
427 return {...hand};
428 }

430 List<int> choosePass(int seat) {
431 final hand = round.hands[seat];
432 final low = hand.toList()..sort(byRank);
433 final high = hand.toList()..sort(byRankDescending);
434 final hearts = high.where((card) => card ~/ 13 == 3);
435 final special = [
436 queenOfSpades,
437 kingOfSpades,
438 aceOfSpades,
439 ].where(hand.contains);
440 final List<int> order;
441 switch (bots[seat]) {
442 case Strategy.low:
443 order = high;
444 case Strategy.high:
445 order = low;
446 case Strategy.moon:
447 order = [
448 ...low.where((card) => card ~/ 13 != 3),
449 ...low.where((card) => card ~/ 13 == 3),
450 ];
451 case Strategy.smart:
452 final isProtected =
453 hand.contains(queenOfSpades) && countSuit(hand, 2) >= 5;
454 final minorSuits = [0, 1].where((suit) => countSuit(hand, suit) > 0);
455 final shortSuit = minorSuits.isEmpty
456 ? null
457 : minorSuits.reduce(
458 (best, suit) =>
459 countSuit(hand, suit) < countSuit(hand, best) ? suit : best,
460 );
461 order = [
462 if (!isProtected) ...special,
463 if (shortSuit != null)
464 ...high.where((card) => card ~/ 13 == shortSuit),
465 ...hearts,
466 ...high,
467 ];
468 case Strategy.basic:
469 order = [...special, ...hearts, ...high];
470 }
471 return order.toSet().take(3).toList();
472 }
473
474 bool queenPlayed() =>
475 round.taken.any((cards) => cards.contains(queenOfSpades)) ||
476 round.current.any((play) => play.card == queenOfSpades);
477
478 bool othersTookPenalty(int seat) => [
479 0,
480 1,
481 2,
482 3,
483 ].any((other) => other != seat && round.taken[other].any(isPenalty));

485 int choosePlay() {
486 final seat = actor;
487 final hand = round.hands[seat];
488 final playable = legal();
489 var strategy = bots[seat];
490 if (strategy == Strategy.low) return lowest(playable);
491 if (strategy == Strategy.high) return highest(playable);
492 if (strategy == Strategy.moon && othersTookPenalty(seat)) {
493 strategy = Strategy.basic;
494 }
495 if (round.current.isEmpty) {
496 if (strategy == Strategy.smart &&
497 !queenPlayed() &&
498 !hand.contains(queenOfSpades)) {
499 final smallSpades = playable.where(
500 (card) => card ~/ 13 == 2 && card % 13 < 10,
501 );
502 if (smallSpades.isNotEmpty) return lowest(smallSpades);
503 }
504 final moon = strategy == Strategy.moon;
505 final playableSuits = playable.map((card) => card ~/ 13).toSet().toList()
506 ..sort();
507 final suit = playableSuits.reduce((best, candidate) {
508 final difference =
509 countSuit(playable, candidate) - countSuit(playable, best);
510 return (moon ? difference > 0 : difference < 0) ? candidate : best;
511 });
512 final suited = playable.where((card) => card ~/ 13 == suit);
513 return moon ? highest(suited) : lowest(suited);
514 }
515 final led = round.current.first.card ~/ 13;
516 if (countSuit(hand, led) > 0) {
517 final winning = round.current
518 .where((play) => play.card ~/ 13 == led)
519 .map((play) => play.card % 13)
520 .reduce(max);
521 if (strategy == Strategy.moon) {
522 final high = highest(playable);
523 return high % 13 > winning ? high : lowest(playable);
524 }
525 final lower = playable.where((card) => card % 13 < winning);
526 if (lower.isNotEmpty) return highest(lower);
527 if (strategy == Strategy.smart && round.current.length == 3) {
528 final withoutQueen = playable.difference({queenOfSpades});
529 return withoutQueen.isEmpty ? queenOfSpades : highest(withoutQueen);
530 }
531 return lowest(playable);
532 }
533 if (strategy == Strategy.moon) {
534 final safe = playable.where((card) => !isPenalty(card));
535 return lowest(safe.isEmpty ? playable : safe);
536 }
537 if (playable.contains(queenOfSpades)) return queenOfSpades;
538 if (strategy == Strategy.smart && !queenPlayed()) {
539 if (playable.contains(aceOfSpades)) return aceOfSpades;
540 if (playable.contains(kingOfSpades)) return kingOfSpades;
541 }
542 final hearts = playable.where((card) => card ~/ 13 == 3);
543 return highest(hearts.isEmpty ? playable : hearts);
544 }

550 void play(int card) {
551 final seat = actor;
552 undoStack.add((
553 round: round.copy(),
554 logLength: log.length,
555 play: (seat: seat, card: card),
556 ));
557 round.hands[seat].remove(card);
558 round.current.add((seat: seat, card: card));
559 if (card ~/ 13 == 3 || (card == queenOfSpades && rules.qsBreak)) {
560 round.broken = true;
561 }
562 if (round.current.length == 4) {
563 final winner = trickWinner(round.current);
564 round.taken[winner].addAll(round.current.map((play) => play.card));
565 round.trickCounts[winner]++;
566 round.tricks.add(round.current);
567 write(2, trickLine(round.current, round.tricks.length));
568 round.current = [];
569 round.leader = winner;
570 if (round.tricks.length == 13) finishHand();
571 }
572 }

574 Play undoCard() {
575 final snapshot = undoStack.removeLast();
576 round = snapshot.round;
577 log.removeRange(snapshot.logLength, log.length);
578 return snapshot.play;
579 }

581 void claim(int seat) {
582 final remaining = round.hands.expand((hand) => hand).toList();
583 for (final hand in round.hands) {
584 hand.clear();
585 }
586 final count = 13 - round.tricks.length;
587 round.taken[seat].addAll(remaining);
588 round.trickCounts[seat] += count;
589 write(
590 2,
591 'claim ${seatName(seat)} tricks $count points ${points(remaining)}',
592 );
593 finishHand();
594 }

596 bool gameOver(List<int> totals) =>
597 number >= rules.limit || totals.reduce(max) >= rules.target;
598
599 void finishHand() {
600 final handPoints = [for (final cards in round.taken) points(cards)];
601 final shooter = [0, 1, 2, 3]
602 .where((seat) => round.taken[seat].where(isPenalty).length == 14)
603 .firstOrNull;
604 if (shooter != null) {
605 handPoints[shooter] -= 26;
606 var method = rules.moon;
607 if (method == 'AUTO') {
608 final added = [
609 for (var seat = 0; seat < 4; seat++)
610 scores[seat] + handPoints[seat] + (seat == shooter ? 0 : 26),
611 ];
612 method = gameOver(added) && ranking(added).first != shooter
613 ? 'SUB'
614 : 'ADD';
615 }
616 if (method == 'SUB') {
617 handPoints[shooter] -= 26;
618 } else {
619 for (var seat = 0; seat < 4; seat++) {
620 if (seat != shooter) handPoints[seat] += 26;
621 }
622 }
623 write(1, 'moon ${seatName(shooter)} $method');
624 }
625 for (var seat = 0; seat < 4; seat++) {
626 final cards = round.taken[seat];
627 scores[seat] += handPoints[seat];
628 stats[seat] += Stats(
629 round.trickCounts[seat],
630 countSuit(cards, 3),
631 cards.contains(queenOfSpades) ? 1 : 0,
632 shooter == seat ? 1 : 0,
633 );
634 }
635 write(1, pairs('points', handPoints));
636 write(1, pairs('scores', scores));
637 history.add((
638 number: number,
639 direction: direction,
640 points: handPoints,
641 shooter: shooter,
642 record: [...round.record],
643 ));
644 final over = gameOver(scores);
645 state = over ? State.none : State.between;
646 if (over) {
647 write(0, 'game over hands $number');
648 final order = ranking(scores);
649 for (var place = 0; place < 4; place++) {
650 final seat = order[place];
651 write(0, 'rank ${place + 1} ${seatName(seat)} ${scores[seat]}');
652 }
653 for (final line in statsLines()) {
654 write(0, line);
655 }
656 }
657 }

659 List<String> statsLines() => [
660 for (var seat = 0; seat < 4; seat++)
661 'stats ${seatName(seat)} ${stats[seat].text}',
662 ];

664 void autoPass() {
665 for (var seat = 0; seat < 4; seat++) {
666 if (!passes.containsKey(seat)) passCards(seat, choosePass(seat));
667 }
668 }
669
670 void autoHand() {
671 if (state == State.passing) autoPass();
672 while (state == State.playing) {
673 play(choosePlay());
674 }
675 }
676
677 void autoGame() {
678 if (state == State.passing || state == State.playing) autoHand();
679 while (state != State.none) {
680 deal();
681 autoHand();
682 }
683 }

726 List<String> simulate(String command, int seed, int count) {
727 final generator = Generator(seed);
728 if (command == 'GAME') {
729 final game = Game(rules, bots, generator)..autoGame();
730 return [
731 'game seed $seed',
732 for (final line in game.log)
733 if (line.level <= count) line.text,
734 ];
735 }
736 final rounds = command == 'ROTATE' ? 4 : 1;
737 final places = List.generate(4, (_) => [0, 0, 0, 0]);
738 final totals = [0, 0, 0, 0];
739 final stats = List.filled(4, const Stats());
740 var hands = 0;
741 for (var rotation = 0; rotation < rounds; rotation++) {
742 final strategies = [
743 for (var seat = 0; seat < 4; seat++) bots[(seat - rotation) % 4],
744 ];
745 for (var i = 0; i < count; i++) {
746 final game = Game(rules, strategies, generator)..autoGame();
747 hands += game.number;
748 final order = ranking(game.scores);
749 for (var original = 0; original < 4; original++) {
750 final actual = (original + rotation) % 4;
751 places[original][order.indexOf(actual)]++;
752 totals[original] += game.scores[actual];
753 stats[original] += game.stats[actual];
754 }
755 }
756 }
757 final games = rounds * count;
758 return [
759 '${command.toLowerCase()} seed $seed games $games hands $hands',
760 for (var seat = 0; seat < 4; seat++)
761 '${seatName(seat)}${command == 'ROTATE' ? ' ${bots[seat].label}' : ''} '
762 'places ${places[seat].join(' ')} total ${totals[seat]} '
763 'average ${averageText(totals[seat], games)} ${stats[seat].text}',
764 ];
765 }
766
767 List<String> command(List<String> words) {
768 final name = words.first;
769 final args = words.sublist(1);
770 if (!commands.contains(name)) fail('unknown command $name');
771 switch (name) {
772 case 'RULE':
773 final changed = rules.withRule(args[0], args[1]);
774 if (active) fail('game in progress');
775 rules = changed;
776 return ['rule ${args[0]} ${args[1]}'];
777 case 'BOT':
778 final seat = seatValue(args[0]);
779 bots[seat] = strategyValue(args[1]);
780 return ['bot ${seatName(seat)} ${args[1]}'];
781 case 'SETTINGS':
782 return [
783 rules.text,
784 pairs('bots', [for (final bot in bots) bot.label]),
785 ];
786 case 'GAME' || 'MATCH' || 'ROTATE':
787 final seed = bounded(args[0], 1, 2147483646);
788 final count = switch (name) {
789 'GAME' => bounded(args[1], 0, 2),
790 'MATCH' => bounded(args[1], 1, 50),
791 _ => bounded(args[1], 1, 10),
792 };
793 return simulate(name, seed, count);
794 case 'NEW':
795 final seed = bounded(args[0], 1, 2147483646);
796 if (active) fail('game in progress');
797 game = Game(rules, bots, Generator(seed));
798 return ['new game seed $seed'];
799 case 'ABANDON':
800 requireGame();
801 game = null;
802 return ['game abandoned'];
803 }
804 return tableCommand(name, args);
805 }

807 List<String> tableCommand(String name, List<String> args) {
808 final seat = const {'STACK', 'PASS', 'PLAY', 'CLAIM'}.contains(name)
809 ? seatValue(args[0])
810 : -1;
811 final cards = const {'STACK', 'PASS', 'PLAY'}.contains(name)
812 ? args.skip(1).map(cardValue).toList()
813 : <int>[];
814 if (name == 'STACK' && cards.length != 13) fail('need 13 cards');
815 if (name == 'AUTO' &&
816 !const {'PASS', 'CARD', 'TRICK', 'HAND', 'GAME'}.contains(args[0])) {
817 fail('bad value ${args[0]}');
818 }
819
820 final game = switch (name) {
821 'STACK' || 'DEAL' => requireBetween(),
822 'SHOW' => requireHand(),
823 'PASS' => requirePassing(),
824 'PLAY' ||
825 'UNDO' ||
826 'CLAIM' ||
827 'LEGAL' ||
828 'PREVIEW' ||
829 'TAKEN' ||
830 'TRICKS' ||
831 'VOIDS' ||
832 'UNSEEN' => requirePlaying(),
833 'AUTO' => switch (args[0]) {
834 'PASS' => requirePassing(),
835 'CARD' || 'TRICK' => requirePlaying(),
836 'HAND' => requireHand(),
837 _ => requireGame(),
838 },
839 _ => requireGame(),
840 };
841 final round = game.round;
842 final before = game.log.length;
843 final prefix = <String>[];
844 switch (name) {
845 case 'STACK':
846 checkDistinct(cards);
847 for (final card in cards) {
848 for (var other = 0; other < 4; other++) {
849 if (other != seat &&
850 (game.stacked[other]?.contains(card) ?? false)) {
851 fail('${cardText(card)} stacked for ${seatName(other)}');
852 }
853 }
854 }
855 game.stacked[seat] = cards.toSet();
856 return ['stacked ${seatName(seat)}'];
857 case 'DEAL':
858 game.deal();
859 case 'SHOW':
860 return [
861 for (var other = 0; other < 4; other++)
862 '${seatName(other)}: ${cardsText(round.hands[other])}',
863 ];
864 case 'PASS':
865 if (game.passes.containsKey(seat)) {
866 fail('${seatName(seat)} already passed');
867 }
868 checkDistinct(cards);
869 for (final card in cards) {
870 if (!round.hands[seat].contains(card)) {
871 fail('${seatName(seat)} does not hold ${cardText(card)}');
872 }
873 }
874 game.passCards(seat, cards);
875 case 'PLAY':
876 final card = cards.first;
877 if (seat != game.actor) fail('waiting for ${seatName(game.actor)}');
878 if (!round.hands[seat].contains(card)) {
879 fail('${seatName(seat)} does not hold ${cardText(card)}');
880 }
881 if (!game.legal().contains(card)) {
882 if (round.current.isEmpty) {
883 fail(round.tricks.isEmpty ? 'must lead 2C' : 'hearts not broken');
884 }
885 final led = round.current.first.card ~/ 13;
886 if (countSuit(round.hands[seat], led) > 0) {
887 fail('must follow ${suitLetters[led]}');
888 }
889 fail('no points on first trick');
890 }
891 prefix.add('${seatName(seat)} plays ${cardText(card)}');
892 game.play(card);
893 case 'AUTO':
894 switch (args[0]) {
895 case 'PASS':
896 game.autoPass();
897 case 'CARD':
898 final actor = game.actor;
899 final card = game.choosePlay();
900 prefix.add('${seatName(actor)} plays ${cardText(card)}');
901 game.play(card);
902 case 'TRICK':
903 final count = round.tricks.length;
904 while (game.state == State.playing &&
905 game.round.tricks.length == count) {
906 game.play(game.choosePlay());
907 }
908 case 'HAND':
909 game.autoHand();
910 default:
911 game.autoGame();
912 }
913 case 'UNDO':
914 if (game.undoStack.isEmpty) fail('nothing to undo');
915 final undone = game.undoCard();
916 return ['undo ${seatName(undone.seat)} ${cardText(undone.card)}'];
917 case 'CLAIM':
918 if (seat != game.actor) fail('waiting for ${seatName(game.actor)}');
919 if (round.current.isNotEmpty) fail('trick in progress');
920 for (final card in round.hands[seat]) {
921 for (var other = 0; other < 4; other++) {
922 if (other != seat &&
923 round.hands[other].any(
924 (held) => held ~/ 13 == card ~/ 13 && held % 13 > card % 13,
925 )) {
926 fail('claim rejected');
927 }
928 }
929 }
930 game.claim(seat);
931 case 'LEGAL':
932 final bot = cardText(game.choosePlay());
933 return [
934 'legal ${seatName(game.actor)}: ${cardsText(game.legal())} bot $bot',
935 ];
936 case 'PREVIEW':
937 final output = <String>[];
938 for (final card in game.legal().toList()..sort()) {
939 final copy = game.copy();
940 final count = copy.round.tricks.length;
941 copy.play(card);
942 while (copy.state == State.playing &&
943 copy.round.tricks.length == count) {
944 copy.play(copy.choosePlay());
945 }
946 final trick = copy.round.tricks.last;
947 output.add(
948 'preview ${cardText(card)} -> ${seatName(trickWinner(trick))} '
949 '${copy.trickPoints(trick)}',
950 );
951 }
952 return output;
953 case 'SCORE':
954 return [
955 pairs('scores', game.scores),
956 if (game.state == State.playing)
957 pairs('taken', [
958 for (final cards in round.taken) game.points(cards),
959 ]),
960 ];
961 case 'TAKEN':
962 return [
963 for (var other = 0; other < 4; other++)
964 '${seatName(other)}: '
965 '${cardsText(round.taken[other].where((card) => game.point(card) != 0))}',
966 ];
967 case 'TRICKS':
968 final output = [
969 for (var i = 0; i < round.tricks.length; i++)
970 game.trickLine(round.tricks[i], i + 1),
971 if (round.current.isNotEmpty) 'current: ${playsText(round.current)}',
972 ];
973 return output.isEmpty ? ['no tricks'] : output;
974 case 'VOIDS':
975 final voids = List.generate(4, (_) => <int>{});
976 for (final trick in [
977 ...round.tricks,
978 if (round.current.isNotEmpty) round.current,
979 ]) {
980 final led = trick.first.card ~/ 13;
981 for (final play in trick) {
982 if (play.card ~/ 13 != led) voids[play.seat].add(led);
983 }
984 }
985 final texts = [
986 for (final missing in voids)
987 missing.isEmpty
988 ? '-'
989 : (missing.toList()..sort())
990 .map((suit) => suitLetters[suit])
991 .join(),
992 ];
993 return [pairs('voids', texts)];
994 case 'UNSEEN':
995 final actor = game.actor;
996 final played = {
997 for (final cards in round.taken) ...cards,
998 for (final play in round.current) play.card,
999 };
1000 final unseen = [
1001 for (var card = 0; card < 52; card++)
1002 if (!round.hands[actor].contains(card) && !played.contains(card))
1003 card,
1004 ];
1005 return ['unseen ${seatName(actor)}: ${cardsText(unseen)}'];
1006 case 'STATS':
1007 return game.statsLines();
1008 case 'HISTORY':
1009 final output = <String>[];
1010 for (final hand in game.history) {
1011 final moon = switch (hand.shooter) {
1012 final shooter? => ' moon ${seatName(shooter)}',
1013 null => '',
1014 };
1015 output.add(
1016 'hand ${hand.number} pass ${hand.direction.label} '
1017 '${pairs('points', hand.points)}$moon',
1018 );
1019 }
1020 return output.isEmpty ? ['no hands'] : output;
1021 case 'REPLAY':
1022 final hand = bounded(args[0], 1, game.history.length);
1023 return game.history[hand - 1].record;
1024 }
1025 return [...prefix, for (final line in game.log.skip(before)) line.text];
1026 }
1027}
