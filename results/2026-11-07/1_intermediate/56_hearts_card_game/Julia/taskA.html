1const SEATS = "NESW"
2const SUITS = "CDSH"
3const RANKS = "23456789TJQKA"
4const QS, KS, AS, JD = 36, 37, 38, 22
5const DIRECTIONS = Dict('L'=>"LEFT", 'R'=>"RIGHT", 'A'=>"ACROSS", 'H'=>"HOLD")
6const RULE_NAMES = ["TARGET", "LIMIT", "MOON", "JACK", "QSBREAK", "BLOOD", "PASS"]
7struct CommandError <: Exception
8 reason::String
9end
10fail(reason) = throw(CommandError(reason))
11seattext(s::Int) = string(SEATS[s])
12cardtext(c::Int) = string(RANKS[mod(c,13)+1], SUITS[div(c,13)+1])
13cardstext(cards) = isempty(cards) ? "-" : join(cardtext.(sort(collect(cards))), " ")
14rankkey(c::Int) = (mod(c,13), div(c,13))
15highkey(c::Int) = (-mod(c,13), div(c,13))
16lowest(cards) = first(sort(collect(cards); by=rankkey))
17highest(cards) = first(sort(collect(cards); by=highkey))
18penalty(c::Int) = c == QS || div(c,13) == 3
19nextseat(s::Int, offset::Int) = mod(s-1+offset,4)+1
20function seatvalue(word::String)
21 length(word) == 1 && first(word) in SEATS || fail("bad seat " * word)
22 return findfirst(==(first(word)), SEATS)
23end
24function cardvalue(word::String)
25 length(word) == 2 || fail("bad card " * word)
26 word[1] in RANKS && word[2] in SUITS || fail("bad card " * word)
27 return 13*(findfirst(==(word[2]),SUITS)-1)+findfirst(==(word[1]),RANKS)-1
28end
29function bounded(word::String, low::Int, high::Int)
30 value = tryparse(Int,word)
31 value !== nothing && low <= value <= high || fail("bad value " * word)
32 return value
33end
34pairs(label, values) = label * " " * join((seattext(s)*" "*string(values[s]) for s in 1:4), " ")

87 HandRecord[],Tuple{Int,String}[],String[],[Set{Int}() for _ in 1:4],

91point(g::Game,c::Int) = c == QS ? 13 : div(c,13) == 3 ? 1 : c == JD && g.rules["JACK"] == "ON" ? -10 : 0
92points(g::Game,cards) = sum((point(g,c) for c in cards); init=0)
93function record!(g::Game, level::Int, line::String)
94 push!(g.log,(level,line))
95 level > 0 && push!(g.record,line)
96end
97function beginplay!(g::Game)
98 g.state = "playing"
99 g.leader = findfirst(hand -> 0 in hand,g.hands)
100 record!(g,2,"lead "*seattext(g.leader))
101end
102function deal!(g::Game)
103 g.number += 1
104 cycle = g.rules["PASS"]
105 g.direction = cycle[mod(g.number-1,length(cycle))+1]
106 g.hands = [copy(get(g.stacked,s,Set{Int}())) for s in 1:4]
107 used = union(g.hands...)
108 deck = [c for c in 0:51 if !(c in used)]
109 for i in length(deck):-1:2
110 j = Int(mod(draw!(g.rng),i))+1
111 deck[i],deck[j] = deck[j],deck[i]
112 end
113 free = [s for s in 1:4 if !haskey(g.stacked,s)]
114 for (i,c) in enumerate(deck)
115 push!(g.hands[free[mod(i-1,length(free))+1]],c)
116 end
117 empty!(g.stacked)
118 empty!(g.passed)
119 g.taken = [Set{Int}() for _ in 1:4]
120 g.tricks = Vector{Tuple{Int,Int}}[]
121 g.counts = zeros(Int,4)
122 g.current = Tuple{Int,Int}[]
123 empty!(g.undo)
124 g.record = String[]
125 g.broken = false
126 record!(g,1,"hand $(g.number) pass $(DIRECTIONS[g.direction])")
127 for s in 1:4
128 record!(g,2,seattext(s)*": "*cardstext(g.hands[s]))
129 end
130 if g.direction == 'H'
131 beginplay!(g)
132 else
133 g.state = "passing"
134 end
135end
136function passcards!(g::Game,s::Int,cards::Vector{Int})
137 setdiff!(g.hands[s],cards)
138 g.passed[s] = copy(cards)
139 record!(g,2,seattext(s)*" passes "*cardstext(cards))
140 if length(g.passed) == 4
141 offset = Dict('L'=>1,'R'=>3,'A'=>2)[g.direction]
142 for source in 1:4
143 union!(g.hands[nextseat(source,offset)],g.passed[source])
144 end
145 beginplay!(g)
146 end
147end
148actor(g::Game) = nextseat(g.leader,length(g.current))
149function legal(g::Game)
150 hand = g.hands[actor(g)]
151 if isempty(g.current)
152 isempty(g.tricks) && return Set([0])
153 safe = Set(c for c in hand if div(c,13) != 3)
154 return !g.broken && !isempty(safe) ? safe : copy(hand)
155 end
156 led = div(g.current[1][2],13)
157 followers = Set(c for c in hand if div(c,13) == led)
158 !isempty(followers) && return followers
159 if isempty(g.tricks) && g.rules["BLOOD"] == "OFF"
160 safe = Set(c for c in hand if !penalty(c))
161 !isempty(safe) && return safe
162 end
163 return copy(hand)
164end
165function choosepass(g::Game,s::Int)
166 hand, strategy = g.hands[s], g.bots[s]
167 low = sort(collect(hand); by=rankkey)
168 high = sort(collect(hand); by=highkey)
169 hearts = [c for c in high if div(c,13) == 3]
170 special = [c for c in (QS,KS,AS) if c in hand]
171 order = Int[]
172 if strategy == "LOW"
173 order = high
174 elseif strategy == "HIGH"
175 order = low
176 elseif strategy == "MOON"
177 order = vcat([c for c in low if div(c,13) != 3],[c for c in low if div(c,13) == 3])
178 elseif strategy == "SMART"
179 protected = QS in hand && count(c -> div(c,13) == 2,hand) >= 5
180 !protected && append!(order,special)
181 available = [suit for suit in 0:1 if any(c -> div(c,13) == suit,hand)]
182 if !isempty(available)
183 suit = first(sort(available; by=t -> (count(c -> div(c,13) == t,hand),t)))
184 append!(order,[c for c in high if div(c,13) == suit])
185 end
186 append!(order,hearts)
187 append!(order,high)
188 else
189 order = vcat(special,hearts,high)
190 end
191 return unique(order)[1:3]
192end
193queenplayed(g::Game) = any(cards -> QS in cards,g.taken) || any(p -> p[2] == QS,g.current)
194function chooseplay(g::Game)
195 s = actor(g)
196 hand, playable, strategy = g.hands[s], legal(g), g.bots[s]
197 strategy == "LOW" && return lowest(playable)
198 strategy == "HIGH" && return highest(playable)
199 if strategy == "MOON" && any(other -> other != s && any(penalty,g.taken[other]),1:4)
200 strategy = "BASIC"
201 end
202 if isempty(g.current)
203 if strategy == "SMART" && !queenplayed(g) && !(QS in hand)
204 small = Set(c for c in playable if div(c,13) == 2 && mod(c,13) < 10)
205 !isempty(small) && return lowest(small)
206 end
207 available = unique([div(c,13) for c in playable])
208 suit = first(sort(available; by=t -> ((strategy == "MOON" ? -1 : 1)*count(c -> div(c,13) == t,playable),t)))
209 subset = Set(c for c in playable if div(c,13) == suit)
210 return strategy == "MOON" ? highest(subset) : lowest(subset)
211 end
212 led = div(g.current[1][2],13)
213 if any(c -> div(c,13) == led,hand)
214 winning = maximum(mod(c,13) for (_,c) in g.current if div(c,13) == led)
215 lower = Set(c for c in playable if mod(c,13) < winning)
216 if strategy == "MOON"
217 high = highest(playable)
218 return mod(high,13) > winning ? high : lowest(playable)
219 end
220 !isempty(lower) && return highest(lower)
221 if strategy == "SMART" && length(g.current) == 3
222 safe = setdiff(playable,Set([QS]))
223 return isempty(safe) ? QS : highest(safe)
224 end
225 return lowest(playable)
226 end
227 if strategy == "MOON"
228 safe = Set(c for c in playable if !penalty(c))
229 return lowest(isempty(safe) ? playable : safe)
230 end
231 QS in playable && return QS
232 if strategy == "SMART" && !queenplayed(g)
233 AS in playable && return AS
234 KS in playable && return KS
235 end
236 hearts = Set(c for c in playable if div(c,13) == 3)
237 return highest(isempty(hearts) ? playable : hearts)
238end
239function winner(plays::Vector{Tuple{Int,Int}})
240 best = plays[1]
241 for p in plays
242 if div(p[2],13) == div(best[2],13) && mod(p[2],13) > mod(best[2],13)
243 best = p
244 end
245 end
246 return best[1]
247end
248playstext(plays) = join((seattext(s)*" "*cardtext(c) for (s,c) in plays)," ")
249function trickline(g::Game,plays,number::Int)
250 value = sum(point(g,c) for (_,c) in plays)
251 return "trick $number: $(playstext(plays)) -> $(seattext(winner(plays))) $value"
252end
253function play!(g::Game,c::Int)
254 s = actor(g)
255 push!(g.undo,Snapshot(deepcopy(g.hands),deepcopy(g.taken),copy(g.tricks),copy(g.counts),
256 copy(g.current),g.leader,g.broken,copy(g.record),length(g.log),s,c))
257 delete!(g.hands[s],c)
258 push!(g.current,(s,c))
259 if div(c,13) == 3 || c == QS && g.rules["QSBREAK"] == "ON"
260 g.broken = true
261 end
262 if length(g.current) == 4
263 w = winner(g.current)
264 union!(g.taken[w],(card for (_,card) in g.current))
265 g.counts[w] += 1
266 push!(g.tricks,g.current)
267 record!(g,2,trickline(g,g.current,length(g.tricks)))
268 g.current = Tuple{Int,Int}[]
269 g.leader = w
270 length(g.tricks) == 13 && finishhand!(g)
271 end
272end
273function undocard!(g::Game)
274 u = pop!(g.undo)
275 g.hands,g.taken,g.tricks,g.counts,g.current = u.hands,u.taken,u.tricks,u.counts,u.current
276 g.leader,g.broken,g.record = u.leader,u.broken,u.record
277 resize!(g.log,u.logsize)
278 return u.seat,u.card
279end
280ranking(g::Game) = sort(collect(1:4); by=s -> (g.scores[s],s))
281ended(g::Game,scores) = g.number >= parse(Int,g.rules["LIMIT"]) || maximum(scores) >= parse(Int,g.rules["TARGET"])
282function statslines(g::Game)
283 ["stats $(seattext(s)) tricks $(g.stats[s,1]) hearts $(g.stats[s,2]) queens $(g.stats[s,3]) moons $(g.stats[s,4])" for s in 1:4]
284end
285function finishhand!(g::Game)
286 values = [points(g,cards) for cards in g.taken]
287 shooter = something(findfirst(cards -> count(penalty,cards) == 14,g.taken),0)
288 if shooter != 0
289 values[shooter] -= 26
290 method = g.rules["MOON"]
291 if method == "AUTO"
292 addition = [g.scores[s]+values[s]+(s == shooter ? 0 : 26) for s in 1:4]
293 best = first(sort(collect(1:4); by=s -> (addition[s],s)))
294 method = ended(g,addition) && best != shooter ? "SUB" : "ADD"
295 end
296 if method == "SUB"
297 values[shooter] -= 26
298 else
299 for s in 1:4
300 s != shooter && (values[s] += 26)
301 end
302 end
303 record!(g,1,"moon $(seattext(shooter)) $method")
304 end
305 for s in 1:4
306 g.scores[s] += values[s]
307 g.stats[s,1] += g.counts[s]
308 g.stats[s,2] += count(c -> div(c,13) == 3,g.taken[s])
309 g.stats[s,3] += QS in g.taken[s]
310 g.stats[s,4] += shooter == s
311 end
312 record!(g,1,pairs("points",values))
313 record!(g,1,pairs("scores",g.scores))
314 push!(g.history,HandRecord(g.number,DIRECTIONS[g.direction],values,shooter,copy(g.record)))
315 g.state = ended(g,g.scores) ? "none" : "between"
316 if g.state == "none"
317 record!(g,0,"game over hands $(g.number)")
318 for (place,s) in enumerate(ranking(g))
319 record!(g,0,"rank $place $(seattext(s)) $(g.scores[s])")
320 end
321 for line in statslines(g)
322 record!(g,0,line)
323 end
324 end
325end
326function autopass!(g::Game)
327 for s in 1:4
328 !haskey(g.passed,s) && passcards!(g,s,choosepass(g,s))
329 end
330end
331function autohand!(g::Game)
332 g.state == "passing" && autopass!(g)
333 while g.state == "playing"
334 play!(g,chooseplay(g))
335 end
336end
337function autogame!(g::Game)
338 g.state in ("passing","playing") && autohand!(g)
339 while g.state != "none"
340 deal!(g)
341 autohand!(g)
342 end
343end
344function duplicate(cards)
345 seen = Set{Int}()
346 for c in cards
347 c in seen && fail("duplicate card "*cardtext(c))
348 push!(seen,c)
349 end

356active(t::Table) = t.game !== nothing && t.game.state != "none"
357function requirestate(t::Table,state::String)
358 if state == "game"
359 active(t) || fail("no game")
360 elseif state == "between"
361 requirestate(t,"game")
362 t.game.state == "between" || fail("hand in progress")
363 elseif state == "hand"
364 active(t) && t.game.state in ("passing","playing") || fail("no hand")
365 else
366 active(t) && t.game.state == state || fail("not "*state)
367 end
368end
369function simulate(t::Table,cmd::String,seed::Int,repetitions::Int)
370 rng = Generator(seed)
371 if cmd == "GAME"
372 g = Game(t.rules,t.bots,rng)
373 autogame!(g)
374 return vcat(["game seed $seed"],[line for (level,line) in g.log if level <= repetitions])
375 end
376 rounds = cmd == "ROTATE" ? 4 : 1
377 places, totals, statistics = zeros(Int,4,4),zeros(Int,4),zeros(Int,4,4)
378 hands = 0
379 for rotation in 0:rounds-1
380 bots = Vector{String}(undef,4)
381 for s in 1:4
382 bots[nextseat(s,rotation)] = t.bots[s]
383 end
384 for _ in 1:repetitions
385 g = Game(t.rules,bots,rng)
386 autogame!(g)
387 hands += g.number
388 order = ranking(g)
389 for s in 1:4
390 actual = nextseat(s,rotation)
391 places[s,findfirst(==(actual),order)] += 1
392 totals[s] += g.scores[actual]
393 statistics[s,:] .+= g.stats[actual,:]
394 end
395 end
396 end
397 games = rounds*repetitions
398 output = ["$(lowercase(cmd)) seed $seed games $games hands $hands"]
399 for s in 1:4
400 cents = fld(100*totals[s],games)
401 magnitude = abs(cents)
402 average = (cents < 0 ? "-" : "")*string(div(magnitude,100))*"."*lpad(string(mod(magnitude,100)),2,'0')
403 name = seattext(s)*(cmd == "ROTATE" ? " "*t.bots[s] : "")
404 push!(output,"$name places $(join(places[s,:]," ")) total $(totals[s]) average $average tricks $(statistics[s,1]) hearts $(statistics[s,2]) queens $(statistics[s,3]) moons $(statistics[s,4])")
405 end
406 return output
407end
408function command!(t::Table,words::Vector{String})
409 cmd,args = words[1],words[2:end]
410 known = split("RULE BOT SETTINGS GAME MATCH ROTATE NEW ABANDON STACK DEAL SHOW PASS PLAY AUTO UNDO CLAIM LEGAL PREVIEW SCORE TAKEN TRICKS VOIDS UNSEEN STATS HISTORY REPLAY")
411 cmd in known || fail("unknown command "*cmd)
412 if cmd == "RULE"
413 name,value = args
414 name in RULE_NAMES || fail("unknown rule "*name)
415 if name in ("TARGET","LIMIT")
416 bounded(value,1,name == "TARGET" ? 1000 : 100)
417 else
418 good = name == "PASS" ? 1 <= length(value) <= 8 && all(c -> c in "LRAH",value) :
419 name == "MOON" ? value in ("ADD","SUB","AUTO") : value in ("ON","OFF")
420 good || fail("bad value "*value)
421 end
422 active(t) && fail("game in progress")
423 t.rules[name] = value
424 return ["rule $name $value"]
425 elseif cmd == "BOT"
426 s = seatvalue(args[1])
427 args[2] in ("BASIC","LOW","HIGH","SMART","MOON") || fail("bad value "*args[2])
428 t.bots[s] = args[2]
429 return ["bot $(seattext(s)) $(args[2])"]
430 elseif cmd == "SETTINGS"
431 return ["rules "*join((name*" "*t.rules[name] for name in RULE_NAMES)," "),pairs("bots",t.bots)]
432 elseif cmd in ("GAME","MATCH","ROTATE")
433 seed = bounded(args[1],1,2147483646)
434 repetitions = bounded(args[2],cmd == "GAME" ? 0 : 1,cmd == "GAME" ? 2 : cmd == "MATCH" ? 50 : 10)
435 return simulate(t,cmd,seed,repetitions)
436 elseif cmd == "NEW"
437 seed = bounded(args[1],1,2147483646)
438 active(t) && fail("game in progress")
439 t.game = Game(t.rules,t.bots,Generator(seed))
440 return ["new game seed $seed"]
441 elseif cmd == "ABANDON"
442 requirestate(t,"game")
443 t.game = nothing
444 return ["game abandoned"]
445 end
446 s = cmd in ("STACK","PASS","PLAY","CLAIM") ? seatvalue(args[1]) : 0
447 cards = cmd in ("STACK","PASS","PLAY") ? cardvalue.(args[2:end]) : Int[]
448 cmd == "STACK" && length(cards) != 13 && fail("need 13 cards")
449 cmd == "AUTO" && !(args[1] in ("PASS","CARD","TRICK","HAND","GAME")) && fail("bad value "*args[1])
450 if cmd in ("STACK","DEAL")
451 requirestate(t,"between")
452 elseif cmd == "SHOW"
453 requirestate(t,"hand")
454 elseif cmd == "PASS"
455 requirestate(t,"passing")
456 elseif cmd in ("PLAY","UNDO","CLAIM","LEGAL","PREVIEW","TAKEN","TRICKS","VOIDS","UNSEEN")
457 requirestate(t,"playing")
458 elseif cmd in ("SCORE","STATS","HISTORY","REPLAY")
459 requirestate(t,"game")
460 elseif cmd == "AUTO"
461 mode = args[1]
462 requirestate(t,mode == "PASS" ? "passing" : mode in ("CARD","TRICK") ? "playing" : mode == "HAND" ? "hand" : "game")
463 end
464 g = t.game
465 before = length(g.log)
466 output = String[]
467 if cmd == "STACK"
468 duplicate(cards)
469 for c in cards, other in 1:4
470 other != s && c in get(g.stacked,other,Set{Int}()) && fail(cardtext(c)*" stacked for "*seattext(other))
471 end
472 g.stacked[s] = Set(cards)
473 return ["stacked "*seattext(s)]
474 elseif cmd == "DEAL"
475 deal!(g)
476 elseif cmd == "SHOW"
477 return [seattext(s)*": "*cardstext(g.hands[s]) for s in 1:4]
478 elseif cmd == "PASS"
479 haskey(g.passed,s) && fail(seattext(s)*" already passed")
480 duplicate(cards)
481 for c in cards
482 c in g.hands[s] || fail(seattext(s)*" does not hold "*cardtext(c))
483 end
484 passcards!(g,s,cards)
485 elseif cmd == "PLAY"
486 c = cards[1]
487 s == actor(g) || fail("waiting for "*seattext(actor(g)))
488 c in g.hands[s] || fail(seattext(s)*" does not hold "*cardtext(c))
489 if !(c in legal(g))
490 isempty(g.current) && fail(isempty(g.tricks) ? "must lead 2C" : "hearts not broken")
491 led = div(g.current[1][2],13)
492 any(held -> div(held,13) == led,g.hands[s]) && fail("must follow "*string(SUITS[led+1]))
493 fail("no points on first trick")
494 end
495 push!(output,seattext(s)*" plays "*cardtext(c))
496 play!(g,c)
497 elseif cmd == "AUTO"
498 mode = args[1]
499 if mode == "PASS"
500 autopass!(g)
501 elseif mode == "CARD"
502 s,c = actor(g),chooseplay(g)
503 push!(output,seattext(s)*" plays "*cardtext(c))
504 play!(g,c)
505 elseif mode == "TRICK"
506 completed = length(g.tricks)
507 while g.state == "playing" && length(g.tricks) == completed
508 play!(g,chooseplay(g))
509 end
510 elseif mode == "HAND"
511 autohand!(g)
512 else
513 autogame!(g)
514 end
515 elseif cmd == "UNDO"
516 isempty(g.undo) && fail("nothing to undo")
517 s,c = undocard!(g)
518 return ["undo $(seattext(s)) $(cardtext(c))"]
519 elseif cmd == "CLAIM"
520 s == actor(g) || fail("waiting for "*seattext(actor(g)))
521 isempty(g.current) || fail("trick in progress")
522 for c in g.hands[s], other in 1:4
523 if other != s && any(d -> div(d,13) == div(c,13) && mod(d,13) > mod(c,13),g.hands[other])
524 fail("claim rejected")
525 end
526 end
527 remaining = union(g.hands...)
528 countleft = 13-length(g.tricks)
529 union!(g.taken[s],remaining)
530 g.counts[s] += countleft
531 g.hands = [Set{Int}() for _ in 1:4]
532 record!(g,2,"claim $(seattext(s)) tricks $countleft points $(points(g,remaining))")
533 finishhand!(g)
534 elseif cmd == "LEGAL"
535 return ["legal $(seattext(actor(g))): $(cardstext(legal(g))) bot $(cardtext(chooseplay(g)))"]
536 elseif cmd == "PREVIEW"
537 for c in sort(collect(legal(g)))
538 trial = deepcopy(g)
539 completed = length(trial.tricks)
540 play!(trial,c)
541 while trial.state == "playing" && length(trial.tricks) == completed
542 play!(trial,chooseplay(trial))
543 end
544 trick = trial.tricks[end]
545 value = sum(point(g,card) for (_,card) in trick)
546 push!(output,"preview $(cardtext(c)) -> $(seattext(winner(trick))) $value")
547 end
548 return output
549 elseif cmd == "SCORE"
550 output = [pairs("scores",g.scores)]
551 g.state == "playing" && push!(output,pairs("taken",[points(g,cards) for cards in g.taken]))
552 return output
553 elseif cmd == "TAKEN"
554 return [seattext(s)*": "*cardstext(Set(c for c in g.taken[s] if point(g,c) != 0)) for s in 1:4]
555 elseif cmd == "TRICKS"
556 output = [trickline(g,trick,n) for (n,trick) in enumerate(g.tricks)]
557 !isempty(g.current) && push!(output,"current: "*playstext(g.current))
558 return isempty(output) ? ["no tricks"] : output
559 elseif cmd == "VOIDS"
560 missing = [Set{Int}() for _ in 1:4]
561 for trick in vcat(g.tricks,[g.current])
562 isempty(trick) && continue
563 led = div(trick[1][2],13)
564 for (who,c) in trick
565 div(c,13) != led && push!(missing[who],led)
566 end
567 end
568 values = [isempty(item) ? "-" : join((SUITS[suit+1] for suit in sort(collect(item)))) for item in missing]
569 return [pairs("voids",values)]
570 elseif cmd == "UNSEEN"
571 who = actor(g)
572 unseen = setdiff(Set(0:51),g.hands[who],union(g.taken...),Set(c for (_,c) in g.current))
573 return ["unseen $(seattext(who)): $(cardstext(unseen))"]
574 elseif cmd == "STATS"
575 return statslines(g)
576 elseif cmd == "HISTORY"
577 for h in g.history
578 line = "hand $(h.number) pass $(h.direction) "*pairs("points",h.points)
579 h.shooter != 0 && (line *= " moon "*seattext(h.shooter))
580 push!(output,line)
581 end
582 return isempty(output) ? ["no hands"] : output
583 elseif cmd == "REPLAY"
584 h = bounded(args[1],1,length(g.history))
585 return copy(g.history[h].record)
586 end
587 append!(output,[line for (_,line) in g.log[before+1:end]])
588 return output
589end
