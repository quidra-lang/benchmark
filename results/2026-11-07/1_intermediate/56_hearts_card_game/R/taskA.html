75deal <- function(g) {
76 g$number <- g$number + 1L
77 cycle <- g$rules[["PASS"]]
78 position <- ((g$number - 1L) %% nchar(cycle)) + 1L
79 g$direction <- substr(cycle, position, position)
80 g$hands <- lapply(g$stacked, function(cards) if (is.null(cards)) integer() else cards)
81 deck <- setdiff(0:51, unlist(g$hands, use.names = FALSE))
82 if (length(deck) > 1L) {
83 for (i in seq.int(length(deck), 2L)) {
84 j <- (draw(g$rng) %% i) + 1L
85 tmp <- deck[i]
86 deck[i] <- deck[j]
87 deck[j] <- tmp
88 }
89 }
90 free <- which(vapply(g$stacked, is.null, logical(1)))
91 for (i in seq_along(deck)) {
92 seat <- free[((i - 1L) %% length(free)) + 1L]
93 g$hands[[seat]] <- c(g$hands[[seat]], deck[i])
94 }
95 g$stacked <- vector("list", 4)
96 g$passed <- rep(FALSE, 4)
97 g$passes <- vector("list", 4)
98 g$taken <- replicate(4, integer(), simplify = FALSE)
99 g$tricks <- list()
100 g$counts <- integer(4)
101 g$current <- list()
102 g$broken <- FALSE
103 g$undo <- list()
104 g$record <- character()
105 record_line(g, 1L, paste("hand", g$number, "pass", directions[[g$direction]]))
106 for (seat in 1:4) record_line(g, 2L, paste0(seats[seat], ": ", cards_text(g$hands[[seat]])))
107 if (g$direction == "H") begin_play(g) else g$state <- "passing"
108}
109pass_cards <- function(g, seat, cards) {
110 g$hands[[seat]] <- setdiff(g$hands[[seat]], cards)
111 g$passed[seat] <- TRUE
112 g$passes[[seat]] <- cards
113 record_line(g, 2L, paste(seats[seat], "passes", cards_text(cards)))
114 if (all(g$passed)) {
115 offset <- c(L = 1L, R = 3L, A = 2L)[[g$direction]]
116 for (source in 1:4) {
117 target <- next_seat(source, offset)
118 g$hands[[target]] <- c(g$hands[[target]], g$passes[[source]])
119 }
120 begin_play(g)
121 }

123actor <- function(g) next_seat(g$leader, length(g$current))
124legal <- function(g) {
125 hand <- g$hands[[actor(g)]]
126 if (!length(g$current)) {
127 if (!length(g$tricks)) return(0L)
128 safe <- hand[(hand %/% 13L) != 3L]
129 return(if (!g$broken && length(safe)) safe else hand)
130 }
131 led <- g$current[[1]]$card %/% 13L
132 followers <- hand[(hand %/% 13L) == led]
133 if (length(followers)) return(followers)
134 if (!length(g$tricks) && g$rules[["BLOOD"]] == "OFF") {
135 safe <- hand[!penalty(hand)]
136 if (length(safe)) return(safe)
137 }
138 hand

140choose_pass <- function(g, seat) {
141 hand <- g$hands[[seat]]
142 strategy <- g$bots$values[seat]
143 low <- ordered(hand)
144 high <- ordered(hand, TRUE)
145 hearts <- high[(high %/% 13L) == 3L]
146 special <- c(QS, KS, AS)
147 special <- special[special %in% hand]
148 if (strategy == "LOW") {
149 priority <- high
150 } else if (strategy == "HIGH") {
151 priority <- low
152 } else if (strategy == "MOON") {
153 priority <- c(low[(low %/% 13L) != 3L], low[(low %/% 13L) == 3L])
154 } else if (strategy == "SMART") {
155 protected <- QS %in% hand && sum((hand %/% 13L) == 2L) >= 5L
156 priority <- if (protected) integer() else special
157 counts <- vapply(0:1, function(suit) sum((hand %/% 13L) == suit), integer(1))
158 available <- which(counts > 0L)
159 if (length(available)) {
160 suit <- available[order(counts[available], available)][1] - 1L
161 priority <- c(priority, high[(high %/% 13L) == suit])
162 }
163 priority <- c(priority, hearts, high)
164 } else {
165 priority <- c(special, hearts, high)
166 }
167 unique(priority)[1:3]
168}
169queen_played <- function(g) {
170 any(vapply(g$taken, function(cards) QS %in% cards, logical(1))) ||
171 any(vapply(g$current, function(play) play$card == QS, logical(1)))
172}
173choose_play <- function(g) {
174 seat <- actor(g)
175 hand <- g$hands[[seat]]
176 playable <- legal(g)
177 strategy <- g$bots$values[seat]
178 if (strategy == "LOW") return(lowest(playable))
179 if (strategy == "HIGH") return(highest(playable))
180 if (strategy == "MOON" && any(vapply(setdiff(1:4, seat), function(other) any(penalty(g$taken[[other]])), logical(1)))) {
181 strategy <- "BASIC"
182 }
183 if (!length(g$current)) {
184 if (strategy == "SMART" && !queen_played(g) && !(QS %in% hand)) {
185 small <- playable[(playable %/% 13L) == 2L & (playable %% 13L) < 10L]
186 if (length(small)) return(lowest(small))
187 }
188 available <- sort(unique(playable %/% 13L))
189 counts <- vapply(available, function(suit) sum((playable %/% 13L) == suit), integer(1))
190 suit <- available[order(if (strategy == "MOON") -counts else counts, available)][1]
191 subset <- playable[(playable %/% 13L) == suit]
192 return(if (strategy == "MOON") highest(subset) else lowest(subset))
193 }
194 led <- g$current[[1]]$card %/% 13L
195 if (any((hand %/% 13L) == led)) {
196 played <- vapply(g$current, function(play) play$card, integer(1))
197 winning <- max((played[played %/% 13L == led]) %% 13L)
198 lower <- playable[(playable %% 13L) < winning]
199 if (strategy == "MOON") {
200 high <- highest(playable)
201 return(if ((high %% 13L) > winning) high else lowest(playable))
202 }
203 if (length(lower)) return(highest(lower))
204 if (strategy == "SMART" && length(g$current) == 3L) {
205 safe <- playable[playable != QS]
206 return(if (length(safe)) highest(safe) else QS)
207 }
208 return(lowest(playable))
209 }
210 if (strategy == "MOON") {
211 safe <- playable[!penalty(playable)]
212 return(lowest(if (length(safe)) safe else playable))
213 }
214 if (QS %in% playable) return(QS)
215 if (strategy == "SMART" && !queen_played(g)) {
216 if (AS %in% playable) return(AS)
217 if (KS %in% playable) return(KS)
218 }
219 hearts <- playable[(playable %/% 13L) == 3L]
220 highest(if (length(hearts)) hearts else playable)
221}
222winner <- function(plays) {
223 led <- plays[[1]]$card %/% 13L
224 candidates <- Filter(function(play) (play$card %/% 13L) == led, plays)
225 candidates[[which.max(vapply(candidates, function(play) play$card %% 13L, integer(1)))]]$seat
226}

232play_card <- function(g, card) {
233 seat <- actor(g)
234 fields <- c("hands", "taken", "tricks", "counts", "current", "leader", "broken", "record")
235 snapshot <- lapply(fields, function(name) g[[name]])
236 names(snapshot) <- fields
237 snapshot$log_size <- length(g$log)
238 snapshot$seat <- seat
239 snapshot$card <- card
240 g$undo[[length(g$undo) + 1L]] <- snapshot
241 g$hands[[seat]] <- setdiff(g$hands[[seat]], card)
242 g$current[[length(g$current) + 1L]] <- list(seat = seat, card = card)
243 if ((card %/% 13L) == 3L || card == QS && g$rules[["QSBREAK"]] == "ON") g$broken <- TRUE
244 if (length(g$current) == 4L) {
245 who <- winner(g$current)
246 g$taken[[who]] <- c(g$taken[[who]], vapply(g$current, function(play) play$card, integer(1)))
247 g$counts[who] <- g$counts[who] + 1L
248 g$tricks[[length(g$tricks) + 1L]] <- g$current
249 record_line(g, 2L, trick_line(g, g$current, length(g$tricks)))
250 g$current <- list()
251 g$leader <- who
252 if (length(g$tricks) == 13L) finish_hand(g)
253 }

260 g[[name]] <- snapshot[[name]]

263 c(snapshot$seat, snapshot$card)

270finish_hand <- function(g) {
271 values <- vapply(g$taken, function(cards) points(g, cards), numeric(1))
272 shooters <- which(vapply(g$taken, function(cards) sum(penalty(cards)) == 14L, logical(1)))
273 shooter <- if (length(shooters)) shooters[1] else 0L
274 if (shooter != 0L) {
275 values[shooter] <- values[shooter] - 26L
276 method <- g$rules[["MOON"]]
277 if (method == "AUTO") {
278 addition <- g$scores + values + ifelse(1:4 == shooter, 0L, 26L)
279 method <- if (ended(g, addition) && order(addition, 1:4)[1] != shooter) "SUB" else "ADD"
280 }
281 if (method == "SUB") values[shooter] <- values[shooter] - 26L else values[-shooter] <- values[-shooter] + 26L
282 record_line(g, 1L, paste("moon", seats[shooter], method))
283 }
284 for (seat in 1:4) {
285 g$scores[seat] <- g$scores[seat] + values[seat]
286 g$stats[seat, ] <- g$stats[seat, ] + c(g$counts[seat], sum((g$taken[[seat]] %/% 13L) == 3L), QS %in% g$taken[[seat]], shooter == seat)
287 }
288 record_line(g, 1L, pairs("points", values))
289 record_line(g, 1L, pairs("scores", g$scores))
290 g$history[[length(g$history) + 1L]] <- list(number = g$number, direction = directions[[g$direction]], points = values, shooter = shooter, record = g$record)
291 g$state <- if (ended(g, g$scores)) "none" else "between"
292 if (g$state == "none") {
293 record_line(g, 0L, paste("game over hands", g$number))
294 order_seats <- ranking(g)
295 for (place in 1:4) record_line(g, 0L, paste("rank", place, seats[order_seats[place]], g$scores[order_seats[place]]))
296 for (line in stats_lines(g)) record_line(g, 0L, line)
297 }
298}

302auto_hand <- function(g) {

310 auto_hand(g)
311 }

321require_state <- function(table, state) {
322 if (state == "game") {
323 if (!active(table)) fail("no game")

327 } else if (state == "hand") {

333simulate <- function(table, command, seed, repetitions) {
334 rng <- new_generator(seed)
335 if (command == "GAME") {
336 g <- new_game(table$rules, table$bots, rng)
337 auto_game(g)
338 selected <- Filter(function(entry) entry$level <= repetitions, g$log)
339 return(c(paste("game seed", seed), vapply(selected, function(entry) entry$text, character(1))))
340 }
341 rounds <- if (command == "ROTATE") 4L else 1L
342 places <- matrix(0, 4, 4)
343 totals <- numeric(4)
344 statistics <- matrix(0, 4, 4)
345 hands <- 0L
346 for (rotation in seq_len(rounds) - 1L) {
347 bots <- new.env(parent = emptyenv())
348 bots$values <- character(4)
349 for (seat in 1:4) bots$values[next_seat(seat, rotation)] <- table$bots$values[seat]
350 for (i in seq_len(repetitions)) {
351 g <- new_game(table$rules, bots, rng)
352 auto_game(g)
353 hands <- hands + g$number
354 order_seats <- ranking(g)
355 for (seat in 1:4) {
356 actual <- next_seat(seat, rotation)
357 place <- match(actual, order_seats)
358 places[seat, place] <- places[seat, place] + 1L
359 totals[seat] <- totals[seat] + g$scores[actual]
360 statistics[seat, ] <- statistics[seat, ] + g$stats[actual, ]
361 }
362 }
363 }
364 games <- rounds * repetitions
365 output <- paste(tolower(command), "seed", seed, "games", games, "hands", hands)
366 for (seat in 1:4) {
367 cents <- floor(100 * totals[seat] / games)
368 magnitude <- abs(cents)
369 average <- paste0(if (cents < 0) "-" else "", magnitude %/% 100, ".", sprintf("%02d", as.integer(magnitude %% 100)))
370 name <- if (command == "ROTATE") paste(seats[seat], table$bots$values[seat]) else seats[seat]
371 output <- c(output, paste(name, "places", paste(places[seat, ], collapse = " "), "total", totals[seat], "average", average,
372 "tricks", statistics[seat, 1], "hearts", statistics[seat, 2], "queens", statistics[seat, 3], "moons", statistics[seat, 4]))
373 }
374 output
375}
376command <- function(table, words) {
377 cmd <- words[1]
378 args <- words[-1]
379 known <- strsplit("RULE BOT SETTINGS GAME MATCH ROTATE NEW ABANDON STACK DEAL SHOW PASS PLAY AUTO UNDO CLAIM LEGAL PREVIEW SCORE TAKEN TRICKS VOIDS UNSEEN STATS HISTORY REPLAY", " ", fixed = TRUE)[[1]]
380 if (!(cmd %in% known)) fail(paste("unknown command", cmd))
381 if (cmd == "RULE") {
382 name <- args[1]
383 value <- args[2]
384 if (!(name %in% rule_names)) fail(paste("unknown rule", name))
385 if (name %in% c("TARGET", "LIMIT")) {
386 bounded(value, 1, if (name == "TARGET") 1000 else 100)
387 } else {
388 good <- if (name == "PASS") nchar(value) <= 8L && grepl("^[LRAH]+$", value) else if (name == "MOON") value %in% c("ADD", "SUB", "AUTO") else value %in% c("ON", "OFF")
389 if (!good) fail(paste("bad value", value))
390 }
391 if (active(table)) fail("game in progress")
392 table$rules[[name]] <- value
393 return(paste("rule", name, value))
394 }
395 if (cmd == "BOT") {
396 seat <- seat_value(args[1])
397 if (!(args[2] %in% c("BASIC", "LOW", "HIGH", "SMART", "MOON"))) fail(paste("bad value", args[2]))
398 table$bots$values[seat] <- args[2]
399 return(paste("bot", seats[seat], args[2]))
400 }
401 if (cmd == "SETTINGS") return(c(paste("rules", paste(rule_names, unlist(table$rules[rule_names]), collapse = " ")), pairs("bots", table$bots$values)))
402 if (cmd %in% c("GAME", "MATCH", "ROTATE")) {
403 seed <- bounded(args[1], 1, 2147483646)
404 repetitions <- bounded(args[2], if (cmd == "GAME") 0 else 1, if (cmd == "GAME") 2 else if (cmd == "MATCH") 50 else 10)
405 return(simulate(table, cmd, seed, repetitions))
406 }
407 if (cmd == "NEW") {
408 seed <- bounded(args[1], 1, 2147483646)
409 if (active(table)) fail("game in progress")
410 table$game <- new_game(table$rules, table$bots, new_generator(seed))
411 return(paste("new game seed", seed))
412 }
413 if (cmd == "ABANDON") {
414 require_state(table, "game")
415 table$game <- NULL
416 return("game abandoned")
417 }
418 seat <- if (cmd %in% c("STACK", "PASS", "PLAY", "CLAIM")) seat_value(args[1]) else 0L
419 cards <- if (cmd %in% c("STACK", "PASS", "PLAY")) vapply(args[-1], card_value, integer(1)) else integer()
420 if (cmd == "STACK" && length(cards) != 13L) fail("need 13 cards")
421 if (cmd == "AUTO" && !(args[1] %in% c("PASS", "CARD", "TRICK", "HAND", "GAME"))) fail(paste("bad value", args[1]))
422 if (cmd %in% c("STACK", "DEAL")) {
423 require_state(table, "between")
424 } else if (cmd == "SHOW") {
425 require_state(table, "hand")
426 } else if (cmd == "PASS") {
427 require_state(table, "passing")
428 } else if (cmd %in% c("PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS", "VOIDS", "UNSEEN")) {
429 require_state(table, "playing")
430 } else if (cmd %in% c("SCORE", "STATS", "HISTORY", "REPLAY")) {
431 require_state(table, "game")
432 } else if (cmd == "AUTO") {
433 mode <- args[1]
434 require_state(table, if (mode == "PASS") "passing" else if (mode %in% c("CARD", "TRICK")) "playing" else if (mode == "HAND") "hand" else "game")
435 }
436 g <- table$game
437 before <- length(g$log)
438 output <- character()
439 if (cmd == "STACK") {
440 duplicate(cards)
441 for (card in cards) for (other in setdiff(1:4, seat)) {
442 if (card %in% g$stacked[[other]]) fail(paste(card_text(card), "stacked for", seats[other]))
443 }
444 g$stacked[[seat]] <- cards
445 return(paste("stacked", seats[seat]))
446 } else if (cmd == "DEAL") {
447 deal(g)
448 } else if (cmd == "SHOW") {
449 return(vapply(1:4, function(seat) paste0(seats[seat], ": ", cards_text(g$hands[[seat]])), character(1)))
450 } else if (cmd == "PASS") {
451 if (g$passed[seat]) fail(paste(seats[seat], "already passed"))
452 duplicate(cards)
453 for (card in cards) if (!(card %in% g$hands[[seat]])) fail(paste(seats[seat], "does not hold", card_text(card)))
454 pass_cards(g, seat, cards)
455 } else if (cmd == "PLAY") {
456 card <- cards[1]
457 if (seat != actor(g)) fail(paste("waiting for", seats[actor(g)]))
458 if (!(card %in% g$hands[[seat]])) fail(paste(seats[seat], "does not hold", card_text(card)))
459 if (!(card %in% legal(g))) {
460 if (!length(g$current)) fail(if (!length(g$tricks)) "must lead 2C" else "hearts not broken")
461 led <- g$current[[1]]$card %/% 13L
462 if (any((g$hands[[seat]] %/% 13L) == led)) fail(paste("must follow", suits[led + 1L]))
463 fail("no points on first trick")
464 }
465 output <- paste(seats[seat], "plays", card_text(card))
466 play_card(g, card)
467 } else if (cmd == "AUTO") {
468 mode <- args[1]
469 if (mode == "PASS") {
470 auto_pass(g)
471 } else if (mode == "CARD") {
472 seat <- actor(g)
473 card <- choose_play(g)
474 output <- paste(seats[seat], "plays", card_text(card))
475 play_card(g, card)
476 } else if (mode == "TRICK") {
477 completed <- length(g$tricks)
478 while (g$state == "playing" && length(g$tricks) == completed) play_card(g, choose_play(g))
479 } else if (mode == "HAND") {
480 auto_hand(g)
481 } else {
482 auto_game(g)
483 }
484 } else if (cmd == "UNDO") {
485 if (!length(g$undo)) fail("nothing to undo")
486 result <- undo_card(g)
487 return(paste("undo", seats[result[1]], card_text(result[2])))
488 } else if (cmd == "CLAIM") {
489 if (seat != actor(g)) fail(paste("waiting for", seats[actor(g)]))
490 if (length(g$current)) fail("trick in progress")
491 for (card in g$hands[[seat]]) for (other in setdiff(1:4, seat)) {
492 hand <- g$hands[[other]]
493 if (any((hand %/% 13L) == (card %/% 13L) & (hand %% 13L) > (card %% 13L))) fail("claim rejected")
494 }
495 remaining <- unlist(g$hands, use.names = FALSE)
496 count <- 13L - length(g$tricks)
497 g$taken[[seat]] <- c(g$taken[[seat]], remaining)
498 g$counts[seat] <- g$counts[seat] + count
499 g$hands <- replicate(4, integer(), simplify = FALSE)
500 record_line(g, 2L, paste("claim", seats[seat], "tricks", count, "points", points(g, remaining)))
501 finish_hand(g)
502 } else if (cmd == "LEGAL") {
503 return(paste0("legal ", seats[actor(g)], ": ", cards_text(legal(g)), " bot ", card_text(choose_play(g))))
504 } else if (cmd == "PREVIEW") {
505 for (card in sort(legal(g))) {
506 trial <- list2env(as.list(g), parent = emptyenv())
507 completed <- length(trial$tricks)
508 play_card(trial, card)
509 while (trial$state == "playing" && length(trial$tricks) == completed) play_card(trial, choose_play(trial))
510 trick <- trial$tricks[[length(trial$tricks)]]
511 value <- sum(vapply(trick, function(play) point(g, play$card), numeric(1)))
512 output <- c(output, paste("preview", card_text(card), "->", seats[winner(trick)], value))
513 }
514 return(output)
515 } else if (cmd == "SCORE") {
516 output <- pairs("scores", g$scores)
517 if (g$state == "playing") output <- c(output, pairs("taken", vapply(g$taken, function(cards) points(g, cards), numeric(1))))
518 return(output)
519 } else if (cmd == "TAKEN") {
520 return(vapply(1:4, function(seat) {
521 cards <- g$taken[[seat]]
522 paste0(seats[seat], ": ", cards_text(cards[point(g, cards) != 0L]))
523 }, character(1)))
524 } else if (cmd == "TRICKS") {
525 output <- vapply(seq_along(g$tricks), function(number) trick_line(g, g$tricks[[number]], number), character(1))
526 if (length(g$current)) output <- c(output, paste("current:", plays_text(g$current)))
527 return(if (length(output)) output else "no tricks")
528 } else if (cmd == "VOIDS") {
529 missing <- replicate(4, integer(), simplify = FALSE)
530 for (trick in c(g$tricks, list(g$current))) {
531 if (!length(trick)) next
532 led <- trick[[1]]$card %/% 13L
533 for (play in trick) if ((play$card %/% 13L) != led) missing[[play$seat]] <- union(missing[[play$seat]], led)
534 }
535 values <- vapply(missing, function(item) if (length(item)) paste(suits[sort(item) + 1L], collapse = "") else "-", character(1))
536 return(pairs("voids", values))
537 } else if (cmd == "UNSEEN") {
538 who <- actor(g)
539 played <- c(unlist(g$taken, use.names = FALSE), vapply(g$current, function(play) play$card, integer(1)))
540 return(paste0("unseen ", seats[who], ": ", cards_text(setdiff(0:51, c(g$hands[[who]], played)))))
541 } else if (cmd == "STATS") {
542 return(stats_lines(g))
543 } else if (cmd == "HISTORY") {
544 for (hand in g$history) {
545 line <- paste("hand", hand$number, "pass", hand$direction, pairs("points", hand$points))
546 if (hand$shooter != 0L) line <- paste(line, "moon", seats[hand$shooter])
547 output <- c(output, line)
548 }
549 return(if (length(output)) output else "no hands")
550 } else if (cmd == "REPLAY") {
551 hand <- bounded(args[1], 1, length(g$history))
552 return(g$history[[hand]]$record)
553 }
554 entries <- if (length(g$log) > before) g$log[seq.int(before + 1L, length(g$log))] else list()
555 c(output, vapply(entries, function(entry) entry$text, character(1)))
556}
