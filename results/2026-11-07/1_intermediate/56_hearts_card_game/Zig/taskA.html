137fn rankOrder(cards: CardSet, descending: bool, buffer: *[52]Card) []Card {
138 var count: usize = 0;
139 for (0..13) |step| {
140 const rank = if (descending) 12 - step else step;
141 for (0..4) |suit| {
142 const card = makeCard(suit, rank);
143 if (has(cards, card)) {
144 buffer[count] = card;
145 count += 1;
146 }
147 }
148 }
149 return buffer[0..count];
150}

152fn lowest(cards: CardSet) Card {
153 var buffer: [52]Card = undefined;
154 return rankOrder(cards, false, &buffer)[0];
155}
156
157fn highest(cards: CardSet) Card {
158 var buffer: [52]Card = undefined;
159 return rankOrder(cards, true, &buffer)[0];
160}

162fn chooseSuit(cards: CardSet, most: bool) ?u6 {
163 var best: ?u6 = null;
164 var best_count: u7 = 0;
165 for (0..4) |index| {
166 const suit: u6 = @intCast(index);
167 const count = @popCount(suitSet(cards, suit));
168 if (count == 0) continue;
169 if (best == null or (most and count > best_count) or (!most and count < best_count)) {
170 best = suit;
171 best_count = count;
172 }
173 }
174 return best;
175}

218 fn point(rules: Rules, card: Card) i32 {
219 if (card == queen_spades) return 13;
220 if (suitOf(card) == heart_suit) return 1;

228 return total;

246const Generator = struct {
247 state: u64,
248
249 fn draw(generator: *Generator) u64 {
250 generator.state = 48271 * generator.state % 2147483647;
251 return generator.state;
252 }

254 fn shuffle(generator: *Generator, deck: []Card) void {
255 var i = deck.len;
256 while (i > 1) {
257 i -= 1;
258 const j: usize = @intCast(generator.draw() % (i + 1));
259 std.mem.swap(Card, &deck[i], &deck[j]);
260 }
261 }

341const Round = struct {
342 hands: [4]CardSet,
343 taken: [4]CardSet = @splat(0),
344 tricks: [13]Trick = undefined,
345 completed: usize = 0,
346 won: [4]usize = @splat(0),
347 current: Trick = .{},
348 leader: usize = 0,
349 broken: bool = false,
350
351 fn completedTricks(round: *const Round) []const Trick {
352 return round.tricks[0..round.completed];
353 }
354
355 fn actor(round: *const Round) usize {
356 return (round.leader + round.current.len) % 4;
357 }
358
359 fn queenPlayed(round: *const Round) bool {
360 var played = round.current.cards();
361 for (round.taken) |taken| played |= taken;
362 return has(played, queen_spades);
363 }
364
365 fn legal(round: *const Round, rules: Rules) CardSet {
366 const hand = round.hands[round.actor()];
367 if (round.current.len == 0) {
368 if (round.completed == 0) return bit(two_clubs);
369 const non_hearts = hand & ~heart_cards;
370 return if (!round.broken and non_hearts != 0) non_hearts else hand;
371 }
372 const followers = suitSet(hand, round.current.ledSuit());
373 if (followers != 0) return followers;
374 if (round.completed == 0 and !rules.blood) {
375 const safe = hand & ~penalty_cards;
376 if (safe != 0) return safe;
377 }
378 return hand;
379 }

381 fn choosePlay(round: *const Round, rules: Rules, bots: *const [4]Strategy) Card {
382 const seat = round.actor();
383 const hand = round.hands[seat];
384 const playable = round.legal(rules);
385 var strategy = bots[seat];
386 if (strategy == .LOW) return lowest(playable);
387 if (strategy == .HIGH) return highest(playable);
388 if (strategy == .MOON) {
389 for (round.taken, 0..) |taken, other| {
390 if (other != seat and taken & penalty_cards != 0) strategy = .BASIC;
391 }
392 }
393 if (round.current.len == 0) {
394 if (strategy == .SMART and !round.queenPlayed() and !has(hand, queen_spades)) {
395 const small_spades = suitSet(playable, spade_suit) & (bit(queen_spades) - 1);
396 if (small_spades != 0) return lowest(small_spades);
397 }
398 if (strategy == .MOON) return highest(suitSet(playable, chooseSuit(playable, true).?));
399 return lowest(suitSet(playable, chooseSuit(playable, false).?));
400 }
401 const led = round.current.ledSuit();
402 if (suitSet(hand, led) != 0) {
403 var winning: u6 = 0;
404 for (round.current.items()) |play| {
405 if (suitOf(play.card) == led) winning = @max(winning, rankOf(play.card));
406 }
407 if (strategy == .MOON) {
408 const high = highest(playable);
409 return if (rankOf(high) > winning) high else lowest(playable);
410 }
411 var lower: CardSet = 0;
412 var iterator = each(playable);
413 while (iterator.next()) |card| {
414 if (rankOf(card) < winning) lower |= bit(card);
415 }
416 if (lower != 0) return highest(lower);
417 if (strategy == .SMART and round.current.len == 3) {
418 const others = playable & ~bit(queen_spades);
419 return if (others != 0) highest(others) else queen_spades;
420 }
421 return lowest(playable);
422 }
423 if (strategy == .MOON) {
424 const safe = playable & ~penalty_cards;
425 return lowest(if (safe != 0) safe else playable);
426 }
427 if (has(playable, queen_spades)) return queen_spades;
428 if (strategy == .SMART and !round.queenPlayed()) {
429 if (has(playable, ace_spades)) return ace_spades;
430 if (has(playable, king_spades)) return king_spades;
431 }
432 const heart_choices = playable & heart_cards;
433 return highest(if (heart_choices != 0) heart_choices else playable);
434 }

436 fn add(round: *Round, rules: Rules, card: Card) void {
437 const seat = round.actor();
438 round.hands[seat] &= ~bit(card);
439 round.current.plays[round.current.len] = .{ .seat = seat, .card = card };
440 round.current.len += 1;
441 if (suitOf(card) == heart_suit or (card == queen_spades and rules.qsbreak)) {
442 round.broken = true;
443 }
444 if (round.current.len < 4) return;
445 const winner = round.current.winner();
446 round.taken[winner] |= round.current.cards();
447 round.won[winner] += 1;
448 round.tricks[round.completed] = round.current;
449 round.completed += 1;
450 round.current = .{};
451 round.leader = winner;
452 }
453};

458
459 fn add(choice: *PassChoice, card: Card) void {

522 fn deal(game: *Game) Allocator.Error!void {
523 game.number += 1;
524 game.pass_code = game.rules.pass[(game.number - 1) % game.rules.pass.len];
525 var hands: [4]CardSet = @splat(0);
526 var used: CardSet = 0;
527 var free: [4]usize = undefined;
528 var free_count: usize = 0;
529 for (game.stacked, 0..) |stacked, seat| {
530 if (stacked) |cards| {
531 hands[seat] = cards;
532 used |= cards;
533 } else {
534 free[free_count] = seat;
535 free_count += 1;
536 }
537 }
538 var deck: [52]Card = undefined;
539 var size: usize = 0;
540 var remaining = each(all_cards & ~used);
541 while (remaining.next()) |card| {
542 deck[size] = card;
543 size += 1;
544 }
545 game.generator.shuffle(deck[0..size]);
546 for (deck[0..size], 0..) |card, index| hands[free[index % free_count]] |= bit(card);
547 game.stacked = @splat(null);
548 game.passes = @splat(null);
549 game.round = .{ .hands = hands };
550 game.undo.clearRetainingCapacity();
551 game.record.clearRetainingCapacity();
552 try game.write(1, "hand {d} pass {s}", .{ game.number, directionName(game.pass_code) });
553 for (hands, seat_names) |cards, name| {
554 try game.write(2, "{c}: {f}", .{ name, cardList(cards) });
555 }
556 if (game.pass_code == 'H') {
557 try game.beginPlay();
558 } else {
559 game.state = .passing;
560 }
561 }

563 fn beginPlay(game: *Game) Allocator.Error!void {
564 game.state = .playing;
565 for (game.round.hands, 0..) |hand, seat| {
566 if (has(hand, two_clubs)) game.round.leader = seat;
567 }
568 try game.write(2, "lead {c}", .{seat_names[game.round.leader]});
569 }
570
571 fn passCards(game: *Game, seat: usize, cards: CardSet) Allocator.Error!void {
572 game.round.hands[seat] &= ~cards;
573 game.passes[seat] = cards;
574 try game.write(2, "{c} passes {f}", .{ seat_names[seat], cardList(cards) });
575 for (game.passes) |passed| if (passed == null) return;
576 const offset: usize = switch (game.pass_code) {
577 'L' => 1,
578 'R' => 3,
579 else => 2,
580 };
581 for (game.passes, 0..) |passed, source| game.round.hands[(source + offset) % 4] |= passed.?;
582 try game.beginPlay();
583 }

585 fn choosePass(game: *const Game, seat: usize) CardSet {
586 const hand = game.round.hands[seat];
587 var choice: PassChoice = .{};
588 switch (game.bots[seat]) {
589 .LOW => choice.addAll(hand, true),
590 .HIGH => choice.addAll(hand, false),
591 .MOON => {
592 choice.addAll(hand & ~heart_cards, false);
593 choice.addAll(hand & heart_cards, false);
594 },
595 .SMART => {
596 const spades = @popCount(suitSet(hand, spade_suit));
597 const protected = has(hand, queen_spades) and spades >= 5;
598 if (!protected) choice.addSpecial(hand);
599 if (chooseSuit(hand & clubs_and_diamonds, false)) |suit| {
600 choice.addAll(suitSet(hand, suit), true);
601 }
602 choice.addAll(hand & heart_cards, true);
603 choice.addAll(hand, true);
604 },
605 .BASIC => {
606 choice.addSpecial(hand);
607 choice.addAll(hand & heart_cards, true);
608 choice.addAll(hand, true);
609 },
610 }
611 return choice.cards;
612 }

618 fn play(game: *Game, card: Card) Allocator.Error!void {
619 const round = &game.round;
620 try game.undo.append(game.gpa, .{
621 .round = round.*,
622 .log_len = game.log.items.len,
623 .record_len = game.record.items.len,
624 .seat = round.actor(),
625 .card = card,
626 });
627 round.add(game.rules, card);
628 if (round.current.len > 0) return;
629 const number = round.completed;
630 try game.write(2, "{f}", .{game.trickLine(&round.tricks[number - 1], number)});
631 if (number == 13) try game.finishHand();
632 }

646 fn claim(game: *Game, seat: usize) Allocator.Error!void {
647 var remaining: CardSet = 0;
648 for (game.round.hands) |hand| remaining |= hand;
649 const count = 13 - game.round.completed;

662 fn finishHand(game: *Game) Allocator.Error!void {
663 var points: [4]i32 = undefined;
664 var shooter: ?usize = null;
665 for (game.round.taken, 0..) |taken, seat| {
666 points[seat] = game.rules.points(taken);
667 if (taken & penalty_cards == penalty_cards) shooter = seat;
668 }
669 if (shooter) |moon| {
670 points[moon] -= 26;
671 var method = game.rules.moon;
672 if (method == .AUTO) {
673 var added = game.scores;
674 for (&added, points, 0..) |*total, hand_points, seat| {
675 total.* += hand_points;
676 if (seat != moon) total.* += 26;
677 }
678 var winner: usize = 0;
679 for (added, 0..) |total, seat| {
680 if (total < added[winner]) winner = seat;
681 }
682 method = if (game.ended(added) and winner != moon) .SUB else .ADD;
683 }
684 if (method == .SUB) {
685 points[moon] -= 26;
686 } else {
687 for (&points, 0..) |*value, seat| {
688 if (seat != moon) value.* += 26;
689 }
690 }
691 try game.write(1, "moon {c} {t}", .{ seat_names[moon], method });
692 game.stats[moon].moons += 1;
693 }
694 for (0..4) |seat| {
695 const taken = game.round.taken[seat];
696 game.scores[seat] += points[seat];
697 game.stats[seat].tricks += game.round.won[seat];
698 game.stats[seat].hearts += @popCount(taken & heart_cards);
699 if (has(taken, queen_spades)) game.stats[seat].queens += 1;
700 }
701 try game.write(1, "points {f}", .{perSeat("{d}", points)});
702 try game.write(1, "scores {f}", .{perSeat("{d}", game.scores)});
703 try game.history.append(game.gpa, .{
704 .number = game.number,
705 .direction = directionName(game.pass_code),
706 .points = points,
707 .shooter = shooter,
708 .record = try game.gpa.dupe([]const u8, game.record.items),
709 });
710 const over = game.ended(game.scores);
711 game.state = if (over) .none else .between;
712 if (!over) return;
713 try game.write(0, "game over hands {d}", .{game.number});
714 const order = game.ranking();
715 for (order, 1..) |seat, rank| {
716 try game.write(0, "rank {d} {c} {d}", .{ rank, seat_names[seat], game.scores[seat] });
717 }
718 for (game.stats, seat_names) |stats, name| {
719 try game.write(0, "stats {c} {f}", .{ name, stats });
720 }
721 }
722
723 fn ranking(game: *const Game) [4]usize {
724 var order: [4]usize = .{ 0, 1, 2, 3 };
725 std.mem.sort(usize, &order, &game.scores, scoreLess);
726 return order;
727 }
728
729 fn autoPass(game: *Game) Allocator.Error!void {
730 for (0..4) |seat| {
731 if (game.passes[seat] == null) try game.passCards(seat, game.choosePass(seat));
732 }
733 }
734
735 fn autoHand(game: *Game) Allocator.Error!void {
736 if (game.state == .passing) try game.autoPass();
737 while (game.state == .playing) try game.play(game.choosePlay());
738 }
739
740 fn autoGame(game: *Game) Allocator.Error!void {
741 if (game.state == .passing or game.state == .playing) try game.autoHand();
742 while (game.state != .none) {
743 try game.deal();
744 try game.autoHand();
745 }
746 }
747};

774 fn parseSeat(table: *Table, word: []const u8) Error!usize {
775 if (word.len == 1) {
776 if (std.mem.findScalar(u8, seat_names, word[0])) |seat| return seat;

781 fn parseCard(table: *Table, word: []const u8) Error!Card {
782 if (word.len == 2) {
783 const rank = std.mem.findScalar(u8, rank_names, word[0]);
784 const suit = std.mem.findScalar(u8, suit_names, word[1]);
785 if (rank != null and suit != null) return makeCard(suit.?, rank.?);
786 }
787 return table.fail("bad card {s}", .{word});
788 }

790 fn parseCards(table: *Table, words: []const []const u8) Error![]const Card {
791 const cards = try table.gpa.alloc(Card, words.len);
792 for (cards, words) |*card, word| card.* = try table.parseCard(word);
793 return cards;
794 }

796 fn bounded(table: *Table, word: []const u8, low: i64, high: i64) Error!i64 {
797 const value = std.fmt.parseInt(i64, word, 10) catch return table.badValue(word);
798 if (value < low or value > high) return table.badValue(word);
799 return value;
800 }

802 fn parseSwitch(table: *Table, word: []const u8) Error!bool {
803 if (std.mem.eql(u8, word, "ON")) return true;
804 if (std.mem.eql(u8, word, "OFF")) return false;
805 return table.badValue(word);
806 }

808 fn setRule(table: *Table, rules: *Rules, name: RuleName, word: []const u8) Error!void {
809 switch (name) {
810 .TARGET => rules.target = @intCast(try table.bounded(word, 1, 1000)),
811 .LIMIT => rules.limit = @intCast(try table.bounded(word, 1, 100)),
812 .MOON => rules.moon = std.meta.stringToEnum(MoonMethod, word) orelse
813 return table.badValue(word),
814 .JACK => rules.jack = try table.parseSwitch(word),
815 .QSBREAK => rules.qsbreak = try table.parseSwitch(word),
816 .BLOOD => rules.blood = try table.parseSwitch(word),
817 .PASS => {
818 if (word.len == 0 or word.len > 8) return table.badValue(word);
819 for (word) |char| {
820 if (std.mem.findScalar(u8, "LRAH", char) == null) return table.badValue(word);
821 }
822 rules.pass = word;
823 },
824 }
825 }

827 fn requireTurn(table: *Table, round: *const Round, seat: usize) Error!void {

829 return table.fail("waiting for {c}", .{seat_names[round.actor()]});
830 }

833 fn checkDuplicates(table: *Table, cards: []const Card) Error!void {
834 var seen: CardSet = 0;

836 if (has(seen, card)) return table.fail("duplicate card {s}", .{&cardName(card)});
837 seen |= bit(card);

841 fn checkHolds(table: *Table, hand: CardSet, seat: usize, cards: []const Card) Error!void {
842 for (cards) |card| {
843 if (!has(hand, card)) {
844 return table.fail("{c} does not hold {s}", .{ seat_names[seat], &cardName(card) });
845 }
846 }

849 fn simulate(table: *Table, command: Command, seed: i64, count: usize, out: *Writer) Error!void {
850 var generator: Generator = .{ .state = @intCast(seed) };
851 if (command == .GAME) {
852 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
853 defer arena.deinit();
854 var game: Game = .{
855 .gpa = arena.allocator(),
856 .rules = table.rules,
857 .bots = &table.bots,
858 .generator = &generator,
859 };
860 try game.autoGame();
861 try out.print("game seed {d}\n", .{seed});
862 for (game.log.items) |entry| {
863 if (entry.level <= count) try out.print("{s}\n", .{entry.text});
864 }
865 return;
866 }
867 const rounds: usize = if (command == .ROTATE) 4 else 1;
868 var places: [4][4]usize = @splat(@splat(0));
869 var totals: [4]i64 = @splat(0);
870 var stats: [4]Stats = @splat(.{});
871 var hands: usize = 0;
872 for (0..rounds) |rotation| {
873 var strategies: [4]Strategy = undefined;
874 for (table.bots, 0..) |bot, seat| strategies[(seat + rotation) % 4] = bot;
875 for (0..count) |_| {
876 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
877 defer arena.deinit();
878 var game: Game = .{
879 .gpa = arena.allocator(),
880 .rules = table.rules,
881 .bots = &strategies,
882 .generator = &generator,
883 };
884 try game.autoGame();
885 hands += game.number;
886 const order = game.ranking();
887 for (0..4) |original| {
888 const actual = (original + rotation) % 4;
889 places[original][std.mem.findScalar(usize, &order, actual).?] += 1;
890 totals[original] += game.scores[actual];
891 stats[original].add(game.stats[actual]);
892 }
893 }
894 }
895 const games: i64 = @intCast(rounds * count);
896 const label = if (command == .ROTATE) "rotate" else "match";
897 try out.print("{s} seed {d} games {d} hands {d}\n", .{ label, seed, games, hands });
898 for (0..4) |seat| {
899 const cents = @divFloor(100 * totals[seat], games);
900 const magnitude = @abs(cents);
901 const sign = if (cents < 0) "-" else "";
902 try out.print("{c}", .{seat_names[seat]});
903 if (command == .ROTATE) try out.print(" {t}", .{table.bots[seat]});
904 try out.print(" places {d} {d} {d} {d} total {d} average {s}{d}.{d:0>2} {f}\n", .{
905 places[seat][0], places[seat][1], places[seat][2], places[seat][3], totals[seat],
906 sign, magnitude / 100, magnitude % 100, stats[seat],
907 });
908 }
909 }
910
911 fn execute(table: *Table, words: []const []const u8, out: *Writer) Error!void {
912 const command = std.meta.stringToEnum(Command, words[0]) orelse
913 return table.fail("unknown command {s}", .{words[0]});
914 const args = words[1..];
915 switch (command) {
916 .RULE => {
917 const name = std.meta.stringToEnum(RuleName, args[0]) orelse
918 return table.fail("unknown rule {s}", .{args[0]});
919 var rules = table.rules;
920 try table.setRule(&rules, name, args[1]);
921 try table.require(table.tableState() == .none, "game in progress");
922 table.rules = rules;
923 try out.print("rule {s} {s}\n", .{ args[0], args[1] });
924 },
925 .BOT => {
926 const seat = try table.parseSeat(args[0]);
927 table.bots[seat] = std.meta.stringToEnum(Strategy, args[1]) orelse
928 return table.badValue(args[1]);
929 try out.print("bot {c} {s}\n", .{ seat_names[seat], args[1] });
930 },
931 .SETTINGS => {
932 try out.print("rules {f}\n", .{table.rules});
933 try out.print("bots {f}\n", .{perSeat("{t}", table.bots)});
934 },
935 .GAME, .MATCH, .ROTATE => {
936 const seed = try table.bounded(args[0], 1, 2147483646);
937 const high: i64 = switch (command) {
938 .GAME => 2,
939 .MATCH => 50,
940 else => 10,
941 };
942 const count = try table.bounded(args[1], if (command == .GAME) 0 else 1, high);
943 try table.simulate(command, seed, @intCast(count), out);
944 },
945 .NEW => {
946 const seed = try table.bounded(args[0], 1, 2147483646);
947 try table.require(table.tableState() == .none, "game in progress");
948 table.generator = .{ .state = @intCast(seed) };
949 table.game = .{
950 .gpa = table.gpa,
951 .rules = table.rules,
952 .bots = &table.bots,
953 .generator = &table.generator,
954 };
955 try out.print("new game seed {d}\n", .{seed});
956 },
957 .ABANDON => {
958 try table.require(table.tableState() != .none, "no game");
959 table.game = null;
960 try out.writeAll("game abandoned\n");
961 },
962 else => try table.tableCommand(command, args, out),
963 }
964 }

966 fn tableCommand(
967 table: *Table,
968 command: Command,
969 args: []const []const u8,
970 out: *Writer,
971 ) Error!void {
972 const seat = switch (command) {
973 .STACK, .PASS, .PLAY, .CLAIM => try table.parseSeat(args[0]),
974 else => 0,
975 };
976 const cards = switch (command) {
977 .STACK, .PASS, .PLAY => try table.parseCards(args[1..]),
978 else => &.{},
979 };
980 if (command == .STACK and cards.len != 13) return table.fail("need 13 cards", .{});
981 var mode: AutoMode = .PASS;
982 if (command == .AUTO) {
983 mode = std.meta.stringToEnum(AutoMode, args[0]) orelse return table.badValue(args[0]);
984 }
985
986 const state = table.tableState();
987 const in_hand = state == .passing or state == .playing;
988 switch (command) {
989 .STACK, .DEAL => {
990 try table.require(state != .none, "no game");
991 try table.require(state == .between, "hand in progress");
992 },
993 .SHOW => try table.require(in_hand, "no hand"),
994 .PASS => try table.require(state == .passing, "not passing"),
995 .PLAY, .UNDO, .CLAIM, .LEGAL, .PREVIEW, .TAKEN, .TRICKS, .VOIDS, .UNSEEN => {
996 try table.require(state == .playing, "not playing");
997 },
998 .SCORE, .STATS, .HISTORY, .REPLAY => try table.require(state != .none, "no game"),
999 .AUTO => switch (mode) {
1000 .PASS => try table.require(state == .passing, "not passing"),
1001 .CARD, .TRICK => try table.require(state == .playing, "not playing"),
1002 .HAND => try table.require(in_hand, "no hand"),
1003 .GAME => try table.require(state != .none, "no game"),
1004 },
1005 else => unreachable,
1006 }
1007 const game = &table.game.?;
1008 const round = &game.round;
1009 const before = game.log.items.len;
1010 switch (command) {
1011 .STACK => {
1012 try table.checkDuplicates(cards);
1013 for (cards) |card| {
1014 for (game.stacked, seat_names, 0..) |stacked, name, owner| {
1015 if (owner == seat or !has(stacked orelse 0, card)) continue;
1016 return table.fail("{s} stacked for {c}", .{ &cardName(card), name });
1017 }
1018 }
1019 game.stacked[seat] = setOf(cards);
1020 return out.print("stacked {c}\n", .{seat_names[seat]});
1021 },
1022 .DEAL => try game.deal(),
1023 .SHOW => {
1024 for (round.hands, seat_names) |hand, name| {
1025 try out.print("{c}: {f}\n", .{ name, cardList(hand) });
1026 }
1027 return;
1028 },
1029 .PASS => {
1030 if (game.passes[seat] != null) {
1031 return table.fail("{c} already passed", .{seat_names[seat]});
1032 }
1033 try table.checkDuplicates(cards);
1034 try table.checkHolds(round.hands[seat], seat, cards);
1035 try game.passCards(seat, setOf(cards));
1036 },
1037 .PLAY => {
1038 const card = cards[0];
1039 try table.requireTurn(round, seat);
1040 try table.checkHolds(round.hands[seat], seat, cards);
1041 if (!has(round.legal(game.rules), card)) {
1042 if (round.current.len == 0) {
1043 if (round.completed == 0) return table.fail("must lead 2C", .{});
1044 return table.fail("hearts not broken", .{});
1045 }
1046 const led = round.current.ledSuit();
1047 if (suitSet(round.hands[seat], led) != 0) {
1048 return table.fail("must follow {c}", .{suit_names[led]});
1049 }
1050 return table.fail("no points on first trick", .{});
1051 }
1052 try out.print("{c} plays {s}\n", .{ seat_names[seat], &cardName(card) });
1053 try game.play(card);
1054 },
1055 .AUTO => switch (mode) {
1056 .PASS => try game.autoPass(),
1057 .CARD => {
1058 const actor = round.actor();
1059 const card = game.choosePlay();
1060 try out.print("{c} plays {s}\n", .{ seat_names[actor], &cardName(card) });
1061 try game.play(card);
1062 },
1063 .TRICK => {
1064 const count = round.completed;
1065 while (game.state == .playing and round.completed == count) {
1066 try game.play(game.choosePlay());
1067 }
1068 },
1069 .HAND => try game.autoHand(),
1070 .GAME => try game.autoGame(),
1071 },
1072 .UNDO => {
1073 if (game.undo.items.len == 0) return table.fail("nothing to undo", .{});
1074 const last = game.undoCard();
1075 const card_name = cardName(last.card);
1076 return out.print("undo {c} {s}\n", .{ seat_names[last.seat], &card_name });
1077 },
1078 .CLAIM => {
1079 try table.requireTurn(round, seat);
1080 if (round.current.len > 0) return table.fail("trick in progress", .{});
1081 var others: CardSet = 0;
1082 for (round.hands, 0..) |hand, other| {
1083 if (other != seat) others |= hand;
1084 }
1085 var held = each(round.hands[seat]);
1086 while (held.next()) |card| {
1087 const higher = suitSet(others, suitOf(card)) & ~(bit(card) - 1);
1088 if (higher != 0) return table.fail("claim rejected", .{});
1089 }
1090 try game.claim(seat);
1091 },
1092 .LEGAL => return out.print("legal {c}: {f} bot {s}\n", .{
1093 seat_names[round.actor()],
1094 cardList(round.legal(game.rules)),
1095 &cardName(game.choosePlay()),
1096 }),
1097 .PREVIEW => {
1098 var choices = each(round.legal(game.rules));
1099 while (choices.next()) |card| {
1100 var preview = round.*;
1101 const count = preview.completed;
1102 preview.add(game.rules, card);
1103 while (preview.completed == count) {
1104 preview.add(game.rules, preview.choosePlay(game.rules, game.bots));
1105 }
1106 const trick = &preview.tricks[count];
1107 try out.print("preview {s} -> {c} {d}\n", .{
1108 &cardName(card),
1109 seat_names[trick.winner()],
1110 game.rules.points(trick.cards()),
1111 });
1112 }
1113 return;
1114 },
1115 .SCORE => {
1116 try out.print("scores {f}\n", .{perSeat("{d}", game.scores)});
1117 if (game.state == .playing) {
1118 var taken: [4]i32 = undefined;
1119 for (&taken, round.taken) |*value, cards_taken| {
1120 value.* = game.rules.points(cards_taken);
1121 }
1122 try out.print("taken {f}\n", .{perSeat("{d}", taken)});
1123 }
1124 return;
1125 },
1126 .TAKEN => {
1127 for (round.taken, seat_names) |taken, name| {
1128 const scoring = taken & game.rules.pointCards();
1129 try out.print("{c}: {f}\n", .{ name, cardList(scoring) });
1130 }
1131 return;
1132 },
1133 .TRICKS => {
1134 for (round.completedTricks(), 1..) |*trick, number| {
1135 try out.print("{f}\n", .{game.trickLine(trick, number)});
1136 }
1137 if (round.current.len > 0) try out.print("current: {f}\n", .{round.current});
1138 if (round.completed == 0 and round.current.len == 0) {
1139 try out.writeAll("no tricks\n");
1140 }
1141 return;
1142 },
1143 .VOIDS => {
1144 var voids: [4]SuitList = @splat(.{});
1145 for (round.completedTricks()) |*trick| markVoids(&voids, trick);
1146 markVoids(&voids, &round.current);
1147 return out.print("voids {f}\n", .{perSeat("{f}", voids)});
1148 },
1149 .UNSEEN => {
1150 var seen = round.hands[round.actor()] | round.current.cards();
1151 for (round.taken) |taken| seen |= taken;
1152 return out.print("unseen {c}: {f}\n", .{
1153 seat_names[round.actor()],
1154 cardList(all_cards & ~seen),
1155 });
1156 },
1157 .STATS => {
1158 for (game.stats, seat_names) |stats, name| {
1159 try out.print("stats {c} {f}\n", .{ name, stats });
1160 }
1161 return;
1162 },
1163 .HISTORY => {
1164 for (game.history.items) |hand| {
1165 try out.print("hand {d} pass {s} points {f}", .{
1166 hand.number,
1167 hand.direction,
1168 perSeat("{d}", hand.points),
1169 });
1170 if (hand.shooter) |shooter| try out.print(" moon {c}", .{seat_names[shooter]});
1171 try out.writeByte('\n');
1172 }
1173 if (game.history.items.len == 0) try out.writeAll("no hands\n");
1174 return;
1175 },
1176 .REPLAY => {
1177 const number = try table.bounded(args[0], 1, @intCast(game.history.items.len));
1178 for (game.history.items[@intCast(number - 1)].record) |text| {
1179 try out.print("{s}\n", .{text});
1180 }
1181 return;
1182 },
1183 else => unreachable,
1184 }
1185 for (game.log.items[before..]) |entry| try out.print("{s}\n", .{entry.text});
1186 }
1187};

1189pub fn main(init: std.process.Init) !void {
1190 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1191 defer storage.deinit();
1192 const gpa = storage.allocator();
1193 var input_buffer: [4096]u8 = undefined;
1194 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
1195 var output_buffer: [4096]u8 = undefined;
1196 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1197 const out = &output.interface;
1198 const data = try input.interface.allocRemaining(gpa, .unlimited);
1199 var lines = std.mem.tokenizeAny(u8, data, "\r\n");
1200 const count = try std.fmt.parseInt(usize, lines.next() orelse return, 10);
1201 var table: Table = .{ .gpa = gpa };
1202 for (0..count) |_| {
1203 const text = lines.next() orelse break;
1204 var words: std.ArrayList([]const u8) = .empty;
1205 var parts = std.mem.splitScalar(u8, text, ' ');
1206 while (parts.next()) |word| try words.append(gpa, word);
1207 table.execute(words.items, out) catch |err| {
1208 if (err != error.Invalid) return err;
1209 try out.print("error: {s}\n", .{table.problem});
1210 };
1211 }
1212 try out.flush();
1213}
