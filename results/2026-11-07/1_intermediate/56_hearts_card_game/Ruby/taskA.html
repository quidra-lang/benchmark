1SEATS = 'NESW'
2SUITS = 'CDSH'
3RANKS = '23456789TJQKA'
4STRATEGIES = %w[BASIC LOW HIGH SMART MOON]
5DIRECTIONS = {'L' => 'LEFT', 'R' => 'RIGHT', 'A' => 'ACROSS', 'H' => 'HOLD'}
6DEFAULT_RULES = {'TARGET' => 100, 'LIMIT' => 100, 'MOON' => 'ADD', 'JACK' => 'OFF',
7 'QSBREAK' => 'OFF', 'BLOOD' => 'OFF', 'PASS' => 'LRAH'}
8QS, KS, AS, JD = 36, 37, 38, 22
9class CommandError < StandardError; end
10
11def fail_command(message)
12 raise CommandError, message
13end
14
15def card_text(card)
16 RANKS[card % 13] + SUITS[card / 13]
17end
18
19def cards_text(cards)
20 cards.empty? ? '-' : cards.sort.map { |card| card_text(card) }.join(' ')
21end
22
23def seat_value(word)
24 fail_command("bad seat #{word}") unless word.size == 1 && SEATS.include?(word)
25 SEATS.index(word)
26end
27
28def card_value(word)
29 unless word.size == 2 && RANKS.include?(word[0]) && SUITS.include?(word[1])
30 fail_command("bad card #{word}")
31 end
32 SUITS.index(word[1]) * 13 + RANKS.index(word[0])
33end
34
35def bounded(word, low, high)
36 value = word.to_i
37 fail_command("bad value #{word}") unless value.between?(low, high)
38 value
39end
40
41def lowest(cards)
42 cards.min_by { |card| [card % 13, card / 13] }
43end
44
45def highest(cards)
46 cards.min_by { |card| [-(card % 13), card / 13] }
47end
48
49def penalty?(card)
50 card == QS || card / 13 == 3
51end
52
53def pairs(label, values)
54 label + ' ' + values.each_with_index.map { |value, seat| "#{SEATS[seat]} #{value}" }.join(' ')
55end
56
57def duplicate_cards(cards)
58 seen = []
59 cards.each do |card|
60 fail_command("duplicate card #{card_text(card)}") if seen.include?(card)
61 seen << card
62 end
63end

65class Generator
66 def initialize(seed)
67 @state = seed
68 end
69
70 def shuffle(deck)
71 (deck.size - 1).downto(1) do |i|
72 @state = 48271 * @state % 2147483647
73 j = @state % (i + 1)
74 deck[i], deck[j] = deck[j], deck[i]
75 end
76 end
77end

79class Game
80 attr_accessor :bots
81 attr_reader :state, :number, :scores, :stats, :stacked, :history, :log, :record,
82 :hands, :passes, :taken, :tricks, :current, :undo
83
84 def initialize(rules, bots, generator)
85 @rules, @bots, @generator = rules, bots, generator
86 @state = 'between'
87 @number = 0
88 @scores = Array.new(4, 0)
89 @stats = Array.new(4) { Array.new(4, 0) }
90 @stacked = {}
91 @history, @log, @record = [], [], []
92 end
93
94 def point(card)
95 return 13 if card == QS
96 return 1 if card / 13 == 3
97 card == JD && @rules['JACK'] == 'ON' ? -10 : 0
98 end
99
100 def write(level, line)
101 @log << [level, line]
102 @record << line if level > 0
103 end
104
105 def deal
106 @number += 1
107 cycle = @rules['PASS']
108 @pass_code = cycle[(@number - 1) % cycle.size]
109 @hands = Array.new(4) { |seat| (@stacked[seat] || []).dup }
110 deck = (0...52).to_a - @hands.flatten
111 @generator.shuffle(deck)
112 free = (0...4).reject { |seat| @stacked.key?(seat) }
113 deck.each_with_index { |card, i| @hands[free[i % free.size]] << card }
114 @stacked.clear
115 @passes = {}
116 @taken = Array.new(4) { [] }
117 @tricks, @current, @undo = [], [], []
118 @trick_counts = Array.new(4, 0)
119 @broken = false
120 @record = []
121 write(1, "hand #{@number} pass #{DIRECTIONS[@pass_code]}")
122 4.times { |seat| write(2, "#{SEATS[seat]}: #{cards_text(@hands[seat])}") }
123 if @pass_code == 'H'
124 begin_play
125 else
126 @state = 'passing'
127 end
128 end
129
130 def begin_play
131 @state = 'playing'
132 @leader = (0...4).find { |seat| @hands[seat].include?(0) }
133 write(2, "lead #{SEATS[@leader]}")
134 end
135
136 def pass_cards(seat, cards)
137 @hands[seat] -= cards
138 @passes[seat] = cards.dup
139 write(2, "#{SEATS[seat]} passes #{cards_text(cards)}")
140 return unless @passes.size == 4
141 offset = {'L' => 1, 'R' => 3, 'A' => 2}[@pass_code]
142 @passes.each { |source, passed| @hands[(source + offset) % 4].concat(passed) }
143 begin_play
144 end
145
146 def actor
147 (@leader + @current.size) % 4
148 end
149
150 def legal
151 hand = @hands[actor]
152 if @current.empty?
153 return [0] if @tricks.empty?
154 safe = hand.reject { |card| card / 13 == 3 }
155 return !@broken && !safe.empty? ? safe : hand.dup
156 end
157 suit = @current[0][1] / 13
158 followers = hand.select { |card| card / 13 == suit }
159 return followers unless followers.empty?
160 if @tricks.empty? && @rules['BLOOD'] == 'OFF'
161 safe = hand.reject { |card| penalty?(card) }
162 return safe unless safe.empty?
163 end
164 hand.dup
165 end
166
167 def choose_pass(seat)
168 hand = @hands[seat]
169 low = hand.sort_by { |card| [card % 13, card / 13] }
170 high = hand.sort_by { |card| [-(card % 13), card / 13] }
171 hearts = high.select { |card| card / 13 == 3 }
172 special = [QS, KS, AS].select { |card| hand.include?(card) }
173 order = case @bots[seat]
174 when 'LOW' then high
175 when 'HIGH' then low
176 when 'MOON'
177 low.reject { |card| card / 13 == 3 } + low.select { |card| card / 13 == 3 }
178 when 'SMART'
179 protected = hand.include?(QS) && hand.count { |card| card / 13 == 2 } >= 5
180 list = protected ? [] : special.dup
181 suits = [0, 1].select { |suit| hand.any? { |card| card / 13 == suit } }
182 unless suits.empty?
183 suit = suits.min_by { |s| [hand.count { |card| card / 13 == s }, s] }
184 list += high.select { |card| card / 13 == suit }
185 end
186 list + hearts + high
187 else special + hearts + high
188 end
189 order.uniq.first(3)
190 end
191
192 def queen_played?
193 @taken.any? { |cards| cards.include?(QS) } || @current.any? { |_, card| card == QS }
194 end
195
196 def choose_play
197 seat = actor
198 hand = @hands[seat]
199 choices = legal
200 strategy = @bots[seat]
201 return lowest(choices) if strategy == 'LOW'
202 return highest(choices) if strategy == 'HIGH'
203 if strategy == 'MOON' && (0...4).any? { |other| other != seat && @taken[other].any? { |card| penalty?(card) } }
204 strategy = 'BASIC'
205 end
206 if @current.empty?
207 if strategy == 'SMART' && !queen_played? && !hand.include?(QS)
208 small = choices.select { |card| card / 13 == 2 && card % 13 < 10 }
209 return lowest(small) unless small.empty?
210 end
211 suits = choices.map { |card| card / 13 }.uniq
212 suit = suits.min_by do |s|
213 count = choices.count { |card| card / 13 == s }
214 [strategy == 'MOON' ? -count : count, s]
215 end
216 suited = choices.select { |card| card / 13 == suit }
217 return strategy == 'MOON' ? highest(suited) : lowest(suited)
218 end
219 led = @current[0][1] / 13
220 if hand.any? { |card| card / 13 == led }
221 winning = @current.select { |_, card| card / 13 == led }.map { |_, card| card % 13 }.max
222 lower = choices.select { |card| card % 13 < winning }
223 if strategy == 'MOON'
224 high = highest(choices)
225 return high % 13 > winning ? high : lowest(choices)
226 end
227 return highest(lower) unless lower.empty?
228 if strategy == 'SMART' && @current.size == 3
229 safe = choices - [QS]
230 return safe.empty? ? QS : highest(safe)
231 end
232 return lowest(choices)
233 end
234 if strategy == 'MOON'
235 safe = choices.reject { |card| penalty?(card) }
236 return lowest(safe.empty? ? choices : safe)
237 end
238 return QS if choices.include?(QS)
239 if strategy == 'SMART' && !queen_played?
240 return AS if choices.include?(AS)
241 return KS if choices.include?(KS)
242 end
243 hearts = choices.select { |card| card / 13 == 3 }
244 highest(hearts.empty? ? choices : hearts)
245 end
246
247 def winner(plays)
248 led = plays[0][1] / 13
249 plays.select { |_, card| card / 13 == led }.max_by { |_, card| card % 13 }[0]
250 end
251
252 def trick_line(plays, number)
253 sequence = plays.map { |seat, card| "#{SEATS[seat]} #{card_text(card)}" }.join(' ')
254 "trick #{number}: #{sequence} -> #{SEATS[winner(plays)]} #{plays.sum { |_, card| point(card) }}"
255 end
256
257 def play(card)
258 seat = actor
259 fields = [@hands, @taken, @tricks, @trick_counts, @current, @leader, @broken, @record]
260 @undo << [Marshal.load(Marshal.dump(fields)), @log.size, seat, card]
261 @hands[seat].delete(card)
262 @current << [seat, card]
263 @broken = true if card / 13 == 3 || card == QS && @rules['QSBREAK'] == 'ON'
264 return unless @current.size == 4
265 won = winner(@current)
266 @taken[won].concat(@current.map(&:last))
267 @trick_counts[won] += 1
268 @tricks << @current
269 write(2, trick_line(@current, @tricks.size))
270 @current = []
271 @leader = won
272 finish_hand if @tricks.size == 13
273 end
274
275 def undo_card
276 fields, length, seat, card = @undo.pop
277 @hands, @taken, @tricks, @trick_counts, @current, @leader, @broken, @record = fields
278 @log.slice!(length..-1)
279 [seat, card]
280 end
281
282 def claim(seat)
283 remaining = @hands.flatten
284 count = 13 - @tricks.size
285 @taken[seat].concat(remaining)
286 @trick_counts[seat] += count
287 @hands = Array.new(4) { [] }
288 write(2, "claim #{SEATS[seat]} tricks #{count} points #{remaining.sum { |card| point(card) }}")
289 finish_hand
290 end
291
292 def ranking
293 (0...4).sort_by { |seat| [@scores[seat], seat] }
294 end
295
296 def stats_lines
297 @stats.each_with_index.map do |s, seat|
298 "stats #{SEATS[seat]} tricks #{s[0]} hearts #{s[1]} queens #{s[2]} moons #{s[3]}"
299 end
300 end
301
302 def finish_hand
303 points = @taken.map { |cards| cards.sum { |card| point(card) } }
304 shooter = (0...4).find { |seat| @taken[seat].count { |card| penalty?(card) } == 14 }
305 unless shooter.nil?
306 points[shooter] -= 26
307 method = @rules['MOON']
308 if method == 'AUTO'
309 add = (0...4).map { |seat| @scores[seat] + points[seat] + (seat == shooter ? 0 : 26) }
310 ended = @number >= @rules['LIMIT'] || add.max >= @rules['TARGET']
311 first = (0...4).min_by { |seat| [add[seat], seat] }
312 method = ended && first != shooter ? 'SUB' : 'ADD'
313 end
314 if method == 'SUB'
315 points[shooter] -= 26
316 else
317 4.times { |seat| points[seat] += 26 if seat != shooter }
318 end
319 write(1, "moon #{SEATS[shooter]} #{method}")
320 end
321 4.times do |seat|
322 @scores[seat] += points[seat]
323 @stats[seat][0] += @trick_counts[seat]
324 @stats[seat][1] += @taken[seat].count { |card| card / 13 == 3 }
325 @stats[seat][2] += 1 if @taken[seat].include?(QS)
326 @stats[seat][3] += 1 if shooter == seat
327 end
328 write(1, pairs('points', points))
329 write(1, pairs('scores', @scores))
330 @history << {number: @number, direction: DIRECTIONS[@pass_code], points: points,
331 shooter: shooter, record: @record.dup}
332 ended = @number >= @rules['LIMIT'] || @scores.max >= @rules['TARGET']
333 @state = ended ? 'none' : 'between'
334 return unless ended
335 write(0, "game over hands #{@number}")
336 ranking.each_with_index { |seat, i| write(0, "rank #{i + 1} #{SEATS[seat]} #{@scores[seat]}") }
337 stats_lines.each { |line| write(0, line) }
338 end
339
340 def auto_pass
341 4.times { |seat| pass_cards(seat, choose_pass(seat)) unless @passes.key?(seat) }
342 end
343
344 def auto_hand
345 auto_pass if @state == 'passing'
346 play(choose_play) while @state == 'playing'
347 end
348
349 def auto_game
350 auto_hand if %w[passing playing].include?(@state)
351 until @state == 'none'
352 deal
353 auto_hand
354 end
355 end
356end

358class Table
359 def initialize
360 @rules = DEFAULT_RULES.dup
361 @bots = Array.new(4, 'BASIC')
362 @game = nil
363 end
364
365 def active?
366 @game && @game.state != 'none'
367 end
368
369 def require_state(kind)
370 case kind
371 when 'game' then fail_command('no game') unless active?
372 when 'between'
373 require_state('game')
374 fail_command('hand in progress') unless @game.state == 'between'
375 when 'hand'
376 fail_command('no hand') unless active? && %w[passing playing].include?(@game.state)
377 else
378 fail_command("not #{kind}") unless active? && @game.state == kind
379 end
380 end
381
382 def rule_value(name, word)
383 fail_command("unknown rule #{name}") unless @rules.key?(name)
384 if %w[TARGET LIMIT].include?(name)
385 fail_command("bad value #{word}") unless word.match?(/\A-?\d+\z/)
386 return bounded(word, 1, name == 'TARGET' ? 1000 : 100)
387 end
388 good = if name == 'PASS'
389 word.match?(/\A[LRAH]{1,8}\z/)
390 elsif name == 'MOON'
391 %w[ADD SUB AUTO].include?(word)
392 else
393 %w[ON OFF].include?(word)
394 end
395 fail_command("bad value #{word}") unless good
396 word
397 end
398
399 def simulate(command, seed, count)
400 generator = Generator.new(seed)
401 if command == 'GAME'
402 game = Game.new(@rules, @bots, generator)
403 game.auto_game
404 return ["game seed #{seed}"] + game.log.select { |level, _| level <= count }.map(&:last)
405 end
406 rounds = command == 'ROTATE' ? 4 : 1
407 places = Array.new(4) { Array.new(4, 0) }
408 totals = Array.new(4, 0)
409 stats = Array.new(4) { Array.new(4, 0) }
410 hands = 0
411 rounds.times do |rotation|
412 strategies = Array.new(4)
413 4.times { |seat| strategies[(seat + rotation) % 4] = @bots[seat] }
414 count.times do
415 game = Game.new(@rules, strategies, generator)
416 game.auto_game
417 hands += game.number
418 ranking = game.ranking
419 4.times do |original|
420 actual = (original + rotation) % 4
421 places[original][ranking.index(actual)] += 1
422 totals[original] += game.scores[actual]
423 4.times { |i| stats[original][i] += game.stats[actual][i] }
424 end
425 end
426 end
427 games = rounds * count
428 output = ["#{command.downcase} seed #{seed} games #{games} hands #{hands}"]
429 4.times do |seat|
430 cents = 100 * totals[seat] / games
431 average = "#{cents < 0 ? '-' : ''}#{cents.abs / 100}.#{format('%02d', cents.abs % 100)}"
432 name = SEATS[seat] + (command == 'ROTATE' ? " #{@bots[seat]}" : '')
433 s = stats[seat]
434 output << "#{name} places #{places[seat].join(' ')} total #{totals[seat]} average #{average} tricks #{s[0]} hearts #{s[1]} queens #{s[2]} moons #{s[3]}"
435 end
436 output
437 end

439 def command(words)
440 command, *args = words
441 known = %w[RULE BOT SETTINGS GAME MATCH ROTATE NEW ABANDON STACK DEAL SHOW PASS PLAY AUTO UNDO CLAIM LEGAL PREVIEW SCORE TAKEN TRICKS VOIDS UNSEEN STATS HISTORY REPLAY]
442 fail_command("unknown command #{command}") unless known.include?(command)
443 case command
444 when 'RULE'
445 value = rule_value(args[0], args[1])
446 fail_command('game in progress') if active?
447 @rules[args[0]] = value
448 return ["rule #{args.join(' ')}"]
449 when 'BOT'
450 seat = seat_value(args[0])
451 fail_command("bad value #{args[1]}") unless STRATEGIES.include?(args[1])
452 @bots[seat] = args[1]
453 return ["bot #{args.join(' ')}"]
454 when 'SETTINGS'
455 return ['rules ' + @rules.map { |name, value| "#{name} #{value}" }.join(' '), pairs('bots', @bots)]
456 when 'GAME', 'MATCH', 'ROTATE'
457 seed = bounded(args[0], 1, 2147483646)
458 low, high = {'GAME' => [0, 2], 'MATCH' => [1, 50], 'ROTATE' => [1, 10]}[command]
459 return simulate(command, seed, bounded(args[1], low, high))
460 when 'NEW'
461 seed = bounded(args[0], 1, 2147483646)
462 fail_command('game in progress') if active?
463 @game = Game.new(@rules, @bots, Generator.new(seed))
464 return ["new game seed #{seed}"]
465 when 'ABANDON'
466 require_state('game')
467 @game = nil
468 return ['game abandoned']
469 end
470 seat = seat_value(args[0]) if %w[STACK PASS PLAY CLAIM].include?(command)
471 cards = %w[STACK PASS PLAY].include?(command) ? args.drop(1).map { |word| card_value(word) } : []
472 fail_command('need 13 cards') if command == 'STACK' && cards.size != 13
473 if command == 'AUTO' && !%w[PASS CARD TRICK HAND GAME].include?(args[0])
474 fail_command("bad value #{args[0]}")
475 end
476 kind = case command
477 when 'STACK', 'DEAL' then 'between'
478 when 'SHOW' then 'hand'
479 when 'PASS' then 'passing'
480 when 'SCORE', 'STATS', 'HISTORY', 'REPLAY' then 'game'
481 when 'AUTO'
482 {'PASS' => 'passing', 'CARD' => 'playing', 'TRICK' => 'playing', 'HAND' => 'hand', 'GAME' => 'game'}[args[0]]
483 else 'playing'
484 end
485 require_state(kind)
486 game = @game
487 before = game.log.size
488 prefix = []
489 case command
490 when 'STACK'
491 duplicate_cards(cards)
492 cards.each do |card|
493 owner = (0...4).find { |other| other != seat && (game.stacked[other] || []).include?(card) }
494 fail_command("#{card_text(card)} stacked for #{SEATS[owner]}") unless owner.nil?
495 end
496 game.stacked[seat] = cards.dup
497 return ["stacked #{SEATS[seat]}"]
498 when 'DEAL' then game.deal
499 when 'SHOW'
500 return (0...4).map { |s| "#{SEATS[s]}: #{cards_text(game.hands[s])}" }
501 when 'PASS'
502 fail_command("#{SEATS[seat]} already passed") if game.passes.key?(seat)
503 duplicate_cards(cards)
504 cards.each { |card| fail_command("#{SEATS[seat]} does not hold #{card_text(card)}") unless game.hands[seat].include?(card) }
505 game.pass_cards(seat, cards)
506 when 'PLAY'
507 card = cards[0]
508 fail_command("waiting for #{SEATS[game.actor]}") if seat != game.actor
509 fail_command("#{SEATS[seat]} does not hold #{card_text(card)}") unless game.hands[seat].include?(card)
510 unless game.legal.include?(card)
511 fail_command(game.tricks.empty? ? 'must lead 2C' : 'hearts not broken') if game.current.empty?
512 led = game.current[0][1] / 13
513 fail_command("must follow #{SUITS[led]}") if game.hands[seat].any? { |held| held / 13 == led }
514 fail_command('no points on first trick')
515 end
516 prefix << "#{SEATS[seat]} plays #{card_text(card)}"
517 game.play(card)
518 when 'AUTO'
519 case args[0]
520 when 'PASS' then game.auto_pass
521 when 'CARD'
522 card = game.choose_play
523 prefix << "#{SEATS[game.actor]} plays #{card_text(card)}"
524 game.play(card)
525 when 'TRICK'
526 count = game.tricks.size
527 game.play(game.choose_play) while game.state == 'playing' && game.tricks.size == count
528 when 'HAND' then game.auto_hand
529 when 'GAME' then game.auto_game
530 end
531 when 'UNDO'
532 fail_command('nothing to undo') if game.undo.empty?
533 s, card = game.undo_card
534 return ["undo #{SEATS[s]} #{card_text(card)}"]
535 when 'CLAIM'
536 fail_command("waiting for #{SEATS[game.actor]}") if seat != game.actor
537 fail_command('trick in progress') unless game.current.empty?
538 game.hands[seat].each do |card|
539 (0...4).each do |other|
540 next if other == seat
541 if game.hands[other].any? { |held| held / 13 == card / 13 && held % 13 > card % 13 }
542 fail_command('claim rejected')
543 end
544 end
545 end
546 game.claim(seat)
547 when 'LEGAL'
548 return ["legal #{SEATS[game.actor]}: #{cards_text(game.legal)} bot #{card_text(game.choose_play)}"]
549 when 'PREVIEW'
550 return game.legal.sort.map do |card|
551 copy = Marshal.load(Marshal.dump(game))
552 count = copy.tricks.size
553 copy.play(card)
554 copy.play(copy.choose_play) while copy.state == 'playing' && copy.tricks.size == count
555 trick = copy.tricks.last
556 "preview #{card_text(card)} -> #{SEATS[copy.winner(trick)]} #{trick.sum { |_, played| copy.point(played) }}"
557 end
558 when 'SCORE'
559 output = [pairs('scores', game.scores)]
560 output << pairs('taken', game.taken.map { |held| held.sum { |card| game.point(card) } }) if game.state == 'playing'
561 return output
562 when 'TAKEN'
563 return (0...4).map { |s| "#{SEATS[s]}: #{cards_text(game.taken[s].select { |card| game.point(card) != 0 })}" }
564 when 'TRICKS'
565 output = game.tricks.each_with_index.map { |trick, i| game.trick_line(trick, i + 1) }
566 output << 'current: ' + game.current.map { |s, card| "#{SEATS[s]} #{card_text(card)}" }.join(' ') unless game.current.empty?
567 return output.empty? ? ['no tricks'] : output
568 when 'VOIDS'
569 voids = Array.new(4) { [] }
570 (game.tricks + (game.current.empty? ? [] : [game.current])).each do |trick|
571 led = trick[0][1] / 13
572 trick.each { |s, card| voids[s] << led if card / 13 != led }
573 end
574 return [pairs('voids', voids.map { |missing| missing.empty? ? '-' : missing.uniq.sort.map { |s| SUITS[s] }.join })]
575 when 'UNSEEN'
576 played = game.taken.flatten + game.current.map(&:last)
577 return ["unseen #{SEATS[game.actor]}: #{cards_text((0...52).to_a - game.hands[game.actor] - played)}"]
578 when 'STATS' then return game.stats_lines
579 when 'HISTORY'
580 output = game.history.map do |hand|
581 line = "hand #{hand[:number]} pass #{hand[:direction]} " + pairs('points', hand[:points])
582 line += " moon #{SEATS[hand[:shooter]]}" unless hand[:shooter].nil?
583 line
584 end
585 return output.empty? ? ['no hands'] : output
586 when 'REPLAY'
587 hand = args[0].to_i
588 fail_command("bad value #{args[0]}") unless hand.between?(1, game.history.size)
589 return game.history[hand - 1][:record].dup
590 end
591 prefix + game.log.drop(before).map(&:last)
592 end
593end
