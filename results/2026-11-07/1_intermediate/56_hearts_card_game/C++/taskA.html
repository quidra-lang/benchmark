86int seat_value(const std::string &word) {
87 if (word.size() != 1 || SEATS.find(word[0]) == std::string::npos)
88 fail("bad seat " + word);
89 return static_cast<int>(SEATS.find(word[0]));
90}
91
92int card_value(const std::string &word) {
93 if (word.size() != 2 || RANKS.find(word[0]) == std::string::npos ||
94 SUITS.find(word[1]) == std::string::npos) {
95 fail("bad card " + word);
96 }
97 return static_cast<int>(SUITS.find(word[1]) * 13 + RANKS.find(word[0]));
98}
99
100Strategy strategy_value(const std::string &word) {
101 auto found = std::find(STRATEGY_NAMES.begin(), STRATEGY_NAMES.end(), word);
102 if (found == STRATEGY_NAMES.end())
103 fail("bad value " + word);
104 return static_cast<Strategy>(found - STRATEGY_NAMES.begin());
105}

107int bounded(const std::string &word, int low, int high) {
108 int value = 0;
109 const char *end = word.data() + word.size();
110 auto [last, error] = std::from_chars(word.data(), end, value);
111 if (error != std::errc() || last != end || value < low || value > high) {
112 fail("bad value " + word);
113 }
114 return value;
115}

117bool by_rank(int a, int b) {
118 return std::make_pair(a % 13, a / 13) < std::make_pair(b % 13, b / 13);
119}
120
121bool by_rank_descending(int a, int b) {
122 return std::make_pair(-(a % 13), a / 13) < std::make_pair(-(b % 13), b / 13);
123}
124
125int lowest(const CardSet &cards) {
126 return *std::min_element(cards.begin(), cards.end(), by_rank);
127}
128
129int highest(const CardSet &cards) {
130 return *std::min_element(cards.begin(), cards.end(), by_rank_descending);
131}

133std::vector<int> sorted_cards(const CardSet &cards, bool (*order)(int, int)) {
134 std::vector<int> sorted(cards.begin(), cards.end());
135 std::sort(sorted.begin(), sorted.end(), order);
136 return sorted;

139template <typename Predicate> CardSet filter_cards(const CardSet &cards, Predicate predicate) {
140 CardSet result;
141 std::copy_if(cards.begin(), cards.end(), std::inserter(result, result.end()), predicate);
142 return result;
143}

162std::array<int, 4> ranking(const std::array<int, 4> &scores) {
163 std::array<int, 4> seats = {0, 1, 2, 3};
164 std::stable_sort(seats.begin(), seats.end(),
165 [&scores](int a, int b) { return scores[a] < scores[b]; });
166 return seats;
167}

169class Generator {
170 public:
171 explicit Generator(std::int64_t seed) : state_(seed) {
172 }
173
174 std::int64_t draw() {
175 state_ = 48271 * state_ % 2147483647;
176 return state_;
177 }
178
179 void shuffle(std::vector<int> &deck) {
180 for (int i = static_cast<int>(deck.size()) - 1; i > 0; --i) {
181 int j = static_cast<int>(draw() % (i + 1));
182 std::swap(deck[i], deck[j]);
183 }
184 }
185
186 private:
187 std::int64_t state_;
188};

200struct Rules {
201 int target = 100;
202 int limit = 100;
203 std::string moon = "ADD";
204 bool jack = false;
205 bool qs_break = false;
206 bool blood = false;
207 std::string pass = "LRAH";
208
209 Rules with(const std::string &name, const std::string &word) const {
210 Rules next = *this;
211 if (name == "TARGET") {
212 next.target = bounded(word, 1, 1000);
213 } else if (name == "LIMIT") {
214 next.limit = bounded(word, 1, 100);
215 } else if (name == "MOON") {
216 if (!one_of(word, {"ADD", "SUB", "AUTO"}))
217 fail("bad value " + word);
218 next.moon = word;
219 } else if (name == "JACK") {
220 next.jack = switch_value(word);
221 } else if (name == "QSBREAK") {
222 next.qs_break = switch_value(word);
223 } else if (name == "BLOOD") {
224 next.blood = switch_value(word);
225 } else if (name == "PASS") {
226 if (word.empty() || word.size() > 8 ||
227 word.find_first_not_of("LRAH") != std::string::npos) {
228 fail("bad value " + word);
229 }
230 next.pass = word;
231 } else {
232 fail("unknown rule " + name);
233 }
234 return next;
235 }
236
237 std::string text() const {
238 return "rules TARGET " + std::to_string(target) + " LIMIT " + std::to_string(limit) +
239 " MOON " + moon + " JACK " + switch_text(jack) + " QSBREAK " +
240 switch_text(qs_break) + " BLOOD " + switch_text(blood) + " PASS " + pass;
241 }
242};

271int trick_winner(const Trick &plays) {
272 int led = plays[0].card / 13;
273 const Play *best = &plays[0];
274 for (const Play &play : plays) {
275 if (play.card / 13 == led && play.card % 13 > best->card % 13)
276 best = &play;
277 }
278 return best->seat;
279}

369 void deal() {
370 ++number;
371 pass_code = rules.pass[(number - 1) % rules.pass.size()];
372 round = Round();
373 CardSet used;
374 std::vector<int> free;
375 for (int seat = 0; seat < 4; ++seat) {
376 auto found = stacked.find(seat);
377 if (found == stacked.end()) {
378 free.push_back(seat);
379 } else {
380 round.hands[seat] = found->second;
381 used.insert(found->second.begin(), found->second.end());
382 }
383 }
384 std::vector<int> deck;
385 for (int card = 0; card < 52; ++card) {
386 if (!used.count(card))
387 deck.push_back(card);
388 }
389 generator.shuffle(deck);
390 for (std::size_t i = 0; i < deck.size(); ++i) {
391 round.hands[free[i % free.size()]].insert(deck[i]);
392 }
393 stacked.clear();
394 passes.clear();
395 undo_stack.clear();
396 write(1, "hand " + std::to_string(number) + " pass " + direction_name(pass_code));
397 for (int seat = 0; seat < 4; ++seat) {
398 write(2, seat_name(seat) + ": " + cards_text(round.hands[seat]));
399 }
400 if (pass_code == 'H') {
401 begin_play();
402 } else {
403 state = State::Passing;
404 }
405 }
406
407 void begin_play() {
408 state = State::Playing;
409 for (int seat = 0; seat < 4; ++seat) {
410 if (round.hands[seat].count(TWO_OF_CLUBS))
411 round.leader = seat;
412 }
413 write(2, "lead " + seat_name(round.leader));
414 }
415
416 void pass_cards(int seat, const CardSet &cards) {
417 for (int card : cards)
418 round.hands[seat].erase(card);
419 passes[seat] = cards;
420 write(2, seat_name(seat) + " passes " + cards_text(cards));
421 if (passes.size() == 4) {
422 int offset = pass_code == 'L' ? 1 : pass_code == 'R' ? 3 : 2;
423 for (const auto &[source, passed] : passes) {
424 round.hands[(source + offset) % 4].insert(passed.begin(), passed.end());
425 }
426 begin_play();
427 }
428 }

430 int actor() const {
431 return (round.leader + static_cast<int>(round.current.size())) % 4;
432 }
433
434 CardSet legal() const {
435 const CardSet &hand = round.hands[actor()];
436 if (round.current.empty()) {
437 if (round.tricks.empty())
438 return {TWO_OF_CLUBS};
439 CardSet non_hearts = filter_cards(hand, [](int card) { return card / 13 != 3; });
440 return !round.broken && !non_hearts.empty() ? non_hearts : hand;
441 }
442 int suit = round.current[0].card / 13;
443 CardSet followers = filter_cards(hand, [suit](int card) { return card / 13 == suit; });
444 if (!followers.empty())
445 return followers;
446 if (round.tricks.empty() && !rules.blood) {
447 CardSet safe = filter_cards(hand, [](int card) { return !penalty(card); });
448 if (!safe.empty())
449 return safe;
450 }
451 return hand;
452 }
453
454 CardSet choose_pass(int seat) const {
455 const CardSet &hand = round.hands[seat];
456 std::vector<int> low = sorted_cards(hand, by_rank);
457 std::vector<int> high = sorted_cards(hand, by_rank_descending);
458 std::vector<int> hearts;
459 std::copy_if(high.begin(), high.end(), std::back_inserter(hearts),
460 [](int card) { return card / 13 == 3; });
461 std::vector<int> special;
462 for (int card : {QS, KS, AS}) {
463 if (hand.count(card))
464 special.push_back(card);
465 }
466 std::vector<int> order;
467 switch (bots[seat]) {
468 case Strategy::Low:
469 order = high;
470 break;
471 case Strategy::High:
472 order = low;
473 break;
474 case Strategy::Moon:
475 std::copy_if(low.begin(), low.end(), std::back_inserter(order),
476 [](int card) { return card / 13 != 3; });
477 std::copy_if(low.begin(), low.end(), std::back_inserter(order),
478 [](int card) { return card / 13 == 3; });
479 break;
480 case Strategy::Smart: {
481 bool is_protected = hand.count(QS) && count_suit(hand, 2) >= 5;
482 if (!is_protected)
483 order = special;
484 int clubs = count_suit(hand, 0);
485 int diamonds = count_suit(hand, 1);
486 if (clubs > 0 || diamonds > 0) {
487 int suit = diamonds > 0 && (clubs == 0 || diamonds < clubs) ? 1 : 0;
488 std::copy_if(high.begin(), high.end(), std::back_inserter(order),
489 [suit](int card) { return card / 13 == suit; });
490 }
491 order.insert(order.end(), hearts.begin(), hearts.end());
492 order.insert(order.end(), high.begin(), high.end());
493 break;
494 }
495 case Strategy::Basic:
496 order = special;
497 order.insert(order.end(), hearts.begin(), hearts.end());
498 order.insert(order.end(), high.begin(), high.end());
499 break;
500 }
501 CardSet chosen;
502 for (int card : order) {
503 if (chosen.size() == 3)
504 break;
505 chosen.insert(card);
506 }
507 return chosen;
508 }

510 bool queen_played() const {
511 for (const CardSet &cards : round.taken) {
512 if (cards.count(QS))
513 return true;
514 }
515 return std::any_of(round.current.begin(), round.current.end(),
516 [](const Play &play) { return play.card == QS; });
517 }
518
519 bool others_took_penalty(int seat) const {
520 for (int other = 0; other < 4; ++other) {
521 const CardSet &cards = round.taken[other];
522 if (other != seat && std::any_of(cards.begin(), cards.end(), penalty))
523 return true;
524 }
525 return false;
526 }
527
528 int choose_play() const {
529 int seat = actor();
530 const CardSet &hand = round.hands[seat];
531 CardSet playable = legal();
532 Strategy strategy = bots[seat];
533 if (strategy == Strategy::Low)
534 return lowest(playable);
535 if (strategy == Strategy::High)
536 return highest(playable);
537 if (strategy == Strategy::Moon && others_took_penalty(seat))
538 strategy = Strategy::Basic;
539 if (round.current.empty()) {
540 if (strategy == Strategy::Smart && !queen_played() && !hand.count(QS)) {
541 CardSet small_spades = filter_cards(
542 playable, [](int card) { return card / 13 == 2 && card % 13 < 10; });
543 if (!small_spades.empty())
544 return lowest(small_spades);
545 }
546 int chosen_suit = -1;
547 int chosen_count = 0;
548 for (int suit = 0; suit < 4; ++suit) {
549 int count = count_suit(playable, suit);
550 if (count == 0)
551 continue;
552 bool better =
553 strategy == Strategy::Moon ? count > chosen_count : count < chosen_count;
554 if (chosen_suit < 0 || better) {
555 chosen_suit = suit;
556 chosen_count = count;
557 }
558 }
559 CardSet suited = filter_cards(
560 playable, [chosen_suit](int card) { return card / 13 == chosen_suit; });
561 return strategy == Strategy::Moon ? highest(suited) : lowest(suited);
562 }
563 int led = round.current[0].card / 13;
564 if (count_suit(hand, led) > 0) {
565 int winning = 0;
566 for (const Play &play : round.current) {
567 if (play.card / 13 == led)
568 winning = std::max(winning, play.card % 13);
569 }
570 if (strategy == Strategy::Moon) {
571 int high = highest(playable);
572 return high % 13 > winning ? high : lowest(playable);
573 }
574 CardSet lower =
575 filter_cards(playable, [winning](int card) { return card % 13 < winning; });
576 if (!lower.empty())
577 return highest(lower);
578 if (strategy == Strategy::Smart && round.current.size() == 3) {
579 CardSet without_queen = playable;
580 without_queen.erase(QS);
581 return without_queen.empty() ? QS : highest(without_queen);
582 }
583 return lowest(playable);
584 }
585 if (strategy == Strategy::Moon) {
586 CardSet safe = filter_cards(playable, [](int card) { return !penalty(card); });
587 return lowest(safe.empty() ? playable : safe);
588 }
589 if (playable.count(QS))
590 return QS;
591 if (strategy == Strategy::Smart && !queen_played()) {
592 if (playable.count(AS))
593 return AS;
594 if (playable.count(KS))
595 return KS;
596 }
597 CardSet hearts = filter_cards(playable, [](int card) { return card / 13 == 3; });
598 return highest(hearts.empty() ? playable : hearts);
599 }

606 void play(int card) {
607 int seat = actor();
608 undo_stack.push_back({round, log.size(), {seat, card}});
609 round.hands[seat].erase(card);
610 round.current.push_back({seat, card});
611 if (card / 13 == 3 || (card == QS && rules.qs_break))
612 round.broken = true;
613 if (round.current.size() == 4) {
614 int winner = trick_winner(round.current);
615 for (const Play &played : round.current)
616 round.taken[winner].insert(played.card);
617 ++round.trick_counts[winner];
618 round.tricks.push_back(round.current);
619 write(2, trick_line(round.current, round.tricks.size()));
620 round.current.clear();
621 round.leader = winner;
622 if (round.tricks.size() == 13)
623 finish_hand();
624 }
625 }
626
627 Play undo_card() {
628 Snapshot snapshot = undo_stack.back();
629 undo_stack.pop_back();
630 round = snapshot.round;
631 log.resize(snapshot.log_length);
632 return snapshot.play;
633 }

635 void claim(int seat) {
636 CardSet remaining;
637 for (CardSet &hand : round.hands) {
638 remaining.insert(hand.begin(), hand.end());
639 hand.clear();
640 }
641 int count = 13 - static_cast<int>(round.tricks.size());
642 round.taken[seat].insert(remaining.begin(), remaining.end());
643 round.trick_counts[seat] += count;
644 write(2, "claim " + seat_name(seat) + " tricks " + std::to_string(count) + " points " +
645 std::to_string(points(remaining)));
646 finish_hand();
647 }

649 bool game_over(const std::array<int, 4> &totals) const {
650 return number >= rules.limit ||
651 *std::max_element(totals.begin(), totals.end()) >= rules.target;
652 }
653
654 void finish_hand() {
655 std::array<int, 4> hand_points{};
656 std::optional<int> shooter;
657 for (int seat = 0; seat < 4; ++seat) {
658 const CardSet &cards = round.taken[seat];
659 hand_points[seat] = points(cards);
660 if (std::count_if(cards.begin(), cards.end(), penalty) == 14)
661 shooter = seat;
662 }
663 if (shooter) {
664 int moon_seat = *shooter;
665 hand_points[moon_seat] -= 26;
666 std::string method = rules.moon;
667 if (method == "AUTO") {
668 std::array<int, 4> added{};
669 for (int seat = 0; seat < 4; ++seat) {
670 added[seat] = scores[seat] + hand_points[seat] + (seat == moon_seat ? 0 : 26);
671 }
672 method = game_over(added) && ranking(added)[0] != moon_seat ? "SUB" : "ADD";
673 }
674 if (method == "SUB") {
675 hand_points[moon_seat] -= 26;
676 } else {
677 for (int seat = 0; seat < 4; ++seat) {
678 if (seat != moon_seat)
679 hand_points[seat] += 26;
680 }
681 }
682 write(1, "moon " + seat_name(moon_seat) + " " + method);
683 }
684 for (int seat = 0; seat < 4; ++seat) {
685 const CardSet &cards = round.taken[seat];
686 scores[seat] += hand_points[seat];
687 stats[seat] += Stats{round.trick_counts[seat], count_suit(cards, 3),
688 static_cast<int>(cards.count(QS)), shooter == seat ? 1 : 0};
689 }
690 write(1, pairs("points", hand_points));
691 write(1, pairs("scores", scores));
692 history.push_back({number, pass_code, hand_points, shooter, round.record});
693 bool over = game_over(scores);
694 state = over ? State::None : State::Between;
695 if (over) {
696 write(0, "game over hands " + std::to_string(number));
697 std::array<int, 4> order = ranking(scores);
698 for (int place = 0; place < 4; ++place) {
699 int seat = order[place];
700 write(0, "rank " + std::to_string(place + 1) + " " + seat_name(seat) + " " +
701 std::to_string(scores[seat]));
702 }
703 for (const std::string &line : stats_lines())
704 write(0, line);
705 }
706 }

714 }

716 void auto_pass() {
717 for (int seat = 0; seat < 4; ++seat) {
718 if (!passes.count(seat))
719 pass_cards(seat, choose_pass(seat));
720 }
721 }
722
723 void auto_hand() {
724 if (state == State::Passing)
725 auto_pass();
726 while (state == State::Playing)
727 play(choose_play());
728 }
729
730 void auto_game() {
731 if (state == State::Passing || state == State::Playing)
732 auto_hand();
733 while (state != State::None) {
734 deal();
735 auto_hand();
736 }
737 }
738};

740CardSet distinct_cards(const std::vector<int> &cards) {
741 CardSet seen;
742 for (int card : cards) {
743 if (!seen.insert(card).second)
744 fail("duplicate card " + card_text(card));
745 }
746 return seen;
747}

749std::string average_text(long long total, int games) {
750 long long numerator = 100 * total;
751 long long cents = numerator / games;
752 if (numerator % games != 0 && numerator < 0)
753 --cents;
754 std::ostringstream out;
755 if (cents < 0)
756 out << '-';
757 out << std::llabs(cents) / 100 << '.' << std::setw(2) << std::setfill('0')
758 << std::llabs(cents) % 100;
759 return out.str();
760}

806std::vector<std::string> Table::simulate(const std::string &name, int seed, int count) const {
807 Generator generator(seed);
808 if (name == "GAME") {
809 Game simulated(rules, bots, generator);
810 simulated.auto_game();
811 std::vector<std::string> output = {"game seed " + std::to_string(seed)};
812 for (const LogLine &line : simulated.log) {
813 if (line.level <= count)
814 output.push_back(line.text);
815 }
816 return output;
817 }
818 int rounds = name == "ROTATE" ? 4 : 1;
819 std::array<std::array<int, 4>, 4> places{};
820 std::array<long long, 4> totals{};
821 std::array<Stats, 4> stats{};
822 int hands = 0;
823 for (int rotation = 0; rotation < rounds; ++rotation) {
824 std::array<Strategy, 4> strategies{};
825 for (int seat = 0; seat < 4; ++seat)
826 strategies[(seat + rotation) % 4] = bots[seat];
827 for (int i = 0; i < count; ++i) {
828 Game simulated(rules, strategies, generator);
829 simulated.auto_game();
830 generator = simulated.generator;
831 hands += simulated.number;
832 std::array<int, 4> order = ranking(simulated.scores);
833 for (int original = 0; original < 4; ++original) {
834 int actual = (original + rotation) % 4;
835 ++places[original][std::find(order.begin(), order.end(), actual) - order.begin()];
836 totals[original] += simulated.scores[actual];
837 stats[original] += simulated.stats[actual];
838 }
839 }
840 }
841 int games = rounds * count;
842 std::string label = name == "ROTATE" ? "rotate" : "match";
843 std::vector<std::string> output = {label + " seed " + std::to_string(seed) + " games " +
844 std::to_string(games) + " hands " + std::to_string(hands)};
845 for (int seat = 0; seat < 4; ++seat) {
846 std::ostringstream line;
847 line << SEATS[seat];
848 if (name == "ROTATE")
849 line << ' ' << strategy_name(bots[seat]);
850 line << " places";
851 for (int place : places[seat])
852 line << ' ' << place;
853 line << " total " << totals[seat] << " average " << average_text(totals[seat], games) << ' '
854 << stats[seat].text();
855 output.push_back(line.str());
856 }
857 return output;
858}

860std::vector<std::string> Table::command(const std::vector<std::string> &words) {
861 const std::string &name = words[0];
862 std::vector<std::string> args(words.begin() + 1, words.end());
863 if (!one_of(name, {"RULE", "BOT", "SETTINGS", "GAME", "MATCH", "ROTATE", "NEW",
864 "ABANDON", "STACK", "DEAL", "SHOW", "PASS", "PLAY", "AUTO",
865 "UNDO", "CLAIM", "LEGAL", "PREVIEW", "SCORE", "TAKEN", "TRICKS",
866 "VOIDS", "UNSEEN", "STATS", "HISTORY", "REPLAY"})) {
867 fail("unknown command " + name);
868 }
869 if (name == "RULE") {
870 Rules changed = rules.with(args[0], args[1]);
871 if (active())
872 fail("game in progress");
873 rules = changed;
874 return {"rule " + args[0] + " " + args[1]};
875 }
876 if (name == "BOT") {
877 int seat = seat_value(args[0]);
878 bots[seat] = strategy_value(args[1]);
879 return {"bot " + seat_name(seat) + " " + args[1]};
880 }
881 if (name == "SETTINGS") {
882 std::array<std::string, 4> names;
883 for (int seat = 0; seat < 4; ++seat)
884 names[seat] = strategy_name(bots[seat]);
885 return {rules.text(), pairs("bots", names)};
886 }
887 if (name == "GAME" || name == "MATCH" || name == "ROTATE") {
888 int seed = bounded(args[0], 1, 2147483646);
889 int count = name == "GAME" ? bounded(args[1], 0, 2)
890 : name == "MATCH" ? bounded(args[1], 1, 50)
891 : bounded(args[1], 1, 10);
892 return simulate(name, seed, count);
893 }
894 if (name == "NEW") {
895 int seed = bounded(args[0], 1, 2147483646);
896 if (active())
897 fail("game in progress");
898 game.emplace(rules, bots, Generator(seed));
899 return {"new game seed " + std::to_string(seed)};
900 }
901 if (name == "ABANDON") {
902 require_game();
903 game.reset();
904 return {"game abandoned"};
905 }
906
907 int seat = -1;
908 std::vector<int> cards;
909 if (one_of(name, {"STACK", "PASS", "PLAY", "CLAIM"}))
910 seat = seat_value(args[0]);
911 if (one_of(name, {"STACK", "PASS", "PLAY"})) {
912 for (std::size_t i = 1; i < args.size(); ++i)
913 cards.push_back(card_value(args[i]));
914 }
915 if (name == "STACK" && cards.size() != 13)
916 fail("need 13 cards");
917 if (name == "AUTO" && !one_of(args[0], {"PASS", "CARD", "TRICK", "HAND", "GAME"})) {
918 fail("bad value " + args[0]);
919 }
920
921 if (name == "STACK" || name == "DEAL") {
922 require_between();
923 } else if (name == "SHOW") {
924 require_hand();
925 } else if (name == "PASS") {
926 require_passing();
927 } else if (one_of(name, {"PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS",
928 "VOIDS", "UNSEEN"})) {
929 require_playing();
930 } else if (one_of(name, {"SCORE", "STATS", "HISTORY", "REPLAY"})) {
931 require_game();
932 } else if (name == "AUTO") {
933 const std::string &mode = args[0];
934 if (mode == "PASS") {
935 require_passing();
936 } else if (mode == "CARD" || mode == "TRICK") {
937 require_playing();
938 } else if (mode == "HAND") {
939 require_hand();
940 } else {
941 require_game();
942 }
943 }
944 Game &current = *game;
945 Round &round = current.round;
946 std::size_t before = current.log.size();
947 std::vector<std::string> output;
948 if (name == "STACK") {
949 CardSet stacked = distinct_cards(cards);
950 for (int card : cards) {
951 for (const auto &[other, held] : current.stacked) {
952 if (other != seat && held.count(card)) {
953 fail(card_text(card) + " stacked for " + seat_name(other));
954 }
955 }
956 }
957 current.stacked[seat] = stacked;
958 return {"stacked " + seat_name(seat)};
959 }
960 if (name == "DEAL") {
961 current.deal();
962 } else if (name == "SHOW") {
963 for (int other = 0; other < 4; ++other) {
964 output.push_back(seat_name(other) + ": " + cards_text(round.hands[other]));
965 }
966 return output;
967 } else if (name == "PASS") {
968 if (current.passes.count(seat))
969 fail(seat_name(seat) + " already passed");
970 CardSet chosen = distinct_cards(cards);
971 for (int card : cards) {
972 if (!round.hands[seat].count(card)) {
973 fail(seat_name(seat) + " does not hold " + card_text(card));
974 }
975 }
976 current.pass_cards(seat, chosen);
977 } else if (name == "PLAY") {
978 int card = cards[0];
979 if (seat != current.actor())
980 fail("waiting for " + seat_name(current.actor()));
981 if (!round.hands[seat].count(card)) {
982 fail(seat_name(seat) + " does not hold " + card_text(card));
983 }
984 if (!current.legal().count(card)) {
985 if (round.current.empty()) {
986 fail(round.tricks.empty() ? "must lead 2C" : "hearts not broken");
987 }
988 int led = round.current[0].card / 13;
989 if (count_suit(round.hands[seat], led) > 0) {
990 fail(std::string("must follow ") + SUITS[led]);
991 }
992 fail("no points on first trick");
993 }
994 output.push_back(seat_name(seat) + " plays " + card_text(card));
995 current.play(card);
996 } else if (name == "AUTO") {
997 const std::string &mode = args[0];
998 if (mode == "PASS") {
999 current.auto_pass();
1000 } else if (mode == "CARD") {
1001 int actor = current.actor();
1002 int card = current.choose_play();
1003 output.push_back(seat_name(actor) + " plays " + card_text(card));
1004 current.play(card);
1005 } else if (mode == "TRICK") {
1006 std::size_t count = round.tricks.size();
1007 while (current.state == State::Playing && round.tricks.size() == count) {
1008 current.play(current.choose_play());
1009 }
1010 } else if (mode == "HAND") {
1011 current.auto_hand();
1012 } else {
1013 current.auto_game();
1014 }
1015 } else if (name == "UNDO") {
1016 if (current.undo_stack.empty())
1017 fail("nothing to undo");
1018 Play undone = current.undo_card();
1019 return {"undo " + seat_name(undone.seat) + " " + card_text(undone.card)};
1020 } else if (name == "CLAIM") {
1021 if (seat != current.actor())
1022 fail("waiting for " + seat_name(current.actor()));
1023 if (!round.current.empty())
1024 fail("trick in progress");
1025 for (int card : round.hands[seat]) {
1026 for (int other = 0; other < 4; ++other) {
1027 if (other == seat)
1028 continue;
1029 for (int other_card : round.hands[other]) {
1030 if (other_card / 13 == card / 13 && other_card % 13 > card % 13) {
1031 fail("claim rejected");
1032 }
1033 }
1034 }
1035 }
1036 current.claim(seat);
1037 } else if (name == "LEGAL") {
1038 return {"legal " + seat_name(current.actor()) + ": " + cards_text(current.legal()) +
1039 " bot " + card_text(current.choose_play())};
1040 } else if (name == "PREVIEW") {
1041 for (int card : current.legal()) {
1042 Game copy = current;
1043 std::size_t count = copy.round.tricks.size();
1044 copy.play(card);
1045 while (copy.state == State::Playing && copy.round.tricks.size() == count) {
1046 copy.play(copy.choose_play());
1047 }
1048 const Trick &trick = copy.round.tricks.back();
1049 output.push_back("preview " + card_text(card) + " -> " +
1050 seat_name(trick_winner(trick)) + " " +
1051 std::to_string(copy.trick_points(trick)));
1052 }
1053 return output;
1054 } else if (name == "SCORE") {
1055 output.push_back(pairs("scores", current.scores));
1056 if (current.state == State::Playing) {
1057 std::array<int, 4> taken{};
1058 for (int other = 0; other < 4; ++other) {
1059 taken[other] = current.points(round.taken[other]);
1060 }
1061 output.push_back(pairs("taken", taken));
1062 }
1063 return output;
1064 } else if (name == "TAKEN") {
1065 for (int other = 0; other < 4; ++other) {
1066 CardSet scoring = filter_cards(
1067 round.taken[other], [&current](int card) { return current.point(card) != 0; });
1068 output.push_back(seat_name(other) + ": " + cards_text(scoring));
1069 }
1070 return output;
1071 } else if (name == "TRICKS") {
1072 for (std::size_t i = 0; i < round.tricks.size(); ++i) {
1073 output.push_back(current.trick_line(round.tricks[i], i + 1));
1074 }
1075 if (!round.current.empty())
1076 output.push_back("current: " + plays_text(round.current));
1077 if (output.empty())
1078 output.push_back("no tricks");
1079 return output;
1080 } else if (name == "VOIDS") {
1081 std::vector<Trick> tricks = round.tricks;
1082 if (!round.current.empty())
1083 tricks.push_back(round.current);
1084 std::array<std::set<int>, 4> voids;
1085 for (const Trick &trick : tricks) {
1086 int led = trick[0].card / 13;
1087 for (const Play &play : trick) {
1088 if (play.card / 13 != led)
1089 voids[play.seat].insert(led);
1090 }
1091 }
1092 std::array<std::string, 4> texts;
1093 for (int other = 0; other < 4; ++other) {
1094 for (int suit : voids[other])
1095 texts[other] += SUITS[suit];
1096 if (texts[other].empty())
1097 texts[other] = "-";
1098 }
1099 return {pairs("voids", texts)};
1100 } else if (name == "UNSEEN") {
1101 int actor = current.actor();
1102 CardSet seen = round.hands[actor];
1103 for (const CardSet &taken : round.taken)
1104 seen.insert(taken.begin(), taken.end());
1105 for (const Play &play : round.current)
1106 seen.insert(play.card);
1107 CardSet unseen;
1108 for (int card = 0; card < 52; ++card) {
1109 if (!seen.count(card))
1110 unseen.insert(card);
1111 }
1112 return {"unseen " + seat_name(actor) + ": " + cards_text(unseen)};
1113 } else if (name == "STATS") {
1114 return current.stats_lines();
1115 } else if (name == "HISTORY") {
1116 for (const HandRecord &hand : current.history) {
1117 std::string line = "hand " + std::to_string(hand.number) + " pass " +
1118 direction_name(hand.pass_code) + " " + pairs("points", hand.points);
1119 if (hand.shooter)
1120 line += " moon " + seat_name(*hand.shooter);
1121 output.push_back(line);
1122 }
1123 if (output.empty())
1124 output.push_back("no hands");
1125 return output;
1126 } else if (name == "REPLAY") {
1127 int hand = bounded(args[0], 1, static_cast<int>(current.history.size()));
1128 return current.history[hand - 1].record;
1129 }
1130 for (std::size_t i = before; i < current.log.size(); ++i)
1131 output.push_back(current.log[i].text);
1132 return output;
1133}

1135std::vector<std::string> split(const std::string &line) {
1136 std::vector<std::string> words;
1137 std::istringstream in(line);
1138 std::string word;
1139 while (in >> word)
1140 words.push_back(word);
1141 return words;
1142}
