130 deal() {
131 this.number++;
132 this.passCode = this.rules.PASS[(this.number - 1) % this.rules.PASS.length];
133 this.hands = [0, 1, 2, 3].map((s) => (this.stacked.get(s) ?? []).slice());
134 const used = new Set(this.hands.flat());
135 const deck = Array.from({ length: 52 }, (_, c) => c).filter((c) => !used.has(c));
136 this.generator.shuffle(deck);
137 const free = [0, 1, 2, 3].filter((s) => !this.stacked.has(s));
138 deck.forEach((c, i) => this.hands[free[i % free.length]].push(c));
139 this.stacked.clear();
140 this.passes.clear();
141 this.taken = rows();
142 this.tricks = [];
143 this.counts = four();
144 this.current = [];
145 this.broken = false;
146 this.undo = [];

149 for (let s = 0; s < 4; s++) this.write(2, `${seats[s]}: ${display(this.hands[s])}`);
150 if (this.passCode === "H") this.beginPlay();

153 beginPlay() {
154 this.state = "playing";
155 this.leader = this.hands.findIndex((h) => h.includes(0));

158 passCards(s, cs) {
159 this.hands[s] = this.hands[s].filter((c) => !cs.includes(c));
160 this.passes.set(s, cs.slice());
161 this.write(2, `${seats[s]} passes ${display(cs)}`);
162 if (this.passes.size === 4) {
163 const offset = this.passCode === "L" ? 1 : this.passCode === "R" ? 3 : 2;
164 for (const [source, cards] of this.passes)

169 actor() {
170 return (this.leader + this.current.length) % 4;
171 }
172 legal() {
173 const hand = this.hands[this.actor()];
174 if (!this.current.length) {
175 if (!this.tricks.length) return [0];
176 const safe = hand.filter((c) => suit(c) !== 3);
177 return !this.broken && safe.length ? safe : hand.slice();
178 }
179 const led = suit(this.current[0][1]);
180 const follow = hand.filter((c) => suit(c) === led);
181 if (follow.length) return follow;
182 if (!this.tricks.length && this.rules.BLOOD === "OFF") {
183 const safe = hand.filter((c) => !penalty(c));
184 if (safe.length) return safe;
185 }
186 return hand.slice();
187 }
188 choosePass(s) {
189 const hand = this.hands[s],
190 strategy = this.bots[s];
191 const ascending = hand.slice().sort(lowOrder),
192 descending = hand.slice().sort(highOrder);
193 const hearts = descending.filter((c) => suit(c) === 3);
194 const special = [QS, KS, AS].filter((c) => hand.includes(c));
195 let order;
196 if (strategy === "LOW") order = descending;
197 else if (strategy === "HIGH") order = ascending;
198 else if (strategy === "MOON")
199 order = ascending
200 .filter((c) => suit(c) !== 3)
201 .concat(ascending.filter((c) => suit(c) === 3));
202 else if (strategy === "SMART") {
203 const protectedQueen =
204 hand.includes(QS) && hand.filter((c) => suit(c) === 2).length >= 5;
205 order = protectedQueen ? [] : special;
206 const choices = [0, 1].filter((su) => hand.some((c) => suit(c) === su));
207 choices.sort(
208 (a, b) =>
209 hand.filter((c) => suit(c) === a).length -
210 hand.filter((c) => suit(c) === b).length || a - b,
211 );
212 if (choices.length)
213 order = order.concat(descending.filter((c) => suit(c) === choices[0]));
214 order = order.concat(hearts, descending);
215 } else order = special.concat(hearts, descending);
216 return Array.from(new Set(order)).slice(0, 3);
217 }
218 queenPlayed() {
219 return this.taken.some((cs) => cs.includes(QS)) || this.current.some((p) => p[1] === QS);
220 }
221 choosePlay() {
222 const seat = this.actor(),
223 hand = this.hands[seat],
224 legal = this.legal();
225 let strategy = this.bots[seat];
226 if (strategy === "LOW") return low(legal);
227 if (strategy === "HIGH") return high(legal);
228 if (strategy === "MOON" && this.taken.some((cs, s) => s !== seat && cs.some(penalty)))
229 strategy = "BASIC";
230 if (!this.current.length) {
231 if (strategy === "SMART" && !this.queenPlayed() && !hand.includes(QS)) {
232 const spades = legal.filter((c) => suit(c) === 2 && c % 13 < 10);
233 if (spades.length) return low(spades);
234 }
235 const choices = [0, 1, 2, 3].filter((s) => legal.some((c) => suit(c) === s));
236 choices.sort(
237 (a, b) =>
238 (strategy === "MOON" ? -1 : 1) *
239 (legal.filter((c) => suit(c) === a).length -
240 legal.filter((c) => suit(c) === b).length) || a - b,
241 );
242 const cs = legal.filter((c) => suit(c) === choices[0]);
243 return strategy === "MOON" ? high(cs) : low(cs);
244 }
245 const led = suit(this.current[0][1]);
246 if (hand.some((c) => suit(c) === led)) {
247 const winning =
248 high(this.current.filter((p) => suit(p[1]) === led).map((p) => p[1])) % 13;
249 const lower = legal.filter((c) => c % 13 < winning);
250 if (strategy === "MOON") return high(legal) % 13 > winning ? high(legal) : low(legal);
251 if (lower.length) return high(lower);
252 if (strategy === "SMART" && this.current.length === 3) {
253 const safe = legal.filter((c) => c !== QS);
254 return safe.length ? high(safe) : QS;
255 }
256 return low(legal);
257 }
258 if (strategy === "MOON") {
259 const safe = legal.filter((c) => !penalty(c));
260 return low(safe.length ? safe : legal);
261 }
262 if (legal.includes(QS)) return QS;
263 if (strategy === "SMART" && !this.queenPlayed()) {
264 if (legal.includes(AS)) return AS;
265 if (legal.includes(KS)) return KS;
266 }
267 const hearts = legal.filter((c) => suit(c) === 3);
268 return high(hearts.length ? hearts : legal);
269 }
270 winner(plays) {
271 const led = suit(plays[0][1]);
272 return plays
273 .filter((p) => suit(p[1]) === led)
274 .sort((a, b) => (b[1] % 13) - (a[1] % 13))[0][0];
275 }
276 trickLine(plays, n) {
277 return `trick ${n}: ${plays.map(([s, c]) => `${seats[s]} ${cardText(c)}`).join(" ")} -> ${seats[this.winner(plays)]} ${sum(plays.map((p) => this.point(p[1])))}`;
278 }
279 play(c) {
280 const seat = this.actor();
281 this.undo.push({ position: this.snapshot(), logLength: this.log.length, seat, card: c });
282 this.hands[seat] = this.hands[seat].filter((v) => v !== c);
283 this.current.push([seat, c]);
284 if (suit(c) === 3 || (c === QS && this.rules.QSBREAK === "ON")) this.broken = true;
285 if (this.current.length === 4) {
286 const winner = this.winner(this.current);
287 this.taken[winner] = this.taken[winner].concat(this.current.map((p) => p[1]));
288 this.counts[winner]++;
289 this.tricks.push(this.current);
290 this.write(2, this.trickLine(this.current, this.tricks.length));
291 this.current = [];
292 this.leader = winner;
293 if (this.tricks.length === 13) this.finish();
294 }
295 }
296 undoCard() {
297 const u = this.undo.pop();
298 this.restore(u.position);
299 this.log.length = u.logLength;
300 return [u.seat, u.card];
301 }
302 claim(s) {
303 const remaining = this.hands.flat(),
304 n = 13 - this.tricks.length;
305 this.taken[s] = this.taken[s].concat(remaining);
306 this.counts[s] += n;
307 this.hands = rows();
308 this.write(
309 2,
310 `claim ${seats[s]} tricks ${n} points ${sum(remaining.map((c) => this.point(c)))}`,
311 );
312 this.finish();
313 }
314 finish() {
315 let points = this.taken.map((cs) => sum(cs.map((c) => this.point(c))));
316 const shooter = this.taken.findIndex((cs) => cs.filter(penalty).length === 14);
317 if (shooter >= 0) {
318 points[shooter] -= 26;
319 let method = this.rules.MOON;
320 if (method === "AUTO") {
321 const add = points.map((p, s) => this.scores[s] + p + (s === shooter ? 0 : 26));
322 const ended =
323 this.number >= this.rules.LIMIT || add.some((p) => p >= this.rules.TARGET);
324 const winner = [0, 1, 2, 3].sort((a, b) => add[a] - add[b] || a - b)[0];
325 method = ended && winner !== shooter ? "SUB" : "ADD";
326 }
327 if (method === "SUB") points[shooter] -= 26;
328 else points = points.map((p, s) => p + (s === shooter ? 0 : 26));
329 this.write(1, `moon ${seats[shooter]} ${method}`);
330 }
331 for (let s = 0; s < 4; s++) {
332 this.scores[s] += points[s];
333 this.stats[s][0] += this.counts[s];
334 this.stats[s][1] += this.taken[s].filter((c) => suit(c) === 3).length;
335 this.stats[s][2] += Number(this.taken[s].includes(QS));
336 this.stats[s][3] += Number(s === shooter);
337 }
338 this.write(1, pairs("points", points));
339 this.write(1, pairs("scores", this.scores));
340 this.history.push({
341 number: this.number,
342 direction: directions[this.passCode],
343 points,
344 shooter,
345 record: this.record.slice(),
346 });
347 const ended =
348 this.number >= this.rules.LIMIT || this.scores.some((p) => p >= this.rules.TARGET);
349 this.state = ended ? "none" : "between";
350 if (ended) {
351 this.write(0, `game over hands ${this.number}`);
352 this.ranking().forEach((s, i) =>
353 this.write(0, `rank ${i + 1} ${seats[s]} ${this.scores[s]}`),
354 );
355 for (const line of this.statsLines()) this.write(0, line);
356 }
357 }
358 ranking() {
359 return [0, 1, 2, 3].sort((a, b) => this.scores[a] - this.scores[b] || a - b);
360 }
361 statsLines() {
362 return this.stats.map(
363 (s, i) =>
364 `stats ${seats[i]} tricks ${s[0]} hearts ${s[1]} queens ${s[2]} moons ${s[3]}`,
365 );
366 }
367 autoPass() {
368 for (let s = 0; s < 4; s++) if (!this.passes.has(s)) this.passCards(s, this.choosePass(s));
369 }
370 autoHand() {
371 if (this.state === "passing") this.autoPass();
372 while (this.state === "playing") this.play(this.choosePlay());
373 }
374 autoGame() {
375 if (this.state === "passing" || this.state === "playing") this.autoHand();
376 while (this.state !== "none") {
377 this.deal();
378 this.autoHand();
379 }
380 }
381 preview(c) {
382 const copy = new Game(this.rules, this.bots, new Generator(this.generator.state));
383 copy.restore(this.snapshot());
384 copy.state = "playing";
385 copy.number = this.number;
386 copy.passCode = this.passCode;
387 copy.scores = this.scores.slice();
388 copy.stats = this.stats.map((s) => s.slice());
389 const count = copy.tricks.length;
390 copy.play(c);
391 while (copy.state === "playing" && copy.tricks.length === count)
392 copy.play(copy.choosePlay());
393 const trick = copy.tricks[copy.tricks.length - 1];
394 return [copy.winner(trick), sum(trick.map((p) => copy.point(p[1])))];
395 }

397class Table {
398 rules = { ...defaults };
399 bots = ["BASIC", "BASIC", "BASIC", "BASIC"];
400 game = null;
401 active() {
402 return this.game !== null && this.game.state !== "none";
403 }
404 require(kind) {
405 const state = this.active() ? this.game.state : "none";
406 if (kind === "game" && state === "none") fail("no game");
407 if (kind === "hand" && !["passing", "playing"].includes(state)) fail("no hand");
408 if (kind === "playing" && state !== "playing") fail("not playing");
409 if (kind === "passing" && state !== "passing") fail("not passing");
410 if (kind === "between") {
411 if (state === "none") fail("no game");
412 if (state !== "between") fail("hand in progress");
413 }

415 simulate(command, seed, count) {
416 const generator = new Generator(seed);
417 if (command === "GAME") {
418 const game = new Game(this.rules, this.bots, generator);
419 game.autoGame();
420 return [`game seed ${seed}`].concat(
421 game.log.filter((l) => l.level <= count).map((l) => l.text),
422 );
423 }
424 const rounds = command === "ROTATE" ? 4 : 1,
425 games = rounds * count;
426 const places = [four(), four(), four(), four()],
427 totals = four(),
428 stats = [four(), four(), four(), four()];
429 let hands = 0;
430 for (let r = 0; r < rounds; r++) {
431 const bots = this.bots.slice();
432 for (let i = 0; i < 4; i++) bots[(i + r) % 4] = this.bots[i];
433 for (let g = 0; g < count; g++) {
434 const game = new Game(this.rules, bots, generator);
435 game.autoGame();
436 hands += game.number;
437 const ranking = game.ranking();
438 for (let i = 0; i < 4; i++) {
439 const actual = (i + r) % 4;
440 places[i][ranking.indexOf(actual)]++;
441 totals[i] += game.scores[actual];
442 for (let k = 0; k < 4; k++) stats[i][k] += game.stats[actual][k];
443 }
444 }
445 }
446 const out = [`${command.toLowerCase()} seed ${seed} games ${games} hands ${hands}`];
447 for (let i = 0; i < 4; i++) {
448 const cents = Math.floor((100 * totals[i]) / games),
449 abs = Math.abs(cents);
450 const average = `${cents < 0 ? "-" : ""}${Math.floor(abs / 100)}.${String(abs % 100).padStart(2, "0")}`;
451 const name = seats[i] + (command === "ROTATE" ? ` ${this.bots[i]}` : "");
452 const s = stats[i];
453 out.push(
454 `${name} places ${places[i].join(" ")} total ${totals[i]} average ${average} tricks ${s[0]} hearts ${s[1]} queens ${s[2]} moons ${s[3]}`,
455 );
456 }
457 return out;
458 }
459 command(words) {
460 const [cmd, ...args] = words;
461 const known =
462 "RULE BOT SETTINGS GAME MATCH ROTATE NEW ABANDON STACK DEAL SHOW PASS PLAY AUTO UNDO CLAIM LEGAL PREVIEW SCORE TAKEN TRICKS VOIDS UNSEEN STATS HISTORY REPLAY".split(
463 " ",
464 );
465 if (!known.includes(cmd)) fail(`unknown command ${cmd}`);
466 if (cmd === "RULE") {
467 const [name, word] = args;
468 if (!Object.hasOwn(defaults, name)) fail(`unknown rule ${name}`);
469 let value = word;
470 if (name === "TARGET" || name === "LIMIT")
471 value = bounded(word, 1, name === "TARGET" ? 1000 : 100);
472 else {
473 const valid =
474 name === "PASS"
475 ? /^[LRAH]{1,8}$/.test(word)
476 : (name === "MOON" ? ["ADD", "SUB", "AUTO"] : ["ON", "OFF"]).includes(word);
477 if (!valid) fail(`bad value ${word}`);
478 }
479 if (this.active()) fail("game in progress");
480 if (name === "TARGET") this.rules.TARGET = value;
481 else if (name === "LIMIT") this.rules.LIMIT = value;
482 else this.rules[name] = value;
483 return [`rule ${name} ${word}`];
484 }
485 if (cmd === "BOT") {
486 const s = seatValue(args[0]);
487 if (!strategies.includes(args[1])) fail(`bad value ${args[1]}`);
488 this.bots[s] = args[1];
489 return [`bot ${seats[s]} ${args[1]}`];
490 }
491 if (cmd === "SETTINGS")
492 return [
493 "rules " +
494 Object.entries(this.rules)
495 .map(([k, v]) => `${k} ${v}`)
496 .join(" "),
497 pairs("bots", this.bots),
498 ];
499 if (["GAME", "MATCH", "ROTATE"].includes(cmd)) {
500 const seed = bounded(args[0], 1, 2147483646),
501 count = bounded(
502 args[1],
503 cmd === "GAME" ? 0 : 1,
504 cmd === "GAME" ? 2 : cmd === "MATCH" ? 50 : 10,
505 );
506 return this.simulate(cmd, seed, count);
507 }
508 if (cmd === "NEW") {
509 const seed = bounded(args[0], 1, 2147483646);
510 if (this.active()) fail("game in progress");
511 this.game = new Game(this.rules, this.bots, new Generator(seed));
512 return [`new game seed ${seed}`];
513 }
514 if (cmd === "ABANDON") {
515 this.require("game");
516 this.game = null;
517 return ["game abandoned"];
518 }
519 let s = -1,
520 cards = [];
521 if (["STACK", "PASS", "PLAY", "CLAIM"].includes(cmd)) s = seatValue(args[0]);
522 if (["STACK", "PASS", "PLAY"].includes(cmd)) cards = args.slice(1).map(cardValue);
523 if (cmd === "STACK" && cards.length !== 13) fail("need 13 cards");
524 if (cmd === "AUTO" && !["PASS", "CARD", "TRICK", "HAND", "GAME"].includes(args[0]))
525 fail(`bad value ${args[0]}`);
526 if (["STACK", "DEAL"].includes(cmd)) this.require("between");
527 else if (cmd === "SHOW") this.require("hand");
528 else if (cmd === "PASS") this.require("passing");
529 else if (
530 [
531 "PLAY",
532 "UNDO",
533 "CLAIM",
534 "LEGAL",
535 "PREVIEW",
536 "TAKEN",
537 "TRICKS",
538 "VOIDS",
539 "UNSEEN",
540 ].includes(cmd)
541 )
542 this.require("playing");
543 else if (["SCORE", "STATS", "HISTORY", "REPLAY"].includes(cmd)) this.require("game");
544 else if (cmd === "AUTO")
545 this.require(
546 args[0] === "PASS"
547 ? "passing"
548 : ["CARD", "TRICK"].includes(args[0])
549 ? "playing"
550 : args[0] === "HAND"
551 ? "hand"
552 : "game",
553 );
554 const game = this.game,
555 before = game.log.length,
556 prefix = [];
557 switch (cmd) {
558 case "STACK":
559 duplicate(cards);
560 for (const c of cards) {
561 for (let o = 0; o < 4; o++) {
562 if (o !== s && (game.stacked.get(o) ?? []).includes(c)) {
563 fail(`${cardText(c)} stacked for ${seats[o]}`);
564 }
565 }
566 }
567 game.stacked.set(s, cards.slice());
568 return [`stacked ${seats[s]}`];
569 case "DEAL":
570 game.deal();
571 break;
572 case "SHOW":
573 return game.hands.map((cs, i) => `${seats[i]}: ${display(cs)}`);
574 case "PASS":
575 if (game.passes.has(s)) fail(`${seats[s]} already passed`);
576 duplicate(cards);
577 for (const c of cards)
578 if (!game.hands[s].includes(c))
579 fail(`${seats[s]} does not hold ${cardText(c)}`);
580 game.passCards(s, cards);
581 break;
582 case "PLAY": {
583 const c = cards[0];
584 if (s !== game.actor()) fail(`waiting for ${seats[game.actor()]}`);
585 if (!game.hands[s].includes(c)) fail(`${seats[s]} does not hold ${cardText(c)}`);
586 if (!game.legal().includes(c)) {
587 if (!game.current.length)
588 fail(!game.tricks.length ? "must lead 2C" : "hearts not broken");
589 const led = suit(game.current[0][1]);
590 if (game.hands[s].some((c) => suit(c) === led))
591 fail(`must follow ${suits[led]}`);
592 fail("no points on first trick");
593 }
594 prefix.push(`${seats[s]} plays ${cardText(c)}`);
595 game.play(c);
596 break;
597 }
598 case "AUTO":
599 if (args[0] === "PASS") game.autoPass();
600 else if (args[0] === "CARD") {
601 const seat = game.actor(),
602 c = game.choosePlay();
603 prefix.push(`${seats[seat]} plays ${cardText(c)}`);
604 game.play(c);
605 } else if (args[0] === "TRICK") {
606 const count = game.tricks.length;
607 while (game.state === "playing" && game.tricks.length === count)
608 game.play(game.choosePlay());
609 } else if (args[0] === "HAND") game.autoHand();
610 else game.autoGame();
611 break;
612 case "UNDO": {
613 if (!game.undo.length) fail("nothing to undo");
614 const [seat, c] = game.undoCard();
615 return [`undo ${seats[seat]} ${cardText(c)}`];
616 }
617 case "CLAIM":
618 if (s !== game.actor()) fail(`waiting for ${seats[game.actor()]}`);
619 if (game.current.length) fail("trick in progress");
620 for (const c of game.hands[s]) {
621 if (
622 game.hands.some(
623 (cs, o) =>
624 o !== s && cs.some((v) => suit(v) === suit(c) && v % 13 > c % 13),
625 )
626 ) {
627 fail("claim rejected");
628 }
629 }
630 game.claim(s);
631 break;
632 case "LEGAL":
633 return [
634 `legal ${seats[game.actor()]}: ${display(game.legal())} bot ${cardText(game.choosePlay())}`,
635 ];
636 case "PREVIEW":
637 return game
638 .legal()
639 .sort((a, b) => a - b)
640 .map((c) => {
641 const [w, p] = game.preview(c);
642 return `preview ${cardText(c)} -> ${seats[w]} ${p}`;
643 });
644 case "SCORE":
645 return [pairs("scores", game.scores)].concat(
646 game.state === "playing"
647 ? [
648 pairs(
649 "taken",
650 game.taken.map((cs) => sum(cs.map((c) => game.point(c)))),
651 ),
652 ]
653 : [],
654 );
655 case "TAKEN":
656 return game.taken.map(
657 (cs, i) => `${seats[i]}: ${display(cs.filter((c) => game.point(c) !== 0))}`,
658 );
659 case "TRICKS": {
660 const out = game.tricks.map((t, i) => game.trickLine(t, i + 1));
661 if (game.current.length)
662 out.push(
663 "current: " +
664 game.current.map(([s, c]) => `${seats[s]} ${cardText(c)}`).join(" "),
665 );
666 return out.length ? out : ["no tricks"];
667 }
668 case "VOIDS": {
669 const missing = [new Set(), new Set(), new Set(), new Set()];
670 for (const trick of game.tricks.concat(game.current.length ? [game.current] : [])) {
671 const led = suit(trick[0][1]);
672 for (const [s, c] of trick) if (suit(c) !== led) missing[s].add(led);
673 }
674 return [
675 pairs(
676 "voids",
677 missing.map(
678 (ms) =>
679 Array.from(ms)
680 .sort()
681 .map((s) => suits[s])
682 .join("") || "-",
683 ),
684 ),
685 ];
686 }
687 case "UNSEEN": {
688 const actor = game.actor(),
689 seen = new Set(
690 game.taken.flat().concat(
691 game.current.map((p) => p[1]),
692 game.hands[actor],
693 ),
694 );
695 return [
696 `unseen ${seats[actor]}: ${display(Array.from({ length: 52 }, (_, c) => c).filter((c) => !seen.has(c)))}`,
697 ];
698 }
699 case "STATS":
700 return game.statsLines();
701 case "HISTORY":
702 return game.history.length
703 ? game.history.map(
704 (h) =>
705 `hand ${h.number} pass ${h.direction} ${pairs("points", h.points)}${h.shooter >= 0 ? ` moon ${seats[h.shooter]}` : ""}`,
706 )
707 : ["no hands"];
708 case "REPLAY": {
709 const n = Number(args[0]);
710 if (n < 1 || n > game.history.length) fail(`bad value ${args[0]}`);
711 return game.history[n - 1].record.slice();
712 }
713 }
714 return prefix.concat(game.log.slice(before).map((l) => l.text));
715 }
716}
717const lines = readFileSync(0, "utf8").split(/\r?\n/),
718 table = new Table();
719for (const line of lines.slice(1, Number(lines[0]) + 1)) {
720 try {
721 for (const result of table.command(line.split(" "))) writeFileSync(1, result + "\n");
722 } catch (e) {
723 if (e instanceof CommandError) writeFileSync(1, `error: ${e.message}\n`);
724 else throw e;
725 }
