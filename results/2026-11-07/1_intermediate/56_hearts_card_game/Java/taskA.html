212 record Rules(int target, int limit, String moon, boolean jack, boolean qsBreak, boolean blood,
213 String pass) {
214 static final Rules DEFAULT = new Rules(100, 100, "ADD", false, false, false, "LRAH");
215
216 Rules withRule(String name, String word) {
217 return switch (name) {
218 case "TARGET" ->
219 new Rules(bounded(word, 1, 1000), limit, moon, jack, qsBreak, blood, pass);
220 case "LIMIT" ->
221 new Rules(target, bounded(word, 1, 100), moon, jack, qsBreak, blood, pass);
222 case "MOON" -> new Rules(target, limit, oneOf(word, "ADD", "SUB", "AUTO"), jack,
223 qsBreak, blood, pass);
224 case "JACK" ->
225 new Rules(target, limit, moon, switchValue(word), qsBreak, blood, pass);
226 case "QSBREAK" ->
227 new Rules(target, limit, moon, jack, switchValue(word), blood, pass);
228 case "BLOOD" ->
229 new Rules(target, limit, moon, jack, qsBreak, switchValue(word), pass);
230 case "PASS" ->
231 new Rules(target, limit, moon, jack, qsBreak, blood, passValue(word));
232 default -> throw new CommandError("unknown rule " + name);
233 };
234 }

237 return "rules TARGET " + target + " LIMIT " + limit + " MOON " + moon + " JACK "
238 + onOff(jack) + " QSBREAK " + onOff(qsBreak) + " BLOOD " + onOff(blood)
239 + " PASS " + pass;
240 }

368 void deal() {
369 number++;
370 direction = Direction.of(rules.pass().charAt((number - 1) % rules.pass().length()));
371 round = new Round();
372 Set<Integer> used = new HashSet<>();
373 List<Integer> free = new ArrayList<>();
374 for (int seat = 0; seat < 4; seat++) {
375 if (stacked.containsKey(seat)) {
376 round.hands.get(seat).addAll(stacked.get(seat));
377 used.addAll(stacked.get(seat));
378 } else {
379 free.add(seat);
380 }
381 }
382 List<Integer> deck = IntStream.range(0, 52)
383 .filter(card -> !used.contains(card))
384 .boxed()
385 .collect(Collectors.toCollection(ArrayList::new));
386 generator.shuffle(deck);
387 for (int i = 0; i < deck.size(); i++) {
388 round.hands.get(free.get(i % free.size())).add(deck.get(i));
389 }
390 stacked.clear();
391 passes.clear();
392 undoStack.clear();
393 write(1, "hand " + number + " pass " + direction);
394 for (int seat = 0; seat < 4; seat++) {
395 write(2, seatName(seat) + ": " + cardsText(round.hands.get(seat)));
396 }
397 if (direction == Direction.HOLD) {
398 beginPlay();
399 } else {
400 state = State.PASSING;
401 }
402 }

404 void beginPlay() {
405 state = State.PLAYING;
406 round.leader = IntStream.range(0, 4)
407 .filter(seat -> round.hands.get(seat).contains(TWO_OF_CLUBS))
408 .findFirst()
409 .orElseThrow();
410 write(2, "lead " + seatName(round.leader));
411 }

413 void passCards(int seat, List<Integer> cards) {
414 round.hands.get(seat).removeAll(cards);
415 passes.put(seat, cards);
416 write(2, seatName(seat) + " passes " + cardsText(cards));
417 if (passes.size() == 4) {
418 passes.forEach((source, passed) ->
419 round.hands.get((source + direction.offset) % 4).addAll(passed));
420 beginPlay();
421 }

424 int actor() {
425 return (round.leader + round.current.size()) % 4;

428 Set<Integer> legal() {
429 Set<Integer> hand = round.hands.get(actor());
430 if (round.current.isEmpty()) {
431 if (round.tricks.isEmpty()) {
432 return Set.of(TWO_OF_CLUBS);
433 }
434 Set<Integer> nonHearts = filter(hand, card -> card / 13 != 3);
435 return !round.broken && !nonHearts.isEmpty() ? nonHearts : new TreeSet<>(hand);
436 }
437 int suit = round.current.getFirst().card() / 13;
438 Set<Integer> followers = filter(hand, card -> card / 13 == suit);
439 if (!followers.isEmpty()) {
440 return followers;
441 }
442 if (round.tricks.isEmpty() && !rules.blood()) {
443 Set<Integer> safe = filter(hand, card -> !penalty(card));
444 if (!safe.isEmpty()) {
445 return safe;
446 }
447 }
448 return new TreeSet<>(hand);
449 }

451 List<Integer> choosePass(int seat) {
452 Set<Integer> hand = round.hands.get(seat);
453 List<Integer> low = hand.stream().sorted(BY_RANK).toList();
454 List<Integer> high = hand.stream().sorted(BY_RANK_DESCENDING).toList();
455 List<Integer> hearts = high.stream().filter(card -> card / 13 == 3).toList();
456 List<Integer> special = Stream.of(QS, KS, AS).filter(hand::contains).toList();
457 List<Integer> order = new ArrayList<>();
458 switch (bots[seat]) {
459 case LOW -> order.addAll(high);
460 case HIGH -> order.addAll(low);
461 case MOON -> {
462 low.stream().filter(card -> card / 13 != 3).forEach(order::add);
463 low.stream().filter(card -> card / 13 == 3).forEach(order::add);
464 }
465 case SMART -> {
466 boolean isProtected = hand.contains(QS) && countSuit(hand, 2) >= 5;
467 if (!isProtected) {
468 order.addAll(special);
469 }
470 Stream.of(0, 1)
471 .filter(suit -> countSuit(hand, suit) > 0)
472 .min(Comparator.comparingInt(suit -> countSuit(hand, suit)))
473 .ifPresent(suit -> high.stream()
474 .filter(card -> card / 13 == suit)
475 .forEach(order::add));
476 order.addAll(hearts);
477 order.addAll(high);
478 }
479 case BASIC -> {
480 order.addAll(special);
481 order.addAll(hearts);
482 order.addAll(high);
483 }
484 }
485 return order.stream().distinct().limit(3).toList();
486 }

488 boolean queenPlayed() {
489 return round.taken.stream().anyMatch(cards -> cards.contains(QS))
490 || round.current.stream().anyMatch(play -> play.card() == QS);

498 int choosePlay() {
499 int seat = actor();
500 Set<Integer> hand = round.hands.get(seat);
501 Set<Integer> playable = legal();
502 Strategy strategy = bots[seat];
503 if (strategy == Strategy.LOW) {
504 return lowest(playable);
505 }
506 if (strategy == Strategy.HIGH) {
507 return highest(playable);
508 }
509 if (strategy == Strategy.MOON && othersTookPenalty(seat)) {
510 strategy = Strategy.BASIC;
511 }
512 if (round.current.isEmpty()) {
513 if (strategy == Strategy.SMART && !queenPlayed() && !hand.contains(QS)) {
514 Set<Integer> smallSpades =
515 filter(playable, card -> card / 13 == 2 && card % 13 < 10);
516 if (!smallSpades.isEmpty()) {
517 return lowest(smallSpades);
518 }
519 }
520 Comparator<Integer> bySize =
521 Comparator.comparingInt(candidate -> countSuit(playable, candidate));
522 Set<Integer> suits =
523 new TreeSet<>(playable.stream().map(card -> card / 13).toList());
524 int suit = Collections.min(suits,
525 strategy == Strategy.MOON ? bySize.reversed() : bySize);
526 Set<Integer> suited = filter(playable, card -> card / 13 == suit);
527 return strategy == Strategy.MOON ? highest(suited) : lowest(suited);
528 }
529 int led = round.current.getFirst().card() / 13;
530 if (countSuit(hand, led) > 0) {
531 int winning = round.current.stream()
532 .filter(play -> play.card() / 13 == led)
533 .mapToInt(play -> play.card() % 13)
534 .max()
535 .orElseThrow();
536 if (strategy == Strategy.MOON) {
537 int high = highest(playable);
538 return high % 13 > winning ? high : lowest(playable);
539 }
540 Set<Integer> lower = filter(playable, card -> card % 13 < winning);
541 if (!lower.isEmpty()) {
542 return highest(lower);
543 }
544 if (strategy == Strategy.SMART && round.current.size() == 3) {
545 Set<Integer> withoutQueen = filter(playable, card -> card != QS);
546 return withoutQueen.isEmpty() ? QS : highest(withoutQueen);
547 }
548 return lowest(playable);
549 }
550 if (strategy == Strategy.MOON) {
551 Set<Integer> safe = filter(playable, card -> !penalty(card));
552 return lowest(safe.isEmpty() ? playable : safe);
553 }
554 if (playable.contains(QS)) {
555 return QS;
556 }
557 if (strategy == Strategy.SMART && !queenPlayed()) {
558 if (playable.contains(AS)) {
559 return AS;
560 }
561 if (playable.contains(KS)) {
562 return KS;
563 }
564 }
565 Set<Integer> hearts = filter(playable, card -> card / 13 == 3);
566 return highest(hearts.isEmpty() ? playable : hearts);
567 }

574 void play(int card) {
575 int seat = actor();
576 undoStack.push(new Snapshot(round.copy(), log.size(), new Play(seat, card)));
577 round.hands.get(seat).remove(card);
578 round.current.add(new Play(seat, card));
579 if (card / 13 == 3 || (card == QS && rules.qsBreak())) {
580 round.broken = true;
581 }
582 if (round.current.size() == 4) {
583 int winner = trickWinner(round.current);
584 round.current.forEach(played -> round.taken.get(winner).add(played.card()));
585 round.trickCounts[winner]++;
586 round.tricks.add(round.current);
587 write(2, trickLine(round.current, round.tricks.size()));
588 round.current = new ArrayList<>();
589 round.leader = winner;
590 if (round.tricks.size() == 13) {
591 finishHand();
592 }
593 }
594 }

596 Play undoCard() {
597 Snapshot snapshot = undoStack.pop();
598 round = snapshot.round();
599 log.subList(snapshot.logLength(), log.size()).clear();
600 return snapshot.play();
601 }

603 void claim(int seat) {
604 Set<Integer> remaining = new TreeSet<>();
605 for (Set<Integer> hand : round.hands) {
606 remaining.addAll(hand);
607 hand.clear();
608 }
609 int count = 13 - round.tricks.size();
610 round.taken.get(seat).addAll(remaining);
611 round.trickCounts[seat] += count;
612 write(2, "claim " + seatName(seat) + " tricks " + count + " points "
613 + points(remaining));

618 return number >= rules.limit()
619 || Arrays.stream(totals).max().orElseThrow() >= rules.target();

622 void finishHand() {
623 int[] handPoints = new int[4];
624 int shooter = -1;
625 for (int seat = 0; seat < 4; seat++) {
626 Set<Integer> cards = round.taken.get(seat);
627 handPoints[seat] = points(cards);
628 if (cards.stream().filter(Main::penalty).count() == 14) {
629 shooter = seat;
630 }
631 }
632 if (shooter >= 0) {
633 handPoints[shooter] -= 26;
634 String method = rules.moon();
635 if (method.equals("AUTO")) {
636 int[] added = new int[4];
637 for (int seat = 0; seat < 4; seat++) {
638 added[seat] = scores[seat] + handPoints[seat] + (seat == shooter ? 0 : 26);
639 }
640 method = gameOver(added) && ranking(added).getFirst() != shooter
641 ? "SUB"
642 : "ADD";
643 }
644 if (method.equals("SUB")) {
645 handPoints[shooter] -= 26;
646 } else {
647 for (int seat = 0; seat < 4; seat++) {
648 if (seat != shooter) {
649 handPoints[seat] += 26;
650 }
651 }
652 }
653 write(1, "moon " + seatName(shooter) + " " + method);
654 }
655 for (int seat = 0; seat < 4; seat++) {
656 Set<Integer> cards = round.taken.get(seat);
657 scores[seat] += handPoints[seat];
658 Stats handStats = new Stats(round.trickCounts[seat], countSuit(cards, 3),
659 cards.contains(QS) ? 1 : 0, shooter == seat ? 1 : 0);
660 stats[seat] = stats[seat].plus(handStats);
661 }
662 write(1, pairs("points", seat -> handPoints[seat]));
663 write(1, pairs("scores", seat -> scores[seat]));
664 history.add(new HandRecord(number, direction, handPoints, shooter,
665 List.copyOf(round.record)));
666 boolean over = gameOver(scores);
667 state = over ? State.NONE : State.BETWEEN;
668 if (over) {
669 write(0, "game over hands " + number);
670 List<Integer> order = ranking(scores);
671 for (int place = 0; place < 4; place++) {
672 int seat = order.get(place);
673 write(0, "rank " + (place + 1) + " " + seatName(seat) + " " + scores[seat]);
674 }
675 statsLines().forEach(line -> write(0, line));
676 }
677 }

693 void autoHand() {
694 if (state == State.PASSING) {
695 autoPass();
696 }
697 while (state == State.PLAYING) {
698 play(choosePlay());

703 if (state == State.PASSING || state == State.PLAYING) {
704 autoHand();
705 }
706 while (state != State.NONE) {
707 deal();
708 autoHand();
709 }

754 private List<String> simulate(String command, int seed, int count) {
755 Generator generator = new Generator(seed);
756 if (command.equals("GAME")) {
757 Game simulated = new Game(rules, bots, generator);
758 simulated.autoGame();
759 List<String> output = new ArrayList<>(List.of("game seed " + seed));
760 for (LogLine line : simulated.log) {
761 if (line.level() <= count) {
762 output.add(line.text());
763 }
764 }
765 return output;
766 }
767 int rounds = command.equals("ROTATE") ? 4 : 1;
768 int[][] places = new int[4][4];
769 int[] totals = new int[4];
770 Stats[] stats = {Stats.ZERO, Stats.ZERO, Stats.ZERO, Stats.ZERO};
771 int hands = 0;
772 for (int rotation = 0; rotation < rounds; rotation++) {
773 Strategy[] strategies = new Strategy[4];
774 for (int seat = 0; seat < 4; seat++) {
775 strategies[(seat + rotation) % 4] = bots[seat];
776 }
777 for (int i = 0; i < count; i++) {
778 Game simulated = new Game(rules, strategies, generator);
779 simulated.autoGame();
780 hands += simulated.number;
781 List<Integer> order = ranking(simulated.scores);
782 for (int original = 0; original < 4; original++) {
783 int actual = (original + rotation) % 4;
784 places[original][order.indexOf(actual)]++;
785 totals[original] += simulated.scores[actual];
786 stats[original] = stats[original].plus(simulated.stats[actual]);
787 }
788 }
789 }
790 int games = rounds * count;
791 List<String> output = new ArrayList<>();
792 output.add(command.toLowerCase() + " seed " + seed + " games " + games + " hands "
793 + hands);
794 for (int seat = 0; seat < 4; seat++) {
795 String name = seatName(seat) + (command.equals("ROTATE") ? " " + bots[seat] : "");
796 String positions = Arrays.stream(places[seat])
797 .mapToObj(String::valueOf)
798 .collect(Collectors.joining(" "));
799 output.add(name + " places " + positions + " total " + totals[seat] + " average "
800 + averageText(totals[seat], games) + " " + stats[seat].text());
801 }
802 return output;
803 }

805 List<String> command(String[] words) {
806 String name = words[0];
807 List<String> args = List.of(words).subList(1, words.length);
808 if (!COMMANDS.contains(name)) {
809 throw new CommandError("unknown command " + name);
810 }
811 switch (name) {
812 case "RULE" -> {
813 Rules changed = rules.withRule(args.get(0), args.get(1));
814 if (active()) {
815 throw new CommandError("game in progress");
816 }
817 rules = changed;
818 return List.of("rule " + args.get(0) + " " + args.get(1));
819 }
820 case "BOT" -> {
821 int seat = seatValue(args.get(0));
822 bots[seat] = strategyValue(args.get(1));
823 return List.of("bot " + seatName(seat) + " " + args.get(1));
824 }
825 case "SETTINGS" -> {
826 return List.of(rules.text(), pairs("bots", seat -> bots[seat]));
827 }
828 case "GAME", "MATCH", "ROTATE" -> {
829 int seed = bounded(args.get(0), 1, 2147483646);
830 int count = switch (name) {
831 case "GAME" -> bounded(args.get(1), 0, 2);
832 case "MATCH" -> bounded(args.get(1), 1, 50);
833 default -> bounded(args.get(1), 1, 10);
834 };
835 return simulate(name, seed, count);
836 }
837 case "NEW" -> {
838 int seed = bounded(args.get(0), 1, 2147483646);
839 if (active()) {
840 throw new CommandError("game in progress");
841 }
842 game = new Game(rules, bots, new Generator(seed));
843 return List.of("new game seed " + seed);
844 }
845 case "ABANDON" -> {
846 requireGame();
847 game = null;
848 return List.of("game abandoned");
849 }
850 default -> {
851 return tableCommand(name, args);
852 }
853 }
854 }

856 private List<String> tableCommand(String name, List<String> args) {
857 int seat = Set.of("STACK", "PASS", "PLAY", "CLAIM").contains(name)
858 ? seatValue(args.get(0))
859 : -1;
860 List<Integer> cards = Set.of("STACK", "PASS", "PLAY").contains(name)
861 ? args.stream().skip(1).map(Main::cardValue).toList()
862 : List.of();
863 if (name.equals("STACK") && cards.size() != 13) {
864 throw new CommandError("need 13 cards");
865 }
866 if (name.equals("AUTO")
867 && !Set.of("PASS", "CARD", "TRICK", "HAND", "GAME").contains(args.get(0))) {
868 throw new CommandError("bad value " + args.get(0));
869 }
870
871 switch (name) {
872 case "STACK", "DEAL" -> requireBetween();
873 case "SHOW" -> requireHand();
874 case "PASS" -> requirePassing();
875 case "PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS", "VOIDS",
876 "UNSEEN" -> requirePlaying();
877 case "AUTO" -> {
878 switch (args.get(0)) {
879 case "PASS" -> requirePassing();
880 case "CARD", "TRICK" -> requirePlaying();
881 case "HAND" -> requireHand();
882 default -> requireGame();
883 }
884 }
885 default -> requireGame();
886 }
887 Round round = game.round;
888 int before = game.log.size();
889 List<String> output = new ArrayList<>();
890 switch (name) {
891 case "STACK" -> {
892 Set<Integer> stacked = distinctCards(cards);
893 for (int card : cards) {
894 for (int other = 0; other < 4; other++) {
895 if (other != seat
896 && game.stacked.getOrDefault(other, Set.of()).contains(card)) {
897 throw new CommandError(
898 cardText(card) + " stacked for " + seatName(other));
899 }
900 }
901 }
902 game.stacked.put(seat, stacked);
903 return List.of("stacked " + seatName(seat));
904 }
905 case "DEAL" -> game.deal();
906 case "SHOW" -> {
907 for (int other = 0; other < 4; other++) {
908 output.add(seatName(other) + ": " + cardsText(round.hands.get(other)));
909 }
910 return output;
911 }
912 case "PASS" -> {
913 if (game.passes.containsKey(seat)) {
914 throw new CommandError(seatName(seat) + " already passed");
915 }
916 distinctCards(cards);
917 for (int card : cards) {
918 if (!round.hands.get(seat).contains(card)) {
919 throw new CommandError(
920 seatName(seat) + " does not hold " + cardText(card));
921 }
922 }
923 game.passCards(seat, cards);
924 }
925 case "PLAY" -> {
926 int card = cards.getFirst();
927 if (seat != game.actor()) {
928 throw new CommandError("waiting for " + seatName(game.actor()));
929 }
930 if (!round.hands.get(seat).contains(card)) {
931 throw new CommandError(seatName(seat) + " does not hold " + cardText(card));
932 }
933 if (!game.legal().contains(card)) {
934 if (round.current.isEmpty()) {
935 throw new CommandError(
936 round.tricks.isEmpty() ? "must lead 2C" : "hearts not broken");
937 }
938 int led = round.current.getFirst().card() / 13;
939 if (countSuit(round.hands.get(seat), led) > 0) {
940 throw new CommandError("must follow " + SUITS.charAt(led));
941 }
942 throw new CommandError("no points on first trick");
943 }
944 output.add(seatName(seat) + " plays " + cardText(card));
945 game.play(card);
946 }
947 case "AUTO" -> {
948 switch (args.get(0)) {
949 case "PASS" -> game.autoPass();
950 case "CARD" -> {
951 int actor = game.actor();
952 int card = game.choosePlay();
953 output.add(seatName(actor) + " plays " + cardText(card));
954 game.play(card);
955 }
956 case "TRICK" -> {
957 int count = round.tricks.size();
958 while (game.state == State.PLAYING
959 && game.round.tricks.size() == count) {
960 game.play(game.choosePlay());
961 }
962 }
963 case "HAND" -> game.autoHand();
964 default -> game.autoGame();
965 }
966 }
967 case "UNDO" -> {
968 if (game.undoStack.isEmpty()) {
969 throw new CommandError("nothing to undo");
970 }
971 Play undone = game.undoCard();
972 return List.of(
973 "undo " + seatName(undone.seat()) + " " + cardText(undone.card()));
974 }
975 case "CLAIM" -> {
976 if (seat != game.actor()) {
977 throw new CommandError("waiting for " + seatName(game.actor()));
978 }
979 if (!round.current.isEmpty()) {
980 throw new CommandError("trick in progress");
981 }
982 for (int card : round.hands.get(seat)) {
983 for (int other = 0; other < 4; other++) {
984 if (other != seat && round.hands.get(other).stream().anyMatch(held ->
985 held / 13 == card / 13 && held % 13 > card % 13)) {
986 throw new CommandError("claim rejected");
987 }
988 }
989 }
990 game.claim(seat);
991 }
992 case "LEGAL" -> {
993 return List.of("legal " + seatName(game.actor()) + ": "
994 + cardsText(game.legal()) + " bot " + cardText(game.choosePlay()));
995 }
996 case "PREVIEW" -> {
997 for (int card : game.legal().stream().sorted().toList()) {
998 Game copy = game.copy();
999 int count = copy.round.tricks.size();
1000 copy.play(card);
1001 while (copy.state == State.PLAYING && copy.round.tricks.size() == count) {
1002 copy.play(copy.choosePlay());
1003 }
1004 List<Play> trick = copy.round.tricks.getLast();
1005 output.add("preview " + cardText(card) + " -> "
1006 + seatName(trickWinner(trick)) + " " + copy.trickPoints(trick));
1007 }
1008 return output;
1009 }
1010 case "SCORE" -> {
1011 output.add(pairs("scores", other -> game.scores[other]));
1012 if (game.state == State.PLAYING) {
1013 output.add(pairs("taken", other -> game.points(round.taken.get(other))));
1014 }
1015 return output;
1016 }
1017 case "TAKEN" -> {
1018 for (int other = 0; other < 4; other++) {
1019 Set<Integer> scoring =
1020 filter(round.taken.get(other), card -> game.point(card) != 0);
1021 output.add(seatName(other) + ": " + cardsText(scoring));
1022 }
1023 return output;
1024 }
1025 case "TRICKS" -> {
1026 for (int i = 0; i < round.tricks.size(); i++) {
1027 output.add(game.trickLine(round.tricks.get(i), i + 1));
1028 }
1029 if (!round.current.isEmpty()) {
1030 output.add("current: " + playsText(round.current));
1031 }
1032 return output.isEmpty() ? List.of("no tricks") : output;
1033 }
1034 case "VOIDS" -> {
1035 List<List<Play>> tricks = new ArrayList<>(round.tricks);
1036 if (!round.current.isEmpty()) {
1037 tricks.add(round.current);
1038 }
1039 List<Set<Integer>> voids = seatSets();
1040 for (List<Play> trick : tricks) {
1041 int led = trick.getFirst().card() / 13;
1042 for (Play play : trick) {
1043 if (play.card() / 13 != led) {
1044 voids.get(play.seat()).add(led);
1045 }
1046 }
1047 }
1048 return List.of(pairs("voids", other -> voids.get(other).stream()
1049 .map(suit -> String.valueOf(SUITS.charAt(suit)))
1050 .reduce(String::concat)
1051 .orElse("-")));
1052 }
1053 case "UNSEEN" -> {
1054 int actor = game.actor();
1055 Set<Integer> seen = new HashSet<>(round.hands.get(actor));
1056 round.taken.forEach(seen::addAll);
1057 round.current.forEach(play -> seen.add(play.card()));
1058 List<Integer> unseen = IntStream.range(0, 52)
1059 .filter(card -> !seen.contains(card))
1060 .boxed()
1061 .toList();
1062 return List.of("unseen " + seatName(actor) + ": " + cardsText(unseen));
1063 }
1064 case "STATS" -> {
1065 return game.statsLines();
1066 }
1067 case "HISTORY" -> {
1068 for (HandRecord hand : game.history) {
1069 String moon = hand.shooter() < 0 ? "" : " moon " + seatName(hand.shooter());
1070 output.add("hand " + hand.number() + " pass " + hand.direction() + " "
1071 + pairs("points", other -> hand.points()[other]) + moon);
1072 }
1073 return output.isEmpty() ? List.of("no hands") : output;
1074 }
1075 case "REPLAY" -> {
1076 int hand = bounded(args.get(0), 1, game.history.size());
1077 return game.history.get(hand - 1).record();
1078 }
1079 }
1080 for (LogLine line : game.log.subList(before, game.log.size())) {
1081 output.add(line.text());
1082 }
1083 return output;
1084 }
1085 }

1088 List<String> lines = new BufferedReader(new InputStreamReader(System.in)).lines().toList();
1089 int count = Integer.parseInt(lines.getFirst().trim());
1090 Table table = new Table();
1091 StringBuilder output = new StringBuilder();
1092 for (String line : lines.subList(1, Math.min(lines.size(), count + 1))) {

1096 }
1097 } catch (CommandError error) {
1098 output.append("error: ").append(error.getMessage()).append('\n');
1099 }
1100 }
