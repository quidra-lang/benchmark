3enum CardNames {
4 static let seats = Array("NESW")
5 static let suits = Array("CDSH")
6 static let ranks = Array("23456789TJQKA")
7 static let strategies = ["BASIC", "LOW", "HIGH", "SMART", "MOON"]
8}
9struct CommandError: Error { let reason: String }
10func fail(_ reason: String) throws -> Never {
11 throw CommandError(reason: reason)
12}
13func cardText(_ c: Int) -> String {
14 String(CardNames.ranks[c % 13]) + String(CardNames.suits[c / 13])
15}
16func cardsText(_ cards: Set<Int>) -> String {
17 cards.isEmpty ? "-" : cards.sorted().map(cardText).joined(separator: " ")
18}
19func penalty(_ c: Int) -> Bool {
20 c == 36 || c / 13 == 3
21}
22func ordered(_ cards: Set<Int>, high: Bool = false) -> [Int] {
23 cards.sorted { a, b in a % 13 == b % 13 ? a / 13 < b / 13 : (high ? a % 13 > b % 13 : a % 13 < b % 13) }
24}
25func extreme(_ cards: Set<Int>, high: Bool = false) -> Int {
26 ordered(cards, high: high)[0]
27}
28func direction(_ c: Character) -> String {
29 c == "L" ? "LEFT" : c == "R" ? "RIGHT" : c == "A" ? "ACROSS" : "HOLD"
30}
31func pairs<T>(_ label: String, _ values: [T]) -> String {
32 label + " " + values.enumerated().map { "\(CardNames.seats[$0.offset]) \($0.element)" }.joined(separator: " ")
33}
34func parseSeat(_ s: String) throws -> Int {
35 guard s.count == 1, let c = s.first, let seat = CardNames.seats.firstIndex(of: c) else {
36 try fail("bad seat " + s)
37 }
38 return seat
39}
40func parseCard(_ s: String) throws -> Int {
41 let c = Array(s)
42 guard c.count == 2, let rank = CardNames.ranks.firstIndex(of: c[0]),
43 let suit = CardNames.suits.firstIndex(of: c[1])
44 else {
45 try fail("bad card " + s)
46 }
47 return suit * 13 + rank
48}
49func bounded(_ s: String, _ low: Int, _ high: Int) throws -> Int {
50 guard let n = Int(s), n >= low && n <= high else {
51 try fail("bad value " + s)
52 }
53 return n
54}
55func checkedSet(_ cards: [Int]) throws -> Set<Int> {
56 var seen = Set<Int>()
57 for c in cards {
58 if !seen.insert(c).inserted {
59 try fail("duplicate card " + cardText(c))
60 }
61 }
62 return seen
63}
64struct Rules {
65 var target = 100, limit = 100
66 var moon = "ADD", pass = "LRAH"
67 var jack = false, qsBreak = false, blood = false
68 func validate(_ name: String, _ value: String) throws {
69 let valid: Bool
70 switch name {
71 case "TARGET":
72 _ = try bounded(value, 1, 1000)
73 return
74 case "LIMIT":
75 _ = try bounded(value, 1, 100)
76 return
77 case "MOON":
78 valid = ["ADD", "SUB", "AUTO"].contains(value)
79 case "JACK", "QSBREAK", "BLOOD":
80 valid = ["ON", "OFF"].contains(value)
81 case "PASS":
82 valid = (1...8).contains(value.count) && value.allSatisfy { "LRAH".contains($0) }
83 default:
84 try fail("unknown rule " + name)
85 }
86 if !valid {
87 try fail("bad value " + value)
88 }
89 }
90 mutating func set(_ name: String, _ value: String) {
91 switch name {
92 case "TARGET":
93 target = Int(value)!
94 case "LIMIT":
95 limit = Int(value)!
96 case "MOON":
97 moon = value
98 case "PASS":
99 pass = value
100 case "JACK":
101 jack = value == "ON"
102 case "QSBREAK":
103 qsBreak = value == "ON"
104 case "BLOOD":
105 blood = value == "ON"
106 default:
107 break
108 }
109 }
110 var text: String {
111 "rules TARGET \(target) LIMIT \(limit) MOON \(moon) JACK \(jack ? "ON" : "OFF") QSBREAK \(qsBreak ? "ON" : "OFF") BLOOD \(blood ? "ON" : "OFF") PASS \(pass)"
112 }
113}
114enum TableState { case none, between, passing, playing }
115struct Play { let seat: Int, card: Int }
116struct LiveHand {
117 var hands = Array(repeating: Set<Int>(), count: 4)
118 var taken = Array(repeating: Set<Int>(), count: 4)
119 var tricks: [[Play]] = []
120 var current: [Play] = []
121 var counts = [0, 0, 0, 0]
122 var leader = 0, broken = false
123}
124struct Snapshot { let hand: LiveHand, logLength: Int, seat: Int, card: Int }
125struct History { let number: Int, pass: Character, points: [Int], shooter: Int?, begin: Int, end: Int }
126struct RecordLine { let level: Int, text: String }
127struct Game {
128 let rules: Rules
129 var bots: [Int]
130 var random: Int64
131 var state = TableState.between
132 var number = 0
133 var scores = [0, 0, 0, 0]
134 var stats = Array(repeating: [0, 0, 0, 0], count: 4)
135 var stacked: [Int: Set<Int>] = [:]
136 var passes: [Int: Set<Int>] = [:]
137 var pass: Character = "L"
138 var hand = LiveHand()
139 var undo: [Snapshot] = []
140 var history: [History] = []
141 var log: [RecordLine] = []
142 var handBegin = 0
143 init(_ rules: Rules, _ bots: [Int], _ seed: Int64) {
144 self.rules = rules
145 self.bots = bots
146 random = seed
147 }
148 func point(_ c: Int) -> Int {
149 c == 36 ? 13 : c / 13 == 3 ? 1 : c == 22 && rules.jack ? -10 : 0
150 }
151 func points(_ cards: Set<Int>) -> Int {
152 cards.reduce(0) { $0 + point($1) }
153 }
154 mutating func write(_ level: Int, _ text: String) {
155 log.append(RecordLine(level: level, text: text))
156 }
157 mutating func deal() {
158 number += 1
159 let cycle = Array(rules.pass)
160 pass = cycle[(number - 1) % cycle.count]
161 hand = LiveHand()
162 var used = Set<Int>(), free: [Int] = []
163 for s in 0..<4 {
164 if let cards = stacked[s] {
165 hand.hands[s] = cards
166 used.formUnion(cards)
167 } else {
168 free.append(s)
169 }
170 }
171 var deck = (0..<52).filter { !used.contains($0) }
172 if deck.count >= 2 {
173 for i in stride(from: deck.count - 1, through: 1, by: -1) {
174 random = random * 48271 % 2147483647
175 deck.swapAt(i, Int(random % Int64(i + 1)))
176 }
177 }
178 for (i, c) in deck.enumerated() {
179 hand.hands[free[i % free.count]].insert(c)
180 }
181 stacked = [:]
182 passes = [:]
183 undo = []
184 handBegin = log.count
185 write(1, "hand \(number) pass \(direction(pass))")
186 for s in 0..<4 {
187 write(2, "\(CardNames.seats[s]): \(cardsText(hand.hands[s]))")
188 }
189 if pass == "H" {
190 beginPlay()
191 } else {
192 state = .passing
193 }
194 }
195 mutating func beginPlay() {
196 state = .playing
197 hand.leader = (0..<4).first { hand.hands[$0].contains(0) }!
198 write(2, "lead \(CardNames.seats[hand.leader])")
199 }
200 mutating func passCards(_ seat: Int, _ cards: Set<Int>) {
201 hand.hands[seat].subtract(cards)
202 passes[seat] = cards
203 write(2, "\(CardNames.seats[seat]) passes \(cardsText(cards))")
204 if passes.count == 4 {
205 let offset = pass == "L" ? 1 : pass == "R" ? 3 : 2
206 for s in 0..<4 {
207 hand.hands[(s + offset) % 4].formUnion(passes[s]!)
208 }
209 beginPlay()
210 }
211 }
212 var actor: Int {
213 (hand.leader + hand.current.count) % 4
214 }
215 func legal() -> Set<Int> {
216 let cards = hand.hands[actor]
217 if hand.current.isEmpty {
218 if hand.tricks.isEmpty {
219 return [0]
220 }
221 let safe = cards.filter { $0 / 13 != 3 }
222 return !hand.broken && !safe.isEmpty ? safe : cards
223 }
224 let follow = cards.filter { $0 / 13 == hand.current[0].card / 13 }
225 if !follow.isEmpty {
226 return follow
227 }
228 if hand.tricks.isEmpty && !rules.blood {
229 let safe = cards.filter { !penalty($0) }
230 if !safe.isEmpty {
231 return safe
232 }
233 }
234 return cards
235 }
236 func choosePass(_ seat: Int) -> Set<Int> {
237 let cards = hand.hands[seat], strategy = bots[seat]
238 let low = ordered(cards), high = ordered(cards, high: true)
239 var order: [Int] = []
240 switch strategy {
241 case 1:
242 order = high
243 case 2:
244 order = low
245 case 4:
246 order = low.filter { $0 / 13 != 3 } + low.filter { $0 / 13 == 3 }
247 default:
248 let protect = strategy == 3 && cards.contains(36) && cards.filter { $0 / 13 == 2 }.count >= 5
249 if !protect {
250 order += [36, 37, 38].filter { cards.contains($0) }
251 }
252 if strategy == 3 {
253 let suits = [0, 1].filter { s in cards.contains { $0 / 13 == s } }.sorted { a, b in
254 let ca = cards.filter { $0 / 13 == a }.count, cb = cards.filter { $0 / 13 == b }.count
255 return ca == cb ? a < b : ca < cb
256 }
257 if let suit = suits.first {
258 order += high.filter { $0 / 13 == suit }
259 }
260 }
261 order += high.filter { $0 / 13 == 3 }
262 order += high
263 }
264 var out = Set<Int>()
265 for c in order {
266 out.insert(c)
267 if out.count == 3 {
268 break
269 }
270 }
271 return out
272 }
273 func queenPlayed() -> Bool {
274 hand.taken.contains { $0.contains(36) } || hand.current.contains { $0.card == 36 }
275 }
276 func choosePlay() -> Int {
277 let seat = actor, cards = hand.hands[seat], choices = legal()
278 var strategy = bots[seat]
279 if strategy == 1 || strategy == 2 {
280 return extreme(choices, high: strategy == 2)
281 }
282 if strategy == 4 && (0..<4)
283 .contains(where: { $0 != seat && hand.taken[$0].contains(where: penalty) })
284 {
285 strategy = 0
286 }
287 if hand.current.isEmpty {
288 let spades = choices.filter { $0 / 13 == 2 && $0 % 13 < 10 }
289 if strategy == 3 && !queenPlayed() && !cards.contains(36) && !spades.isEmpty {
290 return extreme(spades)
291 }
292 let suits = (0..<4).filter { s in choices.contains { $0 / 13 == s } }.sorted { a, b in
293 let ca = choices.filter { $0 / 13 == a }.count, cb = choices.filter { $0 / 13 == b }.count
294 return ca == cb ? a < b : (strategy == 4 ? ca > cb : ca < cb)
295 }
296 return extreme(choices.filter { $0 / 13 == suits[0] }, high: strategy == 4)
297 }
298 let led = hand.current[0].card / 13
299 if cards.contains(where: { $0 / 13 == led }) {
300 let winning = hand.current.filter { $0.card / 13 == led }.map { $0.card % 13 }.max()!
301 let lower = choices.filter { $0 % 13 < winning }
302 if strategy == 4 {
303 let high = extreme(choices, high: true)
304 return high % 13 > winning ? high : extreme(choices)
305 }
306 if !lower.isEmpty {
307 return extreme(lower, high: true)
308 }
309 if strategy == 3 && hand.current.count == 3 {
310 let safe = choices.subtracting([36])
311 return safe.isEmpty ? 36 : extreme(safe, high: true)
312 }
313 return extreme(choices)
314 }
315 if strategy == 4 {
316 let safe = choices.filter { !penalty($0) }
317 return extreme(safe.isEmpty ? choices : safe)
318 }
319 if choices.contains(36) {
320 return 36
321 }
322 if strategy == 3 && !queenPlayed() {
323 if choices.contains(38) {
324 return 38
325 }
326 if choices.contains(37) {
327 return 37
328 }
329 }
330 let hearts = choices.filter { $0 / 13 == 3 }
331 return extreme(hearts.isEmpty ? choices : hearts, high: true)
332 }
333 func winner(_ plays: [Play]) -> Int {
334 plays.filter { $0.card / 13 == plays[0].card / 13 }
335 .max { $0.card % 13 < $1.card % 13 }!.seat
336 }
337 func trickLine(_ plays: [Play], _ number: Int) -> String {
338 let sequence = plays.map { "\(CardNames.seats[$0.seat]) \(cardText($0.card))" }.joined(separator: " ")
339 return "trick \(number): \(sequence) -> \(CardNames.seats[winner(plays)]) \(plays.reduce(0) { $0 + point($1.card) })"
340 }
341 mutating func play(_ card: Int) {
342 let seat = actor
343 undo.append(Snapshot(hand: hand, logLength: log.count, seat: seat, card: card))
344 hand.hands[seat].remove(card)
345 hand.current.append(Play(seat: seat, card: card))
346 if card / 13 == 3 || (card == 36 && rules.qsBreak) {
347 hand.broken = true
348 }
349 if hand.current.count == 4 {
350 let seat = winner(hand.current)
351 hand.taken[seat].formUnion(hand.current.map(\.card))
352 hand.counts[seat] += 1
353 hand.tricks.append(hand.current)
354 write(2, trickLine(hand.current, hand.tricks.count))
355 hand.current = []
356 hand.leader = seat
357 if hand.tricks.count == 13 {
358 finishHand()
359 }
360 }
361 }
362 mutating func undoCard() -> Play {
363 let s = undo.removeLast()
364 hand = s.hand
365 log.removeSubrange(s.logLength..<log.count)
366 return Play(seat: s.seat, card: s.card)
367 }
368 mutating func claim(_ seat: Int) {
369 let all = hand.hands.reduce(Set<Int>()) { $0.union($1) }, count = 13 - hand.tricks.count
370 hand.taken[seat].formUnion(all)
371 hand.counts[seat] += count
372 hand.hands = Array(repeating: [], count: 4)
373 write(2, "claim \(CardNames.seats[seat]) tricks \(count) points \(points(all))")
374 finishHand()
375 }
376 func ranking(_ values: [Int]) -> [Int] {
377 (0..<4).sorted { a, b in
378 values[a] == values[b] ? a < b : values[a] < values[b]
379 }
380 }
381 func ended(_ values: [Int]) -> Bool {
382 number >= rules.limit || values.contains { $0 >= rules.target }
383 }
384 mutating func finishHand() {
385 var points = hand.taken.map { self.points($0) }
386 let shooter = (0..<4).first { hand.taken[$0].filter(penalty).count == 14 }
387 if let shooter {
388 points[shooter] -= 26
389 var method = rules.moon
390 if method == "AUTO" {
391 let add = (0..<4).map { scores[$0] + points[$0] + ($0 == shooter ? 0 : 26) }
392 method = ended(add) && ranking(add)[0] != shooter ? "SUB" : "ADD"
393 }
394 if method == "SUB" {
395 points[shooter] -= 26
396 } else {
397 for s in 0..<4 where s != shooter {
398 points[s] += 26
399 }
400 }
401 write(1, "moon \(CardNames.seats[shooter]) \(method)")
402 }
403 for s in 0..<4 {
404 scores[s] += points[s]
405 stats[s][0] += hand.counts[s]
406 stats[s][1] += hand.taken[s].filter { $0 / 13 == 3 }.count
407 stats[s][2] += hand.taken[s].contains(36) ? 1 : 0
408 stats[s][3] += shooter == s ? 1 : 0
409 }
410 write(1, pairs("points", points))
411 write(1, pairs("scores", scores))
412 history.append(History(
413 number: number,
414 pass: pass,
415 points: points,
416 shooter: shooter,
417 begin: handBegin,
418 end: log.count
419 ))
420 state = ended(scores) ? .none : .between
421 if state == .none {
422 write(0, "game over hands \(number)")
423 for (rank, seat) in ranking(scores).enumerated() {
424 write(
425 0,
426 "rank \(rank + 1) \(CardNames.seats[seat]) \(scores[seat])"
427 )
428 }
429 for s in statsLines() {
430 write(0, s)
431 }
432 }
433 }
434 func statsLines() -> [String] {
435 (0..<4)
436 .map { s in
437 "stats \(CardNames.seats[s]) tricks \(stats[s][0]) hearts \(stats[s][1]) queens \(stats[s][2]) moons \(stats[s][3])"
438 }
439 }
440 mutating func autoPass() {
441 for s in 0..<4 where passes[s] == nil {
442 let cards = choosePass(s)
443 passCards(s, cards)
444 }
445 }
446 mutating func autoHand() {
447 if state == .passing {
448 autoPass()
449 }
450 while state == .playing {
451 let c = choosePlay()
452 play(c)
453 }
454 }
455 mutating func autoGame() {
456 if state == .passing || state == .playing {
457 autoHand()
458 }
459 while state != .none {
460 deal()
461 autoHand()
462 }
463 }
464}
465final class Table {
466 var rules = Rules(), bots = [0, 0, 0, 0]
467 var game: Game?
468 var active: Bool {
469 game != nil && game!.state != .none
470 }
471 func simulate(_ command: String, _ seed: Int, _ count: Int) -> [String] {
472 var random = Int64(seed)
473 if command == "GAME" {
474 var g = Game(rules, bots, random)
475 g.autoGame()
476 return ["game seed \(seed)"] + g.log.filter { $0.level <= count }.map(\.text)
477 }
478 let rounds = command == "ROTATE" ? 4 : 1, games = (command == "ROTATE" ? 4 : 1) * count
479 var hands = 0, places = Array(repeating: [0, 0, 0, 0], count: 4), totals = [0, 0, 0, 0], stats = Array(
480 repeating: [0, 0, 0, 0],
481 count: 4
482 )
483 for r in 0..<rounds {
484 var strategies = [0, 0, 0, 0]
485 for s in 0..<4 {
486 strategies[(s + r) % 4] = bots[s]
487 }
488 for _ in 0..<count {
489 var g = Game(rules, strategies, random)
490 g.autoGame()
491 random = g.random
492 hands += g.number
493 let ranks = g.ranking(g.scores)
494 for s in 0..<4 {
495 let actual = (s + r) % 4
496 places[s][ranks.firstIndex(of: actual)!] += 1
497 totals[s] += g.scores[actual]
498 for k in 0..<4 {
499 stats[s][k] += g.stats[actual][k]
500 }
501 }
502 }
503 }
504 var output = ["\(command.lowercased()) seed \(seed) games \(games) hands \(hands)"]
505 for s in 0..<4 {
506 let numerator = 100 * totals[s]
507 var cents = numerator / games
508 if numerator < 0 && numerator % games != 0 {
509 cents -= 1
510 }
511 let magnitude = abs(cents), average = "\(cents < 0 ? "-" : "")\(magnitude / 100)." + String(
512 format: "%02d",
513 magnitude % 100
514 )
515 let name = String(CardNames.seats[s]) + (command == "ROTATE" ? " " + CardNames.strategies[bots[s]] : "")
516 output
517 .append(
518 "\(name) places \(places[s].map(String.init).joined(separator: " ")) total \(totals[s]) average \(average) tricks \(stats[s][0]) hearts \(stats[s][1]) queens \(stats[s][2]) moons \(stats[s][3])"
519 )
520 }
521 return output
522 }
523 func command(_ line: String) throws -> [String] {
524 let words = line.split(separator: " ").map(String.init)
525 let command = words[0]
526 let a = Array(words.dropFirst())
527 let known = [
528 "RULE",
529 "BOT",
530 "SETTINGS",
531 "GAME",
532 "MATCH",
533 "ROTATE",
534 "NEW",
535 "ABANDON",
536 "STACK",
537 "DEAL",
538 "SHOW",
539 "PASS",
540 "PLAY",
541 "AUTO",
542 "UNDO",
543 "CLAIM",
544 "LEGAL",
545 "PREVIEW",
546 "SCORE",
547 "TAKEN",
548 "TRICKS",
549 "VOIDS",
550 "UNSEEN",
551 "STATS",
552 "HISTORY",
553 "REPLAY"
554 ]
555 if !known.contains(command) {
556 try fail("unknown command " + command)
557 }
558 switch command {
559 case "RULE":
560 try rules.validate(a[0], a[1])
561 if active {
562 try fail("game in progress")
563 }
564 rules.set(a[0], a[1])
565 return ["rule \(a[0]) \(a[1])"]
566 case "BOT":
567 let seat = try parseSeat(a[0])
568 guard let strategy = CardNames.strategies.firstIndex(of: a[1]) else {
569 try fail("bad value " + a[1])
570 }
571 bots[seat] = strategy
572 game?.bots[seat] = strategy
573 return ["bot \(a[0]) \(a[1])"]
574 case "SETTINGS":
575 return [rules.text, pairs("bots", bots.map { CardNames.strategies[$0] })]
576 case "GAME", "MATCH", "ROTATE":
577 let seed = try bounded(a[0], 1, 2147483646), count = try bounded(
578 a[1],
579 command == "GAME" ? 0 : 1,
580 command == "GAME" ? 2 : command == "MATCH" ? 50 : 10
581 )
582 return simulate(command, seed, count)
583 case "NEW":
584 let seed = try bounded(a[0], 1, 2147483646)
585 if active {
586 try fail("game in progress")
587 }
588 game = Game(rules, bots, Int64(seed))
589 return ["new game seed \(seed)"]
590 case "ABANDON":
591 if !active {
592 try fail("no game")
593 }
594 game = nil
595 return ["game abandoned"]
596 default:
597 break
598 }
599 var seat = 0, cards: [Int] = []
600 if ["STACK", "PASS", "PLAY", "CLAIM"].contains(command) {
601 seat = try parseSeat(a[0])
602 }
603 if ["STACK", "PASS", "PLAY"]
604 .contains(command)
605 {
606 for word in a.dropFirst() {
607 cards.append(try parseCard(word))
608 }
609 }
610 if command == "STACK" && cards.count != 13 {
611 try fail("need 13 cards")
612 }
613 let mode = command == "AUTO" ? a[0] : ""
614 if command == "AUTO" && !["PASS", "CARD", "TRICK", "HAND", "GAME"]
615 .contains(mode)
616 {
617 try fail("bad value " + mode)
618 }
619 let state = game?.state ?? .none
620 if command == "STACK" || command == "DEAL" {
621 if state == .none {
622 try fail("no game")
623 }
624 if state != .between {
625 try fail("hand in progress")
626 }
627 } else if command == "SHOW" || mode == "HAND" {
628 if state != .passing && state != .playing {
629 try fail("no hand")
630 }
631 } else if command == "PASS" || mode == "PASS" {
632 if state != .passing {
633 try fail("not passing")
634 }
635 } else if ["PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS", "VOIDS", "UNSEEN"]
636 .contains(command) || mode == "CARD" || mode == "TRICK"
637 {
638 if state != .playing {
639 try fail("not playing")
640 }
641 } else if state == .none {
642 try fail("no game")
643 }
644 var g = game!, output: [String] = []
645 let before = g.log.count
646 switch command {
647 case "STACK":
648 let set = try checkedSet(cards)
649 for c in cards {
650 for other in 0..<4 where other != seat {
651 if g.stacked[other]?
652 .contains(c) == true
653 {
654 try fail("\(cardText(c)) stacked for \(CardNames.seats[other])")
655 }
656 }
657 }
658 g.stacked[seat] = set
659 game = g
660 return ["stacked \(CardNames.seats[seat])"]
661 case "DEAL":
662 g.deal()
663 case "SHOW":
664 return (0..<4).map { "\(CardNames.seats[$0]): \(cardsText(g.hand.hands[$0]))" }
665 case "PASS":
666 if g.passes[seat] != nil {
667 try fail("\(CardNames.seats[seat]) already passed")
668 }
669 let set = try checkedSet(cards)
670 for c in cards
671 where !g.hand.hands[seat]
672 .contains(c)
673 {
674 try fail("\(CardNames.seats[seat]) does not hold \(cardText(c))")
675 }
676 g.passCards(seat, set)
677 case "PLAY":
678 let card = cards[0]
679 if seat != g.actor {
680 try fail("waiting for \(CardNames.seats[g.actor])")
681 }
682 if !g.hand.hands[seat]
683 .contains(card)
684 {
685 try fail("\(CardNames.seats[seat]) does not hold \(cardText(card))")
686 }
687 if !g.legal().contains(card) {
688 if g.hand.current.isEmpty {
689 try fail(g.hand.tricks.isEmpty ? "must lead 2C" : "hearts not broken")
690 }
691 let led = g.hand.current[0].card / 13
692 if g.hand.hands[seat]
693 .contains(where: { $0 / 13 == led })
694 {
695 try fail("must follow \(CardNames.suits[led])")
696 }
697 try fail("no points on first trick")
698 }
699 output.append("\(CardNames.seats[seat]) plays \(cardText(card))")
700 g.play(card)
701 case "AUTO":
702 switch mode {
703 case "PASS":
704 g.autoPass()
705 case "CARD":
706 let seat = g.actor, card = g.choosePlay()
707 output.append("\(CardNames.seats[seat]) plays \(cardText(card))")
708 g.play(card)
709 case "TRICK":
710 let count = g.hand.tricks.count
711 while g.state == .playing && g.hand.tricks.count == count {
712 let c = g.choosePlay()
713 g.play(c)
714 }
715 case "HAND":
716 g.autoHand()
717 default:
718 g.autoGame()
719 }
720 case "UNDO":
721 if g.undo.isEmpty {
722 try fail("nothing to undo")
723 }
724 let p = g.undoCard()
725 game = g
726 return ["undo \(CardNames.seats[p.seat]) \(cardText(p.card))"]
727 case "CLAIM":
728 if seat != g.actor {
729 try fail("waiting for \(CardNames.seats[g.actor])")
730 }
731 if !g.hand.current.isEmpty {
732 try fail("trick in progress")
733 }
734 for c in g.hand.hands[seat] {
735 for other in 0..<4 where other != seat {
736 if g.hand.hands[other]
737 .contains(where: { $0 / 13 == c / 13 && $0 % 13 > c % 13 })
738 {
739 try fail("claim rejected")
740 }
741 }
742 }
743 g.claim(seat)
744 case "LEGAL":
745 return [
746 "legal \(CardNames.seats[g.actor]): \(cardsText(g.legal())) bot \(cardText(g.choosePlay()))"
747 ]
748 case "PREVIEW":
749 for c in g.legal().sorted() {
750 var copy = g
751 let count = copy.hand.tricks.count
752 copy.play(c)
753 while copy.state == .playing && copy.hand.tricks.count == count {
754 let c = copy.choosePlay()
755 copy.play(c)
756 }
757 let trick = copy.hand.tricks.last!
758 output
759 .append(
760 "preview \(cardText(c)) -> \(CardNames.seats[copy.winner(trick)]) \(trick.reduce(0) { $0 + copy.point($1.card) })"
761 )
762 }
763 return output
764 case "SCORE":
765 output.append(pairs("scores", g.scores))
766 if g.state == .playing {
767 output.append(pairs("taken", g.hand.taken.map { g.points($0) }))
768 }
769 return output
770 case "TAKEN":
771 return (0..<4)
772 .map { s in "\(CardNames.seats[s]): \(cardsText(g.hand.taken[s].filter { g.point($0) != 0 }))" }
773 case "TRICKS":
774 for (i, trick) in g.hand.tricks.enumerated() {
775 output.append(g.trickLine(trick, i + 1))
776 }
777 if !g.hand.current
778 .isEmpty
779 {
780 output
781 .append("current: " + g.hand.current.map { "\(CardNames.seats[$0.seat]) \(cardText($0.card))" }
782 .joined(separator: " "))
783 }
784 return output.isEmpty ? ["no tricks"] : output
785 case "VOIDS":
786 var missing = Array(repeating: Set<Int>(), count: 4)
787 for trick in g.hand.tricks + [g.hand.current] where !trick.isEmpty {
788 let led = trick[0].card / 13
789 for p in trick where p.card / 13 != led {
790 missing[p.seat].insert(led)
791 }
792 }
793 return [pairs("voids", missing.map { $0.isEmpty ? "-" : String($0.sorted().map { CardNames.suits[$0] }) })]
794 case "UNSEEN":
795 let seen = g.hand.taken.reduce(g.hand.hands[g.actor]) { $0.union($1) }.union(g.hand.current.map(\.card))
796 return ["unseen \(CardNames.seats[g.actor]): \(cardsText(Set(0..<52).subtracting(seen)))"]
797 case "STATS":
798 return g.statsLines()
799 case "HISTORY":
800 return g.history.isEmpty ? ["no hands"] : g.history.map { h in
801 "hand \(h.number) pass \(direction(h.pass)) \(pairs("points", h.points))" +
802 (h.shooter.map { " moon \(CardNames.seats[$0])" } ?? "")
803 }
804 case "REPLAY":
805 let number = try bounded(a[0], 1, g.history.count), h = g.history[number - 1]
806 return g.log[h.begin..<h.end].map(\.text)
807 default:
808 break
809 }
810 output += g.log.dropFirst(before).map(\.text)
811 game = g
812 return output
813 }
814}
