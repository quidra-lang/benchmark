159 deal(): void {
160 this.number++;
161 this.passCode = this.rules.PASS[(this.number - 1) % this.rules.PASS.length];
162 this.hands = [0, 1, 2, 3].map((s) => (this.stacked.get(s) ?? []).slice());
163 const used = new Set(this.hands.flat());
164 const deck = Array.from({ length: 52 }, (_, c) => c).filter((c) => !used.has(c));
165 this.generator.shuffle(deck);
166 const free = [0, 1, 2, 3].filter((s) => !this.stacked.has(s));
167 deck.forEach((c, i) => this.hands[free[i % free.length]].push(c));
168 this.stacked.clear();
169 this.passes.clear();
170 this.taken = rows();
171 this.tricks = [];
172 this.counts = four();
173 this.current = [];
174 this.broken = false;
175 this.undo = [];
176 this.record = [];
177 this.write(1, `hand ${this.number} pass ${directions[this.passCode]}`);
178 for (let s = 0; s < 4; s++) this.write(2, `${seats[s]}: ${display(this.hands[s])}`);
179 if (this.passCode === "H") this.beginPlay();

187 passCards(s: number, cs: number[]): void {
188 this.hands[s] = this.hands[s].filter((c) => !cs.includes(c));
189 this.passes.set(s, cs.slice());

191 if (this.passes.size === 4) {
192 const offset = this.passCode === "L" ? 1 : this.passCode === "R" ? 3 : 2;
193 for (const [source, cards] of this.passes)
194 this.hands[(source + offset) % 4] = this.hands[(source + offset) % 4].concat(cards);
195 this.beginPlay();
196 }

198 actor(): number {
199 return (this.leader + this.current.length) % 4;
200 }
201 legal(): number[] {
202 const hand = this.hands[this.actor()];
203 if (!this.current.length) {
204 if (!this.tricks.length) return [0];
205 const safe = hand.filter((c) => suit(c) !== 3);
206 return !this.broken && safe.length ? safe : hand.slice();
207 }
208 const led = suit(this.current[0][1]);
209 const follow = hand.filter((c) => suit(c) === led);
210 if (follow.length) return follow;
211 if (!this.tricks.length && this.rules.BLOOD === "OFF") {
212 const safe = hand.filter((c) => !penalty(c));
213 if (safe.length) return safe;
214 }
215 return hand.slice();
216 }
217 choosePass(s: number): number[] {
218 const hand = this.hands[s],
219 strategy = this.bots[s];
220 const ascending = hand.slice().sort(lowOrder),
221 descending = hand.slice().sort(highOrder);
222 const hearts = descending.filter((c) => suit(c) === 3);
223 const special = [QS, KS, AS].filter((c) => hand.includes(c));
224 let order: number[];
225 if (strategy === "LOW") order = descending;
226 else if (strategy === "HIGH") order = ascending;
227 else if (strategy === "MOON")
228 order = ascending
229 .filter((c) => suit(c) !== 3)
230 .concat(ascending.filter((c) => suit(c) === 3));
231 else if (strategy === "SMART") {
232 const protectedQueen =
233 hand.includes(QS) && hand.filter((c) => suit(c) === 2).length >= 5;
234 order = protectedQueen ? [] : special;
235 const choices = [0, 1].filter((su) => hand.some((c) => suit(c) === su));
236 choices.sort(
237 (a, b) =>
238 hand.filter((c) => suit(c) === a).length -
239 hand.filter((c) => suit(c) === b).length || a - b,
240 );
241 if (choices.length)
242 order = order.concat(descending.filter((c) => suit(c) === choices[0]));
243 order = order.concat(hearts, descending);
244 } else order = special.concat(hearts, descending);
245 return Array.from(new Set(order)).slice(0, 3);
246 }
247 queenPlayed(): boolean {
248 return this.taken.some((cs) => cs.includes(QS)) || this.current.some((p) => p[1] === QS);
249 }
250 choosePlay(): number {
251 const seat = this.actor(),
252 hand = this.hands[seat],
253 legal = this.legal();
254 let strategy = this.bots[seat];
255 if (strategy === "LOW") return low(legal);
256 if (strategy === "HIGH") return high(legal);
257 if (strategy === "MOON" && this.taken.some((cs, s) => s !== seat && cs.some(penalty)))
258 strategy = "BASIC";
259 if (!this.current.length) {
260 if (strategy === "SMART" && !this.queenPlayed() && !hand.includes(QS)) {
261 const spades = legal.filter((c) => suit(c) === 2 && c % 13 < 10);
262 if (spades.length) return low(spades);
263 }
264 const choices = [0, 1, 2, 3].filter((s) => legal.some((c) => suit(c) === s));
265 choices.sort(
266 (a, b) =>
267 (strategy === "MOON" ? -1 : 1) *
268 (legal.filter((c) => suit(c) === a).length -
269 legal.filter((c) => suit(c) === b).length) || a - b,
270 );
271 const cs = legal.filter((c) => suit(c) === choices[0]);
272 return strategy === "MOON" ? high(cs) : low(cs);
273 }
274 const led = suit(this.current[0][1]);
275 if (hand.some((c) => suit(c) === led)) {
276 const winning =
277 high(this.current.filter((p) => suit(p[1]) === led).map((p) => p[1])) % 13;
278 const lower = legal.filter((c) => c % 13 < winning);
279 if (strategy === "MOON") return high(legal) % 13 > winning ? high(legal) : low(legal);
280 if (lower.length) return high(lower);
281 if (strategy === "SMART" && this.current.length === 3) {
282 const safe = legal.filter((c) => c !== QS);
283 return safe.length ? high(safe) : QS;
284 }
285 return low(legal);
286 }
287 if (strategy === "MOON") {
288 const safe = legal.filter((c) => !penalty(c));
289 return low(safe.length ? safe : legal);
290 }
291 if (legal.includes(QS)) return QS;
292 if (strategy === "SMART" && !this.queenPlayed()) {
293 if (legal.includes(AS)) return AS;
294 if (legal.includes(KS)) return KS;
295 }
296 const hearts = legal.filter((c) => suit(c) === 3);
297 return high(hearts.length ? hearts : legal);
298 }
299 winner(plays: Play[]): number {
300 const led = suit(plays[0][1]);
301 return plays
302 .filter((p) => suit(p[1]) === led)
303 .sort((a, b) => (b[1] % 13) - (a[1] % 13))[0][0];
304 }
305 trickLine(plays: Play[], n: number): string {
306 return `trick ${n}: ${plays.map(([s, c]) => `${seats[s]} ${cardText(c)}`).join(" ")} -> ${seats[this.winner(plays)]} ${sum(plays.map((p) => this.point(p[1])))}`;
307 }
308 play(c: number): void {
309 const seat = this.actor();
310 this.undo.push({ position: this.snapshot(), logLength: this.log.length, seat, card: c });
311 this.hands[seat] = this.hands[seat].filter((v) => v !== c);
312 this.current.push([seat, c]);
313 if (suit(c) === 3 || (c === QS && this.rules.QSBREAK === "ON")) this.broken = true;
314 if (this.current.length === 4) {
315 const winner = this.winner(this.current);
316 this.taken[winner] = this.taken[winner].concat(this.current.map((p) => p[1]));
317 this.counts[winner]++;
318 this.tricks.push(this.current);
319 this.write(2, this.trickLine(this.current, this.tricks.length));
320 this.current = [];
321 this.leader = winner;
322 if (this.tricks.length === 13) this.finish();
323 }
324 }
325 undoCard(): Play {
326 const u = this.undo.pop()!;
327 this.restore(u.position);
328 this.log.length = u.logLength;
329 return [u.seat, u.card];
330 }
331 claim(s: number): void {
332 const remaining = this.hands.flat(),
333 n = 13 - this.tricks.length;
334 this.taken[s] = this.taken[s].concat(remaining);
335 this.counts[s] += n;
336 this.hands = rows();
337 this.write(
338 2,
339 `claim ${seats[s]} tricks ${n} points ${sum(remaining.map((c) => this.point(c)))}`,
340 );
341 this.finish();
342 }
343 finish(): void {
344 let points = this.taken.map((cs) => sum(cs.map((c) => this.point(c))));
345 const shooter = this.taken.findIndex((cs) => cs.filter(penalty).length === 14);
346 if (shooter >= 0) {
347 points[shooter] -= 26;
348 let method = this.rules.MOON;
349 if (method === "AUTO") {
350 const add = points.map((p, s) => this.scores[s] + p + (s === shooter ? 0 : 26));
351 const ended =
352 this.number >= this.rules.LIMIT || add.some((p) => p >= this.rules.TARGET);
353 const winner = [0, 1, 2, 3].sort((a, b) => add[a] - add[b] || a - b)[0];
354 method = ended && winner !== shooter ? "SUB" : "ADD";
355 }
356 if (method === "SUB") points[shooter] -= 26;
357 else points = points.map((p, s) => p + (s === shooter ? 0 : 26));
358 this.write(1, `moon ${seats[shooter]} ${method}`);
359 }
360 for (let s = 0; s < 4; s++) {
361 this.scores[s] += points[s];
362 this.stats[s][0] += this.counts[s];
363 this.stats[s][1] += this.taken[s].filter((c) => suit(c) === 3).length;
364 this.stats[s][2] += Number(this.taken[s].includes(QS));
365 this.stats[s][3] += Number(s === shooter);
366 }
367 this.write(1, pairs("points", points));
368 this.write(1, pairs("scores", this.scores));
369 this.history.push({
370 number: this.number,
371 direction: directions[this.passCode],
372 points,
373 shooter,
374 record: this.record.slice(),
375 });
376 const ended =
377 this.number >= this.rules.LIMIT || this.scores.some((p) => p >= this.rules.TARGET);
378 this.state = ended ? "none" : "between";
379 if (ended) {
380 this.write(0, `game over hands ${this.number}`);
381 this.ranking().forEach((s, i) =>
382 this.write(0, `rank ${i + 1} ${seats[s]} ${this.scores[s]}`),
383 );
384 for (const line of this.statsLines()) this.write(0, line);
385 }
386 }
387 ranking(): number[] {
388 return [0, 1, 2, 3].sort((a, b) => this.scores[a] - this.scores[b] || a - b);
389 }
390 statsLines(): string[] {
391 return this.stats.map(
392 (s, i) =>
393 `stats ${seats[i]} tricks ${s[0]} hearts ${s[1]} queens ${s[2]} moons ${s[3]}`,
394 );
395 }
396 autoPass(): void {
397 for (let s = 0; s < 4; s++) if (!this.passes.has(s)) this.passCards(s, this.choosePass(s));
398 }
399 autoHand(): void {
400 if (this.state === "passing") this.autoPass();
401 while (this.state === "playing") this.play(this.choosePlay());
402 }
403 autoGame(): void {
404 if (this.state === "passing" || this.state === "playing") this.autoHand();
405 while (this.state !== "none") {
406 this.deal();
407 this.autoHand();
408 }
409 }
410 preview(c: number): Play {
411 const copy = new Game(this.rules, this.bots, new Generator(this.generator.state));
412 copy.restore(this.snapshot());
413 copy.state = "playing";
414 copy.number = this.number;
415 copy.passCode = this.passCode;
416 copy.scores = this.scores.slice();
417 copy.stats = this.stats.map((s) => s.slice());
418 const count = copy.tricks.length;
419 copy.play(c);
420 while (copy.state === "playing" && copy.tricks.length === count)
421 copy.play(copy.choosePlay());
422 const trick = copy.tricks[copy.tricks.length - 1];
423 return [copy.winner(trick), sum(trick.map((p) => copy.point(p[1])))];
424 }

426class Table {
427 rules: Rules = { ...defaults };
428 bots: Strategy[] = ["BASIC", "BASIC", "BASIC", "BASIC"];
429 game: Game | null = null;
430 active(): boolean {
431 return this.game !== null && this.game.state !== "none";
432 }
433 require(kind: string): void {
434 const state = this.active() ? this.game!.state : "none";
435 if (kind === "game" && state === "none") fail("no game");
436 if (kind === "hand" && !["passing", "playing"].includes(state)) fail("no hand");
437 if (kind === "playing" && state !== "playing") fail("not playing");
438 if (kind === "passing" && state !== "passing") fail("not passing");
439 if (kind === "between") {
440 if (state === "none") fail("no game");
441 if (state !== "between") fail("hand in progress");
442 }
443 }
444 simulate(command: string, seed: number, count: number): string[] {
445 const generator = new Generator(seed);
446 if (command === "GAME") {
447 const game = new Game(this.rules, this.bots, generator);
448 game.autoGame();
449 return [`game seed ${seed}`].concat(
450 game.log.filter((l) => l.level <= count).map((l) => l.text),
451 );
452 }
453 const rounds = command === "ROTATE" ? 4 : 1,
454 games = rounds * count;
455 const places = [four(), four(), four(), four()],
456 totals = four(),
457 stats = [four(), four(), four(), four()];
458 let hands = 0;
459 for (let r = 0; r < rounds; r++) {
460 const bots = this.bots.slice();
461 for (let i = 0; i < 4; i++) bots[(i + r) % 4] = this.bots[i];
462 for (let g = 0; g < count; g++) {
463 const game = new Game(this.rules, bots, generator);
464 game.autoGame();
465 hands += game.number;
466 const ranking = game.ranking();
467 for (let i = 0; i < 4; i++) {
468 const actual = (i + r) % 4;
469 places[i][ranking.indexOf(actual)]++;
470 totals[i] += game.scores[actual];
471 for (let k = 0; k < 4; k++) stats[i][k] += game.stats[actual][k];
472 }
473 }
474 }
475 const out = [`${command.toLowerCase()} seed ${seed} games ${games} hands ${hands}`];
476 for (let i = 0; i < 4; i++) {
477 const cents = Math.floor((100 * totals[i]) / games),
478 abs = Math.abs(cents);
479 const average = `${cents < 0 ? "-" : ""}${Math.floor(abs / 100)}.${String(abs % 100).padStart(2, "0")}`;
480 const name = seats[i] + (command === "ROTATE" ? ` ${this.bots[i]}` : "");
481 const s = stats[i];
482 out.push(
483 `${name} places ${places[i].join(" ")} total ${totals[i]} average ${average} tricks ${s[0]} hearts ${s[1]} queens ${s[2]} moons ${s[3]}`,
484 );
485 }
486 return out;
487 }
488 command(words: string[]): string[] {
489 const [cmd, ...args] = words;
490 const known =
491 "RULE BOT SETTINGS GAME MATCH ROTATE NEW ABANDON STACK DEAL SHOW PASS PLAY AUTO UNDO CLAIM LEGAL PREVIEW SCORE TAKEN TRICKS VOIDS UNSEEN STATS HISTORY REPLAY".split(
492 " ",
493 );
494 if (!known.includes(cmd)) fail(`unknown command ${cmd}`);
495 if (cmd === "RULE") {
496 const [name, word] = args;
497 if (!Object.hasOwn(defaults, name)) fail(`unknown rule ${name}`);
498 let value: string | number = word;
499 if (name === "TARGET" || name === "LIMIT")
500 value = bounded(word, 1, name === "TARGET" ? 1000 : 100);
501 else {
502 const valid =
503 name === "PASS"
504 ? /^[LRAH]{1,8}$/.test(word)
505 : (name === "MOON" ? ["ADD", "SUB", "AUTO"] : ["ON", "OFF"]).includes(word);
506 if (!valid) fail(`bad value ${word}`);
507 }
508 if (this.active()) fail("game in progress");
509 if (name === "TARGET") this.rules.TARGET = value as number;
510 else if (name === "LIMIT") this.rules.LIMIT = value as number;
511 else
512 this.rules[name as "MOON" | "JACK" | "QSBREAK" | "BLOOD" | "PASS"] =
513 value as string;
514 return [`rule ${name} ${word}`];
515 }
516 if (cmd === "BOT") {
517 const s = seatValue(args[0]);
518 if (!strategies.includes(args[1] as Strategy)) fail(`bad value ${args[1]}`);
519 this.bots[s] = args[1] as Strategy;
520 return [`bot ${seats[s]} ${args[1]}`];
521 }
522 if (cmd === "SETTINGS")
523 return [
524 "rules " +
525 Object.entries(this.rules)
526 .map(([k, v]) => `${k} ${v}`)
527 .join(" "),
528 pairs("bots", this.bots),
529 ];
530 if (["GAME", "MATCH", "ROTATE"].includes(cmd)) {
531 const seed = bounded(args[0], 1, 2147483646),
532 count = bounded(
533 args[1],
534 cmd === "GAME" ? 0 : 1,
535 cmd === "GAME" ? 2 : cmd === "MATCH" ? 50 : 10,
536 );
537 return this.simulate(cmd, seed, count);
538 }
539 if (cmd === "NEW") {
540 const seed = bounded(args[0], 1, 2147483646);
541 if (this.active()) fail("game in progress");
542 this.game = new Game(this.rules, this.bots, new Generator(seed));
543 return [`new game seed ${seed}`];
544 }
545 if (cmd === "ABANDON") {
546 this.require("game");
547 this.game = null;
548 return ["game abandoned"];
549 }
550 let s = -1,
551 cards: number[] = [];
552 if (["STACK", "PASS", "PLAY", "CLAIM"].includes(cmd)) s = seatValue(args[0]);
553 if (["STACK", "PASS", "PLAY"].includes(cmd)) cards = args.slice(1).map(cardValue);
554 if (cmd === "STACK" && cards.length !== 13) fail("need 13 cards");
555 if (cmd === "AUTO" && !["PASS", "CARD", "TRICK", "HAND", "GAME"].includes(args[0]))
556 fail(`bad value ${args[0]}`);
557 if (["STACK", "DEAL"].includes(cmd)) this.require("between");
558 else if (cmd === "SHOW") this.require("hand");
559 else if (cmd === "PASS") this.require("passing");
560 else if (
561 [
562 "PLAY",
563 "UNDO",
564 "CLAIM",
565 "LEGAL",
566 "PREVIEW",
567 "TAKEN",
568 "TRICKS",
569 "VOIDS",
570 "UNSEEN",
571 ].includes(cmd)
572 )
573 this.require("playing");
574 else if (["SCORE", "STATS", "HISTORY", "REPLAY"].includes(cmd)) this.require("game");
575 else if (cmd === "AUTO")
576 this.require(
577 args[0] === "PASS"
578 ? "passing"
579 : ["CARD", "TRICK"].includes(args[0])
580 ? "playing"
581 : args[0] === "HAND"
582 ? "hand"
583 : "game",
584 );
585 const game = this.game!,
586 before = game.log.length,
587 prefix: string[] = [];
588 switch (cmd) {
589 case "STACK":
590 duplicate(cards);
591 for (const c of cards) {
592 for (let o = 0; o < 4; o++) {
593 if (o !== s && (game.stacked.get(o) ?? []).includes(c)) {
594 fail(`${cardText(c)} stacked for ${seats[o]}`);
595 }
596 }
597 }
598 game.stacked.set(s, cards.slice());
599 return [`stacked ${seats[s]}`];
600 case "DEAL":
601 game.deal();
602 break;
603 case "SHOW":
604 return game.hands.map((cs, i) => `${seats[i]}: ${display(cs)}`);
605 case "PASS":
606 if (game.passes.has(s)) fail(`${seats[s]} already passed`);
607 duplicate(cards);
608 for (const c of cards)
609 if (!game.hands[s].includes(c))
610 fail(`${seats[s]} does not hold ${cardText(c)}`);
611 game.passCards(s, cards);
612 break;
613 case "PLAY": {
614 const c = cards[0];
615 if (s !== game.actor()) fail(`waiting for ${seats[game.actor()]}`);
616 if (!game.hands[s].includes(c)) fail(`${seats[s]} does not hold ${cardText(c)}`);
617 if (!game.legal().includes(c)) {
618 if (!game.current.length)
619 fail(!game.tricks.length ? "must lead 2C" : "hearts not broken");
620 const led = suit(game.current[0][1]);
621 if (game.hands[s].some((c) => suit(c) === led))
622 fail(`must follow ${suits[led]}`);
623 fail("no points on first trick");
624 }
625 prefix.push(`${seats[s]} plays ${cardText(c)}`);
626 game.play(c);
627 break;
628 }
629 case "AUTO":
630 if (args[0] === "PASS") game.autoPass();
631 else if (args[0] === "CARD") {
632 const seat = game.actor(),
633 c = game.choosePlay();
634 prefix.push(`${seats[seat]} plays ${cardText(c)}`);
635 game.play(c);
636 } else if (args[0] === "TRICK") {
637 const count = game.tricks.length;
638 while (game.state === "playing" && game.tricks.length === count)
639 game.play(game.choosePlay());
640 } else if (args[0] === "HAND") game.autoHand();
641 else game.autoGame();
642 break;
643 case "UNDO": {
644 if (!game.undo.length) fail("nothing to undo");
645 const [seat, c] = game.undoCard();
646 return [`undo ${seats[seat]} ${cardText(c)}`];
647 }
648 case "CLAIM":
649 if (s !== game.actor()) fail(`waiting for ${seats[game.actor()]}`);
650 if (game.current.length) fail("trick in progress");
651 for (const c of game.hands[s]) {
652 if (
653 game.hands.some(
654 (cs, o) =>
655 o !== s && cs.some((v) => suit(v) === suit(c) && v % 13 > c % 13),
656 )
657 ) {
658 fail("claim rejected");
659 }
660 }
661 game.claim(s);
662 break;
663 case "LEGAL":
664 return [
665 `legal ${seats[game.actor()]}: ${display(game.legal())} bot ${cardText(game.choosePlay())}`,
666 ];
667 case "PREVIEW":
668 return game
669 .legal()
670 .sort((a, b) => a - b)
671 .map((c) => {
672 const [w, p] = game.preview(c);
673 return `preview ${cardText(c)} -> ${seats[w]} ${p}`;
674 });
675 case "SCORE":
676 return [pairs("scores", game.scores)].concat(
677 game.state === "playing"
678 ? [
679 pairs(
680 "taken",
681 game.taken.map((cs) => sum(cs.map((c) => game.point(c)))),
682 ),
683 ]
684 : [],
685 );
686 case "TAKEN":
687 return game.taken.map(
688 (cs, i) => `${seats[i]}: ${display(cs.filter((c) => game.point(c) !== 0))}`,
689 );
690 case "TRICKS": {
691 const out = game.tricks.map((t, i) => game.trickLine(t, i + 1));
692 if (game.current.length)
693 out.push(
694 "current: " +
695 game.current.map(([s, c]) => `${seats[s]} ${cardText(c)}`).join(" "),
696 );
697 return out.length ? out : ["no tricks"];
698 }
699 case "VOIDS": {
700 const missing = [
701 new Set<number>(),
702 new Set<number>(),
703 new Set<number>(),
704 new Set<number>(),
705 ];
706 for (const trick of game.tricks.concat(game.current.length ? [game.current] : [])) {
707 const led = suit(trick[0][1]);
708 for (const [s, c] of trick) if (suit(c) !== led) missing[s].add(led);
709 }
710 return [
711 pairs(
712 "voids",
713 missing.map(
714 (ms) =>
715 Array.from(ms)
716 .sort()
717 .map((s) => suits[s])
718 .join("") || "-",
719 ),
720 ),
721 ];
722 }
723 case "UNSEEN": {
724 const actor = game.actor(),
725 seen = new Set(
726 game.taken.flat().concat(
727 game.current.map((p) => p[1]),
728 game.hands[actor],
729 ),
730 );
731 return [
732 `unseen ${seats[actor]}: ${display(Array.from({ length: 52 }, (_, c) => c).filter((c) => !seen.has(c)))}`,
733 ];
734 }
735 case "STATS":
736 return game.statsLines();
737 case "HISTORY":
738 return game.history.length
739 ? game.history.map(
740 (h) =>
741 `hand ${h.number} pass ${h.direction} ${pairs("points", h.points)}${h.shooter >= 0 ? ` moon ${seats[h.shooter]}` : ""}`,
742 )
743 : ["no hands"];
744 case "REPLAY": {
745 const n = Number(args[0]);
746 if (n < 1 || n > game.history.length) fail(`bad value ${args[0]}`);
747 return game.history[n - 1].record.slice();
748 }
749 }
750 return prefix.concat(game.log.slice(before).map((l) => l.text));
751 }
752}
753const lines = readFileSync(0, "utf8").split(/\r?\n/),
754 table = new Table();
755for (const line of lines.slice(1, Number(lines[0]) + 1)) {
756 try {
757 for (const result of table.command(line.split(" "))) writeFileSync(1, result + "\n");
758 } catch (e) {
759 if (e instanceof CommandError) writeFileSync(1, `error: ${e.message}\n`);
760 else throw e;
761 }
