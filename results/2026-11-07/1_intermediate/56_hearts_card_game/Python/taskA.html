82 def shuffle(self, deck):
83 for i in range(len(deck) - 1, 0, -1):
84 j = self.draw() % (i + 1)
85 deck[i], deck[j] = deck[j], deck[i]

88class Game:
89 def __init__(self, rules, bots, generator):
90 self.rules = rules
91 self.bots = bots
92 self.generator = generator
93 self.state = "between"
94 self.number = 0
95 self.scores = [0] * 4
96 self.stats = [[0] * 4 for _ in SEATS]
97 self.stacked = {}
98 self.history = []
99 self.log = []

102 def point(self, card):
103 if card == QS:
104 return 13
105 if card // 13 == 3:
106 return 1
107 return -10 if card == JD and self.rules["JACK"] == "ON" else 0

114 def deal(self):
115 self.number += 1
116 cycle = self.rules["PASS"]
117 self.pass_code = cycle[(self.number - 1) % len(cycle)]
118 self.hands = [set(self.stacked.get(seat, ())) for seat in range(4)]
119 used = set().union(*self.hands)
120 deck = [card for card in range(52) if card not in used]
121 self.generator.shuffle(deck)
122 free = [seat for seat in range(4) if seat not in self.stacked]
123 for i, card in enumerate(deck):
124 self.hands[free[i % len(free)]].add(card)
125 self.stacked.clear()
126 self.passes = {}
127 self.taken = [set() for _ in SEATS]
128 self.tricks = []
129 self.trick_counts = [0] * 4
130 self.current = []
131 self.broken = False
132 self.undo = []
133 self.record = []
134 self.write(1, f"hand {self.number} pass {DIRECTIONS[self.pass_code]}")
135 for seat in range(4):
136 self.write(2, f"{SEATS[seat]}: {cards_text(self.hands[seat])}")
137 if self.pass_code == "H":
138 self.begin_play()
139 else:
140 self.state = "passing"

142 def begin_play(self):
143 self.state = "playing"
144 self.leader = next(seat for seat in range(4) if 0 in self.hands[seat])
145 self.write(2, f"lead {SEATS[self.leader]}")

147 def pass_cards(self, seat, cards):
148 self.hands[seat].difference_update(cards)
149 self.passes[seat] = list(cards)
150 self.write(2, f"{SEATS[seat]} passes {cards_text(cards)}")
151 if len(self.passes) == 4:
152 offset = {"L": 1, "R": 3, "A": 2}[self.pass_code]
153 for source, passed in self.passes.items():
154 self.hands[(source + offset) % 4].update(passed)
155 self.begin_play()

157 def actor(self):
158 return (self.leader + len(self.current)) % 4

160 def legal(self):
161 hand = self.hands[self.actor()]
162 if not self.current:
163 if not self.tricks:
164 return {0}
165 non_hearts = {card for card in hand if card // 13 != 3}
166 return non_hearts if not self.broken and non_hearts else set(hand)
167 suit = self.current[0][1] // 13
168 followers = {card for card in hand if card // 13 == suit}
169 if followers:
170 return followers
171 if not self.tricks and self.rules["BLOOD"] == "OFF":
172 safe = {card for card in hand if not penalty(card)}
173 if safe:
174 return safe
175 return set(hand)

177 def choose_pass(self, seat):
178 hand = self.hands[seat]
179 strategy = self.bots[seat]
180 low = sorted(hand, key=rank_key)
181 high = sorted(hand, key=high_key)
182 hearts = sorted((card for card in hand if card // 13 == 3), key=high_key)
183 special = [card for card in (QS, KS, AS) if card in hand]
184 if strategy == "LOW":
185 order = high
186 elif strategy == "HIGH":
187 order = low
188 elif strategy == "MOON":
189 order = [card for card in low if card // 13 != 3] + [card for card in low if card // 13 == 3]
190 elif strategy == "SMART":
191 protected = QS in hand and sum(card // 13 == 2 for card in hand) >= 5
192 order = [] if protected else special
193 suits = [suit for suit in (0, 1) if any(card // 13 == suit for card in hand)]
194 if suits:
195 suit = min(suits, key=lambda s: (sum(card // 13 == s for card in hand), s))
196 order += [card for card in high if card // 13 == suit]
197 order += hearts + high
198 else:
199 order = special + hearts + high
200 return list(dict.fromkeys(order))[:3]

202 def choose_play(self):
203 seat = self.actor()
204 hand = self.hands[seat]
205 legal = self.legal()
206 strategy = self.bots[seat]
207 if strategy == "LOW":
208 return lowest(legal)
209 if strategy == "HIGH":
210 return highest(legal)
211 if strategy == "MOON" and any(any(penalty(card) for card in self.taken[other])
212 for other in range(4) if other != seat):
213 strategy = "BASIC"
214 if not self.current:
215 if strategy == "SMART" and not self.queen_played() and QS not in hand:
216 small_spades = {card for card in legal if card // 13 == 2 and card % 13 < 10}
217 if small_spades:
218 return lowest(small_spades)
219 suits = {card // 13 for card in legal}
220 if strategy == "MOON":
221 suit = min(suits, key=lambda s: (-sum(card // 13 == s for card in legal), s))
222 return highest(card for card in legal if card // 13 == suit)
223 suit = min(suits, key=lambda s: (sum(card // 13 == s for card in legal), s))
224 return lowest(card for card in legal if card // 13 == suit)
225 led = self.current[0][1] // 13
226 if any(card // 13 == led for card in hand):
227 winning = max(card % 13 for _, card in self.current if card // 13 == led)
228 lower = {card for card in legal if card % 13 < winning}
229 if strategy == "MOON":
230 high = highest(legal)
231 return high if high % 13 > winning else lowest(legal)
232 if lower:
233 return highest(lower)
234 if strategy == "SMART" and len(self.current) == 3:
235 return highest(legal - {QS}) if legal - {QS} else QS
236 return lowest(legal)
237 if strategy == "MOON":
238 safe = {card for card in legal if not penalty(card)}
239 return lowest(safe or legal)
240 if QS in legal:
241 return QS
242 if strategy == "SMART" and not self.queen_played():
243 if AS in legal:
244 return AS
245 if KS in legal:
246 return KS
247 hearts = {card for card in legal if card // 13 == 3}
248 return highest(hearts or legal)

250 def queen_played(self):
251 return any(QS in cards for cards in self.taken) or any(card == QS for _, card in self.current)

253 def winner(self, plays):
254 led = plays[0][1] // 13
255 return max((play for play in plays if play[1] // 13 == led), key=lambda play: play[1] % 13)[0]

257 def trick_line(self, plays, number):
258 sequence = " ".join(f"{SEATS[seat]} {card_text(card)}" for seat, card in plays)
259 points = sum(self.point(card) for _, card in plays)
260 return f"trick {number}: {sequence} -> {SEATS[self.winner(plays)]} {points}"

262 def play(self, card):
263 seat = self.actor()
264 fields = (self.hands, self.taken, self.tricks, self.trick_counts, self.current,
265 self.leader, self.broken, self.record)
266 self.undo.append((deepcopy(fields), len(self.log), seat, card))
267 self.hands[seat].remove(card)
268 self.current.append((seat, card))
269 if card // 13 == 3 or card == QS and self.rules["QSBREAK"] == "ON":
270 self.broken = True
271 if len(self.current) == 4:
272 winner = self.winner(self.current)
273 self.taken[winner].update(card for _, card in self.current)
274 self.trick_counts[winner] += 1
275 self.tricks.append(self.current)
276 self.write(2, self.trick_line(self.current, len(self.tricks)))
277 self.current = []
278 self.leader = winner
279 if len(self.tricks) == 13:
280 self.finish_hand()

282 def undo_card(self):
283 fields, length, seat, card = self.undo.pop()
284 (self.hands, self.taken, self.tricks, self.trick_counts, self.current,
285 self.leader, self.broken, self.record) = fields
286 del self.log[length:]
287 return seat, card

289 def claim(self, seat):
290 remaining = set().union(*self.hands)
291 count = 13 - len(self.tricks)
292 self.taken[seat].update(remaining)
293 self.trick_counts[seat] += count
294 self.hands = [set() for _ in SEATS]
295 self.write(2, f"claim {SEATS[seat]} tricks {count} points {sum(self.point(card) for card in remaining)}")
296 self.finish_hand()

298 def finish_hand(self):
299 points = [sum(self.point(card) for card in cards) for cards in self.taken]
300 shooter = next((seat for seat in range(4) if sum(penalty(card) for card in self.taken[seat]) == 14), None)
301 if shooter is not None:
302 points[shooter] -= 26
303 method = self.rules["MOON"]
304 if method == "AUTO":
305 add = [self.scores[seat] + points[seat] + (26 if seat != shooter else 0) for seat in range(4)]
306 ended = self.number >= self.rules["LIMIT"] or max(add) >= self.rules["TARGET"]
307 winner = min(range(4), key=lambda seat: (add[seat], seat))
308 method = "SUB" if ended and winner != shooter else "ADD"
309 if method == "SUB":
310 points[shooter] -= 26
311 else:
312 points = [value + (26 if seat != shooter else 0) for seat, value in enumerate(points)]
313 self.write(1, f"moon {SEATS[shooter]} {method}")
314 for seat in range(4):
315 self.scores[seat] += points[seat]
316 stats = self.stats[seat]
317 stats[0] += self.trick_counts[seat]
318 stats[1] += sum(card // 13 == 3 for card in self.taken[seat])
319 stats[2] += QS in self.taken[seat]
320 stats[3] += shooter == seat
321 self.write(1, pairs("points", points))
322 self.write(1, pairs("scores", self.scores))
323 self.history.append(dict(number=self.number, direction=DIRECTIONS[self.pass_code],
324 points=points, shooter=shooter, record=self.record.copy()))
325 ended = self.number >= self.rules["LIMIT"] or max(self.scores) >= self.rules["TARGET"]
326 self.state = "none" if ended else "between"
327 if ended:
328 self.write(0, f"game over hands {self.number}")
329 for rank, seat in enumerate(self.ranking(), 1):
330 self.write(0, f"rank {rank} {SEATS[seat]} {self.scores[seat]}")
331 for line in self.stats_lines():
332 self.write(0, line)

338 return [f"stats {SEATS[seat]} tricks {s[0]} hearts {s[1]} queens {s[2]} moons {s[3]}"

342 for seat in range(4):
343 if seat not in self.passes:

346 def auto_hand(self):
347 if self.state == "passing":
348 self.auto_pass()

368def validate_rule(name, word):
369 if name not in DEFAULT_RULES:
370 fail(f"unknown rule {name}")
371 if name in ("TARGET", "LIMIT"):
372 try:
373 value = int(word)
374 except ValueError:
375 fail(f"bad value {word}")
376 if not 1 <= value <= (1000 if name == "TARGET" else 100):
377 fail(f"bad value {word}")
378 return value
379 allowed = ("ADD", "SUB", "AUTO") if name == "MOON" else ("ON", "OFF")
380 if name == "PASS":
381 good = 1 <= len(word) <= 8 and all(char in "LRAH" for char in word)
382 else:
383 good = word in allowed
384 if not good:
385 fail(f"bad value {word}")
386 return word

419 def simulate(self, command, seed, count):
420 generator = Generator(seed)
421 if command == "GAME":
422 game = Game(self.rules, self.bots, generator)
423 game.auto_game()
424 return [f"game seed {seed}"] + [line for level, line in game.log if level <= count]
425 rounds = 4 if command == "ROTATE" else 1
426 places = [[0] * 4 for _ in SEATS]
427 totals = [0] * 4
428 stats = [[0] * 4 for _ in SEATS]
429 hands = 0
430 for rotation in range(rounds):
431 strategies = [None] * 4
432 for seat in range(4):
433 strategies[(seat + rotation) % 4] = self.bots[seat]
434 for _ in range(count):
435 game = Game(self.rules, strategies, generator)
436 game.auto_game()
437 hands += game.number
438 ranking = game.ranking()
439 for original in range(4):
440 actual = (original + rotation) % 4
441 places[original][ranking.index(actual)] += 1
442 totals[original] += game.scores[actual]
443 stats[original] = [a + b for a, b in zip(stats[original], game.stats[actual])]
444 games = rounds * count
445 output = [f"{command.lower()} seed {seed} games {games} hands {hands}"]
446 for seat in range(4):
447 cents = 100 * totals[seat] // games
448 average = f"{'-' if cents < 0 else ''}{abs(cents) // 100}.{abs(cents) % 100:02d}"
449 name = SEATS[seat] + (f" {self.bots[seat]}" if command == "ROTATE" else "")
450 positions = " ".join(map(str, places[seat]))
451 tricks, hearts, queens, moons = stats[seat]
452 output.append(f"{name} places {positions} total {totals[seat]} average {average} "
453 f"tricks {tricks} hearts {hearts} queens {queens} moons {moons}")
454 return output

456 def command(self, words):
457 command, args = words[0], words[1:]
458 known = {"RULE", "BOT", "SETTINGS", "GAME", "MATCH", "ROTATE", "NEW", "ABANDON", "STACK",
459 "DEAL", "SHOW", "PASS", "PLAY", "AUTO", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "SCORE",
460 "TAKEN", "TRICKS", "VOIDS", "UNSEEN", "STATS", "HISTORY", "REPLAY"}
461 if command not in known:
462 fail(f"unknown command {command}")
463 if command == "RULE":
464 name, word = args
465 value = validate_rule(name, word)
466 if self.active():
467 fail("game in progress")
468 self.rules[name] = value
469 return [f"rule {name} {word}"]
470 if command == "BOT":
471 seat = seat_value(args[0])
472 if args[1] not in STRATEGIES:
473 fail(f"bad value {args[1]}")
474 self.bots[seat] = args[1]
475 return [f"bot {SEATS[seat]} {args[1]}"]
476 if command == "SETTINGS":
477 rule_line = "rules " + " ".join(f"{name} {value}" for name, value in self.rules.items())
478 return [rule_line, pairs("bots", self.bots)]
479 if command in ("GAME", "MATCH", "ROTATE"):
480 seed = bounded(args[0], 1, 2147483646)
481 limits = {"GAME": (0, 2), "MATCH": (1, 50), "ROTATE": (1, 10)}
482 low, high = limits[command]
483 count = bounded(args[1], low, high)
484 return self.simulate(command, seed, count)
485 if command == "NEW":
486 seed = bounded(args[0], 1, 2147483646)
487 if self.active():
488 fail("game in progress")
489 self.game = Game(self.rules, self.bots, Generator(seed))
490 return [f"new game seed {seed}"]
491 if command == "ABANDON":
492 self.require_game()
493 self.game = None
494 return ["game abandoned"]
495
496 seat = None
497 cards = []
498 if command in ("STACK", "PASS", "PLAY", "CLAIM"):
499 seat = seat_value(args[0])
500 if command in ("STACK", "PASS", "PLAY"):
501 cards = [card_value(word) for word in args[1:]]
502 if command == "STACK" and len(cards) != 13:
503 fail("need 13 cards")
504 if command == "AUTO" and args[0] not in ("PASS", "CARD", "TRICK", "HAND", "GAME"):
505 fail(f"bad value {args[0]}")
506
507 if command in ("STACK", "DEAL"):
508 self.require_between()
509 elif command == "SHOW":
510 self.require_hand()
511 elif command == "PASS":
512 self.require_passing()
513 elif command in ("PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS", "VOIDS", "UNSEEN"):
514 self.require_playing()
515 elif command in ("SCORE", "STATS", "HISTORY", "REPLAY"):
516 self.require_game()
517 elif command == "AUTO":
518 mode = args[0]
519 if mode == "PASS":
520 self.require_passing()
521 elif mode in ("CARD", "TRICK"):
522 self.require_playing()
523 elif mode == "HAND":
524 self.require_hand()
525 else:
526 self.require_game()
527 game = self.game
528 before = len(game.log)
529 prefix = []
530 if command == "STACK":
531 duplicate_card(cards)
532 for card in cards:
533 owner = next((other for other in range(4) if other != seat and card in game.stacked.get(other, ())), None)
534 if owner is not None:
535 fail(f"{card_text(card)} stacked for {SEATS[owner]}")
536 game.stacked[seat] = set(cards)
537 return [f"stacked {SEATS[seat]}"]
538 if command == "DEAL":
539 game.deal()
540 elif command == "SHOW":
541 return [f"{SEATS[seat]}: {cards_text(game.hands[seat])}" for seat in range(4)]
542 elif command == "PASS":
543 if seat in game.passes:
544 fail(f"{SEATS[seat]} already passed")
545 duplicate_card(cards)
546 for card in cards:
547 if card not in game.hands[seat]:
548 fail(f"{SEATS[seat]} does not hold {card_text(card)}")
549 game.pass_cards(seat, cards)
550 elif command == "PLAY":
551 card = cards[0]
552 if seat != game.actor():
553 fail(f"waiting for {SEATS[game.actor()]}")
554 if card not in game.hands[seat]:
555 fail(f"{SEATS[seat]} does not hold {card_text(card)}")
556 if card not in game.legal():
557 if not game.current:
558 fail("must lead 2C" if not game.tricks else "hearts not broken")
559 led = game.current[0][1] // 13
560 if any(held // 13 == led for held in game.hands[seat]):
561 fail(f"must follow {SUITS[led]}")
562 fail("no points on first trick")
563 prefix.append(f"{SEATS[seat]} plays {card_text(card)}")
564 game.play(card)
565 elif command == "AUTO":
566 mode = args[0]
567 if mode == "PASS":
568 game.auto_pass()
569 elif mode == "CARD":
570 seat = game.actor()
571 card = game.choose_play()
572 prefix.append(f"{SEATS[seat]} plays {card_text(card)}")
573 game.play(card)
574 elif mode == "TRICK":
575 count = len(game.tricks)
576 while game.state == "playing" and len(game.tricks) == count:
577 game.play(game.choose_play())
578 elif mode == "HAND":
579 game.auto_hand()
580 else:
581 game.auto_game()
582 elif command == "UNDO":
583 if not game.undo:
584 fail("nothing to undo")
585 seat, card = game.undo_card()
586 return [f"undo {SEATS[seat]} {card_text(card)}"]
587 elif command == "CLAIM":
588 if seat != game.actor():
589 fail(f"waiting for {SEATS[game.actor()]}")
590 if game.current:
591 fail("trick in progress")
592 for card in game.hands[seat]:
593 if any(other_card // 13 == card // 13 and other_card % 13 > card % 13
594 for other in range(4) if other != seat for other_card in game.hands[other]):
595 fail("claim rejected")
596 game.claim(seat)
597 elif command == "LEGAL":
598 return [f"legal {SEATS[game.actor()]}: {cards_text(game.legal())} bot {card_text(game.choose_play())}"]
599 elif command == "PREVIEW":
600 output = []
601 for card in sorted(game.legal()):
602 copy = deepcopy(game)
603 count = len(copy.tricks)
604 copy.play(card)
605 while copy.state == "playing" and len(copy.tricks) == count:
606 copy.play(copy.choose_play())
607 trick = copy.tricks[-1]
608 points = sum(copy.point(played) for _, played in trick)
609 output.append(f"preview {card_text(card)} -> {SEATS[copy.winner(trick)]} {points}")
610 return output
611 elif command == "SCORE":
612 output = [pairs("scores", game.scores)]
613 if game.state == "playing":
614 output.append(pairs("taken", [sum(game.point(card) for card in cards) for cards in game.taken]))
615 return output
616 elif command == "TAKEN":
617 return [f"{SEATS[seat]}: {cards_text(card for card in game.taken[seat] if game.point(card) != 0)}"
618 for seat in range(4)]
619 elif command == "TRICKS":
620 output = [game.trick_line(trick, number) for number, trick in enumerate(game.tricks, 1)]
621 if game.current:
622 output.append("current: " + " ".join(f"{SEATS[seat]} {card_text(card)}" for seat, card in game.current))
623 return output or ["no tricks"]
624 elif command == "VOIDS":
625 voids = [set() for _ in SEATS]
626 for trick in game.tricks + ([game.current] if game.current else []):
627 led = trick[0][1] // 13
628 for seat, card in trick:
629 if card // 13 != led:
630 voids[seat].add(led)
631 return [pairs("voids", ["".join(SUITS[suit] for suit in sorted(missing)) or "-" for missing in voids])]
632 elif command == "UNSEEN":
633 seat = game.actor()
634 played = set().union(*game.taken) | {card for _, card in game.current}
635 unseen = set(range(52)) - game.hands[seat] - played
636 return [f"unseen {SEATS[seat]}: {cards_text(unseen)}"]
637 elif command == "STATS":

639 elif command == "HISTORY":
640 output = []

643 if hand["shooter"] is not None:

647 elif command == "REPLAY":
648 hand = int(args[0])
649 if not 1 <= hand <= len(game.history):
650 fail(f"bad value {args[0]}")
651 return game.history[hand - 1]["record"].copy()
652 return prefix + [line for _, line in game.log[before:]]
