51fun cardValue(word: String): Int {
52 if (word.length != 2 || word[0] !in RANKS || word[1] !in SUITS) fail("bad card $word")

92fun averageText(total: Int, games: Int): String {
93 val cents = (100 * total).floorDiv(games)
94 val sign = if (cents < 0) "-" else ""

98class Generator(var state: Long) {
99 fun draw(): Long {
100 state = 48271 * state % 2147483647
101 return state
102 }
103
104 fun shuffle(deck: MutableList<Int>) {
105 for (i in deck.lastIndex downTo 1) {
106 val j = (draw() % (i + 1)).toInt()
107 deck[i] = deck[j].also { deck[j] = deck[i] }
108 }
109 }
110}

112data class Rules(
113 val target: Int = 100,
114 val limit: Int = 100,
115 val moon: String = "ADD",
116 val jack: Boolean = false,
117 val qsBreak: Boolean = false,
118 val blood: Boolean = false,
119 val pass: String = "LRAH",
120) {
121 fun withRule(name: String, word: String) = when (name) {
122 "TARGET" -> copy(target = bounded(word, 1, 1000))
123 "LIMIT" -> copy(limit = bounded(word, 1, 100))
124 "MOON" -> copy(moon = choice(word, "ADD", "SUB", "AUTO"))
125 "JACK" -> copy(jack = switchValue(word))
126 "QSBREAK" -> copy(qsBreak = switchValue(word))
127 "BLOOD" -> copy(blood = switchValue(word))
128 "PASS" -> copy(pass = passValue(word))
129 else -> fail("unknown rule $name")
130 }
131
132 fun text() = "rules TARGET $target LIMIT $limit MOON $moon JACK ${onOff(jack)} " +
133 "QSBREAK ${onOff(qsBreak)} BLOOD ${onOff(blood)} PASS $pass"
134}

196class Game(val rules: Rules, val bots: Array<Strategy>, val generator: Generator) {
197 var state = State.BETWEEN
198 var number = 0
199 var direction = Direction.HOLD
200 val scores = IntArray(4)
201 val stats = Array(4) { Stats() }
202 val stacked = mutableMapOf<Int, Set<Int>>()
203 val passes = mutableMapOf<Int, List<Int>>()
204 var round = Round()
205 val undoStack = ArrayDeque<Snapshot>()
206 val history = mutableListOf<HandRecord>()
207 val log = mutableListOf<LogLine>()
208
209 fun copy() = Game(rules, bots, Generator(generator.state)).also {
210 it.state = state
211 it.number = number
212 it.direction = direction
213 scores.copyInto(it.scores)
214 stats.copyInto(it.stats)
215 it.stacked += stacked
216 it.passes += passes
217 it.round = round.copy()
218 it.undoStack += undoStack
219 it.history += history
220 it.log += log
221 }

223 fun point(card: Int) = when {
224 card == QS -> 13
225 card / 13 == 3 -> 1
226 card == JD && rules.jack -> -10
227 else -> 0
228 }

230 fun points(cards: Iterable<Int>) = cards.sumOf(::point)

232 fun trickPoints(plays: List<Play>) = plays.sumOf { point(it.card) }

234 fun write(level: Int, line: String) {
235 log += LogLine(level, line)
236 if (level > 0) round.record += line

239 fun deal() {
240 number++
241 direction = Direction.of(rules.pass[(number - 1) % rules.pass.length])
242 round = Round()
243 val used = stacked.values.flatten().toSet()
244 val deck = (0 until 52).filter { it !in used }.toMutableList()
245 generator.shuffle(deck)
246 val free = (0 until 4).filter { it !in stacked }
247 for ((seat, cards) in stacked) round.hands[seat] += cards
248 deck.forEachIndexed { i, card -> round.hands[free[i % free.size]] += card }
249 stacked.clear()
250 passes.clear()
251 undoStack.clear()
252 write(1, "hand $number pass $direction")
253 for (seat in 0 until 4) write(2, "${seatName(seat)}: ${cardsText(round.hands[seat])}")
254 if (direction == Direction.HOLD) beginPlay() else state = State.PASSING
255 }
256
257 fun beginPlay() {
258 state = State.PLAYING
259 round.leader = (0 until 4).first { TWO_OF_CLUBS in round.hands[it] }
260 write(2, "lead ${seatName(round.leader)}")
261 }
262
263 fun passCards(seat: Int, cards: List<Int>) {
264 round.hands[seat] -= cards.toSet()
265 passes[seat] = cards
266 write(2, "${seatName(seat)} passes ${cardsText(cards)}")
267 if (passes.size == 4) {
268 for ((source, passed) in passes) round.hands[(source + direction.offset) % 4] += passed
269 beginPlay()
270 }
271 }
272
273 fun actor() = (round.leader + round.current.size) % 4
274
275 fun legal(): Set<Int> {
276 val hand = round.hands[actor()]
277 if (round.current.isEmpty()) {
278 if (round.tricks.isEmpty()) return setOf(TWO_OF_CLUBS)
279 val nonHearts = hand.filter { it / 13 != 3 }.toSet()
280 return if (!round.broken && nonHearts.isNotEmpty()) nonHearts else hand.toSet()
281 }
282 val suit = round.current[0].card / 13
283 val followers = hand.filter { it / 13 == suit }.toSet()
284 if (followers.isNotEmpty()) return followers
285 if (round.tricks.isEmpty() && !rules.blood) {
286 val safe = hand.filterNot(::isPenalty).toSet()
287 if (safe.isNotEmpty()) return safe
288 }
289 return hand.toSet()
290 }

292 fun choosePass(seat: Int): List<Int> {
293 val hand = round.hands[seat]
294 val low = hand.sortedWith(BY_RANK)
295 val high = hand.sortedWith(BY_RANK_DESCENDING)
296 val hearts = high.filter { it / 13 == 3 }
297 val special = listOf(QS, KS, AS).filter { it in hand }
298 val order = when (bots[seat]) {
299 Strategy.LOW -> high
300 Strategy.HIGH -> low
301 Strategy.MOON -> low.filter { it / 13 != 3 } + low.filter { it / 13 == 3 }
302 Strategy.SMART -> {
303 val isProtected = QS in hand && countSuit(hand, 2) >= 5
304 val shortSuit = listOf(0, 1)
305 .filter { countSuit(hand, it) > 0 }
306 .minByOrNull { countSuit(hand, it) }
307 val shortCards = high.filter { it / 13 == shortSuit }
308 (if (isProtected) emptyList() else special) + shortCards + hearts + high
309 }
310 Strategy.BASIC -> special + hearts + high
311 }
312 return order.distinct().take(3)
313 }
314
315 fun queenPlayed() = round.taken.any { QS in it } || round.current.any { it.card == QS }
316
317 fun othersTookPenalty(seat: Int) =
318 (0 until 4).any { it != seat && round.taken[it].any(::isPenalty) }
319
320 fun choosePlay(): Int {
321 val seat = actor()
322 val hand = round.hands[seat]
323 val playable = legal()
324 var strategy = bots[seat]
325 if (strategy == Strategy.LOW) return lowest(playable)
326 if (strategy == Strategy.HIGH) return highest(playable)
327 if (strategy == Strategy.MOON && othersTookPenalty(seat)) strategy = Strategy.BASIC
328 if (round.current.isEmpty()) {
329 if (strategy == Strategy.SMART && !queenPlayed() && QS !in hand) {
330 val smallSpades = playable.filter { it / 13 == 2 && it % 13 < 10 }
331 if (smallSpades.isNotEmpty()) return lowest(smallSpades)
332 }
333 val suits = playable.map { it / 13 }.distinct().sorted()
334 if (strategy == Strategy.MOON) {
335 val suit = suits.maxBy { countSuit(playable, it) }
336 return highest(playable.filter { it / 13 == suit })
337 }
338 val suit = suits.minBy { countSuit(playable, it) }
339 return lowest(playable.filter { it / 13 == suit })
340 }
341 val led = round.current[0].card / 13
342 if (countSuit(hand, led) > 0) {
343 val winning = round.current.filter { it.card / 13 == led }.maxOf { it.card % 13 }
344 if (strategy == Strategy.MOON) {
345 val high = highest(playable)
346 return if (high % 13 > winning) high else lowest(playable)
347 }
348 val lower = playable.filter { it % 13 < winning }
349 if (lower.isNotEmpty()) return highest(lower)
350 if (strategy == Strategy.SMART && round.current.size == 3) {
351 val withoutQueen = playable - QS
352 return if (withoutQueen.isEmpty()) QS else highest(withoutQueen)
353 }
354 return lowest(playable)
355 }
356 if (strategy == Strategy.MOON) {
357 val safe = playable.filterNot(::isPenalty)
358 return lowest(safe.ifEmpty { playable })
359 }
360 if (QS in playable) return QS
361 if (strategy == Strategy.SMART && !queenPlayed()) {
362 if (AS in playable) return AS
363 if (KS in playable) return KS
364 }
365 val hearts = playable.filter { it / 13 == 3 }
366 return highest(hearts.ifEmpty { playable })
367 }

369 fun trickLine(plays: List<Play>, number: Int) = "trick $number: ${playsText(plays)} -> " +
370 "${seatName(trickWinner(plays))} ${trickPoints(plays)}"

372 fun play(card: Int) {
373 val seat = actor()
374 undoStack.addLast(Snapshot(round.copy(), log.size, Play(seat, card)))
375 round.hands[seat] -= card
376 round.current.add(Play(seat, card))
377 if (card / 13 == 3 || card == QS && rules.qsBreak) round.broken = true
378 if (round.current.size == 4) {
379 val winner = trickWinner(round.current)
380 round.taken[winner] += round.current.map { it.card }
381 round.trickCounts[winner]++
382 round.tricks.add(round.current)
383 write(2, trickLine(round.current, round.tricks.size))
384 round.current = mutableListOf()
385 round.leader = winner
386 if (round.tricks.size == 13) finishHand()
387 }
388 }
389
390 fun undoCard(): Play {
391 val snapshot = undoStack.removeLast()
392 round = snapshot.round
393 log.subList(snapshot.logLength, log.size).clear()
394 return snapshot.play
395 }

397 fun claim(seat: Int) {
398 val remaining = round.hands.flatten()
399 round.hands.forEach { it.clear() }
400 val count = 13 - round.tricks.size
401 round.taken[seat] += remaining
402 round.trickCounts[seat] += count
403 write(2, "claim ${seatName(seat)} tricks $count points ${points(remaining)}")
404 finishHand()
405 }

407 fun gameOver(totals: IntArray) = number >= rules.limit || totals.max() >= rules.target
408
409 fun finishHand() {
410 val handPoints = IntArray(4) { points(round.taken[it]) }
411 val shooter = (0 until 4).firstOrNull { round.taken[it].count(::isPenalty) == 14 }
412 if (shooter != null) {
413 handPoints[shooter] -= 26
414 var method = rules.moon
415 if (method == "AUTO") {
416 val added = IntArray(4) {
417 scores[it] + handPoints[it] + if (it == shooter) 0 else 26
418 }
419 method = if (gameOver(added) && ranking(added)[0] != shooter) "SUB" else "ADD"
420 }
421 if (method == "SUB") {
422 handPoints[shooter] -= 26
423 } else {
424 for (seat in 0 until 4) {
425 if (seat != shooter) handPoints[seat] += 26
426 }
427 }
428 write(1, "moon ${seatName(shooter)} $method")
429 }
430 for (seat in 0 until 4) {
431 val cards = round.taken[seat]
432 scores[seat] += handPoints[seat]
433 stats[seat] += Stats(
434 round.trickCounts[seat],
435 countSuit(cards, 3),
436 if (QS in cards) 1 else 0,
437 if (shooter == seat) 1 else 0,
438 )
439 }
440 write(1, pairs("points", handPoints.asList()))
441 write(1, pairs("scores", scores.asList()))
442 history +=
443 HandRecord(number, direction, handPoints.asList(), shooter, round.record.toList())
444 val over = gameOver(scores)
445 state = if (over) State.NONE else State.BETWEEN
446 if (over) {
447 write(0, "game over hands $number")
448 ranking(scores).forEachIndexed { place, seat ->
449 write(0, "rank ${place + 1} ${seatName(seat)} ${scores[seat]}")
450 }
451 statsLines().forEach { write(0, it) }
452 }
453 }
454
455 fun statsLines() = (0 until 4).map { "stats ${seatName(it)} ${stats[it].text()}" }
456
457 fun autoPass() {
458 for (seat in 0 until 4) {
459 if (seat !in passes) passCards(seat, choosePass(seat))
460 }
461 }
462
463 fun autoHand() {
464 if (state == State.PASSING) autoPass()
465 while (state == State.PLAYING) play(choosePlay())
466 }
467
468 fun autoGame() {
469 if (state == State.PASSING || state == State.PLAYING) autoHand()
470 while (state != State.NONE) {
471 deal()
472 autoHand()
473 }
474 }

477class Table {
478 private var rules = Rules()
479 private val bots = Array(4) { Strategy.BASIC }
480 private var game: Game? = null
481
482 private fun activeGame() = game?.takeIf { it.state != State.NONE }
483
484 private fun requireGame() = activeGame() ?: fail("no game")
485
486 private fun requireHand() = activeGame()
487 ?.takeIf { it.state == State.PASSING || it.state == State.PLAYING }
488 ?: fail("no hand")
489
490 private fun requirePlaying() =
491 activeGame()?.takeIf { it.state == State.PLAYING } ?: fail("not playing")
492
493 private fun requirePassing() =
494 activeGame()?.takeIf { it.state == State.PASSING } ?: fail("not passing")
495
496 private fun requireBetween(): Game {
497 val game = requireGame()
498 if (game.state != State.BETWEEN) fail("hand in progress")
499 return game
500 }
501
502 private fun simulate(command: String, seed: Int, count: Int): List<String> {
503 val generator = Generator(seed.toLong())
504 if (command == "GAME") {
505 val game = Game(rules, bots, generator)
506 game.autoGame()
507 return listOf("game seed $seed") + game.log.filter { it.level <= count }.map { it.text }
508 }
509 val rounds = if (command == "ROTATE") 4 else 1
510 val places = Array(4) { IntArray(4) }
511 val totals = IntArray(4)
512 val stats = Array(4) { Stats() }
513 var hands = 0
514 for (rotation in 0 until rounds) {
515 val strategies = Array(4) { bots[(it - rotation).mod(4)] }
516 repeat(count) {
517 val game = Game(rules, strategies, generator)
518 game.autoGame()
519 hands += game.number
520 val order = ranking(game.scores)
521 for (original in 0 until 4) {
522 val actual = (original + rotation) % 4
523 places[original][order.indexOf(actual)]++
524 totals[original] += game.scores[actual]
525 stats[original] += game.stats[actual]
526 }
527 }
528 }
529 val games = rounds * count
530 val output = mutableListOf("${command.lowercase()} seed $seed games $games hands $hands")
531 for (seat in 0 until 4) {
532 val name = seatName(seat) + if (command == "ROTATE") " ${bots[seat]}" else ""
533 output += "$name places ${places[seat].joinToString(" ")} total ${totals[seat]} " +
534 "average ${averageText(totals[seat], games)} ${stats[seat].text()}"
535 }
536 return output
537 }

539 fun command(words: List<String>): List<String> {
540 val name = words[0]
541 val args = words.drop(1)
542 if (name !in COMMANDS) fail("unknown command $name")
543 return when (name) {
544 "RULE" -> {
545 val changed = rules.withRule(args[0], args[1])
546 if (activeGame() != null) fail("game in progress")
547 rules = changed
548 listOf("rule ${args[0]} ${args[1]}")
549 }
550 "BOT" -> {
551 val seat = seatValue(args[0])
552 bots[seat] = strategyValue(args[1])
553 listOf("bot ${seatName(seat)} ${args[1]}")
554 }
555 "SETTINGS" -> listOf(rules.text(), pairs("bots", bots.asList()))
556 "GAME", "MATCH", "ROTATE" -> {
557 val seed = bounded(args[0], 1, 2147483646)
558 val count = when (name) {
559 "GAME" -> bounded(args[1], 0, 2)
560 "MATCH" -> bounded(args[1], 1, 50)
561 else -> bounded(args[1], 1, 10)
562 }
563 simulate(name, seed, count)
564 }
565 "NEW" -> {
566 val seed = bounded(args[0], 1, 2147483646)
567 if (activeGame() != null) fail("game in progress")
568 game = Game(rules, bots, Generator(seed.toLong()))
569 listOf("new game seed $seed")
570 }
571 "ABANDON" -> {
572 requireGame()
573 game = null
574 listOf("game abandoned")
575 }
576 else -> tableCommand(name, args)
577 }
578 }

580 private fun tableCommand(name: String, args: List<String>): List<String> {
581 val seat = if (name in listOf("STACK", "PASS", "PLAY", "CLAIM")) seatValue(args[0]) else -1
582 val cards =
583 if (name in listOf("STACK", "PASS", "PLAY")) args.drop(1).map(::cardValue) else listOf()
584 if (name == "STACK" && cards.size != 13) fail("need 13 cards")
585 if (name == "AUTO" && args[0] !in listOf("PASS", "CARD", "TRICK", "HAND", "GAME")) {
586 fail("bad value ${args[0]}")
587 }
588
589 val game = when (name) {
590 "STACK", "DEAL" -> requireBetween()
591 "SHOW" -> requireHand()
592 "PASS" -> requirePassing()
593 "PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS", "VOIDS", "UNSEEN" ->
594 requirePlaying()
595 "AUTO" -> when (args[0]) {
596 "PASS" -> requirePassing()
597 "CARD", "TRICK" -> requirePlaying()
598 "HAND" -> requireHand()
599 else -> requireGame()
600 }
601 else -> requireGame()
602 }
603 val round = game.round
604 val before = game.log.size
605 val prefix = mutableListOf<String>()
606 when (name) {
607 "STACK" -> {
608 checkDistinct(cards)
609 for (card in cards) {
610 val owner = (0 until 4).firstOrNull {
611 it != seat && card in game.stacked[it].orEmpty()
612 }
613 if (owner != null) fail("${cardText(card)} stacked for ${seatName(owner)}")
614 }
615 game.stacked[seat] = cards.toSet()
616 return listOf("stacked ${seatName(seat)}")
617 }
618 "DEAL" -> game.deal()
619 "SHOW" -> return (0 until 4).map { "${seatName(it)}: ${cardsText(round.hands[it])}" }
620 "PASS" -> {
621 if (seat in game.passes) fail("${seatName(seat)} already passed")
622 checkDistinct(cards)
623 val missing = cards.firstOrNull { it !in round.hands[seat] }
624 if (missing != null) fail("${seatName(seat)} does not hold ${cardText(missing)}")
625 game.passCards(seat, cards)
626 }
627 "PLAY" -> {
628 val card = cards[0]
629 if (seat != game.actor()) fail("waiting for ${seatName(game.actor())}")
630 if (card !in round.hands[seat]) {
631 fail("${seatName(seat)} does not hold ${cardText(card)}")
632 }
633 if (card !in game.legal()) {
634 if (round.current.isEmpty()) {
635 fail(if (round.tricks.isEmpty()) "must lead 2C" else "hearts not broken")
636 }
637 val led = round.current[0].card / 13
638 if (countSuit(round.hands[seat], led) > 0) fail("must follow ${SUITS[led]}")
639 fail("no points on first trick")
640 }
641 prefix += "${seatName(seat)} plays ${cardText(card)}"
642 game.play(card)
643 }
644 "AUTO" -> when (args[0]) {
645 "PASS" -> game.autoPass()
646 "CARD" -> {
647 val actor = game.actor()
648 val card = game.choosePlay()
649 prefix += "${seatName(actor)} plays ${cardText(card)}"
650 game.play(card)
651 }
652 "TRICK" -> {
653 val count = round.tricks.size
654 while (game.state == State.PLAYING && game.round.tricks.size == count) {
655 game.play(game.choosePlay())
656 }
657 }
658 "HAND" -> game.autoHand()
659 else -> game.autoGame()
660 }
661 "UNDO" -> {
662 if (game.undoStack.isEmpty()) fail("nothing to undo")
663 val (undoneSeat, undoneCard) = game.undoCard()
664 return listOf("undo ${seatName(undoneSeat)} ${cardText(undoneCard)}")
665 }
666 "CLAIM" -> {
667 if (seat != game.actor()) fail("waiting for ${seatName(game.actor())}")
668 if (round.current.isNotEmpty()) fail("trick in progress")
669 val beaten = round.hands[seat].any { card ->
670 (0 until 4).any { other ->
671 other != seat &&
672 round.hands[other].any { it / 13 == card / 13 && it % 13 > card % 13 }
673 }
674 }
675 if (beaten) fail("claim rejected")
676 game.claim(seat)
677 }
678 "LEGAL" -> {
679 val bot = cardText(game.choosePlay())
680 val playable = cardsText(game.legal())
681 return listOf("legal ${seatName(game.actor())}: $playable bot $bot")
682 }
683 "PREVIEW" -> return game.legal().sorted().map { card ->
684 val copy = game.copy()
685 val count = copy.round.tricks.size
686 copy.play(card)
687 while (copy.state == State.PLAYING && copy.round.tricks.size == count) {
688 copy.play(copy.choosePlay())
689 }
690 val trick = copy.round.tricks.last()
691 "preview ${cardText(card)} -> ${seatName(trickWinner(trick))} " +
692 copy.trickPoints(trick)
693 }
694 "SCORE" -> {
695 val output = mutableListOf(pairs("scores", game.scores.asList()))
696 if (game.state == State.PLAYING) {
697 output += pairs("taken", round.taken.map(game::points))
698 }
699 return output
700 }
701 "TAKEN" -> return (0 until 4).map { other ->
702 val scoring = round.taken[other].filter { game.point(it) != 0 }
703 "${seatName(other)}: ${cardsText(scoring)}"
704 }
705 "TRICKS" -> {
706 val output = round.tricks
707 .mapIndexed { i, trick -> game.trickLine(trick, i + 1) }
708 .toMutableList()
709 if (round.current.isNotEmpty()) output += "current: ${playsText(round.current)}"
710 return output.ifEmpty { listOf("no tricks") }
711 }
712 "VOIDS" -> {
713 val voids = List(4) { sortedSetOf<Int>() }
714 val tricks = (round.tricks + listOf(round.current)).filter { it.isNotEmpty() }
715 for (trick in tricks) {
716 val led = trick[0].card / 13
717 for (play in trick) {
718 if (play.card / 13 != led) voids[play.seat] += led
719 }
720 }
721 val texts = voids.map { missing ->
722 missing.joinToString("") { SUITS[it].toString() }.ifEmpty { "-" }
723 }
724 return listOf(pairs("voids", texts))
725 }
726 "UNSEEN" -> {
727 val actor = game.actor()
728 val played = round.taken.flatten() + round.current.map { it.card }
729 val unseen = (0 until 52).filter { it !in round.hands[actor] && it !in played }
730 return listOf("unseen ${seatName(actor)}: ${cardsText(unseen)}")
731 }
732 "STATS" -> return game.statsLines()
733 "HISTORY" -> return game.history.map { hand ->
734 val moon = hand.shooter?.let { " moon ${seatName(it)}" }.orEmpty()
735 "hand ${hand.number} pass ${hand.direction} ${pairs("points", hand.points)}$moon"
736 }.ifEmpty { listOf("no hands") }
737 "REPLAY" -> return game.history[bounded(args[0], 1, game.history.size) - 1].record
738 }
739 return prefix + game.log.drop(before).map { it.text }
740 }
741}

743fun main() {
744 val lines = generateSequence(::readlnOrNull).toList()
745 val table = Table()
746 val output = StringBuilder()
747 for (line in lines.drop(1).take(lines[0].trim().toInt())) {
748 val result = try {
749 table.command(line.split(" "))
750 } catch (error: CommandError) {
751 listOf("error: ${error.message}")
752 }
753 result.forEach { output.appendLine(it) }
754 }
755 print(output)
756}
