5static class Cards {
6 public const string Seats = "NESW", Suits = "CDSH", Ranks = "23456789TJQKA";
7 public static readonly string[] Strategies = { "BASIC", "LOW", "HIGH", "SMART", "MOON" };
8 public static string Text(int c) => $"{Ranks[c%13]}{Suits[c/13]}";
9 public static string Text(IEnumerable<int> cards) {
10 string s = string.Join(" ", cards.Order().Select(Text));
11 return s.Length == 0 ? "-" : s;
12 }
13 public static bool Penalty(int c) => c == 36 || c / 13 == 3;
14 public static IEnumerable<int>
15 Low(IEnumerable<int> c) => c.OrderBy(c => c % 13).ThenBy(c => c / 13);
16 public static IEnumerable<int>
17 High(IEnumerable<int> c) => c.OrderByDescending(c => c % 13).ThenBy(c => c / 13);
18 public static string Pairs<T>(string label, IEnumerable<T> values) =>
19 label + " " + string.Join(" ", values.Select((v, i) => $"{Seats[i]} {v}"));
20 public static string Direction(char c) => c switch { 'L' => "LEFT", 'R' => "RIGHT",
21 'A' => "ACROSS",
22 _ => "HOLD" };
23 public static void Fail(string reason) => throw new InvalidOperationException(reason);
24 public static int Seat(string s) {
25 int i = s.Length == 1 ? Seats.IndexOf(s[0]) : -1;
26 if (i < 0)
27 Fail("bad seat " + s);
28 return i;
29 }
30 public static int Parse(string s) {
31 if (s.Length != 2 || !Ranks.Contains(s[0]) || !Suits.Contains(s[1]))
32 Fail("bad card " + s);
33 return Suits.IndexOf(s[1]) * 13 + Ranks.IndexOf(s[0]);
34 }
35 public static int Bounded(string s, int low, int high) {
36 if (!int.TryParse(s, out int n) || n < low || n > high)
37 Fail("bad value " + s);
38 return n;
39 }
40 public static void Duplicates(IEnumerable<int> cards) {
41 HashSet<int> seen = new();
42 foreach (int c in cards)
43 if (!seen.Add(c))
44 Fail("duplicate card " + Text(c));
45 }
46}
47sealed class Rules {
48 public int Target = 100, Limit = 100;
49 public string Moon = "ADD", Pass = "LRAH";
50 public bool Jack, QsBreak, Blood;
51 public void Validate(string name, string value) {
52 switch (name) {
53 case "TARGET":
54 Cards.Bounded(value, 1, 1000);
55 return;
56 case "LIMIT":
57 Cards.Bounded(value, 1, 100);
58 return;
59 case "MOON":
60 if (new[] { "ADD", "SUB", "AUTO" }.Contains(value))
61 return;
62 break;
63 case "JACK":
64 case "QSBREAK":
65 case "BLOOD":
66 if (value == "ON" || value == "OFF")
67 return;
68 break;
69 case "PASS":
70 if (value.Length >= 1 && value.Length <= 8 && value.All(c => "LRAH".Contains(c)))
71 return;
72 break;
73 default:
74 Cards.Fail("unknown rule " + name);
75 return;
76 }
77 Cards.Fail("bad value " + value);
78 }
79 public void Set(string name, string value) {
80 switch (name) {
81 case "TARGET":
82 Target = int.Parse(value);
83 break;
84 case "LIMIT":
85 Limit = int.Parse(value);
86 break;
87 case "MOON":
88 Moon = value;
89 break;
90 case "PASS":
91 Pass = value;
92 break;
93 case "JACK":
94 Jack = value == "ON";
95 break;
96 case "QSBREAK":
97 QsBreak = value == "ON";
98 break;
99 case "BLOOD":
100 Blood = value == "ON";
101 break;
102 }
103 }
104 public override string ToString() =>
105 $"rules TARGET {Target} LIMIT {Limit} MOON {Moon} JACK {(Jack ? "ON" : "OFF")} QSBREAK {(QsBreak ? "ON" : "OFF")} BLOOD {(Blood ? "ON" : "OFF")} PASS {Pass}";
106}

112 public void Shuffle(List<int> deck) {
113 for (int i = deck.Count - 1; i > 0; --i) {
114 state = state * 48271 % 2147483647;
115 int j = (int)(state % (i + 1));
116 (deck[i], deck[j]) = (deck[j], deck[i]);
117 }
118 }

148 static HashSet<int>[] EmptySets() => Enumerable.Range(0, 4)

154 public Game(Rules rules, string[] bots, Generator generator) {
155 Rules = rules;
156 Bots = bots;
157 this.generator = generator;

159 public int Point(int c) => c == 36 ? 13 : c / 13 == 3 ? 1 : c == 22 && Rules.Jack ? -10 : 0;
160 public void Write(int level, string line) {
161 Log.Add((level, line));
162 if (level > 0)
163 record.Add(line);

165 public void Deal() {
166 ++Number;
167 PassCode = Rules.Pass[(Number - 1) % Rules.Pass.Length];
168 Hands = EmptySets();
169 foreach (var pair in Stacked)
170 Hands[pair.Key] = new(pair.Value);
171 HashSet<int> used = Hands.SelectMany(s => s).ToHashSet();
172 List<int> deck = Enumerable.Range(0, 52).Where(c => !used.Contains(c)).ToList();
173 generator.Shuffle(deck);
174 int[] free = Enumerable.Range(0, 4).Where(s => !Stacked.ContainsKey(s)).ToArray();
175 for (int i = 0; i < deck.Count; ++i)
176 Hands[free[i % free.Length]].Add(deck[i]);
177 Stacked.Clear();
178 Passes.Clear();
179 Taken = EmptySets();
180 Tricks.Clear();
181 TrickCounts = new int[4];
182 Current.Clear();
183 Broken = false;
184 undo.Clear();
185 record = new();
186 Write(1, $"hand {Number} pass {Cards.Direction(PassCode)}");
187 for (int s = 0; s < 4; ++s)
188 Write(2, $"{Cards.Seats[s]}: {Cards.Text(Hands[s])}");
189 if (PassCode == 'H')
190 BeginPlay();
191 else
192 State = TableState.Passing;
193 }
194 void BeginPlay() {
195 State = TableState.Playing;
196 Leader = Array.FindIndex(Hands, h => h.Contains(0));
197 Write(2, "lead " + Cards.Seats[Leader]);
198 }
199 public void PassCards(int seat, IEnumerable<int> cards) {
200 List<int> chosen = cards.ToList();
201 Hands[seat].ExceptWith(chosen);
202 Passes[seat] = chosen;
203 Write(2, $"{Cards.Seats[seat]} passes {Cards.Text(chosen)}");
204 if (Passes.Count == 4) {
205 int offset = PassCode == 'L' ? 1 : PassCode == 'R' ? 3 : 2;
206 foreach (var pair in Passes)
207 Hands[(pair.Key + offset) % 4].UnionWith(pair.Value);
208 BeginPlay();
209 }
210 }
211 public int Actor => (Leader + Current.Count) % 4;
212 public HashSet<int> Legal() {
213 HashSet<int> hand = Hands[Actor];
214 if (Current.Count == 0) {
215 if (Tricks.Count == 0)
216 return new() { 0 };
217 HashSet<int> safe = hand.Where(c => c / 13 != 3).ToHashSet();
218 return !Broken && safe.Count > 0 ? safe : new(hand);
219 }
220 int suit = Current[0].Card / 13;
221 HashSet<int> follow = hand.Where(c => c / 13 == suit).ToHashSet();
222 if (follow.Count > 0)
223 return follow;
224 if (Tricks.Count == 0 && !Rules.Blood) {
225 HashSet<int> safe = hand.Where(c => !Cards.Penalty(c)).ToHashSet();
226 if (safe.Count > 0)
227 return safe;
228 }
229 return new(hand);
230 }
231 public List<int> ChoosePass(int seat) {
232 HashSet<int> hand = Hands[seat];
233 string strategy = Bots[seat];
234 List<int> low = Cards.Low(hand).ToList(), high = Cards.High(hand).ToList(), order = new();
235 if (strategy == "LOW")
236 order.AddRange(high);
237 else if (strategy == "HIGH")
238 order.AddRange(low);
239 else if (strategy == "MOON") {
240 order.AddRange(low.Where(c => c / 13 != 3));
241 order.AddRange(low.Where(c => c / 13 == 3));
242 } else {
243 bool protect =
244 strategy == "SMART" && hand.Contains(36) && hand.Count(c => c / 13 == 2) >= 5;
245 if (!protect)
246 order.AddRange(new[] { 36, 37, 38 }.Where(hand.Contains));
247 if (strategy == "SMART") {
248 int[] suits = new[] { 0, 1 }
249 .Where(s => hand.Any(c => c / 13 == s))
250 .OrderBy(s => hand.Count(c => c / 13 == s))
251 .ThenBy(s => s)
252 .ToArray();
253 if (suits.Length > 0)
254 order.AddRange(high.Where(c => c / 13 == suits[0]));
255 }
256 order.AddRange(high.Where(c => c / 13 == 3));
257 order.AddRange(high);
258 }
259 return order.Distinct().Take(3).ToList();
260 }
261 bool QueenPlayed() => Taken.Any(h => h.Contains(36)) || Current.Any(p => p.Card == 36);
262 public int ChoosePlay() {
263 int seat = Actor;
264 HashSet<int> legal = Legal(), hand = Hands[seat];
265 string strategy = Bots[seat];
266 int Low(IEnumerable<int> c) => Cards.Low(c).First();
267 int High(IEnumerable<int> c) => Cards.High(c).First();
268 if (strategy == "LOW")
269 return Low(legal);
270 if (strategy == "HIGH")
271 return High(legal);
272 if (strategy == "MOON" &&
273 Enumerable.Range(0, 4).Any(s => s != seat && Taken[s].Any(Cards.Penalty)))
274 strategy = "BASIC";
275 if (Current.Count == 0) {
276 int[] spades = legal.Where(c => c / 13 == 2 && c % 13 < 10).ToArray();
277 if (strategy == "SMART" && !QueenPlayed() && !hand.Contains(36) && spades.Length > 0)
278 return Low(spades);
279 int suit =
280 legal.Select(c => c / 13)
281 .Distinct()
282 .OrderBy(s => (strategy == "MOON" ? -1 : 1) * legal.Count(c => c / 13 == s))
283 .ThenBy(s => s)
284 .First();
285 return strategy == "MOON" ? High(legal.Where(c => c / 13 == suit))
286 : Low(legal.Where(c => c / 13 == suit));
287 }
288 int led = Current[0].Card / 13;
289 if (hand.Any(c => c / 13 == led)) {
290 int winning = Current.Where(p => p.Card / 13 == led).Max(p => p.Card % 13);
291 int[] lower = legal.Where(c => c % 13 < winning).ToArray();
292 if (strategy == "MOON") {
293 int high = High(legal);
294 return high % 13 > winning ? high : Low(legal);
295 }
296 if (lower.Length > 0)
297 return High(lower);
298 if (strategy == "SMART" && Current.Count == 3) {
299 int[] safe = legal.Where(c => c != 36).ToArray();
300 return safe.Length > 0 ? High(safe) : 36;
301 }
302 return Low(legal);
303 }
304 if (strategy == "MOON") {
305 int[] safe = legal.Where(c => !Cards.Penalty(c)).ToArray();
306 return Low(safe.Length > 0 ? safe : legal);
307 }
308 if (legal.Contains(36))
309 return 36;
310 if (strategy == "SMART" && !QueenPlayed()) {
311 if (legal.Contains(38))
312 return 38;
313 if (legal.Contains(37))
314 return 37;
315 }
316 int[] hearts = legal.Where(c => c / 13 == 3).ToArray();
317 return High(hearts.Length > 0 ? hearts : legal);
318 }
319 public static int Winner(List<Play> plays) =>
320 plays.Where(p => p.Card / 13 == plays[0].Card / 13).MaxBy(p => p.Card % 13)!.Seat;
321 public string TrickLine(List<Play> plays, int number) =>
322 $"trick {number}: {string.Join(" ", plays.Select(p => $"{Cards.Seats[p.Seat]} {Cards.Text(p.Card)}"))} -> {Cards.Seats[Winner(plays)]} {plays.Sum(p => Point(p.Card))}";
323 public void PlayCard(int card) {
324 int seat = Actor;
325 undo.Add(new(CopySets(Hands), CopySets(Taken),
326 Tricks.Select(t => new List<Play>(t)).ToList(), (int[])TrickCounts.Clone(),
327 new(Current), Leader, Broken, record.Count, Log.Count, seat, card));
328 Hands[seat].Remove(card);
329 Current.Add(new(seat, card));
330 if (card / 13 == 3 || card == 36 && Rules.QsBreak)
331 Broken = true;
332 if (Current.Count == 4) {
333 int winner = Winner(Current);
334 Taken[winner].UnionWith(Current.Select(p => p.Card));
335 ++TrickCounts[winner];
336 Tricks.Add(Current);
337 Write(2, TrickLine(Current, Tricks.Count));
338 Current = new();
339 Leader = winner;
340 if (Tricks.Count == 13)
341 FinishHand();
342 }
343 }
344 public Play Undo() {
345 Snapshot s = undo[^1];
346 undo.RemoveAt(undo.Count - 1);
347 Hands = s.Hands;
348 Taken = s.Taken;
349 Tricks = s.Tricks;
350 TrickCounts = s.Counts;
351 Current = s.Current;
352 Leader = s.Leader;
353 Broken = s.Broken;
354 record.RemoveRange(s.RecordLength, record.Count - s.RecordLength);
355 Log.RemoveRange(s.LogLength, Log.Count - s.LogLength);
356 return new(s.Seat, s.Card);
357 }
358 public bool CanUndo => undo.Count > 0;
359 public void Claim(int seat) {
360 HashSet<int> all = Hands.SelectMany(h => h).ToHashSet();
361 int count = 13 - Tricks.Count;
362 Taken[seat].UnionWith(all);
363 TrickCounts[seat] += count;
364 Hands = EmptySets();
365 Write(2, $"claim {Cards.Seats[seat]} tricks {count} points {all.Sum(Point)}");
366 FinishHand();
367 }
368 void FinishHand() {
369 int[] points = Taken.Select(h => h.Sum(Point)).ToArray();
370 int shooter = Array.FindIndex(Taken, h => h.Count(Cards.Penalty) == 14);
371 if (shooter >= 0) {
372 points[shooter] -= 26;
373 string method = Rules.Moon;
374 if (method == "AUTO") {
375 int[] add = Enumerable.Range(0, 4)
376 .Select(s => Scores[s] + points[s] + (s == shooter ? 0 : 26))
377 .ToArray();
378 bool ended = Number >= Rules.Limit || add.Max() >= Rules.Target;
379 int winner = Enumerable.Range(0, 4).OrderBy(s => add[s]).ThenBy(s => s).First();
380 method = ended && winner != shooter ? "SUB" : "ADD";
381 }
382 if (method == "SUB")
383 points[shooter] -= 26;
384 else
385 for (int s = 0; s < 4; ++s)
386 if (s != shooter)
387 points[s] += 26;
388 Write(1, $"moon {Cards.Seats[shooter]} {method}");
389 }
390 for (int s = 0; s < 4; ++s) {
391 Scores[s] += points[s];
392 Stats[s][0] += TrickCounts[s];
393 Stats[s][1] += Taken[s].Count(c => c / 13 == 3);
394 Stats[s][2] += Taken[s].Contains(36) ? 1 : 0;
395 Stats[s][3] += s == shooter ? 1 : 0;
396 }
397 Write(1, Cards.Pairs("points", points));
398 Write(1, Cards.Pairs("scores", Scores));
399 History.Add(new(Number, PassCode, points, shooter, new(record)));
400 bool over = Number >= Rules.Limit || Scores.Max() >= Rules.Target;
401 State = over ? TableState.None : TableState.Between;
402 if (over) {
403 Write(0, "game over hands " + Number);
404 int rank = 0;
405 foreach (int s in Ranking())
406 Write(0, $"rank {++rank} {Cards.Seats[s]} {Scores[s]}");
407 foreach (string line in StatsLines())
408 Write(0, line);
409 }
410 }
411 public int[] Ranking() =>
412 Enumerable.Range(0, 4).OrderBy(s => Scores[s]).ThenBy(s => s).ToArray();
413 public List<string> StatsLines() =>
414 Enumerable.Range(0, 4)
415 .Select(
416 s =>
417 $"stats {Cards.Seats[s]} tricks {Stats[s][0]} hearts {Stats[s][1]} queens {Stats[s][2]} moons {Stats[s][3]}")
418 .ToList();
419 public void AutoPass() {
420 for (int s = 0; s < 4; ++s)
421 if (!Passes.ContainsKey(s))
422 PassCards(s, ChoosePass(s));
423 }
424 public void AutoHand() {
425 if (State == TableState.Passing)
426 AutoPass();
427 while (State == TableState.Playing)
428 PlayCard(ChoosePlay());
429 }
430 public void AutoGame() {
431 if (State == TableState.Passing || State == TableState.Playing)
432 AutoHand();
433 while (State != TableState.None) {
434 Deal();
435 AutoHand();
436 }
437 }
438 public (int Winner, int Points) Preview(int card) {
439 Game copy = new(Rules, Bots,
440 new Generator(1)) { Hands = CopySets(Hands),
441 Taken = CopySets(Taken),
442 Tricks = Tricks.Select(t => new List<Play>(t)).ToList(),
443 Current = new(Current),
444 Leader = Leader,
445 Broken = Broken,
446 State = TableState.Playing,
447 Number = Number,
448 TrickCounts = (int[])TrickCounts.Clone() };
449 int count = copy.Tricks.Count;
450 copy.PlayCard(card);
451 while (copy.State == TableState.Playing && copy.Tricks.Count == count)
452 copy.PlayCard(copy.ChoosePlay());
453 List<Play> trick = copy.Tricks[^1];
454 return (Winner(trick), trick.Sum(p => Point(p.Card)));
455 }

457sealed class Table {
458 readonly Rules rules = new();
459 readonly string[] bots = { "BASIC", "BASIC", "BASIC", "BASIC" };
460 Game? game;
461 bool Active => game != null && game.State != TableState.None;
462 void Require(string error, params TableState[] states) {
463 if (!Active || !states.Contains(game!.State))
464 Cards.Fail(error);
465 }
466 List<string> Simulate(string command, int seed, int count) {
467 Generator random = new(seed);
468 if (command == "GAME") {
469 Game g = new(rules, bots, random);
470 g.AutoGame();
471 return new[] { "game seed " + seed }
472 .Concat(g.Log.Where(x => x.Level <= count).Select(x => x.Line))
473 .ToList();
474 }
475 int rounds = command == "ROTATE" ? 4 : 1, hands = 0, games = rounds * count;
476 int[][] places = Enumerable.Range(0, 4)
477 .Select(
478 _ => new int[4])
479 .ToArray(),
480 stats = Enumerable.Range(0, 4)
481 .Select(
482 _ => new int[4])
483 .ToArray();
484 int[] totals = new int[4];
485 for (int r = 0; r < rounds; ++r) {
486 string[] strategies = new string[4];
487 for (int s = 0; s < 4; ++s)
488 strategies[(s + r) % 4] = bots[s];
489 for (int i = 0; i < count; ++i) {
490 Game g = new(rules, strategies, random);
491 g.AutoGame();
492 hands += g.Number;
493 int[] ranking = g.Ranking();
494 for (int s = 0; s < 4; ++s) {
495 int actual = (s + r) % 4;
496 ++places[s][Array.IndexOf(ranking, actual)];
497 totals[s] += g.Scores[actual];
498 for (int k = 0; k < 4; ++k)
499 stats[s][k] += g.Stats[actual][k];
500 }
501 }
502 }
503 List<string> output =
504 new() { $"{command.ToLowerInvariant()} seed {seed} games {games} hands {hands}" };
505 for (int s = 0; s < 4; ++s) {
506 long numerator = 100L * totals[s], cents = numerator / games;
507 if (numerator < 0 && numerator % games != 0)
508 --cents;
509 string average =
510 $"{(cents < 0 ? "-" : "")}{Math.Abs(cents)/100}.{Math.Abs(cents)%100:00}";
511 string name = Cards.Seats[s] + (command == "ROTATE" ? " " + bots[s] : "");
512 output.Add(
513 $"{name} places {string.Join(" ", places[s])} total {totals[s]} average {average} tricks {stats[s][0]} hearts {stats[s][1]} queens {stats[s][2]} moons {stats[s][3]}");
514 }
515 return output;
516 }
517 public List<string> Command(string line) {
518 string[] words = line.Split(' ');
519 string command = words[0];
520 string[] a = words.Skip(1).ToArray();
521 string[] known = { "RULE", "BOT", "SETTINGS", "GAME", "MATCH", "ROTATE", "NEW",
522 "ABANDON", "STACK", "DEAL", "SHOW", "PASS", "PLAY", "AUTO",
523 "UNDO", "CLAIM", "LEGAL", "PREVIEW", "SCORE", "TAKEN", "TRICKS",
524 "VOIDS", "UNSEEN", "STATS", "HISTORY", "REPLAY" };
525 if (!known.Contains(command))
526 Cards.Fail("unknown command " + command);
527 if (command == "RULE") {
528 rules.Validate(a[0], a[1]);
529 if (Active)
530 Cards.Fail("game in progress");
531 rules.Set(a[0], a[1]);
532 return new() { $"rule {a[0]} {a[1]}" };
533 }
534 if (command == "BOT") {
535 int s = Cards.Seat(a[0]);
536 if (!Cards.Strategies.Contains(a[1]))
537 Cards.Fail("bad value " + a[1]);
538 bots[s] = a[1];
539 return new() { $"bot {a[0]} {a[1]}" };
540 }
541 if (command == "SETTINGS")
542 return new() { rules.ToString(), Cards.Pairs("bots", bots) };
543 if (command == "GAME" || command == "MATCH" || command == "ROTATE") {
544 int seed = Cards.Bounded(a[0], 1, 2147483646);
545 int count = Cards.Bounded(a[1], command == "GAME" ? 0 : 1,
546 command == "GAME" ? 2
547 : command == "MATCH" ? 50
548 : 10);
549 return Simulate(command, seed, count);
550 }
551 if (command == "NEW") {
552 int seed = Cards.Bounded(a[0], 1, 2147483646);
553 if (Active)
554 Cards.Fail("game in progress");
555 game = new(rules, bots, new Generator(seed));
556 return new() { "new game seed " + seed };
557 }
558 if (command == "ABANDON") {
559 if (!Active)
560 Cards.Fail("no game");
561 game = null;
562 return new() { "game abandoned" };
563 }
564 int seat = -1;
565 if (new[] { "STACK", "PASS", "PLAY", "CLAIM" }.Contains(command))
566 seat = Cards.Seat(a[0]);
567 List<int> cards = new();
568 if (new[] { "STACK", "PASS", "PLAY" }.Contains(command))
569 foreach (string word in a.Skip(1))
570 cards.Add(Cards.Parse(word));
571 if (command == "STACK" && cards.Count != 13)
572 Cards.Fail("need 13 cards");
573 if (command == "AUTO" && !new[] { "PASS", "CARD", "TRICK", "HAND", "GAME" }.Contains(a[0]))
574 Cards.Fail("bad value " + a[0]);
575 if (command == "STACK" || command == "DEAL") {
576 if (!Active)
577 Cards.Fail("no game");
578 Require("hand in progress", TableState.Between);
579 } else if (command == "SHOW")
580 Require("no hand", TableState.Passing, TableState.Playing);
581 else if (command == "PASS")
582 Require("not passing", TableState.Passing);
583 else if (new[] { "PLAY", "UNDO", "CLAIM", "LEGAL", "PREVIEW", "TAKEN", "TRICKS", "VOIDS",
584 "UNSEEN" }
585 .Contains(command))
586 Require("not playing", TableState.Playing);
587 else if (command == "AUTO") {
588 if (a[0] == "PASS")
589 Require("not passing", TableState.Passing);
590 else if (a[0] == "CARD" || a[0] == "TRICK")
591 Require("not playing", TableState.Playing);
592 else if (a[0] == "HAND")
593 Require("no hand", TableState.Passing, TableState.Playing);
594 else if (!Active)
595 Cards.Fail("no game");
596 } else if (!Active)
597 Cards.Fail("no game");
598 Game g = game!;
599 int before = g.Log.Count;
600 List<string> output = new();
601 switch (command) {
602 case "STACK":
603 Cards.Duplicates(cards);
604 foreach (int stackedCard in cards) {
605 for (int other = 0; other < 4; ++other) {
606 if (other != seat && g.Stacked.TryGetValue(other, out var held) &&
607 held.Contains(stackedCard))
608 Cards.Fail($"{Cards.Text(stackedCard)} stacked for {Cards.Seats[other]}");
609 }
610 }
611 g.Stacked[seat] = cards.ToHashSet();
612 return new() { "stacked " + Cards.Seats[seat] };
613 case "DEAL":
614 g.Deal();
615 break;
616 case "SHOW":
617 return Enumerable.Range(0, 4)
618 .Select(s => $"{Cards.Seats[s]}: {Cards.Text(g.Hands[s])}")
619 .ToList();
620 case "PASS":
621 if (g.Passes.ContainsKey(seat))
622 Cards.Fail($"{Cards.Seats[seat]} already passed");
623 Cards.Duplicates(cards);
624 foreach (int c in cards)
625 if (!g.Hands[seat].Contains(c))
626 Cards.Fail($"{Cards.Seats[seat]} does not hold {Cards.Text(c)}");
627 g.PassCards(seat, cards);
628 break;
629 case "PLAY":
630 int playedCard = cards[0];
631 if (seat != g.Actor)
632 Cards.Fail("waiting for " + Cards.Seats[g.Actor]);
633 if (!g.Hands[seat].Contains(playedCard))
634 Cards.Fail($"{Cards.Seats[seat]} does not hold {Cards.Text(playedCard)}");
635 if (!g.Legal().Contains(playedCard)) {
636 if (g.Current.Count == 0)
637 Cards.Fail(g.Tricks.Count == 0 ? "must lead 2C" : "hearts not broken");
638 int led = g.Current[0].Card / 13;
639 if (g.Hands[seat].Any(c => c / 13 == led))
640 Cards.Fail("must follow " + Cards.Suits[led]);
641 Cards.Fail("no points on first trick");
642 }
643 output.Add($"{Cards.Seats[seat]} plays {Cards.Text(playedCard)}");
644 g.PlayCard(playedCard);
645 break;
646 case "AUTO":
647 switch (a[0]) {
648 case "PASS":
649 g.AutoPass();
650 break;
651 case "CARD":
652 int s = g.Actor, c = g.ChoosePlay();
653 output.Add($"{Cards.Seats[s]} plays {Cards.Text(c)}");
654 g.PlayCard(c);
655 break;
656 case "TRICK":
657 int count = g.Tricks.Count;
658 while (g.State == TableState.Playing && g.Tricks.Count == count)
659 g.PlayCard(g.ChoosePlay());
660 break;
661 case "HAND":
662 g.AutoHand();
663 break;
664 case "GAME":
665 g.AutoGame();
666 break;
667 }
668 break;
669 case "UNDO":
670 if (!g.CanUndo)
671 Cards.Fail("nothing to undo");
672 Play undone = g.Undo();
673 return new() { $"undo {Cards.Seats[undone.Seat]} {Cards.Text(undone.Card)}" };
674 case "CLAIM":
675 if (seat != g.Actor)
676 Cards.Fail("waiting for " + Cards.Seats[g.Actor]);
677 if (g.Current.Count > 0)
678 Cards.Fail("trick in progress");
679 if (g.Hands[seat].Any(c => Enumerable.Range(0, 4).Any(
680 s => s != seat && g.Hands[s].Any(d => c / 13 == d / 13 &&
681 d % 13 > c % 13))))
682 Cards.Fail("claim rejected");
683 g.Claim(seat);
684 break;
685 case "LEGAL":
686 return new() {
687 $"legal {Cards.Seats[g.Actor]}: {Cards.Text(g.Legal())} bot {Cards.Text(g.ChoosePlay())}"
688 };
689 case "PREVIEW":
690 return g.Legal()
691 .Order()
692 .Select(c => {
693 var prediction = g.Preview(c);
694 return $"preview {Cards.Text(c)} -> {Cards.Seats[prediction.Winner]} {prediction.Points}";
695 })
696 .ToList();
697 case "SCORE":
698 output.Add(Cards.Pairs("scores", g.Scores));
699 if (g.State == TableState.Playing)
700 output.Add(Cards.Pairs("taken", g.Taken.Select(h => h.Sum(g.Point))));
701 return output;
702 case "TAKEN":
703 return Enumerable.Range(0, 4)
704 .Select(
705 s => $"{Cards.Seats[s]}: {Cards.Text(g.Taken[s].Where(c => g.Point(c) != 0))}")
706 .ToList();
707 case "TRICKS":
708 output = g.Tricks.Select((t, i) => g.TrickLine(t, i + 1)).ToList();
709 if (g.Current.Count > 0)
710 output.Add(
711 "current: " +
712 string.Join(
713 " ", g.Current.Select(p => $"{Cards.Seats[p.Seat]} {Cards.Text(p.Card)}")));
714 return output.Count > 0 ? output : new() { "no tricks" };
715 case "VOIDS":
716 HashSet<int>[] missing = Enumerable.Range(0, 4)
717 .Select(
718 _ => new HashSet<int>())
719 .ToArray();
720 foreach (List<Play> t in g.Tricks.Concat(
721 g.Current.Count == 0 ? Array.Empty<List<Play>>() : new[] { g.Current }))
722 foreach (Play p in t)
723 if (p.Card / 13 != t[0].Card / 13)
724 missing[p.Seat].Add(t[0].Card / 13);
725 return new() { Cards.Pairs(
726 "voids", missing.Select(h => h.Count == 0 ? "-"
727 : string.Concat(h.Order().Select(
728 s => Cards.Suits[s])))) };
729 case "UNSEEN":
730 HashSet<int> seen = g.Taken.SelectMany(h => h)
731 .Concat(g.Current.Select(p => p.Card))
732 .Concat(g.Hands[g.Actor])
733 .ToHashSet();
734 return new() {
735 $"unseen {Cards.Seats[g.Actor]}: {Cards.Text(Enumerable.Range(0, 52).Where(c => !seen.Contains(c)))}"
736 };
737 case "STATS":
738 return g.StatsLines();
739 case "HISTORY":
740 return g.History.Count == 0
741 ? new() { "no hands" }
742 : g.History
743 .Select(
744 h =>
745 $"hand {h.Number} pass {Cards.Direction(h.Direction)} {Cards.Pairs("points", h.Points)}" +
746 (h.Shooter >= 0 ? " moon " + Cards.Seats[h.Shooter] : ""))
747 .ToList();
748 case "REPLAY":
749 int number = Cards.Bounded(a[0], 1, g.History.Count);
750 return new(g.History[number - 1].Lines);
751 }
752 output.AddRange(g.Log.Skip(before).Select(x => x.Line));
753 return output;
754 }
755}
