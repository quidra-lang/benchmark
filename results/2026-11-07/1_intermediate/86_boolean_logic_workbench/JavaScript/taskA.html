73function parseFormula(text, start, definitions) {
74 const tokens = [];
75 for (let i = 0; i < text.length;) {
76 const c = text[i],
77 position = start + i;
78 if (c === " ") {
79 i++;
80 continue;
81 }
82 if (text.startsWith("<->", i)) {
83 tokens.push({ text: "<->", position });
84 i += 3;
85 continue;
86 }
87 if (text.startsWith("->", i)) {
88 tokens.push({ text: "->", position });
89 i += 2;
90 continue;
91 }
92 if (c >= "a" && c <= "z") {
93 const begin = i++;
94 while (i < text.length && /[a-z0-9]/.test(text[i])) i++;
95 tokens.push({ text: text.slice(begin, i), position });
96 continue;
97 }
98 if (/[A-Z01~&^|()]/.test(c)) {
99 tokens.push({ text: c, position });
100 i++;
101 continue;
102 }
103 fail(`invalid character '${c}' at ${position}`);
104 }
105 tokens.push({ text: "", position: start + text.length });
106 let at = 0;
107 const unexpected = () => {
108 const token = tokens[at];
109 fail(
110 token.text === ""
111 ? "unexpected end of formula at " + token.position
112 : `unexpected '${token.text}' at ${token.position}`,
113 );
114 };
115 function unary() {
116 const token = tokens[at];
117 if (token.text === "~") {
118 at++;
119 return negate(unary());
120 }
121 if (token.text === "(") {
122 at++;
123 const result = expression(1);
124 if (tokens[at].text !== ")") unexpected();
125 at++;
126 return result;
127 }
128 if (/^[A-Z]$/.test(token.text)) {
129 at++;
130 return variable(token.text);
131 }
132 if (token.text === "0" || token.text === "1") {
133 at++;
134 return constant(Number(token.text));
135 }
136 if (/^[a-z]/.test(token.text)) {
137 at++;
138 const definition = definitions.get(token.text);
139 if (definition === undefined)
140 fail(`undefined name '${token.text}' at ${token.position}`);
141 return definition;
142 }
143 unexpected();
144 }
145 function expression(minimum) {
146 let left = unary();
147 while (precedence.has(tokens[at].text) && precedence.get(tokens[at].text) >= minimum) {
148 const operator = tokens[at++].text,
149 level = precedence.get(operator);
150 left = binary(operator, left, expression(level + (operator === "->" ? 0 : 1)));
151 }
152 return left;
153 }
154 const result = expression(1);
155 if (tokens[at].text !== "") unexpected();
156 return result;
157}
158function equal(a, b) {
159 if (a === b) return true;
160 if (a.kind !== b.kind) return false;
161 if (a.kind === "variable" && b.kind === "variable") return a.name === b.name;
162 if (a.kind === "constant" && b.kind === "constant") return a.value === b.value;
163 if (a.kind === "not" && b.kind === "not") return equal(a.child, b.child);
164 return (
165 a.kind === "binary" &&
166 b.kind === "binary" &&
167 a.operator === b.operator &&
168 equal(a.left, b.left) &&
169 equal(a.right, b.right)
170 );
171}
172function reducedNegation(a) {
173 return a.kind === "constant" ? constant(1 - a.value) : a.kind === "not" ? a.child : negate(a);

175function simplifyBinary(operator, a, b) {
176 const is = (node, value) => node.kind === "constant" && node.value === value;
177 const same = equal(a, b),
178 complementary =
179 (a.kind === "not" && equal(a.child, b)) || (b.kind === "not" && equal(a, b.child));
180 switch (operator) {
181 case "&":
182 if (is(a, 0) || is(b, 0)) return constant(0);
183 if (is(a, 1)) return b;
184 if (is(b, 1)) return a;
185 if (same) return a;
186 if (complementary) return constant(0);
187 break;
188 case "|":
189 if (is(a, 1) || is(b, 1)) return constant(1);
190 if (is(a, 0)) return b;
191 if (is(b, 0)) return a;
192 if (same) return a;
193 if (complementary) return constant(1);
194 break;
195 case "^":
196 if (is(a, 0)) return b;
197 if (is(b, 0)) return a;
198 if (is(a, 1)) return reducedNegation(b);
199 if (is(b, 1)) return reducedNegation(a);
200 if (same) return constant(0);
201 if (complementary) return constant(1);
202 break;
203 case "->":
204 if (is(a, 0) || is(b, 1)) return constant(1);
205 if (is(a, 1)) return b;
206 if (is(b, 0)) return reducedNegation(a);
207 if (same) return constant(1);
208 if (complementary) return b;
209 break;
210 case "<->":
211 if (is(a, 1)) return b;
212 if (is(b, 1)) return a;
213 if (is(a, 0)) return reducedNegation(b);
214 if (is(b, 0)) return reducedNegation(a);
215 if (same) return constant(1);
216 if (complementary) return constant(0);
217 break;
218 }
219 return binary(operator, a, b);
220}
221function simplify(node, replacements = new Map()) {
222 if (node.kind === "constant") return node;
223 if (node.kind === "variable")
224 return replacements.has(node.name) ? constant(replacements.get(node.name)) : node;
225 if (node.kind === "not") return reducedNegation(simplify(node.child, replacements));
226 return simplifyBinary(
227 node.operator,
228 simplify(node.left, replacements),
229 simplify(node.right, replacements),
230 );

241function nnfCompute(node, positive) {
242 if (node.kind === "constant") return constant(positive ? node.value : 1 - node.value);
243 if (node.kind === "variable") return positive ? node : negate(node);
244 if (node.kind === "not") return nnf(node.child, !positive);
245 const a = node.left,
246 b = node.right,
247 operator = node.operator;
248 if (operator === "&" || operator === "|")
249 return binary(
250 positive ? operator : operator === "&" ? "|" : "&",
251 nnf(a, positive),
252 nnf(b, positive),
253 );
254 if (operator === "->") return binary(positive ? "|" : "&", nnf(a, !positive), nnf(b, positive));
255 const same = operator === "<->" ? positive : !positive;
256 return binary(
257 "|",
258 binary("&", nnf(a, true), nnf(b, same)),
259 binary("&", nnf(a, false), nnf(b, !same)),
260 );

262function dual(node) {
263 if (node.kind === "variable") return node;
264 if (node.kind === "constant") return constant(1 - node.value);
265 if (node.kind === "not") return negate(dual(node.child));
266 const a = dual(node.left),
267 b = dual(node.right);
268 return node.operator === "->"
269 ? binary("&", negate(a), b)
270 : binary(
271 node.operator === "&"
272 ? "|"
273 : node.operator === "|"
274 ? "&"
275 : node.operator === "^"
276 ? "<->"
277 : "^",
278 a,
279 b,
280 );

282function evaluate(node, names, row) {
283 if (node.kind === "constant") return node.value;
284 if (node.kind === "variable") return (row >> (names.length - names.indexOf(node.name) - 1)) & 1;
285 if (node.kind === "not") return 1 - evaluate(node.child, names, row);
286 const a = evaluate(node.left, names, row),
287 b = evaluate(node.right, names, row);
288 return node.operator === "&"
289 ? a & b
290 : node.operator === "|"
291 ? a | b
292 : node.operator === "^"
293 ? a ^ b
294 : node.operator === "->"
295 ? (1 - a) | b
296 : Number(a === b);
297}

308function patternCompare(a, b) {
309 const rank = { 1: 0, 0: 1, "-": 2 };
310 for (let i = 0; i < a.length; i++) if (a[i] !== b[i]) return rank[a[i]] - rank[b[i]];
311 return 0;

325function coverText(patterns, names, product) {
326 if (patterns.length === 0) return product ? "0" : "1";
327 return patterns

329 const text = patternText(pattern, names, product);
330 return !product && patterns.length >= 2 && literalCount(pattern) >= 2

336function coefficients(values) {
337 const result = [...values];
338 for (let bit = 1; bit < values.length; bit *= 2)
339 for (let row = 0; row < values.length; row++)
340 if (row & bit) result[row] ^= result[row ^ bit];
341 return result;

343function minimize(names, on, dc, pos) {

345 const primes = [],
346 output = [];
347 for (let number = 0; ; number++) {
348 const used = new Set(),
349 next = new Set();
350 for (const pattern of stage)
351 for (let i = 0; i < pattern.length; i++)
352 if (pattern[i] === "0") {
353 const partner = pattern.slice(0, i) + "1" + pattern.slice(i + 1);
354 if (stage.has(partner)) {
355 used.add(pattern);
356 used.add(partner);
357 next.add(pattern.slice(0, i) + "-" + pattern.slice(i + 1));
358 }
359 }
360 const current = [...stage].filter((pattern) => !used.has(pattern));
361 primes.push(...current);
362 if (!pos) output.push(`stage ${number}: implicants ${stage.size} primes ${current.length}`);
363 if (next.size === 0) break;
364 stage = next;
365 }
366 const rows = on.map((row) => bits(row, names.length));
367 const target = primes
368 .filter((pattern) => rows.some((row) => contains(pattern, row)))
369 .sort(patternCompare);
370 const masks = target.map((pattern) =>
371 rows.reduce(
372 (mask, row, i) => (contains(pattern, row) ? mask | (1n << BigInt(i)) : mask),
373 0n,
374 ),
375 );
376 const essential = new Set();
377 for (let i = 0; i < rows.length; i++) {
378 const candidates = target
379 .map((_, j) => j)
380 .filter((j) => (masks[j] & (1n << BigInt(i))) !== 0n);
381 if (candidates.length === 1) essential.add(candidates[0]);
382 }
383 const fixed = [...essential],
384 optional = target.map((_, i) => i).filter((i) => !essential.has(i));
385 const full = (1n << BigInt(rows.length)) - 1n;
386 let best = null,
387 bestLiterals = 0;
388 const ordered = (indices) =>
389 indices.map((i) => (pos ? flip(target[i]) : target[i])).sort(patternCompare);
390 function better(candidate, literals) {
391 if (best === null || candidate.length !== best.length)
392 return best === null || candidate.length < best.length;
393 if (literals !== bestLiterals) return literals < bestLiterals;
394 const a = ordered(candidate),
395 b = ordered(best);
396 for (let i = 0; i < a.length; i++) {
397 const order = patternCompare(a[i], b[i]);
398 if (order !== 0) return order < 0;
399 }
400 return false;
401 }
402 for (let subset = 0; subset < 2 ** optional.length; subset++) {
403 const candidate = [...fixed];
404 let covered = 0n,
405 literals = 0;
406 for (let i = 0; i < optional.length; i++)
407 if (subset & (1 << i)) candidate.push(optional[i]);
408 if (best !== null && candidate.length > best.length) continue;
409 for (const index of candidate) {
410 covered |= masks[index];
411 literals += literalCount(target[index]);
412 }
413 if (covered === full && better(candidate, literals)) {
414 best = candidate;
415 bestLiterals = literals;
416 }
417 }
418 const chosen = ordered(best);
419 if (pos)
420 return [
421 `minimal: ${coverText(chosen, names, false)}`,
422 `clauses ${chosen.length} literals ${bestLiterals}`,
423 ];
424 output.push(
425 "primes: " +
426 (target.map((pattern) => patternText(pattern, names, true)).join(", ") || "none"),
427 );
428 output.push(
429 "essential: " +
430 (fixed
431 .map((i) => target[i])
432 .sort(patternCompare)
433 .map((pattern) => patternText(pattern, names, true))
434 .join(", ") || "none"),
435 );
436 output.push(
437 "minimal: " + coverText(chosen, names, true),
438 `terms ${chosen.length} literals ${bestLiterals}`,
439 );
440 return output;
441}
442function mergeClauses(a, b) {
443 let result = "";
444 for (let i = 0; i < a.length; i++) {
445 if (a[i] !== "-" && b[i] !== "-" && a[i] !== b[i]) return null;
446 result += a[i] === "-" ? b[i] : a[i];
447 }
448 return result;

450function tidy(patterns) {
451 const unique = [...new Set(patterns)];
452 return unique.filter(
453 (pattern) =>
454 !unique.some(
455 (other) =>
456 literalCount(other) < literalCount(pattern) &&
457 [...other].every((c, i) => c === "-" || c === pattern[i]),

459 );

461function clauses(node, names, cache = new Map()) {
462 if (cache.has(node)) return cache.get(node);
463 const result = clausesCompute(node, names, cache);
464 cache.set(node, result);

467function clausesCompute(node, names, cache) {
468 if (node.kind === "constant") return node.value ? [] : ["-".repeat(names.length)];
469 if (node.kind === "variable" || node.kind === "not") {
470 const name = node.kind === "variable" ? node.name : node.child.name,
471 i = names.indexOf(name);
472 return [
473 "-".repeat(i) + (node.kind === "not" ? "0" : "1") + "-".repeat(names.length - i - 1),
474 ];
475 }

477 b = clauses(node.right, names, cache);

481 for (const y of b) {
482 const value = mergeClauses(x, y);

484 }

755 weights = new Map();

757 symmetric = true;

759 const bit = 1 << (names.length - 1 - i);
760 if (values.some((value, row) => value !== values[row ^ bit]))
761 depends.push(names[i]);
762 if (values.some((value, row) => !(row & bit) && value > values[row | bit]))
763 monotone = false;

766 const count = [...bits(row, names.length)].filter((c) => c === "1").length;
767 if (weights.has(count) && weights.get(count) !== values[row]) symmetric = false;
768 weights.set(count, values[row]);

773 "zero-preserving: " + yes(values[0] === 0),

779 "symmetric: " + yes(symmetric),

783 unique = new Map();
784 function build(index, list) {
785 if (index === names.length) return list[0];

787 lo = build(index + 1, list.slice(0, half)),
788 hi = build(index + 1, list.slice(half));
789 if (lo === hi) return lo;
790 const key = [index, lo, hi].join(",");
791 if (unique.has(key)) return unique.get(key);
792 const id = nodes.length + 2;
793 nodes.push({ index, lo, hi });
794 unique.set(key, id);
795 return id;
796 }
797 const root = build(0, values),

800 [1, 1],
801 ]);

803 if (memo.has(id)) return memo.get(id);
804 const node = nodes[id - 2],
805 count = paths(node.lo) + paths(node.hi);
806 memo.set(id, count);
807 return count;
808 }
809 return [
810 "nodes: " + nodes.length,
811 ...nodes.map(
812 (node, i) => `${i + 2} = ite(${names[node.index]}, ${node.hi}, ${node.lo})`,
813 ),
814 "root: " + root,
815 "paths: " + paths(root),
816 ];
817 }
