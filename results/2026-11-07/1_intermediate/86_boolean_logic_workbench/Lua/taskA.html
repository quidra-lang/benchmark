1local function leaf(kind)
2 return {kind = kind}
3end
4local function unary(child)
5 return {kind = "~", left = child}
6end
7local function binary(kind, left, right)
8 return {kind = kind, left = left, right = right}
9end
10local priorities = {["<->"] = 1, ["->"] = 2, ["|"] = 3, ["^"] = 4, ["&"] = 5}
11local function precedence(kind)
12 return priorities[kind] or 0
13end
14local function notation(node)
15 if not node.left then
16 return node.kind
17 end
18 if node.kind == "~" then
19 local text = notation(node.left)
20 return "~" .. (precedence(node.left.kind) > 0 and "(" .. text .. ")" or text)
21 end
22 local left, right = notation(node.left), notation(node.right)
23 local p = precedence(node.kind)
24 if
25 precedence(node.left.kind) > 0 and
26 (precedence(node.left.kind) < p or node.kind == "->" and node.left.kind == "->")
27 then
28 left = "(" .. left .. ")"
29 end
30 if
31 precedence(node.right.kind) > 0 and
32 (precedence(node.right.kind) < p or node.kind == node.right.kind and node.kind ~= "->")
33 then
34 right = "(" .. right .. ")"
35 end
36 return left .. " " .. node.kind .. " " .. right
37end
38local function equal(a, b)
39 return a == b or
40 a.kind == b.kind and
41 (not a.left or equal(a.left, b.left) and (not a.right or equal(a.right, b.right)))
42end

48end

67local function lexical(text, start)
68 local result, i = {}, 1
69 while i <= #text do
70 local c, begin = text:sub(i, i), i
71 if c == " " then
72 i = i + 1
73 else
74 if text:sub(i, i + 2) == "<->" then
75 i = i + 3
76 elseif text:sub(i, i + 1) == "->" then
77 i = i + 2
78 elseif c >= "a" and c <= "z" then
79 repeat
80 i = i + 1
81 c = text:sub(i, i)
82 until c == "" or not (c >= "a" and c <= "z" or c >= "0" and c <= "9")
83 elseif ("ABCDEFGHIJKLMNOPQRSTUVWXYZ01~&^|()"):find(c, 1, true) then
84 i = i + 1
85 else
86 fail("invalid character '%s' at %d", c, start + i - 1)
87 end
88 result[#result + 1] = {text = text:sub(begin, i - 1), position = start + begin - 1}
89 end
90 end
91 return result
92end
93local Parser = {}
94Parser.__index = Parser
95function Parser:unexpected()
96 local token = self.tokens[self.at]
97 if not token then
98 fail("unexpected end of formula at %d", self.ending)
99 end
100 fail("unexpected '%s' at %d", token.text, token.position)
101end
102function Parser:unary()
103 local token = self.tokens[self.at]
104 if not token then
105 self:unexpected()
106 end
107 if token.text == "~" then
108 self.at = self.at + 1
109 return unary(self:unary())
110 end
111 if token.text == "(" then
112 self.at = self.at + 1
113 local value = self:expression(1)
114 if not self.tokens[self.at] or self.tokens[self.at].text ~= ")" then
115 self:unexpected()
116 end
117 self.at = self.at + 1
118 return value
119 end
120 if variable(token.text) or token.text == "0" or token.text == "1" then
121 self.at = self.at + 1
122 return leaf(token.text)
123 end
124 if token.text:sub(1, 1) >= "a" and token.text:sub(1, 1) <= "z" then
125 local value = self.names[token.text]
126 if not value then
127 fail("undefined name '%s' at %d", token.text, token.position)
128 end
129 self.at = self.at + 1
130 return value
131 end
132 self:unexpected()
133end
134function Parser:expression(minimum)
135 local value = self:unary()
136 while self.tokens[self.at] do
137 local operator = self.tokens[self.at].text
138 local priority = precedence(operator)
139 if priority < minimum then
140 break
141 end
142 self.at = self.at + 1
143 value =
144 binary(operator, value, self:expression(operator == "->" and priority or priority + 1))
145 end
146 return value
147end
148local function parse(text, start, names)
149 local parser =
150 setmetatable(
151 {tokens = lexical(text, start), at = 1, ending = start + #text, names = names},
152 Parser
153 )
154 local value = parser:expression(1)
155 if parser.tokens[parser.at] then
156 parser:unexpected()
157 end
158 return value
159end
160local function variable_list(...)

163 if variable(node.kind) then
164 set[node.kind] = true
165 end
166 if node.left then

168 end
169 if node.right then

171 end
172 end

183local function size(node)
184 return not node.left and 1 or not node.right and 1 + size(node.left) or
185 1 + size(node.left) + size(node.right)
186end
187local function depth(node)
188 return not node.left and 1 or not node.right and 1 + depth(node.left) or
189 1 + math.max(depth(node.left), depth(node.right))

191local function truth(node, assignment)
192 if variable(node.kind) then
193 return assignment[node.kind]
194 end
195 if node.kind == "0" then
196 return 0
197 elseif node.kind == "1" then
198 return 1
199 end
200 local a = truth(node.left, assignment)
201 if node.kind == "~" then
202 return 1 - a
203 end
204 local b = truth(node.right, assignment)
205 if node.kind == "&" then
206 return a & b
207 elseif node.kind == "|" then
208 return a | b
209 elseif node.kind == "^" then
210 return a ~ b
211 elseif node.kind == "->" then
212 return (1 - a) | b
213 else
214 return 1 - (a ~ b)
215 end
216end
217local function truth_table(node, names)
218 local result = {}
219 for row = 0, (1 << #names) - 1 do
220 local assignment = {}
221 for i, name in ipairs(names) do
222 assignment[name] = (row >> (#names - i)) & 1
223 end
224 result[row + 1] = truth(node, assignment)
225 end
226 return result
227end
228local function assignment_text(names, row)
229 local parts = {}
230 for i, name in ipairs(names) do
231 parts[#parts + 1] = name .. "=" .. ((row >> (#names - i)) & 1)
232 end
233 return "{" .. table.concat(parts, " ") .. "}"
234end
235local function bit_pattern(row, n)
236 local parts = {}
237 for i = 1, n do
238 parts[i] = tostring((row >> (n - i)) & 1)
239 end
240 return table.concat(parts)
241end
242local function literal_count(pattern)
243 local _, count = pattern:gsub("%-", "")
244 return #pattern - count
245end
246local pattern_rank = {["1"] = 0, ["0"] = 1, ["-"] = 2}
247local function pattern_less(a, b)
248 for i = 1, #a do
249 local x, y = pattern_rank[a:sub(i, i)], pattern_rank[b:sub(i, i)]
250 if x ~= y then
251 return x < y
252 end
253 end
254 return false
255end
256local function sort_patterns(patterns)

258end
259local function flipped(pattern)
260 return (pattern:gsub("[01]", {["0"] = "1", ["1"] = "0"}))
261end
262local function term_text(pattern, names, clause)
263 local literals = {}
264 for i = 1, #pattern do
265 local c = pattern:sub(i, i)
266 if c ~= "-" then
267 literals[#literals + 1] = (c == "0" and "~" or "") .. names[i]
268 end
269 end
270 if #literals == 0 then
271 return clause and "0" or "1"
272 end
273 return table.concat(literals, clause and " | " or " & ")
274end
275local function form_text(patterns, names, clauses)
276 if #patterns == 0 then
277 return clauses and "1" or "0"
278 end
279 local parts = {}
280 for _, pattern in ipairs(patterns) do
281 local text = term_text(pattern, names, clauses)
282 if clauses and #patterns >= 2 and literal_count(pattern) >= 2 then
283 text = "(" .. text .. ")"
284 end
285 parts[#parts + 1] = text
286 end
287 return table.concat(parts, clauses and " & " or " | ")
288end
289local function reduce_negation(node)
290 if node.kind == "0" then
291 return leaf("1")
292 elseif node.kind == "1" then
293 return leaf("0")
294 elseif node.kind == "~" then
295 return node.left
296 else
297 return unary(node)
298 end
299end
300local function simplify_binary(operator, a, b)
301 local same = equal(a, b)
302 local complement = a.kind == "~" and equal(a.left, b) or b.kind == "~" and equal(b.left, a)
303 if operator == "&" then
304 if a.kind == "0" or b.kind == "0" then
305 return leaf("0")
306 elseif a.kind == "1" then
307 return b
308 elseif b.kind == "1" or same then
309 return a
310 elseif complement then
311 return leaf("0")
312 end
313 elseif operator == "|" then
314 if a.kind == "1" or b.kind == "1" then
315 return leaf("1")
316 elseif a.kind == "0" then
317 return b
318 elseif b.kind == "0" or same then
319 return a
320 elseif complement then
321 return leaf("1")
322 end
323 elseif operator == "^" then
324 if a.kind == "0" then
325 return b
326 elseif b.kind == "0" then
327 return a
328 elseif a.kind == "1" then
329 return reduce_negation(b)
330 elseif b.kind == "1" then
331 return reduce_negation(a)
332 elseif same then
333 return leaf("0")
334 elseif complement then
335 return leaf("1")
336 end
337 elseif operator == "->" then
338 if a.kind == "0" or b.kind == "1" then
339 return leaf("1")
340 elseif a.kind == "1" then
341 return b
342 elseif b.kind == "0" then
343 return reduce_negation(a)
344 elseif same then
345 return leaf("1")
346 elseif complement then
347 return b
348 end
349 elseif operator == "<->" then
350 if a.kind == "1" then
351 return b
352 elseif b.kind == "1" then
353 return a
354 elseif a.kind == "0" then
355 return reduce_negation(b)
356 elseif b.kind == "0" then
357 return reduce_negation(a)
358 elseif same then
359 return leaf("1")
360 elseif complement then
361 return leaf("0")
362 end
363 end
364 return binary(operator, a, b)
365end
366local function simplify(node, assignments)
367 if not node.left then
368 return assignments and assignments[node.kind] and leaf(assignments[node.kind]) or node
369 end
370 local a = simplify(node.left, assignments)
371 if node.kind == "~" then
372 return reduce_negation(a)
373 end
374 return simplify_binary(node.kind, a, simplify(node.right, assignments))
375end
376local function nnf(node, positive, memo)
377 memo[node] = memo[node] or {}
378 local sign = positive and 1 or 2
379 if memo[node][sign] then
380 return memo[node][sign]
381 end
382 local kind, result = node.kind
383 if kind == "0" or kind == "1" then
384 result = positive and node or reduce_negation(node)
385 elseif kind == "~" then
386 result = nnf(node.left, not positive, memo)
387 elseif kind == "&" or kind == "|" then
388 local operator = kind
389 if not positive then
390 operator = kind == "&" and "|" or "&"
391 end
392 result = binary(operator, nnf(node.left, positive, memo), nnf(node.right, positive, memo))
393 elseif kind == "->" then
394 result =
395 binary(
396 positive and "|" or "&",
397 nnf(node.left, not positive, memo),
398 nnf(node.right, positive, memo)
399 )
400 elseif kind == "^" or kind == "<->" then
401 local same_sign = positive
402 if kind == "^" then
403 same_sign = not same_sign
404 end
405 result =
406 binary(
407 "|",
408 binary("&", nnf(node.left, true, memo), nnf(node.right, same_sign, memo)),
409 binary("&", nnf(node.left, false, memo), nnf(node.right, not same_sign, memo))
410 )
411 else
412 result = positive and node or unary(node)
413 end
414 memo[node][sign] = result
415 return result
416end
417local function dual(node)
418 if variable(node.kind) then
419 return node
420 elseif node.kind == "0" or node.kind == "1" then
421 return reduce_negation(node)
422 elseif node.kind == "~" then
423 return unary(dual(node.left))
424 end
425 local a, b = dual(node.left), dual(node.right)
426 local swap = {["&"] = "|", ["|"] = "&", ["^"] = "<->", ["<->"] = "^"}
427 if node.kind == "->" then
428 return binary("&", unary(a), b)
429 end
430 return binary(swap[node.kind], a, b)
431end
432local function covers(pattern, row)
433 local bits = bit_pattern(row, #pattern)
434 for i = 1, #pattern do
435 local c = pattern:sub(i, i)
436 if c ~= "-" and c ~= bits:sub(i, i) then
437 return false
438 end
439 end
440 return true
441end
442local function pattern_subset(a, b)
443 for i = 1, #a do
444 local c = a:sub(i, i)
445 if c ~= "-" and c ~= b:sub(i, i) then
446 return false
447 end
448 end
449 return true
450end
451local function tidy(patterns)
452 local unique, result = {}, {}
453 for _, pattern in ipairs(patterns) do
454 if not contains(unique, pattern) then
455 unique[#unique + 1] = pattern
456 end
457 end
458 for i, pattern in ipairs(unique) do
459 local redundant = false
460 for j, other in ipairs(unique) do
461 if i ~= j and pattern_subset(other, pattern) then
462 redundant = true
463 break
464 end
465 end
466 if not redundant then
467 result[#result + 1] = pattern
468 end
469 end
470 return result
471end
472local function merge_clauses(a, b)
473 local result = {}
474 for i = 1, #a do
475 local x, y = a:sub(i, i), b:sub(i, i)
476 if x ~= "-" and y ~= "-" and x ~= y then
477 return nil
478 end
479 result[i] = y ~= "-" and y or x
480 end
481 return table.concat(result)
482end
483local function clauses(node, names, memo)
484 if memo[node] then
485 return memo[node]
486 end
487 local empty = string.rep("-", #names)
488 local result = {}
489 if node.kind == "1" then
490 elseif node.kind == "0" then
491 result = {empty}
492 elseif node.kind == "&" then
493 for _, pattern in ipairs(clauses(node.left, names, memo)) do
494 result[#result + 1] = pattern
495 end
496 for _, pattern in ipairs(clauses(node.right, names, memo)) do
497 result[#result + 1] = pattern
498 end
499 result = tidy(result)
500 elseif node.kind == "|" then
501 for _, a in ipairs(clauses(node.left, names, memo)) do
502 for _, b in ipairs(clauses(node.right, names, memo)) do
503 local merged = merge_clauses(a, b)
504 if merged then
505 result[#result + 1] = merged
506 end
507 end
508 end
509 result = tidy(result)
510 else
511 local name, sign = node.kind, "1"
512 if node.kind == "~" then
513 name, sign = node.left.kind, "0"
514 end
515 local pattern = {}
516 for i, variable_name in ipairs(names) do
517 pattern[i] = variable_name == name and sign or "-"
518 end
519 result = {table.concat(pattern)}
520 end
521 memo[node] = result
522 return result
523end

531local function minimize(values, dc, names, pos)
532 local on, stage = {}, {}
533 for index, value in ipairs(values) do
534 if value == 1 and (not dc or dc[index] == 0) then
535 on[#on + 1] = index - 1
536 end
537 if value == 1 or dc and dc[index] == 1 then
538 stage[#stage + 1] = bit_pattern(index - 1, #names)
539 end
540 end
541 local out, primes, level = {}, {}, 0
542 while true do
543 local used, next_stage = {}, {}
544 for i, a in ipairs(stage) do
545 for j = i + 1, #stage do
546 local b, differing = stage[j], nil
547 for k = 1, #a do
548 local x, y = a:sub(k, k), b:sub(k, k)
549 if x ~= y then
550 if x == "-" or y == "-" or differing then
551 differing = false
552 break
553 end
554 differing = k
555 end
556 end
557 if differing then
558 local pattern = a:sub(1, differing - 1) .. "-" .. a:sub(differing + 1)
559 used[a], used[b] = true, true
560 if not contains(next_stage, pattern) then
561 next_stage[#next_stage + 1] = pattern
562 end
563 end
564 end
565 end
566 local count = 0
567 for _, pattern in ipairs(stage) do
568 if not used[pattern] then
569 primes[#primes + 1] = pattern
570 count = count + 1
571 end
572 end
573 if not pos then
574 out[#out + 1] = string.format("stage %d: implicants %d primes %d", level, #stage, count)
575 end
576 if #next_stage == 0 then
577 break
578 end
579 stage, level = next_stage, level + 1
580 end
581 local target = {}
582 for _, pattern in ipairs(primes) do
583 for _, row in ipairs(on) do
584 if covers(pattern, row) then
585 target[#target + 1] = pattern
586 break
587 end
588 end
589 end
590 sort_patterns(target)
591 local essential = {}
592 for _, row in ipairs(on) do
593 local options = {}
594 for _, pattern in ipairs(target) do
595 if covers(pattern, row) then
596 options[#options + 1] = pattern
597 end
598 end
599 if #options == 1 and not contains(essential, options[1]) then
600 essential[#essential + 1] = options[1]
601 end
602 end
603 sort_patterns(essential)
604 local optional = {}
605 for _, pattern in ipairs(target) do
606 if not contains(essential, pattern) then
607 optional[#optional + 1] = pattern
608 end
609 end
610 local best, best_count, best_literals = nil, math.huge, math.huge
611 for mask = 0, (1 << #optional) - 1 do
612 local chosen = list_copy(essential)
613 for i, pattern in ipairs(optional) do
614 if (mask >> (i - 1)) & 1 ~= 0 then
615 chosen[#chosen + 1] = pattern
616 end
617 end
618 local valid = true
619 for _, row in ipairs(on) do
620 local found = false
621 for _, pattern in ipairs(chosen) do
622 if covers(pattern, row) then
623 found = true
624 break
625 end
626 end
627 if not found then
628 valid = false
629 break
630 end
631 end
632 if valid then
633 local literals = 0
634 for _, pattern in ipairs(chosen) do
635 literals = literals + literal_count(pattern)
636 end
637 local candidate = list_copy(chosen)
638 if pos then
639 for i, pattern in ipairs(candidate) do
640 candidate[i] = flipped(pattern)
641 end
642 end
643 sort_patterns(candidate)
644 local better =
645 #chosen < best_count or #chosen == best_count and literals < best_literals
646 if #chosen == best_count and literals == best_literals then
647 for i, pattern in ipairs(candidate) do
648 if pattern ~= best[i] then
649 better = pattern_less(pattern, best[i])
650 break
651 end
652 end
653 end
654 if better then
655 best, best_count, best_literals = candidate, #chosen, literals
656 end
657 end
658 end
659 if pos then
660 return {
661 "minimal: " .. form_text(best, names, true),
662 string.format("clauses %d literals %d", best_count, best_literals)
663 }
664 end

681local function coefficients(values, n)
682 local result = list_copy(values)
683 for bit = 0, n - 1 do
684 for row = 0, #values - 1 do
685 if (row >> bit) & 1 ~= 0 then
686 result[row + 1] = result[row + 1] ~ result[(row ~ (1 << bit)) + 1]
687 end
688 end
689 end
690 return result
691end

787 not word:match("^%d+$") or word ~= "0" and word:sub(1, 1) == "0" or not row or
788 row >= (1 << #front) or
789 row <= previous
790 then
791 fail("bad minterm '%s'", word)
792 end
793 previous = row
794 local product
795 for i, variable_name in ipairs(front) do
796 local literal = leaf(variable_name)
797 if (row >> (#front - i)) & 1 == 0 then
798 literal = unary(literal)
799 end
800 product = product and binary("&", product, literal) or literal
801 end
802 formula = formula and binary("|", formula, product) or product
803 end
804 formula = formula or leaf("0")
805 local prefix = self.names[name] and "redefined" or "defined"
806 self.names[name] = formula
807 return {prefix .. " " .. name .. " = " .. notation(formula)}
808 end
809 local front, back, start, back_start = argument, "", #command + 2, 0
810 if contains({"RESTRICT", "COFACTOR", "MINDC", "EQUIV", "IMPLIES"}, command) then
811 local semicolon = argument:find(";", 1, true)
812 if not semicolon then
813 fail("missing ';'")
814 end
815 front, back, back_start =
816 argument:sub(1, semicolon - 1),
817 argument:sub(semicolon + 1),
818 start + semicolon
819 end
820 local formula = parse(front, start, self.names)
821 local second
822 if command == "MINDC" or command == "EQUIV" or command == "IMPLIES" then
823 second = parse(back, back_start, self.names)
824 end
825 if command == "RESTRICT" then
826 local assignments, parts = {}, words(back)
827 if #parts == 0 then
828 fail("missing assignment")
829 end
830 for _, word in ipairs(parts) do
831 if
832 #word ~= 3 or not variable(word:sub(1, 1)) or word:sub(2, 2) ~= "=" or
833 word:sub(3, 3) ~= "0" and word:sub(3, 3) ~= "1"
834 then
835 fail("bad assignment '%s'", word)
836 end
837 local name = word:sub(1, 1)
838 if assignments[name] then
839 fail("duplicate variable %s", name)
840 end
841 assignments[name] = word:sub(3)
842 end
843 return {"restricted: " .. notation(simplify(formula, assignments))}
844 elseif command == "COFACTOR" then
845 local parts = words(back)
846 if #parts ~= 1 or not variable(parts[1]) then
847 fail("bad variable")
848 end
849 local name = parts[1]
850 local one, zero = simplify(formula, {[name] = "1"}), simplify(formula, {[name] = "0"})
851 return {
852 "cofactor " .. name .. "=1: " .. notation(one),
853 "cofactor " .. name .. "=0: " .. notation(zero),
854 "difference: " .. notation(simplify_binary("^", one, zero))
855 }
856 end
857 local names = second and variable_list(formula, second) or variable_list(formula)
858 if contains(truth_commands, command) and #names >= 9 then
859 fail("too many variables: %d", #names)
860 end
861 if command == "PRINT" then
862 return {
863 "formula: " .. notation(formula),
864 "variables: " .. (#names == 0 and "none" or table.concat(names, " ")),
865 string.format("size %d depth %d", size(formula), depth(formula))
866 }
867 elseif command == "SIMPLIFY" then
868 return {"simplified: " .. notation(simplify(formula))}
869 elseif command == "NNF" then
870 return {"nnf: " .. notation(nnf(formula, true, {}))}
871 elseif command == "DUAL" then
872 return {"dual: " .. notation(dual(formula))}
873 elseif command == "CLAUSES" or command == "DP" then
874 local set = clauses(nnf(formula, true, {}), names, {})
875 sort_patterns(set)
876 if command == "CLAUSES" then
877 return {"clauses: " .. form_text(set, names, true)}
878 end
879 local out = {"start: clauses " .. #set}
880 local empty = string.rep("-", #names)
881 for i, name in ipairs(names) do
882 if #set == 0 or contains(set, empty) then
883 break
884 end
885 local p, q, rest = {}, {}, {}
886 for _, pattern in ipairs(set) do
887 local c = pattern:sub(i, i)
888 local target = c == "1" and p or c == "0" and q or rest
889 target[#target + 1] = pattern
890 end
891 local resolvents = {}
892 for _, a in ipairs(p) do
893 for _, b in ipairs(q) do
894 local merged =
895 merge_clauses(
896 a:sub(1, i - 1) .. "-" .. a:sub(i + 1),
897 b:sub(1, i - 1) .. "-" .. b:sub(i + 1)
898 )
899 if merged and not contains(resolvents, merged) then
900 resolvents[#resolvents + 1] = merged
901 end
902 end
903 end
904 for _, pattern in ipairs(resolvents) do
905 rest[#rest + 1] = pattern
906 end
907 set = tidy(rest)
908 out[#out + 1] =
909 string.format("eliminate %s: resolvents %d clauses %d", name, #resolvents, #set)
910 end
911 out[#out + 1] = "result: " .. (contains(set, empty) and "unsatisfiable" or "satisfiable")
912 return out
913 end
914 local values = truth_table(formula, names)
915 local models, false_rows = {}, {}
916 for index, value in ipairs(values) do
917 local target = value == 1 and models or false_rows
918 target[#target + 1] = index - 1
919 end
920 local function row_patterns(rows, flip)
921 local result = {}
922 for _, row in ipairs(rows) do
923 local pattern = bit_pattern(row, #names)
924 result[#result + 1] = flip and flipped(pattern) or pattern
925 end
926 return result
927 end
928 if command == "TABLE" then
929 local out = {(#names == 0 and "" or table.concat(names, " ") .. " ") .. "| F"}
930 for index, value in ipairs(values) do
931 local parts = {}
932 for c in bit_pattern(index - 1, #names):gmatch(".") do
933 parts[#parts + 1] = c
934 end
935 parts[#parts + 1] = "|"
936 parts[#parts + 1] = value
937 out[#out + 1] = table.concat(parts, " ")
938 end
939 return out
940 elseif command == "MINTERMS" or command == "MAXTERMS" then
941 local rows = command == "MINTERMS" and models or false_rows
942 return {command:lower() .. ": " .. (#rows == 0 and "none" or table.concat(rows, " "))}
943 elseif command == "DNF" then
944 return {"dnf: " .. form_text(row_patterns(models), names, false)}
945 elseif command == "CNF" then
946 return {"cnf: " .. form_text(row_patterns(false_rows, true), names, true)}
947 elseif command == "ANF" then
948 local patterns = {}
949 for index, coefficient in ipairs(coefficients(values, #names)) do
950 if coefficient == 1 then
951 patterns[#patterns + 1] = (bit_pattern(index - 1, #names):gsub("0", "-"))
952 end
953 end
954 table.sort(
955 patterns,
956 function(a, b)
957 local x, y = literal_count(a), literal_count(b)
958 return x < y or x == y and pattern_less(a, b)
959 end
960 )
961 local parts = {}
962 for _, pattern in ipairs(patterns) do
963 parts[#parts + 1] = term_text(pattern, names, false)
964 end
965 return {"anf: " .. (#parts == 0 and "0" or table.concat(parts, " ^ "))}
966 elseif command == "SAT" then
967 return {
968 #models == 0 and "unsatisfiable" or "satisfiable: " .. assignment_text(names, models[1])
969 }
970 elseif command == "TAUT" then
971 return {
972 #false_rows == 0 and "tautology" or
973 "not a tautology: " .. assignment_text(names, false_rows[1])
974 }
975 elseif command == "COUNT" then
976 return {string.format("models: %d of %d", #models, #values)}
977 elseif command == "EQUIV" or command == "IMPLIES" then
978 local other = truth_table(second, names)
979 for index, value in ipairs(values) do
980 if command == "EQUIV" and value ~= other[index] then

992 end
993 end
994 return {command == "EQUIV" and "equivalent" or "implies"}
995 elseif command == "MINIMIZE" then
996 return minimize(values, nil, names, false)
997 elseif command == "MINDC" then

999 elseif command == "MINPOS" then
1000 local inverted = {}

1003 end

1005 elseif command == "KMAP" then

1009 end

1101 i + 1,
1102 names[node.variable],

1105 )

1107 end
