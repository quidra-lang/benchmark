106static void append(Buffer *b, const char *text) {
107 size_t n = strlen(text);
108 if (b->length + n + 1 > b->capacity) {
109 size_t capacity = b->capacity ? b->capacity : 64;
110 while (capacity < b->length + n + 1) capacity *= 2;
111 char *p = realloc(b->text, capacity);
112 if (!p) exit(EXIT_FAILURE);
113 b->text = p;
114 b->capacity = capacity;
115 }
116 memcpy(b->text + b->length, text, n + 1);
117 b->length += n;
118}

157static Node *make(char op, Node *left, Node *right) {
158 unsigned long hash = (unsigned char)op;
159 hash = hash * 1000003 + (unsigned long)(left ? left->id + 1 : 0);
160 hash = hash * 1000003 + (unsigned long)(right ? right->id + 1 : 0);
161 size_t bucket = hash % 16381;
162 for (Node *n = node_buckets[bucket]; n; n = n->next) if (n->op == op && n->left == left && n->right == right) return n;
163 Node *n = persist(sizeof *n);
164 n->op = op;
165 n->left = left;
166 n->right = right;
167 n->id = node_count++;
168 n->variables = upper(op) ? 1u << (op - 'A') : (left ? left->variables : 0) | (right ? right->variables : 0);
169 n->next = node_buckets[bucket];
170 node_buckets[bucket] = n;
171 return n;
172}

190static const char *print(Node *n) {
191 if (n->printed) return n->printed;
192 Buffer b = {0};
193 if (!n->left) append_char(&b, n->op);
194 else if (!n->right) {
195 append(&b, "~");
196 if (n->left->right) append(&b, "(");
197 append(&b, print(n->left));
198 if (n->left->right) append(&b, ")");
199 } else {
200 int p = precedence(n->op);
201 bool left = n->left->right && (precedence(n->left->op) < p || (n->op == '>' && n->left->op == '>'));
202 bool right = n->right->right && (precedence(n->right->op) < p || (n->op == n->right->op && n->op != '>'));
203 if (left) append(&b, "(");
204 append(&b, print(n->left));
205 if (left) append(&b, ")");
206 append(&b, " ");
207 append(&b, operator_text(n->op));
208 append(&b, " ");
209 if (right) append(&b, "(");
210 append(&b, print(n->right));
211 if (right) append(&b, ")");
212 }
213 n->printed = finish(&b, true);
214 return n->printed;
215}

277static Parser parser(const char *text, int start) {
278 Parser p = {.end = start + (int)strlen(text)};
279 for (size_t i = 0; text[i];) {
280 char c = text[i];
281 if (c == ' ') {
282 i++;
283 continue;
284 }
285 size_t first = i;
286 char kind;
287 if (!strncmp(text + i, "<->", 3)) {
288 i += 3;
289 kind = '=';
290 } else if (!strncmp(text + i, "->", 2)) {
291 i += 2;
292 kind = '>';
293 } else if (lower(c)) {
294 i++;
295 while (lower(text[i]) || digit(text[i])) i++;
296 kind = 'n';
297 } else if (upper(c) || strchr("01~&^|()", c)) {
298 i++;
299 kind = c;
300 } else {
301 fail("invalid character '%c' at %d", c, start + (int)i);
302 kind = 0;
303 }
304 p.tokens[p.count++] = (Token){kind, part(text + first, i - first), start + (int)first};
305 }
306 return p;
307}

309static char peek(Parser *p) {
310 return p->at == p->count ? '\0' : p->tokens[p->at].kind;
311}

313static void unexpected(Parser *p) {
314 if (p->at == p->count) fail("unexpected end of formula at %d", p->end);
315 Token t = p->tokens[p->at];
316 fail("unexpected '%s' at %d", t.text, t.position);
317}
318static Node *expression(Parser *p, int level);
319
320static Node *unary(Parser *p) {
321 char c = peek(p);
322 if (c == '~') {
323 p->at++;
324 return negate(unary(p));
325 }
326 if (c == '(') {
327 p->at++;
328 Node *n = expression(p, 1);
329 if (peek(p) != ')') unexpected(p);
330 p->at++;
331 return n;
332 }
333 if (upper(c) || c == '0' || c == '1') {
334 p->at++;
335 return leaf(c);
336 }
337 if (c == 'n') {
338 Token t = p->tokens[p->at++];
339 int index = find(t.text);
340 if (index < 0) fail("undefined name '%s' at %d", t.text, t.position);
341 return definitions[index].node;
342 }
343 unexpected(p);
344 return NULL;
345}

347static Node *expression(Parser *p, int level) {
348 if (level == 6) return unary(p);
349 Node *left = expression(p, level + 1);
350 const char operators[] = "=>|^&";
351 char op = operators[level - 1];
352 while (peek(p) == op) {
353 p->at++;
354 Node *right = expression(p, level == 2 ? level : level + 1);
355 left = make(op, left, right);
356 }
357 return left;
358}

360static Node *parse(const char *text, int start) {
361 Parser p = parser(text, start);
362 Node *n = expression(&p, 1);
363 if (p.at != p.count) unexpected(&p);
364 return n;
365}

367static Node *nnf(Node *n, bool positive) {
368 if (n->normal[positive]) return n->normal[positive];
369 Node *out;
370 if (!n->left) out = n->op == '0' || n->op == '1' ? leaf(positive ? n->op : n->op == '0' ? '1' : '0') : positive ? n : negate(n);
371 else if (n->op == '~') out = nnf(n->left, !positive);
372 else if (n->op == '&' || n->op == '|') out = make(positive ? n->op : n->op == '&' ? '|' : '&', nnf(n->left, positive), nnf(n->right, positive));
373 else if (n->op == '>') out = make(positive ? '|' : '&', nnf(n->left, !positive), nnf(n->right, positive));
374 else {
375 bool same = (n->op == '=') == positive;
376 Node *a = make('&', nnf(n->left, true), nnf(n->right, same));
377 Node *b = make('&', nnf(n->left, false), nnf(n->right, !same));
378 out = make('|', a, b);
379 }
380 n->normal[positive] = out;
381 return out;
382}

384static Node *dual(Node *n) {
385 if (n->dual) return n->dual;
386 Node *out;
387 if (!n->left) out = n->op == '0' ? leaf('1') : n->op == '1' ? leaf('0') : n;
388 else if (n->op == '~') out = negate(dual(n->left));
389 else {
390 Node *a = dual(n->left), *b = dual(n->right);
391 out = n->op == '&' ? make('|', a, b) : n->op == '|' ? make('&', a, b) : n->op == '^' ? make('=', a, b) : n->op == '=' ? make('^', a, b) : make('&', negate(a), b);
392 }
393 n->dual = out;
394 return out;
395}

397static Node *reduce_not(Node *n) {

402 return (a->op == '~' && a->left == b) || (b->op == '~' && b->left == a);

405static Node *simplify_binary(char op, Node *a, Node *b) {
406 bool a0 = a->op == '0', a1 = a->op == '1', b0 = b->op == '0', b1 = b->op == '1';
407 switch (op) {
408 case '&':
409 if (a0 || b0) return leaf('0');
410 if (a1) return b;
411 if (b1 || a == b) return a;
412 if (complementary(a, b)) return leaf('0');
413 break;
414 case '|':
415 if (a1 || b1) return leaf('1');
416 if (a0) return b;
417 if (b0 || a == b) return a;
418 if (complementary(a, b)) return leaf('1');
419 break;
420 case '^':
421 if (a0) return b;
422 if (b0) return a;
423 if (a1) return reduce_not(b);
424 if (b1) return reduce_not(a);
425 if (a == b) return leaf('0');
426 if (complementary(a, b)) return leaf('1');
427 break;
428 case '>':
429 if (a0 || b1) return leaf('1');
430 if (a1) return b;
431 if (b0) return reduce_not(a);
432 if (a == b) return leaf('1');
433 if (complementary(a, b)) return b;
434 break;
435 case '=':
436 if (a1) return b;
437 if (b1) return a;
438 if (a0) return reduce_not(b);
439 if (b0) return reduce_not(a);
440 if (a == b) return leaf('1');
441 if (complementary(a, b)) return leaf('0');
442 break;
443 }
444 return make(op, a, b);
445}

447static Node *simplify_walk(Node *n, const int *substitutions) {
448 if (n->simplify_generation == simplify_generation) return n->simplified;
449 Node *out;
450 if (!n->left) out = upper(n->op) && substitutions[n->op - 'A'] >= 0 ? leaf((char)('0' + substitutions[n->op - 'A'])) : n;
451 else if (n->op == '~') out = reduce_not(simplify_walk(n->left, substitutions));
452 else {
453 Node *a = simplify_walk(n->left, substitutions), *b = simplify_walk(n->right, substitutions);
454 out = simplify_binary(n->op, a, b);
455 }
456 n->simplify_generation = simplify_generation;
457 n->simplified = out;
458 return out;
459}

461static Node *simplify(Node *n, const int *substitutions) {
462 simplify_generation++;
463 return simplify_walk(n, substitutions);
464}

466static int evaluate(Node *n, unsigned true_variables) {
467 if (n->evaluate_generation == evaluate_generation) return n->evaluated;
468 int out;
469 if (!n->left) out = upper(n->op) ? !!(true_variables & 1u << (n->op - 'A')) : n->op - '0';
470 else if (n->op == '~') out = 1 - evaluate(n->left, true_variables);
471 else {
472 int a = evaluate(n->left, true_variables), b = evaluate(n->right, true_variables);
473 out = n->op == '&' ? a & b : n->op == '|' ? a | b : n->op == '^' ? a ^ b : n->op == '>' ? (1 - a) | b : 1 - (a ^ b);
474 }
475 n->evaluate_generation = evaluate_generation;
476 n->evaluated = out;
477 return out;
478}

593typedef struct {
594 char pattern[9];
595 unsigned mask, values;
596 Coverage coverage;
597} Implicant;

599 Implicant *items;
600 int count, capacity;
601} Implicants;

603static void all_implicants(int at, int n, unsigned mask, unsigned values, char pattern[9], const bool *allowed, Implicants *out) {
604 if (at == n) {
605 Coverage coverage = {
606 {0}
607 };
608 for (int row = 0; row < 1 << n; row++) if ((((unsigned)row ^ values) & mask) == 0) {
609 if (!allowed[row]) return;
610 cover_set(&coverage, row);
611 }
612 if (out->count == out->capacity) {
613 int next = out->capacity ? out->capacity * 2 : 8;
614 Implicant *items = allocate((size_t)next * sizeof(Implicant));
615 if (out->count) memcpy(items, out->items, (size_t)out->count * sizeof(Implicant));
616 out->items = items;
617 out->capacity = next;
618 }
619 Implicant p = {.mask = mask, .values = values, .coverage = coverage};
620 pattern[n] = '\0';
621 memcpy(p.pattern, pattern, (size_t)n + 1);
622 out->items[out->count++] = p;
623 return;
624 }
625 unsigned bit = 1u << (n - 1 - at);
626 pattern[at] = '1';
627 all_implicants(at + 1, n, mask | bit, values | bit, pattern, allowed, out);
628 pattern[at] = '0';
629 all_implicants(at + 1, n, mask | bit, values, pattern, allowed, out);
630 pattern[at] = '-';
631 all_implicants(at + 1, n, mask, values, pattern, allowed, out);
632}
633static bool product_of_sums;
634
635static int implicant_compare(const void *a, const void *b) {
636 const char *x = (*(Implicant *const *)a)->pattern, *y = (*(Implicant *const *)b)->pattern;
637 if (!product_of_sums) return pattern_compare(x, y);
638 for (; *x; x++, y++) {
639 int p = *x == '0' ? 0 : *x == '1' ? 1 : 2, q = *y == '0' ? 0 : *y == '1' ? 1 : 2;
640 if (p != q) return p - q;
641 }
642 return 0;
643}

648 if (count > 1) qsort(patterns, (size_t)count, sizeof(char *), pattern_pointer_compare);

652static int cover_literals(Implicant **cover, int count) {
653 int total = 0;
654 for (int i = 0; i < count; i++) total += bits(cover[i]->mask);

658static void minimize(const char *vars, int n, const int *f, const int *dc, bool pos) {
659 bool allowed[256] = {false};
660 Coverage on = {
661 {0}
662 };
663 for (int row = 0; row < 1 << n; row++) {
664 bool selected = (pos ? 1 - f[row] : f[row]) == 1 && (!dc || dc[row] == 0);
665 if (selected) cover_set(&on, row);
666 allowed[row] = selected || (dc && dc[row] == 1);
667 }
668 Implicants all = {0};
669 char pattern[9];
670 all_implicants(0, n, 0, 0, pattern, allowed, &all);
671 bool *keys = allocate(65536 * sizeof(bool));
672 for (int i = 0; i < all.count; i++) keys[(all.items[i].mask << 8) | all.items[i].values] = true;
673 Implicant **primes = allocate((size_t)all.count * sizeof(Implicant *));
674 int prime_count = 0, stages[9] = {0}, prime_stages[9] = {0};
675 for (int i = 0; i < all.count; i++) {
676 Implicant *p = &all.items[i];
677 int stage = n - bits(p->mask);
678 stages[stage]++;
679 bool prime = true;
680 for (unsigned bit = 1; bit < 1u << n; bit <<= 1) if ((p->mask & bit) && keys[((p->mask ^ bit) << 8) | (p->values & ~bit)]) {
681 prime = false;
682 break;
683 }
684 if (prime) {
685 prime_stages[stage]++;
686 if (cover_intersects(p->coverage, on)) primes[prime_count++] = p;
687 }
688 }
689 product_of_sums = false;
690 if (prime_count > 1) qsort(primes, (size_t)prime_count, sizeof(Implicant *), implicant_compare);
691 bool *essential_flags = allocate((size_t)prime_count * sizeof(bool));
692 for (int row = 0; row < 1 << n; row++) if (cover_has(on, row)) {
693 int only = -1, count = 0;
694 for (int i = 0; i < prime_count; i++) if (cover_has(primes[i]->coverage, row)) {
695 only = i;
696 count++;
697 }
698 if (count == 1) essential_flags[only] = true;
699 }
700 Implicant **essential = allocate((size_t)prime_count * sizeof(Implicant *)), **optional = allocate((size_t)prime_count * sizeof(Implicant *));
701 int essential_count = 0, optional_count = 0;
702 for (int i = 0; i < prime_count; i++) if (essential_flags[i]) essential[essential_count++] = primes[i];
703 else optional[optional_count++] = primes[i];
704 Implicant **candidate = allocate((size_t)prime_count * sizeof(Implicant *)), **best = allocate((size_t)prime_count * sizeof(Implicant *));
705 int best_count = -1, best_literals = 0;
706 product_of_sums = pos;
707 for (unsigned mask = 0; mask < 1u << optional_count; mask++) {
708 int count = essential_count;
709 if (count) memcpy(candidate, essential, (size_t)count * sizeof(Implicant *));
710 for (int i = 0; i < optional_count; i++) if (mask & 1u << i) candidate[count++] = optional[i];
711 if (best_count >= 0 && count > best_count) continue;
712 Coverage covered = {
713 {0}
714 };
715 for (int i = 0; i < count; i++) cover_union(&covered, candidate[i]->coverage);
716 if (!cover_contains(covered, on)) continue;
717 int total = cover_literals(candidate, count);
718 if (count > 1) qsort(candidate, (size_t)count, sizeof(Implicant *), implicant_compare);
719 bool better = best_count < 0 || count < best_count || (count == best_count && total < best_literals);
720 if (!better && count == best_count && total == best_literals) for (int i = 0; i < count; i++) {
721 int c = implicant_compare(&candidate[i], &best[i]);
722 if (c) {
723 better = c < 0;
724 break;
725 }
726 }
727 if (better) {
728 best_count = count;
729 best_literals = total;
730 if (count) memcpy(best, candidate, (size_t)count * sizeof(Implicant *));
731 }
732 }
733 if (!pos) {
734 for (int stage = 0; stage <= n; stage++) {
735 printf("stage %d: implicants %d primes %d\n", stage, stages[stage], prime_stages[stage]);
736 if (stage == n || !stages[stage + 1]) break;
737 }
738 fputs("primes: ", stdout);
739 if (!prime_count) fputs("none", stdout);
740 else for (int i = 0; i < prime_count; i++) {
741 if (i) fputs(", ", stdout);
742 fputs(term(primes[i]->pattern, vars, false), stdout);
743 }
744 putchar('\n');
745 fputs("essential: ", stdout);
746 if (!essential_count) fputs("none", stdout);
747 else for (int i = 0; i < essential_count; i++) {
748 if (i) fputs(", ", stdout);
749 fputs(term(essential[i]->pattern, vars, false), stdout);
750 }
751 putchar('\n');
752 }
753 printf("minimal: %s\n", form(cover_patterns(best, best_count, pos), best_count, vars, pos));
754 printf("%s %d literals %d\n", pos ? "clauses" : "terms", best_count, best_literals);
755}

757static bool clause_subset(Clause a, Clause b) {
758 return (a.positive & b.positive) == a.positive && (a.negative & b.negative) == a.negative;
759}
760
761static bool tautological(Clause c) {
762 return (c.positive & c.negative) != 0;
763}

765static void add_clause(Clauses *set, Clause clause) {
766 if (set->count == set->capacity) {
767 int next = set->capacity ? set->capacity * 2 : 8;
768 Clause *items = allocate((size_t)next * sizeof(Clause));
769 if (set->count) memcpy(items, set->items, (size_t)set->count * sizeof(Clause));
770 set->items = items;
771 set->capacity = next;
772 }
773 set->items[set->count++] = clause;
774}
775
776static int clause_compare(const void *a, const void *b) {
777 const Clause *x = a, *y = b;
778 if (x->positive != y->positive) return x->positive < y->positive ? -1 : 1;
779 return (x->negative > y->negative) - (x->negative < y->negative);
780}
781
782static int clause_length_compare(const void *a, const void *b) {
783 const Clause *x = a, *y = b;
784 int c = bits(x->positive | x->negative) - bits(y->positive | y->negative);
785 return c ? c : clause_compare(a, b);
786}

788static Clauses unique_clauses(Clauses set) {
789 if (set.count > 1) qsort(set.items, (size_t)set.count, sizeof(Clause), clause_compare);
790 int count = 0;
791 for (int i = 0; i < set.count; i++) if (!tautological(set.items[i]) && (!count || clause_compare(&set.items[i], &set.items[count - 1]))) set.items[count++] = set.items[i];
792 set.count = count;
793 return set;
794}
795
796static Clauses tidy(Clauses set) {
797 if (set.count > 1) qsort(set.items, (size_t)set.count, sizeof(Clause), clause_length_compare);
798 Clauses out = {0};
799 for (int i = 0; i < set.count; i++) if (!tautological(set.items[i])) {
800 bool contains = false;
801 for (int j = 0; j < out.count; j++) if (clause_subset(out.items[j], set.items[i])) {
802 contains = true;
803 break;
804 }
805 if (!contains) add_clause(&out, set.items[i]);
806 }
807 return out;
808}

810static Clauses clauses(Node *n) {
811 if (n->clause_generation == generation) return n->clause_cache;
812 Clauses out = {0};
813 if (n->op == '0') add_clause(&out, (Clause){0, 0});
814 else if (n->op == '1') {
815 } else if (!n->left) add_clause(&out, (Clause){1u << (n->op - 'A'), 0});
816 else if (n->op == '~') add_clause(&out, (Clause){0, 1u << (n->left->op - 'A')});
817 else {
818 Clauses a = clauses(n->left), b = clauses(n->right);
819 if (n->op == '&') {
820 for (int i = 0; i < a.count; i++) add_clause(&out, a.items[i]);
821 for (int i = 0; i < b.count; i++) add_clause(&out, b.items[i]);
822 } else for (int i = 0; i < a.count; i++) for (int j = 0; j < b.count; j++) {
823 Clause c = {a.items[i].positive | b.items[j].positive, a.items[i].negative | b.items[j].negative};
824 if (!tautological(c)) add_clause(&out, c);
825 }
826 out = tidy(out);
827 }
828 n->clause_generation = generation;
829 n->clause_cache = out;
830 return out;
831}

834 for (int i = 0; i < set.count; i++) if (!(set.items[i].positive | set.items[i].negative)) return true;
835 return false;
