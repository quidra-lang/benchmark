5private class Formula(val op: String, val left: Formula? = null, val right: Formula? = null)

7private fun leaf(op: String) = Formula(op)
8
9private fun negation(a: Formula) = Formula("~", a)
10
11private fun binary(op: String, a: Formula, b: Formula) = Formula(op, a, b)

13private fun same(a: Formula, b: Formula): Boolean = a === b || a.op == b.op && when {
14 a.left == null && b.left == null -> true
15 a.left == null || b.left == null -> false
16 a.right == null && b.right == null -> same(a.left, b.left)
17 a.right == null || b.right == null -> false
18 else -> same(a.left, b.left) && same(a.right, b.right)
19}
20private fun precedence(op: String) = when (op) {
21 "<->" -> 1
22 "->" -> 2
23 "|" -> 3
24 "^" -> 4
25 "&" -> 5
26 "~" -> 6
27 else -> 7
28}
29private fun notation(f: Formula): String {
30 val a = f.left ?: return f.op
31 if (f.op == "~") return "~" + if (a.right != null) "(${notation(a)})" else notation(a)
32 val b = f.right!!
33 val left = if (a.right != null && (precedence(a.op) < precedence(f.op) || a.op == "->" && f.op == "->")) "(${notation(a)})" else notation(a)
34 val right = if (b.right != null && (precedence(b.op) < precedence(f.op) || b.op == f.op && f.op != "->")) "(${notation(b)})" else notation(b)
35 return "$left ${f.op} $right"
36}
37private fun variables(f: Formula): Set<Char> = if (f.left == null) (if (f.op[0] in 'A'..'Z') setOf(f.op[0]) else emptySet()) else variables(f.left) + (f.right?.let(::variables) ?: emptySet())
38
39private fun evaluate(f: Formula, vars: List<Char>, row: Int): Int {
40 val a = f.left ?: return when (f.op) {
41 "0" -> 0
42 "1" -> 1
43 else -> row shr (vars.size - 1 - vars.indexOf(f.op[0])) and 1
44 }
45 val x = evaluate(a, vars, row)
46 if (f.op == "~") return 1 - x
47 val y = evaluate(f.right!!, vars, row)
48 return when (f.op) {
49 "&" -> x and y
50 "|" -> x or y
51 "^" -> x xor y
52 "->" -> (1 - x) or y
53 else -> if (x == y) 1 else 0
54 }
55}
56private data class Token(val text: String, val position: Int)

58private class Parser(text: String, start: Int, val definitions: Map<String, Formula>) {
59 val tokens = mutableListOf<Token>()
60 var at = 0
61 val end = start + text.length
62 init {
63 var i = 0
64 while (i < text.length) {
65 if (text[i] == ' ') {
66 i++
67 continue
68 }
69 val begin = i
70 when {
71 text.startsWith("<->", i) -> i += 3
72 text.startsWith("->", i) -> i += 2
73 text[i] in 'a'..'z' -> {
74 i++
75 while (i < text.length && (text[i] in 'a'..'z' || text[i] in '0'..'9')) i++
76 }
77 text[i] in 'A'..'Z' || text[i] in "01~&^|()" -> i++
78 else -> fail("invalid character '${text[i]}' at ${start + i}")
79 }
80 tokens.add(Token(text.substring(begin, i), start + begin))
81 }
82 }
83 fun peek() = tokens.getOrNull(at)?.text ?: ""
84
85 fun error(): Nothing = fail(tokens.getOrNull(at)?.let {
86 "unexpected '${it.text}' at ${it.position}"
87 } ?: "unexpected end of formula at $end")
88
89 fun unary(): Formula {
90 if (peek() == "~") {
91 at++
92 return negation(unary())
93 }
94 if (peek() == "(") {
95 at++
96 val f = expression(1)
97 if (peek() != ")") error()
98 at++
99 return f
100 }
101 val t = tokens.getOrNull(at) ?: error()
102 if (t.text[0] in 'a'..'z') {
103 val f = definitions[t.text] ?: fail("undefined name '${t.text}' at ${t.position}")
104 at++
105 return f
106 }
107 if (t.text == "0" || t.text == "1" || t.text.length == 1 && t.text[0] in 'A'..'Z') {
108 at++
109 return leaf(t.text)
110 }
111 error()
112 }
113 fun expression(level: Int): Formula {
114 if (level == 6) return unary()
115 var f = expression(level + 1)
116 val op = listOf("", "<->", "->", "|", "^", "&")[level]
117 while (peek() == op) {
118 at++
119 f = binary(op, f, expression(if (op == "->") level else level + 1))
120 }
121 return f
122 }
123 fun parse(): Formula {
124 val f = expression(1)
125 if (at != tokens.size) error()
126 return f
127 }
128}
129private fun negate(a: Formula) = when (a.op) {
130 "0" -> leaf("1")
131 "1" -> leaf("0")
132 "~" -> a.left!!
133 else -> negation(a)
134}
135private fun simplifyPair(op: String, a: Formula, b: Formula): Formula {
136 val eq = same(a, b)
137 val complementary = a.op == "~" && same(a.left!!, b) || b.op == "~" && same(a, b.left!!)
138 val result = when (op) {
139 "&" -> when {
140 a.op == "0" || b.op == "0" -> leaf("0")
141 a.op == "1" -> b
142 b.op == "1" || eq -> a
143 complementary -> leaf("0")
144 else -> null
145 }
146 "|" -> when {
147 a.op == "1" || b.op == "1" -> leaf("1")
148 a.op == "0" -> b
149 b.op == "0" || eq -> a
150 complementary -> leaf("1")
151 else -> null
152 }
153 "^" -> when {
154 a.op == "0" -> b
155 b.op == "0" -> a
156 a.op == "1" -> negate(b)
157 b.op == "1" -> negate(a)
158 eq -> leaf("0")
159 complementary -> leaf("1")
160 else -> null
161 }
162 "->" -> when {
163 a.op == "0" || b.op == "1" -> leaf("1")
164 a.op == "1" -> b
165 b.op == "0" -> negate(a)
166 eq -> leaf("1")
167 complementary -> b
168 else -> null
169 } else -> when {
170 a.op == "1" -> b
171 b.op == "1" -> a
172 a.op == "0" -> negate(b)
173 b.op == "0" -> negate(a)
174 eq -> leaf("1")
175 complementary -> leaf("0")
176 else -> null
177 }
178 }
179 return result ?: binary(op, a, b)
180}
181private fun simplify(f: Formula, assignments: Map<Char, Int> = emptyMap()): Formula {
182 val child = f.left ?: return assignments[f.op[0]]?.let {
183 leaf(it.toString())
184 } ?: f
185 val a = simplify(child, assignments)
186 return if (f.op == "~") negate(a) else simplifyPair(f.op, a, simplify(f.right!!, assignments))
187}
188private fun dual(f: Formula): Formula {
189 val child = f.left ?: return when (f.op) {
190 "0" -> leaf("1")
191 "1" -> leaf("0")
192 else -> f
193 }
194 val a = dual(child)
195 if (f.op == "~") return negation(a)
196 val b = dual(f.right!!)
197 return when (f.op) {
198 "&" -> binary("|", a, b)
199 "|" -> binary("&", a, b)
200 "^" -> binary("<->", a, b)
201 "<->" -> binary("^", a, b)
202 else -> binary("&", negation(a), b)
203 }
204}
205private fun nnf(f: Formula, positive: Boolean = true, memo: MutableMap<Pair<Formula, Boolean>, Formula> = mutableMapOf()): Formula {
206 val key = f to positive
207 memo[key]?.let {
208 return it
209 }
210 val result = when {
211 f.left == null -> if (positive) f else when (f.op) {
212 "0" -> leaf("1")
213 "1" -> leaf("0")
214 else -> negation(f)
215 }
216 f.op == "~" -> nnf(f.left, !positive, memo)
217 f.op == "&" || f.op == "|" -> binary(if ((f.op == "&") == positive) "&" else "|", nnf(f.left, positive, memo), nnf(f.right!!, positive, memo))
218 f.op == "->" -> binary(if (positive) "|" else "&", nnf(f.left, !positive, memo), nnf(f.right!!, positive, memo))
219 else -> {
220 val equal = (f.op == "<->") == positive
221 binary("|", binary("&", nnf(f.left, true, memo), nnf(f.right!!, equal, memo)), binary("&", nnf(f.left, false, memo), nnf(f.right, !equal, memo)))
222 }
223 }
224 memo[key] = result
225 return result
226}
227private fun pattern(row: Int, n: Int) = (0 until n).joinToString("") {

230private fun invert(p: String) = p.map {
231 when (it) {
232 '0' -> '1'
233 '1' -> '0'
234 else -> '-'

238private fun patternKey(p: String) = p.map {

240 '1' -> '0'

246private fun sortedPatterns(ps: Collection<String>) = ps.sortedBy(::patternKey)

248private fun literals(p: String) = p.count {
249 it != '-'

251private fun contains(p: String, row: Int) = p.zip(pattern(row, p.length)).all {

254private fun termText(p: String, vars: List<Char>, clause: Boolean): String {
255 val parts = p.zip(vars).mapNotNull {
256 (c, v) -> when (c) {
257 '1' -> "$v"
258 '0' -> "~$v"
259 else -> null
260 }
261 }
262 return if (parts.isEmpty()) (if (clause) "0" else "1") else parts.joinToString(if (clause) " | " else " & ")
263}
264private fun coverText(ps: List<String>, vars: List<Char>, clause: Boolean): String {
265 if (ps.isEmpty()) return if (clause) "1" else "0"
266 return ps.joinToString(if (clause) " & " else " | ") {
267 p -> val text = termText(p, vars, clause)
268 if (clause && ps.size >= 2 && literals(p) >= 2) "($text)" else text
269 }
270}
271private fun coefficients(values: List<Int>): List<Int> {
272 val c = values.toMutableList()
273 var bit = 1
274 while (bit < c.size) {
275 for (s in c.indices) if (s and bit != 0) c[s] = c[s] xor c[s xor bit]
276 bit *= 2
277 }
278 return c
279}
280private fun assignment(row: Int, vars: List<Char>) = vars.zip(pattern(row, vars.size).toList()).joinToString(" ", "{", "}") {
281 (v, n) -> "$v=$n"
282}
283private fun minimize(vars: List<Char>, on: List<Int>, dc: List<Int>, pos: Boolean): List<String> {
284 var stage = (on + dc).map {
285 pattern(it, vars.size)
286 }.toSet()
287 val primes = mutableListOf<String>()
288 val output = mutableListOf<String>()
289 var k = 0
290 while (true) {
291 val list = stage.toList()
292 val used = mutableSetOf<Int>()
293 val next = mutableSetOf<String>()
294 for (i in list.indices) for (j in i + 1 until list.size) {
295 val different = list[i].indices.filter {
296 list[i][it] != list[j][it]
297 }
298 if (different.size == 1) {
299 val d = different[0]
300 if (list[i][d] != '-' && list[j][d] != '-') {
301 val p = list[i].toCharArray()
302 p[d] = '-'
303 next.add(String(p))
304 used.add(i)
305 used.add(j)
306 }
307 }
308 }
309 val added = list.filterIndexed {
310 i, _ -> i !in used
311 }
312 if (!pos) output.add("stage $k: implicants ${stage.size} primes ${added.size}")
313 primes.addAll(added)
314 if (next.isEmpty()) break
315 stage = next
316 k++
317 }
318 val targets = sortedPatterns(primes.filter {
319 p -> on.any {
320 contains(p, it)
321 }
322 })
323 val essential = sortedSetOf<Int>()
324 for (r in on) {
325 val list = targets.indices.filter {
326 contains(targets[it], r)
327 }
328 if (list.size == 1) essential.add(list[0])
329 }
330 val optional = targets.indices.filter {
331 it !in essential
332 }
333 var best: List<String>? = null
334 var bestLiterals = 0
335
336 fun earlier(a: List<String>, b: List<String>): Boolean {
337 for (i in a.indices) {
338 val cmp = patternKey(a[i]).compareTo(patternKey(b[i]))
339 if (cmp != 0) return cmp < 0
340 }
341 return false
342 }
343 for (mask in 0 until (1 shl optional.size)) {
344 val chosen = essential.toMutableSet()
345 for (i in optional.indices) if (mask and (1 shl i) != 0) chosen.add(optional[i])
346 if (best != null && chosen.size > best!!.size) continue
347 if (!on.all {
348 r -> chosen.any {
349 contains(targets[it], r)
350 }
351 })
352 continue
353 val ps = sortedPatterns(chosen.map {
354 if (pos) invert(targets[it]) else targets[it]
355 })
356 val count = ps.sumOf(::literals)
357 if (best == null || ps.size < best!!.size || ps.size == best!!.size && (count < bestLiterals || count == bestLiterals && earlier(ps, best!!))) {
358 best = ps
359 bestLiterals = count
360 }
361 }
362 val cover = best!!
363 if (!pos) {
364 for ((heading, ps) in listOf("primes" to targets, "essential" to essential.map {
365 targets[it]
366 }))
367 output.add("$heading: " + if (ps.isEmpty()) "none" else ps.joinToString(", ") {
368 termText(it, vars, false)
369 })
370 }
371 output.add("minimal: ${coverText(cover, vars, pos)}")
372 output.add("${if (pos) "clauses" else "terms"} ${cover.size} literals $bestLiterals")
373 return output
374}
375private data class Clause(val positive: Int = 0, val negative: Int = 0)

377private fun tidy(cs: Set<Clause>) = cs.filter {
378 c -> c.positive and c.negative == 0 && cs.none {
379 e -> e != c && e.positive and c.positive == e.positive && e.negative and c.negative == e.negative
380 }
381}.toSet()

383private fun clauses(f: Formula, vars: List<Char>, memo: MutableMap<Formula, Set<Clause>> = mutableMapOf()): Set<Clause> {
384 memo[f]?.let {
385 return it
386 }
387 val result = when {
388 f.op == "1" -> emptySet()
389 f.op == "0" -> setOf(Clause())
390 f.op == "~" || f.left == null -> {
391 val bit = 1 shl vars.indexOf(if (f.op == "~") f.left!!.op[0] else f.op[0])
392 setOf(if (f.op == "~") Clause(negative = bit) else Clause(positive = bit))
393 } else -> {
394 val a = clauses(f.left, vars, memo)
395 val b = clauses(f.right!!, vars, memo)
396 tidy(if (f.op == "&") a + b else a.flatMap {
397 x -> b.map {
398 y -> Clause(x.positive or y.positive, x.negative or y.negative)
399 }
400 }.filter {
401 it.positive and it.negative == 0
402 }.toSet())
403 }
404 }
405 memo[f] = result
406 return result
407}

410 i -> when {
411 c.positive and (1 shl i) != 0 -> "1"
412 c.negative and (1 shl i) != 0 -> "0"
413 else -> "-"
414 }
415 }

433 fun command(line: String): List<String> {

437 if (command !in known) fail("unknown command $command")
438 if (command == "LIST") {

497 if (command == "RESTRICT" || command == "COFACTOR") {

499 if (command == "RESTRICT") {

610 "SAT" -> listOf(models.firstOrNull()?.let {
611 "satisfiable: ${assignment(it, vars)}"
612 } ?: "unsatisfiable")
613 "TAUT" -> listOf(falseRows.firstOrNull()?.let {
614 "not a tautology: ${assignment(it, vars)}"
615 } ?: "tautology")
616 "COUNT" -> listOf("models: ${models.size} of ${values.size}")
617 "EQUIV", "IMPLIES" -> {
618 val r = values.indices.firstOrNull {
619 if (command == "EQUIV") values[it] != other[it] else values[it] == 1 && other[it] == 0
620 }
621 listOf(if (r == null) (if (command == "EQUIV") "equivalent" else "implies") else if (command == "EQUIV") "not equivalent: ${assignment(r, vars)} f=${values[r]} g=${other[r]}" else "does not imply: ${assignment(r, vars)}")
622 }
623 "MINIMIZE", "MINDC", "MINPOS" -> {
624 val on = values.indices.filter {
625 if (command == "MINPOS") values[it] == 0 else values[it] == 1 && (other.isEmpty() || other[it] == 0)
626 }
627 minimize(vars, on, other.indices.filter {
628 other[it] == 1
629 },
630 command == "MINPOS")
631 }
632 "PROPS" -> {
633 val depends = vars.indices.filter {
634 i -> values.indices.any {
635 values[it] != values[it xor (1 shl (vars.size - 1 - i))]
636 }
637 }.map {
638 vars[it]
639 }
640 val c = coefficients(values)
641 val symmetric = values.indices.groupBy {
642 Integer.bitCount(it)
643 }.values.all {
644 rows -> rows.map {
645 values[it]
646 }.distinct().size == 1
647 }
648 val properties = listOf("zero-preserving" to (values[0] == 0), "one-preserving" to (values.last() == 1), "self-dual" to values.indices.all {
649 values[it] != values[values.lastIndex - it]
650 }, "monotone" to values.indices.all {
651 r -> vars.indices.all {
652 i -> r and (1 shl i) != 0 || values[r] <= values[r or (1 shl i)]
653 }
654 }, "linear" to c.indices.all {
655 c[it] == 0 || Integer.bitCount(it) <= 1
656 }, "symmetric" to symmetric)
657 listOf("depends: ${depends.joinToString(" ").ifEmpty { "none" }}") + properties.map {
658 (name, value) -> "$name: ${if (value) "yes" else "no"}"
659 }
660 }
661 "KMAP" -> {
662 if (vars.size !in 2..4) fail("kmap needs 2 to 4 variables")
663 val p = vars.size / 2
664 val q = vars.size - p
665 val output = mutableListOf(vars.take(p).joinToString("") + "\\" + vars.drop(p).joinToString("") + (0 until (1 shl q)).joinToString("") {
666 " " + pattern(it xor (it / 2), q)
667 })
668 for (i in 0 until (1 shl p)) {
669 val r = i xor (i / 2)
670 output.add(pattern(r, p) + " ".repeat(q + 1) + (0 until (1 shl q)).joinToString("") {
671 j -> " ".repeat(q) + values[(r shl q) or (j xor (j / 2))]
672 })
673 }
674 output
675 }
676 "BDD" -> {
677 val nodes = mutableListOf<Triple<Int, Int, Int>>()
678
679 fun build(level: Int, list: List<Int>): Int {
680 if (list.size == 1) return list[0]
681 val lo = build(level + 1, list.take(list.size / 2))
682 val hi = build(level + 1, list.drop(list.size / 2))
683 if (lo == hi) return lo
684 val triple = Triple(level, lo, hi)
685 val old = nodes.indexOf(triple)
686 if (old >= 0) return old + 2
687 nodes.add(triple)
688 return nodes.size + 1
689 }
690 val root = build(0, values)
691 val paths = mutableListOf(0, 1)
692 val output = mutableListOf("nodes: ${nodes.size}")
693 for ((i, node) in nodes.withIndex()) {
694 val (v, lo, hi) = node
695 output.add("${i + 2} = ite(${vars[v]}, $hi, $lo)")
696 paths.add(paths[lo] + paths[hi])
697 }
698 output.add("root: $root")
699 output.add("paths: ${paths[root]}")
700 output
701 } else -> error("unreachable command")
702 }
703 }
704}
