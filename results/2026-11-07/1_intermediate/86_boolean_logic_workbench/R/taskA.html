42parse_formula <- function(text, start, definitions) {
43 tokens <- list()
44 i <- 1L
45 while (i <= nchar(text)) {
46 c <- substr(text, i, i)
47 if (c == " ") {
48 i <- i + 1L
49 next
50 }
51 from <- i
52 if (substr(text, i, i + 2L) == "<->") {
53 i <- i + 3L
54 } else if (substr(text, i, i + 1L) == "->") {
55 i <- i + 2L
56 } else if (grepl("^[a-z]$", c)) {
57 i <- i + 1L
58 while (i <= nchar(text) && grepl("^[a-z0-9]$", substr(text, i, i))) i <- i + 1L
59 } else if (c %in% c(LETTERS, "0", "1", "~", "&", "^", "|", "(", ")")) {
60 i <- i + 1L
61 } else {
62 logic_error(paste0("invalid character '", c, "' at ", start + i - 1L))
63 }
64 tokens[[length(tokens) + 1L]] <- list(text = substr(text, from, i - 1L), position = start + from - 1L)
65 }
66 tokens[[length(tokens) + 1L]] <- list(text = "", position = start + nchar(text))
67 position <- 1L
68 peek <- function() tokens[[position]]$text
69 advance <- function() {
70 result <- peek()
71 position <<- position + 1L
72 result
73 }
74 unexpected <- function() {
75 token <- tokens[[position]]
76 logic_error(if (token$text == "") paste0("unexpected end of formula at ", token$position) else paste0("unexpected '", token$text, "' at ", token$position))
77 }
78 unary <- function() {
79 if (peek() == "~") {
80 advance()
81 return(bool_not(unary()))
82 }
83 if (peek() == "(") {
84 advance()
85 result <- expression(1L)
86 if (peek() != ")") unexpected()
87 advance()
88 return(result)
89 }
90 if (grepl("^[A-Z]$", peek())) return(bool_variable(advance()))
91 if (peek() %in% c("0", "1")) return(bool_constant(as.integer(advance())))
92 if (grepl("^[a-z][a-z0-9]*$", peek())) {
93 token <- tokens[[position]]
94 if (is.null(definitions[[token$text]])) logic_error(paste0("undefined name '", token$text, "' at ", token$position))
95 advance()
96 return(definitions[[token$text]])
97 }
98 unexpected()
99 }
100 expression <- function(level) {
101 if (level == 6L) return(unary())
102 op <- c("<->", "->", "|", "^", "&")[[level]]
103 result <- expression(level + 1L)
104 while (peek() == op) {
105 advance()
106 result <- bool_binary(op, result, expression(if (op == "->") level else level + 1L))
107 }
108 result
109 }
110 result <- expression(1L)
111 if (peek() != "") unexpected()
112 result

114nnf <- function(node, positive = TRUE) {
115 memo <- list()
116 serial <- 0L
117 visit <- function(current, sign) {
118 key <- paste0(if (sign) "p" else "n", paste(as.integer(serialize(current, NULL)), collapse = "."))
119 if (!is.null(memo[[key]])) return(memo[[key]])
120 if (current$kind == "variable") {
121 result <- if (sign) current else bool_not(current)
122 } else if (current$kind == "constant") {
123 result <- bool_constant(if (sign) current$value else 1L - current$value)
124 } else if (current$kind == "not") {
125 result <- visit(current$child, !sign)
126 } else if (current$kind %in% c("&", "|")) {
127 op <- if (sign) current$kind else if (current$kind == "&") "|" else "&"
128 result <- bool_binary(op, visit(current$left, sign), visit(current$right, sign))
129 } else if (current$kind == "->") {
130 result <- bool_binary(if (sign) "|" else "&", visit(current$left, !sign), visit(current$right, sign))
131 } else {
132 same <- if (current$kind == "<->") sign else !sign
133 result <- bool_binary("|", bool_binary("&", visit(current$left, TRUE), visit(current$right, same)),
134 bool_binary("&", visit(current$left, FALSE), visit(current$right, !same)))
135 }
136 serial <<- serial + 1L
137 attr(result, "memo_id") <- serial
138 memo[[key]] <<- result
139 result
140 }
141 visit(node, positive)

144dual <- function(node) {
145 if (node$kind == "variable") return(node)
146 if (node$kind == "constant") return(bool_constant(1L - node$value))
147 if (node$kind == "not") return(bool_not(dual(node$child)))
148 if (node$kind == "->") return(bool_binary("&", bool_not(dual(node$left)), dual(node$right)))
149 op <- c("&" = "|", "|" = "&", "^" = "<->", "<->" = "^")[[node$kind]]
150 bool_binary(op, dual(node$left), dual(node$right))
151}
152reduce_negation <- function(node) {

156simplify_binary <- function(op, a, b) {
157 zero <- bool_constant(0L)
158 one <- bool_constant(1L)
159 equal <- identical(a, b)
160 complementary <- identical(a, bool_not(b)) || identical(b, bool_not(a))
161 if (op == "&") {
162 if (identical(a, zero) || identical(b, zero)) return(zero)
163 if (identical(a, one)) return(b)
164 if (identical(b, one) || equal) return(a)
165 if (complementary) return(zero)
166 } else if (op == "|") {
167 if (identical(a, one) || identical(b, one)) return(one)
168 if (identical(a, zero)) return(b)
169 if (identical(b, zero) || equal) return(a)
170 if (complementary) return(one)
171 } else if (op == "^") {
172 if (identical(a, zero)) return(b)
173 if (identical(b, zero)) return(a)
174 if (identical(a, one)) return(reduce_negation(b))
175 if (identical(b, one)) return(reduce_negation(a))
176 if (equal) return(zero)
177 if (complementary) return(one)
178 } else if (op == "->") {
179 if (identical(a, zero) || identical(b, one)) return(one)
180 if (identical(a, one)) return(b)
181 if (identical(b, zero)) return(reduce_negation(a))
182 if (equal) return(one)
183 if (complementary) return(b)
184 } else {
185 if (identical(a, one)) return(b)
186 if (identical(b, one)) return(a)
187 if (identical(a, zero)) return(reduce_negation(b))
188 if (identical(b, zero)) return(reduce_negation(a))
189 if (equal) return(one)
190 if (complementary) return(zero)
191 }
192 bool_binary(op, a, b)
193}
194simplify <- function(node, assignments = list()) {
195 if (node$kind == "variable") return(if (is.null(assignments[[node$value]])) node else bool_constant(assignments[[node$value]]))
196 if (node$kind == "constant") return(node)
197 if (node$kind == "not") return(reduce_negation(simplify(node$child, assignments)))
198 simplify_binary(node$kind, simplify(node$left, assignments), simplify(node$right, assignments))
199}
200evaluate <- function(node, assignments) {
201 if (node$kind == "variable") return(assignments[[node$value]])
202 if (node$kind == "constant") return(node$value)
203 if (node$kind == "not") return(1L - evaluate(node$child, assignments))
204 a <- evaluate(node$left, assignments)
205 b <- evaluate(node$right, assignments)
206 switch(node$kind, "&" = bitwAnd(a, b), "|" = bitwOr(a, b), "^" = bitwXor(a, b), "->" = if (a == 1L && b == 0L) 0L else 1L, "<->" = if (a == b) 1L else 0L)
207}

251minimize <- function(on, dc, names, pos = FALSE) {
252 rows <- setNames(vapply(on, pattern, character(1), size = length(names)), as.character(on))
253 stage <- unique(vapply(unique(c(on, dc)), pattern, character(1), size = length(names)))
254 primes <- character()
255 lines <- character()
256 number <- 0L
257 repeat {
258 used <- next_stage <- character()
259 for (text in stage) {
260 for (i in seq_len(nchar(text))) {
261 if (substr(text, i, i) != "0") next
262 partner <- text
263 substr(partner, i, i) <- "1"
264 if (!(partner %in% stage)) next
265 combined <- text
266 substr(combined, i, i) <- "-"
267 used <- c(used, text, partner)
268 next_stage <- c(next_stage, combined)
269 }
270 }
271 fresh <- setdiff(stage, used)
272 primes <- union(primes, fresh)
273 if (!pos) lines <- c(lines, paste0("stage ", number, ": implicants ", length(stage), " primes ", length(fresh)))
274 next_stage <- unique(next_stage)
275 if (length(next_stage) == 0L) break
276 stage <- next_stage
277 number <- number + 1L
278 }
279 targets <- sort_patterns(primes[vapply(primes, function(text) any(vapply(rows, function(row) contains_row(text, row), logical(1))), logical(1))])
280 coverage <- lapply(targets, function(text) on[vapply(on, function(row) contains_row(text, rows[[as.character(row)]]), logical(1))])
281 essential <- integer()
282 for (row in on) {
283 candidates <- which(vapply(coverage, function(covered) row %in% covered, logical(1)))
284 if (length(candidates) == 1L) essential <- union(essential, candidates)
285 }
286 optional <- setdiff(seq_along(targets), essential)
287 best <- best_key <- NULL
288 for (mask in seq_len(2 ^ length(optional)) - 1L) {
289 chosen <- c(essential, optional[vapply(seq_along(optional) - 1L, function(i) bitwAnd(mask, bitwShiftL(1L, i)) != 0L, logical(1))])
290 covered <- unique(unlist(coverage[chosen], use.names = FALSE))
291 if (!all(on %in% covered)) next
292 texts <- if (pos) complement_pattern(targets[chosen]) else targets[chosen]
293 texts <- sort_patterns(texts)
294 key <- list(count = length(texts), literals = sum(literal_count(texts)), patterns = pattern_key(texts))
295 if (is.null(best_key) || cover_key_compare(key, best_key) < 0L) {
296 best_key <- key
297 best <- texts
298 }
299 }
300 if (pos) return(c(paste0("minimal: ", cover_text(best, names, FALSE)), paste0("clauses ", length(best), " literals ", best_key$literals)))
301 terms <- vapply(targets, pattern_part, character(1), names = names)
302 essentials <- vapply(targets[sort(essential)], pattern_part, character(1), names = names)
303 c(lines, paste0("primes: ", if (length(terms) == 0L) "none" else paste(terms, collapse = ", ")),
304 paste0("essential: ", if (length(essentials) == 0L) "none" else paste(essentials, collapse = ", ")),
305 paste0("minimal: ", cover_text(best, names)), paste0("terms ", length(best), " literals ", best_key$literals))
306}

486 names <- sort(unique(c(variable_names(formula), if (is.null(other)) character() else variable_names(other))), method = "radix")
487 if (command %in% table_commands && length(names) >= 9L) logic_error(paste0("too many variables: ", length(names)))
488 if (command == "PRINT") {
489 measure <- measures(formula)
490 return(c(paste0("formula: ", notation(formula)), paste0("variables: ", if (length(names) == 0L) "none" else paste(names, collapse = " ")), paste0("size ", measure[[1]], " depth ", measure[[2]])))
491 }
492 if (command == "NNF") return(paste0("nnf: ", notation(nnf(formula))))
493 if (command == "DUAL") return(paste0("dual: ", notation(dual(formula))))
494 if (command == "SIMPLIFY") return(paste0("simplified: ", notation(simplify(formula))))
495 if (command %in% c("CLAUSES", "DP")) {
496 set <- clauses(nnf(formula), names)
497 if (command == "CLAUSES") return(paste0("clauses: ", cover_text(set, names, FALSE)))
498 lines <- paste0("start: clauses ", length(set))
499 empty <- strrep("-", length(names))
500 for (i in seq_along(names)) {
501 if (length(set) == 0L || empty %in% set) break
502 positive <- set[substr(set, i, i) == "1"]
503 negative <- set[substr(set, i, i) == "0"]
504 resolvents <- character()
505 for (a in positive) {
506 for (b in negative) {
507 substr(a, i, i) <- substr(b, i, i) <- "-"
508 merged <- merge_clause(a, b)
509 if (!is.null(merged)) resolvents <- c(resolvents, merged)
510 }
511 }
512 resolvents <- unique(resolvents)
513 set <- tidy(c(set[substr(set, i, i) == "-"], resolvents))
514 lines <- c(lines, paste0("eliminate ", names[[i]], ": resolvents ", length(resolvents), " clauses ", length(set)))
515 }
516 return(c(lines, if (empty %in% set) "result: unsatisfiable" else "result: satisfiable"))
517 }
518 rows <- seq_len(2 ^ length(names)) - 1L
519 values <- vapply(rows, function(row) evaluate(formula, assignments(names, row)), integer(1))
520 other_values <- if (is.null(other)) NULL else vapply(rows, function(row) evaluate(other, assignments(names, row)), integer(1))
521 on <- rows[values == 1L]
522 off <- rows[values == 0L]
523 if (command == "TABLE") return(c(paste(c(names, "|", "F"), collapse = " "), vapply(rows, function(row) paste(c(strsplit(pattern(row, length(names)), "", fixed = TRUE)[[1]], "|", values[[row + 1L]]), collapse = " "), character(1))))
524 if (command %in% c("MINTERMS", "MAXTERMS")) {
525 selected <- if (command == "MINTERMS") on else off
526 return(paste0(tolower(command), ": ", if (length(selected) == 0L) "none" else paste(selected, collapse = " ")))
527 }
528 if (command %in% c("DNF", "CNF")) {
529 texts <- vapply(if (command == "DNF") on else off, pattern, character(1), size = length(names))
530 if (command == "CNF") texts <- complement_pattern(texts)
531 return(paste0(tolower(command), ": ", cover_text(texts, names, command == "DNF")))
532 }
533 if (command == "ANF") {
534 coeff <- coefficients(values, length(names))
535 texts <- chartr("0", "-", vapply(rows[coeff == 1L], pattern, character(1), size = length(names)))
536 texts <- texts[order(literal_count(texts), pattern_key(texts), method = "radix")]
537 return(paste0("anf: ", if (length(texts) == 0L) "0" else paste(vapply(texts, pattern_part, character(1), names = names), collapse = " ^ ")))
538 }
539 if (command == "SAT") return(if (length(on) == 0L) "unsatisfiable" else paste0("satisfiable: ", assignment_text(names, on[[1]])))
540 if (command == "TAUT") return(if (length(off) == 0L) "tautology" else paste0("not a tautology: ", assignment_text(names, off[[1]])))
541 if (command == "COUNT") return(paste0("models: ", length(on), " of ", length(values)))
542 if (command == "EQUIV") {
543 differing <- which(values != other_values)
544 if (length(differing) == 0L) return("equivalent")
545 i <- differing[[1]]
546 return(paste0("not equivalent: ", assignment_text(names, i - 1L), " f=", values[[i]], " g=", other_values[[i]]))
547 }
548 if (command == "IMPLIES") {
549 failing <- which(values == 1L & other_values == 0L)
550 if (length(failing) == 0L) return("implies")
551 return(paste0("does not imply: ", assignment_text(names, failing[[1]] - 1L)))
552 }
553 if (command == "PROPS") {
554 depends <- character()
555 for (i in seq_along(names)) {
556 bit <- bitwShiftL(1L, length(names) - i)
557 if (any(values != values[bitwXor(rows, bit) + 1L])) depends <- c(depends, names[[i]])
558 }
559 monotone <- self_dual <- symmetric <- TRUE
560 weights <- list()
561 for (row in rows) {
562 value <- values[[row + 1L]]
563 if (value == values[[length(values) - row]]) self_dual <- FALSE
564 for (i in seq_along(names) - 1L) {
565 bit <- bitwShiftL(1L, i)
566 if (bitwAnd(row, bit) == 0L && value > values[[bitwOr(row, bit) + 1L]]) monotone <- FALSE
567 }
568 weight <- as.character(sum(as.integer(intToBits(row))))
569 if (!is.null(weights[[weight]]) && weights[[weight]] != value) symmetric <- FALSE
570 weights[[weight]] <- value
571 }
572 coeff <- coefficients(values, length(names))
573 linear <- all(vapply(rows[coeff == 1L], function(row) sum(as.integer(intToBits(row))) <= 1L, logical(1)))
574 properties <- c("zero-preserving" = values[[1]] == 0L, "one-preserving" = tail(values, 1L) == 1L, "self-dual" = self_dual, monotone = monotone, linear = linear, symmetric = symmetric)
575 return(c(paste0("depends: ", if (length(depends) == 0L) "none" else paste(depends, collapse = " ")), paste0(names(properties), ": ", ifelse(properties, "yes", "no"))))
576 }
577 if (command == "KMAP") {
578 if (length(names) < 2L || length(names) > 4L) logic_error("kmap needs 2 to 4 variables")
579 p <- length(names) %/% 2L
580 q <- length(names) - p
581 row_codes <- vapply(seq_len(2 ^ p) - 1L, function(i) pattern(bitwXor(i, i %/% 2L), p), character(1))
582 columns <- vapply(seq_len(2 ^ q) - 1L, function(i) pattern(bitwXor(i, i %/% 2L), q), character(1))
583 lines <- paste0(paste(head(names, p), collapse = ""), "\\", paste(tail(names, q), collapse = ""), " ", paste(columns, collapse = " "))
584 for (row in row_codes) {
585 text <- paste0(row, strrep(" ", length(names) + 1L - p))
586 for (column in columns) {
587 bits <- strsplit(paste0(row, column), "", fixed = TRUE)[[1]]
588 index <- sum(as.integer(bits) * 2 ^ rev(seq_along(bits) - 1L))
589 text <- paste0(text, strrep(" ", q), values[[index + 1L]])
590 }
591 lines <- c(lines, text)
592 }
593 return(lines)
594 }
595 if (command == "BDD") return(bdd(values, names))
596 if (command == "MINIMIZE") return(minimize(on, integer(), names))
597 if (command == "MINDC") {

601 minimize(off, integer(), names, TRUE)
602 }
