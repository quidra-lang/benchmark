40 {
41 Shape shape = new(op, left?.Id ?? -1, right?.Id ?? -1);
42 if (nodes.TryGetValue(shape, out Node? old)) return old;

44 nodes[shape] = n;
45 return n;

50 static int Precedence(string op) => op switch
51 {
52 "<->" => 1,
53 "->" => 2,
54 "|" => 3,
55 "^" => 4,
56 "&" => 5,
57 _ => 6
58 };

60 string Print(Node n)
61 {
62 if (printed.TryGetValue(n.Id, out string? old)) return old;
63 string result;
64 if (n.Left == null) result = n.Op;
65 else if (!n.Binary) result = "~" + (n.Left.Binary ? "(" + Print(n.Left) + ")" : Print(n.Left));
66 else
67 {
68 string a = Print(n.Left), b = Print(n.Right!);
69 int p = Precedence(n.Op);
70 if (n.Left.Binary && (Precedence(n.Left.Op) < p || n.Op == "->" && n.Left.Op == "->")) a = "(" + a + ")";
71 if (n.Right!.Binary && (Precedence(n.Right.Op) < p || n.Right.Op == n.Op && n.Op != "->")) b = "(" + b + ")";
72 result = a + " " + n.Op + " " + b;
73 }
74 printed[n.Id] = result;
75 return result;
76 }

86 class Parser
87 {
88 readonly Program workbench;
89 readonly List<Token> tokens = new();
90 readonly int end;
91 int at;
92 public Parser(Program workbench, string text, int start)
93 {
94 this.workbench = workbench;
95 end = start + text.Length;
96 for (int i = 0; i < text.Length;)
97 {
98 char c = text[i];
99 if (c == ' ')
100 {
101 i++;
102 continue;
103 }
104 int first = i;
105 if (text.AsSpan(i).StartsWith("<->")) i += 3;
106 else if (text.AsSpan(i).StartsWith("->")) i += 2;
107 else if (Lower(c))
108 {
109 i++;
110 while (i < text.Length && (Lower(text[i]) || Digit(text[i]))) i++;
111 }
112 else if (Upper(c) || "01~&^|()".Contains(c)) i++;
113 else
114 {
115 Fail($"invalid character '{c}' at {start + i}");
116 return;
117 }
118 tokens.Add(new(text[first..i], start + first));
119 }
120 }
121 string Peek() => at == tokens.Count ? "" : tokens[at].Text;
122 void Unexpected()
123 {
124 if (at == tokens.Count) Fail("unexpected end of formula at " + end);
125 Token t = tokens[at];
126 Fail($"unexpected '{t.Text}' at {t.Position}");
127 }
128 Node Unary()
129 {
130 string t = Peek();
131 if (t == "~")
132 {
133 at++;
134 return workbench.Not(Unary());
135 }
136 if (t == "(")
137 {
138 at++;
139 Node n = Expression(1);
140 if (Peek() != ")") Unexpected();
141 at++;
142 return n;
143 }
144 if (t.Length > 0 && (Upper(t[0]) || t is "0" or "1"))
145 {
146 at++;
147 return workbench.Make(t);
148 }
149 if (t.Length > 0 && Lower(t[0]))
150 {
151 Token token = tokens[at++];
152 if (!workbench.definitions.TryGetValue(t, out Node? n)) Fail($"undefined name '{t}' at {token.Position}");
153 return n!;
154 }
155 Unexpected();
156 return null!;
157 }
158 Node Expression(int level)
159 {
160 if (level == 6) return Unary();
161 Node left = Expression(level + 1);
162 string op = level switch
163 {
164 1 => "<->",
165 2 => "->",
166 3 => "|",
167 4 => "^",
168 _ => "&"
169 };
170 while (Peek() == op)
171 {
172 at++;
173 left = workbench.Make(op, left, Expression(level == 2 ? level : level + 1));
174 }
175 return left;
176 }
177 public Node Parse()
178 {
179 Node n = Expression(1);
180 if (at != tokens.Count) Unexpected();
181 return n;
182 }
183 }

187 Node Nnf(Node n, bool positive)
188 {
189 if (normalForms.TryGetValue((n.Id, positive), out Node? old)) return old;
190 Node result;
191 if (n.Left == null) result = n.Op is "0" or "1" ? Make(positive ? n.Op : n.Op == "0" ? "1" : "0") : positive ? n : Not(n);
192 else if (n.Op == "~") result = Nnf(n.Left, !positive);
193 else if (n.Op is "&" or "|") result = Make(positive ? n.Op : n.Op == "&" ? "|" : "&", Nnf(n.Left, positive), Nnf(n.Right!, positive));
194 else if (n.Op == "->") result = Make(positive ? "|" : "&", Nnf(n.Left, !positive), Nnf(n.Right!, positive));
195 else
196 {
197 bool equal = (n.Op == "<->") == positive;
198 result = Make("|", Make("&", Nnf(n.Left, true), Nnf(n.Right!, equal)), Make("&", Nnf(n.Left, false), Nnf(n.Right!, !equal)));
199 }
200 normalForms[(n.Id, positive)] = result;
201 return result;
202 }

204 Node Dual(Node n, Dictionary<int, Node> memo)
205 {
206 if (memo.TryGetValue(n.Id, out Node? old)) return old;
207 Node result;
208 if (n.Left == null) result = n.Op == "0" ? Make("1") : n.Op == "1" ? Make("0") : n;
209 else if (n.Op == "~") result = Not(Dual(n.Left, memo));
210 else
211 {
212 Node a = Dual(n.Left, memo), b = Dual(n.Right!, memo);
213 result = n.Op switch
214 {
215 "&" => Make("|", a, b),
216 "|" => Make("&", a, b),
217 "^" => Make("<->", a, b),
218 "<->" => Make("^", a, b),
219 _ => Make("&", Not(a), b)
220 };
221 }
222 memo[n.Id] = result;
223 return result;
224 }

226 Node ReduceNot(Node n) => n.Op == "0" ? Make("1") : n.Op == "1" ? Make("0") : n.Op == "~" ? n.Left! : Not(n);

228 static bool Complement(Node a, Node b) => a.Op == "~" && a.Left == b || b.Op == "~" && b.Left == a;
229
230 Node SimplifyBinary(string op, Node a, Node b)
231 {
232 bool a0 = a.Op == "0", a1 = a.Op == "1", b0 = b.Op == "0", b1 = b.Op == "1";
233 switch (op)
234 {
235 case "&":
236 if (a0 || b0) return Make("0");
237 if (a1) return b;
238 if (b1 || a == b) return a;
239 if (Complement(a, b)) return Make("0");
240 break;
241 case "|":
242 if (a1 || b1) return Make("1");
243 if (a0) return b;
244 if (b0 || a == b) return a;
245 if (Complement(a, b)) return Make("1");
246 break;
247 case "^":
248 if (a0) return b;
249 if (b0) return a;
250 if (a1) return ReduceNot(b);
251 if (b1) return ReduceNot(a);
252 if (a == b) return Make("0");
253 if (Complement(a, b)) return Make("1");
254 break;
255 case "->":
256 if (a0 || b1) return Make("1");
257 if (a1) return b;
258 if (b0) return ReduceNot(a);
259 if (a == b) return Make("1");
260 if (Complement(a, b)) return b;
261 break;
262 case "<->":
263 if (a1) return b;
264 if (b1) return a;
265 if (a0) return ReduceNot(b);
266 if (b0) return ReduceNot(a);
267 if (a == b) return Make("1");
268 if (Complement(a, b)) return Make("0");
269 break;
270 }
271 return Make(op, a, b);
272 }

274 Node Simplify(Node n, Dictionary<char, int> substitutions, Dictionary<int, Node> memo)
275 {
276 if (memo.TryGetValue(n.Id, out Node? old)) return old;
277 Node result;
278 if (n.Left == null) result = substitutions.TryGetValue(n.Op[0], out int value) ? Make(value.ToString()) : n;
279 else if (n.Op == "~") result = ReduceNot(Simplify(n.Left, substitutions, memo));
280 else result = SimplifyBinary(n.Op, Simplify(n.Left, substitutions, memo), Simplify(n.Right!, substitutions, memo));
281 memo[n.Id] = result;
282 return result;
283 }

285 static int Evaluate(Node n, int trueVariables, Dictionary<int, int> memo)
286 {
287 if (memo.TryGetValue(n.Id, out int old)) return old;
288 int result;
289 if (n.Left == null) result = Upper(n.Op[0]) ? (trueVariables >> (n.Op[0] - 'A')) & 1 : n.Op[0] - '0';
290 else if (n.Op == "~") result = 1 - Evaluate(n.Left, trueVariables, memo);
291 else
292 {
293 int a = Evaluate(n.Left, trueVariables, memo), b = Evaluate(n.Right!, trueVariables, memo);
294 result = n.Op switch
295 {
296 "&" => a & b,
297 "|" => a | b,
298 "^" => a ^ b,
299 "->" => (1 - a) | b,
300 _ => 1 - (a ^ b)
301 };
302 }
303 memo[n.Id] = result;
304 return result;
305 }

307 static int[] Truth(Node n, string vars)
308 {
309 int[] table = new int[1 << vars.Length];
310 for (int r = 0; r < table.Length; r++)
311 {
312 int mask = 0;
313 for (int i = 0; i < vars.Length; i++) if ((r & 1 << (vars.Length - 1 - i)) != 0) mask |= 1 << (vars[i] - 'A');
314 table[r] = Evaluate(n, mask, new());
315 }
316 return table;
317 }

351 int[] output = (int[])table.Clone();

358 static void Implicants(int at, int n, int mask, int values, string prefix, List<Implicant> output, HashSet<int> allowed)
359 {
360 if (at == n)
361 {
362 HashSet<int> coverage = new();
363 for (int row = 0; row < 1 << n; row++) if (((row ^ values) & mask) == 0)
364 {
365 if (!allowed.Contains(row)) return;
366 coverage.Add(row);
367 }
368 output.Add(new(prefix, mask, values, coverage));
369 return;
370 }
371 int bit = 1 << (n - 1 - at);
372 Implicants(at + 1, n, mask | bit, values | bit, prefix + "1", output, allowed);
373 Implicants(at + 1, n, mask | bit, values, prefix + "0", output, allowed);
374 Implicants(at + 1, n, mask, values, prefix + "-", output, allowed);
375 }

384 static int CoverLiterals(IEnumerable<Implicant> cover) => cover.Sum(p => Bits(p.Mask));

386 static bool Better(List<Implicant> candidate, List<Implicant>? best, bool pos)
387 {
388 if (best == null || candidate.Count != best.Count) return best == null || candidate.Count < best.Count;
389 int a = CoverLiterals(candidate), b = CoverLiterals(best);
390 if (a != b) return a < b;
391 List<string> x = CoverPatterns(candidate, pos), y = CoverPatterns(best, pos);
392 for (int i = 0; i < x.Count; i++)
393 {
394 int c = PatternCompare(x[i], y[i]);
395 if (c != 0) return c < 0;
396 }
397 return false;
398 }

400 static List<string> Minimize(string vars, int[] f, int[]? dc, bool pos)
401 {
402 int n = vars.Length;
403 HashSet<int> on = new(), allowed = new();
404 for (int r = 0; r < f.Length; r++)
405 {
406 if ((pos ? 1 - f[r] : f[r]) == 1 && (dc == null || dc[r] == 0)) on.Add(r);
407 if (on.Contains(r) || dc != null && dc[r] == 1) allowed.Add(r);
408 }
409 List<Implicant> all = new();
410 Implicants(0, n, 0, 0, "", all, allowed);
411 HashSet<int> keys = new(all.Select(p => (p.Mask << 8) | p.Values));
412 List<Implicant> primes = new();
413 int[] stages = new int[n + 1], primeStages = new int[n + 1];
414 foreach (Implicant p in all)
415 {
416 int stage = n - Bits(p.Mask);
417 stages[stage]++;
418 bool prime = true;
419 for (int bit = 1; bit < 1 << n; bit <<= 1) if ((p.Mask & bit) != 0 && keys.Contains(((p.Mask ^ bit) << 8) | (p.Values & ~bit)))
420 {
421 prime = false;
422 break;
423 }
424 if (prime)
425 {
426 primeStages[stage]++;
427 if (p.Coverage.Overlaps(on)) primes.Add(p);
428 }
429 }
430 primes.Sort((a, b) => PatternCompare(a.Pattern, b.Pattern));
431 HashSet<Implicant> essentialSet = new();
432 foreach (int r in on)
433 {
434 var covering = primes.Where(p => p.Coverage.Contains(r)).ToList();
435 if (covering.Count == 1) essentialSet.Add(covering[0]);
436 }
437 var essential = primes.Where(essentialSet.Contains).ToList();
438 var optional = primes.Where(p => !essentialSet.Contains(p)).ToList();
439 List<Implicant>? best = null;
440 for (int mask = 0; mask < 1 << optional.Count; mask++)
441 {
442 List<Implicant> candidate = new(essential);
443 HashSet<int> covered = new();
444 for (int i = 0; i < optional.Count; i++) if ((mask & 1 << i) != 0) candidate.Add(optional[i]);
445 if (best != null && candidate.Count > best.Count) continue;
446 foreach (Implicant p in candidate) covered.UnionWith(p.Coverage);
447 if (on.IsSubsetOf(covered) && Better(candidate, best, pos)) best = candidate;
448 }
449 List<string> output = new();
450 if (!pos)
451 {
452 for (int stage = 0; stage <= n; stage++)
453 {
454 output.Add($"stage {stage}: implicants {stages[stage]} primes {primeStages[stage]}");
455 if (stage == n || stages[stage + 1] == 0) break;
456 }
457 output.Add("primes: " + (primes.Count == 0 ? "none" : string.Join(", ", CoverPatterns(primes, false).Select(p => Term(p, vars, false)))));
458 output.Add("essential: " + (essential.Count == 0 ? "none" : string.Join(", ", CoverPatterns(essential, false).Select(p => Term(p, vars, false)))));
459 }
460 output.Add("minimal: " + Form(CoverPatterns(best!, pos), vars, pos));
461 output.Add((pos ? "clauses " : "terms ") + best!.Count + " literals " + CoverLiterals(best));
462 return output;
463 }
464 readonly record struct Clause(int Positive, int Negative)
465 {
466 public bool Subset(Clause other) => (Positive & other.Positive) == Positive && (Negative & other.Negative) == Negative;
467 public Clause Union(Clause other) => new(Positive | other.Positive, Negative | other.Negative);
468 public bool Tautology => (Positive & Negative) != 0;
469 public string Pattern(string vars)
470 {
471 int positive = Positive, negative = Negative;
472 return string.Concat(vars.Select(c => (positive & 1 << (c - 'A')) != 0 ? '1' : (negative & 1 << (c - 'A')) != 0 ? '0' : '-'));
473 }
474 }

476 static HashSet<Clause> Tidy(HashSet<Clause> input)
477 {
478 HashSet<Clause> output = new();
479 foreach (Clause c in input.Where(c => !c.Tautology).OrderBy(c => Bits(c.Positive | c.Negative))) if (!output.Any(e => e.Subset(c))) output.Add(c);
480 return output;
481 }

483 HashSet<Clause> Clauses(Node n, Dictionary<int, HashSet<Clause>> memo)
484 {
485 if (memo.TryGetValue(n.Id, out var old)) return old;
486 HashSet<Clause> result = new();
487 if (n.Op == "0") result.Add(new(0, 0));
488 else if (n.Op == "1")
489 {
490 }
491 else if (n.Left == null) result.Add(new(1 << (n.Op[0] - 'A'), 0));
492 else if (n.Op == "~") result.Add(new(0, 1 << (n.Left.Op[0] - 'A')));
493 else
494 {
495 var a = Clauses(n.Left, memo);
496 var b = Clauses(n.Right!, memo);
497 if (n.Op == "&")
498 {
499 result.UnionWith(a);
500 result.UnionWith(b);
501 }
502 else foreach (Clause x in a) foreach (Clause y in b)
503 {
504 Clause c = x.Union(y);
505 if (!c.Tautology) result.Add(c);
506 }
507 result = Tidy(result);
508 }
509 memo[n.Id] = result;
510 return result;
511 }

543 List<string> Command(string line)
544 {
545 int space = line.IndexOf(' ');
546 string command = space < 0 ? line : line[..space], arg = space < 0 ? "" : line[(space + 1)..];
547 HashSet<string> single = new(StringComparer.Ordinal) { "PRINT", "TABLE", "MINTERMS", "MAXTERMS", "DNF", "CNF", "NNF", "DUAL", "ANF", "SIMPLIFY", "MINIMIZE", "MINPOS", "KMAP", "SAT", "TAUT", "COUNT", "PROPS", "BDD", "CLAUSES", "DP" };
548 HashSet<string> split = new(StringComparer.Ordinal) { "RESTRICT", "COFACTOR", "MINDC", "EQUIV", "IMPLIES" };
549 if (!single.Contains(command) && !split.Contains(command) && !new[] { "DEF", "DEFSUM", "DEL", "LIST" }.Contains(command)) Fail("unknown command " + command);
550 if (command == "LIST" && space >= 0) Fail("unexpected argument");
551 if (command != "LIST" && space < 0) Fail("missing argument");
552 int semicolon = arg.IndexOf(';');
553 if (split.Contains(command) && semicolon < 0) Fail("missing ';'");
554 List<string> output = new();
555 if (command == "LIST")
556 {
557 output.Add("names: " + definitions.Count);
558 foreach (var e in definitions) output.Add(e.Key + " = " + Print(e.Value));
559 return output;
560 }
561 if (command == "DEL")
562 {
563 if (!ValidName(arg)) Fail("invalid name '" + arg + "'");
564 if (!definitions.ContainsKey(arg)) Fail("undefined name '" + arg + "'");
565 definitions.Remove(arg);
566 return new() { "deleted " + arg };
567 }
568 if (command == "DEF")
569 {
570 int at = arg.IndexOf(' ');
571 if (at < 0) Fail("missing argument");
572 string name = arg[..at];
573 if (!ValidName(name)) Fail("invalid name '" + name + "'");
574 Node formula = Parse(arg[(at + 1)..], name.Length + 6);
575 Define(name, formula, output);
576 return output;
577 }
578 if (command == "DEFSUM")
579 {
580 if (semicolon < 0) Fail("missing ';'");
581 string[] front = Words(arg[..semicolon]), back = Words(arg[(semicolon + 1)..]);
582 if (front.Length == 0) Fail("missing argument");
583 if (!ValidName(front[0])) Fail("invalid name '" + front[0] + "'");
584 if (front.Length == 1) Fail("bad variable list");
585 for (int i = 1; i < front.Length; i++) if (front[i].Length != 1 || !Upper(front[i][0]) || i > 1 && StringComparer.Ordinal.Compare(front[i - 1], front[i]) >= 0) Fail("bad variable list");
586 int n = front.Length - 1;
587 if (n >= 9) Fail("too many variables: " + n);
588 int previous = -1;
589 Node? result = null;
590 foreach (string word in back)
591 {
592 if (word.Length == 0 || word.Length > 1 && word[0] == '0' || !word.All(Digit)) Fail("bad minterm '" + word + "'");
593 int row = int.Parse(word);
594 if (row >= 1 << n || row <= previous) Fail("bad minterm '" + word + "'");
595 previous = row;
596 Node? term = null;
597 for (int i = 0; i < n; i++)
598 {
599 Node literal = Make(front[i + 1]);
600 if ((row & 1 << (n - i - 1)) == 0) literal = Not(literal);
601 term = term == null ? literal : Make("&", term, literal);
602 }
603 result = result == null ? term : Make("|", result, term);
604 }
605 Define(front[0], result ?? Make("0"), output);
606 return output;
607 }
608 int start = command.Length + 2;
609 string first = split.Contains(command) ? arg[..semicolon] : arg, second = split.Contains(command) ? arg[(semicolon + 1)..] : "";
610 Node f = Parse(first, start);
611 Node? g = null;
612 if (command is "MINDC" or "EQUIV" or "IMPLIES") g = Parse(second, start + semicolon + 1);
613 if (command == "RESTRICT")
614 {
615 string[] assignments = Words(second);
616 if (assignments.Length == 0) Fail("missing assignment");
617 Dictionary<char, int> values = new();
618 foreach (string a in assignments)
619 {
620 if (a.Length != 3 || !Upper(a[0]) || a[1] != '=' || !"01".Contains(a[2])) Fail("bad assignment '" + a + "'");
621 if (!values.TryAdd(a[0], a[2] - '0')) Fail("duplicate variable " + a[0]);
622 }
623 return new() { "restricted: " + Print(Simplify(f, values, new())) };
624 }
625 if (command == "COFACTOR")
626 {
627 string[] x = Words(second);
628 if (x.Length != 1 || x[0].Length != 1 || !Upper(x[0][0])) Fail("bad variable");
629 char c = x[0][0];
630 Node f1 = Simplify(f, new() { [c] = 1 }, new()), f0 = Simplify(f, new() { [c] = 0 }, new());
631 return new() { $"cofactor {c}=1: {Print(f1)}", $"cofactor {c}=0: {Print(f0)}", "difference: " + Print(SimplifyBinary("^", f1, f0)) };
632 }
633 string vars = Variables(f.Variables | (g?.Variables ?? 0));
634 int variableCount = vars.Length;
635 if (command == "PRINT") return new() { "formula: " + Print(f), "variables: " + (vars.Length == 0 ? "none" : string.Join(" ", vars.ToCharArray())), $"size {Size(f)} depth {Depth(f)}" };
636 if (command == "NNF") return new() { "nnf: " + Print(Nnf(f, true)) };
637 if (command == "DUAL") return new() { "dual: " + Print(Dual(f, new())) };
638 if (command == "SIMPLIFY") return new() { "simplified: " + Print(Simplify(f, new(), new())) };
639 if (command is "CLAUSES" or "DP")
640 {
641 var clauses = Clauses(Nnf(f, true), new());
642 if (command == "CLAUSES") return new() { "clauses: " + Form(ClausePatterns(clauses, vars), vars, true) };
643 output.Add("start: clauses " + clauses.Count);
644 foreach (char c in vars)
645 {
646 if (clauses.Count == 0 || clauses.Contains(new(0, 0))) break;
647 int bit = 1 << (c - 'A');
648 var positive = clauses.Where(x => (x.Positive & bit) != 0).ToList();
649 var negative = clauses.Where(x => (x.Negative & bit) != 0).ToList();
650 HashSet<Clause> resolvents = new(), next = new();
651 foreach (Clause a in positive) foreach (Clause b in negative)
652 {
653 Clause r = new((a.Positive | b.Positive) & ~bit, (a.Negative | b.Negative) & ~bit);
654 if (!r.Tautology) resolvents.Add(r);
655 }
656 foreach (Clause a in clauses) if (((a.Positive | a.Negative) & bit) == 0) next.Add(a);
657 next.UnionWith(resolvents);
658 clauses = Tidy(next);
659 output.Add($"eliminate {c}: resolvents {resolvents.Count} clauses {clauses.Count}");
660 }
661 output.Add("result: " + (clauses.Contains(new(0, 0)) ? "unsatisfiable" : "satisfiable"));
662 return output;
663 }
664 if (variableCount >= 9) Fail("too many variables: " + variableCount);
665 int[] table = Truth(f, vars);
666 int[]? other = g == null ? null : Truth(g, vars);
667 int rows = table.Length;
668 if (command is "MINIMIZE" or "MINDC" or "MINPOS") return Minimize(vars, table, other, command == "MINPOS");
669 switch (command)
670 {
671 case "TABLE":
672 output.Add((variableCount == 0 ? "" : string.Join(" ", vars.ToCharArray()) + " ") + "| F");
673 for (int r = 0; r < rows; r++) output.Add((variableCount == 0 ? "" : string.Join(" ", Binary(r, variableCount).ToCharArray()) + " ") + "| " + table[r]);
674 break;
675 case "MINTERMS":
676 case "MAXTERMS":
677 case "DNF":
678 case "CNF":
679 {
680 bool positive = command is "MINTERMS" or "DNF";
681 List<string> patterns = new(), numbers = new();
682 for (int r = 0; r < rows; r++) if (table[r] == (positive ? 1 : 0))
683 {
684 string p = Binary(r, variableCount);
685 patterns.Add(positive ? p : Opposite(p));
686 numbers.Add(r.ToString());
687 }
688 output.Add(command.ToLowerInvariant() + ": " + (command is "DNF" or "CNF" ? Form(patterns, vars, !positive) : numbers.Count == 0 ? "none" : string.Join(" ", numbers)));
689 break;
690 }
691 case "ANF":
692 {
693 int[] coeff = Coefficients(table);
694 List<string> patterns = new();
695 for (int r = 0; r < rows; r++) if (coeff[r] == 1) patterns.Add(Binary(r, variableCount).Replace('0', '-'));
696 patterns.Sort((a, b) => Literals(a) != Literals(b) ? Literals(a).CompareTo(Literals(b)) : PatternCompare(a, b));
697 output.Add("anf: " + (patterns.Count == 0 ? "0" : string.Join(" ^ ", patterns.Select(p => Term(p, vars, false)))));
698 break;
699 }
700 case "SAT":
701 case "TAUT":
702 {
703 bool sat = command == "SAT";
704 int firstRow = -1;
705 for (int r = 0; r < rows; r++) if (table[r] == (sat ? 1 : 0))
706 {
707 firstRow = r;
708 break;
709 }
710 output.Add(firstRow < 0 ? sat ? "unsatisfiable" : "tautology" : (sat ? "satisfiable: " : "not a tautology: ") + Assignment(firstRow, vars));
711 break;
712 }
713 case "COUNT":
714 output.Add($"models: {table.Sum()} of {rows}");
715 break;
716 case "EQUIV":
717 case "IMPLIES":
718 {
719 bool equivalent = command == "EQUIV";
720 int firstRow = -1;
721 for (int r = 0; r < rows; r++) if (equivalent ? table[r] != other![r] : table[r] == 1 && other![r] == 0)
722 {
723 firstRow = r;
724 break;
725 }
726 output.Add(firstRow < 0 ? equivalent ? "equivalent" : "implies" : equivalent ? $"not equivalent: {Assignment(firstRow, vars)} f={table[firstRow]} g={other![firstRow]}" : "does not imply: " + Assignment(firstRow, vars));
727 break;
728 }
729 case "PROPS":
730 {
731 List<char> dependencies = new();
732 bool selfDual = true, monotone = true, linear = true, symmetric = true;
733 for (int i = 0; i < variableCount; i++)
734 {
735 int bit = 1 << (variableCount - 1 - i);
736 for (int r = 0; r < rows; r++) if (table[r] != table[r ^ bit])
737 {
738 dependencies.Add(vars[i]);
739 break;
740 }
741 }
742 int[] coeff = Coefficients(table), weights = Enumerable.Repeat(-1, variableCount + 1).ToArray();
743 for (int r = 0; r < rows; r++)
744 {
745 selfDual &= table[r] != table[rows - 1 - r];
746 if (coeff[r] == 1 && Bits(r) > 1) linear = false;
747 int weight = Bits(r);
748 if (weights[weight] >= 0 && weights[weight] != table[r]) symmetric = false;
749 weights[weight] = table[r];
750 for (int bit = 1; bit < rows; bit <<= 1) if ((r & bit) == 0 && table[r] > table[r | bit]) monotone = false;
751 }
752 output.Add("depends: " + (dependencies.Count == 0 ? "none" : string.Join(" ", dependencies)));
753 string[] labels = { "zero-preserving", "one-preserving", "self-dual", "monotone", "linear", "symmetric" };
754 bool[] values = { table[0] == 0, table[rows - 1] == 1, selfDual, monotone, linear, symmetric };
755 for (int i = 0; i < labels.Length; i++) output.Add(labels[i] + ": " + (values[i] ? "yes" : "no"));
756 break;
757 }
758 case "KMAP":
759 {
760 int n = variableCount;
761 if (n < 2 || n > 4) Fail("kmap needs 2 to 4 variables");
762 int p = n / 2, q = n - p;
763 StringBuilder header = new(vars[..p] + "\\" + vars[p..]);
764 for (int i = 0; i < 1 << q; i++) header.Append(' ').Append(Binary(i ^ i / 2, q));
765 output.Add(header.ToString());
766 for (int i = 0; i < 1 << p; i++)
767 {
768 int r = i ^ i / 2;
769 StringBuilder row = new(Binary(r, p) + new string(' ', q + 1));
770 for (int j = 0; j < 1 << q; j++) row.Append(' ', q).Append(table[(r << q) | (j ^ j / 2)]);
771 output.Add(row.ToString());
772 }
773 break;
774 }
775 case "BDD":
776 {
777 List<Decision> decisions = new();
778 int root = Bdd(table, 0, rows, 0, decisions, new());
779 int[] paths = new int[decisions.Count + 2];
780 paths[1] = 1;
781 output.Add("nodes: " + decisions.Count);
782 for (int i = 0; i < decisions.Count; i++)
783 {
784 Decision d = decisions[i];
785 paths[i + 2] = paths[d.Low] + paths[d.High];
786 output.Add($"{i + 2} = ite({vars[d.Variable]}, {d.High}, {d.Low})");
787 }
788 output.Add("root: " + root);
789 output.Add("paths: " + paths[root]);
790 break;
791 }
792 }
793 return output;
794 }
