61def tokenize(text, start):
62 tokens = []
63 index = 0
64 while index < len(text):
65 if text[index] == ' ':
66 index += 1
67 continue
68 token = TOKEN.match(text, index)
69 if token is None:
70 raise CommandError(f"invalid character '{text[index]}' at {start + index}")
71 tokens.append((token.group(), start + index))
72 index = token.end()
73 return tokens

76class Parser:
77 def __init__(self, tokens, end, names):
78 self.tokens = tokens
79 self.end = end
80 self.names = names
81 self.index = 0
82
83 def parse(self):
84 tree = self.equivalence()
85 if self.index < len(self.tokens):
86 raise self.unexpected()
87 return tree
88
89 def peek(self):
90 if self.index < len(self.tokens):
91 return self.tokens[self.index][0]
92 return None
93
94 def unexpected(self):
95 if self.index < len(self.tokens):
96 text, position = self.tokens[self.index]
97 return CommandError(f"unexpected '{text}' at {position}")
98 return CommandError(f'unexpected end of formula at {self.end}')
99
100 def left_assoc(self, op, operand):
101 tree = operand()
102 while self.peek() == op:
103 self.index += 1
104 tree = Bin(op, tree, operand())
105 return tree
106
107 def equivalence(self):
108 return self.left_assoc('<->', self.implication)
109
110 def implication(self):
111 tree = self.disjunction()
112 if self.peek() == '->':
113 self.index += 1
114 return Bin('->', tree, self.implication())
115 return tree
116
117 def disjunction(self):
118 return self.left_assoc('|', self.xor)
119
120 def xor(self):
121 return self.left_assoc('^', self.conjunction)
122
123 def conjunction(self):
124 return self.left_assoc('&', self.unary)
125
126 def unary(self):
127 if self.peek() == '~':
128 self.index += 1
129 return Not(self.unary())
130 return self.atom()
131
132 def atom(self):
133 if self.index == len(self.tokens):
134 raise self.unexpected()
135 text, position = self.tokens[self.index]
136 if text == '(':
137 self.index += 1
138 tree = self.equivalence()
139 if self.peek() != ')':
140 raise self.unexpected()
141 elif text in ('0', '1'):
142 tree = Const(int(text))
143 elif text.isupper():
144 tree = Var(text)
145 elif text[0].islower():
146 if text not in self.names:
147 raise CommandError(f"undefined name '{text}' at {position}")
148 tree = self.names[text]
149 else:
150 raise self.unexpected()
151 self.index += 1
152 return tree

191def notation(f):
192 match f:
193 case Var(name):
194 return name
195 case Const(value):
196 return str(value)
197 case Not(child):
198 text = notation(child)
199 return f'~({text})' if isinstance(child, Bin) else '~' + text
200 case Bin(op, left, right):
201 left_text = notation(left)
202 right_text = notation(right)
203 if isinstance(left, Bin) and (
204 PRECEDENCE[left.op] < PRECEDENCE[op] or left.op == op == '->'
205 ):
206 left_text = f'({left_text})'
207 if isinstance(right, Bin) and (
208 PRECEDENCE[right.op] < PRECEDENCE[op] or (right.op == op and op != '->')
209 ):
210 right_text = f'({right_text})'
211 return f'{left_text} {op} {right_text}'

214def nnf(f, positive=True):
215 match f:
216 case Var():
217 return f if positive else Not(f)
218 case Const(value):
219 return f if positive else Const(1 - value)
220 case Not(child):
221 return nnf(child, not positive)
222 case Bin('&' | '|' as op, left, right):
223 if not positive:
224 op = DUAL_OPERATORS[op]
225 return Bin(op, nnf(left, positive), nnf(right, positive))
226 case Bin('->', left, right):
227 if positive:
228 return Bin('|', nnf(left, False), nnf(right))
229 return Bin('&', nnf(left), nnf(right, False))
230 case Bin(op, left, right):
231 left_true, left_false = nnf(left), nnf(left, False)
232 right_true, right_false = nnf(right), nnf(right, False)
233 if (op == '<->') == positive:
234 return Bin('|', Bin('&', left_true, right_true), Bin('&', left_false, right_false))
235 return Bin('|', Bin('&', left_true, right_false), Bin('&', left_false, right_true))

238def dual(f):
239 match f:
240 case Var():
241 return f
242 case Const(value):
243 return Const(1 - value)
244 case Not(child):
245 return Not(dual(child))
246 case Bin('->', left, right):
247 return Bin('&', Not(dual(left)), dual(right))
248 case Bin(op, left, right):
249 return Bin(DUAL_OPERATORS[op], dual(left), dual(right))

262def combine(op, a, b):
263 equal = a == b
264 complementary = a == Not(b) or b == Not(a)
265 match op:
266 case '&':
267 if a == ZERO or b == ZERO:
268 return ZERO
269 if a == ONE:
270 return b
271 if b == ONE:
272 return a
273 if equal:
274 return a
275 if complementary:
276 return ZERO
277 case '|':
278 if a == ONE or b == ONE:
279 return ONE
280 if a == ZERO:
281 return b
282 if b == ZERO:
283 return a
284 if equal:
285 return a
286 if complementary:
287 return ONE
288 case '^':
289 if a == ZERO:
290 return b
291 if b == ZERO:
292 return a
293 if a == ONE:
294 return negate(b)
295 if b == ONE:
296 return negate(a)
297 if equal:
298 return ZERO
299 if complementary:
300 return ONE
301 case '->':
302 if a == ZERO or b == ONE:
303 return ONE
304 if a == ONE:
305 return b
306 if b == ZERO:
307 return negate(a)
308 if equal:
309 return ONE
310 if complementary:
311 return b
312 case '<->':
313 if a == ONE:
314 return b
315 if b == ONE:
316 return a
317 if a == ZERO:
318 return negate(b)
319 if b == ZERO:
320 return negate(a)
321 if equal:
322 return ONE
323 if complementary:
324 return ZERO
325 return Bin(op, a, b)

328def simplify(f, constants=None):
329 match f:
330 case Var(name) if constants and name in constants:
331 return constants[name]
332 case Var() | Const():
333 return f
334 case Not(child):
335 return negate(simplify(child, constants))
336 case Bin(op, left, right):
337 return combine(op, simplify(left, constants), simplify(right, constants))

341 match f:
342 case Var(name):
343 return env[name]

345 return value

369def anf_coefficients(values, n):
370 coefficients = list(values)
371 for i in range(n):
372 bit = 1 << i
373 for s in range(len(coefficients)):
374 if s & bit:
375 coefficients[s] ^= coefficients[s ^ bit]

514def properties(f, variables):
515 n = len(variables)
516 values = truth_table(f, variables)
517 rows = range(len(values))
518 bits = [1 << (n - 1 - i) for i in range(n)]
519 depends = [
520 variable
521 for variable, bit in zip(variables, bits)
522 if any(values[row] != values[row ^ bit] for row in rows)

525 self_dual = all(values[row] != values[-1 - row] for row in rows)

527 linear = all(s.bit_count() <= 1 for s in rows if coefficients[s])
528 symmetric = all(values[row] == values[(1 << row.bit_count()) - 1] for row in rows)
529 return [
530 'depends: ' + (' '.join(depends) or 'none'),
531 'zero-preserving: ' + yes_no(values[0] == 0),
532 'one-preserving: ' + yes_no(values[-1] == 1),
533 'self-dual: ' + yes_no(self_dual),
534 'monotone: ' + yes_no(monotone),

557def decision_diagram(f, variables):
558 nodes = []
559 numbers = {}
560
561 def build(level, values):
562 if len(values) == 1:
563 return values[0]
564 half = len(values) // 2
565 low = build(level + 1, values[:half])
566 high = build(level + 1, values[half:])
567 if low == high:
568 return low
569 node = (level, low, high)

581 lines += [f'root: {root}', f'paths: {paths[root]}']

585def quine_mccluskey(rows, n):
586 stage = {row_pattern(row, n) for row in rows}
587 stages = []
588 primes = []
589 while True:
590 merged = set()
591 combined = set()
592 for pattern in stage:
593 for i, c in enumerate(pattern):
594 if c == '-':
595 continue
596 partner = pattern[:i] + ('0' if c == '1' else '1') + pattern[i + 1:]
597 if partner in stage:
598 merged.add(pattern[:i] + '-' + pattern[i + 1:])
599 combined.add(pattern)
600 stages.append((len(stage), len(stage) - len(combined)))
601 primes.extend(stage - combined)
602 if not merged:
603 return stages, primes
604 stage = merged

607def choose_cover(primes, on, n, swap):
608 bits = {row: row_pattern(row, n) for row in on}
609 coverage = {p: {row for row in on if contains(p, bits[row])} for p in primes}
610 targets = [p for p in primes if coverage[p]]
611 essential = set()
612 for row in on:
613 covering = [p for p in targets if row in coverage[p]]
614 if len(covering) == 1:
615 essential.add(covering[0])
616 uncovered = on - set().union(*(coverage[p] for p in essential))
617 optional = [p for p in targets if p not in essential]
618
619 def cover_key(cover):
620 shown = [p.translate(SWAP) if swap else p for p in cover]
621 return sum(map(literal_count, cover)), sorted(map(pattern_key, shown))
622
623 for count in range(len(optional) + 1):
624 covers = [
625 essential.union(chosen)
626 for chosen in combinations(optional, count)
627 if uncovered <= set().union(*(coverage[p] for p in chosen))
628 ]
629 if covers:
630 return targets, essential, min(covers, key=cover_key)

633def minimization_report(on, dont_care, variables):
634 n = len(variables)
635 stages, primes = quine_mccluskey(on | dont_care, n)
636 targets, essential, cover = choose_cover(primes, on, n, swap=False)
637 lines = [
638 f'stage {k}: implicants {count} primes {prime_count}'
639 for k, (count, prime_count) in enumerate(stages)
640 ]
641 lines.append('primes: ' + term_list(targets, variables))
642 lines.append('essential: ' + term_list(essential, variables))
643 lines.append('minimal: ' + sum_of_products(sorted(cover, key=pattern_key), variables))
644 lines.append(f'terms {len(cover)} literals {sum(map(literal_count, cover))}')
645 return lines

671def clauses_of(f):
672 match f:
673 case Var(name):
674 return {frozenset([name])}
675 case Not(Var(name)):
676 return {frozenset(['~' + name])}
677 case Const(value):
678 return set() if value else {frozenset()}
679 case Bin('&', left, right):
680 return clauses_of(left) | clauses_of(right)
681 case Bin('|', left, right):
682 right_clauses = clauses_of(right)
683 return {c | e for c in clauses_of(left) for e in right_clauses}

707def davis_putnam(f, variables):
708 clauses = tidy(clauses_of(nnf(f)))
709 lines = [f'start: clauses {len(clauses)}']
710 for variable in variables:
711 if not clauses or frozenset() in clauses:
712 break
713 negated = '~' + variable
714 positive = {c for c in clauses if variable in c}
715 negative = {c for c in clauses if negated in c}
716 resolvents = {(p - {variable}) | (q - {negated}) for p in positive for q in negative}
717 resolvents = {r for r in resolvents if not is_tautological(r)}
718 clauses = tidy((clauses - positive - negative) | resolvents)
719 lines.append(f'eliminate {variable}: resolvents {len(resolvents)} clauses {len(clauses)}')
720 lines.append('result: unsatisfiable' if frozenset() in clauses else 'result: satisfiable')
721 return lines

738def cofactor(f, text):
739 words = text.split()
740 if len(words) != 1 or not VARIABLE.fullmatch(words[0]):
741 raise CommandError('bad variable')
742 variable = words[0]
743 high = simplify(f, {variable: ONE})

746 f'cofactor {variable}=1: {notation(high)}',
747 f'cofactor {variable}=0: {notation(low)}',
748 f'difference: {notation(combine("^", high, low))}',
749 ]
750

752FORMULA_COMMANDS = {
753 'PRINT': print_formula,
754 'TABLE': show_table,
755 'MINTERMS': show_minterms,
756 'MAXTERMS': show_maxterms,
757 'DNF': canonical_dnf,
758 'CNF': canonical_cnf,
759 'NNF': lambda f, variables: [f'nnf: {notation(nnf(f))}'],
760 'DUAL': lambda f, variables: [f'dual: {notation(dual(f))}'],
761 'ANF': algebraic_normal_form,
762 'SIMPLIFY': lambda f, variables: [f'simplified: {notation(simplify(f))}'],
763 'MINIMIZE': minimize_sum,
764 'MINPOS': minimize_product,
765 'KMAP': karnaugh_map,
766 'SAT': satisfiable,
767 'TAUT': tautology,
768 'COUNT': count_models,
769 'PROPS': properties,
770 'BDD': decision_diagram,
771 'CLAUSES': clause_form,
772 'DP': davis_putnam,
773}
774PAIR_COMMANDS = {'MINDC': minimize_dont_care, 'EQUIV': equivalent, 'IMPLIES': implies}
775SUBSTITUTION_COMMANDS = {'RESTRICT': restrict, 'COFACTOR': cofactor}
776COMMAND_NAMES = {
777 'DEF', 'DEFSUM', 'DEL', 'LIST', *FORMULA_COMMANDS, *PAIR_COMMANDS, *SUBSTITUTION_COMMANDS,
778}

780
781class Workbench:
782 def __init__(self):
783 self.names = {}
784
785 def execute(self, line):
786 name, has_argument, argument = line.partition(' ')
787 if name not in COMMAND_NAMES:
788 raise CommandError(f'unknown command {name}')
789 if name == 'LIST':
790 if has_argument:
791 raise CommandError('unexpected argument')
792 return self.list_names()
793 if not has_argument:
794 raise CommandError('missing argument')
795 if name == 'DEF':
796 return self.define(argument)
797 if name == 'DEFSUM':
798 return self.define_sum(argument)
799 if name == 'DEL':
800 return self.delete(argument)
801 start = len(name) + 2
802 if name in FORMULA_COMMANDS:
803 formulas = [self.parse(argument, start)]
804 handler = FORMULA_COMMANDS[name]
805 else:
806 if ';' not in argument:
807 raise CommandError("missing ';'")
808 front, _, back = argument.partition(';')
809 f = self.parse(front, start)
810 if name in SUBSTITUTION_COMMANDS:
811 return SUBSTITUTION_COMMANDS[name](f, back)
812 formulas = [f, self.parse(back, start + len(front) + 1)]
813 handler = PAIR_COMMANDS[name]
814 variables = variables_of(*formulas)
815 if name in TABLE_COMMANDS and len(variables) >= 9:
816 raise CommandError(f'too many variables: {len(variables)}')
817 return handler(*formulas, variables)
818
819 def parse(self, text, start):
820 return Parser(tokenize(text, start), start + len(text), self.names).parse()
821
822 def assign(self, name, f):
823 verb = 'redefined' if name in self.names else 'defined'
824 self.names[name] = f
825 return [f'{verb} {name} = {notation(f)}']
826
827 def define(self, argument):
828 name, has_formula, text = argument.partition(' ')
829 if not has_formula:
830 raise CommandError('missing argument')
831 if not NAME.fullmatch(name):
832 raise CommandError(f"invalid name '{name}'")
833 return self.assign(name, self.parse(text, len(name) + 6))
834
835 def define_sum(self, argument):
836 if ';' not in argument:
837 raise CommandError("missing ';'")
838 front, _, back = argument.partition(';')
839 words = front.split()
840 if not words:
841 raise CommandError('missing argument')
842 name, variables = words[0], words[1:]
843 if not NAME.fullmatch(name):
844 raise CommandError(f"invalid name '{name}'")
845 if (
846 not variables
847 or not all(VARIABLE.fullmatch(v) for v in variables)
848 or any(a >= b for a, b in pairwise(variables))
849 ):
850 raise CommandError('bad variable list')
851 if len(variables) >= 9:
852 raise CommandError(f'too many variables: {len(variables)}')
853 minterms = []
854 for word in back.split():
855 if (
856 not MINTERM.fullmatch(word)
857 or int(word) >= 1 << len(variables)
858 or (minterms and int(word) <= minterms[-1])
859 ):
860 raise CommandError(f"bad minterm '{word}'")
861 minterms.append(int(word))
862 terms = []
863 for minterm in minterms:
864 bits = row_pattern(minterm, len(variables))
865 factors = [Var(v) if bit == '1' else Not(Var(v)) for v, bit in zip(variables, bits)]
866 terms.append(chain('&', factors))
867 return self.assign(name, chain('|', terms) if terms else ZERO)
868
869 def delete(self, name):
870 if not NAME.fullmatch(name):
871 raise CommandError(f"invalid name '{name}'")
872 if name not in self.names:
873 raise CommandError(f"undefined name '{name}'")
874 del self.names[name]
875 return [f'deleted {name}']
876
877 def list_names(self):
878 lines = [f'names: {len(self.names)}']
879 lines += [f'{name} = {notation(self.names[name])}' for name in sorted(self.names)]
880 return lines

885 workbench = Workbench()
886 for line in lines[1:int(lines[0]) + 1]:
887 try:
888 output = workbench.execute(line)
889 except CommandError as error:
890 output = [f'error: {error}']
891 print('\n'.join(output))

895 main()
