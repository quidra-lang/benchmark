6PRECEDENCE = { '<->' => 1, '->' => 2, '|' => 3, '^' => 4, '&' => 5 }.freeze
7
8def variable(name) = [:variable, name]
9def constant(value) = [:constant, value]
10def negate(node) = [:not, node]
11def binary(operation, left, right) = [operation, left, right]
12def binary?(node) = PRECEDENCE.key?(node[0])
13
14def notation(node)
15 return node[1].to_s if %i[variable constant].include?(node[0])
16 if node[0] == :not
17 text = notation(node[1])
18 return '~' + (binary?(node[1]) ? '(' + text + ')' : text)
19 end
20 operation, a, b = node
21 left = notation(a)
22 right = notation(b)
23 left = '(' + left + ')' if binary?(a) && (PRECEDENCE[a[0]] < PRECEDENCE[operation] || a[0] == '->' && operation == '->')
24 right = '(' + right + ')' if binary?(b) && (PRECEDENCE[b[0]] < PRECEDENCE[operation] || b[0] == operation && operation != '->')
25 left + ' ' + operation + ' ' + right
26end
27
28def variables(node)
29 return [node[1]] if node[0] == :variable
30 return [] if node[0] == :constant
31 return variables(node[1]) if node[0] == :not
32 (variables(node[1]) + variables(node[2])).uniq.sort
33end
34
35def measures(node)
36 return [1, 1] if %i[variable constant].include?(node[0])
37 a = measures(node[1])
38 return [a[0] + 1, a[1] + 1] if node[0] == :not
39 b = measures(node[2])
40 [a[0] + b[0] + 1, [a[1], b[1]].max + 1]
41end

43class FormulaParser
44 def initialize(text, start, definitions)
45 @definitions = definitions
46 @tokens = []
47 i = 0
48 while i < text.length
49 c = text[i]
50 if c == ' '
51 i += 1
52 next
53 end
54 from = i
55 if text[i, 3] == '<->'
56 i += 3
57 elsif text[i, 2] == '->'
58 i += 2
59 elsif c.match?(/[a-z]/)
60 i += 1
61 i += 1 while i < text.length && text[i].match?(/[a-z0-9]/)
62 elsif 'ABCDEFGHIJKLMNOPQRSTUVWXYZ01~&^|()'.include?(c)
63 i += 1
64 else
65 raise LogicError, "invalid character '#{c}' at #{start + i}"
66 end
67 @tokens << [text[from...i], start + from]
68 end
69 @tokens << ['', start + text.length]
70 @position = 0
71 end
72
73 def peek = @tokens[@position][0]
74 def advance
75 text = peek
76 @position += 1
77 text
78 end
79
80 def unexpected
81 text, position = @tokens[@position]
82 raise LogicError, text.empty? ? "unexpected end of formula at #{position}" : "unexpected '#{text}' at #{position}"
83 end
84
85 def unary
86 if peek == '~'
87 advance
88 return negate(unary)
89 end
90 if peek == '('
91 advance
92 result = expression(0)
93 unexpected unless peek == ')'
94 advance
95 return result
96 end
97 if peek.match?(/\A[A-Z]\z/)
98 return variable(advance)
99 end
100 return constant(advance.to_i) if %w[0 1].include?(peek)
101 if peek.match?(/\A[a-z][a-z0-9]*\z/)
102 text, position = @tokens[@position]
103 raise LogicError, "undefined name '#{text}' at #{position}" unless @definitions.key?(text)
104 advance
105 return @definitions[text]
106 end
107 unexpected
108 end
109
110 def expression(level)
111 return unary if level == 5
112 operation = %w[<-> -> | ^ &][level]
113 result = expression(level + 1)
114 while peek == operation
115 advance
116 result = binary(operation, result, expression(operation == '->' ? level : level + 1))
117 end
118 result
119 end
120
121 def parse
122 result = expression(0)
123 unexpected unless peek.empty?
124 result
125 end
126end

128def nnf(node, positive = true, memo = {})
129 key = [node.object_id, positive]
130 return memo[key] if memo.key?(key)
131 result = case node[0]
132 when :variable
133 positive ? node : negate(node)
134 when :constant
135 constant(positive ? node[1] : 1 - node[1])
136 when :not
137 nnf(node[1], !positive, memo)
138 when '&', '|'
139 operation = positive ? node[0] : node[0] == '&' ? '|' : '&'
140 binary(operation, nnf(node[1], positive, memo), nnf(node[2], positive, memo))
141 when '->'
142 binary(positive ? '|' : '&', nnf(node[1], !positive, memo), nnf(node[2], positive, memo))
143 when '<->', '^'
144 same = node[0] == '<->' ? positive : !positive
145 binary('|', binary('&', nnf(node[1], true, memo), nnf(node[2], same, memo)),
146 binary('&', nnf(node[1], false, memo), nnf(node[2], !same, memo)))
147 end
148 memo[key] = result
149end

151def dual(node)
152 case node[0]
153 when :variable then node
154 when :constant then constant(1 - node[1])
155 when :not then negate(dual(node[1]))
156 when '->' then binary('&', negate(dual(node[1])), dual(node[2]))
157 else
158 operation = { '&' => '|', '|' => '&', '^' => '<->', '<->' => '^' }[node[0]]
159 binary(operation, dual(node[1]), dual(node[2]))
160 end
161end

163def reduce_negation(node)
164 return constant(1 - node[1]) if node[0] == :constant
165 node[0] == :not ? node[1] : negate(node)
166end

168def simplify_binary(operation, a, b)
169 zero = constant(0)
170 one = constant(1)
171 equal = a == b
172 complementary = a == negate(b) || b == negate(a)
173 case operation
174 when '&'
175 return zero if a == zero || b == zero
176 return b if a == one
177 return a if b == one || equal
178 return zero if complementary
179 when '|'
180 return one if a == one || b == one
181 return b if a == zero
182 return a if b == zero || equal
183 return one if complementary
184 when '^'
185 return b if a == zero
186 return a if b == zero
187 return reduce_negation(b) if a == one
188 return reduce_negation(a) if b == one
189 return zero if equal
190 return one if complementary
191 when '->'
192 return one if a == zero || b == one
193 return b if a == one
194 return reduce_negation(a) if b == zero
195 return one if equal
196 return b if complementary
197 when '<->'
198 return b if a == one
199 return a if b == one
200 return reduce_negation(b) if a == zero
201 return reduce_negation(a) if b == zero
202 return one if equal
203 return zero if complementary
204 end
205 binary(operation, a, b)
206end

208def simplify(node, assignments = {})
209 return assignments.key?(node[1]) ? constant(assignments[node[1]]) : node if node[0] == :variable
210 return node if node[0] == :constant
211 return reduce_negation(simplify(node[1], assignments)) if node[0] == :not
212 simplify_binary(node[0], simplify(node[1], assignments), simplify(node[2], assignments))
213end
214
215def evaluate(node, assignments)
216 return assignments[node[1]] if node[0] == :variable
217 return node[1] if node[0] == :constant
218 return 1 - evaluate(node[1], assignments) if node[0] == :not
219 a = evaluate(node[1], assignments)
220 b = evaluate(node[2], assignments)
221 case node[0]
222 when '&' then a & b
223 when '|' then a | b
224 when '^' then a ^ b
225 when '->' then a == 1 && b == 0 ? 0 : 1
226 when '<->' then a == b ? 1 : 0
227 end
228end

230def pattern(row, size)
231 size == 0 ? '' : row.to_s(2).rjust(size, '0')
232end
233
234def pattern_key(text) = text.tr('10-', '012')
235def literal_count(text) = text.count('10')
236def complement_pattern(text) = text.tr('01', '10')
237def contains?(text, row) = text.each_char.with_index.all? { |c, i| c == '-' || c == row[i] }
238
239def pattern_part(text, names, product = true)
240 literals = text.each_char.with_index.filter_map { |c, i| c == '-' ? nil : (c == '1' ? names[i] : '~' + names[i]) }
241 literals.empty? ? (product ? '1' : '0') : literals.join(product ? ' & ' : ' | ')
242end
243
244def cover_text(patterns, names, product = true)
245 return product ? '0' : '1' if patterns.empty?
246 patterns.map do |text|
247 part = pattern_part(text, names, product)
248 !product && patterns.length >= 2 && literal_count(text) >= 2 ? '(' + part + ')' : part
249 end.join(product ? ' | ' : ' & ')
250end

252def coefficients(values, size)
253 result = values.dup
254 size.times do |i|
255 bit = 1 << i
256 result.each_index { |row| result[row] ^= result[row ^ bit] if (row & bit) != 0 }
257 end
258 result
259end

261def minimize(on, dc, names, pos = false)
262 rows = on.to_h { |row| [row, pattern(row, names.length)] }
263 stage = (on + dc).uniq.map { |row| pattern(row, names.length) }.to_set
264 primes = Set.new
265 lines = []
266 number = 0
267 loop do
268 used = Set.new
269 next_stage = Set.new
270 stage.each do |text|
271 text.length.times do |i|
272 next unless text[i] == '0'
273 partner = text.dup
274 partner[i] = '1'
275 next unless stage.include?(partner)
276 combined = text.dup
277 combined[i] = '-'
278 used << text << partner
279 next_stage << combined
280 end
281 end
282 fresh_primes = stage - used
283 primes.merge(fresh_primes)
284 lines << "stage #{number}: implicants #{stage.length} primes #{fresh_primes.length}" unless pos
285 break if next_stage.empty?
286 stage = next_stage
287 number += 1
288 end
289 targets = primes.select { |text| rows.values.any? { |row| contains?(text, row) } }.sort_by { |text| pattern_key(text) }
290 coverage = targets.map { |text| on.select { |row| contains?(text, rows[row]) }.to_set }
291 essential = Set.new
292 on.each do |row|
293 candidates = coverage.each_index.select { |i| coverage[i].include?(row) }
294 essential << candidates[0] if candidates.length == 1
295 end
296 optional = targets.each_index.reject { |i| essential.include?(i) }
297 best = nil
298 best_key = nil
299 (1 << optional.length).times do |mask|
300 chosen = essential.to_a + optional.each_with_index.filter_map { |index, i| index if mask[i] == 1 }
301 covered = chosen.reduce(Set.new) { |set, index| set | coverage[index] }
302 next unless on.all? { |row| covered.include?(row) }
303 texts = chosen.map { |i| pos ? complement_pattern(targets[i]) : targets[i] }.sort_by { |text| pattern_key(text) }
304 key = [texts.length, texts.sum { |text| literal_count(text) }, texts.map { |text| pattern_key(text) }]
305 if best_key.nil? || (key <=> best_key) < 0
306 best_key = key
307 best = texts
308 end
309 end
310 if pos
311 ["minimal: #{cover_text(best, names, false)}", "clauses #{best.length} literals #{best_key[1]}"]
312 else
313 terms = targets.map { |text| pattern_part(text, names) }
314 essentials = essential.to_a.sort.map { |i| pattern_part(targets[i], names) }
315 lines + ["primes: #{terms.empty? ? 'none' : terms.join(', ')}",
316 "essential: #{essentials.empty? ? 'none' : essentials.join(', ')}",
317 "minimal: #{cover_text(best, names)}", "terms #{best.length} literals #{best_key[1]}"]
318 end
319end

321def merge_clause(a, b)
322 result = ''
323 a.length.times do |i|
324 return nil if a[i] != '-' && b[i] != '-' && a[i] != b[i]
325 result << (a[i] == '-' ? b[i] : a[i])
326 end
327 result
328end

330def tidy(clauses)
331 unique = clauses.uniq
332 unique.reject do |clause|
333 unique.any? { |other| clause != other && other.each_char.with_index.all? { |c, i| c == '-' || c == clause[i] } }
334 end.sort_by { |text| pattern_key(text) }
335end

337def clauses(node, names, memo = {})
338 key = node.object_id
339 return memo[key] if memo.key?(key)
340 result = case node[0]
341 when :variable, :not
342 name = node[0] == :variable ? node[1] : node[1][1]
343 text = '-' * names.length
344 text[names.index(name)] = node[0] == :variable ? '1' : '0'
345 [text]
346 when :constant
347 node[1] == 1 ? [] : ['-' * names.length]
348 when '&'
349 tidy(clauses(node[1], names, memo) + clauses(node[2], names, memo))
350 when '|'
351 left = clauses(node[1], names, memo)
352 right = clauses(node[2], names, memo)
353 tidy(left.flat_map { |a| right.filter_map { |b| merge_clause(a, b) } })
354 end
355 memo[key] = result
356end

359 values = pattern(row, names.length)
360 '{' + names.each_index.map { |i| names[i] + '=' + values[i] }.join(' ') + '}'
361end

363class Workbench
364 COMMANDS = %w[DEF DEFSUM DEL LIST PRINT TABLE MINTERMS MAXTERMS DNF CNF NNF DUAL ANF SIMPLIFY RESTRICT COFACTOR MINIMIZE MINDC MINPOS KMAP SAT TAUT COUNT EQUIV IMPLIES PROPS BDD CLAUSES DP].freeze

381 def bdd(values, names)
382 nodes = []
383 unique = {}
384 build = lambda do |i, table|
385 return table[0] if i == names.length
386 half = table.length / 2
387 low = build.call(i + 1, table.take(half))
388 high = build.call(i + 1, table.drop(half))
389 return low if low == high
390 key = [i, low, high]
391 unless unique.key?(key)
392 unique[key] = nodes.length + 2
393 nodes << key
394 end
395 unique[key]
396 end
397 root = build.call(0, values)
398 counts = { 0 => 0, 1 => 1 }
399 paths = lambda do |id|
400 counts[id] ||= paths.call(nodes[id - 2][1]) + paths.call(nodes[id - 2][2])
401 end
402 ["nodes: #{nodes.length}"] + nodes.each_with_index.map { |(i, low, high), j| "#{j + 2} = ite(#{names[i]}, #{high}, #{low})" } + ["root: #{root}", "paths: #{paths.call(root)}"]
403 end

405 def command(line)
406 separator = line.index(' ')
407 command = separator ? line[0...separator] : line
408 argument = separator ? line[(separator + 1)..] : ''
409 raise LogicError, "unknown command #{command}" unless COMMANDS.include?(command)
410 raise LogicError, 'unexpected argument' if command == 'LIST' && separator
411 raise LogicError, 'missing argument' if command != 'LIST' && separator.nil?
412 if command == 'LIST'
413 return ["names: #{@definitions.length}"] + @definitions.keys.sort.map { |name| name + ' = ' + notation(@definitions[name]) }
414 end
415 if command == 'DEF'
416 split = argument.index(' ')
417 raise LogicError, 'missing argument' unless split
418 name = argument[0...split]
419 raise LogicError, "invalid name '#{name}'" unless valid_name?(name)
420 return define(name, parse(argument[(split + 1)..], name.length + 6))
421 end
422 if command == 'DEL'
423 raise LogicError, "invalid name '#{argument}'" unless valid_name?(argument)
424 raise LogicError, "undefined name '#{argument}'" unless @definitions.key?(argument)
425 @definitions.delete(argument)
426 return ["deleted #{argument}"]
427 end
428 if command == 'DEFSUM'
429 split = argument.index(';')
430 raise LogicError, "missing ';'" unless split
431 words = argument[0...split].split(' ')
432 raise LogicError, 'missing argument' if words.empty?
433 name, *names = words
434 raise LogicError, "invalid name '#{name}'" unless valid_name?(name)
435 raise LogicError, 'bad variable list' if names.empty? || names.any? { |item| !item.match?(/\A[A-Z]\z/) } || names.each_cons(2).any? { |a, b| a >= b }
436 raise LogicError, "too many variables: #{names.length}" if names.length >= 9
437 previous = -1
438 rows = argument[(split + 1)..].split(' ').map do |word|
439 unless word.match?(/\A(?:0|[1-9][0-9]*)\z/) && word.to_i < (1 << names.length) && word.to_i > previous
440 raise LogicError, "bad minterm '#{word}'"
441 end
442 previous = word.to_i
443 end
444 terms = rows.map do |row|
445 pattern(row, names.length).each_char.with_index.map { |c, i| c == '1' ? variable(names[i]) : negate(variable(names[i])) }.reduce { |a, b| binary('&', a, b) }
446 end
447 return define(name, terms.empty? ? constant(0) : terms.reduce { |a, b| binary('|', a, b) })
448 end
449 second = nil
450 second_start = nil
451 front = argument
452 if SPLIT.include?(command)
453 split = argument.index(';')
454 raise LogicError, "missing ';'" unless split
455 front = argument[0...split]
456 second = argument[(split + 1)..]
457 second_start = command.length + split + 3
458 end
459 formula = parse(front, command.length + 2)
460 if command == 'RESTRICT'
461 words = second.split(' ')
462 raise LogicError, 'missing assignment' if words.empty?
463 assignments = {}
464 words.each do |word|
465 raise LogicError, "bad assignment '#{word}'" unless word.match?(/\A[A-Z]=[01]\z/)
466 raise LogicError, "duplicate variable #{word[0]}" if assignments.key?(word[0])
467 assignments[word[0]] = word[2].to_i
468 end
469 return ['restricted: ' + notation(simplify(formula, assignments))]
470 end
471 if command == 'COFACTOR'
472 words = second.split(' ')
473 raise LogicError, 'bad variable' unless words.length == 1 && words[0].match?(/\A[A-Z]\z/)
474 name = words[0]
475 high = simplify(formula, name => 1)
476 low = simplify(formula, name => 0)
477 return ["cofactor #{name}=1: #{notation(high)}", "cofactor #{name}=0: #{notation(low)}", 'difference: ' + notation(simplify_binary('^', high, low))]
478 end
479 other = second ? parse(second, second_start) : nil
480 names = (variables(formula) + (other ? variables(other) : [])).uniq.sort
481 raise LogicError, "too many variables: #{names.length}" if TABLE.include?(command) && names.length >= 9
482 case command
483 when 'PRINT'
484 size, depth = measures(formula)
485 return ['formula: ' + notation(formula), 'variables: ' + (names.empty? ? 'none' : names.join(' ')), "size #{size} depth #{depth}"]
486 when 'NNF' then return ['nnf: ' + notation(nnf(formula))]
487 when 'DUAL' then return ['dual: ' + notation(dual(formula))]
488 when 'SIMPLIFY' then return ['simplified: ' + notation(simplify(formula))]
489 when 'CLAUSES', 'DP'
490 set = clauses(nnf(formula), names)
491 return ['clauses: ' + cover_text(set, names, false)] if command == 'CLAUSES'
492 lines = ["start: clauses #{set.length}"]
493 empty = '-' * names.length
494 names.each_with_index do |name, i|
495 break if set.empty? || set.include?(empty)
496 positive = set.select { |text| text[i] == '1' }
497 negative = set.select { |text| text[i] == '0' }
498 resolvents = positive.flat_map do |a|
499 negative.filter_map do |b|
500 x, y = a.dup, b.dup
501 x[i] = y[i] = '-'
502 merge_clause(x, y)
503 end
504 end.uniq
505 set = tidy(set.select { |text| text[i] == '-' } + resolvents)
506 lines << "eliminate #{name}: resolvents #{resolvents.length} clauses #{set.length}"
507 end
508 return lines + [set.include?(empty) ? 'result: unsatisfiable' : 'result: satisfiable']
509 end
510 values = Array.new(1 << names.length) do |row|
511 assignments = names.each_with_index.to_h { |name, i| [name, row[names.length - 1 - i]] }
512 evaluate(formula, assignments)
513 end
514 other_values = other ? Array.new(values.length) { |row| evaluate(other, names.each_with_index.to_h { |name, i| [name, row[names.length - 1 - i]] }) } : nil
515 on = values.each_index.select { |row| values[row] == 1 }
516 off = values.each_index.select { |row| values[row] == 0 }
517 case command
518 when 'TABLE'
519 [([*names, '|', 'F']).join(' ')] + values.each_index.map { |row| ([*pattern(row, names.length).chars, '|', values[row]]).join(' ') }
520 when 'MINTERMS', 'MAXTERMS'
521 rows = command == 'MINTERMS' ? on : off
522 [command.downcase + ': ' + (rows.empty? ? 'none' : rows.join(' '))]
523 when 'DNF', 'CNF'
524 texts = (command == 'DNF' ? on : off).map { |row| pattern(row, names.length) }
525 texts.map! { |text| complement_pattern(text) } if command == 'CNF'
526 [command.downcase + ': ' + cover_text(texts, names, command == 'DNF')]
527 when 'ANF'
528 coeff = coefficients(values, names.length)
529 texts = coeff.each_index.select { |row| coeff[row] == 1 }.map { |row| pattern(row, names.length).tr('0', '-') }.sort_by { |text| [literal_count(text), pattern_key(text)] }
530 ['anf: ' + (texts.empty? ? '0' : texts.map { |text| pattern_part(text, names) }.join(' ^ '))]
531 when 'SAT'
532 [on.empty? ? 'unsatisfiable' : 'satisfiable: ' + assignment_text(names, on[0])]
533 when 'TAUT'
534 [off.empty? ? 'tautology' : 'not a tautology: ' + assignment_text(names, off[0])]
535 when 'COUNT'
536 ["models: #{on.length} of #{values.length}"]
537 when 'EQUIV'
538 row = values.each_index.find { |i| values[i] != other_values[i] }
539 [row ? "not equivalent: #{assignment_text(names, row)} f=#{values[row]} g=#{other_values[row]}" : 'equivalent']
540 when 'IMPLIES'
541 row = values.each_index.find { |i| values[i] == 1 && other_values[i] == 0 }
542 [row ? 'does not imply: ' + assignment_text(names, row) : 'implies']
543 when 'PROPS'
544 depends = names.each_index.select do |i|
545 bit = 1 << (names.length - 1 - i)
546 values.each_index.any? { |row| values[row] != values[row ^ bit] }
547 end.map { |i| names[i] }
548 monotone = values.each_index.all? do |row|
549 names.each_index.all? do |i|
550 bit = 1 << i
551 (row & bit) != 0 || values[row] <= values[row | bit]
552 end
553 end
554 linear = coefficients(values, names.length).each_with_index.all? { |value, row| value == 0 || row.to_s(2).count('1') <= 1 }
555 weights = {}
556 symmetric = values.each_index.all? do |row|
557 weight = row.to_s(2).count('1')
558 previous = weights[weight]
559 weights[weight] = values[row]
560 previous.nil? || previous == values[row]
561 end
562 properties = [['zero-preserving', values[0] == 0], ['one-preserving', values[-1] == 1], ['self-dual', values.each_index.all? { |row| values[row] != values[-1 - row] }], ['monotone', monotone], ['linear', linear], ['symmetric', symmetric]]
563 ['depends: ' + (depends.empty? ? 'none' : depends.join(' '))] + properties.map { |label, value| label + ': ' + (value ? 'yes' : 'no') }
564 when 'KMAP'
565 raise LogicError, 'kmap needs 2 to 4 variables' unless (2..4).cover?(names.length)
566 p = names.length / 2
567 q = names.length - p
568 rows = Array.new(1 << p) { |i| pattern(i ^ (i / 2), p) }
569 columns = Array.new(1 << q) { |i| pattern(i ^ (i / 2), q) }
