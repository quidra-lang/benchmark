34String notation(Formula node) {
35 if (node is Variable) return node.name;
36 if (node is Constant) return '${node.value}';
37 if (node is Negation)
38 return '~${node.child is Binary ? '(${notation(node.child)})' : notation(node.child)}';
39 final binary = node as Binary, a = binary.left, b = binary.right;
40 final level = precedence[binary.op]!;
41 final left =
42 a is Binary &&
43 (precedence[a.op]! < level || binary.op == '->' && a.op == '->');
44 final right =
45 b is Binary &&
46 (precedence[b.op]! < level || binary.op == b.op && binary.op != '->');
47 return '${left ? '(${notation(a)})' : notation(a)} ${binary.op} ${right ? '(${notation(b)})' : notation(b)}';
48}

60(int, int) measure(Formula node) {
61 if (node is Variable || node is Constant) return (1, 1);
62 if (node is Negation) {
63 final child = measure(node.child);
64 return (child.$1 + 1, child.$2 + 1);
65 }
66 final binary = node as Binary,
67 left = measure(binary.left),
68 right = measure(binary.right);
69 return (
70 left.$1 + right.$1 + 1,
71 (left.$2 > right.$2 ? left.$2 : right.$2) + 1,
72 );

81List<Token> lex(String text, int start) {
82 final tokens = <Token>[];
83 var i = 0;
84 while (i < text.length) {
85 final c = text[i], position = start + i;
86 if (c == ' ') {
87 i++;
88 continue;
89 }
90 if (text.startsWith('<->', i)) {
91 tokens.add(Token('<->', position));
92 i += 3;
93 continue;
94 }
95 if (text.startsWith('->', i)) {
96 tokens.add(Token('->', position));
97 i += 2;
98 continue;
99 }
100 if (RegExp(r'[a-z]').hasMatch(c)) {
101 final begin = i++;
102 while (i < text.length && RegExp(r'[a-z0-9]').hasMatch(text[i])) {
103 i++;
104 }
105 tokens.add(Token(text.substring(begin, i), position));
106 continue;
107 }
108 if (RegExp(r'[A-Z01~&^|()]').hasMatch(c)) {
109 tokens.add(Token(c, position));
110 i++;
111 continue;
112 }
113 fail("invalid character '$c' at $position");
114 }
115 tokens.add(Token('', start + text.length));
116 return tokens;
117}

119class FormulaParser {
120 final List<Token> tokens;
121 final Map<String, Formula> definitions;
122 int at = 0;
123 FormulaParser(this.tokens, this.definitions);
124 Never unexpected() {
125 final token = tokens[at];
126 fail(
127 token.text.isEmpty
128 ? 'unexpected end of formula at ${token.position}'
129 : "unexpected '${token.text}' at ${token.position}",
130 );
131 }
132
133 Formula unary() {
134 final token = tokens[at];
135 if (token.text == '~') {
136 at++;
137 return Negation(unary());
138 }
139 if (token.text == '(') {
140 at++;
141 final result = expression(1);
142 if (tokens[at].text != ')') unexpected();
143 at++;
144 return result;
145 }
146 if (RegExp(r'^[A-Z]$').hasMatch(token.text)) {
147 at++;
148 return Variable(token.text);
149 }
150 if (token.text == '0' || token.text == '1') {
151 at++;
152 return Constant(int.parse(token.text));
153 }
154 if (RegExp(r'^[a-z]').hasMatch(token.text)) {
155 at++;
156 final definition = definitions[token.text];
157 if (definition == null)
158 fail("undefined name '${token.text}' at ${token.position}");
159 return definition;
160 }
161 unexpected();
162 }
163
164 Formula expression(int minimum) {
165 Formula left = unary();
166 while (precedence.containsKey(tokens[at].text) &&
167 precedence[tokens[at].text]! >= minimum) {
168 final op = tokens[at++].text, level = precedence[op]!;
169 left = Binary(op, left, expression(level + (op == '->' ? 0 : 1)));
170 }
171 return left;
172 }
173
174 Formula parse() {
175 final result = expression(1);
176 if (tokens[at].text.isNotEmpty) unexpected();
177 return result;
178 }
179}

186bool equal(Formula a, Formula b) {
187 if (identical(a, b)) return true;
188 if (a is Variable && b is Variable) return a.name == b.name;
189 if (a is Constant && b is Constant) return a.value == b.value;
190 if (a is Negation && b is Negation) return equal(a.child, b.child);
191 return a is Binary &&
192 b is Binary &&
193 a.op == b.op &&
194 equal(a.left, b.left) &&
195 equal(a.right, b.right);
196}

198Formula reducedNegation(Formula a) => a is Constant
199 ? Constant(1 - a.value)
200 : a is Negation
201 ? a.child
202 : Negation(a);
203Formula simplifyBinary(String op, Formula a, Formula b) {
204 bool isValue(Formula node, int value) =>
205 node is Constant && node.value == value;
206 final same = equal(a, b),
207 complementary =
208 a is Negation && equal(a.child, b) ||
209 b is Negation && equal(a, b.child);
210 switch (op) {
211 case '&':
212 if (isValue(a, 0) || isValue(b, 0)) return Constant(0);
213 if (isValue(a, 1)) return b;
214 if (isValue(b, 1)) return a;
215 if (same) return a;
216 if (complementary) return Constant(0);
217 break;
218 case '|':
219 if (isValue(a, 1) || isValue(b, 1)) return Constant(1);
220 if (isValue(a, 0)) return b;
221 if (isValue(b, 0)) return a;
222 if (same) return a;
223 if (complementary) return Constant(1);
224 break;
225 case '^':
226 if (isValue(a, 0)) return b;
227 if (isValue(b, 0)) return a;
228 if (isValue(a, 1)) return reducedNegation(b);
229 if (isValue(b, 1)) return reducedNegation(a);
230 if (same) return Constant(0);
231 if (complementary) return Constant(1);
232 break;
233 case '->':
234 if (isValue(a, 0) || isValue(b, 1)) return Constant(1);
235 if (isValue(a, 1)) return b;
236 if (isValue(b, 0)) return reducedNegation(a);
237 if (same) return Constant(1);
238 if (complementary) return b;
239 break;
240 case '<->':
241 if (isValue(a, 1)) return b;
242 if (isValue(b, 1)) return a;
243 if (isValue(a, 0)) return reducedNegation(b);
244 if (isValue(b, 0)) return reducedNegation(a);
245 if (same) return Constant(1);
246 if (complementary) return Constant(0);
247 break;
248 }
249 return Binary(op, a, b);
250}

252Formula simplify(Formula node, [Map<String, int> replacements = const {}]) {
253 if (node is Constant) return node;
254 if (node is Variable)
255 return replacements.containsKey(node.name)
256 ? Constant(replacements[node.name]!)
257 : node;
258 if (node is Negation)
259 return reducedNegation(simplify(node.child, replacements));
260 final binary = node as Binary;
261 return simplifyBinary(
262 binary.op,
263 simplify(binary.left, replacements),
264 simplify(binary.right, replacements),
265 );
266}

278Formula nnfCompute(Formula node, bool positive) {
279 if (node is Constant) return Constant(positive ? node.value : 1 - node.value);
280 if (node is Variable) return positive ? node : Negation(node);
281 if (node is Negation) return nnf(node.child, !positive);
282 final binary = node as Binary,
283 a = binary.left,
284 b = binary.right,
285 op = binary.op;
286 if (op == '&' || op == '|')
287 return Binary(
288 positive
289 ? op
290 : op == '&'
291 ? '|'
292 : '&',
293 nnf(a, positive),
294 nnf(b, positive),
295 );
296 if (op == '->')
297 return Binary(positive ? '|' : '&', nnf(a, !positive), nnf(b, positive));
298 final same = op == '<->' ? positive : !positive;
299 return Binary(
300 '|',
301 Binary('&', nnf(a), nnf(b, same)),
302 Binary('&', nnf(a, false), nnf(b, !same)),
303 );
304}

306Formula dual(Formula node) {
307 if (node is Variable) return node;
308 if (node is Constant) return Constant(1 - node.value);
309 if (node is Negation) return Negation(dual(node.child));
310 final binary = node as Binary, a = dual(binary.left), b = dual(binary.right);
311 return binary.op == '->'
312 ? Binary('&', Negation(a), b)
313 : Binary(
314 binary.op == '&'
315 ? '|'
316 : binary.op == '|'
317 ? '&'
318 : binary.op == '^'
319 ? '<->'
320 : '^',
321 a,
322 b,
323 );
324}

326int evaluate(Formula node, List<String> names, int row) {
327 if (node is Constant) return node.value;
328 if (node is Variable)
329 return row >> (names.length - names.indexOf(node.name) - 1) & 1;
330 if (node is Negation) return 1 - evaluate(node.child, names, row);
331 final binary = node as Binary,
332 a = evaluate(binary.left, names, row),
333 b = evaluate(binary.right, names, row);
334 return binary.op == '&'
335 ? a & b
336 : binary.op == '|'
337 ? a | b
338 : binary.op == '^'
339 ? a ^ b
340 : binary.op == '->'
341 ? (1 - a) | b
342 : a == b
343 ? 1
344 : 0;
345}

394 if (patterns.isEmpty) return product ? '0' : '1';
395 return patterns

405List<int> coefficients(List<int> values) {
406 final result = List<int>.of(values);
407 for (var bit = 1; bit < values.length; bit *= 2) {
408 for (var row = 0; row < values.length; row++) {
409 if ((row & bit) != 0) result[row] ^= result[row ^ bit];
410 }
411 }
412 return result;
413}

415List<String> minimize(
416 List<String> names,
417 List<int> on,
418 List<int> dc,
419 bool pos,
420) {
421 var stage = [...on, ...dc].map((row) => bits(row, names.length)).toSet();
422 final primes = <String>[], output = <String>[];
423 for (var number = 0; ; number++) {
424 final used = <String>{}, next = <String>{};
425 for (final pattern in stage) {
426 for (var i = 0; i < pattern.length; i++) {
427 if (pattern[i] != '0') continue;
428 final partner =
429 '${pattern.substring(0, i)}1${pattern.substring(i + 1)}';
430 if (stage.contains(partner)) {
431 used.addAll([pattern, partner]);
432 next.add('${pattern.substring(0, i)}-${pattern.substring(i + 1)}');
433 }
434 }
435 }
436 final current = stage.where((pattern) => !used.contains(pattern)).toList();
437 primes.addAll(current);
438 if (!pos)
439 output.add(
440 'stage $number: implicants ${stage.length} primes ${current.length}',
441 );
442 if (next.isEmpty) break;
443 stage = next;
444 }
445 final rows = on.map((row) => bits(row, names.length)).toList();
446 final target =
447 primes
448 .where((pattern) => rows.any((row) => contains(pattern, row)))
449 .toList()
450 ..sort(patternCompare);
451 final masks = <BigInt>[];
452 for (final pattern in target) {
453 var mask = BigInt.zero;
454 for (var i = 0; i < rows.length; i++) {
455 if (contains(pattern, rows[i])) mask |= BigInt.one << i;
456 }
457 masks.add(mask);
458 }
459 final essential = <int>{};
460 for (var i = 0; i < rows.length; i++) {
461 final candidates = <int>[
462 for (var j = 0; j < target.length; j++)
463 if ((masks[j] & (BigInt.one << i)) != BigInt.zero) j,
464 ];
465 if (candidates.length == 1) essential.add(candidates.first);
466 }
467 final fixed = essential.toList(),
468 optional = <int>[
469 for (var i = 0; i < target.length; i++)
470 if (!essential.contains(i)) i,
471 ];
472 final full = (BigInt.one << rows.length) - BigInt.one;
473 List<int>? best;
474 var bestLiterals = 0;
475 List<String> ordered(List<int> indices) =>
476 indices.map((i) => pos ? flip(target[i]) : target[i]).toList()
477 ..sort(patternCompare);
478 bool better(List<int> candidate, int literals) {
479 final previous = best;
480 if (previous == null || candidate.length != previous.length)
481 return previous == null || candidate.length < previous.length;
482 if (literals != bestLiterals) return literals < bestLiterals;
483 final a = ordered(candidate), b = ordered(previous);
484 for (var i = 0; i < a.length; i++) {
485 final order = patternCompare(a[i], b[i]);
486 if (order != 0) return order < 0;
487 }
488 return false;
489 }
490
491 for (var subset = 0; subset < (1 << optional.length); subset++) {
492 final candidate = List<int>.of(fixed);
493 var covered = BigInt.zero, literals = 0;
494 for (var i = 0; i < optional.length; i++) {
495 if ((subset & (1 << i)) != 0) candidate.add(optional[i]);
496 }
497 final previous = best;
498 if (previous != null && candidate.length > previous.length) continue;
499 for (final index in candidate) {
500 covered |= masks[index];
501 literals += literalCount(target[index]);
502 }
503 if (covered == full && better(candidate, literals)) {
504 best = candidate;
505 bestLiterals = literals;
506 }
507 }
508 final chosen = ordered(best!);
509 if (pos)
510 return [
511 'minimal: ${coverText(chosen, names, false)}',
512 'clauses ${chosen.length} literals $bestLiterals',
513 ];
514 final primeText = target
515 .map((pattern) => patternText(pattern, names, true))
516 .join(', ');
517 final essentials = fixed.map((i) => target[i]).toList()..sort(patternCompare);
518 final essentialText = essentials
519 .map((pattern) => patternText(pattern, names, true))
520 .join(', ');
521 output.addAll([
522 'primes: ${primeText.isEmpty ? 'none' : primeText}',
523 'essential: ${essentialText.isEmpty ? 'none' : essentialText}',
524 'minimal: ${coverText(chosen, names, true)}',
525 'terms ${chosen.length} literals $bestLiterals',
526 ]);
527 return output;
528}

530String? mergeClauses(String a, String b) {
531 var result = '';
532 for (var i = 0; i < a.length; i++) {
533 if (a[i] != '-' && b[i] != '-' && a[i] != b[i]) return null;
534 result += a[i] == '-' ? b[i] : a[i];
535 }
536 return result;
537}

539List<String> tidy(List<String> patterns) {
540 final unique = patterns.toSet().toList();
541 bool includes(String pattern, String other) {
542 for (var i = 0; i < pattern.length; i++) {
543 if (other[i] != '-' && other[i] != pattern[i]) return false;
544 }
545 return true;
546 }
547
548 return unique
549 .where(

551 (other) =>

572List<String> clausesCompute(
573 Formula node,
574 List<String> names,
575 Map<Formula, List<String>> cache,
576) {
577 if (node is Constant) return node.value == 1 ? [] : ['-' * names.length];
578 if (node is Variable || node is Negation) {
579 final name = node is Variable
580 ? node.name
581 : ((node as Negation).child as Variable).name;
582 final i = names.indexOf(name);
583 return [
584 '${'-' * i}${node is Negation ? '0' : '1'}${'-' * (names.length - i - 1)}',
585 ];
586 }
587 final binary = node as Binary,
588 a = clauses(binary.left, names, cache),
589 b = clauses(binary.right, names, cache);
590 if (binary.op == '&') return tidy([...a, ...b]);
591 final merged = <String>[];
592 for (final x in a) {
593 for (final y in b) {
594 final value = mergeClauses(x, y);
595 if (value != null) merged.add(value);
596 }
597 }
598 return tidy(merged);
599}

615 List<String> command(String line) {
616 final split = line.indexOf(' '),
617 command = split < 0 ? line : line.substring(0, split);
618 final argument = split < 0 ? '' : line.substring(split + 1),
619 start = command.length + 2;
620 final single = 'PRINT TABLE MINTERMS MAXTERMS DNF CNF NNF DUAL ANF SIMPLIFY MINIMIZE MINPOS KMAP SAT TAUT COUNT PROPS BDD CLAUSES DP'
621 .split(' ')
622 .toSet();
623 const divided = {'RESTRICT', 'COFACTOR', 'MINDC', 'EQUIV', 'IMPLIES'};
624 if (!single.contains(command) &&
625 !divided.contains(command) &&
626 !['DEF', 'DEFSUM', 'DEL', 'LIST'].contains(command))
627 fail('unknown command $command');
628 if (command == 'LIST' && split >= 0) fail('unexpected argument');
629 if (command != 'LIST' && split < 0) fail('missing argument');
630 final separator = argument.indexOf(';');
631 if (divided.contains(command) && separator < 0) fail("missing ';'");
632 bool validName(String name) =>
633 RegExp(r'^[a-z][a-z0-9]{0,7}$').hasMatch(name);
634 List<String> words(String text) =>
635 text.split(' ').where((word) => word.isNotEmpty).toList();
636 if (command == 'LIST') {
637 final names = definitions.keys.toList()..sort();
638 return [
639 'names: ${definitions.length}',
640 ...names.map((name) => '$name = ${notation(definitions[name]!)}'),
641 ];
642 }
643 if (command == 'DEL') {
644 if (!validName(argument)) fail("invalid name '$argument'");
645 if (!definitions.containsKey(argument))
646 fail("undefined name '$argument'");
647 definitions.remove(argument);
648 return ['deleted $argument'];
649 }
650 if (command == 'DEF') {
651 final space = argument.indexOf(' ');
652 if (space < 0) fail('missing argument');
653 final name = argument.substring(0, space);
654 if (!validName(name)) fail("invalid name '$name'");
655 return define(
656 name,
657 parseFormula(
658 argument.substring(space + 1),
659 name.length + 6,
660 definitions,
661 ),
662 );
663 }
664 if (command == 'DEFSUM') {
665 if (separator < 0) fail("missing ';'");
666 final front = words(argument.substring(0, separator)),
667 back = words(argument.substring(separator + 1));
668 if (front.isEmpty) fail('missing argument');
669 final name = front[0], names = front.sublist(1);
670 if (!validName(name)) fail("invalid name '$name'");
671 if (names.isEmpty) fail('bad variable list');
672 for (var i = 0; i < names.length; i++) {
673 if (!RegExp(r'^[A-Z]$').hasMatch(names[i]) ||
674 i > 0 && names[i - 1].compareTo(names[i]) >= 0)
675 fail('bad variable list');
676 }
677 if (names.length >= 9) fail('too many variables: ${names.length}');
678 final rows = <int>[];
679 var previous = -1;
680 for (final word in back) {
681 if (!RegExp(r'^(0|[1-9][0-9]*)$').hasMatch(word) ||
682 int.parse(word) >= (1 << names.length) ||
683 int.parse(word) <= previous)
684 fail("bad minterm '$word'");
685 previous = int.parse(word);
686 rows.add(previous);
687 }
688 Formula? formula;
689 for (final row in rows) {
690 final pattern = bits(row, names.length);
691 Formula? term;
692 for (var i = 0; i < names.length; i++) {
693 final literal = pattern[i] == '1'
694 ? Variable(names[i])
695 : Negation(Variable(names[i]));
696 term = term == null ? literal : Binary('&', term, literal);
697 }
698 formula = formula == null ? term : Binary('|', formula, term!);
699 }
700 return define(name, formula ?? Constant(0));
701 }
702 final formula = parseFormula(
703 divided.contains(command) ? argument.substring(0, separator) : argument,
704 start,
705 definitions,
706 );
707 final back = divided.contains(command)
708 ? argument.substring(separator + 1)
709 : '';
710 Formula? second;
711 if (['MINDC', 'EQUIV', 'IMPLIES'].contains(command))
712 second = parseFormula(back, start + separator + 1, definitions);
713 if (command == 'RESTRICT') {
714 final assignments = words(back), replacements = <String, int>{};
715 if (assignments.isEmpty) fail('missing assignment');
716 for (final assignment in assignments) {
717 if (!RegExp(r'^[A-Z]=[01]$').hasMatch(assignment))
718 fail("bad assignment '$assignment'");
719 if (replacements.containsKey(assignment[0]))
720 fail('duplicate variable ${assignment[0]}');
721 replacements[assignment[0]] = int.parse(assignment[2]);
722 }
723 return ['restricted: ${notation(simplify(formula, replacements))}'];
724 }
725 if (command == 'COFACTOR') {
726 final list = words(back);
727 if (list.length != 1 || !RegExp(r'^[A-Z]$').hasMatch(list[0]))
728 fail('bad variable');
729 final name = list[0],
730 one = simplify(formula, {list[0]: 1}),
731 zero = simplify(formula, {list[0]: 0});
732 return [
733 'cofactor $name=1: ${notation(one)}',
734 'cofactor $name=0: ${notation(zero)}',
735 'difference: ${notation(simplifyBinary('^', one, zero))}',
736 ];
737 }
738 final names = variables([formula, if (second != null) second]);
739 if (command == 'PRINT') {
740 final info = measure(formula);
741 return [
742 'formula: ${notation(formula)}',
743 'variables: ${names.isEmpty ? 'none' : names.join(' ')}',
744 'size ${info.$1} depth ${info.$2}',
745 ];
746 }
747 if (command == 'NNF') return ['nnf: ${notation(nnf(formula))}'];
748 if (command == 'DUAL') return ['dual: ${notation(dual(formula))}'];
749 if (command == 'SIMPLIFY')
750 return ['simplified: ${notation(simplify(formula))}'];
751 if (command == 'CLAUSES' || command == 'DP') {
752 var set = clauses(nnf(formula), names);
753 if (command == 'CLAUSES')
754 return [
755 'clauses: ${coverText(set..sort(patternCompare), names, false)}',
756 ];
757 final output = ['start: clauses ${set.length}'],
758 empty = '-' * names.length;
759 for (var i = 0; i < names.length; i++) {
760 if (set.isEmpty || set.contains(empty)) break;
761 final positive = set.where((pattern) => pattern[i] == '1').toList(),
762 negative = set.where((pattern) => pattern[i] == '0').toList();
763 final resolvents = <String>{};
764 for (final a in positive) {
765 for (final b in negative) {
766 final value = mergeClauses(
767 '${a.substring(0, i)}-${a.substring(i + 1)}',
768 '${b.substring(0, i)}-${b.substring(i + 1)}',
769 );
770 if (value != null) resolvents.add(value);
771 }
772 }
773 set = tidy([
774 ...set.where((pattern) => pattern[i] == '-'),
775 ...resolvents,
776 ]);
777 output.add(
778 'eliminate ${names[i]}: resolvents ${resolvents.length} clauses ${set.length}',
779 );
780 }
781 output.add(
782 'result: ${set.contains(empty) ? 'unsatisfiable' : 'satisfiable'}',
783 );
784 return output;
785 }
786 if (names.length >= 9) fail('too many variables: ${names.length}');
787 if (command == 'KMAP' && (names.length < 2 || names.length > 4))
788 fail('kmap needs 2 to 4 variables');
789 final values = List.generate(
790 1 << names.length,
791 (row) => evaluate(formula, names, row),
792 );
793 final on = <int>[
794 for (var row = 0; row < values.length; row++)
795 if (values[row] == 1) row,
796 ];
797 final off = <int>[
798 for (var row = 0; row < values.length; row++)
799 if (values[row] == 0) row,
800 ];
801 if (command == 'TABLE') {
802 return [
803 '${names.isEmpty ? '' : '${names.join(' ')} '}| F',
804 for (var row = 0; row < values.length; row++)
805 '${names.isEmpty ? '' : '${bits(row, names.length).split('').join(' ')} '}| ${values[row]}',
806 ];
807 }
808 if (command == 'MINTERMS' || command == 'MAXTERMS') {
809 final rows = command == 'MINTERMS' ? on : off;
810 return [
811 '${command == 'MINTERMS' ? 'minterms' : 'maxterms'}: ${rows.isEmpty ? 'none' : rows.join(' ')}',
812 ];
813 }
814 if (command == 'DNF')
815 return [
816 'dnf: ${coverText(on.map((row) => bits(row, names.length)).toList(), names, true)}',
817 ];
818 if (command == 'CNF')
819 return [
820 'cnf: ${coverText(off.map((row) => flip(bits(row, names.length))).toList(), names, false)}',
821 ];
822 final anf = coefficients(values);
823 if (command == 'ANF') {
824 final patterns = <String>[
825 for (var row = 0; row < anf.length; row++)
826 if (anf[row] == 1) bits(row, names.length).replaceAll('0', '-'),
827 ];
828 patterns.sort((a, b) {
829 final order = literalCount(a).compareTo(literalCount(b));
830 return order != 0 ? order : patternCompare(a, b);
831 });
832 return [
833 'anf: ${patterns.isEmpty ? '0' : patterns.map((pattern) => patternText(pattern, names, true)).join(' ^ ')}',
834 ];
835 }
836 if (command == 'SAT')
837 return on.isEmpty
838 ? ['unsatisfiable']
839 : ['satisfiable: ${assignmentText(names, on.first)}'];
840 if (command == 'TAUT')
841 return off.isEmpty
842 ? ['tautology']
843 : ['not a tautology: ${assignmentText(names, off.first)}'];
844 if (command == 'COUNT') return ['models: ${on.length} of ${values.length}'];
845 if (command == 'EQUIV' || command == 'IMPLIES') {
846 for (var row = 0; row < values.length; row++) {
847 final a = values[row], b = evaluate(second!, names, row);
848 if (command == 'EQUIV' && a != b)
849 return ['not equivalent: ${assignmentText(names, row)} f=$a g=$b'];
850 if (command == 'IMPLIES' && a == 1 && b == 0)
851 return ['does not imply: ${assignmentText(names, row)}'];
852 }
853 return [command == 'EQUIV' ? 'equivalent' : 'implies'];
854 }
855 if (command == 'MINIMIZE' || command == 'MINPOS')
856 return minimize(
857 names,
858 command == 'MINPOS' ? off : on,
859 [],
860 command == 'MINPOS',
861 );
862 if (command == 'MINDC') {
863 final dc = <int>[
864 for (var row = 0; row < values.length; row++)
865 if (evaluate(second!, names, row) == 1) row,
866 ],
867 ignored = <int>{};
868 ignored.addAll(dc);
869 return minimize(
870 names,
871 on.where((row) => !ignored.contains(row)).toList(),
872 dc,
873 false,
874 );
875 }
876 if (command == 'KMAP') {
877 final p = names.length ~/ 2,
878 q = names.length - p,
879 label = '${names.sublist(0, p).join()}\\${names.sublist(p).join()}';
880 final rows = List.generate(1 << p, (i) => i ^ (i >> 1)),
881 columns = List.generate(1 << q, (i) => i ^ (i >> 1));
882 return [
883 label + columns.map((column) => ' ${bits(column, q)}').join(),
884 ...rows.map(
885 (row) =>
886 bits(row, p).padRight(label.length, ' ') +
887 columns
888 .map(
889 (column) => '${' ' * q}${values[row * (1 << q) + column]}',
890 )
891 .join(),
892 ),
893 ];
894 }
895 if (command == 'PROPS') {
896 final depends = <String>[], weights = <int, int>{};
897 var monotone = true, symmetric = true;
898 for (var i = 0; i < names.length; i++) {
899 final bit = 1 << (names.length - 1 - i);
900 var dependent = false;
901 for (var row = 0; row < values.length; row++) {
902 if (values[row] != values[row ^ bit]) dependent = true;
903 if ((row & bit) == 0 && values[row] > values[row | bit])
904 monotone = false;
905 }
906 if (dependent) depends.add(names[i]);
907 }
908 for (var row = 0; row < values.length; row++) {
909 final count = bits(
910 row,
911 names.length,
912 ).split('').where((c) => c == '1').length;
913 if (weights.containsKey(count) && weights[count] != values[row])
914 symmetric = false;
915 weights[count] = values[row];
916 }
917 var selfDual = true, linear = true;
918 for (var row = 0; row < values.length; row++) {
919 if (values[row] == values[values.length - 1 - row]) selfDual = false;
920 if (anf[row] == 1 && (row & (row - 1)) != 0) linear = false;
921 }
922 String yes(bool value) => value ? 'yes' : 'no';
923 return [
924 'depends: ${depends.isEmpty ? 'none' : depends.join(' ')}',
925 'zero-preserving: ${yes(values[0] == 0)}',
926 'one-preserving: ${yes(values.last == 1)}',
927 'self-dual: ${yes(selfDual)}',
928 'monotone: ${yes(monotone)}',
929 'linear: ${yes(linear)}',
930 'symmetric: ${yes(symmetric)}',
931 ];
932 }
933 final nodes = <BDDNode>[], unique = <String, int>{};
934 int build(int index, List<int> list) {
935 if (index == names.length) return list[0];
936 final half = list.length ~/ 2;
937 final lo = build(index + 1, list.sublist(0, half)),
938 hi = build(index + 1, list.sublist(half));
939 if (lo == hi) return lo;
940 final key = '$index,$lo,$hi';
941 if (unique.containsKey(key)) return unique[key]!;
942 final id = nodes.length + 2;
943 nodes.add(BDDNode(index, lo, hi));
944 unique[key] = id;
945 return id;
946 }
947
948 final root = build(0, values), memo = <int, int>{0: 0, 1: 1};
949 int paths(int id) {
950 if (memo.containsKey(id)) return memo[id]!;
951 final node = nodes[id - 2];
952 final count = paths(node.lo) + paths(node.hi);
953 memo[id] = count;
954 return count;
955 }
956
957 return [
958 'nodes: ${nodes.length}',
959 for (var i = 0; i < nodes.length; i++)
960 '${i + 2} = ite(${names[nodes[i].index]}, ${nodes[i].hi}, ${nodes[i].lo})',
961 'root: $root',
962 'paths: ${paths(root)}',
963 ];
964 }
