39bool validName(const string& name) {
40 return !name.empty() && name.size() <= 8 && lower(name[0])
41 && all_of(name.begin(), name.end(), [](char c) {
42 return lower(c) || digit(c);
43 });
44}

55 if (!result.empty()) result += separator;

60int precedence(const string& op) {
61 if (op == "<->") return 1;
62 if (op == "->") return 2;
63 if (op == "|") return 3;
64 if (op == "^") return 4;
65 if (op == "&") return 5;
66 return op == "~" ? 6 : 7;
67}
68string notation(const Formula& f) {
69 if (!f->left) return f->op;
70 if (f->op == "~") {
71 string child = notation(f->left);
72 return "~" + (f->left->right ? "(" + child + ")" : child);
73 }
74 string a = notation(f->left), b = notation(f->right);
75 if (f->left->right && (precedence(f->left->op) < precedence(f->op)
76 || (f->left->op == "->" && f->op == "->"))) a = "(" + a + ")";
77 if (f->right->right && (precedence(f->right->op) < precedence(f->op)
78 || (f->right->op == f->op && f->op != "->"))) b = "(" + b + ")";
79 return a + " " + f->op + " " + b;
80}
81bool equalTree(const Formula& a, const Formula& b) {
82 if (a == b) return true;
83 if (!a || !b || a->op != b->op) return false;
84 return equalTree(a->left, b->left) && equalTree(a->right, b->right);
85}
86Formula negate(const Formula& f) {
87 if (f->op == "0") return leaf("1");
88 if (f->op == "1") return leaf("0");
89 if (f->op == "~") return f->left;
90 return make("~", f);
91}
92Formula binary(const string& op, const Formula& a, const Formula& b) {
93 bool same = equalTree(a, b);
94 bool complement = (a->op == "~" && equalTree(a->left, b)) || (b->op == "~" && equalTree(b->left, a));
95 if (op == "&") {
96 if (a->op == "0" || b->op == "0") return leaf("0");
97 if (a->op == "1") return b;
98 if (b->op == "1" || same) return a;
99 if (complement) return leaf("0");
100 } else if (op == "|") {
101 if (a->op == "1" || b->op == "1") return leaf("1");
102 if (a->op == "0") return b;
103 if (b->op == "0" || same) return a;
104 if (complement) return leaf("1");
105 } else if (op == "^") {
106 if (a->op == "0") return b;
107 if (b->op == "0") return a;
108 if (a->op == "1") return negate(b);
109 if (b->op == "1") return negate(a);
110 if (same) return leaf("0");
111 if (complement) return leaf("1");
112 } else if (op == "->") {
113 if (a->op == "0" || b->op == "1") return leaf("1");
114 if (a->op == "1") return b;
115 if (b->op == "0") return negate(a);
116 if (same) return leaf("1");
117 if (complement) return b;
118 } else if (op == "<->") {
119 if (a->op == "1") return b;
120 if (b->op == "1") return a;
121 if (a->op == "0") return negate(b);
122 if (b->op == "0") return negate(a);
123 if (same) return leaf("1");
124 if (complement) return leaf("0");
125 }
126 return make(op, a, b);
127}
128Formula simplify(const Formula& f, const map<char, int>& assignment = {}) {
129 if (!f->left) {
130 auto it = assignment.find(f->op[0]);
131 return it == assignment.end() ? f : leaf(to_string(it->second));
132 }
133 auto a = simplify(f->left, assignment);
134 if (f->op == "~") return negate(a);
135 auto b = simplify(f->right, assignment);
136 return binary(f->op, a, b);
137}
138Formula nnf(const Formula& f, bool positive, map<pair<Formula, bool>, Formula>& memo) {
139 pair<Formula, bool> key{f, positive};
140 auto cached = memo.find(key);
141 if (cached != memo.end()) return cached->second;
142 const string& op = f->op;
143 Formula result;
144 if (!f->left) {
145 result = op == "0" || op == "1"
146 ? leaf(to_string((op == "1") == positive)) : positive ? f : make("~", f);
147 } else if (op == "~") {
148 result = nnf(f->left, !positive, memo);
149 } else if (op == "&" || op == "|") {
150 auto a = nnf(f->left, positive, memo), b = nnf(f->right, positive, memo);
151 result = make(positive ? op : (op == "&" ? "|" : "&"), a, b);
152 } else if (op == "->") {
153 auto a = nnf(f->left, !positive, memo), b = nnf(f->right, positive, memo);
154 result = make(positive ? "|" : "&", a, b);
155 } else {
156 bool agreement = positive == (op == "<->");
157 auto ap = nnf(f->left, true, memo), an = nnf(f->left, false, memo);
158 auto bp = nnf(f->right, true, memo), bn = nnf(f->right, false, memo);
159 result = make("|", make("&", ap, agreement ? bp : bn), make("&", an, agreement ? bn : bp));
160 }
161 memo.emplace(key, result);
162 return result;
163}
164Formula nnf(const Formula& f) {
165 map<pair<Formula, bool>, Formula> memo;
166 return nnf(f, true, memo);
167}
168Formula dual(const Formula& f) {
169 if (!f->left) return f->op == "0" ? leaf("1") : f->op == "1" ? leaf("0") : f;
170 auto a = dual(f->left);
171 if (f->op == "~") return make("~", a);
172 auto b = dual(f->right);
173 if (f->op == "->") return make("&", make("~", a), b);
174 string op = f->op == "&" ? "|" : f->op == "|" ? "&" : f->op == "^" ? "<->" : "^";
175 return make(op, a, b);
176}
177void collect(const Formula& f, set<char>& vars) {
178 if (!f->left) {
179 if (variable(f->op[0])) vars.insert(f->op[0]);
180 } else {
181 collect(f->left, vars);
182 if (f->right) collect(f->right, vars);
183 }
184}
185pair<int, int> dimensions(const Formula& f) {
186 if (!f->left) return {1, 1};
187 auto a = dimensions(f->left);
188 if (!f->right) return {a.first + 1, a.second + 1};
189 auto b = dimensions(f->right);
190 return {a.first + b.first + 1, max(a.second, b.second) + 1};
191}
192int evaluate(const Formula& f, const map<char, int>& assignment) {
193 if (!f->left) return variable(f->op[0]) ? assignment.at(f->op[0]) : f->op == "1";
194 int a = evaluate(f->left, assignment);
195 if (f->op == "~") return !a;
196 int b = evaluate(f->right, assignment);
197 if (f->op == "&") return a & b;
198 if (f->op == "^") return a ^ b;
199 if (f->op == "|") return a | b;
200 if (f->op == "->") return !a || b;
201 return a == b;
202}
203struct Token {
204 string text;
205 int position;
206};
207class Parser {
208 vector<Token> tokens;
209 size_t index = 0;
210 int endPosition;
211 const map<string, Formula>& names;
212 [[noreturn]] void error() const {
213 if (index == tokens.size()) throw runtime_error("unexpected end of formula at " + to_string(endPosition));
214 throw runtime_error("unexpected '" + tokens[index].text + "' at " + to_string(tokens[index].position));
215 }
216 Formula unary() {
217 if (index == tokens.size()) error();
218 Token token = tokens[index];
219 if (token.text == "~") {
220 ++index;
221 return make("~", unary());
222 }
223 if (token.text == "(") {
224 ++index;
225 auto result = expression(1);
226 if (index == tokens.size() || tokens[index].text != ")") error();
227 ++index;
228 return result;
229 }
230 if (variable(token.text[0]) || token.text == "0" || token.text == "1") {
231 ++index;
232 return leaf(token.text);
233 }
234 if (lower(token.text[0])) {
235 auto found = names.find(token.text);
236 if (found == names.end()) throw runtime_error("undefined name '" + token.text + "' at " + to_string(token.position));
237 ++index;
238 return found->second;
239 }
240 error();
241 }
242 Formula expression(int level) {
243 if (level == 6) return unary();
244 auto result = expression(level + 1);
245 while (index < tokens.size() && precedence(tokens[index].text) == level) {
246 string op = tokens[index++].text;
247 auto right = expression(op == "->" ? level : level + 1);
248 result = make(op, result, right);
249 }
250 return result;
251 }
252public:
253 Parser(const string& text, int start, const map<string, Formula>& definitions)
254 : endPosition(start + static_cast<int>(text.size())), names(definitions) {
255 for (size_t i = 0; i < text.size();) {
256 if (text[i] == ' ') {
257 ++i;
258 continue;
259 }
260 size_t begin = i;
261 if (text.compare(i, 3, "<->") == 0) i += 3;
262 else if (text.compare(i, 2, "->") == 0) i += 2;
263 else if (lower(text[i])) {
264 ++i;
265 while (i < text.size() && (lower(text[i]) || digit(text[i]))) ++i;
266 } else if (variable(text[i]) || string("01~&^|()").find(text[i]) != string::npos) ++i;
267 else throw runtime_error("invalid character '" + string(1, text[i]) + "' at " + to_string(start + i));
268 tokens.push_back({text.substr(begin, i - begin), start + static_cast<int>(begin)});
269 }
270 }
271 Formula parse() {
272 auto result = expression(1);
273 if (index != tokens.size()) error();
274 return result;
275 }
276};
277string pattern(int row, int n) {
278 string result(n, '0');
279 for (int i = n - 1; i >= 0; --i) {
280 result[i] += row & 1;
281 row >>= 1;
282 }
283 return result;
284}
285int literals(const string& p) {
286 return count_if(p.begin(), p.end(), [](char c) {
287 return c != '-';
288 });
289}
290int rankPattern(char c) {
291 return c == '1' ? 0 : c == '0' ? 1 : 2;
292}
293struct PatternLess {
294 bool operator()(const string& a, const string& b) const {
295 return lexicographical_compare(a.begin(), a.end(), b.begin(), b.end(), [](char x, char y) {
296 return rankPattern(x) < rankPattern(y);
297 });
298 }
299};
300using Patterns = set<string, PatternLess>;
301bool contains(const string& p, const string& row) {
302 for (size_t i = 0; i < p.size(); ++i) if (p[i] != '-' && p[i] != row[i]) return false;
303 return true;
304}
305string swapBits(string p) {
306 for (char& c : p) if (c != '-') c = c == '0' ? '1' : '0';
307 return p;
308}
309string term(const string& p, const string& vars, bool clause = false) {
310 vector<string> parts;
311 for (size_t i = 0; i < p.size(); ++i) if (p[i] != '-') parts.push_back((p[i] == '0' ? "~" : "") + string(1, vars[i]));
312 return parts.empty() ? (clause ? "0" : "1") : join(parts, clause ? " | " : " & ");
313}
314string coverNotation(const vector<string>& ps, const string& vars, bool clauses = false) {
315 vector<string> parts;
316 for (const string& p : ps) {
317 string text = term(p, vars, clauses);
318 if (clauses && ps.size() >= 2 && literals(p) >= 2) text = "(" + text + ")";
319 parts.push_back(text);
320 }
321 return parts.empty() ? (clauses ? "1" : "0") : join(parts, clauses ? " & " : " | ");
322}
323string listTerms(const Patterns& ps, const string& vars) {
324 vector<string> parts;
325 for (const string& p : ps) parts.push_back(term(p, vars));
326 return parts.empty() ? "none" : join(parts, ", ");
327}
328void minimize(ostream& out, const vector<int>& f, const vector<int>& dc, const string& vars, bool pos) {
329 Patterns stage, primes;
330 vector<string> on;
331 for (size_t r = 0; r < f.size(); ++r) {
332 bool care = dc.empty() || !dc[r];
333 if ((pos ? !f[r] : f[r]) && care) on.push_back(pattern(r, vars.size()));
334 if ((pos ? !f[r] : f[r]) || !care) stage.insert(pattern(r, vars.size()));
335 }
336 for (int k = 0;; ++k) {
337 Patterns next, combined;
338 for (auto a = stage.begin(); a != stage.end(); ++a) for (auto b = std::next(a); b != stage.end(); ++b) {
339 int differences = 0, position = -1;
340 for (size_t i = 0; i < a->size(); ++i) if ((*a)[i] != (*b)[i]) {
341 if ((*a)[i] == '-' || (*b)[i] == '-') {
342 differences = 2;
343 break;
344 }
345 ++differences;
346 position = i;
347 }
348 if (differences == 1) {
349 string merged = *a;
350 merged[position] = '-';
351 next.insert(move(merged));
352 combined.insert(*a);
353 combined.insert(*b);
354 }
355 }
356 int primeCount = 0;
357 for (const string& p : stage) if (!combined.count(p)) {
358 primes.insert(p);
359 ++primeCount;
360 }
361 if (!pos) out << "stage " << k << ": implicants " << stage.size() << " primes " << primeCount << '\n';
362 if (next.empty()) break;
363 stage = move(next);
364 }
365 Patterns target, essential;
366 for (const string& p : primes) for (const string& row : on) if (contains(p, row)) {
367 target.insert(p);
368 break;
369 }
370 for (const string& row : on) {
371 vector<string> covering;
372 for (const string& p : target) if (contains(p, row)) covering.push_back(p);
373 if (covering.size() == 1) essential.insert(covering[0]);
374 }
375 vector<string> candidates;
376 for (const string& p : target) if (!essential.count(p)) candidates.push_back(p);
377 vector<string> best;
378 int bestCount = 1000000, bestLiterals = 1000000;
379 PatternLess less;
380 for (int mask = 0; mask < (1 << candidates.size()); ++mask) {
381 vector<string> cover(essential.begin(), essential.end());
382 for (size_t i = 0; i < candidates.size(); ++i) if (mask & (1 << i)) cover.push_back(candidates[i]);
383 if (static_cast<int>(cover.size()) > bestCount) continue;
384 bool complete = true;
385 for (const string& row : on) {
386 bool covered = false;
387 for (const string& p : cover) if (contains(p, row)) {
388 covered = true;
389 break;
390 }
391 if (!covered) {
392 complete = false;
393 break;
394 }
395 }
396 if (!complete) continue;
397 int count = cover.size(), literalCount = 0;
398 for (string& p : cover) {
399 literalCount += literals(p);
400 if (pos) p = swapBits(p);
401 }
402 sort(cover.begin(), cover.end(), less);
403 if (count < bestCount || (count == bestCount && (literalCount < bestLiterals
404 || (literalCount == bestLiterals && lexicographical_compare(cover.begin(), cover.end(), best.begin(), best.end(), less))))) {
405 bestCount = count;
406 bestLiterals = literalCount;
407 best = move(cover);
408 }
409 }
410 if (!pos) {
411 out << "primes: " << listTerms(target, vars) << '\n';
412 out << "essential: " << listTerms(essential, vars) << '\n';
413 }
414 out << "minimal: " << coverNotation(best, vars, pos) << '\n';
415 out << (pos ? "clauses " : "terms ") << bestCount << " literals " << bestLiterals << '\n';
416}
417Patterns tidy(const Patterns& input) {
418 Patterns result;
419 for (const string& p : input) {
420 bool absorbed = false;
421 for (const string& q : input) if (p != q && literals(q) < literals(p)) {
422 bool subset = true;
423 for (size_t i = 0; i < p.size(); ++i) if (q[i] != '-' && q[i] != p[i]) {
424 subset = false;
425 break;
426 }
427 if (subset) {
428 absorbed = true;
429 break;
430 }
431 }
432 if (!absorbed) result.insert(p);
433 }
434 return result;
435}
436optional<string> mergeClause(const string& a, const string& b) {
437 string result = a;
438 for (size_t i = 0; i < a.size(); ++i) if (b[i] != '-') {
439 if (result[i] != '-' && result[i] != b[i]) return nullopt;
440 result[i] = b[i];
441 }
442 return result;
443}
444Patterns clauses(const Formula& f, const string& vars, map<Formula, Patterns>& memo) {
445 auto cached = memo.find(f);
446 if (cached != memo.end()) return cached->second;
447 Patterns result;
448 if (f->op == "0") {
449 result.insert(string(vars.size(), '-'));
450 } else if (f->op != "1" && (!f->left || f->op == "~")) {
451 string p(vars.size(), '-');
452 p[vars.find(f->op == "~" ? f->left->op[0] : f->op[0])] = f->op == "~" ? '0' : '1';
453 result.insert(p);
454 } else if (f->op != "1") {
455 auto a = clauses(f->left, vars, memo), b = clauses(f->right, vars, memo);
456 if (f->op == "&") {
457 result = move(a);
458 result.insert(b.begin(), b.end());
459 } else for (const string& p : a) for (const string& q : b) {
460 auto merged = mergeClause(p, q);
461 if (merged) result.insert(*merged);
462 }
463 result = tidy(result);
464 }
465 memo.emplace(f, result);
466 return result;
467}
468Patterns clauses(const Formula& f, const string& vars) {
469 map<Formula, Patterns> memo;
470 return clauses(f, vars, memo);
471}
472vector<int> coefficients(vector<int> f) {
473 for (size_t bit = 1; bit < f.size(); bit <<= 1) {
474 for (size_t row = 0; row < f.size(); ++row) if (row & bit) f[row] ^= f[row ^ bit];
475 }
476 return f;
477}
478string assignment(int row, const string& vars) {
479 string result = "{";
480 string bits = pattern(row, vars.size());
481 for (size_t i = 0; i < vars.size(); ++i) {
482 if (i) result += ' ';
483 result += string(1, vars[i]) + "=" + bits[i];
484 }
485 return result + '}';
486}
487class Workbench {
488 map<string, Formula> names;
489 void define(const string& name, Formula f, ostream& out) {
490 out << (names.count(name) ? "redefined " : "defined ") << name << " = " << notation(f) << '\n';
491 names[name] = move(f);
492 }
493public:
494 void execute(const string& line, ostream& out) {
495 size_t space = line.find(' ');
496 string command = line.substr(0, space);
497 bool hasArgument = space != string::npos;
498 string argument = hasArgument ? line.substr(space + 1) : "";
499 const set<string> commands = {"DEF", "DEFSUM", "DEL", "LIST", "PRINT", "TABLE", "MINTERMS", "MAXTERMS",
500 "DNF", "CNF", "NNF", "DUAL", "ANF", "SIMPLIFY", "RESTRICT", "COFACTOR", "MINIMIZE", "MINDC",
501 "MINPOS", "KMAP", "SAT", "TAUT", "COUNT", "EQUIV", "IMPLIES", "PROPS", "BDD", "CLAUSES", "DP"};
502 if (!commands.count(command)) throw runtime_error("unknown command " + command);
503 if (command == "LIST" && hasArgument) throw runtime_error("unexpected argument");
504 if (command != "LIST" && !hasArgument) throw runtime_error("missing argument");
505 if (command == "LIST") {
506 out << "names: " << names.size() << '\n';
507 for (const auto& entry : names) out << entry.first << " = " << notation(entry.second) << '\n';
508 return;
509 }
510 if (command == "DEL") {
511 if (!validName(argument)) throw runtime_error("invalid name '" + argument + "'");
512 if (!names.count(argument)) throw runtime_error("undefined name '" + argument + "'");
513 names.erase(argument);
514 out << "deleted " << argument << '\n';
515 return;
516 }
517 if (command == "DEF") {
518 size_t split = argument.find(' ');
519 if (split == string::npos) throw runtime_error("missing argument");
520 string name = argument.substr(0, split);
521 if (!validName(name)) throw runtime_error("invalid name '" + name + "'");
522 auto f = Parser(argument.substr(split + 1), name.size() + 6, names).parse();
523 define(name, f, out);
524 return;
525 }
526 if (command == "DEFSUM") {
527 size_t split = argument.find(';');
528 if (split == string::npos) throw runtime_error("missing ';'");
529 auto front = words(argument.substr(0, split)), back = words(argument.substr(split + 1));
530 if (front.empty()) throw runtime_error("missing argument");
531 if (!validName(front[0])) throw runtime_error("invalid name '" + front[0] + "'");
532 bool valid = front.size() >= 2;
533 for (size_t i = 1; i < front.size(); ++i) {
534 valid = valid && front[i].size() == 1 && variable(front[i][0]) && (i == 1 || front[i - 1] < front[i]);
535 }
536 if (!valid) throw runtime_error("bad variable list");
537 int n = front.size() - 1;
538 if (n >= 9) throw runtime_error("too many variables: " + to_string(n));
539 Formula f;
540 int previous = -1;
541 for (const string& m : back) {
542 bool validM = m == "0" || (!m.empty() && m[0] != '0' && all_of(m.begin(), m.end(), digit));
543 long long value = validM ? stoll(m) : -1;
544 if (!validM || value >= (1 << n) || value <= previous) throw runtime_error("bad minterm '" + m + "'");
545 previous = value;
546 string bits = pattern(value, n);
547 Formula product;
548 for (int i = 0; i < n; ++i) {
549 auto literal = leaf(front[i + 1]);
550 if (bits[i] == '0') literal = make("~", literal);
551 product = product ? make("&", product, literal) : literal;
552 }
553 f = f ? make("|", f, product) : product;
554 }
555 define(front[0], f ? f : leaf("0"), out);
556 return;
557 }
558 const set<string> splitCommands = {"RESTRICT", "COFACTOR", "MINDC", "EQUIV", "IMPLIES"};
559 string first = argument, second;
560 int firstStart = command.size() + 2, secondStart = 0;
561 if (splitCommands.count(command)) {
562 size_t split = argument.find(';');
563 if (split == string::npos) throw runtime_error("missing ';'");
564 first = argument.substr(0, split);
565 second = argument.substr(split + 1);
566 secondStart = firstStart + split + 1;
567 }
568 auto f = Parser(first, firstStart, names).parse();
569 Formula g;
570 if (command == "MINDC" || command == "EQUIV" || command == "IMPLIES") g = Parser(second, secondStart, names).parse();
571 map<char, int> substitutions;
572 char cofactor = 0;
573 if (command == "RESTRICT") {
574 auto assignments = words(second);
575 if (assignments.empty()) throw runtime_error("missing assignment");
576 for (const string& a : assignments) {
577 if (a.size() != 3 || !variable(a[0]) || a[1] != '=' || (a[2] != '0' && a[2] != '1')) throw runtime_error("bad assignment '" + a + "'");
578 if (substitutions.count(a[0])) throw runtime_error("duplicate variable " + string(1, a[0]));
579 substitutions[a[0]] = a[2] - '0';
580 }
581 } else if (command == "COFACTOR") {
582 auto vs = words(second);
583 if (vs.size() != 1 || vs[0].size() != 1 || !variable(vs[0][0])) throw runtime_error("bad variable");
584 cofactor = vs[0][0];
585 }
586 set<char> variables;
587 collect(f, variables);
588 if (g) collect(g, variables);
589 string vars(variables.begin(), variables.end());
590 int n = vars.size();
591 const set<string> tableCommands = {"TABLE", "MINTERMS", "MAXTERMS", "DNF", "CNF", "ANF", "SAT", "TAUT", "COUNT",
592 "EQUIV", "IMPLIES", "PROPS", "KMAP", "BDD", "MINIMIZE", "MINDC", "MINPOS"};
593 if (tableCommands.count(command) && n >= 9) throw runtime_error("too many variables: " + to_string(n));
594 if (command == "PRINT") {
595 vector<string> letters;
596 for (char v : vars) letters.push_back(string(1, v));
597 auto size = dimensions(f);
598 out << "formula: " << notation(f) << "\nvariables: " << (vars.empty() ? "none" : join(letters, " "))
599 << "\nsize " << size.first << " depth " << size.second << '\n';
600 } else if (command == "NNF") out << "nnf: " << notation(nnf(f)) << '\n';
601 else if (command == "DUAL") out << "dual: " << notation(dual(f)) << '\n';
602 else if (command == "SIMPLIFY") out << "simplified: " << notation(simplify(f)) << '\n';
603 else if (command == "RESTRICT") out << "restricted: " << notation(simplify(f, substitutions)) << '\n';
604 else if (command == "COFACTOR") {
605 auto one = simplify(f, {{cofactor, 1}}), zero = simplify(f, {{cofactor, 0}});
606 out << "cofactor " << cofactor << "=1: " << notation(one) << "\ncofactor " << cofactor << "=0: " << notation(zero)
607 << "\ndifference: " << notation(binary("^", one, zero)) << '\n';
608 } else if (command == "CLAUSES" || command == "DP") {
609 auto current = clauses(nnf(f), vars);
610 if (command == "CLAUSES") out << "clauses: " << coverNotation(vector<string>(current.begin(), current.end()), vars, true) << '\n';
611 else {
612 string empty(vars.size(), '-');
613 out << "start: clauses " << current.size() << '\n';
614 for (size_t i = 0; i < vars.size() && !current.empty() && !current.count(empty); ++i) {
615 Patterns positive, negative, retained, resolvents;
616 for (const string& p : current) {
617 if (p[i] == '1') positive.insert(p);
618 else if (p[i] == '0') negative.insert(p);
619 else retained.insert(p);
620 }
621 for (string p : positive) for (string q : negative) {
622 p[i] = q[i] = '-';
623 auto merged = mergeClause(p, q);
624 if (merged) resolvents.insert(*merged);
625 }
626 retained.insert(resolvents.begin(), resolvents.end());
627 current = tidy(retained);
628 out << "eliminate " << vars[i] << ": resolvents " << resolvents.size() << " clauses " << current.size() << '\n';
629 }
630 out << "result: " << (current.count(empty) ? "unsatisfiable" : "satisfiable") << '\n';
631 }
632 } else {
633 vector<int> values(1 << n), other(g ? 1 << n : 0);
634 for (int row = 0; row < (1 << n); ++row) {
635 map<char, int> assignment;
636 for (int i = 0; i < n; ++i) assignment[vars[i]] = (row >> (n - 1 - i)) & 1;
637 values[row] = evaluate(f, assignment);
638 if (g) other[row] = evaluate(g, assignment);
639 }
640 if (command == "TABLE") {
641 for (char v : vars) out << v << ' ';
642 out << "| F\n";
643 for (int r = 0; r < (1 << n); ++r) {
644 for (char bit : pattern(r, n)) out << bit << ' ';
645 out << "| " << values[r] << '\n';
646 }
647 } else if (command == "MINTERMS" || command == "MAXTERMS") {
648 vector<string> rows;
649 for (int r = 0; r < (1 << n); ++r) if (values[r] == (command == "MINTERMS")) rows.push_back(to_string(r));
650 out << (command == "MINTERMS" ? "minterms: " : "maxterms: ") << (rows.empty() ? "none" : join(rows, " ")) << '\n';
651 } else if (command == "DNF" || command == "CNF") {
652 vector<string> ps;
653 bool cnf = command == "CNF";
654 for (int r = 0; r < (1 << n); ++r) if (values[r] != cnf) ps.push_back(cnf ? swapBits(pattern(r, n)) : pattern(r, n));
655 out << (cnf ? "cnf: " : "dnf: ") << coverNotation(ps, vars, cnf) << '\n';
656 } else if (command == "ANF") {
657 auto cs = coefficients(values);
658 vector<string> ps, terms;
659 for (int r = 0; r < (1 << n); ++r) if (cs[r]) {
660 string p = pattern(r, n);
661 replace(p.begin(), p.end(), '0', '-');
662 ps.push_back(p);
663 }
664 sort(ps.begin(), ps.end(), [](const string& a, const string& b) {
665 return literals(a) != literals(b) ? literals(a) < literals(b) : PatternLess{}(a, b);
666 });
667 for (const string& p : ps) terms.push_back(term(p, vars));
668 out << "anf: " << (terms.empty() ? "0" : join(terms, " ^ ")) << '\n';
669 } else if (command == "SAT" || command == "TAUT") {
670 int wanted = command == "SAT";
671 auto found = find(values.begin(), values.end(), wanted);
672 if (command == "SAT") out << (found == values.end() ? "unsatisfiable" : "satisfiable: " + assignment(found - values.begin(), vars)) << '\n';
673 else out << (found == values.end() ? "tautology" : "not a tautology: " + assignment(found - values.begin(), vars)) << '\n';
674 } else if (command == "COUNT") out << "models: " << count(values.begin(), values.end(), 1) << " of " << values.size() << '\n';
675 else if (command == "EQUIV" || command == "IMPLIES") {
676 int witness = -1;
677 for (int r = 0; r < (1 << n); ++r) if (command == "EQUIV" ? values[r] != other[r] : values[r] && !other[r]) {
678 witness = r;
679 break;
680 }
681 if (command == "EQUIV") {
682 if (witness < 0) out << "equivalent\n";
683 else out << "not equivalent: " << assignment(witness, vars) << " f=" << values[witness] << " g=" << other[witness] << '\n';
684 } else out << (witness < 0 ? "implies" : "does not imply: " + assignment(witness, vars)) << '\n';
685 } else if (command == "PROPS") {
686 vector<string> depends;
687 bool selfDual = true, monotone = true, linear = true, symmetric = true;
688 for (int i = 0; i < n; ++i) {
689 bool dependent = false;
690 int bit = 1 << (n - 1 - i);
691 for (int r = 0; r < (1 << n); ++r) {
692 if (values[r] != values[r ^ bit]) dependent = true;
693 if (!(r & bit) && values[r] > values[r | bit]) monotone = false;
694 }
695 if (dependent) depends.push_back(string(1, vars[i]));
696 }
697 auto cs = coefficients(values);
698 map<int, int> weights;
699 for (int r = 0; r < (1 << n); ++r) {
700 if (values[r] == values[(1 << n) - 1 - r]) selfDual = false;
701 string bits = pattern(r, n);
702 int weight = count(bits.begin(), bits.end(), '1');
703 if (cs[r] && weight > 1) linear = false;
704 if (weights.count(weight) && weights[weight] != values[r]) symmetric = false;
705 weights[weight] = values[r];
706 }
707 auto yes = [](bool b) {
708 return b ? "yes" : "no";
709 };
710 out << "depends: " << (depends.empty() ? "none" : join(depends, " ")) << "\nzero-preserving: " << yes(!values.front())
711 << "\none-preserving: " << yes(values.back()) << "\nself-dual: " << yes(selfDual) << "\nmonotone: " << yes(monotone)
712 << "\nlinear: " << yes(linear) << "\nsymmetric: " << yes(symmetric) << '\n';
713 } else if (command == "KMAP") {
714 if (n < 2 || n > 4) throw runtime_error("kmap needs 2 to 4 variables");
715 int p = n / 2, q = n - p;
716 string label = vars.substr(0, p) + "\\" + vars.substr(p);
717 out << label;
718 for (int i = 0; i < (1 << q); ++i) out << ' ' << pattern(i ^ (i / 2), q);
719 out << '\n';
720 for (int i = 0; i < (1 << p); ++i) {
721 int r = i ^ (i / 2);
722 out << pattern(r, p) << string(q + 1, ' ');
723 for (int j = 0; j < (1 << q); ++j) out << string(q, ' ') << values[(r << q) | (j ^ (j / 2))];
724 out << '\n';
725 }
726 } else if (command == "BDD") {
727 vector<tuple<int, int, int>> nodes;
728 map<tuple<int, int, int>, int> unique;
729 function<int(int, int, int)> build = [&](int level, int begin, int length) {
730 if (level == n) return values[begin];
731 int lo = build(level + 1, begin, length / 2);
732 int hi = build(level + 1, begin + length / 2, length / 2);
733 if (lo == hi) return lo;
734 auto key = make_tuple(level, lo, hi);
735 auto found = unique.find(key);
736 if (found != unique.end()) return found->second;
737 int id = nodes.size() + 2;
738 unique[key] = id;
739 nodes.push_back(key);
740 return id;
741 };
742 int root = build(0, 0, values.size());
743 vector<int> paths{0, 1};
744 out << "nodes: " << nodes.size() << '\n';
745 for (size_t i = 0; i < nodes.size(); ++i) {
746 auto [level, lo, hi] = nodes[i];
747 out << i + 2 << " = ite(" << vars[level] << ", " << hi << ", " << lo << ")\n";
748 paths.push_back(paths[lo] + paths[hi]);
749 }
750 out << "root: " << root << "\npaths: " << paths[root] << '\n';
751 } else minimize(out, values, command == "MINDC" ? other : vector<int>{}, vars, command == "MINPOS");
752 }
753 }
754};
