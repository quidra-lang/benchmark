6mutable struct Formula
7 operation::String
8 children::Vector{Formula}
9end
10b_leaf(text) = Formula(string(text), Formula[])
11b_node(operation, children...) = Formula(operation, collect(children))
12b_constant(value) = b_leaf(value ? "1" : "0")
13b_isconstant(f, value) = isempty(f.children) && f.operation == (value ? "1" : "0")
14b_variable(f) = isempty(f.children) && f.operation != "0" && f.operation != "1"
15const B_PRECEDENCE = Dict("<->"=>1, "->"=>2, "|"=>3, "^"=>4, "&"=>5, "~"=>6)
16b_precedence(f) = get(B_PRECEDENCE, f.operation, 7)
17function b_equal(a, b)
18 a === b || (a.operation == b.operation && length(a.children) == length(b.children) && all(b_equal(x,y) for (x,y) in zip(a.children,b.children)))
19end
20function b_text(f)
21 isempty(f.children) && return f.operation
22 if f.operation == "~"
23 child = f.children[1]
24 text = b_text(child)
25 return "~" * (length(child.children) == 2 ? "(" * text * ")" : text)
26 end
27 a, b = f.children
28 left, right = b_text(a), b_text(b)
29 if b_precedence(a) < b_precedence(f) || a.operation == f.operation == "->"
30 left = "(" * left * ")"
31 end
32 if b_precedence(b) < b_precedence(f) || (b.operation == f.operation && f.operation != "->")
33 right = "(" * right * ")"
34 end
35 left * " " * f.operation * " " * right
36end
37function b_variables(formulas...)
38 result = Set{Char}()
39 function visit(f)
40 if b_variable(f)
41 push!(result, only(f.operation))
42 else
43 foreach(visit, f.children)
44 end
45 end
46 foreach(visit, formulas)
47 sort!(collect(result))
48end
49b_size(f) = 1 + sum(b_size(child) for child in f.children; init=0)
50b_depth(f) = 1 + maximum((b_depth(child) for child in f.children); init=0)

52struct LogicToken
53 text::String
54 position::Int
55end
56mutable struct LogicParser
57 tokens::Vector{LogicToken}
58 position::Int
59 finish::Int
60 definitions::Dict{String,Formula}

62function b_lex(text, start)
63 tokens = LogicToken[]
64 i = 1
65 while i <= length(text)
66 c = text[i]
67 if c == ' '
68 i += 1
69 continue
70 end
71 first = i
72 if i + 2 <= length(text) && text[i:i+2] == "<->"
73 i += 3
74 elseif i + 1 <= length(text) && text[i:i+1] == "->"
75 i += 2
76 elseif 'a' <= c <= 'z'
77 i += 1
78 while i <= length(text) && ('a' <= text[i] <= 'z' || '0' <= text[i] <= '9')
79 i += 1
80 end
81 elseif 'A' <= c <= 'Z' || c in "01~&^|()"
82 i += 1
83 else
84 b_fail("invalid character '$c' at $(start+i-1)")
85 end
86 push!(tokens, LogicToken(text[first:i-1], start + first - 1))
87 end
88 tokens
89end
90b_peek(p) = p.position <= length(p.tokens) ? p.tokens[p.position].text : ""
91function b_unexpected(p)
92 p.position > length(p.tokens) && b_fail("unexpected end of formula at $(p.finish)")
93 token = p.tokens[p.position]
94 b_fail("unexpected '$(token.text)' at $(token.position)")
95end
96function b_unary!(p)
97 text = b_peek(p)
98 if text == "~"
99 p.position += 1
100 return b_node("~", b_unary!(p))
101 elseif text == "("
102 p.position += 1
103 result = b_binary!(p, 1)
104 b_peek(p) == ")" || b_unexpected(p)
105 p.position += 1
106 return result
107 elseif length(text) == 1 && ('A' <= text[1] <= 'Z' || text in ("0", "1"))
108 p.position += 1
109 return b_leaf(text)
110 elseif !isempty(text) && 'a' <= text[1] <= 'z'
111 token = p.tokens[p.position]
112 haskey(p.definitions, text) || b_fail("undefined name '$text' at $(token.position)")
113 p.position += 1
114 return p.definitions[text]
115 end
116 b_unexpected(p)
117end
118function b_binary!(p, minimum)
119 left = b_unary!(p)
120 while true
121 operation = b_peek(p)
122 precedence = get(B_PRECEDENCE, operation, 0)
123 (operation == "~" || precedence < minimum) && break
124 p.position += 1
125 right = b_binary!(p, precedence + Int(operation != "->"))
126 left = b_node(operation, left, right)
127 end
128 left
129end
130function b_parse(text, start, definitions)
131 p = LogicParser(b_lex(text, start), 1, start + length(text), definitions)
132 result = b_binary!(p, 1)
133 isempty(b_peek(p)) || b_unexpected(p)
134 result
135end

137function b_nnf(formula)
138 positive, negative = IdDict{Formula,Formula}(), IdDict{Formula,Formula}()
139 function visit(f, sign)
140 memo = sign ? positive : negative
141 haskey(memo, f) && return memo[f]
142 operation = f.operation
143 if b_variable(f)
144 result = sign ? f : b_node("~", f)
145 elseif isempty(f.children)
146 result = b_constant((operation == "1") == sign)
147 elseif operation == "~"
148 result = visit(f.children[1], !sign)
149 else
150 a, b = f.children
151 if operation in ("&", "|")
152 result = b_node(sign ? operation : (operation == "&" ? "|" : "&"), visit(a,sign), visit(b,sign))
153 elseif operation == "->"
154 result = b_node(sign ? "|" : "&", visit(a,!sign), visit(b,sign))
155 else
156 equal = operation == "<->" ? sign : !sign
157 result = b_node("|", b_node("&",visit(a,true),visit(b,equal)), b_node("&",visit(a,false),visit(b,!equal)))
158 end
159 end
160 memo[f] = result
161 end
162 visit(formula, true)
163end
164function b_dual(f)
165 b_variable(f) && return f
166 isempty(f.children) && return b_constant(f.operation == "0")
167 children = b_dual.(f.children)
168 f.operation == "~" && return b_node("~", children[1])
169 f.operation == "->" && return b_node("&", b_node("~",children[1]), children[2])
170 operation = Dict("&"=>"|", "|"=>"&", "^"=>"<->", "<->"=>"^")[f.operation]
171 b_node(operation, children...)
172end
173function b_negate(f)
174 b_isconstant(f,false) && return b_leaf("1")
175 b_isconstant(f,true) && return b_leaf("0")
176 f.operation == "~" ? f.children[1] : b_node("~",f)
177end
178function b_simplify_binary(operation, a, b)
179 zeroa, zerob = b_isconstant(a,false), b_isconstant(b,false)
180 onea, oneb = b_isconstant(a,true), b_isconstant(b,true)
181 equal = b_equal(a,b)
182 complementary = (a.operation == "~" && b_equal(a.children[1],b)) || (b.operation == "~" && b_equal(b.children[1],a))
183 if operation == "&"
184 (zeroa || zerob) && return b_leaf("0")
185 onea && return b
186 oneb && return a
187 equal && return a
188 complementary && return b_leaf("0")
189 elseif operation == "|"
190 (onea || oneb) && return b_leaf("1")
191 zeroa && return b
192 zerob && return a
193 equal && return a
194 complementary && return b_leaf("1")
195 elseif operation == "^"
196 zeroa && return b
197 zerob && return a
198 onea && return b_negate(b)
199 oneb && return b_negate(a)
200 equal && return b_leaf("0")
201 complementary && return b_leaf("1")
202 elseif operation == "->"
203 (zeroa || oneb) && return b_leaf("1")
204 onea && return b
205 zerob && return b_negate(a)
206 equal && return b_leaf("1")
207 complementary && return b
208 else
209 onea && return b
210 oneb && return a
211 zeroa && return b_negate(b)
212 zerob && return b_negate(a)
213 equal && return b_leaf("1")
214 complementary && return b_leaf("0")
215 end
216 b_node(operation,a,b)
217end
218function b_simplify(f, assignments=Dict{Char,Bool}())
219 if isempty(f.children)
220 return b_variable(f) && haskey(assignments,only(f.operation)) ? b_constant(assignments[only(f.operation)]) : f
221 elseif f.operation == "~"
222 return b_negate(b_simplify(f.children[1],assignments))
223 end
224 b_simplify_binary(f.operation,b_simplify(f.children[1],assignments),b_simplify(f.children[2],assignments))
225end
226function b_evaluate(f, variables, row)
227 operation = f.operation
228 operation == "0" && return false
229 operation == "1" && return true
230 if b_variable(f)
231 index = findfirst(==(only(operation)),variables)
232 return (row & (1 << (length(variables)-index))) != 0
233 end
234 a = b_evaluate(f.children[1],variables,row)
235 operation == "~" && return !a
236 b = b_evaluate(f.children[2],variables,row)
237 operation == "&" ? a && b : operation == "|" ? a || b : operation == "^" ? xor(a,b) : operation == "->" ? !a || b : a == b
238end
239b_truth(f, variables) = [b_evaluate(f,variables,row) for row in 0:(1<<length(variables))-1]
240b_pattern(row, n) = join(((row >> i) & 1 for i in n-1:-1:0))
241b_pattern_key(pattern) = replace(pattern,'1'=>'0','0'=>'1','-'=>'2')
242b_sorted(patterns) = sort!(collect(patterns); by=b_pattern_key)
243b_literals(pattern) = count(!=('-'),pattern)
244b_flip(pattern) = replace(pattern,'1'=>'0','0'=>'1')
245function b_term(pattern, variables; clause=false)
246 literals = [string(c == '0' ? "~" : "",variables[i]) for (i,c) in enumerate(pattern) if c != '-']
247 isempty(literals) ? (clause ? "0" : "1") : join(literals,clause ? " | " : " & ")
248end
249function b_normal_text(patterns, variables; clause=false)
250 isempty(patterns) && return clause ? "1" : "0"
251 texts = [b_term(p,variables;clause=clause) for p in patterns]
252 if clause && length(patterns) >= 2
253 texts = [b_literals(p) >= 2 ? "(" * text * ")" : text for (p,text) in zip(patterns,texts)]
254 end
255 join(texts,clause ? " & " : " | ")
256end
257b_assignment(row,variables) = "{" * join((string(v,"=",(row >> (length(variables)-i)) & 1) for (i,v) in enumerate(variables))," ") * "}"
258function b_anf(values, n)
259 coefficients = copy(values)
260 for bit in 0:n-1, mask in 0:length(values)-1
261 mask & (1<<bit) != 0 && (coefficients[mask+1] = xor(coefficients[mask+1],coefficients[(mask ⊻ (1<<bit))+1]))
262 end
263 coefficients
264end
265function b_anf_text(values, variables)
266 coefficients = b_anf(values,length(variables))
267 patterns = [replace(b_pattern(row,length(variables)),'0'=>'-') for row in 0:length(values)-1 if coefficients[row+1]]
268 sort!(patterns; by=p -> (b_literals(p),b_pattern_key(p)))
269 isempty(patterns) ? "0" : join((b_term(p,variables) for p in patterns)," ^ ")
270end
271function b_covers(pattern, row, n)
272 all(c == '-' || Int(c-'0') == ((row >> (n-i)) & 1) for (i,c) in enumerate(pattern))
273end
274function b_minimize(on, dc, variables; product=false)
275 n = length(variables)
276 stage = Set(b_pattern(row,n) for row in union(on,dc))
277 primes = Set{String}()
278 lines = String[]
279 number = 0
280 while true
281 patterns = collect(stage)
282 combined = Set{String}()
283 next = Set{String}()
284 for i in eachindex(patterns), j in i+1:length(patterns)
285 a, b = patterns[i], patterns[j]
286 positions = [k for k in 1:n if a[k] != b[k]]
287 if length(positions) == 1
288 k = only(positions)
289 if a[k] != '-' && b[k] != '-'
290 chars = collect(a)
291 chars[k] = '-'
292 push!(next,String(chars))
293 push!(combined,a,b)
294 end
295 end
296 end
297 uncombined = setdiff(stage,combined)
298 union!(primes,uncombined)
299 !product && push!(lines,"stage $number: implicants $(length(stage)) primes $(length(uncombined))")
300 isempty(next) && break
301 stage = next
302 number += 1
303 end
304 target = b_sorted(p for p in primes if any(row -> b_covers(p,row,n),on))
305 essential = Set{Int}()
306 coverage = Dict(row => [i for i in eachindex(target) if b_covers(target[i],row,n)] for row in on)
307 for indices in values(coverage)
308 length(indices) == 1 && push!(essential,only(indices))
309 end
310 optional = [i for i in eachindex(target) if !(i in essential)]
311 best = Int[]
312 best_score = nothing
313 for mask in 0:(1<<length(optional))-1
314 chosen = vcat(collect(essential),[optional[i] for i in eachindex(optional) if mask & (1<<(i-1)) != 0])
315 all(any(i -> i in chosen,coverage[row]) for row in on) || continue
316 patterns = b_sorted(product ? b_flip(target[i]) : target[i] for i in chosen)
317 score = (length(chosen),sum(b_literals(p) for p in patterns;init=0),Tuple(b_pattern_key.(patterns)))
318 if best_score === nothing || isless(score,best_score)
319 best, best_score = chosen, score
320 end
321 end
322 cover = b_sorted(product ? b_flip(target[i]) : target[i] for i in best)
323 if product
324 push!(lines,"minimal: " * b_normal_text(cover,variables;clause=true))
325 push!(lines,"clauses $(length(cover)) literals $(sum(b_literals(p) for p in cover;init=0))")
326 else
327 essentials = b_sorted(target[i] for i in essential)
328 push!(lines,"primes: " * (isempty(target) ? "none" : join((b_term(p,variables) for p in target),", ")))
329 push!(lines,"essential: " * (isempty(essentials) ? "none" : join((b_term(p,variables) for p in essentials),", ")))
330 push!(lines,"minimal: " * b_normal_text(cover,variables))
331 push!(lines,"terms $(length(cover)) literals $(sum(b_literals(p) for p in cover;init=0))")
332 end
333 lines
334end
335function b_merge_clause(a,b)
336 result = collect(a)
337 for i in eachindex(result)
338 b[i] == '-' && continue
339 result[i] != '-' && result[i] != b[i] && return nothing
340 result[i] = b[i]
341 end
342 String(result)
343end
344function b_tidy(patterns)
345 unique_patterns = unique(patterns)
346 [p for p in unique_patterns if !any(q -> q != p && all(q[i] == '-' || q[i] == p[i] for i in 1:length(p)),unique_patterns)]
347end
348function b_clauses(formula,variables)
349 memo = IdDict{Formula,Vector{String}}()
350 function visit(f)
351 haskey(memo,f) && return memo[f]
352 if f.operation == "1"
353 result = String[]
354 elseif f.operation == "0"
355 result = [repeat("-",length(variables))]
356 elseif b_variable(f) || f.operation == "~"
357 variable = only(f.operation == "~" ? f.children[1].operation : f.operation)
358 result = [join((v == variable ? (f.operation == "~" ? '0' : '1') : '-' for v in variables))]
359 else
360 left, right = visit(f.children[1]), visit(f.children[2])
361 if f.operation == "&"
362 result = b_tidy(vcat(left,right))
363 else
364 result = String[]
365 for a in left, b in right
366 merged = b_merge_clause(a,b)
367 merged !== nothing && push!(result,merged)
368 end
369 result = b_tidy(result)
370 end
371 end
372 memo[f] = result
373 end
374 visit(b_nnf(formula))
375end
376function b_dp(formula,variables)
377 clauses = b_clauses(formula,variables)
378 empty_clause = repeat("-",length(variables))
379 lines = ["start: clauses $(length(clauses))"]
380 for (i,variable) in enumerate(variables)
381 (isempty(clauses) || empty_clause in clauses) && break
382 positive = [p for p in clauses if p[i] == '1']
383 negative = [p for p in clauses if p[i] == '0']
384 resolvents = Set{String}()
385 for p in positive, q in negative
386 a, b = collect(p), collect(q)
387 a[i] = b[i] = '-'
388 merged = b_merge_clause(String(a),String(b))
389 merged !== nothing && push!(resolvents,merged)
390 end
391 clauses = b_tidy(vcat([p for p in clauses if p[i] == '-'],collect(resolvents)))
392 push!(lines,"eliminate $variable: resolvents $(length(resolvents)) clauses $(length(clauses))")
393 end
394 push!(lines,"result: " * (empty_clause in clauses ? "unsatisfiable" : "satisfiable"))
395 lines
396end
397function b_bdd(values,variables)
398 nodes = Tuple{Int,Int,Int}[]
399 unique_nodes = Dict{Tuple{Int,Int,Int},Int}()
400 function build(level, first, amount)
401 amount == 1 && return Int(values[first])
402 half = div(amount,2)
403 low = build(level+1,first,half)
404 high = build(level+1,first+half,half)
405 low == high && return low
406 key = (level,low,high)
407 if !haskey(unique_nodes,key)
408 push!(nodes,key)
409 unique_nodes[key] = length(nodes)+1
410 end
411 unique_nodes[key]
412 end
413 root = build(1,1,length(values))
414 lines = ["nodes: $(length(nodes))"]
415 paths = [0,1]
416 for (i,(level,low,high)) in enumerate(nodes)
417 push!(lines,"$(i+1) = ite($(variables[level]), $high, $low)")
418 push!(paths,paths[low+1]+paths[high+1])
419 end
420 push!(lines,"root: $root","paths: $(paths[root+1])")
421 lines
422end
423function b_properties(values,variables)
424 n = length(variables)
425 depends = Char[]
426 monotone = true
427 for (i,variable) in enumerate(variables)
428 bit = 1 << (n-i)
429 any(values[row+1] != values[(row ⊻ bit)+1] for row in 0:length(values)-1) && push!(depends,variable)
430 for row in 0:length(values)-1
431 row & bit == 0 && values[row+1] && !values[(row|bit)+1] && (monotone=false)
432 end
433 end
434 coefficients = b_anf(values,n)
435 linear = all(!coefficients[row+1] || count_ones(row) <= 1 for row in 0:length(values)-1)
436 symmetric = all(length(unique(values[row+1] for row in 0:length(values)-1 if count_ones(row) == k)) <= 1 for k in 0:n)
437 yesno(value) = value ? "yes" : "no"
438 ["depends: " * (isempty(depends) ? "none" : join(depends," ")),
439 "zero-preserving: " * yesno(!first(values)),
440 "one-preserving: " * yesno(last(values)),
441 "self-dual: " * yesno(all(values[i] != values[end-i+1] for i in eachindex(values))),
442 "monotone: " * yesno(monotone),
443 "linear: " * yesno(linear),
444 "symmetric: " * yesno(symmetric)]
445end
446const B_SINGLE = Set(split("PRINT TABLE MINTERMS MAXTERMS DNF CNF NNF DUAL ANF SIMPLIFY MINIMIZE MINPOS KMAP SAT TAUT COUNT PROPS BDD CLAUSES DP"))
447const B_SPLIT = Set(split("RESTRICT COFACTOR MINDC EQUIV IMPLIES"))
448const B_TRUTH = Set(split("TABLE MINTERMS MAXTERMS DNF CNF ANF SAT TAUT COUNT EQUIV IMPLIES PROPS KMAP BDD MINIMIZE MINDC MINPOS"))
449b_name(name) = occursin(r"^[a-z][a-z0-9]{0,7}$",name)
450b_words(text) = String.(split(text,' ';keepempty=false))
451function b_define!(definitions,name,formula)
452 prefix = haskey(definitions,name) ? "redefined " : "defined "
453 result = prefix * name * " = " * b_text(formula)

455 [result]

457function b_sum_formula(rows,variables)
458 terms = Formula[]
459 for row in rows
460 literals = [((row >> (length(variables)-i)) & 1) == 1 ? b_leaf(variable) : b_node("~",b_leaf(variable)) for (i,variable) in enumerate(variables)]
461 push!(terms,reduce((a,b)->b_node("&",a,b),literals))
462 end
463 isempty(terms) ? b_leaf("0") : reduce((a,b)->b_node("|",a,b),terms)
464end
465function b_command!(definitions,line)
466 at = findfirst(' ',line)
467 command = at === nothing ? line : line[1:at-1]
468 argument = at === nothing ? "" : line[at+1:end]
469 command in union(B_SINGLE,B_SPLIT,Set(["DEF","DEFSUM","DEL","LIST"])) || b_fail("unknown command $command")
470 command == "LIST" && at !== nothing && b_fail("unexpected argument")
471 command != "LIST" && at === nothing && b_fail("missing argument")
472 if command == "LIST"
473 return vcat(["names: $(length(definitions))"],[name * " = " * b_text(definitions[name]) for name in sort!(collect(keys(definitions)))])
474 elseif command == "DEL"
475 b_name(argument) || b_fail("invalid name '$argument'")
476 haskey(definitions,argument) || b_fail("undefined name '$argument'")
477 delete!(definitions,argument)
478 return ["deleted $argument"]
479 elseif command == "DEF"
480 split_at = findfirst(' ',argument)
481 split_at === nothing && b_fail("missing argument")
482 name = argument[1:split_at-1]
483 b_name(name) || b_fail("invalid name '$name'")
484 formula = b_parse(argument[split_at+1:end],length(name)+6,definitions)
485 return b_define!(definitions,name,formula)
486 elseif command == "DEFSUM"
487 semicolon = findfirst(';',argument)
488 semicolon === nothing && b_fail("missing ';'")
489 words = b_words(argument[1:semicolon-1])
490 isempty(words) && b_fail("missing argument")
491 name = first(words)
492 b_name(name) || b_fail("invalid name '$name'")
493 variables = words[2:end]
494 valid = !isempty(variables) && all(word -> length(word) == 1 && 'A' <= word[1] <= 'Z',variables)
495 valid = valid && all(variables[i] < variables[i+1] for i in 1:length(variables)-1)
496 valid || b_fail("bad variable list")
497 length(variables) >= 9 && b_fail("too many variables: $(length(variables))")
498 rows = Int[]
499 for word in b_words(argument[semicolon+1:end])
500 occursin(r"^(0|[1-9][0-9]*)$",word) || b_fail("bad minterm '$word'")
501 value = parse(Int,word)
502 (value >= (1<<length(variables)) || (!isempty(rows) && value <= last(rows))) && b_fail("bad minterm '$word'")
503 push!(rows,value)
504 end
505 return b_define!(definitions,name,b_sum_formula(rows,only.(variables)))
506 end
507 front, back = argument, ""
508 second_start = 0
509 if command in B_SPLIT
510 semicolon = findfirst(';',argument)
511 semicolon === nothing && b_fail("missing ';'")
512 front, back = argument[1:semicolon-1],argument[semicolon+1:end]
513 second_start = length(command)+2+semicolon
514 end
515 formula = b_parse(front,length(command)+2,definitions)
516 second = command in ("MINDC","EQUIV","IMPLIES") ? b_parse(back,second_start,definitions) : nothing
517 if command == "RESTRICT"
518 words = b_words(back)
519 isempty(words) && b_fail("missing assignment")
520 assignments = Dict{Char,Bool}()
521 for word in words
522 occursin(r"^[A-Z]=[01]$",word) || b_fail("bad assignment '$word'")
523 variable = word[1]
524 haskey(assignments,variable) && b_fail("duplicate variable $variable")
525 assignments[variable] = word[3] == '1'
526 end
527 return ["restricted: " * b_text(b_simplify(formula,assignments))]
528 elseif command == "COFACTOR"
529 words = b_words(back)
530 (length(words) == 1 && length(words[1]) == 1 && 'A' <= words[1][1] <= 'Z') || b_fail("bad variable")
531 variable = only(words[1])
532 high = b_simplify(formula,Dict(variable=>true))
533 low = b_simplify(formula,Dict(variable=>false))
534 return ["cofactor $variable=1: " * b_text(high),"cofactor $variable=0: " * b_text(low),"difference: " * b_text(b_simplify_binary("^",high,low))]
535 end
536 variables = second === nothing ? b_variables(formula) : b_variables(formula,second)
537 command in B_TRUTH && length(variables) >= 9 && b_fail("too many variables: $(length(variables))")
538 if command == "PRINT"
539 return ["formula: " * b_text(formula),"variables: " * (isempty(variables) ? "none" : join(variables," ")),"size $(b_size(formula)) depth $(b_depth(formula))"]
540 elseif command == "NNF"
541 return ["nnf: " * b_text(b_nnf(formula))]
542 elseif command == "DUAL"
543 return ["dual: " * b_text(b_dual(formula))]
544 elseif command == "SIMPLIFY"
545 return ["simplified: " * b_text(b_simplify(formula))]
546 elseif command == "CLAUSES"
547 return ["clauses: " * b_normal_text(b_sorted(b_clauses(formula,variables)),variables;clause=true)]
548 elseif command == "DP"
549 return b_dp(formula,variables)
550 end
551 values = b_truth(formula,variables)
552 n = length(variables)
553 rows = collect(0:length(values)-1)
554 on = [row for row in rows if values[row+1]]
555 off = [row for row in rows if !values[row+1]]
556 if command == "TABLE"
557 lines = [join(vcat(string.(variables),["|","F"])," ")]
558 append!(lines,[join(vcat(string.(collect(b_pattern(row,n))),["|",string(Int(values[row+1]))])," ") for row in rows])
559 return lines
560 elseif command in ("MINTERMS","MAXTERMS")
561 selected = command == "MINTERMS" ? on : off
562 return [lowercase(command) * ": " * (isempty(selected) ? "none" : join(selected," "))]
563 elseif command == "DNF"
564 return ["dnf: " * b_normal_text([b_pattern(row,n) for row in on],variables)]
565 elseif command == "CNF"
566 return ["cnf: " * b_normal_text([b_flip(b_pattern(row,n)) for row in off],variables;clause=true)]
567 elseif command == "ANF"
568 return ["anf: " * b_anf_text(values,variables)]
569 elseif command == "SAT"
570 return [isempty(on) ? "unsatisfiable" : "satisfiable: " * b_assignment(first(on),variables)]
571 elseif command == "TAUT"
572 return [isempty(off) ? "tautology" : "not a tautology: " * b_assignment(first(off),variables)]
573 elseif command == "COUNT"
574 return ["models: $(length(on)) of $(length(values))"]
575 elseif command in ("EQUIV","IMPLIES")
576 other = b_truth(second,variables)
577 mismatch = findfirst(row -> command == "EQUIV" ? values[row+1] != other[row+1] : values[row+1] && !other[row+1],rows)
578 mismatch === nothing && return [command == "EQUIV" ? "equivalent" : "implies"]
579 row = rows[mismatch]
580 return [command == "EQUIV" ? "not equivalent: " * b_assignment(row,variables) * " f=$(Int(values[row+1])) g=$(Int(other[row+1]))" : "does not imply: " * b_assignment(row,variables)]
581 elseif command == "PROPS"
582 return b_properties(values,variables)
583 elseif command == "BDD"
584 return b_bdd(values,variables)
585 elseif command in ("MINIMIZE","MINDC","MINPOS")
586 dc = second === nothing ? Int[] : [row for row in rows if b_evaluate(second,variables,row)]
587 return b_minimize(command == "MINPOS" ? off : setdiff(on,dc),dc,variables;product=command == "MINPOS")
588 elseif command == "KMAP"
589 2 <= n <= 4 || b_fail("kmap needs 2 to 4 variables")
590 p, q = div(n,2),n-div(n,2)
591 row_codes = [xor(i,div(i,2)) for i in 0:(1<<p)-1]
592 column_codes = [xor(i,div(i,2)) for i in 0:(1<<q)-1]
593 label = join(variables[1:p]) * "\\" * join(variables[p+1:end])
594 lines = [label * " " * join((b_pattern(c,q) for c in column_codes)," ")]
595 for row in row_codes
596 text = b_pattern(row,p) * repeat(" ",length(label)-p)
597 for column in column_codes
598 text *= repeat(" ",q) * string(Int(values[(row<<q)+column+1]))
599 end
600 push!(lines,text)
601 end
602 return lines
603 end
604 String[]
605end
