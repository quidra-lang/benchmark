33function notation(array $node): string
34{
35 if ($node[0] === 'variable' || $node[0] === 'constant') {
36 return (string)$node[1];
37 }
38 if ($node[0] === 'not') {
39 $text = notation($node[1]);
40 return '~' . (isBinary($node[1]) ? '(' . $text . ')' : $text);
41 }
42 [$op, $a, $b] = $node;
43 $left = notation($a);
44 $right = notation($b);
45 if (isBinary($a) && (PRECEDENCE[$a[0]] < PRECEDENCE[$op] || ($a[0] === '->' && $op === '->'))) {
46 $left = '(' . $left . ')';
47 }
48 if (isBinary($b) && (PRECEDENCE[$b[0]] < PRECEDENCE[$op] || ($b[0] === $op && $op !== '->'))) {
49 $right = '(' . $right . ')';
50 }
51 return $left . ' ' . $op . ' ' . $right;
52}

54function variableNames(array $node): array
55{
56 if ($node[0] === 'variable') {
57 return [$node[1]];
58 }
59 if ($node[0] === 'constant') {
60 return [];
61 }
62 if ($node[0] === 'not') {
63 return variableNames($node[1]);

70function measures(array $node): array
71{
72 if ($node[0] === 'variable' || $node[0] === 'constant') {
73 return [1, 1];
74 }
75 $a = measures($node[1]);
76 if ($node[0] === 'not') {
77 return [$a[0] + 1, $a[1] + 1];
78 }
79 $b = measures($node[2]);
80 return [$a[0] + $b[0] + 1, max($a[1], $b[1]) + 1];
81}

83class FormulaParser
84{
85 private array $tokens = [];
86 private int $position = 0;
87
88 public function __construct(string $text, int $start, private array $definitions)
89 {
90 $i = 0;
91 while ($i < strlen($text)) {
92 $c = $text[$i];
93 if ($c === ' ') {
94 $i++;
95 continue;
96 }
97 $from = $i;
98 if (substr($text, $i, 3) === '<->') {
99 $i += 3;
100 } elseif (substr($text, $i, 2) === '->') {
101 $i += 2;
102 } elseif ($c >= 'a' && $c <= 'z') {
103 $i++;
104 while ($i < strlen($text) && preg_match('/[a-z0-9]/', $text[$i])) {
105 $i++;
106 }
107 } elseif (str_contains('ABCDEFGHIJKLMNOPQRSTUVWXYZ01~&^|()', $c)) {
108 $i++;
109 } else {
110 throw new LogicError("invalid character '" . $c . "' at " . ($start + $i));
111 }
112 $this->tokens[] = [substr($text, $from, $i - $from), $start + $from];
113 }
114 $this->tokens[] = ['', $start + strlen($text)];
115 }
116
117 private function peek(): string
118 {
119 return $this->tokens[$this->position][0];
120 }
121
122 private function advance(): string
123 {
124 return $this->tokens[$this->position++][0];
125 }
126
127 private function unexpected(): never
128 {
129 [$text, $position] = $this->tokens[$this->position];
130 throw new LogicError($text === '' ? 'unexpected end of formula at ' . $position : "unexpected '" . $text . "' at " . $position);
131 }
132
133 private function unary(): array
134 {
135 if ($this->peek() === '~') {
136 $this->advance();
137 return boolNot($this->unary());
138 }
139 if ($this->peek() === '(') {
140 $this->advance();
141 $result = $this->expression(0);
142 if ($this->peek() !== ')') {
143 $this->unexpected();
144 }
145 $this->advance();
146 return $result;
147 }
148 if (preg_match('/\A[A-Z]\z/', $this->peek())) {
149 return boolVar($this->advance());
150 }
151 if ($this->peek() === '0' || $this->peek() === '1') {
152 return boolConst((int)$this->advance());
153 }
154 if (preg_match('/\A[a-z][a-z0-9]*\z/', $this->peek())) {
155 [$text, $position] = $this->tokens[$this->position];
156 if (!isset($this->definitions[$text])) {
157 throw new LogicError("undefined name '" . $text . "' at " . $position);
158 }
159 $this->advance();
160 return $this->definitions[$text];
161 }
162 $this->unexpected();
163 }
164
165 private function expression(int $level): array
166 {
167 if ($level === 5) {
168 return $this->unary();
169 }
170 $op = ['<->', '->', '|', '^', '&'][$level];
171 $result = $this->expression($level + 1);
172 while ($this->peek() === $op) {
173 $this->advance();
174 $result = boolBinary($op, $result, $this->expression($op === '->' ? $level : $level + 1));
175 }
176 return $result;
177 }
178
179 public function parse(): array
180 {
181 $result = $this->expression(0);
182 if ($this->peek() !== '') {
183 $this->unexpected();
184 }
185 return $result;
186 }
187}

189function nnf(array $node, bool $positive = true): array
190{
191 $memo = [];
192 $serial = 0;
193 $visit = function (array $current, bool $sign) use (&$visit, &$memo, &$serial): array {
194 $key = ($sign ? 'p' : 'n') . serialize($current);
195 if (isset($memo[$key])) {
196 return $memo[$key];
197 }
198 if ($current[0] === 'variable') {
199 $result = $sign ? $current : boolNot($current);
200 } elseif ($current[0] === 'constant') {
201 $result = boolConst($sign ? $current[1] : 1 - $current[1]);
202 } elseif ($current[0] === 'not') {
203 $result = $visit($current[1], !$sign);
204 } elseif ($current[0] === '&' || $current[0] === '|') {
205 $op = $sign ? $current[0] : ($current[0] === '&' ? '|' : '&');
206 $result = boolBinary($op, $visit($current[1], $sign), $visit($current[2], $sign));
207 } elseif ($current[0] === '->') {
208 $result = boolBinary($sign ? '|' : '&', $visit($current[1], !$sign), $visit($current[2], $sign));
209 } else {
210 $same = $current[0] === '<->' ? $sign : !$sign;
211 $result = boolBinary('|',
212 boolBinary('&', $visit($current[1], true), $visit($current[2], $same)),
213 boolBinary('&', $visit($current[1], false), $visit($current[2], !$same)));
214 }
215 $result['memo_id'] = ++$serial;
216 $memo[$key] = $result;
217 return $result;
218 };
219 return $visit($node, $positive);
220}

222function dual(array $node): array
223{
224 return match ($node[0]) {
225 'variable' => $node,
226 'constant' => boolConst(1 - $node[1]),
227 'not' => boolNot(dual($node[1])),
228 '->' => boolBinary('&', boolNot(dual($node[1])), dual($node[2])),
229 default => boolBinary(['&' => '|', '|' => '&', '^' => '<->', '<->' => '^'][$node[0]], dual($node[1]), dual($node[2]))
230 };
231}

233function reduceNegation(array $node): array
234{
235 if ($node[0] === 'constant') {
236 return boolConst(1 - $node[1]);
237 }
238 return $node[0] === 'not' ? $node[1] : boolNot($node);

241function simplifyBinary(string $op, array $a, array $b): array
242{
243 $zero = boolConst(0);
244 $one = boolConst(1);
245 $equal = $a === $b;
246 $complementary = $a === boolNot($b) || $b === boolNot($a);
247 switch ($op) {
248 case '&':
249 if ($a === $zero || $b === $zero) {
250 return $zero;
251 }
252 if ($a === $one) {
253 return $b;
254 }
255 if ($b === $one || $equal) {
256 return $a;
257 }
258 if ($complementary) {
259 return $zero;
260 }
261 break;
262 case '|':
263 if ($a === $one || $b === $one) {
264 return $one;
265 }
266 if ($a === $zero) {
267 return $b;
268 }
269 if ($b === $zero || $equal) {
270 return $a;
271 }
272 if ($complementary) {
273 return $one;
274 }
275 break;
276 case '^':
277 if ($a === $zero) {
278 return $b;
279 }
280 if ($b === $zero) {
281 return $a;
282 }
283 if ($a === $one) {
284 return reduceNegation($b);
285 }
286 if ($b === $one) {
287 return reduceNegation($a);
288 }
289 if ($equal) {
290 return $zero;
291 }
292 if ($complementary) {
293 return $one;
294 }
295 break;
296 case '->':
297 if ($a === $zero || $b === $one) {
298 return $one;
299 }
300 if ($a === $one) {
301 return $b;
302 }
303 if ($b === $zero) {
304 return reduceNegation($a);
305 }
306 if ($equal) {
307 return $one;
308 }
309 if ($complementary) {
310 return $b;
311 }
312 break;
313 case '<->':
314 if ($a === $one) {
315 return $b;
316 }
317 if ($b === $one) {
318 return $a;
319 }
320 if ($a === $zero) {
321 return reduceNegation($b);
322 }
323 if ($b === $zero) {
324 return reduceNegation($a);
325 }
326 if ($equal) {
327 return $one;
328 }
329 if ($complementary) {
330 return $zero;
331 }
332 break;
333 }
334 return boolBinary($op, $a, $b);
335}

337function simplify(array $node, array $assignments = []): array
338{
339 if ($node[0] === 'variable') {
340 return isset($assignments[$node[1]]) ? boolConst($assignments[$node[1]]) : $node;
341 }
342 if ($node[0] === 'constant') {
343 return $node;
344 }
345 if ($node[0] === 'not') {
346 return reduceNegation(simplify($node[1], $assignments));
347 }
348 return simplifyBinary($node[0], simplify($node[1], $assignments), simplify($node[2], $assignments));
349}

351function evaluate(array $node, array $assignments): int
352{
353 if ($node[0] === 'variable') {
354 return $assignments[$node[1]];
355 }
356 if ($node[0] === 'constant') {
357 return $node[1];
358 }
359 if ($node[0] === 'not') {
360 return 1 - evaluate($node[1], $assignments);
361 }
362 $a = evaluate($node[1], $assignments);
363 $b = evaluate($node[2], $assignments);
364 return match ($node[0]) {
365 '&' => $a & $b, '|' => $a | $b, '^' => $a ^ $b,
366 '->' => $a === 1 && $b === 0 ? 0 : 1,
367 '<->' => $a === $b ? 1 : 0
368 };
369}

401function patternPart(string $text, array $names, bool $product = true): string
402{
403 $literals = [];
404 for ($i = 0; $i < strlen($text); $i++) {
405 if ($text[$i] !== '-') {
406 $literals[] = ($text[$i] === '1' ? '' : '~') . $names[$i];
407 }
408 }
409 return $literals === [] ? ($product ? '1' : '0') : implode($product ? ' & ' : ' | ', $literals);

412function coverText(array $patterns, array $names, bool $product = true): string
413{
414 if ($patterns === []) {
415 return $product ? '0' : '1';
416 }
417 $parts = [];
418 foreach ($patterns as $text) {
419 $part = patternPart($text, $names, $product);
420 $parts[] = !$product && count($patterns) >= 2 && literalCount($text) >= 2 ? '(' . $part . ')' : $part;
421 }
422 return implode($product ? ' | ' : ' & ', $parts);
423}

431function coefficients(array $values, int $size): array
432{
433 for ($i = 0; $i < $size; $i++) {
434 $bit = 1 << $i;
435 for ($row = 0; $row < count($values); $row++) {
436 if (($row & $bit) !== 0) {
437 $values[$row] ^= $values[$row ^ $bit];
438 }
439 }
440 }
441 return $values;
442}

444function coverKeyCompare(array $a, array $b): int
445{
446 if ($a[0] !== $b[0]) {
447 return $a[0] <=> $b[0];
448 }
449 if ($a[1] !== $b[1]) {
450 return $a[1] <=> $b[1];
451 }
452 foreach ($a[2] as $i => $key) {
453 $comparison = strcmp($key, $b[2][$i]);
454 if ($comparison !== 0) {
455 return $comparison;
456 }
457 }
458 return 0;
459}

461function minimize(array $on, array $dc, array $names, bool $pos = false): array
462{
463 $rows = [];
464 $stage = [];
465 foreach (array_unique(array_merge($on, $dc)) as $row) {
466 $stage[pattern($row, count($names))] = true;
467 }
468 foreach ($on as $row) {
469 $rows[$row] = pattern($row, count($names));
470 }
471 $primes = [];
472 $lines = [];
473 $number = 0;
474 while (true) {
475 $used = [];
476 $next = [];
477 foreach ($stage as $raw => $ignored) {
478 $text = (string)$raw;
479 for ($i = 0; $i < strlen($text); $i++) {
480 if ($text[$i] !== '0') {
481 continue;
482 }
483 $partner = $text;
484 $partner[$i] = '1';
485 if (!isset($stage[$partner])) {
486 continue;
487 }
488 $combined = $text;
489 $combined[$i] = '-';
490 $used[$text] = $used[$partner] = true;
491 $next[$combined] = true;
492 }
493 }
494 $fresh = array_diff_key($stage, $used);
495 $primes += $fresh;
496 if (!$pos) {
497 $lines[] = 'stage ' . $number . ': implicants ' . count($stage) . ' primes ' . count($fresh);
498 }
499 if ($next === []) {
500 break;
501 }
502 $stage = $next;
503 $number++;
504 }
505 $targets = [];
506 foreach (array_keys($primes) as $raw) {
507 $text = (string)$raw;
508 foreach ($rows as $row) {
509 if (containsRow($text, $row)) {
510 $targets[] = $text;
511 break;
512 }
513 }
514 }
515 $targets = sortPatterns($targets);
516 $coverage = [];
517 foreach ($targets as $text) {
518 $covered = [];
519 foreach ($on as $row) {
520 if (containsRow($text, $rows[$row])) {
521 $covered[$row] = true;
522 }
523 }
524 $coverage[] = $covered;
525 }
526 $essential = [];
527 foreach ($on as $row) {
528 $candidates = [];
529 foreach ($coverage as $i => $covered) {
530 if (isset($covered[$row])) {
531 $candidates[] = $i;
532 }
533 }
534 if (count($candidates) === 1) {
535 $essential[$candidates[0]] = true;
536 }
537 }
538 $optional = [];
539 foreach ($targets as $i => $text) {
540 if (!isset($essential[$i])) {
541 $optional[] = $i;
542 }
543 }
544 $best = null;
545 $bestKey = null;
546 for ($mask = 0; $mask < (1 << count($optional)); $mask++) {
547 $chosen = array_keys($essential);
548 foreach ($optional as $i => $index) {
549 if (($mask & (1 << $i)) !== 0) {
550 $chosen[] = $index;
551 }
552 }
553 $covered = [];
554 foreach ($chosen as $i) {
555 $covered += $coverage[$i];
556 }
557 $complete = true;
558 foreach ($on as $row) {
559 if (!isset($covered[$row])) {
560 $complete = false;
561 break;
562 }
563 }
564 if (!$complete) {
565 continue;
566 }
567 $texts = sortPatterns(array_map(fn($i) => $pos ? complementPattern($targets[$i]) : $targets[$i], $chosen));
568 $key = [count($texts), array_sum(array_map('literalCount', $texts)), array_map('patternKey', $texts)];
569 if ($bestKey === null || coverKeyCompare($key, $bestKey) < 0) {
570 $bestKey = $key;
571 $best = $texts;
572 }
573 }
574 if ($pos) {
575 return ['minimal: ' . coverText($best, $names, false), 'clauses ' . count($best) . ' literals ' . $bestKey[1]];
576 }
577 $terms = array_map(fn($text) => patternPart($text, $names), $targets);
578 $indices = array_keys($essential);
579 sort($indices, SORT_NUMERIC);
580 $essentials = array_map(fn($i) => patternPart($targets[$i], $names), $indices);
581 return array_merge($lines, ['primes: ' . ($terms === [] ? 'none' : implode(', ', $terms)),
582 'essential: ' . ($essentials === [] ? 'none' : implode(', ', $essentials)),
583 'minimal: ' . coverText($best, $names), 'terms ' . count($best) . ' literals ' . $bestKey[1]]);
584}

586function mergeClause(string $a, string $b): ?string
587{
588 $result = '';
589 for ($i = 0; $i < strlen($a); $i++) {
590 if ($a[$i] !== '-' && $b[$i] !== '-' && $a[$i] !== $b[$i]) {
591 return null;
592 }
593 $result .= $a[$i] === '-' ? $b[$i] : $a[$i];
594 }
595 return $result;
596}

598function tidy(array $clauses): array
599{
600 $unique = array_values(array_unique($clauses));
601 $result = [];
602 foreach ($unique as $clause) {
603 $absorbed = false;
604 foreach ($unique as $other) {
605 if ($clause === $other) {
606 continue;
607 }
608 $subset = true;
609 for ($i = 0; $i < strlen($other); $i++) {
610 if ($other[$i] !== '-' && $other[$i] !== $clause[$i]) {
611 $subset = false;
612 break;
613 }
614 }
615 if ($subset) {
616 $absorbed = true;
617 break;
618 }
619 }
620 if (!$absorbed) {
621 $result[] = $clause;
622 }
623 }
624 return sortPatterns($result);
625}

627function clauses(array $node, array $names): array
628{
629 $memo = [];
630 $visit = function (array $current) use (&$visit, &$memo, $names): array {
631 $key = $current['memo_id'] ?? null;
632 if ($key !== null && isset($memo[$key])) {
633 return $memo[$key];
634 }
635 if ($current[0] === 'variable' || $current[0] === 'not') {
636 $name = $current[0] === 'variable' ? $current[1] : $current[1][1];
637 $text = str_repeat('-', count($names));
638 $text[array_search($name, $names, true)] = $current[0] === 'variable' ? '1' : '0';
639 $result = [$text];
640 } elseif ($current[0] === 'constant') {
641 $result = $current[1] === 1 ? [] : [str_repeat('-', count($names))];
642 } else {
643 $left = $visit($current[1]);
644 $right = $visit($current[2]);
645 if ($current[0] === '&') {
646 $result = tidy(array_merge($left, $right));
647 } else {
648 $result = [];
649 foreach ($left as $a) {
650 foreach ($right as $b) {
651 $merged = mergeClause($a, $b);
652 if ($merged !== null) {
653 $result[] = $merged;
654 }
655 }
656 }
657 $result = tidy($result);
658 }
659 }
660 if ($key !== null) {
661 $memo[$key] = $result;
662 }
663 return $result;
664 };
665 return $visit($node);
666}

715 private function bdd(array $values, array $names): array
716 {
717 $nodes = [];
718 $unique = [];
719 $build = function (int $i, array $table) use (&$build, &$nodes, &$unique, $names): int {
720 if ($i === count($names)) {
721 return $table[0];
722 }
723 $half = intdiv(count($table), 2);
724 $low = $build($i + 1, array_slice($table, 0, $half));
725 $high = $build($i + 1, array_slice($table, $half));
726 if ($low === $high) {
727 return $low;
728 }
729 $key = $i . ',' . $low . ',' . $high;
730 if (!isset($unique[$key])) {
731 $unique[$key] = count($nodes) + 2;
732 $nodes[] = [$i, $low, $high];
733 }
734 return $unique[$key];
735 };
736 $root = $build(0, $values);
737 $counts = [0 => 0, 1 => 1];
738 $paths = function (int $id) use (&$paths, &$counts, $nodes): int {
739 if (!isset($counts[$id])) {
740 $counts[$id] = $paths($nodes[$id - 2][1]) + $paths($nodes[$id - 2][2]);
741 }
742 return $counts[$id];
743 };
744 $lines = ['nodes: ' . count($nodes)];
745 foreach ($nodes as $j => [$i, $low, $high]) {
746 $lines[] = ($j + 2) . ' = ite(' . $names[$i] . ', ' . $high . ', ' . $low . ')';
747 }
748 return array_merge($lines, ['root: ' . $root, 'paths: ' . $paths($root)]);
749 }

751 public function command(string $line): array
752 {
753 $separator = strpos($line, ' ');
754 $command = $separator === false ? $line : substr($line, 0, $separator);
755 $argument = $separator === false ? '' : substr($line, $separator + 1);
756 if (!in_array($command, self::COMMANDS, true)) {
757 throw new LogicError('unknown command ' . $command);
758 }
759 if ($command === 'LIST' && $separator !== false) {
760 throw new LogicError('unexpected argument');
761 }
762 if ($command !== 'LIST' && $separator === false) {
763 throw new LogicError('missing argument');
764 }
765 if ($command === 'LIST') {
766 $names = array_keys($this->definitions);
767 sort($names, SORT_STRING);
768 return array_merge(['names: ' . count($names)], array_map(fn($name) => $name . ' = ' . notation($this->definitions[$name]), $names));
769 }
770 if ($command === 'DEF') {
771 $split = strpos($argument, ' ');
772 if ($split === false) {
773 throw new LogicError('missing argument');
774 }
775 $name = substr($argument, 0, $split);
776 if (!$this->validName($name)) {
777 throw new LogicError("invalid name '" . $name . "'");
778 }
779 return $this->define($name, $this->parse(substr($argument, $split + 1), strlen($name) + 6));
780 }
781 if ($command === 'DEL') {
782 if (!$this->validName($argument)) {
783 throw new LogicError("invalid name '" . $argument . "'");
784 }
785 if (!isset($this->definitions[$argument])) {
786 throw new LogicError("undefined name '" . $argument . "'");
787 }
788 unset($this->definitions[$argument]);
789 return ['deleted ' . $argument];
790 }
791 if ($command === 'DEFSUM') {
792 $split = strpos($argument, ';');
793 if ($split === false) {
794 throw new LogicError("missing ';'");
795 }
796 $words = $this->words(substr($argument, 0, $split));
797 if ($words === []) {
798 throw new LogicError('missing argument');
799 }
800 $name = array_shift($words);
801 if (!$this->validName($name)) {
802 throw new LogicError("invalid name '" . $name . "'");
803 }
804 $previous = '';
805 if ($words === []) {
806 throw new LogicError('bad variable list');
807 }
808 foreach ($words as $word) {
809 if (!preg_match('/\A[A-Z]\z/', $word) || strcmp($previous, $word) >= 0) {
810 throw new LogicError('bad variable list');
811 }
812 $previous = $word;
813 }
814 if (count($words) >= 9) {
815 throw new LogicError('too many variables: ' . count($words));
816 }
817 $rows = [];
818 $previous = -1;
819 foreach ($this->words(substr($argument, $split + 1)) as $word) {
820 if (!preg_match('/\A(?:0|[1-9][0-9]*)\z/', $word) || (int)$word >= (1 << count($words)) || (int)$word <= $previous) {
821 throw new LogicError("bad minterm '" . $word . "'");
822 }
823 $rows[] = $previous = (int)$word;
824 }
825 $terms = [];
826 foreach ($rows as $row) {
827 $text = pattern($row, count($words));
828 $term = null;
829 foreach ($words as $i => $variable) {
830 $literal = $text[$i] === '1' ? boolVar($variable) : boolNot(boolVar($variable));
831 $term = $term === null ? $literal : boolBinary('&', $term, $literal);
832 }
833 $terms[] = $term;
834 }
835 $formula = null;
836 foreach ($terms as $term) {
837 $formula = $formula === null ? $term : boolBinary('|', $formula, $term);
838 }
839 return $this->define($name, $formula ?? boolConst(0));
840 }
841 $front = $argument;
842 $second = null;
843 $secondStart = null;
844 if (in_array($command, self::SPLIT, true)) {
845 $split = strpos($argument, ';');
846 if ($split === false) {
847 throw new LogicError("missing ';'");
848 }
849 $front = substr($argument, 0, $split);
850 $second = substr($argument, $split + 1);
851 $secondStart = strlen($command) + $split + 3;
852 }
853 $formula = $this->parse($front, strlen($command) + 2);
854 if ($command === 'RESTRICT') {
855 $words = $this->words($second);
856 if ($words === []) {
857 throw new LogicError('missing assignment');
858 }
859 $assignments = [];
860 foreach ($words as $word) {
861 if (!preg_match('/\A[A-Z]=[01]\z/', $word)) {
862 throw new LogicError("bad assignment '" . $word . "'");
863 }
864 if (isset($assignments[$word[0]])) {
865 throw new LogicError('duplicate variable ' . $word[0]);
866 }
867 $assignments[$word[0]] = (int)$word[2];
868 }
869 return ['restricted: ' . notation(simplify($formula, $assignments))];
870 }
871 if ($command === 'COFACTOR') {
872 $words = $this->words($second);
873 if (count($words) !== 1 || !preg_match('/\A[A-Z]\z/', $words[0])) {
874 throw new LogicError('bad variable');
875 }
876 $name = $words[0];
877 $high = simplify($formula, [$name => 1]);
878 $low = simplify($formula, [$name => 0]);
879 return ['cofactor ' . $name . '=1: ' . notation($high), 'cofactor ' . $name . '=0: ' . notation($low), 'difference: ' . notation(simplifyBinary('^', $high, $low))];
880 }
881 $other = $second === null ? null : $this->parse($second, $secondStart);
882 $names = array_values(array_unique(array_merge(variableNames($formula), $other === null ? [] : variableNames($other))));
883 sort($names, SORT_STRING);
884 if (in_array($command, self::TABLE, true) && count($names) >= 9) {
885 throw new LogicError('too many variables: ' . count($names));
886 }
887 if ($command === 'PRINT') {
888 [$size, $depth] = measures($formula);
889 return ['formula: ' . notation($formula), 'variables: ' . ($names === [] ? 'none' : implode(' ', $names)), 'size ' . $size . ' depth ' . $depth];
890 }
891 if ($command === 'NNF') {
892 return ['nnf: ' . notation(nnf($formula))];
893 }
894 if ($command === 'DUAL') {
895 return ['dual: ' . notation(dual($formula))];
896 }
897 if ($command === 'SIMPLIFY') {
898 return ['simplified: ' . notation(simplify($formula))];
899 }
900 if ($command === 'CLAUSES' || $command === 'DP') {
901 $set = clauses(nnf($formula), $names);
902 if ($command === 'CLAUSES') {
903 return ['clauses: ' . coverText($set, $names, false)];
904 }
905 $lines = ['start: clauses ' . count($set)];
906 $empty = str_repeat('-', count($names));
907 foreach ($names as $i => $name) {
908 if ($set === [] || in_array($empty, $set, true)) {
909 break;
910 }
911 $positive = array_filter($set, fn($text) => $text[$i] === '1');
912 $negative = array_filter($set, fn($text) => $text[$i] === '0');
913 $resolvents = [];
914 foreach ($positive as $a) {
915 foreach ($negative as $b) {
916 $a[$i] = $b[$i] = '-';
917 $merged = mergeClause($a, $b);
918 if ($merged !== null) {
919 $resolvents[$merged] = true;
920 }
921 }
922 }
923 $fresh = array_map('strval', array_keys($resolvents));
924 $set = tidy(array_merge(array_values(array_filter($set, fn($text) => $text[$i] === '-')), $fresh));
925 $lines[] = 'eliminate ' . $name . ': resolvents ' . count($resolvents) . ' clauses ' . count($set);
926 }
927 return array_merge($lines, [in_array($empty, $set, true) ? 'result: unsatisfiable' : 'result: satisfiable']);
928 }
929 $values = [];
930 $otherValues = [];
931 for ($row = 0; $row < (1 << count($names)); $row++) {
932 $assignment = assignments($names, $row);
933 $values[] = evaluate($formula, $assignment);
934 if ($other !== null) {
935 $otherValues[] = evaluate($other, $assignment);
936 }
937 }
938 $on = $off = [];
939 foreach ($values as $row => $value) {
940 if ($value === 1) {
941 $on[] = $row;
942 } else {
943 $off[] = $row;
944 }
945 }
946 switch ($command) {
947 case 'TABLE':
948 $lines = [implode(' ', array_merge($names, ['|', 'F']))];
949 foreach ($values as $row => $value) {
950 $lines[] = implode(' ', array_merge(str_split(pattern($row, count($names))), ['|', (string)$value]));
951 }
952 return $lines;
953 case 'MINTERMS':
954 case 'MAXTERMS':
955 $rows = $command === 'MINTERMS' ? $on : $off;
956 return [strtolower($command) . ': ' . ($rows === [] ? 'none' : implode(' ', $rows))];
957 case 'DNF':
958 case 'CNF':
959 $texts = array_map(fn($row) => pattern($row, count($names)), $command === 'DNF' ? $on : $off);
960 if ($command === 'CNF') {
961 $texts = array_map('complementPattern', $texts);
962 }
963 return [strtolower($command) . ': ' . coverText($texts, $names, $command === 'DNF')];
964 case 'ANF':
965 $texts = [];
966 foreach (coefficients($values, count($names)) as $row => $value) {
967 if ($value === 1) {
968 $texts[] = strtr(pattern($row, count($names)), '0', '-');
969 }
970 }
971 usort($texts, fn($a, $b) => (literalCount($a) <=> literalCount($b)) ?: strcmp(patternKey($a), patternKey($b)));
972 return ['anf: ' . ($texts === [] ? '0' : implode(' ^ ', array_map(fn($text) => patternPart($text, $names), $texts)))];
973 case 'SAT':
974 return [$on === [] ? 'unsatisfiable' : 'satisfiable: ' . assignmentText($names, $on[0])];
975 case 'TAUT':
976 return [$off === [] ? 'tautology' : 'not a tautology: ' . assignmentText($names, $off[0])];
977 case 'COUNT':
978 return ['models: ' . count($on) . ' of ' . count($values)];
979 case 'EQUIV':
980 foreach ($values as $row => $value) {
981 if ($value !== $otherValues[$row]) {
982 return ['not equivalent: ' . assignmentText($names, $row) . ' f=' . $value . ' g=' . $otherValues[$row]];
983 }
984 }
985 return ['equivalent'];
986 case 'IMPLIES':
987 foreach ($values as $row => $value) {
988 if ($value === 1 && $otherValues[$row] === 0) {
989 return ['does not imply: ' . assignmentText($names, $row)];
990 }
991 }
992 return ['implies'];
993 case 'PROPS':
994 $depends = [];
995 foreach ($names as $i => $name) {
996 $bit = 1 << (count($names) - 1 - $i);
997 foreach ($values as $row => $value) {
998 if ($value !== $values[$row ^ $bit]) {
999 $depends[] = $name;
1000 break;
1001 }
1002 }
1003 }
1004 $monotone = $selfDual = $symmetric = true;
1005 $weights = [];
1006 foreach ($values as $row => $value) {
1007 if ($value === $values[count($values) - 1 - $row]) {
1008 $selfDual = false;
1009 }
1010 foreach ($names as $i => $name) {
1011 $bit = 1 << $i;
1012 if (($row & $bit) === 0 && $value > $values[$row | $bit]) {
1013 $monotone = false;
1014 }
1015 }
1016 $weight = substr_count(decbin($row), '1');
1017 if (isset($weights[$weight]) && $weights[$weight] !== $value) {
1018 $symmetric = false;
1019 }
1020 $weights[$weight] = $value;
1021 }
1022 $linear = true;
1023 foreach (coefficients($values, count($names)) as $row => $value) {
1024 if ($value === 1 && substr_count(decbin($row), '1') > 1) {
1025 $linear = false;
1026 }
1027 }
1028 $lines = ['depends: ' . ($depends === [] ? 'none' : implode(' ', $depends))];
1029 foreach (['zero-preserving' => $values[0] === 0, 'one-preserving' => $values[array_key_last($values)] === 1, 'self-dual' => $selfDual, 'monotone' => $monotone, 'linear' => $linear, 'symmetric' => $symmetric] as $label => $value) {
1030 $lines[] = $label . ': ' . ($value ? 'yes' : 'no');
1031 }
1032 return $lines;
1033 case 'KMAP':
1034 if (count($names) < 2 || count($names) > 4) {
1035 throw new LogicError('kmap needs 2 to 4 variables');
1036 }
1037 $p = intdiv(count($names), 2);
1038 $q = count($names) - $p;
1039 $rows = $columns = [];
1040 for ($i = 0; $i < (1 << $p); $i++) {
1041 $rows[] = pattern($i ^ intdiv($i, 2), $p);
1042 }
1043 for ($i = 0; $i < (1 << $q); $i++) {
1044 $columns[] = pattern($i ^ intdiv($i, 2), $q);
1045 }
1046 $lines = [implode('', array_slice($names, 0, $p)) . '\\' . implode('', array_slice($names, $p)) . ' ' . implode(' ', $columns)];
1047 foreach ($rows as $row) {
1048 $text = str_pad($row, count($names) + 1);
1049 foreach ($columns as $column) {
1050 $text .= str_repeat(' ', $q) . $values[bindec($row . $column)];
1051 }
1052 $lines[] = $text;
1053 }
1054 return $lines;
1055 case 'BDD':
1056 return $this->bdd($values, $names);
1057 case 'MINIMIZE':
1058 return minimize($on, [], $names);
1059 case 'MINDC':
1060 $dc = [];
1061 foreach ($otherValues as $row => $value) {
1062 if ($value === 1) {
1063 $dc[] = $row;
1064 }
1065 }
1066 return minimize(array_values(array_diff($on, $dc)), $dc, $names);
1067 case 'MINPOS':
1068 return minimize($off, [], $names, true);
1069 }
1070 return [];
1071 }

1073 public function run(): void
1074 {
1075 $count = (int)trim(fgets(STDIN));
1076 for ($i = 0; $i < $count; $i++) {
1077 $input = fgets(STDIN);
1078 $line = str_ends_with($input, "\n") ? substr($input, 0, -1) : $input;
1079 try {
1080 foreach ($this->command($line) as $text) {
1081 echo $text, "\n";
1082 }
1083 } catch (LogicError $error) {
1084 echo 'error: ', $error->getMessage(), "\n";
1085 }
1086 }
1087 }
