48fn isVariable(text: Text) bool {
49 return text.len == 1 and text[0] >= 'A' and text[0] <= 'Z';
50}
51fn validName(text: Text) bool {
52 if (text.len == 0 or text.len > 8 or text[0] < 'a' or text[0] > 'z') return false;
53 for (text) |c| if (!(c >= 'a' and c <= 'z' or c >= '0' and c <= '9')) return false;
54 return true;
55}
56fn priority(kind: Text) usize {
57 for ([_]Text{ "<->", "->", "|", "^", "&" }, 1..) |operator, i| if (equalText(operator, kind)) return i;
58 return 0;
59}
60fn notationInto(node: *const Node, out: *Builder) Error!void {
61 if (node.left == null) {
62 try out.add(node.kind);
63 return;
64 }
65 if (equalText(node.kind, "~")) {
66 try out.add("~");
67 const parentheses = priority(node.left.?.kind) > 0;
68 if (parentheses) try out.add("(");
69 try notationInto(node.left.?, out);
70 if (parentheses) try out.add(")");
71 return;
72 }
73 const p = priority(node.kind);
74 const left = node.left.?;
75 const right = node.right.?;
76 const lp = priority(left.kind) > 0 and (priority(left.kind) < p or equalText(node.kind, "->") and equalText(left.kind, "->"));
77 const rp = priority(right.kind) > 0 and (priority(right.kind) < p or equalText(node.kind, right.kind) and !equalText(node.kind, "->"));
78 if (lp) try out.add("(");
79 try notationInto(left, out);
80 if (lp) try out.add(")");
81 try out.print(" {s} ", .{node.kind});
82 if (rp) try out.add("(");
83 try notationInto(right, out);
84 if (rp) try out.add(")");
85}
86fn notation(gpa: Allocator, node: *const Node) Error!Text {
87 var out: Builder = .{ .gpa = gpa };
88 try notationInto(node, &out);
89 return out.bytes.items;
90}
91fn sameTree(a: *const Node, b: *const Node) bool {
92 return a == b or equalText(a.kind, b.kind) and (a.left == null or sameTree(a.left.?, b.left.?) and (a.right == null or sameTree(a.right.?, b.right.?)));
93}
94const Token = struct { text: Text, position: usize };
95fn lexical(gpa: Allocator, source: Text, start: usize) Error![]const Token {
96 var result: std.ArrayList(Token) = .empty;
97 var i: usize = 0;
98 while (i < source.len) {
99 if (source[i] == ' ') {
100 i += 1;
101 continue;
102 }
103 const begin = i;
104 if (std.mem.startsWith(u8, source[i..], "<->")) {
105 i += 3;
106 } else if (std.mem.startsWith(u8, source[i..], "->")) {
107 i += 2;
108 } else if (source[i] >= 'a' and source[i] <= 'z') {
109 while (i < source.len and (source[i] >= 'a' and source[i] <= 'z' or std.ascii.isDigit(source[i]))) i += 1;
110 } else if (std.mem.findScalar(u8, "ABCDEFGHIJKLMNOPQRSTUVWXYZ01~&^|()", source[i]) != null) {
111 i += 1;
112 } else return bad(gpa, "invalid character '{c}' at {d}", .{ source[i], start + i });
113 try result.append(gpa, .{ .text = source[begin..i], .position = start + begin });
114 }
115 return result.items;
116}
117const Names = std.StringHashMapUnmanaged(*Node);
118const Parser = struct {
119 gpa: Allocator,
120 names: Names,
121 tokens: []const Token,
122 position: usize = 0,
123 end: usize,
124 fn unexpected(self: *Parser) Error {
125 return if (self.position == self.tokens.len) bad(self.gpa, "unexpected end of formula at {d}", .{self.end}) else bad(self.gpa, "unexpected '{s}' at {d}", .{ self.tokens[self.position].text, self.tokens[self.position].position });
126 }
127 fn readUnary(self: *Parser) Error!*Node {
128 if (self.position == self.tokens.len) return self.unexpected();
129 const token = self.tokens[self.position];
130 if (equalText(token.text, "~")) {
131 self.position += 1;
132 return negation(self.gpa, try self.readUnary());
133 }
134 if (equalText(token.text, "(")) {
135 self.position += 1;
136 const result = try self.expression(1);
137 if (self.position == self.tokens.len or !equalText(self.tokens[self.position].text, ")")) return self.unexpected();
138 self.position += 1;
139 return result;
140 }
141 if (isVariable(token.text) or includes(token.text, &.{ "0", "1" })) {
142 self.position += 1;
143 return leaf(self.gpa, token.text);
144 }
145 if (token.text[0] >= 'a' and token.text[0] <= 'z') {
146 const result = self.names.get(token.text) orelse return bad(self.gpa, "undefined name '{s}' at {d}", .{ token.text, token.position });
147 self.position += 1;
148 return result;
149 }
150 return self.unexpected();
151 }
152 fn expression(self: *Parser, minimum: usize) Error!*Node {
153 var result = try self.readUnary();
154 while (self.position < self.tokens.len) {
155 const operator = self.tokens[self.position].text;
156 const p = priority(operator);
157 if (p < minimum) break;
158 self.position += 1;
159 result = try makeNode(self.gpa, operator, result, try self.expression(if (equalText(operator, "->")) p else p + 1));
160 }
161 return result;
162 }
163};
164fn parse(gpa: Allocator, source: Text, start: usize, names: Names) Error!*Node {
165 var parser: Parser = .{ .gpa = gpa, .names = names, .tokens = try lexical(gpa, source, start), .end = start + source.len };
166 const result = try parser.expression(1);
167 if (parser.position != parser.tokens.len) return parser.unexpected();
168 return result;
169}
170fn collectVariables(node: *const Node, seen: *[26]bool) void {
171 if (isVariable(node.kind)) seen[node.kind[0] - 'A'] = true;
172 if (node.left) |child| collectVariables(child, seen);
173 if (node.right) |child| collectVariables(child, seen);
174}
175fn variableList(gpa: Allocator, first: *const Node, second: ?*const Node) Error![]const Text {
176 var seen: [26]bool = @splat(false);
177 collectVariables(first, &seen);
178 if (second) |node| collectVariables(node, &seen);
179 var names: std.ArrayList(Text) = .empty;
180 for (seen, 0..) |present, i| if (present) try names.append(gpa, "ABCDEFGHIJKLMNOPQRSTUVWXYZ"[i .. i + 1]);
181 return names.items;
182}
183fn treeSize(node: *const Node) usize {
184 if (node.left == null) return 1;
185 return 1 + treeSize(node.left.?) + (if (node.right) |child| treeSize(child) else 0);
186}
187fn treeDepth(node: *const Node) usize {
188 if (node.left == null) return 1;
189 return 1 + @max(treeDepth(node.left.?), if (node.right) |child| treeDepth(child) else 0);
190}
191fn truth(node: *const Node, assignment: *const [26]u8) u8 {
192 if (isVariable(node.kind)) return assignment[node.kind[0] - 'A'];
193 if (equalText(node.kind, "0")) return 0;
194 if (equalText(node.kind, "1")) return 1;
195 const a = truth(node.left.?, assignment);
196 if (equalText(node.kind, "~")) return 1 - a;
197 const b = truth(node.right.?, assignment);
198 if (equalText(node.kind, "&")) return a & b;
199 if (equalText(node.kind, "|")) return a | b;
200 if (equalText(node.kind, "^")) return a ^ b;
201 if (equalText(node.kind, "->")) return (1 - a) | b;
202 return 1 - (a ^ b);
203}
204fn power2(n: usize) usize {
205 return @as(usize, 1) << @as(std.math.Log2Int(usize), @intCast(n));
206}
207fn bitAt(row: usize, bit: usize) u8 {
208 return @intCast((row >> @as(std.math.Log2Int(usize), @intCast(bit))) & 1);
209}
210fn truthTable(gpa: Allocator, node: *const Node, names: []const Text) Error![]u8 {
211 const result = try gpa.alloc(u8, power2(names.len));
212 for (result, 0..) |*value, row| {
213 var assignment: [26]u8 = @splat(0);
214 for (names, 0..) |name, i| assignment[name[0] - 'A'] = bitAt(row, names.len - 1 - i);
215 value.* = truth(node, &assignment);
216 }
217 return result;
218}
219fn assignmentText(gpa: Allocator, names: []const Text, row: usize) Error!Text {
220 var out: Builder = .{ .gpa = gpa };
221 try out.add("{");
222 for (names, 0..) |name, i| {
223 if (i > 0) try out.add(" ");
224 try out.print("{s}={d}", .{ name, bitAt(row, names.len - 1 - i) });
225 }
226 try out.add("}");
227 return out.bytes.items;
228}
229fn bitPattern(gpa: Allocator, row: usize, n: usize) Error!Text {
230 const result = try gpa.alloc(u8, n);
231 for (result, 0..) |*c, i| c.* = '0' + bitAt(row, n - 1 - i);
232 return result;
233}
234fn literalCount(pattern: Text) usize {
235 var result: usize = 0;
236 for (pattern) |c| if (c != '-') {
237 result += 1;
238 };
239 return result;
240}
241fn patternRank(c: u8) u8 {
242 return if (c == '1') 0 else if (c == '0') 1 else 2;
243}
244fn patternLess(_: void, a: Text, b: Text) bool {
245 for (a, b) |x, y| if (patternRank(x) != patternRank(y)) return patternRank(x) < patternRank(y);
246 return false;
247}
248fn sortPatterns(patterns: []Text) void {
249 std.mem.sort(Text, patterns, {}, patternLess);
250}
251fn flipped(gpa: Allocator, pattern: Text) Error!Text {
252 const result = try gpa.dupe(u8, pattern);
253 for (result) |*c| {
254 if (c.* == '1') {
255 c.* = '0';
256 } else if (c.* == '0') c.* = '1';
257 }
258 return result;
259}
260fn termText(gpa: Allocator, pattern: Text, names: []const Text, clause: bool) Error!Text {
261 var out: Builder = .{ .gpa = gpa };
262 var first = true;
263 for (pattern, names) |c, name| {
264 if (c == '-') continue;
265 if (!first) try out.add(if (clause) " | " else " & ");
266 first = false;
267 if (c == '0') try out.add("~");
268 try out.add(name);
269 }
270 if (first) try out.add(if (clause) "0" else "1");
271 return out.bytes.items;
272}
273fn formText(gpa: Allocator, patterns: []const Text, names: []const Text, clausesMode: bool) Error!Text {
274 if (patterns.len == 0) return if (clausesMode) "1" else "0";
275 var out: Builder = .{ .gpa = gpa };
276 for (patterns, 0..) |pattern, i| {
277 if (i > 0) try out.add(if (clausesMode) " & " else " | ");
278 const parentheses = clausesMode and patterns.len >= 2 and literalCount(pattern) >= 2;
279 if (parentheses) try out.add("(");
280 try out.add(try termText(gpa, pattern, names, clausesMode));
281 if (parentheses) try out.add(")");
282 }
283 return out.bytes.items;
284}
285fn reduceNegation(gpa: Allocator, node: *Node) Error!*Node {
286 if (equalText(node.kind, "0")) return leaf(gpa, "1");
287 if (equalText(node.kind, "1")) return leaf(gpa, "0");
288 if (equalText(node.kind, "~")) return node.left.?;
289 return negation(gpa, node);
290}
291fn simplifyBinary(gpa: Allocator, operator: Text, a: *Node, b: *Node) Error!*Node {
292 const same = sameTree(a, b);
293 const complement = equalText(a.kind, "~") and sameTree(a.left.?, b) or equalText(b.kind, "~") and sameTree(b.left.?, a);
294 if (equalText(operator, "&")) {
295 if (equalText(a.kind, "0") or equalText(b.kind, "0")) return leaf(gpa, "0");
296 if (equalText(a.kind, "1")) return b;
297 if (equalText(b.kind, "1") or same) return a;
298 if (complement) return leaf(gpa, "0");
299 } else if (equalText(operator, "|")) {
300 if (equalText(a.kind, "1") or equalText(b.kind, "1")) return leaf(gpa, "1");
301 if (equalText(a.kind, "0")) return b;
302 if (equalText(b.kind, "0") or same) return a;
303 if (complement) return leaf(gpa, "1");
304 } else if (equalText(operator, "^")) {
305 if (equalText(a.kind, "0")) return b;
306 if (equalText(b.kind, "0")) return a;
307 if (equalText(a.kind, "1")) return reduceNegation(gpa, b);
308 if (equalText(b.kind, "1")) return reduceNegation(gpa, a);
309 if (same) return leaf(gpa, "0");
310 if (complement) return leaf(gpa, "1");
311 } else if (equalText(operator, "->")) {
312 if (equalText(a.kind, "0") or equalText(b.kind, "1")) return leaf(gpa, "1");
313 if (equalText(a.kind, "1")) return b;
314 if (equalText(b.kind, "0")) return reduceNegation(gpa, a);
315 if (same) return leaf(gpa, "1");
316 if (complement) return b;
317 } else {
318 if (equalText(a.kind, "1")) return b;
319 if (equalText(b.kind, "1")) return a;
320 if (equalText(a.kind, "0")) return reduceNegation(gpa, b);
321 if (equalText(b.kind, "0")) return reduceNegation(gpa, a);
322 if (same) return leaf(gpa, "1");
323 if (complement) return leaf(gpa, "0");
324 }
325 return makeNode(gpa, operator, a, b);
326}
327fn simplify(gpa: Allocator, node: *Node, assignments: *const [26]?Text) Error!*Node {
328 if (node.left == null) {
329 if (isVariable(node.kind)) {
330 if (assignments[node.kind[0] - 'A']) |value| return leaf(gpa, value);
331 }
332 return node;
333 }
334 const a = try simplify(gpa, node.left.?, assignments);
335 if (equalText(node.kind, "~")) return reduceNegation(gpa, a);
336 return simplifyBinary(gpa, node.kind, a, try simplify(gpa, node.right.?, assignments));
337}
338const SignedNode = struct { node: *Node, positive: bool };
339const NnfMemo = std.AutoHashMapUnmanaged(SignedNode, *Node);
340fn nnf(gpa: Allocator, node: *Node, positive: bool, memo: *NnfMemo) Error!*Node {
341 const key: SignedNode = .{ .node = node, .positive = positive };
342 if (memo.get(key)) |result| return result;
343 var result: *Node = undefined;
344 if (includes(node.kind, &.{ "0", "1" })) {
345 result = if (positive) node else try reduceNegation(gpa, node);
346 } else if (equalText(node.kind, "~")) {
347 result = try nnf(gpa, node.left.?, !positive, memo);
348 } else if (includes(node.kind, &.{ "&", "|" })) {
349 const operator = if (positive) node.kind else if (equalText(node.kind, "&")) "|" else "&";
350 result = try makeNode(gpa, operator, try nnf(gpa, node.left.?, positive, memo), try nnf(gpa, node.right.?, positive, memo));
351 } else if (equalText(node.kind, "->")) {
352 result = try makeNode(gpa, if (positive) "|" else "&", try nnf(gpa, node.left.?, !positive, memo), try nnf(gpa, node.right.?, positive, memo));
353 } else if (includes(node.kind, &.{ "^", "<->" })) {
354 const sameSign = if (equalText(node.kind, "^")) !positive else positive;
355 const a = try makeNode(gpa, "&", try nnf(gpa, node.left.?, true, memo), try nnf(gpa, node.right.?, sameSign, memo));
356 const b = try makeNode(gpa, "&", try nnf(gpa, node.left.?, false, memo), try nnf(gpa, node.right.?, !sameSign, memo));
357 result = try makeNode(gpa, "|", a, b);
358 } else {
359 result = if (positive) node else try negation(gpa, node);
360 }
361 try memo.put(gpa, key, result);
362 return result;
363}
364fn dual(gpa: Allocator, node: *Node) Error!*Node {
365 if (isVariable(node.kind)) return node;
366 if (includes(node.kind, &.{ "0", "1" })) return reduceNegation(gpa, node);
367 if (equalText(node.kind, "~")) return negation(gpa, try dual(gpa, node.left.?));
368 const a = try dual(gpa, node.left.?);
369 const b = try dual(gpa, node.right.?);
370 if (equalText(node.kind, "->")) return makeNode(gpa, "&", try negation(gpa, a), b);
371 const operator: Text = if (equalText(node.kind, "&")) "|" else if (equalText(node.kind, "|")) "&" else if (equalText(node.kind, "^")) "<->" else "^";
372 return makeNode(gpa, operator, a, b);
373}
374fn covers(pattern: Text, row: usize) bool {
375 for (pattern, 0..) |c, i| if (c != '-' and c != '0' + bitAt(row, pattern.len - 1 - i)) return false;
376 return true;
377}
378fn patternSubset(a: Text, b: Text) bool {
379 for (a, b) |x, y| if (x != '-' and x != y) return false;
380 return true;
381}
382fn tidy(gpa: Allocator, patterns: []const Text) Error![]Text {
383 var unique: std.ArrayList(Text) = .empty;
384 for (patterns) |pattern| if (!includes(pattern, unique.items)) try unique.append(gpa, pattern);
385 var result: std.ArrayList(Text) = .empty;
386 for (unique.items, 0..) |pattern, i| {
387 var redundant = false;
388 for (unique.items, 0..) |other, j| if (i != j and patternSubset(other, pattern)) {
389 redundant = true;
390 break;
391 };
392 if (!redundant) try result.append(gpa, pattern);
393 }
394 return result.items;
395}
396fn mergeClauses(gpa: Allocator, a: Text, b: Text) Error!?Text {
397 const result = try gpa.dupe(u8, a);
398 for (result, b) |*x, y| {
399 if (y == '-') continue;
400 if (x.* != '-' and x.* != y) return null;
401 x.* = y;
402 }
403 return result;
404}
405const ClauseMemo = std.AutoHashMapUnmanaged(*Node, []const Text);
406fn clauseSet(gpa: Allocator, node: *Node, names: []const Text, memo: *ClauseMemo) Error![]const Text {
407 if (memo.get(node)) |result| return result;
408 var patterns: std.ArrayList(Text) = .empty;
409 if (equalText(node.kind, "1")) {} else if (equalText(node.kind, "0")) {
410 const empty = try gpa.alloc(u8, names.len);
411 @memset(empty, '-');
412 try patterns.append(gpa, empty);
413 } else if (includes(node.kind, &.{ "&", "|" })) {
414 const a = try clauseSet(gpa, node.left.?, names, memo);
415 const b = try clauseSet(gpa, node.right.?, names, memo);
416 if (equalText(node.kind, "&")) {
417 try patterns.appendSlice(gpa, a);
418 try patterns.appendSlice(gpa, b);
419 } else for (a) |x| for (b) |y| {
420 if (try mergeClauses(gpa, x, y)) |merged| try patterns.append(gpa, merged);
421 };
422 } else {
423 const negative = equalText(node.kind, "~");
424 const name = if (negative) node.left.?.kind else node.kind;
425 const pattern = try gpa.alloc(u8, names.len);
426 for (names, pattern) |variable, *c| c.* = if (equalText(name, variable)) (if (negative) @as(u8, '0') else '1') else '-';
427 try patterns.append(gpa, pattern);
428 }
429 const result = try tidy(gpa, patterns.items);
430 try memo.put(gpa, node, result);
431 return result;
432}
433fn patternsTerms(gpa: Allocator, patterns: []const Text, names: []const Text) Error!Text {
434 if (patterns.len == 0) return "none";
435 var out: Builder = .{ .gpa = gpa };
436 for (patterns, 0..) |pattern, i| {
437 if (i > 0) try out.add(", ");
438 try out.add(try termText(gpa, pattern, names, false));
439 }
440 return out.bytes.items;
441}
442fn minimize(gpa: Allocator, values: []const u8, dc: ?[]const u8, names: []const Text, pos: bool, out: *Builder) Error!void {
443 var on: std.ArrayList(usize) = .empty;
444 var stage: std.ArrayList(Text) = .empty;
445 for (values, 0..) |value, row| {
446 const dontCare = if (dc) |table| table[row] == 1 else false;
447 if (value == 1 and !dontCare) try on.append(gpa, row);
448 if (value == 1 or dontCare) try stage.append(gpa, try bitPattern(gpa, row, names.len));
449 }
450 var primes: std.ArrayList(Text) = .empty;
451 var level: usize = 0;
452 while (true) : (level += 1) {
453 var used: std.StringHashMapUnmanaged(void) = .empty;
454 var next: std.ArrayList(Text) = .empty;
455 for (stage.items, 0..) |a, i| for (stage.items[i + 1 ..]) |b| {
456 var differing: ?usize = null;
457 var valid = true;
458 for (a, b, 0..) |x, y, j| if (x != y) {
459 if (x == '-' or y == '-' or differing != null) {
460 valid = false;
461 break;
462 }
463 differing = j;
464 };
465 if (valid and differing != null) {
466 const pattern = try gpa.dupe(u8, a);
467 pattern[differing.?] = '-';
468 try used.put(gpa, a, {});
469 try used.put(gpa, b, {});
470 if (!includes(pattern, next.items)) try next.append(gpa, pattern);
471 }
472 };
473 var count: usize = 0;
474 for (stage.items) |pattern| if (!used.contains(pattern)) {
475 try primes.append(gpa, pattern);
476 count += 1;
477 };
478 if (!pos) try out.line("stage {d}: implicants {d} primes {d}", .{ level, stage.items.len, count });
479 if (next.items.len == 0) break;
480 stage = next;
481 }
482 var target: std.ArrayList(Text) = .empty;
483 for (primes.items) |pattern| for (on.items) |row| {
484 if (covers(pattern, row)) {
485 try target.append(gpa, pattern);
486 break;
487 }
488 };
489 sortPatterns(target.items);
490 var essential: std.ArrayList(Text) = .empty;
491 for (on.items) |row| {
492 var options: usize = 0;
493 var selected: Text = "";
494 for (target.items) |pattern| if (covers(pattern, row)) {
495 options += 1;
496 selected = pattern;
497 };
498 if (options == 1 and !includes(selected, essential.items)) try essential.append(gpa, selected);
499 }
500 sortPatterns(essential.items);
501 var optional: std.ArrayList(Text) = .empty;
502 for (target.items) |pattern| if (!includes(pattern, essential.items)) try optional.append(gpa, pattern);
503 var best: []Text = &.{};
504 var bestCount: usize = std.math.maxInt(usize);
505 var bestLiterals: usize = std.math.maxInt(usize);
506 for (0..power2(optional.items.len)) |mask| {
507 var chosen: std.ArrayList(Text) = .empty;
508 try chosen.appendSlice(gpa, essential.items);
509 for (optional.items, 0..) |pattern, i| if (bitAt(mask, i) != 0) try chosen.append(gpa, pattern);
510 var valid = true;
511 for (on.items) |row| {
512 var found = false;
513 for (chosen.items) |pattern| if (covers(pattern, row)) {
514 found = true;
515 break;
516 };
517 if (!found) {
518 valid = false;
519 break;
520 }
521 }
522 if (!valid) continue;
523 var literals: usize = 0;
524 for (chosen.items) |pattern| literals += literalCount(pattern);
525 const displayed = try gpa.alloc(Text, chosen.items.len);
526 for (displayed, chosen.items) |*pattern, original| pattern.* = if (pos) try flipped(gpa, original) else original;
527 sortPatterns(displayed);
528 var better = chosen.items.len < bestCount or chosen.items.len == bestCount and literals < bestLiterals;
529 if (chosen.items.len == bestCount and literals == bestLiterals) for (displayed, best) |a, b| {
530 if (!equalText(a, b)) {
531 better = patternLess({}, a, b);
532 break;
533 }
534 };
535 if (better) {
536 best = displayed;
537 bestCount = chosen.items.len;
538 bestLiterals = literals;
539 }
540 }
541 if (pos) {
542 try out.line("minimal: {s}", .{try formText(gpa, best, names, true)});
543 try out.line("clauses {d} literals {d}", .{ bestCount, bestLiterals });
544 } else {
545 try out.line("primes: {s}", .{try patternsTerms(gpa, target.items, names)});
546 try out.line("essential: {s}", .{try patternsTerms(gpa, essential.items, names)});
547 try out.line("minimal: {s}", .{try formText(gpa, best, names, false)});
548 try out.line("terms {d} literals {d}", .{ bestCount, bestLiterals });
549 }
550}
551fn coefficients(gpa: Allocator, values: []const u8, n: usize) Error![]u8 {
552 const result = try gpa.dupe(u8, values);
553 for (0..n) |bit| for (0..result.len) |row| {
554 if (bitAt(row, bit) != 0) result[row] ^= result[row ^ power2(bit)];
555 };
556 return result;
557}
558fn bitCount(value: usize) usize {
559 return @popCount(value);
560}
561fn yes(value: bool) Text {
562 return if (value) "yes" else "no";

570 fn build(self: *Bdd, level: usize, rows: []const u8) Error!usize {
571 if (level == self.names.len) return rows[0];
572 const low = try self.build(level + 1, rows[0 .. rows.len / 2]);
573 const high = try self.build(level + 1, rows[rows.len / 2 ..]);
574 if (low == high) return low;

576 if (self.unique.get(key)) |index| return index;
577 const index = self.nodes.items.len + 2;
578 try self.nodes.append(self.gpa, key);
579 try self.unique.put(self.gpa, key, index);
580 return index;

638 if (names.len >= 9) return bad(self.scratch, "too many variables: {d}", .{names.len});

642 var valid = word.len > 0 and (equalText(word, "0") or word[0] >= '1' and word[0] <= '9');

649 previous = row;

652 var literal = try leaf(self.gpa, variable);

656 formula = if (formula) |old| try makeNode(self.gpa, "|", old, product.?) else product.?;

664 var backStart: usize = 0;
665 if (includes(command, &.{ "RESTRICT", "COFACTOR", "MINDC", "EQUIV", "IMPLIES" })) {

667 front = argument[0..semicolon];

673 if (equalText(command, "RESTRICT")) {

679 const index = word[0] - 'A';

693 assignments[name[0] - 'A'] = "0";
694 const zero = try simplify(self.scratch, formula, &assignments);

696 try self.out.line("cofactor {s}=0: {s}", .{ name, try notation(self.scratch, zero) });

871 try self.out.add("\n");
872 for (0..power2(p)) |i| {
873 const row = i ^ (i / 2);
874 try self.out.add(try bitPattern(self.scratch, row, p));
875 for (0..n + 1 - p) |_| try self.out.add(" ");
876 for (0..power2(q)) |j| {
877 for (0..q) |_| try self.out.add(" ");

879 }
880 try self.out.add("\n");

887 var monotone = true;
888 var linear = true;
889 var symmetric = true;
890 for (names, 0..) |name, i| {
891 const bit = power2(names.len - 1 - i);
892 for (values, 0..) |value, row| if (value != values[row ^ bit]) {
893 try depends.append(self.scratch, name);
894 break;
895 };
896 }

898 for (values, 0..) |value, row| {
899 if (value == values[values.len - 1 - row]) selfDual = false;
900 for (0..names.len) |i| {
901 const bit = power2(i);
902 if (row & bit == 0 and value > values[row | bit]) monotone = false;
903 }
904 const weight = bitCount(row);
905 if (weights[weight]) |previous| {
906 if (previous != value) symmetric = false;
907 }
908 weights[weight] = value;
909 }
910 for (try coefficients(self.scratch, values, names.len), 0..) |coefficient, row| if (coefficient == 1 and bitCount(row) > 1) {
911 linear = false;
912 };

914 if (depends.items.len == 0) try self.out.add(" none");
915 for (depends.items) |name| try self.out.print(" {s}", .{name});
916 try self.out.add("\n");
917 try self.out.line("zero-preserving: {s}", .{yes(values[0] == 0)});
918 try self.out.line("one-preserving: {s}", .{yes(values[values.len - 1] == 1)});
919 try self.out.line("self-dual: {s}", .{yes(selfDual)});
920 try self.out.line("monotone: {s}", .{yes(monotone)});
921 try self.out.line("linear: {s}", .{yes(linear)});
922 try self.out.line("symmetric: {s}", .{yes(symmetric)});
923 return;
924 }
925 if (equalText(command, "BDD")) {
926 var bdd: Bdd = .{ .gpa = self.scratch, .names = names };
927 const root = try bdd.build(0, values);
928 var paths: std.ArrayList(usize) = .empty;
929 try paths.appendSlice(self.scratch, &.{ 0, 1 });
930 try self.out.line("nodes: {d}", .{bdd.nodes.items.len});
931 for (bdd.nodes.items, 0..) |node, i| {
932 try self.out.line("{d} = ite({s}, {d}, {d})", .{ i + 2, names[node.variable], node.high, node.low });
933 try paths.append(self.scratch, paths.items[node.low] + paths.items[node.high]);
934 }
935 try self.out.line("root: {d}", .{root});
936 try self.out.line("paths: {d}", .{paths.items[root]});
937 return;
938 }
939 }

941pub fn main(init: std.process.Init) !void {
942 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
943 defer arena.deinit();
944 const gpa = arena.allocator();
945 var inputBuffer: [8192]u8 = undefined;
946 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
947 const text = try input.interface.allocRemaining(gpa, .unlimited);
948 var lines = std.mem.splitScalar(u8, text, '\n');
949 const count = try std.fmt.parseInt(usize, std.mem.trim(u8, lines.next().?, "\r"), 10);
950 var outputBuffer: [8192]u8 = undefined;
951 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
952 var unused: Builder = .{ .gpa = gpa };
