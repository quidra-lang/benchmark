124 class Parser {
125 final List<Token> tokens = new ArrayList<>();
126 int at, end;
127 Parser(String text, int start) {
128 end = start + text.length();
129 for (int i = 0; i < text.length();) {
130 char c = text.charAt(i);
131 if (c == ' ') {
132 i++;
133 continue;
134 }
135 int first = i;
136 if (text.startsWith("<->", i)) i += 3;
137 else if (text.startsWith("->", i)) i += 2;
138 else if (lower(c)) {
139 i++;
140 while (i < text.length() && (lower(text.charAt(i)) || digit(text.charAt(i)))) i++;
141 } else if (upper(c) || "01~&^|()".indexOf(c) >= 0) i++;
142 else {
143 fail("invalid character '" + c + "' at " + (start + i));
144 return;
145 }
146 tokens.add(new Token(text.substring(first, i), start + first));
147 }
148 }
149 String peek() {
150 return at == tokens.size() ? "" : tokens.get(at).text;
151 }
152 void unexpected() {
153 if (at == tokens.size()) fail("unexpected end of formula at " + end);
154 Token t = tokens.get(at);
155 fail("unexpected '" + t.text + "' at " + t.position);
156 }
157 Node unary() {
158 String t = peek();
159 if (t.equals("~")) {
160 at++;
161 return not(unary());
162 }
163 if (t.equals("(")) {
164 at++;
165 Node n = expression(1);
166 if (!peek().equals(")")) unexpected();
167 at++;
168 return n;
169 }
170 if (!t.isEmpty() && (upper(t.charAt(0)) || t.equals("0") || t.equals("1"))) {
171 at++;
172 return leaf(t);
173 }
174 if (!t.isEmpty() && lower(t.charAt(0))) {
175 Token token = tokens.get(at++);
176 Node n = definitions.get(t);
177 if (n == null) fail("undefined name '" + t + "' at " + token.position);
178 return n;
179 }
180 unexpected();
181 return null;
182 }
183 Node expression(int level) {
184 if (level == 6) return unary();
185 Node left = expression(level + 1);
186 String op = switch (level) {
187 case 1 -> "<->";
188 case 2 -> "->";
189 case 3 -> "|";
190 case 4 -> "^";
191 default -> "&";
192 };
193 while (peek().equals(op)) {
194 at++;
195 left = node(op, left, expression(level == 2 ? level : level + 1));
196 }
197 return left;
198 }
199 Node parse() {
200 Node n = expression(1);
201 if (at != tokens.size()) unexpected();
202 return n;
203 }
204 }

210 Node nnf(Node n, boolean positive) {
211 long key = 2L * n.id + (positive ? 1 : 0);
212 Node old = normalForms.get(key);
213 if (old != null) return old;
214 Node result;
215 if (n.left == null) result = n.op.equals("0") || n.op.equals("1") ? leaf(positive ? n.op : n.op.equals("0") ? "1" : "0") : positive ? n : not(n);
216 else if (n.op.equals("~")) result = nnf(n.left, !positive);
217 else if (n.op.equals("&") || n.op.equals("|")) result = node(positive ? n.op : n.op.equals("&") ? "|" : "&", nnf(n.left, positive), nnf(n.right, positive));
218 else if (n.op.equals("->")) result = node(positive ? "|" : "&", nnf(n.left, !positive), nnf(n.right, positive));
219 else {
220 boolean equal = n.op.equals("<->") == positive;
221 result = node("|", node("&", nnf(n.left, true), nnf(n.right, equal)), node("&", nnf(n.left, false), nnf(n.right, !equal)));
222 }
223 normalForms.put(key, result);
224 return result;
225 }

227 Node dual(Node n, Map<Integer, Node> memo) {
228 Node old = memo.get(n.id);
229 if (old != null) return old;
230 Node result;
231 if (n.left == null) result = n.op.equals("0") ? leaf("1") : n.op.equals("1") ? leaf("0") : n;
232 else if (n.op.equals("~")) result = not(dual(n.left, memo));
233 else {
234 Node a = dual(n.left, memo), b = dual(n.right, memo);
235 result = switch (n.op) {
236 case "&" -> node("|", a, b);
237 case "|" -> node("&", a, b);
238 case "^" -> node("<->", a, b);
239 case "<->" -> node("^", a, b);
240 default -> node("&", not(a), b);
241 };
242 }
243 memo.put(n.id, result);
244 return result;
245 }

255 Node simplifyBinary(String op, Node a, Node b) {
256 boolean a0 = a.op.equals("0"), a1 = a.op.equals("1"), b0 = b.op.equals("0"), b1 = b.op.equals("1");
257 switch (op) {
258 case "&":
259 if (a0 || b0) return leaf("0");
260 if (a1) return b;
261 if (b1 || a == b) return a;
262 if (complement(a, b)) return leaf("0");
263 break;
264 case "|":
265 if (a1 || b1) return leaf("1");
266 if (a0) return b;
267 if (b0 || a == b) return a;
268 if (complement(a, b)) return leaf("1");
269 break;
270 case "^":
271 if (a0) return b;
272 if (b0) return a;
273 if (a1) return reduceNot(b);
274 if (b1) return reduceNot(a);
275 if (a == b) return leaf("0");
276 if (complement(a, b)) return leaf("1");
277 break;
278 case "->":
279 if (a0 || b1) return leaf("1");
280 if (a1) return b;
281 if (b0) return reduceNot(a);
282 if (a == b) return leaf("1");
283 if (complement(a, b)) return b;
284 break;
285 case "<->":
286 if (a1) return b;
287 if (b1) return a;
288 if (a0) return reduceNot(b);
289 if (b0) return reduceNot(a);
290 if (a == b) return leaf("1");
291 if (complement(a, b)) return leaf("0");
292 break;
293 }
294 return node(op, a, b);
295 }

308 static int evaluate(Node n, int trueVariables, Map<Integer, Integer> memo) {
309 Integer old = memo.get(n.id);
310 if (old != null) return old;
311 int result;
312 if (n.left == null) result = upper(n.op.charAt(0)) ? (trueVariables >>> (n.op.charAt(0) - 'A')) & 1 : n.op.charAt(0) - '0';
313 else if (n.op.equals("~")) result = 1 - evaluate(n.left, trueVariables, memo);
314 else {
315 int a = evaluate(n.left, trueVariables, memo), b = evaluate(n.right, trueVariables, memo);
316 result = switch (n.op) {
317 case "&" -> a & b;
318 case "|" -> a | b;
319 case "^" -> a ^ b;
320 case "->" -> (1 - a) | b;
321 default -> 1 - (a ^ b);
322 };
323 }
324 memo.put(n.id, result);
325 return result;
326 }

393 static void implicants(int at, int n, int mask, int bits, String prefix, List<Implicant> out, BitSet allowed) {
394 if (at == n) {
395 BitSet coverage = new BitSet(1 << n);
396 for (int row = 0; row < 1 << n; row++) if (((row ^ bits) & mask) == 0) {
397 if (!allowed.get(row)) return;
398 coverage.set(row);
399 }
400 out.add(new Implicant(prefix, mask, bits, coverage));
401 return;
402 }
403 int bit = 1 << (n - 1 - at);
404 implicants(at + 1, n, mask | bit, bits | bit, prefix + "1", out, allowed);
405 implicants(at + 1, n, mask | bit, bits, prefix + "0", out, allowed);
406 implicants(at + 1, n, mask, bits, prefix + "-", out, allowed);
407 }

432 static List<String> minimize(String vars, int[] f, int[] dc, boolean pos) {
433 int n = vars.length();
434 BitSet on = new BitSet(f.length), allowed = new BitSet(f.length);
435 for (int r = 0; r < f.length; r++) {
436 if ((pos ? 1 - f[r] : f[r]) == 1 && (dc == null || dc[r] == 0)) on.set(r);
437 if (on.get(r) || dc != null && dc[r] == 1) allowed.set(r);
438 }
439 List<Implicant> all = new ArrayList<>();
440 implicants(0, n, 0, 0, "", all, allowed);
441 Set<Integer> keys = new HashSet<>();
442 for (Implicant p : all) keys.add((p.mask << 8) | p.bits);
443 List<Implicant> primes = new ArrayList<>();
444 int[] stages = new int[n + 1], primeStages = new int[n + 1];
445 for (Implicant p : all) {
446 int stage = n - Integer.bitCount(p.mask);
447 stages[stage]++;
448 boolean prime = true;
449 for (int bit = 1; bit < 1 << n; bit <<= 1) if ((p.mask & bit) != 0 && keys.contains(((p.mask ^ bit) << 8) | (p.bits & ~bit))) {
450 prime = false;
451 break;
452 }
453 if (prime) {
454 primeStages[stage]++;
455 if (p.coverage.intersects(on)) primes.add(p);
456 }
457 }
458 primes.sort((a, b) -> patternCompare(a.pattern, b.pattern));
459 Set<Implicant> essentialSet = new HashSet<>();
460 for (int r = on.nextSetBit(0); r >= 0; r = on.nextSetBit(r + 1)) {
461 Implicant only = null;
462 int count = 0;
463 for (Implicant p : primes) if (p.coverage.get(r)) {
464 only = p;
465 count++;
466 }
467 if (count == 1) essentialSet.add(only);
468 }
469 List<Implicant> essential = primes.stream().filter(essentialSet::contains).toList(), optional = primes.stream().filter(p -> !essentialSet.contains(p)).toList();
470 List<Implicant> best = null;
471 for (int mask = 0; mask < 1 << optional.size(); mask++) {
472 List<Implicant> candidate = new ArrayList<>(essential);
473 BitSet covered = new BitSet(f.length);
474 for (int i = 0; i < optional.size(); i++) if ((mask & 1 << i) != 0) candidate.add(optional.get(i));
475 if (best != null && candidate.size() > best.size()) continue;
476 for (Implicant p : candidate) covered.or(p.coverage);
477 BitSet missing = (BitSet)on.clone();
478 missing.andNot(covered);
479 if (missing.isEmpty() && better(candidate, best, pos)) best = candidate;
480 }
481 List<String> out = new ArrayList<>();
482 if (!pos) {
483 for (int stage = 0; stage <= n; stage++) {
484 out.add("stage " + stage + ": implicants " + stages[stage] + " primes " + primeStages[stage]);
485 if (stage == n || stages[stage + 1] == 0) break;
486 }
487 out.add("primes: " + (primes.isEmpty() ? "none" : String.join(", ", coverPatterns(primes, false).stream().map(p -> term(p, vars, false)).toList())));
488 out.add("essential: " + (essential.isEmpty() ? "none" : String.join(", ", coverPatterns(essential, false).stream().map(p -> term(p, vars, false)).toList())));
489 }
490 out.add("minimal: " + form(coverPatterns(best, pos), vars, pos));
491 out.add((pos ? "clauses " : "terms ") + best.size() + " literals " + coverLiterals(best));
492 return out;
493 }

577 List<String> command(String line) {

579 String command = space < 0 ? line : line.substring(0, space), arg = space < 0 ? "" : line.substring(space + 1);
580 Set<String> single = Set.of("PRINT", "TABLE", "MINTERMS", "MAXTERMS", "DNF", "CNF", "NNF", "DUAL", "ANF", "SIMPLIFY", "MINIMIZE", "MINPOS", "KMAP", "SAT", "TAUT", "COUNT", "PROPS", "BDD", "CLAUSES", "DP");
581 Set<String> split = Set.of("RESTRICT", "COFACTOR", "MINDC", "EQUIV", "IMPLIES");
582 if (!single.contains(command) && !split.contains(command) && !Set.of("DEF", "DEFSUM", "DEL", "LIST").contains(command)) fail("unknown command " + command);
583 if (command.equals("LIST") && space >= 0) fail("unexpected argument");
584 if (!command.equals("LIST") && space < 0) fail("missing argument");
585 int semicolon = arg.indexOf(';');
586 if (split.contains(command) && semicolon < 0) fail("missing ';'");
587 List<String> out = new ArrayList<>();
588 if (command.equals("LIST")) {
589 out.add("names: " + definitions.size());
590 for (var e : definitions.entrySet()) out.add(e.getKey() + " = " + print(e.getValue()));

592 }
593 if (command.equals("DEL")) {
594 if (!name(arg)) fail("invalid name '" + arg + "'");
595 if (!definitions.containsKey(arg)) fail("undefined name '" + arg + "'");
596 definitions.remove(arg);
597 return List.of("deleted " + arg);
598 }
599 if (command.equals("DEF")) {
600 int at = arg.indexOf(' ');
601 if (at < 0) fail("missing argument");
602 String name = arg.substring(0, at);
603 if (!name(name)) fail("invalid name '" + name + "'");
604 Node formula = parse(arg.substring(at + 1), name.length() + 6);
605 define(name, formula, out);
606 return out;
607 }
608 if (command.equals("DEFSUM")) {
609 if (semicolon < 0) fail("missing ';'");
610 String[] front = words(arg.substring(0, semicolon)), back = words(arg.substring(semicolon + 1));
611 if (front.length == 0) fail("missing argument");
612 if (!name(front[0])) fail("invalid name '" + front[0] + "'");
613 if (front.length == 1) fail("bad variable list");
614 for (int i = 1; i < front.length; i++) if (front[i].length() != 1 || !upper(front[i].charAt(0)) || i > 1 && front[i - 1].compareTo(front[i]) >= 0) fail("bad variable list");
615 int n = front.length - 1;
616 if (n >= 9) fail("too many variables: " + n);
617 int previous = -1;
618 Node result = null;
619 for (String word : back) {
620 if (word.isEmpty() || word.length() > 1 && word.charAt(0) == '0' || !word.chars().allMatch(c -> c >= '0' && c <= '9')) fail("bad minterm '" + word + "'");
621 int row = Integer.parseInt(word);
622 if (row >= 1 << n || row <= previous) fail("bad minterm '" + word + "'");
623 previous = row;
624 Node term = null;
625 for (int i = 0; i < n; i++) {
626 Node literal = leaf(front[i + 1]);
627 if ((row & 1 << (n - i - 1)) == 0) literal = not(literal);
628 term = term == null ? literal : node("&", term, literal);
629 }
630 result = result == null ? term : node("|", result, term);
631 }
632 define(front[0], result == null ? leaf("0") : result, out);
633 return out;
634 }
635 int start = command.length() + 2;
636 String first = split.contains(command) ? arg.substring(0, semicolon) : arg, second = split.contains(command) ? arg.substring(semicolon + 1) : "";
637 Node f = parse(first, start), g = null;
638 if (Set.of("MINDC", "EQUIV", "IMPLIES").contains(command)) g = parse(second, start + semicolon + 1);
639 if (command.equals("RESTRICT")) {
640 String[] assignments = words(second);
641 if (assignments.length == 0) fail("missing assignment");
642 Map<Character, Integer> values = new HashMap<>();
643 for (String a : assignments) {
644 if (a.length() != 3 || !upper(a.charAt(0)) || a.charAt(1) != '=' || "01".indexOf(a.charAt(2)) < 0) fail("bad assignment '" + a + "'");
645 if (values.putIfAbsent(a.charAt(0), a.charAt(2) - '0') != null) fail("duplicate variable " + a.charAt(0));
646 }
647 return List.of("restricted: " + print(simplify(f, values, new HashMap<>())));
648 }
649 if (command.equals("COFACTOR")) {
650 String[] x = words(second);
651 if (x.length != 1 || x[0].length() != 1 || !upper(x[0].charAt(0))) fail("bad variable");
652 char c = x[0].charAt(0);
653 Node f1 = simplify(f, Map.of(c, 1), new HashMap<>()), f0 = simplify(f, Map.of(c, 0), new HashMap<>());
654 return List.of("cofactor " + c + "=1: " + print(f1), "cofactor " + c + "=0: " + print(f0), "difference: " + print(simplifyBinary("^", f1, f0)));
655 }
656 String vars = variables(f.variables | (g == null ? 0 : g.variables));
657 int n = vars.length();
658 if (command.equals("PRINT")) return List.of("formula: " + print(f), "variables: " + (vars.isEmpty() ? "none" : String.join(" ", vars.split(""))), "size " + size(f) + " depth " + depth(f));
659 if (command.equals("NNF")) return List.of("nnf: " + print(nnf(f, true)));
660 if (command.equals("DUAL")) return List.of("dual: " + print(dual(f, new HashMap<>())));
661 if (command.equals("SIMPLIFY")) return List.of("simplified: " + print(simplify(f, Map.of(), new HashMap<>())));
662 if (command.equals("CLAUSES") || command.equals("DP")) {
663 Set<Clause> clauses = clauses(nnf(f, true), new HashMap<>());
664 if (command.equals("CLAUSES")) return List.of("clauses: " + form(clausePatterns(clauses, vars), vars, true));
665 out.add("start: clauses " + clauses.size());
666 for (char c : vars.toCharArray()) {
667 if (clauses.isEmpty() || clauses.contains(new Clause(0, 0))) break;
668 int bit = 1 << (c - 'A');
669 List<Clause> positive = clauses.stream().filter(x -> (x.positive & bit) != 0).toList(), negative = clauses.stream().filter(x -> (x.negative & bit) != 0).toList();
670 Set<Clause> resolvents = new HashSet<>(), next = new HashSet<>();
671 for (Clause a : positive) for (Clause b : negative) {
672 Clause r = new Clause((a.positive | b.positive) & ~bit, (a.negative | b.negative) & ~bit);
673 if (!r.tautology()) resolvents.add(r);
674 }
675 for (Clause a : clauses) if (((a.positive | a.negative) & bit) == 0) next.add(a);
676 next.addAll(resolvents);
677 clauses = tidy(next);
678 out.add("eliminate " + c + ": resolvents " + resolvents.size() + " clauses " + clauses.size());
679 }
680 out.add("result: " + (clauses.contains(new Clause(0, 0)) ? "unsatisfiable" : "satisfiable"));
681 return out;
682 }
683 if (n >= 9) fail("too many variables: " + n);
684 int[] table = truth(f, vars), other = g == null ? null : truth(g, vars);
685 int rows = table.length;
686 if (command.equals("MINIMIZE") || command.equals("MINDC") || command.equals("MINPOS")) return minimize(vars, table, other, command.equals("MINPOS"));
687 switch (command) {
688 case "TABLE" -> {
689 out.add((n == 0 ? "" : String.join(" ", vars.split("")) + " ") + "| F");
690 for (int r = 0; r < rows; r++) out.add((n == 0 ? "" : String.join(" ", binary(r, n).split("")) + " ") + "| " + table[r]);
691 }
692 case "MINTERMS", "MAXTERMS", "DNF", "CNF" -> {
693 boolean positive = command.equals("MINTERMS") || command.equals("DNF");
694 List<String> patterns = new ArrayList<>(), numbers = new ArrayList<>();
695 for (int r = 0; r < rows; r++) if (table[r] == (positive ? 1 : 0)) {
696 String p = binary(r, n);
697 patterns.add(positive ? p : opposite(p));
698 numbers.add(Integer.toString(r));
699 }
700 String label = command.toLowerCase(java.util.Locale.ROOT);
701 out.add(label + ": " + (command.equals("DNF") || command.equals("CNF") ? form(patterns, vars, !positive) : numbers.isEmpty() ? "none" : String.join(" ", numbers)));
702 }
703 case "ANF" -> {
704 int[] coeff = coefficients(table);
705 List<String> patterns = new ArrayList<>();
706 for (int r = 0; r < rows; r++) if (coeff[r] == 1) patterns.add(binary(r, n).replace('0', '-'));
707 patterns.sort((a, b) -> literals(a) != literals(b) ? Integer.compare(literals(a), literals(b)) : patternCompare(a, b));
708 out.add("anf: " + (patterns.isEmpty() ? "0" : String.join(" ^ ", patterns.stream().map(p -> term(p, vars, false)).toList())));
709 }
710 case "SAT", "TAUT" -> {
711 boolean sat = command.equals("SAT");
712 int firstRow = -1;
713 for (int r = 0; r < rows; r++) if (table[r] == (sat ? 1 : 0)) {
714 firstRow = r;
715 break;
716 }
717 out.add(firstRow < 0 ? sat ? "unsatisfiable" : "tautology" : (sat ? "satisfiable: " : "not a tautology: ") + assignment(firstRow, vars));
718 }
719 case "COUNT" -> {
720 int count = 0;
721 for (int v : table) count += v;
722 out.add("models: " + count + " of " + rows);
723 }
724 case "EQUIV", "IMPLIES" -> {
725 boolean equivalent = command.equals("EQUIV");
726 int firstRow = -1;
727 for (int r = 0; r < rows; r++) if (equivalent ? table[r] != other[r] : table[r] == 1 && other[r] == 0) {
728 firstRow = r;
729 break;
730 }
731 out.add(firstRow < 0 ? equivalent ? "equivalent" : "implies" : equivalent ? "not equivalent: " + assignment(firstRow, vars) + " f=" + table[firstRow] + " g=" + other[firstRow] : "does not imply: " + assignment(firstRow, vars));
732 }
733 case "PROPS" -> {
734 List<String> dependencies = new ArrayList<>();
735 boolean selfDual = true, monotone = true, linear = true, symmetric = true;
736 for (int i = 0; i < n; i++) {
737 int bit = 1 << (n - 1 - i);
738 for (int r = 0; r < rows; r++) if (table[r] != table[r ^ bit]) {
739 dependencies.add(Character.toString(vars.charAt(i)));
740 break;
741 }
742 }
743 int[] coeff = coefficients(table), weights = new int[n + 1];
744 java.util.Arrays.fill(weights, -1);
745 for (int r = 0; r < rows; r++) {
746 selfDual &= table[r] != table[rows - 1 - r];
747 if (coeff[r] == 1 && Integer.bitCount(r) > 1) linear = false;
748 int weight = Integer.bitCount(r);
749 if (weights[weight] >= 0 && weights[weight] != table[r]) symmetric = false;
750 weights[weight] = table[r];
751 for (int bit = 1; bit < rows; bit <<= 1) if ((r & bit) == 0 && table[r] > table[r | bit]) monotone = false;
752 }
753 out.add("depends: " + (dependencies.isEmpty() ? "none" : String.join(" ", dependencies)));
754 String[] labels = {"zero-preserving", "one-preserving", "self-dual", "monotone", "linear", "symmetric"};
755 boolean[] values = {table[0] == 0, table[rows - 1] == 1, selfDual, monotone, linear, symmetric};
756 for (int i = 0; i < labels.length; i++) out.add(labels[i] + ": " + (values[i] ? "yes" : "no"));
757 }
758 case "KMAP" -> {
759 if (n < 2 || n > 4) fail("kmap needs 2 to 4 variables");
760 int p = n / 2, q = n - p;
761 String label = vars.substring(0, p) + "\\" + vars.substring(p);
762 StringBuilder header = new StringBuilder(label);
763 for (int i = 0; i < 1 << q; i++) header.append(' ').append(binary(i ^ i / 2, q));
764 out.add(header.toString());
765 for (int i = 0; i < 1 << p; i++) {
766 int r = i ^ i / 2;
767 StringBuilder row = new StringBuilder(binary(r, p) + " ".repeat(q + 1));
768 for (int j = 0; j < 1 << q; j++) row.append(" ".repeat(q)).append(table[(r << q) | (j ^ j / 2)]);
769 out.add(row.toString());
770 }
771 }
772 case "BDD" -> {
773 List<Decision> decisions = new ArrayList<>();
774 int root = bdd(table, 0, rows, 0, decisions, new HashMap<>());
775 int[] paths = new int[decisions.size() + 2];
776 paths[1] = 1;
777 out.add("nodes: " + decisions.size());
778 for (int i = 0; i < decisions.size(); i++) {
779 Decision d = decisions.get(i);
780 paths[i + 2] = paths[d.low] + paths[d.high];
781 out.add((i + 2) + " = ite(" + vars.charAt(d.variable) + ", " + d.high + ", " + d.low + ")");
782 }
783 out.add("root: " + root);
784 out.add("paths: " + paths[root]);
785 }
786 default -> throw new IllegalStateException();
787 }
788 return out;
789 }
