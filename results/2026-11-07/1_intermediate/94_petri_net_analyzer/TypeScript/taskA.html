39 m = matrix[0].length;

44 if (r[j] > 0)
45 for (const s of rows)

47 let v = r.map((x, k) => -s[j] * x + r[j] * s[k]);
48 const d = v.reduce((g, x) => gcd(g, x), 0);

51 }

625 case "FIRE": {
626 const ts = args.map((a) => {
627 const t = this.t.indexOf(a);
628 if (t < 0) fail("unknown transition " + a);
629 return t;
630 });
631 for (let i = 0; i < ts.length; ++i) {
632 const reason = this.reason(this.current, ts[i]);
633 if (reason) {
634 out(
635 `step ${i + 1}: ${args[i]} not enabled (${reason}) at ${marking(this.current)}`,
636 );
637 return;
638 }
639 this.commit(ts[i]);
640 }
641 out(`fired ${ts.length} -> ${marking(this.current)}`);
642 break;
643 }
644 case "UNDO": {
645 const k = integer(args[0], 1, 1000);
646 if (k < 0) fail("invalid count " + args[0]);
647 if (k > this.history.length)
648 fail(`cannot undo ${k}: history size ${this.history.length}`);
649 this.current = this.history[this.history.length - k];
650 this.history.length -= k;
651 out("marking " + marking(this.current));
652 break;
653 }
654 case "SIMULATE": {
655 const steps = integer(args[0], 1, 1000);
656 let x = integer(args[1], 1, 2147483646);
657 if (steps < 0) fail("invalid steps " + args[0]);
658 if (x < 0) fail("invalid seed " + args[1]);
659 if (!["PRIORITY", "RANDOM"].includes(args[2])) fail("invalid mode " + args[2]);
660 const counts = this.t.map(() => 0);
661 let done = 0;
662 for (; done < steps; ++done) {
663 const e = this.enabled(this.current);
664 if (!e.length) {
665 out(`dead (steps=${done})`);
666 break;
667 }
668 let t = e[0];
669 if (args[2] === "RANDOM") {
670 x = (x * 48271) % 2147483647;
671 t = e[x % e.length];
672 } else for (const u of e) if (this.prio[u] > this.prio[t]) t = u;
673 this.commit(t);
674 ++counts[t];
675 out(`step ${done + 1}: ${this.t[t]} -> ${marking(this.current)}`);
676 }
677 if (done === steps) out(`completed (steps=${steps})`);
678 out("counts: " + this.t.map((n, t) => n + "=" + counts[t]).join(" "));
679 break;
680 }
681 case "CONFLICTS": {
682 const structural = [],
683 effective = [];
684 for (let u = 0; u < this.t.length; ++u)
685 for (let v = u + 1; v < this.t.length; ++v) {
686 const pair = this.t[u] + "/" + this.t[v];
687 if (this.p.some((_, p) => this.pre[p][u] && this.pre[p][v]))
688 structural.push(pair);
689 if (
690 !this.reason(this.current, u) &&
691 !this.reason(this.current, v) &&
692 (this.reason(this.fire(this.current, u), v) ||
693 this.reason(this.fire(this.current, v), u))
694 )
695 effective.push(pair);
696 }
697 out("structural: " + listing(structural));
698 out("effective: " + listing(effective));
699 break;
700 }
701 case "REACH":
702 case "GRAPH":
703 this.reach(limit, cmd === "GRAPH");
704 break;
705 case "COVER":
706 this.cover(limit);
707 break;
708 case "PATH": {
709 const entries = args[1].split(",");
710 if (entries.length !== this.p.length) fail(`target needs ${this.p.length} entries`);
711 const target: Target = entries.map((s) => {
712 if (s === "*") return [0, -1];
713 const plus = s.endsWith("+"),
714 v = integer(plus ? s.slice(0, -1) : s, 0, 9999999999);
715 if (v < 0) fail("invalid target entry " + s);
716 return [v, plus ? -1 : v];
717 });
718 const g = this.graph(this.current, limit, target);
719 if (g.found >= 0) {
720 const path = this.names(this.path(g, g.found));
721 out(
722 `reached ${marking(g.states[g.found])} length=${path.length}` +
723 (path.length ? ": " + path.join(" ") : ""),
724 );
725 } else if (g.aborted) out(`search limit ${limit} exceeded`);
726 else out(`unreachable (markings explored: ${g.states.length})`);
727 break;
728 }
729 case "INCIDENCE": {
730 const rows = [
731 ["", ...this.t],
732 ...this.p.map((name, p) => [
733 name,
734 ...this.t.map((_, t) => String(this.post[p][t] - this.pre[p][t])),
735 ]),
736 ],
737 width = rows[0].map((_, i) => Math.max(...rows.map((r) => r[i].length)));
738 for (const r of rows)
739 out(
740 r
741 .map((s, i) => (i ? s.padStart(width[i]) : s.padEnd(width[i])))
742 .join(" ")
743 .trimEnd(),
744 );
745 break;
746 }
747 case "PINV":
748 case "TINV":
749 this.invariants(cmd === "TINV");
750 break;
751 case "SIPHONS":
752 case "TRAPS": {
753 if (this.p.length > 12) fail("more than 12 places");
754 const sets = this.sets(cmd === "TRAPS");
755 out(`minimal ${cmd === "TRAPS" ? "traps" : "siphons"}: ${sets.length}`);
756 for (const s of sets)
757 out(
758 " " +
759 this.setText(s) +
760 " tokens=" +
761 this.current.reduce((sum, v, p) => sum + (s & (1 << p) ? v : 0), 0),
762 );
763 break;
764 }
765 case "CLASSIFY":
766 this.classify();
767 break;
768 case "WORKFLOW":
769 this.workflow(limit);
770 break;
771 case "SHOW":
772 this.p.forEach((name, p) =>
773 out(
774 `place ${name} tokens=${this.initial[p]} capacity=${this.cap[p] || "none"}`,
775 ),
776 );
777 this.t.forEach((name, t) => {
778 const list = (mat: number[][]) =>
779 this.p
780 .map((n, p) => (mat[p][t] ? n + ":" + mat[p][t] : ""))
781 .filter(Boolean)
782 .join(",") || "-";
783 out(
784 `transition ${name} priority=${this.prio[t]} pre=${list(this.pre)} post=${list(this.post)} inhibit=${list(this.inh)}`,
785 );
786 });
787 break;
788 }
789 }
790}
791const blocks = +lines[cursor++];
792for (let b = 0; b < blocks; ++b) {
793 const net = new Net(lines[cursor++].split(" ")[1]);
794 while (lines[cursor] !== "END NET") {
795 try {
796 net.define(lines[cursor++].split(" "));
797 } catch (e) {
798 out("error: " + (e instanceof Error ? e.message : String(e)));
799 }
800 }
801 ++cursor;
802 net.current = net.initial.slice();
803 out(
804 `net ${net.name}: places=${net.p.length} transitions=${net.t.length} arcs=${net.arcs} inhibitors=${net.inhibitors}`,
805 );
806 const n = +lines[cursor++];
807 for (let i = 0; i < n; ++i) {
808 try {
809 net.execute(lines[cursor++].split(" "));
810 } catch (e) {
811 out("error: " + (e instanceof Error ? e.message : String(e)));
812 }
813 }
