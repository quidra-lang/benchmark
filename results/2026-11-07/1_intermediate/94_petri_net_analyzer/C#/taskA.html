36 static List<long[]> Flows(long[][] a)
37 {
38 int n = a.Length, m = a[0].Length;
39 List<long[]> rows = a.Select((r,i) => r.Concat(Enumerable.Range(0,n).Select(j => i == j ? 1L : 0L)).ToArray()).ToList();
40 for (int j = 0; j < m; ++j)
41 {
42 Dictionary<string,long[]> unique = [];
43 foreach (long[] r in rows) if (r[j] == 0) unique[string.Join(",",r)] = r;
44 foreach (long[] r in rows) if (r[j] > 0) foreach (long[] s in rows) if (s[j] < 0)
45 {
46 long[] v = r.Select((x,k) => -s[j]*x+r[j]*s[k]).ToArray();
47 long d = v.Aggregate(0L,Gcd);
48 for (int k = 0; k < v.Length; ++k) v[k] /= d;
49 unique[string.Join(",",v)] = v;
50 }
51 List<long[]> candidates = unique.Values.ToList();
52 int[] supports = candidates.Select(r => Enumerable.Range(0,n).Where(i => r[m+i] > 0).Aggregate(0,(s,i) => s | 1<<i)).ToArray();
53 rows = candidates.Where((r,i) => !supports.Any(s => s != supports[i] && (s & supports[i]) == s)).ToList();
54 }
55 List<long[]> result = rows.Select(r => r[m..]).ToList();
56 result.Sort((a,b) => Compare(b,a));
57 return result;

320 List<int>? Realize(long[] counts)
321 {
322 long[] used = new long[counts.Length];
323 List<long[]> stack = [Current];
324 List<int> choices = [0], path = [];
325 HashSet<string> seen = [];
326 long total = counts.Sum();
327 while (stack.Count > 0) {
328 int d = path.Count;
329 if (d == total) return path;
330 int chosen = -1;
331 for (int t = choices[d]; t < counts.Length; ++t) {
332 choices[d] = t+1;
333 if (used[t] >= counts[t] || Reason(stack[d],t).Length > 0) continue;
334 ++used[t];
335 bool fresh = seen.Add(string.Join(",",used));
336 --used[t];
337 if (fresh) {
338 chosen = t;
339 break;
340 }
341 }
342 if (chosen >= 0) {
343 ++used[chosen];
344 path.Add(chosen);
345 stack.Add(Fire(stack[d],chosen));
346 choices.Add(0);
347 } else {
348 stack.RemoveAt(stack.Count-1);
349 choices.RemoveAt(choices.Count-1);
350 if (path.Count > 0) {
351 --used[path[^1]];
352 path.RemoveAt(path.Count-1);
353 }
354 }
355 }
356 return null;

422 void Workflow(int limit)
423 {

432 return;
433 }

437 for (int p = 0; p < P; ++p) for (int t = 0; t < T; ++t) {

460 }

465 for (int u = 0; u < g.States.Count; ++u) {
466 if (g.States[u][o] >= 1 && !g.States[u].SequenceEqual(goal)) proper = false;
467 foreach (Edge e in g.Edges[u]) {
468 rev[e.Target].Add(u);
469 used[e.Transition] = true;
470 }
471 }

473 Say("option to complete: " + Yes(option));
474 Say("proper completion: " + Yes(proper));
475 Say("no dead transitions: " + Yes(live));
476 Say("sound: " + Yes(option && proper && live));

478 public void Execute(string[] w)
479 {
480 Dictionary<string,int> arity = new() { ["MARKING"]=0,["RESET"]=0,["SET"]=2,["ENABLED"]=0,["FIRE"]=-1,["UNDO"]=1,["SIMULATE"]=3,["CONFLICTS"]=0,["REACH"]=1,["GRAPH"]=1,["COVER"]=1,["PATH"]=2,["INCIDENCE"]=0,["PINV"]=0,["TINV"]=0,["SIPHONS"]=0,["TRAPS"]=0,["CLASSIFY"]=0,["WORKFLOW"]=1,["SHOW"]=0 };
481 string cmd = w[0];
482 if (!arity.TryGetValue(cmd,out int expected)) Bad("unknown command " + cmd);
483 if (expected < 0 ? w.Length < 2 : w.Length != expected+1) Bad("wrong number of arguments");
484 int limit = 0;
485 if (new[]{
486 "REACH","GRAPH","COVER","PATH","WORKFLOW"
487 }.Contains(cmd)) {
488 limit = (int)Number(w[1],1,2000);
489 if (limit < 0) Bad("invalid limit " + w[1]);
490 }
491 switch (cmd)
492 {
493 case "MARKING": case "RESET": if (cmd == "RESET") {
494 Current = Initial.ToArray();
495 history.Clear();
496 }
497 Say("marking " + Mark(Current));
498 break;
499 case "SET": {
500 int p = P(w[1]);
501 if (p < 0) Bad("unknown place " + w[1]);
502 long v = Number(w[2],0,999);
503 if (v < 0) Bad("invalid tokens " + w[2]);
504 if (Capacities[p] > 0 && v > Capacities[p]) Bad("tokens exceed capacity of " + w[1]);
505 Current = (long[])Current.Clone();
506 Current[p] = v;
507 history.Clear();
508 Say("marking " + Mark(Current));
509 break;
510 }
511 case "ENABLED": Say("enabled: " + Listed(Names(Enabled(Current))));
512 break;
513 case "FIRE": {
514 List<int> ts = [];
515 foreach (string name in w[1..]) {
516 int t = T(name);
517 if (t < 0) Bad("unknown transition " + name);
518 ts.Add(t);
519 }
520 for (int i = 0; i < ts.Count; ++i) {
521 string r = Reason(Current,ts[i]);
522 if (r.Length > 0) {
523 Say($"step {i+1}: {Transitions[ts[i]]} not enabled ({r}) at {Mark(Current)}");
524 return;
525 }
526 Commit(ts[i]);
527 }
528 Say($"fired {ts.Count} -> {Mark(Current)}");
529 break;
530 }
531 case "UNDO": {
532 int k = (int)Number(w[1],1,1000);
533 if (k < 0) Bad("invalid count " + w[1]);
534 if (k > history.Count) Bad($"cannot undo {k}: history size {history.Count}");
535 Current = history[history.Count-k];
536 history.RemoveRange(history.Count-k,k);
537 Say("marking " + Mark(Current));
538 break;
539 }
540 case "SIMULATE": {
541 int steps = (int)Number(w[1],1,1000);
542 long x = Number(w[2],1,2147483646);
543 if (steps < 0) Bad("invalid steps " + w[1]);
544 if (x < 0) Bad("invalid seed " + w[2]);
545 if (w[3] is not ("PRIORITY" or "RANDOM")) Bad("invalid mode " + w[3]);
546 int[] counts = new int[Transitions.Count];
547 int done = 0;
548 for (; done < steps; ++done) {
549 List<int> es = Enabled(Current);
550 if (es.Count == 0) {
551 Say($"dead (steps={done})");
552 break;
553 }
554 int t = es[0];
555 if (w[3] == "RANDOM") {
556 x = x*48271%2147483647;
557 t = es[(int)(x%es.Count)];
558 } else foreach (int u in es) if (Priorities[u] > Priorities[t]) t = u;
559 Commit(t);
560 ++counts[t];
561 Say($"step {done+1}: {Transitions[t]} -> {Mark(Current)}");
562 }
563 if (done == steps) Say($"completed (steps={steps})");
564 Say("counts: " + string.Join(" ",Transitions.Select((name,t) => name+"="+counts[t])));
565 break;
566 }
567 case "CONFLICTS": {
568 List<string> s = [], e = [];
569 for (int u = 0; u < Transitions.Count; ++u) for (int v = u+1; v < Transitions.Count; ++v) {
570 string pair = Transitions[u]+"/"+Transitions[v];
571 if (Enumerable.Range(0,Places.Count).Any(p => pre[p,u] > 0 && pre[p,v] > 0)) s.Add(pair);
572 if (Reason(Current,u).Length == 0 && Reason(Current,v).Length == 0 && (Reason(Fire(Current,u),v).Length > 0 || Reason(Fire(Current,v),u).Length > 0)) e.Add(pair);
573 }
574 Say("structural: " + Listed(s));
575 Say("effective: " + Listed(e));
576 break;
577 }
578 case "REACH": case "GRAPH": Reach(limit,cmd == "GRAPH");
579 break;
580 case "COVER": Cover(limit);
581 break;
582 case "PATH": {
583 string[] entries = w[2].Split(',');
584 if (entries.Length != Places.Count) Bad($"target needs {Places.Count} entries");
585 long[][] target = new long[entries.Length][];
586 for (int p = 0; p < entries.Length; ++p) {
587 string s = entries[p];
588 if (s == "*") {
589 target[p] = [0,-1];
590 continue;
591 }
592 bool plus = s.EndsWith('+');
593 long v = Number(plus ? s[..^1] : s,0,9999999999);
594 if (v < 0) Bad("invalid target entry " + s);
595 target[p] = [v,plus ? -1 : v];
596 }
597 Graph g = Build(Current,limit,target);
598 if (g.Found >= 0) {
599 List<string> a = Names(Path(g,g.Found));
600 Say($"reached {Mark(g.States[g.Found])} length={a.Count}" + (a.Count == 0 ? "" : ": "+string.Join(" ",a)));
601 } else if (g.Aborted) Say($"search limit {limit} exceeded");
602 else Say($"unreachable (markings explored: {g.States.Count})");
603 break;
604 }
605 case "INCIDENCE": {
606 List<List<string>> rows = [["",..Transitions]];
607 for (int p = 0; p < Places.Count; ++p) rows.Add([Places[p],..Enumerable.Range(0,Transitions.Count).Select(t => (post[p,t]-pre[p,t]).ToString())]);
608 int[] width = Enumerable.Range(0,Transitions.Count+1).Select(i => rows.Max(r => r[i].Length)).ToArray();
609 foreach (List<string> r in rows) Say(string.Join(" ",r.Select((s,i) => i == 0 ? s.PadRight(width[i]) : s.PadLeft(width[i]))).TrimEnd());
610 break;
611 }
612 case "PINV": case "TINV": Invariants(cmd == "TINV");
613 break;
614 case "SIPHONS": case "TRAPS": {
615 if (Places.Count > 12) Bad("more than 12 places");
616 List<int> ss = Sets(cmd == "TRAPS");
617 Say("minimal " + (cmd == "TRAPS" ? "traps" : "siphons") + $": {ss.Count}");
618 foreach (int s in ss) Say($" {SetText(s)} tokens={Bits(s).Sum(p => Current[p])}");
619 break;
620 }
621 case "CLASSIFY": Classify();
622 break;
623 case "WORKFLOW": Workflow(limit);
624 break;
625 case "SHOW": for (int p = 0; p < Places.Count; ++p) Say($"place {Places[p]} tokens={Initial[p]} capacity=" + (Capacities[p] == 0 ? "none" : Capacities[p].ToString()));
626 for (int t = 0; t < Transitions.Count; ++t) {
627 string List(long[,] matrix) {
628 string a = string.Join(",",Enumerable.Range(0,Places.Count).Where(p => matrix[p,t] > 0).Select(p => Places[p]+":"+matrix[p,t]));
629 return a.Length == 0 ? "-" : a;
630 }
631 Say($"transition {Transitions[t]} priority={Priorities[t]} pre={List(pre)} post={List(post)} inhibit={List(inhibitor)}");
632 }
633 break;
634 }
635 }
