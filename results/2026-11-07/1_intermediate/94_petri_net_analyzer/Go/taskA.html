62func visit(adj [][]int, start int) []bool {
63 seen := make([]bool, len(adj))
64 q := []int{start}
65 seen[start] = true
66 for i := 0; i < len(q); i++ {
67 for _, v := range adj[q[i]] {
68 if !seen[v] {
69 seen[v] = true
70 q = append(q, v)
71 }
72 }
73 }
74 return seen
75}

102func flows(a [][]int64) [][]int64 {
103 n, m := len(a), len(a[0])
104 rows := make([][]int64, n)
105 for i, r := range a {
106 rows[i] = append(slices.Clone(r), make([]int64, n)...)
107 rows[i][m+i] = 1
108 }
109 for j := 0; j < m; j++ {
110 unique := map[string][]int64{}
111 for _, r := range rows {
112 if r[j] == 0 {
113 unique[key(r)] = r
114 }
115 }
116 for _, r := range rows {
117 if r[j] <= 0 {
118 continue
119 }
120 for _, s := range rows {
121 if s[j] >= 0 {
122 continue
123 }
124 v := make([]int64, m+n)
125 d := int64(0)
126 for k := range v {
127 v[k] = -s[j]*r[k] + r[j]*s[k]
128 d = gcd(d, v[k])
129 }
130 for k := range v {
131 v[k] /= d
132 }
133 unique[key(v)] = v
134 }
135 }
136 candidates := [][]int64{}
137 supports := []int{}
138 for _, r := range unique {
139 candidates = append(candidates, r)
140 mask := 0
141 for i := 0; i < n; i++ {
142 if r[m+i] > 0 {
143 mask |= 1 << i
144 }
145 }
146 supports = append(supports, mask)
147 }
148 rows = nil
149 for i, r := range candidates {
150 minimal := true
151 for _, s := range supports {
152 if s != supports[i] && s&supports[i] == s {
153 minimal = false
154 break
155 }
156 }
157 if minimal {
158 rows = append(rows, r)
159 }
160 }
161 }
162 result := [][]int64{}
163 for _, r := range rows {
164 result = append(result, slices.Clone(r[m:]))
165 }
166 sort.Slice(result, func(i, j int) bool { return slices.Compare(result[i], result[j]) > 0 })
167 return result
168}

195func (n *Net) define(w []string) error {
196 cmd := w[0]
197 arity := len(w) - 1
198 if cmd != "place" && cmd != "transition" && cmd != "arc" && cmd != "inhibit" {
199 return fmt.Errorf("unknown definition %s", cmd)
200 }
201 valid := arity == 3
202 if cmd == "place" {
203 valid = arity == 2 || arity == 3
204 }
205 if cmd == "transition" {
206 valid = arity == 1 || arity == 2
207 }
208 if !valid {
209 return fmt.Errorf("wrong number of arguments")
210 }
211 a := w[1]
212 if cmd == "place" || cmd == "transition" {
213 if slices.Contains(n.places, a) || slices.Contains(n.transitions, a) {
214 return fmt.Errorf("duplicate name %s", a)
215 }
216 if cmd == "place" {
217 tokens, _ := strconv.ParseInt(w[2], 10, 64)
218 cap := int64(0)
219 if arity == 3 {
220 cap, _ = strconv.ParseInt(w[3], 10, 64)
221 if tokens > cap {
222 return fmt.Errorf("tokens exceed capacity of %s", a)
223 }
224 }
225 n.places = append(n.places, a)
226 n.initial = append(n.initial, tokens)
227 n.capacity = append(n.capacity, cap)
228 } else {
229 priority := int64(0)
230 if arity == 2 {
231 priority, _ = strconv.ParseInt(w[2], 10, 64)
232 }
233 n.transitions = append(n.transitions, a)
234 n.priority = append(n.priority, priority)
235 }
236 } else if cmd == "arc" {
237 b := w[2]
238 for _, x := range []string{a, b} {
239 if !slices.Contains(n.places, x) && !slices.Contains(n.transitions, x) {
240 return fmt.Errorf("unknown node %s", x)
241 }
242 }
243 pa, pb := slices.Index(n.places, a), slices.Index(n.places, b)
244 if (pa >= 0) == (pb >= 0) {
245 return fmt.Errorf("arc must connect a place and a transition")
246 }
247 p := pa
248 t := slices.Index(n.transitions, b)
249 matrix := &n.pre
250 if pa < 0 {
251 p = pb
252 t = slices.Index(n.transitions, a)
253 matrix = &n.post
254 }
255 if matrix[p][t] != 0 {
256 return fmt.Errorf("duplicate arc %s %s", a, b)
257 }
258 matrix[p][t], _ = strconv.ParseInt(w[3], 10, 64)
259 n.arcs++
260 } else {
261 p, t := slices.Index(n.places, a), slices.Index(n.transitions, w[2])
262 if p < 0 {
263 return fmt.Errorf("unknown place %s", a)
264 }
265 if t < 0 {
266 return fmt.Errorf("unknown transition %s", w[2])
267 }
268 if n.inhibitor[p][t] != 0 {
269 return fmt.Errorf("duplicate inhibitor %s %s", a, w[2])
270 }
271 n.inhibitor[p][t], _ = strconv.ParseInt(w[3], 10, 64)
272 n.inhibitors++
273 }
274 return nil
275}
276func (n *Net) reason(m []int64, t int) string {
277 for p, name := range n.places {
278 if m[p] >= 0 && m[p] < n.pre[p][t] {
279 return "insufficient tokens in " + name
280 }
281 }
282 for p, name := range n.places {
283 if n.inhibitor[p][t] > 0 && (m[p] < 0 || m[p] >= n.inhibitor[p][t]) {
284 return "inhibited by " + name
285 }
286 }
287 for p, name := range n.places {
288 if n.capacity[p] > 0 && (m[p] < 0 || m[p]-n.pre[p][t]+n.post[p][t] > n.capacity[p]) {
289 return "capacity of " + name + " exceeded"
290 }
291 }
292 return ""
293}
294func (n *Net) enabled(m []int64) []int {

301 return r

303func (n *Net) fire(m []int64, t int) []int64 {
304 r := slices.Clone(m)
305 for p := range r {
306 if r[p] >= 0 {
307 r[p] += n.post[p][t] - n.pre[p][t]
308 }
309 }
310 return r
311}

323func (n *Net) graph(start []int64, limit int, target [][2]int64) Graph {
324 g := Graph{found: -1}
325 g.add(slices.Clone(start), -1, -1)
326 index := map[string]int{key(start): 0}
327 matches := func(m []int64) bool {
328 if target == nil {
329 return false
330 }
331 for p, v := range m {
332 if v < target[p][0] || target[p][1] >= 0 && v != target[p][1] {
333 return false
334 }
335 }
336 return true
337 }
338 if matches(start) {
339 g.found = 0
340 return g
341 }
342 for q := 0; q < len(g.states); q++ {
343 for _, t := range n.enabled(g.states[q]) {
344 m := n.fire(g.states[q], t)
345 k, known := index[key(m)]
346 if !known {
347 if len(g.states) == limit {
348 g.aborted = true
349 return g
350 }
351 k = len(g.states)
352 index[key(m)] = k
353 g.add(m, q, t)
354 if matches(m) {
355 g.found = k
356 return g
357 }
358 }
359 g.edges[q] = append(g.edges[q], Edge{t, k})
360 }
361 }
362 return g
363}
364func path(g Graph, k int) []int {
365 r := []int{}
366 for g.parent[k] >= 0 {
367 r = append(r, g.via[k])
368 k = g.parent[k]
369 }
370 slices.Reverse(r)
371 return r
372}
373func (n *Net) reach(limit int, display bool) {
374 g := n.graph(n.current, limit, nil)
375 if g.aborted {
376 say(fmt.Sprintf("reachability graph limit %d exceeded", limit))
377 return
378 }
379 N := len(g.states)
380 if display {
381 for i, m := range g.states {
382 a := []string{}
383 for _, e := range g.edges[i] {
384 a = append(a, fmt.Sprintf("%s->M%d", n.transitions[e.transition], e.target))
385 }
386 text := strings.Join(a, " ")
387 if len(a) == 0 {
388 text = "dead"
389 }
390 say(fmt.Sprintf("M%d %s: %s", i, marking(m), text))
391 }
392 return
393 }
394 adj := make([][]int, N)
395 for i, e := range g.edges {
396 for _, v := range e {
397 adj[i] = append(adj[i], v.target)
398 }
399 }
400 closure := make([][]bool, N)
401 for i := range closure {
402 closure[i] = visit(adj, i)
403 }
404 comp := make([]int, N)
405 for i := range comp {
406 comp[i] = -1
407 }
408 components := 0
409 for i := range comp {
410 if comp[i] >= 0 {
411 continue
412 }
413 for j := i; j < N; j++ {
414 if closure[i][j] && closure[j][i] {
415 comp[j] = components
416 }
417 }
418 components++
419 }
420 terminal, home := make([]bool, components), make([]bool, N)
421 for i := range terminal {
422 terminal[i] = true
423 }
424 for i := range home {
425 home[i] = true
426 }
427 maximum := make([]int64, len(n.places))
428 dead := []int{}
429 edges := 0
430 persistent := true
431 for i, m := range g.states {
432 if len(g.edges[i]) == 0 {
433 dead = append(dead, i)
434 }
435 edges += len(g.edges[i])
436 for p, v := range m {
437 maximum[p] = max(maximum[p], v)
438 }
439 for j := range home {
440 home[j] = home[j] && closure[i][j]
441 }
442 for _, e := range g.edges[i] {
443 if comp[i] != comp[e.target] {
444 terminal[comp[i]] = false
445 }
446 for _, u := range g.edges[i] {
447 if e.transition != u.transition && n.reason(g.states[e.target], u.transition) != "" {
448 persistent = false
449 }
450 }
451 }
452 }
453 say(fmt.Sprintf("reachable markings: %d\nedges: %d\ncomponents: %d (terminal %d)\ndead markings: %d", N, edges, components, count(terminal), len(dead)))
454 for _, i := range dead {
455 a := n.names(path(g, i))
456 text := strings.Join(a, " ")
457 if len(a) == 0 {
458 text = "empty"
459 }
460 say(fmt.Sprintf(" M%d %s path %s", i, marking(g.states[i]), text))
461 }
462 bound := slices.Max(maximum)
463 say(fmt.Sprintf("max tokens: %s\nbound: %d\nsafe: %s\nhome markings: %d\nreversible: %s\npersistent: %s", marking(maximum), bound, yes(bound <= 1), count(home), yes(home[0]), yes(persistent)))
464 levels := []string{}
465 for t, name := range n.transitions {
466 present := make([]bool, components)
467 used, cycle := false, false
468 for i, e := range g.edges {
469 for _, v := range e {
470 if v.transition == t {
471 used = true
472 present[comp[i]] = true
473 if comp[i] == comp[v.target] {
474 cycle = true
475 }
476 }
477 }
478 }
479 live := true
480 for c, v := range terminal {
481 if v && !present[c] {
482 live = false
483 }
484 }
485 level := "L0"
486 if live {
487 level = "L4"
488 } else if cycle {
489 level = "L3"
490 } else if used {
491 level = "L1"
492 }
493 levels = append(levels, name+"="+level)
494 }
495 say("liveness: " + strings.Join(levels, " "))
496}
497func rank(v int64) int64 {
498 if v < 0 {
499 return 1<<63 - 1
500 }
501 return v
502}
503func (n *Net) cover(limit int) {
504 g := Graph{}
505 g.add(slices.Clone(n.current), -1, -1)
506 old, dead := []bool{false}, []bool{false}
507 seen := map[string]bool{key(n.current): true}
508 for q := 0; q < len(g.states); q++ {
509 if old[q] {
510 continue
511 }
512 es := n.enabled(g.states[q])
513 dead[q] = len(es) == 0
514 anc := []int{}
515 for a := q; a >= 0; a = g.parent[a] {
516 anc = append(anc, a)
517 }
518 slices.Reverse(anc)
519 for _, t := range es {
520 m := n.fire(g.states[q], t)
521 for _, a := range anc {
522 ge, gt := true, false
523 for p, v := range m {
524 ge = ge && rank(v) >= rank(g.states[a][p])
525 gt = gt || rank(v) > rank(g.states[a][p])
526 }
527 if ge && gt {
528 for p, v := range m {
529 if n.capacity[p] == 0 && rank(v) > rank(g.states[a][p]) {
530 m[p] = -1
531 }
532 }
533 }
534 }
535 if len(g.states) == limit {
536 say(fmt.Sprintf("coverability tree limit %d exceeded", limit))
537 return
538 }
539 old = append(old, seen[key(m)])
540 seen[key(m)] = true
541 dead = append(dead, false)
542 g.add(m, q, t)
543 }
544 }
545 say(fmt.Sprintf("coverability tree nodes: %d", len(g.states)))
546 maximum := make([]int64, len(n.places))
547 for i, m := range g.states {
548 text := "root"
549 if i > 0 {
550 text = fmt.Sprintf("<- %d %s", g.parent[i], n.transitions[g.via[i]])
551 }
552 if old[i] {
553 text += " old"
554 }
555 if dead[i] {
556 text += " dead"
557 }
558 say(fmt.Sprintf(" %d %s %s", i, marking(m), text))
559 for p, v := range m {
560 if rank(v) > rank(maximum[p]) {
561 maximum[p] = v
562 }
563 }
564 }
565 a := []string{}
566 for p, v := range maximum {
567 if v < 0 {
568 a = append(a, n.places[p])
569 }
570 }
571 say("bounds: " + marking(maximum))
572 say("unbounded: " + listed(a))
573}

583func (n *Net) sets(trap bool) []int {
584 found := []int{}
585 for mask := 1; mask < 1<<len(n.places); mask++ {
586 valid := true
587 for t := range n.transitions {
588 input, output := false, false
589 for _, p := range n.bits(mask) {
590 input = input || n.post[p][t] > 0
591 output = output || n.pre[p][t] > 0
592 }
593 if trap && output && !input || !trap && input && !output {
594 valid = false
595 break
596 }
597 }
598 if valid {
599 found = append(found, mask)
600 }
601 }
602 r := []int{}
603 for _, m := range found {
604 minimal := true
605 for _, s := range found {
606 if s != m && s&m == s {
607 minimal = false
608 break
609 }
610 }
611 if minimal {
612 r = append(r, m)
613 }
614 }
615 sort.Slice(r, func(i, j int) bool {
616 a, b := r[i], r[j]
617 x, y := bits.OnesCount(uint(a)), bits.OnesCount(uint(b))
618 if x != y {
619 return x < y
620 }
621 return slices.Compare(n.bits(a), n.bits(b)) < 0
622 })
623 return r
624}

632func (n *Net) realize(counts []int64) ([]int, bool) {
633 used := make([]int64, len(counts))
634 stack := [][]int64{n.current}
635 choices, path := []int{0}, []int{}
636 seen := map[string]bool{}
637 total := int64(0)
638 for _, c := range counts {
639 total += c
640 }
641 for len(stack) > 0 {
642 d := len(path)
643 if int64(d) == total {
644 return path, true
645 }
646 chosen := -1
647 for t := choices[d]; t < len(counts); t++ {
648 choices[d] = t + 1
649 if used[t] >= counts[t] || n.reason(stack[d], t) != "" {
650 continue
651 }
652 used[t]++
653 k := key(used)
654 used[t]--
655 if seen[k] {
656 continue
657 }
658 seen[k] = true
659 chosen = t
660 break
661 }
662 if chosen >= 0 {
663 used[chosen]++
664 path = append(path, chosen)
665 stack = append(stack, n.fire(stack[d], chosen))
666 choices = append(choices, 0)
667 } else {
668 stack = stack[:len(stack)-1]
669 choices = choices[:len(choices)-1]
670 if len(path) > 0 {
671 used[path[len(path)-1]]--
672 path = path[:len(path)-1]
673 }
674 }
675 }
676 return nil, false
677}
678func (n *Net) invariants(transpose bool) {
679 a := [][]int64{}
680 ns := n.places
681 if transpose {
682 ns = n.transitions
683 for t := range n.transitions {
684 r := []int64{}
685 for p := range n.places {
686 r = append(r, n.post[p][t]-n.pre[p][t])
687 }
688 a = append(a, r)
689 }
690 } else {
691 for p := range n.places {
692 r := []int64{}
693 for t := range n.transitions {
694 r = append(r, n.post[p][t]-n.pre[p][t])
695 }
696 a = append(a, r)
697 }
698 }
699 f := flows(a)
700 tag := "P"
701 if transpose {
702 tag = "T"
703 }
704 say(fmt.Sprintf("%s-semiflows: %d", tag, len(f)))
705 bounds := make([]int64, len(ns))
706 covered := make([]bool, len(ns))
707 for i := range bounds {
708 bounds[i] = -1
709 }
710 for _, c := range f {
711 terms := []string{}
712 value := int64(0)
713 for i, v := range c {
714 if v > 0 {
715 covered[i] = true
716 prefix := ""
717 if v != 1 {
718 prefix = fmt.Sprintf("%d*", v)
719 }
720 terms = append(terms, prefix+ns[i])
721 if !transpose {
722 value += v * n.current[i]
723 }
724 }
725 }
726 text := " " + strings.Join(terms, " + ")
727 if transpose {
728 path, ok := n.realize(c)
729 if ok {
730 say(text + ": realizable " + strings.Join(n.names(path), " "))
731 } else {
732 say(text + ": not realizable")
733 }
734 } else {
735 say(fmt.Sprintf("%s = %d", text, value))
736 for p, v := range c {
737 if v > 0 {
738 b := value / v
739 if bounds[p] < 0 || b < bounds[p] {
740 bounds[p] = b
741 }
742 }
743 }
744 }
745 }
746 if !transpose {
747 s := []string{}
748 for _, b := range bounds {
749 if b < 0 {
750 s = append(s, "-")
751 } else {
752 s = append(s, strconv.FormatInt(b, 10))
753 }
754 }
755 say("bounds: (" + strings.Join(s, ",") + ")")
756 }
757 label := "conservative: "
758 if transpose {
759 label = "consistent: "
760 }
761 say(label + yes(all(covered)))
762}
763func (n *Net) classify() {
764 P, T := len(n.places), len(n.transitions)
765 inputs, outputs := make([][]int, T), make([][]int, P)
766 ordinary, pure, sm, mg, fc, efc := true, true, true, true, true, true
767 for p := range n.places {
768 for t := range n.transitions {
769 ordinary = ordinary && n.pre[p][t] <= 1 && n.post[p][t] <= 1
770 pure = pure && !(n.pre[p][t] > 0 && n.post[p][t] > 0)
771 if n.pre[p][t] > 0 {
772 inputs[t] = append(inputs[t], p)
773 outputs[p] = append(outputs[p], t)
774 }
775 }
776 }
777 for t := range n.transitions {
778 c := 0
779 for p := range n.places {
780 if n.post[p][t] > 0 {
781 c++
782 }
783 }
784 sm = sm && len(inputs[t]) == 1 && c == 1
785 }
786 for p, a := range outputs {
787 c := 0
788 for t := range n.transitions {
789 if n.post[p][t] > 0 {
790 c++
791 }
792 }
793 mg = mg && len(a) == 1 && c == 1
794 for _, t := range a {
795 if len(a) != 1 && len(inputs[t]) != 1 {
796 fc = false
797 }
798 for q, b := range outputs {
799 if n.pre[q][t] > 0 && !slices.Equal(a, b) {
800 efc = false
801 }
802 }
803 }
804 }
805 sm = sm && ordinary
806 mg = mg && ordinary
807 fc = fc && ordinary
808 efc = efc && ordinary
809 labels := []string{"ordinary", "pure", "state machine", "marked graph", "free choice", "extended free choice"}
810 values := []bool{ordinary, pure, sm, mg, fc, efc}
811 for i, label := range labels {
812 say(label + ": " + yes(values[i]))
813 }
814 verdict := "holds"
815 if !fc {
816 verdict = "not applicable"
817 } else if P > 12 {
818 verdict = "too many places"
819 } else {
820 traps := n.sets(true)
821 for _, s := range n.sets(false) {
822 holds := false
823 for _, q := range traps {
824 if s&q == q {
825 for _, p := range n.bits(q) {
826 if n.current[p] > 0 {
827 holds = true
828 }
829 }
830 }
831 }
832 if !holds {
833 verdict = "fails at " + n.setText(s)
834 break
835 }
836 }
837 }
838 say("commoner: " + verdict)
839}
840func (n *Net) workflow(limit int) {
841 P, T := len(n.places), len(n.transitions)
842 sources, sinks := []int{}, []int{}
843 for p := range n.places {
844 input, output := false, false
845 for t := range n.transitions {
846 input = input || n.post[p][t] > 0
847 output = output || n.pre[p][t] > 0
848 }
849 if !input {
850 sources = append(sources, p)
851 }
852 if !output {
853 sinks = append(sinks, p)
854 }
855 }
856 if len(sources) != 1 {
857 say(fmt.Sprintf("workflow: no (%d source places)", len(sources)))
858 return
859 }
860 if len(sinks) != 1 {
861 say(fmt.Sprintf("workflow: no (%d sink places)", len(sinks)))
862 return
863 }
864 i, o := sources[0], sinks[0]
865 adj, rev := make([][]int, P+T), make([][]int, P+T)
866 for p := range n.places {
867 for t := range n.transitions {
868 if n.pre[p][t] > 0 {
869 adj[p] = append(adj[p], P+t)
870 rev[P+t] = append(rev[P+t], p)
871 }
872 if n.post[p][t] > 0 {
873 adj[P+t] = append(adj[P+t], p)
874 rev[p] = append(rev[p], P+t)
875 }
876 }
877 }
878 ns := append(slices.Clone(n.places), n.transitions...)
879 f, b := visit(adj, i), visit(rev, o)
880 for v, name := range ns {
881 if !f[v] || !b[v] {
882 say(fmt.Sprintf("workflow: no (%s is not on a path from %s to %s)", name, n.places[i], n.places[o]))
883 return
884 }
885 }
886 say(fmt.Sprintf("workflow: source %s sink %s", n.places[i], n.places[o]))
887 start, goal := make([]int64, P), make([]int64, P)
888 start[i] = 1
889 goal[o] = 1
890 g := n.graph(start, limit, nil)
891 if g.aborted {
892 say(fmt.Sprintf("reachability graph limit %d exceeded", limit))
893 return
894 }
895 finish := -1
896 rev = make([][]int, len(g.states))
897 used := make([]bool, T)
898 proper := true
899 for u, m := range g.states {
900 if slices.Equal(m, goal) {
901 finish = u
902 }
903 if m[o] >= 1 && !slices.Equal(m, goal) {
904 proper = false
905 }
906 for _, e := range g.edges[u] {
907 rev[e.target] = append(rev[e.target], u)
908 used[e.transition] = true
909 }
910 }
911 option := finish >= 0 && all(visit(rev, finish))
912 live := all(used)
913 say("option to complete: " + yes(option))
914 say("proper completion: " + yes(proper))
915 say("no dead transitions: " + yes(live))
916 say("sound: " + yes(option && proper && live))
917}
918func (n *Net) execute(w []string) error {
919 arity := map[string]int{"MARKING": 0, "RESET": 0, "SET": 2, "ENABLED": 0, "FIRE": -1, "UNDO": 1, "SIMULATE": 3, "CONFLICTS": 0, "REACH": 1, "GRAPH": 1, "COVER": 1, "PATH": 2, "INCIDENCE": 0, "PINV": 0, "TINV": 0, "SIPHONS": 0, "TRAPS": 0, "CLASSIFY": 0, "WORKFLOW": 1, "SHOW": 0}
920 cmd := w[0]
921 expected, known := arity[cmd]
922 if !known {
923 return fmt.Errorf("unknown command %s", cmd)
924 }
925 if expected < 0 && len(w) < 2 || expected >= 0 && len(w) != expected+1 {
926 return fmt.Errorf("wrong number of arguments")
927 }
928 limit := 0
929 if slices.Contains([]string{"REACH", "GRAPH", "COVER", "PATH", "WORKFLOW"}, cmd) {
930 limit = int(number(w[1], 1, 2000))
931 if limit < 0 {
932 return fmt.Errorf("invalid limit %s", w[1])
933 }
934 }
935 switch cmd {
936 case "MARKING", "RESET":
937 if cmd == "RESET" {
938 n.current = slices.Clone(n.initial)
939 n.history = nil
940 }
941 say("marking " + marking(n.current))
942 case "SET":
943 p := slices.Index(n.places, w[1])
944 if p < 0 {
945 return fmt.Errorf("unknown place %s", w[1])
946 }
947 v := number(w[2], 0, 999)
948 if v < 0 {
949 return fmt.Errorf("invalid tokens %s", w[2])
950 }
951 if n.capacity[p] > 0 && v > n.capacity[p] {
952 return fmt.Errorf("tokens exceed capacity of %s", w[1])
953 }
954 n.current = slices.Clone(n.current)
955 n.current[p] = v
956 n.history = nil
957 say("marking " + marking(n.current))
958 case "ENABLED":
959 say("enabled: " + listed(n.names(n.enabled(n.current))))
960 case "FIRE":
961 ts := []int{}
962 for _, name := range w[1:] {
963 t := slices.Index(n.transitions, name)
964 if t < 0 {
965 return fmt.Errorf("unknown transition %s", name)
966 }
967 ts = append(ts, t)
968 }
969 for i, t := range ts {
970 r := n.reason(n.current, t)
971 if r != "" {
972 say(fmt.Sprintf("step %d: %s not enabled (%s) at %s", i+1, n.transitions[t], r, marking(n.current)))
973 return nil
974 }
975 n.commit(t)
976 }
977 say(fmt.Sprintf("fired %d -> %s", len(ts), marking(n.current)))
978 case "UNDO":
979 k := int(number(w[1], 1, 1000))
980 if k < 0 {
981 return fmt.Errorf("invalid count %s", w[1])
982 }
983 h := len(n.history)
984 if k > h {
985 return fmt.Errorf("cannot undo %d: history size %d", k, h)
986 }
987 n.current = n.history[h-k]
988 n.history = n.history[:h-k]
989 say("marking " + marking(n.current))
990 case "SIMULATE":
991 steps := int(number(w[1], 1, 1000))
992 x := number(w[2], 1, 2147483646)
993 if steps < 0 {
994 return fmt.Errorf("invalid steps %s", w[1])
995 }
996 if x < 0 {
997 return fmt.Errorf("invalid seed %s", w[2])
998 }
999 if w[3] != "PRIORITY" && w[3] != "RANDOM" {
1000 return fmt.Errorf("invalid mode %s", w[3])
1001 }
1002 counts := make([]int, len(n.transitions))
1003 done := 0
1004 for ; done < steps; done++ {
1005 es := n.enabled(n.current)
1006 if len(es) == 0 {
1007 say(fmt.Sprintf("dead (steps=%d)", done))
1008 break
1009 }
1010 t := es[0]
1011 if w[3] == "RANDOM" {
1012 x = x * 48271 % 2147483647
1013 t = es[x%int64(len(es))]
1014 } else {
1015 for _, u := range es {
1016 if n.priority[u] > n.priority[t] {
1017 t = u
1018 }
1019 }
1020 }
1021 n.commit(t)
1022 counts[t]++
1023 say(fmt.Sprintf("step %d: %s -> %s", done+1, n.transitions[t], marking(n.current)))
1024 }
1025 if done == steps {
1026 say(fmt.Sprintf("completed (steps=%d)", steps))
1027 }
1028 a := []string{}
1029 for t, name := range n.transitions {
1030 a = append(a, fmt.Sprintf("%s=%d", name, counts[t]))
1031 }
1032 say("counts: " + strings.Join(a, " "))
1033 case "CONFLICTS":
1034 s, e := []string{}, []string{}
1035 for u := range n.transitions {
1036 for v := u + 1; v < len(n.transitions); v++ {
1037 pair := n.transitions[u] + "/" + n.transitions[v]
1038 conflict := false
1039 for p := range n.places {
1040 if n.pre[p][u] > 0 && n.pre[p][v] > 0 {
1041 conflict = true
1042 }
1043 }
1044 if conflict {
1045 s = append(s, pair)
1046 }
1047 if n.reason(n.current, u) == "" && n.reason(n.current, v) == "" && (n.reason(n.fire(n.current, u), v) != "" || n.reason(n.fire(n.current, v), u) != "") {
1048 e = append(e, pair)
1049 }
1050 }
1051 }
1052 say("structural: " + listed(s))
1053 say("effective: " + listed(e))
1054 case "REACH", "GRAPH":
1055 n.reach(limit, cmd == "GRAPH")
1056 case "COVER":
1057 n.cover(limit)
1058 case "PATH":
1059 entries := strings.Split(w[2], ",")
1060 if len(entries) != len(n.places) {
1061 return fmt.Errorf("target needs %d entries", len(n.places))
1062 }
1063 target := make([][2]int64, len(entries))
1064 for p, s := range entries {
1065 if s == "*" {
1066 target[p] = [2]int64{0, -1}
1067 continue
1068 }
1069 plus := strings.HasSuffix(s, "+")
1070 raw := s
1071 if plus {
1072 raw = s[:len(s)-1]
1073 }
1074 v := number(raw, 0, 9999999999)
1075 if v < 0 {
1076 return fmt.Errorf("invalid target entry %s", s)
1077 }
1078 target[p] = [2]int64{v, v}
1079 if plus {
1080 target[p][1] = -1
1081 }
1082 }
1083 g := n.graph(n.current, limit, target)
1084 if g.found >= 0 {
1085 a := n.names(path(g, g.found))
1086 suffix := ""
1087 if len(a) > 0 {
1088 suffix = ": " + strings.Join(a, " ")
1089 }
1090 say(fmt.Sprintf("reached %s length=%d%s", marking(g.states[g.found]), len(a), suffix))
1091 } else if g.aborted {
1092 say(fmt.Sprintf("search limit %d exceeded", limit))
1093 } else {
1094 say(fmt.Sprintf("unreachable (markings explored: %d)", len(g.states)))
1095 }
1096 case "INCIDENCE":
1097 rows := [][]string{append([]string{""}, n.transitions...)}
1098 for p, name := range n.places {
1099 r := []string{name}
1100 for t := range n.transitions {
1101 r = append(r, strconv.FormatInt(n.post[p][t]-n.pre[p][t], 10))
1102 }
1103 rows = append(rows, r)
1104 }
1105 width := make([]int, len(n.transitions)+1)
1106 for _, r := range rows {
1107 for i, s := range r {
1108 width[i] = max(width[i], len(s))
1109 }
1110 }
1111 for _, r := range rows {
1112 a := []string{}
1113 for i, s := range r {
1114 pad := strings.Repeat(" ", width[i]-len(s))
1115 if i == 0 {
1116 a = append(a, s+pad)
1117 } else {
1118 a = append(a, pad+s)
1119 }
1120 }
1121 say(strings.TrimRight(strings.Join(a, " "), " "))
1122 }
1123 case "PINV", "TINV":
1124 n.invariants(cmd == "TINV")
1125 case "SIPHONS", "TRAPS":
1126 if len(n.places) > 12 {
1127 return fmt.Errorf("more than 12 places")
1128 }
1129 ss := n.sets(cmd == "TRAPS")
1130 label := "siphons"
1131 if cmd == "TRAPS" {
1132 label = "traps"
1133 }
1134 say(fmt.Sprintf("minimal %s: %d", label, len(ss)))
1135 for _, s := range ss {
1136 tokens := int64(0)
1137 for _, p := range n.bits(s) {
1138 tokens += n.current[p]
1139 }
1140 say(fmt.Sprintf(" %s tokens=%d", n.setText(s), tokens))
1141 }
1142 case "CLASSIFY":
1143 n.classify()
1144 case "WORKFLOW":
1145 n.workflow(limit)
1146 case "SHOW":
1147 for p, name := range n.places {
1148 cap := "none"
1149 if n.capacity[p] > 0 {
1150 cap = strconv.FormatInt(n.capacity[p], 10)
1151 }
1152 say(fmt.Sprintf("place %s tokens=%d capacity=%s", name, n.initial[p], cap))
1153 }
1154 for t, name := range n.transitions {
1155 list := func(matrix [16][16]int64) string {
1156 a := []string{}
1157 for p, pn := range n.places {
1158 if matrix[p][t] > 0 {
1159 a = append(a, fmt.Sprintf("%s:%d", pn, matrix[p][t]))
1160 }
1161 }
1162 if len(a) == 0 {
1163 return "-"
1164 }
1165 return strings.Join(a, ",")
1166 }
1167 say(fmt.Sprintf("transition %s priority=%d pre=%s post=%s inhibit=%s", name, n.priority[t], list(n.pre), list(n.post), list(n.inhibitor)))
1168 }
1169 }
1170 return nil
1171}
