44flows <- function(a) {
45 n <- nrow(a)
46 m <- ncol(a)
47 rows <- lapply(seq_len(n), function(i) c(a[i, ], as.numeric(seq_len(n) == i)))
48 for (j in seq_len(m)) {
49 collected <- Filter(function(r) r[j] == 0, rows)
50 for (r in rows) {
51 if (r[j] > 0) {
52 for (s in rows) {
53 if (s[j] < 0) {
54 v <- -s[j] * r + r[j] * s
55 d <- Reduce(gcd, v, init = 0)
56 collected[[length(collected) + 1L]] <- v / d
57 }
58 }
59 }
60 }
61 keys <- vapply(collected, paste, character(1), collapse = ',')
62 unique_rows <- collected[!duplicated(keys)]
63 support <- vapply(
64 unique_rows,
65 function(r) sum(2^(seq_len(n) - 1L)[r[m + seq_len(n)] > 0]),
66 numeric(1)
67 )
68 rows <- unique_rows[vapply(
69 seq_along(unique_rows),
70 function(i) {
71 !any(support != support[i] & bitwAnd(support, support[i]) == support)
72 },
73 logical(1)
74 )]
75 }
76 result <- lapply(rows, function(r) r[m + seq_len(n)])
77 if (length(result) > 1L) {
78 order <- do.call(
79 order,
80 c(
81 lapply(seq_len(n), function(i) -vapply(result, `[`, numeric(1), i)),
82 list(method = 'radix')
83 )
84 )
85 result <- result[order]
86 }
87 result

100define <- function(n, w) {
101 cmd <- w[1]
102 count <- length(w) - 1L
103 if (!cmd %in% c('place', 'transition', 'arc', 'inhibit')) {
104 bad(paste('unknown definition', cmd))
105 }
106 if (
107 !(if (cmd == 'place') {
108 count %in% 2:3
109 } else if (cmd == 'transition') {
110 count %in% 1:2
111 } else {
112 count == 3
113 })
114 ) {
115 bad('wrong number of arguments')
116 }
117 a <- w[2]
118 if (cmd %in% c('place', 'transition')) {
119 if (a %in% c(n$p, n$t)) {
120 bad(paste('duplicate name', a))
121 }
122 if (cmd == 'place') {
123 if (count == 3 && as.numeric(w[3]) > as.numeric(w[4])) {
124 bad(paste('tokens exceed capacity of', a))
125 }
126 n$p <- c(n$p, a)
127 n$initial <- c(n$initial, as.numeric(w[3]))
128 n$cap <- c(n$cap, if (count == 3) as.numeric(w[4]) else 0)
129 } else {
130 n$t <- c(n$t, a)
131 n$priority <- c(n$priority, if (count == 2) as.numeric(w[3]) else 0)
132 }
133 } else if (cmd == 'arc') {
134 b <- w[3]
135 for (x in c(a, b)) {
136 if (!x %in% c(n$p, n$t)) bad(paste('unknown node', x))
137 }
138 pa <- match(a, n$p)
139 pb <- match(b, n$p)
140 if (is.na(pa) == is.na(pb)) {
141 bad('arc must connect a place and a transition')
142 }
143 p <- if (!is.na(pa)) pa else pb
144 t <- match(if (!is.na(pa)) b else a, n$t)
145 field <- if (!is.na(pa)) 'pre' else 'post'
146 if (n[[field]][p, t] > 0) {
147 bad(paste('duplicate arc', a, b))
148 }
149 n[[field]][p, t] <- as.numeric(w[4])
150 n$arcs <- n$arcs + 1L
151 } else {
152 p <- match(a, n$p)
153 t <- match(w[3], n$t)
154 if (is.na(p)) {
155 bad(paste('unknown place', a))
156 }
157 if (is.na(t)) {
158 bad(paste('unknown transition', w[3]))
159 }
160 if (n$inh[p, t] > 0) {
161 bad(paste('duplicate inhibitor', a, w[3]))
162 }
163 n$inh[p, t] <- as.numeric(w[4])
164 n$inhibitors <- n$inhibitors + 1L
165 }

167reason <- function(n, m, t) {
168 for (p in seq_along(n$p)) {
169 if (m[p] >= 0 && m[p] < n$pre[p, t]) {
170 return(paste('insufficient tokens in', n$p[p]))
171 }
172 }
173 for (p in seq_along(n$p)) {
174 if (n$inh[p, t] > 0 && (m[p] < 0 || m[p] >= n$inh[p, t])) {
175 return(paste('inhibited by', n$p[p]))
176 }
177 }
178 for (p in seq_along(n$p)) {
179 if (
180 n$cap[p] > 0 && (m[p] < 0 || m[p] - n$pre[p, t] + n$post[p, t] > n$cap[p])
181 ) {
182 return(paste('capacity of', n$p[p], 'exceeded'))
183 }
184 }
185 ''

198graph <- function(n, start, limit, target = NULL) {
199 states <- list(start)
200 parent <- via <- 0L
201 edges <- list(list())
202 index <- new.env(hash = TRUE, parent = emptyenv())
203 index[[key(start)]] <- 1L
204 matches <- function(m) {
205 !is.null(target) &&
206 all(m >= target[, 1] & (target[, 2] < 0 | m == target[, 2]))

208 result <- function(found = 0L, aborted = FALSE) {
209 list(
210 states = states,
211 parent = parent,
212 via = via,
213 edges = edges,
214 found = found,
215 aborted = aborted
216 )
217 }
218 if (matches(start)) {
219 return(result(1L))
220 }
221 q <- 1L
222 while (q <= length(states)) {
223 for (t in enabled(n, states[[q]])) {
224 m <- fire(n, states[[q]], t)
225 k <- index[[key(m)]]
226 if (is.null(k)) {
227 if (length(states) == limit) {
228 return(result(aborted = TRUE))
229 }

231 index[[key(m)]] <- k
232 states[[k]] <- m
233 parent[k] <- q
234 via[k] <- t
235 edges[[k]] <- list()
236 if (matches(m)) return(result(k))
237 }
238 edges[[q]][[length(edges[[q]]) + 1L]] <- c(t, k)
239 }
240 q <- q + 1L
241 }

252reach <- function(n, limit, display) {
253 g <- graph(n, n$current, limit)
254 if (g$aborted) {
255 say('reachability graph limit ', limit, ' exceeded')
256 return()
257 }
258 N <- length(g$states)
259 if (display) {
260 for (i in seq_len(N)) {
261 a <- vapply(
262 g$edges[[i]],
263 function(e) paste0(n$t[e[1]], '->M', e[2] - 1L),
264 character(1)
265 )
266 say(
267 'M',
268 i - 1L,
269 ' ',
270 mark(g$states[[i]]),
271 ': ',
272 if (length(a)) paste(a, collapse = ' ') else 'dead'
273 )
274 }
275 return()
276 }
277 adj <- lapply(g$edges, function(e) vapply(e, `[`, numeric(1), 2))
278 closure <- lapply(seq_len(N), function(i) visit(adj, i))
279 comp <- integer(N)
280 components <- 0L
281 for (i in seq_len(N)) {
282 if (comp[i] == 0) {
283 components <- components + 1L
284 for (j in i:N) {
285 if (closure[[i]][j] && closure[[j]][i]) comp[j] <- components
286 }
287 }
288 }
289 terminal <- rep(TRUE, components)
290 home <- rep(TRUE, N)
291 maximum <- numeric(length(n$p))
292 dead <- integer()
293 edges <- 0L
294 persistent <- TRUE
295 for (i in seq_len(N)) {
296 if (!length(g$edges[[i]])) {
297 dead <- c(dead, i)
298 }
299 edges <- edges + length(g$edges[[i]])
300 maximum <- pmax(maximum, g$states[[i]])
301 home <- home & closure[[i]]
302 for (e in g$edges[[i]]) {
303 if (comp[i] != comp[e[2]]) {
304 terminal[comp[i]] <- FALSE
305 }
306 for (u in g$edges[[i]]) {
307 if (e[1] != u[1] && reason(n, g$states[[e[2]]], u[1]) != '') {
308 persistent <- FALSE
309 }
310 }
311 }
312 }
313 say('reachable markings: ', N)
314 say('edges: ', edges)
315 say('components: ', components, ' (terminal ', sum(terminal), ')')
316 say('dead markings: ', length(dead))
317 for (i in dead) {
318 a <- n$t[path(g, i)]
319 say(
320 ' M',
321 i - 1L,
322 ' ',
323 mark(g$states[[i]]),
324 ' path ',
325 if (length(a)) paste(a, collapse = ' ') else 'empty'
326 )
327 }
328 bound <- max(maximum)
329 say('max tokens: ', mark(maximum))
330 say('bound: ', num(bound))
331 say('safe: ', yes(bound <= 1))
332 say('home markings: ', sum(home))
333 say('reversible: ', yes(home[1]))
334 say('persistent: ', yes(persistent))
335 levels <- character()
336 for (t in seq_along(n$t)) {
337 present <- rep(FALSE, components)
338 used <- cycle <- FALSE
339 for (i in seq_len(N)) {
340 for (e in g$edges[[i]]) {
341 if (e[1] == t) {
342 used <- TRUE
343 present[comp[i]] <- TRUE
344 if (comp[i] == comp[e[2]]) cycle <- TRUE
345 }
346 }
347 }
348 level <- if (all(!terminal | present)) {
349 'L4'
350 } else if (cycle) {
351 'L3'
352 } else if (used) {
353 'L1'
354 } else {
355 'L0'
356 }
357 levels <- c(levels, paste0(n$t[t], '=', level))
358 }
359 say('liveness: ', paste(levels, collapse = ' '))

361cover <- function(n, limit) {
362 states <- list(n$current)
363 parent <- via <- 0L
364 old <- dead <- FALSE
365 rank <- function(v) ifelse(v < 0, Inf, v)
366 q <- 1L
367 while (q <= length(states)) {
368 if (!old[q]) {
369 es <- enabled(n, states[[q]])
370 dead[q] <- !length(es)
371 ancestors <- integer()
372 a <- q
373 while (a > 0) {
374 ancestors <- c(a, ancestors)
375 a <- parent[a]
376 }
377 for (t in es) {
378 m <- fire(n, states[[q]], t)
379 for (a in ancestors) {
380 if (
381 all(rank(m) >= rank(states[[a]])) &&
382 any(rank(m) > rank(states[[a]]))
383 ) {
384 m[n$cap == 0 & rank(m) > rank(states[[a]])] <- -1
385 }
386 }
387 if (length(states) == limit) {
388 say('coverability tree limit ', limit, ' exceeded')
389 return()
390 }
391 old <- c(old, any(vapply(states, identical, logical(1), m)))
392 states[[length(states) + 1L]] <- m
393 parent <- c(parent, q)
394 via <- c(via, t)
395 dead <- c(dead, FALSE)
396 }
397 }
398 q <- q + 1L
399 }
400 say('coverability tree nodes: ', length(states))
401 maximum <- numeric(length(n$p))
402 for (i in seq_along(states)) {
403 say(
404 ' ',
405 i - 1L,
406 ' ',
407 mark(states[[i]]),
408 ' ',
409 if (i == 1) 'root' else paste('<-', parent[i] - 1L, n$t[via[i]]),
410 if (old[i]) ' old' else '',
411 if (dead[i]) ' dead' else ''
412 )
413 mask <- rank(states[[i]]) > rank(maximum)
414 maximum[mask] <- states[[i]][mask]
415 }
416 say('bounds: ', mark(maximum))
417 say('unbounded: ', listed(n$p[maximum < 0]))

422sets <- function(n, trap) {
423 found <- integer()
424 for (mask in seq_len(2^length(n$p) - 1L)) {
425 ps <- bits(n, mask)
426 valid <- TRUE
427 for (t in seq_along(n$t)) {
428 input <- any(n$post[ps, t] > 0)
429 output <- any(n$pre[ps, t] > 0)
430 if (if (trap) output && !input else input && !output) {
431 valid <- FALSE
432 break
433 }
434 }
435 if (valid) found <- c(found, mask)
436 }
437 result <- found[vapply(
438 found,
439 function(m) !any(found != m & bitwAnd(found, m) == found),
440 logical(1)
441 )]
442 if (length(result) > 1L) {
443 for (i in 2:length(result)) {
444 j <- i
445 while (j > 1L) {
446 a <- bits(n, result[j])
447 b <- bits(n, result[j - 1L])
448 smaller <- if (length(a) != length(b)) {
449 length(a) < length(b)
450 } else {
451 lexless(a, b)
452 }
453 if (!smaller) {
454 break
455 }
456 result[c(j - 1L, j)] <- result[c(j, j - 1L)]
457 j <- j - 1L
458 }
459 }
460 }
461 result

466realize <- function(n, counts) {
467 used <- numeric(length(counts))
468 stack <- list(n$current)
469 choices <- 1L
470 result <- integer()
471 seen <- new.env(hash = TRUE, parent = emptyenv())
472 total <- sum(counts)
473 while (length(stack)) {
474 d <- length(result) + 1L
475 if (length(result) == total) {
476 return(result)
477 }
478 chosen <- 0L
479 if (choices[d] <= length(counts)) {
480 for (t in choices[d]:length(counts)) {
481 choices[d] <- t + 1L
482 if (used[t] >= counts[t] || reason(n, stack[[d]], t) != '') {
483 next
484 }
485 used[t] <- used[t] + 1
486 k <- key(used)
487 used[t] <- used[t] - 1
488 if (!is.null(seen[[k]])) {
489 next
490 }
491 seen[[k]] <- TRUE
492 chosen <- t
493 break
494 }
495 }
496 if (chosen > 0) {
497 used[chosen] <- used[chosen] + 1
498 result <- c(result, chosen)
499 stack[[length(stack) + 1L]] <- fire(n, stack[[d]], chosen)
500 choices <- c(choices, 1L)
501 } else {
502 stack <- head(stack, -1L)
503 choices <- head(choices, -1L)
504 if (length(result)) {
505 t <- tail(result, 1L)
506 used[t] <- used[t] - 1
507 result <- head(result, -1L)
508 }
509 }
510 }
511 NULL

563classify <- function(n) {
564 P <- length(n$p)
565 T <- length(n$t)
566 inputs <- lapply(seq_len(T), function(t) which(n$pre[seq_len(P), t] > 0))
567 outputs <- lapply(seq_len(P), function(p) which(n$pre[p, seq_len(T)] > 0))
568 ordinary <- all(n$pre <= 1 & n$post <= 1)
569 pure <- !any(n$pre > 0 & n$post > 0)
570 sm <- ordinary &&
571 all(vapply(
572 seq_len(T),
573 function(t) {
574 length(inputs[[t]]) == 1L && sum(n$post[seq_len(P), t] > 0) == 1
575 },
576 logical(1)
577 ))
578 mg <- ordinary &&
579 all(vapply(
580 seq_len(P),
581 function(p) {
582 length(outputs[[p]]) == 1L && sum(n$post[p, seq_len(T)] > 0) == 1
583 },
584 logical(1)
585 ))
586 fc <- ordinary
587 efc <- ordinary
588 for (p in seq_len(P)) {
589 for (t in outputs[[p]]) {
590 if (length(outputs[[p]]) != 1L && length(inputs[[t]]) != 1L) fc <- FALSE
591 }
592 for (q in seq_len(P)) {
593 if (
594 length(intersect(outputs[[p]], outputs[[q]])) &&
595 !identical(outputs[[p]], outputs[[q]])
596 ) {
597 efc <- FALSE
598 }
599 }
600 }
601 labels <- c(
602 'ordinary',
603 'pure',
604 'state machine',
605 'marked graph',
606 'free choice',
607 'extended free choice'
608 )
609 values <- c(ordinary, pure, sm, mg, fc, efc)
610 for (i in seq_along(labels)) {
611 say(labels[i], ': ', yes(values[i]))
612 }
613 verdict <- if (!fc) {
614 'not applicable'
615 } else if (P > 12) {
616 'too many places'
617 } else {
618 'holds'
619 }
620 if (fc && P <= 12) {
621 traps <- sets(n, TRUE)
622 for (s in sets(n, FALSE)) {
623 if (
624 !any(vapply(
625 traps,
626 function(q) bitwAnd(s, q) == q && any(n$current[bits(n, q)] > 0),
627 logical(1)
628 ))
629 ) {
630 verdict <- paste('fails at', set_text(n, s))
631 break
632 }
633 }
634 }
635 say('commoner: ', verdict)
636}
637workflow <- function(n, limit) {
638 P <- length(n$p)
639 T <- length(n$t)
640 sources <- which(rowSums(n$post[seq_len(P), seq_len(T), drop = FALSE]) == 0)
641 sinks <- which(rowSums(n$pre[seq_len(P), seq_len(T), drop = FALSE]) == 0)
642 if (length(sources) != 1) {
643 say('workflow: no (', length(sources), ' source places)')
644 return()
645 }
646 if (length(sinks) != 1) {
647 say('workflow: no (', length(sinks), ' sink places)')
648 return()
649 }
650 i <- sources[1]
651 o <- sinks[1]
652 ns <- c(n$p, n$t)
653 adj <- rev <- rep(list(integer()), P + T)
654 for (p in seq_len(P)) {
655 for (t in seq_len(T)) {
656 if (n$pre[p, t] > 0) {
657 adj[[p]] <- c(adj[[p]], P + t)
658 rev[[P + t]] <- c(rev[[P + t]], p)
659 }
660 if (n$post[p, t] > 0) {
661 adj[[P + t]] <- c(adj[[P + t]], p)
662 rev[[p]] <- c(rev[[p]], P + t)
663 }
664 }
665 }
666 f <- visit(adj, i)
667 b <- visit(rev, o)
668 for (v in seq_along(ns)) {
669 if (!f[v] || !b[v]) {
670 say(
671 'workflow: no (',
672 ns[v],
673 ' is not on a path from ',
674 n$p[i],
675 ' to ',
676 n$p[o],
677 ')'
678 )
679 return()
680 }
681 }
682 say('workflow: source ', n$p[i], ' sink ', n$p[o])
683 start <- goal <- numeric(P)
684 start[i] <- goal[o] <- 1
685 g <- graph(n, start, limit)
686 if (g$aborted) {
687 say('reachability graph limit ', limit, ' exceeded')
688 return()
689 }
690 finish <- which(vapply(g$states, identical, logical(1), goal))
691 rev <- rep(list(integer()), length(g$states))
692 used <- rep(FALSE, T)
693 for (u in seq_along(g$states)) {
694 for (e in g$edges[[u]]) {
695 rev[[e[2]]] <- c(rev[[e[2]]], u)
696 used[e[1]] <- TRUE
697 }
698 }
699 option <- length(finish) > 0 && all(visit(rev, finish[1]))
700 proper <- all(vapply(
701 g$states,
702 function(m) m[o] < 1 || identical(m, goal),
703 logical(1)
704 ))
705 live <- all(used)
706 say('option to complete: ', yes(option))
707 say('proper completion: ', yes(proper))
708 say('no dead transitions: ', yes(live))
709 say('sound: ', yes(option && proper && live))
710}
