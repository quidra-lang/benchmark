1def yn(value) = value ? "yes" : "no"
2def mark(values) = "(" + values.map { |v| v.negative? ? "w" : v.to_s }.join(",") + ")"
3def listing(values) = values.empty? ? "none" : values.join(" ")
4def number(word, low, high)
5 return nil unless /\A(?:0|[1-9][0-9]{0,9})\z/.match?(word)
6 value = word.to_i
7 (low..high).cover?(value) ? value : nil
8end

10def reachable(adj, start)
11 seen = Array.new(adj.size, false)
12 queue = [start]
13 seen[start] = true
14 queue.each { |u|
15 adj[u].each { |v|
16 unless seen[v]
17 seen[v] = true
18 queue << v
19 end
20 }
21 }
22 seen
23end

25def semiflows(matrix)
26 n = matrix.size
27 m = matrix[0].size
28 rows = matrix.each_with_index.map { |r, i| r + Array.new(n) { |j| i == j ? 1 : 0 } }
29 m.times do |j|
30 next_rows = rows.select { |r| r[j].zero? }
31 rows.select { |r| r[j].positive? }.each do |r|
32 rows.select { |s| s[j].negative? }.each do |s|
33 v = r.zip(s).map { |a, b| -s[j] * a + r[j] * b }
34 d = v.reduce(0) { |g, x| g.gcd(x) }
35 next_rows << v.map { |x| x / d }
36 end
37 end
38
39 unique = next_rows.uniq
40 supports = unique.map { |r| r.drop(m).each_index.reduce(0) { |s, i| r[m + i].positive? ? s | (1 << i) : s } }
41 rows = unique.each_with_index.filter_map { |r, i|
42 r unless supports.any? { |s| s != supports[i] && s & supports[i] == s }
43 }
44 end
45
46 rows.map { |r| r.drop(m) }.sort.reverse
47end

49class Net
50 attr_reader :places, :transitions, :initial, :arcs, :inhibitors
51 def initialize(name)
52 @name = name
53 @places = []
54 @transitions = []
55 @initial = []
56 @capacity = []
57 @priority = []
58 @pre, @post, @inhibit = Array.new(3) { Array.new(16) { Array.new(16, 0) } }
59 @arcs = @inhibitors = 0
60 @current = []
61 @history = []
62 end

64 def define(words)
65 cmd, a, b, c = words
66 n = words.size - 1
67 raise "unknown definition #{cmd}" unless %w[place transition arc inhibit].include?(cmd)
68 valid = case cmd
69 when "place"
70 [2, 3].include?(n)
71 when "transition"
72 [1, 2].include?(n)
73 else
74 n == 3
75 end
76
77 raise "wrong number of arguments" unless valid
78 if %w[place transition].include?(cmd)
79 raise "duplicate name #{a}" if (@places + @transitions).include?(a)
80 if cmd == "place"
81 raise "tokens exceed capacity of #{a}" if n == 3 && b.to_i > c.to_i
82 @places << a
83 @initial << b.to_i
84 @capacity << (n == 3 ? c.to_i : 0)
85 else
86 @transitions << a
87 @priority << (n == 2 ? b.to_i : 0)
88 end
89 elsif cmd == "arc"
90 [a, b].each { |x| raise "unknown node #{x}" unless (@places + @transitions).include?(x) }
91 pa = @places.index(a)
92 pb = @places.index(b)
93 raise "arc must connect a place and a transition" if pa.nil? == pb.nil?
94 p = pa || pb
95 t = @transitions.index(pa ? b : a)
96 matrix = pa ? @pre : @post
97 raise "duplicate arc #{a} #{b}" unless matrix[p][t].zero?
98 matrix[p][t] = c.to_i
99 @arcs += 1
100 else
101 p = @places.index(a)
102 t = @transitions.index(b)
103 raise "unknown place #{a}" unless p
104 raise "unknown transition #{b}" unless t
105 raise "duplicate inhibitor #{a} #{b}" unless @inhibit[p][t].zero?
106 @inhibit[p][t] = c.to_i
107 @inhibitors += 1
108 end
109 end

112 @current = @initial.dup
113 puts(
114 "net #{@name}: places=#{@places.size} transitions=#{@transitions.size} arcs=#{@arcs} inhibitors=#{@inhibitors}"
115 )

118 def reason(m, t)
119 @places.each_index { |p| return "insufficient tokens in #{@places[p]}" if m[p] >= 0 && m[p] < @pre[p][t] }
120 @places.each_index { |p|
121 return "inhibited by #{@places[p]}" if @inhibit[p][t] > 0 && (m[p] < 0 || m[p] >= @inhibit[p][t])
122 }
123 @places.each_index { |p|
124 if @capacity[p] > 0 && (m[p] < 0 || m[p] - @pre[p][t] + @post[p][t] > @capacity[p])
125 return "capacity of #{@places[p]} exceeded"
126 end
127 }
128 nil
129 end

131 def enabled(m) = @transitions.each_index.select { |t| !reason(m, t) }
132 def fire(m, t) = m.each_index.map { |p| m[p] < 0 ? -1 : m[p] - @pre[p][t] + @post[p][t] }
133 def names(indices) = indices.map { |i| @transitions[i] }
134 def commit(t)
135 @history << @current
136 @current = fire(@current, t)
137 end

139 def graph(start, limit, target = nil)
140 g = {states: [start.dup], parent: [-1], via: [-1], edges: [[]], found: nil, aborted: false}
141 matches = -> (m) {
142 target && m.each_index.all? { |p| m[p] >= target[p][0] && (target[p][1] < 0 || m[p] == target[p][1]) }
143 }
144 if matches.call(start)
145 g[:found] = 0
146 return g
147 end
148
149 index = {start => 0}
150 q = 0
151 while q < g[:states].size
152 enabled(g[:states][q]).each do |t|
153 m = fire(g[:states][q], t)
154 k = index[m]
155 unless k
156 if g[:states].size == limit
157 g[:aborted] = true
158 return g
159 end
160
161 k = g[:states].size
162 index[m] = k
163 g[:states] << m
164 g[:parent] << q
165 g[:via] << t
166 g[:edges] << []
167 if matches.call(m)
168 g[:found] = k
169 return g
170 end
171 end
172
173 g[:edges][q] << [t, k]
174 end
175
176 q += 1
177 end
178
179 g
180 end

182 def path(g, k)
183 result = []
184 while g[:parent][k] >= 0
185 result << g[:via][k]
186 k = g[:parent][k]
187 end
188
189 result.reverse
190 end

192 def reach(limit, display)
193 g = graph(@current, limit)
194 states = g[:states]
195 edges = g[:edges]
196 if g[:aborted]
197 puts("reachability graph limit #{limit} exceeded")
198 return
199 end
200
201 if display
202 states.each_with_index { |m, i|
203 puts(
204 "M#{i} #{mark(m)}: " +
205 (edges[i].empty? ? "dead" : edges[i].map { |t, k| "#{@transitions[t]}->M#{k}" }.join(" "))
206 )
207 }
208 return
209 end
210
211 adj = edges.map { |e| e.map(&:last) }
212 closure = states.each_index.map { |i| reachable(adj, i) }
213 comp = Array.new(states.size, -1)
214 count = 0
215 states.each_index do |i|
216 next if comp[i] >= 0
217 states.each_index { |j| comp[j] = count if closure[i][j] && closure[j][i] }
218 count += 1
219 end
220
221 terminal = Array.new(count, true)
222 edges.each_with_index { |e, i| e.each { |_, j| terminal[comp[i]] = false if comp[i] != comp[j] } }
223 dead = states.each_index.select { |i| edges[i].empty? }
224 maximum = @places.each_index.map { |p| states.map { |m| m[p] }.max }
225 bound = maximum.max
226 home = states.each_index.map { |j| closure.all? { |r| r[j] } }
227 persistent = states.each_index.all? { |i|
228 edges[i].all? { |t, k| edges[i].all? { |u, _| t == u || !reason(states[k], u) } }
229 }
230 levels = @transitions.each_index.map do |t|
231 present = Array.new(count, false)
232 used = cycle = false
233 edges.each_with_index { |e, i|
234 e.each { |u, k|
235 if t == u
236 used = true
237 present[comp[i]] = true
238 cycle = true if comp[i] == comp[k]
239 end
240 }
241 }
242 level = terminal.each_index.all? { |c| !terminal[c] || present[c] } ? "L4" : cycle ? "L3" : used ? "L1" : "L0"
243 "#{@transitions[t]}=#{level}"
244 end
245
246 puts(
247 "reachable markings: #{states.size}",
248 "edges: #{edges.sum(&:size)}",
249 "components: #{count} (terminal #{terminal.count(true)})",
250 "dead markings: #{dead.size}"
251 )
252 dead.each { |i|
253 p = names(path(g, i))
254 puts(" M#{i} #{mark(states[i])} path #{p.empty? ? "empty" : p.join(" ")}")
255 }
256 puts(
257 "max tokens: #{mark(maximum)}",
258 "bound: #{bound}",
259 "safe: #{yn(bound <= 1)}",
260 "home markings: #{home.count(true)}",
261 "reversible: #{yn(home[0])}",
262 "persistent: #{yn(persistent)}",
263 "liveness: #{levels.join(" ")}"
264 )
265 end

267 def cover(limit)
268 states = [@current.dup]
269 parents = [-1]
270 via = [-1]
271 old = [false]
272 dead = [false]
273 rank = -> (v) { v < 0 ? Float::INFINITY : v }
274 q = 0
275 while q < states.size
276 unless old[q]
277 es = enabled(states[q])
278 dead[q] = es.empty?
279 ancestors = []
280 a = q
281 while a >= 0
282 ancestors << a
283 a = parents[a]
284 end
285
286 es.each do |t|
287 m = fire(states[q], t)
288 ancestors.reverse_each do |ancestor|
289 pairs = m.zip(states[ancestor])
290 if pairs.all? { |x, y| rank.call(x) >= rank.call(y) } && pairs.any? { |x, y| rank.call(x) > rank.call(y) }
291 m.each_index { |p| m[p] = -1 if @capacity[p].zero? && rank.call(m[p]) > rank.call(states[ancestor][p]) }
292 end
293 end
294
295 if states.size == limit
296 puts("coverability tree limit #{limit} exceeded")
297 return
298 end
299
300 old << states.include?(m)
301 states << m
302 parents << q
303 via << t
304 dead << false
305 end
306 end
307
308 q += 1
309 end
310
311 puts("coverability tree nodes: #{states.size}")
312 states.each_with_index { |m, i|
313 puts(
314 " #{i} #{mark(m)} " +
315 (i.zero? ? "root" : "<- #{parents[i]} #{@transitions[via[i]]}") +
316 (old[i] ? " old" : "") +
317 (dead[i] ? " dead" : "")
318 )
319 }
320 maximum = @places.each_index.map { |p| states.any? { |m| m[p] < 0 } ? -1 : states.map { |m| m[p] }.max }
321 puts(
322 "bounds: #{mark(maximum)}",
323 "unbounded: #{listing(@places.each_index.filter_map { |p| @places[p] if maximum[p] < 0 })}"
324 )
325 end

327 def sets(trap)
328 found = (1...(1 << @places.size)).select do |mask|
329 @transitions.each_index.all? do |t|
330 input = output = false
331 @places.each_index { |p|
332 if mask & (1 << p) != 0
333 input ||= @post[p][t] > 0
334 output ||= @pre[p][t] > 0
335 end
336 }
337 trap ? !output || input : !input || output
338 end
339 end
340
341 found.reject { |m| found.any? { |s| s != m && s & m == s } }.sort_by { |m|
342 p = @places.each_index.select { |i| m & (1 << i) != 0 }
343 [p.size, p]
344 }
345 end

347 def set_text(mask) = "{" + @places.each_index.filter_map { |p| @places[p] if mask & (1 << p) != 0 }.join(",") + "}"
348 def realize(counts)
349 used = Array.new(counts.size, 0)
350 stack = [@current]
351 choices = [0]
352 result = []
353 seen = {}
354 total = counts.sum
355 until stack.empty?
356 depth = result.size
357 return result if depth == total
358 chosen = nil
359 (choices[depth]...counts.size).each do |t|
360 choices[depth] = t + 1
361 next if used[t] >= counts[t] || reason(stack[depth], t)
362 key = used.dup
363 key[t] += 1
364 next if seen[key]
365 seen[key] = true
366 chosen = t
367 break
368 end
369
370 if chosen
371 used[chosen] += 1
372 result << chosen
373 stack << fire(stack[depth], chosen)
374 choices << 0
375 else
376 stack.pop
377 choices.pop
378 used[result.pop] -= 1 unless result.empty?
379 end
380 end
381
382 nil
383 end

385 def invariants(transpose)
386 matrix = @places.each_index.map { |p| @transitions.each_index.map { |t| @post[p][t] - @pre[p][t] } }
387 matrix = matrix.transpose if transpose
388 flows = semiflows(matrix)
389 names = transpose ? @transitions : @places
390 covered = Array.new(names.size, false)
391 bounds = Array.new(names.size)
392 puts("#{transpose ? "T" : "P"}-semiflows: #{flows.size}")
393 flows.each do |coeff|
394 terms = coeff
395 .each_index
396 .filter_map { |i| (coeff[i] == 1 ? "" : "#{coeff[i]}*") + names[i] if coeff[i] > 0 }
397 .join(" + ")
398 coeff.each_index { |i| covered[i] = true if coeff[i] > 0 }
399 if transpose
400 sequence = realize(coeff)
401 puts(" #{terms}: " + (sequence ? "realizable " + self.names(sequence).join(" ") : "not realizable"))
402 else
403 value = coeff.zip(@current).sum { |a, b| a * b }
404 puts(" #{terms} = #{value}")
405 coeff.each_index { |p| bounds[p] = [bounds[p] || value / coeff[p], value / coeff[p]].min if coeff[p] > 0 }
406 end
407 end
408
409 puts("bounds: (" + bounds.map { |b| b ? b.to_s : "-" }.join(",") + ")") unless transpose
410 puts((transpose ? "consistent: " : "conservative: ") + yn(covered.all?))
411 end

413 def classify
414 inputs = @transitions.each_index.map { |t| @places.each_index.select { |p| @pre[p][t] > 0 } }
415 outputs = @places.each_index.map { |p| @transitions.each_index.select { |t| @pre[p][t] > 0 } }
416 ordinary = @places.each_index.all? { |p|
417 @transitions.each_index.all? { |t| @pre[p][t] <= 1 && @post[p][t] <= 1 }
418 }
419 pure = @places.each_index.all? { |p| @transitions.each_index.all? { |t| @pre[p][t].zero? || @post[p][t].zero? } }
420 sm = ordinary &&
421 inputs.each_index.all? { |t| inputs[t].size == 1 && @places.each_index.count { |p| @post[p][t] > 0 } == 1 }
422 mg = ordinary &&
423 outputs.each_index.all? { |p|
424 outputs[p].size == 1 && @transitions.each_index.count { |t| @post[p][t] > 0 } == 1
425 }
426 fc = ordinary && outputs.all? { |a| a.all? { |t| a.size == 1 || inputs[t].size == 1 } }
427 efc = ordinary && outputs.all? { |a| outputs.all? { |b| (a & b).empty? || a == b } }
428 [
429 ["ordinary", ordinary],
430 ["pure", pure],
431 ["state machine", sm],
432 ["marked graph", mg],
433 ["free choice", fc],
434 ["extended free choice", efc]
435 ].each { |name, v| puts("#{name}: #{yn(v)}") }
436 verdict = !fc ? "not applicable" : @places.size > 12 ? "too many places" : "holds"
437 if fc && @places.size <= 12
438 traps = sets(true)
439 sets(false).each do |s|
440 unless traps.any? { |q| q & s == q && @current.each_index.any? { |p| @current[p] > 0 && q & (1 << p) != 0 } }
441 verdict = "fails at " + set_text(s)
442 break
443 end
444 end
445 end
446
447 puts("commoner: " + verdict)
448 end

450 def workflow(limit)
451 source = @places.each_index.select { |p| @transitions.each_index.all? { |t| @post[p][t].zero? } }
452 sink = @places.each_index.select { |p| @transitions.each_index.all? { |t| @pre[p][t].zero? } }
453 if source.size != 1
454 puts("workflow: no (#{source.size} source places)")
455 return
456 end
457
458 if sink.size != 1
459 puts("workflow: no (#{sink.size} sink places)")
460 return
461 end
462
463 i = source[0]
464 o = sink[0]
465 n = @places.size
466 names = @places + @transitions
467 adj = Array.new(names.size) { [] }
468 rev = Array.new(names.size) { [] }
469 @places.each_index { |p|
470 @transitions.each_index { |t|
471 if @pre[p][t] > 0
472 adj[p] << n + t
473 rev[n + t] << p
474 end
475
476 if @post[p][t] > 0
477 adj[n + t] << p
478 rev[p] << n + t
479 end
480 }
481 }
482 f = reachable(adj, i)
483 b = reachable(rev, o)
484 names.each_index { |v|
485 unless f[v] && b[v]
486 puts("workflow: no (#{names[v]} is not on a path from #{@places[i]} to #{@places[o]})")
487 return
488 end
489 }
490 puts("workflow: source #{@places[i]} sink #{@places[o]}")
491 start = Array.new(n, 0)
492 start[i] = 1
493 goal = Array.new(n, 0)
494 goal[o] = 1
495 g = graph(start, limit)
496 if g[:aborted]
497 puts("reachability graph limit #{limit} exceeded")
498 return
499 end
500
501 goal_index = g[:states].index(goal)
502 rev = Array.new(g[:states].size) { [] }
503 g[:edges].each_with_index { |e, u| e.each { |_, v| rev[v] << u } }
504 option = goal_index && reachable(rev, goal_index).all?
505 proper = g[:states].all? { |m| m[o] < 1 || m == goal }
506 live = @transitions.each_index.all? { |t| g[:edges].any? { |e| e.any? { |u, _| t == u } } }
507 puts(
508 "option to complete: #{yn(option)}",
509 "proper completion: #{yn(proper)}",
510 "no dead transitions: #{yn(live)}",
511 "sound: #{yn(option && proper && live)}"
512 )
513 end

515 ARITY = {
516 "MARKING" => 0,
517 "RESET" => 0,
518 "SET" => 2,
519 "ENABLED" => 0,
520 "FIRE" => -1,
521 "UNDO" => 1,
522 "SIMULATE" => 3,
523 "CONFLICTS" => 0,
524 "REACH" => 1,
525 "GRAPH" => 1,
526 "COVER" => 1,
527 "PATH" => 2,
528 "INCIDENCE" => 0,
529 "PINV" => 0,
530 "TINV" => 0,
531 "SIPHONS" => 0,
532 "TRAPS" => 0,
533 "CLASSIFY" => 0,
534 "WORKFLOW" => 1,
535 "SHOW" => 0
536 }.freeze
537 def execute(words)
538 cmd, *args = words
539 raise "unknown command #{cmd}" unless ARITY.key?(cmd)
540 raise "wrong number of arguments" unless ARITY[cmd] < 0 ? !args.empty? : args.size == ARITY[cmd]
541 if %w[REACH GRAPH COVER PATH WORKFLOW].include?(cmd)
542 limit = number(args[0], 1, 2000)
543 raise "invalid limit #{args[0]}" unless limit
544 end
545
546 case cmd
547 when "MARKING"
548 puts("marking #{mark(@current)}")
549 when "RESET"
550 @current = @initial.dup
551 @history = []
552 puts("marking #{mark(@current)}")
553 when "SET"
554 p = @places.index(args[0])
555 raise "unknown place #{args[0]}" unless p
556 v = number(args[1], 0, 999)
557 raise "invalid tokens #{args[1]}" unless v
558 raise "tokens exceed capacity of #{args[0]}" if @capacity[p] > 0 && v > @capacity[p]
559 @current = @current.dup
560 @current[p] = v
561 @history = []
562 puts("marking #{mark(@current)}")
563 when "ENABLED"
564 puts("enabled: " + listing(names(enabled(@current))))
565 when "FIRE"
566 ts = args.map { |name| @transitions.index(name) || (raise("unknown transition #{name}")) }
567 ts.each_with_index { |t, i|
568 if (why = reason(@current, t))
569 puts("step #{i + 1}: #{args[i]} not enabled (#{why}) at #{mark(@current)}")
570 return
571 end
572
573 commit(t)
574 }
575 puts("fired #{ts.size} -> #{mark(@current)}")
576 when "UNDO"
577 k = number(args[0], 1, 1000)
578 raise "invalid count #{args[0]}" unless k
579 raise "cannot undo #{k}: history size #{@history.size}" if k > @history.size
580 @current = @history[-k]
581 @history.pop(k)
582 puts("marking #{mark(@current)}")
583 when "SIMULATE"
584 steps = number(args[0], 1, 1000)
585 x = number(args[1], 1, 2147483646)
586 raise "invalid steps #{args[0]}" unless steps
587 raise "invalid seed #{args[1]}" unless x
588 raise "invalid mode #{args[2]}" unless %w[PRIORITY RANDOM].include?(args[2])
589 counts = Array.new(@transitions.size, 0)
590 done = 0
591 while done < steps
592 e = enabled(@current)
593 if e.empty?
594 puts("dead (steps=#{done})")
595 break
596 end
597
598 if args[2] == "RANDOM"
599 x = x * 48271 % 2147483647
600 t = e[x % e.size]
601 else
602 t = e.max_by { |u| [@priority[u], -u] }
603 end
604
605 commit(t)
606 counts[t] += 1
607 done += 1
608 puts("step #{done}: #{@transitions[t]} -> #{mark(@current)}")
609 end
610
611 puts("completed (steps=#{steps})") if done == steps
612 puts("counts: " + @transitions.each_index.map { |t| "#{@transitions[t]}=#{counts[t]}" }.join(" "))
613 when "CONFLICTS"
614 structural = []
615 effective = []
616 @transitions.each_index { |u|
617 ((u + 1)...@transitions.size).each { |v|
618 pair = "#{@transitions[u]}/#{@transitions[v]}"
619 structural << pair if @places.each_index.any? { |p| @pre[p][u] > 0 && @pre[p][v] > 0 }
620 if !reason(@current, u) &&
621 !reason(@current, v) &&
622 (reason(fire(@current, u), v) || reason(fire(@current, v), u))
623 effective << pair
624 end
625 }
626 }
627 puts("structural: " + listing(structural), "effective: " + listing(effective))
628 when "REACH", "GRAPH"
629 reach(limit, cmd == "GRAPH")
630 when "COVER"
631 cover(limit)
632 when "PATH"
633 entries = args[1].split(",")
634 raise "target needs #{@places.size} entries" unless entries.size == @places.size
635 target = entries.map { |s|
636 if s == "*"
637 [0, -1]
638 else
639 plus = s.end_with?("+")
640 v = number(plus ? s[0...-1] : s, 0, 9999999999)
641 raise "invalid target entry #{s}" unless v
642 [v, plus ? -1 : v]
643 end
644 }
645 g = graph(@current, limit, target)
646 if g[:found]
647 k = g[:found]
648 sequence = names(path(g, k))
649 puts(
650 "reached #{mark(g[:states][k])} length=#{sequence.size}" + (sequence.empty? ? "" : ": " + sequence.join(" "))
651 )
652 elsif g[:aborted]
653 puts("search limit #{limit} exceeded")
654 else
655 puts("unreachable (markings explored: #{g[:states].size})")
656 end
657
658 when "INCIDENCE"
659 rows = [[""] + @transitions] +
