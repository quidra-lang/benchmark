35List<List<int>> flows(List<List<int>> a) {
36 final n = a.length, m = a.first.length;
37 var rows = [
38 for (var i = 0; i < n; ++i)
39 [...a[i], for (var j = 0; j < n; ++j) i == j ? 1 : 0],
40 ];
41 for (var j = 0; j < m; ++j) {
42 final unique = <String, List<int>>{};
43 for (final r in rows) {
44 if (r[j] == 0) unique[r.join(',')] = r;
45 }
46 for (final r in rows) {
47 if (r[j] > 0) {
48 for (final s in rows) {
49 if (s[j] < 0) {
50 final v = [
51 for (var k = 0; k < m + n; ++k) -s[j] * r[k] + r[j] * s[k],
52 ];
53 final d = v.fold(0, (g, x) => g.gcd(x));
54 for (var k = 0; k < v.length; ++k) {
55 v[k] ~/= d;
56 }
57 unique[v.join(',')] = v;
58 }
59 }
60 }
61 }
62 final candidates = unique.values.toList();
63 final supports = [
64 for (final r in candidates)
65 [
66 for (var i = 0; i < n; ++i)
67 if (r[m + i] > 0) 1 << i,
68 ].fold(0, (s, v) => s | v),
69 ];
70 rows = [
71 for (var i = 0; i < candidates.length; ++i)
72 if (!supports.any((s) => s != supports[i] && (s & supports[i]) == s))
73 candidates[i],
74 ];
75 }
76 return rows.map((r) => r.sublist(m)).toList()..sort((a, b) => lex(b, a));
77}

118 bad('duplicate name $a');

665 'GRAPH': 1,
666 'COVER': 1,
667 'PATH': 2,
668 'INCIDENCE': 0,
669 'PINV': 0,
670 'TINV': 0,
671 'SIPHONS': 0,
672 'TRAPS': 0,
673 'CLASSIFY': 0,
674 'WORKFLOW': 1,
675 'SHOW': 0,
676 };
677 final cmd = w[0], expected = arity[w[0]];
678 if (expected == null) bad('unknown command $cmd');
679 if (expected < 0 ? w.length < 2 : w.length != expected + 1)
680 bad('wrong number of arguments');
681 final limit = ['REACH', 'GRAPH', 'COVER', 'PATH', 'WORKFLOW'].contains(cmd)
682 ? number(w[1], 1, 2000) ?? bad('invalid limit ${w[1]}')
683 : 0;
684 switch (cmd) {
685 case 'MARKING':
686 case 'RESET':
687 if (cmd == 'RESET') {
688 current = List.of(initial);
689 history.clear();
690 }
691 print('marking ${mark(current)}');
692 case 'SET':
693 final p = places.indexOf(w[1]);
694 if (p < 0) bad('unknown place ${w[1]}');
695 final v = number(w[2], 0, 999) ?? bad('invalid tokens ${w[2]}');
696 if (capacity[p] > 0 && v > capacity[p])
697 bad('tokens exceed capacity of ${w[1]}');
698 current = List.of(current);
699 current[p] = v;
700 history.clear();
701 print('marking ${mark(current)}');
702 case 'ENABLED':
703 print('enabled: ${listed(names(enabled(current)))}');
704 case 'FIRE':
705 final ts = [
706 for (final name in w.skip(1))
707 transitions.indexOf(name) >= 0
708 ? transitions.indexOf(name)
709 : bad('unknown transition $name'),
710 ];
711 for (var i = 0; i < ts.length; ++i) {
712 final r = reason(current, ts[i]);
713 if (r != null) {
714 print(
715 'step ${i + 1}: ${transitions[ts[i]]} not enabled ($r) at ${mark(current)}',
716 );
717 return;
718 }
719 commit(ts[i]);
720 }
721 print('fired ${ts.length} -> ${mark(current)}');
722 case 'UNDO':
723 final k = number(w[1], 1, 1000) ?? bad('invalid count ${w[1]}');
724 if (k > history.length)
725 bad('cannot undo $k: history size ${history.length}');
726 current = history[history.length - k];
727 history.removeRange(history.length - k, history.length);
728 print('marking ${mark(current)}');
729 case 'SIMULATE':
730 final steps = number(w[1], 1, 1000) ?? bad('invalid steps ${w[1]}');
731 var x = number(w[2], 1, 2147483646) ?? bad('invalid seed ${w[2]}');
732 if (!['PRIORITY', 'RANDOM'].contains(w[3])) bad('invalid mode ${w[3]}');
733 final counts = List.filled(transitions.length, 0);
734 var done = 0;
735 for (; done < steps; ++done) {
736 final es = enabled(current);
737 if (es.isEmpty) {
738 print('dead (steps=$done)');
739 break;
740 }
741 var t = es.first;
742 if (w[3] == 'RANDOM') {
743 x = x * 48271 % 2147483647;
744 t = es[x % es.length];
745 } else {
746 for (final u in es) {
747 if (priority[u] > priority[t]) t = u;
748 }
749 }
750 commit(t);
751 ++counts[t];
752 print('step ${done + 1}: ${transitions[t]} -> ${mark(current)}');
753 }
754 if (done == steps) print('completed (steps=$steps)');
755 print(
756 'counts: ${[for (var t = 0; t < transitions.length; ++t) '${transitions[t]}=${counts[t]}'].join(' ')}',
757 );
758 case 'CONFLICTS':
759 final s = <String>[], e = <String>[];
760 for (var u = 0; u < transitions.length; ++u) {
761 for (var v = u + 1; v < transitions.length; ++v) {
762 final pair = '${transitions[u]}/${transitions[v]}';
763 if ([
764 for (var p = 0; p < places.length; ++p)
765 pre[p][u] > 0 && pre[p][v] > 0,
766 ].any((v) => v))
767 s.add(pair);
768 if (reason(current, u) == null &&
769 reason(current, v) == null &&
770 (reason(fire(current, u), v) != null ||
771 reason(fire(current, v), u) != null))
772 e.add(pair);
773 }
774 }
775 print('structural: ${listed(s)}\neffective: ${listed(e)}');
776 case 'REACH':
777 case 'GRAPH':
778 reach(limit, cmd == 'GRAPH');
779 case 'COVER':
780 cover(limit);
781 case 'PATH':
782 final entries = w[2].split(',');
783 if (entries.length != places.length)
784 bad('target needs ${places.length} entries');
785 final target = <(int, int)>[];
786 for (final s in entries) {
787 if (s == '*') {
788 target.add((0, -1));
789 continue;
790 }
791 final plus = s.endsWith('+'),
792 v =
793 number(
794 plus ? s.substring(0, s.length - 1) : s,
795 0,
796 9999999999,
797 ) ??
798 bad('invalid target entry $s');
799 target.add((v, plus ? -1 : v));
800 }
801 final g = graph(current, limit, target);
802 if (g.found >= 0) {
803 final a = names(path(g, g.found));
804 print(
805 'reached ${mark(g.states[g.found])} length=${a.length}${a.isEmpty ? '' : ': ${a.join(' ')}'}',
806 );
807 } else if (g.aborted) {
808 print('search limit $limit exceeded');
809 } else {
810 print('unreachable (markings explored: ${g.states.length})');
811 }
812 case 'INCIDENCE':
813 final rows = [
814 ['', ...transitions],
815 for (var p = 0; p < places.length; ++p)
816 [
817 places[p],
818 for (var t = 0; t < transitions.length; ++t)
819 '${post[p][t] - pre[p][t]}',
820 ],
821 ];
822 final width = [
823 for (var i = 0; i < rows.first.length; ++i)
824 rows.map((r) => r[i].length).reduce(math.max),
825 ];
826 for (final r in rows) {
827 print(
828 [
829 for (var i = 0; i < r.length; ++i)
830 i == 0 ? r[i].padRight(width[i]) : r[i].padLeft(width[i]),
831 ].join(' ').trimRight(),
832 );
833 }
834 case 'PINV':
835 case 'TINV':
836 invariants(cmd == 'TINV');
837 case 'SIPHONS':
838 case 'TRAPS':
839 if (places.length > 12) bad('more than 12 places');
840 final ss = sets(cmd == 'TRAPS');
841 print('minimal ${cmd == 'TRAPS' ? 'traps' : 'siphons'}: ${ss.length}');
842 for (final s in ss) {
843 print(
844 ' ${setText(s)} tokens=${bits(s).fold(0, (sum, p) => sum + current[p])}',
845 );
846 }
847 case 'CLASSIFY':
848 classify();
849 case 'WORKFLOW':
850 workflow(limit);
851 case 'SHOW':
852 for (var p = 0; p < places.length; ++p) {
853 print(
854 'place ${places[p]} tokens=${initial[p]} capacity=${capacity[p] == 0 ? 'none' : capacity[p]}',
855 );
856 }
857 for (var t = 0; t < transitions.length; ++t) {
858 String list(List<List<int>> mat) {
859 final a = [
860 for (var p = 0; p < places.length; ++p)
861 if (mat[p][t] > 0) '${places[p]}:${mat[p][t]}',
862 ];
863 return a.isEmpty ? '-' : a.join(',');
864 }
865
866 print(
867 'transition ${transitions[t]} priority=${priority[t]} pre=${list(pre)} post=${list(post)} inhibit=${list(inhibitor)}',
868 );
869 }
870 }
871 }
