14fun visit(adj: List<List<Int>>, start: Int): BooleanArray {
15 val seen = BooleanArray(adj.size)
16 val queue = ArrayDeque<Int>()
17 queue.add(start)
18 seen[start] = true
19 while (queue.isNotEmpty()) for (v in adj[queue.removeFirst()]) if (!seen[v]) {
20 seen[v] = true
21 queue.add(v)
22 }
23 return seen
24}

33fun flows(a: List<LongArray>): List<LongArray> {
34 val n = a.size
35 val m = a[0].size
36 var rows = a.mapIndexed { i, r -> r + LongArray(n) { j -> if (i == j) 1 else 0 } }
37 for (j in 0 until m) {
38 val unique = linkedMapOf<List<Long>, LongArray>()
39 for (r in rows) if (r[j] == 0L) unique[r.toList()] = r
40 for (r in rows) if (r[j] > 0)
41 for (s in rows) if (s[j] < 0) {
42 val v = LongArray(m + n) { k -> -s[j] * r[k] + r[j] * s[k] }
43 val d = v.fold(0L, ::gcd)
44 for (k in v.indices) v[k] /= d
45 unique[v.toList()] = v
46 }
47 val candidates = unique.values.toList()
48 val supports = candidates.map { r ->
49 (0 until n).fold(0) { mask, i -> if (r[m + i] > 0) mask or (1 shl i) else mask }
50 }
51 rows = candidates.filterIndexed { i, _ ->
52 supports.none { it != supports[i] && it and supports[i] == it }
53 }
54 }
55 return rows.map { it.copyOfRange(m, m + n) }.sortedWith { a, b -> lex(b, a) }
56}

90 fun define(w: List<String>) {
91 val cmd = w[0]
92 val n = w.size - 1
93 if (cmd !in listOf("place", "transition", "arc", "inhibit")) bad("unknown definition $cmd")
94 if (
95 !(when (cmd) {
96 "place" -> n in 2..3
97 "transition" -> n in 1..2
98 else -> n == 3
99 })
100 )
101 bad("wrong number of arguments")
102 val a = w[1]
103 when (cmd) {
104 "place",
105 "transition" -> {
106 if (a in places || a in transitions) bad("duplicate name $a")
107 if (cmd == "place") {
108 if (n == 3 && w[2].toLong() > w[3].toLong()) bad("tokens exceed capacity of $a")
109 places.add(a)
110 initial.add(w[2].toLong())
111 capacity.add(if (n == 3) w[3].toLong() else 0)
112 } else {
113 transitions.add(a)
114 priority.add(if (n == 2) w[2].toLong() else 0)
115 }
116 }
117 "arc" -> {
118 val b = w[2]
119 for (x in listOf(a, b)) if (x !in places && x !in transitions)
120 bad("unknown node $x")
121 val pa = places.indexOf(a)
122 val pb = places.indexOf(b)
123 if ((pa >= 0) == (pb >= 0)) bad("arc must connect a place and a transition")
124 val p = if (pa >= 0) pa else pb
125 val t = transitions.indexOf(if (pa >= 0) b else a)
126 val matrix = if (pa >= 0) pre else post
127 if (matrix[p][t] > 0) bad("duplicate arc $a $b")
128 matrix[p][t] = w[3].toLong()
129 ++arcs
130 }
131 else -> {
132 val p = places.indexOf(a)
133 val t = transitions.indexOf(w[2])
134 if (p < 0) bad("unknown place $a")
135 if (t < 0) bad("unknown transition ${w[2]}")
136 if (inhibitor[p][t] > 0) bad("duplicate inhibitor $a ${w[2]}")
137 inhibitor[p][t] = w[3].toLong()
138 ++inhibitors
139 }
140 }
141 }

143 fun reason(m: LongArray, t: Int): String? {
144 for (p in places.indices) if (m[p] >= 0 && m[p] < pre[p][t])
145 return "insufficient tokens in ${places[p]}"
146 for (p in places.indices) if (inhibitor[p][t] > 0 && (m[p] < 0 || m[p] >= inhibitor[p][t]))
147 return "inhibited by ${places[p]}"
148 for (p in places.indices) if (
149 capacity[p] > 0 && (m[p] < 0 || m[p] - pre[p][t] + post[p][t] > capacity[p])
150 )
151 return "capacity of ${places[p]} exceeded"
152 return null
153 }

167 fun graph(start: LongArray, limit: Int, target: List<Pair<Long, Long>>? = null): Graph {
168 val g = Graph()
169 g.add(start.copyOf(), -1, -1)
170 val index = mutableMapOf(start.toList() to 0)
171 fun matches(m: LongArray) =
172 target != null &&
173 m.indices.all { p ->
174 m[p] >= target[p].first && (target[p].second < 0 || m[p] == target[p].second)
175 }
176 if (matches(start)) {
177 g.found = 0
178 return g
179 }
180 var q = 0
181 while (q < g.states.size) {
182 for (t in enabled(g.states[q])) {
183 val m = fire(g.states[q], t)
184 var k = index[m.toList()]
185 if (k == null) {
186 if (g.states.size == limit) {
187 g.aborted = true
188 return g
189 }
190 k = g.states.size
191 index[m.toList()] = k
192 g.add(m, q, t)
193 if (matches(m)) {
194 g.found = k
195 return g
196 }
197 }
198 g.edges[q].add(Edge(t, k))
199 }
200 ++q
201 }
202 return g
203 }

215 fun reach(limit: Int, display: Boolean) {
216 val g = graph(current, limit)
217 if (g.aborted) {
218 println("reachability graph limit $limit exceeded")
219 return
220 }
221 val n = g.states.size
222 if (display) {
223 for (i in 0 until n) {
224 val a = g.edges[i].map { "${transitions[it.transition]}->M${it.target}" }
225 println(
226 "M$i ${mark(g.states[i])}: " + if (a.isEmpty()) "dead" else a.joinToString(" "),
227 )
228 }
229 return
230 }
231 val adj = g.edges.map { e -> e.map { it.target } }
232 val closure = List(n) { visit(adj, it) }
233 val comp = IntArray(n) { -1 }
234 var components = 0
235 for (i in 0 until n) if (comp[i] < 0) {
236 for (j in i until n) if (closure[i][j] && closure[j][i]) comp[j] = components
237 ++components
238 }
239 val terminal = BooleanArray(components) { true }
240 val home = BooleanArray(n) { true }
241 val maximum = LongArray(places.size)
242 val dead = mutableListOf<Int>()
243 var edgeCount = 0
244 var persistent = true
245 for (i in 0 until n) {
246 if (g.edges[i].isEmpty()) dead.add(i)
247 edgeCount += g.edges[i].size
248 for (p in places.indices) maximum[p] = maxOf(maximum[p], g.states[i][p])
249 for (j in 0 until n) home[j] = home[j] && closure[i][j]
250 for (e in g.edges[i]) {
251 if (comp[i] != comp[e.target]) terminal[comp[i]] = false
252 for (u in g.edges[i]) if (
253 u.transition != e.transition && reason(g.states[e.target], u.transition) != null
254 )
255 persistent = false
256 }
257 }
258 println(
259 "reachable markings: $n\nedges: $edgeCount\ncomponents: $components (terminal ${terminal.count { it }})\ndead markings: ${dead.size}",
260 )
261 for (i in dead) {
262 val a = names(path(g, i))
263 println(
264 " M$i ${mark(g.states[i])} path " +
265 if (a.isEmpty()) "empty" else a.joinToString(" "),
266 )
267 }
268 val bound = maximum.maxOrNull()!!
269 println(
270 "max tokens: ${mark(maximum)}\nbound: $bound\nsafe: ${yes(bound<=1)}\nhome markings: ${home.count { it }}\nreversible: ${yes(home[0])}\npersistent: ${yes(persistent)}",
271 )
272 val levels =
273 transitions.indices.map { t ->
274 val present = BooleanArray(components)
275 var used = false
276 var cycle = false
277 for (i in 0 until n) for (e in g.edges[i]) if (e.transition == t) {
278 used = true
279 present[comp[i]] = true
280 if (comp[i] == comp[e.target]) cycle = true
281 }
282 val live = terminal.indices.all { !terminal[it] || present[it] }
283 transitions[t] +
284 "=" +
285 if (live) "L4" else if (cycle) "L3" else if (used) "L1" else "L0"
286 }
287 println("liveness: " + levels.joinToString(" "))
288 }

292 fun cover(limit: Int) {
293 val g = Graph()
294 g.add(current.copyOf(), -1, -1)
295 val old = mutableListOf(false)
296 val dead = mutableListOf(false)
297 val seen = mutableSetOf(current.toList())
298 var q = 0
299 while (q < g.states.size) {
300 if (!old[q]) {
301 val es = enabled(g.states[q])
302 dead[q] = es.isEmpty()
303 val ancestors = mutableListOf<Int>()
304 var a = q
305 while (a >= 0) {
306 ancestors.add(a)
307 a = g.parent[a]
308 }
309 ancestors.reverse()
310 for (t in es) {
311 val m = fire(g.states[q], t)
312 for (ancestor in ancestors) if (
313 m.indices.all { rank(m[it]) >= rank(g.states[ancestor][it]) } &&
314 m.indices.any { rank(m[it]) > rank(g.states[ancestor][it]) }
315 )
316 for (p in m.indices) if (
317 capacity[p] == 0L && rank(m[p]) > rank(g.states[ancestor][p])
318 )
319 m[p] = -1
320 if (g.states.size == limit) {
321 println("coverability tree limit $limit exceeded")
322 return
323 }
324 old.add(!seen.add(m.toList()))
325 dead.add(false)
326 g.add(m, q, t)
327 }
328 }
329 ++q
330 }
331 println("coverability tree nodes: ${g.states.size}")
332 val maximum = LongArray(places.size)
333 for (i in g.states.indices) {
334 println(
335 " $i ${mark(g.states[i])} " +
336 (if (i == 0) "root" else "<- ${g.parent[i]} ${transitions[g.via[i]]}") +
337 (if (old[i]) " old" else "") +
338 (if (dead[i]) " dead" else ""),
339 )
340 for (p in places.indices) if (rank(g.states[i][p]) > rank(maximum[p]))
341 maximum[p] = g.states[i][p]
342 }
343 println(
344 "bounds: ${mark(maximum)}\nunbounded: " +
345 listed(places.indices.filter { maximum[it] < 0 }.map { places[it] }),
346 )
347 }

351 fun sets(trap: Boolean): List<Int> {
352 val found =
353 (1 until (1 shl places.size)).filter { mask ->
354 transitions.indices.all { t ->
355 val input = bits(mask).any { post[it][t] > 0 }
356 val output = bits(mask).any { pre[it][t] > 0 }
357 if (trap) !output || input else !input || output
358 }
359 }
360 return found
361 .filter { m -> found.none { it != m && it and m == it } }
362 .sortedWith { a, b ->
363 val x = bits(a)
364 val y = bits(b)
365 if (x.size != y.size) x.size.compareTo(y.size)
366 else lex(x.map { it.toLong() }.toLongArray(), y.map { it.toLong() }.toLongArray())
367 }

372 fun realize(counts: LongArray): List<Int>? {
373 val used = LongArray(counts.size)
374 val stack = mutableListOf(current)
375 val choices = mutableListOf(0)
376 val result = mutableListOf<Int>()
377 val seen = mutableSetOf<List<Long>>()
378 val total = counts.sum()
379 while (stack.isNotEmpty()) {
380 val d = result.size
381 if (d.toLong() == total) return result
382 var chosen = -1
383 for (t in choices[d] until counts.size) {
384 choices[d] = t + 1
385 if (used[t] >= counts[t] || reason(stack[d], t) != null) continue
386 ++used[t]
387 val fresh = seen.add(used.toList())
388 --used[t]
389 if (fresh) {
390 chosen = t
391 break
392 }
393 }
394 if (chosen >= 0) {
395 ++used[chosen]
396 result.add(chosen)
397 stack.add(fire(stack[d], chosen))
398 choices.add(0)
399 } else {
400 stack.removeAt(stack.lastIndex)
401 choices.removeAt(choices.lastIndex)
402 if (result.isNotEmpty()) --used[result.removeAt(result.lastIndex)]
403 }
404 }
405 return null
406 }

408 fun invariants(transpose: Boolean) {
409 val matrix =
410 if (transpose)
411 transitions.indices.map { t ->
412 LongArray(places.size) { p -> post[p][t] - pre[p][t] }
413 }
414 else
415 places.indices.map { p ->
416 LongArray(transitions.size) { t -> post[p][t] - pre[p][t] }
417 }
418 val f = flows(matrix)
419 val ns = if (transpose) transitions else places
420 val covered = BooleanArray(ns.size)
421 val bounds = LongArray(ns.size) { -1 }
422 println((if (transpose) "T" else "P") + "-semiflows: ${f.size}")
423 for (c in f) {
424 val terms =
425 c.indices
426 .filter { c[it] > 0 }
427 .joinToString(" + ") { i -> (if (c[i] == 1L) "" else "${c[i]}*") + ns[i] }
428 for (i in c.indices) if (c[i] > 0) covered[i] = true
429 if (transpose) {
430 val sequence = realize(c)
431 println(
432 " $terms: " +
433 if (sequence == null) "not realizable"
434 else "realizable " + names(sequence).joinToString(" "),
435 )
436 } else {
437 val value = c.indices.sumOf { c[it] * current[it] }
438 println(" $terms = $value")
439 for (p in c.indices) if (c[p] > 0)
440 bounds[p] = if (bounds[p] < 0) value / c[p] else minOf(bounds[p], value / c[p])
441 }
442 }
443 if (!transpose)
444 println(
445 "bounds: " +
446 bounds.joinToString(",", "(", ")") { if (it < 0) "-" else it.toString() },
447 )
448 println((if (transpose) "consistent: " else "conservative: ") + yes(covered.all { it }))
449 }

451 fun classify() {
452 val inputs = transitions.indices.map { t -> places.indices.filter { pre[it][t] > 0 } }
453 val outputs = places.indices.map { p -> transitions.indices.filter { pre[p][it] > 0 } }
454 val ordinary =
455 places.indices.all { p ->
456 transitions.indices.all { pre[p][it] <= 1 && post[p][it] <= 1 }
457 }
458 val pure =
459 places.indices.all { p ->
460 transitions.indices.all { pre[p][it] == 0L || post[p][it] == 0L }
461 }
462 val sm =
463 ordinary &&
464 transitions.indices.all { t ->
465 inputs[t].size == 1 && places.indices.count { post[it][t] > 0 } == 1
466 }
467 val mg =
468 ordinary &&
469 places.indices.all { p ->
470 outputs[p].size == 1 && transitions.indices.count { post[p][it] > 0 } == 1
471 }
472 val fc = ordinary && outputs.all { a -> a.all { a.size == 1 || inputs[it].size == 1 } }
473 val efc =
474 ordinary &&
475 outputs.all { a -> outputs.all { b -> a.intersect(b.toSet()).isEmpty() || a == b } }
476 for ((label, value) in
477 listOf(
478 "ordinary" to ordinary,
479 "pure" to pure,
480 "state machine" to sm,
481 "marked graph" to mg,
482 "free choice" to fc,
483 "extended free choice" to efc,
484 )) println("$label: ${yes(value)}")
485 var verdict =
486 if (!fc) "not applicable" else if (places.size > 12) "too many places" else "holds"
487 if (fc && places.size <= 12) {
488 val traps = sets(true)
489 for (s in sets(false)) if (
490 traps.none { q -> s and q == q && bits(q).any { current[it] > 0 } }
491 ) {
492 verdict = "fails at " + setText(s)
493 break
494 }
495 }
496 println("commoner: $verdict")
497 }

499 fun workflow(limit: Int) {
500 val P = places.size
501 val T = transitions.size
502 val sources = places.indices.filter { p -> transitions.indices.all { post[p][it] == 0L } }
503 val sinks = places.indices.filter { p -> transitions.indices.all { pre[p][it] == 0L } }
504 if (sources.size != 1) {
505 println("workflow: no (${sources.size} source places)")
506 return
507 }
508 if (sinks.size != 1) {
509 println("workflow: no (${sinks.size} sink places)")
510 return
511 }
512 val i = sources[0]
513 val o = sinks[0]
514 val ns = places + transitions
515 val adj = List(P + T) { mutableListOf<Int>() }
516 val rev = List(P + T) { mutableListOf<Int>() }
517 for (p in places.indices) for (t in transitions.indices) {
518 if (pre[p][t] > 0) {
519 adj[p].add(P + t)
520 rev[P + t].add(p)
521 }
522 if (post[p][t] > 0) {
523 adj[P + t].add(p)
524 rev[p].add(P + t)
525 }
526 }
527 val f = visit(adj, i)
528 val b = visit(rev, o)
529 for (v in ns.indices) if (!f[v] || !b[v]) {
530 println("workflow: no (${ns[v]} is not on a path from ${places[i]} to ${places[o]})")
531 return
532 }
533 println("workflow: source ${places[i]} sink ${places[o]}")
534 val start = LongArray(P)
535 val goal = LongArray(P)
536 start[i] = 1
537 goal[o] = 1
538 val g = graph(start, limit)
539 if (g.aborted) {
540 println("reachability graph limit $limit exceeded")
541 return
542 }
543 val finish = g.states.indexOfFirst { it.contentEquals(goal) }
544 val back = List(g.states.size) { mutableListOf<Int>() }
545 val used = BooleanArray(T)
546 for (u in g.states.indices) for (e in g.edges[u]) {
547 back[e.target].add(u)
548 used[e.transition] = true
549 }
550 val option = finish >= 0 && visit(back, finish).all { it }
551 val proper = g.states.all { it[o] < 1 || it.contentEquals(goal) }
552 val live = used.all { it }
553 println(
554 "option to complete: ${yes(option)}\nproper completion: ${yes(proper)}\nno dead transitions: ${yes(live)}\nsound: ${yes(option&&proper&&live)}",
555 )
556 }

558 fun execute(w: List<String>) {
559 val arity =
560 mapOf(
561 "MARKING" to 0,
562 "RESET" to 0,
563 "SET" to 2,
564 "ENABLED" to 0,
565 "FIRE" to -1,
566 "UNDO" to 1,
567 "SIMULATE" to 3,
568 "CONFLICTS" to 0,
569 "REACH" to 1,
570 "GRAPH" to 1,
571 "COVER" to 1,
572 "PATH" to 2,
573 "INCIDENCE" to 0,
574 "PINV" to 0,
575 "TINV" to 0,
576 "SIPHONS" to 0,
577 "TRAPS" to 0,
578 "CLASSIFY" to 0,
579 "WORKFLOW" to 1,
580 "SHOW" to 0,
581 )
582 val cmd = w[0]
583 val expected = arity[cmd] ?: bad("unknown command $cmd")
584 if (if (expected < 0) w.size < 2 else w.size != expected + 1)
585 bad("wrong number of arguments")
586 val limit =
587 if (cmd in listOf("REACH", "GRAPH", "COVER", "PATH", "WORKFLOW"))
588 (number(w[1], 1, 2000) ?: bad("invalid limit ${w[1]}")).toInt()
589 else 0
590 when (cmd) {
591 "MARKING",
592 "RESET" -> {
593 if (cmd == "RESET") {
594 current = initial.toLongArray()
595 history.clear()
596 }
597 println("marking " + mark(current))
598 }
599 "SET" -> {
600 val p = places.indexOf(w[1])
601 if (p < 0) bad("unknown place ${w[1]}")
602 val v = number(w[2], 0, 999) ?: bad("invalid tokens ${w[2]}")
603 if (capacity[p] > 0 && v > capacity[p]) bad("tokens exceed capacity of ${w[1]}")
604 current = current.copyOf()
605 current[p] = v
606 history.clear()
607 println("marking " + mark(current))
608 }
609 "ENABLED" -> println("enabled: " + listed(names(enabled(current))))
610 "FIRE" -> {
611 val ts =
612 w.drop(1).map { name ->
613 transitions.indexOf(name).takeIf { it >= 0 }
614 ?: bad("unknown transition $name")
615 }
616 for ((i, t) in ts.withIndex()) {
617 val reason = reason(current, t)
618 if (reason != null) {
619 println(
620 "step ${i+1}: ${transitions[t]} not enabled ($reason) at ${mark(current)}",
621 )
622 return
623 }
624 commit(t)
625 }
626 println("fired ${ts.size} -> ${mark(current)}")
627 }
628 "UNDO" -> {
629 val k = (number(w[1], 1, 1000) ?: bad("invalid count ${w[1]}")).toInt()
630 if (k > history.size) bad("cannot undo $k: history size ${history.size}")
631 current = history[history.size - k]
632 history.subList(history.size - k, history.size).clear()
633 println("marking " + mark(current))
634 }
635 "SIMULATE" -> {
636 val steps = (number(w[1], 1, 1000) ?: bad("invalid steps ${w[1]}")).toInt()
637 var x = number(w[2], 1, 2147483646) ?: bad("invalid seed ${w[2]}")
638 if (w[3] !in listOf("PRIORITY", "RANDOM")) bad("invalid mode ${w[3]}")
639 val counts = IntArray(transitions.size)
640 var done = 0
641 while (done < steps) {
642 val es = enabled(current)
643 if (es.isEmpty()) {
644 println("dead (steps=$done)")
645 break
646 }
647 var t = es[0]
648 if (w[3] == "RANDOM") {
649 x = x * 48271 % 2147483647
650 t = es[(x % es.size).toInt()]
651 } else for (u in es) if (priority[u] > priority[t]) t = u
652 commit(t)
653 ++counts[t]
654 ++done
655 println("step $done: ${transitions[t]} -> ${mark(current)}")
656 }
657 if (done == steps) println("completed (steps=$steps)")
658 println(
659 "counts: " +
660 transitions.indices.joinToString(" ") {
661 "${transitions[it]}=${counts[it]}"
662 },
663 )
664 }
665 "CONFLICTS" -> {
666 val s = mutableListOf<String>()
667 val e = mutableListOf<String>()
668 for (u in transitions.indices) for (v in u + 1 until transitions.size) {
669 val pair = "${transitions[u]}/${transitions[v]}"
670 if (places.indices.any { pre[it][u] > 0 && pre[it][v] > 0 }) s.add(pair)
671 if (
672 reason(current, u) == null &&
673 reason(current, v) == null &&
674 (reason(fire(current, u), v) != null ||
675 reason(fire(current, v), u) != null)
676 )
677 e.add(pair)
678 }
679 println("structural: ${listed(s)}\neffective: ${listed(e)}")
680 }
681 "REACH",
682 "GRAPH" -> reach(limit, cmd == "GRAPH")
683 "COVER" -> cover(limit)
684 "PATH" -> {
685 val entries = w[2].split(',')
686 if (entries.size != places.size) bad("target needs ${places.size} entries")
687 val target = entries.map { s ->
688 if (s == "*") 0L to -1L
689 else {
690 val plus = s.endsWith('+')
691 val v =
692 number(if (plus) s.dropLast(1) else s, 0, 9999999999)
693 ?: bad("invalid target entry $s")
694 v to if (plus) -1L else v
695 }
696 }
697 val g = graph(current, limit, target)
698 if (g.found >= 0) {
699 val a = names(path(g, g.found))
700 println(
701 "reached ${mark(g.states[g.found])} length=${a.size}" +
702 (if (a.isEmpty()) "" else ": " + a.joinToString(" ")),
703 )
704 } else if (g.aborted) println("search limit $limit exceeded")
705 else println("unreachable (markings explored: ${g.states.size})")
706 }
707 "INCIDENCE" -> {
708 val rows =
709 listOf(listOf("") + transitions) +
710 places.indices.map { p ->
711 listOf(places[p]) +
712 transitions.indices.map { (post[p][it] - pre[p][it]).toString() }
713 }
714 val width = rows[0].indices.map { i -> rows.maxOf { it[i].length } }
715 for (r in rows) println(
716 r.indices
717 .joinToString(" ") { i ->
718 if (i == 0) r[i].padEnd(width[i]) else r[i].padStart(width[i])
719 }
720 .trimEnd(),
721 )
722 }
723 "PINV",
724 "TINV" -> invariants(cmd == "TINV")
725 "SIPHONS",
726 "TRAPS" -> {
727 if (places.size > 12) bad("more than 12 places")
728 val ss = sets(cmd == "TRAPS")
729 println("minimal " + (if (cmd == "TRAPS") "traps" else "siphons") + ": ${ss.size}")
730 for (s in ss) println(" ${setText(s)} tokens=${bits(s).sumOf { current[it] }}")
731 }
732 "CLASSIFY" -> classify()
733 "WORKFLOW" -> workflow(limit)
734 "SHOW" -> {
735 for (p in places.indices) println(
736 "place ${places[p]} tokens=${initial[p]} capacity=" +
737 (if (capacity[p] == 0L) "none" else capacity[p].toString()),
738 )
739 for (t in transitions.indices) {
740 fun list(mat: Array<LongArray>): String {
741 val a =
742 places.indices
743 .filter { mat[it][t] > 0 }
744 .joinToString(",") { "${places[it]}:${mat[it][t]}" }
745 return a.ifEmpty { "-" }
746 }
747 println(
748 "transition ${transitions[t]} priority=${priority[t]} pre=${list(pre)} post=${list(post)} inhibit=${list(inhibitor)}",
749 )
750 }
751 }
752 }
753 }
754}
