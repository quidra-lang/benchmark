60 static List<long[]> flows(long[][] a) {
61 int n = a.length, m = a[0].length;
62 List<long[]> rows = new ArrayList<>();
63 for (int i = 0; i < n; ++i) {
64 long[] r = Arrays.copyOf(a[i], m + n);
65 r[m + i] = 1;
66 rows.add(r);
67 }
68 for (int j = 0; j < m; ++j) {
69 Map<String,long[]> unique = new LinkedHashMap<>();
70 for (long[] r : rows) if (r[j] == 0) unique.put(Arrays.toString(r), r);
71 for (long[] r : rows) if (r[j] > 0) for (long[] s : rows) if (s[j] < 0) {
72 long[] v = new long[m + n];
73 long d = 0;
74 for (int k = 0; k < v.length; ++k) {
75 v[k] = -s[j] * r[k] + r[j] * s[k];
76 d = gcd(d, v[k]);
77 }
78 for (int k = 0; k < v.length; ++k) v[k] /= d;
79 unique.put(Arrays.toString(v), v);
80 }
81 List<long[]> candidates = new ArrayList<>(unique.values());
82 int[] supports = new int[candidates.size()];
83 for (int i = 0; i < candidates.size(); ++i) for (int k = 0; k < n; ++k) if (candidates.get(i)[m + k] > 0) supports[i] |= 1 << k;
84 rows = new ArrayList<>();
85 for (int i = 0; i < candidates.size(); ++i) {
86 boolean minimal = true;
87 for (int s : supports) if (s != supports[i] && (s & supports[i]) == s) {
88 minimal = false;
89 break;
90 }
91 if (minimal) rows.add(candidates.get(i));
92 }
93 }
94 List<long[]> result = new ArrayList<>();
95 for (long[] r : rows) result.add(Arrays.copyOfRange(r, m, m + n));
96 result.sort((x,y) -> Arrays.compare(y,x));
97 return result;

131 void define(String[] w) {
132 String cmd = w[0];
133 int n = w.length - 1;
134 if (!Set.of("place","transition","arc","inhibit").contains(cmd)) bad("unknown definition " + cmd);
135 if (!(cmd.equals("place") ? n == 2 || n == 3 : cmd.equals("transition") ? n == 1 || n == 2 : n == 3)) bad("wrong number of arguments");
136 String a = w[1];
137 if (cmd.equals("place") || cmd.equals("transition")) {
138 if (p(a) >= 0 || t(a) >= 0) bad("duplicate name " + a);
139 if (cmd.equals("place")) {
140 if (n == 3 && Long.parseLong(w[2]) > Long.parseLong(w[3])) bad("tokens exceed capacity of " + a);
141 places.add(a);
142 initial.add(Long.parseLong(w[2]));
143 capacities.add(n == 3 ? Long.parseLong(w[3]) : 0L);
144 }
145 else {
146 transitions.add(a);
147 priorities.add(n == 2 ? Long.parseLong(w[2]) : 0L);
148 }

159 } else {

368 List<Integer> sets(boolean trap) {
369 List<Integer> found = new ArrayList<>();
370 for (int mask = 1; mask < 1 << places.size(); ++mask) {
371 boolean valid = true;
372 for (int t = 0; t < transitions.size(); ++t) {
373 boolean in = false, out = false;
374 for (int p = 0; p < places.size(); ++p) if ((mask & 1 << p) != 0) {
375 in |= post[p][t] > 0;
376 out |= pre[p][t] > 0;
377 }
378 if (trap ? out && !in : in && !out) {
379 valid = false;
380 break;
381 }
382 }
383 if (valid) found.add(mask);
384 }
385 List<Integer> result = new ArrayList<>();
386 for (int m : found) {
387 boolean minimal = true;
388 for (int s : found) if (s != m && (s & m) == s) {
389 minimal = false;

393 }

422 choices.set(d,t+1);

428 chosen = t;

433 ++used[chosen];
434 path.add(chosen);
435 stack.add(fire(stack.get(d),chosen));

562 }
563 }
564 boolean[] f = visit(adj,i), b = visit(rev,o);
565 for (int v = 0; v < ns.size(); ++v) if (!f[v] || !b[v]) {
566 out("workflow: no (" + ns.get(v) + " is not on a path from " + places.get(i) + " to " + places.get(o) + ")");
567 return;
568 }
569 out("workflow: source " + places.get(i) + " sink " + places.get(o));
570 long[] start = new long[P], goal = new long[P];
571 start[i] = 1;
572 goal[o] = 1;
573 Graph g = graph(start,limit,null);
574 if (g.aborted) {
575 out("reachability graph limit " + limit + " exceeded");
576 return;
577 }
578 int finish = -1;
579 rev = adjacency(g.states.size());
580 boolean[] used = new boolean[T];
581 boolean proper = true;
582 for (int u = 0; u < g.states.size(); ++u) {
583 long[] m = g.states.get(u);
584 if (Arrays.equals(m,goal)) finish = u;
585 if (m[o] >= 1 && !Arrays.equals(m,goal)) proper = false;
586 for (Edge e : g.edges.get(u)) {
587 rev.get(e.target()).add(u);
588 used[e.transition()] = true;
589 }
590 }
591 boolean option = finish >= 0 && all(visit(rev,finish)), live = all(used);
592 out("option to complete: " + yes(option));
593 out("proper completion: " + yes(proper));
594 out("no dead transitions: " + yes(live));
595 out("sound: " + yes(option && proper && live));
596 }
597 void execute(String[] w) {
598 Map<String,Integer> arity = Map.ofEntries(Map.entry("MARKING",0),Map.entry("RESET",0),Map.entry("SET",2),Map.entry("ENABLED",0),Map.entry("FIRE",-1),Map.entry("UNDO",1),Map.entry("SIMULATE",3),Map.entry("CONFLICTS",0),Map.entry("REACH",1),Map.entry("GRAPH",1),Map.entry("COVER",1),Map.entry("PATH",2),Map.entry("INCIDENCE",0),Map.entry("PINV",0),Map.entry("TINV",0),Map.entry("SIPHONS",0),Map.entry("TRAPS",0),Map.entry("CLASSIFY",0),Map.entry("WORKFLOW",1),Map.entry("SHOW",0));
599 String cmd = w[0];
600 if (!arity.containsKey(cmd)) bad("unknown command " + cmd);
601 int expected = arity.get(cmd);
602 if (expected < 0 ? w.length < 2 : w.length != expected+1) bad("wrong number of arguments");
603 int limit = 0;
604 if (Set.of("REACH","GRAPH","COVER","PATH","WORKFLOW").contains(cmd)) {
605 limit = (int)number(w[1],1,2000);
606 if (limit < 0) bad("invalid limit " + w[1]);
607 }
608 switch (cmd) {
609 case "MARKING", "RESET" -> {
610 if (cmd.equals("RESET")) {
611 current = initialMarking();
612 history.clear();
613 }
614 out("marking " + mark(current));
615 }
616 case "SET" -> {
617 int p = p(w[1]);
618 if (p < 0) bad("unknown place " + w[1]);
619 long v = number(w[2],0,999);
620 if (v < 0) bad("invalid tokens " + w[2]);
621 if (capacities.get(p) > 0 && v > capacities.get(p)) bad("tokens exceed capacity of " + w[1]);
622 current = current.clone();
623 current[p] = v;
624 history.clear();
625 out("marking " + mark(current));
626 }
627 case "ENABLED" -> out("enabled: " + listed(names(enabled(current))));
628 case "FIRE" -> {
629 List<Integer> ts = new ArrayList<>();
630 for (int i = 1; i < w.length; ++i) {
631 int t = t(w[i]);
632 if (t < 0) bad("unknown transition " + w[i]);
633 ts.add(t);
634 }
635 for (int i = 0; i < ts.size(); ++i) {
636 int t = ts.get(i);
637 String reason = reason(current,t);
638 if (!reason.isEmpty()) {
639 out("step " + (i+1) + ": " + transitions.get(t) + " not enabled (" + reason + ") at " + mark(current));
640 return;
641 }
642 commit(t);
643 }
644 out("fired " + ts.size() + " -> " + mark(current));
645 }
646 case "UNDO" -> {
647 int k = (int)number(w[1],1,1000);
648 if (k < 0) bad("invalid count " + w[1]);
649 if (k > history.size()) bad("cannot undo " + k + ": history size " + history.size());
650 current = history.get(history.size()-k);
651 history.subList(history.size()-k,history.size()).clear();
652 out("marking " + mark(current));
653 }
654 case "SIMULATE" -> {
655 int steps = (int)number(w[1],1,1000);
656 long x = number(w[2],1,2147483646);
657 if (steps < 0) bad("invalid steps " + w[1]);
658 if (x < 0) bad("invalid seed " + w[2]);
659 if (!Set.of("PRIORITY","RANDOM").contains(w[3])) bad("invalid mode " + w[3]);
660 int[] counts = new int[transitions.size()];
661 int done = 0;
662 for (; done < steps; ++done) {
663 List<Integer> es = enabled(current);
664 if (es.isEmpty()) {
665 out("dead (steps=" + done + ")");
666 break;
667 }
668 int t = es.getFirst();
669 if (w[3].equals("RANDOM")) {
670 x = x*48271%2147483647;
671 t = es.get((int)(x%es.size()));
672 } else for (int u : es) if (priorities.get(u) > priorities.get(t)) t = u;
673 commit(t);
674 ++counts[t];
675 out("step " + (done+1) + ": " + transitions.get(t) + " -> " + mark(current));
676 }
677 if (done == steps) out("completed (steps=" + steps + ")");
678 List<String> a = new ArrayList<>();
679 for (int t = 0; t < counts.length; ++t) a.add(transitions.get(t) + "=" + counts[t]);
680 out("counts: " + String.join(" ",a));
681 }
682 case "CONFLICTS" -> {
683 List<String> s = new ArrayList<>(), e = new ArrayList<>();
684 for (int u = 0; u < transitions.size(); ++u) for (int v = u+1; v < transitions.size(); ++v) {
685 String pair = transitions.get(u) + "/" + transitions.get(v);
686 boolean conflict = false;
687 for (int p = 0; p < places.size(); ++p) if (pre[p][u] > 0 && pre[p][v] > 0) conflict = true;
688 if (conflict) s.add(pair);
689 if (reason(current,u).isEmpty() && reason(current,v).isEmpty() && (!reason(fire(current,u),v).isEmpty() || !reason(fire(current,v),u).isEmpty())) e.add(pair);
690 }
691 out("structural: " + listed(s));
692 out("effective: " + listed(e));
693 }
694 case "REACH", "GRAPH" -> reach(limit,cmd.equals("GRAPH"));
695 case "COVER" -> cover(limit);
696 case "PATH" -> {
697 String[] entries = w[2].split(",",-1);
698 if (entries.length != places.size()) bad("target needs " + places.size() + " entries");
699 long[][] target = new long[entries.length][2];
700 for (int p = 0; p < entries.length; ++p) {
701 String s = entries[p];
702 if (s.equals("*")) {
703 target[p][1] = -1;
704 continue;
705 }
706 boolean plus = s.endsWith("+");
707 long v = number(plus ? s.substring(0,s.length()-1) : s,0,9999999999L);
708 if (v < 0) bad("invalid target entry " + s);
709 target[p][0] = v;
710 target[p][1] = plus ? -1 : v;
711 }
712 Graph g = graph(current,limit,target);
713 if (g.found >= 0) {
714 List<String> a = names(path(g,g.found));
715 out("reached " + mark(g.states.get(g.found)) + " length=" + a.size() + (a.isEmpty() ? "" : ": " + String.join(" ",a)));
716 } else if (g.aborted) out("search limit " + limit + " exceeded");
717 else out("unreachable (markings explored: " + g.states.size() + ")");
718 }
719 case "INCIDENCE" -> {
720 List<List<String>> rows = new ArrayList<>();
721 List<String> header = new ArrayList<>();
722 header.add("");
723 header.addAll(transitions);
724 rows.add(header);
725 for (int p = 0; p < places.size(); ++p) {
726 List<String> r = new ArrayList<>();
727 r.add(places.get(p));
728 for (int t = 0; t < transitions.size(); ++t) r.add(Long.toString(post[p][t]-pre[p][t]));
729 rows.add(r);
730 }
731 int[] widths = new int[transitions.size()+1];
732 for (List<String> r : rows) for (int i = 0; i < r.size(); ++i) widths[i] = Math.max(widths[i],r.get(i).length());
733 for (List<String> r : rows) {
734 List<String> a = new ArrayList<>();
735 for (int i = 0; i < r.size(); ++i) {
736 String pad = " ".repeat(widths[i]-r.get(i).length());
737 a.add(i == 0 ? r.get(i)+pad : pad+r.get(i));
738 }
739 out(String.join(" ",a).stripTrailing());
740 }
741 }
742 case "PINV", "TINV" -> invariants(cmd.equals("TINV"));
743 case "SIPHONS", "TRAPS" -> {
744 if (places.size() > 12) bad("more than 12 places");
745 List<Integer> ss = sets(cmd.equals("TRAPS"));
746 out("minimal " + (cmd.equals("TRAPS") ? "traps" : "siphons") + ": " + ss.size());
747 for (int s : ss) {
748 long tokens = 0;
749 for (int p : bits(s)) tokens += current[p];
750 out(" " + setText(s) + " tokens=" + tokens);
751 }
752 }
753 case "CLASSIFY" -> classify();
754 case "WORKFLOW" -> workflow(limit);
755 case "SHOW" -> {
756 for (int p = 0; p < places.size(); ++p) out("place " + places.get(p) + " tokens=" + initial.get(p) + " capacity=" + (capacities.get(p) == 0 ? "none" : capacities.get(p)));
757 for (int t = 0; t < transitions.size(); ++t) {
758 List<String> lists = new ArrayList<>();
759 for (long[][] matrix : List.of(pre,post,inhibitor)) {
760 List<String> a = new ArrayList<>();
761 for (int p = 0; p < places.size(); ++p) if (matrix[p][t] > 0) a.add(places.get(p) + ":" + matrix[p][t]);
762 lists.add(a.isEmpty() ? "-" : String.join(",",a));
763 }
764 out("transition " + transitions.get(t) + " priority=" + priorities.get(t) + " pre=" + lists.get(0) + " post=" + lists.get(1) + " inhibit=" + lists.get(2));
765 }
766 }
767 }
768 }
769 }
