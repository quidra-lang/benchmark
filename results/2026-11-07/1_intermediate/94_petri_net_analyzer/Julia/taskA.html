4function number(s, lo, hi)
5 occursin(r"^(0|[1-9][0-9]{0,9})$", s) || return -1
6 v = parse(Int64, s)
7 lo <= v <= hi ? v : -1
8end
9function visit(adj, start)
10 seen = falses(length(adj))
11 q = [start]
12 seen[start] = true
13 cursor = 1
14 while cursor <= length(q)
15 for v in adj[q[cursor]]
16 if !seen[v]
17 seen[v] = true
18 push!(q, v)
19 end
20 end
21 cursor += 1
22 end
23 seen
24end
25function lexless(a, b)
26 for i in eachindex(a)
27 a[i] != b[i] && return a[i] < b[i]
28 end
29 false
30end
31function flows(a)
32 n = length(a)
33 m = length(a[1])
34 rows = [vcat(a[i], [Int64(i == j) for j in 1:n]) for i in 1:n]
35 for j in 1:m
36 allrows = [r for r in rows if r[j] == 0]
37 for r in rows, s in rows
38 r[j] > 0 && s[j] < 0 || continue
39 v = -s[j] .* r .+ r[j] .* s
40 d = foldl(gcd, v; init = 0)
41 push!(allrows, div.(v, d))
42 end
43 unique_rows = collect(values(Dict(Tuple(r) => r for r in allrows)))
44 support = [sum(1 << (i-1) for i in 1:n if r[m+i] > 0; init = 0) for r in unique_rows]
45 rows = [
46 r for (i, r) in enumerate(unique_rows) if
47 !any(s != support[i] && s & support[i] == s for s in support)
48 ]
49 end
50 sort!([r[(m+1):end] for r in rows]; lt = (a, b)->lexless(b, a))
51end

92function define!(n, w)
93 cmd = w[1]
94 count = length(w)-1
95 cmd in ("place", "transition", "arc", "inhibit") || error("unknown definition $cmd")
96 (cmd == "place" ? count in (2, 3) : cmd == "transition" ? count in (1, 2) : count == 3) ||
97 error("wrong number of arguments")
98 a = w[2]
99 if cmd in ("place", "transition")
100 (place(n, a) > 0 || transition(n, a) > 0) && error("duplicate name $a")
101 if cmd == "place"
102 count == 3 &&
103 parse(Int, w[3]) > parse(Int, w[4]) &&
104 error("tokens exceed capacity of $a")
105 push!(n.p, a)
106 push!(n.initial, parse(Int64, w[3]))
107 push!(n.cap, count == 3 ? parse(Int64, w[4]) : 0)
108 else
109 push!(n.t, a)
110 push!(n.priority, count == 2 ? parse(Int64, w[3]) : 0)
111 end
112 elseif cmd == "arc"
113 b = w[3]
114 for x in (a, b)
115 place(n, x) == 0 && transition(n, x) == 0 && error("unknown node $x")
116 end
117 pa = place(n, a)
118 pb = place(n, b)
119 (pa > 0) == (pb > 0) && error("arc must connect a place and a transition")
120 p = pa > 0 ? pa : pb
121 t = transition(n, pa > 0 ? b : a)
122 mat = pa > 0 ? n.pre : n.post
123 mat[p, t] > 0 && error("duplicate arc $a $b")
124 mat[p, t] = parse(Int64, w[4])
125 n.arcs += 1
126 else
127 p = place(n, a)
128 t = transition(n, w[3])
129 p == 0 && error("unknown place $a")
130 t == 0 && error("unknown transition $(w[3])")
131 n.inh[p, t] > 0 && error("duplicate inhibitor $a $(w[3])")
132 n.inh[p, t] = parse(Int64, w[4])
133 n.inhibitors += 1
134 end
135end
136function reason(n, m, t)
137 for p in eachindex(n.p)
138 m[p] >= 0 && m[p] < n.pre[p, t] && return "insufficient tokens in $(n.p[p])"
139 end
140 for p in eachindex(n.p)
141 n.inh[p, t] > 0 && (m[p] < 0 || m[p] >= n.inh[p, t]) && return "inhibited by $(n.p[p])"
142 end
143 for p in eachindex(n.p)
144 n.cap[p] > 0 &&
145 (m[p] < 0 || m[p]-n.pre[p, t]+n.post[p, t] > n.cap[p]) &&
146 return "capacity of $(n.p[p]) exceeded"
147 end
148 ""
149end
150enabled(n, m) = [t for t in eachindex(n.t) if isempty(reason(n, m, t))]
151fire(n, m, t) = [m[p] < 0 ? -1 : m[p]-n.pre[p, t]+n.post[p, t] for p in eachindex(n.p)]
152names(n, ts) = [n.t[t] for t in ts]
153function commit!(n, t)
154 push!(n.history, n.current)
155 n.current = fire(n, n.current, t)
156end
157function graph(n, start, limit, target = nothing)
158 g = Graph([copy(start)], [0], [0], [Tuple{Int,Int}[]], 0, false)
159 index = Dict{Tuple,Int}(Tuple(start)=>1)
160 matches(m) =
161 target !== nothing && all(
162 m[p] >= target[p][1] && (target[p][2] < 0 || m[p] == target[p][2]) for p in eachindex(m)
163 )
164 if matches(start)
165 g.found = 1
166 return g
167 end
168 q = 1
169 while q <= length(g.states)
170 for t in enabled(n, g.states[q])
171 m = fire(n, g.states[q], t)
172 k = get(index, Tuple(m), 0)
173 if k == 0
174 if length(g.states) == limit
175 g.aborted = true
176 return g
177 end
178 k = length(g.states)+1
179 index[Tuple(m)] = k
180 push!(g.states, m)
181 push!(g.parent, q)
182 push!(g.via, t)
183 push!(g.edges, Tuple{Int,Int}[])
184 if matches(m)
185 g.found = k
186 return g
187 end
188 end
189 push!(g.edges[q], (t, k))
190 end
191 q += 1
192 end
193 g
194end
195function path(g, k)
196 result = Int[]
197 while g.parent[k] > 0
198 push!(result, g.via[k])
199 k = g.parent[k]
200 end
201 reverse(result)
202end
203function reach(n, limit, display)
204 g = graph(n, n.current, limit)
205 if g.aborted
206 println("reachability graph limit $limit exceeded")
207 return
208 end
209 N = length(g.states)
210 if display
211 for i in 1:N
212 a = ["$(n.t[t])->M$(k-1)" for (t, k) in g.edges[i]]
213 println("M$(i-1) $(mark(g.states[i])): ", isempty(a) ? "dead" : join(a, " "))
214 end
215 return
216 end
217 adj = [[k for (_, k) in e] for e in g.edges]
218 closure = [visit(adj, i) for i in 1:N]
219 comp = zeros(Int, N)
220 count = 0
221 for i in 1:N
222 comp[i] > 0 && continue
223 count += 1
224 for j in i:N
225 closure[i][j] && closure[j][i] && (comp[j] = count)
226 end
227 end
228 terminal = trues(count)
229 home = trues(N)
230 dead = Int[]
231 maximum = zeros(Int64, length(n.p))
232 edges = 0
233 persistent = true
234 for i in 1:N
235 isempty(g.edges[i]) && push!(dead, i)
236 edges += length(g.edges[i])
237 maximum = max.(maximum, g.states[i])
238 home .&= closure[i]
239 for (t, k) in g.edges[i]
240 comp[i] != comp[k] && (terminal[comp[i]] = false)
241 for (u, _) in g.edges[i]
242 t != u && !isempty(reason(n, g.states[k], u)) && (persistent = false)
243 end
244 end
245 end
246 println(
247 "reachable markings: $N\nedges: $edges\ncomponents: $count (terminal $(sum(terminal)))\ndead markings: $(length(dead))",
248 )
249 for i in dead
250 a = names(n, path(g, i))
251 println(" M$(i-1) $(mark(g.states[i])) path ", isempty(a) ? "empty" : join(a, " "))
252 end
253 bound = Base.maximum(maximum)
254 println(
255 "max tokens: $(mark(maximum))\nbound: $bound\nsafe: $(yes(bound<=1))\nhome markings: $(sum(home))\nreversible: $(yes(home[1]))\npersistent: $(yes(persistent))",
256 )
257 levels = String[]
258 for t in eachindex(n.t)
259 present = falses(count)
260 used = cycle = false
261 for i in 1:N, (u, k) in g.edges[i]
262 u == t || continue
263 used = true
264 present[comp[i]] = true
265 comp[i] == comp[k] && (cycle = true)
266 end
267 level =
268 all(!terminal[c] || present[c] for c in 1:count) ? "L4" :
269 cycle ? "L3" : used ? "L1" : "L0"
270 push!(levels, n.t[t]*"="*level)
271 end
272 println("liveness: ", join(levels, " "))
273end
274function cover(n, limit)
275 states = [copy(n.current)]
276 parent = [0]
277 via = [0]
278 old = [false]
279 dead = [false]
280 q = 1
281 rank(v) = v < 0 ? typemax(Int64) : v
282 while q <= length(states)
283 if !old[q]
284 es = enabled(n, states[q])
285 dead[q] = isempty(es)
286 ancestors = Int[]
287 a = q
288 while a > 0
289 push!(ancestors, a)
290 a = parent[a]
291 end
292 for t in es
293 m = fire(n, states[q], t)
294 for a in reverse(ancestors)
295 if all(rank(m[p]) >= rank(states[a][p]) for p in eachindex(m)) &&
296 any(rank(m[p]) > rank(states[a][p]) for p in eachindex(m))
297 for p in eachindex(m)
298 n.cap[p] == 0 && rank(m[p]) > rank(states[a][p]) && (m[p] = -1)
299 end
300 end
301 end
302 if length(states) == limit
303 println("coverability tree limit $limit exceeded")
304 return
305 end
306 push!(old, m in states)
307 push!(states, m)
308 push!(parent, q)
309 push!(via, t)
310 push!(dead, false)
311 end
312 end
313 q += 1
314 end
315 println("coverability tree nodes: $(length(states))")
316 maximum = zeros(Int64, length(n.p))
317 for i in eachindex(states)
318 prefix = i == 1 ? "root" : "<- $(parent[i]-1) $(n.t[via[i]])"
319 println(" $(i-1) $(mark(states[i])) $prefix", old[i] ? " old" : "", dead[i] ? " dead" : "")
320 for p in eachindex(n.p)
321 rank(states[i][p]) > rank(maximum[p]) && (maximum[p] = states[i][p])
322 end
323 end
324 println(
325 "bounds: $(mark(maximum))\nunbounded: ",
326 listed([n.p[p] for p in eachindex(n.p) if maximum[p] < 0]),
327 )
328end
329function sets(n, trap)
330 found = Int[]
331 for mask in 1:((1<<length(n.p))-1)
332 valid = true
333 for t in eachindex(n.t)
334 input = any(mask & (1<<(p-1)) != 0 && n.post[p, t] > 0 for p in eachindex(n.p))
335 output = any(mask & (1<<(p-1)) != 0 && n.pre[p, t] > 0 for p in eachindex(n.p))
336 if trap ? output && !input : input && !output
337 valid = false
338 break
339 end
340 end
341 valid && push!(found, mask)
342 end
343 result = [m for m in found if !any(s != m && s & m == s for s in found)]
344 bits(m) = [p for p in eachindex(n.p) if m & (1<<(p-1)) != 0]
345 sort!(
346 result;
347 lt = (a, b)->count_ones(a) != count_ones(b) ? count_ones(a)<count_ones(b) :
348 lexless(bits(a), bits(b)),
349 )
350end
351settext(n, mask) = "{"*join([n.p[p] for p in eachindex(n.p) if mask & (1<<(p-1)) != 0], ",")*"}"
352function realize(n, counts)
353 used = zeros(Int64, length(n.t))
354 stack = [n.current]
355 choices = [1]
356 result = Int[]
357 seen = Set{Tuple}()
358 total = sum(counts)
359 while !isempty(stack)
360 d = length(result)+1
361 length(result) == total && return result
362 chosen = 0
363 for t in choices[d]:length(n.t)
364 choices[d] = t+1
365 (used[t] >= counts[t] || !isempty(reason(n, stack[d], t))) && continue
366 used[t] += 1
367 key = Tuple(used)
368 used[t] -= 1
369 key in seen && continue
370 push!(seen, key)
371 chosen = t
372 break
373 end
374 if chosen > 0
375 used[chosen] += 1
376 push!(result, chosen)
377 push!(stack, fire(n, stack[d], chosen))
378 push!(choices, 1)
379 else
380 pop!(stack)
381 pop!(choices)
382 !isempty(result) && (used[pop!(result)] -= 1)
383 end
384 end
385 nothing
386end
387function invariants(n, transpose)
388 a =
389 transpose ? [[n.post[p, t]-n.pre[p, t] for p in eachindex(n.p)] for t in eachindex(n.t)] :
390 [[n.post[p, t]-n.pre[p, t] for t in eachindex(n.t)] for p in eachindex(n.p)]
391 f = flows(a)
392 ns = transpose ? n.t : n.p
393 covered = falses(length(ns))
394 bounds = fill(Int64(-1), length(ns))
395 println(transpose ? "T" : "P", "-semiflows: ", length(f))
396 for c in f
397 terms = join([(c[i] == 1 ? "" : "$(c[i])*")*ns[i] for i in eachindex(c) if c[i] > 0], " + ")
398 covered .|= c .> 0
399 if transpose
400 sequence = realize(n, c)
401 println(
402 " $terms: ",
403 sequence === nothing ? "not realizable" :
404 "realizable "*join(names(n, sequence), " "),
405 )
406 else
407 value = sum(c .* n.current)
408 println(" $terms = $value")
409 for p in eachindex(c)
410 c[p] > 0 && (
411 bounds[p] =
412 bounds[p] < 0 ? div(value, c[p]) : min(bounds[p], div(value, c[p]))
413 )
414 end
415 end
416 end
417 !transpose && println("bounds: (", join([b < 0 ? "-" : string(b) for b in bounds], ","), ")")
418 println(transpose ? "consistent: " : "conservative: ", yes(all(covered)))
419end
420function classify(n)
421 P = length(n.p)
422 T = length(n.t)
423 inputs = [[p for p in 1:P if n.pre[p, t] > 0] for t in 1:T]
424 outputs = [[t for t in 1:T if n.pre[p, t] > 0] for p in 1:P]
425 ordinary = all(n.pre[p, t] <= 1 && n.post[p, t] <= 1 for p in 1:P, t in 1:T)
426 pure = all(n.pre[p, t] == 0 || n.post[p, t] == 0 for p in 1:P, t in 1:T)
427 sm = ordinary && all(length(inputs[t]) == 1 && count(>(0), n.post[1:P, t]) == 1 for t in 1:T)
428 mg = ordinary && all(length(outputs[p]) == 1 && count(>(0), n.post[p, 1:T]) == 1 for p in 1:P)
429 fc =
430 ordinary &&
431 all(length(outputs[p]) == 1 || length(inputs[t]) == 1 for p in 1:P for t in outputs[p])
432 efc = ordinary && all(isempty(intersect(a, b)) || a == b for a in outputs, b in outputs)
433 for (label, v) in [
434 ("ordinary", ordinary),
435 ("pure", pure),
436 ("state machine", sm),
437 ("marked graph", mg),
438 ("free choice", fc),
439 ("extended free choice", efc),
440 ]
441 println(label, ": ", yes(v))
442 end
443 verdict = !fc ? "not applicable" : P > 12 ? "too many places" : "holds"
444 if fc && P <= 12
445 traps = sets(n, true)
446 for s in sets(n, false)
447 if !any(
448 q & s == q && any(n.current[p] > 0 && q & (1<<(p-1)) != 0 for p in 1:P) for
449 q in traps
450 )
451 verdict = "fails at "*settext(n, s)
452 break
453 end
454 end
455 end
456 println("commoner: ", verdict)
457end
458function workflow(n, limit)
459 P = length(n.p)
460 T = length(n.t)
461 sources = [p for p in 1:P if all(n.post[p, t] == 0 for t in 1:T)]
462 sinks = [p for p in 1:P if all(n.pre[p, t] == 0 for t in 1:T)]
463 if length(sources) != 1
464 println("workflow: no ($(length(sources)) source places)")
465 return
466 end
467 if length(sinks) != 1
468 println("workflow: no ($(length(sinks)) sink places)")
469 return
470 end
471 i = sources[1]
472 o = sinks[1]
473 ns = vcat(n.p, n.t)
474 adj = [Int[] for _ in ns]
475 rev = [Int[] for _ in ns]
476 for p in 1:P, t in 1:T
477 if n.pre[p, t] > 0
478 push!(adj[p], P+t)
479 push!(rev[P+t], p)
480 end
481 if n.post[p, t] > 0
482 push!(adj[P+t], p)
483 push!(rev[p], P+t)
484 end
485 end
486 f = visit(adj, i)
487 b = visit(rev, o)
488 for v in eachindex(ns)
489 if !f[v] || !b[v]
490 println("workflow: no ($(ns[v]) is not on a path from $(n.p[i]) to $(n.p[o]))")
491 return
492 end
493 end
494 println("workflow: source $(n.p[i]) sink $(n.p[o])")
495 start = zeros(Int64, P)
496 goal = zeros(Int64, P)
497 start[i] = goal[o] = 1
498 g = graph(n, start, limit)
499 if g.aborted
500 println("reachability graph limit $limit exceeded")
501 return
502 end
503 finish = findfirst(==(goal), g.states)
504 rev = [Int[] for _ in g.states]
505 used = falses(T)
506 for u in eachindex(g.states), (t, v) in g.edges[u]
507 push!(rev[v], u)
508 used[t] = true
509 end
510 option = finish !== nothing && all(visit(rev, finish))
511 proper = all(m[o] < 1 || m == goal for m in g.states)
512 live = all(used)
513 println(
514 "option to complete: $(yes(option))\nproper completion: $(yes(proper))\nno dead transitions: $(yes(live))\nsound: $(yes(option&&proper&&live))",
515 )
516end
517const ARITY = Dict(
518 "MARKING"=>0,
519 "RESET"=>0,
520 "SET"=>2,
521 "ENABLED"=>0,
522 "FIRE"=>-1,
523 "UNDO"=>1,
524 "SIMULATE"=>3,
525 "CONFLICTS"=>0,
526 "REACH"=>1,
527 "GRAPH"=>1,
528 "COVER"=>1,
529 "PATH"=>2,
530 "INCIDENCE"=>0,
531 "PINV"=>0,

539function execute!(n, w)
540 cmd = w[1]
541 haskey(ARITY, cmd) || error("unknown command $cmd")
542 (ARITY[cmd] < 0 ? length(w)>1 : length(w)==ARITY[cmd]+1) || error("wrong number of arguments")
543 limit = 0
544 if cmd in ("REACH", "GRAPH", "COVER", "PATH", "WORKFLOW")
545 limit = number(w[2], 1, 2000)
546 limit < 0 && error("invalid limit $(w[2])")
547 end
548 if cmd in ("MARKING", "RESET")
549 if cmd == "RESET"
550 n.current = copy(n.initial)
551 empty!(n.history)
552 end
553 println("marking ", mark(n.current))
554 elseif cmd == "SET"
555 p = place(n, w[2])
556 p == 0 && error("unknown place $(w[2])")
557 v = number(w[3], 0, 999)
558 v < 0 && error("invalid tokens $(w[3])")
559 n.cap[p] > 0 && v > n.cap[p] && error("tokens exceed capacity of $(w[2])")
560 n.current = copy(n.current)
561 n.current[p] = v
562 empty!(n.history)
563 println("marking ", mark(n.current))
564 elseif cmd == "ENABLED"
565 println("enabled: ", listed(names(n, enabled(n, n.current))))
566 elseif cmd == "FIRE"
567 ts = Int[]
568 for a in w[2:end]
569 t = transition(n, a)
570 t == 0 && error("unknown transition $a")
571 push!(ts, t)
572 end
573 for (i, t) in enumerate(ts)
574 r = reason(n, n.current, t)
575 if !isempty(r)
576 println("step $i: $(n.t[t]) not enabled ($r) at $(mark(n.current))")
577 return
578 end
579 commit!(n, t)
580 end
581 println("fired $(length(ts)) -> $(mark(n.current))")
582 elseif cmd == "UNDO"
583 k = number(w[2], 1, 1000)
584 k < 0 && error("invalid count $(w[2])")
585 h = length(n.history)
586 k > h && error("cannot undo $k: history size $h")
587 n.current = n.history[h-k+1]
588 resize!(n.history, h-k)
589 println("marking ", mark(n.current))
590 elseif cmd == "SIMULATE"
591 steps = number(w[2], 1, 1000)
592 x = number(w[3], 1, 2147483646)
593 steps < 0 && error("invalid steps $(w[2])")
594 x < 0 && error("invalid seed $(w[3])")
595 w[4] in ("PRIORITY", "RANDOM") || error("invalid mode $(w[4])")
596 counts = zeros(Int, length(n.t))
597 done = 0
598 while done < steps
599 es = enabled(n, n.current)
600 if isempty(es)
601 println("dead (steps=$done)")
602 break
603 end
604 t = es[1]
605 if w[4] == "RANDOM"
606 x = x*48271%2147483647
607 t = es[x%length(es)+1]
608 else
609 for u in es
610 n.priority[u] > n.priority[t] && (t = u)
611 end
612 end
613 commit!(n, t)
614 counts[t] += 1
615 done += 1
616 println("step $done: $(n.t[t]) -> $(mark(n.current))")
617 end
618 done == steps && println("completed (steps=$steps)")
619 println("counts: ", join(["$(n.t[t])=$(counts[t])" for t in eachindex(n.t)], " "))
620 elseif cmd == "CONFLICTS"
621 structural = String[]
622 effective = String[]
623 for u in eachindex(n.t), v in (u+1):length(n.t)
624 pair = n.t[u]*"/"*n.t[v]
625 any(n.pre[p, u] > 0 && n.pre[p, v] > 0 for p in eachindex(n.p)) &&
626 push!(structural, pair)
627 isempty(reason(n, n.current, u)) &&
628 isempty(reason(n, n.current, v)) &&
629 (
630 !isempty(reason(n, fire(n, n.current, u), v)) ||
631 !isempty(reason(n, fire(n, n.current, v), u))
632 ) &&
633 push!(effective, pair)
634 end
635 println("structural: ", listed(structural), "\neffective: ", listed(effective))
636 elseif cmd in ("REACH", "GRAPH")
637 reach(n, limit, cmd=="GRAPH")
638 elseif cmd == "COVER"
639 cover(n, limit)
640 elseif cmd == "PATH"
641 entries = split(w[3], ',')
642 length(entries) == length(n.p) || error("target needs $(length(n.p)) entries")
643 target = Tuple{Int64,Int64}[]
644 for s in entries
645 if s == "*"
646 push!(target, (0, -1))
647 continue
648 end
649 plus = endswith(s, "+")
650 v = number(plus ? s[1:(end-1)] : s, 0, 9999999999)
651 v < 0 && error("invalid target entry $s")
652 push!(target, (v, plus ? -1 : v))
653 end
654 g = graph(n, n.current, limit, target)
655 if g.found > 0
656 sequence = names(n, path(g, g.found))
657 println(
658 "reached $(mark(g.states[g.found])) length=$(length(sequence))",
659 isempty(sequence) ? "" : ": "*join(sequence, " "),
660 )
661 elseif g.aborted
662 println("search limit $limit exceeded")
663 else
664 println("unreachable (markings explored: $(length(g.states)))")
665 end
666 elseif cmd == "INCIDENCE"
667 rows = [vcat([""], n.t)]
668 append!(
669 rows,
670 [
671 [n.p[p]; [string(n.post[p, t]-n.pre[p, t]) for t in eachindex(n.t)]] for
672 p in eachindex(n.p)
673 ],
674 )
675 width = [maximum(length(r[i]) for r in rows) for i in eachindex(rows[1])]
676 for r in rows
677 println(
678 rstrip(
679 join(
680 [
681 i == 1 ? rpad(r[i], width[i]) : lpad(r[i], width[i]) for
682 i in eachindex(r)
683 ],
684 " ",
685 ),
686 ),
687 )
688 end
689 elseif cmd in ("PINV", "TINV")
690 invariants(n, cmd=="TINV")
691 elseif cmd in ("SIPHONS", "TRAPS")
692 length(n.p) > 12 && error("more than 12 places")
693 ss = sets(n, cmd=="TRAPS")
694 println("minimal ", cmd=="TRAPS" ? "traps" : "siphons", ": ", length(ss))
695 for s in ss
696 println(
697 " $(settext(n,s)) tokens=",
698 sum(n.current[p] for p in eachindex(n.p) if s & (1<<(p-1)) != 0; init = 0),
699 )
700 end
701 elseif cmd == "CLASSIFY"
702 classify(n)
703 elseif cmd == "WORKFLOW"
704 workflow(n, limit)
705 elseif cmd == "SHOW"
706 for p in eachindex(n.p)
707 println(
708 "place $(n.p[p]) tokens=$(n.initial[p]) capacity=",
709 n.cap[p] == 0 ? "none" : n.cap[p],
710 )
711 end
712 for t in eachindex(n.t)
713 list(mat) = (
714 a = ["$(n.p[p]):$(mat[p,t])" for p in eachindex(n.p) if mat[p, t] > 0];
715 isempty(a) ? "-" : join(a, ",")
716 )
717 println(
718 "transition $(n.t[t]) priority=$(n.priority[t]) pre=$(list(n.pre)) post=$(list(n.post)) inhibit=$(list(n.inh))",
719 )
720 end
721 end
722end
723function main()
724 blocks = parse(Int, readline())
725 for _ in 1:blocks
726 n = Net(split(readline())[2])
727 while true
728 line = readline()
729 line == "END NET" && break
730 try
731 define!(n, split(line))
732 catch e
733 e isa ErrorException || rethrow()
734 println("error: ", e.msg)
735 end
736 end
737 n.current = copy(n.initial)
738 println(
739 "net $(n.name): places=$(length(n.p)) transitions=$(length(n.t)) arcs=$(n.arcs) inhibitors=$(n.inhibitors)",
740 )
741 for _ in 1:parse(Int, readline())
742 try
743 execute!(n, split(readline()))
744 catch e
745 e isa ErrorException || rethrow()
746 println("error: ", e.msg)
747 end
748 end
749 end
750end
