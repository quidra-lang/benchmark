static void commit(Net *n, int t) {
    n->history[n->history_size++] = n->current;
    n->current = fire(n, n->current, t);
    case 6: {
        int steps = (int)integer(w[1], 1, 1000);
        int64_t x = integer(w[2], 1, 2147483646);
        if (steps < 0)
            ERROR("invalid steps %s", w[1]);
        if (x < 0)
            ERROR("invalid seed %s", w[2]);
        if (strcmp(w[3], "PRIORITY") && strcmp(w[3], "RANDOM"))
            ERROR("invalid mode %s", w[3]);
        int counts[16] = {0}, done = 0;
        for (; done < steps; ++done) {
            int es[16], size = 0;
            for (int t = 0; t < n->T; ++t)
                if (enabled(n, n->current, t))
                    es[size++] = t;
            if (!size) {
                printf("dead (steps=%d)\n", done);
                break;
            }
            int t = es[0];
            if (!strcmp(w[3], "RANDOM")) {
                x = x * 48271 % 2147483647;
                t = es[x % size];
            } else
                for (int i = 1; i < size; ++i)
                    if (n->priority[es[i]] > n->priority[t])
                        t = es[i];
            commit(n, t);
            ++counts[t];
            printf("step %d: %s -> ", done + 1, n->t[t]);
            marking(n->current, n->P);
            putchar('\n');
        }
        if (done == steps)
            printf("completed (steps=%d)\n", steps);
        printf("counts: ");
        for (int t = 0; t < n->T; ++t)
            printf("%s%s=%d", t ? " " : "", n->t[t], counts[t]);
        putchar('\n');
