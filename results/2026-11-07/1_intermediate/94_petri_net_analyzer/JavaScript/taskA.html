16 queue = [start];
17 seen[start] = true;

26function flows(matrix) {
27 const n = matrix.length,
28 m = matrix[0].length;
29 let rows = matrix.map((r, i) => r.concat(Array.from({ length: n }, (_, j) => +(i === j))));
30 for (let j = 0; j < m; ++j) {
31 const next = rows.filter((r) => r[j] === 0);
32 for (const r of rows)
33 if (r[j] > 0)
34 for (const s of rows)
35 if (s[j] < 0) {
36 let v = r.map((x, k) => -s[j] * x + r[j] * s[k]);
37 const d = v.reduce((g, x) => gcd(g, x), 0);
38 v = v.map((x) => x / d);
39 next.push(v);
40 }
41 const unique = [...new Map(next.map((r) => [r.join(","), r])).values()];
42 const supports = unique.map((r) =>
43 r.slice(m).reduce((mask, v, i) => mask | (v > 0 ? 1 << i : 0), 0),
44 );
45 rows = unique.filter(
46 (_, i) => !supports.some((s) => s !== supports[i] && (s & supports[i]) === s),
47 );
48 }
49 return rows
50 .map((r) => r.slice(m))
51 .sort((a, b) => {
52 for (let i = 0; i < n; ++i) if (a[i] !== b[i]) return b[i] - a[i];
53 return 0;
54 });

544 execute(w) {
545 const cmd = w[0],

547 arity = {
548 MARKING: 0,
549 RESET: 0,
550 SET: 2,
551 ENABLED: 0,
552 FIRE: -1,
553 UNDO: 1,
554 SIMULATE: 3,
555 CONFLICTS: 0,
556 REACH: 1,

558 COVER: 1,

581 case "RESET":

592 this.current = this.current.slice();

596 break;
597 }

601 case "FIRE": {
602 const ts = args.map((a) => {
603 const t = this.t.indexOf(a);
604 if (t < 0) fail("unknown transition " + a);

606 });
607 for (let i = 0; i < ts.length; ++i) {
608 const reason = this.reason(this.current, ts[i]);
609 if (reason) {
610 out(
611 `step ${i + 1}: ${args[i]} not enabled (${reason}) at ${marking(this.current)}`,
612 );
613 return;
614 }

616 }
617 out(`fired ${ts.length} -> ${marking(this.current)}`);
618 break;
619 }
620 case "UNDO": {
621 const k = integer(args[0], 1, 1000);
622 if (k < 0) fail("invalid count " + args[0]);

624 fail(`cannot undo ${k}: history size ${this.history.length}`);
625 this.current = this.history[this.history.length - k];
626 this.history.length -= k;
627 out("marking " + marking(this.current));
628 break;
629 }
630 case "SIMULATE": {

634 if (x < 0) fail("invalid seed " + args[1]);

659 effective = [];

690 v = integer(plus ? s.slice(0, -1) : s, 0, 9999999999);

700 );

702 else out(`unreachable (markings explored: ${g.states.length})`);

704 }
705 case "INCIDENCE": {
706 const rows = [
707 ["", ...this.t],
708 ...this.p.map((name, p) => [
709 name,
710 ...this.t.map((_, t) => String(this.post[p][t] - this.pre[p][t])),
711 ]),
712 ],
713 width = rows[0].map((_, i) => Math.max(...rows.map((r) => r[i].length)));
714 for (const r of rows)
715 out(
716 r
717 .map((s, i) => (i ? s.padStart(width[i]) : s.padEnd(width[i])))
718 .join(" ")
719 .trimEnd(),
720 );
721 break;
722 }
723 case "PINV":
724 case "TINV":
725 this.invariants(cmd === "TINV");
726 break;
727 case "SIPHONS":
728 case "TRAPS": {

731 out(`minimal ${cmd === "TRAPS" ? "traps" : "siphons"}: ${sets.length}`);
732 for (const s of sets)
733 out(
734 " " +
735 this.setText(s) +
736 " tokens=" +
737 this.current.reduce((sum, v, p) => sum + (s & (1 << p) ? v : 0), 0),
738 );
739 break;
740 }
741 case "CLASSIFY":
742 this.classify();
743 break;
744 case "WORKFLOW":
745 this.workflow(limit);
746 break;
747 case "SHOW":

749 out(
750 `place ${name} tokens=${this.initial[p]} capacity=${this.cap[p] || "none"}`,
751 ),
752 );

754 const list = (mat) =>
755 this.p
756 .map((n, p) => (mat[p][t] ? n + ":" + mat[p][t] : ""))

761 );
