152 seen = {start}

163def strongly_connected_components(successors):
164 visited = [False] * len(successors)
165 finish_order = []
166 for root in range(len(successors)):
167 if visited[root]:
168 continue
169 visited[root] = True
170 stack = [(root, iter(successors[root]))]
171 while stack:
172 node, children = stack[-1]
173 for child in children:
174 if not visited[child]:
175 visited[child] = True
176 stack.append((child, iter(successors[child])))
177 break
178 else:
179 stack.pop()
180 finish_order.append(node)
181
182 predecessors = invert(successors)
183 component = [-1] * len(successors)
184 count = 0
185 for root in reversed(finish_order):
186 if component[root] != -1:
187 continue
188 component[root] = count
189 stack = [root]
190 while stack:
191 node = stack.pop()
192 for parent in predecessors[node]:
193 if component[parent] == -1:
194 component[parent] = count
195 stack.append(parent)
196 count += 1
197 return component, count

200def farkas(matrix):
201 size = len(matrix)
202 width = len(matrix[0])
203 rows = [
204 tuple(row) + tuple(1 if i == k else 0 for i in range(size))
205 for k, row in enumerate(matrix)
206 ]
207 for j in range(width):
208 positives = [row for row in rows if row[j] > 0]
209 negatives = [row for row in rows if row[j] < 0]
210 collected = [row for row in rows if row[j] == 0]
211 for positive in positives:
212 for negative in negatives:
213 a, b = positive[j], -negative[j]
214 combined = [b * x + a * y for x, y in zip(positive, negative)]
215 divisor = math.gcd(*combined)
216 collected.append(tuple(value // divisor for value in combined))
217 unique = list(dict.fromkeys(collected))
218 supports = [
219 frozenset(i for i in range(size) if row[width + i] > 0) for row in unique
220 ]
221 rows = [
222 row
223 for row, support in zip(unique, supports)
224 if not any(other < support for other in supports)
225 ]
226 return [row[width:] for row in rows]

239 def define(self, words):
240 keyword, args = words[0], words[1:]
241 if keyword not in DEFINITION_ARITY:
242 raise InputError(f"unknown definition {keyword}")
243 if len(args) not in DEFINITION_ARITY[keyword]:
244 raise InputError("wrong number of arguments")
245 if keyword == "place":
246 self.add_place(*args)
247 elif keyword == "transition":
248 self.add_transition(*args)
249 elif keyword == "arc":
250 self.add_arc(*args)
251 else:
252 self.add_inhibitor(*args)

254 def is_node(self, name):
255 return name in self.place_index or name in self.transition_index
256
257 def add_place(self, name, tokens, capacity=None):
258 if self.is_node(name):
259 raise InputError(f"duplicate name {name}")
260 tokens = int(tokens)
261 if capacity is not None:
262 capacity = int(capacity)
263 if tokens > capacity:
264 raise InputError(f"tokens exceed capacity of {name}")
265 self.place_index[name] = len(self.places)
266 self.places.append(Place(name, tokens, capacity))

268 def add_transition(self, name, priority="0"):
269 if self.is_node(name):
270 raise InputError(f"duplicate name {name}")
271 self.transition_index[name] = len(self.transitions)
272 self.transitions.append(Transition(name, int(priority)))
273
274 def add_arc(self, source, target, weight):
275 for node in (source, target):
276 if not self.is_node(node):
277 raise InputError(f"unknown node {node}")
278 if (source in self.place_index) == (target in self.place_index):
279 raise InputError("arc must connect a place and a transition")
280 if (source, target) in self.arcs:
281 raise InputError(f"duplicate arc {source} {target}")
282 self.arcs[source, target] = int(weight)

284 def add_inhibitor(self, place, transition, threshold):
285 if place not in self.place_index:
286 raise InputError(f"unknown place {place}")
287 if transition not in self.transition_index:
288 raise InputError(f"unknown transition {transition}")
289 if (place, transition) in self.inhibitors:
290 raise InputError(f"duplicate inhibitor {place} {transition}")
291 self.inhibitors[place, transition] = int(threshold)

293 def finish(self):
294 places = range(len(self.places))
295 transitions = range(len(self.transitions))
296 self.pre = [[0] * len(places) for _ in transitions]
297 self.post = [[0] * len(places) for _ in transitions]
298 self.inhibit = [[0] * len(places) for _ in transitions]
299 for (source, target), weight in self.arcs.items():
300 if source in self.place_index:
301 self.pre[self.transition_index[target]][self.place_index[source]] = weight
302 else:
303 self.post[self.transition_index[source]][self.place_index[target]] = weight
304 for (place, transition), threshold in self.inhibitors.items():
305 self.inhibit[self.transition_index[transition]][self.place_index[place]] = threshold
306
307 self.place_names = [place.name for place in self.places]
308 self.transition_names = [transition.name for transition in self.transitions]
309 self.initial_marking = tuple(place.tokens for place in self.places)
310 self.incidence = [
311 [self.post[t][p] - self.pre[t][p] for t in transitions] for p in places
312 ]
313 self.input_places = [[p for p in places if self.pre[t][p]] for t in transitions]
314 self.output_places = [[p for p in places if self.post[t][p]] for t in transitions]
315 self.input_transitions = [[t for t in transitions if self.post[t][p]] for p in places]
316 self.output_transitions = [[t for t in transitions if self.pre[t][p]] for p in places]

325 def disabled_reason(self, marking, t):
326 pre, post, inhibit = self.pre[t], self.post[t], self.inhibit[t]
327 for p, place in enumerate(self.places):
328 if marking[p] < pre[p]:
329 return f"insufficient tokens in {place.name}"
330 for p, place in enumerate(self.places):
331 if 0 < inhibit[p] <= marking[p]:
332 return f"inhibited by {place.name}"
333 for p, place in enumerate(self.places):
334 if place.capacity is not None and marking[p] - pre[p] + post[p] > place.capacity:
335 return f"capacity of {place.name} exceeded"
336 return None

349 def reachability_graph(self, start, limit, goal=None):
350 graph = ReachabilityGraph(start)
351 if goal is not None and goal(start):
352 graph.found = 0
353 return graph
354 position = 0
355 while position < len(graph.markings):
356 marking = graph.markings[position]
357 edges = []
358 for t in self.enabled(marking):
359 successor = self.fire(marking, t)
360 target = graph.index.get(successor)
361 if target is None:
362 if len(graph.markings) == limit:
363 return None
364 target = graph.add(successor, position, t)
365 if goal is not None and goal(successor):
366 graph.found = target
367 return graph
368 edges.append((t, target))
369 graph.edges.append(edges)
370 position += 1
371 return graph

373 def is_persistent(self, graph):
374 for edges in graph.edges:
375 for t, target in edges:
376 successor = graph.markings[target]
377 if any(u != t and not self.is_enabled(successor, u) for u, _ in edges):
378 return False

381 def coverability_tree(self, start, limit):
382 nodes = [TreeNode(start, None, None, False)]
383 seen = {start}
384 index = 0
385 while index < len(nodes):
386 node = nodes[index]
387 if not node.old:
388 ancestors = []
389 ancestor = index
390 while ancestor is not None:
391 ancestors.append(nodes[ancestor].marking)
392 ancestor = nodes[ancestor].parent
393 ancestors.reverse()
394 enabled = self.enabled(node.marking)
395 node.dead = not enabled
396 for t in enabled:
397 successor = self.accelerate(list(self.fire(node.marking, t)), ancestors)
398 if len(nodes) == limit:
399 return None
400 nodes.append(TreeNode(successor, index, t, successor in seen))
401 seen.add(successor)
402 index += 1
403 return nodes

405 def accelerate(self, marking, ancestors):
406 for previous in ancestors:
407 if strictly_covers(marking, previous):
408 for p, place in enumerate(self.places):
409 if place.capacity is None and marking[p] > previous[p]:
410 marking[p] = OMEGA
411 return tuple(marking)

413 def find_firing_sequence(self, marking, counts):
414 sequence = []
415 if self.extend_sequence(marking, list(counts), sequence, set()):
416 return sequence
417 return None

419 def extend_sequence(self, marking, remaining, sequence, failed):
420 if not any(remaining):
421 return True
422 state = tuple(remaining)
423 if state in failed:
424 return False
425 for t in range(len(remaining)):
426 if remaining[t] > 0 and self.is_enabled(marking, t):
427 remaining[t] -= 1
428 sequence.append(t)
429 if self.extend_sequence(self.fire(marking, t), remaining, sequence, failed):
430 return True
431 sequence.pop()
432 remaining[t] += 1
433 failed.add(state)
434 return False

556 def do_simulate(self, steps_text, seed_text, mode):
557 steps = parse_int(steps_text, 1, 1000)
558 if steps is None:
559 raise InputError(f"invalid steps {steps_text}")
560 seed = parse_int(seed_text, 1, RANDOM_MODULUS - 1)
561 if seed is None:
562 raise InputError(f"invalid seed {seed_text}")
563 if mode not in ("PRIORITY", "RANDOM"):
564 raise InputError(f"invalid mode {mode}")
565
566 net = self.net
567 counts = [0] * len(net.transitions)
568 state = seed
569 for step in range(1, steps + 1):
570 enabled = net.enabled(self.current)
571 if not enabled:
572 print(f"dead (steps={step - 1})")
573 break
574 if mode == "PRIORITY":
575 chosen = max(enabled, key=lambda t: net.transitions[t].priority)
576 else:
577 state = RANDOM_MULTIPLIER * state % RANDOM_MODULUS
578 chosen = enabled[state % len(enabled)]
579 self.fire_transition(chosen)
580 counts[chosen] += 1
581 name = net.transition_names[chosen]
582 print(f"step {step}: {name} -> {format_marking(self.current)}")
583 else:
584 print(f"completed (steps={steps})")
585 firings = (f"{name}={count}" for name, count in zip(net.transition_names, counts))
586 print("counts: " + " ".join(firings))

588 def do_conflicts(self):

605 def do_reach(self, limit):
606 net = self.net
607 graph = net.reachability_graph(self.current, limit)
608 if graph is None:
609 print(f"reachability graph limit {limit} exceeded")
610 return
611 successors = graph.successors()
612 component, component_count = strongly_connected_components(successors)
613 terminal = [True] * component_count
614 for node, targets in enumerate(successors):
615 for target in targets:
616 if component[target] != component[node]:
617 terminal[component[node]] = False
618 terminal_components = {c for c in range(component_count) if terminal[c]}
619 dead = [k for k, targets in enumerate(successors) if not targets]
620
621 print(f"reachable markings: {len(graph.markings)}")
622 print(f"edges: {sum(len(targets) for targets in successors)}")
623 print(f"components: {component_count} (terminal {len(terminal_components)})")
624 print(f"dead markings: {len(dead)}")
625 for k in dead:
626 path = " ".join(net.transition_names[t] for t in graph.path_to(k)) or "empty"
627 print(f" M{k} {format_marking(graph.markings[k])} path {path}")
628 maximum = tuple(max(column) for column in zip(*graph.markings))
629 print(f"max tokens: {format_marking(maximum)}")
630 print(f"bound: {max(maximum)}")
631 print(f"safe: {yes_no(max(maximum) <= 1)}")
632
633 home = []
634 if len(terminal_components) == 1:
635 home = [k for k in range(len(graph.markings)) if component[k] in terminal_components]
636 print(f"home markings: {len(home)}")
637 print(f"reversible: {yes_no(0 in home)}")
638 print(f"persistent: {yes_no(net.is_persistent(graph))}")
639 levels = self.liveness_levels(graph, component, terminal_components)
640 print("liveness: " + " ".join(levels))

642 def liveness_levels(self, graph, component, terminal_components):
643 net = self.net
644 enabled_in = [set() for _ in net.transitions]
645 cyclic = [False] * len(net.transitions)
646 for node, edges in enumerate(graph.edges):
647 for t, target in edges:
648 enabled_in[t].add(component[node])
649 if component[target] == component[node]:
650 cyclic[t] = True
651 levels = []
652 for t, name in enumerate(net.transition_names):
653 if terminal_components <= enabled_in[t]:
654 level = "L4"
655 elif cyclic[t]:
656 level = "L3"
657 elif enabled_in[t]:
658 level = "L1"
659 else:
660 level = "L0"
661 levels.append(f"{name}={level}")
662 return levels

704 def matches(marking):
705 return all(
706 low <= value and (high is None or value <= high)
707 for value, (low, high) in zip(marking, ranges)
708 )
709
710 graph = net.reachability_graph(self.current, limit, matches)
711 if graph is None:
712 print(f"search limit {limit} exceeded")
713 elif graph.found is None:
714 print(f"unreachable (markings explored: {len(graph.markings)})")
715 else:
716 path = graph.path_to(graph.found)
717 line = f"reached {format_marking(graph.markings[graph.found])} length={len(path)}"
718 if path:
719 line += ": " + " ".join(net.transition_names[t] for t in path)
720 print(line)

729 cells = [row[0].ljust(widths[0])]

734 net = self.net
735 semiflows = sorted(farkas(net.incidence), reverse=True)
736 values = [sum(c * m for c, m in zip(y, self.current)) for y in semiflows]

743 bounds.append(str(min(limits)) if limits else "-")

778 def do_classify(self):
779 net = self.net
780 places = range(len(net.places))
781 transitions = range(len(net.transitions))
782 ordinary = all(weight == 1 for weight in net.arcs.values())
783 pure = not any(net.pre[t][p] and net.post[t][p] for t in transitions for p in places)
784 state_machine = ordinary and all(
785 len(net.input_places[t]) == 1 and len(net.output_places[t]) == 1
786 for t in transitions
787 )
788 marked_graph = ordinary and all(
789 len(net.input_transitions[p]) == 1 and len(net.output_transitions[p]) == 1
790 for p in places
791 )
792 free_choice = ordinary and all(
793 len(net.output_transitions[p]) == 1 or len(net.input_places[t]) == 1
794 for p in places
795 for t in net.output_transitions[p]
796 )
797 extended_free_choice = ordinary and all(
798 net.output_transitions[p] == net.output_transitions[q]
799 for t in transitions
800 for p in net.input_places[t]
801 for q in net.input_places[t]
802 )
803 print(f"ordinary: {yes_no(ordinary)}")
804 print(f"pure: {yes_no(pure)}")
805 print(f"state machine: {yes_no(state_machine)}")
806 print(f"marked graph: {yes_no(marked_graph)}")
807 print(f"free choice: {yes_no(free_choice)}")
808 print(f"extended free choice: {yes_no(extended_free_choice)}")
809 print(f"commoner: {self.commoner_verdict(free_choice)}")

811 def commoner_verdict(self, free_choice):
812 net = self.net
813 if not free_choice:
814 return "not applicable"
815 if len(net.places) > MAX_SET_PLACES:
816 return "too many places"
817 for siphon in net.minimal_sets(net.is_siphon):
818 marked_trap = any(
819 net.is_trap(trap) and self.tokens_in(trap) > 0 for trap in submasks(siphon)
820 )
821 if not marked_trap:
822 return f"fails at {net.format_set(siphon)}"
823 return "holds"

825 def do_workflow(self, limit):
826 net = self.net
827 place_count = len(net.places)
828 sources = [p for p in range(place_count) if not net.input_transitions[p]]
829 sinks = [p for p in range(place_count) if not net.output_transitions[p]]
830 if len(sources) != 1:
831 print(f"workflow: no ({len(sources)} source places)")
832 return
833 if len(sinks) != 1:
834 print(f"workflow: no ({len(sinks)} sink places)")
835 return
836 source, sink = sources[0], sinks[0]
837
838 successors = [
839 [place_count + t for t in net.output_transitions[p]] for p in range(place_count)
840 ]
841 successors += net.output_places
842 forward = reachable(source, successors)
843 backward = reachable(sink, invert(successors))
844 names = net.place_names + net.transition_names
845 for vertex, name in enumerate(names):
846 if vertex not in forward or vertex not in backward:
847 route = f"from {names[source]} to {names[sink]}"
848 print(f"workflow: no ({name} is not on a path {route})")
849 return
850 print(f"workflow: source {names[source]} sink {names[sink]}")
851
852 start = tuple(1 if p == source else 0 for p in range(place_count))
853 final = tuple(1 if p == sink else 0 for p in range(place_count))
854 graph = net.reachability_graph(start, limit)
855 if graph is None:
856 print(f"reachability graph limit {limit} exceeded")
857 return
858 completes = False
859 if final in graph.index:
860 predecessors = invert(graph.successors())
861 completes = len(reachable(graph.index[final], predecessors)) == len(graph.markings)
862 proper = all(marking == final for marking in graph.markings if marking[sink] >= 1)
863 fired = {t for edges in graph.edges for t, _ in edges}
864 no_dead = len(fired) == len(net.transitions)
865 print(f"option to complete: {yes_no(completes)}")
866 print(f"proper completion: {yes_no(proper)}")
867 print(f"no dead transitions: {yes_no(no_dead)}")
868 print(f"sound: {yes_no(completes and proper and no_dead)}")
