    nat[] enabled(int[] values)
        return [for trans_index := (nat(0):<#transitions)[] if reason(values, trans_index) == "" then trans_index]
    void .commit(nat trans_index)
        history = history.append(current)
        current = fired(current, trans_index)
        if command_name == "SIMULATE"
            steps := parse_number(words[1], 1, 1000)
            random_value := parse_number(words[2], 1, 2147483646)
            if steps < 0
                return ["error: invalid steps " + words[1]]
            if random_value < 0
                return ["error: invalid seed " + words[2]]
            if words[3] != "PRIORITY" and words[3] != "RANDOM"
                return ["error: invalid mode " + words[3]]
            nat[] counts := [for trans_index := (nat(0):<#transitions)[] then 0]
            string[] output := []
            done := 0
            while done < steps
                nat[] available := enabled(current)
                if #available == 0
                    output = output.append("dead (steps={done})")
                    break
                selected := available[0]
                if words[3] == "RANDOM"
                    random_value = random_value * 48271 % 2147483647
                    selected = available[nat(random_value % int(#available))]
                else
                    for trans_index := available[]
                        if priorities[trans_index] > priorities[selected]
                            selected = trans_index
                .commit(selected)
                counts[selected] += 1
                done += 1
                output = output.append("step {done}: " + transitions[selected] + " -> " + mark(current))
            if done == steps
                output = output.append("completed (steps={steps})")
            return output.append("counts: " + [for trans_index := (nat(0):<#transitions)[] then transitions[trans_index] + "={counts[trans_index]}"].join(" "))
