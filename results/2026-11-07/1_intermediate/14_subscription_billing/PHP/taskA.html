9function cents(string $text): int
10{
11 [$whole, $fraction] = explode('.', $text);

15function money(int $value): string
16{
17 $sign = $value < 0 ? '-' : '';
18 $absolute = abs($value);
19 return $sign . intdiv($absolute, 100) . '.' . str_pad((string) ($absolute % 100), 2, '0', STR_PAD_LEFT);
20}

32function floorDiv(int $numerator, int $denominator): int
33{
34 $quotient = intdiv($numerator, $denominator);
35 return $numerator < 0 && $numerator % $denominator !== 0 ? $quotient - 1 : $quotient;

38final class Plan
39{
40 public int $length;
41 public int $trial = 0;
42 public int $setup = 0;
43 public bool $perSeat = false;
44 public ?string $mode = null;
45 public array $tiers = [];
46
47 public function __construct(
48 public string $name,
49 public int $price,
50 public string $cycle
51 ) {
52 $this->length = PERIOD_LENGTHS[$cycle];
53 }
54
55 public function fee(int $seats): int
56 {
57 return $this->price * ($this->perSeat ? $seats : 1);
58 }
59
60 public function usageCharge(int $units): int
61 {
62 $lower = 0;
63 $total = 0;
64 $count = count($this->tiers);
65 foreach ($this->tiers as $index => [$upper, $unitPrice, $flatFee]) {
66 $last = $index === $count - 1;
67 if ($this->mode === 'VOLUME') {
68 if ($last || $upper === null || $units <= $upper) {
69 return $units * $unitPrice + $flatFee;
70 }
71 } else {
72 if ($units <= $lower) {
73 break;
74 }
75 $end = $last || $upper === null ? $units : min($units, $upper);
76 $total += ($end - $lower) * $unitPrice + $flatFee;
77 if ($last || $upper === null) {
78 break;
79 }
80 $lower = $upper;
81 }
82 }
83 return $total;
84 }
85}

324 private function invoiceTotals(Subscription $subscription, array $items): array
325 {
326 $subtotal = 0;
327 foreach ($items as $item) {
328 $subtotal += $item[1];
329 }
330 $discount = 0;
331 if ($this->couponUsable($subscription, $subtotal)) {
332 $coupon = $subscription->coupon;
333 $discount = $coupon->kind === 'PERCENT'
334 ? floorDiv($subtotal * $coupon->value, 10000)
335 : min($subtotal, $coupon->value);
336 }
337 $tax = floorDiv(
338 ($subtotal - $discount) * ($this->taxRates[$subscription->customer->region] ?? 0),
339 10000
340 );
341 $total = $subtotal - $discount + $tax;
342 $credit = min($subscription->customer->balance, $total);
343 return [$subtotal, $discount, $tax, $total, $credit, $total - $credit];
344 }

346 private function issue(Subscription $subscription, string $kind, array $items): void
347 {
348 $totals = $this->invoiceTotals($subscription, $items);
349 $invoice = new Invoice(
350 count($this->invoices) + 1, $subscription, $this->day, $kind, $items, $totals
351 );
352 if ($this->couponUsable($subscription, $invoice->subtotal) &&
353 $subscription->remaining !== null) {
354 --$subscription->remaining;
355 }
356 $this->invoices[] = $invoice;
357 $subscription->invoices[] = $invoice;
358 $subscription->customer->invoices[] = $invoice;
359 $this->balanceChange(
360 $subscription->customer, -$invoice->credit, 'applied invoice ' . $invoice->number
361 );
362
363 if ($invoice->due === 0) {
364 $result = 'PAID';
365 } elseif ($subscription->state === 'UNPAID') {
366 $result = 'OPEN';
367 } else {
368 $invoice->attempts = 1;
369 if ($subscription->customer->method === 'OK') {
370 $invoice->paid = $invoice->due;
371 $invoice->outstanding = 0;
372 $result = 'PAID';
373 } else {
374 $result = 'FAILED';
375 }
376 }
377 $this->notice(
378 'invoice', $invoice->number, 'sub', $subscription->number,
379 $kind, 'total', money($invoice->total), 'credit', money($invoice->credit),
380 'due', money($invoice->due), $result
381 );
382 if ($result === 'FAILED') {
383 if ($subscription->state === 'ACTIVE') {
384 $this->changeState($subscription, 'PAST_DUE');
385 }
386 if ($subscription->state === 'PAST_DUE') {
387 $invoice->retries = [$this->day + 3, $this->day + 5, $this->day + 7];
388 }
389 }

421 private function retry(Invoice $invoice): void
422 {
423 array_shift($invoice->retries);
424 ++$invoice->attempts;
425 $number = $invoice->attempts - 1;
426 if ($invoice->customer->method === 'OK') {
427 $invoice->paid += $invoice->outstanding;
428 $invoice->outstanding = 0;
429 $this->notice('retry', $number, 'invoice', $invoice->number, 'PAID');
430 $this->paymentComplete($invoice);
431 } else {
432 $this->notice('retry', $number, 'invoice', $invoice->number, 'FAILED');
433 if (!$invoice->retries) {
434 $this->changeState($invoice->subscription, 'UNPAID');
435 $invoice->subscription->cancelOn = $this->day + 14;
436 }
437 }

440 private function advanceSubscription(Subscription $subscription): void
441 {
442 if ($subscription->state === 'PAUSED' || $subscription->state === 'CANCELED') {
443 return;
444 }
445 foreach ($subscription->invoices as $invoice) {
446 if ($invoice->retries && $invoice->retries[0] === $this->day) {
447 $this->retry($invoice);
448 }
449 }
450 if ($subscription->state === 'UNPAID' && $subscription->cancelOn === $this->day) {
451 $this->cancelSubscription($subscription);
452 }
453 if ($subscription->state === 'TRIALING' && $subscription->trialEnd === $this->day) {
454 if ($subscription->cancelEnd) {
455 $this->cancelSubscription($subscription);
456 } else {
457 $this->changeState($subscription, 'ACTIVE');
458 $subscription->start = $this->day;
459 $subscription->end = $this->day + $subscription->plan->length;
460 $this->issue(
461 $subscription, 'first', $this->openingItems($subscription, $this->day)
462 );
463 }
464 }
465 if (in_array($subscription->state, ['ACTIVE', 'PAST_DUE', 'UNPAID'], true) &&
466 $subscription->end === $this->day) {
467 if ($subscription->cancelEnd) {
468 $this->cancelSubscription($subscription);
469 } else {
470 $items = $this->usageItems($subscription, true);
471 if ($subscription->pending !== null) {
472 [$plan, $seats] = $subscription->pending;
473 $this->notice(
474 'sub', $subscription->number, $subscription->plan->name,
475 'seats', $subscription->seats, '->', $plan->name, 'seats', $seats
476 );
477 $subscription->plan = $plan;
478 $subscription->seats = $seats;
479 $subscription->pending = null;
480 }
481 $subscription->start = $this->day;
482 $subscription->end = $this->day + $subscription->plan->length;
483 $items[] = $this->periodItem(
484 $subscription->plan, $subscription->seats, $this->day
485 );
486 $this->issue($subscription, 'renewal', $items);
487 }
488 }

491 private function cmdPlan(array $args): void
492 {
493 [$name, $amount, $cycle] = array_slice($args, 0, 3);
494 if (isset($this->plans[$name])) {
495 throw new CommandError('duplicate plan ' . $name);
496 }
497 $plan = new Plan($name, cents($amount), $cycle);
498 $options = array_slice($args, 3);
499 for ($i = 0; $i < count($options);) {
500 $option = $options[$i];
501 if ($option === 'TRIAL') {
502 $plan->trial = (int) $options[$i + 1];
503 $i += 2;
504 } elseif ($option === 'SETUP') {
505 $plan->setup = cents($options[$i + 1]);
506 $i += 2;
507 } else {
508 if ($option === 'PERSEAT') {
509 $plan->perSeat = true;
510 } else {
511 $plan->mode = $option;
512 }
513 ++$i;
514 }
515 }
516 $this->plans[$name] = $plan;
517 $parts = ['PLAN', $name, $cycle, $amount];
518 if ($plan->trial !== 0) {
519 array_push($parts, 'trial', $plan->trial);
520 }
521 if ($plan->setup !== 0) {
522 array_push($parts, 'setup', money($plan->setup));
523 }
524 if ($plan->perSeat) {
525 $parts[] = 'perseat';
526 }
527 if ($plan->mode !== null) {
528 $parts[] = strtolower($plan->mode);
529 }
530 $this->emit(...$parts);
531 }

633 private function cmdSubscribe(array $args): void
634 {
635 [$name, $planName] = array_slice($args, 0, 2);
636 $customer = $this->customer($name);
637 $plan = $this->plan($planName);
638 $seats = 1;
639 $hasSeats = false;
640 $couponName = null;
641 $options = array_slice($args, 2);
642 for ($i = 0; $i < count($options); $i += 2) {
643 if ($options[$i] === 'SEATS') {
644 $seats = (int) $options[$i + 1];
645 $hasSeats = true;
646 } else {
647 $couponName = $options[$i + 1];
648 }
649 }
650 if ($hasSeats && !$plan->perSeat) {
651 throw new CommandError('not per-seat ' . $planName);
652 }
653 if ($seats === 0) {
654 throw new CommandError('invalid seats');
655 }
656 $coupon = $couponName === null ? null : $this->redeemableCoupon($couponName);
657 $subscription = new Subscription(
658 count($this->subscriptions) + 1, $customer, $plan, $seats, $coupon
659 );
660 if ($coupon !== null) {
661 ++$coupon->redemptions;
662 }
663 $this->subscriptions[] = $subscription;
664 $customer->subscriptions[] = $subscription;
665 $parts = ['SUBSCRIBE', 'sub', $subscription->number, $name, $planName,
666 'seats', $seats];
667 if ($coupon !== null) {
668 array_push($parts, 'coupon', $coupon->name);
669 }
670 $this->emit(...$parts);
671 if ($plan->trial !== 0) {
672 $subscription->trialEnd = $this->day + $plan->trial;
673 $this->changeState($subscription, 'TRIALING');
674 } else {
675 $this->changeState($subscription, 'ACTIVE');
676 $subscription->start = $this->day;
677 $subscription->end = $this->day + $plan->length;
678 $this->issue($subscription, 'first', $this->openingItems($subscription, $this->day));
679 }

682 private function cmdChange(array $args): void
683 {
684 [$text, $kind, $value] = $args;
685 $subscription = $this->subscription($text, ['ACTIVE', 'TRIALING']);
686 $oldPlan = $subscription->plan;
687 $oldSeats = $subscription->seats;
688 if ($kind === 'PLAN') {
689 $plan = $this->plan($value);
690 $seats = $plan->perSeat ? $oldSeats : 1;
691 } else {
692 $plan = $oldPlan;
693 if (!$plan->perSeat) {
694 throw new CommandError('not per-seat ' . $plan->name);
695 }
696 $seats = (int) $value;
697 if ($seats === 0) {
698 throw new CommandError('invalid seats');
699 }
700 }
701 if ($plan === $oldPlan && $seats === $oldSeats) {
702 throw new CommandError('no change ' . $text);
703 }
704 $head = ['CHANGE', 'sub', $subscription->number, $oldPlan->name,
705 'seats', $oldSeats, '->', $plan->name, 'seats', $seats];
706 $subscription->pending = null;
707 $upgrade = $plan->fee($seats) * $oldPlan->length >
708 $oldPlan->fee($oldSeats) * $plan->length;
709 if ($subscription->state === 'TRIALING') {
710 $subscription->plan = $plan;
711 $subscription->seats = $seats;
712 $this->emit(...[...$head, 'now']);
713 } elseif ($upgrade) {
714 $this->emit(...[...$head, 'now']);
715 $days = $subscription->end - $this->day;
716 $items = $this->usageItems($subscription, true);
717 $items[] = [
718 'unused ' . $oldPlan->name . ' seats ' . $oldSeats . ' ' . $days . ' days',
719 -intdiv($oldPlan->fee($oldSeats) * $days, $oldPlan->length)
720 ];
721 $items[] = [
722 'remaining ' . $plan->name . ' seats ' . $seats . ' ' . $days . ' days',
723 intdiv($plan->fee($seats) * $days, $plan->length)
724 ];
725 $subscription->plan = $plan;
726 $subscription->seats = $seats;
727 $this->issue($subscription, 'upgrade', $items);
728 } else {
729 $subscription->pending = [$plan, $seats];
730 $this->emit(...[...$head, 'at', $subscription->end]);
731 }
732 }
