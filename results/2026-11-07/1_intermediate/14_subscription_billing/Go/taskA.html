14func number(text string) int {
15 value, _ := strconv.Atoi(text)
16 return value
17}

19func cents(text string) int {
20 pieces := strings.Split(text, ".")
21 return number(pieces[0])*100 + number(pieces[1])
22}

24func money(value int) string {
25 sign := ""
26 if value < 0 {
27 sign = "-"
28 value = -value
29 }
30 return fmt.Sprintf("%s%d.%02d", sign, value/100, value%100)
31}

33func signedMoney(value int) string {
34 if value > 0 {
35 return "+" + money(value)
36 }
37 return money(value)
38}

40func percentage(basisPoints int) string {
41 return money(basisPoints) + "%"

44func among(value string, permitted ...string) bool {
45 for _, candidate := range permitted {
46 if value == candidate {
47 return true
48 }
49 }
50 return false
51}

85func (plan *Plan) usageCharge(units int) int {
86 lower, total := 0, 0
87 for i, tier := range plan.tiers {
88 last := i == len(plan.tiers)-1
89 if plan.mode == "VOLUME" {
90 if last || tier.upper == nil || units <= *tier.upper {
91 return units*tier.unit + tier.flat
92 }
93 } else {
94 if units <= lower {
95 break
96 }
97 end := units
98 if !last && tier.upper != nil {
99 end = smaller(units, *tier.upper)
100 }
101 total += (end-lower)*tier.unit + tier.flat
102 if last || tier.upper == nil {
103 break
104 }
105 lower = *tier.upper
106 }
107 }
108 return total
109}

281func (service *Service) requireSubscription(text string, permitted ...string) (*Subscription, error) {
282 id := number(text)
283 if id < 1 || id > len(service.subscriptions) {
284 return nil, fmt.Errorf("unknown sub %s", text)
285 }
286 subscription := service.subscriptions[id-1]
287 if len(permitted) > 0 && !among(subscription.state, permitted...) {
288 return nil, fmt.Errorf("invalid state %s %s", text, subscription.state)
289 }
290 return subscription, nil
291}

355func couponUsable(subscription *Subscription, subtotal int) bool {
356 return subtotal > 0 && subscription.coupon != nil &&
357 (subscription.remaining == nil || *subscription.remaining > 0)
358}

360func (service *Service) invoiceTotals(subscription *Subscription, items []Item) Totals {
361 var totals Totals
362 for _, item := range items {
363 totals.subtotal += item.amount
364 }
365 if couponUsable(subscription, totals.subtotal) {
366 coupon := subscription.coupon
367 if coupon.kind == "PERCENT" {
368 totals.discount = totals.subtotal * coupon.value / 10000
369 } else {
370 totals.discount = smaller(totals.subtotal, coupon.value)
371 }
372 }
373 totals.tax = (totals.subtotal - totals.discount) * service.taxes[subscription.customer.region] / 10000
374 totals.total = totals.subtotal - totals.discount + totals.tax
375 totals.credit = smaller(subscription.customer.balance, totals.total)
376 totals.due = totals.total - totals.credit
377 return totals
378}

380func (service *Service) issue(subscription *Subscription, kind string, items []Item) {
381 totals := service.invoiceTotals(subscription, items)
382 invoice := &Invoice{
383 number: len(service.invoices) + 1,
384 subscription: subscription,
385 customer: subscription.customer,
386 day: service.day,
387 kind: kind,
388 items: items,
389 Totals: totals,
390 outstanding: totals.due,
391 }
392 if couponUsable(subscription, invoice.subtotal) && subscription.remaining != nil {
393 *subscription.remaining--
394 }
395 service.invoices = append(service.invoices, invoice)
396 subscription.invoices = append(subscription.invoices, invoice)
397 subscription.customer.invoices = append(subscription.customer.invoices, invoice)
398 service.balanceChange(subscription.customer, -invoice.credit, fmt.Sprintf("applied invoice %d", invoice.number))
399 result := "PAID"
400 if invoice.due == 0 {
401 result = "PAID"
402 } else if subscription.state == "UNPAID" {
403 result = "OPEN"
404 } else {
405 invoice.attempts = 1
406 if subscription.customer.method == "OK" {
407 invoice.paid = invoice.due
408 invoice.outstanding = 0
409 result = "PAID"
410 } else {
411 result = "FAILED"
412 }
413 }
414 service.notice("invoice", invoice.number, "sub", subscription.number, kind,
415 "total", money(invoice.total), "credit", money(invoice.credit),
416 "due", money(invoice.due), result)
417 if result == "FAILED" {
418 if subscription.state == "ACTIVE" {
419 service.changeState(subscription, "PAST_DUE")
420 }
421 if subscription.state == "PAST_DUE" {
422 invoice.retries = []int{service.day + 3, service.day + 5, service.day + 7}
423 }
424 }
425}

427func (service *Service) paymentComplete(invoice *Invoice) {
428 invoice.retries = nil
429 subscription := invoice.subscription
430 if !among(subscription.state, "PAST_DUE", "UNPAID") {
431 return
432 }
433 for _, other := range subscription.invoices {
434 if other.outstanding != 0 {
435 return
436 }
437 }
438 subscription.cancelOn = 0
439 service.changeState(subscription, "ACTIVE")
440}

442func (service *Service) cancelSubscription(subscription *Subscription) {
443 service.changeState(subscription, "CANCELED")
444 subscription.canceledOn = service.day
445 subscription.pending = nil
446 subscription.cancelEnd = false
447 subscription.cancelOn = 0
448 items := usageItems(subscription, true)
449 if len(items) > 0 {
450 service.issue(subscription, "final", items)
451 }
452}

454func (service *Service) retry(invoice *Invoice) {
455 invoice.retries = invoice.retries[1:]
456 invoice.attempts++
457 attempt := invoice.attempts - 1
458 if invoice.customer.method == "OK" {
459 invoice.paid += invoice.outstanding
460 invoice.outstanding = 0
461 service.notice("retry", attempt, "invoice", invoice.number, "PAID")
462 service.paymentComplete(invoice)
463 } else {
464 service.notice("retry", attempt, "invoice", invoice.number, "FAILED")
465 if len(invoice.retries) == 0 {
466 service.changeState(invoice.subscription, "UNPAID")
467 invoice.subscription.cancelOn = service.day + 14
468 }
469 }
470}

472func (service *Service) advanceSubscription(subscription *Subscription) {
473 if among(subscription.state, "PAUSED", "CANCELED") {
474 return
475 }
476 for _, invoice := range subscription.invoices {
477 if len(invoice.retries) > 0 && invoice.retries[0] == service.day {
478 service.retry(invoice)
479 }
480 }
481 if subscription.state == "UNPAID" && subscription.cancelOn == service.day {
482 service.cancelSubscription(subscription)
483 }
484 if subscription.state == "TRIALING" && subscription.trialEnd == service.day {
485 if subscription.cancelEnd {
486 service.cancelSubscription(subscription)
487 } else {
488 service.changeState(subscription, "ACTIVE")
489 subscription.start = service.day
490 subscription.end = service.day + subscription.plan.length
491 service.issue(subscription, "first", openingItems(subscription, service.day))
492 }
493 }
494 if among(subscription.state, "ACTIVE", "PAST_DUE", "UNPAID") &&
495 subscription.end == service.day {
496 if subscription.cancelEnd {
497 service.cancelSubscription(subscription)
498 } else {
499 items := usageItems(subscription, true)
500 if subscription.pending != nil {
501 change := subscription.pending
502 service.notice("sub", subscription.number, subscription.plan.name,
503 "seats", subscription.seats, "->", change.plan.name, "seats", change.seats)
504 subscription.plan = change.plan
505 subscription.seats = change.seats
506 subscription.pending = nil
507 }
508 subscription.start = service.day
509 subscription.end = service.day + subscription.plan.length
510 items = append(items, periodItem(subscription.plan, subscription.seats, service.day))
511 service.issue(subscription, "renewal", items)
512 }
513 }
514}

516func (service *Service) cmdPlan(args []string) error {
517 name, amount, cycle := args[0], args[1], args[2]
518 if service.plans[name] != nil {
519 return fmt.Errorf("duplicate plan %s", name)
520 }
521 plan := &Plan{name: name, price: cents(amount), cycle: cycle, length: periodLengths[cycle]}
522 for i := 3; i < len(args); {
523 switch args[i] {
524 case "TRIAL":
525 plan.trial = number(args[i+1])
526 i += 2
527 case "SETUP":
528 plan.setup = cents(args[i+1])
529 i += 2
530 case "PERSEAT":
531 plan.perSeat = true
532 i++
533 default:
534 plan.mode = args[i]
535 i++
536 }
537 }
538 service.plans[name] = plan
539 parts := []any{"PLAN", name, cycle, amount}
540 if plan.trial > 0 {
541 parts = append(parts, "trial", plan.trial)
542 }
543 if plan.setup > 0 {
544 parts = append(parts, "setup", money(plan.setup))
545 }
546 if plan.perSeat {
547 parts = append(parts, "perseat")
548 }
549 if plan.mode != "" {
550 parts = append(parts, strings.ToLower(plan.mode))
551 }
552 service.emit(parts...)
553 return nil
554}

556func (service *Service) cmdTier(args []string) error {
557 name, limit, unit, flat := args[0], args[1], args[2], args[3]
558 plan, err := service.requirePlan(name)
559 if err != nil {
560 return err
561 }
562 if plan.mode == "" {
563 return fmt.Errorf("not metered %s", name)
564 }
565 var upper *int
566 if limit != "INF" {
567 value := number(limit)
568 upper = &value
569 }
570 if len(plan.tiers) > 0 {
571 previous := plan.tiers[len(plan.tiers)-1].upper
572 if previous == nil || (upper != nil && *upper <= *previous) {
573 return fmt.Errorf("tier order %s", name)
574 }
575 }
576 plan.tiers = append(plan.tiers, Tier{upper: upper, unit: cents(unit), flat: cents(flat)})
577 service.emit("TIER", name, len(plan.tiers), "upto", limit, "unit", unit, "flat", flat)
578 return nil
579}

615 count := number(args[index+1])

617 }
618 numeric := number(value)
619 if kind == "AMOUNT" {

621 }
622 coupon := &Coupon{
623 name: name, kind: kind, value: numeric,
624 applications: applications, maximum: maximum,
625 }
626 service.coupons[name] = coupon
627 shown := money(numeric)
628 if kind == "PERCENT" {
629 shown = percentage(numeric)
630 }
631 parts := []any{"COUPON", name, strings.ToLower(kind), shown, strings.ToLower(application)}
632 if application == "REPEATING" {
633 parts = append(parts, *applications)
634 }
635 if maximum != nil {
636 parts = append(parts, "max", *maximum)
637 }
638 service.emit(parts...)

681func (service *Service) cmdSubscribe(args []string) error {
682 name, planName := args[0], args[1]
683 customer, err := service.requireCustomer(name)
684 if err != nil {
685 return err
686 }
687 plan, err := service.requirePlan(planName)
688 if err != nil {
689 return err
690 }
691 seats, hasSeats, couponName := 1, false, ""
692 for i := 2; i < len(args); i += 2 {
693 if args[i] == "SEATS" {
694 seats = number(args[i+1])
695 hasSeats = true
696 } else {
697 couponName = args[i+1]
698 }
699 }
700 if hasSeats && !plan.perSeat {
701 return fmt.Errorf("not per-seat %s", planName)
702 }
703 if seats == 0 {
704 return fmt.Errorf("invalid seats")
705 }
706 var coupon *Coupon
707 if couponName != "" {
708 coupon, err = service.redeemableCoupon(couponName)
709 if err != nil {
710 return err
711 }
712 }
713 subscription := &Subscription{
714 number: len(service.subscriptions) + 1, customer: customer,
715 plan: plan, seats: seats, state: "NONE", coupon: coupon,
716 }
717 if coupon != nil {
718 coupon.redemptions++
719 if coupon.applications != nil {
720 remaining := *coupon.applications
721 subscription.remaining = &remaining
722 }
723 }
724 service.subscriptions = append(service.subscriptions, subscription)
725 customer.subscriptions = append(customer.subscriptions, subscription)
726 parts := []any{"SUBSCRIBE", "sub", subscription.number, name, planName, "seats", seats}
727 if coupon != nil {
728 parts = append(parts, "coupon", coupon.name)
729 }
730 service.emit(parts...)
731 if plan.trial > 0 {
732 subscription.trialEnd = service.day + plan.trial
733 service.changeState(subscription, "TRIALING")
734 } else {
735 service.changeState(subscription, "ACTIVE")
736 subscription.start = service.day
737 subscription.end = service.day + plan.length
738 service.issue(subscription, "first", openingItems(subscription, service.day))
739 }
740 return nil
741}

743func (service *Service) cmdChange(args []string) error {
744 text, kind, value := args[0], args[1], args[2]
745 subscription, err := service.requireSubscription(text, "ACTIVE", "TRIALING")
746 if err != nil {
747 return err
748 }
749 oldPlan, oldSeats := subscription.plan, subscription.seats
750 var plan *Plan
751 var seats int
752 if kind == "PLAN" {
753 plan, err = service.requirePlan(value)
754 if err != nil {
755 return err
756 }
757 seats = 1
758 if plan.perSeat {
759 seats = oldSeats
760 }
761 } else {
762 plan = oldPlan
763 if !plan.perSeat {
764 return fmt.Errorf("not per-seat %s", plan.name)
765 }
766 seats = number(value)
767 if seats == 0 {
768 return fmt.Errorf("invalid seats")
769 }
770 }
771 if plan == oldPlan && seats == oldSeats {
772 return fmt.Errorf("no change %s", text)
773 }
774 head := []any{"CHANGE", "sub", subscription.number, oldPlan.name, "seats", oldSeats,
775 "->", plan.name, "seats", seats}
776 subscription.pending = nil
777 upgrade := plan.fee(seats)*oldPlan.length > oldPlan.fee(oldSeats)*plan.length
778 if subscription.state == "TRIALING" {
779 subscription.plan = plan
780 subscription.seats = seats
781 service.emit(append(head, "now")...)
782 } else if upgrade {
783 service.emit(append(head, "now")...)
784 days := subscription.end - service.day
785 items := usageItems(subscription, true)
786 items = append(items, Item{
787 description: fmt.Sprintf("unused %s seats %d %d days", oldPlan.name, oldSeats, days),
788 amount: -(oldPlan.fee(oldSeats) * days / oldPlan.length),
789 })
790 items = append(items, Item{
791 description: fmt.Sprintf("remaining %s seats %d %d days", plan.name, seats, days),
792 amount: plan.fee(seats) * days / plan.length,
793 })
794 subscription.plan = plan
795 subscription.seats = seats
796 service.issue(subscription, "upgrade", items)
797 } else {
798 subscription.pending = &PendingChange{plan: plan, seats: seats}
799 service.emit(append(head, "at", subscription.end)...)
800 }
801 return nil
802}

815func (service *Service) cmdResume(args []string) error {
816 subscription, err := service.requireSubscription(args[0], "PAUSED")
817 if err != nil {
818 return err
819 }
820 elapsed := service.day - subscription.pauseStart
821 subscription.end += elapsed
822 subscription.pauseStart = 0

828func (service *Service) cmdCancel(args []string) error {
829 text, when := args[0], args[1]
830 subscription, err := service.requireSubscription(text)
831 if err != nil {
832 return err
833 }
834 if when == "END" {
835 if !among(subscription.state, "ACTIVE", "TRIALING") {
836 return fmt.Errorf("invalid state %s %s", text, subscription.state)
837 }
838 if subscription.cancelEnd {
839 return fmt.Errorf("already canceling %s", text)
840 }
841 subscription.cancelEnd = true
842 day := subscription.end
843 if subscription.state == "TRIALING" {
844 day = subscription.trialEnd
845 }
846 service.emit("CANCEL", "sub", subscription.number, "end", day)
847 } else {
848 if subscription.state == "CANCELED" {
849 return fmt.Errorf("invalid state %s %s", text, subscription.state)
850 }
851 refund := 0
852 if among(subscription.state, "ACTIVE", "PAUSED") {
853 start := service.day
854 if subscription.state == "PAUSED" {
855 start = subscription.pauseStart
856 }
857 refund = subscription.fee() * (subscription.end - start) / subscription.plan.length
858 }
859 service.balanceChange(subscription.customer, refund, fmt.Sprintf("cancel sub %d", subscription.number))
860 service.emit("CANCEL", "sub", subscription.number, "now", "credit", money(refund))
861 service.cancelSubscription(subscription)
862 }
863 return nil
864}

923func (service *Service) cmdRefund(args []string) error {
924 text, amount := args[0], args[1]
925 invoice, err := service.requireInvoice(text)
926 if err != nil {
927 return err
928 }
929 value := cents(amount)
930 if value == 0 {
931 return fmt.Errorf("invalid amount")
932 }
933 maximum := invoice.total - invoice.refunded
934 if value > maximum {
935 return fmt.Errorf("refund exceeds %s %s", text, money(maximum))
936 }
937 applied := smaller(value, invoice.outstanding)
938 invoice.refunded += value
939 invoice.outstanding -= applied
940 service.balanceChange(invoice.customer, value-applied, fmt.Sprintf("refund invoice %d", invoice.number))
941 service.emit("REFUND", "invoice", invoice.number, "amount", amount,
942 "applied", money(applied), "balance", money(invoice.customer.balance))
943 if applied > 0 && invoice.outstanding == 0 {
944 service.paymentComplete(invoice)
945 }
946 return nil
947}

1104func (service *Service) cmdMrr(_ []string) error {
1105 type Aggregate struct{ count, monthly int }
1106 byPlan := make(map[string]*Aggregate)
1107 for _, subscription := range service.subscriptions {
1108 if !among(subscription.state, "ACTIVE", "PAST_DUE") {
1109 continue
1110 }
1111 fee := subscription.fee()
1112 var monthly int
1113 switch subscription.plan.cycle {
1114 case "MONTHLY":
1115 monthly = fee
1116 case "YEARLY":
1117 monthly = fee / 12
1118 default:
1119 monthly = fee * 30 / 7
1120 }
1121 name := subscription.plan.name
1122 if byPlan[name] == nil {
1123 byPlan[name] = &Aggregate{}
1124 }
1125 byPlan[name].count++
1126 byPlan[name].monthly += monthly
1127 }
1128 keys := make([]string, 0, len(byPlan))
1129 for name := range byPlan {
1130 keys = append(keys, name)
1131 }
1132 sort.Strings(keys)
1133 service.emit("MRR")
1134 count, total := 0, 0
1135 for _, name := range keys {
1136 row := byPlan[name]
1137 service.indented(name, "subs", row.count, "mrr", money(row.monthly))
1138 count += row.count
1139 total += row.monthly
1140 }
1141 service.indented("total subs", count, "mrr", money(total))
1142 return nil
1143}

1150func (service *Service) cmdAging(_ []string) error {
1151 byCustomer := make(map[string]*[3]int)
1152 var totals [3]int
1153 for _, invoice := range service.invoices {
1154 if invoice.outstanding == 0 {
1155 continue
1156 }
1157 age := service.day - invoice.day
1158 bucket := 0
1159 if age > 60 {
1160 bucket = 2
1161 } else if age > 30 {
1162 bucket = 1
1163 }
1164 name := invoice.customer.name
1165 if byCustomer[name] == nil {
1166 byCustomer[name] = &[3]int{}
1167 }
1168 byCustomer[name][bucket] += invoice.outstanding
1169 totals[bucket] += invoice.outstanding
1170 }
1171 keys := make([]string, 0, len(byCustomer))
1172 for name := range byCustomer {
1173 keys = append(keys, name)
1174 }
1175 sort.Strings(keys)
1176 service.emit("AGING", "day", service.day)
1177 for _, name := range keys {
1178 service.indented(append([]any{name}, agingParts(*byCustomer[name])...)...)
1179 }
1180 service.indented(append([]any{"total"}, agingParts(totals)...)...)
1181 return nil
