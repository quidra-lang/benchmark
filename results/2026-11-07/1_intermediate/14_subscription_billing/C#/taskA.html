18sealed class Plan
19{
20 public string Name { get; }
21 public long Price { get; }
22 public string Cycle { get; }
23 public int Length { get; }
24 public int Trial { get; set; }
25 public long Setup { get; set; }
26 public bool PerSeat { get; set; }
27 public string? Mode { get; set; }
28 public List<Tier> Tiers { get; } = new();
29
30 public Plan(string name, long price, string cycle)
31 {
32 Name = name;
33 Price = price;
34 Cycle = cycle;
35 Length = cycle switch
36 {
37 "WEEKLY" => 7,
38 "MONTHLY" => 30,
39 _ => 360
40 };
41 }
42
43 public long Fee(int seats) => PerSeat ? Price * seats : Price;
44
45 public long UsageCharge(long units)
46 {
47 long total = 0;
48 long lower = 0;
49 for (int index = 0; index < Tiers.Count; ++index)
50 {
51 var tier = Tiers[index];
52 bool last = index == Tiers.Count - 1;
53 if (Mode == "VOLUME")
54 {
55 if (last || tier.Upper is null || units <= tier.Upper.Value)
56 return units * tier.Unit + tier.Flat;
57 }
58 else
59 {
60 if (units <= lower) break;
61 long end = last || tier.Upper is null ? units : Math.Min(units, tier.Upper.Value);
62 total += (end - lower) * tier.Unit + tier.Flat;
63 if (last || tier.Upper is null) break;
64 lower = tier.Upper.Value;
65 }
66 }
67 return total;
68 }
69}

195 private static long Cents(string amount)
196 {
197 var parts = amount.Split('.');
198 return long.Parse(parts[0], CultureInfo.InvariantCulture) * 100
199 + long.Parse(parts[1], CultureInfo.InvariantCulture);

202 private static string Money(long amount)
203 {
204 long absolute = Math.Abs(amount);
205 return (amount < 0 ? "-" : "") + (absolute / 100).ToString(CultureInfo.InvariantCulture)
206 + "." + (absolute % 100).ToString("D2", CultureInfo.InvariantCulture);
207 }

288 private static List<LineItem> OpeningItems(Subscription subscription, int start)

296 private static List<LineItem> UsageItems(Subscription subscription, bool close = false)
297 {
298 if (subscription.Plan.Mode is null || subscription.Usage == 0)
299 return new List<LineItem>();
300 long units = subscription.Usage;
301 var result = new List<LineItem> {
302 new($"usage {subscription.Plan.Name} {units} units",
303 subscription.Plan.UsageCharge(units))
304 };
305 if (close) subscription.Usage = 0;
306 return result;

309 private static bool CouponUsable(Subscription subscription, long subtotal) =>
310 subtotal > 0 && subscription.Coupon is not null &&
311 (subscription.Remaining is null || subscription.Remaining > 0);
312
313 private Totals InvoiceTotals(Subscription subscription, List<LineItem> items)
314 {
315 long subtotal = items.Sum(item => item.Amount);
316 long discount = 0;
317 if (CouponUsable(subscription, subtotal))
318 {
319 Coupon coupon = subscription.Coupon!;
320 discount = coupon.Kind == "PERCENT"
321 ? subtotal * coupon.Value / 10000
322 : Math.Min(subtotal, coupon.Value);
323 }
324 long tax = (subtotal - discount) * TaxRate(subscription.Customer.Region) / 10000;
325 long total = subtotal - discount + tax;
326 long credit = Math.Min(subscription.Customer.Balance, total);
327 return new Totals(subtotal, discount, tax, total, credit, total - credit);
328 }

330 private void Issue(Subscription subscription, string kind, List<LineItem> items)
331 {
332 Totals totals = InvoiceTotals(subscription, items);
333 var invoice = new Invoice(invoices.Count + 1, subscription, day, kind, items, totals);
334 if (CouponUsable(subscription, invoice.Subtotal) && subscription.Remaining.HasValue)
335 subscription.Remaining = subscription.Remaining.Value - 1;
336 invoices.Add(invoice);
337 subscription.Invoices.Add(invoice);
338 subscription.Customer.Invoices.Add(invoice);
339 BalanceChange(subscription.Customer, -invoice.Credit, "applied invoice " + invoice.Number);
340
341 string result;
342 if (invoice.Due == 0)
343 {
344 result = "PAID";
345 }
346 else if (subscription.State == "UNPAID")
347 {
348 result = "OPEN";
349 }
350 else
351 {
352 invoice.Attempts = 1;
353 if (subscription.Customer.Method == "OK")
354 {
355 invoice.Paid = invoice.Due;
356 invoice.Outstanding = 0;
357 result = "PAID";
358 }
359 else
360 {
361 result = "FAILED";
362 }
363 }
364
365 Notice("invoice", invoice.Number, "sub", subscription.Number, kind,
366 "total", Money(invoice.Total), "credit", Money(invoice.Credit),
367 "due", Money(invoice.Due), result);
368 if (result == "FAILED")
369 {
370 if (subscription.State == "ACTIVE") ChangeState(subscription, "PAST_DUE");
371 if (subscription.State == "PAST_DUE")
372 invoice.Retries.AddRange(new[] { day + 3, day + 5, day + 7 });
373 }
374 }

376 private void PaymentComplete(Invoice invoice)
377 {
378 invoice.Retries.Clear();
379 Subscription subscription = invoice.Subscription;
380 if ((subscription.State == "PAST_DUE" || subscription.State == "UNPAID") &&
381 subscription.Invoices.All(entry => entry.Outstanding == 0))
382 {
383 subscription.CancelOn = null;
384 ChangeState(subscription, "ACTIVE");
385 }
386 }

388 private void CancelSubscription(Subscription subscription)
389 {
390 ChangeState(subscription, "CANCELED");
391 subscription.CanceledOn = day;
392 subscription.Pending = null;
393 subscription.CancelEnd = false;
394 subscription.CancelOn = null;
395 var items = UsageItems(subscription, true);
396 if (items.Count > 0) Issue(subscription, "final", items);
397 }
398
399 private void Retry(Invoice invoice)
400 {
401 invoice.Retries.RemoveAt(0);
402 ++invoice.Attempts;
403 int retryNumber = invoice.Attempts - 1;
404 if (invoice.Customer.Method == "OK")
405 {
406 invoice.Paid += invoice.Outstanding;
407 invoice.Outstanding = 0;
408 Notice("retry", retryNumber, "invoice", invoice.Number, "PAID");
409 PaymentComplete(invoice);
410 }
411 else
412 {
413 Notice("retry", retryNumber, "invoice", invoice.Number, "FAILED");
414 if (invoice.Retries.Count == 0)
415 {
416 ChangeState(invoice.Subscription, "UNPAID");
417 invoice.Subscription.CancelOn = day + 14;
418 }
419 }
420 }
421
422 private void AdvanceSubscription(Subscription subscription)
423 {
424 if (subscription.State is "PAUSED" or "CANCELED") return;
425 foreach (var invoice in subscription.Invoices)
426 {
427 if (invoice.Retries.Count > 0 && invoice.Retries[0] == day)
428 Retry(invoice);
429 }
430 if (subscription.State == "UNPAID" && subscription.CancelOn == day)
431 CancelSubscription(subscription);
432 if (subscription.State == "TRIALING" && subscription.TrialEnd == day)
433 {
434 if (subscription.CancelEnd)
435 {
436 CancelSubscription(subscription);
437 }
438 else
439 {
440 ChangeState(subscription, "ACTIVE");
441 subscription.Start = day;
442 subscription.End = day + subscription.Plan.Length;
443 Issue(subscription, "first", OpeningItems(subscription, day));
444 }
445 }
446 if ((subscription.State is "ACTIVE" or "PAST_DUE" or "UNPAID")
447 && subscription.End == day)
448 {
449 if (subscription.CancelEnd)
450 {
451 CancelSubscription(subscription);
452 }
453 else
454 {
455 var items = UsageItems(subscription, true);
456 if (subscription.Pending is { } pending)
457 {
458 Notice("sub", subscription.Number, subscription.Plan.Name, "seats",
459 subscription.Seats, "->", pending.Plan.Name, "seats", pending.Seats);
460 subscription.Plan = pending.Plan;
461 subscription.Seats = pending.Seats;
462 subscription.Pending = null;
463 }
464 subscription.Start = day;
465 subscription.End = day + subscription.Plan.Length;
466 items.Add(PeriodItem(subscription.Plan, subscription.Seats, day));
467 Issue(subscription, "renewal", items);
468 }
469 }

472 private void CmdPlan(string[] args)
473 {
474 string name = args[0];
475 if (plans.ContainsKey(name)) Fail("duplicate plan " + name);
476 var plan = new Plan(name, Cents(args[1]), args[2]);
477 for (int i = 3; i < args.Length;)
478 {
479 switch (args[i])
480 {
481 case "TRIAL":
482 plan.Trial = Integer(args[i + 1]);
483 i += 2;
484 break;
485 case "SETUP":
486 plan.Setup = Cents(args[i + 1]);
487 i += 2;
488 break;
489 case "PERSEAT":
490 plan.PerSeat = true;
491 ++i;
492 break;
493 default:
494 plan.Mode = args[i];
495 ++i;
496 break;
497 }
498 }
499 plans.Add(name, plan);
500 var parts = new List<object?> { "PLAN", name, args[2], args[1] };
501 if (plan.Trial != 0) parts.AddRange(new object?[] { "trial", plan.Trial });
502 if (plan.Setup != 0) parts.AddRange(new object?[] { "setup", Money(plan.Setup) });
503 if (plan.PerSeat) parts.Add("perseat");
504 if (plan.Mode is not null) parts.Add(plan.Mode.ToLowerInvariant());
505 Emit(parts.ToArray());
506 }

508 private void CmdTier(string[] args)
509 {
510 Plan plan = RequirePlan(args[0]);
511 if (plan.Mode is null) Fail("not metered " + plan.Name);
512 long? upper = args[1] == "INF" ? null : long.Parse(args[1], CultureInfo.InvariantCulture);
513 if (plan.Tiers.Count > 0)

519 plan.Tiers.Add(new Tier(upper, Cents(args[2]), Cents(args[3])));
520 Emit("TIER", plan.Name, plan.Tiers.Count, "upto", args[1],
521 "unit", args[2], "flat", args[3]);
522 }

556 "COUPON", name, coupon.Kind.ToLowerInvariant(),
557 coupon.Kind == "PERCENT" ? Percent(value) : Money(value),
558 application.ToLowerInvariant()
559 };
560 if (application == "REPEATING") parts.Add(applications);
561 if (maximum is not null) parts.AddRange(new object?[] { "max", maximum });
562 Emit(parts.ToArray());

593 private void CmdSubscribe(string[] args)
594 {
595 Customer customer = RequireCustomer(args[0]);
596 Plan plan = RequirePlan(args[1]);
597 int seats = 1;
598 bool hasSeats = false;
599 string? couponName = null;
600 for (int i = 2; i < args.Length; i += 2)
601 {
602 if (args[i] == "SEATS")
603 {
604 seats = Integer(args[i + 1]);
605 hasSeats = true;
606 }
607 else
608 {
609 couponName = args[i + 1];
610 }
611 }
612 if (hasSeats && !plan.PerSeat) Fail("not per-seat " + plan.Name);
613 if (seats == 0) Fail("invalid seats");
614 Coupon? coupon = couponName is null ? null : RedeemableCoupon(couponName);
615 var subscription = new Subscription(subscriptions.Count + 1, customer, plan, seats, coupon);
616 if (coupon is not null) ++coupon.Redemptions;
617 subscriptions.Add(subscription);
618 customer.Subscriptions.Add(subscription);
619 var parts = new List<object?> {
620 "SUBSCRIBE", "sub", subscription.Number, customer.Name, plan.Name, "seats", seats
621 };
622 if (coupon is not null) parts.AddRange(new object?[] { "coupon", coupon.Name });
623 Emit(parts.ToArray());
624 if (plan.Trial != 0)
625 {
626 subscription.TrialEnd = day + plan.Trial;
627 ChangeState(subscription, "TRIALING");
628 }
629 else
630 {
631 ChangeState(subscription, "ACTIVE");
632 subscription.Start = day;
633 subscription.End = day + plan.Length;
634 Issue(subscription, "first", OpeningItems(subscription, day));
635 }
636 }

638 private void CmdChange(string[] args)
639 {
640 Subscription subscription = RequireSubscription(args[0], "ACTIVE", "TRIALING");
641 Plan plan;
642 int seats;
643 if (args[1] == "PLAN")
644 {
645 plan = RequirePlan(args[2]);
646 seats = plan.PerSeat ? subscription.Seats : 1;
647 }
648 else
649 {
650 plan = subscription.Plan;
651 if (!plan.PerSeat) Fail("not per-seat " + plan.Name);
652 seats = Integer(args[2]);
653 if (seats == 0) Fail("invalid seats");
654 }
655 if (ReferenceEquals(plan, subscription.Plan) && seats == subscription.Seats)
656 Fail("no change " + args[0]);
657
658 Plan oldPlan = subscription.Plan;
659 int oldSeats = subscription.Seats;
660 var head = new object?[] {
661 "CHANGE", "sub", subscription.Number, oldPlan.Name, "seats", oldSeats,
662 "->", plan.Name, "seats", seats
663 };
664 subscription.Pending = null;
665 bool upgrade = plan.Fee(seats) * oldPlan.Length > oldPlan.Fee(oldSeats) * plan.Length;
666 if (subscription.State == "TRIALING")
667 {
668 subscription.Plan = plan;
669 subscription.Seats = seats;
670 Emit(head.Concat(new object?[] { "now" }).ToArray());
671 }
672 else if (upgrade)
673 {
674 Emit(head.Concat(new object?[] { "now" }).ToArray());
675 int days = subscription.End!.Value - day;
676 var items = UsageItems(subscription, true);
677 items.Add(new LineItem($"unused {oldPlan.Name} seats {oldSeats} {days} days",
678 -(oldPlan.Fee(oldSeats) * days / oldPlan.Length)));
679 items.Add(new LineItem($"remaining {plan.Name} seats {seats} {days} days",
680 plan.Fee(seats) * days / plan.Length));
681 subscription.Plan = plan;
682 subscription.Seats = seats;
683 Issue(subscription, "upgrade", items);
684 }
685 else
686 {
687 subscription.Pending = (plan, seats);
688 Emit(head.Concat(new object?[] { "at", subscription.End }).ToArray());
689 }
690 }

711 private void CmdCancel(string[] args)
712 {
713 Subscription subscription = RequireSubscription(args[0]);
714 if (args[1] == "END")
715 {
716 if (subscription.State is not ("ACTIVE" or "TRIALING"))
717 Fail("invalid state " + args[0] + " " + subscription.State);
718 if (subscription.CancelEnd) Fail("already canceling " + args[0]);
719 subscription.CancelEnd = true;
720 int until = (subscription.State == "TRIALING"
721 ? subscription.TrialEnd : subscription.End)!.Value;
722 Emit("CANCEL", "sub", subscription.Number, "end", until);
723 }
724 else
725 {
726 if (subscription.State == "CANCELED")
727 Fail("invalid state " + args[0] + " " + subscription.State);
728 long refund = 0;
729 if (subscription.State is "ACTIVE" or "PAUSED")
730 {
731 int start = subscription.State == "ACTIVE"
732 ? day : subscription.PauseStart!.Value;
733 refund = subscription.Fee() * (subscription.End!.Value - start)
734 / subscription.Plan.Length;
735 }
736 BalanceChange(subscription.Customer, refund, "cancel sub " + subscription.Number);
737 Emit("CANCEL", "sub", subscription.Number, "now", "credit", Money(refund));
738 CancelSubscription(subscription);
739 }
740 }

775 private void CmdRefund(string[] args)
776 {
777 Invoice invoice = RequireInvoice(args[0]);
778 long value = Cents(args[1]);
779 if (value == 0) Fail("invalid amount");
780 long maximum = invoice.Total - invoice.Refunded;
781 if (value > maximum) Fail("refund exceeds " + args[0] + " " + Money(maximum));
782 long applied = Math.Min(value, invoice.Outstanding);
783 invoice.Refunded += value;
784 invoice.Outstanding -= applied;
785 BalanceChange(invoice.Customer, value - applied, "refund invoice " + invoice.Number);
786 Emit("REFUND", "invoice", invoice.Number, "amount", args[1],
787 "applied", Money(applied), "balance", Money(invoice.Customer.Balance));
788 if (applied > 0 && invoice.Outstanding == 0) PaymentComplete(invoice);
789 }

907 private void CmdMrr()
908 {
909 var grouped = new SortedDictionary<string, (int Subs, long Monthly)>(StringComparer.Ordinal);
910 foreach (var subscription in subscriptions)
911 {
912 if (subscription.State is not ("ACTIVE" or "PAST_DUE")) continue;
913 long fee = subscription.Fee();
914 long monthly = subscription.Plan.Cycle switch
915 {
916 "MONTHLY" => fee,
917 "YEARLY" => fee / 12,
918 _ => fee * 30 / 7
919 };
920 grouped.TryGetValue(subscription.Plan.Name, out var previous);
921 grouped[subscription.Plan.Name] = (previous.Subs + 1, previous.Monthly + monthly);
922 }
923 Emit("MRR");
924 foreach (var (name, row) in grouped)
925 Indent(name, "subs", row.Subs, "mrr", Money(row.Monthly));
926 Indent("total subs", grouped.Sum(row => row.Value.Subs),
927 "mrr", Money(grouped.Sum(row => row.Value.Monthly)));
928 }

930 private static object?[] AgingParts(long[] amounts) => new object?[] {
931 "0-30", Money(amounts[0]), "31-60", Money(amounts[1]),
932 "61+", Money(amounts[2]), "sum", Money(amounts.Sum())
933 };

935 private void CmdAging()
936 {
937 var byCustomer = new SortedDictionary<string, long[]>(StringComparer.Ordinal);
938 var totals = new long[3];
939 foreach (var invoice in invoices)
940 {
941 if (invoice.Outstanding == 0) continue;
942 int age = day - invoice.Day;
943 int bucket = age <= 30 ? 0 : age <= 60 ? 1 : 2;
944 if (!byCustomer.TryGetValue(invoice.Customer.Name, out var amounts))
945 {
946 amounts = new long[3];
947 byCustomer[invoice.Customer.Name] = amounts;
948 }
949 amounts[bucket] += invoice.Outstanding;
950 totals[bucket] += invoice.Outstanding;
951 }
952 Emit("AGING", "day", day);
953 foreach (var (name, amounts) in byCustomer)
954 Indent(new object?[] { name }.Concat(AgingParts(amounts)).ToArray());
955 Indent(new object?[] { "total" }.Concat(AgingParts(totals)).ToArray());
956 }

1004 case "TIER": CmdTier(args); break;
1005 case "TAX": CmdTax(args); break;
1006 case "CUSTOMER": CmdCustomer(args); break;
1007 case "COUPON": CmdCoupon(args); break;

1010 case "DAY": CmdDay(args); break;
1011 case "SUBSCRIBE": CmdSubscribe(args); break;
1012 case "CHANGE": CmdChange(args); break;
1013 case "PAUSE": CmdPause(args); break;
1014 case "RESUME": CmdResume(args); break;
1015 case "CANCEL": CmdCancel(args); break;
1016 case "USAGE": CmdUsage(args); break;
1017 case "APPLY": CmdApply(args); break;
1018 case "PAY": CmdPay(args); break;
1019 case "REFUND": CmdRefund(args); break;
1020 case "INVOICE": CmdInvoice(args); break;
