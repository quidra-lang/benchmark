8private fun cents(text: String): Long {
9 val parts = text.split(".")
10 return parts[0].toLong() * 100 + parts[1].toLong()
11}

13private fun money(amount: Long): String {
14 val absolute = kotlin.math.abs(amount)
15 return (if (amount < 0) "-" else "") +
16 (absolute / 100).toString() + "." + (absolute % 100).toString().padStart(2, '0')
17}

19private fun signedMoney(amount: Long): String =

22private fun percentage(basisPoints: Int): String = money(basisPoints.toLong()) + "%"

24private data class Tier(val upper: Long?, val unitPrice: Long, val flatFee: Long)

26private class Plan(
27 val name: String,
28 val price: Long,
29 val cycle: String
30) {
31 val length: Int = when (cycle) {
32 "WEEKLY" -> 7
33 "MONTHLY" -> 30
34 else -> 360
35 }
36 var trial = 0
37 var setup = 0L
38 var perSeat = false
39 var mode: String? = null
40 val tiers = mutableListOf<Tier>()
41
42 fun fee(seats: Int): Long = if (perSeat) price * seats else price
43
44 fun usageCharge(units: Long): Long {
45 var total = 0L
46 var lower = 0L
47 for ((index, tier) in tiers.withIndex()) {
48 val last = index == tiers.lastIndex
49 if (mode == "VOLUME") {
50 if (last || tier.upper == null || units <= tier.upper) {
51 return units * tier.unitPrice + tier.flatFee
52 }
53 } else {
54 if (units <= lower) break
55 val end = if (last || tier.upper == null) units else minOf(units, tier.upper)
56 total += (end - lower) * tier.unitPrice + tier.flatFee
57 if (last || tier.upper == null) break
58 lower = tier.upper
59 }
60 }
61 return total
62 }
63}

249 private fun couponUsable(subscription: Subscription, subtotal: Long): Boolean =
250 subtotal > 0L && subscription.coupon != null &&
251 (subscription.remaining == null || subscription.remaining!! > 0)
252
253 private fun invoiceTotals(subscription: Subscription, items: List<BillItem>): Totals {
254 val subtotal = items.sumOf { it.second }
255 var discount = 0L
256 if (couponUsable(subscription, subtotal)) {
257 val coupon = subscription.coupon!!
258 discount = if (coupon.kind == "PERCENT") {
259 subtotal * coupon.value / 10000
260 } else {
261 minOf(subtotal, coupon.value)
262 }
263 }
264 val tax = (subtotal - discount) * (taxRates[subscription.customer.region] ?: 0) / 10000
265 val total = subtotal - discount + tax
266 val credit = minOf(subscription.customer.balance, total)
267 return Totals(subtotal, discount, tax, total, credit, total - credit)
268 }

270 private fun issue(subscription: Subscription, kind: String, items: List<BillItem>) {
271 val totals = invoiceTotals(subscription, items)
272 val invoice = Invoice(invoices.size + 1, subscription, day, kind, items.toList(), totals)
273 if (couponUsable(subscription, invoice.subtotal) && subscription.remaining != null) {
274 subscription.remaining = subscription.remaining!! - 1
275 }
276 invoices.add(invoice)
277 subscription.invoices.add(invoice)
278 subscription.customer.invoices.add(invoice)
279 balanceChange(subscription.customer, -invoice.credit, "applied invoice ${invoice.number}")
280
281 val result: String
282 if (invoice.due == 0L) {
283 result = "PAID"
284 } else if (subscription.state == "UNPAID") {
285 result = "OPEN"
286 } else {
287 invoice.attempts = 1
288 if (subscription.customer.method == "OK") {
289 invoice.paid = invoice.due
290 invoice.outstanding = 0L
291 result = "PAID"
292 } else {
293 result = "FAILED"
294 }
295 }
296
297 notice(
298 "invoice", invoice.number, "sub", subscription.number, kind,
299 "total", money(invoice.total), "credit", money(invoice.credit),
300 "due", money(invoice.due), result
301 )
302 if (result == "FAILED") {
303 if (subscription.state == "ACTIVE") changeState(subscription, "PAST_DUE")
304 if (subscription.state == "PAST_DUE") {
305 invoice.retries.addAll(listOf(day + 3, day + 5, day + 7))
306 }
307 }
308 }

310 private fun paymentComplete(invoice: Invoice) {
311 invoice.retries.clear()
312 val subscription = invoice.subscription
313 if (subscription.state in setOf("PAST_DUE", "UNPAID") &&
314 subscription.invoices.all { it.outstanding == 0L }) {
315 subscription.cancelOn = null
316 changeState(subscription, "ACTIVE")
317 }
318 }

320 private fun cancelSubscription(subscription: Subscription) {
321 changeState(subscription, "CANCELED")
322 subscription.canceledOn = day
323 subscription.pending = null
324 subscription.cancelEnd = false
325 subscription.cancelOn = null
326 val items = usageItems(subscription, close = true)
327 if (items.isNotEmpty()) issue(subscription, "final", items)
328 }
329
330 private fun retry(invoice: Invoice) {
331 invoice.retries.removeAt(0)
332 invoice.attempts++
333 val number = invoice.attempts - 1
334 if (invoice.customer.method == "OK") {
335 invoice.paid += invoice.outstanding
336 invoice.outstanding = 0L
337 notice("retry", number, "invoice", invoice.number, "PAID")
338 paymentComplete(invoice)
339 } else {
340 notice("retry", number, "invoice", invoice.number, "FAILED")
341 if (invoice.retries.isEmpty()) {
342 changeState(invoice.subscription, "UNPAID")
343 invoice.subscription.cancelOn = day + 14
344 }
345 }
346 }

348 private fun advanceSubscription(subscription: Subscription) {
349 if (subscription.state in setOf("PAUSED", "CANCELED")) return
350
351 for (invoice in subscription.invoices.toList()) {
352 if (invoice.retries.firstOrNull() == day) retry(invoice)
353 }
354 if (subscription.state == "UNPAID" && subscription.cancelOn == day) {
355 cancelSubscription(subscription)
356 }
357 if (subscription.state == "TRIALING" && subscription.trialEnd == day) {
358 if (subscription.cancelEnd) {
359 cancelSubscription(subscription)
360 } else {
361 changeState(subscription, "ACTIVE")
362 subscription.start = day
363 subscription.end = day + subscription.plan.length
364 issue(subscription, "first", openingItems(subscription, day))
365 }
366 }
367 if (subscription.state in setOf("ACTIVE", "PAST_DUE", "UNPAID") &&
368 subscription.end == day) {
369 if (subscription.cancelEnd) {
370 cancelSubscription(subscription)
371 } else {
372 val items = usageItems(subscription, close = true)
373 val pending = subscription.pending
374 if (pending != null) {
375 val (plan, seats) = pending
376 notice(
377 "sub", subscription.number, subscription.plan.name, "seats",
378 subscription.seats, "->", plan.name, "seats", seats
379 )
380 subscription.plan = plan
381 subscription.seats = seats
382 subscription.pending = null
383 }
384 subscription.start = day
385 subscription.end = day + subscription.plan.length
386 items.add(periodItem(subscription.plan, subscription.seats, day))
387 issue(subscription, "renewal", items)
388 }
389 }

392 private fun commandPlan(a: List<String>) {
393 val name = a[0]
394 val amount = a[1]
395 val cycle = a[2]
396 if (plans.containsKey(name)) fail("duplicate plan $name")
397 val plan = Plan(name, cents(amount), cycle)
398 var index = 3
399 while (index < a.size) {
400 when (a[index]) {
401 "TRIAL" -> {
402 plan.trial = a[index + 1].toInt()
403 index += 2
404 }
405 "SETUP" -> {
406 plan.setup = cents(a[index + 1])
407 index += 2
408 }
409 "PERSEAT" -> {
410 plan.perSeat = true
411 index++
412 }
413 else -> {
414 plan.mode = a[index]
415 index++
416 }
417 }
418 }
419 plans[name] = plan
420 var line = "PLAN $name $cycle $amount"
421 if (plan.trial != 0) line += " trial ${plan.trial}"
422 if (plan.setup != 0L) line += " setup ${money(plan.setup)}"
423 if (plan.perSeat) line += " perseat"
424 if (plan.mode != null) line += " ${plan.mode!!.lowercase()}"
425 emit(line)
426 }

428 private fun commandTier(a: List<String>) {
429 val name = a[0]
430 val plan = plan(name)
431 if (plan.mode == null) fail("not metered $name")
432 val upper: Long? = if (a[1] == "INF") null else a[1].toLong()
433 if (plan.tiers.isNotEmpty()) {
434 val previous = plan.tiers.last().upper
435 if (previous == null || (upper != null && upper <= previous)) {
436 fail("tier order $name")
437 }
438 }
439 plan.tiers.add(Tier(upper, cents(a[2]), cents(a[3])))
440 emit("TIER", name, plan.tiers.size, "upto", a[1], "unit", a[2], "flat", a[3])
441 }

456 private fun commandCoupon(a: List<String>) {
457 val name = a[0]
458 val kind = a[1]
459 val application = a[3]
460 if (coupons.containsKey(name)) fail("duplicate coupon $name")
461 var applications: Int? = if (application == "ONCE") 1 else null
462 var index = 4
463 if (application == "REPEATING") {
464 applications = a[index].toInt()
465 index++
466 }
467 val maximum: Int? = if (index < a.size) a[index + 1].toInt() else null
468 val numericValue = if (kind == "PERCENT") a[2].toLong() else cents(a[2])
469 val coupon = Coupon(name, kind, numericValue, applications, maximum)
470 coupons[name] = coupon
471 val shownValue = if (kind == "PERCENT") percentage(numericValue.toInt()) else money(numericValue)
472 var line = "COUPON $name ${kind.lowercase()} $shownValue ${application.lowercase()}"
473 if (application == "REPEATING") line += " $applications"
474 if (maximum != null) line += " max $maximum"
475 emit(line)
476 }

502 private fun commandSubscribe(a: List<String>) {
503 val customer = customer(a[0])
504 val plan = plan(a[1])
505 var seats = 1
506 var hasSeats = false
507 var couponName: String? = null
508 var index = 2
509 while (index < a.size) {
510 if (a[index] == "SEATS") {
511 seats = a[index + 1].toInt()
512 hasSeats = true
513 } else {
514 couponName = a[index + 1]
515 }
516 index += 2
517 }
518 if (hasSeats && !plan.perSeat) fail("not per-seat ${plan.name}")
519 if (seats == 0) fail("invalid seats")
520 val coupon = if (couponName == null) null else redeemableCoupon(couponName)
521 val subscription = Subscription(subscriptions.size + 1, customer, plan, seats, coupon)
522 if (coupon != null) coupon.redemptions++
523 subscriptions.add(subscription)
524 customer.subscriptions.add(subscription)
525 var line = "SUBSCRIBE sub ${subscription.number} ${customer.name} ${plan.name} seats $seats"
526 if (coupon != null) line += " coupon ${coupon.name}"
527 emit(line)
528 if (plan.trial != 0) {
529 subscription.trialEnd = day + plan.trial
530 changeState(subscription, "TRIALING")
531 } else {
532 changeState(subscription, "ACTIVE")
533 subscription.start = day
534 subscription.end = day + plan.length
535 issue(subscription, "first", openingItems(subscription, day))
536 }

539 private fun commandChange(a: List<String>) {
540 val subscription = subscription(a[0], setOf("ACTIVE", "TRIALING"))
541 val plan: Plan
542 val seats: Int
543 if (a[1] == "PLAN") {
544 plan = plan(a[2])
545 seats = if (plan.perSeat) subscription.seats else 1
546 } else {
547 plan = subscription.plan
548 if (!plan.perSeat) fail("not per-seat ${plan.name}")
549 seats = a[2].toInt()
550 if (seats == 0) fail("invalid seats")
551 }
552 if (plan === subscription.plan && seats == subscription.seats) {
553 fail("no change ${a[0]}")
554 }
555 val oldPlan = subscription.plan
556 val oldSeats = subscription.seats
557 val head = "CHANGE sub ${subscription.number} ${oldPlan.name} seats $oldSeats -> ${plan.name} seats $seats"
558 subscription.pending = null
559 val upgrade = plan.fee(seats) * oldPlan.length > oldPlan.fee(oldSeats) * plan.length
560 if (subscription.state == "TRIALING") {
561 subscription.plan = plan
562 subscription.seats = seats
563 emit("$head now")
564 } else if (upgrade) {
565 emit("$head now")
566 val days = subscription.end!! - day
567 val items = usageItems(subscription, close = true)
568 items.add("unused ${oldPlan.name} seats $oldSeats $days days" to
569 -(oldPlan.fee(oldSeats) * days / oldPlan.length))
570 items.add("remaining ${plan.name} seats $seats $days days" to
571 (plan.fee(seats) * days / plan.length))
572 subscription.plan = plan
573 subscription.seats = seats
574 issue(subscription, "upgrade", items)
575 } else {
576 subscription.pending = plan to seats
577 emit("$head at ${subscription.end}")
578 }
579 }

597 private fun commandCancel(a: List<String>) {
598 val subscription = subscription(a[0])
599 if (a[1] == "END") {
600 if (subscription.state !in setOf("ACTIVE", "TRIALING")) {
601 fail("invalid state ${a[0]} ${subscription.state}")
602 }
603 if (subscription.cancelEnd) fail("already canceling ${a[0]}")
604 subscription.cancelEnd = true
605 val at = if (subscription.state == "TRIALING") subscription.trialEnd else subscription.end
606 emit("CANCEL", "sub", subscription.number, "end", at!!)
607 } else {
608 if (subscription.state == "CANCELED") {
609 fail("invalid state ${a[0]} ${subscription.state}")
610 }
611 var refund = 0L
612 if (subscription.state in setOf("ACTIVE", "PAUSED")) {
613 val start = if (subscription.state == "ACTIVE") day else subscription.pauseStart!!
614 refund = subscription.fee() * (subscription.end!! - start) / subscription.plan.length
615 }
616 balanceChange(subscription.customer, refund, "cancel sub ${subscription.number}")
617 emit("CANCEL", "sub", subscription.number, "now", "credit", money(refund))
618 cancelSubscription(subscription)
619 }

653 val invoice = invoice(a[0])
654 val value = cents(a[1])

656 val maximum = invoice.total - invoice.refunded
657 if (value > maximum) {
658 fail("refund exceeds ${a[0]} ${money(maximum)}")

660 val applied = minOf(value, invoice.outstanding)

663 balanceChange(invoice.customer, value - applied, "refund invoice ${invoice.number}")
664 emit(
665 "REFUND", "invoice", invoice.number, "amount", a[1],
666 "applied", money(applied), "balance", money(invoice.customer.balance)
667 )
668 if (applied != 0L && invoice.outstanding == 0L) paymentComplete(invoice)

699 private fun commandSub(a: List<String>) {
700 val subscription = subscription(a[0])
701 emit(
702 "SUB", subscription.number, subscription.customer.name, subscription.plan.name,
703 "seats", subscription.seats, subscription.state
704 )
705 if (subscription.state == "TRIALING") detail("trial until", subscription.trialEnd!!)
706 if (subscription.state in setOf("ACTIVE", "PAST_DUE", "UNPAID", "PAUSED")) {
707 detail("period", subscription.period())
708 }
709 if (subscription.state == "PAUSED") detail("paused since", subscription.pauseStart!!)
710 subscription.pending?.let { (plan, seats) ->
711 detail("pending", plan.name, "seats", seats)
712 }
713 if (subscription.cancelEnd) detail("cancel at end")
714 subscription.cancelOn?.let { detail("cancel on", it) }
715 if (subscription.state == "CANCELED") detail("canceled on", subscription.canceledOn!!)
716 subscription.coupon?.let { coupon ->
717 if (subscription.remaining == null) {
718 detail("coupon", coupon.name, "forever")
719 } else if (subscription.remaining!! > 0) {
720 detail("coupon", coupon.name, "left", subscription.remaining!!)
721 }
722 }
723 if (subscription.plan.mode != null && subscription.state != "CANCELED") {
724 detail("usage", subscription.usage)
725 }
726 val open = subscription.invoices.filter { it.outstanding != 0L }.map { it.number }
727 if (open.isNotEmpty()) detail("open", open.joinToString(" "))
728 }

730 private fun commandPreview(a: List<String>) {
731 val subscription = subscription(a[0], setOf("TRIALING", "ACTIVE", "PAST_DUE", "UNPAID"))
732 val date = if (subscription.state == "TRIALING") subscription.trialEnd!! else subscription.end!!
733 val kind: String
734 val items: MutableList<BillItem>
735 if (subscription.cancelEnd) {
736 kind = "final"
737 items = if (subscription.state == "TRIALING") mutableListOf() else usageItems(subscription)
738 } else if (subscription.state == "TRIALING") {
739 kind = "first"
740 items = openingItems(subscription, date)
741 } else {
742 kind = "renewal"
743 val (plan, seats) = subscription.pending ?: (subscription.plan to subscription.seats)
744 items = usageItems(subscription)
745 items.add(periodItem(plan, seats, date))
746 }
747 emit("PREVIEW", "sub", subscription.number, "day", date, kind)
748 if (items.isEmpty()) {
749 detail("(none)")
750 return
751 }
752 emitItems(items)
753 val totals = invoiceTotals(subscription, items)
754 emitTotals(totals)
755 detail("credit", money(totals.credit), "due", money(totals.due))
756 }

790 private fun commandMrr() {
791 val byPlan = mutableMapOf<String, Pair<Int, Long>>()
792 for (subscription in subscriptions) {
793 if (subscription.state !in setOf("ACTIVE", "PAST_DUE")) continue
794 val fee = subscription.fee()
795 val monthly = when (subscription.plan.cycle) {
796 "MONTHLY" -> fee
797 "YEARLY" -> fee / 12
798 else -> fee * 30 / 7
799 }
800 val row = byPlan[subscription.plan.name] ?: (0 to 0L)
801 byPlan[subscription.plan.name] = (row.first + 1) to (row.second + monthly)
802 }
803 emit("MRR")
804 for (name in byPlan.keys.sorted()) {
805 val (count, monthly) = byPlan.getValue(name)
806 detail(name, "subs", count, "mrr", money(monthly))
807 }
808 detail(
809 "total subs", byPlan.values.sumOf { it.first },
810 "mrr", money(byPlan.values.sumOf { it.second })
811 )
812 }

815 "0-30 ${money(amounts[0])} 31-60 ${money(amounts[1])} " +
816 "61+ ${money(amounts[2])} sum ${money(amounts.sum())}"

818 private fun commandAging() {
819 val byCustomer = mutableMapOf<String, LongArray>()
820 val totals = LongArray(3)
821 for (invoice in invoices) {
822 if (invoice.outstanding == 0L) continue
823 val age = day - invoice.day
824 val bucket = if (age <= 30) 0 else if (age <= 60) 1 else 2
825 byCustomer.getOrPut(invoice.customer.name) { LongArray(3) }[bucket] += invoice.outstanding
826 totals[bucket] += invoice.outstanding
827 }
828 emit("AGING", "day", day)
829 for (name in byCustomer.keys.sorted()) detail(name, agingParts(byCustomer.getValue(name)))
830 detail("total", agingParts(totals))
831 }

873 fun execute(line: String) {
874 val words = line.trim().split(" ")
875 val args = words.drop(1)
876 try {
877 when (words[0]) {
878 "PLAN" -> commandPlan(args)
879 "TIER" -> commandTier(args)
880 "TAX" -> commandTax(args)
881 "CUSTOMER" -> commandCustomer(args)
882 "COUPON" -> commandCoupon(args)
883 "PAYMETHOD" -> commandPayMethod(args)
884 "CREDIT" -> commandCredit(args)
885 "DAY" -> commandDay(args)
886 "SUBSCRIBE" -> commandSubscribe(args)
887 "CHANGE" -> commandChange(args)
888 "PAUSE" -> commandPause(args)
889 "RESUME" -> commandResume(args)
890 "CANCEL" -> commandCancel(args)
891 "USAGE" -> commandUsage(args)
892 "APPLY" -> commandApply(args)
893 "PAY" -> commandPay(args)
894 "REFUND" -> commandRefund(args)
895 "INVOICE" -> commandInvoice(args)
896 "SUB" -> commandSub(args)
897 "PREVIEW" -> commandPreview(args)
898 "STATEMENT" -> commandStatement(args)
899 "LEDGER" -> commandLedger(args)
900 "MRR" -> commandMrr()
901 "AGING" -> commandAging()
902 "DUNNING" -> commandDunning()
903 "REVENUE" -> commandRevenue()
904 "EVENTS" -> commandEvents(args)
905 }
906 } catch (error: CommandError) {
907 emit("ERROR", error.message ?: "")
908 }
