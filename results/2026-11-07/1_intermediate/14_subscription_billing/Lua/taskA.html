3local function cents(text)
4 local whole, fraction = text:match("^(%d+)%.(%d%d)$")
5 return tonumber(whole) * 100 + tonumber(fraction)
6end
7
8local function money(value)
9 local sign = value < 0 and "-" or ""
10 local absolute = math.abs(value)
11 return sign .. string.format("%d.%02d", absolute // 100, absolute % 100)
12end
13
14local function signed_money(value)
15 return (value > 0 and "+" or "") .. money(value)
16end
17
18local function percentage(bp)
19 return money(bp) .. "%"
20end
21
22local function error_command(reason)
23 error({reason = reason}, 0)
24end
25
26local function contains(items, item)
27 for _, value in ipairs(items) do
28 if value == item then return true end
29 end
30 return false
31end
32
33local function fee(plan, seats)
34 return plan.price * (plan.per_seat and seats or 1)
35end
36
37local function usage_charge(plan, units)
38 local total = 0
39 local lower = 0
40 for index, tier in ipairs(plan.tiers) do
41 local last = index == #plan.tiers
42 if plan.mode == "VOLUME" then
43 if last or tier.upper == nil or units <= tier.upper then
44 return units * tier.unit + tier.flat
45 end
46 else
47 if units <= lower then break end
48 local upper = (last or tier.upper == nil) and units or tier.upper
49 total = total + (math.min(units, upper) - lower) * tier.unit + tier.flat
50 if last or tier.upper == nil then break end
51 lower = tier.upper
52 end
53 end
54 return total
55end

135function Billing:change_state(subscription, state)
136 local old = subscription.state
137 if old == "PAST_DUE" then
138 for _, invoice in ipairs(subscription.invoices) do
139 invoice.retries = {}

141 end
142 subscription.state = state

150 return {
151 description = "charge " .. plan.name .. " seats " .. seats .. " " ..

157local function opening_items(subscription, start)
158 local plan = subscription.plan
159 local items = {period_item(plan, subscription.seats, start)}

161 items[#items + 1] = {description = "setup " .. plan.name, amount = plan.setup}

166local function usage_items(subscription, close)
167 local plan = subscription.plan
168 if not plan.mode or subscription.usage == 0 then return {} end
169 local units = subscription.usage
170 local item = {
171 description = "usage " .. plan.name .. " " .. units .. " units",
172 amount = usage_charge(plan, units)
173 }
174 if close then subscription.usage = 0 end
175 return {item}

178local function coupon_usable(subscription, subtotal)
179 return subtotal > 0 and subscription.coupon ~= nil and
180 (subscription.remaining == nil or subscription.remaining > 0)
181end
182
183function Billing:invoice_totals(subscription, items)
184 local subtotal = 0
185 for _, item in ipairs(items) do subtotal = subtotal + item.amount end
186 local discount = 0
187 if coupon_usable(subscription, subtotal) then
188 local coupon = subscription.coupon
189 if coupon.kind == "PERCENT" then
190 discount = (subtotal * coupon.value) // 10000
191 else
192 discount = math.min(subtotal, coupon.value)
193 end
194 end
195 local tax = ((subtotal - discount) * (self.taxes[subscription.customer.region] or 0)) // 10000
196 local total = subtotal - discount + tax
197 local credit = math.min(subscription.customer.balance, total)
198 return {subtotal, discount, tax, total, credit, total - credit}
199end

201function Billing:issue(subscription, kind, items)
202 local totals = self:invoice_totals(subscription, items)
203 local invoice = {
204 number = #self.invoices + 1, subscription = subscription,
205 customer = subscription.customer, day = self.day, kind = kind, items = items,
206 subtotal = totals[1], discount = totals[2], tax = totals[3],
207 total = totals[4], credit = totals[5], due = totals[6],
208 paid = 0, refunded = 0, outstanding = totals[6], attempts = 0, retries = {}
209 }
210 if coupon_usable(subscription, invoice.subtotal) and subscription.remaining ~= nil then
211 subscription.remaining = subscription.remaining - 1
212 end
213 self.invoices[#self.invoices + 1] = invoice
214 subscription.invoices[#subscription.invoices + 1] = invoice
215 subscription.customer.invoices[#subscription.customer.invoices + 1] = invoice
216 self:balance_change(subscription.customer, -invoice.credit, "applied invoice " .. invoice.number)
217
218 local result
219 if invoice.due == 0 then
220 result = "PAID"
221 elseif subscription.state == "UNPAID" then
222 result = "OPEN"
223 else
224 invoice.attempts = 1
225 if subscription.customer.method == "OK" then
226 invoice.paid = invoice.due
227 invoice.outstanding = 0
228 result = "PAID"
229 else
230 result = "FAILED"
231 end
232 end
233 self:notice("invoice", invoice.number, "sub", subscription.number,
234 kind, "total", money(invoice.total), "credit", money(invoice.credit),
235 "due", money(invoice.due), result)
236 if result == "FAILED" then
237 if subscription.state == "ACTIVE" then
238 self:change_state(subscription, "PAST_DUE")
239 end
240 if subscription.state == "PAST_DUE" then
241 invoice.retries = {self.day + 3, self.day + 5, self.day + 7}
242 end
243 end
244end
245
246function Billing:payment_complete(invoice)
247 invoice.retries = {}
248 local subscription = invoice.subscription
249 if subscription.state == "PAST_DUE" or subscription.state == "UNPAID" then
250 for _, other in ipairs(subscription.invoices) do
251 if other.outstanding ~= 0 then return end
252 end
253 subscription.cancel_on = nil
254 self:change_state(subscription, "ACTIVE")
255 end
256end

258function Billing:cancel_subscription(subscription)
259 self:change_state(subscription, "CANCELED")
260 subscription.canceled_on = self.day
261 subscription.pending = nil
262 subscription.cancel_end = false
263 subscription.cancel_on = nil
264 local items = usage_items(subscription, true)
265 if #items > 0 then
266 self:issue(subscription, "final", items)
267 end
268end
269
270function Billing:retry(invoice)
271 table.remove(invoice.retries, 1)
272 invoice.attempts = invoice.attempts + 1
273 local retry_number = invoice.attempts - 1
274 if invoice.customer.method == "OK" then
275 invoice.paid = invoice.paid + invoice.outstanding
276 invoice.outstanding = 0
277 self:notice("retry", retry_number, "invoice", invoice.number, "PAID")
278 self:payment_complete(invoice)
279 else
280 self:notice("retry", retry_number, "invoice", invoice.number, "FAILED")
281 if #invoice.retries == 0 then
282 self:change_state(invoice.subscription, "UNPAID")
283 invoice.subscription.cancel_on = self.day + 14
284 end
285 end
286end

288function Billing:advance_subscription(subscription)
289 if subscription.state == "PAUSED" or subscription.state == "CANCELED" then return end
290 for _, invoice in ipairs(subscription.invoices) do
291 if #invoice.retries > 0 and invoice.retries[1] == self.day then
292 self:retry(invoice)
293 end
294 end
295 if subscription.state == "UNPAID" and subscription.cancel_on == self.day then
296 self:cancel_subscription(subscription)
297 end
298 if subscription.state == "TRIALING" and subscription.trial_end == self.day then
299 if subscription.cancel_end then
300 self:cancel_subscription(subscription)
301 else
302 self:change_state(subscription, "ACTIVE")
303 subscription.start = self.day
304 subscription.finish = self.day + subscription.plan.length
305 self:issue(subscription, "first", opening_items(subscription, self.day))
306 end
307 end
308 if contains({"ACTIVE", "PAST_DUE", "UNPAID"}, subscription.state) and
309 subscription.finish == self.day then
310 if subscription.cancel_end then
311 self:cancel_subscription(subscription)
312 else
313 local items = usage_items(subscription, true)
314 if subscription.pending then
315 local plan, seats = subscription.pending[1], subscription.pending[2]
316 self:notice("sub", subscription.number, subscription.plan.name,
317 "seats", subscription.seats, "->", plan.name, "seats", seats)
318 subscription.plan, subscription.seats = plan, seats
319 subscription.pending = nil
320 end
321 subscription.start = self.day
322 subscription.finish = self.day + subscription.plan.length
323 items[#items + 1] = period_item(subscription.plan, subscription.seats, self.day)
324 self:issue(subscription, "renewal", items)
325 end
326 end
327end

329function Billing:cmd_plan(args)
330 local name, amount, cycle = args[1], args[2], args[3]
331 if self.plans[name] then error_command("duplicate plan " .. name) end
332 local plan = {
333 name = name, price = cents(amount), cycle = cycle,
334 length = period_lengths[cycle], trial = 0, setup = 0,
335 per_seat = false, mode = nil, tiers = {}
336 }
337 local i = 4
338 while i <= #args do
339 local option = args[i]
340 if option == "TRIAL" then
341 plan.trial = tonumber(args[i + 1])
342 i = i + 2
343 elseif option == "SETUP" then
344 plan.setup = cents(args[i + 1])
345 i = i + 2
346 else
347 if option == "PERSEAT" then plan.per_seat = true
348 else plan.mode = option end
349 i = i + 1
350 end
351 end
352 self.plans[name] = plan
353 local line = {"PLAN", name, cycle, amount}
354 if plan.trial ~= 0 then line[#line + 1] = "trial"; line[#line + 1] = plan.trial end
355 if plan.setup ~= 0 then line[#line + 1] = "setup"; line[#line + 1] = money(plan.setup) end
356 if plan.per_seat then line[#line + 1] = "perseat" end
357 if plan.mode then line[#line + 1] = plan.mode:lower() end
358 self:emit(table.unpack(line))
359end

361function Billing:cmd_tier(args)
362 local name, limit, unit, flat = table.unpack(args)
363 local plan = self:plan(name)
364 if not plan.mode then error_command("not metered " .. name) end
365 local upper = limit ~= "INF" and tonumber(limit) or nil
366 if #plan.tiers > 0 then
367 local previous = plan.tiers[#plan.tiers].upper
368 if previous == nil or (upper ~= nil and upper <= previous) then
369 error_command("tier order " .. name)
370 end
371 end
372 plan.tiers[#plan.tiers + 1] = {
373 upper = upper, unit = cents(unit), flat = cents(flat)
374 }
375 self:emit("TIER", name, #plan.tiers, "upto", limit, "unit", unit, "flat", flat)
376end

394function Billing:cmd_coupon(args)
395 local name, kind, value, application = args[1], args[2], args[3], args[4]
396 if self.coupons[name] then error_command("duplicate coupon " .. name) end
397 local applications = application == "ONCE" and 1 or nil
398 local i = 5
399 if application == "REPEATING" then
400 applications = tonumber(args[5])
401 i = 6
402 end
403 local maximum = i <= #args and tonumber(args[i + 1]) or nil
404 local numeric = kind == "PERCENT" and tonumber(value) or cents(value)
405 local coupon = {
406 name = name, kind = kind, value = numeric, applications = applications,
407 maximum = maximum, redemptions = 0
408 }
409 self.coupons[name] = coupon
410 local shown = kind == "PERCENT" and percentage(numeric) or money(numeric)
411 local line = {"COUPON", name, kind:lower(), shown, application:lower()}
412 if application == "REPEATING" then line[#line + 1] = applications end
413 if maximum then line[#line + 1] = "max"; line[#line + 1] = maximum end
414 self:emit(table.unpack(line))
415end

443function Billing:cmd_subscribe(args)
444 local name, plan_name = args[1], args[2]
445 local customer = self:customer(name)
446 local plan = self:plan(plan_name)
447 local seats, has_seats, coupon_name = 1, false, nil
448 for i = 3, #args, 2 do
449 if args[i] == "SEATS" then
450 seats = tonumber(args[i + 1])
451 has_seats = true
452 else
453 coupon_name = args[i + 1]
454 end
455 end
456 if has_seats and not plan.per_seat then error_command("not per-seat " .. plan_name) end
457 if seats == 0 then error_command("invalid seats") end
458 local coupon = coupon_name and self:redeemable_coupon(coupon_name) or nil
459 local subscription = {
460 number = #self.subscriptions + 1, customer = customer, plan = plan,
461 seats = seats, state = "NONE", start = nil, finish = nil,
462 trial_end = nil, pause_start = nil, pending = nil,
463 cancel_end = false, cancel_on = nil, canceled_on = nil,
464 coupon = coupon, remaining = coupon and coupon.applications or nil,
465 usage = 0, invoices = {}
466 }
467 if coupon then coupon.redemptions = coupon.redemptions + 1 end
468 self.subscriptions[#self.subscriptions + 1] = subscription
469 customer.subscriptions[#customer.subscriptions + 1] = subscription
470 local line = {"SUBSCRIBE", "sub", subscription.number, name, plan_name, "seats", seats}
471 if coupon then line[#line + 1] = "coupon"; line[#line + 1] = coupon.name end
472 self:emit(table.unpack(line))
473 if plan.trial > 0 then
474 subscription.trial_end = self.day + plan.trial
475 self:change_state(subscription, "TRIALING")
476 else
477 self:change_state(subscription, "ACTIVE")
478 subscription.start = self.day
479 subscription.finish = self.day + plan.length
480 self:issue(subscription, "first", opening_items(subscription, self.day))
481 end
482end

484function Billing:cmd_change(args)
485 local text, kind, value = args[1], args[2], args[3]
486 local subscription = self:subscription(text, {"ACTIVE", "TRIALING"})
487 local old_plan, old_seats = subscription.plan, subscription.seats
488 local plan, seats
489 if kind == "PLAN" then
490 plan = self:plan(value)
491 seats = plan.per_seat and old_seats or 1
492 else
493 plan = old_plan
494 if not plan.per_seat then error_command("not per-seat " .. plan.name) end
495 seats = tonumber(value)
496 if seats == 0 then error_command("invalid seats") end
497 end
498 if plan == old_plan and seats == old_seats then error_command("no change " .. text) end
499 local head = {"CHANGE", "sub", subscription.number, old_plan.name, "seats",
500 old_seats, "->", plan.name, "seats", seats}
501 subscription.pending = nil
502 local upgrade = fee(plan, seats) * old_plan.length > fee(old_plan, old_seats) * plan.length
503 if subscription.state == "TRIALING" then
504 subscription.plan, subscription.seats = plan, seats
505 head[#head + 1] = "now"
506 self:emit(table.unpack(head))
507 elseif upgrade then
508 head[#head + 1] = "now"
509 self:emit(table.unpack(head))
510 local days = subscription.finish - self.day
511 local items = usage_items(subscription, true)
512 items[#items + 1] = {
513 description = "unused " .. old_plan.name .. " seats " .. old_seats ..
514 " " .. days .. " days",
515 amount = -(fee(old_plan, old_seats) * days // old_plan.length)
516 }
517 items[#items + 1] = {
518 description = "remaining " .. plan.name .. " seats " .. seats ..
519 " " .. days .. " days",
520 amount = fee(plan, seats) * days // plan.length
521 }
522 subscription.plan, subscription.seats = plan, seats
523 self:issue(subscription, "upgrade", items)
524 else
525 subscription.pending = {plan, seats}
526 head[#head + 1] = "at"
527 head[#head + 1] = subscription.finish
528 self:emit(table.unpack(head))
529 end
530end

549function Billing:cmd_cancel(args)
550 local text, when = args[1], args[2]
551 local subscription = self:subscription(text)
552 if when == "END" then
553 if not contains({"ACTIVE", "TRIALING"}, subscription.state) then
554 error_command("invalid state " .. text .. " " .. subscription.state)
555 end
556 if subscription.cancel_end then error_command("already canceling " .. text) end
557 subscription.cancel_end = true
558 local day = subscription.state == "TRIALING" and
559 subscription.trial_end or subscription.finish
560 self:emit("CANCEL", "sub", subscription.number, "end", day)
561 else
562 if subscription.state == "CANCELED" then
563 error_command("invalid state " .. text .. " " .. subscription.state)
564 end
565 local refund = 0
566 if subscription.state == "ACTIVE" or subscription.state == "PAUSED" then
567 local start = subscription.state == "ACTIVE" and self.day or subscription.pause_start
568 refund = fee(subscription.plan, subscription.seats) *
569 (subscription.finish - start) // subscription.plan.length
570 end
571 self:balance_change(subscription.customer, refund, "cancel sub " .. subscription.number)
572 self:emit("CANCEL", "sub", subscription.number, "now", "credit", money(refund))
573 self:cancel_subscription(subscription)
574 end
575end

602function Billing:cmd_pay(args)

605 if invoice.outstanding == 0 then error_command("already paid " .. text) end
606 self:emit("PAY", "invoice", invoice.number, "amount", money(invoice.outstanding))

612function Billing:cmd_refund(args)
613 local text, amount = args[1], args[2]
614 local invoice = self:invoice(text)
615 local value = cents(amount)
616 if value == 0 then error_command("invalid amount") end
617 local maximum = invoice.total - invoice.refunded
618 if value > maximum then
619 error_command("refund exceeds " .. text .. " " .. money(maximum))
620 end
621 local applied = math.min(value, invoice.outstanding)
622 invoice.refunded = invoice.refunded + value
623 invoice.outstanding = invoice.outstanding - applied
624 self:balance_change(invoice.customer, value - applied, "refund invoice " .. invoice.number)
625 self:emit("REFUND", "invoice", invoice.number, "amount", amount,
626 "applied", money(applied), "balance", money(invoice.customer.balance))
627 if applied > 0 and invoice.outstanding == 0 then self:payment_complete(invoice) end
628end

655 local state = subscription.state
656 self:emit("SUB", subscription.number, subscription.customer.name,
657 subscription.plan.name, "seats", subscription.seats, state)
658 if state == "TRIALING" then
659 self:indented("trial until", subscription.trial_end)
660 end
661 if contains({"ACTIVE", "PAST_DUE", "UNPAID", "PAUSED"}, state) then
662 self:indented("period", subscription.start .. "-" .. subscription.finish)
663 end
664 if state == "PAUSED" then self:indented("paused since", subscription.pause_start) end
665 if subscription.pending then
666 self:indented("pending", subscription.pending[1].name,
667 "seats", subscription.pending[2])
668 end
669 if subscription.cancel_end then self:indented("cancel at end") end
670 if subscription.cancel_on then self:indented("cancel on", subscription.cancel_on) end
671 if state == "CANCELED" then self:indented("canceled on", subscription.canceled_on) end
672 if subscription.coupon then
673 if subscription.remaining == nil then
674 self:indented("coupon", subscription.coupon.name, "forever")
675 elseif subscription.remaining > 0 then
676 self:indented("coupon", subscription.coupon.name, "left", subscription.remaining)
677 end
678 end
679 if subscription.plan.mode and state ~= "CANCELED" then
680 self:indented("usage", subscription.usage)
681 end
682 local open = {}
683 for _, invoice in ipairs(subscription.invoices) do
684 if invoice.outstanding > 0 then open[#open + 1] = invoice.number end
685 end
686 if #open > 0 then
687 local parts = {"open"}
688 for _, number in ipairs(open) do parts[#parts + 1] = number end
689 self:indented(table.unpack(parts))
690 end
691end

693function Billing:cmd_preview(args)
694 local subscription = self:subscription(args[1],
695 {"TRIALING", "ACTIVE", "PAST_DUE", "UNPAID"})
696 local day = subscription.state == "TRIALING" and subscription.trial_end or
697 subscription.finish
698 local kind, items
699 if subscription.cancel_end then
700 kind = "final"
701 items = subscription.state == "TRIALING" and {} or usage_items(subscription)
702 elseif subscription.state == "TRIALING" then
703 kind = "first"
704 items = opening_items(subscription, day)
705 else
706 kind = "renewal"
707 local plan = subscription.pending and subscription.pending[1] or subscription.plan
708 local seats = subscription.pending and subscription.pending[2] or subscription.seats
709 items = usage_items(subscription)
710 items[#items + 1] = period_item(plan, seats, day)
711 end
712 self:emit("PREVIEW", "sub", subscription.number, "day", day, kind)
713 if #items == 0 then
714 self:indented("(none)")
715 return
716 end
717 self:emit_items(items)
718 local totals = self:invoice_totals(subscription, items)
719 self:emit_totals(totals)
720 self:indented("credit", money(totals[5]), "due", money(totals[6]))
721end

723function Billing:cmd_statement(args)
724 local customer = self:customer(args[1])
725 self:emit("STATEMENT", customer.name, "region", customer.region,
726 "tax", percentage(self.taxes[customer.region] or 0),

731 end
732 local outstanding = 0

735 "sub", invoice.subscription.number, invoice.kind,

752function Billing:cmd_mrr()
753 local by_plan = {}
754 for _, subscription in ipairs(self.subscriptions) do
755 if subscription.state == "ACTIVE" or subscription.state == "PAST_DUE" then
756 local price = fee(subscription.plan, subscription.seats)
757 local cycle = subscription.plan.cycle
758 local monthly = cycle == "MONTHLY" and price or
759 (cycle == "YEARLY" and price // 12 or price * 30 // 7)
760 local name = subscription.plan.name
761 local row = by_plan[name]
762 if not row then row = {0, 0}; by_plan[name] = row end
763 row[1] = row[1] + 1
764 row[2] = row[2] + monthly
765 end
766 end
767 local names = {}
768 for name in pairs(by_plan) do names[#names + 1] = name end
769 table.sort(names)
770 self:emit("MRR")
771 local count, total = 0, 0
772 for _, name in ipairs(names) do
773 local row = by_plan[name]
774 self:indented(name, "subs", row[1], "mrr", money(row[2]))
775 count, total = count + row[1], total + row[2]
776 end
777 self:indented("total subs", count, "mrr", money(total))
778end

780local function aging_parts(amounts)
781 local total = amounts[1] + amounts[2] + amounts[3]
782 return "0-30", money(amounts[1]), "31-60", money(amounts[2]),
783 "61+", money(amounts[3]), "sum", money(total)
784end

786function Billing:cmd_aging()
787 local by_customer = {}
788 local totals = {0, 0, 0}
789 for _, invoice in ipairs(self.invoices) do
790 if invoice.outstanding ~= 0 then
791 local age = self.day - invoice.day
792 local bucket = age <= 30 and 1 or (age <= 60 and 2 or 3)
793 local name = invoice.customer.name
794 if not by_customer[name] then by_customer[name] = {0, 0, 0} end
795 by_customer[name][bucket] = by_customer[name][bucket] + invoice.outstanding
796 totals[bucket] = totals[bucket] + invoice.outstanding
797 end
798 end
799 local names = {}
800 for name in pairs(by_customer) do names[#names + 1] = name end
801 table.sort(names)
802 self:emit("AGING", "day", self.day)
803 for _, name in ipairs(names) do
804 self:indented(name, aging_parts(by_customer[name]))
805 end
806 self:indented("total", aging_parts(totals))
807end

825 0, 0, 0, 0, 0, 0, 0, 0
826 for _, invoice in ipairs(self.invoices) do
827 subtotal = subtotal + invoice.subtotal
828 discount = discount + invoice.discount
829 tax = tax + invoice.tax
830 total = total + invoice.total
831 credit = credit + invoice.credit

833 refunded = refunded + invoice.refunded

835 end
836 local balances = 0
837 for _, customer in pairs(self.customers) do balances = balances + customer.balance end
838 self:emit("REVENUE")
839 self:indented("invoices", #self.invoices, "subtotal", money(subtotal),
840 "discount", money(discount), "tax", money(tax), "total", money(total))
841 self:indented("credit", money(credit), "paid", money(paid),
842 "refunded", money(refunded), "outstanding", money(outstanding),
843 "balances", money(balances))

891function Billing:execute(line)
892 local words = {}
893 for word in line:gmatch("%S+") do words[#words + 1] = word end
894 local command = table.remove(words, 1)
895 local ok, reason = pcall(function() handlers[command](self, words) end)
896 if not ok then
897 if type(reason) == "table" and reason.reason then
898 self:emit("ERROR", reason.reason)
899 else
900 error(reason)
901 end
902 end
903end

905local number_of_commands = tonumber(io.read("*l"))
906local billing = Billing.new()
907for _ = 1, number_of_commands do
908 billing:execute(io.read("*l"))
909end
910if #billing.output > 0 then
911 io.write(table.concat(billing.output, "\n"), "\n")
