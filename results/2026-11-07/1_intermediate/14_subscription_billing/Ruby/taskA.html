1LENGTH = {'WEEKLY' => 7, 'MONTHLY' => 30, 'YEARLY' => 360}.freeze
2Plan = Struct.new(:name, :price, :cycle, :length, :trial, :setup, :per_seat, :mode, :tiers)
3Customer = Struct.new(:name, :region, :method, :balance, :log)
4Coupon = Struct.new(:name, :kind, :value, :applications, :maximum, :redeemed)
5Subscription = Struct.new(:id, :customer, :plan, :seats, :state, :start, :finish, :trial_end,
6 :pause_start, :pending, :cancel_end, :cancel_on, :canceled_on,
7 :coupon, :left, :usage)
8class CommandError < StandardError; end
9
10def cents(text)
11 text.delete('.').to_i
12end
13
14def money(value)
15 (value.negative? ? '-' : '') + format('%d.%02d', value.abs / 100, value.abs % 100)
16end
17
18def percent(value)
19 money(value) + '%'
20end
21
22def fee(plan, seats)
23 plan.price * (plan.per_seat ? seats : 1)
24end
25
26def period(s)
27 "#{s.start}-#{s.finish}"
28end
29
30class Billing
31 attr_reader :output
32
33 def initialize
34 @day = 1
35 @plans = {}
36 @taxes = {}
37 @customers = {}
38 @coupons = {}
39 @subs = []
40 @invoices = []
41 @events = []
42 @output = []
43 end
44
45 def emit(indent, *parts)
46 @output << ' ' * indent + parts.join(' ')
47 end
48
49 def plan(name)
50 raise CommandError, "unknown plan #{name}" unless @plans.key?(name)
51 @plans[name]
52 end
53
54 def customer(name)
55 raise CommandError, "unknown customer #{name}" unless @customers.key?(name)
56 @customers[name]
57 end
58
59 def coupon(name)
60 c = @coupons[name]
61 raise CommandError, "unknown coupon #{name}" unless c
62 raise CommandError, "coupon exhausted #{name}" if c.maximum && c.redeemed == c.maximum
63 c
64 end
65
66 def sub(text)
67 id = text.to_i
68 raise CommandError, "unknown sub #{text}" unless (1..@subs.length).cover?(id)
69 @subs[id - 1]
70 end
71
72 def invoice(text)
73 id = text.to_i
74 raise CommandError, "unknown invoice #{text}" unless (1..@invoices.length).cover?(id)
75 @invoices[id - 1]
76 end
77
78 def state(s, allowed)
79 raise CommandError, "invalid state #{s.id} #{s.state}" unless allowed.include?(s.state)
80 end
81
82 def balance(c, description, change)
83 return if change.zero?
84 c.balance += change
85 c.log << [@day, description, change, c.balance]
86 end
87
88 def sub_invoices(s)
89 @invoices.select { |i| i[:sub].equal?(s) }
90 end
91
92 def change_state(s, target)
93 from = s.state
94 emit(2, 'day', @day, 'sub', s.id, from, '->', target)
95 @events << [@day, s.id, from, target]
96 sub_invoices(s).each { |i| i[:retries] = [] } if from == 'PAST_DUE'
97 s.state = target
98 end
99
100 def usage_item(s, close = false)
101 return [] if s.plan.mode.nil? || s.usage.zero?
102 p = s.plan
103 units = s.usage
104 charge = 0
105 lower = 0
106 p.tiers.each_with_index do |(limit, unit, flat), k|
107 last = k == p.tiers.length - 1
108 if p.mode == 'VOLUME'
109 if last || limit.nil? || limit >= units
110 charge = units * unit + flat
111 break
112 end
113 elsif lower < units
114 upper = last || limit.nil? ? units : [units, limit].min
115 charge += (upper - lower) * unit + flat
116 end
117 lower = limit unless limit.nil?
118 end
119 s.usage = 0 if close
120 [["usage #{p.name} #{units} units", charge]]
121 end
122
123 def charge_item(p, seats, start)
124 ["charge #{p.name} seats #{seats} #{start}-#{start + p.length}", fee(p, seats)]
125 end
126
127 def opening(s, start)
128 items = [charge_item(s.plan, s.seats, start)]
129 items << ["setup #{s.plan.name}", s.plan.setup] if s.plan.setup.positive?
130 items
131 end
132
133 def quote(s, items)
134 subtotal = items.sum(&:last)
135 discount = 0
136 if s.coupon && (s.left.nil? || s.left.positive?) && subtotal.positive?
137 discount = s.coupon.kind == 'PERCENT' ? subtotal * s.coupon.value / 10000 : [s.coupon.value, subtotal].min
138 end
139 tax = (subtotal - discount) * @taxes.fetch(s.customer.region, 0) / 10000
140 total = subtotal - discount + tax
141 credit = [s.customer.balance, total].min
142 {subtotal: subtotal, discount: discount, tax: tax, total: total, credit: credit, due: total - credit}
143 end
144
145 def issue(s, kind, items)
146 q = quote(s, items)
147 if s.coupon && (s.left.nil? || s.left.positive?) && q[:subtotal].positive? && !s.left.nil?
148 s.left -= 1
149 end
150 i = q.merge(id: @invoices.length + 1, sub: s, day: @day, kind: kind, items: items,
151 paid: 0, refunded: 0, outstanding: q[:due], attempts: 0, retries: [])
152 @invoices << i
153 balance(s.customer, "applied invoice #{i[:id]}", -q[:credit])
154 result = 'PAID'
155 if q[:due].positive?
156 if s.state == 'UNPAID'
157 result = 'OPEN'
158 else
159 i[:attempts] = 1
160 if s.customer.method == 'OK'
161 i[:paid] = q[:due]
162 i[:outstanding] = 0
163 else
164 result = 'FAILED'
165 end
166 end
167 end
168 emit(2, 'day', @day, 'invoice', i[:id], 'sub', s.id, kind,
169 'total', money(q[:total]), 'credit', money(q[:credit]), 'due', money(q[:due]), result)
170 if result == 'FAILED'
171 change_state(s, 'PAST_DUE') if s.state == 'ACTIVE'
172 i[:retries] = [@day + 3, @day + 5, @day + 7] if s.state == 'PAST_DUE'
173 end
174 end
175
176 def complete(i)
177 i[:retries] = []
178 s = i[:sub]
179 if %w[PAST_DUE UNPAID].include?(s.state) && sub_invoices(s).all? { |i| i[:outstanding].zero? }
180 s.cancel_on = nil
181 change_state(s, 'ACTIVE')
182 end
183 end
184
185 def cancel(s)
186 change_state(s, 'CANCELED')
187 s.canceled_on = @day
188 s.pending = nil
189 s.cancel_end = false
190 s.cancel_on = nil
191 items = usage_item(s, true)
192 issue(s, 'final', items) unless items.empty?
193 end
194
195 def advance(target)
196 while @day < target
197 @day += 1
198 @subs.each do |s|
199 next if %w[PAUSED CANCELED].include?(s.state)
200 sub_invoices(s).each do |i|
201 next if i[:retries].first != @day
202 i[:retries].shift
203 i[:attempts] += 1
204 if s.customer.method == 'OK'
205 i[:paid] += i[:outstanding]
206 i[:outstanding] = 0
207 emit(2, 'day', @day, 'retry', i[:attempts] - 1, 'invoice', i[:id], 'PAID')
208 complete(i)
209 else
210 emit(2, 'day', @day, 'retry', i[:attempts] - 1, 'invoice', i[:id], 'FAILED')
211 if i[:retries].empty?
212 change_state(s, 'UNPAID')
213 s.cancel_on = @day + 14
214 end
215 end
216 end
217 cancel(s) if s.state == 'UNPAID' && s.cancel_on == @day
218 if s.state == 'TRIALING' && s.trial_end == @day
219 if s.cancel_end
220 cancel(s)
221 else
222 change_state(s, 'ACTIVE')
223 s.start = @day
224 s.finish = @day + s.plan.length
225 issue(s, 'first', opening(s, @day))
226 end
227 end
228 if %w[ACTIVE PAST_DUE UNPAID].include?(s.state) && s.finish == @day
229 if s.cancel_end
230 cancel(s)
231 else
232 items = usage_item(s, true)
233 if s.pending
234 p, seats = s.pending
235 emit(2, 'day', @day, 'sub', s.id, s.plan.name, 'seats', s.seats, '->', p.name, 'seats', seats)
236 s.plan = p
237 s.seats = seats
238 s.pending = nil
239 end
240 s.start = @day
241 s.finish = @day + s.plan.length
242 items << charge_item(s.plan, s.seats, @day)
243 issue(s, 'renewal', items)
244 end
245 end
246 end
247 end
248 end
249
250 def totals(q)
251 emit(2, 'subtotal', money(q[:subtotal]), 'discount', money(q[:discount]), 'tax', money(q[:tax]), 'total', money(q[:total]))
252 end
253
254 def show_sub(s)
255 emit(0, 'SUB', s.id, s.customer.name, s.plan.name, 'seats', s.seats, s.state)
256 emit(2, 'trial', 'until', s.trial_end) if s.state == 'TRIALING'
257 emit(2, 'period', period(s)) if %w[ACTIVE PAST_DUE UNPAID PAUSED].include?(s.state)
258 emit(2, 'paused', 'since', s.pause_start) if s.state == 'PAUSED'
259 emit(2, 'pending', s.pending[0].name, 'seats', s.pending[1]) if s.pending
260 emit(2, 'cancel', 'at', 'end') if s.cancel_end
261 emit(2, 'cancel', 'on', s.cancel_on) if s.cancel_on
262 emit(2, 'canceled', 'on', s.canceled_on) if s.state == 'CANCELED'
263 if s.coupon && (s.left.nil? || s.left.positive?)
264 emit(2, 'coupon', s.coupon.name, *(s.left.nil? ? ['forever'] : ['left', s.left]))
265 end
266 emit(2, 'usage', s.usage) if s.plan.mode && s.state != 'CANCELED'
267 open = sub_invoices(s).select { |i| i[:outstanding].positive? }
268 emit(2, 'open', *open.map { |i| i[:id] }) unless open.empty?
269 end
270
271 def preview(s)
272 state(s, %w[TRIALING ACTIVE PAST_DUE UNPAID])
273 day = s.state == 'TRIALING' ? s.trial_end : s.finish
274 if s.cancel_end
275 kind = 'final'
276 items = s.state == 'TRIALING' ? [] : usage_item(s)
277 elsif s.state == 'TRIALING'
278 kind = 'first'
279 items = opening(s, day)
280 else
281 kind = 'renewal'
282 items = usage_item(s)
283 p, seats = s.pending || [s.plan, s.seats]
284 items << charge_item(p, seats, day)
285 end
286 emit(0, 'PREVIEW', 'sub', s.id, 'day', day, kind)
287 items.each { |description, amount| emit(2, description, money(amount)) }
288 if items.empty?
289 emit(2, '(none)')
290 else
291 q = quote(s, items)
292 totals(q)
293 emit(2, 'credit', money(q[:credit]), 'due', money(q[:due]))
294 end
295 end
296
297 def aging_line(label, row)
298 emit(2, label, '0-30', money(row[0]), '31-60', money(row[1]), '61+', money(row[2]), 'sum', money(row.sum))
299 end

301 def command(cmd, a)
302 case cmd
303 when 'PLAN'
304 raise CommandError, "duplicate plan #{a[0]}" if @plans.key?(a[0])
305 p = Plan.new(a[0], cents(a[1]), a[2], LENGTH[a[2]], 0, 0, false, nil, [])
306 i = 3
307 while i < a.length
308 case a[i]
309 when 'TRIAL'
310 i += 1
311 p.trial = a[i].to_i
312 when 'SETUP'
313 i += 1
314 p.setup = cents(a[i])
315 when 'PERSEAT'
316 p.per_seat = true
317 else
318 p.mode = a[i]
319 end
320 i += 1
321 end
322 @plans[p.name] = p
323 parts = ['PLAN', p.name, p.cycle, money(p.price)]
324 parts.push('trial', p.trial) if p.trial.positive?
325 parts.push('setup', money(p.setup)) if p.setup.positive?
326 parts << 'perseat' if p.per_seat
327 parts << p.mode.downcase if p.mode
328 emit(0, *parts)
329 when 'TIER'
330 p = plan(a[0])
331 raise CommandError, "not metered #{a[0]}" if p.mode.nil?
332 limit = a[1] == 'INF' ? nil : a[1].to_i
333 last = p.tiers.last
334 raise CommandError, "tier order #{a[0]}" if last && (last[0].nil? || (!limit.nil? && limit <= last[0]))
335 p.tiers << [limit, cents(a[2]), cents(a[3])]
336 emit(0, 'TIER', p.name, p.tiers.length, 'upto', a[1], 'unit', money(cents(a[2])), 'flat', money(cents(a[3])))
337 when 'TAX'
338 @taxes[a[0]] = a[1].to_i
339 emit(0, 'TAX', a[0], percent(a[1].to_i))
340 when 'CUSTOMER'
341 raise CommandError, "duplicate customer #{a[0]}" if @customers.key?(a[0])
342 @customers[a[0]] = Customer.new(a[0], a[1], 'OK', 0, [])
343 emit(0, 'CUSTOMER', a[0], 'region', a[1])
344 when 'COUPON'
345 raise CommandError, "duplicate coupon #{a[0]}" if @coupons.key?(a[0])
346 applications = a[3] == 'FOREVER' ? nil : a[3] == 'ONCE' ? 1 : a[4].to_i
347 max_index = a.index('MAX')
348 maximum = max_index ? a[max_index + 1].to_i : nil
349 value = a[1] == 'PERCENT' ? a[2].to_i : cents(a[2])
350 @coupons[a[0]] = Coupon.new(a[0], a[1], value, applications, maximum, 0)
351 parts = ['COUPON', a[0], a[1].downcase, a[1] == 'PERCENT' ? percent(value) : money(value), a[3].downcase]
352 parts << applications if a[3] == 'REPEATING'
353 parts.push('max', maximum) if maximum
354 emit(0, *parts)
355 when 'PAYMETHOD'
356 c = customer(a[0])
357 c.method = a[1]
358 emit(0, 'PAYMETHOD', c.name, c.method)
359 when 'CREDIT'
360 c = customer(a[0])
361 amount = cents(a[1])
362 raise CommandError, 'invalid amount' if amount.zero?
363 balance(c, 'grant', amount)
364 emit(0, 'CREDIT', c.name, 'amount', money(amount), 'balance', money(c.balance))
365 when 'DAY'
366 day = a[0].to_i
367 raise CommandError, "invalid day #{a[0]}" if day <= @day
368 emit(0, 'DAY', day)
369 advance(day)
370 when 'SUBSCRIBE'
371 c = customer(a[0])
372 p = plan(a[1])
373 seat_index = a.index('SEATS')
374 raise CommandError, "not per-seat #{p.name}" if seat_index && !p.per_seat
375 seats = seat_index ? a[seat_index + 1].to_i : 1
376 raise CommandError, 'invalid seats' if seats.zero?
377 coupon_index = a.index('COUPON')
378 discount = coupon_index ? coupon(a[coupon_index + 1]) : nil
379 discount.redeemed += 1 if discount
380 s = Subscription.new(@subs.length + 1, c, p, seats, 'NONE', 0, 0, nil,
381 nil, nil, false, nil, nil, discount, discount&.applications, 0)
382 @subs << s
383 emit(0, 'SUBSCRIBE', 'sub', s.id, c.name, p.name, 'seats', seats, *(discount ? ['coupon', discount.name] : []))
384 if p.trial.positive?
385 s.trial_end = @day + p.trial
386 change_state(s, 'TRIALING')
387 else
388 change_state(s, 'ACTIVE')
389 s.start = @day
390 s.finish = @day + p.length
391 issue(s, 'first', opening(s, @day))
392 end
393 when 'CHANGE'
394 s = sub(a[0])
395 state(s, %w[ACTIVE TRIALING])
396 p = s.plan
397 seats = s.seats
398 if a[1] == 'PLAN'
399 p = plan(a[2])
400 seats = 1 unless p.per_seat
401 else
402 raise CommandError, "not per-seat #{p.name}" unless p.per_seat
403 seats = a[2].to_i
404 raise CommandError, 'invalid seats' if seats.zero?
405 end
406 raise CommandError, "no change #{s.id}" if p.equal?(s.plan) && seats == s.seats
407 s.pending = nil
408 old = s.plan
409 old_seats = s.seats
410 old_fee = fee(old, old_seats)
411 new_fee = fee(p, seats)
412 head = ['CHANGE', 'sub', s.id, old.name, 'seats', old_seats, '->', p.name, 'seats', seats]
413 if s.state == 'TRIALING'
414 s.plan = p
415 s.seats = seats
416 emit(0, *head, 'now')
417 elsif new_fee * old.length > old_fee * p.length
418 emit(0, *head, 'now')
419 days = s.finish - @day
420 items = usage_item(s, true)
421 items << ["unused #{old.name} seats #{old_seats} #{days} days", -(old_fee * days / old.length)]
422 items << ["remaining #{p.name} seats #{seats} #{days} days", new_fee * days / p.length]
423 s.plan = p
424 s.seats = seats
425 issue(s, 'upgrade', items)
426 else
427 s.pending = [p, seats]
428 emit(0, *head, 'at', s.finish)
429 end
430 when 'PAUSE'
431 s = sub(a[0])
432 state(s, ['ACTIVE'])
433 emit(0, 'PAUSE', 'sub', s.id)
434 s.pause_start = @day
435 change_state(s, 'PAUSED')
436 when 'RESUME'
437 s = sub(a[0])
438 state(s, ['PAUSED'])
439 shift = @day - s.pause_start
440 s.finish += shift
441 emit(0, 'RESUME', 'sub', s.id, 'shift', shift, 'period', period(s))
442 s.pause_start = nil
443 change_state(s, 'ACTIVE')
444 when 'CANCEL'
445 s = sub(a[0])
446 if a[1] == 'NOW'
447 state(s, %w[TRIALING ACTIVE PAST_DUE UNPAID PAUSED])
448 remaining = s.state == 'ACTIVE' ? s.finish - @day : s.state == 'PAUSED' ? s.finish - s.pause_start : 0
449 refund = fee(s.plan, s.seats) * remaining / s.plan.length
450 balance(s.customer, "cancel sub #{s.id}", refund)
451 emit(0, 'CANCEL', 'sub', s.id, 'now', 'credit', money(refund))
452 cancel(s)
453 else
454 state(s, %w[ACTIVE TRIALING])
455 raise CommandError, "already canceling #{s.id}" if s.cancel_end
456 s.cancel_end = true
457 emit(0, 'CANCEL', 'sub', s.id, 'end', s.state == 'TRIALING' ? s.trial_end : s.finish)
458 end
459 when 'USAGE'
460 s = sub(a[0])
461 state(s, %w[ACTIVE PAST_DUE])
462 raise CommandError, "not metered #{s.plan.name}" unless s.plan.mode
463 units = a[1].to_i
464 raise CommandError, 'invalid units' if units.zero?
465 s.usage += units
466 emit(0, 'USAGE', 'sub', s.id, 'units', units, 'total', s.usage)
467 when 'APPLY'
468 s = sub(a[0])
469 state(s, %w[TRIALING ACTIVE PAST_DUE UNPAID PAUSED])
470 discount = coupon(a[1])
471 discount.redeemed += 1
472 s.coupon = discount
473 s.left = discount.applications
474 emit(0, 'APPLY', 'sub', s.id, 'coupon', discount.name)
475 when 'PAY'
476 i = invoice(a[0])
477 raise CommandError, "already paid #{a[0]}" if i[:outstanding].zero?
478 emit(0, 'PAY', 'invoice', i[:id], 'amount', money(i[:outstanding]))
479 i[:paid] += i[:outstanding]
480 i[:outstanding] = 0
481 complete(i)
482 when 'REFUND'
483 i = invoice(a[0])
484 amount = cents(a[1])
485 raise CommandError, 'invalid amount' if amount.zero?
486 raise CommandError, "refund exceeds #{a[0]} #{money(i[:total] - i[:refunded])}" if amount > i[:total] - i[:refunded]
487 applied = [amount, i[:outstanding]].min
488 i[:refunded] += amount
489 i[:outstanding] -= applied
490 balance(i[:sub].customer, "refund invoice #{i[:id]}", amount - applied)
491 emit(0, 'REFUND', 'invoice', i[:id], 'amount', money(amount), 'applied', money(applied), 'balance', money(i[:sub].customer.balance))
492 complete(i) if applied.positive? && i[:outstanding].zero?
493 when 'INVOICE'
494 i = invoice(a[0])
495 emit(0, 'INVOICE', i[:id], 'sub', i[:sub].id, i[:sub].customer.name, 'day', i[:day], i[:kind], i[:outstanding].positive? ? 'OPEN' : 'PAID')
496 i[:items].each { |description, amount| emit(2, description, money(amount)) }
497 totals(i)
498 emit(2, 'credit', money(i[:credit]), 'due', money(i[:due]), 'paid', money(i[:paid]),
499 'refunded', money(i[:refunded]), 'outstanding', money(i[:outstanding]))
500 when 'SUB'
501 show_sub(sub(a[0]))
502 when 'PREVIEW'
503 preview(sub(a[0]))
504 when 'STATEMENT', 'LEDGER'
505 c = customer(a[0])
506 if cmd == 'LEDGER'
507 emit(0, 'LEDGER', c.name, 'balance', money(c.balance))
508 c.log.each do |day, description, change, after|
509 emit(2, 'day', day, description, (change.positive? ? '+' : '') + money(change), 'balance', money(after))
510 end
511 emit(2, '(none)') if c.log.empty?
512 else
513 emit(0, 'STATEMENT', c.name, 'region', c.region, 'tax', percent(@taxes.fetch(c.region, 0)), 'method', c.method, 'balance', money(c.balance))
514 @subs.select { |s| s.customer.equal?(c) }.each { |s| emit(2, 'sub', s.id, s.plan.name, s.state) }
515 invoices = @invoices.select { |i| i[:sub].customer.equal?(c) }
516 invoices.each do |i|
517 emit(2, 'invoice', i[:id], 'day', i[:day], 'sub', i[:sub].id, i[:kind], 'total', money(i[:total]), 'outstanding', money(i[:outstanding]))
518 end
519 emit(2, 'outstanding', money(invoices.sum { |i| i[:outstanding] }))
520 end
521 when 'MRR'
522 active = @subs.select { |s| %w[ACTIVE PAST_DUE].include?(s.state) }
523 rows = active.group_by { |s| s.plan.name }.transform_values do |subs|
524 monthly = subs.sum do |s|
525 amount = fee(s.plan, s.seats)
526 s.plan.cycle == 'MONTHLY' ? amount : s.plan.cycle == 'YEARLY' ? amount / 12 : amount * 30 / 7
527 end
528 [subs.length, monthly]
529 end
530 emit(0, 'MRR')
531 rows.keys.sort.each { |name| emit(2, name, 'subs', rows[name][0], 'mrr', money(rows[name][1])) }
532 emit(2, 'total', 'subs', rows.values.sum(&:first), 'mrr', money(rows.values.sum(&:last)))
533 when 'AGING'
534 rows = {}
535 total = [0, 0, 0]
536 @invoices.each do |i|
537 next if i[:outstanding].zero?
538 row = (rows[i[:sub].customer.name] ||= [0, 0, 0])
539 age = @day - i[:day]
540 bucket = age <= 30 ? 0 : age <= 60 ? 1 : 2
541 row[bucket] += i[:outstanding]
542 total[bucket] += i[:outstanding]
543 end
544 emit(0, 'AGING', 'day', @day)
545 rows.keys.sort.each { |name| aging_line(name, rows[name]) }
546 aging_line('total', total)
547 when 'DUNNING'
548 emit(0, 'DUNNING')
549 invoices = @invoices.reject { |i| i[:retries].empty? }
550 invoices.each do |i|
551 emit(2, 'invoice', i[:id], 'sub', i[:sub].id, i[:sub].customer.name,
552 'outstanding', money(i[:outstanding]), 'attempts', i[:attempts], 'next', i[:retries].first)
553 end
554 emit(2, '(none)') if invoices.empty?
555 when 'REVENUE'
556 total = ->(field) { money(@invoices.sum { |i| i[field] }) }
557 emit(0, 'REVENUE')
558 emit(2, 'invoices', @invoices.length, 'subtotal', total.call(:subtotal), 'discount', total.call(:discount),
559 'tax', total.call(:tax), 'total', total.call(:total))
560 emit(2, 'credit', total.call(:credit), 'paid', total.call(:paid), 'refunded', total.call(:refunded),
561 'outstanding', total.call(:outstanding), 'balances', money(@customers.values.sum(&:balance)))
562 when 'EVENTS'
563 s = a.empty? ? nil : sub(a[0])
564 emit(0, 'EVENTS', *(s ? ['sub', s.id] : []))
565 events = @events.select { |_, id, _, _| s.nil? || id == s.id }
566 events.each { |day, id, from, to| emit(2, 'day', day, 'sub', id, from, '->', to) }
567 emit(2, '(none)') if events.empty?
568 end
569 end
