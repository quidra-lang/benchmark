3type Tier = {limit: number | null; unit: bigint; flat: bigint};
4type Plan = {name: string; price: bigint; cycle: string; length: number; trial: number; setup: bigint; perSeat: boolean; mode: string | null; tiers: Tier[]};
5type Customer = {name: string; region: string; method: string; balance: bigint; log: {day: number; description: string; change: bigint; balance: bigint}[]};
6type Coupon = {name: string; kind: string; value: bigint; applications: number | null; maximum: number | null; redeemed: number};
7type Subscription = {id: number; customer: Customer; plan: Plan; seats: number; state: string; start: number; end: number; trialEnd: number | null; pauseStart: number | null; pending: {plan: Plan; seats: number} | null; cancelEnd: boolean; cancelOn: number | null; canceledOn: number | null; coupon: Coupon | null; left: number | null; usage: number};
8type Item = {description: string; amount: bigint};
9type Quote = {subtotal: bigint; discount: bigint; tax: bigint; total: bigint; credit: bigint; due: bigint};
10type Invoice = Quote & {id: number; sub: Subscription; day: number; kind: string; items: Item[]; paid: bigint; refunded: bigint; outstanding: bigint; attempts: number; retries: number[]};
11type EventRow = {day: number; id: number; from: string; to: string};
12const LENGTH: Record<string, number> = {WEEKLY: 7, MONTHLY: 30, YEARLY: 360};
13const cents = (s: string): bigint => BigInt(s.replace(".", ""));
14const min = (a: bigint, b: bigint): bigint => a < b ? a : b;
15const fee = (p: Plan, seats: number): bigint => p.price * BigInt(p.perSeat ? seats : 1);
16const period = (s: Subscription): string => s.start + "-" + s.end;
17const sum = (v: bigint[]): bigint => v.reduce((a, b) => a + b, 0n);
18function money(v: bigint): string {
19 const a = v < 0n ? -v : v;
20 return (v < 0n ? "-" : "") + a / 100n + "." + String(a % 100n).padStart(2, "0");
21}
22const percent = (v: bigint): string => money(v) + "%";
23class CommandError extends Error {}

85 changeState(s: Subscription, to: string): void {
86 const from = s.state;
87 this.emit(2, "day", this.day, "sub", s.id, from, "->", to);
88 this.events.push({day: this.day, id: s.id, from, to});
89 if (from === "PAST_DUE") {
90 for (const i of this.subInvoices(s)) i.retries = [];
91 }

95 usageItem(s: Subscription, close = false): Item[] {
96 if (s.plan.mode === null || s.usage === 0) return [];
97 const p = s.plan;
98 const units = s.usage;
99 let charge = 0n;
100 let lower = 0;
101 for (let k = 0; k < p.tiers.length; k++) {
102 const tier = p.tiers[k];
103 const last = k === p.tiers.length - 1;
104 if (p.mode === "VOLUME") {
105 if (last || tier.limit === null || tier.limit >= units) {
106 charge = BigInt(units) * tier.unit + tier.flat;
107 break;
108 }
109 } else if (lower < units) {
110 const upper = last || tier.limit === null ? units : Math.min(units, tier.limit);
111 charge += BigInt(upper - lower) * tier.unit + tier.flat;
112 }
113 if (tier.limit !== null) lower = tier.limit;
114 }
115 if (close) s.usage = 0;
116 return [{description: "usage " + p.name + " " + units + " units", amount: charge}];
117 }

119 chargeItem(p: Plan, seats: number, start: number): Item {
120 return {description: "charge " + p.name + " seats " + seats + " " + start + "-" + (start + p.length), amount: fee(p, seats)};
121 }

123 opening(s: Subscription, start: number): Item[] {
124 const items = [this.chargeItem(s.plan, s.seats, start)];
125 if (s.plan.setup > 0n) items.push({description: "setup " + s.plan.name, amount: s.plan.setup});
126 return items;
127 }

129 quote(s: Subscription, items: Item[]): Quote {
130 const subtotal = sum(items.map(i => i.amount));
131 let discount = 0n;
132 if (s.coupon !== null && (s.left === null || s.left > 0) && subtotal > 0n) {
133 discount = s.coupon.kind === "PERCENT" ? subtotal * s.coupon.value / 10000n : min(s.coupon.value, subtotal);
134 }
135 const tax = (subtotal - discount) * (this.taxes.get(s.customer.region) ?? 0n) / 10000n;
136 const total = subtotal - discount + tax;
137 const credit = min(s.customer.balance, total);
138 return {subtotal, discount, tax, total, credit, due: total - credit};
139 }
140
141 issue(s: Subscription, kind: string, items: Item[]): void {
142 const q = this.quote(s, items);
143 if (s.coupon !== null && (s.left === null || s.left > 0) && q.subtotal > 0n && s.left !== null) s.left--;
144 const i: Invoice = {...q, id: this.invoices.length + 1, sub: s, day: this.day, kind, items,
145 paid: 0n, refunded: 0n, outstanding: q.due, attempts: 0, retries: []};
146 this.invoices.push(i);
147 this.balance(s.customer, "applied invoice " + i.id, -q.credit);
148 let result = "PAID";
149 if (q.due > 0n) {
150 if (s.state === "UNPAID") result = "OPEN";
151 else {
152 i.attempts = 1;
153 if (s.customer.method === "OK") {
154 i.paid = q.due;
155 i.outstanding = 0n;
156 } else result = "FAILED";
157 }
158 }
159 this.emit(2, "day", this.day, "invoice", i.id, "sub", s.id, kind,
160 "total", money(q.total), "credit", money(q.credit), "due", money(q.due), result);
161 if (result === "FAILED") {
162 if (s.state === "ACTIVE") this.changeState(s, "PAST_DUE");
163 if (s.state === "PAST_DUE") i.retries = [this.day + 3, this.day + 5, this.day + 7];
164 }
165 }
166
167 complete(i: Invoice): void {
168 i.retries = [];
169 const s = i.sub;
170 if (["PAST_DUE", "UNPAID"].includes(s.state) && this.subInvoices(s).every(i => i.outstanding === 0n)) {
171 s.cancelOn = null;
172 this.changeState(s, "ACTIVE");
173 }
174 }
175
176 cancel(s: Subscription): void {
177 this.changeState(s, "CANCELED");
178 s.canceledOn = this.day;
179 s.pending = null;
180 s.cancelEnd = false;
181 s.cancelOn = null;
182 const items = this.usageItem(s, true);
183 if (items.length) this.issue(s, "final", items);
184 }
185
186 advance(target: number): void {
187 while (this.day < target) {
188 this.day++;
189 for (const s of this.subs) {
190 if (["PAUSED", "CANCELED"].includes(s.state)) continue;
191 for (const i of this.subInvoices(s)) {
192 if (i.retries[0] !== this.day) continue;
193 i.retries.shift();
194 i.attempts++;
195 if (s.customer.method === "OK") {
196 i.paid += i.outstanding;
197 i.outstanding = 0n;
198 this.emit(2, "day", this.day, "retry", i.attempts - 1, "invoice", i.id, "PAID");
199 this.complete(i);
200 } else {
201 this.emit(2, "day", this.day, "retry", i.attempts - 1, "invoice", i.id, "FAILED");
202 if (!i.retries.length) {
203 this.changeState(s, "UNPAID");
204 s.cancelOn = this.day + 14;
205 }
206 }
207 }
208 if (s.state === "UNPAID" && s.cancelOn === this.day) this.cancel(s);
209 if (s.state === "TRIALING" && s.trialEnd === this.day) {
210 if (s.cancelEnd) this.cancel(s);
211 else {
212 this.changeState(s, "ACTIVE");
213 s.start = this.day;
214 s.end = this.day + s.plan.length;
215 this.issue(s, "first", this.opening(s, this.day));
216 }
217 }
218 if (["ACTIVE", "PAST_DUE", "UNPAID"].includes(s.state) && s.end === this.day) {
219 if (s.cancelEnd) this.cancel(s);
220 else {
221 const items = this.usageItem(s, true);
222 if (s.pending) {
223 this.emit(2, "day", this.day, "sub", s.id, s.plan.name, "seats", s.seats,
224 "->", s.pending.plan.name, "seats", s.pending.seats);
225 s.plan = s.pending.plan;
226 s.seats = s.pending.seats;
227 s.pending = null;
228 }
229 s.start = this.day;
230 s.end = this.day + s.plan.length;
231 items.push(this.chargeItem(s.plan, s.seats, this.day));
232 this.issue(s, "renewal", items);
233 }
234 }
235 }
236 }

243 showSub(s: Subscription): void {
244 this.emit(0, "SUB", s.id, s.customer.name, s.plan.name, "seats", s.seats, s.state);
245 if (s.state === "TRIALING") this.emit(2, "trial", "until", s.trialEnd!);
246 if (["ACTIVE", "PAST_DUE", "UNPAID", "PAUSED"].includes(s.state)) this.emit(2, "period", period(s));
247 if (s.state === "PAUSED") this.emit(2, "paused", "since", s.pauseStart!);
248 if (s.pending) this.emit(2, "pending", s.pending.plan.name, "seats", s.pending.seats);
249 if (s.cancelEnd) this.emit(2, "cancel", "at", "end");
250 if (s.cancelOn !== null) this.emit(2, "cancel", "on", s.cancelOn);
251 if (s.state === "CANCELED") this.emit(2, "canceled", "on", s.canceledOn!);
252 if (s.coupon && (s.left === null || s.left > 0)) this.emit(2, "coupon", s.coupon.name, ...(s.left === null ? ["forever"] : ["left", s.left]));
253 if (s.plan.mode !== null && s.state !== "CANCELED") this.emit(2, "usage", s.usage);
254 const open = this.subInvoices(s).filter(i => i.outstanding > 0n);
255 if (open.length) this.emit(2, "open", ...open.map(i => i.id));
256 }

258 preview(s: Subscription): void {
259 this.state(s, ["TRIALING", "ACTIVE", "PAST_DUE", "UNPAID"]);
260 const day = s.state === "TRIALING" ? s.trialEnd! : s.end;
261 let kind: string;
262 let items: Item[];
263 if (s.cancelEnd) {
264 kind = "final";
265 items = s.state === "TRIALING" ? [] : this.usageItem(s);
266 } else if (s.state === "TRIALING") {
267 kind = "first";
268 items = this.opening(s, day);
269 } else {
270 kind = "renewal";
271 items = this.usageItem(s);
272 const p = s.pending?.plan ?? s.plan;
273 const seats = s.pending?.seats ?? s.seats;
274 items.push(this.chargeItem(p, seats, day));
275 }
276 this.emit(0, "PREVIEW", "sub", s.id, "day", day, kind);
277 for (const item of items) this.emit(2, item.description, money(item.amount));
278 if (!items.length) this.emit(2, "(none)");
279 else {
280 const q = this.quote(s, items);
281 this.totals(q);
282 this.emit(2, "credit", money(q.credit), "due", money(q.due));
283 }

286 command(cmd: string, a: string[]): void {
287 switch (cmd) {
288 case "PLAN": {
289 if (this.plans.has(a[0])) throw new CommandError("duplicate plan " + a[0]);
290 const p: Plan = {name: a[0], price: cents(a[1]), cycle: a[2], length: LENGTH[a[2]],
291 trial: 0, setup: 0n, perSeat: false, mode: null, tiers: []};
292 for (let i = 3; i < a.length; i++) {
293 if (a[i] === "TRIAL") p.trial = Number(a[++i]);
294 else if (a[i] === "SETUP") p.setup = cents(a[++i]);
295 else if (a[i] === "PERSEAT") p.perSeat = true;
296 else p.mode = a[i];
297 }
298 this.plans.set(p.name, p);
299 const parts: (string | number)[] = ["PLAN", p.name, p.cycle, money(p.price)];
300 if (p.trial) parts.push("trial", p.trial);
301 if (p.setup > 0n) parts.push("setup", money(p.setup));
302 if (p.perSeat) parts.push("perseat");
303 if (p.mode) parts.push(p.mode.toLowerCase());
304 this.emit(0, ...parts);
305 break;
306 }
307 case "TIER": {
308 const p = this.plan(a[0]);
309 if (p.mode === null) throw new CommandError("not metered " + a[0]);
310 const limit = a[1] === "INF" ? null : Number(a[1]);
311 const last = p.tiers[p.tiers.length - 1];
312 if (last && (last.limit === null || (limit !== null && limit <= last.limit))) throw new CommandError("tier order " + a[0]);
313 p.tiers.push({limit, unit: cents(a[2]), flat: cents(a[3])});
314 this.emit(0, "TIER", p.name, p.tiers.length, "upto", a[1], "unit", money(cents(a[2])), "flat", money(cents(a[3])));
315 break;
316 }
317 case "TAX":
318 this.taxes.set(a[0], BigInt(a[1]));
319 this.emit(0, "TAX", a[0], percent(BigInt(a[1])));
320 break;
321 case "CUSTOMER":
322 if (this.customers.has(a[0])) throw new CommandError("duplicate customer " + a[0]);
323 this.customers.set(a[0], {name: a[0], region: a[1], method: "OK", balance: 0n, log: []});
324 this.emit(0, "CUSTOMER", a[0], "region", a[1]);
325 break;
326 case "COUPON": {
327 if (this.coupons.has(a[0])) throw new CommandError("duplicate coupon " + a[0]);
328 const applications = a[3] === "FOREVER" ? null : a[3] === "ONCE" ? 1 : Number(a[4]);
329 const maxIndex = a.indexOf("MAX");
330 const maximum = maxIndex < 0 ? null : Number(a[maxIndex + 1]);
331 const value = a[1] === "PERCENT" ? BigInt(a[2]) : cents(a[2]);
332 this.coupons.set(a[0], {name: a[0], kind: a[1], value, applications, maximum, redeemed: 0});
333 this.emit(0, "COUPON", a[0], a[1].toLowerCase(), a[1] === "PERCENT" ? percent(value) : money(value),
334 a[3].toLowerCase(), ...(a[3] === "REPEATING" ? [applications!] : []), ...(maximum === null ? [] : ["max", maximum]));
335 break;
336 }
337 case "PAYMETHOD": {
338 const c = this.customer(a[0]);
339 c.method = a[1];
340 this.emit(0, "PAYMETHOD", c.name, c.method);
341 break;
342 }
343 case "CREDIT": {
344 const c = this.customer(a[0]);
345 const amount = cents(a[1]);
346 if (amount === 0n) throw new CommandError("invalid amount");
347 this.balance(c, "grant", amount);
348 this.emit(0, "CREDIT", c.name, "amount", money(amount), "balance", money(c.balance));
349 break;
350 }
351 case "DAY": {
352 const day = Number(a[0]);
353 if (day <= this.day) throw new CommandError("invalid day " + a[0]);
354 this.emit(0, "DAY", day);
355 this.advance(day);
356 break;
357 }
358 case "SUBSCRIBE": {
359 const c = this.customer(a[0]);
360 const p = this.plan(a[1]);
361 const seatIndex = a.indexOf("SEATS");
362 if (seatIndex >= 0 && !p.perSeat) throw new CommandError("not per-seat " + p.name);
363 const seats = seatIndex < 0 ? 1 : Number(a[seatIndex + 1]);
364 if (seats === 0) throw new CommandError("invalid seats");
365 const couponIndex = a.indexOf("COUPON");
366 const coupon = couponIndex < 0 ? null : this.coupon(a[couponIndex + 1]);
367 if (coupon) coupon.redeemed++;
368 const s: Subscription = {id: this.subs.length + 1, customer: c, plan: p, seats, state: "NONE",
369 start: 0, end: 0, trialEnd: null, pauseStart: null, pending: null, cancelEnd: false,
370 cancelOn: null, canceledOn: null, coupon, left: coupon?.applications ?? null, usage: 0};
371 this.subs.push(s);
372 this.emit(0, "SUBSCRIBE", "sub", s.id, c.name, p.name, "seats", seats, ...(coupon ? ["coupon", coupon.name] : []));
373 if (p.trial) {
374 s.trialEnd = this.day + p.trial;
375 this.changeState(s, "TRIALING");
376 } else {
377 this.changeState(s, "ACTIVE");
378 s.start = this.day;
379 s.end = this.day + p.length;
380 this.issue(s, "first", this.opening(s, this.day));
381 }
382 break;
383 }
384 case "CHANGE": {
385 const s = this.sub(a[0]);
386 this.state(s, ["ACTIVE", "TRIALING"]);
387 let p = s.plan;
388 let seats = s.seats;
389 if (a[1] === "PLAN") {
390 p = this.plan(a[2]);
391 if (!p.perSeat) seats = 1;
392 } else {
393 if (!p.perSeat) throw new CommandError("not per-seat " + p.name);
394 seats = Number(a[2]);
395 if (seats === 0) throw new CommandError("invalid seats");
396 }
397 if (p === s.plan && seats === s.seats) throw new CommandError("no change " + s.id);
398 s.pending = null;
399 const old = s.plan;
400 const oldSeats = s.seats;
401 const oldFee = fee(old, oldSeats);
402 const newFee = fee(p, seats);
403 const head: (string | number)[] = ["CHANGE", "sub", s.id, old.name, "seats", oldSeats, "->", p.name, "seats", seats];
404 if (s.state === "TRIALING") {
405 s.plan = p;
406 s.seats = seats;
407 this.emit(0, ...head, "now");
408 } else if (newFee * BigInt(old.length) > oldFee * BigInt(p.length)) {
409 this.emit(0, ...head, "now");
410 const days = s.end - this.day;
411 const items = this.usageItem(s, true);
412 items.push({description: "unused " + old.name + " seats " + oldSeats + " " + days + " days", amount: -(oldFee * BigInt(days) / BigInt(old.length))});
413 items.push({description: "remaining " + p.name + " seats " + seats + " " + days + " days", amount: newFee * BigInt(days) / BigInt(p.length)});
414 s.plan = p;
415 s.seats = seats;
416 this.issue(s, "upgrade", items);
417 } else {
418 s.pending = {plan: p, seats};
419 this.emit(0, ...head, "at", s.end);
420 }
421 break;
422 }
423 case "PAUSE": {
424 const s = this.sub(a[0]);
425 this.state(s, ["ACTIVE"]);
426 this.emit(0, "PAUSE", "sub", s.id);
427 s.pauseStart = this.day;
428 this.changeState(s, "PAUSED");
429 break;
430 }
431 case "RESUME": {
432 const s = this.sub(a[0]);
433 this.state(s, ["PAUSED"]);
434 const shift = this.day - s.pauseStart!;
435 s.end += shift;
436 this.emit(0, "RESUME", "sub", s.id, "shift", shift, "period", period(s));
437 s.pauseStart = null;
438 this.changeState(s, "ACTIVE");
439 break;
440 }
441 case "CANCEL": {
442 const s = this.sub(a[0]);
443 if (a[1] === "NOW") {
444 this.state(s, ["TRIALING", "ACTIVE", "PAST_DUE", "UNPAID", "PAUSED"]);
445 const remaining = s.state === "ACTIVE" ? s.end - this.day : s.state === "PAUSED" ? s.end - s.pauseStart! : 0;
446 const refund = fee(s.plan, s.seats) * BigInt(remaining) / BigInt(s.plan.length);
447 this.balance(s.customer, "cancel sub " + s.id, refund);
448 this.emit(0, "CANCEL", "sub", s.id, "now", "credit", money(refund));
449 this.cancel(s);
450 } else {
451 this.state(s, ["ACTIVE", "TRIALING"]);
452 if (s.cancelEnd) throw new CommandError("already canceling " + s.id);
453 s.cancelEnd = true;
454 this.emit(0, "CANCEL", "sub", s.id, "end", s.state === "TRIALING" ? s.trialEnd! : s.end);
455 }
456 break;
457 }
458 case "USAGE": {
459 const s = this.sub(a[0]);
460 this.state(s, ["ACTIVE", "PAST_DUE"]);
461 if (s.plan.mode === null) throw new CommandError("not metered " + s.plan.name);
462 const units = Number(a[1]);
463 if (units === 0) throw new CommandError("invalid units");
464 s.usage += units;
465 this.emit(0, "USAGE", "sub", s.id, "units", units, "total", s.usage);
466 break;
467 }
468 case "APPLY": {
469 const s = this.sub(a[0]);
470 this.state(s, ["TRIALING", "ACTIVE", "PAST_DUE", "UNPAID", "PAUSED"]);
471 const coupon = this.coupon(a[1]);
472 coupon.redeemed++;
473 s.coupon = coupon;
474 s.left = coupon.applications;
475 this.emit(0, "APPLY", "sub", s.id, "coupon", coupon.name);
476 break;
477 }
478 case "PAY": {
479 const i = this.invoice(a[0]);
480 if (i.outstanding === 0n) throw new CommandError("already paid " + a[0]);
481 this.emit(0, "PAY", "invoice", i.id, "amount", money(i.outstanding));
482 i.paid += i.outstanding;
483 i.outstanding = 0n;
484 this.complete(i);
485 break;
486 }
487 case "REFUND": {
488 const i = this.invoice(a[0]);
489 const amount = cents(a[1]);
490 if (amount === 0n) throw new CommandError("invalid amount");
491 if (amount > i.total - i.refunded) throw new CommandError("refund exceeds " + a[0] + " " + money(i.total - i.refunded));
492 const applied = min(amount, i.outstanding);
493 i.refunded += amount;
494 i.outstanding -= applied;
495 this.balance(i.sub.customer, "refund invoice " + i.id, amount - applied);
496 this.emit(0, "REFUND", "invoice", i.id, "amount", money(amount), "applied", money(applied), "balance", money(i.sub.customer.balance));
497 if (applied > 0n && i.outstanding === 0n) this.complete(i);
498 break;
499 }
500 case "INVOICE": {
501 const i = this.invoice(a[0]);
502 this.emit(0, "INVOICE", i.id, "sub", i.sub.id, i.sub.customer.name, "day", i.day, i.kind, i.outstanding > 0n ? "OPEN" : "PAID");
503 for (const item of i.items) this.emit(2, item.description, money(item.amount));
504 this.totals(i);
505 this.emit(2, "credit", money(i.credit), "due", money(i.due), "paid", money(i.paid), "refunded", money(i.refunded), "outstanding", money(i.outstanding));
506 break;
507 }
508 case "SUB":
509 this.showSub(this.sub(a[0]));
510 break;
511 case "PREVIEW":
512 this.preview(this.sub(a[0]));
513 break;
514 case "STATEMENT":
515 case "LEDGER": {
516 const c = this.customer(a[0]);
517 if (cmd === "LEDGER") {
518 this.emit(0, "LEDGER", c.name, "balance", money(c.balance));
519 for (const row of c.log) this.emit(2, "day", row.day, row.description, (row.change > 0n ? "+" : "") + money(row.change), "balance", money(row.balance));
520 if (!c.log.length) this.emit(2, "(none)");
521 } else {
522 this.emit(0, "STATEMENT", c.name, "region", c.region, "tax", percent(this.taxes.get(c.region) ?? 0n), "method", c.method, "balance", money(c.balance));
523 for (const s of this.subs.filter(s => s.customer === c)) this.emit(2, "sub", s.id, s.plan.name, s.state);
524 const invoices = this.invoices.filter(i => i.sub.customer === c);
525 for (const i of invoices) this.emit(2, "invoice", i.id, "day", i.day, "sub", i.sub.id, i.kind, "total", money(i.total), "outstanding", money(i.outstanding));
526 this.emit(2, "outstanding", money(sum(invoices.map(i => i.outstanding))));
527 }
528 break;
529 }
530 case "MRR": {
531 const active = this.subs.filter(s => ["ACTIVE", "PAST_DUE"].includes(s.state));
532 const groups = Map.groupBy(active, s => s.plan.name);
533 const rows = new Map([...groups].map(([name, subs]) => [name, {
534 count: subs.length,
535 value: sum(subs.map(s => {
536 const amount = fee(s.plan, s.seats);
537 return s.plan.cycle === "MONTHLY" ? amount : s.plan.cycle === "YEARLY" ? amount / 12n : amount * 30n / 7n;
538 })),
539 }]));
540 this.emit(0, "MRR");
541 for (const name of [...rows.keys()].sort()) {
542 const row = rows.get(name)!;
543 this.emit(2, name, "subs", row.count, "mrr", money(row.value));
544 }
545 this.emit(2, "total", "subs", [...rows.values()].reduce((n, r) => n + r.count, 0), "mrr", money(sum([...rows.values()].map(r => r.value))));
546 break;
547 }
548 case "AGING": {
549 const rows = new Map<string, bigint[]>();
550 const total = [0n, 0n, 0n];
551 for (const i of this.invoices) {
552 if (i.outstanding === 0n) continue;
553 if (!rows.has(i.sub.customer.name)) rows.set(i.sub.customer.name, [0n, 0n, 0n]);
554 const age = this.day - i.day;
555 const bucket = age <= 30 ? 0 : age <= 60 ? 1 : 2;
556 rows.get(i.sub.customer.name)![bucket] += i.outstanding;
557 total[bucket] += i.outstanding;
558 }
559 this.emit(0, "AGING", "day", this.day);
560 for (const name of [...rows.keys()].sort()) {
561 const row = rows.get(name)!;
562 this.emit(2, name, "0-30", money(row[0]), "31-60", money(row[1]), "61+", money(row[2]), "sum", money(sum(row)));
563 }
564 this.emit(2, "total", "0-30", money(total[0]), "31-60", money(total[1]), "61+", money(total[2]), "sum", money(sum(total)));
565 break;
566 }
567 case "DUNNING": {
568 this.emit(0, "DUNNING");
569 const invoices = this.invoices.filter(i => i.retries.length > 0);
570 for (const i of invoices) this.emit(2, "invoice", i.id, "sub", i.sub.id, i.sub.customer.name,
571 "outstanding", money(i.outstanding), "attempts", i.attempts, "next", i.retries[0]);
572 if (!invoices.length) this.emit(2, "(none)");
573 break;
574 }
575 case "REVENUE": {
576 const total = (field: keyof Quote | "paid" | "refunded" | "outstanding"): string => money(sum(this.invoices.map(i => i[field])));
577 this.emit(0, "REVENUE");
578 this.emit(2, "invoices", this.invoices.length, "subtotal", total("subtotal"), "discount", total("discount"), "tax", total("tax"), "total", total("total"));
579 this.emit(2, "credit", total("credit"), "paid", total("paid"), "refunded", total("refunded"), "outstanding", total("outstanding"),
580 "balances", money(sum([...this.customers.values()].map(c => c.balance))));
581 break;
582 }
583 case "EVENTS": {
584 const s = a.length ? this.sub(a[0]) : null;
585 this.emit(0, "EVENTS", ...(s ? ["sub", s.id] : []));
586 const events = this.events.filter(e => s === null || e.id === s.id);
587 for (const e of events) this.emit(2, "day", e.day, "sub", e.id, e.from, "->", e.to);
588 if (!events.length) this.emit(2, "(none)");
589 break;
590 }
591 }
592 }
