23Money cents(const string& text) {
24 size_t dot = text.find('.');

28string money(Money value) {
29 string sign = value < 0 ? "-" : "";
30 value = llabs(value);
31 string fraction = to_string(value % 100);
32 return sign + to_string(value / 100) + "." +
33 (fraction.size() == 1 ? "0" : "") + fraction;
34}

36string signed_money(Money value) {
37 return (value > 0 ? "+" : "") + money(value);

50 Money unit;
51 Money flat;

54struct Plan {
55 string name;
56 Money price = 0;
57 string cycle;
58 int length = 0;
59 int trial = 0;
60 Money setup = 0;
61 bool per_seat = false;
62 string mode;
63 vector<Tier> tiers;
64
65 Money fee(int seats) const {
66 return price * (per_seat ? seats : 1);
67 }
68
69 Money usage_charge(int units) const {
70 Money amount = 0;
71 int lower = 0;
72 for (size_t i = 0; i < tiers.size(); ++i) {
73 const Tier& tier = tiers[i];
74 bool last = i + 1 == tiers.size();
75 if (mode == "VOLUME") {
76 if (last || tier.upper < 0 || units <= tier.upper)
77 return units * tier.unit + tier.flat;
78 } else {
79 if (units <= lower) break;
80 int end = last || tier.upper < 0 ? units : min(units, tier.upper);
81 amount += static_cast<Money>(end - lower) * tier.unit + tier.flat;
82 if (last || tier.upper < 0) break;
83 lower = tier.upper;
84 }
85 }
86 return amount;
87 }

298 bool coupon_usable(const Subscription& s, Money subtotal) const {
299 return subtotal > 0 && !s.coupon.empty() && s.remaining != 0;
300 }
301
302 Totals invoice_totals(const Subscription& s, const vector<Item>& items) const {
303 Totals totals;
304 for (const Item& item : items) totals.subtotal += item.amount;
305 if (coupon_usable(s, totals.subtotal)) {
306 const Coupon& coupon = coupons.at(s.coupon);
307 totals.discount = coupon.kind == "PERCENT"
308 ? totals.subtotal * coupon.value / 10000
309 : min(totals.subtotal, coupon.value);
310 }
311 const Customer& c = customers.at(s.customer);
312 int tax_rate = 0;
313 auto it = tax_rates.find(c.region);
314 if (it != tax_rates.end()) tax_rate = it->second;
315 totals.tax = (totals.subtotal - totals.discount) * tax_rate / 10000;
316 totals.total = totals.subtotal - totals.discount + totals.tax;
317 totals.credit = min(c.balance, totals.total);
318 totals.due = totals.total - totals.credit;
319 return totals;
320 }
321
322 void issue(Subscription& s, const string& kind, const vector<Item>& items) {
323 Totals totals = invoice_totals(s, items);
324 int id = static_cast<int>(invoices.size()) + 1;
325 Invoice entry{id, s.number, s.customer, day, kind, items, totals};
326 entry.outstanding = totals.due;
327 if (coupon_usable(s, totals.subtotal) && s.remaining > 0)
328 --s.remaining;
329 invoices.push_back(move(entry));
330 s.invoices.push_back(id);
331 Customer& c = customers.at(s.customer);
332 c.invoices.push_back(id);
333 balance_change(c, -totals.credit, "applied invoice " + to_string(id));
334 Invoice& bill = invoices.back();
335 string result;
336 if (bill.outstanding == 0) {
337 result = "PAID";
338 } else if (s.state == "UNPAID") {
339 result = "OPEN";
340 } else {
341 bill.attempts = 1;
342 if (c.method == "OK") {
343 bill.paid = bill.outstanding;
344 bill.outstanding = 0;
345 result = "PAID";
346 } else {
347 result = "FAILED";
348 }
349 }
350 notice("invoice", id, "sub", s.number, kind, "total", money(totals.total),
351 "credit", money(totals.credit), "due", money(totals.due), result);
352 if (result == "FAILED") {
353 if (s.state == "ACTIVE") change_state(s, "PAST_DUE");
354 if (s.state == "PAST_DUE")
355 bill.retries = {day + 3, day + 5, day + 7};
356 }
357 }
358
359 void payment_complete(Invoice& bill) {
360 bill.retries.clear();
361 Subscription& s = subscriptions[bill.subscription - 1];
362 if (s.state == "PAST_DUE" || s.state == "UNPAID") {
363 bool settled = true;
364 for (int id : s.invoices)
365 if (invoices[id - 1].outstanding)
366 settled = false;
367 if (settled) {
368 s.cancel_on = -1;
369 change_state(s, "ACTIVE");
370 }
371 }
372 }
373
374 void cancel_subscription(Subscription& s) {
375 change_state(s, "CANCELED");
376 s.canceled_on = day;
377 s.pending_plan.clear();
378 s.cancel_end = false;
379 s.cancel_on = -1;
380 vector<Item> items = close_usage(s);
381 if (!items.empty()) issue(s, "final", items);
382 }
383
384 void retry(Invoice& bill) {
385 bill.retries.erase(bill.retries.begin());
386 ++bill.attempts;
387 int attempt = bill.attempts - 1;
388 if (customers.at(bill.customer).method == "OK") {
389 bill.paid += bill.outstanding;
390 bill.outstanding = 0;
391 notice("retry", attempt, "invoice", bill.number, "PAID");
392 payment_complete(bill);
393 } else {
394 notice("retry", attempt, "invoice", bill.number, "FAILED");
395 if (bill.retries.empty()) {
396 Subscription& s = subscriptions[bill.subscription - 1];
397 change_state(s, "UNPAID");
398 s.cancel_on = day + 14;
399 }
400 }
401 }
402
403 void advance_subscription(Subscription& s) {
404 if (s.state == "PAUSED" || s.state == "CANCELED") return;
405 for (int id : s.invoices) {
406 Invoice& bill = invoices[id - 1];
407 if (!bill.retries.empty() && bill.retries.front() == day)
408 retry(bill);
409 }
410 if (s.state == "UNPAID" && s.cancel_on == day)
411 cancel_subscription(s);
412 if (s.state == "TRIALING" && s.trial_end == day) {
413 if (s.cancel_end) {
414 cancel_subscription(s);
415 } else {
416 change_state(s, "ACTIVE");
417 s.start = day;
418 s.end = day + plans.at(s.plan).length;
419 issue(s, "first", opening_items(s, day));
420 }
421 }
422 if ((s.state == "ACTIVE" || s.state == "PAST_DUE" || s.state == "UNPAID") &&
423 s.end == day) {
424 if (s.cancel_end) {
425 cancel_subscription(s);
426 } else {
427 vector<Item> items = close_usage(s);
428 if (!s.pending_plan.empty()) {
429 notice("sub", s.number, s.plan, "seats", s.seats, "->",
430 s.pending_plan, "seats", s.pending_seats);
431 s.plan = s.pending_plan;
432 s.seats = s.pending_seats;
433 s.pending_plan.clear();
434 }
435 s.start = day;
436 s.end = day + plans.at(s.plan).length;
437 items.push_back(period_item(plans.at(s.plan), s.seats, day));
438 issue(s, "renewal", items);
439 }
440 }

443 void cmd_plan(const vector<string>& a) {
444 const string& name = a[0];
445 if (plans.count(name)) fail("duplicate plan " + name);
446 Plan p;
447 p.name = name;
448 p.price = cents(a[1]);
449 p.cycle = a[2];
450 p.length = period_length(a[2]);
451 for (size_t i = 3; i < a.size(); ++i) {
452 if (a[i] == "TRIAL") {
453 p.trial = number(a[++i]);
454 } else if (a[i] == "SETUP") {
455 p.setup = cents(a[++i]);
456 } else if (a[i] == "PERSEAT") {
457 p.per_seat = true;
458 } else if (a[i] == "GRADUATED" || a[i] == "VOLUME") {
459 p.mode = a[i];
460 }
461 }
462 plans.emplace(name, p);
463 string text = "PLAN " + name + " " + p.cycle + " " + money(p.price);
464 if (p.trial) text += " trial " + to_string(p.trial);
465 if (p.setup) text += " setup " + money(p.setup);
466 if (p.per_seat) text += " perseat";
467 if (!p.mode.empty()) {
468 text += p.mode == "GRADUATED" ? " graduated" : " volume";
469 }
470 emit(0, text);
471 }

473 void cmd_tier(const vector<string>& a) {
474 Plan& p = plan(a[0]);
475 if (p.mode.empty()) fail("not metered " + p.name);
476 int upper = a[1] == "INF" ? -1 : number(a[1]);
477 if (!p.tiers.empty()) {
478 int previous = p.tiers.back().upper;
479 if (previous < 0 || (upper >= 0 && upper <= previous))
480 fail("tier order " + p.name);
481 }
482 p.tiers.push_back({upper, cents(a[2]), cents(a[3])});
483 emit(0, "TIER", p.name, p.tiers.size(), "upto", a[1], "unit", a[2], "flat", a[3]);
484 }

501 void cmd_coupon(const vector<string>& a) {
502 if (coupons.count(a[0])) fail("duplicate coupon " + a[0]);
503 Coupon c;
504 c.name = a[0];
505 c.kind = a[1];
506 c.value = a[1] == "PERCENT" ? number(a[2]) : cents(a[2]);
507 c.applications = a[3] == "ONCE" ? 1 : -1;
508 size_t i = 4;
509 if (a[3] == "REPEATING") c.applications = number(a[i++]);
510 if (i < a.size()) c.maximum = number(a[i + 1]);
511 coupons.emplace(c.name, c);
512 string shown = c.kind == "PERCENT" ? percentage(static_cast<int>(c.value)) : money(c.value);
513 string text = "COUPON " + c.name + " " +
514 (c.kind == "PERCENT" ? "percent" : "amount") + " " +
515 shown + " " + (a[3] == "REPEATING" ? "repeating " +
516 to_string(c.applications) : a[3] == "ONCE" ? "once" : "forever");
517 if (c.maximum >= 0) text += " max " + to_string(c.maximum);
518 emit(0, text);
519 }

546 void cmd_subscribe(const vector<string>& a) {
547 Customer& c = customer(a[0]);
548 Plan& p = plan(a[1]);
549 int seats = 1;
550 bool has_seats = false;
551 string coupon_name;
552 for (size_t i = 2; i < a.size(); i += 2) {
553 if (a[i] == "SEATS") {
554 seats = number(a[i + 1]);
555 has_seats = true;
556 } else {
557 coupon_name = a[i + 1];
558 }
559 }
560 if (has_seats && !p.per_seat) fail("not per-seat " + p.name);
561 if (seats == 0) fail("invalid seats");
562 Coupon* coupon = coupon_name.empty() ? nullptr : &redeemable_coupon(coupon_name);
563 int number = static_cast<int>(subscriptions.size()) + 1;
564 Subscription s;
565 s.number = number;
566 s.customer = c.name;
567 s.plan = p.name;
568 s.seats = seats;
569 s.coupon = coupon_name;
570 if (coupon != nullptr) {
571 ++coupon->redemptions;
572 s.remaining = coupon->applications;
573 }
574 subscriptions.push_back(move(s));
575 c.subscriptions.push_back(number);
576 Subscription& subscription = subscriptions.back();
577 string text = "SUBSCRIBE sub " + to_string(number) + " " + c.name +
578 " " + p.name + " seats " + to_string(seats);
579 if (coupon != nullptr) text += " coupon " + coupon_name;
580 emit(0, text);
581 if (p.trial) {
582 subscription.trial_end = day + p.trial;
583 change_state(subscription, "TRIALING");
584 } else {
585 change_state(subscription, "ACTIVE");
586 subscription.start = day;
587 subscription.end = day + p.length;
588 issue(subscription, "first", opening_items(subscription, day));
589 }
590 }

592 void cmd_change(const vector<string>& a) {
593 Subscription& s = require_state(a[0], {"ACTIVE", "TRIALING"});
594 Plan* p;
595 int seats = s.seats;
596 if (a[1] == "PLAN") {
597 p = &plan(a[2]);
598 seats = p->per_seat ? s.seats : 1;
599 } else {
600 p = &plan(s.plan);
601 if (!p->per_seat) fail("not per-seat " + p->name);
602 seats = number(a[2]);
603 if (!seats) fail("invalid seats");
604 }
605 if (p->name == s.plan && seats == s.seats)
606 fail("no change " + a[0]);
607 const Plan& old = plans.at(s.plan);
608 int old_seats = s.seats;
609 string text = "CHANGE sub " + to_string(s.number) + " " + old.name +
610 " seats " + to_string(old_seats) + " -> " + p->name +
611 " seats " + to_string(seats);
612 s.pending_plan.clear();
613 bool upgrade = p->fee(seats) * old.length > old.fee(old_seats) * p->length;
614 if (s.state == "TRIALING") {
615 s.plan = p->name;
616 s.seats = seats;
617 emit(0, text + " now");
618 } else if (upgrade) {
619 emit(0, text + " now");
620 int days = s.end - day;
621 vector<Item> items = close_usage(s);
622 items.push_back({"unused " + old.name + " seats " + to_string(old_seats) +
623 " " + to_string(days) + " days",
624 -(old.fee(old_seats) * days / old.length)});
625 items.push_back({"remaining " + p->name + " seats " + to_string(seats) +
626 " " + to_string(days) + " days",
627 p->fee(seats) * days / p->length});
628 s.plan = p->name;
629 s.seats = seats;
630 issue(s, "upgrade", items);
631 } else {
632 s.pending_plan = p->name;
633 s.pending_seats = seats;
634 emit(0, text, "at", s.end);
635 }
636 }

654 void cmd_cancel(const vector<string>& a) {
655 Subscription& s = subscription(a[0]);
656 if (a[1] == "END") {
657 if (s.state != "ACTIVE" && s.state != "TRIALING")
658 fail("invalid state " + a[0] + " " + s.state);
659 if (s.cancel_end) fail("already canceling " + a[0]);
660 s.cancel_end = true;
661 int end = s.state == "TRIALING" ? s.trial_end : s.end;
662 emit(0, "CANCEL", "sub", s.number, "end", end);
663 } else {
664 if (s.state == "CANCELED")
665 fail("invalid state " + a[0] + " " + s.state);
666 Money refund = 0;
667 const Plan& p = plans.at(s.plan);
668 if (s.state == "ACTIVE" || s.state == "PAUSED") {
669 int start = s.state == "ACTIVE" ? day : s.pause_start;
670 refund = fee(s) * (s.end - start) / p.length;
671 }
672 balance_change(customers.at(s.customer), refund,
673 "cancel sub " + to_string(s.number));
674 emit(0, "CANCEL", "sub", s.number, "now", "credit", money(refund));
675 cancel_subscription(s);
676 }

679 void cmd_usage(const vector<string>& a) {
680 Subscription& s = require_state(a[0], {"ACTIVE", "PAST_DUE"});
681 if (plans.at(s.plan).mode.empty()) fail("not metered " + s.plan);
682 int units = number(a[1]);
683 if (!units) fail("invalid units");
684 s.usage += units;
685 emit(0, "USAGE", "sub", s.number, "units", units, "total", s.usage);
686 }

688 void cmd_apply(const vector<string>& a) {
689 Subscription& s = subscription(a[0]);
690 if (s.state == "CANCELED") fail("invalid state " + a[0] + " " + s.state);
691 Coupon& coupon = redeemable_coupon(a[1]);
692 ++coupon.redemptions;
693 s.coupon = coupon.name;
694 s.remaining = coupon.applications;
695 emit(0, "APPLY", "sub", s.number, "coupon", coupon.name);
696 }

698 void cmd_pay(const vector<string>& a) {
699 Invoice& bill = invoice(a[0]);
700 if (!bill.outstanding) fail("already paid " + a[0]);
701 emit(0, "PAY", "invoice", bill.number, "amount", money(bill.outstanding));
702 bill.paid += bill.outstanding;
703 bill.outstanding = 0;
704 payment_complete(bill);
705 }

707 void cmd_refund(const vector<string>& a) {
708 Invoice& bill = invoice(a[0]);
709 Money value = cents(a[1]);
710 if (!value) fail("invalid amount");
711 Money maximum = bill.totals.total - bill.refunded;
712 if (value > maximum)
713 fail("refund exceeds " + a[0] + " " + money(maximum));
714 Money applied = min(value, bill.outstanding);
715 bill.refunded += value;
716 bill.outstanding -= applied;
717 Customer& c = customers.at(bill.customer);
718 balance_change(c, value - applied, "refund invoice " + to_string(bill.number));
719 emit(0, "REFUND", "invoice", bill.number, "amount", a[1], "applied",
720 money(applied), "balance", money(c.balance));
721 if (applied && !bill.outstanding) payment_complete(bill);
722 }

745 void cmd_sub(const vector<string>& a) {
746 const Subscription& s = subscription(a[0]);
747 emit(0, "SUB", s.number, s.customer, s.plan, "seats", s.seats, s.state);
748 if (s.state == "TRIALING") emit(2, "trial", "until", s.trial_end);
749 if (s.state == "ACTIVE" || s.state == "PAST_DUE" ||
750 s.state == "UNPAID" || s.state == "PAUSED")
751 emit(2, "period", period(s));
752 if (s.state == "PAUSED") emit(2, "paused", "since", s.pause_start);
753 if (!s.pending_plan.empty())
754 emit(2, "pending", s.pending_plan, "seats", s.pending_seats);
755 if (s.cancel_end) emit(2, "cancel", "at", "end");
756 if (s.cancel_on >= 0) emit(2, "cancel", "on", s.cancel_on);
757 if (s.state == "CANCELED") emit(2, "canceled", "on", s.canceled_on);
758 if (!s.coupon.empty()) {
759 if (s.remaining < 0) emit(2, "coupon", s.coupon, "forever");
760 else if (s.remaining > 0) emit(2, "coupon", s.coupon, "left", s.remaining);
761 }
762 if (!plans.at(s.plan).mode.empty() && s.state != "CANCELED")
763 emit(2, "usage", s.usage);
764 string open;
765 for (int id : s.invoices) {
766 if (invoices[id - 1].outstanding > 0)
767 open += " " + to_string(id);
768 }
769 if (!open.empty()) emit(2, "open" + open);
770 }

772 void cmd_preview(const vector<string>& a) {
773 const Subscription& s = require_state(a[0],
774 {"TRIALING", "ACTIVE", "PAST_DUE", "UNPAID"});
775 int due_day = s.state == "TRIALING" ? s.trial_end : s.end;
776 vector<Item> items;
777 string kind;
778 if (s.cancel_end) {
779 kind = "final";
780 if (s.state != "TRIALING") items = usage_items(s);
781 } else if (s.state == "TRIALING") {
782 kind = "first";
783 items = opening_items(s, due_day);
784 } else {
785 kind = "renewal";
786 items = usage_items(s);
787 string name = s.pending_plan.empty() ? s.plan : s.pending_plan;
788 int seats = s.pending_plan.empty() ? s.seats : s.pending_seats;
789 items.push_back(period_item(plans.at(name), seats, due_day));
790 }
791 emit(0, "PREVIEW", "sub", s.number, "day", due_day, kind);
792 if (items.empty()) {
793 emit(2, "(none)");
794 return;
795 }
796 emit_items(items);
797 Totals totals = invoice_totals(s, items);
798 emit_totals(totals);
799 emit(2, "credit", money(totals.credit), "due", money(totals.due));
800 }

802 void cmd_statement(const vector<string>& a) {
803 const Customer& c = customer(a[0]);
804 int rate = tax_rates.count(c.region) ? tax_rates.at(c.region) : 0;
805 emit(0, "STATEMENT", c.name, "region", c.region, "tax",
806 percentage(rate), "method", c.method, "balance", money(c.balance));
807 Money outstanding = 0;

809 const Subscription& s = subscriptions[id - 1];

815 bill.kind, "total", money(bill.totals.total),
816 "outstanding", money(bill.outstanding));

818 }
819 emit(2, "outstanding", money(outstanding));
820 }

831 void cmd_mrr(const vector<string>&) {
832 map<string, pair<int, Money>> rows;
833 for (const Subscription& s : subscriptions) {
834 if (s.state != "ACTIVE" && s.state != "PAST_DUE") continue;
835 const Plan& p = plans.at(s.plan);
836 Money amount = p.fee(s.seats);
837 Money monthly = p.cycle == "MONTHLY" ? amount :
838 p.cycle == "YEARLY" ? amount / 12 : amount * 30 / 7;
839 auto& row = rows[p.name];
840 ++row.first;
841 row.second += monthly;
842 }
843 emit(0, "MRR");
844 int count = 0;
845 Money sum = 0;
846 for (const auto& [name, row] : rows) {
847 emit(2, name, "subs", row.first, "mrr", money(row.second));
848 count += row.first;
849 sum += row.second;
850 }
851 emit(2, "total", "subs", count, "mrr", money(sum));
852 }

854 void cmd_aging(const vector<string>&) {
855 map<string, vector<Money>> rows;
856 vector<Money> totals(3, 0);
857 for (const Invoice& bill : invoices) {
858 if (!bill.outstanding) continue;
859 int age = day - bill.day;
860 int bucket = age <= 30 ? 0 : age <= 60 ? 1 : 2;
861 auto& row = rows[bill.customer];
862 if (row.empty()) row.assign(3, 0);
863 row[bucket] += bill.outstanding;
864 totals[bucket] += bill.outstanding;
865 }
866 emit(0, "AGING", "day", day);
867 for (const auto& [name, row] : rows) {
868 emit(2, name, "0-30", money(row[0]), "31-60", money(row[1]),
869 "61+", money(row[2]), "sum", money(row[0] + row[1] + row[2]));
870 }
871 emit(2, "total", "0-30", money(totals[0]), "31-60", money(totals[1]),
872 "61+", money(totals[2]), "sum", money(totals[0] + totals[1] + totals[2]));
873 }

888 void cmd_revenue(const vector<string>&) {
889 Totals totals;
890 Money paid = 0, refunded = 0, outstanding = 0, balances = 0;
891 for (const Invoice& bill : invoices) {
892 totals.subtotal += bill.totals.subtotal;
893 totals.discount += bill.totals.discount;
894 totals.tax += bill.totals.tax;
895 totals.total += bill.totals.total;
896 totals.credit += bill.totals.credit;
897 paid += bill.paid;
898 refunded += bill.refunded;
899 outstanding += bill.outstanding;
900 }
901 for (const auto& [id, c] : customers) balances += c.balance;
902 emit(0, "REVENUE");
903 emit(2, "invoices", invoices.size(), "subtotal", money(totals.subtotal),
904 "discount", money(totals.discount), "tax", money(totals.tax),
905 "total", money(totals.total));
906 emit(2, "credit", money(totals.credit), "paid", money(paid),
907 "refunded", money(refunded), "outstanding", money(outstanding),
908 "balances", money(balances));

911 void cmd_events(const vector<string>& a) {
912 int selected = 0;
913 if (!a.empty()) selected = subscription(a[0]).number;
914 if (selected) emit(0, "EVENTS", "sub", selected);
915 else emit(0, "EVENTS");
916 int count = 0;
917 for (const Event& event : events) {
918 if (selected && event.subscription != selected) continue;
919 emit(2, "day", event.day, "sub", event.subscription,
920 event.previous, "->", event.current);

922 }
923 if (!count) emit(2, "(none)");
924 }

926public:
927 void execute(const string& line) {
928 istringstream stream(line);
929 vector<string> a;
930 string word;
931 while (stream >> word) a.push_back(word);
932 if (a.empty()) return;
933 string command = a[0];
934 a.erase(a.begin());
935 try {
936 if (command == "PLAN") cmd_plan(a);
937 else if (command == "TIER") cmd_tier(a);
938 else if (command == "TAX") cmd_tax(a);
939 else if (command == "CUSTOMER") cmd_customer(a);
940 else if (command == "COUPON") cmd_coupon(a);
941 else if (command == "PAYMETHOD") cmd_paymethod(a);
942 else if (command == "CREDIT") cmd_credit(a);
943 else if (command == "DAY") cmd_day(a);
944 else if (command == "SUBSCRIBE") cmd_subscribe(a);
945 else if (command == "CHANGE") cmd_change(a);
946 else if (command == "PAUSE") cmd_pause(a);
947 else if (command == "RESUME") cmd_resume(a);
948 else if (command == "CANCEL") cmd_cancel(a);
949 else if (command == "USAGE") cmd_usage(a);
950 else if (command == "APPLY") cmd_apply(a);
951 else if (command == "PAY") cmd_pay(a);
952 else if (command == "REFUND") cmd_refund(a);
953 else if (command == "INVOICE") cmd_invoice(a);
954 else if (command == "SUB") cmd_sub(a);
955 else if (command == "PREVIEW") cmd_preview(a);
956 else if (command == "STATEMENT") cmd_statement(a);
957 else if (command == "LEDGER") cmd_ledger(a);
958 else if (command == "MRR") cmd_mrr(a);
959 else if (command == "AGING") cmd_aging(a);
960 else if (command == "DUNNING") cmd_dunning(a);
961 else if (command == "REVENUE") cmd_revenue(a);
962 else if (command == "EVENTS") cmd_events(a);
963 } catch (const CommandError& error) {
964 emit(0, "ERROR", error.what());
965 }
966 }
