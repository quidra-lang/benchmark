11private func cents(_ text: String) -> Int {
12 let parts = text.split(separator: ".")
13 return Int(parts[0])! * 100 + Int(parts[1])!
14}
15
16private func money(_ value: Int) -> String {
17 let absolute = abs(value)
18 return (value < 0 ? "-" : "") +
19 "\(absolute / 100)." + String(format: "%02d", absolute % 100)
20}
21
22private func signedMoney(_ value: Int) -> String {
23 (value > 0 ? "+" : "") + money(value)
24}
25
26private func percentage(_ basisPoints: Int) -> String {
27 money(basisPoints) + "%"
28}

32 let unitPrice: Int
33 let flatFee: Int

36private final class Plan {
37 let name: String
38 let price: Int
39 let cycle: String
40 let length: Int
41 var trial = 0
42 var setup = 0
43 var perSeat = false
44 var mode: String?
45 var tiers: [Tier] = []
46
47 init(name: String, price: Int, cycle: String) {
48 self.name = name
49 self.price = price
50 self.cycle = cycle
51 switch cycle {
52 case "WEEKLY": length = 7
53 case "MONTHLY": length = 30
54 default: length = 360
55 }
56 }
57
58 func fee(_ seats: Int) -> Int {
59 perSeat ? price * seats : price
60 }
61
62 func usageCharge(_ units: Int) -> Int {
63 var total = 0
64 var lower = 0
65 for (index, tier) in tiers.enumerated() {
66 let last = index == tiers.count - 1
67 if mode == "VOLUME" {
68 if last || tier.upper == nil || units <= tier.upper! {
69 return units * tier.unitPrice + tier.flatFee
70 }
71 } else {
72 if units <= lower { break }
73 let end = last || tier.upper == nil ? units : min(units, tier.upper!)
74 total += (end - lower) * tier.unitPrice + tier.flatFee
75 if last || tier.upper == nil { break }
76 lower = tier.upper!
77 }
78 }
79 return total
80 }
81}

329 private func couponUsable(_ subscription: Subscription, subtotal: Int) -> Bool {
330 subtotal > 0 && subscription.coupon != nil &&
331 (subscription.remaining == nil || subscription.remaining! > 0)
332 }
333
334 private func invoiceTotals(_ subscription: Subscription, items: [BillItem]) -> Totals {
335 let subtotal = items.reduce(0) { $0 + $1.amount }
336 var discount = 0
337 if couponUsable(subscription, subtotal: subtotal) {
338 let coupon = subscription.coupon!
339 discount = coupon.kind == "PERCENT"
340 ? subtotal * coupon.value / 10000
341 : min(subtotal, coupon.value)
342 }
343 let tax = (subtotal - discount) *
344 (taxRates[subscription.customer.region] ?? 0) / 10000
345 let total = subtotal - discount + tax
346 let credit = min(subscription.customer.balance, total)
347 return Totals(
348 subtotal: subtotal, discount: discount, tax: tax, total: total,
349 credit: credit, due: total - credit
350 )
351 }

353 private func issue(_ subscription: Subscription, kind: String, items: [BillItem]) {
354 let totals = invoiceTotals(subscription, items: items)
355 let invoice = Invoice(
356 number: invoices.count + 1, subscription: subscription,
357 day: day, kind: kind, items: items, totals: totals
358 )
359 if couponUsable(subscription, subtotal: invoice.subtotal),
360 let remaining = subscription.remaining {
361 subscription.remaining = remaining - 1
362 }
363 invoices.append(invoice)
364 subscription.invoices.append(invoice)
365 subscription.customer.invoices.append(invoice)
366 balanceChange(subscription.customer, by: -invoice.credit,
367 description: "applied invoice \(invoice.number)")
368
369 let result: String
370 if invoice.due == 0 {
371 result = "PAID"
372 } else if subscription.state == "UNPAID" {
373 result = "OPEN"
374 } else {
375 invoice.attempts = 1
376 if subscription.customer.method == "OK" {
377 invoice.paid = invoice.due
378 invoice.outstanding = 0
379 result = "PAID"
380 } else {
381 result = "FAILED"
382 }
383 }
384
385 notice(
386 "invoice", invoice.number, "sub", subscription.number, kind,
387 "total", money(invoice.total), "credit", money(invoice.credit),
388 "due", money(invoice.due), result
389 )
390 if result == "FAILED" {
391 if subscription.state == "ACTIVE" { changeState(subscription, to: "PAST_DUE") }
392 if subscription.state == "PAST_DUE" {
393 invoice.retries = [day + 3, day + 5, day + 7]
394 }
395 }
396 }

398 private func paymentComplete(_ invoice: Invoice) {
399 invoice.retries.removeAll()
400 let subscription = invoice.subscription
401 if ["PAST_DUE", "UNPAID"].contains(subscription.state) &&
402 subscription.invoices.allSatisfy({ $0.outstanding == 0 }) {
403 subscription.cancelOn = nil
404 changeState(subscription, to: "ACTIVE")
405 }
406 }
407
408 private func cancelSubscription(_ subscription: Subscription) {
409 changeState(subscription, to: "CANCELED")
410 subscription.canceledOn = day
411 subscription.pending = nil
412 subscription.cancelEnd = false
413 subscription.cancelOn = nil
414 let items = usageItems(subscription, close: true)
415 if !items.isEmpty { issue(subscription, kind: "final", items: items) }
416 }
417
418 private func retry(_ invoice: Invoice) {
419 invoice.retries.removeFirst()
420 invoice.attempts += 1
421 let number = invoice.attempts - 1
422 if invoice.customer.method == "OK" {
423 invoice.paid += invoice.outstanding
424 invoice.outstanding = 0
425 notice("retry", number, "invoice", invoice.number, "PAID")
426 paymentComplete(invoice)
427 } else {
428 notice("retry", number, "invoice", invoice.number, "FAILED")
429 if invoice.retries.isEmpty {
430 changeState(invoice.subscription, to: "UNPAID")
431 invoice.subscription.cancelOn = day + 14
432 }
433 }
434 }
435
436 private func advanceSubscription(_ subscription: Subscription) {
437 if ["PAUSED", "CANCELED"].contains(subscription.state) { return }
438
439 for invoice in subscription.invoices {
440 if invoice.retries.first == day { retry(invoice) }
441 }
442 if subscription.state == "UNPAID" && subscription.cancelOn == day {
443 cancelSubscription(subscription)
444 }
445 if subscription.state == "TRIALING" && subscription.trialEnd == day {
446 if subscription.cancelEnd {
447 cancelSubscription(subscription)
448 } else {
449 changeState(subscription, to: "ACTIVE")
450 subscription.start = day
451 subscription.end = day + subscription.plan.length
452 issue(subscription, kind: "first",
453 items: openingItems(subscription, start: day))
454 }
455 }
456 if ["ACTIVE", "PAST_DUE", "UNPAID"].contains(subscription.state) &&
457 subscription.end == day {
458 if subscription.cancelEnd {
459 cancelSubscription(subscription)
460 } else {
461 var items = usageItems(subscription, close: true)
462 if let (plan, seats) = subscription.pending {
463 notice(
464 "sub", subscription.number, subscription.plan.name, "seats",
465 subscription.seats, "->", plan.name, "seats", seats
466 )
467 subscription.plan = plan
468 subscription.seats = seats
469 subscription.pending = nil
470 }
471 subscription.start = day
472 subscription.end = day + subscription.plan.length
473 items.append(periodItem(subscription.plan, seats: subscription.seats, start: day))
474 issue(subscription, kind: "renewal", items: items)
475 }
476 }
477 }

479 private func commandPlan(_ a: [String]) throws {
480 let name = a[0]
481 let amount = a[1]
482 let cycle = a[2]
483 if plans[name] != nil { throw fail("duplicate plan \(name)") }
484 let plan = Plan(name: name, price: cents(amount), cycle: cycle)
485 var index = 3
486 while index < a.count {
487 switch a[index] {
488 case "TRIAL":
489 plan.trial = Int(a[index + 1])!
490 index += 2
491 case "SETUP":
492 plan.setup = cents(a[index + 1])
493 index += 2
494 case "PERSEAT":
495 plan.perSeat = true
496 index += 1
497 default:
498 plan.mode = a[index]
499 index += 1
500 }
501 }
502 plans[name] = plan
503 var text = "PLAN \(name) \(cycle) \(amount)"
504 if plan.trial != 0 { text += " trial \(plan.trial)" }
505 if plan.setup != 0 { text += " setup \(money(plan.setup))" }
506 if plan.perSeat { text += " perseat" }
507 if let mode = plan.mode { text += " \(mode.lowercased())" }
508 emit(text)
509 }

512 let name = a[0]
513 let plan = try self.plan(name)
514 guard plan.mode != nil else { throw fail("not metered \(name)") }

522 emit("TIER", name, plan.tiers.count, "upto", a[1], "unit", a[2], "flat", a[3])
523 }

537 private func commandCoupon(_ a: [String]) throws {
538 let name = a[0]
539 let kind = a[1]
540 let application = a[3]
541 if coupons[name] != nil { throw fail("duplicate coupon \(name)") }
542 var applications: Int? = application == "ONCE" ? 1 : nil
543 var index = 4
544 if application == "REPEATING" {
545 applications = Int(a[index])!
546 index += 1
547 }
548 let maximum = index < a.count ? Int(a[index + 1])! : nil
549 let value = kind == "PERCENT" ? Int(a[2])! : cents(a[2])
550 let coupon = Coupon(
551 name: name, kind: kind, value: value,
552 applications: applications, maximum: maximum
553 )
554 coupons[name] = coupon
555 let shown = kind == "PERCENT" ? percentage(value) : money(value)
556 var text = "COUPON \(name) \(kind.lowercased()) \(shown) \(application.lowercased())"
557 if application == "REPEATING" { text += " \(applications!)" }
558 if let maximum { text += " max \(maximum)" }
559 emit(text)
560 }

586 private func commandSubscribe(_ a: [String]) throws {
587 let customer = try self.customer(a[0])
588 let plan = try self.plan(a[1])
589 var seats = 1
590 var hasSeats = false
591 var couponName: String?
592 var index = 2
593 while index < a.count {
594 if a[index] == "SEATS" {
595 seats = Int(a[index + 1])!
596 hasSeats = true
597 } else {
598 couponName = a[index + 1]
599 }
600 index += 2
601 }
602 if hasSeats && !plan.perSeat { throw fail("not per-seat \(plan.name)") }
603 if seats == 0 { throw fail("invalid seats") }
604 let coupon = try couponName.map { try redeemableCoupon($0) }
605 let subscription = Subscription(
606 number: subscriptions.count + 1, customer: customer,
607 plan: plan, seats: seats, coupon: coupon
608 )
609 coupon?.redemptions += 1
610 subscriptions.append(subscription)
611 customer.subscriptions.append(subscription)
612 var text = "SUBSCRIBE sub \(subscription.number) \(customer.name) \(plan.name) seats \(seats)"
613 if let coupon { text += " coupon \(coupon.name)" }
614 emit(text)
615 if plan.trial != 0 {
616 subscription.trialEnd = day + plan.trial
617 changeState(subscription, to: "TRIALING")
618 } else {
619 changeState(subscription, to: "ACTIVE")
620 subscription.start = day
621 subscription.end = day + plan.length
622 issue(subscription, kind: "first", items: openingItems(subscription, start: day))
623 }
624 }

626 private func commandChange(_ a: [String]) throws {
627 let subscription = try self.subscription(a[0], allowed: ["ACTIVE", "TRIALING"])
628 let newPlan: Plan
629 let seats: Int
630 if a[1] == "PLAN" {
631 newPlan = try self.plan(a[2])
632 seats = newPlan.perSeat ? subscription.seats : 1
633 } else {
634 newPlan = subscription.plan
635 if !newPlan.perSeat { throw fail("not per-seat \(newPlan.name)") }
636 seats = Int(a[2])!
637 if seats == 0 { throw fail("invalid seats") }
638 }
639 if newPlan === subscription.plan && seats == subscription.seats {
640 throw fail("no change \(a[0])")
641 }
642 let oldPlan = subscription.plan
643 let oldSeats = subscription.seats
644 let head = "CHANGE sub \(subscription.number) \(oldPlan.name) seats \(oldSeats) -> \(newPlan.name) seats \(seats)"
645 subscription.pending = nil
646 let upgrade = newPlan.fee(seats) * oldPlan.length >
647 oldPlan.fee(oldSeats) * newPlan.length
648 if subscription.state == "TRIALING" {
649 subscription.plan = newPlan
650 subscription.seats = seats
651 emit(head, "now")
652 } else if upgrade {
653 emit(head, "now")
654 let days = subscription.end! - day
655 var items = usageItems(subscription, close: true)
656 items.append(BillItem(
657 description: "unused \(oldPlan.name) seats \(oldSeats) \(days) days",
658 amount: -(oldPlan.fee(oldSeats) * days / oldPlan.length)
659 ))
660 items.append(BillItem(
661 description: "remaining \(newPlan.name) seats \(seats) \(days) days",
662 amount: newPlan.fee(seats) * days / newPlan.length
663 ))
664 subscription.plan = newPlan
665 subscription.seats = seats
666 issue(subscription, kind: "upgrade", items: items)
667 } else {
668 subscription.pending = (newPlan, seats)
669 emit(head, "at", subscription.end!)
670 }
671 }

690 private func commandCancel(_ a: [String]) throws {
691 let subscription = try self.subscription(a[0])
692 if a[1] == "END" {
693 if !["ACTIVE", "TRIALING"].contains(subscription.state) {
694 throw fail("invalid state \(a[0]) \(subscription.state)")
695 }
696 if subscription.cancelEnd { throw fail("already canceling \(a[0])") }
697 subscription.cancelEnd = true
698 let date = subscription.state == "TRIALING" ?
699 subscription.trialEnd! : subscription.end!
700 emit("CANCEL", "sub", subscription.number, "end", date)
701 } else {
702 if subscription.state == "CANCELED" {
703 throw fail("invalid state \(a[0]) \(subscription.state)")
704 }
705 var refund = 0
706 if ["ACTIVE", "PAUSED"].contains(subscription.state) {
707 let first = subscription.state == "ACTIVE" ? day : subscription.pauseStart!
708 refund = subscription.fee() * (subscription.end! - first) / subscription.plan.length
709 }
710 balanceChange(subscription.customer, by: refund,
711 description: "cancel sub \(subscription.number)")
712 emit("CANCEL", "sub", subscription.number, "now", "credit", money(refund))
713 cancelSubscription(subscription)
714 }
715 }

747 paymentComplete(invoice)
748 }

750 private func commandRefund(_ a: [String]) throws {
751 let invoice = try self.invoice(a[0])
752 let amount = cents(a[1])
753 if amount == 0 { throw fail("invalid amount") }
754 let maximum = invoice.total - invoice.refunded
755 if amount > maximum {
756 throw fail("refund exceeds \(a[0]) \(money(maximum))")
757 }
758 let applied = min(amount, invoice.outstanding)
759 invoice.refunded += amount
760 invoice.outstanding -= applied
761 balanceChange(invoice.customer, by: amount - applied,
762 description: "refund invoice \(invoice.number)")
763 emit(
764 "REFUND", "invoice", invoice.number, "amount", a[1], "applied",
765 money(applied), "balance", money(invoice.customer.balance)
766 )
767 if applied != 0 && invoice.outstanding == 0 { paymentComplete(invoice) }
768 }

902 private func commandMRR() {
903 var byPlan: [String: (Int, Int)] = [:]
904 for subscription in subscriptions {
905 guard ["ACTIVE", "PAST_DUE"].contains(subscription.state) else { continue }
906 let fee = subscription.fee()
907 let monthly: Int
908 switch subscription.plan.cycle {
909 case "MONTHLY": monthly = fee
910 case "YEARLY": monthly = fee / 12
911 default: monthly = fee * 30 / 7
912 }
913 let previous = byPlan[subscription.plan.name] ?? (0, 0)
914 byPlan[subscription.plan.name] = (previous.0 + 1, previous.1 + monthly)
915 }
916 emit("MRR")
917 for name in byPlan.keys.sorted() {
918 let (count, monthly) = byPlan[name]!
919 detail(name, "subs", count, "mrr", money(monthly))
920 }
921 let count = byPlan.values.reduce(0) { $0 + $1.0 }
922 let total = byPlan.values.reduce(0) { $0 + $1.1 }
923 detail("total subs", count, "mrr", money(total))
924 }

931 private func commandAging() {
932 var byCustomer: [String: [Int]] = [:]
933 var totals = [0, 0, 0]
934 for invoice in invoices where invoice.outstanding != 0 {
935 let age = day - invoice.day
936 let bucket = age <= 30 ? 0 : (age <= 60 ? 1 : 2)
937 var amounts = byCustomer[invoice.customer.name] ?? [0, 0, 0]
938 amounts[bucket] += invoice.outstanding
939 byCustomer[invoice.customer.name] = amounts
940 totals[bucket] += invoice.outstanding
941 }
942 emit("AGING", "day", day)
943 for name in byCustomer.keys.sorted() {
944 detail(name, agingParts(byCustomer[name]!))
945 }
946 detail("total", agingParts(totals))
947 }

994 func execute(_ line: String) {
995 let fields = line.split(separator: " ").map(String.init)
996 if fields.isEmpty { return }
997 let arguments = Array(fields.dropFirst())
998 do {
999 switch fields[0] {
1000 case "PLAN": try commandPlan(arguments)
1001 case "TIER": try commandTier(arguments)
1002 case "TAX": commandTax(arguments)
1003 case "CUSTOMER": try commandCustomer(arguments)
1004 case "COUPON": try commandCoupon(arguments)
1005 case "PAYMETHOD": try commandPayMethod(arguments)
1006 case "CREDIT": try commandCredit(arguments)
1007 case "DAY": try commandDay(arguments)
1008 case "SUBSCRIBE": try commandSubscribe(arguments)
1009 case "CHANGE": try commandChange(arguments)
1010 case "PAUSE": try commandPause(arguments)
1011 case "RESUME": try commandResume(arguments)
1012 case "CANCEL": try commandCancel(arguments)
1013 case "USAGE": try commandUsage(arguments)
1014 case "APPLY": try commandApply(arguments)
1015 case "PAY": try commandPay(arguments)
1016 case "REFUND": try commandRefund(arguments)
1017 case "INVOICE": try commandInvoice(arguments)
1018 case "SUB": try commandSub(arguments)
1019 case "PREVIEW": try commandPreview(arguments)
1020 case "STATEMENT": try commandStatement(arguments)
1021 case "LEDGER": try commandLedger(arguments)
1022 case "MRR": commandMRR()
1023 case "AGING": commandAging()
1024 case "DUNNING": commandDunning()
1025 case "REVENUE": commandRevenue()
1026 case "EVENTS": try commandEvents(arguments)
1027 default: break
1028 }
1029 } catch BillingError.invalid(let reason) {
1030 emit("ERROR", reason)
1031 } catch {
1032 fatalError("Unexpected error: \(error)")
1033 }
1034 }
1035}
