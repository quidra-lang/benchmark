9const PERIOD_LENGTHS = Dict("WEEKLY" => 7, "MONTHLY" => 30, "YEARLY" => 360)
10const ITEMS = Vector{Tuple{String,Int}}

12cents(text::String) = let parts = split(text, '.')
13 parse(Int, parts[1]) * 100 + parse(Int, parts[2])
14end
15
16function money(value::Int)
17 absolute = abs(value)
18 (value < 0 ? "-" : "") * @sprintf("%d.%02d", div(absolute, 100), mod(absolute, 100))
19end
20
21signed_money(value::Int) = (value > 0 ? "+" : "") * money(value)
22percentage(value::Int) = money(value) * "%"

36function Plan(name::String, price::Int, cycle::String)
37 Plan(name, price, cycle, PERIOD_LENGTHS[cycle], 0, 0, false, nothing,
38 Tuple{Union{Nothing,Int},Int,Int}[])
39end

41fee(plan::Plan, seats::Int) = plan.price * (plan.per_seat ? seats : 1)
42
43function usage_charge(plan::Plan, units::Int)
44 total, lower = 0, 0
45 for (index, (upper, unit_price, flat)) in enumerate(plan.tiers)
46 last = index == length(plan.tiers)
47 if plan.mode == "VOLUME"
48 if last || upper === nothing || units <= upper
49 return units * unit_price + flat
50 end
51 else
52 units <= lower && break
53 finish = last || upper === nothing ? units : min(units, upper)
54 total += (finish - lower) * unit_price + flat
55 if last || upper === nothing
56 break
57 end
58 lower = upper
59 end
60 end
61 total
62end

169 haskey(billing.coupons, name) || fail("unknown coupon " * name)

224function usage_items!(billing::Billing, sub::Subscription; close::Bool=false)
225 plan = billing.plans[sub.plan]
226 if plan.mode === nothing || sub.usage == 0
227 return ITEMS()
228 end
229 units = sub.usage
230 items = ITEMS([("usage $(plan.name) $units units", usage_charge(plan, units))])
231 close && (sub.usage = 0)
232 items

234
235coupon_usable(sub::Subscription, subtotal::Int) =
236 subtotal > 0 && sub.coupon !== nothing &&
237 (sub.remaining === nothing || sub.remaining > 0)
238
239function invoice_totals(billing::Billing, sub::Subscription, items::ITEMS)
240 subtotal = sum(amount for (_, amount) in items; init=0)
241 discount = 0
242 if coupon_usable(sub, subtotal)
243 coupon = billing.coupons[sub.coupon]
244 discount = coupon.kind == "PERCENT" ?
245 div(subtotal * coupon.value, 10000) : min(subtotal, coupon.value)
246 end
247 customer = billing.customers[sub.customer]
248 tax = div((subtotal - discount) * get(billing.taxes, customer.region, 0), 10000)
249 total = subtotal - discount + tax
250 credit = min(customer.balance, total)
251 (subtotal, discount, tax, total, credit, total - credit)
252end
253
254function issue!(billing::Billing, sub::Subscription, kind::String, items::ITEMS)
255 subtotal, discount, tax, total, credit, due = invoice_totals(billing, sub, items)
256 number = length(billing.invoices) + 1
257 invoice = Invoice(number, sub.number, sub.customer, billing.day, kind, items,
258 subtotal, discount, tax, total, credit, due,
259 0, 0, due, 0, Int[])
260 if coupon_usable(sub, subtotal) && sub.remaining !== nothing
261 sub.remaining -= 1
262 end
263 push!(billing.invoices, invoice)
264 push!(sub.invoices, number)
265 customer = billing.customers[sub.customer]
266 push!(customer.invoices, number)
267 balance_change!(billing, customer, -credit, "applied invoice $number")
268 result = "PAID"
269 if due == 0
270 result = "PAID"
271 elseif sub.state == "UNPAID"
272 result = "OPEN"
273 else
274 invoice.attempts = 1
275 if customer.method == "OK"
276 invoice.paid = due
277 invoice.outstanding = 0
278 result = "PAID"
279 else
280 result = "FAILED"
281 end
282 end
283 notice!(billing, "invoice", number, "sub", sub.number, kind, "total", money(total),
284 "credit", money(credit), "due", money(due), result)
285 if result == "FAILED"
286 if sub.state == "ACTIVE"
287 change_state!(billing, sub, "PAST_DUE")
288 end
289 if sub.state == "PAST_DUE"
290 append!(invoice.retries, (billing.day + 3, billing.day + 5, billing.day + 7))
291 end
292 end
293end
294
295function payment_complete!(billing::Billing, invoice::Invoice)
296 empty!(invoice.retries)
297 sub = billing.subscriptions[invoice.subscription]
298 if sub.state in ("PAST_DUE", "UNPAID") &&
299 all(billing.invoices[id].outstanding == 0 for id in sub.invoices)
300 sub.cancel_on = 0
301 change_state!(billing, sub, "ACTIVE")
302 end
303end
304
305function cancel_subscription!(billing::Billing, sub::Subscription)
306 change_state!(billing, sub, "CANCELED")
307 sub.canceled_on = billing.day
308 sub.pending = nothing
309 sub.cancel_end = false
310 sub.cancel_on = 0
311 items = usage_items!(billing, sub; close=true)
312 !isempty(items) && issue!(billing, sub, "final", items)
313end
314
315function retry!(billing::Billing, invoice::Invoice)
316 popfirst!(invoice.retries)
317 invoice.attempts += 1
318 attempt = invoice.attempts - 1
319 customer = billing.customers[invoice.customer]
320 if customer.method == "OK"
321 invoice.paid += invoice.outstanding
322 invoice.outstanding = 0
323 notice!(billing, "retry", attempt, "invoice", invoice.number, "PAID")
324 payment_complete!(billing, invoice)
325 else
326 notice!(billing, "retry", attempt, "invoice", invoice.number, "FAILED")
327 if isempty(invoice.retries)
328 sub = billing.subscriptions[invoice.subscription]
329 change_state!(billing, sub, "UNPAID")
330 sub.cancel_on = billing.day + 14
331 end
332 end
333end
334
335function advance_subscription!(billing::Billing, sub::Subscription)
336 sub.state in ("PAUSED", "CANCELED") && return
337 for id in sub.invoices
338 invoice = billing.invoices[id]
339 if !isempty(invoice.retries) && first(invoice.retries) == billing.day
340 retry!(billing, invoice)
341 end
342 end
343 if sub.state == "UNPAID" && sub.cancel_on == billing.day
344 cancel_subscription!(billing, sub)
345 end
346 if sub.state == "TRIALING" && sub.trial_end == billing.day
347 if sub.cancel_end
348 cancel_subscription!(billing, sub)
349 else
350 change_state!(billing, sub, "ACTIVE")
351 sub.start = billing.day
352 sub.finish = billing.day + billing.plans[sub.plan].length
353 issue!(billing, sub, "first", opening_items(billing, sub, billing.day))
354 end
355 end
356 if sub.state in ("ACTIVE", "PAST_DUE", "UNPAID") && sub.finish == billing.day
357 if sub.cancel_end
358 cancel_subscription!(billing, sub)
359 else
360 items = usage_items!(billing, sub; close=true)
361 if sub.pending !== nothing
362 plan, seats = sub.pending
363 notice!(billing, "sub", sub.number, sub.plan, "seats", sub.seats,
364 "->", plan, "seats", seats)
365 sub.plan, sub.seats = plan, seats
366 sub.pending = nothing
367 end
368 sub.start = billing.day
369 sub.finish = billing.day + billing.plans[sub.plan].length
370 push!(items, period_item(billing.plans[sub.plan], sub.seats, billing.day))
371 issue!(billing, sub, "renewal", items)
372 end
373 end
374end

376function cmd_plan!(billing::Billing, a::Vector{String})
377 name, amount, cycle = a[1:3]
378 haskey(billing.plans, name) && fail("duplicate plan " * name)
379 plan = Plan(name, cents(amount), cycle)
380 i = 4
381 while i <= length(a)
382 option = a[i]
383 if option == "TRIAL"
384 plan.trial = parse(Int, a[i+1])
385 i += 2
386 elseif option == "SETUP"
387 plan.setup = cents(a[i+1])
388 i += 2
389 else
390 if option == "PERSEAT"
391 plan.per_seat = true
392 else
393 plan.mode = option
394 end
395 i += 1
396 end
397 end
398 billing.plans[name] = plan
399 parts = Any["PLAN", name, cycle, amount]
400 plan.trial > 0 && append!(parts, ("trial", plan.trial))
401 plan.setup > 0 && append!(parts, ("setup", money(plan.setup)))
402 plan.per_seat && push!(parts, "perseat")
403 plan.mode !== nothing && push!(parts, lowercase(plan.mode))
404 emit!(billing, parts...)
405end

407function cmd_tier!(billing::Billing, a::Vector{String})
408 name, limit, unit, flat = a
409 plan = require_plan(billing, name)
410 plan.mode === nothing && fail("not metered " * name)
411 upper = limit == "INF" ? nothing : parse(Int, limit)
412 if !isempty(plan.tiers)
413 previous = last(plan.tiers)[1]
414 if previous === nothing || (upper !== nothing && upper <= previous)
415 fail("tier order " * name)
416 end
417 end
418 push!(plan.tiers, (upper, cents(unit), cents(flat)))
419 emit!(billing, "TIER", name, length(plan.tiers), "upto", limit, "unit", unit, "flat", flat)
420end

436function cmd_coupon!(billing::Billing, a::Vector{String})
437 name, kind, value, application = a[1:4]
438 haskey(billing.coupons, name) && fail("duplicate coupon " * name)
439 applications = application == "ONCE" ? 1 : nothing
440 index = 5
441 if application == "REPEATING"
442 applications = parse(Int, a[5])
443 index = 6
444 end
445 maximum = index <= length(a) ? parse(Int, a[index+1]) : nothing
446 numeric = kind == "PERCENT" ? parse(Int, value) : cents(value)
447 coupon = Coupon(name, kind, numeric, applications, maximum, 0)
448 billing.coupons[name] = coupon
449 shown = kind == "PERCENT" ? percentage(numeric) : money(numeric)
450 parts = Any["COUPON", name, lowercase(kind), shown, lowercase(application)]
451 application == "REPEATING" && push!(parts, applications)
452 maximum !== nothing && append!(parts, ("max", maximum))
453 emit!(billing, parts...)
454end

482function cmd_subscribe!(billing::Billing, a::Vector{String})
483 name, plan_name = a[1:2]
484 customer = require_customer(billing, name)
485 plan = require_plan(billing, plan_name)
486 seats, has_seats = 1, false
487 coupon_name = nothing
488 for i in 3:2:length(a)
489 if a[i] == "SEATS"
490 seats = parse(Int, a[i+1])
491 has_seats = true
492 else
493 coupon_name = a[i+1]
494 end
495 end
496 has_seats && !plan.per_seat && fail("not per-seat " * plan_name)
497 seats == 0 && fail("invalid seats")
498 coupon = coupon_name === nothing ? nothing : redeemable_coupon(billing, coupon_name)
499 remaining = coupon === nothing ? nothing : coupon.applications
500 sub = Subscription(length(billing.subscriptions)+1, name, plan_name, seats,
501 coupon_name, remaining)
502 coupon !== nothing && (coupon.redemptions += 1)
503 push!(billing.subscriptions, sub)
504 push!(customer.subscriptions, sub.number)
505 parts = Any["SUBSCRIBE", "sub", sub.number, name, plan_name, "seats", seats]
506 coupon !== nothing && append!(parts, ("coupon", coupon.name))
507 emit!(billing, parts...)
508 if plan.trial > 0
509 sub.trial_end = billing.day + plan.trial
510 change_state!(billing, sub, "TRIALING")
511 else
512 change_state!(billing, sub, "ACTIVE")
513 sub.start = billing.day
514 sub.finish = billing.day + plan.length
515 issue!(billing, sub, "first", opening_items(billing, sub, billing.day))
516 end
517end

519function cmd_change!(billing::Billing, a::Vector{String})
520 text, kind, value = a
521 sub = require_sub(billing, text, "ACTIVE", "TRIALING")
522 old_plan, old_seats = billing.plans[sub.plan], sub.seats
523 plan, seats = old_plan, old_seats
524 if kind == "PLAN"
525 plan = require_plan(billing, value)
526 seats = plan.per_seat ? old_seats : 1
527 else
528 old_plan.per_seat || fail("not per-seat " * old_plan.name)
529 seats = parse(Int, value)
530 seats == 0 && fail("invalid seats")
531 end
532 plan === old_plan && seats == old_seats && fail("no change " * text)
533 head = Any["CHANGE", "sub", sub.number, old_plan.name, "seats", old_seats,
534 "->", plan.name, "seats", seats]
535 sub.pending = nothing
536 upgrade = fee(plan, seats) * old_plan.length > fee(old_plan, old_seats) * plan.length
537 if sub.state == "TRIALING"
538 sub.plan, sub.seats = plan.name, seats
539 emit!(billing, head..., "now")
540 elseif upgrade
541 emit!(billing, head..., "now")
542 days = sub.finish - billing.day
543 items = usage_items!(billing, sub; close=true)
544 push!(items, ("unused $(old_plan.name) seats $old_seats $days days",
545 -div(fee(old_plan, old_seats)*days, old_plan.length)))
546 push!(items, ("remaining $(plan.name) seats $seats $days days",
547 div(fee(plan, seats)*days, plan.length)))
548 sub.plan, sub.seats = plan.name, seats
549 issue!(billing, sub, "upgrade", items)
550 else
551 sub.pending = (plan.name, seats)
552 emit!(billing, head..., "at", sub.finish)
553 end
554end

572function cmd_cancel!(billing::Billing, a::Vector{String})
573 text, when = a
574 sub = require_sub(billing, text)
575 if when == "END"
576 sub.state in ("ACTIVE", "TRIALING") || fail("invalid state $text $(sub.state)")
577 sub.cancel_end && fail("already canceling " * text)
578 sub.cancel_end = true
579 until = sub.state == "TRIALING" ? sub.trial_end : sub.finish
580 emit!(billing, "CANCEL", "sub", sub.number, "end", until)
581 else
582 sub.state == "CANCELED" && fail("invalid state $text $(sub.state)")
583 refund = 0
584 if sub.state in ("ACTIVE", "PAUSED")
585 start = sub.state == "ACTIVE" ? billing.day : sub.pause_start
586 plan = billing.plans[sub.plan]
587 refund = div(fee(plan, sub.seats)*(sub.finish - start), plan.length)
588 end
589 balance_change!(billing, billing.customers[sub.customer], refund,
590 "cancel sub $(sub.number)")
591 emit!(billing, "CANCEL", "sub", sub.number, "now", "credit", money(refund))
592 cancel_subscription!(billing, sub)
593 end
594end

626function cmd_refund!(billing::Billing, a::Vector{String})
627 text, amount = a
628 invoice = require_invoice(billing, text)
629 value = cents(amount)
630 value == 0 && fail("invalid amount")
631 maximum = invoice.total - invoice.refunded
632 value > maximum && fail("refund exceeds $text $(money(maximum))")
633 applied = min(value, invoice.outstanding)
634 invoice.refunded += value
635 invoice.outstanding -= applied
636 customer = billing.customers[invoice.customer]
637 balance_change!(billing, customer, value - applied, "refund invoice $(invoice.number)")
638 emit!(billing, "REFUND", "invoice", invoice.number, "amount", amount,
639 "applied", money(applied), "balance", money(customer.balance))
640 applied > 0 && invoice.outstanding == 0 && payment_complete!(billing, invoice)
641end

665function cmd_sub!(billing::Billing, a::Vector{String})
666 sub = require_sub(billing, a[1])
667 emit!(billing, "SUB", sub.number, sub.customer, sub.plan, "seats", sub.seats, sub.state)
668 if sub.state == "TRIALING"
669 indented!(billing, "trial until", sub.trial_end)
670 end
671 if sub.state in ("ACTIVE", "PAST_DUE", "UNPAID", "PAUSED")
672 indented!(billing, "period", period(sub))
673 end
674 if sub.state == "PAUSED"
675 indented!(billing, "paused since", sub.pause_start)
676 end
677 if sub.pending !== nothing
678 plan, seats = sub.pending
679 indented!(billing, "pending", plan, "seats", seats)
680 end
681 sub.cancel_end && indented!(billing, "cancel at end")
682 sub.cancel_on != 0 && indented!(billing, "cancel on", sub.cancel_on)
683 sub.state == "CANCELED" && indented!(billing, "canceled on", sub.canceled_on)
684 if sub.coupon !== nothing
685 if sub.remaining === nothing
686 indented!(billing, "coupon", sub.coupon, "forever")
687 elseif sub.remaining > 0
688 indented!(billing, "coupon", sub.coupon, "left", sub.remaining)
689 end
690 end
691 if billing.plans[sub.plan].mode !== nothing && sub.state != "CANCELED"
692 indented!(billing, "usage", sub.usage)
693 end
694 open = [id for id in sub.invoices if billing.invoices[id].outstanding > 0]
695 !isempty(open) && indented!(billing, "open", join(open, " "))
696end

698function cmd_preview!(billing::Billing, a::Vector{String})
699 sub = require_sub(billing, a[1], "TRIALING", "ACTIVE", "PAST_DUE", "UNPAID")
700 day = sub.state == "TRIALING" ? sub.trial_end : sub.finish
701 items = ITEMS()
702 kind = "renewal"
703 if sub.cancel_end
704 kind = "final"
705 items = sub.state == "TRIALING" ? ITEMS() : usage_items!(billing, sub)
706 elseif sub.state == "TRIALING"
707 kind = "first"
708 items = opening_items(billing, sub, day)
709 else
710 plan, seats = sub.pending === nothing ? (sub.plan, sub.seats) : sub.pending
711 items = usage_items!(billing, sub)
712 push!(items, period_item(billing.plans[plan], seats, day))
713 end
714 emit!(billing, "PREVIEW", "sub", sub.number, "day", day, kind)
715 if isempty(items)
716 indented!(billing, "(none)")
717 return
718 end
719 emit_items!(billing, items)
720 subtotal, discount, tax, total, credit, due = invoice_totals(billing, sub, items)
721 emit_totals!(billing, subtotal, discount, tax, total)
722 indented!(billing, "credit", money(credit), "due", money(due))
723end

725function cmd_statement!(billing::Billing, a::Vector{String})
726 customer = require_customer(billing, a[1])
727 emit!(billing, "STATEMENT", customer.name, "region", customer.region,
728 "tax", percentage(get(billing.taxes, customer.region, 0)),
729 "method", customer.method, "balance", money(customer.balance))
730 for id in customer.subscriptions
731 sub = billing.subscriptions[id]
732 indented!(billing, "sub", sub.number, sub.plan, sub.state)
733 end
734 outstanding = 0
735 for id in customer.invoices
736 invoice = billing.invoices[id]
737 indented!(billing, "invoice", id, "day", invoice.day, "sub", invoice.subscription,
738 invoice.kind, "total", money(invoice.total),
739 "outstanding", money(invoice.outstanding))
740 outstanding += invoice.outstanding
741 end
742 indented!(billing, "outstanding", money(outstanding))
743end

757function cmd_mrr!(billing::Billing, a::Vector{String})
758 rows = Dict{String,Tuple{Int,Int}}()
759 for sub in billing.subscriptions
760 sub.state in ("ACTIVE", "PAST_DUE") || continue
761 plan = billing.plans[sub.plan]
762 amount = fee(plan, sub.seats)
763 monthly = plan.cycle == "MONTHLY" ? amount :
764 plan.cycle == "YEARLY" ? div(amount, 12) : div(amount * 30, 7)
765 count, total = get(rows, plan.name, (0, 0))
766 rows[plan.name] = (count + 1, total + monthly)
767 end
768 emit!(billing, "MRR")
769 count, total = 0, 0
770 for name in sort(collect(keys(rows)))
771 subscriptions, monthly = rows[name]
772 indented!(billing, name, "subs", subscriptions, "mrr", money(monthly))
773 count += subscriptions
774 total += monthly
775 end
776 indented!(billing, "total subs", count, "mrr", money(total))
777end

779function aging_parts(amounts)
780 ("0-30", money(amounts[1]), "31-60", money(amounts[2]),
781 "61+", money(amounts[3]), "sum", money(sum(amounts)))
782end

784function cmd_aging!(billing::Billing, a::Vector{String})
785 rows = Dict{String,Vector{Int}}()
786 totals = zeros(Int, 3)
787 for invoice in billing.invoices
788 invoice.outstanding == 0 && continue
789 age = billing.day - invoice.day
790 bucket = age <= 30 ? 1 : age <= 60 ? 2 : 3
791 amounts = get!(rows, invoice.customer, zeros(Int, 3))
792 amounts[bucket] += invoice.outstanding
793 totals[bucket] += invoice.outstanding
794 end
795 emit!(billing, "AGING", "day", billing.day)
796 for name in sort(collect(keys(rows)))
797 indented!(billing, name, aging_parts(rows[name])...)
798 end
799 indented!(billing, "total", aging_parts(totals)...)
800end

831function cmd_events!(billing::Billing, a::Vector{String})
832 selected = isempty(a) ? nothing : require_sub(billing, a[1]).number
833 if selected === nothing
834 emit!(billing, "EVENTS")
835 else
836 emit!(billing, "EVENTS", "sub", selected)
837 end
838 found = false
839 for (day, number, old, new) in billing.events
840 if selected === nothing || number == selected
841 found = true
842 indented!(billing, "day", day, "sub", number, old, "->", new)
843 end
844 end
845 !found && indented!(billing, "(none)")
846end

861function execute!(billing::Billing, line::String)
862 words = String.(split(line))
863 command = words[1]
864 try
865 COMMANDS[command](billing, words[2:end])
866 catch error
867 if error isa BillingError
868 emit!(billing, "ERROR", error.message)
869 else
870 rethrow()
871 end
872 end
873end
