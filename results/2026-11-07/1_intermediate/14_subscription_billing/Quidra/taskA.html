78 return plan.price * (if plan.per_seat then seats else 1)

80int usage_charge(const Plan plan, int units)
81 result := 0
82 lower := 0
83 for: 0 <= k < int(#plan.tiers)
84 tier := plan.tiers[k]
85 last := k + 1 == int(#plan.tiers)
86 if plan.mode == "VOLUME"
87 if last or tier.upper < 0 or units <= tier.upper
88 return units * tier.unit_price + tier.flat_fee
89 else
90 if units <= lower
91 break
92 upper := if last or tier.upper < 0 then units else math.min(units, tier.upper)
93 result += (upper - lower) * tier.unit_price + tier.flat_fee
94 if last or tier.upper < 0
95 break
96 lower = tier.upper
97 return result

143 return subtotal > 0 and subscription.coupon >= 0 and subscription.remaining != 0

145int[] invoice_totals(const State state, int sub, const Item[] items)
146 subtotal := 0
147 for item := items[]
148 subtotal += item.amount
149 subscription := state.subscriptions[sub]
150 discount := 0
151 if coupon_usable(subscription, subtotal)
152 coupon := state.coupons[subscription.coupon]
153 discount = if coupon.kind == "PERCENT" then subtotal * coupon.value / 10000 else math.min(subtotal, coupon.value)
154 customer := state.customers[subscription.customer]
155 tax := (subtotal - discount) * tax_rate(state, customer.region) / 10000
156 total := subtotal - discount + tax
157 credit := math.min(customer.balance, total)
158 return [subtotal, discount, tax, total, credit, total - credit]

160void issue(State &state, int sub, string kind, const Item[] items)
161 totals := invoice_totals(state, sub, items)
162 index := int(#state.invoices)
163 subscription := state.subscriptions[sub]
164 if coupon_usable(subscription, totals[0]) and subscription.remaining > 0
165 state.subscriptions[sub].remaining -= 1
166 state.invoices = state.invoices.append(Invoice(sub, subscription.customer, state.day, kind, items, totals, 0, 0, totals[5], 0, []))
167 balance_change(&state, subscription.customer, -totals[4], "applied invoice " + string(index + 1))
168 result := "PAID"
169 if totals[5] > 0
170 if subscription.status == "UNPAID"
171 result = "OPEN"
172 else
173 state.invoices[index].attempts = 1
174 if state.customers[subscription.customer].method == "OK"
175 state.invoices[index].paid = totals[5]
176 state.invoices[index].outstanding = 0
177 else
178 result = "FAILED"
179 notice(state, "invoice " + string(index + 1) + " sub " + string(sub + 1) + " " + kind + " total " + money(totals[3]) + " credit " + money(totals[4]) + " due " + money(totals[5]) + " " + result)
180 if result == "FAILED"
181 if state.subscriptions[sub].status == "ACTIVE"
182 change_state(&state, sub, "PAST_DUE")
183 if state.subscriptions[sub].status == "PAST_DUE"
184 state.invoices[index].retries = [state.day + 3, state.day + 5, state.day + 7]

208void retry(State &state, int invoice)
209 bill := state.invoices[invoice]
210 state.invoices[invoice].retries = if #bill.retries <= 1 then [] else bill.retries[1:]
211 state.invoices[invoice].attempts += 1
212 attempt := state.invoices[invoice].attempts - 1
213 if state.customers[bill.customer].method == "OK"
214 state.invoices[invoice].paid += state.invoices[invoice].outstanding
215 state.invoices[invoice].outstanding = 0
216 notice(state, "retry " + string(attempt) + " invoice " + string(invoice + 1) + " PAID")
217 payment_complete(&state, invoice)
218 else
219 notice(state, "retry " + string(attempt) + " invoice " + string(invoice + 1) + " FAILED")
220 if #state.invoices[invoice].retries == 0
221 change_state(&state, bill.sub, "UNPAID")
222 state.subscriptions[bill.sub].cancel_on = state.day + 14

224void advance_subscription(State &state, int sub)
225 status := state.subscriptions[sub].status
226 if status == "PAUSED" or status == "CANCELED"
227 return
228 existing := int(#state.invoices)
229 for: 0 <= i < existing
230 if state.invoices[i].sub == sub and #state.invoices[i].retries > 0
231 if state.invoices[i].retries[0] == state.day
232 retry(&state, i)
233 if state.subscriptions[sub].status == "UNPAID" and state.subscriptions[sub].cancel_on == state.day
234 cancel_subscription(&state, sub)
235 if state.subscriptions[sub].status == "TRIALING" and state.subscriptions[sub].trial_end == state.day
236 if state.subscriptions[sub].cancel_end
237 cancel_subscription(&state, sub)
238 else
239 change_state(&state, sub, "ACTIVE")
240 state.subscriptions[sub].start = state.day
241 state.subscriptions[sub].end = state.day + period_length(state.plans[state.subscriptions[sub].plan].cycle)
242 issue(&state, sub, "first", opening_items(state, sub, state.day))
243 status = state.subscriptions[sub].status
244 if (status == "ACTIVE" or status == "PAST_DUE" or status == "UNPAID") and state.subscriptions[sub].end == state.day
245 if state.subscriptions[sub].cancel_end
246 cancel_subscription(&state, sub)
247 else
248 items := close_usage(&state, sub)
249 pending := state.subscriptions[sub].pending_plan
250 if pending >= 0
251 old := state.subscriptions[sub]
252 notice(state, "sub " + string(sub + 1) + " " + state.plans[old.plan].name + " seats " + string(old.seats) + " -> " + state.plans[pending].name + " seats " + string(old.pending_seats))
253 state.subscriptions[sub].plan = pending
254 state.subscriptions[sub].seats = old.pending_seats
255 state.subscriptions[sub].pending_plan = -1
256 plan := state.plans[state.subscriptions[sub].plan]
257 state.subscriptions[sub].start = state.day
258 state.subscriptions[sub].end = state.day + period_length(plan.cycle)
259 items = items.append(period_item(plan, state.subscriptions[sub].seats, state.day))
260 issue(&state, sub, "renewal", items)

269void command(State &state, string line)
270 words := line.split(" ")
271 action := words[0]
272 if action == "PLAN"
273 if plan_id(state, words[1]) >= 0
274 fail("duplicate plan " + words[1])
275 return
276 price := cents(words[2])
277 cycle := words[3]
278 trial := 0
279 setup := 0
280 per_seat := false
281 mode := ""
282 i := 4
283 while i < int(#words)
284 if words[i] == "TRIAL"
285 trial = number(words[i + 1])
286 i += 2
287 elif words[i] == "SETUP"
288 setup = cents(words[i + 1])
289 i += 2
290 elif words[i] == "PERSEAT"
291 per_seat = true
292 i += 1
293 else
294 mode = words[i]
295 i += 1
296 state.plans = state.plans.append(Plan(words[1], price, cycle, trial, setup, per_seat, mode, []))
297 emit("PLAN " + words[1] + " " + cycle + " " + money(price) + (if trial > 0 then " trial " + string(trial) else "") + (if setup > 0 then " setup " + money(setup) else "") + (if per_seat then " perseat" else "") + (if mode == "" then "" elif mode == "GRADUATED" then " graduated" else " volume"))
298 return
299 if action == "TIER"
300 plan := plan_id(state, words[1])
301 if plan < 0
302 fail("unknown plan " + words[1])
303 return
304 if state.plans[plan].mode == ""
305 fail("not metered " + words[1])
306 return
307 upper := if words[2] == "INF" then -1 else number(words[2])
308 tiers := state.plans[plan].tiers
309 if #tiers > 0
310 previous := tiers[-1].upper
311 if previous < 0 or (upper >= 0 and upper <= previous)
312 fail("tier order " + words[1])
313 return
314 state.plans[plan].tiers = tiers.append(Tier(upper, cents(words[3]), cents(words[4])))
315 emit("TIER " + words[1] + " " + string(#state.plans[plan].tiers) + " upto " + words[2] + " unit " + words[3] + " flat " + words[4])
316 return
317 if action == "TAX"
318 found := false
319 for: 0 <= i < int(#state.taxes)
320 if state.taxes[i].region == words[1]
321 state.taxes[i].rate = number(words[2])
322 found = true
323 if not found
324 state.taxes = state.taxes.append(Tax(words[1], number(words[2])))
325 emit("TAX " + words[1] + " " + percentage(number(words[2])))
326 return
327 if action == "CUSTOMER"
328 if customer_id(state, words[1]) >= 0
329 fail("duplicate customer " + words[1])
330 else
331 state.customers = state.customers.append(Customer(words[1], words[2], "OK", 0))
332 emit("CUSTOMER " + words[1] + " region " + words[2])
333 return
334 if action == "COUPON"
335 if coupon_id(state, words[1]) >= 0
336 fail("duplicate coupon " + words[1])
337 return
338 kind := words[2]
339 value := if kind == "PERCENT" then number(words[3]) else cents(words[3])
340 applications := if words[4] == "ONCE" then 1 else -1
341 offset := 5
342 if words[4] == "REPEATING"
343 applications = number(words[5])
344 offset = 6
345 maximum := if offset < int(#words) then number(words[offset + 1]) else -1
346 state.coupons = state.coupons.append(Coupon(words[1], kind, value, applications, maximum, 0))
347 emit("COUPON " + words[1] + " " + (if kind == "PERCENT" then "percent " + percentage(value) else "amount " + money(value)) + " " + (if words[4] == "ONCE" then "once" elif words[4] == "FOREVER" then "forever" else "repeating " + string(applications)) + (if maximum >= 0 then " max " + string(maximum) else ""))
348 return
349 if action == "DAY"
350 target := number(words[1])
351 if target <= state.day
352 fail("invalid day " + words[1])
353 return
354 emit("DAY " + string(target))
355 while state.day < target
356 state.day += 1
357 for: 0 <= sub < int(#state.subscriptions)
358 advance_subscription(&state, sub)
359 return
360 if action == "MRR"
361 emit("MRR")
362 names := [for plan := state.plans[] then plan.name]
363 names = names.sorted()
364 total_subs := 0
365 total_mrr := 0
366 for name := names[]
367 plan := plan_id(state, name)
368 entry := state.plans[plan]
369 count := 0
370 amount := 0
371 for sub := state.subscriptions[]
372 if sub.plan == plan and (sub.status == "ACTIVE" or sub.status == "PAST_DUE")
373 fee := plan_fee(entry, sub.seats)
374 amount += if entry.cycle == "MONTHLY" then fee elif entry.cycle == "YEARLY" then fee / 12 else fee * 30 / 7
375 count += 1
376 if count > 0
377 emit(" " + name + " subs " + string(count) + " mrr " + money(amount))
378 total_subs += count
379 total_mrr += amount
380 emit(" total subs " + string(total_subs) + " mrr " + money(total_mrr))
381 return
382 if action == "AGING"
383 emit("AGING day " + string(state.day))
384 names := [for customer := state.customers[] then customer.name]
385 names = names.sorted()
386 totals := array(3, fill = 0)
387 for name := names[]
388 customer := customer_id(state, name)
389 amounts := array(3, fill = 0)
390 for invoice := state.invoices[]
391 if invoice.customer == customer and invoice.outstanding > 0
392 age := state.day - invoice.day
393 bucket := if age <= 30 then 0 elif age <= 60 then 1 else 2
394 amounts[bucket] += invoice.outstanding
395 if amounts.sum() > 0
396 emit(" " + name + " 0-30 " + money(amounts[0]) + " 31-60 " + money(amounts[1]) + " 61+ " + money(amounts[2]) + " sum " + money(amounts.sum()))
397 for: 0 <= k < 3
398 totals[k] += amounts[k]
399 emit(" total 0-30 " + money(totals[0]) + " 31-60 " + money(totals[1]) + " 61+ " + money(totals[2]) + " sum " + money(totals.sum()))
400 return
401 if action == "DUNNING"
402 emit("DUNNING")
403 found := false
404 for: 0 <= k < int(#state.invoices)
405 invoice := state.invoices[k]
406 if #invoice.retries > 0
407 found = true
408 emit(" invoice " + string(k + 1) + " sub " + string(invoice.sub + 1) + " " + state.customers[invoice.customer].name + " outstanding " + money(invoice.outstanding) + " attempts " + string(invoice.attempts) + " next " + string(invoice.retries[0]))
409 if not found
410 emit(" (none)")
411 return
412 if action == "REVENUE"
413 sums := array(6, fill = 0)
414 paid := 0
415 refunded := 0
416 outstanding := 0
417 balances := 0
418 for invoice := state.invoices[]
419 for: 0 <= k < 6
420 sums[k] += invoice.totals[k]
421 paid += invoice.paid
422 refunded += invoice.refunded
423 outstanding += invoice.outstanding
424 for customer := state.customers[]
425 balances += customer.balance
426 emit("REVENUE")
427 emit(" invoices " + string(#state.invoices) + " subtotal " + money(sums[0]) + " discount " + money(sums[1]) + " tax " + money(sums[2]) + " total " + money(sums[3]))
428 emit(" credit " + money(sums[4]) + " paid " + money(paid) + " refunded " + money(refunded) + " outstanding " + money(outstanding) + " balances " + money(balances))
429 return
430 if action == "EVENTS"
431 selected := -1
432 if #words > 1
433 selected = number(words[1]) - 1
434 if selected < 0 or selected >= int(#state.subscriptions)
435 fail("unknown sub " + words[1])
436 return
437 emit("EVENTS" + (if selected >= 0 then " sub " + string(selected + 1) else ""))
438 found := false
439 for event := state.events[]
440 if selected < 0 or event.sub == selected
441 emit(" day " + string(event.day) + " sub " + string(event.sub + 1) + " " + event.before + " -> " + event.after)
442 found = true
443 if not found
444 emit(" (none)")
445 return
446 if action == "PAYMETHOD" or action == "CREDIT" or action == "STATEMENT" or action == "LEDGER"
447 customer := customer_id(state, words[1])
448 if customer < 0
449 fail("unknown customer " + words[1])
450 return
451 if action == "PAYMETHOD"
452 state.customers[customer].method = words[2]
453 emit("PAYMETHOD " + words[1] + " " + words[2])
454 elif action == "CREDIT"
455 amount := cents(words[2])
456 if amount == 0
457 fail("invalid amount")
458 else
459 balance_change(&state, customer, amount, "grant")
460 emit("CREDIT " + words[1] + " amount " + money(amount) + " balance " + money(state.customers[customer].balance))
461 elif action == "STATEMENT"
462 item := state.customers[customer]
463 emit("STATEMENT " + item.name + " region " + item.region + " tax " + percentage(tax_rate(state, item.region)) + " method " + item.method + " balance " + money(item.balance))
464 for: 0 <= k < int(#state.subscriptions)
465 sub := state.subscriptions[k]
466 if sub.customer == customer
467 emit(" sub " + string(k + 1) + " " + state.plans[sub.plan].name + " " + sub.status)
468 total := 0
469 for: 0 <= k < int(#state.invoices)
470 invoice := state.invoices[k]
471 if invoice.customer == customer
472 emit(" invoice " + string(k + 1) + " day " + string(invoice.day) + " sub " + string(invoice.sub + 1) + " " + invoice.kind + " total " + money(invoice.totals[3]) + " outstanding " + money(invoice.outstanding))
473 total += invoice.outstanding
474 emit(" outstanding " + money(total))
475 else
476 emit("LEDGER " + words[1] + " balance " + money(state.customers[customer].balance))
477 found := false
478 for row := state.balances[]
479 if row.customer == customer
480 emit(" day " + string(row.day) + " " + row.description + " " + signed_money(row.change) + " balance " + money(row.balance))
481 found = true
482 if not found
483 emit(" (none)")
484 return
485 if action == "SUBSCRIBE"
486 customer := customer_id(state, words[1])
487 if customer < 0
488 fail("unknown customer " + words[1])
489 return
490 plan := plan_id(state, words[2])
491 if plan < 0
492 fail("unknown plan " + words[2])
493 return
494 seats := 1
495 has_seats := false
496 coupon := -1
497 coupon_name := ""
498 for: 3 <= k < int(#words) step 2
499 if words[k] == "SEATS"
500 has_seats = true
501 seats = number(words[k + 1])
502 else
503 coupon_name = words[k + 1]
504 if has_seats and not state.plans[plan].per_seat
505 fail("not per-seat " + words[2])
506 return
507 if seats == 0
508 fail("invalid seats")
509 return
510 if coupon_name != ""
511 coupon = coupon_id(state, coupon_name)
512 if coupon < 0
513 fail("unknown coupon " + coupon_name)
514 return
515 item := state.coupons[coupon]
516 if item.maximum >= 0 and item.redemptions == item.maximum
517 fail("coupon exhausted " + coupon_name)
518 return
519 sub := int(#state.subscriptions)
520 remaining := -1
521 if coupon >= 0
522 remaining = state.coupons[coupon].applications
523 state.coupons[coupon].redemptions += 1
524 state.subscriptions = state.subscriptions.append(Subscription(customer, plan, seats, "NONE", -1, -1, -1, -1, -1, 0, false, -1, -1, coupon, remaining, 0))
525 emit("SUBSCRIBE sub " + string(sub + 1) + " " + words[1] + " " + words[2] + " seats " + string(seats) + (if coupon >= 0 then " coupon " + coupon_name else ""))
526 if state.plans[plan].trial > 0
527 state.subscriptions[sub].trial_end = state.day + state.plans[plan].trial
528 change_state(&state, sub, "TRIALING")
529 else
530 change_state(&state, sub, "ACTIVE")
531 state.subscriptions[sub].start = state.day
532 state.subscriptions[sub].end = state.day + period_length(state.plans[plan].cycle)
533 issue(&state, sub, "first", opening_items(state, sub, state.day))
534 return
535 if action == "INVOICE" or action == "PAY" or action == "REFUND"
536 invoice := number(words[1]) - 1
537 if invoice < 0 or invoice >= int(#state.invoices)
538 fail("unknown invoice " + words[1])
539 return
540 if action == "INVOICE"
541 bill := state.invoices[invoice]
542 emit("INVOICE " + words[1] + " sub " + string(bill.sub + 1) + " " + state.customers[bill.customer].name + " day " + string(bill.day) + " " + bill.kind + " " + (if bill.outstanding == 0 then "PAID" else "OPEN"))
543 emit_items(bill.items)
544 emit_totals(bill.totals)
545 emit(" credit " + money(bill.totals[4]) + " due " + money(bill.totals[5]) + " paid " + money(bill.paid) + " refunded " + money(bill.refunded) + " outstanding " + money(bill.outstanding))
546 elif action == "PAY"
547 if state.invoices[invoice].outstanding == 0
548 fail("already paid " + words[1])
549 else
550 amount := state.invoices[invoice].outstanding
551 emit("PAY invoice " + words[1] + " amount " + money(amount))
552 state.invoices[invoice].paid += amount
553 state.invoices[invoice].outstanding = 0
