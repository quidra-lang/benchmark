class Tier(int upper, int unit_price, int flat_fee)
class Plan(string name, int price, string cycle, int trial, int setup, bool per_seat, string mode, Tier[] tiers)
class Tax(string region, int rate)
class Customer(string name, string region, string method, int balance)
class Coupon(string name, string kind, int value, int applications, int maximum, int redemptions)
class Subscription(int customer, int plan, int seats, string status, int start, int end, int trial_end, int pause_start, int pending_plan, int pending_seats, bool cancel_end, int cancel_on, int canceled_on, int coupon, int remaining, int usage)
class Item(string description, int amount)
void emit(string text)
    println(text)
int tax_rate(const State state, string region)
    for item := state.taxes[]
        if item.region == region
            return item.rate
    return 0

int plan_fee(const Plan plan, int seats)
    return plan.price * (if plan.per_seat then seats else 1)
int usage_charge(const Plan plan, int units)
    for: 0 <= k < int(#plan.tiers)
        tier := plan.tiers[k]
        last := k + 1 == int(#plan.tiers)
        if plan.mode == "VOLUME"
            if last or tier.upper < 0 or units <= tier.upper
                return units * tier.unit_price + tier.flat_fee
Item period_item(const Plan plan, int seats, int start)
    return Item("charge " + plan.name + " seats " + string(seats) + " " + string(start) + "-" + string(start + period_length(plan.cycle)), plan_fee(plan, seats))

Item[] opening_items(const State state, int sub, int start)
    subscription := state.subscriptions[sub]
    plan := state.plans[subscription.plan]
    items := [period_item(plan, subscription.seats, start)]
    if plan.setup > 0
        items = items.append(Item("setup " + plan.name, plan.setup))
    return items
Item[] usage_items(const State state, int sub)
    subscription := state.subscriptions[sub]
    plan := state.plans[subscription.plan]
    if plan.mode == "" or subscription.usage == 0
        return []
    return [Item("usage " + plan.name + " " + string(subscription.usage) + " units", usage_charge(plan, subscription.usage))]
bool coupon_usable(const Subscription subscription, int subtotal)
    return subtotal > 0 and subscription.coupon >= 0 and subscription.remaining != 0

int[] invoice_totals(const State state, int sub, const Item[] items)
    subtotal := 0
    for item := items[]
        subtotal += item.amount
    subscription := state.subscriptions[sub]
    discount := 0
    if coupon_usable(subscription, subtotal)
        coupon := state.coupons[subscription.coupon]
        discount = if coupon.kind == "PERCENT" then subtotal * coupon.value / 10000 else math.min(subtotal, coupon.value)
    customer := state.customers[subscription.customer]
    tax := (subtotal - discount) * tax_rate(state, customer.region) / 10000
    total := subtotal - discount + tax
    credit := math.min(customer.balance, total)
    return [subtotal, discount, tax, total, credit, total - credit]
void emit_items(const Item[] items)
    for item := items[]
        emit("  " + item.description + " " + money(item.amount))
