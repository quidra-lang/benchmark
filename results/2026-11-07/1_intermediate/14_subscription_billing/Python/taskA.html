11def cents(text):

16def money(value):
17 sign = "-" if value < 0 else ""
18 value = abs(value)
19 return f"{sign}{value // 100}.{value % 100:02d}"

30class Plan:
31 def __init__(self, name, price, cycle):
32 self.name = name
33 self.price = price
34 self.cycle = cycle
35 self.length = PERIOD_LENGTHS[cycle]
36 self.trial = 0
37 self.setup = 0
38 self.per_seat = False
39 self.mode = None
40 self.tiers = []
41
42 def fee(self, seats):
43 return self.price * seats if self.per_seat else self.price
44
45 def usage_charge(self, units):
46 total = 0
47 lower = 0
48 for index, (upper, unit_price, flat_fee) in enumerate(self.tiers):
49 last = index == len(self.tiers) - 1
50 if self.mode == "VOLUME":
51 if last or upper is None or units <= upper:
52 return units * unit_price + flat_fee
53 else:
54 if units <= lower:
55 break
56 end = units if last or upper is None else min(units, upper)
57 total += (end - lower) * unit_price + flat_fee
58 if last or upper is None:
59 break
60 lower = upper
61 return total

207 def usage_items(self, subscription, close=False):
208 if subscription.plan.mode is None or subscription.usage == 0:
209 return []
210 units = subscription.usage
211 item = f"usage {subscription.plan.name} {units} units", subscription.plan.usage_charge(units)
212 if close:

216 def coupon_usable(self, subscription, subtotal):
217 return subtotal > 0 and subscription.coupon is not None and (subscription.remaining is None or subscription.remaining > 0)
218
219 def invoice_totals(self, subscription, items):
220 subtotal = sum(amount for _, amount in items)
221 discount = 0
222 if self.coupon_usable(subscription, subtotal):
223 coupon = subscription.coupon
224 discount = subtotal * coupon.value // 10000 if coupon.kind == "PERCENT" else min(subtotal, coupon.value)
225 tax = (subtotal - discount) * self.tax_rates.get(subscription.customer.region, 0) // 10000
226 total = subtotal - discount + tax
227 credit = min(subscription.customer.balance, total)
228 return subtotal, discount, tax, total, credit, total - credit

230 def issue(self, subscription, kind, items):
231 totals = self.invoice_totals(subscription, items)
232 invoice = Invoice(len(self.invoices) + 1, subscription, self.day, kind, items, totals)
233 if self.coupon_usable(subscription, invoice.subtotal) and subscription.remaining is not None:
234 subscription.remaining -= 1
235 self.invoices.append(invoice)
236 subscription.invoices.append(invoice)
237 subscription.customer.invoices.append(invoice)
238 self.balance_change(subscription.customer, -invoice.credit, f"applied invoice {invoice.number}")
239 if invoice.due == 0:
240 result = "PAID"
241 elif subscription.state == "UNPAID":
242 result = "OPEN"
243 else:
244 invoice.attempts = 1
245 if subscription.customer.method == "OK":
246 invoice.paid = invoice.due
247 invoice.outstanding = 0
248 result = "PAID"
249 else:
250 result = "FAILED"
251 self.notice(
252 "invoice", invoice.number, "sub", subscription.number, kind, "total", money(invoice.total),
253 "credit", money(invoice.credit), "due", money(invoice.due), result,
254 )
255 if result == "FAILED":
256 if subscription.state == "ACTIVE":
257 self.change_state(subscription, "PAST_DUE")
258 if subscription.state == "PAST_DUE":
259 invoice.retries = [self.day + offset for offset in (3, 5, 7)]

261 def payment_complete(self, invoice):
262 invoice.retries.clear()
263 subscription = invoice.subscription
264 if subscription.state in ("PAST_DUE", "UNPAID") and all(i.outstanding == 0 for i in subscription.invoices):
265 subscription.cancel_on = None
266 self.change_state(subscription, "ACTIVE")

268 def cancel_subscription(self, subscription):
269 self.change_state(subscription, "CANCELED")
270 subscription.canceled_on = self.day
271 subscription.pending = None
272 subscription.cancel_end = False
273 subscription.cancel_on = None
274 items = self.usage_items(subscription, close=True)
275 if items:
276 self.issue(subscription, "final", items)

278 def retry(self, invoice):
279 invoice.retries.pop(0)
280 invoice.attempts += 1
281 retry_number = invoice.attempts - 1
282 if invoice.customer.method == "OK":
283 invoice.paid += invoice.outstanding
284 invoice.outstanding = 0
285 self.notice("retry", retry_number, "invoice", invoice.number, "PAID")
286 self.payment_complete(invoice)
287 else:
288 self.notice("retry", retry_number, "invoice", invoice.number, "FAILED")
289 if not invoice.retries:
290 self.change_state(invoice.subscription, "UNPAID")
291 invoice.subscription.cancel_on = self.day + 14

293 def advance_subscription(self, subscription):
294 if subscription.state in ("PAUSED", "CANCELED"):
295 return
296 for invoice in subscription.invoices:
297 if invoice.retries and invoice.retries[0] == self.day:
298 self.retry(invoice)
299 if subscription.state == "UNPAID" and subscription.cancel_on == self.day:
300 self.cancel_subscription(subscription)
301 if subscription.state == "TRIALING" and subscription.trial_end == self.day:
302 if subscription.cancel_end:
303 self.cancel_subscription(subscription)
304 else:
305 self.change_state(subscription, "ACTIVE")
306 subscription.start = self.day
307 subscription.end = self.day + subscription.plan.length
308 self.issue(subscription, "first", self.opening_items(subscription, self.day))
309 if subscription.state in ("ACTIVE", "PAST_DUE", "UNPAID") and subscription.end == self.day:
310 if subscription.cancel_end:
311 self.cancel_subscription(subscription)
312 else:
313 items = self.usage_items(subscription, close=True)
314 if subscription.pending is not None:
315 plan, seats = subscription.pending
316 self.notice(
317 "sub", subscription.number, subscription.plan.name, "seats", subscription.seats,
318 "->", plan.name, "seats", seats,
319 )
320 subscription.plan = plan
321 subscription.seats = seats
322 subscription.pending = None
323 subscription.start = self.day
324 subscription.end = self.day + subscription.plan.length
325 items.append(self.period_item(subscription.plan, subscription.seats, self.day))
326 self.issue(subscription, "renewal", items)

328 def cmd_plan(self, name, amount, cycle, *options):
329 if name in self.plans:
330 raise CommandError(f"duplicate plan {name}")
331 plan = Plan(name, cents(amount), cycle)
332 index = 0
333 while index < len(options):
334 option = options[index]
335 if option == "TRIAL":
336 plan.trial = int(options[index + 1])
337 index += 2
338 elif option == "SETUP":
339 plan.setup = cents(options[index + 1])
340 index += 2
341 else:
342 if option == "PERSEAT":
343 plan.per_seat = True
344 else:
345 plan.mode = option
346 index += 1
347 self.plans[name] = plan
348 parts = ["PLAN", name, cycle, amount]
349 if plan.trial:
350 parts.extend(("trial", plan.trial))
351 if plan.setup:
352 parts.extend(("setup", money(plan.setup)))
353 if plan.per_seat:
354 parts.append("perseat")
355 if plan.mode is not None:
356 parts.append(plan.mode.lower())
357 self.emit(*parts)

359 def cmd_tier(self, name, limit, unit, flat):
360 plan = self.plan(name)
361 if plan.mode is None:
362 raise CommandError(f"not metered {name}")
363 upper = None if limit == "INF" else int(limit)
364 if plan.tiers:
365 previous = plan.tiers[-1][0]
366 if previous is None or (upper is not None and upper <= previous):
367 raise CommandError(f"tier order {name}")
368 plan.tiers.append((upper, cents(unit), cents(flat)))
369 self.emit("TIER", name, len(plan.tiers), "upto", limit, "unit", unit, "flat", flat)

424 def cmd_subscribe(self, name, plan_name, *options):
425 customer = self.customer(name)
426 plan = self.plan(plan_name)
427 seats = 1
428 has_seats = False
429 coupon_name = None
430 for index in range(0, len(options), 2):
431 if options[index] == "SEATS":
432 seats = int(options[index + 1])
433 has_seats = True
434 else:
435 coupon_name = options[index + 1]
436 if has_seats and not plan.per_seat:
437 raise CommandError(f"not per-seat {plan_name}")
438 if seats == 0:
439 raise CommandError("invalid seats")
440 coupon = None if coupon_name is None else self.redeemable_coupon(coupon_name)
441 subscription = Subscription(len(self.subscriptions) + 1, customer, plan, seats, coupon)
442 if coupon is not None:
443 coupon.redemptions += 1
444 self.subscriptions.append(subscription)
445 customer.subscriptions.append(subscription)
446 parts = ["SUBSCRIBE", "sub", subscription.number, name, plan_name, "seats", seats]
447 if coupon is not None:
448 parts.extend(("coupon", coupon.name))
449 self.emit(*parts)
450 if plan.trial:
451 subscription.trial_end = self.day + plan.trial
452 self.change_state(subscription, "TRIALING")
453 else:
454 self.change_state(subscription, "ACTIVE")
455 subscription.start = self.day
456 subscription.end = self.day + plan.length
457 self.issue(subscription, "first", self.opening_items(subscription, self.day))

459 def cmd_change(self, text, kind, value):
460 subscription = self.subscription(text, ("ACTIVE", "TRIALING"))
461 if kind == "PLAN":
462 plan = self.plan(value)
463 seats = subscription.seats if plan.per_seat else 1
464 else:
465 plan = subscription.plan
466 if not plan.per_seat:
467 raise CommandError(f"not per-seat {plan.name}")
468 seats = int(value)
469 if seats == 0:
470 raise CommandError("invalid seats")
471 if plan is subscription.plan and seats == subscription.seats:
472 raise CommandError(f"no change {text}")
473 old_plan, old_seats = subscription.plan, subscription.seats
474 head = ["CHANGE", "sub", subscription.number, old_plan.name, "seats", old_seats, "->", plan.name, "seats", seats]
475 subscription.pending = None
476 upgrade = plan.fee(seats) * old_plan.length > old_plan.fee(old_seats) * plan.length
477 if subscription.state == "TRIALING":
478 subscription.plan, subscription.seats = plan, seats
479 self.emit(*head, "now")
480 elif upgrade:
481 self.emit(*head, "now")
482 days = subscription.end - self.day
483 items = self.usage_items(subscription, close=True)
484 items.append((f"unused {old_plan.name} seats {old_seats} {days} days", -(old_plan.fee(old_seats) * days // old_plan.length)))
485 items.append((f"remaining {plan.name} seats {seats} {days} days", plan.fee(seats) * days // plan.length))
486 subscription.plan, subscription.seats = plan, seats
487 self.issue(subscription, "upgrade", items)
488 else:
489 subscription.pending = plan, seats
490 self.emit(*head, "at", subscription.end)

506 def cmd_cancel(self, text, when):
507 subscription = self.subscription(text)
508 if when == "END":
509 if subscription.state not in ("ACTIVE", "TRIALING"):
510 raise CommandError(f"invalid state {text} {subscription.state}")
511 if subscription.cancel_end:
512 raise CommandError(f"already canceling {text}")
513 subscription.cancel_end = True
514 day = subscription.trial_end if subscription.state == "TRIALING" else subscription.end
515 self.emit("CANCEL", "sub", subscription.number, "end", day)
516 else:
517 if subscription.state == "CANCELED":
518 raise CommandError(f"invalid state {text} {subscription.state}")
519 refund = 0
520 if subscription.state in ("ACTIVE", "PAUSED"):
521 start = self.day if subscription.state == "ACTIVE" else subscription.pause_start
522 refund = subscription.fee() * (subscription.end - start) // subscription.plan.length
523 self.balance_change(subscription.customer, refund, f"cancel sub {subscription.number}")
524 self.emit("CANCEL", "sub", subscription.number, "now", "credit", money(refund))
525 self.cancel_subscription(subscription)

556 def cmd_refund(self, text, amount):

562 if value > maximum:
563 raise CommandError(f"refund exceeds {text} {money(maximum)}")
564 applied = min(value, invoice.outstanding)

567 self.balance_change(invoice.customer, value - applied, f"refund invoice {invoice.number}")
568 self.emit(
569 "REFUND", "invoice", invoice.number, "amount", amount,
570 "applied", money(applied), "balance", money(invoice.customer.balance),

628 def cmd_preview(self, text):
629 subscription = self.subscription(text, ("TRIALING", "ACTIVE", "PAST_DUE", "UNPAID"))
630 day = subscription.trial_end if subscription.state == "TRIALING" else subscription.end
631 if subscription.cancel_end:
632 kind = "final"
633 items = [] if subscription.state == "TRIALING" else self.usage_items(subscription)
634 elif subscription.state == "TRIALING":
635 kind = "first"
636 items = self.opening_items(subscription, day)
637 else:
638 kind = "renewal"
639 plan, seats = subscription.pending if subscription.pending is not None else (subscription.plan, subscription.seats)
640 items = self.usage_items(subscription)
641 items.append(self.period_item(plan, seats, day))
642 self.emit("PREVIEW", "sub", subscription.number, "day", day, kind)
643 if not items:
644 self.emit("(none)", indent=2)
645 return

674 def cmd_mrr(self):
675 by_plan = {}
676 for subscription in self.subscriptions:
677 if subscription.state not in ("ACTIVE", "PAST_DUE"):
678 continue
679 fee = subscription.fee()
680 cycle = subscription.plan.cycle
681 monthly = fee if cycle == "MONTHLY" else fee // 12 if cycle == "YEARLY" else fee * 30 // 7
682 row = by_plan.setdefault(subscription.plan.name, [0, 0])
683 row[0] += 1
684 row[1] += monthly
685 self.emit("MRR")
686 for name in sorted(by_plan):
687 count, monthly = by_plan[name]
688 self.emit(name, "subs", count, "mrr", money(monthly), indent=2)
689 self.emit(
690 "total subs", sum(row[0] for row in by_plan.values()),
691 "mrr", money(sum(row[1] for row in by_plan.values())), indent=2,
692 )

697 def cmd_aging(self):
698 by_customer = {}
699 totals = [0, 0, 0]
700 for invoice in self.invoices:
701 if invoice.outstanding == 0:
702 continue
703 age = self.day - invoice.day
704 bucket = 0 if age <= 30 else 1 if age <= 60 else 2
705 amounts = by_customer.setdefault(invoice.customer.name, [0, 0, 0])
706 amounts[bucket] += invoice.outstanding
707 totals[bucket] += invoice.outstanding
708 self.emit("AGING", "day", self.day)
709 for name in sorted(by_customer):
710 self.emit(name, *self.aging_parts(by_customer[name]), indent=2)
711 self.emit("total", *self.aging_parts(totals), indent=2)

737 def cmd_events(self, *arguments):

749 def execute(self, line):

751 try:

753 except CommandError as error:
754 self.emit("ERROR", str(error))
