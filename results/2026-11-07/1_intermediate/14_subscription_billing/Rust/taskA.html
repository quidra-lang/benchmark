6fn number(text: &str) -> i64 {
7 text.parse().unwrap()
8}
9
10fn cents(text: &str) -> i64 {
11 let (whole, fraction) = text.split_once('.').unwrap();
12 number(whole) * 100 + number(fraction)
13}
14
15fn money(amount: i64) -> String {
16 let absolute = amount.abs();
17 format!("{}{}.{:02}", if amount < 0 { "-" } else { "" }, absolute / 100, absolute % 100)
18}
19
20fn percent(bp: i64) -> String {
21 format!("{}%", money(bp))
22}

62struct Plan {
63 price: i64,
64 length: i64,
65 trial: i64,
66 setup: i64,
67 per_seat: bool,
68 metering: Metering,
69 tiers: Vec<Tier>,
70}

72impl Plan {
73 fn fee(&self, seats: i64) -> i64 {
74 self.price * if self.per_seat { seats } else { 1 }
75 }
76
77 fn usage_charge(&self, usage: i64) -> i64 {
78 let mut lower = 0;
79 let mut charge = 0;
80 for (index, tier) in self.tiers.iter().enumerate() {
81 let upper = if index + 1 == self.tiers.len() {
82 usage
83 } else {
84 tier.limit.unwrap()
85 };
86 if self.metering == Metering::Volume {
87 if usage <= upper {
88 return usage * tier.unit + tier.flat;
89 }
90 } else if lower < usage {
91 charge += (usage.min(upper) - lower) * tier.unit + tier.flat;
92 }
93 lower = upper;
94 }
95 charge
96 }
97}

298 fn calculate(&self, index: usize, items: &[Item]) -> Charges {
299 let sub = &self.subs[index];
300 let customer = &self.customers[&sub.customer];
301 let subtotal: i64 = items.iter().map(|item| item.1).sum();
302 let mut discount = 0;
303 if let Some(applied) = &sub.coupon {
304 if applied.remaining != Some(0) && subtotal > 0 {
305 let coupon = &self.coupons[&applied.name];
306 discount = if coupon.percentage { subtotal * coupon.value / 10000 }
307 else { subtotal.min(coupon.value) };
308 }
309 }
310 let tax = (subtotal - discount) * self.taxes.get(&customer.region).copied().unwrap_or(0) / 10000;
311 let total = subtotal - discount + tax;
312 let credit = customer.balance.min(total);
313 Charges { subtotal, discount, tax, total, credit, due: total - credit }
314 }
315
316 fn issue(&mut self, index: usize, kind: &'static str, items: Vec<Item>) {
317 let charges = self.calculate(index, &items);
318 if charges.subtotal > 0 {
319 if let Some(applied) = &mut self.subs[index].coupon {
320 if let Some(left) = &mut applied.remaining {
321 if *left > 0 {
322 *left -= 1;
323 }
324 }
325 }
326 }
327 let customer = self.subs[index].customer.clone();
328 let id = self.invoices.len() + 1;
329 self.balance_change(&customer, format!("applied invoice {id}"), -charges.credit);
330 let (result, attempts, paid) = if charges.due == 0 {
331 ("PAID", 0, 0)
332 } else if self.subs[index].state == State::Unpaid {
333 ("OPEN", 0, 0)
334 } else if self.customers[&customer].payment_ok {
335 ("PAID", 1, charges.due)
336 } else {
337 ("FAILED", 1, 0)
338 };
339 println!(" day {} invoice {id} sub {} {kind} total {} credit {} due {} {result}",
340 self.day, index + 1, money(charges.total), money(charges.credit), money(charges.due));
341 self.invoices.push(Invoice {
342 sub: index, customer, day: self.day, kind, items,
343 outstanding: charges.due - paid, charges, paid, refunded: 0,
344 attempts, retries: VecDeque::new(),
345 });
346 if result == "FAILED" {
347 if self.subs[index].state == State::Active {
348 self.change_state(index, State::PastDue);
349 }
350 if self.subs[index].state == State::PastDue {
351 self.invoices[id - 1].retries = [self.day + 3, self.day + 5, self.day + 7].into();
352 }
353 }
354 }
355
356 fn payment_complete(&mut self, invoice: usize) {
357 self.invoices[invoice].retries.clear();
358 let index = self.invoices[invoice].sub;
359 if matches!(self.subs[index].state, State::PastDue | State::Unpaid)
360 && self.invoices.iter().all(|bill| bill.sub != index || bill.outstanding == 0)
361 {
362 self.subs[index].cancel_day = None;
363 self.change_state(index, State::Active);
364 }
365 }
366
367 fn cancel(&mut self, index: usize) {
368 self.change_state(index, State::Canceled);
369 let sub = &mut self.subs[index];
370 sub.canceled_day = Some(self.day);
371 sub.pending = None;
372 sub.cancel_end = false;
373 sub.cancel_day = None;
374 let items = self.close_usage(index);
375 if !items.is_empty() {
376 self.issue(index, "final", items);
377 }
378 }

380 fn advance(&mut self, target: i64) {
381 while self.day < target {
382 self.day += 1;
383 for index in 0..self.subs.len() {
384 if matches!(self.subs[index].state, State::Paused | State::Canceled) {
385 continue;
386 }
387 for invoice in 0..self.invoices.len() {
388 if self.invoices[invoice].sub != index
389 || self.invoices[invoice].retries.front() != Some(&self.day)
390 {
391 continue;
392 }
393 let bill = &mut self.invoices[invoice];
394 bill.retries.pop_front();
395 bill.attempts += 1;
396 let retry = bill.attempts - 1;
397 if self.customers[&bill.customer].payment_ok {
398 bill.paid += bill.outstanding;
399 bill.outstanding = 0;
400 println!(" day {} retry {retry} invoice {} PAID", self.day, invoice + 1);
401 self.payment_complete(invoice);
402 } else {
403 println!(" day {} retry {retry} invoice {} FAILED", self.day, invoice + 1);
404 if bill.retries.is_empty() {
405 self.change_state(index, State::Unpaid);
406 self.subs[index].cancel_day = Some(self.day + 14);
407 }
408 }
409 }
410 if self.subs[index].state == State::Unpaid && self.subs[index].cancel_day == Some(self.day) {
411 self.cancel(index);
412 }
413 if self.subs[index].state == State::Trialing && self.subs[index].trial_end == Some(self.day) {
414 if self.subs[index].cancel_end {
415 self.cancel(index);
416 } else {
417 self.change_state(index, State::Active);
418 let length = self.plans[&self.subs[index].plan].length;
419 self.subs[index].period = (self.day, self.day + length);
420 self.issue(index, "first", self.opening_items(index, self.day));
421 }
422 }
423 if matches!(self.subs[index].state, State::Active | State::PastDue | State::Unpaid)
424 && self.subs[index].period.1 == self.day
425 {
426 if self.subs[index].cancel_end {
427 self.cancel(index);
428 } else {
429 let mut items = self.close_usage(index);
430 let sub = &mut self.subs[index];
431 if let Some((plan, seats)) = sub.pending.take() {
432 println!(" day {} sub {} {} seats {} -> {plan} seats {seats}",
433 self.day, index + 1, sub.plan, sub.seats);
434 sub.plan = plan;
435 sub.seats = seats;
436 }
437 sub.period = (self.day, self.day + self.plans[&sub.plan].length);
438 items.push(self.charge_item(&self.subs[index].plan, self.subs[index].seats, self.day));
439 self.issue(index, "renewal", items);
440 }
441 }
442 }
443 }
444 }

501 fn preview(&self, index: usize) {
502 let sub = &self.subs[index];
503 let due = if sub.state == State::Trialing { sub.trial_end.unwrap() } else { sub.period.1 };
504 let (kind, items) = if sub.cancel_end {

506 else { self.usage_item(index).into_iter().collect() })
507 } else if sub.state == State::Trialing {
508 ("first", self.opening_items(index, due))
509 } else {
510 let mut items: Vec<Item> = self.usage_item(index).into_iter().collect();
511 let (plan, seats) = sub.pending.as_ref()
512 .map(|(plan, seats)| (plan.as_str(), *seats)).unwrap_or((&sub.plan, sub.seats));
513 items.push(self.charge_item(plan, seats, due));
514 ("renewal", items)
515 };
516 println!("PREVIEW sub {} day {due} {kind}", index + 1);
517 if items.is_empty() {
518 println!(" (none)");
519 } else {
520 for (description, amount) in &items {
521 println!(" {description} {}", money(*amount));
522 }
523 let charges = self.calculate(index, &items);
524 charges.print();
525 println!(" credit {} due {}", money(charges.credit), money(charges.due));
526 }

550 fn mrr(&self) {
551 let mut plans: BTreeMap<&str, (usize, i64)> = BTreeMap::new();
552 let (mut count, mut total) = (0, 0);
553 for sub in &self.subs {
554 if matches!(sub.state, State::Active | State::PastDue) {
555 let plan = &self.plans[&sub.plan];
556 let fee = plan.fee(sub.seats);
557 let monthly = match plan.length {
558 7 => fee * 30 / 7,
559 360 => fee / 12,
560 _ => fee,
561 };
562 let row = plans.entry(&sub.plan).or_default();
563 row.0 += 1;
564 row.1 += monthly;
565 count += 1;
566 total += monthly;
567 }
568 }
569 println!("MRR");
570 for (plan, (count, total)) in plans {
571 println!(" {plan} subs {count} mrr {}", money(total));
572 }
573 println!(" total subs {count} mrr {}", money(total));
574 }

576 fn aging(&self) {
577 let mut customers: BTreeMap<&str, [i64; 3]> = BTreeMap::new();
578 let mut total = [0; 3];
579 for bill in &self.invoices {
580 if bill.outstanding > 0 {
581 let age = self.day - bill.day;
582 let bucket = if age <= 30 { 0 } else if age <= 60 { 1 } else { 2 };
583 customers.entry(&bill.customer).or_default()[bucket] += bill.outstanding;
584 total[bucket] += bill.outstanding;
585 }
586 }
587 println!("AGING day {}", self.day);
588 for (name, buckets) in customers.into_iter().chain(std::iter::once(("total", total))) {
589 println!(" {name} 0-30 {} 31-60 {} 61+ {} sum {}", money(buckets[0]),
590 money(buckets[1]), money(buckets[2]), money(buckets.iter().sum()));
591 }

611 fn command(&mut self, parts: &[&str]) -> Result<(), String> {
612 let a = &parts[1..];
613 match parts[0] {
614 "PLAN" => {
615 if self.plans.contains_key(a[0]) {
616 return Err(format!("duplicate plan {}", a[0]));
617 }
618 let mut plan = Plan {
619 price: cents(a[1]), length: match a[2] { "WEEKLY" => 7, "MONTHLY" => 30, _ => 360 },
620 trial: 0, setup: 0, per_seat: false, metering: Metering::None, tiers: Vec::new(),
621 };
622 let mut position = 3;
623 while position < a.len() {
624 match a[position] {
625 "TRIAL" => { plan.trial = number(a[position + 1]); position += 1; }
626 "SETUP" => { plan.setup = cents(a[position + 1]); position += 1; }
627 "PERSEAT" => plan.per_seat = true,
628 "GRADUATED" => plan.metering = Metering::Graduated,
629 "VOLUME" => plan.metering = Metering::Volume,
630 _ => unreachable!(),
631 }
632 position += 1;
633 }
634 let mut line = format!("PLAN {} {} {}", a[0], a[2], money(plan.price));
635 if plan.trial > 0 { line.push_str(&format!(" trial {}", plan.trial)); }
636 if plan.setup > 0 { line.push_str(&format!(" setup {}", money(plan.setup))); }
637 if plan.per_seat { line.push_str(" perseat"); }
638 match plan.metering {
639 Metering::Graduated => line.push_str(" graduated"),
640 Metering::Volume => line.push_str(" volume"),
641 Metering::None => {}
642 }
643 self.plans.insert(a[0].into(), plan);
644 println!("{line}");
645 }
646 "TIER" => {
647 let plan = self.plan(a[0])?;
648 if plan.metering == Metering::None { return Err(format!("not metered {}", a[0])); }
649 let limit = if a[1] == "INF" { None } else { Some(number(a[1])) };
650 if let Some(last) = plan.tiers.last() {
651 if last.limit.is_none() || limit.is_some_and(|value| value <= last.limit.unwrap()) {
652 return Err(format!("tier order {}", a[0]));
653 }
654 }
655 let plan = self.plans.get_mut(a[0]).unwrap();
656 plan.tiers.push(Tier { limit, unit: cents(a[2]), flat: cents(a[3]) });
657 println!("TIER {} {} upto {} unit {} flat {}", a[0], plan.tiers.len(), a[1], a[2], a[3]);
658 }
659 "TAX" => {
660 let rate = number(a[1]);
661 self.taxes.insert(a[0].into(), rate);
662 println!("TAX {} {}", a[0], percent(rate));
663 }
664 "CUSTOMER" => {
665 if self.customers.contains_key(a[0]) { return Err(format!("duplicate customer {}", a[0])); }
666 self.customers.insert(a[0].into(), Customer {
667 region: a[1].into(), payment_ok: true, balance: 0, ledger: Vec::new(),
668 });
669 println!("CUSTOMER {} region {}", a[0], a[1]);
670 }
671 "COUPON" => {
672 if self.coupons.contains_key(a[0]) { return Err(format!("duplicate coupon {}", a[0])); }
673 let percentage = a[1] == "PERCENT";
674 let value = if percentage { number(a[2]) } else { cents(a[2]) };
675 let (applications, application, position) = match a[3] {
676 "ONCE" => (Some(1), "once".to_string(), 4),
677 "FOREVER" => (None, "forever".to_string(), 4),
678 _ => (Some(number(a[4])), format!("repeating {}", a[4]), 5),
679 };
680 let limit = if position < a.len() { Some(number(a[position + 1])) } else { None };
681 self.coupons.insert(a[0].into(), Coupon { percentage, value, applications, limit, redemptions: 0 });
682 let discount = if percentage { format!("percent {}", percent(value)) }
683 else { format!("amount {}", money(value)) };
684 println!("COUPON {} {discount} {application}{}", a[0],
685 limit.map(|count| format!(" max {count}")).unwrap_or_default());
686 }
687 "PAYMETHOD" => {
688 self.customer(a[0])?;
689 self.customers.get_mut(a[0]).unwrap().payment_ok = a[1] == "OK";
690 println!("PAYMETHOD {} {}", a[0], a[1]);
691 }
692 "CREDIT" => {
693 self.customer(a[0])?;
694 let amount = cents(a[1]);
695 if amount == 0 { return Err("invalid amount".into()); }
696 self.balance_change(a[0], "grant".into(), amount);
697 println!("CREDIT {} amount {} balance {}", a[0], money(amount), money(self.customers[a[0]].balance));
698 }
699 "DAY" => {
700 let day = number(a[0]);
701 if day <= self.day { return Err(format!("invalid day {}", a[0])); }
702 println!("DAY {day}");
703 self.advance(day);
704 }
705 "SUBSCRIBE" => {
706 self.customer(a[0])?;
707 let plan = self.plan(a[1])?;
708 let mut position = 2;
709 let explicit_seats = a.get(position) == Some(&"SEATS");
710 if explicit_seats && !plan.per_seat { return Err(format!("not per-seat {}", a[1])); }
711 let seats = if explicit_seats { position += 2; number(a[3]) } else { 1 };
712 if seats == 0 { return Err("invalid seats".into()); }
713 let coupon_name = if position < a.len() { Some(a[position + 1]) } else { None };
714 if let Some(name) = coupon_name { self.check_coupon(name)?; }
715 let (trial, length) = (plan.trial, plan.length);
716 let coupon = coupon_name.map(|name| self.redeem(name));
717 let index = self.subs.len();
718 self.subs.push(Subscription {
719 customer: a[0].into(), plan: a[1].into(), seats, state: State::None,
720 period: (self.day, self.day + length),
721 trial_end: if trial > 0 { Some(self.day + trial) } else { None },
722 paused_since: None, pending: None, cancel_end: false,
723 cancel_day: None, canceled_day: None, coupon, usage: 0,
724 });
725 println!("SUBSCRIBE sub {} {} {} seats {seats}{}", index + 1, a[0], a[1],
726 coupon_name.map(|name| format!(" coupon {name}")).unwrap_or_default());
727 if trial > 0 {
728 self.change_state(index, State::Trialing);
729 } else {
730 self.change_state(index, State::Active);
731 self.issue(index, "first", self.opening_items(index, self.day));
732 }
733 }
734 "CHANGE" => {
735 let index = self.sub_index(a[0])?;
736 self.check_state(index, &[State::Active, State::Trialing])?;
737 let old = self.subs[index].clone();
738 let old_plan = &self.plans[&old.plan];
739 let (new_plan, seats) = if a[1] == "PLAN" {
740 let plan = self.plan(a[2])?;
741 (a[2], if plan.per_seat { old.seats } else { 1 })
742 } else {
743 if !old_plan.per_seat { return Err(format!("not per-seat {}", old.plan)); }
744 let seats = number(a[2]);
745 if seats == 0 { return Err("invalid seats".into()); }
746 (old.plan.as_str(), seats)
747 };
748 if new_plan == old.plan && seats == old.seats { return Err(format!("no change {}", a[0])); }
749 let (old_fee, old_length) = (old_plan.fee(old.seats), old_plan.length);
750 let plan = &self.plans[new_plan];
751 let (new_fee, new_length) = (plan.fee(seats), plan.length);
752 self.subs[index].pending = None;
753 let head = format!("CHANGE sub {} {} seats {} -> {new_plan} seats {seats}", a[0], old.plan, old.seats);
754 if old.state == State::Trialing {
755 self.subs[index].plan = new_plan.into();
756 self.subs[index].seats = seats;
757 println!("{head} now");
758 } else if new_fee * old_length > old_fee * new_length {
759 println!("{head} now");
760 let remaining = old.period.1 - self.day;
761 let mut items = self.close_usage(index);
762 items.push((format!("unused {} seats {} {remaining} days", old.plan, old.seats),
763 -(old_fee * remaining / old_length)));
764 items.push((format!("remaining {new_plan} seats {seats} {remaining} days"),
765 new_fee * remaining / new_length));
766 self.subs[index].plan = new_plan.into();
767 self.subs[index].seats = seats;
768 self.issue(index, "upgrade", items);
769 } else {
770 self.subs[index].pending = Some((new_plan.into(), seats));
771 println!("{head} at {}", old.period.1);
772 }
773 }
774 "PAUSE" => {
775 let index = self.sub_index(a[0])?;
776 self.check_state(index, &[State::Active])?;
777 self.subs[index].paused_since = Some(self.day);
778 println!("PAUSE sub {}", a[0]);
779 self.change_state(index, State::Paused);
780 }
781 "RESUME" => {
782 let index = self.sub_index(a[0])?;
783 self.check_state(index, &[State::Paused])?;
784 let sub = &mut self.subs[index];
785 let shift = self.day - sub.paused_since.take().unwrap();
786 sub.period.1 += shift;
787 println!("RESUME sub {} shift {shift} period {}-{}", a[0], sub.period.0, sub.period.1);
788 self.change_state(index, State::Active);
789 }
790 "CANCEL" => {
791 let index = self.sub_index(a[0])?;
792 let sub = self.subs[index].clone();
793 if a[1] == "END" {
794 self.check_state(index, &[State::Active, State::Trialing])?;
795 if sub.cancel_end { return Err(format!("already canceling {}", a[0])); }
796 self.subs[index].cancel_end = true;
797 let day = if sub.state == State::Trialing { sub.trial_end.unwrap() } else { sub.period.1 };
798 println!("CANCEL sub {} end {day}", a[0]);
799 } else {
800 if sub.state == State::Canceled { return Err(format!("invalid state {} CANCELED", a[0])); }
801 let plan = &self.plans[&sub.plan];
802 let remaining = match sub.state {
803 State::Active => sub.period.1 - self.day,
804 State::Paused => sub.period.1 - sub.paused_since.unwrap(),
805 _ => 0,
806 };
807 let refund = plan.fee(sub.seats) * remaining / plan.length;
808 self.balance_change(&sub.customer, format!("cancel sub {}", a[0]), refund);
809 println!("CANCEL sub {} now credit {}", a[0], money(refund));
810 self.cancel(index);
811 }
812 }
813 "USAGE" => {
814 let index = self.sub_index(a[0])?;
815 self.check_state(index, &[State::Active, State::PastDue])?;
816 let sub = &mut self.subs[index];
817 if self.plans[&sub.plan].metering == Metering::None { return Err(format!("not metered {}", sub.plan)); }
818 let units = number(a[1]);
819 if units == 0 { return Err("invalid units".into()); }
820 sub.usage += units;
821 println!("USAGE sub {} units {units} total {}", a[0], sub.usage);
822 }
823 "APPLY" => {
824 let index = self.sub_index(a[0])?;
825 if self.subs[index].state == State::Canceled { return Err(format!("invalid state {} CANCELED", a[0])); }
826 self.check_coupon(a[1])?;
827 let coupon = self.redeem(a[1]);
828 self.subs[index].coupon = Some(coupon);
829 println!("APPLY sub {} coupon {}", a[0], a[1]);
830 }
831 "PAY" => {
832 let index = self.invoice_index(a[0])?;
833 let bill = &mut self.invoices[index];
834 if bill.outstanding == 0 { return Err(format!("already paid {}", a[0])); }
835 println!("PAY invoice {} amount {}", a[0], money(bill.outstanding));
836 bill.paid += bill.outstanding;
837 bill.outstanding = 0;
838 self.payment_complete(index);
839 }
840 "REFUND" => {
841 let index = self.invoice_index(a[0])?;
842 let amount = cents(a[1]);
843 if amount == 0 { return Err("invalid amount".into()); }
844 let bill = &mut self.invoices[index];
845 if amount > bill.charges.total - bill.refunded {
846 return Err(format!("refund exceeds {} {}", a[0], money(bill.charges.total - bill.refunded)));
847 }
848 let applied = amount.min(bill.outstanding);
849 bill.refunded += amount;
850 bill.outstanding -= applied;
851 let completed = applied > 0 && bill.outstanding == 0;
852 let customer = bill.customer.clone();
853 self.balance_change(&customer, format!("refund invoice {}", a[0]), amount - applied);
854 println!("REFUND invoice {} amount {} applied {} balance {}", a[0], money(amount),
855 money(applied), money(self.customers[&customer].balance));
856 if completed { self.payment_complete(index); }
857 }
858 "INVOICE" => self.show_invoice(self.invoice_index(a[0])?),
859 "SUB" => self.show_sub(self.sub_index(a[0])?),
860 "PREVIEW" => {
861 let index = self.sub_index(a[0])?;
862 self.check_state(index, &[State::Trialing, State::Active, State::PastDue, State::Unpaid])?;
863 self.preview(index);
864 }
865 "STATEMENT" => {
866 self.customer(a[0])?;
867 self.statement(a[0]);
868 }
869 "LEDGER" => {
870 let customer = self.customer(a[0])?;
871 println!("LEDGER {} balance {}", a[0], money(customer.balance));
872 for (day, description, change, balance) in &customer.ledger {
873 println!(" day {day} {description} {}{} balance {}", if *change > 0 { "+" } else { "" },
874 money(*change), money(*balance));
875 }
876 if customer.ledger.is_empty() { println!(" (none)"); }
877 }
878 "MRR" => self.mrr(),
879 "AGING" => self.aging(),
880 "DUNNING" => {
881 println!("DUNNING");
882 let mut found = false;
883 for (index, bill) in self.invoices.iter().enumerate() {
884 if let Some(next) = bill.retries.front() {
885 found = true;
886 println!(" invoice {} sub {} {} outstanding {} attempts {} next {next}",
887 index + 1, bill.sub + 1, bill.customer, money(bill.outstanding), bill.attempts);
888 }
889 }
890 if !found { println!(" (none)"); }
891 }
892 "REVENUE" => self.revenue(),
893 "EVENTS" => {
894 let selected = if a.is_empty() { None } else { Some(self.sub_index(a[0])?) };
895 println!("EVENTS{}", selected.map(|id| format!(" sub {}", id + 1)).unwrap_or_default());
896 let mut found = false;
897 for (day, index, old, new) in &self.events {
898 if selected.is_none() || selected == Some(*index) {
899 found = true;
900 println!(" day {day} sub {} {} -> {}", index + 1, old.name(), new.name());
901 }
902 }
903 if !found { println!(" (none)"); }
904 }
905 _ => unreachable!(),
906 }
907 Ok(())
908 }
909}
