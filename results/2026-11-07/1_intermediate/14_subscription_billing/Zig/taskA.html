20fn integer(s: []const u8) i64 {
21 return std.fmt.parseInt(i64, s, 10) catch unreachable;
22}
23fn amount(s: []const u8) i64 {
24 const dot = std.mem.findScalar(u8, s, '.').?;
25 return integer(s[0..dot]) * 100 + integer(s[dot + 1 ..]);
26}
27fn equal(a: []const u8, b: []const u8) bool {

30const Tier = struct { limit: ?i64, unit: i64, flat: i64 };
31const Plan = struct {
32 name: []const u8,
33 price: i64,
34 cycle: Cycle,
35 trial: i64 = 0,
36 setup: i64 = 0,
37 per_seat: bool = false,
38 mode: ?Mode = null,
39 tiers: std.ArrayList(Tier) = .empty,
40 fn fee(p: *const Plan, seats: i64) i64 {
41 return p.price * (if (p.per_seat) seats else 1);
42 }
43 fn usageCharge(p: *const Plan, units: i64) i64 {
44 var total: i64 = 0;
45 var lower: i64 = 0;
46 for (p.tiers.items, 0..) |t, i| {
47 const upper = if (i + 1 == p.tiers.items.len) units else t.limit.?;
48 if (p.mode.? == .VOLUME) {
49 if (units <= upper) return units * t.unit + t.flat;
50 } else if (lower < units) {
51 total += (@min(units, upper) - lower) * t.unit + t.flat;
52 }
53 lower = upper;
54 }
55 return total;
56 }

124 fn fail(self: *Billing, comptime fmt: []const u8, args: anytype) Error {
125 self.error_message = std.fmt.allocPrint(self.gpa, fmt, args) catch return error.OutOfMemory;
126 return error.BadCommand;

128 fn money(self: *Billing, n: i64) Allocator.Error![]const u8 {
129 const u = @abs(n);
130 return std.fmt.allocPrint(self.gpa, "{s}{d}.{d:0>2}", .{ if (n < 0) "-" else "", u / 100, u % 100 });
131 }
132 fn rate(self: *Billing, n: i64) Allocator.Error![]const u8 {
133 return std.fmt.allocPrint(self.gpa, "{s}%", .{try self.money(n)});
134 }
135 fn ordered(self: *Billing, comptime T: type, values: []const *T) Allocator.Error![]*T {
136 const result = try self.gpa.dupe(*T, values);
137 std.mem.sort(*T, result, {}, struct {
138 fn less(_: void, a: *T, b: *T) bool {
139 return std.mem.order(u8, a.name, b.name) == .lt;
140 }
141 }.less);
142 return result;
143 }

153 if (c.redemptions == maximum) return self.fail("coupon exhausted {s}", .{name});
154 }
155 return c;

167 fn requireState(self: *Billing, s: *Subscription, states: []const State) Error!void {
168 for (states) |state| if (s.state == state) return;
169 return self.fail("invalid state {d} {s}", .{ s.id, @tagName(s.state) });
170 }
171 fn balance(self: *Billing, c: *Customer, change: i64, comptime fmt: []const u8, args: anytype) Error!void {
172 if (change == 0) return;
173 c.balance += change;
174 try c.log.append(self.gpa, .{ .day = self.day, .description = try std.fmt.allocPrint(self.gpa, fmt, args), .change = change, .balance = c.balance });
175 }
176 fn transition(self: *Billing, s: *Subscription, to: State) Error!void {
177 const from = s.state;
178 try self.out.print(" day {d} sub {d} {s} -> {s}\n", .{ self.day, s.id, @tagName(from), @tagName(to) });
179 try self.events.append(self.gpa, .{ .day = self.day, .sub_id = s.id, .from = from, .to = to });
180 if (from == .PAST_DUE) {
181 for (s.invoices.items) |inv| inv.next_retry = 3;
182 }
183 s.state = to;
184 }
185 fn completed(self: *Billing, inv: *Invoice) Error!void {
186 inv.next_retry = 3;
187 const s = inv.sub;
188 if (s.state != .PAST_DUE and s.state != .UNPAID) return;
189 for (s.invoices.items) |other| if (other.outstanding > 0) return;
190 s.cancel_day = null;
191 try self.transition(s, .ACTIVE);
192 }
193 fn item(self: *Billing, items: *std.ArrayList(Item), value: i64, comptime fmt: []const u8, args: anytype) Error!void {
194 try items.append(self.gpa, .{ .description = try std.fmt.allocPrint(self.gpa, fmt, args), .amount = value });
195 }
196 fn charge(self: *Billing, items: *std.ArrayList(Item), p: *Plan, seats: i64, start: i64) Error!void {
197 try self.item(items, p.fee(seats), "charge {s} seats {d} {d}-{d}", .{ p.name, seats, start, start + p.cycle.days() });
198 }
199 fn opening(self: *Billing, s: *Subscription, start: i64) Error!std.ArrayList(Item) {
200 var items: std.ArrayList(Item) = .empty;
201 try self.charge(&items, s.plan, s.seats, start);
202 if (s.plan.setup > 0) try self.item(&items, s.plan.setup, "setup {s}", .{s.plan.name});
203 return items;
204 }
205 fn usageItem(self: *Billing, items: *std.ArrayList(Item), s: *Subscription, close: bool) Error!void {
206 if (s.plan.mode == null or s.usage == 0) return;
207 try self.item(items, s.plan.usageCharge(s.usage), "usage {s} {d} units", .{ s.plan.name, s.usage });
208 if (close) s.usage = 0;
209 }
210 fn totals(self: *Billing, s: *Subscription, items: []const Item) Totals {
211 var t: Totals = .{};
212 for (items) |line| t.subtotal += line.amount;
213 if (s.coupon) |c| {
214 if ((s.left == null or s.left.? > 0) and t.subtotal > 0) {
215 t.discount = if (c.percent) @divTrunc(t.subtotal * c.value, 10000) else @min(c.value, t.subtotal);
216 }
217 }
218 t.tax = @divTrunc((t.subtotal - t.discount) * (self.taxes.get(s.customer.region) orelse 0), 10000);
219 t.total = t.subtotal - t.discount + t.tax;
220 t.credit = @min(s.customer.balance, t.total);
221 t.due = t.total - t.credit;
222 return t;
223 }
224 fn issue(self: *Billing, s: *Subscription, kind: Kind, items: std.ArrayList(Item)) Error!void {
225 const t = self.totals(s, items.items);
226 if (s.coupon != null and t.subtotal > 0) {
227 if (s.left) |left| if (left > 0) {
228 s.left = left - 1;
229 };
230 }
231 const inv = try self.gpa.create(Invoice);
232 inv.* = .{ .id = self.invoices.items.len + 1, .sub = s, .day = self.day, .kind = kind, .items = items.items, .sums = t, .outstanding = t.due };
233 try self.invoices.append(self.gpa, inv);
234 try s.invoices.append(self.gpa, inv);
235 try self.balance(s.customer, -t.credit, "applied invoice {d}", .{inv.id});
236 var result: []const u8 = "PAID";
237 if (t.due > 0) {
238 if (s.state == .UNPAID) {
239 result = "OPEN";
240 } else {
241 inv.attempts = 1;
242 if (s.customer.method_ok) {
243 inv.paid = t.due;
244 inv.outstanding = 0;
245 } else result = "FAILED";
246 }
247 }
248 try self.out.print(" day {d} invoice {d} sub {d} {s} total {s} credit {s} due {s} {s}\n", .{ self.day, inv.id, s.id, @tagName(kind), try self.money(t.total), try self.money(t.credit), try self.money(t.due), result });
249 if (equal(result, "FAILED")) {
250 if (s.state == .ACTIVE) try self.transition(s, .PAST_DUE);
251 if (s.state == .PAST_DUE) {
252 inv.retries = .{ self.day + 3, self.day + 5, self.day + 7 };
253 inv.next_retry = 0;
254 }
255 }
256 }
257 fn cancel(self: *Billing, s: *Subscription) Error!void {
258 try self.transition(s, .CANCELED);
259 s.canceled_on = self.day;
260 s.pending = null;
261 s.cancel_end = false;
262 s.cancel_day = null;
263 var items: std.ArrayList(Item) = .empty;
264 try self.usageItem(&items, s, true);
265 if (items.items.len > 0) try self.issue(s, .final, items);
266 }
267 fn advance(self: *Billing, target: i64) Error!void {
268 while (self.day < target) {
269 self.day += 1;
270 for (self.subscriptions.items) |s| {
271 if (s.state == .PAUSED or s.state == .CANCELED) continue;
272 for (s.invoices.items) |inv| {
273 if (inv.next_retry == 3 or inv.retries[inv.next_retry] != self.day) continue;
274 inv.next_retry += 1;
275 inv.attempts += 1;
276 try self.out.print(" day {d} retry {d} invoice {d} {s}\n", .{ self.day, inv.attempts - 1, inv.id, if (s.customer.method_ok) "PAID" else "FAILED" });
277 if (s.customer.method_ok) {
278 inv.paid += inv.outstanding;
279 inv.outstanding = 0;
280 try self.completed(inv);
281 } else if (inv.next_retry == 3) {
282 try self.transition(s, .UNPAID);
283 s.cancel_day = self.day + 14;
284 }
285 }
286 if (s.state == .UNPAID and s.cancel_day == self.day) try self.cancel(s);
287 if (s.state == .TRIALING and s.trial_end == self.day) {
288 if (s.cancel_end) {
289 try self.cancel(s);
290 } else {
291 try self.transition(s, .ACTIVE);
292 s.start = self.day;
293 s.end = self.day + s.plan.cycle.days();
294 try self.issue(s, .first, try self.opening(s, self.day));
295 }
296 }
297 if ((s.state == .ACTIVE or s.state == .PAST_DUE or s.state == .UNPAID) and s.end == self.day) {
298 if (s.cancel_end) {
299 try self.cancel(s);
300 } else {
301 var items: std.ArrayList(Item) = .empty;
302 try self.usageItem(&items, s, true);
303 if (s.pending) |pending| {
304 try self.out.print(" day {d} sub {d} {s} seats {d} -> {s} seats {d}\n", .{ self.day, s.id, s.plan.name, s.seats, pending.plan.name, pending.seats });
305 s.plan = pending.plan;
306 s.seats = pending.seats;
307 s.pending = null;
308 }
309 s.start = self.day;
310 s.end = self.day + s.plan.cycle.days();
311 try self.charge(&items, s.plan, s.seats, self.day);
312 try self.issue(s, .renewal, items);
313 }
314 }
315 }
316 }
317 }
318 fn printItems(self: *Billing, items: []const Item, t: Totals) Error!void {
319 for (items) |line| try self.out.print(" {s} {s}\n", .{ line.description, try self.money(line.amount) });
320 try self.out.print(" subtotal {s} discount {s} tax {s} total {s}\n", .{ try self.money(t.subtotal), try self.money(t.discount), try self.money(t.tax), try self.money(t.total) });
321 }
322 fn preview(self: *Billing, s: *Subscription) Error!void {
323 try self.requireState(s, &.{ .TRIALING, .ACTIVE, .PAST_DUE, .UNPAID });
324 const day = if (s.state == .TRIALING) s.trial_end else s.end;
325 const kind: Kind = if (s.cancel_end) .final else if (s.state == .TRIALING) .first else .renewal;
326 var items: std.ArrayList(Item) = .empty;
327 if (kind == .first) {
328 items = try self.opening(s, day);
329 } else if (s.state != .TRIALING) {
330 try self.usageItem(&items, s, false);
331 if (kind == .renewal) {
332 const selection = s.pending orelse Selection{ .plan = s.plan, .seats = s.seats };
333 try self.charge(&items, selection.plan, selection.seats, day);
334 }
335 }
336 try self.out.print("PREVIEW sub {d} day {d} {s}\n", .{ s.id, day, @tagName(kind) });
337 if (items.items.len == 0) return self.out.writeAll(" (none)\n");
338 const t = self.totals(s, items.items);
339 try self.printItems(items.items, t);
340 try self.out.print(" credit {s} due {s}\n", .{ try self.money(t.credit), try self.money(t.due) });
341 }
342 fn printSub(self: *Billing, s: *Subscription) Error!void {
343 try self.out.print("SUB {d} {s} {s} seats {d} {s}\n", .{ s.id, s.customer.name, s.plan.name, s.seats, @tagName(s.state) });
344 if (s.state == .TRIALING) try self.out.print(" trial until {d}\n", .{s.trial_end});
345 if (s.state == .ACTIVE or s.state == .PAST_DUE or s.state == .UNPAID or s.state == .PAUSED) try self.out.print(" period {d}-{d}\n", .{ s.start, s.end });
346 if (s.state == .PAUSED) try self.out.print(" paused since {d}\n", .{s.pause_day});
347 if (s.pending) |p| try self.out.print(" pending {s} seats {d}\n", .{ p.plan.name, p.seats });
348 if (s.cancel_end) try self.out.writeAll(" cancel at end\n");
349 if (s.cancel_day) |day| try self.out.print(" cancel on {d}\n", .{day});
350 if (s.state == .CANCELED) try self.out.print(" canceled on {d}\n", .{s.canceled_on});
351 if (s.coupon) |c| {
352 if (s.left) |left| {
353 if (left > 0) try self.out.print(" coupon {s} left {d}\n", .{ c.name, left });
354 } else try self.out.print(" coupon {s} forever\n", .{c.name});
355 }
356 if (s.plan.mode != null and s.state != .CANCELED) try self.out.print(" usage {d}\n", .{s.usage});
357 var found = false;
358 for (s.invoices.items) |inv| {
359 if (inv.outstanding == 0) continue;
360 if (!found) try self.out.writeAll(" open");
361 found = true;
362 try self.out.print(" {d}", .{inv.id});
363 }
364 if (found) try self.out.writeAll("\n");
365 }
366 fn agingLine(self: *Billing, name: []const u8, buckets: [3]i64) Error!void {

368 }
369 fn execute(self: *Billing, a: []const []const u8) Error!void {
370 const op = std.meta.stringToEnum(Command, a[0]).?;
371 const id: []const u8 = if (a.len > 1) a[1] else "";
372 switch (op) {
373 .PLAN => {
374 if (self.plans.contains(id)) return self.fail("duplicate plan {s}", .{id});
375 const p = try self.gpa.create(Plan);
376 p.* = .{ .name = id, .price = amount(a[2]), .cycle = std.meta.stringToEnum(Cycle, a[3]).? };
377 var i: usize = 4;
378 while (i < a.len) : (i += 1) {
379 if (equal(a[i], "TRIAL")) {
380 i += 1;
381 p.trial = integer(a[i]);
382 } else if (equal(a[i], "SETUP")) {
383 i += 1;
384 p.setup = amount(a[i]);
385 } else if (equal(a[i], "PERSEAT")) {
386 p.per_seat = true;
387 } else {
388 p.mode = std.meta.stringToEnum(Mode, a[i]).?;
389 }
390 }
391 try self.plans.put(self.gpa, id, p);
392 try self.out.print("PLAN {s} {s} {s}", .{ id, a[3], try self.money(p.price) });
393 if (p.trial > 0) try self.out.print(" trial {d}", .{p.trial});
394 if (p.setup > 0) try self.out.print(" setup {s}", .{try self.money(p.setup)});
395 if (p.per_seat) try self.out.writeAll(" perseat");
396 if (p.mode) |mode| try self.out.writeAll(if (mode == .GRADUATED) " graduated" else " volume");
397 try self.out.writeAll("\n");
398 },
399 .TIER => {
400 const p = try self.plan(id);
401 if (p.mode == null) return self.fail("not metered {s}", .{id});
402 const limit: ?i64 = if (equal(a[2], "INF")) null else integer(a[2]);
403 if (p.tiers.items.len > 0) {
404 const last = p.tiers.items[p.tiers.items.len - 1].limit;
405 if (last == null or (limit != null and limit.? <= last.?)) return self.fail("tier order {s}", .{id});
406 }
407 try p.tiers.append(self.gpa, .{ .limit = limit, .unit = amount(a[3]), .flat = amount(a[4]) });
408 try self.out.print("TIER {s} {d} upto {s} unit {s} flat {s}\n", .{ id, p.tiers.items.len, a[2], a[3], a[4] });
409 },
410 .TAX => {
411 const bp = integer(a[2]);
412 try self.taxes.put(self.gpa, id, bp);
413 try self.out.print("TAX {s} {s}\n", .{ id, try self.rate(bp) });
414 },
415 .CUSTOMER => {
416 if (self.customers.contains(id)) return self.fail("duplicate customer {s}", .{id});
417 const c = try self.gpa.create(Customer);
418 c.* = .{ .name = id, .region = a[2] };
419 try self.customers.put(self.gpa, id, c);
420 try self.out.print("CUSTOMER {s} region {s}\n", .{ id, c.region });
421 },
422 .COUPON => {
423 if (self.coupons.contains(id)) return self.fail("duplicate coupon {s}", .{id});
424 const c = try self.gpa.create(Coupon);
425 const percent = equal(a[2], "PERCENT");
426 c.* = .{ .name = id, .percent = percent, .value = if (percent) integer(a[3]) else amount(a[3]), .applications = if (equal(a[4], "FOREVER")) null else if (equal(a[4], "ONCE")) 1 else integer(a[5]) };
427 const i: usize = if (equal(a[4], "REPEATING")) 6 else 5;
428 if (i < a.len) c.maximum = integer(a[i + 1]);
429 try self.coupons.put(self.gpa, id, c);
430 try self.out.print("COUPON {s} {s} {s} ", .{ id, if (percent) "percent" else "amount", if (percent) try self.rate(c.value) else try self.money(c.value) });
431 if (equal(a[4], "REPEATING")) try self.out.print("repeating {d}", .{c.applications.?}) else try self.out.writeAll(if (c.applications == null) "forever" else "once");
432 if (c.maximum) |maximum| try self.out.print(" max {d}", .{maximum});
433 try self.out.writeAll("\n");
434 },
435 .PAYMETHOD, .CREDIT, .STATEMENT, .LEDGER => {
436 const c = try self.customer(id);
437 switch (op) {
438 .PAYMETHOD => {
439 c.method_ok = equal(a[2], "OK");
440 try self.out.print("PAYMETHOD {s} {s}\n", .{ id, a[2] });
441 },
442 .CREDIT => {
443 const value = amount(a[2]);
444 if (value == 0) return self.fail("invalid amount", .{});
445 try self.balance(c, value, "grant", .{});
446 try self.out.print("CREDIT {s} amount {s} balance {s}\n", .{ id, a[2], try self.money(c.balance) });
447 },
448 .STATEMENT => {
449 try self.out.print("STATEMENT {s} region {s} tax {s} method {s} balance {s}\n", .{ id, c.region, try self.rate(self.taxes.get(c.region) orelse 0), if (c.method_ok) "OK" else "FAIL", try self.money(c.balance) });
450 for (self.subscriptions.items) |s| if (s.customer == c) {
451 try self.out.print(" sub {d} {s} {s}\n", .{ s.id, s.plan.name, @tagName(s.state) });
452 };
453 var outstanding: i64 = 0;
454 for (self.invoices.items) |inv| {
455 if (inv.sub.customer != c) continue;
456 outstanding += inv.outstanding;
457 try self.out.print(" invoice {d} day {d} sub {d} {s} total {s} outstanding {s}\n", .{ inv.id, inv.day, inv.sub.id, @tagName(inv.kind), try self.money(inv.sums.total), try self.money(inv.outstanding) });
458 }
459 try self.out.print(" outstanding {s}\n", .{try self.money(outstanding)});
460 },
461 .LEDGER => {
462 try self.out.print("LEDGER {s} balance {s}\n", .{ id, try self.money(c.balance) });
463 for (c.log.items) |entry| try self.out.print(" day {d} {s} {s}{s} balance {s}\n", .{ entry.day, entry.description, if (entry.change > 0) "+" else "", try self.money(entry.change), try self.money(entry.balance) });
464 if (c.log.items.len == 0) try self.out.writeAll(" (none)\n");
465 },
466 else => unreachable,
467 }
468 },
469 .DAY => {
470 const day = integer(id);
471 if (day <= self.day) return self.fail("invalid day {s}", .{id});
472 try self.out.print("DAY {d}\n", .{day});
473 try self.advance(day);
474 },
475 .SUBSCRIBE => {
476 const c = try self.customer(id);
477 const p = try self.plan(a[2]);
478 var i: usize = 3;
479 var seats: i64 = 1;
480 if (i < a.len and equal(a[i], "SEATS")) {
481 if (!p.per_seat) return self.fail("not per-seat {s}", .{p.name});
482 seats = integer(a[i + 1]);
483 i += 2;
484 }
485 if (seats == 0) return self.fail("invalid seats", .{});
486 const cp: ?*Coupon = if (i < a.len) try self.coupon(a[i + 1]) else null;
487 const s = try self.gpa.create(Subscription);
488 s.* = .{ .id = self.subscriptions.items.len + 1, .customer = c, .plan = p, .seats = seats, .coupon = cp };
489 if (cp) |coupon_value| {
490 coupon_value.redemptions += 1;
491 s.left = coupon_value.applications;
492 }
493 try self.subscriptions.append(self.gpa, s);
494 try self.out.print("SUBSCRIBE sub {d} {s} {s} seats {d}", .{ s.id, id, p.name, seats });
495 if (cp) |coupon_value| try self.out.print(" coupon {s}", .{coupon_value.name});
496 try self.out.writeAll("\n");
497 if (p.trial > 0) {
498 s.trial_end = self.day + p.trial;
499 try self.transition(s, .TRIALING);
500 } else {
501 try self.transition(s, .ACTIVE);
502 s.start = self.day;
503 s.end = self.day + p.cycle.days();
504 try self.issue(s, .first, try self.opening(s, self.day));
505 }
506 },
507 .PAY, .REFUND, .INVOICE => {
508 const inv = try self.invoice(id);
509 if (op == .PAY) {
510 if (inv.outstanding == 0) return self.fail("already paid {s}", .{id});
511 try self.out.print("PAY invoice {d} amount {s}\n", .{ inv.id, try self.money(inv.outstanding) });
512 inv.paid += inv.outstanding;
513 inv.outstanding = 0;
514 try self.completed(inv);
515 } else if (op == .REFUND) {
516 const value = amount(a[2]);
517 if (value == 0) return self.fail("invalid amount", .{});
518 if (value > inv.sums.total - inv.refunded) return self.fail("refund exceeds {s} {s}", .{ id, try self.money(inv.sums.total - inv.refunded) });
519 const applied = @min(value, inv.outstanding);
520 inv.refunded += value;
521 inv.outstanding -= applied;
522 try self.balance(inv.sub.customer, value - applied, "refund invoice {d}", .{inv.id});
523 try self.out.print("REFUND invoice {d} amount {s} applied {s} balance {s}\n", .{ inv.id, a[2], try self.money(applied), try self.money(inv.sub.customer.balance) });
524 if (applied > 0 and inv.outstanding == 0) try self.completed(inv);
525 } else {
526 try self.out.print("INVOICE {d} sub {d} {s} day {d} {s} {s}\n", .{ inv.id, inv.sub.id, inv.sub.customer.name, inv.day, @tagName(inv.kind), if (inv.outstanding == 0) "PAID" else "OPEN" });
527 try self.printItems(inv.items, inv.sums);
528 try self.out.print(" credit {s} due {s} paid {s} refunded {s} outstanding {s}\n", .{ try self.money(inv.sums.credit), try self.money(inv.sums.due), try self.money(inv.paid), try self.money(inv.refunded), try self.money(inv.outstanding) });
529 }
530 },
531 .CHANGE, .PAUSE, .RESUME, .CANCEL, .USAGE, .APPLY, .SUB, .PREVIEW => {
532 const s = try self.subscription(id);
533 switch (op) {
534 .CHANGE => {
535 try self.requireState(s, &.{ .ACTIVE, .TRIALING });
536 var p = s.plan;
537 var seats = s.seats;
538 if (equal(a[2], "PLAN")) {
539 p = try self.plan(a[3]);
540 if (!p.per_seat) seats = 1;
541 } else {
542 if (!p.per_seat) return self.fail("not per-seat {s}", .{p.name});
543 seats = integer(a[3]);
544 if (seats == 0) return self.fail("invalid seats", .{});
545 }
546 if (p == s.plan and seats == s.seats) return self.fail("no change {s}", .{id});
547 const old_fee = s.plan.fee(s.seats);
548 const new_fee = p.fee(seats);
549 const upgrade = new_fee * s.plan.cycle.days() > old_fee * p.cycle.days();
550 s.pending = null;
551 try self.out.print("CHANGE sub {d} {s} seats {d} -> {s} seats {d}", .{ s.id, s.plan.name, s.seats, p.name, seats });
552 if (s.state == .TRIALING) {
553 s.plan = p;
554 s.seats = seats;
555 try self.out.writeAll(" now\n");
556 } else if (upgrade) {
557 try self.out.writeAll(" now\n");
558 const remaining = s.end - self.day;
559 var items: std.ArrayList(Item) = .empty;
560 try self.usageItem(&items, s, true);
561 try self.item(&items, -@divTrunc(old_fee * remaining, s.plan.cycle.days()), "unused {s} seats {d} {d} days", .{ s.plan.name, s.seats, remaining });
562 try self.item(&items, @divTrunc(new_fee * remaining, p.cycle.days()), "remaining {s} seats {d} {d} days", .{ p.name, seats, remaining });
563 s.plan = p;
564 s.seats = seats;
565 try self.issue(s, .upgrade, items);
566 } else {
567 s.pending = .{ .plan = p, .seats = seats };
568 try self.out.print(" at {d}\n", .{s.end});
569 }
570 },
571 .PAUSE => {
572 try self.requireState(s, &.{.ACTIVE});
573 try self.out.print("PAUSE sub {d}\n", .{s.id});
574 s.pause_day = self.day;
575 try self.transition(s, .PAUSED);
576 },
577 .RESUME => {
578 try self.requireState(s, &.{.PAUSED});
579 const shift = self.day - s.pause_day;
580 s.end += shift;
581 s.pause_day = 0;
582 try self.out.print("RESUME sub {d} shift {d} period {d}-{d}\n", .{ s.id, shift, s.start, s.end });
583 try self.transition(s, .ACTIVE);
584 },
585 .CANCEL => {
586 if (equal(a[2], "END")) {
587 try self.requireState(s, &.{ .ACTIVE, .TRIALING });
588 if (s.cancel_end) return self.fail("already canceling {s}", .{id});
589 s.cancel_end = true;
590 try self.out.print("CANCEL sub {d} end {d}\n", .{ s.id, if (s.state == .TRIALING) s.trial_end else s.end });
591 } else {
592 try self.requireState(s, &.{ .TRIALING, .ACTIVE, .PAST_DUE, .UNPAID, .PAUSED });
593 const remaining = if (s.state == .ACTIVE) s.end - self.day else if (s.state == .PAUSED) s.end - s.pause_day else 0;
594 const credit = @divTrunc(s.plan.fee(s.seats) * remaining, s.plan.cycle.days());
595 try self.balance(s.customer, credit, "cancel sub {d}", .{s.id});
596 try self.out.print("CANCEL sub {d} now credit {s}\n", .{ s.id, try self.money(credit) });
597 try self.cancel(s);
598 }
599 },
600 .USAGE => {
601 try self.requireState(s, &.{ .ACTIVE, .PAST_DUE });
602 if (s.plan.mode == null) return self.fail("not metered {s}", .{s.plan.name});
603 const units = integer(a[2]);
604 if (units == 0) return self.fail("invalid units", .{});
605 s.usage += units;
606 try self.out.print("USAGE sub {d} units {d} total {d}\n", .{ s.id, units, s.usage });
607 },
608 .APPLY => {
609 try self.requireState(s, &.{ .TRIALING, .ACTIVE, .PAST_DUE, .UNPAID, .PAUSED });
610 const cp = try self.coupon(a[2]);
611 cp.redemptions += 1;
612 s.coupon = cp;
613 s.left = cp.applications;
614 try self.out.print("APPLY sub {d} coupon {s}\n", .{ s.id, cp.name });
615 },
616 .SUB => try self.printSub(s),
617 .PREVIEW => try self.preview(s),
618 else => unreachable,
619 }
620 },
621 .MRR => {
622 try self.out.writeAll("MRR\n");
623 var total: i64 = 0;
624 var count: usize = 0;
625 for (try self.ordered(Plan, self.plans.values())) |p| {
626 var plan_total: i64 = 0;
627 var plan_count: usize = 0;
628 for (self.subscriptions.items) |s| {
629 if (s.plan != p or (s.state != .ACTIVE and s.state != .PAST_DUE)) continue;
630 const fee = p.fee(s.seats);
631 plan_total += switch (p.cycle) {
632 .MONTHLY => fee,
633 .YEARLY => @divTrunc(fee, 12),
634 .WEEKLY => @divTrunc(fee * 30, 7),
635 };
636 plan_count += 1;
637 }
638 if (plan_count > 0) try self.out.print(" {s} subs {d} mrr {s}\n", .{ p.name, plan_count, try self.money(plan_total) });
639 total += plan_total;
640 count += plan_count;
641 }
642 try self.out.print(" total subs {d} mrr {s}\n", .{ count, try self.money(total) });
643 },
644 .AGING => {
645 try self.out.print("AGING day {d}\n", .{self.day});
646 var totals_by_age: [3]i64 = @splat(0);
647 for (try self.ordered(Customer, self.customers.values())) |c| {
648 var buckets: [3]i64 = @splat(0);
649 for (self.invoices.items) |inv| {
650 if (inv.sub.customer != c or inv.outstanding == 0) continue;
651 const age = self.day - inv.day;
652 const bucket: usize = if (age <= 30) 0 else if (age <= 60) 1 else 2;
653 buckets[bucket] += inv.outstanding;
654 }
655 if (buckets[0] + buckets[1] + buckets[2] > 0) try self.agingLine(c.name, buckets);
656 for (0..3) |i| totals_by_age[i] += buckets[i];
657 }
658 try self.agingLine("total", totals_by_age);
659 },
660 .DUNNING => {
661 try self.out.writeAll("DUNNING\n");
662 var found = false;
663 for (self.invoices.items) |inv| {
664 if (inv.next_retry == 3) continue;
665 found = true;
666 try self.out.print(" invoice {d} sub {d} {s} outstanding {s} attempts {d} next {d}\n", .{ inv.id, inv.sub.id, inv.sub.customer.name, try self.money(inv.outstanding), inv.attempts, inv.retries[inv.next_retry] });
667 }
668 if (!found) try self.out.writeAll(" (none)\n");
669 },
670 .REVENUE => {
671 var sums: Totals = .{};
672 var paid: i64 = 0;
673 var refunded: i64 = 0;
674 var outstanding: i64 = 0;
675 var balances: i64 = 0;
676 for (self.invoices.items) |inv| {
677 sums.subtotal += inv.sums.subtotal;
678 sums.discount += inv.sums.discount;
679 sums.tax += inv.sums.tax;
680 sums.total += inv.sums.total;
681 sums.credit += inv.sums.credit;
682 paid += inv.paid;
683 refunded += inv.refunded;
684 outstanding += inv.outstanding;
685 }
686 for (self.customers.values()) |c| balances += c.balance;
687 try self.out.print("REVENUE\n invoices {d} subtotal {s} discount {s} tax {s} total {s}\n credit {s} paid {s} refunded {s} outstanding {s} balances {s}\n", .{ self.invoices.items.len, try self.money(sums.subtotal), try self.money(sums.discount), try self.money(sums.tax), try self.money(sums.total), try self.money(sums.credit), try self.money(paid), try self.money(refunded), try self.money(outstanding), try self.money(balances) });
688 },
689 .EVENTS => {
690 const s: ?*Subscription = if (a.len == 2) try self.subscription(id) else null;
691 if (s) |sub| try self.out.print("EVENTS sub {d}\n", .{sub.id}) else try self.out.writeAll("EVENTS\n");
692 var found = false;
693 for (self.events.items) |event| {
694 if (s != null and event.sub_id != s.?.id) continue;
695 found = true;
696 try self.out.print(" day {d} sub {d} {s} -> {s}\n", .{ event.day, event.sub_id, @tagName(event.from), @tagName(event.to) });
697 }
698 if (!found) try self.out.writeAll(" (none)\n");
699 },
700 }
701 }
702};
703pub fn main(init: std.process.Init) !void {
704 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
705 defer arena.deinit();
706 const gpa = arena.allocator();
707 var input_buffer: [4096]u8 = undefined;
708 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
709 var output_buffer: [8192]u8 = undefined;
710 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
711 var billing = Billing{ .gpa = gpa, .out = &output.interface };
712 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
713 for (0..n) |_| {
714 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
715 var tokens: std.ArrayList([]const u8) = .empty;
716 var it = std.mem.splitScalar(u8, line, ' ');
717 while (it.next()) |token| try tokens.append(gpa, token);
718 billing.execute(tokens.items) catch |err| {
719 if (err != error.BadCommand) return err;
720 try output.interface.print("ERROR {s}\n", .{billing.error_message});
721 };
722 }
723 try output.interface.flush();
724}
