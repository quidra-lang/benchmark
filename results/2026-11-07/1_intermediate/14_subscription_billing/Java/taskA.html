36 this.name = name;
37 this.price = price;

46 long fee(int seats) { return perSeat ? price * seats : price; }
47
48 long usageCharge(long units) {
49 long total = 0;
50 long lower = 0;
51 for (int i = 0; i < tiers.size(); i++) {
52 Tier tier = tiers.get(i);
53 boolean last = i == tiers.size() - 1;
54 if ("VOLUME".equals(mode)) {
55 if (last || tier.upper() == null || units <= tier.upper())
56 return units * tier.unit() + tier.flat();
57 } else {
58 if (units <= lower) break;
59 long end = last || tier.upper() == null
60 ? units : Math.min(units, tier.upper());
61 total += (end - lower) * tier.unit() + tier.flat();
62 if (last || tier.upper() == null) break;
63 lower = tier.upper();
64 }
65 }
66 return total;
67 }

298 static boolean couponUsable(Subscription sub, long subtotal) {
299 return subtotal > 0 && sub.coupon != null &&
300 (sub.remaining == null || sub.remaining > 0);
301 }
302
303 Totals invoiceTotals(Subscription sub, List<Line> items) {
304 long subtotal = 0;
305 for (Line item : items) subtotal += item.amount();
306 long discount = 0;
307 if (couponUsable(sub, subtotal)) {
308 Coupon coupon = sub.coupon;
309 discount = coupon.kind.equals("PERCENT")
310 ? subtotal * coupon.value / 10000
311 : Math.min(subtotal, coupon.value);
312 }
313 long tax = (subtotal - discount) * taxRate(sub.customer.region) / 10000;
314 long total = subtotal - discount + tax;
315 long credit = Math.min(sub.customer.balance, total);
316 return new Totals(subtotal, discount, tax, total, credit, total - credit);
317 }

319 void issue(Subscription sub, String kind, List<Line> items) {
320 Totals totals = invoiceTotals(sub, items);
321 Invoice inv = new Invoice(invoices.size() + 1, sub, day, kind, items, totals);
322 if (couponUsable(sub, inv.subtotal) && sub.remaining != null)
323 --sub.remaining;
324 invoices.add(inv);
325 sub.invoices.add(inv);
326 sub.customer.invoices.add(inv);
327 balanceChange(sub.customer, -inv.credit, "applied invoice " + inv.number);
328
329 String result;
330 if (inv.due == 0) {
331 result = "PAID";
332 } else if (sub.state.equals("UNPAID")) {
333 result = "OPEN";
334 } else {
335 inv.attempts = 1;
336 if (sub.customer.method.equals("OK")) {
337 inv.paid = inv.due;
338 inv.outstanding = 0;
339 result = "PAID";
340 } else {
341 result = "FAILED";
342 }
343 }
344 notice("invoice", inv.number, "sub", sub.number, kind,
345 "total", money(inv.total), "credit", money(inv.credit),
346 "due", money(inv.due), result);
347 if (result.equals("FAILED")) {
348 if (sub.state.equals("ACTIVE")) changeState(sub, "PAST_DUE");
349 if (sub.state.equals("PAST_DUE"))
350 Collections.addAll(inv.retries, day + 3, day + 5, day + 7);
351 }
352 }

378 void retry(Invoice inv) {
379 inv.retries.remove(0);
380 ++inv.attempts;
381 int retryNumber = inv.attempts - 1;
382 if (inv.customer.method.equals("OK")) {
383 inv.paid += inv.outstanding;
384 inv.outstanding = 0;
385 notice("retry", retryNumber, "invoice", inv.number, "PAID");
386 paymentComplete(inv);
387 } else {
388 notice("retry", retryNumber, "invoice", inv.number, "FAILED");
389 if (inv.retries.isEmpty()) {
390 changeState(inv.subscription, "UNPAID");
391 inv.subscription.cancelOn = day + 14;
392 }
393 }

396 void advanceSubscription(Subscription sub) {
397 if (sub.state.equals("PAUSED") || sub.state.equals("CANCELED")) return;
398 for (Invoice inv : sub.invoices)
399 if (!inv.retries.isEmpty() && inv.retries.get(0) == day) retry(inv);
400
401 if (sub.state.equals("UNPAID") && sub.cancelOn != null && sub.cancelOn == day)
402 cancelSubscription(sub);
403
404 if (sub.state.equals("TRIALING") && sub.trialEnd != null && sub.trialEnd == day) {
405 if (sub.cancelEnd) {
406 cancelSubscription(sub);
407 } else {
408 changeState(sub, "ACTIVE");
409 sub.start = day;
410 sub.end = day + sub.plan.length;
411 issue(sub, "first", openingItems(sub, day));
412 }
413 }
414
415 if ((sub.state.equals("ACTIVE") || sub.state.equals("PAST_DUE") ||
416 sub.state.equals("UNPAID")) && sub.end != null && sub.end == day) {
417 if (sub.cancelEnd) {
418 cancelSubscription(sub);
419 } else {
420 List<Line> items = usageItems(sub, true);
421 if (sub.pending != null) {
422 Plan next = sub.pending.plan();
423 int seats = sub.pending.seats();
424 notice("sub", sub.number, sub.plan.name, "seats", sub.seats,
425 "->", next.name, "seats", seats);
426 sub.plan = next;
427 sub.seats = seats;
428 sub.pending = null;
429 }
430 sub.start = day;
431 sub.end = day + sub.plan.length;
432 items.add(periodItem(sub.plan, sub.seats, day));
433 issue(sub, "renewal", items);
434 }
435 }

438 void cmdPlan(String[] a) {
439 String name = a[0];
440 if (plans.containsKey(name)) fail("duplicate plan " + name);
441 Plan plan = new Plan(name, cents(a[1]), a[2]);
442 for (int i = 3; i < a.length;) {
443 switch (a[i]) {
444 case "TRIAL" -> {
445 plan.trial = Integer.parseInt(a[i + 1]);
446 i += 2;
447 }
448 case "SETUP" -> {
449 plan.setup = cents(a[i + 1]);
450 i += 2;
451 }
452 case "PERSEAT" -> {
453 plan.perSeat = true;
454 ++i;
455 }
456 default -> {
457 plan.mode = a[i];
458 ++i;
459 }
460 }
461 }
462 plans.put(name, plan);
463 List<Object> parts = new ArrayList<>(List.of("PLAN", name, a[2], a[1]));
464 if (plan.trial != 0) Collections.addAll(parts, "trial", plan.trial);
465 if (plan.setup != 0) Collections.addAll(parts, "setup", money(plan.setup));
466 if (plan.perSeat) parts.add("perseat");
467 if (plan.mode != null) parts.add(plan.mode.toLowerCase(java.util.Locale.ROOT));
468 emit(parts.toArray());
469 }

481 emit("TIER", plan.name, plan.tiers.size(), "upto", a[1],
482 "unit", a[2], "flat", a[3]);
483 }

497 void cmdCoupon(String[] a) {
498 String name = a[0];
499 if (coupons.containsKey(name)) fail("duplicate coupon " + name);
500 String application = a[3];
501 Integer applications = application.equals("ONCE") ? 1 : null;
502 int index = 4;
503 if (application.equals("REPEATING")) {
504 applications = Integer.parseInt(a[index]);
505 ++index;
506 }
507 Integer maximum = index < a.length ? Integer.parseInt(a[index + 1]) : null;
508 long value = a[1].equals("PERCENT") ? Long.parseLong(a[2]) : cents(a[2]);
509 Coupon coupon = new Coupon(name, a[1], value, applications, maximum);
510 coupons.put(name, coupon);
511 List<Object> parts = new ArrayList<>();
512 Collections.addAll(parts, "COUPON", name,
513 coupon.kind.toLowerCase(java.util.Locale.ROOT),
514 coupon.kind.equals("PERCENT") ? percent(value) : money(value),
515 application.toLowerCase(java.util.Locale.ROOT));
516 if (application.equals("REPEATING")) parts.add(applications);
517 if (maximum != null) Collections.addAll(parts, "max", maximum);
518 emit(parts.toArray());

545 void cmdSubscribe(String[] a) {
546 Customer customer = customer(a[0]);
547 Plan plan = plan(a[1]);
548 int seats = 1;
549 boolean hasSeats = false;
550 String couponName = null;
551 for (int i = 2; i < a.length; i += 2) {
552 if (a[i].equals("SEATS")) {
553 seats = Integer.parseInt(a[i + 1]);
554 hasSeats = true;
555 } else {
556 couponName = a[i + 1];
557 }
558 }
559 if (hasSeats && !plan.perSeat) fail("not per-seat " + plan.name);
560 if (seats == 0) fail("invalid seats");
561 Coupon coupon = couponName == null ? null : redeemableCoupon(couponName);
562 Subscription sub = new Subscription(subscriptions.size() + 1,
563 customer, plan, seats, coupon);
564 if (coupon != null) ++coupon.redemptions;
565 subscriptions.add(sub);
566 customer.subscriptions.add(sub);
567 List<Object> parts = new ArrayList<>();
568 Collections.addAll(parts, "SUBSCRIBE", "sub", sub.number, customer.name,
569 plan.name, "seats", seats);
570 if (coupon != null) Collections.addAll(parts, "coupon", coupon.name);
571 emit(parts.toArray());
572 if (plan.trial > 0) {
573 sub.trialEnd = day + plan.trial;
574 changeState(sub, "TRIALING");
575 } else {
576 changeState(sub, "ACTIVE");
577 sub.start = day;
578 sub.end = day + plan.length;
579 issue(sub, "first", openingItems(sub, day));
580 }

583 void cmdChange(String[] a) {
584 Subscription sub = subscription(a[0], "ACTIVE", "TRIALING");
585 Plan plan;
586 int seats;
587 if (a[1].equals("PLAN")) {
588 plan = plan(a[2]);
589 seats = plan.perSeat ? sub.seats : 1;
590 } else {
591 plan = sub.plan;
592 if (!plan.perSeat) fail("not per-seat " + plan.name);
593 seats = Integer.parseInt(a[2]);
594 if (seats == 0) fail("invalid seats");
595 }
596 if (plan == sub.plan && seats == sub.seats) fail("no change " + a[0]);
597 Plan oldPlan = sub.plan;
598 int oldSeats = sub.seats;
599 List<Object> head = new ArrayList<>();
600 Collections.addAll(head, "CHANGE", "sub", sub.number, oldPlan.name, "seats",
601 oldSeats, "->", plan.name, "seats", seats);
602 sub.pending = null;
603 boolean upgrade = plan.fee(seats) * oldPlan.length >
604 oldPlan.fee(oldSeats) * plan.length;
605 if (sub.state.equals("TRIALING")) {
606 sub.plan = plan;
607 sub.seats = seats;
608 head.add("now");
609 emit(head.toArray());
610 } else if (upgrade) {
611 head.add("now");
612 emit(head.toArray());
613 int days = sub.end - day;
614 List<Line> items = usageItems(sub, true);
615 items.add(new Line("unused " + oldPlan.name + " seats " + oldSeats +
616 " " + days + " days",
617 -(oldPlan.fee(oldSeats) * days / oldPlan.length)));
618 items.add(new Line("remaining " + plan.name + " seats " + seats +
619 " " + days + " days",
620 plan.fee(seats) * days / plan.length));
621 sub.plan = plan;
622 sub.seats = seats;
623 issue(sub, "upgrade", items);
624 } else {
625 sub.pending = new Pending(plan, seats);
626 Collections.addAll(head, "at", sub.end);
627 emit(head.toArray());
628 }
629 }

662 refund = sub.fee() * (sub.end - start) / sub.plan.length;
663 }

665 emit("CANCEL", "sub", sub.number, "now", "credit", money(refund));
666 cancelSubscription(sub);
