69 static String name(String word, boolean wildcard, String origin) {
70 if (origin != null && word.equals("@")) return origin;
71 if (word.equals(".")) return ".";
72 String text = word.endsWith(".") ? word.substring(0, word.length() - 1) : word;
73 String[] labels = text.split("\\.", -1);
74 for (int i = 0; i < labels.length; i++) require(wildcard && i == 0 && labels[i].equals("*") || labels[i].matches("[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?"), "bad name " + word);
75 String result = text.toLowerCase(Locale.ROOT) + ".";
76 if (origin != null && !word.endsWith(".") && !origin.equals(".")) result += origin;
77 return result;
78 }

97 static int compareNames(String left, String right) {
98 String[] a = labels(left), b = labels(right);
99 for (int i = 1; i <= Math.min(a.length, b.length); i++) {
100 int c = a[a.length - i].compareTo(b[b.length - i]);
101 if (c != 0) return c;
102 }
103 return Integer.compare(a.length, b.length);
104 }

106 static long ipv4(String word) {
107 String[] fields = word.split("\\.", -1);
108 require(fields.length == 4, "bad address " + word);
109 long value = 0;
110 for (String field : fields) {
111 require(field.matches("0|[1-9][0-9]{0,2}"), "bad address " + word);
112 int n = Integer.parseInt(field);
113 require(n <= 255, "bad address " + word);
114 value = value * 256 + n;
115 }
116 return value;
117 }

119 static String address(long value) {
120 return (value >>> 24) + "." + ((value >>> 16) & 255) + "." + ((value >>> 8) & 255) + "." + (value & 255);
121 }

123 static int[] ipv6(String word) {
124 int doubleColon = word.indexOf("::");
125 String left = doubleColon < 0 ? word : word.substring(0, doubleColon), right = doubleColon < 0 ? "" : word.substring(doubleColon + 2);
126 require(doubleColon < 0 || doubleColon == word.lastIndexOf("::"), "bad address " + word);
127 String[] a = left.isEmpty() ? new String[0] : left.split(":", -1), b = right.isEmpty() ? new String[0] : right.split(":", -1);
128 require(doubleColon < 0 ? a.length == 8 : a.length + b.length <= 7, "bad address " + word);
129 int[] groups = new int[8];
130 for (int i = 0; i < a.length; i++) {
131 require(a[i].matches("[0-9a-fA-F]{1,4}"), "bad address " + word);
132 groups[i] = Integer.parseInt(a[i], 16);
133 }
134 for (int i = 0; i < b.length; i++) {
135 require(b[i].matches("[0-9a-fA-F]{1,4}"), "bad address " + word);
136 groups[8 - b.length + i] = Integer.parseInt(b[i], 16);
137 }
138 return groups;
139 }

141 static String address6(int[] groups) {
142 int best = -1, count = 1;
143 for (int i = 0; i < 8;) {
144 if (groups[i] != 0) {
145 i++;
146 continue;
147 }
148 int end = i;
149 while (end < 8 && groups[end] == 0) end++;
150 if (end - i > count) {
151 best = i;
152 count = end - i;
153 }
154 i = end;
155 }
156 List<String> left = new ArrayList<>(), right = new ArrayList<>();
157 if (best < 0) {
158 for (int group : groups) left.add(Integer.toHexString(group));
159 return String.join(":", left);
160 }
161 for (int i = 0; i < best; i++) left.add(Integer.toHexString(groups[i]));
162 for (int i = best + count; i < 8; i++) right.add(Integer.toHexString(groups[i]));
163 return String.join(":", left) + "::" + String.join(":", right);
164 }

170 static int ttl(String word) {
171 Integer plain = integer(word);
172 long total = plain == null ? 0 : plain;
173 if (plain == null) {
174 int at = 0, previous = -1;
175 String units = "wdhms";
176 int[] seconds = {604800, 86400, 3600, 60, 1};
177 while (at < word.length()) {
178 int start = at;
179 while (at < word.length() && word.charAt(at) >= '0' && word.charAt(at) <= '9') at++;
180 String number = word.substring(start, at);
181 require(number.length() >= 1 && number.length() <= 3 && integer(number) != null && at < word.length(), "bad ttl " + word);
182 int unit = units.indexOf(Character.toLowerCase(word.charAt(at++)));
183 require(unit > previous, "bad ttl " + word);
184 previous = unit;
185 total += (long)Integer.parseInt(number) * seconds[unit];
186 }
187 }
188 require(total >= 1 && total <= 604800, "bad ttl " + word);
189 return (int)total;
190 }

192 static int compareData(Record a, Record b) {
193 return switch (a.type()) {
194 case 0 -> Long.compare(ipv4(a.data()), ipv4(b.data()));
195 case 7 -> Arrays.compare(ipv6(a.data()), ipv6(b.data()));
196 case 1, 2, 4 -> compareNames(a.data(), b.data());
197 case 5 -> {
198 String[] x = a.data().split(" "), y = b.data().split(" ");
199 int order = Integer.compare(Integer.parseInt(x[0]), Integer.parseInt(y[0]));
200 yield order == 0 ? compareNames(x[1], y[1]) : order;
201 }
202 case 6 -> a.data().compareTo(b.data());
203 default -> 0;
204 };
205 }

215 static void zoneLine(String[] words, int line) {
216 if (words[0].equals("ROOT")) {
217 List<Long> additions = new ArrayList<>();
218 for (String word : Arrays.copyOfRange(words, 1, words.length)) {
219 long value = ipv4(word);
220 require(!roots.contains(value) && !additions.contains(value), "duplicate root " + word);
221 additions.add(value);
222 }
223 roots.addAll(additions);
224 return;
225 }
226 if (words[0].equals("ZONE")) {
227 current = null;
228 String origin = name(words[1], false, null);
229 long server = ipv4(words[2]);
230 require(zones.stream().noneMatch(z -> z.origin.equals(origin) && z.server == server), "duplicate zone " + origin + " at " + words[2]);
231 current = new Zone(origin, server, line);
232 zones.add(current);
233 return;
234 }
235 require(current != null, "no current zone");
236 String owner = name(words[0], true, current.origin);
237 require(below(owner, current.origin), "out of zone " + owner);
238 int ttl = ttl(words[1]), type = type(words[2]);
239 String[] data = Arrays.copyOfRange(words, 3, words.length);
240 require(type == 6 || data.length == (type == 5 ? 2 : type == 3 ? 7 : 1), "wrong field count");
241 String text;
242 if (type == 0) text = address(ipv4(data[0]));
243 else if (type == 7) text = address6(ipv6(data[0]));
244 else if (type == 1 || type == 2 || type == 4) text = name(data[0], false, current.origin);
245 else if (type == 5) {
246 Integer preference = integer(data[0]);
247 require(preference != null && preference <= 65535, "bad preference " + data[0]);
248 text = preference + " " + name(data[1], false, current.origin);
249 } else if (type == 6) {
250 text = String.join(" ", data);
251 require(text.length() >= 2 && text.startsWith("\"") && text.endsWith("\"") && !text.substring(1, text.length() - 1).contains("\""), "bad text " + text);
252 } else {
253 List<String> fields = new ArrayList<>();
254 fields.add(name(data[0], false, current.origin));
255 fields.add(name(data[1], false, current.origin));
256 for (int i = 2; i < 6; i++) {
257 require(integer(data[i]) != null, "bad number " + data[i]);
258 fields.add(data[i]);
259 }
260 Integer minimum = integer(data[6]);
261 require(minimum != null && minimum >= 1 && minimum <= 86400, "bad minimum " + data[6]);
262 fields.add(data[6]);
263 text = String.join(" ", fields);
264 }
265 require(type != 1 || !owner.startsWith("*."), "ns at wildcard");
266 require(type != 3 || owner.equals(current.origin), "soa not at origin");
267 require(type != 3 || current.soa() == null, "duplicate soa");
268 String normalized = text;
269 require(current.records.stream().noneMatch(r -> r.owner().equals(owner) && r.type() == type && r.data().equals(normalized)), "duplicate record");
270 require(current.records.stream().noneMatch(r -> r.owner().equals(owner) && (type == 2 || r.type() == 2)), "cname conflict " + owner);
271 current.records.add(new Record(owner, ttl, type, text));
272 }

290 static Response authoritative(long server, String name, int type, String cut) {
291 if (down.contains(server) || zones.stream().noneMatch(z -> z.server == server)) return new Response("TIMEOUT");
292 Zone zone = null;
293 for (Zone candidate : zones) if (candidate.server == server && below(name, candidate.origin) && (zone == null || length(candidate.origin) > length(zone.origin))) zone = candidate;
294 if (zone == null || length(zone.origin) < length(cut)) return new Response("LAME");
295 String delegation = null;
296 for (Record record : zone.records) if (record.type() == 1 && !record.owner().equals(zone.origin) && below(name, record.owner()) && (delegation == null || length(record.owner()) < length(delegation))) delegation = record.owner();
297 if (delegation != null) {
298 Response response = new Response("REFERRAL");
299 response.cut = delegation;
300 response.authority = records(zone, delegation, 1);
301 for (Record ns : response.authority) response.additional.addAll(records(zone, ns.data(), 0));
302 return response;
303 }
304 List<Record> chosen = new ArrayList<>();
305 for (Record record : zone.records) if (record.owner().equals(name)) chosen.add(record);
306 if (chosen.isEmpty() && !exists(zone, name)) {
307 String ancestor = parent(name);
308 while (!exists(zone, ancestor)) ancestor = parent(ancestor);
309 String wildcard = ancestor.equals(".") ? "*." : "*." + ancestor;
310 for (Record record : zone.records) if (record.owner().equals(wildcard)) chosen.add(record);
311 }
312 Response response;
313 if (chosen.isEmpty()) response = new Response(exists(zone, name) ? "NODATA" : "NXDOMAIN");
314 else {
315 List<Record> answer = chosen.stream().filter(r -> r.type() == type).map(r -> r.owner(name)).toList();
316 if (!answer.isEmpty()) {
317 response = new Response("ANSWER");
318 response.answer = sorted(answer);
319 return response;
320 }
321 if (type != 2) for (Record record : chosen) if (record.type() == 2) {
322 response = new Response("CNAME");
323 response.answer.add(record.owner(name));
324 return response;
325 }
326 response = new Response("NODATA");
327 }
328 response.authority.add(zone.soa());
329 return response;
330 }

353 static Response queryServers(String name, int type, String cut, List<String> servers, int depth, List<Key> resolving) {
354 if (servers == null) {
355 for (long server : roots) {
356 Response response = ask(server, name, type, cut);
357 trace(depth, "ASK " + address(server) + " " + name + " " + TYPES[type] + " -> " + response.description());
358 if (!response.kind.equals("LAME") && !response.kind.equals("TIMEOUT")) return response;
359 }
360 } else {
361 List<String> ordered = new ArrayList<>(new HashSet<>(servers));
362 ordered.sort(Main::compareNames);
363 for (String host : ordered) {
364 TreeSet<Long> addresses = new TreeSet<>();
365 CacheEntry entry = cache.get(new Key(host, 0));
366 if (entry != null && entry.kind().equals("POS")) for (Record record : entry.records()) addresses.add(ipv4(record.data()));
367 else if (resolving.contains(new Key(host, 0))) trace(depth, "GLUELESS " + host + " -> LOOP");
368 else {
369 trace(depth, "GLUELESS " + host);
370 Result result = resolve(host, 0, depth + 1, resolving);
371 if (result.status().equals("NOERROR")) for (Record record : result.answer()) if (record.type() == 0) addresses.add(ipv4(record.data()));
372 }
373 for (long server : addresses) {
374 Response response = ask(server, name, type, cut);
375 trace(depth, "ASK " + address(server) + " " + name + " " + TYPES[type] + " -> " + response.description());
376 if (!response.kind.equals("LAME") && !response.kind.equals("TIMEOUT")) return response;
377 }
378 }
379 }
380 return null;
381 }

383 static Response lookup(String name, int type, int depth, List<Key> resolving) {
384 CacheEntry exact = cache.get(new Key(name, type));
385 Response cached = null;
386 if (exact != null) {
387 cached = new Response(exact.kind().equals("POS") ? "ANSWER" : exact.kind());
388 if (exact.kind().equals("POS")) cached.answer = cachedRecords(exact);
389 }
390 if (cached == null) for (var entry : cache.entrySet()) if (entry.getValue().kind().equals("NXDOMAIN") && below(name, entry.getKey().name())) {
391 cached = new Response("NXDOMAIN");
392 break;
393 }
394 if (cached == null && type != 2) {
395 CacheEntry cname = cache.get(new Key(name, 2));
396 if (cname != null && cname.kind().equals("POS")) {
397 cached = new Response("CNAME");
398 cached.answer = cachedRecords(cname);
399 }
400 }
401 if (cached != null) {
402 hits++;
403 trace(depth, "CACHE HIT " + name + " " + TYPES[type] + " -> " + cached.description());
404 return cached;
405 }
406 String cut = ".";
407 List<String> servers = null;
408 for (var entry : cache.entrySet()) if (entry.getKey().type() == 1 && entry.getValue().kind().equals("POS") && below(name, entry.getKey().name()) && (servers == null || length(entry.getKey().name()) > length(cut))) {
409 cut = entry.getKey().name();
410 servers = entry.getValue().records().stream().map(Record::data).toList();
411 }
412 if (servers != null) {
413 hits++;
414 trace(depth, "CACHE HIT " + name + " " + TYPES[type] + " -> REFERRAL " + cut);
415 }
416 while (true) {
417 Response response = queryServers(name, type, cut, servers, depth, resolving);
418 if (response == null) return new Response("SERVFAIL no server for " + cut);
419 if (response.kind.equals("REFERRAL")) {
420 store(response.authority);
421 Map<String, List<Record>> glue = new LinkedHashMap<>();
422 for (Record record : response.additional) glue.computeIfAbsent(record.owner(), ignored -> new ArrayList<>()).add(record);
423 for (List<Record> group : glue.values()) store(group);
424 cut = response.cut;
425 servers = response.authority.stream().map(Record::data).toList();
426 continue;
427 }
428 if (response.kind.equals("ANSWER") || response.kind.equals("CNAME")) {
429 int ttl = store(response.answer);
430 response.answer = response.answer.stream().map(r -> r.ttl(ttl)).toList();
431 } else {
432 Record soa = response.authority.getFirst();
433 int minimum = Integer.parseInt(soa.data().split(" ")[6]);
434 cache.put(new Key(name, type), new CacheEntry(response.kind, List.of(), now + Math.min(negTtl, Math.min(soa.ttl(), minimum))));
435 }
436 return response;
437 }

440 static Result resolve(String name, int type, int depth, List<Key> resolving) {
441 resolving.add(new Key(name, type));
442 List<Record> aliases = new ArrayList<>();
443 Set<String> seen = new HashSet<>();
444 seen.add(name);
445 String current = name, status;
446 List<Record> answer;
447 while (true) {
448 Response response = lookup(current, type, depth, resolving);
449 if (response.kind.equals("CNAME")) {
450 Record alias = response.answer.getFirst();
451 aliases.add(alias);
452 String target = alias.data();
453 if (seen.contains(target)) {
454 status = "SERVFAIL cname loop";
455 answer = List.of();
456 break;
457 }
458 if (aliases.size() == 9) {
459 status = "SERVFAIL cname chain too long";
460 answer = List.of();
461 break;
462 }
463 seen.add(target);
464 current = target;
465 continue;
466 }
467 if (response.kind.equals("ANSWER")) {
468 status = "NOERROR";
469 answer = new ArrayList<>(aliases);
470 answer.addAll(response.answer);
471 } else if (response.kind.equals("NODATA") || response.kind.equals("NXDOMAIN")) {
472 status = response.kind;
473 answer = aliases;
474 } else {
475 status = response.kind;
476 answer = List.of();
477 }
478 break;
479 }
480 resolving.removeLast();
481 resolutions[status.equals("NOERROR") ? 0 : status.equals("NODATA") ? 1 : status.equals("NXDOMAIN") ? 2 : 3]++;
482 return new Result(status, answer);
483 }

508 static void command(String[] words) {
509 switch (words[0]) {
510 case "RESOLVE":
511 {
512 String name = name(words[1], true, null);
513 int type = type(words[2]);
514 System.out.println("resolve " + name + " " + TYPES[type]);
515 answer(resolve(name, type));
516 break;
517 }
518 case "REVERSE":
519 {
520 String output, reverse;
521 if (words[1].contains(":")) {
522 int[] groups = ipv6(words[1]);
523 output = address6(groups);
524 StringBuilder digits = new StringBuilder();
525 for (int group : groups) digits.append(String.format(Locale.ROOT, "%04x", group));
526 List<String> parts = new ArrayList<>();
527 for (int i = 31; i >= 0; i--) parts.add("" + digits.charAt(i));
528 reverse = String.join(".", parts) + ".ip6.arpa.";
529 } else {
530 long ip = ipv4(words[1]);
531 output = address(ip);
532 String[] fields = output.split("\\.");
533 reverse = fields[3] + "." + fields[2] + "." + fields[1] + "." + fields[0] + ".in-addr.arpa.";
534 }
535 System.out.println("reverse " + output + " " + reverse);
536 answer(resolve(reverse, 4));
537 break;
538 }
539 case "MX":
540 {
541 String name = name(words[1], true, null);
542 System.out.println("mx " + name);
543 Result result = resolve(name, 5);
544 answer(result);
545 if (result.status().equals("NOERROR")) for (Record record : result.answer()) if (record.type() == 5) {
546 String[] fields = record.data().split(" ");
547 if (fields[1].equals(".")) System.out.println("exchange " + fields[0] + " . -> null");
548 else addresses(fields[1], "exchange " + fields[0] + " ");
549 }
550 if (result.status().equals("NODATA")) {
551 String host = name;
552 for (Record record : result.answer()) if (record.type() == 2) host = record.data();
553 addresses(host, "implicit ");
554 }
555 break;
556 }
557 case "QUERY":
558 {
559 long server = ipv4(words[1]);
560 String name = name(words[2], true, null);
561 int type = type(words[3]);
562 Response response = ask(server, name, type, ".");
563 System.out.println("query " + address(server) + " " + name + " " + TYPES[type] + " -> " + response.description());
564 for (Record record : response.answer) System.out.println("answer " + record);
565 for (Record record : response.authority) System.out.println("authority " + record);
566 for (Record record : response.additional) System.out.println("additional " + record);
567 break;
568 }
569 case "DUMP":
570 {
571 long server = ipv4(words[1]);
572 String name = name(words[2], false, null);
573 if (down.contains(server) || zones.stream().noneMatch(z -> z.server == server)) {
574 System.out.println("dump " + address(server) + " " + name + " -> TIMEOUT");
575 break;
576 }
577 Zone zone = zones.stream().filter(z -> z.server == server && z.origin.equals(name)).findFirst().orElse(null);
578 if (zone == null) System.out.println("dump " + address(server) + " " + name + " -> REFUSED");
579 else {
580 System.out.println("dump " + address(server) + " " + name + " records=" + zone.records.size());
581 zone.records.stream().sorted((a, b) -> {
582 int order = compareNames(a.owner(), b.owner()); if (order == 0) order = Integer.compare(a.type(), b.type()); return order == 0 ? compareData(a, b) : order;
583 }).forEach(r -> System.out.println(" " + r));
584 }
585 break;
586 }
587 case "DOWN", "UP":
588 {
589 long server = ipv4(words[1]);
590 boolean disable = words[0].equals("DOWN");
591 require(disable ? !down.contains(server) : down.contains(server), "server " + words[1] + (disable ? " already down" : " not down"));
592 if (disable) down.add(server);
593 else down.remove(server);
594 System.out.println((disable ? "down " : "up ") + address(server));
595 break;
596 }
597 case "SET":
598 {
599 require(Set.of("MINTTL", "MAXTTL", "NEGTTL").contains(words[1]), "bad setting " + words[1]);
600 Integer value = integer(words[2]);
601 require(value != null && value >= 1 && value <= 604800, "bad value " + words[2]);
602 require((words[1].equals("MINTTL") ? value : minTtl) <= (words[1].equals("MAXTTL") ? value : maxTtl), "minttl above maxttl");
603 if (words[1].equals("MINTTL")) minTtl = value;
604 else if (words[1].equals("MAXTTL")) maxTtl = value;
605 else negTtl = value;
606 System.out.println("set " + words[1] + "=" + value);
607 break;
608 }
609 case "TIME":
610 {
611 Integer value = integer(words[1]);
612 require(value != null && value >= 1 && value <= 1000000, "bad time " + words[1]);
613 now += value;
614 int before = cache.size();
615 cache.entrySet().removeIf(e -> e.getValue().expiry() <= now);
616 System.out.println("time now=" + now + " expired=" + (before - cache.size()));
617 break;
618 }
619 case "CACHE", "FLUSH":
620 {
621 String name = words.length == 2 ? name(words[1], false, null) : null;
622 var entries = selected(name);
623 if (words[0].equals("FLUSH")) {
624 for (var entry : entries) cache.remove(entry.getKey());
625 System.out.println("flush removed=" + entries.size());
626 } else {
627 System.out.println("cache entries=" + entries.size());
628 for (var entry : entries) if (entry.getValue().kind().equals("POS")) for (Record record : cachedRecords(entry.getValue())) System.out.println(" " + record);
629 else System.out.println(" " + entry.getKey().name() + " " + (entry.getValue().expiry() - now) + " " + TYPES[entry.getKey().type()] + " " + entry.getValue().kind());
630 }
631 break;
632 }
633 case "STATS": System.out.println("stats queries=" + queries.values().stream().mapToInt(a -> Arrays.stream(a).sum()).sum() + " hits=" + hits);
634 queries.forEach((server, counts) -> {
635 StringBuilder line = new StringBuilder(" " + address(server) + " sent=" + Arrays.stream(counts).sum()); for (int i = 0; i < KINDS.length; i++) line.append(" ").append(KINDS[i].toLowerCase(Locale.ROOT)).append("=").append(counts[i]); System.out.println(line);
636 });
637 System.out.println("resolutions noerror=" + resolutions[0] + " nodata=" + resolutions[1] + " nxdomain=" + resolutions[2] + " servfail=" + resolutions[3]);
638 }
