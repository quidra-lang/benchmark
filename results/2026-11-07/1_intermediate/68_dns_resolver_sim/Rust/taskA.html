7fn integer(s: &str) -> Option<i64> {
8 if s.is_empty() || s.len()>9 || (s.len()>1 && s.starts_with('0')) || !s.bytes().all(|c|c.is_ascii_digit()) {
9 None
10 } else {
11 s.parse().ok()
12 }
13}
14fn ip(s: &str) -> Option<u32> {
15 let fields:Vec<_>=s.split('.').collect();
16 if fields.len()!=4 {
17 return None;
18 }
19 let mut a=0;
20 for p in fields {
21 if p.len()>3 {
22 return None;
23 }
24 let n=integer(p)?;
25 if n>255 {
26 return None;
27 }
28 a=a*256+n as u32;
29 }
30 Some(a)
31}
32fn address(a: u32) -> String {
33 format!("{}.{}.{}.{}",a>>24,(a>>16)&255,(a>>8)&255,a&255)
34}
35fn ipv6(s: &str) -> Option<[u16; 8]> {
36 fn groups(s: &str) -> Option<Vec<u16>> {
37 if s.is_empty() {
38 return Some(Vec::new());
39 }
40 s.split(':').map(|p| if (1..=4).contains(&p.len()) && p.bytes().all(|c|c.is_ascii_hexdigit()) {
41 u16::from_str_radix(p,16).ok()
42 } else {
43 None
44 }).collect()
45 }
46 let values=if let Some((a,b))=s.split_once("::") {
47 let left=groups(a)?;
48 let right=groups(b)?;
49 if left.len()+right.len()>7 {
50 return None;
51 }
52 let mut values=left;
53 values.resize(8-right.len(),0);
54 values.extend(right);
55 values
56 } else {
57 let values=groups(s)?;
58 if values.len()!=8 {
59 return None;
60 }
61 values
62 };
63 values.try_into().ok()
64}
65fn address6(a: &[u16; 8]) -> String {
66 let mut best=(0,0);
67 let mut at=0;
68 while at<8 {
69 if a[at]!=0 {
70 at+=1;
71 continue;
72 }
73 let start=at;
74 while at<8 && a[at]==0 {
75 at+=1;
76 }
77 if at-start>=2 && at-start>best.1 {
78 best=(start,at-start);
79 }
80 }
81 let text=|part:&[u16]|part.iter().map(|n|format!("{n:x}")).collect::<Vec<_>>().join(":");
82 if best.1==0 {
83 text(a)
84 } else {
85 format!("{}::{}",text(&a[..best.0]),text(&a[best.0+best.1..]))
86 }
87}
88fn name(word: &str, wild: bool, origin: Option<&str>) -> Result<String,String> {
89 if word=="@" {
90 if let Some(origin)=origin {
91 return Ok(origin.into());
92 }
93 }
94 if word=="." {
95 return Ok(".".into());
96 }
97 let labels:Vec<_>=word.strip_suffix('.').unwrap_or(word).split('.').collect();
98 for (i,label) in labels.iter().enumerate() {
99 if wild && i==0 && *label=="*" {
100 continue;
101 }
102 if label.is_empty() || label.len()>63 || label.starts_with('-') || label.ends_with('-') || !label.bytes().all(|c|c.is_ascii_alphanumeric() || c==b'-') {
103 return Err(format!("bad name {word}"));
104 }
105 }
106 let mut result=labels.join(".").to_ascii_lowercase()+".";
107 if !word.ends_with('.') {
108 if let Some(origin)=origin.filter(|&s|s!=".") {
109 result+=origin;
110 }
111 }
112 Ok(result)
113}
114fn labels(s: &str) -> Vec<&str> {
115 if s=="." {
116 Vec::new()
117 } else {
118 s[..s.len()-1].split('.').collect()
119 }
120}
121fn name_order(a: &str,b: &str) -> Ordering {
122 labels(a).into_iter().rev().cmp(labels(b).into_iter().rev())
123}
124fn below(a: &str,b: &str) -> bool {
125 b=="." || a==b || a.ends_with(&(".".to_string()+b))
126}
127fn ancestors(s: &str) -> Vec<String> {
128 let parts=labels(s);
129 let mut out:Vec<_>=(0..parts.len()).map(|i|parts[i..].join(".")+".").collect();
130 out.push(".".into());
131 out
132}

143fn ttl(s: &str) -> Result<i64,String> {
144 let parsed=integer(s).or_else(|| {
145 let b=s.as_bytes();
146 let mut at=0;
147 let mut previous=None;
148 let mut total=0;
149 while at<b.len() {
150 let start=at;
151 while at<b.len() && b[at].is_ascii_digit() {
152 at+=1;
153 }
154 if at==b.len() || at-start>3 {
155 return None;
156 }
157 let n=integer(&s[start..at])?;
158 let unit=b"wdhms".iter().position(|&c|c==b[at].to_ascii_lowercase())?;
159 if previous.is_some_and(|p|unit<=p) {
160 return None;
161 }
162 previous=Some(unit);
163 total+=n*[604800,86400,3600,60,1][unit];
164 at+=1;
165 }
166 previous.map(|_|total)
167 });
168 parsed.filter(|&n|(1..=604800).contains(&n)).ok_or_else(||format!("bad ttl {s}"))
169}
170#[derive(Clone,PartialEq,Eq)]
171enum Data {
172 A(u32), Aaaa([u16; 8]), Name(String), Mx(i64,String), Txt(String), Soa(String,String,[i64; 5])
173}
174impl Data {
175 fn text(&self) -> String {
176 match self {
177 Self::A(a)=>address(*a),Self::Aaaa(a)=>address6(a),Self::Name(n)=>n.clone(),Self::Mx(p,n)=>format!("{p} {n}"),Self::Txt(s)=>s.clone(),Self::Soa(a,b,n)=>format!("{a} {b} {}",n.iter().map(i64::to_string).collect::<Vec<_>>().join(" "))
178 }
179 }
180 fn compare(&self,b: &Self) -> Ordering {
181 match (self,b) {
182 (Self::A(a),Self::A(b))=>a.cmp(b),(Self::Aaaa(a),Self::Aaaa(b))=>a.cmp(b),(Self::Name(a),Self::Name(b))=>name_order(a,b),(Self::Mx(a,x),Self::Mx(b,y))=>a.cmp(b).then_with(||name_order(x,y)),(Self::Txt(a),Self::Txt(b))=>a.cmp(b),_=>Ordering::Equal
183 }
184 }
185}

195fn record_order(a: &Record,b: &Record) -> Ordering {
196 name_order(&a.owner,&b.owner).then(TYPES.iter().position(|&t|t==a.kind).cmp(&TYPES.iter().position(|&t|t==b.kind))).then_with(||a.data.compare(&b.data))
197}

230 fn zone_line(&mut self,w: &[&str],line:usize) -> Result<(),String> {
231 if w[0]=="ROOT" {
232 let mut added=Vec::new();
233 for word in &w[1..] {
234 let a=checked_ip(word)?;
235 if self.roots.contains(&a) || added.contains(&a) {
236 return Err(format!("duplicate root {word}"));
237 }
238 added.push(a);
239 }
240 self.roots.extend(added);
241 return Ok(());
242 }
243 if w[0]=="ZONE" {
244 self.current=None;
245 let origin=name(w[1],false,None)?;
246 let server=checked_ip(w[2])?;
247 if self.zones.iter().any(|z|z.origin==origin && z.server==server) {
248 return Err(format!("duplicate zone {origin} at {}",w[2]));
249 }
250 self.current=Some(self.zones.len());
251 self.zones.push(Zone {
252 origin,server,line,records:Vec::new()
253 });
254 return Ok(());
255 }
256 let i=self.current.ok_or("no current zone")?;
257 let z=&self.zones[i];
258 let origin=z.origin.as_str();
259 let owner=name(w[0],true,Some(origin))?;
260 if !below(&owner,origin) {
261 return Err(format!("out of zone {owner}"));
262 }
263 let seconds=ttl(w[1])?;
264 let kind=w[2];
265 checked_type(kind)?;
266 let fields=&w[3..];
267 let expected=match kind {
268 "MX"=>2,"SOA"=>7,"TXT"=>fields.len(),_=>1
269 };
270 if fields.len()!=expected {
271 return Err("wrong field count".into());
272 }
273 let data=match kind {
274 "A"=>Data::A(checked_ip(fields[0])?),"AAAA"=>Data::Aaaa(ipv6(fields[0]).ok_or_else(||format!("bad address {}",fields[0]))?),
275 "NS"|"CNAME"|"PTR"=>Data::Name(name(fields[0],false,Some(origin))?),
276 "MX"=>{
277 let p=integer(fields[0]).filter(|&n|n<=65535).ok_or_else(||format!("bad preference {}",fields[0]))?;
278 Data::Mx(p,name(fields[1],false,Some(origin))?)
279 },
280 "TXT"=>{
281 let s=fields.join(" ");
282 if s.len()<2 || !s.starts_with('"') || !s.ends_with('"') || s[1..s.len()-1].contains('"') {
283 return Err(format!("bad text {s}"));
284 }
285 Data::Txt(s)
286 },
287 _=>{
288 let a=name(fields[0],false,Some(origin))?;
289 let b=name(fields[1],false,Some(origin))?;
290 let mut nums=[0; 5];
291 for j in 0..4 {
292 nums[j]=integer(fields[j+2]).ok_or_else(||format!("bad number {}",fields[j+2]))?;
293 }
294 nums[4]=integer(fields[6]).filter(|&n|(1..=86400).contains(&n)).ok_or_else(||format!("bad minimum {}",fields[6]))?;
295 Data::Soa(a,b,nums)
296 },
297 };
298 if kind=="NS" && owner.starts_with("*.") {
299 return Err("ns at wildcard".into());
300 }
301 if kind=="SOA" && owner!=origin {
302 return Err("soa not at origin".into());
303 }
304 if kind=="SOA" && z.records.iter().any(|r|r.kind=="SOA") {
305 return Err("duplicate soa".into());
306 }
307 if z.records.iter().any(|r|r.owner==owner && r.kind==kind && r.data==data) {
308 return Err("duplicate record".into());
309 }
310 if z.records.iter().any(|r|r.owner==owner && (kind=="CNAME" || r.kind=="CNAME")) {
311 return Err(format!("cname conflict {owner}"));
312 }
313 self.zones[i].records.push(Record {
314 owner,ttl:seconds,kind:kind.into(),data
315 });
316 Ok(())
317 }
318 fn finish_zones(&mut self) {
319 let zones=std::mem::take(&mut self.zones);
320 for z in zones {
321 if z.records.iter().any(|r|r.kind=="SOA") {
322 self.out.push(format!("zone {} at {} records={}",z.origin,address(z.server),z.records.len()));
323 self.zones.push(z);
324 } else {
325 self.out.push(format!("error line {}: zone {} at {} has no soa",z.line,z.origin,address(z.server)));
326 }
327 }
328 self.out.push(format!("loaded roots={} zones={} records={}",self.roots.len(),self.zones.len(),self.zones.iter().map(|z|z.records.len()).sum::<usize>()));
329 }
330 fn records(z: &Zone,n: &str,t: &str) -> Vec<Record> {
331 let mut r:Vec<_>=z.records.iter().filter(|r|r.owner==n && r.kind==t).cloned().collect();
332 r.sort_by(|a,b|a.data.compare(&b.data));
333 r

335 fn authoritative(&self,server:u32,n: &str,t: &str,cut: &str) -> Response {
336 if self.down.contains(&server) || !self.zones.iter().any(|z|z.server==server) {
337 return Response::empty(6);
338 }
339 let Some(z)=self.zones.iter().filter(|z|z.server==server && below(n,&z.origin)).max_by_key(|z|labels(&z.origin).len()) else {
340 return Response::empty(5);
341 };
342 if labels(&z.origin).len()<labels(cut).len() {
343 return Response::empty(5);
344 }
345 let mut ancestors=ancestors(n);
346 ancestors.reverse();
347 for d in ancestors.iter().filter(|d|d.as_str()!=z.origin && below(d,&z.origin)) {
348 let ns=Self::records(z,d,"NS");
349 if ns.is_empty() {
350 continue;
351 }
352 let mut response=Response::empty(2);
353 response.cut=d.clone();
354 for r in &ns {
355 if let Data::Name(host)=&r.data {
356 response.additional.extend(Self::records(z,host,"A"));
357 }
358 }
359 response.authority=ns;
360 return response;
361 }
362 let exists=|x:&str|z.records.iter().any(|r|below(&r.owner,x));
363 let mut owner=n.to_string();
364 if !z.records.iter().any(|r|r.owner==n) {
365 if exists(n) {
366 let mut r=Response::empty(3);
367 r.authority=Self::records(z,&z.origin,"SOA");
368 return r;
369 }
370 let ancestor=ancestors.iter().rev().find(|a|a.as_str()!=n && exists(a)).unwrap();
371 owner=if ancestor=="." {
372 "*.".into()
373 } else {
374 "*.".to_string()+ancestor
375 };
376 if !z.records.iter().any(|r|r.owner==owner) {
377 let mut r=Response::empty(4);
378 r.authority=Self::records(z,&z.origin,"SOA");
379 return r;
380 }
381 }
382 let mut answer=Self::records(z,&owner,t);
383 let mut kind=0;
384 if answer.is_empty() && t!="CNAME" {
385 answer=Self::records(z,&owner,"CNAME");
386 kind=1;
387 }
388 if answer.is_empty() {
389 let mut r=Response::empty(3);
390 r.authority=Self::records(z,&z.origin,"SOA");
391 return r;
392 }
393 for r in &mut answer {
394 r.owner=n.into();
395 }
396 let mut response=Response::empty(kind);
397 response.answer=answer;
398 response
399 }

401 let r=self.authoritative(a,n,t,cut);
402 self.queries.entry(a).or_insert([0; 7])[r.kind]+=1;
403 r

408 fn positive(&mut self,records:Vec<Record>) -> i64 {
409 let seconds=records.iter().map(|r|r.ttl).min().unwrap().max(self.settings[0]).min(self.settings[1]);
410 let key=(records[0].owner.clone(),records[0].kind.clone());
411 self.cache.insert(key,Cache {
412 kind:0,records,expiry:self.clock+seconds
413 });
414 seconds

417 let entry=self.cache.get(key).filter(|e|e.kind==0)?;
418 Some(entry.records.iter().cloned().map(|mut r| {
419 r.ttl=entry.expiry-self.clock;

422 }
423 fn query_servers(&mut self,n:&str,t:&str,cut:&str,servers:Option<Vec<String>>,depth:usize,stack:&mut Vec<(String,String)>) -> Option<Response> {
424 if let Some(mut servers)=servers {
425 servers.sort_by(|a,b|name_order(a,b));
426 for host in servers {
427 let records=if let Some(records)=self.cached_records(&(host.clone(),"A".into())) {
428 records
429 }
430 else if stack.contains(&(host.clone(),"A".into())) {
431 self.trace(depth,format!("GLUELESS {host} -> LOOP"));
432 Vec::new()
433 }
434 else {
435 self.trace(depth,format!("GLUELESS {host}"));
436 let (status,answer)=self.resolve(&host,"A",depth+1,stack);
437 if status=="NOERROR" {
438 answer
439 } else {
440 Vec::new()
441 }
442 };
443 let addresses:BTreeSet<_>=records.iter().filter_map(|r|if let Data::A(a)=r.data {
444 Some(a)
445 } else {
446 None
447 }).collect();
448 for a in addresses {
449 let r=self.ask(a,n,t,cut);
450 self.trace(depth,format!("ASK {} {n} {t} -> {}",address(a),r.text()));
451 if r.kind<5 {
452 return Some(r);
453 }
454 }
455 }
456 } else {
457 for a in self.roots.clone() {
458 let r=self.ask(a,n,t,cut);
459 self.trace(depth,format!("ASK {} {n} {t} -> {}",address(a),r.text()));
460 if r.kind<5 {
461 return Some(r);
462 }
463 }
464 }
465 None
466 }
467 fn lookup(&mut self,n:&str,t:&str,depth:usize,stack:&mut Vec<(String,String)>) -> Result<Response,String> {
468 if let Some(entry)=self.cache.get(&(n.into(),t.into())).cloned() {
469 let mut r=Response::empty(if entry.kind==0 {
470 0
471 } else {
472 entry.kind
473 });
474 if entry.kind==0 {
475 r.answer=self.cached_records(&(n.into(),t.into())).unwrap();
476 }
477 self.hits+=1;
478 self.trace(depth,format!("CACHE HIT {n} {t} -> {}",KINDS[r.kind]));
479 return Ok(r);
480 }
481 if self.cache.iter().any(|((m,_),e)|e.kind==4 && below(n,m)) {
482 self.hits+=1;
483 self.trace(depth,format!("CACHE HIT {n} {t} -> NXDOMAIN"));
484 return Ok(Response::empty(4));
485 }
486 if t!="CNAME" {
487 if let Some(records)=self.cached_records(&(n.into(),"CNAME".into())) {
488 let mut r=Response::empty(1);
489 r.answer=records;
490 self.hits+=1;
491 self.trace(depth,format!("CACHE HIT {n} {t} -> {}",r.text()));
492 return Ok(r);
493 }
494 }
495 let referral=self.cache.iter().filter(|((m,t),e)|t=="NS" && e.kind==0 && below(n,m)).max_by_key(|((m,_),_)|labels(m).len()).map(|((m,_),e)|(m.clone(),e.records.clone()));
496 let (mut cut,mut servers)=if let Some((cut,records))=referral {
497 self.hits+=1;
498 self.trace(depth,format!("CACHE HIT {n} {t} -> REFERRAL {cut}"));
499 (cut,Some(records.iter().map(|r|r.data.text()).collect()))
500 } else {
501 (".".into(),None)
502 };
503 loop {
504 let mut r=self.query_servers(n,t,&cut,servers,depth,stack).ok_or_else(||format!("SERVFAIL no server for {cut}"))?;
505 match r.kind {
506 2=>{
507 self.positive(r.authority.clone());
508 let mut groups:BTreeMap<String,Vec<Record>>=BTreeMap::new();
509 for record in r.additional {
510 groups.entry(record.owner.clone()).or_default().push(record);
511 }
512 for records in groups.into_values() {
513 self.positive(records);
514 }
515 cut=r.cut;
516 servers=Some(r.authority.iter().map(|r|r.data.text()).collect());
517 }
518 0|1=>{
519 let seconds=self.positive(r.answer.clone());
520 for record in &mut r.answer {
521 record.ttl=seconds;
522 }
523 return Ok(r);
524 }
525 3|4=>{
526 let soa=&r.authority[0];
527 let minimum=if let Data::Soa(_,_,n)=&soa.data {
528 n[4]
529 } else {
530 unreachable!()
531 };
532 let seconds=soa.ttl.min(minimum).min(self.settings[2]);
533 self.cache.insert((n.into(),t.into()),Cache {
534 kind:r.kind,records:Vec::new(),expiry:self.clock+seconds
535 });
536 return Ok(r);
537 }
538 _=>unreachable!(),
539 }
540 }
541 }
542 fn resolve(&mut self,n:&str,t:&str,depth:usize,stack:&mut Vec<(String,String)>) -> (String,Vec<Record>) {
543 stack.push((n.into(),t.into()));
544 let mut chain=Vec::new();
545 let mut seen=BTreeSet::from([n.to_string()]);
546 let mut current=n.to_string();
547 let result=loop {
548 match self.lookup(&current,t,depth,stack) {
549 Err(reason)=>break (reason,Vec::new()),
550 Ok(r)=>match r.kind {
551 1=>{
552 let target=r.answer[0].data.text();
553 chain.extend(r.answer);
554 if seen.contains(&target) {
555 break ("SERVFAIL cname loop".into(),Vec::new());
556 }
557 if chain.len()==9 {
558 break ("SERVFAIL cname chain too long".into(),Vec::new());
559 }
560 seen.insert(target.clone());
561 current=target;
562 }
563 0=>{
564 chain.extend(r.answer);
565 break ("NOERROR".into(),chain);
566 }
567 _=>break (KINDS[r.kind].into(),chain),
568 },
569 }
570 };
571 stack.pop();
572 let index=match result.0.as_str() {
573 "NOERROR"=>0,"NODATA"=>1,"NXDOMAIN"=>2,_=>3
574 };
575 self.resolutions[index]+=1;
576 result
577 }
578 fn output_resolution(&mut self,n:&str,t:&str) -> (String,Vec<Record>) {
579 let result=self.resolve(n,t,0,&mut Vec::new());
580 for r in &result.1 {
581 self.out.push("answer ".to_string()+&r.text());
582 }
583 self.out.push("status ".to_string()+&result.0);
584 result
585 }

596 fn command(&mut self,w:&[&str]) -> Result<(),String> {
597 match w[0] {
598 "RESOLVE"=>{
599 let n=name(w[1],true,None)?;
600 checked_type(w[2])?;
601 self.out.push(format!("resolve {n} {}",w[2]));
602 self.output_resolution(&n,w[2]);
603 }
604 "REVERSE"=>{
605 let (a,n)=if w[1].contains(':') {
606 let a=ipv6(w[1]).ok_or_else(||format!("bad address {}",w[1]))?;
607 let digits=a.iter().map(|n|format!("{n:04x}")).collect::<String>();
608 (address6(&a),digits.chars().rev().map(|c|c.to_string()).collect::<Vec<_>>().join(".")+".ip6.arpa.")
609 }
610 else {
611 let a=checked_ip(w[1])?;
612 let s=address(a);
613 (s.clone(),s.split('.').rev().collect::<Vec<_>>().join(".")+".in-addr.arpa.")
614 };
615 self.out.push(format!("reverse {a} {n}"));
616 self.output_resolution(&n,"PTR");
617 }
618 "MX"=>{
619 let n=name(w[1],true,None)?;
620 self.out.push(format!("mx {n}"));
621 let (status,answer)=self.output_resolution(&n,"MX");
622 if status=="NOERROR" {
623 for r in answer {
624 if let Data::Mx(p,host)=r.data {
625 if host=="." {
626 self.out.push(format!("exchange {p} . -> null"));
627 } else {
628 let addresses=self.host_addresses(&host);
629 self.out.push(format!("exchange {p} {host} -> {}",if addresses.is_empty() {
630 "no address".into()
631 } else {
632 addresses.join(" ")
633 }));
634 }
635 }
636 }
637 }
638 else if status=="NODATA" {
639 let host=answer.iter().rev().find(|r|r.kind=="CNAME").map_or(n.clone(),|r|r.data.text());
640 let addresses=self.host_addresses(&host);
641 self.out.push(format!("implicit {host} -> {}",if addresses.is_empty() {
642 "no address".into()
643 } else {
644 addresses.join(" ")
645 }));
646 }
647 }
648 "QUERY"=>{
649 let a=checked_ip(w[1])?;
650 let n=name(w[2],true,None)?;
651 checked_type(w[3])?;
652 let r=self.ask(a,&n,w[3],".");
653 self.out.push(format!("query {} {n} {} -> {}",address(a),w[3],r.text()));
654 let sections=match r.kind {
655 0|1=>vec![("answer",r.answer)],2=>vec![("authority",r.authority),("additional",r.additional)],3|4=>vec![("authority",r.authority)],_=>Vec::new()
656 };
657 for (section,records) in sections {
658 for record in records {
659 self.out.push(format!("{section} {}",record.text()));
660 }
661 }
662 }
663 "DUMP"=>{
664 let a=checked_ip(w[1])?;
665 let n=name(w[2],false,None)?;
666 if self.down.contains(&a) || !self.zones.iter().any(|z|z.server==a) {
667 self.out.push(format!("dump {} {n} -> TIMEOUT",address(a)));
668 } else if let Some(z)=self.zones.iter().find(|z|z.server==a && z.origin==n) {
669 let mut records=z.records.clone();
670 records.sort_by(record_order);
671 self.out.push(format!("dump {} {n} records={}",address(a),records.len()));
672 for r in records {
673 self.out.push(" ".to_string()+&r.text());
674 }
675 } else {
676 self.out.push(format!("dump {} {n} -> REFUSED",address(a)));
677 }
678 }
679 "UP"|"DOWN"=>{
680 let a=checked_ip(w[1])?;
681 if w[0]=="DOWN" {
682 if self.down.contains(&a) {
683 return Err(format!("server {} already down",w[1]));
684 }
685 self.down.insert(a);
686 } else if !self.down.remove(&a) {
687 return Err(format!("server {} not down",w[1]));
688 }
689 self.out.push(format!("{} {}",w[0].to_ascii_lowercase(),address(a)));
690 }
691 "SET"=>{
692 let i=["MINTTL","MAXTTL","NEGTTL"].iter().position(|&s|s==w[1]).ok_or_else(||format!("bad setting {}",w[1]))?;
693 let n=integer(w[2]).filter(|&n|(1..=604800).contains(&n)).ok_or_else(||format!("bad value {}",w[2]))?;
694 let mut settings=self.settings;
695 settings[i]=n;
696 if settings[0]>settings[1] {
697 return Err("minttl above maxttl".into());
698 }
699 self.settings=settings;
700 self.out.push(format!("set {}={n}",w[1]));
701 }
702 "TIME"=>{
703 let n=integer(w[1]).filter(|&n|(1..=1000000).contains(&n)).ok_or_else(||format!("bad time {}",w[1]))?;
704 self.clock+=n;
705 let old=self.cache.len();
706 let clock=self.clock;
707 self.cache.retain(|_,e|e.expiry>clock);
708 self.out.push(format!("time now={} expired={}",self.clock,old-self.cache.len()));
709 }
710 "CACHE"|"FLUSH"=>{
711 let selected=if w.len()==2 {
712 Some(name(w[1],false,None)?)
713 } else {
714 None
715 };
716 let mut keys:Vec<_>=self.cache.keys().filter(|(n,_)|selected.as_ref().is_none_or(|parent|below(n,parent))).cloned().collect();
717 keys.sort_by(|(a,t),(b,u)|name_order(a,b).then(TYPES.iter().position(|x|x==t).cmp(&TYPES.iter().position(|x|x==u))));
718 if w[0]=="FLUSH" {
719 self.out.push(format!("flush removed={}",keys.len()));
720 for key in keys {
721 self.cache.remove(&key);
722 }
723 }
724 else {
725 self.out.push(format!("cache entries={}",keys.len()));
726 for key in keys {
727 let e=&self.cache[&key];
728 if e.kind==0 {
729 for r in self.cached_records(&key).unwrap() {
730 self.out.push(" ".to_string()+&r.text());
731 }
732 } else {
733 self.out.push(format!(" {} {} {} {}",key.0,e.expiry-self.clock,key.1,KINDS[e.kind]));
734 }
735 }
736 }
737 }
738 _=>{
739 let total:usize=self.queries.values().map(|counts|counts.iter().sum::<usize>()).sum();
740 self.out.push(format!("stats queries={total} hits={}",self.hits));
741 for (a,c) in &self.queries {
742 self.out.push(format!(" {} sent={} answer={} cname={} referral={} nodata={} nxdomain={} lame={} timeout={}",address(*a),c.iter().sum::<usize>(),c[0],c[1],c[2],c[3],c[4],c[5],c[6]));
743 }
744 let c=self.resolutions;
745 self.out.push(format!("resolutions noerror={} nodata={} nxdomain={} servfail={}",c[0],c[1],c[2],c[3]));
746 }
747 }
748 Ok(())
749 }
750}
751fn main() {
752 let mut input=String::new();
753 io::stdin().read_to_string(&mut input).unwrap();
754 let mut resolver=Resolver::new();
755 let mut zones=true;
756 for (i,line) in input.lines().enumerate() {
757 if line=="END" {
758 break;
759 }
760 if line=="END ZONES" {
761 resolver.finish_zones();
762 zones=false;
763 continue;
764 }
765 let words:Vec<_>=line.split(' ').collect();
766 let result=if zones {
767 resolver.zone_line(&words,i+1)
768 } else {
769 resolver.command(&words)
770 };
771 if let Err(reason)=result {
772 resolver.out.push(format!("error line {}: {reason}",i+1));
773 }
774 }
775 for line in resolver.out {
776 println!("{line}");
777 }
778}
