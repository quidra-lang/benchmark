16func indexOf(xs []string, s string) int {
17 for i, x := range xs {
18 if x == s {
19 return i
20 }
21 }
22 return -1
23}
24func integerWord(s string) bool {
25 if len(s) < 1 || len(s) > 9 || len(s) > 1 && s[0] == '0' {
26 return false
27 }
28 for i := range s {
29 if s[i] < '0' || s[i] > '9' {
30 return false
31 }
32 }
33 return true
34}
35func number(s string) int {
36 n, _ := strconv.Atoi(s)
37 return n
38}
39func nameLabels(name string) []string {
40 if name == "." {
41 return nil
42 }
43 return strings.Split(name[:len(name)-1], ".")
44}
45func nameLength(name string) int {
46 return len(nameLabels(name))
47}
48func below(name, ancestor string) bool {
49 return ancestor == "." || name == ancestor || strings.HasSuffix(name, "."+ancestor)
50}
51func nameCompare(a, b string) int {
52 x, y := nameLabels(a), nameLabels(b)
53 for i, j := len(x)-1, len(y)-1; i >= 0 && j >= 0; i, j = i-1, j-1 {
54 if x[i] != y[j] {
55 return strings.Compare(x[i], y[j])
56 }
57 }
58 if len(x) < len(y) {
59 return -1
60 }
61 if len(x) > len(y) {
62 return 1
63 }
64 return 0
65}
66func ancestors(name string) []string {
67 out := []string{name}
68 for name != "." {
69 at := strings.IndexByte(name, '.')
70 name = name[at+1:]
71 if name == "" {
72 name = "."
73 }
74 out = append(out, name)
75 }
76 return out
77}
78func parseName(word, origin string, wildcard bool) (string, error) {
79 if origin != "" && word == "@" {
80 return origin, nil
81 }
82 if word == "." {
83 return ".", nil
84 }
85 absolute := strings.HasSuffix(word, ".")
86 text := strings.TrimSuffix(word, ".")
87 labels := strings.Split(text, ".")
88 for i, label := range labels {
89 if wildcard && i == 0 && label == "*" {
90 continue
91 }
92 if len(label) < 1 || len(label) > 63 || label[0] == '-' || label[len(label)-1] == '-' {
93 return "", fmt.Errorf("bad name %s", word)
94 }
95 for j := range label {
96 c := label[j]
97 if !(c >= 'a' && c <= 'z' || c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || c == '-') {
98 return "", fmt.Errorf("bad name %s", word)
99 }
100 }
101 }
102 result := strings.ToLower(text) + "."
103 if origin != "" && origin != "." && !absolute {
104 result += origin
105 }
106 return result, nil
107}
108func parseIPv4(s string) (uint32, error) {
109 fields := strings.Split(s, ".")
110 if len(fields) != 4 {
111 return 0, fmt.Errorf("bad address %s", s)
112 }
113 var a uint32
114 for _, field := range fields {
115 if len(field) > 3 || !integerWord(field) || number(field) > 255 {
116 return 0, fmt.Errorf("bad address %s", s)
117 }
118 a = a*256 + uint32(number(field))
119 }
120 return a, nil
121}
122func ipv4(a uint32) string {
123 return fmt.Sprintf("%d.%d.%d.%d", a>>24, a>>16&255, a>>8&255, a&255)
124}
125func parseIPv6(s string) ([8]uint16, error) {
126 var result [8]uint16
127 invalid := fmt.Errorf("bad address %s", s)
128 left, right, compressed := strings.Cut(s, "::")
129 parse := func(text string) ([]uint16, bool) {
130 if text == "" {
131 return []uint16{}, true
132 }
133 values := []uint16{}
134 for _, group := range strings.Split(text, ":") {
135 if len(group) < 1 || len(group) > 4 {
136 return nil, false
137 }
138 for i := range group {
139 if !strings.ContainsRune("0123456789abcdefABCDEF", rune(group[i])) {
140 return nil, false
141 }
142 }
143 n, _ := strconv.ParseUint(group, 16, 16)
144 values = append(values, uint16(n))
145 }
146 return values, true
147 }
148 a, ok := parse(left)
149 if !ok {
150 return result, invalid
151 }
152 if !compressed {
153 if len(a) != 8 {
154 return result, invalid
155 }
156 copy(result[:], a)
157 return result, nil
158 }
159 b, ok := parse(right)
160 if !ok || len(a)+len(b) > 7 {
161 return result, invalid
162 }
163 copy(result[:], a)
164 copy(result[8-len(b):], b)
165 return result, nil
166}
167func ipv6(a [8]uint16) string {
168 best, length := -1, 0
169 for i := 0; i < 8; {
170 if a[i] != 0 {
171 i++
172 continue
173 }
174 j := i
175 for j < 8 && a[j] == 0 {
176 j++
177 }
178 if j-i >= 2 && j-i > length {
179 best, length = i, j-i
180 }
181 i = j
182 }
183 groups := make([]string, 8)
184 for i, n := range a {
185 groups[i] = strconv.FormatUint(uint64(n), 16)
186 }
187 if best < 0 {
188 return strings.Join(groups, ":")
189 }
190 return strings.Join(groups[:best], ":") + "::" + strings.Join(groups[best+length:], ":")
191}
192func parseTTL(s string) (int, error) {
193 invalid := fmt.Errorf("bad ttl %s", s)
194 if integerWord(s) {
195 n := number(s)
196 if n >= 1 && n <= 604800 {
197 return n, nil
198 }
199 return 0, invalid
200 }
201 units := "wdhms"
202 multipliers := []int{604800, 86400, 3600, 60, 1}
203 total, previous, pos := 0, -1, 0
204 for pos < len(s) {
205 start := pos
206 for pos < len(s) && s[pos] >= '0' && s[pos] <= '9' {
207 pos++
208 }
209 digits := s[start:pos]
210 if len(digits) > 3 || !integerWord(digits) || pos == len(s) {
211 return 0, invalid
212 }
213 unit := strings.IndexByte(units, strings.ToLower(s[pos : pos+1])[0])
214 if unit <= previous {
215 return 0, invalid
216 }
217 total += number(digits) * multipliers[unit]
218 previous = unit
219 pos++
220 }
221 if total < 1 || total > 604800 {
222 return 0, invalid
223 }
224 return total, nil
225}

236func (r record) String() string {
237 return fmt.Sprintf("%s %d %s %s", r.owner, r.ttl, r.kind, r.data)

239func dataCompare(a, b record) int {
240 switch a.kind {
241 case "A":
242 if a.address < b.address {
243 return -1
244 }
245 if a.address > b.address {
246 return 1
247 }
248 case "AAAA":
249 for i := 0; i < 8; i++ {
250 if a.address6[i] < b.address6[i] {
251 return -1
252 }
253 if a.address6[i] > b.address6[i] {
254 return 1
255 }
256 }
257 case "NS", "CNAME", "PTR":
258 return nameCompare(a.target, b.target)
259 case "MX":
260 if a.preference < b.preference {
261 return -1
262 }
263 if a.preference > b.preference {
264 return 1
265 }
266 return nameCompare(a.target, b.target)
267 default:
268 return strings.Compare(a.data, b.data)
269 }
270 return 0
271}
272func sortedRecords(records []record) []record {
273 result := append([]record{}, records...)
274 sort.Slice(result, func(i, j int) bool {
275 a, b := result[i], result[j]
276 if c := nameCompare(a.owner, b.owner); c != 0 {
277 return c < 0
278 }
279 if a.kind != b.kind {
280 return indexOf(types, a.kind) < indexOf(types, b.kind)
281 }
282 return dataCompare(a, b) < 0
283 })
284 return result
285}

363func (r *resolver) zoneLine(words []string, line int) error {
364 if words[0] == "ROOT" {
365 pending := []uint32{}
366 for _, s := range words[1:] {
367 a, err := parseIPv4(s)
368 if err != nil {
369 return err
370 }
371 for _, old := range append(append([]uint32{}, r.roots...), pending...) {
372 if old == a {
373 return fmt.Errorf("duplicate root %s", s)
374 }
375 }
376 pending = append(pending, a)
377 }
378 r.roots = append(r.roots, pending...)
379 return nil
380 }
381 if words[0] == "ZONE" {
382 r.current = nil
383 origin, err := parseName(words[1], "", false)
384 if err != nil {
385 return err
386 }
387 server, err := parseIPv4(words[2])
388 if err != nil {
389 return err
390 }
391 for _, z := range r.zones {
392 if z.origin == origin && z.server == server {
393 return fmt.Errorf("duplicate zone %s at %s", origin, words[2])
394 }
395 }
396 z := &zone{origin: origin, server: server, line: line}
397 r.zones = append(r.zones, z)
398 r.current = z
399 return nil
400 }
401 z := r.current
402 if z == nil {
403 return fmt.Errorf("no current zone")
404 }
405 owner, err := parseName(words[0], z.origin, true)
406 if err != nil {
407 return err
408 }
409 if !below(owner, z.origin) {
410 return fmt.Errorf("out of zone %s", owner)
411 }
412 ttl, err := parseTTL(words[1])
413 if err != nil {
414 return err
415 }
416 kind, fields := words[2], words[3:]
417 if indexOf(types, kind) < 0 {
418 return fmt.Errorf("bad type %s", kind)
419 }
420 required := 1
421 if kind == "MX" {
422 required = 2
423 }
424 if kind == "SOA" {
425 required = 7
426 }
427 if kind != "TXT" && len(fields) != required {
428 return fmt.Errorf("wrong field count")
429 }
430 rec := record{owner: owner, ttl: ttl, kind: kind}
431 switch kind {
432 case "A":
433 rec.address, err = parseIPv4(fields[0])
434 rec.data = ipv4(rec.address)
435 case "AAAA":
436 rec.address6, err = parseIPv6(fields[0])
437 rec.data = ipv6(rec.address6)
438 case "NS", "CNAME", "PTR":
439 rec.target, err = parseName(fields[0], z.origin, false)
440 rec.data = rec.target
441 case "MX":
442 if !integerWord(fields[0]) || number(fields[0]) > 65535 {
443 return fmt.Errorf("bad preference %s", fields[0])
444 }
445 rec.preference = number(fields[0])
446 rec.target, err = parseName(fields[1], z.origin, false)
447 rec.data = fmt.Sprintf("%d %s", rec.preference, rec.target)
448 case "TXT":
449 rec.data = strings.Join(fields, " ")
450 if len(rec.data) < 2 || rec.data[0] != '"' || rec.data[len(rec.data)-1] != '"' || strings.Contains(rec.data[1:len(rec.data)-1], "\"") {
451 return fmt.Errorf("bad text %s", rec.data)
452 }
453 case "SOA":
454 data := append([]string{}, fields...)
455 data[0], err = parseName(fields[0], z.origin, false)
456 if err != nil {
457 return err
458 }
459 data[1], err = parseName(fields[1], z.origin, false)
460 if err != nil {
461 return err
462 }
463 for i := 2; i < 6; i++ {
464 if !integerWord(fields[i]) {
465 return fmt.Errorf("bad number %s", fields[i])
466 }
467 }
468 if !integerWord(fields[6]) || number(fields[6]) < 1 || number(fields[6]) > 86400 {
469 return fmt.Errorf("bad minimum %s", fields[6])
470 }
471 rec.minimum = number(fields[6])
472 rec.data = strings.Join(data, " ")
473 }
474 if err != nil {
475 return err
476 }
477 if kind == "NS" && strings.HasPrefix(owner, "*.") {
478 return fmt.Errorf("ns at wildcard")
479 }
480 if kind == "SOA" && owner != z.origin {
481 return fmt.Errorf("soa not at origin")
482 }
483 if kind == "SOA" {
484 for _, old := range z.records {
485 if old.kind == "SOA" {
486 return fmt.Errorf("duplicate soa")
487 }
488 }
489 }
490 for _, old := range z.records {
491 if old.owner == owner && old.kind == kind && old.data == rec.data {
492 return fmt.Errorf("duplicate record")
493 }
494 }
495 for _, old := range z.records {
496 if old.owner == owner && (kind == "CNAME" || old.kind == "CNAME") {
497 return fmt.Errorf("cname conflict %s", owner)
498 }
499 }
500 z.records = append(z.records, rec)
501 return nil
502}
503func (r *resolver) finishZones() {
504 kept := []*zone{}
505 count := 0
506 for _, z := range r.zones {
507 if z.soa().kind == "" {
508 fmt.Fprintf(r.out, "error line %d: zone %s at %s has no soa\n", z.line, z.origin, ipv4(z.server))
509 } else {
510 kept = append(kept, z)
511 count += len(z.records)
512 fmt.Fprintf(r.out, "zone %s at %s records=%d\n", z.origin, ipv4(z.server), len(z.records))
513 }
514 }
515 r.zones = kept
516 r.current = nil
517 fmt.Fprintf(r.out, "loaded roots=%d zones=%d records=%d\n", len(r.roots), len(r.zones), count)
518}
519func (r *resolver) known(server uint32) bool {
520 for _, z := range r.zones {
521 if z.server == server {
522 return true
523 }
524 }
525 return false
526}
527func (r *resolver) authoritative(server uint32, name, kind, cut string) response {
528 if r.down[server] || !r.known(server) {
529 return response{kind: "TIMEOUT"}
530 }
531 var z *zone
532 for _, candidate := range r.zones {
533 if candidate.server == server && below(name, candidate.origin) && (z == nil || nameLength(candidate.origin) > nameLength(z.origin)) {
534 z = candidate
535 }
536 }
537 if z == nil || nameLength(z.origin) < nameLength(cut) {
538 return response{kind: "LAME"}
539 }
540 chain := ancestors(name)
541 for i := len(chain) - 1; i >= 0; i-- {
542 d := chain[i]
543 if d == z.origin || !below(d, z.origin) {
544 continue
545 }
546 ns := z.recordsAt(d, "NS")
547 if len(ns) > 0 {
548 additional := []record{}
549 for _, n := range ns {
550 additional = append(additional, z.recordsAt(n.target, "A")...)
551 }
552 return response{kind: "REFERRAL", referral: d, authority: ns, additional: additional}
553 }
554 }
555 records := z.recordsAt(name, "")
556 if len(records) == 0 {
557 if z.exists(name) {
558 return response{kind: "NODATA", authority: []record{z.soa()}}
559 }
560 wildcard := ""
561 for _, ancestor := range chain[1:] {
562 if z.exists(ancestor) {
563 wildcard = "*."
564 if ancestor != "." {
565 wildcard += ancestor
566 }
567 break
568 }
569 }
570 records = z.recordsAt(wildcard, "")
571 if len(records) == 0 {
572 return response{kind: "NXDOMAIN", authority: []record{z.soa()}}
573 }
574 }
575 answers := []record{}
576 for _, rec := range records {
577 if rec.kind == kind {
578 rec.owner = name
579 answers = append(answers, rec)
580 }
581 }
582 if len(answers) > 0 {
583 return response{kind: "ANSWER", answer: sortedRecords(answers)}
584 }
585 if kind != "CNAME" {
586 for _, rec := range records {
587 if rec.kind == "CNAME" {
588 rec.owner = name
589 return response{kind: "CNAME", answer: []record{rec}}
590 }
591 }
592 }
593 return response{kind: "NODATA", authority: []record{z.soa()}}
594}
595func (r *resolver) ask(server uint32, name, kind, cut string) response {
596 answer := r.authoritative(server, name, kind, cut)
597 if r.stats[server] == nil {
598 r.stats[server] = &[7]int{}
599 }
600 r.stats[server][indexOf(responseKinds, answer.kind)]++
601 return answer
602}
603func (r *resolver) trace(depth int, format string, args ...any) {
604 fmt.Fprint(r.out, strings.Repeat(" ", 2*(depth+1)))
605 fmt.Fprintf(r.out, format+"\n", args...)
606}
607func (r *resolver) store(records []record) int {
608 ttl := records[0].ttl
609 for _, rec := range records {
610 if rec.ttl < ttl {
611 ttl = rec.ttl
612 }
613 }
614 if ttl < r.settings["MINTTL"] {
615 ttl = r.settings["MINTTL"]
616 }
617 if ttl > r.settings["MAXTTL"] {
618 ttl = r.settings["MAXTTL"]
619 }
620 first := records[0]
621 r.cache[cacheKey{first.owner, first.kind}] = entry{kind: "POS", records: sortedRecords(records), expiry: r.clock + ttl}
622 return ttl
623}
624func withTTL(records []record, ttl int) []record {
625 out := append([]record{}, records...)
626 for i := range out {
627 out[i].ttl = ttl
628 }
629 return out
630}
631func (r *resolver) queryServers(name, kind, cut string, names []string, roots bool, depth int, resolving []cacheKey) *response {
632 query := func(a uint32) *response {
633 response := r.ask(a, name, kind, cut)
634 r.trace(depth, "ASK %s %s %s -> %s", ipv4(a), name, kind, response.description())
635 if response.kind == "LAME" || response.kind == "TIMEOUT" {
636 return nil
637 }
638 return &response
639 }
640 if roots {
641 for _, a := range r.roots {
642 if answer := query(a); answer != nil {
643 return answer
644 }
645 }
646 return nil
647 }
648 names = append([]string{}, names...)
649 sort.Slice(names, func(i, j int) bool {
650 return nameCompare(names[i], names[j]) < 0
651 })
652 for _, host := range names {
653 addresses := map[uint32]bool{}
654 key := cacheKey{host, "A"}
655 if cached, ok := r.cache[key]; ok && cached.kind == "POS" {
656 for _, rec := range cached.records {
657 addresses[rec.address] = true
658 }
659 } else {
660 loop := false
661 for _, pair := range resolving {
662 if pair == key {
663 loop = true
664 break
665 }
666 }
667 if loop {
668 r.trace(depth, "GLUELESS %s -> LOOP", host)
669 } else {
670 r.trace(depth, "GLUELESS %s", host)
671 answer := r.resolve(host, "A", depth+1, resolving)
672 if answer.status == "NOERROR" {
673 for _, rec := range answer.answer {
674 if rec.kind == "A" {
675 addresses[rec.address] = true
676 }
677 }
678 }
679 }
680 }
681 ordered := []uint32{}
682 for a := range addresses {
683 ordered = append(ordered, a)
684 }
685 sort.Slice(ordered, func(i, j int) bool {
686 return ordered[i] < ordered[j]
687 })
688 for _, a := range ordered {
689 if answer := query(a); answer != nil {
690 return answer
691 }
692 }
693 }
694 return nil
695}
696func (r *resolver) lookup(name, kind string, depth int, resolving []cacheKey) response {
697 key := cacheKey{name, kind}
698 if cached, ok := r.cache[key]; ok {
699 r.hits++
700 result := response{kind: cached.kind}
701 if cached.kind == "POS" {
702 result.kind = "ANSWER"
703 result.answer = withTTL(cached.records, cached.expiry-r.clock)
704 }
705 r.trace(depth, "CACHE HIT %s %s -> %s", name, kind, result.kind)
706 return result
707 }
708 for k, cached := range r.cache {
709 if cached.kind == "NXDOMAIN" && below(name, k.name) {
710 r.hits++
711 r.trace(depth, "CACHE HIT %s %s -> NXDOMAIN", name, kind)
712 return response{kind: "NXDOMAIN"}
713 }
714 }
715 if kind != "CNAME" {
716 if cached, ok := r.cache[cacheKey{name, "CNAME"}]; ok && cached.kind == "POS" {
717 r.hits++
718 records := withTTL(cached.records, cached.expiry-r.clock)
719 r.trace(depth, "CACHE HIT %s %s -> CNAME %s", name, kind, records[0].target)
720 return response{kind: "CNAME", answer: records}
721 }
722 }
723 cut, names, roots := ".", []string{}, true
724 for _, ancestor := range ancestors(name) {
725 if cached, ok := r.cache[cacheKey{ancestor, "NS"}]; ok && cached.kind == "POS" {
726 cut = ancestor
727 roots = false
728 for _, rec := range cached.records {
729 names = append(names, rec.target)
730 }
731 r.hits++
732 r.trace(depth, "CACHE HIT %s %s -> REFERRAL %s", name, kind, cut)
733 break
734 }
735 }
736 for {
737 obtained := r.queryServers(name, kind, cut, names, roots, depth, resolving)
738 if obtained == nil {
739 return response{kind: "SERVFAIL no server for " + cut}
740 }
741 answer := *obtained
742 if answer.kind == "REFERRAL" {
743 r.store(answer.authority)
744 grouped := map[string][]record{}
745 for _, rec := range answer.additional {
746 grouped[rec.owner] = append(grouped[rec.owner], rec)
747 }
748 for _, records := range grouped {
749 r.store(records)
750 }
751 cut, names, roots = answer.referral, nil, false
752 for _, rec := range answer.authority {
753 names = append(names, rec.target)
754 }
755 } else if answer.kind == "ANSWER" || answer.kind == "CNAME" {
756 answer.answer = withTTL(answer.answer, r.store(answer.answer))
757 return answer
758 } else {
759 soa := answer.authority[0]
760 ttl := soa.ttl
761 if soa.minimum < ttl {
762 ttl = soa.minimum
763 }
764 if r.settings["NEGTTL"] < ttl {
765 ttl = r.settings["NEGTTL"]
766 }
767 r.cache[key] = entry{kind: answer.kind, expiry: r.clock + ttl}
768 return answer
769 }
770 }
771}
772func (r *resolver) resolve(name, kind string, depth int, resolving []cacheKey) result {
773 resolving = append(append([]cacheKey{}, resolving...), cacheKey{name, kind})
774 chain := []record{}
775 seen := map[string]bool{name: true}
776 current := name
777 var result result
778 for {
779 lookup := r.lookup(current, kind, depth, resolving)
780 if lookup.kind == "CNAME" {
781 rec := lookup.answer[0]
782 chain = append(chain, rec)
783 if seen[rec.target] {
784 result.status = "SERVFAIL cname loop"
785 break
786 }
787 if len(chain) == 9 {
788 result.status = "SERVFAIL cname chain too long"
789 break
790 }
791 seen[rec.target] = true
792 current = rec.target
793 } else if lookup.kind == "ANSWER" {
794 result = resultWith("NOERROR", append(chain, lookup.answer...))
795 break
796 } else if lookup.kind == "NODATA" || lookup.kind == "NXDOMAIN" {
797 result = resultWith(lookup.kind, chain)
798 break
799 } else {
800 result.status = lookup.kind
801 break
802 }
803 }
804 statusIndex := indexOf([]string{"NOERROR", "NODATA", "NXDOMAIN"}, result.status)
805 if statusIndex < 0 {
806 statusIndex = 3
807 result.answer = nil
808 }
809 r.resolutions[statusIndex]++
810 return result
811}
812func resultWith(status string, records []record) result {
813 return result{status, records}
814}
815func (r *resolver) emitResult(answer result) {
816 for _, rec := range answer.answer {
817 fmt.Fprintln(r.out, "answer "+rec.String())
818 }
819 fmt.Fprintln(r.out, "status "+answer.status)
820}
821func (r *resolver) command(words []string) error {
822 command := words[0]
823 switch command {
824 case "RESOLVE", "MX":
825 name, err := parseName(words[1], "", true)
826 if err != nil {
827 return err
828 }
829 kind := "MX"
830 if command == "RESOLVE" {
831 kind = words[2]
832 if indexOf(types, kind) < 0 {
833 return fmt.Errorf("bad type %s", kind)
834 }
835 fmt.Fprintf(r.out, "resolve %s %s\n", name, kind)
836 } else {
837 fmt.Fprintln(r.out, "mx "+name)
838 }
839 answer := r.resolve(name, kind, 0, nil)
840 r.emitResult(answer)
841 if command == "MX" {
842 addressResolution := func(host, prefix string) {
843 a := r.resolve(host, "A", 0, nil)
844 aaaa := r.resolve(host, "AAAA", 0, nil)
845 addresses := []string{}
846 for _, result := range []result{a, aaaa} {
847 if result.status == "NOERROR" {
848 for _, rec := range result.answer {
849 if rec.kind == "A" || rec.kind == "AAAA" {
850 addresses = append(addresses, rec.data)
851 }
852 }
853 }
854 }
855 text := "no address"
856 if len(addresses) > 0 {
857 text = strings.Join(addresses, " ")
858 }
859 fmt.Fprintln(r.out, prefix+" -> "+text)
860 }
861 if answer.status == "NOERROR" {
862 for _, rec := range answer.answer {
863 if rec.kind == "MX" {
864 prefix := fmt.Sprintf("exchange %d %s", rec.preference, rec.target)
865 if rec.target == "." {
866 fmt.Fprintln(r.out, prefix+" -> null")
867 } else {
868 addressResolution(rec.target, prefix)
869 }
870 }
871 }
872 } else if answer.status == "NODATA" {
873 host := name
874 for _, rec := range answer.answer {
875 if rec.kind == "CNAME" {
876 host = rec.target
877 }
878 }
879 addressResolution(host, "implicit "+host)
880 }
881 }
882 case "REVERSE":
883 text, name := "", ""
884 if strings.Contains(words[1], ":") {
885 a, err := parseIPv6(words[1])
886 if err != nil {
887 return err
888 }
889 text = ipv6(a)
890 digits := ""
891 for _, n := range a {
892 digits += fmt.Sprintf("%04x", n)
893 }
894 labels := []string{}
895 for i := len(digits) - 1; i >= 0; i-- {
896 labels = append(labels, digits[i:i+1])
897 }
898 name = strings.Join(labels, ".") + ".ip6.arpa."
899 } else {
900 a, err := parseIPv4(words[1])
901 if err != nil {
902 return err
903 }
904 text = ipv4(a)
905 fields := strings.Split(text, ".")
906 name = fields[3] + "." + fields[2] + "." + fields[1] + "." + fields[0] + ".in-addr.arpa."
907 }
908 fmt.Fprintf(r.out, "reverse %s %s\n", text, name)
909 r.emitResult(r.resolve(name, "PTR", 0, nil))
910 case "QUERY", "DUMP":
911 server, err := parseIPv4(words[1])
912 if err != nil {
913 return err
914 }
915 name, err := parseName(words[2], "", command == "QUERY")
916 if err != nil {
917 return err
918 }
919 if command == "QUERY" {
920 kind := words[3]
921 if indexOf(types, kind) < 0 {
922 return fmt.Errorf("bad type %s", kind)
923 }
924 answer := r.ask(server, name, kind, ".")
925 fmt.Fprintf(r.out, "query %s %s %s -> %s\n", ipv4(server), name, kind, answer.description())
926 for _, group := range []struct {
927 label string
928 records []record
929 }{{"answer", answer.answer}, {"authority", answer.authority}, {"additional", answer.additional}} {
930 for _, rec := range group.records {
931 fmt.Fprintln(r.out, group.label+" "+rec.String())
932 }
933 }
934 } else {
935 if r.down[server] || !r.known(server) {
936 fmt.Fprintf(r.out, "dump %s %s -> TIMEOUT\n", ipv4(server), name)
937 return nil
938 }
939 for _, z := range r.zones {
940 if z.server == server && z.origin == name {
941 fmt.Fprintf(r.out, "dump %s %s records=%d\n", ipv4(server), name, len(z.records))
942 for _, rec := range sortedRecords(z.records) {
943 fmt.Fprintln(r.out, " "+rec.String())
944 }
945 return nil
946 }
947 }
948 fmt.Fprintf(r.out, "dump %s %s -> REFUSED\n", ipv4(server), name)
949 }
950 case "DOWN", "UP":
951 server, err := parseIPv4(words[1])
952 if err != nil {
953 return err
954 }
955 if command == "DOWN" {
956 if r.down[server] {
957 return fmt.Errorf("server %s already down", words[1])
958 }
959 r.down[server] = true
960 fmt.Fprintln(r.out, "down "+ipv4(server))
961 } else {
962 if !r.down[server] {
963 return fmt.Errorf("server %s not down", words[1])
964 }
965 delete(r.down, server)
966 fmt.Fprintln(r.out, "up "+ipv4(server))
967 }
968 case "SET":
969 key := words[1]
970 if _, ok := r.settings[key]; !ok {
971 return fmt.Errorf("bad setting %s", key)
972 }
973 if !integerWord(words[2]) || number(words[2]) < 1 || number(words[2]) > 604800 {
974 return fmt.Errorf("bad value %s", words[2])
975 }
976 n := number(words[2])
977 minimum, maximum := r.settings["MINTTL"], r.settings["MAXTTL"]
978 if key == "MINTTL" {
979 minimum = n
980 }
981 if key == "MAXTTL" {
982 maximum = n
983 }
984 if minimum > maximum {
985 return fmt.Errorf("minttl above maxttl")
986 }
987 r.settings[key] = n
988 fmt.Fprintf(r.out, "set %s=%d\n", key, n)
989 case "TIME":
990 if !integerWord(words[1]) || number(words[1]) < 1 || number(words[1]) > 1000000 {
991 return fmt.Errorf("bad time %s", words[1])
992 }
993 r.clock += number(words[1])
994 expired := 0
995 for key, cached := range r.cache {
996 if cached.expiry <= r.clock {
997 delete(r.cache, key)
998 expired++
999 }
1000 }
1001 fmt.Fprintf(r.out, "time now=%d expired=%d\n", r.clock, expired)
1002 case "CACHE", "FLUSH":
1003 ancestor := "."
1004 if len(words) == 2 {
1005 var err error
1006 ancestor, err = parseName(words[1], "", false)
1007 if err != nil {
1008 return err
1009 }
1010 }
1011 keys := []cacheKey{}
1012 for key := range r.cache {
1013 if below(key.name, ancestor) {
1014 keys = append(keys, key)
1015 }
1016 }
1017 sort.Slice(keys, func(i, j int) bool {
1018 a, b := keys[i], keys[j]
1019 if c := nameCompare(a.name, b.name); c != 0 {
1020 return c < 0
1021 }
1022 return indexOf(types, a.kind) < indexOf(types, b.kind)
1023 })
1024 if command == "FLUSH" {
1025 for _, key := range keys {
1026 delete(r.cache, key)
1027 }
1028 fmt.Fprintf(r.out, "flush removed=%d\n", len(keys))
1029 } else {
1030 fmt.Fprintf(r.out, "cache entries=%d\n", len(keys))
1031 for _, key := range keys {
1032 cached := r.cache[key]
1033 if cached.kind == "POS" {
1034 for _, rec := range withTTL(cached.records, cached.expiry-r.clock) {
1035 fmt.Fprintln(r.out, " "+rec.String())
1036 }
1037 } else {
1038 fmt.Fprintf(r.out, " %s %d %s %s\n", key.name, cached.expiry-r.clock, key.kind, cached.kind)
1039 }
1040 }
1041 }
1042 case "STATS":
1043 servers, total := []uint32{}, 0
1044 for server, stats := range r.stats {
1045 servers = append(servers, server)
1046 for _, n := range stats {
1047 total += n
1048 }
1049 }
1050 sort.Slice(servers, func(i, j int) bool {
1051 return servers[i] < servers[j]
1052 })
1053 fmt.Fprintf(r.out, "stats queries=%d hits=%d\n", total, r.hits)
1054 for _, server := range servers {
1055 stats := r.stats[server]
1056 sent := 0
1057 for _, n := range stats {
1058 sent += n
1059 }
1060 fmt.Fprintf(r.out, " %s sent=%d", ipv4(server), sent)
1061 for i, kind := range responseKinds {
1062 fmt.Fprintf(r.out, " %s=%d", strings.ToLower(kind), stats[i])
1063 }
1064 fmt.Fprintln(r.out)
1065 }
1066 fmt.Fprintf(r.out, "resolutions noerror=%d nodata=%d nxdomain=%d servfail=%d\n", r.resolutions[0], r.resolutions[1], r.resolutions[2], r.resolutions[3])
1067 }
1068 return nil
1069}
1070func main() {
1071 data, _ := io.ReadAll(os.Stdin)
1072 lines := strings.Split(string(data), "\n")
1073 out := bufio.NewWriter(os.Stdout)
1074 defer out.Flush()
1075 r := newResolver(out)
1076 position := 0
1077 for lines[position] != "END ZONES" {
1078 if err := r.zoneLine(strings.Split(lines[position], " "), position+1); err != nil {
1079 fmt.Fprintf(out, "error line %d: %s\n", position+1, err)
1080 }
1081 position++
1082 }
1083 r.finishZones()
1084 position++
1085 for lines[position] != "END" {
1086 if err := r.command(strings.Split(lines[position], " ")); err != nil {
1087 fmt.Fprintf(out, "error line %d: %s\n", position+1, err)
1088 }
1089 position++
1090 }
1091}
