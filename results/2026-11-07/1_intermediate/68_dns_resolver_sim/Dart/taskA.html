98List<String> labels(String name) =>
99 name == '.' ? [] : name.substring(0, name.length - 1).split('.');

103 final parts = labels(name);

105 ...List.generate(parts.length, (i) => '${parts.sublist(i).join('.')}.'),

110int nameOrder(String a, String b) {
111 final left = labels(a).reversed.toList();
112 final right = labels(b).reversed.toList();
113 final length = left.length < right.length ? left.length : right.length;
114 for (var i = 0; i < length; i++) {
115 final comparison = left[i].compareTo(right[i]);
116 if (comparison != 0) return comparison;
117 }
118 return left.length.compareTo(right.length);
119}

121String parseName(String word, {bool wildcard = false, String? origin}) {
122 if (origin != null && word == '@') return origin;
123 if (word == '.') return '.';
124 final absolute = word.endsWith('.');
125 final parts = (absolute ? word.substring(0, word.length - 1) : word).split(
126 '.',
127 );
128 for (var i = 0; i < parts.length; i++) {
129 if (!(wildcard && i == 0 && parts[i] == '*') &&
130 !RegExp(r'^[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?$')
131 .hasMatch(parts[i]))
132 throw DnsFailure('bad name $word');
133 }
134 final name = '${parts.join('.').toLowerCase()}.';
135 return origin != null && !absolute && origin != '.' ? name + origin : name;
136}

138int ipv4(String word) {
139 final parts = word.split('.');
140 if (parts.length != 4 ||
141 parts.any(
142 (part) =>
143 !RegExp(r'^(?:0|[1-9][0-9]{0,2})$').hasMatch(part) ||
144 int.parse(part) > 255,
145 ))
146 throw DnsFailure('bad address $word');
147 return parts.fold<int>(0, (value, part) => value * 256 + int.parse(part));
148}
149
150String addressText(int value) =>
151 [24, 16, 8, 0].map((shift) => (value >> shift) & 255).join('.');
152List<int> ipv6(String word) {
153 final halves = word.split('::');
154 if (halves.length > 2) throw DnsFailure('bad address $word');
155 final left = halves[0].isEmpty ? <String>[] : halves[0].split(':');
156 final right = halves.length == 2 && halves[1].isNotEmpty
157 ? halves[1].split(':')
158 : <String>[];
159 if ([
160 ...left,
161 ...right,
162 ].any((group) => !RegExp(r'^[0-9A-Fa-f]{1,4}$').hasMatch(group)) ||
163 (halves.length == 1 ? left.length != 8 : left.length + right.length > 7))
164 throw DnsFailure('bad address $word');
165 return [
166 ...left.map((group) => int.parse(group, radix: 16)),
167 ...List.filled(halves.length == 2 ? 8 - left.length - right.length : 0, 0),
168 ...right.map((group) => int.parse(group, radix: 16)),
169 ];
170}

172String ipv6Text(List<int> groups) {
173 var best = -1;
174 var length = 0;
175 for (var start = 0; start < 8;) {
176 if (groups[start] != 0) {
177 start++;
178 continue;
179 }
180 var end = start;
181 while (end < 8 && groups[end] == 0) end++;
182 if (end - start >= 2 && end - start > length) {
183 best = start;
184 length = end - start;
185 }
186 start = end;
187 }
188 final words = groups.map((group) => group.toRadixString(16)).toList();
189 return best < 0
190 ? words.join(':')
191 : '${words.sublist(0, best).join(':')}::${words.sublist(best + length).join(':')}';
192}

194int ttl(String word) {
195 var result = 0;
196 if (integerWord.hasMatch(word)) {
197 result = int.parse(word);
198 } else {
199 const units = 'wdhms';
200 const seconds = [604800, 86400, 3600, 60, 1];
201 var position = 0;
202 var previous = -1;
203 for (final match in RegExp(
204 r'(0|[1-9][0-9]{0,2})([wdhms])',
205 caseSensitive: false,
206 ).allMatches(word)) {
207 final unit = units.indexOf(match[2]!.toLowerCase());
208 if (match.start != position || unit <= previous)
209 throw DnsFailure('bad ttl $word');
210 result += int.parse(match[1]!) * seconds[unit];
211 previous = unit;
212 position += match[0]!.length;
213 }
214 if (position != word.length || position == 0)
215 throw DnsFailure('bad ttl $word');
216 }
217 if (result < 1 || result > 604800) throw DnsFailure('bad ttl $word');
218 return result;
219}

225int dataOrder(DnsRecord a, DnsRecord b) {
226 if (a.type == 'A') return a.address!.compareTo(b.address!);
227 if (a.type == 'AAAA') {
228 for (var i = 0; i < 8; i++) {
229 if (a.groups![i] != b.groups![i])
230 return a.groups![i].compareTo(b.groups![i]);
231 }
232 return 0;
233 }
234 if (['NS', 'CNAME', 'PTR'].contains(a.type))
235 return nameOrder(a.data[0], b.data[0]);
236 if (a.type == 'MX') {
237 final preference = a.preference!.compareTo(b.preference!);
238 return preference == 0 ? nameOrder(a.data[1], b.data[1]) : preference;
239 }
240 return a.data.join(' ').compareTo(b.data.join(' '));
241}

243int recordOrder(DnsRecord a, DnsRecord b) {
244 final name = nameOrder(a.owner, b.owner);
245 if (name != 0) return name;
246 final type = types.indexOf(a.type).compareTo(types.indexOf(b.type));
247 return type == 0 ? dataOrder(a, b) : type;
248}

281 void zoneLine(List<String> words, int line) {
282 if (words[0] == 'ROOT') {
283 final addresses = <int>[];
284 for (final word in words.skip(1)) {
285 final address = ipv4(word);
286 if (roots.contains(address) || addresses.contains(address))
287 throw DnsFailure('duplicate root $word');
288 addresses.add(address);
289 }
290 roots.addAll(addresses);
291 return;
292 }
293 if (words[0] == 'ZONE') {
294 current = null;
295 final origin = parseName(words[1]);
296 final server = ipv4(words[2]);
297 if (zones.any((zone) => zone.origin == origin && zone.server == server))
298 throw DnsFailure('duplicate zone $origin at ${words[2]}');
299 final zone = Zone(origin, server, line);
300 current = zone;
301 zones.add(zone);
302 return;
303 }
304 final zone = current;
305 if (zone == null) throw DnsFailure('no current zone');
306 final owner = parseName(words[0], wildcard: true, origin: zone.origin);
307 if (!below(owner, zone.origin)) throw DnsFailure('out of zone $owner');
308 final seconds = ttl(words[1]);
309 final type = words[2];
310 checkType(type);
311 final fields = words.sublist(3);
312 final required = type == 'MX'
313 ? 2
314 : type == 'SOA'
315 ? 7
316 : 1;
317 if (type != 'TXT' && fields.length != required)
318 throw DnsFailure('wrong field count');
319 final record = DnsRecord(owner, seconds, type, []);
320 String name(String word) => parseName(word, origin: zone.origin);
321 if (type == 'A') {
322 record.address = ipv4(fields[0]);
323 record.data = [addressText(record.address!)];
324 } else if (type == 'AAAA') {
325 record.groups = ipv6(fields[0]);
326 record.data = [ipv6Text(record.groups!)];
327 } else if (['NS', 'CNAME', 'PTR'].contains(type)) {
328 record.data = [name(fields[0])];
329 } else if (type == 'MX') {
330 if (!integerWord.hasMatch(fields[0]) || int.parse(fields[0]) > 65535)
331 throw DnsFailure('bad preference ${fields[0]}');
332 record.preference = int.parse(fields[0]);
333 record.data = ['${record.preference}', name(fields[1])];
334 } else if (type == 'TXT') {
335 final text = fields.join(' ');
336 if (text.length < 2 ||
337 !text.startsWith('"') ||
338 !text.endsWith('"') ||
339 text.substring(1, text.length - 1).contains('"'))
340 throw DnsFailure('bad text $text');
341 record.data = [text];
342 } else {
343 record.data = [name(fields[0]), name(fields[1])];
344 for (final field in fields.sublist(2, 6)) {
345 if (!integerWord.hasMatch(field)) throw DnsFailure('bad number $field');
346 record.data.add(int.parse(field).toString());
347 }
348 if (!integerWord.hasMatch(fields[6]) ||
349 int.parse(fields[6]) < 1 ||
350 int.parse(fields[6]) > 86400)
351 throw DnsFailure('bad minimum ${fields[6]}');
352 record.minimum = int.parse(fields[6]);
353 record.data.add('${record.minimum}');
354 }
355 if (type == 'NS' && owner.startsWith('*.'))
356 throw DnsFailure('ns at wildcard');
357 if (type == 'SOA' && owner != zone.origin)
358 throw DnsFailure('soa not at origin');
359 if (type == 'SOA' && zone.records.any((record) => record.type == 'SOA'))
360 throw DnsFailure('duplicate soa');
361 if (zone.records.any(
362 (previous) =>
363 previous.owner == owner &&
364 previous.type == type &&
365 jsonEncode(previous.data) == jsonEncode(record.data),
366 ))
367 throw DnsFailure('duplicate record');
368 if (zone.records.any(
369 (previous) =>
370 previous.owner == owner &&
371 (type == 'CNAME' || previous.type == 'CNAME'),
372 ))
373 throw DnsFailure('cname conflict $owner');
374 zone.records.add(record);
375 }

397 Reply authoritative(int server, String name, String type, String cut) {
398 if (down.contains(server) || !zones.any((zone) => zone.server == server))
399 return Reply('TIMEOUT');
400 final candidates =
401 zones
402 .where((zone) => zone.server == server && below(name, zone.origin))
403 .toList()
404 ..sort(
405 (a, b) =>
406 labels(b.origin).length.compareTo(labels(a.origin).length),
407 );
408 if (candidates.isEmpty ||
409 labels(candidates[0].origin).length < labels(cut).length)
410 return Reply('LAME');
411 final zone = candidates[0];
412 final soa = zone.records.firstWhere((record) => record.type == 'SOA');
413 final delegation = find(
414 ancestors(name).reversed,
415 (ancestor) =>
416 ancestor != zone.origin &&
417 below(ancestor, zone.origin) &&
418 zone.records.any(
419 (record) => record.owner == ancestor && record.type == 'NS',
420 ),
421 );
422 if (delegation != null) {
423 final authority =
424 zone.records
425 .where(
426 (record) => record.owner == delegation && record.type == 'NS',
427 )
428 .toList()
429 ..sort(dataOrder);
430 final additional = <DnsRecord>[];
431 for (final record in authority) {
432 additional.addAll(
433 zone.records
434 .where((glue) => glue.owner == record.data[0] && glue.type == 'A')
435 .toList()
436 ..sort(dataOrder),
437 );
438 }
439 return Reply(
440 'REFERRAL',
441 authority: authority,
442 additional: additional,
443 cut: delegation,
444 );
445 }
446 bool exists(String candidate) =>
447 zone.records.any((record) => below(record.owner, candidate));
448 var records = zone.records.where((record) => record.owner == name).toList();
449 if (records.isEmpty) {
450 if (exists(name)) return Reply('NODATA', authority: [soa]);
451 final ancestor = ancestors(name).skip(1).firstWhere(exists);
452 final wildcard = ancestor == '.' ? '*.' : '*.$ancestor';
453 records = zone.records
454 .where((record) => record.owner == wildcard)
455 .toList();
456 if (records.isEmpty) return Reply('NXDOMAIN', authority: [soa]);
457 }
458 final answers = records.where((record) => record.type == type).toList()
459 ..sort(dataOrder);
460 if (answers.isNotEmpty)
461 return Reply(
462 'ANSWER',
463 answer: answers.map((record) => record.copy(owner: name)).toList(),
464 );
465 final cname = find(records, (record) => record.type == 'CNAME');
466 if (type != 'CNAME' && cname != null)
467 return Reply('CNAME', answer: [cname.copy(owner: name)]);
468 return Reply('NODATA', authority: [soa]);
469 }

472 final response = authoritative(server, name, type, cut);
473 final counts = queries.putIfAbsent(
474 server,
475 () => List.filled(kinds.length, 0),
476 );
477 counts[kinds.indexOf(response.kind)]++;
478 return response;

481 List<DnsRecord> positive(List<DnsRecord> records) {
482 var seconds = records
483 .map((record) => record.ttl)
484 .reduce((a, b) => a < b ? a : b);
485 if (seconds < settings['MINTTL']!) seconds = settings['MINTTL']!;
486 if (seconds > settings['MAXTTL']!) seconds = settings['MAXTTL']!;
487 final ordered = records.toList()..sort(dataOrder);
488 cache[cacheKey(records[0].owner, records[0].type)] = CacheEntry(
489 records[0].owner,
490 records[0].type,
491 'POS',
492 ordered,
493 clock + seconds,
494 );
495 return records.map((record) => record.copy(ttl: seconds)).toList();
496 }

502 Reply? queryServers(
503 String name,
504 String type,
505 String cut,
506 List<String>? nameservers,
507 int depth,
508 List<String> stack,
509 ) {
510 Reply? query(int server) {
511 final response = ask(server, name, type, cut);
512 trace(
513 depth,
514 'ASK ${addressText(server)} $name $type -> ${replyText(response)}',
515 );
516 return response.kind == 'LAME' || response.kind == 'TIMEOUT'
517 ? null
518 : response;
519 }
520
521 if (nameservers == null) {
522 for (final server in roots) {
523 final response = query(server);
524 if (response != null) return response;
525 }
526 return null;
527 }
528 final ordered = nameservers.toList()..sort(nameOrder);
529 for (final host in ordered) {
530 final cached = cache[cacheKey(host, 'A')];
531 var addresses = <int>[];
532 if (cached?.kind == 'POS') {
533 addresses = cached!.records.map((record) => record.address!).toList();
534 } else if (stack.contains(cacheKey(host, 'A'))) {
535 trace(depth, 'GLUELESS $host -> LOOP');
536 } else {
537 trace(depth, 'GLUELESS $host');
538 final resolution = resolve(host, 'A', depth: depth + 1, stack: stack);
539 if (resolution.status == 'NOERROR')
540 addresses = resolution.answer
541 .where((record) => record.type == 'A')
542 .map((record) => record.address!)
543 .toList();
544 }
545 final unique = addresses.toSet().toList()..sort();
546 for (final server in unique) {
547 final response = query(server);
548 if (response != null) return response;
549 }
550 }
551 return null;
552 }

554 Reply lookup(String name, String type, int depth, List<String> stack) {
555 Reply hit(Reply response, String text) {
556 hits++;
557 trace(depth, 'CACHE HIT $name $type -> $text');
558 return response;
559 }
560
561 final exact = cache[cacheKey(name, type)];
562 if (exact != null)
563 return hit(
564 exact.kind == 'POS'
565 ? Reply(
566 'ANSWER',
567 answer: exact.records
568 .map((record) => record.copy(ttl: exact.expiry - clock))
569 .toList(),
570 )
571 : Reply(exact.kind),
572 exact.kind == 'POS' ? 'ANSWER' : exact.kind,
573 );
574 if (cache.values.any(
575 (entry) => entry.kind == 'NXDOMAIN' && below(name, entry.name),
576 ))
577 return hit(Reply('NXDOMAIN'), 'NXDOMAIN');
578 final cname = cache[cacheKey(name, 'CNAME')];
579 if (type != 'CNAME' && cname != null && cname.kind == 'POS')
580 return hit(
581 Reply(
582 'CNAME',
583 answer: cname.records
584 .map((record) => record.copy(ttl: cname.expiry - clock))
585 .toList(),
586 ),
587 'CNAME ${cname.records[0].data[0]}',
588 );
589 final ns = find(
590 ancestors(name)
591 .map((ancestor) => cache[cacheKey(ancestor, 'NS')])
592 .whereType<CacheEntry>(),
593 (entry) => entry.kind == 'POS',
594 );
595 var cut = ns?.name ?? '.';
596 List<String>? nameservers = ns?.records
597 .map((record) => record.data[0])
598 .toList();
599 if (ns != null) {
600 hits++;
601 trace(depth, 'CACHE HIT $name $type -> REFERRAL $cut');
602 }
603 while (true) {
604 final response = queryServers(name, type, cut, nameservers, depth, stack);
605 if (response == null)
606 return Reply('SERVFAIL', reason: 'no server for $cut');
607 if (response.kind == 'REFERRAL') {
608 positive(response.authority);
609 final groups = <String, List<DnsRecord>>{};
610 for (final record in response.additional)
611 groups.putIfAbsent(record.owner, () => []).add(record);
612 for (final group in groups.values) positive(group);
613 cut = response.cut!;
614 nameservers = response.authority
615 .map((record) => record.data[0])
616 .toList();
617 } else if (response.kind == 'ANSWER' || response.kind == 'CNAME') {
618 return Reply(response.kind, answer: positive(response.answer));
619 } else {
620 final soa = response.authority[0];
621 var seconds = soa.ttl;
622 if (seconds > soa.minimum!) seconds = soa.minimum!;
623 if (seconds > settings['NEGTTL']!) seconds = settings['NEGTTL']!;
624 cache[cacheKey(name, type)] = CacheEntry(
625 name,
626 type,
627 response.kind,
628 [],
629 clock + seconds,
630 );
631 return response;
632 }
633 }
634 }

636 Resolution resolve(
637 String name,
638 String type, {
639 int depth = 0,
640 List<String>? stack,
641 }) {
642 final pending = stack ?? <String>[];
643 pending.add(cacheKey(name, type));
644 final chain = <DnsRecord>[];
645 final seen = {name};
646 var current = name;
647 var status = '';
648 var answer = <DnsRecord>[];
649 while (true) {
650 final result = lookup(current, type, depth, pending);
651 if (result.kind == 'CNAME') {
652 final record = result.answer[0];
653 chain.add(record);
654 final target = record.data[0];
655 if (seen.contains(target)) {
656 status = 'SERVFAIL cname loop';
657 break;
658 }
659 if (chain.length == 9) {
660 status = 'SERVFAIL cname chain too long';
661 break;
662 }
663 seen.add(target);
664 current = target;
665 } else {
666 status = result.kind == 'ANSWER'
667 ? 'NOERROR'
668 : result.kind == 'SERVFAIL'
669 ? 'SERVFAIL ${result.reason}'
670 : result.kind;
671 if (!status.startsWith('SERVFAIL'))
672 answer = [...chain, if (result.kind == 'ANSWER') ...result.answer];
673 break;
674 }
675 }
676 pending.removeLast();
677 final key = status.split(' ')[0];
678 resolutions[key] = resolutions[key]! + 1;
679 return Resolution(status, answer);
680 }

700 void command(List<String> words) {
701 final command = words[0];
702 if (command == 'RESOLVE' || command == 'MX') {
703 final name = parseName(words[1], wildcard: true);
704 final type = command == 'MX' ? 'MX' : words[2];
705 checkType(type);
706 output.add(command == 'MX' ? 'mx $name' : 'resolve $name $type');
707 final result = resolve(name, type);
708 emitResolution(result);
709 if (command == 'MX') {
710 if (result.status == 'NOERROR') {
711 for (final record in result.answer.where(
712 (record) => record.type == 'MX',
713 )) {
714 final host = record.data[1];
715 output.add(
716 host == '.'
717 ? 'exchange ${record.preference} . -> null'
718 : 'exchange ${record.preference} $host -> ${addresses(host)}',
719 );
720 }
721 } else if (result.status == 'NODATA') {
722 final aliases = result.answer
723 .where((record) => record.type == 'CNAME')
724 .toList();
725 final host = aliases.isEmpty ? name : aliases.last.data[0];
726 output.add('implicit $host -> ${addresses(host)}');
727 }
728 }
729 } else if (command == 'REVERSE') {
730 final String address;
731 final String name;
732 if (words[1].contains(':')) {
733 final groups = ipv6(words[1]);
734 address = ipv6Text(groups);
735 name =
736 '${groups.map((group) => group.toRadixString(16).padLeft(4, '0')).join().split('').reversed.join('.')}.ip6.arpa.';
737 } else {
738 address = addressText(ipv4(words[1]));
739 name = '${address.split('.').reversed.join('.')}.in-addr.arpa.';
740 }
741 output.add('reverse $address $name');
742 emitResolution(resolve(name, 'PTR'));
743 } else if (command == 'QUERY' || command == 'DUMP') {
744 final server = ipv4(words[1]);
745 final name = parseName(words[2], wildcard: command == 'QUERY');
746 if (command == 'QUERY') {
747 checkType(words[3]);
748 final result = ask(server, name, words[3], '.');
749 output.add(
750 'query ${addressText(server)} $name ${words[3]} -> ${replyText(result)}',
751 );
752 for (final record in result.answer)
753 output.add('answer ${recordText(record)}');
754 for (final record in result.authority)
755 output.add('authority ${recordText(record)}');
756 for (final record in result.additional)
757 output.add('additional ${recordText(record)}');
758 } else {
759 final zone = find(
760 zones,
761 (zone) => zone.server == server && zone.origin == name,
762 );
763 if (down.contains(server) ||
764 !zones.any((zone) => zone.server == server)) {
765 output.add('dump ${addressText(server)} $name -> TIMEOUT');
766 } else if (zone == null) {
767 output.add('dump ${addressText(server)} $name -> REFUSED');
768 } else {
769 output.add(
770 'dump ${addressText(server)} $name records=${zone.records.length}',
771 );
772 final records = zone.records.toList()..sort(recordOrder);
773 for (final record in records) output.add(' ${recordText(record)}');
774 }
775 }
776 } else if (command == 'DOWN' || command == 'UP') {
777 final address = ipv4(words[1]);
778 if (command == 'DOWN' && down.contains(address))
779 throw DnsFailure('server ${words[1]} already down');
780 if (command == 'UP' && !down.contains(address))
781 throw DnsFailure('server ${words[1]} not down');
782 if (command == 'DOWN') {
783 down.add(address);
784 } else {
785 down.remove(address);
786 }
787 output.add('${command.toLowerCase()} ${addressText(address)}');
788 } else if (command == 'SET') {
789 if (!settings.containsKey(words[1]))
790 throw DnsFailure('bad setting ${words[1]}');
791 if (!integerWord.hasMatch(words[2]) ||
792 int.parse(words[2]) < 1 ||
793 int.parse(words[2]) > 604800)
794 throw DnsFailure('bad value ${words[2]}');
795 final value = int.parse(words[2]);
796 final minimum = words[1] == 'MINTTL' ? value : settings['MINTTL']!;
797 final maximum = words[1] == 'MAXTTL' ? value : settings['MAXTTL']!;
798 if (minimum > maximum) throw DnsFailure('minttl above maxttl');
799 settings[words[1]] = value;
800 output.add('set ${words[1]}=$value');
801 } else if (command == 'TIME') {
802 if (!integerWord.hasMatch(words[1]) ||
803 int.parse(words[1]) < 1 ||
804 int.parse(words[1]) > 1000000)
805 throw DnsFailure('bad time ${words[1]}');
806 clock += int.parse(words[1]);
807 final before = cache.length;
808 cache.removeWhere((key, entry) => entry.expiry <= clock);
809 output.add('time now=$clock expired=${before - cache.length}');
810 } else if (command == 'CACHE' || command == 'FLUSH') {
811 final name = words.length == 1 ? '.' : parseName(words[1]);
812 final entries =
813 cache.values.where((entry) => below(entry.name, name)).toList()
814 ..sort((a, b) {
815 final comparison = nameOrder(a.name, b.name);
816 return comparison == 0
817 ? types.indexOf(a.type).compareTo(types.indexOf(b.type))
818 : comparison;
819 });
820 if (command == 'FLUSH') {
821 for (final entry in entries)
822 cache.remove(cacheKey(entry.name, entry.type));
823 output.add('flush removed=${entries.length}');
824 } else {
825 output.add('cache entries=${entries.length}');
826 for (final entry in entries) {
827 if (entry.kind == 'POS') {
828 for (final record in entry.records)
829 output.add(
830 ' ${recordText(record.copy(ttl: entry.expiry - clock))}',
831 );
832 } else {
833 output.add(
834 ' ${entry.name} ${entry.expiry - clock} ${entry.type} ${entry.kind}',
835 );
836 }
837 }
838 }
839 } else {
840 final count = queries.values.fold<int>(
841 0,
842 (total, counts) => total + counts.fold<int>(0, (a, b) => a + b),
843 );
844 output.add('stats queries=$count hits=$hits');
845 final servers = queries.keys.toList()..sort();
846 for (final server in servers) {
847 final counts = queries[server]!;
848 output.add(
849 ' ${addressText(server)} sent=${counts.fold<int>(0, (a, b) => a + b)} ${List.generate(kinds.length, (i) => '${kinds[i].toLowerCase()}=${counts[i]}').join(' ')}',
850 );
851 }
852 output.add(
853 'resolutions noerror=${resolutions['NOERROR']} nodata=${resolutions['NODATA']} nxdomain=${resolutions['NXDOMAIN']} servfail=${resolutions['SERVFAIL']}',
854 );
855 }
856 }

858 void run(List<String> lines) {
859 var position = 0;
860 while (lines[position] != 'END ZONES') {
861 try {
862 zoneLine(lines[position].split(' '), position + 1);
863 } on DnsFailure catch (error) {
864 output.add('error line ${position + 1}: ${error.message}');
865 }
866 position++;
867 }
868 position++;
869 finishZones();
870 while (lines[position] != 'END') {
871 try {
872 command(lines[position].split(' '));
873 } on DnsFailure catch (error) {
874 output.add('error line ${position + 1}: ${error.message}');
875 }
876 position++;
877 }
878 }
879}
