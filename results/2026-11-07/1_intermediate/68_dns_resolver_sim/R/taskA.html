29parse_name <- function(word, wildcard = FALSE, origin = NULL) {
30 if (!is.null(origin) && word == "@") return(origin)
31 if (word == ".") return(".")
32 parts <- split_keep(if (endsWith(word, ".")) substr(word, 1, nchar(word) - 1) else word, ".")
33 for (i in seq_along(parts)) {
34 part <- parts[i]
35 if (wildcard && i == 1 && part == "*") next
36 if (nchar(part) < 1 || nchar(part) > 63 || !grepl("^[A-Za-z0-9]([A-Za-z0-9-]*[A-Za-z0-9])?$", part)) fail(paste0("bad name ", word))
37 }
38 result <- paste0(chartr("A-Z", "a-z", paste(parts, collapse = ".")), ".")
39 if (!is.null(origin) && origin != "." && !endsWith(word, ".")) result <- paste0(result, origin)

42address <- function(word) {
43 if (!grepl("^(0|[1-9][0-9]{0,2})(\\.(0|[1-9][0-9]{0,2})){3}$", word)) fail(paste0("bad address ", word))
44 parts <- as.numeric(strsplit(word, ".", fixed = TRUE)[[1]])
45 if (any(parts > 255)) fail(paste0("bad address ", word))
46 sum(parts * 256^(3:0))

48ip <- function(number) paste(floor(number / 256^(3:0)) %% 256, collapse = ".")
49ipv6 <- function(word) {
50 parts <- split_keep(word, "::")
51 if (length(parts) > 2) fail(paste0("bad address ", word))
52 sides <- lapply(parts, function(part) if (part == "") character() else split_keep(part, ":"))
53 groups <- unlist(sides, use.names = FALSE)
54 if (any(!grepl("^[0-9A-Fa-f]{1,4}$", groups)) || (if (length(parts) == 1) length(groups) != 8 else length(groups) > 7)) fail(paste0("bad address ", word))
55 if (length(parts) == 1) as.numeric(strtoi(groups, 16)) else c(as.numeric(strtoi(sides[[1]], 16)), rep(0, 8 - length(groups)), as.numeric(strtoi(sides[[2]], 16)))
56}
57ipv6_text <- function(groups) {
58 best_start <- 0
59 best_length <- 1
60 index <- 1
61 while (index <= 8) {
62 if (groups[index] != 0) {
63 index <- index + 1
64 next
65 }
66 finish <- index
67 while (finish <= 8 && groups[finish] == 0) finish <- finish + 1
68 if (finish - index > best_length) {
69 best_start <- index
70 best_length <- finish - index
71 }
72 index <- finish
73 }
74 words <- sprintf("%x", as.integer(groups))
75 if (!best_start) return(paste(words, collapse = ":"))
76 left <- if (best_start > 1) words[seq_len(best_start - 1)] else character()
77 right <- if (best_start + best_length <= 8) words[seq.int(best_start + best_length, 8)] else character()
78 paste0(paste(left, collapse = ":"), "::", paste(right, collapse = ":"))
79}
80integer_word <- function(word) grepl("^(0|[1-9][0-9]{0,8})$", word)
81parse_ttl <- function(word) {
82 if (integer_word(word)) seconds <- as.numeric(word) else {
83 text <- chartr("A-Z", "a-z", word)
84 if (!grepl("^((0|[1-9][0-9]{0,2})[wdhms])+$", text)) fail(paste0("bad ttl ", word))
85 pairs <- regmatches(text, gregexpr("[0-9]+[wdhms]", text))[[1]]
86 units <- c(w = 604800, d = 86400, h = 3600, m = 60, s = 1)
87 previous <- 0
88 seconds <- 0
89 for (pair in pairs) {
90 unit <- substr(pair, nchar(pair), nchar(pair))
91 rank <- match(unit, names(units))
92 if (rank <= previous) fail(paste0("bad ttl ", word))
93 previous <- rank
94 seconds <- seconds + as.numeric(substr(pair, 1, nchar(pair) - 1)) * units[unit]
95 }
96 }
97 if (seconds < 1 || seconds > 604800) fail(paste0("bad ttl ", word))
98 unname(seconds)

104sorted_records <- function(records) {

106 type <- records[[1]]$type
107 if (type == "A") indices <- order(vapply(records, function(record) record$data, numeric(1)))
108 else if (type == "AAAA") {
109 groups <- do.call(rbind, lapply(records, function(record) record$data))
110 indices <- do.call(order, as.data.frame(groups))
111 } else if (type == "MX") indices <- order(vapply(records, function(record) record$data[[1]], numeric(1)), vapply(records, function(record) name_key(record$data[[2]]), character(1)), method = "radix")
112 else if (type == "TXT") indices <- order(vapply(records, function(record) record$data, character(1)), method = "radix")
113 else if (type == "SOA") indices <- seq_along(records)
114 else indices <- order(vapply(records, function(record) name_key(record$data), character(1)), method = "radix")
115 records[indices]

123zone_line <- function(words, number) {
124 if (words[1] == "ROOT") {
125 added <- numeric()
126 for (word in words[-1]) {
127 number <- address(word)
128 if (number %in% c(roots, added)) fail(paste0("duplicate root ", word))
129 added <- c(added, number)
130 }
131 roots <<- c(roots, added)
132 } else if (words[1] == "ZONE") {
133 current_zone <<- NULL
134 origin <- parse_name(words[2])
135 server <- address(words[3])
136 for (zone in zones) if (zone$origin == origin && zone$server == server) fail(paste0("duplicate zone ", origin, " at ", words[3]))
137 current_zone <<- list2env(list(origin = origin, server = server, line = number, records = list()), parent = emptyenv())
138 zones[[length(zones) + 1]] <<- current_zone
139 } else {
140 if (is.null(current_zone)) fail("no current zone")
141 owner <- parse_name(words[1], TRUE, current_zone$origin)
142 if (!below(owner, current_zone$origin)) fail(paste0("out of zone ", owner))
143 ttl <- parse_ttl(words[2])
144 type <- check_type(words[3])
145 fields <- words[-(1:3)]
146 count <- switch(type, MX = 2, SOA = 7, 1)
147 if (type != "TXT" && length(fields) != count) fail("wrong field count")
148 if (type == "A") data <- address(fields[1])
149 else if (type == "AAAA") data <- ipv6(fields[1])
150 else if (type %in% c("NS", "CNAME", "PTR")) data <- parse_name(fields[1], FALSE, current_zone$origin)
151 else if (type == "MX") {
152 if (!integer_word(fields[1]) || as.numeric(fields[1]) > 65535) fail(paste0("bad preference ", fields[1]))
153 data <- list(as.numeric(fields[1]), parse_name(fields[2], FALSE, current_zone$origin))
154 } else if (type == "TXT") {
155 data <- paste(fields, collapse = " ")
156 if (!grepl('^"[^"]*"$', data)) fail(paste0("bad text ", data))
157 } else {
158 data <- lapply(fields[1:2], parse_name, origin = current_zone$origin)
159 for (field in fields[3:6]) if (!integer_word(field)) fail(paste0("bad number ", field))
160 if (!integer_word(fields[7]) || as.numeric(fields[7]) < 1 || as.numeric(fields[7]) > 86400) fail(paste0("bad minimum ", fields[7]))
161 data <- c(data, as.list(as.numeric(fields[3:7])))
162 }
163 records <- current_zone$records
164 if (type == "NS" && length(labels(owner)) && labels(owner)[1] == "*") fail("ns at wildcard")
165 if (type == "SOA" && owner != current_zone$origin) fail("soa not at origin")
166 if (type == "SOA" && any(vapply(records, function(record) record$type == "SOA", logical(1)))) fail("duplicate soa")
167 if (any(vapply(records, function(record) record$owner == owner && record$type == type && identical(record$data, data), logical(1)))) fail("duplicate record")
168 if (any(vapply(records, function(record) record$owner == owner && (type == "CNAME" || record$type == "CNAME"), logical(1)))) fail(paste0("cname conflict ", owner))
169 current_zone$records[[length(records) + 1]] <- list(owner = owner, ttl = ttl, type = type, data = data)
170 }
171}

174authoritative <- function(server, name, type, cut) {
175 if (server %in% down || !known(server)) return(response("TIMEOUT"))
176 chosen <- NULL
177 for (zone in zones) if (zone$server == server && below(name, zone$origin) && (is.null(chosen) || length(labels(zone$origin)) > length(labels(chosen$origin)))) chosen <- zone
178 if (is.null(chosen) || length(labels(chosen$origin)) < length(labels(cut))) return(response("LAME"))
179 records <- chosen$records
180 soa <- Filter(function(record) record$type == "SOA", records)[[1]]
181 for (ancestor in rev(ancestors(name))) {
182 if (ancestor == chosen$origin || !below(ancestor, chosen$origin)) next
183 authority <- sorted_records(Filter(function(record) record$owner == ancestor && record$type == "NS", records))
184 if (!length(authority)) next
185 glue <- list()
186 for (record in authority) glue <- c(glue, sorted_records(Filter(function(item) item$owner == record$data && item$type == "A", records)))
187 return(response("REFERRAL", authority = authority, additional = glue, cut = ancestor))
188 }
189 found <- Filter(function(record) record$owner == name, records)
190 if (!length(found)) {
191 exists <- function(ancestor) any(vapply(records, function(record) below(record$owner, ancestor), logical(1)))
192 if (exists(name)) return(response("NODATA", authority = list(soa)))
193 closest <- "."
194 for (ancestor in ancestors(name)[-1]) if (exists(ancestor)) {
195 closest <- ancestor
196 break
197 }
198 wildcard <- if (closest == ".") "*." else paste0("*.", closest)
199 found <- Filter(function(record) record$owner == wildcard, records)
200 if (!length(found)) return(response("NXDOMAIN", authority = list(soa)))
201 }
202 answer <- Filter(function(record) record$type == type, found)
203 kind <- "ANSWER"
204 if (!length(answer) && type != "CNAME") {
205 answer <- Filter(function(record) record$type == "CNAME", found)
206 kind <- "CNAME"
207 }
208 if (!length(answer)) return(response("NODATA", authority = list(soa)))
209 answer <- lapply(sorted_records(answer), function(record) {
210 record$owner <- name
211 record
212 })
213 response(kind, answer)
214}

240query_servers <- function(name, type, cut, nameservers, depth, resolving) {
241 if (is.null(nameservers)) {
242 for (server in roots) {
243 result <- ask(server, name, type, cut)
244 trace(depth, paste0("ASK ", ip(server), " ", name, " ", type, " -> ", describe(result)))
245 if (!(result$kind %in% c("LAME", "TIMEOUT"))) return(result)
246 }
247 } else {
248 nameservers <- nameservers[order(vapply(nameservers, name_key, character(1)), method = "radix")]
249 for (host in nameservers) {
250 entry <- cache[[cache_key(host, "A")]]
251 if (!is.null(entry) && entry$kind == "POS") addresses <- vapply(entry$records, function(record) record$data, numeric(1))
252 else if (cache_key(host, "A") %in% resolving) {
253 trace(depth, paste0("GLUELESS ", host, " -> LOOP"))
254 addresses <- numeric()
255 } else {
256 trace(depth, paste0("GLUELESS ", host))
257 result <- resolve(host, "A", depth + 1, resolving)
258 addresses <- if (result$status == "NOERROR") vapply(Filter(function(record) record$type == "A", result$answer), function(record) record$data, numeric(1)) else numeric()
259 }
260 for (server in sort(unique(addresses))) {
261 result <- ask(server, name, type, cut)
262 trace(depth, paste0("ASK ", ip(server), " ", name, " ", type, " -> ", describe(result)))
263 if (!(result$kind %in% c("LAME", "TIMEOUT"))) return(result)
264 }
265 }
266 }
267 NULL

269lookup <- function(name, type, depth, resolving) {
270 entry <- cache[[cache_key(name, type)]]
271 result <- NULL
272 if (!is.null(entry)) result <- if (entry$kind == "POS") response("ANSWER", cache_records(entry)) else response(entry$kind)
273 else if (any(vapply(cache, function(item) item$kind == "NXDOMAIN" && below(name, item$name), logical(1)))) result <- response("NXDOMAIN")
274 if (is.null(result) && type != "CNAME") {
275 entry <- cache[[cache_key(name, "CNAME")]]
276 if (!is.null(entry) && entry$kind == "POS") result <- response("CNAME", cache_records(entry))
277 }
278 if (!is.null(result)) {
279 hits <<- hits + 1
280 trace(depth, paste0("CACHE HIT ", name, " ", type, " -> ", describe(result)))
281 return(result)
282 }
283 cut <- "."
284 servers <- NULL
285 for (ancestor in ancestors(name)) {
286 entry <- cache[[cache_key(ancestor, "NS")]]
287 if (!is.null(entry) && entry$kind == "POS") {
288 cut <- ancestor
289 servers <- vapply(entry$records, function(record) record$data, character(1))
290 hits <<- hits + 1
291 trace(depth, paste0("CACHE HIT ", name, " ", type, " -> REFERRAL ", cut))
292 break
293 }
294 }
295 repeat {
296 result <- query_servers(name, type, cut, servers, depth, resolving)
297 if (is.null(result)) return(response(paste0("SERVFAIL no server for ", cut)))
298 if (result$kind == "REFERRAL") {
299 store(result$authority)
300 owners <- unique(vapply(result$additional, function(record) record$owner, character(1)))
301 for (owner in owners) store(Filter(function(record) record$owner == owner, result$additional))
302 cut <- result$cut
303 servers <- vapply(result$authority, function(record) record$data, character(1))
304 } else if (result$kind %in% c("ANSWER", "CNAME")) {
305 result$answer <- store(result$answer)
306 return(result)
307 } else {
308 soa <- result$authority[[1]]
309 ttl <- min(soa$ttl, soa$data[[7]], settings["NEGTTL"])
310 cache[[cache_key(name, type)]] <<- list(name = name, type = type, kind = result$kind, expiry = clock + ttl)
311 return(result)
312 }
313 }

315resolve <- function(name, type, depth = 0, resolving = character()) {
316 resolving <- c(resolving, cache_key(name, type))
317 seen <- name
318 chain <- list()
319 current <- name
320 repeat {
321 result <- lookup(current, type, depth, resolving)
322 if (result$kind == "CNAME") {
323 record <- result$answer[[1]]
324 chain[[length(chain) + 1]] <- record
325 target <- record$data
326 if (target %in% seen) {
327 status <- "SERVFAIL cname loop"
328 answer <- list()
329 break
330 }
331 if (length(chain) == 9) {
332 status <- "SERVFAIL cname chain too long"
333 answer <- list()
334 break
335 }
336 seen <- c(seen, target)
337 current <- target
338 } else {
339 status <- if (result$kind == "ANSWER") "NOERROR" else result$kind
340 answer <- if (startsWith(status, "SERVFAIL")) list() else c(chain, result$answer)
341 break
342 }
343 }
344 key <- if (startsWith(status, "SERVFAIL")) "SERVFAIL" else status
345 resolutions[key] <<- resolutions[key] + 1
346 list(status = status, answer = answer)
347}

389 } else {

417 server <- address(w[2])
418 if (kind == "DOWN" && server %in% down) fail(paste0("server ", w[2], " already down"))
419 if (kind == "UP" && !(server %in% down)) fail(paste0("server ", w[2], " not down"))
420 down <<- if (kind == "DOWN") c(down, server) else down[down != server]
421 emit(tolower(kind), " ", ip(server))
422 } else if (kind == "SET") {
423 key <- w[2]
424 if (!(key %in% names(settings))) fail(paste0("bad setting ", key))
425 if (!integer_word(w[3]) || as.numeric(w[3]) < 1 || as.numeric(w[3]) > 604800) fail(paste0("bad value ", w[3]))
426 changed <- settings
427 changed[key] <- as.numeric(w[3])
428 if (changed["MINTTL"] > changed["MAXTTL"]) fail("minttl above maxttl")
429 settings <<- changed
430 emit("set ", key, "=", fmt(changed[key]))
431 } else if (kind == "TIME") {
432 if (!integer_word(w[2]) || as.numeric(w[2]) < 1 || as.numeric(w[2]) > 1000000) fail(paste0("bad time ", w[2]))
433 clock <<- clock + as.numeric(w[2])
434 before <- length(cache)
435 cache <<- Filter(function(entry) entry$expiry > clock, cache)

437 } else if (kind %in% c("CACHE", "FLUSH")) {
438 name <- if (length(w) > 1) parse_name(w[2]) else NULL
439 entries <- Filter(function(entry) is.null(name) || below(entry$name, name), cache)
440 if (kind == "FLUSH") {

442 emit("flush removed=", length(entries))
443 } else {
444 emit("cache entries=", length(entries))
445 indices <- order(vapply(entries, function(entry) name_key(entry$name), character(1)), vapply(entries, function(entry) match(entry$type, types), integer(1)), method = "radix")
446 for (entry in entries[indices]) {
447 remaining <- entry$expiry - clock
448 if (entry$kind == "POS") for (record in entry$records) emit(" ", record_text(record, remaining))
449 else emit(" ", entry$name, " ", fmt(remaining), " ", entry$type, " ", entry$kind)
450 }
451 }
452 } else if (kind == "STATS") {
453 emit("stats queries=", sum(vapply(queries, function(record) sum(record$counts), numeric(1))), " hits=", hits)
454 indices <- order(vapply(queries, function(record) record$server, numeric(1)))
455 for (record in queries[indices]) emit(" ", ip(record$server), " sent=", sum(record$counts), " ", paste(paste0(tolower(kinds), "=", record$counts[kinds]), collapse = " "))
456 emit("resolutions ", paste(paste0(tolower(names(resolutions)), "=", resolutions), collapse = " "))
457 }
