1class DnsError < StandardError
2end
3
4class Resolver
5 TYPES = %w[A NS CNAME SOA PTR MX TXT AAAA].freeze
6 KINDS = %w[ANSWER CNAME REFERRAL NODATA NXDOMAIN LAME TIMEOUT].freeze
7 def initialize
8 @roots = []
9 @zones = []
10 @current = nil
11 @down = []
12 @cache = {}
13 @settings = { 'MINTTL' => 1, 'MAXTTL' => 86400, 'NEGTTL' => 3600 }
14 @clock = @hits = 0
15 @queries = {}
16 @resolutions = Hash.new(0)
17 end
18 def error(message)
19 raise DnsError, message
20 end
21 def labels(name)
22 name == '.' ? [] : name[0...-1].split('.')
23 end
24 def name_key(name)
25 labels(name).reverse
26 end
27 def below?(name, ancestor)
28 ancestor == '.' || name == ancestor || name.end_with?('.' + ancestor)
29 end
30 def ancestors(name)
31 parts = labels(name)
32 (0..parts.length).map { |i| i == parts.length ? '.' : parts[i..].join('.') + '.' }
33 end
34 def name(word, wildcard = false, origin = nil)
35 return origin if origin && word == '@'
36 return '.' if word == '.'
37 parts = (word.end_with?('.') ? word[0...-1] : word).split('.', -1)
38 valid = parts.each_with_index.all? do |part, index|
39 (wildcard && index == 0 && part == '*') || (part.length.between?(1, 63) && part.match?(/\A[A-Za-z0-9](?:[A-Za-z0-9-]*[A-Za-z0-9])?\z/))
40 end
41 error("bad name #{word}") unless valid
42 result = parts.map(&:downcase).join('.') + '.'
43 result += origin if origin && origin != '.' && !word.end_with?('.')
44 result
45 end
46 def address(word)
47 error("bad address #{word}") unless word.match?(/\A(?:0|[1-9][0-9]{0,2})(?:\.(?:0|[1-9][0-9]{0,2})){3}\z/) && word.split('.').all? { |part| part.to_i <= 255 }
48 word.split('.').reduce(0) { |n, part| n * 256 + part.to_i }
49 end
50 def ip(number)
51 [24, 16, 8, 0].map { |shift| (number >> shift) & 255 }.join('.')
52 end
53 def ipv6(word)
54 pieces = word.split('::', -1)
55 valid = pieces.length <= 2
56 sides = pieces.map { |side| side.empty? ? [] : side.split(':', -1) }
57 valid &&= sides.flatten.all? { |group| group.match?(/\A[0-9A-Fa-f]{1,4}\z/) }
58 count = sides.flatten.length
59 valid &&= pieces.length == 1 ? count == 8 : count <= 7
60 error("bad address #{word}") unless valid
61 groups = sides[0].map { |group| group.to_i(16) }
62 groups += [0] * (8 - count) + sides[1].map { |group| group.to_i(16) } if pieces.length == 2
63 groups
64 end
65 def ipv6_text(groups)
66 best_start = nil
67 best_length = 1
68 index = 0
69 while index < 8
70 if groups[index] == 0
71 finish = index
72 finish += 1 while finish < 8 && groups[finish] == 0
73 if finish - index > best_length
74 best_start = index
75 best_length = finish - index
76 end
77 index = finish
78 else
79 index += 1
80 end
81 end
82 words = groups.map { |group| group.to_s(16) }
83 return words.join(':') unless best_start
84 words[0...best_start].join(':') + '::' + words[(best_start + best_length)..].join(':')
85 end
86 def integer?(word)
87 word.match?(/\A(?:0|[1-9][0-9]{0,8})\z/)
88 end
89 def ttl(word)
90 if integer?(word)
91 seconds = word.to_i
92 else
93 lower = word.downcase
94 error("bad ttl #{word}") unless lower.match?(/\A(?:(?:0|[1-9][0-9]{0,2})[wdhms])+\z/)
95 order = 'wdhms'
96 previous = -1
97 seconds = 0
98 lower.scan(/([0-9]+)([wdhms])/).each do |number, unit|
99 rank = order.index(unit)
100 error("bad ttl #{word}") unless rank > previous
101 previous = rank
102 seconds += number.to_i * { 'w' => 604800, 'd' => 86400, 'h' => 3600, 'm' => 60, 's' => 1 }[unit]
103 end
104 end
105 error("bad ttl #{word}") unless seconds.between?(1, 604800)
106 seconds
107 end
108 def check_type(type)
109 error("bad type #{type}") unless TYPES.include?(type)
110 type
111 end
112 def data_key(record)
113 data = record[:data]
114 case record[:type]
115 when 'A', 'AAAA', 'TXT' then data
116 when 'MX' then [data[0], name_key(data[1])]
117 when 'SOA' then []
118 else name_key(data)
119 end
120 end
121 def sorted(records)
122 records.sort_by { |record| data_key(record) }
123 end
124 def record_text(record, remaining = nil)
125 data = record[:data]
126 text = case record[:type]
127 when 'A' then ip(data)
128 when 'AAAA' then ipv6_text(data)
129 when 'MX', 'SOA' then data.join(' ')
130 else data
131 end
132 "#{record[:owner]} #{remaining || record[:ttl]} #{record[:type]} #{text}"
133 end
134 def zone_line(words, number)
135 if words[0] == 'ROOT'
136 added = []
137 words.drop(1).each do |word|
138 value = address(word)
139 error("duplicate root #{word}") if @roots.include?(value) || added.include?(value)
140 added << value
141 end
142 @roots.concat(added)
143 elsif words[0] == 'ZONE'
144 @current = nil
145 origin = name(words[1])
146 server = address(words[2])
147 error("duplicate zone #{origin} at #{words[2]}") if @zones.any? { |zone| zone[:origin] == origin && zone[:server] == server }
148 @current = { origin: origin, server: server, line: number, records: [] }
149 @zones << @current
150 else
151 error('no current zone') unless @current
152 owner = name(words[0], true, @current[:origin])
153 error("out of zone #{owner}") unless below?(owner, @current[:origin])
154 lifetime = ttl(words[1])
155 type = check_type(words[2])
156 fields = words.drop(3)
157 count = { 'MX' => 2, 'SOA' => 7 }.fetch(type, 1)
158 error('wrong field count') unless type == 'TXT' || fields.length == count
159 data = case type
160 when 'A' then address(fields[0])
161 when 'AAAA' then ipv6(fields[0])
162 when 'NS', 'CNAME', 'PTR' then name(fields[0], false, @current[:origin])
163 when 'MX'
164 error("bad preference #{fields[0]}") unless integer?(fields[0]) && fields[0].to_i <= 65535
165 [fields[0].to_i, name(fields[1], false, @current[:origin])]
166 when 'TXT'
167 text = fields.join(' ')
168 error("bad text #{text}") unless text.match?(/\A"[^"]*"\z/)
169 text
170 when 'SOA'
171 values = fields[0, 2].map { |field| name(field, false, @current[:origin]) }
172 fields[2, 4].each { |field| error("bad number #{field}") unless integer?(field) }
173 error("bad minimum #{fields[6]}") unless integer?(fields[6]) && fields[6].to_i.between?(1, 86400)
174 values + fields.drop(2).map(&:to_i)
175 end
176 records = @current[:records]
177 error('ns at wildcard') if type == 'NS' && labels(owner)[0] == '*'
178 error('soa not at origin') if type == 'SOA' && owner != @current[:origin]
179 error('duplicate soa') if type == 'SOA' && records.any? { |record| record[:type] == 'SOA' }
180 error('duplicate record') if records.any? { |record| record[:owner] == owner && record[:type] == type && record[:data] == data }
181 conflict = records.any? { |record| record[:owner] == owner && (type == 'CNAME' || record[:type] == 'CNAME') }
182 error("cname conflict #{owner}") if conflict
183 records << { owner: owner, ttl: lifetime, type: type, data: data }
184 end
185 end
186 def response(kind, answer = [], authority = [], additional = [], cut = nil)
187 { kind: kind, answer: answer, authority: authority, additional: additional, cut: cut }
188 end
189 def authoritative(server, queried_name, type, cut)
190 return response('TIMEOUT') if @down.include?(server) || @zones.none? { |zone| zone[:server] == server }
191 zone = @zones.select { |item| item[:server] == server && below?(queried_name, item[:origin]) }.max_by { |item| labels(item[:origin]).length }
192 return response('LAME') unless zone && labels(zone[:origin]).length >= labels(cut).length
193 records = zone[:records]
194 soa = records.find { |record| record[:type] == 'SOA' }
195 delegation = ancestors(queried_name).reverse.find do |ancestor|
196 ancestor != zone[:origin] && below?(ancestor, zone[:origin]) && records.any? { |record| record[:owner] == ancestor && record[:type] == 'NS' }
197 end
198 if delegation
199 authority = sorted(records.select { |record| record[:owner] == delegation && record[:type] == 'NS' })
200 glue = authority.flat_map { |record| sorted(records.select { |item| item[:owner] == record[:data] && item[:type] == 'A' }) }
201 return response('REFERRAL', [], authority, glue, delegation)
202 end
203 found = records.select { |record| record[:owner] == queried_name }
204 if found.empty?
205 exists = ->(ancestor) { records.any? { |record| below?(record[:owner], ancestor) } }
206 return response('NODATA', [], [soa]) if exists.call(queried_name)
207 closest = ancestors(queried_name).drop(1).find { |ancestor| exists.call(ancestor) }
208 wildcard = closest == '.' ? '*.' : '*.' + closest
209 found = records.select { |record| record[:owner] == wildcard }
210 return response('NXDOMAIN', [], [soa]) if found.empty?
211 end
212 answer = found.select { |record| record[:type] == type }
213 kind = 'ANSWER'
214 if answer.empty? && type != 'CNAME'
215 answer = found.select { |record| record[:type] == 'CNAME' }
216 kind = 'CNAME'
217 end
218 return response('NODATA', [], [soa]) if answer.empty?
219 response(kind, sorted(answer).map { |record| record.merge(owner: queried_name) })
220 end
221 def describe(result)
222 case result[:kind]
223 when 'REFERRAL' then 'REFERRAL ' + result[:cut]
224 when 'CNAME' then 'CNAME ' + result[:answer][0][:data]
225 else result[:kind]
226 end
227 end
228 def ask(server, queried_name, type, cut)
229 result = authoritative(server, queried_name, type, cut)
230 @queries[server] ||= Hash.new(0)
231 @queries[server][result[:kind]] += 1
232 result
233 end
234 def trace(depth, text)
235 puts ' ' * (depth + 1) + text
236 end
237 def store(records)
238 lifetime = [[records.map { |record| record[:ttl] }.min, @settings['MINTTL']].max, @settings['MAXTTL']].min
239 first = records[0]
240 @cache[[first[:owner], first[:type]]] = { kind: 'POS', records: sorted(records), expiry: @clock + lifetime }
241 records.map { |record| record.merge(ttl: lifetime) }
242 end
243 def cache_records(entry)
244 entry[:records].map { |record| record.merge(ttl: entry[:expiry] - @clock) }
245 end
246 def query_servers(queried_name, type, cut, nameservers, depth, resolving)
247 if nameservers.nil?
248 @roots.each do |server|
249 result = ask(server, queried_name, type, cut)
250 trace(depth, "ASK #{ip(server)} #{queried_name} #{type} -> #{describe(result)}")
251 return result unless %w[LAME TIMEOUT].include?(result[:kind])
252 end
253 else
254 nameservers.sort_by { |host| name_key(host) }.each do |host|
255 cached = @cache[[host, 'A']]
256 if cached && cached[:kind] == 'POS'
257 addresses = cached[:records].map { |record| record[:data] }
258 elsif resolving.include?([host, 'A'])
259 trace(depth, "GLUELESS #{host} -> LOOP")
260 addresses = []
261 else
262 trace(depth, "GLUELESS #{host}")
263 status, answer = resolve(host, 'A', depth + 1, resolving)
264 addresses = status == 'NOERROR' ? answer.select { |record| record[:type] == 'A' }.map { |record| record[:data] } : []
265 end
266 addresses.uniq.sort.each do |server|
267 result = ask(server, queried_name, type, cut)
268 trace(depth, "ASK #{ip(server)} #{queried_name} #{type} -> #{describe(result)}")
269 return result unless %w[LAME TIMEOUT].include?(result[:kind])
270 end
271 end
272 end
273 nil
274 end
275 def lookup(queried_name, type, depth, resolving)
276 entry = @cache[[queried_name, type]]
277 if entry
278 result = entry[:kind] == 'POS' ? response('ANSWER', cache_records(entry)) : response(entry[:kind])
279 elsif @cache.any? { |(host, _), item| item[:kind] == 'NXDOMAIN' && below?(queried_name, host) }
280 result = response('NXDOMAIN')
281 elsif type != 'CNAME' && (entry = @cache[[queried_name, 'CNAME']]) && entry[:kind] == 'POS'
282 result = response('CNAME', cache_records(entry))
283 end
284 if result
285 @hits += 1
286 trace(depth, "CACHE HIT #{queried_name} #{type} -> #{describe(result)}")
287 return result
288 end
289 cut = ancestors(queried_name).find { |ancestor| (entry = @cache[[ancestor, 'NS']]) && entry[:kind] == 'POS' }
290 if cut
291 @hits += 1
292 trace(depth, "CACHE HIT #{queried_name} #{type} -> REFERRAL #{cut}")
293 servers = @cache[[cut, 'NS']][:records].map { |record| record[:data] }
294 else
295 cut = '.'
296 servers = nil
297 end
298 loop do
299 result = query_servers(queried_name, type, cut, servers, depth, resolving)
300 return response("SERVFAIL no server for #{cut}") unless result
301 case result[:kind]
302 when 'REFERRAL'
303 store(result[:authority])
304 result[:additional].group_by { |record| record[:owner] }.each_value { |records| store(records) }
305 cut = result[:cut]
306 servers = result[:authority].map { |record| record[:data] }
307 when 'ANSWER', 'CNAME'
308 result[:answer] = store(result[:answer])
309 return result
310 else
311 soa = result[:authority][0]
312 lifetime = [soa[:ttl], soa[:data][-1], @settings['NEGTTL']].min
313 @cache[[queried_name, type]] = { kind: result[:kind], expiry: @clock + lifetime }
314 return result
315 end
316 end
317 end
318 def resolve(queried_name, type, depth = 0, resolving = [])
319 resolving = resolving + [[queried_name, type]]
320 seen = [queried_name]
321 chain = []
322 current = queried_name
323 status = nil
324 answer = []
325 loop do
326 result = lookup(current, type, depth, resolving)
327 kind = result[:kind]
328 if kind == 'CNAME'
329 record = result[:answer][0]
330 chain << record
331 target = record[:data]
332 if seen.include?(target)
333 status = 'SERVFAIL cname loop'
334 answer = []
335 break
336 elsif chain.length == 9
337 status = 'SERVFAIL cname chain too long'
338 answer = []
339 break
340 end
341 seen << target
342 current = target
343 else
344 status = kind == 'ANSWER' ? 'NOERROR' : kind
345 answer = status.start_with?('SERVFAIL') ? [] : chain + result[:answer]
346 break
347 end
348 end
349 @resolutions[status.start_with?('SERVFAIL') ? 'SERVFAIL' : status] += 1
350 [status, answer]
351 end
352 def print_result(status, answer)
353 answer.each { |record| puts 'answer ' + record_text(record) }
354 puts "status #{status}"
355 end
356 def address_resolution(host)
357 output = []
358 %w[A AAAA].each do |type|
359 status, answer = resolve(host, type)
360 next unless status == 'NOERROR'
361 output.concat(answer.select { |record| record[:type] == type }.map { |record| type == 'A' ? ip(record[:data]) : ipv6_text(record[:data]) })
362 end
363 output.empty? ? 'no address' : output.join(' ')
364 end
365 def command(words)
366 kind = words[0]
367 case kind
368 when 'RESOLVE', 'MX'
369 queried_name = name(words[1], true)
370 type = kind == 'MX' ? 'MX' : check_type(words[2])
371 puts "#{kind.downcase} #{queried_name}" + (kind == 'MX' ? '' : " #{type}")
372 status, answer = resolve(queried_name, type)
373 print_result(status, answer)
374 if kind == 'MX' && status == 'NOERROR'
375 answer.select { |record| record[:type] == 'MX' }.each do |record|
376 preference, host = record[:data]
377 puts "exchange #{preference} #{host} -> #{host == '.' ? 'null' : address_resolution(host)}"
378 end
379 elsif kind == 'MX' && status == 'NODATA'
380 cnames = answer.select { |record| record[:type] == 'CNAME' }
381 host = cnames.empty? ? queried_name : cnames[-1][:data]
382 puts "implicit #{host} -> #{address_resolution(host)}"
383 end
384 when 'REVERSE'
385 if words[1].include?(':')
386 groups = ipv6(words[1])
387 text = ipv6_text(groups)
388 reverse = groups.map { |group| format('%04x', group) }.join.reverse.chars.join('.') + '.ip6.arpa.'
389 else
390 value = address(words[1])
391 text = ip(value)
392 reverse = text.split('.').reverse.join('.') + '.in-addr.arpa.'
393 end
394 puts "reverse #{text} #{reverse}"
395 print_result(*resolve(reverse, 'PTR'))
396 when 'QUERY', 'DUMP'
397 server = address(words[1])
398 queried_name = name(words[2], kind == 'QUERY')
399 if kind == 'QUERY'
400 type = check_type(words[3])
401 result = ask(server, queried_name, type, '.')
402 puts "query #{ip(server)} #{queried_name} #{type} -> #{describe(result)}"
403 %i[answer authority additional].each { |section| result[section].each { |record| puts "#{section} #{record_text(record)}" } }
404 else
405 zone = @zones.find { |item| item[:server] == server && item[:origin] == queried_name }
406 if @down.include?(server) || @zones.none? { |item| item[:server] == server }
407 puts "dump #{ip(server)} #{queried_name} -> TIMEOUT"
408 elsif !zone
409 puts "dump #{ip(server)} #{queried_name} -> REFUSED"
410 else
411 puts "dump #{ip(server)} #{queried_name} records=#{zone[:records].length}"
412 zone[:records].sort_by { |record| [name_key(record[:owner]), TYPES.index(record[:type]), data_key(record)] }.each { |record| puts ' ' + record_text(record) }
413 end
414 end
415 when 'DOWN', 'UP'
416 server = address(words[1])
417 error("server #{words[1]} already down") if kind == 'DOWN' && @down.include?(server)
418 error("server #{words[1]} not down") if kind == 'UP' && !@down.include?(server)
419 kind == 'DOWN' ? @down.push(server) : @down.delete(server)
420 puts "#{kind.downcase} #{ip(server)}"
421 when 'SET'
422 key, word = words[1, 2]
423 error("bad setting #{key}") unless @settings.key?(key)
424 error("bad value #{word}") unless integer?(word) && word.to_i.between?(1, 604800)
425 settings = @settings.merge(key => word.to_i)
426 error('minttl above maxttl') if settings['MINTTL'] > settings['MAXTTL']
427 @settings = settings
428 puts "set #{key}=#{word.to_i}"
429 when 'TIME'
430 word = words[1]
431 error("bad time #{word}") unless integer?(word) && word.to_i.between?(1, 1000000)
432 @clock += word.to_i
433 before = @cache.length
434 @cache.delete_if { |_, item| item[:expiry] <= @clock }
435 puts "time now=#{@clock} expired=#{before - @cache.length}"
436 when 'CACHE', 'FLUSH'
437 selected_name = words[1] ? name(words[1]) : nil
438 entries = @cache.select { |(host, _), _| selected_name.nil? || below?(host, selected_name) }
439 if kind == 'FLUSH'
440 entries.each_key { |key| @cache.delete(key) }
441 puts "flush removed=#{entries.length}"
442 else
443 puts "cache entries=#{entries.length}"
444 entries.sort_by { |(host, type), _| [name_key(host), TYPES.index(type)] }.each do |(host, type), entry|
445 remaining = entry[:expiry] - @clock
446 if entry[:kind] == 'POS'
447 entry[:records].each { |record| puts ' ' + record_text(record, remaining) }
448 else
449 puts " #{host} #{remaining} #{type} #{entry[:kind]}"
450 end
451 end
452 end
453 when 'STATS'
454 puts "stats queries=#{@queries.values.sum { |counts| counts.values.sum }} hits=#{@hits}"
455 @queries.keys.sort.each do |server|
456 counts = @queries[server]
457 puts " #{ip(server)} sent=#{counts.values.sum} " + KINDS.map { |response| "#{response.downcase}=#{counts[response]}" }.join(' ')
458 end
459 puts 'resolutions ' + %w[NOERROR NODATA NXDOMAIN SERVFAIL].map { |status| "#{status.downcase}=#{@resolutions[status]}" }.join(' ')
460 end
461 end
462 def run
463 lines = STDIN.each_line.map { |line| line.chomp("\n").delete_suffix("\r") }
464 zones = true
465 lines.each_with_index do |line, index|
466 break if !zones && line == 'END'
467 if zones && line == 'END ZONES'
468 @zones = @zones.select do |zone|
469 valid = zone[:records].any? { |record| record[:type] == 'SOA' }
470 if valid
471 puts "zone #{zone[:origin]} at #{ip(zone[:server])} records=#{zone[:records].length}"
472 else
473 puts "error line #{zone[:line]}: zone #{zone[:origin]} at #{ip(zone[:server])} has no soa"
474 end
475 valid
476 end
477 puts "loaded roots=#{@roots.length} zones=#{@zones.length} records=#{@zones.sum { |zone| zone[:records].length }}"
478 zones = false
479 next
480 end
481 begin
482 zones ? zone_line(line.split(' '), index + 1) : command(line.split(' '))
483 rescue DnsError => exception
484 puts "error line #{index + 1}: #{exception.message}"
485 end
486 end
487 end
488end
