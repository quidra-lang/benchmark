2private val kinds = listOf("ANSWER", "CNAME", "REFERRAL", "NODATA", "NXDOMAIN", "LAME", "TIMEOUT")
3private class Problem(val reason: String) : RuntimeException(reason)
4private fun integer(s: String): Int? = if (s.matches(Regex("0|[1-9][0-9]{0,8}"))) s.toInt() else null
5private fun ip(s: String): Long? {
6 val parts = s.split('.')
7 if (parts.size != 4) return null
8 var a = 0L
9 for (p in parts) {
10 if (p.length > 3) return null
11 val n = integer(p) ?: return null
12 if (n > 255) return null
13 a = a*256+n
14 }
15 return a
16}
17private fun address(a: Long) = "${a shr 24}.${(a shr 16) and 255}.${(a shr 8) and 255}.${a and 255}"
18private fun ipv6(s: String): List<Int>? {
19 fun groups(s: String): List<Int>? {
20 if (s.isEmpty()) return emptyList()
21 val out = mutableListOf<Int>()
22 for (p in s.split(':')) {
23 if (!p.matches(Regex("[0-9A-Fa-f]{1,4}"))) return null
24 out.add(p.toInt(16))
25 }
26 return out
27 }
28 val pieces = s.split("::")
29 if (pieces.size == 1) return groups(s)?.takeIf {
30 it.size == 8
31 }
32 if (pieces.size != 2) return null
33 val left = groups(pieces[0]) ?: return null
34 val right = groups(pieces[1]) ?: return null
35 if (left.size+right.size > 7) return null
36 return left+List(8-left.size-right.size) {
37 0
38 }+right
39}
40private fun address6(a: List<Int>): String {
41 var best = 0
42 var size = 0
43 var at = 0
44 while (at < 8) {
45 if (a[at] != 0) {
46 at++
47 continue
48 }
49 val start = at
50 while (at < 8 && a[at] == 0) at++
51 if (at-start >= 2 && at-start > size) {
52 best = start
53 size = at-start
54 }
55 }
56 fun text(a: List<Int>) = a.joinToString(":") {
57 it.toString(16)
58 }
59 return if (size == 0) text(a) else text(a.take(best))+"::"+text(a.drop(best+size))
60}
61private fun name(word: String, wild: Boolean, origin: String? = null): String {
62 if (word == "@" && origin != null) return origin
63 if (word == ".") return "."
64 val parts = word.removeSuffix(".").split('.')
65 for ((i, p) in parts.withIndex()) {
66 if (wild && i == 0 && p == "*") continue
67 if (p.length !in 1..63 || p.startsWith('-') || p.endsWith('-') || p.any {
68 it !in 'A'..'Z' && it !in 'a'..'z' && it !in '0'..'9' && it != '-'
69 }) throw Problem("bad name $word")
70 }
71 return parts.joinToString(".").lowercase()+"."+(if (!word.endsWith('.') && origin != null && origin != ".") origin else "")
72}
73private fun labels(s: String) = if (s == ".") emptyList() else s.dropLast(1).split('.')
74private fun nameOrder(a: String, b: String): Int {
75 val x = labels(a).asReversed()
76 val y = labels(b).asReversed()
77 for (i in 0 until minOf(x.size, y.size)) {
78 val c = x[i].compareTo(y[i])
79 if (c != 0) return c
80 }
81 return x.size.compareTo(y.size)
82}
83private fun below(a: String, b: String) = b == "." || a == b || a.endsWith(".$b")
84private fun ancestors(s: String): List<String> {
85 val p = labels(s)
86 return p.indices.map {
87 p.drop(it).joinToString(".")+"."
88 }+"."
89}
90private fun checkedIp(s: String) = ip(s) ?: throw Problem("bad address $s")
91private fun checkedType(s: String) {
92 if (s !in types) throw Problem("bad type $s")
93}
94private fun ttl(s: String): Int {
95 var seconds = integer(s)
96 if (seconds == null) {
97 var at = 0
98 var previous = -1
99 var total = 0
100 var valid = true
101 while (at < s.length) {
102 val start = at
103 while (at < s.length && s[at] in '0'..'9') at++
104 val n = integer(s.substring(start, at))
105 if (at == s.length || at-start > 3 || n == null) {
106 valid = false
107 break
108 }
109 val unit = "wdhms".indexOf(s[at].lowercaseChar())
110 if (unit <= previous) {
111 valid = false
112 break
113 }
114 total += n*intArrayOf(604800, 86400, 3600, 60, 1)[unit]
115 previous = unit
116 at++
117 }
118 if (valid && previous >= 0) seconds = total
119 }
120 return seconds?.takeIf {
121 it in 1..604800
122 } ?: throw Problem("bad ttl $s")
123}
124private sealed interface Data
125private data class A(val address: Long) : Data
126private data class Aaaa(val address: List<Int>) : Data
127private data class Target(val name: String) : Data
128private data class Mx(val preference: Int, val host: String) : Data
129private data class Txt(val text: String) : Data
130private data class Soa(val mname: String, val rname: String, val numbers: List<Int>) : Data
131private fun dataText(d: Data): String = when (d) {
132 is A -> address(d.address)
133 is Aaaa -> address6(d.address)
134 is Target -> d.name
135 is Mx -> "${d.preference} ${d.host}"
136 is Txt -> d.text
137 is Soa -> "${d.mname} ${d.rname} ${d.numbers.joinToString(" ")}"
138}
139private fun dataOrder(a: Data, b: Data): Int = when {
140 a is A && b is A -> a.address.compareTo(b.address)
141 a is Aaaa && b is Aaaa -> {
142 var c = 0
143 for (i in 0..7) {
144 c = a.address[i].compareTo(b.address[i])
145 if (c != 0) break
146 }
147 c
148 }
149 a is Target && b is Target -> nameOrder(a.name, b.name)
150 a is Mx && b is Mx -> a.preference.compareTo(b.preference).takeIf {
151 it != 0
152 } ?: nameOrder(a.host, b.host)
153 a is Txt && b is Txt -> a.text.compareTo(b.text)
154 else -> 0
155}
156private data class Record(val owner: String, val ttl: Int, val type: String, val data: Data) {

158}
159private val recordOrder = Comparator<Record> {
160 a, b -> nameOrder(a.owner, b.owner).takeIf {
161 it != 0
162 } ?: types.indexOf(a.type).compareTo(types.indexOf(b.type)).takeIf {
163 it != 0
164 } ?: dataOrder(a.data, b.data)
165}

188 fun zoneLine(w: List<String>, line: Int) {
189 if (w[0] == "ROOT") {
190 val added = mutableListOf<Long>()
191 for (word in w.drop(1)) {
192 val a = checkedIp(word)
193 if (a in roots || a in added) throw Problem("duplicate root $word")
194 added.add(a)
195 }
196 roots.addAll(added)
197 return
198 }
199 if (w[0] == "ZONE") {
200 current = null
201 val origin = name(w[1], false)
202 val server = checkedIp(w[2])
203 if (zones.any {
204 it.origin == origin && it.server == server
205 }) throw Problem("duplicate zone $origin at ${w[2]}")
206 current = zones.size
207 zones.add(Zone(origin, server, line))
208 return
209 }
210 val z = zones[current ?: throw Problem("no current zone")]
211 val owner = name(w[0], true, z.origin)
212 if (!below(owner, z.origin)) throw Problem("out of zone $owner")
213 val seconds = ttl(w[1])
214 val type = w[2]
215 checkedType(type)
216 val fields = w.drop(3)
217 val count = when (type) {
218 "MX" -> 2
219 "SOA" -> 7
220 "TXT" -> fields.size
221 else -> 1
222 }
223 if (fields.size != count) throw Problem("wrong field count")
224 val data: Data = when (type) {
225 "A" -> A(checkedIp(fields[0]))
226 "AAAA" -> Aaaa(ipv6(fields[0]) ?: throw Problem("bad address ${fields[0]}"))
227 "NS", "CNAME", "PTR" -> Target(name(fields[0], false, z.origin))
228 "MX" -> {
229 val p = integer(fields[0])?.takeIf {
230 it <= 65535
231 } ?: throw Problem("bad preference ${fields[0]}")
232 Mx(p, name(fields[1], false, z.origin))
233 }
234 "TXT" -> {
235 val s = fields.joinToString(" ")
236 if (s.length < 2 || !s.startsWith('"') || !s.endsWith('"') || '"' in s.substring(1, s.lastIndex)) throw Problem("bad text $s")
237 Txt(s)
238 }
239 else -> {
240 val a = name(fields[0], false, z.origin)
241 val b = name(fields[1], false, z.origin)
242 val n = (2..5).map {
243 integer(fields[it]) ?: throw Problem("bad number ${fields[it]}")
244 }+
245 (integer(fields[6])?.takeIf {
246 it in 1..86400
247 } ?: throw Problem("bad minimum ${fields[6]}"))
248 Soa(a, b, n)
249 }
250 }
251 if (type == "NS" && owner.startsWith("*.")) throw Problem("ns at wildcard")
252 if (type == "SOA" && owner != z.origin) throw Problem("soa not at origin")
253 if (type == "SOA" && z.records.any {
254 it.type == "SOA"
255 }) throw Problem("duplicate soa")
256 if (z.records.any {
257 it.owner == owner && it.type == type && it.data == data
258 }) throw Problem("duplicate record")
259 if (z.records.any {
260 it.owner == owner && (type == "CNAME" || it.type == "CNAME")
261 }) throw Problem("cname conflict $owner")
262 z.records.add(Record(owner, seconds, type, data))
263 }
264 fun finishZones() {
265 val kept = mutableListOf<Zone>()
266 for (z in zones) {
267 if (z.records.any {
268 it.type == "SOA"
269 }) {
270 out.add("zone ${z.origin} at ${address(z.server)} records=${z.records.size}")
271 kept.add(z)
272 }
273 else out.add("error line ${z.line}: zone ${z.origin} at ${address(z.server)} has no soa")
274 }

278 }
279 fun records(z: Zone, n: String, t: String) = z.records.filter {
280 it.owner == n && it.type == t
281 }.sortedWith {
282 a, b -> dataOrder(a.data, b.data)
283 }
284 fun authoritative(server: Long, n: String, t: String, cut: String): Response {
285 if (server in down || zones.none {
286 it.server == server
287 }) return Response(6)
288 val z = zones.filter {
289 it.server == server && below(n, it.origin)
290 }.maxByOrNull {
291 labels(it.origin).size
292 } ?: return Response(5)
293 if (labels(z.origin).size < labels(cut).size) return Response(5)
294 for (d in ancestors(n).asReversed().filter {
295 it != z.origin && below(it, z.origin)
296 }) {
297 val ns = records(z, d, "NS")
298 if (ns.isNotEmpty()) return Response(2, authority = ns, additional = ns.flatMap {
299 records(z, (it.data as Target).name, "A")
300 }, cut = d)
301 }
302 fun exists(x: String) = z.records.any {
303 below(it.owner, x)
304 }
305 val soa = records(z, z.origin, "SOA")
306 var owner = n
307 if (z.records.none {
308 it.owner == n
309 }) {
310 if (exists(n)) return Response(3, authority = soa)
311 val e = ancestors(n).first {
312 it != n && exists(it)
313 }
314 owner = if (e == ".") "*." else "*.$e"
315 if (z.records.none {
316 it.owner == owner
317 }) return Response(4, authority = soa)
318 }
319 var answer = records(z, owner, t)
320 var kind = 0
321 if (answer.isEmpty() && t != "CNAME") {
322 answer = records(z, owner, "CNAME")
323 kind = 1
324 }
325 if (answer.isEmpty()) return Response(3, authority = soa)
326 return Response(kind, answer.map {
327 it.copy(owner = n)
328 })
329 }
330 fun ask(a: Long, n: String, t: String, cut: String): Response {
331 val r = authoritative(a, n, t, cut)
332 queries.getOrPut(a) {
333 IntArray(7)
334 }[r.kind]++
335 return r
336 }
337 fun trace(depth: Int, text: String) {
338 out.add(" ".repeat(2*(depth+1))+text)
339 }
340 fun positive(records: List<Record>): Int {
341 val seconds = records.minOf {
342 it.ttl
343 }.coerceAtLeast(settings[0]).coerceAtMost(settings[1])
344 cache[records[0].owner to records[0].type] = Cache(0, records, clock+seconds)
345 return seconds
346 }
347 fun cachedRecords(key: Pair<String, String>): List<Record>? = cache[key]?.takeIf {
348 it.kind == 0
349 }?.let {
350 e -> e.records.map {
351 it.copy(ttl = e.expiry-clock)
352 }
353 }
354 fun queryServers(n: String, t: String, cut: String, servers: List<String>?, depth: Int, stack: MutableList<Pair<String, String>>): Response? {
355 if (servers == null) {
356 for (a in roots) {
357 val r = ask(a, n, t, cut)
358 trace(depth, "ASK ${address(a)} $n $t -> ${r.text()}")
359 if (r.kind < 5) return r
360 }
361 } else for (host in servers.sortedWith {
362 a, b -> nameOrder(a, b)
363 }) {
364 val records = cachedRecords(host to "A") ?: if ((host to "A") in stack) {
365 trace(depth, "GLUELESS $host -> LOOP")
366 emptyList()
367 } else {
368 trace(depth, "GLUELESS $host")
369 val r = resolve(host, "A", depth+1, stack)
370 if (r.status == "NOERROR") r.answer else emptyList()
371 }
372 val addresses = records.mapNotNull {
373 (it.data as? A)?.address
374 }.toSortedSet()
375 for (a in addresses) {
376 val r = ask(a, n, t, cut)
377 trace(depth, "ASK ${address(a)} $n $t -> ${r.text()}")
378 if (r.kind < 5) return r
379 }
380 }
381 return null
382 }
383 fun lookup(n: String, t: String, depth: Int, stack: MutableList<Pair<String, String>>): Response {
384 cache[n to t]?.let {
385 e -> hits++
386 trace(depth, "CACHE HIT $n $t -> ${if (e.kind == 0) "ANSWER" else kinds[e.kind]}")
387 return Response(e.kind, if (e.kind == 0) cachedRecords(n to t)!! else emptyList())
388 }
389 if (cache.any {
390 (key, e) -> e.kind == 4 && below(n, key.first)
391 }) {
392 hits++
393 trace(depth, "CACHE HIT $n $t -> NXDOMAIN")
394 return Response(4)
395 }
396 if (t != "CNAME") cachedRecords(n to "CNAME")?.let {
397 hits++
398 val r = Response(1, it)
399 trace(depth, "CACHE HIT $n $t -> ${r.text()}")
400 return r
401 }
402 val referral = cache.entries.filter {
403 it.key.second == "NS" && it.value.kind == 0 && below(n, it.key.first)
404 }.maxByOrNull {
405 labels(it.key.first).size
406 }
407 var cut = referral?.key?.first ?: "."
408 var servers = referral?.value?.records?.map {
409 dataText(it.data)
410 }
411 if (referral != null) {
412 hits++
413 trace(depth, "CACHE HIT $n $t -> REFERRAL $cut")
414 }
415 while (true) {
416 val r = queryServers(n, t, cut, servers, depth, stack) ?: throw Problem("SERVFAIL no server for $cut")
417 when (r.kind) {
418 2 -> {
419 positive(r.authority)
420 r.additional.groupBy {
421 it.owner
422 }.values.forEach {
423 positive(it)
424 }
425 cut = r.cut
426 servers = r.authority.map {
427 dataText(it.data)
428 }
429 }
430 0, 1 -> {
431 val seconds = positive(r.answer)
432 return r.copy(answer = r.answer.map {
433 it.copy(ttl = seconds)
434 })
435 }
436 3, 4 -> {
437 val soa = r.authority[0]
438 val seconds = minOf(soa.ttl, (soa.data as Soa).numbers[4], settings[2])
439 cache[n to t] = Cache(r.kind, emptyList(), clock+seconds)
440 return r
441 }
442 else -> error("unexpected response")
443 }
444 }
445 }
446 fun resolve(n: String, t: String, depth: Int, stack: MutableList<Pair<String, String>>): Result {
447 stack.add(n to t)
448 val chain = mutableListOf<Record>()
449 val seen = mutableSetOf(n)
450 var current = n
451 var result: Result
452 while (true) {
453 val r = try {
454 lookup(current, t, depth, stack)
455 } catch (p: Problem) {
456 result = Result(p.reason, emptyList())
457 break
458 }
459 if (r.kind == 1) {
460 val target = dataText(r.answer[0].data)
461 chain.addAll(r.answer)
462 if (target in seen) {
463 result = Result("SERVFAIL cname loop", emptyList())
464 break
465 }
466 if (chain.size == 9) {
467 result = Result("SERVFAIL cname chain too long", emptyList())
468 break
469 }
470 seen.add(target)
471 current = target
472 } else {
473 if (r.kind == 0) chain.addAll(r.answer)
474 result = Result(if (r.kind == 0) "NOERROR" else kinds[r.kind], chain.toList())
475 break
476 }
477 }
478 stack.removeAt(stack.lastIndex)
479 resolutions[when (result.status) {
480 "NOERROR" -> 0
481 "NODATA" -> 1
482 "NXDOMAIN" -> 2
483 else -> 3
484 }]++
485 return result
486 }
487 fun outputResolution(n: String, t: String): Result {
488 val r = resolve(n, t, 0, mutableListOf())
489 r.answer.forEach {
490 out.add("answer ${it.text()}")
491 }
492 out.add("status ${r.status}")
493 return r
494 }
495 fun hostAddresses(n: String): List<String> = buildList {
496 for (t in listOf("A", "AAAA")) {
497 val r = resolve(n, t, 0, mutableListOf())
498 if (r.status == "NOERROR") addAll(r.answer.filter {
499 it.type == t
500 }.map {
501 dataText(it.data)
502 })
503 }
504 }
505 fun command(w: List<String>) {
506 when (w[0]) {
507 "RESOLVE" -> {
508 val n = name(w[1], true)
509 checkedType(w[2])
510 out.add("resolve $n ${w[2]}")
511 outputResolution(n, w[2])
512 }
513 "REVERSE" -> {
514 val a: String
515 val n: String
516 if (':' in w[1]) {
517 val groups = ipv6(w[1]) ?: throw Problem("bad address ${w[1]}")
518 a = address6(groups)
519 n = groups.joinToString("") {
520 "%04x".format(it)
521 }.reversed().map(Char::toString).joinToString(".")+".ip6.arpa."
522 }
523 else {
524 val value = checkedIp(w[1])
525 a = address(value)
526 n = a.split('.').reversed().joinToString(".")+".in-addr.arpa."
527 }
528 out.add("reverse $a $n")
529 outputResolution(n, "PTR")
530 }
531 "MX" -> {
532 val n = name(w[1], true)
533 out.add("mx $n")
534 val r = outputResolution(n, "MX")
535 if (r.status == "NOERROR") for (record in r.answer) {
536 val data = record.data as? Mx ?: continue
537 if (data.host == ".") out.add("exchange ${data.preference} . -> null")
538 else {
539 val addresses = hostAddresses(data.host)
540 out.add("exchange ${data.preference} ${data.host} -> ${addresses.takeIf { it.isNotEmpty() }?.joinToString(" ") ?: "no address"}")
541 }
542 } else if (r.status == "NODATA") {
543 val host = r.answer.lastOrNull {
544 it.type == "CNAME"
545 }?.let {
546 dataText(it.data)
547 } ?: n
548 val addresses = hostAddresses(host)
549 out.add("implicit $host -> ${addresses.takeIf { it.isNotEmpty() }?.joinToString(" ") ?: "no address"}")
550 }
551 }
552 "QUERY" -> {
553 val a = checkedIp(w[1])
554 val n = name(w[2], true)
555 checkedType(w[3])
556 val r = ask(a, n, w[3], ".")
557 out.add("query ${address(a)} $n ${w[3]} -> ${r.text()}")
558 val sections = when (r.kind) {
559 0, 1 -> listOf("answer" to r.answer)
560 2 -> listOf("authority" to r.authority, "additional" to r.additional)
561 3, 4 -> listOf("authority" to r.authority)
562 else -> emptyList()
563 }
564 for ((section, records) in sections) records.forEach {
565 out.add("$section ${it.text()}")
566 }
567 }
568 "DUMP" -> {
569 val a = checkedIp(w[1])
570 val n = name(w[2], false)
571 val z = zones.firstOrNull {
572 it.server == a && it.origin == n
573 }
574 if (a in down || zones.none {
575 it.server == a
576 }) out.add("dump ${address(a)} $n -> TIMEOUT")
577 else if (z == null) out.add("dump ${address(a)} $n -> REFUSED")
578 else {
579 out.add("dump ${address(a)} $n records=${z.records.size}")
580 z.records.sortedWith(recordOrder).forEach {
581 out.add(" ${it.text()}")
582 }
583 }
584 }
585 "UP", "DOWN" -> {
586 val a = checkedIp(w[1])
587 if (w[0] == "DOWN") {
588 if (a in down) throw Problem("server ${w[1]} already down")
589 down.add(a)
590 }
591 else if (!down.remove(a)) throw Problem("server ${w[1]} not down")
592 out.add("${w[0].lowercase()} ${address(a)}")
593 }
594 "SET" -> {
595 val i = listOf("MINTTL", "MAXTTL", "NEGTTL").indexOf(w[1])
596 if (i < 0) throw Problem("bad setting ${w[1]}")
597 val n = integer(w[2])?.takeIf {
598 it in 1..604800
599 } ?: throw Problem("bad value ${w[2]}")
600 val changed = settings.copyOf()
601 changed[i] = n
602 if (changed[0] > changed[1]) throw Problem("minttl above maxttl")
603 settings[i] = n
604 out.add("set ${w[1]}=$n")
605 }
606 "TIME" -> {
607 val n = integer(w[1])?.takeIf {
608 it in 1..1000000
609 } ?: throw Problem("bad time ${w[1]}")
610 clock += n
611 val old = cache.size
612 cache.entries.removeIf {
613 it.value.expiry <= clock
614 }
615 out.add("time now=$clock expired=${old-cache.size}")
616 }
617 "CACHE", "FLUSH" -> {
618 val parent = if (w.size == 2) name(w[1], false) else null
619 val keys = cache.keys.filter {
620 parent == null || below(it.first, parent)
621 }.sortedWith {
622 a, b -> nameOrder(a.first, b.first).takeIf {
623 it != 0
624 } ?: types.indexOf(a.second).compareTo(types.indexOf(b.second))
625 }
626 if (w[0] == "FLUSH") {
627 out.add("flush removed=${keys.size}")
628 keys.forEach {
629 cache.remove(it)
630 }
631 }
632 else {
633 out.add("cache entries=${keys.size}")
634 for (key in keys) {
635 val e = cache.getValue(key)
636 if (e.kind == 0) cachedRecords(key)!!.forEach {
637 out.add(" ${it.text()}")
638 } else out.add(" ${key.first} ${e.expiry-clock} ${key.second} ${kinds[e.kind]}")
639 }
640 }
641 }
642 else -> {
643 out.add("stats queries=${queries.values.sumOf { it.sum() }} hits=$hits")
644 for ((a, c) in queries) out.add(" ${address(a)} sent=${c.sum()} answer=${c[0]} cname=${c[1]} referral=${c[2]} nodata=${c[3]} nxdomain=${c[4]} lame=${c[5]} timeout=${c[6]}")
645 val c = resolutions
646 out.add("resolutions noerror=${c[0]} nodata=${c[1]} nxdomain=${c[2]} servfail=${c[3]}")
647 }
648 }
649 }
650}
651fun main() {
652 val resolver = Resolver()
653 var zones = true
654 generateSequence(::readLine).forEachIndexed {
655 i, line ->
656 if (line == "END") return@forEachIndexed
657 if (line == "END ZONES") {
658 resolver.finishZones()
659 zones = false
660 }
661 else try {
662 if (zones) resolver.zoneLine(line.split(' '), i+1) else resolver.command(line.split(' '))
663 }
664 catch (p: Problem) {
665 resolver.out.add("error line ${i+1}: ${p.reason}")
666 }
667 }
668 resolver.out.forEach(::println)
669}
