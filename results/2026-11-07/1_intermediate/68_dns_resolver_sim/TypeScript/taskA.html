54const compareText = (a: string, b: string): number =>
55 a === b ? 0 : a < b ? -1 : 1;
56function nameOrder(a: string, b: string): number {
57 const left = labels(a).reverse();
58 const right = labels(b).reverse();
59 for (let i = 0; i < Math.min(left.length, right.length); i++) {
60 const order = compareText(left[i], right[i]);
61 if (order !== 0) return order;
62 }
63 return left.length - right.length;
64}
65function parseName(
66 word: string,
67 wildcard = false,
68 origin: string | null = null,
69): string {
70 if (origin !== null && word === "@") return origin;
71 if (word === ".") return ".";
72 const absolute = word.endsWith(".");
73 const parts = (absolute ? word.slice(0, -1) : word).split(".");
74 if (
75 parts.some(
76 (part, i) =>
77 !(wildcard && i === 0 && part === "*") &&
78 !/^[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?$/.test(part),
79 )
80 )
81 throw new Error(`bad name ${word}`);
82 const name = parts.join(".").toLowerCase() + ".";
83 return origin !== null && !absolute && origin !== "." ? name + origin : name;
84}
85function ipv4(word: string): number {
86 const parts = word.split(".");
87 if (
88 parts.length !== 4 ||
89 parts.some(
90 (part) => !/^(?:0|[1-9][0-9]{0,2})$/.test(part) || Number(part) > 255,
91 )
92 )
93 throw new Error(`bad address ${word}`);
94 return parts.reduce((value, part) => value * 256 + Number(part), 0);
95}
96function addressText(value: number): string {
97 return [24, 16, 8, 0]
98 .map((shift) => Math.floor(value / 2 ** shift) % 256)
99 .join(".");
100}
101function ipv6(word: string): number[] {
102 const halves = word.split("::");
103 if (halves.length > 2) throw new Error(`bad address ${word}`);
104 const left = halves[0] === "" ? [] : halves[0].split(":");
105 const right =
106 halves.length === 2 && halves[1] !== "" ? halves[1].split(":") : [];
107 if (
108 [...left, ...right].some((group) => !/^[0-9A-Fa-f]{1,4}$/.test(group)) ||
109 (halves.length === 1 ? left.length !== 8 : left.length + right.length > 7)
110 )
111 throw new Error(`bad address ${word}`);
112 return [
113 ...left.map((group) => parseInt(group, 16)),
114 ...Array<number>(
115 halves.length === 2 ? 8 - left.length - right.length : 0,
116 ).fill(0),
117 ...right.map((group) => parseInt(group, 16)),
118 ];
119}
120function ipv6Text(groups: number[]): string {
121 let best = -1;
122 let length = 0;
123 for (let start = 0; start < 8;) {
124 if (groups[start] !== 0) {
125 start++;
126 continue;
127 }
128 let end = start;
129 while (end < 8 && groups[end] === 0) end++;
130 if (end - start >= 2 && end - start > length) {
131 best = start;
132 length = end - start;
133 }
134 start = end;
135 }
136 const words = groups.map((group) => group.toString(16));
137 return best < 0
138 ? words.join(":")
139 : words.slice(0, best).join(":") +
140 "::" +
141 words.slice(best + length).join(":");
142}
143function ttl(word: string): number {
144 let result = 0;
145 if (integerWord.test(word)) result = Number(word);
146 else {
147 const units = "wdhms";
148 const seconds = [604800, 86400, 3600, 60, 1];
149 let position = 0;
150 let previous = -1;
151 for (const match of word.matchAll(/(0|[1-9][0-9]{0,2})([wdhms])/gi)) {
152 const unit = units.indexOf(match[2].toLowerCase());
153 if (match.index !== position || unit <= previous)
154 throw new Error(`bad ttl ${word}`);
155 result += Number(match[1]) * seconds[unit];
156 previous = unit;
157 position += match[0].length;
158 }
159 if (position !== word.length || position === 0)
160 throw new Error(`bad ttl ${word}`);
161 }
162 if (result < 1 || result > 604800) throw new Error(`bad ttl ${word}`);
163 return result;
164}
165function checkType(type: string): void {
166 if (!types.includes(type)) throw new Error(`bad type ${type}`);
167}
168function dataOrder(a: DnsRecord, b: DnsRecord): number {
169 if (a.type === "A") return a.address! - b.address!;
170 if (a.type === "AAAA") {
171 for (let i = 0; i < 8; i++)
172 if (a.groups![i] !== b.groups![i]) return a.groups![i] - b.groups![i];
173 return 0;
174 }
175 if (["NS", "CNAME", "PTR"].includes(a.type))
176 return nameOrder(a.data[0], b.data[0]);
177 if (a.type === "MX")
178 return a.preference! - b.preference! || nameOrder(a.data[1], b.data[1]);
179 return compareText(a.data.join(" "), b.data.join(" "));
180}
181function recordOrder(a: DnsRecord, b: DnsRecord): number {
182 return (
183 nameOrder(a.owner, b.owner) ||
184 types.indexOf(a.type) - types.indexOf(b.type) ||
185 dataOrder(a, b)
186 );
187}
188function recordText(record: DnsRecord): string {
189 return `${record.owner} ${record.ttl} ${record.type} ${record.data.join(" ")}`;

226 zoneLine(words: string[], line: number): void {
227 if (words[0] === "ROOT") {
228 const addresses: number[] = [];
229 for (const word of words.slice(1)) {
230 const address = ipv4(word);
231 if (this.roots.includes(address) || addresses.includes(address))
232 throw new Error(`duplicate root ${word}`);
233 addresses.push(address);
234 }
235 this.roots.push(...addresses);
236 return;
237 }
238 if (words[0] === "ZONE") {
239 this.current = null;
240 const origin = parseName(words[1]);
241 const server = ipv4(words[2]);
242 if (
243 this.zones.some(
244 (zone) => zone.origin === origin && zone.server === server,
245 )
246 )
247 throw new Error(`duplicate zone ${origin} at ${words[2]}`);
248 this.current = { origin, server, line, records: [] };
249 this.zones.push(this.current);
250 return;
251 }
252 const zone = this.current;
253 if (zone === null) throw new Error("no current zone");
254 const owner = parseName(words[0], true, zone.origin);
255 if (!below(owner, zone.origin)) throw new Error("out of zone " + owner);
256 const seconds = ttl(words[1]);
257 const type = words[2];
258 checkType(type);
259 const fields = words.slice(3);
260 const required = type === "MX" ? 2 : type === "SOA" ? 7 : 1;
261 if (type !== "TXT" && fields.length !== required)
262 throw new Error("wrong field count");
263 const record: DnsRecord = { owner, ttl: seconds, type, data: [] };
264 const name = (word: string): string => parseName(word, false, zone.origin);
265 if (type === "A") {
266 record.address = ipv4(fields[0]);
267 record.data = [addressText(record.address)];
268 } else if (type === "AAAA") {
269 record.groups = ipv6(fields[0]);
270 record.data = [ipv6Text(record.groups)];
271 } else if (["NS", "CNAME", "PTR"].includes(type))
272 record.data = [name(fields[0])];
273 else if (type === "MX") {
274 if (!integerWord.test(fields[0]) || Number(fields[0]) > 65535)
275 throw new Error("bad preference " + fields[0]);
276 record.preference = Number(fields[0]);
277 record.data = [String(record.preference), name(fields[1])];
278 } else if (type === "TXT") {
279 const text = fields.join(" ");
280 if (
281 text.length < 2 ||
282 !text.startsWith('"') ||
283 !text.endsWith('"') ||
284 text.slice(1, -1).includes('"')
285 )
286 throw new Error("bad text " + text);
287 record.data = [text];
288 } else {
289 record.data = [name(fields[0]), name(fields[1])];
290 for (const field of fields.slice(2, 6)) {
291 if (!integerWord.test(field)) throw new Error("bad number " + field);
292 record.data.push(String(Number(field)));
293 }
294 if (
295 !integerWord.test(fields[6]) ||
296 Number(fields[6]) < 1 ||
297 Number(fields[6]) > 86400
298 )
299 throw new Error("bad minimum " + fields[6]);
300 record.minimum = Number(fields[6]);
301 record.data.push(String(record.minimum));
302 }
303 if (type === "NS" && labels(owner)[0] === "*")
304 throw new Error("ns at wildcard");
305 if (type === "SOA" && owner !== zone.origin)
306 throw new Error("soa not at origin");
307 if (type === "SOA" && zone.records.some((record) => record.type === "SOA"))
308 throw new Error("duplicate soa");
309 if (
310 zone.records.some(
311 (previous) =>
312 previous.owner === owner &&
313 previous.type === type &&
314 JSON.stringify(previous.data) === JSON.stringify(record.data),
315 )
316 )
317 throw new Error("duplicate record");
318 if (
319 zone.records.some(
320 (previous) =>
321 previous.owner === owner &&
322 (type === "CNAME" || previous.type === "CNAME"),
323 )
324 )
325 throw new Error("cname conflict " + owner);
326 zone.records.push(record);
327 }

347 authoritative(
348 server: number,
349 name: string,
350 type: string,
351 cut: string,
352 ): Reply {
353 if (
354 this.down.has(server) ||
355 !this.zones.some((zone) => zone.server === server)
356 )
357 return reply("TIMEOUT");
358 const candidates = this.zones
359 .filter((zone) => zone.server === server && below(name, zone.origin))
360 .sort((a, b) => labels(b.origin).length - labels(a.origin).length);
361 const zone = candidates[0];
362 if (zone === undefined || labels(zone.origin).length < labels(cut).length)
363 return reply("LAME");
364 const soa = zone.records.find((record) => record.type === "SOA")!;
365 const delegation = ancestors(name)
366 .reverse()
367 .find(
368 (ancestor) =>
369 ancestor !== zone.origin &&
370 below(ancestor, zone.origin) &&
371 zone.records.some(
372 (record) => record.owner === ancestor && record.type === "NS",
373 ),
374 );
375 if (delegation !== undefined) {
376 const authority = zone.records
377 .filter((record) => record.owner === delegation && record.type === "NS")
378 .sort(dataOrder);
379 const additional = authority.flatMap((record) =>
380 zone.records
381 .filter((glue) => glue.owner === record.data[0] && glue.type === "A")
382 .sort(dataOrder),
383 );
384 return {
385 ...reply("REFERRAL", [], authority, additional),
386 cut: delegation,
387 };
388 }
389 const exists = (candidate: string): boolean =>
390 zone.records.some((record) => below(record.owner, candidate));
391 let records = zone.records.filter((record) => record.owner === name);
392 if (records.length === 0) {
393 if (exists(name)) return reply("NODATA", [], [soa]);
394 const ancestor = ancestors(name).slice(1).find(exists)!;
395 const wildcard = ancestor === "." ? "*." : "*." + ancestor;
396 records = zone.records.filter((record) => record.owner === wildcard);
397 if (records.length === 0) return reply("NXDOMAIN", [], [soa]);
398 }
399 const answers = records
400 .filter((record) => record.type === type)
401 .sort(dataOrder)
402 .map((record) => ({ ...record, owner: name }));
403 if (answers.length > 0) return reply("ANSWER", answers);
404 const cname = records.find((record) => record.type === "CNAME");
405 if (type !== "CNAME" && cname !== undefined)
406 return reply("CNAME", [{ ...cname, owner: name }]);
407 return reply("NODATA", [], [soa]);
408 }
409 ask(server: number, name: string, type: string, cut: string): Reply {
410 const response = this.authoritative(server, name, type, cut);
411 const counts =
412 this.queries.get(server) ?? Array<number>(kinds.length).fill(0);
413 counts[kinds.indexOf(response.kind)]++;
414 this.queries.set(server, counts);
415 return response;
416 }
417 positive(records: DnsRecord[]): DnsRecord[] {
418 const seconds = Math.min(
419 this.settings.MAXTTL,
420 Math.max(
421 this.settings.MINTTL,
422 Math.min(...records.map((record) => record.ttl)),
423 ),
424 );
425 this.cache.set(cacheKey(records[0].owner, records[0].type), {
426 name: records[0].owner,
427 type: records[0].type,
428 kind: "POS",
429 records: [...records].sort(dataOrder),
430 expiry: this.clock + seconds,
431 });
432 return records.map((record) => ({ ...record, ttl: seconds }));
433 }

437 queryServers(
438 name: string,
439 type: string,
440 cut: string,
441 nameservers: string[] | null,
442 depth: number,
443 stack: string[],
444 ): Reply | null {
445 const query = (server: number): Reply | null => {
446 const response = this.ask(server, name, type, cut);
447 this.trace(
448 depth,
449 `ASK ${addressText(server)} ${name} ${type} -> ${replyText(response)}`,
450 );
451 return response.kind === "LAME" || response.kind === "TIMEOUT"
452 ? null
453 : response;
454 };
455 if (nameservers === null) {
456 for (const server of this.roots) {
457 const response = query(server);
458 if (response !== null) return response;
459 }
460 return null;

465 if (cached?.kind === "POS")
466 addresses = cached.records.map((record) => record.address!);
467 else if (stack.includes(cacheKey(host, "A")))

484 lookup(name: string, type: string, depth: number, stack: string[]): Reply {
485 const hit = (response: Reply, text: string): Reply => {
486 this.hits++;
487 this.trace(depth, `CACHE HIT ${name} ${type} -> ${text}`);
488 return response;
489 };
490 const exact = this.cache.get(cacheKey(name, type));
491 if (exact !== undefined)
492 return hit(
493 exact.kind === "POS"
494 ? reply(
495 "ANSWER",
496 exact.records.map((record) => ({
497 ...record,
498 ttl: exact.expiry - this.clock,
499 })),
500 )
501 : reply(exact.kind),
502 exact.kind === "POS" ? "ANSWER" : exact.kind,
503 );
504 if (
505 [...this.cache.values()].some(
506 (entry) => entry.kind === "NXDOMAIN" && below(name, entry.name),
507 )
508 )
509 return hit(reply("NXDOMAIN"), "NXDOMAIN");
510 const cname = this.cache.get(cacheKey(name, "CNAME"));
511 if (type !== "CNAME" && cname?.kind === "POS")
512 return hit(
513 reply(
514 "CNAME",
515 cname.records.map((record) => ({
516 ...record,
517 ttl: cname.expiry - this.clock,
518 })),
519 ),
520 `CNAME ${cname.records[0].data[0]}`,
521 );
522 const ns = ancestors(name)
523 .map((ancestor) => this.cache.get(cacheKey(ancestor, "NS")))
524 .find((entry) => entry?.kind === "POS");
525 let cut = ns?.name ?? ".";
526 let nameservers =
527 ns === undefined ? null : ns.records.map((record) => record.data[0]);
528 if (ns !== undefined) {
529 this.hits++;
530 this.trace(depth, `CACHE HIT ${name} ${type} -> REFERRAL ${cut}`);
531 }
532 while (true) {
533 const response = this.queryServers(
534 name,
535 type,
536 cut,
537 nameservers,
538 depth,
539 stack,
540 );
541 if (response === null)
542 return { ...reply("SERVFAIL"), reason: "no server for " + cut };
543 if (response.kind === "REFERRAL") {
544 this.positive(response.authority);
545 const groups = new Map<string, DnsRecord[]>();
546 for (const record of response.additional) {
547 const group = groups.get(record.owner) ?? [];
548 group.push(record);
549 groups.set(record.owner, group);
550 }
551 for (const group of groups.values()) this.positive(group);
552 cut = response.cut!;
553 nameservers = response.authority.map((record) => record.data[0]);
554 } else if (response.kind === "ANSWER" || response.kind === "CNAME")
555 return { ...response, answer: this.positive(response.answer) };
556 else {
557 const soa = response.authority[0];
558 const seconds = Math.min(soa.ttl, soa.minimum!, this.settings.NEGTTL);
559 this.cache.set(cacheKey(name, type), {
560 name,
561 type,
562 kind: response.kind,
563 records: [],
564 expiry: this.clock + seconds,
565 });
566 return response;
567 }

570 resolve(
571 name: string,
572 type: string,
573 depth = 0,
574 stack: string[] = [],
575 ): Resolution {
576 stack.push(cacheKey(name, type));
577 const chain: DnsRecord[] = [];
578 const seen = new Set([name]);
579 let current = name;
580 let status = "";
581 let answer: DnsRecord[] = [];
582 while (true) {
583 const result = this.lookup(current, type, depth, stack);
584 if (result.kind === "CNAME") {
585 const record = result.answer[0];
586 chain.push(record);
587 const target = record.data[0];
588 if (seen.has(target)) {
589 status = "SERVFAIL cname loop";
590 break;
591 }
592 if (chain.length === 9) {
593 status = "SERVFAIL cname chain too long";
594 break;
595 }
596 seen.add(target);
597 current = target;
598 } else {
599 status =
600 result.kind === "ANSWER"
601 ? "NOERROR"
602 : result.kind === "SERVFAIL"
603 ? "SERVFAIL " + result.reason
604 : result.kind;
605 if (!status.startsWith("SERVFAIL"))
606 answer = [
607 ...chain,
608 ...(result.kind === "ANSWER" ? result.answer : []),
609 ];
610 break;
611 }
612 }
613 stack.pop();
614 this.resolutions[status.split(" ")[0]]++;
615 return { status, answer };
616 }

635 command(words: string[]): void {
636 const command = words[0];
637 if (command === "RESOLVE" || command === "MX") {
638 const name = parseName(words[1], true);
639 const type = command === "MX" ? "MX" : words[2];
640 checkType(type);
641 this.output.push(
642 command === "MX" ? `mx ${name}` : `resolve ${name} ${type}`,
643 );
644 const result = this.resolve(name, type);
645 this.emitResolution(result);
646 if (command === "MX") {
647 if (result.status === "NOERROR")
648 for (const record of result.answer.filter(
649 (record) => record.type === "MX",
650 )) {
651 const host = record.data[1];
652 this.output.push(
653 host === "."
654 ? `exchange ${record.preference} . -> null`
655 : `exchange ${record.preference} ${host} -> ${this.addresses(host)}`,
656 );
657 }
658 else if (result.status === "NODATA") {
659 const aliases = result.answer.filter(
660 (record) => record.type === "CNAME",
661 );
662 const host =
663 aliases.length === 0 ? name : aliases[aliases.length - 1].data[0];
664 this.output.push(`implicit ${host} -> ${this.addresses(host)}`);
665 }
666 }
667 } else if (command === "REVERSE") {
668 let address: string;
669 let name: string;
670 if (words[1].includes(":")) {
671 const groups = ipv6(words[1]);
672 address = ipv6Text(groups);
673 name =
674 [
675 ...groups
676 .map((group) => group.toString(16).padStart(4, "0"))
677 .join(""),
678 ]
679 .reverse()
680 .join(".") + ".ip6.arpa.";
681 } else {
682 address = addressText(ipv4(words[1]));
683 name = address.split(".").reverse().join(".") + ".in-addr.arpa.";
684 }
685 this.output.push(`reverse ${address} ${name}`);
686 this.emitResolution(this.resolve(name, "PTR"));
687 } else if (command === "QUERY" || command === "DUMP") {
688 const server = ipv4(words[1]);
689 const name = parseName(words[2], command === "QUERY");
690 if (command === "QUERY") {
691 checkType(words[3]);
692 const result = this.ask(server, name, words[3], ".");
693 this.output.push(
694 `query ${addressText(server)} ${name} ${words[3]} -> ${replyText(result)}`,
695 );
696 for (const record of result.answer)
697 this.output.push("answer " + recordText(record));
698 for (const record of result.authority)
699 this.output.push("authority " + recordText(record));
700 for (const record of result.additional)
701 this.output.push("additional " + recordText(record));
702 } else {
703 const zone = this.zones.find(
704 (zone) => zone.server === server && zone.origin === name,
705 );
706 if (
707 this.down.has(server) ||
708 !this.zones.some((zone) => zone.server === server)
709 )
710 this.output.push(`dump ${addressText(server)} ${name} -> TIMEOUT`);
711 else if (zone === undefined)
712 this.output.push(`dump ${addressText(server)} ${name} -> REFUSED`);
713 else {
714 this.output.push(
715 `dump ${addressText(server)} ${name} records=${zone.records.length}`,
716 );
717 for (const record of [...zone.records].sort(recordOrder))
718 this.output.push(" " + recordText(record));
719 }
720 }
721 } else if (command === "DOWN" || command === "UP") {
722 const address = ipv4(words[1]);
723 if (command === "DOWN" && this.down.has(address))
724 throw new Error(`server ${words[1]} already down`);
725 if (command === "UP" && !this.down.has(address))
726 throw new Error(`server ${words[1]} not down`);
727 if (command === "DOWN") this.down.add(address);
728 else this.down.delete(address);
729 this.output.push(command.toLowerCase() + " " + addressText(address));
730 } else if (command === "SET") {
731 if (!Object.hasOwn(this.settings, words[1]))
732 throw new Error("bad setting " + words[1]);
733 if (
734 !integerWord.test(words[2]) ||
735 Number(words[2]) < 1 ||
736 Number(words[2]) > 604800
737 )
738 throw new Error("bad value " + words[2]);
739 const value = Number(words[2]);
740 const minimum = words[1] === "MINTTL" ? value : this.settings.MINTTL;
741 const maximum = words[1] === "MAXTTL" ? value : this.settings.MAXTTL;
742 if (minimum > maximum) throw new Error("minttl above maxttl");
743 this.settings[words[1]] = value;
744 this.output.push(`set ${words[1]}=${value}`);
745 } else if (command === "TIME") {
746 if (
747 !integerWord.test(words[1]) ||
748 Number(words[1]) < 1 ||
749 Number(words[1]) > 1000000
750 )
751 throw new Error("bad time " + words[1]);
752 this.clock += Number(words[1]);
753 let expired = 0;
754 for (const [key, entry] of this.cache)
755 if (entry.expiry <= this.clock) {
756 this.cache.delete(key);
757 expired++;
758 }
759 this.output.push(`time now=${this.clock} expired=${expired}`);
760 } else if (command === "CACHE" || command === "FLUSH") {
761 const name = words.length === 1 ? "." : parseName(words[1]);
762 const entries = [...this.cache.values()]
763 .filter((entry) => below(entry.name, name))
764 .sort(
765 (a, b) =>
766 nameOrder(a.name, b.name) ||
767 types.indexOf(a.type) - types.indexOf(b.type),
768 );
769 if (command === "FLUSH") {
770 for (const entry of entries)
771 this.cache.delete(cacheKey(entry.name, entry.type));
772 this.output.push(`flush removed=${entries.length}`);
773 } else {
774 this.output.push(`cache entries=${entries.length}`);
775 for (const entry of entries) {
776 if (entry.kind === "POS")
777 for (const record of entry.records)
778 this.output.push(
779 " " +
780 recordText({ ...record, ttl: entry.expiry - this.clock }),
781 );
782 else
783 this.output.push(
784 ` ${entry.name} ${entry.expiry - this.clock} ${entry.type} ${entry.kind}`,
785 );
786 }
787 }
788 } else {
789 const queryCount = [...this.queries.values()].reduce(
790 (sum, counts) => sum + counts.reduce((a, b) => a + b, 0),
791 0,
792 );
793 this.output.push(`stats queries=${queryCount} hits=${this.hits}`);
794 for (const [server, counts] of [...this.queries].sort(
795 (a, b) => a[0] - b[0],
796 ))
797 this.output.push(
798 ` ${addressText(server)} sent=${counts.reduce((a, b) => a + b, 0)} ` +
799 kinds
800 .map((kind, index) => `${kind.toLowerCase()}=${counts[index]}`)
801 .join(" "),
802 );
803 this.output.push(
804 `resolutions noerror=${this.resolutions.NOERROR} nodata=${this.resolutions.NODATA} nxdomain=${this.resolutions.NXDOMAIN} servfail=${this.resolutions.SERVFAIL}`,
805 );
806 }
807 }
808 run(lines: string[]): void {
809 let position = 0;
810 while (lines[position] !== "END ZONES") {
811 try {
812 this.zoneLine(lines[position].split(" "), position + 1);
813 } catch (error) {
814 this.output.push(
815 `error line ${position + 1}: ${(error as Error).message}`,
816 );
817 }
818 position++;
819 }
820 position++;
821 this.finishZones();
822 while (lines[position] !== "END") {
823 try {
824 this.command(lines[position].split(" "));
825 } catch (error) {
826 this.output.push(
827 `error line ${position + 1}: ${(error as Error).message}`,
828 );
829 }
830 position++;
831 }
832 }
