110static void append(Buffer *b, const char *text) {
111 size_t n = strlen(text);
112 if (b->length + n + 1 > b->capacity) {
113 b->capacity = (b->length + n + 1) * 2;
114 b->data = resize(b->data, b->capacity);
115 }
116 memcpy(b->data + b->length, text, n);
117 b->length += n;
118 b->data[b->length] = 0;
119}

125static char **split(const char *s, char delimiter, size_t *count) {
126 char **out = NULL;
127 size_t start = 0;
128 *count = 0;
129 for (size_t i = 0;; i++) if (!s[i] || s[i] == delimiter) {
130 APPEND(out, *count, slice(s + start, i - start));
131 start = i + 1;
132 if (!s[i]) break;
133 }
134 return out;
135}

137static char *join(char **parts, size_t count, const char *separator) {
138 Buffer b = {0};
139 for (size_t i = 0; i < count; i++) {
140 if (i) append(&b, separator);
141 append(&b, parts[i]);
142 }
143 return finish(&b);
144}

159static char *name(const char *word, bool wildcard, const char *origin) {
160 if (origin && !strcmp(word, "@")) return (char *)origin;
161 if (!strcmp(word, ".")) return ".";
162 size_t n = strlen(word);
163 bool absolute = word[n - 1] == '.';
164 char *text = slice(word, n - (absolute ? 1 : 0));
165 size_t count;
166 char **labels = split(text, '.', &count);
167 for (size_t i = 0; i < count; i++) {
168 if (wildcard && !i && !strcmp(labels[i], "*")) continue;
169 size_t length = strlen(labels[i]);
170 if (!length || length > 63 || labels[i][0] == '-' || labels[i][length - 1] == '-') error("bad name %s", word);
171 for (size_t j = 0; j < length; j++) if (!letter(labels[i][j]) && !digit(labels[i][j]) && labels[i][j] != '-') error("bad name %s", word);
172 }
173 for (size_t i = 0; text[i]; i++) if (text[i] >= 'A' && text[i] <= 'Z') text[i] += 'a' - 'A';
174 return format("%s.%s", text, origin && !absolute && strcmp(origin, ".") ? origin : "");
175}

193static int compare_names(const char *left, const char *right) {
194 if (!strcmp(left, ".")) return !strcmp(right, ".") ? 0 : -1;
195 if (!strcmp(right, ".")) return 1;
196 size_t a_count, b_count;
197 char **a = split(slice(left, strlen(left) - 1), '.', &a_count), **b = split(slice(right, strlen(right) - 1), '.', &b_count);
198 size_t n = a_count < b_count ? a_count : b_count;
199 for (size_t i = 1; i <= n; i++) {
200 int order = strcmp(a[a_count - i], b[b_count - i]);
201 if (order) return order;
202 }
203 return (a_count > b_count) - (a_count < b_count);
204}

206static uint32_t ipv4(const char *word) {
207 size_t count;
208 char **fields = split(word, '.', &count);
209 if (count != 4) error("bad address %s", word);
210 uint32_t value = 0;
211 for (size_t i = 0; i < 4; i++) {
212 size_t n = strlen(fields[i]);
213 if (!n || n > 3 || (n > 1 && fields[i][0] == '0')) error("bad address %s", word);
214 for (size_t j = 0; j < n; j++) if (!digit(fields[i][j])) error("bad address %s", word);
215 int part = atoi(fields[i]);
216 if (part > 255) error("bad address %s", word);
217 value = value * 256 + (uint32_t)part;
218 }
219 return value;
220}

230static int *ipv6(const char *word) {
231 const char *colon = strstr(word, "::");
232 char *left = colon ? slice(word, (size_t)(colon - word)) : copy(word), *right = colon ? copy(colon + 2) : "";
233 if (colon && strstr(colon + 1, "::")) error("bad address %s", word);
234 size_t a_count = 0, b_count = 0;
235 char **a = *left ? split(left, ':', &a_count) : NULL, **b = *right ? split(right, ':', &b_count) : NULL;
236 if (colon ? a_count + b_count > 7 : a_count != 8) error("bad address %s", word);
237 int *groups = allocate(8 * sizeof(*groups));
238 for (int side = 0; side < 2; side++) {
239 size_t count = side ? b_count : a_count;
240 char **parts = side ? b : a;
241 for (size_t i = 0; i < count; i++) {
242 size_t n = strlen(parts[i]);
243 if (!n || n > 4) error("bad address %s", word);
244 int value = 0;
245 for (size_t j = 0; j < n; j++) {
246 int h = hex(parts[i][j]);
247 if (h < 0) error("bad address %s", word);
248 value = value * 16 + h;
249 }
250 groups[side ? 8 - b_count + i : i] = value;
251 }
252 }
253 return groups;
254}

256static char *address6(const int *groups) {
257 int best = -1, count = 1;
258 for (int i = 0; i < 8;) {
259 if (groups[i]) {
260 i++;
261 continue;
262 }
263 int end = i;
264 while (end < 8 && !groups[end]) end++;
265 if (end - i > count) {
266 best = i;
267 count = end - i;
268 }
269 i = end;
270 }
271 char **left = NULL, **right = NULL;
272 size_t left_count = 0, right_count = 0;
273 if (best < 0) {
274 for (int i = 0; i < 8; i++) APPEND(left, left_count, format("%x", groups[i]));
275 return join(left, left_count, ":");
276 }
277 for (int i = 0; i < best; i++) APPEND(left, left_count, format("%x", groups[i]));
278 for (int i = best + count; i < 8; i++) APPEND(right, right_count, format("%x", groups[i]));
279 return format("%s::%s", join(left, left_count, ":"), join(right, right_count, ":"));
280}

295static int ttl(const char *word) {
296 int plain = integer(word);
297 int64_t total = plain >= 0 ? plain : 0;
298 if (plain < 0) {
299 size_t at = 0;
300 int previous = -1;
301 const char *units = "wdhms";
302 int seconds[] = {604800, 86400, 3600, 60, 1};
303 while (word[at]) {
304 size_t start = at;
305 while (digit(word[at])) at++;
306 char *number = slice(word + start, at - start);
307 if (!*number || strlen(number) > 3 || integer(number) < 0 || !word[at]) error("bad ttl %s", word);
308 char c = (char)tolower((unsigned char)word[at++]);
309 const char *unit = strchr(units, c);
310 int index = unit ? (int)(unit - units) : -1;
311 if (index <= previous) error("bad ttl %s", word);
312 previous = index;
313 total += (int64_t)atoi(number) * seconds[index];
314 }
315 }
316 if (total < 1 || total > 604800) error("bad ttl %s", word);
317 return (int)total;
318}

334static int compare_data(Record *a, Record *b) {
335 if (a->type == 0) {
336 uint32_t x = ipv4(a->data), y = ipv4(b->data);
337 return (x > y) - (x < y);
338 }
339 if (a->type == 7) {
340 int *x = ipv6(a->data), *y = ipv6(b->data);
341 for (int i = 0; i < 8; i++) if (x[i] != y[i]) return x[i] - y[i];
342 return 0;
343 }
344 if (a->type == 1 || a->type == 2 || a->type == 4) return compare_names(a->data, b->data);
345 if (a->type == 5) {
346 char *x = strchr(a->data, ' '), *y = strchr(b->data, ' ');
347 int order = atoi(a->data) - atoi(b->data);
348 return order ? order : compare_names(x + 1, y + 1);
349 }
350 return a->type == 6 ? strcmp(a->data, b->data) : 0;
351}

384static void zone_line(char **w, size_t argc, int line) {
385 if (!strcmp(w[0], "ROOT")) {
386 uint32_t *additions = NULL;
387 size_t count = 0;
388 for (size_t i = 1; i < argc; i++) {
389 uint32_t value = ipv4(w[i]);
390 if (contains_address(roots, root_count, value) || contains_address(additions, count, value)) error("duplicate root %s", w[i]);
391 APPEND(additions, count, value);
392 }
393 for (size_t i = 0; i < count; i++) APPEND(roots, root_count, additions[i]);
394 return;
395 }
396 if (!strcmp(w[0], "ZONE")) {
397 current = NULL;
398 char *origin = name(w[1], false, NULL);
399 uint32_t server = ipv4(w[2]);
400 for (size_t i = 0; i < zone_count; i++) if (!strcmp(zones[i]->origin, origin) && zones[i]->server == server) error("duplicate zone %s at %s", origin, w[2]);
401 current = allocate(sizeof(*current));
402 current->origin = origin;
403 current->server = server;
404 current->line = line;
405 APPEND(zones, zone_count, current);
406 return;
407 }
408 if (!current) error("no current zone");
409 char *owner = name(w[0], true, current->origin);
410 if (!below(owner, current->origin)) error("out of zone %s", owner);
411 int seconds = ttl(w[1]);
412 int type = type_index(w[2]);
413 char **data = w + 3;
414 size_t count = argc - 3;
415 if (type != 6 && count != (size_t)(type == 5 ? 2 : type == 3 ? 7 : 1)) error("wrong field count");
416 char *text;
417 if (type == 0) text = address(ipv4(data[0]));
418 else if (type == 7) text = address6(ipv6(data[0]));
419 else if (type == 1 || type == 2 || type == 4) text = name(data[0], false, current->origin);
420 else if (type == 5) {
421 int preference = integer(data[0]);
422 if (preference < 0 || preference > 65535) error("bad preference %s", data[0]);
423 text = format("%d %s", preference, name(data[1], false, current->origin));
424 } else if (type == 6) {
425 text = join(data, count, " ");
426 size_t n = strlen(text);
427 if (n < 2 || *text != '"' || text[n - 1] != '"' || memchr(text + 1, '"', n - 2)) error("bad text %s", text);
428 } else {
429 char **fields = NULL;
430 size_t fields_count = 0;
431 APPEND(fields, fields_count, name(data[0], false, current->origin));
432 APPEND(fields, fields_count, name(data[1], false, current->origin));
433 for (int i = 2; i < 6; i++) {
434 if (integer(data[i]) < 0) error("bad number %s", data[i]);
435 APPEND(fields, fields_count, data[i]);
436 }
437 int minimum = integer(data[6]);
438 if (minimum < 1 || minimum > 86400) error("bad minimum %s", data[6]);
439 APPEND(fields, fields_count, data[6]);
440 text = join(fields, fields_count, " ");
441 }
442 if (type == 1 && !strncmp(owner, "*.", 2)) error("ns at wildcard");
443 if (type == 3 && strcmp(owner, current->origin)) error("soa not at origin");
444 if (type == 3 && soa(current)) error("duplicate soa");
445 for (size_t i = 0; i < current->records.count; i++) {
446 Record *r = current->records.items[i];
447 if (!strcmp(r->owner, owner) && r->type == type && !strcmp(r->data, text)) error("duplicate record");
448 }
449 for (size_t i = 0; i < current->records.count; i++) {
450 Record *r = current->records.items[i];
451 if (!strcmp(r->owner, owner) && (type == 2 || r->type == 2)) error("cname conflict %s", owner);
452 }
453 APPEND(current->records.items, current->records.count, record(owner, seconds, type, text));
454}

491static Response *authoritative(uint32_t server, const char *name, int type, const char *cut) {
492 if (contains_address(down, down_count, server) || !known(server)) return response("TIMEOUT");
493 Zone *zone = NULL;
494 for (size_t i = 0; i < zone_count; i++) if (zones[i]->server == server && below(name, zones[i]->origin) && (!zone || name_length(zones[i]->origin) > name_length(zone->origin))) zone = zones[i];
495 if (!zone || name_length(zone->origin) < name_length(cut)) return response("LAME");
496 char *delegation = NULL;
497 for (size_t i = 0; i < zone->records.count; i++) {
498 Record *r = zone->records.items[i];
499 if (r->type == 1 && strcmp(r->owner, zone->origin) && below(name, r->owner) && (!delegation || name_length(r->owner) < name_length(delegation))) delegation = r->owner;
500 }
501 if (delegation) {
502 Response *r = response("REFERRAL");
503 r->cut = delegation;
504 r->authority = records(zone, delegation, 1);
505 for (size_t i = 0; i < r->authority.count; i++) {
506 Records glue = records(zone, r->authority.items[i]->data, 0);
507 for (size_t j = 0; j < glue.count; j++) APPEND(r->additional.items, r->additional.count, glue.items[j]);
508 }
509 return r;
510 }
511 Records chosen = {0};
512 for (size_t i = 0; i < zone->records.count; i++) if (!strcmp(zone->records.items[i]->owner, name)) APPEND(chosen.items, chosen.count, zone->records.items[i]);
513 if (!chosen.count && !exists(zone, name)) {
514 char *ancestor = parent(name);
515 while (!exists(zone, ancestor)) ancestor = parent(ancestor);
516 char *wildcard = !strcmp(ancestor, ".") ? "*." : format("*.%s", ancestor);
517 for (size_t i = 0; i < zone->records.count; i++) if (!strcmp(zone->records.items[i]->owner, wildcard)) APPEND(chosen.items, chosen.count, zone->records.items[i]);
518 }
519 Response *r;
520 if (!chosen.count) r = response(exists(zone, name) ? "NODATA" : "NXDOMAIN");
521 else {
522 r = response("ANSWER");
523 for (size_t i = 0; i < chosen.count; i++) if (chosen.items[i]->type == type) APPEND(r->answer.items, r->answer.count, record((char *)name, chosen.items[i]->ttl, type, chosen.items[i]->data));
524 if (r->answer.count) {
525 r->answer = sorted(r->answer);
526 return r;
527 }
528 if (type != 2) for (size_t i = 0; i < chosen.count; i++) if (chosen.items[i]->type == 2) {
529 r->kind = "CNAME";
530 APPEND(r->answer.items, r->answer.count, record((char *)name, chosen.items[i]->ttl, 2, chosen.items[i]->data));
531 return r;
532 }
533 r->kind = "NODATA";
534 }
535 APPEND(r->authority.items, r->authority.count, soa(zone));
536 return r;
537}

572static int store(Records records) {
573 int ttl = records.items[0]->ttl;
574 for (size_t i = 1; i < records.count; i++) if (records.items[i]->ttl < ttl) ttl = records.items[i]->ttl;
575 if (ttl < min_ttl) ttl = min_ttl;
576 if (ttl > max_ttl) ttl = max_ttl;
577 cache_put(records.items[0]->owner, records.items[0]->type, "POS", sorted(records), now + ttl);
578 return ttl;
579}

597static Response *query_servers(const char *name, int type, const char *cut, char **servers, size_t server_count, bool root_hints, int depth, Key *resolving, size_t resolving_count) {
598 if (root_hints) {
599 for (size_t i = 0; i < root_count; i++) {
600 Response *r = ask(roots[i], name, type, cut);
601 trace(depth, format("ASK %s %s %s -> %s", address(roots[i]), name, types[type], description(r)));
602 if (strcmp(r->kind, "LAME") && strcmp(r->kind, "TIMEOUT")) return r;
603 }
604 } else {
605 char **ordered = allocate(server_count * sizeof(*ordered));
606 if (server_count) memcpy(ordered, servers, server_count * sizeof(*ordered));
607 qsort(ordered, server_count, sizeof(*ordered), name_compare);
608 for (size_t i = 0; i < server_count; i++) {
609 char *host = ordered[i];
610 if (i && !strcmp(host, ordered[i - 1])) continue;
611 uint32_t *addresses = NULL;
612 size_t count = 0;
613 CacheEntry *entry = cached(host, 0);
614 bool loop = false;
615 for (size_t j = 0; j < resolving_count; j++) if (resolving[j].type == 0 && !strcmp(resolving[j].name, host)) loop = true;
616 if (entry && !strcmp(entry->kind, "POS")) for (size_t j = 0; j < entry->records.count; j++) {
617 uint32_t a = ipv4(entry->records.items[j]->data);
618 if (!contains_address(addresses, count, a)) APPEND(addresses, count, a);
619 } else if (loop) trace(depth, format("GLUELESS %s -> LOOP", host));
620 else {
621 trace(depth, format("GLUELESS %s", host));
622 Result result = resolve(host, 0, depth + 1, resolving, resolving_count);
623 if (!strcmp(result.status, "NOERROR")) for (size_t j = 0; j < result.answer.count; j++) if (result.answer.items[j]->type == 0) {
624 uint32_t a = ipv4(result.answer.items[j]->data);
625 if (!contains_address(addresses, count, a)) APPEND(addresses, count, a);
626 }
627 }
628 if (count) qsort(addresses, count, sizeof(*addresses), address_compare);
629 for (size_t j = 0; j < count; j++) {
630 Response *r = ask(addresses[j], name, type, cut);
631 trace(depth, format("ASK %s %s %s -> %s", address(addresses[j]), name, types[type], description(r)));
632 if (strcmp(r->kind, "LAME") && strcmp(r->kind, "TIMEOUT")) return r;
633 }
634 }
635 }
636 return NULL;
637}

639static Response *lookup(const char *name, int type, int depth, Key *resolving, size_t resolving_count) {
640 CacheEntry *entry = cached(name, type);
641 Response *r = NULL;
642 if (entry) {
643 r = response(!strcmp(entry->kind, "POS") ? "ANSWER" : entry->kind);
644 if (!strcmp(entry->kind, "POS")) r->answer = cached_records(entry);
645 }
646 if (!r) for (size_t i = 0; i < cache_count; i++) if (!strcmp(cache[i]->kind, "NXDOMAIN") && below(name, cache[i]->key.name)) {
647 r = response("NXDOMAIN");
648 break;
649 }
650 if (!r && type != 2) {
651 entry = cached(name, 2);
652 if (entry && !strcmp(entry->kind, "POS")) {
653 r = response("CNAME");
654 r->answer = cached_records(entry);
655 }
656 }
657 if (r) {
658 hits++;
659 trace(depth, format("CACHE HIT %s %s -> %s", name, types[type], description(r)));
660 return r;
661 }
662 char *cut = ".", **servers = NULL;
663 size_t server_count = 0;
664 bool hints = true;
665 for (size_t i = 0; i < cache_count; i++) {
666 entry = cache[i];
667 if (entry->key.type == 1 && !strcmp(entry->kind, "POS") && below(name, entry->key.name) && (hints || name_length(entry->key.name) > name_length(cut))) {
668 cut = entry->key.name;
669 servers = NULL;
670 server_count = 0;
671 hints = false;
672 for (size_t j = 0; j < entry->records.count; j++) APPEND(servers, server_count, entry->records.items[j]->data);
673 }
674 }
675 if (!hints) {
676 hits++;
677 trace(depth, format("CACHE HIT %s %s -> REFERRAL %s", name, types[type], cut));
678 }
679 while (true) {
680 r = query_servers(name, type, cut, servers, server_count, hints, depth, resolving, resolving_count);
681 if (!r) return response(format("SERVFAIL no server for %s", cut));
682 if (!strcmp(r->kind, "REFERRAL")) {
683 store(r->authority);
684 for (size_t i = 0; i < r->additional.count; i++) {
685 Record *item = r->additional.items[i];
686 bool earlier = false;
687 for (size_t j = 0; j < i; j++) if (!strcmp(r->additional.items[j]->owner, item->owner)) earlier = true;
688 if (earlier) continue;
689 Records group = {0};
690 for (size_t j = i; j < r->additional.count; j++) if (!strcmp(r->additional.items[j]->owner, item->owner)) APPEND(group.items, group.count, r->additional.items[j]);
691 store(group);
692 }
693 cut = r->cut;
694 servers = NULL;
695 server_count = 0;
696 hints = false;
697 for (size_t i = 0; i < r->authority.count; i++) APPEND(servers, server_count, r->authority.items[i]->data);
698 continue;
699 }
700 if (!strcmp(r->kind, "ANSWER") || !strcmp(r->kind, "CNAME")) {
701 int ttl = store(r->answer);
702 for (size_t i = 0; i < r->answer.count; i++) r->answer.items[i] = with_ttl(r->answer.items[i], ttl);
703 } else {
704 Record *soa_record = r->authority.items[0];
705 size_t count;
706 char **fields = split(soa_record->data, ' ', &count);
707 int ttl = atoi(fields[6]);
708 if (ttl > soa_record->ttl) ttl = soa_record->ttl;
709 if (ttl > neg_ttl) ttl = neg_ttl;
710 cache_put((char *)name, type, r->kind, (Records){0}, now + ttl);
711 }
712 return r;
713 }
714}

716static Result resolve(const char *name, int type, int depth, Key *resolving, size_t resolving_count) {
717 Key *stack = allocate((resolving_count + 1) * sizeof(*stack));
718 if (resolving_count) memcpy(stack, resolving, resolving_count * sizeof(*stack));
719 stack[resolving_count++] = (Key){(char *)name, type};
720 Records aliases = {0};
721 char **seen = NULL;
722 size_t seen_count = 0;
723 APPEND(seen, seen_count, (char *)name);
724 const char *current_name = name;
725 Result result = {0};
726 while (true) {
727 Response *r = lookup(current_name, type, depth, stack, resolving_count);
728 if (!strcmp(r->kind, "CNAME")) {
729 Record *alias = r->answer.items[0];
730 APPEND(aliases.items, aliases.count, alias);
731 bool loop = false;
732 for (size_t i = 0; i < seen_count; i++) if (!strcmp(seen[i], alias->data)) loop = true;
733 if (loop) {
734 result.status = "SERVFAIL cname loop";
735 break;
736 }
737 if (aliases.count == 9) {
738 result.status = "SERVFAIL cname chain too long";
739 break;
740 }
741 APPEND(seen, seen_count, alias->data);
742 current_name = alias->data;
743 continue;
744 }
745 if (!strcmp(r->kind, "ANSWER")) {
746 result.status = "NOERROR";
747 result.answer = aliases;
748 for (size_t i = 0; i < r->answer.count; i++) APPEND(result.answer.items, result.answer.count, r->answer.items[i]);
749 } else if (!strcmp(r->kind, "NODATA") || !strcmp(r->kind, "NXDOMAIN")) {
750 result.status = r->kind;
751 result.answer = aliases;
752 } else result.status = r->kind;
753 break;
754 }
755 resolutions[!strcmp(result.status, "NOERROR") ? 0 : !strcmp(result.status, "NODATA") ? 1 : !strcmp(result.status, "NXDOMAIN") ? 2 : 3]++;
756 return result;
757}

778static int cache_compare(const void *a, const void *b) {
779 CacheEntry *x = *(CacheEntry *const *)a, *y = *(CacheEntry *const *)b;
780 int order = compare_names(x->key.name, y->key.name);
781 return order ? order : x->key.type - y->key.type;
782}

784static int record_compare(const void *a, const void *b) {
785 Record *x = *(Record *const *)a, *y = *(Record *const *)b;
786 int order = compare_names(x->owner, y->owner);
787 if (!order) order = x->type - y->type;
788 return order ? order : compare_data(x, y);
789}

796static void command(char **w, size_t argc) {
797 if (!strcmp(w[0], "RESOLVE")) {
798 char *n = name(w[1], true, NULL);
799 int t = type_index(w[2]);
800 printf("resolve %s %s\n", n, types[t]);
801 answer(resolve_root(n, t));
802 } else if (!strcmp(w[0], "REVERSE")) {
803 char *output, *reverse;
804 if (strchr(w[1], ':')) {
805 int *groups = ipv6(w[1]);
806 output = address6(groups);
807 Buffer digits = {0};
808 for (int i = 0; i < 8; i++) append(&digits, format("%04x", groups[i]));
809 Buffer n = {0};
810 for (int i = 31; i >= 0; i--) append(&n, format("%c.", digits.data[i]));
811 append(&n, "ip6.arpa.");
812 reverse = finish(&n);
813 } else {
814 uint32_t a = ipv4(w[1]);
815 output = address(a);
816 reverse = format("%u.%u.%u.%u.in-addr.arpa.", a & 255, (a >> 8) & 255, (a >> 16) & 255, a >> 24);
817 }
818 printf("reverse %s %s\n", output, reverse);
819 answer(resolve_root(reverse, 4));
820 } else if (!strcmp(w[0], "MX")) {
821 char *n = name(w[1], true, NULL);
822 printf("mx %s\n", n);
823 Result result = resolve_root(n, 5);
824 answer(result);
825 if (!strcmp(result.status, "NOERROR")) for (size_t i = 0; i < result.answer.count; i++) {
826 Record *r = result.answer.items[i];
827 if (r->type != 5) continue;
828 char *space = strchr(r->data, ' ');
829 char *preference = slice(r->data, (size_t)(space - r->data));
830 if (!strcmp(space + 1, ".")) printf("exchange %s . -> null\n", preference);
831 else addresses(space + 1, format("exchange %s ", preference));
832 }
833 if (!strcmp(result.status, "NODATA")) {
834 char *host = n;
835 for (size_t i = 0; i < result.answer.count; i++) if (result.answer.items[i]->type == 2) host = result.answer.items[i]->data;
836 addresses(host, "implicit ");
837 }
838 } else if (!strcmp(w[0], "QUERY")) {
839 uint32_t server = ipv4(w[1]);
840 char *n = name(w[2], true, NULL);
841 int t = type_index(w[3]);
842 Response *r = ask(server, n, t, ".");
843 printf("query %s %s %s -> %s\n", address(server), n, types[t], description(r));
844 for (size_t i = 0; i < r->answer.count; i++) printf("answer %s\n", record_text(r->answer.items[i]));
845 for (size_t i = 0; i < r->authority.count; i++) printf("authority %s\n", record_text(r->authority.items[i]));
846 for (size_t i = 0; i < r->additional.count; i++) printf("additional %s\n", record_text(r->additional.items[i]));
847 } else if (!strcmp(w[0], "DUMP")) {
848 uint32_t server = ipv4(w[1]);
849 char *n = name(w[2], false, NULL);
850 if (contains_address(down, down_count, server) || !known(server)) {
851 printf("dump %s %s -> TIMEOUT\n", address(server), n);
852 return;
853 }
854 Zone *zone = NULL;
855 for (size_t i = 0; i < zone_count; i++) if (zones[i]->server == server && !strcmp(zones[i]->origin, n)) zone = zones[i];
856 if (!zone) printf("dump %s %s -> REFUSED\n", address(server), n);
857 else {
858 printf("dump %s %s records=%zu\n", address(server), n, zone->records.count);
859 Records ordered = {.count = zone->records.count};
860 ordered.items = allocate(ordered.count * sizeof(*ordered.items));
861 if (ordered.count) memcpy(ordered.items, zone->records.items, ordered.count * sizeof(*ordered.items));
862 qsort(ordered.items, ordered.count, sizeof(*ordered.items), record_compare);
863 for (size_t i = 0; i < ordered.count; i++) printf(" %s\n", record_text(ordered.items[i]));
864 }
865 } else if (!strcmp(w[0], "DOWN") || !strcmp(w[0], "UP")) {
866 uint32_t server = ipv4(w[1]);
867 bool disable = !strcmp(w[0], "DOWN"), present = contains_address(down, down_count, server);
868 if (disable ? present : !present) error("server %s %s", w[1], disable ? "already down" : "not down");
869 if (disable) APPEND(down, down_count, server);
870 else {
871 size_t next = 0;
872 for (size_t i = 0; i < down_count; i++) if (down[i] != server) down[next++] = down[i];
873 down_count = next;
874 }
875 printf("%s %s\n", disable ? "down" : "up", address(server));
876 } else if (!strcmp(w[0], "SET")) {
877 if (strcmp(w[1], "MINTTL") && strcmp(w[1], "MAXTTL") && strcmp(w[1], "NEGTTL")) error("bad setting %s", w[1]);
878 int value = integer(w[2]);
879 if (value < 1 || value > 604800) error("bad value %s", w[2]);
880 if ((!strcmp(w[1], "MINTTL") ? value : min_ttl) > (!strcmp(w[1], "MAXTTL") ? value : max_ttl)) error("minttl above maxttl");
881 if (!strcmp(w[1], "MINTTL")) min_ttl = value;
882 else if (!strcmp(w[1], "MAXTTL")) max_ttl = value;
883 else neg_ttl = value;
884 printf("set %s=%d\n", w[1], value);
885 } else if (!strcmp(w[0], "TIME")) {
886 int value = integer(w[1]);
887 if (value < 1 || value > 1000000) error("bad time %s", w[1]);
888 now += value;
889 size_t old = cache_count, next = 0;
890 for (size_t i = 0; i < cache_count; i++) if (cache[i]->expiry > now) cache[next++] = cache[i];
891 cache_count = next;
892 printf("time now=%d expired=%zu\n", now, old - cache_count);
893 } else if (!strcmp(w[0], "CACHE") || !strcmp(w[0], "FLUSH")) {
894 char *n = argc == 2 ? name(w[1], false, NULL) : NULL;
895 CacheEntry **selected = NULL;
896 size_t count = 0;
897 for (size_t i = 0; i < cache_count; i++) if (!n || below(cache[i]->key.name, n)) APPEND(selected, count, cache[i]);
898 if (!strcmp(w[0], "FLUSH")) {
899 size_t next = 0;
900 for (size_t i = 0; i < cache_count; i++) if (n && !below(cache[i]->key.name, n)) cache[next++] = cache[i];
901 cache_count = next;
902 printf("flush removed=%zu\n", count);
903 } else {
904 printf("cache entries=%zu\n", count);
905 if (count) qsort(selected, count, sizeof(*selected), cache_compare);
906 for (size_t i = 0; i < count; i++) {
907 CacheEntry *e = selected[i];
908 if (!strcmp(e->kind, "POS")) {
909 Records r = cached_records(e);
910 for (size_t j = 0; j < r.count; j++) printf(" %s\n", record_text(r.items[j]));
911 } else printf(" %s %d %s %s\n", e->key.name, e->expiry - now, types[e->key.type], e->kind);
912 }
913 }
914 } else {
915 int total = 0;
916 for (size_t i = 0; i < query_count; i++) for (int j = 0; j < 7; j++) total += query_counts[i]->counts[j];
917 printf("stats queries=%d hits=%d\n", total, hits);
918 if (query_count) qsort(query_counts, query_count, sizeof(*query_counts), query_compare);
919 for (size_t i = 0; i < query_count; i++) {
920 QueryCount *q = query_counts[i];
921 int sent = 0;
922 for (int j = 0; j < 7; j++) sent += q->counts[j];
923 printf(" %s sent=%d", address(q->server), sent);
924 for (int j = 0; j < 7; j++) {
925 char *kind = copy(kinds[j]);
926 for (size_t k = 0; kind[k]; k++) kind[k] = (char)tolower((unsigned char)kind[k]);
927 printf(" %s=%d", kind, q->counts[j]);
928 }
929 putchar('\n');
930 }
931 printf("resolutions noerror=%d nodata=%d nxdomain=%d servfail=%d\n", resolutions[0], resolutions[1], resolutions[2], resolutions[3]);
932 }
933}
