    int ttl = records.items[0]->ttl;

    if (ttl < min_ttl) ttl = min_ttl;
    if (ttl > max_ttl) ttl = max_ttl;
    cache_put(records.items[0]->owner, records.items[0]->type, "POS", sorted(records), now + ttl);
