1const TYPES=("A", "NS", "CNAME", "SOA", "PTR", "MX", "TXT", "AAAA")
2const KINDS=("ANSWER", "CNAME", "REFERRAL", "NODATA", "NXDOMAIN", "LAME", "TIMEOUT")
3const STATUSES=("NOERROR", "NODATA", "NXDOMAIN", "SERVFAIL")
4const Name=Tuple{Vararg{String}}
5const ROOT=()
6struct InputError <: Exception
7 message::String
8end
9fail(s) = throw(InputError(s))
10require(b, s) = b ? nothing : fail(s)
11part(s, a, b = length(s)) = a>b ? "" : String(s[a:b])
12integerword(s) = occursin(r"^(?:0|[1-9][0-9]{0,8})$", s)
13function ipv4(s)
14 fields=split(s, '.'; keepempty = true)
15 length(fields)==4 && all(x->occursin(r"^(?:0|[1-9][0-9]{0,2})$", x), fields) ||
16 fail("bad address $s")
17 values=parse.(Int, fields)
18 all(<=(255), values) || fail("bad address $s")
19 foldl((a, b)->a*256+b, values; init = 0)
20end
21ip4text(n) = join([div(n, 256^i)%256 for i in 3:-1:0], '.')
22function groups(s)
23 isempty(s) && return Int[]
24 parts=split(s, ':'; keepempty = true)
25 all(x->occursin(r"^[0-9a-fA-F]{1,4}$", x), parts) || return nothing
26 parse.(Int, parts; base = 16)
27end
28function ipv6(s)
29 parts=split(s, "::"; limit = 2, keepempty = true)
30 if length(parts)==1
31 values=groups(s)
32 values!==nothing && length(values)==8 || fail("bad address $s")
33 return Tuple(values)
34 end
35 front=groups(parts[1])
36 back=groups(parts[2])
37 front!==nothing && back!==nothing && length(front)+length(back)<=7 || fail("bad address $s")
38 Tuple([front; zeros(Int, 8-length(front)-length(back)); back])
39end
40function ip6text(values)
41 best=0
42 width=1
43 at=1
44 while at<=8
45 stop=at
46 while stop<=8 && values[stop]==0
47 stop+=1
48 end
49 if stop-at>width
50 best=at
51 width=stop-at
52 end
53 at=max(stop, at+1)
54 end
55 parts=[string(x; base = 16) for x in values]
56 best==0 ? join(parts, ':') :
57 join(parts[1:(best-1)], ':')*"::"*join(parts[(best+width):end], ':')
58end
59function parsename(s; wildcard = false, origin = nothing)
60 origin!==nothing && s=="@" && return origin
61 s=="." && return ROOT
62 absolute=endswith(s, ".")
63 fields=split(absolute ? part(s, 1, length(s)-1) : s, '.'; keepempty = true)
64 for (i, label) in enumerate(fields)
65 wildcard && i==1 && label=="*" && continue
66 occursin(r"^[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?$", label) || fail("bad name $s")
67 end
68 name=Tuple(lowercase.(String.(fields)))
69 origin!==nothing && !absolute ? (name..., origin...) : name
70end
71nametext(n) = isempty(n) ? "." : join(n, '.')*"."
72namekey(n) = reverse(n)
73below(n, a) = length(n)>=length(a) && n[(length(n)-length(a)+1):end]==a
74ancestors(n) = [n[i:end] for i in 1:(length(n)+1)]
75function ttlword(s)
76 if integerword(s)
77 n=parse(Int, s)
78 else
79 pairs=collect(eachmatch(r"(0|[1-9][0-9]{0,2})([wWdDhHmMsS])", s))
80 join(m.match for m in pairs)==s && !isempty(pairs) || fail("bad ttl $s")
81 previous=0
82 n=0
83 for m in pairs
84 at=findfirst(==(lowercase(m.captures[2])[1]), "wdhms")
85 at>previous || fail("bad ttl $s")
86 previous=at
87 n+=parse(Int, m.captures[1])*(604800, 86400, 3600, 60, 1)[at]
88 end
89 end
90 require(1<=n<=604800, "bad ttl $s")
91 n
92end
93checktype(s) = require(s in TYPES, "bad type $s")
94function parsedata(kind, fields, origin)
95 kind=="A" && return ipv4(fields[1])
96 kind=="AAAA" && return ipv6(fields[1])
97 kind in ("NS", "CNAME", "PTR") && return parsename(fields[1]; origin)
98 if kind=="MX"
99 require(
100 integerword(fields[1]) && parse(Int, fields[1])<=65535,
101 "bad preference $(fields[1])",
102 )
103 return (parse(Int, fields[1]), parsename(fields[2]; origin))
104 end
105 if kind=="TXT"
106 s=join(fields, ' ')
107 require(
108 length(s)>=2 && s[1]=='"' && s[end]=='"' && !('"' in part(s, 2, length(s)-1)),
109 "bad text $s",
110 )
111 return s
112 end
113 first=parsename(fields[1]; origin)
114 second=parsename(fields[2]; origin)
115 numbers=Int[]
116 for s in fields[3:6]
117 require(integerword(s), "bad number $s")
118 push!(numbers, parse(Int, s))
119 end
120 s=fields[7]
121 require(integerword(s) && 1<=parse(Int, s)<=86400, "bad minimum $s")
122 (first, second, numbers..., parse(Int, s))
123end
124struct Record
125 owner::Name
126 ttl::Int
127 kind::String
128 data::Any
129end
130withttl(r, t) = Record(r.owner, t, r.kind, r.data)
131withowner(r, n) = Record(n, r.ttl, r.kind, r.data)
132samerecord(a, b) = a.owner==b.owner && a.kind==b.kind && a.data==b.data
133datakey(r) =
134 r.kind in ("NS", "CNAME", "PTR") ? namekey(r.data) :
135 r.kind=="MX" ? (r.data[1], namekey(r.data[2])) : r.data
136recordkey(r) = (namekey(r.owner), findfirst(==(r.kind), TYPES), datakey(r))
137dataorder(records) = sort(records; by = datakey)
138function datatext(r)
139 r.kind=="A" && return ip4text(r.data)
140 r.kind=="AAAA" && return ip6text(r.data)
141 r.kind in ("NS", "CNAME", "PTR") && return nametext(r.data)
142 r.kind=="MX" && return string(r.data[1])*" "*nametext(r.data[2])
143 r.kind=="SOA" && return nametext(r.data[1])*" "*nametext(r.data[2])*" "*join(r.data[3:end], ' ')
144 r.data
145end
146recordtext(r) = nametext(r.owner)*" $(r.ttl) $(r.kind) "*datatext(r)
147mutable struct Zone
148 origin::Name
149 server::Int
150 line::Int
151 records::Vector{Record}
152end
153atowner(z, n) = [r for r in z.records if r.owner==n]
154findrecords(z, n, t) = dataorder([r for r in z.records if r.owner==n && r.kind==t])
155exists(z, n) = any(r->below(r.owner, n), z.records)
156soa(z) = z.records[findfirst(r->r.kind=="SOA", z.records)]
157struct Response
158 kind::String
159 answer::Vector{Record}
160 authority::Vector{Record}
161 additional::Vector{Record}
162 referral::Name
163end
164Response(k; answer = Record[], authority = Record[], additional = Record[], referral = ROOT) =
165 Response(k, answer, authority, additional, referral)
166response_summary(r) =
167 r.kind=="REFERRAL" ? "REFERRAL "*nametext(r.referral) :
168 r.kind=="CNAME" ? "CNAME "*nametext(r.answer[1].data) : r.kind
169struct Entry
170 kind::String
171 records::Vector{Record}
172 expiry::Int
173end
174mutable struct Resolver
175 roots::Vector{Int}
176 zones::Vector{Zone}
177 current::Int
178 down::Set{Int}
179 clock::Int
180 cache::Dict{Tuple{Name,String},Entry}
181 settings::Dict{String,Int}
182 queries::Dict{Int,Vector{Int}}
183 hits::Int
184 resolutions::Vector{Int}
185end
186Resolver() = Resolver(
187 Int[],
188 Zone[],
189 0,
190 Set{Int}(),
191 0,
192 Dict{Tuple{Name,String},Entry}(),
193 Dict("MINTTL"=>1, "MAXTTL"=>86400, "NEGTTL"=>3600),
194 Dict{Int,Vector{Int}}(),
195 0,
196 zeros(Int, 4),
197)
198function zoneinput!(s, line, number)
199 words=String.(split(line, ' '))
200 command=words[1]
201 if command=="ROOT"
202 addresses=Int[]
203 for word in words[2:end]
204 n=ipv4(word)
205 require(!(n in s.roots || n in addresses), "duplicate root $word")
206 push!(addresses, n)
207 end
208 append!(s.roots, addresses)
209 elseif command=="ZONE"
210 s.current=0
211 origin=parsename(words[2])
212 server=ipv4(words[3])
213 require(
214 !any(z->z.origin==origin && z.server==server, s.zones),
215 "duplicate zone $(nametext(origin)) at $(words[3])",
216 )
217 push!(s.zones, Zone(origin, server, number, Record[]))
218 s.current=length(s.zones)
219 else
220 require(s.current!=0, "no current zone")
221 z=s.zones[s.current]
222 owner=parsename(words[1]; wildcard = true, origin = z.origin)
223 require(below(owner, z.origin), "out of zone $(nametext(owner))")
224 ttl=ttlword(words[2])
225 kind=words[3]
226 checktype(kind)
227 fields=words[4:end]
228 counts=Dict("A"=>1, "AAAA"=>1, "NS"=>1, "CNAME"=>1, "PTR"=>1, "MX"=>2, "SOA"=>7)
229 require(!haskey(counts, kind) || length(fields)==counts[kind], "wrong field count")
230 r=Record(owner, ttl, kind, parsedata(kind, fields, z.origin))
231 require(!(kind=="NS" && !isempty(owner) && owner[1]=="*"), "ns at wildcard")
232 require(kind!="SOA" || owner==z.origin, "soa not at origin")
233 require(kind!="SOA" || !any(x->x.kind=="SOA", z.records), "duplicate soa")
234 require(!any(x->samerecord(x, r), z.records), "duplicate record")
235 existing=atowner(z, owner)
236 require(
237 !(
238 (kind=="CNAME" && !isempty(existing)) ||
239 (kind!="CNAME" && any(x->x.kind=="CNAME", existing))
240 ),
241 "cname conflict $(nametext(owner))",
242 )
243 push!(z.records, r)
244 end
245end
246function finishzones!(s)
247 kept=Zone[]
248 for z in s.zones
249 label="zone $(nametext(z.origin)) at $(ip4text(z.server))"
250 if any(r->r.kind=="SOA", z.records)
251 println(label, " records=$(length(z.records))")
252 push!(kept, z)
253 else
254 println("error line $(z.line): $label has no soa")
255 end
256 end
257 s.zones=kept
258 println(
259 "loaded roots=$(length(s.roots)) zones=$(length(kept)) records=$(sum(z->length(z.records),kept;init=0))",
260 )
261end
262reachable(s, a) = !(a in s.down) && any(z->z.server==a, s.zones)
263function dnsresponse(s, server, name, kind, cut)
264 reachable(s, server) || return Response("TIMEOUT")
265 candidates=[z for z in s.zones if z.server==server && below(name, z.origin)]
266 isempty(candidates) && return Response("LAME")
267 z=candidates[argmax(length(z.origin) for z in candidates)]
268 length(z.origin)>=length(cut) || return Response("LAME")
269 for n in (length(z.origin)+1):length(name)
270 point=name[(length(name)-n+1):end]
271 ns=findrecords(z, point, "NS")
272 if !isempty(ns)
273 glue=Record[]
274 for r in ns
275 append!(glue, findrecords(z, r.data, "A"))
276 end
277 return Response("REFERRAL"; authority = ns, additional = glue, referral = point)
278 end
279 end
280 records=atowner(z, name)
281 if isempty(records)
282 exists(z, name) && return Response("NODATA"; authority = [soa(z)])
283 closest=first(a for a in ancestors(name)[2:end] if exists(z, a))
284 records=atowner(z, ("*", closest...))
285 isempty(records) && return Response("NXDOMAIN"; authority = [soa(z)])
286 end
287 records=withowner.(records, Ref(name))
288 matching=dataorder([r for r in records if r.kind==kind])
289 !isempty(matching) && return Response("ANSWER"; answer = matching)
290 cnames=[r for r in records if r.kind=="CNAME"]
291 kind!="CNAME" && !isempty(cnames) && return Response("CNAME"; answer = cnames)
292 Response("NODATA"; authority = [soa(z)])
293end
294function dnsask!(s, a, n, t, c)
295 response=dnsresponse(s, a, n, t, c)
296 counts=get!(()->zeros(Int, 7), s.queries, a)
297 counts[findfirst(==(response.kind), KINDS)]+=1
298 response
299end
300function positive!(s, records)
301 ttl=min(max(minimum(r.ttl for r in records), s.settings["MINTTL"]), s.settings["MAXTTL"])
302 first=records[1]
303 s.cache[(first.owner, first.kind)]=Entry("POS", dataorder(records), s.clock+ttl)
304 ttl
305end
306function negative!(s, name, kind, r)
307 soa=r.authority[1]
308 ttl=min(soa.ttl, soa.data[end], s.settings["NEGTTL"])
309 s.cache[(name, kind)]=Entry(r.kind, Record[], s.clock+ttl)
310end
311function positiveentry(s, name, kind)
312 e=get(s.cache, (name, kind), nothing)
313 e!==nothing && e.kind=="POS" ? e : nothing
314end
315aged(s, e) = [withttl(r, e.expiry-s.clock) for r in e.records]
316trace(d, message) = println(repeat(" ", d+1), message)
317function dnslookup!(s, name, kind, depth, pending)
318 entry=get(s.cache, (name, kind), nothing)
319 cached=nothing
320 if entry!==nothing
321 cached=Response(
322 entry.kind=="POS" ? "ANSWER" : entry.kind;
323 answer = entry.kind=="POS" ? aged(s, entry) : Record[],
324 )
325 elseif any(kv->kv.second.kind=="NXDOMAIN" && below(name, kv.first[1]), s.cache)
326 cached=Response("NXDOMAIN")
327 elseif kind!="CNAME"
328 e=positiveentry(s, name, "CNAME")
329 e!==nothing && (cached=Response("CNAME"; answer = aged(s, e)))
330 end
331 query=nametext(name)*" "*kind
332 if cached!==nothing
333 s.hits+=1
334 trace(depth, "CACHE HIT $query -> $(response_summary(cached))")
335 return cached
336 end
337 cut=ROOT
338 nameservers=nothing
339 for ancestor in ancestors(name)
340 e=positiveentry(s, ancestor, "NS")
341 if e!==nothing
342 s.hits+=1
343 trace(depth, "CACHE HIT $query -> REFERRAL $(nametext(ancestor))")
344 cut=ancestor
345 nameservers=[r.data for r in e.records]
346 break
347 end
348 end
349 while true
350 response=nothing
351 hosts=nameservers===nothing ? [nothing] : sort(nameservers; by = namekey)
352 for host in hosts
353 addresses=Int[]
354 if host===nothing
355 addresses=s.roots
356 else
357 e=positiveentry(s, host, "A")
358 if e!==nothing
359 addresses=sort(unique([r.data for r in e.records]))
360 elseif (host, "A") in pending
361 trace(depth, "GLUELESS $(nametext(host)) -> LOOP")
362 else
363 trace(depth, "GLUELESS $(nametext(host))")
364 status, answer=dnsresolve!(s, host, "A", depth+1, pending)
365 status=="NOERROR" &&
366 (addresses=sort(unique([r.data for r in answer if r.kind=="A"])))
367 end
368 end
369 for address in addresses
370 r=dnsask!(s, address, name, kind, cut)
371 trace(depth, "ASK $(ip4text(address)) $query -> $(response_summary(r))")
372 if !(r.kind in ("LAME", "TIMEOUT"))
373 response=r
374 break
375 end
376 end
377 response!==nothing && break
378 end
379 response===nothing && return Response("SERVFAIL no server for "*nametext(cut))
380 if response.kind=="REFERRAL"
381 positive!(s, response.authority)
382 owners=unique([r.owner for r in response.additional])
383 for owner in owners
384 positive!(s, [r for r in response.additional if r.owner==owner])
385 end
386 cut=response.referral
387 nameservers=[r.data for r in response.authority]
388 elseif response.kind in ("ANSWER", "CNAME")
389 ttl=positive!(s, response.answer)
390 return Response(response.kind; answer = [withttl(r, ttl) for r in response.answer])
391 else
392 negative!(s, name, kind, response)
393 return Response(response.kind)
394 end
395 end
396end
397function dnsresolve!(s, name, kind, depth = 0, pending = Tuple{Name,String}[])
398 pending=[pending; [(name, kind)]]
399 chain=Record[]
400 seen=Set{Name}([name])
401 current=name
402 response=Response("")
403 while true
404 response=dnslookup!(s, current, kind, depth, pending)
405 response.kind!="CNAME" && break
406 append!(chain, response.answer)
407 target=response.answer[1].data
408 if target in seen
409 response=Response("SERVFAIL cname loop")
410 break
411 end
412 if length(chain)==9
413 response=Response("SERVFAIL cname chain too long")
414 break
415 end
416 push!(seen, target)
417 current=target
418 end
419 status=response.kind=="ANSWER" ? "NOERROR" : response.kind
420 answer=startswith(status, "SERVFAIL") ? Record[] : [chain; response.answer]
421 label=startswith(status, "SERVFAIL") ? "SERVFAIL" : status
422 s.resolutions[findfirst(==(label), STATUSES)]+=1
423 status, answer
424end
425function report!(s, name, kind)
426 status, answer=dnsresolve!(s, name, kind)
427 for r in answer
428 println("answer ", recordtext(r))
429 end
430 println("status ", status)
431 status, answer
432end
433function addressresolution!(s, host)
434 addresses=String[]
435 for kind in ("A", "AAAA")
436 status, answer=dnsresolve!(s, host, kind)
437 status=="NOERROR" && append!(addresses, [datatext(r) for r in answer if r.kind==kind])
438 end
439 isempty(addresses) ? "no address" : join(addresses, ' ')
440end
441function execute!(s, line)
442 w=String.(split(line, ' '))
443 command=w[1]
444 if command in ("RESOLVE", "MX")
445 name=parsename(w[2]; wildcard = true)
446 kind=command=="MX" ? "MX" : w[3]
447 checktype(kind)
448 println(command=="MX" ? "mx $(nametext(name))" : "resolve $(nametext(name)) $kind")
449 status, answer=report!(s, name, kind)
450 if command=="MX"
451 if status=="NOERROR"
452 for r in answer
453 r.kind=="MX" || continue
454 preference, host=r.data
455 println(
456 "exchange $preference $(nametext(host)) -> ",
457 isempty(host) ? "null" : addressresolution!(s, host),
458 )
459 end
460 elseif status=="NODATA"
461 host=isempty(answer) ? name : answer[end].data
462 addresses=addressresolution!(s, host)
463 println("implicit $(nametext(host)) -> $addresses")
464 end
465 end
466 elseif command=="REVERSE"
467 if ':' in w[2]
468 hextets=ipv6(w[2])
469 address=ip6text(hextets)
470 digits=join(lpad(string(x; base = 16), 4, '0') for x in hextets)
471 name=(string.(reverse(collect(digits)))..., "ip6", "arpa")
472 else
473 n=ipv4(w[2])
474 address=ip4text(n)
475 name=(reverse(String.(split(address, '.')))..., "in-addr", "arpa")
476 end
477 println("reverse $address $(nametext(name))")
478 report!(s, name, "PTR")
479 elseif command=="QUERY"
480 address=ipv4(w[2])
481 name=parsename(w[3]; wildcard = true)
482 checktype(w[4])
483 response=dnsask!(s, address, name, w[4], ROOT)
484 println(
485 "query $(ip4text(address)) $(nametext(name)) $(w[4]) -> $(response_summary(response))",
486 )
487 for (label, records) in (
488 ("answer", response.answer),
489 ("authority", response.authority),
490 ("additional", response.additional),
491 ),
492 r in records
493
494 println(label, " ", recordtext(r))
495 end
496 elseif command=="DUMP"
497 address=ipv4(w[2])
498 name=parsename(w[3])
499 prefix="dump $(ip4text(address)) $(nametext(name))"
500 if !reachable(s, address)
501 println(prefix, " -> TIMEOUT")
502 else
503 at=findfirst(z->z.server==address && z.origin==name, s.zones)
504 if at===nothing
505 println(prefix, " -> REFUSED")
506 else
507 z=s.zones[at]
508 println(prefix, " records=$(length(z.records))")
509 for r in sort(z.records; by = recordkey)
510 println(" ", recordtext(r))
511 end
512 end
513 end
514 elseif command in ("DOWN", "UP")
515 address=ipv4(w[2])
516 require(
517 command=="DOWN" ? !(address in s.down) : address in s.down,
518 "server $(w[2]) "*(command=="DOWN" ? "already down" : "not down"),
519 )
520 command=="DOWN" ? push!(s.down, address) : delete!(s.down, address)
521 println(lowercase(command), " ", ip4text(address))
522 elseif command=="SET"
523 key=w[2]
524 require(haskey(s.settings, key), "bad setting $key")
525 require(integerword(w[3]) && 1<=parse(Int, w[3])<=604800, "bad value $(w[3])")
526 value=parse(Int, w[3])
527 next=copy(s.settings)
528 next[key]=value
529 require(next["MINTTL"]<=next["MAXTTL"], "minttl above maxttl")
530 s.settings=next
531 println("set $key=$value")
532 elseif command=="TIME"
533 require(integerword(w[2]) && 1<=parse(Int, w[2])<=1000000, "bad time $(w[2])")
534 s.clock+=parse(Int, w[2])
535 expired=[k for (k, e) in s.cache if e.expiry<=s.clock]
536 for k in expired
537 delete!(s.cache, k)
538 end
539 println("time now=$(s.clock) expired=$(length(expired))")
540 elseif command in ("CACHE", "FLUSH")
541 name=length(w)==1 ? nothing : parsename(w[2])
542 selected=[k for k in keys(s.cache) if name===nothing || below(k[1], name)]
543 if command=="FLUSH"
544 for k in selected
545 delete!(s.cache, k)
546 end
547 println("flush removed=$(length(selected))")
548 else
549 sort!(selected; by = k->(namekey(k[1]), findfirst(==(k[2]), TYPES)))
550 println("cache entries=$(length(selected))")
551 for k in selected
552 e=s.cache[k]
553 if e.kind=="POS"
554 for r in aged(s, e)
555 println(" ", recordtext(r))
556 end
557 else
558 println(" $(nametext(k[1])) $(e.expiry-s.clock) $(k[2]) $(e.kind)")
559 end
560 end
561 end
562 elseif command=="STATS"
563 println("stats queries=$(sum(x->sum(x),values(s.queries);init=0)) hits=$(s.hits)")
564 for a in sort(collect(keys(s.queries)))
565 counts=s.queries[a]
566 println(
567 " $(ip4text(a)) sent=$(sum(counts)) ",
568 join([lowercase(KINDS[i])*"=$(counts[i])" for i in 1:7], ' '),
569 )
570 end
571 println(
572 "resolutions ",
573 join([lowercase(STATUSES[i])*"=$(s.resolutions[i])" for i in 1:4], ' '),
574 )
575 end
576end
577function main()
578 resolver=Resolver()
579 zones=true
580 for (number, line) in enumerate(eachline())
581 line=="END" && !zones && break
582 if zones && line=="END ZONES"
583 finishzones!(resolver)
584 zones=false
585 continue
586 end
587 try
588 zones ? zoneinput!(resolver, line, number) : execute!(resolver, line)
589 catch e
590 e isa InputError || rethrow()
591 println("error line $number: $(e.message)")
592 end
593 end
594end
