37 private function name(string $word, bool $wildcard = false, ?string $origin = null): string
38 {
39 if ($origin !== null && $word === '@') return $origin;
40 if ($word === '.') return '.';
41 $parts = explode('.', str_ends_with($word, '.') ? substr($word, 0, -1) : $word);
42 foreach ($parts as $index => $part) {
43 if ($wildcard && $index === 0 && $part === '*') continue;
44 if (strlen($part) < 1 || strlen($part) > 63 || !preg_match('/\A[A-Za-z0-9](?:[A-Za-z0-9-]*[A-Za-z0-9])?\z/', $part)) $this->fail("bad name $word");
45 }
46 $result = strtolower(implode('.', $parts)) . '.';
47 if ($origin !== null && $origin !== '.' && !str_ends_with($word, '.')) $result .= $origin;
48 return $result;

50 private function address(string $word): int
51 {
52 if (!preg_match('/\A(?:0|[1-9][0-9]{0,2})(?:\.(?:0|[1-9][0-9]{0,2})){3}\z/', $word)) $this->fail("bad address $word");
53 $result = 0;
54 foreach (explode('.', $word) as $part) {
55 if ((int)$part > 255) $this->fail("bad address $word");
56 $result = $result * 256 + (int)$part;
57 }
58 return $result;

60 private function ip(int $number): string {
61 return implode('.', array_map(fn($shift) => ($number >> $shift) & 255, [24, 16, 8, 0]));

63 private function ipv6(string $word): array
64 {
65 $parts = explode('::', $word);
66 if (count($parts) > 2) $this->fail("bad address $word");
67 $sides = array_map(fn($part) => $part === '' ? [] : explode(':', $part), $parts);
68 $groups = array_merge(...$sides);
69 foreach ($groups as $group) if (!preg_match('/\A[0-9A-Fa-f]{1,4}\z/', $group)) $this->fail("bad address $word");
70 $count = count($groups);
71 if (count($parts) === 1 ? $count !== 8 : $count > 7) $this->fail("bad address $word");
72 return count($parts) === 1 ? array_map('hexdec', $groups) : [...array_map('hexdec', $sides[0]), ...array_fill(0, 8 - $count, 0), ...array_map('hexdec', $sides[1])];
73 }
74 private function ipv6Text(array $groups): string
75 {
76 $bestStart = -1;
77 $bestLength = 1;
78 $index = 0;
79 while ($index < 8) {
80 if ($groups[$index] !== 0) {
81 $index++;
82 continue;
83 }
84 $finish = $index;
85 while ($finish < 8 && $groups[$finish] === 0) $finish++;
86 if ($finish - $index > $bestLength) {
87 $bestStart = $index;
88 $bestLength = $finish - $index;
89 }
90 $index = $finish;
91 }
92 $words = array_map('dechex', $groups);
93 return $bestStart < 0 ? implode(':', $words) : implode(':', array_slice($words, 0, $bestStart)) . '::' . implode(':', array_slice($words, $bestStart + $bestLength));
94 }
95 private function integer(string $word): bool {

98 private function ttl(string $word): int
99 {
100 if ($this->integer($word)) $seconds = (int)$word;
101 else {
102 $text = strtolower($word);
103 if (!preg_match('/\A(?:(?:0|[1-9][0-9]{0,2})[wdhms])+\z/', $text)) $this->fail("bad ttl $word");
104 preg_match_all('/([0-9]+)([wdhms])/', $text, $pairs, PREG_SET_ORDER);
105 $previous = -1;
106 $seconds = 0;
107 $units = ['w' => 604800, 'd' => 86400, 'h' => 3600, 'm' => 60, 's' => 1];
108 foreach ($pairs as $pair) {
109 $rank = strpos('wdhms', $pair[2]);
110 if ($rank <= $previous) $this->fail("bad ttl $word");
111 $previous = $rank;
112 $seconds += (int)$pair[1] * $units[$pair[2]];
113 }
114 }
115 if ($seconds < 1 || $seconds > 604800) $this->fail("bad ttl $word");
116 return $seconds;
117 }

123 private function compareData(array $a, array $b): int
124 {
125 return match ($a['type']) {
126 'A', 'AAAA' => $a['data'] <=> $b['data'],
127 'MX' => ($a['data'][0] <=> $b['data'][0]) ?: strcmp($this->nameKey($a['data'][1]), $this->nameKey($b['data'][1])),
128 'TXT' => strcmp($a['data'], $b['data']), 'SOA' => 0,
129 default => strcmp($this->nameKey($a['data']), $this->nameKey($b['data']))
130 };
131 }
132 private function sorted(array $records): array {
133 usort($records, fn($a, $b) => $this->compareData($a, $b));

136 private function recordText(array $record, ?int $ttl = null): string
137 {
138 $data = match ($record['type']) {
139 'A' => $this->ip($record['data']), 'AAAA' => $this->ipv6Text($record['data']),

144 private function zoneLine(array $w, int $number): void
145 {
146 if ($w[0] === 'ROOT') {
147 $added = [];
148 foreach (array_slice($w, 1) as $word) {
149 $value = $this->address($word);
150 if (in_array($value, $this->roots, true) || in_array($value, $added, true)) $this->fail("duplicate root $word");
151 $added[] = $value;
152 }
153 array_push($this->roots, ...$added);
154 }
155 elseif ($w[0] === 'ZONE') {
156 $this->current = null;
157 $origin = $this->name($w[1]);
158 $server = $this->address($w[2]);
159 foreach ($this->zones as $zone) if ($zone->origin === $origin && $zone->server === $server) $this->fail("duplicate zone $origin at $w[2]");
160 $this->current = (object)['origin' => $origin, 'server' => $server, 'line' => $number, 'records' => []];
161 $this->zones[] = $this->current;
162 }
163 else {
164 if ($this->current === null) $this->fail('no current zone');
165 $owner = $this->name($w[0], true, $this->current->origin);
166 if (!$this->below($owner, $this->current->origin)) $this->fail("out of zone $owner");
167 $ttl = $this->ttl($w[1]);
168 $type = $this->checkType($w[2]);
169 $fields = array_slice($w, 3);
170 $count = ['MX' => 2, 'SOA' => 7][$type] ?? 1;
171 if ($type !== 'TXT' && count($fields) !== $count) $this->fail('wrong field count');
172 if ($type === 'A') $data = $this->address($fields[0]);
173 elseif ($type === 'AAAA') $data = $this->ipv6($fields[0]);
174 elseif (in_array($type, ['NS', 'CNAME', 'PTR'], true)) $data = $this->name($fields[0], false, $this->current->origin);
175 elseif ($type === 'MX') {
176 if (!$this->integer($fields[0]) || (int)$fields[0] > 65535) $this->fail("bad preference $fields[0]");
177 $data = [(int)$fields[0], $this->name($fields[1], false, $this->current->origin)];
178 }
179 elseif ($type === 'TXT') {
180 $data = implode(' ', $fields);
181 if (!preg_match('/\A"[^"]*"\z/', $data)) $this->fail("bad text $data");
182 }
183 else {
184 $data = [$this->name($fields[0], false, $this->current->origin), $this->name($fields[1], false, $this->current->origin)];
185 for ($i = 2; $i <= 5; $i++) if (!$this->integer($fields[$i])) $this->fail("bad number $fields[$i]");
186 if (!$this->integer($fields[6]) || (int)$fields[6] < 1 || (int)$fields[6] > 86400) $this->fail("bad minimum $fields[6]");
187 array_push($data, ...array_map('intval', array_slice($fields, 2)));
188 }
189 if ($type === 'NS' && str_starts_with($owner, '*.')) $this->fail('ns at wildcard');
190 if ($type === 'SOA' && $owner !== $this->current->origin) $this->fail('soa not at origin');
191 if ($type === 'SOA') foreach ($this->current->records as $record) if ($record['type'] === 'SOA') $this->fail('duplicate soa');
192 foreach ($this->current->records as $record) if ($record['owner'] === $owner && $record['type'] === $type && $record['data'] === $data) $this->fail('duplicate record');
193 foreach ($this->current->records as $record) if ($record['owner'] === $owner && ($type === 'CNAME' || $record['type'] === 'CNAME')) $this->fail("cname conflict $owner");
194 $this->current->records[] = ['owner' => $owner, 'ttl' => $ttl, 'type' => $type, 'data' => $data];
195 }

206 private function authoritative(int $server, string $name, string $type, string $cut): array
207 {
208 if (isset($this->down[$server]) || !$this->known($server)) return $this->response('TIMEOUT');
209 $chosen = null;
210 foreach ($this->zones as $zone) if ($zone->server === $server && $this->below($name, $zone->origin) && ($chosen === null || count($this->labels($zone->origin)) > count($this->labels($chosen->origin)))) $chosen = $zone;
211 if ($chosen === null || count($this->labels($chosen->origin)) < count($this->labels($cut))) return $this->response('LAME');
212 $records = $chosen->records;
213 $soa = array_values(array_filter($records, fn($record) => $record['type'] === 'SOA'))[0];
214 foreach (array_reverse($this->ancestors($name)) as $ancestor) {
215 if ($ancestor === $chosen->origin || !$this->below($ancestor, $chosen->origin)) continue;
216 $authority = $this->sorted(array_values(array_filter($records, fn($record) => $record['owner'] === $ancestor && $record['type'] === 'NS')));
217 if ($authority === []) continue;
218 $glue = [];
219 foreach ($authority as $record) array_push($glue, ...$this->sorted(array_values(array_filter($records, fn($item) => $item['owner'] === $record['data'] && $item['type'] === 'A'))));
220 return $this->response('REFERRAL', [], $authority, $glue, $ancestor);
221 }
222 $found = array_values(array_filter($records, fn($record) => $record['owner'] === $name));
223 if ($found === []) {
224 $exists = function ($ancestor) use ($records) {
225 foreach ($records as $record) if ($this->below($record['owner'], $ancestor)) return true;
226 return false;
227 };
228 if ($exists($name)) return $this->response('NODATA', [], [$soa]);
229 $closest = '.';
230 foreach (array_slice($this->ancestors($name), 1) as $ancestor) if ($exists($ancestor)) {
231 $closest = $ancestor;
232 break;
233 }
234 $wildcard = $closest === '.' ? '*.' : '*.' . $closest;
235 $found = array_values(array_filter($records, fn($record) => $record['owner'] === $wildcard));
236 if ($found === []) return $this->response('NXDOMAIN', [], [$soa]);
237 }
238 $answer = array_values(array_filter($found, fn($record) => $record['type'] === $type));
239 $kind = 'ANSWER';
240 if ($answer === [] && $type !== 'CNAME') {
241 $answer = array_values(array_filter($found, fn($record) => $record['type'] === 'CNAME'));
242 $kind = 'CNAME';
243 }
244 if ($answer === []) return $this->response('NODATA', [], [$soa]);
245 return $this->response($kind, array_map(fn($record) => [...$record, 'owner' => $name], $this->sorted($answer)));
246 }

266 private function store(array $records): array
267 {
268 $ttl = min(max(min(array_column($records, 'ttl')), $this->settings['MINTTL']), $this->settings['MAXTTL']);
269 $first = $records[0];

276 private function queryServers(string $name, string $type, string $cut, ?array $nameservers, int $depth, array $resolving): ?array
277 {
278 if ($nameservers === null) {
279 foreach ($this->roots as $server) {
280 $result = $this->ask($server, $name, $type, $cut);
281 $this->trace($depth, 'ASK ' . $this->ip($server) . " $name $type -> " . $this->describe($result));
282 if (!in_array($result['kind'], ['LAME', 'TIMEOUT'], true)) return $result;
283 }
284 }
285 else {
286 usort($nameservers, fn($a, $b) => strcmp($this->nameKey($a), $this->nameKey($b)));
287 foreach ($nameservers as $host) {
288 $entry = $this->cache[$this->key($host, 'A')] ?? null;
289 if ($entry !== null && $entry['kind'] === 'POS') $addresses = array_column($entry['records'], 'data');
290 elseif (in_array([$host, 'A'], $resolving, true)) {
291 $this->trace($depth, "GLUELESS $host -> LOOP");
292 $addresses = [];
293 }
294 else {
295 $this->trace($depth, "GLUELESS $host");
296 [$status, $answer] = $this->resolve($host, 'A', $depth + 1, $resolving);
297 $addresses = $status === 'NOERROR' ? array_column(array_filter($answer, fn($record) => $record['type'] === 'A'), 'data') : [];
298 }
299 $addresses = array_unique($addresses);
300 sort($addresses, SORT_NUMERIC);
301 foreach ($addresses as $server) {
302 $result = $this->ask($server, $name, $type, $cut);
303 $this->trace($depth, 'ASK ' . $this->ip($server) . " $name $type -> " . $this->describe($result));
304 if (!in_array($result['kind'], ['LAME', 'TIMEOUT'], true)) return $result;
305 }
306 }
307 }
308 return null;
309 }
310 private function lookup(string $name, string $type, int $depth, array $resolving): array
311 {
312 $entry = $this->cache[$this->key($name, $type)] ?? null;
313 $result = null;
314 if ($entry !== null) $result = $entry['kind'] === 'POS' ? $this->response('ANSWER', $this->cacheRecords($entry)) : $this->response($entry['kind']);
315 else foreach ($this->cache as $item) if ($item['kind'] === 'NXDOMAIN' && $this->below($name, $item['name'])) {
316 $result = $this->response('NXDOMAIN');
317 break;
318 }
319 if ($result === null && $type !== 'CNAME') {
320 $entry = $this->cache[$this->key($name, 'CNAME')] ?? null;
321 if ($entry !== null && $entry['kind'] === 'POS') $result = $this->response('CNAME', $this->cacheRecords($entry));
322 }
323 if ($result !== null) {
324 $this->hits++;
325 $this->trace($depth, "CACHE HIT $name $type -> " . $this->describe($result));
326 return $result;
327 }
328 $cut = '.';
329 $servers = null;
330 foreach ($this->ancestors($name) as $ancestor) {
331 $entry = $this->cache[$this->key($ancestor, 'NS')] ?? null;
332 if ($entry !== null && $entry['kind'] === 'POS') {
333 $cut = $ancestor;
334 $servers = array_column($entry['records'], 'data');
335 $this->hits++;
336 $this->trace($depth, "CACHE HIT $name $type -> REFERRAL $cut");
337 break;
338 }
339 }
340 while (true) {
341 $result = $this->queryServers($name, $type, $cut, $servers, $depth, $resolving);
342 if ($result === null) return $this->response("SERVFAIL no server for $cut");
343 if ($result['kind'] === 'REFERRAL') {
344 $this->store($result['authority']);
345 $groups = [];
346 foreach ($result['additional'] as $record) $groups[$record['owner']][] = $record;
347 foreach ($groups as $records) $this->store($records);
348 $cut = $result['cut'];
349 $servers = array_column($result['authority'], 'data');
350 }
351 elseif (in_array($result['kind'], ['ANSWER', 'CNAME'], true)) {
352 $result['answer'] = $this->store($result['answer']);
353 return $result;
354 }
355 else {
356 $soa = $result['authority'][0];
357 $ttl = min($soa['ttl'], $soa['data'][6], $this->settings['NEGTTL']);
358 $this->cache[$this->key($name, $type)] = ['name' => $name, 'type' => $type, 'kind' => $result['kind'], 'expiry' => $this->clock + $ttl];
359 return $result;
360 }
361 }

363 private function resolve(string $name, string $type, int $depth = 0, array $resolving = []): array
364 {
365 $resolving[] = [$name, $type];
366 $seen = [$name];
367 $chain = [];
368 $current = $name;
369 while (true) {
370 $result = $this->lookup($current, $type, $depth, $resolving);
371 if ($result['kind'] === 'CNAME') {
372 $record = $result['answer'][0];
373 $chain[] = $record;
374 $target = $record['data'];
375 if (in_array($target, $seen, true)) {
376 $status = 'SERVFAIL cname loop';
377 $answer = [];
378 break;
379 }
380 if (count($chain) === 9) {
381 $status = 'SERVFAIL cname chain too long';
382 $answer = [];
383 break;
384 }
385 $seen[] = $target;
386 $current = $target;
387 }
388 else {
389 $status = $result['kind'] === 'ANSWER' ? 'NOERROR' : $result['kind'];
390 $answer = str_starts_with($status, 'SERVFAIL') ? [] : [...$chain, ...$result['answer']];
391 break;
392 }
393 }
394 $this->resolutions[str_starts_with($status, 'SERVFAIL') ? 'SERVFAIL' : $status]++;
395 return [$status, $answer];
396 }

403 private function addressResolution(string $host): string
404 {
405 $output = [];
406 foreach (['A', 'AAAA'] as $type) {
407 [$status, $answer] = $this->resolve($host, $type);
408 if ($status !== 'NOERROR') continue;
409 foreach ($answer as $record) if ($record['type'] === $type) $output[] = $type === 'A' ? $this->ip($record['data']) : $this->ipv6Text($record['data']);
410 }
411 return $output === [] ? 'no address' : implode(' ', $output);
412 }
413 private function command(array $w): void
414 {
415 $kind = $w[0];
416 if ($kind === 'RESOLVE' || $kind === 'MX') {
417 $name = $this->name($w[1], true);
418 $type = $kind === 'MX' ? 'MX' : $this->checkType($w[2]);
419 echo strtolower($kind) . " $name" . ($kind === 'MX' ? '' : " $type") . "\n";
420 [$status, $answer] = $this->resolve($name, $type);
421 $this->printResult([$status, $answer]);
422 if ($kind === 'MX' && $status === 'NOERROR') foreach ($answer as $record) {
423 if ($record['type'] !== 'MX') continue;
424 [$preference, $host] = $record['data'];
425 $addresses = $host === '.' ? 'null' : $this->addressResolution($host);
426 echo "exchange $preference $host -> $addresses\n";
427 }
428 elseif ($kind === 'MX' && $status === 'NODATA') {
429 $host = $name;
430 foreach ($answer as $record) if ($record['type'] === 'CNAME') $host = $record['data'];
431 $addresses = $this->addressResolution($host);
432 echo "implicit $host -> $addresses\n";
433 }
434 }
435 elseif ($kind === 'REVERSE') {
436 if (str_contains($w[1], ':')) {
437 $groups = $this->ipv6($w[1]);
438 $text = $this->ipv6Text($groups);
439 $digits = implode('', array_map(fn($group) => sprintf('%04x', $group), $groups));
440 $reverse = implode('.', str_split(strrev($digits))) . '.ip6.arpa.';
441 }
442 else {
443 $text = $this->ip($this->address($w[1]));
444 $reverse = implode('.', array_reverse(explode('.', $text))) . '.in-addr.arpa.';
445 }
446 echo "reverse $text $reverse\n";
447 $this->printResult($this->resolve($reverse, 'PTR'));
448 }
449 elseif ($kind === 'QUERY' || $kind === 'DUMP') {
450 $server = $this->address($w[1]);
451 $name = $this->name($w[2], $kind === 'QUERY');
452 $serverText = $this->ip($server);
453 if ($kind === 'QUERY') {
454 $type = $this->checkType($w[3]);
455 $result = $this->ask($server, $name, $type, '.');
456 echo "query $serverText $name $type -> " . $this->describe($result) . "\n";
457 foreach (['answer', 'authority', 'additional'] as $section) foreach ($result[$section] as $record) echo $section . ' ' . $this->recordText($record) . "\n";
458 }
459 else {
460 $chosen = null;
461 foreach ($this->zones as $zone) if ($zone->server === $server && $zone->origin === $name) $chosen = $zone;
462 if (isset($this->down[$server]) || !$this->known($server)) echo "dump $serverText $name -> TIMEOUT\n";
463 elseif ($chosen === null) echo "dump $serverText $name -> REFUSED\n";
464 else {
465 echo "dump $serverText $name records=" . count($chosen->records) . "\n";
466 $records = $chosen->records;
467 usort($records, fn($a, $b) => strcmp($this->nameKey($a['owner']), $this->nameKey($b['owner'])) ?: (array_search($a['type'], self::TYPES, true) <=> array_search($b['type'], self::TYPES, true)) ?: $this->compareData($a, $b));
468 foreach ($records as $record) echo ' ' . $this->recordText($record) . "\n";
469 }
470 }
471 }
472 elseif ($kind === 'DOWN' || $kind === 'UP') {
473 $server = $this->address($w[1]);
474 if ($kind === 'DOWN' && isset($this->down[$server])) $this->fail("server $w[1] already down");
475 if ($kind === 'UP' && !isset($this->down[$server])) $this->fail("server $w[1] not down");
476 if ($kind === 'DOWN') $this->down[$server] = true;
477 else unset($this->down[$server]);
478 echo strtolower($kind) . ' ' . $this->ip($server) . "\n";
479 }
480 elseif ($kind === 'SET') {
481 [$key, $word] = array_slice($w, 1);
482 if (!isset($this->settings[$key])) $this->fail("bad setting $key");
483 if (!$this->integer($word) || (int)$word < 1 || (int)$word > 604800) $this->fail("bad value $word");
484 $settings = [...$this->settings, $key => (int)$word];
485 if ($settings['MINTTL'] > $settings['MAXTTL']) $this->fail('minttl above maxttl');
486 $this->settings = $settings;
487 echo "set $key=" . (int)$word . "\n";
488 }
489 elseif ($kind === 'TIME') {
490 if (!$this->integer($w[1]) || (int)$w[1] < 1 || (int)$w[1] > 1000000) $this->fail("bad time $w[1]");
491 $this->clock += (int)$w[1];
492 $before = count($this->cache);
493 $this->cache = array_filter($this->cache, fn($entry) => $entry['expiry'] > $this->clock);
494 echo "time now=$this->clock expired=" . ($before - count($this->cache)) . "\n";
495 }
496 elseif ($kind === 'CACHE' || $kind === 'FLUSH') {
497 $name = isset($w[1]) ? $this->name($w[1]) : null;
498 $entries = array_filter($this->cache, fn($entry) => $name === null || $this->below($entry['name'], $name));
499 if ($kind === 'FLUSH') {
500 foreach ($entries as $key => $entry) unset($this->cache[$key]);
501 echo 'flush removed=' . count($entries) . "\n";
502 }
503 else {
504 echo 'cache entries=' . count($entries) . "\n";
505 usort($entries, fn($a, $b) => strcmp($this->nameKey($a['name']), $this->nameKey($b['name'])) ?: (array_search($a['type'], self::TYPES, true) <=> array_search($b['type'], self::TYPES, true)));
506 foreach ($entries as $entry) {
507 $ttl = $entry['expiry'] - $this->clock;
508 if ($entry['kind'] === 'POS') foreach ($entry['records'] as $record) echo ' ' . $this->recordText($record, $ttl) . "\n";
509 else echo ' ' . $entry['name'] . " $ttl " . $entry['type'] . ' ' . $entry['kind'] . "\n";
510 }
511 }
512 }
513 elseif ($kind === 'STATS') {
514 $total = array_sum(array_map('array_sum', $this->queries));
515 echo "stats queries=$total hits=$this->hits\n";
516 ksort($this->queries, SORT_NUMERIC);
517 foreach ($this->queries as $server => $counts) echo ' ' . $this->ip($server) . ' sent=' . array_sum($counts) . ' ' . implode(' ', array_map(fn($response) => strtolower($response) . '=' . $counts[$response], self::KINDS)) . "\n";
518 echo 'resolutions ' . implode(' ', array_map(fn($status) => strtolower($status) . '=' . $this->resolutions[$status], ['NOERROR', 'NODATA', 'NXDOMAIN', 'SERVFAIL'])) . "\n";
519 }
520 }
521 public function run(): void
522 {
523 $zoneSection = true;
524 $number = 0;
525 while (($line = fgets(STDIN)) !== false) {
526 $number++;
527 $line = rtrim($line, "\r\n");
528 if (!$zoneSection && $line === 'END') break;
529 if ($zoneSection && $line === 'END ZONES') {
530 $kept = [];
531 foreach ($this->zones as $zone) {
532 $soa = array_filter($zone->records, fn($record) => $record['type'] === 'SOA');
533 if ($soa === []) echo "error line $zone->line: zone $zone->origin at " . $this->ip($zone->server) . " has no soa\n";
534 else {
535 $kept[] = $zone;
536 echo "zone $zone->origin at " . $this->ip($zone->server) . ' records=' . count($zone->records) . "\n";
537 }
538 }
539 $this->zones = $kept;
540 echo 'loaded roots=' . count($this->roots) . ' zones=' . count($this->zones) . ' records=' . array_sum(array_map(fn($zone) => count($zone->records), $this->zones)) . "\n";
541 $zoneSection = false;
542 continue;
543 }
544 try {
545 if ($zoneSection) $this->zoneLine(explode(' ', $line), $number);
546 else $this->command(explode(' ', $line));
547 }
548 catch (DnsFault $error) {
549 echo "error line $number: {$error->getMessage()}\n";
550 }
551 }
552 }
553}
