7TYPE_ORDER = ("A", "NS", "CNAME", "SOA", "PTR", "MX", "TXT", "AAAA")
8NAME_TYPES = ("NS", "CNAME", "PTR")
9FIELD_COUNTS = {"A": 1, "AAAA": 1, "NS": 1, "CNAME": 1, "PTR": 1, "MX": 2, "SOA": 7}

11FAILURES = ("LAME", "TIMEOUT")

13DEFAULT_SETTINGS = {"MINTTL": 1, "MAXTTL": 86400, "NEGTTL": 3600}
14ROOT = ()
15WILDCARD = "*"

17MAX_MINIMUM = 86400
18MAX_PREFERENCE = 65535

22LABEL_PATTERN = re.compile(r"[A-Za-z0-9]([A-Za-z0-9-]{0,61}[A-Za-z0-9])?")
23INTEGER_PATTERN = re.compile(r"0|[1-9][0-9]{0,8}")
24OCTET_PATTERN = re.compile(r"0|[1-9][0-9]{0,2}")
25GROUP_PATTERN = re.compile(r"[0-9a-fA-F]{1,4}")
26TTL_NUMBER = "(0|[1-9][0-9]{0,2})"
27TTL_PATTERN = re.compile(
28 "".join(f"(?:{TTL_NUMBER}[{unit}])?" for unit in ("wW", "dD", "hH", "mM", "sS"))
29)
30TTL_UNIT_SECONDS = (604800, 86400, 3600, 60, 1)

37def parse_integer(word):
38 return int(word) if INTEGER_PATTERN.fullmatch(word) else None

41def parse_ipv4(word):
42 fields = word.split(".")
43 if len(fields) != 4 or not all(OCTET_PATTERN.fullmatch(field) for field in fields):
44 return None
45 value = 0
46 for field in fields:
47 if int(field) > 255:
48 return None
49 value = value * 256 + int(field)
50 return value

60def ipv4_text(value):
61 return ".".join(str(value >> shift & 255) for shift in (24, 16, 8, 0))

64def parse_groups(text):
65 if not text:
66 return []
67 groups = text.split(":")
68 if not all(GROUP_PATTERN.fullmatch(group) for group in groups):
69 return None
70 return [int(group, 16) for group in groups]

73def parse_ipv6(word):
74 if "::" not in word:
75 groups = parse_groups(word)
76 return tuple(groups) if groups is not None and len(groups) == 8 else None
77 head, tail = word.split("::", 1)
78 front, back = parse_groups(head), parse_groups(tail)
79 if front is None or back is None or len(front) + len(back) > 7:
80 return None
81 return tuple(front + [0] * (8 - len(front) - len(back)) + back)

84def require_ipv6(word):
85 address = parse_ipv6(word)
86 if address is None:
87 raise InputError(f"bad address {word}")
88 return address

91def ipv6_text(groups):
92 best_start, best_length = -1, 1
93 index = 0
94 while index < 8:
95 end = index
96 while end < 8 and groups[end] == 0:
97 end += 1
98 if end - index > best_length:
99 best_start, best_length = index, end - index
100 index = max(end, index + 1)
101 parts = [f"{group:x}" for group in groups]
102 if best_start < 0:
103 return ":".join(parts)
104 head = ":".join(parts[:best_start])
105 tail = ":".join(parts[best_start + best_length:])
106 return f"{head}::{tail}"

109def parse_name(word, wildcard=False, origin=None):
110 if origin is not None and word == "@":
111 return origin
112 if word == ".":
113 return ROOT
114 absolute = word.endswith(".")
115 parts = (word[:-1] if absolute else word).split(".")
116 for index, part in enumerate(parts):
117 allowed_wildcard = wildcard and index == 0 and part == WILDCARD
118 if not allowed_wildcard and not LABEL_PATTERN.fullmatch(part):
119 raise InputError(f"bad name {word}")
120 labels = tuple(part.lower() for part in parts)
121 if origin is not None and not absolute:
122 labels += origin
123 return labels

146def parse_ttl(word):
147 seconds = parse_integer(word)
148 if seconds is None and (match := TTL_PATTERN.fullmatch(word)):
149 seconds = sum(
150 int(number) * unit
151 for number, unit in zip(match.groups(), TTL_UNIT_SECONDS)
152 if number is not None
153 )
154 if seconds is None or not 1 <= seconds <= MAX_TTL:
155 raise InputError(f"bad ttl {word}")
156 return seconds

159def require_type(word):
160 if word not in TYPE_ORDER:
161 raise InputError(f"bad type {word}")

164def parse_data(rtype, fields, origin):
165 if rtype == "A":
166 return require_ipv4(fields[0])
167 if rtype == "AAAA":
168 return require_ipv6(fields[0])
169 if rtype in NAME_TYPES:
170 return parse_name(fields[0], origin=origin)
171 if rtype == "MX":
172 preference = parse_integer(fields[0])
173 if preference is None or preference > MAX_PREFERENCE:
174 raise InputError(f"bad preference {fields[0]}")
175 return preference, parse_name(fields[1], origin=origin)
176 if rtype == "TXT":
177 text = " ".join(fields)
178 if len(text) < 2 or text[0] != '"' or text[-1] != '"' or '"' in text[1:-1]:
179 raise InputError(f"bad text {text}")
180 return text
181 mname = parse_name(fields[0], origin=origin)
182 rname = parse_name(fields[1], origin=origin)
183 numbers = []
184 for word in fields[2:6]:
185 number = parse_integer(word)
186 if number is None:
187 raise InputError(f"bad number {word}")
188 numbers.append(number)
189 minimum = parse_integer(fields[6])
190 if minimum is None or not 1 <= minimum <= MAX_MINIMUM:
191 raise InputError(f"bad minimum {fields[6]}")
192 return (mname, rname, *numbers, minimum)

195class Record:
196 def __init__(self, owner, ttl, rtype, data):
197 self.owner = owner
198 self.ttl = ttl
199 self.rtype = rtype
200 self.data = data
201
202 def with_ttl(self, ttl):
203 return Record(self.owner, ttl, self.rtype, self.data)
204
205 def with_owner(self, owner):
206 return Record(owner, self.ttl, self.rtype, self.data)
207
208 def same_as(self, other):
209 return (self.owner, self.rtype, self.data) == (other.owner, other.rtype, other.data)
210
211 def data_key(self):
212 if self.rtype in NAME_TYPES:
213 return name_key(self.data)
214 if self.rtype == "MX":
215 preference, host = self.data
216 return preference, name_key(host)
217 return self.data
218
219 def order_key(self):
220 return name_key(self.owner), TYPE_ORDER.index(self.rtype), self.data_key()
221
222 def data_text(self):
223 if self.rtype == "A":
224 return ipv4_text(self.data)
225 if self.rtype == "AAAA":
226 return ipv6_text(self.data)
227 if self.rtype in NAME_TYPES:
228 return name_text(self.data)
229 if self.rtype == "MX":
230 preference, host = self.data
231 return f"{preference} {name_text(host)}"
232 if self.rtype == "SOA":
233 mname, rname, *numbers = self.data
234 return " ".join([name_text(mname), name_text(rname), *map(str, numbers)])
235 return self.data
236
237 def text(self):
238 return f"{name_text(self.owner)} {self.ttl} {self.rtype} {self.data_text()}"

299class Resolver:
300 def __init__(self):
301 self.output = []
302 self.root_hints = []
303 self.zones = []
304 self.current_zone = None
305 self.zones_by_server = {}
306 self.down = set()
307 self.clock = 0
308 self.cache = {}
309 self.settings = dict(DEFAULT_SETTINGS)
310 self.query_counts = defaultdict(Counter)
311 self.hits = 0
312 self.resolution_counts = Counter()
313
314 def emit(self, line):
315 self.output.append(line)
316
317 def trace(self, depth, line):
318 self.emit(" " * (depth + 1) + line)

320 def load(self, line_number, line):
321 words = line.split(" ")
322 try:
323 if words[0] == "ROOT":
324 self.add_roots(words[1:])
325 elif words[0] == "ZONE":
326 self.current_zone = None
327 self.add_zone(line_number, words[1], words[2])
328 else:
329 self.add_record(words)
330 except InputError as error:
331 self.emit(f"error line {line_number}: {error}")
332
333 def add_roots(self, words):
334 addresses = []
335 for word in words:
336 address = require_ipv4(word)
337 if address in self.root_hints or address in addresses:
338 raise InputError(f"duplicate root {word}")
339 addresses.append(address)
340 self.root_hints.extend(addresses)
341
342 def add_zone(self, line_number, origin_word, server_word):
343 origin = parse_name(origin_word)
344 server = require_ipv4(server_word)
345 if any(zone.origin == origin and zone.server == server for zone in self.zones):
346 raise InputError(f"duplicate zone {name_text(origin)} at {server_word}")
347 self.current_zone = Zone(origin, server, line_number)
348 self.zones.append(self.current_zone)
349
350 def add_record(self, words):
351 zone = self.current_zone
352 if zone is None:
353 raise InputError("no current zone")
354 owner_word, ttl_word, rtype, *fields = words
355 owner = parse_name(owner_word, wildcard=True, origin=zone.origin)
356 if not is_at_or_below(owner, zone.origin):
357 raise InputError(f"out of zone {name_text(owner)}")
358 ttl = parse_ttl(ttl_word)
359 require_type(rtype)
360 if rtype in FIELD_COUNTS and len(fields) != FIELD_COUNTS[rtype]:
361 raise InputError("wrong field count")
362 record = Record(owner, ttl, rtype, parse_data(rtype, fields, zone.origin))
363 if rtype == "NS" and is_wildcard(owner):
364 raise InputError("ns at wildcard")
365 if rtype == "SOA" and owner != zone.origin:
366 raise InputError("soa not at origin")
367 if rtype == "SOA" and zone.soa is not None:
368 raise InputError("duplicate soa")
369 existing = zone.records_at(owner)
370 if any(other.same_as(record) for other in existing):
371 raise InputError("duplicate record")
372 has_cname = any(other.rtype == "CNAME" for other in existing)
373 if (rtype == "CNAME" and existing) or (rtype != "CNAME" and has_cname):
374 raise InputError(f"cname conflict {name_text(owner)}")
375 zone.add(record)
376
377 def finish_zones(self):
378 kept = []
379 for zone in self.zones:
380 label = f"zone {name_text(zone.origin)} at {ipv4_text(zone.server)}"
381 if zone.soa is None:
382 self.emit(f"error line {zone.line_number}: {label} has no soa")
383 else:
384 self.emit(f"{label} records={len(zone.records)}")
385 kept.append(zone)
386 self.zones = kept
387 for zone in kept:
388 self.zones_by_server.setdefault(zone.server, []).append(zone)
389 total = sum(len(zone.records) for zone in kept)
390 self.emit(f"loaded roots={len(self.root_hints)} zones={len(kept)} records={total}")
391
392 def reachable(self, server):
393 return server not in self.down and server in self.zones_by_server
394
395 def respond(self, server, name, rtype, cut):
396 if not self.reachable(server):
397 return Response("TIMEOUT")
398 candidates = [
399 zone for zone in self.zones_by_server[server] if is_at_or_below(name, zone.origin)
400 ]
401 if not candidates:
402 return Response("LAME")
403 zone = max(candidates, key=lambda candidate: len(candidate.origin))
404 if len(zone.origin) < len(cut):
405 return Response("LAME")
406 for length in range(len(zone.origin) + 1, len(name) + 1):
407 point = name[len(name) - length:]
408 nameservers = zone.find(point, "NS")
409 if nameservers:
410 glue = [record for ns in nameservers for record in zone.find(ns.data, "A")]
411 return Response(
412 "REFERRAL", authority=nameservers, additional=glue, referral=point
413 )
414 records = zone.records_at(name)
415 if not records:
416 if zone.exists(name):
417 return Response("NODATA", authority=[zone.soa])
418 closest = next(ancestor for ancestor in ancestors(name)[1:] if zone.exists(ancestor))
419 records = zone.records_at((WILDCARD,) + closest)
420 if not records:
421 return Response("NXDOMAIN", authority=[zone.soa])
422 records = [record.with_owner(name) for record in records]
423 matching = in_data_order(record for record in records if record.rtype == rtype)
424 if matching:
425 return Response("ANSWER", answer=matching)
426 cnames = [record for record in records if record.rtype == "CNAME"]
427 if rtype != "CNAME" and cnames:
428 return Response("CNAME", answer=cnames)
429 return Response("NODATA", authority=[zone.soa])

431 def ask(self, server, name, rtype, cut):
432 response = self.respond(server, name, rtype, cut)
433 self.query_counts[server][response.kind] += 1
434 return response

436 def store_positive(self, records):
437 lowest = min(record.ttl for record in records)
438 cache_ttl = min(max(lowest, self.settings["MINTTL"]), self.settings["MAXTTL"])
439 key = (records[0].owner, records[0].rtype)
440 self.cache[key] = Entry("POS", in_data_order(records), self.clock + cache_ttl)
441 return cache_ttl
442
443 def store_negative(self, name, rtype, response):
444 soa = response.authority[0]
445 cache_ttl = min(soa.ttl, soa.data[-1], self.settings["NEGTTL"])
446 self.cache[(name, rtype)] = Entry(response.kind, [], self.clock + cache_ttl)

448 def positive_entry(self, name, rtype):
449 entry = self.cache.get((name, rtype))
450 return entry if entry is not None and entry.kind == "POS" else None
451
452 def aged(self, entry):
453 return [record.with_ttl(entry.expiry - self.clock) for record in entry.records]
454
455 def cached_result(self, name, rtype):
456 entry = self.cache.get((name, rtype))
457 if entry is not None:
458 if entry.kind == "POS":
459 return "ANSWER", self.aged(entry)
460 return entry.kind, []
461 if any(
462 entry.kind == "NXDOMAIN" and is_at_or_below(name, cached_name)
463 for (cached_name, _), entry in self.cache.items()
464 ):
465 return "NXDOMAIN", []
466 entry = self.positive_entry(name, "CNAME")
467 if rtype != "CNAME" and entry is not None:
468 return "CNAME", self.aged(entry)
469 return None

471 def lookup(self, name, rtype, depth, pending):
472 query = f"{name_text(name)} {rtype}"
473 result = self.cached_result(name, rtype)
474 if result is not None:
475 self.hits += 1
476 self.trace(depth, f"CACHE HIT {query} -> {outcome_text(*result)}")
477 return result
478 cut, nameservers = ROOT, None
479 for ancestor in ancestors(name):
480 entry = self.positive_entry(ancestor, "NS")
481 if entry is not None:
482 self.hits += 1
483 self.trace(depth, f"CACHE HIT {query} -> REFERRAL {name_text(ancestor)}")
484 cut = ancestor
485 nameservers = [record.data for record in entry.records]
486 break
487 while True:
488 response = self.query_servers(name, rtype, cut, nameservers, depth, pending)
489 if response is None:
490 return f"SERVFAIL no server for {name_text(cut)}", []
491 if response.kind == "REFERRAL":
492 self.store_positive(response.authority)
493 for _, glue in groupby(response.additional, key=lambda record: record.owner):
494 self.store_positive(list(glue))
495 cut = response.referral
496 nameservers = [record.data for record in response.authority]
497 elif response.kind in ("ANSWER", "CNAME"):
498 cache_ttl = self.store_positive(response.answer)
499 return response.kind, [record.with_ttl(cache_ttl) for record in response.answer]
500 else:
501 self.store_negative(name, rtype, response)
502 return response.kind, []

504 def query_servers(self, name, rtype, cut, nameservers, depth, pending):
505 for address in self.destinations(nameservers, depth, pending):
506 response = self.ask(address, name, rtype, cut)
507 self.trace(
508 depth,
509 f"ASK {ipv4_text(address)} {name_text(name)} {rtype} -> {response.summary()}",
510 )
511 if response.kind not in FAILURES:
512 return response
513 return None

515 def destinations(self, nameservers, depth, pending):
516 if nameservers is None:
517 yield from self.root_hints
518 return
519 for host in sorted(nameservers, key=name_key):
520 yield from self.host_addresses(host, depth, pending)

522 def host_addresses(self, host, depth, pending):
523 entry = self.positive_entry(host, "A")
524 if entry is not None:
525 return sorted(record.data for record in entry.records)
526 if (host, "A") in pending:
527 self.trace(depth, f"GLUELESS {name_text(host)} -> LOOP")
528 return []
529 self.trace(depth, f"GLUELESS {name_text(host)}")
530 status, answer = self.resolve(host, "A", depth + 1, pending)
531 if status != "NOERROR":
532 return []
533 return sorted(record.data for record in answer if record.rtype == "A")

535 def resolve(self, name, rtype, depth, pending):
536 pending.append((name, rtype))
537 chain = []
538 seen = {name}
539 current = name
540 while True:
541 kind, records = self.lookup(current, rtype, depth, pending)
542 if kind != "CNAME":
543 break
544 chain.extend(records)
545 target = records[0].data
546 if target in seen:
547 kind = "SERVFAIL cname loop"
548 break
549 if len(chain) == MAX_CHAIN:
550 kind = "SERVFAIL cname chain too long"
551 break
552 seen.add(target)
553 current = target
554 pending.pop()
555 if kind == "ANSWER":
556 status, answer = "NOERROR", chain + records
557 elif kind.startswith("SERVFAIL"):
558 status, answer = kind, []
559 else:
560 status, answer = kind, chain
561 self.resolution_counts[status.split(" ")[0]] += 1
562 return status, answer
563
564 def resolve_and_report(self, name, rtype):
565 status, answer = self.resolve(name, rtype, 0, [])
566 for record in answer:
567 self.emit(f"answer {record.text()}")
568 self.emit(f"status {status}")
569 return status, answer

571 def address_resolution(self, host):
572 addresses = []
573 for rtype in ("A", "AAAA"):
574 status, answer = self.resolve(host, rtype, 0, [])
575 if status == "NOERROR":
576 addresses.extend(record.data_text() for record in answer if record.rtype == rtype)
577 return " ".join(addresses) if addresses else "no address"

580 command, *arguments = line.split(" ")

606 def reverse_command(self, address_word):
607 if ":" in address_word:
608 groups = require_ipv6(address_word)
609 address = ipv6_text(groups)
610 digits = "".join(f"{group:04x}" for group in groups)
611 name = tuple(reversed(digits)) + ("ip6", "arpa")
612 else:
613 address = ipv4_text(require_ipv4(address_word))
614 name = tuple(reversed(address.split("."))) + ("in-addr", "arpa")
615 self.emit(f"reverse {address} {name_text(name)}")
616 self.resolve_and_report(name, "PTR")

618 def mx_command(self, name_word):
619 name = parse_name(name_word, wildcard=True)
620 self.emit(f"mx {name_text(name)}")
621 status, answer = self.resolve_and_report(name, "MX")
622 if status == "NOERROR":
623 for record in answer:
624 if record.rtype != "MX":
625 continue
626 preference, host = record.data
627 if host == ROOT:
628 self.emit(f"exchange {preference} . -> null")
629 else:
630 addresses = self.address_resolution(host)
631 self.emit(f"exchange {preference} {name_text(host)} -> {addresses}")
632 elif status == "NODATA":
633 host = answer[-1].data if answer else name
634 addresses = self.address_resolution(host)
635 self.emit(f"implicit {name_text(host)} -> {addresses}")

637 def query_command(self, server_word, name_word, rtype):
638 server = require_ipv4(server_word)
639 name = parse_name(name_word, wildcard=True)
640 require_type(rtype)
641 response = self.ask(server, name, rtype, ROOT)
642 self.emit(f"query {server_word} {name_text(name)} {rtype} -> {response.summary()}")
643 sections = (
644 ("answer", response.answer),
645 ("authority", response.authority),
646 ("additional", response.additional),
647 )
648 for section, records in sections:
649 for record in records:
650 self.emit(f"{section} {record.text()}")

652 def dump_command(self, server_word, name_word):
653 server = require_ipv4(server_word)
654 name = parse_name(name_word)
655 prefix = f"dump {server_word} {name_text(name)}"
656 if not self.reachable(server):
657 self.emit(f"{prefix} -> TIMEOUT")
658 return
659 zone = next((zone for zone in self.zones_by_server[server] if zone.origin == name), None)
660 if zone is None:
661 self.emit(f"{prefix} -> REFUSED")
662 return
663 self.emit(f"{prefix} records={len(zone.records)}")
664 for record in sorted(zone.records, key=Record.order_key):
665 self.emit(f" {record.text()}")

667 def down_command(self, server_word):
668 server = require_ipv4(server_word)
669 if server in self.down:
670 raise InputError(f"server {server_word} already down")
671 self.down.add(server)
672 self.emit(f"down {server_word}")
673
674 def up_command(self, server_word):
675 server = require_ipv4(server_word)
676 if server not in self.down:
677 raise InputError(f"server {server_word} not down")
678 self.down.remove(server)
679 self.emit(f"up {server_word}")

681 def set_command(self, key, value_word):
682 if key not in self.settings:
683 raise InputError(f"bad setting {key}")
684 value = parse_integer(value_word)
685 if value is None or not 1 <= value <= MAX_TTL:
686 raise InputError(f"bad value {value_word}")
687 updated = {**self.settings, key: value}
688 if updated["MINTTL"] > updated["MAXTTL"]:
689 raise InputError("minttl above maxttl")
690 self.settings = updated
691 self.emit(f"set {key}={value}")
692
693 def time_command(self, seconds_word):
694 seconds = parse_integer(seconds_word)
695 if seconds is None or not 1 <= seconds <= MAX_TIME_STEP:
696 raise InputError(f"bad time {seconds_word}")
697 self.clock += seconds
698 expired = [key for key, entry in self.cache.items() if entry.expiry <= self.clock]
699 for key in expired:
700 del self.cache[key]
701 self.emit(f"time now={self.clock} expired={len(expired)}")
702
703 def selected_keys(self, name_word):
704 if name_word is None:
705 return list(self.cache)
706 name = parse_name(name_word)
707 return [key for key in self.cache if is_at_or_below(key[0], name)]
708
709 def cache_command(self, name_word=None):
710 keys = sorted(
711 self.selected_keys(name_word),
712 key=lambda key: (name_key(key[0]), TYPE_ORDER.index(key[1])),
713 )
714 self.emit(f"cache entries={len(keys)}")
715 for name, rtype in keys:
716 entry = self.cache[(name, rtype)]
717 if entry.kind == "POS":
718 for record in self.aged(entry):
719 self.emit(f" {record.text()}")
720 else:
721 remaining = entry.expiry - self.clock
722 self.emit(f" {name_text(name)} {remaining} {rtype} {entry.kind}")
723
724 def flush_command(self, name_word=None):
725 keys = self.selected_keys(name_word)
726 for key in keys:
727 del self.cache[key]
728 self.emit(f"flush removed={len(keys)}")

730 def stats_command(self):
731 total = sum(sum(counts.values()) for counts in self.query_counts.values())
732 self.emit(f"stats queries={total} hits={self.hits}")
733 for address in sorted(self.query_counts):
734 counts = self.query_counts[address]
735 details = " ".join(f"{kind.lower()}={counts[kind]}" for kind in RESPONSE_KINDS)
736 self.emit(f" {ipv4_text(address)} sent={sum(counts.values())} {details}")
737 details = " ".join(
738 f"{status.lower()}={self.resolution_counts[status]}" for status in STATUSES
739 )
740 self.emit(f"resolutions {details}")
