20std::vector<std::string> split(const std::string& text, char separator) {
21 std::vector<std::string> result;
22 std::size_t start = 0;
23 for (std::size_t i = 0; i <= text.size(); ++i) {
24 if (i == text.size() || text[i] == separator) {
25 result.push_back(text.substr(start, i - start));
26 start = i + 1;
27 }
28 }
29 return result;
30}

32std::string join(const std::vector<std::string>& parts, const std::string& separator) {
33 std::string result;
34 for (std::size_t i = 0; i < parts.size(); ++i) {
35 if (i) result += separator;
36 result += parts[i];
37 }
38 return result;
39}

41std::string lower(std::string text) {
42 for (char& c : text) if (c >= 'A' && c <= 'Z') c += 'a' - 'A';
43 return text;
44}

46std::optional<int> decimal(const std::string& text, std::size_t maximum = 9) {
47 if (text.empty() || text.size() > maximum || (text.size() > 1 && text[0] == '0')) return std::nullopt;
48 int value = 0;
49 for (char c : text) {
50 if (c < '0' || c > '9') return std::nullopt;
51 value = value * 10 + c - '0';
52 }
53 return value;
54}

56std::optional<Address> address4(const std::string& text) {
57 std::vector<std::string> fields = split(text, '.');
58 if (fields.size() != 4) return std::nullopt;
59 Address address = 0;
60 for (const std::string& field : fields) {
61 std::optional<int> value = decimal(field, 3);
62 if (!value || *value > 255) return std::nullopt;
63 address = address * 256 + static_cast<Address>(*value);
64 }
65 return address;
66}

68std::string address_text(Address address) {
69 return std::to_string(address >> 24) + "." + std::to_string((address >> 16) & 255) + "." +
70 std::to_string((address >> 8) & 255) + "." + std::to_string(address & 255);
71}

79std::optional<std::vector<unsigned>> groups(const std::string& text) {
80 std::vector<unsigned> result;
81 if (text.empty()) return result;
82 for (const std::string& group : split(text, ':')) {
83 if (group.empty() || group.size() > 4) return std::nullopt;
84 unsigned value = 0;
85 for (char c : group) {
86 unsigned h;
87 if (c >= '0' && c <= '9') h = static_cast<unsigned>(c - '0');
88 else if (c >= 'a' && c <= 'f') h = static_cast<unsigned>(c - 'a' + 10);
89 else if (c >= 'A' && c <= 'F') h = static_cast<unsigned>(c - 'A' + 10);
90 else return std::nullopt;
91 value = value * 16 + h;
92 }
93 result.push_back(value);
94 }
95 return result;
96}

98std::optional<Address6> address6(const std::string& text) {
99 std::size_t colon = text.find("::");
100 Address6 result{};
101 if (colon == std::string::npos) {
102 std::optional<std::vector<unsigned>> values = groups(text);
103 if (!values || values->size() != 8) return std::nullopt;
104 std::copy(values->begin(), values->end(), result.begin());
105 } else {
106 if (text.find("::", colon + 2) != std::string::npos) return std::nullopt;
107 std::optional<std::vector<unsigned>> first = groups(text.substr(0, colon));
108 std::optional<std::vector<unsigned>> last = groups(text.substr(colon + 2));
109 if (!first || !last || first->size() + last->size() > 7) return std::nullopt;
110 std::copy(first->begin(), first->end(), result.begin());
111 std::copy(last->begin(), last->end(), result.end() - static_cast<std::ptrdiff_t>(last->size()));
112 }
113 return result;
114}

116std::string address_text(const Address6& address) {
117 std::vector<std::string> fields;
118 for (unsigned group : address) {
119 std::ostringstream text;
120 text << std::hex << group;
121 fields.push_back(text.str());
122 }
123 std::size_t best_start = 0, best_length = 0;
124 for (std::size_t i = 0; i < 8;) {
125 if (address[i] != 0) {
126 ++i;
127 continue;
128 }
129 std::size_t end = i;
130 while (end < 8 && address[end] == 0) ++end;
131 if (end - i > best_length && end - i >= 2) {
132 best_start = i;
133 best_length = end - i;
134 }
135 i = end;
136 }
137 if (!best_length) return join(fields, ":");
138 return join(std::vector<std::string>(fields.begin(), fields.begin() + best_start), ":") + "::" +
139 join(std::vector<std::string>(fields.begin() + best_start + best_length, fields.end()), ":");
140}

142struct Name {
143 std::vector<std::string> labels;
144 std::string text() const { return labels.empty() ? "." : join(labels, ".") + "."; }
145 bool operator==(const Name& other) const { return labels == other.labels; }
146 bool operator!=(const Name& other) const { return !(*this == other); }
147 bool operator<(const Name& other) const {
148 return std::lexicographical_compare(labels.rbegin(), labels.rend(), other.labels.rbegin(), other.labels.rend());
149 }
150 Name suffix(std::size_t length) const {
151 return Name{std::vector<std::string>(labels.end() - static_cast<std::ptrdiff_t>(length), labels.end())};
152 }
153};

155bool below(const Name& name, const Name& ancestor) {
156 return name.labels.size() >= ancestor.labels.size() && name.suffix(ancestor.labels.size()) == ancestor;
157}

159Name require_name(const std::string& word, bool wildcard = false, const Name* origin = nullptr) {
160 if (origin && word == "@") return *origin;
161 Name name;
162 if (word == ".") return name;
163 bool absolute = !word.empty() && word.back() == '.';
164 std::string base = absolute ? word.substr(0, word.size() - 1) : word;
165 std::vector<std::string> labels = split(base, '.');
166 for (std::size_t i = 0; i < labels.size(); ++i) {
167 std::string label = lower(labels[i]);
168 if (i == 0 && wildcard && label == "*") {
169 name.labels.push_back(label);
170 continue;
171 }
172 if (label.empty() || label.size() > 63 || label.front() == '-' || label.back() == '-')
173 throw std::runtime_error("bad name " + word);
174 for (char c : label)
175 if (!((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '-'))
176 throw std::runtime_error("bad name " + word);
177 name.labels.push_back(label);
178 }
179 if (origin && !absolute) name.labels.insert(name.labels.end(), origin->labels.begin(), origin->labels.end());
180 return name;
181}

183int require_type(const std::string& word) {
184 auto found = std::find(types.begin(), types.end(), word);
185 if (found == types.end()) throw std::runtime_error("bad type " + word);
186 return static_cast<int>(found - types.begin());
187}

189int ttl_value(const std::string& word) {
190 std::optional<int> integer = decimal(word);
191 long long total = 0;
192 bool valid = true;
193 if (integer) total = *integer;
194 else {
195 static const std::string units = "wdhms";
196 static const std::array<int, 5> seconds{604800, 86400, 3600, 60, 1};
197 std::size_t position = 0;
198 int previous = -1;
199 while (position < word.size()) {
200 std::size_t start = position;
201 while (position < word.size() && word[position] >= '0' && word[position] <= '9') ++position;
202 std::optional<int> number = decimal(word.substr(start, position - start), 3);
203 if (!number || position == word.size()) {
204 valid = false;
205 break;
206 }
207 char unit = lower(word.substr(position++, 1))[0];
208 std::size_t index = units.find(unit);
209 if (index == std::string::npos || static_cast<int>(index) <= previous) {
210 valid = false;
211 break;
212 }
213 previous = static_cast<int>(index);
214 total += static_cast<long long>(*number) * seconds[index];
215 }
216 }
217 if (!valid || total < 1 || total > 604800) throw std::runtime_error("bad ttl " + word);
218 return static_cast<int>(total);
219}

221struct Record {
222 Name owner;
223 int ttl = 0, type = 0;
224 Address ipv4 = 0;
225 Address6 ipv6{};
226 Name target, rname;
227 int preference = 0;
228 std::vector<int> numbers;
229 std::string text_data;
230
231 std::string data() const {
232 if (type == 0) return address_text(ipv4);
233 if (type == 7) return address_text(ipv6);
234 if (type == 5) return std::to_string(preference) + " " + target.text();
235 if (type == 6) return text_data;
236 if (type != 3) return target.text();
237 std::string result = target.text() + " " + rname.text();
238 for (int number : numbers) result += " " + std::to_string(number);
239 return result;
240 }
241 std::string line() const { return owner.text() + " " + std::to_string(ttl) + " " + types[type] + " " + data(); }
242};
243
244bool data_less(const Record& a, const Record& b) {
245 if (a.type == 0) return a.ipv4 < b.ipv4;
246 if (a.type == 7) return a.ipv6 < b.ipv6;
247 if (a.type == 6) return a.text_data < b.text_data;
248 if (a.type == 5 && a.preference != b.preference) return a.preference < b.preference;
249 if (a.target != b.target) return a.target < b.target;
250 if (a.type == 3) {
251 if (a.rname != b.rname) return a.rname < b.rname;
252 return a.numbers < b.numbers;
253 }
254 return false;
255}
256
257bool record_less(const Record& a, const Record& b) {
258 if (a.owner != b.owner) return a.owner < b.owner;
259 if (a.type != b.type) return a.type < b.type;
260 return data_less(a, b);
261}

291class Resolver {
292 std::vector<Address> roots;
293 std::vector<Zone> zones;
294 std::optional<std::size_t> current;
295 std::set<Address> down;
296 std::map<Key, Entry> cache;
297 int clock = 0, minttl = 1, maxttl = 86400, negttl = 3600, hits = 0;

301 bool known(Address server) const {
302 return std::any_of(zones.begin(), zones.end(), [server](const Zone& zone) { return zone.server == server; });
303 }
304 std::vector<Record> records(const Zone& zone, const Name& owner, std::optional<int> type = std::nullopt) const {
305 std::vector<Record> result;
306 for (const Record& record : zone.records) if (record.owner == owner && (!type || record.type == *type)) result.push_back(record);
307 std::sort(result.begin(), result.end(), record_less);
308 return result;
309 }
310 bool exists(const Zone& zone, const Name& name) const {
311 return std::any_of(zone.records.begin(), zone.records.end(), [&](const Record& record) { return below(record.owner, name); });
312 }
313 Response negative(const Zone& zone, const std::string& kind) const {
314 Response result(kind);
315 for (const Record& record : zone.records) if (record.type == 3) {
316 result.authority.push_back(record);
317 break;
318 }
319 return result;
320 }
321 Response authoritative(Address server, const Name& name, int type, const Name& cut) const {
322 if (down.count(server) || !known(server)) return Response("TIMEOUT");
323 const Zone* chosen = nullptr;
324 for (const Zone& zone : zones)
325 if (zone.server == server && below(name, zone.origin) && (!chosen || zone.origin.labels.size() > chosen->origin.labels.size())) chosen = &zone;
326 if (!chosen || chosen->origin.labels.size() < cut.labels.size()) return Response("LAME");
327 const Zone& zone = *chosen;
328 for (std::size_t length = zone.origin.labels.size() + 1; length <= name.labels.size(); ++length) {
329 Name ancestor = name.suffix(length);
330 std::vector<Record> nameservers = records(zone, ancestor, 1);
331 if (nameservers.empty()) continue;
332 Response result("REFERRAL");
333 result.referral = ancestor;
334 result.authority = nameservers;
335 for (const Record& ns : nameservers) {
336 std::vector<Record> glue = records(zone, ns.target, 0);
337 result.additional.insert(result.additional.end(), glue.begin(), glue.end());
338 }
339 return result;
340 }
341 std::vector<Record> candidates = records(zone, name);
342 if (candidates.empty()) {
343 if (exists(zone, name)) return negative(zone, "NODATA");
344 Name ancestor;
345 for (std::size_t removed = 1; removed <= name.labels.size(); ++removed) {
346 ancestor = name.suffix(name.labels.size() - removed);
347 if (exists(zone, ancestor)) break;
348 }
349 Name wildcard = ancestor;
350 wildcard.labels.insert(wildcard.labels.begin(), "*");
351 candidates = records(zone, wildcard);
352 if (candidates.empty()) return negative(zone, "NXDOMAIN");
353 }
354 Response result("ANSWER");
355 for (Record record : candidates) if (record.type == type) {
356 record.owner = name;
357 result.answer.push_back(std::move(record));
358 }
359 if (!result.answer.empty()) return result;
360 if (type != 2) for (Record record : candidates) if (record.type == 2) {
361 result.kind = "CNAME";
362 record.owner = name;
363 result.answer.push_back(std::move(record));
364 return result;
365 }
366 return negative(zone, "NODATA");
367 }
368 Response ask(Address server, const Name& name, int type, const Name& cut) {
369 Response result = authoritative(server, name, type, cut);
370 auto found = std::find(response_kinds.begin(), response_kinds.end(), result.kind);
371 ++queries[server][static_cast<std::size_t>(found - response_kinds.begin())];
372 return result;
373 }
374 void trace(int depth, const std::string& text) const { std::cout << std::string(2 * (depth + 1), ' ') << text << '\n'; }
375 std::vector<Record> cached_records(const Entry& entry) const {
376 std::vector<Record> result = entry.records;
377 for (Record& record : result) record.ttl = entry.expiry - clock;
378 return result;
379 }
380 int store(std::vector<Record> records) {
381 int ttl = records.front().ttl;
382 for (const Record& record : records) ttl = std::min(ttl, record.ttl);
383 ttl = std::min(std::max(ttl, minttl), maxttl);
384 std::sort(records.begin(), records.end(), data_less);
385 Key key{records.front().owner, records.front().type};
386 cache[key] = {"POS", std::move(records), clock + ttl};
387 return ttl;
388 }
389 std::optional<Response> query_servers(const Name& name, int type, const Name& cut,
390 const std::optional<std::vector<Name>>& nameservers,
391 int depth, std::vector<Key>& resolving) {
392 auto query = [&](Address address) -> std::optional<Response> {
393 Response result = ask(address, name, type, cut);
394 trace(depth, "ASK " + address_text(address) + " " + name.text() + " " + types[type] + " -> " + result.description());
395 if (result.kind == "LAME" || result.kind == "TIMEOUT") return std::nullopt;
396 return result;
397 };
398 if (!nameservers) {
399 for (Address address : roots) {
400 std::optional<Response> result = query(address);
401 if (result) return result;
402 }
403 } else {
404 std::vector<Name> ordered = *nameservers;
405 std::sort(ordered.begin(), ordered.end());
406 for (const Name& host : ordered) {
407 std::set<Address> addresses;
408 auto cached = cache.find({host, 0});
409 if (cached != cache.end() && cached->second.kind == "POS") {
410 for (const Record& record : cached->second.records) addresses.insert(record.ipv4);
411 } else if (std::find(resolving.begin(), resolving.end(), Key{host, 0}) != resolving.end()) {
412 trace(depth, "GLUELESS " + host.text() + " -> LOOP");
413 } else {
414 trace(depth, "GLUELESS " + host.text());
415 Resolution result = resolve(host, 0, depth + 1, resolving);
416 if (result.status == "NOERROR") for (const Record& record : result.answer) if (record.type == 0) addresses.insert(record.ipv4);
417 }
418 for (Address address : addresses) {
419 std::optional<Response> result = query(address);
420 if (result) return result;
421 }
422 }
423 }
424 return std::nullopt;
425 }
426 Response lookup(const Name& name, int type, int depth, std::vector<Key>& resolving) {
427 auto cache_hit = [&](Response result) {
428 ++hits;
429 trace(depth, "CACHE HIT " + name.text() + " " + types[type] + " -> " + result.description());
430 return result;
431 };
432 auto exact = cache.find({name, type});
433 if (exact != cache.end()) {
434 Response result(exact->second.kind == "POS" ? "ANSWER" : exact->second.kind);
435 if (exact->second.kind == "POS") result.answer = cached_records(exact->second);
436 return cache_hit(std::move(result));
437 }
438 for (const auto& entry : cache)
439 if (entry.second.kind == "NXDOMAIN" && below(name, entry.first.first)) return cache_hit(Response("NXDOMAIN"));
440 if (type != 2) {
441 auto cname = cache.find({name, 2});
442 if (cname != cache.end() && cname->second.kind == "POS") {
443 Response result("CNAME");
444 result.answer = cached_records(cname->second);
445 return cache_hit(std::move(result));
446 }
447 }
448 Name cut;
449 std::optional<std::vector<Name>> nameservers;
450 for (const auto& entry : cache) {
451 if (entry.first.second == 1 && entry.second.kind == "POS" && below(name, entry.first.first) &&
452 (!nameservers || entry.first.first.labels.size() > cut.labels.size())) {
453 cut = entry.first.first;
454 nameservers.emplace();
455 for (const Record& record : entry.second.records) nameservers->push_back(record.target);
456 }
457 }
458 if (nameservers) {
459 ++hits;
460 trace(depth, "CACHE HIT " + name.text() + " " + types[type] + " -> REFERRAL " + cut.text());
461 }
462 while (true) {
463 std::optional<Response> obtained = query_servers(name, type, cut, nameservers, depth, resolving);
464 if (!obtained) return Response("SERVFAIL no server for " + cut.text());
465 Response result = std::move(*obtained);
466 if (result.kind == "REFERRAL") {
467 store(result.authority);
468 std::map<Name, std::vector<Record>> glue;
469 for (const Record& record : result.additional) glue[record.owner].push_back(record);
470 for (const auto& entry : glue) store(entry.second);
471 cut = result.referral;
472 nameservers.emplace();
473 for (const Record& record : result.authority) nameservers->push_back(record.target);
474 } else if (result.kind == "ANSWER" || result.kind == "CNAME") {
475 int ttl = store(result.answer);
476 for (Record& record : result.answer) record.ttl = ttl;
477 return result;
478 } else {
479 const Record& soa = result.authority.front();
480 int ttl = std::min({soa.ttl, soa.numbers.back(), negttl});
481 cache[{name, type}] = {result.kind, {}, clock + ttl};
482 return result;
483 }
484 }
485 }
486 Resolution resolve(const Name& name, int type, int depth, std::vector<Key>& resolving) {
487 resolving.emplace_back(name, type);
488 std::vector<Record> chain;
489 std::set<Name> seen{name};
490 Name current_name = name;
491 Resolution resolution;
492 while (true) {
493 Response result = lookup(current_name, type, depth, resolving);
494 if (result.kind == "CNAME") {
495 const Record& cname = result.answer.front();
496 chain.push_back(cname);
497 if (seen.count(cname.target)) {
498 resolution.status = "SERVFAIL cname loop";
499 break;
500 }
501 if (chain.size() == 9) {
502 resolution.status = "SERVFAIL cname chain too long";
503 break;
504 }
505 seen.insert(cname.target);
506 current_name = cname.target;
507 } else {
508 resolution.status = result.kind == "ANSWER" ? "NOERROR" : result.kind;
509 if (resolution.status.compare(0, 8, "SERVFAIL") != 0) {
510 resolution.answer = std::move(chain);
511 if (result.kind == "ANSWER") resolution.answer.insert(resolution.answer.end(), result.answer.begin(), result.answer.end());
512 }
513 break;
514 }
515 }
516 resolving.pop_back();
517 ++resolutions[resolution.status.compare(0, 8, "SERVFAIL") == 0 ? "SERVFAIL" : resolution.status];
518 return resolution;
519 }
520 Resolution resolve(const Name& name, int type) {
521 std::vector<Key> resolving;
522 return resolve(name, type, 0, resolving);
523 }

536 void add_zone_line(const std::vector<std::string>& w, int line) {
537 if (w[0] == "ROOT") {
538 std::vector<Address> added;
539 for (std::size_t i = 1; i < w.size(); ++i) {
540 Address address = require_address(w[i]);
541 if (std::find(roots.begin(), roots.end(), address) != roots.end() ||
542 std::find(added.begin(), added.end(), address) != added.end()) throw std::runtime_error("duplicate root " + w[i]);
543 added.push_back(address);
544 }
545 roots.insert(roots.end(), added.begin(), added.end());
546 return;
547 }
548 if (w[0] == "ZONE") {
549 current.reset();
550 Name origin = require_name(w[1]);
551 Address server = require_address(w[2]);
552 for (const Zone& zone : zones)
553 if (zone.origin == origin && zone.server == server)
554 throw std::runtime_error("duplicate zone " + origin.text() + " at " + w[2]);
555 zones.push_back({origin, server, line, {}});
556 current = zones.size() - 1;
557 return;
558 }
559 if (!current) throw std::runtime_error("no current zone");
560 Zone& zone = zones[*current];
561 Record record;
562 record.owner = require_name(w[0], true, &zone.origin);
563 if (!below(record.owner, zone.origin)) throw std::runtime_error("out of zone " + record.owner.text());
564 record.ttl = ttl_value(w[1]);
565 record.type = require_type(w[2]);
566 std::size_t count = w.size() - 3;
567 std::size_t expected = record.type == 5 ? 2 : record.type == 3 ? 7 : 1;
568 if (record.type != 6 && count != expected) throw std::runtime_error("wrong field count");
569 if (record.type == 0) record.ipv4 = require_address(w[3]);
570 else if (record.type == 7) {
571 std::optional<Address6> address = address6(w[3]);
572 if (!address) throw std::runtime_error("bad address " + w[3]);
573 record.ipv6 = *address;
574 } else if (record.type == 1 || record.type == 2 || record.type == 4) record.target = require_name(w[3], false, &zone.origin);
575 else if (record.type == 5) {
576 std::optional<int> preference = decimal(w[3]);
577 if (!preference || *preference > 65535) throw std::runtime_error("bad preference " + w[3]);
578 record.preference = *preference;
579 record.target = require_name(w[4], false, &zone.origin);
580 } else if (record.type == 6) {
581 record.text_data = join(std::vector<std::string>(w.begin() + 3, w.end()), " ");
582 const std::string& text = record.text_data;
583 if (text.size() < 2 || text.front() != '"' || text.back() != '"' || text.find('"', 1) != text.size() - 1)
584 throw std::runtime_error("bad text " + text);
585 } else {
586 record.target = require_name(w[3], false, &zone.origin);
587 record.rname = require_name(w[4], false, &zone.origin);
588 for (std::size_t i = 5; i <= 8; ++i) {
589 std::optional<int> number = decimal(w[i]);
590 if (!number) throw std::runtime_error("bad number " + w[i]);
591 record.numbers.push_back(*number);
592 }
593 std::optional<int> minimum = decimal(w[9]);
594 if (!minimum || *minimum < 1 || *minimum > 86400) throw std::runtime_error("bad minimum " + w[9]);
595 record.numbers.push_back(*minimum);
596 }
597 if (record.type == 1 && !record.owner.labels.empty() && record.owner.labels[0] == "*") throw std::runtime_error("ns at wildcard");
598 if (record.type == 3 && record.owner != zone.origin) throw std::runtime_error("soa not at origin");
599 if (record.type == 3 && std::any_of(zone.records.begin(), zone.records.end(), [](const Record& r) { return r.type == 3; }))
600 throw std::runtime_error("duplicate soa");
601 for (const Record& existing : zone.records)
602 if (existing.owner == record.owner && existing.type == record.type && !data_less(existing, record) && !data_less(record, existing))
603 throw std::runtime_error("duplicate record");
604 for (const Record& existing : zone.records)
605 if (existing.owner == record.owner && (record.type == 2 || existing.type == 2))
606 throw std::runtime_error("cname conflict " + record.owner.text());
607 zone.records.push_back(std::move(record));
608 }

610 void execute(const std::vector<std::string>& w) {
611 const std::string& command = w[0];
612 if (command == "RESOLVE" || command == "MX") {
613 Name name = require_name(w[1], true);
614 int type = command == "MX" ? 5 : require_type(w[2]);
615 std::cout << (command == "MX" ? "mx " : "resolve ") << name.text();
616 if (command != "MX") std::cout << ' ' << types[type];
617 std::cout << '\n';
618 Resolution result = resolve(name, type);
619 write_result(result);
620 if (command == "MX") {
621 if (result.status == "NOERROR") for (const Record& record : result.answer) if (record.type == 5) {
622 if (record.target.labels.empty()) std::cout << "exchange " << record.preference << " . -> null\n";
623 else addresses(record.target, "exchange " + std::to_string(record.preference) + " ");
624 }
625 if (result.status == "NODATA") {
626 Name host = name;
627 for (const Record& record : result.answer) if (record.type == 2) host = record.target;
628 addresses(host, "implicit ");
629 }
630 }
631 } else if (command == "REVERSE") {
632 Name name;
633 std::string address;
634 if (w[1].find(':') == std::string::npos) {
635 Address value = require_address(w[1]);
636 address = address_text(value);
637 std::vector<std::string> parts = split(address, '.');
638 std::reverse(parts.begin(), parts.end());
639 name = require_name(join(parts, ".") + ".in-addr.arpa.");
640 } else {
641 std::optional<Address6> value = address6(w[1]);
642 if (!value) throw std::runtime_error("bad address " + w[1]);
643 address = address_text(*value);
644 std::ostringstream digits;
645 digits << std::hex << std::setfill('0');
646 for (unsigned group : *value) digits << std::setw(4) << group;
647 std::string reversed;
648 std::string raw = digits.str();
649 for (auto i = raw.rbegin(); i != raw.rend(); ++i) {
650 reversed += *i;
651 reversed += '.';
652 }
653 name = require_name(reversed + "ip6.arpa.");
654 }
655 std::cout << "reverse " << address << ' ' << name.text() << '\n';
656 write_result(resolve(name, 4));
657 } else if (command == "QUERY" || command == "DUMP") {
658 Address server = require_address(w[1]);
659 Name name = require_name(w[2], command == "QUERY");
660 if (command == "QUERY") {
661 int type = require_type(w[3]);
662 Response result = ask(server, name, type, Name{});
663 std::cout << "query " << address_text(server) << ' ' << name.text() << ' ' << types[type] << " -> " << result.description() << '\n';
664 for (const Record& record : result.answer) std::cout << "answer " << record.line() << '\n';
665 for (const Record& record : result.authority) std::cout << "authority " << record.line() << '\n';
666 for (const Record& record : result.additional) std::cout << "additional " << record.line() << '\n';
667 } else {
668 std::string prefix = "dump " + address_text(server) + " " + name.text();
669 if (down.count(server) || !known(server)) {
670 std::cout << prefix << " -> TIMEOUT\n";
671 return;
672 }
673 for (const Zone& zone : zones) if (zone.server == server && zone.origin == name) {
674 std::vector<Record> ordered = zone.records;
675 std::sort(ordered.begin(), ordered.end(), record_less);
676 std::cout << prefix << " records=" << ordered.size() << '\n';
677 for (const Record& record : ordered) std::cout << " " << record.line() << '\n';
678 return;
679 }
680 std::cout << prefix << " -> REFUSED\n";
681 }
682 } else if (command == "DOWN" || command == "UP") {
683 Address server = require_address(w[1]);
684 if (command == "DOWN" && down.count(server)) throw std::runtime_error("server " + w[1] + " already down");
685 if (command == "UP" && !down.count(server)) throw std::runtime_error("server " + w[1] + " not down");
686 if (command == "DOWN") down.insert(server);
687 else down.erase(server);
688 std::cout << lower(command) << ' ' << address_text(server) << '\n';
689 } else if (command == "SET") {
690 if (w[1] != "MINTTL" && w[1] != "MAXTTL" && w[1] != "NEGTTL") throw std::runtime_error("bad setting " + w[1]);
691 std::optional<int> value = decimal(w[2]);
692 if (!value || *value < 1 || *value > 604800) throw std::runtime_error("bad value " + w[2]);
693 if ((w[1] == "MINTTL" && *value > maxttl) || (w[1] == "MAXTTL" && *value < minttl))
694 throw std::runtime_error("minttl above maxttl");
695 if (w[1] == "MINTTL") minttl = *value;
696 else if (w[1] == "MAXTTL") maxttl = *value;
697 else negttl = *value;
698 std::cout << "set " << w[1] << '=' << *value << '\n';
699 } else if (command == "TIME") {
700 std::optional<int> seconds = decimal(w[1]);
701 if (!seconds || *seconds < 1 || *seconds > 1000000) throw std::runtime_error("bad time " + w[1]);
702 clock += *seconds;
703 std::size_t removed = 0;
704 for (auto it = cache.begin(); it != cache.end();) {
705 if (it->second.expiry <= clock) {
706 it = cache.erase(it);
707 ++removed;
708 }
709 else ++it;
710 }
711 std::cout << "time now=" << clock << " expired=" << removed << '\n';
712 } else if (command == "CACHE" || command == "FLUSH") {
713 std::optional<Name> name;
714 if (w.size() > 1) name = require_name(w[1]);
715 std::vector<Key> selected;
716 for (const auto& entry : cache) if (!name || below(entry.first.first, *name)) selected.push_back(entry.first);
717 if (command == "FLUSH") {
718 for (const Key& key : selected) cache.erase(key);
719 std::cout << "flush removed=" << selected.size() << '\n';
720 } else {
721 std::cout << "cache entries=" << selected.size() << '\n';
722 for (const Key& key : selected) {
723 const Entry& entry = cache.at(key);
724 if (entry.kind == "POS") for (const Record& record : cached_records(entry)) std::cout << " " << record.line() << '\n';
725 else std::cout << " " << key.first.text() << ' ' << entry.expiry - clock << ' ' << types[key.second] << ' ' << entry.kind << '\n';
726 }
727 }
728 } else if (command == "STATS") {
729 int total = 0;
730 for (const auto& entry : queries) for (int count : entry.second) total += count;
731 std::cout << "stats queries=" << total << " hits=" << hits << '\n';
732 for (const auto& entry : queries) {
733 int sent = 0;
734 for (int count : entry.second) sent += count;
735 std::cout << " " << address_text(entry.first) << " sent=" << sent;
736 for (std::size_t i = 0; i < response_kinds.size(); ++i)
737 std::cout << ' ' << lower(response_kinds[i]) << '=' << entry.second[i];
738 std::cout << '\n';
739 }
740 std::cout << "resolutions noerror=" << resolutions["NOERROR"] << " nodata=" << resolutions["NODATA"]
741 << " nxdomain=" << resolutions["NXDOMAIN"] << " servfail=" << resolutions["SERVFAIL"] << '\n';
742 }
743 }
744public:
745 void zone_line(const std::string& line, int number) {
746 try { add_zone_line(split(line, ' '), number); }
747 catch (const std::runtime_error& error) { std::cout << "error line " << number << ": " << error.what() << '\n'; }
748 }
749 void finish_zones() {
750 std::vector<Zone> valid;
751 std::size_t total = 0;
752 for (Zone& zone : zones) {
753 bool soa = std::any_of(zone.records.begin(), zone.records.end(), [](const Record& record) { return record.type == 3; });
754 if (!soa) std::cout << "error line " << zone.line << ": zone " << zone.origin.text() << " at " << address_text(zone.server) << " has no soa\n";
755 else {
756 std::cout << "zone " << zone.origin.text() << " at " << address_text(zone.server) << " records=" << zone.records.size() << '\n';
757 total += zone.records.size();
758 valid.push_back(std::move(zone));
759 }
760 }
761 zones = std::move(valid);
762 current.reset();
763 std::cout << "loaded roots=" << roots.size() << " zones=" << zones.size() << " records=" << total << '\n';
764 }
765 void command(const std::string& line, int number) {
766 try { execute(split(line, ' ')); }
767 catch (const std::runtime_error& error) { std::cout << "error line " << number << ": " << error.what() << '\n'; }
768 }
769};
