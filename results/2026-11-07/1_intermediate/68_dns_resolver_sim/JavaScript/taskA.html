13const integerWord = /^(?:0|[1-9][0-9]{0,8})$/;
14const labels = (name) => (name === "." ? [] : name.slice(0, -1).split("."));
15const below = (name, ancestor) =>

17const ancestors = (name) => {

21const compareText = (a, b) => (a === b ? 0 : a < b ? -1 : 1);
22function nameOrder(a, b) {
23 const left = labels(a).reverse();
24 const right = labels(b).reverse();
25 for (let i = 0; i < Math.min(left.length, right.length); i++) {
26 const order = compareText(left[i], right[i]);
27 if (order !== 0) return order;
28 }
29 return left.length - right.length;
30}
31function parseName(word, wildcard = false, origin = null) {
32 if (origin !== null && word === "@") return origin;
33 if (word === ".") return ".";
34 const absolute = word.endsWith(".");
35 const parts = (absolute ? word.slice(0, -1) : word).split(".");
36 if (
37 parts.some(
38 (part, i) =>
39 !(wildcard && i === 0 && part === "*") &&
40 !/^[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?$/.test(part),
41 )
42 )
43 throw new Error(`bad name ${word}`);
44 const name = parts.join(".").toLowerCase() + ".";
45 return origin !== null && !absolute && origin !== "." ? name + origin : name;
46}
47function ipv4(word) {
48 const parts = word.split(".");
49 if (
50 parts.length !== 4 ||
51 parts.some(
52 (part) => !/^(?:0|[1-9][0-9]{0,2})$/.test(part) || Number(part) > 255,
53 )
54 )
55 throw new Error(`bad address ${word}`);
56 return parts.reduce((value, part) => value * 256 + Number(part), 0);
57}
58function addressText(value) {
59 return [24, 16, 8, 0]
60 .map((shift) => Math.floor(value / 2 ** shift) % 256)
61 .join(".");
62}
63function ipv6(word) {
64 const halves = word.split("::");
65 if (halves.length > 2) throw new Error(`bad address ${word}`);
66 const left = halves[0] === "" ? [] : halves[0].split(":");
67 const right =
68 halves.length === 2 && halves[1] !== "" ? halves[1].split(":") : [];
69 if (
70 [...left, ...right].some((group) => !/^[0-9A-Fa-f]{1,4}$/.test(group)) ||
71 (halves.length === 1 ? left.length !== 8 : left.length + right.length > 7)
72 )
73 throw new Error(`bad address ${word}`);
74 return [
75 ...left.map((group) => parseInt(group, 16)),
76 ...Array(halves.length === 2 ? 8 - left.length - right.length : 0).fill(0),
77 ...right.map((group) => parseInt(group, 16)),
78 ];
79}
80function ipv6Text(groups) {
81 let best = -1;
82 let length = 0;
83 for (let start = 0; start < 8;) {
84 if (groups[start] !== 0) {
85 start++;
86 continue;
87 }
88 let end = start;
89 while (end < 8 && groups[end] === 0) end++;
90 if (end - start >= 2 && end - start > length) {
91 best = start;
92 length = end - start;
93 }
94 start = end;
95 }
96 const words = groups.map((group) => group.toString(16));
97 return best < 0
98 ? words.join(":")
99 : words.slice(0, best).join(":") +
100 "::" +
101 words.slice(best + length).join(":");
102}
103function ttl(word) {
104 let result = 0;
105 if (integerWord.test(word)) result = Number(word);
106 else {
107 const units = "wdhms";
108 const seconds = [604800, 86400, 3600, 60, 1];
109 let position = 0;
110 let previous = -1;
111 for (const match of word.matchAll(/(0|[1-9][0-9]{0,2})([wdhms])/gi)) {
112 const unit = units.indexOf(match[2].toLowerCase());
113 if (match.index !== position || unit <= previous)
114 throw new Error(`bad ttl ${word}`);
115 result += Number(match[1]) * seconds[unit];
116 previous = unit;
117 position += match[0].length;
118 }
119 if (position !== word.length || position === 0)
120 throw new Error(`bad ttl ${word}`);
121 }
122 if (result < 1 || result > 604800) throw new Error(`bad ttl ${word}`);
123 return result;
124}
125function checkType(type) {
126 if (!types.includes(type)) throw new Error(`bad type ${type}`);
127}
128function dataOrder(a, b) {
129 if (a.type === "A") return a.address - b.address;
130 if (a.type === "AAAA") {
131 for (let i = 0; i < 8; i++)
132 if (a.groups[i] !== b.groups[i]) return a.groups[i] - b.groups[i];
133 return 0;
134 }
135 if (["NS", "CNAME", "PTR"].includes(a.type))
136 return nameOrder(a.data[0], b.data[0]);
137 if (a.type === "MX")
138 return a.preference - b.preference || nameOrder(a.data[1], b.data[1]);
139 return compareText(a.data.join(" "), b.data.join(" "));
140}
141function recordOrder(a, b) {
142 return (
143 nameOrder(a.owner, b.owner) ||
144 types.indexOf(a.type) - types.indexOf(b.type) ||
145 dataOrder(a, b)
146 );
147}
148function recordText(record) {
149 return `${record.owner} ${record.ttl} ${record.type} ${record.data.join(" ")}`;
150}

255 if (type === "SOA" && owner !== zone.origin)

257 if (type === "SOA" && zone.records.some((record) => record.type === "SOA"))
258 throw new Error("duplicate soa");

260 zone.records.some(

297 authoritative(server, name, type, cut) {

299 this.down.has(server) ||
300 !this.zones.some((zone) => zone.server === server)

309 const soa = zone.records.find((record) => record.type === "SOA");
310 const delegation = ancestors(name)
311 .reverse()
312 .find(
313 (ancestor) =>
314 ancestor !== zone.origin &&

316 zone.records.some(
317 (record) => record.owner === ancestor && record.type === "NS",
318 ),
319 );
320 if (delegation !== undefined) {
321 const authority = zone.records
322 .filter((record) => record.owner === delegation && record.type === "NS")
323 .sort(dataOrder);

326 .filter((glue) => glue.owner === record.data[0] && glue.type === "A")
327 .sort(dataOrder),

333 }

336 let records = zone.records.filter((record) => record.owner === name);

339 const ancestor = ancestors(name).slice(1).find(exists);

351 return reply("CNAME", [{ ...cname, owner: name }]);

421 lookup(name, type, depth, stack) {
422 const hit = (response, text) => {
423 this.hits++;
424 this.trace(depth, `CACHE HIT ${name} ${type} -> ${text}`);
425 return response;
426 };
427 const exact = this.cache.get(cacheKey(name, type));
428 if (exact !== undefined)
429 return hit(
430 exact.kind === "POS"
431 ? reply(
432 "ANSWER",
433 exact.records.map((record) => ({
434 ...record,
435 ttl: exact.expiry - this.clock,
436 })),
437 )
438 : reply(exact.kind),
439 exact.kind === "POS" ? "ANSWER" : exact.kind,
440 );
441 if (
442 [...this.cache.values()].some(
443 (entry) => entry.kind === "NXDOMAIN" && below(name, entry.name),
444 )
445 )
446 return hit(reply("NXDOMAIN"), "NXDOMAIN");
447 const cname = this.cache.get(cacheKey(name, "CNAME"));
448 if (type !== "CNAME" && cname?.kind === "POS")
449 return hit(
450 reply(
451 "CNAME",
452 cname.records.map((record) => ({
453 ...record,
454 ttl: cname.expiry - this.clock,
455 })),
456 ),
457 `CNAME ${cname.records[0].data[0]}`,
458 );
459 const ns = ancestors(name)
460 .map((ancestor) => this.cache.get(cacheKey(ancestor, "NS")))
461 .find((entry) => entry?.kind === "POS");
462 let cut = ns?.name ?? ".";
463 let nameservers =
464 ns === undefined ? null : ns.records.map((record) => record.data[0]);

466 this.hits++;
467 this.trace(depth, `CACHE HIT ${name} ${type} -> REFERRAL ${cut}`);
468 }
469 while (true) {
470 const response = this.queryServers(
471 name,
472 type,
473 cut,
474 nameservers,
475 depth,
476 stack,

492 return { ...response, answer: this.positive(response.answer) };
493 else {

497 name,
498 type,
499 kind: response.kind,
500 records: [],
501 expiry: this.clock + seconds,
502 });

507 resolve(name, type, depth = 0, stack = []) {
508 stack.push(cacheKey(name, type));
509 const chain = [];
510 const seen = new Set([name]);
511 let current = name;
512 let status = "";
513 let answer = [];
514 while (true) {
515 const result = this.lookup(current, type, depth, stack);
516 if (result.kind === "CNAME") {
517 const record = result.answer[0];
518 chain.push(record);
519 const target = record.data[0];
520 if (seen.has(target)) {
521 status = "SERVFAIL cname loop";
522 break;
523 }
524 if (chain.length === 9) {
525 status = "SERVFAIL cname chain too long";
526 break;
527 }
528 seen.add(target);
529 current = target;
530 } else {
531 status =
532 result.kind === "ANSWER"
533 ? "NOERROR"
534 : result.kind === "SERVFAIL"
535 ? "SERVFAIL " + result.reason
536 : result.kind;
537 if (!status.startsWith("SERVFAIL"))
538 answer = [
539 ...chain,
540 ...(result.kind === "ANSWER" ? result.answer : []),
541 ];
542 break;
543 }
544 }
545 stack.pop();
546 this.resolutions[status.split(" ")[0]]++;
547 return { status, answer };
548 }

654 const address = ipv4(words[1]);
655 if (command === "DOWN" && this.down.has(address))
656 throw new Error(`server ${words[1]} already down`);
657 if (command === "UP" && !this.down.has(address))
658 throw new Error(`server ${words[1]} not down`);
659 if (command === "DOWN") this.down.add(address);
660 else this.down.delete(address);

662 } else if (command === "SET") {

665 if (
666 !integerWord.test(words[2]) ||
667 Number(words[2]) < 1 ||
668 Number(words[2]) > 604800
669 )
670 throw new Error("bad value " + words[2]);
671 const value = Number(words[2]);
672 const minimum = words[1] === "MINTTL" ? value : this.settings.MINTTL;
673 const maximum = words[1] === "MAXTTL" ? value : this.settings.MAXTTL;
674 if (minimum > maximum) throw new Error("minttl above maxttl");
675 this.settings[words[1]] = value;
676 this.output.push(`set ${words[1]}=${value}`);
677 } else if (command === "TIME") {
678 if (
679 !integerWord.test(words[1]) ||
680 Number(words[1]) < 1 ||
681 Number(words[1]) > 1000000
682 )
683 throw new Error("bad time " + words[1]);
684 this.clock += Number(words[1]);
685 let expired = 0;
686 for (const [key, entry] of this.cache)
687 if (entry.expiry <= this.clock) {
688 this.cache.delete(key);
689 expired++;
690 }
691 this.output.push(`time now=${this.clock} expired=${expired}`);
692 } else if (command === "CACHE" || command === "FLUSH") {
693 const name = words.length === 1 ? "." : parseName(words[1]);
694 const entries = [...this.cache.values()]
695 .filter((entry) => below(entry.name, name))
696 .sort(
697 (a, b) =>
698 nameOrder(a.name, b.name) ||
699 types.indexOf(a.type) - types.indexOf(b.type),
700 );
701 if (command === "FLUSH") {
702 for (const entry of entries)
703 this.cache.delete(cacheKey(entry.name, entry.type));
704 this.output.push(`flush removed=${entries.length}`);
705 } else {
706 this.output.push(`cache entries=${entries.length}`);
707 for (const entry of entries) {
708 if (entry.kind === "POS")
709 for (const record of entry.records)
710 this.output.push(
711 " " +
712 recordText({ ...record, ttl: entry.expiry - this.clock }),
713 );
714 else
715 this.output.push(
716 ` ${entry.name} ${entry.expiry - this.clock} ${entry.type} ${entry.kind}`,
717 );
718 }
719 }
720 } else {
721 const queryCount = [...this.queries.values()].reduce(
722 (sum, counts) => sum + counts.reduce((a, b) => a + b, 0),
723 0,
724 );
725 this.output.push(`stats queries=${queryCount} hits=${this.hits}`);
726 for (const [server, counts] of [...this.queries].sort(
727 (a, b) => a[0] - b[0],
728 ))
729 this.output.push(
730 ` ${addressText(server)} sent=${counts.reduce((a, b) => a + b, 0)} ` +
731 kinds
732 .map((kind, index) => `${kind.toLowerCase()}=${counts[index]}`)
733 .join(" "),
734 );
735 this.output.push(
736 `resolutions noerror=${this.resolutions.NOERROR} nodata=${this.resolutions.NODATA} nxdomain=${this.resolutions.NXDOMAIN} servfail=${this.resolutions.SERVFAIL}`,
737 );
738 }
739 }
740 run(lines) {
741 let position = 0;
742 while (lines[position] !== "END ZONES") {
743 try {
744 this.zoneLine(lines[position].split(" "), position + 1);
745 } catch (error) {
746 this.output.push(`error line ${position + 1}: ${error.message}`);
747 }
748 position++;
749 }
750 position++;
751 this.finishZones();
752 while (lines[position] !== "END") {
753 try {
754 this.command(lines[position].split(" "));
755 } catch (error) {
756 this.output.push(`error line ${position + 1}: ${error.message}`);
757 }
758 position++;
759 }
760 }
761}
