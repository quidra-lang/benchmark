7func integer(_ s:String) -> Int? {
8 s.range(of:"^(?:0|[1-9][0-9]{0,8})$",options:.regularExpression) != nil ? Int(s) : nil

10func ip(_ s:String) -> Int? {
11 let fields=s.components(separatedBy:".")
12 guard fields.count==4 else {
13 return nil
14 }
15 var a=0
16 for p in fields {
17 guard p.count<=3, let n=integer(p), n<=255 else {
18 return nil
19 }
20 a=a*256+n
21 }
22 return a
23}
24func address(_ a:Int) -> String {
25 "\(a>>24).\((a>>16)&255).\((a>>8)&255).\(a&255)"
26}
27func ipv6(_ s:String) -> [Int]? {
28 func groups(_ s:String) -> [Int]? {
29 if s.isEmpty {
30 return []
31 }
32 let parts=s.components(separatedBy:":")
33 guard parts.allSatisfy({
34 $0.range(of:"^[0-9A-Fa-f]{1,4}$",options:.regularExpression) != nil
35 }) else {
36 return nil
37 }
38 return parts.map {
39 Int($0,radix:16)!
40 }
41 }
42 let parts=s.components(separatedBy:"::")
43 if parts.count==1 {
44 guard let a=groups(s), a.count==8 else {
45 return nil
46 }
47 return a
48 }
49 guard parts.count==2, let left=groups(parts[0]), let right=groups(parts[1]), left.count+right.count<=7 else {
50 return nil
51 }
52 return left+Array(repeating:0,count:8-left.count-right.count)+right
53}
54func address6(_ a:[Int]) -> String {
55 var best=0, size=0, at=0
56 while at<8 {
57 if a[at] != 0 {
58 at += 1
59 continue
60 }
61 let start=at
62 while at<8 && a[at]==0 {
63 at += 1
64 }
65 if at-start>=2 && at-start>size {
66 best=start
67 size=at-start
68 }
69 }
70 func text(_ a:[Int]) -> String {
71 a.map {
72 String($0,radix:16)
73 }.joined(separator:":")
74 }
75 return size==0 ? text(a) : text(Array(a[..<best]))+"::"+text(Array(a[(best+size)...]))
76}
77func name(_ word:String,_ wild:Bool,_ origin:String?=nil) throws -> String {
78 if word=="@", let origin {
79 return origin
80 }
81 if word=="." {
82 return "."
83 }
84 let parts=(word.hasSuffix(".") ? String(word.dropLast()) : word).components(separatedBy:".")
85 for (i,p) in parts.enumerated() {
86 if wild && i==0 && p=="*" {
87 continue
88 }
89 guard (1...63).contains(p.count), !p.hasPrefix("-"), !p.hasSuffix("-"), p.utf8.allSatisfy({
90 (65...90).contains($0) || (97...122).contains($0) || (48...57).contains($0) || $0==45
91 }) else {
92 throw Problem(reason:"bad name \(word)")
93 }
94 }
95 let suffix = !word.hasSuffix(".") && origin != nil && origin != "." ? origin! : ""
96 return parts.joined(separator:".").lowercased()+"."+suffix
97}
98func labels(_ s:String) -> [String] {
99 s=="." ? [] : String(s.dropLast()).components(separatedBy:".")
100}
101func nameOrder(_ a:String,_ b:String) -> Int {
102 let x=Array(labels(a).reversed()), y=Array(labels(b).reversed())
103 for i in 0..<min(x.count,y.count) {
104 if x[i] != y[i] {
105 return x[i]<y[i] ? -1 : 1
106 }
107 }
108 return x.count==y.count ? 0 : (x.count<y.count ? -1 : 1)
109}
110func below(_ a:String,_ b:String) -> Bool {

130func ttl(_ s:String) throws -> Int {
131 var seconds=integer(s)
132 if seconds==nil {
133 let b=Array(s.utf8), units=Array("wdhms".utf8), factors=[604800,86400,3600,60,1]
134 var at=0, previous = -1, total=0, valid=true
135 while at<b.count {
136 let start=at
137 while at<b.count && (48...57).contains(b[at]) {
138 at += 1
139 }
140 guard at<b.count, at-start<=3, let n=integer(String(decoding:b[start..<at],as:UTF8.self)), let unit=units.firstIndex(of:b[at]>=65 && b[at]<=90 ? b[at]+32 : b[at]), unit>previous else {
141 valid=false
142 break
143 }
144 total += n*factors[unit]
145 previous=unit
146 at += 1
147 }
148 if valid && previous>=0 {
149 seconds=total
150 }
151 }
152 guard let seconds, (1...604800).contains(seconds) else {
153 throw Problem(reason:"bad ttl \(s)")
154 }
155 return seconds

157enum Data: Equatable {
158 case a(Int), aaaa([Int]), target(String), mx(Int,String), txt(String), soa(String,String,[Int])
159 var text:String {
160 switch self {
161 case .a(let a): return address(a)
162 case .aaaa(let a): return address6(a)
163 case .target(let n): return n
164 case .mx(let p,let n): return "\(p) \(n)"
165 case .txt(let s): return s
166 case .soa(let a,let b,let n): return a+" "+b+" "+n.map(String.init).joined(separator:" ")
167 }
168 }
169}
170func dataOrder(_ a:Data,_ b:Data) -> Int {
171 switch (a,b) {
172 case (.a(let a),.a(let b)): return a==b ? 0 : (a<b ? -1 : 1)
173 case (.aaaa(let a),.aaaa(let b)): return a==b ? 0 : (a.lexicographicallyPrecedes(b) ? -1 : 1)
174 case (.target(let a),.target(let b)): return nameOrder(a,b)
175 case (.mx(let a,let x),.mx(let b,let y)): return a==b ? nameOrder(x,y) : (a<b ? -1 : 1)
176 case (.txt(let a),.txt(let b)): return a==b ? 0 : (a<b ? -1 : 1)
177 default: return 0
178 }
179}

186func recordBefore(_ a:Record,_ b:Record) -> Bool {
187 let order=nameOrder(a.owner,b.owner)
188 if order != 0 {
189 return order<0
190 }
191 let x=types.firstIndex(of:a.type)!, y=types.firstIndex(of:b.type)!
192 return x==y ? dataOrder(a.data,b.data)<0 : x<y
193}

223 var roots:[Int]=[], zones:[Zone]=[], current:Int?=nil, down=Set<Int>(), cache:[CacheKey:Cache]=[:], settings=[1,86400,3600], clock=0, queries:[Int:[Int]]=[:], hits=0, resolutions=[0,0,0,0], out:[String]=[]
224 func zoneLine(_ w:[String],_ line:Int) throws {
225 if w[0]=="ROOT" {
226 var added:[Int]=[]
227 for word in w.dropFirst() {
228 let a=try checkedIP(word)
229 if roots.contains(a) || added.contains(a) {
230 throw Problem(reason:"duplicate root \(word)")
231 }
232 added.append(a)
233 }
234 roots += added
235 return
236 }
237 if w[0]=="ZONE" {
238 current=nil
239 let origin=try name(w[1],false), server=try checkedIP(w[2])
240 if zones.contains(where: {
241 $0.origin==origin && $0.server==server
242 }) {
243 throw Problem(reason:"duplicate zone \(origin) at \(w[2])")
244 }
245 current=zones.count
246 zones.append(Zone(origin,server,line))
247 return
248 }
249 guard let i=current else {
250 throw Problem(reason:"no current zone")
251 }
252 let z=zones[i], owner=try name(w[0],true,z.origin)
253 if !below(owner,z.origin) {
254 throw Problem(reason:"out of zone \(owner)")
255 }
256 let seconds=try ttl(w[1]), type=w[2]
257 try checkedType(type)
258 let fields=Array(w.dropFirst(3)), count=type=="MX" ? 2 : (type=="SOA" ? 7 : (type=="TXT" ? fields.count : 1))
259 guard fields.count==count else {
260 throw Problem(reason:"wrong field count")
261 }
262 let data:Data
263 switch type {
264 case "A": data = .a(try checkedIP(fields[0]))
265 case "AAAA": guard let a=ipv6(fields[0]) else {
266 throw Problem(reason:"bad address \(fields[0])")
267 }
268 data = .aaaa(a)
269 case "NS","CNAME","PTR": data = .target(try name(fields[0],false,z.origin))
270 case "MX": guard let p=integer(fields[0]), p<=65535 else {
271 throw Problem(reason:"bad preference \(fields[0])")
272 }
273 data = .mx(p,try name(fields[1],false,z.origin))
274 case "TXT":
275 let text=fields.joined(separator:" ")
276 guard text.count>=2, text.hasPrefix("\""), text.hasSuffix("\""), !text.dropFirst().dropLast().contains("\"") else {
277 throw Problem(reason:"bad text \(text)")
278 }
279 data = .txt(text)
280 default:
281 let a=try name(fields[0],false,z.origin), b=try name(fields[1],false,z.origin)
282 var nums:[Int]=[]
283 for j in 2...5 {
284 guard let n=integer(fields[j]) else {
285 throw Problem(reason:"bad number \(fields[j])")
286 }
287 nums.append(n)
288 }
289 guard let minimum=integer(fields[6]), (1...86400).contains(minimum) else {
290 throw Problem(reason:"bad minimum \(fields[6])")
291 }
292 nums.append(minimum)
293 data = .soa(a,b,nums)
294 }
295 if type=="NS" && owner.hasPrefix("*.") {
296 throw Problem(reason:"ns at wildcard")
297 }
298 if type=="SOA" && owner != z.origin {
299 throw Problem(reason:"soa not at origin")
300 }
301 if type=="SOA" && z.records.contains(where: {
302 $0.type=="SOA"
303 }) {
304 throw Problem(reason:"duplicate soa")
305 }
306 if z.records.contains(where: {
307 $0.owner==owner && $0.type==type && $0.data==data
308 }) {
309 throw Problem(reason:"duplicate record")
310 }
311 if z.records.contains(where: {
312 $0.owner==owner && (type=="CNAME" || $0.type=="CNAME")
313 }) {
314 throw Problem(reason:"cname conflict \(owner)")
315 }
316 z.records.append(Record(owner:owner,ttl:seconds,type:type,data:data))
317 }
318 func finishZones() {
319 var kept:[Zone]=[]
320 for z in zones {
321 if z.records.contains(where: {
322 $0.type=="SOA"
323 }) {
324 out.append("zone \(z.origin) at \(address(z.server)) records=\(z.records.count)")
325 kept.append(z)
326 }
327 else {
328 out.append("error line \(z.line): zone \(z.origin) at \(address(z.server)) has no soa")
329 }
330 }
331 zones=kept
332 out.append("loaded roots=\(roots.count) zones=\(zones.count) records=\(zones.reduce(0) { $0+$1.records.count })")
333 }

341 func authoritative(_ server:Int,_ n:String,_ t:String,_ cut:String) -> Response {
342 if down.contains(server) || !zones.contains(where: {
343 $0.server==server
344 }) {
345 return Response(kind:6)
346 }
347 guard let z=zones.filter({
348 $0.server==server && below(n,$0.origin)
349 }).max(by: {
350 labels($0.origin).count<labels($1.origin).count
351 }), labels(z.origin).count>=labels(cut).count else {
352 return Response(kind:5)
353 }
354 for d in ancestors(n).reversed() where d != z.origin && below(d,z.origin) {
355 let ns=records(z,d,"NS")
356 if !ns.isEmpty {
357 return Response(kind:2,authority:ns,additional:ns.flatMap {
358 records(z,$0.data.text,"A")
359 },cut:d)
360 }
361 }
362 func exists(_ x:String) -> Bool {
363 z.records.contains {
364 below($0.owner,x)
365 }
366 }
367 let soa=records(z,z.origin,"SOA")
368 var owner=n
369 if !z.records.contains(where: {
370 $0.owner==n
371 }) {
372 if exists(n) {
373 return Response(kind:3,authority:soa)
374 }
375 let e=ancestors(n).first {
376 $0 != n && exists($0)
377 }!
378 owner=e=="." ? "*." : "*."+e
379 if !z.records.contains(where: {
380 $0.owner==owner
381 }) {
382 return Response(kind:4,authority:soa)
383 }
384 }
385 var answer=records(z,owner,t), kind=0
386 if answer.isEmpty && t != "CNAME" {
387 answer=records(z,owner,"CNAME")
388 kind=1
389 }
390 if answer.isEmpty {
391 return Response(kind:3,authority:soa)
392 }
393 return Response(kind:kind,answer:answer.map {
394 var r=$0
395 r.owner=n
396 return r
397 })
398 }
399 func ask(_ a:Int,_ n:String,_ t:String,_ cut:String) -> Response {
400 let r=authoritative(a,n,t,cut)
401 queries[a,default:Array(repeating:0,count:7)][r.kind] += 1
402 return r
403 }
404 func trace(_ depth:Int,_ text:String) {
405 out.append(String(repeating:" ",count:2*(depth+1))+text)

407 @discardableResult func positive(_ records:[Record]) -> Int {
408 let seconds=min(max(records.map(\.ttl).min()!,settings[0]),settings[1])
409 cache[CacheKey(records[0].owner,records[0].type)]=Cache(kind:0,records:records,expiry:clock+seconds)
410 return seconds
411 }
412 func cachedRecords(_ key:CacheKey) -> [Record]? {
413 guard let e=cache[key], e.kind==0 else {
414 return nil
415 }
416 return e.records.map {
417 var r=$0
418 r.ttl=e.expiry-clock
419 return r
420 }

422 func queryServers(_ n:String,_ t:String,_ cut:String,_ servers:[String]?,_ depth:Int,_ stack:inout [CacheKey]) -> Response? {
423 if let servers {
424 for host in servers.sorted(by: {
425 nameOrder($0,$1)<0
426 }) {
427 let records:[Record]
428 if let cached=cachedRecords(CacheKey(host,"A")) {
429 records=cached
430 }
431 else if stack.contains(CacheKey(host,"A")) {
432 trace(depth,"GLUELESS \(host) -> LOOP")
433 records=[]
434 }
435 else {
436 trace(depth,"GLUELESS \(host)")
437 let r=resolve(host,"A",depth+1,&stack)
438 records=r.status=="NOERROR" ? r.answer : []
439 }
440 let addresses=Set(records.compactMap {
441 if case .a(let a)=$0.data {
442 return a
443 }
444 return nil
445 }).sorted()
446 for a in addresses {
447 let r=ask(a,n,t,cut)
448 trace(depth,"ASK \(address(a)) \(n) \(t) -> \(r.text)")
449 if r.kind<5 {
450 return r
451 }
452 }
453 }
454 } else {
455 for a in roots {
456 let r=ask(a,n,t,cut)
457 trace(depth,"ASK \(address(a)) \(n) \(t) -> \(r.text)")
458 if r.kind<5 {
459 return r
460 }
461 }
462 }
463 return nil
464 }
465 func lookup(_ n:String,_ t:String,_ depth:Int,_ stack:inout [CacheKey]) throws -> Response {
466 if let e=cache[CacheKey(n,t)] {
467 hits += 1
468 trace(depth,"CACHE HIT \(n) \(t) -> \(e.kind==0 ? "ANSWER" : kinds[e.kind])")
469 return Response(kind:e.kind,answer:e.kind==0 ? cachedRecords(CacheKey(n,t))! : [])
470 }
471 if cache.contains(where: {
472 $0.value.kind==4 && below(n,$0.key.name)
473 }) {
474 hits += 1
475 trace(depth,"CACHE HIT \(n) \(t) -> NXDOMAIN")
476 return Response(kind:4)
477 }
478 if t != "CNAME", let records=cachedRecords(CacheKey(n,"CNAME")) {
479 let r=Response(kind:1,answer:records)
480 hits += 1
481 trace(depth,"CACHE HIT \(n) \(t) -> \(r.text)")
482 return r
483 }
484 let referral=cache.filter {
485 $0.key.type=="NS" && $0.value.kind==0 && below(n,$0.key.name)
486 }.max {
487 labels($0.key.name).count<labels($1.key.name).count
488 }
489 var cut=referral?.key.name ?? ".", servers=referral?.value.records.map {
490 $0.data.text
491 }
492 if referral != nil {
493 hits += 1
494 trace(depth,"CACHE HIT \(n) \(t) -> REFERRAL \(cut)")
495 }
496 while true {
497 guard var r=queryServers(n,t,cut,servers,depth,&stack) else {
498 throw Problem(reason:"SERVFAIL no server for \(cut)")
499 }
500 switch r.kind {
501 case 2:
502 positive(r.authority)
503 for records in Dictionary(grouping:r.additional,by: \.owner).values {
504 positive(records)
505 }
506 cut=r.cut
507 servers=r.authority.map {
508 $0.data.text
509 }
510 case 0,1:
511 let seconds=positive(r.answer)
512 r.answer=r.answer.map {
513 var r=$0
514 r.ttl=seconds
515 return r
516 }
517 return r
518 case 3,4:
519 let soa=r.authority[0]
520 guard case .soa(_,_,let nums)=soa.data else {
521 fatalError("SOA")
522 }
523 let seconds=min(soa.ttl,min(nums[4],settings[2]))
524 cache[CacheKey(n,t)]=Cache(kind:r.kind,records:[],expiry:clock+seconds)
525 return r
526 default: fatalError("unexpected response")
527 }
528 }
529 }
530 func resolve(_ n:String,_ t:String,_ depth:Int,_ stack:inout [CacheKey]) -> Resolution {
531 stack.append(CacheKey(n,t))
532 var chain:[Record]=[], seen:Set<String>=[n], current=n, result:Resolution
533 while true {
534 let r:Response
535 do {
536 r=try lookup(current,t,depth,&stack)
537 }
538 catch let p as Problem {
539 result=Resolution(status:p.reason,answer:[])
540 break
541 }
542 catch {
543 fatalError("unexpected error")
544 }
545 if r.kind==1 {
546 let target=r.answer[0].data.text
547 chain += r.answer
548 if seen.contains(target) {
549 result=Resolution(status:"SERVFAIL cname loop",answer:[])
550 break
551 }
552 if chain.count==9 {
553 result=Resolution(status:"SERVFAIL cname chain too long",answer:[])
554 break
555 }
556 seen.insert(target)
557 current=target
558 } else {
559 if r.kind==0 {
560 chain += r.answer
561 }
562 result=Resolution(status:r.kind==0 ? "NOERROR" : kinds[r.kind],answer:chain)
563 break
564 }
565 }
566 stack.removeLast()
567 let i=["NOERROR","NODATA","NXDOMAIN"].firstIndex(of:result.status) ?? 3
568 resolutions[i] += 1
569 return result
570 }
571 @discardableResult func outputResolution(_ n:String,_ t:String) -> Resolution {
572 var stack:[CacheKey]=[]
573 let r=resolve(n,t,0,&stack)
574 for record in r.answer {
575 out.append("answer "+record.text)
576 }
577 out.append("status "+r.status)
578 return r
579 }
580 func hostAddresses(_ n:String) -> [String] {
581 var addresses:[String]=[]
582 for t in ["A","AAAA"] {
583 var stack:[CacheKey]=[]
584 let r=resolve(n,t,0,&stack)
585 if r.status=="NOERROR" {
586 addresses += r.answer.filter {
587 $0.type==t
588 }.map {
589 $0.data.text
590 }
591 }
592 }
593 return addresses
594 }
595 func command(_ w:[String]) throws {
596 switch w[0] {
597 case "RESOLVE": let n=try name(w[1],true)
598 try checkedType(w[2])
599 out.append("resolve \(n) \(w[2])")
600 outputResolution(n,w[2])
601 case "REVERSE":
602 let a:String, n:String
603 if w[1].contains(":") {
604 guard let groups=ipv6(w[1]) else {
605 throw Problem(reason:"bad address \(w[1])")
606 }
607 a=address6(groups)
608 n=groups.map {
609 String(format:"%04x",$0)
610 }.joined().reversed().map(String.init).joined(separator:".")+".ip6.arpa."
611 }
612 else {
613 let value=try checkedIP(w[1])
614 a=address(value)
615 n=a.components(separatedBy:".").reversed().joined(separator:".")+".in-addr.arpa."
616 }
617 out.append("reverse \(a) \(n)")
618 outputResolution(n,"PTR")
619 case "MX":
620 let n=try name(w[1],true)
621 out.append("mx \(n)")
622 let r=outputResolution(n,"MX")
623 if r.status=="NOERROR" {
624 for record in r.answer {
625 if case .mx(let p,let host)=record.data {
626 if host=="." {
627 out.append("exchange \(p) . -> null")
628 } else {
629 let addresses=hostAddresses(host)
630 out.append("exchange \(p) \(host) -> \(addresses.isEmpty ? "no address" : addresses.joined(separator:" "))")
631 }
632 }
633 }
634 }
635 else if r.status=="NODATA" {
636 let host=r.answer.last {
637 $0.type=="CNAME"
638 }?.data.text ?? n, addresses=hostAddresses(host)
639 out.append("implicit \(host) -> \(addresses.isEmpty ? "no address" : addresses.joined(separator:" "))")
640 }
641 case "QUERY":
642 let a=try checkedIP(w[1]), n=try name(w[2],true)
643 try checkedType(w[3])
644 let r=ask(a,n,w[3],".")
645 out.append("query \(address(a)) \(n) \(w[3]) -> \(r.text)")
646 let sections:[(String,[Record])]
647 switch r.kind {
648 case 0,1: sections=[("answer",r.answer)]
649 case 2: sections=[("authority",r.authority),("additional",r.additional)]
650 case 3,4: sections=[("authority",r.authority)]
651 default: sections=[]
652 }
653 for (section,records) in sections {
654 for record in records {
655 out.append(section+" "+record.text)
656 }
657 }
658 case "DUMP":
659 let a=try checkedIP(w[1]), n=try name(w[2],false)
660 if down.contains(a) || !zones.contains(where: {
661 $0.server==a
662 }) {
663 out.append("dump \(address(a)) \(n) -> TIMEOUT")
664 }
665 else if let z=zones.first(where: {
666 $0.server==a && $0.origin==n
667 }) {
668 out.append("dump \(address(a)) \(n) records=\(z.records.count)")
669 for r in z.records.sorted(by:recordBefore) {
670 out.append(" "+r.text)
671 }
672 }
673 else {
674 out.append("dump \(address(a)) \(n) -> REFUSED")
675 }
676 case "UP","DOWN":
677 let a=try checkedIP(w[1])
678 if w[0]=="DOWN" {
679 if down.contains(a) {
680 throw Problem(reason:"server \(w[1]) already down")
681 }
682 down.insert(a)
683 }
684 else if down.remove(a)==nil {
685 throw Problem(reason:"server \(w[1]) not down")
686 }
687 out.append(w[0].lowercased()+" "+address(a))
688 case "SET":
689 guard let i=["MINTTL","MAXTTL","NEGTTL"].firstIndex(of:w[1]) else {
690 throw Problem(reason:"bad setting \(w[1])")
691 }
692 guard let n=integer(w[2]), (1...604800).contains(n) else {
693 throw Problem(reason:"bad value \(w[2])")
694 }
695 var changed=settings
696 changed[i]=n
697 if changed[0]>changed[1] {
698 throw Problem(reason:"minttl above maxttl")
699 }
700 settings=changed
701 out.append("set \(w[1])=\(n)")
702 case "TIME":
703 guard let n=integer(w[1]), (1...1000000).contains(n) else {
704 throw Problem(reason:"bad time \(w[1])")
705 }
706 clock += n
707 let old=cache.count
708 cache=cache.filter {
709 $0.value.expiry>clock
710 }
711 out.append("time now=\(clock) expired=\(old-cache.count)")
712 case "CACHE","FLUSH":
713 let parent: String?
714 if w.count == 2 {
715 parent = try name(w[1], false)
716 }
717 else {
718 parent = nil
719 }
720 let keys=cache.keys.filter {
721 parent==nil || below($0.name,parent!)
722 }.sorted {
723 a,b in let order=nameOrder(a.name,b.name)
724 return order==0 ? types.firstIndex(of:a.type)!<types.firstIndex(of:b.type)! : order<0
725 }
726 if w[0]=="FLUSH" {
727 out.append("flush removed=\(keys.count)")
728 for key in keys {
729 cache.removeValue(forKey:key)
730 }
731 }
732 else {
733 out.append("cache entries=\(keys.count)")
734 for key in keys {
735 let e=cache[key]!
736 if e.kind==0 {
737 for r in cachedRecords(key)! {
738 out.append(" "+r.text)
739 }
740 } else {
741 out.append(" \(key.name) \(e.expiry-clock) \(key.type) \(kinds[e.kind])")
742 }
743 }
744 }
745 default:
746 let total=queries.values.reduce(0) {
747 $0+$1.reduce(0,+)
748 }
749 out.append("stats queries=\(total) hits=\(hits)")
750 for a in queries.keys.sorted() {
751 let c=queries[a]!
752 out.append(" \(address(a)) sent=\(c.reduce(0,+)) answer=\(c[0]) cname=\(c[1]) referral=\(c[2]) nodata=\(c[3]) nxdomain=\(c[4]) lame=\(c[5]) timeout=\(c[6])")
753 }
754 let c=resolutions
755 out.append("resolutions noerror=\(c[0]) nodata=\(c[1]) nxdomain=\(c[2]) servfail=\(c[3])")
756 }
757 }
758}
759let resolver=Resolver()
760var zones=true, lineNumber=0
761while let line=readLine() {
762 lineNumber += 1
763 if line=="END" {
764 break
765 }
766 if line=="END ZONES" {
767 resolver.finishZones()
768 zones=false
769 continue
770 }
771 do {
772 let w=line.components(separatedBy:" ")
773 if zones {
774 try resolver.zoneLine(w,lineNumber)
775 } else {
776 try resolver.command(w)
777 }
778 }
779 catch let p as Problem {
780 resolver.out.append("error line \(lineNumber): \(p.reason)")
781 }
782 catch {
783 fatalError("unexpected error")
784 }
785}
