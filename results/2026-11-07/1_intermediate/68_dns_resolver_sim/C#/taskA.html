62 static string Name(string word, bool wildcard, string? origin)
63 {
64 if (origin is not null && word == "@") return origin;
65 if (word == ".") return ".";
66 string text = word.EndsWith('.') ? word[..^1] : word;
67 string[] labels = text.Split('.');
68 for (int i = 0; i < labels.Length; i++) Require(wildcard && i == 0 && labels[i] == "*" || Matches(labels[i], "[A-Za-z0-9](?:[A-Za-z0-9-]{0,61}[A-Za-z0-9])?"), "bad name " + word);
69 string result = text.ToLowerInvariant() + ".";
70 if (origin is not null && !word.EndsWith('.') && origin != ".") result += origin;
71 return result;
72 }

86 static int CompareNames(string left, string right)
87 {
88 string[] a = Labels(left), b = Labels(right);
89 for (int i = 1; i <= Math.Min(a.Length, b.Length); i++)
90 {
91 int order = StringComparer.Ordinal.Compare(a[^i], b[^i]);
92 if (order != 0) return order;
93 }
94 return a.Length.CompareTo(b.Length);
95 }

97 static uint Ipv4(string word)
98 {
99 string[] fields = word.Split('.');
100 Require(fields.Length == 4, "bad address " + word);
101 uint value = 0;
102 foreach (string field in fields)
103 {
104 Require(Matches(field, "0|[1-9][0-9]{0,2}"), "bad address " + word);
105 uint n = uint.Parse(field);
106 Require(n <= 255, "bad address " + word);
107 value = value * 256 + n;
108 }
109 return value;
110 }

112 static string Address(uint value) => $"{value >> 24}.{(value >> 16) & 255}.{(value >> 8) & 255}.{value & 255}";

114 static int[] Ipv6(string word)
115 {
116 int colon = word.IndexOf("::", StringComparison.Ordinal);
117 string left = colon < 0 ? word : word[..colon], right = colon < 0 ? "" : word[(colon + 2)..];
118 Require(colon < 0 || colon == word.LastIndexOf("::", StringComparison.Ordinal), "bad address " + word);
119 string[] a = left.Length == 0 ? [] : left.Split(':'), b = right.Length == 0 ? [] : right.Split(':');
120 Require(colon < 0 ? a.Length == 8 : a.Length + b.Length <= 7, "bad address " + word);
121 int[] groups = new int[8];
122 for (int i = 0; i < a.Length; i++)
123 {
124 Require(Matches(a[i], "[0-9a-fA-F]{1,4}"), "bad address " + word);
125 groups[i] = int.Parse(a[i], NumberStyles.HexNumber, CultureInfo.InvariantCulture);
126 }
127 for (int i = 0; i < b.Length; i++)
128 {
129 Require(Matches(b[i], "[0-9a-fA-F]{1,4}"), "bad address " + word);
130 groups[8 - b.Length + i] = int.Parse(b[i], NumberStyles.HexNumber, CultureInfo.InvariantCulture);
131 }
132 return groups;
133 }

135 static string Address6(int[] groups)
136 {
137 int best = -1, count = 1;
138 for (int i = 0; i < 8;)
139 {
140 if (groups[i] != 0)
141 {
142 i++;
143 continue;
144 }
145 int end = i;
146 while (end < 8 && groups[end] == 0) end++;
147 if (end - i > count)
148 {
149 best = i;
150 count = end - i;
151 }
152 i = end;
153 }
154 string Group(int index) => groups[index].ToString("x", CultureInfo.InvariantCulture);
155 if (best < 0) return string.Join(":", Enumerable.Range(0, 8).Select(Group));
156 return string.Join(":", Enumerable.Range(0, best).Select(Group)) + "::" + string.Join(":", Enumerable.Range(best + count, 8 - best - count).Select(Group));
157 }

159 static int? Integer(string word) => Matches(word, "0|[1-9][0-9]{0,8}") ? int.Parse(word) : null;

161 static int Ttl(string word)
162 {
163 int? plain = Integer(word);
164 long total = plain ?? 0;
165 if (plain is null)
166 {
167 int position = 0, previous = -1;
168 const string units = "wdhms";
169 int[] seconds = [604800, 86400, 3600, 60, 1];
170 while (position < word.Length)
171 {
172 int start = position;
173 while (position < word.Length && word[position] is >= '0' and <= '9') position++;
174 string number = word[start..position];
175 Require(number.Length is >= 1 and <= 3 && Integer(number) is not null && position < word.Length, "bad ttl " + word);
176 int unit = units.IndexOf(char.ToLowerInvariant(word[position++]));
177 Require(unit > previous, "bad ttl " + word);
178 previous = unit;
179 total += (long)int.Parse(number) * seconds[unit];
180 }
181 }
182 Require(total is >= 1 and <= 604800, "bad ttl " + word);
183 return (int)total;
184 }

186 static int CompareData(ResourceRecord left, ResourceRecord right)
187 {
188 switch (left.Type)
189 {
190 case 0:
191 return Ipv4(left.Data).CompareTo(Ipv4(right.Data));
192 case 7:
193 int[] a = Ipv6(left.Data), b = Ipv6(right.Data);
194 for (int i = 0; i < 8; i++)
195 {
196 int order = a[i].CompareTo(b[i]);
197 if (order != 0) return order;
198 }
199 return 0;
200 case 1:
201 case 2:
202 case 4:
203 return CompareNames(left.Data, right.Data);
204 case 5:
205 string[] x = left.Data.Split(' '), y = right.Data.Split(' ');
206 int comparison = int.Parse(x[0]).CompareTo(int.Parse(y[0]));
207 return comparison == 0 ? CompareNames(x[1], y[1]) : comparison;
208 case 6:
209 return StringComparer.Ordinal.Compare(left.Data, right.Data);
210 default:
211 return 0;
212 }
213 }

219 static void ZoneLine(string[] words, int line)
220 {
221 if (words[0] == "ROOT")
222 {
223 List<uint> additions = [];
224 foreach (string word in words.Skip(1))
225 {
226 uint value = Ipv4(word);
227 Require(!Roots.Contains(value) && !additions.Contains(value), "duplicate root " + word);
228 additions.Add(value);
229 }
230 Roots.AddRange(additions);
231 return;
232 }
233 if (words[0] == "ZONE")
234 {
235 current = null;
236 string origin = Name(words[1], false, null);
237 uint server = Ipv4(words[2]);
238 Require(!Zones.Any(zone => zone.Origin == origin && zone.Server == server), $"duplicate zone {origin} at {words[2]}");
239 current = new(origin, server, line);
240 Zones.Add(current);
241 return;
242 }
243 Require(current is not null, "no current zone");
244 Zone selected = current!;
245 string owner = Name(words[0], true, selected.Origin);
246 Require(Below(owner, selected.Origin), "out of zone " + owner);
247 int ttl = Ttl(words[1]), type = TypeIndex(words[2]);
248 string[] data = words[3..];
249 Require(type == 6 || data.Length == (type == 5 ? 2 : type == 3 ? 7 : 1), "wrong field count");
250 string text;
251 if (type == 0) text = Address(Ipv4(data[0]));
252 else if (type == 7) text = Address6(Ipv6(data[0]));
253 else if (type is 1 or 2 or 4) text = Name(data[0], false, selected.Origin);
254 else if (type == 5)
255 {
256 int? preference = Integer(data[0]);
257 Require(preference is not null && preference <= 65535, "bad preference " + data[0]);
258 text = preference + " " + Name(data[1], false, selected.Origin);
259 }
260 else if (type == 6)
261 {
262 text = string.Join(" ", data);
263 Require(text.Length >= 2 && text.StartsWith('"') && text.EndsWith('"') && !text[1..^1].Contains('"'), "bad text " + text);
264 }
265 else
266 {
267 List<string> fields = [Name(data[0], false, selected.Origin), Name(data[1], false, selected.Origin)];
268 for (int i = 2; i < 6; i++)
269 {
270 Require(Integer(data[i]) is not null, "bad number " + data[i]);
271 fields.Add(data[i]);
272 }
273 int? minimum = Integer(data[6]);
274 Require(minimum is >= 1 and <= 86400, "bad minimum " + data[6]);
275 fields.Add(data[6]);
276 text = string.Join(" ", fields);
277 }
278 Require(type != 1 || !owner.StartsWith("*.", StringComparison.Ordinal), "ns at wildcard");
279 Require(type != 3 || owner == selected.Origin, "soa not at origin");
280 Require(type != 3 || selected.Soa is null, "duplicate soa");
281 Require(!selected.Records.Any(record => record.Owner == owner && record.Type == type && record.Data == text), "duplicate record");
282 Require(!selected.Records.Any(record => record.Owner == owner && (type == 2 || record.Type == 2)), "cname conflict " + owner);
283 selected.Records.Add(new(owner, ttl, type, text));
284 }

302 static Response Authoritative(uint server, string name, int type, string cut)
303 {
304 if (Down.Contains(server) || !Zones.Any(candidate => candidate.Server == server)) return new("TIMEOUT");
305 Zone? zone = null;
306 foreach (Zone candidate in Zones) if (candidate.Server == server && Below(name, candidate.Origin) && (zone is null || Length(candidate.Origin) > Length(zone.Origin))) zone = candidate;
307 if (zone is null || Length(zone.Origin) < Length(cut)) return new("LAME");
308 string? delegation = null;
309 foreach (ResourceRecord record in zone.Records) if (record.Type == 1 && record.Owner != zone.Origin && Below(name, record.Owner) && (delegation is null || Length(record.Owner) < Length(delegation))) delegation = record.Owner;
310 if (delegation is not null)
311 {
312 Response response = new("REFERRAL") { Cut = delegation, Authority = Records(zone, delegation, 1) };
313 foreach (ResourceRecord ns in response.Authority) response.Additional.AddRange(Records(zone, ns.Data, 0));
314 return response;
315 }
316 List<ResourceRecord> chosen = zone.Records.Where(record => record.Owner == name).ToList();
317 if (chosen.Count == 0 && !Exists(zone, name))
318 {
319 string ancestor = Parent(name);
320 while (!Exists(zone, ancestor)) ancestor = Parent(ancestor);
321 string wildcard = ancestor == "." ? "*." : "*." + ancestor;
322 chosen = zone.Records.Where(record => record.Owner == wildcard).ToList();
323 }
324 Response result;
325 if (chosen.Count == 0) result = new(Exists(zone, name) ? "NODATA" : "NXDOMAIN");
326 else
327 {
328 List<ResourceRecord> answer = chosen.Where(record => record.Type == type).Select(record => record with { Owner = name }).ToList();
329 if (answer.Count > 0) return new("ANSWER") { Answer = Sorted(answer) };
330 if (type != 2)
331 {
332 ResourceRecord? alias = chosen.Find(record => record.Type == 2);
333 if (alias is not null) return new("CNAME")
334 {
335 Answer = [alias with { Owner = name }]
336 };
337 }
338 result = new("NODATA");
339 }
340 result.Authority.Add(zone.Soa!);
341 return result;
342 }

353 return response;

368 static Response? QueryServers(string name, int type, string cut, List<string>? servers, int depth, List<(string Name, int Type)> resolving)
369 {
370 if (servers is null)
371 {
372 foreach (uint server in Roots)
373 {
374 Response response = Ask(server, name, type, cut);
375 Trace(depth, $"ASK {Address(server)} {name} {Types[type]} -> {response.Description}");
376 if (response.Kind is not ("LAME" or "TIMEOUT")) return response;
377 }
378 }
379 else foreach (string host in servers.Distinct(StringComparer.Ordinal).OrderBy(entry => entry, Comparer<string>.Create(CompareNames)))
380 {
381 SortedSet<uint> addresses = [];
382 if (Cache.TryGetValue((host, 0), out CacheEntry? cached) && cached.Kind == "POS") foreach (ResourceRecord record in cached.Records) addresses.Add(Ipv4(record.Data));
383 else if (resolving.Contains((host, 0))) Trace(depth, "GLUELESS " + host + " -> LOOP");
384 else
385 {
386 Trace(depth, "GLUELESS " + host);
387 Result result = Resolve(host, 0, depth + 1, resolving);
388 if (result.Status == "NOERROR") foreach (ResourceRecord record in result.Answer) if (record.Type == 0) addresses.Add(Ipv4(record.Data));
389 }
390 foreach (uint server in addresses)
391 {
392 Response response = Ask(server, name, type, cut);
393 Trace(depth, $"ASK {Address(server)} {name} {Types[type]} -> {response.Description}");
394 if (response.Kind is not ("LAME" or "TIMEOUT")) return response;
395 }
396 }
397 return null;
398 }

400 static Response Lookup(string name, int type, int depth, List<(string Name, int Type)> resolving)
401 {
402 Response? cached = null;
403 if (Cache.TryGetValue((name, type), out CacheEntry? exact)) cached = new(exact.Kind == "POS" ? "ANSWER" : exact.Kind) { Answer = exact.Kind == "POS" ? CachedRecords(exact) : [] };
404 if (cached is null && Cache.Any(entry => entry.Value.Kind == "NXDOMAIN" && Below(name, entry.Key.Name))) cached = new("NXDOMAIN");
405 if (cached is null && type != 2 && Cache.TryGetValue((name, 2), out CacheEntry? alias) && alias.Kind == "POS") cached = new("CNAME") { Answer = CachedRecords(alias) };
406 if (cached is not null)
407 {
408 hits++;
409 Trace(depth, $"CACHE HIT {name} {Types[type]} -> {cached.Description}");
410 return cached;
411 }
412 string cut = ".";
413 List<string>? servers = null;
414 foreach (var entry in Cache) if (entry.Key.Type == 1 && entry.Value.Kind == "POS" && Below(name, entry.Key.Name) && (servers is null || Length(entry.Key.Name) > Length(cut)))
415 {
416 cut = entry.Key.Name;
417 servers = entry.Value.Records.Select(record => record.Data).ToList();
418 }
419 if (servers is not null)
420 {
421 hits++;
422 Trace(depth, $"CACHE HIT {name} {Types[type]} -> REFERRAL {cut}");
423 }
424 while (true)
425 {
426 Response? response = QueryServers(name, type, cut, servers, depth, resolving);
427 if (response is null) return new("SERVFAIL no server for " + cut);
428 if (response.Kind == "REFERRAL")
429 {
430 Store(response.Authority);
431 foreach (var group in response.Additional.GroupBy(record => record.Owner, StringComparer.Ordinal)) Store(group.ToList());
432 cut = response.Cut;
433 servers = response.Authority.Select(record => record.Data).ToList();
434 continue;
435 }
436 if (response.Kind is "ANSWER" or "CNAME")
437 {
438 int ttl = Store(response.Answer);
439 response.Answer = response.Answer.Select(record => record with { Ttl = ttl }).ToList();
440 }
441 else
442 {
443 ResourceRecord soa = response.Authority[0];
444 int minimum = int.Parse(soa.Data.Split(' ')[6]);
445 Cache[(name, type)] = new(response.Kind, [], now + Math.Min(negTtl, Math.Min(soa.Ttl, minimum)));
446 }
447 return response;
448 }
449 }

451 static Result Resolve(string name, int type, int depth, List<(string Name, int Type)> resolving)
452 {
453 resolving.Add((name, type));
454 List<ResourceRecord> aliases = [];
455 HashSet<string> seen = new(StringComparer.Ordinal) { name };
456 string currentName = name, status;
457 List<ResourceRecord> answer;
458 while (true)
459 {
460 Response response = Lookup(currentName, type, depth, resolving);
461 if (response.Kind == "CNAME")
462 {
463 ResourceRecord alias = response.Answer[0];
464 aliases.Add(alias);
465 string target = alias.Data;
466 if (seen.Contains(target))
467 {
468 status = "SERVFAIL cname loop";
469 answer = [];
470 break;
471 }
472 if (aliases.Count == 9)
473 {
474 status = "SERVFAIL cname chain too long";
475 answer = [];
476 break;
477 }
478 seen.Add(target);
479 currentName = target;
480 continue;
481 }
482 if (response.Kind == "ANSWER")
483 {
484 status = "NOERROR";
485 answer = [.. aliases, .. response.Answer];
486 }
487 else if (response.Kind is "NODATA" or "NXDOMAIN")
488 {
489 status = response.Kind;
490 answer = aliases;
491 }
492 else
493 {
494 status = response.Kind;
495 answer = [];
496 }
497 break;
498 }
499 resolving.RemoveAt(resolving.Count - 1);
500 Resolutions[status == "NOERROR" ? 0 : status == "NODATA" ? 1 : status == "NXDOMAIN" ? 2 : 3]++;
501 return new(status, answer);
502 }

523 static void Command(string[] words)
524 {
525 switch (words[0])
526 {
527 case "RESOLVE":
528 {
529 string name = Name(words[1], true, null);
530 int type = TypeIndex(words[2]);
531 Console.WriteLine($"resolve {name} {Types[type]}");
532 Answer(Resolve(name, type));
533 break;
534 }
535 case "REVERSE":
536 {
537 string output, reverse;
538 if (words[1].Contains(':'))
539 {
540 int[] groups = Ipv6(words[1]);
541 output = Address6(groups);
542 string digits = string.Concat(groups.Select(group => group.ToString("x4", CultureInfo.InvariantCulture)));
543 reverse = string.Join(".", digits.Reverse()) + ".ip6.arpa.";
544 }
545 else
546 {
547 output = Address(Ipv4(words[1]));
548 string[] fields = output.Split('.');
549 reverse = $"{fields[3]}.{fields[2]}.{fields[1]}.{fields[0]}.in-addr.arpa.";
550 }
551 Console.WriteLine($"reverse {output} {reverse}");
552 Answer(Resolve(reverse, 4));
553 break;
554 }
555 case "MX":
556 {
557 string name = Name(words[1], true, null);
558 Console.WriteLine("mx " + name);
559 Result result = Resolve(name, 5);
560 Answer(result);
561 if (result.Status == "NOERROR") foreach (ResourceRecord record in result.Answer.Where(entry => entry.Type == 5))
562 {
563 string[] fields = record.Data.Split(' ');
564 if (fields[1] == ".") Console.WriteLine($"exchange {fields[0]} . -> null");
565 else Addresses(fields[1], "exchange " + fields[0] + " ");
566 }
567 if (result.Status == "NODATA")
568 {
569 string host = name;
570 foreach (ResourceRecord record in result.Answer) if (record.Type == 2) host = record.Data;
571 Addresses(host, "implicit ");
572 }
573 break;
574 }
575 case "QUERY":
576 {
577 uint server = Ipv4(words[1]);
578 string name = Name(words[2], true, null);
579 int type = TypeIndex(words[3]);
580 Response response = Ask(server, name, type, ".");
581 Console.WriteLine($"query {Address(server)} {name} {Types[type]} -> {response.Description}");
582 foreach (ResourceRecord record in response.Answer) Console.WriteLine("answer " + record);
583 foreach (ResourceRecord record in response.Authority) Console.WriteLine("authority " + record);
584 foreach (ResourceRecord record in response.Additional) Console.WriteLine("additional " + record);
585 break;
586 }
587 case "DUMP":
588 {
589 uint server = Ipv4(words[1]);
590 string name = Name(words[2], false, null);
591 if (Down.Contains(server) || !Zones.Any(candidate => candidate.Server == server))
592 {
593 Console.WriteLine($"dump {Address(server)} {name} -> TIMEOUT");
594 break;
595 }
596 Zone? zone = Zones.Find(candidate => candidate.Server == server && candidate.Origin == name);
597 if (zone is null) Console.WriteLine($"dump {Address(server)} {name} -> REFUSED");
598 else
599 {
600 Console.WriteLine($"dump {Address(server)} {name} records={zone.Records.Count}");
601 foreach (ResourceRecord record in zone.Records.OrderBy(entry => entry.Owner, Comparer<string>.Create(CompareNames)).ThenBy(entry => entry.Type).ThenBy(entry => entry, Comparer<ResourceRecord>.Create(CompareData))) Console.WriteLine(" " + record);
602 }
603 break;
604 }
605 case "DOWN":
606 case "UP":
607 {
608 uint server = Ipv4(words[1]);
609 bool disable = words[0] == "DOWN";
610 Require(disable ? !Down.Contains(server) : Down.Contains(server), "server " + words[1] + (disable ? " already down" : " not down"));
611 if (disable) Down.Add(server);
612 else Down.Remove(server);
613 Console.WriteLine((disable ? "down " : "up ") + Address(server));
614 break;
615 }
616 case "SET":
617 {
618 Require(words[1] is "MINTTL" or "MAXTTL" or "NEGTTL", "bad setting " + words[1]);
619 int? value = Integer(words[2]);
620 Require(value is >= 1 and <= 604800, "bad value " + words[2]);
621 Require((words[1] == "MINTTL" ? value!.Value : minTtl) <= (words[1] == "MAXTTL" ? value!.Value : maxTtl), "minttl above maxttl");
622 if (words[1] == "MINTTL") minTtl = value!.Value;
623 else if (words[1] == "MAXTTL") maxTtl = value!.Value;
624 else negTtl = value!.Value;
625 Console.WriteLine($"set {words[1]}={value}");
626 break;
627 }
628 case "TIME":
629 {
630 int? value = Integer(words[1]);
631 Require(value is >= 1 and <= 1000000, "bad time " + words[1]);
632 now += value!.Value;
633 int before = Cache.Count;
634 foreach (var key in Cache.Where(entry => entry.Value.Expiry <= now).Select(entry => entry.Key).ToList()) Cache.Remove(key);
635 Console.WriteLine($"time now={now} expired={before - Cache.Count}");
636 break;
637 }
638 case "CACHE":
639 case "FLUSH":
640 {
641 string? name = words.Length == 2 ? Name(words[1], false, null) : null;
642 var entries = Selected(name);
643 if (words[0] == "FLUSH")
644 {
645 foreach (var entry in entries) Cache.Remove(entry.Key);
646 Console.WriteLine($"flush removed={entries.Count}");
647 }
648 else
649 {
650 Console.WriteLine($"cache entries={entries.Count}");
651 foreach (var entry in entries) if (entry.Value.Kind == "POS") foreach (ResourceRecord record in CachedRecords(entry.Value)) Console.WriteLine(" " + record);
652 else Console.WriteLine($" {entry.Key.Name} {entry.Value.Expiry - now} {Types[entry.Key.Type]} {entry.Value.Kind}");
653 }
654 break;
655 }
656 case "STATS": Console.WriteLine($"stats queries={Queries.Values.Sum(counts => counts.Sum())} hits={hits}");
657 foreach (var entry in Queries) Console.WriteLine($" {Address(entry.Key)} sent={entry.Value.Sum()} " + string.Join(" ", Kinds.Select((kind, i) => kind.ToLowerInvariant() + "=" + entry.Value[i])));
658 Console.WriteLine($"resolutions noerror={Resolutions[0]} nodata={Resolutions[1]} nxdomain={Resolutions[2]} servfail={Resolutions[3]}");
659 break;
660 }
661 }
