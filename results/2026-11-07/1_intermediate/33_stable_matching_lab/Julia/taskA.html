47rank_in(preference::Vector{Int}, partner::Int) = something(findfirst(==(partner), preference), length(preference) + 1)
48list_text(values) = isempty(values) ? "-" : join(values, " ")
49receivers(market::Market) = (market.proposers + 1):length(market.names)
50partner_name(market::Market, partner::Int) = partner == 0 ? "*" : market.names[partner]
51pair_text(market::Market, matching::Vector{Int}) =
52 join((market.names[p] * "-" * partner_name(market, matching[p]) for p in 1:market.proposers), " ")
53rotation_text(market::Market, matching::Vector{Int}, rotation::Vector{Int}) =
54 join((market.names[p] * "-" * market.names[matching[p]] for p in rotation), " ")
55allocation_text(market::Market, allocation::Vector{Int}) =
56 join((market.names[a] * "-" * (allocation[a] == 0 ? "*" : market.houses[allocation[a]]) for a in eachindex(market.names)), " ")

61function agent_index(market::Market, name::AbstractString, side::Side = ANY_AGENT)
62 index = findfirst(==(name), market.names)
63 isnothing(index) && fail("UNKNOWN AGENT $name")
64 side == PROPOSER && index > market.proposers && fail("NOT PROPOSER $name")
65 side == RECEIVER && index <= market.proposers && fail("NOT RECEIVER $name")
66 index
67end

69function deferred(market::Market; reverse::Bool = false, verbose::Bool = false)
70 n = length(market.names)
71 next_position = ones(Int, n)
72 counts = zeros(Int, n)
73 holding = [Int[] for _ in 1:n]
74 queued = falses(n)
75 queue = Int[]
76 able(agent) = counts[agent] < market.capacities[agent] && next_position[agent] <= length(market.preferences[agent])
77 function enqueue!(agent)
78 if able(agent) && !queued[agent]
79 push!(queue, agent)
80 queued[agent] = true
81 end
82 end
83 for agent in (reverse ? receivers(market) : (1:market.proposers))
84 enqueue!(agent)
85 end
86 history = String[]
87 head = 1
88 while head <= length(queue)
89 proposer = queue[head]
90 head += 1
91 queued[proposer] = false
92 receiver = market.preferences[proposer][next_position[proposer]]
93 next_position[proposer] += 1
94 action = "REJECT"
95 preference = market.preferences[receiver]
96 held = holding[receiver]
97 if proposer in preference
98 if length(held) < market.capacities[receiver]
99 push!(held, proposer)
100 counts[proposer] += 1
101 action = "HOLD"
102 else
103 worst = argmax([rank_in(preference, partner) for partner in held])
104 rejected = held[worst]
105 if rank_in(preference, proposer) < rank_in(preference, rejected)
106 held[worst] = proposer
107 counts[rejected] -= 1
108 counts[proposer] += 1
109 action = "HOLD REJECT $(market.names[rejected])"
110 enqueue!(rejected)
111 end
112 end
113 end
114 verbose && push!(history, "PROPOSE $(market.names[proposer]) $(market.names[receiver]) $action")
115 enqueue!(proposer)
116 end
117 matching = zeros(Int, market.proposers)
118 for receiver in (reverse ? (1:market.proposers) : receivers(market))
119 for proposer in holding[receiver]
120 if reverse
121 matching[receiver] = proposer
122 else
123 matching[proposer] = receiver
124 end
125 end
126 end
127 Run(matching, head - 1, history)
128end

130function boston(market::Market)
131 matching = zeros(Int, market.proposers)
132 remaining = copy(market.capacities)
133 rounds = maximum(length, market.preferences[1:market.proposers])
134 applications = 0
135 for round in 1:rounds
136 applied = zeros(Int, market.proposers)
137 for proposer in 1:market.proposers
138 if matching[proposer] == 0 && length(market.preferences[proposer]) >= round
139 applied[proposer] = market.preferences[proposer][round]
140 applications += 1
141 end
142 end
143 for receiver in receivers(market), proposer in market.preferences[receiver]
144 if remaining[receiver] > 0 && applied[proposer] == receiver
145 matching[proposer] = receiver
146 remaining[receiver] -= 1
147 end
148 end
149 end
150 Run(matching, applications, String[])
151end

153function exposed_rotations(market::Market, matching::Vector{Int})
154 owner = zeros(Int, length(market.names))
155 successor = zeros(Int, market.proposers)
156 for proposer in findall(>(0), matching)
157 owner[matching[proposer]] = proposer
158 end
159 for proposer in findall(>(0), matching)
160 preference = market.preferences[proposer]
161 for receiver in preference[(rank_in(preference, matching[proposer]) + 1):end]
162 ranking = market.preferences[receiver]
163 if proposer in ranking && (owner[receiver] == 0 || rank_in(ranking, proposer) < rank_in(ranking, owner[receiver]))
164 successor[proposer] = owner[receiver]
165 break
166 end
167 end
168 end
169 covered = falses(market.proposers)
170 rotations = Vector{Int}[]
171 for start in 1:market.proposers
172 covered[start] && continue
173 path = Int[]
174 point = start
175 while point > 0 && !(point in path)
176 push!(path, point)
177 point = successor[point]
178 end
179 if point == start
180 push!(rotations, path)
181 covered[path] .= true
182 end
183 end
184 rotations
185end

187function candidate(market::Market, matching::Vector{Int}, origin::Int = 0, via::Int = 0)
188 ranks = Int[]
189 for proposer in findall(>(0), matching)
190 receiver = matching[proposer]
191 push!(ranks, rank_in(market.preferences[proposer], receiver), rank_in(market.preferences[receiver], proposer))
192 end
193 Candidate(matching, origin, via, sum(ranks), maximum(ranks; init = 0))
194end

196function enumerate_matchings(market::Market)
197 all(==(1), market.capacities[receivers(market)]) || fail("NOT ONE TO ONE")
198 matchings = [candidate(market, deferred(market).matching)]
199 rotations = String[]
200 index = 1
201 while index <= length(matchings)
202 matching = matchings[index].matching
203 for rotation in exposed_rotations(market, matching)
204 notation = rotation_text(market, matching, rotation)
205 identifier = findfirst(==(notation), rotations)
206 if isnothing(identifier)
207 push!(rotations, notation)
208 identifier = length(rotations)
209 end
210 following = copy(matching)
211 for i in eachindex(rotation)
212 following[rotation[i]] = matching[rotation[mod1(i + 1, length(rotation))]]
213 end
214 any(old -> old.matching == following, matchings) && continue
215 length(matchings) == 50 && fail("TOO MANY MATCHINGS")
216 push!(matchings, candidate(market, following, index, identifier))
217 end
218 index += 1
219 end
220 Lattice(matchings, rotations)
221end

223function check_pairs(market::Market, words)
224 market.kind == HOUSING && fail("WRONG TYPE $(market.name)")
225 matching = zeros(Int, length(market.names))
226 counts = zeros(Int, length(market.names))
227 two = market.kind == TWO_SIDED
228 for word in words[3:end]
229 left, right = split(word, '-')
230 first = agent_index(market, left, two ? PROPOSER : ANY_AGENT)
231 second = agent_index(market, right, two ? RECEIVER : ANY_AGENT)
232 matching[first] > 0 && fail("DUPLICATE AGENT $left")
233 if two
234 counts[second] += 1
235 counts[second] > market.capacities[second] && fail("OVER CAPACITY $right")
236 else
237 (first == second || matching[second] > 0) && fail("DUPLICATE AGENT $right")
238 matching[second] = first
239 end
240 matching[first] = second
241 end
242 unacceptable = String[]
243 blocking = String[]
244 if two
245 for proposer in 1:market.proposers
246 receiver = matching[proposer]
247 if receiver > 0 && (!(receiver in market.preferences[proposer]) || !(proposer in market.preferences[receiver]))
248 push!(unacceptable, "UNACCEPTABLE $(market.names[proposer]) $(market.names[receiver])")
249 end
250 for other in receivers(market)
251 (receiver == other || !(other in market.preferences[proposer]) || !(proposer in market.preferences[other])) && continue
252 rank_in(market.preferences[proposer], other) >= rank_in(market.preferences[proposer], receiver) && continue
253 partners = findall(==(other), matching[1:market.proposers])
254 worst = maximum((rank_in(market.preferences[other], p) for p in partners); init = 0)
255 if length(partners) < market.capacities[other] || rank_in(market.preferences[other], proposer) < worst
256 push!(blocking, "BLOCKING $(market.names[proposer]) $(market.names[other])")
257 end
258 end
259 end
260 else
261 for first in eachindex(market.names), second in (first + 1):length(market.names)
262 mutual = second in market.preferences[first] && first in market.preferences[second]
263 if matching[first] == second
264 !mutual && push!(unacceptable, "UNACCEPTABLE $(market.names[first]) $(market.names[second])")
265 elseif mutual && rank_in(market.preferences[first], second) < rank_in(market.preferences[first], matching[first]) &&
266 rank_in(market.preferences[second], first) < rank_in(market.preferences[second], matching[second])
267 push!(blocking, "BLOCKING $(market.names[first]) $(market.names[second])")
268 end
269 end
270 end
271 isempty(unacceptable) && isempty(blocking) && return ["CHECK $(market.name) STABLE"]
272 vcat(["CHECK $(market.name) UNSTABLE unacceptable=$(length(unacceptable)) blocking=$(length(blocking))"], unacceptable, blocking)
273end

275function remove_pair!(lists::Vector{Vector{Int}}, first::Int, second::Int)
276 filter!(!=(second), lists[first])
277 filter!(!=(first), lists[second])

282function roommates(market::Market, verbose::Bool)
283 require_kind(market, ROOMMATE)
284 incomplete = findfirst(list -> length(list) < length(market.names) - 1, market.preferences)
285 isnothing(incomplete) || fail("INCOMPLETE $(market.names[incomplete])")
286 lists = copy.(market.preferences)
287 held = zeros(Int, length(market.names))
288 queue = collect(eachindex(market.names))
289 proposals = 0
290 failure = nothing
291 while proposals < length(queue)
292 proposer = queue[proposals + 1]
293 proposals += 1
294 receiver = first(lists[proposer])
295 held[receiver] > 0 && push!(queue, held[receiver])
296 held[receiver] = proposer
297 position = something(findfirst(==(proposer), lists[receiver]))
298 removed = lists[receiver][(position + 1):end]
299 for other in removed
300 remove_pair!(lists, receiver, other)
301 end
302 failure = findfirst(isempty, lists)
303 isnothing(failure) || break
304 end
305 history = isnothing(failure) && verbose ? table_lines(market, lists) : String[]
306 rotations = 0
307 while isnothing(failure)
308 point = findfirst(list -> length(list) >= 2, lists)
309 isnothing(point) && break
310 path = Int[]
311 while !(point in path)
312 push!(path, point)
313 second = lists[point][2]
314 point = last(lists[second])
315 end
316 rotation = path[something(findfirst(==(point), path)):end]
317 front = [first(lists[a]) for a in rotation]
318 notation = join((market.names[rotation[i]] * "-" * market.names[front[i]] for i in eachindex(rotation)), " ")
319 rotations += 1
320 push!(history, "ROTATION $rotations $notation")
321 removals = Tuple{Int,Int}[]
322 for i in eachindex(rotation)
323 receiver = front[mod1(i + 1, length(front))]
324 position = something(findfirst(==(rotation[i]), lists[receiver]))
325 for other in lists[receiver][(position + 1):end]
326 push!(removals, (receiver, other))
327 end
328 end
329 for (left, right) in removals
330 remove_pair!(lists, left, right)
331 end
332 verbose && append!(history, table_lines(market, lists))
333 failure = findfirst(isempty, lists)
334 end
335 rows = vcat(["ROOMMATES $(market.name) proposals=$proposals"], history)
336 if !isnothing(failure)
337 push!(rows, "NO STABLE MATCHING $(market.names[failure])")
338 else
339 pairs = [market.names[a] * "-" * market.names[first(lists[a])] for a in eachindex(market.names) if a < first(lists[a])]
340 push!(rows, "MATCHING " * join(pairs, " "))
341 end
342 rows
343end

345function effective_list(market::Market, agent::Int)
346 preference = market.preferences[agent]
347 position = findfirst(==(agent), preference)
348 isnothing(position) ? vcat(preference, agent) : preference[1:position]

351function ttc(market::Market)
352 remaining = trues(length(market.names))
353 allocation = zeros(Int, length(market.names))
354 effective = [effective_list(market, agent) for agent in eachindex(market.names)]
355 cycles = String[]
356 while any(remaining)
357 pointing = zeros(Int, length(market.names))
358 for agent in findall(remaining)
359 pointing[agent] = first(house for house in effective[agent] if remaining[house])
360 end
361 point = something(findfirst(remaining))
362 path = Int[]
363 while !(point in path)
364 push!(path, point)
365 point = pointing[point]
366 end
367 cycle = path[something(findfirst(==(point), path)):end]
368 allocation[cycle] = pointing[cycle]
369 remaining[cycle] .= false
370 notation = join((market.names[a] * ":" * market.houses[pointing[a]] for a in cycle), " ")
371 push!(cycles, "CYCLE $(length(cycles) + 1) $notation")
372 end
373 vcat(["TTC $(market.name) cycles=$(length(cycles))"], cycles, ["ALLOCATION " * allocation_text(market, allocation)])
374end

376function serial(market::Market, words)
377 order = Int[]
378 for word in words[3:end]
379 agent = agent_index(market, word)
380 agent in order && fail("DUPLICATE AGENT $word")
381 push!(order, agent)
382 end
383 length(order) == length(market.names) || fail("BAD ORDER")
384 remaining = trues(length(market.names))
385 allocation = zeros(Int, length(market.names))
386 for agent in order
387 for house in market.preferences[agent]
388 if remaining[house]
389 allocation[agent] = house
390 remaining[house] = false
391 break
392 end
393 end
394 end
395 worse = [market.names[a] for a in eachindex(market.names) if !(allocation[a] in effective_list(market, a))]
396 ["SERIAL $(market.name) assigned=$(count(>(0), allocation))", "ALLOCATION " * allocation_text(market, allocation), "WORSE " * list_text(worse)]
397end

399function hospital_report(market::Market)
400 resident = deferred(market).matching
401 hospital = deferred(market; reverse = true).matching
402 unmatched = market.names[findall(==(0), resident)]
403 rows = ["HR $(market.name) residents=$(market.proposers) hospitals=$(length(market.names) - market.proposers)",
404 "RESIDENT-OPTIMAL " * pair_text(market, resident), "HOSPITAL-OPTIMAL " * pair_text(market, hospital),
405 "UNMATCHED " * list_text(unmatched)]
406 rural = (resident .> 0) == (hospital .> 0)
407 for receiver in receivers(market)
408 first = findall(==(receiver), resident)
409 second = findall(==(receiver), hospital)
410 same = first == second
411 capacity = market.capacities[receiver]
412 label = same ? "YES" : "NO"
413 push!(rows, "HOSPITAL $(market.names[receiver]) cap=$capacity resident=$(length(first)) hospital=$(length(second)) same=$label")
414 rural &= length(first) == length(second) && (length(first) == capacity || same)
415 end
416 push!(rows, rural ? "RURAL HOLDS" : "RURAL FAILS")
417end

419function manipulate(market::Market, proposer::Int)
420 truth = market.preferences[proposer]
421 baseline_b = boston(market).matching[proposer]
422 baseline_g = deferred(market).matching[proposer]
423 swaps = max(0, length(truth) - 1)
424 rows = ["MANIPULATE $(market.name) $(market.names[proposer]) boston=$(partner_name(market, baseline_b)) gs=$(partner_name(market, baseline_g)) swaps=$swaps"]
425 for index in 1:swaps
426 preferences = copy(market.preferences)
427 preferences[proposer] = copy(truth)
428 preferences[proposer][index], preferences[proposer][index + 1] = truth[index + 1], truth[index]
429 trial = Market(market.name, market.kind, market.names, market.proposers, market.houses, preferences, market.capacities, market.current)
430 results = (boston(trial).matching[proposer], deferred(trial).matching[proposer])
431 baselines = (baseline_b, baseline_g)
432 for (kind, label) in enumerate(("BOSTON", "GS"))
433 if rank_in(truth, results[kind]) < rank_in(truth, baselines[kind])
434 push!(rows, "IMPROVE $label $index $(market.names[truth[index]]) $(market.names[truth[index + 1]]) result=$(partner_name(market, results[kind]))")
435 end
436 end
437 end
438 length(rows) == 1 && push!(rows, "NO IMPROVEMENT")
439 rows
440end

442function statistics(market::Market)
443 saved = market.current
444 isnothing(saved) && fail("NO MATCHING")
445 matched = findall(>(0), saved.matching)
446 pairs = length(matched)
447 proposer_ranks = [rank_in(market.preferences[p], saved.matching[p]) for p in matched]
448 receiver_ranks = [rank_in(market.preferences[saved.matching[p]], p) for p in matched]
449 ranks = (proposer_ranks, receiver_ranks)
450 unmatched = (market.proposers - pairs, length(market.names) - market.proposers - length(unique(saved.matching[matched])))
451 rows = ["STATS $(market.name) source=$(saved.source) pairs=$pairs"]
452 for (side, label) in enumerate(("PROPOSERS", "RECEIVERS"))
453 total = sum(ranks[side])
454 average = truncated = "-"
455 if pairs > 0
456 divisor = gcd(total, pairs)
457 average = "$(div(total, divisor))/$(div(pairs, divisor))"
458 cents = div(100 * total, pairs)
459 truncated = @sprintf("%d.%02d", div(cents, 100), mod(cents, 100))
460 end
461 push!(rows, "$label sum=$total average=$average truncated=$truncated first=$(count(==(1), ranks[side])) unmatched=$(unmatched[side])")
462 end
463 rows
464end

466function show_market(market::Market)
467 rows = if market.kind == TWO_SIDED
468 ["SHOW $(market.name) MARKET proposers=$(market.proposers) receivers=$(length(market.names) - market.proposers)"]
469 else
470 kind = market.kind == ROOMMATE ? "ROOMS" : "HOUSES"
471 ["SHOW $(market.name) $kind agents=$(length(market.names))"]
472 end

474 prefix = if market.kind == TWO_SIDED

484 push!(rows, "CURRENT $source")

489function new_market(words)
490 command, name = words[1:2]
491 kind = command == "ROOMS" ? ROOMMATE : command == "HOUSES" ? HOUSING : TWO_SIDED
492 if command == "RANDOM"
493 n = parse(Int, words[3])
494 state = parse(Int64, words[4])
495 names = vcat(["P$i" for i in 1:n], ["R$i" for i in 1:n])
496 market = Market(name, TWO_SIDED, names, n)
497 for agent in eachindex(names)
498 preference = collect(agent <= n ? ((n + 1):(2 * n)) : (1:n))
499 for size in n:-1:2
500 state = mod(48271 * state, 2147483647)
501 index = Int(mod(state, size)) + 1
502 preference[size], preference[index] = preference[index], preference[size]
503 end
504 market.preferences[agent] = preference
505 end
506 return market
507 end
508 names = String[]
509 houses = String[]
510 proposers = 0
511 for word in words[3:end]
512 if word == "|"
513 proposers = length(names)
514 continue
515 end
516 pair = split(word, ':')
517 pair[1] in names && fail("DUPLICATE AGENT $(pair[1])")
518 if kind == HOUSING
519 pair[2] in houses && fail("DUPLICATE HOUSE $(pair[2])")
520 push!(houses, String(pair[2]))
521 end
522 push!(names, String(pair[1]))
523 end
524 Market(name, kind, names, proposers, houses)
525end

532function execute!(markets::Dict{String,Market}, line::AbstractString)
533 words = split(line, ' ')
534 command, name = words[1:2]
535 if command in ("MARKET", "ROOMS", "HOUSES", "RANDOM")
536 haskey(markets, name) && fail("DUPLICATE MARKET $name")
537 market = new_market(words)
538 response = if command == "RANDOM"
539 "RANDOM $name n=$(market.proposers) seed=$(words[4])"
540 elseif market.kind == TWO_SIDED
541 "MARKET $name proposers=$(market.proposers) receivers=$(length(market.names) - market.proposers)"
542 else
543 "$command $name agents=$(length(market.names))"
544 end
545 markets[String(name)] = market
546 return [response]
547 end
548 haskey(markets, name) || fail("UNKNOWN MARKET $name")
549 market = markets[name]
550 if command == "SHOW"
551 return show_market(market)
552 elseif command == "PREF"
553 agent = agent_index(market, words[3])
554 names = market.kind == HOUSING ? market.houses : market.names
555 preference = Int[]
556 for word in words[4:end]
557 choice = findfirst(==(word), names)
558 if isnothing(choice) || (market.kind == ROOMMATE && choice == agent) ||
559 (market.kind == TWO_SIDED && (agent <= market.proposers) == (choice <= market.proposers))
560 fail("BAD CHOICE $word")
561 end
562 choice in preference && fail("DUPLICATE CHOICE $word")
563 push!(preference, choice)
564 end
565 market.preferences[agent] = preference
566 market.kind == TWO_SIDED && (market.current = nothing)
567 return ["PREF $name $(words[3]) length=$(length(preference))"]
568 elseif command == "CHECK"
569 return check_pairs(market, words)
570 elseif command == "ROOMMATES"
571 return roommates(market, length(words) == 3)
572 elseif command in ("TTC", "SERIAL")
573 require_kind(market, HOUSING)
574 return command == "TTC" ? ttc(market) : serial(market, words)
575 end
576 require_kind(market, TWO_SIDED)
577 if command == "CAPACITY"
578 receiver = agent_index(market, words[3], RECEIVER)
579 capacity = parse(Int, words[4])
580 market.capacities[receiver] = capacity
581 market.current = nothing
582 return ["CAPACITY $name $(words[3]) $capacity"]
583 elseif command in ("GS", "GSR", "BOSTON")
584 run = command == "BOSTON" ? boston(market) : deferred(market; reverse = command == "GSR", verbose = length(words) == 3)
585 save!(market, command, run.matching)
586 label = command == "BOSTON" ? "applications" : "proposals"
587 return vcat(["$command $name $label=$(run.count)"], run.history, ["MATCHING " * pair_text(market, run.matching)])
588 elseif command == "HR"
589 return hospital_report(market)
590 elseif command == "MANIPULATE"
591 return manipulate(market, agent_index(market, words[3], PROPOSER))
592 elseif command == "STATS"
593 return statistics(market)
594 end
595 lattice = enumerate_matchings(market)
596 if command == "LATTICE"
597 rows = ["LATTICE $name matchings=$(length(lattice.matchings)) rotations=$(length(lattice.rotations))"]
598 for (index, result) in enumerate(lattice.matchings)
599 origin = result.origin == 0 ? "-" : string(result.origin)
600 via = result.via == 0 ? "-" : string(result.via)
601 push!(rows, "MATCHING $index from=$origin via=$via cost=$(result.cost) regret=$(result.regret) " * pair_text(market, result.matching))
602 end
603 for (index, notation) in enumerate(lattice.rotations)
604 push!(rows, "ROTATION $index $notation")
605 end
606 return rows
607 end
608 values = [command == "EGALITARIAN" ? result.cost : result.regret for result in lattice.matchings]
609 best = argmin(values)
610 selected = lattice.matchings[best].matching
611 save!(market, command, selected)
612 label = command == "EGALITARIAN" ? "cost" : "regret"
613 ["$command $name matching=$best $label=$(values[best])", "MATCHING " * pair_text(market, selected)]
614end
