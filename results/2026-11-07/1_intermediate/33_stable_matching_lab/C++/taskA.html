26template<class T> int position_of(const vector<T>& values, const T& value) {
27 auto found = std::find(values.begin(), values.end(), value);
28 return found == values.end() ? -1 : static_cast<int>(found - values.begin());
29}
30int rank_in(const Preference& preference, int partner) {
31 int index = position_of(preference, partner);
32 return index < 0 ? static_cast<int>(preference.size()) + 1 : index + 1;
33}
34vector<int> sequence(int start, int end) {
35 vector<int> result(end - start);
36 std::iota(result.begin(), result.end(), start);
37 return result;
38}

46 string result;

48 if (i) result += separator;

58struct Market {
59 string name;
60 Kind kind;
61 vector<string> names;
62 int proposers;
63 vector<string> houses;
64 vector<Preference> preferences;
65 vector<int> capacities;
66 std::optional<Saved> current;

68 Market(string name, Kind kind, vector<string> names, int proposers = 0, vector<string> houses = {})
69 : name(std::move(name)), kind(kind), names(std::move(names)), proposers(proposers), houses(std::move(houses)),
70 preferences(this->names.size()), capacities(this->names.size(), 1) {}
71 int size() const { return static_cast<int>(names.size()); }
72 void require_kind(Kind expected) const { if (kind != expected) fail("WRONG TYPE " + name); }
73 int agent(const string& name, std::optional<Side> side = std::nullopt) const {
74 int index = position_of(names, name);
75 if (index < 0) fail("UNKNOWN AGENT " + name);
76 if (side == Side::Proposer && index >= proposers) fail("NOT PROPOSER " + name);
77 if (side == Side::Receiver && index < proposers) fail("NOT RECEIVER " + name);
78 return index;
79 }
80 Preference effective(int agent) const {
81 Preference result = preferences[agent];
82 int own = position_of(result, agent);
83 if (own < 0) result.push_back(agent);
84 else result.resize(own + 1);
85 return result;
86 }
87 string partner_name(int partner) const { return partner < 0 ? "*" : names[partner]; }
88};
89string pair_text(const Market& market, const Matching& matching) {
90 vector<string> pairs;
91 for (int p = 0; p < market.proposers; ++p) pairs.push_back(market.names[p] + "-" + market.partner_name(matching[p]));
92 return join(pairs);
93}
94string rotation_text(const Market& market, const Matching& matching, const vector<int>& rotation) {
95 vector<string> pairs;
96 for (int p : rotation) pairs.push_back(market.names[p] + "-" + market.names[matching[p]]);
97 return join(pairs);
98}
99string allocation_text(const Market& market, const vector<int>& allocation) {
100 vector<string> parts;
101 for (int a = 0; a < market.size(); ++a) parts.push_back(market.names[a] + "-" + (allocation[a] < 0 ? "*" : market.houses[allocation[a]]));
102 return join(parts);
103}
104vector<int> partners_of(const Matching& matching, int receiver) {
105 vector<int> partners;
106 for (std::size_t p = 0; p < matching.size(); ++p) if (matching[p] == receiver) partners.push_back(static_cast<int>(p));
107 return partners;

110struct DeferredResult { Matching matching; int proposals; vector<string> history; };
111DeferredResult deferred(const Market& market, bool reverse = false, bool verbose = false) {
112 vector<int> next(market.size()), counts(market.size());
113 vector<vector<int>> holding(market.size());
114 vector<bool> queued(market.size());
115 auto able = [&](int agent) {
116 return counts[agent] < market.capacities[agent] && next[agent] < static_cast<int>(market.preferences[agent].size());
117 };
118 vector<int> proposing = reverse ? sequence(market.proposers, market.size()) : sequence(0, market.proposers);
119 vector<int> queue;
120 for (int agent : proposing) if (able(agent)) { queue.push_back(agent); queued[agent] = true; }
121 DeferredResult result{Matching(market.proposers, -1), 0, {}};
122 for (std::size_t head = 0; head < queue.size(); ++head) {
123 int proposer = queue[head];
124 queued[proposer] = false;
125 int receiver = market.preferences[proposer][next[proposer]++];
126 const Preference& preference = market.preferences[receiver];
127 ++result.proposals;
128 string action = "REJECT";
129 if (position_of(preference, proposer) >= 0) {
130 if (holding[receiver].size() < static_cast<std::size_t>(market.capacities[receiver])) {
131 holding[receiver].push_back(proposer);
132 ++counts[proposer];
133 action = "HOLD";
134 } else {
135 auto worst = std::max_element(holding[receiver].begin(), holding[receiver].end(), [&](int a, int b) {
136 return rank_in(preference, a) < rank_in(preference, b);
137 });
138 int rejected = *worst;
139 if (rank_in(preference, proposer) < rank_in(preference, rejected)) {
140 *worst = proposer;
141 ++counts[proposer];
142 --counts[rejected];
143 action = "HOLD REJECT " + market.names[rejected];
144 if (able(rejected) && !queued[rejected]) { queue.push_back(rejected); queued[rejected] = true; }
145 }
146 }
147 }
148 if (verbose) result.history.push_back("PROPOSE " + market.names[proposer] + " " + market.names[receiver] + " " + action);
149 if (able(proposer)) { queue.push_back(proposer); queued[proposer] = true; }
150 }
151 vector<int> receiving = reverse ? sequence(0, market.proposers) : sequence(market.proposers, market.size());
152 for (int receiver : receiving) {
153 for (int proposer : holding[receiver]) {
154 if (reverse) result.matching[receiver] = proposer;
155 else result.matching[proposer] = receiver;
156 }
157 }
158 return result;
159}
160struct BostonResult { Matching matching; int applications; };
161BostonResult boston(const Market& market) {
162 BostonResult result{Matching(market.proposers, -1), 0};
163 vector<int> remaining = market.capacities;
164 int rounds = 0;
165 for (int p = 0; p < market.proposers; ++p) rounds = std::max(rounds, static_cast<int>(market.preferences[p].size()));
166 for (int round = 0; round < rounds; ++round) {
167 vector<int> applied(market.proposers, -1);
168 for (int p = 0; p < market.proposers; ++p) {
169 if (result.matching[p] < 0 && round < static_cast<int>(market.preferences[p].size())) {
170 applied[p] = market.preferences[p][round];
171 ++result.applications;
172 }
173 }
174 for (int r = market.proposers; r < market.size(); ++r) {
175 for (int p : market.preferences[r]) {
176 if (remaining[r] > 0 && applied[p] == r) {
177 result.matching[p] = r;
178 --remaining[r];
179 }
180 }
181 }
182 }
183 return result;
184}

186vector<vector<int>> exposed_rotations(const Market& market, const Matching& matching) {
187 vector<int> owner(market.size(), -1), successor(market.proposers, -1);
188 for (int p = 0; p < market.proposers; ++p) if (matching[p] >= 0) owner[matching[p]] = p;
189 for (int p = 0; p < market.proposers; ++p) {
190 if (matching[p] < 0) continue;
191 const Preference& preference = market.preferences[p];
192 for (std::size_t i = rank_in(preference, matching[p]); i < preference.size(); ++i) {
193 int r = preference[i];
194 const Preference& ranking = market.preferences[r];
195 if (position_of(ranking, p) >= 0 && (owner[r] < 0 || rank_in(ranking, p) < rank_in(ranking, owner[r]))) {
196 successor[p] = owner[r];
197 break;
198 }
199 }
200 }
201 vector<bool> covered(market.proposers);
202 vector<vector<int>> rotations;
203 for (int start = 0; start < market.proposers; ++start) {
204 if (covered[start]) continue;
205 vector<int> path;
206 int point = start;
207 while (point >= 0 && position_of(path, point) < 0) { path.push_back(point); point = successor[point]; }
208 if (point == start) {
209 for (int p : path) covered[p] = true;
210 rotations.push_back(path);
211 }
212 }
213 return rotations;
214}
215struct Metrics { int cost = 0; int regret = 0; };
216Metrics metrics(const Market& market, const Matching& matching) {
217 Metrics result;
218 for (int p = 0; p < market.proposers; ++p) {
219 int r = matching[p];
220 if (r < 0) continue;
221 int first = rank_in(market.preferences[p], r), second = rank_in(market.preferences[r], p);
222 result.cost += first + second;
223 result.regret = std::max({result.regret, first, second});
224 }
225 return result;
226}
227struct Entry { Matching matching; int origin; int via; };
228struct Lattice { vector<Entry> entries; vector<string> rotations; };
229Lattice enumerate(const Market& market) {
230 for (int r = market.proposers; r < market.size(); ++r) if (market.capacities[r] != 1) fail("NOT ONE TO ONE");
231 Lattice result{{{deferred(market).matching, 0, 0}}, {}};
232 std::set<Matching> seen{result.entries.front().matching};
233 for (std::size_t index = 0; index < result.entries.size(); ++index) {
234 Matching matching = result.entries[index].matching;
235 for (const vector<int>& rotation : exposed_rotations(market, matching)) {
236 string notation = rotation_text(market, matching, rotation);
237 int identifier = position_of(result.rotations, notation);
238 if (identifier < 0) { identifier = static_cast<int>(result.rotations.size()); result.rotations.push_back(notation); }
239 Matching following = matching;
240 for (std::size_t i = 0; i < rotation.size(); ++i) following[rotation[i]] = matching[rotation[(i + 1) % rotation.size()]];
241 if (seen.count(following)) continue;
242 if (result.entries.size() == 50) fail("TOO MANY MATCHINGS");
243 seen.insert(following);
244 result.entries.push_back({following, static_cast<int>(index) + 1, identifier + 1});
245 }
246 }
247 return result;
248}
249vector<string> check_pairs(const Market& market, const vector<string>& pairs) {
250 if (market.kind == Kind::Housing) fail("WRONG TYPE " + market.name);
251 Matching matching(market.size(), -1);
252 vector<int> counts(market.size());
253 for (const string& word : pairs) {
254 vector<string> pair = split(word, '-');
255 std::optional<Side> first_side = market.kind == Kind::TwoSided ? std::optional<Side>(Side::Proposer) : std::nullopt;
256 std::optional<Side> second_side = market.kind == Kind::TwoSided ? std::optional<Side>(Side::Receiver) : std::nullopt;
257 int first = market.agent(pair[0], first_side);
258 int second = market.agent(pair[1], second_side);
259 if (matching[first] >= 0) fail("DUPLICATE AGENT " + pair[0]);
260 if (market.kind == Kind::TwoSided) {
261 if (++counts[second] > market.capacities[second]) fail("OVER CAPACITY " + pair[1]);
262 } else {
263 if (first == second || matching[second] >= 0) fail("DUPLICATE AGENT " + pair[1]);
264 matching[second] = first;
265 }
266 matching[first] = second;
267 }
268 vector<string> unacceptable, blocking;
269 if (market.kind == Kind::TwoSided) {
270 for (int p = 0; p < market.proposers; ++p) {
271 int r = matching[p];
272 const Preference& preference = market.preferences[p];
273 if (r >= 0 && (position_of(preference, r) < 0 || position_of(market.preferences[r], p) < 0)) {
274 unacceptable.push_back("UNACCEPTABLE " + market.names[p] + " " + market.names[r]);
275 }
276 for (int other = market.proposers; other < market.size(); ++other) {
277 const Preference& ranking = market.preferences[other];
278 if (r == other || position_of(preference, other) < 0 || position_of(ranking, p) < 0) continue;
279 if (rank_in(preference, other) >= rank_in(preference, r)) continue;
280 int worst = 0;
281 for (int partner = 0; partner < market.proposers; ++partner) {
282 if (matching[partner] == other) worst = std::max(worst, rank_in(ranking, partner));
283 }
284 if (counts[other] < market.capacities[other] || rank_in(ranking, p) < worst) {
285 blocking.push_back("BLOCKING " + market.names[p] + " " + market.names[other]);
286 }
287 }
288 }
289 } else {
290 for (int first = 0; first < market.size(); ++first) {
291 for (int second = first + 1; second < market.size(); ++second) {
292 bool mutual = position_of(market.preferences[first], second) >= 0 && position_of(market.preferences[second], first) >= 0;
293 string notation = market.names[first] + " " + market.names[second];
294 if (matching[first] == second) {
295 if (!mutual) unacceptable.push_back("UNACCEPTABLE " + notation);
296 } else if (mutual && rank_in(market.preferences[first], second) < rank_in(market.preferences[first], matching[first])
297 && rank_in(market.preferences[second], first) < rank_in(market.preferences[second], matching[second])) {
298 blocking.push_back("BLOCKING " + notation);
299 }
300 }
301 }
302 }
303 if (unacceptable.empty() && blocking.empty()) return {"CHECK " + market.name + " STABLE"};
304 vector<string> result{"CHECK " + market.name + " UNSTABLE unacceptable=" + to_string(unacceptable.size()) + " blocking=" + to_string(blocking.size())};
305 result.insert(result.end(), unacceptable.begin(), unacceptable.end());
306 result.insert(result.end(), blocking.begin(), blocking.end());
307 return result;
308}
309void remove_pair(vector<Preference>& lists, int first, int second) {
310 auto erase = [&](int a, int b) { lists[a].erase(std::remove(lists[a].begin(), lists[a].end(), b), lists[a].end()); };
311 erase(first, second);
312 erase(second, first);
313}

321vector<string> roommates(const Market& market, bool verbose) {
322 market.require_kind(Kind::Roommate);
323 for (int a = 0; a < market.size(); ++a) {
324 if (static_cast<int>(market.preferences[a].size()) < market.size() - 1) fail("INCOMPLETE " + market.names[a]);
325 }
326 vector<Preference> lists = market.preferences;
327 vector<int> held(market.size(), -1), queue = sequence(0, market.size());
328 int proposals = 0, failure = -1;
329 for (std::size_t head = 0; head < queue.size(); ++head) {
330 int proposer = queue[head], receiver = lists[proposer].front();
331 if (held[receiver] >= 0) queue.push_back(held[receiver]);
332 held[receiver] = proposer;
333 ++proposals;
334 Preference preference = lists[receiver];
335 for (std::size_t i = position_of(preference, proposer) + 1; i < preference.size(); ++i) remove_pair(lists, receiver, preference[i]);
336 failure = empty_list(lists);
337 if (failure >= 0) break;
338 }
339 vector<string> history;
340 if (failure < 0 && verbose) append_table(history, market, lists);
341 int rotations = 0;
342 while (failure < 0) {
343 int point = -1;
344 for (int a = 0; a < market.size(); ++a) if (lists[a].size() >= 2) { point = a; break; }
345 if (point < 0) break;
346 vector<int> path;
347 while (position_of(path, point) < 0) { path.push_back(point); point = lists[lists[point][1]].back(); }
348 vector<int> rotation(path.begin() + position_of(path, point), path.end()), front;
349 vector<string> pairs;
350 for (int a : rotation) { front.push_back(lists[a].front()); pairs.push_back(market.names[a] + "-" + market.names[lists[a].front()]); }
351 history.push_back("ROTATION " + to_string(++rotations) + " " + join(pairs));
352 vector<std::pair<int, int>> removals;
353 for (std::size_t i = 0; i < rotation.size(); ++i) {
354 int receiver = front[(i + 1) % front.size()];
355 const Preference& preference = lists[receiver];
356 for (std::size_t j = position_of(preference, rotation[i]) + 1; j < preference.size(); ++j) removals.emplace_back(receiver, preference[j]);
357 }
358 for (auto [first, second] : removals) remove_pair(lists, first, second);
359 if (verbose) append_table(history, market, lists);
360 failure = empty_list(lists);
361 }
362 vector<string> result{"ROOMMATES " + market.name + " proposals=" + to_string(proposals)};
363 result.insert(result.end(), history.begin(), history.end());
364 if (failure >= 0) { result.push_back("NO STABLE MATCHING " + market.names[failure]); return result; }
365 vector<string> pairs;
366 for (int a = 0; a < market.size(); ++a) if (a < lists[a].front()) pairs.push_back(market.names[a] + "-" + market.names[lists[a].front()]);
367 result.push_back("MATCHING " + join(pairs));
368 return result;
369}

371vector<string> ttc(const Market& market) {
372 std::set<int> remaining;
373 for (int a = 0; a < market.size(); ++a) remaining.insert(a);
374 vector<int> allocation(market.size(), -1);
375 vector<string> cycles;
376 while (!remaining.empty()) {
377 vector<int> pointing(market.size(), -1);
378 for (int a : remaining) {
379 for (int house : market.effective(a)) if (remaining.count(house)) { pointing[a] = house; break; }
380 }
381 vector<int> path;
382 int point = *remaining.begin();
383 while (position_of(path, point) < 0) { path.push_back(point); point = pointing[point]; }
384 vector<int> cycle(path.begin() + position_of(path, point), path.end());
385 vector<string> parts;
386 for (int a : cycle) {
387 allocation[a] = pointing[a];
388 remaining.erase(a);
389 parts.push_back(market.names[a] + ":" + market.houses[allocation[a]]);
390 }
391 cycles.push_back("CYCLE " + to_string(cycles.size() + 1) + " " + join(parts));
392 }
393 vector<string> result{"TTC " + market.name + " cycles=" + to_string(cycles.size())};
394 result.insert(result.end(), cycles.begin(), cycles.end());
395 result.push_back("ALLOCATION " + allocation_text(market, allocation));
396 return result;
397}
398vector<string> serial(const Market& market, const vector<string>& names) {
399 vector<int> order;
400 for (const string& name : names) {
401 int a = market.agent(name);
402 if (position_of(order, a) >= 0) fail("DUPLICATE AGENT " + name);
403 order.push_back(a);
404 }
405 if (order.size() != market.names.size()) fail("BAD ORDER");
406 vector<bool> remaining(market.size(), true);
407 vector<int> allocation(market.size(), -1);
408 int assigned = 0;
409 for (int a : order) {
410 for (int house : market.preferences[a]) {
411 if (remaining[house]) { allocation[a] = house; remaining[house] = false; ++assigned; break; }
412 }
413 }
414 vector<int> worse;
415 for (int a = 0; a < market.size(); ++a) if (position_of(market.effective(a), allocation[a]) < 0) worse.push_back(a);
416 return {"SERIAL " + market.name + " assigned=" + to_string(assigned), "ALLOCATION " + allocation_text(market, allocation), "WORSE " + list_text(market.names, worse)};
417}
418vector<string> hospital_report(const Market& market) {
419 Matching resident = deferred(market).matching, hospital = deferred(market, true).matching;
420 vector<int> unmatched;
421 bool rural = true;
422 for (int p = 0; p < market.proposers; ++p) {
423 if (resident[p] < 0) unmatched.push_back(p);
424 if ((resident[p] >= 0) != (hospital[p] >= 0)) rural = false;
425 }
426 vector<string> result{
427 "HR " + market.name + " residents=" + to_string(market.proposers) + " hospitals=" + to_string(market.size() - market.proposers),
428 "RESIDENT-OPTIMAL " + pair_text(market, resident), "HOSPITAL-OPTIMAL " + pair_text(market, hospital),
429 "UNMATCHED " + list_text(market.names, unmatched),
430 };
431 for (int r = market.proposers; r < market.size(); ++r) {
432 vector<int> first = partners_of(resident, r), second = partners_of(hospital, r);
433 int capacity = market.capacities[r];
434 bool same = first == second;
435 result.push_back("HOSPITAL " + market.names[r] + " cap=" + to_string(capacity) + " resident=" + to_string(first.size())
436 + " hospital=" + to_string(second.size()) + " same=" + (same ? "YES" : "NO"));
437 if (first.size() != second.size() || (first.size() < static_cast<std::size_t>(capacity) && !same)) rural = false;
438 }
439 result.push_back(rural ? "RURAL HOLDS" : "RURAL FAILS");
440 return result;
441}
442vector<string> manipulate(const Market& market, int proposer) {
443 const Preference& truth = market.preferences[proposer];
444 int baseline_b = boston(market).matching[proposer], baseline_g = deferred(market).matching[proposer];
445 int swaps = std::max(0, static_cast<int>(truth.size()) - 1);
446 vector<string> output{"MANIPULATE " + market.name + " " + market.names[proposer] + " boston=" + market.partner_name(baseline_b)
447 + " gs=" + market.partner_name(baseline_g) + " swaps=" + to_string(swaps)};
448 for (int i = 0; i < swaps; ++i) {
449 Market trial = market;
450 std::swap(trial.preferences[proposer][i], trial.preferences[proposer][i + 1]);
451 struct Outcome { string label; int partner; int baseline; };
452 vector<Outcome> outcomes{{"BOSTON", boston(trial).matching[proposer], baseline_b}, {"GS", deferred(trial).matching[proposer], baseline_g}};
453 for (const Outcome& result : outcomes) {
454 if (rank_in(truth, result.partner) < rank_in(truth, result.baseline)) {
455 output.push_back("IMPROVE " + result.label + " " + to_string(i + 1) + " " + market.names[truth[i]] + " " + market.names[truth[i + 1]]
456 + " result=" + market.partner_name(result.partner));
457 }
458 }
459 }
460 if (output.size() == 1) output.push_back("NO IMPROVEMENT");
461 return output;
462}
463vector<string> statistics(const Market& market) {
464 if (!market.current) fail("NO MATCHING");
465 const Saved& saved = *market.current;
466 vector<int> proposer_ranks, receiver_ranks;
467 std::set<int> occupied;
468 for (int p = 0; p < market.proposers; ++p) {
469 int r = saved.matching[p];
470 if (r < 0) continue;
471 proposer_ranks.push_back(rank_in(market.preferences[p], r));
472 receiver_ranks.push_back(rank_in(market.preferences[r], p));
473 occupied.insert(r);
474 }
475 int pairs = static_cast<int>(proposer_ranks.size());
476 vector<string> output{"STATS " + market.name + " source=" + saved.source + " pairs=" + to_string(pairs)};
477 struct Summary { string label; vector<int> ranks; int unmatched; };
478 vector<Summary> sides{{"PROPOSERS", proposer_ranks, market.proposers - pairs},
479 {"RECEIVERS", receiver_ranks, market.size() - market.proposers - static_cast<int>(occupied.size())}};
480 for (const Summary& side : sides) {
481 int sum = std::accumulate(side.ranks.begin(), side.ranks.end(), 0);
482 int first = static_cast<int>(std::count(side.ranks.begin(), side.ranks.end(), 1));
483 string average = "-", truncated = "-";
484 if (pairs > 0) {
485 int divisor = std::gcd(sum, pairs), cents = 100 * sum / pairs;
486 average = to_string(sum / divisor) + "/" + to_string(pairs / divisor);
487 truncated = to_string(cents / 100) + "." + (cents % 100 < 10 ? "0" : "") + to_string(cents % 100);
488 }
489 output.push_back(side.label + " sum=" + to_string(sum) + " average=" + average + " truncated=" + truncated
490 + " first=" + to_string(first) + " unmatched=" + to_string(side.unmatched));
491 }
492 return output;
493}

495vector<string> show_market(const Market& market) {
496 string header = "SHOW " + market.name;
497 if (market.kind == Kind::TwoSided) header += " MARKET proposers=" + to_string(market.proposers) + " receivers=" + to_string(market.size() - market.proposers);
498 else header += string(market.kind == Kind::Roommate ? " ROOMS" : " HOUSES") + " agents=" + to_string(market.size());
499 vector<string> rows{header};
500 for (int a = 0; a < market.size(); ++a) {
501 string prefix;
502 if (market.kind == Kind::TwoSided) prefix = a < market.proposers ? "PROPOSER " + market.names[a]
503 : "RECEIVER " + market.names[a] + " cap=" + to_string(market.capacities[a]);
504 else {
505 prefix = "AGENT " + market.names[a];
506 if (market.kind == Kind::Housing) prefix += " house=" + market.houses[a];
507 }
508 rows.push_back(prefix + " " + list_text(market.kind == Kind::Housing ? market.houses : market.names, market.preferences[a]));
509 }
510 if (market.kind == Kind::TwoSided) rows.push_back("CURRENT " + (market.current ? market.current->source : "-"));
511 return rows;
512}
513Market new_market(const string& command, const string& name, const vector<string>& args) {
514 if (command == "RANDOM") {
515 int count = std::stoi(args[0]);
516 std::int64_t state = std::stoll(args[1]);
517 vector<string> names;
518 for (const string& prefix : {string("P"), string("R")}) {
519 for (int i = 1; i <= count; ++i) names.push_back(prefix + to_string(i));
520 }
521 Market market(name, Kind::TwoSided, names, count);
522 for (int agent = 0; agent < market.size(); ++agent) {
523 Preference preference = agent < count ? sequence(count, 2 * count) : sequence(0, count);
524 for (int length = count; length >= 2; --length) {
525 state = 48271 * state % 2147483647;
526 int index = static_cast<int>(state % length);
527 std::swap(preference[length - 1], preference[index]);
528 }
529 market.preferences[agent] = preference;
530 }
531 return market;
532 }
533 Kind kind = command == "ROOMS" ? Kind::Roommate : command == "HOUSES" ? Kind::Housing : Kind::TwoSided;
534 vector<string> names, houses;
535 int proposers = 0;
536 for (const string& word : args) {
537 if (word == "|") { proposers = static_cast<int>(names.size()); continue; }
538 vector<string> pair = split(word, ':');
539 if (position_of(names, pair[0]) >= 0) fail("DUPLICATE AGENT " + pair[0]);
540 if (kind == Kind::Housing) {
541 if (position_of(houses, pair[1]) >= 0) fail("DUPLICATE HOUSE " + pair[1]);
542 houses.push_back(pair[1]);
543 }
544 names.push_back(pair[0]);
545 }
546 return Market(name, kind, names, proposers, houses);
547}
548class Laboratory {
549 std::map<string, Market> markets_;
550public:
551 vector<string> execute(const string& line) {
552 vector<string> words = split(line);
553 string command = words[0], name = words[1];
554 vector<string> args(words.begin() + 2, words.end());
555 if (command == "MARKET" || command == "ROOMS" || command == "HOUSES" || command == "RANDOM") {
556 if (markets_.count(name)) fail("DUPLICATE MARKET " + name);
557 Market market = new_market(command, name, args);
558 string response;
559 if (command == "RANDOM") response = "RANDOM " + name + " n=" + to_string(market.proposers) + " seed=" + args[1];
560 else if (command == "MARKET") response = "MARKET " + name + " proposers=" + to_string(market.proposers) + " receivers=" + to_string(market.size() - market.proposers);
561 else response = command + " " + name + " agents=" + to_string(market.size());
562 markets_.emplace(name, std::move(market));
563 return {response};
564 }
565 auto found = markets_.find(name);
566 if (found == markets_.end()) fail("UNKNOWN MARKET " + name);
567 Market& market = found->second;
568 if (command == "SHOW") return show_market(market);
569 if (command == "PREF") {
570 int agent = market.agent(args[0]);
571 const vector<string>& names = market.kind == Kind::Housing ? market.houses : market.names;
572 Preference preference;
573 for (std::size_t i = 1; i < args.size(); ++i) {
574 int choice = position_of(names, args[i]);
575 if (choice < 0 || (market.kind == Kind::Roommate && choice == agent)
576 || (market.kind == Kind::TwoSided && ((agent < market.proposers) == (choice < market.proposers)))) fail("BAD CHOICE " + args[i]);
577 if (position_of(preference, choice) >= 0) fail("DUPLICATE CHOICE " + args[i]);
578 preference.push_back(choice);
579 }
580 market.preferences[agent] = preference;
581 if (market.kind == Kind::TwoSided) market.current.reset();
582 return {"PREF " + name + " " + args[0] + " length=" + to_string(preference.size())};
583 }
584 if (command == "CHECK") return check_pairs(market, args);
585 if (command == "ROOMMATES") return roommates(market, args.size() == 1);
586 if (command == "TTC" || command == "SERIAL") {
587 market.require_kind(Kind::Housing);
588 return command == "TTC" ? ttc(market) : serial(market, args);
589 }
590 market.require_kind(Kind::TwoSided);
591 if (command == "CAPACITY") {
592 int receiver = market.agent(args[0], Side::Receiver), capacity = std::stoi(args[1]);
593 market.capacities[receiver] = capacity;
594 market.current.reset();
595 return {"CAPACITY " + name + " " + args[0] + " " + to_string(capacity)};
596 }
597 if (command == "GS" || command == "GSR") {
598 DeferredResult run = deferred(market, command == "GSR", args.size() == 1);
599 market.current = Saved{command, run.matching};
600 vector<string> result{command + " " + name + " proposals=" + to_string(run.proposals)};
601 result.insert(result.end(), run.history.begin(), run.history.end());
602 result.push_back("MATCHING " + pair_text(market, run.matching));
603 return result;
604 }
605 if (command == "BOSTON") {
606 BostonResult run = boston(market);
607 market.current = Saved{command, run.matching};
608 return {"BOSTON " + name + " applications=" + to_string(run.applications), "MATCHING " + pair_text(market, run.matching)};
609 }
610 if (command == "HR") return hospital_report(market);
611 if (command == "MANIPULATE") return manipulate(market, market.agent(args[0], Side::Proposer));
612 if (command == "STATS") return statistics(market);
613 Lattice lattice = enumerate(market);
614 if (command == "LATTICE") {
615 vector<string> result{"LATTICE " + name + " matchings=" + to_string(lattice.entries.size()) + " rotations=" + to_string(lattice.rotations.size())};
616 for (std::size_t i = 0; i < lattice.entries.size(); ++i) {
617 const Entry& entry = lattice.entries[i];
618 Metrics measure = metrics(market, entry.matching);
619 string origin = i == 0 ? "-" : to_string(entry.origin), via = i == 0 ? "-" : to_string(entry.via);
620 result.push_back("MATCHING " + to_string(i + 1) + " from=" + origin + " via=" + via + " cost=" + to_string(measure.cost)
621 + " regret=" + to_string(measure.regret) + " " + pair_text(market, entry.matching));
622 }
623 for (std::size_t i = 0; i < lattice.rotations.size(); ++i) result.push_back("ROTATION " + to_string(i + 1) + " " + lattice.rotations[i]);
624 return result;
625 }
626 vector<int> values;
627 for (const Entry& entry : lattice.entries) {
628 Metrics measure = metrics(market, entry.matching);
629 values.push_back(command == "EGALITARIAN" ? measure.cost : measure.regret);
630 }
631 int best = static_cast<int>(std::min_element(values.begin(), values.end()) - values.begin());
632 const Matching& matching = lattice.entries[best].matching;
633 market.current = Saved{command, matching};
634 string label = command == "EGALITARIAN" ? "cost" : "regret";
635 return {command + " " + name + " matching=" + to_string(best + 1) + " " + label + "=" + to_string(values[best]), "MATCHING " + pair_text(market, matching)};
636 }
637};
