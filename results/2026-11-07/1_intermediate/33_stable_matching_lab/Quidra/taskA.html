88Run deferred(const Market market, bool reverse, verbose := false)
89 count := int(#market.names)
90 next := array(nat(count), fill = 0)
91 counts := array(nat(count), fill = 0)
92 int[][] holding := array(nat(count), fill = [])
93 side := if reverse then sequence(market.proposers, count) else sequence(0, market.proposers)
94 int[] queue := []
95 for agent := side[]
96 if able(market, agent, next, counts)
97 queue = queue.append(agent)
98 proposals := 0
99 string[] history := []
100 while #queue != 0
101 proposer := queue[0]
102 queue = queue[1:]
103 receiver := market.preferences[proposer][next[proposer]]
104 next[proposer] += 1
105 proposals += 1
106 action := "REJECT"
107 preference := market.preferences[receiver]
108 if position_of(preference, proposer) >= 0
109 if int(#holding[receiver]) < market.capacities[receiver]
110 holding[receiver] = holding[receiver].append(proposer)
111 counts[proposer] += 1
112 action = "HOLD"
113 else
114 worst := 0
115 for: 1 <= index < int(#holding[receiver])
116 if rank_in(preference, holding[receiver][index]) > rank_in(preference, holding[receiver][worst])
117 worst = index
118 rejected := holding[receiver][worst]
119 if rank_in(preference, proposer) < rank_in(preference, rejected)
120 holding[receiver][worst] = proposer
121 counts[rejected] -= 1
122 counts[proposer] += 1
123 action = "HOLD REJECT " + market.names[rejected]
124 if able(market, rejected, next, counts) and position_of(queue, rejected) < 0
125 queue = queue.append(rejected)
126 if verbose
127 history = history.append("PROPOSE " + market.names[proposer] + " " + market.names[receiver] + " " + action)
128 if able(market, proposer, next, counts)
129 queue = queue.append(proposer)
130 matching := array(nat(market.proposers), fill = -1)
131 receiving := if reverse then sequence(0, market.proposers) else sequence(market.proposers, count)
132 for receiver := receiving[]
133 for proposer := holding[receiver][]
134 if reverse
135 matching[receiver] = proposer
136 else
137 matching[proposer] = receiver
138 return Run(matching, proposals, history)

189 rotations = rotations.append(path)
190 return rotations
