class Run(int[] matching, int proposals, string[] history := [])
Run boston(const Market market)
    matching := array(nat(market.proposers), fill = -1)
    int[] remaining := market.capacities
    int[] lengths := []
    for: 0 <= proposer < market.proposers
        lengths = lengths.append(int(#market.preferences[proposer]))
    rounds := lengths.max()
    applications := 0
    for: 0 <= round < rounds
        applied := array(nat(market.proposers), fill = -1)
        for: 0 <= proposer < market.proposers
            if matching[proposer] < 0 and round < lengths[proposer]
                applied[proposer] = market.preferences[proposer][round]
                applications += 1
        for: market.proposers <= receiver < int(#market.names)
            for proposer := market.preferences[receiver][]
                if remaining[receiver] > 0 and applied[proposer] == receiver
                    matching[proposer] = receiver
                    remaining[receiver] -= 1
    return Run(matching, applications)
