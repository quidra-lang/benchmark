50static void append(Text *text, const char *format, ...) {
51 va_list args, copy;
52 va_start(args, format);
53 va_copy(copy, args);
54 int n = vsnprintf(NULL, 0, format, copy);
55 va_end(copy);
56 size_t needed = text->length + (size_t)n + 1;
57 if (needed > text->capacity) {
58 size_t capacity = text->capacity ? text->capacity * 2 : 256;
59 if (capacity < needed) capacity = needed;
60 char *p = realloc(text->data, capacity);
61 if (!p) exit(EXIT_FAILURE);
62 text->data = p;
63 text->capacity = capacity;
64 }
65 vsnprintf(text->data + text->length, text->capacity - text->length, format, args);
66 va_end(args);
67 text->length += (size_t)n;
68}

75static void enqueue(Queue *queue, int agent) {
76 queue->items[(queue->head + queue->count) % MAX_AGENTS] = agent;
77 ++queue->count;
78 queue->present[agent] = true;

81static int dequeue(Queue *queue) {
82 int agent = queue->items[queue->head];
83 queue->head = (queue->head + 1) % MAX_AGENTS;
84 --queue->count;
85 queue->present[agent] = false;
86 return agent;
87}

158static Deferred deferred(const Market *market, bool reverse, Text *history) {
159 Deferred result = { .matching = empty_matching() };
160 bool holding[MAX_AGENTS][MAX_AGENTS] = {{false}};
161 int next[MAX_AGENTS] = {0}, held[MAX_AGENTS] = {0};
162 Queue queue = {0};
163 int begin = reverse ? market->proposers : 0;
164 int end = reverse ? market->count : market->proposers;
165 for (int x = begin; x < end; ++x)
166 if (able(market, x, next, held)) enqueue(&queue, x);
167 while (queue.count) {
168 int x = dequeue(&queue), y = market->agents[x].preference[next[x]++];
169 int rejected = -1;
170 bool accepted = false;
171 ++result.proposals;
172 if (accepts(market, y, x)) {
173 if (held[y] < market->agents[y].capacity) accepted = true;
174 else {
175 int worst = -1;
176 for (int w = begin; w < end; ++w)
177 if (holding[y][w] && (worst < 0 || rank_of(market, y, w) > rank_of(market, y, worst)))
178 worst = w;
179 if (rank_of(market, y, x) < rank_of(market, y, worst)) {
180 accepted = true;
181 rejected = worst;
182 holding[y][worst] = false;
183 --held[y];
184 --held[worst];
185 if (able(market, worst, next, held) && !queue.present[worst]) enqueue(&queue, worst);
186 }
187 }
188 }
189 if (accepted) {
190 holding[y][x] = true;
191 ++held[x];
192 ++held[y];
193 }
194 if (history) {
195 append(history, "PROPOSE %s %s %s", market->agents[x].name, market->agents[y].name,
196 accepted ? "HOLD" : "REJECT");
197 if (rejected >= 0) append(history, " REJECT %s", market->agents[rejected].name);
198 append(history, "\n");
199 }
200 if (able(market, x, next, held)) enqueue(&queue, x);
201 }
202 for (int y = 0; y < market->count; ++y)
203 for (int x = begin; x < end; ++x)
204 if (holding[y][x]) result.matching.partner[reverse ? y : x] = reverse ? x : y;
205 return result;
206}

208static Boston boston(const Market *market) {
209 Boston result = { .matching = empty_matching() };
210 int remaining[MAX_AGENTS], rounds = 0;
211 for (int r = market->proposers; r < market->count; ++r) remaining[r] = market->agents[r].capacity;
212 for (int p = 0; p < market->proposers; ++p)
213 if (market->agents[p].length > rounds) rounds = market->agents[p].length;
214 for (int round = 0; round < rounds; ++round) {
215 int applications[MAX_SIDE];
216 for (int p = 0; p < market->proposers; ++p) {
217 applications[p] = -1;
218 if (result.matching.partner[p] < 0 && market->agents[p].length > round) {
219 applications[p] = market->agents[p].preference[round];
220 ++result.applications;
221 }
222 }
223 for (int r = market->proposers; r < market->count; ++r)
224 for (int i = 0; i < market->agents[r].length && remaining[r]; ++i) {
225 int p = market->agents[r].preference[i];
226 if (applications[p] == r) {
227 result.matching.partner[p] = r;
228 --remaining[r];
229 }
230 }
231 }
232 return result;
233}

235static int rotations(const Market *market, Matching matching, Rotation result[]) {
236 int owner[MAX_AGENTS], successor[MAX_SIDE];
237 for (int r = 0; r < market->count; ++r) owner[r] = -1;
238 for (int p = 0; p < market->proposers; ++p)
239 if (matching.partner[p] >= 0) owner[matching.partner[p]] = p;
240 for (int p = 0; p < market->proposers; ++p) {
241 successor[p] = -1;
242 if (matching.partner[p] < 0) continue;
243 const Agent *a = &market->agents[p];
244 for (int i = rank_of(market, p, matching.partner[p]); i < a->length; ++i) {
245 int r = a->preference[i];
246 if (accepts(market, r, p) && (owner[r] < 0 || rank_of(market, r, p) < rank_of(market, r, owner[r]))) {
247 successor[p] = owner[r];
248 break;
249 }
250 }
251 }
252 bool covered[MAX_SIDE] = {false};
253 int count = 0;
254 for (int start = 0; start < market->proposers; ++start) {
255 if (covered[start]) continue;
256 bool seen[MAX_SIDE] = {false};
257 Rotation rotation = {0};
258 int p = start;
259 while (p >= 0 && !seen[p]) {
260 seen[p] = true;
261 rotation.proposer[rotation.count] = p;
262 rotation.receiver[rotation.count++] = matching.partner[p];
263 p = successor[p];
264 }
265 if (p == start) {
266 result[count++] = rotation;
267 for (int i = 0; i < rotation.count; ++i) covered[rotation.proposer[i]] = true;
268 }
269 }
270 return count;
271}

273static bool same_rotation(const Rotation *a, const Rotation *b) {
274 if (a->count != b->count) return false;
275 for (int i = 0; i < a->count; ++i)
276 if (a->proposer[i] != b->proposer[i] || a->receiver[i] != b->receiver[i]) return false;
277 return true;
278}

280static bool enumerate(const Market *market, Lattice *lattice) {
281 lattice->count = 1;
282 lattice->rotation_count = 0;
283 lattice->nodes[0] = (Node){ .matching = deferred(market, false, NULL).matching };
284 for (int i = 0; i < lattice->count; ++i) {
285 Rotation sequence[MAX_SIDE];
286 Matching matching = lattice->nodes[i].matching;
287 int count = rotations(market, matching, sequence);
288 for (int j = 0; j < count; ++j) {
289 Rotation *rotation = &sequence[j];
290 int number = 0;
291 while (number < lattice->rotation_count && !same_rotation(rotation, &lattice->rotations[number])) ++number;
292 if (number == lattice->rotation_count) lattice->rotations[lattice->rotation_count++] = *rotation;
293 Matching next = matching;
294 for (int k = 0; k < rotation->count; ++k)
295 next.partner[rotation->proposer[k]] = rotation->receiver[(k + 1) % rotation->count];
296 int previous = 0;
297 while (previous < lattice->count && memcmp(next.partner, lattice->nodes[previous].matching.partner,
298 (size_t)market->proposers * sizeof(int))) ++previous;
299 if (previous == lattice->count) {
300 if (lattice->count == MAX_MATCHINGS) return false;
301 lattice->nodes[lattice->count++] = (Node){next, i + 1, number + 1};
302 }
303 }
304 }
305 return true;
306}

308static void metrics(const Market *market, Matching matching, int *cost, int *regret) {
309 *cost = *regret = 0;
310 for (int p = 0; p < market->proposers; ++p) {
311 int r = matching.partner[p];
312 if (r < 0) continue;
313 int a = rank_of(market, p, r), b = rank_of(market, r, p);
314 *cost += a + b;
315 if (a > *regret) *regret = a;
316 if (b > *regret) *regret = b;
317 }
318}

320static void check_matching(const Market *market, const int partner[]) {
321 Pair unacceptable[MAX_SIDE * MAX_SIDE], blocking[MAX_SIDE * MAX_SIDE];
322 int bad = 0, blocks = 0;
323 if (market->type == TWO_SIDED) {
324 int counts[MAX_AGENTS] = {0}, worst[MAX_AGENTS] = {0};
325 for (int p = 0; p < market->proposers; ++p) {
326 int r = partner[p];
327 if (r < 0) continue;
328 ++counts[r];
329 int rank = rank_of(market, r, p);
330 if (rank > worst[r]) worst[r] = rank;
331 if (!accepts(market, p, r) || !accepts(market, r, p)) unacceptable[bad++] = (Pair){p, r};
332 }
333 for (int p = 0; p < market->proposers; ++p)
334 for (int r = market->proposers; r < market->count; ++r)
335 if (partner[p] != r && accepts(market, p, r) && accepts(market, r, p) &&
336 rank_of(market, p, r) < rank_of(market, p, partner[p]) &&
337 (counts[r] < market->agents[r].capacity || rank_of(market, r, p) < worst[r]))
338 blocking[blocks++] = (Pair){p, r};
339 } else {
340 for (int a = 0; a < market->count; ++a)
341 for (int b = a + 1; b < market->count; ++b) {
342 bool mutual = accepts(market, a, b) && accepts(market, b, a);
343 if (partner[a] == b) {
344 if (!mutual) unacceptable[bad++] = (Pair){a, b};
345 } else if (mutual && rank_of(market, a, b) < rank_of(market, a, partner[a]) &&
346 rank_of(market, b, a) < rank_of(market, b, partner[b]))
347 blocking[blocks++] = (Pair){a, b};
348 }
349 }
350 if (!bad && !blocks) printf("CHECK %s STABLE\n", market->name);
351 else {
352 printf("CHECK %s UNSTABLE unacceptable=%d blocking=%d\n", market->name, bad, blocks);
353 for (int i = 0; i < bad; ++i)
354 printf("UNACCEPTABLE %s %s\n", market->agents[unacceptable[i].left].name, market->agents[unacceptable[i].right].name);
355 for (int i = 0; i < blocks; ++i)
356 printf("BLOCKING %s %s\n", market->agents[blocking[i].left].name, market->agents[blocking[i].right].name);
357 }
358}

387static void roommates(const Market *market, bool verbose) {
388 Working working = {0};
389 Queue queue = {0};
390 int held[MAX_AGENTS], proposals = 0, failure = -1;
391 for (int a = 0; a < market->count; ++a) {
392 held[a] = -1;
393 enqueue(&queue, a);
394 for (int i = 0; i < market->agents[a].length; ++i)
395 working.present[a][market->agents[a].preference[i]] = true;
396 }
397 while (queue.count) {
398 int x = dequeue(&queue), list[MAX_AGENTS];
399 working_list(market, &working, x, list);
400 int y = list[0];
401 if (held[y] >= 0) enqueue(&queue, held[y]);
402 held[y] = x;
403 ++proposals;
404 int count = working_list(market, &working, y, list);
405 bool after = false;
406 for (int i = 0; i < count; ++i) {
407 int w = list[i];
408 if (after) working.present[y][w] = working.present[w][y] = false;
409 if (w == x) after = true;
410 }
411 failure = first_empty(market, &working);
412 if (failure >= 0) break;
413 }
414 Text history = {0};
415 int rotation_count = 0;
416 if (failure < 0 && verbose) table(market, &working, &history);
417 while (failure < 0) {
418 int list[MAX_AGENTS], start = -1;
419 for (int a = 0; a < market->count; ++a)
420 if (working_list(market, &working, a, list) >= 2) { start = a; break; }
421 if (start < 0) break;
422 int path[MAX_AGENTS], position[MAX_AGENTS], count = 0, p = start;
423 for (int a = 0; a < market->count; ++a) position[a] = -1;
424 while (position[p] < 0) {
425 position[p] = count;
426 path[count++] = p;
427 working_list(market, &working, p, list);
428 int q = list[1];
429 int length = working_list(market, &working, q, list);
430 p = list[length - 1];
431 }
432 int begin = position[p], size = count - begin, front[MAX_AGENTS];
433 append(&history, "ROTATION %d", ++rotation_count);
434 for (int i = 0; i < size; ++i) {
435 int x = path[begin + i];
436 working_list(market, &working, x, list);
437 front[i] = list[0];
438 append(&history, " %s-%s", market->agents[x].name, market->agents[front[i]].name);
439 }
440 append(&history, "\n");
441 Working removed = {0};
442 for (int i = 0; i < size; ++i) {
443 int u = front[(i + 1) % size], length = working_list(market, &working, u, list);
444 bool after = false;
445 for (int j = 0; j < length; ++j) {
446 int w = list[j];
447 if (after) removed.present[u][w] = removed.present[w][u] = true;
448 if (w == path[begin + i]) after = true;
449 }
450 }
451 for (int a = 0; a < market->count; ++a)
452 for (int b = 0; b < market->count; ++b)
453 if (removed.present[a][b]) working.present[a][b] = false;
454 if (verbose) table(market, &working, &history);
455 failure = first_empty(market, &working);
456 }
457 printf("ROOMMATES %s proposals=%d\n", market->name, proposals);
458 emit(&history);
459 if (failure >= 0) printf("NO STABLE MATCHING %s\n", market->agents[failure].name);
460 else {
461 fputs("MATCHING", stdout);
462 for (int a = 0; a < market->count; ++a) {
463 int list[MAX_AGENTS];
464 working_list(market, &working, a, list);
465 if (a < list[0]) printf(" %s-%s", market->agents[a].name, market->agents[list[0]].name);
466 }
467 putchar('\n');
468 }
469}

488static void ttc(const Market *market) {
489 bool remaining[MAX_AGENTS] = {false};
490 int allocation[MAX_AGENTS], left = market->count, cycles = 0;
491 for (int a = 0; a < market->count; ++a) remaining[a] = true;
492 Text history = {0};
493 while (left) {
494 int points[MAX_AGENTS], start = -1;
495 for (int a = 0; a < market->count; ++a) if (remaining[a]) {
496 if (start < 0) start = a;
497 int list[MAX_AGENTS], length = effective(market, a, list);
498 for (int i = 0; i < length; ++i)
499 if (remaining[list[i]]) { points[a] = list[i]; break; }
500 }
501 bool seen[MAX_AGENTS] = {false};
502 int p = start;
503 while (!seen[p]) { seen[p] = true; p = points[p]; }
504 int first = p;
505 append(&history, "CYCLE %d", ++cycles);
506 do {
507 allocation[p] = points[p];
508 append(&history, " %s:%s", market->agents[p].name, market->houses[allocation[p]]);
509 remaining[p] = false;
510 --left;
511 p = points[p];
512 } while (p != first);
513 append(&history, "\n");
514 }
515 printf("TTC %s cycles=%d\n", market->name, cycles);
516 emit(&history);
517 print_allocation(market, allocation);
518}

520static void serial(const Market *market, const int order[]) {
521 bool taken[MAX_AGENTS] = {false}, worse[MAX_AGENTS] = {false};
522 int allocation[MAX_AGENTS], assigned = 0;
523 for (int a = 0; a < market->count; ++a) allocation[a] = -1;
524 for (int i = 0; i < market->count; ++i) {
525 int a = order[i];
526 for (int j = 0; j < market->agents[a].length; ++j) {
527 int house = market->agents[a].preference[j];
528 if (!taken[house]) {
529 allocation[a] = house;
530 taken[house] = true;
531 ++assigned;
532 break;
533 }
534 }
535 int list[MAX_AGENTS], length = effective(market, a, list);
536 worse[a] = true;
537 for (int j = 0; j < length; ++j)
538 if (list[j] == allocation[a]) worse[a] = false;
539 }
540 printf("SERIAL %s assigned=%d\n", market->name, assigned);
541 print_allocation(market, allocation);
542 fputs("WORSE", stdout);
543 bool any = false;
544 for (int a = 0; a < market->count; ++a) if (worse[a]) { printf(" %s", market->agents[a].name); any = true; }
545 puts(any ? "" : " -");
546}

548static void hr(const Market *market) {
549 Matching resident = deferred(market, false, NULL).matching;
550 Matching hospital = deferred(market, true, NULL).matching;
551 printf("HR %s residents=%d hospitals=%d\n", market->name, market->proposers, market->receivers);
552 fputs("RESIDENT-OPTIMAL ", stdout); print_matching(market, resident);
553 fputs("HOSPITAL-OPTIMAL ", stdout); print_matching(market, hospital);
554 fputs("UNMATCHED", stdout);
555 bool any = false, rural = true;
556 for (int p = 0; p < market->proposers; ++p) {
557 if (resident.partner[p] < 0) { printf(" %s", market->agents[p].name); any = true; }
558 if ((resident.partner[p] < 0) != (hospital.partner[p] < 0)) rural = false;
559 }
560 puts(any ? "" : " -");
561 for (int r = market->proposers; r < market->count; ++r) {
562 int a = 0, h = 0;
563 bool same = true;
564 for (int p = 0; p < market->proposers; ++p) {
565 a += resident.partner[p] == r;
566 h += hospital.partner[p] == r;
567 if ((resident.partner[p] == r) != (hospital.partner[p] == r)) same = false;
568 }
569 printf("HOSPITAL %s cap=%d resident=%d hospital=%d same=%s\n", market->agents[r].name,
570 market->agents[r].capacity, a, h, same ? "YES" : "NO");
571 if (a != h || (a < market->agents[r].capacity && !same)) rural = false;
572 }
573 printf("RURAL %s\n", rural ? "HOLDS" : "FAILS");
574}

576static void manipulate(const Market *market, int a) {
577 int baseline_b = boston(market).matching.partner[a];
578 int baseline_g = deferred(market, false, NULL).matching.partner[a];
579 const Agent *agent = &market->agents[a];
580 printf("MANIPULATE %s %s boston=%s gs=%s swaps=%d\n", market->name, agent->name,
581 partner_name(market, baseline_b), partner_name(market, baseline_g), agent->length ? agent->length - 1 : 0);
582 bool improved = false;
583 for (int i = 0; i + 1 < agent->length; ++i) {
584 Market trial = *market;
585 trial.agents[a].preference[i] = agent->preference[i + 1];
586 trial.agents[a].preference[i + 1] = agent->preference[i];
587 int b = boston(&trial).matching.partner[a];
588 int g = deferred(&trial, false, NULL).matching.partner[a];
589 int result[] = {b, g}, baseline[] = {baseline_b, baseline_g};
590 const char *labels[] = {"BOSTON", "GS"};
591 for (int k = 0; k < 2; ++k) if (rank_of(market, a, result[k]) < rank_of(market, a, baseline[k])) {
592 printf("IMPROVE %s %d %s %s result=%s\n", labels[k], i + 1,
593 market->agents[agent->preference[i]].name, market->agents[agent->preference[i + 1]].name,
594 partner_name(market, result[k]));
595 improved = true;
596 }
597 }
598 if (!improved) puts("NO IMPROVEMENT");
599}

601static int gcd(int a, int b) {
602 while (b) { int remainder = a % b; a = b; b = remainder; }
603 return a;
604}

606static void stats(const Market *market) {
607 int pairs = 0, sum[2] = {0}, first[2] = {0}, receivers[MAX_AGENTS] = {0};
608 for (int p = 0; p < market->proposers; ++p) {
609 int r = market->current.partner[p];
610 if (r < 0) continue;
611 ++pairs;
612 ++receivers[r];
613 int ranks[] = {rank_of(market, p, r), rank_of(market, r, p)};
614 for (int i = 0; i < 2; ++i) { sum[i] += ranks[i]; first[i] += ranks[i] == 1; }
615 }
616 int unmatched[] = {market->proposers - pairs, 0};
617 for (int r = market->proposers; r < market->count; ++r) unmatched[1] += !receivers[r];
618 printf("STATS %s source=%s pairs=%d\n", market->name, sources[market->source], pairs);
619 const char *labels[] = {"PROPOSERS", "RECEIVERS"};
620 for (int i = 0; i < 2; ++i) {
621 printf("%s sum=%d average=", labels[i], sum[i]);
622 if (pairs) {
623 int divisor = gcd(sum[i], pairs), hundredths = 100 * sum[i] / pairs;
624 printf("%d/%d truncated=%d.%02d", sum[i] / divisor, pairs / divisor, hundredths / 100, hundredths % 100);
625 } else fputs("- truncated=-", stdout);
626 printf(" first=%d unmatched=%d\n", first[i], unmatched[i]);
627 }
628}

630static void create_market(char *words[], int count) {
631 const char *command = words[0], *name = words[1];
632 if (find_market(name)) { printf("ERROR DUPLICATE MARKET %s\n", name); return; }
633 Market market = {0};
634 strcpy(market.name, name);
635 market.type = !strcmp(command, "ROOMS") ? ROOMMATE : !strcmp(command, "HOUSES") ? HOUSING : TWO_SIDED;
636 if (!strcmp(command, "RANDOM")) {
637 int n = atoi(words[2]);
638 market.proposers = market.receivers = n;
639 market.count = 2 * n;
640 int64_t state = strtoll(words[3], NULL, 10);
641 for (int a = 0; a < market.count; ++a) {
642 Agent *agent = &market.agents[a];
643 snprintf(agent->name, sizeof agent->name, "%c%d", a < n ? 'P' : 'R', a % n + 1);
644 agent->length = n;
645 agent->capacity = 1;
646 for (int i = 0; i < n; ++i) agent->preference[i] = (a < n ? n : 0) + i;
647 for (int i = n; i > 1; --i) {
648 state = 48271 * state % 2147483647;
649 int j = (int)(state % i), temporary = agent->preference[i - 1];
650 agent->preference[i - 1] = agent->preference[j];
651 agent->preference[j] = temporary;
652 }
653 }
654 printf("RANDOM %s n=%d seed=%lld\n", name, n, (long long)strtoll(words[3], NULL, 10));
655 } else {
656 for (int i = 2; i < count; ++i) {
657 if (!strcmp(words[i], "|")) { market.proposers = market.count; continue; }
658 char *house = NULL;
659 if (market.type == HOUSING) { house = strchr(words[i], ':'); *house++ = '\0'; }
660 if (index_of(&market, words[i]) >= 0) { printf("ERROR DUPLICATE AGENT %s\n", words[i]); return; }
661 if (house && house_index(&market, house) >= 0) { printf("ERROR DUPLICATE HOUSE %s\n", house); return; }
662 strcpy(market.agents[market.count].name, words[i]);
663 market.agents[market.count].capacity = 1;
664 if (house) strcpy(market.houses[market.count], house);
665 ++market.count;
666 }
667 if (market.type == TWO_SIDED) {
668 market.receivers = market.count - market.proposers;
669 printf("MARKET %s proposers=%d receivers=%d\n", name, market.proposers, market.receivers);
670 } else printf("%s %s agents=%d\n", command, name, market.count);
671 }
672 Market *saved = allocate(sizeof *saved);
673 *saved = market;
674 markets[market_count++] = saved;
675}

677static void show(const Market *market) {
678 if (market->type == TWO_SIDED)
679 printf("SHOW %s MARKET proposers=%d receivers=%d\n", market->name, market->proposers, market->receivers);
680 else printf("SHOW %s %s agents=%d\n", market->name, market->type == ROOMMATE ? "ROOMS" : "HOUSES", market->count);
681 for (int a = 0; a < market->count; ++a) {
682 const Agent *agent = &market->agents[a];
683 if (market->type == TWO_SIDED) {
684 printf("%s %s", a < market->proposers ? "PROPOSER" : "RECEIVER", agent->name);
685 if (a >= market->proposers) printf(" cap=%d", agent->capacity);
686 } else {
687 printf("AGENT %s", agent->name);
688 if (market->type == HOUSING) printf(" house=%s", market->houses[a]);
689 }
690 if (!agent->length) fputs(" -", stdout);
691 for (int i = 0; i < agent->length; ++i) {
692 int b = agent->preference[i];
693 printf(" %s", market->type == HOUSING ? market->houses[b] : market->agents[b].name);
694 }
695 putchar('\n');
696 }
697 if (market->type == TWO_SIDED) printf("CURRENT %s\n", sources[market->source]);
698}

700static void command(char *words[], int count) {
701 const char *op = words[0];
702 if (!strcmp(op, "MARKET") || !strcmp(op, "ROOMS") || !strcmp(op, "HOUSES") || !strcmp(op, "RANDOM")) {
703 create_market(words, count);
704 return;
705 }
706 Market *market = find_market(words[1]);
707 if (!market) { printf("ERROR UNKNOWN MARKET %s\n", words[1]); return; }
708 if (!strcmp(op, "SHOW")) show(market);
709 else if (!strcmp(op, "PREF")) {
710 int a = require_agent(market, words[2], -1);
711 if (a < 0) return;
712 int preference[MAX_AGENTS], length = 0;
713 bool used[MAX_AGENTS] = {false};
714 for (int i = 3; i < count; ++i) {
715 int b = market->type == HOUSING ? house_index(market, words[i]) : index_of(market, words[i]);
716 bool possible = b >= 0;
717 if (market->type == TWO_SIDED) possible = possible && ((a < market->proposers) != (b < market->proposers));
718 if (market->type == ROOMMATE) possible = possible && b != a;
719 if (!possible) { printf("ERROR BAD CHOICE %s\n", words[i]); return; }
720 if (used[b]) { printf("ERROR DUPLICATE CHOICE %s\n", words[i]); return; }
721 used[b] = true;
722 preference[length++] = b;
723 }
724 memcpy(market->agents[a].preference, preference, (size_t)length * sizeof(int));
725 market->agents[a].length = length;
726 market->source = NONE;
727 printf("PREF %s %s length=%d\n", market->name, market->agents[a].name, length);
728 } else if (!strcmp(op, "CHECK")) {
729 if (market->type == HOUSING) { printf("ERROR WRONG TYPE %s\n", market->name); return; }
730 int partner[MAX_AGENTS], counts[MAX_AGENTS] = {0};
731 for (int a = 0; a < market->count; ++a) partner[a] = -1;
732 for (int i = 2; i < count; ++i) {
733 char *right = strchr(words[i], '-');
734 *right++ = '\0';
735 bool two = market->type == TWO_SIDED;
736 int x = require_agent(market, words[i], two ? 0 : -1);
737 if (x < 0) return;
738 int y = require_agent(market, right, two ? 1 : -1);
739 if (y < 0) return;
740 if (partner[x] >= 0) { printf("ERROR DUPLICATE AGENT %s\n", words[i]); return; }
741 if (two) {
742 if (++counts[y] > market->agents[y].capacity) { printf("ERROR OVER CAPACITY %s\n", right); return; }
743 } else {
744 if (y == x || partner[y] >= 0) { printf("ERROR DUPLICATE AGENT %s\n", right); return; }
745 partner[y] = x;
746 }
747 partner[x] = y;
748 }
749 check_matching(market, partner);
750 } else if (!strcmp(op, "ROOMMATES")) {
751 if (!require_type(market, ROOMMATE)) return;
752 for (int a = 0; a < market->count; ++a)
753 if (market->agents[a].length < market->count - 1) { printf("ERROR INCOMPLETE %s\n", market->agents[a].name); return; }
754 roommates(market, count == 3);
755 } else if (!strcmp(op, "TTC") || !strcmp(op, "SERIAL")) {
756 if (!require_type(market, HOUSING)) return;
757 if (!strcmp(op, "TTC")) ttc(market);
758 else {
759 int order[MAX_AGENTS], length = 0;
760 bool used[MAX_AGENTS] = {false};
761 for (int i = 2; i < count; ++i) {
762 int a = require_agent(market, words[i], -1);
763 if (a < 0) return;
764 if (used[a]) { printf("ERROR DUPLICATE AGENT %s\n", words[i]); return; }
765 used[a] = true;
766 order[length++] = a;
767 }
768 if (length != market->count) { puts("ERROR BAD ORDER"); return; }
769 serial(market, order);
770 }
771 } else {
772 if (!require_type(market, TWO_SIDED)) return;
773 if (!strcmp(op, "CAPACITY")) {
774 int r = require_agent(market, words[2], 1);
775 if (r < 0) return;
776 market->agents[r].capacity = atoi(words[3]);
777 market->source = NONE;
778 printf("CAPACITY %s %s %d\n", market->name, market->agents[r].name, market->agents[r].capacity);
779 } else if (!strcmp(op, "GS") || !strcmp(op, "GSR")) {
780 Text history = {0};
781 bool reverse = !strcmp(op, "GSR");
782 Deferred result = deferred(market, reverse, count == 3 ? &history : NULL);
783 market->current = result.matching;
784 market->source = reverse ? GSR : GS;
785 printf("%s %s proposals=%d\n", op, market->name, result.proposals);
786 emit(&history);
787 fputs("MATCHING ", stdout); print_matching(market, result.matching);
788 } else if (!strcmp(op, "BOSTON")) {
789 Boston result = boston(market);
790 market->current = result.matching;
791 market->source = BOSTON;
792 printf("BOSTON %s applications=%d\n", market->name, result.applications);
793 fputs("MATCHING ", stdout); print_matching(market, result.matching);
794 } else if (!strcmp(op, "HR")) hr(market);
795 else if (!strcmp(op, "MANIPULATE")) {
796 int a = require_agent(market, words[2], 0);
797 if (a >= 0) manipulate(market, a);
798 } else if (!strcmp(op, "STATS")) {
799 if (market->source == NONE) puts("ERROR NO MATCHING");
800 else stats(market);
801 } else {
802 for (int r = market->proposers; r < market->count; ++r)
803 if (market->agents[r].capacity != 1) { puts("ERROR NOT ONE TO ONE"); return; }
804 Lattice *lattice = allocate(sizeof *lattice);
805 if (!enumerate(market, lattice)) { free(lattice); puts("ERROR TOO MANY MATCHINGS"); return; }
806 if (!strcmp(op, "LATTICE")) {
807 printf("LATTICE %s matchings=%d rotations=%d\n", market->name, lattice->count, lattice->rotation_count);
808 for (int i = 0; i < lattice->count; ++i) {
809 Node node = lattice->nodes[i];
810 int cost, regret;
811 metrics(market, node.matching, &cost, &regret);
812 printf("MATCHING %d from=", i + 1);
813 if (i) printf("%d via=%d", node.origin, node.via);
814 else fputs("- via=-", stdout);
815 printf(" cost=%d regret=%d ", cost, regret);
816 print_matching(market, node.matching);
817 }
818 for (int i = 0; i < lattice->rotation_count; ++i) {
819 const Rotation *rotation = &lattice->rotations[i];
820 printf("ROTATION %d", i + 1);
821 for (int j = 0; j < rotation->count; ++j)
822 printf(" %s-%s", market->agents[rotation->proposer[j]].name, market->agents[rotation->receiver[j]].name);
823 putchar('\n');
824 }
825 } else {
826 bool egalitarian = !strcmp(op, "EGALITARIAN");
827 int best = 0, value = 0;
828 for (int i = 0; i < lattice->count; ++i) {
829 int cost, regret;
830 metrics(market, lattice->nodes[i].matching, &cost, &regret);
831 int candidate = egalitarian ? cost : regret;
832 if (!i || candidate < value) { best = i; value = candidate; }
833 }
834 market->current = lattice->nodes[best].matching;
835 market->source = egalitarian ? EGALITARIAN : MINREGRET;
836 printf("%s %s matching=%d %s=%d\n", op, market->name, best + 1, egalitarian ? "cost" : "regret", value);
837 fputs("MATCHING ", stdout); print_matching(market, market->current);
838 }
839 free(lattice);
840 }
841 }
842}
