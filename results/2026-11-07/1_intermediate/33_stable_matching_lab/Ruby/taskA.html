98 def deferred(reverse: false, verbose: false)
99 n = @agents.length
100 next_position = Array.new(n, 0)
101 counts = Array.new(n, 0)
102 holding = Array.new(n) { [] }
103 queued = Array.new(n, false)
104 queue = []
105 able = ->(a) { counts[a] < @agents[a].capacity && next_position[a] < @agents[a].preference.length }
106 enqueue = lambda do |a|
107 if able.call(a) && !queued[a]
108 queue << a
109 queued[a] = true
110 end
111 end
112 side = reverse ? (@proposers...n) : (0...@proposers)
113 side.each { |a| enqueue.call(a) }
114 history = []
115 head = 0
116 while head < queue.length
117 x = queue[head]
118 head += 1
119 queued[x] = false
120 y = @agents[x].preference[next_position[x]]
121 next_position[x] += 1
122 action = 'REJECT'
123 held = holding[y]
124 if accepts?(y, x)
125 if held.length < @agents[y].capacity
126 held << x
127 counts[x] += 1
128 action = 'HOLD'
129 else
130 worst = held.each_index.max_by { |i| rank(y, held[i]) }
131 rejected = held[worst]
132 if rank(y, x) < rank(y, rejected)
133 held[worst] = x
134 counts[rejected] -= 1
135 counts[x] += 1
136 action = "HOLD REJECT #{@agents[rejected].name}"
137 enqueue.call(rejected)
138 end
139 end
140 end
141 history << "PROPOSE #{@agents[x].name} #{@agents[y].name} #{action}" if verbose
142 enqueue.call(x)
143 end
144 partners = Array.new(@proposers)
145 holding.each_with_index do |held, y|
146 held.each { |x| partners[reverse ? y : x] = reverse ? x : y }
147 end
148 Run.new(partners.freeze, head, history.freeze)
149 end

151 def boston
152 partners = Array.new(@proposers)
153 remaining = @agents.map(&:capacity)
154 rounds = @agents.take(@proposers).map { |a| a.preference.length }.max
155 applications = 0
156 rounds.times do |round|
157 applicants = Array.new(@proposers)
158 @proposers.times do |p|
159 next unless partners[p].nil? && round < @agents[p].preference.length
160 applicants[p] = @agents[p].preference[round]
161 applications += 1
162 end
163 (@proposers...@agents.length).each do |r|
164 @agents[r].preference.each do |p|
165 next unless remaining[r].positive? && applicants[p] == r
166 partners[p] = r
167 remaining[r] -= 1
168 end
169 end
170 end
171 Run.new(partners.freeze, applications, [].freeze)
172 end

174 def rotations(partners)
175 owners = Array.new(@agents.length)
176 partners.each_with_index { |r, p| owners[r] = p unless r.nil? }
177 successors = Array.new(@proposers)
178 @proposers.times do |p|
179 next if partners[p].nil?
180 @agents[p].preference.drop(rank(p, partners[p])).each do |r|
181 next unless accepts?(r, p) && (owners[r].nil? || rank(r, p) < rank(r, owners[r]))
182 successors[p] = owners[r]
183 break
184 end
185 end
186 result = []
187 covered = Array.new(@proposers, false)
188 @proposers.times do |start|
189 next if covered[start]
190 seen = Array.new(@proposers, false)
191 path = []
192 p = start
193 until p.nil? || seen[p]
194 seen[p] = true
195 path << p
196 p = successors[p]
197 end
198 next unless p == start
199 path.each { |a| covered[a] = true }
200 result << Rotation.new(path.map { |a| [a, partners[a]].freeze }.freeze)
201 end
202 result
203 end

205 def candidate(partners, origin, via)
206 cost = regret = 0
207 partners.each_with_index do |r, p|
208 next if r.nil?
209 a, b = rank(p, r), rank(r, p)
210 cost += a + b
211 regret = [regret, a, b].max
212 end
213 Candidate.new(partners.freeze, origin, via, cost, regret)
214 end

216 def enumerate
217 raise CommandError, 'NOT ONE TO ONE' if @agents.drop(@proposers).any? { |a| a.capacity != 1 }
218 matchings = [candidate(deferred.partners, nil, nil)]
219 numbered = []
220 index = 0
221 while index < matchings.length
222 partners = matchings[index].partners
223 rotations(partners).each do |rotation|
224 number = numbered.index(rotation)
225 if number.nil?
226 number = numbered.length
227 numbered << rotation
228 end
229 next_matching = partners.dup
230 rotation.pairs.each_with_index do |(p, _), i|
231 next_matching[p] = rotation.pairs[(i + 1) % rotation.pairs.length][1]
232 end
233 next if matchings.any? { |old| old.partners == next_matching }
234 raise CommandError, 'TOO MANY MATCHINGS' if matchings.length == 50
235 matchings << candidate(next_matching, index + 1, number + 1)
236 end
237 index += 1
238 end
239 Lattice.new(matchings.freeze, numbered.freeze)
240 end

242 def check(words)
243 raise CommandError, "WRONG TYPE #{@name}" if @kind == :housing
244 partners = Array.new(@agents.length)
245 counts = Array.new(@agents.length, 0)
246 words.drop(2).each do |word|
247 left, right = word.split('-')
248 two = @kind == :two_sided
249 x = agent(left, two ? :proposer : :any)
250 y = agent(right, two ? :receiver : :any)
251 raise CommandError, "DUPLICATE AGENT #{left}" unless partners[x].nil?
252 if two
253 counts[y] += 1
254 raise CommandError, "OVER CAPACITY #{right}" if counts[y] > @agents[y].capacity
255 else
256 raise CommandError, "DUPLICATE AGENT #{right}" if x == y || !partners[y].nil?
257 partners[y] = x
258 end
259 partners[x] = y
260 end
261 unacceptable, blocking = [], []
262 if @kind == :two_sided
263 @proposers.times do |p|
264 r = partners[p]
265 unacceptable << [p, r] if !r.nil? && (!accepts?(p, r) || !accepts?(r, p))
266 (@proposers...@agents.length).each do |receiver|
267 next if partners[p] == receiver || !accepts?(p, receiver) || !accepts?(receiver, p) || rank(p, receiver) >= rank(p, partners[p])
268 worst = @proposers.times.filter_map { |a| rank(receiver, a) if partners[a] == receiver }.max || 0
269 blocking << [p, receiver] if counts[receiver] < @agents[receiver].capacity || rank(receiver, p) < worst
270 end
271 end
272 else
273 @agents.each_index do |a|
274 (a + 1...@agents.length).each do |b|
275 mutual = accepts?(a, b) && accepts?(b, a)
276 if partners[a] == b
277 unacceptable << [a, b] unless mutual
278 elsif mutual && rank(a, b) < rank(a, partners[a]) && rank(b, a) < rank(b, partners[b])
279 blocking << [a, b]
280 end
281 end
282 end
283 end
284 return ["CHECK #{@name} STABLE"] if unacceptable.empty? && blocking.empty?
285 output = ["CHECK #{@name} UNSTABLE unacceptable=#{unacceptable.length} blocking=#{blocking.length}"]
286 unacceptable.each { |a, b| output << "UNACCEPTABLE #{@agents[a].name} #{@agents[b].name}" }
287 blocking.each { |a, b| output << "BLOCKING #{@agents[a].name} #{@agents[b].name}" }
288 output
289 end

300 def roommates(verbose)
301 require_kind(:roommate)
302 n = @agents.length
303 @agents.each do |agent|
304 raise CommandError, "INCOMPLETE #{agent.name}" if agent.preference.length < n - 1
305 end
306 lists = @agents.map { |a| a.preference.dup }
307 held = Array.new(n)
308 queue = (0...n).to_a
309 proposals = 0
310 failure = nil
311 while proposals < queue.length
312 x = queue[proposals]
313 proposals += 1
314 y = lists[x].first
315 queue << held[y] unless held[y].nil?
316 held[y] = x
317 removed = lists[y].drop(lists[y].index(x) + 1)
318 removed.each { |w| remove_pair(lists, y, w) }
319 failure = lists.index(&:empty?)
320 break unless failure.nil?
321 end
322 history = []
323 rotations = 0
324 history.concat(table(lists)) if failure.nil? && verbose
325 while failure.nil?
326 p = lists.index { |list| list.length >= 2 }
327 break if p.nil?
328 positions = Array.new(n)
329 path = []
330 while positions[p].nil?
331 positions[p] = path.length
332 path << p
333 q = lists[p][1]
334 p = lists[q].last
335 end
336 cycle = path.drop(positions[p])
337 front = cycle.map { |x| lists[x].first }
338 rotations += 1
339 notation = cycle.each_index.map { |i| "#{@agents[cycle[i]].name}-#{@agents[front[i]].name}" }.join(' ')
340 history << "ROTATION #{rotations} #{notation}"
341 removals = []
342 cycle.each_with_index do |x, i|
343 u = front[(i + 1) % front.length]
344 lists[u].drop(lists[u].index(x) + 1).each { |w| removals << [u, w] }
345 end
346 removals.each { |a, b| remove_pair(lists, a, b) }
347 history.concat(table(lists)) if verbose
348 failure = lists.index(&:empty?)
349 end
350 output = ["ROOMMATES #{@name} proposals=#{proposals}", *history]
351 if failure.nil?
352 pairs = @agents.each_index.filter_map do |a|
353 b = lists[a].first
354 "#{@agents[a].name}-#{@agents[b].name}" if a < b
355 end
356 output << "MATCHING #{pairs.join(' ')}"
357 else
358 output << "NO STABLE MATCHING #{@agents[failure].name}"
359 end
360 output
361 end

363 def ttc
364 n = @agents.length
365 remaining = Array.new(n, true)
366 effective = @agents.each_index.map { |a| self.effective(a) }
367 allocation = Array.new(n)
368 history = []
369 while (start = remaining.index(true))
370 points = Array.new(n)
371 n.times { |a| points[a] = effective[a].find { |house| remaining[house] } if remaining[a] }
372 seen = Array.new(n, false)
373 p = start
374 until seen[p]
375 seen[p] = true
376 p = points[p]
377 end
378 first = p
379 cycle = []
380 loop do
381 allocation[p] = points[p]
382 remaining[p] = false
383 cycle << "#{@agents[p].name}:#{@houses[points[p]]}"
384 p = points[p]
385 break if p == first
386 end
387 history << "CYCLE #{history.length + 1} #{cycle.join(' ')}"
388 end
389 ["TTC #{@name} cycles=#{history.length}", *history, "ALLOCATION #{allocation_text(allocation)}"]
390 end

392 def serial(words)
393 order = []
394 words.drop(2).each do |name|
395 a = agent(name)
396 raise CommandError, "DUPLICATE AGENT #{name}" if order.include?(a)
397 order << a
398 end
399 raise CommandError, 'BAD ORDER' unless order.length == @agents.length
400 taken = Array.new(@agents.length, false)
401 allocation = Array.new(@agents.length)
402 order.each do |a|
403 house = @agents[a].preference.find { |h| !taken[h] }
404 next if house.nil?
405 allocation[a] = house
406 taken[house] = true
407 end
408 worse = @agents.each_index.filter_map do |a|
409 @agents[a].name if allocation[a].nil? || !effective(a).include?(allocation[a])
410 end
411 ["SERIAL #{@name} assigned=#{allocation.count { |house| !house.nil? }}", "ALLOCATION #{allocation_text(allocation)}", "WORSE #{list_text(worse)}"]
412 end

414 def hr
415 a = deferred.partners
416 h = deferred(reverse: true).partners
417 unmatched = @proposers.times.filter_map { |p| @agents[p].name if a[p].nil? }
418 rural = @proposers.times.all? { |p| a[p].nil? == h[p].nil? }
419 output = ["HR #{@name} residents=#{@proposers} hospitals=#{@agents.length - @proposers}",
420 "RESIDENT-OPTIMAL #{pairs(a)}", "HOSPITAL-OPTIMAL #{pairs(h)}", "UNMATCHED #{list_text(unmatched)}"]
421 (@proposers...@agents.length).each do |r|
422 first = @proposers.times.select { |p| a[p] == r }
423 second = @proposers.times.select { |p| h[p] == r }
424 same = first == second
425 capacity = @agents[r].capacity
426 output << "HOSPITAL #{@agents[r].name} cap=#{capacity} resident=#{first.length} hospital=#{second.length} same=#{same ? 'YES' : 'NO'}"
427 rural &&= first.length == second.length && (first.length == capacity || same)
428 end
429 output << (rural ? 'RURAL HOLDS' : 'RURAL FAILS')
430 end

432 def manipulate(a)
433 truth = @agents[a].preference
434 b0 = boston.partners[a]
435 g0 = deferred.partners[a]
436 output = ["MANIPULATE #{@name} #{@agents[a].name} boston=#{partner_name(b0)} gs=#{partner_name(g0)} swaps=#{[0, truth.length - 1].max}"]
437 (truth.length - 1).times do |i|
438 trial = dup
439 swapped = truth.dup
440 swapped[i], swapped[i + 1] = truth[i + 1], truth[i]
441 trial.agents[a].preference = swapped.freeze
442 b = trial.boston.partners[a]
443 g = trial.deferred.partners[a]
444 swap = "#{i + 1} #{@agents[truth[i]].name} #{@agents[truth[i + 1]].name}"
445 output << "IMPROVE BOSTON #{swap} result=#{partner_name(b)}" if rank(a, b) < rank(a, b0)
446 output << "IMPROVE GS #{swap} result=#{partner_name(g)}" if rank(a, g) < rank(a, g0)
447 end
448 output << 'NO IMPROVEMENT' if output.length == 1
449 output
450 end

452 def stats
453 raise CommandError, 'NO MATCHING' if @current.nil?
454 sums = [0, 0]
455 first = [0, 0]
456 occupied = Array.new(@agents.length, false)
457 pairs = 0
458 @current.partners.each_with_index do |r, p|
459 next if r.nil?
460 pairs += 1
461 occupied[r] = true
462 [rank(p, r), rank(r, p)].each_with_index do |value, i|
463 sums[i] += value
464 first[i] += 1 if value == 1
465 end
466 end
467 unmatched = [@proposers - pairs, occupied.drop(@proposers).count(false)]
468 output = ["STATS #{@name} source=#{@current.source} pairs=#{pairs}"]
469 %w[PROPOSERS RECEIVERS].each_with_index do |label, i|
470 average = truncated = '-'
471 if pairs.positive?
472 divisor = sums[i].gcd(pairs)
473 cents = 100 * sums[i] / pairs
474 average = "#{sums[i] / divisor}/#{pairs / divisor}"
475 truncated = format('%d.%02d', cents / 100, cents % 100)
476 end
477 output << "#{label} sum=#{sums[i]} average=#{average} truncated=#{truncated} first=#{first[i]} unmatched=#{unmatched[i]}"
478 end
479 output
480 end

485 when :roommate then "SHOW #{@name} ROOMS agents=#{@agents.length}"
486 when :housing then "SHOW #{@name} HOUSES agents=#{@agents.length}"

493 when :roommate then "AGENT #{agent.name}"

509 def create(words)
510 op, name = words.take(2)
511 raise CommandError, "DUPLICATE MARKET #{name}" if @markets.key?(name)
512 kind = { 'ROOMS' => :roommate, 'HOUSES' => :housing }.fetch(op, :two_sided)
513 if op == 'RANDOM'
514 n, seed = words.drop(2).map { |value| Integer(value, 10) }
515 state = seed
516 names = (0...2 * n).map { |a| "#{a < n ? 'P' : 'R'}#{a % n + 1}" }
517 market = Market.new(name, kind, names, n)
518 market.agents.each_with_index do |agent, a|
519 preference = n.times.map { |i| (a < n ? n : 0) + i }
520 n.downto(2) do |i|
521 state = 48271 * state % 2147483647
522 j = state % i
523 preference[i - 1], preference[j] = preference[j], preference[i - 1]
524 end
525 agent.preference = preference.freeze
526 end
527 @markets[name] = market
528 return ["RANDOM #{name} n=#{n} seed=#{seed}"]
529 end
530 names, houses = [], []
531 proposers = 0
532 words.drop(2).each do |word|
533 if word == '|'
534 proposers = names.length
535 next
536 end
537 agent, house = kind == :housing ? word.split(':') : [word]
538 raise CommandError, "DUPLICATE AGENT #{agent}" if names.include?(agent)
539 if kind == :housing
540 raise CommandError, "DUPLICATE HOUSE #{house}" if houses.include?(house)
541 houses << house
542 end
543 names << agent
544 end
545 @markets[name] = Market.new(name, kind, names, proposers, houses)
546 [kind == :two_sided ? "MARKET #{name} proposers=#{proposers} receivers=#{names.length - proposers}" : "#{op} #{name} agents=#{names.length}"]
547 end

549 def execute(line)
550 words = line.split(' ')
551 op, name = words.take(2)
552 return create(words) if %w[MARKET ROOMS HOUSES RANDOM].include?(op)
553 market = @markets[name]
554 raise CommandError, "UNKNOWN MARKET #{name}" if market.nil?
555 case op
556 when 'SHOW' then return market.show
557 when 'PREF'
558 a = market.agent(words[2])
559 preference = []
560 words.drop(3).each do |word|
561 b = market.kind == :housing ? market.houses.index(word) : market.find_agent(word)
562 if b.nil? || (market.kind == :two_sided && (a < market.proposers) == (b < market.proposers)) || (market.kind == :roommate && a == b)
563 raise CommandError, "BAD CHOICE #{word}"
564 end
565 raise CommandError, "DUPLICATE CHOICE #{word}" if preference.include?(b)
566 preference << b
567 end
568 market.agents[a].preference = preference.freeze
569 market.current = nil if market.kind == :two_sided
570 return ["PREF #{name} #{words[2]} length=#{preference.length}"]
571 when 'CHECK' then return market.check(words)
572 when 'ROOMMATES' then return market.roommates(words.length == 3)
573 when 'TTC', 'SERIAL'
574 market.require_kind(:housing)
575 return op == 'TTC' ? market.ttc : market.serial(words)
576 end
577 market.require_kind(:two_sided)
578 case op
579 when 'CAPACITY'
580 r = market.agent(words[2], :receiver)
581 capacity = Integer(words[3], 10)
582 market.agents[r].capacity = capacity
583 market.current = nil
584 ["CAPACITY #{name} #{words[2]} #{capacity}"]
585 when 'GS', 'GSR', 'BOSTON'
586 result = op == 'BOSTON' ? market.boston : market.deferred(reverse: op == 'GSR', verbose: words.length == 3)
587 market.save(op, result.partners)
588 ["#{op} #{name} #{op == 'BOSTON' ? 'applications' : 'proposals'}=#{result.count}", *result.history, "MATCHING #{market.pairs(result.partners)}"]
589 when 'HR' then market.hr
590 when 'MANIPULATE' then market.manipulate(market.agent(words[2], :proposer))
591 when 'STATS' then market.stats
592 when 'LATTICE', 'EGALITARIAN', 'MINREGRET'
593 lattice = market.enumerate
594 if op == 'LATTICE'
595 output = ["LATTICE #{name} matchings=#{lattice.matchings.length} rotations=#{lattice.rotations.length}"]
596 lattice.matchings.each_with_index do |candidate, i|
597 output << "MATCHING #{i + 1} from=#{candidate.origin || '-'} via=#{candidate.via || '-'} cost=#{candidate.cost} regret=#{candidate.regret} #{market.pairs(candidate.partners)}"
598 end
599 lattice.rotations.each_with_index do |rotation, i|
600 notation = rotation.pairs.map { |p, r| "#{market.agents[p].name}-#{market.agents[r].name}" }.join(' ')
601 output << "ROTATION #{i + 1} #{notation}"
602 end
603 output
604 else
605 egalitarian = op == 'EGALITARIAN'
606 best = lattice.matchings.each_index.min_by do |i|
607 candidate = lattice.matchings[i]
608 [egalitarian ? candidate.cost : candidate.regret, i]
609 end
610 selected = lattice.matchings[best]
611 market.save(op, selected.partners)
612 metric = egalitarian ? "cost=#{selected.cost}" : "regret=#{selected.regret}"
613 ["#{op} #{name} matching=#{best + 1} #{metric}", "MATCHING #{market.pairs(selected.partners)}"]
614 end
615 else
616 raise 'Unknown command in well-formed input'
617 end
618 end
619end
