33 fun agent(name: String, side: Side = Side.ANY): Int {

36 if (side == Side.PROPOSER && index >= proposers) fail("NOT PROPOSER $name")

64 fun deferred(reverse: Boolean = false, verbose: Boolean = false): Run {
65 val n = agents.size
66 val next = IntArray(n)
67 val counts = IntArray(n)
68 val holding = Array(n) { mutableListOf<Int>() }
69 val queue = ArrayDeque<Int>()
70 fun able(a: Int) = counts[a] < agents[a].capacity && next[a] < agents[a].preference.size
71 for (a in if (reverse) proposers until n else 0 until proposers) if (able(a)) queue.addLast(a)
72 var proposals = 0
73 val history = mutableListOf<String>()
74 while (queue.isNotEmpty()) {
75 val x = queue.removeFirst()
76 val y = agents[x].preference[next[x]++]
77 proposals++
78 var action = "REJECT"
79 val held = holding[y]
80 if (accepts(y, x)) {
81 if (held.size < agents[y].capacity) {
82 held.add(x)
83 counts[x]++
84 action = "HOLD"
85 } else {
86 val worst = held.indices.maxBy { rank(y, held[it]) }
87 val rejected = held[worst]
88 if (rank(y, x) < rank(y, rejected)) {
89 held[worst] = x
90 counts[rejected]--
91 counts[x]++
92 action = "HOLD REJECT ${agents[rejected].name}"
93 if (able(rejected) && rejected !in queue) queue.addLast(rejected)
94 }
95 }
96 }
97 if (verbose) history.add("PROPOSE ${agents[x].name} ${agents[y].name} $action")
98 if (able(x)) queue.addLast(x)
99 }
100 val partners = emptyMatching(proposers)
101 for (y in agents.indices) for (x in holding[y]) {
102 if (reverse) partners[y] = x else partners[x] = y
103 }
104 return Run(partners, proposals, history)
105 }

107 fun boston(): Run {
108 val partners = emptyMatching(proposers)
109 val remaining = IntArray(agents.size) { agents[it].capacity }
110 val rounds = (0 until proposers).maxOf { agents[it].preference.size }
111 var applications = 0
112 for (round in 0 until rounds) {
113 val applicants = emptyMatching(proposers)
114 for (p in 0 until proposers) if (partners[p] < 0 && round < agents[p].preference.size) {
115 applicants[p] = agents[p].preference[round]
116 applications++
117 }
118 for (r in proposers until agents.size) for (p in agents[r].preference) {
119 if (remaining[r] > 0 && applicants[p] == r) {
120 partners[p] = r
121 remaining[r]--
122 }
123 }
124 }
125 return Run(partners, applications)
126 }

128 fun rotations(partners: IntArray): List<Rotation> {
129 val owners = emptyMatching(agents.size)
130 val successors = emptyMatching(proposers)
131 for (p in 0 until proposers) if (partners[p] >= 0) owners[partners[p]] = p
132 for (p in 0 until proposers) {
133 if (partners[p] < 0) continue
134 for (r in agents[p].preference.drop(rank(p, partners[p]))) {
135 if (accepts(r, p) && (owners[r] < 0 || rank(r, p) < rank(r, owners[r]))) {
136 successors[p] = owners[r]
137 break
138 }
139 }
140 }
141 val result = mutableListOf<Rotation>()
142 val covered = BooleanArray(proposers)
143 for (start in 0 until proposers) {
144 if (covered[start]) continue
145 val seen = BooleanArray(proposers)
146 val path = mutableListOf<Int>()
147 var p = start
148 while (p >= 0 && !seen[p]) {
149 seen[p] = true
150 path.add(p)
151 p = successors[p]
152 }
153 if (p == start) {
154 for (a in path) covered[a] = true
155 result.add(Rotation(path.map { it to partners[it] }))
156 }
157 }
158 return result
159 }

161 fun candidate(partners: IntArray, origin: Int, via: Int): Candidate {
162 var cost = 0
163 var regret = 0
164 for (p in 0 until proposers) if (partners[p] >= 0) {
165 val r = partners[p]
166 val a = rank(p, r)
167 val b = rank(r, p)
168 cost += a + b
169 regret = maxOf(regret, a, b)
170 }
171 return Candidate(partners, origin, via, cost, regret)
172 }

174 fun enumerate(): Lattice {
175 if (agents.drop(proposers).any { it.capacity != 1 }) fail("NOT ONE TO ONE")
176 val matchings = mutableListOf(candidate(deferred().partners, 0, 0))
177 val numbered = mutableListOf<Rotation>()
178 var index = 0
179 while (index < matchings.size) {
180 val partners = matchings[index].partners
181 for (rotation in rotations(partners)) {
182 var number = numbered.indexOf(rotation)
183 if (number < 0) {
184 number = numbered.size
185 numbered.add(rotation)
186 }
187 val next = partners.copyOf()
188 val pairs = rotation.pairs
189 for (i in pairs.indices) next[pairs[i].first] = pairs[(i + 1) % pairs.size].second
190 if (matchings.any { it.partners.contentEquals(next) }) continue
191 if (matchings.size == 50) fail("TOO MANY MATCHINGS")
192 matchings.add(candidate(next, index + 1, number + 1))
193 }
194 index++
195 }
196 return Lattice(matchings, numbered)
197 }

199 fun check(words: List<String>): List<String> {
200 if (kind == Kind.HOUSING) fail("WRONG TYPE $name")
201 val partners = emptyMatching(agents.size)
202 val counts = IntArray(agents.size)
203 for (word in words.drop(2)) {
204 val (left, right) = word.split('-')
205 val two = kind == Kind.TWO_SIDED
206 val x = agent(left, if (two) Side.PROPOSER else Side.ANY)
207 val y = agent(right, if (two) Side.RECEIVER else Side.ANY)
208 if (partners[x] >= 0) fail("DUPLICATE AGENT $left")
209 if (two) {
210 if (++counts[y] > agents[y].capacity) fail("OVER CAPACITY $right")
211 } else {
212 if (x == y || partners[y] >= 0) fail("DUPLICATE AGENT $right")
213 partners[y] = x
214 }
215 partners[x] = y
216 }
217 val unacceptable = mutableListOf<Pair<Int, Int>>()
218 val blocking = mutableListOf<Pair<Int, Int>>()
219 if (kind == Kind.TWO_SIDED) {
220 for (p in 0 until proposers) {
221 val partner = partners[p]
222 if (partner >= 0 && (!accepts(p, partner) || !accepts(partner, p))) unacceptable.add(p to partner)
223 for (r in proposers until agents.size) {
224 if (partners[p] == r || !accepts(p, r) || !accepts(r, p) || rank(p, r) >= rank(p, partners[p])) continue
225 val worst = (0 until proposers).filter { partners[it] == r }.maxOfOrNull { rank(r, it) } ?: 0
226 if (counts[r] < agents[r].capacity || rank(r, p) < worst) blocking.add(p to r)
227 }
228 }
229 } else {
230 for (a in agents.indices) for (b in a + 1 until agents.size) {
231 val mutual = accepts(a, b) && accepts(b, a)
232 if (partners[a] == b) {
233 if (!mutual) unacceptable.add(a to b)
234 } else if (mutual && rank(a, b) < rank(a, partners[a]) && rank(b, a) < rank(b, partners[b])) blocking.add(a to b)
235 }
236 }
237 if (unacceptable.isEmpty() && blocking.isEmpty()) return listOf("CHECK $name STABLE")
238 return buildList {
239 add("CHECK $name UNSTABLE unacceptable=${unacceptable.size} blocking=${blocking.size}")
240 for ((a, b) in unacceptable) add("UNACCEPTABLE ${agents[a].name} ${agents[b].name}")
241 for ((a, b) in blocking) add("BLOCKING ${agents[a].name} ${agents[b].name}")
242 }
243 }

249 fun roommates(verbose: Boolean): List<String> {
250 requireKind(Kind.ROOMMATE)
251 val n = agents.size
252 for (agent in agents) if (agent.preference.size < n - 1) fail("INCOMPLETE ${agent.name}")
253 val lists = agents.map { it.preference.toMutableList() }
254 val held = emptyMatching(n)
255 val queue = ArrayDeque((0 until n).toList())
256 var proposals = 0
257 var failure = -1
258 while (queue.isNotEmpty()) {
259 val x = queue.removeFirst()
260 val y = lists[x].first()
261 if (held[y] >= 0) queue.addLast(held[y])
262 held[y] = x
263 proposals++
264 val removed = lists[y].drop(lists[y].indexOf(x) + 1)
265 for (w in removed) removePair(lists, y, w)
266 failure = lists.indexOfFirst { it.isEmpty() }
267 if (failure >= 0) break
268 }
269 val history = mutableListOf<String>()
270 var rotations = 0
271 if (failure < 0 && verbose) history.addAll(table(lists))
272 while (failure < 0) {
273 var p = lists.indexOfFirst { it.size >= 2 }
274 if (p < 0) break
275 val positions = emptyMatching(n)
276 val path = mutableListOf<Int>()
277 while (positions[p] < 0) {
278 positions[p] = path.size
279 path.add(p)
280 val q = lists[p][1]
281 p = lists[q].last()
282 }
283 val cycle = path.drop(positions[p])
284 val front = cycle.map { lists[it].first() }
285 rotations++
286 val notation = cycle.indices.joinToString(" ") { "${agents[cycle[it]].name}-${agents[front[it]].name}" }
287 history.add("ROTATION $rotations $notation")
288 val removals = mutableListOf<Pair<Int, Int>>()
289 for (i in cycle.indices) {
290 val u = front[(i + 1) % front.size]
291 for (w in lists[u].drop(lists[u].indexOf(cycle[i]) + 1)) removals.add(u to w)
292 }
293 for ((a, b) in removals) removePair(lists, a, b)
294 if (verbose) history.addAll(table(lists))
295 failure = lists.indexOfFirst { it.isEmpty() }
296 }
297 return buildList {
298 add("ROOMMATES $name proposals=$proposals")
299 addAll(history)
300 if (failure >= 0) add("NO STABLE MATCHING ${agents[failure].name}")
301 else add("MATCHING " + agents.indices.filter { it < lists[it].first() }
302 .joinToString(" ") { "${agents[it].name}-${agents[lists[it].first()].name}" })
303 }
304 }

306 fun ttc(): List<String> {
307 val n = agents.size
308 val remaining = BooleanArray(n) { true }
309 val effective = agents.indices.map { effective(it) }
310 val allocation = emptyMatching(n)
311 val history = mutableListOf<String>()
312 while (true) {
313 val start = remaining.indexOfFirst { it }
314 if (start < 0) break
315 val points = IntArray(n)
316 for (a in agents.indices) if (remaining[a]) points[a] = effective[a].first { remaining[it] }
317 val seen = BooleanArray(n)
318 var p = start
319 while (!seen[p]) { seen[p] = true; p = points[p] }
320 val first = p
321 val cycle = mutableListOf<String>()
322 do {
323 allocation[p] = points[p]
324 remaining[p] = false
325 cycle.add("${agents[p].name}:${houses[points[p]]}")
326 p = points[p]
327 } while (p != first)
328 history.add("CYCLE ${history.size + 1} ${cycle.joinToString(" ")}")
329 }
330 return listOf("TTC $name cycles=${history.size}") + history + "ALLOCATION ${allocation(allocation)}"
331 }

333 fun serial(words: List<String>): List<String> {
334 val order = mutableListOf<Int>()
335 for (name in words.drop(2)) {
336 val a = agent(name)
337 if (a in order) fail("DUPLICATE AGENT $name")
338 order.add(a)
339 }
340 if (order.size != agents.size) fail("BAD ORDER")
341 val taken = BooleanArray(agents.size)
342 val allocation = emptyMatching(agents.size)
343 for (a in order) {
344 val house = agents[a].preference.firstOrNull { !taken[it] } ?: continue
345 allocation[a] = house
346 taken[house] = true
347 }
348 val worse = agents.indices.filter { allocation[it] < 0 || allocation[it] !in effective(it) }.map { agents[it].name }
349 return listOf("SERIAL $name assigned=${allocation.count { it >= 0 }}", "ALLOCATION ${allocation(allocation)}", "WORSE ${listText(worse)}")
350 }

352 fun hr(): List<String> {
353 val a = deferred().partners
354 val h = deferred(reverse = true).partners
355 val unmatched = (0 until proposers).filter { a[it] < 0 }.map { agents[it].name }
356 val output = mutableListOf("HR $name residents=$proposers hospitals=${agents.size - proposers}",
357 "RESIDENT-OPTIMAL ${pairs(a)}", "HOSPITAL-OPTIMAL ${pairs(h)}", "UNMATCHED ${listText(unmatched)}")
358 var rural = (0 until proposers).all { (a[it] < 0) == (h[it] < 0) }
359 for (r in proposers until agents.size) {
360 val first = (0 until proposers).filter { a[it] == r }
361 val second = (0 until proposers).filter { h[it] == r }
362 val same = first == second
363 val capacity = agents[r].capacity
364 output.add("HOSPITAL ${agents[r].name} cap=$capacity resident=${first.size} hospital=${second.size} same=${if (same) "YES" else "NO"}")
365 rural = rural && first.size == second.size && (first.size == capacity || same)
366 }
367 output.add(if (rural) "RURAL HOLDS" else "RURAL FAILS")
368 return output
369 }

371 fun manipulate(a: Int): List<String> {
372 val truth = agents[a].preference
373 val b0 = boston().partners[a]
374 val g0 = deferred().partners[a]
375 val output = mutableListOf("MANIPULATE $name ${agents[a].name} boston=${partnerName(b0)} gs=${partnerName(g0)} swaps=${maxOf(0, truth.size - 1)}")
376 for (i in 0 until truth.size - 1) {
377 val trial = Market(name, kind, agents.map { it.copy() }, proposers, houses, current)
378 val swapped = truth.toMutableList()
379 swapped[i] = truth[i + 1]
380 swapped[i + 1] = truth[i]
381 trial.agents[a].preference = swapped.toList()
382 val b = trial.boston().partners[a]
383 val g = trial.deferred().partners[a]
384 val swap = "${i + 1} ${agents[truth[i]].name} ${agents[truth[i + 1]].name}"
385 if (rank(a, b) < rank(a, b0)) output.add("IMPROVE BOSTON $swap result=${partnerName(b)}")
386 if (rank(a, g) < rank(a, g0)) output.add("IMPROVE GS $swap result=${partnerName(g)}")
387 }
388 if (output.size == 1) output.add("NO IMPROVEMENT")
389 return output
390 }

392 fun stats(): List<String> {
393 val saved = current ?: fail("NO MATCHING")
394 val sums = IntArray(2)
395 val first = IntArray(2)
396 val occupied = BooleanArray(agents.size)
397 var pairs = 0
398 for (p in 0 until proposers) {
399 val r = saved.partners[p]
400 if (r < 0) continue
401 pairs++
402 occupied[r] = true
403 val ranks = intArrayOf(rank(p, r), rank(r, p))
404 for (i in 0..1) { sums[i] += ranks[i]; if (ranks[i] == 1) first[i]++ }
405 }
406 val unmatched = intArrayOf(proposers - pairs, (proposers until agents.size).count { !occupied[it] })
407 val output = mutableListOf("STATS $name source=${saved.source} pairs=$pairs")
408 for ((i, label) in listOf("PROPOSERS", "RECEIVERS").withIndex()) {
409 var average = "-"
410 var truncated = "-"
411 if (pairs > 0) {
412 val divisor = gcd(sums[i], pairs)
413 val cents = 100 * sums[i] / pairs
414 average = "${sums[i] / divisor}/${pairs / divisor}"
415 truncated = "%d.%02d".format(Locale.ROOT, cents / 100, cents % 100)
416 }
417 output.add("$label sum=${sums[i]} average=$average truncated=$truncated first=${first[i]} unmatched=${unmatched[i]}")
418 }
419 return output
420 }

446private fun gcd(first: Int, second: Int): Int {
447 var a = first
448 var b = second
449 while (b != 0) { val remainder = a % b; a = b; b = remainder }
450 return a
451}

456 fun create(words: List<String>): List<String> {
457 val op = words[0]
458 val name = words[1]
459 if (name in markets) fail("DUPLICATE MARKET $name")
460 val kind = when (op) { "ROOMS" -> Kind.ROOMMATE; "HOUSES" -> Kind.HOUSING; else -> Kind.TWO_SIDED }
461 if (op == "RANDOM") {
462 val n = words[2].toInt()
463 val seed = words[3].toLong()
464 var state = seed
465 val agents = List(2 * n) { Agent("${if (it < n) "P" else "R"}${it % n + 1}") }
466 for (a in agents.indices) {
467 val preference = MutableList(n) { (if (a < n) n else 0) + it }
468 for (i in n downTo 2) {
469 state = 48271L * state % 2147483647L
470 val j = (state % i).toInt()
471 val first = preference[i - 1]
472 preference[i - 1] = preference[j]
473 preference[j] = first
474 }
475 agents[a].preference = preference.toList()
476 }
477 markets[name] = Market(name, kind, agents, n)
478 return listOf("RANDOM $name n=$n seed=$seed")
479 }
480 val names = mutableListOf<String>()
481 val houses = mutableListOf<String>()
482 var proposers = 0
483 for (word in words.drop(2)) {
484 if (word == "|") { proposers = names.size; continue }
485 val pair = if (kind == Kind.HOUSING) word.split(':') else listOf(word)
486 if (pair[0] in names) fail("DUPLICATE AGENT ${pair[0]}")
487 if (kind == Kind.HOUSING) {
488 if (pair[1] in houses) fail("DUPLICATE HOUSE ${pair[1]}")
489 houses.add(pair[1])
490 }
491 names.add(pair[0])
492 }
493 markets[name] = Market(name, kind, names.map { Agent(it) }, proposers, houses.toList())
494 return listOf(if (kind == Kind.TWO_SIDED) "MARKET $name proposers=$proposers receivers=${names.size - proposers}" else "$op $name agents=${names.size}")
495 }

497 fun execute(line: String): List<String> {
498 val words = line.split(' ')
499 val op = words[0]
500 val name = words[1]
501 if (op in setOf("MARKET", "ROOMS", "HOUSES", "RANDOM")) return create(words)
502 val market = markets[name] ?: fail("UNKNOWN MARKET $name")
503 when (op) {
504 "SHOW" -> return market.show()
505 "PREF" -> {
506 val a = market.agent(words[2])
507 val preference = mutableListOf<Int>()
508 for (word in words.drop(3)) {
509 val b = if (market.kind == Kind.HOUSING) market.houses.indexOf(word) else market.findAgent(word)
510 if (b < 0 || (market.kind == Kind.TWO_SIDED && (a < market.proposers) == (b < market.proposers))
511 || (market.kind == Kind.ROOMMATE && a == b)) fail("BAD CHOICE $word")
512 if (b in preference) fail("DUPLICATE CHOICE $word")
513 preference.add(b)
514 }
515 market.agents[a].preference = preference.toList()
516 if (market.kind == Kind.TWO_SIDED) market.current = null
517 return listOf("PREF $name ${words[2]} length=${preference.size}")
518 }
519 "CHECK" -> return market.check(words)
520 "ROOMMATES" -> return market.roommates(words.size == 3)
521 "TTC", "SERIAL" -> {
522 market.requireKind(Kind.HOUSING)
523 return if (op == "TTC") market.ttc() else market.serial(words)
524 }
525 }
526 market.requireKind(Kind.TWO_SIDED)
527 return when (op) {
528 "CAPACITY" -> {
529 val r = market.agent(words[2], Side.RECEIVER)
530 val capacity = words[3].toInt()
531 market.agents[r].capacity = capacity
532 market.current = null
533 listOf("CAPACITY $name ${words[2]} $capacity")
534 }
535 "GS", "GSR", "BOSTON" -> {
536 val result = if (op == "BOSTON") market.boston() else market.deferred(op == "GSR", words.size == 3)
537 market.save(Source.valueOf(op), result.partners)
538 listOf("$op $name ${if (op == "BOSTON") "applications" else "proposals"}=${result.count}") + result.history + "MATCHING ${market.pairs(result.partners)}"
539 }
540 "HR" -> market.hr()
541 "MANIPULATE" -> market.manipulate(market.agent(words[2], Side.PROPOSER))
542 "STATS" -> market.stats()
543 "LATTICE", "EGALITARIAN", "MINREGRET" -> {
544 val lattice = market.enumerate()
545 if (op == "LATTICE") buildList {
546 add("LATTICE $name matchings=${lattice.matchings.size} rotations=${lattice.rotations.size}")
547 for ((i, candidate) in lattice.matchings.withIndex()) {
548 val origin = if (candidate.origin == 0) "-" else candidate.origin.toString()
549 val via = if (candidate.via == 0) "-" else candidate.via.toString()
550 add("MATCHING ${i + 1} from=$origin via=$via cost=${candidate.cost} regret=${candidate.regret} ${market.pairs(candidate.partners)}")
551 }
552 for ((i, rotation) in lattice.rotations.withIndex()) {
553 val notation = rotation.pairs.joinToString(" ") { (p, r) -> "${market.agents[p].name}-${market.agents[r].name}" }
554 add("ROTATION ${i + 1} $notation")
555 }
556 } else {
557 val egalitarian = op == "EGALITARIAN"
558 val best = lattice.matchings.indices.minBy { if (egalitarian) lattice.matchings[it].cost else lattice.matchings[it].regret }
559 val selected = lattice.matchings[best]
560 market.save(Source.valueOf(op), selected.partners)
561 val metric = if (egalitarian) "cost=${selected.cost}" else "regret=${selected.regret}"
562 listOf("$op $name matching=${best + 1} $metric", "MATCHING ${market.pairs(selected.partners)}")
563 }
564 }
565 else -> error("Unknown command in well-formed input")
566 }
567 }
568}
