92 if (side == Side.PROPOSER && a >= proposers) throw new CommandError("NOT PROPOSER " + name);

131 Run deferred(boolean reverse, boolean verbose) {
132 int n = agents.size(), proposals = 0;
133 int[] next = new int[n], counts = new int[n];
134 List<List<Integer>> holding = new ArrayList<>();
135 for (int a = 0; a < n; a++) holding.add(new ArrayList<>());
136 ArrayDeque<Integer> queue = new ArrayDeque<>();
137 int begin = reverse ? proposers : 0, end = reverse ? n : proposers;
138 for (int a = begin; a < end; a++) if (able(a, next, counts)) queue.addLast(a);
139 List<String> history = new ArrayList<>();
140 while (!queue.isEmpty()) {
141 int x = queue.removeFirst(), y = agents.get(x).preference.get(next[x]++);
142 proposals++;
143 String action = "REJECT";
144 List<Integer> held = holding.get(y);
145 if (accepts(y, x)) {
146 if (held.size() < agents.get(y).capacity) {
147 held.add(x);
148 counts[x]++;
149 action = "HOLD";
150 } else {
151 int worst = 0;
152 for (int i = 1; i < held.size(); i++) if (rank(y, held.get(i)) > rank(y, held.get(worst))) worst = i;
153 int rejected = held.get(worst);
154 if (rank(y, x) < rank(y, rejected)) {
155 held.set(worst, x);
156 counts[rejected]--;
157 counts[x]++;
158 action = "HOLD REJECT " + agents.get(rejected).name;
159 if (able(rejected, next, counts) && !queue.contains(rejected)) queue.addLast(rejected);
160 }
161 }
162 }
163 if (verbose) history.add("PROPOSE " + agents.get(x).name + " " + agents.get(y).name + " " + action);
164 if (able(x, next, counts)) queue.addLast(x);
165 }
166 int[] partners = emptyMatching(proposers);
167 for (int y = 0; y < n; y++) for (int x : holding.get(y)) {
168 if (reverse) partners[y] = x;
169 else partners[x] = y;
170 }
171 return new Run(partners, proposals, history);
172 }

174 Run boston() {
175 int[] partners = emptyMatching(proposers), remaining = new int[agents.size()];
176 int rounds = 0, applications = 0;
177 for (int p = 0; p < proposers; p++) rounds = Math.max(rounds, agents.get(p).preference.size());
178 for (int r = proposers; r < agents.size(); r++) remaining[r] = agents.get(r).capacity;
179 for (int round = 0; round < rounds; round++) {
180 int[] applicants = emptyMatching(proposers);
181 for (int p = 0; p < proposers; p++) if (partners[p] < 0 && round < agents.get(p).preference.size()) {
182 applicants[p] = agents.get(p).preference.get(round);
183 applications++;
184 }
185 for (int r = proposers; r < agents.size(); r++) for (int p : agents.get(r).preference) {
186 if (remaining[r] > 0 && applicants[p] == r) { partners[p] = r; remaining[r]--; }
187 }
188 }
189 return new Run(partners, applications, List.of());
190 }

192 List<Rotation> rotations(int[] partners) {
193 int[] owners = emptyMatching(agents.size()), successors = emptyMatching(proposers);
194 for (int p = 0; p < proposers; p++) if (partners[p] >= 0) owners[partners[p]] = p;
195 for (int p = 0; p < proposers; p++) {
196 if (partners[p] < 0) continue;
197 List<Integer> preference = agents.get(p).preference;
198 for (int i = rank(p, partners[p]); i < preference.size(); i++) {
199 int r = preference.get(i);
200 if (accepts(r, p) && (owners[r] < 0 || rank(r, p) < rank(r, owners[r]))) {
201 successors[p] = owners[r];
202 break;
203 }
204 }
205 }
206 List<Rotation> result = new ArrayList<>();
207 boolean[] covered = new boolean[proposers];
208 for (int start = 0; start < proposers; start++) {
209 if (covered[start]) continue;
210 List<Integer> path = new ArrayList<>();
211 boolean[] seen = new boolean[proposers];
212 int p = start;
213 while (p >= 0 && !seen[p]) { seen[p] = true; path.add(p); p = successors[p]; }
214 if (p == start) {
215 List<Pair> pairs = new ArrayList<>();
216 for (int a : path) { covered[a] = true; pairs.add(new Pair(a, partners[a])); }
217 result.add(new Rotation(List.copyOf(pairs)));
218 }
219 }
220 return result;
221 }

223 Candidate candidate(int[] partners, int origin, int via) {
224 int cost = 0, regret = 0;
225 for (int p = 0; p < proposers; p++) if (partners[p] >= 0) {
226 int r = partners[p], a = rank(p, r), b = rank(r, p);
227 cost += a + b;
228 regret = Math.max(regret, Math.max(a, b));
229 }
230 return new Candidate(partners, origin, via, cost, regret);
231 }

233 Lattice enumerate() {
234 for (int r = proposers; r < agents.size(); r++) if (agents.get(r).capacity != 1) throw new CommandError("NOT ONE TO ONE");
235 List<Candidate> matchings = new ArrayList<>();
236 List<Rotation> numbered = new ArrayList<>();
237 matchings.add(candidate(deferred(false, false).partners, 0, 0));
238 for (int index = 0; index < matchings.size(); index++) {
239 int[] partners = matchings.get(index).partners;
240 for (Rotation rotation : rotations(partners)) {
241 int number = numbered.indexOf(rotation);
242 if (number < 0) { number = numbered.size(); numbered.add(rotation); }
243 int[] next = partners.clone();
244 List<Pair> pairs = rotation.pairs;
245 for (int i = 0; i < pairs.size(); i++) next[pairs.get(i).left] = pairs.get((i + 1) % pairs.size()).right;
246 if (matchings.stream().anyMatch(old -> Arrays.equals(old.partners, next))) continue;
247 if (matchings.size() == 50) throw new CommandError("TOO MANY MATCHINGS");
248 matchings.add(candidate(next, index + 1, number + 1));
249 }
250 }
251 return new Lattice(matchings, numbered);
252 }

254 List<String> check(String[] words) {
255 if (kind == Kind.HOUSING) throw new CommandError("WRONG TYPE " + name);
256 int[] partners = emptyMatching(agents.size()), counts = new int[agents.size()];
257 for (int i = 2; i < words.length; i++) {
258 String[] pair = words[i].split("-");
259 boolean two = kind == Kind.TWO_SIDED;
260 int x = agent(pair[0], two ? Side.PROPOSER : Side.ANY);
261 int y = agent(pair[1], two ? Side.RECEIVER : Side.ANY);
262 if (partners[x] >= 0) throw new CommandError("DUPLICATE AGENT " + pair[0]);
263 if (two) {
264 if (++counts[y] > agents.get(y).capacity) throw new CommandError("OVER CAPACITY " + pair[1]);
265 } else {
266 if (x == y || partners[y] >= 0) throw new CommandError("DUPLICATE AGENT " + pair[1]);
267 partners[y] = x;
268 }
269 partners[x] = y;
270 }
271 List<Pair> unacceptable = new ArrayList<>(), blocking = new ArrayList<>();
272 if (kind == Kind.TWO_SIDED) {
273 for (int p = 0; p < proposers; p++) {
274 if (partners[p] >= 0 && (!accepts(p, partners[p]) || !accepts(partners[p], p))) unacceptable.add(new Pair(p, partners[p]));
275 for (int r = proposers; r < agents.size(); r++) {
276 if (partners[p] == r || !accepts(p, r) || !accepts(r, p) || rank(p, r) >= rank(p, partners[p])) continue;
277 int worst = 0;
278 for (int a = 0; a < proposers; a++) if (partners[a] == r) worst = Math.max(worst, rank(r, a));
279 if (counts[r] < agents.get(r).capacity || rank(r, p) < worst) blocking.add(new Pair(p, r));
280 }
281 }
282 } else {
283 for (int a = 0; a < agents.size(); a++) for (int b = a + 1; b < agents.size(); b++) {
284 boolean mutual = accepts(a, b) && accepts(b, a);
285 if (partners[a] == b) {
286 if (!mutual) unacceptable.add(new Pair(a, b));
287 } else if (mutual && rank(a, b) < rank(a, partners[a]) && rank(b, a) < rank(b, partners[b])) blocking.add(new Pair(a, b));
288 }
289 }
290 if (unacceptable.isEmpty() && blocking.isEmpty()) return List.of("CHECK " + name + " STABLE");
291 List<String> output = new ArrayList<>();
292 output.add("CHECK " + name + " UNSTABLE unacceptable=" + unacceptable.size() + " blocking=" + blocking.size());
293 for (Pair pair : unacceptable) output.add("UNACCEPTABLE " + agents.get(pair.left).name + " " + agents.get(pair.right).name);
294 for (Pair pair : blocking) output.add("BLOCKING " + agents.get(pair.left).name + " " + agents.get(pair.right).name);
295 return output;
296 }

305 List<String> roommates(boolean verbose) {
306 require(Kind.ROOMMATE);
307 int n = agents.size();
308 for (Agent agent : agents) if (agent.preference.size() < n - 1) throw new CommandError("INCOMPLETE " + agent.name);
309 List<List<Integer>> lists = new ArrayList<>();
310 ArrayDeque<Integer> queue = new ArrayDeque<>();
311 for (int a = 0; a < n; a++) { lists.add(new ArrayList<>(agents.get(a).preference)); queue.addLast(a); }
312 int[] held = emptyMatching(n);
313 int proposals = 0, failure = -1;
314 while (!queue.isEmpty()) {
315 int x = queue.removeFirst(), y = lists.get(x).getFirst();
316 if (held[y] >= 0) queue.addLast(held[y]);
317 held[y] = x;
318 proposals++;
319 List<Integer> targets = lists.get(y);
320 List<Integer> removed = new ArrayList<>(targets.subList(targets.indexOf(x) + 1, targets.size()));
321 for (int w : removed) removePair(lists, y, w);
322 failure = firstEmpty(lists);
323 if (failure >= 0) break;
324 }
325 List<String> history = new ArrayList<>();
326 int rotations = 0;
327 if (failure < 0 && verbose) table(lists, history);
328 while (failure < 0) {
329 int p = 0;
330 while (p < n && lists.get(p).size() < 2) p++;
331 if (p == n) break;
332 int[] positions = emptyMatching(n);
333 List<Integer> path = new ArrayList<>();
334 while (positions[p] < 0) {
335 positions[p] = path.size();
336 path.add(p);
337 int q = lists.get(p).get(1);
338 p = lists.get(q).getLast();
339 }
340 List<Integer> cycle = path.subList(positions[p], path.size());
341 int[] front = new int[cycle.size()];
342 List<String> notation = new ArrayList<>();
343 for (int i = 0; i < cycle.size(); i++) {
344 int x = cycle.get(i);
345 front[i] = lists.get(x).getFirst();
346 notation.add(agents.get(x).name + "-" + agents.get(front[i]).name);
347 }
348 history.add("ROTATION " + (++rotations) + " " + String.join(" ", notation));
349 List<Pair> removals = new ArrayList<>();
350 for (int i = 0; i < cycle.size(); i++) {
351 int u = front[(i + 1) % front.length];
352 List<Integer> targets = lists.get(u);
353 for (int w : targets.subList(targets.indexOf(cycle.get(i)) + 1, targets.size())) removals.add(new Pair(u, w));
354 }
355 for (Pair pair : removals) removePair(lists, pair.left, pair.right);
356 if (verbose) table(lists, history);
357 failure = firstEmpty(lists);
358 }
359 List<String> output = new ArrayList<>();
360 output.add("ROOMMATES " + name + " proposals=" + proposals);
361 output.addAll(history);
362 if (failure >= 0) output.add("NO STABLE MATCHING " + agents.get(failure).name);
363 else {
364 List<String> pairs = new ArrayList<>();
365 for (int a = 0; a < n; a++) if (a < lists.get(a).getFirst()) pairs.add(agents.get(a).name + "-" + agents.get(lists.get(a).getFirst()).name);
366 output.add("MATCHING " + String.join(" ", pairs));
367 }
368 return output;
369 }

371 List<String> ttc() {
372 int n = agents.size();
373 boolean[] remaining = new boolean[n];
374 Arrays.fill(remaining, true);
375 List<List<Integer>> effective = new ArrayList<>();
376 for (int a = 0; a < n; a++) effective.add(effective(a));
377 int[] allocation = emptyMatching(n);
378 List<String> history = new ArrayList<>();
379 while (true) {
380 int start = 0;
381 while (start < n && !remaining[start]) start++;
382 if (start == n) break;
383 int[] points = new int[n];
384 for (int a = 0; a < n; a++) if (remaining[a]) {
385 for (int house : effective.get(a)) if (remaining[house]) { points[a] = house; break; }
386 }
387 boolean[] seen = new boolean[n];
388 int p = start;
389 while (!seen[p]) { seen[p] = true; p = points[p]; }
390 int first = p;
391 List<String> cycle = new ArrayList<>();
392 do {
393 allocation[p] = points[p];
394 remaining[p] = false;
395 cycle.add(agents.get(p).name + ":" + houses.get(points[p]));
396 p = points[p];
397 } while (p != first);
398 history.add("CYCLE " + (history.size() + 1) + " " + String.join(" ", cycle));
399 }
400 List<String> output = new ArrayList<>();
401 output.add("TTC " + name + " cycles=" + history.size());
402 output.addAll(history);
403 output.add("ALLOCATION " + allocation(allocation));
404 return output;
405 }

407 List<String> serial(String[] words) {
408 List<Integer> order = new ArrayList<>();
409 for (int i = 2; i < words.length; i++) {
410 int a = agent(words[i], Side.ANY);
411 if (order.contains(a)) throw new CommandError("DUPLICATE AGENT " + words[i]);
412 order.add(a);
413 }
414 if (order.size() != agents.size()) throw new CommandError("BAD ORDER");
415 boolean[] taken = new boolean[agents.size()];
416 int[] allocation = emptyMatching(agents.size());
417 int assigned = 0;
418 for (int a : order) for (int house : agents.get(a).preference) if (!taken[house]) {
419 allocation[a] = house;
420 taken[house] = true;
421 assigned++;
422 break;
423 }
424 List<String> worse = new ArrayList<>();
425 for (int a = 0; a < agents.size(); a++) if (allocation[a] < 0 || !effective(a).contains(allocation[a])) worse.add(agents.get(a).name);
426 return List.of("SERIAL " + name + " assigned=" + assigned, "ALLOCATION " + allocation(allocation), "WORSE " + listText(worse));
427 }

429 List<String> hr() {
430 int[] a = deferred(false, false).partners, h = deferred(true, false).partners;
431 List<String> unmatched = new ArrayList<>();
432 boolean rural = true;
433 for (int p = 0; p < proposers; p++) {
434 if (a[p] < 0) unmatched.add(agents.get(p).name);
435 if ((a[p] < 0) != (h[p] < 0)) rural = false;
436 }
437 List<String> output = new ArrayList<>(List.of("HR " + name + " residents=" + proposers + " hospitals=" + (agents.size() - proposers),
438 "RESIDENT-OPTIMAL " + pairs(a), "HOSPITAL-OPTIMAL " + pairs(h), "UNMATCHED " + listText(unmatched)));
439 for (int r = proposers; r < agents.size(); r++) {
440 int resident = 0, hospital = 0;
441 boolean same = true;
442 for (int p = 0; p < proposers; p++) {
443 if (a[p] == r) resident++;
444 if (h[p] == r) hospital++;
445 if ((a[p] == r) != (h[p] == r)) same = false;
446 }
447 int capacity = agents.get(r).capacity;
448 output.add("HOSPITAL " + agents.get(r).name + " cap=" + capacity + " resident=" + resident + " hospital=" + hospital + " same=" + (same ? "YES" : "NO"));
449 if (resident != hospital || (resident < capacity && !same)) rural = false;
450 }
451 output.add(rural ? "RURAL HOLDS" : "RURAL FAILS");
452 return output;
453 }

455 List<String> manipulate(int a) {
456 List<Integer> truth = agents.get(a).preference;
457 int b0 = boston().partners[a], g0 = deferred(false, false).partners[a];
458 List<String> output = new ArrayList<>();
459 output.add("MANIPULATE " + name + " " + agents.get(a).name + " boston=" + partnerName(b0) + " gs=" + partnerName(g0) + " swaps=" + Math.max(0, truth.size() - 1));
460 for (int i = 0; i + 1 < truth.size(); i++) {
461 Market trial = new Market(this);
462 List<Integer> swapped = new ArrayList<>(truth);
463 Collections.swap(swapped, i, i + 1);
464 trial.agents.get(a).preference = List.copyOf(swapped);
465 int b = trial.boston().partners[a], g = trial.deferred(false, false).partners[a];
466 String swap = (i + 1) + " " + agents.get(truth.get(i)).name + " " + agents.get(truth.get(i + 1)).name;
467 if (rank(a, b) < rank(a, b0)) output.add("IMPROVE BOSTON " + swap + " result=" + partnerName(b));
468 if (rank(a, g) < rank(a, g0)) output.add("IMPROVE GS " + swap + " result=" + partnerName(g));
469 }
470 if (output.size() == 1) output.add("NO IMPROVEMENT");
471 return output;
472 }

474 List<String> stats() {
475 if (current == null) throw new CommandError("NO MATCHING");
476 int[] sum = new int[2], first = new int[2];
477 boolean[] occupied = new boolean[agents.size()];
478 int pairs = 0;
479 for (int p = 0; p < proposers; p++) {
480 int r = current.partners[p];
481 if (r < 0) continue;
482 pairs++;
483 occupied[r] = true;
484 int[] ranks = {rank(p, r), rank(r, p)};
485 for (int i = 0; i < 2; i++) { sum[i] += ranks[i]; if (ranks[i] == 1) first[i]++; }
486 }
487 int[] unmatched = {proposers - pairs, 0};
488 for (int r = proposers; r < agents.size(); r++) if (!occupied[r]) unmatched[1]++;
489 List<String> output = new ArrayList<>();
490 output.add("STATS " + name + " source=" + current.source + " pairs=" + pairs);
491 String[] labels = {"PROPOSERS", "RECEIVERS"};
492 for (int i = 0; i < 2; i++) {
493 String average = "-", truncated = "-";
494 if (pairs > 0) {
495 int divisor = gcd(sum[i], pairs), cents = 100 * sum[i] / pairs;
496 average = sum[i] / divisor + "/" + pairs / divisor;
497 truncated = String.format(Locale.ROOT, "%d.%02d", cents / 100, cents % 100);
498 }
499 output.add(labels[i] + " sum=" + sum[i] + " average=" + average + " truncated=" + truncated + " first=" + first[i] + " unmatched=" + unmatched[i]);
500 }
501 return output;
502 }

537 while (b != 0) { int remainder = a % b; a = b; b = remainder; }

543 static List<String> create(String[] words) {
544 String op = words[0], name = words[1];
545 if (markets.containsKey(name)) throw new CommandError("DUPLICATE MARKET " + name);
546 Kind kind = switch (op) { case "ROOMS" -> Kind.ROOMMATE; case "HOUSES" -> Kind.HOUSING; default -> Kind.TWO_SIDED; };
547 List<String> names = new ArrayList<>(), houses = new ArrayList<>();
548 if (op.equals("RANDOM")) {
549 int n = Integer.parseInt(words[2]);
550 long seed = Long.parseLong(words[3]), state = seed;
551 for (int a = 0; a < 2 * n; a++) names.add((a < n ? "P" : "R") + (a % n + 1));
552 Market market = new Market(name, kind, names, n, houses);
553 for (int a = 0; a < 2 * n; a++) {
554 List<Integer> preference = new ArrayList<>();
555 for (int i = 0; i < n; i++) preference.add((a < n ? n : 0) + i);
556 for (int i = n; i > 1; i--) {
557 state = 48271 * state % 2147483647;
558 Collections.swap(preference, i - 1, (int)(state % i));
559 }
560 market.agents.get(a).preference = List.copyOf(preference);
561 }
562 markets.put(name, market);
563 return List.of("RANDOM " + name + " n=" + n + " seed=" + seed);
564 }
565 Set<String> agentNames = new HashSet<>(), houseNames = new HashSet<>();
566 int proposers = 0;
567 for (int i = 2; i < words.length; i++) {
568 if (words[i].equals("|")) { proposers = names.size(); continue; }
569 String[] pair = kind == Kind.HOUSING ? words[i].split(":") : new String[]{words[i]};
570 if (!agentNames.add(pair[0])) throw new CommandError("DUPLICATE AGENT " + pair[0]);
571 if (kind == Kind.HOUSING) {
572 if (!houseNames.add(pair[1])) throw new CommandError("DUPLICATE HOUSE " + pair[1]);
573 houses.add(pair[1]);
574 }
575 names.add(pair[0]);
576 }
577 Market market = new Market(name, kind, names, proposers, houses);
578 markets.put(name, market);
579 return List.of(kind == Kind.TWO_SIDED ? "MARKET " + name + " proposers=" + proposers + " receivers=" + (names.size() - proposers)
580 : op + " " + name + " agents=" + names.size());
581 }

583 static List<String> execute(String line) {
584 String[] words = line.split(" ");
585 String op = words[0], name = words[1];
586 if (Set.of("MARKET", "ROOMS", "HOUSES", "RANDOM").contains(op)) return create(words);
587 Market market = markets.get(name);
588 if (market == null) throw new CommandError("UNKNOWN MARKET " + name);
589 switch (op) {
590 case "SHOW" -> { return market.show(); }
591 case "PREF" -> {
592 int a = market.agent(words[2], Side.ANY);
593 List<Integer> preference = new ArrayList<>();
594 for (int i = 3; i < words.length; i++) {
595 int b = market.kind == Kind.HOUSING ? market.houses.indexOf(words[i]) : market.findAgent(words[i]);
596 if (b < 0 || (market.kind == Kind.TWO_SIDED && (a < market.proposers) == (b < market.proposers))
597 || (market.kind == Kind.ROOMMATE && a == b)) throw new CommandError("BAD CHOICE " + words[i]);
598 if (preference.contains(b)) throw new CommandError("DUPLICATE CHOICE " + words[i]);
599 preference.add(b);
600 }
601 market.agents.get(a).preference = List.copyOf(preference);
602 if (market.kind == Kind.TWO_SIDED) market.current = null;
603 return List.of("PREF " + name + " " + words[2] + " length=" + preference.size());
604 }
605 case "CHECK" -> { return market.check(words); }
606 case "ROOMMATES" -> { return market.roommates(words.length == 3); }
607 case "TTC", "SERIAL" -> {
608 market.require(Kind.HOUSING);
609 return op.equals("TTC") ? market.ttc() : market.serial(words);
610 }
611 }
612 market.require(Kind.TWO_SIDED);
613 switch (op) {
614 case "CAPACITY" -> {
615 int r = market.agent(words[2], Side.RECEIVER), capacity = Integer.parseInt(words[3]);
616 market.agents.get(r).capacity = capacity;
617 market.current = null;
618 return List.of("CAPACITY " + name + " " + words[2] + " " + capacity);
619 }
620 case "GS", "GSR", "BOSTON" -> {
621 Run result = op.equals("BOSTON") ? market.boston() : market.deferred(op.equals("GSR"), words.length == 3);
622 market.current = new Saved(Source.valueOf(op), result.partners);
623 List<String> output = new ArrayList<>();
624 output.add(op + " " + name + " " + (op.equals("BOSTON") ? "applications" : "proposals") + "=" + result.count);
625 output.addAll(result.history);
626 output.add("MATCHING " + market.pairs(result.partners));
627 return output;
628 }
629 case "HR" -> { return market.hr(); }
630 case "MANIPULATE" -> { return market.manipulate(market.agent(words[2], Side.PROPOSER)); }
631 case "STATS" -> { return market.stats(); }
632 case "LATTICE", "EGALITARIAN", "MINREGRET" -> {
633 Lattice lattice = market.enumerate();
634 if (op.equals("LATTICE")) {
635 List<String> output = new ArrayList<>();
636 output.add("LATTICE " + name + " matchings=" + lattice.matchings.size() + " rotations=" + lattice.rotations.size());
637 for (int i = 0; i < lattice.matchings.size(); i++) {
638 Candidate candidate = lattice.matchings.get(i);
639 String origin = candidate.origin == 0 ? "-" : Integer.toString(candidate.origin);
640 String via = candidate.via == 0 ? "-" : Integer.toString(candidate.via);
641 output.add("MATCHING " + (i + 1) + " from=" + origin + " via=" + via + " cost=" + candidate.cost + " regret=" + candidate.regret + " " + market.pairs(candidate.partners));
642 }
643 for (int i = 0; i < lattice.rotations.size(); i++) {
644 List<String> notation = lattice.rotations.get(i).pairs.stream().map(pair -> market.agents.get(pair.left).name + "-" + market.agents.get(pair.right).name).toList();
645 output.add("ROTATION " + (i + 1) + " " + String.join(" ", notation));
646 }
647 return output;
648 }
649 boolean egalitarian = op.equals("EGALITARIAN");
650 int best = 0, value = 0;
651 for (int i = 0; i < lattice.matchings.size(); i++) {
652 Candidate candidate = lattice.matchings.get(i);
653 int metric = egalitarian ? candidate.cost : candidate.regret;
654 if (i == 0 || metric < value) { best = i; value = metric; }
655 }
656 int[] selected = lattice.matchings.get(best).partners;
657 market.current = new Saved(Source.valueOf(op), selected);
658 return List.of(op + " " + name + " matching=" + (best + 1) + " " + (egalitarian ? "cost" : "regret") + "=" + value,
659 "MATCHING " + market.pairs(selected));
660 }
661 default -> throw new IllegalStateException("Unknown command in well-formed input");
662 }
663 }

665 public static void main(String[] args) throws Exception {
666 BufferedReader input = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.UTF_8));
667 BufferedWriter output = new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.UTF_8));
668 int count = Integer.parseInt(input.readLine());
669 for (int i = 0; i < count; i++) {
670 String line = input.readLine();
671 try {
672 for (String row : execute(line)) { output.write(row); output.write('\n'); }
673 } catch (CommandError error) { output.write(error.getMessage()); output.write('\n'); }
674 }
675 output.flush();
676 }
