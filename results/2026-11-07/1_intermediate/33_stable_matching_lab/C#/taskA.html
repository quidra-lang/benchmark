6sealed class CommandError(string message) : Exception("ERROR " + message);
7enum MarketKind { TwoSided, Roommates, Housing }

9sealed class Market(string name, MarketKind kind, string[] agents, int proposerCount = 0, string[]? houses = null)
10{
11 public string Name { get; } = name;
12 public MarketKind Kind { get; } = kind;
13 public string[] Agents { get; } = agents;
14 public int ProposerCount { get; } = proposerCount;
15 public string[] Houses { get; } = houses ?? [];
16 public int[][] Preferences { get; } = agents.Select(_ => Array.Empty<int>()).ToArray();
17 public int[] Capacities { get; } = Enumerable.Repeat(1, agents.Length).ToArray();
18 public int[]? Current { get; private set; }
19 public string Source { get; private set; } = "-";
20 public int Count => Agents.Length;
21 public IEnumerable<int> Proposers => Enumerable.Range(0, ProposerCount);
22 public IEnumerable<int> Receivers => Enumerable.Range(ProposerCount, Count - ProposerCount);
23 public int Agent(string text)
24 {
25 int index = Array.IndexOf(Agents, text);
26 if (index < 0) throw new CommandError($"UNKNOWN AGENT {text}");
27 return index;
28 }
29 public int Proposer(string text)
30 {
31 int agent = Agent(text);
32 if (agent >= ProposerCount) throw new CommandError($"NOT PROPOSER {text}");
33 return agent;
34 }
35 public int Receiver(string text)
36 {
37 int agent = Agent(text);
38 if (agent < ProposerCount) throw new CommandError($"NOT RECEIVER {text}");
39 return agent;
40 }
41 public int Choice(int agent, string text)
42 {
43 int choice = Array.IndexOf(Kind == MarketKind.Housing ? Houses : Agents, text);
44 bool valid = choice >= 0 && (Kind switch
45 {
46 MarketKind.TwoSided => (agent < ProposerCount) != (choice < ProposerCount),
47 MarketKind.Roommates => choice != agent,
48 _ => true
49 });
50 if (!valid) throw new CommandError($"BAD CHOICE {text}");
51 return choice;
52 }
53 public void ClearMatching() { Current = null; Source = "-"; }
54 public void SetMatching(int[] partners, string source) { Current = (int[])partners.Clone(); Source = source; }
55 public string Partner(int agent) => agent < 0 ? "*" : Agents[agent];
56 public string Pairs(int[] partners) => string.Join(' ', Proposers.Select(p => $"{Agents[p]}-{Partner(partners[p])}"));
57 public string Allocation(int[] allocation) => string.Join(' ', Enumerable.Range(0, Count)
58 .Select(a => $"{Agents[a]}-{(allocation[a] < 0 ? "*" : Houses[allocation[a]])}"));
59 public int[] Effective(int agent)
60 {
61 int[] preference = Preferences[agent];
62 int own = Array.IndexOf(preference, agent);
63 return own >= 0 ? preference[..(own + 1)] : [..preference, agent];
64 }
65}

74 public static int Rank(int[] preference, int partner)
75 {
76 int index = Array.IndexOf(preference, partner);
77 return index < 0 ? preference.Length + 1 : index + 1;
78 }
79 static bool Acceptable(int[][] preferences, int a, int b) =>
80 Array.IndexOf(preferences[a], b) >= 0 && Array.IndexOf(preferences[b], a) >= 0;
81 public static MatchingResult Deferred(Market market, bool reverse = false, bool verbose = false, int[][]? preferences = null)
82 {
83 int[][] lists = preferences ?? market.Preferences;
84 int[] next = new int[market.Count], counts = new int[market.Count];
85 bool[] queued = new bool[market.Count];
86 List<int>[] holding = Enumerable.Range(0, market.Count).Select(_ => new List<int>()).ToArray();
87 Queue<int> queue = new();
88 bool Able(int agent) => counts[agent] < market.Capacities[agent] && next[agent] < lists[agent].Length;
89 void Enqueue(int agent)
90 {
91 if (Able(agent) && !queued[agent]) { queue.Enqueue(agent); queued[agent] = true; }
92 }
93 foreach (int agent in reverse ? market.Receivers : market.Proposers) Enqueue(agent);
94 int proposals = 0;
95 List<string> log = [];
96 while (queue.Count > 0)
97 {
98 int x = queue.Dequeue();
99 queued[x] = false;
100 int y = lists[x][next[x]++];
101 proposals++;
102 string action = "REJECT";
103 if (Array.IndexOf(lists[y], x) >= 0)
104 {
105 if (holding[y].Count < market.Capacities[y])
106 {
107 holding[y].Add(x); counts[x]++;
108 action = "HOLD";
109 }
110 else
111 {
112 int worst = 0;
113 for (int i = 1; i < holding[y].Count; i++)
114 if (Rank(lists[y], holding[y][i]) > Rank(lists[y], holding[y][worst])) worst = i;
115 int rejected = holding[y][worst];
116 if (Rank(lists[y], x) < Rank(lists[y], rejected))
117 {
118 holding[y][worst] = x;
119 counts[rejected]--; counts[x]++;
120 action = $"HOLD REJECT {market.Agents[rejected]}";
121 Enqueue(rejected);
122 }
123 }
124 }
125 if (verbose) log.Add($"PROPOSE {market.Agents[x]} {market.Agents[y]} {action}");
126 Enqueue(x);
127 }
128 int[] partners = Enumerable.Repeat(-1, market.ProposerCount).ToArray();
129 foreach (int y in reverse ? market.Proposers : market.Receivers)
130 foreach (int x in holding[y]) partners[reverse ? y : x] = reverse ? x : y;
131 return new(partners, proposals, log);
132 }
133 public static MatchingResult Boston(Market market, int[][]? preferences = null)
134 {
135 int[][] lists = preferences ?? market.Preferences;
136 int[] partners = Enumerable.Repeat(-1, market.ProposerCount).ToArray();
137 int[] remaining = (int[])market.Capacities.Clone();
138 int applications = 0, rounds = market.Proposers.Max(p => lists[p].Length);
139 for (int round = 0; round < rounds; round++)
140 {
141 List<int>[] applicants = Enumerable.Range(0, market.Count).Select(_ => new List<int>()).ToArray();
142 foreach (int p in market.Proposers)
143 {
144 if (partners[p] >= 0 || round >= lists[p].Length) continue;
145 applicants[lists[p][round]].Add(p);
146 applications++;
147 }
148 foreach (int r in market.Receivers)
149 {
150 applicants[r].Sort((a, b) => Rank(lists[r], a).CompareTo(Rank(lists[r], b)));
151 foreach (int p in applicants[r])
152 {
153 if (remaining[r] == 0 || Array.IndexOf(lists[r], p) < 0) continue;
154 partners[p] = r; remaining[r]--;
155 }
156 }
157 }
158 return new(partners, applications, []);
159 }
160 public static List<int> Residents(Market market, int[] partners, int receiver) =>
161 market.Proposers.Where(p => partners[p] == receiver).ToList();
162 public static (List<Pair> Unacceptable, List<Pair> Blocking) Check(Market market, int[] partners)
163 {
164 List<Pair> unacceptable = [], blocking = [];
165 int[][] preferences = market.Preferences;
166 if (market.Kind == MarketKind.Roommates)
167 {
168 for (int a = 0; a < market.Count; a++)
169 for (int b = a + 1; b < market.Count; b++)
170 {
171 bool acceptable = Acceptable(preferences, a, b);
172 if (partners[a] == b)
173 {
174 if (!acceptable) unacceptable.Add(new(a, b));
175 }
176 else if (acceptable && Rank(preferences[a], b) < Rank(preferences[a], partners[a])
177 && Rank(preferences[b], a) < Rank(preferences[b], partners[b])) blocking.Add(new(a, b));
178 }
179 }
180 else
181 {
182 foreach (int p in market.Proposers)
183 {
184 if (partners[p] >= 0 && !Acceptable(preferences, p, partners[p])) unacceptable.Add(new(p, partners[p]));
185 foreach (int r in market.Receivers)
186 {
187 if (partners[p] == r || !Acceptable(preferences, p, r)
188 || Rank(preferences[p], r) >= Rank(preferences[p], partners[p])) continue;
189 List<int> residents = Residents(market, partners, r);
190 if (residents.Count < market.Capacities[r] || Rank(preferences[r], p) < residents.Max(a => Rank(preferences[r], a)))
191 blocking.Add(new(p, r));
192 }
193 }
194 }
195 return (unacceptable, blocking);
196 }
197 static Candidate CandidateFor(Market market, int[] partners, int origin, int via)
198 {
199 int cost = 0, regret = 0;
200 foreach (int p in market.Proposers)
201 {
202 int r = partners[p];
203 if (r < 0) continue;
204 int a = Rank(market.Preferences[p], r), b = Rank(market.Preferences[r], p);
205 cost += a + b; regret = Math.Max(regret, Math.Max(a, b));
206 }
207 return new(partners, origin, via, cost, regret);
208 }
209 static List<int[]> Rotations(Market market, int[] partners)
210 {
211 int[] owners = Enumerable.Repeat(-1, market.Count).ToArray();
212 foreach (int p in market.Proposers) if (partners[p] >= 0) owners[partners[p]] = p;
213 int[] successors = Enumerable.Repeat(-1, market.ProposerCount).ToArray();
214 foreach (int p in market.Proposers)
215 {
216 if (partners[p] < 0) continue;
217 int[] preference = market.Preferences[p];
218 for (int i = Array.IndexOf(preference, partners[p]) + 1; i < preference.Length; i++)
219 {
220 int r = preference[i], owner = owners[r];
221 if (Array.IndexOf(market.Preferences[r], p) >= 0 &&
222 (owner < 0 || Rank(market.Preferences[r], p) < Rank(market.Preferences[r], owner)))
223 {
224 successors[p] = owner;
225 break;
226 }
227 }
228 }
229 List<int[]> rotations = [];
230 bool[] covered = new bool[market.ProposerCount];
231 foreach (int start in market.Proposers)
232 {
233 if (covered[start]) continue;
234 List<int> path = [];
235 bool[] seen = new bool[market.ProposerCount];
236 int current = start;
237 while (current >= 0 && !seen[current])
238 {
239 seen[current] = true; path.Add(current); current = successors[current];
240 }
241 if (current != start) continue;
242 rotations.Add(path.ToArray());
243 foreach (int p in path) covered[p] = true;
244 }
245 return rotations;
246 }
247 public static Enumeration Enumerate(Market market)
248 {
249 if (market.Receivers.Any(r => market.Capacities[r] != 1)) throw new CommandError("NOT ONE TO ONE");
250 List<Candidate> matchings = [CandidateFor(market, Deferred(market).Partners, 0, 0)];
251 List<string> rotations = [];
252 Dictionary<string, int> numbers = new(StringComparer.Ordinal);
253 for (int index = 0; index < matchings.Count; index++)
254 {
255 int[] partners = matchings[index].Partners;
256 foreach (int[] rotation in Rotations(market, partners))
257 {
258 string notation = string.Join(' ', rotation.Select(p => $"{market.Agents[p]}-{market.Agents[partners[p]]}"));
259 if (!numbers.TryGetValue(notation, out int number))
260 {
261 number = rotations.Count + 1;
262 numbers.Add(notation, number); rotations.Add(notation);
263 }
264 int[] next = (int[])partners.Clone();
265 for (int i = 0; i < rotation.Length; i++) next[rotation[i]] = partners[rotation[(i + 1) % rotation.Length]];
266 if (matchings.Any(candidate => candidate.Partners.SequenceEqual(next))) continue;
267 if (matchings.Count == 50) throw new CommandError("TOO MANY MATCHINGS");
268 matchings.Add(CandidateFor(market, next, index + 1, number));
269 }
270 }
271 return new(matchings, rotations);
272 }
273 public static List<string> Roommates(Market market, bool verbose)
274 {
275 for (int a = 0; a < market.Count; a++)
276 if (market.Preferences[a].Length != market.Count - 1) throw new CommandError($"INCOMPLETE {market.Agents[a]}");
277 List<int>[] lists = market.Preferences.Select(list => list.ToList()).ToArray();
278 int[] held = Enumerable.Repeat(-1, market.Count).ToArray();
279 Queue<int> queue = new(Enumerable.Range(0, market.Count));
280 List<string> trace = [];
281 int proposals = 0, failure = -1, rotations = 0;
282 void Remove(int a, int b) { lists[a].Remove(b); lists[b].Remove(a); }
283 void Table()
284 {
285 for (int a = 0; a < market.Count; a++) trace.Add($"TABLE {market.Agents[a]} {Program.ListText(lists[a].Select(b => market.Agents[b]))}");
286 }
287 int Empty() => Array.FindIndex(lists, list => list.Count == 0);
288 while (queue.Count > 0)
289 {
290 int x = queue.Dequeue(), y = lists[x][0];
291 proposals++;
292 if (held[y] >= 0) queue.Enqueue(held[y]);
293 held[y] = x;
294 int[] removed = lists[y].Skip(lists[y].IndexOf(x) + 1).ToArray();
295 foreach (int w in removed) Remove(y, w);
296 failure = Empty();
297 if (failure >= 0) break;
298 }
299 if (failure < 0)
300 {
301 if (verbose) Table();
302 while (lists.Any(list => list.Count > 1))
303 {
304 int p = Array.FindIndex(lists, list => list.Count > 1);
305 int[] seen = Enumerable.Repeat(-1, market.Count).ToArray();
306 List<int> path = [];
307 while (seen[p] < 0)
308 {
309 seen[p] = path.Count; path.Add(p);
310 int q = lists[p][1];
311 p = lists[q][^1];
312 }
313 int[] xs = path.Skip(seen[p]).ToArray(), ys = xs.Select(x => lists[x][0]).ToArray();
314 trace.Add($"ROTATION {++rotations} " + string.Join(' ', xs.Select((x, i) => $"{market.Agents[x]}-{market.Agents[ys[i]]}")));
315 List<Pair> removals = [];
316 for (int i = 0; i < xs.Length; i++)
317 {
318 int u = ys[(i + 1) % ys.Length];
319 foreach (int w in lists[u].Skip(lists[u].IndexOf(xs[i]) + 1)) removals.Add(new(u, w));
320 }
321 foreach (Pair pair in removals) Remove(pair.First, pair.Second);
322 if (verbose) Table();
323 failure = Empty();
324 if (failure >= 0) break;
325 }
326 }
327 List<string> output = [$"ROOMMATES {market.Name} proposals={proposals}", ..trace];
328 if (failure >= 0) output.Add($"NO STABLE MATCHING {market.Agents[failure]}");
329 else output.Add("MATCHING " + string.Join(' ', Enumerable.Range(0, market.Count)
330 .Where(a => a < lists[a][0]).Select(a => $"{market.Agents[a]}-{market.Agents[lists[a][0]]}")));
331 return output;
332 }
333 public static List<string> TradingCycles(Market market)
334 {
335 bool[] remaining = Enumerable.Repeat(true, market.Count).ToArray();
336 int[][] effective = Enumerable.Range(0, market.Count).Select(market.Effective).ToArray();
337 int[] allocation = Enumerable.Repeat(-1, market.Count).ToArray();
338 List<string> cycles = [];
339 while (remaining.Any(value => value))
340 {
341 int[] pointers = new int[market.Count];
342 for (int a = 0; a < market.Count; a++) if (remaining[a]) pointers[a] = effective[a].First(h => remaining[h]);
343 int[] seen = Enumerable.Repeat(-1, market.Count).ToArray();
344 List<int> path = [];
345 int current = Array.FindIndex(remaining, value => value);
346 while (seen[current] < 0)
347 {
348 seen[current] = path.Count; path.Add(current); current = pointers[current];
349 }
350 int[] cycle = path.Skip(seen[current]).ToArray();
351 foreach (int a in cycle) { allocation[a] = pointers[a]; remaining[a] = false; }
352 cycles.Add($"CYCLE {cycles.Count + 1} " + string.Join(' ', cycle.Select(a => $"{market.Agents[a]}:{market.Houses[allocation[a]]}")));
353 }
354 return [$"TTC {market.Name} cycles={cycles.Count}", ..cycles, $"ALLOCATION {market.Allocation(allocation)}"];
355 }

382 static string RankSummary(List<int> ranks)
383 {
384 int total = ranks.Sum(), first = ranks.Count(rank => rank == 1);
385 string average = "-", truncated = "-";
386 if (ranks.Count > 0)
387 {
388 int a = total, b = ranks.Count;
389 while (b != 0) (a, b) = (b, a % b);
390 average = $"{total / a}/{ranks.Count / a}";
391 int cents = total * 100 / ranks.Count;
392 truncated = $"{cents / 100}.{cents % 100:D2}";
393 }
394 return $"sum={total} average={average} truncated={truncated} first={first}";

416 static List<string> Check(Market market, string[] words)
417 {
418 if (market.Kind == MarketKind.Housing) throw new CommandError($"WRONG TYPE {market.Name}");
419 int[] partners = Enumerable.Repeat(-1, market.Kind == MarketKind.TwoSided ? market.ProposerCount : market.Count).ToArray();
420 bool[] used = new bool[market.Count];
421 int[] counts = new int[market.Count];
422 foreach (string word in words.Skip(2))
423 {
424 string[] pair = word.Split('-');
425 if (market.Kind == MarketKind.TwoSided)
426 {
427 int p = market.Proposer(pair[0]);
428 int r = market.Receiver(pair[1]);
429 if (used[p]) throw new CommandError($"DUPLICATE AGENT {pair[0]}");
430 if (counts[r] + 1 > market.Capacities[r]) throw new CommandError($"OVER CAPACITY {pair[1]}");
431 used[p] = true; counts[r]++; partners[p] = r;
432 }
433 else
434 {
435 int a = market.Agent(pair[0]);
436 int b = market.Agent(pair[1]);
437 if (used[a]) throw new CommandError($"DUPLICATE AGENT {pair[0]}");
438 if (a == b || used[b]) throw new CommandError($"DUPLICATE AGENT {pair[1]}");
439 used[a] = used[b] = true; partners[a] = b; partners[b] = a;
440 }
441 }
442 var (unacceptable, blocking) = Algorithms.Check(market, partners);
443 if (unacceptable.Count == 0 && blocking.Count == 0) return [$"CHECK {market.Name} STABLE"];
444 List<string> output = [$"CHECK {market.Name} UNSTABLE unacceptable={unacceptable.Count} blocking={blocking.Count}"];
445 foreach (Pair pair in unacceptable) output.Add($"UNACCEPTABLE {market.Agents[pair.First]} {market.Agents[pair.Second]}");
446 foreach (Pair pair in blocking) output.Add($"BLOCKING {market.Agents[pair.First]} {market.Agents[pair.Second]}");
447 return output;
448 }

488 static List<string> Manipulate(Market market, int agent)
489 {
490 int[] truth = market.Preferences[agent];
491 int b0 = Algorithms.Boston(market).Partners[agent], g0 = Algorithms.Deferred(market).Partners[agent];
492 List<string> output = [$"MANIPULATE {market.Name} {market.Agents[agent]} boston={market.Partner(b0)} gs={market.Partner(g0)} swaps={Math.Max(0, truth.Length - 1)}"];
493 for (int i = 0; i + 1 < truth.Length; i++)
494 {
495 int[][] preferences = (int[][])market.Preferences.Clone();
496 preferences[agent] = (int[])truth.Clone();
497 (preferences[agent][i], preferences[agent][i + 1]) = (preferences[agent][i + 1], preferences[agent][i]);
498 int b = Algorithms.Boston(market, preferences).Partners[agent];
499 int g = Algorithms.Deferred(market, preferences: preferences).Partners[agent];
500 string swap = $"{i + 1} {market.Agents[truth[i]]} {market.Agents[truth[i + 1]]}";
501 if (Algorithms.Rank(truth, b) < Algorithms.Rank(truth, b0)) output.Add($"IMPROVE BOSTON {swap} result={market.Partner(b)}");
502 if (Algorithms.Rank(truth, g) < Algorithms.Rank(truth, g0)) output.Add($"IMPROVE GS {swap} result={market.Partner(g)}");
503 }
504 if (output.Count == 1) output.Add("NO IMPROVEMENT");
505 return output;
506 }

525 static List<string> Execute(string line)
526 {
527 string[] words = line.Split(' ');
528 string command = words[0], name = words[1];
529 switch (command)
530 {
531 case "MARKET":
532 {
533 NewName(name);
534 int bar = Array.IndexOf(words, "|");
535 string[] proposers = words[2..bar], receivers = words[(bar + 1)..];
536 string[] agents = [..proposers, ..receivers];
537 DistinctAgents(agents);
538 Markets.Add(name, new(name, MarketKind.TwoSided, agents, proposers.Length));
539 return [$"MARKET {name} proposers={proposers.Length} receivers={receivers.Length}"];
540 }
541 case "ROOMS":
542 {
543 NewName(name);
544 string[] agents = words[2..];
545 DistinctAgents(agents);
546 Markets.Add(name, new(name, MarketKind.Roommates, agents));
547 return [$"ROOMS {name} agents={agents.Length}"];
548 }
549 case "HOUSES":
550 {
551 NewName(name);
552 List<string> agents = [], houses = [];
553 HashSet<string> agentNames = new(StringComparer.Ordinal), houseNames = new(StringComparer.Ordinal);
554 foreach (string word in words.Skip(2))
555 {
556 string[] pair = word.Split(':');
557 if (!agentNames.Add(pair[0])) throw new CommandError($"DUPLICATE AGENT {pair[0]}");
558 if (!houseNames.Add(pair[1])) throw new CommandError($"DUPLICATE HOUSE {pair[1]}");
559 agents.Add(pair[0]); houses.Add(pair[1]);
560 }
561 Markets.Add(name, new(name, MarketKind.Housing, agents.ToArray(), houses: houses.ToArray()));
562 return [$"HOUSES {name} agents={agents.Count}"];
563 }
564 case "RANDOM":
565 {
566 NewName(name);
567 int size = Integer(words[2]);
568 long seed = long.Parse(words[3], CultureInfo.InvariantCulture), state = seed;
569 string[] agents = [..Enumerable.Range(1, size).Select(i => $"P{i}"), ..Enumerable.Range(1, size).Select(i => $"R{i}")];
570 Market market = new(name, MarketKind.TwoSided, agents, size);
571 for (int a = 0; a < market.Count; a++)
572 {
573 int[] preferences = (a < size ? market.Receivers : market.Proposers).ToArray();
574 for (int i = size - 1; i > 0; i--)
575 {
576 state = 48271 * state % 2147483647;
577 int j = (int)(state % (i + 1));
578 (preferences[i], preferences[j]) = (preferences[j], preferences[i]);
579 }
580 market.Preferences[a] = preferences;
581 }
582 Markets.Add(name, market);
583 return [$"RANDOM {name} n={size} seed={seed}"];
584 }
585 case "PREF":
586 {
587 Market market = Get(name);
588 int agent = market.Agent(words[2]);
589 List<int> choices = [];
590 HashSet<int> seen = [];
591 foreach (string word in words.Skip(3))
592 {
593 int choice = market.Choice(agent, word);
594 if (!seen.Add(choice)) throw new CommandError($"DUPLICATE CHOICE {word}");
595 choices.Add(choice);
596 }
597 market.Preferences[agent] = choices.ToArray();
598 if (market.Kind == MarketKind.TwoSided) market.ClearMatching();
599 return [$"PREF {name} {words[2]} length={choices.Count}"];
600 }
601 case "CAPACITY":
602 {
603 Market market = Get(name, MarketKind.TwoSided);
604 int receiver = market.Receiver(words[2]), capacity = Integer(words[3]);
605 market.Capacities[receiver] = capacity;
606 market.ClearMatching();
607 return [$"CAPACITY {name} {words[2]} {capacity}"];
608 }
609 case "SHOW": return Show(Get(name));
610 case "CHECK": return Check(Get(name), words);
611 case "GS": case "GSR": case "BOSTON":
612 {
613 Market market = Get(name, MarketKind.TwoSided);
614 MatchingResult result = command == "BOSTON" ? Algorithms.Boston(market)
615 : Algorithms.Deferred(market, command == "GSR", words.Length == 3);
616 market.SetMatching(result.Partners, command);
617 return [$"{command} {name} {(command == "BOSTON" ? "applications" : "proposals")}={result.Proposals}",
618 ..result.Log, $"MATCHING {market.Pairs(result.Partners)}"];
619 }
620 case "LATTICE": case "EGALITARIAN": case "MINREGRET":
621 {
622 Market market = Get(name, MarketKind.TwoSided);
623 Enumeration result = Algorithms.Enumerate(market);
624 if (command == "LATTICE")
625 {
626 List<string> output = [$"LATTICE {name} matchings={result.Matchings.Count} rotations={result.Rotations.Count}"];
627 for (int i = 0; i < result.Matchings.Count; i++)
628 {
629 Candidate matching = result.Matchings[i];
630 string origin = matching.Origin == 0 ? "-" : matching.Origin.ToString(CultureInfo.InvariantCulture);
631 string via = matching.Via == 0 ? "-" : matching.Via.ToString(CultureInfo.InvariantCulture);
632 output.Add($"MATCHING {i + 1} from={origin} via={via} cost={matching.Cost} regret={matching.Regret} {market.Pairs(matching.Partners)}");
633 }
634 for (int i = 0; i < result.Rotations.Count; i++) output.Add($"ROTATION {i + 1} {result.Rotations[i]}");
635 return output;
636 }
637 int best = 0;
638 for (int i = 1; i < result.Matchings.Count; i++)
639 if (command == "EGALITARIAN" ? result.Matchings[i].Cost < result.Matchings[best].Cost
640 : result.Matchings[i].Regret < result.Matchings[best].Regret) best = i;
641 Candidate selected = result.Matchings[best];
642 market.SetMatching(selected.Partners, command);
643 return [$"{command} {name} matching={best + 1} {(command == "EGALITARIAN" ? $"cost={selected.Cost}" : $"regret={selected.Regret}")}",
644 $"MATCHING {market.Pairs(selected.Partners)}"];
645 }
646 case "HR": return Hospitals(Get(name, MarketKind.TwoSided));
647 case "ROOMMATES": return Algorithms.Roommates(Get(name, MarketKind.Roommates), words.Length == 3);
648 case "TTC": return Algorithms.TradingCycles(Get(name, MarketKind.Housing));
649 case "SERIAL": return Serial(Get(name, MarketKind.Housing), words);
650 case "MANIPULATE":
651 {
652 Market market = Get(name, MarketKind.TwoSided);
653 return Manipulate(market, market.Proposer(words[2]));
654 }
655 case "STATS": return Stats(Get(name, MarketKind.TwoSided));
656 default: throw new InvalidOperationException("Unknown command in well-formed input");
657 }
658 }
659 public static void Main()
660 {
661 int count = Integer(Console.ReadLine()!);
662 for (int i = 0; i < count; i++)
663 {
664 string line = Console.ReadLine()!;
665 try { foreach (string output in Execute(line)) Console.WriteLine(output); }
666 catch (CommandError error) { Console.WriteLine(error.Message); }
667 }
668 }
