102MatchingResult deferred(Market market, {bool reverse = false, bool verbose = false, List<List<int>>? preferences}) {
103 final lists = preferences ?? market.preferences;
104 final next = List.filled(market.count, 0), counts = List.filled(market.count, 0);
105 final queued = List.filled(market.count, false);
106 final holding = List.generate(market.count, (_) => <int>[]);
107 final queue = Queue<int>();
108 bool able(int agent) => counts[agent] < market.capacities[agent] && next[agent] < lists[agent].length;
109 void enqueue(int agent) {
110 if (able(agent) && !queued[agent]) { queue.addLast(agent); queued[agent] = true; }
111 }
112 for (final agent in reverse ? market.receivers : market.proposers) { enqueue(agent); }
113 var proposals = 0;
114 final log = <String>[];
115 while (queue.isNotEmpty) {
116 final x = queue.removeFirst();
117 queued[x] = false;
118 final y = lists[x][next[x]++];
119 proposals++;
120 var action = 'REJECT';
121 if (lists[y].contains(x)) {
122 if (holding[y].length < market.capacities[y]) {
123 holding[y].add(x); counts[x]++;
124 action = 'HOLD';
125 } else {
126 var worst = 0;
127 for (var i = 1; i < holding[y].length; i++) {
128 if (rank(lists[y], holding[y][i]) > rank(lists[y], holding[y][worst])) worst = i;
129 }
130 final rejected = holding[y][worst];
131 if (rank(lists[y], x) < rank(lists[y], rejected)) {
132 holding[y][worst] = x;
133 counts[rejected]--; counts[x]++;
134 action = 'HOLD REJECT ${market.agents[rejected]}';
135 enqueue(rejected);
136 }
137 }
138 }
139 if (verbose) log.add('PROPOSE ${market.agents[x]} ${market.agents[y]} $action');
140 enqueue(x);
141 }
142 final partners = List.filled(market.proposerCount, -1);
143 for (final y in reverse ? market.proposers : market.receivers) {
144 for (final x in holding[y]) { partners[reverse ? y : x] = reverse ? x : y; }
145 }
146 return MatchingResult(partners, proposals, log);
147}

149MatchingResult boston(Market market, {List<List<int>>? preferences}) {
150 final lists = preferences ?? market.preferences;
151 final partners = List.filled(market.proposerCount, -1);
152 final remaining = List.of(market.capacities);
153 final rounds = market.proposers.map((p) => lists[p].length).reduce(math.max);
154 var applications = 0;
155 for (var round = 0; round < rounds; round++) {
156 final applicants = List.generate(market.count, (_) => <int>[]);
157 for (final p in market.proposers) {
158 if (partners[p] >= 0 || round >= lists[p].length) continue;
159 applicants[lists[p][round]].add(p);
160 applications++;
161 }
162 for (final r in market.receivers) {
163 applicants[r].sort((a, b) => rank(lists[r], a).compareTo(rank(lists[r], b)));
164 for (final p in applicants[r]) {
165 if (remaining[r] == 0 || !lists[r].contains(p)) continue;
166 partners[p] = r; remaining[r]--;
167 }
168 }
169 }
170 return MatchingResult(partners, applications, []);
171}

176({List<(int, int)> unacceptable, List<(int, int)> blocking}) checkMatching(Market market, List<int> partners) {
177 final unacceptable = <(int, int)>[], blocking = <(int, int)>[];
178 final lists = market.preferences;
179 bool acceptable(int a, int b) => lists[a].contains(b) && lists[b].contains(a);
180 if (market.kind == MarketKind.roommates) {
181 for (var a = 0; a < market.count; a++) {
182 for (var b = a + 1; b < market.count; b++) {
183 if (partners[a] == b) {
184 if (!acceptable(a, b)) unacceptable.add((a, b));
185 } else if (acceptable(a, b) && rank(lists[a], b) < rank(lists[a], partners[a]) &&
186 rank(lists[b], a) < rank(lists[b], partners[b])) {
187 blocking.add((a, b));
188 }
189 }
190 }
191 } else {
192 for (final p in market.proposers) {
193 if (partners[p] >= 0 && !acceptable(p, partners[p])) unacceptable.add((p, partners[p]));
194 for (final r in market.receivers) {
195 if (partners[p] == r || !acceptable(p, r) || rank(lists[p], r) >= rank(lists[p], partners[p])) continue;
196 final held = residents(market, partners, r);
197 if (held.length < market.capacities[r] || rank(lists[r], p) < held.map((a) => rank(lists[r], a)).reduce(math.max)) {
198 blocking.add((p, r));
199 }
200 }
201 }
202 }
203 return (unacceptable: unacceptable, blocking: blocking);
204}

206Candidate candidateFor(Market market, List<int> partners, int origin, int via) {
207 var cost = 0, regret = 0;
208 for (final p in market.proposers) {
209 final r = partners[p];
210 if (r < 0) continue;
211 final a = rank(market.preferences[p], r), b = rank(market.preferences[r], p);
212 cost += a + b;
213 regret = math.max(regret, math.max(a, b));
214 }
215 return Candidate(partners, origin, via, cost, regret);
216}

218List<List<int>> rotations(Market market, List<int> partners) {
219 final owners = List.filled(market.count, -1), successors = List.filled(market.proposerCount, -1);
220 for (final p in market.proposers) { if (partners[p] >= 0) owners[partners[p]] = p; }
221 for (final p in market.proposers) {
222 if (partners[p] < 0) continue;
223 final preference = market.preferences[p];
224 for (var i = preference.indexOf(partners[p]) + 1; i < preference.length; i++) {
225 final r = preference[i], owner = owners[r];
226 if (market.preferences[r].contains(p) &&
227 (owner < 0 || rank(market.preferences[r], p) < rank(market.preferences[r], owner))) {
228 successors[p] = owner;
229 break;
230 }
231 }
232 }
233 final result = <List<int>>[];
234 final covered = List.filled(market.proposerCount, false);
235 for (final start in market.proposers) {
236 if (covered[start]) continue;
237 final path = <int>[], seen = <int>{};
238 var current = start;
239 while (current >= 0 && seen.add(current)) { path.add(current); current = successors[current]; }
240 if (current != start) continue;
241 result.add(path);
242 for (final p in path) { covered[p] = true; }
243 }
244 return result;
245}

247Enumeration enumerate(Market market) {
248 if (market.receivers.any((r) => market.capacities[r] != 1)) throw const CommandError('NOT ONE TO ONE');
249 final matchings = [candidateFor(market, deferred(market).partners, 0, 0)];
250 final notations = <String>[], numbers = <String, int>{};
251 for (var index = 0; index < matchings.length; index++) {
252 final partners = matchings[index].partners;
253 for (final rotation in rotations(market, partners)) {
254 final notation = rotation.map((p) => '${market.agents[p]}-${market.agents[partners[p]]}').join(' ');
255 final number = numbers.putIfAbsent(notation, () { notations.add(notation); return notations.length; });
256 final next = List.of(partners);
257 for (var i = 0; i < rotation.length; i++) { next[rotation[i]] = partners[rotation[(i + 1) % rotation.length]]; }
258 if (matchings.any((candidate) => equalLists(candidate.partners, next))) continue;
259 if (matchings.length == 50) throw const CommandError('TOO MANY MATCHINGS');
260 matchings.add(candidateFor(market, next, index + 1, number));
261 }
262 }
263 return Enumeration(matchings, notations);
264}

266List<String> roommates(Market market, bool verbose) {
267 for (var a = 0; a < market.count; a++) {
268 if (market.preferences[a].length != market.count - 1) throw CommandError('INCOMPLETE ${market.agents[a]}');
269 }
270 final lists = market.preferences.map((list) => List<int>.of(list)).toList();
271 final held = List.filled(market.count, -1);
272 final queue = Queue<int>.of(Iterable<int>.generate(market.count));
273 final trace = <String>[];
274 var proposals = 0, failure = -1, rotationCount = 0;
275 void remove(int a, int b) { lists[a].remove(b); lists[b].remove(a); }
276 void table() {
277 for (var a = 0; a < market.count; a++) {
278 trace.add('TABLE ${market.agents[a]} ${listText(lists[a].map((b) => market.agents[b]))}');
279 }
280 }
281 int empty() => lists.indexWhere((list) => list.isEmpty);
282 while (queue.isNotEmpty) {
283 final x = queue.removeFirst(), y = lists[x].first;
284 proposals++;
285 if (held[y] >= 0) queue.addLast(held[y]);
286 held[y] = x;
287 final removed = lists[y].sublist(lists[y].indexOf(x) + 1);
288 for (final w in removed) { remove(y, w); }
289 failure = empty();
290 if (failure >= 0) break;
291 }
292 if (failure < 0) {
293 if (verbose) table();
294 while (lists.any((list) => list.length > 1)) {
295 var p = lists.indexWhere((list) => list.length > 1);
296 final seen = List.filled(market.count, -1), path = <int>[];
297 while (seen[p] < 0) {
298 seen[p] = path.length; path.add(p);
299 final q = lists[p][1];
300 p = lists[q].last;
301 }
302 final xs = path.sublist(seen[p]);
303 final ys = xs.map((x) => lists[x].first).toList();
304 rotationCount++;
305 trace.add('ROTATION $rotationCount ${Iterable.generate(xs.length, (i) => '${market.agents[xs[i]]}-${market.agents[ys[i]]}').join(' ')}');
306 final removals = <(int, int)>[];
307 for (var i = 0; i < xs.length; i++) {
308 final u = ys[(i + 1) % ys.length];
309 for (final w in lists[u].skip(lists[u].indexOf(xs[i]) + 1)) { removals.add((u, w)); }
310 }
311 for (final (a, b) in removals) { remove(a, b); }
312 if (verbose) table();
313 failure = empty();
314 if (failure >= 0) break;
315 }
316 }
317 final output = ['ROOMMATES ${market.name} proposals=$proposals', ...trace];
318 if (failure >= 0) {
319 output.add('NO STABLE MATCHING ${market.agents[failure]}');
320 } else {
321 final pairs = Iterable<int>.generate(market.count).where((a) => a < lists[a].first)
322 .map((a) => '${market.agents[a]}-${market.agents[lists[a].first]}');
323 output.add('MATCHING ${pairs.join(' ')}');
324 }
325 return output;
326}

328List<String> tradingCycles(Market market) {
329 final remaining = List.filled(market.count, true);
330 final effective = List.generate(market.count, market.effective);
331 final assigned = List.filled(market.count, -1);
332 final cycles = <String>[];
333 while (remaining.any((value) => value)) {
334 final pointers = List.filled(market.count, -1);
335 for (var a = 0; a < market.count; a++) {
336 if (remaining[a]) pointers[a] = effective[a].firstWhere((house) => remaining[house]);
337 }
338 final seen = List.filled(market.count, -1), path = <int>[];
339 var current = remaining.indexOf(true);
340 while (seen[current] < 0) {
341 seen[current] = path.length; path.add(current); current = pointers[current];
342 }
343 final cycle = path.sublist(seen[current]);
344 for (final a in cycle) { assigned[a] = pointers[a]; remaining[a] = false; }
345 cycles.add('CYCLE ${cycles.length + 1} ${cycle.map((a) => '${market.agents[a]}:${market.houses[assigned[a]]}').join(' ')}');
346 }
347 return ['TTC ${market.name} cycles=${cycles.length}', ...cycles, 'ALLOCATION ${market.allocation(assigned)}'];
348}

350String rankSummary(List<int> ranks) {
351 final total = ranks.fold(0, (sum, rank) => sum + rank);
352 var average = '-', truncated = '-';
353 if (ranks.isNotEmpty) {
354 final divisor = total.gcd(ranks.length), cents = total * 100 ~/ ranks.length;
355 average = '${total ~/ divisor}/${ranks.length ~/ divisor}';
356 truncated = '${cents ~/ 100}.${(cents % 100).toString().padLeft(2, '0')}';
357 }
358 return 'sum=$total average=$average truncated=$truncated first=${ranks.where((rank) => rank == 1).length}';
359}

463 List<String> manipulate(Market market, int agent) {
464 final truth = market.preferences[agent];
465 final b0 = boston(market).partners[agent], g0 = deferred(market).partners[agent];
466 final output = ['MANIPULATE ${market.name} ${market.agents[agent]} boston=${market.partner(b0)} gs=${market.partner(g0)} swaps=${math.max(0, truth.length - 1)}'];
467 for (var i = 0; i + 1 < truth.length; i++) {
468 final preferences = List.of(market.preferences);
469 preferences[agent] = List.of(truth);
470 final first = preferences[agent][i];
471 preferences[agent][i] = preferences[agent][i + 1];
472 preferences[agent][i + 1] = first;
473 final b = boston(market, preferences: preferences).partners[agent];
474 final g = deferred(market, preferences: preferences).partners[agent];
475 final swap = '${i + 1} ${market.agents[truth[i]]} ${market.agents[truth[i + 1]]}';
476 if (rank(truth, b) < rank(truth, b0)) output.add('IMPROVE BOSTON $swap result=${market.partner(b)}');
477 if (rank(truth, g) < rank(truth, g0)) output.add('IMPROVE GS $swap result=${market.partner(g)}');
478 }
479 if (output.length == 1) output.add('NO IMPROVEMENT');
480 return output;

498 List<String> execute(String line) {
499 final words = line.split(' ');
500 final command = words[0], name = words[1];
501 switch (command) {
502 case 'MARKET':
503 newName(name);
504 final bar = words.indexOf('|');
505 final proposers = words.sublist(2, bar), receivers = words.sublist(bar + 1);
506 final agents = [...proposers, ...receivers];
507 distinctAgents(agents);
508 markets[name] = Market(name, MarketKind.twoSided, agents, proposerCount: proposers.length);
509 return ['MARKET $name proposers=${proposers.length} receivers=${receivers.length}'];
510 case 'ROOMS':
511 newName(name);
512 final agents = words.sublist(2);
513 distinctAgents(agents);
514 markets[name] = Market(name, MarketKind.roommates, agents);
515 return ['ROOMS $name agents=${agents.length}'];
516 case 'HOUSES':
517 newName(name);
518 final agents = <String>[], houses = <String>[];
519 final agentNames = <String>{}, houseNames = <String>{};
520 for (final word in words.skip(2)) {
521 final pair = word.split(':');
522 if (!agentNames.add(pair[0])) throw CommandError('DUPLICATE AGENT ${pair[0]}');
523 if (!houseNames.add(pair[1])) throw CommandError('DUPLICATE HOUSE ${pair[1]}');
524 agents.add(pair[0]); houses.add(pair[1]);
525 }
526 markets[name] = Market(name, MarketKind.housing, agents, houses: houses);
527 return ['HOUSES $name agents=${agents.length}'];
528 case 'RANDOM':
529 newName(name);
530 final size = int.parse(words[2]), seed = int.parse(words[3]);
531 var state = seed;
532 final agents = [for (var i = 1; i <= size; i++) 'P$i', for (var i = 1; i <= size; i++) 'R$i'];
533 final market = Market(name, MarketKind.twoSided, agents, proposerCount: size);
534 for (var a = 0; a < market.count; a++) {
535 final preferences = (a < size ? market.receivers : market.proposers).toList();
536 for (var i = size - 1; i > 0; i--) {
537 state = 48271 * state % 2147483647;
538 final j = state % (i + 1);
539 final first = preferences[i];
540 preferences[i] = preferences[j];
541 preferences[j] = first;
542 }
543 market.preferences[a] = preferences;
544 }
545 markets[name] = market;
546 return ['RANDOM $name n=$size seed=$seed'];
547 case 'PREF':
548 final market = get(name);
549 final agent = market.agent(words[2]);
550 final choices = <int>[], seen = <int>{};
551 for (final word in words.skip(3)) {
552 final choice = market.choice(agent, word);
553 if (!seen.add(choice)) throw CommandError('DUPLICATE CHOICE $word');
554 choices.add(choice);
555 }
556 market.preferences[agent] = choices;
557 if (market.kind == MarketKind.twoSided) market.clearMatching();
558 return ['PREF $name ${words[2]} length=${choices.length}'];
559 case 'CAPACITY':
560 final market = get(name, MarketKind.twoSided);
561 final receiver = market.receiver(words[2]), capacity = int.parse(words[3]);
562 market.capacities[receiver] = capacity;
563 market.clearMatching();
564 return ['CAPACITY $name ${words[2]} $capacity'];
565 case 'SHOW': return show(get(name));
566 case 'CHECK': return check(get(name), words);
567 case 'GS':
568 case 'GSR':
569 case 'BOSTON':
570 final market = get(name, MarketKind.twoSided);
571 final result = command == 'BOSTON' ? boston(market)
572 : deferred(market, reverse: command == 'GSR', verbose: words.length == 3);
573 market.setMatching(result.partners, command);
574 return ['$command $name ${command == 'BOSTON' ? 'applications' : 'proposals'}=${result.proposals}',
575 ...result.log, 'MATCHING ${market.pairs(result.partners)}'];
576 case 'LATTICE':
577 case 'EGALITARIAN':
578 case 'MINREGRET':
579 final market = get(name, MarketKind.twoSided);
580 final result = enumerate(market);
581 if (command == 'LATTICE') {
582 final output = ['LATTICE $name matchings=${result.matchings.length} rotations=${result.rotations.length}'];
583 for (var i = 0; i < result.matchings.length; i++) {
584 final matching = result.matchings[i];
585 final origin = matching.origin == 0 ? '-' : '${matching.origin}';
586 final via = matching.via == 0 ? '-' : '${matching.via}';
587 output.add('MATCHING ${i + 1} from=$origin via=$via cost=${matching.cost} regret=${matching.regret} ${market.pairs(matching.partners)}');
588 }
589 for (var i = 0; i < result.rotations.length; i++) { output.add('ROTATION ${i + 1} ${result.rotations[i]}'); }
590 return output;
591 }
592 var best = 0;
593 for (var i = 1; i < result.matchings.length; i++) {
594 if (command == 'EGALITARIAN' ? result.matchings[i].cost < result.matchings[best].cost
595 : result.matchings[i].regret < result.matchings[best].regret) best = i;
596 }
597 final selected = result.matchings[best];
598 market.setMatching(selected.partners, command);
599 final criterion = command == 'EGALITARIAN' ? 'cost=${selected.cost}' : 'regret=${selected.regret}';
600 return ['$command $name matching=${best + 1} $criterion', 'MATCHING ${market.pairs(selected.partners)}'];
601 case 'HR': return hospitals(get(name, MarketKind.twoSided));
602 case 'ROOMMATES': return roommates(get(name, MarketKind.roommates), words.length == 3);
603 case 'TTC': return tradingCycles(get(name, MarketKind.housing));
604 case 'SERIAL': return serial(get(name, MarketKind.housing), words);
605 case 'MANIPULATE':
606 final market = get(name, MarketKind.twoSided);
607 return manipulate(market, market.proposer(words[2]));
608 case 'STATS': return stats(get(name, MarketKind.twoSided));
609 default: throw StateError('Unknown command in well-formed input');
610 }
611 }
