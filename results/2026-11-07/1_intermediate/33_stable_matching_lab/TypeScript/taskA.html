44const sequence = (start: number, end: number): number[] => Array.from({ length: end - start }, (_, i) => start + i);
45const rank = (preference: readonly number[], partner: number): number => {
46 const index = preference.indexOf(partner);

49const listText = (names: readonly string[], indices: readonly number[]): string => indices.length ? indices.map(i => names[i]).join(' ') : '-';
50const partnerName = (market: Market, partner: number): string => partner < 0 ? '*' : market.names[partner];
51const pairText = (market: Market, matching: Matching): string => matching.map((partner, p) => `${market.names[p]}-${partnerName(market, partner)}`).join(' ');

56class Market {
57 readonly name: string;
58 readonly kind: MarketKind;
59 readonly names: readonly string[];
60 readonly proposers: number;
61 readonly houses: readonly string[];
62 preferences: (readonly number[])[];
63 capacities: number[];
64 current: Saved | null = null;

66 constructor(name: string, kind: MarketKind, names: readonly string[], proposers = 0, houses: readonly string[] = []) {
67 this.name = name;
68 this.kind = kind;
69 this.names = names.slice();
70 this.proposers = proposers;
71 this.houses = houses.slice();
72 this.preferences = names.map(() => []);
73 this.capacities = names.map(() => 1);
74 this.current = null;
75 }
76 requireKind(kind: MarketKind): void { if (this.kind !== kind) fail(`WRONG TYPE ${this.name}`); }
77 agent(name: string, side: Side | null = null): number {
78 const index = this.names.indexOf(name);
79 if (index < 0) fail(`UNKNOWN AGENT ${name}`);
80 if (side === 'PROPOSER' && index >= this.proposers) fail(`NOT PROPOSER ${name}`);
81 if (side === 'RECEIVER' && index < this.proposers) fail(`NOT RECEIVER ${name}`);
82 return index;
83 }
84 trial(proposer: number, preference: readonly number[]): Market {
85 const result = new Market(this.name, this.kind, this.names, this.proposers, this.houses);
86 result.preferences = this.preferences.map((list, i) => (i === proposer ? preference : list).slice());
87 result.capacities = this.capacities.slice();
88 return result;
89 }
90 save(source: Source, matching: Matching): void { this.current = { source, matching: matching.slice() }; }
91 effective(agent: number): readonly number[] {
92 const preference = this.preferences[agent], own = preference.indexOf(agent);
93 return own < 0 ? [...preference, agent] : preference.slice(0, own + 1);
94 }

97function deferred(market: Market, reverse = false, verbose = false): DeferredRun {
98 const count = market.names.length, next = Array<number>(count).fill(0), counts = Array<number>(count).fill(0);
99 const holding: number[][] = market.names.map(() => []), queued = Array<boolean>(count).fill(false);
100 const able = (agent: number): boolean => counts[agent] < market.capacities[agent] && next[agent] < market.preferences[agent].length;
101 const proposing = reverse ? sequence(market.proposers, count) : sequence(0, market.proposers);
102 const queue = proposing.filter(able), history: string[] = [];
103 for (const agent of queue) queued[agent] = true;
104 let proposals = 0;
105 for (let head = 0; head < queue.length; head++) {
106 const proposer = queue[head];
107 queued[proposer] = false;
108 const receiver = market.preferences[proposer][next[proposer]++], preference = market.preferences[receiver];
109 proposals++;
110 let action = 'REJECT';
111 if (preference.includes(proposer)) {
112 if (holding[receiver].length < market.capacities[receiver]) {
113 holding[receiver].push(proposer);
114 counts[proposer]++;
115 action = 'HOLD';
116 } else {
117 let worst = 0;
118 for (let i = 1; i < holding[receiver].length; i++) {
119 if (rank(preference, holding[receiver][i]) > rank(preference, holding[receiver][worst])) worst = i;
120 }
121 const rejected = holding[receiver][worst];
122 if (rank(preference, proposer) < rank(preference, rejected)) {
123 holding[receiver][worst] = proposer;
124 counts[proposer]++;
125 counts[rejected]--;
126 action = `HOLD REJECT ${market.names[rejected]}`;
127 if (able(rejected) && !queued[rejected]) { queue.push(rejected); queued[rejected] = true; }
128 }
129 }
130 }
131 if (verbose) history.push(`PROPOSE ${market.names[proposer]} ${market.names[receiver]} ${action}`);
132 if (able(proposer)) { queue.push(proposer); queued[proposer] = true; }
133 }
134 const matching = Array<number>(market.proposers).fill(-1);
135 const receiving = reverse ? sequence(0, market.proposers) : sequence(market.proposers, count);
136 for (const receiver of receiving) {
137 for (const proposer of holding[receiver]) {
138 if (reverse) matching[receiver] = proposer;
139 else matching[proposer] = receiver;
140 }
141 }
142 return { matching, proposals, history };
143}
144function boston(market: Market): BostonRun {
145 const matching = Array<number>(market.proposers).fill(-1), remaining = market.capacities.slice();
146 const rounds = Math.max(...market.preferences.slice(0, market.proposers).map(list => list.length));
147 let applications = 0;
148 for (let round = 0; round < rounds; round++) {
149 const applied = Array<number>(market.proposers).fill(-1);
150 for (let proposer = 0; proposer < market.proposers; proposer++) {
151 if (matching[proposer] < 0 && round < market.preferences[proposer].length) {
152 applied[proposer] = market.preferences[proposer][round];
153 applications++;
154 }
155 }
156 for (let receiver = market.proposers; receiver < market.names.length; receiver++) {
157 for (const proposer of market.preferences[receiver]) {
158 if (remaining[receiver] > 0 && applied[proposer] === receiver) {
159 matching[proposer] = receiver;
160 remaining[receiver]--;
161 }
162 }
163 }
164 }
165 return { matching, applications };
166}
167function exposedRotations(market: Market, matching: Matching): number[][] {
168 const owner = Array<number>(market.names.length).fill(-1), successor = Array<number>(market.proposers).fill(-1);
169 matching.forEach((receiver, proposer) => { if (receiver >= 0) owner[receiver] = proposer; });
170 for (let proposer = 0; proposer < market.proposers; proposer++) {
171 if (matching[proposer] < 0) continue;
172 const preference = market.preferences[proposer];
173 for (const receiver of preference.slice(rank(preference, matching[proposer]))) {
174 const ranking = market.preferences[receiver];
175 if (ranking.includes(proposer) && (owner[receiver] < 0 || rank(ranking, proposer) < rank(ranking, owner[receiver]))) {
176 successor[proposer] = owner[receiver];
177 break;
178 }
179 }
180 }
181 const covered = new Set<number>(), rotations: number[][] = [];
182 for (let start = 0; start < market.proposers; start++) {
183 if (covered.has(start)) continue;
184 const path: number[] = [];
185 let point = start;
186 while (point >= 0 && !path.includes(point)) { path.push(point); point = successor[point]; }
187 if (point === start) {
188 path.forEach(proposer => covered.add(proposer));
189 rotations.push(path);
190 }
191 }
192 return rotations;
193}
194function metrics(market: Market, matching: Matching): Metrics {
195 let cost = 0, regret = 0;
196 matching.forEach((receiver, proposer) => {
197 if (receiver < 0) return;
198 const first = rank(market.preferences[proposer], receiver), second = rank(market.preferences[receiver], proposer);
199 cost += first + second;
200 regret = Math.max(regret, first, second);
201 });
202 return { cost, regret };
203}
204function enumerate(market: Market): Lattice {
205 if (market.capacities.slice(market.proposers).some(capacity => capacity !== 1)) fail('NOT ONE TO ONE');
206 const initial = deferred(market).matching;
207 const records: MatchingRecord[] = [{ matching: initial, origin: null, via: null }];
208 const rotations: string[] = [], seen = new Set([initial.join(',')]);
209 for (let index = 0; index < records.length; index++) {
210 const { matching } = records[index];
211 for (const rotation of exposedRotations(market, matching)) {
212 const notation = rotationText(market, matching, rotation);
213 let identifier = rotations.indexOf(notation);
214 if (identifier < 0) { identifier = rotations.length; rotations.push(notation); }
215 const following = matching.slice();
216 rotation.forEach((proposer, i) => { following[proposer] = matching[rotation[(i + 1) % rotation.length]]; });
217 const key = following.join(',');
218 if (seen.has(key)) continue;
219 if (records.length === 50) fail('TOO MANY MATCHINGS');
220 seen.add(key);
221 records.push({ matching: following, origin: index + 1, via: identifier + 1 });
222 }
223 }
224 return { records, rotations };
225}

227function checkPairs(market: Market, pairs: readonly string[]): string[] {
228 if (market.kind === 'HOUSES') fail(`WRONG TYPE ${market.name}`);
229 const matching = Array<number>(market.names.length).fill(-1), counts = Array<number>(market.names.length).fill(0);
230 for (const word of pairs) {
231 const [x, y] = word.split('-');
232 const first = market.agent(x, market.kind === 'MARKET' ? 'PROPOSER' : null);
233 const second = market.agent(y, market.kind === 'MARKET' ? 'RECEIVER' : null);
234 if (matching[first] >= 0) fail(`DUPLICATE AGENT ${x}`);
235 if (market.kind === 'MARKET') {
236 if (++counts[second] > market.capacities[second]) fail(`OVER CAPACITY ${y}`);
237 } else {
238 if (first === second || matching[second] >= 0) fail(`DUPLICATE AGENT ${y}`);
239 matching[second] = first;
240 }
241 matching[first] = second;
242 }
243 const unacceptable: string[] = [], blocking: string[] = [];
244 if (market.kind === 'MARKET') {
245 for (let proposer = 0; proposer < market.proposers; proposer++) {
246 const receiver = matching[proposer], preference = market.preferences[proposer];
247 if (receiver >= 0 && (!preference.includes(receiver) || !market.preferences[receiver].includes(proposer))) {
248 unacceptable.push(`UNACCEPTABLE ${market.names[proposer]} ${market.names[receiver]}`);
249 }
250 for (let other = market.proposers; other < market.names.length; other++) {
251 const ranking = market.preferences[other];
252 if (receiver === other || !preference.includes(other) || !ranking.includes(proposer)) continue;
253 if (rank(preference, other) >= rank(preference, receiver)) continue;
254 const held = partnersOf(matching.slice(0, market.proposers), other);
255 if (held.length < market.capacities[other] || rank(ranking, proposer) < Math.max(...held.map(p => rank(ranking, p)))) {
256 blocking.push(`BLOCKING ${market.names[proposer]} ${market.names[other]}`);
257 }
258 }
259 }
260 } else {
261 for (let first = 0; first < market.names.length; first++) {
262 for (let second = first + 1; second < market.names.length; second++) {
263 const mutual = market.preferences[first].includes(second) && market.preferences[second].includes(first);
264 const notation = `${market.names[first]} ${market.names[second]}`;
265 if (matching[first] === second) {
266 if (!mutual) unacceptable.push(`UNACCEPTABLE ${notation}`);
267 } else if (mutual && rank(market.preferences[first], second) < rank(market.preferences[first], matching[first])
268 && rank(market.preferences[second], first) < rank(market.preferences[second], matching[second])) {
269 blocking.push(`BLOCKING ${notation}`);
270 }
271 }
272 }
273 }
274 if (!unacceptable.length && !blocking.length) return [`CHECK ${market.name} STABLE`];
275 return [`CHECK ${market.name} UNSTABLE unacceptable=${unacceptable.length} blocking=${blocking.length}`, ...unacceptable, ...blocking];
276}
277function roommates(market: Market, verbose: boolean): string[] {
278 market.requireKind('ROOMS');
279 const count = market.names.length;
280 for (let agent = 0; agent < count; agent++) {
281 if (market.preferences[agent].length < count - 1) fail(`INCOMPLETE ${market.names[agent]}`);
282 }
283 const lists = market.preferences.map(list => list.slice()), held = Array<number>(count).fill(-1), queue = sequence(0, count);
284 const removePair = (a: number, b: number): void => { lists[a] = lists[a].filter(x => x !== b); lists[b] = lists[b].filter(x => x !== a); };
285 const table = (): string[] => lists.map((list, a) => `TABLE ${market.names[a]} ${listText(market.names, list)}`);
286 let proposals = 0, failure = -1;
287 for (let head = 0; head < queue.length; head++) {
288 const proposer = queue[head], receiver = lists[proposer][0];
289 if (held[receiver] >= 0) queue.push(held[receiver]);
290 held[receiver] = proposer;
291 proposals++;
292 for (const other of lists[receiver].slice(lists[receiver].indexOf(proposer) + 1)) removePair(receiver, other);
293 failure = lists.findIndex(list => list.length === 0);
294 if (failure >= 0) break;
295 }
296 const history = failure < 0 && verbose ? table() : [];
297 let rotations = 0;
298 while (failure < 0) {
299 let point = lists.findIndex(list => list.length >= 2);
300 if (point < 0) break;
301 const path: number[] = [];
302 while (!path.includes(point)) {
303 path.push(point);
304 const nextList = lists[lists[point][1]];
305 point = nextList[nextList.length - 1];
306 }
307 const rotation = path.slice(path.indexOf(point)), front = rotation.map(agent => lists[agent][0]);
308 history.push(`ROTATION ${++rotations} ${rotation.map((a, i) => `${market.names[a]}-${market.names[front[i]]}`).join(' ')}`);
309 const removals: [number, number][] = [];
310 rotation.forEach((agent, i) => {
311 const receiver = front[(i + 1) % front.length];
312 for (const other of lists[receiver].slice(lists[receiver].indexOf(agent) + 1)) removals.push([receiver, other]);
313 });
314 for (const [first, second] of removals) removePair(first, second);
315 if (verbose) history.push(...table());
316 failure = lists.findIndex(list => list.length === 0);
317 }
318 const output = [`ROOMMATES ${market.name} proposals=${proposals}`, ...history];
319 if (failure >= 0) return [...output, `NO STABLE MATCHING ${market.names[failure]}`];
320 const pairs: string[] = [];
321 lists.forEach((list, a) => { if (a < list[0]) pairs.push(`${market.names[a]}-${market.names[list[0]]}`); });
322 return [...output, `MATCHING ${pairs.join(' ')}`];
323}
324function ttc(market: Market): string[] {
325 const remaining = new Set(sequence(0, market.names.length)), allocation = Array<number>(market.names.length).fill(-1), cycles: string[] = [];
326 while (remaining.size) {
327 const pointing = Array<number>(market.names.length).fill(-1);
328 for (const agent of remaining) {
329 pointing[agent] = market.effective(agent).find(house => remaining.has(house))!;
330 }
331 const path: number[] = [];
332 let point = remaining.values().next().value!;
333 while (!path.includes(point)) { path.push(point); point = pointing[point]; }
334 const cycle = path.slice(path.indexOf(point));
335 for (const agent of cycle) { allocation[agent] = pointing[agent]; remaining.delete(agent); }
336 cycles.push(`CYCLE ${cycles.length + 1} ${cycle.map(a => `${market.names[a]}:${market.houses[allocation[a]]}`).join(' ')}`);
337 }
338 return [`TTC ${market.name} cycles=${cycles.length}`, ...cycles, `ALLOCATION ${allocationText(market, allocation)}`];

340function serial(market: Market, names: readonly string[]): string[] {
341 const order: number[] = [];
342 for (const name of names) {
343 const agent = market.agent(name);
344 if (order.includes(agent)) fail(`DUPLICATE AGENT ${name}`);
345 order.push(agent);
346 }
347 if (order.length !== market.names.length) fail('BAD ORDER');
348 const remaining = new Set(sequence(0, market.houses.length)), allocation = Array<number>(market.names.length).fill(-1);
349 let assigned = 0;
350 for (const agent of order) {
351 const house = market.preferences[agent].find(candidate => remaining.has(candidate));
352 if (house !== undefined) { allocation[agent] = house; remaining.delete(house); assigned++; }
353 }
354 const worse = sequence(0, market.names.length).filter(a => !market.effective(a).includes(allocation[a]));
355 return [`SERIAL ${market.name} assigned=${assigned}`, `ALLOCATION ${allocationText(market, allocation)}`, `WORSE ${listText(market.names, worse)}`];
356}
357function hospitalReport(market: Market): string[] {
358 const resident = deferred(market).matching, hospital = deferred(market, true).matching;
359 const unmatched = sequence(0, market.proposers).filter(p => resident[p] < 0);
360 let rural = resident.every((partner, p) => (partner >= 0) === (hospital[p] >= 0));
361 const output = [`HR ${market.name} residents=${market.proposers} hospitals=${market.names.length - market.proposers}`,
362 `RESIDENT-OPTIMAL ${pairText(market, resident)}`, `HOSPITAL-OPTIMAL ${pairText(market, hospital)}`, `UNMATCHED ${listText(market.names, unmatched)}`];
363 for (let receiver = market.proposers; receiver < market.names.length; receiver++) {
364 const first = partnersOf(resident, receiver), second = partnersOf(hospital, receiver), capacity = market.capacities[receiver];
365 const same = first.length === second.length && first.every((p, i) => p === second[i]);
366 output.push(`HOSPITAL ${market.names[receiver]} cap=${capacity} resident=${first.length} hospital=${second.length} same=${same ? 'YES' : 'NO'}`);
367 if (first.length !== second.length || (first.length < capacity && !same)) rural = false;
368 }
369 return [...output, `RURAL ${rural ? 'HOLDS' : 'FAILS'}`];
370}

372function manipulate(market: Market, proposer: number): string[] {
373 const truth = market.preferences[proposer], baselineB = boston(market).matching[proposer], baselineG = deferred(market).matching[proposer];
374 const output = [`MANIPULATE ${market.name} ${market.names[proposer]} boston=${partnerName(market, baselineB)} gs=${partnerName(market, baselineG)} swaps=${Math.max(0, truth.length - 1)}`];
375 for (let index = 0; index + 1 < truth.length; index++) {
376 const preference = truth.slice();
377 [preference[index], preference[index + 1]] = [preference[index + 1], preference[index]];
378 const trial = market.trial(proposer, preference);
379 const outcomes: readonly (readonly ['BOSTON' | 'GS', number, number])[] = [
380 ['BOSTON', boston(trial).matching[proposer], baselineB], ['GS', deferred(trial).matching[proposer], baselineG],
381 ];
382 for (const [label, result, baseline] of outcomes) {
383 if (rank(truth, result) < rank(truth, baseline)) {
384 output.push(`IMPROVE ${label} ${index + 1} ${market.names[truth[index]]} ${market.names[truth[index + 1]]} result=${partnerName(market, result)}`);
385 }
386 }
387 }
388 if (output.length === 1) output.push('NO IMPROVEMENT');
389 return output;
390}
391function gcd(first: number, second: number): number {
392 while (second) [first, second] = [second, first % second];
393 return first;

395function statistics(market: Market): string[] {
396 if (market.current === null) fail('NO MATCHING');
397 const { source, matching } = market.current, proposerRanks: number[] = [], receiverRanks: number[] = [], occupied = new Set<number>();
398 matching.forEach((receiver, proposer) => {
399 if (receiver < 0) return;
400 proposerRanks.push(rank(market.preferences[proposer], receiver));
401 receiverRanks.push(rank(market.preferences[receiver], proposer));
402 occupied.add(receiver);
403 });
404 const pairs = proposerRanks.length, output = [`STATS ${market.name} source=${source} pairs=${pairs}`];
405 const sides = [
406 { label: 'PROPOSERS', ranks: proposerRanks, unmatched: market.proposers - pairs },
407 { label: 'RECEIVERS', ranks: receiverRanks, unmatched: market.names.length - market.proposers - occupied.size },
408 ];
409 for (const { label, ranks, unmatched } of sides) {
410 const sum = ranks.reduce((a, b) => a + b, 0), first = ranks.filter(value => value === 1).length;
411 let average = '-', truncated = '-';
412 if (pairs) {
413 const divisor = gcd(sum, pairs), cents = Math.floor(100 * sum / pairs);
414 average = `${sum / divisor}/${pairs / divisor}`;
415 truncated = `${Math.floor(cents / 100)}.${String(cents % 100).padStart(2, '0')}`;
416 }
417 output.push(`${label} sum=${sum} average=${average} truncated=${truncated} first=${first} unmatched=${unmatched}`);
418 }
419 return output;
420}

425 market.names.forEach((name, agent) => {

428 else prefix = `AGENT ${name}${market.kind === 'HOUSES' ? ' house=' + market.houses[agent] : ''}`;

431 if (market.kind === 'MARKET') output.push(`CURRENT ${market.current?.source ?? '-'}`);

434function newMarket(command: CreationCommand, name: string, args: readonly string[]): Market {
435 if (command === 'RANDOM') {
436 const count = Number(args[0]), names = ['P', 'R'].flatMap(prefix => sequence(1, count + 1).map(i => prefix + i));
437 const market = new Market(name, 'MARKET', names, count);
438 let state = Number(args[1]);
439 for (let agent = 0; agent < names.length; agent++) {
440 const preference = agent < count ? sequence(count, 2 * count) : sequence(0, count);
441 for (let length = count; length >= 2; length--) {
442 state = 48271 * state % 2147483647;
443 const index = state % length;
444 [preference[length - 1], preference[index]] = [preference[index], preference[length - 1]];
445 }
446 market.preferences[agent] = preference;
447 }
448 return market;
449 }
450 const names: string[] = [], houses: string[] = [];
451 let proposers = 0;
452 for (const word of args) {
453 if (word === '|') { proposers = names.length; continue; }
454 const [agent, house] = word.split(':');
455 if (names.includes(agent)) fail(`DUPLICATE AGENT ${agent}`);
456 if (command === 'HOUSES') {
457 if (houses.includes(house)) fail(`DUPLICATE HOUSE ${house}`);
458 houses.push(house);
459 }
460 names.push(agent);
461 }
462 return new Market(name, command, names, proposers, houses);

467 execute(line: string): string[] {
468 const [word, name, ...args] = line.split(' ');
469 const command = word as Command;
470 if (isCreation(command)) {
471 if (this.#markets.has(name)) fail(`DUPLICATE MARKET ${name}`);
472 const market = newMarket(command, name, args);
473 this.#markets.set(name, market);
474 if (command === 'RANDOM') return [`RANDOM ${name} n=${market.proposers} seed=${args[1]}`];
475 if (command === 'MARKET') return [`MARKET ${name} proposers=${market.proposers} receivers=${market.names.length - market.proposers}`];
476 return [`${command} ${name} agents=${market.names.length}`];
477 }
478 const market = this.#markets.get(name);
479 if (market === undefined) fail(`UNKNOWN MARKET ${name}`);
480 switch (command) {
481 case 'SHOW': return showMarket(market);
482 case 'PREF': {
483 const agent = market.agent(args[0]), names = market.kind === 'HOUSES' ? market.houses : market.names, preference: number[] = [];
484 for (const word of args.slice(1)) {
485 const choice = names.indexOf(word);
486 if (choice < 0 || (market.kind === 'ROOMS' && choice === agent)
487 || (market.kind === 'MARKET' && (agent < market.proposers) === (choice < market.proposers))) fail(`BAD CHOICE ${word}`);
488 if (preference.includes(choice)) fail(`DUPLICATE CHOICE ${word}`);
489 preference.push(choice);
490 }
491 market.preferences[agent] = preference;
492 if (market.kind === 'MARKET') market.current = null;
493 return [`PREF ${name} ${args[0]} length=${preference.length}`];
494 }
495 case 'CHECK': return checkPairs(market, args);
496 case 'ROOMMATES': return roommates(market, args.length === 1);
497 case 'TTC': case 'SERIAL':
498 market.requireKind('HOUSES');
499 return command === 'TTC' ? ttc(market) : serial(market, args);
500 default: market.requireKind('MARKET');
501 }
502 switch (command) {
503 case 'CAPACITY': {
504 const receiver = market.agent(args[0], 'RECEIVER'), capacity = Number(args[1]);
505 market.capacities[receiver] = capacity;
506 market.current = null;
507 return [`CAPACITY ${name} ${args[0]} ${capacity}`];
508 }
509 case 'GS': case 'GSR': {
510 const run = deferred(market, command === 'GSR', args.length === 1);
511 market.save(command, run.matching);
512 return [`${command} ${name} proposals=${run.proposals}`, ...run.history, `MATCHING ${pairText(market, run.matching)}`];
513 }
514 case 'BOSTON': {
515 const run = boston(market);
516 market.save(command, run.matching);
517 return [`BOSTON ${name} applications=${run.applications}`, `MATCHING ${pairText(market, run.matching)}`];
518 }
519 case 'HR': return hospitalReport(market);
520 case 'MANIPULATE': return manipulate(market, market.agent(args[0], 'PROPOSER'));
521 case 'STATS': return statistics(market);
522 default: {
523 const lattice = enumerate(market);
524 if (command === 'LATTICE') {
525 const output = [`LATTICE ${name} matchings=${lattice.records.length} rotations=${lattice.rotations.length}`];
526 lattice.records.forEach(({ matching, origin, via }, i) => {
527 const { cost, regret } = metrics(market, matching);
528 output.push(`MATCHING ${i + 1} from=${origin ?? '-'} via=${via ?? '-'} cost=${cost} regret=${regret} ${pairText(market, matching)}`);
529 });
530 lattice.rotations.forEach((rotation, i) => output.push(`ROTATION ${i + 1} ${rotation}`));
531 return output;
532 }
533 const label = command === 'EGALITARIAN' ? 'cost' : 'regret';
534 const values = lattice.records.map(record => metrics(market, record.matching)[label]), best = values.indexOf(Math.min(...values));
535 const matching = lattice.records[best].matching;
536 market.save(command, matching);
537 return [`${command} ${name} matching=${best + 1} ${label}=${values[best]}`, `MATCHING ${pairText(market, matching)}`];
538 }
539 }
540 }
541}

544for (let i = 1; i <= count; i++) {

548 output.push(`ERROR ${error.message}`);
