49def rank_summary(ranks):
50 total = sum(ranks)
51 first = ranks.count(1)
52 if ranks:
53 divisor = gcd(total, len(ranks))
54 average = f"{total // divisor}/{len(ranks) // divisor}"
55 hundredths = total * 100 // len(ranks)
56 truncated = f"{hundredths // 100}.{hundredths % 100:02d}"
57 else:
58 average = "-"

60 return f"sum={total} average={average} truncated={truncated} first={first}"

124 def blocks(self, matching, proposer, receiver):
125 if not acceptable(self.preferences, proposer, receiver):
126 return False
127 preference = self.preferences[proposer]
128 if rank(preference, receiver) >= rank(preference, matching[proposer]):
129 return False
130 partners = self.partners(matching, receiver)
131 if len(partners) < self.capacities[receiver]:
132 return True
133 choices = self.preferences[receiver]
134 worst = max(rank(choices, partner) for partner in partners)
135 return rank(choices, proposer) < worst

169class DeferredAcceptance:
170 def __init__(self, market, preferences, receivers_propose):
171 self.market = market
172 self.preferences = preferences
173 self.receivers_propose = receivers_propose
174 if receivers_propose:
175 self.proposing = market.receivers
176 self.receiving = market.proposers
177 else:
178 self.proposing = market.proposers
179 self.receiving = market.receivers
180 self.next_position = {agent: 0 for agent in self.proposing}
181 self.hold_counts = {agent: 0 for agent in self.proposing}
182 self.holding = {agent: [] for agent in self.receiving}
183 self.lines = []
184
185 def able(self, agent):
186 return (
187 self.hold_counts[agent] < self.market.capacity(agent)
188 and self.next_position[agent] < len(self.preferences[agent])
189 )
190
191 def hold(self, receiver, proposer):
192 self.holding[receiver].append(proposer)
193 self.hold_counts[proposer] += 1
194
195 def release(self, receiver, proposer):
196 self.holding[receiver].remove(proposer)
197 self.hold_counts[proposer] -= 1
198
199 def propose(self, proposer, receiver, queue):
200 choices = self.preferences[receiver]
201 held = self.holding[receiver]
202 if proposer not in choices:
203 self.lines.append(f"PROPOSE {proposer} {receiver} REJECT")
204 return
205 if len(held) < self.market.capacity(receiver):
206 self.hold(receiver, proposer)
207 self.lines.append(f"PROPOSE {proposer} {receiver} HOLD")
208 return
209 worst = max(held, key=lambda agent: rank(choices, agent))
210 if rank(choices, proposer) < rank(choices, worst):
211 self.release(receiver, worst)
212 self.hold(receiver, proposer)
213 self.lines.append(f"PROPOSE {proposer} {receiver} HOLD REJECT {worst}")
214 if self.able(worst) and worst not in queue:
215 queue.append(worst)
216 else:
217 self.lines.append(f"PROPOSE {proposer} {receiver} REJECT")
218
219 def run(self):
220 queue = deque(agent for agent in self.proposing if self.able(agent))
221 while queue:
222 proposer = queue.popleft()
223 receiver = self.preferences[proposer][self.next_position[proposer]]
224 self.next_position[proposer] += 1
225 self.propose(proposer, receiver, queue)
226 if self.able(proposer):
227 queue.append(proposer)
228 matching = self.market.empty_matching()
229 for receiver, held in self.holding.items():
230 for proposer in held:
231 if self.receivers_propose:
232 matching[receiver] = proposer
233 else:
234 matching[proposer] = receiver
235 return matching

240 matching = procedure.run()
241 return matching, procedure.lines

244def boston_mechanism(market, preferences):
245 matching = market.empty_matching()
246 slots = dict(market.capacities)
247 rounds = max(len(preferences[proposer]) for proposer in market.proposers)
248 applications = 0
249 for round_index in range(rounds):
250 applied = {}
251 for proposer in market.proposers:
252 preference = preferences[proposer]
253 if matching[proposer] is None and len(preference) > round_index:
254 applied[proposer] = preference[round_index]
255 applications += 1
256 for receiver in market.receivers:
257 for proposer in preferences[receiver]:
258 if applied.get(proposer) == receiver and slots[receiver] > 0:
259 matching[proposer] = receiver
260 slots[receiver] -= 1
261 return matching, applications

264def cost_and_regret(market, matching):
265 cost = 0
266 regret = 0
267 for proposer in market.proposers:
268 receiver = matching[proposer]
269 if receiver is None:
270 continue
271 for value in (
272 rank(market.preferences[proposer], receiver),
273 rank(market.preferences[receiver], proposer),
274 ):
275 cost += value
276 regret = max(regret, value)
277 return cost, regret

280def successor(market, matching, holders, proposer):
281 receiver = matching[proposer]
282 if receiver is None:
283 return None
284 preference = market.preferences[proposer]
285 for candidate in preference[preference.index(receiver) + 1:]:
286 choices = market.preferences[candidate]
287 if proposer not in choices:
288 continue
289 holder = holders.get(candidate)
290 if holder is None:
291 return None
292 if rank(choices, proposer) < rank(choices, holder):
293 return holder
294 return None

297def rotation_sequence(market, matching):
298 holders = {
299 matching[proposer]: proposer for proposer in market.proposers
300 if matching[proposer] is not None
301 }
302 successors = {
303 proposer: successor(market, matching, holders, proposer)
304 for proposer in market.proposers
305 }
306 rotations = []
307 covered = set()
308 for start in market.proposers:
309 if start in covered:
310 continue
311 path = [start]
312 current = successors[start]
313 while current is not None and current not in path:
314 path.append(current)
315 current = successors[current]
316 if current == start:
317 rotations.append(path)
318 covered.update(path)
319 return rotations

342def stable_matchings(market):
343 first, _ = deferred_acceptance(market, market.preferences, False)
344 entries = [LatticeEntry(market, 1, first, "-", "-")]
345 numbers = {}
346 notations = []
347 index = 0
348 while index < len(entries):
349 entry = entries[index]
350 for rotation in rotation_sequence(market, entry.matching):
351 notation = rotation_notation(entry.matching, rotation)
352 if notation not in numbers:
353 notations.append(notation)
354 numbers[notation] = len(notations)
355 eliminated = eliminate_rotation(entry.matching, rotation)
356 if all(other.matching != eliminated for other in entries):
357 if len(entries) == MAX_MATCHINGS:
358 raise CommandError("TOO MANY MATCHINGS")
359 entries.append(LatticeEntry(
360 market, len(entries) + 1, eliminated, entry.number, numbers[notation]
361 ))
362 index += 1
363 return entries, notations

366class Irving:
367 def __init__(self, market, verbose):
368 self.market = market
369 self.verbose = verbose
370 self.working = {agent: list(market.preferences[agent]) for agent in market.agents}
371 self.proposals = 0
372 self.rotations = 0
373 self.lines = []
374
375 def remove_pair(self, first, second):
376 if second in self.working[first]:
377 self.working[first].remove(second)
378 if first in self.working[second]:
379 self.working[second].remove(first)
380
381 def remove_after(self, agent, other):
382 working = self.working[agent]
383 for later in working[working.index(other) + 1:]:
384 self.remove_pair(agent, later)
385
386 def print_table(self):
387 if self.verbose:
388 for agent in self.market.agents:
389 self.lines.append(f"TABLE {agent} {list_notation(self.working[agent])}")
390
391 def empty_agent(self):
392 for agent in self.market.agents:
393 if not self.working[agent]:
394 return agent
395 return None
396
397 def phase_one(self):
398 held = {agent: None for agent in self.market.agents}
399 queue = deque(self.market.agents)
400 while queue:
401 proposer = queue.popleft()
402 receiver = self.working[proposer][0]
403 self.proposals += 1
404 if held[receiver] is not None:
405 queue.append(held[receiver])
406 held[receiver] = proposer
407 self.remove_after(receiver, proposer)
408 failed = self.empty_agent()
409 if failed is not None:
410 return failed
411 return None
412
413 def find_rotation(self):
414 start = next(agent for agent in self.market.agents if len(self.working[agent]) >= 2)
415 sequence = [start]
416 while True:
417 second = self.working[sequence[-1]][1]
418 following = self.working[second][-1]
419 if following in sequence:
420 break
421 sequence.append(following)
422 cycle = sequence[sequence.index(following):]
423 return [(agent, self.working[agent][0]) for agent in cycle]
424
425 def eliminate(self, rotation):
426 removals = []
427 for index, (agent, _) in enumerate(rotation):
428 target = rotation[(index + 1) % len(rotation)][1]
429 working = self.working[target]
430 for later in working[working.index(agent) + 1:]:
431 removals.append((target, later))
432 for first, second in removals:
433 self.remove_pair(first, second)
434
435 def phase_two(self):
436 while any(len(self.working[agent]) >= 2 for agent in self.market.agents):
437 rotation = self.find_rotation()
438 self.rotations += 1
439 pairs = " ".join(f"{agent}-{front}" for agent, front in rotation)
440 self.lines.append(f"ROTATION {self.rotations} {pairs}")
441 self.eliminate(rotation)
442 self.print_table()
443 failed = self.empty_agent()
444 if failed is not None:
445 return failed
446 return None
447
448 def solve(self):
449 failed = self.phase_one()
450 if failed is None:
451 self.print_table()
452 failed = self.phase_two()
453 if failed is not None:
454 self.lines.append(f"NO STABLE MATCHING {failed}")
455 return
456 pairs = []
457 for agent in self.market.agents:
458 partner = self.working[agent][0]
459 if self.market.position(agent) < self.market.position(partner):
460 pairs.append(f"{agent}-{partner}")
461 self.lines.append("MATCHING " + " ".join(pairs))

472def top_trading_cycles(market):
473 remaining = list(market.agents)
474 received = {}
475 cycles = []
476 while remaining:
477 points = {agent: pointed_agent(market, agent, remaining) for agent in remaining}
478 visited = []

683 def check_two_sided(self, market, pairs):
684 proposers = []
685 receivers = []
686 for proposer, receiver in pairs:
687 market.require_proposer(proposer)
688 market.require_receiver(receiver)
689 if proposer in proposers:
690 raise CommandError(f"DUPLICATE AGENT {proposer}")
691 proposers.append(proposer)
692 receivers.append(receiver)
693 if receivers.count(receiver) > market.capacities[receiver]:
694 raise CommandError(f"OVER CAPACITY {receiver}")
695 matching = market.empty_matching()
696 for proposer, receiver in pairs:
697 matching[proposer] = receiver
698 unacceptable = [
699 (proposer, matching[proposer]) for proposer in market.proposers
700 if matching[proposer] is not None
701 and not acceptable(market.preferences, proposer, matching[proposer])
702 ]
703 blocking = [
704 (proposer, receiver)
705 for proposer in market.proposers for receiver in market.receivers
706 if matching[proposer] != receiver and market.blocks(matching, proposer, receiver)
707 ]
708 self.report_check(market.name, unacceptable, blocking)

710 def check_roommates(self, market, pairs):
711 seen = []
712 for first, second in pairs:
713 market.require_agent(first)
714 market.require_agent(second)
715 if first in seen:
716 raise CommandError(f"DUPLICATE AGENT {first}")
717 if second == first or second in seen:
718 raise CommandError(f"DUPLICATE AGENT {second}")
719 seen.extend((first, second))
720 partner = {agent: None for agent in market.agents}
721 for first, second in pairs:
722 partner[first] = second
723 partner[second] = first
724 preferences = market.preferences
725 unacceptable = [
726 market.ordered(first, second) for first, second in pairs
727 if not acceptable(preferences, first, second)
728 ]
729 unacceptable.sort(key=lambda pair: (market.position(pair[0]), market.position(pair[1])))
730 blocking = []
731 for index, first in enumerate(market.agents):
732 for second in market.agents[index + 1:]:
733 if partner[first] == second or not acceptable(preferences, first, second):
734 continue
735 first_gains = (
736 rank(preferences[first], second) < rank(preferences[first], partner[first])
737 )
738 second_gains = (
739 rank(preferences[second], first) < rank(preferences[second], partner[second])
740 )
741 if first_gains and second_gains:
742 blocking.append((first, second))
743 self.report_check(market.name, unacceptable, blocking)

874 def manipulate(self, arguments):
875 name, agent = arguments
876 market = self.market(name, TWO_SIDED)
877 market.require_proposer(agent)

880 gs_base = deferred_acceptance(market, market.preferences, False)[0][agent]
881 self.emit(
882 "MANIPULATE", name, agent, f"boston={partner_text(boston_base)}",
883 f"gs={partner_text(gs_base)}", f"swaps={max(len(truth) - 1, 0)}",
884 )
885 improved = False
886 for i in range(1, len(truth)):
887 report = list(truth)
888 report[i - 1], report[i] = report[i], report[i - 1]
889 preferences = dict(market.preferences)
890 preferences[agent] = report
891 boston_result = boston_mechanism(market, preferences)[0][agent]
892 gs_result = deferred_acceptance(market, preferences, False)[0][agent]
893 if rank(truth, boston_result) < rank(truth, boston_base):
894 self.emit(
895 "IMPROVE", "BOSTON", i, truth[i - 1], truth[i], f"result={boston_result}"
896 )
897 improved = True
898 if rank(truth, gs_result) < rank(truth, gs_base):
899 self.emit("IMPROVE", "GS", i, truth[i - 1], truth[i], f"result={gs_result}")
900 improved = True
901 if not improved:
902 self.emit("NO IMPROVEMENT")

904 def stats(self, arguments):

912 (proposer, matching[proposer]) for proposer in market.proposers

914 ]
915 proposer_ranks = [rank(preferences[proposer], receiver) for proposer, receiver in pairs]
916 receiver_ranks = [rank(preferences[receiver], proposer) for proposer, receiver in pairs]
917 unmatched_proposers = len(market.proposers) - len(pairs)
918 unmatched_receivers = sum(
919 not market.partners(matching, receiver) for receiver in market.receivers
920 )

922 self.emit("PROPOSERS", rank_summary(proposer_ranks), f"unmatched={unmatched_proposers}")
