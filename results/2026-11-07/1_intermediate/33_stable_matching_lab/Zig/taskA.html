54const Sequence = struct {
55 items: [24]usize = undefined,
56 len: usize = 0,
57 fn list(sequence: *const Sequence) []const usize {
58 return sequence.items[0..sequence.len];
59 }
60 fn append(sequence: *Sequence, agent: usize) void {
61 sequence.items[sequence.len] = agent;
62 sequence.len += 1;
63 }
64 fn index(sequence: *const Sequence, agent: usize) ?usize {
65 for (sequence.list(), 0..) |item, i| if (item == agent) return i;
66 return null;
67 }
68 fn remove(sequence: *Sequence, agent: usize) void {
69 const position = sequence.index(agent) orelse return;
70 var i = position;
71 while (i + 1 < sequence.len) : (i += 1) sequence.items[i] = sequence.items[i + 1];
72 sequence.len -= 1;
73 }
74 fn rank(sequence: *const Sequence, partner: ?usize) usize {
75 if (partner) |agent| if (sequence.index(agent)) |i| return i + 1;
76 return sequence.len + 1;

172const MatchingResult = struct { matching: Matching, proposals: usize, log: []const u8 };
173const ProposalState = struct {
174 gpa: Allocator,
175 market: *const Market,
176 next: [24]usize = @splat(0),
177 counts: [24]usize = @splat(0),
178 queued: [24]bool = @splat(false),
179 holding: [24]Sequence = @splat(Sequence{}),
180 queue: std.ArrayList(usize) = .empty,
181 fn able(state: *const ProposalState, agent: usize) bool {
182 return state.counts[agent] < state.market.capacities[agent] and state.next[agent] < state.market.preferences[agent].len;
183 }
184 fn enqueue(state: *ProposalState, agent: usize) Allocator.Error!void {
185 if (state.able(agent) and !state.queued[agent]) {
186 try state.queue.append(state.gpa, agent);
187 state.queued[agent] = true;
188 }
189 }

191fn deferred(gpa: Allocator, market: *const Market, reverse: bool, verbose: bool) Error!MatchingResult {
192 var state: ProposalState = .{ .gpa = gpa, .market = market };
193 const start: usize = if (reverse) market.proposer_count else 0;
194 const end = if (reverse) market.count else market.proposer_count;
195 for (start..end) |a| try state.enqueue(a);
196 var log: Writer.Allocating = .init(gpa);
197 defer log.deinit();
198 var head: usize = 0;
199 while (head < state.queue.items.len) : (head += 1) {
200 const x = state.queue.items[head];
201 state.queued[x] = false;
202 const y = market.preferences[x].items[state.next[x]];
203 state.next[x] += 1;
204 var accepted = false;
205 var rejected: ?usize = null;
206 if (market.preferences[y].index(x) != null) {
207 if (state.holding[y].len < market.capacities[y]) {
208 state.holding[y].append(x);
209 state.counts[x] += 1;
210 accepted = true;
211 } else {
212 var worst: usize = 0;
213 for (state.holding[y].list(), 0..) |a, i| {
214 if (market.rank(y, a) > market.rank(y, state.holding[y].items[worst])) worst = i;
215 }
216 const previous = state.holding[y].items[worst];
217 if (market.rank(y, x) < market.rank(y, previous)) {
218 state.holding[y].items[worst] = x;
219 state.counts[previous] -= 1;
220 state.counts[x] += 1;
221 accepted = true;
222 rejected = previous;
223 try state.enqueue(previous);
224 }
225 }
226 }
227 if (verbose) {
228 try log.writer.print("PROPOSE {f} {f} {s}", .{ market.names[x], market.names[y], if (accepted) "HOLD" else "REJECT" });
229 if (rejected) |a| try log.writer.print(" REJECT {f}", .{market.names[a]});
230 try log.writer.writeByte('\n');
231 }
232 try state.enqueue(x);
233 }
234 var matching: Matching = .{};
235 const receiving_start: usize = if (reverse) 0 else market.proposer_count;
236 const receiving_end = if (reverse) market.proposer_count else market.count;
237 for (receiving_start..receiving_end) |y| {
238 for (state.holding[y].list()) |x| matching.partners[if (reverse) y else x] = if (reverse) x else y;
239 }
240 return .{ .matching = matching, .proposals = head, .log = try log.toOwnedSlice() };
241}
242fn boston(market: *const Market) MatchingResult {
243 var matching: Matching = .{};
244 var remaining = market.capacities;
245 var rounds: usize = 0;
246 for (market.preferences[0..market.proposer_count]) |preference| rounds = @max(rounds, preference.len);
247 var applications: usize = 0;
248 for (0..rounds) |round| {
249 var applicants: [24]Sequence = @splat(Sequence{});
250 for (0..market.proposer_count) |p| {
251 if (matching.partners[p] != null or round >= market.preferences[p].len) continue;
252 applicants[market.preferences[p].items[round]].append(p);
253 applications += 1;
254 }
255 for (market.proposer_count..market.count) |r| {
256 for (market.preferences[r].list()) |p| {
257 if (remaining[r] > 0 and applicants[r].index(p) != null) {
258 matching.partners[p] = r;
259 remaining[r] -= 1;
260 }
261 }
262 }
263 }
264 return .{ .matching = matching, .proposals = applications, .log = "" };
265}
266const CheckResult = struct {

270fn checkMatching(gpa: Allocator, market: *const Market, matching: Matching) Allocator.Error!CheckResult {
271 var result: CheckResult = .{};
272 if (market.kind == .roommates) {
273 for (0..market.count) |a| {
274 for (a + 1..market.count) |b| {
275 if (matching.partners[a] == b) {
276 if (!market.acceptable(a, b)) try result.unacceptable.append(gpa, .{ .first = a, .second = b });
277 } else if (market.acceptable(a, b) and market.rank(a, b) < market.rank(a, matching.partners[a]) and
278 market.rank(b, a) < market.rank(b, matching.partners[b]))
279 {
280 try result.blocking.append(gpa, .{ .first = a, .second = b });
281 }
282 }
283 }
284 } else {
285 for (0..market.proposer_count) |p| {
286 if (matching.partners[p]) |r| {
287 if (!market.acceptable(p, r)) try result.unacceptable.append(gpa, .{ .first = p, .second = r });
288 }
289 for (market.proposer_count..market.count) |r| {
290 if (matching.partners[p] == r or !market.acceptable(p, r) or market.rank(p, r) >= market.rank(p, matching.partners[p])) continue;
291 const held = market.residents(matching, r);
292 var worst: usize = 0;
293 for (held.list()) |a| worst = @max(worst, market.rank(r, a));
294 if (held.len < market.capacities[r] or market.rank(r, p) < worst)
295 try result.blocking.append(gpa, .{ .first = p, .second = r });
296 }
297 }
298 }
299 return result;
300}
301const Candidate = struct {
302 matching: Matching,
303 origin: usize,
304 via: usize,
305 cost: usize = 0,
306 regret: usize = 0,
307 fn init(market: *const Market, matching: Matching, origin: usize, via: usize) Candidate {
308 var result: Candidate = .{ .matching = matching, .origin = origin, .via = via };
309 for (0..market.proposer_count) |p| {
310 if (matching.partners[p]) |r| {
311 const a = market.rank(p, r);
312 const b = market.rank(r, p);
313 result.cost += a + b;
314 result.regret = @max(result.regret, @max(a, b));
315 }
316 }
317 return result;
318 }

320const RotationList = struct { items: [12]Sequence = undefined, len: usize = 0 };
321fn rotations(market: *const Market, matching: Matching) RotationList {
322 var owners: [24]?usize = @splat(null);
323 for (0..market.proposer_count) |p| {
324 if (matching.partners[p]) |r| owners[r] = p;
325 }
326 var successors: [12]?usize = @splat(null);
327 for (0..market.proposer_count) |p| {
328 const partner = matching.partners[p] orelse continue;
329 const preference = &market.preferences[p];
330 for (preference.list()[preference.index(partner).? + 1 ..]) |r| {
331 if (market.preferences[r].index(p) == null) continue;
332 if (owners[r] == null or market.rank(r, p) < market.rank(r, owners[r])) {
333 successors[p] = owners[r];
334 break;
335 }
336 }
337 }
338 var result: RotationList = .{};
339 var covered: [12]bool = @splat(false);
340 for (0..market.proposer_count) |start| {
341 if (covered[start]) continue;
342 var seen: [12]bool = @splat(false);
343 var path: Sequence = .{};
344 var current: ?usize = start;
345 while (current) |p| {
346 if (seen[p]) break;
347 seen[p] = true;
348 path.append(p);
349 current = successors[p];
350 }
351 if (current != start) continue;
352 result.items[result.len] = path;
353 result.len += 1;
354 for (path.list()) |p| covered[p] = true;
355 }
356 return result;
357}

426 fn choice(session: *Session, market: *const Market, a: usize, name: []const u8) Error!usize {
427 const b = (if (market.kind == .housing) market.house(name) else market.find(name)) orelse
428 return session.fail("BAD CHOICE {s}", .{name});
429 const valid = switch (market.kind) {
430 .two_sided => (a < market.proposer_count) != (b < market.proposer_count),
431 .roommates => a != b,
432 .housing => true,
433 };
434 if (!valid) return session.fail("BAD CHOICE {s}", .{name});
435 return b;

454 fn check(session: *Session, market: *const Market, words: []const []const u8) Error!void {
455 if (market.kind == .housing) return session.fail("WRONG TYPE {f}", .{market.name});
456 var matching: Matching = .{};
457 var used: [24]bool = @splat(false);
458 var counts: [24]usize = @splat(0);
459 for (words[2..]) |word| {
460 const separator = std.mem.findScalar(u8, word, '-').?;
461 const x = word[0..separator];
462 const y = word[separator + 1 ..];
463 if (market.kind == .two_sided) {
464 const p = try session.proposer(market, x);
465 const r = try session.receiver(market, y);
466 if (used[p]) return session.fail("DUPLICATE AGENT {s}", .{x});
467 if (counts[r] + 1 > market.capacities[r]) return session.fail("OVER CAPACITY {s}", .{y});
468 used[p] = true;
469 counts[r] += 1;
470 matching.partners[p] = r;
471 } else {
472 const a = try session.agent(market, x);
473 const b = try session.agent(market, y);
474 if (used[a]) return session.fail("DUPLICATE AGENT {s}", .{x});
475 if (a == b or used[b]) return session.fail("DUPLICATE AGENT {s}", .{y});
476 used[a] = true;
477 used[b] = true;
478 matching.partners[a] = b;
479 matching.partners[b] = a;
480 }
481 }
482 const result = try checkMatching(session.gpa, market, matching);
483 if (result.unacceptable.items.len == 0 and result.blocking.items.len == 0)
484 return line(session.out, "CHECK {f} STABLE", .{market.name});
485 try line(session.out, "CHECK {f} UNSTABLE unacceptable={d} blocking={d}", .{ market.name, result.unacceptable.items.len, result.blocking.items.len });
486 for (result.unacceptable.items) |pair| try line(session.out, "UNACCEPTABLE {f} {f}", .{ market.names[pair.first], market.names[pair.second] });
487 for (result.blocking.items) |pair| try line(session.out, "BLOCKING {f} {f}", .{ market.names[pair.first], market.names[pair.second] });
488 }
489 fn enumerate(session: *Session, market: *const Market) Error!Enumeration {
490 for (market.capacities[market.proposer_count..market.count]) |capacity| {
491 if (capacity != 1) return session.fail("NOT ONE TO ONE", .{});
492 }
493 var result: Enumeration = .{};
494 const first = try deferred(session.gpa, market, false, false);
495 try result.matchings.append(session.gpa, Candidate.init(market, first.matching, 0, 0));
496 var index: usize = 0;
497 while (index < result.matchings.items.len) : (index += 1) {
498 const matching = result.matchings.items[index].matching;
499 const sequence = rotations(market, matching);
500 for (sequence.items[0..sequence.len]) |rotation| {
501 var buffer: Writer.Allocating = .init(session.gpa);
502 defer buffer.deinit();
503 for (rotation.list(), 0..) |p, i| {
504 if (i > 0) try buffer.writer.writeByte(' ');
505 try buffer.writer.print("{f}-{f}", .{ market.names[p], market.names[matching.partners[p].?] });
506 }
507 const notation = try buffer.toOwnedSlice();
508 var number: usize = 0;
509 for (result.notations.items, 0..) |previous, i| {
510 if (equal(previous, notation)) {
511 number = i + 1;
512 break;
513 }
514 }
515 if (number == 0) {
516 try result.notations.append(session.gpa, notation);
517 number = result.notations.items.len;
518 }
519 var next = matching;
520 for (rotation.list(), 0..) |p, i| next.partners[p] = matching.partners[rotation.items[(i + 1) % rotation.len]];
521 var seen = false;
522 for (result.matchings.items) |previous| {
523 if (previous.matching.same(next, market.proposer_count)) {
524 seen = true;
525 break;
526 }
527 }
528 if (seen) continue;
529 if (result.matchings.items.len == 50) return session.fail("TOO MANY MATCHINGS", .{});
530 try result.matchings.append(session.gpa, Candidate.init(market, next, index + 1, number));
531 }
532 }
533 return result;
534 }
535 fn hospitals(session: *Session, market: *const Market) Error!void {
536 const a = (try deferred(session.gpa, market, false, false)).matching;
537 const h = (try deferred(session.gpa, market, true, false)).matching;
538 try line(session.out, "HR {f} residents={d} hospitals={d}", .{ market.name, market.proposer_count, market.count - market.proposer_count });
539 try line(session.out, "RESIDENT-OPTIMAL {f}", .{Pairs{ .market = market, .matching = a }});
540 try line(session.out, "HOSPITAL-OPTIMAL {f}", .{Pairs{ .market = market, .matching = h }});
541 var unmatched: Sequence = .{};
542 var rural = true;
543 for (0..market.proposer_count) |p| {
544 if (a.partners[p] == null) unmatched.append(p);
545 if ((a.partners[p] == null) != (h.partners[p] == null)) rural = false;
546 }
547 try line(session.out, "UNMATCHED {f}", .{NamedList{ .names = market.names[0..market.count], .sequence = unmatched }});
548 for (market.proposer_count..market.count) |r| {
549 const first = market.residents(a, r);
550 const second = market.residents(h, r);
551 const same = std.mem.eql(usize, first.list(), second.list());
552 try line(session.out, "HOSPITAL {f} cap={d} resident={d} hospital={d} same={s}", .{ market.names[r], market.capacities[r], first.len, second.len, if (same) "YES" else "NO" });
553 if (first.len != second.len or (first.len < market.capacities[r] and !same)) rural = false;
554 }
555 try line(session.out, "RURAL {s}", .{if (rural) "HOLDS" else "FAILS"});
556 }
557 fn printTable(out: *Writer, market: *const Market, lists: *const [24]Sequence) Writer.Error!void {
558 for (0..market.count) |a| try line(out, "TABLE {f} {f}", .{ market.names[a], NamedList{ .names = market.names[0..market.count], .sequence = lists[a] } });
559 }
560 fn empty(lists: *const [24]Sequence, count: usize) ?usize {
561 for (lists[0..count], 0..) |list_value, a| if (list_value.len == 0) return a;
562 return null;
563 }
564 fn roommates(session: *Session, market: *const Market, verbose: bool) Error!void {
565 for (market.preferences[0..market.count], 0..) |preference, a| {
566 if (preference.len != market.count - 1) return session.fail("INCOMPLETE {f}", .{market.names[a]});
567 }
568 var lists = market.preferences;
569 var held: [24]?usize = @splat(null);
570 var queue: std.ArrayList(usize) = .empty;
571 for (0..market.count) |a| try queue.append(session.gpa, a);
572 var trace: Writer.Allocating = .init(session.gpa);
573 defer trace.deinit();
574 var proposals: usize = 0;
575 var failure: ?usize = null;
576 while (proposals < queue.items.len) {
577 const x = queue.items[proposals];
578 proposals += 1;
579 const y = lists[x].items[0];
580 if (held[y]) |previous| try queue.append(session.gpa, previous);
581 held[y] = x;
582 const before = lists[y];
583 for (before.list()[before.index(x).? + 1 ..]) |w| {
584 lists[y].remove(w);
585 lists[w].remove(y);
586 }
587 failure = empty(&lists, market.count);
588 if (failure != null) break;
589 }
590 var rotation_count: usize = 0;
591 if (failure == null) {
592 if (verbose) try printTable(&trace.writer, market, &lists);
593 while (true) {
594 var start: ?usize = null;
595 for (lists[0..market.count], 0..) |list_value, a| {
596 if (list_value.len > 1) {
597 start = a;
598 break;
599 }
600 }
601 var p = start orelse break;
602 var seen: [24]?usize = @splat(null);
603 var path: Sequence = .{};
604 while (seen[p] == null) {
605 seen[p] = path.len;
606 path.append(p);
607 const q = lists[p].items[1];
608 p = lists[q].items[lists[q].len - 1];
609 }
610 var xs: Sequence = .{};
611 var ys: Sequence = .{};
612 for (path.list()[seen[p].?..]) |a| {
613 xs.append(a);
614 ys.append(lists[a].items[0]);
615 }
616 rotation_count += 1;
617 try trace.writer.print("ROTATION {d}", .{rotation_count});
618 for (xs.list(), ys.list()) |x, y| try trace.writer.print(" {f}-{f}", .{ market.names[x], market.names[y] });
619 try trace.writer.writeByte('\n');
620 var removals: std.ArrayList(Pair) = .empty;
621 for (xs.list(), 0..) |x, i| {
622 const u = ys.items[(i + 1) % ys.len];
623 for (lists[u].list()[lists[u].index(x).? + 1 ..]) |w| try removals.append(session.gpa, .{ .first = u, .second = w });
624 }
625 for (removals.items) |pair| {
626 lists[pair.first].remove(pair.second);
627 lists[pair.second].remove(pair.first);
628 }
629 if (verbose) try printTable(&trace.writer, market, &lists);
630 failure = empty(&lists, market.count);
631 if (failure != null) break;
632 }
633 }
634 try line(session.out, "ROOMMATES {f} proposals={d}", .{ market.name, proposals });
635 try session.out.writeAll(trace.written());
636 if (failure) |a| return line(session.out, "NO STABLE MATCHING {f}", .{market.names[a]});
637 try session.out.writeAll("MATCHING");
638 for (0..market.count) |a| {
639 const b = lists[a].items[0];
640 if (a < b) try session.out.print(" {f}-{f}", .{ market.names[a], market.names[b] });
641 }
642 try session.out.writeByte('\n');
643 }
644 fn tradingCycles(session: *Session, market: *const Market) Error!void {
645 var remaining: [24]bool = @splat(true);
646 var assigned: [24]?usize = @splat(null);
647 var count = market.count;
648 var cycles: usize = 0;
649 var trace: Writer.Allocating = .init(session.gpa);
650 defer trace.deinit();
651 while (count > 0) {
652 var pointers: [24]usize = undefined;
653 var start: ?usize = null;
654 for (0..market.count) |a| {
655 if (!remaining[a]) continue;
656 if (start == null) start = a;
657 const effective = market.effective(a);
658 for (effective.list()) |h| {
659 if (remaining[h]) {
660 pointers[a] = h;
661 break;
662 }
663 }
664 }
665 var seen: [24]?usize = @splat(null);
666 var path: Sequence = .{};
667 var current = start.?;
668 while (seen[current] == null) {
669 seen[current] = path.len;
670 path.append(current);
671 current = pointers[current];
672 }
673 cycles += 1;
674 try trace.writer.print("CYCLE {d}", .{cycles});
675 for (path.list()[seen[current].?..]) |a| {
676 assigned[a] = pointers[a];
677 remaining[a] = false;
678 count -= 1;
679 try trace.writer.print(" {f}:{f}", .{ market.names[a], market.houses[pointers[a]] });
680 }
681 try trace.writer.writeByte('\n');
682 }
683 try line(session.out, "TTC {f} cycles={d}", .{ market.name, cycles });
684 try session.out.writeAll(trace.written());
685 try line(session.out, "ALLOCATION {f}", .{Allocation{ .market = market, .houses = assigned }});
686 }
687 fn serial(session: *Session, market: *const Market, words: []const []const u8) Error!void {
688 var order: Sequence = .{};
689 for (words[2..]) |word| {
690 const a = try session.agent(market, word);
691 if (order.index(a) != null) return session.fail("DUPLICATE AGENT {s}", .{word});
692 order.append(a);
693 }
694 if (order.len != market.count) return session.fail("BAD ORDER", .{});
695 var assigned: [24]?usize = @splat(null);
696 var available: [12]bool = @splat(true);
697 var count: usize = 0;
698 for (order.list()) |a| {
699 for (market.preferences[a].list()) |h| {
700 if (available[h]) {
701 assigned[a] = h;
702 available[h] = false;
703 count += 1;
704 break;
705 }
706 }
707 }
708 var worse: Sequence = .{};
709 for (0..market.count) |a| {
710 const effective = market.effective(a);
711 if (assigned[a] == null or effective.index(assigned[a].?) == null) worse.append(a);
712 }
713 try line(session.out, "SERIAL {f} assigned={d}", .{ market.name, count });
714 try line(session.out, "ALLOCATION {f}", .{Allocation{ .market = market, .houses = assigned }});
715 try line(session.out, "WORSE {f}", .{NamedList{ .names = market.names[0..market.count], .sequence = worse }});
716 }
717 fn manipulate(session: *Session, market: *const Market, agent_value: usize) Error!void {
718 const truth = market.preferences[agent_value];
719 const b0 = boston(market).matching.partners[agent_value];
720 const g0 = (try deferred(session.gpa, market, false, false)).matching.partners[agent_value];
721 const swaps = if (truth.len == 0) 0 else truth.len - 1;
722 try line(session.out, "MANIPULATE {f} {f} boston={f} gs={f} swaps={d}", .{ market.name, market.names[agent_value], Partner{ .market = market, .agent = b0 }, Partner{ .market = market, .agent = g0 }, swaps });
723 var improved = false;
724 for (0..swaps) |i| {
725 var trial = market.*;
726 trial.preferences[agent_value].items[i] = truth.items[i + 1];
727 trial.preferences[agent_value].items[i + 1] = truth.items[i];
728 const b = boston(&trial).matching.partners[agent_value];
729 const g = (try deferred(session.gpa, &trial, false, false)).matching.partners[agent_value];
730 if (truth.rank(b) < truth.rank(b0)) {
731 improved = true;
732 try line(session.out, "IMPROVE BOSTON {d} {f} {f} result={f}", .{ i + 1, market.names[truth.items[i]], market.names[truth.items[i + 1]], Partner{ .market = market, .agent = b } });
733 }
734 if (truth.rank(g) < truth.rank(g0)) {
735 improved = true;
736 try line(session.out, "IMPROVE GS {d} {f} {f} result={f}", .{ i + 1, market.names[truth.items[i]], market.names[truth.items[i + 1]], Partner{ .market = market, .agent = g } });
737 }
738 }
739 if (!improved) try line(session.out, "NO IMPROVEMENT", .{});
740 }
741 fn stats(session: *Session, market: *const Market) Error!void {
742 const matching = market.current orelse return session.fail("NO MATCHING", .{});
743 var proposer_ranks: Sequence = .{};
744 var receiver_ranks: Sequence = .{};
745 var occupied: [24]bool = @splat(false);
746 for (0..market.proposer_count) |p| {
747 if (matching.partners[p]) |r| {
748 proposer_ranks.append(market.rank(p, r));
749 receiver_ranks.append(market.rank(r, p));
750 occupied[r] = true;
751 }
752 }
753 var unmatched: usize = 0;
754 for (occupied[market.proposer_count..market.count]) |has_partner| unmatched += @intFromBool(!has_partner);
755 try line(session.out, "STATS {f} source={s} pairs={d}", .{ market.name, @tagName(market.source.?), proposer_ranks.len });
756 try line(session.out, "PROPOSERS {f}", .{RankSummary{ .ranks = proposer_ranks, .unmatched = market.proposer_count - proposer_ranks.len }});
757 try line(session.out, "RECEIVERS {f}", .{RankSummary{ .ranks = receiver_ranks, .unmatched = unmatched }});
758 }
759 fn execute(session: *Session, text: []const u8) Error!void {
760 var words: [32][]const u8 = undefined;
761 var count: usize = 0;
762 var iterator = std.mem.tokenizeScalar(u8, text, ' ');
763 while (iterator.next()) |word| {
764 words[count] = word;
765 count += 1;
766 }
767 const w = words[0..count];
768 const command = std.meta.stringToEnum(Command, w[0]).?;
769 const name = w[1];
770 const out = session.out;
771 switch (command) {
772 .MARKET, .ROOMS => {
773 try session.newName(name);
774 var market: Market = .{ .name = Name.init(name), .kind = if (command == .MARKET) .two_sided else .roommates };
775 for (w[2..]) |word| {
776 if (equal(word, "|")) {
777 market.proposer_count = market.count;
778 continue;
779 }
780 if (market.find(word) != null) return session.fail("DUPLICATE AGENT {s}", .{word});
781 market.names[market.count] = Name.init(word);
782 market.count += 1;
783 }
784 try session.markets.append(session.storage, market);
785 if (command == .MARKET) {
786 try line(out, "MARKET {s} proposers={d} receivers={d}", .{ name, market.proposer_count, market.count - market.proposer_count });
787 } else try line(out, "ROOMS {s} agents={d}", .{ name, market.count });
788 },
789 .HOUSES => {
790 try session.newName(name);
791 var market: Market = .{ .name = Name.init(name), .kind = .housing };
792 for (w[2..]) |word| {
793 const separator = std.mem.findScalar(u8, word, ':').?;
794 const agent_name = word[0..separator];
795 const house_name = word[separator + 1 ..];
796 if (market.find(agent_name) != null) return session.fail("DUPLICATE AGENT {s}", .{agent_name});
797 if (market.house(house_name) != null) return session.fail("DUPLICATE HOUSE {s}", .{house_name});
798 market.names[market.count] = Name.init(agent_name);
799 market.houses[market.count] = Name.init(house_name);
800 market.count += 1;
801 }
802 try session.markets.append(session.storage, market);
803 try line(out, "HOUSES {s} agents={d}", .{ name, market.count });
804 },
805 .RANDOM => {
806 try session.newName(name);
807 const size = integer(w[2]);
808 const seed: u64 = @intCast(integer(w[3]));
809 var state = seed;
810 var market: Market = .{ .name = Name.init(name), .kind = .two_sided, .count = size * 2, .proposer_count = size };
811 for (0..market.count) |a| {
812 var bytes: [8]u8 = undefined;
813 const agent_name = std.fmt.bufPrint(&bytes, "{s}{d}", .{ if (a < size) "P" else "R", a % size + 1 }) catch unreachable;
814 market.names[a] = Name.init(agent_name);
815 var preference: Sequence = .{};
816 for (0..size) |i| preference.append(if (a < size) i + size else i);
817 var length = size;
818 while (length > 1) : (length -= 1) {
819 state = 48271 * state % 2147483647;
820 const j: usize = @intCast(state % @as(u64, @intCast(length)));
821 const first = preference.items[length - 1];
822 preference.items[length - 1] = preference.items[j];
823 preference.items[j] = first;
824 }
825 market.preferences[a] = preference;
826 }
827 try session.markets.append(session.storage, market);
828 try line(out, "RANDOM {s} n={d} seed={d}", .{ name, size, seed });
829 },
830 .PREF => {
831 const market = try session.get(name, null);
832 const agent_value = try session.agent(market, w[2]);
833 var choices: Sequence = .{};
834 for (w[3..]) |word| {
835 const choice_value = try session.choice(market, agent_value, word);
836 if (choices.index(choice_value) != null) return session.fail("DUPLICATE CHOICE {s}", .{word});
837 choices.append(choice_value);
838 }
839 market.preferences[agent_value] = choices;
840 if (market.kind == .two_sided) market.clear();
841 try line(out, "PREF {s} {s} length={d}", .{ name, w[2], choices.len });
842 },
843 .CAPACITY => {
844 const market = try session.get(name, .two_sided);
845 const receiver_value = try session.receiver(market, w[2]);
846 const capacity = integer(w[3]);
847 market.capacities[receiver_value] = capacity;
848 market.clear();
849 try line(out, "CAPACITY {s} {s} {d}", .{ name, w[2], capacity });
850 },
851 .SHOW => try session.show(try session.get(name, null)),
852 .CHECK => try session.check(try session.get(name, null), w),
853 .GS, .GSR, .BOSTON => {
854 const market = try session.get(name, .two_sided);
855 const result = if (command == .BOSTON) boston(market) else try deferred(session.gpa, market, command == .GSR, count == 3);
856 market.save(result.matching, command);
857 try line(out, "{s} {s} {s}={d}", .{ @tagName(command), name, if (command == .BOSTON) "applications" else "proposals", result.proposals });
858 try out.writeAll(result.log);
859 try line(out, "MATCHING {f}", .{Pairs{ .market = market, .matching = result.matching }});
860 },
861 .LATTICE, .EGALITARIAN, .MINREGRET => {
862 const market = try session.get(name, .two_sided);
863 const result = try session.enumerate(market);
864 if (command == .LATTICE) {
865 try line(out, "LATTICE {s} matchings={d} rotations={d}", .{ name, result.matchings.items.len, result.notations.items.len });
866 for (result.matchings.items, 0..) |candidate, i| {
867 try out.print("MATCHING {d} from=", .{i + 1});
868 if (candidate.origin == 0) try out.writeByte('-') else try out.print("{d}", .{candidate.origin});
869 try out.writeAll(" via=");
870 if (candidate.via == 0) try out.writeByte('-') else try out.print("{d}", .{candidate.via});
871 try line(out, " cost={d} regret={d} {f}", .{ candidate.cost, candidate.regret, Pairs{ .market = market, .matching = candidate.matching } });
872 }
873 for (result.notations.items, 0..) |notation, i| try line(out, "ROTATION {d} {s}", .{ i + 1, notation });
874 } else {
875 var best: usize = 0;
876 for (result.matchings.items, 0..) |candidate, i| {
877 if (if (command == .EGALITARIAN) candidate.cost < result.matchings.items[best].cost else candidate.regret < result.matchings.items[best].regret) best = i;
878 }
879 const selected = result.matchings.items[best];
880 market.save(selected.matching, command);
881 try line(out, "{s} {s} matching={d} {s}={d}", .{ @tagName(command), name, best + 1, if (command == .EGALITARIAN) "cost" else "regret", if (command == .EGALITARIAN) selected.cost else selected.regret });
882 try line(out, "MATCHING {f}", .{Pairs{ .market = market, .matching = selected.matching }});
883 }
884 },
885 .HR => try session.hospitals(try session.get(name, .two_sided)),
886 .ROOMMATES => try session.roommates(try session.get(name, .roommates), count == 3),
887 .TTC => try session.tradingCycles(try session.get(name, .housing)),
888 .SERIAL => try session.serial(try session.get(name, .housing), w),
889 .MANIPULATE => {
890 const market = try session.get(name, .two_sided);
891 const agent_value = try session.proposer(market, w[2]);
892 try session.manipulate(market, agent_value);
893 },
894 .STATS => try session.stats(try session.get(name, .two_sided)),
895 }
896 }
897};

899fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
900 var buffer: Writer.Allocating = .init(gpa);
901 defer buffer.deinit();
902 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
903 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
904 const text = try buffer.toOwnedSlice();
905 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
906}
907pub fn main(init: std.process.Init) !void {
908 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
909 defer storage.deinit();
910 var markets: std.ArrayList(Market) = .empty;
911 var input_buffer: [1024]u8 = undefined;
912 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
913 var output_buffer: [4096]u8 = undefined;
914 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
915 const count = integer(try readLine(storage.allocator(), &input.interface));
916 for (0..count) |_| {
917 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
918 defer scratch.deinit();
919 const gpa = scratch.allocator();
920 const text = try readLine(gpa, &input.interface);
921 var response: Writer.Allocating = .init(gpa);
922 defer response.deinit();
923 var session: Session = .{ .gpa = gpa, .storage = storage.allocator(), .markets = &markets, .out = &response.writer };
924 session.execute(text) catch |err| {
925 if (err != error.Invalid) return err;
926 try line(&output.interface, "ERROR {s}", .{session.problem});
927 continue;
928 };
929 try output.interface.writeAll(response.written());
930 }
931 try output.interface.flush();
932}
