36deferred <- function(market, reverse = FALSE, verbose = FALSE) {
37 n <- length(market$names)
38 next_position <- rep.int(1L, n)
39 counts <- integer(n)
40 holding <- rep(list(integer()), n)
41 preference_lengths <- lengths(market$preferences)
42 able <- function(agent) counts[agent] < market$capacities[agent] && next_position[agent] <= preference_lengths[agent]
43 side <- if (reverse) receivers(market) else seq_len(market$proposers)
44 queue <- side[vapply(side, able, logical(1))]
45 proposals <- 0L
46 history <- character()
47 while (length(queue)) {
48 proposer <- queue[1L]
49 queue <- queue[-1L]
50 receiver <- market$preferences[[proposer]][next_position[proposer]]
51 next_position[proposer] <- next_position[proposer] + 1L
52 proposals <- proposals + 1L
53 action <- "REJECT"
54 preference <- market$preferences[[receiver]]
55 if (proposer %in% preference) {
56 if (length(holding[[receiver]]) < market$capacities[receiver]) {
57 holding[[receiver]] <- c(holding[[receiver]], proposer)
58 counts[proposer] <- counts[proposer] + 1L
59 action <- "HOLD"
60 } else {
61 worst <- which.max(rank_in(preference, holding[[receiver]]))
62 rejected <- holding[[receiver]][worst]
63 if (rank_in(preference, proposer) < rank_in(preference, rejected)) {
64 holding[[receiver]][worst] <- proposer
65 counts[rejected] <- counts[rejected] - 1L
66 counts[proposer] <- counts[proposer] + 1L
67 action <- paste("HOLD REJECT", market$names[rejected])
68 if (able(rejected) && !rejected %in% queue) queue <- c(queue, rejected)
69 }
70 }
71 }
72 if (verbose) history <- c(history, paste("PROPOSE", market$names[proposer], market$names[receiver], action))
73 if (able(proposer)) queue <- c(queue, proposer)
74 }
75 matching <- integer(market$proposers)
76 receiving <- if (reverse) seq_len(market$proposers) else receivers(market)
77 for (receiver in receiving) {
78 for (proposer in holding[[receiver]]) {
79 if (reverse) matching[receiver] <- proposer else matching[proposer] <- receiver
80 }
81 }
82 list(matching = matching, proposals = proposals, history = history)
83}

85boston <- function(market) {
86 matching <- integer(market$proposers)
87 remaining <- market$capacities
88 rounds <- max(lengths(market$preferences[seq_len(market$proposers)]))
89 applications <- 0L
90 for (round in seq_len(rounds)) {
91 applied <- integer(market$proposers)
92 for (proposer in seq_len(market$proposers)) {
93 if (matching[proposer] == 0L && length(market$preferences[[proposer]]) >= round) {
94 applied[proposer] <- market$preferences[[proposer]][round]
95 applications <- applications + 1L
96 }
97 }
98 for (receiver in receivers(market)) {
99 for (proposer in market$preferences[[receiver]]) {
100 if (remaining[receiver] > 0L && applied[proposer] == receiver) {
101 matching[proposer] <- receiver
102 remaining[receiver] <- remaining[receiver] - 1L
103 }
104 }
105 }
106 }
107 list(matching = matching, proposals = applications, history = character())
108}

110exposed_rotations <- function(market, matching) {
111 owner <- integer(length(market$names))
112 successor <- integer(market$proposers)
113 matched <- which(matching > 0L)
114 owner[matching[matched]] <- matched
115 for (proposer in matched) {
116 preference <- market$preferences[[proposer]]
117 later <- tail(preference, -rank_in(preference, matching[proposer]))
118 for (receiver in later) {
119 ranking <- market$preferences[[receiver]]
120 if (proposer %in% ranking && (owner[receiver] == 0L || rank_in(ranking, proposer) < rank_in(ranking, owner[receiver]))) {
121 successor[proposer] <- owner[receiver]
122 break
123 }
124 }
125 }
126 covered <- rep(FALSE, market$proposers)
127 rotations <- list()
128 for (start in seq_len(market$proposers)) {
129 if (covered[start]) next
130 path <- integer()
131 point <- start
132 while (point > 0L && !point %in% path) {
133 path <- c(path, point)
134 point <- successor[point]
135 }
136 if (point == start) {
137 rotations[[length(rotations) + 1L]] <- path
138 covered[path] <- TRUE
139 }
140 }
141 rotations
142}

153enumerate_matchings <- function(market) {
154 if (any(market$capacities[receivers(market)] != 1L)) fail("NOT ONE TO ONE")
155 matchings <- list(deferred(market)$matching)
156 origins <- vias <- 0L
157 rotations <- character()
158 index <- 1L
159 while (index <= length(matchings)) {
160 matching <- matchings[[index]]
161 for (rotation in exposed_rotations(market, matching)) {
162 notation <- rotation_text(market, matching, rotation)
163 identifier <- match(notation, rotations, nomatch = 0L)
164 if (identifier == 0L) {
165 rotations <- c(rotations, notation)
166 identifier <- length(rotations)
167 }
168 following <- matching
169 following[rotation] <- matching[c(rotation[-1L], rotation[1L])]
170 if (!any(vapply(matchings, identical, logical(1), following))) {
171 if (length(matchings) == 50L) fail("TOO MANY MATCHINGS")
172 matchings[[length(matchings) + 1L]] <- following
173 origins <- c(origins, index)
174 vias <- c(vias, identifier)
175 }
176 }
177 index <- index + 1L
178 }
179 list(matchings = matchings, origins = origins, vias = vias, rotations = rotations)
180}

182check_pairs <- function(market, words) {
183 if (market$kind == "HOUSES") fail(paste("WRONG TYPE", market$name))
184 matching <- integer(length(market$names))
185 counts <- integer(length(market$names))
186 two <- market$kind == "MARKET"
187 for (word in tail(words, -2L)) {
188 pair <- strsplit(word, "-", fixed = TRUE)[[1L]]
189 first <- agent_index(market, pair[1L], if (two) "PROPOSER" else NULL)
190 second <- agent_index(market, pair[2L], if (two) "RECEIVER" else NULL)
191 if (matching[first] > 0L) fail(paste("DUPLICATE AGENT", pair[1L]))
192 if (two) {
193 counts[second] <- counts[second] + 1L
194 if (counts[second] > market$capacities[second]) fail(paste("OVER CAPACITY", pair[2L]))
195 } else {
196 if (first == second || matching[second] > 0L) fail(paste("DUPLICATE AGENT", pair[2L]))
197 matching[second] <- first
198 }
199 matching[first] <- second
200 }
201 unacceptable <- blocking <- character()
202 if (two) {
203 for (proposer in seq_len(market$proposers)) {
204 receiver <- matching[proposer]
205 if (receiver > 0L && (!receiver %in% market$preferences[[proposer]] || !proposer %in% market$preferences[[receiver]])) {
206 unacceptable <- c(unacceptable, paste("UNACCEPTABLE", market$names[proposer], market$names[receiver]))
207 }
208 for (other in receivers(market)) {
209 if (other == receiver || !other %in% market$preferences[[proposer]] || !proposer %in% market$preferences[[other]]) next
210 if (rank_in(market$preferences[[proposer]], other) >= rank_in(market$preferences[[proposer]], receiver)) next
211 partners <- which(matching[seq_len(market$proposers)] == other)
212 worst <- max(0L, rank_in(market$preferences[[other]], partners))
213 if (length(partners) < market$capacities[other] || rank_in(market$preferences[[other]], proposer) < worst) {
214 blocking <- c(blocking, paste("BLOCKING", market$names[proposer], market$names[other]))
215 }
216 }
217 }
218 } else {
219 for (first in seq_along(market$names)) {
220 for (second in which(seq_along(market$names) > first)) {
221 mutual <- second %in% market$preferences[[first]] && first %in% market$preferences[[second]]
222 if (matching[first] == second) {
223 if (!mutual) unacceptable <- c(unacceptable, paste("UNACCEPTABLE", market$names[first], market$names[second]))
224 } else if (mutual && rank_in(market$preferences[[first]], second) < rank_in(market$preferences[[first]], matching[first]) &&
225 rank_in(market$preferences[[second]], first) < rank_in(market$preferences[[second]], matching[second])) {
226 blocking <- c(blocking, paste("BLOCKING", market$names[first], market$names[second]))
227 }
228 }
229 }
230 }
231 if (!length(unacceptable) && !length(blocking)) return(paste("CHECK", market$name, "STABLE"))
232 c(sprintf("CHECK %s UNSTABLE unacceptable=%d blocking=%d", market$name, length(unacceptable), length(blocking)), unacceptable, blocking)
233}

247roommates <- function(market, verbose) {
248 require_kind(market, "ROOMS")
249 incomplete <- which(lengths(market$preferences) < length(market$names) - 1L)
250 if (length(incomplete)) fail(paste("INCOMPLETE", market$names[incomplete[1L]]))
251 lists <- market$preferences
252 held <- integer(length(market$names))
253 queue <- seq_along(market$names)
254 proposals <- 0L
255 failure <- 0L
256 while (length(queue)) {
257 proposer <- queue[1L]
258 queue <- queue[-1L]
259 receiver <- lists[[proposer]][1L]
260 if (held[receiver] > 0L) queue <- c(queue, held[receiver])
261 held[receiver] <- proposer
262 proposals <- proposals + 1L
263 removed <- tail(lists[[receiver]], -match(proposer, lists[[receiver]]))
264 for (other in removed) lists <- remove_pair(lists, receiver, other)
265 failure <- first_empty(lists)
266 if (failure > 0L) break
267 }
268 history <- if (failure == 0L && verbose) table_lines(market, lists) else character()
269 rotations <- 0L
270 while (failure == 0L) {
271 multiple <- which(lengths(lists) >= 2L)
272 if (!length(multiple)) break
273 point <- multiple[1L]
274 path <- integer()
275 while (!point %in% path) {
276 path <- c(path, point)
277 second <- lists[[point]][2L]
278 point <- tail(lists[[second]], 1L)
279 }
280 rotation <- path[seq.int(match(point, path), length(path))]
281 front <- vapply(rotation, function(agent) lists[[agent]][1L], integer(1))
282 rotations <- rotations + 1L
283 notation <- paste(paste0(market$names[rotation], "-", market$names[front]), collapse = " ")
284 history <- c(history, paste("ROTATION", rotations, notation))
285 removals <- list()
286 for (index in seq_along(rotation)) {
287 receiver <- front[index %% length(front) + 1L]
288 for (other in tail(lists[[receiver]], -match(rotation[index], lists[[receiver]]))) {
289 removals[[length(removals) + 1L]] <- c(receiver, other)
290 }
291 }
292 for (pair in removals) lists <- remove_pair(lists, pair[1L], pair[2L])
293 if (verbose) history <- c(history, table_lines(market, lists))
294 failure <- first_empty(lists)
295 }
296 rows <- c(sprintf("ROOMMATES %s proposals=%d", market$name, proposals), history)
297 if (failure > 0L) return(c(rows, paste("NO STABLE MATCHING", market$names[failure])))
298 partners <- vapply(lists, `[`, integer(1), 1L)
299 first <- which(seq_along(partners) < partners)
300 c(rows, paste("MATCHING", paste(paste0(market$names[first], "-", market$names[partners[first]]), collapse = " ")))
301}

309ttc <- function(market) {
310 remaining <- rep(TRUE, length(market$names))
311 allocation <- integer(length(market$names))
312 effective <- lapply(seq_along(market$names), function(agent) effective_list(market, agent))
313 cycles <- character()

324 allocation[cycle] <- pointing[cycle]
325 remaining[cycle] <- FALSE
326 notation <- paste(paste0(market$names[cycle], ":", market$houses[pointing[cycle]]), collapse = " ")
327 cycles <- c(cycles, paste("CYCLE", length(cycles) + 1L, notation))
328 }
329 c(sprintf("TTC %s cycles=%d", market$name, length(cycles)), cycles, paste("ALLOCATION", allocation_text(market, allocation)))

332serial <- function(market, words) {
333 order <- integer()
334 for (word in tail(words, -2L)) {
335 agent <- agent_index(market, word)
336 if (agent %in% order) fail(paste("DUPLICATE AGENT", word))
337 order <- c(order, agent)
338 }
339 if (length(order) != length(market$names)) fail("BAD ORDER")
340 remaining <- rep(TRUE, length(market$names))
341 allocation <- integer(length(market$names))
342 for (agent in order) {
343 preference <- market$preferences[[agent]]
344 available <- preference[remaining[preference]]
345 if (length(available)) {
346 allocation[agent] <- available[1L]
347 remaining[available[1L]] <- FALSE
348 }
349 }
350 worse <- which(vapply(seq_along(market$names), function(agent) {
351 !allocation[agent] %in% effective_list(market, agent)
352 }, logical(1)))
353 c(sprintf("SERIAL %s assigned=%d", market$name, sum(allocation > 0L)),
354 paste("ALLOCATION", allocation_text(market, allocation)), paste("WORSE", list_text(market$names[worse])))
355}

359 hospital <- deferred(market, TRUE)$matching
360 rows <- c(sprintf("HR %s residents=%d hospitals=%d", market$name, market$proposers, length(market$names) - market$proposers),

366 second <- which(hospital == receiver)
367 same <- identical(first, second)

369 rows <- c(rows, sprintf("HOSPITAL %s cap=%d resident=%d hospital=%d same=%s", market$names[receiver], capacity,
370 length(first), length(second), if (same) "YES" else "NO"))
371 rural <- rural && length(first) == length(second) && (length(first) == capacity || same)
372 }
373 c(rows, if (rural) "RURAL HOLDS" else "RURAL FAILS")

376manipulate <- function(market, proposer) {
377 truth <- market$preferences[[proposer]]
378 baseline_b <- boston(market)$matching[proposer]
379 baseline_g <- deferred(market)$matching[proposer]
380 swaps <- max(0L, length(truth) - 1L)
381 rows <- sprintf("MANIPULATE %s %s boston=%s gs=%s swaps=%d", market$name, market$names[proposer],
382 partner_name(market, baseline_b), partner_name(market, baseline_g), swaps)
383 for (index in seq_len(swaps)) {
384 trial <- market
385 trial$preferences[[proposer]][c(index, index + 1L)] <- truth[c(index + 1L, index)]
386 results <- c(boston(trial)$matching[proposer], deferred(trial)$matching[proposer])
387 baselines <- c(baseline_b, baseline_g)
388 for (kind in seq_len(2L)) {
389 if (rank_in(truth, results[kind]) < rank_in(truth, baselines[kind])) {
390 rows <- c(rows, sprintf("IMPROVE %s %d %s %s result=%s", c("BOSTON", "GS")[kind], index,
391 market$names[truth[index]], market$names[truth[index + 1L]], partner_name(market, results[kind])))
392 }
393 }
394 }
395 if (length(rows) == 1L) rows <- c(rows, "NO IMPROVEMENT")
396 rows

407statistics <- function(market) {
408 saved <- market$current
409 if (is.null(saved)) fail("NO MATCHING")
410 matched <- which(saved$matching > 0L)
411 pairs <- length(matched)
412 proposer_ranks <- vapply(matched, function(p) rank_in(market$preferences[[p]], saved$matching[p]), integer(1))
413 receiver_ranks <- vapply(matched, function(p) rank_in(market$preferences[[saved$matching[p]]], p), integer(1))
414 ranks <- list(proposer_ranks, receiver_ranks)
415 unmatched <- c(market$proposers - pairs, length(market$names) - market$proposers - length(unique(saved$matching[matched])))
416 rows <- sprintf("STATS %s source=%s pairs=%d", market$name, saved$source, pairs)
417 for (side in seq_len(2L)) {
418 total <- sum(ranks[[side]])
419 average <- truncated <- "-"
420 if (pairs > 0L) {
421 divisor <- gcd(total, pairs)
422 average <- sprintf("%d/%d", total %/% divisor, pairs %/% divisor)
423 cents <- (100L * total) %/% pairs
424 truncated <- sprintf("%d.%02d", cents %/% 100L, cents %% 100L)
425 }
426 rows <- c(rows, sprintf("%s sum=%d average=%s truncated=%s first=%d unmatched=%d", c("PROPOSERS", "RECEIVERS")[side],
427 total, average, truncated, sum(ranks[[side]] == 1L), unmatched[side]))
428 }
429 rows

452new_market <- function(words) {
453 command <- words[1L]
454 kind <- if (command == "RANDOM") "MARKET" else command
455 names <- houses <- character()
456 proposers <- 0L
457 if (command == "RANDOM") {
458 proposers <- as.integer(words[3L])
459 names <- c(paste0("P", seq_len(proposers)), paste0("R", seq_len(proposers)))
460 } else {
461 for (word in tail(words, -2L)) {
462 if (word == "|") {
463 proposers <- length(names)
464 next
465 }
466 pair <- strsplit(word, ":", fixed = TRUE)[[1L]]
467 if (pair[1L] %in% names) fail(paste("DUPLICATE AGENT", pair[1L]))
468 if (kind == "HOUSES") {
469 if (pair[2L] %in% houses) fail(paste("DUPLICATE HOUSE", pair[2L]))
470 houses <- c(houses, pair[2L])
471 }
472 names <- c(names, pair[1L])
473 }
474 }
475 market <- list(name = words[2L], kind = kind, names = names, houses = houses, proposers = proposers,
476 preferences = rep(list(integer()), length(names)), capacities = rep.int(1L, length(names)), current = NULL)
477 if (command == "RANDOM") {
478 state <- as.bigz(words[4L])
479 for (agent in seq_along(names)) {
480 preference <- if (agent <= proposers) receivers(market) else seq_len(proposers)
481 for (size in rev(seq_len(proposers)[-1L])) {
482 state <- (48271L * state) %% 2147483647L
483 index <- as.integer(state %% size) + 1L
484 preference[c(size, index)] <- preference[c(index, size)]
485 }
486 market$preferences[[agent]] <- preference
487 }
488 }
489 market
490}

492execute <- function(markets, line) {
493 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
494 command <- words[1L]
495 name <- words[2L]
496 if (command %in% c("MARKET", "ROOMS", "HOUSES", "RANDOM")) {
497 if (exists(name, markets, inherits = FALSE)) fail(paste("DUPLICATE MARKET", name))
498 market <- new_market(words)
499 rows <- if (command == "RANDOM") sprintf("RANDOM %s n=%d seed=%s", name, market$proposers, words[4L]) else {
500 if (market$kind == "MARKET") sprintf("MARKET %s proposers=%d receivers=%d", name, market$proposers, length(market$names) - market$proposers) else {
501 sprintf("%s %s agents=%d", command, name, length(market$names))
502 }
503 }
504 assign(name, market, markets)
505 return(rows)
506 }
507 if (!exists(name, markets, inherits = FALSE)) fail(paste("UNKNOWN MARKET", name))
508 market <- get(name, markets, inherits = FALSE)
509 if (command == "SHOW") return(show_market(market))
510 if (command == "PREF") {
511 agent <- agent_index(market, words[3L])
512 names <- if (market$kind == "HOUSES") market$houses else market$names
513 preference <- integer()
514 for (word in tail(words, -3L)) {
515 choice <- match(word, names, nomatch = 0L)
516 if (choice == 0L || (market$kind == "ROOMS" && choice == agent) ||
517 (market$kind == "MARKET" && (agent <= market$proposers) == (choice <= market$proposers))) fail(paste("BAD CHOICE", word))
518 if (choice %in% preference) fail(paste("DUPLICATE CHOICE", word))
519 preference <- c(preference, choice)
520 }
521 market$preferences[[agent]] <- preference
522 if (market$kind == "MARKET") market$current <- NULL
523 rows <- sprintf("PREF %s %s length=%d", name, words[3L], length(preference))
524 } else if (command == "CHECK") {
525 return(check_pairs(market, words))
526 } else if (command == "ROOMMATES") {
527 return(roommates(market, length(words) == 3L))
528 } else if (command %in% c("TTC", "SERIAL")) {
529 require_kind(market, "HOUSES")
530 return(if (command == "TTC") ttc(market) else serial(market, words))
531 } else {
532 require_kind(market, "MARKET")
533 if (command == "CAPACITY") {
534 receiver <- agent_index(market, words[3L], "RECEIVER")
535 capacity <- as.integer(words[4L])
536 market$capacities[receiver] <- capacity
537 market$current <- NULL
538 rows <- sprintf("CAPACITY %s %s %d", name, words[3L], capacity)
539 } else if (command %in% c("GS", "GSR", "BOSTON")) {
540 run <- if (command == "BOSTON") boston(market) else deferred(market, command == "GSR", length(words) == 3L)
541 market$current <- list(source = command, matching = run$matching)
542 rows <- c(sprintf("%s %s %s=%d", command, name, if (command == "BOSTON") "applications" else "proposals", run$proposals),
543 run$history, paste("MATCHING", pair_text(market, run$matching)))
544 } else if (command == "HR") {
545 return(hospital_report(market))
546 } else if (command == "MANIPULATE") {
547 return(manipulate(market, agent_index(market, words[3L], "PROPOSER")))
548 } else if (command == "STATS") {
549 return(statistics(market))
550 } else {
551 lattice <- enumerate_matchings(market)
552 if (command == "LATTICE") {
553 rows <- sprintf("LATTICE %s matchings=%d rotations=%d", name, length(lattice$matchings), length(lattice$rotations))
554 for (index in seq_along(lattice$matchings)) {
555 matching <- lattice$matchings[[index]]
556 metrics <- matching_metrics(market, matching)
557 origin <- if (index == 1L) "-" else as.character(lattice$origins[index])
558 via <- if (index == 1L) "-" else as.character(lattice$vias[index])
559 rows <- c(rows, sprintf("MATCHING %d from=%s via=%s cost=%d regret=%d %s", index, origin, via,
560 metrics["cost"], metrics["regret"], pair_text(market, matching)))
561 }
562 for (index in seq_along(lattice$rotations)) rows <- c(rows, paste("ROTATION", index, lattice$rotations[index]))
563 return(rows)
564 }
565 criterion <- if (command == "EGALITARIAN") "cost" else "regret"
566 values <- vapply(lattice$matchings, function(matching) matching_metrics(market, matching)[criterion], integer(1))
567 best <- which.min(values)
568 market$current <- list(source = command, matching = lattice$matchings[[best]])
569 rows <- c(sprintf("%s %s matching=%d %s=%d", command, name, best, criterion, values[best]),
570 paste("MATCHING", pair_text(market, market$current$matching)))
571 }
572 }
573 assign(name, market, markets)
574 rows
575}

581 rows <- tryCatch(execute(markets, input[index + 1L]), command_error = function(problem) paste("ERROR", conditionMessage(problem)))
582 cat(paste(rows, collapse = "\n"), "\n", sep = "")
583 }
584}
