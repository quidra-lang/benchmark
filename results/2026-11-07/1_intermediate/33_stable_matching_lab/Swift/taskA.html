18func rank(_ preference: Preference,
19 _ partner: Int) -> Int
20{
21 preference.firstIndex(of: partner).map { $0 + 1 } ?? (preference.count + 1)

23func listText(_ names: [String],
24 _ indices: [Int]) -> String
25{
26 indices.isEmpty ? "-" : indices.map { names[$0] }.joined(separator: " ")

28struct Market {

38 init(_ name: String, _ kind: MarketKind, _ names: [String], proposers: Int = 0, houses: [String] = []) {
39 self.name = name
40 self.kind = kind
41 self.names = names
42 self.proposers = proposers
43 self.houses = houses
44 preferences = names.map { _ in [] }
45 capacities = names.map { _ in 1 }
46 current = nil
47 }
48 func requireKind(_ expected: MarketKind) throws {
49 if kind != expected {
50 try fail("WRONG TYPE \(name)")

53 func agent(_ name: String, side: Side? = nil) throws -> Int {
54 guard let index = names.firstIndex(of: name) else {
55 try fail("UNKNOWN AGENT \(name)")
56 }
57 if side == .proposer && index >= proposers {
58 try fail("NOT PROPOSER \(name)")
59 }
60 if side == .receiver && index < proposers {
61 try fail("NOT RECEIVER \(name)")
62 }
63 return index
64 }
65 func effective(_ agent: Int) -> Preference {
66 let preference = preferences[agent]
67 if let own = preference.firstIndex(of: agent) {
68 return Array(preference.prefix(own + 1))
69 }
70 return preference + [agent]
71 }
72 func partnerName(_ partner: Int) -> String {
73 partner < 0 ? "*" : names[partner]
74 }

90struct DeferredResult { let matching: Matching
91 let proposals: Int
92 let history: [String]
93}
94func deferred(_ market: Market, reverse: Bool = false, verbose: Bool = false) -> DeferredResult {
95 let count = market.names.count
96 var next = Array(repeating: 0, count: count), counts = Array(repeating: 0, count: count)
97 var holding = Array(repeating: [Int](), count: count), queued = Array(repeating: false, count: count)
98 let able = { (agent: Int) in
99 counts[agent] < market.capacities[agent] && next[agent] < market.preferences[agent].count
100 }
101 let proposing = reverse ? Array(market.proposers..<count) : Array(0..<market.proposers)
102 var queue = proposing.filter(able), history: [String] = []
103 for agent in queue {
104 queued[agent] = true
105 }
106 var head = 0, proposals = 0
107 while head < queue.count {
108 let proposer = queue[head]
109 head += 1
110 queued[proposer] = false
111 let receiver = market.preferences[proposer][next[proposer]]
112 let preference = market.preferences[receiver]
113 next[proposer] += 1
114 proposals += 1
115 var action = "REJECT"
116 if preference.contains(proposer) {
117 if holding[receiver].count < market.capacities[receiver] {
118 holding[receiver].append(proposer)
119 counts[proposer] += 1
120 action = "HOLD"
121 } else {
122 let worst = holding[receiver].indices.max { rank(preference, holding[receiver][$0]) < rank(
123 preference,
124 holding[receiver][$1]
125 ) }!
126 let rejected = holding[receiver][worst]
127 if rank(preference, proposer) < rank(preference, rejected) {
128 holding[receiver][worst] = proposer
129 counts[proposer] += 1
130 counts[rejected] -= 1
131 action = "HOLD REJECT \(market.names[rejected])"
132 if able(rejected) && !queued[rejected] {
133 queue.append(rejected)
134 queued[rejected] = true
135 }
136 }
137 }
138 }
139 if verbose {
140 history.append("PROPOSE \(market.names[proposer]) \(market.names[receiver]) \(action)")
141 }
142 if able(proposer) {
143 queue.append(proposer)
144 queued[proposer] = true
145 }
146 }
147 var matching = Array(repeating: -1, count: market.proposers)
148 let receiving = reverse ? Array(0..<market.proposers) : Array(market.proposers..<count)
149 for receiver in receiving {
150 for proposer in holding[receiver] {
151 if reverse {
152 matching[receiver] = proposer
153 } else {
154 matching[proposer] = receiver
155 }
156 }
157 }
158 return DeferredResult(matching: matching, proposals: proposals, history: history)
159}
160struct BostonResult { let matching: Matching
161 let applications: Int
162}
163func boston(_ market: Market) -> BostonResult {
164 var matching = Array(repeating: -1, count: market.proposers), remaining = market.capacities
165 let rounds = market.preferences.prefix(market.proposers).map(\.count).max()!
166 var applications = 0
167 for round in 0..<rounds {
168 var applied = Array(repeating: -1, count: market.proposers)
169 for proposer in 0..<market.proposers
170 where matching[proposer] < 0 && round < market.preferences[proposer].count
171 {
172 applied[proposer] = market.preferences[proposer][round]
173 applications += 1
174 }
175 for receiver in market.proposers..<market.names.count {
176 for proposer in market.preferences[receiver]
177 where remaining[receiver] > 0 && applied[proposer] == receiver
178 {
179 matching[proposer] = receiver
180 remaining[receiver] -= 1
181 }
182 }
183 }
184 return BostonResult(matching: matching, applications: applications)
185}

187func exposedRotations(_ market: Market, _ matching: Matching) -> [[Int]] {
188 var owner = Array(repeating: -1, count: market.names.count), successor = Array(
189 repeating: -1,
190 count: market.proposers
191 )
192 for proposer in 0..<market.proposers where matching[proposer] >= 0 {
193 owner[matching[proposer]] = proposer
194 }
195 for proposer in 0..<market.proposers {
196 if matching[proposer] < 0 {
197 continue
198 }
199 let preference = market.preferences[proposer]
200 for receiver in preference.dropFirst(rank(preference, matching[proposer])) {
201 let ranking = market.preferences[receiver]
202 if ranking.contains(proposer) && (owner[receiver] < 0 || rank(ranking, proposer) < rank(
203 ranking,
204 owner[receiver]
205 )) {
206 successor[proposer] = owner[receiver]
207 break
208 }
209 }
210 }
211 var covered: Set<Int> = [], rotations: [[Int]] = []
212 for start in 0..<market.proposers where !covered.contains(start) {
213 var path: [Int] = [], point = start
214 while point >= 0 && !path.contains(point) {
215 path.append(point)
216 point = successor[point]
217 }
218 if point == start {
219 covered.formUnion(path)
220 rotations.append(path)
221 }
222 }
223 return rotations
224}
225struct Metrics { let cost: Int
226 let regret: Int
227}
228func metrics(_ market: Market, _ matching: Matching) -> Metrics {
229 var cost = 0, regret = 0
230 for proposer in 0..<market.proposers {
231 let receiver = matching[proposer]
232 if receiver < 0 {
233 continue
234 }
235 let first = rank(market.preferences[proposer], receiver), second = rank(market.preferences[receiver], proposer)
236 cost += first + second
237 regret = max(regret, max(first, second))
238 }
239 return Metrics(cost: cost, regret: regret)

241struct Entry { let matching: Matching
242 let origin: Int?
243 let via: Int?
244}
245struct Lattice { let entries: [Entry]
246 let rotations: [String]
247}
248func enumerate(_ market: Market) throws -> Lattice {
249 guard market.capacities.dropFirst(market.proposers).allSatisfy({ $0 == 1 }) else {
250 try fail("NOT ONE TO ONE")
251 }
252 let initial = deferred(market).matching
253 var entries = [Entry(matching: initial, origin: nil, via: nil)], rotations: [String] = []
254 var seen: Set<Matching> = [initial], index = 0
255 while index < entries.count {
256 let matching = entries[index].matching
257 for rotation in exposedRotations(market, matching) {
258 let notation = rotationText(market, matching, rotation)
259 let identifier: Int
260 if let found = rotations.firstIndex(of: notation) {
261 identifier = found
262 } else {
263 identifier = rotations.count
264 rotations.append(notation)
265 }
266 var following = matching
267 for i in rotation.indices {
268 following[rotation[i]] = matching[rotation[(i + 1) % rotation.count]]
269 }
270 if seen.contains(following) {
271 continue
272 }
273 if entries.count == 50 {
274 try fail("TOO MANY MATCHINGS")
275 }
276 seen.insert(following)
277 entries.append(Entry(matching: following, origin: index + 1, via: identifier + 1))
278 }
279 index += 1
280 }
281 return Lattice(entries: entries, rotations: rotations)
282}
283func checkPairs(_ market: Market, _ pairs: [String]) throws -> [String] {
284 if market.kind == .housing {
285 try fail("WRONG TYPE \(market.name)")
286 }
287 var matching = Array(repeating: -1, count: market.names.count), counts = Array(
288 repeating: 0,
289 count: market.names.count
290 )
291 for word in pairs {
292 let pair = word.split(separator: "-").map(String.init)
293 let first = try market.agent(pair[0], side: market.kind == .twoSided ? .proposer : nil)
294 let second = try market.agent(pair[1], side: market.kind == .twoSided ? .receiver : nil)
295 if matching[first] >= 0 {
296 try fail("DUPLICATE AGENT \(pair[0])")
297 }
298 if market.kind == .twoSided {
299 counts[second] += 1
300 if counts[second] > market.capacities[second] {
301 try fail("OVER CAPACITY \(pair[1])")
302 }
303 } else {
304 if first == second || matching[second] >= 0 {
305 try fail("DUPLICATE AGENT \(pair[1])")
306 }
307 matching[second] = first
308 }
309 matching[first] = second
310 }
311 var unacceptable: [String] = [], blocking: [String] = []
312 if market.kind == .twoSided {
313 for proposer in 0..<market.proposers {
314 let receiver = matching[proposer], preference = market.preferences[proposer]
315 if receiver >= 0 && (!preference.contains(receiver) || !market.preferences[receiver].contains(proposer)) {
316 unacceptable.append("UNACCEPTABLE \(market.names[proposer]) \(market.names[receiver])")
317 }
318 for other in market.proposers..<market.names.count {
319 let ranking = market.preferences[other]
320 if receiver == other || !preference.contains(other) || !ranking.contains(proposer) {
321 continue
322 }
323 if rank(preference, other) >= rank(preference, receiver) {
324 continue
325 }
326 var worst = 0
327 for partner in 0..<market.proposers where matching[partner] == other {
328 worst = max(
329 worst,
330 rank(ranking, partner)
331 )
332 }
333 if counts[other] < market.capacities[other] || rank(ranking, proposer) < worst {
334 blocking.append("BLOCKING \(market.names[proposer]) \(market.names[other])")
335 }
336 }
337 }
338 } else {
339 for first in market.names.indices {
340 for second in (first + 1)..<market.names.count {
341 let mutual = market.preferences[first].contains(second) && market.preferences[second].contains(first)
342 let notation = "\(market.names[first]) \(market.names[second])"
343 if matching[first] == second {
344 if !mutual {
345 unacceptable.append("UNACCEPTABLE \(notation)")
346 }
347 } else if mutual && rank(market.preferences[first], second) < rank(
348 market.preferences[first],
349 matching[first]
350 )
351 && rank(market.preferences[second], first) < rank(market.preferences[second], matching[second])
352 {
353 blocking.append("BLOCKING \(notation)")
354 }
355 }
356 }
357 }
358 if unacceptable.isEmpty && blocking.isEmpty {
359 return ["CHECK \(market.name) STABLE"]
360 }
361 return ["CHECK \(market.name) UNSTABLE unacceptable=\(unacceptable.count) blocking=\(blocking.count)"] +
362 unacceptable + blocking
363}
364func removePair(_ lists: inout [Preference], _ first: Int, _ second: Int) {
365 lists[first].removeAll { $0 == second }
366 lists[second].removeAll { $0 == first }
367}

370}

372func roommates(_ market: Market, verbose: Bool) throws -> [String] {
373 try market.requireKind(.roommate)
374 let count = market.names.count
375 for agent in market.names.indices
376 where market.preferences[agent].count < count - 1
377 {
378 try fail("INCOMPLETE \(market.names[agent])")
379 }
380 var lists = market.preferences, held = Array(repeating: -1, count: count), queue = Array(0..<count)
381 var head = 0, proposals = 0, failure: Int?
382 while head < queue.count {
383 let proposer = queue[head]
384 head += 1
385 let receiver = lists[proposer][0]
386 if held[receiver] >= 0 {
387 queue.append(held[receiver])
388 }
389 held[receiver] = proposer
390 proposals += 1
391 let preference = lists[receiver]
392 for other in preference
393 .dropFirst(preference.firstIndex(of: proposer)! + 1)
394 {
395 removePair(&lists, receiver, other)
396 }
397 failure = lists.firstIndex(where: { $0.isEmpty })
398 if failure != nil {
399 break
400 }
401 }
402 var history = failure == nil && verbose ? tableLines(market, lists) : []
403 var rotations = 0
404 while failure == nil {
405 guard let start = lists.firstIndex(where: { $0.count >= 2 }) else {
406 break
407 }
408 var path: [Int] = [], point = start
409 while !path.contains(point) {
410 path.append(point)
411 point = lists[lists[point][1]].last!
412 }
413 let rotation = Array(path.dropFirst(path.firstIndex(of: point)!))
414 let front = rotation.map { lists[$0][0] }
415 rotations += 1
416 let pairs = rotation.indices.map { "\(market.names[rotation[$0]])-\(market.names[front[$0]])" }
417 history.append("ROTATION \(rotations) \(pairs.joined(separator: " "))")
418 var removals: [(Int, Int)] = []
419 for index in rotation.indices {
420 let receiver = front[(index + 1) % front.count], agent = rotation[index]
421 let preference = lists[receiver]
422 for other in preference
423 .dropFirst(preference.firstIndex(of: agent)! + 1)
424 {
425 removals.append((receiver, other))
426 }
427 }
428 for (first, second) in removals {
429 removePair(&lists, first, second)
430 }
431 if verbose {
432 history += tableLines(market, lists)
433 }
434 failure = lists.firstIndex(where: { $0.isEmpty })
435 }
436 let output = ["ROOMMATES \(market.name) proposals=\(proposals)"] + history
437 if let agent = failure {
438 return output + ["NO STABLE MATCHING \(market.names[agent])"]
439 }
440 let pairs = market.names.indices.filter { $0 < lists[$0][0] }
441 .map { "\(market.names[$0])-\(market.names[lists[$0][0]])" }
442 return output + ["MATCHING \(pairs.joined(separator: " "))"]
443}
444func ttc(_ market: Market) -> [String] {
445 var remaining = Set(market.names.indices), allocation = Array(repeating: -1, count: market.names.count),
446 cycles: [String] = []
447 while !remaining.isEmpty {
448 var pointing = Array(repeating: -1, count: market.names.count)
449 for agent in market.names.indices where remaining.contains(agent) {
450 pointing[agent] = market.effective(agent).first(where: { remaining.contains($0) })!
451 }
452 var path: [Int] = [], point = remaining.min()!
453 while !path.contains(point) {
454 path.append(point)
455 point = pointing[point]
456 }
457 let cycle = Array(path.dropFirst(path.firstIndex(of: point)!))
458 for agent in cycle {
459 allocation[agent] = pointing[agent]
460 remaining.remove(agent)
461 }
462 let parts = cycle.map { "\(market.names[$0]):\(market.houses[allocation[$0]])" }
463 cycles.append("CYCLE \(cycles.count + 1) \(parts.joined(separator: " "))")
464 }
465 return ["TTC \(market.name) cycles=\(cycles.count)"] + cycles + ["ALLOCATION \(allocationText(market, allocation))"]
466}
467func serial(_ market: Market, _ names: [String]) throws -> [String] {
468 var order: [Int] = []
469 for name in names {
470 let agent = try market.agent(name)
471 if order.contains(agent) {
472 try fail("DUPLICATE AGENT \(name)")
473 }
474 order.append(agent)
475 }
476 guard order.count == market.names.count else {
477 try fail("BAD ORDER")
478 }
479 var remaining = Set(market.houses.indices), allocation = Array(repeating: -1, count: market.names.count),
480 assigned = 0
481 for agent in order {
482 if let house = market.preferences[agent].first(where: { remaining.contains($0) }) {
483 allocation[agent] = house
484 remaining.remove(house)
485 assigned += 1
486 }
487 }
488 let worse = market.names.indices.filter { !market.effective($0).contains(allocation[$0]) }
489 return [
490 "SERIAL \(market.name) assigned=\(assigned)",
491 "ALLOCATION \(allocationText(market, allocation))",
492 "WORSE \(listText(market.names, worse))"
493 ]
494}
495func hospitalReport(_ market: Market) -> [String] {
496 let resident = deferred(market).matching, hospital = deferred(market, reverse: true).matching
497 let unmatched = (0..<market.proposers).filter { resident[$0] < 0 }
498 var rural = (0..<market.proposers).allSatisfy { (resident[$0] >= 0) == (hospital[$0] >= 0) }
499 var output = ["HR \(market.name) residents=\(market.proposers) hospitals=\(market.names.count - market.proposers)",
500 "RESIDENT-OPTIMAL \(pairText(market, resident))", "HOSPITAL-OPTIMAL \(pairText(market, hospital))",
501 "UNMATCHED \(listText(market.names, unmatched))"]
502 for receiver in market.proposers..<market.names.count {
503 let first = partnersOf(resident, receiver), second = partnersOf(hospital, receiver),
504 capacity = market.capacities[receiver]
505 let same = first == second
506 output
507 .append(
508 "HOSPITAL \(market.names[receiver]) cap=\(capacity) resident=\(first.count) hospital=\(second.count) same=\(same ? "YES" : "NO")"
509 )
510 if first.count != second.count || (first.count < capacity && !same) {
511 rural = false
512 }
513 }
514 return output + [rural ? "RURAL HOLDS" : "RURAL FAILS"]
515}
516func manipulate(_ market: Market, _ proposer: Int) -> [String] {
517 let truth = market.preferences[proposer]
518 let baselineB = boston(market).matching[proposer], baselineG = deferred(market).matching[proposer]
519 let swaps = max(0, truth.count - 1)
520 var output =
521 [
522 "MANIPULATE \(market.name) \(market.names[proposer]) boston=\(market.partnerName(baselineB)) gs=\(market.partnerName(baselineG)) swaps=\(swaps)"
523 ]
524 for index in 0..<swaps {
525 var trial = market
526 trial.preferences[proposer].swapAt(index, index + 1)
527 let outcomes = [
528 ("BOSTON", boston(trial).matching[proposer], baselineB),
529 ("GS", deferred(trial).matching[proposer], baselineG)
530 ]
531 for (label, partner, baseline) in outcomes where rank(truth, partner) < rank(truth, baseline) {
532 output
533 .append(
534 "IMPROVE \(label) \(index + 1) \(market.names[truth[index]]) \(market.names[truth[index + 1]]) result=\(market.partnerName(partner))"
535 )
536 }
537 }
538 if output.count == 1 {
539 output.append("NO IMPROVEMENT")
540 }
541 return output
542}
543func gcd(_ first: Int, _ second: Int) -> Int {
544 var a = first, b = second
545 while b != 0 {
546 (a, b) = (b, a % b)
547 }
548 return a
549}
550func statistics(_ market: Market) throws -> [String] {
551 guard let saved = market.current else {
552 try fail("NO MATCHING")
553 }
554 var proposerRanks: [Int] = [], receiverRanks: [Int] = [], occupied: Set<Int> = []
555 for proposer in 0..<market.proposers {
556 let receiver = saved.matching[proposer]
557 if receiver < 0 {
558 continue
559 }
560 proposerRanks.append(rank(market.preferences[proposer], receiver))
561 receiverRanks.append(rank(market.preferences[receiver], proposer))
562 occupied.insert(receiver)
563 }
564 let pairs = proposerRanks.count
565 var output = ["STATS \(market.name) source=\(saved.source.rawValue) pairs=\(pairs)"]
566 let summaries = [
567 ("PROPOSERS", proposerRanks, market.proposers - pairs),
568 ("RECEIVERS", receiverRanks, market.names.count - market.proposers - occupied.count)
569 ]
570 for (label, ranks, unmatched) in summaries {
571 let sum = ranks.reduce(0, +), first = ranks.filter { $0 == 1 }.count
572 var average = "-", truncated = "-"
573 if pairs > 0 {
574 let divisor = gcd(sum, pairs), cents = 100 * sum / pairs
575 average = "\(sum / divisor)/\(pairs / divisor)"
576 truncated = "\(cents / 100).\(cents % 100 < 10 ? "0" : "")\(cents % 100)"
577 }
578 output
579 .append(
580 "\(label) sum=\(sum) average=\(average) truncated=\(truncated) first=\(first) unmatched=\(unmatched)"
581 )
582 }
583 return output
584}
585func showMarket(_ market: Market) -> [String] {
586 let detail = market
587 .kind == .twoSided ? "proposers=\(market.proposers) receivers=\(market.names.count - market.proposers)" : "agents=\(market.names.count)"
588 var rows = ["SHOW \(market.name) \(market.kind.rawValue) \(detail)"]
589 for agent in market.names.indices {
590 var prefix: String
591 if market.kind == .twoSided {
592 prefix = agent < market.proposers ? "PROPOSER \(market.names[agent])" : "RECEIVER \(market.names[agent]) cap=\(market.capacities[agent])"
593 } else {
594 prefix = "AGENT \(market.names[agent])"
595 if market.kind == .housing {
596 prefix += " house=\(market.houses[agent])"
597 }
598 }
599 rows.append(prefix + " " + listText(
600 market.kind == .housing ? market.houses : market.names,
601 market.preferences[agent]
602 ))
603 }
604 if market.kind == .twoSided {
605 rows.append("CURRENT \(market.current?.source.rawValue ?? "-")")
606 }
607 return rows

609func newMarket(_ command: String, _ name: String, _ args: [String]) throws -> Market {
610 if command == "RANDOM" {
611 let count = Int(args[0])!
612 var state = Int64(args[1])!
613 let names = ["P", "R"].flatMap { prefix in (1...count).map { "\(prefix)\($0)" } }
614 var market = Market(name, .twoSided, names, proposers: count)
615 for agent in names.indices {
616 var preference = agent < count ? Array(count..<(2 * count)) : Array(0..<count)
617 for length in stride(from: count, through: 2, by: -1) {
618 state = 48271 * state % 2147483647
619 preference.swapAt(length - 1, Int(state % Int64(length)))
620 }
621 market.preferences[agent] = preference
622 }
623 return market
624 }
625 let kind: MarketKind = command == "ROOMS" ? .roommate : command == "HOUSES" ? .housing : .twoSided
626 var names: [String] = [], houses: [String] = [], proposers = 0
627 for word in args {
628 if word == "|" {
629 proposers = names.count
630 continue
631 }
632 let pair = word.split(separator: ":").map(String.init)
633 if names.contains(pair[0]) {
634 try fail("DUPLICATE AGENT \(pair[0])")
635 }
636 if kind == .housing {
637 if houses.contains(pair[1]) {
638 try fail("DUPLICATE HOUSE \(pair[1])")
639 }
640 houses.append(pair[1])
641 }
642 names.append(pair[0])
643 }
644 return Market(name, kind, names, proposers: proposers, houses: houses)
645}
646func executeMarket(_ market: inout Market, _ command: String, _ args: [String]) throws -> [String] {
647 let name = market.name
648 if command == "SHOW" {
649 return showMarket(market)
650 }
651 if command == "PREF" {
652 let agent = try market.agent(args[0])
653 let names = market.kind == .housing ? market.houses : market.names
654 var preference: Preference = []
655 for word in args.dropFirst() {
656 guard let choice = names.firstIndex(of: word) else {
657 try fail("BAD CHOICE \(word)")
658 }
659 if (market.kind == .roommate && choice == agent) ||
660 (market.kind == .twoSided && ((agent < market.proposers) == (choice < market.proposers)))
661 {
662 try fail("BAD CHOICE \(word)")
663 }
664 if preference.contains(choice) {
665 try fail("DUPLICATE CHOICE \(word)")
666 }
667 preference.append(choice)
668 }
669 market.preferences[agent] = preference
670 if market.kind == .twoSided {
671 market.current = nil
672 }
673 return ["PREF \(name) \(args[0]) length=\(preference.count)"]
674 }
675 if command == "CHECK" {
676 return try checkPairs(market, args)
677 }
678 if command == "ROOMMATES" {
679 return try roommates(market, verbose: args.count == 1)
680 }
681 if command == "TTC" || command == "SERIAL" {
682 try market.requireKind(.housing)
683 if command == "TTC" {
684 return ttc(market)
685 }
686 return try serial(market, args)
687 }
688 try market.requireKind(.twoSided)
689 if command == "CAPACITY" {
690 let receiver = try market.agent(args[0], side: .receiver), capacity = Int(args[1])!
691 market.capacities[receiver] = capacity
692 market.current = nil
693 return ["CAPACITY \(name) \(args[0]) \(capacity)"]
694 }
695 if command == "GS" || command == "GSR" {
696 let run = deferred(market, reverse: command == "GSR", verbose: args.count == 1)
697 market.current = Saved(source: command == "GS" ? .gs : .gsr, matching: run.matching)
698 return ["\(command) \(name) proposals=\(run.proposals)"] + run
699 .history + ["MATCHING \(pairText(market, run.matching))"]
700 }
701 if command == "BOSTON" {
702 let run = boston(market)
703 market.current = Saved(source: .boston, matching: run.matching)
704 return ["BOSTON \(name) applications=\(run.applications)", "MATCHING \(pairText(market, run.matching))"]
705 }
706 if command == "HR" {
707 return hospitalReport(market)
708 }
709 if command == "MANIPULATE" {
710 return try manipulate(market, market.agent(args[0], side: .proposer))
711 }
712 if command == "STATS" {
713 return try statistics(market)
714 }
715 let lattice = try enumerate(market)
716 if command == "LATTICE" {
717 var output = ["LATTICE \(name) matchings=\(lattice.entries.count) rotations=\(lattice.rotations.count)"]
718 for (index, entry) in lattice.entries.enumerated() {
719 let measure = metrics(market, entry.matching)
720 let origin = entry.origin.map(String.init) ?? "-", via = entry.via.map(String.init) ?? "-"
721 output
722 .append(
723 "MATCHING \(index + 1) from=\(origin) via=\(via) cost=\(measure.cost) regret=\(measure.regret) \(pairText(market, entry.matching))"
724 )
725 }
726 for (index, rotation) in lattice.rotations.enumerated() {
727 output.append("ROTATION \(index + 1) \(rotation)")
728 }
729 return output
730 }
731 let values = lattice.entries.map { entry -> Int in
732 let measure = metrics(market, entry.matching)
733 return command == "EGALITARIAN" ? measure.cost : measure.regret
734 }
735 let best = values.indices.min { values[$0] < values[$1] }!
736 let matching = lattice.entries[best].matching, label = command == "EGALITARIAN" ? "cost" : "regret"
737 market.current = Saved(source: command == "EGALITARIAN" ? .egalitarian : .minRegret, matching: matching)
738 return [
739 "\(command) \(name) matching=\(best + 1) \(label)=\(values[best])",
740 "MATCHING \(pairText(market, matching))"
741 ]
742}
743final class Laboratory {
744 private var markets: [String: Market] = [:]
745
746 func execute(_ line: String) throws -> [String] {
747 let words = line.split(separator: " ").map(String.init)
748 let command = words[0], name = words[1], args = Array(words.dropFirst(2))
749 if ["MARKET", "ROOMS", "HOUSES", "RANDOM"].contains(command) {
750 if markets[name] != nil {
751 try fail("DUPLICATE MARKET \(name)")
752 }
753 let market = try newMarket(command, name, args)
754 let response: String
755 if command == "RANDOM" {
756 response = "RANDOM \(name) n=\(market.proposers) seed=\(args[1])"
757 } else if command ==
758 "MARKET"
759 {
760 response =
761 "MARKET \(name) proposers=\(market.proposers) receivers=\(market.names.count - market.proposers)"
762 } else {
763 response = "\(command) \(name) agents=\(market.names.count)"
764 }
765 markets[name] = market
766 return [response]
767 }
768 guard var market = markets[name] else {
769 try fail("UNKNOWN MARKET \(name)")
770 }
771 let output = try executeMarket(&market, command, args)
772 markets[name] = market
773 return output
774 }
775}
776
777let count = Int(readLine()!)!
778let laboratory = Laboratory()
779var output: [String] = []
780for _ in 0..<count {
781 do { output += try laboratory.execute(readLine()!) }
782 catch let error as CommandError { output.append("ERROR \(error.message)") }
783 catch { fatalError("Unexpected error: \(error)") }
784}
785print(output.joined(separator: "\n"))
