1local function fail(message) error("ERROR " .. message, 0) end
2local function copy(values)
3 local result = {}
4 for i, value in ipairs(values) do result[i] = value end
5 return result
6end
7local function filled(count, value)
8 local result = {}
9 for i = 1, count do result[i] = value end
10 return result
11end
12local function slice(values, first, last)
13 local result = {}
14 for i = first, last or #values do result[#result + 1] = values[i] end
15 return result
16end
17local function indexOf(values, value)
18 for i, item in ipairs(values) do if item == value then return i end end
19 return 0
20end
21local function sameValues(first, second)
22 if #first ~= #second then return false end
23 for i, value in ipairs(first) do if value ~= second[i] then return false end end
24 return true
25end
26local function splitPair(text, separator)
27 local position = text:find(separator, 1, true)
28 return text:sub(1, position - 1), text:sub(position + 1)
29end
30local function listText(values) return #values == 0 and "-" or table.concat(values, " ") end
31local function rankOf(preference, partner)
32 local position = indexOf(preference, partner)
33 return position == 0 and #preference + 1 or position
34end

38function Market.new(name, kind, agents, proposerCount, houses)
39 local preferences = {}
40 for a = 1, #agents do preferences[a] = {} end
41 return setmetatable({name = name, kind = kind, agents = agents, count = #agents,
42 proposerCount = proposerCount or 0, houses = houses or {}, preferences = preferences,
43 capacities = filled(#agents, 1), current = false, source = "-"}, Market)
44end
45function Market:agent(name)
46 local agent = indexOf(self.agents, name)
47 if agent == 0 then fail("UNKNOWN AGENT " .. name) end
48 return agent
49end
50function Market:proposer(name)
51 local agent = self:agent(name)
52 if agent > self.proposerCount then fail("NOT PROPOSER " .. name) end
53 return agent
54end
55function Market:receiver(name)
56 local agent = self:agent(name)
57 if agent <= self.proposerCount then fail("NOT RECEIVER " .. name) end
58 return agent
59end
60function Market:choice(agent, name)
61 local choice = indexOf(self.kind == "HOUSES" and self.houses or self.agents, name)
62 local valid = choice > 0
63 if self.kind == "MARKET" then
64 valid = valid and ((agent <= self.proposerCount) ~= (choice <= self.proposerCount))
65 elseif self.kind == "ROOMS" then
66 valid = valid and choice ~= agent
67 end
68 if not valid then fail("BAD CHOICE " .. name) end
69 return choice
70end
71function Market:clearMatching() self.current, self.source = false, "-" end
72function Market:setMatching(partners, source) self.current, self.source = copy(partners), source end
73function Market:partner(agent) return agent == 0 and "*" or self.agents[agent] end
74function Market:pairs(partners)
75 local parts = {}
76 for p = 1, self.proposerCount do parts[p] = self.agents[p] .. "-" .. self:partner(partners[p]) end
77 return table.concat(parts, " ")
78end
79function Market:allocation(allocation)
80 local parts = {}
81 for a, house in ipairs(allocation) do
82 parts[a] = self.agents[a] .. "-" .. (house == 0 and "*" or self.houses[house])
83 end
84 return table.concat(parts, " ")
85end
86function Market:effective(agent)
87 local preference = self.preferences[agent]
88 local own = indexOf(preference, agent)
89 if own > 0 then return slice(preference, 1, own) end
90 local result = copy(preference)
91 result[#result + 1] = agent
92 return result
93end

95local function acceptable(preferences, a, b)
96 return indexOf(preferences[a], b) > 0 and indexOf(preferences[b], a) > 0
97end
98local function residents(market, partners, receiver)
99 local result = {}
100 for p = 1, market.proposerCount do if partners[p] == receiver then result[#result + 1] = p end end
101 return result
102end
103local function deferred(market, reverse, verbose, preferences)
104 local lists = preferences or market.preferences
105 local positions, counts = filled(market.count, 1), filled(market.count, 0)
106 local holding, queued, queue = {}, filled(market.count, false), {}
107 for a = 1, market.count do holding[a] = {} end
108 local function enqueue(agent)
109 if not queued[agent] and counts[agent] < market.capacities[agent] and positions[agent] <= #lists[agent] then
110 queue[#queue + 1], queued[agent] = agent, true
111 end
112 end
113 local first = reverse and market.proposerCount + 1 or 1
114 local last = reverse and market.count or market.proposerCount
115 for a = first, last do enqueue(a) end
116 local proposals, log, head = 0, {}, 1
117 while head <= #queue do
118 local x = queue[head]
119 head, queued[x] = head + 1, false
120 local y = lists[x][positions[x]]
121 positions[x], proposals = positions[x] + 1, proposals + 1
122 local action = "REJECT"
123 if indexOf(lists[y], x) > 0 then
124 if #holding[y] < market.capacities[y] then
125 holding[y][#holding[y] + 1] = x
126 counts[x] = counts[x] + 1
127 action = "HOLD"
128 else
129 local worst = 1
130 for i = 2, #holding[y] do
131 if rankOf(lists[y], holding[y][i]) > rankOf(lists[y], holding[y][worst]) then worst = i end
132 end
133 local rejected = holding[y][worst]
134 if rankOf(lists[y], x) < rankOf(lists[y], rejected) then
135 holding[y][worst] = x
136 counts[rejected], counts[x] = counts[rejected] - 1, counts[x] + 1
137 action = "HOLD REJECT " .. market.agents[rejected]
138 enqueue(rejected)
139 end
140 end
141 end
142 if verbose then log[#log + 1] = "PROPOSE " .. market.agents[x] .. " " .. market.agents[y] .. " " .. action end
143 enqueue(x)
144 end
145 local partners = filled(market.proposerCount, 0)
146 first = reverse and 1 or market.proposerCount + 1
147 last = reverse and market.proposerCount or market.count
148 for y = first, last do
149 for _, x in ipairs(holding[y]) do partners[reverse and y or x] = reverse and x or y end
150 end
151 return {partners = partners, count = proposals, log = log}
152end
153local function boston(market, preferences)
154 local lists = preferences or market.preferences
155 local partners, remaining = filled(market.proposerCount, 0), copy(market.capacities)
156 local rounds, applications = 0, 0
157 for p = 1, market.proposerCount do rounds = math.max(rounds, #lists[p]) end
158 for round = 1, rounds do
159 local applied = filled(market.proposerCount, 0)
160 for p = 1, market.proposerCount do
161 if partners[p] == 0 and round <= #lists[p] then
162 applied[p], applications = lists[p][round], applications + 1
163 end
164 end
165 for r = market.proposerCount + 1, market.count do
166 for _, p in ipairs(lists[r]) do
167 if remaining[r] == 0 then break end
168 if applied[p] == r then
169 partners[p], remaining[r] = r, remaining[r] - 1
170 end
171 end
172 end
173 end
174 return {partners = partners, count = applications}
175end
176local function checkMatching(market, partners)
177 local unacceptable, blocking, lists = {}, {}, market.preferences
178 if market.kind == "ROOMS" then
179 for a = 1, market.count do
180 for b = a + 1, market.count do
181 local mutual = acceptable(lists, a, b)
182 if partners[a] == b then
183 if not mutual then unacceptable[#unacceptable + 1] = {a, b} end
184 elseif mutual and rankOf(lists[a], b) < rankOf(lists[a], partners[a])
185 and rankOf(lists[b], a) < rankOf(lists[b], partners[b]) then
186 blocking[#blocking + 1] = {a, b}
187 end
188 end
189 end
190 else
191 for p = 1, market.proposerCount do
192 if partners[p] > 0 and not acceptable(lists, p, partners[p]) then
193 unacceptable[#unacceptable + 1] = {p, partners[p]}
194 end
195 for r = market.proposerCount + 1, market.count do
196 if partners[p] ~= r and acceptable(lists, p, r) and rankOf(lists[p], r) < rankOf(lists[p], partners[p]) then
197 local held, worst = residents(market, partners, r), 0
198 for _, a in ipairs(held) do worst = math.max(worst, rankOf(lists[r], a)) end
199 if #held < market.capacities[r] or rankOf(lists[r], p) < worst then
200 blocking[#blocking + 1] = {p, r}
201 end
202 end
203 end
204 end
205 end
206 return unacceptable, blocking
207end
208local function candidateFor(market, partners, origin, via)
209 local cost, regret = 0, 0
210 for p, r in ipairs(partners) do
211 if r > 0 then
212 local a, b = rankOf(market.preferences[p], r), rankOf(market.preferences[r], p)
213 cost, regret = cost + a + b, math.max(regret, a, b)
214 end
215 end
216 return {partners = partners, origin = origin, via = via, cost = cost, regret = regret}
217end
218local function exposedRotations(market, partners)
219 local owners, successors = filled(market.count, 0), filled(market.proposerCount, 0)
220 for p, r in ipairs(partners) do if r > 0 then owners[r] = p end end
221 for p, r in ipairs(partners) do
222 if r > 0 then
223 local preference = market.preferences[p]
224 for i = indexOf(preference, r) + 1, #preference do
225 local w = preference[i]
226 local owner = owners[w]
227 if indexOf(market.preferences[w], p) > 0 and (owner == 0
228 or rankOf(market.preferences[w], p) < rankOf(market.preferences[w], owner)) then
229 successors[p] = owner
230 break
231 end
232 end
233 end
234 end
235 local rotations, covered = {}, filled(market.proposerCount, false)
236 for start = 1, market.proposerCount do
237 if not covered[start] then
238 local path, seen, current = {}, filled(market.proposerCount, false), start
239 while current > 0 and not seen[current] do
240 seen[current], path[#path + 1] = true, current
241 current = successors[current]
242 end
243 if current == start then
244 rotations[#rotations + 1] = path
245 for _, p in ipairs(path) do covered[p] = true end
246 end
247 end
248 end
249 return rotations
250end
251local function enumerateStable(market)
252 for r = market.proposerCount + 1, market.count do if market.capacities[r] ~= 1 then fail("NOT ONE TO ONE") end end
253 local matchings = {candidateFor(market, deferred(market).partners, 0, 0)}
254 local rotations, numbers, index = {}, {}, 1
255 while index <= #matchings do
256 local partners = matchings[index].partners
257 for _, rotation in ipairs(exposedRotations(market, partners)) do
258 local parts = {}
259 for i, p in ipairs(rotation) do parts[i] = market.agents[p] .. "-" .. market.agents[partners[p]] end
260 local notation = table.concat(parts, " ")
261 if not numbers[notation] then
262 rotations[#rotations + 1] = notation
263 numbers[notation] = #rotations
264 end
265 local following = copy(partners)
266 for i, p in ipairs(rotation) do following[p] = partners[rotation[i % #rotation + 1]] end
267 local duplicate = false
268 for _, candidate in ipairs(matchings) do
269 if sameValues(candidate.partners, following) then duplicate = true; break end
270 end
271 if not duplicate then
272 if #matchings == 50 then fail("TOO MANY MATCHINGS") end
273 matchings[#matchings + 1] = candidateFor(market, following, index, numbers[notation])
274 end
275 end
276 index = index + 1
277 end
278 return matchings, rotations
279end
280local function removePair(lists, a, b)
281 local position = indexOf(lists[a], b)
282 if position > 0 then table.remove(lists[a], position) end
283 position = indexOf(lists[b], a)
284 if position > 0 then table.remove(lists[b], position) end
285end
286local function firstEmpty(lists)
287 for a, list in ipairs(lists) do if #list == 0 then return a end end
288 return 0
289end

291 for a, list in ipairs(lists) do
292 local names = {}

294 output[#output + 1] = "TABLE " .. market.agents[a] .. " " .. listText(names)

297local function roommates(market, verbose)
298 local lists = {}
299 for a, list in ipairs(market.preferences) do
300 if #list < market.count - 1 then fail("INCOMPLETE " .. market.agents[a]) end
301 lists[a] = copy(list)
302 end
303 local held, queue = filled(market.count, 0), {}
304 for a = 1, market.count do queue[a] = a end
305 local proposals, head, failure, rotations = 0, 1, 0, 0
306 local trace = {}
307 while head <= #queue do
308 local x = queue[head]
309 head = head + 1
310 local y = lists[x][1]
311 proposals = proposals + 1
312 if held[y] > 0 then queue[#queue + 1] = held[y] end
313 held[y] = x
314 local removed = slice(lists[y], indexOf(lists[y], x) + 1)
315 for _, w in ipairs(removed) do removePair(lists, y, w) end
316 failure = firstEmpty(lists)
317 if failure > 0 then break end
318 end
319 if failure == 0 then
320 if verbose then addTable(trace, market, lists) end
321 while true do
322 local p = 0
323 for a, list in ipairs(lists) do if #list > 1 then p = a; break end end
324 if p == 0 then break end
325 local seen, path = filled(market.count, 0), {}
326 while seen[p] == 0 do
327 path[#path + 1] = p
328 seen[p] = #path
329 local q = lists[p][2]
330 p = lists[q][#lists[q]]
331 end
332 local xs, ys, parts = slice(path, seen[p]), {}, {}
333 for i, x in ipairs(xs) do
334 ys[i] = lists[x][1]
335 parts[i] = market.agents[x] .. "-" .. market.agents[ys[i]]
336 end
337 rotations = rotations + 1
338 trace[#trace + 1] = "ROTATION " .. rotations .. " " .. table.concat(parts, " ")
339 local removals = {}
340 for i, x in ipairs(xs) do
341 local u = ys[i % #ys + 1]
342 for j = indexOf(lists[u], x) + 1, #lists[u] do removals[#removals + 1] = {u, lists[u][j]} end
343 end
344 for _, pair in ipairs(removals) do removePair(lists, pair[1], pair[2]) end
345 if verbose then addTable(trace, market, lists) end
346 failure = firstEmpty(lists)
347 if failure > 0 then break end
348 end
349 end
350 local output = {"ROOMMATES " .. market.name .. " proposals=" .. proposals}
351 for _, line in ipairs(trace) do output[#output + 1] = line end
352 if failure > 0 then output[#output + 1] = "NO STABLE MATCHING " .. market.agents[failure]
353 else
354 local pairs = {}
355 for a, list in ipairs(lists) do
356 if a < list[1] then pairs[#pairs + 1] = market.agents[a] .. "-" .. market.agents[list[1]] end
357 end
358 output[#output + 1] = "MATCHING " .. table.concat(pairs, " ")
359 end
360 return output
361end
362local function tradingCycles(market)
363 local remaining, allocation = filled(market.count, true), filled(market.count, 0)
364 local effective, cycles, left = {}, {}, market.count
365 for a = 1, market.count do effective[a] = market:effective(a) end
366 while left > 0 do
367 local pointers = filled(market.count, 0)
368 for a = 1, market.count do
369 if remaining[a] then
370 for _, h in ipairs(effective[a]) do if remaining[h] then pointers[a] = h; break end end
371 end
372 end
373 local seen, path, current = filled(market.count, 0), {}, indexOf(remaining, true)
374 while seen[current] == 0 do
375 path[#path + 1] = current
376 seen[current] = #path
377 current = pointers[current]
378 end
379 local cycle, parts = slice(path, seen[current]), {}
380 for i, a in ipairs(cycle) do
381 allocation[a], remaining[a], left = pointers[a], false, left - 1
382 parts[i] = market.agents[a] .. ":" .. market.houses[allocation[a]]
383 end
384 cycles[#cycles + 1] = "CYCLE " .. (#cycles + 1) .. " " .. table.concat(parts, " ")
385 end
386 local output = {"TTC " .. market.name .. " cycles=" .. #cycles}
387 for _, line in ipairs(cycles) do output[#output + 1] = line end
388 output[#output + 1] = "ALLOCATION " .. market:allocation(allocation)
389 return output
390end

392local function show(market)
393 local output = {}
394 if market.kind == "MARKET" then
395 output[1] = "SHOW " .. market.name .. " MARKET proposers=" .. market.proposerCount
396 .. " receivers=" .. (market.count - market.proposerCount)
397 else
398 output[1] = "SHOW " .. market.name .. " " .. market.kind .. " agents=" .. market.count
399 end
400 for a, name in ipairs(market.agents) do
401 local prefix = "AGENT " .. name
402 if market.kind == "MARKET" then
403 prefix = a <= market.proposerCount and "PROPOSER " .. name
404 or "RECEIVER " .. name .. " cap=" .. market.capacities[a]
405 elseif market.kind == "HOUSES" then prefix = prefix .. " house=" .. market.houses[a] end
406 local names = {}
407 for i, choice in ipairs(market.preferences[a]) do
408 names[i] = market.kind == "HOUSES" and market.houses[choice] or market.agents[choice]
409 end
410 output[#output + 1] = prefix .. " " .. listText(names)
411 end
412 if market.kind == "MARKET" then output[#output + 1] = "CURRENT " .. market.source end
413 return output

415local function check(market, words)
416 if market.kind == "HOUSES" then fail("WRONG TYPE " .. market.name) end
417 local partners = filled(market.kind == "MARKET" and market.proposerCount or market.count, 0)
418 local used, counts = filled(market.count, false), filled(market.count, 0)
419 for i = 3, #words do
420 local first, second = splitPair(words[i], "-")
421 if market.kind == "MARKET" then
422 local p = market:proposer(first)
423 local r = market:receiver(second)
424 if used[p] then fail("DUPLICATE AGENT " .. first) end
425 if counts[r] + 1 > market.capacities[r] then fail("OVER CAPACITY " .. second) end
426 used[p], counts[r], partners[p] = true, counts[r] + 1, r
427 else
428 local a = market:agent(first)
429 local b = market:agent(second)
430 if used[a] then fail("DUPLICATE AGENT " .. first) end
431 if a == b or used[b] then fail("DUPLICATE AGENT " .. second) end
432 used[a], used[b], partners[a], partners[b] = true, true, b, a
433 end
434 end
435 local unacceptable, blocking = checkMatching(market, partners)
436 if #unacceptable == 0 and #blocking == 0 then return {"CHECK " .. market.name .. " STABLE"} end
437 local output = {"CHECK " .. market.name .. " UNSTABLE unacceptable=" .. #unacceptable .. " blocking=" .. #blocking}
438 for _, pair in ipairs(unacceptable) do
439 output[#output + 1] = "UNACCEPTABLE " .. market.agents[pair[1]] .. " " .. market.agents[pair[2]]
440 end
441 for _, pair in ipairs(blocking) do
442 output[#output + 1] = "BLOCKING " .. market.agents[pair[1]] .. " " .. market.agents[pair[2]]
443 end
444 return output
445end
446local function hospitals(market)
447 local a, h = deferred(market).partners, deferred(market, true).partners
448 local unmatched, rural = {}, true
449 for p = 1, market.proposerCount do
450 if a[p] == 0 then unmatched[#unmatched + 1] = market.agents[p] end
451 if (a[p] > 0) ~= (h[p] > 0) then rural = false end
452 end
453 local output = {"HR " .. market.name .. " residents=" .. market.proposerCount .. " hospitals=" .. (market.count - market.proposerCount),
454 "RESIDENT-OPTIMAL " .. market:pairs(a), "HOSPITAL-OPTIMAL " .. market:pairs(h), "UNMATCHED " .. listText(unmatched)}
455 for r = market.proposerCount + 1, market.count do
456 local first, second = residents(market, a, r), residents(market, h, r)
457 local same = sameValues(first, second)
458 output[#output + 1] = "HOSPITAL " .. market.agents[r] .. " cap=" .. market.capacities[r]
459 .. " resident=" .. #first .. " hospital=" .. #second .. " same=" .. (same and "YES" or "NO")
460 rural = rural and #first == #second and (#first == market.capacities[r] or same)
461 end
462 output[#output + 1] = rural and "RURAL HOLDS" or "RURAL FAILS"
463 return output
464end
465local function serial(market, words)
466 local order, seen = {}, {}
467 for i = 3, #words do
468 local a = market:agent(words[i])
469 if seen[a] then fail("DUPLICATE AGENT " .. words[i]) end
470 seen[a], order[#order + 1] = true, a
471 end
472 if #order ~= market.count then fail("BAD ORDER") end
473 local allocation, remaining, assigned = filled(market.count, 0), filled(market.count, true), 0
474 for _, a in ipairs(order) do
475 for _, house in ipairs(market.preferences[a]) do
476 if remaining[house] then
477 allocation[a], remaining[house], assigned = house, false, assigned + 1
478 break
479 end
480 end
481 end
482 local worse = {}
483 for a, house in ipairs(allocation) do
484 if house == 0 or indexOf(market:effective(a), house) == 0 then worse[#worse + 1] = market.agents[a] end
485 end
486 return {"SERIAL " .. market.name .. " assigned=" .. assigned,
487 "ALLOCATION " .. market:allocation(allocation), "WORSE " .. listText(worse)}
488end
489local function manipulate(market, agent)
490 local truth = market.preferences[agent]
491 local b0, g0 = boston(market).partners[agent], deferred(market).partners[agent]
492 local output = {"MANIPULATE " .. market.name .. " " .. market.agents[agent] .. " boston=" .. market:partner(b0)
493 .. " gs=" .. market:partner(g0) .. " swaps=" .. math.max(0, #truth - 1)}
494 for i = 1, #truth - 1 do
495 local preferences = copy(market.preferences)
496 preferences[agent] = copy(truth)
497 preferences[agent][i], preferences[agent][i + 1] = truth[i + 1], truth[i]
498 local b = boston(market, preferences).partners[agent]
499 local g = deferred(market, false, false, preferences).partners[agent]
500 local swap = i .. " " .. market.agents[truth[i]] .. " " .. market.agents[truth[i + 1]]
501 if rankOf(truth, b) < rankOf(truth, b0) then output[#output + 1] = "IMPROVE BOSTON " .. swap .. " result=" .. market:partner(b) end
502 if rankOf(truth, g) < rankOf(truth, g0) then output[#output + 1] = "IMPROVE GS " .. swap .. " result=" .. market:partner(g) end
503 end
504 if #output == 1 then output[#output + 1] = "NO IMPROVEMENT" end
505 return output
506end
507local function rankSummary(ranks)
508 local sum, first = 0, 0
509 for _, rank in ipairs(ranks) do
510 sum = sum + rank
511 if rank == 1 then first = first + 1 end
512 end
513 local average, truncated = "-", "-"
514 if #ranks > 0 then
515 local a, b = sum, #ranks
516 while b ~= 0 do a, b = b, a % b end
517 average = string.format("%d/%d", sum // a, #ranks // a)
518 local cents = sum * 100 // #ranks
519 truncated = string.format("%d.%02d", cents // 100, cents % 100)
520 end
521 return "sum=" .. sum .. " average=" .. average .. " truncated=" .. truncated .. " first=" .. first
522end
523local function stats(market)
524 if not market.current then fail("NO MATCHING") end
525 local proposerRanks, receiverRanks, occupied = {}, {}, filled(market.count, false)
526 local hospitalCount = 0
527 for p, r in ipairs(market.current) do
528 if r > 0 then
529 proposerRanks[#proposerRanks + 1] = rankOf(market.preferences[p], r)
530 receiverRanks[#receiverRanks + 1] = rankOf(market.preferences[r], p)
531 if not occupied[r] then occupied[r], hospitalCount = true, hospitalCount + 1 end
532 end
533 end
534 return {"STATS " .. market.name .. " source=" .. market.source .. " pairs=" .. #proposerRanks,
535 "PROPOSERS " .. rankSummary(proposerRanks) .. " unmatched=" .. (market.proposerCount - #proposerRanks),
536 "RECEIVERS " .. rankSummary(receiverRanks) .. " unmatched=" .. (market.count - market.proposerCount - hospitalCount)}
537end

554local function execute(line)
555 local words = {}
556 for word in line:gmatch("%S+") do words[#words + 1] = word end
557 local command, name = words[1], words[2]
558 if command == "MARKET" then
559 newName(name)
560 local bar = indexOf(words, "|")
561 local agents = slice(words, 3, bar - 1)
562 local proposers = #agents
563 for i = bar + 1, #words do agents[#agents + 1] = words[i] end
564 distinctAgents(agents)
565 markets[name] = Market.new(name, "MARKET", agents, proposers)
566 return {"MARKET " .. name .. " proposers=" .. proposers .. " receivers=" .. (#agents - proposers)}
567 elseif command == "ROOMS" then
568 newName(name)
