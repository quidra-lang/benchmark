67fn list_text(values: impl IntoIterator<Item = String>) -> String {
68 let text = values.into_iter().collect::<Vec<_>>().join(" ");
69 if text.is_empty() { "-".into() } else { text }

72impl Market {
73 fn require(&self, kind: Kind) -> Result<(), String> {
74 if self.kind == kind { Ok(()) } else { Err(format!("WRONG TYPE {}", self.name)) }
75 }

77 fn agent(&self, name: &str, side: Option<Side>) -> Result<usize, String> {
78 let index = self.agents.iter().position(|a| a.name == name)
79 .ok_or_else(|| format!("UNKNOWN AGENT {name}"))?;
80 match side {
81 Some(Side::Proposer) if index >= self.proposers => Err(format!("NOT PROPOSER {name}")),
82 Some(Side::Receiver) if index < self.proposers => Err(format!("NOT RECEIVER {name}")),
83 _ => Ok(index),
84 }
85 }

87 fn rank(&self, agent: usize, partner: Option<usize>) -> usize {
88 let preference = &self.agents[agent].preference;
89 preference.iter().position(|&p| Some(p) == partner).map_or(preference.len() + 1, |i| i + 1)
90 }

92 fn accepts(&self, agent: usize, partner: usize) -> bool {
93 self.agents[agent].preference.contains(&partner)
94 }

96 fn partner_name(&self, partner: Option<usize>) -> &str {
97 partner.map_or("*", |a| self.agents[a].name.as_str())
98 }

100 fn pairs(&self, matching: &Matching) -> String {
101 (0..self.proposers).map(|p| format!("{}-{}", self.agents[p].name, self.partner_name(matching[p])))
102 .collect::<Vec<_>>().join(" ")

105 fn allocation(&self, allocation: &Matching) -> String {
106 self.agents.iter().enumerate().map(|(a, agent)| {
107 let house = allocation[a].map_or("*", |h| self.houses[h].as_str());
108 format!("{}-{house}", agent.name)

112 fn effective(&self, agent: usize) -> Vec<usize> {
113 let preference = &self.agents[agent].preference;
114 if let Some(i) = preference.iter().position(|&house| house == agent) {
115 preference[..=i].to_vec()
116 } else {
117 let mut result = preference.clone();
118 result.push(agent);
119 result
120 }
121 }

123 fn able(&self, agent: usize, next: &[usize], counts: &[usize]) -> bool {
124 counts[agent] < self.agents[agent].capacity && next[agent] < self.agents[agent].preference.len()
125 }

127 fn deferred(&self, reverse: bool, verbose: bool) -> Run {
128 let n = self.agents.len();
129 let mut next = vec![0; n];
130 let mut counts = vec![0; n];
131 let mut holding = vec![Vec::new(); n];
132 let proposing = if reverse { self.proposers..n } else { 0..self.proposers };
133 let mut queue: VecDeque<_> = proposing.filter(|&a| self.able(a, &next, &counts)).collect();
134 let mut proposals = 0;
135 let mut history = Vec::new();
136 while let Some(x) = queue.pop_front() {
137 let y = self.agents[x].preference[next[x]];
138 next[x] += 1;
139 proposals += 1;
140 let mut action = "REJECT".to_string();
141 if self.accepts(y, x) {
142 if holding[y].len() < self.agents[y].capacity {
143 holding[y].push(x);
144 counts[x] += 1;
145 action = "HOLD".into();
146 } else {
147 let worst = (0..holding[y].len()).max_by_key(|&i| self.rank(y, Some(holding[y][i]))).unwrap();
148 let rejected = holding[y][worst];
149 if self.rank(y, Some(x)) < self.rank(y, Some(rejected)) {
150 holding[y][worst] = x;
151 counts[rejected] -= 1;
152 counts[x] += 1;
153 action = format!("HOLD REJECT {}", self.agents[rejected].name);
154 if self.able(rejected, &next, &counts) && !queue.contains(&rejected) { queue.push_back(rejected); }
155 }
156 }
157 }
158 if verbose { history.push(format!("PROPOSE {} {} {action}", self.agents[x].name, self.agents[y].name)); }
159 if self.able(x, &next, &counts) { queue.push_back(x); }
160 }
161 let mut matching = vec![None; self.proposers];
162 let receiving = if reverse { 0..self.proposers } else { self.proposers..n };
163 for y in receiving {
164 for &x in &holding[y] {
165 if reverse { matching[y] = Some(x); } else { matching[x] = Some(y); }
166 }
167 }
168 Run { matching, proposals, history }
169 }

171 fn boston(&self) -> Run {
172 let mut matching = vec![None; self.proposers];
173 let mut remaining: Vec<_> = self.agents.iter().map(|a| a.capacity).collect();
174 let rounds = self.agents[..self.proposers].iter().map(|a| a.preference.len()).max().unwrap();
175 let mut applications = 0;
176 for round in 0..rounds {
177 let mut applicants = vec![None; self.proposers];
178 for p in 0..self.proposers {
179 if matching[p].is_none() && round < self.agents[p].preference.len() {
180 applicants[p] = Some(self.agents[p].preference[round]);
181 applications += 1;
182 }
183 }
184 for r in self.proposers..self.agents.len() {
185 for &p in &self.agents[r].preference {
186 if remaining[r] > 0 && applicants[p] == Some(r) {
187 matching[p] = Some(r);
188 remaining[r] -= 1;
189 }
190 }
191 }
192 }
193 Run { matching, proposals: applications, history: Vec::new() }
194 }

196 fn rotations(&self, matching: &Matching) -> Vec<Rotation> {
197 let mut owners = vec![None; self.agents.len()];
198 for (p, &partner) in matching.iter().enumerate() {
199 if let Some(r) = partner { owners[r] = Some(p); }
200 }
201 let mut successors = vec![None; self.proposers];
202 for p in 0..self.proposers {
203 if matching[p].is_none() { continue; }
204 for &r in self.agents[p].preference.iter().skip(self.rank(p, matching[p])) {
205 if self.accepts(r, p) && (owners[r].is_none() || self.rank(r, Some(p)) < self.rank(r, owners[r])) {
206 successors[p] = owners[r];
207 break;
208 }
209 }
210 }
211 let mut rotations = Vec::new();
212 let mut covered = vec![false; self.proposers];
213 for start in 0..self.proposers {
214 if covered[start] { continue; }
215 let mut seen = vec![false; self.proposers];
216 let mut path = Vec::new();
217 let mut current = Some(start);
218 while let Some(p) = current {
219 if seen[p] { break; }
220 seen[p] = true;
221 path.push(p);
222 current = successors[p];
223 }
224 if current == Some(start) {
225 for &p in &path { covered[p] = true; }
226 rotations.push(Rotation { pairs: path.into_iter().map(|p| (p, matching[p].unwrap())).collect() });
227 }
228 }
229 rotations
230 }

232 fn candidate(&self, matching: Matching, origin: Option<usize>, via: Option<usize>) -> Candidate {
233 let mut cost = 0;
234 let mut regret = 0;
235 for (p, &partner) in matching.iter().enumerate() {
236 if let Some(r) = partner {
237 let a = self.rank(p, Some(r));
238 let b = self.rank(r, Some(p));
239 cost += a + b;
240 regret = regret.max(a).max(b);
241 }
242 }
243 Candidate { matching, origin, via, cost, regret }
244 }

246 fn enumerate(&self) -> Result<Lattice, String> {
247 if self.agents[self.proposers..].iter().any(|a| a.capacity != 1) { return Err("NOT ONE TO ONE".into()); }
248 let mut matchings = vec![self.candidate(self.deferred(false, false).matching, None, None)];
249 let mut rotations = Vec::new();
250 let mut index = 0;
251 while index < matchings.len() {
252 let matching = matchings[index].matching.clone();
253 for rotation in self.rotations(&matching) {
254 let number = if let Some(i) = rotations.iter().position(|old| old == &rotation) { i } else {
255 rotations.push(rotation.clone());
256 rotations.len() - 1
257 };
258 let mut next = matching.clone();
259 for (i, &(p, _)) in rotation.pairs.iter().enumerate() {
260 next[p] = Some(rotation.pairs[(i + 1) % rotation.pairs.len()].1);
261 }
262 if matchings.iter().any(|old| old.matching == next) { continue; }
263 if matchings.len() == 50 { return Err("TOO MANY MATCHINGS".into()); }
264 matchings.push(self.candidate(next, Some(index + 1), Some(number + 1)));
265 }
266 index += 1;
267 }
268 Ok(Lattice { matchings, rotations })
269 }

271 fn check(&self, words: &[&str]) -> Response {
272 if self.kind == Kind::Housing { return Err(format!("WRONG TYPE {}", self.name)); }
273 let mut partners = vec![None; self.agents.len()];
274 let mut counts = vec![0; self.agents.len()];
275 for word in &words[2..] {
276 let (left, right) = word.split_once('-').unwrap();
277 let two = self.kind == Kind::TwoSided;
278 let x = self.agent(left, if two { Some(Side::Proposer) } else { None })?;
279 let y = self.agent(right, if two { Some(Side::Receiver) } else { None })?;
280 if partners[x].is_some() { return Err(format!("DUPLICATE AGENT {left}")); }
281 if two {
282 counts[y] += 1;
283 if counts[y] > self.agents[y].capacity { return Err(format!("OVER CAPACITY {right}")); }
284 } else {
285 if x == y || partners[y].is_some() { return Err(format!("DUPLICATE AGENT {right}")); }
286 partners[y] = Some(x);
287 }
288 partners[x] = Some(y);
289 }
290 let mut unacceptable = Vec::new();
291 let mut blocking = Vec::new();
292 if self.kind == Kind::TwoSided {
293 for p in 0..self.proposers {
294 if let Some(r) = partners[p] {
295 if !self.accepts(p, r) || !self.accepts(r, p) { unacceptable.push((p, r)); }
296 }
297 for r in self.proposers..self.agents.len() {
298 if partners[p] == Some(r) || !self.accepts(p, r) || !self.accepts(r, p)
299 || self.rank(p, Some(r)) >= self.rank(p, partners[p]) { continue; }
300 let worst = (0..self.proposers).filter(|&a| partners[a] == Some(r)).map(|a| self.rank(r, Some(a))).max();
301 if counts[r] < self.agents[r].capacity || self.rank(r, Some(p)) < worst.unwrap() { blocking.push((p, r)); }
302 }
303 }
304 } else {
305 for a in 0..self.agents.len() {
306 for b in a + 1..self.agents.len() {
307 let mutual = self.accepts(a, b) && self.accepts(b, a);
308 if partners[a] == Some(b) {
309 if !mutual { unacceptable.push((a, b)); }
310 } else if mutual && self.rank(a, Some(b)) < self.rank(a, partners[a])
311 && self.rank(b, Some(a)) < self.rank(b, partners[b]) { blocking.push((a, b)); }
312 }
313 }
314 }
315 if unacceptable.is_empty() && blocking.is_empty() { return Ok(vec![format!("CHECK {} STABLE", self.name)]); }
316 let mut output = vec![format!("CHECK {} UNSTABLE unacceptable={} blocking={}", self.name, unacceptable.len(), blocking.len())];
317 for (label, pairs) in [("UNACCEPTABLE", unacceptable), ("BLOCKING", blocking)] {
318 for (a, b) in pairs { output.push(format!("{label} {} {}", self.agents[a].name, self.agents[b].name)); }
319 }
320 Ok(output)
321 }

329 fn roommates(&self, verbose: bool) -> Response {
330 self.require(Kind::Roommate)?;
331 let n = self.agents.len();
332 for agent in &self.agents {
333 if agent.preference.len() < n - 1 { return Err(format!("INCOMPLETE {}", agent.name)); }
334 }
335 let mut lists: Vec<_> = self.agents.iter().map(|a| a.preference.clone()).collect();
336 let mut held = vec![None; n];
337 let mut queue: VecDeque<_> = (0..n).collect();
338 let mut proposals = 0;
339 let mut failure = None;
340 while let Some(x) = queue.pop_front() {
341 let y = lists[x][0];
342 if let Some(z) = held[y] { queue.push_back(z); }
343 held[y] = Some(x);
344 proposals += 1;
345 let cut = lists[y].iter().position(|&a| a == x).unwrap();
346 let removed = lists[y][cut + 1..].to_vec();
347 for w in removed { remove_pair(&mut lists, y, w); }
348 failure = lists.iter().position(Vec::is_empty);
349 if failure.is_some() { break; }
350 }
351 let mut history = Vec::new();
352 let mut rotations = 0;
353 if failure.is_none() && verbose { self.table(&lists, &mut history); }
354 while failure.is_none() {
355 let Some(mut p) = lists.iter().position(|list| list.len() >= 2) else { break; };
356 let mut path = Vec::new();
357 let mut positions = vec![None; n];
358 while positions[p].is_none() {
359 positions[p] = Some(path.len());
360 path.push(p);
361 let q = lists[p][1];
362 p = *lists[q].last().unwrap();
363 }
364 let cycle = &path[positions[p].unwrap()..];
365 let front: Vec<_> = cycle.iter().map(|&x| lists[x][0]).collect();
366 rotations += 1;
367 let notation = cycle.iter().zip(&front).map(|(&x, &y)| format!("{}-{}", self.agents[x].name, self.agents[y].name))
368 .collect::<Vec<_>>().join(" ");
369 history.push(format!("ROTATION {rotations} {notation}"));
370 let mut removals = Vec::new();
371 for (i, &x) in cycle.iter().enumerate() {
372 let u = front[(i + 1) % front.len()];
373 let cut = lists[u].iter().position(|&a| a == x).unwrap();
374 for &w in &lists[u][cut + 1..] { removals.push((u, w)); }
375 }
376 for (a, b) in removals { remove_pair(&mut lists, a, b); }
377 if verbose { self.table(&lists, &mut history); }
378 failure = lists.iter().position(Vec::is_empty);
379 }
380 let mut output = vec![format!("ROOMMATES {} proposals={proposals}", self.name)];
381 output.extend(history);
382 if let Some(a) = failure { output.push(format!("NO STABLE MATCHING {}", self.agents[a].name)); }
383 else {
384 let pairs = (0..n).filter(|&a| a < lists[a][0]).map(|a| format!("{}-{}", self.agents[a].name, self.agents[lists[a][0]].name))
385 .collect::<Vec<_>>().join(" ");
386 output.push(format!("MATCHING {pairs}"));
387 }
388 Ok(output)
389 }

391 fn ttc(&self) -> Vec<String> {
392 let n = self.agents.len();
393 let effective: Vec<_> = (0..n).map(|a| self.effective(a)).collect();
394 let mut remaining = vec![true; n];
395 let mut allocation = vec![None; n];
396 let mut history = Vec::new();
397 while let Some(start) = remaining.iter().position(|&value| value) {
398 let mut points = vec![0; n];
399 for a in 0..n {
400 if remaining[a] { points[a] = *effective[a].iter().find(|&&house| remaining[house]).unwrap(); }
401 }
402 let mut seen = vec![false; n];
403 let mut p = start;
404 while !seen[p] { seen[p] = true; p = points[p]; }
405 let first = p;
406 let mut cycle = Vec::new();
407 loop {
408 allocation[p] = Some(points[p]);
409 remaining[p] = false;
410 cycle.push(format!("{}:{}", self.agents[p].name, self.houses[points[p]]));
411 p = points[p];
412 if p == first { break; }
413 }
414 history.push(format!("CYCLE {} {}", history.len() + 1, cycle.join(" ")));
415 }
416 let mut output = vec![format!("TTC {} cycles={}", self.name, history.len())];
417 output.extend(history);
418 output.push(format!("ALLOCATION {}", self.allocation(&allocation)));
419 output
420 }

422 fn serial(&self, words: &[&str]) -> Response {
423 let n = self.agents.len();
424 let mut order = Vec::new();
425 for name in &words[2..] {
426 let a = self.agent(name, None)?;
427 if order.contains(&a) { return Err(format!("DUPLICATE AGENT {name}")); }
428 order.push(a);
429 }
430 if order.len() != n { return Err("BAD ORDER".into()); }
431 let mut taken = vec![false; n];
432 let mut allocation = vec![None; n];
433 for a in order {
434 if let Some(&house) = self.agents[a].preference.iter().find(|&&house| !taken[house]) {
435 allocation[a] = Some(house);
436 taken[house] = true;
437 }
438 }
439 let assigned = allocation.iter().filter(|house| house.is_some()).count();
440 let worse = list_text((0..n).filter(|&a| allocation[a].is_none_or(|house| !self.effective(a).contains(&house)))
441 .map(|a| self.agents[a].name.clone()));
442 Ok(vec![format!("SERIAL {} assigned={assigned}", self.name), format!("ALLOCATION {}", self.allocation(&allocation)), format!("WORSE {worse}")])
443 }

445 fn hr(&self) -> Vec<String> {
446 let resident = self.deferred(false, false).matching;
447 let hospital = self.deferred(true, false).matching;
448 let unmatched = list_text((0..self.proposers).filter(|&p| resident[p].is_none()).map(|p| self.agents[p].name.clone()));
449 let mut output = vec![format!("HR {} residents={} hospitals={}", self.name, self.proposers, self.agents.len() - self.proposers),
450 format!("RESIDENT-OPTIMAL {}", self.pairs(&resident)), format!("HOSPITAL-OPTIMAL {}", self.pairs(&hospital)), format!("UNMATCHED {unmatched}")];
451 let mut rural = (0..self.proposers).all(|p| resident[p].is_some() == hospital[p].is_some());
452 for r in self.proposers..self.agents.len() {
453 let a: Vec<_> = (0..self.proposers).filter(|&p| resident[p] == Some(r)).collect();
454 let h: Vec<_> = (0..self.proposers).filter(|&p| hospital[p] == Some(r)).collect();
455 let same = a == h;
456 let capacity = self.agents[r].capacity;
457 output.push(format!("HOSPITAL {} cap={capacity} resident={} hospital={} same={}", self.agents[r].name, a.len(), h.len(), if same { "YES" } else { "NO" }));
458 rural &= a.len() == h.len() && (a.len() == capacity || same);
459 }
460 output.push(format!("RURAL {}", if rural { "HOLDS" } else { "FAILS" }));
461 output
462 }

464 fn manipulate(&self, a: usize) -> Vec<String> {
465 let truth = &self.agents[a].preference;
466 let baseline_b = self.boston().matching[a];
467 let baseline_g = self.deferred(false, false).matching[a];
468 let mut output = vec![format!("MANIPULATE {} {} boston={} gs={} swaps={}", self.name, self.agents[a].name,
469 self.partner_name(baseline_b), self.partner_name(baseline_g), truth.len().saturating_sub(1))];
470 for i in 0..truth.len().saturating_sub(1) {
471 let mut trial = self.clone();
472 trial.agents[a].preference.swap(i, i + 1);
473 let b = trial.boston().matching[a];
474 let g = trial.deferred(false, false).matching[a];
475 for (label, result, baseline) in [("BOSTON", b, baseline_b), ("GS", g, baseline_g)] {
476 if self.rank(a, result) < self.rank(a, baseline) {
477 output.push(format!("IMPROVE {label} {} {} {} result={}", i + 1,
478 self.agents[truth[i]].name, self.agents[truth[i + 1]].name, self.partner_name(result)));
479 }
480 }
481 }
482 if output.len() == 1 { output.push("NO IMPROVEMENT".into()); }
483 output
484 }

486 fn stats(&self) -> Response {
487 let (source, matching) = self.current.as_ref().ok_or("NO MATCHING")?;
488 let mut sums = [0; 2];
489 let mut first = [0; 2];
490 let mut pairs = 0;
491 let mut occupied = vec![false; self.agents.len()];
492 for (p, &partner) in matching.iter().enumerate() {
493 if let Some(r) = partner {
494 pairs += 1;
495 occupied[r] = true;
496 let ranks = [self.rank(p, Some(r)), self.rank(r, Some(p))];
497 for i in 0..2 { sums[i] += ranks[i]; first[i] += usize::from(ranks[i] == 1); }
498 }
499 }
500 let unmatched = [self.proposers - pairs, occupied[self.proposers..].iter().filter(|&&value| !value).count()];
501 let mut output = vec![format!("STATS {} source={} pairs={pairs}", self.name, source.name())];
502 for (i, label) in ["PROPOSERS", "RECEIVERS"].iter().enumerate() {
503 let (average, truncated) = if pairs == 0 { ("-".into(), "-".into()) } else {
504 let divisor = gcd(sums[i], pairs);
505 let cents = 100 * sums[i] / pairs;
506 (format!("{}/{}", sums[i] / divisor, pairs / divisor), format!("{}.{:02}", cents / 100, cents % 100))
507 };
508 output.push(format!("{label} sum={} average={average} truncated={truncated} first={} unmatched={}", sums[i], first[i], unmatched[i]));
509 }
510 Ok(output)
511 }

513 fn show(&self) -> Vec<String> {
514 let mut output = vec![match self.kind {
515 Kind::TwoSided => format!("SHOW {} MARKET proposers={} receivers={}", self.name, self.proposers, self.agents.len() - self.proposers),
516 Kind::Roommate => format!("SHOW {} ROOMS agents={}", self.name, self.agents.len()),
517 Kind::Housing => format!("SHOW {} HOUSES agents={}", self.name, self.agents.len()),
518 }];
519 for (a, agent) in self.agents.iter().enumerate() {
520 let preference = list_text(agent.preference.iter().map(|&b| if self.kind == Kind::Housing { self.houses[b].clone() } else { self.agents[b].name.clone() }));
521 let prefix = match self.kind {
522 Kind::TwoSided if a < self.proposers => format!("PROPOSER {}", agent.name),
523 Kind::TwoSided => format!("RECEIVER {} cap={}", agent.name, agent.capacity),
524 Kind::Roommate => format!("AGENT {}", agent.name),
525 Kind::Housing => format!("AGENT {} house={}", agent.name, self.houses[a]),
526 };
527 output.push(format!("{prefix} {preference}"));
528 }
529 if self.kind == Kind::TwoSided { output.push(format!("CURRENT {}", self.current.as_ref().map_or("-", |(source, _)| source.name()))); }
530 output
531 }

534fn remove_pair(lists: &mut [Vec<usize>], a: usize, b: usize) {
535 lists[a].retain(|&x| x != b);
536 lists[b].retain(|&x| x != a);

539fn gcd(mut a: usize, mut b: usize) -> usize {

544fn new_market(words: &[&str]) -> Result<(Market, String), String> {
545 let op = words[0];
546 let name = words[1];
547 let kind = match op { "ROOMS" => Kind::Roommate, "HOUSES" => Kind::Housing, _ => Kind::TwoSided };
548 let mut market = Market { name: name.into(), kind, agents: Vec::new(), houses: Vec::new(), proposers: 0, current: None };
549 let output;
550 if op == "RANDOM" {
551 let n: usize = words[2].parse().unwrap();
552 let seed: i64 = words[3].parse().unwrap();
553 let mut state = seed;
554 market.proposers = n;
555 for a in 0..2 * n {
556 let mut agent = Agent::new(&format!("{}{}", if a < n { 'P' } else { 'R' }, a % n + 1));
557 agent.preference = if a < n { (n..2 * n).collect() } else { (0..n).collect() };
558 for i in (2..=n).rev() {
559 state = 48271 * state % 2147483647;
560 agent.preference.swap(i - 1, (state % i as i64) as usize);
561 }
562 market.agents.push(agent);
563 }
564 output = format!("RANDOM {name} n={n} seed={seed}");
565 } else {
566 for word in &words[2..] {
567 if *word == "|" { market.proposers = market.agents.len(); continue; }
568 let (agent, house) = if kind == Kind::Housing {
569 let (a, h) = word.split_once(':').unwrap();
570 (a, Some(h))
571 } else { (*word, None) };
572 if market.agents.iter().any(|a| a.name == agent) { return Err(format!("DUPLICATE AGENT {agent}")); }
573 if let Some(h) = house {
574 if market.houses.iter().any(|old| old == h) { return Err(format!("DUPLICATE HOUSE {h}")); }
575 market.houses.push(h.into());
576 }
577 market.agents.push(Agent::new(agent));
578 }
579 output = if kind == Kind::TwoSided {
580 format!("MARKET {name} proposers={} receivers={}", market.proposers, market.agents.len() - market.proposers)
581 } else { format!("{op} {name} agents={}", market.agents.len()) };
582 }
583 Ok((market, output))
584}

586fn execute(markets: &mut HashMap<String, Market>, words: &[&str]) -> Response {
587 let op = words[0];
588 let name = words[1];
589 if matches!(op, "MARKET" | "ROOMS" | "HOUSES" | "RANDOM") {
590 if markets.contains_key(name) { return Err(format!("DUPLICATE MARKET {name}")); }
591 let (market, output) = new_market(words)?;
592 markets.insert(name.into(), market);
593 return Ok(vec![output]);
594 }
595 let market = markets.get_mut(name).ok_or_else(|| format!("UNKNOWN MARKET {name}"))?;
596 match op {
597 "SHOW" => Ok(market.show()),
598 "PREF" => {
599 let a = market.agent(words[2], None)?;
600 let mut preference = Vec::new();
601 for word in &words[3..] {
602 let choice = if market.kind == Kind::Housing { market.houses.iter().position(|h| h.as_str() == *word) }
603 else { market.agents.iter().position(|agent| agent.name == *word) };
604 let b = choice.ok_or_else(|| format!("BAD CHOICE {word}"))?;
605 if (market.kind == Kind::TwoSided && (a < market.proposers) == (b < market.proposers))
606 || (market.kind == Kind::Roommate && a == b) { return Err(format!("BAD CHOICE {word}")); }
607 if preference.contains(&b) { return Err(format!("DUPLICATE CHOICE {word}")); }
608 preference.push(b);
609 }
610 let length = preference.len();
611 market.agents[a].preference = preference;
612 if market.kind == Kind::TwoSided { market.current = None; }
613 Ok(vec![format!("PREF {name} {} length={length}", words[2])])
614 }
615 "CHECK" => market.check(words),
616 "ROOMMATES" => market.roommates(words.len() == 3),
617 "TTC" | "SERIAL" => {
618 market.require(Kind::Housing)?;
619 if op == "TTC" { Ok(market.ttc()) } else { market.serial(words) }
620 }
621 _ => {
622 market.require(Kind::TwoSided)?;
623 match op {
624 "CAPACITY" => {
625 let r = market.agent(words[2], Some(Side::Receiver))?;
626 let capacity = words[3].parse().unwrap();
627 market.agents[r].capacity = capacity;
628 market.current = None;
629 Ok(vec![format!("CAPACITY {name} {} {capacity}", words[2])])
630 }
631 "GS" | "GSR" | "BOSTON" => {
632 let source = match op { "GS" => Source::Gs, "GSR" => Source::Gsr, _ => Source::Boston };
633 let result = if op == "BOSTON" { market.boston() } else { market.deferred(op == "GSR", words.len() == 3) };
634 let mut output = vec![format!("{op} {name} {}={}", if op == "BOSTON" { "applications" } else { "proposals" }, result.proposals)];
635 output.extend(result.history);
636 output.push(format!("MATCHING {}", market.pairs(&result.matching)));
637 market.current = Some((source, result.matching));
638 Ok(output)
639 }
640 "HR" => Ok(market.hr()),
641 "MANIPULATE" => {
642 let a = market.agent(words[2], Some(Side::Proposer))?;
643 Ok(market.manipulate(a))
644 }
645 "STATS" => market.stats(),
646 "LATTICE" | "EGALITARIAN" | "MINREGRET" => {
647 let lattice = market.enumerate()?;
648 if op == "LATTICE" {
649 let mut output = vec![format!("LATTICE {name} matchings={} rotations={}", lattice.matchings.len(), lattice.rotations.len())];
650 for (i, candidate) in lattice.matchings.iter().enumerate() {
651 let origin = candidate.origin.map_or_else(|| "-".into(), |n| n.to_string());
652 let via = candidate.via.map_or_else(|| "-".into(), |n| n.to_string());
653 output.push(format!("MATCHING {} from={origin} via={via} cost={} regret={} {}", i + 1,
654 candidate.cost, candidate.regret, market.pairs(&candidate.matching)));
655 }
656 for (i, rotation) in lattice.rotations.iter().enumerate() {
657 let notation = rotation.pairs.iter().map(|&(p, r)| format!("{}-{}", market.agents[p].name, market.agents[r].name))
658 .collect::<Vec<_>>().join(" ");
659 output.push(format!("ROTATION {} {notation}", i + 1));
660 }
661 Ok(output)
662 } else {
663 let egalitarian = op == "EGALITARIAN";
664 let (index, candidate) = lattice.matchings.iter().enumerate()
665 .min_by_key(|(_, c)| if egalitarian { c.cost } else { c.regret }).unwrap();
666 let value = if egalitarian { candidate.cost } else { candidate.regret };
667 let output = vec![format!("{op} {name} matching={} {}={value}", index + 1, if egalitarian { "cost" } else { "regret" }),
668 format!("MATCHING {}", market.pairs(&candidate.matching))];
669 market.current = Some((if egalitarian { Source::Egalitarian } else { Source::MinRegret }, candidate.matching.clone()));
670 Ok(output)
671 }
672 }
673 _ => unreachable!(),
674 }
675 }
676 }
677}

679fn main() {
680 let mut input = String::new();
681 io::stdin().read_to_string(&mut input).unwrap();
682 let mut lines = input.lines();
683 let count: usize = lines.next().unwrap().parse().unwrap();
684 let mut markets = HashMap::new();
685 let mut output = String::new();
686 for line in lines.take(count) {
687 let words: Vec<_> = line.split(' ').collect();
688 match execute(&mut markets, &words) {
689 Ok(response) => for text in response { output.push_str(&text); output.push('\n'); },
690 Err(error) => { output.push_str("ERROR "); output.push_str(&error); output.push('\n'); }
691 }
692 }
693 io::stdout().lock().write_all(output.as_bytes()).unwrap();
694}
