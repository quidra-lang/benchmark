1class Version
2 include Comparable
3 attr_reader :core, :pre
4 NUMBER = /\A(?:0|[1-9][0-9]{0,3})\z/
5
6 def initialize(core, pre = nil)
7 @core = core
8 @pre = pre
9 end
10
11 def self.parse(text)
12 match = /\A(0|[1-9][0-9]{0,3})\.(0|[1-9][0-9]{0,3})\.(0|[1-9][0-9]{0,3})(?:-([a-z0-9]+(?:\.[a-z0-9]+)*))?\z/.match(text)
13 return nil unless match
14 pre = match[4]&.split('.')
15 return nil if pre&.any? { |part| part.match?(/\A[0-9]+\z/) && !NUMBER.match?(part) }
16 new(match.captures.take(3).map(&:to_i), pre)
17 end
18
19 def <=>(other)
20 comparison = @core <=> other.core
21 return comparison unless comparison.zero?
22 return 0 if @pre.nil? && other.pre.nil?
23 return 1 if @pre.nil?
24 return -1 if other.pre.nil?
25 @pre.zip(other.pre).each do |a, b|
26 return 1 unless b
27 next if a == b
28 an = a.match?(/\A[0-9]+\z/)
29 bn = b.match?(/\A[0-9]+\z/)
30 return an && bn ? a.to_i <=> b.to_i : (an == bn ? a <=> b : an ? -1 : 1)
31 end
32 @pre.length <=> other.pre.length
33 end
34
35 def to_s
36 @core.join('.') + (@pre ? '-' + @pre.join('.') : '')
37 end
38end

40class Constraint
41 attr_reader :alternatives
42
43 def initialize(alternatives)
44 @alternatives = alternatives
45 end
46
47 def self.parse(text)
48 alternatives = []
49 text.split(' || ', -1).each do |alternative|
50 return nil if alternative.empty?
51 comparators = []
52 gates = []
53 alternative.split(/ /, -1).each do |term|
54 operator = %w[>= <= > < = ^ ~].find { |op| term.start_with?(op) } || ''
55 pattern = term[operator.length..]
56 version = Version.parse(pattern)
57 if version && version.pre
58 core = version.core
59 k = 3
60 gates << core
61 else
62 parts = pattern.split('.', -1)
63 return nil unless parts.length.between?(1, 3)
64 numbers = []
65 wildcard = false
66 parts.each do |part|
67 if %w[x *].include?(part)
68 wildcard = true
69 elsif Version::NUMBER.match?(part) && !wildcard
70 numbers << part.to_i
71 else
72 return nil
73 end
74 end
75 k = numbers.length
76 core = numbers + Array.new(3 - k, 0)
77 version = Version.new(core)
78 end
79 upper = lambda do |i|
80 values = core.dup
81 values[i - 1] += 1
82 (i...3).each { |j| values[j] = 0 }
83 Version.new(values)
84 end
85 if k.zero?
86 comparators << ['<', Version.new([0, 0, 0], ['0'])] if %w[> <].include?(operator)
87 else
88 case operator
89 when '', '='
90 comparators.concat(k == 3 ? [['=', version]] : [['>=', version], ['<', upper.call(k)]])
91 when '>'
92 comparators << [k == 3 ? '>' : '>=', k == 3 ? version : upper.call(k)]
93 when '<='
94 comparators << [k == 3 ? '<=' : '<', k == 3 ? version : upper.call(k)]
95 when '>=', '<'
96 comparators << [operator, version]
97 when '~'
98 comparators.concat([['>=', version], ['<', upper.call(k == 1 ? 1 : 2)]])
99 when '^'
100 i = core[0] != 0 || k == 1 ? 1 : (core[1] != 0 || k == 2 ? 2 : 3)
101 comparators.concat([['>=', version], ['<', upper.call(i)]])
102 end
103 end
104 end
105 alternatives << [comparators, gates]
106 end
107 new(alternatives)
108 end
109
110 def match?(version)
111 @alternatives.any? do |comparators, gates|
112 (version.pre.nil? || gates.include?(version.core)) && comparators.all? do |operator, bound|
113 comparison = version <=> bound
114 case operator
115 when '=' then comparison.zero?
116 when '>' then comparison.positive?
117 when '>=' then comparison >= 0
118 when '<' then comparison.negative?
119 else comparison <= 0
120 end
121 end
122 end
123 end
124
125 def to_s
126 @alternatives.map do |comparators, _|
127 comparators.empty? ? '*' : comparators.map { |operator, version| operator + version.to_s }.join(' ')
128 end.join(' || ')
129 end
130end

134class PackageManager
135 def initialize
136 @registry = {}
137 @constraints = {}
138 @requests = []
139 @installed = {}
140 @held = []
141 @advisories = {}
142 @history = []
143 end
144
145 def version(text)
146 value = Version.parse(text)
147 raise PackageError, "invalid version '#{text}'" unless value
148 value
149 end
150
151 def constraint(text)
152 @constraints[text] ||= Constraint.parse(text)
153 raise PackageError, "invalid constraint '#{text}'" unless @constraints[text]
154 @constraints[text]
155 end
156
157 def matches?(text, value)
158 constraint(text).match?(value.is_a?(Version) ? value : Version.parse(value))
159 end
160
161 def record(name, value)
162 @registry[name]&.[](value)
163 end
164
165 def versions(name)
166 (@registry[name] || {}).keys.sort { |a, b| Version.parse(b) <=> Version.parse(a) }
167 end
168
169 def advisories(name, value)
170 @advisories.keys.sort.select { |id| @advisories[id][0] == name && matches?(@advisories[id][1], value) }
171 end
172
173 def display(name, value)
174 flags = []
175 flags << 'yanked' if record(name, value)[:yanked]
176 flags << 'installed' if @installed[name] == value
177 "#{name}@#{value}#{flags.empty? ? '' : " (#{flags.join(', ')})"}"
178 end
179
180 def dependencies(name, installed = @installed)
181 record(name, installed[name])[:deps]
182 end

184 def resolve(requests, preferred, audit = false)
185 selected = {}
186 reasons = {}
187 failure = nil
188 failure_depth = -1
189 usable = lambda { |name, value| (!record(name, value)[:yanked] || @installed[name] == value) && (!audit || advisories(name, value).empty?) }
190 save_failure = lambda do |chain, reason, other = nil|
191 if selected.length > failure_depth
192 failure_depth = selected.length
193 failure = ['UNSATISFIABLE'] + chain.map { |line| ' ' + line } + [' ' + reason]
194 failure += other.map { |line| ' ' + line } if other
195 end
196 end
197 search = lambda do |requirements|
198 return true if requirements.empty?
199 name, text, chain = requirements.first
200 rest = requirements.drop(1)
201 if selected.key?(name)
202 return search.call(rest) if matches?(text, selected[name])
203 save_failure.call(chain, "#{name}@#{selected[name]} was selected by:", reasons[name])
204 return false
205 end
206 if @held.include?(name)
207 value = @installed[name]
208 unless matches?(text, value) && usable.call(name, value)
209 save_failure.call(chain, "#{name}@#{value} is held")
210 return false
211 end
212 candidates = [value]
213 else
214 candidates = versions(name).select { |v| matches?(text, v) && usable.call(name, v) }
215 if preferred.include?(name) && candidates.include?(@installed[name])
216 candidates.delete(@installed[name])
217 candidates.unshift(@installed[name])
218 end
219 if candidates.empty?
220 save_failure.call(chain, "no version of #{name} matches #{text}")
221 return false
222 end
223 end
224 candidates.each do |value|
225 conflicting = selected.keys.sort.find do |q|
226 record(name, value)[:conflicts].any? { |d, c| d == q && matches?(c, selected[q]) } ||
227 record(q, selected[q])[:conflicts].any? { |d, c| d == name && matches?(c, value) }
228 end
229 if conflicting
230 save_failure.call(chain, "#{name}@#{value} conflicts with #{conflicting}@#{selected[conflicting]}, which was selected by:", reasons[conflicting])
231 next
232 end
233 selected[name] = value
234 reasons[name] = chain
235 extra = record(name, value)[:deps].map { |d, c| [d, c, chain + ["#{name}@#{value} requires #{d} #{c}"]] }
236 return true if search.call(extra + rest)
237 selected.delete(name)
238 reasons.delete(name)
239 end
240 false
241 end
242 requirements = requests.map { |name, text| [name, text, ["requested #{name} #{text}"]] }
243 search.call(requirements) ? [selected, nil] : [nil, failure]
244 end

246 def plan(new_state)
247 old = @installed
248 arranged = []
249 counts = Hash.new(0)
250 until arranged.length == new_state.length
251 name = (new_state.keys - arranged).sort.find { |n| dependencies(n, new_state).all? { |d, _| arranged.include?(d) } }
252 arranged << name
253 value = new_state[name]
254 if !old.key?(name)
255 puts "install #{name}@#{value}"
256 counts[:installed] += 1
257 elsif old[name] != value
258 kind = Version.parse(value) > Version.parse(old[name]) ? :upgraded : :downgraded
259 puts "#{kind == :upgraded ? 'upgrade' : 'downgrade'} #{name} #{old[name]} -> #{value}"
260 counts[kind] += 1
261 end
262 end
263 remaining = old.keys - new_state.keys
264 until remaining.empty?
265 name = remaining.sort.find { |n| remaining.none? { |q| dependencies(q, old).any? { |d, _| d == n } } }
266 puts "remove #{name}@#{old[name]}"
267 counts[:removed] += 1
268 remaining.delete(name)
269 end
270 if counts.empty?
271 puts 'no changes'
272 else
273 puts "#{counts[:installed]} installed, #{counts[:upgraded]} upgraded, #{counts[:downgraded]} downgraded, #{counts[:removed]} removed"
274 end
275 end
276
277 def update(requests, installed)
278 @history << [@requests.map(&:dup), @installed.dup]
279 @requests = requests
280 @installed = installed
281 @held &= installed.keys
282 end
283
284 def resolve_and_plan(requests, preferred, audit = false, simulate = false)
285 installed, failure = resolve(requests, preferred, audit)
286 if failure
287 puts failure
288 else
289 plan(installed)
290 update(requests, installed) unless simulate
291 end
292 end

294 def require_installed(name)
295 raise PackageError, "#{name} is not installed" unless @installed.key?(name)
296 end
297
298 def required_by(name)
299 @installed.keys.sort.find { |n| dependencies(n).any? { |d, _| d == name } }
300 end
301
302 def tree(name)
303 roots = name ? [name] : @requests.map(&:first).sort
304 require_installed(name) if name
305 if roots.empty?
306 puts '(empty)'
307 return
308 end
309 seen = []
310 visit = lambda do |n, text, prefix, indent|
311 body = "#{prefix}#{n}@#{@installed[n]}#{text ? " [#{text}]" : ''}"
312 if seen.include?(n)
313 puts body + ' (*)'
314 return
315 end
316 puts body
317 seen << n
318 deps = dependencies(n)
319 deps.each_with_index do |(d, c), i|
320 last = i == deps.length - 1
321 visit.call(d, c, indent + (last ? '`-- ' : '|-- '), indent + (last ? ' ' : '| '))
322 end
323 end
324 roots.each { |root| visit.call(root, @requests.find { |n, _| n == root }&.last, '', '') }
325 end

327 def execute(line)
328 command, rest = line.split(/ /, 2)
329 args = rest ? rest.split(/ /) : []
330 name = args[0]
331 case command
332 when 'PKG'
333 clauses = rest.split('; ')
334 name, value = clauses.shift.split(' ')
335 parsed = version(value)
336 raise PackageError, "#{name}@#{value} is already registered" if record(name, value)
337 deps = []
338 conflicts = []
339 clauses.each do |clause|
340 kind, other, text = clause.split(/ /, 3)
341 constraint(text)
342 (kind == 'DEP' ? deps : conflicts) << [other, text]
343 end
344 @registry[name] ||= {}
345 @registry[name][value] = { version: parsed, deps: deps, conflicts: conflicts, yanked: false }
346 puts "registered #{name}@#{value}"
347 when 'YANK', 'INFO'
348 value = args[1]
349 entry = record(name, value)
350 raise PackageError, "#{name}@#{value} is not registered" unless entry
351 if command == 'YANK'
352 raise PackageError, "#{name}@#{value} is already yanked" if entry[:yanked]
353 entry[:yanked] = true
354 puts "yanked #{name}@#{value}"
355 else
356 puts display(name, value)
357 entry[:deps].each { |d, c| puts " requires #{d} #{c}" }
358 entry[:conflicts].each { |d, c| puts " conflicts with #{d} #{c}" }
359 advisories(name, value).each { |id| puts " advisory #{id}" }
360 end
361 when 'COMPARE'
362 a = version(name)
363 b = version(args[1])
364 puts "#{name} #{['<', '=', '>'][(a <=> b) + 1]} #{args[1]}"
365 when 'MATCH'
366 value, text = rest.split(/ /, 2)
367 parsed = version(value)
368 puts(constraint(text).match?(parsed) ? 'yes' : 'no')
369 when 'RANGE'
370 puts constraint(rest).to_s
371 when 'VERSIONS'
372 name, text = rest.split(/ /, 2)
373 constraint(text) if text
374 raise PackageError, "unknown package #{name}" unless @registry.key?(name)
375 selected = versions(name).select { |v| !text || matches?(text, v) }
376 puts 'no matching versions' if selected.empty?
377 selected.each { |v| puts display(name, v) }
378 when 'DEPENDENTS'
379 value = args[1]
380 version(value) if value
381 lines = []
382 @registry.keys.sort.each do |n|
383 versions(n).each do |v|
384 record(n, v)[:deps].each { |d, c| lines << "#{n}@#{v} requires #{name} #{c}" if d == name && (!value || matches?(c, value)) }
385 end
386 end
387 puts(lines.empty? ? ['no dependents'] : lines)
388 when 'INSTALL', 'SIMULATE'
389 name, text = rest.split(/ /, 2)
390 text ||= '*'
391 constraint(text)
392 raise PackageError, "unknown package #{name}" unless @registry.key?(name)
393 requests = @requests.map(&:dup)
394 index = requests.index { |n, _| n == name }
395 index ? requests[index] = [name, text] : requests << [name, text]
396 resolve_and_plan(requests, @installed.keys, false, command == 'SIMULATE')
397 when 'REMOVE'
398 unless @requests.any? { |n, _| n == name }
399 raise PackageError, @installed.key?(name) ? "#{name} is required by #{required_by(name)}" : "#{name} is not installed"
400 end
401 requests = @requests.reject { |n, _| n == name }
402 keep = requests.map(&:first)
403 keep.each { |n| dependencies(n).each { |d, _| keep << d unless keep.include?(d) } }
404 if keep.include?(name)
405 puts "#{name}@#{@installed[name]} is still required by #{required_by(name)}"
406 update(requests, @installed.dup)
407 else
408 installed = @installed.select { |n, _| keep.include?(n) }
409 plan(installed)
410 update(requests, installed)
411 end
412 when 'UPGRADE'
413 require_installed(name) if name
414 resolve_and_plan(@requests.map(&:dup), name ? @installed.keys - [name] : [])
415 when 'ROLLBACK'
416 raise PackageError, 'nothing to roll back' if @history.empty?
417 requests, installed = @history.pop
418 plan(installed)
419 @requests = requests
420 @installed = installed
421 @held &= installed.keys
422 when 'HOLD'
423 require_installed(name)
424 raise PackageError, "#{name} is already held" if @held.include?(name)
425 @held << name
426 puts "held #{name}@#{@installed[name]}"
427 when 'UNHOLD'
428 raise PackageError, "#{name} is not held" unless @held.include?(name)
429 @held.delete(name)
430 puts "released #{name}"
431 when 'WHY'
432 require_installed(name)
433 if @requests.any? { |n, _| n == name }
434 puts "#{name}@#{@installed[name]} is explicitly installed"
435 else
436 queue = @requests.map(&:first).sort
437 parent = queue.to_h { |n| [n, nil] }
438 queue.each do |n|
439 dependencies(n).map(&:first).sort.each do |d|
440 next if parent.key?(d)
441 parent[d] = n
442 queue << d
443 end
444 end
445 path = [name]
446 path.unshift(parent[path.first]) while parent[path.first]
447 puts path.map { |n| "#{n}@#{@installed[n]}" }.join(' -> ')
448 end
449 when 'TREE'
450 tree(name)
451 when 'OUTDATED'
452 lines = []
453 @installed.keys.sort.each do |n|
454 current = @installed[n]
455 constraints = @requests.select { |d, _| d == n }.map(&:last)
456 @installed.each_key { |q| dependencies(q).each { |d, c| constraints << c if d == n } }
457 wanted = versions(n).find { |v| (!record(n, v)[:yanked] || v == current) && constraints.all? { |c| matches?(c, v) } }
458 latest = versions(n).find { |v| !record(n, v)[:yanked] && Version.parse(v).pre.nil? }
459 lines << "#{n} #{current} #{wanted} #{latest || '-'}" if wanted != current || (latest && Version.parse(latest) > Version.parse(current))
460 end
461 puts(lines.empty? ? ['all packages are up to date'] : lines)
462 when 'LOCK'
463 puts '(empty)' if @installed.empty?
464 @installed.keys.sort.each { |n| puts "#{n}@#{@installed[n]}" }
465 when 'STATUS'
466 puts 'no requests' if @requests.empty?
467 @requests.each { |n, c| puts "#{n} #{c} -> #{@installed[n]}" }
468 puts "#{@installed.length} installed, #{@held.length} held, #{@history.length} snapshots"
469 when 'ADVISORY'
470 id, n, text = rest.split(/ /, 3)
471 raise PackageError, "advisory #{id} already exists" if @advisories.key?(id)
472 constraint(text)
473 @advisories[id] = [n, text]
474 puts "advisory #{id} recorded"
475 when 'AUDIT'
476 if name == 'FIX'
477 resolve_and_plan(@requests.map(&:dup), @installed.keys, true)
478 else
479 lines = []
480 @installed.keys.sort.each do |n|
481 current = @installed[n]
482 ids = advisories(n, current)
483 next if ids.empty?
484 fix = versions(n).reverse.find { |v| Version.parse(v) > Version.parse(current) && Version.parse(v).pre.nil? && !record(n, v)[:yanked] && advisories(n, v).empty? }
485 lines << "#{n}@#{current}: #{ids.join(', ')} (#{fix ? "fix: #{fix}" : 'no fix'})"
486 end
487 puts(lines.empty? ? ['no known vulnerabilities'] : lines)
488 end
489 end
490 rescue PackageError => error
491 puts "error: #{error.message}"
492 end
493end

496count = STDIN.gets.to_i
497count.times { manager.execute(STDIN.gets.chomp) }
