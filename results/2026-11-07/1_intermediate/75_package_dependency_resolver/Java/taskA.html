23 static class Version implements Comparable<Version> {
24 int[] core;
25 List<String> prerelease;
26 String text;
27 Version(int[] core, List<String> prerelease) {
28 this.core = core;
29 this.prerelease = prerelease;
30 text = core[0] + "." + core[1] + "." + core[2] + (prerelease.isEmpty() ? "" : "-" + String.join(".", prerelease));
31 }
32 String coreText() {
33 return core[0] + "." + core[1] + "." + core[2];
34 }
35 boolean stable() {
36 return prerelease.isEmpty();
37 }
38 public int compareTo(Version other) {
39 for (int i = 0; i < 3; i++) if (core[i] != other.core[i]) return Integer.compare(core[i], other.core[i]);
40 if (stable() || other.stable()) return Boolean.compare(stable(), other.stable());
41 for (int i = 0; i < Math.min(prerelease.size(), other.prerelease.size()); i++) {
42 String a = prerelease.get(i), b = other.prerelease.get(i);
43 boolean an = digits(a), bn = digits(b);
44 int order = an && bn ? Integer.compare(Integer.parseInt(a), Integer.parseInt(b)) : an != bn ? (an ? -1 : 1) : a.compareTo(b);
45 if (order != 0) return order;
46 }
47 return Integer.compare(prerelease.size(), other.prerelease.size());
48 }
49 }

51 static Version parseVersion(String text) {
52 String[] halves = text.split("-", -1);
53 if (halves.length > 2) return null;
54 String[] parts = halves[0].split("\\.", -1);
55 if (parts.length != 3 || Arrays.stream(parts).anyMatch(p -> !number(p))) return null;
56 List<String> pre = new ArrayList<>();
57 if (halves.length == 2) for (String p : halves[1].split("\\.", -1)) {
58 if (!p.matches("[a-z0-9]+") || (digits(p) && !number(p))) return null;
59 pre.add(p);
60 }
61 return new Version(Arrays.stream(parts).mapToInt(Integer::parseInt).toArray(), pre);
62 }

67 return result;

70 record ComparatorTerm(String operator, Version version) {
71 boolean matches(Version value) {
72 int c = value.compareTo(version);
73 return switch (operator) {
74 case "=" -> c == 0;
75 case ">" -> c > 0;
76 case ">=" -> c >= 0;
77 case "<" -> c < 0;
78 default -> c <= 0;
79 };
80 }
81 public String toString() {
82 return operator + version.text;
83 }
84 }

86 static class Alternative {
87 List<ComparatorTerm> comparators = new ArrayList<>();
88 Set<String> prereleaseCores = new HashSet<>();
89 void add(String operator, Version version) {
90 comparators.add(new ComparatorTerm(operator, version));
91 }
92 boolean matches(Version v) {
93 return (v.stable() || prereleaseCores.contains(v.coreText())) && comparators.stream().allMatch(c -> c.matches(v));
94 }
95 String normal() {
96 return comparators.isEmpty() ? "*" : String.join(" ", comparators.stream().map(Object::toString).toList());
97 }
98 }

100 static class Constraint {
101 List<Alternative> alternatives = new ArrayList<>();
102 boolean matches(Version v) {
103 return alternatives.stream().anyMatch(a -> a.matches(v));
104 }
105 String normal() {
106 return String.join(" || ", alternatives.stream().map(Alternative::normal).toList());

110 static Constraint parseConstraint(String text) {
111 Constraint result = new Constraint();
112 for (String branch : text.split(" \\|\\| ", -1)) {
113 Alternative a = new Alternative();
114 for (String term : branch.split(" ", -1)) {
115 if (term.isEmpty()) return null;
116 String operator = "", pattern = term;
117 for (String op : List.of(">=", "<=", ">", "<", "=", "^", "~")) if (term.startsWith(op)) {
118 operator = op;
119 pattern = term.substring(op.length());
120 break;
121 }
122 Version lower = parseVersion(pattern);
123 int count;
124 if (lower != null && !lower.stable()) {
125 count = 3;
126 a.prereleaseCores.add(lower.coreText());
127 } else {
128 String[] parts = pattern.split("\\.", -1);
129 if (parts.length < 1 || parts.length > 3) return null;
130 int[] core = new int[3];
131 count = 0;
132 boolean wildcard = false;
133 for (int i = 0; i < parts.length; i++) {
134 if (parts[i].equals("x") || parts[i].equals("*")) wildcard = true;
135 else {
136 if (wildcard || !number(parts[i])) return null;
137 core[i] = Integer.parseInt(parts[i]);
138 count++;
139 }
140 }
141 lower = new Version(core, List.of());
142 }
143 if (count == 0) {
144 if (operator.equals(">") || operator.equals("<")) a.add("<", new Version(new int[]{0, 0, 0}, List.of("0")));
145 continue;
146 }
147 Version[] upper = new Version[3];
148 for (int i = 0; i < 3; i++) {
149 int[] core = lower.core.clone();
150 core[i]++;
151 for (int j = i + 1; j < 3; j++) core[j] = 0;
152 upper[i] = new Version(core, List.of());
153 }
154 switch (operator) {
155 case "", "=" -> {
156 if (count == 3) a.add("=", lower);
157 else {
158 a.add(">=", lower);
159 a.add("<", upper[count - 1]);
160 }
161 }
162 case ">" -> a.add(count == 3 ? ">" : ">=", count == 3 ? lower : upper[count - 1]);
163 case "<=" -> a.add(count == 3 ? "<=" : "<", count == 3 ? lower : upper[count - 1]);
164 case ">=", "<" -> a.add(operator, lower);
165 case "~" -> {
166 a.add(">=", lower);
167 a.add("<", upper[count == 1 ? 0 : 1]);
168 }
169 case "^" -> {
170 int i = lower.core[0] != 0 || count == 1 ? 0 : lower.core[1] != 0 || count == 2 ? 1 : 2;
171 a.add(">=", lower);
172 a.add("<", upper[i]);
173 }
174 }
175 }
176 result.alternatives.add(a);
177 }
178 return result;
179 }

249 class Resolution {
250 TreeMap<String, PackageVersion> selected = new TreeMap<>();
251 Map<String, Requirement> reasons = new HashMap<>();
252 Set<String> preferred;
253 boolean audit;
254 int failureDepth = -1;
255 List<String> failure;
256 Resolution(Set<String> preferred, boolean audit) {
257 this.preferred = preferred;
258 this.audit = audit;
259 }
260 boolean usable(PackageVersion p) {
261 return (!p.yanked || installed.get(p.name) == p) && (!audit || !vulnerable(p));
262 }
263 void record(Requirement r, String reason, Requirement other) {
264 if (selected.size() <= failureDepth) return;
265 failureDepth = selected.size();
266 failure = new ArrayList<>(List.of("UNSATISFIABLE"));
267 for (String line : r.chain()) failure.add(" " + line);
268 failure.add(" " + reason);
269 if (other != null) for (String line : other.chain()) failure.add(" " + line);
270 }
271 boolean search(List<Requirement> pending) {
272 if (pending.isEmpty()) return true;
273 Requirement r = pending.getFirst();
274 Clause clause = r.clause;
275 String name = clause.name;
276 List<Requirement> rest = pending.subList(1, pending.size());
277 if (selected.containsKey(name)) {
278 PackageVersion p = selected.get(name);
279 if (clause.constraint.matches(p.version)) return search(rest);
280 record(r, p + " was selected by:", reasons.get(name));
281 return false;
282 }
283 List<PackageVersion> candidates = new ArrayList<>();
284 if (held.contains(name)) {
285 PackageVersion p = installed.get(name);
286 if (!clause.constraint.matches(p.version) || !usable(p)) {
287 record(r, p + " is held", null);
288 return false;
289 }
290 candidates.add(p);
291 } else {
292 for (PackageVersion p : versions(name)) if (clause.constraint.matches(p.version) && usable(p)) candidates.add(p);
293 PackageVersion current = installed.get(name);
294 if (preferred.contains(name) && candidates.remove(current)) candidates.addFirst(current);
295 if (candidates.isEmpty()) {
296 record(r, "no version of " + name + " matches " + clause.text, null);
297 return false;
298 }
299 }
300 for (PackageVersion p : candidates) {
301 PackageVersion conflict = null;
302 for (PackageVersion q : selected.values()) {
303 if (p.conflicts.stream().anyMatch(d -> d.name.equals(q.name) && d.constraint.matches(q.version)) || q.conflicts.stream().anyMatch(d -> d.name.equals(name) && d.constraint.matches(p.version))) {
304 conflict = q;
305 break;
306 }
307 }
308 if (conflict != null) {
309 record(r, p + " conflicts with " + conflict + ", which was selected by:", reasons.get(conflict.name));
310 continue;
311 }
312 selected.put(name, p);
313 reasons.put(name, r);
314 List<Requirement> next = new ArrayList<>();
315 for (Clause d : p.dependencies) next.add(new Requirement(d, p, r));
316 next.addAll(rest);
317 if (search(next)) return true;
318 selected.remove(name);
319 reasons.remove(name);
320 }
321 return false;
322 }
323 boolean solve(List<Clause> list) {
324 return search(list.stream().map(c -> new Requirement(c, null, null)).toList());
325 }
326 }

328 List<String> plan(TreeMap<String, PackageVersion> next) {
329 List<String> out = new ArrayList<>();
330 int[] counts = new int[4];
331 Set<String> arranged = new HashSet<>();
332 while (arranged.size() < next.size()) {
333 String name = next.keySet().stream().filter(n -> !arranged.contains(n) && next.get(n).dependencies.stream().allMatch(d -> arranged.contains(d.name))).findFirst().orElseThrow();
334 arranged.add(name);
335 PackageVersion p = next.get(name), old = installed.get(name);
336 if (old == null) {
337 out.add("install " + p);
338 counts[0]++;
339 } else if (old != p) {
340 boolean newer = p.version.compareTo(old.version) > 0;
341 out.add((newer ? "upgrade " : "downgrade ") + name + " " + old.version.text + " -> " + p.version.text);
342 counts[newer ? 1 : 2]++;
343 }
344 }
345 TreeSet<String> removed = new TreeSet<>(installed.keySet());
346 removed.removeAll(next.keySet());
347 while (!removed.isEmpty()) {
348 String name = removed.stream().filter(n -> removed.stream().noneMatch(q -> installed.get(q).dependencies.stream().anyMatch(d -> d.name.equals(n)))).findFirst().orElseThrow();
349 out.add("remove " + installed.get(name));
350 counts[3]++;
351 removed.remove(name);
352 }
353 if (out.isEmpty()) out.add("no changes");
354 else out.add(counts[0] + " installed, " + counts[1] + " upgraded, " + counts[2] + " downgraded, " + counts[3] + " removed");
355 return out;
356 }

358 void update(List<Clause> list, TreeMap<String, PackageVersion> next) {
359 history.push(new Snapshot(new ArrayList<>(requests), new TreeMap<>(installed)));
360 requests = new ArrayList<>(list);
361 installed = new TreeMap<>(next);
362 held.retainAll(installed.keySet());
363 }

365 List<String> resolve(List<Clause> list, Set<String> preferred, boolean audit, boolean change) {
366 Resolution r = new Resolution(preferred, audit);
367 if (!r.solve(list)) return r.failure;
368 List<String> result = plan(r.selected);
369 if (change) update(list, r.selected);
370 return result;
371 }

381 Set<String> reachable(List<Clause> list) {
382 Set<String> seen = new TreeSet<>();
383 ArrayDeque<String> queue = new ArrayDeque<>();
384 for (Clause c : list) if (seen.add(c.name)) queue.add(c.name);
385 while (!queue.isEmpty()) for (Clause d : installed.get(queue.remove()).dependencies) if (seen.add(d.name)) queue.add(d.name);

389 void tree(String name, String constraint, String prefix, String indent, Set<String> seen, List<String> out) {
390 PackageVersion p = installed.get(name);
391 String body = prefix + p + (constraint == null ? "" : " [" + constraint + "]");
392 if (!seen.add(name)) {
393 out.add(body + " (*)");
394 return;
395 }
396 out.add(body);
397 for (int i = 0; i < p.dependencies.size(); i++) {
398 Clause d = p.dependencies.get(i);
399 boolean last = i + 1 == p.dependencies.size();
400 tree(d.name, d.text, indent + (last ? "`-- " : "|-- "), indent + (last ? " " : "| "), seen, out);
401 }

404 List<String> command(String line) {
405 int space = line.indexOf(' ');
406 String c = space < 0 ? line : line.substring(0, space), rest = space < 0 ? "" : line.substring(space + 1);
407 String[] w = rest.split(" ", 3);
408 List<String> out = new ArrayList<>();
409 switch (c) {
410 case "PKG" -> {
411 String[] clauses = rest.split("; ", -1), header = clauses[0].split(" ");
412 Version v = checkedVersion(header[1]);
413 if (registered(header[0], header[1]) != null) fail(header[0] + "@" + header[1] + " is already registered");
414 PackageVersion p = new PackageVersion(header[0], v);
415 for (int i = 1; i < clauses.length; i++) {
416 String[] a = clauses[i].split(" ", 3);
417 Clause d = clause(a[1], a[2]);
418 (a[0].equals("DEP") ? p.dependencies : p.conflicts).add(d);
419 }
420 registry.computeIfAbsent(p.name, n -> new ArrayList<>()).add(p);
421 registry.get(p.name).sort((a, b) -> b.version.compareTo(a.version));
422 out.add("registered " + p);
423 }
424 case "YANK", "INFO" -> {
425 PackageVersion p = checkedPackage(w[0], w[1]);
426 if (c.equals("YANK")) {
427 if (p.yanked) fail(p + " is already yanked");
428 p.yanked = true;
429 out.add("yanked " + p);
430 } else {
431 out.add(display(p));
432 for (Clause d : p.dependencies) out.add(" requires " + d.name + " " + d.text);
433 for (Clause d : p.conflicts) out.add(" conflicts with " + d.name + " " + d.text);
434 for (var a : advisories.entrySet()) if (a.getValue().name.equals(p.name) && a.getValue().constraint.matches(p.version)) out.add(" advisory " + a.getKey());
435 }
436 }
437 case "VERSIONS", "INSTALL", "SIMULATE" -> {
438 String name = w[0], text = rest.length() == name.length() ? null : rest.substring(name.length() + 1);
439 Constraint constraint = text == null ? null : checkedConstraint(text);
440 if (!registry.containsKey(name)) fail("unknown package " + name);
441 if (c.equals("VERSIONS")) {
442 for (PackageVersion p : versions(name)) if (constraint == null || constraint.matches(p.version)) out.add(display(p));
443 if (out.isEmpty()) out.add("no matching versions");
444 } else {
445 Clause requirement = text == null ? clause(name, "*") : new Clause(name, text, constraint);
446 List<Clause> list = new ArrayList<>(requests);
447 Clause old = request(name);
448 if (old == null) list.add(requirement);
449 else list.set(list.indexOf(old), requirement);
450 out = resolve(list, new HashSet<>(installed.keySet()), false, c.equals("INSTALL"));
451 }
452 }
453 case "DEPENDENTS" -> {
454 Version v = w.length > 1 ? checkedVersion(w[1]) : null;
455 for (List<PackageVersion> list : registry.values()) for (PackageVersion p : list) for (Clause d : p.dependencies) if (d.name.equals(w[0]) && (v == null || d.constraint.matches(v))) out.add(p + " requires " + d.name + " " + d.text);
456 if (out.isEmpty()) out.add("no dependents");
457 }
458 case "COMPARE" -> {
459 Version a = checkedVersion(w[0]), b = checkedVersion(w[1]);
460 out.add(w[0] + (a.compareTo(b) < 0 ? " < " : a.compareTo(b) > 0 ? " > " : " = ") + w[1]);
461 }
462 case "MATCH" -> {
463 Version v = checkedVersion(w[0]);
464 Constraint constraint = checkedConstraint(rest.substring(w[0].length() + 1));
465 out.add(constraint.matches(v) ? "yes" : "no");
466 }
467 case "RANGE" -> out.add(checkedConstraint(rest).normal());
468 case "REMOVE" -> {
469 Clause old = request(rest);
470 if (old == null) fail(installed.containsKey(rest) ? rest + " is required by " + requiredBy(rest) : rest + " is not installed");
471 List<Clause> list = new ArrayList<>(requests);
472 list.remove(old);
473 Set<String> keep = reachable(list);
474 if (keep.contains(rest)) {
475 out.add(installed.get(rest) + " is still required by " + requiredBy(rest));
476 update(list, installed);
477 } else {
478 TreeMap<String, PackageVersion> next = new TreeMap<>(installed);
479 next.keySet().retainAll(keep);
480 out = plan(next);
481 update(list, next);
482 }
483 }
484 case "UPGRADE" -> {
485 if (!rest.isEmpty() && !installed.containsKey(rest)) fail(rest + " is not installed");
486 Set<String> preferred = new HashSet<>();
487 if (!rest.isEmpty()) {
488 preferred.addAll(installed.keySet());
489 preferred.remove(rest);
490 }
491 out = resolve(requests, preferred, false, true);
492 }
493 case "ROLLBACK" -> {
494 if (history.isEmpty()) fail("nothing to roll back");
495 Snapshot snapshot = history.pop();
496 out = plan(snapshot.installed);
497 requests = snapshot.requests;
498 installed = snapshot.installed;
499 held.retainAll(installed.keySet());
500 }
501 case "HOLD" -> {
502 if (!installed.containsKey(rest)) fail(rest + " is not installed");
503 if (!held.add(rest)) fail(rest + " is already held");
504 out.add("held " + installed.get(rest));
505 }
506 case "UNHOLD" -> {
507 if (!held.remove(rest)) fail(rest + " is not held");
508 out.add("released " + rest);
509 }
510 case "WHY" -> {
511 if (!installed.containsKey(rest)) fail(rest + " is not installed");
512 if (request(rest) != null) out.add(installed.get(rest) + " is explicitly installed");
513 else {
514 TreeSet<String> roots = new TreeSet<>();
515 for (Clause r : requests) roots.add(r.name);
516 Map<String, String> parent = new HashMap<>();
517 for (String root : roots) parent.put(root, null);
518 ArrayDeque<String> queue = new ArrayDeque<>(roots);
519 while (!queue.isEmpty()) {
520 String name = queue.remove();
521 List<String> children = installed.get(name).dependencies.stream().map(Clause::name).sorted().toList();
522 for (String child : children) if (!parent.containsKey(child)) {
523 parent.put(child, name);
524 queue.add(child);
525 }
526 }
527 List<String> path = new ArrayList<>();
528 for (String name = rest; name != null; name = parent.get(name)) path.add(installed.get(name).toString());
529 Collections.reverse(path);
530 out.add(String.join(" -> ", path));
531 }
532 }
533 case "TREE" -> {
534 if (!rest.isEmpty() && !installed.containsKey(rest)) fail(rest + " is not installed");
535 Set<String> seen = new HashSet<>();
536 List<String> roots = rest.isEmpty() ? requests.stream().map(Clause::name).sorted().toList() : List.of(rest);
537 if (roots.isEmpty()) out.add("(empty)");
538 for (String root : roots) {
539 Clause r = request(root);
540 tree(root, r == null ? null : r.text, "", "", seen, out);
541 }
542 }
543 case "OUTDATED" -> {
544 for (PackageVersion p : installed.values()) {
545 List<Constraint> relevant = new ArrayList<>();
546 Clause r = request(p.name);
547 if (r != null) relevant.add(r.constraint);
548 for (PackageVersion q : installed.values()) for (Clause d : q.dependencies) if (d.name.equals(p.name)) relevant.add(d.constraint);
549 PackageVersion wanted = versions(p.name).stream().filter(q -> (!q.yanked || q == p) && relevant.stream().allMatch(a -> a.matches(q.version))).findFirst().orElseThrow();
550 PackageVersion latest = versions(p.name).stream().filter(q -> !q.yanked && q.version.stable()).findFirst().orElse(null);
551 if (wanted != p || (latest != null && latest.version.compareTo(p.version) > 0)) out.add(p.name + " " + p.version.text + " " + wanted.version.text + " " + (latest == null ? "-" : latest.version.text));
552 }
553 if (out.isEmpty()) out.add("all packages are up to date");
554 }
555 case "LOCK" -> {
556 for (PackageVersion p : installed.values()) out.add(p.toString());
557 if (out.isEmpty()) out.add("(empty)");
558 }
559 case "STATUS" -> {
560 for (Clause r : requests) out.add(r.name + " " + r.text + " -> " + installed.get(r.name).version.text);
561 if (requests.isEmpty()) out.add("no requests");
562 out.add(installed.size() + " installed, " + held.size() + " held, " + history.size() + " snapshots");
563 }
564 case "ADVISORY" -> {
565 if (advisories.containsKey(w[0])) fail("advisory " + w[0] + " already exists");
566 Clause a = clause(w[1], w[2]);
567 advisories.put(w[0], a);
568 out.add("advisory " + w[0] + " recorded");
569 }
570 case "AUDIT" -> {
571 if (rest.equals("FIX")) out = resolve(requests, new HashSet<>(installed.keySet()), true, true);
572 else {
573 for (PackageVersion p : installed.values()) {
574 List<String> ids = advisories.entrySet().stream().filter(e -> e.getValue().name.equals(p.name) && e.getValue().constraint.matches(p.version)).map(Map.Entry::getKey).toList();
575 if (ids.isEmpty()) continue;
576 PackageVersion fix = null;
577 for (PackageVersion q : versions(p.name)) if (q.version.compareTo(p.version) > 0 && q.version.stable() && !q.yanked && !vulnerable(q)) fix = q;
578 out.add(p + ": " + String.join(", ", ids) + (fix == null ? " (no fix)" : " (fix: " + fix.version.text + ")"));
579 }
580 if (out.isEmpty()) out.add("no known vulnerabilities");
581 }
582 }
583 default -> throw new IllegalStateException();
584 }
585 return out;
586 }
