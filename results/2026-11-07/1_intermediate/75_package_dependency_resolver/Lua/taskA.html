17local function split(text, separator, limit)
18 local result, start = {}, 1
19 while not limit or #result + 1 < limit do
20 local at = text:find(separator, start, true)
21 if not at then
22 break
23 end
24 result[#result + 1] = text:sub(start, at - 1)
25 start = at + #separator
26 end
27 result[#result + 1] = text:sub(start)
28 return result
29end

44local function numeric(text)
45 return text ~= "" and not text:find("[^0-9]")
46end
47local function number_word(text)
48 return #text >= 1 and #text <= 4 and numeric(text) and (#text == 1 or text:sub(1, 1) ~= "0")
49end
50local function make_version(core, pre)
51 local text = table.concat(core, ".")
52 if #pre > 0 then
53 text = text .. "-" .. table.concat(pre, ".")
54 end
55 return {core = core, pre = pre, text = text}
56end
57local function parse_version(text)
58 local fields = split(text, "-", 2)
59 local parts, core, pre = split(fields[1], "."), {}, {}
60 if #parts ~= 3 then
61 return nil
62 end
63 for i, part in ipairs(parts) do
64 if not number_word(part) then
65 return nil
66 end
67 core[i] = tonumber(part)
68 end
69 if fields[2] then
70 pre = split(fields[2], ".")
71 for _, part in ipairs(pre) do
72 if part == "" or part:find("[^a-z0-9]") or numeric(part) and not number_word(part) then
73 return nil
74 end
75 end
76 end
77 return make_version(core, pre)
78end
79local function compare(a, b)
80 for i = 1, 3 do
81 if a.core[i] ~= b.core[i] then
82 return a.core[i] < b.core[i] and -1 or 1
83 end
84 end
85 if #a.pre == 0 and #b.pre > 0 then
86 return 1
87 elseif #b.pre == 0 and #a.pre > 0 then
88 return -1
89 end
90 for i = 1, math.min(#a.pre, #b.pre) do
91 local x, y = a.pre[i], b.pre[i]
92 if x ~= y then
93 local nx, ny = numeric(x), numeric(y)
94 if nx and ny then
95 return tonumber(x) < tonumber(y) and -1 or 1
96 elseif nx ~= ny then
97 return nx and -1 or 1
98 end
99 return x < y and -1 or 1
100 end
101 end
102 return #a.pre == #b.pre and 0 or #a.pre < #b.pre and -1 or 1
103end
104local function parse_constraint(text)
105 local result, forms = {alternatives = {}}, {}
106 for _, branch in ipairs(split(text, " || ")) do
107 local item = {comparisons = {}, prerelease = {}}
108 for _, term in ipairs(split(branch, " ")) do
109 if term == "" then
110 return nil
111 end
112 local op, pattern = "", term
113 for _, candidate in ipairs({">=", "<=", ">", "<", "=", "^", "~"}) do
114 if term:sub(1, #candidate) == candidate then
115 op, pattern = candidate, term:sub(#candidate + 1)
116 break
117 end
118 end
119 local core, pre, count = {0, 0, 0}, {}, 0
120 if pattern:find("-", 1, true) then
121 local value = parse_version(pattern)
122 if not value or #value.pre == 0 then
123 return nil
124 end
125 core, pre, count = value.core, value.pre, 3
126 item.prerelease[table.concat(core, ".")] = true
127 else
128 local parts, wildcard = split(pattern, "."), false
129 if #parts > 3 then
130 return nil
131 end
132 for i, part in ipairs(parts) do
133 if part == "x" or part == "*" then
134 wildcard = true
135 else
136 if wildcard or not number_word(part) then
137 return nil
138 end
139 core[i], count = tonumber(part), count + 1
140 end
141 end
142 end
143 local lower = make_version(core, pre)
144 local function upper(at)
145 local next_core = copy(core)
146 next_core[at] = next_core[at] + 1
147 for i = at + 1, 3 do
148 next_core[i] = 0
149 end
150 return make_version(next_core, {})
151 end
152 local function add(operator, value)
153 item.comparisons[#item.comparisons + 1] = {op = operator, value = value}
154 end
155 if count == 0 then
156 if op == ">" or op == "<" then
157 add("<", make_version({0, 0, 0}, {"0"}))
158 end
159 elseif op == "" or op == "=" then
160 if count == 3 then
161 add("=", lower)
162 else
163 add(">=", lower)
164 add("<", upper(count))
165 end
166 elseif op == ">" then
167 add(count == 3 and ">" or ">=", count == 3 and lower or upper(count))
168 elseif op == "<=" then
169 add(count == 3 and "<=" or "<", count == 3 and lower or upper(count))
170 elseif op == ">=" or op == "<" then
171 add(op, lower)
172 elseif op == "~" then
173 add(">=", lower)
174 add("<", upper(count == 1 and 1 or 2))
175 elseif op == "^" then
176 local at
177 if core[1] ~= 0 or count == 1 then
178 at = 1
179 elseif core[2] ~= 0 or count == 2 then
180 at = 2
181 else
182 at = 3
183 end
184 add(">=", lower)
185 add("<", upper(at))
186 end
187 end
188 local words = {}
189 for _, comparator in ipairs(item.comparisons) do
190 words[#words + 1] = comparator.op .. comparator.value.text
191 end
192 forms[#forms + 1] = #words > 0 and table.concat(words, " ") or "*"
193 result.alternatives[#result.alternatives + 1] = item
194 end
195 result.normal = table.concat(forms, " || ")
196 return result
197end
198local function constraint(text)
199 if not parsed[text] then
200 parsed[text] = parse_constraint(text)
201 if not parsed[text] then
202 fail("invalid constraint '%s'", text)
203 end
204 end
205 return parsed[text]
206end
207local function checked_version(text)
208 local value = parse_version(text)

214local function matches(value, text)
215 for _, branch in ipairs(parsed[text].alternatives) do
216 if #value.pre == 0 or branch.prerelease[table.concat(value.core, ".")] then
217 local valid = true
218 for _, comparator in ipairs(branch.comparisons) do
219 local order, op = compare(value, comparator.value), comparator.op
220 valid = op == "=" and order == 0 or op == ">" and order > 0 or op == ">=" and order >= 0 or
221 op == "<" and order < 0 or op == "<=" and order <= 0
222 if not valid then
223 break
224 end
225 end
226 if valid then
227 return true
228 end
229 end
230 end
231 return false
232end

258 local flags = {}

267local function resolve(new_requests, preferred, audit)
268 local selected, origins, failure, depth = {}, {}, nil, -1
269 local function record(requirement, reason, extra)
270 local count = #keys(selected)
271 if count <= depth then
272 return
273 end
274 depth, failure = count, {"UNSATISFIABLE"}
275 for _, line in ipairs(requirement.chain) do
276 failure[#failure + 1] = " " .. line
277 end
278 failure[#failure + 1] = " " .. reason
279 for _, line in ipairs(extra or {}) do
280 failure[#failure + 1] = " " .. line
281 end
282 end
283 local function search(todo)
284 if #todo == 0 then
285 return true
286 end
287 local requirement, rest = todo[1], {}
288 for i = 2, #todo do
289 rest[#rest + 1] = todo[i]
290 end
291 local name, text = requirement.name, requirement.text
292 if selected[name] then
293 if matches(selected[name].version, text) then
294 return search(rest)
295 end
296 record(requirement, label(selected[name]) .. " was selected by:", origins[name].chain)
297 return false
298 end
299 local candidates = {}
300 if held[name] then
301 local value = installed[name]
302 if matches(value.version, text) and usable(value, audit) then
303 candidates[1] = value
304 else
305 record(requirement, label(value) .. " is held")
306 return false
307 end
308 else
309 for _, value in ipairs(registry[name] or {}) do
310 if matches(value.version, text) and usable(value, audit) then
311 candidates[#candidates + 1] = value
312 end
313 end
314 if preferred[name] then
315 for i, value in ipairs(candidates) do
316 if installed[name] == value then
317 table.remove(candidates, i)
318 table.insert(candidates, 1, value)
319 break
320 end
321 end
322 end
323 if #candidates == 0 then
324 record(requirement, "no version of " .. name .. " matches " .. text)
325 return false
326 end
327 end
328 for _, candidate in ipairs(candidates) do
329 local conflicting
330 for _, other_name in ipairs(keys(selected)) do
331 local chosen, conflict = selected[other_name], false
332 for _, pair in ipairs(candidate.conflicts) do
333 if pair.name == other_name and matches(chosen.version, pair.text) then
334 conflict = true
335 break
336 end
337 end
338 if not conflict then
339 for _, pair in ipairs(chosen.conflicts) do
340 if pair.name == name and matches(candidate.version, pair.text) then
341 conflict = true
342 break
343 end
344 end
345 end
346 if conflict then
347 conflicting = other_name
348 break
349 end
350 end
351 if conflicting then
352 record(requirement, label(candidate) .. " conflicts with " .. label(selected[conflicting]) ..
353 ", which was selected by:", origins[conflicting].chain)
354 else
355 selected[name], origins[name] = candidate, requirement
356 local following = {}
357 for _, pair in ipairs(candidate.dependencies) do
358 local chain = copy(requirement.chain)
359 chain[#chain + 1] = label(candidate) .. " requires " .. pair.name .. " " .. pair.text
360 following[#following + 1] = {name = pair.name, text = pair.text, chain = chain}
361 end
362 for _, item in ipairs(rest) do
363 following[#following + 1] = item
364 end
365 if search(following) then
366 return true
367 end
368 selected[name], origins[name] = nil, nil
369 end
370 end
371 return false
372 end
373 local initial = {}
374 for _, pair in ipairs(new_requests) do
375 initial[#initial + 1] = {name = pair.name, text = pair.text, chain = {"requested " .. pair.name .. " " .. pair.text}}
376 end
377 if search(initial) then
378 return selected, nil
379 end
380 return nil, failure
381end

387local function plan(next_installed)
388 local arranged, added, upgraded, downgraded, removed = {}, 0, 0, 0, 0
389 local names = keys(next_installed)
390 while #keys(arranged) < #names do
391 for _, name in ipairs(names) do
392 if not arranged[name] then
393 local ready = true
394 for _, pair in ipairs(next_installed[name].dependencies) do
395 if not arranged[pair.name] then
396 ready = false
397 break
398 end
399 end
400 if ready then
401 local value, old = next_installed[name], installed[name]
402 if not old then
403 emit("install %s", label(value))
404 added = added + 1
405 elseif old ~= value then
406 if compare(value.version, old.version) > 0 then
407 emit("upgrade %s %s -> %s", name, old.version.text, value.version.text)
408 upgraded = upgraded + 1
409 else
410 emit("downgrade %s %s -> %s", name, old.version.text, value.version.text)
411 downgraded = downgraded + 1
412 end
413 end
414 arranged[name] = true
415 break
416 end
417 end
418 end
419 end
420 local dropping = {}
421 for name in pairs(installed) do
422 if not next_installed[name] then
423 dropping[name] = true
424 end
425 end
426 while next(dropping) do
427 for _, name in ipairs(keys(dropping)) do
428 local ready = true
429 for other in pairs(dropping) do
430 for _, pair in ipairs(installed[other].dependencies) do
431 if pair.name == name then
432 ready = false
433 break
434 end
435 end
436 if not ready then
437 break
438 end
439 end
440 if ready then
441 emit("remove %s", label(installed[name]))
442 dropping[name], removed = nil, removed + 1
443 break
444 end
445 end
446 end
447 if added + upgraded + downgraded + removed == 0 then
448 emit("no changes")
449 else
450 emit("%d installed, %d upgraded, %d downgraded, %d removed", added, upgraded, downgraded, removed)
451 end
452end

460 held[name] = nil

462 end
463end

480local function tree(root, seen, text, prefix, indent)
481 local value = installed[root]
482 local body = prefix .. label(value) .. (text and " [" .. text .. "]" or "")
483 if seen[root] then
484 emit("%s (*)", body)
485 return
486 end
487 emit("%s", body)
488 seen[root] = true
489 for i, pair in ipairs(value.dependencies) do
490 local mark, following = "|-- ", "| "
491 if i == #value.dependencies then
492 mark, following = "`-- ", " "
493 end
494 tree(pair.name, seen, pair.text, indent .. mark, indent .. following)
495 end
496end

505local function command(line)
506 local w = split(line, " ")
507 local cmd = w[1]
508 if cmd == "PKG" then
509 local clauses = split(line, "; ")
510 local head = split(clauses[1], " ")
511 local value = checked_version(head[3])
512 if find(head[2], head[3]) then
513 fail("%s@%s is already registered", head[2], head[3])
514 end
515 local candidate = {name = head[2], version = value, dependencies = {}, conflicts = {}}
516 for i = 2, #clauses do
517 local parts = split(clauses[i], " ", 3)
518 constraint(parts[3])
519 local list = parts[1] == "DEP" and candidate.dependencies or candidate.conflicts
520 list[#list + 1] = {name = parts[2], text = parts[3]}
521 end
522 registry[candidate.name] = registry[candidate.name] or {}
523 table.insert(registry[candidate.name], candidate)
524 table.sort(registry[candidate.name], function(a, b) return compare(a.version, b.version) > 0 end)
525 emit("registered %s", label(candidate))
526 elseif cmd == "YANK" or cmd == "INFO" then
527 local value = find(w[2], w[3])
528 if not value then
529 fail("%s@%s is not registered", w[2], w[3])
530 end
531 if cmd == "YANK" then
532 if value.yanked then
533 fail("%s is already yanked", label(value))
534 end
535 value.yanked = true
536 emit("yanked %s", label(value))
537 else
538 emit("%s", display(value))
539 for _, pair in ipairs(value.dependencies) do
540 emit(" requires %s %s", pair.name, pair.text)
541 end
542 for _, pair in ipairs(value.conflicts) do
543 emit(" conflicts with %s %s", pair.name, pair.text)
544 end
545 for _, id in ipairs(applicable(value)) do
546 emit(" advisory %s", id)
547 end
548 end
549 elseif cmd == "COMPARE" then
550 local a = checked_version(w[2])
551 local b = checked_version(w[3])
552 local relation = compare(a, b)
553 emit("%s %s %s", w[2], relation < 0 and "<" or relation > 0 and ">" or "=", w[3])
554 elseif cmd == "MATCH" then
555 local parts = split(line, " ", 3)
556 local value = checked_version(parts[2])
557 constraint(parts[3])
558 emit("%s", matches(value, parts[3]) and "yes" or "no")
559 elseif cmd == "RANGE" then
560 emit("%s", constraint(line:sub(7)).normal)
561 elseif cmd == "VERSIONS" or cmd == "INSTALL" or cmd == "SIMULATE" then
562 local parts = split(line, " ", 3)
563 local text = parts[3] or "*"
564 constraint(text)
565 if not registry[w[2]] then
566 fail("unknown package %s", w[2])
567 end
568 if cmd == "VERSIONS" then
569 local count = 0
570 for _, value in ipairs(registry[w[2]]) do
571 if not parts[3] or matches(value.version, text) then
572 emit("%s", display(value))
573 count = count + 1
574 end
575 end
576 if count == 0 then
577 emit("no matching versions")
578 end
579 else
580 local new_requests, found = copy_requests(requests), false
581 for _, pair in ipairs(new_requests) do
582 if pair.name == w[2] then
583 pair.text, found = text, true
584 break
585 end
586 end
587 if not found then
588 new_requests[#new_requests + 1] = {name = w[2], text = text}
589 end
590 local next_installed, failure = resolve(new_requests, all_preferred(), false)
591 if failure then
592 report_failure(failure)
593 else
594 plan(next_installed)
595 if cmd == "INSTALL" then
596 update(new_requests, next_installed, true)
597 end
598 end
599 end
600 elseif cmd == "DEPENDENTS" then
601 local selected = w[3] and checked_version(w[3]) or nil
602 local count = 0
603 for _, name in ipairs(keys(registry)) do
604 for _, value in ipairs(registry[name]) do
605 for _, pair in ipairs(value.dependencies) do
606 if pair.name == w[2] and (not selected or matches(selected, pair.text)) then
607 emit("%s requires %s %s", label(value), w[2], pair.text)
608 count = count + 1
609 end
610 end
611 end
612 end
613 if count == 0 then
614 emit("no dependents")
615 end
616 elseif cmd == "REMOVE" then
617 if not requested(w[2]) then
618 if installed[w[2]] then
619 fail("%s is required by %s", w[2], parent(w[2]))
620 end
621 fail("%s is not installed", w[2])
622 end
623 local new_requests, keep, queue = {}, {}, {}
624 for _, pair in ipairs(requests) do
625 if pair.name ~= w[2] then
626 new_requests[#new_requests + 1], queue[#queue + 1] = pair, pair.name
627 end
628 end
629 local at = 1
630 while at <= #queue do
631 local name = queue[at]
632 at = at + 1
633 if not keep[name] then
634 keep[name] = installed[name]
635 for _, pair in ipairs(installed[name].dependencies) do
636 queue[#queue + 1] = pair.name
637 end
638 end
639 end
640 if keep[w[2]] then
641 emit("%s is still required by %s", label(installed[w[2]]), parent(w[2]))
642 update(new_requests, installed, true)
643 else
644 plan(keep)
645 update(new_requests, keep, true)
646 end
647 elseif cmd == "UPGRADE" then
648 if w[2] and not installed[w[2]] then
649 fail("%s is not installed", w[2])
650 end
651 local preferred = w[2] and all_preferred() or {}
652 if w[2] then
653 preferred[w[2]] = nil
654 end
655 local next_installed, failure = resolve(requests, preferred, false)
656 if failure then
657 report_failure(failure)
658 else
659 plan(next_installed)
660 update(requests, next_installed, true)
661 end
662 elseif cmd == "ROLLBACK" then
663 if #history == 0 then
664 fail("nothing to roll back")
665 end
666 local saved = table.remove(history)
667 plan(saved.installed)
668 update(saved.requests, saved.installed, false)
669 elseif cmd == "HOLD" then
670 if not installed[w[2]] then
671 fail("%s is not installed", w[2])
672 end
673 if held[w[2]] then
674 fail("%s is already held", w[2])
675 end
676 held[w[2]] = true
677 emit("held %s", label(installed[w[2]]))
678 elseif cmd == "UNHOLD" then
679 if not held[w[2]] then
680 fail("%s is not held", w[2])
681 end
682 held[w[2]] = nil
683 emit("released %s", w[2])
684 elseif cmd == "WHY" then
685 local value = installed[w[2]]
686 if not value then
687 fail("%s is not installed", w[2])
688 end
689 if requested(w[2]) then
690 emit("%s is explicitly installed", label(value))
691 return
692 end
693 local previous, seen, queue, roots = {}, {}, {}, {}
694 for _, pair in ipairs(requests) do
695 roots[pair.name] = true
696 end
697 for _, name in ipairs(keys(roots)) do
698 queue[#queue + 1], seen[name] = name, true
699 end
700 local at = 1
701 while at <= #queue do
702 local name, deps = queue[at], {}
703 at = at + 1
704 for _, pair in ipairs(installed[name].dependencies) do
705 deps[pair.name] = true
706 end
707 for _, dep in ipairs(keys(deps)) do
708 if not seen[dep] then
709 seen[dep], previous[dep], queue[#queue + 1] = true, name, dep
710 end
711 end
712 end
713 local path, name = {}, w[2]
714 while name do
715 table.insert(path, 1, label(installed[name]))
716 name = previous[name]
717 end
718 emit("%s", table.concat(path, " -> "))
719 elseif cmd == "TREE" then
720 if w[2] and not installed[w[2]] then
721 fail("%s is not installed", w[2])
722 end
723 local roots = {}
724 if w[2] then
725 roots[1] = w[2]
726 else
727 for _, pair in ipairs(requests) do
728 roots[#roots + 1] = pair.name
729 end
730 table.sort(roots)
731 end
732 if #roots == 0 then
733 emit("(empty)")
734 end
735 local seen = {}
736 for _, name in ipairs(roots) do
737 tree(name, seen, requested(name), "", "")
738 end
739 elseif cmd == "OUTDATED" then
740 local count = 0
741 for _, name in ipairs(keys(installed)) do
742 local current, conditions = installed[name], {}
743 local text = requested(name)
744 if text then
745 conditions[#conditions + 1] = text
746 end
747 for _, value in pairs(installed) do
748 for _, pair in ipairs(value.dependencies) do
749 if pair.name == name then
750 conditions[#conditions + 1] = pair.text
751 end
752 end
753 end
754 local wanted, latest
755 for _, value in ipairs(registry[name]) do
756 if not value.yanked and #value.version.pre == 0 and not latest then
757 latest = value
758 end
759 if not wanted and (not value.yanked or value == current) then
760 local valid = true
761 for _, condition in ipairs(conditions) do
762 if not matches(value.version, condition) then
763 valid = false
764 break
765 end
766 end
767 if valid then
768 wanted = value
769 end
770 end
771 end
772 if wanted ~= current or latest and compare(latest.version, current.version) > 0 then
773 emit("%s %s %s %s", name, current.version.text, wanted.version.text, latest and latest.version.text or "-")
774 count = count + 1
775 end
776 end
777 if count == 0 then
778 emit("all packages are up to date")
779 end
780 elseif cmd == "LOCK" then
781 if not next(installed) then
782 emit("(empty)")
783 end
784 for _, name in ipairs(keys(installed)) do
785 emit("%s", label(installed[name]))
786 end
787 elseif cmd == "STATUS" then
788 if #requests == 0 then
789 emit("no requests")
790 end
791 for _, pair in ipairs(requests) do
792 emit("%s %s -> %s", pair.name, pair.text, installed[pair.name].version.text)
793 end
794 emit("%d installed, %d held, %d snapshots", #keys(installed), #keys(held), #history)
795 elseif cmd == "ADVISORY" then
796 local parts = split(line, " ", 4)
797 if advisories[w[2]] then
798 fail("advisory %s already exists", w[2])
799 end
800 constraint(parts[4])
801 advisories[w[2]] = {name = w[3], text = parts[4]}
802 emit("advisory %s recorded", w[2])
803 elseif cmd == "AUDIT" then
804 if w[2] then
805 local next_installed, failure = resolve(requests, all_preferred(), true)
806 if failure then
807 report_failure(failure)
808 else
809 plan(next_installed)
810 update(requests, next_installed, true)
811 end
812 else
813 local count = 0
814 for _, name in ipairs(keys(installed)) do
815 local current = installed[name]
816 local ids = applicable(current)
817 if #ids > 0 then
818 local fix
819 for i = #registry[name], 1, -1 do
820 local value = registry[name][i]
821 if compare(value.version, current.version) > 0 and #value.version.pre == 0 and
822 not value.yanked and #applicable(value) == 0 then
823 fix = value
824 break
825 end
826 end
827 emit("%s: %s (%s)", label(current), table.concat(ids, ", "), fix and "fix: " .. fix.version.text or "no fix")
828 count = count + 1
829 end
830 end
831 if count == 0 then
832 emit("no known vulnerabilities")
833 end
834 end
835 end
836end
837
838local count = tonumber(io.read("*l"))
839for _ = 1, count do
840 local ok, message = pcall(command, io.read("*l"))
841 if not ok then
842 emit("error: %s", message)
843 end
844end
845io.write(table.concat(output, "\n"), "\n")
