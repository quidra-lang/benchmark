5private fun number(s: String): Int? = if (s.length in 1..4 && s.all {
6 it in '0'..'9'
7}
8&& (s.length == 1 || s[0] != '0')) s.toInt() else null

10private data class Version(val core: List<Int>,val pre: List<String> = emptyList()) : Comparable<Version> {
11 val text get() = core.joinToString(".")+if (pre.isEmpty()) "" else "-"+pre.joinToString(".")
12 override fun compareTo(other: Version): Int {
13 for (i in 0..2) if (core[i] != other.core[i]) return core[i].compareTo(other.core[i])
14 if (pre.isEmpty() || other.pre.isEmpty()) return if (pre.isEmpty()) {
15 if (other.pre.isEmpty()) 0 else 1
16 } else -1
17 for (i in 0 until minOf(pre.size,other.pre.size)) {
18 val a = pre[i]
19 val b = other.pre[i]
20 if (a == b) continue
21 val na = a.all {
22 it in '0'..'9'
23 }
24 val nb = b.all {
25 it in '0'..'9'
26 }
27 return if (na && nb) a.toInt().compareTo(b.toInt()) else if (na != nb) {
28 if (na) -1 else 1
29 } else a.compareTo(b)
30 }
31 return pre.size.compareTo(other.pre.size)
32 }
33}

35private fun version(s: String): Version? {
36 val p = s.split('-')
37 if (p.size !in 1..2) return null
38 val parts = p[0].split('.')
39 if (parts.size != 3) return null
40 val core = parts.map {
41 number(it) ?: return null
42 }
43 val pre = if (p.size == 2) p[1].split('.') else emptyList()
44 for (id in pre) if (id.isEmpty() || !id.all {
45 it in 'a'..'z' || it in '0'..'9'
46 }
47 || (id.all {
48 it in '0'..'9'
49 }
50 && number(id) == null)) return null
51 return Version(core,pre)
52}

54private fun checkedVersion(s: String) = version(s) ?: fail("invalid version '$s'")

56private data class Comparator(val op: String,val version: Version) {
57 fun accepts(v: Version): Boolean {
58 val d = v.compareTo(version)
59 return when(op) {
60 "=" -> d == 0
61 ">" -> d > 0
62 ">=" -> d >= 0
63 "<" -> d < 0
64 else -> d <= 0
65 }
66 }
67}
68
69private data class Alternative(val comparisons: List<Comparator>,val prereleaseCores: Set<List<Int>>)
70
71private data class Constraint(val alternatives: List<Alternative>) {
72 fun accepts(v: Version) = alternatives.any {
73 (v.pre.isEmpty() || v.core in it.prereleaseCores) && it.comparisons.all {
74 c -> c.accepts(v)
75 }
76 }
77 fun normal() = alternatives.joinToString(" || ") {
78 a -> a.comparisons.joinToString(" ") {
79 it.op+it.version.text
80 }.ifEmpty {
81 "*"
82 }
83 }
84}
85
86private fun constraint(text: String): Constraint {
87 fun invalid(): Nothing = fail("invalid constraint '$text'")
88 val alternatives = mutableListOf<Alternative>()
89 for (alternative in text.split(" || ")) {
90 val comparisons = mutableListOf<Comparator>()
91 val prereleaseCores = mutableSetOf<List<Int>>()
92 for (term in alternative.split(' ')) {
93 val op = listOf(">=","<=",">","<","=","^","~").firstOrNull {
94 term.startsWith(it)
95 }?: ""
96 val pattern = term.drop(op.length)
97 var k: Int
98 val lower: Version
99 if ('-' in pattern) {
100 lower = version(pattern) ?: invalid()
101 if (lower.pre.isEmpty()) invalid()
102 k = 3
103 prereleaseCores.add(lower.core)
104 } else {
105 val parts = pattern.split('.')
106 if (parts.size !in 1..3) invalid()
107 val core = mutableListOf<Int>()
108 var wildcard = false
109 for (part in parts) {
110 if (part == "x" || part == "*") wildcard = true else {
111 val n = number(part) ?: invalid()
112 if (wildcard) invalid()
113 core.add(n)
114 }
115 }
116 k = core.size
117 while (core.size < 3) core.add(0)
118 lower = Version(core)
119 }
120 fun upper(i: Int): Version {
121 val core = lower.core.toMutableList()
122 core[i-1]++
123 for (j in i..2) core[j] = 0
124 return Version(core)
125 }
126 fun add(op: String,v: Version) {
127 comparisons.add(Comparator(op,v))
128 }
129 if (k == 0) {
130 if (op == ">" || op == "<") add("<",Version(listOf(0,0,0),listOf("0")))
131 } else when(op) {
132 "","=" -> if (k == 3) add("=",lower) else {
133 add(">=",lower)
134 add("<",upper(k))
135 }
136 ">" -> add(if (k == 3) ">" else ">=",if (k == 3) lower else upper(k))
137 "<=" -> add(if (k == 3) "<=" else "<",if (k == 3) lower else upper(k))
138 ">=","<" -> add(op,lower)
139 "~" -> {
140 add(">=",lower)
141 add("<",upper(if (k >= 2) 2 else 1))
142 }
143 "^" -> {
144 add(">=",lower)
145 add("<",upper(if (lower.core[0] != 0 || k == 1) 1 else if (lower.core[1] != 0 || k == 2) 2 else 3))
146 }
147 }
148 }
149 alternatives.add(Alternative(comparisons,prereleaseCores))
150 }
151 return Constraint(alternatives)
152}

162private class Manager {
163 val registry = sortedMapOf<String,MutableList<Release>>()
164 var requests = emptyList<Rule>()
165 var installed = sortedMapOf<String,Release>()
166 val held = mutableSetOf<String>()
167 val advisories = sortedMapOf<String,Rule>()
168 val history = mutableListOf<Snapshot>()
169 fun releases(n: String) = registry[n]?.sortedByDescending {
170 it.version
171 }?: emptyList()
172 fun registered(n: String,v: String) = registry[n]?.firstOrNull {
173 it.version.text == v
174 }?: fail("$n@$v is not registered")
175 fun checkInstalled(n: String) = installed[n] ?: fail("$n is not installed")
176 fun applicable(n: String,v: Version) = advisories.filterValues {
177 it.name == n && it.constraint.accepts(v)
178 }.keys.toList()
179 fun display(n: String,r: Release): String {
180 val flags = mutableListOf<String>()
181 if (r.yanked) flags.add("yanked")
182 if (installed[n]?.version == r.version) flags.add("installed")
183 return "$n@${r.version.text}"+if (flags.isEmpty()) "" else flags.joinToString(", ",prefix=" (",postfix=")")
184 }
185 fun resolve(newRequests: List<Rule>,preferred: Set<String>,audit: Boolean): Pair<Map<String,Release>?,List<String>> {
186 val selected = sortedMapOf<String,Release>()
187 val origins = mutableMapOf<String,Requirement>()
188 var failureDepth = -1
189 var failure = emptyList<String>()
190 fun chain(r: Requirement): List<String> {
191 val origin = r.origin
192 return if (origin == null) listOf("requested ${r.rule.name} ${r.rule.text}") else chain(origins.getValue(origin.first))+"${origin.first}@${origin.second.version.text} requires ${r.rule.name} ${r.rule.text}"
193 }
194 fun record(r: Requirement,reason: String,other: String? = null) {
195 if (selected.size > failureDepth) {
196 failureDepth = selected.size
197 failure = listOf("UNSATISFIABLE")+chain(r).map {
198 " $it"
199 }+" $reason"+(other?.let {
200 chain(origins.getValue(it)).map {
201 line -> " $line"
202 }
203 }?: emptyList())
204 }
205 }
206 fun usable(n: String,r: Release) = (!r.yanked || installed[n]?.version == r.version) && (!audit || applicable(n,r.version).isEmpty())
207 fun search(queue: List<Requirement>): Boolean {
208 if (queue.isEmpty()) return true
209 val r = queue.first()
210 val n = r.rule.name
211 val rest = queue.drop(1)
212 val old = selected[n]
213 if (old != null) {
214 if (r.rule.constraint.accepts(old.version)) return search(rest)
215 record(r,"$n@${old.version.text} was selected by:",n)
216 return false
217 }
218 val candidates: MutableList<Release>
219 if (n in held) {
220 val fixed = installed.getValue(n)
221 if (!r.rule.constraint.accepts(fixed.version) || !usable(n,fixed)) {
222 record(r,"$n@${fixed.version.text} is held")
223 return false
224 }
225 candidates = mutableListOf(fixed)
226 } else {
227 candidates = releases(n).filter {
228 r.rule.constraint.accepts(it.version) && usable(n,it)
229 }.toMutableList()
230 if (n in preferred) {
231 val index = candidates.indexOfFirst {
232 it.version == installed[n]?.version
233 }
234 if (index >= 0) candidates.add(0,candidates.removeAt(index))
235 }
236 if (candidates.isEmpty()) {
237 record(r,"no version of $n matches ${r.rule.text}")
238 return false
239 }
240 }
241 for (candidate in candidates) {
242 val conflict = selected.entries.firstOrNull {
243 (q,w) -> candidate.conflicts.any {
244 it.name == q && it.constraint.accepts(w.version)
245 }
246 || w.conflicts.any {
247 it.name == n && it.constraint.accepts(candidate.version)
248 }
249 }
250 if (conflict != null) {
251 record(r,"$n@${candidate.version.text} conflicts with ${conflict.key}@${conflict.value.version.text}, which was selected by:",conflict.key)
252 continue
253 }
254 selected[n] = candidate
255 origins[n] = r
256 if (search(candidate.deps.map {
257 Requirement(it,n to candidate)
258 }+rest)) return true
259 selected.remove(n)
260 origins.remove(n)
261 }
262 return false
263 }
264 return if (search(newRequests.map {
265 Requirement(it)
266 })) selected.toSortedMap() to emptyList() else null to failure
267 }
268 fun plan(next: Map<String,Release>): List<String> {
269 val out = mutableListOf<String>()
270 val counts = IntArray(4)
271 val remaining = next.keys.toSortedSet()
272 while (remaining.isNotEmpty()) {
273 val n = remaining.first {
274 next.getValue(it).deps.none {
275 d -> d.name in remaining
276 }
277 }
278 remaining.remove(n)
279 val v = next.getValue(n).version
280 val old = installed[n]?.version
281 if (old == null) {
282 out.add("install $n@${v.text}")
283 counts[0]++
284 } else if (v != old) {
285 val up = v > old
286 out.add("${if (up) "upgrade" else "downgrade"} $n ${old.text} -> ${v.text}")
287 counts[if (up) 1 else 2]++
288 }
289 }
290 remaining.addAll(installed.keys-next.keys)
291 while (remaining.isNotEmpty()) {
292 val n = remaining.first {
293 name -> remaining.none {
294 q -> installed.getValue(q).deps.any {
295 it.name == name
296 }
297 }
298 }
299 remaining.remove(n)
300 out.add("remove $n@${installed.getValue(n).version.text}")
301 counts[3]++
302 }
303 if (out.isEmpty()) out.add("no changes") else out.add("${counts[0]} installed, ${counts[1]} upgraded, ${counts[2]} downgraded, ${counts[3]} removed")
304 return out
305 }
306 fun update(newRequests: List<Rule>,next: Map<String,Release>) {
307 history.add(Snapshot(requests,installed.toMap()))
308 requests = newRequests
309 installed = next.toSortedMap()
310 held.retainAll(installed.keys)
311 }
312 fun parent(n: String) = installed.entries.first {
313 (_,r) -> r.deps.any {
314 it.name == n
315 }
316 }.key
317 fun command(line: String): List<String> {
318 val clauses = line.split("; ")
319 val w = clauses[0].split(' ')
320 val c = w[0]
321 val a = w.drop(1)
322 val out = mutableListOf<String>()
323 fun rest(i: Int) = a.drop(i).joinToString(" ")
324 when(c) {
325 "PKG" -> {
326 val v = checkedVersion(a[1])
327 if (registry[a[0]]?.any {
328 it.version == v
329 }
330 == true) fail("${a[0]}@${a[1]} is already registered")
331 val deps = mutableListOf<Rule>()
332 val conflicts = mutableListOf<Rule>()
333 for (clause in clauses.drop(1)) {
334 val words = clause.split(' ')
335 val text = words.drop(2).joinToString(" ")
336 val rule = Rule(words[1],text,constraint(text))
337 if (words[0] == "DEP") deps.add(rule) else conflicts.add(rule)
338 }
339 registry.getOrPut(a[0]) {
340 mutableListOf()
341 }.add(Release(v,deps,conflicts))
342 return listOf("registered ${a[0]}@${a[1]}")
343 }
344 "YANK","INFO" -> {
345 val r = registered(a[0],a[1])
346 if (c == "YANK") {
347 if (r.yanked) fail("${a[0]}@${a[1]} is already yanked")
348 r.yanked = true
349 return listOf("yanked ${a[0]}@${a[1]}")
350 }
351 out.add(display(a[0],r))
352 out.addAll(r.deps.map {
353 " requires ${it.name} ${it.text}"
354 })
355 out.addAll(r.conflicts.map {
356 " conflicts with ${it.name} ${it.text}"
357 })
358 out.addAll(applicable(a[0],r.version).map {
359 " advisory $it"
360 })
361 return out
362 }
363 "VERSIONS" -> {
364 val con = if (a.size > 1) constraint(rest(1)) else null
365 if (a[0] !in registry) fail("unknown package ${a[0]}")
366 out.addAll(releases(a[0]).filter {
367 con == null || con.accepts(it.version)
368 }.map {
369 display(a[0],it)
370 })
371 return out.ifEmpty {
372 listOf("no matching versions")
373 }
374 }
375 "DEPENDENTS" -> {
376 val v = if (a.size > 1) checkedVersion(a[1]) else null
377 for (n in registry.keys) for (r in releases(n)) for (d in r.deps) if (d.name == a[0] && (v == null || d.constraint.accepts(v))) out.add("$n@${r.version.text} requires ${d.name} ${d.text}")
378 return out.ifEmpty {
379 listOf("no dependents")
380 }
381 }
382 "COMPARE" -> {
383 val x = checkedVersion(a[0])
384 val y = checkedVersion(a[1])
385 return listOf("${a[0]} ${if (x < y) "<" else if (x > y) ">" else "="} ${a[1]}")
386 }
387 "MATCH" -> {
388 val v = checkedVersion(a[0])
389 val con = constraint(rest(1))
390 return listOf(if (con.accepts(v)) "yes" else "no")
391 }
392 "RANGE" -> return listOf(constraint(rest(0)).normal())
393 "INSTALL","SIMULATE" -> {
394 val text = if (a.size > 1) rest(1) else "*"
395 val con = constraint(text)
396 if (a[0] !in registry) fail("unknown package ${a[0]}")
397 val rule = Rule(a[0],text,con)
398 val nextRequests = requests.toMutableList()
399 val i = nextRequests.indexOfFirst {
400 it.name == a[0]
401 }
402 if (i < 0) nextRequests.add(rule) else nextRequests[i] = rule
403 val (next,failure) = resolve(nextRequests,installed.keys,false)
404 if (next == null) return failure
405 val plan = plan(next)
406 if (c == "INSTALL") update(nextRequests,next)
407 return plan
408 }
409 "REMOVE" -> {
410 val n = a[0]
411 if (requests.none {
412 it.name == n
413 }) {
414 if (n in installed) fail("$n is required by ${parent(n)}") else fail("$n is not installed")
415 }
416 val nextRequests = requests.filter {
417 it.name != n
418 }
419 val keep = nextRequests.map {
420 it.name
421 }.toMutableSet()
422 while (true) {
423 val next = keep.flatMap {
424 installed.getValue(it).deps.map {
425 d -> d.name
426 }
427 }
428 if (!keep.addAll(next)) break
429 }
430 if (n in keep) {
431 out.add("$n@${installed.getValue(n).version.text} is still required by ${parent(n)}")
432 update(nextRequests,installed)
433 } else {
434 val next = installed.filterKeys {
435 it in keep
436 }
437 out.addAll(plan(next))
438 update(nextRequests,next)
439 }
440 return out
441 }
442 "UPGRADE" -> {
443 if (a.isNotEmpty()) checkInstalled(a[0])
444 val preferred = if (a.isEmpty()) emptySet() else installed.keys-a[0]
445 val (next,failure) = resolve(requests,preferred,false)
446 if (next == null) return failure
447 out.addAll(plan(next))
448 update(requests,next)
449 return out
450 }
451 "ROLLBACK" -> {
452 if (history.isEmpty()) fail("nothing to roll back")
453 val snapshot = history.removeAt(history.lastIndex)
454 out.addAll(plan(snapshot.installed))
455 requests = snapshot.requests
456 installed = snapshot.installed.toSortedMap()
457 held.retainAll(installed.keys)
458 return out
459 }
460 "HOLD" -> {
461 val r = checkInstalled(a[0])
462 if (!held.add(a[0])) fail("${a[0]} is already held")
463 return listOf("held ${a[0]}@${r.version.text}")
464 }
465 "UNHOLD" -> {
466 if (!held.remove(a[0])) fail("${a[0]} is not held")
467 return listOf("released ${a[0]}")
468 }
469 "WHY" -> {
470 val n = a[0]
471 val r = checkInstalled(n)
472 val roots = requests.map {
473 it.name
474 }.toSortedSet()
475 if (n in roots) return listOf("$n@${r.version.text} is explicitly installed")
476 val seen = roots.toMutableSet()
477 val queue = ArrayDeque<String>()
478 queue.addAll(roots)
479 val parents = mutableMapOf<String,String>()
480 while (queue.isNotEmpty()) {
481 val u = queue.removeFirst()
482 for (v in installed.getValue(u).deps.map {
483 it.name
484 }.sorted()) if (seen.add(v)) {
485 parents[v] = u
486 queue.addLast(v)
487 }
488 }
489 val path = mutableListOf(n)
490 while (path.last() !in roots) path.add(parents.getValue(path.last()))
491 return listOf(path.asReversed().joinToString(" -> ") {
492 "$it@${installed.getValue(it).version.text}"
493 })
494 }
495 "TREE" -> {
496 if (a.isNotEmpty()) checkInstalled(a[0])
497 val roots = if (a.isEmpty()) requests.map {
498 it.name
499 }.sorted() else listOf(a[0])
500 if (roots.isEmpty()) return listOf("(empty)")
501 val seen = mutableSetOf<String>()
502 fun visit(n: String,text: String?,prefix: String,indent: String) {
503 val r = installed.getValue(n)
504 val body = prefix+"$n@${r.version.text}"+(text?.let {
505 " [$it]"
506 }?: "")
507 if (!seen.add(n)) {
508 out.add("$body (*)")
509 return
510 }
511 out.add(body)
512 for ((i,d) in r.deps.withIndex()) {
513 val last = i == r.deps.lastIndex
514 visit(d.name,d.text,indent+if (last) "`-- " else "|-- ",indent+if (last) " " else "| ")
515 }
516 }
517 for (n in roots) visit(n,requests.firstOrNull {
518 it.name == n
519 }?.text,"","")
520 return out
521 }
522 "OUTDATED" -> {
523 for ((n,r) in installed) {
524 val constraints = requests.filter {
525 it.name == n
526 }.map {
527 it.constraint
528 }+installed.values.flatMap {
529 it.deps
530 }.filter {
531 it.name == n
532 }.map {
533 it.constraint
534 }
535 val versions = releases(n)
536 val wanted = versions.first {
537 (!it.yanked || it.version == r.version) && constraints.all {
538 con -> con.accepts(it.version)
539 }
540 }
541 val latest = versions.firstOrNull {
542 !it.yanked && it.version.pre.isEmpty()
543 }
544 if (wanted.version != r.version || (latest != null && latest.version > r.version)) out.add("$n ${r.version.text} ${wanted.version.text} ${latest?.version?.text ?: "-"}")
545 }
546 return out.ifEmpty {
547 listOf("all packages are up to date")
548 }
549 }
550 "LOCK" -> return installed.map {
551 (n,r) -> "$n@${r.version.text}"
552 }.ifEmpty {
553 listOf("(empty)")
554 }
555 "STATUS" -> {
556 out.addAll(requests.map {
557 "${it.name} ${it.text} -> ${installed.getValue(it.name).version.text}"
558 }.ifEmpty {
559 listOf("no requests")
560 })
561 out.add("${installed.size} installed, ${held.size} held, ${history.size} snapshots")
562 return out
563 }
564 "ADVISORY" -> {
565 if (a[0] in advisories) fail("advisory ${a[0]} already exists")
566 val text = rest(2)
567 val con = constraint(text)
568 advisories[a[0]] = Rule(a[1],text,con)
569 return listOf("advisory ${a[0]} recorded")
570 }
571 "AUDIT" -> {
572 if (a.isNotEmpty()) {
573 val (next,failure) = resolve(requests,installed.keys,true)
574 if (next == null) return failure
575 out.addAll(plan(next))
576 update(requests,next)
577 return out
578 }
579 for ((n,r) in installed) {
580 val ids = applicable(n,r.version)
581 if (ids.isEmpty()) continue
582 val fix = releases(n).asReversed().firstOrNull {
583 it.version > r.version && it.version.pre.isEmpty() && !it.yanked && applicable(n,it.version).isEmpty()
584 }
585 out.add("$n@${r.version.text}: ${ids.joinToString(", ")} "+if (fix == null) "(no fix)" else "(fix: ${fix.version.text})")
586 }
587 return out.ifEmpty {
588 listOf("no known vulnerabilities")
589 }
590 }
591 }
592 return out
593 }
594}
