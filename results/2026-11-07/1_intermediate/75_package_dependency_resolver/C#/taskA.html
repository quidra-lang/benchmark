8class VersionValue(int[] core, List<string> prerelease) : IComparable<VersionValue>
9{
10 public int[] Core = core;
11 public List<string> Prerelease = prerelease;
12 public bool Stable => Prerelease.Count == 0;
13 public string CoreText => string.Join(".", Core);
14 public string Text => CoreText + (Stable ? "" : "-" + string.Join(".", Prerelease));
15
16 public int CompareTo(VersionValue? other)
17 {
18 if (other == null) return 1;
19 for (int i = 0; i < 3; i++) if (Core[i] != other.Core[i]) return Core[i].CompareTo(other.Core[i]);
20 if (Stable || other.Stable) return Stable.CompareTo(other.Stable);
21 for (int i = 0; i < Math.Min(Prerelease.Count, other.Prerelease.Count); i++)
22 {
23 string a = Prerelease[i], b = other.Prerelease[i];
24 bool an = Syntax.Digits(a), bn = Syntax.Digits(b);
25 int order = an && bn ? int.Parse(a).CompareTo(int.Parse(b)) : an != bn ? an ? -1 : 1 : string.CompareOrdinal(a, b);
26 if (order != 0) return order;
27 }
28 return Prerelease.Count.CompareTo(other.Prerelease.Count);
29 }
30}

32record ComparatorTerm(string Operator, VersionValue Version)
33{
34 public bool Matches(VersionValue value)
35 {
36 int c = value.CompareTo(Version);
37 return Operator switch { "=" => c == 0, ">" => c > 0, ">=" => c >= 0, "<" => c < 0, _ => c <= 0 };
38 }
39 public override string ToString() => Operator + Version.Text;
40}
41
42class Alternative
43{
44 public List<ComparatorTerm> Comparators = [];
45 public HashSet<string> PrereleaseCores = new(StringComparer.Ordinal);
46 public void Add(string op, VersionValue version) => Comparators.Add(new(op, version));
47 public bool Matches(VersionValue v) => (v.Stable || PrereleaseCores.Contains(v.CoreText)) && Comparators.All(c => c.Matches(v));
48 public string Normal() => Comparators.Count == 0 ? "*" : string.Join(" ", Comparators);
49}
50
51class Constraint
52{
53 public List<Alternative> Alternatives = [];
54 public bool Matches(VersionValue v) => Alternatives.Any(a => a.Matches(v));
55 public string Normal() => string.Join(" || ", Alternatives.Select(a => a.Normal()));
56}

58static class Syntax
59{
60 public static bool Digits(string text) => text.Length != 0 && text.All(c => c >= '0' && c <= '9');
61 public static bool Number(string text) => text.Length is >= 1 and <= 4 && Digits(text) && (text.Length == 1 || text[0] != '0');
62
63 public static VersionValue? Version(string text)
64 {
65 string[] halves = text.Split('-'), parts = halves[0].Split('.');
66 if (halves.Length > 2 || parts.Length != 3 || parts.Any(p => !Number(p))) return null;
67 List<string> pre = [];
68 if (halves.Length == 2) foreach (string p in halves[1].Split('.'))
69 {
70 if (p.Length == 0 || p.Any(c => !(c >= 'a' && c <= 'z' || c >= '0' && c <= '9')) || (Digits(p) && !Number(p))) return null;
71 pre.Add(p);
72 }
73 return new(parts.Select(int.Parse).ToArray(), pre);
74 }
75 public static VersionValue CheckedVersion(string text) => Version(text) ?? throw new Problem($"invalid version '{text}'");
76
77 public static Constraint? ParseConstraint(string text)
78 {
79 Constraint result = new();
80 foreach (string branch in text.Split(" || ", StringSplitOptions.None))
81 {
82 Alternative a = new();
83 foreach (string term in branch.Split(' '))
84 {
85 if (term.Length == 0) return null;
86 string op = "", pattern = term;
87 foreach (string candidate in new[] { ">=", "<=", ">", "<", "=", "^", "~" }) if (term.StartsWith(candidate, StringComparison.Ordinal))
88 {
89 op = candidate;
90 pattern = term[candidate.Length..];
91 break;
92 }
93 VersionValue? lower = Version(pattern);
94 int count;
95 if (lower != null && !lower.Stable)
96 {
97 count = 3;
98 a.PrereleaseCores.Add(lower.CoreText);
99 }
100 else
101 {
102 string[] parts = pattern.Split('.');
103 if (parts.Length < 1 || parts.Length > 3) return null;
104 int[] core = new int[3];
105 count = 0;
106 bool wildcard = false;
107 for (int i = 0; i < parts.Length; i++)
108 {
109 if (parts[i] is "x" or "*") wildcard = true;
110 else
111 {
112 if (wildcard || !Number(parts[i])) return null;
113 core[i] = int.Parse(parts[i]);
114 count++;
115 }
116 }
117 lower = new(core, []);
118 }
119 if (count == 0)
120 {
121 if (op is ">" or "<") a.Add("<", new([0, 0, 0], ["0"]));
122 continue;
123 }
124 VersionValue[] upper = new VersionValue[3];
125 for (int i = 0; i < 3; i++)
126 {
127 int[] core = (int[])lower.Core.Clone();
128 core[i]++;
129 for (int j = i + 1; j < 3; j++) core[j] = 0;
130 upper[i] = new(core, []);
131 }
132 switch (op)
133 {
134 case "":
135 case "=":
136 if (count == 3) a.Add("=", lower);
137 else
138 {
139 a.Add(">=", lower);
140 a.Add("<", upper[count - 1]);
141 }
142 break;
143 case ">":
144 a.Add(count == 3 ? ">" : ">=", count == 3 ? lower : upper[count - 1]);
145 break;
146 case "<=":
147 a.Add(count == 3 ? "<=" : "<", count == 3 ? lower : upper[count - 1]);
148 break;
149 case ">=":
150 case "<":
151 a.Add(op, lower);
152 break;
153 case "~":
154 a.Add(">=", lower);
155 a.Add("<", upper[count == 1 ? 0 : 1]);
156 break;
157 case "^":
158 int k = lower.Core[0] != 0 || count == 1 ? 0 : lower.Core[1] != 0 || count == 2 ? 1 : 2;
159 a.Add(">=", lower);
160 a.Add("<", upper[k]);
161 break;
162 }
163 }
164 result.Alternatives.Add(a);
165 }
166 return result;
167 }
168 public static Constraint CheckedConstraint(string text) => ParseConstraint(text) ?? throw new Problem($"invalid constraint '{text}'");
169}

196class Program
197{
198 readonly SortedDictionary<string, List<PackageVersion>> registry = new(StringComparer.Ordinal);
199 List<Clause> requests = [];
200 SortedDictionary<string, PackageVersion> installed = new(StringComparer.Ordinal);
201 readonly HashSet<string> held = new(StringComparer.Ordinal);
202 readonly SortedDictionary<string, Clause> advisories = new(StringComparer.Ordinal);
203 readonly Stack<Snapshot> history = new();
204
205 static void Fail(string message) => throw new Problem(message);
206
207 static Clause NewClause(string name, string text) => new(name, text, Syntax.CheckedConstraint(text));
208 List<PackageVersion> Versions(string name) => registry.GetValueOrDefault(name) ?? [];
209 PackageVersion? Registered(string name, string version) => Versions(name).FirstOrDefault(p => p.Version.Text == version);
210 PackageVersion CheckedPackage(string name, string version) => Registered(name, version) ?? throw new Problem($"{name}@{version} is not registered");
211 bool Vulnerable(PackageVersion p) => advisories.Values.Any(a => a.Name == p.Name && a.Constraint.Matches(p.Version));
212
213 string Display(PackageVersion p)
214 {
215 bool current = installed.GetValueOrDefault(p.Name) == p;
216 return p + (p.Yanked ? current ? " (yanked, installed)" : " (yanked)" : current ? " (installed)" : "");
217 }

219 class Resolution(Program manager, HashSet<string> preferred, bool audit)
220 {
221 public SortedDictionary<string, PackageVersion> Selected = new(StringComparer.Ordinal);
222 readonly Dictionary<string, Requirement> reasons = new(StringComparer.Ordinal);
223 int failureDepth = -1;
224 public List<string> Failure = [];
225 bool Usable(PackageVersion p) => (!p.Yanked || manager.installed.GetValueOrDefault(p.Name) == p) && (!audit || !manager.Vulnerable(p));
226 void Record(Requirement r, string reason, Requirement? other)
227 {
228 if (Selected.Count <= failureDepth) return;
229 failureDepth = Selected.Count;
230 Failure = ["UNSATISFIABLE", .. r.Chain().Select(line => " " + line), " " + reason];
231 if (other != null) Failure.AddRange(other.Chain().Select(line => " " + line));
232 }
233 bool Search(List<Requirement> pending)
234 {
235 if (pending.Count == 0) return true;
236 Requirement r = pending[0];
237 Clause clause = r.Clause;
238 string name = clause.Name;
239 List<Requirement> rest = pending.Skip(1).ToList();
240 if (Selected.TryGetValue(name, out PackageVersion? already))
241 {
242 if (clause.Constraint.Matches(already.Version)) return Search(rest);
243 Record(r, $"{already} was selected by:", reasons[name]);
244 return false;
245 }
246 List<PackageVersion> candidates = [];
247 if (manager.held.Contains(name))
248 {
249 PackageVersion p = manager.installed[name];
250 if (!clause.Constraint.Matches(p.Version) || !Usable(p))
251 {
252 Record(r, $"{p} is held", null);
253 return false;
254 }
255 candidates.Add(p);
256 }
257 else
258 {
259 candidates = manager.Versions(name).Where(p => clause.Constraint.Matches(p.Version) && Usable(p)).ToList();
260 PackageVersion? current = manager.installed.GetValueOrDefault(name);
261 if (preferred.Contains(name) && current != null && candidates.Remove(current)) candidates.Insert(0, current);
262 if (candidates.Count == 0)
263 {
264 Record(r, $"no version of {name} matches {clause.Text}", null);
265 return false;
266 }
267 }
268 foreach (PackageVersion p in candidates)
269 {
270 PackageVersion? conflict = Selected.Values.FirstOrDefault(q => p.Conflicts.Any(d => d.Name == q.Name && d.Constraint.Matches(q.Version)) || q.Conflicts.Any(d => d.Name == name && d.Constraint.Matches(p.Version)));
271 if (conflict != null)
272 {
273 Record(r, $"{p} conflicts with {conflict}, which was selected by:", reasons[conflict.Name]);
274 continue;
275 }
276 Selected[name] = p;
277 reasons[name] = r;
278 List<Requirement> next = p.Dependencies.Select(d => new Requirement(d, p, r)).ToList();
279 next.AddRange(rest);
280 if (Search(next)) return true;
281 Selected.Remove(name);
282 reasons.Remove(name);
283 }
284 return false;
285 }
286 public bool Solve(List<Clause> list) => Search(list.Select(c => new Requirement(c, null, null)).ToList());
287 }

289 List<string> Plan(SortedDictionary<string, PackageVersion> next)
290 {
291 List<string> output = [];
292 int[] counts = new int[4];
293 HashSet<string> arranged = new(StringComparer.Ordinal);
294 while (arranged.Count < next.Count)
295 {
296 string name = next.Keys.First(n => !arranged.Contains(n) && next[n].Dependencies.All(d => arranged.Contains(d.Name)));
297 arranged.Add(name);
298 PackageVersion p = next[name];
299 if (!installed.TryGetValue(name, out PackageVersion? old))
300 {
301 output.Add("install " + p);
302 counts[0]++;
303 }
304 else if (old != p)
305 {
306 bool newer = p.Version.CompareTo(old.Version) > 0;
307 output.Add((newer ? "upgrade " : "downgrade ") + $"{name} {old.Version.Text} -> {p.Version.Text}");
308 counts[newer ? 1 : 2]++;
309 }
310 }
311 SortedSet<string> removed = new(installed.Keys.Except(next.Keys), StringComparer.Ordinal);
312 while (removed.Count != 0)
313 {
314 string name = removed.First(n => removed.All(q => installed[q].Dependencies.All(d => d.Name != n)));
315 output.Add("remove " + installed[name]);
316 counts[3]++;
317 removed.Remove(name);
318 }
319 if (output.Count == 0) output.Add("no changes");
320 else output.Add($"{counts[0]} installed, {counts[1]} upgraded, {counts[2]} downgraded, {counts[3]} removed");
321 return output;
322 }

324 void Update(List<Clause> list, SortedDictionary<string, PackageVersion> next)
325 {
326 history.Push(new([.. requests], new(installed, StringComparer.Ordinal)));
327 requests = [.. list];
328 installed = new(next, StringComparer.Ordinal);
329 held.IntersectWith(installed.Keys);
330 }

332 List<string> Resolve(List<Clause> list, HashSet<string> preferred, bool audit, bool change)
333 {
334 Resolution r = new(this, preferred, audit);
335 if (!r.Solve(list)) return r.Failure;
336 List<string> result = Plan(r.Selected);
337 if (change) Update(list, r.Selected);
338 return result;
339 }
340 Clause? Request(string name) => requests.FirstOrDefault(r => r.Name == name);
341 string RequiredBy(string name) => installed.Keys.First(n => installed[n].Dependencies.Any(d => d.Name == name));
342
343 HashSet<string> Reachable(List<Clause> list)
344 {
345 HashSet<string> seen = new(StringComparer.Ordinal);
346 Queue<string> queue = new();
347 foreach (Clause c in list) if (seen.Add(c.Name)) queue.Enqueue(c.Name);
348 while (queue.Count != 0) foreach (Clause d in installed[queue.Dequeue()].Dependencies) if (seen.Add(d.Name)) queue.Enqueue(d.Name);
349 return seen;
350 }
351
352 void Tree(string name, string? constraint, string prefix, string indent, HashSet<string> seen, List<string> output)
353 {
354 PackageVersion p = installed[name];
355 string body = prefix + p + (constraint == null ? "" : $" [{constraint}]");
356 if (!seen.Add(name))
357 {
358 output.Add(body + " (*)");
359 return;
360 }
361 output.Add(body);
362 for (int i = 0; i < p.Dependencies.Count; i++)
363 {
364 Clause d = p.Dependencies[i];
365 bool last = i + 1 == p.Dependencies.Count;
366 Tree(d.Name, d.Text, indent + (last ? "`-- " : "|-- "), indent + (last ? " " : "| "), seen, output);
367 }
368 }

370 List<string> Command(string line)
371 {
372 int space = line.IndexOf(' ');
373 string c = space < 0 ? line : line[..space], rest = space < 0 ? "" : line[(space + 1)..];
374 string[] w = rest.Split(' ', 3);
375 List<string> output = [];
376 switch (c)
377 {
378 case "PKG":
379 {
380 string[] clauses = rest.Split("; ", StringSplitOptions.None), header = clauses[0].Split(' ');
381 VersionValue v = Syntax.CheckedVersion(header[1]);
382 if (Registered(header[0], header[1]) != null) Fail($"{header[0]}@{header[1]} is already registered");
383 PackageVersion p = new(header[0], v);
384 foreach (string text in clauses.Skip(1))
385 {
386 string[] a = text.Split(' ', 3);
387 Clause d = NewClause(a[1], a[2]);
388 (a[0] == "DEP" ? p.Dependencies : p.Conflicts).Add(d);
389 }
390 if (!registry.TryGetValue(p.Name, out List<PackageVersion>? versions)) registry[p.Name] = versions = [];
391 versions.Add(p);
392 versions.Sort((a, b) => b.Version.CompareTo(a.Version));
393 output.Add("registered " + p);
394 break;
395 }
396 case "YANK":
397 case "INFO":
398 {
399 PackageVersion p = CheckedPackage(w[0], w[1]);
400 if (c == "YANK")
401 {
402 if (p.Yanked) Fail($"{p} is already yanked");
403 p.Yanked = true;
404 output.Add("yanked " + p);
405 }
406 else
407 {
408 output.Add(Display(p));
409 foreach (Clause d in p.Dependencies) output.Add($" requires {d.Name} {d.Text}");
410 foreach (Clause d in p.Conflicts) output.Add($" conflicts with {d.Name} {d.Text}");
411 foreach ((string id, Clause a) in advisories) if (a.Name == p.Name && a.Constraint.Matches(p.Version)) output.Add(" advisory " + id);
412 }
413 break;
414 }
415 case "VERSIONS":
416 case "INSTALL":
417 case "SIMULATE":
418 {
419 string name = w[0];
420 string? text = rest.Length == name.Length ? null : rest[(name.Length + 1)..];
421 Constraint? constraint = text == null ? null : Syntax.CheckedConstraint(text);
422 if (!registry.ContainsKey(name)) Fail("unknown package " + name);
423 if (c == "VERSIONS")
424 {
425 foreach (PackageVersion p in Versions(name)) if (constraint == null || constraint.Matches(p.Version)) output.Add(Display(p));
426 if (output.Count == 0) output.Add("no matching versions");
427 }
428 else
429 {
430 Clause requirement = text == null ? NewClause(name, "*") : new(name, text, constraint!);
431 List<Clause> list = [.. requests];
432 Clause? old = Request(name);
433 if (old == null) list.Add(requirement);
434 else list[list.IndexOf(old)] = requirement;
435 output = Resolve(list, new(installed.Keys, StringComparer.Ordinal), false, c == "INSTALL");
436 }
437 break;
438 }
439 case "DEPENDENTS":
440 {
441 VersionValue? v = w.Length > 1 ? Syntax.CheckedVersion(w[1]) : null;
442 foreach (List<PackageVersion> versions in registry.Values) foreach (PackageVersion p in versions) foreach (Clause d in p.Dependencies) if (d.Name == w[0] && (v == null || d.Constraint.Matches(v))) output.Add($"{p} requires {d.Name} {d.Text}");
443 if (output.Count == 0) output.Add("no dependents");
444 break;
445 }
446 case "COMPARE":
447 {
448 VersionValue a = Syntax.CheckedVersion(w[0]), b = Syntax.CheckedVersion(w[1]);
449 output.Add(w[0] + (a.CompareTo(b) < 0 ? " < " : a.CompareTo(b) > 0 ? " > " : " = ") + w[1]);
450 break;
451 }
452 case "MATCH":
453 {
454 VersionValue v = Syntax.CheckedVersion(w[0]);
455 Constraint constraint = Syntax.CheckedConstraint(rest[(w[0].Length + 1)..]);
456 output.Add(constraint.Matches(v) ? "yes" : "no");
457 break;
458 }
459 case "RANGE":
460 output.Add(Syntax.CheckedConstraint(rest).Normal());
461 break;
462 case "REMOVE":
463 {
464 Clause? old = Request(rest);
465 if (old == null) Fail(installed.ContainsKey(rest) ? $"{rest} is required by {RequiredBy(rest)}" : $"{rest} is not installed");
466 List<Clause> list = [.. requests];
467 list.Remove(old!);
468 HashSet<string> keep = Reachable(list);
469 if (keep.Contains(rest))
470 {
471 output.Add($"{installed[rest]} is still required by {RequiredBy(rest)}");
472 Update(list, installed);
473 }
474 else
475 {
476 SortedDictionary<string, PackageVersion> next = new(StringComparer.Ordinal);
477 foreach ((string name, PackageVersion p) in installed) if (keep.Contains(name)) next[name] = p;
478 output = Plan(next);
479 Update(list, next);
480 }
481 break;
482 }
483 case "UPGRADE":
484 {
485 if (rest.Length != 0 && !installed.ContainsKey(rest)) Fail(rest + " is not installed");
486 HashSet<string> preferred = new(StringComparer.Ordinal);
487 if (rest.Length != 0)
488 {
489 preferred.UnionWith(installed.Keys);
490 preferred.Remove(rest);
491 }
492 output = Resolve(requests, preferred, false, true);
493 break;
494 }
495 case "ROLLBACK":
496 {
497 if (history.Count == 0) Fail("nothing to roll back");
498 Snapshot snapshot = history.Pop();
499 output = Plan(snapshot.Installed);
500 requests = snapshot.Requests;
501 installed = snapshot.Installed;
502 held.IntersectWith(installed.Keys);
503 break;
504 }
505 case "HOLD":
506 if (!installed.ContainsKey(rest)) Fail(rest + " is not installed");
507 if (!held.Add(rest)) Fail(rest + " is already held");
508 output.Add("held " + installed[rest]);
509 break;
510 case "UNHOLD":
511 if (!held.Remove(rest)) Fail(rest + " is not held");
512 output.Add("released " + rest);
513 break;
514 case "WHY":
515 {
516 if (!installed.ContainsKey(rest)) Fail(rest + " is not installed");
517 if (Request(rest) != null) output.Add(installed[rest] + " is explicitly installed");
518 else
519 {
520 List<string> roots = requests.Select(r => r.Name).OrderBy(n => n, StringComparer.Ordinal).ToList();
521 Dictionary<string, string?> parent = new(StringComparer.Ordinal);
522 foreach (string root in roots) parent[root] = null;
523 Queue<string> queue = new(roots);
524 while (queue.Count != 0)
525 {
526 string name = queue.Dequeue();
527 foreach (string child in installed[name].Dependencies.Select(d => d.Name).OrderBy(n => n, StringComparer.Ordinal)) if (!parent.ContainsKey(child))
528 {
529 parent[child] = name;
530 queue.Enqueue(child);
531 }
532 }
533 List<string> path = [];
534 for (string? name = rest; name != null; name = parent[name]) path.Add(installed[name].ToString());
535 path.Reverse();
536 output.Add(string.Join(" -> ", path));
537 }
538 break;
539 }
540 case "TREE":
541 {
542 if (rest.Length != 0 && !installed.ContainsKey(rest)) Fail(rest + " is not installed");
543 HashSet<string> seen = new(StringComparer.Ordinal);
544 List<string> roots = rest.Length == 0 ? requests.Select(r => r.Name).OrderBy(n => n, StringComparer.Ordinal).ToList() : [rest];
545 if (roots.Count == 0) output.Add("(empty)");
546 foreach (string root in roots) Tree(root, Request(root)?.Text, "", "", seen, output);
547 break;
548 }
549 case "OUTDATED":
550 foreach (PackageVersion p in installed.Values)
551 {
552 List<Constraint> relevant = [];
553 Clause? r = Request(p.Name);
554 if (r != null) relevant.Add(r.Constraint);
555 foreach (PackageVersion q in installed.Values) foreach (Clause d in q.Dependencies) if (d.Name == p.Name) relevant.Add(d.Constraint);
556 PackageVersion wanted = Versions(p.Name).First(q => (!q.Yanked || q == p) && relevant.All(a => a.Matches(q.Version)));
557 PackageVersion? latest = Versions(p.Name).FirstOrDefault(q => !q.Yanked && q.Version.Stable);
558 if (wanted != p || latest != null && latest.Version.CompareTo(p.Version) > 0) output.Add($"{p.Name} {p.Version.Text} {wanted.Version.Text} {latest?.Version.Text ?? "-"}");
559 }
560 if (output.Count == 0) output.Add("all packages are up to date");
561 break;
562 case "LOCK": output.AddRange(installed.Values.Select(p => p.ToString()));
563 if (output.Count == 0) output.Add("(empty)");
564 break;
565 case "STATUS": foreach (Clause r in requests) output.Add($"{r.Name} {r.Text} -> {installed[r.Name].Version.Text}");
566 if (requests.Count == 0) output.Add("no requests");
567 output.Add($"{installed.Count} installed, {held.Count} held, {history.Count} snapshots");
568 break;
569 case "ADVISORY":
570 if (advisories.ContainsKey(w[0])) Fail($"advisory {w[0]} already exists");
571 advisories[w[0]] = NewClause(w[1], w[2]);
572 output.Add($"advisory {w[0]} recorded");
573 break;
574 case "AUDIT":
575 if (rest == "FIX") output = Resolve(requests, new(installed.Keys, StringComparer.Ordinal), true, true);
576 else
577 {
578 foreach (PackageVersion p in installed.Values)
579 {
580 List<string> ids = advisories.Where(e => e.Value.Name == p.Name && e.Value.Constraint.Matches(p.Version)).Select(e => e.Key).ToList();
581 if (ids.Count == 0) continue;
582 PackageVersion? fix = null;
583 foreach (PackageVersion q in Versions(p.Name)) if (q.Version.CompareTo(p.Version) > 0 && q.Version.Stable && !q.Yanked && !Vulnerable(q)) fix = q;
584 output.Add($"{p}: {string.Join(", ", ids)}" + (fix == null ? " (no fix)" : $" (fix: {fix.Version.Text})"));
585 }
586 if (output.Count == 0) output.Add("no known vulnerabilities");
587 }
588 break;
589 }
590 return output;
591 }
