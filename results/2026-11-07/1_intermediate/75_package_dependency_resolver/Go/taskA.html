57func keys[T any](items map[string]T) []string {
58 result := []string{}
59 for name := range items {
60 result = append(result, name)
61 }
62 sort.Strings(result)
63 return result
64}
65func numeric(text string) bool {
66 if len(text) == 0 {
67 return false
68 }
69 for i := range text {
70 if text[i] < '0' || text[i] > '9' {
71 return false
72 }
73 }
74 return true
75}
76func numberWord(text string) bool {
77 return len(text) >= 1 && len(text) <= 4 && numeric(text) && (len(text) == 1 || text[0] != '0')
78}
79func integer(text string) int { value, _ := strconv.Atoi(text); return value }
80func makeVersion(core [3]int, pre []string) *version {
81 text := fmt.Sprintf("%d.%d.%d", core[0], core[1], core[2])
82 if len(pre) > 0 {
83 text += "-" + strings.Join(pre, ".")
84 }
85 return &version{core, pre, text}
86}
87func parseVersion(text string) *version {
88 coreText, suffix, hasSuffix := strings.Cut(text, "-")
89 parts := strings.Split(coreText, ".")
90 if len(parts) != 3 {
91 return nil
92 }
93 core := [3]int{}
94 for i, part := range parts {
95 if !numberWord(part) {
96 return nil
97 }
98 core[i] = integer(part)
99 }
100 pre := []string{}
101 if hasSuffix {
102 pre = strings.Split(suffix, ".")
103 for _, part := range pre {
104 if len(part) == 0 {
105 return nil
106 }
107 for i := range part {
108 if !(part[i] >= 'a' && part[i] <= 'z' || part[i] >= '0' && part[i] <= '9') {
109 return nil
110 }
111 }
112 if numeric(part) && !numberWord(part) {
113 return nil
114 }
115 }
116 }
117 return makeVersion(core, pre)
118}
119func compare(a, b *version) int {
120 for i := 0; i < 3; i++ {
121 if a.core[i] < b.core[i] {
122 return -1
123 }
124 if a.core[i] > b.core[i] {
125 return 1
126 }
127 }
128 if len(a.pre) == 0 && len(b.pre) > 0 {
129 return 1
130 }
131 if len(b.pre) == 0 && len(a.pre) > 0 {
132 return -1
133 }
134 for i := 0; i < len(a.pre) && i < len(b.pre); i++ {
135 x, y := a.pre[i], b.pre[i]
136 if x == y {
137 continue
138 }
139 nx, ny := numeric(x), numeric(y)
140 if nx && ny {
141 if integer(x) < integer(y) {
142 return -1
143 }
144 return 1
145 }
146 if nx != ny {
147 if nx {
148 return -1
149 }
150 return 1
151 }
152 if x < y {
153 return -1
154 }
155 return 1
156 }
157 if len(a.pre) < len(b.pre) {
158 return -1
159 }
160 if len(a.pre) > len(b.pre) {
161 return 1
162 }
163 return 0
164}
165func parseConstraint(text string) *constraint {
166 result, forms := &constraint{}, []string{}
167 for _, branch := range strings.Split(text, " || ") {
168 item := alternative{prerelease: map[[3]int]bool{}}
169 for _, term := range strings.Split(branch, " ") {
170 if term == "" {
171 return nil
172 }
173 op, pattern := "", term
174 for _, candidate := range []string{">=", "<=", ">", "<", "=", "^", "~"} {
175 if strings.HasPrefix(term, candidate) {
176 op, pattern = candidate, term[len(candidate):]
177 break
178 }
179 }
180 core, pre, count := [3]int{}, []string{}, 0
181 if strings.Contains(pattern, "-") {
182 value := parseVersion(pattern)
183 if value == nil || len(value.pre) == 0 {
184 return nil
185 }
186 core, pre, count = value.core, value.pre, 3
187 item.prerelease[core] = true
188 } else {
189 parts, wildcard := strings.Split(pattern, "."), false
190 if len(parts) < 1 || len(parts) > 3 {
191 return nil
192 }
193 for i, part := range parts {
194 if part == "x" || part == "*" {
195 wildcard = true
196 } else {
197 if wildcard || !numberWord(part) {
198 return nil
199 }
200 core[i], count = integer(part), count+1
201 }
202 }
203 }
204 lower := makeVersion(core, pre)
205 upper := func(at int) *version {
206 next := core
207 next[at]++
208 for i := at + 1; i < 3; i++ {
209 next[i] = 0
210 }
211 return makeVersion(next, nil)
212 }
213 add := func(operator string, value *version) {
214 item.comparisons = append(item.comparisons, comparator{operator, value})
215 }
216 if count == 0 {
217 if op == ">" || op == "<" {
218 add("<", makeVersion([3]int{}, []string{"0"}))
219 }
220 } else {
221 switch op {
222 case "", "=":
223 if count == 3 {
224 add("=", lower)
225 } else {
226 add(">=", lower)
227 add("<", upper(count-1))
228 }
229 case ">":
230 if count == 3 {
231 add(">", lower)
232 } else {
233 add(">=", upper(count-1))
234 }
235 case "<=":
236 if count == 3 {
237 add("<=", lower)
238 } else {
239 add("<", upper(count-1))
240 }
241 case ">=", "<":
242 add(op, lower)
243 case "~":
244 at := 1
245 if count == 1 {
246 at = 0
247 }
248 add(">=", lower)
249 add("<", upper(at))
250 case "^":
251 at := 2
252 if core[0] != 0 || count == 1 {
253 at = 0
254 } else if core[1] != 0 || count == 2 {
255 at = 1
256 }
257 add(">=", lower)
258 add("<", upper(at))
259 }
260 }
261 }
262 words := []string{}
263 for _, comparison := range item.comparisons {
264 words = append(words, comparison.op+comparison.value.text)
265 }
266 if len(words) == 0 {
267 words = append(words, "*")
268 }
269 forms = append(forms, strings.Join(words, " "))
270 result.alternatives = append(result.alternatives, item)
271 }
272 result.normal = strings.Join(forms, " || ")
273 return result
274}
275func (c *constraint) matches(v *version) bool {
276 for _, branch := range c.alternatives {
277 if len(v.pre) > 0 && !branch.prerelease[v.core] {
278 continue
279 }
280 valid := true
281 for _, comparison := range branch.comparisons {
282 order := compare(v, comparison.value)
283 switch comparison.op {
284 case "=":
285 valid = order == 0
286 case ">":
287 valid = order > 0
288 case ">=":
289 valid = order >= 0
290 case "<":
291 valid = order < 0
292 case "<=":
293 valid = order <= 0
294 }
295 if !valid {
296 break
297 }
298 }
299 if valid {
300 return true
301 }
302 }
303 return false
304}
305func (m *manager) constraint(text string) (*constraint, error) {
306 if value := m.parsed[text]; value != nil {
307 return value, nil
308 }
309 value := parseConstraint(text)
310 if value == nil {
311 return nil, fmt.Errorf("invalid constraint '%s'", text)
312 }
313 m.parsed[text] = value
314 return value, nil
315}
316func checkedVersion(text string) (*version, error) {
317 value := parseVersion(text)
318 if value == nil {
319 return nil, fmt.Errorf("invalid version '%s'", text)
320 }
321 return value, nil
322}
323func (m *manager) matches(v *version, text string) bool { return m.parsed[text].matches(v) }
324func (m *manager) find(name, text string) *release {
325 for _, value := range m.registry[name] {
326 if value.version.text == text {
327 return value
328 }
329 }
330 return nil
331}

334 for name, value := range items {

339func (m *manager) applicable(value *release) []string {
340 result := []string{}
341 for _, id := range keys(m.advisories) {
342 advisory := m.advisories[id]
343 if advisory.name == value.name && m.matches(value.version, advisory.text) {
344 result = append(result, id)
345 }
346 }
347 return result
348}
349func (m *manager) usable(value *release, audit bool) bool {
350 return (!value.yanked || m.installed[value.name] == value) && (!audit || len(m.applicable(value)) == 0)
351}
352func (m *manager) display(value *release) string {
353 flags := []string{}
354 if value.yanked {
355 flags = append(flags, "yanked")
356 }
357 if m.installed[value.name] == value {
358 flags = append(flags, "installed")
359 }
360 text := value.name + "@" + value.version.text
361 if len(flags) > 0 {
362 text += " (" + strings.Join(flags, ", ") + ")"
363 }
364 return text
365}
366func label(value *release) string { return value.name + "@" + value.version.text }

368func (m *manager) resolve(requests []requirementPair, preferred map[string]bool, audit bool) (map[string]*release, []string) {
369 selected, origins := map[string]*release{}, map[string]requirement{}
370 failure, depth := []string(nil), -1
371 record := func(r requirement, reason string, extra []string) {
372 if len(selected) <= depth {
373 return
374 }
375 depth, failure = len(selected), []string{"UNSATISFIABLE"}
376 for _, line := range r.chain {
377 failure = append(failure, " "+line)
378 }
379 failure = append(failure, " "+reason)
380 for _, line := range extra {
381 failure = append(failure, " "+line)
382 }
383 }
384 var search func([]requirement) bool
385 search = func(todo []requirement) bool {
386 if len(todo) == 0 {
387 return true
388 }
389 r, rest := todo[0], todo[1:]
390 if chosen := selected[r.name]; chosen != nil {
391 if m.matches(chosen.version, r.text) {
392 return search(rest)
393 }
394 record(r, label(chosen)+" was selected by:", origins[r.name].chain)
395 return false
396 }
397 candidates := []*release{}
398 if m.held[r.name] {
399 value := m.installed[r.name]
400 if m.matches(value.version, r.text) && m.usable(value, audit) {
401 candidates = append(candidates, value)
402 } else {
403 record(r, label(value)+" is held", nil)
404 return false
405 }
406 } else {
407 for _, value := range m.registry[r.name] {
408 if m.matches(value.version, r.text) && m.usable(value, audit) {
409 candidates = append(candidates, value)
410 }
411 }
412 if preferred[r.name] {
413 for i, value := range candidates {
414 if m.installed[r.name] == value {
415 candidates = append([]*release{value}, append(candidates[:i:i], candidates[i+1:]...)...)
416 break
417 }
418 }
419 }
420 if len(candidates) == 0 {
421 record(r, fmt.Sprintf("no version of %s matches %s", r.name, r.text), nil)
422 return false
423 }
424 }
425 for _, candidate := range candidates {
426 conflicting := ""
427 for _, name := range keys(selected) {
428 chosen := selected[name]
429 conflict := false
430 for _, pair := range candidate.conflicts {
431 if pair.name == name && m.matches(chosen.version, pair.text) {
432 conflict = true
433 break
434 }
435 }
436 if !conflict {
437 for _, pair := range chosen.conflicts {
438 if pair.name == r.name && m.matches(candidate.version, pair.text) {
439 conflict = true
440 break
441 }
442 }
443 }
444 if conflict {
445 conflicting = name
446 break
447 }
448 }
449 if conflicting != "" {
450 record(r, fmt.Sprintf("%s conflicts with %s, which was selected by:", label(candidate), label(selected[conflicting])), origins[conflicting].chain)
451 continue
452 }
453 selected[r.name], origins[r.name] = candidate, r
454 following := []requirement{}
455 for _, pair := range candidate.dependencies {
456 chain := append(append([]string{}, r.chain...), fmt.Sprintf("%s requires %s %s", label(candidate), pair.name, pair.text))
457 following = append(following, requirement{pair, chain})
458 }
459 following = append(following, rest...)
460 if search(following) {
461 return true
462 }
463 delete(selected, r.name)
464 delete(origins, r.name)
465 }
466 return false
467 }
468 initial := []requirement{}
469 for _, pair := range requests {
470 initial = append(initial, requirement{pair, []string{fmt.Sprintf("requested %s %s", pair.name, pair.text)}})
471 }
472 if search(initial) {
473 return selected, nil
474 }
475 return nil, failure
476}
477func (m *manager) reportFailure(lines []string) {
478 for _, line := range lines {
479 fmt.Fprintln(m.out, line)
480 }
481}
482func (m *manager) plan(next map[string]*release) {
483 arranged, installed, upgraded, downgraded, removed := map[string]bool{}, 0, 0, 0, 0
484 for len(arranged) < len(next) {
485 for _, name := range keys(next) {
486 if arranged[name] {
487 continue
488 }
489 ready := true
490 for _, dep := range next[name].dependencies {
491 if !arranged[dep.name] {
492 ready = false
493 break
494 }
495 }
496 if !ready {
497 continue
498 }
499 value, old := next[name], m.installed[name]
500 if old == nil {
501 fmt.Fprintln(m.out, "install "+label(value))
502 installed++
503 } else if old != value {
504 action := "upgrade"
505 if compare(value.version, old.version) < 0 {
506 action = "downgrade"
507 downgraded++
508 } else {
509 upgraded++
510 }
511 fmt.Fprintf(m.out, "%s %s %s -> %s\n", action, name, old.version.text, value.version.text)
512 }
513 arranged[name] = true
514 break
515 }
516 }
517 dropping := map[string]bool{}
518 for name := range m.installed {
519 if next[name] == nil {
520 dropping[name] = true
521 }
522 }
523 for len(dropping) > 0 {
524 for _, name := range keys(dropping) {
525 ready := true
526 for other := range dropping {
527 for _, dep := range m.installed[other].dependencies {
528 if dep.name == name {
529 ready = false
530 break
531 }
532 }
533 if !ready {
534 break
535 }
536 }
537 if ready {
538 fmt.Fprintln(m.out, "remove "+label(m.installed[name]))
539 delete(dropping, name)
540 removed++
541 break
542 }
543 }
544 }
545 if installed+upgraded+downgraded+removed == 0 {
546 fmt.Fprintln(m.out, "no changes")
547 } else {
548 fmt.Fprintf(m.out, "%d installed, %d upgraded, %d downgraded, %d removed\n", installed, upgraded, downgraded, removed)
549 }
550}
551func (m *manager) update(requests []requirementPair, installed map[string]*release, save bool) {
552 if save {
553 m.history = append(m.history, snapshot{append([]requirementPair{}, m.requests...), clone(m.installed)})
554 }
555 m.requests, m.installed = append([]requirementPair{}, requests...), clone(installed)
556 for name := range m.held {
557 if installed[name] == nil {
558 delete(m.held, name)
559 }
560 }
561}
562func (m *manager) parent(name string) string {
563 for _, candidate := range keys(m.installed) {
564 for _, dep := range m.installed[candidate].dependencies {
565 if dep.name == name {
566 return candidate
567 }
568 }
569 }
570 return ""
571}
572func (m *manager) requested(name string) (string, bool) {
573 for _, pair := range m.requests {
574 if pair.name == name {
575 return pair.text, true
576 }
577 }
578 return "", false

580func (m *manager) tree(root string, seen map[string]bool, text *string, prefix, indent string) {
581 value := m.installed[root]
582 body := prefix + label(value)
583 if text != nil {
584 body += " [" + *text + "]"
585 }
586 if seen[root] {
587 fmt.Fprintln(m.out, body+" (*)")
588 return
589 }
590 fmt.Fprintln(m.out, body)
591 seen[root] = true
592 for i, pair := range value.dependencies {
593 mark, following := "|-- ", "| "
594 if i+1 == len(value.dependencies) {
595 mark, following = "`-- ", " "
596 }
597 m.tree(pair.name, seen, &pair.text, indent+mark, indent+following)
598 }
599}

601func (m *manager) command(line string) error {
602 w := strings.Split(line, " ")
603 cmd := w[0]
604 switch cmd {
605 case "PKG":
606 clauses := strings.Split(line, "; ")
607 head := strings.Split(clauses[0], " ")
608 value, err := checkedVersion(head[2])
609 if err != nil {
610 return err
611 }
612 if m.find(head[1], head[2]) != nil {
613 return fmt.Errorf("%s@%s is already registered", head[1], head[2])
614 }
615 candidate := &release{name: head[1], version: value}
616 for _, clause := range clauses[1:] {
617 parts := strings.SplitN(clause, " ", 3)
618 if _, err := m.constraint(parts[2]); err != nil {
619 return err
620 }
621 pair := requirementPair{parts[1], parts[2]}
622 if parts[0] == "DEP" {
623 candidate.dependencies = append(candidate.dependencies, pair)
624 } else {
625 candidate.conflicts = append(candidate.conflicts, pair)
626 }
627 }
628 m.registry[candidate.name] = append(m.registry[candidate.name], candidate)
629 sort.Slice(m.registry[candidate.name], func(i, j int) bool {
630 return compare(m.registry[candidate.name][i].version, m.registry[candidate.name][j].version) > 0
631 })
632 fmt.Fprintln(m.out, "registered "+label(candidate))
633 case "YANK", "INFO":
634 value := m.find(w[1], w[2])
635 if value == nil {
636 return fmt.Errorf("%s@%s is not registered", w[1], w[2])
637 }
638 if cmd == "YANK" {
639 if value.yanked {
640 return fmt.Errorf("%s is already yanked", label(value))
641 }
642 value.yanked = true
643 fmt.Fprintln(m.out, "yanked "+label(value))
644 } else {
645 fmt.Fprintln(m.out, m.display(value))
646 for _, pair := range value.dependencies {
647 fmt.Fprintf(m.out, " requires %s %s\n", pair.name, pair.text)
648 }
649 for _, pair := range value.conflicts {
650 fmt.Fprintf(m.out, " conflicts with %s %s\n", pair.name, pair.text)
651 }
652 for _, id := range m.applicable(value) {
653 fmt.Fprintln(m.out, " advisory "+id)
654 }
655 }
656 case "COMPARE":
657 a, err := checkedVersion(w[1])
658 if err != nil {
659 return err
660 }
661 b, err := checkedVersion(w[2])
662 if err != nil {
663 return err
664 }
665 relation := "="
666 if compare(a, b) < 0 {
667 relation = "<"
668 } else if compare(a, b) > 0 {
669 relation = ">"
670 }
671 fmt.Fprintf(m.out, "%s %s %s\n", w[1], relation, w[2])
672 case "MATCH":
673 parts := strings.SplitN(line, " ", 3)
674 value, err := checkedVersion(parts[1])
675 if err != nil {
676 return err
677 }
678 condition, err := m.constraint(parts[2])
679 if err != nil {
680 return err
681 }
682 if condition.matches(value) {
683 fmt.Fprintln(m.out, "yes")
684 } else {
685 fmt.Fprintln(m.out, "no")
686 }
687 case "RANGE":
688 condition, err := m.constraint(line[6:])
689 if err != nil {
690 return err
691 }
692 fmt.Fprintln(m.out, condition.normal)
693 case "VERSIONS", "INSTALL", "SIMULATE":
694 parts := strings.SplitN(line, " ", 3)
695 text := "*"
696 if len(parts) == 3 {
697 text = parts[2]
698 }
699 condition, err := m.constraint(text)
700 if err != nil {
701 return err
702 }
703 if len(m.registry[w[1]]) == 0 {
704 return fmt.Errorf("unknown package %s", w[1])
705 }
706 if cmd == "VERSIONS" {
707 count := 0
708 for _, value := range m.registry[w[1]] {
709 if len(parts) == 2 || condition.matches(value.version) {
710 fmt.Fprintln(m.out, m.display(value))
711 count++
712 }
713 }
714 if count == 0 {
715 fmt.Fprintln(m.out, "no matching versions")
716 }
717 } else {
718 requests, found := append([]requirementPair{}, m.requests...), false
719 for i := range requests {
720 if requests[i].name == w[1] {
721 requests[i].text, found = text, true
722 break
723 }
724 }
725 if !found {
726 requests = append(requests, requirementPair{w[1], text})
727 }
728 preferred := map[string]bool{}
729 for name := range m.installed {
730 preferred[name] = true
731 }
732 next, failure := m.resolve(requests, preferred, false)
733 if failure != nil {
734 m.reportFailure(failure)
735 } else {
736 m.plan(next)
737 if cmd == "INSTALL" {
738 m.update(requests, next, true)
739 }
740 }
741 }
742 case "DEPENDENTS":
743 var selected *version
744 if len(w) == 3 {
745 var err error
746 selected, err = checkedVersion(w[2])
747 if err != nil {
748 return err
749 }
750 }
751 count := 0
752 for _, name := range keys(m.registry) {
753 for _, value := range m.registry[name] {
754 for _, pair := range value.dependencies {
755 if pair.name == w[1] && (selected == nil || m.matches(selected, pair.text)) {
756 fmt.Fprintf(m.out, "%s requires %s %s\n", label(value), w[1], pair.text)
757 count++
758 }
759 }
760 }
761 }
762 if count == 0 {
763 fmt.Fprintln(m.out, "no dependents")
764 }
765 case "REMOVE":
766 _, requested := m.requested(w[1])
767 if !requested {
768 if m.installed[w[1]] != nil {
769 return fmt.Errorf("%s is required by %s", w[1], m.parent(w[1]))
770 }
771 return fmt.Errorf("%s is not installed", w[1])
772 }
773 requests, keep, queue := []requirementPair{}, map[string]*release{}, []string{}
774 for _, pair := range m.requests {
775 if pair.name != w[1] {
776 requests = append(requests, pair)
777 queue = append(queue, pair.name)
778 }
779 }
780 for at := 0; at < len(queue); at++ {
781 name := queue[at]
782 if keep[name] != nil {
783 continue
784 }
785 keep[name] = m.installed[name]
786 for _, dep := range m.installed[name].dependencies {
787 queue = append(queue, dep.name)
788 }
789 }
790 if keep[w[1]] != nil {
791 fmt.Fprintf(m.out, "%s is still required by %s\n", label(m.installed[w[1]]), m.parent(w[1]))
792 m.update(requests, m.installed, true)
793 } else {
794 m.plan(keep)
795 m.update(requests, keep, true)
796 }
797 case "UPGRADE":
798 if len(w) == 2 && m.installed[w[1]] == nil {
799 return fmt.Errorf("%s is not installed", w[1])
800 }
801 preferred := map[string]bool{}
802 if len(w) == 2 {
803 for name := range m.installed {
804 if name != w[1] {
805 preferred[name] = true
806 }
807 }
808 }
809 next, failure := m.resolve(m.requests, preferred, false)
810 if failure != nil {
811 m.reportFailure(failure)
812 } else {
813 m.plan(next)
814 m.update(m.requests, next, true)
815 }
816 case "ROLLBACK":
817 if len(m.history) == 0 {
818 return fmt.Errorf("nothing to roll back")
819 }
820 saved := m.history[len(m.history)-1]
821 m.history = m.history[:len(m.history)-1]
822 m.plan(saved.installed)
823 m.update(saved.requests, saved.installed, false)
824 case "HOLD":
825 if m.installed[w[1]] == nil {
826 return fmt.Errorf("%s is not installed", w[1])
827 }
828 if m.held[w[1]] {
829 return fmt.Errorf("%s is already held", w[1])
830 }
831 m.held[w[1]] = true
832 fmt.Fprintln(m.out, "held "+label(m.installed[w[1]]))
833 case "UNHOLD":
834 if !m.held[w[1]] {
835 return fmt.Errorf("%s is not held", w[1])
836 }
837 delete(m.held, w[1])
838 fmt.Fprintln(m.out, "released "+w[1])
839 case "WHY":
840 value := m.installed[w[1]]
841 if value == nil {
842 return fmt.Errorf("%s is not installed", w[1])
843 }
844 if _, requested := m.requested(w[1]); requested {
845 fmt.Fprintln(m.out, label(value)+" is explicitly installed")
846 return nil
847 }
848 previous, seen, queue := map[string]string{}, map[string]bool{}, []string{}
849 roots := map[string]bool{}
850 for _, pair := range m.requests {
851 roots[pair.name] = true
852 }
853 for _, name := range keys(roots) {
854 queue = append(queue, name)
855 seen[name] = true
856 }
857 for at := 0; at < len(queue); at++ {
858 name, deps := queue[at], map[string]bool{}
859 for _, pair := range m.installed[name].dependencies {
860 deps[pair.name] = true
861 }
862 for _, dep := range keys(deps) {
863 if !seen[dep] {
864 seen[dep], previous[dep] = true, name
865 queue = append(queue, dep)
866 }
867 }
868 }
869 path := []string{}
870 for name := w[1]; name != ""; name = previous[name] {
871 path = append(path, label(m.installed[name]))
872 }
873 for a, b := 0, len(path)-1; a < b; a, b = a+1, b-1 {
874 path[a], path[b] = path[b], path[a]
875 }
876 fmt.Fprintln(m.out, strings.Join(path, " -> "))
877 case "TREE":
878 if len(w) == 2 && m.installed[w[1]] == nil {
879 return fmt.Errorf("%s is not installed", w[1])
880 }
881 roots := []string{}
882 if len(w) == 2 {
883 roots = append(roots, w[1])
884 } else {
885 for _, pair := range m.requests {
886 roots = append(roots, pair.name)
887 }
888 sort.Strings(roots)
889 }
890 if len(roots) == 0 {
891 fmt.Fprintln(m.out, "(empty)")
892 }
893 seen := map[string]bool{}
894 for _, name := range roots {
895 var text *string
896 if value, requested := m.requested(name); requested {
897 text = &value
898 }
899 m.tree(name, seen, text, "", "")
900 }
901 case "OUTDATED":
902 count := 0
903 for _, name := range keys(m.installed) {
904 current, conditions := m.installed[name], []string{}
905 if text, requested := m.requested(name); requested {
906 conditions = append(conditions, text)
907 }
908 for _, value := range m.installed {
909 for _, pair := range value.dependencies {
910 if pair.name == name {
911 conditions = append(conditions, pair.text)
912 }
913 }
914 }
915 var wanted, latest *release
916 for _, value := range m.registry[name] {
917 if !value.yanked && len(value.version.pre) == 0 && latest == nil {
918 latest = value
919 }
920 if wanted != nil || value.yanked && value != current {
921 continue
922 }
923 valid := true
924 for _, text := range conditions {
925 if !m.matches(value.version, text) {
926 valid = false
927 break
928 }
929 }
930 if valid {
931 wanted = value
932 }
933 }
934 if wanted != current || latest != nil && compare(latest.version, current.version) > 0 {
935 latestText := "-"
936 if latest != nil {
937 latestText = latest.version.text
938 }
939 fmt.Fprintf(m.out, "%s %s %s %s\n", name, current.version.text, wanted.version.text, latestText)
940 count++
941 }
942 }
943 if count == 0 {
944 fmt.Fprintln(m.out, "all packages are up to date")
945 }
946 case "LOCK":
947 if len(m.installed) == 0 {
948 fmt.Fprintln(m.out, "(empty)")
949 }
950 for _, name := range keys(m.installed) {
951 fmt.Fprintln(m.out, label(m.installed[name]))
952 }
953 case "STATUS":
954 if len(m.requests) == 0 {
955 fmt.Fprintln(m.out, "no requests")
956 }
957 for _, pair := range m.requests {
958 fmt.Fprintf(m.out, "%s %s -> %s\n", pair.name, pair.text, m.installed[pair.name].version.text)
959 }
960 fmt.Fprintf(m.out, "%d installed, %d held, %d snapshots\n", len(m.installed), len(m.held), len(m.history))
961 case "ADVISORY":
962 parts := strings.SplitN(line, " ", 4)
963 if _, exists := m.advisories[w[1]]; exists {
964 return fmt.Errorf("advisory %s already exists", w[1])
965 }
966 if _, err := m.constraint(parts[3]); err != nil {
967 return err
968 }
969 m.advisories[w[1]] = advisory{w[2], parts[3]}
970 fmt.Fprintf(m.out, "advisory %s recorded\n", w[1])
971 case "AUDIT":
972 if len(w) == 2 {
973 preferred := map[string]bool{}
974 for name := range m.installed {
975 preferred[name] = true
976 }
977 next, failure := m.resolve(m.requests, preferred, true)
978 if failure != nil {
979 m.reportFailure(failure)
980 } else {
981 m.plan(next)
982 m.update(m.requests, next, true)
983 }
984 } else {
985 count := 0
986 for _, name := range keys(m.installed) {
987 current := m.installed[name]
988 ids := m.applicable(current)
989 if len(ids) == 0 {
990 continue
991 }
992 var fix *release
993 values := m.registry[name]
994 for i := len(values) - 1; i >= 0; i-- {
995 value := values[i]
996 if compare(value.version, current.version) > 0 && len(value.version.pre) == 0 && !value.yanked && len(m.applicable(value)) == 0 {
997 fix = value
998 break
999 }
1000 }
1001 suffix := "no fix"
1002 if fix != nil {
1003 suffix = "fix: " + fix.version.text
1004 }
1005 fmt.Fprintf(m.out, "%s: %s (%s)\n", label(current), strings.Join(ids, ", "), suffix)
1006 count++
1007 }
1008 if count == 0 {
1009 fmt.Fprintln(m.out, "no known vulnerabilities")
1010 }
1011 }
1012 }
1013 return nil
1014}
