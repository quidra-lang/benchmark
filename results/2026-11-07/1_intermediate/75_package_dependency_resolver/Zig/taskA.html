7fn equal(a: Text, b: Text) bool {
8 return std.mem.eql(u8, a, b);

10fn bad(gpa: Allocator, comptime pattern: Text, args: anytype) Error {
11 failure = std.fmt.allocPrint(gpa, pattern, args) catch return error.OutOfMemory;

14const Builder = struct {
15 gpa: Allocator,
16 bytes: std.ArrayList(u8) = .empty,
17 fn add(self: *Builder, text: Text) Allocator.Error!void {
18 try self.bytes.appendSlice(self.gpa, text);
19 }
20 fn print(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
21 try self.add(try std.fmt.allocPrint(self.gpa, pattern, args));
22 }
23 fn line(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
24 try self.print(pattern ++ "\n", args);
25 }

27fn split(gpa: Allocator, text: Text, separator: Text, limit: usize) Allocator.Error![]Text {
28 var result: std.ArrayList(Text) = .empty;
29 var start: usize = 0;
30 while (limit == 0 or result.items.len + 1 < limit) {
31 const offset = std.mem.find(u8, text[start..], separator) orelse break;
32 try result.append(gpa, text[start .. start + offset]);
33 start += offset + separator.len;
34 }
35 try result.append(gpa, text[start..]);
36 return result.items;
37}
38fn textLess(_: void, a: Text, b: Text) bool {
39 return std.mem.order(u8, a, b) == .lt;
40}
41fn keys(gpa: Allocator, map: anytype) Allocator.Error![]Text {
42 var result: std.ArrayList(Text) = .empty;
43 var iterator = map.keyIterator();
44 while (iterator.next()) |key| try result.append(gpa, key.*);
45 std.mem.sort(Text, result.items, {}, textLess);
46 return result.items;
47}
48fn join(gpa: Allocator, parts: []const Text, separator: Text) Allocator.Error!Text {
49 var out: Builder = .{ .gpa = gpa };
50 for (parts, 0..) |part, i| {
51 if (i > 0) try out.add(separator);
52 try out.add(part);
53 }
54 return out.bytes.items;
55}
56fn numeric(text: Text) bool {
57 if (text.len == 0) return false;
58 for (text) |c| if (!std.ascii.isDigit(c)) return false;
59 return true;
60}
61fn numberWord(text: Text) bool {
62 return text.len >= 1 and text.len <= 4 and numeric(text) and (text.len == 1 or text[0] != '0');
63}
64fn integer(text: Text) usize {
65 return std.fmt.parseInt(usize, text, 10) catch unreachable;
66}
67const Version = struct { core: [3]usize, pre: []const Text, text: Text };
68fn makeVersion(gpa: Allocator, core: [3]usize, pre: []const Text) Allocator.Error!Version {
69 const base = try std.fmt.allocPrint(gpa, "{d}.{d}.{d}", .{ core[0], core[1], core[2] });
70 const text = if (pre.len == 0) base else try std.fmt.allocPrint(gpa, "{s}-{s}", .{ base, try join(gpa, pre, ".") });
71 return .{ .core = core, .pre = pre, .text = text };
72}
73fn parseVersion(gpa: Allocator, text: Text) Allocator.Error!?Version {
74 const fields = try split(gpa, text, "-", 2);
75 const parts = try split(gpa, fields[0], ".", 0);
76 if (parts.len != 3) return null;
77 var core: [3]usize = undefined;
78 for (parts, 0..) |part, i| {
79 if (!numberWord(part)) return null;
80 core[i] = integer(part);
81 }
82 var pre: []const Text = &.{};
83 if (fields.len == 2) {
84 pre = try split(gpa, fields[1], ".", 0);
85 for (pre) |part| {
86 if (part.len == 0) return null;
87 for (part) |c| if ((c < 'a' or c > 'z') and !std.ascii.isDigit(c)) return null;
88 if (numeric(part) and !numberWord(part)) return null;
89 }
90 }
91 return try makeVersion(gpa, core, pre);
92}
93fn compare(a: Version, b: Version) std.math.Order {
94 for (a.core, b.core) |x, y| {
95 if (x != y) return if (x < y) .lt else .gt;
96 }
97 if (a.pre.len == 0 and b.pre.len > 0) return .gt;
98 if (b.pre.len == 0 and a.pre.len > 0) return .lt;
99 for (0..@min(a.pre.len, b.pre.len)) |i| {
100 const x = a.pre[i];
101 const y = b.pre[i];
102 if (equal(x, y)) continue;
103 const nx = numeric(x);
104 const ny = numeric(y);
105 if (nx and ny) return if (integer(x) < integer(y)) .lt else .gt;
106 if (nx != ny) return if (nx) .lt else .gt;
107 return std.mem.order(u8, x, y);
108 }
109 return if (a.pre.len < b.pre.len) .lt else if (a.pre.len > b.pre.len) .gt else .eq;
110}
111const Operator = enum { eq, gt, ge, lt, le };
112const Comparator = struct { op: Operator, value: Version };
113const Alternative = struct { comparisons: []const Comparator, prerelease: []const [3]usize };
114const Constraint = struct {
115 alternatives: []const Alternative,
116 normal: Text,
117 fn matches(self: *const Constraint, value: Version) bool {
118 for (self.alternatives) |branch| {
119 var allowed = value.pre.len == 0;
120 for (branch.prerelease) |core| {
121 if (std.mem.eql(usize, &core, &value.core)) {
122 allowed = true;
123 break;
124 }
125 }
126 if (!allowed) continue;
127 var valid = true;
128 for (branch.comparisons) |comparison| {
129 const order = compare(value, comparison.value);
130 valid = switch (comparison.op) {
131 .eq => order == .eq,
132 .gt => order == .gt,
133 .ge => order != .lt,
134 .lt => order == .lt,
135 .le => order != .gt,
136 };
137 if (!valid) break;
138 }
139 if (valid) return true;
140 }
141 return false;
142 }
143};
144fn operatorText(op: Operator) Text {
145 return switch (op) {
146 .eq => "=",
147 .gt => ">",
148 .ge => ">=",
149 .lt => "<",
150 .le => "<=",
151 };
152}
153fn upperVersion(gpa: Allocator, core: [3]usize, at: usize) Allocator.Error!Version {
154 var next = core;
155 next[at] += 1;
156 for (at + 1..3) |i| next[i] = 0;
157 return makeVersion(gpa, next, &.{});
158}
159fn parseConstraint(gpa: Allocator, text: Text) Allocator.Error!?*Constraint {
160 var alternatives: std.ArrayList(Alternative) = .empty;
161 var forms: std.ArrayList(Text) = .empty;
162 for (try split(gpa, text, " || ", 0)) |branch| {
163 var comparisons: std.ArrayList(Comparator) = .empty;
164 var prerelease: std.ArrayList([3]usize) = .empty;
165 for (try split(gpa, branch, " ", 0)) |term| {
166 if (term.len == 0) return null;
167 var op: Text = "";
168 var pattern = term;
169 for ([_]Text{ ">=", "<=", ">", "<", "=", "^", "~" }) |candidate| {
170 if (std.mem.startsWith(u8, term, candidate)) {
171 op = candidate;
172 pattern = term[candidate.len..];
173 break;
174 }
175 }
176 var core = [_]usize{ 0, 0, 0 };
177 var pre: []const Text = &.{};
178 var count: usize = 0;
179 if (std.mem.findScalar(u8, pattern, '-') != null) {
180 const value = (try parseVersion(gpa, pattern)) orelse return null;
181 if (value.pre.len == 0) return null;
182 core = value.core;
183 pre = value.pre;
184 count = 3;
185 try prerelease.append(gpa, core);
186 } else {
187 const parts = try split(gpa, pattern, ".", 0);
188 if (parts.len > 3) return null;
189 var wildcard = false;
190 for (parts, 0..) |part, i| {
191 if (equal(part, "x") or equal(part, "*")) wildcard = true else {
192 if (wildcard or !numberWord(part)) return null;
193 core[i] = integer(part);
194 count += 1;
195 }
196 }
197 }
198 const lower = try makeVersion(gpa, core, pre);
199 if (count == 0) {
200 if (equal(op, ">") or equal(op, "<")) try comparisons.append(gpa, .{ .op = .lt, .value = try makeVersion(gpa, .{ 0, 0, 0 }, &.{"0"}) });
201 } else if (op.len == 0 or equal(op, "=")) {
202 if (count == 3) try comparisons.append(gpa, .{ .op = .eq, .value = lower }) else {
203 try comparisons.append(gpa, .{ .op = .ge, .value = lower });
204 try comparisons.append(gpa, .{ .op = .lt, .value = try upperVersion(gpa, core, count - 1) });
205 }
206 } else if (equal(op, ">")) {
207 try comparisons.append(gpa, .{ .op = if (count == 3) .gt else .ge, .value = if (count == 3) lower else try upperVersion(gpa, core, count - 1) });
208 } else if (equal(op, "<=")) {
209 try comparisons.append(gpa, .{ .op = if (count == 3) .le else .lt, .value = if (count == 3) lower else try upperVersion(gpa, core, count - 1) });
210 } else if (equal(op, ">=") or equal(op, "<")) {
211 try comparisons.append(gpa, .{ .op = if (equal(op, ">=")) .ge else .lt, .value = lower });
212 } else {
213 const at: usize = if (equal(op, "~")) (if (count == 1) @as(usize, 0) else 1) else if (core[0] != 0 or count == 1) 0 else if (core[1] != 0 or count == 2) 1 else 2;
214 try comparisons.append(gpa, .{ .op = .ge, .value = lower });
215 try comparisons.append(gpa, .{ .op = .lt, .value = try upperVersion(gpa, core, at) });
216 }
217 }
218 var form: Builder = .{ .gpa = gpa };
219 for (comparisons.items, 0..) |comparison, i| {
220 if (i > 0) try form.add(" ");
221 try form.print("{s}{s}", .{ operatorText(comparison.op), comparison.value.text });
222 }
223 if (comparisons.items.len == 0) try form.add("*");
224 try forms.append(gpa, form.bytes.items);
225 try alternatives.append(gpa, .{ .comparisons = comparisons.items, .prerelease = prerelease.items });
226 }
227 const result = try gpa.create(Constraint);
228 result.* = .{ .alternatives = alternatives.items, .normal = try join(gpa, forms.items, " || ") };
229 return result;
230}

251const Resolver = struct {
252 manager: *Manager,
253 gpa: Allocator,
254 preferred: Set,
255 audit: bool,
256 selected: Installation = .empty,
257 origins: std.StringHashMapUnmanaged(Requirement) = .empty,
258 failure: ?[]const Text = null,
259 depth: usize = 0,
260 fn record(self: *Resolver, requirement: Requirement, reason: Text, extra: []const Text) Allocator.Error!void {
261 if (self.failure != null and self.selected.count() <= self.depth) return;
262 self.depth = self.selected.count();
263 var lines: std.ArrayList(Text) = .empty;
264 try lines.append(self.gpa, "UNSATISFIABLE");
265 for (requirement.chain) |line| try lines.append(self.gpa, try std.fmt.allocPrint(self.gpa, " {s}", .{line}));
266 try lines.append(self.gpa, try std.fmt.allocPrint(self.gpa, " {s}", .{reason}));
267 for (extra) |line| try lines.append(self.gpa, try std.fmt.allocPrint(self.gpa, " {s}", .{line}));
268 self.failure = lines.items;
269 }
270 fn search(self: *Resolver, todo: []const Requirement) Allocator.Error!bool {
271 if (todo.len == 0) return true;
272 const requirement = todo[0];
273 const name = requirement.pair.name;
274 const text = requirement.pair.text;
275 if (self.selected.get(name)) |chosen| {
276 if (self.manager.matches(chosen.version, text)) return self.search(todo[1..]);
277 try self.record(requirement, try std.fmt.allocPrint(self.gpa, "{s} was selected by:", .{try label(self.gpa, chosen)}), self.origins.get(name).?.chain);
278 return false;
279 }
280 var candidates: std.ArrayList(*Release) = .empty;
281 if (self.manager.held.contains(name)) {
282 const value = self.manager.installed.get(name).?;
283 if (self.manager.matches(value.version, text) and try self.manager.usable(self.gpa, value, self.audit)) {
284 try candidates.append(self.gpa, value);
285 } else {
286 try self.record(requirement, try std.fmt.allocPrint(self.gpa, "{s} is held", .{try label(self.gpa, value)}), &.{});
287 return false;
288 }
289 } else {
290 if (self.manager.registry.get(name)) |versions| {
291 for (versions.items) |value| {
292 if (self.manager.matches(value.version, text) and try self.manager.usable(self.gpa, value, self.audit)) try candidates.append(self.gpa, value);
293 }
294 }
295 if (self.preferred.contains(name)) {
296 for (candidates.items, 0..) |value, i| {
297 if (self.manager.installed.get(name) == value) {
298 _ = candidates.orderedRemove(i);
299 try candidates.insert(self.gpa, 0, value);
300 break;
301 }
302 }
303 }
304 if (candidates.items.len == 0) {
305 try self.record(requirement, try std.fmt.allocPrint(self.gpa, "no version of {s} matches {s}", .{ name, text }), &.{});
306 return false;
307 }
308 }
309 for (candidates.items) |candidate| {
310 var conflicting: ?Text = null;
311 for (try keys(self.gpa, self.selected)) |other| {
312 const chosen = self.selected.get(other).?;
313 var conflict = false;
314 for (candidate.conflicts) |pair| {
315 if (equal(pair.name, other) and self.manager.matches(chosen.version, pair.text)) {
316 conflict = true;
317 break;
318 }
319 }
320 if (!conflict) {
321 for (chosen.conflicts) |pair| {
322 if (equal(pair.name, name) and self.manager.matches(candidate.version, pair.text)) {
323 conflict = true;
324 break;
325 }
326 }
327 }
328 if (conflict) {
329 conflicting = other;
330 break;
331 }
332 }
333 if (conflicting) |other| {
334 try self.record(requirement, try std.fmt.allocPrint(self.gpa, "{s} conflicts with {s}, which was selected by:", .{ try label(self.gpa, candidate), try label(self.gpa, self.selected.get(other).?) }), self.origins.get(other).?.chain);
335 continue;
336 }
337 try self.selected.put(self.gpa, name, candidate);
338 try self.origins.put(self.gpa, name, requirement);
339 var following: std.ArrayList(Requirement) = .empty;
340 for (candidate.dependencies) |pair| {
341 var chain: std.ArrayList(Text) = .empty;
342 try chain.appendSlice(self.gpa, requirement.chain);
343 try chain.append(self.gpa, try std.fmt.allocPrint(self.gpa, "{s} requires {s} {s}", .{ try label(self.gpa, candidate), pair.name, pair.text }));
344 try following.append(self.gpa, .{ .pair = pair, .chain = chain.items });
345 }
346 try following.appendSlice(self.gpa, todo[1..]);
347 if (try self.search(following.items)) return true;
348 _ = self.selected.remove(name);
349 _ = self.origins.remove(name);
350 }
351 return false;
352 }
353};
354const Manager = struct {
355 gpa: Allocator,
356 out: *Builder,
357 registry: std.StringHashMapUnmanaged(std.ArrayList(*Release)) = .empty,
358 parsed: std.StringHashMapUnmanaged(*Constraint) = .empty,
359 requests: []const Pair = &.{},
360 installed: Installation = .empty,
361 held: Set = .empty,
362 advisories: std.StringHashMapUnmanaged(Advisory) = .empty,
363 history: std.ArrayList(Snapshot) = .empty,
364 fn constraint(self: *Manager, text: Text) Error!*Constraint {
365 if (self.parsed.get(text)) |value| return value;
366 const value = (try parseConstraint(self.gpa, text)) orelse return bad(self.gpa, "invalid constraint '{s}'", .{text});
367 try self.parsed.put(self.gpa, text, value);
368 return value;
369 }
370 fn checkedVersion(self: *Manager, text: Text) Error!Version {
371 return (try parseVersion(self.gpa, text)) orelse return bad(self.gpa, "invalid version '{s}'", .{text});
372 }
373 fn matches(self: *const Manager, value: Version, text: Text) bool {
374 return self.parsed.get(text).?.matches(value);
375 }
376 fn find(self: *const Manager, name: Text, text: Text) ?*Release {
377 if (self.registry.get(name)) |versions| {
378 for (versions.items) |value| if (equal(value.version.text, text)) return value;
379 }
380 return null;
381 }
382 fn applicable(self: *const Manager, gpa: Allocator, value: *Release) Allocator.Error![]Text {
383 var result: std.ArrayList(Text) = .empty;
384 for (try keys(gpa, self.advisories)) |id| {
385 const advisory = self.advisories.get(id).?;
386 if (equal(advisory.name, value.name) and self.matches(value.version, advisory.text)) try result.append(gpa, id);
387 }
388 return result.items;
389 }
390 fn usable(self: *const Manager, gpa: Allocator, value: *Release, audit: bool) Allocator.Error!bool {
391 return (!value.yanked or self.installed.get(value.name) == value) and (!audit or (try self.applicable(gpa, value)).len == 0);
392 }
393 fn display(self: *const Manager, gpa: Allocator, value: *Release) Allocator.Error!Text {
394 const active = self.installed.get(value.name) == value;
395 const suffix: Text = if (value.yanked and active) " (yanked, installed)" else if (value.yanked) " (yanked)" else if (active) " (installed)" else "";
396 return std.fmt.allocPrint(gpa, "{s}{s}", .{ try label(gpa, value), suffix });
397 }
398 fn resolve(self: *Manager, scratch: Allocator, requests: []const Pair, preferred: Set, audit: bool) Allocator.Error!Resolution {
399 var resolver: Resolver = .{ .manager = self, .gpa = scratch, .preferred = preferred, .audit = audit };
400 var initial: std.ArrayList(Requirement) = .empty;
401 for (requests) |pair| {
402 const chain = try scratch.alloc(Text, 1);
403 chain[0] = try std.fmt.allocPrint(scratch, "requested {s} {s}", .{ pair.name, pair.text });
404 try initial.append(scratch, .{ .pair = pair, .chain = chain });
405 }
406 if (try resolver.search(initial.items)) return .{ .selected = resolver.selected };
407 return .{ .failure = resolver.failure.? };
408 }
409 fn reportFailure(self: *Manager, lines: []const Text) Allocator.Error!void {
410 for (lines) |line| try self.out.line("{s}", .{line});
411 }
412 fn plan(self: *Manager, scratch: Allocator, next: Installation) Allocator.Error!void {
413 var arranged: Set = .empty;
414 var added: usize = 0;
415 var upgraded: usize = 0;
416 var downgraded: usize = 0;
417 var removed: usize = 0;
418 const names = try keys(scratch, next);
419 while (arranged.count() < next.count()) {
420 for (names) |name| {
421 if (arranged.contains(name)) continue;
422 const value = next.get(name).?;
423 var ready = true;
424 for (value.dependencies) |pair| if (!arranged.contains(pair.name)) {
425 ready = false;
426 break;
427 };
428 if (!ready) continue;
429 if (self.installed.get(name)) |old| {
430 if (old != value) {
431 const newer = compare(value.version, old.version) == .gt;
432 if (newer) upgraded += 1 else downgraded += 1;
433 try self.out.line("{s} {s} {s} -> {s}", .{ if (newer) @as(Text, "upgrade") else "downgrade", name, old.version.text, value.version.text });
434 }
435 } else {
436 try self.out.line("install {s}", .{try label(scratch, value)});
437 added += 1;
438 }
439 try arranged.put(scratch, name, {});
440 break;
441 }
442 }
443 var dropping: Set = .empty;
444 for (try keys(scratch, self.installed)) |name| if (!next.contains(name)) {
445 try dropping.put(scratch, name, {});
446 };
447 while (dropping.count() > 0) {
448 for (try keys(scratch, dropping)) |name| {
449 var ready = true;
450 for (try keys(scratch, dropping)) |other| {
451 for (self.installed.get(other).?.dependencies) |pair| {
452 if (equal(pair.name, name)) {
453 ready = false;
454 break;
455 }
456 }
457 if (!ready) break;
458 }
459 if (ready) {
460 try self.out.line("remove {s}", .{try label(scratch, self.installed.get(name).?)});
461 _ = dropping.remove(name);
462 removed += 1;
463 break;
464 }
465 }
466 }
467 if (added + upgraded + downgraded + removed == 0) try self.out.add("no changes\n") else try self.out.line("{d} installed, {d} upgraded, {d} downgraded, {d} removed", .{ added, upgraded, downgraded, removed });
468 }
469 fn update(self: *Manager, requests: []const Pair, installed: Installation, save: bool) Allocator.Error!void {
470 if (save) try self.history.append(self.gpa, .{ .requests = try self.gpa.dupe(Pair, self.requests), .installed = try clone(self.gpa, self.installed) });
471 self.requests = try self.gpa.dupe(Pair, requests);
472 self.installed = try clone(self.gpa, installed);
473 var keep: Set = .empty;
474 var iterator = self.held.keyIterator();
475 while (iterator.next()) |name| if (installed.contains(name.*)) {
476 try keep.put(self.gpa, name.*, {});
477 };
478 self.held = keep;
479 }
480 fn requested(self: *const Manager, name: Text) ?Text {

482 return null;

485 for (try keys(scratch, self.installed)) |candidate| {
486 for (self.installed.get(candidate).?.dependencies) |pair| if (equal(pair.name, name)) return candidate;

488 unreachable;

492 for (try keys(scratch, self.installed)) |name| try result.put(scratch, name, {});

495 fn tree(self: *Manager, scratch: Allocator, root: Text, seen: *Set, text: ?Text, prefix: Text, indent: Text) Allocator.Error!void {
496 const value = self.installed.get(root).?;
497 try self.out.print("{s}{s}", .{ prefix, try label(scratch, value) });
498 if (text) |condition| try self.out.print(" [{s}]", .{condition});
499 if (seen.contains(root)) {
500 try self.out.add(" (*)\n");
501 return;
502 }
503 try self.out.add("\n");
504 try seen.put(scratch, root, {});
505 for (value.dependencies, 0..) |pair, i| {
506 const last = i + 1 == value.dependencies.len;
507 const next_prefix = try std.fmt.allocPrint(scratch, "{s}{s}", .{ indent, if (last) @as(Text, "`-- ") else "|-- " });
508 const next_indent = try std.fmt.allocPrint(scratch, "{s}{s}", .{ indent, if (last) @as(Text, " ") else "| " });
509 try self.tree(scratch, pair.name, seen, pair.text, next_prefix, next_indent);
510 }
511 }
512 fn command(self: *Manager, scratch: Allocator, line: Text) Error!void {
513 const w = try split(scratch, line, " ", 0);
514 const cmd = w[0];
515 if (equal(cmd, "PKG")) {
516 const clauses = try split(scratch, line, "; ", 0);
517 const head = try split(scratch, clauses[0], " ", 0);
518 const version = try self.checkedVersion(head[2]);
519 if (self.find(head[1], head[2]) != null) return bad(self.gpa, "{s}@{s} is already registered", .{ head[1], head[2] });
520 var dependencies: std.ArrayList(Pair) = .empty;
521 var conflicts: std.ArrayList(Pair) = .empty;
522 for (clauses[1..]) |clause| {
523 const parts = try split(scratch, clause, " ", 3);
524 _ = try self.constraint(parts[2]);
525 const pair: Pair = .{ .name = parts[1], .text = parts[2] };
526 if (equal(parts[0], "DEP")) try dependencies.append(self.gpa, pair) else try conflicts.append(self.gpa, pair);
527 }
528 const value = try self.gpa.create(Release);
529 value.* = .{ .name = head[1], .version = version, .dependencies = dependencies.items, .conflicts = conflicts.items };
530 var versions = self.registry.get(value.name) orelse std.ArrayList(*Release).empty;
531 try versions.append(self.gpa, value);
532 std.mem.sort(*Release, versions.items, {}, releaseLess);
533 try self.registry.put(self.gpa, value.name, versions);
534 try self.out.line("registered {s}", .{try label(scratch, value)});
535 } else if (equal(cmd, "YANK") or equal(cmd, "INFO")) {
536 const value = self.find(w[1], w[2]) orelse return bad(self.gpa, "{s}@{s} is not registered", .{ w[1], w[2] });
537 if (equal(cmd, "YANK")) {
538 if (value.yanked) return bad(self.gpa, "{s} is already yanked", .{try label(scratch, value)});
539 value.yanked = true;
540 try self.out.line("yanked {s}", .{try label(scratch, value)});
541 } else {
542 try self.out.line("{s}", .{try self.display(scratch, value)});
543 for (value.dependencies) |pair| try self.out.line(" requires {s} {s}", .{ pair.name, pair.text });
544 for (value.conflicts) |pair| try self.out.line(" conflicts with {s} {s}", .{ pair.name, pair.text });
545 for (try self.applicable(scratch, value)) |id| try self.out.line(" advisory {s}", .{id});
546 }
547 } else if (equal(cmd, "COMPARE")) {
548 const a = try self.checkedVersion(w[1]);
549 const b = try self.checkedVersion(w[2]);
550 const relation: Text = switch (compare(a, b)) {
551 .lt => "<",
552 .eq => "=",
553 .gt => ">",
554 };
555 try self.out.line("{s} {s} {s}", .{ w[1], relation, w[2] });
556 } else if (equal(cmd, "MATCH")) {
557 const parts = try split(scratch, line, " ", 3);
558 const value = try self.checkedVersion(parts[1]);
559 const condition = try self.constraint(parts[2]);
560 try self.out.line("{s}", .{if (condition.matches(value)) @as(Text, "yes") else "no"});
561 } else if (equal(cmd, "RANGE")) {
562 const condition = try self.constraint(line[6..]);
563 try self.out.line("{s}", .{condition.normal});
564 } else if (equal(cmd, "VERSIONS") or equal(cmd, "INSTALL") or equal(cmd, "SIMULATE")) {
565 const parts = try split(scratch, line, " ", 3);
566 const text: Text = if (parts.len == 3) parts[2] else "*";
567 const condition = try self.constraint(text);
568 const versions = self.registry.get(w[1]) orelse return bad(self.gpa, "unknown package {s}", .{w[1]});
569 if (equal(cmd, "VERSIONS")) {
570 var count: usize = 0;
571 for (versions.items) |value| {
572 if (parts.len == 2 or condition.matches(value.version)) {
573 try self.out.line("{s}", .{try self.display(scratch, value)});
574 count += 1;
575 }
576 }
577 if (count == 0) try self.out.add("no matching versions\n");
578 } else {
579 var requests: std.ArrayList(Pair) = .empty;
580 try requests.appendSlice(scratch, self.requests);
581 var found = false;
582 for (requests.items) |*pair| {
583 if (equal(pair.name, w[1])) {
584 pair.text = text;
585 found = true;
586 break;
587 }
588 }
589 if (!found) try requests.append(scratch, .{ .name = w[1], .text = text });
590 const result = try self.resolve(scratch, requests.items, try self.allPreferred(scratch), false);
591 if (result.selected) |selected| {
592 try self.plan(scratch, selected);
593 if (equal(cmd, "INSTALL")) try self.update(requests.items, selected, true);
594 } else try self.reportFailure(result.failure);
595 }
596 } else if (equal(cmd, "DEPENDENTS")) {
597 const selected: ?Version = if (w.len == 3) try self.checkedVersion(w[2]) else null;
598 var count: usize = 0;
599 for (try keys(scratch, self.registry)) |name| {
600 for (self.registry.get(name).?.items) |value| {
601 for (value.dependencies) |pair| {
602 if (equal(pair.name, w[1]) and (selected == null or self.matches(selected.?, pair.text))) {
603 try self.out.line("{s} requires {s} {s}", .{ try label(scratch, value), w[1], pair.text });
604 count += 1;
605 }
606 }
607 }
608 }
609 if (count == 0) try self.out.add("no dependents\n");
610 } else if (equal(cmd, "REMOVE")) {
611 if (self.requested(w[1]) == null) {
612 if (self.installed.contains(w[1])) return bad(self.gpa, "{s} is required by {s}", .{ w[1], try self.parent(scratch, w[1]) });
613 return bad(self.gpa, "{s} is not installed", .{w[1]});
614 }
615 var requests: std.ArrayList(Pair) = .empty;
616 var keep: Installation = .empty;
617 var queue: std.ArrayList(Text) = .empty;
618 for (self.requests) |pair| {
619 if (!equal(pair.name, w[1])) {
620 try requests.append(scratch, pair);
621 try queue.append(scratch, pair.name);
622 }
623 }
624 var at: usize = 0;
625 while (at < queue.items.len) : (at += 1) {
626 const name = queue.items[at];
627 if (keep.contains(name)) continue;
628 const value = self.installed.get(name).?;
629 try keep.put(scratch, name, value);
630 for (value.dependencies) |pair| try queue.append(scratch, pair.name);
631 }
632 if (keep.contains(w[1])) {
633 try self.out.line("{s} is still required by {s}", .{ try label(scratch, self.installed.get(w[1]).?), try self.parent(scratch, w[1]) });
634 try self.update(requests.items, self.installed, true);
635 } else {
636 try self.plan(scratch, keep);
637 try self.update(requests.items, keep, true);
638 }
639 } else if (equal(cmd, "UPGRADE")) {
640 if (w.len == 2 and !self.installed.contains(w[1])) return bad(self.gpa, "{s} is not installed", .{w[1]});
641 var preferred: Set = .empty;
642 if (w.len == 2) {
643 preferred = try self.allPreferred(scratch);
644 _ = preferred.remove(w[1]);
645 }
646 const result = try self.resolve(scratch, self.requests, preferred, false);
647 if (result.selected) |selected| {
648 try self.plan(scratch, selected);
649 try self.update(self.requests, selected, true);
650 } else try self.reportFailure(result.failure);
651 } else if (equal(cmd, "ROLLBACK")) {
652 if (self.history.items.len == 0) return bad(self.gpa, "nothing to roll back", .{});
653 const saved = self.history.pop().?;
654 try self.plan(scratch, saved.installed);
655 try self.update(saved.requests, saved.installed, false);
656 } else if (equal(cmd, "HOLD")) {
657 const value = self.installed.get(w[1]) orelse return bad(self.gpa, "{s} is not installed", .{w[1]});
658 if (self.held.contains(w[1])) return bad(self.gpa, "{s} is already held", .{w[1]});
659 try self.held.put(self.gpa, w[1], {});
660 try self.out.line("held {s}", .{try label(scratch, value)});
661 } else if (equal(cmd, "UNHOLD")) {
662 if (!self.held.contains(w[1])) return bad(self.gpa, "{s} is not held", .{w[1]});
663 _ = self.held.remove(w[1]);
664 try self.out.line("released {s}", .{w[1]});
665 } else if (equal(cmd, "WHY")) {
666 const value = self.installed.get(w[1]) orelse return bad(self.gpa, "{s} is not installed", .{w[1]});
667 if (self.requested(w[1]) != null) {
668 try self.out.line("{s} is explicitly installed", .{try label(scratch, value)});
669 return;
670 }
671 var previous: std.StringHashMapUnmanaged(Text) = .empty;
672 var seen: Set = .empty;
673 var roots: Set = .empty;
674 var queue: std.ArrayList(Text) = .empty;
675 for (self.requests) |pair| try roots.put(scratch, pair.name, {});
676 for (try keys(scratch, roots)) |name| {
677 try queue.append(scratch, name);
678 try seen.put(scratch, name, {});
679 }
680 var at: usize = 0;
681 while (at < queue.items.len) : (at += 1) {
682 const name = queue.items[at];
683 var deps: Set = .empty;
684 for (self.installed.get(name).?.dependencies) |pair| try deps.put(scratch, pair.name, {});
685 for (try keys(scratch, deps)) |dep| {
686 if (!seen.contains(dep)) {
687 try seen.put(scratch, dep, {});
688 try previous.put(scratch, dep, name);
689 try queue.append(scratch, dep);
690 }
691 }
692 }
693 var path: std.ArrayList(Text) = .empty;
694 var current: ?Text = w[1];
695 while (current) |name| {
696 try path.append(scratch, try label(scratch, self.installed.get(name).?));
697 current = previous.get(name);
698 }
699 std.mem.reverse(Text, path.items);
700 try self.out.line("{s}", .{try join(scratch, path.items, " -> ")});
701 } else if (equal(cmd, "TREE")) {
702 if (w.len == 2 and !self.installed.contains(w[1])) return bad(self.gpa, "{s} is not installed", .{w[1]});
703 var roots: std.ArrayList(Text) = .empty;
704 if (w.len == 2) try roots.append(scratch, w[1]) else {
705 for (self.requests) |pair| try roots.append(scratch, pair.name);
706 std.mem.sort(Text, roots.items, {}, textLess);
707 }
708 if (roots.items.len == 0) try self.out.add("(empty)\n");
709 var seen: Set = .empty;
710 for (roots.items) |name| try self.tree(scratch, name, &seen, self.requested(name), "", "");
711 } else if (equal(cmd, "OUTDATED")) {
712 var count: usize = 0;
713 for (try keys(scratch, self.installed)) |name| {
714 const current = self.installed.get(name).?;
715 var conditions: std.ArrayList(Text) = .empty;
716 if (self.requested(name)) |text| try conditions.append(scratch, text);
717 var iterator = self.installed.valueIterator();
718 while (iterator.next()) |other| {
719 for (other.*.dependencies) |pair| if (equal(pair.name, name)) {
720 try conditions.append(scratch, pair.text);
721 };
722 }
723 var wanted: ?*Release = null;
724 var latest: ?*Release = null;
725 for (self.registry.get(name).?.items) |value| {
726 if (!value.yanked and value.version.pre.len == 0 and latest == null) latest = value;
727 if (wanted != null or (value.yanked and value != current)) continue;
728 var valid = true;
729 for (conditions.items) |text| if (!self.matches(value.version, text)) {
730 valid = false;
731 break;
732 };
733 if (valid) wanted = value;
734 }
735 if (wanted.? != current or (latest != null and compare(latest.?.version, current.version) == .gt)) {
736 try self.out.line("{s} {s} {s} {s}", .{ name, current.version.text, wanted.?.version.text, if (latest) |value| value.version.text else "-" });
737 count += 1;
738 }
739 }
740 if (count == 0) try self.out.add("all packages are up to date\n");
741 } else if (equal(cmd, "LOCK")) {
742 if (self.installed.count() == 0) try self.out.add("(empty)\n");
743 for (try keys(scratch, self.installed)) |name| try self.out.line("{s}", .{try label(scratch, self.installed.get(name).?)});
744 } else if (equal(cmd, "STATUS")) {
745 if (self.requests.len == 0) try self.out.add("no requests\n");
746 for (self.requests) |pair| try self.out.line("{s} {s} -> {s}", .{ pair.name, pair.text, self.installed.get(pair.name).?.version.text });
747 try self.out.line("{d} installed, {d} held, {d} snapshots", .{ self.installed.count(), self.held.count(), self.history.items.len });
748 } else if (equal(cmd, "ADVISORY")) {
749 const parts = try split(scratch, line, " ", 4);
750 if (self.advisories.contains(w[1])) return bad(self.gpa, "advisory {s} already exists", .{w[1]});
751 _ = try self.constraint(parts[3]);
752 try self.advisories.put(self.gpa, w[1], .{ .name = w[2], .text = parts[3] });
753 try self.out.line("advisory {s} recorded", .{w[1]});
754 } else if (equal(cmd, "AUDIT")) {
755 if (w.len == 2) {
756 const result = try self.resolve(scratch, self.requests, try self.allPreferred(scratch), true);
757 if (result.selected) |selected| {
758 try self.plan(scratch, selected);
759 try self.update(self.requests, selected, true);
760 } else try self.reportFailure(result.failure);
761 } else {
762 var count: usize = 0;
763 for (try keys(scratch, self.installed)) |name| {
764 const current = self.installed.get(name).?;
765 const ids = try self.applicable(scratch, current);
766 if (ids.len == 0) continue;
767 var fix: ?*Release = null;
768 const values = self.registry.get(name).?.items;
769 var i = values.len;
770 while (i > 0) {
771 i -= 1;
772 const value = values[i];
773 if (compare(value.version, current.version) == .gt and value.version.pre.len == 0 and !value.yanked and
774 (try self.applicable(scratch, value)).len == 0)
775 {
776 fix = value;
777 break;
778 }
779 }
780 const suffix = if (fix) |value| try std.fmt.allocPrint(scratch, "fix: {s}", .{value.version.text}) else "no fix";
781 try self.out.line("{s}: {s} ({s})", .{ try label(scratch, current), try join(scratch, ids, ", "), suffix });
782 count += 1;
783 }
784 if (count == 0) try self.out.add("no known vulnerabilities\n");
785 }
786 }
787 }
788};

790pub fn main(init: std.process.Init) !void {
791 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
792 defer arena.deinit();
793 const gpa = arena.allocator();
794 var input_buffer: [8192]u8 = undefined;
795 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
796 const text = try input.interface.allocRemaining(gpa, .unlimited);
797 var lines = std.mem.splitScalar(u8, text, '\n');
798 const count = integer(lines.next().?);
799 var out: Builder = .{ .gpa = gpa };
800 var manager: Manager = .{ .gpa = gpa, .out = &out };
801 for (0..count) |_| {
802 var scratch_arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
803 defer scratch_arena.deinit();
804 const scratch = scratch_arena.allocator();
805 manager.command(scratch, lines.next().?) catch |err| switch (err) {
806 error.Invalid => try out.line("error: {s}", .{failure}),
807 else => return err,
808 };
809 }
810 var output_buffer: [8192]u8 = undefined;
811 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
812 try output.interface.writeAll(out.bytes.items);
813 try output.interface.flush();
814}
