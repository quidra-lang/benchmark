11func number(_ s: String) -> Int? {
12 let b = Array(s.utf8)
13 return (1...4).contains(b.count) && b.allSatisfy {
14 (48...57).contains($0)
15 }
16 && (b.count == 1 || b[0] != 48) ? Int(s) : nil
17}

19struct Version: Comparable {
20 let core: [Int]
21 var pre = [String]()
22 var text: String {
23 core.map(String.init).joined(separator: ".")+(pre.isEmpty ? "" : "-"+pre.joined(separator: "."))
24 }
25 static func < (a: Version,b: Version) -> Bool {
26 for i in 0..<3 {
27 if a.core[i] != b.core[i] {
28 return a.core[i] < b.core[i]
29 }
30 }
31 if a.pre.isEmpty || b.pre.isEmpty {
32 return !a.pre.isEmpty && b.pre.isEmpty
33 }
34 for i in 0..<min(a.pre.count,b.pre.count) {
35 let x = a.pre[i],y = b.pre[i]
36 if x == y {
37 continue
38 }
39 let nx = x.utf8.allSatisfy {
40 (48...57).contains($0)
41 },ny = y.utf8.allSatisfy {
42 (48...57).contains($0)
43 }
44 if nx && ny {
45 return Int(x)! < Int(y)!
46 }
47 if nx != ny {
48 return nx
49 }
50 return x < y
51 }
52 return a.pre.count < b.pre.count
53 }
54}

56func version(_ s: String) -> Version? {
57 let p = s.components(separatedBy: "-")
58 if p.count < 1 || p.count > 2 {
59 return nil
60 }
61 let parts = p[0].components(separatedBy: ".")
62 if parts.count != 3 {
63 return nil
64 }
65 let core = parts.compactMap {
66 number($0)
67 }
68 if core.count != 3 {
69 return nil
70 }
71 let pre = p.count == 2 ? p[1].components(separatedBy: ".") : []
72 for id in pre {
73 if id.isEmpty || !id.utf8.allSatisfy( {
74 (97...122).contains($0) || (48...57).contains($0)
75 }) || (id.utf8.allSatisfy {
76 (48...57).contains($0)
77 }
78 && number(id) == nil) {
79 return nil
80 }
81 }
82 return Version(core: core,pre: pre)
83}

92struct Comparison {
93 let op: String
94 let version: Version
95 func accepts(_ v: Version) -> Bool {
96 switch op {
97 case "=":
98 return v == version
99 case ">":
100 return v > version
101 case ">=":
102 return v >= version
103 case "<":
104 return v < version
105 default:
106 return v <= version
107 }
108 }
109}
110
111struct Alternative {
112 let comparisons: [Comparison]
113 let prereleaseCores: [[Int]]
114}
115
116struct Constraint {
117 let alternatives: [Alternative]
118 func accepts(_ v: Version) -> Bool {
119 alternatives.contains {
120 (v.pre.isEmpty || $0.prereleaseCores.contains(v.core)) && $0.comparisons.allSatisfy {
121 $0.accepts(v)
122 }
123 }
124 }
125 func normal() -> String {
126 alternatives.map {
127 a in let s = a.comparisons.map {
128 $0.op+$0.version.text
129 }.joined(separator: " ")
130 return s.isEmpty ? "*" : s
131 }.joined(separator: " || ")
132 }
133}
134
135func constraint(_ text: String) throws -> Constraint {
136 func invalid() throws -> Never {
137 try fail("invalid constraint '\(text)'")
138 }
139 var alternatives = [Alternative]()
140 for alternative in text.components(separatedBy: " || ") {
141 var comparisons = [Comparison]()
142 var prereleaseCores = [[Int]]()
143 for term in alternative.components(separatedBy: " ") {
144 let op = [">=","<=",">","<","=","^","~"].first {
145 term.hasPrefix($0)
146 } ?? ""
147 let pattern = String(term.dropFirst(op.count))
148 let k: Int
149 let lower: Version
150 if pattern.contains("-") {
151 guard let v = version(pattern),!v.pre.isEmpty else {
152 try invalid()
153 }
154 lower = v
155 k = 3
156 prereleaseCores.append(v.core)
157 } else {
158 let parts = pattern.components(separatedBy: ".")
159 if parts.count < 1 || parts.count > 3 {
160 try invalid()
161 }
162 var core = [Int]()
163 var wildcard = false
164 for part in parts {
165 if part == "x" || part == "*" {
166 wildcard = true
167 } else {
168 guard let n = number(part),!wildcard else {
169 try invalid()
170 }
171 core.append(n)
172 }
173 }
174 k = core.count
175 while core.count < 3 {
176 core.append(0)
177 }
178 lower = Version(core: core)
179 }
180 func upper(_ i: Int) -> Version {
181 var core = lower.core
182 core[i-1] += 1
183 if i < 3 {
184 for j in i..<3 {
185 core[j] = 0
186 }
187 }
188 return Version(core: core)
189 }
190 func add(_ op: String,_ v: Version) {
191 comparisons.append(Comparison(op: op,version: v))
192 }
193 if k == 0 {
194 if op == ">" || op == "<" {
195 add("<",Version(core: [0,0,0],pre: ["0"]))
196 }
197 } else {
198 switch op {
199 case "","=":
200 if k == 3 {
201 add("=",lower)
202 } else {
203 add(">=",lower)
204 add("<",upper(k))
205 }
206 case ">":
207 add(k == 3 ? ">" : ">=",k == 3 ? lower : upper(k))
208 case "<=":
209 add(k == 3 ? "<=" : "<",k == 3 ? lower : upper(k))
210 case ">=","<":
211 add(op,lower)
212 case "~":
213 add(">=",lower)
214 add("<",upper(k >= 2 ? 2 : 1))
215 case "^":
216 add(">=",lower)
217 add("<",upper(lower.core[0] != 0 || k == 1 ? 1 : lower.core[1] != 0 || k == 2 ? 2 : 3))
218 default:
219 break
220 }
221 }
222 }
223 alternatives.append(Alternative(comparisons: comparisons,prereleaseCores: prereleaseCores))
224 }
225 return Constraint(alternatives: alternatives)
226}

256final class Manager {
257 var registry = [String:[Release]]()
258 var requests = [Rule]()
259 var installed = [String:Release]()
260 var held = Set<String>()
261 var advisories = [String:Rule]()
262 var history = [Snapshot]()
263 func releases(_ n: String) -> [Release] {
264 (registry[n] ?? []).sorted {
265 $0.version > $1.version
266 }
267 }
268 func registered(_ n: String,_ v: String) throws -> Release {
269 guard let r = registry[n]?.first(where: {
270 $0.version.text == v
271 }) else {
272 try fail("\(n)@\(v) is not registered")
273 }
274 return r
275 }
276 func checkInstalled(_ n: String) throws -> Release {
277 guard let r = installed[n] else {
278 try fail("\(n) is not installed")
279 }
280 return r
281 }
282 func applicable(_ n: String,_ v: Version) -> [String] {
283 advisories.keys.filter {
284 advisories[$0]!.name == n && advisories[$0]!.constraint.accepts(v)
285 }.sorted()
286 }
287 func display(_ n: String,_ r: Release) -> String {
288 var flags = [String]()
289 if r.yanked {
290 flags.append("yanked")
291 }
292 if installed[n]?.version == r.version {
293 flags.append("installed")
294 }
295 return "\(n)@\(r.version.text)"+(flags.isEmpty ? "" : " ("+flags.joined(separator: ", ")+")")
296 }
297 func resolve(_ newRequests: [Rule],_ preferred: Set<String>,_ audit: Bool) -> ([String:Release]?,[String]) {
298 var selected = [String:Release]()
299 var origins = [String:Requirement]()
300 var failureDepth = -1
301 var failure = [String]()
302 func chain(_ r: Requirement) -> [String] {
303 if let (n,v) = r.origin {
304 return chain(origins[n]!)+["\(n)@\(v.version.text) requires \(r.rule.name) \(r.rule.text)"]
305 }
306 return ["requested \(r.rule.name) \(r.rule.text)"]
307 }
308 func record(_ r: Requirement,_ reason: String,_ other: String? = nil) {
309 if selected.count > failureDepth {
310 failureDepth = selected.count
311 failure = ["UNSATISFIABLE"]+chain(r).map {
312 " "+$0
313 }+[" "+reason]
314 if let n = other {
315 failure += chain(origins[n]!).map {
316 " "+$0
317 }
318 }
319 }
320 }
321 func usable(_ n: String,_ r: Release) -> Bool {
322 (!r.yanked || installed[n]?.version == r.version) && (!audit || applicable(n,r.version).isEmpty)
323 }
324 func search(_ queue: [Requirement]) -> Bool {
325 if queue.isEmpty {
326 return true
327 }
328 let r = queue[0]
329 let n = r.rule.name
330 let rest = Array(queue.dropFirst())
331 if let old = selected[n] {
332 if r.rule.constraint.accepts(old.version) {
333 return search(rest)
334 }
335 record(r,"\(n)@\(old.version.text) was selected by:",n)
336 return false
337 }
338 var candidates = [Release]()
339 if held.contains(n) {
340 let fixed = installed[n]!
341 if !r.rule.constraint.accepts(fixed.version) || !usable(n,fixed) {
342 record(r,"\(n)@\(fixed.version.text) is held")
343 return false
344 }
345 candidates = [fixed]
346 } else {
347 candidates = releases(n).filter {
348 r.rule.constraint.accepts($0.version) && usable(n,$0)
349 }
350 if preferred.contains(n),let i = candidates.firstIndex(where: {
351 $0.version == installed[n]?.version
352 }) {
353 let first = candidates.remove(at: i)
354 candidates.insert(first,at: 0)
355 }
356 if candidates.isEmpty {
357 record(r,"no version of \(n) matches \(r.rule.text)")
358 return false
359 }
360 }
361 for candidate in candidates {
362 let conflict = selected.keys.sorted().first {
363 q in let w = selected[q]!
364 return candidate.conflicts.contains {
365 $0.name == q && $0.constraint.accepts(w.version)
366 }
367 || w.conflicts.contains {
368 $0.name == n && $0.constraint.accepts(candidate.version)
369 }
370 }
371 if let q = conflict {
372 record(r,"\(n)@\(candidate.version.text) conflicts with \(q)@\(selected[q]!.version.text), which was selected by:",q)
373 continue
374 }
375 selected[n] = candidate
376 origins[n] = r
377 if search(candidate.deps.map {
378 Requirement(rule: $0,origin: (n,candidate))
379 }+rest) {
380 return true
381 }
382 selected.removeValue(forKey: n)
383 origins.removeValue(forKey: n)
384 }
385 return false
386 }
387 return search(newRequests.map {
388 Requirement(rule: $0)
389 }) ? (selected,[]) : (nil,failure)
390 }
391 func plan(_ next: [String:Release]) -> [String] {
392 var out = [String]()
393 var counts = [0,0,0,0]
394 var remaining = Set(next.keys)
395 while !remaining.isEmpty {
396 let n = remaining.sorted().first {
397 next[$0]!.deps.allSatisfy {
398 !remaining.contains($0.name)
399 }
400 }!
401 remaining.remove(n)
402 let v = next[n]!.version
403 if let old = installed[n]?.version {
404 if v != old {
405 let up = v > old
406 out.append("\(up ? "upgrade" : "downgrade") \(n) \(old.text) -> \(v.text)")
407 counts[up ? 1 : 2] += 1
408 }
409 } else {
410 out.append("install \(n)@\(v.text)")
411 counts[0] += 1
412 }
413 }
414 remaining = Set(installed.keys).subtracting(next.keys)
415 while !remaining.isEmpty {
416 let n = remaining.sorted().first {
417 n in !remaining.contains {
418 q in installed[q]!.deps.contains {
419 $0.name == n
420 }
421 }
422 }!
423 remaining.remove(n)
424 out.append("remove \(n)@\(installed[n]!.version.text)")
425 counts[3] += 1
426 }
427 out.append(out.isEmpty ? "no changes" : "\(counts[0]) installed, \(counts[1]) upgraded, \(counts[2]) downgraded, \(counts[3]) removed")
428 return out
429 }
430 func update(_ newRequests: [Rule],_ next: [String:Release]) {
431 history.append(Snapshot(requests: requests,installed: installed))
432 requests = newRequests
433 installed = next
434 held.formIntersection(installed.keys)
435 }
436 func parent(_ n: String) -> String {
437 installed.keys.sorted().first {
438 installed[$0]!.deps.contains {
439 $0.name == n
440 }
441 }!
442 }
443 func command(_ line: String) throws -> [String] {
444 let clauses = line.components(separatedBy: "; ")
445 let w = clauses[0].components(separatedBy: " ")
446 let c = w[0]
447 let a = Array(w.dropFirst())
448 var out = [String]()
449 func rest(_ i: Int) -> String {
450 a.dropFirst(i).joined(separator: " ")
451 }
452 switch c {
453 case "PKG":
454 let v = try checkedVersion(a[1])
455 if registry[a[0]]?.contains(where: {
456 $0.version == v
457 }) == true {
458 try fail("\(a[0])@\(a[1]) is already registered")
459 }
460 var deps = [Rule]()
461 var conflicts = [Rule]()
462 for clause in clauses.dropFirst() {
463 let words = clause.components(separatedBy: " ")
464 let text = words.dropFirst(2).joined(separator: " ")
465 let rule = Rule(name: words[1],text: text,constraint: try constraint(text))
466 if words[0] == "DEP" {
467 deps.append(rule)
468 } else {
469 conflicts.append(rule)
470 }
471 }
472 registry[a[0],default: []].append(Release(v,deps,conflicts))
473 return ["registered \(a[0])@\(a[1])"]
474 case "YANK","INFO":
475 let r = try registered(a[0],a[1])
476 if c == "YANK" {
477 if r.yanked {
478 try fail("\(a[0])@\(a[1]) is already yanked")
479 }
480 r.yanked = true
481 return ["yanked \(a[0])@\(a[1])"]
482 }
483 out.append(display(a[0],r))
484 out += r.deps.map {
485 " requires \($0.name) \($0.text)"
486 }
487 out += r.conflicts.map {
488 " conflicts with \($0.name) \($0.text)"
489 }
490 out += applicable(a[0],r.version).map {
491 " advisory "+$0
492 }
493 return out
494 case "VERSIONS":
495 let con: Constraint?
496 if a.count > 1 {
497 con = try constraint(rest(1))
498 } else {
499 con = nil
500 }
501 if registry[a[0]] == nil {
502 try fail("unknown package \(a[0])")
503 }
504 out = releases(a[0]).filter {
505 con == nil || con!.accepts($0.version)
506 }.map {
507 display(a[0],$0)
508 }
509 return out.isEmpty ? ["no matching versions"] : out
510 case "DEPENDENTS":
511 let v: Version?
512 if a.count > 1 {
513 v = try checkedVersion(a[1])
514 } else {
515 v = nil
516 }
517 for n in registry.keys.sorted() {
518 for r in releases(n) {
519 for d in r.deps where d.name == a[0] && (v == nil || d.constraint.accepts(v!)) {
520 out.append("\(n)@\(r.version.text) requires \(d.name) \(d.text)")
521 }
522 }
523 }
524 return out.isEmpty ? ["no dependents"] : out
525 case "COMPARE":
526 let x = try checkedVersion(a[0])
527 let y = try checkedVersion(a[1])
528 return ["\(a[0]) \(x < y ? "<" : x > y ? ">" : "=") \(a[1])"]
529 case "MATCH":
530 let v = try checkedVersion(a[0])
531 let con = try constraint(rest(1))
532 return [con.accepts(v) ? "yes" : "no"]
533 case "RANGE":
534 return [try constraint(rest(0)).normal()]
535 case "INSTALL","SIMULATE":
536 let text = a.count > 1 ? rest(1) : "*"
537 let con = try constraint(text)
538 if registry[a[0]] == nil {
539 try fail("unknown package \(a[0])")
540 }
541 let rule = Rule(name: a[0],text: text,constraint: con)
542 var nextRequests = requests
543 if let i = nextRequests.firstIndex(where: {
544 $0.name == a[0]
545 }) {
546 nextRequests[i] = rule
547 } else {
548 nextRequests.append(rule)
549 }
550 let (next,failure) = resolve(nextRequests,Set(installed.keys),false)
551 guard let next = next else {
552 return failure
553 }
554 out = plan(next)
555 if c == "INSTALL" {
556 update(nextRequests,next)
557 }
558 return out
559 case "REMOVE":
560 let n = a[0]
561 if !requests.contains(where: {
562 $0.name == n
563 }) {
564 if installed[n] != nil {
565 try fail("\(n) is required by \(parent(n))")
566 }
567 try fail("\(n) is not installed")
568 }
569 let nextRequests = requests.filter {
570 $0.name != n
571 }
572 var keep = Set(nextRequests.map(\.name))
573 while true {
574 let next = keep.flatMap {
575 installed[$0]!.deps.map(\.name)
576 }
577 let before = keep.count
578 keep.formUnion(next)
579 if keep.count == before {
580 break
581 }
582 }
583 if keep.contains(n) {
584 out.append("\(n)@\(installed[n]!.version.text) is still required by \(parent(n))")
585 update(nextRequests,installed)
586 } else {
587 let next = installed.filter {
588 keep.contains($0.key)
589 }
590 out = plan(next)
591 update(nextRequests,next)
592 }
593 return out
594 case "UPGRADE":
595 if !a.isEmpty {
596 _ = try checkInstalled(a[0])
597 }
598 let preferred = a.isEmpty ? Set<String>() : Set(installed.keys).subtracting([a[0]])
599 let (next,failure) = resolve(requests,preferred,false)
600 guard let next = next else {
601 return failure
602 }
603 out = plan(next)
604 update(requests,next)
605 return out
606 case "ROLLBACK":
607 guard let snapshot = history.popLast() else {
608 try fail("nothing to roll back")
609 }
610 out = plan(snapshot.installed)
611 requests = snapshot.requests
612 installed = snapshot.installed
613 held.formIntersection(installed.keys)
614 return out
615 case "HOLD":
616 let r = try checkInstalled(a[0])
617 if held.contains(a[0]) {
618 try fail("\(a[0]) is already held")
619 }
620 held.insert(a[0])
621 return ["held \(a[0])@\(r.version.text)"]
622 case "UNHOLD":
623 if !held.contains(a[0]) {
624 try fail("\(a[0]) is not held")
625 }
626 held.remove(a[0])
627 return ["released \(a[0])"]
628 case "WHY":
629 let n = a[0]
630 let r = try checkInstalled(n)
631 let roots = Set(requests.map(\.name))
632 if roots.contains(n) {
633 return ["\(n)@\(r.version.text) is explicitly installed"]
634 }
635 var seen = roots
636 var queue = roots.sorted()
637 var head = 0
638 var parents = [String:String]()
639 while head < queue.count {
640 let u = queue[head]
641 head += 1
642 for v in installed[u]!.deps.map(\.name).sorted() where seen.insert(v).inserted {
643 parents[v] = u
644 queue.append(v)
645 }
646 }
647 var p = [n]
648 while !roots.contains(p.last!) {
649 p.append(parents[p.last!]!)
650 }
651 return [p.reversed().map {
652 "\($0)@\(installed[$0]!.version.text)"
653 }.joined(separator: " -> ")]
654 case "TREE":
655 if !a.isEmpty {
656 _ = try checkInstalled(a[0])
657 }
658 let roots = a.isEmpty ? requests.map(\.name).sorted() : [a[0]]
659 if roots.isEmpty {
660 return ["(empty)"]
661 }
662 var seen = Set<String>()
663 func visit(_ n: String,_ text: String?,_ prefix: String,_ indent: String) {
664 let r = installed[n]!
665 let body = prefix+"\(n)@\(r.version.text)"+(text.map {
666 " [\($0)]"
667 } ?? "")
668 if !seen.insert(n).inserted {
669 out.append(body+" (*)")
670 return
671 }
672 out.append(body)
673 for (i,d) in r.deps.enumerated() {
674 let last = i == r.deps.count-1
675 visit(d.name,d.text,indent+(last ? "`-- " : "|-- "),indent+(last ? " " : "| "))
676 }
677 }
678 for n in roots {
679 visit(n,requests.first {
680 $0.name == n
681 }?.text,"","")
682 }
683 return out
684 case "OUTDATED":
685 for n in installed.keys.sorted() {
686 let r = installed[n]!
687 let constraints = requests.filter {
688 $0.name == n
689 }.map(\.constraint)+installed.values.flatMap(\.deps).filter {
690 $0.name == n
691 }.map(\.constraint)
692 let versions = releases(n)
693 let wanted = versions.first {
694 candidate in (!candidate.yanked || candidate.version == r.version) && constraints.allSatisfy {
695 $0.accepts(candidate.version)
696 }
697 }!
698 let latest = versions.first {
699 !$0.yanked && $0.version.pre.isEmpty
700 }
701 if wanted.version != r.version || (latest != nil && latest!.version > r.version) {
702 out.append("\(n) \(r.version.text) \(wanted.version.text) \(latest?.version.text ?? "-")")
703 }
704 }
705 return out.isEmpty ? ["all packages are up to date"] : out
706 case "LOCK":
707 out = installed.keys.sorted().map {
708 "\($0)@\(installed[$0]!.version.text)"
709 }
710 return out.isEmpty ? ["(empty)"] : out
711 case "STATUS":
712 out = requests.map {
713 "\($0.name) \($0.text) -> \(installed[$0.name]!.version.text)"
714 }
715 if out.isEmpty {
716 out = ["no requests"]
717 }
718 out.append("\(installed.count) installed, \(held.count) held, \(history.count) snapshots")
719 return out
720 case "ADVISORY":
721 if advisories[a[0]] != nil {
722 try fail("advisory \(a[0]) already exists")
723 }
724 let text = rest(2)
725 let con = try constraint(text)
726 advisories[a[0]] = Rule(name: a[1],text: text,constraint: con)
727 return ["advisory \(a[0]) recorded"]
728 case "AUDIT":
729 if !a.isEmpty {
730 let (next,failure) = resolve(requests,Set(installed.keys),true)
731 guard let next = next else {
732 return failure
733 }
734 out = plan(next)
735 update(requests,next)
736 return out
737 }
738 for n in installed.keys.sorted() {
739 let r = installed[n]!
740 let ids = applicable(n,r.version)
741 if ids.isEmpty {
742 continue
743 }
744 let fix = releases(n).reversed().first {
745 $0.version > r.version && $0.version.pre.isEmpty && !$0.yanked && applicable(n,$0.version).isEmpty
746 }
747 out.append("\(n)@\(r.version.text): "+ids.joined(separator: ", ")+" "+(fix.map {
748 "(fix: \($0.version.text))"
749 } ?? "(no fix)"))
750 }
751 return out.isEmpty ? ["no known vulnerabilities"] : out
752 default:
753 return out
754 }
755 }
756}
