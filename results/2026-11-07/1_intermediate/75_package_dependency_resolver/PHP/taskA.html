2class Version
3{
4 public const NUMBER = '~\A(?:0|[1-9][0-9]{0,3})\z~';
5 public function __construct(public array $core, public ?array $pre = null) {}
6
7 public static function parse(string $text): ?Version
8 {
9 if (!preg_match('~\A(0|[1-9][0-9]{0,3})\.(0|[1-9][0-9]{0,3})\.(0|[1-9][0-9]{0,3})(?:-([a-z0-9]+(?:\.[a-z0-9]+)*))?\z~', $text, $m)) {
10 return null;
11 }
12 $pre = isset($m[4]) ? explode('.', $m[4]) : null;
13 foreach ($pre ?? [] as $part) {
14 if (ctype_digit($part) && !preg_match(self::NUMBER, $part)) {
15 return null;
16 }
17 }
18 return new Version([(int)$m[1], (int)$m[2], (int)$m[3]], $pre);
19 }
20
21 public function compare(Version $other): int
22 {
23 for ($i = 0; $i < 3; $i++) {
24 if ($this->core[$i] !== $other->core[$i]) {
25 return $this->core[$i] <=> $other->core[$i];
26 }
27 }
28 if ($this->pre === null || $other->pre === null) {
29 return $this->pre === $other->pre ? 0 : ($this->pre === null ? 1 : -1);
30 }
31 for ($i = 0; $i < min(count($this->pre), count($other->pre)); $i++) {
32 $a = $this->pre[$i];
33 $b = $other->pre[$i];
34 if ($a === $b) {
35 continue;
36 }
37 $an = ctype_digit($a);
38 $bn = ctype_digit($b);
39 return $an && $bn ? (int)$a <=> (int)$b : ($an === $bn ? strcmp($a, $b) : ($an ? -1 : 1));
40 }
41 return count($this->pre) <=> count($other->pre);
42 }
43
44 public function __toString(): string
45 {
46 return implode('.', $this->core) . ($this->pre === null ? '' : '-' . implode('.', $this->pre));
47 }
48}

50class Constraint
51{
52 public function __construct(public array $alternatives) {}
53
54 public static function parse(string $text): ?Constraint
55 {
56 $alternatives = [];
57 foreach (explode(' || ', $text) as $alternative) {
58 if ($alternative === '') {
59 return null;
60 }
61 $comparators = [];
62 $gates = [];
63 foreach (explode(' ', $alternative) as $term) {
64 $operator = '';
65 foreach (['>=', '<=', '>', '<', '=', '^', '~'] as $op) {
66 if (str_starts_with($term, $op)) {
67 $operator = $op;
68 break;
69 }
70 }
71 $pattern = substr($term, strlen($operator));
72 $version = Version::parse($pattern);
73 if ($version !== null && $version->pre !== null) {
74 $core = $version->core;
75 $k = 3;
76 $gates[] = $core;
77 } else {
78 $parts = explode('.', $pattern);
79 if (count($parts) < 1 || count($parts) > 3) {
80 return null;
81 }
82 $numbers = [];
83 $wildcard = false;
84 foreach ($parts as $part) {
85 if ($part === 'x' || $part === '*') {
86 $wildcard = true;
87 } elseif (!$wildcard && preg_match(Version::NUMBER, $part)) {
88 $numbers[] = (int)$part;
89 } else {
90 return null;
91 }
92 }
93 $k = count($numbers);
94 $core = array_pad($numbers, 3, 0);
95 $version = new Version($core);
96 }
97 $upper = function (int $i) use ($core): Version {
98 $values = $core;
99 $values[$i - 1]++;
100 for ($j = $i; $j < 3; $j++) {
101 $values[$j] = 0;
102 }
103 return new Version($values);
104 };
105 if ($k === 0) {
106 if ($operator === '>' || $operator === '<') {
107 $comparators[] = ['<', new Version([0, 0, 0], ['0'])];
108 }
109 continue;
110 }
111 switch ($operator) {
112 case '':
113 case '=':
114 if ($k === 3) {
115 $comparators[] = ['=', $version];
116 } else {
117 $comparators[] = ['>=', $version];
118 $comparators[] = ['<', $upper($k)];
119 }
120 break;
121 case '>':
122 $comparators[] = [$k === 3 ? '>' : '>=', $k === 3 ? $version : $upper($k)];
123 break;
124 case '<=':
125 $comparators[] = [$k === 3 ? '<=' : '<', $k === 3 ? $version : $upper($k)];
126 break;
127 case '>=':
128 case '<':
129 $comparators[] = [$operator, $version];
130 break;
131 case '~':
132 $comparators[] = ['>=', $version];
133 $comparators[] = ['<', $upper($k === 1 ? 1 : 2)];
134 break;
135 case '^':
136 $i = $core[0] !== 0 || $k === 1 ? 1 : ($core[1] !== 0 || $k === 2 ? 2 : 3);
137 $comparators[] = ['>=', $version];
138 $comparators[] = ['<', $upper($i)];
139 break;
140 }
141 }
142 $alternatives[] = [$comparators, $gates];
143 }
144 return new Constraint($alternatives);
145 }
146
147 public function matches(Version $version): bool
148 {
149 foreach ($this->alternatives as [$comparators, $gates]) {
150 if ($version->pre !== null && !in_array($version->core, $gates, true)) {
151 continue;
152 }
153 foreach ($comparators as [$operator, $bound]) {
154 $c = $version->compare($bound);
155 $ok = match ($operator) {
156 '=' => $c === 0, '>' => $c > 0, '>=' => $c >= 0, '<' => $c < 0, '<=' => $c <= 0
157 };
158 if (!$ok) {
159 continue 2;
160 }
161 }
162 return true;
163 }
164 return false;
165 }
166
167 public function __toString(): string
168 {
169 return implode(' || ', array_map(fn($a) => $a[0] ?
170 implode(' ', array_map(fn($c) => $c[0] . $c[1], $a[0])) : '*', $this->alternatives));
171 }

258 private function resolve(array $requests, array $preferred, bool $audit = false): array
259 {
260 $selected = [];
261 $reasons = [];
262 $failure = null;
263 $failureDepth = -1;
264 $usable = fn($n, $v) => (!$this->record($n, $v)->yanked || ($this->installed[$n] ?? null) === $v) &&
265 (!$audit || !$this->applicable($n, $v));
266 $save = function (array $chain, string $reason, ?array $other = null) use (&$selected, &$failure, &$failureDepth): void {
267 if (count($selected) > $failureDepth) {
268 $failureDepth = count($selected);
269 $failure = ['UNSATISFIABLE', ...array_map(fn($s) => " $s", $chain), " $reason"];
270 if ($other !== null) {
271 $failure = [...$failure, ...array_map(fn($s) => " $s", $other)];
272 }
273 }
274 };
275 $search = function (array $requirements) use (&$search, &$selected, &$reasons, $preferred, $usable, $save): bool {
276 if (!$requirements) {
277 return true;
278 }
279 [$name, $text, $chain] = $requirements[0];
280 $rest = array_slice($requirements, 1);
281 if (isset($selected[$name])) {
282 if ($this->matches($text, $selected[$name])) {
283 return $search($rest);
284 }
285 $save($chain, "$name@{$selected[$name]} was selected by:", $reasons[$name]);
286 return false;
287 }
288 if (isset($this->held[$name])) {
289 $value = $this->installed[$name];
290 if (!$this->matches($text, $value) || !$usable($name, $value)) {
291 $save($chain, "$name@$value is held");
292 return false;
293 }
294 $candidates = [$value];
295 } else {
296 $candidates = array_values(array_filter($this->versions($name), fn($v) => $this->matches($text, $v) && $usable($name, $v)));
297 $current = $this->installed[$name] ?? null;
298 if (in_array($name, $preferred, true) && in_array($current, $candidates, true)) {
299 $candidates = array_values(array_diff($candidates, [$current]));
300 array_unshift($candidates, $current);
301 }
302 if (!$candidates) {
303 $save($chain, "no version of $name matches $text");
304 return false;
305 }
306 }
307 foreach ($candidates as $value) {
308 $conflicting = null;
309 foreach ($this->keys($selected) as $q) {
310 foreach ($this->record($name, $value)->conflicts as [$d, $c]) {
311 if ($d === $q && $this->matches($c, $selected[$q])) {
312 $conflicting = $q;
313 break;
314 }
315 }
316 foreach ($this->record($q, $selected[$q])->conflicts as [$d, $c]) {
317 if ($d === $name && $this->matches($c, $value)) {
318 $conflicting = $q;
319 break;
320 }
321 }
322 if ($conflicting !== null) {
323 break;
324 }
325 }
326 if ($conflicting !== null) {
327 $save($chain, "$name@$value conflicts with $conflicting@{$selected[$conflicting]}, which was selected by:", $reasons[$conflicting]);
328 continue;
329 }
330 $selected[$name] = $value;
331 $reasons[$name] = $chain;
332 $extra = [];
333 foreach ($this->record($name, $value)->deps as [$d, $c]) {
334 $extra[] = [$d, $c, [...$chain, "$name@$value requires $d $c"]];
335 }
336 if ($search([...$extra, ...$rest])) {
337 return true;
338 }
339 unset($selected[$name], $reasons[$name]);
340 }
341 return false;
342 };
343 $requirements = array_map(fn($r) => [$r[0], $r[1], ["requested {$r[0]} {$r[1]}"]], $requests);
344 return $search($requirements) ? [$selected, null] : [null, $failure];
345 }

347 private function plan(array $new): void
348 {
349 $old = $this->installed;
350 $arranged = [];
351 $counts = ['install' => 0, 'upgrade' => 0, 'downgrade' => 0, 'remove' => 0];
352 while (count($arranged) < count($new)) {
353 foreach ($this->keys($new) as $name) {
354 if (isset($arranged[$name])) {
355 continue;
356 }
357 foreach ($this->dependencies($name, $new) as [$d, $_]) {
358 if (!isset($arranged[$d])) {
359 continue 2;
360 }
361 }
362 break;
363 }
364 $arranged[$name] = true;
365 $value = $new[$name];
366 if (!isset($old[$name])) {
367 echo "install $name@$value\n";
368 $counts['install']++;
369 } elseif ($old[$name] !== $value) {
370 $kind = Version::parse($value)->compare(Version::parse($old[$name])) > 0 ? 'upgrade' : 'downgrade';
371 echo "$kind $name {$old[$name]} -> $value\n";
372 $counts[$kind]++;
373 }
374 }
375 $remaining = array_diff_key($old, $new);
376 while ($remaining) {
377 foreach ($this->keys($remaining) as $name) {
378 foreach ($remaining as $q => $_) {
379 foreach ($this->dependencies($q, $old) as [$d, $_]) {
380 if ($d === $name) {
381 continue 3;
382 }
383 }
384 }
385 break;
386 }
387 echo "remove $name@{$old[$name]}\n";
388 $counts['remove']++;
389 unset($remaining[$name]);
390 }
391 if (array_sum($counts) === 0) {
392 echo "no changes\n";
393 } else {
394 echo "{$counts['install']} installed, {$counts['upgrade']} upgraded, {$counts['downgrade']} downgraded, {$counts['remove']} removed\n";
395 }
396 }

437 private function tree(?string $name): void
438 {
439 if ($name !== null) {
440 $this->requireInstalled($name);
441 }
442 $roots = $name !== null ? [$name] : array_column($this->requests, 0);
443 if ($name === null) {
444 sort($roots, SORT_STRING);
445 }
446 if (!$roots) {
447 echo "(empty)\n";
448 return;
449 }
450 $seen = [];
451 $visit = function (string $n, ?string $text, string $prefix, string $indent) use (&$visit, &$seen): void {
452 $body = "$prefix$n@{$this->installed[$n]}" . ($text === null ? '' : " [$text]");
453 if (isset($seen[$n])) {
454 echo "$body (*)\n";
455 return;
456 }
457 echo "$body\n";
458 $seen[$n] = true;
459 $deps = $this->dependencies($n);
460 foreach ($deps as $i => [$d, $c]) {
461 $last = $i === count($deps) - 1;
462 $visit($d, $c, $indent . ($last ? '`-- ' : '|-- '), $indent . ($last ? ' ' : '| '));
463 }
464 };
465 foreach ($roots as $root) {
466 $text = null;
467 foreach ($this->requests as [$n, $c]) {
468 if ($n === $root) {
469 $text = $c;
470 break;
471 }
472 }
473 $visit($root, $text, '', '');
474 }

477 public function execute(string $line): void
478 {
479 [$command, $rest] = array_pad(explode(' ', $line, 2), 2, '');
480 $args = $rest === '' ? [] : explode(' ', $rest);
481 $name = $args[0] ?? null;
482 try {
483 switch ($command) {
484 case 'PKG':
485 $clauses = explode('; ', $rest);
486 [$name, $value] = explode(' ', array_shift($clauses));
487 $parsed = $this->version($value);
488 if ($this->record($name, $value) !== null) {
489 throw new PackageError("$name@$value is already registered");
490 }
491 $deps = [];
492 $conflicts = [];
493 foreach ($clauses as $clause) {
494 [$kind, $other, $text] = explode(' ', $clause, 3);
495 $this->constraint($text);
496 if ($kind === 'DEP') {
497 $deps[] = [$other, $text];
498 } else {
499 $conflicts[] = [$other, $text];
500 }
501 }
502 $this->registry[$name][$value] = new PackageEntry($parsed, $deps, $conflicts);
503 echo "registered $name@$value\n";
504 break;
505 case 'YANK':
506 case 'INFO':
507 $value = $args[1];
508 $entry = $this->record($name, $value);
509 if ($entry === null) {
510 throw new PackageError("$name@$value is not registered");
511 }
512 if ($command === 'YANK') {
513 if ($entry->yanked) {
514 throw new PackageError("$name@$value is already yanked");
515 }
516 $entry->yanked = true;
517 echo "yanked $name@$value\n";
518 } else {
519 echo $this->display($name, $value), "\n";
520 foreach ($entry->deps as [$d, $c]) {
521 echo " requires $d $c\n";
522 }
523 foreach ($entry->conflicts as [$d, $c]) {
524 echo " conflicts with $d $c\n";
525 }
526 foreach ($this->applicable($name, $value) as $id) {
527 echo " advisory $id\n";
528 }
529 }
530 break;
531 case 'COMPARE':
532 $a = $this->version($name);
533 $b = $this->version($args[1]);
534 $comparison = $a->compare($b);
535 echo "$name ", $comparison < 0 ? '<' : ($comparison === 0 ? '=' : '>'), " {$args[1]}\n";
536 break;
537 case 'MATCH':
538 [$value, $text] = explode(' ', $rest, 2);
539 $parsed = $this->version($value);
540 echo $this->constraint($text)->matches($parsed) ? "yes\n" : "no\n";
541 break;
542 case 'RANGE':
543 echo $this->constraint($rest), "\n";
544 break;
545 case 'VERSIONS':
546 [$name, $text] = array_pad(explode(' ', $rest, 2), 2, null);
547 if ($text !== null) {
548 $this->constraint($text);
549 }
550 if (!isset($this->registry[$name])) {
551 throw new PackageError("unknown package $name");
552 }
553 $selected = array_filter($this->versions($name), fn($v) => $text === null || $this->matches($text, $v));
554 if (!$selected) {
555 echo "no matching versions\n";
556 }
557 foreach ($selected as $value) {
558 echo $this->display($name, $value), "\n";
559 }
560 break;
561 case 'DEPENDENTS':
562 $value = $args[1] ?? null;
563 if ($value !== null) {
564 $this->version($value);
565 }
566 $lines = [];
567 foreach ($this->keys($this->registry) as $n) {
568 foreach ($this->versions($n) as $v) {
569 foreach ($this->record($n, $v)->deps as [$d, $c]) {
570 if ($d === $name && ($value === null || $this->matches($c, $value))) {
571 $lines[] = "$n@$v requires $name $c";
572 }
573 }
574 }
575 }
576 echo implode("\n", $lines ?: ['no dependents']), "\n";
577 break;
578 case 'INSTALL':
579 case 'SIMULATE':
580 [$name, $text] = array_pad(explode(' ', $rest, 2), 2, '*');
581 $this->constraint($text);
582 if (!isset($this->registry[$name])) {
583 throw new PackageError("unknown package $name");
584 }
585 $requests = $this->requests;
586 $index = array_search($name, array_column($requests, 0), true);
587 if ($index === false) {
588 $requests[] = [$name, $text];
589 } else {
590 $requests[$index] = [$name, $text];
591 }
592 $this->resolveAndPlan($requests, array_keys($this->installed), false, $command === 'SIMULATE');
593 break;
594 case 'REMOVE':
595 if (!in_array($name, array_column($this->requests, 0), true)) {
596 throw new PackageError(isset($this->installed[$name]) ? "$name is required by " . $this->requiredBy($name) : "$name is not installed");
597 }
598 $requests = array_values(array_filter($this->requests, fn($r) => $r[0] !== $name));
599 $keep = array_column($requests, 0);
600 for ($i = 0; $i < count($keep); $i++) {
601 foreach ($this->dependencies($keep[$i]) as [$d, $_]) {
602 if (!in_array($d, $keep, true)) {
603 $keep[] = $d;
604 }
605 }
606 }
607 if (in_array($name, $keep, true)) {
608 echo "$name@{$this->installed[$name]} is still required by ", $this->requiredBy($name), "\n";
609 $this->update($requests, $this->installed);
610 } else {
611 $installed = array_intersect_key($this->installed, array_fill_keys($keep, true));
612 $this->plan($installed);
613 $this->update($requests, $installed);
614 }
615 break;
616 case 'UPGRADE':
617 if ($name !== null) {
618 $this->requireInstalled($name);
619 }
620 $this->resolveAndPlan($this->requests, $name === null ? [] : array_values(array_diff(array_keys($this->installed), [$name])));
621 break;
622 case 'ROLLBACK':
623 if (!$this->history) {
624 throw new PackageError('nothing to roll back');
625 }
626 [$requests, $installed] = array_pop($this->history);
627 $this->plan($installed);
628 $this->requests = $requests;
629 $this->installed = $installed;
630 $this->held = array_intersect_key($this->held, $installed);
631 break;
632 case 'HOLD':
633 $this->requireInstalled($name);
634 if (isset($this->held[$name])) {
635 throw new PackageError("$name is already held");
636 }
637 $this->held[$name] = true;
638 echo "held $name@{$this->installed[$name]}\n";
639 break;
640 case 'UNHOLD':
641 if (!isset($this->held[$name])) {
642 throw new PackageError("$name is not held");
643 }
644 unset($this->held[$name]);
645 echo "released $name\n";
646 break;
647 case 'WHY':
648 $this->requireInstalled($name);
649 $queue = array_column($this->requests, 0);
650 if (in_array($name, $queue, true)) {
651 echo "$name@{$this->installed[$name]} is explicitly installed\n";
652 break;
653 }
654 sort($queue, SORT_STRING);
655 $parent = array_fill_keys($queue, null);
656 for ($i = 0; $i < count($queue); $i++) {
657 $n = $queue[$i];
658 $deps = array_column($this->dependencies($n), 0);
659 sort($deps, SORT_STRING);
660 foreach ($deps as $d) {
661 if (!array_key_exists($d, $parent)) {
662 $parent[$d] = $n;
663 $queue[] = $d;
664 }
665 }
666 }
667 $path = [$name];
668 while ($parent[$path[0]] !== null) {
669 array_unshift($path, $parent[$path[0]]);
670 }
671 echo implode(' -> ', array_map(fn($n) => "$n@{$this->installed[$n]}", $path)), "\n";
672 break;
673 case 'TREE':
674 $this->tree($name);
675 break;
676 case 'OUTDATED':
677 $lines = [];
678 foreach ($this->keys($this->installed) as $n) {
679 $current = $this->installed[$n];
680 $constraints = [];
681 foreach ($this->requests as [$d, $c]) {
682 if ($d === $n) {
683 $constraints[] = $c;
684 }
685 }
686 foreach ($this->installed as $q => $_) {
687 foreach ($this->dependencies($q) as [$d, $c]) {
688 if ($d === $n) {
689 $constraints[] = $c;
690 }
691 }
692 }
693 $wanted = null;
694 $latest = null;
695 foreach ($this->versions($n) as $v) {
696 $entry = $this->record($n, $v);
697 if ($latest === null && !$entry->yanked && $entry->version->pre === null) {
698 $latest = $v;
699 }
700 if ($wanted !== null || ($entry->yanked && $v !== $current)) {
701 continue;
702 }
703 foreach ($constraints as $c) {
704 if (!$this->matches($c, $v)) {
705 continue 2;
706 }
707 }
708 $wanted = $v;
709 }
710 if ($wanted !== $current || ($latest !== null && Version::parse($latest)->compare(Version::parse($current)) > 0)) {
711 $lines[] = "$n $current $wanted " . ($latest ?? '-');
712 }
713 }
714 echo implode("\n", $lines ?: ['all packages are up to date']), "\n";
715 break;
716 case 'LOCK':
717 if (!$this->installed) {
718 echo "(empty)\n";
719 }
720 foreach ($this->keys($this->installed) as $n) {
721 echo "$n@{$this->installed[$n]}\n";
722 }
723 break;
724 case 'STATUS':
725 if (!$this->requests) {
726 echo "no requests\n";
727 }
728 foreach ($this->requests as [$n, $c]) {
729 echo "$n $c -> {$this->installed[$n]}\n";
730 }
731 echo count($this->installed), ' installed, ', count($this->held), ' held, ', count($this->history), " snapshots\n";
732 break;
733 case 'ADVISORY':
734 [$id, $n, $text] = explode(' ', $rest, 3);
735 if (isset($this->advisories[$id])) {
736 throw new PackageError("advisory $id already exists");
737 }
738 $this->constraint($text);
739 $this->advisories[$id] = [$n, $text];
740 echo "advisory $id recorded\n";
741 break;
742 case 'AUDIT':
743 if ($name === 'FIX') {
744 $this->resolveAndPlan($this->requests, array_keys($this->installed), true);
745 break;
746 }
747 $lines = [];
748 foreach ($this->keys($this->installed) as $n) {
749 $current = $this->installed[$n];
750 $ids = $this->applicable($n, $current);
751 if (!$ids) {
752 continue;
753 }
754 $fix = null;
755 foreach (array_reverse($this->versions($n)) as $v) {
756 $entry = $this->record($n, $v);
757 if ($entry->version->compare(Version::parse($current)) > 0 && $entry->version->pre === null &&
758 !$entry->yanked && !$this->applicable($n, $v)) {
759 $fix = $v;
760 break;
761 }
762 }
763 $lines[] = "$n@$current: " . implode(', ', $ids) . ' (' . ($fix === null ? 'no fix' : "fix: $fix") . ')';
764 }
765 echo implode("\n", $lines ?: ['no known vulnerabilities']), "\n";
766 break;
767 }
768 } catch (PackageError $error) {
769 echo 'error: ', $error->getMessage(), "\n";
770 }
771 }
772}
