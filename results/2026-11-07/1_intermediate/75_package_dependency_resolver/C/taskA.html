108static int split(char *text, char separator, char **parts) {
109 int count = 0;
110 parts[count++] = text;
111 for (char *p = text; *p; p++) if (*p == separator) {
112 *p = '\0';
113 parts[count++] = p + 1;
114 }
115 return count;
116}

118static bool parse_version(const char *text, Version *v) {
119 memset(v, 0, sizeof *v);
120 char copy[MAX_CONSTRAINT + 1];
121 if (strlen(text) > MAX_CONSTRAINT) return false;
122 strcpy(copy, text);
123 char *dash = strchr(copy, '-');
124 if (dash) {
125 *dash++ = '\0';
126 if (!*dash) return false;
127 strcpy(v->pre, dash);
128 char pre[MAX_CONSTRAINT + 1], *ids[MAX_CONSTRAINT + 1];
129 strcpy(pre, dash);
130 int count = split(pre, '.', ids);
131 for (int i = 0; i < count; i++) {
132 if (!*ids[i]) return false;
133 for (char *p = ids[i]; *p; p++) if (!(*p >= 'a' && *p <= 'z') && !(*p >= '0' && *p <= '9')) return false;
134 if (digits(ids[i]) && !number(ids[i])) return false;
135 }
136 }
137 char *parts[MAX_CONSTRAINT + 1];
138 int count = split(copy, '.', parts);
139 if (count != 3) return false;
140 for (int i = 0; i < 3; i++) {
141 if (!number(parts[i])) return false;
142 v->core[i] = atoi(parts[i]);
143 }
144 version_text(v);
145 return true;
146}

148static Version checked_version(const char *text) {

151 return v;
152}

154static int identifier_compare(const char *a, int na, const char *b, int nb) {
155 bool an = true, bn = true;
156 for (int i = 0; i < na; i++) if (a[i] < '0' || a[i] > '9') an = false;
157 for (int i = 0; i < nb; i++) if (b[i] < '0' || b[i] > '9') bn = false;
158 if (an != bn) return an ? -1 : 1;
159 if (an && na != nb) return na < nb ? -1 : 1;
160 int length = na < nb ? na : nb, c = strncmp(a, b, (size_t)length);
161 return c ? c : (na > nb) - (na < nb);
162}

164static int version_compare(const Version *a, const Version *b) {
165 for (int i = 0; i < 3; i++) if (a->core[i] != b->core[i]) return a->core[i] < b->core[i] ? -1 : 1;
166 if (!*a->pre || !*b->pre) return (!*a->pre) - (!*b->pre);
167 const char *x = a->pre, *y = b->pre;
168 while (*x && *y) {
169 const char *xe = strchr(x, '.'), *ye = strchr(y, '.');
170 int nx = xe ? (int)(xe - x) : (int)strlen(x), ny = ye ? (int)(ye - y) : (int)strlen(y);
171 int c = identifier_compare(x, nx, y, ny);
172 if (c) return c;
173 x += nx;
174 y += ny;
175 if (*x) x++;
176 if (*y) y++;
177 }
178 return (*x != '\0') - (*y != '\0');
179}

189static Constraint *parse_constraint(const char *text) {
190 Constraint *result = allocate(sizeof *result);
191 int branches = 1;
192 for (const char *p = text; (p = strstr(p, " || ")) != NULL; p += 4) branches++;
193 result->alternatives = allocate((size_t)branches * sizeof *result->alternatives);
194 char copy[MAX_CONSTRAINT + 1];
195 strcpy(copy, text);
196 char *branch = copy;
197 for (;;) {
198 char *next = strstr(branch, " || ");
199 if (next) {
200 *next = '\0';
201 next += 4;
202 }
203 Alternative *a = &result->alternatives[result->count++];
204 char *terms[MAX_CONSTRAINT + 1];
205 int term_count = split(branch, ' ', terms);
206 a->comparators = allocate((size_t)(2 * term_count) * sizeof *a->comparators);
207 a->pre_cores = allocate((size_t)term_count * sizeof *a->pre_cores);
208 for (int t = 0; t < term_count; t++) {
209 char *pattern = terms[t];
210 int op = -1;
211 if (!*pattern) return NULL;
212 const char *prefixes[] = {">=", "<=", ">", "<", "=", "^", "~"};
213 const int codes[] = {2, 4, 1, 3, 0, 5, 6};
214 for (int i = 0; i < 7; i++) if (!strncmp(pattern, prefixes[i], strlen(prefixes[i]))) {
215 op = codes[i];
216 pattern += strlen(prefixes[i]);
217 break;
218 }
219 Version lower;
220 int count;
221 if (parse_version(pattern, &lower) && *lower.pre) {
222 count = 3;
223 memcpy(a->pre_cores[a->pre_count++], lower.core, sizeof lower.core);
224 } else {
225 char parts_copy[MAX_CONSTRAINT + 1], *parts[MAX_CONSTRAINT + 1];
226 strcpy(parts_copy, pattern);
227 int parts_count = split(parts_copy, '.', parts);
228 if (parts_count < 1 || parts_count > 3) return NULL;
229 memset(&lower, 0, sizeof lower);
230 count = 0;
231 bool wildcard = false;
232 for (int i = 0; i < parts_count; i++) {
233 if (!strcmp(parts[i], "x") || !strcmp(parts[i], "*")) wildcard = true;
234 else {
235 if (wildcard || !number(parts[i])) return NULL;
236 lower.core[i] = atoi(parts[i]);
237 count++;
238 }
239 }
240 version_text(&lower);
241 }
242 if (count == 0) {
243 if (op == 1 || op == 3) {
244 Version smallest = {0};
245 strcpy(smallest.pre, "0");
246 version_text(&smallest);
247 add_comparator(a, 3, smallest);
248 }
249 continue;
250 }
251 Version upper[3];
252 for (int i = 0; i < 3; i++) {
253 upper[i] = lower;
254 upper[i].pre[0] = '\0';
255 upper[i].core[i]++;
256 for (int j = i + 1; j < 3; j++) upper[i].core[j] = 0;
257 version_text(&upper[i]);
258 }
259 if (op == -1 || op == 0) {
260 if (count == 3) add_comparator(a, 0, lower);
261 else {
262 add_comparator(a, 2, lower);
263 add_comparator(a, 3, upper[count - 1]);
264 }
265 } else if (op == 1) add_comparator(a, count == 3 ? 1 : 2, count == 3 ? lower : upper[count - 1]);
266 else if (op == 4) add_comparator(a, count == 3 ? 4 : 3, count == 3 ? lower : upper[count - 1]);
267 else if (op == 2 || op == 3) add_comparator(a, op, lower);
268 else if (op == 6) {
269 add_comparator(a, 2, lower);
270 add_comparator(a, 3, upper[count == 1 ? 0 : 1]);
271 } else {
272 int i = lower.core[0] != 0 || count == 1 ? 0 : lower.core[1] != 0 || count == 2 ? 1 : 2;
273 add_comparator(a, 2, lower);
274 add_comparator(a, 3, upper[i]);
275 }
276 }
277 if (!next) break;
278 branch = next;
279 }
280 return result;
281}

289static bool matches(Constraint *c, const Version *v) {
290 for (int i = 0; i < c->count; i++) {
291 Alternative *a = &c->alternatives[i];
292 bool allowed = !*v->pre;
293 for (int j = 0; j < a->pre_count; j++) if (same_core(a->pre_cores[j], v->core)) allowed = true;
294 if (!allowed) continue;
295 bool valid = true;
296 for (int j = 0; j < a->count; j++) {
297 Comparator *cmp = &a->comparators[j];
298 int order = version_compare(v, &cmp->version);
299 bool passes = cmp->op == 0 ? order == 0 : cmp->op == 1 ? order > 0 : cmp->op == 2 ? order >= 0 : cmp->op == 3 ? order < 0 : order <= 0;
300 if (!passes) {
301 valid = false;
302 break;
303 }
304 }
305 if (valid) return true;
306 }
307 return false;
308}

325 for (int i = 0; i < known_count; i++) order[i] = i;
326 for (int i = 1; i < known_count; i++) {
327 int v = order[i], j = i;
328 while (j && strcmp(known_names[order[j - 1]], known_names[v]) > 0) {
329 order[j] = order[j - 1];

416static void record_failure(Resolution *r, Requirement *requirement, int kind, Package *p, Package *q) {
417 int count = 0;
418 for (int i = 0; i < known_count; i++) count += r->selected[i] != NULL;
419 if (count <= r->failure_depth) return;
420 r->failure_depth = count;
421 clear_lines(&r->failure);
422 append_line(&r->failure, "UNSATISFIABLE");
423 requirement_chain(&r->failure, requirement, " ");
424 if (kind == 0) append_line(&r->failure, " %s@%s was selected by:", p->name, p->version.text);
425 else if (kind == 1) append_line(&r->failure, " %s@%s is held", p->name, p->version.text);
426 else if (kind == 2) append_line(&r->failure, " no version of %s matches %s", requirement->clause->name, requirement->clause->text);
427 else append_line(&r->failure, " %s@%s conflicts with %s@%s, which was selected by:", p->name, p->version.text, q->name, q->version.text);
428 if (kind == 0 || kind == 3) requirement_chain(&r->failure, r->reasons[known(kind == 0 ? p->name : q->name)], " ");
429}

431static bool conflicts(Package *p, Package *q) {
432 for (int i = 0; i < p->conflict_count; i++) if (!strcmp(p->conflicts[i].name, q->name) && matches(p->conflicts[i].constraint, &q->version)) return true;
433 for (int i = 0; i < q->conflict_count; i++) if (!strcmp(q->conflicts[i].name, p->name) && matches(q->conflicts[i].constraint, &p->version)) return true;
434 return false;
435}

437static bool search(Resolution *r, Requirement **pending, int count) {
438 if (!count) return true;
439 Requirement *requirement = pending[0];
440 Clause *d = requirement->clause;
441 int n = known(d->name);
442 if (n >= 0 && r->selected[n]) {
443 Package *p = r->selected[n];
444 if (matches(d->constraint, &p->version)) return search(r, pending + 1, count - 1);
445 record_failure(r, requirement, 0, p, NULL);
446 return false;
447 }
448 Package *candidates[MAX_VERSIONS];
449 int candidate_count = 0;
450 if (n >= 0 && held[n]) {
451 Package *p = installed[n];
452 if (!matches(d->constraint, &p->version) || !usable(r, p)) {
453 record_failure(r, requirement, 1, p, NULL);
454 return false;
455 }
456 candidates[candidate_count++] = p;
457 } else {
458 Package *all[MAX_VERSIONS];
459 int all_count = versions(d->name, all);
460 for (int i = 0; i < all_count; i++) if (matches(d->constraint, &all[i]->version) && usable(r, all[i])) candidates[candidate_count++] = all[i];
461 if (n >= 0 && r->preferred[n]) for (int i = 0; i < candidate_count; i++) if (candidates[i] == installed[n]) {
462 Package *p = candidates[i];
463 memmove(candidates + 1, candidates, (size_t)i * sizeof *candidates);
464 candidates[0] = p;
465 break;
466 }
467 if (!candidate_count) {
468 record_failure(r, requirement, 2, NULL, NULL);
469 return false;
470 }
471 }
472 int order[MAX_PACKAGES];
473 name_order(order);
474 for (int i = 0; i < candidate_count; i++) {
475 Package *p = candidates[i], *conflict = NULL;
476 for (int j = 0; j < known_count; j++) {
477 Package *q = r->selected[order[j]];
478 if (q && conflicts(p, q)) {
479 conflict = q;
480 break;
481 }
482 }
483 if (conflict) {
484 record_failure(r, requirement, 3, p, conflict);
485 continue;
486 }
487 r->selected[n] = p;
488 r->reasons[n] = requirement;
489 Requirement children[MAX_CLAUSES];
490 int next_count = p->dependency_count + count - 1;
491 Requirement **next = malloc((size_t)(next_count ? next_count : 1) * sizeof *next);
492 if (!next) exit(EXIT_FAILURE);
493 for (int j = 0; j < p->dependency_count; j++) {
494 children[j] = (Requirement){&p->dependencies[j], p, requirement};
495 next[j] = &children[j];
496 }
497 for (int j = 1; j < count; j++) next[p->dependency_count + j - 1] = pending[j];
498 bool success = search(r, next, next_count);
499 free(next);
500 if (success) return true;
501 r->selected[n] = NULL;
502 r->reasons[n] = NULL;
503 }
504 return false;
505}

507static void plan(Package **next) {
508 int order[MAX_PACKAGES];
509 name_order(order);
510 bool arranged[MAX_PACKAGES] = {false}, removed[MAX_PACKAGES] = {false};
511 int counts[4] = {0}, total = 0;
512 for (int i = 0; i < known_count; i++) {
513 total += next[i] != NULL;
514 removed[i] = installed[i] && !next[i];
515 }
516 for (int step = 0; step < total; step++) {
517 int n = -1;
518 for (int j = 0; j < known_count; j++) {
519 int i = order[j];
520 if (!next[i] || arranged[i]) continue;
521 bool ready = true;
522 for (int k = 0; k < next[i]->dependency_count; k++) if (!arranged[known(next[i]->dependencies[k].name)]) ready = false;
523 if (ready) {
524 n = i;
525 break;
526 }
527 }
528 arranged[n] = true;
529 Package *p = next[n], *old = installed[n];
530 if (!old) {
531 printf("install %s@%s\n", p->name, p->version.text);
532 counts[0]++;
533 } else if (old != p) {
534 bool newer = version_compare(&p->version, &old->version) > 0;
535 printf("%s %s %s -> %s\n", newer ? "upgrade" : "downgrade", p->name, old->version.text, p->version.text);
536 counts[newer ? 1 : 2]++;
537 }
538 }
539 for (;;) {
540 int n = -1;
541 for (int j = 0; j < known_count; j++) {
542 int i = order[j];
543 if (!removed[i]) continue;
544 bool ready = true;
545 for (int q = 0; q < known_count; q++) if (removed[q]) for (int k = 0; k < installed[q]->dependency_count; k++) if (!strcmp(installed[q]->dependencies[k].name, known_names[i])) ready = false;
546 if (ready) {
547 n = i;
548 break;
549 }
550 }
551 if (n < 0) break;
552 printf("remove %s@%s\n", installed[n]->name, installed[n]->version.text);
553 counts[3]++;
554 removed[n] = false;
555 }
556 if (counts[0] + counts[1] + counts[2] + counts[3]) printf("%d installed, %d upgraded, %d downgraded, %d removed\n", counts[0], counts[1], counts[2], counts[3]);
557 else puts("no changes");
558}

564static void update(Clause **list, int count, Package **next) {
565 Snapshot *snapshot = calloc(1, sizeof *snapshot);
566 if (!snapshot) exit(EXIT_FAILURE);
567 memcpy(snapshot->requests, requests, sizeof requests);
568 snapshot->request_count = request_count;
569 memcpy(snapshot->installed, installed, sizeof installed);
570 snapshot->previous = history;
571 history = snapshot;
572 history_count++;
573 memmove(requests, list, (size_t)count * sizeof *requests);
574 request_count = count;
575 memmove(installed, next, sizeof installed);
576 prune_holds();
577}

579static void resolve(Clause **list, int count, bool *preferred, bool audit, bool change) {
580 Resolution r = {0};
581 r.audit = audit;
582 r.failure_depth = -1;
583 memcpy(r.preferred, preferred, sizeof r.preferred);
584 Requirement requirements[MAX_PACKAGES], *pending[MAX_PACKAGES];
585 for (int i = 0; i < count; i++) {
586 requirements[i] = (Requirement){list[i], NULL, NULL};
587 pending[i] = &requirements[i];
588 }
589 if (search(&r, pending, count)) {
590 plan(r.selected);
591 if (change) update(list, count, r.selected);
592 } else for (int i = 0; i < r.failure.count; i++) puts(r.failure.items[i]);
593 clear_lines(&r.failure);
594}

608 }
609 return "";
610}

612static void reachable(Clause **list, int count, bool *seen) {
613 int queue[MAX_PACKAGES], head = 0, tail = 0;
614 memset(seen, 0, MAX_PACKAGES * sizeof *seen);
615 for (int i = 0; i < count; i++) {
616 int n = known(list[i]->name);
617 if (!seen[n]) {
618 seen[n] = true;
619 queue[tail++] = n;
620 }
621 }
622 while (head < tail) {
623 Package *p = installed[queue[head++]];
624 for (int i = 0; i < p->dependency_count; i++) {
625 int n = known(p->dependencies[i].name);
626 if (!seen[n]) {
627 seen[n] = true;
628 queue[tail++] = n;
629 }
630 }
631 }
632}

634static void tree(int n, const char *constraint, const char *prefix, const char *indent, bool *seen) {
635 Package *p = installed[n];
636 printf("%s%s@%s", prefix, p->name, p->version.text);
637 if (constraint) printf(" [%s]", constraint);
638 if (seen[n]) {
639 puts(" (*)");
640 return;
641 }
642 putchar('\n');
643 seen[n] = true;
644 for (int i = 0; i < p->dependency_count; i++) {
645 Clause *d = &p->dependencies[i];
646 bool last = i + 1 == p->dependency_count;
647 char child_prefix[4 * MAX_PACKAGES + 5], child_indent[4 * MAX_PACKAGES + 5];
648 snprintf(child_prefix, sizeof child_prefix, "%s%s", indent, last ? "`-- " : "|-- ");
649 snprintf(child_indent, sizeof child_indent, "%s%s", indent, last ? " " : "| ");
650 tree(known(d->name), d->text, child_prefix, child_indent, seen);
651 }
652}

654static char *token(char **rest) {
655 char *result = *rest, *space = strchr(result, ' ');
656 if (space) {
657 *space = '\0';
658 *rest = space + 1;
659 } else *rest = result + strlen(result);
660 return result;
661}

663static void command(char *line) {
664 char *rest = line, *c = token(&rest);
665 if (!strcmp(c, "PKG")) {
666 char *clauses[MAX_CLAUSES], *next = strstr(rest, "; ");
667 int count = 0;
668 if (next) {
669 *next = '\0';
670 next += 2;
671 }
672 char *name = token(&rest), *version = rest;
673 Version v = checked_version(version);
674 if (registered(name, version)) fail("%s@%s is already registered", name, version);
675 while (next) {
676 clauses[count++] = next;
677 char *p = strstr(next, "; ");
678 if (p) {
679 *p = '\0';
680 next = p + 2;
681 } else next = NULL;
682 }
683 Package *p = allocate(sizeof *p);
684 strcpy(p->name, name);
685 p->version = v;
686 for (int i = 0; i < count; i++) {
687 char *r = clauses[i], *kind = token(&r), *other = token(&r);
688 Clause *d = new_clause(other, r);
689 if (!strcmp(kind, "DEP")) p->dependencies[p->dependency_count++] = *d;
690 else p->conflicts[p->conflict_count++] = *d;
691 }
692 if (known(name) < 0) strcpy(known_names[known_count++], name);
693 registry[version_count++] = p;
694 printf("registered %s@%s\n", p->name, p->version.text);
695 } else if (!strcmp(c, "YANK") || !strcmp(c, "INFO")) {
696 char *name = token(&rest);
697 Package *p = checked_package(name, rest);
698 if (!strcmp(c, "YANK")) {
699 if (p->yanked) fail("%s@%s is already yanked", p->name, p->version.text);
700 p->yanked = true;
701 printf("yanked %s@%s\n", p->name, p->version.text);
702 } else {
703 display(p);
704 for (int i = 0; i < p->dependency_count; i++) printf(" requires %s %s\n", p->dependencies[i].name, p->dependencies[i].text);
705 for (int i = 0; i < p->conflict_count; i++) printf(" conflicts with %s %s\n", p->conflicts[i].name, p->conflicts[i].text);
706 for (int i = 0; i < advisory_count; i++) if (applies(&advisories[i], p)) printf(" advisory %s\n", advisories[i].id);
707 }
708 } else if (!strcmp(c, "VERSIONS") || !strcmp(c, "INSTALL") || !strcmp(c, "SIMULATE")) {
709 char *name = token(&rest);
710 Constraint *constraint = *rest ? checked_constraint(rest) : NULL;
711 if (known(name) < 0) fail("unknown package %s", name);
712 if (!strcmp(c, "VERSIONS")) {
713 Package *list[MAX_VERSIONS];
714 int count = versions(name, list), shown = 0;
715 for (int i = 0; i < count; i++) if (!constraint || matches(constraint, &list[i]->version)) {
716 display(list[i]);
717 shown++;
718 }
719 if (!shown) puts("no matching versions");
720 } else {
721 Clause *r;
722 if (!constraint) r = new_clause(name, "*");
723 else {
724 r = allocate(sizeof *r);
725 strcpy(r->name, name);
726 strcpy(r->text, rest);
727 r->constraint = constraint;
728 }
729 Clause *list[MAX_PACKAGES];
730 memcpy(list, requests, sizeof list);
731 int count = request_count, i = request_index(name);
732 if (i < 0) list[count++] = r;
733 else list[i] = r;
734 bool preferred[MAX_PACKAGES] = {false};
735 for (int j = 0; j < known_count; j++) preferred[j] = installed[j] != NULL;
736 resolve(list, count, preferred, false, !strcmp(c, "INSTALL"));
737 }
738 } else if (!strcmp(c, "DEPENDENTS")) {
739 char *name = token(&rest);
740 bool specified = *rest != '\0';
741 Version v = {0};
742 if (specified) v = checked_version(rest);
743 int order[MAX_PACKAGES], shown = 0;
744 name_order(order);
745 for (int n = 0; n < known_count; n++) {
746 Package *list[MAX_VERSIONS];
747 int count = versions(known_names[order[n]], list);
748 for (int j = 0; j < count; j++) for (int i = 0; i < list[j]->dependency_count; i++) {
749 Clause *d = &list[j]->dependencies[i];
750 if (!strcmp(d->name, name) && (!specified || matches(d->constraint, &v))) {
751 printf("%s@%s requires %s %s\n", list[j]->name, list[j]->version.text, name, d->text);
752 shown++;
753 }
754 }
755 }
756 if (!shown) puts("no dependents");
757 } else if (!strcmp(c, "COMPARE")) {
758 char *first = token(&rest);
759 Version a = checked_version(first), b = checked_version(rest);
760 int order = version_compare(&a, &b);
761 printf("%s %c %s\n", first, order < 0 ? '<' : order > 0 ? '>' : '=', rest);
762 } else if (!strcmp(c, "MATCH")) {
763 char *text = token(&rest);
764 Version v = checked_version(text);
765 Constraint *constraint = checked_constraint(rest);
766 puts(matches(constraint, &v) ? "yes" : "no");
767 } else if (!strcmp(c, "RANGE")) {
768 Constraint *constraint = checked_constraint(rest);
769 for (int i = 0; i < constraint->count; i++) {
770 Alternative *a = &constraint->alternatives[i];
771 if (i) fputs(" || ", stdout);
772 if (!a->count) putchar('*');
773 for (int j = 0; j < a->count; j++) {
774 if (j) putchar(' ');
775 printf("%s%s", operators[a->comparators[j].op], a->comparators[j].version.text);
776 }
777 }
778 putchar('\n');
779 } else if (!strcmp(c, "REMOVE")) {
780 int index = request_index(rest), n = known(rest);
781 if (index < 0) {
782 if (n >= 0 && installed[n]) fail("%s is required by %s", rest, required_by(rest));
783 fail("%s is not installed", rest);
784 }
785 Clause *list[MAX_PACKAGES];
786 int count = 0;
787 for (int i = 0; i < request_count; i++) if (i != index) list[count++] = requests[i];
788 bool keep[MAX_PACKAGES];
789 reachable(list, count, keep);
790 if (keep[n]) {
791 printf("%s@%s is still required by %s\n", rest, installed[n]->version.text, required_by(rest));
792 update(list, count, installed);
793 } else {
794 Package *next[MAX_PACKAGES] = {0};
795 for (int i = 0; i < known_count; i++) if (keep[i]) next[i] = installed[i];
796 plan(next);
797 update(list, count, next);
798 }
799 } else if (!strcmp(c, "UPGRADE")) {
800 int n = *rest ? known(rest) : -1;
801 if (*rest && (n < 0 || !installed[n])) fail("%s is not installed", rest);
802 bool preferred[MAX_PACKAGES] = {false};
803 if (*rest) for (int i = 0; i < known_count; i++) preferred[i] = installed[i] && i != n;
804 resolve(requests, request_count, preferred, false, true);
805 } else if (!strcmp(c, "ROLLBACK")) {
806 if (!history) fail("nothing to roll back");
807 Snapshot *snapshot = history;
808 history = snapshot->previous;
809 history_count--;
810 plan(snapshot->installed);
811 memcpy(requests, snapshot->requests, sizeof requests);
812 request_count = snapshot->request_count;
813 memcpy(installed, snapshot->installed, sizeof installed);
814 prune_holds();
815 free(snapshot);
816 } else if (!strcmp(c, "HOLD")) {
817 int n = known(rest);
818 if (n < 0 || !installed[n]) fail("%s is not installed", rest);
819 if (held[n]) fail("%s is already held", rest);
820 held[n] = true;
821 printf("held %s@%s\n", rest, installed[n]->version.text);
822 } else if (!strcmp(c, "UNHOLD")) {
823 int n = known(rest);
824 if (n < 0 || !held[n]) fail("%s is not held", rest);
825 held[n] = false;
826 printf("released %s\n", rest);
827 } else if (!strcmp(c, "WHY")) {
828 int n = known(rest);
829 if (n < 0 || !installed[n]) fail("%s is not installed", rest);
830 if (request_index(rest) >= 0) printf("%s@%s is explicitly installed\n", rest, installed[n]->version.text);
831 else {
832 int parent[MAX_PACKAGES], queue[MAX_PACKAGES], head = 0, tail = 0, order[MAX_PACKAGES];
833 name_order(order);
834 for (int i = 0; i < known_count; i++) parent[i] = -2;
835 for (int j = 0; j < known_count; j++) {
836 int i = order[j];
837 if (request_index(known_names[i]) >= 0) {
838 parent[i] = -1;
839 queue[tail++] = i;
840 }
841 }
842 while (head < tail) {
843 int u = queue[head++];
844 Package *p = installed[u];
845 for (int j = 0; j < known_count; j++) {
846 int v = order[j];
847 if (parent[v] != -2) continue;
848 for (int i = 0; i < p->dependency_count; i++) if (!strcmp(p->dependencies[i].name, known_names[v])) {
849 parent[v] = u;
850 queue[tail++] = v;
851 break;
852 }
853 }
854 }
855 int path[MAX_PACKAGES], count = 0;
856 for (int v = n; v >= 0; v = parent[v]) path[count++] = v;
857 for (int i = count - 1; i >= 0; i--) {
858 Package *p = installed[path[i]];
859 if (i != count - 1) fputs(" -> ", stdout);
860 printf("%s@%s", p->name, p->version.text);
861 }
862 putchar('\n');
863 }
864 } else if (!strcmp(c, "TREE")) {
865 bool seen[MAX_PACKAGES] = {false};
866 if (*rest) {
867 int n = known(rest);
868 if (n < 0 || !installed[n]) fail("%s is not installed", rest);
869 int r = request_index(rest);
870 tree(n, r < 0 ? NULL : requests[r]->text, "", "", seen);
871 } else if (!request_count) puts("(empty)");
872 else {
873 int order[MAX_PACKAGES];
874 name_order(order);
875 for (int j = 0; j < known_count; j++) {
876 int n = order[j], r = request_index(known_names[n]);
877 if (r >= 0) tree(n, requests[r]->text, "", "", seen);
878 }
879 }
880 } else if (!strcmp(c, "OUTDATED")) {
881 int order[MAX_PACKAGES], shown = 0;
882 name_order(order);
883 for (int j = 0; j < known_count; j++) {
884 int n = order[j];
885 Package *p = installed[n];
886 if (!p) continue;
887 Package *list[MAX_VERSIONS], *wanted = NULL, *latest = NULL;
888 int count = versions(p->name, list), r = request_index(p->name);
889 for (int k = 0; k < count; k++) {
890 Package *q = list[k];
891 if (!latest && !q->yanked && !*q->version.pre) latest = q;
892 if (wanted || (q->yanked && q != p)) continue;
893 bool valid = r < 0 || matches(requests[r]->constraint, &q->version);
894 for (int u = 0; u < known_count && valid; u++) if (installed[u]) for (int i = 0; i < installed[u]->dependency_count; i++) {
895 Clause *d = &installed[u]->dependencies[i];
896 if (!strcmp(d->name, p->name) && !matches(d->constraint, &q->version)) valid = false;
897 }
898 if (valid) wanted = q;
899 }
900 if (wanted != p || (latest && version_compare(&latest->version, &p->version) > 0)) {
901 printf("%s %s %s %s\n", p->name, p->version.text, wanted->version.text, latest ? latest->version.text : "-");
902 shown++;
903 }
904 }
905 if (!shown) puts("all packages are up to date");
906 } else if (!strcmp(c, "LOCK")) {
907 int order[MAX_PACKAGES], count = 0;
908 name_order(order);
909 for (int j = 0; j < known_count; j++) {
910 Package *p = installed[order[j]];
911 if (p) {
912 printf("%s@%s\n", p->name, p->version.text);
913 count++;
914 }
915 }
916 if (!count) puts("(empty)");
917 } else if (!strcmp(c, "STATUS")) {
918 for (int i = 0; i < request_count; i++) {
919 Clause *r = requests[i];
920 printf("%s %s -> %s\n", r->name, r->text, installed[known(r->name)]->version.text);
921 }
922 if (!request_count) puts("no requests");
923 int count = 0, held_count = 0;
924 for (int i = 0; i < known_count; i++) {
925 count += installed[i] != NULL;
926 held_count += held[i];
927 }
928 printf("%d installed, %d held, %d snapshots\n", count, held_count, history_count);
929 } else if (!strcmp(c, "ADVISORY")) {
930 char *id = token(&rest), *name = token(&rest);
931 for (int i = 0; i < advisory_count; i++) if (!strcmp(advisories[i].id, id)) fail("advisory %s already exists", id);
932 Clause *d = new_clause(name, rest);
933 Advisory *a = &advisories[advisory_count++];
934 strcpy(a->id, id);
935 a->clause = *d;
936 qsort(advisories, (size_t)advisory_count, sizeof *advisories, advisory_compare);
937 printf("advisory %s recorded\n", id);
938 } else if (!strcmp(c, "AUDIT")) {
939 if (!strcmp(rest, "FIX")) {
940 bool preferred[MAX_PACKAGES] = {false};
941 for (int i = 0; i < known_count; i++) preferred[i] = installed[i] != NULL;
942 resolve(requests, request_count, preferred, true, true);
943 } else {
944 int order[MAX_PACKAGES], shown = 0;
945 name_order(order);
946 for (int j = 0; j < known_count; j++) {
947 Package *p = installed[order[j]];
948 if (!p || !vulnerable(p)) continue;
949 printf("%s@%s: ", p->name, p->version.text);
950 bool first = true;
951 for (int i = 0; i < advisory_count; i++) if (applies(&advisories[i], p)) {
952 if (!first) fputs(", ", stdout);
953 fputs(advisories[i].id, stdout);
954 first = false;
955 }
956 Package *list[MAX_VERSIONS], *fix = NULL;
957 int count = versions(p->name, list);
958 for (int k = 0; k < count; k++) {
959 Package *q = list[k];
960 if (version_compare(&q->version, &p->version) > 0 && !*q->version.pre && !q->yanked && !vulnerable(q)) fix = q;
961 }
962 if (fix) printf(" (fix: %s)\n", fix->version.text);
963 else puts(" (no fix)");
964 shown++;
965 }
966 if (!shown) puts("no known vulnerabilities");
967 }
968 }
969}

971static char *read_line(void) {
972 size_t length = 0, capacity = 128;
973 char *line = malloc(capacity);
974 int ch;
975 if (!line) exit(EXIT_FAILURE);
976 while ((ch = getchar()) != EOF && ch != '\n') {
977 if (length + 1 == capacity) {
978 capacity *= 2;
979 char *p = realloc(line, capacity);
980 if (!p) exit(EXIT_FAILURE);
981 line = p;
982 }
983 line[length++] = (char)ch;
984 }
985 line[length] = '\0';
986 return line;
987}
988int main(void) {
989 char *line = read_line();
990 int n = atoi(line);
991 free(line);
992 for (int i = 0; i < n; i++) {
993 line = read_line();
994 if (setjmp(command_error) == 0) command(line);
995 free(line);
996 }
997 while (history) {
998 Snapshot *previous = history->previous;
999 free(history);
1000 history = previous;
1001 }
1002 while (allocations) {
1003 Allocation *next = allocations->next;
1004 free(allocations->pointer);
1005 free(allocations);
1006 allocations = next;
1007 }
1008 return 0;
