18function version(text) {
19 const match = /^([0-9]+)\.([0-9]+)\.([0-9]+)(?:-([a-z0-9]+(?:\.[a-z0-9]+)*))?$/.exec(text);
20 if (match === null || !match.slice(1, 4).every((part) => numberPattern.test(part))) return null;
21 const pre = match[4] === undefined ? [] : match[4].split(".");
22 if (pre.some((part) => /^[0-9]+$/.test(part) && !numberPattern.test(part))) return null;
23 return { text, core: match.slice(1, 4).map(Number), pre };
24}
25function requireVersion(text) {
26 const result = version(text);
27 if (result === null) fail(`invalid version '${text}'`);
28 return result;

30function compareVersions(a, b) {
31 for (let i = 0; i < 3; i++) if (a.core[i] !== b.core[i]) return a.core[i] - b.core[i];
32 if (!a.pre.length || !b.pre.length) return (b.pre.length ? 1 : 0) - (a.pre.length ? 1 : 0);
33 for (let i = 0; i < Math.min(a.pre.length, b.pre.length); i++) {
34 const x = a.pre[i],
35 y = b.pre[i],
36 nx = /^[0-9]+$/.test(x),
37 ny = /^[0-9]+$/.test(y);
38 const order =
39 nx && ny ? Number(x) - Number(y) : nx !== ny ? (nx ? -1 : 1) : compareText(x, y);
40 if (order !== 0) return order;
41 }
42 return a.pre.length - b.pre.length;
43}
44function synthetic(core, pre = []) {
45 return { core, pre, text: core.join(".") + (pre.length ? "-" + pre.join(".") : "") };

47function parseConstraint(text) {
48 const alternatives = [];
49 for (const alternative of text.split(" || ")) {
50 const comparators = [],
51 prereleases = new Set();
52 for (const term of alternative.split(" ")) {
53 if (!term) return null;
54 const op =
55 [">=", "<=", ">", "<", "=", "^", "~"].find((op) => term.startsWith(op)) ?? "";
56 const pattern = term.slice(op.length),
57 prerelease = version(pattern);
58 let core = [0, 0, 0],
59 count = 0,
60 lower;
61 if (prerelease !== null && prerelease.pre.length) {
62 core = prerelease.core;
63 count = 3;
64 lower = prerelease;
65 prereleases.add(core.join("."));
66 } else {
67 const parts = pattern.split(".");
68 if (parts.length < 1 || parts.length > 3) return null;
69 let wildcard = false;
70 for (const part of parts) {
71 if (part === "x" || part === "*") wildcard = true;
72 else if (!wildcard && numberPattern.test(part)) core[count++] = Number(part);
73 else return null;
74 }
75 lower = synthetic(core);
76 }
77 const add = (operator, bound) => comparators.push({ op: operator, bound });
78 const upper = (index) =>
79 synthetic(
80 core.map((value, i) =>
81 i < index - 1 ? value : i === index - 1 ? value + 1 : 0,
82 ),
83 );
84 if (count === 0) {
85 if (op === ">" || op === "<") add("<", synthetic([0, 0, 0], ["0"]));
86 } else if (op === "" || op === "=") {
87 if (count === 3) add("=", lower);
88 else {
89 add(">=", lower);
90 add("<", upper(count));
91 }
92 } else if (op === ">")
93 add(count === 3 ? ">" : ">=", count === 3 ? lower : upper(count));
94 else if (op === "<=") add(count === 3 ? "<=" : "<", count === 3 ? lower : upper(count));
95 else if (op === ">=" || op === "<") add(op, lower);
96 else if (op === "~") {
97 add(">=", lower);
98 add("<", upper(count === 1 ? 1 : 2));
99 } else if (op === "^") {
100 add(">=", lower);
101 add(
102 "<",
103 upper(core[0] !== 0 || count === 1 ? 1 : core[1] !== 0 || count === 2 ? 2 : 3),
104 );
105 } else return null;
106 }
107 alternatives.push({ comparators, prereleases });
108 }
109 return {
110 alternatives,
111 normal: alternatives
112 .map(
113 (alternative) =>
114 alternative.comparators
115 .map((comparator) => comparator.op + comparator.bound.text)
116 .join(" ") || "*",
117 )
118 .join(" || "),
119 };
120}
121function constraint(text) {
122 if (constraintCache.has(text)) return constraintCache.get(text);
123 const result = parseConstraint(text);
124 if (result === null) fail(`invalid constraint '${text}'`);
125 constraintCache.set(text, result);
126 return result;
127}
128function satisfies(value, range) {
129 return range.alternatives.some(
130 (alternative) =>
131 (!value.pre.length || alternative.prereleases.has(value.core.join("."))) &&
132 alternative.comparators.every((comparator) => {
133 const order = compareVersions(value, comparator.bound);
134 return comparator.op === "="
135 ? order === 0
136 : comparator.op === ">"
137 ? order > 0
138 : comparator.op === ">="
139 ? order >= 0
140 : comparator.op === "<"
141 ? order < 0
142 : order <= 0;
143 }),
144 );
145}

178function requirementChain(requirement) {
179 const result = [];
180 for (let current = requirement; current !== null; current = current.parent)
181 result.push(

188function resolve(wanted, preferred, audit) {
189 const selected = new Map();
190 let failure = null;
191 const usable = (item) =>
192 (!item.yanked || installed.get(item.name)?.version.text === item.version.text) &&
193 (!audit || !applicable(item).length);
194 function record(requirement, reason, other = null) {
195 if (failure === null || selected.size > failure.size)
196 failure = {
197 size: selected.size,
198 chain: requirementChain(requirement),
199 reason,
200 other: other === null ? null : requirementChain(other),
201 };

203 function search(requirements) {
204 if (!requirements.length) return true;
205 const [requirement, ...rest] = requirements,
206 name = requirement.name;
207 const existing = selected.get(name);
208 if (existing !== undefined) {
209 if (satisfies(existing.item.version, requirement.range)) return search(rest);
210 record(
211 requirement,
212 `${packageText(existing.item)} was selected by:`,
213 existing.requirement,
214 );
215 return false;
216 }
217 let candidates;
218 if (held.has(name)) {
219 const item = installed.get(name);
220 if (!satisfies(item.version, requirement.range) || !usable(item)) {
221 record(requirement, `${packageText(item)} is held`);
222 return false;
223 }
224 candidates = [item];
225 } else {
226 candidates = versions(name).filter(
227 (item) => satisfies(item.version, requirement.range) && usable(item),
228 );
229 if (preferred.has(name)) {
230 const at = candidates.findIndex(
231 (item) => item.version.text === installed.get(name)?.version.text,
232 );
233 if (at >= 0) candidates.unshift(candidates.splice(at, 1)[0]);
234 }
235 if (!candidates.length) {
236 record(requirement, `no version of ${name} matches ${requirement.raw}`);
237 return false;
238 }
239 }
240 for (const item of candidates) {
241 let conflict = null;
242 for (const otherName of [...selected.keys()].sort()) {
243 const other = selected.get(otherName);
244 if (
245 item.conflicts.some(
246 (dep) => dep.name === otherName && satisfies(other.item.version, dep.range),
247 ) ||
248 other.item.conflicts.some(
249 (dep) => dep.name === name && satisfies(item.version, dep.range),
250 )
251 ) {
252 conflict = other;
253 break;
254 }
255 }
256 if (conflict !== null) {
257 record(
258 requirement,
259 `${packageText(item)} conflicts with ${packageText(conflict.item)}, which was selected by:`,
260 conflict.requirement,
261 );
262 continue;
263 }
264 selected.set(name, { item, requirement });
265 const dependencies = item.deps.map((dep) => ({
266 ...dep,
267 origin: packageText(item),
268 parent: requirement,
269 }));
270 if (search([...dependencies, ...rest])) return true;
271 selected.delete(name);
272 }
273 return false;
274 }
275 const requirements = wanted.map((request) => ({ ...request, origin: null, parent: null }));
276 if (search(requirements))
277 return {
278 installation: new Map([...selected].map(([name, entry]) => [name, entry.item])),
279 failure: null,
280 };
281 return { installation: null, failure };
282}

291function plan(next) {
292 const remaining = new Set([...next.keys()].sort()),
293 arranged = new Set(),
294 counts = [0, 0, 0, 0],
295 lines = [];
296 while (remaining.size) {
297 const name = [...remaining].find((name) =>
298 next.get(name).deps.every((dep) => arranged.has(dep.name)),
299 ),
300 item = next.get(name),
301 old = installed.get(name);
302 if (old === undefined) {
303 lines.push("install " + packageText(item));
304 counts[0]++;
305 } else if (old.version.text !== item.version.text) {
306 const upgrading = compareVersions(item.version, old.version) > 0;
307 lines.push(
308 `${upgrading ? "upgrade" : "downgrade"} ${name} ${old.version.text} -> ${item.version.text}`,
309 );
310 counts[upgrading ? 1 : 2]++;
311 }
312 remaining.delete(name);
313 arranged.add(name);
314 }
315 const removals = new Set([...installed.keys()].filter((name) => !next.has(name)).sort());
316 while (removals.size) {
317 const name = [...removals].find(
318 (name) =>
319 ![...removals].some((other) =>
320 installed.get(other).deps.some((dep) => dep.name === name),
321 ),
322 );
323 lines.push("remove " + packageText(installed.get(name)));
324 counts[3]++;
325 removals.delete(name);
326 }
327 output.push(
328 ...(lines.length
329 ? [
330 ...lines,
331 `${counts[0]} installed, ${counts[1]} upgraded, ${counts[2]} downgraded, ${counts[3]} removed`,
332 ]
333 : ["no changes"]),
334 );
335}

362function cut(text) {
363 const at = text.indexOf(" ");

366function execute(commandLine) {
367 const [command, argument] = cut(commandLine),
368 words = argument === null ? [] : argument.split(" ");
369 const name = words[0];
370 if (command === "PKG") {
371 const parts = commandLine.split("; "),
372 main = parts[0].split(" "),
373 packageName = main[1],
374 text = main[2],
375 parsed = requireVersion(text);
376 if (registry.get(packageName)?.has(text))
377 fail(`${packageName}@${text} is already registered`);
378 const deps = [],
379 conflicts = [];
380 for (const clause of parts.slice(1)) {
381 const [kind, rest] = cut(clause),
382 [other, raw] = cut(rest),
383 range = constraint(raw);
384 (kind === "DEP" ? deps : conflicts).push({ name: other, raw, range });
385 }
386 if (!registry.has(packageName)) registry.set(packageName, new Map());
387 registry
388 .get(packageName)
389 .set(text, { name: packageName, version: parsed, deps, conflicts, yanked: false });
390 output.push(`registered ${packageName}@${text}`);
391 return;
392 }
393 if (command === "YANK" || command === "INFO") {
394 const item = registered(name, words[1]);
395 if (command === "YANK") {
396 if (item.yanked) fail(packageText(item) + " is already yanked");
397 item.yanked = true;
398 output.push("yanked " + packageText(item));
399 } else
400 output.push(
401 display(item),
402 ...item.deps.map((dep) => ` requires ${dep.name} ${dep.raw}`),
403 ...item.conflicts.map((dep) => ` conflicts with ${dep.name} ${dep.raw}`),
404 ...applicable(item).map((id) => " advisory " + id),
405 );
406 return;
407 }
408 if (command === "COMPARE") {
409 const a = requireVersion(words[0]),
410 b = requireVersion(words[1]),
411 order = compareVersions(a, b);
412 output.push(`${words[0]} ${order < 0 ? "<" : order > 0 ? ">" : "="} ${words[1]}`);
413 return;
414 }
415 if (command === "MATCH") {
416 const [text, raw] = cut(argument),
417 parsed = requireVersion(text),
418 range = constraint(raw);
419 output.push(satisfies(parsed, range) ? "yes" : "no");
420 return;
421 }
422 if (command === "RANGE") {
423 output.push(constraint(argument).normal);
424 return;
425 }
426 if (command === "VERSIONS" || command === "INSTALL" || command === "SIMULATE") {
427 const [packageName, rest] = cut(argument),
428 raw = rest ?? "*";
429 const range = constraint(raw);
430 if (!registry.has(packageName)) fail("unknown package " + packageName);
431 if (command === "VERSIONS") {
432 const selected = versions(packageName).filter(
433 (item) => rest === null || satisfies(item.version, range),
434 );
435 output.push(...(selected.length ? selected.map(display) : ["no matching versions"]));
436 return;
437 }
438 const wanted = requests.map((request) => ({ ...request })),
439 at = wanted.findIndex((request) => request.name === packageName),
440 request = { name: packageName, raw, range };
441 if (at >= 0) wanted[at] = request;
442 else wanted.push(request);
443 const result = resolve(wanted, new Set(installed.keys()), false);
444 if (result.installation === null) reportFailure(result.failure);
445 else {
446 plan(result.installation);
447 if (command === "INSTALL") update(wanted, result.installation);
448 }
449 return;
450 }
451 if (command === "DEPENDENTS") {
452 const parsed = words.length > 1 ? requireVersion(words[1]) : null,
453 lines = [];
454 for (const other of [...registry.keys()].sort())
455 for (const item of versions(other))
456 for (const dep of item.deps)
457 if (dep.name === name && (parsed === null || satisfies(parsed, dep.range)))
458 lines.push(`${packageText(item)} requires ${name} ${dep.raw}`);
459 output.push(...(lines.length ? lines : ["no dependents"]));
460 return;
461 }
462 if (command === "REMOVE") {
463 const at = requests.findIndex((request) => request.name === name);
464 if (at < 0)
465 fail(
466 installed.has(name)
467 ? `${name} is required by ${requiredBy(name)}`
468 : `${name} is not installed`,
469 );
470 const wanted = requests.filter((request) => request.name !== name),
471 keep = reach(wanted);
472 if (keep.has(name)) {
473 output.push(
474 `${packageText(installed.get(name))} is still required by ${requiredBy(name)}`,
475 );
476 update(wanted, installed);
477 } else {
478 const next = new Map([...installed].filter(([packageName]) => keep.has(packageName)));
479 plan(next);
480 update(wanted, next);
481 }
482 return;
483 }
484 if (command === "UPGRADE" || (command === "AUDIT" && argument === "FIX")) {
485 if (command === "UPGRADE" && argument !== null && !installed.has(name))
486 fail(`${name} is not installed`);
487 const preferred =
488 command === "AUDIT" || argument !== null
489 ? new Set(
490 [...installed.keys()].filter(
491 (packageName) => command === "AUDIT" || packageName !== name,
492 ),
493 )
494 : new Set();
495 const result = resolve(requests, preferred, command === "AUDIT");
496 if (result.installation === null) reportFailure(result.failure);
497 else {
498 plan(result.installation);
499 update(requests, result.installation);
500 }
501 return;
502 }
503 if (command === "ROLLBACK") {
504 if (!history.length) fail("nothing to roll back");
505 const snapshot = history.pop();
506 plan(snapshot.installed);
507 update(snapshot.requests, snapshot.installed, false);
508 return;
509 }
510 if (command === "HOLD") {
511 if (!installed.has(name)) fail(`${name} is not installed`);
512 if (held.has(name)) fail(`${name} is already held`);
513 held.add(name);
514 output.push("held " + packageText(installed.get(name)));
515 return;
516 }
517 if (command === "UNHOLD") {
518 if (!held.has(name)) fail(`${name} is not held`);
519 held.delete(name);
520 output.push("released " + name);
521 return;
522 }
523 if (command === "WHY") {
524 if (!installed.has(name)) fail(`${name} is not installed`);
525 if (requests.some((request) => request.name === name)) {
526 output.push(packageText(installed.get(name)) + " is explicitly installed");
527 return;
528 }
529 const queue = requests.map((request) => request.name).sort(),
530 parents = new Map(queue.map((packageName) => [packageName, null]));
531 for (let i = 0; i < queue.length; i++) {
532 const current = queue[i];
533 for (const dep of [...installed.get(current).deps].sort((a, b) =>
534 compareText(a.name, b.name),
535 ))
536 if (!parents.has(dep.name)) {
537 parents.set(dep.name, current);
538 queue.push(dep.name);
539 }
540 }
541 const path = [];
542 for (let current = name; current !== null; current = parents.get(current))
543 path.push(packageText(installed.get(current)));
544 output.push(path.reverse().join(" -> "));
545 return;
546 }
547 if (command === "TREE") {
548 if (argument !== null && !installed.has(name)) fail(`${name} is not installed`);
549 const roots = argument !== null ? [name] : requests.map((request) => request.name).sort(),
550 seen = new Set();
551 function tree(packageName, raw, prefix, indent) {
552 const item = installed.get(packageName),
553 body = prefix + packageText(item) + (raw === null ? "" : ` [${raw}]`);
554 if (seen.has(packageName)) {
555 output.push(body + " (*)");
556 return;
557 }
558 seen.add(packageName);
559 output.push(body);
560 item.deps.forEach((dep, i) => {
561 const last = i === item.deps.length - 1;
562 tree(
563 dep.name,
564 dep.raw,
565 indent + (last ? "`-- " : "|-- "),
566 indent + (last ? " " : "| "),
567 );
568 });
569 }
570 if (!roots.length) output.push("(empty)");
571 for (const root of roots)
572 tree(root, requests.find((request) => request.name === root)?.raw ?? null, "", "");
573 return;
574 }
575 if (command === "OUTDATED") {
576 const lines = [];
577 for (const packageName of [...installed.keys()].sort()) {
578 const current = installed.get(packageName),
579 ranges = requests
580 .filter((request) => request.name === packageName)
581 .map((request) => request.range);
582 for (const item of installed.values())
583 for (const dep of item.deps) if (dep.name === packageName) ranges.push(dep.range);
584 const available = versions(packageName),
585 wanted = available.find(
586 (item) =>
587 (!item.yanked || item.version.text === current.version.text) &&
588 ranges.every((range) => satisfies(item.version, range)),
589 );
590 const latest = available.find((item) => !item.yanked && !item.version.pre.length);
591 if (
592 wanted.version.text !== current.version.text ||
593 (latest !== undefined && compareVersions(latest.version, current.version) > 0)
594 )
595 lines.push(
596 `${packageName} ${current.version.text} ${wanted.version.text} ${latest?.version.text ?? "-"}`,
597 );
598 }
599 output.push(...(lines.length ? lines : ["all packages are up to date"]));
600 return;
601 }
602 if (command === "LOCK") {
603 output.push(
604 ...(installed.size
605 ? [...installed.keys()].sort().map((name) => packageText(installed.get(name)))
606 : ["(empty)"]),
607 );
608 return;
609 }
610 if (command === "STATUS") {
611 output.push(
612 ...(requests.length
613 ? requests.map(
614 (request) =>
615 `${request.name} ${request.raw} -> ${installed.get(request.name).version.text}`,
616 )
617 : ["no requests"]),
618 `${installed.size} installed, ${held.size} held, ${history.length} snapshots`,
619 );
620 return;
621 }
622 if (command === "ADVISORY") {
623 const [id, rest] = cut(argument),
624 [packageName, raw] = cut(rest);
625 if (advisories.has(id)) fail(`advisory ${id} already exists`);
626 const range = constraint(raw);
627 advisories.set(id, { name: packageName, raw, range });
628 output.push(`advisory ${id} recorded`);
629 return;
630 }
631 if (command === "AUDIT") {
632 const lines = [];
633 for (const packageName of [...installed.keys()].sort()) {
634 const item = installed.get(packageName),
635 ids = applicable(item);
636 if (!ids.length) continue;
637 const fix = versions(packageName)
638 .reverse()
639 .find(
640 (candidate) =>
641 compareVersions(candidate.version, item.version) > 0 &&
642 !candidate.version.pre.length &&
643 !candidate.yanked &&
644 !applicable(candidate).length,
645 );
646 lines.push(
647 `${packageText(item)}: ${ids.join(", ")} (${fix === undefined ? "no fix" : "fix: " + fix.version.text})`,
648 );
649 }
650 output.push(...(lines.length ? lines : ["no known vulnerabilities"]));
651 }
652}
653const input = fs.readFileSync(0, "utf8").split(/\r?\n/),
654 count = Number(input[0]);
655for (const command of input.slice(1, count + 1)) {
656 try {
657 execute(command);
658 } catch (error) {
659 output.push(error.message);
660 }
661}
662if (output.length) process.stdout.write(output.join("\n") + "\n");
