11NUMBER = "0|[1-9][0-9]{0,3}"
12NUMBER_PATTERN = re.compile(NUMBER)
13VERSION_PATTERN = re.compile(
14 rf"({NUMBER})\.({NUMBER})\.({NUMBER})(?:-([0-9a-z]+(?:\.[0-9a-z]+)*))?"
15)
16WILDCARDS = ("x", "*")
17OPERATORS = (">=", "<=", ">", "<", "=", "^", "~")
18COMPARISONS = {
19 "=": operator.eq,
20 ">": operator.gt,
21 ">=": operator.ge,
22 "<": operator.lt,
23 "<=": operator.le,
24}
25PLAN_WORDS = ("install", "upgrade", "downgrade", "remove")

32@total_ordering
33@dataclass(frozen=True)
34class Version:
35 core: tuple
36 prerelease: tuple = ()
37
38 @property
39 def stable(self):
40 return not self.prerelease
41
42 def sort_key(self):
43 return self.core, self.stable, self.prerelease
44
45 def __lt__(self, other):
46 return self.sort_key() < other.sort_key()
47
48 def __str__(self):
49 text = ".".join(map(str, self.core))
50 if self.prerelease:
51 text += "-" + ".".join(str(value) for _, value in self.prerelease)
52 return text

55MIN_VERSION = Version((0, 0, 0), ((0, 0),))

58def parse_version(text):
59 match = VERSION_PATTERN.fullmatch(text)
60 if match is None:
61 return None
62 major, minor, patch, prerelease = match.groups()
63 identifiers = []
64 if prerelease is not None:
65 for identifier in prerelease.split("."):
66 if not identifier.isdigit():
67 identifiers.append((1, identifier))
68 elif NUMBER_PATTERN.fullmatch(identifier):
69 identifiers.append((0, int(identifier)))
70 else:
71 return None
72 return Version((int(major), int(minor), int(patch)), tuple(identifiers))

75def upper_bound(core, position):
76 index = position - 1
77 return Version(core[:index] + (core[index] + 1,) + (0,) * (2 - index))
78
79
80def parse_term(term):
81 symbol = next((candidate for candidate in OPERATORS if term.startswith(candidate)), "")
82 pattern = term[len(symbol):]
83 version = parse_version(pattern)
84 if version is not None and not version.stable:
85 count = 3
86 lower = version
87 prerelease_core = version.core
88 else:
89 parts = pattern.split(".")
90 count = len(list(takewhile(NUMBER_PATTERN.fullmatch, parts)))
91 if len(parts) > 3 or not all(part in WILDCARDS for part in parts[count:]):
92 return None
93 numbers = [int(part) for part in parts[:count]] + [0] * (3 - count)
94 lower = Version(tuple(numbers))
95 prerelease_core = None
96 core = lower.core
97 if count == 0:
98 comparators = [("<", MIN_VERSION)] if symbol in (">", "<") else []
99 return comparators, prerelease_core
100 match symbol:
101 case "" | "=":
102 if count == 3:
103 comparators = [("=", lower)]
104 else:
105 comparators = [(">=", lower), ("<", upper_bound(core, count))]
106 case ">":
107 comparators = [(">", lower)] if count == 3 else [(">=", upper_bound(core, count))]
108 case "<=":
109 comparators = [("<=", lower)] if count == 3 else [("<", upper_bound(core, count))]
110 case ">=":
111 comparators = [(">=", lower)]
112 case "<":
113 comparators = [("<", lower)]
114 case "~":
115 comparators = [(">=", lower), ("<", upper_bound(core, 1 if count == 1 else 2))]
116 case _:
117 major, minor, _ = core
118 if major != 0 or count == 1:
119 position = 1
120 elif minor != 0 or count == 2:
121 position = 2
122 else:
123 position = 3
124 comparators = [(">=", lower), ("<", upper_bound(core, position))]
125 return comparators, prerelease_core

128@dataclass
129class Alternative:
130 comparators: list
131 prerelease_cores: set
132
133 def allows(self, version):
134 if not version.stable and version.core not in self.prerelease_cores:
135 return False
136 return all(COMPARISONS[symbol](version, bound) for symbol, bound in self.comparators)
137
138 def __str__(self):
139 return " ".join(f"{symbol}{bound}" for symbol, bound in self.comparators) or "*"

142class Constraint:
143 def __init__(self, text, alternatives):
144 self.text = text
145 self.alternatives = alternatives
146
147 @classmethod
148 def parse(cls, text):
149 alternatives = []
150 for alternative_text in text.split(" || "):
151 alternative = Alternative([], set())
152 for term in alternative_text.split(" "):
153 parsed = parse_term(term)
154 if parsed is None:
155 return None
156 comparators, prerelease_core = parsed
157 alternative.comparators.extend(comparators)
158 if prerelease_core is not None:
159 alternative.prerelease_cores.add(prerelease_core)
160 alternatives.append(alternative)
161 return cls(text, alternatives)
162
163 def allows(self, version):
164 return any(alternative.allows(version) for alternative in self.alternatives)
165
166 def normal_form(self):
167 return " || ".join(str(alternative) for alternative in self.alternatives)

198class Resolver:
199 def __init__(self, manager, preferred, audit):
200 self.manager = manager
201 self.preferred = preferred
202 self.audit = audit
203 self.selection = {}
204 self.reasons = {}
205 self.failure_size = -1
206 self.failure = None
207
208 def usable(self, name, version):
209 manager = self.manager
210 if manager.release(name, version).yanked and manager.installed.get(name) != version:
211 return False
212 return not (self.audit and manager.advisories_for(name, version))
213
214 def chain(self, requirement, indent):
215 lines = []
216 while requirement is not None:
217 lines.append(indent + requirement.describe())
218 origin = requirement.origin
219 requirement = None if origin is None else self.reasons[origin[0]]
220 return lines[::-1]
221
222 def fail(self, requirement, reason, culprit=None):
223 size = len(self.selection)
224 if size > self.failure_size:
225 self.failure_size = size
226 lines = self.chain(requirement, " ")
227 lines.append(" " + reason)
228 if culprit is not None:
229 lines.extend(self.chain(self.reasons[culprit], " "))
230 self.failure = lines
231 return False
232
233 def find_conflict(self, name, version):
234 release = self.manager.release(name, version)
235 for other in sorted(self.selection):
236 selected = self.selection[other]
237 if release.conflicts_with(other, selected):
238 return other
239 if self.manager.release(other, selected).conflicts_with(name, version):
240 return other
241 return None
242
243 def candidates(self, name, constraint):
244 manager = self.manager
245 versions = manager.packages.get(name, {})
246 candidates = sorted(
247 (
248 version
249 for version in versions
250 if constraint.allows(version) and self.usable(name, version)
251 ),
252 reverse=True,
253 )
254 installed = manager.installed.get(name)
255 if name in self.preferred and installed in candidates:
256 candidates.remove(installed)
257 candidates.insert(0, installed)
258 return candidates
259
260 def search(self, pending):
261 if not pending:
262 return True
263 requirement, rest = pending[0], pending[1:]
264 name, constraint = requirement.name, requirement.constraint
265 if name in self.selection:
266 selected = self.selection[name]
267 if constraint.allows(selected):
268 return self.search(rest)
269 return self.fail(requirement, f"{name}@{selected} was selected by:", name)
270 if name in self.manager.held:
271 installed = self.manager.installed[name]
272 if not (constraint.allows(installed) and self.usable(name, installed)):
273 return self.fail(requirement, f"{name}@{installed} is held")
274 candidates = [installed]
275 else:
276 candidates = self.candidates(name, constraint)
277 if not candidates:
278 return self.fail(requirement, f"no version of {name} matches {constraint.text}")
279 for version in candidates:
280 culprit = self.find_conflict(name, version)
281 if culprit is not None:
282 reason = (
283 f"{name}@{version} conflicts with {culprit}@{self.selection[culprit]}, "
284 "which was selected by:"
285 )
286 self.fail(requirement, reason, culprit)
287 continue
288 self.selection[name] = version
289 self.reasons[name] = requirement
290 release = self.manager.release(name, version)
291 dependencies = [
292 Requirement(dependency, dependency_constraint, (name, version))
293 for dependency, dependency_constraint in release.dependencies
294 ]
295 if self.search(dependencies + rest):
296 return True
297 del self.selection[name]
298 del self.reasons[name]
299 return False

302def arrange(names, blocked):
303 pending = sorted(names)
304 order = []
305 while pending:
306 name = next(name for name in pending if not blocked(name, pending))
307 pending.remove(name)
308 order.append(name)
309 return order

326class PackageManager:
327 def __init__(self):
328 self.packages = {}
329 self.requests = {}
330 self.installed = {}
331 self.held = set()
332 self.advisories = {}
333 self.history = []
334
335 def execute(self, line):
336 command, _, rest = line.partition(" ")
337 arguments = rest.split(" ")
338 try:
339 match command:
340 case "PKG":
341 return self.register(rest)
342 case "YANK":
343 return self.yank(*arguments)
344 case "INFO":
345 return self.info(*arguments)
346 case "VERSIONS":
347 return self.versions(*rest.split(" ", 1))
348 case "DEPENDENTS":
349 return self.dependents(*arguments)
350 case "COMPARE":
351 return self.compare(*arguments)
352 case "MATCH":
353 return self.match_version(*rest.split(" ", 1))
354 case "RANGE":
355 return [require_constraint(rest).normal_form()]
356 case "INSTALL":
357 return self.install(*rest.split(" ", 1))
358 case "SIMULATE":
359 return self.install(*rest.split(" ", 1), simulate=True)
360 case "REMOVE":
361 return self.remove(rest)
362 case "UPGRADE":
363 return self.upgrade(rest or None)
364 case "ROLLBACK":
365 return self.rollback()
366 case "HOLD":
367 return self.hold(rest)
368 case "UNHOLD":
369 return self.unhold(rest)
370 case "WHY":
371 return self.why(rest)
372 case "TREE":
373 return self.tree(rest or None)
374 case "OUTDATED":
375 return self.outdated()
376 case "LOCK":
377 return self.lock()
378 case "STATUS":
379 return self.status()
380 case "ADVISORY":
381 return self.add_advisory(*rest.split(" ", 2))
382 case "AUDIT":
383 return self.audit_fix() if rest == "FIX" else self.audit()
384 except CommandError as error:
385 return [f"error: {error}"]

387 def release(self, name, version):
388 return self.packages[name][version]

390 def dependency_names(self, name, version):

393 def newest_first(self, name):
394 return sorted(self.packages.get(name, {}), reverse=True)

396 def find_release(self, name, text):
397 version = parse_version(text)
398 release = self.packages.get(name, {}).get(version)
399 if release is None:
400 raise CommandError(f"{name}@{text} is not registered")

403 def installed_version(self, name):
404 if name not in self.installed:
405 raise CommandError(f"{name} is not installed")
406 return self.installed[name]

408 def advisories_for(self, name, version):
409 return sorted(
410 identifier
411 for identifier, (package, constraint) in self.advisories.items()
412 if package == name and constraint.allows(version)
413 )

415 def display(self, name, version):
416 notes = []
417 if self.release(name, version).yanked:
418 notes.append("yanked")
419 if self.installed.get(name) == version:
420 notes.append("installed")
421 text = f"{name}@{version}"
422 return f"{text} ({', '.join(notes)})" if notes else text

424 def first_dependent(self, name):
425 return min(
426 package
427 for package, version in self.installed.items()
428 if name in self.dependency_names(package, version)
429 )

431 def reachable(self, roots):
432 found = set()
433 stack = list(roots)
434 while stack:
435 name = stack.pop()
436 if name not in found:
437 found.add(name)
438 stack.extend(self.dependency_names(name, self.installed[name]))
439 return found

441 def plan(self, new):
442 old = self.installed
443
444 def waits_for_dependency(name, pending):
445 return any(
446 dependency in pending for dependency in self.dependency_names(name, new[name])
447 )
448
449 def needed_by_other(name, pending):
450 return any(
451 name in self.dependency_names(other, old[other])
452 for other in pending
453 if other != name
454 )
455
456 lines = []
457 for name in arrange(new, waits_for_dependency):
458 version = new[name]
459 previous = old.get(name)
460 if previous is None:
461 lines.append(f"install {name}@{version}")
462 elif version > previous:
463 lines.append(f"upgrade {name} {previous} -> {version}")
464 elif version < previous:
465 lines.append(f"downgrade {name} {previous} -> {version}")
466 removed = [name for name in old if name not in new]
467 for name in arrange(removed, needed_by_other):
468 lines.append(f"remove {name}@{old[name]}")
469 if not lines:
470 return ["no changes"]
471 tallies = [sum(line.startswith(word) for line in lines) for word in PLAN_WORDS]
472 lines.append("{} installed, {} upgraded, {} downgraded, {} removed".format(*tallies))
473 return lines
474
475 def update_state(self, requests, installed):
476 self.history.append((self.requests, self.installed))
477 self.requests = requests
478 self.installed = installed
479 self.held.intersection_update(installed)
480
481 def resolve(self, requests, preferred, audit=False, apply=True):
482 resolver = Resolver(self, preferred, audit)
483 pending = [Requirement(name, constraint, None) for name, constraint in requests.items()]
484 if not resolver.search(pending):
485 return ["UNSATISFIABLE", *resolver.failure]
486 lines = self.plan(resolver.selection)
487 if apply:
488 self.update_state(requests, resolver.selection)
489 return lines

491 def register(self, arguments):
492 head, *clauses = arguments.split("; ")
493 name, text = head.split(" ")
494 version = require_version(text)
495 if version in self.packages.get(name, {}):
496 raise CommandError(f"{name}@{text} is already registered")
497 release = Release()
498 for clause in clauses:
499 kind, other, constraint_text = clause.split(" ", 2)
500 entry = (other, require_constraint(constraint_text))
501 if kind == "DEP":
502 release.dependencies.append(entry)
503 else:
504 release.conflicts.append(entry)
505 self.packages.setdefault(name, {})[version] = release
506 return [f"registered {name}@{text}"]
507
508 def yank(self, name, text):
509 _, release = self.find_release(name, text)
510 if release.yanked:
511 raise CommandError(f"{name}@{text} is already yanked")
512 release.yanked = True
513 return [f"yanked {name}@{text}"]
514
515 def info(self, name, text):
516 version, release = self.find_release(name, text)
517 lines = [self.display(name, version)]
518 lines.extend(
519 f" requires {other} {constraint.text}" for other, constraint in release.dependencies
520 )
521 lines.extend(
522 f" conflicts with {other} {constraint.text}"
523 for other, constraint in release.conflicts
524 )
525 lines.extend(f" advisory {identifier}" for identifier in self.advisories_for(name, version))
526 return lines

528 def versions(self, name, text=None):
529 constraint = None if text is None else require_constraint(text)
530 if name not in self.packages:
531 raise CommandError(f"unknown package {name}")
532 lines = [
533 self.display(name, version)
534 for version in self.newest_first(name)
535 if constraint is None or constraint.allows(version)
536 ]
537 return lines or ["no matching versions"]
538
539 def dependents(self, name, text=None):
540 version = None if text is None else require_version(text)
541 lines = []
542 for package in sorted(self.packages):
543 for candidate in self.newest_first(package):
544 for other, constraint in self.release(package, candidate).dependencies:
545 if other == name and (version is None or constraint.allows(version)):
546 lines.append(f"{package}@{candidate} requires {name} {constraint.text}")
547 return lines or ["no dependents"]

549 def compare(self, first_text, second_text):
550 first = require_version(first_text)
551 second = require_version(second_text)
552 if first < second:
553 symbol = "<"
554 elif first == second:
555 symbol = "="
556 else:
557 symbol = ">"
558 return [f"{first_text} {symbol} {second_text}"]

560 def match_version(self, version_text, constraint_text):
561 version = require_version(version_text)
562 constraint = require_constraint(constraint_text)
563 return ["yes" if constraint.allows(version) else "no"]

565 def install(self, name, text="*", simulate=False):
566 constraint = require_constraint(text)
567 if name not in self.packages:
568 raise CommandError(f"unknown package {name}")
569 requests = dict(self.requests)
570 requests[name] = constraint
571 return self.resolve(requests, set(self.installed), apply=not simulate)
572
573 def remove(self, name):
574 if name not in self.requests:
575 if name in self.installed:
576 raise CommandError(f"{name} is required by {self.first_dependent(name)}")
577 raise CommandError(f"{name} is not installed")
578 requests = {package: c for package, c in self.requests.items() if package != name}
579 kept = self.reachable(requests)
580 if name in kept:
581 version = self.installed[name]
582 lines = [f"{name}@{version} is still required by {self.first_dependent(name)}"]
583 self.update_state(requests, self.installed)
584 return lines
585 installed = {package: v for package, v in self.installed.items() if package in kept}
586 lines = self.plan(installed)
587 self.update_state(requests, installed)
588 return lines

590 def upgrade(self, name=None):
591 if name is None:
592 preferred = set()
593 else:
594 self.installed_version(name)
595 preferred = set(self.installed) - {name}
596 return self.resolve(self.requests, preferred)

598 def rollback(self):
599 if not self.history:
600 raise CommandError("nothing to roll back")
601 requests, installed = self.history.pop()
602 lines = self.plan(installed)
603 self.requests = requests
604 self.installed = installed
605 self.held.intersection_update(installed)
606 return lines

608 def hold(self, name):
609 version = self.installed_version(name)
610 if name in self.held:
611 raise CommandError(f"{name} is already held")
612 self.held.add(name)
613 return [f"held {name}@{version}"]

615 def unhold(self, name):
616 if name not in self.held:
617 raise CommandError(f"{name} is not held")
618 self.held.remove(name)
619 return [f"released {name}"]

621 def why(self, name):
622 version = self.installed_version(name)
623 if name in self.requests:
624 return [f"{name}@{version} is explicitly installed"]
625 parents = dict.fromkeys(self.requests)
626 queue = deque(sorted(self.requests))
627 while queue:
628 current = queue.popleft()
629 for dependency in sorted(self.dependency_names(current, self.installed[current])):
630 if dependency not in parents:
631 parents[dependency] = current
632 queue.append(dependency)
633 path = [name]
634 while parents[path[-1]] is not None:
635 path.append(parents[path[-1]])
636 return [" -> ".join(f"{package}@{self.installed[package]}" for package in reversed(path))]

638 def tree(self, name=None):
639 if name is not None:
640 self.installed_version(name)
641 roots = [name]
642 elif self.requests:
643 roots = sorted(self.requests)
644 else:
645 return ["(empty)"]
646 lines = []
647 shown = set()
648 for root in roots:
649 self.show_tree(root, self.requests.get(root), "", "", shown, lines)
650 return lines

652 def show_tree(self, name, constraint, prefix, indent, shown, lines):
653 version = self.installed[name]
654 body = f"{prefix}{name}@{version}"
655 if constraint is not None:
656 body += f" [{constraint.text}]"
657 if name in shown:
658 lines.append(body + " (*)")
659 return
660 shown.add(name)
661 lines.append(body)
662 dependencies = self.release(name, version).dependencies
663 for index, (dependency, dependency_constraint) in enumerate(dependencies):
664 if index < len(dependencies) - 1:
665 branch, extension = "|-- ", "| "
666 else:
667 branch, extension = "`-- ", " "
668 self.show_tree(
669 dependency, dependency_constraint, indent + branch, indent + extension, shown, lines
670 )

672 def outdated(self):
673 lines = []
674 for name in sorted(self.installed):
675 current = self.installed[name]
676 constraints = [
677 constraint
678 for package, version in self.installed.items()
679 for other, constraint in self.release(package, version).dependencies
680 if other == name
681 ]
682 if name in self.requests:
683 constraints.append(self.requests[name])
684 versions = self.packages[name]
685 wanted = max(
686 version
687 for version, release in versions.items()
688 if (not release.yanked or version == current)
689 and all(constraint.allows(version) for constraint in constraints)
690 )
691 latest = max(
692 (
693 version
694 for version, release in versions.items()
695 if version.stable and not release.yanked
696 ),
697 default=None,
698 )
699 if wanted != current or (latest is not None and latest > current):
700 latest_text = "-" if latest is None else latest
701 lines.append(f"{name} {current} {wanted} {latest_text}")
702 return lines or ["all packages are up to date"]

704 def lock(self):
705 lines = [f"{name}@{self.installed[name]}" for name in sorted(self.installed)]

708 def status(self):

712 ]

714 lines.append("no requests")

721 def add_advisory(self, identifier, name, text):
722 if identifier in self.advisories:
723 raise CommandError(f"advisory {identifier} already exists")
724 self.advisories[identifier] = (name, require_constraint(text))

727 def audit(self):
728 lines = []
729 for name in sorted(self.installed):
730 version = self.installed[name]
731 identifiers = self.advisories_for(name, version)
732 if not identifiers:
733 continue
734 fixes = [
735 candidate
736 for candidate, release in self.packages[name].items()
737 if candidate > version
738 and candidate.stable
739 and not release.yanked
740 and not self.advisories_for(name, candidate)
741 ]
742 fix = f"fix: {min(fixes)}" if fixes else "no fix"
743 lines.append(f"{name}@{version}: {', '.join(identifiers)} ({fix})")
744 return lines or ["no known vulnerabilities"]
