1struct CommandError <: Exception
2 message::String
3end
4fail(s) = throw(CommandError(s))
5check(b, s) = b ? nothing : fail(s)
6struct Version
7 core::NTuple{3,Int}
8 pre::Vector{String}
9end
10numberpart(s) =
11 !isempty(s) && length(s)<=4 && all(c->'0'<=c<='9', s) && (length(s)==1 || first(s)!='0')
12digitpart(s) = !isempty(s) && all(c->'0'<=c<='9', s)
13function parseversion(s)
14 pieces=split(s, '-'; keepempty = true)
15 length(pieces)<=2 || return nothing
16 core=split(pieces[1], '.'; keepempty = true)
17 length(core)==3 && all(numberpart, core) || return nothing
18 pre=String[]
19 if length(pieces)==2
20 pre=String.(split(pieces[2], '.'; keepempty = true))
21 all(
22 x->!isempty(x) &&
23 all(c->'0'<=c<='9' || 'a'<=c<='z', x) &&
24 (!digitpart(x) || numberpart(x)),
25 pre,
26 ) || return nothing
27 end
28 Version(Tuple(parse.(Int, core)), pre)
29end
30versiontext(v) = join(v.core, '.')*(isempty(v.pre) ? "" : "-"*join(v.pre, '.'))
31function compareversions(a, b)
32 a.core!=b.core && return a.core<b.core ? -1 : 1
33 isempty(a.pre)!=isempty(b.pre) && return isempty(a.pre) ? 1 : -1
34 for (x, y) in zip(a.pre, b.pre)
35 x==y && continue
36 xn=digitpart(x)
37 yn=digitpart(y)
38 xn!=yn && return xn ? -1 : 1
39 return (xn ? parse(Int, x)<parse(Int, y) : x<y) ? -1 : 1
40 end
41 length(a.pre)==length(b.pre) ? 0 : length(a.pre)<length(b.pre) ? -1 : 1
42end
43struct Alternative
44 comparisons::Vector{Tuple{String,Version}}
45 precore::Set{NTuple{3,Int}}
46end
47struct Constraint
48 text::String
49 alternatives::Vector{Alternative}
50end
51function upperversion(v, k)
52 core=collect(v.core)
53 core[k]+=1
54 core[(k+1):end].=0
55 Version(Tuple(core), String[])
56end
57function parseterm(term)
58 op=""
59 pattern=term
60 for symbol in (">=", "<=", ">", "<", "=", "^", "~")
61 if startswith(term, symbol)
62 op=symbol
63 pattern=term[(length(symbol)+1):end]
64 break
65 end
66 end
67 parsed=parseversion(pattern)
68 precore=nothing
69 if parsed!==nothing && !isempty(parsed.pre)
70 k=3
71 lower=parsed
72 precore=parsed.core
73 else
74 parts=split(pattern, '.'; keepempty = true)
75 1<=length(parts)<=3 || return nothing
76 k=0
77 wildcard=false
78 numbers=Int[]
79 for part in parts
80 if numberpart(part) && !wildcard
81 k+=1
82 push!(numbers, parse(Int, part))
83 elseif part in ("x", "*")
84 wildcard=true
85 else
86 return nothing
87 end
88 end
89 append!(numbers, zeros(Int, 3-k))
90 lower=Version(Tuple(numbers), String[])
91 end
92 comparisons=Tuple{String,Version}[]
93 if k==0
94 op in (">", "<") && push!(comparisons, ("<", Version((0, 0, 0), ["0"])))
95 elseif op in ("", "=")
96 if k==3
97 push!(comparisons, ("=", lower))
98 else
99 append!(comparisons, [(">=", lower), ("<", upperversion(lower, k))])
100 end
101 elseif op==">"
102 push!(comparisons, k==3 ? (">", lower) : (">=", upperversion(lower, k)))
103 elseif op=="<="
104 push!(comparisons, k==3 ? ("<=", lower) : ("<", upperversion(lower, k)))
105 elseif op in (">=", "<")
106 push!(comparisons, (op, lower))
107 else
108 at=op=="~" ? (k==1 ? 1 : 2) :
109 (lower.core[1]!=0 || k==1 ? 1 : lower.core[2]!=0 || k==2 ? 2 : 3)
110 append!(comparisons, [(">=", lower), ("<", upperversion(lower, at))])
111 end
112 comparisons, precore
113end
114function parseconstraint(text)
115 alternatives=Alternative[]
116 for part in split(text, " || "; keepempty = true)
117 comparisons=Tuple{String,Version}[]
118 cores=Set{NTuple{3,Int}}()
119 for term in split(part, ' '; keepempty = true)
120 parsed=parseterm(term)
121 parsed===nothing && return nothing
122 append!(comparisons, parsed[1])
123 parsed[2]!==nothing && push!(cores, parsed[2])
124 end
125 push!(alternatives, Alternative(comparisons, cores))
126 end
127 Constraint(text, alternatives)
128end
129function allows(c, v)
130 any(c.alternatives) do alt
131 (isempty(v.pre) || v.core in alt.precore) && all(alt.comparisons) do (op, bound)
132 cmp=compareversions(v, bound)
133 op=="=" ? cmp==0 : op==">" ? cmp>0 : op==">=" ? cmp>=0 : op=="<" ? cmp<0 : cmp<=0
134 end
135 end
136end
137normalform(c) = join(
138 [
139 isempty(a.comparisons) ? "*" : join([op*versiontext(v) for (op, v) in a.comparisons], ' ') for a in c.alternatives
140 ],
141 " || ",
142)
143function requireversion(s)
144 v=parseversion(s)
145 check(v!==nothing, "invalid version '$s'")
146 v
147end
148function requireconstraint(s)
149 c=parseconstraint(s)
150 check(c!==nothing, "invalid constraint '$s'")
151 c
152end

177Manager() =
178 Manager(Release[], Link[], Dict{String,Int}(), Set{String}(), Dict{String,Link}(), Snapshot[])
179knownnames(m) = sort!(unique([r.name for r in m.registry]))
180newest(m, name) = sort!(
181 [i for i in eachindex(m.registry) if m.registry[i].name==name];
182 lt = (a, b)->compareversions(m.registry[a].version, m.registry[b].version)>0,
183)
184function advisoryids(m, index)
185 r=m.registry[index]
186 sort!([
187 id for (id, link) in m.advisories if link.name==r.name && allows(link.constraint, r.version)
188 ])
189end
190function findrelease(m, name, text)
191 at=findfirst(r->r.name==name && r.text==text, m.registry)
192 check(at!==nothing, "$name@$text is not registered")
193 at
194end
195function requireinstalled(m, name)
196 check(haskey(m.installed, name), "$name is not installed")
197 m.installed[name]
198end
199function displayrelease(m, index)
200 r=m.registry[index]
201 notes=String[]
202 r.yanked && push!(notes, "yanked")
203 get(m.installed, r.name, 0)==index && push!(notes, "installed")
204 "$(r.name)@$(r.text)"*(isempty(notes) ? "" : " ("*join(notes, ", ")*")")
205end
206function resolve(m, requests, preferred, audit)
207 selected=Dict{String,Int}()
208 chains=Dict{String,Vector{String}}()
209 bestsize=-1
210 failure=String[]
211 usable(i) =
212 (!m.registry[i].yanked || get(m.installed, m.registry[i].name, 0)==i) &&
213 (!audit || isempty(advisoryids(m, i)))
214 function recordfailure(chain, reason, culprit = "")
215 if length(selected)>bestsize
216 bestsize=length(selected)
217 failure=["UNSATISFIABLE"; [" "*s for s in chain]; " "*reason]
218 !isempty(culprit) && append!(failure, [" "*s for s in chains[culprit]])
219 end
220 false
221 end
222 function search(pending)
223 isempty(pending) && return true
224 link, chain=first(pending)
225 rest=pending[2:end]
226 name=link.name
227 c=link.constraint
228 if haskey(selected, name)
229 r=m.registry[selected[name]]
230 allows(c, r.version) && return search(rest)
231 return recordfailure(chain, "$name@$(r.text) was selected by:", name)
232 end
233 if name in m.held
234 index=m.installed[name]
235 allows(c, m.registry[index].version) && usable(index) ||
236 return recordfailure(chain, "$name@$(m.registry[index].text) is held")
237 candidates=[index]
238 else
239 candidates=[i for i in newest(m, name) if allows(c, m.registry[i].version) && usable(i)]
240 if name in preferred && get(m.installed, name, 0) in candidates
241 index=m.installed[name]
242 candidates=[index; filter(!=(index), candidates)]
243 end
244 isempty(candidates) &&
245 return recordfailure(chain, "no version of $name matches $(c.text)")
246 end
247 for index in candidates
248 r=m.registry[index]
249 culprit=""
250 for other in sort!(collect(keys(selected)))
251 s=m.registry[selected[other]]
252 if any(x->x.name==other && allows(x.constraint, s.version), r.conflicts) ||
253 any(x->x.name==name && allows(x.constraint, r.version), s.conflicts)
254 culprit=other
255 break
256 end
257 end
258 if !isempty(culprit)
259 recordfailure(
260 chain,
261 "$name@$(r.text) conflicts with $culprit@$(m.registry[selected[culprit]].text), which was selected by:",
262 culprit,
263 )
264 continue
265 end
266 selected[name]=index
267 chains[name]=chain
268 dependencies=[
269 (dep, [chain; "$name@$(r.text) requires $(dep.name) $(dep.constraint.text)"]) for
270 dep in r.dependencies
271 ]
272 search([dependencies; rest]) && return true
273 delete!(selected, name)
274 delete!(chains, name)
275 end
276 false
277 end
278 pending=[(link, ["requested $(link.name) $(link.constraint.text)"]) for link in requests]
279 success=search(pending)
280 success, selected, failure
281end
282function dependency_names(m, name, installed)
283 [link.name for link in m.registry[installed[name]].dependencies]
284end
285function plan(m, new)
286 lines=String[]
287 counts=zeros(Int, 4)
288 pending=Set(keys(new))
289 while !isempty(pending)
290 name=first([
291 n for
292 n in sort!(collect(pending)) if all(d->!(d in pending), dependency_names(m, n, new))
293 ])
294 delete!(pending, name)
295 r=m.registry[new[name]]
296 if !haskey(m.installed, name)
297 push!(lines, "install $name@$(r.text)")
298 counts[1]+=1
299 elseif new[name]!=m.installed[name]
300 old=m.registry[m.installed[name]]
301 up=compareversions(r.version, old.version)>0
302 word=up ? "upgrade" : "downgrade"
303 push!(lines, "$word $name $(old.text) -> $(r.text)")
304 counts[up ? 2 : 3]+=1
305 end
306 end
307 pending=setdiff(Set(keys(m.installed)), Set(keys(new)))
308 while !isempty(pending)
309 name=first([
310 n for n in sort!(collect(pending)) if
311 all(other->!(n in dependency_names(m, other, m.installed)), pending)
312 ])
313 delete!(pending, name)
314 push!(lines, "remove $name@$(m.registry[m.installed[name]].text)")
315 counts[4]+=1
316 end
317 isempty(lines) ? ["no changes"] :
318 [
319 lines;
320 "$(counts[1]) installed, $(counts[2]) upgraded, $(counts[3]) downgraded, $(counts[4]) removed"
321 ]
322end
323function update!(m, requests, installed)
324 push!(m.history, Snapshot(m.requests, m.installed))
325 m.requests=requests
326 m.installed=installed
327 intersect!(m.held, keys(installed))
328end
329function resolved!(m, requests, preferred; audit = false, apply = true)
330 success, installed, failure=resolve(m, requests, preferred, audit)
331 success || return failure
332 lines=plan(m, installed)
333 apply && update!(m, requests, installed)
334 lines
335end
336firstdependent(m, name) = minimum([
337 other for other in keys(m.installed) if name in dependency_names(m, other, m.installed)
338])
339function reachable(m, roots)
340 found=Set{String}()
341 queue=collect(roots)
342 while !isempty(queue)
343 name=pop!(queue)
344 name in found && continue
345 push!(found, name)
346 append!(queue, dependency_names(m, name, m.installed))
347 end
348 found
349end
350function treelines(m, roots)
351 shown=Set{String}()
352 lines=String[]
353 function visit(name, c, prefix, indent)
354 r=m.registry[m.installed[name]]
355 body=prefix*"$name@$(r.text)"*(c===nothing ? "" : " [$(c.text)]")
356 if name in shown
357 push!(lines, body*" (*)")
358 return
359 end
360 push!(shown, name)
361 push!(lines, body)
362 for (i, dep) in enumerate(r.dependencies)
363 last=i==length(r.dependencies)
364 visit(
365 dep.name,
366 dep.constraint,
367 indent*(last ? "`-- " : "|-- "),
368 indent*(last ? " " : "| "),
369 )
370 end
371 end
372 for root in roots
373 at=findfirst(x->x.name==root, m.requests)
374 visit(root, at===nothing ? nothing : m.requests[at].constraint, "", "")
375 end
376 isempty(lines) ? ["(empty)"] : lines
377end
378function execute!(m, line)
379 pieces=split(line, ' '; limit = 2)
380 command=pieces[1]
381 rest=length(pieces)==2 ? String(pieces[2]) : ""
382 args=String.(split(rest, ' '; keepempty = true))
383 lines=String[]
384 if command=="PKG"
385 clauses=split(rest, "; ")
386 name, text=String.(split(clauses[1], ' '))
387 v=requireversion(text)
388 check(
389 !any(r->r.name==name && r.text==text, m.registry),
390 "$name@$text is already registered",
391 )
392 deps=Link[]
393 conflicts=Link[]
394 for clause in clauses[2:end]
395 kind, other, c=String.(split(clause, ' '; limit = 3))
396 link=Link(other, requireconstraint(c))
397 push!(kind=="DEP" ? deps : conflicts, link)
398 end
399 push!(m.registry, Release(name, text, v, deps, conflicts, false))
400 return ["registered $name@$text"]
401 elseif command in ("YANK", "INFO")
402 name, text=args
403 index=findrelease(m, name, text)
404 r=m.registry[index]
405 if command=="YANK"
406 check(!r.yanked, "$name@$text is already yanked")
407 r.yanked=true
408 return ["yanked $name@$text"]
409 end
410 return [
411 displayrelease(m, index);
412 [" requires $(x.name) $(x.constraint.text)" for x in r.dependencies];
413 [" conflicts with $(x.name) $(x.constraint.text)" for x in r.conflicts];
414 [" advisory $id" for id in advisoryids(m, index)]
415 ]
416 elseif command=="VERSIONS"
417 parts=String.(split(rest, ' '; limit = 2))
418 name=parts[1]
419 c=length(parts)==2 ? requireconstraint(parts[2]) : nothing
420 check(name in knownnames(m), "unknown package $name")
421 lines=[
422 displayrelease(m, i) for
423 i in newest(m, name) if c===nothing || allows(c, m.registry[i].version)
424 ]
425 return isempty(lines) ? ["no matching versions"] : lines
426 elseif command=="DEPENDENTS"
427 name=args[1]
428 v=length(args)==2 ? requireversion(args[2]) : nothing
429 for other in knownnames(m), i in newest(m, other), dep in m.registry[i].dependencies
430 if dep.name==name && (v===nothing || allows(dep.constraint, v))
431 push!(lines, "$other@$(m.registry[i].text) requires $name $(dep.constraint.text)")
432 end
433 end
434 return isempty(lines) ? ["no dependents"] : lines
435 elseif command=="COMPARE"
436 a=requireversion(args[1])
437 b=requireversion(args[2])
438 cmp=compareversions(a, b)
439 return ["$(args[1]) $(cmp<0 ? "<" : cmp>0 ? ">" : "=") $(args[2])"]
440 elseif command=="MATCH"
441 parts=String.(split(rest, ' '; limit = 2))
442 v=requireversion(parts[1])
443 c=requireconstraint(parts[2])
444 return [allows(c, v) ? "yes" : "no"]
445 elseif command=="RANGE"
446 return [normalform(requireconstraint(rest))]
447 elseif command in ("INSTALL", "SIMULATE")
448 parts=String.(split(rest, ' '; limit = 2))
449 name=parts[1]
450 c=requireconstraint(length(parts)==2 ? parts[2] : "*")
451 check(name in knownnames(m), "unknown package $name")
452 requests=copy(m.requests)
453 at=findfirst(x->x.name==name, requests)
454 if at===nothing
455 push!(requests, Link(name, c))
456 else
457 requests[at]=Link(name, c)
458 end
459 return resolved!(m, requests, Set(keys(m.installed)); apply = command=="INSTALL")
460 elseif command=="REMOVE"
461 name=rest
462 at=findfirst(x->x.name==name, m.requests)
463 if at===nothing
464 haskey(m.installed, name) && fail("$name is required by $(firstdependent(m,name))")
465 fail("$name is not installed")
466 end
467 requests=filter(x->x.name!=name, m.requests)
468 kept=reachable(m, [x.name for x in requests])
469 installed=Dict(n=>i for (n, i) in m.installed if n in kept)
470 if name in kept
471 lines=[
472 "$name@$(m.registry[m.installed[name]].text) is still required by $(firstdependent(m,name))",
473 ]
474 installed=m.installed
475 else
476 lines=plan(m, installed)
477 end
478 update!(m, requests, installed)
479 return lines
480 elseif command=="UPGRADE"
481 preferred=Set{String}()
482 if !isempty(rest)
483 requireinstalled(m, rest)
484 preferred=setdiff(Set(keys(m.installed)), Set([rest]))
485 end
486 return resolved!(m, m.requests, preferred)
487 elseif command=="ROLLBACK"
488 check(!isempty(m.history), "nothing to roll back")
489 snapshot=pop!(m.history)
490 lines=plan(m, snapshot.installed)
491 m.requests=snapshot.requests
492 m.installed=snapshot.installed
493 intersect!(m.held, keys(m.installed))
494 return lines
495 elseif command=="HOLD"
496 index=requireinstalled(m, rest)
497 check(!(rest in m.held), "$rest is already held")
498 push!(m.held, rest)
499 return ["held $rest@$(m.registry[index].text)"]
500 elseif command=="UNHOLD"
501 check(rest in m.held, "$rest is not held")
502 delete!(m.held, rest)
503 return ["released $rest"]
504 elseif command=="WHY"
505 index=requireinstalled(m, rest)
506 roots=[x.name for x in m.requests]
507 rest in roots && return ["$rest@$(m.registry[index].text) is explicitly installed"]
508 parents=Dict(x=>"" for x in roots)
509 queue=sort(roots)
510 at=1
511 while at<=length(queue)
512 name=queue[at]
513 at+=1
514 for dep in sort(dependency_names(m, name, m.installed))
515 if !haskey(parents, dep)
516 parents[dep]=name
517 push!(queue, dep)
518 end
519 end
520 end
521 path=[rest]
522 while !isempty(parents[last(path)])
523 push!(path, parents[last(path)])
524 end
525 return [
526 join(["$name@$(m.registry[m.installed[name]].text)" for name in reverse(path)], " -> "),
527 ]
528 elseif command=="TREE"
529 if !isempty(rest)
530 requireinstalled(m, rest)
531 end
532 return treelines(m, isempty(rest) ? sort([x.name for x in m.requests]) : [rest])
533 elseif command=="OUTDATED"
534 for name in sort!(collect(keys(m.installed)))
535 cur=m.installed[name]
536 constraints=[link.constraint for link in m.requests if link.name==name]
537 for i in values(m.installed), dep in m.registry[i].dependencies
538 dep.name==name && push!(constraints, dep.constraint)
539 end
540 versions=newest(m, name)
541 wanted=first(
542 i for i in versions if (!m.registry[i].yanked || i==cur) &&
543 all(c->allows(c, m.registry[i].version), constraints)
544 )
545 latest=findfirst(
546 i->isempty(m.registry[i].version.pre) && !m.registry[i].yanked,
547 versions,
548 )
549 latestindex=latest===nothing ? 0 : versions[latest]
550 if wanted!=cur || (
551 latestindex!=0 &&
552 compareversions(m.registry[latestindex].version, m.registry[cur].version)>0
553 )
554 push!(
555 lines,
556 "$name $(m.registry[cur].text) $(m.registry[wanted].text) $(latestindex==0 ? "-" : m.registry[latestindex].text)",
557 )
558 end
559 end
560 return isempty(lines) ? ["all packages are up to date"] : lines
561 elseif command=="LOCK"
562 lines=[
563 "$name@$(m.registry[m.installed[name]].text)" for
564 name in sort!(collect(keys(m.installed)))
565 ]
566 return isempty(lines) ? ["(empty)"] : lines
567 elseif command=="STATUS"
568 lines=[
569 "$(r.name) $(r.constraint.text) -> $(m.registry[m.installed[r.name]].text)" for
570 r in m.requests
571 ]
572 isempty(lines) && push!(lines, "no requests")
573 push!(
574 lines,
575 "$(length(m.installed)) installed, $(length(m.held)) held, $(length(m.history)) snapshots",
576 )
577 return lines
578 elseif command=="ADVISORY"
579 id, name, text=String.(split(rest, ' '; limit = 3))
580 check(!haskey(m.advisories, id), "advisory $id already exists")
581 c=requireconstraint(text)
582 m.advisories[id]=Link(name, c)
583 return ["advisory $id recorded"]
584 elseif command=="AUDIT"
585 rest=="FIX" && return resolved!(m, m.requests, Set(keys(m.installed)); audit = true)
586 for name in sort!(collect(keys(m.installed)))
587 cur=m.installed[name]
588 ids=advisoryids(m, cur)
589 isempty(ids) && continue
590 fixes=[
591 i for i in reverse(newest(m, name)) if
592 compareversions(m.registry[i].version, m.registry[cur].version)>0 &&
593 isempty(m.registry[i].version.pre) &&
594 !m.registry[i].yanked &&
595 isempty(advisoryids(m, i))
596 ]
597 push!(
598 lines,
599 "$name@$(m.registry[cur].text): $(join(ids,", ")) ($(isempty(fixes) ? "no fix" : "fix: "*m.registry[first(fixes)].text))",
600 )
601 end
602 return isempty(lines) ? ["no known vulnerabilities"] : lines
603 end
604 lines
605end
606function main()
607 manager=Manager()
608 count=parse(Int, readline())
609 for _ in 1:count
610 line=readline()
611 try
612 foreach(println, execute!(manager, line))
613 catch e
614 e isa CommandError || rethrow()
615 println("error: $(e.message)")
616 end
617 end
618end
