85Version? version(String text) {
86 final match = RegExp(r'^([0-9]+)\.([0-9]+)\.([0-9]+)(?:-([a-z0-9]+(?:\.[a-z0-9]+)*))?$').firstMatch(text);
87 if (match == null || ![1, 2, 3].every((i) => numberPattern.hasMatch(match.group(i)!))) return null;
88 final pre = match.group(4)?.split('.') ?? <String>[];
89 if (pre.any((part) => digitsPattern.hasMatch(part) && !numberPattern.hasMatch(part))) return null;
90 return Version(text, [for (var i = 1; i <= 3; i++) int.parse(match.group(i)!)], pre);
91}
92Version requireVersion(String text) {
93 final result = version(text);
94 if (result == null) fail("invalid version '$text'");
95 return result;
96}
97int compareVersions(Version a, Version b) {
98 for (var i = 0; i < 3; i++) {
99 if (a.core[i] != b.core[i]) return a.core[i].compareTo(b.core[i]);
100 }
101 if (a.pre.isEmpty || b.pre.isEmpty) return (b.pre.isNotEmpty ? 1 : 0) - (a.pre.isNotEmpty ? 1 : 0);
102 for (var i = 0; i < math.min(a.pre.length, b.pre.length); i++) {
103 final x = a.pre[i], y = b.pre[i];
104 final nx = digitsPattern.hasMatch(x), ny = digitsPattern.hasMatch(y);
105 final order = nx && ny ? int.parse(x).compareTo(int.parse(y))
106 : nx != ny ? nx ? -1 : 1 : x.compareTo(y);
107 if (order != 0) return order;
108 }
109 return a.pre.length.compareTo(b.pre.length);
110}
111Version synthetic(List<int> core, [List<String> pre = const []]) =>
112 Version('${core.join('.')}${pre.isEmpty ? '' : '-${pre.join('.')}'}', core, pre);
113Constraint? parseConstraint(String text) {
114 final alternatives = <Alternative>[];
115 for (final alternative in text.split(' || ')) {
116 final comparators = <Comparator>[], prereleases = <String>{};
117 for (final term in alternative.split(' ')) {
118 if (term.isEmpty) return null;
119 var op = '';
120 for (final candidate in ['>=', '<=', '>', '<', '=', '^', '~']) {
121 if (term.startsWith(candidate)) { op = candidate; break; }
122 }
123 final pattern = term.substring(op.length), prerelease = version(term.substring(op.length));
124 var core = <int>[0, 0, 0], count = 0;
125 late Version lower;
126 if (prerelease != null && prerelease.pre.isNotEmpty) {
127 core = prerelease.core;
128 count = 3;
129 lower = prerelease;
130 prereleases.add(core.join('.'));
131 } else {
132 final parts = pattern.split('.');
133 if (parts.isEmpty || parts.length > 3) return null;
134 var wildcard = false;
135 for (final part in parts) {
136 if (part == 'x' || part == '*') { wildcard = true; }
137 else if (!wildcard && numberPattern.hasMatch(part)) { core[count++] = int.parse(part); }
138 else { return null; }
139 }
140 lower = synthetic(core);
141 }
142 void add(String operator, Version bound) { comparators.add(Comparator(operator, bound)); }
143 Version upper(int index) => synthetic([for (var i = 0; i < 3; i++)
144 i < index - 1 ? core[i] : i == index - 1 ? core[i] + 1 : 0]);
145 if (count == 0) {
146 if (op == '>' || op == '<') add('<', synthetic([0, 0, 0], ['0']));
147 } else if (op.isEmpty || op == '=') {
148 if (count == 3) { add('=', lower); }
149 else { add('>=', lower); add('<', upper(count)); }
150 } else if (op == '>') {
151 add(count == 3 ? '>' : '>=', count == 3 ? lower : upper(count));
152 } else if (op == '<=') {
153 add(count == 3 ? '<=' : '<', count == 3 ? lower : upper(count));
154 } else if (op == '>=' || op == '<') {
155 add(op, lower);
156 } else if (op == '~') {
157 add('>=', lower);
158 add('<', upper(count == 1 ? 1 : 2));
159 } else if (op == '^') {
160 add('>=', lower);
161 add('<', upper(core[0] != 0 || count == 1 ? 1 : core[1] != 0 || count == 2 ? 2 : 3));
162 } else {
163 return null;
164 }
165 }
166 alternatives.add(Alternative(comparators, prereleases));
167 }
168 return Constraint(alternatives);
169}
170Constraint constraint(String text) {
171 final cached = constraintCache[text];
172 if (cached != null) return cached;
173 final result = parseConstraint(text);
174 if (result == null) fail("invalid constraint '$text'");
175 constraintCache[text] = result;
176 return result;
177}
178bool satisfies(Version value, Constraint range) => range.alternatives.any((alternative) =>
179 (value.pre.isEmpty || alternative.prereleases.contains(value.core.join('.'))) &&
180 alternative.comparators.every((comparator) {
181 final order = compareVersions(value, comparator.bound);
182 return switch (comparator.op) {
183 '=' => order == 0, '>' => order > 0, '>=' => order >= 0,
184 '<' => order < 0, _ => order <= 0
185 };
186 }));
187List<Package> versions(String name) => (registry[name]?.values.toList() ?? <Package>[])
188 ..sort((a, b) => compareVersions(b.version, a.version));

194String packageText(Package item) => '${item.name}@${item.version.text}';
195List<String> applicable(Package item) => (advisories.keys.toList()..sort()).where((id) {
196 final advisory = advisories[id]!;
197 return advisory.name == item.name && satisfies(item.version, advisory.range);
198}).toList();
199String display(Package item) {
200 final current = installed[item.name]?.version.text == item.version.text;
201 return packageText(item) + (item.yanked ? current ? ' (yanked, installed)' : ' (yanked)' : current ? ' (installed)' : '');
202}
203List<String> requirementChain(Requirement requirement) {
204 final result = <String>[];
205 for (Requirement? current = requirement; current != null; current = current.parent) {
206 result.add(current.origin == null ? 'requested ${current.name} ${current.raw}'
207 : '${current.origin} requires ${current.name} ${current.raw}');
208 }
209 return result.reversed.toList();
210}
211Resolution resolve(List<Declaration> wanted, Set<String> preferred, bool audit) {
212 final selected = <String, Selection>{};
213 Failure? failure;
214 bool usable(Package item) => (!item.yanked || installed[item.name]?.version.text == item.version.text) &&
215 (!audit || applicable(item).isEmpty);
216 void record(Requirement requirement, String reason, [Requirement? other]) {
217 if (failure == null || selected.length > failure!.size) {
218 failure = Failure(selected.length, requirementChain(requirement), reason,
219 other == null ? null : requirementChain(other));
220 }
221 }
222 bool search(List<Requirement> requirements) {
223 if (requirements.isEmpty) return true;
224 final requirement = requirements.first, rest = requirements.skip(1).toList();
225 final name = requirement.name, existing = selected[requirement.name];
226 if (existing != null) {
227 if (satisfies(existing.item.version, requirement.range)) return search(rest);
228 record(requirement, '${packageText(existing.item)} was selected by:', existing.requirement);
229 return false;
230 }
231 List<Package> candidates;
232 if (held.contains(name)) {
233 final item = installed[name]!;
234 if (!satisfies(item.version, requirement.range) || !usable(item)) {
235 record(requirement, '${packageText(item)} is held');
236 return false;
237 }
238 candidates = [item];
239 } else {
240 candidates = versions(name).where((item) => satisfies(item.version, requirement.range) && usable(item)).toList();
241 if (preferred.contains(name)) {
242 final at = candidates.indexWhere((item) => item.version.text == installed[name]?.version.text);
243 if (at >= 0) candidates.insert(0, candidates.removeAt(at));
244 }
245 if (candidates.isEmpty) {
246 record(requirement, 'no version of $name matches ${requirement.raw}');
247 return false;
248 }
249 }
250 for (final item in candidates) {
251 Selection? conflict;
252 for (final otherName in selected.keys.toList()..sort()) {
253 final other = selected[otherName]!;
254 if (item.conflicts.any((dep) => dep.name == otherName && satisfies(other.item.version, dep.range)) ||
255 other.item.conflicts.any((dep) => dep.name == name && satisfies(item.version, dep.range))) {
256 conflict = other;
257 break;
258 }
259 }
260 if (conflict != null) {
261 record(requirement, '${packageText(item)} conflicts with ${packageText(conflict.item)}, which was selected by:', conflict.requirement);
262 continue;
263 }
264 selected[name] = Selection(item, requirement);
265 final dependencies = item.deps.map((dep) => Requirement(dep, packageText(item), requirement));
266 if (search([...dependencies, ...rest])) return true;
267 selected.remove(name);
268 }
269 return false;
270 }
271 final requirements = wanted.map((request) => Requirement(request, null, null)).toList();
272 if (search(requirements)) return Resolution({for (final entry in selected.entries) entry.key: entry.value.item}, null);
273 return Resolution(null, failure);
274}
275void reportFailure(Failure failure) {
276 output.add('UNSATISFIABLE');
277 output.addAll(failure.chain.map((line) => ' $line'));
278 output.add(' ${failure.reason}');
279 if (failure.other != null) output.addAll(failure.other!.map((line) => ' $line'));
280}
281void plan(Map<String, Package> next) {
282 final remaining = (next.keys.toList()..sort()).toSet(), arranged = <String>{};
283 final counts = List<int>.filled(4, 0), lines = <String>[];
284 while (remaining.isNotEmpty) {
285 final name = remaining.firstWhere((name) => next[name]!.deps.every((dep) => arranged.contains(dep.name)));
286 final item = next[name]!, old = installed[name];
287 if (old == null) {
288 lines.add('install ${packageText(item)}');
289 counts[0]++;
290 } else if (old.version.text != item.version.text) {
291 final upgrading = compareVersions(item.version, old.version) > 0;
292 lines.add('${upgrading ? 'upgrade' : 'downgrade'} $name ${old.version.text} -> ${item.version.text}');
293 counts[upgrading ? 1 : 2]++;
294 }
295 remaining.remove(name);
296 arranged.add(name);
297 }
298 final removals = (installed.keys.where((name) => !next.containsKey(name)).toList()..sort()).toSet();
299 while (removals.isNotEmpty) {
300 final name = removals.firstWhere((name) => !removals.any((other) => installed[other]!.deps.any((dep) => dep.name == name)));
301 lines.add('remove ${packageText(installed[name]!)}');
302 counts[3]++;
303 removals.remove(name);
304 }
305 output.addAll(lines.isEmpty ? ['no changes'] : [...lines,
306 '${counts[0]} installed, ${counts[1]} upgraded, ${counts[2]} downgraded, ${counts[3]} removed']);
307}
308void update(List<Declaration> wanted, Map<String, Package> next, [bool snapshot = true]) {
309 if (snapshot) history.add(Snapshot(List.of(requests), Map.of(installed)));
310 requests = List.of(wanted);
311 installed = Map.of(next);
312 held.removeWhere((name) => !installed.containsKey(name));
313}
314Set<String> reach(List<Declaration> wanted) {
315 final seen = wanted.map((request) => request.name).toSet(), queue = <String>[];
316 queue.addAll(seen);
317 for (var i = 0; i < queue.length; i++) {
318 for (final dep in installed[queue[i]]!.deps) {
319 if (seen.add(dep.name)) queue.add(dep.name);
320 }
321 }
322 return seen;
323}
324String requiredBy(String name) => (installed.keys.toList()..sort())
325 .firstWhere((other) => installed[other]!.deps.any((dep) => dep.name == name));
326(String, String?) cut(String text) {
327 final at = text.indexOf(' ');
328 return at < 0 ? (text, null) : (text.substring(0, at), text.substring(at + 1));
329}
330void execute(String commandLine) {
331 final (command, argument) = cut(commandLine);
332 final words = argument?.split(' ') ?? <String>[], name = argument?.split(' ').first ?? '';
333 if (command == 'PKG') {
334 final parts = commandLine.split('; '), main = parts.first.split(' ');
335 final packageName = main[1], text = main[2], parsed = requireVersion(main[2]);
336 if (registry[packageName]?.containsKey(text) ?? false) fail('$packageName@$text is already registered');
337 final deps = <Declaration>[], conflicts = <Declaration>[];
338 for (final clause in parts.skip(1)) {
339 final (kind, rest) = cut(clause);
340 final (other, raw) = cut(rest!);
341 final range = constraint(raw!);
342 (kind == 'DEP' ? deps : conflicts).add(Declaration(other, raw, range));
343 }
344 registry.putIfAbsent(packageName, () => {})[text] = Package(packageName, parsed, deps, conflicts);
345 output.add('registered $packageName@$text');
346 return;
347 }
348 if (command == 'YANK' || command == 'INFO') {
349 final item = registered(name, words[1]);
350 if (command == 'YANK') {
351 if (item.yanked) fail('${packageText(item)} is already yanked');
352 item.yanked = true;
353 output.add('yanked ${packageText(item)}');
354 } else {
355 output.add(display(item));
356 output.addAll(item.deps.map((dep) => ' requires ${dep.name} ${dep.raw}'));
357 output.addAll(item.conflicts.map((dep) => ' conflicts with ${dep.name} ${dep.raw}'));
358 output.addAll(applicable(item).map((id) => ' advisory $id'));
359 }
360 return;
361 }
362 if (command == 'COMPARE') {
363 final a = requireVersion(words[0]), b = requireVersion(words[1]), order = compareVersions(a, b);
364 output.add('${words[0]} ${order < 0 ? '<' : order > 0 ? '>' : '='} ${words[1]}');
365 return;
366 }
367 if (command == 'MATCH') {
368 final (text, raw) = cut(argument!);
369 final parsed = requireVersion(text), range = constraint(raw!);
370 output.add(satisfies(parsed, range) ? 'yes' : 'no');
371 return;
372 }
373 if (command == 'RANGE') { output.add(constraint(argument!).normal); return; }
374 if (command == 'VERSIONS' || command == 'INSTALL' || command == 'SIMULATE') {
375 final (packageName, rest) = cut(argument!);
376 final raw = rest ?? '*', range = constraint(rest ?? '*');
377 if (!registry.containsKey(packageName)) fail('unknown package $packageName');
378 if (command == 'VERSIONS') {
379 final selected = versions(packageName).where((item) => rest == null || satisfies(item.version, range)).toList();
380 output.addAll(selected.isEmpty ? ['no matching versions'] : selected.map(display));
381 return;
382 }
383 final wanted = List<Declaration>.of(requests), at = requests.indexWhere((request) => request.name == packageName);
384 final request = Declaration(packageName, raw, range);
385 if (at >= 0) { wanted[at] = request; } else { wanted.add(request); }
386 final result = resolve(wanted, installed.keys.toSet(), false);
387 if (result.installation == null) { reportFailure(result.failure!); }
388 else {
389 plan(result.installation!);
390 if (command == 'INSTALL') update(wanted, result.installation!);
391 }
392 return;
393 }
394 if (command == 'DEPENDENTS') {
395 final parsed = words.length > 1 ? requireVersion(words[1]) : null, lines = <String>[];
396 for (final other in registry.keys.toList()..sort()) {
397 for (final item in versions(other)) {
398 for (final dep in item.deps) {
399 if (dep.name == name && (parsed == null || satisfies(parsed, dep.range))) lines.add('${packageText(item)} requires $name ${dep.raw}');
400 }
401 }
402 }
403 output.addAll(lines.isEmpty ? ['no dependents'] : lines);
404 return;
405 }
406 if (command == 'REMOVE') {
407 final at = requests.indexWhere((request) => request.name == name);
408 if (at < 0) fail(installed.containsKey(name) ? '$name is required by ${requiredBy(name)}' : '$name is not installed');
409 final wanted = requests.where((request) => request.name != name).toList(), keep = reach(requests.where((request) => request.name != name).toList());
410 if (keep.contains(name)) {
411 output.add('${packageText(installed[name]!)} is still required by ${requiredBy(name)}');
412 update(wanted, installed);
413 } else {
414 final next = <String, Package>{for (final entry in installed.entries) if (keep.contains(entry.key)) entry.key: entry.value};
415 plan(next);
416 update(wanted, next);
417 }
418 return;
419 }
420 if (command == 'UPGRADE' || command == 'AUDIT' && argument == 'FIX') {
421 if (command == 'UPGRADE' && argument != null && !installed.containsKey(name)) fail('$name is not installed');
422 final preferred = command == 'AUDIT' || argument != null
423 ? installed.keys.where((packageName) => command == 'AUDIT' || packageName != name).toSet() : <String>{};
424 final result = resolve(requests, preferred, command == 'AUDIT');
425 if (result.installation == null) { reportFailure(result.failure!); }
426 else { plan(result.installation!); update(requests, result.installation!); }
427 return;
428 }
429 if (command == 'ROLLBACK') {
430 if (history.isEmpty) fail('nothing to roll back');
431 final snapshot = history.removeLast();
432 plan(snapshot.installed);
433 update(snapshot.requests, snapshot.installed, false);
434 return;
435 }
436 if (command == 'HOLD') {
437 if (!installed.containsKey(name)) fail('$name is not installed');
438 if (held.contains(name)) fail('$name is already held');
439 held.add(name);
440 output.add('held ${packageText(installed[name]!)}');
441 return;
442 }
443 if (command == 'UNHOLD') {
444 if (!held.contains(name)) fail('$name is not held');
445 held.remove(name);
446 output.add('released $name');
447 return;
448 }
449 if (command == 'WHY') {
450 if (!installed.containsKey(name)) fail('$name is not installed');
451 if (requests.any((request) => request.name == name)) {
452 output.add('${packageText(installed[name]!)} is explicitly installed');
453 return;
454 }
455 final queue = requests.map((request) => request.name).toList()..sort();
456 final parents = <String, String?>{for (final packageName in queue) packageName: null};
457 for (var i = 0; i < queue.length; i++) {
458 final current = queue[i];
459 final deps = installed[current]!.deps.toList()..sort((a, b) => a.name.compareTo(b.name));
460 for (final dep in deps) {
461 if (!parents.containsKey(dep.name)) { parents[dep.name] = current; queue.add(dep.name); }
462 }
463 }
464 final path = <String>[];
465 for (String? current = name; current != null; current = parents[current]) {
466 path.add(packageText(installed[current]!));
467 }
468 output.add(path.reversed.join(' -> '));
469 return;
470 }
471 if (command == 'TREE') {
472 if (argument != null && !installed.containsKey(name)) fail('$name is not installed');
473 final roots = argument != null ? [name] : (requests.map((request) => request.name).toList()..sort());
474 final seen = <String>{};
475 void tree(String packageName, String? raw, String prefix, String indent) {
476 final item = installed[packageName]!;
477 final body = '$prefix${packageText(item)}${raw == null ? '' : ' [$raw]'}';
478 if (!seen.add(packageName)) { output.add('$body (*)'); return; }
479 output.add(body);
480 for (var i = 0; i < item.deps.length; i++) {
481 final dep = item.deps[i], last = i == item.deps.length - 1;
482 tree(dep.name, dep.raw, indent + (last ? '`-- ' : '|-- '), indent + (last ? ' ' : '| '));
483 }
484 }
485 if (roots.isEmpty) output.add('(empty)');
486 for (final root in roots) {
487 tree(root, firstOrNull(requests.where((request) => request.name == root))?.raw, '', '');
488 }
489 return;
490 }
491 if (command == 'OUTDATED') {
492 final lines = <String>[];
493 for (final packageName in installed.keys.toList()..sort()) {
494 final current = installed[packageName]!;
495 final ranges = requests.where((request) => request.name == packageName).map((request) => request.range).toList();
496 for (final item in installed.values) {
497 for (final dep in item.deps) { if (dep.name == packageName) ranges.add(dep.range); }
498 }
499 final available = versions(packageName);
500 final wanted = available.firstWhere((item) => (!item.yanked || item.version.text == current.version.text) && ranges.every((range) => satisfies(item.version, range)));
501 final latest = firstOrNull(available.where((item) => !item.yanked && item.version.pre.isEmpty));
502 if (wanted.version.text != current.version.text || latest != null && compareVersions(latest.version, current.version) > 0) {
503 lines.add('$packageName ${current.version.text} ${wanted.version.text} ${latest?.version.text ?? '-'}');
504 }
505 }
506 output.addAll(lines.isEmpty ? ['all packages are up to date'] : lines);
507 return;
508 }
509 if (command == 'LOCK') {
510 output.addAll(installed.isEmpty ? ['(empty)'] : (installed.keys.toList()..sort()).map((name) => packageText(installed[name]!)));
511 return;
512 }
513 if (command == 'STATUS') {
514 output.addAll(requests.isEmpty ? ['no requests'] : requests.map((request) => '${request.name} ${request.raw} -> ${installed[request.name]!.version.text}'));
515 output.add('${installed.length} installed, ${held.length} held, ${history.length} snapshots');
516 return;
517 }
518 if (command == 'ADVISORY') {
519 final (id, rest) = cut(argument!);
520 final (packageName, raw) = cut(rest!);
521 if (advisories.containsKey(id)) fail('advisory $id already exists');
522 final range = constraint(raw!);
523 advisories[id] = Declaration(packageName, raw, range);
524 output.add('advisory $id recorded');
525 return;
526 }
527 if (command == 'AUDIT') {
528 final lines = <String>[];
529 for (final packageName in installed.keys.toList()..sort()) {
530 final item = installed[packageName]!, ids = applicable(installed[packageName]!);
531 if (ids.isEmpty) continue;
532 final fix = firstOrNull(versions(packageName).reversed.where((candidate) => compareVersions(candidate.version, item.version) > 0 &&
533 candidate.version.pre.isEmpty && !candidate.yanked && applicable(candidate).isEmpty));
534 lines.add('${packageText(item)}: ${ids.join(', ')} (${fix == null ? 'no fix' : 'fix: ${fix.version.text}'})');
535 }
536 output.addAll(lines.isEmpty ? ['no known vulnerabilities'] : lines);
537 }
538}
539void main() {
540 final count = int.parse(stdin.readLineSync()!);
541 for (var i = 0; i < count; i++) {
542 try { execute(stdin.readLineSync()!); }
543 on PackageError catch (error) { output.add(error.message); }
544 }
545 if (output.isNotEmpty) stdout.write('${output.join('\n')}\n');
546}
