4split_keep <- function(text, separator) {
5 result <- character()
6 repeat {
7 at <- regexpr(separator, text, fixed = TRUE)[[1]]
8 if (at < 0L) return(c(result, text))
9 result <- c(result, substr(text, 1L, at - 1L))
10 text <- substring(text, at + nchar(separator))
11 }
12}
13bounded <- function(text, count) {
14 result <- character()
15 while (length(result) + 1L < count) {
16 at <- regexpr(" ", text, fixed = TRUE)[[1]]
17 if (at < 0L) break
18 result <- c(result, substr(text, 1L, at - 1L))
19 text <- substring(text, at + 1L)
20 }
21 c(result, text)
22}
23ascii_compare <- function(a, b) {
24 ac <- as.integer(charToRaw(a))
25 bc <- as.integer(charToRaw(b))
26 for (i in seq_len(min(length(ac), length(bc)))) {
27 if (ac[[i]] != bc[[i]]) return(sign(ac[[i]] - bc[[i]]))
28 }
29 sign(length(ac) - length(bc))
30}
31number_pattern <- "\\A(?:0|[1-9][0-9]{0,3})\\z"
32version_cache <- new.env(parent = emptyenv())
33parse_version <- function(text) {
34 key <- paste0("v", text)
35 if (!is.null(version_cache[[key]])) return(version_cache[[key]])
36 pattern <- "\\A(0|[1-9][0-9]{0,3})\\.(0|[1-9][0-9]{0,3})\\.(0|[1-9][0-9]{0,3})(?:-([a-z0-9]+(?:\\.[a-z0-9]+)*))?\\z"
37 match <- regmatches(text, regexec(pattern, text, perl = TRUE))[[1]]
38 if (!length(match)) return(NULL)
39 pre <- if (length(match) >= 5L && match[[5]] != "") strsplit(match[[5]], ".", fixed = TRUE)[[1]] else NULL
40 if (any(vapply(pre, function(part) grepl("\\A[0-9]+\\z", part, perl = TRUE) && !grepl(number_pattern, part, perl = TRUE), logical(1)))) return(NULL)
41 result <- list(core = as.integer(match[2:4]), pre = pre)
42 version_cache[[key]] <- result
43 result

45compare_version <- function(a, b) {
46 for (i in 1:3) if (a$core[[i]] != b$core[[i]]) return(sign(a$core[[i]] - b$core[[i]]))
47 if (is.null(a$pre) || is.null(b$pre)) return(if (is.null(a$pre) && is.null(b$pre)) 0L else if (is.null(a$pre)) 1L else -1L)
48 for (i in seq_len(min(length(a$pre), length(b$pre)))) {
49 x <- a$pre[[i]]
50 y <- b$pre[[i]]
51 if (x == y) next
52 xn <- grepl("\\A[0-9]+\\z", x, perl = TRUE)
53 yn <- grepl("\\A[0-9]+\\z", y, perl = TRUE)
54 if (xn && yn) return(sign(as.integer(x) - as.integer(y)))
55 if (xn != yn) return(if (xn) -1L else 1L)
56 return(ascii_compare(x, y))
57 }
58 sign(length(a$pre) - length(b$pre))
59}
60version_text <- function(value) paste0(paste(value$core, collapse = "."), if (is.null(value$pre)) "" else paste0("-", paste(value$pre, collapse = ".")))
61parse_constraint <- function(text) {
62 alternatives <- list()
63 for (alternative in split_keep(text, " || ")) {
64 if (alternative == "") return(NULL)
65 comparators <- list()
66 gates <- list()
67 for (term in split_keep(alternative, " ")) {
68 operators <- c(">=", "<=", ">", "<", "=", "^", "~")
69 found <- operators[startsWith(term, operators)]
70 operator <- if (length(found)) found[[1]] else ""
71 pattern <- substring(term, nchar(operator) + 1L)
72 value <- parse_version(pattern)
73 if (!is.null(value) && !is.null(value$pre)) {
74 core <- value$core
75 k <- 3L
76 gates[[length(gates) + 1L]] <- core
77 } else {
78 parts <- split_keep(pattern, ".")
79 if (length(parts) < 1L || length(parts) > 3L) return(NULL)
80 numbers <- integer()
81 wildcard <- FALSE
82 for (part in parts) {
83 if (part %in% c("x", "*")) wildcard <- TRUE else if (!wildcard && grepl(number_pattern, part, perl = TRUE)) {
84 numbers <- c(numbers, as.integer(part))
85 } else return(NULL)
86 }
87 k <- length(numbers)
88 core <- c(numbers, rep(0L, 3L - k))
89 value <- list(core = core, pre = NULL)
90 }
91 upper <- function(i) {
92 values <- core
93 values[[i]] <- values[[i]] + 1L
94 if (i < 3L) values[(i + 1L):3L] <- 0L
95 list(core = values, pre = NULL)
96 }
97 add <- function(op, bound) comparators[[length(comparators) + 1L]] <<- list(op = op, bound = bound)
98 if (k == 0L) {
99 if (operator %in% c(">", "<")) add("<", list(core = c(0L, 0L, 0L), pre = "0"))
100 } else if (operator %in% c("", "=")) {
101 if (k == 3L) add("=", value) else {
102 add(">=", value)
103 add("<", upper(k))
104 }
105 } else if (operator == ">") {
106 add(if (k == 3L) ">" else ">=", if (k == 3L) value else upper(k))
107 } else if (operator == "<=") {
108 add(if (k == 3L) "<=" else "<", if (k == 3L) value else upper(k))
109 } else if (operator %in% c(">=", "<")) {
110 add(operator, value)
111 } else if (operator == "~") {
112 add(">=", value)
113 add("<", upper(if (k == 1L) 1L else 2L))
114 } else if (operator == "^") {
115 i <- if (core[[1]] != 0L || k == 1L) 1L else if (core[[2]] != 0L || k == 2L) 2L else 3L
116 add(">=", value)
117 add("<", upper(i))
118 }
119 }
120 alternatives[[length(alternatives) + 1L]] <- list(comparators = comparators, gates = gates)
121 }
122 alternatives

124constraint_matches <- function(constraint, version) {
125 for (alternative in constraint) {
126 if (!is.null(version$pre) && !any(vapply(alternative$gates, function(core) identical(core, version$core), logical(1)))) next
127 valid <- TRUE
128 for (comparator in alternative$comparators) {
129 comparison <- compare_version(version, comparator$bound)
130 ok <- switch(comparator$op, `=` = comparison == 0L, `>` = comparison > 0L,
131 `>=` = comparison >= 0L, `<` = comparison < 0L, `<=` = comparison <= 0L)
132 if (!ok) {
133 valid <- FALSE
134 break
135 }
136 }
137 if (valid) return(TRUE)
138 }
139 FALSE
140}

200 search <- function(requirements) {
201 if (!length(requirements)) return(TRUE)
202 r <- requirements[[1]]
203 name <- r$name
204 text <- r$constraint
205 chain <- r$chain
206 rest <- requirements[-1]
207 if (!is.null(selected[[name]])) {
208 if (matches(text, selected[[name]])) return(search(rest))
209 save(chain, paste0(name, "@", selected[[name]], " was selected by:"), reasons[[name]])
210 return(FALSE)
211 }
212 if (name %in% state$held) {
213 value <- state$installed[[name]]
214 if (!matches(text, value) || !usable(name, value)) {
215 save(chain, paste0(name, "@", value, " is held"))
216 return(FALSE)
217 }
218 candidates <- value
219 } else {
220 candidates <- versions(name)
221 candidates <- candidates[vapply(candidates, function(v) matches(text, v) && usable(name, v), logical(1))]
222 current <- state$installed[[name]]
223 if (name %in% preferred && !is.null(current) && current %in% candidates) candidates <- c(current, setdiff(candidates, current))
224 if (!length(candidates)) {
225 save(chain, paste("no version of", name, "matches", text))
226 return(FALSE)
227 }
228 }
229 for (value in candidates) {
230 conflicting <- NULL
231 for (q in ordered(names(selected))) {
232 ours <- any(vapply(record(name, value)$conflicts, function(d) d$name == q && matches(d$constraint, selected[[q]]), logical(1)))
233 theirs <- any(vapply(record(q, selected[[q]])$conflicts, function(d) d$name == name && matches(d$constraint, value), logical(1)))
234 if (ours || theirs) {
235 conflicting <- q
236 break
237 }
238 }
239 if (!is.null(conflicting)) {
240 save(chain, paste0(name, "@", value, " conflicts with ", conflicting, "@", selected[[conflicting]], ", which was selected by:"), reasons[[conflicting]])
241 next
242 }
243 selected[[name]] <<- value
244 reasons[[name]] <<- chain
245 extra <- lapply(record(name, value)$deps, function(d) list(name = d$name, constraint = d$constraint,
246 chain = c(chain, paste0(name, "@", value, " requires ", d$name, " ", d$constraint))))
247 if (search(c(extra, rest))) return(TRUE)
248 selected[[name]] <<- NULL
249 reasons[[name]] <<- NULL
250 }
251 FALSE
252 }
253 requirements <- lapply(requests, function(r) list(name = r$name, constraint = r$constraint, chain = paste("requested", r$name, r$constraint)))
254 if (search(requirements)) list(installed = selected, failure = NULL) else list(installed = NULL, failure = failure)

256plan <- function(new) {
257 old <- state$installed
258 arranged <- character()
259 counts <- c(install = 0L, upgrade = 0L, downgrade = 0L, remove = 0L)
260 while (length(arranged) < length(new)) {
261 eligible <- ordered(setdiff(names(new), arranged))
262 eligible <- eligible[vapply(eligible, function(n) all(vapply(dependencies(n, new), function(d) d$name %in% arranged, logical(1))), logical(1))]
263 name <- eligible[[1]]
264 arranged <- c(arranged, name)
265 value <- new[[name]]
266 if (is.null(old[[name]])) {
267 say("install ", name, "@", value)
268 counts[["install"]] <- counts[["install"]] + 1L
269 } else if (old[[name]] != value) {
270 kind <- if (compare_version(parse_version(value), parse_version(old[[name]])) > 0L) "upgrade" else "downgrade"
271 say(kind, " ", name, " ", old[[name]], " -> ", value)
272 counts[[kind]] <- counts[[kind]] + 1L
273 }
274 }
275 remaining <- setdiff(names(old), names(new))
276 while (length(remaining)) {
277 eligible <- ordered(remaining)
278 eligible <- eligible[!vapply(eligible, function(n) any(vapply(remaining, function(q) {
279 any(vapply(dependencies(q, old), function(d) d$name == n, logical(1)))
280 }, logical(1))), logical(1))]
281 name <- eligible[[1]]
282 say("remove ", name, "@", old[[name]])
283 counts[["remove"]] <- counts[["remove"]] + 1L
284 remaining <- setdiff(remaining, name)
285 }
286 if (sum(counts) == 0L) say("no changes") else say(counts[[1]], " installed, ", counts[[2]], " upgraded, ", counts[[3]], " downgraded, ", counts[[4]], " removed")
287}

340execute <- function(line) {
341 head <- bounded(line, 2L)
342 command <- head[[1]]
343 rest <- if (length(head) == 2L) head[[2]] else ""
344 args <- if (rest == "") character() else strsplit(rest, " ", fixed = TRUE)[[1]]
345 name <- if (length(args)) args[[1]] else NULL
346 switch(command,
347 PKG = {
348 clauses <- strsplit(rest, "; ", fixed = TRUE)[[1]]
349 pair <- strsplit(clauses[[1]], " ", fixed = TRUE)[[1]]
350 name <- pair[[1]]
351 value <- pair[[2]]
352 parsed <- version(value)
353 if (!is.null(record(name, value))) fail(paste0(name, "@", value, " is already registered"))
354 deps <- list()
355 conflicts <- list()
356 for (clause in clauses[-1]) {
357 fields <- bounded(clause, 3L)
358 constraint(fields[[3]])
359 entry <- list(name = fields[[2]], constraint = fields[[3]])
360 if (fields[[1]] == "DEP") deps[[length(deps) + 1L]] <- entry else conflicts[[length(conflicts) + 1L]] <- entry
361 }
362 state$registry[[name]][[value]] <- list(version = parsed, deps = deps, conflicts = conflicts, yanked = FALSE)
363 say("registered ", name, "@", value)
364 },
365 YANK = show_or_yank(command, args),
366 INFO = show_or_yank(command, args),
367 COMPARE = {
368 a <- version(name)
369 b <- version(args[[2]])
370 c <- compare_version(a, b)
371 say(name, " ", if (c < 0L) "<" else if (c == 0L) "=" else ">", " ", args[[2]])
372 },
373 MATCH = {
374 fields <- bounded(rest, 2L)
375 value <- version(fields[[1]])
376 say(if (constraint_matches(constraint(fields[[2]]), value)) "yes" else "no")
377 },
378 RANGE = say(constraint_text(constraint(rest))),
379 VERSIONS = {
380 fields <- bounded(rest, 2L)
381 name <- fields[[1]]
382 text <- if (length(fields) == 2L) fields[[2]] else NULL
383 if (!is.null(text)) constraint(text)
384 if (is.null(state$registry[[name]])) fail(paste("unknown package", name))
385 selected <- versions(name)
386 selected <- selected[vapply(selected, function(v) is.null(text) || matches(text, v), logical(1))]
387 if (!length(selected)) say("no matching versions")
388 for (value in selected) say(display(name, value))
389 },
390 DEPENDENTS = {
391 value <- if (length(args) == 2L) args[[2]] else NULL
392 if (!is.null(value)) version(value)
393 lines <- character()
394 for (n in ordered(names(state$registry))) {
395 for (v in versions(n)) {
396 for (d in record(n, v)$deps) {
397 if (d$name == name && (is.null(value) || matches(d$constraint, value))) lines <- c(lines, paste0(n, "@", v, " requires ", name, " ", d$constraint))
398 }
399 }
400 }
401 if (!length(lines)) lines <- "no dependents"
402 for (text in lines) say(text)
403 },
404 INSTALL = install_or_simulate(rest, FALSE),
405 SIMULATE = install_or_simulate(rest, TRUE),
406 REMOVE = {
407 if (!(name %in% request_names())) fail(if (is.null(state$installed[[name]])) paste(name, "is not installed") else paste(name, "is required by", required_by(name)))
408 requests <- Filter(function(r) r$name != name, state$requests)
409 keep <- request_names(requests)
410 at <- 1L
411 while (at <= length(keep)) {
412 for (d in dependencies(keep[[at]])) if (!(d$name %in% keep)) keep <- c(keep, d$name)
413 at <- at + 1L
414 }
415 if (name %in% keep) {
416 say(name, "@", state$installed[[name]], " is still required by ", required_by(name))
417 update(requests, state$installed)
418 } else {
419 installed <- state$installed[intersect(names(state$installed), keep)]
420 plan(installed)
421 update(requests, installed)
422 }
423 },
424 UPGRADE = {
425 if (!is.null(name)) require_installed(name)
426 preferred <- if (is.null(name)) character() else setdiff(names(state$installed), name)
427 resolve_and_plan(state$requests, preferred)
428 },
429 ROLLBACK = {
430 if (!length(state$history)) fail("nothing to roll back")
431 snapshot <- state$history[[length(state$history)]]
432 state$history <- head(state$history, -1L)
433 plan(snapshot$installed)
434 state$requests <- snapshot$requests
435 state$installed <- snapshot$installed
436 state$held <- intersect(state$held, names(snapshot$installed))
437 },
438 HOLD = {
439 require_installed(name)
440 if (name %in% state$held) fail(paste(name, "is already held"))
441 state$held <- c(state$held, name)
442 say("held ", name, "@", state$installed[[name]])
443 },
444 UNHOLD = {
445 if (!(name %in% state$held)) fail(paste(name, "is not held"))
446 state$held <- setdiff(state$held, name)
447 say("released ", name)
448 },
449 WHY = {
450 require_installed(name)
451 if (name %in% request_names()) {
452 say(name, "@", state$installed[[name]], " is explicitly installed")
453 } else {
454 queue <- ordered(request_names())
455 parent <- list()
456 seen <- queue
457 at <- 1L
458 while (at <= length(queue)) {
459 n <- queue[[at]]
460 at <- at + 1L
461 deps <- ordered(vapply(dependencies(n), function(d) d$name, character(1)))
462 for (d in deps) {
463 if (d %in% seen) next
464 seen <- c(seen, d)
465 parent[[d]] <- n
466 queue <- c(queue, d)
467 }
468 }
469 path <- name
470 while (!is.null(parent[[path[[1]]]])) path <- c(parent[[path[[1]]]], path)
471 say(paste(vapply(path, function(n) paste0(n, "@", state$installed[[n]]), character(1)), collapse = " -> "))
472 }
473 },
474 TREE = tree(name),
475 OUTDATED = {
476 lines <- character()
477 for (n in ordered(names(state$installed))) {
478 current <- state$installed[[n]]
479 relevant <- vapply(Filter(function(r) r$name == n, state$requests), function(r) r$constraint, character(1))
480 for (q in names(state$installed)) {
481 for (d in dependencies(q)) if (d$name == n) relevant <- c(relevant, d$constraint)
482 }
483 candidates <- versions(n)
484 wanted <- candidates[vapply(candidates, function(v) (!record(n, v)$yanked || v == current) && all(vapply(relevant, function(c) matches(c, v), logical(1))), logical(1))][[1]]
485 stable <- candidates[vapply(candidates, function(v) !record(n, v)$yanked && is.null(record(n, v)$version$pre), logical(1))]
486 latest <- if (length(stable)) stable[[1]] else NULL
487 if (wanted != current || (!is.null(latest) && compare_version(parse_version(latest), parse_version(current)) > 0L)) lines <- c(lines, paste(n, current, wanted, if (is.null(latest)) "-" else latest))
488 }
489 if (!length(lines)) lines <- "all packages are up to date"
490 for (text in lines) say(text)
491 },
492 LOCK = {
493 if (!length(state$installed)) say("(empty)")
494 for (n in ordered(names(state$installed))) say(n, "@", state$installed[[n]])
495 },
496 STATUS = {
497 if (!length(state$requests)) say("no requests")
498 for (r in state$requests) say(r$name, " ", r$constraint, " -> ", state$installed[[r$name]])
499 say(length(state$installed), " installed, ", length(state$held), " held, ", length(state$history), " snapshots")
500 },
501 ADVISORY = {
502 fields <- bounded(rest, 3L)
503 id <- fields[[1]]
504 if (!is.null(state$advisories[[id]])) fail(paste("advisory", id, "already exists"))
505 constraint(fields[[3]])
506 state$advisories[[id]] <- list(name = fields[[2]], constraint = fields[[3]])
507 say("advisory ", id, " recorded")
508 },
509 AUDIT = {
510 if (identical(name, "FIX")) resolve_and_plan(state$requests, names(state$installed), TRUE) else {
511 lines <- character()
512 for (n in ordered(names(state$installed))) {
513 current <- state$installed[[n]]
514 ids <- applicable(n, current)
515 if (!length(ids)) next
516 candidates <- rev(versions(n))
517 fixes <- candidates[vapply(candidates, function(v) compare_version(parse_version(v), parse_version(current)) > 0L &&
518 is.null(record(n, v)$version$pre) && !record(n, v)$yanked && !length(applicable(n, v)), logical(1))]
519 fix <- if (length(fixes)) paste0("fix: ", fixes[[1]]) else "no fix"
520 lines <- c(lines, paste0(n, "@", current, ": ", paste(ids, collapse = ", "), " (", fix, ")"))
521 }
522 if (!length(lines)) lines <- "no known vulnerabilities"
523 for (text in lines) say(text)
524 }
525 }

528show_or_yank <- function(command, args) {
529 name <- args[[1]]
530 value <- args[[2]]
531 entry <- record(name, value)
532 if (is.null(entry)) fail(paste0(name, "@", value, " is not registered"))
533 if (command == "YANK") {
534 if (entry$yanked) fail(paste0(name, "@", value, " is already yanked"))
535 state$registry[[name]][[value]]$yanked <- TRUE
536 say("yanked ", name, "@", value)
537 } else {
538 say(display(name, value))
539 for (d in entry$deps) say(" requires ", d$name, " ", d$constraint)
540 for (d in entry$conflicts) say(" conflicts with ", d$name, " ", d$constraint)
541 for (id in applicable(name, value)) say(" advisory ", id)
542 }
543}
544install_or_simulate <- function(rest, simulate) {
545 fields <- bounded(rest, 2L)
546 name <- fields[[1]]
547 text <- if (length(fields) == 2L) fields[[2]] else "*"
548 constraint(text)
549 if (is.null(state$registry[[name]])) fail(paste("unknown package", name))
550 requests <- state$requests
551 index <- match(name, request_names(requests))
552 entry <- list(name = name, constraint = text)
553 if (is.na(index)) requests[[length(requests) + 1L]] <- entry else requests[[index]] <- entry
554 resolve_and_plan(requests, names(state$installed), FALSE, simulate)
555}
