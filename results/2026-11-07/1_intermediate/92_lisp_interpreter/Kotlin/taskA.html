8private class Env(val parent: Env? = null){
9 val frame = mutableMapOf<String, Any>()
10 fun find(n: String): Env{
11 var e: Env? = this;
12 while (e != null){
13 if (e.frame.containsKey(n))return e;
14 e = e.parent
15 };
16 error("unbound variable: $n")
17 }

19 val v = find(n).frame.getValue(n);
20 if (v === Uninitialized)error("uninitialized variable: $n");
21 return v

24private class Procedure(var name: String?, val minimum: Int, val variadic: Boolean, val rest: String? = null, val params: List<String> = emptyList(), val body: List<Any> = emptyList(), val env: Env? = null, val builtin: ((List<Any>)->Any)? = null){
25 fun call(a: List<Any>): Any{
26 if (a.size<minimum || (!variadic && a.size != minimum))error("${name?:"lambda"}: wrong number of arguments (expected ${if(variadic)"at least " else ""}$minimum, got ${a.size})")
27 builtin?.let{
28 return it(a)
29 }
30 val e = Env(env);
31 params.forEachIndexed{
32 i, n->e.frame[n] = a[i]
33 };
34 if (rest != null)e.frame[rest] = makeList(a.drop(minimum));
35 return evalBody(body, e)
36 }
37}
38private fun makeList(a: List<Any>, tail: Any = Nil): Any{
39 var v = tail;
40 for (x in a.asReversed())v = Pair(x, v);
41 return v
42}
43private fun items(value: Any): List<Any>?{
44 var v = value;
45 val a = mutableListOf<Any>();
46 while (v is Pair){
47 a.add(v.car);
48 v = v.cdr
49 };
50 return if (v === Nil)a else null
51}
52private fun truth(v: Any) = v != false
53private fun repr(value: Any, quoted: Boolean = true): String{
54 if (value === Nil)return "()";
55 if (value === VoidValue)return "#<void>"
56 when(value){
57 is Boolean->return if (value)"#t" else "#f";
58 is Long->return value.toString();
59 is String->return if (quoted)"\""+value.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n")+"\"" else value;
60 is Symbol->return value.name;
61 is Procedure->return value.name?.let{
62 "#<procedure $it>"
63 }?: "#<procedure>"
64 }
65 var v = value;
66 val a = mutableListOf<String>();
67 while (v is Pair){
68 a.add(repr(v.car, quoted));
69 v = v.cdr
70 };
71 if (v !== Nil){
72 a.add(".");
73 a.add(repr(v, quoted))
74 };
75 return "("+a.joinToString(" ")+")"
76}
77private fun same(a: Any, b: Any, deep: Boolean = false): Boolean{
78 if (a is Pair && b is Pair)return if (deep)same(a.car, b.car, true) && same(a.cdr, b.cdr, true) else a === b;
79 return a == b
80}
81private data class Parameters(val names: List<String>, val rest: String?)
82private fun parameters(value: Any): Parameters?{
83 var v = value;
84 val names = mutableListOf<String>();
85 while (v is Pair){
86 val s = v.car as? Symbol?: return null;
87 names.add(s.name);
88 v = v.cdr
89 };
90 val r = (v as? Symbol)?.name;
91 if (v !== Nil && r == null)return null;
92 val all = names+listOfNotNull(r);
93 return if (all.distinct().size == all.size)Parameters(names, r) else null
94}
95private fun closure(p: Parameters, b: List<Any>, e: Env, n: String? = null) = Procedure(n, p.names.size, p.rest != null, p.rest, p.names, b, e)
96private data class Binding(val name: String, val expression: Any)
97private fun bindings(v: Any, unique: Boolean): List<Binding>?{
98 val a = items(v)?: return null;
99 val bs = mutableListOf<Binding>();
100 val seen = mutableSetOf<String>();
101 for (x in a){
102 val b = items(x)?: return null;
103 if (b.size != 2)return null;
104 val s = b[0] as? Symbol?: return null;
105 if (unique && !seen.add(s.name))return null;
106 bs.add(Binding(s.name, b[1]))
107 };
108 return bs
109}
110private fun shape(ok: Boolean, x: Any){
111 if (!ok)error("bad syntax: "+repr(x))
112}
113private fun evalBody(a: List<Any>, e: Env): Any{
114 var v: Any = VoidValue;
115 for (x in a)v = evaluate(x, e);
116 return v
117}
118private fun evaluate(x: Any, e: Env): Any{
119 if (x is Symbol)return e.get(x.name);
120 if (x !is Pair){
121 shape(x !== Nil, x);
122 return x
123 };
124 val parsed = items(x);
125 shape(parsed != null, x);
126 val a = parsed!!;
127 val n = a.size;
128 val h = (a[0] as? Symbol)?.name?: ""
129 when(h){
130 "quote"->{
131 shape(n == 2, x);
132 return a[1]
133 }
134 "if"->{
135 shape(n == 3 || n == 4, x);
136 return if (truth(evaluate(a[1], e)))evaluate(a[2], e) else if (n == 4)evaluate(a[3], e) else VoidValue
137 }
138 "begin"->return evalBody(a.drop(1), e)
139 "lambda"->{
140 val p = if (n >= 3)parameters(a[1]) else null;
141 shape(p != null, x);
142 return closure(p!!, a.drop(2), e)
143 }
144 "define"->{
145 if (n>1 && a[1] is Symbol){
146 shape(n == 3, x);
147 val name = (a[1] as Symbol).name;
148 val v = evaluate(a[2], e);
149 if (v is Procedure && v.builtin == null && v.name == null)v.name = name;
150 e.frame[name] = v
151 }
152 else{
153 val t = a.getOrNull(1) as? Pair;
154 val name = t?.car as? Symbol;
155 val p = if (n >= 3 && name != null)parameters(t!!.cdr) else null;
156 shape(p != null, x);
157 e.frame[name!!.name] = closure(p!!, a.drop(2), e, name.name)
158 };
159 return VoidValue
160 }
161 "set!"->{
162 shape(n == 3 && a[1] is Symbol, x);
163 val name = (a[1] as Symbol).name;
164 val v = evaluate(a[2], e);
165 e.find(name).frame[name] = v;
166 return VoidValue
167 }
168 "let", "let*", "letrec"->{
169 val named = h == "let" && n>1 && a[1] is Symbol;
170 val at = if (named)2 else 1;
171 val bs = if (n >= at+2)bindings(a[at], h != "let*") else null;
172 shape(bs != null, x);
173 val bindings = bs!!;
174 var scope = Env(e)
175 if (h == "let*"){
176 if (bindings.isNotEmpty())scope = e;
177 for (b in bindings){
178 val v = evaluate(b.expression, scope);
179 scope = Env(scope);
180 scope.frame[b.name] = v
181 }
182 }
183 else if (h == "letrec"){
184 for (b in bindings)scope.frame[b.name] = Uninitialized;
185 for (b in bindings)scope.frame[b.name] = evaluate(b.expression, scope)
186 }
187 else{
188 val vs = bindings.map{
189 evaluate(it.expression, e)
190 };
191 if (named){
192 val name = (a[1] as Symbol).name;
193 val f = closure(Parameters(bindings.map{
194 it.name
195 }, null), a.drop(3), scope, name);
196 scope.frame[name] = f;
197 return f.call(vs)
198 };
199 bindings.forEachIndexed{
200 i, b->scope.frame[b.name] = vs[i]
201 }
202 }
203 return evalBody(a.drop(at+1), scope)
204 }
205 "cond"->{
206 val cs = a.drop(1).map{
207 items(it)
208 };
209 cs.forEachIndexed{
210 i, c->shape(c != null && c.isNotEmpty(), x);
211 if (c!![0] == Symbol("else"))shape(c.size >= 2 && i == cs.size-1, x)
212 };
213 for (c0 in cs){
214 val c = c0!!;
215 if (c[0] == Symbol("else"))return evalBody(c.drop(1), e);
216 val v = evaluate(c[0], e);
217 if (truth(v))return if (c.size == 1)v else evalBody(c.drop(1), e)
218 };
219 return VoidValue
220 }
221 "and"->{
222 var v: Any = true;
223 for (z in a.drop(1)){
224 v = evaluate(z, e);
225 if (!truth(v))break
226 };
227 return v
228 }
229 "or"->{
230 for (z in a.drop(1)){
231 val v = evaluate(z, e);
232 if (truth(v))return v
233 };
234 return false
235 }
236 "when", "unless"->{
237 shape(n >= 3, x);
238 return if (truth(evaluate(a[1], e)) == (h == "when"))evalBody(a.drop(2), e) else VoidValue
239 }
240 }
241 val f = evaluate(a[0], e);
242 val args = a.drop(1).map{
243 evaluate(it, e)
244 };
245 if (f !is Procedure)error("not a procedure: "+repr(f));
246 return f.call(args)
247}

257private fun expect(n: String, v: Any, k: String): Any{
258 val ok = when(k){
259 "integer"->v is Long;
260 "string"->v is String;
261 "pair"->v is Pair;
262 "procedure"->v is Procedure;
263 else->items(v) != null
264 };
265 if (!ok)error("$n: expected $k, got "+repr(v));
266 return v

272private fun global(): Env{
273 val e = Env();
274 fun add(n: String, m: Int, r: Boolean, fn: (List<Any>)->Any){
275 e.frame[n] = Procedure(n, m, r, builtin = fn)
276 }
277 for (name in listOf("+", "*", "-", "quotient", "remainder", "modulo", "abs", "min", "max", "gcd", "=", "<", ">", "<=", ">=")){
278 val fixed = if (name in listOf("quotient", "remainder", "modulo"))2 else if (name == "abs")1 else -1
279 add(name, if (fixed >= 0)fixed else if (name in listOf("+", "*", "gcd"))0 else 1, fixed<0){
280 a->
281 val z = a.map{
282 expect(name, it, "integer") as Long
283 };
284 val v = z.firstOrNull()?: 0L
285 when(name){
286 "+"->z.sum();
287 "*"->z.fold(1L){
288 s, x->s*x
289 };
290 "-"->if (z.size == 1)-v else v-z.drop(1).sum();
291 "abs"->kotlin.math.abs(v);
292 "min"->z.min();
293 "max"->z.max();
294 "gcd"->z.fold(0L){
295 g0, x->var g = g0;
296 var b = kotlin.math.abs(x);
297 while (b != 0L){
298 val r = g%b;
299 g = b;
300 b = r
301 };
302 g
303 };
304 "quotient", "remainder", "modulo"->{
305 val b = z[1];
306 if (b == 0L)error(name+": division by zero");
307 val r = v%b;
308 if (name == "quotient")v/b else if (name == "modulo" && r != 0L && ((r<0) != (b<0)))r+b else r
309 }
310 else->z.zipWithNext().all{
311 (l, r)->when(name){
312 "="->l == r;
313 "<"->l<r;
314 ">"->l>r;
315 "<="->l <= r;
316 else->l >= r
317 }
318 }
319 }
320 }
321 }
322 add("not", 1, false){
323 !truth(it[0])
324 };
325 add("eq?", 2, false){
326 same(it[0], it[1])
327 };
328 add("equal?", 2, false){
329 same(it[0], it[1], true)
330 }
331 for (n in listOf("null?", "pair?", "list?", "number?", "symbol?", "procedure?"))add(n, 1, false){
332 a->when(n){
333 "null?"->a[0] === Nil;
334 "pair?"->a[0] is Pair;
335 "list?"->items(a[0]) != null;
336 "number?"->a[0] is Long;
337 "symbol?"->a[0] is Symbol;
338 else->a[0] is Procedure
339 }
340 }
341 add("cons", 2, false){
342 Pair(it[0], it[1])
343 };
344 add("list", 0, true){
345 makeList(it)
346 }
347 for (n in listOf("car", "cdr", "cadr"))add(n, 1, false){
348 a->val p = expect(n, a[0], "pair") as Pair;
349 when(n){
350 "car"->p.car;
351 "cdr"->p.cdr;
352 else->(expect(n, p.cdr, "pair") as Pair).car
353 }
354 }
355 for (n in listOf("length", "reverse", "list-ref"))add(n, if (n == "list-ref")2 else 1, false){
356 a->val z = asList(n, a[0]);
357 when(n){
358 "length"->z.size.toLong();
359 "reverse"->makeList(z.asReversed());
360 else->{
361 val k = expect(n, a[1], "integer") as Long;
362 if (k<0 || k >= z.size)error("list-ref: index out of range: $k");
363 z[k.toInt()]
364 }
365 }
366 }
367 add("append", 0, true){
368 a->if (a.isEmpty())Nil else makeList(a.dropLast(1).flatMap{
369 asList("append", it)
370 }, a.last())
371 }
372 for (n in listOf("map", "filter", "fold-left", "apply"))add(n, if (n == "fold-left")3 else 2, n != "filter"){
373 a->
374 val f = expect(n, a[0], "procedure") as Procedure
375 if (n == "apply")f.call(a.drop(1).dropLast(1)+asList(n, a.last()))
376 else{
377 val ls = a.drop(if (n == "fold-left")2 else 1).map{
378 asList(n, it)
379 };
380 val out = mutableListOf<Any>();
381 var acc = a[1];
382 for (i in 0 until ls.minOf{
383 it.size
384 }){
385 if (n == "filter"){
386 if (truth(f.call(listOf(ls[0][i]))))out.add(ls[0][i])
387 }else{
388 val args = ls.map{
389 it[i]
390 };
391 val v = f.call(if (n == "fold-left")listOf(acc)+args else args);
392 if (n == "map")out.add(v) else acc = v
393 }
394 };
395 if (n == "fold-left")acc else makeList(out)
396 }
397 }
398 add("string-append", 0, true){
399 a->a.joinToString(""){
400 expect("string-append", it, "string") as String
401 }
402 };
403 add("number->string", 1, false){
404 expect("number->string", it[0], "integer").toString()
405 }
406 add("display", 1, false){
407 write(repr(it[0], false));
408 VoidValue
409 };
410 add("newline", 0, false){
411 write("\n");
412 VoidValue
413 };
414 add("error", 1, true){
415 a->error((expect("error", a[0], "string") as String)+a.drop(1).joinToString(""){
416 " "+repr(it)
417 })
418 };
419 return e
420}
421private data class Token(val kind: Char, val value: Any? = null)
422private class Reader(val lines: List<String>){
423 var row = 0;
424 var col = 0
425 fun skip(){
426 row++;
427 col = 0
428 }
429 fun end(): Boolean{
430 while (row<lines.size){
431 val s = lines[row];
432 while (col<s.length && s[col] == ' ')col++;
433 if (col<s.length && s[col] != ';')return false;
434 skip()
435 };
436 return true
437 }
438 fun token(): Token{
439 if (end())error("unexpected end of input");
440 val s = lines[row];
441 var c = s[col++];
442 if (c in "()'")return Token(c);
443 if (c == '"'){
444 val v = StringBuilder();
445 while (col<s.length){
446 c = s[col++];
447 if (c == '"')return Token('v', v.toString());
448 if (c == '\\'){
449 if (col == s.length)break;
450 c = s[col++];
451 if (c !in "\"\\n")error("bad escape");
452 if (c == 'n')c = '\n'
453 };
454 v.append(c)
455 };
456 error("unterminated string")
457 };
458 val start = col-1;
459 while (col<s.length && s[col] !in " ()'\";")col++;
460 val a = s.substring(start, col);
461 if (a == ".")return Token('.');
462 if (a.matches(Regex("[+-]?[0-9]+")))return Token('v', a.toLong());
463 if (a == "#t" || a == "#f")return Token('v', a == "#t");
464 if (a.startsWith('#'))error("bad token: $a");
465 return Token('v', Symbol(a))
466 }
467 fun datum(t: Token = token()): Any{
468 if (t.kind == 'v')return t.value!!;
469 if (t.kind == '\'')return makeList(listOf(Symbol("quote"), datum()));
470 if (t.kind != '(')error("unexpected ${t.kind}");
471 val a = mutableListOf<Any>();
472 while (true){
473 val next = token();
474 if (next.kind == ')')return makeList(a);
475 if (next.kind == '.' && a.isNotEmpty()){
476 val tail = datum();
477 if (token().kind != ')')error("expected )");
478 return makeList(a, tail)
479 };
480 a.add(datum(next))
481 }
482 }
483}
484fun main(){
485 val lines = mutableListOf<String>();
486 val input = BufferedReader(InputStreamReader(System.`in`));
487 while (true){
488 val s = input.readLine()?: break;
489 if (s == "(exit)")break;
490 lines.add(s)
491 };
492 val r = Reader(lines);
493 val e = global();
494 while (!r.end()){
495 val x: Any;
496 try{
497 x = r.datum()
498 }catch (err: IllegalStateException){
499 line("Error: read: "+err.message);
500 if (err.message == "unexpected end of input")break;
501 r.skip();
502 continue
503 };
504 try{
505 val v = evaluate(x, e);
506 if (v !== VoidValue)line(repr(v))
507 }catch (err: IllegalStateException){
508 line("Error: "+err.message)
509 }
510 };
511 if (mid)write("\n")
512}
