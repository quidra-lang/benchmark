6class Env
7 attr_reader :frame, :parent
8 def initialize(parent = nil)
9 @parent, @frame = parent, {}
10 end

12 def find(name)
13 e = self
14 while e
15 return e if e.frame.key?(name)
16 e = e.parent
17 end
18
19 raise "unbound variable: #{name}"
20 end

22 def get(name)
23 v = find(name).frame[name]
24 raise "uninitialized variable: #{name}" if v.equal?(UNINIT)
25 v
26 end
27end

29class Procedure
30 attr_accessor :name
31 attr_reader :builtin
32 def initialize(name, minimum, rest, params = [], body = [], env = nil, &builtin)
33 @name, @minimum, @rest, @params, @body, @env, @builtin = name, minimum, rest, params, body, env, builtin
34 end
35
36 def call(args)
37 if args.length < @minimum || (!@rest && args.length != @minimum)
38 raise(
39 "#{@name || "lambda"}: wrong number of arguments (expected #{@rest ? "at least " : ""}#{@minimum}, got #{args.length})"
40 )
41 end
42
43 return @builtin.call(args) if @builtin
44 scope = Env.new(@env)
45 @params.zip(args).each { |name, value| scope.frame[name] = value }
46 scope.frame[@rest] = make_list(args.drop(@minimum)) if @rest.is_a?(Symbol)
47 evaluate_body(@body, scope)
48 end
49end

51def make_list(a, tail = NIL)
52 a.reverse_each { |x| tail = Pair.new(x, tail) }
53 tail
54end

56def list_items(v)
57 a = []
58 while v.is_a?(Pair)
59 a << v.car
60 v = v.cdr
61 end
62
63 v.equal?(NIL) ? a : nil
64end

66def represent(v, quoted = true)
67 return "()" if v.equal?(NIL)
68 return "#<void>" if v.equal?(VOID)
69 case v
70 when true
71 "#t"
72 when false
73 "#f"
74 when Integer, Symbol
75 v.to_s
76 when String
77 quoted ? "\"" + v.each_char.map { |c| {"\"" => "\\\"", "\\" => "\\\\", "\n" => "\\n"}.fetch(c, c) }.join + "\"" : v
78 when Procedure
79 v.name ? "#<procedure #{v.name}>" : "#<procedure>"
80 when Pair
81 a = []
82 while v.is_a?(Pair)
83 a << represent(v.car, quoted)
84 v = v.cdr
85 end
86
87 a += [".", represent(v, quoted)] unless v.equal?(NIL)
88 "(" + a.join(" ") + ")"
89 end
90end

92def same(a, b, deep = false)
93 return same(a.car, b.car, true) && same(a.cdr, b.cdr, true) if deep && a.is_a?(Pair) && b.is_a?(Pair)
94 return a == b if a.class == b.class && [Integer, String, Symbol, TrueClass, FalseClass].include?(a.class)
95 a.equal?(b)
96end

98def parameters(v)
99 names = []
100 while v.is_a?(Pair)
101 return nil unless v.car.is_a?(Symbol)
102 names << v.car
103 v = v.cdr
104 end
105
106 return nil unless v.equal?(NIL) || v.is_a?(Symbol)
107 rest = v.equal?(NIL) ? nil : v
108 all = rest ? names + [rest] : names
109 all.uniq.length == all.length ? [names, rest] : nil
110end

112def bindings(v, unique)
113 a = list_items(v)
114 return nil unless a
115 a = a.map { |x| list_items(x) }
116 return nil unless a.all? { |b| b && b.length == 2 && b[0].is_a?(Symbol) }
117 return nil if unique && a.map(&:first).uniq.length != a.length
118 a
119end

121def shape(ok, x)
122 raise "bad syntax: #{represent(x)}" unless ok
123end

125def compound(p, body, env, name = nil)
126 Procedure.new(name, p[0].length, p[1], p[0], body, env)
127end

129def evaluate_body(a, e)
130 a.reduce(VOID) { |_, x| evaluate(x, e) }
131end

133def evaluate(x, e)
134 return e.get(x) if x.is_a?(Symbol)
135 unless x.is_a?(Pair)
136 shape(!x.equal?(NIL), x)
137 return x
138 end
139
140 a = list_items(x)
141 shape(a, x)
142 n = a.length
143 case a[0]
144 when :quote
145 shape(n == 2, x)
146 return a[1]
147 when :if
148 shape([3, 4].include?(n), x)
149 return evaluate(a[1], e) != false ? evaluate(a[2], e) : n == 4 ? evaluate(a[3], e) : VOID
150 when :begin
151 return evaluate_body(a.drop(1), e)
152 when :lambda
153 p = n >= 3 ? parameters(a[1]) : nil
154 shape(p, x)
155 return compound(p, a.drop(2), e)
156 when :define
157 if a[1].is_a?(Symbol)
158 shape(n == 3, x)
159 v = evaluate(a[2], e)
160 v.name = a[1] if v.is_a?(Procedure) && !v.builtin && !v.name
161 e.frame[a[1]] = v
162 else
163 t = a[1]
164 p = n >= 3 && t.is_a?(Pair) && t.car.is_a?(Symbol) ? parameters(t.cdr) : nil
165 shape(p, x)
166 e.frame[t.car] = compound(p, a.drop(2), e, t.car)
167 end
168
169 return VOID
170 when :"set!"
171 shape(n == 3 && a[1].is_a?(Symbol), x)
172 v = evaluate(a[2], e)
173 e.find(a[1]).frame[a[1]] = v
174 return VOID
175 when :let, :"let*", :letrec
176 named = a[0] == :let && a[1].is_a?(Symbol)
177 at = named ? 2 : 1
178 bs = n >= at + 2 ? bindings(a[at], a[0] != :"let*") : nil
179 shape(bs, x)
180 scope = Env.new(e)
181 if a[0] == :"let*"
182 scope = e unless bs.empty?
183 bs.each { |name, expr|
184 v = evaluate(expr, scope)
185 scope = Env.new(scope)
186 scope.frame[name] = v
187 }
188 elsif a[0] == :letrec
189 bs.each { |name, _| scope.frame[name] = UNINIT }
190 bs.each { |name, expr| scope.frame[name] = evaluate(expr, scope) }
191 else
192 values = bs.map { |_, expr| evaluate(expr, e) }
193 if named
194 p = compound([bs.map(&:first), nil], a.drop(3), scope, a[1])
195 scope.frame[a[1]] = p
196 return p.call(values)
197 end
198
199 bs.zip(values).each { |b, v| scope.frame[b[0]] = v }
200 end
201
202 return evaluate_body(a.drop(at + 1), scope)
203 when :cond
204 cs = a.drop(1).map { |c| list_items(c) }
205 cs.each_with_index { |c, i| shape(c && !c.empty? && (c[0] != :else || (c.length >= 2 && i == cs.length - 1)), x) }
206 cs.each do |c|
207 return evaluate_body(c.drop(1), e) if c[0] == :else
208 v = evaluate(c[0], e)
209 return c.length == 1 ? v : evaluate_body(c.drop(1), e) if v != false
210 end
211
212 return VOID
213 when :and
214 v = true
215 a.drop(1).each { |z|
216 v = evaluate(z, e)
217 break if v == false
218 }
219 return v
220 when :or
221 a.drop(1).each { |z|
222 v = evaluate(z, e)
223 return v if v != false
224 }
225 return false
226 when :when, :unless
227 shape(n >= 3, x)
228 return (evaluate(a[1], e) != false) == (a[0] == :when) ? evaluate_body(a.drop(2), e) : VOID
229 end
230
231 f = evaluate(a[0], e)
232 args = a.drop(1).map { |z| evaluate(z, e) }
233 raise "not a procedure: #{represent(f)}" unless f.is_a?(Procedure)
234 f.call(args)
235end

249def expect(name, v, kind)
250 ok = case kind
251 when "integer"
252 v.is_a?(Integer)
253 when "string"
254 v.is_a?(String)
255 when "pair"
256 v.is_a?(Pair)
257 when "procedure"
258 v.is_a?(Procedure)
259 when "list"
260 !list_items(v).nil?
261 end
262
263 raise "#{name}: expected #{kind}, got #{represent(v)}" unless ok
264 v
265end

267def global_env
268 e = Env.new
269 add = -> (name, min, rest, &fn) { e.frame[name.to_sym] = Procedure.new(name.to_sym, min, rest, &fn) }
270 %w[+ * - quotient remainder modulo abs min max gcd = < > <= >=].each do |name|
271 fixed = %w[quotient remainder modulo].include?(name) ? 2 : name == "abs" ? 1 : nil
272 add.call(name, fixed || (%w[+ * gcd].include?(name) ? 0 : 1), fixed.nil?) do |a|
273 a.each { |v| expect(name, v, "integer") }
274 case name
275 when "+"
276 a.sum
277 when "*"
278 a.reduce(1, :*)
279 when "-"
280 a.length == 1 ? -a[0] : a.drop(1).reduce(a[0], :-)
281 when "abs"
282 a[0].abs
283 when "min"
284 a.min
285 when "max"
286 a.max
287 when "gcd"
288 a.reduce(0) { |g, v| g.gcd(v) }
289 when "quotient", "remainder", "modulo"
290 raise "#{name}: division by zero" if a[1] == 0
291 q = a[0].abs / a[1].abs
292 q = -q if (a[0] < 0) != (a[1] < 0)
293 name == "quotient" ? q : name == "modulo" ? a[0] % a[1] : a[0] - a[1] * q
294 else
295 a.each_cons(2).all? { |u, v| name == "=" ? u == v : u.public_send(name, v) }
296 end
297 end
298 end
299
300 add.call("not", 1, nil) { |a| a[0] == false }
301 add.call("eq?", 2, nil) { |a| same(a[0], a[1]) }
302 add.call("equal?", 2, nil) { |a| same(a[0], a[1], true) }
303 %w[null? pair? list? number? symbol? procedure?].each do |name|
304 add.call(name, 1, nil) do |a|
305 case name
306 when "null?"
307 a[0].equal?(NIL)
308 when "pair?"
309 a[0].is_a?(Pair)
310 when "list?"
311 !list_items(a[0]).nil?
312 when "number?"
313 a[0].is_a?(Integer)
314 when "symbol?"
315 a[0].is_a?(Symbol)
316 else
317 a[0].is_a?(Procedure)
318 end
319 end
320 end
321
322 add.call("cons", 2, nil) { |a| Pair.new(*a) }
323 add.call("list", 0, true) { |a| make_list(a) }
324 %w[car cdr cadr].each do |name|
325 add.call(name, 1, nil) { |a|
326 p = expect(name, a[0], "pair")
327 name == "car" ? p.car : name == "cdr" ? p.cdr : expect(name, p.cdr, "pair").car
328 }
329 end
330
331 %w[length reverse list-ref].each do |name|
332 add.call(name, name == "list-ref" ? 2 : 1, nil) do |a|
333 ls = list_items(expect(name, a[0], "list"))
334 if name == "length"
335 ls.length
336 elsif name == "reverse"
337 make_list(ls.reverse)
338 else
339 k = expect(name, a[1], "integer")
340 raise "list-ref: index out of range: #{k}" unless k >= 0 && k < ls.length
341 ls[k]
342 end
343 end
344 end
345
346 add.call("append", 0, true) { |a|
347 a.empty? ? NIL : make_list(a[0...-1].flat_map { |v| list_items(expect("append", v, "list")) }, a[-1])
348 }
349 %w[map filter fold-left apply].each do |name|
350 add.call(name, name == "fold-left" ? 3 : 2, name != "filter") do |a|
351 f = expect(name, a[0], "procedure")
352 if name == "apply"
353 f.call(a[1...-1] + list_items(expect(name, a[-1], "list")))
354 else
355 ls = a.drop(name == "fold-left" ? 2 : 1).map { |v| list_items(expect(name, v, "list")) }
356 if name == "filter"
357 make_list(ls[0].select { |v| f.call([v]) != false })
358 else
359 acc = a[1]
360 values = []
361 ls.map(&:length).min.times do |i|
362 args = ls.map { |l| l[i] }
363 if name == "map"
364 values << f.call(args)
365 else
366 acc = f.call([acc] + args)
367 end
368 end
369
370 name == "map" ? make_list(values) : acc
371 end
372 end
373 end
374 end
375
376 add.call("string-append", 0, true) { |a| a.map { |v| expect("string-append", v, "string") }.join }
377 add.call("number->string", 1, nil) { |a| expect("number->string", a[0], "integer").to_s }
378 add.call("display", 1, nil) { |a|
379 write_text(represent(a[0], false))
380 VOID
381 }
382 add.call("newline", 0, nil) { |_|
383 write_text("\n")
384 VOID
385 }
386 add.call("error", 1, true) { |a|
387 raise expect("error", a[0], "string") + a.drop(1).map { |v| " " + represent(v) }.join
388 }
389 e
390end

392class Reader
393 def initialize(lines)
394 @lines, @row, @column = lines, 0, 0
395 end
396
397 def skip_line
398 @row += 1
399 @column = 0
400 end
401
402 def at_end?
403 while @row < @lines.length
404 s = @lines[@row]
405 @column += 1 while s[@column] == " "
406 return false if @column < s.length && s[@column] != ";"
407 skip_line
408 end
409
410 true
411 end
412
413 def token
414 raise "unexpected end of input" if at_end?
415 s = @lines[@row]
416 c = s[@column]
417 @column += 1
418 return [c, nil] if "()'".include?(c)
419 if c == "\""
420 v = ""
421 while @column < s.length
422 c = s[@column]
423 @column += 1
424 return ["value", v] if c == "\""
425 if c == "\\"
426 break if @column == s.length
427 c = s[@column]
428 @column += 1
429 raise "bad escape" unless ["\"", "\\", "n"].include?(c)
430 v += c == "n" ? "\n" : c
431 else
432 v += c
433 end
434 end
435
436 raise "unterminated string"
437 end
438
439 start = @column - 1
440 @column += 1 while @column < s.length && !" ()'\";".include?(s[@column])
441 a = s[start...@column]
442 return [".", nil] if a == "."
443 return ["value", a.to_i] if /\A[+-]?[0-9]+\z/.match?(a)
444 return ["value", a == "#t"] if %w[#t #f].include?(a)
445 raise "bad token: #{a}" if a.start_with?("#")
446 ["value", a.to_sym]
447 end
448
449 def datum(t = token)
450 return t[1] if t[0] == "value"
451 return make_list([:quote, datum]) if t[0] == "'"
452 raise "unexpected #{t[0]}" unless t[0] == "("
453 a = []
454 loop do
455 t = token
456 return make_list(a) if t[0] == ")"
457 if t[0] == "." && !a.empty?
458 tail = datum
459 raise "expected )" unless token[0] == ")"
460 return make_list(a, tail)
461 end
462
463 a << datum(t)
464 end
465 end
466end

469reader, env = Reader.new(lines), global_env
470until reader.at_end?
471 begin
472 x = reader.datum
473 rescue => error
474 write_line("Error: read: " + error.message)
475 break if error.message == "unexpected end of input"
476 reader.skip_line
