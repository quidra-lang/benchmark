43function repr_value(v,quoted=true)
44 v===NIL && return "()"
45 v===VOID && return "#<void>"
46 v isa Bool && return v ? "#t" : "#f"
47 v isa Integer && return string(v)
48 v isa String && return quoted ? "\""*replace(v,"\\"=>"\\\\","\""=>"\\\"","\n"=>"\\n")*"\"" : v
49 v isa Symbol && return string(v)
50 v isa Procedure && return isnothing(v.name) ? "#<procedure>" : "#<procedure $(v.name)>"
51 a=String[]
52 while v isa Pair;
53 push!(a,repr_value(v.car,quoted));
54 v=v.cdr;
55 end
56 v!==NIL && append!(a,[".",repr_value(v,quoted)])
57 "("*join(a," ")*")"
58end

62 end
63 typeof(a)==typeof(b) && (a isa String ? a==b : a===b)
64end

90function bindings(v,unique_names)
91 a=items(v);
92 isnothing(a) && return nothing
93 bs=Tuple{Symbol,Any}[]
94 for x in a
95 b=items(x)
96 (isnothing(b)||length(b)!=2||!(b[1] isa Symbol)) && return nothing
97 push!(bs,(b[1],b[2]))
98 end
99 unique_names && length(unique(first.(bs)))!=length(bs) && return nothing
100 bs
101end

110function call_proc(f,a)
111 if length(a)<f.minimum || (f.rest===false && length(a)!=f.minimum)
112 fail("$(isnothing(f.name) ? :lambda : f.name): wrong number of arguments (expected $(f.rest===false ? "" : "at least ")$(f.minimum), got $(length(a)))")
113 end
114 !isnothing(f.builtin) && return f.builtin(a)
115 e=Env(f.env)
116 for (n,v) in zip(f.params,a);
117 e.frame[n]=v;
118 end
119 f.rest isa Symbol && (e.frame[f.rest]=make_list(a[f.minimum+1:end]))
120 eval_body(f.body,e)
121end
122function evaluate(x,e)
123 x isa Symbol && return lookup(e,x)
124 if !(x isa Pair);
125 shape(x!==NIL,x);
126 return x;
127 end
128 a=items(x);
129 shape(!isnothing(a),x);
130 n=length(a);
131 h=a[1] isa Symbol ? string(a[1]) : ""
132 if h=="quote";
133 shape(n==2,x);
134 return a[2]
135 elseif h=="if";
136 shape(n==3||n==4,x);
137 return truth(evaluate(a[2],e)) ? evaluate(a[3],e) : n==4 ? evaluate(a[4],e) : VOID
138 elseif h=="begin";
139 return eval_body(a[2:end],e)
140 elseif h=="lambda";
141 p=n>=3 ? parameters(a[2]) : nothing;
142 shape(!isnothing(p),x);
143 return compound(p,a[3:end],e)
144 elseif h=="define"
145 if n>1 && a[2] isa Symbol
146 shape(n==3,x);
147 v=evaluate(a[3],e)
148 v isa Procedure && isnothing(v.builtin) && isnothing(v.name) && (v.name=a[2])
149 e.frame[a[2]]=v
150 else
151 t=n>1 ? a[2] : NIL
152 p=n>=3 && t isa Pair && t.car isa Symbol ? parameters(t.cdr) : nothing
153 shape(!isnothing(p),x);
154 e.frame[t.car]=compound(p,a[3:end],e,t.car)
155 end
156 return VOID
157 elseif h=="set!";
158 shape(n==3&&a[2] isa Symbol,x);
159 v=evaluate(a[3],e);
160 findenv(e,a[2]).frame[a[2]]=v;
161 return VOID
162 elseif h in ("let","let*","letrec")
163 named=h=="let"&&n>1&&a[2] isa Symbol;
164 at=named ? 3 : 2
165 bs=n>=at+1 ? bindings(a[at],h!="let*") : nothing;
166 shape(!isnothing(bs),x);
167 s=Env(e)
168 if h=="let*"
169 !isempty(bs) && (s=e)
170 for (name,expr) in bs;
171 v=evaluate(expr,s);
172 s=Env(s);
173 s.frame[name]=v;
174 end
175 elseif h=="letrec"
176 for (name,_) in bs;
177 s.frame[name]=UNINIT;
178 end
179 for (name,expr) in bs;
180 s.frame[name]=evaluate(expr,s);
181 end
182 else
183 vs=Any[evaluate(b[2],e) for b in bs]
184 if named
185 f=compound((Symbol[first(b) for b in bs],false),a[4:end],s,a[2]);
186 s.frame[a[2]]=f;
187 return call_proc(f,vs)
188 end
189 for (b,v) in zip(bs,vs);
190 s.frame[b[1]]=v;
191 end
192 end
193 return eval_body(a[at+1:end],s)
194 elseif h=="cond"
195 cs=items.(a[2:end])
196 for (i,c) in enumerate(cs)
197 shape(!isnothing(c)&&!isempty(c),x)
198 c[1]===:else && shape(length(c)>=2&&i==length(cs),x)
199 end
200 for c in cs
201 c[1]===:else && return eval_body(c[2:end],e)
202 v=evaluate(c[1],e);
203 truth(v) && return length(c)==1 ? v : eval_body(c[2:end],e)
204 end
205 return VOID
206 elseif h=="and"
207 v=true;
208 for z in a[2:end];
209 v=evaluate(z,e);
210 !truth(v)&&break;
211 end;
212 return v
213 elseif h=="or"
214 for z in a[2:end];
215 v=evaluate(z,e);
216 truth(v)&&return v;
217 end;
218 return false
219 elseif h=="when"||h=="unless"
220 shape(n>=3,x);
221 return truth(evaluate(a[2],e))==(h=="when") ? eval_body(a[3:end],e) : VOID
222 end
223 f=evaluate(a[1],e);
224 args=Any[evaluate(z,e) for z in a[2:end]]
225 f isa Procedure || fail("not a procedure: "*repr_value(f));
226 call_proc(f,args)
227end

237function expect(n,v,k)
238 ok=k=="integer" ? v isa Integer && !(v isa Bool) : k=="string" ? v isa String : k=="pair" ? v isa Pair : k=="procedure" ? v isa Procedure : !isnothing(items(v))
239 ok||fail("$n: expected $k, got "*repr_value(v));
240 v
241end
242function global_env()
243 e=Env()
244 add(n,m,r,fn)=(e.frame[Symbol(n)]=Procedure(Symbol(n),m,r,Symbol[],Any[],nothing,fn))
245 for name in ("+","*","-","quotient","remainder","modulo","abs","min","max","gcd","=","<",">","<=",">=")
246 fixed=name in ("quotient","remainder","modulo") ? 2 : name=="abs" ? 1 : -1
247 add(name,fixed>=0 ? fixed : name in ("+","*","gcd") ? 0 : 1,fixed<0,a->begin
248 z=Int64[expect(name,v,"integer") for v in a]
249 name=="+"&&return sum(z);
250 name=="*"&&return prod(z);
251 name=="-"&&return length(z)==1 ? -z[1] : z[1]-sum(z[2:end])
252 name=="abs"&&return abs(z[1]);
253 name=="min"&&return minimum(z);
254 name=="max"&&return maximum(z)
255 if name=="gcd";
256 g=0;
257 for v in z;
258 g=gcd(g,v);
259 end;
260 return g;
261 end
262 if name in ("quotient","remainder","modulo")
263 z[2]==0&&fail(name*": division by zero")
264 return name=="quotient" ? div(z[1],z[2]) : name=="remainder" ? rem(z[1],z[2]) : mod(z[1],z[2])
265 end
266 all(i->name=="=" ? z[i-1]==z[i] : name=="<" ? z[i-1]<z[i] : name==">" ? z[i-1]>z[i] : name=="<=" ? z[i-1]<=z[i] : z[i-1]>=z[i],2:length(z))
267 end)
268 end
269 add("not",1,false,a->!truth(a[1]));
270 add("eq?",2,false,a->same(a[1],a[2]));
271 add("equal?",2,false,a->same(a[1],a[2],true))
272 for name in ("null?","pair?","list?","number?","symbol?","procedure?")
273 add(name,1,false,a->name=="null?" ? a[1]===NIL : name=="pair?" ? a[1] isa Pair : name=="list?" ? !isnothing(items(a[1])) : name=="number?" ? a[1] isa Integer&&!(a[1] isa Bool) : name=="symbol?" ? a[1] isa Symbol : a[1] isa Procedure)
274 end
275 add("cons",2,false,a->Pair(a[1],a[2]));
276 add("list",0,true,make_list)
277 for name in ("car","cdr","cadr")
278 add(name,1,false,a->begin
279 p=expect(name,a[1],"pair");
280 name=="car" ? p.car : name=="cdr" ? p.cdr : expect(name,p.cdr,"pair").car
281 end)
282 end
283 for name in ("length","reverse","list-ref")
284 add(name,name=="list-ref" ? 2 : 1,false,a->begin
285 z=items(expect(name,a[1],"list"));
286 name=="length"&&return length(z);
287 name=="reverse"&&return make_list(reverse(z))
288 k=expect(name,a[2],"integer");
289 0<=k<length(z)||fail("list-ref: index out of range: $k");
290 z[k+1]
291 end)
292 end
293 add("append",0,true,a->begin
294 isempty(a)&&return NIL;
295 z=Any[];
296 for v in a[1:end-1];
297 append!(z,items(expect("append",v,"list")));
298 end;
299 make_list(z,a[end])
300 end)
301 for name in ("map","filter","fold-left","apply")
302 add(name,name=="fold-left" ? 3 : 2,name!="filter",a->begin
303 f=expect(name,a[1],"procedure")
304 name=="apply"&&return call_proc(f,vcat(a[2:end-1],items(expect(name,a[end],"list"))))
305 ls=[items(expect(name,v,"list")) for v in a[(name=="fold-left" ? 3 : 2):end]];
306 out=Any[];
307 acc=a[2]
308 for i in 1:minimum(length.(ls))
309 if name=="filter";
310 truth(call_proc(f,Any[ls[1][i]]))&&push!(out,ls[1][i])
311 else
312 args=Any[l[i] for l in ls];
313 name=="fold-left"&&pushfirst!(args,acc);
314 v=call_proc(f,args)
315 if name=="map";
316 push!(out,v);
317 else;
318 acc=v;
319 end
320 end
321 end
322 name=="fold-left" ? acc : make_list(out)
323 end)
324 end
325 add("string-append",0,true,a->join([expect("string-append",v,"string") for v in a]))
326 add("number->string",1,false,a->string(expect("number->string",a[1],"integer")))
327 add("display",1,false,a->begin
328 write_text(repr_value(a[1],false));
329 VOID
330 end);
331 add("newline",0,false,a->begin
332 write_text("\n");
333 VOID
334 end)
335 add("error",1,true,a->fail(expect("error",a[1],"string")*join([" "*repr_value(v) for v in a[2:end]])))
336 e
337end
338mutable struct Reader
339 lines::Vector{String}
340 row::Int
341 col::Int
342end
343skip_line!(r)=(r.row+=1;r.col=1)
344function at_end(r)
345 while r.row<=length(r.lines)
346 s=r.lines[r.row];
347 while r.col<=length(s)&&s[r.col]==' ';
348 r.col+=1;
349 end
350 r.col<=length(s)&&s[r.col]!=';'&&return false
351 skip_line!(r)
352 end
353 true
354end
355function token(r)
356 at_end(r)&&fail("unexpected end of input");
357 s=r.lines[r.row];
358 c=s[r.col];
359 r.col+=1
360 c in ('(',')','\'')&&return (c,nothing)
361 if c=='"'
362 v=""
363 while r.col<=length(s)
364 c=s[r.col];
365 r.col+=1;
366 c=='"'&&return ('v',v)
367 if c=='\\'
368 r.col>length(s)&&break;
369 c=s[r.col];
370 r.col+=1
371 c in ('"','\\','n')||fail("bad escape");
372 c=='n'&&(c='\n')
373 end
374 v*=c
375 end
376 fail("unterminated string")
377 end
378 first=r.col-1;
379 while r.col<=length(s)&&!(s[r.col] in " ()'\";");
380 r.col+=1;
381 end
382 a=s[first:r.col-1];
383 a=="."&&return ('.',nothing)
384 occursin(r"^[+-]?[0-9]+$",a)&&return ('v',parse(Int64,a))
385 a in ("#t","#f")&&return ('v',a=="#t")
386 startswith(a,"#")&&fail("bad token: "*a);
387 ('v',Symbol(a))
388end
389function datum(r,t=token(r))
390 t[1]=='v'&&return t[2];
391 t[1]=='\''&&return make_list(Any[:quote,datum(r)])
392 t[1]=='('||fail("unexpected $(t[1])");
393 a=Any[]
394 while true
395 t=token(r);
396 t[1]==')'&&return make_list(a)
397 if t[1]=='.'&&!isempty(a);
398 tail=datum(r);
399 token(r)[1]==')'||fail("expected )");
400 return make_list(a,tail);
401 end
402 push!(a,datum(r,t))
403 end
404end
405function main_program()
406 lines=String[];
407 for s in eachline(stdin);
408 s=="(exit)"&&break;
409 push!(lines,s);
410 end
411 r=Reader(lines,1,1);
412 e=global_env()
413 while !at_end(r)
414 x=try
415 datum(r)
416 catch err
417 write_line("Error: read: "*err.msg);
418 err.msg=="unexpected end of input"&&break;
419 skip_line!(r);
420 continue
421 end
422 try
423 v=evaluate(x,e);
424 v!==VOID&&write_line(repr_value(v))
425 catch err
426 write_line("Error: "*err.msg)
427 end
428 end
429 mid[]&&write_text("\n")
430end
