        public Environment(Environment? parent = null)=>Parent = parent;
        public Environment Find(string name){
            for (Environment? e = this; e != null; e = e.Parent)if (e.Frame.ContainsKey(name))return e;
        public object Get(string name){
            object v = Find(name).Frame[name];
            if (ReferenceEquals(v, Uninitialized))throw new Exception("uninitialized variable: "+name);
            return v;
