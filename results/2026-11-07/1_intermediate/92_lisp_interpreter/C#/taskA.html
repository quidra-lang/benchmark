7 static readonly object Nil = new(), Void = new(), Uninitialized = new();
8 sealed record Symbol(string Name);
9 sealed record Pair(object Car, object Cdr);
10 sealed class Environment
11 {
12 public readonly Dictionary<string, object> Frame = new();
13 public readonly Environment? Parent;
14 public Environment(Environment? parent = null)=>Parent = parent;
15 public Environment Find(string name){
16 for (Environment? e = this; e != null; e = e.Parent)if (e.Frame.ContainsKey(name))return e;
17 throw new Exception("unbound variable: "+name);
18 }
19 public object Get(string name){
20 object v = Find(name).Frame[name];
21 if (ReferenceEquals(v, Uninitialized))throw new Exception("uninitialized variable: "+name);
22 return v;
23 }
24 }
25 sealed class Procedure
26 {
27 public string? Name, Rest;
28 public int Minimum;
29 public bool Variadic;
30 public List<string> Parameters = new();
31 public List<object> Body = new();
32 public Environment? Scope;
33 public Func<List<object>, object>? Builtin;
34 public object Call(List<object> args)
35 {
36 if (args.Count<Minimum || (!Variadic && args.Count != Minimum))throw new Exception($"{Name??"lambda"}: wrong number of arguments (expected {(Variadic?"at least ":"")}{Minimum}, got {args.Count})");
37 if (Builtin != null)return Builtin(args);
38 Environment e = new(Scope);
39 for (int i = 0; i<Parameters.Count; i++)e.Frame[Parameters[i]] = args[i];
40 if (Rest != null)e.Frame[Rest] = List(args.Skip(Minimum));
41 return EvalBody(Body, e);
42 }
43 }
44 static object List(IEnumerable<object> values, object? tail = null){
45 object v = tail??Nil;
46 foreach(object x in values.Reverse())v = new Pair(x, v);
47 return v;
48 }
49 static List<object>? Items(object v){
50 List<object>a = new();
51 while (v is Pair p){
52 a.Add(p.Car);
53 v = p.Cdr;
54 }
55 return ReferenceEquals(v, Nil)?a: null;
56 }
57 static bool Truth(object v)=>v is not false;
58 static string Repr(object v, bool quoted = true)
59 {
60 if (ReferenceEquals(v, Nil))return "()";
61 if (ReferenceEquals(v, Void))return "#<void>";
62 if (v is bool b)return b?"#t": "#f";
63 if (v is long n)return n.ToString();
64 if (v is string s)return quoted?'"'+s.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\n", "\\n")+'"': s;
65 if (v is Symbol symbol)return symbol.Name;
66 if (v is Procedure f)return f.Name == null?"#<procedure>": $"#<procedure {f.Name}>";
67 List<string>a = new();
68 while (v is Pair p){
69 a.Add(Repr(p.Car, quoted));
70 v = p.Cdr;
71 }
72 if (!ReferenceEquals(v, Nil)){
73 a.Add(".");
74 a.Add(Repr(v, quoted));
75 }
76 return "("+string.Join(" ", a)+")";
77 }
78 static bool Same(object a, object b, bool deep = false){
79 if (a is Pair p && b is Pair q)return deep?Same(p.Car, q.Car, true) && Same(p.Cdr, q.Cdr, true): ReferenceEquals(a, b);
80 if (a is Procedure || b is Procedure)return ReferenceEquals(a, b);
81 return Equals(a, b);
82 }
83 sealed record Formals(List<string> Names, string? Rest);
84 static Formals? Parameters(object v){
85 List<string>names = new();
86 while (v is Pair p){
87 if (p.Car is not Symbol s)return null;
88 names.Add(s.Name);
89 v = p.Cdr;
90 }
91 string? r = (v as Symbol)?.Name;
92 if (!ReferenceEquals(v, Nil) && r == null)return null;
93 List<string>all = new(names);
94 if (r != null)all.Add(r);
95 return all.Distinct().Count() == all.Count?new(names, r): null;
96 }
97 static Procedure Closure(Formals p, List<object>b, Environment e, string? name = null)=>new(){
98 Name = name, Minimum = p.Names.Count, Variadic = p.Rest != null, Rest = p.Rest, Parameters = p.Names, Body = b, Scope = e
99 };
100 sealed record Binding(string Name, object Expression);
101 static List<Binding>? Bindings(object v, bool unique){
102 List<object>? a = Items(v);
103 if (a == null)return null;
104 List<Binding>bs = new();
105 HashSet<string>seen = new();
106 foreach(object x in a){
107 List<object>? b = Items(x);
108 if (b == null || b.Count != 2 || b[0] is not Symbol s)return null;
109 if (unique && !seen.Add(s.Name))return null;
110 bs.Add(new(s.Name, b[1]));
111 }
112 return bs;
113 }
114 static void Shape(bool valid, object x){
115 if (!valid)throw new Exception("bad syntax: "+Repr(x));
116 }
117 static object EvalBody(IEnumerable<object>a, Environment e){
118 object v = Void;
119 foreach(object x in a)v = Evaluate(x, e);
120 return v;
121 }
122 static object Evaluate(object x, Environment e)
123 {
124 if (x is Symbol symbol)return e.Get(symbol.Name);
125 if (x is not Pair){
126 Shape(!ReferenceEquals(x, Nil), x);
127 return x;
128 }
129 List<object>? parsed = Items(x);
130 Shape(parsed != null, x);
131 List<object>a = parsed!;
132 int n = a.Count;
133 string h = (a[0] as Symbol)?.Name??"";
134 switch (h)
135 {
136 case "quote":
137 Shape(n == 2, x);
138 return a[1];
139 case "if":
140 Shape(n == 3 || n == 4, x);
141 return Truth(Evaluate(a[1], e))?Evaluate(a[2], e): n == 4?Evaluate(a[3], e): Void;
142 case "begin":
143 return EvalBody(a.Skip(1), e);
144 case "lambda":
145 {
146 Formals? p = n >= 3?Parameters(a[1]): null;
147 Shape(p != null, x);
148 return Closure(p!, a.Skip(2).ToList(), e);
149 }
150 case "define":
151 if (n>1 && a[1] is Symbol s){
152 Shape(n == 3, x);
153 object v = Evaluate(a[2], e);
154 if (v is Procedure f && f.Builtin == null && f.Name == null)f.Name = s.Name;
155 e.Frame[s.Name] = v;
156 }
157 else{
158 object target = n>1?a[1]: Nil;
159 Formals? p = n >= 3 && target is Pair q && q.Car is Symbol?Parameters(q.Cdr): null;
160 Shape(p != null, x);
161 string name = ((Symbol)((Pair)target).Car).Name;
162 e.Frame[name] = Closure(p!, a.Skip(2).ToList(), e, name);
163 }
164 return Void;
165 case "set!":
166 Shape(n == 3 && a[1] is Symbol, x);
167 {
168 string name = ((Symbol)a[1]).Name;
169 object v = Evaluate(a[2], e);
170 e.Find(name).Frame[name] = v;
171 }
172 return Void;
173 case "let":
174 case "let*":
175 case "letrec":
176 {
177 bool named = h == "let" && n>1 && a[1] is Symbol;
178 int at = named?2: 1;
179 List<Binding>? bs = n >= at+2?Bindings(a[at], h != "let*"): null;
180 Shape(bs != null, x);
181 Environment scope = new(e);
182 if (h == "let*"){
183 if (bs!.Count>0)scope = e;
184 foreach(Binding b in bs){
185 object v = Evaluate(b.Expression, scope);
186 scope = new(scope);
187 scope.Frame[b.Name] = v;
188 }
189 }
190 else if (h == "letrec"){
191 foreach(Binding b in bs!)scope.Frame[b.Name] = Uninitialized;
192 foreach(Binding b in bs)scope.Frame[b.Name] = Evaluate(b.Expression, scope);
193 }
194 else{
195 List<object>values = bs!.Select(b=>Evaluate(b.Expression, e)).ToList();
196 if (named){
197 string name = ((Symbol)a[1]).Name;
198 Procedure f = Closure(new(bs.Select(b=>b.Name).ToList(), null), a.Skip(3).ToList(), scope, name);
199 scope.Frame[name] = f;
200 return f.Call(values);
201 }
202 for (int i = 0; i<bs.Count; i++)scope.Frame[bs[i].Name] = values[i];
203 }
204 return EvalBody(a.Skip(at+1), scope);
205 }
206 case "cond":
207 {
208 List<List<object>?>cs = a.Skip(1).Select(Items).ToList();
209 for (int i = 0; i<cs.Count; i++){
210 List<object>? c = cs[i];
211 Shape(c != null && c.Count>0, x);
212 if (c![0] is Symbol s && s.Name == "else")Shape(c.Count >= 2 && i == cs.Count-1, x);
213 }
214 foreach(List<object>? c in cs){
215 if (c![0] is Symbol s && s.Name == "else")return EvalBody(c.Skip(1), e);
216 object v = Evaluate(c[0], e);
217 if (Truth(v))return c.Count == 1?v: EvalBody(c.Skip(1), e);
218 }
219 return Void;
220 }
221 case "and":
222 {
223 object v = true;
224 foreach(object z in a.Skip(1)){
225 v = Evaluate(z, e);
226 if (!Truth(v))break;
227 }
228 return v;
229 }
230 case "or":
231 foreach(object z in a.Skip(1)){
232 object v = Evaluate(z, e);
233 if (Truth(v))return v;
234 }
235 return false;
236 case "when":
237 case "unless":
238 Shape(n >= 3, x);
239 return Truth(Evaluate(a[1], e)) == (h == "when")?EvalBody(a.Skip(2), e): Void;
240 }
241 object fn = Evaluate(a[0], e);
242 List<object>args = a.Skip(1).Select(z=>Evaluate(z, e)).ToList();
243 if (fn is not Procedure proc)throw new Exception("not a procedure: "+Repr(fn));
244 return proc.Call(args);
245 }

248 Console.Write(s);
249 if (s.Length>0)mid = !s.EndsWith('\n');

252 if (mid)Write("\n");
253 Write(s+"\n");

255 static object Expect(string name, object v, string kind){
256 bool ok = kind switch{
257 "integer"=>v is long, "string"=>v is string, "pair"=>v is Pair, "procedure"=>v is Procedure, _=>Items(v) != null
258 };
259 if (!ok)throw new Exception($"{name}: expected {kind}, got {Repr(v)}");
260 return v;
261 }
262 static List<object> AsList(string n, object v){
263 Expect(n, v, "list");
264 return Items(v)!;
265 }
266 static void Add(Environment e, string n, int minimum, bool rest, Func<List<object>, object>fn)=>e.Frame[n] = new Procedure{
267 Name = n, Minimum = minimum, Variadic = rest, Builtin = fn
268 };
269 static Environment Global()
270 {
271 Environment e = new();
272 foreach(string name in new[]{
273 "+", "*", "-", "quotient", "remainder", "modulo", "abs", "min", "max", "gcd", "=", "<", ">", "<=", ">="
274 })
275 {
276 int fixedCount = new[]{
277 "quotient", "remainder", "modulo"
278 }.Contains(name)?2: name == "abs"?1: -1;
279 Add(e, name, fixedCount >= 0?fixedCount: new[]{
280 "+", "*", "gcd"
281 }.Contains(name)?0: 1, fixedCount<0, a=>{
282 foreach(object x in a)Expect(name, x, "integer");
283 long v = a.Count>0?(long)a[0]: 0;
284 switch (name){
285 case "+":
286 case "*":
287 v = name == "+"?0: 1;
288 foreach(object x in a)v = name == "+"?v+(long)x: v*(long)x;
289 return v;
290 case "-":
291 if (a.Count == 1)return -v;
292 foreach(object x in a.Skip(1))v -= (long)x;
293 return v;
294 case "abs":
295 return Math.Abs(v);
296 case "min":
297 case "max":
298 foreach(object x in a)v = name == "min"?Math.Min(v, (long)x): Math.Max(v, (long)x);
299 return v;
300 case "gcd":
301 v = 0;
302 foreach(object x in a){
303 long b = Math.Abs((long)x);
304 while (b != 0){
305 long r = v%b;
306 v = b;
307 b = r;
308 }
309 }
310 return v;
311 case "quotient":
312 case "remainder":
313 case "modulo":
314 long divisor = (long)a[1];
315 if (divisor == 0)throw new Exception(name+": division by zero");
316 long rem = v%divisor;
317 return name == "quotient"?v/divisor: name == "modulo" && rem != 0 && ((rem<0) != (divisor<0))?rem+divisor: rem;
318 }
319 for (int i = 1; i<a.Count; i++){
320 long l = (long)a[i-1], r = (long)a[i];
321 bool ok = name switch{
322 "="=>l == r, "<"=>l<r, ">"=>l>r, "<="=>l <= r, _=>l >= r
323 };
324 if (!ok)return false;
325 }
326 return true;
327 });
328 }
329 Add(e, "not", 1, false, a=>!Truth(a[0]));
330 foreach(string n in new[]{
331 "eq?", "equal?"
332 })Add(e, n, 2, false, a=>Same(a[0], a[1], n == "equal?"));
333 foreach(string n in new[]{
334 "null?", "pair?", "list?", "number?", "symbol?", "procedure?"
335 })Add(e, n, 1, false, a=>n switch{
336 "null?"=>ReferenceEquals(a[0], Nil), "pair?"=>a[0] is Pair, "list?"=>Items(a[0]) != null, "number?"=>a[0] is long, "symbol?"=>a[0] is Symbol, _=>a[0] is Procedure
337 });
338 Add(e, "cons", 2, false, a=>new Pair(a[0], a[1]));
339 Add(e, "list", 0, true, a=>List(a));
340 foreach(string n in new[]{
341 "car", "cdr", "cadr"
342 })Add(e, n, 1, false, a=>{
343 Pair p = (Pair)Expect(n, a[0], "pair");
344 return n == "car"?p.Car: n == "cdr"?p.Cdr: ((Pair)Expect(n, p.Cdr, "pair")).Car;
345 });
346 foreach(string n in new[]{
347 "length", "reverse", "list-ref"
348 })Add(e, n, n == "list-ref"?2: 1, false, a=>{
349 List<object>z = AsList(n, a[0]);
350 if (n == "length")return (long)z.Count;
351 if (n == "reverse")return List(z.AsEnumerable().Reverse());
352 long k = (long)Expect(n, a[1], "integer");
353 if (k<0 || k >= z.Count)throw new Exception("list-ref: index out of range: "+k);
354 return z[(int)k];
355 });
356 Add(e, "append", 0, true, a=>{
357 if (a.Count == 0)return Nil;
358 List<object>z = new();
359 foreach(object v in a.Take(a.Count-1))z.AddRange(AsList("append", v));
360 return List(z, a[^1]);
361 });
362 foreach(string n in new[]{
363 "map", "filter", "fold-left", "apply"
364 })Add(e, n, n == "fold-left"?3: 2, n != "filter", a=>{
365 Procedure f = (Procedure)Expect(n, a[0], "procedure");
366 if (n == "apply")return f.Call(a.Skip(1).Take(a.Count-2).Concat(AsList(n, a[^1])).ToList());
367 List<List<object>>ls = a.Skip(n == "fold-left"?2: 1).Select(v=>AsList(n, v)).ToList();
368 int count = ls.Min(z=>z.Count);
369 List<object>result = new();
370 object acc = a[1];
371 for (int i = 0; i<count; i++){
372 if (n == "filter"){
373 if (Truth(f.Call(new(){
374 ls[0][i]
375 })))result.Add(ls[0][i]);
376 }else{
377 List<object>args = new();
378 if (n == "fold-left")args.Add(acc);
379 args.AddRange(ls.Select(z=>z[i]));
380 object v = f.Call(args);
381 if (n == "map")result.Add(v);
382 else acc = v;
383 }
384 }
385 return n == "fold-left"?acc: List(result);
386 });
387 Add(e, "string-append", 0, true, a=>string.Concat(a.Select(v=>(string)Expect("string-append", v, "string"))));
388 Add(e, "number->string", 1, false, a=>Expect("number->string", a[0], "integer").ToString()!);
389 Add(e, "display", 1, false, a=>{
390 Write(Repr(a[0], false));
391 return Void;
392 });
393 Add(e, "newline", 0, false, a=>{
394 Write("\n");
395 return Void;
396 });
397 Add(e, "error", 1, true, a=>throw new Exception((string)Expect("error", a[0], "string")+string.Concat(a.Skip(1).Select(v=>" "+Repr(v)))));
398 return e;
399 }
400 readonly record struct Token(char Kind, object? Value);
401 sealed class Reader
402 {
403 readonly List<string> lines;
404 int row, col;
405 public Reader(List<string> lines)=>this.lines = lines;
406 public void Skip(){
407 row++;
408 col = 0;
409 }
410 public bool End(){
411 while (row<lines.Count){
412 string s = lines[row];
413 while (col<s.Length && s[col] == ' ')col++;
414 if (col<s.Length && s[col] != ';')return false;
415 Skip();
416 }
417 return true;
418 }
419 Token Next(){
420 if (End())throw new Exception("unexpected end of input");
421 string s = lines[row];
422 char c = s[col++];
423 if ("()'".Contains(c))return new(c, null);
424 if (c == '"'){
425 StringBuilder v = new();
426 while (col<s.Length){
427 c = s[col++];
428 if (c == '"')return new('v', v.ToString());
429 if (c == '\\'){
430 if (col == s.Length)break;
431 c = s[col++];
432 if (c != '"' && c != '\\' && c != 'n')throw new Exception("bad escape");
433 if (c == 'n')c = '\n';
434 }
435 v.Append(c);
436 }
437 throw new Exception("unterminated string");
438 }
439 int start = col-1;
440 while (col<s.Length && !" ()'\";".Contains(s[col]))col++;
441 string atom = s[start..col];
442 if (atom == ".")return new('.', null);
443 if (System.Text.RegularExpressions.Regex.IsMatch(atom, "^[+-]?[0-9]+$"))return new('v', long.Parse(atom));
444 if (atom == "#t" || atom == "#f")return new('v', atom == "#t");
445 if (atom[0] == '#')throw new Exception("bad token: "+atom);
446 return new('v', new Symbol(atom));
447 }
448 public object Datum()=>From(Next());
449 object From(Token t){
450 if (t.Kind == 'v')return t.Value!;
451 if (t.Kind == '\'')return List(new object[]{
452 new Symbol("quote"), Datum()
453 });
454 if (t.Kind != '(')throw new Exception("unexpected "+t.Kind);
455 List<object>a = new();
456 while (true){
457 t = Next();
458 if (t.Kind == ')')return List(a);
459 if (t.Kind == '.' && a.Count>0){
460 object tail = Datum();
461 if (Next().Kind != ')')throw new Exception("expected )");
462 return List(a, tail);
463 }
464 a.Add(From(t));
465 }
466 }
467 }
468 static void Main(){
469 List<string>lines = new();
470 string? s;
471 while ((s = Console.ReadLine()) != null && s != "(exit)")lines.Add(s);
472 Reader r = new(lines);
473 Environment e = Global();
474 while (!r.End()){
475 object x;
476 try{
477 x = r.Datum();
478 }catch (Exception err){
479 Line("Error: read: "+err.Message);
480 if (err.Message == "unexpected end of input")break;
481 r.Skip();
482 continue;
483 }
484 try{
485 object v = Evaluate(x, e);
486 if (!ReferenceEquals(v, Void))Line(Repr(v));
487 }catch (Exception err){
488 Line("Error: "+err.Message);
489 }
490 }
491 if (mid)Write("\n");
492 }
