44final class Procedure {
45 var name: String?
46 let minimum: Int, variadic: Bool
47 let rest: String?, params: [String], expressions: [Value], env: Env?
48 let builtin: (([Value]) throws -> Value)?
49 init(
50 _ name: String?,
51 _ minimum: Int,
52 _ variadic: Bool,
53 rest: String? = nil,
54 params: [String] = [],
55 body: [Value] = [],
56 env: Env? = nil,
57 builtin: (([Value]) throws -> Value)? = nil
58 ) {
59 self.name = name
60 self.minimum = minimum
61 self.variadic = variadic
62 self.rest = rest
63 self.params = params
64 self.expressions = body

68 func call(_ a: [Value]) throws -> Value {
69 if a.count < minimum || (!variadic && a.count != minimum) {
70 try fail(
71 "\(name ?? "lambda"): wrong number of arguments (expected \(variadic ? "at least " : "")\(minimum), got \(a.count))"
72 )
73 }
74 if let fn = builtin {
75 return try fn(a)
76 }
77 let e = Env(env)
78 for (i, n) in params.enumerated() {
79 e.frame[n] = a[i]
80 }
81 if let r = rest {
82 e.frame[r] = makeList(Array(a.dropFirst(minimum)))
83 }
84 return try body(expressions, e)
85 }

110 case .integer(let n):

112 case .boolean(let b):
113 return b ? "#t" : "#f"
114 case .string(let s):

116 of: "\"",
117 with: "\\\""
118 ).replacingOccurrences(of: "\n", with: "\\n") + "\"" : s
119 case .symbol(let s):
120 return s
121 case .nilValue:
122 return "()"
123 case .procedure(let f):

126 } ?? "#<procedure>"
127 case .pair:

143func same(_ a: Value, _ b: Value, _ deep: Bool = false) -> Bool {
144 switch (a, b) {
145 case(.integer(let a), .integer(let b)):
146 return a == b
147 case(.boolean(let a), .boolean(let b)):
148 return a == b
149 case(.string(let a), .string(let b)), (.symbol(let a), .symbol(let b)):
150 return a == b
151 case(.nilValue, .nilValue), (.void, .void):
152 return true
153 case(.pair(let p), .pair(let q)):
154 return deep ? same(p.car, q.car, true) && same(p.cdr, q.cdr, true) : p === q
155 case(.procedure(let p), .procedure(let q)):
156 return p === q
157 default:
158 return false
159 }

166func parameters(_ value: Value) -> ([String], String?)? {
167 var v = value, names = [String]()
168 while case .pair(let p) = v {
169 guard case .symbol(let s) = p.car else {
170 return nil
171 }
172 names.append(s)
173 v = p.cdr
174 }
175 let rest: String?
176 switch v {
177 case .nilValue:
178 rest = nil
179 case .symbol(let n):
180 rest = n
181 default:
182 return nil
183 }
184 let all = names + (rest.map {
185 [$0]
186 } ?? [])
187 return Set(all).count == all.count ? (names, rest) : nil

200func bindings(_ value: Value, _ unique: Bool) -> [(String, Value)]? {
201 guard let a = items(value) else {

205 for x in a {
206 guard let b = items(x), b.count == 2, case .symbol(let name) = b[0] else {

223func evaluate(_ x: Value, _ e: Env) throws -> Value {
224 if case .symbol(let n) = x {
225 return try e.get(n)
226 }
227 guard case .pair = x else {
228 if case .nilValue = x {
229 try shape(false, x)
230 }
231 return x
232 }
233 guard let a = items(x) else {
234 try fail("bad syntax: " + repr(x))
235 }
236 let n = a.count
237 let h: String
238 if case .symbol(let name) = a[0] {
239 h = name
240 } else {
241 h = ""
242 }
243 switch h {
244 case "quote":
245 try shape(n == 2, x)
246 return a[1]
247 case "if":
248 try shape(n == 3 || n == 4, x)
249 if truth(try evaluate(a[1], e)) {
250 return try evaluate(a[2], e)
251 }
252 return n == 4 ? try evaluate(a[3], e) : .void
253 case "begin":
254 return try body(Array(a.dropFirst()), e)
255 case "lambda":
256 let p = n >= 3 ? parameters(a[1]) : nil
257 try shape(p != nil, x)
258 return closure(p!, Array(a.dropFirst(2)), e)
259 case "define":
260 if n > 1, case .symbol(let name) = a[1] {
261 try shape(n == 3, x)
262 let v = try evaluate(a[2], e)
263 if case .procedure(let f) = v, f.builtin == nil, f.name == nil {
264 f.name = name
265 }
266 e.frame[name] = v
267 } else {
268 guard n >= 3, case .pair(let t) = a[1], case .symbol(let name) = t.car, let p = parameters(t.cdr) else {
269 try fail("bad syntax: " + repr(x))
270 }
271 e.frame[name] = closure(p, Array(a.dropFirst(2)), e, name)
272 }
273 return .void
274 case "set!":
275 guard n == 3, case .symbol(let name) = a[1] else {
276 try fail("bad syntax: " + repr(x))
277 }
278 let v = try evaluate(a[2], e)
279 let target = try e.find(name)
280 target.frame[name] = v
281 return .void
282 case "let", "let*", "letrec":
283 var loop: String?
284 if h == "let", n > 1, case .symbol(let name) = a[1] {
285 loop = name
286 }
287 let at = loop == nil ? 1 : 2
288 let bs = n >= at + 2 ? bindings(a[at], h != "let*") : nil
289 try shape(bs != nil, x)
290 var scope = Env(e)
291 if h == "let*" {
292 if !bs!.isEmpty {
293 scope = e
294 }
295 for (name, expr) in bs! {
296 let v = try evaluate(expr, scope)
297 scope = Env(scope)
298 scope.frame[name] = v
299 }
300 } else if h == "letrec" {
301 for (name, _) in bs! {
302 scope.frame[name] = .uninitialized
303 }
304 for (name, expr) in bs! {
305 scope.frame[name] = try evaluate(expr, scope)
306 }
307 } else {
308 var vs = [Value]()
309 for (_, expr) in bs! {
310 vs.append(try evaluate(expr, e))
311 }
312 if let name = loop {
313 let f = closure((bs!.map {
314 $0.0
315 }, nil), Array(a.dropFirst(3)), scope, name)
316 scope.frame[name] = f
317 if case .procedure(let p) = f {
318 return try p.call(vs)
319 }
320 }
321 for (i, b) in bs!.enumerated() {
322 scope.frame[b.0] = vs[i]
323 }
324 }
325 return try body(Array(a.dropFirst(at + 1)), scope)
326 case "cond":
327 var cs = [[Value]]()
328 for (i, c0) in a.dropFirst().enumerated() {
329 guard let c = items(c0),!c.isEmpty else {
330 try fail("bad syntax: " + repr(x))
331 }
332 if case .symbol("else") = c[0] {
333 try shape(c.count >= 2 && i == n - 2, x)
334 }
335 cs.append(c)
336 }
337 for c in cs {
338 if case .symbol("else") = c[0] {
339 return try body(Array(c.dropFirst()), e)
340 }
341 let v = try evaluate(c[0], e)
342 if truth(v) {
343 return c.count == 1 ? v : try body(Array(c.dropFirst()), e)
344 }
345 }
346 return .void
347 case "and":
348 var v = Value.boolean(true)
349 for z in a.dropFirst() {
350 v = try evaluate(z, e)
351 if !truth(v) {
352 break
353 }
354 }
355 return v
356 case "or":
357 for z in a.dropFirst() {
358 let v = try evaluate(z, e)
359 if truth(v) {
360 return v
361 }
362 }
363 return .boolean(false)
364 case "when", "unless":
365 try shape(n >= 3, x)
366 return truth(try evaluate(a[1], e)) == (h == "when") ? try body(Array(a.dropFirst(2)), e) : .void
367 default:
368 break
369 }
370 let f = try evaluate(a[0], e)
371 var args = [Value]()
372 for z in a.dropFirst() {
373 args.append(try evaluate(z, e))
374 }
375 guard case .procedure(let p) = f else {
376 try fail("not a procedure: " + repr(f))
377 }
378 return try p.call(args)
379}

631final class Reader {
632 let lines: [[UInt8]]
633 var row = 0, col = 0
634 init(_ lines: [String]) {
635 self.lines = lines.map {
636 Array($0.utf8)
637 }
638 }
639 func skip() {
640 row += 1
641 col = 0
642 }
643 func end() -> Bool {
644 while row < lines.count {
645 let s = lines[row]
646 while col < s.count && s[col] == 32 {
647 col += 1
648 }
649 if col < s.count && s[col] != 59 {
650 return false
651 }
652 skip()
653 }
654 return true
655 }
656 func token() throws -> (UInt8, Value?) {
657 if end() {
658 try fail("unexpected end of input")
659 }
660 let s = lines[row]
661 var c = s[col]
662 col += 1
663 if [40, 41, 39].contains(c) {
664 return (c, nil)
665 }
666 if c == 34 {
667 var v = [UInt8]()
668 while col < s.count {
669 c = s[col]
670 col += 1
671 if c == 34 {
672 return (118, .string(String(decoding: v, as: UTF8.self)))
673 }
674 if c == 92 {
675 if col == s.count {
676 break
677 }
678 c = s[col]
679 col += 1
680 if ![34, 92, 110].contains(c) {
681 try fail("bad escape")
682 }
683 if c == 110 {
684 c = 10
685 }
686 }
687 v.append(c)
688 }
689 try fail("unterminated string")
690 }
691 let first = col - 1
692 while col < s.count && ![32, 40, 41, 39, 34, 59].contains(s[col]) {
693 col += 1
694 }
695 let a = String(decoding: s[first..<col], as: UTF8.self)
696 if a == "." {
697 return (46, nil)
698 }
699 if a.range(of: "^[+-]?[0-9]+$", options: .regularExpression) != nil {
700 return (118, .integer(Int64(a)!))
701 }
702 if a == "#t" || a == "#f" {
703 return (118, .boolean(a == "#t"))
704 }
705 if a.hasPrefix("#") {
706 try fail("bad token: " + a)
707 }
708 return (118, .symbol(a))
709 }
710 func datum() throws -> Value {
711 try from(token())
712 }
713 func from(_ t: (UInt8, Value?)) throws -> Value {
714 if t.0 == 118 {
715 return t.1!
716 }
717 if t.0 == 39 {
718 return makeList([.symbol("quote"), try datum()])
719 }
720 if t.0 != 40 {
721 try fail("unexpected " + String(UnicodeScalar(t.0)))
722 }
723 var a = [Value]()
724 while true {
725 let t = try token()
726 if t.0 == 41 {
727 return makeList(a)
728 }
729 if t.0 == 46 && !a.isEmpty {
730 let tail = try datum()
731 if try token().0 != 41 {
732 try fail("expected )")
733 }
734 return makeList(a, tail)
735 }
736 a.append(try from(t))
737 }
738 }
739}
