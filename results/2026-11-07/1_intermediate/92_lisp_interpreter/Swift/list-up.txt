final class Env {
    var frame = [String: Value]()
    let parent: Env?
    init(_ p: Env? = nil) {
        parent = p
    func find(_ n: String) throws -> Env {
        var e: Env? = self
        while let current = e {
            if current.frame[n] != nil {
                return current
            e = current.parent
    func get(_ n: String) throws -> Value {
        let v = try find(n).frame[n]!
        if case .uninitialized = v {
            try fail("uninitialized variable: " + n)
        return v
final class Procedure {
    init(
        _ name: String?,
        _ minimum: Int,
        _ variadic: Bool,
        rest: String? = nil,
        params: [String] = [],
        body: [Value] = [],
        env: Env? = nil,
        builtin: (([Value]) throws -> Value)? = nil
    ) {
        self.expressions = body
        self.env = env
    func call(_ a: [Value]) throws -> Value {
        let e = Env(env)
        return try body(expressions, e)
func closure(_ p: ([String], String?), _ b: [Value], _ e: Env, _ name: String? = nil) -> Value {
    .procedure(Procedure(
        name,
        p.0.count,
        p.1 != nil,
        rest: p.1,
        params: p.0,
        body: b,
        env: e
    ))
func body(_ a: [Value], _ e: Env) throws -> Value {
    var v = Value.void
    for x in a {
        v = try evaluate(x, e)
    return v
func evaluate(_ x: Value, _ e: Env) throws -> Value {
    if case .symbol(let n) = x {
        return try e.get(n)
    case "lambda":
        return closure(p!, Array(a.dropFirst(2)), e)
    case "set!":

        let v = try evaluate(a[2], e)
        let target = try e.find(name)
        target.frame[name] = v
    case "let", "let*", "letrec":
        var loop: String?
        if h == "let", n > 1, case .symbol(let name) = a[1] {
            loop = name
        let at = loop == nil ? 1 : 2
        let bs = n >= at + 2 ? bindings(a[at], h != "let*") : nil
        var scope = Env(e)
        if h == "let*" {
            if !bs!.isEmpty {
                scope = e
            for (name, expr) in bs! {
                let v = try evaluate(expr, scope)
                scope = Env(scope)
                scope.frame[name] = v
        } else if h == "letrec" {
            for (name, _) in bs! {
                scope.frame[name] = .uninitialized
            for (name, expr) in bs! {
                scope.frame[name] = try evaluate(expr, scope)
        } else {

            for (i, b) in bs!.enumerated() {

        return try body(Array(a.dropFirst(at + 1)), scope)
