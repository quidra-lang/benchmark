1use std::cell::RefCell;
2use std::collections::{HashMap, HashSet};
3use std::io::{self, Read, Write};
4use std::rc::Rc;
5type Result<T> = std::result::Result<T, String>;
6type Scope = Rc<Environment>;
7#[derive(Clone)]
8enum Value {
9 Int(i64),
10 Bool(bool),
11 Str(String),
12 Sym(String),
13 Nil,
14 Pair(Rc<(Value, Value)>),
15 Proc(Rc<Procedure>),
16 Void,
17 Uninitialized,
18}
19struct Environment {
20 frame: RefCell<HashMap<String, Value>>,
21 parent: Option<Scope>,
22}
23impl Environment {
24 fn new(parent: Option<Scope>) -> Scope {
25 Rc::new(Self {
26 frame: RefCell::new(HashMap::new()),
27 parent,
28 })
29 }
30 fn find(self: &Rc<Self>, name: &str) -> Result<Scope> {
31 let mut e = Some(self.clone());
32 while let Some(scope) = e {
33 if scope.frame.borrow().contains_key(name) {
34 return Ok(scope);
35 }
36 e = scope.parent.clone();
37 }
38 Err(format!("unbound variable: {name}"))
39 }
40 fn get(self: &Rc<Self>, name: &str) -> Result<Value> {
41 let v = self.find(name)?.frame.borrow()[name].clone();
42 if matches!(v, Value::Uninitialized) {
43 Err(format!("uninitialized variable: {name}"))
44 } else {
45 Ok(v)
46 }
47 }
48 fn bind(&self, name: &str, v: Value) {
49 self.frame.borrow_mut().insert(name.to_owned(), v);
50 }
51}
52struct Procedure {
53 name: RefCell<Option<String>>,
54 minimum: usize,
55 variadic: bool,
56 implementation: Implementation,
57}
58enum Implementation {
59 Builtin(&'static str),
60 Closure {
61 params: Vec<String>,
62 rest: Option<String>,
63 body: Vec<Value>,
64 env: Scope,
65 },
66}
67fn list(a: &[Value], mut tail: Value) -> Value {
68 for x in a.iter().rev() {
69 tail = Value::Pair(Rc::new((x.clone(), tail)));
70 }
71 tail
72}
73fn items(v: &Value) -> Option<Vec<Value>> {
74 let mut a = Vec::new();
75 let mut v = v;
76 while let Value::Pair(p) = v {
77 a.push(p.0.clone());
78 v = &p.1;
79 }
80 if matches!(v, Value::Nil) {
81 Some(a)
82 } else {
83 None
84 }
85}
86fn truth(v: &Value) -> bool {
87 !matches!(v, Value::Bool(false))
88}
89fn repr(v: &Value, quoted: bool) -> String {
90 match v {
91 Value::Int(n) => n.to_string(),
92 Value::Bool(b) => (if *b { "#t" } else { "#f" }).into(),
93 Value::Str(s) => {
94 if quoted {
95 format!(
96 "\"{}\"",
97 s.replace('\\', "\\\\")
98 .replace('"', "\\\"")
99 .replace('\n', "\\n")
100 )
101 } else {
102 s.clone()
103 }
104 }
105 Value::Sym(s) => s.clone(),
106 Value::Nil => "()".into(),
107 Value::Void | Value::Uninitialized => "#<void>".into(),
108 Value::Proc(p) => match &*p.name.borrow() {
109 Some(n) => format!("#<procedure {n}>"),
110 None => "#<procedure>".into(),
111 },
112 Value::Pair(_) => {
113 let mut a = Vec::new();
114 let mut v = v;
115 while let Value::Pair(p) = v {
116 a.push(repr(&p.0, quoted));
117 v = &p.1;
118 }
119 if !matches!(v, Value::Nil) {
120 a.push(".".into());
121 a.push(repr(v, quoted));
122 }
123 format!("({})", a.join(" "))
124 }
125 }
126}
127fn equal(a: &Value, b: &Value, deep: bool) -> bool {
128 match (a, b) {
129 (Value::Int(a), Value::Int(b)) => a == b,
130 (Value::Bool(a), Value::Bool(b)) => a == b,
131 (Value::Str(a), Value::Str(b)) | (Value::Sym(a), Value::Sym(b)) => a == b,
132 (Value::Nil, Value::Nil) | (Value::Void, Value::Void) => true,
133 (Value::Pair(a), Value::Pair(b)) => {
134 if deep {
135 equal(&a.0, &b.0, true) && equal(&a.1, &b.1, true)
136 } else {
137 Rc::ptr_eq(a, b)
138 }
139 }
140 (Value::Proc(a), Value::Proc(b)) => Rc::ptr_eq(a, b),
141 _ => false,
142 }
143}
144fn shape(valid: bool, x: &Value) -> Result<()> {
145 if valid {
146 Ok(())
147 } else {
148 Err(format!("bad syntax: {}", repr(x, true)))
149 }
150}
151fn parameters(v: &Value) -> Option<(Vec<String>, Option<String>)> {
152 let mut names = Vec::new();
153 let mut v = v;
154 while let Value::Pair(p) = v {
155 if let Value::Sym(s) = &p.0 {
156 names.push(s.clone())
157 } else {
158 return None;
159 }
160 v = &p.1;
161 }
162 let rest = match v {
163 Value::Nil => None,
164 Value::Sym(s) => Some(s.clone()),
165 _ => return None,
166 };
167 let mut seen = HashSet::new();
168 for n in names.iter().chain(rest.iter()) {
169 if !seen.insert(n) {
170 return None;
171 }
172 }
173 Some((names, rest))
174}
175fn closure(
176 p: (Vec<String>, Option<String>),
177 body: Vec<Value>,
178 env: Scope,
179 name: Option<String>,
180) -> Value {
181 Value::Proc(Rc::new(Procedure {
182 name: RefCell::new(name),
183 minimum: p.0.len(),
184 variadic: p.1.is_some(),
185 implementation: Implementation::Closure {
186 params: p.0,
187 rest: p.1,
188 body,
189 env,
190 },
191 }))
192}
193fn bindings(v: &Value, unique: bool) -> Option<Vec<(String, Value)>> {
194 let a = items(v)?;
195 let mut bs = Vec::new();
196 let mut seen = HashSet::new();
197 for x in a {
198 let b = items(&x)?;
199 if b.len() != 2 {
200 return None;
201 }
202 if let Value::Sym(n) = &b[0] {
203 if unique && !seen.insert(n.clone()) {
204 return None;
205 }
206 bs.push((n.clone(), b[1].clone()));
207 } else {
208 return None;
209 }
210 }
211 Some(bs)
212}
213fn body(a: &[Value], e: &Scope, out: &mut Output) -> Result<Value> {
214 let mut v = Value::Void;
215 for x in a {
216 v = evaluate(x, e, out)?;
217 }
218 Ok(v)
219}
220fn evaluate(x: &Value, e: &Scope, out: &mut Output) -> Result<Value> {
221 if let Value::Sym(n) = x {
222 return e.get(n);
223 }
224 if !matches!(x, Value::Pair(_)) {
225 shape(!matches!(x, Value::Nil), x)?;
226 return Ok(x.clone());
227 }
228 let a = items(x).ok_or_else(|| format!("bad syntax: {}", repr(x, true)))?;
229 let n = a.len();
230 let h = if let Value::Sym(s) = &a[0] {
231 s.as_str()
232 } else {
233 ""
234 };
235 match h {
236 "quote" => {
237 shape(n == 2, x)?;
238 return Ok(a[1].clone());
239 }
240 "if" => {
241 shape(n == 3 || n == 4, x)?;
242 return if truth(&evaluate(&a[1], e, out)?) {
243 evaluate(&a[2], e, out)
244 } else if n == 4 {
245 evaluate(&a[3], e, out)
246 } else {
247 Ok(Value::Void)
248 };
249 }
250 "begin" => return body(&a[1..], e, out),
251 "lambda" => {
252 let p = if n >= 3 { parameters(&a[1]) } else { None };
253 shape(p.is_some(), x)?;
254 return Ok(closure(p.unwrap(), a[2..].to_vec(), e.clone(), None));
255 }
256 "define" => {
257 if n > 1 {
258 if let Value::Sym(name) = &a[1] {
259 shape(n == 3, x)?;
260 let v = evaluate(&a[2], e, out)?;
261 if let Value::Proc(p) = &v {
262 if matches!(p.implementation, Implementation::Closure { .. })
263 && p.name.borrow().is_none()
264 {
265 *p.name.borrow_mut() = Some(name.clone());
266 }
267 }
268 e.bind(name, v);
269 return Ok(Value::Void);
270 }
271 }
272 shape(n >= 3, x)?;
273 if let Value::Pair(t) = &a[1] {
274 if let Value::Sym(name) = &t.0 {
275 let p = parameters(&t.1);
276 shape(p.is_some(), x)?;
277 e.bind(
278 name,
279 closure(p.unwrap(), a[2..].to_vec(), e.clone(), Some(name.clone())),
280 );
281 return Ok(Value::Void);
282 }
283 }
284 shape(false, x)?;
285 }
286 "set!" => {
287 shape(n == 3 && matches!(&a[1], Value::Sym(_)), x)?;
288 let name = if let Value::Sym(s) = &a[1] {
289 s
290 } else {
291 unreachable!()
292 };
293 let v = evaluate(&a[2], e, out)?;
294 e.find(name)?.bind(name, v);
295 return Ok(Value::Void);
296 }
297 "let" | "let*" | "letrec" => {
298 let named = h == "let" && n > 1 && matches!(&a[1], Value::Sym(_));
299 let at = if named { 2 } else { 1 };
300 let bs = if n >= at + 2 {
301 bindings(&a[at], h != "let*")
302 } else {
303 None
304 };
305 shape(bs.is_some(), x)?;
306 let bs = bs.unwrap();
307 let mut scope = Environment::new(Some(e.clone()));
308 if h == "let*" {
309 if !bs.is_empty() {
310 scope = e.clone()
311 }
312 for (name, expr) in &bs {
313 let v = evaluate(expr, &scope, out)?;
314 scope = Environment::new(Some(scope));
315 scope.bind(name, v);
316 }
317 } else if h == "letrec" {
318 for (name, _) in &bs {
319 scope.bind(name, Value::Uninitialized)
320 }
321 for (name, expr) in &bs {
322 let v = evaluate(expr, &scope, out)?;
323 scope.bind(name, v);
324 }
325 } else {
326 let mut vs = Vec::new();
327 for (_, expr) in &bs {
328 vs.push(evaluate(expr, e, out)?)
329 }
330 if named {
331 let name = if let Value::Sym(s) = &a[1] {
332 s
333 } else {
334 unreachable!()
335 };
336 let f = closure(
337 (bs.iter().map(|b| b.0.clone()).collect(), None),
338 a[3..].to_vec(),
339 scope.clone(),
340 Some(name.clone()),
341 );
342 scope.bind(name, f.clone());
343 return call(&f, &vs, out);
344 }
345 for ((name, _), v) in bs.iter().zip(vs) {
346 scope.bind(name, v)
347 }
348 }
349 return body(&a[at + 1..], &scope, out);
350 }
351 "cond" => {
352 let mut cs = Vec::new();
353 for (i, c) in a[1..].iter().enumerate() {
354 let c = items(c);
355 shape(c.as_ref().is_some_and(|c| !c.is_empty()), x)?;
356 let c = c.unwrap();
357 if matches!(&c[0], Value::Sym(s)if s == "else") {
358 shape(c.len() >= 2 && i == n - 2, x)?
359 }
360 cs.push(c)
361 }
362 for c in cs {
363 if matches!(&c[0], Value::Sym(s)if s == "else") {
364 return body(&c[1..], e, out);
365 }
366 let v = evaluate(&c[0], e, out)?;
367 if truth(&v) {
368 return if c.len() == 1 {
369 Ok(v)
370 } else {
371 body(&c[1..], e, out)
372 };
373 }
374 }
375 return Ok(Value::Void);
376 }
377 "and" => {
378 let mut v = Value::Bool(true);
379 for z in &a[1..] {
380 v = evaluate(z, e, out)?;
381 if !truth(&v) {
382 break;
383 }
384 }
385 return Ok(v);
386 }
387 "or" => {
388 for z in &a[1..] {
389 let v = evaluate(z, e, out)?;
390 if truth(&v) {
391 return Ok(v);
392 }
393 }
394 return Ok(Value::Bool(false));
395 }
396 "when" | "unless" => {
397 shape(n >= 3, x)?;
398 return if truth(&evaluate(&a[1], e, out)?) == (h == "when") {
399 body(&a[2..], e, out)
400 } else {
401 Ok(Value::Void)
402 };
403 }
404 _ => {}
405 }
406 let f = evaluate(&a[0], e, out)?;
407 let mut args = Vec::new();
408 for z in &a[1..] {
409 args.push(evaluate(z, e, out)?)
410 }
411 call(&f, &args, out)
412}
413fn call(f: &Value, a: &[Value], out: &mut Output) -> Result<Value> {
414 let Value::Proc(p) = f else {
415 return Err(format!("not a procedure: {}", repr(f, true)));
416 };
417 if a.len() < p.minimum || (!p.variadic && a.len() != p.minimum) {
418 return Err(format!(
419 "{}: wrong number of arguments (expected {}{}, got {})",
420 p.name.borrow().as_deref().unwrap_or("lambda"),
421 if p.variadic { "at least " } else { "" },
422 p.minimum,
423 a.len()
424 ));
425 }
426 match &p.implementation {
427 Implementation::Builtin(n) => builtin(n, a, out),
428 Implementation::Closure {
429 params,
430 rest,
431 body: b,
432 env,
433 } => {
434 let e = Environment::new(Some(env.clone()));
435 for (n, v) in params.iter().zip(a) {
436 e.bind(n, v.clone())
437 }
438 if let Some(r) = rest {
439 e.bind(r, list(&a[params.len()..], Value::Nil))
440 }
441 body(b, &e, out)
442 }
443 }
444}

464fn expect(n: &str, v: &Value, k: &str) -> Result<()> {
465 let ok = match k {
466 "integer" => matches!(v, Value::Int(_)),
467 "string" => matches!(v, Value::Str(_)),
468 "pair" => matches!(v, Value::Pair(_)),
469 "procedure" => matches!(v, Value::Proc(_)),
470 _ => items(v).is_some(),
471 };
472 if ok {
473 Ok(())
474 } else {
475 Err(format!("{n}: expected {k}, got {}", repr(v, true)))
476 }
477}
478fn as_list(n: &str, v: &Value) -> Result<Vec<Value>> {
479 expect(n, v, "list")?;
480 Ok(items(v).unwrap())
481}
482fn integer(v: &Value) -> i64 {
483 if let Value::Int(n) = v {
484 *n
485 } else {
486 unreachable!()
487 }
488}
489fn builtin(n: &str, a: &[Value], out: &mut Output) -> Result<Value> {
490 if [
491 "+",
492 "*",
493 "-",
494 "quotient",
495 "remainder",
496 "modulo",
497 "abs",
498 "min",
499 "max",
500 "gcd",
501 "=",
502 "<",
503 ">",
504 "<=",
505 ">=",
506 ]
507 .contains(&n)
508 {
509 for v in a {
510 expect(n, v, "integer")?
511 }
512 let z: Vec<i64> = a.iter().map(integer).collect();
513 let v = z.first().copied().unwrap_or(0);
514 return Ok(match n {
515 "+" => Value::Int(z.iter().sum()),
516 "*" => Value::Int(z.iter().product()),
517 "-" => Value::Int(if z.len() == 1 {
518 -v
519 } else {
520 v - z[1..].iter().sum::<i64>()
521 }),
522 "abs" => Value::Int(v.abs()),
523 "min" => Value::Int(*z.iter().min().unwrap()),
524 "max" => Value::Int(*z.iter().max().unwrap()),
525 "gcd" => Value::Int(z.into_iter().fold(0, |mut g, x| {
526 let mut b = x.abs();
527 while b != 0 {
528 (g, b) = (b, g % b)
529 }
530 g
531 })),
532 "quotient" | "remainder" | "modulo" => {
533 let b = z[1];
534 if b == 0 {
535 return Err(format!("{n}: division by zero"));
536 }
537 let r = v % b;
538 Value::Int(if n == "quotient" {
539 v / b
540 } else if n == "modulo" && r != 0 && ((r < 0) != (b < 0)) {
541 r + b
542 } else {
543 r
544 })
545 }
546 _ => Value::Bool(z.windows(2).all(|w| match n {
547 "=" => w[0] == w[1],
548 "<" => w[0] < w[1],
549 ">" => w[0] > w[1],
550 "<=" => w[0] <= w[1],
551 _ => w[0] >= w[1],
552 })),
553 });
554 }
555 Ok(match n {
556 "not" => Value::Bool(!truth(&a[0])),
557 "eq?" | "equal?" => Value::Bool(equal(&a[0], &a[1], n == "equal?")),
558 "null?" => Value::Bool(matches!(a[0], Value::Nil)),
559 "pair?" => Value::Bool(matches!(a[0], Value::Pair(_))),
560 "list?" => Value::Bool(items(&a[0]).is_some()),
561 "number?" => Value::Bool(matches!(a[0], Value::Int(_))),
562 "symbol?" => Value::Bool(matches!(a[0], Value::Sym(_))),
563 "procedure?" => Value::Bool(matches!(a[0], Value::Proc(_))),
564 "cons" => Value::Pair(Rc::new((a[0].clone(), a[1].clone()))),
565 "list" => list(a, Value::Nil),
566 "car" | "cdr" | "cadr" => {
567 expect(n, &a[0], "pair")?;
568 let Value::Pair(p) = &a[0] else {
569 unreachable!()
570 };
571 if n == "car" {
572 p.0.clone()
573 } else if n == "cdr" {
574 p.1.clone()
575 } else {
576 expect(n, &p.1, "pair")?;
577 let Value::Pair(q) = &p.1 else { unreachable!() };
578 q.0.clone()
579 }
580 }
581 "length" | "reverse" | "list-ref" => {
582 let mut z = as_list(n, &a[0])?;
583 if n == "length" {
584 Value::Int(z.len() as i64)
585 } else if n == "reverse" {
586 z.reverse();
587 list(&z, Value::Nil)
588 } else {
589 expect(n, &a[1], "integer")?;
590 let k = integer(&a[1]);
591 if k < 0 || k as usize >= z.len() {
592 return Err(format!("list-ref: index out of range: {k}"));
593 }
594 z[k as usize].clone()
595 }
596 }
597 "append" => {
598 if a.is_empty() {
599 Value::Nil
600 } else {
601 let mut z = Vec::new();
602 for v in &a[..a.len() - 1] {
603 z.extend(as_list(n, v)?)
604 }
605 list(&z, a.last().unwrap().clone())
606 }
607 }
608 "map" | "filter" | "fold-left" | "apply" => {
609 expect(n, &a[0], "procedure")?;
610 if n == "apply" {
611 let mut z = a[1..a.len() - 1].to_vec();
612 z.extend(as_list(n, a.last().unwrap())?);
613 return call(&a[0], &z, out);
614 }
615 let mut ls = Vec::new();
616 for v in &a[if n == "fold-left" { 2 } else { 1 }..] {
617 ls.push(as_list(n, v)?)
618 }
619 let count = ls.iter().map(Vec::len).min().unwrap();
620 let mut result = Vec::new();
621 let mut acc = a[1].clone();
622 for i in 0..count {
623 if n == "filter" {
624 if truth(&call(&a[0], &[ls[0][i].clone()], out)?) {
625 result.push(ls[0][i].clone())
626 }
627 } else {
628 let mut z = Vec::new();
629 if n == "fold-left" {
630 z.push(acc.clone())
631 }
632 z.extend(ls.iter().map(|l| l[i].clone()));
633 let v = call(&a[0], &z, out)?;
634 if n == "map" { result.push(v) } else { acc = v }
635 }
636 }
637 if n == "fold-left" {
638 acc
639 } else {
640 list(&result, Value::Nil)
641 }
642 }
643 "string-append" => {
644 let mut s = String::new();
645 for v in a {
646 expect(n, v, "string")?;
647 if let Value::Str(t) = v {
648 s.push_str(t)
649 }
650 }
651 Value::Str(s)
652 }
653 "number->string" => {
654 expect(n, &a[0], "integer")?;
655 Value::Str(repr(&a[0], true))
656 }
657 "display" => {
658 out.write(&repr(&a[0], false));
659 Value::Void
660 }
661 "newline" => {
662 out.write("\n");
663 Value::Void
664 }
665 "error" => {
666 expect(n, &a[0], "string")?;
667 let Value::Str(mut s) = a[0].clone() else {
668 unreachable!()
669 };
670 for v in &a[1..] {
671 s.push(' ');
672 s.push_str(&repr(v, true))
673 }
674 return Err(s);
675 }
676 _ => unreachable!(),
677 })
678}

738enum Token {
739 Datum(Value),
740 Mark(u8),
741}
742struct Reader {
743 lines: Vec<Vec<u8>>,
744 row: usize,
745 col: usize,
746}
747impl Reader {
748 fn skip(&mut self) {
749 self.row += 1;
750 self.col = 0
751 }
752 fn end(&mut self) -> bool {
753 while self.row < self.lines.len() {
754 let s = &self.lines[self.row];
755 while self.col < s.len() && s[self.col] == b' ' {
756 self.col += 1
757 }
758 if self.col < s.len() && s[self.col] != b';' {
759 return false;
760 }
761 self.skip()
762 }
763 true
764 }
765 fn token(&mut self) -> Result<Token> {
766 if self.end() {
767 return Err("unexpected end of input".into());
768 }
769 let s = &self.lines[self.row];
770 let c = s[self.col];
771 self.col += 1;
772 if b"()'".contains(&c) {
773 return Ok(Token::Mark(c));
774 }
775 if c == b'"' {
776 let mut v = String::new();
777 while self.col < s.len() {
778 let mut c = s[self.col];
779 self.col += 1;
780 if c == b'"' {
781 return Ok(Token::Datum(Value::Str(v)));
782 }
783 if c == b'\\' {
784 if self.col == s.len() {
785 break;
786 }
787 c = s[self.col];
788 self.col += 1;
789 if !b"\"\\n".contains(&c) {
790 return Err("bad escape".into());
791 }
792 if c == b'n' {
793 c = b'\n'
794 }
795 }
796 v.push(c as char)
797 }
798 return Err("unterminated string".into());
799 }
800 let first = self.col - 1;
801 while self.col < s.len() && !b" ()'\";".contains(&s[self.col]) {
802 self.col += 1
803 }
804 let a = String::from_utf8(s[first..self.col].to_vec()).unwrap();
805 if a == "." {
806 return Ok(Token::Mark(b'.'));
807 }
808 let digits = a.trim_start_matches(['+', '-']);
809 let integer = !digits.is_empty()
810 && digits.bytes().all(|c| c.is_ascii_digit())
811 && a.bytes().filter(|c| *c == b'+' || *c == b'-').count() <= 1;
812 if integer {
813 return Ok(Token::Datum(Value::Int(a.parse().unwrap())));
814 }
815 if a == "#t" || a == "#f" {
816 return Ok(Token::Datum(Value::Bool(a == "#t")));
817 }
818 if a.starts_with('#') {
819 return Err(format!("bad token: {a}"));
820 }
821 Ok(Token::Datum(Value::Sym(a)))
822 }
823 fn datum(&mut self) -> Result<Value> {
824 let t = self.token()?;
825 self.from(t)
826 }
827 fn from(&mut self, t: Token) -> Result<Value> {
828 match t {
829 Token::Datum(v) => Ok(v),
830 Token::Mark(b'\'') => Ok(list(
831 &[Value::Sym("quote".into()), self.datum()?],
832 Value::Nil,
833 )),
834 Token::Mark(b'(') => {
835 let mut a = Vec::new();
836 loop {
837 let t = self.token()?;
838 if matches!(t, Token::Mark(b')')) {
839 return Ok(list(&a, Value::Nil));
840 }
841 if matches!(t, Token::Mark(b'.')) && !a.is_empty() {
842 let tail = self.datum()?;
843 if !matches!(self.token()?, Token::Mark(b')')) {
844 return Err("expected )".into());
845 }
846 return Ok(list(&a, tail));
847 }
848 a.push(self.from(t)?)
849 }
850 }
851 Token::Mark(c) => Err(format!("unexpected {}", c as char)),
852 }
853 }
854}
855fn main() {
856 let mut input = String::new();
857 io::stdin().read_to_string(&mut input).unwrap();
858 let lines = input
859 .lines()
860 .take_while(|s| *s != "(exit)")
861 .map(|s| s.as_bytes().to_vec())
862 .collect();
863 let mut r = Reader {
864 lines,
865 row: 0,
866 col: 0,
867 };
868 let e = global();
869 let mut out = Output {
870 text: String::new(),
871 mid: false,
872 };
873 while !r.end() {
874 let x = match r.datum() {
875 Ok(x) => x,
876 Err(err) => {
877 out.line(&format!("Error: read: {err}"));
878 if err == "unexpected end of input" {
879 break;
880 }
881 r.skip();
882 continue;
883 }
884 };
885 match evaluate(&x, &e, &mut out) {
886 Ok(Value::Void) => {}
887 Ok(v) => out.line(&repr(&v, true)),
888 Err(err) => out.line(&format!("Error: {err}")),
889 }
890 }
891 if out.mid {
892 out.write("\n")
893 }
894 io::stdout().write_all(out.text.as_bytes()).unwrap();
895}
