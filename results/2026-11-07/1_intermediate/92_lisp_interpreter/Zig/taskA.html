1const std = @import("std");
2const Allocator = std.mem.Allocator;
3const Values = std.ArrayList(Value);
4const Value = union(enum) { integer: i64, boolean: bool, string: []const u8, symbol: []const u8, nil, pair: *Pair, procedure: *Procedure, void_value, uninitialized };
5const Pair = struct { car: Value, cdr: Value };
6const Environment = struct { frame: std.StringHashMapUnmanaged(Value) = .empty, parent: ?*Environment = null };
7const Procedure = struct { name: []const u8 = "", minimum: usize = 0, variadic: bool = false, builtin: bool = false, params: []const []const u8 = &.{}, rest: []const u8 = "", body: []const Value = &.{}, env: ?*Environment = null };
8fn eqs(a: []const u8, b: []const u8) bool {
9 return std.mem.eql(u8, a, b);
10}
11fn is(v: Value, tag: std.meta.Tag(Value)) bool {
12 return std.meta.activeTag(v) == tag;
13}
14fn truth(v: Value) bool {
15 return !is(v, .boolean) or v.boolean;
16}
17fn proper(v0: Value) bool {
18 var v = v0;
19 while (is(v, .pair)) v = v.pair.cdr;
20 return is(v, .nil);
21}
22fn equal(a: Value, b: Value, deep: bool) bool {
23 if (std.meta.activeTag(a) != std.meta.activeTag(b)) return false;
24 return switch (a) {
25 .integer => |n| n == b.integer,
26 .boolean => |flag| flag == b.boolean,
27 .string => |s| eqs(s, b.string),
28 .symbol => |s| eqs(s, b.symbol),
29 .pair => |p| if (deep) equal(p.car, b.pair.car, true) and equal(p.cdr, b.pair.cdr, true) else p == b.pair,
30 .procedure => |p| p == b.procedure,
31 else => true,
32 };
33}
34const Machine = struct {
35 gpa: Allocator,
36 why: []const u8 = "",
37 output: std.ArrayList(u8) = .empty,
38 mid: bool = false,
39 fn fail(m: *Machine, comptime fmt: []const u8, args: anytype) anyerror {
40 m.why = std.fmt.allocPrint(m.gpa, fmt, args) catch unreachable;
41 return error.Failure;
42 }
43 fn pair(m: *Machine, a: Value, b: Value) anyerror!Value {
44 const p = try m.gpa.create(Pair);
45 p.* = .{ .car = a, .cdr = b };
46 return .{ .pair = p };
47 }
48 fn list(m: *Machine, a: []const Value, tail: Value) anyerror!Value {
49 var v = tail;
50 var i = a.len;
51 while (i > 0) {
52 i -= 1;
53 v = try m.pair(a[i], v);
54 }
55 return v;
56 }
57 fn items(m: *Machine, v0: Value) anyerror!?[]const Value {
58 var a: Values = .empty;
59 var v = v0;
60 while (is(v, .pair)) {
61 try a.append(m.gpa, v.pair.car);
62 v = v.pair.cdr;
63 }
64 return if (is(v, .nil)) try a.toOwnedSlice(m.gpa) else null;
65 }
66 fn appendText(m: *Machine, a: *std.ArrayList(u8), s: []const u8) anyerror!void {
67 try a.appendSlice(m.gpa, s);
68 }
69 fn render(m: *Machine, v0: Value, quoted: bool, a: *std.ArrayList(u8)) anyerror!void {
70 switch (v0) {
71 .integer => |n| try m.appendText(a, try std.fmt.allocPrint(m.gpa, "{d}", .{n})),
72 .boolean => |b| try m.appendText(a, if (b) "#t" else "#f"),
73 .symbol => |s| try m.appendText(a, s),
74 .nil => try m.appendText(a, "()"),
75 .void_value, .uninitialized => try m.appendText(a, "#<void>"),
76 .string => |s| {
77 if (!quoted) {
78 try m.appendText(a, s);
79 return;
80 }
81 try a.append(m.gpa, '"');
82 for (s) |c| {
83 switch (c) {
84 '"' => try m.appendText(a, "\\\""),
85 '\\' => try m.appendText(a, "\\\\"),
86 '\n' => try m.appendText(a, "\\n"),
87 else => try a.append(m.gpa, c),
88 }
89 }
90 try a.append(m.gpa, '"');
91 },
92 .procedure => |p| {
93 try m.appendText(a, "#<procedure");
94 if (p.name.len > 0) {
95 try a.append(m.gpa, ' ');
96 try m.appendText(a, p.name);
97 }
98 try a.append(m.gpa, '>');
99 },
100 .pair => {
101 var v = v0;
102 try a.append(m.gpa, '(');
103 var first = true;
104 while (is(v, .pair)) {
105 if (!first) try a.append(m.gpa, ' ');
106 first = false;
107 try m.render(v.pair.car, quoted, a);
108 v = v.pair.cdr;
109 }
110 if (!is(v, .nil)) {
111 try m.appendText(a, " . ");
112 try m.render(v, quoted, a);
113 }
114 try a.append(m.gpa, ')');
115 },
116 }
117 }
118 fn repr(m: *Machine, v: Value, quoted: bool) anyerror![]const u8 {
119 var a: std.ArrayList(u8) = .empty;
120 try m.render(v, quoted, &a);
121 return a.toOwnedSlice(m.gpa);
122 }
123 fn shape(m: *Machine, ok: bool, x: Value) anyerror!void {
124 if (!ok) return m.fail("bad syntax: {s}", .{try m.repr(x, true)});
125 }
126 fn env(m: *Machine, parent: ?*Environment) anyerror!*Environment {
127 const e = try m.gpa.create(Environment);
128 e.* = .{ .parent = parent };
129 return e;
130 }
131 fn bind(m: *Machine, e: *Environment, n: []const u8, v: Value) anyerror!void {
132 try e.frame.put(m.gpa, n, v);
133 }
134 fn find(m: *Machine, e0: *Environment, n: []const u8) anyerror!*Environment {
135 var e: ?*Environment = e0;
136 while (e) |scope| {
137 if (scope.frame.contains(n)) return scope;
138 e = scope.parent;
139 }
140 return m.fail("unbound variable: {s}", .{n});
141 }
142 fn lookup(m: *Machine, e: *Environment, n: []const u8) anyerror!Value {
143 const scope = try m.find(e, n);
144 const v = scope.frame.get(n).?;
145 if (is(v, .uninitialized)) return m.fail("uninitialized variable: {s}", .{n});
146 return v;
147 }
148 const Parameters = struct { names: []const []const u8, rest: []const u8 };
149 fn parameters(m: *Machine, v0: Value) anyerror!?Parameters {
150 var names: std.ArrayList([]const u8) = .empty;
151 var v = v0;
152 while (is(v, .pair)) {
153 if (!is(v.pair.car, .symbol)) return null;
154 const n = v.pair.car.symbol;
155 for (names.items) |old| if (eqs(n, old)) return null;
156 try names.append(m.gpa, n);
157 v = v.pair.cdr;
158 }
159 var rest: []const u8 = "";
160 if (!is(v, .nil)) {
161 if (!is(v, .symbol)) return null;
162 rest = v.symbol;
163 for (names.items) |old| if (eqs(rest, old)) return null;
164 }
165 return .{ .names = try names.toOwnedSlice(m.gpa), .rest = rest };
166 }
167 fn closure(m: *Machine, p: Parameters, b: []const Value, e: *Environment, n: []const u8) anyerror!Value {
168 const f = try m.gpa.create(Procedure);
169 f.* = .{ .name = n, .minimum = p.names.len, .variadic = p.rest.len > 0, .params = p.names, .rest = p.rest, .body = b, .env = e };
170 return .{ .procedure = f };
171 }
172 const Binding = struct { name: []const u8, expr: Value };
173 fn bindings(m: *Machine, v: Value, unique: bool) anyerror!?[]const Binding {
174 const a = (try m.items(v)) orelse return null;
175 var bs: std.ArrayList(Binding) = .empty;
176 for (a) |x| {
177 const b = (try m.items(x)) orelse return null;
178 if (b.len != 2 or !is(b[0], .symbol)) return null;
179 const n = b[0].symbol;
180 if (unique) {
181 for (bs.items) |old| {
182 if (eqs(n, old.name)) return null;
183 }
184 }
185 try bs.append(m.gpa, .{ .name = n, .expr = b[1] });
186 }
187 return bs.toOwnedSlice(m.gpa);
188 }
189 fn body(m: *Machine, a: []const Value, e: *Environment) anyerror!Value {
190 var v: Value = .void_value;
191 for (a) |x| v = try m.evaluate(x, e);
192 return v;
193 }
194 fn evaluate(m: *Machine, x: Value, e: *Environment) anyerror!Value {
195 if (is(x, .symbol)) return m.lookup(e, x.symbol);
196 if (!is(x, .pair)) {
197 try m.shape(!is(x, .nil), x);
198 return x;
199 }
200 const a = (try m.items(x)) orelse return m.fail("bad syntax: {s}", .{try m.repr(x, true)});
201 const n = a.len;
202 const h = if (is(a[0], .symbol)) a[0].symbol else "";
203 if (eqs(h, "quote")) {
204 try m.shape(n == 2, x);
205 return a[1];
206 }
207 if (eqs(h, "if")) {
208 try m.shape(n == 3 or n == 4, x);
209 if (truth(try m.evaluate(a[1], e))) return m.evaluate(a[2], e);
210 return if (n == 4) m.evaluate(a[3], e) else .void_value;
211 }
212 if (eqs(h, "begin")) return m.body(a[1..], e);
213 if (eqs(h, "lambda")) {
214 try m.shape(n >= 3, x);
215 const p = (try m.parameters(a[1])) orelse return m.fail("bad syntax: {s}", .{try m.repr(x, true)});
216 return m.closure(p, a[2..], e, "");
217 }
218 if (eqs(h, "define")) {
219 try m.shape(n >= 3, x);
220 if (is(a[1], .symbol)) {
221 try m.shape(n == 3, x);
222 const v = try m.evaluate(a[2], e);
223 if (is(v, .procedure) and !v.procedure.builtin and v.procedure.name.len == 0) v.procedure.name = a[1].symbol;
224 try m.bind(e, a[1].symbol, v);
225 } else {
226 try m.shape(is(a[1], .pair) and is(a[1].pair.car, .symbol), x);
227 const p = (try m.parameters(a[1].pair.cdr)) orelse return m.fail("bad syntax: {s}", .{try m.repr(x, true)});
228 const name = a[1].pair.car.symbol;
229 try m.bind(e, name, try m.closure(p, a[2..], e, name));
230 }
231 return .void_value;
232 }
233 if (eqs(h, "set!")) {
234 try m.shape(n == 3 and is(a[1], .symbol), x);
235 const v = try m.evaluate(a[2], e);
236 const scope = try m.find(e, a[1].symbol);
237 try m.bind(scope, a[1].symbol, v);
238 return .void_value;
239 }
240 if (eqs(h, "let") or eqs(h, "let*") or eqs(h, "letrec")) {
241 const named = eqs(h, "let") and n > 1 and is(a[1], .symbol);
242 const at: usize = if (named) 2 else 1;
243 try m.shape(n >= at + 2, x);
244 const bs = (try m.bindings(a[at], !eqs(h, "let*"))) orelse return m.fail("bad syntax: {s}", .{try m.repr(x, true)});
245 var scope = try m.env(e);
246 if (eqs(h, "let*")) {
247 if (bs.len > 0) scope = e;
248 for (bs) |b| {
249 const v = try m.evaluate(b.expr, scope);
250 scope = try m.env(scope);
251 try m.bind(scope, b.name, v);
252 }
253 } else if (eqs(h, "letrec")) {
254 for (bs) |b| try m.bind(scope, b.name, .uninitialized);
255 for (bs) |b| try m.bind(scope, b.name, try m.evaluate(b.expr, scope));
256 } else {
257 var vs: Values = .empty;
258 var names: std.ArrayList([]const u8) = .empty;
259 for (bs) |b| {
260 try vs.append(m.gpa, try m.evaluate(b.expr, e));
261 try names.append(m.gpa, b.name);
262 }
263 if (named) {
264 const name = a[1].symbol;
265 const f = try m.closure(.{ .names = names.items, .rest = "" }, a[3..], scope, name);
266 try m.bind(scope, name, f);
267 return m.call(f, vs.items);
268 }
269 for (bs, vs.items) |b, v| try m.bind(scope, b.name, v);
270 }
271 return m.body(a[at + 1 ..], scope);
272 }
273 if (eqs(h, "cond")) {
274 var cs: std.ArrayList([]const Value) = .empty;
275 for (a[1..], 1..) |x0, i| {
276 const c = (try m.items(x0)) orelse return m.fail("bad syntax: {s}", .{try m.repr(x, true)});
277 try m.shape(c.len > 0, x);
278 if (is(c[0], .symbol) and eqs(c[0].symbol, "else")) try m.shape(c.len >= 2 and i == n - 1, x);
279 try cs.append(m.gpa, c);
280 }
281 for (cs.items) |c| {
282 if (is(c[0], .symbol) and eqs(c[0].symbol, "else")) return m.body(c[1..], e);
283 const v = try m.evaluate(c[0], e);
284 if (truth(v)) return if (c.len == 1) v else m.body(c[1..], e);
285 }
286 return .void_value;
287 }
288 if (eqs(h, "and")) {
289 var v: Value = .{ .boolean = true };
290 for (a[1..]) |z| {
291 v = try m.evaluate(z, e);
292 if (!truth(v)) break;
293 }
294 return v;
295 }
296 if (eqs(h, "or")) {
297 for (a[1..]) |z| {
298 const v = try m.evaluate(z, e);
299 if (truth(v)) return v;
300 }
301 return .{ .boolean = false };
302 }
303 if (eqs(h, "when") or eqs(h, "unless")) {
304 try m.shape(n >= 3, x);
305 return if (truth(try m.evaluate(a[1], e)) == eqs(h, "when")) m.body(a[2..], e) else .void_value;
306 }
307 const f = try m.evaluate(a[0], e);
308 var args: Values = .empty;
309 for (a[1..]) |z| try args.append(m.gpa, try m.evaluate(z, e));
310 return m.call(f, args.items);
311 }
312 fn call(m: *Machine, f: Value, a: []const Value) anyerror!Value {
313 if (!is(f, .procedure)) return m.fail("not a procedure: {s}", .{try m.repr(f, true)});
314 const p = f.procedure;
315 if (a.len < p.minimum or (!p.variadic and a.len != p.minimum)) return m.fail("{s}: wrong number of arguments (expected {s}{d}, got {d})", .{ if (p.name.len == 0) "lambda" else p.name, if (p.variadic) "at least " else "", p.minimum, a.len });
316 if (p.builtin) return m.builtin(p.name, a);
317 const e = try m.env(p.env);
318 for (p.params, 0..) |name, i| try m.bind(e, name, a[i]);
319 if (p.rest.len > 0) try m.bind(e, p.rest, try m.list(a[p.minimum..], .nil));
320 return m.body(p.body, e);
321 }
322 fn write(m: *Machine, s: []const u8) anyerror!void {
323 try m.output.appendSlice(m.gpa, s);
324 if (s.len > 0) m.mid = s[s.len - 1] != '\n';
325 }
326 fn line(m: *Machine, s: []const u8) anyerror!void {
327 if (m.mid) try m.write("\n");
328 try m.write(s);
329 try m.write("\n");
330 }
331 fn expect(m: *Machine, n: []const u8, v: Value, k: []const u8) anyerror!Value {
332 const ok = if (eqs(k, "integer")) is(v, .integer) else if (eqs(k, "string")) is(v, .string) else if (eqs(k, "pair")) is(v, .pair) else if (eqs(k, "procedure")) is(v, .procedure) else proper(v);
333 if (!ok) return m.fail("{s}: expected {s}, got {s}", .{ n, k, try m.repr(v, true) });
334 return v;
335 }
336 fn asList(m: *Machine, n: []const u8, v: Value) anyerror![]const Value {
337 _ = try m.expect(n, v, "list");
338 return (try m.items(v)).?;
339 }
340 fn builtin(m: *Machine, n: []const u8, a: []const Value) anyerror!Value {
341 if (contains(&.{ "+", "*", "-", "quotient", "remainder", "modulo", "abs", "min", "max", "gcd", "=", "<", ">", "<=", ">=" }, n)) {
342 for (a) |v| _ = try m.expect(n, v, "integer");
343 var v: i64 = if (a.len > 0) a[0].integer else 0;
344 if (eqs(n, "+") or eqs(n, "*")) {
345 v = if (eqs(n, "+")) 0 else 1;
346 for (a) |x| v = if (eqs(n, "+")) v + x.integer else v * x.integer;
347 return .{ .integer = v };
348 }
349 if (eqs(n, "-")) {
350 if (a.len == 1) return .{ .integer = -v };
351 for (a[1..]) |x| v -= x.integer;
352 return .{ .integer = v };
353 }
354 if (eqs(n, "abs")) return .{ .integer = if (v < 0) -v else v };
355 if (eqs(n, "min") or eqs(n, "max")) {
356 for (a) |x| v = if (eqs(n, "min")) @min(v, x.integer) else @max(v, x.integer);
357 return .{ .integer = v };
358 }
359 if (eqs(n, "gcd")) {
360 v = 0;
361 for (a) |x| {
362 var b = if (x.integer < 0) -x.integer else x.integer;
363 while (b != 0) {
364 const r = @rem(v, b);
365 v = b;
366 b = r;
367 }
368 }
369 return .{ .integer = v };
370 }
371 if (eqs(n, "quotient") or eqs(n, "remainder") or eqs(n, "modulo")) {
372 const b = a[1].integer;
373 if (b == 0) return m.fail("{s}: division by zero", .{n});
374 const r = @rem(v, b);
375 return .{ .integer = if (eqs(n, "quotient")) @divTrunc(v, b) else if (eqs(n, "modulo") and r != 0 and ((r < 0) != (b < 0))) r + b else r };
376 }
377 for (a[1..], 0..) |x, i| {
378 const l = a[i].integer;
379 const r = x.integer;
380 const ok = if (eqs(n, "=")) l == r else if (eqs(n, "<")) l < r else if (eqs(n, ">")) l > r else if (eqs(n, "<=")) l <= r else l >= r;
381 if (!ok) return .{ .boolean = false };
382 }
383 return .{ .boolean = true };
384 }
385 if (eqs(n, "not")) return .{ .boolean = !truth(a[0]) };
386 if (eqs(n, "eq?") or eqs(n, "equal?")) return .{ .boolean = equal(a[0], a[1], eqs(n, "equal?")) };
387 if (contains(&.{ "null?", "pair?", "list?", "number?", "symbol?", "procedure?" }, n)) return .{ .boolean = if (eqs(n, "null?")) is(a[0], .nil) else if (eqs(n, "pair?")) is(a[0], .pair) else if (eqs(n, "list?")) proper(a[0]) else if (eqs(n, "number?")) is(a[0], .integer) else if (eqs(n, "symbol?")) is(a[0], .symbol) else is(a[0], .procedure) };
388 if (eqs(n, "cons")) return m.pair(a[0], a[1]);
389 if (eqs(n, "list")) return m.list(a, .nil);
390 if (eqs(n, "car") or eqs(n, "cdr") or eqs(n, "cadr")) {
391 const p = (try m.expect(n, a[0], "pair")).pair;
392 return if (eqs(n, "car")) p.car else if (eqs(n, "cdr")) p.cdr else (try m.expect(n, p.cdr, "pair")).pair.car;
393 }
394 if (eqs(n, "length") or eqs(n, "reverse") or eqs(n, "list-ref")) {
395 const z = try m.asList(n, a[0]);
396 if (eqs(n, "length")) return .{ .integer = @intCast(z.len) };
397 if (eqs(n, "reverse")) {
398 var v: Value = .nil;
399 for (z) |x| v = try m.pair(x, v);
400 return v;
401 }
402 const k = (try m.expect(n, a[1], "integer")).integer;
403 if (k < 0 or k >= @as(i64, @intCast(z.len))) return m.fail("list-ref: index out of range: {d}", .{k});
404 return z[@intCast(k)];
405 }
406 if (eqs(n, "append")) {
407 if (a.len == 0) return .nil;
408 var z: Values = .empty;
409 for (a[0 .. a.len - 1]) |v| try z.appendSlice(m.gpa, try m.asList(n, v));
410 return m.list(z.items, a[a.len - 1]);
411 }
412 if (contains(&.{ "map", "filter", "fold-left", "apply" }, n)) {
413 const f = try m.expect(n, a[0], "procedure");
414 if (eqs(n, "apply")) {
415 const spread = try m.asList(n, a[a.len - 1]);
416 var z: Values = .empty;
417 try z.appendSlice(m.gpa, a[1 .. a.len - 1]);
418 try z.appendSlice(m.gpa, spread);
419 return m.call(f, z.items);
420 }
421 var ls: std.ArrayList([]const Value) = .empty;
422 var count: usize = std.math.maxInt(usize);
423 const first: usize = if (eqs(n, "fold-left")) 2 else 1;
424 for (a[first..]) |v| {
425 const z = try m.asList(n, v);
426 try ls.append(m.gpa, z);
427 count = @min(count, z.len);
428 }
429 var result: Values = .empty;
430 var acc = a[1];
431 for (0..count) |i| {
432 if (eqs(n, "filter")) {
433 if (truth(try m.call(f, &.{ls.items[0][i]}))) try result.append(m.gpa, ls.items[0][i]);
434 } else {
435 var args: Values = .empty;
436 if (eqs(n, "fold-left")) try args.append(m.gpa, acc);
437 for (ls.items) |l| try args.append(m.gpa, l[i]);
438 const v = try m.call(f, args.items);
439 if (eqs(n, "map")) try result.append(m.gpa, v) else acc = v;
440 }
441 }
442 return if (eqs(n, "fold-left")) acc else m.list(result.items, .nil);
443 }
444 if (eqs(n, "string-append")) {
445 var s: std.ArrayList(u8) = .empty;
446 for (a) |v| try s.appendSlice(m.gpa, (try m.expect(n, v, "string")).string);
447 return .{ .string = try s.toOwnedSlice(m.gpa) };
448 }
449 if (eqs(n, "number->string")) return .{ .string = try m.repr(try m.expect(n, a[0], "integer"), true) };
450 if (eqs(n, "display")) {
451 try m.write(try m.repr(a[0], false));
452 return .void_value;
453 }
454 if (eqs(n, "newline")) {
455 try m.write("\n");
456 return .void_value;
457 }
458 var s: std.ArrayList(u8) = .empty;
459 try s.appendSlice(m.gpa, (try m.expect(n, a[0], "string")).string);
460 for (a[1..]) |v| {
461 try s.append(m.gpa, ' ');
462 try s.appendSlice(m.gpa, try m.repr(v, true));
463 }
464 return m.fail("{s}", .{s.items});
465 }
466 fn global(m: *Machine) anyerror!*Environment {
467 const e = try m.env(null);
468 const names = [_][]const u8{ "+", "*", "-", "quotient", "remainder", "modulo", "abs", "min", "max", "gcd", "=", "<", ">", "<=", ">=", "not", "eq?", "equal?", "null?", "pair?", "list?", "number?", "symbol?", "procedure?", "cons", "car", "cdr", "cadr", "list", "length", "append", "reverse", "list-ref", "map", "filter", "fold-left", "apply", "string-append", "number->string", "display", "newline", "error" };
469 const minima = [_]usize{ 0, 0, 1, 2, 2, 2, 1, 1, 1, 0, 1, 1, 1, 1, 1, 1, 2, 2, 1, 1, 1, 1, 1, 1, 2, 1, 1, 1, 0, 1, 0, 1, 2, 2, 2, 3, 2, 0, 1, 1, 0, 1 };

482const Token = struct { kind: u8, value: Value = .nil };
483const Reader = struct {
484 m: *Machine,
485 lines: []const []const u8,
486 row: usize = 0,
487 col: usize = 0,
488 fn skip(r: *Reader) void {
489 r.row += 1;
490 r.col = 0;
491 }
492 fn end(r: *Reader) bool {
493 while (r.row < r.lines.len) {
494 const s = r.lines[r.row];
495 while (r.col < s.len and s[r.col] == ' ') r.col += 1;
496 if (r.col < s.len and s[r.col] != ';') return false;
497 r.skip();
498 }
499 return true;
500 }
501 fn token(r: *Reader) anyerror!Token {
502 if (r.end()) return r.m.fail("unexpected end of input", .{});
503 const s = r.lines[r.row];
504 var c = s[r.col];
505 r.col += 1;
506 if (std.mem.indexOfScalar(u8, "()'", c) != null) return .{ .kind = c };
507 if (c == '"') {
508 var v: std.ArrayList(u8) = .empty;
509 while (r.col < s.len) {
510 c = s[r.col];
511 r.col += 1;
512 if (c == '"') return .{ .kind = 'v', .value = .{ .string = try v.toOwnedSlice(r.m.gpa) } };
513 if (c == '\\') {
514 if (r.col == s.len) break;
515 c = s[r.col];
516 r.col += 1;
517 if (c != '"' and c != '\\' and c != 'n') return r.m.fail("bad escape", .{});
518 if (c == 'n') c = '\n';
519 }
520 try v.append(r.m.gpa, c);
521 }
522 return r.m.fail("unterminated string", .{});
523 }
524 const start = r.col - 1;
525 while (r.col < s.len and std.mem.indexOfScalar(u8, " ()'\";", s[r.col]) == null) r.col += 1;
526 const a = s[start..r.col];
527 if (eqs(a, ".")) return .{ .kind = '.' };
528 const first: usize = if (a[0] == '+' or a[0] == '-') 1 else 0;
529 var integer = first < a.len;
530 for (a[first..]) |d| integer = integer and std.ascii.isDigit(d);
531 if (integer) return .{ .kind = 'v', .value = .{ .integer = try std.fmt.parseInt(i64, a, 10) } };
532 if (eqs(a, "#t") or eqs(a, "#f")) return .{ .kind = 'v', .value = .{ .boolean = eqs(a, "#t") } };
533 if (a[0] == '#') return r.m.fail("bad token: {s}", .{a});
534 return .{ .kind = 'v', .value = .{ .symbol = a } };
535 }
536 fn datum(r: *Reader) anyerror!Value {
537 return r.from(try r.token());
538 }
539 fn from(r: *Reader, t: Token) anyerror!Value {
540 if (t.kind == 'v') return t.value;
541 if (t.kind == '\'') return r.m.list(&.{ .{ .symbol = "quote" }, try r.datum() }, .nil);
542 if (t.kind != '(') return r.m.fail("unexpected {c}", .{t.kind});
543 var a: Values = .empty;
544 while (true) {
545 const next = try r.token();
546 if (next.kind == ')') return r.m.list(a.items, .nil);
547 if (next.kind == '.' and a.items.len > 0) {
548 const tail = try r.datum();
549 if ((try r.token()).kind != ')') return r.m.fail("expected )", .{});
550 return r.m.list(a.items, tail);
551 }
552 try a.append(r.m.gpa, try r.from(next));
553 }
554 }
555};
556pub fn main(init: std.process.Init) !void {
557 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
558 defer arena.deinit();
559 const gpa = arena.allocator();
560 var input_buffer: [8192]u8 = undefined;
561 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
562 const text = try input.interface.allocRemaining(gpa, .unlimited);
563 var split = std.mem.splitScalar(u8, text, '\n');
564 var lines: std.ArrayList([]const u8) = .empty;
565 while (split.next()) |s| {
566 if (eqs(s, "(exit)")) break;
567 try lines.append(gpa, s);
568 }
569 var m: Machine = .{ .gpa = gpa };
570 const env = try m.global();
571 var r: Reader = .{ .m = &m, .lines = lines.items };
572 while (!r.end()) {
573 const x = r.datum() catch |err| {
574 if (err != error.Failure) return err;
575 try m.line(try std.fmt.allocPrint(gpa, "Error: read: {s}", .{m.why}));
576 if (eqs(m.why, "unexpected end of input")) break;
577 r.skip();
578 continue;
579 };
580 const v = m.evaluate(x, env) catch |err| {
581 if (err != error.Failure) return err;
582 try m.line(try std.fmt.allocPrint(gpa, "Error: {s}", .{m.why}));
583 continue;
584 };
585 if (!is(v, .void_value)) try m.line(try m.repr(v, true));
586 }
587 if (m.mid) try m.write("\n");
588 var output_buffer: [8192]u8 = undefined;
589 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
590 try output.interface.writeAll(m.output.items);
591 try output.interface.flush();
592}
