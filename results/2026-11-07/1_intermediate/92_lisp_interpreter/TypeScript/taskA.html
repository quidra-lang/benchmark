113 );

164function evaluate(x: Value, e: Env): Value {
165 if (x instanceof Atom) return e.get(x.name);
166 if (!(x instanceof Pair)) {
167 shape(x !== NIL, x);
168 return x;
169 }
170 const a = items(x);
171 shape(a !== null, x);
172 const n = a.length,
173 h = a[0] instanceof Atom ? a[0].name : "";
174 switch (h) {
175 case "quote":
176 shape(n === 2, x);
177 return a[1];
178 case "if":
179 shape(n === 3 || n === 4, x);
180 return evaluate(a[1], e) !== false
181 ? evaluate(a[2], e)
182 : n === 4
183 ? evaluate(a[3], e)
184 : VOID;
185 case "begin":
186 return body(a.slice(1), e);
187 case "lambda": {
188 const p = n >= 3 ? parameters(a[1]) : null;
189 shape(p !== null, x);
190 return closure(p, a.slice(2), e);
191 }
192 case "define": {
193 if (n >= 2 && a[1] instanceof Atom) {
194 shape(n === 3, x);
195 const v = evaluate(a[2], e);
196 if (v instanceof Proc && !v.fn && !v.name) v.name = a[1].name;
197 e.frame.set(a[1].name, v);
198 } else {
199 const target = a[1],
200 p =
201 n >= 3 && target instanceof Pair && target.car instanceof Atom
202 ? parameters(target.cdr)
203 : null;
204 shape(p !== null, x);
205 const pair = target as Pair;
206 const name = (pair.car as Atom).name;
207 e.frame.set(name, closure(p, a.slice(2), e, name));
208 }
209 return VOID;
210 }
211 case "set!":
212 shape(n === 3 && a[1] instanceof Atom, x);
213 {
214 const name = (a[1] as Atom).name;
215 const v = evaluate(a[2], e);
216 e.find(name).frame.set(name, v);
217 }
218 return VOID;
219 case "let":
220 case "let*":
221 case "letrec": {
222 const named = h === "let" && n > 1 && a[1] instanceof Atom,
223 at = named ? 2 : 1;
224 const bs = n >= at + 2 ? bindings(a[at], h !== "let*") : null;
225 shape(bs !== null, x);
226 let scope = new Env(e);
227 if (h === "let*") {
228 scope = bs.length ? e : scope;
229 for (const [name, expr] of bs) {
230 const v = evaluate(expr, scope);
231 scope = new Env(scope);
232 scope.frame.set(name, v);
233 }
234 } else if (h === "letrec") {
235 for (const [name] of bs) scope.frame.set(name, UNINIT);
236 for (const [name, expr] of bs) scope.frame.set(name, evaluate(expr, scope));
237 } else {
238 const vs = bs.map((b) => evaluate(b[1], e));
239 if (named) {
240 const p = closure(
241 {
242 names: bs.map((b) => b[0]),
243 rest: null,
244 },
245 a.slice(3),
246 scope,
247 (a[1] as Atom).name,
248 );
249 scope.frame.set((a[1] as Atom).name, p);
250 return p.call(vs);
251 }
252 bs.forEach((b, i) => scope.frame.set(b[0], vs[i]));
253 }
254 return body(a.slice(at + 1), scope);
255 }
256 case "cond": {
257 const cs = a.slice(1).map((z) => {
258 const c = items(z);
259 shape(c && c.length, x);
260 return c;
261 });
262 cs.forEach((c, i) =>
263 shape(
264 !(c[0] instanceof Atom) ||
265 c[0].name !== "else" ||
266 (c.length >= 2 && i === cs.length - 1),
267 x,
268 ),
269 );
270 for (const c of cs) {
271 if (c[0] instanceof Atom && c[0].name === "else") return body(c.slice(1), e);
272 const v = evaluate(c[0], e);
273 if (v !== false) return c.length === 1 ? v : body(c.slice(1), e);
274 }
275 return VOID;
276 }
277 case "and": {
278 let v: Value = true;
279 for (const z of a.slice(1)) {
280 v = evaluate(z, e);
281 if (v === false) break;
282 }
283 return v;
284 }
285 case "or":
286 for (const z of a.slice(1)) {
287 const v = evaluate(z, e);
288 if (v !== false) return v;
289 }
290 return false;
291 case "when":
292 case "unless":
293 shape(n >= 3, x);
294 return (evaluate(a[1], e) !== false) === (h === "when") ? body(a.slice(2), e) : VOID;
295 }
296 const p = evaluate(a[0], e),
297 args = a.slice(1).map((z) => evaluate(z, e));
298 if (!(p instanceof Proc)) throw Error("not a procedure: " + repr(p));
299 return p.call(args);

500class Reader {
501 lines: string[];
502 row: number;
503 col: number;
504 constructor(lines: string[]) {
505 this.lines = lines;
506 this.row = 0;

509 skip() {
510 this.row++;

512 }
513 end() {
514 while (this.row < this.lines.length) {
515 const s = this.lines[this.row];
516 while (s[this.col] === " ") this.col++;

522 token(): [string, Value | null] {
523 if (this.end()) throw Error("unexpected end of input");
524 const s = this.lines[this.row],
525 c = s[this.col++];
526 if ("()'".includes(c)) return [c, null];
527 if (c === '"') {

530 let c = s[this.col++];
531 if (c === '"') return ["v", v];
532 if (c === "\\") {
533 if (this.col === s.length) break;
534 c = s[this.col++];
535 if (!'"\\n'.includes(c)) throw Error("bad escape");
536 v += c === "n" ? "\n" : c;
537 } else v += c;

546 if (a === "#t" || a === "#f") return ["v", a === "#t"];

550 datum(t = this.token()): Value {
551 if (t[0] === "v") return t[1]!;
552 if (t[0] === "'") return list([sym("quote"), this.datum()]);
553 if (t[0] !== "(") throw Error("unexpected " + t[0]);
