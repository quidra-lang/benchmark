local function env(parent)
    return {parent=parent, frame={}}
local function find(e, n)
    while e do
        if e.frame[n]~=nil then
            return e
        e=e.parent
local function lookup(e, n)
    local v=find(e, n).frame[n]
    if v==UNINIT then
        fail('uninitialized variable: '..n)
    return v
local function body(a, e)
    local v=VOID
    for _, x in ipairs(a) do
        v=evaluate(x, e)
    return v
local function closure(p, b, e, name)
    return {tag='procedure', name=name, min=#p.names, rest=p.rest, params=p.names, body=b, env=e}
call=function(p, a)
    local e=env(p.env)
    return body(p.body, e)
evaluate=function(x, e)
    if kind(x, 'symbol') then
        return lookup(e, x.name)
    elseif h=='lambda' then
        return closure(p, slice(a, 3), e)
    elseif h=='set!' then
        local v=evaluate(a[3], e)
        find(e, a[2].name).frame[a[2].name]=v
    elseif h=='let' or h=='let*' or h=='letrec' then
        local named=h=='let' and kind(a[2], 'symbol')
        local at=named and 3 or 2
        local bs=n>=at+1 and bindings(a[at], h~='let*')
        local scope=env(e)
        if h=='let*' then
            if #bs>0 then
                scope=e
            for _, b in ipairs(bs) do
                local v=evaluate(b[2], scope)
                scope=env(scope)
                scope.frame[b[1]]=v
        elseif h=='letrec' then
            for _, b in ipairs(bs) do
                scope.frame[b[1]]=UNINIT
            for _, b in ipairs(bs) do
                scope.frame[b[1]]=evaluate(b[2], scope)
        else
            for _, b in ipairs(bs) do
                vs[#vs+1]=evaluate(b[2], e)
            for i, b in ipairs(bs) do
                scope.frame[b[1]]=vs[i]
        return body(slice(a, at+1), scope)
    local f=evaluate(a[1], e)
    local args={}
    for i=2, n do
        args[#args+1]=evaluate(a[i], e)
    return call(f, args)
