169evaluate=function(x, e)

183 elseif h=='if' then

194 elseif h=='lambda' then

198 elseif h=='define' then
199 if kind(a[2], 'symbol') then

201 local v=evaluate(a[3], e)
202 if kind(v, 'procedure') and not v.fn and not v.name then
203 v.name=a[2].name
204 end
205 e.frame[a[2].name]=v
206 else

208 local p=n>=3 and kind(t, 'pair') and kind(t.car, 'symbol') and parameters(t.cdr)
209 shape(p, x)

211 end
212 return VOID
213 elseif h=='set!' then
214 shape(n==3 and kind(a[2], 'symbol'), x)
215 local v=evaluate(a[3], e)

217 return VOID
218 elseif h=='let' or h=='let*' or h=='letrec' then
219 local named=h=='let' and kind(a[2], 'symbol')
220 local at=named and 3 or 2
221 local bs=n>=at+1 and bindings(a[at], h~='let*')
222 shape(bs, x)
223 local scope=env(e)
224 if h=='let*' then
225 if #bs>0 then
226 scope=e
227 end
228 for _, b in ipairs(bs) do
229 local v=evaluate(b[2], scope)
230 scope=env(scope)
231 scope.frame[b[1]]=v
232 end
233 elseif h=='letrec' then
234 for _, b in ipairs(bs) do
235 scope.frame[b[1]]=UNINIT
236 end
237 for _, b in ipairs(bs) do
238 scope.frame[b[1]]=evaluate(b[2], scope)
239 end
240 else
241 local vs={}
242 for _, b in ipairs(bs) do
243 vs[#vs+1]=evaluate(b[2], e)
244 end
245 if named then
246 local names={}
247 for _, b in ipairs(bs) do
248 names[#names+1]=b[1]
249 end
250 local p=closure({names=names}, slice(a, 4), scope, a[2].name)
251 scope.frame[a[2].name]=p
252 return call(p, vs)
253 end
254 for i, b in ipairs(bs) do
255 scope.frame[b[1]]=vs[i]
256 end
257 end
258 return body(slice(a, at+1), scope)
259 elseif h=='cond' then
260 local cs={}
261 for i=2, n do
262 local c=items(a[i])
263 shape(c and #c>0, x)
264 if c[1]==symbol('else') then
265 shape(#c>=2 and i==n, x)
266 end
267 cs[#cs+1]=c
268 end
269 for _, c in ipairs(cs) do
270 if c[1]==symbol('else') then
271 return body(slice(c, 2), e)
272 end
273 local v=evaluate(c[1], e)
274 if v~=false then
275 if #c==1 then
276 return v
277 else
278 return body(slice(c, 2), e)
279 end
280 end
281 end
282 return VOID
283 elseif h=='and' then
284 local v=true
285 for i=2, n do
286 v=evaluate(a[i], e)
287 if v==false then
288 break
289 end
290 end
291 return v
292 elseif h=='or' then
293 for i=2, n do
294 local v=evaluate(a[i], e)
295 if v~=false then
296 return v
297 end
298 end
299 return false
300 elseif h=='when' or h=='unless' then
301 shape(n>=3, x)
302 if (evaluate(a[2], e)~=false)==(h=='when') then
303 return body(slice(a, 3), e)
304 end
305 return VOID
306 end

310 args[#args+1]=evaluate(a[i], e)
311 end
312 if not kind(f, 'procedure') then
313 fail('not a procedure: '..repr(f))
314 end
315 return call(f, args)
316end
