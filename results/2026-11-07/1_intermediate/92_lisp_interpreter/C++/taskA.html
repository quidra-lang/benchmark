71string repr(V v, bool quoted = true){
72 switch (v->kind){
73 case Value::Integer:
74 return to_string(v->number);
75 case Value::Boolean:
76 return v->number?"#t": "#f";
77 case Value::String:
78 {
79 if (!quoted)return v->text;
80 string s = "\"";
81 for (char c: v->text)s += c == '\n'?"\\n": c == '\\'?"\\\\": c == '\"'?"\\\"": string(1, c);
82 return s+'"';
83 }
84 case Value::Symbol:
85 return v->text;
86 case Value::Nil:
87 return "()";
88 case Value::Procedure:
89 return v->text.empty()?"#<procedure>": "#<procedure "+v->text+">";
90 case Value::Pair:
91 {
92 string s = "(";
93 bool first = true;
94 while (v->kind == Value::Pair){
95 if (!first)s += ' ';
96 first = false;
97 s += repr(v->car, quoted);
98 v = v->cdr;
99 }
100 if (v != nil)s += " . "+repr(v, quoted);
101 return s+")";
102 }
103 default: return "#<void>";
104 }

112bool eq(V a, V b, bool deep = false){
113 if (a->kind != b->kind)return false;
114 if (a->kind == Value::Pair)return deep?eq(a->car, b->car, true) && eq(a->cdr, b->cdr, true): a == b;
115 if (a->kind == Value::Procedure)return a == b;
116 if (a->kind == Value::Integer || a->kind == Value::Boolean)return a->number == b->number;
117 if (a->kind == Value::String || a->kind == Value::Symbol)return a->text == b->text;
118 return a == b;
119}

149bool parameters(V v, vector<string>&p, string&r){
150 unordered_set<string>seen;
151 while (v->kind == Value::Pair){
152 if (v->car->kind != Value::Symbol || !seen.insert(v->car->text).second)return false;
153 p.push_back(v->car->text);
154 v = v->cdr;
155 }
156 if (v == nil)return true;
157 if (v->kind != Value::Symbol || !seen.insert(v->text).second)return false;
158 r = v->text;
159 return true;
160}
161V closure(const vector<string>&p, const string&r, const vector<V>&b, E e, const string&name = ""){
162 V v = text(Value::Procedure, name);
163 v->params = p;
164 v->rest = r;
165 v->variadic = !r.empty();
166 v->minimum = p.size();
167 v->body = b;
168 v->env = e;
169 return v;
170}
171bool bindings(V v, vector<pair<string, V>>&bs, bool unique){
172 vector<V>a;
173 if (!unpack(v, a))return false;
174 unordered_set<string>seen;
175 for (V x: a){
176 vector<V>b;
177 if (!unpack(x, b) || b.size() != 2 || b[0]->kind != Value::Symbol)return false;
178 string n = b[0]->text;
179 if (unique && !seen.insert(n).second)return false;
180 bs.emplace_back(n, b[1]);
181 }
182 return true;
183}
184V evaluate(V x, E e){
185 if (x->kind == Value::Symbol)return e->get(x->text);
186 if (x->kind != Value::Pair){
187 shape(x != nil, x);
188 return x;
189 }
190 vector<V>a;
191 shape(unpack(x, a), x);
192 size_t n = a.size();
193 string h = a[0]->kind == Value::Symbol?a[0]->text: "";
194 if (h == "quote"){
195 shape(n == 2, x);
196 return a[1];
197 }
198 if (h == "if"){
199 shape(n == 3 || n == 4, x);
200 return truth(evaluate(a[1], e))?evaluate(a[2], e): n == 4?evaluate(a[3], e): nothing;
201 }
202 if (h == "begin")return body(slice(a, 1), e);
203 if (h == "lambda"){
204 vector<string>p;
205 string r;
206 shape(n >= 3 && parameters(a[1], p, r), x);
207 return closure(p, r, slice(a, 2), e);
208 }
209 if (h == "define"){
210 if (n>1 && a[1]->kind == Value::Symbol){
211 shape(n == 3, x);
212 V v = evaluate(a[2], e);
213 if (v->kind == Value::Procedure && !v->builtin && v->text.empty())v->text = a[1]->text;
214 e->frame[a[1]->text] = v;
215 }
216 else {
217 vector<string>p;
218 string r;
219 shape(n >= 3 && a[1]->kind == Value::Pair && a[1]->car->kind == Value::Symbol && parameters(a[1]->cdr, p, r), x);
220 e->frame[a[1]->car->text] = closure(p, r, slice(a, 2), e, a[1]->car->text);
221 }
222 return nothing;
223 }
224 if (h == "set!"){
225 shape(n == 3 && a[1]->kind == Value::Symbol, x);
226 V v = evaluate(a[2], e);
227 e->locate(a[1]->text) = v;
228 return nothing;
229 }
230 if (h == "let" || h == "let*" || h == "letrec"){
231 bool named = h == "let" && n>1 && a[1]->kind == Value::Symbol;
232 size_t at = named?2: 1;
233 vector<pair<string, V>>bs;
234 shape(n >= at+2 && bindings(a[at], bs, h != "let*"), x);
235 E scope = make_shared<Environment>(e);
236 if (h == "let*"){
237 if (!bs.empty())scope = e;
238 for (const auto&b: bs){
239 V v = evaluate(b.second, scope);
240 scope = make_shared<Environment>(scope);
241 scope->frame[b.first] = v;
242 }
243 }
244 else if (h == "letrec"){
245 for (const auto&b: bs)scope->frame[b.first] = uninitialized;
246 for (const auto&b: bs)scope->frame[b.first] = evaluate(b.second, scope);
247 }
248 else {
249 vector<V>vs;
250 vector<string>p;
251 for (const auto&b: bs){
252 vs.push_back(evaluate(b.second, e));
253 p.push_back(b.first);
254 }
255 if (named){
256 V f = closure(p, "", slice(a, 3), scope, a[1]->text);
257 scope->frame[a[1]->text] = f;
258 return call(f, vs);
259 }
260 for (size_t i = 0; i<bs.size(); i++)scope->frame[bs[i].first] = vs[i];
261 }
262 return body(slice(a, at+1), scope);
263 }
264 if (h == "cond"){
265 vector<vector<V>>cs;
266 for (size_t i = 1; i<n; i++){
267 vector<V>c;
268 shape(unpack(a[i], c) && !c.empty(), x);
269 shape(!(c[0]->kind == Value::Symbol && c[0]->text == "else") || (c.size() >= 2 && i+1 == n), x);
270 cs.push_back(c);
271 }
272 for (const auto&c: cs){
273 if (c[0]->kind == Value::Symbol && c[0]->text == "else")return body(slice(c, 1), e);
274 V v = evaluate(c[0], e);
275 if (truth(v))return c.size() == 1?v: body(slice(c, 1), e);
276 }
277 return nothing;
278 }
279 if (h == "and"){
280 V v = boolean(true);
281 for (size_t i = 1; i<n; i++){
282 v = evaluate(a[i], e);
283 if (!truth(v))break;
284 }
285 return v;
286 }
287 if (h == "or"){
288 for (size_t i = 1; i<n; i++){
289 V v = evaluate(a[i], e);
290 if (truth(v))return v;
291 }
292 return boolean(false);
293 }
294 if (h == "when" || h == "unless"){
295 shape(n >= 3, x);
296 return truth(evaluate(a[1], e)) == (h == "when")?body(slice(a, 2), e): nothing;
297 }
298 V f = evaluate(a[0], e);
299 vector<V>args;
300 for (size_t i = 1; i<n; i++)args.push_back(evaluate(a[i], e));
301 if (f->kind != Value::Procedure)fail("not a procedure: "+repr(f));
302 return call(f, args);
303}
304V call(V f, const vector<V>&a){
305 if (a.size()<size_t(f->minimum) || (!f->variadic && a.size() != size_t(f->minimum)))fail((f->text.empty()?"lambda": f->text)+": wrong number of arguments (expected "+(f->variadic?"at least ": "")+to_string(f->minimum)+", got "+to_string(a.size())+")");
306 if (f->builtin)return f->builtin(a);
307 E e = make_shared<Environment>(f->env);
308 for (size_t i = 0; i<f->params.size(); i++)e->frame[f->params[i]] = a[i];
309 if (f->variadic)e->frame[f->rest] = list(slice(a, f->params.size()));
310 return body(f->body, e);
311}

335 auto add = [&](string n, int m, bool r, function<V(const vector<V>&)>fn){
336 V p = text(Value::Procedure, n);
337 p->minimum = m;
338 p->variadic = r;
339 p->builtin = fn;
340 e->frame[n] = p;

342 for (string n: {
343 "+", "*", "-", "quotient", "remainder", "modulo", "abs", "min", "max", "gcd", "=", "<", ">", "<=", ">="
344 }){
345 int fixed = n == "quotient" || n == "remainder" || n == "modulo"?2: n == "abs"?1: -1;
346 add(n, fixed >= 0?fixed: n == "+" || n == "*" || n == "gcd"?0: 1, fixed<0, [n](const vector<V>&a){
347 for (V v: a)expect(n, v, "integer");
348 long long v = a.empty()?0: a[0]->number;
349 if (n == "+" || n == "*"){
350 v = n == "+"?0: 1;
351 for (V x: a)v = n == "+"?v+x->number: v*x->number;
352 return num(v);
353 }
354 if (n == "-"){
355 if (a.size() == 1)return num(-v);
356 for (size_t i = 1; i<a.size(); i++)v -= a[i]->number;
357 return num(v);
358 }
359 if (n == "abs")return num(v<0?-v: v);
360 if (n == "min" || n == "max"){
361 for (V x: a)v = n == "min"?min(v, x->number): max(v, x->number);
362 return num(v);
363 }
364 if (n == "gcd"){
365 v = 0;
366 for (V x: a)v = gcd(v, x->number);
367 return num(v);
368 }
369 if (n == "quotient" || n == "remainder" || n == "modulo"){
370 long long b = a[1]->number;
371 if (!b)fail(n+": division by zero");
372 long long q = v/b, r = v%b;
373 if (n == "quotient")return num(q);
374 if (n == "modulo" && r && ((r<0) != (b<0)))r += b;
375 return num(r);
376 }
377 bool ok = true;
378 for (size_t i = 1; i<a.size(); i++){
379 long long l = a[i-1]->number, r = a[i]->number;
380 ok = ok && (n == "="?l == r: n == "<"?l<r: n == ">"?l>r: n == "<="?l <= r: l >= r);
381 }
382 return boolean(ok);
383 });
384 }

399 });

411 });

424 });

426 if (a.empty())return nil;
427 vector<V>z;

429 vector<V>b = aslist("append", a[i]);
430 z.insert(z.end(), b.begin(), b.end());
431 }
432 return list(z, a.back());
433 });
434 for (string n: {
435 "map", "filter", "fold-left", "apply"
436 })add(n, n == "fold-left"?3: 2, n != "filter", [n](const vector<V>&a){
437 V f = expect(n, a[0], "procedure");
438 if (n == "apply"){
439 vector<V>z(a.begin()+1, a.end()-1), b = aslist(n, a.back());
440 z.insert(z.end(), b.begin(), b.end());
441 return call(f, z);
442 }
443 vector<vector<V>>ls;
444 size_t count = 1000000;
445 for (size_t i = n == "fold-left"?2: 1; i<a.size(); i++){
446 ls.push_back(aslist(n, a[i]));
447 count = min(count, ls.back().size());
448 }
449 vector<V>out;
450 V acc = a[1];
451 for (size_t i = 0; i<count; i++){
452 if (n == "filter"){

463 }
464 }

466 });

491struct Reader {
492 vector<string>lines;
493 size_t row = 0, col = 0;
494 void skip(){
495 row++;
496 col = 0;
497 }
498 bool end(){
499 while (row<lines.size()){
500 while (col<lines[row].size() && lines[row][col] == ' ')col++;
501 if (col<lines[row].size() && lines[row][col] != ';')return false;
502 skip();
503 }
504 return true;
505 }
506 pair<char, V>token(){
507 if (end())fail("unexpected end of input");
508 string&s = lines[row];
509 char c = s[col++];
510 if (c == '(' || c == ')' || c == '\'')return {
511 c, nullptr
512 };
513 if (c == '"'){
514 string v;
515 while (col<s.size()){
516 c = s[col++];
517 if (c == '"')return {
518 'v', text(Value::String, v)
519 };
520 if (c == '\\'){
521 if (col == s.size())break;
522 c = s[col++];
523 if (c != '"' && c != '\\' && c != 'n')fail("bad escape");
524 if (c == 'n')c = '\n';
525 }
526 v += c;
527 }
528 fail("unterminated string");
529 }
530 size_t start = col-1;
531 while (col<s.size() && string(" ()'\";").find(s[col]) == string::npos)col++;
532 string a = s.substr(start, col-start);
533 if (a == ".")return {
534 '.', nullptr
535 };
536 size_t i = (a[0] == '+' || a[0] == '-')?1: 0;
537 bool integer = i<a.size();
538 for (; i<a.size(); i++)integer = integer && a[i] >= '0' && a[i] <= '9';
539 if (integer)return {
540 'v', num(stoll(a))
541 };
542 if (a == "#t" || a == "#f")return {
543 'v', boolean(a == "#t")
544 };
545 if (a[0] == '#')fail("bad token: "+a);
546 return {
547 'v', text(Value::Symbol, a)
548 };
549 }
550 V datum(){
551 return from(token());
552 }
553 V from(pair<char, V>t){
554 if (t.first == 'v')return t.second;
555 if (t.first == '\'')return list({
556 text(Value::Symbol, "quote"), datum()
557 });
558 if (t.first != '(')fail(string("unexpected ")+t.first);
559 vector<V>a;
560 while (true){
561 t = token();
562 if (t.first == ')')return list(a);
563 if (t.first == '.' && !a.empty()){
564 V tail = datum();
565 if (token().first != ')')fail("expected )");
566 return list(a, tail);
567 }
568 a.push_back(from(t));
569 }
570 }
571};
