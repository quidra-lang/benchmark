131Object body(List<Object>a, Env e){

136Object evaluate(Object x, Env e){
137 if (x is SymbolValue)return e.get(x.name);
138 if (x is! Pair){
139 shape(!identical(x, nil), x);
140 return x;
141 }
142 final parsed = items(x);
143 shape(parsed != null, x);
144 final a = parsed!;
145 final n = a.length, h = a[0] is SymbolValue?(a[0] as SymbolValue).name: '';
146 switch (h){
147 case 'quote':
148 shape(n == 2, x);
149 return a[1];
150 case 'if':
151 shape(n == 3 || n == 4, x);
152 return truth(evaluate(a[1], e))?evaluate(a[2], e): n == 4?evaluate(a[3], e): voidValue;
153 case 'begin':
154 return body(a.sublist(1), e);
155 case 'lambda':
156 final p = n >= 3?parameters(a[1]): null;
157 shape(p != null, x);
158 return closure(p!, a.sublist(2), e);
159 case 'define':
160 if (n>1 && a[1] is SymbolValue){
161 shape(n == 3, x);
162 final name = (a[1] as SymbolValue).name;
163 final v = evaluate(a[2], e);
164 if (v is Procedure && v.builtin == null && v.name == null)v.name = name;
165 e.frame[name] = v;
166 }
167 else{
168 final t = n>1?a[1]: nil;
169 final p = n >= 3 && t is Pair && t.car is SymbolValue?parameters(t.cdr): null;
170 shape(p != null, x);
171 final name = ((t as Pair).car as SymbolValue).name;
172 e.frame[name] = closure(p!, a.sublist(2), e, name);
173 }
174 return voidValue;
175 case 'set!':
176 shape(n == 3 && a[1] is SymbolValue, x);
177 final name = (a[1] as SymbolValue).name;
178 final v = evaluate(a[2], e);
179 e.find(name).frame[name] = v;
180 return voidValue;
181 case 'let':
182 case 'let*':
183 case 'letrec':
184 final named = h == 'let' && n>1 && a[1] is SymbolValue, at = named?2: 1;
185 final bs = n >= at+2?bindings(a[at], h != 'let*'): null;
186 shape(bs != null, x);
187 var scope = Env(e);
188 if (h == 'let*'){
189 if (bs!.isNotEmpty)scope = e;
190 for (final b in bs){
191 final v = evaluate(b.expr, scope);
192 scope = Env(scope);
193 scope.frame[b.name] = v;
194 }
195 }
196 else if (h == 'letrec'){
197 for (final b in bs!)scope.frame[b.name] = uninitialized;
198 for (final b in bs)scope.frame[b.name] = evaluate(b.expr, scope);
199 }
200 else{
201 final vs = bs!.map((b)=>evaluate(b.expr, e)).toList();
202 if (named){
203 final name = (a[1] as SymbolValue).name;
204 final f = closure(Parameters(bs.map((b)=>b.name).toList(), null), a.sublist(3), scope, name);
205 scope.frame[name] = f;
206 return f.call(vs);
207 }
208 for (var i = 0; i<bs.length; i++)scope.frame[bs[i].name] = vs[i];
209 }
210 return body(a.sublist(at+1), scope);
211 case 'cond':
212 final cs = a.sublist(1).map(items).toList();
213 for (var i = 0; i<cs.length; i++){
214 final c = cs[i];
215 shape(c != null && c.isNotEmpty, x);
216 if (identical(c![0], sym('else')))shape(c.length >= 2 && i == cs.length-1, x);
217 }
218 for (final c0 in cs){
219 final c = c0!;
220 if (identical(c[0], sym('else')))return body(c.sublist(1), e);
221 final v = evaluate(c[0], e);
222 if (truth(v))return c.length == 1?v: body(c.sublist(1), e);
223 }
224 return voidValue;
225 case 'and':
226 Object v = true;
227 for (final z in a.sublist(1)){
228 v = evaluate(z, e);
229 if (!truth(v))break;
230 }
231 return v;
232 case 'or':
233 for (final z in a.sublist(1)){
234 final v = evaluate(z, e);
235 if (truth(v))return v;
236 }
237 return false;
238 case 'when':
239 case 'unless':
240 shape(n >= 3, x);
241 return truth(evaluate(a[1], e)) == (h == 'when')?body(a.sublist(2), e): voidValue;
242 }
243 final f = evaluate(a[0], e), args = a.sublist(1).map((z)=>evaluate(z, e)).toList();
244 if (f is! Procedure)throw LispError('not a procedure: ${repr(f)}');
245 return f.call(args);
246}

273 final fixed = ['quotient', 'remainder', 'modulo'].contains(name)?2: name == 'abs'?1: -1;
274 add(name, fixed >= 0?fixed: ['+', '*', 'gcd'].contains(name)?0: 1, fixed<0, (a){
275 final z = a.map((v)=>expect(name, v, 'integer') as int).toList();

278 case '+':

280 case '*':

283 return z.length == 1?-v: v-z.sublist(1).fold(0, (s, x)=>s+x);

301 final l = z[i-1], r = z[i];

339 final ls = a.sublist(n == 'fold-left'?2: 1).map((v)=>asList(n, v)).toList();
340 final count = ls.map((z)=>z.length).reduce(math.min), out = <Object>[];

368class Token {
369 final String kind;
370 final Object? value;
371 Token(this.kind, [this.value]);
372}
373class Reader {
374 final List<String> lines;
375 int row = 0, col = 0;
376 Reader(this.lines);
377 void skip(){
378 row++;
379 col = 0;
380 }
381 bool end(){
382 while (row<lines.length){
383 final s = lines[row];
384 while (col<s.length && s[col] == ' ')col++;
385 if (col<s.length && s[col] != ';')return false;
386 skip();
387 }
388 return true;
389 }
390 Token token(){
391 if (end())throw LispError('unexpected end of input');
392 final s = lines[row];
393 var c = s[col++];
394 if ("()'".contains(c))return Token(c);
395 if (c == '"'){
396 var v = '';
397 while (col<s.length){
398 c = s[col++];
399 if (c == '"')return Token('v', v);
400 if (c == '\\'){
401 if (col == s.length)break;
402 c = s[col++];
403 if (!['"', '\\', 'n'].contains(c))throw LispError('bad escape');
404 if (c == 'n')c = '\n';
405 }
406 v += c;
407 }
408 throw LispError('unterminated string');
409 }
410 final start = col-1;
411 while (col<s.length && !" ()'\";".contains(s[col]))col++;
412 final a = s.substring(start, col);
413 if (a == '.')return Token('.');
414 if (RegExp(r'^[+-]?[0-9]+$').hasMatch(a))return Token('v', int.parse(a));
415 if (a == '#t' || a == '#f')return Token('v', a == '#t');
416 if (a.startsWith('#'))throw LispError('bad token: $a');
417 return Token('v', sym(a));
418 }
419 Object datum([Token? first]){
420 final t = first??token();
421 if (t.kind == 'v')return t.value!;
422 if (t.kind == "'")return makeList([sym('quote'), datum()]);
423 if (t.kind != '(')throw LispError('unexpected ${t.kind}');
424 final a = <Object>[];
425 while (true){
426 final t = token();
427 if (t.kind == ')')return makeList(a);
428 if (t.kind == '.' && a.isNotEmpty){
429 final tail = datum();
430 if (token().kind != ')')throw LispError('expected )');
431 return makeList(a, tail);
432 }
433 a.add(datum(t));
434 }
435 }
436}
