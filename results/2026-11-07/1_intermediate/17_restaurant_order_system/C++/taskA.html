34long long cents(const string& text) {
35 size_t point = text.find('.');
36 return stoll(text.substr(0, point)) * 100 + stoll(text.substr(point + 1));
37}

39string money(long long amount) {
40 ostringstream out;
41 out << amount / 100 << '.' << setw(2) << setfill('0') << amount % 100;
42 return out.str();
43}

45string clock_text(int elapsed) {
46 int minutes = 660 + elapsed;
47 ostringstream out;
48 out << setw(2) << setfill('0') << minutes / 60 << ':'
49 << setw(2) << minutes % 60;
50 return out.str();
51}

104struct Check {
105 string party;
106 int size;
107 Table* table;
108 Server* server;
109 int since;
110 vector<OrderLine*> lines;
111 long long paid = 0;
112 int discount = 0;
113 bool closed = false;
114
115 vector<OrderLine*> active() const {
116 vector<OrderLine*> result;
117 for (OrderLine* line : lines)
118 if (line->state != "VOID") result.push_back(line);
119 return result;
120 }
121
122 array<long long, 6> amounts() const {
123 long long subtotal = 0;
124 for (OrderLine* line : lines)
125 if (line->state != "VOID") subtotal += line->amount();
126 long long discount_amount = rounded(subtotal, discount);
127 long long after = subtotal - discount_amount;
128 long long service = size >= 6 ? rounded(after, 18) : 0;
129 long long tax = rounded(after, 8);
130 long long total = after + service + tax;
131 return {subtotal, discount_amount, service, tax, total, total - paid};
132 }
133};

194 static void remove_from(deque<OrderLine*>& queue, OrderLine* line) {
195 auto found = find(queue.begin(), queue.end(), line);
196 if (found != queue.end()) queue.erase(found);

199 Table& table(const string& text, bool occupied = false) {
200 int number = stoi(text);
201 auto found = tables.find(number);
202 if (found == tables.end()) fail("unknown table " + text);
203 if (occupied && !found->second.check) fail("table " + text + " is free");
204 return found->second;
205 }

207 OrderLine& line(const string& text) {
208 int number = stoi(text);
209 if (number < 1 || number > static_cast<int>(lines.size()))
210 fail("unknown line " + text);
211 return *lines[number - 1];
212 }

219 void modification(const OrderLine& order) {
220 if (order.state == "VOID") fail(order.tag() + " is void");
221 if (order.check->closed) fail(order.tag() + " is on a closed check");

225 static void all_served(const Check& check) {
226 int unserved = 0;
227 for (OrderLine* line : check.lines)
228 if (line->state != "VOID" && line->state != "SERVED") ++unserved;
229 if (unserved)
230 fail("table " + to_string(check.table->number) + " has " +
231 to_string(unserved) + " unserved items");
232 }

234 Table* select_table(int size) {
235 Table* best = nullptr;
236 for (auto& [number, table] : tables) {
237 if (!table.check && table.seats >= size &&
238 (!best || table.seats < best->seats ||
239 (table.seats == best->seats && table.number < best->number))) {
240 best = &table;
241 }
242 }
243 return best;
244 }
245
246 Check& seat_party(const string& party, int size, Table& target) {
247 Server* assigned = nullptr;
248 for (const auto& server : servers) {
249 if (!assigned || server->load < assigned->load)
250 assigned = server.get();
251 }
252 checks.push_back(make_unique<Check>(
253 Check{party, size, &target, assigned, now}
254 ));
255 Check& check = *checks.back();
256 target.check = &check;
257 seated[party] = &check;
258 ++assigned->load;
259 ++assigned->parties;
260 guests += size;
261 ++parties;
262 return check;
263 }
264
265 void process_waitlist() {
266 vector<Waiting> remaining;
267 for (const Waiting& entry : waiting) {
268 Table* target = select_table(entry.size);
269 if (!target) {
270 remaining.push_back(entry);
271 } else {
272 Check& check = seat_party(entry.party, entry.size, *target);
273 emit("SEATED", entry.party, entry.size, "table", target->number,
274 "server", check.server->name, "waited", now - entry.joined);
275 }
276 }
277 waiting = move(remaining);
278 }
279
280 void close_check(Check& check) {
281 auto amounts = check.amounts();
282 for (OrderLine* line : check.active())
283 sales[line->course] += line->amount();
284 for (size_t i = 0; i < totals.size(); ++i)
285 totals[i] += amounts[i];
286 int stay = now - check.since;
287 ++closed;
288 stay_total += stay;
289 check.server->sales += amounts[4];
290 --check.server->load;
291 check.closed = true;
292 check.table->check = nullptr;
293 seated.erase(check.party);
294 emit("FREE", "table", check.table->number, check.party, "stayed", stay);
295 process_waitlist();
296 }

298 void start_station(int station_index, int when) {
299 Station& station = stations[station_index];
300 while (station.up && !station.queue.empty() &&
301 static_cast<int>(station.cooking.size()) < station.capacity) {
302 OrderLine* order = station.queue.front();
303 station.queue.pop_front();
304 order->state = "COOKING";
305 order->started = true;
306 order->ready = when + order->prep;
307 station.cooking.push_back(order);
308 }
309 }

311 void start_all(int when) {
312 for (int i = 0; i < 4; ++i) start_station(i, when);
313 }

322 static map<int, long long> allocate(const map<int, long long>& values, int rate) {
323 map<int, long long> amounts;
324 long long subtotal = 0, allocated = 0;
325 vector<int> ranking;
326 for (auto [seat, value] : values) {
327 amounts[seat] = value * rate / 100;
328 subtotal += value;
329 allocated += amounts[seat];
330 ranking.push_back(seat);
331 }
332 long long extra = rounded(subtotal, rate) - allocated;
333 sort(ranking.begin(), ranking.end(), [&](int a, int b) {
334 long long rem_a = values.at(a) * rate % 100;
335 long long rem_b = values.at(b) * rate % 100;
336 return rem_a != rem_b ? rem_a > rem_b : a < b;
337 });
338 for (int i = 0; i < extra; ++i) ++amounts[ranking[i]];
339 return amounts;
340 }

359 void cmd_seat(const vector<string>& a) {
360 string party = a[0];
361 int size = stoi(a[1]);
362 if (seated.count(party))
363 fail(party + " is already seated at table " +
364 to_string(seated.at(party)->table->number));
365 for (const Waiting& entry : waiting)
366 if (entry.party == party) fail(party + " is already waiting");
367 bool capacity = false;
368 for (const auto& [number, table] : tables)
369 if (table.seats >= size) capacity = true;
370 if (!capacity) fail("no table for party of " + a[1]);
371 if (servers.empty()) fail("no server on duty");
372 Table* target = select_table(size);
373 if (!target) {
374 waiting.push_back(Waiting{party, size, now});
375 emit("WAIT", party, size, "position", waiting.size());
376 } else {
377 Check& check = seat_party(party, size, *target);
378 emit("SEAT", party, size, "table", target->number,
379 "server", check.server->name);
380 }

457 void cmd_order(const vector<string>& a) {
458 Table& target = table(a[0], true);
459 Check& check = *target.check;
460 unlocked(check);
461 int seat = stoi(a[1]), quantity = stoi(a[3]);
462 if (seat > check.size)
463 fail("seat " + a[1] + " out of range for table " + a[0]);
464 auto found = menu.find(a[2]);
465 if (found == menu.end()) fail("unknown item " + a[2]);
466 MenuItem& item = found->second;
467 if (item.stock && *item.stock < quantity)
468 fail("only " + to_string(*item.stock) + " " + item.name + " left");
469 bool happy = item.course == 0 && happy_until && now < *happy_until;
470 long long unit_price = item.price - (happy ? rounded(item.price, happy_rate) : 0);
471 int number = static_cast<int>(lines.size()) + 1;
472 lines.push_back(make_unique<OrderLine>(
473 OrderLine{number, &check, seat, item.name, quantity,
474 item.station, item.prep, unit_price, item.course}
475 ));
476 OrderLine& order = *lines.back();
477 check.lines.push_back(&order);
478 string text = "ORDER " + order.tag() + " table " + a[0] + " seat " +
479 a[1] + " " + item.name + " x" + a[3] +
480 " " + money(order.amount());
481 if (happy) text += " HAPPY";
482 emit(text);
483 if (item.stock) {
484 *item.stock -= quantity;
485 if (*item.stock == 0) emit("SOLDOUT", item.name);
486 }
487 }

489 void cmd_fire(const vector<string>& a) {
490 Check& check = *table(a[0], true).check;
491 int course = course_index(a[1]);
492 if (course < 0) fail("unknown course " + a[1]);
493 vector<OrderLine*> selected;
494 for (OrderLine* line : check.lines)
495 if (line->course == course && line->state == "ORDERED") selected.push_back(line);
496 if (selected.empty())
497 fail("nothing to fire for " + a[1] + " at table " + a[0]);
498 if (course == 2 || course == 3) {
499 for (int earlier = 1; earlier < course; ++earlier) {
500 for (OrderLine* line : check.lines)
501 if (line->course == earlier && line->state == "ORDERED")
502 fail("fire " + string(COURSES[earlier]) + " first");
503 }
504 }
505 string text = "FIRE table " + a[0] + " " + a[1];
506 for (OrderLine* line : selected) {
507 line->state = "QUEUED";
508 line->fire = now;
509 stations[line->station].queue.push_back(line);
510 text += " " + line->tag();
511 }
512 emit(text);
513 start_all(now);
514 }

516 void cmd_tick(const vector<string>& a) {
517 int target = now + stoi(a[0]);
518 emit("TICK", clock_text(target));
519 while (true) {
520 int completion = -1;
521 for (const Station& station : stations) {
522 for (OrderLine* order : station.cooking) {
523 if (completion < 0 || order->ready < completion)
524 completion = order->ready;
525 }
526 }
527 if (completion < 0 || completion > target) break;
528 for (Station& station : stations) {
529 vector<OrderLine*> finished;
530 for (OrderLine* order : station.cooking)
531 if (order->ready == completion) finished.push_back(order);
532 sort(finished.begin(), finished.end(), [](OrderLine* a, OrderLine* b) {
533 return a->number < b->number;
534 });
535 for (OrderLine* order : finished) {
536 station.cooking.erase(
537 find(station.cooking.begin(), station.cooking.end(), order));
538 order->state = "READY";
539 int ticket = completion - order->fire;
540 ++station.tickets;
541 station.ticket_total += ticket;
542 station.ticket_max = max(station.ticket_max, ticket);
543 emit("READY", clock_text(completion), station.name, order->tag(),
544 order->name, "x" + to_string(order->quantity), "table",
545 order->check->table->number, "seat", order->seat);
546 }
547 }
548 start_all(completion);
549 }
550 now = target;
551 }

647 void cmd_bill(const vector<string>& a) {
648 Check& check = *table(a[0], true).check;
649 emit("BILL", "table", a[0], check.party, "guests", check.size);
650 auto subtotals = seat_subtotals(check);
651 for (auto [seat, subtotal] : subtotals) {
652 emit("SEAT", seat, money(subtotal));
653 for (OrderLine* order : check.lines) {
654 if (order->seat != seat || order->state == "VOID") continue;
655 string text = order->tag() + " " + order->name + " x" +
656 to_string(order->quantity) + " " + money(order->amount());
657 if (order->comp) text += " COMP " + *order->comp;
658 emit(text);
659 }
660 }
661 auto amounts = check.amounts();
662 emit("SUBTOTAL", money(amounts[0]));
663 if (check.discount)
664 emit("DISCOUNT", to_string(check.discount) + "%", money(amounts[1]));
665 if (check.size >= 6) emit("SERVICE", money(amounts[2]));
666 emit("TAX", money(amounts[3]));
667 emit("TOTAL", money(amounts[4]));
668 if (check.paid) {
669 emit("PAID", money(check.paid));
670 emit("DUE", money(amounts[5]));
671 }
672 }

674 void cmd_split(const vector<string>& a) {
675 Check& check = *table(a[0], true).check;
676 long long due = check.amounts()[5];
677 if (due == 0) fail("nothing to split for table " + a[0]);
678 if (a[1] == "EVEN") {
679 int count = stoi(a[2]);
680 if (count > check.size)
681 fail("cannot split table " + a[0] + " " + a[2] + " ways");
682 emit("SPLIT", "table", a[0], "EVEN", count, "due", money(due));
683 for (int payer = 1; payer <= count; ++payer)
684 emit("PAYER", payer, money(due / count + (payer <= due % count)));
685 return;
686 }
687 if (check.paid) fail("table " + a[0] + " has payments");
688 auto subtotals = seat_subtotals(check);
689 auto discounts = allocate(subtotals, check.discount);
690 map<int, long long> after, services, taxes;
691 for (auto [seat, subtotal] : subtotals)
692 after[seat] = subtotal - discounts[seat];
693 if (check.size >= 6) services = allocate(after, 18);
694 else for (auto [seat, subtotal] : subtotals) services[seat] = 0;
695 taxes = allocate(after, 8);
696 emit("SPLIT", "table", a[0], "BYSEAT", "total", money(check.amounts()[4]));
697 for (auto [seat, subtotal] : subtotals) {
698 long long part = after[seat] + services[seat] + taxes[seat];
699 emit("SEAT", seat, "subtotal", money(subtotal),
700 "discount", money(discounts[seat]), "service", money(services[seat]),
701 "tax", money(taxes[seat]), "total", money(part));
702 }
703 }
704
705 void cmd_pay(const vector<string>& a) {
706 Check& check = *table(a[0], true).check;
707 if (!payments.count(a[1])) fail("unknown method " + a[1]);
708 all_served(check);
709 long long due = check.amounts()[5];
710 if (due == 0) fail("nothing to pay for table " + a[0]);
711 long long value = cents(a[2]);
712 long long applied = min(value, due);
713 check.paid += applied;
714 payments[a[1]] += applied;
715 if (value < due) {
716 emit("PAY", "table", a[0], a[1], a[2], "remaining", money(due - value));
717 } else {
718 long long excess = value - due;
719 emit("PAY", "table", a[0], a[1], a[2],
720 a[1] == "CASH" ? "change" : "tip", money(excess));
721 if (a[1] == "CARD") {
722 tips += excess;
723 check.server->tips += excess;
724 }
725 close_check(check);
726 }
727 }
728
729 void cmd_close(const vector<string>& a) {
730 Check& check = *table(a[0], true).check;
731 all_served(check);
732 long long due = check.amounts()[5];
733 if (due) fail("table " + a[0] + " still owes " + money(due));
734 close_check(check);
735 }

737 void cmd_status(const vector<string>& a) {
738 Check& check = *table(a[0], true).check;
739 emit("STATUS", "table", a[0], check.party, "guests", check.size,
740 "since", clock_text(check.since));
741 for (OrderLine* order : check.lines) {
742 string text = order->tag() + " seat " + to_string(order->seat) + " " +
743 order->name + " x" + to_string(order->quantity) + " " +
744 COURSES[order->course] + " " + order->state;
745 if (order->state == "COOKING") text += " until " + clock_text(order->ready);
746 if (order->comp) text += " COMP " + *order->comp;
747 emit(text);
748 }
749 }

751 void cmd_kitchen(const vector<string>&) {
752 emit("KITCHEN", clock_text(now));
753 for (const Station& station : stations) {
754 vector<OrderLine*> cooking = station.cooking;
755 sort(cooking.begin(), cooking.end(), [](OrderLine* a, OrderLine* b) {
756 return a->ready != b->ready ? a->ready < b->ready : a->number < b->number;
757 });
758 string cooking_text, queue_text;
759 for (OrderLine* order : cooking) {
760 if (!cooking_text.empty()) cooking_text += " ";
761 cooking_text += order->tag() + "@" + clock_text(order->ready);
762 }
763 for (OrderLine* order : station.queue) {
764 if (!queue_text.empty()) queue_text += " ";
765 queue_text += order->tag();
766 }
767 emit(station.name, station.up ? "up" : "down", "cooking",
768 cooking_text.empty() ? "-" : cooking_text, "queue",
769 queue_text.empty() ? "-" : queue_text);
770 }
771 }

773 void cmd_floor(const vector<string>&) {
774 int occupied = 0;
775 for (const auto& [number, table] : tables)
776 if (table.check) ++occupied;
777 emit("FLOOR", clock_text(now), "free", tables.size() - occupied,
778 "seated", occupied, "waiting", waiting.size());
779 for (auto& [number, table] : tables) {
780 if (!table.check) {
781 emit("table", number, "seats", table.seats, "free");
782 continue;
783 }
784 Check& check = *table.check;
785 string text = "table " + to_string(number) + " seats " + to_string(table.seats) +
786 " " + check.party + " guests " + to_string(check.size) +
787 " server " + check.server->name + " since " +
788 clock_text(check.since) + " items " +
789 to_string(check.active().size()) + " total " +
790 money(check.amounts()[4]);
791 if (check.paid) text += " paid " + money(check.paid);
792 emit(text);
793 }
794 int position = 0;
795 for (const Waiting& entry : waiting)
796 emit("WAITING", ++position, entry.party, entry.size,
797 "since", clock_text(entry.joined));
798 }

800 void cmd_report(const vector<string>&) {
801 emit("REPORT", clock_text(now));
802 emit("COVERS", guests, "parties", parties);
803 emit("TURNS", closed, "avg_stay",
804 closed ? to_string(stay_total / closed) : "-");
805 emit("SALES", "DRINK", money(sales[0]), "STARTER", money(sales[1]),
806 "MAIN", money(sales[2]), "DESSERT", money(sales[3]));
807 emit("TOTALS", "subtotal", money(totals[0]), "discount", money(totals[1]),
808 "service", money(totals[2]), "tax", money(totals[3]),
809 "total", money(totals[4]));
810 emit("PAYMENTS", "cash", money(payments["CASH"]),
811 "card", money(payments["CARD"]), "tips", money(tips));
812 emit("COMPS", comps, money(comp_amount), "VOIDS", voids);
813 vector<pair<string, long long>> top(served.begin(), served.end());
814 sort(top.begin(), top.end(), [](const auto& a, const auto& b) {
815 return a.second != b.second ? a.second > b.second : a.first < b.first;
816 });
817 for (size_t i = 0; i < min<size_t>(3, top.size()); ++i)
818 emit("TOP", i + 1, top[i].first, top[i].second);
819 for (const Station& station : stations)
820 emit(station.name, "tickets", station.tickets, "avg",
821 station.tickets ? to_string(station.ticket_total / station.tickets) : "-",
822 "max", station.tickets ? to_string(station.ticket_max) : "-",
823 "remakes", station.remakes);
824 for (const auto& server : servers)
825 emit("SERVER", server->name, "tables", server->load, "parties",
826 server->parties, "sales", money(server->sales), "tips",
827 money(server->tips));
828 }

830public:
831 void execute(const string& input) {
832 istringstream source(input);
833 vector<string> parts;
834 for (string part; source >> part;) parts.push_back(part);
835 if (parts.empty()) return;
836 string command = parts.front();
837 vector<string> a(parts.begin() + 1, parts.end());
838 try {
839 if (command == "TABLE") cmd_table(a);
840 else if (command == "STAFF") cmd_staff(a);
841 else if (command == "SEAT") cmd_seat(a);
842 else if (command == "CANCEL") cmd_cancel(a);
843 else if (command == "MOVE") cmd_move(a);
844 else if (command == "MENU") cmd_menu(a);
845 else if (command == "STOCK") cmd_stock(a);
846 else if (command == "HAPPY") cmd_happy(a);
847 else if (command == "ITEMS") cmd_items(a);
848 else if (command == "ORDER") cmd_order(a);
849 else if (command == "FIRE") cmd_fire(a);
850 else if (command == "TICK") cmd_tick(a);
851 else if (command == "SERVE") cmd_serve(a);
852 else if (command == "RUSH") cmd_rush(a);
853 else if (command == "REMAKE") cmd_remake(a);
854 else if (command == "VOID") cmd_void(a);
855 else if (command == "STATION") cmd_station(a);
856 else if (command == "COMP") cmd_comp(a);
857 else if (command == "TRANSFER") cmd_transfer(a);
858 else if (command == "DISCOUNT") cmd_discount(a);
859 else if (command == "BILL") cmd_bill(a);
860 else if (command == "SPLIT") cmd_split(a);
861 else if (command == "PAY") cmd_pay(a);
862 else if (command == "CLOSE") cmd_close(a);
863 else if (command == "STATUS") cmd_status(a);
864 else if (command == "KITCHEN") cmd_kitchen(a);
865 else if (command == "FLOOR") cmd_floor(a);
866 else if (command == "REPORT") cmd_report(a);
867 } catch (const runtime_error& error) {
868 emit("ERROR", error.what());
869 }
870 }
