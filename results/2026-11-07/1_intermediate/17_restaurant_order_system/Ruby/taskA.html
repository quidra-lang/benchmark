1STATIONS = %w[GRILL FRY COLD BAR].freeze
2COURSES = %w[DRINK STARTER MAIN DESSERT].freeze
3Server = Struct.new(:name, :parties, :sales, :tips)
4Table = Struct.new(:number, :seats, :check)
5Check = Struct.new(:party, :size, :table, :server, :seated, :lines, :paid, :discount, :closed)
6MenuItem = Struct.new(:name, :station, :prep, :price, :course, :stock)
7Line = Struct.new(:id, :check, :seat, :item, :quantity, :station, :prep, :price, :course,
8 :state, :reason, :fire, :ready, :started)
9class CommandError < StandardError; end
10
11def cents(text)
12 text.delete('.').to_i
13end
14
15def money(value)
16 format('%d.%02d', value / 100, value % 100)
17end
18
19def clock(time)
20 format('%02d:%02d', (660 + time) / 60, (660 + time) % 60)
21end
22
23def rounded(value, rate)
24 (value * rate + 50) / 100
25end
26
27def amount(line)
28 line.reason.nil? ? line.price * line.quantity : 0
29end
30
31def active(check)
32 check.lines.reject { |line| line.state == 'VOID' }
33end
34
35class Restaurant
36 attr_reader :output
37
38 def initialize
39 @now = 0
40 @tables = {}
41 @servers = []
42 @waiting = []
43 @menu = {}
44 @lines = []
45 @stations = STATIONS.to_h do |name|
46 [name, {capacity: name == 'BAR' ? 2 : 1, up: true, queue: [], cooking: [], tickets: [], remakes: 0}]
47 end
48 @happy_rate = 0
49 @happy_end = 0
50 @covers = 0
51 @parties = 0
52 @turns = 0
53 @stay = 0
54 @sales = COURSES.to_h { |course| [course, 0] }
55 @totals = %i[subtotal discount service tax total].to_h { |field| [field, 0] }
56 @payments = {'CASH' => 0, 'CARD' => 0}
57 @tips = 0
58 @comps = 0
59 @comp_amount = 0
60 @voids = 0
61 @served = {}
62 @output = []
63 end
64
65 def emit(*parts)
66 @output << parts.join(' ')
67 end
68
69 def table(text)
70 raise CommandError, "unknown table #{text}" unless @tables.key?(text.to_i)
71 @tables[text.to_i]
72 end
73
74 def check(text)
75 t = table(text)
76 raise CommandError, "table #{text} is free" unless t.check
77 t.check
78 end
79
80 def unlocked(c)
81 raise CommandError, "check for table #{c.table.number} is locked" if c.paid.positive?
82 end
83
84 def line(text)
85 id = text.to_i
86 raise CommandError, "unknown line #{text}" unless (1..@lines.length).cover?(id)
87 @lines[id - 1]
88 end
89
90 def modifiable(l)
91 raise CommandError, "##{l.id} is void" if l.state == 'VOID'
92 raise CommandError, "##{l.id} is on a closed check" if l.check.closed
93 unlocked(l.check)
94 end
95
96 def unserved(c)
97 count = active(c).count { |l| l.state != 'SERVED' }
98 raise CommandError, "table #{c.table.number} has #{count} unserved items" if count.positive?
99 end
100
101 def station(name)
102 raise CommandError, "unknown station #{name}" unless @stations.key?(name)
103 @stations[name]
104 end
105
106 def course(name)
107 raise CommandError, "unknown course #{name}" unless COURSES.include?(name)
108 end
109
110 def item(name)
111 raise CommandError, "unknown item #{name}" unless @menu.key?(name)
112 @menu[name]
113 end
114
115 def load(server)
116 @tables.values.count { |t| t.check && t.check.server.equal?(server) }
117 end
118
119 def select_table(size)
120 @tables.values.select { |t| t.check.nil? && t.seats >= size }.min_by { |t| [t.seats, t.number] }
121 end
122
123 def seat(party, size, table)
124 server = @servers.each_with_index.min_by { |s, i| [load(s), i] }[0]
125 c = Check.new(party, size, table, server, @now, [], 0, 0, false)
126 table.check = c
127 server.parties += 1
128 @covers += size
129 @parties += 1
130 c
131 end
132
133 def process_waiting
134 remaining = []
135 @waiting.each do |party, size, joined|
136 t = select_table(size)
137 if t
138 c = seat(party, size, t)
139 emit('SEATED', party, size, 'table', t.number, 'server', c.server.name, 'waited', @now - joined)
140 else
141 remaining << [party, size, joined]
142 end
143 end
144 @waiting = remaining
145 end
146
147 def amounts(c)
148 subtotal = active(c).sum { |l| amount(l) }
149 discount = rounded(subtotal, c.discount)
150 after = subtotal - discount
151 service = c.size >= 6 ? rounded(after, 18) : 0
152 tax = rounded(after, 8)
153 total = after + service + tax
154 {subtotal: subtotal, discount: discount, service: service, tax: tax, total: total, due: total - c.paid}
155 end
156
157 def close(c)
158 a = amounts(c)
159 @turns += 1
160 @stay += @now - c.seated
161 active(c).each { |l| @sales[l.course] += amount(l) }
162 @totals.each_key { |field| @totals[field] += a[field] }
163 c.server.sales += a[:total]
164 c.closed = true
165 c.table.check = nil
166 emit('FREE', 'table', c.table.number, c.party, 'stayed', @now - c.seated)
167 process_waiting
168 end
169
170 def start(s, time)
171 while s[:up] && !s[:queue].empty? && s[:cooking].length < s[:capacity]
172 l = s[:queue].shift
173 l.state = 'COOKING'
174 l.ready = time + l.prep
175 l.started = true
176 s[:cooking] << l
177 end
178 end
179
180 def tick(minutes)
181 finish = @now + minutes
182 emit('TICK', clock(finish))
183 loop do
184 cooking = @stations.values.flat_map { |s| s[:cooking] }
185 break if cooking.empty?
186 time = cooking.map(&:ready).min
187 break if time > finish
188 @stations.each do |name, s|
189 finished = s[:cooking].select { |l| l.ready == time }.sort_by(&:id)
190 s[:cooking].reject! { |l| l.ready == time }
191 finished.each do |l|
192 l.state = 'READY'
193 s[:tickets] << time - l.fire
194 emit('READY', clock(time), name, "##{l.id}", l.item, "x#{l.quantity}", 'table', l.check.table.number, 'seat', l.seat)
195 end
196 end
197 @stations.each_value { |s| start(s, time) }
198 end
199 @now = finish
200 end
201
202 def seats(c)
203 active(c).group_by(&:seat).sort.to_h.transform_values { |lines| lines.sum { |l| amount(l) } }
204 end
205
206 def allocate(values, rate)
207 result = values.to_h { |seat, value| [seat, value * rate / 100] }
208 left = rounded(values.values.sum, rate) - result.values.sum
209 order = values.keys.sort_by { |seat| [-(values[seat] * rate % 100), seat] }
210 order.first(left).each { |seat| result[seat] += 1 }
211 result
212 end
213
214 def bill(c)
215 emit('BILL', 'table', c.table.number, c.party, 'guests', c.size)
216 seats(c).each do |seat, subtotal|
217 emit('SEAT', seat, money(subtotal))
218 active(c).select { |l| l.seat == seat }.each do |l|
219 emit("##{l.id}", l.item, "x#{l.quantity}", money(amount(l)), *(l.reason ? ['COMP', l.reason] : []))
220 end
221 end
222 a = amounts(c)
223 emit('SUBTOTAL', money(a[:subtotal]))
224 emit('DISCOUNT', "#{c.discount}%", money(a[:discount])) unless c.discount.zero?
225 emit('SERVICE', money(a[:service])) if c.size >= 6
226 emit('TAX', money(a[:tax]))
227 emit('TOTAL', money(a[:total]))
228 if c.paid.positive?
229 emit('PAID', money(c.paid))
230 emit('DUE', money(a[:due]))
231 end
232 end
233
234 def report
235 emit('REPORT', clock(@now))
236 emit('COVERS', @covers, 'parties', @parties)
237 emit('TURNS', @turns, 'avg_stay', @turns.positive? ? @stay / @turns : '-')
238 emit('SALES', *COURSES.flat_map { |c| [c, money(@sales[c])] })
239 emit('TOTALS', *@totals.flat_map { |field, value| [field, money(value)] })
240 emit('PAYMENTS', 'cash', money(@payments['CASH']), 'card', money(@payments['CARD']), 'tips', money(@tips))
241 emit('COMPS', @comps, money(@comp_amount), 'VOIDS', @voids)
242 @served.select { |_, count| count.positive? }.sort_by { |name, count| [-count, name] }.first(3).each_with_index do |(name, count), i|
243 emit('TOP', i + 1, name, count)
244 end
245 @stations.each do |name, s|
246 tickets = s[:tickets]
247 emit(name, 'tickets', tickets.length, 'avg', tickets.empty? ? '-' : tickets.sum / tickets.length,
248 'max', tickets.empty? ? '-' : tickets.max, 'remakes', s[:remakes])
249 end
250 @servers.each { |s| emit('SERVER', s.name, 'tables', load(s), 'parties', s.parties, 'sales', money(s.sales), 'tips', money(s.tips)) }
251 end
252
253 def command(cmd, a)
254 case cmd
255 when 'TABLE'
256 number = a[0].to_i
257 raise CommandError, "table #{a[0]} already exists" if @tables.key?(number)
258 @tables[number] = Table.new(number, a[1].to_i, nil)
259 emit('TABLE', number, 'seats', a[1].to_i)
260 process_waiting
261 when 'STAFF'
262 raise CommandError, "#{a[0]} is already on duty" if @servers.any? { |s| s.name == a[0] }
263 @servers << Server.new(a[0], 0, 0, 0)
264 emit('STAFF', a[0])
265 when 'SEAT'
266 seated = @tables.values.find { |t| t.check && t.check.party == a[0] }
267 raise CommandError, "#{a[0]} is already seated at table #{seated.number}" if seated
268 raise CommandError, "#{a[0]} is already waiting" if @waiting.any? { |party, _, _| party == a[0] }
269 size = a[1].to_i
270 raise CommandError, "no table for party of #{size}" unless @tables.values.any? { |t| t.seats >= size }
271 raise CommandError, 'no server on duty' if @servers.empty?
272 t = select_table(size)
273 if t
274 c = seat(a[0], size, t)
275 emit('SEAT', a[0], size, 'table', t.number, 'server', c.server.name)
276 else
277 @waiting << [a[0], size, @now]
278 emit('WAIT', a[0], size, 'position', @waiting.length)
279 end
280 when 'CANCEL'
281 index = @waiting.index { |party, _, _| party == a[0] }
282 raise CommandError, "#{a[0]} is not waiting" unless index
283 party, size, joined = @waiting.delete_at(index)
284 emit('CANCEL', party, size, 'waited', @now - joined)
285 when 'MOVE'
286 source = table(a[0])
287 destination = table(a[1])
288 raise CommandError, "table #{a[0]} is free" unless source.check
289 raise CommandError, "table #{a[1]} is occupied" if destination.check
290 c = source.check
291 raise CommandError, "table #{a[1]} is too small for #{c.party}" if destination.seats < c.size
292 source.check = nil
293 destination.check = c
294 c.table = destination
295 emit('MOVE', c.party, 'table', source.number, '->', destination.number)
296 process_waiting
297 when 'MENU'
298 station(a[1])
299 course(a[4])
300 old = @menu[a[0]]
301 @menu[a[0]] = MenuItem.new(a[0], a[1], a[2].to_i, cents(a[3]), a[4], old&.stock)
302 emit('MENU', old ? 'UPDATE' : 'ADD', a[0], a[1], a[2].to_i, money(cents(a[3])), a[4])
303 when 'STOCK'
304 item = item(a[0])
305 item.stock = a[1].to_i
306 emit('STOCK', item.name, item.stock)
307 when 'HAPPY'
308 @happy_rate = a[0].to_i
309 @happy_end = @now + a[1].to_i
310 emit('HAPPY', "#{@happy_rate}%", 'until', clock(@happy_end))
311 when 'ITEMS'
312 items = @menu.values.sort_by { |i| [COURSES.index(i.course), i.name] }
313 emit('ITEMS', items.length)
314 items.each { |i| emit(i.course, i.name, i.station, i.prep, money(i.price), i.stock.nil? ? '-' : i.stock) }
315 when 'ORDER'
316 c = check(a[0])
317 unlocked(c)
318 seat = a[1].to_i
319 raise CommandError, "seat #{seat} out of range for table #{a[0]}" if seat > c.size
320 item = item(a[2])
321 quantity = a[3].to_i
322 raise CommandError, "only #{item.stock} #{item.name} left" if !item.stock.nil? && item.stock < quantity
323 happy = item.course == 'DRINK' && @now < @happy_end
324 price = item.price - (happy ? rounded(item.price, @happy_rate) : 0)
325 l = Line.new(@lines.length + 1, c, seat, item.name, quantity, item.station, item.prep, price,
326 item.course, 'ORDERED', nil, 0, 0, false)
327 @lines << l
328 c.lines << l
329 emit('ORDER', "##{l.id}", 'table', c.table.number, 'seat', seat, item.name, "x#{quantity}", money(amount(l)), *(happy ? ['HAPPY'] : []))
330 unless item.stock.nil?
331 item.stock -= quantity
332 emit('SOLDOUT', item.name) if item.stock.zero?
333 end
334 when 'FIRE'
335 c = check(a[0])
336 course(a[1])
337 lines = c.lines.select { |l| l.course == a[1] && l.state == 'ORDERED' }
338 raise CommandError, "nothing to fire for #{a[1]} at table #{a[0]}" if lines.empty?
339 if %w[MAIN DESSERT].include?(a[1])
340 %w[STARTER MAIN DESSERT].take_while { |course| course != a[1] }.each do |earlier|
341 raise CommandError, "fire #{earlier} first" if c.lines.any? { |l| l.course == earlier && l.state == 'ORDERED' }
342 end
343 end
344 lines.each do |l|
345 l.state = 'QUEUED'
346 l.fire = @now
347 @stations[l.station][:queue] << l
348 end
349 emit('FIRE', 'table', c.table.number, a[1], *lines.map { |l| "##{l.id}" })
350 @stations.each_value { |s| start(s, @now) }
351 when 'TICK'
352 tick(a[0].to_i)
353 when 'SERVE'
354 c = check(a[0])
355 lines = c.lines.select { |l| l.state == 'READY' }
356 raise CommandError, "nothing ready for table #{a[0]}" if lines.empty?
357 lines.each do |l|
358 l.state = 'SERVED'
359 @served[l.item] = @served.fetch(l.item, 0) + l.quantity
360 end
361 emit('SERVE', 'table', c.table.number, *lines.map { |l| "##{l.id}" })
362 when 'RUSH'
363 l = line(a[0])
364 raise CommandError, "##{l.id} is not queued" unless l.state == 'QUEUED'
365 s = @stations[l.station]
366 s[:queue].delete(l)
367 s[:queue].unshift(l)
368 emit('RUSH', "##{l.id}", l.station)
369 when 'REMAKE'
370 l = line(a[0])
371 raise CommandError, "##{l.id} is on a closed check" if l.check.closed
372 raise CommandError, "##{l.id} is not ready or served" unless %w[READY SERVED].include?(l.state)
373 l.state = 'QUEUED'
374 l.fire = @now
375 s = @stations[l.station]
376 s[:queue].unshift(l)
377 s[:remakes] += 1
378 emit('REMAKE', "##{l.id}", l.item, l.station)
379 start(s, @now)
380 when 'VOID'
381 l = line(a[0])
382 raise CommandError, "##{l.id} is already void" if l.state == 'VOID'
383 raise CommandError, "##{l.id} has already started" if l.started
384 @stations[l.station][:queue].delete(l) if l.state == 'QUEUED'
385 l.state = 'VOID'
386 item = @menu[l.item]
387 item.stock += l.quantity unless item.stock.nil?
388 @voids += 1
389 emit('VOID', "##{l.id}", l.item, "x#{l.quantity}")
390 when 'STATION'
391 s = station(a[0])
392 raise CommandError, "station #{a[0]} is already down" if a[1] == 'DOWN' && !s[:up]
393 raise CommandError, "station #{a[0]} is not down" if a[1] == 'UP' && s[:up]
394 s[:up] = a[1] == 'UP'
395 start(s, @now) if s[:up]
396 emit('STATION', a[0], a[1])
397 when 'COMP', 'TRANSFER'
398 l = line(a[0])
399 modifiable(l)
400 if cmd == 'COMP'
401 raise CommandError, "##{l.id} is already comped" unless l.reason.nil?
402 value = amount(l)
403 emit('COMP', "##{l.id}", l.item, money(value), a[1])
404 @comps += 1
405 @comp_amount += value
406 l.reason = a[1]
407 else
408 seat = a[1].to_i
409 raise CommandError, "seat #{seat} out of range for table #{l.check.table.number}" if seat > l.check.size
410 raise CommandError, "##{l.id} is already on seat #{seat}" if seat == l.seat
411 emit('TRANSFER', "##{l.id}", 'seat', l.seat, '->', seat)
412 l.seat = seat
413 end
414 when 'DISCOUNT'
415 c = check(a[0])
416 unlocked(c)
417 c.discount = a[1].to_i
418 emit('DISCOUNT', 'table', c.table.number, "#{c.discount}%")
419 when 'BILL'
420 bill(check(a[0]))
421 when 'SPLIT'
422 c = check(a[0])
423 amounts = amounts(c)
424 raise CommandError, "nothing to split for table #{a[0]}" if amounts[:due].zero?
425 if a[1] == 'EVEN'
426 count = a[2].to_i
427 raise CommandError, "cannot split table #{a[0]} #{count} ways" if count > c.size
428 emit('SPLIT', 'table', c.table.number, 'EVEN', count, 'due', money(amounts[:due]))
429 q, r = amounts[:due].divmod(count)
430 (1..count).each { |i| emit('PAYER', i, money(q + (i <= r ? 1 : 0))) }
431 else
432 raise CommandError, "table #{a[0]} has payments" if c.paid.positive?
433 values = seats(c)
434 discounts = allocate(values, c.discount)
435 after = values.to_h { |seat, value| [seat, value - discounts[seat]] }
436 services = allocate(after, c.size >= 6 ? 18 : 0)
437 taxes = allocate(after, 8)
438 emit('SPLIT', 'table', c.table.number, 'BYSEAT', 'total', money(amounts[:total]))
439 values.each do |seat, subtotal|
440 emit('SEAT', seat, 'subtotal', money(subtotal), 'discount', money(discounts[seat]),
441 'service', money(services[seat]), 'tax', money(taxes[seat]),
442 'total', money(after[seat] + services[seat] + taxes[seat]))
443 end
444 end
445 when 'PAY'
446 c = check(a[0])
447 raise CommandError, "unknown method #{a[1]}" unless %w[CASH CARD].include?(a[1])
448 unserved(c)
449 due = amounts(c)[:due]
450 raise CommandError, "nothing to pay for table #{a[0]}" if due.zero?
451 value = cents(a[2])
452 applied = [value, due].min
453 c.paid += applied
454 @payments[a[1]] += applied
455 if value < due
456 emit('PAY', 'table', c.table.number, a[1], money(value), 'remaining', money(due - value))
457 else
458 emit('PAY', 'table', c.table.number, a[1], money(value), a[1] == 'CASH' ? 'change' : 'tip', money(value - due))
459 if a[1] == 'CARD'
460 @tips += value - due
461 c.server.tips += value - due
462 end
463 close(c)
464 end
465 when 'CLOSE'
466 c = check(a[0])
467 unserved(c)
468 due = amounts(c)[:due]
469 raise CommandError, "table #{a[0]} still owes #{money(due)}" unless due.zero?
470 close(c)
471 when 'STATUS'
472 c = check(a[0])
473 emit('STATUS', 'table', c.table.number, c.party, 'guests', c.size, 'since', clock(c.seated))
474 c.lines.each do |l|
475 emit("##{l.id}", 'seat', l.seat, l.item, "x#{l.quantity}", l.course, l.state,
476 *(l.state == 'COOKING' ? ['until', clock(l.ready)] : []), *(l.reason ? ['COMP', l.reason] : []))
477 end
478 when 'KITCHEN'
479 emit('KITCHEN', clock(@now))
480 @stations.each do |name, s|
481 cooking = s[:cooking].sort_by { |l| [l.ready, l.id] }.map { |l| "##{l.id}@#{clock(l.ready)}" }
482 queue = s[:queue].map { |l| "##{l.id}" }
483 emit(name, s[:up] ? 'up' : 'down', 'cooking', cooking.empty? ? '-' : cooking.join(' '), 'queue', queue.empty? ? '-' : queue.join(' '))
484 end
485 when 'FLOOR'
486 tables = @tables.values.sort_by(&:number)
487 free = tables.count { |t| t.check.nil? }
488 emit('FLOOR', clock(@now), 'free', free, 'seated', tables.length - free, 'waiting', @waiting.length)
489 tables.each do |t|
490 c = t.check
491 if c
492 emit('table', t.number, 'seats', t.seats, c.party, 'guests', c.size, 'server', c.server.name,
493 'since', clock(c.seated), 'items', active(c).length, 'total', money(amounts(c)[:total]),
494 *(c.paid.positive? ? ['paid', money(c.paid)] : []))
495 else
496 emit('table', t.number, 'seats', t.seats, 'free')
497 end
498 end
499 @waiting.each_with_index { |(party, size, joined), i| emit('WAITING', i + 1, party, size, 'since', clock(joined)) }
500 when 'REPORT'
501 report
502 end
503 end
504
505 def execute(line)
506 cmd, *args = line.split
507 begin
508 command(cmd, args)
509 rescue CommandError => error
510 emit('ERROR', error.message)
511 end
512 end
513end

515restaurant = Restaurant.new
516STDIN.gets.to_i.times { restaurant.execute(STDIN.gets.chomp) }
517STDOUT.write(restaurant.output.join("\n") + "\n")
