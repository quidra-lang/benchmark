4object <- function(...) list2env(list(...), parent = emptyenv())
5number <- function(text) as.numeric(text)
6cents <- function(text) as.numeric(gsub(".", "", text, fixed = TRUE))
7integer_text <- function(value) format(value, scientific = FALSE, trim = TRUE)
8money <- function(value) {
9 whole <- floor(value / 100)
10 paste0(integer_text(whole), ".", sprintf("%02d", as.integer(value %% 100)))
11}
12clock <- function(time) {
13 minutes <- 660 + time
14 sprintf("%02d:%02d", as.integer(floor(minutes / 60)), as.integer(minutes %% 60))
15}
16rounded <- function(amount, rate) floor((amount * rate + 50) / 100)
17fail <- function(reason) {
18 condition <- structure(
19 list(message = reason, call = NULL),
20 class = c("CommandError", "error", "condition")
21 )
22 stop(condition)
23}

25state <- object(

27 menu = list(), lines = list(), stations = list(),

30 sales = setNames(rep(0, length(courses)), courses),

44emit <- function(...) {
45 arguments <- list(...)
46 parts <- unlist(lapply(arguments, function(value) {
47 if (is.numeric(value)) vapply(value, integer_text, character(1))
48 else as.character(value)
49 }), use.names = FALSE)
50 state$output <- c(state$output, paste(parts, collapse = " "))
51}

111select_table <- function(size) {
112 tables <- Filter(function(item) is.null(item$check) && item$seats >= size,
113 state$tables)
114 if (!length(tables)) return(NULL)
115 indexes <- order(
116 vapply(tables, function(item) item$seats, numeric(1)),
117 vapply(tables, function(item) item$number, numeric(1))
118 )
119 tables[[indexes[1]]]

122seat_party <- function(party, size, table) {
123 loads <- vapply(state$servers, load_server, numeric(1))
124 server <- state$servers[[which.min(loads)]]
125 check <- object(
126 party = party, size = size, table = table,
127 server = server, seated = state$now,
128 lines = list(), paid = 0, discount = 0, closed = FALSE
129 )
130 table$check <- check
131 server$parties <- server$parties + 1
132 state$covers <- state$covers + size
133 state$parties <- state$parties + 1
134 check
135}

137process_waiting <- function() {
138 remaining <- list()
139 for (entry in state$waiting) {
140 table <- select_table(entry$size)
141 if (is.null(table)) {
142 remaining[[length(remaining) + 1]] <- entry
143 } else {
144 check <- seat_party(entry$party, entry$size, table)
145 emit("SEATED", entry$party, entry$size, "table", table$number,
146 "server", check$server$name, "waited", state$now - entry$joined)
147 }
148 }
149 state$waiting <- remaining
150}
151
152line_amount <- function(line) {
153 if (is.null(line$reason)) line$price * line$quantity else 0
154}
155
156check_amounts <- function(check) {
157 subtotal <- sum(vapply(active_lines(check), line_amount, numeric(1)))
158 discount <- rounded(subtotal, check$discount)
159 after <- subtotal - discount
160 service <- if (check$size >= 6) rounded(after, 18) else 0
161 tax <- rounded(after, 8)
162 total <- after + service + tax
163 c(subtotal = subtotal, discount = discount, service = service,
164 tax = tax, total = total, due = total - check$paid)
165}

167close_check <- function(check) {
168 amounts <- check_amounts(check)
169 state$turns <- state$turns + 1
170 state$stay <- state$stay + state$now - check$seated
171 for (line in active_lines(check)) {
172 state$sales[line$course] <- state$sales[line$course] + line_amount(line)
173 }
174 for (field in names(state$totals)) {
175 state$totals[field] <- state$totals[field] + amounts[field]
176 }
177 check$server$sales <- check$server$sales + amounts["total"]
178 check$closed <- TRUE
179 check$table$check <- NULL
180 emit("FREE", "table", check$table$number, check$party,
181 "stayed", state$now - check$seated)
182 process_waiting()
183}

185start_station <- function(station, time) {
186 while (station$up && length(station$queue) &&
187 length(station$cooking) < station$capacity) {
188 line <- station$queue[[1]]
189 station$queue <- station$queue[-1]
190 line$status <- "COOKING"
191 line$ready <- time + line$prep
192 line$started <- TRUE
193 station$cooking[[length(station$cooking) + 1]] <- line
194 }
195}

197tick <- function(minutes) {
198 finish <- state$now + minutes
199 emit("TICK", clock(finish))
200 repeat {
201 cooking <- unlist(lapply(state$stations, function(station) station$cooking),
202 recursive = FALSE)
203 if (!length(cooking)) break
204 time <- min(vapply(cooking, function(line) line$ready, numeric(1)))
205 if (time > finish) break
206 for (name in stations_order) {
207 station <- state$stations[[name]]
208 finished <- Filter(function(line) line$ready == time, station$cooking)
209 station$cooking <- Filter(function(line) line$ready != time, station$cooking)
210 if (length(finished)) {
211 finished <- finished[order(vapply(finished, function(line) line$id, numeric(1)))]
212 }
213 for (line in finished) {
214 line$status <- "READY"
215 station$tickets <- c(station$tickets, time - line$fire)
216 emit("READY", clock(time), name, paste0("#", line$id),
217 line$item, paste0("x", line$quantity),
218 "table", line$check$table$number, "seat", line$seat)
219 }
220 }
221 for (name in stations_order) start_station(state$stations[[name]], time)
222 }
223 state$now <- finish
224}

226seat_totals <- function(check) {
227 lines <- active_lines(check)
228 if (!length(lines)) return(setNames(numeric(), character()))
229 seats <- sort(unique(vapply(lines, function(line) line$seat, numeric(1))))
230 setNames(vapply(seats, function(seat)
231 sum(vapply(Filter(function(line) line$seat == seat, lines),
232 line_amount, numeric(1))), numeric(1)), as.character(seats))
233}

235allocate <- function(values, rate) {
236 result <- floor(values * rate / 100)
237 left <- rounded(sum(values), rate) - sum(result)
238 if (left > 0 && length(values)) {
239 seats <- as.numeric(names(values))
240 sorted <- order(-(values * rate %% 100), seats)
241 for (index in head(sorted, left)) result[index] <- result[index] + 1
242 }
243 result
244}

246show_bill <- function(check) {
247 emit("BILL", "table", check$table$number, check$party, "guests", check$size)
248 seat_values <- seat_totals(check)
249 for (seat in names(seat_values)) {
250 emit("SEAT", seat, money(seat_values[[seat]]))
251 for (line in active_lines(check)) {
252 if (as.character(line$seat) != seat) next
253 if (is.null(line$reason)) {
254 emit(paste0("#", line$id), line$item, paste0("x", line$quantity),
255 money(line_amount(line)))
256 } else {
257 emit(paste0("#", line$id), line$item, paste0("x", line$quantity),
258 money(line_amount(line)), "COMP", line$reason)
259 }
260 }
261 }
262 a <- check_amounts(check)
263 emit("SUBTOTAL", money(a["subtotal"]))
264 if (check$discount != 0)
265 emit("DISCOUNT", paste0(check$discount, "%"), money(a["discount"]))
266 if (check$size >= 6) emit("SERVICE", money(a["service"]))
267 emit("TAX", money(a["tax"]))
268 emit("TOTAL", money(a["total"]))
269 if (check$paid > 0) {
270 emit("PAID", money(check$paid))
271 emit("DUE", money(a["due"]))
272 }
273}

275show_report <- function() {
276 emit("REPORT", clock(state$now))
277 emit("COVERS", state$covers, "parties", state$parties)
278 emit("TURNS", state$turns, "avg_stay",
279 if (state$turns > 0) floor(state$stay / state$turns) else "-")
280 sales <- c("SALES")
281 for (course in courses) sales <- c(sales, course, money(state$sales[course]))
282 emit(sales)
283 totals <- c("TOTALS")
284 for (field in names(state$totals)) totals <- c(totals, field, money(state$totals[field]))
285 emit(totals)
286 emit("PAYMENTS", "cash", money(state$payments["CASH"]),
287 "card", money(state$payments["CARD"]), "tips", money(state$tips))
288 emit("COMPS", state$comps, money(state$comp_amount), "VOIDS", state$voids)
289 if (length(state$served)) {
290 names <- names(state$served)
291 ranked <- order(-as.numeric(state$served), names)
292 for (index in seq_len(min(3, length(ranked)))) {
293 item <- ranked[index]

297 for (name in stations_order) {
298 station <- state$stations[[name]]
299 tickets <- station$tickets
300 emit(name, "tickets", length(tickets), "avg",
301 if (length(tickets)) floor(sum(tickets) / length(tickets)) else "-",
302 "max", if (length(tickets)) max(tickets) else "-",
303 "remakes", station$remakes)
304 }
305 for (server in state$servers) {
306 emit("SERVER", server$name, "tables", load_server(server),
307 "parties", server$parties, "sales", money(server$sales),
308 "tips", money(server$tips))
309 }

312run_command <- function(command, a) {
313 if (command == "TABLE") {
314 id <- number(a[1])
315 if (!is.null(state$tables[[as.character(id)]]))
316 fail(paste("table", a[1], "already exists"))
317 table <- object(number = id, seats = number(a[2]), check = NULL)
318 state$tables[[as.character(id)]] <- table
319 emit("TABLE", id, "seats", table$seats)
320 process_waiting()
321 } else if (command == "STAFF") {
322 if (any(vapply(state$servers, function(server) server$name == a[1], logical(1))))
323 fail(paste(a[1], "is already on duty"))
324 state$servers[[length(state$servers) + 1]] <-
325 object(name = a[1], parties = 0, sales = 0, tips = 0)
326 emit("STAFF", a[1])
327 } else if (command == "SEAT") {
328 for (table in state$tables) {
329 if (!is.null(table$check) && table$check$party == a[1])
330 fail(paste(a[1], "is already seated at table", table$number))
331 }
332 for (entry in state$waiting) {
333 if (entry$party == a[1]) fail(paste(a[1], "is already waiting"))
334 }
335 size <- number(a[2])
336 if (!any(vapply(state$tables, function(table) table$seats >= size, logical(1))))
337 fail(paste("no table for party of", size))
338 if (!length(state$servers)) fail("no server on duty")
339 table <- select_table(size)
340 if (is.null(table)) {
341 state$waiting[[length(state$waiting) + 1]] <-
342 list(party = a[1], size = size, joined = state$now)
343 emit("WAIT", a[1], size, "position", length(state$waiting))
344 } else {
345 check <- seat_party(a[1], size, table)
346 emit("SEAT", a[1], size, "table", table$number, "server", check$server$name)
347 }
348 } else if (command == "CANCEL") {
349 index <- which(vapply(state$waiting, function(entry) entry$party == a[1], logical(1)))
350 if (!length(index)) fail(paste(a[1], "is not waiting"))
351 entry <- state$waiting[[index[1]]]
352 state$waiting <- state$waiting[-index[1]]
353 emit("CANCEL", entry$party, entry$size, "waited", state$now - entry$joined)
354 } else if (command == "MOVE") {
355 source <- get_table(a[1])
356 destination <- get_table(a[2])
357 if (is.null(source$check)) fail(paste("table", a[1], "is free"))
358 if (!is.null(destination$check)) fail(paste("table", a[2], "is occupied"))
359 check <- source$check
360 if (destination$seats < check$size)
361 fail(paste("table", a[2], "is too small for", check$party))
362 source$check <- NULL
363 destination$check <- check
364 check$table <- destination
365 emit("MOVE", check$party, "table", source$number, "->", destination$number)
366 process_waiting()
367 } else if (command == "MENU") {
368 get_station(a[2])
369 require_course(a[5])
370 old <- state$menu[[a[1]]]
371 item <- object(
372 name = a[1], station = a[2], prep = number(a[3]),
373 price = cents(a[4]), course = a[5],
374 stock = if (is.null(old)) NULL else old$stock
375 )
376 state$menu[[a[1]]] <- item
377 emit("MENU", if (is.null(old)) "ADD" else "UPDATE",
378 a[1], a[2], item$prep, money(item$price), a[5])
379 } else if (command == "STOCK") {
380 item <- get_item(a[1])
381 item$stock <- number(a[2])
382 emit("STOCK", item$name, item$stock)
383 } else if (command == "HAPPY") {
384 state$happy_rate <- number(a[1])
385 state$happy_end <- state$now + number(a[2])
386 emit("HAPPY", paste0(state$happy_rate, "%"), "until", clock(state$happy_end))
387 } else if (command == "ITEMS") {
388 items <- unname(state$menu)
389 if (length(items)) {
390 ord <- order(
391 vapply(items, function(item) match(item$course, courses), integer(1)),
392 vapply(items, function(item) item$name, character(1))
393 )
394 items <- items[ord]
395 }
396 emit("ITEMS", length(items))
397 for (item in items)
398 emit(item$course, item$name, item$station, item$prep, money(item$price),
399 if (is.null(item$stock)) "-" else item$stock)
400 } else if (command == "ORDER") {
401 check <- get_check(a[1])
402 unlocked(check)
403 seat <- number(a[2])
404 if (seat > check$size)
405 fail(paste("seat", seat, "out of range for table", a[1]))
406 item <- get_item(a[3])
407 quantity <- number(a[4])
408 if (!is.null(item$stock) && item$stock < quantity)
409 fail(paste("only", item$stock, item$name, "left"))
410 happy <- item$course == "DRINK" && state$now < state$happy_end
411 price <- item$price - (if (happy) rounded(item$price, state$happy_rate) else 0)
412 line <- object(
413 id = length(state$lines) + 1, check = check, seat = seat,
414 item = item$name, quantity = quantity, station = item$station,
415 prep = item$prep, price = price, course = item$course,
416 status = "ORDERED", reason = NULL, fire = 0, ready = 0, started = FALSE
417 )
418 state$lines[[length(state$lines) + 1]] <- line
419 check$lines[[length(check$lines) + 1]] <- line
420 if (happy) {
421 emit("ORDER", paste0("#", line$id), "table", check$table$number, "seat", seat,
422 item$name, paste0("x", quantity), money(line_amount(line)), "HAPPY")
423 } else {
424 emit("ORDER", paste0("#", line$id), "table", check$table$number, "seat", seat,
425 item$name, paste0("x", quantity), money(line_amount(line)))
426 }
427 if (!is.null(item$stock)) {
428 item$stock <- item$stock - quantity
429 if (item$stock == 0) emit("SOLDOUT", item$name)
430 }
431 } else if (command == "FIRE") {
432 check <- get_check(a[1])
433 require_course(a[2])
434 lines <- Filter(function(line) line$course == a[2] && line$status == "ORDERED",
435 check$lines)
436 if (!length(lines)) fail(paste("nothing to fire for", a[2], "at table", a[1]))
437 if (a[2] %in% c("MAIN", "DESSERT")) {
438 earlier <- if (a[2] == "MAIN") "STARTER" else c("STARTER", "MAIN")
439 for (previous in earlier) {
440 if (any(vapply(check$lines, function(line)
441 line$course == previous && line$status == "ORDERED", logical(1))))
442 fail(paste("fire", previous, "first"))
443 }
444 }
445 for (line in lines) {
446 line$status <- "QUEUED"
447 line$fire <- state$now
448 station <- state$stations[[line$station]]
449 station$queue[[length(station$queue) + 1]] <- line
450 }
451 emit("FIRE", "table", check$table$number, a[2],
452 vapply(lines, function(line) paste0("#", line$id), character(1)))
453 for (name in stations_order) start_station(state$stations[[name]], state$now)
454 } else if (command == "TICK") {
455 tick(number(a[1]))
456 } else if (command == "SERVE") {
457 check <- get_check(a[1])
458 lines <- Filter(function(line) line$status == "READY", check$lines)
459 if (!length(lines)) fail(paste("nothing ready for table", a[1]))
460 for (line in lines) {
461 line$status <- "SERVED"
462 if (is.na(match(line$item, names(state$served))))
463 state$served[line$item] <- 0
464 state$served[line$item] <- state$served[line$item] + line$quantity
465 }
466 emit("SERVE", "table", check$table$number,
467 vapply(lines, function(line) paste0("#", line$id), character(1)))
468 } else if (command == "RUSH") {
469 line <- get_line(a[1])
470 if (line$status != "QUEUED") fail(paste0("#", line$id, " is not queued"))
471 station <- state$stations[[line$station]]
472 station$queue <- c(list(line),
473 Filter(function(other) !identical(other, line), station$queue))
474 emit("RUSH", paste0("#", line$id), line$station)
475 } else if (command == "REMAKE") {
476 line <- get_line(a[1])
477 if (line$check$closed) fail(paste0("#", line$id, " is on a closed check"))
478 if (!(line$status %in% c("READY", "SERVED")))
479 fail(paste0("#", line$id, " is not ready or served"))
480 line$status <- "QUEUED"
481 line$fire <- state$now
482 station <- state$stations[[line$station]]
483 station$queue <- c(list(line), station$queue)
484 station$remakes <- station$remakes + 1
485 emit("REMAKE", paste0("#", line$id), line$item, line$station)
486 start_station(station, state$now)
487 } else if (command == "VOID") {
488 line <- get_line(a[1])
489 if (line$status == "VOID") fail(paste0("#", line$id, " is already void"))
490 if (line$started) fail(paste0("#", line$id, " has already started"))
491 if (line$status == "QUEUED") {
492 station <- state$stations[[line$station]]
493 station$queue <- Filter(function(other) !identical(other, line), station$queue)
494 }
495 line$status <- "VOID"
496 item <- state$menu[[line$item]]
497 if (!is.null(item$stock)) item$stock <- item$stock + line$quantity
498 state$voids <- state$voids + 1
499 emit("VOID", paste0("#", line$id), line$item, paste0("x", line$quantity))
500 } else if (command == "STATION") {
501 station <- get_station(a[1])
502 if (a[2] == "DOWN" && !station$up)
503 fail(paste("station", a[1], "is already down"))
504 if (a[2] == "UP" && station$up)
505 fail(paste("station", a[1], "is not down"))
506 station$up <- a[2] == "UP"
507 if (station$up) start_station(station, state$now)
508 emit("STATION", a[1], a[2])
509 } else if (command %in% c("COMP", "TRANSFER")) {
510 line <- get_line(a[1])
511 modifiable(line)
512 if (command == "COMP") {
513 if (!is.null(line$reason)) fail(paste0("#", line$id, " is already comped"))
514 value <- line_amount(line)
515 emit("COMP", paste0("#", line$id), line$item, money(value), a[2])
516 state$comps <- state$comps + 1
517 state$comp_amount <- state$comp_amount + value
518 line$reason <- a[2]
519 } else {
520 seat <- number(a[2])
521 if (seat > line$check$size)
522 fail(paste("seat", seat, "out of range for table", line$check$table$number))
523 if (seat == line$seat)
524 fail(paste0("#", line$id, " is already on seat ", seat))
525 emit("TRANSFER", paste0("#", line$id), "seat", line$seat, "->", seat)
526 line$seat <- seat
527 }
528 } else if (command == "DISCOUNT") {
529 check <- get_check(a[1])
530 unlocked(check)
531 check$discount <- number(a[2])
532 emit("DISCOUNT", "table", check$table$number, paste0(check$discount, "%"))
533 } else if (command == "BILL") {
534 show_bill(get_check(a[1]))
535 } else if (command == "SPLIT") {
536 check <- get_check(a[1])
537 amounts <- check_amounts(check)
538 if (amounts["due"] == 0) fail(paste("nothing to split for table", a[1]))
539 if (a[2] == "EVEN") {
540 count <- number(a[3])
541 if (count > check$size)
542 fail(paste("cannot split table", a[1], count, "ways"))
543 emit("SPLIT", "table", check$table$number, "EVEN", count,
544 "due", money(amounts["due"]))
545 quotient <- floor(amounts["due"] / count)
546 remainder <- amounts["due"] %% count
547 for (i in seq_len(count))
548 emit("PAYER", i, money(quotient + if (i <= remainder) 1 else 0))
549 } else {
550 if (check$paid > 0) fail(paste("table", a[1], "has payments"))
551 values <- seat_totals(check)
552 discounts <- allocate(values, check$discount)
553 after <- values - discounts
554 services <- allocate(after, if (check$size >= 6) 18 else 0)
555 taxes <- allocate(after, 8)
556 emit("SPLIT", "table", check$table$number, "BYSEAT",
557 "total", money(amounts["total"]))
558 for (seat in names(values)) {
559 emit("SEAT", seat, "subtotal", money(values[[seat]]),
560 "discount", money(discounts[[seat]]),
561 "service", money(services[[seat]]), "tax", money(taxes[[seat]]),
562 "total", money(after[[seat]] + services[[seat]] + taxes[[seat]]))
563 }
564 }
565 } else if (command == "PAY") {
566 check <- get_check(a[1])
567 if (!(a[2] %in% c("CASH", "CARD")))
568 fail(paste("unknown method", a[2]))
569 unserved(check)
570 due <- check_amounts(check)["due"]
571 if (due == 0) fail(paste("nothing to pay for table", a[1]))
572 value <- cents(a[3])
573 applied <- min(value, due)
574 check$paid <- check$paid + applied
