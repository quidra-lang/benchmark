21String clock(int elapsed) {
22 final minutes = elapsed + 660;

58class Check {
59 final String party;
60 final int size;
61 DiningTable table;
62 final Server server;
63 final int since;
64 final List<OrderLine> lines = [];
65 int paid = 0;
66 int discount = 0;
67 bool closed = false;
68
69 Check(this.party, this.size, this.table, this.server, this.since);
70
71 List<OrderLine> get active =>
72 lines.where((line) => line.state != 'VOID').toList();
73
74 List<int> amounts() {
75 final subtotal = active.fold<int>(0, (sum, line) => sum + line.amount);
76 final reduction = rounded(subtotal, discount);
77 final after = subtotal - reduction;
78 final service = size >= 6 ? rounded(after, 18) : 0;
79 final tax = rounded(after, 8);
80 final total = after + service + tax;
81 return [subtotal, reduction, service, tax, total, total - paid];
82 }
83}

198 DiningTable? selectTable(int size) {
199 DiningTable? best;
200 for (final table in tables.values) {
201 if (table.check == null &&
202 table.seats >= size &&
203 (best == null ||
204 table.seats < best.seats ||
205 (table.seats == best.seats && table.number < best.number))) {
206 best = table;
207 }
208 }
209 return best;
210 }
211
212 Check seatParty(String party, int size, DiningTable table) {
213 var server = servers.first;
214 for (final candidate in servers) {
215 if (candidate.load < server.load) server = candidate;
216 }
217 final check = Check(party, size, table, server, time);
218 table.check = check;
219 seated[party] = check;
220 server.load++;
221 server.parties++;
222 guests += size;
223 parties++;
224 return check;
225 }

227 void processWaitlist() {
228 final remaining = <Waiting>[];
229 for (final entry in waiting) {
230 final table = selectTable(entry.size);
231 if (table == null) {
232 remaining.add(entry);
233 } else {
234 final check = seatParty(entry.party, entry.size, table);
235 emit(['SEATED', entry.party, entry.size, 'table', table.number,
236 'server', check.server.name, 'waited', time - entry.joined]);
237 }
238 }
239 waiting = remaining;
240 }
241
242 void closeCheck(Check check) {
243 final amounts = check.amounts();
244 for (final line in check.active) {
245 sales[courseNames.indexOf(line.course)] += line.amount;
246 }
247 for (var i = 0; i < totals.length; i++) {
248 totals[i] += amounts[i];
249 }
250 final stayed = time - check.since;
251 closed++;
252 stayTotal += stayed;
253 check.server.sales += amounts[4];
254 check.server.load--;
255 check.closed = true;
256 check.table.check = null;
257 seated.remove(check.party);
258 emit(['FREE', 'table', check.table.number, check.party, 'stayed', stayed]);
259 processWaitlist();
260 }

262 void startStation(String name, int when) {
263 final station = stations[name]!;
264 while (station.up &&
265 station.queue.isNotEmpty &&
266 station.cooking.length < station.capacity) {
267 final order = station.queue.removeAt(0);
268 order.state = 'COOKING';
269 order.started = true;
270 order.ready = when + order.prep;
271 station.cooking.add(order);
272 }
273 }

275 void startAll(int when) {
276 for (final name in stationNames) {
277 startStation(name, when);
278 }

284 values[line.seat] = (values[line.seat] ?? 0) + line.amount;

289 Map<int, int> allocate(Map<int, int> values, int rate) {
290 final amounts = <int, int>{};
291 var total = 0;
292 var initiallyAllocated = 0;
293 for (final entry in values.entries) {
294 final part = entry.value * rate ~/ 100;
295 amounts[entry.key] = part;
296 total += entry.value;
297 initiallyAllocated += part;
298 }
299 final extra = rounded(total, rate) - initiallyAllocated;
300 final ranking = values.keys.toList()
301 ..sort((a, b) {
302 final left = values[a]! * rate % 100;
303 final right = values[b]! * rate % 100;
304 return left != right ? right.compareTo(left) : a.compareTo(b);
305 });
306 for (var i = 0; i < extra; i++) {
307 amounts[ranking[i]] = amounts[ranking[i]]! + 1;
308 }
309 return amounts;
310 }

423 void cmdOrder(List<String> a) {
424 final check = table(a[0], true).check!;
425 unlocked(check);
426 final seat = int.parse(a[1]);
427 final quantity = int.parse(a[3]);
428 if (seat > check.size) fail('seat $seat out of range for table ${a[0]}');
429 final item = menu[a[2]];
430 if (item == null) fail('unknown item ${a[2]}');
431 if (item.stock != null && item.stock! < quantity) {
432 fail('only ${item.stock} ${item.name} left');
433 }
434 final happy = item.course == 'DRINK' &&
435 happyUntil != null && time < happyUntil!;
436 final price = item.price - (happy ? rounded(item.price, happyRate) : 0);
437 final order = OrderLine(lines.length + 1, check, seat, item, quantity, price);
438 lines.add(order);
439 check.lines.add(order);
440 emit(['ORDER', order.tag, 'table', a[0], 'seat', seat, item.name,
441 'x$quantity', money(order.amount), if (happy) 'HAPPY']);
442 if (item.stock != null) {
443 item.stock = item.stock! - quantity;
444 if (item.stock == 0) emit(['SOLDOUT', item.name]);
445 }

448 void cmdFire(List<String> a) {
449 final check = table(a[0], true).check!;
450 if (!courseNames.contains(a[1])) fail('unknown course ${a[1]}');
451 final selected = check.lines
452 .where((line) => line.course == a[1] && line.state == 'ORDERED')
453 .toList();
454 if (selected.isEmpty) fail('nothing to fire for ${a[1]} at table ${a[0]}');
455 final course = courseNames.indexOf(a[1]);
456 if (course >= 2) {
457 for (var earlier = 1; earlier < course; earlier++) {
458 if (check.lines.any((line) =>
459 line.course == courseNames[earlier] && line.state == 'ORDERED')) {
460 fail('fire ${courseNames[earlier]} first');
461 }
462 }
463 }
464 for (final order in selected) {
465 order.state = 'QUEUED';
466 order.fired = time;
467 stations[order.station]!.queue.add(order);
468 }
469 emit(['FIRE', 'table', a[0], a[1], ...selected.map((line) => line.tag)]);
470 startAll(time);
471 }

473 void cmdTick(List<String> a) {
474 final target = time + int.parse(a[0]);
475 emit(['TICK', clock(target)]);
476 while (true) {
477 int? completion;
478 for (final station in stations.values) {
479 for (final line in station.cooking) {
480 if (completion == null || line.ready < completion) completion = line.ready;
481 }
482 }
483 if (completion == null || completion > target) break;
484 for (final name in stationNames) {
485 final station = stations[name]!;
486 final finished = station.cooking
487 .where((line) => line.ready == completion).toList()
488 ..sort((a, b) => a.number.compareTo(b.number));
489 for (final order in finished) {
490 station.cooking.remove(order);
491 order.state = 'READY';
492 final ticket = completion - order.fired;
493 station.tickets++;
494 station.ticketTotal += ticket;
495 if (ticket > station.ticketMax) station.ticketMax = ticket;
496 emit(['READY', clock(completion), name, order.tag, order.name,
497 'x${order.quantity}', 'table', order.check.table.number, 'seat', order.seat]);
498 }
499 }
500 startAll(completion);
501 }
502 time = target;
503 }

598 void cmdBill(List<String> a) {
599 final check = table(a[0], true).check!;
600 emit(['BILL', 'table', a[0], check.party, 'guests', check.size]);
601 final subtotals = seatSubtotals(check);
602 final seats = subtotals.keys.toList()..sort();
603 for (final seat in seats) {
604 emit(['SEAT', seat, money(subtotals[seat]!)]);
605 for (final order in check.lines) {
606 if (order.state == 'VOID' || order.seat != seat) continue;
607 emit([order.tag, order.name, 'x${order.quantity}', money(order.amount),
608 if (order.comp != null) ...['COMP', order.comp!]]);
609 }
610 }
611 final amounts = check.amounts();
612 emit(['SUBTOTAL', money(amounts[0])]);
613 if (check.discount != 0) {
614 emit(['DISCOUNT', '${check.discount}%', money(amounts[1])]);
615 }
616 if (check.size >= 6) emit(['SERVICE', money(amounts[2])]);
617 emit(['TAX', money(amounts[3])]);
618 emit(['TOTAL', money(amounts[4])]);
619 if (check.paid != 0) {
620 emit(['PAID', money(check.paid)]);
621 emit(['DUE', money(amounts[5])]);
622 }
623 }

625 void cmdSplit(List<String> a) {
626 final check = table(a[0], true).check!;
627 final due = check.amounts()[5];
628 if (due == 0) fail('nothing to split for table ${a[0]}');
629 if (a[1] == 'EVEN') {
630 final count = int.parse(a[2]);
631 if (count > check.size) fail('cannot split table ${a[0]} $count ways');
632 emit(['SPLIT', 'table', a[0], 'EVEN', count, 'due', money(due)]);
633 for (var payer = 1; payer <= count; payer++) {
634 emit(['PAYER', payer, money(due ~/ count + (payer <= due % count ? 1 : 0))]);
635 }
636 return;
637 }
638 if (check.paid != 0) fail('table ${a[0]} has payments');
639 final subtotals = seatSubtotals(check);
640 final discounts = allocate(subtotals, check.discount);
641 final after = <int, int>{
642 for (final entry in subtotals.entries)
643 entry.key: entry.value - discounts[entry.key]!,
644 };
645 final services = check.size >= 6
646 ? allocate(after, 18)
647 : <int, int>{for (final seat in after.keys) seat: 0};
648 final taxes = allocate(after, 8);
649 emit(['SPLIT', 'table', a[0], 'BYSEAT', 'total', money(check.amounts()[4])]);
650 final seats = subtotals.keys.toList()..sort();
651 for (final seat in seats) {
652 final total = after[seat]! + services[seat]! + taxes[seat]!;
653 emit(['SEAT', seat, 'subtotal', money(subtotals[seat]!),
654 'discount', money(discounts[seat]!), 'service', money(services[seat]!),
655 'tax', money(taxes[seat]!), 'total', money(total)]);
656 }
657 }

659 void cmdPay(List<String> a) {
660 final check = table(a[0], true).check!;
661 if (!payments.containsKey(a[1])) fail('unknown method ${a[1]}');
662 allServed(check);
663 final due = check.amounts()[5];
664 if (due == 0) fail('nothing to pay for table ${a[0]}');
665 final value = cents(a[2]);
666 final applied = value < due ? value : due;
667 check.paid += applied;
668 payments[a[1]] = payments[a[1]]! + applied;
669 if (value < due) {
670 emit(['PAY', 'table', a[0], a[1], a[2], 'remaining', money(due - value)]);
671 } else {
672 final excess = value - due;
673 emit(['PAY', 'table', a[0], a[1], a[2],
674 a[1] == 'CASH' ? 'change' : 'tip', money(excess)]);
675 if (a[1] == 'CARD') {
676 tips += excess;
677 check.server.tips += excess;
678 }
679 closeCheck(check);
680 }
681 }

698 if (order.state == 'COOKING') ...['until', clock(order.ready)],
699 if (order.comp != null) ...['COMP', order.comp!]]);
700 }
701 }

703 void cmdKitchen(List<String> a) {
704 emit(['KITCHEN', clock(time)]);
705 for (final name in stationNames) {
706 final station = stations[name]!;
707 final cooking = station.cooking.toList()
708 ..sort((a, b) {
709 final order = a.ready.compareTo(b.ready);
710 return order != 0 ? order : a.number.compareTo(b.number);
711 });
712 final cookingText = cooking.isEmpty
713 ? '-'
714 : cooking.map((line) => '${line.tag}@${clock(line.ready)}').join(' ');
715 final queueText = station.queue.isEmpty
716 ? '-'
717 : station.queue.map((line) => line.tag).join(' ');
718 emit([name, station.up ? 'up' : 'down', 'cooking', cookingText,
719 'queue', queueText]);
720 }
721 }

723 void cmdFloor(List<String> a) {
724 final occupied = tables.values.where((table) => table.check != null).length;
725 emit(['FLOOR', clock(time), 'free', tables.length - occupied,
726 'seated', occupied, 'waiting', waiting.length]);
727 final numbers = tables.keys.toList()..sort();
728 for (final number in numbers) {
729 final selected = tables[number]!;
730 final check = selected.check;
731 if (check == null) {
732 emit(['table', number, 'seats', selected.seats, 'free']);
733 } else {
734 emit(['table', number, 'seats', selected.seats, check.party,
735 'guests', check.size, 'server', check.server.name,
736 'since', clock(check.since), 'items', check.active.length,
737 'total', money(check.amounts()[4]),
738 if (check.paid != 0) ...['paid', money(check.paid)]]);
739 }
740 }
741 for (var i = 0; i < waiting.length; i++) {
742 final entry = waiting[i];
743 emit(['WAITING', i + 1, entry.party, entry.size,
744 'since', clock(entry.joined)]);
745 }
746 }

748 void cmdReport(List<String> a) {
749 emit(['REPORT', clock(time)]);
750 emit(['COVERS', guests, 'parties', parties]);
751 emit(['TURNS', closed, 'avg_stay',
752 closed == 0 ? '-' : (stayTotal ~/ closed).toString()]);
753 emit(['SALES', for (var i = 0; i < courseNames.length; i++) ...[
754 courseNames[i], money(sales[i]),
755 ]]);
756 emit(['TOTALS', 'subtotal', money(totals[0]), 'discount', money(totals[1]),
757 'service', money(totals[2]), 'tax', money(totals[3]),
758 'total', money(totals[4])]);
759 emit(['PAYMENTS', 'cash', money(payments['CASH']!),
760 'card', money(payments['CARD']!), 'tips', money(tips)]);
761 emit(['COMPS', comps, money(compAmount), 'VOIDS', voids]);
762 final top = served.keys.toList()
763 ..sort((a, b) {
764 final order = served[b]!.compareTo(served[a]!);
765 return order != 0 ? order : a.compareTo(b);
766 });
767 for (var i = 0; i < top.length && i < 3; i++) {
768 emit(['TOP', i + 1, top[i], served[top[i]]!]);
769 }
770 for (final name in stationNames) {
771 final station = stations[name]!;
772 emit([name, 'tickets', station.tickets, 'avg',
773 station.tickets == 0 ? '-' : (station.ticketTotal ~/ station.tickets).toString(),
774 'max', station.tickets == 0 ? '-' : station.ticketMax.toString(),
775 'remakes', station.remakes]);
776 }
777 for (final server in servers) {
778 emit(['SERVER', server.name, 'tables', server.load, 'parties',
779 server.parties, 'sales', money(server.sales), 'tips', money(server.tips)]);
780 }
781 }

783 void execute(String input) {
784 final words = input.trim().split(' ');
785 final command = words.first;
786 final args = words.skip(1).toList();
787 try {
788 switch (command) {
789 case 'TABLE': cmdTable(args); break;
790 case 'STAFF': cmdStaff(args); break;
791 case 'SEAT': cmdSeat(args); break;
792 case 'CANCEL': cmdCancel(args); break;
793 case 'MOVE': cmdMove(args); break;
794 case 'MENU': cmdMenu(args); break;
795 case 'STOCK': cmdStock(args); break;
796 case 'HAPPY': cmdHappy(args); break;
797 case 'ITEMS': cmdItems(args); break;
798 case 'ORDER': cmdOrder(args); break;
799 case 'FIRE': cmdFire(args); break;
800 case 'TICK': cmdTick(args); break;
801 case 'SERVE': cmdServe(args); break;
802 case 'RUSH': cmdRush(args); break;
803 case 'REMAKE': cmdRemake(args); break;
804 case 'VOID': cmdVoid(args); break;
805 case 'STATION': cmdStation(args); break;
806 case 'COMP': cmdComp(args); break;
807 case 'TRANSFER': cmdTransfer(args); break;
808 case 'DISCOUNT': cmdDiscount(args); break;
809 case 'BILL': cmdBill(args); break;
810 case 'SPLIT': cmdSplit(args); break;
811 case 'PAY': cmdPay(args); break;
812 case 'CLOSE': cmdClose(args); break;
813 case 'STATUS': cmdStatus(args); break;
814 case 'KITCHEN': cmdKitchen(args); break;
815 case 'FLOOR': cmdFloor(args); break;
816 case 'REPORT': cmdReport(args); break;
817 }
818 } on CommandError catch (error) {
819 emit(['ERROR', error.message]);
820 }
821 }
822
823 String result() => output.isEmpty ? '' : '${output.join('\n')}\n';
824}

831 }
832 stdout.write(restaurant.result());
