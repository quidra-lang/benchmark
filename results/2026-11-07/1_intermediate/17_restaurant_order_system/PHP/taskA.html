3declare(strict_types=1);

5const STATIONS = ['GRILL', 'FRY', 'COLD', 'BAR'];
6const COURSES = ['DRINK', 'STARTER', 'MAIN', 'DESSERT'];
7
8final class CommandError extends RuntimeException {}
9
10function cents(string $value): int
11{
12 return (int) str_replace('.', '', $value);
13}
14
15function money(int $value): string
16{
17 return intdiv($value, 100) . '.' . str_pad((string) ($value % 100), 2, '0', STR_PAD_LEFT);
18}
19
20function clockTime(int $time): string
21{
22 $minutes = 660 + $time;
23 return sprintf('%02d:%02d', intdiv($minutes, 60), $minutes % 60);
24}
25
26function rounded(int $value, int $rate): int
27{
28 return intdiv($value * $rate + 50, 100);
29}
30
31function lineAmount(object $line): int
32{
33 return $line->reason === null ? $line->price * $line->quantity : 0;
34}
35
36function activeLines(object $check): array
37{
38 return array_values(array_filter(
39 $check->lines,
40 static fn(object $line): bool => $line->state !== 'VOID'
41 ));
42}

44final class Restaurant
45{
46 public array $output = [];
47
48 private int $now = 0;
49 private array $tables = [];
50 private array $servers = [];
51 private array $waiting = [];
52 private array $menu = [];
53 private array $lines = [];
54 private array $stations = [];
55
56 private int $happyRate = 0;
57 private int $happyEnd = 0;
58 private int $covers = 0;
59 private int $parties = 0;
60 private int $turns = 0;
61 private int $stay = 0;
62 private array $sales = ['DRINK' => 0, 'STARTER' => 0, 'MAIN' => 0, 'DESSERT' => 0];
63 private array $totals = [
64 'subtotal' => 0, 'discount' => 0, 'service' => 0, 'tax' => 0, 'total' => 0,
65 ];
66 private array $payments = ['CASH' => 0, 'CARD' => 0];
67 private int $tips = 0;
68 private int $comps = 0;

73 public function __construct()
74 {
75 foreach (STATIONS as $name) {
76 $this->stations[$name] = (object) [
77 'name' => $name, 'capacity' => $name === 'BAR' ? 2 : 1,
78 'up' => true, 'queue' => [], 'cooking' => [],
79 'tickets' => [], 'remakes' => 0,
80 ];
81 }

84 private function emit(string|int ...$parts): void
85 {
86 $this->output[] = implode(' ', $parts);

89 private function table(string $text): object
90 {
91 $number = (int) $text;
92 if (!isset($this->tables[$number])) {
93 throw new CommandError('unknown table ' . $text);
94 }
95 return $this->tables[$number];
96 }
97
98 private function check(string $text): object
99 {
100 $table = $this->table($text);
101 if ($table->check === null) {
102 throw new CommandError('table ' . $text . ' is free');
103 }
104 return $table->check;
105 }
106
107 private function unlocked(object $check): void
108 {
109 if ($check->paid > 0) {
110 throw new CommandError('check for table ' . $check->table->number . ' is locked');
111 }
112 }
113
114 private function line(string $text): object
115 {
116 $number = (int) $text;
117 if ($number < 1 || $number > count($this->lines)) {
118 throw new CommandError('unknown line ' . $text);
119 }
120 return $this->lines[$number - 1];
121 }
122
123 private function modifiable(object $line): void
124 {
125 if ($line->state === 'VOID') {
126 throw new CommandError('#' . $line->id . ' is void');
127 }
128 if ($line->check->closed) {
129 throw new CommandError('#' . $line->id . ' is on a closed check');
130 }
131 $this->unlocked($line->check);
132 }
133
134 private function unserved(object $check): void
135 {
136 $count = 0;
137 foreach (activeLines($check) as $line) {
138 if ($line->state !== 'SERVED') {
139 ++$count;
140 }
141 }
142 if ($count > 0) {
143 throw new CommandError(
144 'table ' . $check->table->number . ' has ' . $count . ' unserved items'
145 );
146 }
147 }

149 private function station(string $name): object
150 {
151 if (!isset($this->stations[$name])) {
152 throw new CommandError('unknown station ' . $name);
153 }
154 return $this->stations[$name];
155 }
156
157 private function course(string $name): void
158 {
159 if (!in_array($name, COURSES, true)) {
160 throw new CommandError('unknown course ' . $name);
161 }
162 }
163
164 private function item(string $name): object

166 if (!isset($this->menu[$name])) {
167 throw new CommandError('unknown item ' . $name);
168 }
169 return $this->menu[$name];

172 private function load(object $server): int
173 {
174 $count = 0;
175 foreach ($this->tables as $table) {
176 if ($table->check !== null && $table->check->server === $server) {
177 ++$count;
178 }
179 }
180 return $count;
181 }
182
183 private function selectTable(int $size): ?object
184 {
185 $best = null;
186 foreach ($this->tables as $table) {
187 if ($table->check !== null || $table->seats < $size) {
188 continue;
189 }
190 if ($best === null || $table->seats < $best->seats ||
191 ($table->seats === $best->seats && $table->number < $best->number)) {
192 $best = $table;
193 }
194 }
195 return $best;
196 }
197
198 private function seat(string $party, int $size, object $table): object
199 {
200 $server = $this->servers[0];
201 foreach ($this->servers as $candidate) {
202 if ($this->load($candidate) < $this->load($server)) {
203 $server = $candidate;
204 }
205 }
206 $check = (object) [
207 'party' => $party, 'size' => $size, 'table' => $table,
208 'server' => $server, 'seated' => $this->now, 'lines' => [],
209 'paid' => 0, 'discount' => 0, 'closed' => false,
210 ];
211 $table->check = $check;
212 ++$server->parties;
213 $this->covers += $size;
214 ++$this->parties;
215 return $check;
216 }
217
218 private function processWaiting(): void
219 {
220 $remaining = [];
221 foreach ($this->waiting as $entry) {
222 $table = $this->selectTable($entry->size);
223 if ($table === null) {
224 $remaining[] = $entry;
225 continue;
226 }
227 $check = $this->seat($entry->party, $entry->size, $table);
228 $this->emit(
229 'SEATED', $entry->party, $entry->size, 'table', $table->number,
230 'server', $check->server->name, 'waited', $this->now - $entry->joined
231 );
232 }
233 $this->waiting = $remaining;
234 }
235
236 private function amounts(object $check): array
237 {
238 $subtotal = 0;
239 foreach (activeLines($check) as $line) {
240 $subtotal += lineAmount($line);
241 }
242 $discount = rounded($subtotal, $check->discount);
243 $after = $subtotal - $discount;
244 $service = $check->size >= 6 ? rounded($after, 18) : 0;
245 $tax = rounded($after, 8);
246 $total = $after + $service + $tax;
247 return compact('subtotal', 'discount', 'service', 'tax', 'total') +
248 ['due' => $total - $check->paid];
249 }
250
251 private function close(object $check): void
252 {
253 $amounts = $this->amounts($check);
254 ++$this->turns;
255 $this->stay += $this->now - $check->seated;
256 foreach (activeLines($check) as $line) {
257 $this->sales[$line->course] += lineAmount($line);
258 }
259 foreach (['subtotal', 'discount', 'service', 'tax', 'total'] as $key) {
260 $this->totals[$key] += $amounts[$key];
261 }
262 $check->server->sales += $amounts['total'];
263 $check->closed = true;
264 $check->table->check = null;
265 $this->emit(
266 'FREE', 'table', $check->table->number, $check->party,
267 'stayed', $this->now - $check->seated
268 );
269 $this->processWaiting();
270 }
271
272 private function start(object $station, int $time): void
273 {
274 while ($station->up && $station->queue !== [] &&
275 count($station->cooking) < $station->capacity) {
276 $line = array_shift($station->queue);
277 $line->state = 'COOKING';
278 $line->ready = $time + $line->prep;
279 $line->started = true;
280 $station->cooking[] = $line;
281 }
282 }
283
284 private function tick(int $minutes): void
285 {
286 $end = $this->now + $minutes;
287 $this->emit('TICK', clockTime($end));
288 while (true) {
289 $earliest = null;
290 foreach (STATIONS as $name) {
291 foreach ($this->stations[$name]->cooking as $line) {
292 if ($earliest === null || $line->ready < $earliest) {
293 $earliest = $line->ready;
294 }
295 }
296 }
297 if ($earliest === null || $earliest > $end) {
298 break;
299 }
300 foreach (STATIONS as $name) {
301 $station = $this->stations[$name];
302 $finished = [];
303 $remaining = [];
304 foreach ($station->cooking as $line) {
305 if ($line->ready === $earliest) {
306 $finished[] = $line;
307 } else {
308 $remaining[] = $line;
309 }
310 }
311 $station->cooking = $remaining;
312 usort($finished, static fn(object $a, object $b): int => $a->id <=> $b->id);
313 foreach ($finished as $line) {
314 $line->state = 'READY';
315 $station->tickets[] = $earliest - $line->fire;
316 $this->emit(
317 'READY', clockTime($earliest), $name, '#' . $line->id,
318 $line->item, 'x' . $line->quantity, 'table',
319 $line->check->table->number, 'seat', $line->seat
320 );
321 }
322 }
323 foreach (STATIONS as $name) {
324 $this->start($this->stations[$name], $earliest);
325 }
326 }
327 $this->now = $end;
328 }

330 private function seatSubtotals(object $check): array
331 {
332 $subtotals = [];
333 foreach (activeLines($check) as $line) {
334 $subtotals[$line->seat] = ($subtotals[$line->seat] ?? 0) + lineAmount($line);
335 }
336 ksort($subtotals, SORT_NUMERIC);
337 return $subtotals;
338 }

340 private function allocate(array $values, int $rate): array
341 {
342 $allocated = [];
343 foreach ($values as $seat => $subtotal) {
344 $allocated[$seat] = intdiv($subtotal * $rate, 100);
345 }
346 $left = rounded(array_sum($values), $rate) - array_sum($allocated);
347 $order = array_keys($values);
348 usort($order, static function (int $a, int $b) use ($values, $rate): int {
349 $first = ($values[$a] * $rate) % 100;
350 $second = ($values[$b] * $rate) % 100;
351 return ($second <=> $first) ?: ($a <=> $b);
352 });
353 for ($i = 0; $i < $left; ++$i) {
354 ++$allocated[$order[$i]];
355 }
356 return $allocated;
357 }

359 private function bill(object $check): void
360 {
361 $this->emit('BILL', 'table', $check->table->number, $check->party, 'guests', $check->size);
362 foreach ($this->seatSubtotals($check) as $seat => $subtotal) {
363 $this->emit('SEAT', (int) $seat, money($subtotal));
364 foreach (activeLines($check) as $line) {
365 if ($line->seat !== $seat) {
366 continue;
367 }
368 $parts = ['#' . $line->id, $line->item, 'x' . $line->quantity, money(lineAmount($line))];
369 if ($line->reason !== null) {
370 $parts[] = 'COMP';
371 $parts[] = $line->reason;
372 }
373 $this->emit(...$parts);
374 }
375 }
376 $amounts = $this->amounts($check);
377 $this->emit('SUBTOTAL', money($amounts['subtotal']));
378 if ($check->discount !== 0) {
379 $this->emit('DISCOUNT', $check->discount . '%', money($amounts['discount']));
380 }
381 if ($check->size >= 6) {
382 $this->emit('SERVICE', money($amounts['service']));
383 }
384 $this->emit('TAX', money($amounts['tax']));
385 $this->emit('TOTAL', money($amounts['total']));
386 if ($check->paid > 0) {
387 $this->emit('PAID', money($check->paid));
388 $this->emit('DUE', money($amounts['due']));
389 }
390 }

392 private function report(): void
393 {
394 $this->emit('REPORT', clockTime($this->now));
395 $this->emit('COVERS', $this->covers, 'parties', $this->parties);
396 $average = $this->turns ? intdiv($this->stay, $this->turns) : '-';
397 $this->emit('TURNS', $this->turns, 'avg_stay', $average);
398 $parts = ['SALES'];
399 foreach (COURSES as $course) {
400 $parts[] = $course;
401 $parts[] = money($this->sales[$course]);
402 }
403 $this->emit(...$parts);
404 $parts = ['TOTALS'];
405 foreach (['subtotal', 'discount', 'service', 'tax', 'total'] as $field) {
406 $parts[] = $field;
407 $parts[] = money($this->totals[$field]);
408 }
409 $this->emit(...$parts);
410 $this->emit(
411 'PAYMENTS', 'cash', money($this->payments['CASH']),
412 'card', money($this->payments['CARD']), 'tips', money($this->tips)
413 );
414 $this->emit('COMPS', $this->comps, money($this->compAmount), 'VOIDS', $this->voids);
415 $names = array_keys(array_filter($this->served, static fn(int $count): bool => $count > 0));
416 usort($names, function (string $a, string $b): int {
417 return ($this->served[$b] <=> $this->served[$a]) ?: strcmp($a, $b);
418 });
419 foreach (array_slice($names, 0, 3) as $index => $name) {
420 $this->emit('TOP', $index + 1, $name, $this->served[$name]);
421 }
422 foreach (STATIONS as $name) {
423 $station = $this->stations[$name];
424 $count = count($station->tickets);
425 $this->emit(
426 $name, 'tickets', $count, 'avg', $count ? intdiv(array_sum($station->tickets), $count) : '-',
427 'max', $count ? max($station->tickets) : '-', 'remakes', $station->remakes
428 );
429 }
430 foreach ($this->servers as $server) {
431 $this->emit(
432 'SERVER', $server->name, 'tables', $this->load($server),
433 'parties', $server->parties, 'sales', money($server->sales),
434 'tips', money($server->tips)
435 );
436 }
437 }

439 private function command(string $command, array $a): void
440 {
441 switch ($command) {
442 case 'TABLE':
443 $number = (int) $a[0];
444 if (isset($this->tables[$number])) {
445 throw new CommandError('table ' . $a[0] . ' already exists');
446 }
447 $this->tables[$number] = (object) [
448 'number' => $number, 'seats' => (int) $a[1], 'check' => null,
449 ];
450 $this->emit('TABLE', $number, 'seats', (int) $a[1]);
451 $this->processWaiting();
452 break;
453
454 case 'STAFF':
455 foreach ($this->servers as $server) {
456 if ($server->name === $a[0]) {
457 throw new CommandError($a[0] . ' is already on duty');
458 }
459 }
460 $this->servers[] = (object) [
461 'name' => $a[0], 'parties' => 0, 'sales' => 0, 'tips' => 0,
462 ];
463 $this->emit('STAFF', $a[0]);
464 break;
465
466 case 'SEAT':
467 foreach ($this->tables as $table) {
468 if ($table->check !== null && $table->check->party === $a[0]) {
469 throw new CommandError(
470 $a[0] . ' is already seated at table ' . $table->number
471 );
472 }
473 }
474 foreach ($this->waiting as $entry) {
475 if ($entry->party === $a[0]) {
476 throw new CommandError($a[0] . ' is already waiting');
477 }
478 }
479 $size = (int) $a[1];
480 $accommodates = false;
481 foreach ($this->tables as $table) {
482 if ($table->seats >= $size) {
483 $accommodates = true;
484 break;
485 }
486 }
487 if (!$accommodates) {
488 throw new CommandError('no table for party of ' . $size);
489 }
490 if ($this->servers === []) {
491 throw new CommandError('no server on duty');
492 }
493 $table = $this->selectTable($size);
494 if ($table === null) {
495 $this->waiting[] = (object) [
496 'party' => $a[0], 'size' => $size, 'joined' => $this->now,
497 ];
498 $this->emit('WAIT', $a[0], $size, 'position', count($this->waiting));
499 } else {
500 $check = $this->seat($a[0], $size, $table);
501 $this->emit(
502 'SEAT', $a[0], $size, 'table', $table->number,
503 'server', $check->server->name
504 );
505 }
506 break;
507
508 case 'CANCEL':
509 foreach ($this->waiting as $i => $entry) {
510 if ($entry->party === $a[0]) {
511 array_splice($this->waiting, $i, 1);
512 $this->emit(
513 'CANCEL', $entry->party, $entry->size,
514 'waited', $this->now - $entry->joined
515 );
516 return;
517 }
518 }
519 throw new CommandError($a[0] . ' is not waiting');
520
521 case 'MOVE':
522 $source = $this->table($a[0]);
523 $destination = $this->table($a[1]);
524 if ($source->check === null) {
525 throw new CommandError('table ' . $a[0] . ' is free');
526 }
527 if ($destination->check !== null) {
528 throw new CommandError('table ' . $a[1] . ' is occupied');
529 }
530 $check = $source->check;
531 if ($destination->seats < $check->size) {
532 throw new CommandError('table ' . $a[1] . ' is too small for ' . $check->party);
533 }
534 $source->check = null;
535 $destination->check = $check;
536 $check->table = $destination;
537 $this->emit(
538 'MOVE', $check->party, 'table', $source->number, '->', $destination->number
539 );
540 $this->processWaiting();
541 break;
542
543 case 'MENU':
544 $this->station($a[1]);
545 $this->course($a[4]);
546 $old = $this->menu[$a[0]] ?? null;
547 $this->menu[$a[0]] = (object) [
548 'name' => $a[0], 'station' => $a[1], 'prep' => (int) $a[2],
549 'price' => cents($a[3]), 'course' => $a[4],
550 'stock' => $old?->stock,
551 ];
552 $this->emit(
553 'MENU', $old === null ? 'ADD' : 'UPDATE', $a[0], $a[1],
554 (int) $a[2], money(cents($a[3])), $a[4]
555 );
556 break;
557
558 case 'STOCK':
559 $item = $this->item($a[0]);
560 $item->stock = (int) $a[1];
561 $this->emit('STOCK', $item->name, $item->stock);
562 break;
563
564 case 'HAPPY':
565 $this->happyRate = (int) $a[0];
566 $this->happyEnd = $this->now + (int) $a[1];
567 $this->emit(
568 'HAPPY', $this->happyRate . '%', 'until', clockTime($this->happyEnd)
569 );
570 break;
571
572 case 'ITEMS':
573 $items = array_values($this->menu);
574 usort($items, static function (object $a, object $b): int {
575 return (array_search($a->course, COURSES, true) <=>
576 array_search($b->course, COURSES, true)) ?: strcmp($a->name, $b->name);
577 });
578 $this->emit('ITEMS', count($items));
579 foreach ($items as $item) {
580 $this->emit(
581 $item->course, $item->name, $item->station, $item->prep,
582 money($item->price), $item->stock === null ? '-' : $item->stock
583 );
584 }
585 break;
586
587 case 'ORDER':
588 $check = $this->check($a[0]);
589 $this->unlocked($check);
590 $seat = (int) $a[1];
591 if ($seat > $check->size) {
592 throw new CommandError(
593 'seat ' . $seat . ' out of range for table ' . $a[0]
594 );
595 }
596 $item = $this->item($a[2]);
597 $quantity = (int) $a[3];
598 if ($item->stock !== null && $item->stock < $quantity) {
599 throw new CommandError('only ' . $item->stock . ' ' . $item->name . ' left');
600 }
601 $happy = $item->course === 'DRINK' && $this->now < $this->happyEnd;
602 $price = $item->price - ($happy ? rounded($item->price, $this->happyRate) : 0);
603 $line = (object) [
604 'id' => count($this->lines) + 1, 'check' => $check,
605 'seat' => $seat, 'item' => $item->name, 'quantity' => $quantity,
606 'station' => $item->station, 'prep' => $item->prep,
607 'price' => $price, 'course' => $item->course,
608 'state' => 'ORDERED', 'reason' => null,
609 'fire' => 0, 'ready' => 0, 'started' => false,
610 ];
611 $this->lines[] = $line;
612 $check->lines[] = $line;
613 $parts = [
614 'ORDER', '#' . $line->id, 'table', $check->table->number,
615 'seat', $seat, $item->name, 'x' . $quantity, money(lineAmount($line)),
616 ];
617 if ($happy) {
618 $parts[] = 'HAPPY';
619 }
620 $this->emit(...$parts);
621 if ($item->stock !== null) {
622 $item->stock -= $quantity;
623 if ($item->stock === 0) {
624 $this->emit('SOLDOUT', $item->name);
625 }
626 }
627 break;
628
629 case 'FIRE':
630 $check = $this->check($a[0]);
631 $this->course($a[1]);
632 $lines = [];
633 foreach ($check->lines as $line) {
634 if ($line->course === $a[1] && $line->state === 'ORDERED') {
635 $lines[] = $line;
636 }
637 }
638 if ($lines === []) {
639 throw new CommandError(
640 'nothing to fire for ' . $a[1] . ' at table ' . $a[0]
641 );
642 }
643 $priorCourses = ['STARTER', 'MAIN', 'DESSERT'];
644 $position = array_search($a[1], $priorCourses, true);
645 if ($position !== false) {
646 foreach (array_slice($priorCourses, 0, $position) as $earlier) {
647 foreach ($check->lines as $line) {
648 if ($line->course === $earlier && $line->state === 'ORDERED') {
649 throw new CommandError('fire ' . $earlier . ' first');
650 }
651 }
652 }
653 }
654 $parts = ['FIRE', 'table', $check->table->number, $a[1]];
655 foreach ($lines as $line) {
656 $line->state = 'QUEUED';
657 $line->fire = $this->now;
658 $this->stations[$line->station]->queue[] = $line;
659 $parts[] = '#' . $line->id;
660 }
661 $this->emit(...$parts);
662 foreach (STATIONS as $station) {
663 $this->start($this->stations[$station], $this->now);
664 }
665 break;
666
667 case 'TICK':
668 $this->tick((int) $a[0]);
669 break;
670
671 case 'SERVE':
672 $check = $this->check($a[0]);
673 $lines = array_values(array_filter(
674 $check->lines,
675 static fn(object $line): bool => $line->state === 'READY'
676 ));
677 if ($lines === []) {
678 throw new CommandError('nothing ready for table ' . $a[0]);
679 }
680 $parts = ['SERVE', 'table', $check->table->number];
681 foreach ($lines as $line) {
682 $line->state = 'SERVED';
683 $this->served[$line->item] = ($this->served[$line->item] ?? 0) +
684 $line->quantity;
685 $parts[] = '#' . $line->id;
686 }
687 $this->emit(...$parts);
688 break;
689
690 case 'RUSH':
691 $line = $this->line($a[0]);
692 if ($line->state !== 'QUEUED') {
693 throw new CommandError('#' . $line->id . ' is not queued');
694 }
695 $station = $this->stations[$line->station];
696 $station->queue = array_merge(
697 [$line],
698 array_values(array_filter(
699 $station->queue,
700 static fn(object $candidate): bool => $candidate !== $line
701 ))
702 );
703 $this->emit('RUSH', '#' . $line->id, $line->station);
704 break;
705
706 case 'REMAKE':
707 $line = $this->line($a[0]);
708 if ($line->check->closed) {
709 throw new CommandError('#' . $line->id . ' is on a closed check');
710 }
711 if ($line->state !== 'READY' && $line->state !== 'SERVED') {
712 throw new CommandError('#' . $line->id . ' is not ready or served');
713 }
714 $line->state = 'QUEUED';
715 $line->fire = $this->now;
716 $station = $this->stations[$line->station];
717 array_unshift($station->queue, $line);
718 ++$station->remakes;
719 $this->emit('REMAKE', '#' . $line->id, $line->item, $line->station);
720 $this->start($station, $this->now);
721 break;
722
723 case 'VOID':
724 $line = $this->line($a[0]);
725 if ($line->state === 'VOID') {
726 throw new CommandError('#' . $line->id . ' is already void');
727 }
728 if ($line->started) {
729 throw new CommandError('#' . $line->id . ' has already started');
730 }
731 if ($line->state === 'QUEUED') {
732 $station = $this->stations[$line->station];
733 $station->queue = array_values(array_filter(
734 $station->queue,
735 static fn(object $candidate): bool => $candidate !== $line
736 ));
737 }
738 $line->state = 'VOID';
739 $item = $this->menu[$line->item];
740 if ($item->stock !== null) {
741 $item->stock += $line->quantity;
742 }
743 ++$this->voids;
744 $this->emit('VOID', '#' . $line->id, $line->item, 'x' . $line->quantity);
745 break;
746
747 case 'STATION':
748 $station = $this->station($a[0]);
749 if ($a[1] === 'DOWN' && !$station->up) {
750 throw new CommandError('station ' . $station->name . ' is already down');
751 }
752 if ($a[1] === 'UP' && $station->up) {
753 throw new CommandError('station ' . $station->name . ' is not down');
754 }
755 $station->up = $a[1] === 'UP';
756 if ($station->up) {
757 $this->start($station, $this->now);
758 }
759 $this->emit('STATION', $station->name, $a[1]);
760 break;
761
762 case 'COMP':
763 case 'TRANSFER':
764 $line = $this->line($a[0]);
765 $this->modifiable($line);
766 if ($command === 'COMP') {
767 if ($line->reason !== null) {
768 throw new CommandError('#' . $line->id . ' is already comped');
769 }
770 $value = lineAmount($line);
771 $this->emit('COMP', '#' . $line->id, $line->item, money($value), $a[1]);
772 ++$this->comps;
773 $this->compAmount += $value;
774 $line->reason = $a[1];
775 } else {
776 $seat = (int) $a[1];
777 if ($seat > $line->check->size) {
778 throw new CommandError(
779 'seat ' . $seat . ' out of range for table ' . $line->check->table->number
780 );
781 }
782 if ($seat === $line->seat) {
783 throw new CommandError('#' . $line->id . ' is already on seat ' . $seat);
784 }
785 $this->emit('TRANSFER', '#' . $line->id, 'seat', $line->seat, '->', $seat);
786 $line->seat = $seat;
787 }
788 break;
789
790 case 'DISCOUNT':
791 $check = $this->check($a[0]);
792 $this->unlocked($check);
793 $check->discount = (int) $a[1];
794 $this->emit('DISCOUNT', 'table', $check->table->number, $check->discount . '%');
795 break;
796
797 case 'BILL':
798 $this->bill($this->check($a[0]));
799 break;
800
801 case 'SPLIT':
802 $check = $this->check($a[0]);
803 $amounts = $this->amounts($check);
804 if ($amounts['due'] === 0) {
805 throw new CommandError('nothing to split for table ' . $a[0]);
806 }
807 if ($a[1] === 'EVEN') {
808 $count = (int) $a[2];
809 if ($count > $check->size) {
810 throw new CommandError(
811 'cannot split table ' . $a[0] . ' ' . $count . ' ways'
812 );
813 }
814 $this->emit(
815 'SPLIT', 'table', $check->table->number, 'EVEN', $count,
816 'due', money($amounts['due'])
817 );
818 $quotient = intdiv($amounts['due'], $count);
819 $remainder = $amounts['due'] % $count;
820 for ($i = 1; $i <= $count; ++$i) {
821 $this->emit('PAYER', $i, money($quotient + (int) ($i <= $remainder)));
822 }
823 } else {
824 if ($check->paid > 0) {
825 throw new CommandError('table ' . $a[0] . ' has payments');
826 }
827 $values = $this->seatSubtotals($check);
828 $discounts = $this->allocate($values, $check->discount);
829 $after = [];
830 foreach ($values as $seat => $value) {
831 $after[$seat] = $value - $discounts[$seat];
832 }
833 $services = $this->allocate($after, $check->size >= 6 ? 18 : 0);
834 $taxes = $this->allocate($after, 8);
835 $this->emit(
836 'SPLIT', 'table', $check->table->number, 'BYSEAT',
837 'total', money($amounts['total'])
838 );
839 foreach ($values as $seat => $value) {
840 $this->emit(
841 'SEAT', (int) $seat, 'subtotal', money($value),
842 'discount', money($discounts[$seat]),
843 'service', money($services[$seat]),
844 'tax', money($taxes[$seat]),
845 'total', money($after[$seat] + $services[$seat] + $taxes[$seat])
846 );
847 }
848 }
849 break;
850
851 case 'PAY':
852 $check = $this->check($a[0]);
853 if (!in_array($a[1], ['CASH', 'CARD'], true)) {
854 throw new CommandError('unknown method ' . $a[1]);
855 }
856 $this->unserved($check);
857 $due = $this->amounts($check)['due'];
858 if ($due === 0) {
859 throw new CommandError('nothing to pay for table ' . $a[0]);
860 }
861 $value = cents($a[2]);
862 $applied = min($value, $due);
863 $check->paid += $applied;
864 $this->payments[$a[1]] += $applied;
865 if ($value < $due) {
866 $this->emit(
867 'PAY', 'table', $check->table->number, $a[1],
868 money($value), 'remaining', money($due - $value)
869 );
870 } else {
871 $this->emit(
872 'PAY', 'table', $check->table->number, $a[1],
873 money($value), $a[1] === 'CASH' ? 'change' : 'tip',
874 money($value - $due)
875 );
876 if ($a[1] === 'CARD') {
877 $this->tips += $value - $due;
878 $check->server->tips += $value - $due;
879 }
880 $this->close($check);
881 }
882 break;
883
884 case 'CLOSE':
885 $check = $this->check($a[0]);
886 $this->unserved($check);
887 $due = $this->amounts($check)['due'];
888 if ($due !== 0) {
889 throw new CommandError(
890 'table ' . $a[0] . ' still owes ' . money($due)
891 );
892 }
893 $this->close($check);
894 break;
895
896 case 'STATUS':
897 $check = $this->check($a[0]);
898 $this->emit(
899 'STATUS', 'table', $check->table->number, $check->party,
900 'guests', $check->size, 'since', clockTime($check->seated)
901 );
902 foreach ($check->lines as $line) {
903 $parts = [
904 '#' . $line->id, 'seat', $line->seat, $line->item,
905 'x' . $line->quantity, $line->course, $line->state,
906 ];
907 if ($line->state === 'COOKING') {
908 $parts[] = 'until';
909 $parts[] = clockTime($line->ready);
910 }
911 if ($line->reason !== null) {
912 $parts[] = 'COMP';
913 $parts[] = $line->reason;
914 }
915 $this->emit(...$parts);
916 }
917 break;
918
919 case 'KITCHEN':
920 $this->emit('KITCHEN', clockTime($this->now));
921 foreach (STATIONS as $name) {
922 $station = $this->stations[$name];
923 $cooking = $station->cooking;
924 usort($cooking, static fn(object $a, object $b): int =>
925 ($a->ready <=> $b->ready) ?: ($a->id <=> $b->id)
926 );
927 $ready = [];
928 foreach ($cooking as $line) {
929 $ready[] = '#' . $line->id . '@' . clockTime($line->ready);
930 }
931 $queue = [];
932 foreach ($station->queue as $line) {
933 $queue[] = '#' . $line->id;
934 }
935 $this->emit(
936 $name, $station->up ? 'up' : 'down', 'cooking',
937 $ready ? implode(' ', $ready) : '-', 'queue',
938 $queue ? implode(' ', $queue) : '-'
939 );
940 }
941 break;
942
943 case 'FLOOR':
944 $tables = array_values($this->tables);
945 usort($tables, static fn(object $a, object $b): int => $a->number <=> $b->number);
946 $free = count(array_filter(
947 $tables, static fn(object $table): bool => $table->check === null
948 ));
949 $this->emit(
950 'FLOOR', clockTime($this->now), 'free', $free,
951 'seated', count($tables) - $free, 'waiting', count($this->waiting)
952 );
953 foreach ($tables as $table) {
954 $check = $table->check;
955 if ($check === null) {
956 $this->emit('table', $table->number, 'seats', $table->seats, 'free');
957 } else {
958 $parts = [
959 'table', $table->number, 'seats', $table->seats,
960 $check->party, 'guests', $check->size, 'server',
961 $check->server->name, 'since', clockTime($check->seated),
962 'items', count(activeLines($check)), 'total',
963 money($this->amounts($check)['total']),
964 ];
965 if ($check->paid > 0) {
966 $parts[] = 'paid';
967 $parts[] = money($check->paid);
968 }
969 $this->emit(...$parts);
970 }
971 }
972 foreach ($this->waiting as $index => $entry) {
973 $this->emit(
974 'WAITING', $index + 1, $entry->party, $entry->size,
975 'since', clockTime($entry->joined)
976 );
977 }
978 break;
979
980 case 'REPORT':
981 $this->report();
982 break;
983 }
984 }

986 public function execute(string $text): void
987 {
988 $tokens = explode(' ', $text);
989 $command = array_shift($tokens);
990 try {
991 $this->command($command, $tokens);
992 } catch (CommandError $error) {
993 $this->emit('ERROR', $error->getMessage());
994 }
995 }
996}
