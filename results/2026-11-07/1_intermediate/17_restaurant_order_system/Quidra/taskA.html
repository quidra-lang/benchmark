146Amounts check_amounts(const State state, int check_index)
147 check := state.checks[check_index]
148 amounts := [for index := active_lines(state, check_index)[] then order_amount(state.orders[index])]
149 subtotal := amounts.sum()
150 discount := rounded(subtotal, check.discount)
151 discounted := subtotal - discount
152 service := if check.size >= 6 then rounded(discounted, 18) else 0
153 tax := rounded(discounted, 8)
154 return Amounts(subtotal, discount, service, tax, discounted + service + tax)

156int[] allocate(const int[] amounts, int rate)
157 shares := amounts * rate / 100
158 remainders := amounts * rate % 100
159 extras := rounded(amounts.sum(), rate) - shares.sum()
160 for: 0 <= extra < extras
161 largest := remainders.max()
162 for: 0 <= index < int(#amounts)
163 if remainders[index] == largest
164 shares[index] += 1
165 remainders[index] = -1
166 break
167 return shares

169SeatBill[] seat_bills(const State state, int check_index)
170 lines := active_lines(state, check_index)
171 distinct<int> unique_seats
172 for index := lines[]
173 .unique_seats.add(state.orders[index].seat)
174 seats := unique_seats.sorted()
175 int[] subtotals := []
176 for seat := seats[]
177 amounts := [for index := lines[] if state.orders[index].seat == seat then order_amount(state.orders[index])]
178 subtotals = subtotals.append(amounts.sum())
179 discounts := allocate(subtotals, state.checks[check_index].discount)
180 discounted := subtotals - discounts
181 services := allocate(discounted, if state.checks[check_index].size >= 6 then 18 else 0)
182 taxes := allocate(discounted, 8)
183 SeatBill[] result := []
184 for: 0 <= index < int(#seats)
185 result = result.append(SeatBill(seats[index], subtotals[index], discounts[index], services[index], taxes[index], discounted[index] + services[index] + taxes[index]))
186 return result

192int select_table(const State state, int size)
193 selected := -1
194 for: 0 <= index < int(#state.tables)
195 table := state.tables[index]
196 if table.check < 0 and table.seats >= size
197 if selected < 0 or table.seats < state.tables[selected].seats or (table.seats == state.tables[selected].seats and table.number < state.tables[selected].number)
198 selected = index
199 return selected

201int seat_party(State &state, string name, int size, int table_index)
202 server_index := 0
203 load := server_load(state, 0)
204 for: 1 <= index < int(#state.servers)
205 candidate := server_load(state, index)
206 if candidate < load
207 load = candidate
208 server_index = index
209 state.tables[table_index].check = int(#state.checks)
210 state.checks = state.checks.append(Check(name, size, table_index, server_index, state.minute))
211 state.covers += size
212 state.parties += 1
213 state.servers[server_index].parties += 1
214 return server_index

230 amounts := check_amounts(state, check_index)

250void start_cooking(State &state, int station_index, int minute)
251 &station := &state.stations[station_index]
252 count := int(#cooking_lines(state, station_index))
253 while not station.down and #station.queue > 0 and count < station.capacity
254 index := station.queue[0]
255 station.queue = station.queue[1:]
256 state.orders[index].state = "COOKING"
257 state.orders[index].started = true
258 state.orders[index].ready = minute + state.orders[index].prep
259 count += 1

265void advance_time(State &state, int minutes)
266 ending := state.minute + minutes
267 emit("TICK " + clock_text(ending))
268 while true
269 finishes := [for order := state.orders[] if order.state == "COOKING" then order.ready]
270 if #finishes == 0
271 break
272 completed := finishes.min()
273 if completed > ending
274 break
275 for: 0 <= station_index < 4
276 for index := cooking_lines(state, station_index)[]
277 &order := &state.orders[index]
278 if order.ready == completed
279 order.state = "READY"
280 state.stations[station_index].tickets = state.stations[station_index].tickets.append(completed - order.fired)
281 table_number := state.tables[state.checks[order.check].table].number
282 emit("READY {clock_text(completed)} {state.stations[station_index].name} #{order.number} {state.menu[order.item].name} x{order.quantity} table {table_number} seat {order.seat}")
283 start_all(&state, completed)
284 state.minute = ending

374void | error command(State &state, const string[] words)
375 verb := words[0]
376 course_names := courses()
377 stations := station_names()
378 if verb == "TABLE"
379 table_number := parse_integer(words[1])
380 if table_id(state, table_number) >= 0
381 return error("table {table_number} already exists")
382 state.tables = state.tables.append(Table(table_number, parse_integer(words[2])))
383 emit("TABLE {table_number} seats {words[2]}")
384 seat_waiting(&state)
385 elif verb == "STAFF"
386 for server := state.servers[]
387 if server.name == words[1]
388 return error(words[1] + " is already on duty")
389 state.servers = state.servers.append(Server(words[1]))
390 emit("STAFF " + words[1])
391 elif verb == "SEAT"
392 name := words[1]
393 size := parse_integer(words[2])
394 for check := state.checks[]
395 if not check.closed and check.party == name
396 return error("{name} is already seated at table {state.tables[check.table].number}")
397 for party := state.waiting[]
398 if party.name == name
399 return error(name + " is already waiting")
400 suitable := [for table := state.tables[] if table.seats >= size then table]
401 if #suitable == 0
402 return error("no table for party of {size}")
403 if #state.servers == 0
404 return error("no server on duty")
405 table_index := select_table(state, size)
406 if table_index < 0
407 state.waiting = state.waiting.append(Waiting(name, size, state.minute))
408 emit("WAIT {name} {size} position {#state.waiting}")
409 else
410 server_index := seat_party(&state, name, size, table_index)
411 emit("SEAT {name} {size} table {state.tables[table_index].number} server {state.servers[server_index].name}")
412 elif verb == "CANCEL"
413 found := -1
414 for: 0 <= position < int(#state.waiting)
415 if state.waiting[position].name == words[1]
416 found = position
417 break
418 if found < 0
419 return error(words[1] + " is not waiting")
420 party := state.waiting[found]
421 state.waiting = state.waiting[0:<found].concat(state.waiting[found + 1:])
422 emit("CANCEL {party.name} {party.size} waited {state.minute - party.joined}")
423 elif verb == "MOVE"
424 source := table_id(state, parse_integer(words[1]))
425 destination := table_id(state, parse_integer(words[2]))
426 if source < 0
427 return error("unknown table " + words[1])
428 if destination < 0
429 return error("unknown table " + words[2])
430 if state.tables[source].check < 0
431 return error("table " + words[1] + " is free")
432 if state.tables[destination].check >= 0
433 return error("table " + words[2] + " is occupied")
434 check_index := state.tables[source].check
435 check := state.checks[check_index]
436 if state.tables[destination].seats < check.size
437 return error("table {words[2]} is too small for {check.party}")
438 state.tables[source].check = -1
439 state.tables[destination].check = check_index
440 state.checks[check_index].table = destination
441 emit("MOVE {check.party} table {words[1]} -> {words[2]}")
442 seat_waiting(&state)
443 elif verb == "MENU"
444 station := try named_index(stations, words[2], "station")
445 course := try named_index(course_names, words[5], "course")
446 item := item_id(state, words[1])
447 action := "ADD"
448 if item < 0
449 state.menu = state.menu.append(MenuItem(words[1], station, parse_integer(words[3]), cents(words[4]), course))
450 else
451 action = "UPDATE"
452 state.menu[item].station = station
453 state.menu[item].prep = parse_integer(words[3])
454 state.menu[item].price = cents(words[4])
455 state.menu[item].course = course
456 emit("MENU {action} {words[1]} {words[2]} {words[3]} {money(cents(words[4]))} {words[5]}")
457 elif verb == "STOCK"
458 item := item_id(state, words[1])
459 if item < 0
460 return error("unknown item " + words[1])
461 state.menu[item].stock = parse_integer(words[2])
462 emit("STOCK {words[1]} {words[2]}")
463 elif verb == "HAPPY"
464 state.happy_rate = parse_integer(words[1])
465 state.happy_end = state.minute + parse_integer(words[2])
466 emit("HAPPY {state.happy_rate}% until {clock_text(state.happy_end)}")
467 elif verb == "ITEMS"
468 emit("ITEMS {#state.menu}")
469 for: 0 <= course < 4
470 names := [for item := state.menu[] if item.course == course then item.name]
471 for name := names.sorted()[]
472 item := state.menu[item_id(state, name)]
473 stock := if item.stock < 0 then "-" else string(item.stock)
474 emit("{course_names[course]} {item.name} {stations[item.station]} {item.prep} {money(item.price)} {stock}")
475 elif verb == "TICK"
476 advance_time(&state, parse_integer(words[1]))
477 elif verb == "STATION"
478 station_index := try named_index(stations, words[1], "station")
479 down := words[2] == "DOWN"
480 if down and state.stations[station_index].down
481 return error("station " + words[1] + " is already down")
482 if not down and not state.stations[station_index].down
483 return error("station " + words[1] + " is not down")
484 state.stations[station_index].down = down
485 if not down
486 start_cooking(&state, station_index, state.minute)
487 emit("STATION {words[1]} {words[2]}")
488 elif verb == "RUSH" or verb == "REMAKE" or verb == "VOID" or verb == "COMP" or verb == "TRANSFER"
489 index := try order_id(state, parse_integer(words[1]))
490 order := state.orders[index]
491 check := state.checks[order.check]
492 table_number := state.tables[check.table].number
493 item_name := state.menu[order.item].name
494 if verb == "RUSH"
495 if order.state != "QUEUED"
496 return error("#{order.number} is not queued")
497 remaining := [for queued := state.stations[order.station].queue[] if queued != index then queued]
498 state.stations[order.station].queue = [index].concat(remaining)
499 emit("RUSH #{order.number} {stations[order.station]}")
500 elif verb == "REMAKE"
501 if check.closed
502 return error("#{order.number} is on a closed check")
503 if order.state != "READY" and order.state != "SERVED"
504 return error("#{order.number} is not ready or served")
505 state.orders[index].state = "QUEUED"
506 state.orders[index].fired = state.minute
507 state.stations[order.station].queue = [index].concat(state.stations[order.station].queue)
508 state.stations[order.station].remakes += 1
509 emit("REMAKE #{order.number} {item_name} {stations[order.station]}")
510 start_cooking(&state, order.station, state.minute)
511 elif verb == "VOID"
512 if order.state == "VOID"
513 return error("#{order.number} is already void")
514 if order.started
515 return error("#{order.number} has already started")
516 if order.state == "QUEUED"
517 state.stations[order.station].queue = [for queued := state.stations[order.station].queue[] if queued != index then queued]
518 state.orders[index].state = "VOID"
519 if state.menu[order.item].stock >= 0
520 state.menu[order.item].stock += order.quantity
521 state.voids += 1
522 emit("VOID #{order.number} {item_name} x{order.quantity}")
523 else
524 try modifiable(state, index)
525 if verb == "COMP"
526 if order.comp != ""
527 return error("#{order.number} is already comped")
528 amount := order_amount(order)
529 state.comps += 1
530 state.comp_amount += amount
531 state.orders[index].comp = words[2]
532 emit("COMP #{order.number} {item_name} {money(amount)} {words[2]}")
533 else
534 seat := parse_integer(words[2])
535 if seat > check.size
536 return error("seat {seat} out of range for table {table_number}")
537 if seat == order.seat
538 return error("#{order.number} is already on seat {seat}")
539 state.orders[index].seat = seat
540 emit("TRANSFER #{order.number} seat {order.seat} -> {seat}")
541 elif verb == "KITCHEN"
542 show_kitchen(state)
543 elif verb == "FLOOR"
544 show_floor(state)
545 elif verb == "REPORT"
546 show_report(state)
547 else
548 table_number := parse_integer(words[1])
549 check_index := try occupied_check(state, table_number)
550 check := state.checks[check_index]
551 if verb == "ORDER"
552 try unlocked(state, check_index)
553 seat := parse_integer(words[2])
554 if seat > check.size
555 return error("seat {seat} out of range for table {table_number}")
556 item := item_id(state, words[3])
557 if item < 0
558 return error("unknown item " + words[3])
559 menu_item := state.menu[item]
560 quantity := parse_integer(words[4])
561 if menu_item.stock >= 0 and menu_item.stock < quantity
562 return error("only {menu_item.stock} {menu_item.name} left")
563 happy := menu_item.course == 0 and state.minute < state.happy_end
564 price := menu_item.price - (if happy then rounded(menu_item.price, state.happy_rate) else 0)
565 index := int(#state.orders)
566 state.orders = state.orders.append(Order(index + 1, check_index, seat, item, quantity, menu_item.station, menu_item.prep, price, menu_item.course))
567 state.checks[check_index].lines = state.checks[check_index].lines.append(index)
