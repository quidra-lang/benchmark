106static void insert_order(OrderList *list, int index, Order *order) {
107 if (list->count == list->capacity) {
108 int capacity = list->capacity ? list->capacity * 2 : 8;
109 Order **items = realloc(list->items, (size_t)capacity * sizeof *items);
110 if (!items) exit(EXIT_FAILURE);
111 list->items = items;
112 list->capacity = capacity;
113 }
114 memmove(list->items + index + 1, list->items + index, (size_t)(list->count - index) * sizeof *list->items);
115 list->items[index] = order;
116 ++list->count;
117}

119static void append_order(OrderList *list, Order *order) { insert_order(list, list->count, order); }

121static Order *remove_at(OrderList *list, int index) {
122 Order *order = list->items[index];
123 --list->count;
124 memmove(list->items + index, list->items + index + 1, (size_t)(list->count - index) * sizeof *list->items);
125 return order;
126}
127
128static void remove_order(OrderList *list, Order *order) {
129 for (int i = 0; i < list->count; ++i) {
130 if (list->items[i] == order) { remove_at(list, i); return; }
131 }
132}

212static Totals totals(Check *c) {
213 Totals t = { 0 };
214 for (int i = 0; i < c->orders.count; ++i) {
215 Order *o = c->orders.items[i];
216 if (o->state != VOID) t.subtotal += charge(o);
217 }
218 t.discount = rounded(t.subtotal, c->discount);
219 Money net = t.subtotal - t.discount;
220 t.service = c->size >= 6 ? rounded(net, 18) : 0;
221 t.tax = rounded(net, 8);
222 t.total = net + t.service + t.tax;
223 return t;
224}

226static Seats seats(Check *c) {
227 Seats result = { 0 };
228 for (int i = 0; i < c->orders.count; ++i) {
229 Order *o = c->orders.items[i];
230 if (o->state == VOID) continue;
231 result.present[o->seat] = true;
232 result.amounts[o->seat] += charge(o);
233 }
234 return result;
235}

237static void allocate(const Money *values, const bool *present, int rate, Money *result) {
238 Money total = 0, assigned = 0;
239 bool used[21] = { false };
240 for (int seat = 1; seat <= 20; ++seat) {
241 result[seat] = present[seat] ? values[seat] * rate / 100 : 0;
242 if (present[seat]) total += values[seat];
243 assigned += result[seat];
244 }
245 Money remaining = rounded(total, rate) - assigned;
246 while (remaining-- > 0) {
247 int best = 0;
248 for (int seat = 1; seat <= 20; ++seat) {
249 if (!present[seat] || used[seat]) continue;
250 if (!best || values[seat] * rate % 100 > values[best] * rate % 100) best = seat;
251 }
252 ++result[best];
253 used[best] = true;
254 }
255}

257static Table *choose_table(int size) {
258 Table *best = NULL;
259 for (int i = 1; i <= 99; ++i) {
260 Table *t = &tables[i];
261 if (!t->number || t->check || t->seats < size) continue;
262 if (!best || t->seats < best->seats) best = t;
263 }
264 return best;
265}

267static Check *seat_party(Waiting party, Table *t) {
268 Server *server = &servers[0];
269 for (int i = 1; i < server_count; ++i) if (servers[i].load < server->load) server = &servers[i];
270 Check *c = &checks[check_count++];
271 *c = (Check){ .size = party.size, .seated = now, .table = t, .server = server };
272 strcpy(c->party, party.name);
273 t->check = c;
274 ++server->load;
275 ++server->parties;
276 stats.covers += party.size;
277 ++stats.parties;
278 return c;
279}

281static void seat_waiting(void) {
282 int i = 0;
283 while (i < waiting_count) {
284 Table *t = choose_table(waiting[i].size);
285 if (!t) { ++i; continue; }
286 Waiting party = remove_waiting(i);
287 Check *c = seat_party(party, t);
288 printf("SEATED %s %d table %d server %s waited %d\n", party.name, party.size, t->number, c->server->name, now - party.since);
289 }

292static void close_check(Check *c) {
293 Totals t = totals(c);
294 stats.totals.subtotal += t.subtotal;
295 stats.totals.discount += t.discount;
296 stats.totals.service += t.service;
297 stats.totals.tax += t.tax;
298 stats.totals.total += t.total;
299 for (int i = 0; i < c->orders.count; ++i) {
300 Order *o = c->orders.items[i];
301 if (o->state != VOID) stats.sales[o->course] += charge(o);
302 }
303 c->server->sales += t.total;
304 --c->server->load;
305 ++stats.turns;
306 stats.stay_total += now - c->seated;
307 c->closed = true;
308 c->table->check = NULL;
309 printf("FREE table %d %s stayed %d\n", c->table->number, c->party, now - c->seated);
310 seat_waiting();
311}

313static void start_cooking(int index, int time) {
314 Station *s = &stations[index];
315 int capacity = index == BAR ? 2 : 1;
316 while (s->up && s->queue.count && s->cooking.count < capacity) {
317 Order *o = remove_at(&s->queue, 0);
318 o->state = COOKING;
319 o->started = true;
320 o->ready = time + o->prep;
321 append_order(&s->cooking, o);
322 }
323}

335static void tick(int target) {
336 printf("TICK %s\n", clock_text(target).text);
337 for (;;) {
338 int next = INT_MAX;
339 for (int i = 0; i < 4; ++i)
340 for (int j = 0; j < stations[i].cooking.count; ++j)
341 if (stations[i].cooking.items[j]->ready < next) next = stations[i].cooking.items[j]->ready;
342 if (next > target) break;
343 for (int i = 0; i < 4; ++i) {
344 Station *s = &stations[i];
345 if (s->cooking.count > 1) qsort(s->cooking.items, (size_t)s->cooking.count, sizeof *s->cooking.items, order_by_id);
346 int j = 0;
347 while (j < s->cooking.count) {
348 Order *o = s->cooking.items[j];
349 if (o->ready != next) { ++j; continue; }
350 remove_at(&s->cooking, j);
351 o->state = READY;
352 int ticket = next - o->fired;
353 ++s->tickets;
354 s->ticket_total += ticket;
355 if (ticket > s->ticket_max) s->ticket_max = ticket;
356 printf("READY %s %s #%d %s x%d table %d seat %d\n", clock_text(next).text, station_names[i], o->id, o->item->name, o->quantity, o->check->table->number, o->seat);
357 }
358 }
359 for (int i = 0; i < 4; ++i) start_cooking(i, next);
360 }
361 now = target;
362}

364static void bill(Check *c) {
365 Seats by_seat = seats(c);
366 printf("BILL table %d %s guests %d\n", c->table->number, c->party, c->size);
367 for (int seat = 1; seat <= 20; ++seat) {
368 if (!by_seat.present[seat]) continue;
369 printf("SEAT %d %s\n", seat, money(by_seat.amounts[seat]).text);
370 for (int i = 0; i < c->orders.count; ++i) {
371 Order *o = c->orders.items[i];
372 if (o->state == VOID || o->seat != seat) continue;
373 printf("#%d %s x%d %s", o->id, o->item->name, o->quantity, money(charge(o)).text);
374 if (o->comp[0]) printf(" COMP %s", o->comp);
375 putchar('\n');
376 }
377 }
378 Totals t = totals(c);
379 printf("SUBTOTAL %s\n", money(t.subtotal).text);
380 if (c->discount) printf("DISCOUNT %d%% %s\n", c->discount, money(t.discount).text);
381 if (c->size >= 6) printf("SERVICE %s\n", money(t.service).text);
382 printf("TAX %s\nTOTAL %s\n", money(t.tax).text, money(t.total).text);
383 if (c->paid) printf("PAID %s\nDUE %s\n", money(c->paid).text, money(t.total - c->paid).text);
384}

386static void split(Check *c, char **args) {
387 Totals t = totals(c);
388 Money due = t.total - c->paid;
389 if (!due) { fail("nothing to split for table %d", c->table->number); return; }
390 if (equal(args[0], "EVEN")) {
391 int k = atoi(args[1]);
392 if (k > c->size) { fail("cannot split table %d %d ways", c->table->number, k); return; }
393 printf("SPLIT table %d EVEN %d due %s\n", c->table->number, k, money(due).text);
394 for (int i = 1; i <= k; ++i) printf("PAYER %d %s\n", i, money(due / k + (i <= due % k)).text);
395 } else {
396 if (c->paid) { fail("table %d has payments", c->table->number); return; }
397 Seats by_seat = seats(c);
398 Money discounts[21] = { 0 }, net[21] = { 0 }, service[21] = { 0 }, tax[21] = { 0 };
399 allocate(by_seat.amounts, by_seat.present, c->discount, discounts);
400 for (int i = 1; i <= 20; ++i) net[i] = by_seat.amounts[i] - discounts[i];
401 allocate(net, by_seat.present, c->size >= 6 ? 18 : 0, service);
402 allocate(net, by_seat.present, 8, tax);
403 printf("SPLIT table %d BYSEAT total %s\n", c->table->number, money(t.total).text);
404 for (int i = 1; i <= 20; ++i) {
405 if (!by_seat.present[i]) continue;
406 printf("SEAT %d subtotal %s discount %s service %s tax %s total %s\n", i, money(by_seat.amounts[i]).text,
407 money(discounts[i]).text, money(service[i]).text, money(tax[i]).text, money(net[i] + service[i] + tax[i]).text);
408 }
409 }
410}

448static void execute(char **a) {
449 const char *op = a[0];
450 if (equal(op, "TABLE")) {
451 int number = atoi(a[1]);
452 Table *t = &tables[number];
453 if (t->number) { fail("table %d already exists", number); return; }
454 t->number = number;
455 t->seats = atoi(a[2]);
456 printf("TABLE %d seats %d\n", number, t->seats);
457 seat_waiting();
458 } else if (equal(op, "STAFF")) {
459 for (int i = 0; i < server_count; ++i)
460 if (equal(servers[i].name, a[1])) { fail("%s is already on duty", a[1]); return; }
461 strcpy(servers[server_count++].name, a[1]);
462 printf("STAFF %s\n", a[1]);
463 } else if (equal(op, "SEAT")) {
464 for (int i = 1; i <= 99; ++i) {
465 if (tables[i].check && equal(tables[i].check->party, a[1])) {
466 fail("%s is already seated at table %d", a[1], i); return;
467 }
468 }
469 for (int i = 0; i < waiting_count; ++i)
470 if (equal(waiting[i].name, a[1])) { fail("%s is already waiting", a[1]); return; }
471 int size = atoi(a[2]);
472 bool fits = false;
473 for (int i = 1; i <= 99; ++i) if (tables[i].number && tables[i].seats >= size) fits = true;
474 if (!fits) { fail("no table for party of %d", size); return; }
475 if (!server_count) { fail("no server on duty"); return; }
476 Waiting party = { .size = size, .since = now };
477 strcpy(party.name, a[1]);
478 Table *t = choose_table(size);
479 if (t) {
480 Check *c = seat_party(party, t);
481 printf("SEAT %s %d table %d server %s\n", party.name, size, t->number, c->server->name);
482 } else {
483 waiting[waiting_count++] = party;
484 printf("WAIT %s %d position %d\n", party.name, size, waiting_count);
485 }
486 } else if (equal(op, "CANCEL")) {
487 int i = 0;
488 while (i < waiting_count && !equal(waiting[i].name, a[1])) ++i;
489 if (i == waiting_count) { fail("%s is not waiting", a[1]); return; }
490 Waiting party = remove_waiting(i);
491 printf("CANCEL %s %d waited %d\n", party.name, party.size, now - party.since);
492 } else if (equal(op, "MOVE")) {
493 Table *from = table(a[1]);
494 if (!from) return;
495 Table *to = table(a[2]);
496 if (!to) return;
497 Check *c = from->check;
498 if (!c) { fail("table %s is free", a[1]); return; }
499 if (to->check) { fail("table %s is occupied", a[2]); return; }
500 if (to->seats < c->size) { fail("table %s is too small for %s", a[2], c->party); return; }
501 from->check = NULL;
502 to->check = c;
503 c->table = to;
504 printf("MOVE %s table %d -> %d\n", c->party, from->number, to->number);
505 seat_waiting();
506 } else if (equal(op, "MENU")) {
507 int station = station_index(a[2]);
508 if (station < 0) return;
509 int course = course_index(a[5]);
510 if (course < 0) return;
511 MenuItem *m = find_item(a[1]);
512 bool existing = m != NULL;
513 if (!m) {
514 m = &menu[menu_count++];
515 strcpy(m->name, a[1]);
516 m->stock = -1;
517 }
518 m->station = (StationName)station;
519 m->course = (Course)course;
520 m->prep = atoi(a[3]);
521 m->price = cents(a[4]);
522 printf("MENU %s %s %s %d %s %s\n", existing ? "UPDATE" : "ADD", m->name, a[2], m->prep, money(m->price).text, a[5]);
523 } else if (equal(op, "STOCK")) {
524 MenuItem *m = item(a[1]);
525 if (!m) return;
526 m->stock = atoi(a[2]);
527 printf("STOCK %s %d\n", m->name, m->stock);
528 } else if (equal(op, "HAPPY")) {
529 happy_rate = atoi(a[1]);
530 happy_end = now + atoi(a[2]);
531 printf("HAPPY %d%% until %s\n", happy_rate, clock_text(happy_end).text);
532 } else if (equal(op, "ITEMS")) {
533 MenuItem *items[1000];
534 for (int i = 0; i < menu_count; ++i) items[i] = &menu[i];
535 qsort(items, (size_t)menu_count, sizeof *items, menu_order);
536 printf("ITEMS %d\n", menu_count);
537 for (int i = 0; i < menu_count; ++i) {
538 MenuItem *m = items[i];
539 printf("%s %s %s %d %s ", course_names[m->course], m->name, station_names[m->station], m->prep, money(m->price).text);
540 if (m->stock < 0) puts("-");
541 else printf("%d\n", m->stock);
542 }
543 } else if (equal(op, "TICK")) {
544 tick(now + atoi(a[1]));
545 } else if (equal(op, "STATION")) {
546 int index = station_index(a[1]);
547 if (index < 0) return;
548 Station *s = &stations[index];
549 bool up = equal(a[2], "UP");
550 if (s->up == up) { fail("station %s is %s", a[1], up ? "not down" : "already down"); return; }
551 s->up = up;
552 if (up) start_cooking(index, now);
553 printf("STATION %s %s\n", a[1], a[2]);
554 } else if (equal(op, "RUSH") || equal(op, "REMAKE") || equal(op, "VOID") || equal(op, "COMP") || equal(op, "TRANSFER")) {
555 Order *o = order(a[1]);
556 if (!o) return;
557 Station *s = &stations[o->station];
558 if (equal(op, "RUSH")) {
559 if (o->state != QUEUED) { fail("#%d is not queued", o->id); return; }
560 remove_order(&s->queue, o);
561 insert_order(&s->queue, 0, o);
562 printf("RUSH #%d %s\n", o->id, station_names[o->station]);
563 } else if (equal(op, "REMAKE")) {
564 if (o->check->closed) { fail("#%d is on a closed check", o->id); return; }
565 if (o->state != READY && o->state != SERVED) { fail("#%d is not ready or served", o->id); return; }
566 o->state = QUEUED;
567 o->fired = now;
568 insert_order(&s->queue, 0, o);
569 ++s->remakes;
570 printf("REMAKE #%d %s %s\n", o->id, o->item->name, station_names[o->station]);
571 start_cooking(o->station, now);
572 } else if (equal(op, "VOID")) {
573 if (o->state == VOID) { fail("#%d is already void", o->id); return; }
574 if (o->started) { fail("#%d has already started", o->id); return; }
575 if (o->state == QUEUED) remove_order(&s->queue, o);
576 o->state = VOID;
577 if (o->item->stock >= 0) o->item->stock += o->quantity;
578 ++stats.voids;
579 printf("VOID #%d %s x%d\n", o->id, o->item->name, o->quantity);
580 } else if (equal(op, "COMP")) {
581 if (!modifiable(o)) return;
582 if (o->comp[0]) { fail("#%d is already comped", o->id); return; }
583 Money value = charge(o);
584 printf("COMP #%d %s %s %s\n", o->id, o->item->name, money(value).text, a[2]);
585 ++stats.comps;
586 stats.comp_amount += value;
587 strcpy(o->comp, a[2]);
588 } else {
589 if (!modifiable(o)) return;
590 int seat = atoi(a[2]);
591 if (!valid_seat(o->check, seat)) return;
592 if (seat == o->seat) { fail("#%d is already on seat %d", o->id, seat); return; }
593 printf("TRANSFER #%d seat %d -> %d\n", o->id, o->seat, seat);
594 o->seat = seat;
595 }
596 } else if (equal(op, "KITCHEN")) {
597 printf("KITCHEN %s\n", clock_text(now).text);
598 for (int i = 0; i < 4; ++i) {
599 Station *s = &stations[i];
600 Order *cooking[2];
601 for (int j = 0; j < s->cooking.count; ++j) cooking[j] = s->cooking.items[j];
602 qsort(cooking, (size_t)s->cooking.count, sizeof *cooking, order_by_ready);
603 printf("%s %s cooking", station_names[i], s->up ? "up" : "down");
604 for (int j = 0; j < s->cooking.count; ++j) printf(" #%d@%s", cooking[j]->id, clock_text(cooking[j]->ready).text);
605 if (!s->cooking.count) fputs(" -", stdout);
606 fputs(" queue", stdout);
607 for (int j = 0; j < s->queue.count; ++j) printf(" #%d", s->queue.items[j]->id);
608 if (!s->queue.count) fputs(" -", stdout);
609 putchar('\n');
610 }
611 } else if (equal(op, "FLOOR")) {
612 int free_tables = 0, occupied = 0;
613 for (int i = 1; i <= 99; ++i) {
614 if (!tables[i].number) continue;
615 if (tables[i].check) ++occupied;
616 else ++free_tables;
617 }
618 printf("FLOOR %s free %d seated %d waiting %d\n", clock_text(now).text, free_tables, occupied, waiting_count);
619 for (int i = 1; i <= 99; ++i) {
620 Table *t = &tables[i];
621 if (!t->number) continue;
622 printf("table %d seats %d", i, t->seats);
623 Check *c = t->check;
624 if (!c) fputs(" free", stdout);
625 else {
626 printf(" %s guests %d server %s since %s items %d total %s", c->party, c->size, c->server->name, clock_text(c->seated).text, active_count(c), money(totals(c).total).text);
627 if (c->paid) printf(" paid %s", money(c->paid).text);
628 }
629 putchar('\n');
630 }
631 for (int i = 0; i < waiting_count; ++i)
632 printf("WAITING %d %s %d since %s\n", i + 1, waiting[i].name, waiting[i].size, clock_text(waiting[i].since).text);
633 } else if (equal(op, "REPORT")) {
634 report();
635 } else {
636 Check *c = check(a[1]);
637 if (!c) return;
638 int number = c->table->number;
639 if (equal(op, "ORDER")) {
640 if (!unlocked(c)) return;
641 int seat = atoi(a[2]);
642 if (!valid_seat(c, seat)) return;
643 MenuItem *m = item(a[3]);
644 if (!m) return;
645 int quantity = atoi(a[4]);
646 if (m->stock >= 0 && m->stock < quantity) { fail("only %d %s left", m->stock, m->name); return; }
647 bool happy = m->course == DRINK && now < happy_end;
648 Order *o = &orders[order_count];
649 *o = (Order){ .id = order_count + 1, .seat = seat, .quantity = quantity, .prep = m->prep,
650 .check = c, .item = m, .station = m->station, .course = m->course,
651 .state = ORDERED, .price = m->price - (happy ? rounded(m->price, happy_rate) : 0) };
652 ++order_count;
653 append_order(&c->orders, o);
654 printf("ORDER #%d table %d seat %d %s x%d %s%s\n", o->id, number, seat, m->name, quantity, money(charge(o)).text, happy ? " HAPPY" : "");
655 if (m->stock >= 0) {
656 m->stock -= quantity;
657 if (!m->stock) printf("SOLDOUT %s\n", m->name);
658 }
659 } else if (equal(op, "FIRE")) {
660 int course = course_index(a[2]);
661 if (course < 0) return;
662 bool found = false;
663 for (int i = 0; i < c->orders.count; ++i)
664 if (c->orders.items[i]->course == (Course)course && c->orders.items[i]->state == ORDERED) found = true;
665 if (!found) { fail("nothing to fire for %s at table %d", a[2], number); return; }
666 if (course == MAIN || course == DESSERT) {
667 for (int earlier = STARTER; earlier < course; ++earlier) {
668 for (int i = 0; i < c->orders.count; ++i) {
669 Order *o = c->orders.items[i];
670 if (o->course == (Course)earlier && o->state == ORDERED) { fail("fire %s first", course_names[earlier]); return; }
671 }
672 }
673 }
674 printf("FIRE table %d %s", number, a[2]);
675 for (int i = 0; i < c->orders.count; ++i) {
676 Order *o = c->orders.items[i];
677 if (o->course != (Course)course || o->state != ORDERED) continue;
678 o->state = QUEUED;
679 o->fired = now;
680 append_order(&stations[o->station].queue, o);
681 printf(" #%d", o->id);
682 }
683 putchar('\n');
684 for (int i = 0; i < 4; ++i) start_cooking(i, now);
685 } else if (equal(op, "SERVE")) {
686 bool found = false;
687 for (int i = 0; i < c->orders.count; ++i) if (c->orders.items[i]->state == READY) found = true;
688 if (!found) { fail("nothing ready for table %d", number); return; }
689 printf("SERVE table %d", number);
690 for (int i = 0; i < c->orders.count; ++i) {
691 Order *o = c->orders.items[i];
692 if (o->state != READY) continue;
693 o->state = SERVED;
694 o->item->served += o->quantity;
695 printf(" #%d", o->id);
696 }
697 putchar('\n');
698 } else if (equal(op, "DISCOUNT")) {
699 if (!unlocked(c)) return;
700 c->discount = atoi(a[2]);
701 printf("DISCOUNT table %d %d%%\n", number, c->discount);
702 } else if (equal(op, "BILL")) bill(c);
703 else if (equal(op, "SPLIT")) split(c, a + 2);
704 else if (equal(op, "PAY")) {
705 bool cash = equal(a[2], "CASH");
706 if (!cash && !equal(a[2], "CARD")) { fail("unknown method %s", a[2]); return; }
707 if (!all_served(c)) return;
708 Money due = totals(c).total - c->paid;
709 if (!due) { fail("nothing to pay for table %d", number); return; }
710 Money paid = cents(a[3]), applied = paid < due ? paid : due;
711 c->paid += applied;
712 if (cash) stats.cash += applied;
713 else stats.card += applied;
714 if (paid < due) printf("PAY table %d %s %s remaining %s\n", number, a[2], money(paid).text, money(due - paid).text);
715 else {
716 if (!cash) { stats.tips += paid - due; c->server->tips += paid - due; }
717 printf("PAY table %d %s %s %s %s\n", number, a[2], money(paid).text, cash ? "change" : "tip", money(paid - due).text);
718 close_check(c);
719 }
720 } else if (equal(op, "CLOSE")) {
721 if (!all_served(c)) return;
722 Money due = totals(c).total - c->paid;
723 if (due) { fail("table %d still owes %s", number, money(due).text); return; }
724 close_check(c);
725 } else if (equal(op, "STATUS")) {
726 printf("STATUS table %d %s guests %d since %s\n", number, c->party, c->size, clock_text(c->seated).text);
727 for (int i = 0; i < c->orders.count; ++i) {
728 Order *o = c->orders.items[i];
729 printf("#%d seat %d %s x%d %s %s", o->id, o->seat, o->item->name, o->quantity, course_names[o->course], state_names[o->state]);
730 if (o->state == COOKING) printf(" until %s", clock_text(o->ready).text);
731 if (o->comp[0]) printf(" COMP %s", o->comp);
732 putchar('\n');
733 }
734 }
735 }
736}

740 if (!fgets(line, sizeof line, stdin)) return 0;

749 }

751 for (int i = 0; i < 4; ++i) { free(stations[i].queue.items); free(stations[i].cooking.items); }
752 return 0;
