143sealed class Restaurant
144{
145 private static readonly string[] StationNames = { "GRILL", "FRY", "COLD", "BAR" };
146 private static readonly string[] CourseNames = { "DRINK", "STARTER", "MAIN", "DESSERT" };
147
148 private readonly Dictionary<int, Table> tables = new();
149 private readonly List<Server> servers = new();
150 private readonly List<WaitingParty> waiting = new();
151 private readonly Dictionary<string, MenuItem> menu = new(StringComparer.Ordinal);

153 private readonly Dictionary<string, Station> stations = new(StringComparer.Ordinal);
154 private readonly Dictionary<string, int> sales = new(StringComparer.Ordinal);
155 private readonly Dictionary<string, int> totals = new(StringComparer.Ordinal);

172 public Restaurant()
173 {
174 foreach (string name in StationNames) stations.Add(name, new Station(name));
175 foreach (string name in CourseNames) sales.Add(name, 0);
176 foreach (string name in new[] { "subtotal", "discount", "service", "tax", "total" })
177 totals.Add(name, 0);
178 payments.Add("CASH", 0);
179 payments.Add("CARD", 0);

182 private static int Number(string value) => int.Parse(value, CultureInfo.InvariantCulture);
183
184 private static int Cents(string amount) => Number(amount.Replace(".", "", StringComparison.Ordinal));
185
186 private static string Money(int amount) =>
187 (amount / 100).ToString(CultureInfo.InvariantCulture) + "." +
188 (amount % 100).ToString("00", CultureInfo.InvariantCulture);
189
190 private static string Clock(int time)
191 {
192 int minutes = 660 + time;
193 return (minutes / 60).ToString("00", CultureInfo.InvariantCulture) + ":" +
194 (minutes % 60).ToString("00", CultureInfo.InvariantCulture);
195 }
196
197 private static int Rounded(int value, int rate) => (value * rate + 50) / 100;
198
199 private void Emit(params object[] parts) =>
200 output.Add(string.Join(" ", parts.Select(p => Convert.ToString(p, CultureInfo.InvariantCulture))));

202 private static void Fail(string reason) => throw new CommandError(reason);

231 {
232 if (line.State == "VOID") Fail("#" + line.Number + " is void");
233 if (line.Check.Closed) Fail("#" + line.Number + " is on a closed check");
234 Unlocked(line.Check);

239 int count = check.Active().Count(line => line.State != "SERVED");

242 }

252 if (!CourseNames.Contains(name, StringComparer.Ordinal)) Fail("unknown course " + name);

261 private int Load(Server server) =>
262 tables.Values.Count(table => table.Check?.Server == server);
263
264 private Table? SelectTable(int size) =>
265 tables.Values.Where(table => table.Check is null && table.Seats >= size)
266 .OrderBy(table => table.Seats).ThenBy(table => table.Number).FirstOrDefault();
267
268 private Check Seat(string party, int size, Table table)
269 {
270 Server server = servers[0];
271 foreach (Server candidate in servers)
272 {
273 if (Load(candidate) < Load(server)) server = candidate;
274 }
275 Check check = new(party, size, table, server, now);
276 table.Check = check;
277 server.Parties++;
278 covers += size;
279 parties++;
280 return check;
281 }
282
283 private void ProcessWaiting()
284 {
285 List<WaitingParty> remaining = new();
286 foreach (WaitingParty party in waiting)
287 {
288 Table? table = SelectTable(party.Size);
289 if (table is null)
290 {
291 remaining.Add(party);
292 continue;
293 }
294 Check check = Seat(party.Party, party.Size, table);
295 Emit("SEATED", party.Party, party.Size, "table", table.Number, "server",
296 check.Server.Name, "waited", now - party.Joined);
297 }
298 waiting.Clear();
299 waiting.AddRange(remaining);
300 }
301
302 private static Amounts Calculate(Check check)
303 {
304 int subtotal = check.Active().Sum(line => line.Amount);
305 int discount = Rounded(subtotal, check.Discount);
306 int after = subtotal - discount;
307 int service = check.Size >= 6 ? Rounded(after, 18) : 0;
308 int tax = Rounded(after, 8);
309 int total = after + service + tax;
310 return new Amounts(subtotal, discount, service, tax, total, total - check.Paid);
311 }
312
313 private void Close(Check check)
314 {
315 Amounts amounts = Calculate(check);
316 turns++;
317 stay += now - check.Seated;
318 foreach (OrderLine line in check.Active()) sales[line.Course] += line.Amount;
319 totals["subtotal"] += amounts.Subtotal;
320 totals["discount"] += amounts.Discount;
321 totals["service"] += amounts.Service;
322 totals["tax"] += amounts.Tax;
323 totals["total"] += amounts.Total;
324 check.Server.Sales += amounts.Total;
325 check.Closed = true;
326 check.Table.Check = null;
327 Emit("FREE", "table", check.Table.Number, check.Party, "stayed", now - check.Seated);
328 ProcessWaiting();
329 }

331 private static void Start(Station station, int time)
332 {
333 while (station.Up && station.Queue.Count > 0 &&
334 station.Cooking.Count < station.Capacity)
335 {
336 OrderLine line = station.Queue[0];
337 station.Queue.RemoveAt(0);
338 line.State = "COOKING";
339 line.Ready = time + line.Prep;
340 line.Started = true;
341 station.Cooking.Add(line);
342 }
343 }
344
345 private void Tick(int minutes)
346 {
347 int end = now + minutes;
348 Emit("TICK", Clock(end));
349 while (true)
350 {
351 List<OrderLine> cooking = StationNames.SelectMany(name => stations[name].Cooking).ToList();
352 if (cooking.Count == 0) break;
353 int time = cooking.Min(line => line.Ready);
354 if (time > end) break;
355 foreach (string name in StationNames)
356 {
357 Station station = stations[name];
358 List<OrderLine> finished = station.Cooking
359 .Where(line => line.Ready == time).OrderBy(line => line.Number).ToList();
360 station.Cooking.RemoveAll(line => line.Ready == time);
361 foreach (OrderLine line in finished)
362 {
363 line.State = "READY";
364 station.Tickets.Add(time - line.Fire);
365 Emit("READY", Clock(time), name, "#" + line.Number, line.Item,
366 "x" + line.Quantity, "table", line.Check.Table.Number, "seat", line.Seat);
367 }
368 }
369 foreach (string name in StationNames) Start(stations[name], time);
370 }
371 now = end;
372 }

374 private static SortedDictionary<int, int> SeatSubtotals(Check check)
375 {
376 SortedDictionary<int, int> result = new();
377 foreach (OrderLine line in check.Active())
378 {
379 result[line.Seat] = result.GetValueOrDefault(line.Seat) + line.Amount;
380 }
381 return result;
382 }

384 private static Dictionary<int, int> Allocate(IReadOnlyDictionary<int, int> values, int rate)
385 {
386 Dictionary<int, int> result = values.ToDictionary(
387 entry => entry.Key, entry => entry.Value * rate / 100);
388 int left = Rounded(values.Values.Sum(), rate) - result.Values.Sum();
389 foreach (int seat in values.Keys
390 .OrderByDescending(seat => values[seat] * rate % 100)
391 .ThenBy(seat => seat).Take(left))
392 {
393 result[seat]++;
394 }
395 return result;
396 }

398 private void Bill(Check check)
399 {
400 Emit("BILL", "table", check.Table.Number, check.Party, "guests", check.Size);
401 foreach ((int seat, int subtotal) in SeatSubtotals(check))
402 {
403 Emit("SEAT", seat, Money(subtotal));
404 foreach (OrderLine line in check.Active().Where(line => line.Seat == seat))
405 {
406 if (line.Reason is null)
407 Emit("#" + line.Number, line.Item, "x" + line.Quantity, Money(line.Amount));
408 else
409 Emit("#" + line.Number, line.Item, "x" + line.Quantity, Money(line.Amount),
410 "COMP", line.Reason);
411 }
412 }
413 Amounts amount = Calculate(check);
414 Emit("SUBTOTAL", Money(amount.Subtotal));
415 if (check.Discount != 0) Emit("DISCOUNT", check.Discount + "%", Money(amount.Discount));
416 if (check.Size >= 6) Emit("SERVICE", Money(amount.Service));
417 Emit("TAX", Money(amount.Tax));
418 Emit("TOTAL", Money(amount.Total));
419 if (check.Paid > 0)
420 {
421 Emit("PAID", Money(check.Paid));
422 Emit("DUE", Money(amount.Due));
423 }
424 }

426 private void Report()
427 {
428 Emit("REPORT", Clock(now));
429 Emit("COVERS", covers, "parties", parties);
430 Emit("TURNS", turns, "avg_stay", turns > 0 ? (object) (stay / turns) : "-");
431 List<object> saleParts = new() { "SALES" };
432 foreach (string name in CourseNames)
433 {
434 saleParts.Add(name);
435 saleParts.Add(Money(sales[name]));
436 }
437 Emit(saleParts.ToArray());
438 List<object> totalParts = new() { "TOTALS" };
439 foreach (string name in new[] { "subtotal", "discount", "service", "tax", "total" })
440 {
441 totalParts.Add(name);
442 totalParts.Add(Money(totals[name]));
443 }
444 Emit(totalParts.ToArray());
445 Emit("PAYMENTS", "cash", Money(payments["CASH"]), "card", Money(payments["CARD"]),
446 "tips", Money(tips));
447 Emit("COMPS", comps, Money(compAmount), "VOIDS", voids);
448 int rank = 0;
449 foreach (KeyValuePair<string, int> entry in served.Where(entry => entry.Value > 0)
450 .OrderByDescending(entry => entry.Value)
451 .ThenBy(entry => entry.Key, StringComparer.Ordinal).Take(3))
452 {
453 Emit("TOP", ++rank, entry.Key, entry.Value);
454 }
455 foreach (string name in StationNames)
456 {
457 Station station = stations[name];
458 Emit(name, "tickets", station.Tickets.Count,
459 "avg", station.Tickets.Count > 0
460 ? (object) (station.Tickets.Sum() / station.Tickets.Count) : "-",
461 "max", station.Tickets.Count > 0
462 ? (object) station.Tickets.Max() : "-",
463 "remakes", station.Remakes);
464 }
465 foreach (Server server in servers)
466 {
467 Emit("SERVER", server.Name, "tables", Load(server), "parties", server.Parties,
468 "sales", Money(server.Sales), "tips", Money(server.Tips));
469 }
470 }

472 private void Command(string command, string[] args)
473 {
474 switch (command)
475 {
476 case "TABLE":
477 {
478 int number = Number(args[0]);
479 if (tables.ContainsKey(number)) Fail("table " + args[0] + " already exists");
480 tables[number] = new Table(number, Number(args[1]));
481 Emit("TABLE", number, "seats", Number(args[1]));
482 ProcessWaiting();
483 break;
484 }
485 case "STAFF":
486 if (servers.Any(server => server.Name == args[0]))
487 Fail(args[0] + " is already on duty");
488 servers.Add(new Server(args[0]));
489 Emit("STAFF", args[0]);
490 break;
491
492 case "SEAT":
493 {
494 Table? seated = tables.Values.FirstOrDefault(table => table.Check?.Party == args[0]);
495 if (seated is not null)
496 Fail(args[0] + " is already seated at table " + seated.Number);
497 if (waiting.Any(party => party.Party == args[0]))
498 Fail(args[0] + " is already waiting");
499 int size = Number(args[1]);
500 if (!tables.Values.Any(table => table.Seats >= size))
501 Fail("no table for party of " + size);
502 if (servers.Count == 0) Fail("no server on duty");
503 Table? table = SelectTable(size);
504 if (table is null)
505 {
506 waiting.Add(new WaitingParty(args[0], size, now));
507 Emit("WAIT", args[0], size, "position", waiting.Count);
508 }
509 else
510 {
511 Check check = Seat(args[0], size, table);
512 Emit("SEAT", args[0], size, "table", table.Number, "server", check.Server.Name);
513 }
514 break;
515 }
516 case "CANCEL":
517 {
518 int index = waiting.FindIndex(party => party.Party == args[0]);
519 if (index < 0) Fail(args[0] + " is not waiting");
520 WaitingParty party = waiting[index];
521 waiting.RemoveAt(index);
522 Emit("CANCEL", party.Party, party.Size, "waited", now - party.Joined);
523 break;
524 }
525 case "MOVE":
526 {
527 Table source = RequireTable(args[0]);
528 Table destination = RequireTable(args[1]);
529 if (source.Check is null) Fail("table " + args[0] + " is free");
530 if (destination.Check is not null) Fail("table " + args[1] + " is occupied");
531 Check check = source.Check!;
532 if (destination.Seats < check.Size)
533 Fail("table " + args[1] + " is too small for " + check.Party);
534 source.Check = null;
535 destination.Check = check;
536 check.Table = destination;
537 Emit("MOVE", check.Party, "table", source.Number, "->", destination.Number);
538 ProcessWaiting();
539 break;
540 }
541 case "MENU":
542 {
543 RequireStation(args[1]);
544 RequireCourse(args[4]);
545 menu.TryGetValue(args[0], out MenuItem? previous);
546 MenuItem item = new(args[0], args[1], Number(args[2]), Cents(args[3]), args[4])
547 {
548 Stock = previous?.Stock
549 };
550 menu[args[0]] = item;
551 Emit("MENU", previous is null ? "ADD" : "UPDATE", item.Name,
552 item.Station, item.Prep, Money(item.Price), item.Course);
553 break;
554 }
555 case "STOCK":
556 {
557 MenuItem item = RequireItem(args[0]);
558 item.Stock = Number(args[1]);
559 Emit("STOCK", item.Name, item.Stock.Value);
560 break;
561 }
562 case "HAPPY":
563 happyRate = Number(args[0]);
564 happyEnd = now + Number(args[1]);
565 Emit("HAPPY", happyRate + "%", "until", Clock(happyEnd));
566 break;
567
568 case "ITEMS":
569 {
570 List<MenuItem> items = menu.Values
571 .OrderBy(item => Array.IndexOf(CourseNames, item.Course))
572 .ThenBy(item => item.Name, StringComparer.Ordinal).ToList();
573 Emit("ITEMS", items.Count);
574 foreach (MenuItem item in items)
575 {
576 Emit(item.Course, item.Name, item.Station, item.Prep,
577 Money(item.Price), item.Stock is null ? "-" : item.Stock.Value);
578 }
579 break;
580 }
581 case "ORDER":
582 {
583 Check check = RequireCheck(args[0]);
584 Unlocked(check);
585 int seat = Number(args[1]);
586 if (seat > check.Size) Fail("seat " + seat + " out of range for table " + args[0]);
587 MenuItem item = RequireItem(args[2]);
588 int quantity = Number(args[3]);
589 if (item.Stock is not null && item.Stock.Value < quantity)
590 Fail("only " + item.Stock.Value + " " + item.Name + " left");
591 bool happy = item.Course == "DRINK" && now < happyEnd;
592 int price = item.Price - (happy ? Rounded(item.Price, happyRate) : 0);
593 OrderLine line = new(lines.Count + 1, check, seat, item, quantity, price);
594 lines.Add(line);
595 check.Lines.Add(line);
596 List<object> parts = new()
597 {
598 "ORDER", "#" + line.Number, "table", check.Table.Number, "seat", seat,
599 item.Name, "x" + quantity, Money(line.Amount)
600 };
601 if (happy) parts.Add("HAPPY");
602 Emit(parts.ToArray());
603 if (item.Stock is not null)
604 {
605 item.Stock -= quantity;
606 if (item.Stock == 0) Emit("SOLDOUT", item.Name);
607 }
608 break;
609 }
610 case "FIRE":
611 {
612 Check check = RequireCheck(args[0]);
613 RequireCourse(args[1]);
614 List<OrderLine> firing = check.Lines
615 .Where(line => line.Course == args[1] && line.State == "ORDERED").ToList();
616 if (firing.Count == 0)
617 Fail("nothing to fire for " + args[1] + " at table " + args[0]);
618 string[] orderedCourses = { "STARTER", "MAIN", "DESSERT" };
619 int position = Array.IndexOf(orderedCourses, args[1]);
620 for (int i = 0; i < position; ++i)
621 {
622 string earlier = orderedCourses[i];
623 if (check.Lines.Any(line => line.Course == earlier && line.State == "ORDERED"))
624 Fail("fire " + earlier + " first");
625 }
626 List<object> parts = new() { "FIRE", "table", check.Table.Number, args[1] };
627 foreach (OrderLine line in firing)
628 {
629 line.State = "QUEUED";
630 line.Fire = now;
631 stations[line.Station].Queue.Add(line);
632 parts.Add("#" + line.Number);
633 }
634 Emit(parts.ToArray());
635 foreach (string name in StationNames) Start(stations[name], now);
636 break;
637 }
638 case "TICK":
639 Tick(Number(args[0]));
640 break;
641
642 case "SERVE":
643 {
644 Check check = RequireCheck(args[0]);
645 List<OrderLine> ready = check.Lines.Where(line => line.State == "READY").ToList();
646 if (ready.Count == 0) Fail("nothing ready for table " + args[0]);
647 List<object> parts = new() { "SERVE", "table", check.Table.Number };
648 foreach (OrderLine line in ready)
649 {
650 line.State = "SERVED";
651 served[line.Item] = served.GetValueOrDefault(line.Item) + line.Quantity;
652 parts.Add("#" + line.Number);
653 }
654 Emit(parts.ToArray());
655 break;
656 }
657 case "RUSH":
658 {
659 OrderLine line = RequireLine(args[0]);
660 if (line.State != "QUEUED") Fail("#" + line.Number + " is not queued");
661 Station station = stations[line.Station];
662 station.Queue.Remove(line);
663 station.Queue.Insert(0, line);
664 Emit("RUSH", "#" + line.Number, line.Station);
665 break;
666 }
667 case "REMAKE":
668 {
669 OrderLine line = RequireLine(args[0]);
670 if (line.Check.Closed) Fail("#" + line.Number + " is on a closed check");
671 if (line.State != "READY" && line.State != "SERVED")
672 Fail("#" + line.Number + " is not ready or served");
673 line.State = "QUEUED";
674 line.Fire = now;
675 Station station = stations[line.Station];
676 station.Queue.Insert(0, line);
677 station.Remakes++;
678 Emit("REMAKE", "#" + line.Number, line.Item, line.Station);
679 Start(station, now);
680 break;
681 }
682 case "VOID":
683 {
684 OrderLine line = RequireLine(args[0]);
685 if (line.State == "VOID") Fail("#" + line.Number + " is already void");
686 if (line.Started) Fail("#" + line.Number + " has already started");
687 if (line.State == "QUEUED") stations[line.Station].Queue.Remove(line);
688 line.State = "VOID";
689 MenuItem item = menu[line.Item];
690 if (item.Stock is not null) item.Stock += line.Quantity;
691 voids++;
692 Emit("VOID", "#" + line.Number, line.Item, "x" + line.Quantity);
693 break;
694 }
695 case "STATION":
696 {
697 Station station = RequireStation(args[0]);
698 if (args[1] == "DOWN" && !station.Up)
699 Fail("station " + station.Name + " is already down");
700 if (args[1] == "UP" && station.Up)
701 Fail("station " + station.Name + " is not down");
702 station.Up = args[1] == "UP";
703 if (station.Up) Start(station, now);
704 Emit("STATION", station.Name, args[1]);
705 break;
706 }
707 case "COMP":
708 case "TRANSFER":
709 {
710 OrderLine line = RequireLine(args[0]);
711 Modifiable(line);
712 if (command == "COMP")
713 {
714 if (line.Reason is not null)
715 Fail("#" + line.Number + " is already comped");
716 int amount = line.Amount;
717 Emit("COMP", "#" + line.Number, line.Item, Money(amount), args[1]);
718 comps++;
719 compAmount += amount;
720 line.Reason = args[1];
721 }
722 else
723 {
724 int seat = Number(args[1]);
725 if (seat > line.Check.Size)
726 Fail("seat " + seat + " out of range for table " + line.Check.Table.Number);
727 if (seat == line.Seat)
728 Fail("#" + line.Number + " is already on seat " + seat);
729 Emit("TRANSFER", "#" + line.Number, "seat", line.Seat, "->", seat);
730 line.Seat = seat;
731 }
732 break;
733 }
734 case "DISCOUNT":
735 {
736 Check check = RequireCheck(args[0]);
737 Unlocked(check);
738 check.Discount = Number(args[1]);
739 Emit("DISCOUNT", "table", check.Table.Number, check.Discount + "%");
740 break;
741 }
742 case "BILL":
743 Bill(RequireCheck(args[0]));
744 break;
745
746 case "SPLIT":
747 {
748 Check check = RequireCheck(args[0]);
749 Amounts amount = Calculate(check);
750 if (amount.Due == 0)
751 Fail("nothing to split for table " + args[0]);
752 if (args[1] == "EVEN")
753 {
754 int count = Number(args[2]);
755 if (count > check.Size)
756 Fail("cannot split table " + args[0] + " " + count + " ways");
757 Emit("SPLIT", "table", check.Table.Number, "EVEN", count, "due", Money(amount.Due));
758 int quotient = amount.Due / count;
759 int remainder = amount.Due % count;
760 for (int i = 1; i <= count; ++i)
761 Emit("PAYER", i, Money(quotient + (i <= remainder ? 1 : 0)));
762 }
763 else
764 {
765 if (check.Paid > 0) Fail("table " + args[0] + " has payments");
766 SortedDictionary<int, int> values = SeatSubtotals(check);
767 Dictionary<int, int> discounts = Allocate(values, check.Discount);
768 Dictionary<int, int> after = values.ToDictionary(
769 entry => entry.Key, entry => entry.Value - discounts[entry.Key]);
770 Dictionary<int, int> services = Allocate(after, check.Size >= 6 ? 18 : 0);
771 Dictionary<int, int> taxes = Allocate(after, 8);
772 Emit("SPLIT", "table", check.Table.Number, "BYSEAT", "total", Money(amount.Total));
773 foreach ((int seat, int subtotal) in values)
774 {
775 Emit("SEAT", seat, "subtotal", Money(subtotal),
776 "discount", Money(discounts[seat]), "service", Money(services[seat]),
777 "tax", Money(taxes[seat]),
778 "total", Money(after[seat] + services[seat] + taxes[seat]));
779 }
780 }
781 break;
782 }
783 case "PAY":
784 {
785 Check check = RequireCheck(args[0]);
786 if (args[1] != "CASH" && args[1] != "CARD")
787 Fail("unknown method " + args[1]);
788 Unserved(check);
789 int due = Calculate(check).Due;
790 if (due == 0) Fail("nothing to pay for table " + args[0]);
791 int value = Cents(args[2]);
792 int applied = Math.Min(value, due);
793 check.Paid += applied;
794 payments[args[1]] += applied;
795 if (value < due)
796 {
797 Emit("PAY", "table", check.Table.Number, args[1], Money(value),
798 "remaining", Money(due - value));
799 }
800 else
801 {
802 Emit("PAY", "table", check.Table.Number, args[1], Money(value),
803 args[1] == "CASH" ? "change" : "tip", Money(value - due));
804 if (args[1] == "CARD")
805 {
806 tips += value - due;
807 check.Server.Tips += value - due;
808 }
809 Close(check);
810 }
811 break;
812 }
813 case "CLOSE":
814 {
815 Check check = RequireCheck(args[0]);
816 Unserved(check);
817 int due = Calculate(check).Due;
818 if (due != 0) Fail("table " + args[0] + " still owes " + Money(due));
819 Close(check);
820 break;
821 }
822 case "STATUS":
823 {
824 Check check = RequireCheck(args[0]);
825 Emit("STATUS", "table", check.Table.Number, check.Party, "guests",
826 check.Size, "since", Clock(check.Seated));
827 foreach (OrderLine line in check.Lines)
828 {
829 List<object> parts = new()
830 {
831 "#" + line.Number, "seat", line.Seat, line.Item,
832 "x" + line.Quantity, line.Course, line.State
833 };
834 if (line.State == "COOKING")
835 {
836 parts.Add("until");
837 parts.Add(Clock(line.Ready));
838 }
839 if (line.Reason is not null)
840 {
841 parts.Add("COMP");
842 parts.Add(line.Reason);
843 }
844 Emit(parts.ToArray());
845 }
846 break;
847 }
848 case "KITCHEN":
849 {
850 Emit("KITCHEN", Clock(now));
851 foreach (string name in StationNames)
852 {
853 Station station = stations[name];
854 string cooking = string.Join(" ", station.Cooking
855 .OrderBy(line => line.Ready).ThenBy(line => line.Number)
856 .Select(line => "#" + line.Number + "@" + Clock(line.Ready)));
857 string queue = string.Join(" ", station.Queue.Select(line => "#" + line.Number));
858 Emit(name, station.Up ? "up" : "down", "cooking",
859 cooking.Length > 0 ? cooking : "-", "queue",
860 queue.Length > 0 ? queue : "-");
861 }
862 break;
863 }
864 case "FLOOR":
865 {
866 List<Table> ordered = tables.Values.OrderBy(table => table.Number).ToList();
867 int free = ordered.Count(table => table.Check is null);
868 Emit("FLOOR", Clock(now), "free", free, "seated", ordered.Count - free,
869 "waiting", waiting.Count);
870 foreach (Table table in ordered)
871 {
872 Check? check = table.Check;
873 if (check is null)
874 {
875 Emit("table", table.Number, "seats", table.Seats, "free");
876 continue;
877 }
878 List<object> parts = new()
879 {
880 "table", table.Number, "seats", table.Seats, check.Party,
881 "guests", check.Size, "server", check.Server.Name,
882 "since", Clock(check.Seated), "items", check.Active().Count(),
883 "total", Money(Calculate(check).Total)
884 };
885 if (check.Paid > 0)
886 {
887 parts.Add("paid");
888 parts.Add(Money(check.Paid));
889 }
890 Emit(parts.ToArray());
891 }
892 for (int i = 0; i < waiting.Count; ++i)
893 {
894 WaitingParty party = waiting[i];
895 Emit("WAITING", i + 1, party.Party, party.Size, "since", Clock(party.Joined));
896 }
897 break;
898 }
899 case "REPORT":
900 Report();
901 break;
902 }
903 }

911 }

915 }
916 }
