15func integer(text string) int {
16 value, _ := strconv.Atoi(text)
17 return value
18}
19
20func cents(text string) int {
21 return integer(strings.ReplaceAll(text, ".", ""))
22}
23
24func money(value int) string {
25 return fmt.Sprintf("%d.%02d", value/100, value%100)
26}
27
28func clock(time int) string {
29 minutes := 660 + time
30 return fmt.Sprintf("%02d:%02d", minutes/60, minutes%60)
31}

33func rounded(value, rate int) int {
34 return (value*rate + 50) / 100
35}

71func (check *Check) active() []*OrderLine {
72 active := make([]*OrderLine, 0, len(check.Lines))
73 for _, line := range check.Lines {
74 if line.State != "VOID" {
75 active = append(active, line)
76 }
77 }
78 return active
79}

155func newRestaurant() *Restaurant {
156 r := &Restaurant{
157 Tables: make(map[int]*Table),
158 Menu: make(map[string]*MenuItem),
159 Stations: make(map[string]*Station),
160 Sales: make(map[string]int),
161 Totals: make(map[string]int),
162 Payments: make(map[string]int),
163 Served: make(map[string]int),
164 }
165 for _, name := range stationNames {
166 capacity := 1
167 if name == "BAR" {
168 capacity = 2
169 }
170 r.Stations[name] = &Station{Name: name, Capacity: capacity, Up: true}
171 }
172 for _, name := range courseNames {
173 r.Sales[name] = 0
174 }
175 return r
176}

178func (r *Restaurant) emit(parts ...any) {
179 fields := make([]string, len(parts))
180 for i, part := range parts {
181 fields[i] = fmt.Sprint(part)
182 }
183 r.Output = append(r.Output, strings.Join(fields, " "))
184}

245func modifiable(line *OrderLine) error {
246 if line.State == "VOID" {
247 return fmt.Errorf("#%d is void", line.Number)
248 }
249 if line.Check.Closed {
250 return fmt.Errorf("#%d is on a closed check", line.Number)
251 }
252 return unlocked(line.Check)

255func unserved(check *Check) error {
256 count := 0
257 for _, line := range check.active() {
258 if line.State != "SERVED" {
259 count++
260 }
261 }
262 if count != 0 {
263 return fmt.Errorf("table %d has %d unserved items", check.Table.Number, count)
264 }
265 return nil
266}

268func (r *Restaurant) load(server *Server) int {
269 count := 0
270 for _, table := range r.Tables {
271 if table.Check != nil && table.Check.Server == server {
272 count++
273 }
274 }
275 return count
276}

278func (r *Restaurant) selectTable(size int) *Table {
279 var best *Table
280 for _, table := range r.Tables {
281 if table.Check != nil || table.Seats < size {
282 continue
283 }
284 if best == nil || table.Seats < best.Seats ||
285 table.Seats == best.Seats && table.Number < best.Number {
286 best = table
287 }
288 }
289 return best
290}

292func (r *Restaurant) seat(party string, size int, table *Table) *Check {
293 server := r.Servers[0]
294 for _, candidate := range r.Servers {
295 if r.load(candidate) < r.load(server) {
296 server = candidate
297 }
298 }
299 check := &Check{
300 Party: party, Size: size, Table: table, Server: server, Seated: r.Now,
301 }
302 table.Check = check
303 server.Parties++
304 r.Covers += size
305 r.Parties++
306 return check
307}

309func (r *Restaurant) processWaiting() {
310 remaining := make([]WaitingParty, 0, len(r.Waiting))
311 for _, entry := range r.Waiting {
312 table := r.selectTable(entry.Size)
313 if table == nil {
314 remaining = append(remaining, entry)
315 continue
316 }
317 check := r.seat(entry.Party, entry.Size, table)
318 r.emit("SEATED", entry.Party, entry.Size, "table", table.Number,
319 "server", check.Server.Name, "waited", r.Now-entry.Joined)
320 }
321 r.Waiting = remaining
322}

324func calculate(check *Check) Amounts {
325 result := Amounts{}
326 for _, line := range check.active() {
327 result.Subtotal += line.amount()
328 }
329 result.Discount = rounded(result.Subtotal, check.Discount)
330 after := result.Subtotal - result.Discount
331 if check.Size >= 6 {
332 result.Service = rounded(after, 18)
333 }
334 result.Tax = rounded(after, 8)
335 result.Total = after + result.Service + result.Tax
336 result.Due = result.Total - check.Paid
337 return result
338}

340func (r *Restaurant) close(check *Check) {
341 amounts := calculate(check)
342 r.Turns++
343 r.Stay += r.Now - check.Seated
344 for _, line := range check.active() {
345 r.Sales[line.Course] += line.amount()
346 }
347 r.Totals["subtotal"] += amounts.Subtotal
348 r.Totals["discount"] += amounts.Discount
349 r.Totals["service"] += amounts.Service
350 r.Totals["tax"] += amounts.Tax
351 r.Totals["total"] += amounts.Total
352 check.Server.Sales += amounts.Total
353 check.Closed = true
354 check.Table.Check = nil
355 r.emit("FREE", "table", check.Table.Number, check.Party, "stayed", r.Now-check.Seated)
356 r.processWaiting()
357}

359func start(station *Station, time int) {
360 for station.Up && len(station.Queue) != 0 &&
361 len(station.Cooking) < station.Capacity {
362 line := station.Queue[0]
363 station.Queue = station.Queue[1:]
364 line.State = "COOKING"
365 line.Ready = time + line.Prep
366 line.Started = true
367 station.Cooking = append(station.Cooking, line)
368 }
369}

371func (r *Restaurant) tick(minutes int) {
372 end := r.Now + minutes
373 r.emit("TICK", clock(end))
374 for {
375 var earliest int
376 found := false
377 for _, name := range stationNames {
378 for _, line := range r.Stations[name].Cooking {
379 if !found || line.Ready < earliest {
380 earliest, found = line.Ready, true
381 }
382 }
383 }
384 if !found || earliest > end {
385 break
386 }
387 for _, name := range stationNames {
388 station := r.Stations[name]
389 var finished, remaining []*OrderLine
390 for _, line := range station.Cooking {
391 if line.Ready == earliest {
392 finished = append(finished, line)
393 } else {
394 remaining = append(remaining, line)
395 }
396 }
397 station.Cooking = remaining
398 sort.Slice(finished, func(i, j int) bool {
399 return finished[i].Number < finished[j].Number
400 })
401 for _, line := range finished {
402 line.State = "READY"
403 station.Tickets = append(station.Tickets, earliest-line.Fire)
404 r.emit("READY", clock(earliest), name, "#"+strconv.Itoa(line.Number),
405 line.Item, "x"+strconv.Itoa(line.Quantity), "table",
406 line.Check.Table.Number, "seat", line.Seat)
407 }
408 }
409 for _, name := range stationNames {
410 start(r.Stations[name], earliest)
411 }
412 }
413 r.Now = end
414}

416func seatSubtotals(check *Check) map[int]int {
417 result := make(map[int]int)
418 for _, line := range check.active() {
419 result[line.Seat] += line.amount()
420 }
421 return result

424func sortedSeats(values map[int]int) []int {
425 seats := make([]int, 0, len(values))
426 for seat := range values {
427 seats = append(seats, seat)
428 }
429 sort.Ints(seats)
430 return seats
431}

433func allocate(values map[int]int, rate int) map[int]int {
434 result := make(map[int]int, len(values))
435 total, assigned := 0, 0
436 for seat, value := range values {
437 result[seat] = value * rate / 100
438 total += value
439 assigned += result[seat]
440 }
441 extra := rounded(total, rate) - assigned
442 order := sortedSeats(values)
443 sort.Slice(order, func(i, j int) bool {
444 a, b := order[i], order[j]
445 left, right := values[a]*rate%100, values[b]*rate%100
446 if left != right {
447 return left > right
448 }
449 return a < b
450 })
451 for _, seat := range order[:extra] {
452 result[seat]++
453 }
454 return result
455}

457func (r *Restaurant) bill(check *Check) {
458 r.emit("BILL", "table", check.Table.Number, check.Party, "guests", check.Size)
459 values := seatSubtotals(check)
460 for _, seat := range sortedSeats(values) {
461 r.emit("SEAT", seat, money(values[seat]))
462 for _, line := range check.active() {
463 if line.Seat != seat {
464 continue
465 }
466 parts := []any{
467 "#" + strconv.Itoa(line.Number), line.Item,
468 "x" + strconv.Itoa(line.Quantity), money(line.amount()),
469 }
470 if line.Reason != "" {
471 parts = append(parts, "COMP", line.Reason)
472 }
473 r.emit(parts...)
474 }
475 }
476 amounts := calculate(check)
477 r.emit("SUBTOTAL", money(amounts.Subtotal))
478 if check.Discount != 0 {
479 r.emit("DISCOUNT", strconv.Itoa(check.Discount)+"%", money(amounts.Discount))
480 }
481 if check.Size >= 6 {
482 r.emit("SERVICE", money(amounts.Service))
483 }
484 r.emit("TAX", money(amounts.Tax))
485 r.emit("TOTAL", money(amounts.Total))
486 if check.Paid > 0 {
487 r.emit("PAID", money(check.Paid))
488 r.emit("DUE", money(amounts.Due))
489 }
490}

492func (r *Restaurant) report() {
493 r.emit("REPORT", clock(r.Now))
494 r.emit("COVERS", r.Covers, "parties", r.Parties)
495 stay := "-"
496 if r.Turns != 0 {
497 stay = strconv.Itoa(r.Stay / r.Turns)
498 }
499 r.emit("TURNS", r.Turns, "avg_stay", stay)
500 sales := []any{"SALES"}
501 for _, course := range courseNames {
502 sales = append(sales, course, money(r.Sales[course]))
503 }
504 r.emit(sales...)
505 totals := []any{"TOTALS"}
506 for _, name := range []string{"subtotal", "discount", "service", "tax", "total"} {
507 totals = append(totals, name, money(r.Totals[name]))
508 }
509 r.emit(totals...)
510 r.emit("PAYMENTS", "cash", money(r.Payments["CASH"]),
511 "card", money(r.Payments["CARD"]), "tips", money(r.Tips))
512 r.emit("COMPS", r.Comps, money(r.CompAmount), "VOIDS", r.Voids)
513 names := make([]string, 0, len(r.Served))
514 for name, count := range r.Served {
515 if count > 0 {
516 names = append(names, name)
517 }
518 }
519 sort.Slice(names, func(i, j int) bool {
520 if r.Served[names[i]] != r.Served[names[j]] {
521 return r.Served[names[i]] > r.Served[names[j]]
522 }
523 return names[i] < names[j]
524 })
525 for i := 0; i < len(names) && i < 3; i++ {
526 r.emit("TOP", i+1, names[i], r.Served[names[i]])
527 }
528 for _, name := range stationNames {
529 station := r.Stations[name]
530 if len(station.Tickets) == 0 {
531 r.emit(name, "tickets", 0, "avg", "-", "max", "-", "remakes", station.Remakes)
532 continue
533 }
534 sum, maximum := 0, 0
535 for _, ticket := range station.Tickets {
536 sum += ticket
537 if ticket > maximum {
538 maximum = ticket
539 }
540 }
541 r.emit(name, "tickets", len(station.Tickets), "avg", sum/len(station.Tickets),
542 "max", maximum, "remakes", station.Remakes)
543 }
544 for _, server := range r.Servers {
545 r.emit("SERVER", server.Name, "tables", r.load(server),
546 "parties", server.Parties, "sales", money(server.Sales),
547 "tips", money(server.Tips))
548 }
549}

551func (r *Restaurant) command(cmd string, a []string) error {
552 switch cmd {
553 case "TABLE":
554 number := integer(a[0])
555 if r.Tables[number] != nil {
556 return fmt.Errorf("table %s already exists", a[0])
557 }
558 r.Tables[number] = &Table{Number: number, Seats: integer(a[1])}
559 r.emit("TABLE", number, "seats", integer(a[1]))
560 r.processWaiting()
561
562 case "STAFF":
563 for _, server := range r.Servers {
564 if server.Name == a[0] {
565 return fmt.Errorf("%s is already on duty", a[0])
566 }
567 }
568 r.Servers = append(r.Servers, &Server{Name: a[0]})
569 r.emit("STAFF", a[0])
570
571 case "SEAT":
572 for _, table := range r.Tables {
573 if table.Check != nil && table.Check.Party == a[0] {
574 return fmt.Errorf("%s is already seated at table %d", a[0], table.Number)
575 }
576 }
577 for _, party := range r.Waiting {
578 if party.Party == a[0] {
579 return fmt.Errorf("%s is already waiting", a[0])
580 }
581 }
582 size := integer(a[1])
583 accommodates := false
584 for _, table := range r.Tables {
585 if table.Seats >= size {
586 accommodates = true
587 break
588 }
589 }
590 if !accommodates {
591 return fmt.Errorf("no table for party of %d", size)
592 }
593 if len(r.Servers) == 0 {
594 return fmt.Errorf("no server on duty")
595 }
596 if table := r.selectTable(size); table != nil {
597 check := r.seat(a[0], size, table)
598 r.emit("SEAT", a[0], size, "table", table.Number, "server", check.Server.Name)
599 } else {
600 r.Waiting = append(r.Waiting, WaitingParty{a[0], size, r.Now})
601 r.emit("WAIT", a[0], size, "position", len(r.Waiting))
602 }
603
604 case "CANCEL":
605 for i, party := range r.Waiting {
606 if party.Party == a[0] {
607 r.Waiting = append(r.Waiting[:i], r.Waiting[i+1:]...)
608 r.emit("CANCEL", party.Party, party.Size, "waited", r.Now-party.Joined)
609 return nil
610 }
611 }
612 return fmt.Errorf("%s is not waiting", a[0])
613
614 case "MOVE":
615 source, err := r.table(a[0])
616 if err != nil {
617 return err
618 }
619 destination, err := r.table(a[1])
620 if err != nil {
621 return err
622 }
623 if source.Check == nil {
624 return fmt.Errorf("table %s is free", a[0])
625 }
626 if destination.Check != nil {
627 return fmt.Errorf("table %s is occupied", a[1])
628 }
629 check := source.Check
630 if destination.Seats < check.Size {
631 return fmt.Errorf("table %s is too small for %s", a[1], check.Party)
632 }
633 source.Check = nil
634 destination.Check = check
635 check.Table = destination
636 r.emit("MOVE", check.Party, "table", source.Number, "->", destination.Number)
637 r.processWaiting()
638
639 case "MENU":
640 if _, err := r.station(a[1]); err != nil {
641 return err
642 }
643 if err := requireCourse(a[4]); err != nil {
644 return err
645 }
646 old := r.Menu[a[0]]
647 item := &MenuItem{
648 Name: a[0], Station: a[1], Prep: integer(a[2]),
649 Price: cents(a[3]), Course: a[4],
650 }
651 status := "ADD"
652 if old != nil {
653 item.Stock = old.Stock
654 status = "UPDATE"
655 }
656 r.Menu[a[0]] = item
657 r.emit("MENU", status, item.Name, item.Station,
658 item.Prep, money(item.Price), item.Course)
659
660 case "STOCK":
661 item, err := r.item(a[0])
662 if err != nil {
663 return err
664 }
665 count := integer(a[1])
666 item.Stock = &count
667 r.emit("STOCK", item.Name, count)
668
669 case "HAPPY":
670 r.HappyRate = integer(a[0])
671 r.HappyEnd = r.Now + integer(a[1])
672 r.emit("HAPPY", strconv.Itoa(r.HappyRate)+"%", "until", clock(r.HappyEnd))
673
674 case "ITEMS":
675 items := make([]*MenuItem, 0, len(r.Menu))
676 for _, item := range r.Menu {
677 items = append(items, item)
678 }
679 priority := make(map[string]int)
680 for i, course := range courseNames {
681 priority[course] = i
682 }
683 sort.Slice(items, func(i, j int) bool {
684 if priority[items[i].Course] != priority[items[j].Course] {
685 return priority[items[i].Course] < priority[items[j].Course]
686 }
687 return items[i].Name < items[j].Name
688 })
689 r.emit("ITEMS", len(items))
690 for _, item := range items {
691 stock := "-"
692 if item.Stock != nil {
693 stock = strconv.Itoa(*item.Stock)
694 }
695 r.emit(item.Course, item.Name, item.Station, item.Prep, money(item.Price), stock)
696 }
697
698 case "ORDER":
699 check, err := r.check(a[0])
700 if err != nil {
701 return err
702 }
703 if err := unlocked(check); err != nil {
704 return err
705 }
706 seat := integer(a[1])
707 if seat > check.Size {
708 return fmt.Errorf("seat %d out of range for table %s", seat, a[0])
709 }
710 item, err := r.item(a[2])
711 if err != nil {
712 return err
713 }
714 quantity := integer(a[3])
715 if item.Stock != nil && *item.Stock < quantity {
716 return fmt.Errorf("only %d %s left", *item.Stock, item.Name)
717 }
718 happy := item.Course == "DRINK" && r.Now < r.HappyEnd
719 price := item.Price
720 if happy {
721 price -= rounded(price, r.HappyRate)
722 }
723 line := &OrderLine{
724 Number: len(r.Lines) + 1, Check: check, Seat: seat,
725 Item: item.Name, Quantity: quantity, Station: item.Station,
726 Prep: item.Prep, Price: price, Course: item.Course,
727 State: "ORDERED",
728 }
729 r.Lines = append(r.Lines, line)
730 check.Lines = append(check.Lines, line)
731 parts := []any{
732 "ORDER", "#"+strconv.Itoa(line.Number), "table", check.Table.Number,
733 "seat", seat, item.Name, "x"+strconv.Itoa(quantity), money(line.amount()),
734 }
735 if happy {
736 parts = append(parts, "HAPPY")
737 }
738 r.emit(parts...)
739 if item.Stock != nil {
740 *item.Stock -= quantity
741 if *item.Stock == 0 {
742 r.emit("SOLDOUT", item.Name)
743 }
744 }
745
746 case "FIRE":
747 check, err := r.check(a[0])
748 if err != nil {
749 return err
750 }
751 if err := requireCourse(a[1]); err != nil {
752 return err
753 }
754 firing := make([]*OrderLine, 0)
755 for _, line := range check.Lines {
756 if line.Course == a[1] && line.State == "ORDERED" {
757 firing = append(firing, line)
758 }
759 }
760 if len(firing) == 0 {
761 return fmt.Errorf("nothing to fire for %s at table %s", a[1], a[0])
762 }
763 ordered := []string{"STARTER", "MAIN", "DESSERT"}
764 for _, course := range ordered {
765 if course == a[1] {
766 break
767 }
768 if a[1] == "DRINK" {
769 break
770 }
771 for _, line := range check.Lines {
772 if line.Course == course && line.State == "ORDERED" {
773 return fmt.Errorf("fire %s first", course)
774 }
775 }
776 }
777 parts := []any{"FIRE", "table", check.Table.Number, a[1]}
778 for _, line := range firing {
779 line.State = "QUEUED"
780 line.Fire = r.Now
781 r.Stations[line.Station].Queue = append(r.Stations[line.Station].Queue, line)
782 parts = append(parts, "#"+strconv.Itoa(line.Number))
783 }
784 r.emit(parts...)
785 for _, name := range stationNames {
786 start(r.Stations[name], r.Now)
787 }
788
789 case "TICK":
790 r.tick(integer(a[0]))
791
792 case "SERVE":
793 check, err := r.check(a[0])
794 if err != nil {
795 return err
796 }
797 var ready []*OrderLine
798 for _, line := range check.Lines {
799 if line.State == "READY" {
800 ready = append(ready, line)
801 }
802 }
803 if len(ready) == 0 {
804 return fmt.Errorf("nothing ready for table %s", a[0])
805 }
806 parts := []any{"SERVE", "table", check.Table.Number}
807 for _, line := range ready {
808 line.State = "SERVED"
809 r.Served[line.Item] += line.Quantity
810 parts = append(parts, "#"+strconv.Itoa(line.Number))
811 }
812 r.emit(parts...)
813
814 case "RUSH":
815 line, err := r.line(a[0])
816 if err != nil {
817 return err
818 }
819 if line.State != "QUEUED" {
820 return fmt.Errorf("#%d is not queued", line.Number)
821 }
822 station := r.Stations[line.Station]
823 queue := []*OrderLine{line}
824 for _, current := range station.Queue {
825 if current != line {
826 queue = append(queue, current)
827 }
828 }
829 station.Queue = queue
830 r.emit("RUSH", "#"+strconv.Itoa(line.Number), line.Station)
831
832 case "REMAKE":
833 line, err := r.line(a[0])
834 if err != nil {
835 return err
836 }
837 if line.Check.Closed {
838 return fmt.Errorf("#%d is on a closed check", line.Number)
839 }
840 if line.State != "READY" && line.State != "SERVED" {
841 return fmt.Errorf("#%d is not ready or served", line.Number)
842 }
843 line.State = "QUEUED"
844 line.Fire = r.Now
845 station := r.Stations[line.Station]
846 station.Queue = append([]*OrderLine{line}, station.Queue...)
847 station.Remakes++
848 r.emit("REMAKE", "#"+strconv.Itoa(line.Number), line.Item, line.Station)
849 start(station, r.Now)
850
851 case "VOID":
852 line, err := r.line(a[0])
853 if err != nil {
854 return err
855 }
856 if line.State == "VOID" {
857 return fmt.Errorf("#%d is already void", line.Number)
858 }
859 if line.Started {
860 return fmt.Errorf("#%d has already started", line.Number)
861 }
862 if line.State == "QUEUED" {
863 station := r.Stations[line.Station]
864 queue := make([]*OrderLine, 0, len(station.Queue))
865 for _, current := range station.Queue {
866 if current != line {
867 queue = append(queue, current)
868 }
869 }
870 station.Queue = queue
871 }
872 line.State = "VOID"
873 item := r.Menu[line.Item]
874 if item.Stock != nil {
875 *item.Stock += line.Quantity
876 }
877 r.Voids++
878 r.emit("VOID", "#"+strconv.Itoa(line.Number), line.Item, "x"+strconv.Itoa(line.Quantity))
879
880 case "STATION":
881 station, err := r.station(a[0])
882 if err != nil {
883 return err
884 }
885 if a[1] == "DOWN" && !station.Up {
886 return fmt.Errorf("station %s is already down", a[0])
887 }
888 if a[1] == "UP" && station.Up {
889 return fmt.Errorf("station %s is not down", a[0])
890 }
891 station.Up = a[1] == "UP"
892 if station.Up {
893 start(station, r.Now)
894 }
895 r.emit("STATION", station.Name, a[1])
896
897 case "COMP", "TRANSFER":
898 line, err := r.line(a[0])
899 if err != nil {
900 return err
901 }
902 if err := modifiable(line); err != nil {
903 return err
904 }
905 if cmd == "COMP" {
906 if line.Reason != "" {
907 return fmt.Errorf("#%d is already comped", line.Number)
908 }
909 value := line.amount()
910 r.emit("COMP", "#"+strconv.Itoa(line.Number), line.Item, money(value), a[1])
911 r.Comps++
912 r.CompAmount += value
913 line.Reason = a[1]
914 } else {
915 seat := integer(a[1])
916 if seat > line.Check.Size {
917 return fmt.Errorf("seat %d out of range for table %d",
918 seat, line.Check.Table.Number)
919 }
920 if seat == line.Seat {
921 return fmt.Errorf("#%d is already on seat %d", line.Number, seat)
922 }
923 r.emit("TRANSFER", "#"+strconv.Itoa(line.Number), "seat", line.Seat, "->", seat)
924 line.Seat = seat
925 }
926
927 case "DISCOUNT":
928 check, err := r.check(a[0])
929 if err != nil {
930 return err
931 }
932 if err := unlocked(check); err != nil {
933 return err
934 }
935 check.Discount = integer(a[1])
936 r.emit("DISCOUNT", "table", check.Table.Number,
937 strconv.Itoa(check.Discount)+"%")
938
939 case "BILL":
940 check, err := r.check(a[0])
941 if err != nil {
942 return err
943 }
944 r.bill(check)
945
946 case "SPLIT":
947 check, err := r.check(a[0])
948 if err != nil {
949 return err
950 }
951 amounts := calculate(check)
952 if amounts.Due == 0 {
953 return fmt.Errorf("nothing to split for table %s", a[0])
954 }
955 if a[1] == "EVEN" {
956 count := integer(a[2])
957 if count > check.Size {
958 return fmt.Errorf("cannot split table %s %d ways", a[0], count)
959 }
960 r.emit("SPLIT", "table", check.Table.Number, "EVEN", count,
961 "due", money(amounts.Due))
962 quotient, remainder := amounts.Due/count, amounts.Due%count
963 for i := 1; i <= count; i++ {
964 amount := quotient
965 if i <= remainder {
966 amount++
967 }
968 r.emit("PAYER", i, money(amount))
969 }
970 } else {
971 if check.Paid > 0 {
972 return fmt.Errorf("table %s has payments", a[0])
973 }
974 values := seatSubtotals(check)
975 discounts := allocate(values, check.Discount)
976 after := make(map[int]int)
977 for seat, value := range values {
978 after[seat] = value - discounts[seat]
979 }
980 serviceRate := 0
981 if check.Size >= 6 {
982 serviceRate = 18
983 }
984 services := allocate(after, serviceRate)
985 taxes := allocate(after, 8)
986 r.emit("SPLIT", "table", check.Table.Number, "BYSEAT",
987 "total", money(amounts.Total))
988 for _, seat := range sortedSeats(values) {
989 r.emit("SEAT", seat, "subtotal", money(values[seat]),
990 "discount", money(discounts[seat]),
991 "service", money(services[seat]),
992 "tax", money(taxes[seat]),
993 "total", money(after[seat]+services[seat]+taxes[seat]))
994 }
995 }
996
997 case "PAY":
998 check, err := r.check(a[0])
999 if err != nil {
1000 return err
1001 }
1002 if a[1] != "CASH" && a[1] != "CARD" {
1003 return fmt.Errorf("unknown method %s", a[1])
1004 }
1005 if err := unserved(check); err != nil {
1006 return err
1007 }
1008 due := calculate(check).Due
1009 if due == 0 {
1010 return fmt.Errorf("nothing to pay for table %s", a[0])
1011 }
1012 value := cents(a[2])
1013 applied := value
1014 if applied > due {
1015 applied = due
1016 }
1017 check.Paid += applied
1018 r.Payments[a[1]] += applied
1019 if value < due {
1020 r.emit("PAY", "table", check.Table.Number, a[1],
1021 money(value), "remaining", money(due-value))
1022 } else {
1023 label := "change"
1024 if a[1] == "CARD" {
1025 label = "tip"
1026 r.Tips += value - due
1027 check.Server.Tips += value - due
1028 }
1029 r.emit("PAY", "table", check.Table.Number, a[1],
1030 money(value), label, money(value-due))
1031 r.close(check)
1032 }
1033
1034 case "CLOSE":
1035 check, err := r.check(a[0])
1036 if err != nil {
1037 return err
1038 }
1039 if err := unserved(check); err != nil {
1040 return err
1041 }
1042 due := calculate(check).Due
1043 if due != 0 {
1044 return fmt.Errorf("table %s still owes %s", a[0], money(due))
1045 }
1046 r.close(check)
1047
1048 case "STATUS":
1049 check, err := r.check(a[0])
1050 if err != nil {
1051 return err
1052 }
1053 r.emit("STATUS", "table", check.Table.Number, check.Party, "guests",
1054 check.Size, "since", clock(check.Seated))
1055 for _, line := range check.Lines {
1056 parts := []any{
1057 "#" + strconv.Itoa(line.Number), "seat", line.Seat,
1058 line.Item, "x" + strconv.Itoa(line.Quantity),
1059 line.Course, line.State,
1060 }
1061 if line.State == "COOKING" {
1062 parts = append(parts, "until", clock(line.Ready))
1063 }
1064 if line.Reason != "" {
1065 parts = append(parts, "COMP", line.Reason)
1066 }
1067 r.emit(parts...)
1068 }
1069
1070 case "KITCHEN":
1071 r.emit("KITCHEN", clock(r.Now))
1072 for _, name := range stationNames {
1073 station := r.Stations[name]
1074 cooking := append([]*OrderLine(nil), station.Cooking...)
1075 sort.Slice(cooking, func(i, j int) bool {
1076 if cooking[i].Ready != cooking[j].Ready {
1077 return cooking[i].Ready < cooking[j].Ready
1078 }
1079 return cooking[i].Number < cooking[j].Number
1080 })
1081 cookingText := make([]string, 0, len(cooking))
1082 for _, line := range cooking {
1083 cookingText = append(cookingText,
1084 "#"+strconv.Itoa(line.Number)+"@"+clock(line.Ready))
1085 }
1086 queueText := make([]string, 0, len(station.Queue))
1087 for _, line := range station.Queue {
1088 queueText = append(queueText, "#"+strconv.Itoa(line.Number))
1089 }
1090 status := "down"
1091 if station.Up {
1092 status = "up"
1093 }
1094 first, second := "-", "-"
1095 if len(cookingText) != 0 {
1096 first = strings.Join(cookingText, " ")
1097 }
1098 if len(queueText) != 0 {
1099 second = strings.Join(queueText, " ")
1100 }
1101 r.emit(name, status, "cooking", first, "queue", second)
1102 }
1103
1104 case "FLOOR":
1105 tables := make([]*Table, 0, len(r.Tables))
1106 for _, table := range r.Tables {
1107 tables = append(tables, table)
1108 }
1109 sort.Slice(tables, func(i, j int) bool {
1110 return tables[i].Number < tables[j].Number
1111 })
1112 free := 0
1113 for _, table := range tables {
1114 if table.Check == nil {
1115 free++
1116 }
1117 }
1118 r.emit("FLOOR", clock(r.Now), "free", free,
1119 "seated", len(tables)-free, "waiting", len(r.Waiting))
1120 for _, table := range tables {
1121 check := table.Check
1122 if check == nil {
1123 r.emit("table", table.Number, "seats", table.Seats, "free")
1124 continue
1125 }
1126 parts := []any{
1127 "table", table.Number, "seats", table.Seats,
1128 check.Party, "guests", check.Size, "server", check.Server.Name,
1129 "since", clock(check.Seated), "items", len(check.active()),
1130 "total", money(calculate(check).Total),
1131 }
1132 if check.Paid > 0 {
1133 parts = append(parts, "paid", money(check.Paid))
1134 }
1135 r.emit(parts...)
1136 }
1137 for i, party := range r.Waiting {
1138 r.emit("WAITING", i+1, party.Party, party.Size, "since", clock(party.Joined))
1139 }
1140
1141 case "REPORT":
1142 r.report()
1143 }
1144 return nil
1145}
