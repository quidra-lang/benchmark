8fn integer(s: []const u8) i64 {

11fn amount(s: []const u8) i64 {
12 const dot = std.mem.findScalar(u8, s, '.').?;
13 return integer(s[0..dot]) * 100 + integer(s[dot + 1 ..]);
14}
15fn equal(a: []const u8, b: []const u8) bool {
16 return std.mem.eql(u8, a, b);
17}
18fn rounded(value: i64, rate: i64) i64 {
19 return @divTrunc(value * rate + 50, 100);
20}

71const Totals = struct {
72 subtotal: i64 = 0,
73 discount: i64 = 0,
74 service: i64 = 0,
75 tax: i64 = 0,
76 total: i64 = 0,
77 fn of(c: *const Check) Totals {
78 var t: Totals = .{};
79 for (c.orders.items) |o| if (o.state != .VOID) {
80 t.subtotal += o.charge();
81 };
82 t.discount = rounded(t.subtotal, c.discount);
83 const net = t.subtotal - t.discount;
84 t.service = if (c.size >= 6) rounded(net, 18) else 0;
85 t.tax = rounded(net, 8);
86 t.total = net + t.service + t.tax;
87 return t;
88 }
89 fn add(t: *Totals, other: Totals) void {
90 t.subtotal += other.subtotal;
91 t.discount += other.discount;
92 t.service += other.service;
93 t.tax += other.tax;
94 t.total += other.total;
95 }
96};
97const Seats = struct {
98 present: [21]bool = @splat(false),
99 amounts: [21]i64 = @splat(0),
100 fn of(c: *const Check) Seats {
101 var result: Seats = .{};
102 for (c.orders.items) |o| {
103 if (o.state == .VOID) continue;
104 result.present[o.seat] = true;
105 result.amounts[o.seat] += o.charge();
106 }
107 return result;
108 }
109};
110fn allocate(values: [21]i64, present: [21]bool, rate: i64) [21]i64 {
111 var result: [21]i64 = @splat(0);
112 var total: i64 = 0;
113 var assigned: i64 = 0;
114 for (1..21) |seat| {
115 if (!present[seat]) continue;
116 total += values[seat];
117 result[seat] = @divTrunc(values[seat] * rate, 100);
118 assigned += result[seat];
119 }
120 var remaining = rounded(total, rate) - assigned;
121 var used: [21]bool = @splat(false);
122 while (remaining > 0) : (remaining -= 1) {
123 var best: usize = 0;
124 for (1..21) |seat| {
125 if (!present[seat] or used[seat]) continue;
126 if (best == 0 or @mod(values[seat] * rate, 100) > @mod(values[best] * rate, 100)) best = seat;
127 }
128 result[best] += 1;
129 used[best] = true;
130 }
131 return result;
132}

140 unreachable;

211 fn allServed(self: *Restaurant, c: *Check) Error!void {
212 var count: usize = 0;
213 for (c.orders.items) |o| if (o.state != .VOID and o.state != .SERVED) {
214 count += 1;
215 };
216 if (count > 0) return self.fail("table {d} has {d} unserved items", .{ c.table.number, count });
217 }
218 fn chooseTable(self: *Restaurant, size: usize) ?*Table {
219 var best: ?*Table = null;
220 for (self.tables.values()) |t| {
221 if (t.check != null or t.seats < size) continue;
222 if (best == null or t.seats < best.?.seats or (t.seats == best.?.seats and t.number < best.?.number)) best = t;
223 }
224 return best;
225 }
226 fn seatParty(self: *Restaurant, party: Waiting, t: *Table) Error!*Check {
227 var server = self.servers.values()[0];
228 for (self.servers.values()) |s| if (s.load < server.load) {
229 server = s;
230 };
231 const c = try self.gpa.create(Check);
232 c.* = .{ .party = party.name, .size = party.size, .table = t, .server = server, .seated = self.time };
233 t.check = c;
234 server.load += 1;
235 server.parties += 1;
236 self.covers += @intCast(party.size);
237 self.parties += 1;
238 return c;
239 }
240 fn seatWaiting(self: *Restaurant) Error!void {
241 var i: usize = 0;
242 while (i < self.waiting.items.len) {
243 const p = self.waiting.items[i];
244 const t = self.chooseTable(p.size) orelse {
245 i += 1;
246 continue;
247 };
248 _ = self.waiting.orderedRemove(i);
249 const c = try self.seatParty(p, t);
250 try self.out.print("SEATED {s} {d} table {d} server {s} waited {d}\n", .{ p.name, p.size, t.number, c.server.name, self.time - p.since });
251 }
252 }
253 fn closeCheck(self: *Restaurant, c: *Check) Error!void {
254 const t = Totals.of(c);
255 self.totals.add(t);
256 c.server.sales += t.total;
257 c.server.load -= 1;
258 self.turns += 1;
259 self.stay_total += self.time - c.seated;
260 for (c.orders.items) |o| if (o.state != .VOID) {
261 self.sales[@backingInt(o.course)] += o.charge();
262 };
263 c.closed = true;
264 c.table.check = null;
265 try self.out.print("FREE table {d} {s} stayed {d}\n", .{ c.table.number, c.party, self.time - c.seated });
266 try self.seatWaiting();
267 }
268 fn startCooking(self: *Restaurant, index: usize, time: i64) Error!void {
269 const s = &self.stations[index];
270 const capacity: usize = if (index == @backingInt(StationName.BAR)) 2 else 1;
271 while (s.up and s.queue.items.len > 0 and s.cooking.items.len < capacity) {
272 const o = s.queue.orderedRemove(0);
273 o.state = .COOKING;
274 o.started = true;
275 o.ready = time + o.prep;
276 try s.cooking.append(self.gpa, o);
277 }
278 }
279 fn tick(self: *Restaurant, end: i64) Error!void {
280 try self.out.print("TICK {s}\n", .{try self.clock(end)});
281 while (true) {
282 var next: i64 = std.math.maxInt(i64);
283 for (&self.stations) |*s| for (s.cooking.items) |o| {
284 next = @min(next, o.ready);
285 };
286 if (next > end) break;
287 for (&self.stations, 0..) |*s, index| {
288 std.mem.sort(*Order, s.cooking.items, {}, struct {
289 fn less(_: void, a: *Order, b: *Order) bool {
290 return a.id < b.id;
291 }
292 }.less);
293 var i: usize = 0;
294 while (i < s.cooking.items.len) {
295 const o = s.cooking.items[i];
296 if (o.ready != next) {
297 i += 1;
298 continue;
299 }
300 _ = s.cooking.orderedRemove(i);
301 o.state = .READY;
302 const ticket = next - o.fired;
303 s.tickets += 1;
304 s.ticket_total += ticket;
305 s.ticket_max = @max(s.ticket_max, ticket);
306 const name: StationName = @fromBackingInt(@intCast(index));
307 try self.out.print("READY {s} {s} #{d} {s} x{d} table {d} seat {d}\n", .{ try self.clock(next), @tagName(name), o.id, o.item.name, o.quantity, o.check.table.number, o.seat });
308 }
309 }
310 for (0..4) |i| try self.startCooking(i, next);
311 }
312 self.time = end;
313 }
314 fn bill(self: *Restaurant, c: *Check) Error!void {
315 const seats = Seats.of(c);
316 try self.out.print("BILL table {d} {s} guests {d}\n", .{ c.table.number, c.party, c.size });
317 for (1..21) |seat| {
318 if (!seats.present[seat]) continue;
319 try self.out.print("SEAT {d} {s}\n", .{ seat, try self.money(seats.amounts[seat]) });
320 for (c.orders.items) |o| {
321 if (o.state == .VOID or o.seat != seat) continue;
322 try self.out.print("#{d} {s} x{d} {s}", .{ o.id, o.item.name, o.quantity, try self.money(o.charge()) });
323 if (o.comp) |reason| try self.out.print(" COMP {s}", .{reason});
324 try self.out.writeAll("\n");
325 }
326 }
327 const t = Totals.of(c);
328 try self.out.print("SUBTOTAL {s}\n", .{try self.money(t.subtotal)});
329 if (c.discount > 0) try self.out.print("DISCOUNT {d}% {s}\n", .{ c.discount, try self.money(t.discount) });
330 if (c.size >= 6) try self.out.print("SERVICE {s}\n", .{try self.money(t.service)});
331 try self.out.print("TAX {s}\nTOTAL {s}\n", .{ try self.money(t.tax), try self.money(t.total) });
332 if (c.paid > 0) try self.out.print("PAID {s}\nDUE {s}\n", .{ try self.money(c.paid), try self.money(t.total - c.paid) });
333 }
334 fn split(self: *Restaurant, c: *Check, args: []const []const u8) Error!void {
335 const t = Totals.of(c);
336 const due = t.total - c.paid;
337 if (due == 0) return self.fail("nothing to split for table {d}", .{c.table.number});
338 if (equal(args[0], "EVEN")) {
339 const k = integer(args[1]);
340 if (k > @as(i64, @intCast(c.size))) return self.fail("cannot split table {d} {d} ways", .{ c.table.number, k });
341 try self.out.print("SPLIT table {d} EVEN {d} due {s}\n", .{ c.table.number, k, try self.money(due) });
342 var i: i64 = 1;
343 while (i <= k) : (i += 1) try self.out.print("PAYER {d} {s}\n", .{ i, try self.money(@divTrunc(due, k) + @as(i64, @intFromBool(i <= @mod(due, k)))) });
344 } else {
345 if (c.paid > 0) return self.fail("table {d} has payments", .{c.table.number});
346 const seats = Seats.of(c);
347 const discounts = allocate(seats.amounts, seats.present, c.discount);
348 var net = seats.amounts;
349 for (1..21) |seat| net[seat] -= discounts[seat];
350 const service = allocate(net, seats.present, if (c.size >= 6) 18 else 0);
351 const tax = allocate(net, seats.present, 8);
352 try self.out.print("SPLIT table {d} BYSEAT total {s}\n", .{ c.table.number, try self.money(t.total) });
353 for (1..21) |seat| {
354 if (!seats.present[seat]) continue;
355 try self.out.print("SEAT {d} subtotal {s} discount {s} service {s} tax {s} total {s}\n", .{ seat, try self.money(seats.amounts[seat]), try self.money(discounts[seat]), try self.money(service[seat]), try self.money(tax[seat]), try self.money(net[seat] + service[seat] + tax[seat]) });
356 }
357 }

379 for (self.servers.values()) |s| try self.out.print("SERVER {s} tables {d} parties {d} sales {s} tips {s}\n", .{ s.name, s.load, s.parties, try self.money(s.sales), try self.money(s.tips) });
380 }
381 fn execute(self: *Restaurant, a: []const []const u8) Error!void {
382 const op = std.meta.stringToEnum(Command, a[0]).?;
383 const id: []const u8 = if (a.len > 1) a[1] else "";
384 switch (op) {
385 .TABLE => {
386 const number = integer(id);
387 if (self.tables.contains(number)) return self.fail("table {s} already exists", .{id});
388 const t = try self.gpa.create(Table);
389 t.* = .{ .number = number, .seats = @intCast(integer(a[2])) };
390 try self.tables.put(self.gpa, number, t);
391 try self.out.print("TABLE {d} seats {d}\n", .{ number, t.seats });
392 try self.seatWaiting();
393 },
394 .STAFF => {
395 if (self.servers.contains(id)) return self.fail("{s} is already on duty", .{id});
396 const s = try self.gpa.create(Server);
397 s.* = .{ .name = id };
398 try self.servers.put(self.gpa, id, s);
399 try self.out.print("STAFF {s}\n", .{id});
400 },
401 .SEAT => {
402 for (self.tables.values()) |t| {
403 if (t.check) |c| if (equal(c.party, id)) {
404 return self.fail("{s} is already seated at table {d}", .{ id, t.number });
405 };
406 }
407 for (self.waiting.items) |p| if (equal(p.name, id)) {
408 return self.fail("{s} is already waiting", .{id});
409 };
410 const size: usize = @intCast(integer(a[2]));
411 var fits = false;
412 for (self.tables.values()) |t| if (t.seats >= size) {
413 fits = true;
414 };
415 if (!fits) return self.fail("no table for party of {d}", .{size});
416 if (self.servers.count() == 0) return self.fail("no server on duty", .{});
417 const p = Waiting{ .name = id, .size = size, .since = self.time };
418 if (self.chooseTable(size)) |t| {
419 const c = try self.seatParty(p, t);
420 try self.out.print("SEAT {s} {d} table {d} server {s}\n", .{ id, size, t.number, c.server.name });
421 } else {
422 try self.waiting.append(self.gpa, p);
423 try self.out.print("WAIT {s} {d} position {d}\n", .{ id, size, self.waiting.items.len });
424 }
425 },
426 .CANCEL => {
427 var index: ?usize = null;
428 for (self.waiting.items, 0..) |p, i| if (equal(p.name, id)) {
429 index = i;
430 break;
431 };
432 const p = self.waiting.orderedRemove(index orelse return self.fail("{s} is not waiting", .{id}));
433 try self.out.print("CANCEL {s} {d} waited {d}\n", .{ id, p.size, self.time - p.since });
434 },
435 .MOVE => {
436 const from = try self.table(id);
437 const to = try self.table(a[2]);
438 const c = from.check orelse return self.fail("table {s} is free", .{id});
439 if (to.check != null) return self.fail("table {s} is occupied", .{a[2]});
440 if (to.seats < c.size) return self.fail("table {s} is too small for {s}", .{ a[2], c.party });
441 from.check = null;
442 to.check = c;
443 c.table = to;
444 try self.out.print("MOVE {s} table {d} -> {d}\n", .{ c.party, from.number, to.number });
445 try self.seatWaiting();
446 },
447 .MENU => {
448 const station = try self.stationName(a[2]);
449 const course = try self.courseName(a[5]);
450 const existing = self.menu.get(id);
451 const m = existing orelse try self.gpa.create(MenuItem);
452 if (existing == null) m.* = .{ .name = id, .station = station, .prep = integer(a[3]), .price = amount(a[4]), .course = course } else {
453 m.station = station;
454 m.prep = integer(a[3]);
455 m.price = amount(a[4]);
456 m.course = course;
457 }
458 try self.menu.put(self.gpa, id, m);
459 try self.out.print("MENU {s} {s} {s} {d} {s} {s}\n", .{ if (existing == null) "ADD" else "UPDATE", id, a[2], m.prep, try self.money(m.price), a[5] });
460 },
461 .STOCK => {
462 const m = try self.item(id);
463 m.stock = integer(a[2]);
464 try self.out.print("STOCK {s} {d}\n", .{ id, m.stock.? });
465 },
466 .HAPPY => {
467 self.happy_rate = integer(id);
468 self.happy_end = self.time + integer(a[2]);
469 try self.out.print("HAPPY {d}% until {s}\n", .{ self.happy_rate, try self.clock(self.happy_end) });
470 },
471 .ITEMS => {
472 const items = try self.gpa.dupe(*MenuItem, self.menu.values());
473 std.mem.sort(*MenuItem, items, {}, struct {
474 fn less(_: void, x: *MenuItem, y: *MenuItem) bool {
475 return if (x.course != y.course) @backingInt(x.course) < @backingInt(y.course) else std.mem.order(u8, x.name, y.name) == .lt;
476 }
477 }.less);
478 try self.out.print("ITEMS {d}\n", .{items.len});
479 for (items) |m| {
480 try self.out.print("{s} {s} {s} {d} {s} ", .{ @tagName(m.course), m.name, @tagName(m.station), m.prep, try self.money(m.price) });
481 if (m.stock) |stock| try self.out.print("{d}\n", .{stock}) else try self.out.writeAll("-\n");
482 }
483 },
484 .ORDER, .FIRE, .SERVE, .DISCOUNT, .BILL, .SPLIT, .PAY, .CLOSE, .STATUS => {
485 const c = try self.check(id);
486 switch (op) {
487 .ORDER => {
488 try self.unlocked(c);
489 const seat: usize = @intCast(integer(a[2]));
490 try self.validSeat(c, seat);
491 const m = try self.item(a[3]);
492 const quantity = integer(a[4]);
493 if (m.stock != null and m.stock.? < quantity) return self.fail("only {d} {s} left", .{ m.stock.?, m.name });
494 const happy = m.course == .DRINK and self.time < self.happy_end;
495 const o = try self.gpa.create(Order);
496 o.* = .{ .id = self.orders.items.len + 1, .check = c, .seat = seat, .item = m, .quantity = quantity, .station = m.station, .prep = m.prep, .price = m.price - (if (happy) rounded(m.price, self.happy_rate) else 0), .course = m.course };
497 try self.orders.append(self.gpa, o);
498 try c.orders.append(self.gpa, o);
499 try self.out.print("ORDER #{d} table {d} seat {d} {s} x{d} {s}{s}\n", .{ o.id, c.table.number, seat, m.name, quantity, try self.money(o.charge()), if (happy) " HAPPY" else "" });
500 if (m.stock) |stock| {
501 m.stock = stock - quantity;
502 if (m.stock.? == 0) try self.out.print("SOLDOUT {s}\n", .{m.name});
503 }
504 },
505 .FIRE => {
506 const course = try self.courseName(a[2]);
507 var found = false;
508 for (c.orders.items) |o| if (o.course == course and o.state == .ORDERED) {
509 found = true;
510 };
511 if (!found) return self.fail("nothing to fire for {s} at table {d}", .{ a[2], c.table.number });
512 if (course == .MAIN or course == .DESSERT) {
513 for (1..@backingInt(course)) |earlier| {
514 for (c.orders.items) |o| if (@backingInt(o.course) == earlier and o.state == .ORDERED) {
515 return self.fail("fire {s} first", .{@tagName(o.course)});
516 };
517 }
518 }
519 try self.out.print("FIRE table {d} {s}", .{ c.table.number, a[2] });
520 for (c.orders.items) |o| {
521 if (o.course != course or o.state != .ORDERED) continue;
522 o.state = .QUEUED;
523 o.fired = self.time;
524 try self.stations[@backingInt(o.station)].queue.append(self.gpa, o);
525 try self.out.print(" #{d}", .{o.id});
526 }
527 try self.out.writeAll("\n");
528 for (0..4) |i| try self.startCooking(i, self.time);
529 },
530 .SERVE => {
531 var found = false;
532 for (c.orders.items) |o| if (o.state == .READY) {
533 found = true;
534 };
535 if (!found) return self.fail("nothing ready for table {d}", .{c.table.number});
536 try self.out.print("SERVE table {d}", .{c.table.number});
537 for (c.orders.items) |o| {
538 if (o.state != .READY) continue;
539 o.state = .SERVED;
540 o.item.served += o.quantity;
541 try self.out.print(" #{d}", .{o.id});
542 }
543 try self.out.writeAll("\n");
544 },
545 .DISCOUNT => {
546 try self.unlocked(c);
547 c.discount = integer(a[2]);
548 try self.out.print("DISCOUNT table {d} {d}%\n", .{ c.table.number, c.discount });
549 },
550 .BILL => try self.bill(c),
551 .SPLIT => try self.split(c, a[2..]),
552 .PAY => {
553 const cash = equal(a[2], "CASH");
554 if (!cash and !equal(a[2], "CARD")) return self.fail("unknown method {s}", .{a[2]});
555 try self.allServed(c);
556 const due = Totals.of(c).total - c.paid;
557 if (due == 0) return self.fail("nothing to pay for table {d}", .{c.table.number});
558 const paid = amount(a[3]);
559 const applied = @min(paid, due);
560 c.paid += applied;
561 if (cash) self.cash += applied else self.card += applied;
562 if (paid < due) {
563 try self.out.print("PAY table {d} {s} {s} remaining {s}\n", .{ c.table.number, a[2], try self.money(paid), try self.money(due - paid) });
564 } else {
565 if (!cash) {
566 self.tips += paid - due;
567 c.server.tips += paid - due;
568 }
569 try self.out.print("PAY table {d} {s} {s} {s} {s}\n", .{ c.table.number, a[2], try self.money(paid), if (cash) "change" else "tip", try self.money(paid - due) });
570 try self.closeCheck(c);
571 }
572 },
573 .CLOSE => {
574 try self.allServed(c);
575 const due = Totals.of(c).total - c.paid;
576 if (due != 0) return self.fail("table {d} still owes {s}", .{ c.table.number, try self.money(due) });
577 try self.closeCheck(c);
578 },
579 .STATUS => {
580 try self.out.print("STATUS table {d} {s} guests {d} since {s}\n", .{ c.table.number, c.party, c.size, try self.clock(c.seated) });
581 for (c.orders.items) |o| {
582 try self.out.print("#{d} seat {d} {s} x{d} {s} {s}", .{ o.id, o.seat, o.item.name, o.quantity, @tagName(o.course), @tagName(o.state) });
583 if (o.state == .COOKING) try self.out.print(" until {s}", .{try self.clock(o.ready)});
584 if (o.comp) |reason| try self.out.print(" COMP {s}", .{reason});
585 try self.out.writeAll("\n");
586 }
587 },
588 else => unreachable,
589 }
590 },
591 .TICK => try self.tick(self.time + integer(id)),
592 .RUSH, .REMAKE, .VOID, .COMP, .TRANSFER => {
593 const o = try self.order(id);
594 const s = &self.stations[@backingInt(o.station)];
595 switch (op) {
596 .RUSH => {
597 if (o.state != .QUEUED) return self.fail("#{d} is not queued", .{o.id});
598 removeOrder(&s.queue, o);
599 try s.queue.insert(self.gpa, 0, o);
600 try self.out.print("RUSH #{d} {s}\n", .{ o.id, @tagName(o.station) });
601 },
602 .REMAKE => {
603 if (o.check.closed) return self.fail("#{d} is on a closed check", .{o.id});
604 if (o.state != .READY and o.state != .SERVED) return self.fail("#{d} is not ready or served", .{o.id});
605 o.state = .QUEUED;
606 o.fired = self.time;
607 try s.queue.insert(self.gpa, 0, o);
608 s.remakes += 1;
609 try self.out.print("REMAKE #{d} {s} {s}\n", .{ o.id, o.item.name, @tagName(o.station) });
610 try self.startCooking(@backingInt(o.station), self.time);
611 },
612 .VOID => {
613 if (o.state == .VOID) return self.fail("#{d} is already void", .{o.id});
614 if (o.started) return self.fail("#{d} has already started", .{o.id});
615 if (o.state == .QUEUED) removeOrder(&s.queue, o);
616 o.state = .VOID;
617 if (o.item.stock) |stock| o.item.stock = stock + o.quantity;
618 self.voids += 1;
619 try self.out.print("VOID #{d} {s} x{d}\n", .{ o.id, o.item.name, o.quantity });
620 },
621 .COMP => {
622 try self.modifiable(o);
623 if (o.comp != null) return self.fail("#{d} is already comped", .{o.id});
624 const value = o.charge();
625 self.comps += 1;
626 self.comp_amount += value;
627 o.comp = a[2];
628 try self.out.print("COMP #{d} {s} {s} {s}\n", .{ o.id, o.item.name, try self.money(value), a[2] });
629 },
630 .TRANSFER => {
631 try self.modifiable(o);
632 const seat: usize = @intCast(integer(a[2]));
633 try self.validSeat(o.check, seat);
634 if (seat == o.seat) return self.fail("#{d} is already on seat {d}", .{ o.id, seat });
635 try self.out.print("TRANSFER #{d} seat {d} -> {d}\n", .{ o.id, o.seat, seat });
636 o.seat = seat;
637 },
638 else => unreachable,
639 }
640 },
641 .STATION => {
642 const name = try self.stationName(id);
643 const s = &self.stations[@backingInt(name)];
644 const up = equal(a[2], "UP");
645 if (s.up == up) return self.fail("station {s} is {s}", .{ id, if (up) "not down" else "already down" });
646 s.up = up;
647 if (up) try self.startCooking(@backingInt(name), self.time);
648 try self.out.print("STATION {s} {s}\n", .{ id, a[2] });
649 },
650 .KITCHEN => {
651 try self.out.print("KITCHEN {s}\n", .{try self.clock(self.time)});
652 for (&self.stations, 0..) |*s, i| {
653 const name: StationName = @fromBackingInt(@intCast(i));
654 const cooking = try self.gpa.dupe(*Order, s.cooking.items);
655 std.mem.sort(*Order, cooking, {}, struct {
656 fn less(_: void, x: *Order, y: *Order) bool {
657 return if (x.ready != y.ready) x.ready < y.ready else x.id < y.id;
658 }
659 }.less);
660 try self.out.print("{s} {s} cooking", .{ @tagName(name), if (s.up) "up" else "down" });
661 for (cooking) |o| try self.out.print(" #{d}@{s}", .{ o.id, try self.clock(o.ready) });
662 if (cooking.len == 0) try self.out.writeAll(" -");
663 try self.out.writeAll(" queue");
664 for (s.queue.items) |o| try self.out.print(" #{d}", .{o.id});
665 if (s.queue.items.len == 0) try self.out.writeAll(" -");
666 try self.out.writeAll("\n");
667 }
668 },
669 .FLOOR => {
670 const ordered = try self.gpa.dupe(*Table, self.tables.values());
671 std.mem.sort(*Table, ordered, {}, struct {
672 fn less(_: void, x: *Table, y: *Table) bool {
673 return x.number < y.number;
674 }
675 }.less);
676 var free: usize = 0;
677 for (ordered) |t| if (t.check == null) {
678 free += 1;
679 };
680 try self.out.print("FLOOR {s} free {d} seated {d} waiting {d}\n", .{ try self.clock(self.time), free, ordered.len - free, self.waiting.items.len });
681 for (ordered) |t| {
682 try self.out.print("table {d} seats {d}", .{ t.number, t.seats });
683 if (t.check) |c| {
684 try self.out.print(" {s} guests {d} server {s} since {s} items {d} total {s}", .{ c.party, c.size, c.server.name, try self.clock(c.seated), c.activeCount(), try self.money(Totals.of(c).total) });
685 if (c.paid > 0) try self.out.print(" paid {s}", .{try self.money(c.paid)});
686 } else try self.out.writeAll(" free");
687 try self.out.writeAll("\n");
688 }
689 for (self.waiting.items, 0..) |p, i| try self.out.print("WAITING {d} {s} {d} since {s}\n", .{ i + 1, p.name, p.size, try self.clock(p.since) });
690 },
691 .REPORT => try self.report(),
692 }
693 }
694};
695pub fn main(init: std.process.Init) !void {
696 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
697 defer arena.deinit();
698 const gpa = arena.allocator();
699 var input_buffer: [4096]u8 = undefined;
700 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
701 var output_buffer: [8192]u8 = undefined;
702 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
703 var restaurant = Restaurant{ .gpa = gpa, .out = &output.interface };
704 const n: usize = @intCast(integer((try input.interface.takeDelimiter('\n')).?));
705 for (0..n) |_| {
706 const line = try gpa.dupe(u8, (try input.interface.takeDelimiter('\n')).?);
707 var tokens: std.ArrayList([]const u8) = .empty;
708 var it = std.mem.splitScalar(u8, line, ' ');
709 while (it.next()) |token| try tokens.append(gpa, token);
710 restaurant.execute(tokens.items) catch |err| {
711 if (err != error.BadCommand) return err;
712 try output.interface.print("ERROR {s}\n", .{restaurant.error_message});
713 };
714 }
715 try output.interface.flush();
716}
