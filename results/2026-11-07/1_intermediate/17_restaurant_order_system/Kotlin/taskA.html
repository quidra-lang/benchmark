6private val stations = listOf("GRILL", "FRY", "COLD", "BAR")

9private class CommandError(message: String) : RuntimeException(message)

11private fun cents(text: String): Long = text.replace(".", "").toLong()
12private fun money(value: Long): String {
13 val magnitude = abs(value)
14 return (if (value < 0) "-" else "") + magnitude / 100 + "." +
15 (magnitude % 100).toString().padStart(2, '0')
16}
17private fun clock(time: Int): String {
18 val minutes = 660 + time
19 return (minutes / 60).toString().padStart(2, '0') + ":" +
20 (minutes % 60).toString().padStart(2, '0')
21}
22private fun rounded(value: Long, rate: Int): Long = (value * rate + 50) / 100

69private class Restaurant {
70 private var now = 0
71 private val tables = mutableMapOf<Int, TableInfo>()
72 private val servers = mutableListOf<Staff>()
73 private val waiting = mutableListOf<WaitingEntry>()
74 private val menu = mutableMapOf<String, MenuItem>()

100 private fun fail(message: String): Nothing = throw CommandError(message)

102 private fun table(text: String): TableInfo =
103 tables[text.toInt()] ?: fail("unknown table $text")
104
105 private fun check(text: String): Check =
106 table(text).check ?: fail("table $text is free")

108 private fun line(text: String): OrderLine {
109 val number = text.toInt()
110 if (number !in 1..lines.size) fail("unknown line $text")

114 private fun unlocked(check: Check) {
115 if (check.paid > 0L) fail("check for table " + check.table.number + " is locked")
116 }
117
118 private fun modifiable(line: OrderLine) {
119 if (line.status == "VOID") fail("#" + line.id + " is void")
120 if (line.check.closed) fail("#" + line.id + " is on a closed check")
121 unlocked(line.check)
122 }

124 private fun active(check: Check): List<OrderLine> =
125 check.lines.filter { it.status != "VOID" }
126
127 private fun unserved(check: Check) {
128 val count = active(check).count { it.status != "SERVED" }
129 if (count > 0) fail("table " + check.table.number + " has $count unserved items")
130 }

132 private fun station(name: String): StationState =
133 stationState[name] ?: fail("unknown station $name")
134
135 private fun course(name: String) {
136 if (name !in courses) fail("unknown course $name")
137 }
138
139 private fun item(name: String): MenuItem = menu[name] ?: fail("unknown item $name")
140
141 private fun load(server: Staff): Int =
142 tables.values.count { it.check?.server === server }
143
144 private fun selectTable(size: Int): TableInfo? =
145 tables.values.filter { it.check == null && it.seats >= size }
146 .minWithOrNull(compareBy<TableInfo>({ it.seats }, { it.number }))
147
148 private fun seat(party: String, size: Int, table: TableInfo): Check {
149 val server = servers.withIndex().minWithOrNull(
150 compareBy<IndexedValue<Staff>>({ load(it.value) }, { it.index })
151 )!!.value
152 val check = Check(party, size, table, server, now)
153 table.check = check
154 server.parties++
155 covers += size
156 parties++
157 return check
158 }
159
160 private fun processWaiting() {
161 val remaining = mutableListOf<WaitingEntry>()
162 for (entry in waiting) {
163 val table = selectTable(entry.size)
164 if (table == null) {
165 remaining.add(entry)
166 } else {
167 val check = seat(entry.party, entry.size, table)
168 emit("SEATED", entry.party, entry.size, "table", table.number,
169 "server", check.server.name, "waited", now - entry.joined)
170 }
171 }
172 waiting.clear()
173 waiting.addAll(remaining)
174 }
175
176 private fun amount(line: OrderLine): Long =
177 if (line.reason == null) line.price * line.quantity else 0L
178
179 private fun amounts(check: Check): Amounts {
180 val subtotal = active(check).sumOf { amount(it) }
181 val discount = rounded(subtotal, check.discount)
182 val after = subtotal - discount
183 val service = if (check.size >= 6) rounded(after, 18) else 0L
184 val tax = rounded(after, 8)
185 val total = after + service + tax
186 return Amounts(subtotal, discount, service, tax, total, total - check.paid)
187 }
188
189 private fun close(check: Check) {
190 val amounts = amounts(check)
191 turns++
192 totalStay += now - check.seated
193 for (line in active(check)) {
194 sales[line.course] = sales.getValue(line.course) + amount(line)
195 }
196 val updates = mapOf(
197 "subtotal" to amounts.subtotal, "discount" to amounts.discount,
198 "service" to amounts.service, "tax" to amounts.tax, "total" to amounts.total
199 )
200 for ((name, value) in updates) totals[name] = totals.getValue(name) + value
201 check.server.sales += amounts.total
202 check.closed = true
203 check.table.check = null
204 emit("FREE", "table", check.table.number, check.party, "stayed", now - check.seated)
205 processWaiting()
206 }

208 private fun start(station: StationState, time: Int) {
209 while (station.up && station.queue.isNotEmpty() &&
210 station.cooking.size < station.capacity) {
211 val line = station.queue.removeAt(0)
212 line.status = "COOKING"
213 line.ready = time + line.prep
214 line.started = true
215 station.cooking.add(line)
216 }
217 }
218
219 private fun tick(minutes: Int) {
220 val finish = now + minutes
221 emit("TICK", clock(finish))
222 while (true) {
223 val cooking = stations.flatMap { station(it).cooking }
224 val time = cooking.minOfOrNull { it.ready } ?: break
225 if (time > finish) break
226 for (name in stations) {
227 val station = station(name)
228 val finished = station.cooking.filter { it.ready == time }.sortedBy { it.id }
229 station.cooking.removeAll { it.ready == time }
230 for (line in finished) {
231 line.status = "READY"
232 station.tickets.add(time - line.fire)
233 emit("READY", clock(time), name, "#" + line.id, line.item,
234 "x" + line.quantity, "table", line.check.table.number, "seat", line.seat)
235 }
236 }
237 for (name in stations) start(station(name), time)
238 }
239 now = finish
240 }

242 private fun seatTotals(check: Check): Map<Int, Long> {
243 val sums = mutableMapOf<Int, Long>()
244 for (line in active(check)) {
245 sums[line.seat] = (sums[line.seat] ?: 0L) + amount(line)
246 }
247 return sums
248 }
249
250 private fun allocate(values: Map<Int, Long>, rate: Int): Map<Int, Long> {
251 val result = values.mapValues { (_, value) -> value * rate / 100 }.toMutableMap()
252 val remaining = rounded(values.values.sum(), rate) - result.values.sum()
253 val order = values.keys.sortedWith(
254 compareBy<Int>({ -(values.getValue(it) * rate % 100) }, { it })
255 )
256 for (i in 0 until remaining.toInt()) {
257 val seat = order[i]
258 result[seat] = result.getValue(seat) + 1
259 }
260 return result
261 }
262
263 private fun bill(check: Check) {
264 emit("BILL", "table", check.table.number, check.party, "guests", check.size)
265 val totals = seatTotals(check)
266 for (seat in totals.keys.sorted()) {
267 emit("SEAT", seat, money(totals.getValue(seat)))
268 for (line in active(check)) {
269 if (line.seat != seat) continue
270 val parts = mutableListOf<Any>("#" + line.id, line.item,
271 "x" + line.quantity, money(amount(line)))
272 if (line.reason != null) parts.addAll(listOf("COMP", line.reason!!))
273 emit(*parts.toTypedArray())
274 }
275 }
276 val a = amounts(check)
277 emit("SUBTOTAL", money(a.subtotal))
278 if (check.discount != 0) emit("DISCOUNT", check.discount.toString() + "%", money(a.discount))
279 if (check.size >= 6) emit("SERVICE", money(a.service))
280 emit("TAX", money(a.tax))
281 emit("TOTAL", money(a.total))
282 if (check.paid > 0L) {
283 emit("PAID", money(check.paid))
284 emit("DUE", money(a.due))
285 }
286 }

288 private fun report() {
289 emit("REPORT", clock(now))
290 emit("COVERS", covers, "parties", parties)
291 emit("TURNS", turns, "avg_stay", if (turns > 0) totalStay / turns else "-")
292 val salesLine = mutableListOf<Any>("SALES")
293 for (course in courses) salesLine.addAll(listOf(course, money(sales.getValue(course))))
294 emit(*salesLine.toTypedArray())
295 val totalsLine = mutableListOf<Any>("TOTALS")
296 for (key in listOf("subtotal", "discount", "service", "tax", "total")) {
297 totalsLine.addAll(listOf(key, money(totals.getValue(key))))
298 }
299 emit(*totalsLine.toTypedArray())
300 emit("PAYMENTS", "cash", money(payments.getValue("CASH")),
301 "card", money(payments.getValue("CARD")), "tips", money(tips))
302 emit("COMPS", comps, money(compAmount), "VOIDS", voids)
303 val mostServed = served.entries.filter { it.value > 0 }
304 .sortedWith(compareBy<Map.Entry<String, Int>>({ -it.value }, { it.key }))
305 .take(3)
306 for ((i, entry) in mostServed.withIndex()) emit("TOP", i + 1, entry.key, entry.value)
307 for (name in stations) {
308 val station = station(name)
309 val tickets = station.tickets
310 emit(name, "tickets", tickets.size,
311 "avg", if (tickets.isEmpty()) "-" else (tickets.sum() / tickets.size).toString(),
312 "max", if (tickets.isEmpty()) "-" else tickets.max().toString(),
313 "remakes", station.remakes)
314 }
315 for (server in servers) {
316 emit("SERVER", server.name, "tables", load(server), "parties", server.parties,
317 "sales", money(server.sales), "tips", money(server.tips))
318 }
319 }

321 private fun run(command: String, a: List<String>) {
322 when (command) {
323 "TABLE" -> {
324 val number = a[0].toInt()
325 if (tables.containsKey(number)) fail("table " + a[0] + " already exists")
326 val table = TableInfo(number, a[1].toInt())
327 tables[number] = table
328 emit("TABLE", number, "seats", table.seats)
329 processWaiting()
330 }
331 "STAFF" -> {
332 if (servers.any { it.name == a[0] }) fail(a[0] + " is already on duty")
333 servers.add(Staff(a[0]))
334 emit("STAFF", a[0])
335 }
336 "SEAT" -> {
337 val seated = tables.values.firstOrNull { it.check?.party == a[0] }
338 if (seated != null) fail(a[0] + " is already seated at table " + seated.number)
339 if (waiting.any { it.party == a[0] }) fail(a[0] + " is already waiting")
340 val size = a[1].toInt()
341 if (tables.values.none { it.seats >= size }) fail("no table for party of $size")
342 if (servers.isEmpty()) fail("no server on duty")
343 val table = selectTable(size)
344 if (table == null) {
345 waiting.add(WaitingEntry(a[0], size, now))
346 emit("WAIT", a[0], size, "position", waiting.size)
347 } else {
348 val check = seat(a[0], size, table)
349 emit("SEAT", a[0], size, "table", table.number, "server", check.server.name)
350 }
351 }
352 "CANCEL" -> {
353 val index = waiting.indexOfFirst { it.party == a[0] }
354 if (index < 0) fail(a[0] + " is not waiting")
355 val entry = waiting.removeAt(index)
356 emit("CANCEL", entry.party, entry.size, "waited", now - entry.joined)
357 }
358 "MOVE" -> {
359 val from = table(a[0])
360 val to = table(a[1])
361 val check = from.check ?: fail("table " + a[0] + " is free")
362 if (to.check != null) fail("table " + a[1] + " is occupied")
363 if (to.seats < check.size) fail("table " + a[1] + " is too small for " + check.party)
364 from.check = null
365 to.check = check
366 check.table = to
367 emit("MOVE", check.party, "table", from.number, "->", to.number)
368 processWaiting()
369 }
370 "MENU" -> {
371 station(a[1])
372 course(a[4])
373 val previous = menu[a[0]]
374 val item = MenuItem(a[0], a[1], a[2].toInt(), cents(a[3]), a[4], previous?.stock)
375 menu[a[0]] = item
376 emit("MENU", if (previous == null) "ADD" else "UPDATE",
377 item.name, item.station, item.prep, money(item.price), item.course)
378 }
379 "STOCK" -> {
380 val item = item(a[0])
381 item.stock = a[1].toInt()
382 emit("STOCK", item.name, item.stock!!)
383 }
384 "HAPPY" -> {
385 happyRate = a[0].toInt()
386 happyEnd = now + a[1].toInt()
387 emit("HAPPY", happyRate.toString() + "%", "until", clock(happyEnd))
388 }
389 "ITEMS" -> {
390 val items = menu.values.sortedWith(compareBy<MenuItem>({ courses.indexOf(it.course) }, { it.name }))
391 emit("ITEMS", items.size)
392 for (item in items) {
393 emit(item.course, item.name, item.station, item.prep,
394 money(item.price), item.stock?.toString() ?: "-")
395 }
396 }
397 "ORDER" -> {
398 val check = check(a[0])
399 unlocked(check)
400 val seat = a[1].toInt()
401 if (seat > check.size) fail("seat $seat out of range for table " + a[0])
402 val item = item(a[2])
403 val quantity = a[3].toInt()
404 if (item.stock != null && item.stock!! < quantity) {
405 fail("only " + item.stock + " " + item.name + " left")
406 }
407 val happy = item.course == "DRINK" && now < happyEnd
408 val price = item.price - if (happy) rounded(item.price, happyRate) else 0L
409 val line = OrderLine(lines.size + 1, check, seat, item.name, quantity,
410 item.station, item.prep, price, item.course)
411 lines.add(line)
412 check.lines.add(line)
413 val parts = mutableListOf<Any>("ORDER", "#" + line.id, "table", check.table.number,
414 "seat", seat, item.name, "x" + quantity, money(amount(line)))
415 if (happy) parts.add("HAPPY")
416 emit(*parts.toTypedArray())
417 if (item.stock != null) {
418 item.stock = item.stock!! - quantity
419 if (item.stock == 0) emit("SOLDOUT", item.name)
420 }
421 }
422 "FIRE" -> {
423 val check = check(a[0])
424 course(a[1])
425 val orders = check.lines.filter { it.course == a[1] && it.status == "ORDERED" }
426 if (orders.isEmpty()) fail("nothing to fire for " + a[1] + " at table " + a[0])
427 if (a[1] == "MAIN" || a[1] == "DESSERT") {
428 val earlier = if (a[1] == "MAIN") listOf("STARTER") else listOf("STARTER", "MAIN")
429 for (name in earlier) {
430 if (check.lines.any { it.course == name && it.status == "ORDERED" }) {
431 fail("fire $name first")
432 }
433 }
434 }
435 for (line in orders) {
436 line.status = "QUEUED"
437 line.fire = now
438 station(line.station).queue.add(line)
439 }
440 val ids = orders.map { "#" + it.id }
441 emit("FIRE", "table", check.table.number, a[1], *ids.toTypedArray())
442 for (name in stations) start(station(name), now)
443 }
444 "TICK" -> tick(a[0].toInt())
445 "SERVE" -> {
446 val check = check(a[0])
447 val ready = check.lines.filter { it.status == "READY" }
448 if (ready.isEmpty()) fail("nothing ready for table " + a[0])
449 for (line in ready) {
450 line.status = "SERVED"
451 served[line.item] = (served[line.item] ?: 0) + line.quantity
452 }
453 emit("SERVE", "table", check.table.number, *ready.map { "#" + it.id }.toTypedArray())
454 }
455 "RUSH" -> {
456 val line = line(a[0])
457 if (line.status != "QUEUED") fail("#" + line.id + " is not queued")
458 val station = station(line.station)
459 station.queue.remove(line)
460 station.queue.add(0, line)
461 emit("RUSH", "#" + line.id, line.station)
462 }
463 "REMAKE" -> {
464 val line = line(a[0])
465 if (line.check.closed) fail("#" + line.id + " is on a closed check")
466 if (line.status !in setOf("READY", "SERVED")) {
467 fail("#" + line.id + " is not ready or served")
468 }
469 line.status = "QUEUED"
470 line.fire = now
471 val station = station(line.station)
472 station.queue.add(0, line)
473 station.remakes++
474 emit("REMAKE", "#" + line.id, line.item, line.station)
475 start(station, now)
476 }
477 "VOID" -> {
478 val line = line(a[0])
479 if (line.status == "VOID") fail("#" + line.id + " is already void")
480 if (line.started) fail("#" + line.id + " has already started")
481 if (line.status == "QUEUED") station(line.station).queue.remove(line)
482 line.status = "VOID"
483 val item = menu.getValue(line.item)
484 if (item.stock != null) item.stock = item.stock!! + line.quantity
485 voids++
486 emit("VOID", "#" + line.id, line.item, "x" + line.quantity)
487 }
488 "STATION" -> {
489 val station = station(a[0])
490 if (a[1] == "DOWN" && !station.up) fail("station " + a[0] + " is already down")
491 if (a[1] == "UP" && station.up) fail("station " + a[0] + " is not down")
492 station.up = a[1] == "UP"
493 if (station.up) start(station, now)
494 emit("STATION", a[0], a[1])
495 }
496 "COMP", "TRANSFER" -> {
497 val line = line(a[0])
498 modifiable(line)
499 if (command == "COMP") {
500 if (line.reason != null) fail("#" + line.id + " is already comped")
501 val amount = amount(line)
502 emit("COMP", "#" + line.id, line.item, money(amount), a[1])
503 comps++
504 compAmount += amount
505 line.reason = a[1]
506 } else {
507 val seat = a[1].toInt()
508 if (seat > line.check.size) {
509 fail("seat $seat out of range for table " + line.check.table.number)
510 }
511 if (seat == line.seat) fail("#" + line.id + " is already on seat $seat")
512 emit("TRANSFER", "#" + line.id, "seat", line.seat, "->", seat)
513 line.seat = seat
514 }
515 }
516 "DISCOUNT" -> {
517 val check = check(a[0])
518 unlocked(check)
519 check.discount = a[1].toInt()
520 emit("DISCOUNT", "table", check.table.number, check.discount.toString() + "%")
521 }
522 "BILL" -> bill(check(a[0]))
523 "SPLIT" -> {
524 val check = check(a[0])
525 val amount = amounts(check)
526 if (amount.due == 0L) fail("nothing to split for table " + a[0])
527 if (a[1] == "EVEN") {
528 val ways = a[2].toInt()
529 if (ways > check.size) fail("cannot split table " + a[0] + " $ways ways")
530 emit("SPLIT", "table", check.table.number, "EVEN", ways,
531 "due", money(amount.due))
532 val quotient = amount.due / ways
533 val remainder = amount.due % ways
534 for (i in 1..ways) {
535 emit("PAYER", i, money(quotient + if (i <= remainder) 1L else 0L))
536 }
537 } else {
538 if (check.paid > 0L) fail("table " + a[0] + " has payments")
539 val seatValues = seatTotals(check)
540 val discounts = allocate(seatValues, check.discount)
541 val after = seatValues.mapValues { (seat, amount) -> amount - discounts.getValue(seat) }
542 val services = allocate(after, if (check.size >= 6) 18 else 0)
543 val taxes = allocate(after, 8)
544 emit("SPLIT", "table", check.table.number, "BYSEAT", "total", money(amount.total))
545 for (seat in seatValues.keys.sorted()) {
546 emit("SEAT", seat, "subtotal", money(seatValues.getValue(seat)),
547 "discount", money(discounts.getValue(seat)),
548 "service", money(services.getValue(seat)),
549 "tax", money(taxes.getValue(seat)),
550 "total", money(after.getValue(seat) + services.getValue(seat) + taxes.getValue(seat)))
551 }
552 }
553 }
554 "PAY" -> {
555 val check = check(a[0])
556 if (a[1] !in setOf("CASH", "CARD")) fail("unknown method " + a[1])
557 unserved(check)
558 val due = amounts(check).due
559 if (due == 0L) fail("nothing to pay for table " + a[0])
560 val value = cents(a[2])
561 val applied = min(value, due)
562 check.paid += applied
563 payments[a[1]] = payments.getValue(a[1]) + applied
564 if (value < due) {
565 emit("PAY", "table", check.table.number, a[1], money(value),
566 "remaining", money(due - value))
567 } else {
568 emit("PAY", "table", check.table.number, a[1], money(value),
569 if (a[1] == "CASH") "change" else "tip", money(value - due))
570 if (a[1] == "CARD") {
571 tips += value - due
572 check.server.tips += value - due
573 }
574 close(check)
575 }
576 }
577 "CLOSE" -> {
578 val check = check(a[0])
579 unserved(check)
580 val due = amounts(check).due
581 if (due != 0L) fail("table " + a[0] + " still owes " + money(due))
582 close(check)
583 }
584 "STATUS" -> {
585 val check = check(a[0])
586 emit("STATUS", "table", check.table.number, check.party,
587 "guests", check.size, "since", clock(check.seated))
588 for (line in check.lines) {
589 val parts = mutableListOf<Any>("#" + line.id, "seat", line.seat, line.item,
590 "x" + line.quantity, line.course, line.status)
591 if (line.status == "COOKING") parts.addAll(listOf("until", clock(line.ready)))
592 if (line.reason != null) parts.addAll(listOf("COMP", line.reason!!))
593 emit(*parts.toTypedArray())
594 }
595 }
596 "KITCHEN" -> {
597 emit("KITCHEN", clock(now))
598 for (name in stations) {
599 val station = station(name)
600 val cooking = station.cooking.sortedWith(compareBy<OrderLine>({ it.ready }, { it.id }))
601 .map { "#" + it.id + "@" + clock(it.ready) }
602 val queue = station.queue.map { "#" + it.id }
603 emit(name, if (station.up) "up" else "down", "cooking",
604 if (cooking.isEmpty()) "-" else cooking.joinToString(" "),
605 "queue", if (queue.isEmpty()) "-" else queue.joinToString(" "))
606 }
607 }
608 "FLOOR" -> {
609 val ordered = tables.values.sortedBy { it.number }
610 val free = ordered.count { it.check == null }
611 emit("FLOOR", clock(now), "free", free, "seated", ordered.size - free,
612 "waiting", waiting.size)
613 for (table in ordered) {
614 val check = table.check
615 if (check == null) {
616 emit("table", table.number, "seats", table.seats, "free")
617 } else {
618 val parts = mutableListOf<Any>("table", table.number, "seats", table.seats,
619 check.party, "guests", check.size, "server",
620 check.server.name, "since", clock(check.seated),
621 "items", active(check).size, "total",
622 money(amounts(check).total))
623 if (check.paid > 0L) parts.addAll(listOf("paid", money(check.paid)))
624 emit(*parts.toTypedArray())
625 }
626 }
627 for ((i, entry) in waiting.withIndex()) {
628 emit("WAITING", i + 1, entry.party, entry.size, "since", clock(entry.joined))
629 }
630 }
631 "REPORT" -> report()
632 }
633 }

635 fun execute(line: String) {
636 val fields = line.trim().split(" ")
637 try {
638 run(fields[0], fields.drop(1))
639 } catch (error: CommandError) {
640 emit("ERROR", error.message ?: "")
641 }
642 }
