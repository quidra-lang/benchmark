1use std::collections::{BTreeMap, VecDeque};
2use std::io::{self, Read};
3
4const STATIONS: [&str; 4] = ["GRILL", "FRY", "COLD", "BAR"];
5const COURSES: [&str; 4] = ["DRINK", "STARTER", "MAIN", "DESSERT"];
6
7fn number(text: &str) -> i64 { text.parse().unwrap() }
8
9fn cents(text: &str) -> i64 {
10 let (whole, fraction) = text.split_once('.').unwrap();
11 number(whole) * 100 + number(fraction)
12}
13
14fn money(value: i64) -> String { format!("{}.{:02}", value / 100, value % 100) }
15
16fn clock(time: i64) -> String { format!("{:02}:{:02}", (660 + time) / 60, (660 + time) % 60) }
17
18fn rounded(amount: i64, rate: i64) -> i64 { (amount * rate + 50) / 100 }
19
20fn station_index(name: &str) -> Result<usize, String> {
21 STATIONS.iter().position(|station| *station == name).ok_or_else(|| format!("unknown station {name}"))
22}
23
24fn course_index(name: &str) -> Result<usize, String> {
25 COURSES.iter().position(|course| *course == name).ok_or_else(|| format!("unknown course {name}"))
26}
27
28fn numbers(lines: &[usize]) -> String {
29 lines.iter().map(|index| format!("#{}", index + 1)).collect::<Vec<_>>().join(" ")
30}
31
32fn allocate(values: &BTreeMap<i64, i64>, rate: i64) -> BTreeMap<i64, i64> {
33 let mut shares: BTreeMap<i64, i64> = values.iter().map(|(seat, value)| (*seat, value * rate / 100)).collect();
34 let remaining = rounded(values.values().sum(), rate) - shares.values().sum::<i64>();
35 let mut seats: Vec<i64> = values.keys().copied().collect();
36 seats.sort_by_key(|seat| (-(values[seat] * rate % 100), *seat));
37 for seat in seats.into_iter().take(remaining as usize) { *shares.get_mut(&seat).unwrap() += 1; }
38 shares
39}

41#[derive(Clone, Copy, PartialEq)]
42enum State { Ordered, Queued, Cooking, Ready, Served, Void }
43
44impl State {
45 fn name(self) -> &'static str {
46 match self {
47 Self::Ordered => "ORDERED", Self::Queued => "QUEUED", Self::Cooking => "COOKING",
48 Self::Ready => "READY", Self::Served => "SERVED", Self::Void => "VOID",
49 }
50 }
51}

127#[derive(Default)]
128struct Restaurant {
129 time: i64,
130 tables: BTreeMap<i64, Table>,
131 servers: Vec<Server>,
132 waiting: Vec<WaitingParty>,
133 checks: Vec<Check>,
134 menu: BTreeMap<String, MenuItem>,
135 orders: Vec<Order>,
136 stations: [Station; 4],
137 happy: Option<(i64, i64)>,
138 statistics: Statistics,
139}

141impl Restaurant {
142 fn table(&self, number: i64) -> Result<&Table, String> {
143 self.tables.get(&number).ok_or_else(|| format!("unknown table {number}"))
144 }
145
146 fn check_index(&self, table: i64) -> Result<usize, String> {
147 self.table(table)?.check.ok_or_else(|| format!("table {table} is free"))
148 }
149
150 fn order_index(&self, text: &str) -> Result<usize, String> {
151 let id = number(text) as usize;
152 if id < 1 || id > self.orders.len() { Err(format!("unknown line {text}")) } else { Ok(id - 1) }
153 }
154
155 fn unlocked(&self, index: usize) -> Result<(), String> {
156 let check = &self.checks[index];
157 if check.paid > 0 { Err(format!("check for table {} is locked", check.table)) } else { Ok(()) }
158 }
159
160 fn modifiable(&self, index: usize) -> Result<(), String> {
161 let order = &self.orders[index];
162 if order.state == State::Void { return Err(format!("#{} is void", index + 1)); }
163 if self.checks[order.check].closed { return Err(format!("#{} is on a closed check", index + 1)); }
164 self.unlocked(order.check)
165 }
166
167 fn all_served(&self, index: usize) -> Result<(), String> {
168 let check = &self.checks[index];
169 let count = check.lines.iter().filter(|line| !matches!(self.orders[**line].state, State::Served | State::Void)).count();
170 if count == 0 { Ok(()) } else { Err(format!("table {} has {count} unserved items", check.table)) }
171 }
172
173 fn load(&self, server: usize) -> usize {
174 self.checks.iter().filter(|check| !check.closed && check.server == server).count()
175 }
176
177 fn choose_table(&self, size: i64) -> Option<i64> {
178 self.tables.iter().filter(|(_, table)| table.check.is_none() && table.seats >= size)
179 .min_by_key(|(number, table)| (table.seats, **number)).map(|(number, _)| *number)
180 }
181
182 fn seat(&mut self, party: &str, size: i64, table: i64) -> usize {
183 let server = (0..self.servers.len()).min_by_key(|index| (self.load(*index), *index)).unwrap();
184 let index = self.checks.len();
185 self.checks.push(Check { party: party.into(), size, table, server, seated: self.time,
186 lines: Vec::new(), paid: 0, discount: 0, closed: false });
187 self.tables.get_mut(&table).unwrap().check = Some(index);
188 self.servers[server].parties += 1;
189 self.statistics.covers += size;
190 self.statistics.parties += 1;
191 server
192 }
193
194 fn seat_waiting(&mut self) {
195 let mut index = 0;
196 while index < self.waiting.len() {
197 if let Some(table) = self.choose_table(self.waiting[index].size) {
198 let party = self.waiting.remove(index);
199 let server = self.seat(&party.name, party.size, table);
200 println!("SEATED {} {} table {table} server {} waited {}", party.name, party.size,
201 self.servers[server].name, self.time - party.joined);
202 } else { index += 1; }
203 }
204 }
205
206 fn amounts(&self, index: usize) -> Amounts {
207 let check = &self.checks[index];
208 let subtotal = check.lines.iter().map(|line| &self.orders[*line])
209 .filter(|order| order.state != State::Void).map(Order::amount).sum();
210 let discount = rounded(subtotal, check.discount);
211 let net = subtotal - discount;
212 let service = if check.size >= 6 { rounded(net, 18) } else { 0 };
213 let tax = rounded(net, 8);
214 Amounts { subtotal, discount, service, tax, total: net + service + tax }
215 }
216
217 fn seats(&self, index: usize) -> BTreeMap<i64, i64> {
218 let mut seats = BTreeMap::new();
219 for line in &self.checks[index].lines {
220 let order = &self.orders[*line];
221 if order.state != State::Void { *seats.entry(order.seat).or_default() += order.amount(); }
222 }
223 seats
224 }
225
226 fn close(&mut self, index: usize) {
227 let amounts = self.amounts(index);
228 let check = &mut self.checks[index];
229 let stay = self.time - check.seated;
230 self.statistics.closed += 1;
231 self.statistics.stay += stay;
232 for line in &check.lines {
233 let order = &self.orders[*line];
234 if order.state != State::Void { self.statistics.sales[order.course] += order.amount(); }
235 }
236 let totals = &mut self.statistics.totals;
237 totals.subtotal += amounts.subtotal;
238 totals.discount += amounts.discount;
239 totals.service += amounts.service;
240 totals.tax += amounts.tax;
241 totals.total += amounts.total;
242 self.servers[check.server].sales += amounts.total;
243 check.closed = true;
244 self.tables.get_mut(&check.table).unwrap().check = None;
245 println!("FREE table {} {} stayed {stay}", check.table, check.party);
246 self.seat_waiting();
247 }
248
249 fn start_cooking(&mut self, index: usize, time: i64) {
250 let station = &mut self.stations[index];
251 let capacity = if index == 3 { 2 } else { 1 };
252 while !station.down && !station.queue.is_empty() && station.cooking.len() < capacity {
253 let line = station.queue.pop_front().unwrap();
254 let order = &mut self.orders[line];
255 order.state = State::Cooking;
256 order.started = true;
257 order.ready_time = Some(time + order.prep);
258 station.cooking.push(line);
259 }
260 }
261
262 fn tick(&mut self, minutes: i64) {
263 let target = self.time + minutes;
264 println!("TICK {}", clock(target));
265 loop {
266 let next = self.stations.iter().flat_map(|station| &station.cooking)
267 .map(|line| self.orders[*line].ready_time.unwrap()).min();
268 let Some(time) = next.filter(|time| *time <= target) else { break; };
269 for index in 0..4 {
270 let orders = &self.orders;
271 let station = &mut self.stations[index];
272 let mut finished: Vec<usize> = station.cooking.iter().copied()
273 .filter(|line| orders[*line].ready_time == Some(time)).collect();
274 finished.sort_unstable();
275 station.cooking.retain(|line| orders[*line].ready_time != Some(time));
276 for line in finished {
277 let order = &mut self.orders[line];
278 order.state = State::Ready;
279 let ticket = time - order.fire_time.unwrap();
280 station.tickets += 1;
281 station.ticket_sum += ticket;
282 station.ticket_max = station.ticket_max.max(ticket);
283 println!("READY {} {} #{} {} x{} table {} seat {}", clock(time), STATIONS[index],
284 line + 1, order.item, order.quantity, self.checks[order.check].table, order.seat);
285 }
286 }
287 for index in 0..4 { self.start_cooking(index, time); }
288 }
289 self.time = target;
290 }

292 fn bill(&self, index: usize) {
293 let check = &self.checks[index];
294 println!("BILL table {} {} guests {}", check.table, check.party, check.size);
295 for (seat, subtotal) in self.seats(index) {
296 println!("SEAT {seat} {}", money(subtotal));
297 for line in &check.lines {
298 let order = &self.orders[*line];
299 if order.state != State::Void && order.seat == seat {
300 println!("#{} {} x{} {}{}", line + 1, order.item, order.quantity, money(order.amount()), order.comp_text());
301 }
302 }
303 }
304 let amounts = self.amounts(index);
305 println!("SUBTOTAL {}", money(amounts.subtotal));
306 if check.discount > 0 { println!("DISCOUNT {}% {}", check.discount, money(amounts.discount)); }
307 if check.size >= 6 { println!("SERVICE {}", money(amounts.service)); }
308 println!("TAX {}", money(amounts.tax));
309 println!("TOTAL {}", money(amounts.total));
310 if check.paid > 0 {
311 println!("PAID {}", money(check.paid));
312 println!("DUE {}", money(amounts.total - check.paid));
313 }
314 }

316 fn split_by_seat(&self, index: usize) {
317 let check = &self.checks[index];
318 let subtotals = self.seats(index);
319 let discounts = allocate(&subtotals, check.discount);
320 let net: BTreeMap<i64, i64> = subtotals.iter().map(|(seat, subtotal)| (*seat, subtotal - discounts[seat])).collect();
321 let service = allocate(&net, if check.size >= 6 { 18 } else { 0 });
322 let taxes = allocate(&net, 8);
323 println!("SPLIT table {} BYSEAT total {}", check.table, money(self.amounts(index).total));
324 for (seat, subtotal) in subtotals {
325 println!("SEAT {seat} subtotal {} discount {} service {} tax {} total {}", money(subtotal),
326 money(discounts[&seat]), money(service[&seat]), money(taxes[&seat]),
327 money(net[&seat] + service[&seat] + taxes[&seat]));
328 }
329 }

331 fn status(&self, index: usize) {
332 let check = &self.checks[index];
333 println!("STATUS table {} {} guests {} since {}", check.table, check.party, check.size, clock(check.seated));
334 for line in &check.lines {
335 let order = &self.orders[*line];
336 println!("#{} seat {} {} x{} {} {}{}{}", line + 1, order.seat, order.item, order.quantity,
337 COURSES[order.course], order.state.name(),
338 if order.state == State::Cooking { format!(" until {}", clock(order.ready_time.unwrap())) }
339 else { String::new() }, order.comp_text());
340 }

343 fn kitchen(&self) {
344 println!("KITCHEN {}", clock(self.time));
345 for (name, station) in STATIONS.iter().zip(&self.stations) {
346 let mut cooking = station.cooking.clone();
347 cooking.sort_by_key(|line| (self.orders[*line].ready_time.unwrap(), *line));
348 let cooking = if cooking.is_empty() { "-".into() } else {
349 cooking.iter().map(|line| format!("#{}@{}", line + 1, clock(self.orders[*line].ready_time.unwrap())))
350 .collect::<Vec<_>>().join(" ")
351 };
352 let queue = if station.queue.is_empty() { "-".into() } else {
353 station.queue.iter().map(|line| format!("#{}", line + 1)).collect::<Vec<_>>().join(" ")
354 };
355 println!("{name} {} cooking {cooking} queue {queue}", if station.down { "down" } else { "up" });
356 }
357 }

359 fn floor(&self) {
360 let free = self.tables.values().filter(|table| table.check.is_none()).count();
361 println!("FLOOR {} free {free} seated {} waiting {}", clock(self.time), self.tables.len() - free, self.waiting.len());
362 for (number, table) in &self.tables {
363 if let Some(index) = table.check {
364 let check = &self.checks[index];
365 let items = check.lines.iter().filter(|line| self.orders[**line].state != State::Void).count();
366 println!("table {number} seats {} {} guests {} server {} since {} items {items} total {}{}", table.seats,
367 check.party, check.size, self.servers[check.server].name, clock(check.seated), money(self.amounts(index).total),
368 if check.paid > 0 { format!(" paid {}", money(check.paid)) } else { String::new() });
369 } else { println!("table {number} seats {} free", table.seats); }
370 }
371 for (index, party) in self.waiting.iter().enumerate() {
372 println!("WAITING {} {} {} since {}", index + 1, party.name, party.size, clock(party.joined));
373 }
374 }

376 fn report(&self) {
377 let stats = &self.statistics;
378 println!("REPORT {}", clock(self.time));
379 println!("COVERS {} parties {}", stats.covers, stats.parties);
380 println!("TURNS {} avg_stay {}", stats.closed,
381 if stats.closed == 0 { "-".into() } else { (stats.stay / stats.closed).to_string() });
382 println!("SALES DRINK {} STARTER {} MAIN {} DESSERT {}", money(stats.sales[0]), money(stats.sales[1]),
383 money(stats.sales[2]), money(stats.sales[3]));
384 let amounts = stats.totals;
385 println!("TOTALS subtotal {} discount {} service {} tax {} total {}", money(amounts.subtotal),
386 money(amounts.discount), money(amounts.service), money(amounts.tax), money(amounts.total));
387 println!("PAYMENTS cash {} card {} tips {}", money(stats.payments[0]), money(stats.payments[1]), money(stats.tips));
388 println!("COMPS {} {} VOIDS {}", stats.comps, money(stats.comp_amount), stats.voids);
389 let mut served: Vec<(&String, &i64)> = stats.served.iter().filter(|(_, count)| **count > 0).collect();
390 served.sort_by(|(left_name, left_count), (right_name, right_count)| right_count.cmp(left_count).then(left_name.cmp(right_name)));
391 for (rank, (item, count)) in served.into_iter().take(3).enumerate() { println!("TOP {} {item} {count}", rank + 1); }
392 for (name, station) in STATIONS.iter().zip(&self.stations) {
393 println!("{name} tickets {} avg {} max {} remakes {}", station.tickets,
394 if station.tickets == 0 { "-".into() } else { (station.ticket_sum / station.tickets).to_string() },
395 if station.tickets == 0 { "-".into() } else { station.ticket_max.to_string() }, station.remakes);
396 }
397 for (index, server) in self.servers.iter().enumerate() {
398 println!("SERVER {} tables {} parties {} sales {} tips {}", server.name, self.load(index),
399 server.parties, money(server.sales), money(server.tips));
400 }

403 fn command(&mut self, tokens: &[&str]) -> Result<(), String> {
404 let a = &tokens[1..];
405 match tokens[0] {
406 "TABLE" => {
407 let table = number(a[0]);
408 if self.tables.contains_key(&table) { return Err(format!("table {table} already exists")); }
409 let seats = number(a[1]);
410 self.tables.insert(table, Table { seats, check: None });
411 println!("TABLE {table} seats {seats}");
412 self.seat_waiting();
413 }
414 "STAFF" => {
415 if self.servers.iter().any(|server| server.name == a[0]) { return Err(format!("{} is already on duty", a[0])); }
416 self.servers.push(Server { name: a[0].into(), parties: 0, sales: 0, tips: 0 });
417 println!("STAFF {}", a[0]);
418 }
419 "SEAT" => {
420 if let Some(check) = self.checks.iter().find(|check| !check.closed && check.party == a[0]) {
421 return Err(format!("{} is already seated at table {}", a[0], check.table));
422 }
423 if self.waiting.iter().any(|party| party.name == a[0]) { return Err(format!("{} is already waiting", a[0])); }
424 let size = number(a[1]);
425 if !self.tables.values().any(|table| table.seats >= size) { return Err(format!("no table for party of {size}")); }
426 if self.servers.is_empty() { return Err("no server on duty".into()); }
427 if let Some(table) = self.choose_table(size) {
428 let server = self.seat(a[0], size, table);
429 println!("SEAT {} {size} table {table} server {}", a[0], self.servers[server].name);
430 } else {
431 self.waiting.push(WaitingParty { name: a[0].into(), size, joined: self.time });
432 println!("WAIT {} {size} position {}", a[0], self.waiting.len());
433 }
434 }
435 "CANCEL" => {
436 let index = self.waiting.iter().position(|party| party.name == a[0])
437 .ok_or_else(|| format!("{} is not waiting", a[0]))?;
438 let party = self.waiting.remove(index);
439 println!("CANCEL {} {} waited {}", party.name, party.size, self.time - party.joined);
440 }
441 "MOVE" => {
442 let (from, to) = (number(a[0]), number(a[1]));
443 self.table(from)?;
444 self.table(to)?;
445 let index = self.check_index(from)?;
446 if self.tables[&to].check.is_some() { return Err(format!("table {to} is occupied")); }
447 if self.tables[&to].seats < self.checks[index].size {
448 return Err(format!("table {to} is too small for {}", self.checks[index].party));
449 }
450 self.tables.get_mut(&from).unwrap().check = None;
451 self.tables.get_mut(&to).unwrap().check = Some(index);
452 self.checks[index].table = to;
453 println!("MOVE {} table {from} -> {to}", self.checks[index].party);
454 self.seat_waiting();
455 }
456 "MENU" => {
457 let station = station_index(a[1])?;
458 let course = course_index(a[4])?;
459 let action = if self.menu.contains_key(a[0]) { "UPDATE" } else { "ADD" };
460 let stock = self.menu.get(a[0]).and_then(|item| item.stock);
461 let (prep, price) = (number(a[2]), cents(a[3]));
462 self.menu.insert(a[0].into(), MenuItem { station, course, prep, price, stock });
463 println!("MENU {action} {} {} {prep} {} {}", a[0], a[1], money(price), a[4]);
464 }
465 "STOCK" => {
466 let item = self.menu.get_mut(a[0]).ok_or_else(|| format!("unknown item {}", a[0]))?;
467 let stock = number(a[1]);
468 item.stock = Some(stock);
469 println!("STOCK {} {stock}", a[0]);
470 }
471 "HAPPY" => {
472 let (rate, end) = (number(a[0]), self.time + number(a[1]));
473 self.happy = Some((rate, end));
474 println!("HAPPY {rate}% until {}", clock(end));
475 }
476 "ITEMS" => {
477 let mut items: Vec<(&String, &MenuItem)> = self.menu.iter().collect();
478 items.sort_by(|(left_name, left), (right_name, right)| {
479 left.course.cmp(&right.course).then(left_name.cmp(right_name))
480 });
481 println!("ITEMS {}", items.len());
482 for (name, item) in items {
483 println!("{} {name} {} {} {} {}", COURSES[item.course], STATIONS[item.station], item.prep,
484 money(item.price), item.stock.map(|stock| stock.to_string()).unwrap_or_else(|| "-".into()));
485 }
486 }
487 "ORDER" => {
488 let table = number(a[0]);
489 let index = self.check_index(table)?;
490 self.unlocked(index)?;
491 let seat = number(a[1]);
492 if seat > self.checks[index].size { return Err(format!("seat {seat} out of range for table {table}")); }
493 let item = self.menu.get(a[2]).ok_or_else(|| format!("unknown item {}", a[2]))?;
494 let quantity = number(a[3]);
495 if let Some(stock) = item.stock {
496 if stock < quantity { return Err(format!("only {stock} {} left", a[2])); }
497 }
498 let happy = item.course == 0 && self.happy.is_some_and(|(_, end)| self.time < end);
499 let unit_price = if happy { item.price - rounded(item.price, self.happy.unwrap().0) } else { item.price };
500 let line = self.orders.len();
501 self.orders.push(Order {
502 check: index, seat, item: a[2].into(), quantity, station: item.station,
503 prep: item.prep, course: item.course, unit_price, state: State::Ordered,
504 comp: None, fire_time: None, ready_time: None, started: false,
505 });
506 self.checks[index].lines.push(line);
507 println!("ORDER #{} table {table} seat {seat} {} x{quantity} {}{}", line + 1, a[2],
508 money(unit_price * quantity), if happy { " HAPPY" } else { "" });
509 if let Some(stock) = &mut self.menu.get_mut(a[2]).unwrap().stock {
510 *stock -= quantity;
511 if *stock == 0 { println!("SOLDOUT {}", a[2]); }
512 }
513 }
514 "FIRE" => {
515 let table = number(a[0]);
516 let index = self.check_index(table)?;
517 let course = course_index(a[1])?;
518 let lines: Vec<usize> = self.checks[index].lines.iter().copied()
519 .filter(|line| self.orders[*line].course == course && self.orders[*line].state == State::Ordered).collect();
520 if lines.is_empty() { return Err(format!("nothing to fire for {} at table {table}", a[1])); }
521 for earlier in 1..course {
522 if self.checks[index].lines.iter().any(|line| self.orders[*line].course == earlier && self.orders[*line].state == State::Ordered) {
523 return Err(format!("fire {} first", COURSES[earlier]));
524 }
525 }
526 for line in &lines {
527 let order = &mut self.orders[*line];
528 order.state = State::Queued;
529 order.fire_time = Some(self.time);
530 self.stations[order.station].queue.push_back(*line);
531 }
532 println!("FIRE table {table} {} {}", a[1], numbers(&lines));
533 for station in 0..4 { self.start_cooking(station, self.time); }
534 }
535 "TICK" => self.tick(number(a[0])),
536 "SERVE" => {
537 let table = number(a[0]);
538 let index = self.check_index(table)?;
539 let lines: Vec<usize> = self.checks[index].lines.iter().copied()
540 .filter(|line| self.orders[*line].state == State::Ready).collect();
541 if lines.is_empty() { return Err(format!("nothing ready for table {table}")); }
542 for line in &lines {
543 let order = &mut self.orders[*line];
544 order.state = State::Served;
545 *self.statistics.served.entry(order.item.clone()).or_default() += order.quantity;
546 }
547 println!("SERVE table {table} {}", numbers(&lines));
548 }
549 "RUSH" => {
550 let index = self.order_index(a[0])?;
551 let order = &self.orders[index];
552 if order.state != State::Queued { return Err(format!("#{} is not queued", a[0])); }
553 let station = &mut self.stations[order.station];
554 station.queue.retain(|line| *line != index);
555 station.queue.push_front(index);
556 println!("RUSH #{} {}", a[0], STATIONS[order.station]);
557 }
558 "REMAKE" => {
559 let index = self.order_index(a[0])?;
560 let order = &mut self.orders[index];
561 if self.checks[order.check].closed { return Err(format!("#{} is on a closed check", a[0])); }
562 if !matches!(order.state, State::Ready | State::Served) { return Err(format!("#{} is not ready or served", a[0])); }
563 order.state = State::Queued;
564 order.fire_time = Some(self.time);
565 let station = order.station;
566 self.stations[station].queue.push_front(index);
567 self.stations[station].remakes += 1;
568 println!("REMAKE #{} {} {}", a[0], order.item, STATIONS[station]);
569 self.start_cooking(station, self.time);
570 }
571 "VOID" => {
572 let index = self.order_index(a[0])?;
573 let order = &mut self.orders[index];
574 if order.state == State::Void { return Err(format!("#{} is already void", a[0])); }
575 if order.started { return Err(format!("#{} has already started", a[0])); }
576 if order.state == State::Queued { self.stations[order.station].queue.retain(|line| *line != index); }
577 order.state = State::Void;
578 if let Some(stock) = &mut self.menu.get_mut(&order.item).unwrap().stock { *stock += order.quantity; }
579 self.statistics.voids += 1;
580 println!("VOID #{} {} x{}", a[0], order.item, order.quantity);
581 }
582 "STATION" => {
583 let index = station_index(a[0])?;
584 let down = a[1] == "DOWN";
585 if down && self.stations[index].down { return Err(format!("station {} is already down", a[0])); }
586 if !down && !self.stations[index].down { return Err(format!("station {} is not down", a[0])); }
587 self.stations[index].down = down;
588 if !down { self.start_cooking(index, self.time); }
589 println!("STATION {} {}", a[0], a[1]);
590 }
591 "COMP" => {
592 let index = self.order_index(a[0])?;
593 self.modifiable(index)?;
594 let order = &mut self.orders[index];
595 if order.comp.is_some() { return Err(format!("#{} is already comped", a[0])); }
596 let amount = order.amount();
597 println!("COMP #{} {} {} {}", a[0], order.item, money(amount), a[1]);
598 self.statistics.comps += 1;
599 self.statistics.comp_amount += amount;
600 order.comp = Some(a[1].into());
601 }
602 "TRANSFER" => {
603 let index = self.order_index(a[0])?;
604 self.modifiable(index)?;
605 let order = &mut self.orders[index];
606 let check = &self.checks[order.check];
607 let seat = number(a[1]);
608 if seat > check.size { return Err(format!("seat {seat} out of range for table {}", check.table)); }
609 if seat == order.seat { return Err(format!("#{} is already on seat {seat}", a[0])); }
610 println!("TRANSFER #{} seat {} -> {seat}", a[0], order.seat);
611 order.seat = seat;
612 }
613 "DISCOUNT" => {
614 let table = number(a[0]);
615 let index = self.check_index(table)?;
616 self.unlocked(index)?;
617 let rate = number(a[1]);
618 self.checks[index].discount = rate;
619 println!("DISCOUNT table {table} {rate}%");
620 }
621 "BILL" => self.bill(self.check_index(number(a[0]))?),
622 "SPLIT" => {
623 let table = number(a[0]);
624 let index = self.check_index(table)?;
625 let check = &self.checks[index];
626 let due = self.amounts(index).total - check.paid;
627 if due == 0 { return Err(format!("nothing to split for table {table}")); }
628 if a[1] == "EVEN" {
629 let count = number(a[2]);
630 if count > check.size { return Err(format!("cannot split table {table} {count} ways")); }
631 println!("SPLIT table {table} EVEN {count} due {}", money(due));
632 for payer in 1..=count {
633 println!("PAYER {payer} {}", money(due / count + if payer <= due % count { 1 } else { 0 }));
634 }
635 } else {
636 if check.paid > 0 { return Err(format!("table {table} has payments")); }
637 self.split_by_seat(index);
638 }
639 }
640 "PAY" => {
641 let table = number(a[0]);
642 let index = self.check_index(table)?;
643 let method = match a[1] {
644 "CASH" => 0, "CARD" => 1, _ => return Err(format!("unknown method {}", a[1])),
645 };
646 self.all_served(index)?;
647 let due = self.amounts(index).total - self.checks[index].paid;
648 if due == 0 { return Err(format!("nothing to pay for table {table}")); }
649 let amount = cents(a[2]);
650 let applied = amount.min(due);
651 self.checks[index].paid += applied;
652 self.statistics.payments[method] += applied;
653 if amount < due {
654 println!("PAY table {table} {} {} remaining {}", a[1], money(amount), money(due - amount));
655 } else {
656 if method == 0 {
657 println!("PAY table {table} CASH {} change {}", money(amount), money(amount - due));
658 } else {
659 let tip = amount - due;
660 self.statistics.tips += tip;
661 self.servers[self.checks[index].server].tips += tip;
662 println!("PAY table {table} CARD {} tip {}", money(amount), money(tip));
663 }
664 self.close(index);
665 }
666 }
667 "CLOSE" => {
668 let table = number(a[0]);
669 let index = self.check_index(table)?;
670 self.all_served(index)?;
671 let due = self.amounts(index).total - self.checks[index].paid;
672 if due != 0 { return Err(format!("table {table} still owes {}", money(due))); }
673 self.close(index);
674 }
675 "STATUS" => self.status(self.check_index(number(a[0]))?),
676 "KITCHEN" => self.kitchen(),
677 "FLOOR" => self.floor(),
678 "REPORT" => self.report(),
679 _ => unreachable!(),
680 }
681 Ok(())
682 }
683}

685fn main() {
686 let mut input = String::new();
687 io::stdin().read_to_string(&mut input).unwrap();
688 let mut lines = input.lines();
689 let count: usize = lines.next().unwrap().parse().unwrap();
690 let mut restaurant = Restaurant::default();
691 for line in lines.take(count) {
692 let tokens: Vec<&str> = line.split_ascii_whitespace().collect();
693 if let Err(error) = restaurant.command(&tokens) { println!("ERROR {error}"); }
694 }
