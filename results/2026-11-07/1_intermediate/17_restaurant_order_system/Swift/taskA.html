11func cents(_ text: String) -> Int {
12 let parts = text.split(separator: ".", omittingEmptySubsequences: false)
13 return Int(parts[0])! * 100 + Int(parts[1])!
14}

16func money(_ cents: Int) -> String {
17 return String(format: "%d.%02d", cents / 100, cents % 100)
18}

20func clock(_ elapsed: Int) -> String {
21 let minutes = elapsed + 660
22 return String(format: "%02d:%02d", minutes / 60, minutes % 60)
23}

25func rounded(_ value: Int, _ rate: Int) -> Int {
26 return (value * rate + 50) / 100
27}

87
88 var active: [OrderLine] { lines.filter { $0.state != "VOID" } }
89
90 func amounts() -> [Int] {
91 let subtotal = active.reduce(0) { $0 + $1.amount }
92 let reduction = rounded(subtotal, discount)
93 let afterDiscount = subtotal - reduction
94 let service = size >= 6 ? rounded(afterDiscount, 18) : 0
95 let tax = rounded(afterDiscount, 8)
96 let total = afterDiscount + service + tax
97 return [subtotal, reduction, service, tax, total, total - paid]
98 }
99}

194 guard let number = Int(text), let selected = tables[number] else {

210 func unlocked(_ check: Check) throws {
211 if check.paid != 0 {
212 throw CommandError("check for table \(check.table.number) is locked")

216 func modification(_ order: OrderLine) throws {
217 if order.state == "VOID" {
218 throw CommandError("\(order.tag) is void")

221 throw CommandError("\(order.tag) is on a closed check")

226 func allServed(_ check: Check) throws {
227 let count = check.active.filter { $0.state != "SERVED" }.count

233 func selectTable(_ size: Int) -> DiningTable? {
234 tables.values
235 .filter { $0.check == nil && $0.seats >= size }
236 .min { ($0.seats, $0.number) < ($1.seats, $1.number) }
237 }

239 func seatParty(_ party: String, _ size: Int, _ table: DiningTable) -> Check {
240 let server = servers.min { $0.load < $1.load }!
241 let check = Check(party, size, table, server, time)
242 table.check = check
243 seated[party] = check
244 server.load += 1
245 server.parties += 1
246 guests += size
247 parties += 1
248 return check
249 }
250
251 func processWaitlist() {
252 var remaining: [Waiting] = []
253 for entry in waiting {
254 if let table = selectTable(entry.size) {
255 let check = seatParty(entry.party, entry.size, table)
256 emit("SEATED", entry.party, entry.size, "table", table.number,
257 "server", check.server.name, "waited", time - entry.joined)
258 } else {
259 remaining.append(entry)
260 }
261 }
262 waiting = remaining
263 }
264
265 func closeCheck(_ check: Check) {
266 let amounts = check.amounts()
267 for order in check.active {
268 sales[courseNames.firstIndex(of: order.course)!] += order.amount
269 }
270 for i in totals.indices { totals[i] += amounts[i] }
271 let stayed = time - check.since
272 closed += 1
273 stayTotal += stayed
274 check.server.sales += amounts[4]
275 check.server.load -= 1
276 check.closed = true
277 check.table.check = nil
278 seated.removeValue(forKey: check.party)
279 emit("FREE", "table", check.table.number, check.party, "stayed", stayed)
280 processWaitlist()
281 }

283 func startStation(_ name: String, _ when: Int) {
284 let station = stations[name]!
285 while station.up && !station.queue.isEmpty &&
286 station.cooking.count < station.capacity {
287 let order = station.queue.removeFirst()
288 order.state = "COOKING"
289 order.started = true
290 order.ready = when + order.prep
291 station.cooking.append(order)
292 }
293 }

295 func startAll(_ when: Int) {
296 for name in stationNames { startStation(name, when) }
297 }

299 func seatSubtotals(_ check: Check) -> [Int: Int] {
300 var amounts: [Int: Int] = [:]
301 for order in check.active { amounts[order.seat, default: 0] += order.amount }
302 return amounts
303 }

305 func allocate(_ values: [Int: Int], _ rate: Int) -> [Int: Int] {
306 var amounts: [Int: Int] = [:]
307 var sum = 0
308 var allocated = 0
309 for (seat, value) in values {
310 let amount = value * rate / 100
311 amounts[seat] = amount
312 sum += value
313 allocated += amount
314 }
315 let extra = rounded(sum, rate) - allocated
316 let ranked = values.keys.sorted {
317 let a = values[$0]! * rate % 100
318 let b = values[$1]! * rate % 100
319 return a != b ? a > b : $0 < $1
320 }
321 if extra > 0 {
322 for seat in ranked.prefix(extra) { amounts[seat, default: 0] += 1 }
323 }
324 return amounts
325 }

346 func cmdSeat(_ a: [String]) throws {
347 let party = a[0], size = Int(a[1])!
348 if let existing = seated[party] {
349 throw CommandError("\(party) is already seated at table \(existing.table.number)")
350 }
351 if waiting.contains(where: { $0.party == party }) {
352 throw CommandError("\(party) is already waiting")
353 }
354 if !tables.values.contains(where: { $0.seats >= size }) {
355 throw CommandError("no table for party of \(size)")

357 if servers.isEmpty { throw CommandError("no server on duty") }
358 if let table = selectTable(size) {

360 emit("SEAT", party, size, "table", table.number, "server", check.server.name)
361 } else {
362 waiting.append(Waiting(party: party, size: size, joined: time))
363 emit("WAIT", party, size, "position", waiting.count)
364 }

436 func cmdOrder(_ a: [String]) throws {
437 let check = try table(a[0], occupied: true).check!
438 try unlocked(check)
439 let seat = Int(a[1])!, quantity = Int(a[3])!
440 if seat > check.size {
441 throw CommandError("seat \(seat) out of range for table \(a[0])")
442 }
443 guard let item = menu[a[2]] else {
444 throw CommandError("unknown item \(a[2])")
445 }
446 if let stock = item.stock, stock < quantity {
447 throw CommandError("only \(stock) \(item.name) left")
448 }
449 let happy = item.course == "DRINK" && happyUntil != nil && time < happyUntil!
450 let price = item.price - (happy ? rounded(item.price, happyRate) : 0)
451 let order = OrderLine(lines.count + 1, check, seat, item, quantity, price)
452 lines.append(order)
453 check.lines.append(order)
454 var text = "ORDER \(order.tag) table \(a[0]) seat \(seat) " +
455 "\(item.name) x\(quantity) \(money(order.amount))"
456 if happy { text += " HAPPY" }
457 emit(text)
458 if let stock = item.stock {
459 item.stock = stock - quantity
460 if item.stock == 0 { emit("SOLDOUT", item.name) }
461 }
462 }

464 func cmdFire(_ a: [String]) throws {
465 let check = try table(a[0], occupied: true).check!
466 guard let index = courseNames.firstIndex(of: a[1]) else {
467 throw CommandError("unknown course \(a[1])")
468 }
469 let selected = check.lines.filter {
470 $0.course == a[1] && $0.state == "ORDERED"
471 }
472 if selected.isEmpty {
473 throw CommandError("nothing to fire for \(a[1]) at table \(a[0])")
474 }
475 if index >= 2 {
476 for earlier in 1..<index {
477 if check.lines.contains(where: {
478 $0.course == courseNames[earlier] && $0.state == "ORDERED"
479 }) {
480 throw CommandError("fire \(courseNames[earlier]) first")
481 }
482 }
483 }
484 for order in selected {
485 order.state = "QUEUED"
486 order.fired = time
487 stations[order.station]!.queue.append(order)
488 }
489 emitParts(["FIRE", "table", a[0], a[1]] + selected.map { $0.tag })
490 startAll(time)
491 }

493 func cmdTick(_ a: [String]) {
494 let target = time + Int(a[0])!
495 emit("TICK", clock(target))
496 while true {
497 let cooking = stations.values.flatMap { $0.cooking }
498 guard let completion = cooking.map({ $0.ready }).min(),
499 completion <= target else { break }
500 for name in stationNames {
501 let station = stations[name]!
502 let finished = station.cooking.filter { $0.ready == completion }
503 .sorted { $0.number < $1.number }
504 for order in finished {
505 station.cooking.removeAll { $0 === order }
506 order.state = "READY"
507 let ticket = completion - order.fired
508 station.tickets += 1
509 station.ticketTotal += ticket
510 station.ticketMax = max(station.ticketMax, ticket)
511 emit("READY", clock(completion), name, order.tag, order.name,
512 "x\(order.quantity)", "table", order.check.table.number,
513 "seat", order.seat)
514 }
515 }
516 startAll(completion)
517 }
518 time = target
519 }

619 func cmdBill(_ a: [String]) throws {
620 let check = try table(a[0], occupied: true).check!
621 emit("BILL", "table", a[0], check.party, "guests", check.size)
622 let subtotals = seatSubtotals(check)
623 for seat in subtotals.keys.sorted() {
624 emit("SEAT", seat, money(subtotals[seat]!))
625 for order in check.lines where order.state != "VOID" && order.seat == seat {
626 var text = "\(order.tag) \(order.name) x\(order.quantity) \(money(order.amount))"
627 if let reason = order.comp { text += " COMP \(reason)" }
628 emit(text)
629 }
630 }
631 let amounts = check.amounts()
632 emit("SUBTOTAL", money(amounts[0]))
633 if check.discount != 0 {
634 emit("DISCOUNT", "\(check.discount)%", money(amounts[1]))
635 }
636 if check.size >= 6 { emit("SERVICE", money(amounts[2])) }
637 emit("TAX", money(amounts[3]))
638 emit("TOTAL", money(amounts[4]))
639 if check.paid != 0 {
640 emit("PAID", money(check.paid))
641 emit("DUE", money(amounts[5]))
642 }
643 }

645 func cmdSplit(_ a: [String]) throws {
646 let check = try table(a[0], occupied: true).check!
647 let due = check.amounts()[5]
648 if due == 0 { throw CommandError("nothing to split for table \(a[0])") }
649 if a[1] == "EVEN" {
650 let count = Int(a[2])!
651 if count > check.size {
652 throw CommandError("cannot split table \(a[0]) \(count) ways")
653 }
654 emit("SPLIT", "table", a[0], "EVEN", count, "due", money(due))
655 for payer in 1...count {
656 emit("PAYER", payer, money(due / count + (payer <= due % count ? 1 : 0)))
657 }
658 return
659 }
660 if check.paid != 0 {
661 throw CommandError("table \(a[0]) has payments")
662 }
663 let subtotals = seatSubtotals(check)
664 let discounts = allocate(subtotals, check.discount)
665 var after: [Int: Int] = [:]
666 for (seat, subtotal) in subtotals {
667 after[seat] = subtotal - discounts[seat]!
668 }
669 let services = check.size >= 6 ? allocate(after, 18) :
670 Dictionary(uniqueKeysWithValues: after.keys.map { ($0, 0) })
671 let taxes = allocate(after, 8)
672 emit("SPLIT", "table", a[0], "BYSEAT", "total", money(check.amounts()[4]))
673 for seat in subtotals.keys.sorted() {
674 let total = after[seat]! + services[seat]! + taxes[seat]!
675 emit("SEAT", seat, "subtotal", money(subtotals[seat]!),
676 "discount", money(discounts[seat]!), "service", money(services[seat]!),
677 "tax", money(taxes[seat]!), "total", money(total))
678 }
679 }

681 func cmdPay(_ a: [String]) throws {
682 let check = try table(a[0], occupied: true).check!
683 guard payments[a[1]] != nil else { throw CommandError("unknown method \(a[1])") }
684 try allServed(check)
685 let due = check.amounts()[5]
686 if due == 0 { throw CommandError("nothing to pay for table \(a[0])") }
687 let value = cents(a[2])
688 let applied = min(value, due)
689 check.paid += applied
690 payments[a[1], default: 0] += applied
691 if value < due {
692 emit("PAY", "table", a[0], a[1], a[2], "remaining", money(due - value))
693 } else {
694 let excess = value - due
695 emit("PAY", "table", a[0], a[1], a[2],
696 a[1] == "CASH" ? "change" : "tip", money(excess))
697 if a[1] == "CARD" {
698 tips += excess
699 check.server.tips += excess
700 }
701 closeCheck(check)
702 }
703 }

715 func cmdStatus(_ a: [String]) throws {
716 let check = try table(a[0], occupied: true).check!
717 emit("STATUS", "table", a[0], check.party, "guests", check.size,
718 "since", clock(check.since))
719 for order in check.lines {
720 var text = "\(order.tag) seat \(order.seat) \(order.name) " +
721 "x\(order.quantity) \(order.course) \(order.state)"
722 if order.state == "COOKING" { text += " until \(clock(order.ready))" }
723 if let comp = order.comp { text += " COMP \(comp)" }
724 emit(text)
725 }

728 func cmdKitchen() {
729 emit("KITCHEN", clock(time))
730 for name in stationNames {
731 let station = stations[name]!
732 let cooking = station.cooking.sorted {
733 ($0.ready, $0.number) < ($1.ready, $1.number)
734 }
735 let cookingText = cooking.isEmpty ? "-" :
736 cooking.map { "\($0.tag)@\(clock($0.ready))" }.joined(separator: " ")
737 let queueText = station.queue.isEmpty ? "-" :
738 station.queue.map { $0.tag }.joined(separator: " ")
739 emit(name, station.up ? "up" : "down", "cooking", cookingText,
740 "queue", queueText)
741 }

744 func cmdFloor() {
745 let occupied = tables.values.filter { $0.check != nil }.count
746 emit("FLOOR", clock(time), "free", tables.count - occupied,
747 "seated", occupied, "waiting", waiting.count)
748 for number in tables.keys.sorted() {
749 let table = tables[number]!
750 guard let check = table.check else {
751 emit("table", number, "seats", table.seats, "free")
752 continue
753 }
754 var text = "table \(number) seats \(table.seats) " +
755 "\(check.party) guests \(check.size) server \(check.server.name) " +
756 "since \(clock(check.since)) items \(check.active.count) " +
757 "total \(money(check.amounts()[4]))"
758 if check.paid != 0 { text += " paid \(money(check.paid))" }
759 emit(text)
760 }
761 for (index, entry) in waiting.enumerated() {
762 emit("WAITING", index + 1, entry.party, entry.size,
763 "since", clock(entry.joined))
764 }
765 }

800 func execute(_ input: String) {
801 let parts = input.split(separator: " ").map(String.init)
802 guard let command = parts.first else { return }
803 let args = Array(parts.dropFirst())
804 do {
805 switch command {
806 case "TABLE": try cmdTable(args)
807 case "STAFF": try cmdStaff(args)
808 case "SEAT": try cmdSeat(args)
809 case "CANCEL": try cmdCancel(args)
810 case "MOVE": try cmdMove(args)
811 case "MENU": try cmdMenu(args)
812 case "STOCK": try cmdStock(args)
813 case "HAPPY": cmdHappy(args)
814 case "ITEMS": cmdItems()
815 case "ORDER": try cmdOrder(args)
816 case "FIRE": try cmdFire(args)
817 case "TICK": cmdTick(args)
818 case "SERVE": try cmdServe(args)
819 case "RUSH": try cmdRush(args)
820 case "REMAKE": try cmdRemake(args)
821 case "VOID": try cmdVoid(args)
822 case "STATION": try cmdStation(args)
823 case "COMP": try cmdComp(args)
824 case "TRANSFER": try cmdTransfer(args)
825 case "DISCOUNT": try cmdDiscount(args)
826 case "BILL": try cmdBill(args)
827 case "SPLIT": try cmdSplit(args)
828 case "PAY": try cmdPay(args)
829 case "CLOSE": try cmdClose(args)
830 case "STATUS": try cmdStatus(args)
831 case "KITCHEN": cmdKitchen()
832 case "FLOOR": cmdFloor()
833 case "REPORT": cmdReport()
834 default: break
835 }
836 } catch let error as CommandError {
837 emit("ERROR", error.message)
838 } catch {
839 fatalError("Unexpected error: \(error)")
840 }
841 }
842}
