67 def active(self):
68 return [line for line in self.lines if line.state != "VOID"]

70 def amounts(self):
71 subtotal = sum(line.amount() for line in self.active())
72 discount = rounded(subtotal, self.discount)
73 after_discount = subtotal - discount
74 service = rounded(after_discount, 18) if self.size >= 6 else 0
75 tax = rounded(after_discount, 8)
76 total = after_discount + service + tax
77 return subtotal, discount, service, tax, total, total - self.paid

178 def select_table(self, size):

184 def seat_party(self, party, size, table):
185 server = min(self.servers, key=lambda server: server.load)
186 check = Check(party, size, table, server, self.time)
187 table.check = check
188 self.seated[party] = check
189 server.load += 1
190 server.parties += 1
191 self.guests += size
192 self.parties += 1
193 return check

195 def process_waitlist(self):
196 remaining = []
197 for party, size, joined in self.waiting:
198 table = self.select_table(size)
199 if table is None:
200 remaining.append((party, size, joined))
201 else:
202 check = self.seat_party(party, size, table)
203 self.emit("SEATED", party, size, "table", table.number, "server", check.server.name, "waited", self.time - joined)
204 self.waiting = remaining

206 def close_check(self, check):
207 amounts = check.amounts()
208 for line in check.active():
209 self.sales[line.course] += line.amount()
210 self.totals = [total + value for total, value in zip(self.totals, amounts)]
211 stay = self.time - check.since
212 self.closed += 1
213 self.stay_total += stay
214 check.server.sales += amounts[4]
215 check.server.load -= 1
216 check.closed = True
217 check.table.check = None
218 del self.seated[check.party]
219 self.emit("FREE", "table", check.table.number, check.party, "stayed", stay)
220 self.process_waitlist()

222 def start_station(self, name, when):
223 station = self.stations[name]
224 while station.up and station.queue and len(station.cooking) < station.capacity:
225 line = station.queue.pop(0)
226 line.state = "COOKING"
227 line.started = True
228 line.ready = when + line.prep
229 station.cooking.append(line)

235 def seat_subtotals(self, check):
236 values = {}
237 for line in check.active():
238 values[line.seat] = values.get(line.seat, 0) + line.amount()

241 def allocate(self, values, rate):
242 amounts = {seat: value * rate // 100 for seat, value in values.items()}
243 extra = rounded(sum(values.values()), rate) - sum(amounts.values())
244 ranked = sorted(values, key=lambda seat: (-(values[seat] * rate % 100), seat))
245 for seat in ranked[:extra]:
246 amounts[seat] += 1
247 return amounts

264 def cmd_seat(self, party, size):
265 size = int(size)
266 if party in self.seated:
267 raise CommandError(f"{party} is already seated at table {self.seated[party].table.number}")
268 if any(name == party for name, _, _ in self.waiting):
269 raise CommandError(f"{party} is already waiting")
270 if not any(table.seats >= size for table in self.tables.values()):
271 raise CommandError(f"no table for party of {size}")
272 if not self.servers:
273 raise CommandError("no server on duty")
274 table = self.select_table(size)
275 if table is None:
276 self.waiting.append((party, size, self.time))
277 self.emit("WAIT", party, size, "position", len(self.waiting))
278 else:
279 check = self.seat_party(party, size, table)
280 self.emit("SEAT", party, size, "table", table.number, "server", check.server.name)

336 def cmd_order(self, number, seat, name, quantity):
337 check = self.table(number, occupied=True).check
338 self.unlocked(check)
339 seat, quantity = int(seat), int(quantity)
340 if seat > check.size:
341 raise CommandError(f"seat {seat} out of range for table {number}")
342 if name not in self.menu:
343 raise CommandError(f"unknown item {name}")
344 item = self.menu[name]
345 if item.stock is not None and item.stock < quantity:
346 raise CommandError(f"only {item.stock} {name} left")
347 happy = item.course == "DRINK" and self.happy_until is not None and self.time < self.happy_until
348 price = item.price - rounded(item.price, self.happy_rate) if happy else item.price
349 line = OrderLine(len(self.lines) + 1, check, seat, item, quantity, price)
350 self.lines.append(line)
351 check.lines.append(line)
352 parts = ["ORDER", line.tag(), "table", number, "seat", seat, name, "x" + str(quantity), money(line.amount())]
353 if happy:
354 parts.append("HAPPY")
355 self.emit(*parts)
356 if item.stock is not None:
357 item.stock -= quantity
358 if item.stock == 0:
359 self.emit("SOLDOUT", name)

361 def cmd_fire(self, number, course):
362 check = self.table(number, occupied=True).check
363 if course not in COURSES:
364 raise CommandError(f"unknown course {course}")
365 selected = [line for line in check.lines if line.course == course and line.state == "ORDERED"]
366 if not selected:
367 raise CommandError(f"nothing to fire for {course} at table {number}")
368 if course in ("MAIN", "DESSERT"):
369 for earlier in COURSES[1:COURSES.index(course)]:
370 if any(line.course == earlier and line.state == "ORDERED" for line in check.lines):
371 raise CommandError(f"fire {earlier} first")
372 for line in selected:
373 line.state = "QUEUED"
374 line.fire = self.time
375 self.stations[line.station].queue.append(line)
376 self.emit("FIRE", "table", number, course, *(line.tag() for line in selected))
377 self.start_all(self.time)

379 def cmd_tick(self, minutes):
380 target = self.time + int(minutes)
381 self.emit("TICK", clock(target))
382 while True:
383 cooking = [line for station in self.stations.values() for line in station.cooking]
384 if not cooking:
385 break
386 completion = min(line.ready for line in cooking)
387 if completion > target:
388 break
389 for name in STATIONS:
390 station = self.stations[name]
391 finished = sorted((line for line in station.cooking if line.ready == completion), key=lambda line: line.number)
392 for line in finished:
393 station.cooking.remove(line)
394 line.state = "READY"
395 ticket = completion - line.fire
396 station.tickets += 1
397 station.ticket_total += ticket
398 station.ticket_max = max(station.ticket_max, ticket)
399 self.emit(
400 "READY", clock(completion), name, line.tag(), line.name, "x" + str(line.quantity),
401 "table", line.check.table.number, "seat", line.seat,
402 )
403 self.start_all(completion)
404 self.time = target

519 def cmd_split(self, number, mode, *arguments):
520 check = self.table(number, occupied=True).check
521 due = check.amounts()[5]
522 if due == 0:
523 raise CommandError(f"nothing to split for table {number}")
524 if mode == "EVEN":
525 count = int(arguments[0])
526 if count > check.size:
527 raise CommandError(f"cannot split table {number} {count} ways")
528 self.emit("SPLIT", "table", number, "EVEN", count, "due", money(due))
529 quotient, remainder = divmod(due, count)
530 for payer in range(1, count + 1):
531 self.emit("PAYER", payer, money(quotient + (payer <= remainder)))
532 return
533 if check.paid:
534 raise CommandError(f"table {number} has payments")
535 subtotals = self.seat_subtotals(check)
536 discounts = self.allocate(subtotals, check.discount)
537 after = {seat: subtotal - discounts[seat] for seat, subtotal in subtotals.items()}
538 services = self.allocate(after, 18) if check.size >= 6 else {seat: 0 for seat in subtotals}
539 taxes = self.allocate(after, 8)
540 self.emit("SPLIT", "table", number, "BYSEAT", "total", money(check.amounts()[4]))
541 for seat in sorted(subtotals):
542 total = after[seat] + services[seat] + taxes[seat]
543 self.emit(
544 "SEAT", seat, "subtotal", money(subtotals[seat]), "discount", money(discounts[seat]),
545 "service", money(services[seat]), "tax", money(taxes[seat]), "total", money(total),
546 )

548 def cmd_pay(self, number, method, amount):
549 check = self.table(number, occupied=True).check
550 if method not in self.payments:
551 raise CommandError(f"unknown method {method}")
552 self.all_served(check)
553 due = check.amounts()[5]
554 if due == 0:
555 raise CommandError(f"nothing to pay for table {number}")
556 value = cents(amount)
557 applied = min(value, due)
558 check.paid += applied
559 self.payments[method] += applied
560 if value < due:
561 self.emit("PAY", "table", number, method, amount, "remaining", money(due - value))
562 else:
563 excess = value - due
564 label = "change" if method == "CASH" else "tip"
565 self.emit("PAY", "table", number, method, amount, label, money(excess))
566 if method == "CARD":
567 self.tips += excess
568 check.server.tips += excess
569 self.close_check(check)

618 def cmd_report(self):
619 self.emit("REPORT", clock(self.time))
620 self.emit("COVERS", self.guests, "parties", self.parties)
621 self.emit("TURNS", self.closed, "avg_stay", self.stay_total // self.closed if self.closed else "-")
622 self.emit("SALES", *(part for course in COURSES for part in (course, money(self.sales[course]))))
623 subtotal, discount, service, tax, total = self.totals
624 self.emit(
625 "TOTALS", "subtotal", money(subtotal), "discount", money(discount), "service", money(service),
626 "tax", money(tax), "total", money(total),
627 )
628 self.emit("PAYMENTS", "cash", money(self.payments["CASH"]), "card", money(self.payments["CARD"]), "tips", money(self.tips))
629 self.emit("COMPS", self.comps, money(self.comp_amount), "VOIDS", self.voids)
630 top = sorted(self.served, key=lambda name: (-self.served[name], name))[:3]
631 for rank, name in enumerate(top, 1):
632 self.emit("TOP", rank, name, self.served[name])
633 for name in STATIONS:
