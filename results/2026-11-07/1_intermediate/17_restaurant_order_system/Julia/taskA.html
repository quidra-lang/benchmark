6struct CommandError <: Exception
7 reason::String
8end
9Base.showerror(io::IO, error::CommandError) = print(io, error.reason)
10fail(reason::String) = throw(CommandError(reason))

12cents(text::String) = parse(Int, replace(text, "." => ""))
13money(value::Int) = @sprintf("%d.%02d", fld(value, 100), mod(value, 100))
14clock(time::Int) = @sprintf("%02d:%02d", fld(660 + time, 60), mod(660 + time, 60))
15rounded(value::Int, rate::Int) = fld(value * rate + 50, 100)

108function Restaurant()
109 stations = Dict(
110 name => StationState(name == "BAR" ? 2 : 1, true,
111 OrderLine[], OrderLine[], Int[], 0)
112 for name in STATIONS
113 )
114 Restaurant(
115 0, Dict{Int, TableInfo}(), ServerInfo[], WaitingEntry[],
116 Dict{String, MenuItem}(), OrderLine[], stations,
117 0, 0, 0, 0, 0, 0,
118 Dict(name => 0 for name in COURSES),
119 Dict(key => 0 for key in ("subtotal", "discount", "service", "tax", "total")),
120 Dict("CASH" => 0, "CARD" => 0),
121 0, 0, 0, 0, Dict{String, Int}(), String[]
122 )

127function table_info(r::Restaurant, text::String)
128 number = parse(Int, text)
129 haskey(r.tables, number) || fail("unknown table $text")
130 r.tables[number]
131end
132
133function check_info(r::Restaurant, text::String)
134 item = table_info(r, text)
135 item.check === nothing && fail("table $text is free")
136 item.check::Check
137end
138
139function unlocked(check::Check)
140 check.paid > 0 && fail("check for table $(check.table.number) is locked")
141end
142
143function line_info(r::Restaurant, text::String)
144 number = parse(Int, text)
145 1 <= number <= length(r.lines) || fail("unknown line $text")
146 r.lines[number]
147end

149function modifiable(line::OrderLine)
150 line.status == "VOID" && fail("#$(line.id) is void")
151 line.check.closed && fail("#$(line.id) is on a closed check")
152 unlocked(line.check)
153end

155active_lines(check::Check) = [line for line in check.lines if line.status != "VOID"]
156
157function unserved(check::Check)
158 quantity = count(line -> line.status != "SERVED", active_lines(check))
159 quantity > 0 && fail("table $(check.table.number) has $quantity unserved items")
160end

176function server_load(r::Restaurant, server::ServerInfo)
177 count(item -> item.check !== nothing && item.check.server === server, values(r.tables))
178end
179
180function select_table(r::Restaurant, size::Int)
181 candidates = [item for item in values(r.tables)
182 if item.check === nothing && item.seats >= size]
183 isempty(candidates) && return nothing
184 sort!(candidates; by=item -> (item.seats, item.number))
185 first(candidates)
186end
187
188function seat_party!(r::Restaurant, party::String, size::Int, item::TableInfo)
189 index = argmin([server_load(r, server) for server in r.servers])
190 server = r.servers[index]
191 check = Check(party, size, item, server, r.now, OrderLine[], 0, 0, false)
192 item.check = check
193 server.parties += 1
194 r.covers += size
195 r.parties += 1
196 check
197end
198
199function process_waiting!(r::Restaurant)
200 remaining = WaitingEntry[]
201 for entry in r.waiting
202 item = select_table(r, entry.size)
203 if item === nothing
204 push!(remaining, entry)
205 else
206 check = seat_party!(r, entry.party, entry.size, item)
207 emit!(r, "SEATED", entry.party, entry.size, "table", item.number,
208 "server", check.server.name, "waited", r.now - entry.joined)
209 end
210 end
211 r.waiting = remaining
212end
213
214line_amount(line::OrderLine) = line.reason === nothing ? line.price * line.quantity : 0
215
216function check_amounts(check::Check)
217 subtotal = sum(line_amount, active_lines(check); init=0)
218 discount = rounded(subtotal, check.discount)
219 after = subtotal - discount
220 service = check.size >= 6 ? rounded(after, 18) : 0
221 tax = rounded(after, 8)
222 total = after + service + tax
223 (subtotal=subtotal, discount=discount, service=service,
224 tax=tax, total=total, due=total-check.paid)
225end
226
227function close_check!(r::Restaurant, check::Check)
228 amounts = check_amounts(check)
229 r.turns += 1
230 r.stay += r.now - check.seated
231 for line in active_lines(check)
232 r.sales[line.course] += line_amount(line)
233 end
234 for field in ("subtotal", "discount", "service", "tax", "total")
235 r.totals[field] += getproperty(amounts, Symbol(field))
236 end
237 check.server.sales += amounts.total
238 check.closed = true
239 check.table.check = nothing
240 emit!(r, "FREE", "table", check.table.number, check.party,
241 "stayed", r.now - check.seated)
242 process_waiting!(r)
243end

245function start_station!(station::StationState, time::Int)
246 while station.up && !isempty(station.queue) &&
247 length(station.cooking) < station.capacity
248 line = popfirst!(station.queue)
249 line.status = "COOKING"
250 line.ready = time + line.prep
251 line.started = true
252 push!(station.cooking, line)
253 end
254end
255
256function tick!(r::Restaurant, minutes::Int)
257 finish = r.now + minutes
258 emit!(r, "TICK", clock(finish))
259 while true
260 cooking = [line for name in STATIONS for line in r.stations[name].cooking]
261 isempty(cooking) && break
262 time = minimum(line.ready for line in cooking)
263 time > finish && break
264 for name in STATIONS
265 station = r.stations[name]
266 completed = [line for line in station.cooking if line.ready == time]
267 filter!(line -> line.ready != time, station.cooking)
268 sort!(completed; by=line -> line.id)
269 for line in completed
270 line.status = "READY"
271 push!(station.tickets, time - line.fire)
272 emit!(r, "READY", clock(time), name, "#" * string(line.id), line.item,
273 "x" * string(line.quantity), "table", line.check.table.number,
274 "seat", line.seat)
275 end
276 end
277 for name in STATIONS
278 start_station!(r.stations[name], time)
279 end
280 end
281 r.now = finish
282end

284function seat_totals(check::Check)
285 totals = Dict{Int, Int}()
286 for line in active_lines(check)
287 totals[line.seat] = get(totals, line.seat, 0) + line_amount(line)
288 end
289 totals
290end

292function allocate(values::Dict{Int, Int}, rate::Int)
293 result = Dict(seat => fld(value * rate, 100) for (seat, value) in values)
294 left = rounded(sum(Base.values(values); init=0), rate) -
295 sum(Base.values(result); init=0)
296 order = sort(collect(keys(values)); by=seat -> (-mod(values[seat] * rate, 100), seat))
297 for seat in order[1:left]
298 result[seat] += 1
299 end
300 result
301end

303function show_bill!(r::Restaurant, check::Check)
304 emit!(r, "BILL", "table", check.table.number, check.party, "guests", check.size)
305 totals = seat_totals(check)
306 for seat in sort(collect(keys(totals)))
307 emit!(r, "SEAT", seat, money(totals[seat]))
308 for line in active_lines(check)
309 line.seat == seat || continue
310 if line.reason === nothing
311 emit!(r, "#" * string(line.id), line.item, "x" * string(line.quantity),
312 money(line_amount(line)))
313 else
314 emit!(r, "#" * string(line.id), line.item, "x" * string(line.quantity),
315 money(line_amount(line)), "COMP", line.reason)
316 end
317 end
318 end
319 amounts = check_amounts(check)
320 emit!(r, "SUBTOTAL", money(amounts.subtotal))
321 if check.discount != 0
322 emit!(r, "DISCOUNT", "$(check.discount)%", money(amounts.discount))
323 end
324 check.size >= 6 && emit!(r, "SERVICE", money(amounts.service))
325 emit!(r, "TAX", money(amounts.tax))
326 emit!(r, "TOTAL", money(amounts.total))
327 if check.paid > 0
328 emit!(r, "PAID", money(check.paid))
329 emit!(r, "DUE", money(amounts.due))
330 end
331end

333function show_report!(r::Restaurant)
334 emit!(r, "REPORT", clock(r.now))
335 emit!(r, "COVERS", r.covers, "parties", r.parties)
336 emit!(r, "TURNS", r.turns, "avg_stay",
337 r.turns > 0 ? fld(r.stay, r.turns) : "-")
338 sales = Any["SALES"]
339 for course in COURSES
340 append!(sales, (course, money(r.sales[course])))
341 end
342 emit!(r, sales...)
343 fields = Any["TOTALS"]
344 for field in ("subtotal", "discount", "service", "tax", "total")
345 append!(fields, (field, money(r.totals[field])))
346 end
347 emit!(r, fields...)
348 emit!(r, "PAYMENTS", "cash", money(r.payments["CASH"]),
349 "card", money(r.payments["CARD"]), "tips", money(r.tips))
350 emit!(r, "COMPS", r.comps, money(r.comp_amount), "VOIDS", r.voids)
351 top = sort(collect(r.served); by=kv -> (-kv.second, kv.first))
352 for (rank, entry) in enumerate(top[1:min(3, length(top))])
353 emit!(r, "TOP", rank, entry.first, entry.second)
354 end
355 for name in STATIONS
356 station = r.stations[name]
357 tickets = station.tickets
358 emit!(r, name, "tickets", length(tickets),
359 "avg", isempty(tickets) ? "-" : fld(sum(tickets), length(tickets)),
360 "max", isempty(tickets) ? "-" : maximum(tickets),
361 "remakes", station.remakes)
362 end
363 for server in r.servers
364 emit!(r, "SERVER", server.name, "tables", server_load(r, server),
365 "parties", server.parties, "sales", money(server.sales),
366 "tips", money(server.tips))
367 end
368end

370function run_command!(r::Restaurant, command::String, a::Vector{String})
371 if command == "TABLE"
372 number = parse(Int, a[1])
373 haskey(r.tables, number) && fail("table $(a[1]) already exists")
374 item = TableInfo(number, parse(Int, a[2]), nothing)
375 r.tables[number] = item
376 emit!(r, "TABLE", number, "seats", item.seats)
377 process_waiting!(r)
378 elseif command == "STAFF"
379 any(server.name == a[1] for server in r.servers) &&
380 fail("$(a[1]) is already on duty")
381 push!(r.servers, ServerInfo(a[1], 0, 0, 0))
382 emit!(r, "STAFF", a[1])
383 elseif command == "SEAT"
384 for item in values(r.tables)
385 if item.check !== nothing && item.check.party == a[1]
386 fail("$(a[1]) is already seated at table $(item.number)")
387 end
388 end
389 any(entry.party == a[1] for entry in r.waiting) &&
390 fail("$(a[1]) is already waiting")
391 size = parse(Int, a[2])
392 any(item.seats >= size for item in values(r.tables)) ||
393 fail("no table for party of $size")
394 isempty(r.servers) && fail("no server on duty")
395 item = select_table(r, size)
396 if item === nothing
397 push!(r.waiting, WaitingEntry(a[1], size, r.now))
398 emit!(r, "WAIT", a[1], size, "position", length(r.waiting))
399 else
400 check = seat_party!(r, a[1], size, item)
401 emit!(r, "SEAT", a[1], size, "table", item.number,
402 "server", check.server.name)
403 end
404 elseif command == "CANCEL"
405 index = findfirst(entry -> entry.party == a[1], r.waiting)
406 index === nothing && fail("$(a[1]) is not waiting")
407 entry = r.waiting[index]
408 deleteat!(r.waiting, index)
409 emit!(r, "CANCEL", entry.party, entry.size, "waited", r.now - entry.joined)
410 elseif command == "MOVE"
411 source = table_info(r, a[1])
412 destination = table_info(r, a[2])
413 source.check === nothing && fail("table $(a[1]) is free")
414 destination.check !== nothing && fail("table $(a[2]) is occupied")
415 check = source.check::Check
416 destination.seats < check.size &&
417 fail("table $(a[2]) is too small for $(check.party)")
418 source.check = nothing
419 destination.check = check
420 check.table = destination
421 emit!(r, "MOVE", check.party, "table", source.number, "->", destination.number)
422 process_waiting!(r)
423 elseif command == "MENU"
424 station_info(r, a[2])
425 require_course(a[5])
426 previous = get(r.menu, a[1], nothing)
427 item = MenuItem(a[1], a[2], parse(Int, a[3]), cents(a[4]), a[5],
428 previous === nothing ? nothing : previous.stock)
429 r.menu[a[1]] = item
430 emit!(r, "MENU", previous === nothing ? "ADD" : "UPDATE",
431 item.name, item.station, item.prep, money(item.price), item.course)
432 elseif command == "STOCK"
433 item = menu_item(r, a[1])
434 item.stock = parse(Int, a[2])
435 emit!(r, "STOCK", item.name, item.stock)
436 elseif command == "HAPPY"
437 r.happy_rate = parse(Int, a[1])
438 r.happy_end = r.now + parse(Int, a[2])
439 emit!(r, "HAPPY", "$(r.happy_rate)%", "until", clock(r.happy_end))
440 elseif command == "ITEMS"
441 items = sort(collect(values(r.menu)); by=item ->
442 (findfirst(==(item.course), COURSES), item.name))
443 emit!(r, "ITEMS", length(items))
444 for item in items
445 emit!(r, item.course, item.name, item.station, item.prep,
446 money(item.price), item.stock === nothing ? "-" : item.stock)
447 end
448 elseif command == "ORDER"
449 check = check_info(r, a[1])
450 unlocked(check)
451 seat = parse(Int, a[2])
452 seat > check.size && fail("seat $seat out of range for table $(a[1])")
453 item = menu_item(r, a[3])
454 quantity = parse(Int, a[4])
455 if item.stock !== nothing && item.stock < quantity
456 fail("only $(item.stock) $(item.name) left")
457 end
458 happy = item.course == "DRINK" && r.now < r.happy_end
459 price = item.price - (happy ? rounded(item.price, r.happy_rate) : 0)
460 line = OrderLine(length(r.lines)+1, check, seat, item.name, quantity,
461 item.station, item.prep, price, item.course,
462 "ORDERED", nothing, 0, 0, false)
463 push!(r.lines, line)
464 push!(check.lines, line)
465 entry = Any["ORDER", "#" * string(line.id), "table", check.table.number,
466 "seat", seat, item.name, "x" * string(quantity), money(line_amount(line))]
467 happy && push!(entry, "HAPPY")
468 emit!(r, entry...)
469 if item.stock !== nothing
470 item.stock -= quantity
471 item.stock == 0 && emit!(r, "SOLDOUT", item.name)
472 end
473 elseif command == "FIRE"
474 check = check_info(r, a[1])
475 require_course(a[2])
476 lines = [line for line in check.lines
477 if line.course == a[2] && line.status == "ORDERED"]
478 isempty(lines) && fail("nothing to fire for $(a[2]) at table $(a[1])")
479 if a[2] in ("MAIN", "DESSERT")
480 earlier = a[2] == "MAIN" ? ("STARTER",) : ("STARTER", "MAIN")
481 for before in earlier
482 any(line.course == before && line.status == "ORDERED"
483 for line in check.lines) && fail("fire $before first")
484 end
485 end
486 for line in lines
487 line.status = "QUEUED"
488 line.fire = r.now
489 push!(r.stations[line.station].queue, line)
490 end
491 emit!(r, "FIRE", "table", check.table.number, a[2],
492 ["#" * string(line.id) for line in lines]...)
493 for name in STATIONS
494 start_station!(r.stations[name], r.now)
495 end
496 elseif command == "TICK"
497 tick!(r, parse(Int, a[1]))
498 elseif command == "SERVE"
499 check = check_info(r, a[1])
500 lines = [line for line in check.lines if line.status == "READY"]
501 isempty(lines) && fail("nothing ready for table $(a[1])")
502 for line in lines
503 line.status = "SERVED"
504 r.served[line.item] = get(r.served, line.item, 0) + line.quantity
505 end
506 emit!(r, "SERVE", "table", check.table.number,
507 ["#" * string(line.id) for line in lines]...)
508 elseif command == "RUSH"
509 line = line_info(r, a[1])
510 line.status == "QUEUED" || fail("#$(line.id) is not queued")
511 queue = r.stations[line.station].queue
512 filter!(other -> other !== line, queue)
513 pushfirst!(queue, line)
514 emit!(r, "RUSH", "#" * string(line.id), line.station)
515 elseif command == "REMAKE"
516 line = line_info(r, a[1])
517 line.check.closed && fail("#$(line.id) is on a closed check")
518 line.status in ("READY", "SERVED") ||
519 fail("#$(line.id) is not ready or served")
520 line.status = "QUEUED"
521 line.fire = r.now
522 station = r.stations[line.station]
523 pushfirst!(station.queue, line)
524 station.remakes += 1
525 emit!(r, "REMAKE", "#" * string(line.id), line.item, line.station)
526 start_station!(station, r.now)
527 elseif command == "VOID"
528 line = line_info(r, a[1])
529 line.status == "VOID" && fail("#$(line.id) is already void")
530 line.started && fail("#$(line.id) has already started")
531 if line.status == "QUEUED"
532 filter!(other -> other !== line, r.stations[line.station].queue)
533 end
534 line.status = "VOID"
535 item = r.menu[line.item]
536 if item.stock !== nothing
537 item.stock += line.quantity
538 end
539 r.voids += 1
540 emit!(r, "VOID", "#" * string(line.id), line.item, "x" * string(line.quantity))
541 elseif command == "STATION"
542 station = station_info(r, a[1])
543 a[2] == "DOWN" && !station.up && fail("station $(a[1]) is already down")
544 a[2] == "UP" && station.up && fail("station $(a[1]) is not down")
545 station.up = a[2] == "UP"
546 station.up && start_station!(station, r.now)
547 emit!(r, "STATION", a[1], a[2])
548 elseif command in ("COMP", "TRANSFER")
549 line = line_info(r, a[1])
550 modifiable(line)
551 if command == "COMP"
552 line.reason !== nothing && fail("#$(line.id) is already comped")
553 amount = line_amount(line)
554 emit!(r, "COMP", "#" * string(line.id), line.item, money(amount), a[2])
555 r.comps += 1
556 r.comp_amount += amount
557 line.reason = a[2]
558 else
559 seat = parse(Int, a[2])
560 seat > line.check.size &&
561 fail("seat $seat out of range for table $(line.check.table.number)")
562 seat == line.seat && fail("#$(line.id) is already on seat $seat")
563 emit!(r, "TRANSFER", "#" * string(line.id), "seat", line.seat, "->", seat)
564 line.seat = seat
565 end
566 elseif command == "DISCOUNT"
567 check = check_info(r, a[1])
568 unlocked(check)
569 check.discount = parse(Int, a[2])
570 emit!(r, "DISCOUNT", "table", check.table.number, "$(check.discount)%")
571 elseif command == "BILL"
572 show_bill!(r, check_info(r, a[1]))
573 elseif command == "SPLIT"
574 check = check_info(r, a[1])
575 amounts = check_amounts(check)
576 amounts.due == 0 && fail("nothing to split for table $(a[1])")
577 if a[2] == "EVEN"
578 ways = parse(Int, a[3])
579 ways > check.size && fail("cannot split table $(a[1]) $ways ways")
580 emit!(r, "SPLIT", "table", check.table.number, "EVEN", ways,
581 "due", money(amounts.due))
582 quotient, remainder = divrem(amounts.due, ways)
583 for index in 1:ways
584 emit!(r, "PAYER", index, money(quotient + (index <= remainder ? 1 : 0)))
585 end
586 else
587 check.paid > 0 && fail("table $(a[1]) has payments")
588 seatValues = seat_totals(check)
589 discounts = allocate(seatValues, check.discount)
590 after = Dict(seat => subtotal - discounts[seat] for (seat, subtotal) in seatValues)
591 services = allocate(after, check.size >= 6 ? 18 : 0)
592 taxes = allocate(after, 8)
593 emit!(r, "SPLIT", "table", check.table.number, "BYSEAT",
594 "total", money(amounts.total))
595 for seat in sort(collect(keys(seatValues)))
596 emit!(r, "SEAT", seat, "subtotal", money(seatValues[seat]),
597 "discount", money(discounts[seat]), "service", money(services[seat]),
598 "tax", money(taxes[seat]),
599 "total", money(after[seat]+services[seat]+taxes[seat]))
600 end
601 end
602 elseif command == "PAY"
603 check = check_info(r, a[1])
604 a[2] in ("CASH", "CARD") || fail("unknown method $(a[2])")
605 unserved(check)
606 due = check_amounts(check).due
607 due == 0 && fail("nothing to pay for table $(a[1])")
608 amount = cents(a[3])
609 applied = min(amount, due)
610 check.paid += applied
611 r.payments[a[2]] += applied
612 if amount < due
613 emit!(r, "PAY", "table", check.table.number, a[2], money(amount),
614 "remaining", money(due - amount))
615 else
616 emit!(r, "PAY", "table", check.table.number, a[2], money(amount),
617 a[2] == "CASH" ? "change" : "tip", money(amount - due))
618 if a[2] == "CARD"
619 r.tips += amount - due
620 check.server.tips += amount - due
621 end
622 close_check!(r, check)
623 end
624 elseif command == "CLOSE"
625 check = check_info(r, a[1])
626 unserved(check)
627 due = check_amounts(check).due
628 due != 0 && fail("table $(a[1]) still owes $(money(due))")
629 close_check!(r, check)
630 elseif command == "STATUS"
631 check = check_info(r, a[1])
632 emit!(r, "STATUS", "table", check.table.number, check.party, "guests",
633 check.size, "since", clock(check.seated))
634 for line in check.lines
635 fields = Any["#" * string(line.id), "seat", line.seat, line.item,
636 "x" * string(line.quantity), line.course, line.status]
637 if line.status == "COOKING"
638 append!(fields, ("until", clock(line.ready)))
639 end
640 if line.reason !== nothing
641 append!(fields, ("COMP", line.reason))
642 end
643 emit!(r, fields...)
644 end
645 elseif command == "KITCHEN"
646 emit!(r, "KITCHEN", clock(r.now))
647 for name in STATIONS
648 station = r.stations[name]
649 cooking = sort(station.cooking; by=line -> (line.ready, line.id))
650 cooking_text = isempty(cooking) ? "-" :
651 join(["#$(line.id)@$(clock(line.ready))" for line in cooking], " ")
652 queue_text = isempty(station.queue) ? "-" :
653 join(["#$(line.id)" for line in station.queue], " ")
654 emit!(r, name, station.up ? "up" : "down",
655 "cooking", cooking_text, "queue", queue_text)
656 end
657 elseif command == "FLOOR"
658 tables = sort(collect(values(r.tables)); by=item -> item.number)
659 free = count(item -> item.check === nothing, tables)
660 emit!(r, "FLOOR", clock(r.now), "free", free,
661 "seated", length(tables)-free, "waiting", length(r.waiting))
662 for item in tables
663 if item.check === nothing
664 emit!(r, "table", item.number, "seats", item.seats, "free")
665 else
666 check = item.check::Check
667 fields = Any["table", item.number, "seats", item.seats, check.party,
668 "guests", check.size, "server", check.server.name,
669 "since", clock(check.seated), "items", length(active_lines(check)),
670 "total", money(check_amounts(check).total)]
671 check.paid > 0 && append!(fields, ("paid", money(check.paid)))
672 emit!(r, fields...)
673 end
674 end
675 for (index, entry) in enumerate(r.waiting)
676 emit!(r, "WAITING", index, entry.party, entry.size, "since", clock(entry.joined))
677 end
678 elseif command == "REPORT"
679 show_report!(r)
680 end
681end

683function execute!(r::Restaurant, line::String)
684 tokens = String.(split(line))
685 command = first(tokens)
686 try
687 run_command!(r, command, tokens[2:end])
688 catch error
689 if error isa CommandError
690 emit!(r, "ERROR", error.reason)
691 else
692 rethrow()
693 end
694 end
695end

697function main()
698 count = parse(Int, readline(stdin))
699 restaurant = Restaurant()
700 for _ in 1:count
701 execute!(restaurant, readline(stdin))
702 end
703 if !isempty(restaurant.output)
704 println(join(restaurant.output, "\n"))
705 end
706end
