3type Server = {name: string; parties: number; sales: number; tips: number};
4type Table = {number: number; seats: number; check: Check | null};
5type Check = {party: string; size: number; table: Table; server: Server; seated: number; lines: Line[]; paid: number; discount: number; closed: boolean};
6type MenuItem = {name: string; station: string; prep: number; price: number; course: string; stock: number | null};
7type Line = {id: number; check: Check; seat: number; item: string; quantity: number; station: string; prep: number; price: number; course: string; state: string; reason: string | null; fire: number; ready: number; started: boolean};
8type Station = {name: string; capacity: number; up: boolean; queue: Line[]; cooking: Line[]; tickets: number[]; remakes: number};
9type Amounts = {subtotal: number; discount: number; service: number; tax: number; total: number; due: number};
10type Waiting = {party: string; size: number; joined: number};
11const STATIONS = ["GRILL", "FRY", "COLD", "BAR"];
12const COURSES = ["DRINK", "STARTER", "MAIN", "DESSERT"];
13const cents = (s: string): number => Number(s.replace(".", ""));
14const money = (v: number): string => Math.floor(v / 100) + "." + String(v % 100).padStart(2, "0");
15const clock = (t: number): string => String(Math.floor((660 + t) / 60)).padStart(2, "0") + ":" + String((660 + t) % 60).padStart(2, "0");
16const rounded = (v: number, rate: number): number => Math.floor((v * rate + 50) / 100);
17const amount = (l: Line): number => l.reason === null ? l.price * l.quantity : 0;
18const active = (c: Check): Line[] => c.lines.filter(l => l.state !== "VOID");
19const lex = (a: string, b: string): number => a < b ? -1 : a > b ? 1 : 0;
20class CommandError extends Error {}

22class Restaurant {
23 now = 0;
24 tables = new Map<number, Table>();
25 servers: Server[] = [];
26 waiting: Waiting[] = [];
27 menu = new Map<string, MenuItem>();
28 lines: Line[] = [];
29 stations = new Map<string, Station>(STATIONS.map(name => [name, {name, capacity: name === "BAR" ? 2 : 1, up: true, queue: [], cooking: [], tickets: [], remakes: 0}]));
30 happyRate = 0;
31 happyEnd = 0;
32 covers = 0;
33 parties = 0;
34 turns = 0;
35 stay = 0;
36 sales: Record<string, number> = Object.fromEntries(COURSES.map(c => [c, 0]));
37 totals: Record<string, number> = {subtotal: 0, discount: 0, service: 0, tax: 0, total: 0};
38 payments: Record<string, number> = {CASH: 0, CARD: 0};
39 tips = 0;
40 comps = 0;
41 compAmount = 0;
42 voids = 0;
43 served = new Map<string, number>();
44 output: string[] = [];

46 emit(...parts: (string | number)[]): void {
47 this.output.push(parts.join(" "));

50 table(text: string): Table {
51 const t = this.tables.get(Number(text));
52 if (!t) throw new CommandError("unknown table " + text);
53 return t;
54 }
55
56 check(text: string): Check {
57 const t = this.table(text);
58 if (!t.check) throw new CommandError("table " + text + " is free");
59 return t.check;
60 }
61
62 unlocked(c: Check): void {
63 if (c.paid > 0) throw new CommandError("check for table " + c.table.number + " is locked");
64 }
65
66 line(text: string): Line {
67 const n = Number(text);
68 if (n < 1 || n > this.lines.length) throw new CommandError("unknown line " + text);
69 return this.lines[n - 1];
70 }
71
72 modifiable(l: Line): void {
73 if (l.state === "VOID") throw new CommandError("#" + l.id + " is void");
74 if (l.check.closed) throw new CommandError("#" + l.id + " is on a closed check");
75 this.unlocked(l.check);
76 }
77
78 unserved(c: Check): void {
79 const count = active(c).filter(l => l.state !== "SERVED").length;
80 if (count) throw new CommandError("table " + c.table.number + " has " + count + " unserved items");
81 }
82
83 station(name: string): Station {
84 const s = this.stations.get(name);
85 if (!s) throw new CommandError("unknown station " + name);
86 return s;
87 }
88
89 course(name: string): void {
90 if (!COURSES.includes(name)) throw new CommandError("unknown course " + name);
91 }
92
93 item(name: string): MenuItem {
94 const i = this.menu.get(name);
95 if (!i) throw new CommandError("unknown item " + name);
96 return i;
97 }

99 load(server: Server): number {
100 return [...this.tables.values()].filter(t => t.check?.server === server).length;
101 }
102
103 select(size: number): Table | null {
104 return [...this.tables.values()].filter(t => t.check === null && t.seats >= size).sort((a, b) => a.seats - b.seats || a.number - b.number)[0] ?? null;
105 }
106
107 seat(party: string, size: number, table: Table): Check {
108 let server = this.servers[0];
109 for (const candidate of this.servers) {
110 if (this.load(candidate) < this.load(server)) server = candidate;
111 }
112 const c: Check = {party, size, table, server, seated: this.now, lines: [], paid: 0, discount: 0, closed: false};
113 table.check = c;
114 server.parties++;
115 this.covers += size;
116 this.parties++;
117 return c;
118 }
119
120 processWaiting(): void {
121 const remaining: Waiting[] = [];
122 for (const w of this.waiting) {
123 const t = this.select(w.size);
124 if (t) {
125 const c = this.seat(w.party, w.size, t);
126 this.emit("SEATED", w.party, w.size, "table", t.number, "server", c.server.name, "waited", this.now - w.joined);
127 } else remaining.push(w);
128 }
129 this.waiting = remaining;
130 }
131
132 amounts(c: Check): Amounts {
133 const subtotal = active(c).reduce((v, l) => v + amount(l), 0);
134 const discount = rounded(subtotal, c.discount);
135 const after = subtotal - discount;
136 const service = c.size >= 6 ? rounded(after, 18) : 0;
137 const tax = rounded(after, 8);
138 const total = after + service + tax;
139 return {subtotal, discount, service, tax, total, due: total - c.paid};
140 }
141
142 close(c: Check): void {
143 const a = this.amounts(c);
144 this.turns++;
145 this.stay += this.now - c.seated;
146 for (const l of active(c)) this.sales[l.course] += amount(l);
147 for (const key of ["subtotal", "discount", "service", "tax", "total"]) this.totals[key] += a[key as keyof Amounts];
148 c.server.sales += a.total;
149 c.closed = true;
150 c.table.check = null;
151 this.emit("FREE", "table", c.table.number, c.party, "stayed", this.now - c.seated);
152 this.processWaiting();
153 }
154
155 start(s: Station, time: number): void {
156 while (s.up && s.queue.length && s.cooking.length < s.capacity) {
157 const l = s.queue.shift()!;
158 l.state = "COOKING";
159 l.ready = time + l.prep;
160 l.started = true;
161 s.cooking.push(l);
162 }
163 }
164
165 tick(minutes: number): void {
166 const end = this.now + minutes;
167 this.emit("TICK", clock(end));
168 while (true) {
169 const cooking = [...this.stations.values()].flatMap(s => s.cooking);
170 if (!cooking.length) break;
171 const time = Math.min(...cooking.map(l => l.ready));
172 if (time > end) break;
173 for (const name of STATIONS) {
174 const s = this.stations.get(name)!;
175 const finished = s.cooking.filter(l => l.ready === time).sort((a, b) => a.id - b.id);
176 s.cooking = s.cooking.filter(l => l.ready !== time);
177 for (const l of finished) {
178 l.state = "READY";
179 s.tickets.push(time - l.fire);
180 this.emit("READY", clock(time), name, "#" + l.id, l.item, "x" + l.quantity, "table", l.check.table.number, "seat", l.seat);
181 }
182 }
183 for (const s of this.stations.values()) this.start(s, time);
184 }
185 this.now = end;
186 }

188 seats(c: Check): Map<number, number> {
189 const groups = Map.groupBy(active(c), l => l.seat);
190 return new Map([...groups].sort((a, b) => a[0] - b[0]).map(([seat, lines]) =>
191 [seat, lines.reduce((subtotal, line) => subtotal + amount(line), 0)]));
192 }

194 allocate(values: Map<number, number>, rate: number): Map<number, number> {
195 const result = new Map([...values].map(([seat, v]): [number, number] => [seat, Math.floor(v * rate / 100)]));
196 const total = [...values.values()].reduce((a, b) => a + b, 0);
197 const left = rounded(total, rate) - [...result.values()].reduce((a, b) => a + b, 0);
198 const order = [...values.keys()].sort((a, b) => values.get(b)! * rate % 100 - values.get(a)! * rate % 100 || a - b);
199 for (const seat of order.slice(0, left)) result.set(seat, result.get(seat)! + 1);
200 return result;
201 }

203 bill(c: Check): void {
204 this.emit("BILL", "table", c.table.number, c.party, "guests", c.size);
205 for (const [seat, subtotal] of this.seats(c)) {
206 this.emit("SEAT", seat, money(subtotal));
207 for (const l of active(c).filter(l => l.seat === seat)) this.emit("#" + l.id, l.item, "x" + l.quantity, money(amount(l)), ...(l.reason === null ? [] : ["COMP", l.reason]));
208 }
209 const a = this.amounts(c);
210 this.emit("SUBTOTAL", money(a.subtotal));
211 if (c.discount) this.emit("DISCOUNT", c.discount + "%", money(a.discount));
212 if (c.size >= 6) this.emit("SERVICE", money(a.service));
213 this.emit("TAX", money(a.tax));
214 this.emit("TOTAL", money(a.total));
215 if (c.paid > 0) {
216 this.emit("PAID", money(c.paid));
217 this.emit("DUE", money(a.due));
218 }
219 }

221 report(): void {
222 this.emit("REPORT", clock(this.now));
223 this.emit("COVERS", this.covers, "parties", this.parties);
224 this.emit("TURNS", this.turns, "avg_stay", this.turns ? Math.floor(this.stay / this.turns) : "-");
225 this.emit("SALES", ...COURSES.flatMap(c => [c, money(this.sales[c])]));
226 this.emit("TOTALS", ...["subtotal", "discount", "service", "tax", "total"].flatMap(k => [k, money(this.totals[k])]));
227 this.emit("PAYMENTS", "cash", money(this.payments.CASH), "card", money(this.payments.CARD), "tips", money(this.tips));
228 this.emit("COMPS", this.comps, money(this.compAmount), "VOIDS", this.voids);
229 const top = [...this.served].filter(([, n]) => n > 0).sort((a, b) => b[1] - a[1] || lex(a[0], b[0])).slice(0, 3);
230 top.forEach(([name, count], i) => this.emit("TOP", i + 1, name, count));
231 for (const s of this.stations.values()) {
232 this.emit(s.name, "tickets", s.tickets.length, "avg", s.tickets.length ? Math.floor(s.tickets.reduce((a, b) => a + b, 0) / s.tickets.length) : "-",
233 "max", s.tickets.length ? Math.max(...s.tickets) : "-", "remakes", s.remakes);
234 }
235 for (const s of this.servers) this.emit("SERVER", s.name, "tables", this.load(s), "parties", s.parties, "sales", money(s.sales), "tips", money(s.tips));
236 }

238 command(cmd: string, a: string[]): void {
239 switch (cmd) {
240 case "TABLE": {
241 const number = Number(a[0]);
242 if (this.tables.has(number)) throw new CommandError("table " + a[0] + " already exists");
243 this.tables.set(number, {number, seats: Number(a[1]), check: null});
244 this.emit("TABLE", number, "seats", Number(a[1]));
245 this.processWaiting();
246 break;
247 }
248 case "STAFF":
249 if (this.servers.some(s => s.name === a[0])) throw new CommandError(a[0] + " is already on duty");
250 this.servers.push({name: a[0], parties: 0, sales: 0, tips: 0});
251 this.emit("STAFF", a[0]);
252 break;
253 case "SEAT": {
254 const seated = [...this.tables.values()].find(t => t.check?.party === a[0]);
255 if (seated) throw new CommandError(a[0] + " is already seated at table " + seated.number);
256 if (this.waiting.some(w => w.party === a[0])) throw new CommandError(a[0] + " is already waiting");
257 const size = Number(a[1]);
258 if (![...this.tables.values()].some(t => t.seats >= size)) throw new CommandError("no table for party of " + size);
259 if (!this.servers.length) throw new CommandError("no server on duty");
260 const t = this.select(size);
261 if (t) {
262 const c = this.seat(a[0], size, t);
263 this.emit("SEAT", a[0], size, "table", t.number, "server", c.server.name);
264 } else {
265 this.waiting.push({party: a[0], size, joined: this.now});
266 this.emit("WAIT", a[0], size, "position", this.waiting.length);
267 }
268 break;
269 }
270 case "CANCEL": {
271 const i = this.waiting.findIndex(w => w.party === a[0]);
272 if (i < 0) throw new CommandError(a[0] + " is not waiting");
273 const [w] = this.waiting.splice(i, 1);
274 this.emit("CANCEL", w.party, w.size, "waited", this.now - w.joined);
275 break;
276 }
277 case "MOVE": {
278 const source = this.table(a[0]);
279 const destination = this.table(a[1]);
280 if (!source.check) throw new CommandError("table " + a[0] + " is free");
281 if (destination.check) throw new CommandError("table " + a[1] + " is occupied");
282 const c = source.check;
283 if (destination.seats < c.size) throw new CommandError("table " + a[1] + " is too small for " + c.party);
284 source.check = null;
285 destination.check = c;
286 c.table = destination;
287 this.emit("MOVE", c.party, "table", source.number, "->", destination.number);
288 this.processWaiting();
289 break;
290 }
291 case "MENU": {
292 this.station(a[1]);
293 this.course(a[4]);
294 const old = this.menu.get(a[0]);
295 this.menu.set(a[0], {name: a[0], station: a[1], prep: Number(a[2]), price: cents(a[3]), course: a[4], stock: old?.stock ?? null});
296 this.emit("MENU", old ? "UPDATE" : "ADD", a[0], a[1], Number(a[2]), money(cents(a[3])), a[4]);
297 break;
298 }
299 case "STOCK": {
300 const item = this.item(a[0]);
301 item.stock = Number(a[1]);
302 this.emit("STOCK", item.name, item.stock);
303 break;
304 }
305 case "HAPPY":
306 this.happyRate = Number(a[0]);
307 this.happyEnd = this.now + Number(a[1]);
308 this.emit("HAPPY", this.happyRate + "%", "until", clock(this.happyEnd));
309 break;
310 case "ITEMS": {
311 const items = [...this.menu.values()].sort((a, b) => COURSES.indexOf(a.course) - COURSES.indexOf(b.course) || lex(a.name, b.name));
312 this.emit("ITEMS", items.length);
313 for (const i of items) this.emit(i.course, i.name, i.station, i.prep, money(i.price), i.stock === null ? "-" : i.stock);
314 break;
315 }
316 case "ORDER": {
317 const c = this.check(a[0]);
318 this.unlocked(c);
319 const seat = Number(a[1]);
320 if (seat > c.size) throw new CommandError("seat " + seat + " out of range for table " + a[0]);
321 const item = this.item(a[2]);
322 const quantity = Number(a[3]);
323 if (item.stock !== null && item.stock < quantity) throw new CommandError("only " + item.stock + " " + item.name + " left");
324 const happy = item.course === "DRINK" && this.now < this.happyEnd;
325 const price = item.price - (happy ? rounded(item.price, this.happyRate) : 0);
326 const l: Line = {id: this.lines.length + 1, check: c, seat, item: item.name, quantity, station: item.station,
327 prep: item.prep, price, course: item.course, state: "ORDERED", reason: null, fire: 0, ready: 0, started: false};
328 this.lines.push(l);
329 c.lines.push(l);
330 this.emit("ORDER", "#" + l.id, "table", c.table.number, "seat", seat, item.name, "x" + quantity, money(amount(l)), ...(happy ? ["HAPPY"] : []));
331 if (item.stock !== null) {
332 item.stock -= quantity;
333 if (item.stock === 0) this.emit("SOLDOUT", item.name);
334 }
335 break;
336 }
337 case "FIRE": {
338 const c = this.check(a[0]);
339 this.course(a[1]);
340 const lines = c.lines.filter(l => l.course === a[1] && l.state === "ORDERED");
341 if (!lines.length) throw new CommandError("nothing to fire for " + a[1] + " at table " + a[0]);
342 for (const earlier of ["STARTER", "MAIN", "DESSERT"].slice(0, ["STARTER", "MAIN", "DESSERT"].indexOf(a[1]))) {
343 if (a[1] !== "DRINK" && c.lines.some(l => l.course === earlier && l.state === "ORDERED")) throw new CommandError("fire " + earlier + " first");
344 }
345 for (const l of lines) {
346 l.state = "QUEUED";
347 l.fire = this.now;
348 this.stations.get(l.station)!.queue.push(l);
349 }
350 this.emit("FIRE", "table", c.table.number, a[1], ...lines.map(l => "#" + l.id));
351 for (const s of this.stations.values()) this.start(s, this.now);
352 break;
353 }
354 case "TICK":
355 this.tick(Number(a[0]));
356 break;
357 case "SERVE": {
358 const c = this.check(a[0]);
359 const lines = c.lines.filter(l => l.state === "READY");
360 if (!lines.length) throw new CommandError("nothing ready for table " + a[0]);
361 for (const l of lines) {
362 l.state = "SERVED";
363 this.served.set(l.item, (this.served.get(l.item) ?? 0) + l.quantity);
364 }
365 this.emit("SERVE", "table", c.table.number, ...lines.map(l => "#" + l.id));
366 break;
367 }
368 case "RUSH": {
369 const l = this.line(a[0]);
370 if (l.state !== "QUEUED") throw new CommandError("#" + l.id + " is not queued");
371 const s = this.stations.get(l.station)!;
372 s.queue = [l, ...s.queue.filter(x => x !== l)];
373 this.emit("RUSH", "#" + l.id, l.station);
374 break;
375 }
376 case "REMAKE": {
377 const l = this.line(a[0]);
378 if (l.check.closed) throw new CommandError("#" + l.id + " is on a closed check");
379 if (!["READY", "SERVED"].includes(l.state)) throw new CommandError("#" + l.id + " is not ready or served");
380 l.state = "QUEUED";
381 l.fire = this.now;
382 const s = this.stations.get(l.station)!;
383 s.queue.unshift(l);
384 s.remakes++;
385 this.emit("REMAKE", "#" + l.id, l.item, l.station);
386 this.start(s, this.now);
387 break;
388 }
389 case "VOID": {
390 const l = this.line(a[0]);
391 if (l.state === "VOID") throw new CommandError("#" + l.id + " is already void");
392 if (l.started) throw new CommandError("#" + l.id + " has already started");
393 if (l.state === "QUEUED") {
394 const s = this.stations.get(l.station)!;
395 s.queue = s.queue.filter(x => x !== l);
396 }
397 l.state = "VOID";
398 const item = this.menu.get(l.item)!;
399 if (item.stock !== null) item.stock += l.quantity;
400 this.voids++;
401 this.emit("VOID", "#" + l.id, l.item, "x" + l.quantity);
402 break;
403 }
404 case "STATION": {
405 const s = this.station(a[0]);
406 if (a[1] === "DOWN" && !s.up) throw new CommandError("station " + s.name + " is already down");
407 if (a[1] === "UP" && s.up) throw new CommandError("station " + s.name + " is not down");
408 s.up = a[1] === "UP";
409 if (s.up) this.start(s, this.now);
410 this.emit("STATION", s.name, a[1]);
411 break;
412 }
413 case "COMP":
414 case "TRANSFER": {
415 const l = this.line(a[0]);
416 this.modifiable(l);
417 if (cmd === "COMP") {
418 if (l.reason !== null) throw new CommandError("#" + l.id + " is already comped");
419 const value = amount(l);
420 this.emit("COMP", "#" + l.id, l.item, money(value), a[1]);
421 this.comps++;
422 this.compAmount += value;
423 l.reason = a[1];
424 } else {
425 const seat = Number(a[1]);
426 if (seat > l.check.size) throw new CommandError("seat " + seat + " out of range for table " + l.check.table.number);
427 if (seat === l.seat) throw new CommandError("#" + l.id + " is already on seat " + seat);
428 this.emit("TRANSFER", "#" + l.id, "seat", l.seat, "->", seat);
429 l.seat = seat;
430 }
431 break;
432 }
433 case "DISCOUNT": {
434 const c = this.check(a[0]);
435 this.unlocked(c);
436 c.discount = Number(a[1]);
437 this.emit("DISCOUNT", "table", c.table.number, c.discount + "%");
438 break;
439 }
440 case "BILL":
441 this.bill(this.check(a[0]));
442 break;
443 case "SPLIT": {
444 const c = this.check(a[0]);
445 const amounts = this.amounts(c);
446 if (amounts.due === 0) throw new CommandError("nothing to split for table " + a[0]);
447 if (a[1] === "EVEN") {
448 const count = Number(a[2]);
449 if (count > c.size) throw new CommandError("cannot split table " + a[0] + " " + count + " ways");
450 this.emit("SPLIT", "table", c.table.number, "EVEN", count, "due", money(amounts.due));
451 const q = Math.floor(amounts.due / count);
452 const r = amounts.due % count;
453 for (let i = 1; i <= count; i++) this.emit("PAYER", i, money(q + Number(i <= r)));
454 } else {
455 if (c.paid > 0) throw new CommandError("table " + a[0] + " has payments");
456 const values = this.seats(c);
457 const discounts = this.allocate(values, c.discount);
458 const after = new Map([...values].map(([seat, v]): [number, number] => [seat, v - discounts.get(seat)!]));
459 const services = this.allocate(after, c.size >= 6 ? 18 : 0);
460 const taxes = this.allocate(after, 8);
461 this.emit("SPLIT", "table", c.table.number, "BYSEAT", "total", money(amounts.total));
462 for (const [seat, subtotal] of values) this.emit("SEAT", seat, "subtotal", money(subtotal),
463 "discount", money(discounts.get(seat)!), "service", money(services.get(seat)!),
464 "tax", money(taxes.get(seat)!), "total", money(after.get(seat)! + services.get(seat)! + taxes.get(seat)!));
465 }
466 break;
467 }
468 case "PAY": {
469 const c = this.check(a[0]);
470 if (!["CASH", "CARD"].includes(a[1])) throw new CommandError("unknown method " + a[1]);
471 this.unserved(c);
472 const due = this.amounts(c).due;
473 if (due === 0) throw new CommandError("nothing to pay for table " + a[0]);
474 const value = cents(a[2]);
475 const applied = Math.min(value, due);
476 c.paid += applied;
477 this.payments[a[1]] += applied;
478 if (value < due) this.emit("PAY", "table", c.table.number, a[1], money(value), "remaining", money(due - value));
479 else {
480 this.emit("PAY", "table", c.table.number, a[1], money(value), a[1] === "CASH" ? "change" : "tip", money(value - due));
481 if (a[1] === "CARD") {
482 this.tips += value - due;
483 c.server.tips += value - due;
484 }
485 this.close(c);
486 }
487 break;
488 }
489 case "CLOSE": {
490 const c = this.check(a[0]);
491 this.unserved(c);
492 const due = this.amounts(c).due;
493 if (due !== 0) throw new CommandError("table " + a[0] + " still owes " + money(due));
494 this.close(c);
495 break;
496 }
497 case "STATUS": {
498 const c = this.check(a[0]);
499 this.emit("STATUS", "table", c.table.number, c.party, "guests", c.size, "since", clock(c.seated));
500 for (const l of c.lines) this.emit("#" + l.id, "seat", l.seat, l.item, "x" + l.quantity, l.course, l.state,
501 ...(l.state === "COOKING" ? ["until", clock(l.ready)] : []), ...(l.reason === null ? [] : ["COMP", l.reason]));
502 break;
503 }
504 case "KITCHEN":
505 this.emit("KITCHEN", clock(this.now));
506 for (const s of this.stations.values()) {
507 const cooking = s.cooking.toSorted((a, b) => a.ready - b.ready || a.id - b.id).map(l => "#" + l.id + "@" + clock(l.ready));
508 this.emit(s.name, s.up ? "up" : "down", "cooking", cooking.length ? cooking.join(" ") : "-",
509 "queue", s.queue.length ? s.queue.map(l => "#" + l.id).join(" ") : "-");
510 }
511 break;
512 case "FLOOR": {
513 const tables = [...this.tables.values()].sort((a, b) => a.number - b.number);
514 const free = tables.filter(t => t.check === null).length;
515 this.emit("FLOOR", clock(this.now), "free", free, "seated", tables.length - free, "waiting", this.waiting.length);
516 for (const t of tables) {
517 const c = t.check;
518 if (!c) this.emit("table", t.number, "seats", t.seats, "free");
519 else this.emit("table", t.number, "seats", t.seats, c.party, "guests", c.size, "server", c.server.name,
520 "since", clock(c.seated), "items", active(c).length, "total", money(this.amounts(c).total), ...(c.paid > 0 ? ["paid", money(c.paid)] : []));
521 }
522 this.waiting.forEach((w, i) => this.emit("WAITING", i + 1, w.party, w.size, "since", clock(w.joined)));
523 break;
524 }
525 case "REPORT":
526 this.report();
527 break;
528 }
529 }
