1const std = @import("std");
2const Text = []const u8;
3const I = i64;
4fn eq(a: Text, b: Text) bool {
5 return std.mem.eql(u8, a, b);
6}
7fn num(a: Text) I {
8 return std.fmt.parseInt(I, a, 10) catch unreachable;
9}
10fn ix(a: I) usize {
11 return @intCast(a);
12}
13fn powerTwo(a: I) bool {
14 return a > 0 and (a & (a - 1)) == 0;
15}
16fn sizeClass(a: I) I {
17 var value = a;
18 var result: I = 0;
19 while (value >= 2) : (result += 1) value = @divTrunc(value, 2);
20 return result;
21}
22const Host = struct {
23 gpa: std.mem.Allocator,
24 out: *std.Io.Writer,
25 fn text(h: Host, comptime f: Text, args: anytype) !Text {
26 return std.fmt.allocPrint(h.gpa, f, args);
27 }
28 fn line(h: Host, comptime f: Text, args: anytype) !void {
29 try h.out.print(f ++ "\n", args);
30 }
31};
32const Block = struct { address: I, size: I, owner: ?usize = null, age: I = 0 };
33const Identity = struct {
34 name: Text,
35 state: u8 = 0,
36 requested: I = 0,
37 contents: []I = &.{},
38 root: bool = false,
39 history: std.ArrayList(Text) = .empty,
40};
41const Link = struct { source: usize, target: usize };
42const Freed = struct { address: I, size: I, merges: I };
43const Heap = struct {
44 h: Host,
45 size: I = 0,
46 mode: Text = "FIRST",
47 blocks: std.ArrayList(Block) = .empty,
48 identities: std.ArrayList(Identity) = .empty,
49 links: std.ArrayList(Link) = .empty,
50 serial: I = 0,
51 rover: I = 0,
52 peak: I = 0,
53 counters: [6]I = @splat(0),
54 fn init(h: Host, size: I, mode: Text) !Heap {
55 var result: Heap = .{ .h = h, .size = size, .mode = mode };
56 if (size > 0) try result.blocks.append(h.gpa, result.freeBlock(0, size));
57 return result;
58 }
59 fn identity(s: *Heap, name: Text) !usize {
60 for (s.identities.items, 0..) |id, j| if (eq(id.name, name)) return j;
61 try s.identities.append(s.h.gpa, .{ .name = name });
62 return s.identities.items.len - 1;
63 }
64 fn at(s: *Heap, address: I) ?usize {
65 for (s.blocks.items, 0..) |block, j| if (block.address == address) return j;
66 return null;
67 }
68 fn owned(s: *Heap, owner: usize) usize {
69 for (s.blocks.items, 0..) |block, j| if (block.owner == owner) return j;
70 unreachable;
71 }
72 fn required(s: *Heap, requested: I) I {
73 const needed = @divTrunc(requested + 7, 8) * 8 + 8;
74 if (!eq(s.mode, "BUDDY")) return needed;
75 var result: I = 32;
76 while (result < needed) result *= 2;
77 return result;
78 }
79 fn freeBlock(s: *Heap, address: I, size: I) Block {
80 s.serial += 1;
81 return .{ .address = address, .size = size, .age = s.serial };
82 }
83 fn usedBytes(s: *Heap) I {
84 var total: I = 0;
85 for (s.blocks.items) |block| {
86 if (block.owner != null) total += block.size;
87 }
88 return total;
89 }
90 fn fragmentation(s: *Heap) I {
91 var total: I = 0;
92 var largest: I = 0;
93 for (s.blocks.items) |block| {
94 if (block.owner == null) {
95 total += block.size;
96 largest = @max(largest, block.size);
97 }
98 }
99 return if (total == 0) 0 else @divTrunc((total - largest) * 100, total);
100 }
101 fn allocate(s: *Heap, owner: usize, needed: I, alignment: I) !?I {
102 var chosen: ?usize = null;
103 var placement: I = 0;
104 var start: usize = 0;
105 if (eq(s.mode, "NEXT")) {
106 for (s.blocks.items, 0..) |block, j| if (block.address <= s.rover and s.rover < block.address + block.size) {
107 start = j;
108 break;
109 };
110 }
111 for (0..s.blocks.items.len) |turn| {
112 const candidate = if (eq(s.mode, "NEXT")) (start + turn) % s.blocks.items.len else turn;
113 const block = s.blocks.items[candidate];
114 if (block.owner != null) continue;
115 var location = @divTrunc(block.address + 8 + alignment - 1, alignment) * alignment - 8;
116 if (location > block.address and location - block.address < 16) location += alignment;
117 if (location + needed > block.address + block.size) continue;
118 var better = chosen == null;
119 if (chosen) |previous_index| {
120 const previous = s.blocks.items[previous_index];
121 if (eq(s.mode, "BEST") or eq(s.mode, "BUDDY")) {
122 better = block.size < previous.size;
123 } else if (eq(s.mode, "WORST")) {
124 better = block.size > previous.size;
125 } else if (eq(s.mode, "SEGFIT")) {
126 better = sizeClass(block.size) < sizeClass(previous.size) or
127 (sizeClass(block.size) == sizeClass(previous.size) and block.age > previous.age);
128 }
129 }
130 if (better) {
131 chosen = candidate;
132 placement = location;
133 }
134 if (eq(s.mode, "FIRST") or eq(s.mode, "NEXT")) break;
135 }
136 var offset = chosen orelse return null;
137 const target = s.blocks.orderedRemove(offset);
138 if (eq(s.mode, "BUDDY")) {
139 var remaining = target.size;
140 while (remaining > needed) {
141 remaining = @divTrunc(remaining, 2);
142 const upper = s.freeBlock(target.address + remaining, remaining);
143 try s.blocks.insert(s.h.gpa, offset, upper);
144 }
145 try s.blocks.insert(s.h.gpa, offset, .{ .address = target.address, .size = needed, .owner = owner });
146 return target.address;
147 }
148 if (placement > target.address) {
149 const leading = s.freeBlock(target.address, placement - target.address);
150 try s.blocks.insert(s.h.gpa, offset, leading);
151 offset += 1;
152 }
153 const available = target.address + target.size - placement;
154 const used_size = if (available - needed >= 16) needed else available;
155 try s.blocks.insert(s.h.gpa, offset, .{ .address = placement, .size = used_size, .owner = owner });
156 if (available - needed >= 16) {
157 const trailing = s.freeBlock(placement + needed, available - needed);
158 try s.blocks.insert(s.h.gpa, offset + 1, trailing);
159 }
160 if (eq(s.mode, "NEXT")) s.rover = @mod(placement + used_size, s.size);
161 return placement;
162 }
163 fn release(s: *Heap, address: I) Freed {
164 var at_index = s.at(address).?;
165 var merges: I = 0;
166 s.blocks.items[at_index].owner = null;
167 if (eq(s.mode, "BUDDY")) {
168 while (s.blocks.items[at_index].size < s.size) {
169 const block = s.blocks.items[at_index];
170 const buddy = if (@mod(block.address, 2 * block.size) == 0) block.address + block.size else block.address - block.size;
171 const partner = s.at(buddy) orelse break;
172 if (s.blocks.items[partner].owner != null or s.blocks.items[partner].size != block.size) break;
173 const low = @min(at_index, partner);
174 _ = s.blocks.orderedRemove(@max(at_index, partner));
175 s.blocks.items[low].address = @min(block.address, buddy);
176 s.blocks.items[low].size = block.size * 2;
177 at_index = low;
178 merges += 1;
179 }
180 } else {
181 if (at_index + 1 < s.blocks.items.len and s.blocks.items[at_index + 1].owner == null) {
182 s.blocks.items[at_index].size += s.blocks.items[at_index + 1].size;
183 _ = s.blocks.orderedRemove(at_index + 1);
184 merges += 1;
185 }
186 if (at_index > 0 and s.blocks.items[at_index - 1].owner == null) {
187 s.blocks.items[at_index - 1].size += s.blocks.items[at_index].size;
188 _ = s.blocks.orderedRemove(at_index);
189 at_index -= 1;
190 merges += 1;
191 }
192 }
193 const block = s.blocks.items[at_index];
194 s.blocks.items[at_index] = s.freeBlock(block.address, block.size);
195 return .{ .address = block.address, .size = block.size, .merges = merges };
196 }
197 fn live(s: *Heap, name: Text) !?usize {
198 const owner = try s.identity(name);
199 const state = s.identities.items[owner].state;
200 if (state == 1) return owner;
201 try s.h.line("ERROR {s} {s}", .{ if (state == 2) @as(Text, "USE AFTER FREE") else "UNKNOWN ID", name });
202 return null;
203 }
204 fn remember(s: *Heap, owner: usize, line: Text) !void {
205 try s.identities.items[owner].history.append(s.h.gpa, line);
206 try s.h.line("{s}", .{line});
207 }
208 fn forget(s: *Heap, owner: usize) void {
209 s.identities.items[owner].state = 2;
210 s.identities.items[owner].root = false;
211 var j: usize = 0;
212 while (j < s.links.items.len) {
213 const edge = s.links.items[j];
214 if (edge.source == owner or edge.target == owner) _ = s.links.orderedRemove(j) else j += 1;
215 }
216 }
217 fn freeId(s: *Heap, owner: usize, label: Text) !void {
218 const id = s.identities.items[owner];
219 if (id.state != 1) return s.h.line("ERROR {s} {s}", .{ if (id.state == 2) @as(Text, "DOUBLE FREE") else "UNKNOWN ID", id.name });
220 const block = s.blocks.items[s.owned(owner)];
221 const result = s.release(block.address);
222 s.forget(owner);
223 if (eq(label, "FREE")) s.counters[1] += 1;
224 try s.remember(owner, try s.h.text("{s} {s} block={d} merges={d} free={d}+{d}", .{ label, id.name, block.size, result.merges, result.address, result.size }));
225 }
226 fn reachable(s: *Heap) ![]bool {
227 const marked = try s.h.gpa.alloc(bool, s.identities.items.len);
228 for (s.identities.items, 0..) |id, j| marked[j] = id.state == 1 and id.root;
229 var changed = true;
230 while (changed) {
231 changed = false;
232 for (s.links.items) |edge| {
233 if (marked[edge.source] and !marked[edge.target]) {
234 marked[edge.target] = true;
235 changed = true;
236 }
237 }
238 }
239 return marked;
240 }
241 fn tree(s: *Heap, address: I, length: I, depth: usize) anyerror!void {
242 try s.h.out.splatByteAll(' ', depth * 2);
243 try s.h.out.print("{d}+{d}", .{ address, length });
244 if (s.at(address)) |j| {
245 if (s.blocks.items[j].size == length) {
246 if (s.blocks.items[j].owner) |owner| {
247 try s.h.line(" USED {s}", .{s.identities.items[owner].name});
248 } else {
249 try s.h.line(" FREE", .{});
250 }
251 return;
252 }
253 }
254 try s.h.line(" split", .{});
255 try s.tree(address, @divTrunc(length, 2), depth + 1);
256 try s.tree(address + @divTrunc(length, 2), @divTrunc(length, 2), depth + 1);
257 }
258 fn allocation(s: *Heap, w: []const Text) !void {
259 const op = w[0];
260 var alignment: I = 8;
261 if (eq(op, "MEMALIGN")) {
262 if (eq(s.mode, "BUDDY")) return s.h.line("ERROR UNSUPPORTED IN BUDDY MODE", .{});
263 alignment = num(w[2]);
264 if (alignment < 8 or alignment > 4096 or !powerTwo(alignment)) return s.h.line("ERROR BAD ALIGNMENT", .{});
265 }
266 var requested = num(w[if (eq(op, "MEMALIGN")) 3 else 2]);
267 if (eq(op, "CALLOC")) requested *= num(w[3]);
268 const owner = try s.identity(w[1]);
269 const state = s.identities.items[owner].state;
270 if (eq(op, "REALLOC") and requested == 0) return s.freeId(owner, "FREE");
271 if (requested == 0) return s.h.line("ERROR BAD SIZE", .{});
272 if (eq(op, "REALLOC") and state == 2) return s.h.line("ERROR USE AFTER FREE {s}", .{w[1]});
273 const reallocating = eq(op, "REALLOC") and state == 1;
274 if (!reallocating and state == 1) return s.h.line("ERROR DUPLICATE ID {s}", .{w[1]});
275 const needed = s.required(requested);
276 var address: ?I = null;
277 var original: I = -1;
278 var kind: Text = "kept";
279 const old_contents = s.identities.items[owner].contents;
280 if (reallocating) {
281 const at_index = s.owned(owner);
282 const old = s.blocks.items[at_index];
283 original = old.address;
284 address = original;
285 if (needed <= old.size) {
286 if (eq(s.mode, "BUDDY")) {
287 while (s.blocks.items[at_index].size > needed) {
288 s.blocks.items[at_index].size = @divTrunc(s.blocks.items[at_index].size, 2);
289 const half = s.blocks.items[at_index].size;
290 const upper = s.freeBlock(original + half, half);
291 try s.blocks.insert(s.h.gpa, at_index + 1, upper);
292 kind = "shrunk";
293 }
294 } else if (old.size - needed >= 16) {
295 s.blocks.items[at_index].size = needed;
296 try s.blocks.insert(s.h.gpa, at_index + 1, .{ .address = original + needed, .size = old.size - needed });
297 _ = s.release(original + needed);
298 kind = "shrunk";
299 }
300 } else if (!eq(s.mode, "BUDDY") and at_index + 1 < s.blocks.items.len and s.blocks.items[at_index + 1].owner == null and old.size + s.blocks.items[at_index + 1].size >= needed) {
301 const total = old.size + s.blocks.orderedRemove(at_index + 1).size;
302 s.blocks.items[at_index].size = if (total - needed >= 16) needed else total;
303 if (total - needed >= 16) {
304 const trailing = s.freeBlock(original + needed, total - needed);
305 try s.blocks.insert(s.h.gpa, at_index + 1, trailing);
306 }
307 kind = "grown";
308 } else {
309 var can_grow = eq(s.mode, "BUDDY");
310 var length = old.size;
311 while (can_grow and length < needed) {
312 const partner = s.at(original + length);
313 can_grow = @mod(original, 2 * length) == 0 and partner != null and s.blocks.items[partner.?].owner == null and s.blocks.items[partner.?].size == length;
314 length *= 2;
315 }
316 if (can_grow) {
317 length = old.size;
318 while (length < needed) : (length *= 2) _ = s.blocks.orderedRemove(s.at(original + length).?);
319 s.blocks.items[at_index].size = needed;
320 kind = "grown";
321 } else {
322 address = try s.allocate(owner, needed, 8);
323 if (address != null) {
324 _ = s.release(original);
325 kind = "moved";
326 }
327 }
328 }
329 } else address = try s.allocate(owner, needed, alignment);
330 const final_address = address orelse {
331 s.counters[4] += 1;
332 return s.h.line("ERROR OUT OF MEMORY {s}", .{w[1]});
333 };
334 const id = &s.identities.items[owner];
335 id.state = 1;
336 id.requested = requested;
337 id.contents = try s.h.gpa.alloc(I, ix(requested));
338 @memset(id.contents, if (eq(op, "CALLOC")) @as(I, 0) else -1);
339 if (reallocating) {
340 @memcpy(id.contents[0..@min(old_contents.len, id.contents.len)], old_contents[0..@min(old_contents.len, id.contents.len)]);
341 s.counters[2] += 1;
342 if (eq(kind, "moved")) s.counters[3] += 1;
343 } else s.counters[0] += 1;
344 const block = s.blocks.items[s.at(final_address).?];
345 const label = if (eq(op, "REALLOC") and !reallocating) "MALLOC" else op;
346 const suffix = if (!reallocating) "" else if (eq(kind, "moved")) try s.h.text(" moved from={d}", .{original + 8}) else try s.h.text(" {s}", .{kind});
347 try s.remember(owner, try s.h.text("{s} {s} addr={d} block={d}{s}", .{ label, w[1], final_address + 8, block.size, suffix }));
348 }
349 fn bench(s: *Heap, w: []const Text) !void {
350 const size = num(w[1]);
351 const seed = num(w[2]);
352 const steps = num(w[3]);
353 const maxreq = num(w[4]);
354 if (size < 32 or size > 65536 or !powerTwo(size)) return s.h.line("ERROR BAD HEAP SIZE", .{});
355 try s.h.line("BENCH heap={d} seed={d} ops={d} maxreq={d}", .{ size, seed, steps, maxreq });
356 for ([_]Text{ "FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY" }) |mode| {
357 var experiment = try Heap.init(s.h, size, mode);
358 var random = seed;
359 var active: std.ArrayList(I) = .empty;
360 var ok: I = 0;
361 var failures: I = 0;
362 for (0..ix(steps)) |_| {
363 random = @mod(48271 * random, 2147483647);
364 const allocating = active.items.len == 0 or @mod(random, 3) != 0;
365 random = @mod(48271 * random, 2147483647);
366 if (allocating) {
367 const requested = 1 + @mod(random, maxreq);
368 if (try experiment.allocate(0, experiment.required(requested), 8)) |address| {
369 ok += 1;
370 try active.append(s.h.gpa, address);
371 } else failures += 1;
372 } else {
373 const chosen = ix(@mod(random, @as(I, @intCast(active.items.len))));
374 _ = experiment.release(active.orderedRemove(chosen));
375 }
376 experiment.peak = @max(experiment.peak, experiment.usedBytes());
377 }
378 try s.h.line("BENCH {s} ok={d} fail={d} peak={d} used={d} frag={d} blocks={d}", .{ mode, ok, failures, experiment.peak, experiment.usedBytes(), experiment.fragmentation(), experiment.blocks.items.len });
379 }
380 }
381 fn command(s: *Heap, w: []const Text) !void {
382 const op = w[0];
383 if (eq(op, "HEAP")) {
384 const size = num(w[1]);
385 if (size < 32 or size > 65536 or @mod(size, 8) != 0 or (eq(w[2], "BUDDY") and !powerTwo(size))) return s.h.line("ERROR BAD HEAP SIZE", .{});
386 s.* = try Heap.init(s.h, size, w[2]);
387 return s.h.line("HEAP size={d} mode={s}", .{ size, w[2] });
388 }
389 if (eq(op, "BENCH")) return s.bench(w);
390 if (s.size == 0) return s.h.line("ERROR NO HEAP", .{});
391 if (eq(op, "MALLOC") or eq(op, "CALLOC") or eq(op, "MEMALIGN") or eq(op, "REALLOC")) return s.allocation(w);
392 if (eq(op, "FREE")) return s.freeId(try s.identity(w[1]), "FREE");
393 if (eq(op, "FILL") or eq(op, "READ") or eq(op, "COPY")) {
394 const owner = (try s.live(w[1])) orelse return;
395 const offset = num(w[2]);
396 const length = num(w[if (eq(op, "COPY")) 5 else 3]);
397 if (offset + length > s.identities.items[owner].requested) return s.h.line("ERROR OUT OF BOUNDS {s}", .{w[1]});
398 if (eq(op, "COPY")) {
399 const source = (try s.live(w[3])) orelse return;
400 const source_offset = num(w[4]);
401 if (source_offset + length > s.identities.items[source].requested) return s.h.line("ERROR OUT OF BOUNDS {s}", .{w[3]});
402 const copied = try s.h.gpa.dupe(I, s.identities.items[source].contents[ix(source_offset)..ix(source_offset + length)]);
403 @memcpy(s.identities.items[owner].contents[ix(offset)..ix(offset + length)], copied);
404 return s.h.line("COPY {s} {d}+{d} from {s} {d}", .{ w[1], offset, length, w[3], source_offset });
405 }
406 if (eq(op, "FILL")) {
407 @memset(s.identities.items[owner].contents[ix(offset)..ix(offset + length)], num(w[4]));
408 return s.h.line("FILL {s} {d}+{d} = {d}", .{ w[1], offset, length, num(w[4]) });
409 }
410 try s.h.out.print("READ {s} {d}+{d} =", .{ w[1], offset, length });
411 for (s.identities.items[owner].contents[ix(offset)..ix(offset + length)]) |value| {
412 if (value < 0) try s.h.out.writeAll(" ?") else try s.h.out.print(" {d}", .{value});
413 }
414 return s.h.line("", .{});
415 }
416 if (eq(op, "ROOT") or eq(op, "UNROOT")) {
417 const owner = (try s.live(w[1])) orelse return;
418 if (eq(op, "UNROOT") and !s.identities.items[owner].root) return s.h.line("ERROR NOT ROOT {s}", .{w[1]});
419 s.identities.items[owner].root = eq(op, "ROOT");
420 return s.h.line("{s} {s}", .{ op, w[1] });
421 }
422 if (eq(op, "LINK") or eq(op, "UNLINK")) {
423 const source = (try s.live(w[1])) orelse return;
424 const target = (try s.live(w[2])) orelse return;
425 var found: ?usize = null;
426 for (s.links.items, 0..) |edge, j| if (edge.source == source and edge.target == target) {
427 found = j;
428 break;
429 };
430 if (eq(op, "UNLINK") and found == null) return s.h.line("ERROR NO LINK {s} {s}", .{ w[1], w[2] });
431 if (eq(op, "LINK") and found == null) {
432 try s.links.append(s.h.gpa, .{ .source = source, .target = target });
433 } else if (eq(op, "UNLINK")) {
434 _ = s.links.orderedRemove(found.?);
435 }
436 return s.h.line("{s} {s} {s}", .{ op, w[1], w[2] });
437 }
438 if (eq(op, "GC")) {
439 const marked = try s.reachable();
440 var victims: std.ArrayList(usize) = .empty;
441 var total: I = 0;
442 for (s.blocks.items) |block| {
443 if (block.owner) |owner| {
444 if (!marked[owner]) {
445 try victims.append(s.h.gpa, owner);
446 total += block.size;
447 }
448 }
449 }
450 try s.h.line("GC collected={d} bytes={d}", .{ victims.items.len, total });
451 for (victims.items) |owner| try s.freeId(owner, "COLLECT");
452 s.counters[5] += @intCast(victims.items.len);
453 return;
454 }
455 if (eq(op, "SBRK")) {
456 const amount = num(w[1]);
457 const valid = if (eq(s.mode, "BUDDY")) amount == s.size and s.size * 2 <= 65536 else amount >= 16 and @mod(amount, 8) == 0 and s.size + amount <= 65536;
458 if (!valid) return s.h.line("ERROR BAD SBRK SIZE", .{});
459 const old = s.size;
460 s.size += amount;
461 try s.blocks.append(s.h.gpa, .{ .address = old, .size = amount });
462 const result = s.release(old);
463 return s.h.line("SBRK heap={d} free={d}+{d}", .{ s.size, result.address, result.size });
464 }
465 if (eq(op, "TRIM")) {
466 const old = s.size;
467 while (s.blocks.items.len > 1) {
468 const last = s.blocks.items[s.blocks.items.len - 1];
469 if (last.owner != null or (eq(s.mode, "BUDDY") and last.size != @divTrunc(s.size, 2))) break;
470 s.size = last.address;
471 _ = s.blocks.orderedRemove(s.blocks.items.len - 1);
472 if (!eq(s.mode, "BUDDY")) break;
473 }
474 if (s.rover >= s.size) s.rover = 0;
475 return s.h.line("TRIM heap={d} released={d}", .{ s.size, old - s.size });
476 }
477 if (eq(op, "COMPACT")) {
478 if (eq(s.mode, "BUDDY")) return s.h.line("ERROR UNSUPPORTED IN BUDDY MODE", .{});
479 var compacted: std.ArrayList(Block) = .empty;
480 var moves: std.ArrayList(Text) = .empty;
481 var address: I = 0;
482 for (s.blocks.items) |block| if (block.owner) |owner| {
483 if (block.address != address) {
484 const line = try s.h.text("MOVE {s} {d} -> {d}", .{ s.identities.items[owner].name, block.address + 8, address + 8 });
485 try moves.append(s.h.gpa, line);
486 try s.identities.items[owner].history.append(s.h.gpa, line);
487 }
488 try compacted.append(s.h.gpa, .{ .address = address, .size = block.size, .owner = owner });
489 address += block.size;
490 };
491 if (address < s.size) {
492 const remainder = s.freeBlock(address, s.size - address);
493 try compacted.append(s.h.gpa, remainder);
494 }
495 s.blocks = compacted;
496 s.rover = 0;
497 s.counters[3] += @intCast(moves.items.len);
498 const free_text = if (address == s.size) "none" else try s.h.text("{d}+{d}", .{ address, s.size - address });
499 try s.h.line("COMPACT moved={d} free={s}", .{ moves.items.len, free_text });
500 for (moves.items) |line| try s.h.line("{s}", .{line});
501 return;
502 }
503 if (eq(op, "WHO")) {
504 const address = num(w[1]);
505 if (address >= s.size) return s.h.line("WHO {d} outside", .{address});
506 for (s.blocks.items) |block| if (block.address <= address and address < block.address + block.size) {
507 const owner = block.owner orelse return s.h.line("WHO {d} free {d}+{d}", .{ address, block.address, block.size });
508 const id = s.identities.items[owner];
509 if (address < block.address + 8) return s.h.line("WHO {d} header {s}", .{ address, id.name });
510 const offset = address - block.address - 8;
511 return s.h.line("WHO {d} {s} {s}+{d}", .{ address, if (offset < id.requested) @as(Text, "data") else "slack", id.name, offset });
512 };
513 }
514 if (eq(op, "DUMP")) {
515 const suffix = if (eq(s.mode, "NEXT")) try s.h.text(" rover={d}", .{s.rover}) else "";
516 try s.h.line("DUMP mode={s} heap={d} blocks={d}{s}", .{ s.mode, s.size, s.blocks.items.len, suffix });
517 for (s.blocks.items) |block| {
518 if (block.owner) |owner| {
519 try s.h.line("{d} {d} USED {s} req={d}", .{
520 block.address, block.size,
521 s.identities.items[owner].name, s.identities.items[owner].requested,
522 });
523 } else {
524 try s.h.line("{d} {d} FREE", .{ block.address, block.size });
525 }
526 }
527 return;
528 }
529 if (eq(op, "MAP")) {
530 const units = try s.h.gpa.alloc(u8, ix(@divTrunc(s.size, 8)));
531 for (s.blocks.items) |block| {
532 const start = ix(@divTrunc(block.address, 8));
533 units[start] = if (block.owner == null) 'F' else 'H';
534 for (1..ix(@divTrunc(block.size, 8))) |j| units[start + j] = if (block.owner == null) '.' else if (@as(I, @intCast((j - 1) * 8)) < s.identities.items[block.owner.?].requested) '#' else '-';
535 }
536 try s.h.line("MAP heap={d}", .{s.size});
537 var start: usize = 0;
538 while (start < units.len) : (start += 64) try s.h.line("{d:0>5} {s}", .{ start * 8, units[start..@min(start + 64, units.len)] });
539 return;
540 }
541 if (eq(op, "FREELIST")) {
542 var ordered: std.ArrayList(usize) = .empty;
543 for (s.blocks.items, 0..) |block, j| if (block.owner == null) {
544 try ordered.append(s.h.gpa, j);
545 };
546 if (eq(s.mode, "SEGFIT")) {
547 for (0..ordered.items.len) |j| {
548 const selected = ordered.items[j];
549 var at_index = j;
550 while (at_index > 0 and s.blocks.items[ordered.items[at_index - 1]].age < s.blocks.items[selected].age) : (at_index -= 1) ordered.items[at_index] = ordered.items[at_index - 1];
551 ordered.items[at_index] = selected;
552 }
553 }
554 try s.h.line("FREELIST count={d}", .{ordered.items.len});
555 for (4..17) |category| {
556 var first = true;
557 for (ordered.items) |j| {
558 const block = s.blocks.items[j];
559 if (sizeClass(block.size) != @as(I, @intCast(category))) continue;
560 if (first) {
561 const lower = @as(I, 1) << @intCast(category);
562 try s.h.out.print("{d}-{d}:", .{ lower, lower * 2 - 1 });
563 first = false;
564 }
565 try s.h.out.print(" {d}+{d}", .{ block.address, block.size });
566 }
567 if (!first) try s.h.line("", .{});
568 }
569 return;
570 }
571 if (eq(op, "TREE")) {
572 if (!eq(s.mode, "BUDDY")) return s.h.line("ERROR REQUIRES BUDDY MODE", .{});
573 try s.h.line("TREE heap={d}", .{s.size});
574 return s.tree(0, s.size, 0);
575 }
576 if (eq(op, "STATS")) {
577 var used_count: I = 0;
578 var free_count: I = 0;
579 var largest: I = 0;
580 var requested: I = 0;
581 for (s.blocks.items) |block| {
582 if (block.owner) |owner| {
583 used_count += 1;
584 requested += s.identities.items[owner].requested;
585 } else {
586 free_count += 1;
587 largest = @max(largest, block.size);
588 }
589 }
590 const used = s.usedBytes();
591 try s.h.line("STATS used={d} used_blocks={d} free={d} free_blocks={d} largest={d} frag={d} requested={d} peak={d}", .{ used, used_count, s.size - used, free_count, largest, s.fragmentation(), requested, s.peak });
592 return s.h.line("OPS allocs={d} frees={d} reallocs={d} moves={d} failures={d} collected={d}", .{ s.counters[0], s.counters[1], s.counters[2], s.counters[3], s.counters[4], s.counters[5] });
593 }
594 if (eq(op, "LEAKS")) {
595 const marked = try s.reachable();
596 var names: std.ArrayList(Text) = .empty;
597 var requested: I = 0;
598 var lost: I = 0;
599 for (s.identities.items, 0..) |id, j| {
600 if (id.state == 1) {
601 try names.append(s.h.gpa, id.name);
602 requested += id.requested;
603 if (!marked[j]) lost += 1;
604 }
605 }
606 std.mem.sort(Text, names.items, {}, struct {
607 fn less(_: void, a: Text, b: Text) bool {
608 return std.mem.lessThan(u8, a, b);
609 }
610 }.less);
611 try s.h.line("LEAKS count={d} requested={d} lost={d}", .{ names.items.len, requested, lost });
612 for (names.items) |name| {
613 const owner = try s.identity(name);
614 const block = s.blocks.items[s.owned(owner)];
615 try s.h.line("LEAK {s} addr={d} block={d} req={d} {s}", .{ name, block.address + 8, block.size, s.identities.items[owner].requested, if (marked[owner]) @as(Text, "reachable") else "lost" });
616 }
617 return;
618 }
619 const owner = try s.identity(w[1]);
620 const history = s.identities.items[owner].history.items;
621 try s.h.line("HISTORY {s} events={d}", .{ w[1], history.len });
622 for (history) |line| try s.h.line("{s}", .{line});
623 }
624};
625fn readLine(r: *std.Io.Reader, a: std.mem.Allocator) !?Text {
626 var b: std.Io.Writer.Allocating = .init(a);
627 defer b.deinit();
628 _ = try r.streamDelimiterEnding(&b.writer, '\n');
629 if (r.bufferedLen() == 0 and b.written().len == 0) return null;
630 if (r.bufferedLen() > 0) r.toss(1);
631 return b.toOwnedSlice();
632}
633pub fn main(init: std.process.Init) !void {
634 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
635 defer arena.deinit();
636 const a = arena.allocator();
637 var ib: [4096]u8 = undefined;
638 var ob: [4096]u8 = undefined;
639 var input = std.Io.File.stdin().readerStreaming(init.io, &ib);
640 var output = std.Io.File.stdout().writerStreaming(init.io, &ob);
641 var heap = try Heap.init(.{ .gpa = a, .out = &output.interface }, 0, "FIRST");
642 const count = num(std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r"));
643 for (0..ix(count)) |_| {
644 const line = std.mem.trim(u8, (try readLine(&input.interface, a)).?, "\r");
645 var w: std.ArrayList(Text) = .empty;
646 var it = std.mem.splitScalar(u8, line, ' ');
647 while (it.next()) |word| try w.append(a, word);
648 try heap.command(w.items);
649 heap.peak = @max(heap.peak, heap.usedBytes());
650 }
651 try output.interface.flush();
652}
