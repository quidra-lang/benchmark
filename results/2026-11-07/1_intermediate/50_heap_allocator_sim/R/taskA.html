1block <- function(start, size) {
2 b <- new.env(parent = emptyenv())
3 b$start <- start
4 b$size <- size
5 b$owner <- ""
6 b$requested <- 0
7 b$data <- numeric()
8 b$stamp <- 0
9 b

11used <- function(b) nzchar(b$owner)
12area <- function(b) paste0(b$start, "+", b$size)
13power2 <- function(x) x > 0 && x == 2^floor(log2(x))
14heap <- function(size, mode) {
15 h <- new.env(parent = emptyenv())
16 h$size <- size
17 h$mode <- mode
18 h$rover <- 0
19 h$age <- 0
20 h$blocks <- list(register(h, block(0, size)))
21 h
22}
23register <- function(h, b) {
24 h$age <- h$age + 1
25 b$stamp <- h$age
26 b
27}
28required <- function(h, n) {
29 s <- ceiling(n / 8) * 8 + 8
30 if (h$mode == "BUDDY") s <- max(32, 2^ceiling(log2(s)))
31 s
32}
33used_total <- function(h) sum(vapply(h$blocks, function(b) if (used(b)) b$size else 0, numeric(1)))
34fragmentation <- function(h) {
35 sizes <- vapply(Filter(function(b) !used(b), h$blocks), function(b) b$size, numeric(1))
36 if (!length(sizes)) return(0)
37 floor((sum(sizes) - max(sizes)) * 100 / sum(sizes))
38}
39insert_block <- function(h, i, b) h$blocks <- append(h$blocks, list(b), after = i - 1)
40remove_block <- function(h, i) h$blocks <- h$blocks[-i]
41halve <- function(h, i, s) {
42 b <- h$blocks[[i]]
43 while (b$size > s) {
44 b$size <- b$size / 2
45 insert_block(h, i + 1, register(h, block(b$start + b$size, b$size)))
46 }
47}
48allocate <- function(h, s, al) {
49 ri <- which(vapply(h$blocks, function(b) b$start <= h$rover && h$rover < b$start + b$size, logical(1)))[1]
50 candidates <- list()
51 for (i in seq_along(h$blocks)) {
52 b <- h$blocks[[i]]
53 if (used(b)) next
54 p <- ceiling((b$start + 8) / al) * al - 8
55 if (p > b$start && p - b$start < 16) p <- p + al
56 if (p + s > b$start + b$size) next
57 rank <- switch(h$mode,
58 NEXT = c((i - ri) %% length(h$blocks), 0),
59 BEST = c(b$size, b$start), BUDDY = c(b$size, b$start),
60 WORST = c(-b$size, b$start),
61 SEGFIT = c(floor(log2(b$size)), -b$stamp),
62 c(b$start, 0))
63 candidates[[length(candidates) + 1]] <- c(rank, i, p)
64 }
65 if (!length(candidates)) return(NULL)
66 ranks <- do.call(rbind, candidates)
67 choice <- ranks[order(ranks[, 1], ranks[, 2])[1], ]
68 i <- choice[3]
69 p <- choice[4]
70 if (h$mode == "BUDDY") {
71 halve(h, i, s)
72 b <- h$blocks[[i]]
73 } else {
74 old <- h$blocks[[i]]
75 parts <- list()
76 if (p > old$start) parts <- list(register(h, block(old$start, p - old$start)))
77 b <- block(p, old$start + old$size - p)
78 parts <- c(parts, list(b))
79 if (b$size - s >= 16) {
80 parts <- c(parts, list(register(h, block(p + s, b$size - s))))
81 b$size <- s
82 }
83 remove_block(h, i)
84 h$blocks <- append(h$blocks, parts, after = i - 1)
85 }
86 b$owner <- "@"
87 if (h$mode == "NEXT") h$rover <- (b$start + b$size) %% h$size
88 b
89}
90coalesce <- function(h, i) {
91 merges <- 0
92 if (h$mode == "BUDDY") {
93 while (h$blocks[[i]]$size < h$size) {
94 b <- h$blocks[[i]]
95 a <- if (b$start %% (2 * b$size) == 0) b$start + b$size else b$start - b$size
96 js <- which(vapply(h$blocks, function(x) !used(x) && x$start == a && x$size == b$size, logical(1)))
97 if (!length(js)) break
98 i <- min(i, js[1])
99 h$blocks[[i]]$size <- h$blocks[[i]]$size + h$blocks[[i + 1]]$size
100 remove_block(h, i + 1)
101 merges <- merges + 1
102 }
103 } else {
104 if (i < length(h$blocks) && !used(h$blocks[[i + 1]])) {
105 h$blocks[[i]]$size <- h$blocks[[i]]$size + h$blocks[[i + 1]]$size
106 remove_block(h, i + 1)
107 merges <- merges + 1
108 }
109 if (i > 1 && !used(h$blocks[[i - 1]])) {
110 i <- i - 1
111 h$blocks[[i]]$size <- h$blocks[[i]]$size + h$blocks[[i + 1]]$size
112 remove_block(h, i + 1)
113 merges <- merges + 1
114 }
115 }
116 list(block = register(h, h$blocks[[i]]), merges = merges)
117}
118release <- function(h, b) {
119 b$owner <- ""
120 b$requested <- 0
121 b$data <- numeric()
122 i <- which(vapply(h$blocks, identical, logical(1), b))[1]
123 coalesce(h, i)
124}
125resize <- function(h, b, s) {
126 i <- which(vapply(h$blocks, identical, logical(1), b))[1]
127 if (s <= b$size) {
128 if (h$mode == "BUDDY") {
129 if (s == b$size) return("kept")
130 halve(h, i, s)
131 } else {
132 if (b$size - s < 16) return("kept")
133 insert_block(h, i + 1, block(b$start + s, b$size - s))
134 b$size <- s
135 coalesce(h, i + 1)
136 }
137 return("shrunk")
138 }
139 if (h$mode == "BUDDY") {
140 z <- b$size
141 j <- i
142 while (z < s) {
143 j <- j + 1
144 if (b$start %% (2 * z) != 0 || j > length(h$blocks)) return(NULL)
145 p <- h$blocks[[j]]
146 if (used(p) || p$size != z) return(NULL)
147 z <- 2 * z
148 }
149 h$blocks <- h$blocks[-seq.int(i + 1, j)]
150 b$size <- s
151 return("grown")
152 }
153 if (i < length(h$blocks) && !used(h$blocks[[i + 1]])) {
154 z <- b$size + h$blocks[[i + 1]]$size
155 if (z >= s) {
156 remove_block(h, i + 1)
157 b$size <- z
158 if (z - s >= 16) {
159 b$size <- s
160 insert_block(h, i + 1, register(h, block(b$start + s, z - s)))
161 }
162 return("grown")
163 }
164 }
165 NULL
166}
167allocator <- function(h = NULL) {
168 a <- new.env(parent = emptyenv())
169 a$heap <- h
170 a$live <- list()
171 a$freed <- a$roots <- character()
172 a$refs <- a$histories <- list()
173 a$counts <- setNames(rep(0, 6), c("allocs", "frees", "reallocs", "moves", "failures", "collected"))
174 a$peak <- 0
175 a
176}
177emit <- function(...) cat(paste0(...), "\n", sep = "")
178fail <- function(text) stop(structure(list(message = text), class = c("command_error", "error", "condition")))
179record <- function(a, id, line) {
180 emit(line)
181 a$histories[[id]] <- c(a$histories[[id]], line)

183live <- function(a, id) {
184 if (!is.null(a$live[[id]])) return(a$live[[id]])
185 fail(paste(if (id %in% a$freed) "USE AFTER FREE" else "UNKNOWN ID", id))
186}
187bounds <- function(a, id, off, len) {
188 b <- live(a, id)
189 if (off + len > b$requested) fail(paste("OUT OF BOUNDS", id))
190 b
191}
192reachable <- function(a) {
193 seen <- todo <- a$roots
194 while (length(todo)) {
195 id <- todo[1]
196 todo <- todo[-1]
197 fresh <- setdiff(a$refs[[id]], seen)
198 seen <- c(seen, fresh)
199 todo <- c(todo, fresh)
200 }
201 seen
202}
203malloc_id <- function(a, word, id, n, al = 8, value = -1) {
204 if (n == 0) fail("BAD SIZE")
205 if (!is.null(a$live[[id]])) fail(paste("DUPLICATE ID", id))
206 b <- allocate(a$heap, required(a$heap, n), al)
207 if (is.null(b)) {
208 a$counts["failures"] <- a$counts["failures"] + 1
209 fail(paste("OUT OF MEMORY", id))
210 }
211 b$owner <- id
212 b$requested <- n
213 b$data <- rep(value, n)
214 a$live[[id]] <- b
215 a$freed <- setdiff(a$freed, id)
216 a$counts["allocs"] <- a$counts["allocs"] + 1
217 record(a, id, paste0(word, " ", id, " addr=", b$start + 8, " block=", b$size))
218}
219release_id <- function(a, id) {
220 b <- a$live[[id]]
221 size <- b$size
222 a$live[[id]] <- NULL
223 a$freed <- union(a$freed, id)
224 a$roots <- setdiff(a$roots, id)
225 a$refs[[id]] <- NULL
226 a$refs <- lapply(a$refs, setdiff, id)
227 result <- release(a$heap, b)
228 paste0("block=", size, " merges=", result$merges, " free=", area(result$block))
229}
230free_id <- function(a, id) {
231 if (id %in% a$freed) fail(paste("DOUBLE FREE", id))
232 if (is.null(a$live[[id]])) fail(paste("UNKNOWN ID", id))
233 details <- release_id(a, id)
234 a$counts["frees"] <- a$counts["frees"] + 1
235 record(a, id, paste("FREE", id, details))
236}
237tree <- function(h, start, size, depth = 0) {
238 label <- paste0(strrep(" ", 2 * depth), start, "+", size)
239 leaves <- Filter(function(b) b$start == start && b$size == size, h$blocks)
240 if (length(leaves)) {
241 b <- leaves[[1]]
242 emit(label, if (used(b)) paste0(" USED ", b$owner) else " FREE")
243 } else {
244 emit(label, " split")
245 tree(h, start, size / 2, depth + 1)
246 tree(h, start + size / 2, size / 2, depth + 1)
247 }

249experiment <- function(mode, size, seed, ops, maxreq) {
250 h <- heap(size, mode)
251 x <- seed
252 active <- list()
253 ok <- bad <- peak <- 0
254 for (step in seq_len(ops)) {
255 x <- (48271 * x) %% 2147483647
256 alloc <- !length(active) || x %% 3 != 0
257 x <- (48271 * x) %% 2147483647
258 if (alloc) {
259 b <- allocate(h, required(h, 1 + x %% maxreq), 8)
260 if (is.null(b)) {
261 bad <- bad + 1
262 } else {
263 ok <- ok + 1
264 active <- c(active, list(b))
265 }
266 } else {
267 i <- x %% length(active) + 1
268 release(h, active[[i]])
269 active <- active[-i]
270 }
271 peak <- max(peak, used_total(h))
272 }
273 emit("BENCH ", mode, " ok=", ok, " fail=", bad, " peak=", peak, " used=", used_total(h),
274 " frag=", fragmentation(h), " blocks=", length(h$blocks))
275}
276execute <- function(a, words) {
277 cmd <- words[1]
278 v <- words[-1]
279 num <- function(i) as.numeric(v[i])
280 h <- a$heap
281 if (cmd == "HEAP") {
282 s <- num(1)
283 m <- v[2]
284 if (s %% 8 != 0 || s < 32 || s > 65536 || (m == "BUDDY" && !power2(s))) fail("BAD HEAP SIZE")
285 fresh <- allocator(heap(s, m))
286 for (name in ls(fresh)) a[[name]] <- fresh[[name]]
287 emit("HEAP size=", s, " mode=", m)
288 } else if (cmd == "BENCH") {
289 s <- num(1)
290 if (!power2(s) || s < 32 || s > 65536) fail("BAD HEAP SIZE")
291 emit("BENCH heap=", s, " seed=", num(2), " ops=", num(3), " maxreq=", num(4))
292 for (m in c("FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY")) experiment(m, s, num(2), num(3), num(4))
293 } else if (is.null(h)) {
294 fail("NO HEAP")
295 } else if (cmd %in% c("MALLOC", "CALLOC", "MEMALIGN")) {
296 al <- 8
297 n <- num(2)
298 value <- -1
299 if (cmd == "CALLOC") {
300 n <- num(2) * num(3)
301 value <- 0
302 } else if (cmd == "MEMALIGN") {
303 if (h$mode == "BUDDY") fail("UNSUPPORTED IN BUDDY MODE")
304 al <- num(2)
305 n <- num(3)
306 if (!power2(al) || al < 8 || al > 4096) fail("BAD ALIGNMENT")
307 }
308 malloc_id(a, cmd, v[1], n, al, value)
309 } else if (cmd == "FREE") {
310 free_id(a, v[1])
311 } else if (cmd == "REALLOC") {
312 id <- v[1]
313 n <- num(2)
314 if (n == 0) {
315 free_id(a, id)
316 } else if (id %in% a$freed) {
317 fail(paste("USE AFTER FREE", id))
318 } else if (is.null(a$live[[id]])) {
319 malloc_id(a, "MALLOC", id, n)
320 } else {
321 b <- a$live[[id]]
322 data <- rep(-1, n)
323 k <- min(n, b$requested)
324 if (k > 0) data[seq_len(k)] <- b$data[seq_len(k)]
325 origin <- b$start + 8
326 kind <- resize(h, b, required(h, n))
327 if (is.null(kind)) {
328 moved <- allocate(h, required(h, n), 8)
329 if (is.null(moved)) {
330 a$counts["failures"] <- a$counts["failures"] + 1
331 fail(paste("OUT OF MEMORY", id))
332 }
333 moved$owner <- id
334 a$live[[id]] <- moved
335 release(h, b)
336 b <- moved
337 a$counts["moves"] <- a$counts["moves"] + 1
338 kind <- paste0("moved from=", origin)
339 }
340 b$requested <- n
341 b$data <- data
342 a$counts["reallocs"] <- a$counts["reallocs"] + 1
343 record(a, id, paste0("REALLOC ", id, " addr=", b$start + 8, " block=", b$size, " ", kind))
344 }
345 } else if (cmd %in% c("FILL", "READ")) {
346 id <- v[1]
347 off <- num(2)
348 len <- num(3)
349 b <- bounds(a, id, off, len)
350 ix <- seq.int(off + 1, off + len)
351 if (cmd == "FILL") {
352 b$data[ix] <- num(4)
353 emit("FILL ", id, " ", off, "+", len, " = ", num(4))
354 } else {
355 text <- paste(ifelse(b$data[ix] == -1, "?", b$data[ix]), collapse = " ")
356 emit("READ ", id, " ", off, "+", len, " = ", text)
357 }
358 } else if (cmd == "COPY") {
359 d <- bounds(a, v[1], num(2), num(5))
360 s <- bounds(a, v[3], num(4), num(5))
361 values <- s$data[seq.int(num(4) + 1, num(4) + num(5))]
362 d$data[seq.int(num(2) + 1, num(2) + num(5))] <- values
363 emit("COPY ", v[1], " ", num(2), "+", num(5), " from ", v[3], " ", num(4))
364 } else if (cmd %in% c("ROOT", "UNROOT")) {
365 id <- v[1]
366 live(a, id)
367 if (cmd == "ROOT") {
368 a$roots <- union(a$roots, id)
369 } else {
370 if (!(id %in% a$roots)) fail(paste("NOT ROOT", id))
371 a$roots <- setdiff(a$roots, id)
372 }
373 emit(cmd, " ", id)
374 } else if (cmd %in% c("LINK", "UNLINK")) {
375 live(a, v[1])
376 live(a, v[2])
377 if (cmd == "LINK") {
378 a$refs[[v[1]]] <- union(a$refs[[v[1]]], v[2])
379 } else {
380 if (!(v[2] %in% a$refs[[v[1]]])) fail(paste("NO LINK", v[1], v[2]))
381 a$refs[[v[1]]] <- setdiff(a$refs[[v[1]]], v[2])
382 }
383 emit(cmd, " ", v[1], " ", v[2])
384 } else if (cmd == "GC") {
385 seen <- reachable(a)
386 garbage <- Filter(function(b) !(b$owner %in% seen), h$blocks)
387 garbage <- Filter(used, garbage)
388 bytes <- sum(vapply(garbage, function(b) b$size, numeric(1)))
389 emit("GC collected=", length(garbage), " bytes=", bytes)
390 for (b in garbage) {
391 id <- b$owner
392 record(a, id, paste("COLLECT", id, release_id(a, id)))
393 }
394 a$counts["collected"] <- a$counts["collected"] + length(garbage)
395 } else if (cmd == "SBRK") {
396 k <- num(1)
397 valid <- if (h$mode == "BUDDY") k == h$size && 2 * h$size <= 65536 else k %% 8 == 0 && k >= 16 && h$size + k <= 65536
398 if (!valid) fail("BAD SBRK SIZE")
399 h$blocks <- c(h$blocks, list(block(h$size, k)))
400 h$size <- h$size + k
401 result <- coalesce(h, length(h$blocks))
402 emit("SBRK heap=", h$size, " free=", area(result$block))
403 } else if (cmd == "TRIM") {
404 old <- h$size
405 if (h$mode == "BUDDY") {
406 repeat {
407 b <- h$blocks[[length(h$blocks)]]
408 if (used(b) || 2 * b$size != h$size) break
409 h$size <- h$size - b$size
410 remove_block(h, length(h$blocks))
411 }
412 } else if (length(h$blocks) >= 2 && !used(h$blocks[[length(h$blocks)]])) {
413 h$size <- h$blocks[[length(h$blocks)]]$start
414 remove_block(h, length(h$blocks))
415 }
416 if (h$rover >= h$size) h$rover <- 0
417 emit("TRIM heap=", h$size, " released=", old - h$size)
418 } else if (cmd == "COMPACT") {
419 if (h$mode == "BUDDY") fail("UNSUPPORTED IN BUDDY MODE")
420 packed <- Filter(used, h$blocks)
421 moves <- list()
422 pos <- 0
423 for (b in packed) {
424 if (b$start != pos) {
425 moves <- c(moves, list(list(b = b, origin = b$start + 8)))
426 b$start <- pos
427 }
428 pos <- pos + b$size
429 }
430 free <- if (pos == h$size) "none" else paste0(pos, "+", h$size - pos)
431 if (pos < h$size) packed <- c(packed, list(register(h, block(pos, h$size - pos))))
432 h$blocks <- packed
433 h$rover <- 0
434 a$counts["moves"] <- a$counts["moves"] + length(moves)
435 emit("COMPACT moved=", length(moves), " free=", free)
436 for (m in moves) record(a, m$b$owner, paste0("MOVE ", m$b$owner, " ", m$origin, " -> ", m$b$start + 8))
437 } else if (cmd == "WHO") {
438 p <- num(1)
439 if (p >= h$size) {
440 emit("WHO ", p, " outside")
441 } else {
442 b <- Filter(function(b) b$start <= p && p < b$start + b$size, h$blocks)[[1]]
443 if (!used(b)) {
444 emit("WHO ", p, " free ", area(b))
445 } else if (p < b$start + 8) {
446 emit("WHO ", p, " header ", b$owner)
447 } else {
448 d <- p - b$start - 8
449 emit("WHO ", p, if (d < b$requested) " data " else " slack ", b$owner, "+", d)
450 }
451 }
452 } else if (cmd == "DUMP") {
453 tail <- if (h$mode == "NEXT") paste0(" rover=", h$rover) else ""
454 emit("DUMP mode=", h$mode, " heap=", h$size, " blocks=", length(h$blocks), tail)
455 for (b in h$blocks) emit(b$start, " ", b$size, if (used(b)) paste0(" USED ", b$owner, " req=", b$requested) else " FREE")
456 } else if (cmd == "MAP") {
457 emit("MAP heap=", h$size)
458 chars <- character()
459 for (b in h$blocks) {
460 rest <- b$size / 8 - 1
461 tail <- if (used(b)) ifelse(8 * (seq_len(rest) - 1) < b$requested, "#", "-") else rep(".", rest)
462 chars <- c(chars, if (used(b)) "H" else "F", tail)
463 }
464 for (first in seq.int(1, length(chars), by = 64)) emit(sprintf("%05d", (first - 1) * 8), " ", paste(chars[first:min(first + 63, length(chars))], collapse = ""))
465 } else if (cmd == "FREELIST") {
466 free <- Filter(function(b) !used(b), h$blocks)
467 if (h$mode == "SEGFIT" && length(free)) free <- free[order(vapply(free, function(b) -b$stamp, numeric(1)))]
468 emit("FREELIST count=", length(free))
469 classes <- sort(unique(vapply(free, function(b) floor(log2(b$size)), numeric(1))))
470 for (k in classes) {
471 members <- Filter(function(b) floor(log2(b$size)) == k, free)
472 emit(2^k, "-", 2^(k + 1) - 1, ": ", paste(vapply(members, area, character(1)), collapse = " "))
473 }
474 } else if (cmd == "TREE") {
475 if (h$mode != "BUDDY") fail("REQUIRES BUDDY MODE")
476 emit("TREE heap=", h$size)
477 tree(h, 0, h$size)
478 } else if (cmd == "STATS") {
479 u <- Filter(used, h$blocks)
480 f <- Filter(function(b) !used(b), h$blocks)
481 sizes <- vapply(f, function(b) b$size, numeric(1))
482 req <- sum(vapply(u, function(b) b$requested, numeric(1)))
483 emit("STATS used=", used_total(h), " used_blocks=", length(u), " free=", sum(sizes), " free_blocks=", length(f),
484 " largest=", max(c(0, sizes)), " frag=", fragmentation(h), " requested=", req, " peak=", a$peak)
485 emit("OPS ", paste(paste0(names(a$counts), "=", a$counts), collapse = " "))
486 } else if (cmd == "LEAKS") {
487 seen <- reachable(a)
488 ids <- sort(names(a$live), method = "radix")
489 req <- sum(vapply(a$live, function(b) b$requested, numeric(1)))
490 emit("LEAKS count=", length(ids), " requested=", req, " lost=", sum(!(ids %in% seen)))
491 for (id in ids) {
492 b <- a$live[[id]]
493 emit("LEAK ", id, " addr=", b$start + 8, " block=", b$size, " req=", b$requested, if (id %in% seen) " reachable" else " lost")
494 }
495 } else if (cmd == "HISTORY") {
496 events <- a$histories[[v[1]]]
497 emit("HISTORY ", v[1], " events=", length(events))
498 for (line in events) emit(line)
499 }
500}
501main <- function() {
502 options(scipen = 999)
503 input <- readLines(file("stdin"), warn = FALSE)
504 a <- allocator()
505 for (i in seq_len(as.integer(input[1]))) {
506 words <- strsplit(input[i + 1], " ", fixed = TRUE)[[1]]
507 tryCatch(execute(a, words), command_error = function(e) emit("ERROR ", e$message))
508 if (!is.null(a$heap)) a$peak <- max(a$peak, used_total(a$heap))
509 }
510}
