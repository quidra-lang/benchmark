1use std::collections::{BTreeMap, BTreeSet};
2use std::io::{self, Read};
3
4const MODES: [&str; 6] = ["FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY"];
5#[derive(Clone)]
6struct Block {
7 start: usize,
8 size: usize,
9 owner: Option<String>,
10 stamp: usize,
11}
12impl Block {
13 fn new(start: usize, size: usize) -> Self {
14 Self { start, size, owner: None, stamp: 0 }
15 }
16 fn end(&self) -> usize { self.start + self.size }
17 fn area(&self) -> String { format!("{}+{}", self.start, self.size) }
18}
19struct Heap {
20 mode: String,
21 size: usize,
22 rover: usize,
23 age: usize,
24 blocks: Vec<Block>,
25}
26fn class(size: usize) -> usize { size.ilog2() as usize }
27impl Heap {
28 fn new(size: usize, mode: &str) -> Self {
29 let mut block = Block::new(0, size);
30 block.stamp = 1;
31 Self { mode: mode.into(), size, rover: 0, age: 1, blocks: vec![block] }
32 }
33 fn buddy(&self) -> bool { self.mode == "BUDDY" }
34 fn register(&mut self, mut block: Block) -> Block {
35 self.age += 1;
36 block.stamp = self.age;
37 block
38 }
39 fn required(&self, n: usize) -> usize {
40 let length = n.div_ceil(8) * 8 + 8;
41 if self.buddy() { length.max(32).next_power_of_two() } else { length }
42 }
43 fn locate(&self, address: usize) -> usize {
44 self.blocks.iter().position(|b| b.start <= address && address < b.end()).unwrap()
45 }
46 fn owned(&self, id: &str) -> usize {
47 self.blocks.iter().position(|b| b.owner.as_deref() == Some(id)).unwrap()
48 }
49 fn used(&self) -> usize {
50 self.blocks.iter().filter(|b| b.owner.is_some()).map(|b| b.size).sum()
51 }
52 fn fragmentation(&self) -> usize {
53 let free: Vec<&Block> = self.blocks.iter().filter(|b| b.owner.is_none()).collect();
54 let total: usize = free.iter().map(|b| b.size).sum();
55 let largest = free.iter().map(|b| b.size).max().unwrap_or(0);
56 if total == 0 { 0 } else { (total - largest) * 100 / total }
57 }
58 fn allocate(&mut self, length: usize, alignment: usize, id: &str) -> Option<usize> {
59 let rover = self.locate(self.rover);
60 let mut selected: Option<(i64, i64, usize, usize)> = None;
61 for (i, block) in self.blocks.iter().enumerate() {
62 if block.owner.is_some() { continue; }
63 let mut place = (block.start + 8).div_ceil(alignment) * alignment - 8;
64 if place > block.start && place - block.start < 16 { place += alignment; }
65 if place + length > block.end() { continue; }
66 let (first, second) = match self.mode.as_str() {
67 "NEXT" => (((i + self.blocks.len() - rover) % self.blocks.len()) as i64, 0),
68 "BEST" | "BUDDY" => (block.size as i64, block.start as i64),
69 "WORST" => (-(block.size as i64), block.start as i64),
70 "SEGFIT" => (class(block.size) as i64, -(block.stamp as i64)),
71 _ => (block.start as i64, 0),
72 };
73 let choice = (first, second, i, place);
74 if selected.is_none_or(|old| (first, second) < (old.0, old.1)) { selected = Some(choice); }
75 }
76 let (_, _, mut index, place) = selected?;
77 if self.buddy() {
78 while self.blocks[index].size > length {
79 self.blocks[index].size /= 2;
80 let block = &self.blocks[index];
81 let upper = Block::new(block.end(), block.size);
82 let upper = self.register(upper);
83 self.blocks.insert(index + 1, upper);
84 }
85 } else {
86 let old = self.blocks[index].clone();
87 if place > old.start {
88 let leading = self.register(Block::new(old.start, place - old.start));
89 self.blocks[index] = leading;
90 index += 1;
91 self.blocks.insert(index, Block::new(place, old.end() - place));
92 }
93 let remainder = self.blocks[index].size - length;
94 if remainder >= 16 {
95 self.blocks[index].size = length;
96 let tail = self.register(Block::new(place + length, remainder));
97 self.blocks.insert(index + 1, tail);
98 }
99 }
100 self.blocks[index].owner = Some(id.into());
101 if self.mode == "NEXT" { self.rover = self.blocks[index].end() % self.size; }
102 Some(index)
103 }
104 fn merge_next(&mut self, index: usize) {
105 let upper = self.blocks.remove(index + 1);
106 self.blocks[index].size += upper.size;
107 }
108 fn coalesce(&mut self, mut index: usize) -> (usize, usize) {
109 let mut merges = 0;
110 if self.buddy() {
111 while self.blocks[index].size < self.size {
112 let block = &self.blocks[index];
113 let partner = if block.start % (2 * block.size) == 0 {
114 index + 1
115 } else if index > 0 { index - 1 } else { break; };
116 if partner >= self.blocks.len() || self.blocks[partner].owner.is_some()
117 || self.blocks[partner].size != block.size { break; }
118 index = index.min(partner);
119 self.merge_next(index);
120 merges += 1;
121 }
122 } else {
123 if index + 1 < self.blocks.len() && self.blocks[index + 1].owner.is_none() {
124 self.merge_next(index);
125 merges += 1;
126 }
127 if index > 0 && self.blocks[index - 1].owner.is_none() {
128 index -= 1;
129 self.merge_next(index);
130 merges += 1;
131 }
132 }
133 self.age += 1;
134 self.blocks[index].stamp = self.age;
135 (index, merges)
136 }
137 fn release(&mut self, index: usize) -> (usize, usize) {
138 self.blocks[index].owner = None;
139 self.coalesce(index)
140 }
141 fn resize(&mut self, index: usize, length: usize) -> Option<&'static str> {
142 let start = self.blocks[index].start;
143 let size = self.blocks[index].size;
144 if length <= size {
145 if self.buddy() {
146 if length == size { return Some("kept"); }
147 while self.blocks[index].size > length {
148 self.blocks[index].size /= 2;
149 let size = self.blocks[index].size;
150 let tail = self.register(Block::new(start + size, size));
151 self.blocks.insert(index + 1, tail);
152 }
153 } else {
154 if size - length < 16 { return Some("kept"); }
155 self.blocks[index].size = length;
156 self.blocks.insert(index + 1, Block::new(start + length, size - length));
157 self.coalesce(index + 1);
158 }
159 return Some("shrunk");
160 }
161 if self.buddy() {
162 let mut extent = size;
163 let mut last = index + 1;
164 while extent < length {
165 if start % (2 * extent) != 0 || last >= self.blocks.len()
166 || self.blocks[last].owner.is_some() || self.blocks[last].size != extent
167 || self.blocks[last].start != start + extent { return None; }
168 extent *= 2;
169 last += 1;
170 }
171 self.blocks.drain(index + 1..last);
172 self.blocks[index].size = length;
173 } else {
174 if index + 1 >= self.blocks.len() || self.blocks[index + 1].owner.is_some() { return None; }
175 let total = size + self.blocks[index + 1].size;
176 if total < length { return None; }
177 self.blocks.remove(index + 1);
178 self.blocks[index].size = total;
179 if total - length >= 16 {
180 self.blocks[index].size = length;
181 let tail = self.register(Block::new(start + length, total - length));
182 self.blocks.insert(index + 1, tail);
183 }
184 }
185 Some("grown")
186 }
187}
188#[derive(Default)]
189struct Record {
190 state: u8,
191 contents: Vec<i16>,
192 root: bool,
193 links: BTreeSet<String>,
194 history: Vec<String>,
195}
196#[derive(Default)]
197struct Simulator {
198 heap: Option<Heap>,
199 ids: BTreeMap<String, Record>,
200 counters: [usize; 6],
201 peak: usize,
202 output: Vec<String>,
203}
204fn number(text: &str) -> usize { text.parse().unwrap() }
205impl Simulator {
206 fn emit(&mut self, text: String) { self.output.push(text); }
207 fn record(&mut self, id: &str, text: String) {
208 self.ids.entry(id.into()).or_default().history.push(text.clone());
209 self.emit(text);
210 }
211 fn live(&self, id: &str) -> Result<(), String> {
212 match self.ids.get(id).map_or(0, |r| r.state) {
213 1 => Ok(()),
214 2 => Err(format!("USE AFTER FREE {id}")),
215 _ => Err(format!("UNKNOWN ID {id}")),
216 }
217 }
218 fn range(&self, id: &str, offset: usize, length: usize) -> Result<(), String> {
219 self.live(id)?;
220 if offset + length > self.ids[id].contents.len() { return Err(format!("OUT OF BOUNDS {id}")); }
221 Ok(())
222 }
223 fn reachable(&self) -> BTreeSet<String> {
224 let mut found: BTreeSet<String> = self.ids.iter().filter(|(_, r)| r.state == 1 && r.root).map(|(id, _)| id.clone()).collect();
225 let mut pending: Vec<String> = found.iter().cloned().collect();
226 while let Some(id) = pending.pop() {
227 for target in &self.ids[&id].links {
228 if found.insert(target.clone()) { pending.push(target.clone()); }
229 }
230 }
231 found
232 }
233 fn oom(&mut self, id: &str) -> String {
234 self.counters[4] += 1;
235 format!("OUT OF MEMORY {id}")
236 }
237 fn allocate_id(&mut self, verb: &str, id: &str, n: usize, alignment: usize, fill: i16) -> Result<(), String> {
238 if n == 0 { return Err("BAD SIZE".into()); }
239 if self.ids.get(id).is_some_and(|r| r.state == 1) { return Err(format!("DUPLICATE ID {id}")); }
240 let heap = self.heap.as_mut().unwrap();
241 let length = heap.required(n);
242 let Some(index) = heap.allocate(length, alignment, id) else { return Err(self.oom(id)); };
243 let text = format!("{verb} {id} addr={} block={}", heap.blocks[index].start + 8, heap.blocks[index].size);
244 let record = self.ids.entry(id.into()).or_default();
245 record.state = 1;
246 record.contents = vec![fill; n];
247 self.counters[0] += 1;
248 self.record(id, text);
249 Ok(())
250 }
251 fn free_id(&mut self, id: &str, verb: &str) -> Result<(), String> {
252 let state = self.ids.get(id).map_or(0, |r| r.state);
253 if state != 1 { return Err(format!("{} {id}", if state == 2 { "DOUBLE FREE" } else { "UNKNOWN ID" })); }
254 let heap = self.heap.as_mut().unwrap();
255 let index = heap.owned(id);
256 let size = heap.blocks[index].size;
257 let (index, merges) = heap.release(index);
258 let text = format!("{verb} {id} block={size} merges={merges} free={}", heap.blocks[index].area());
259 let record = self.ids.get_mut(id).unwrap();
260 record.state = 2;
261 record.root = false;
262 record.links.clear();
263 record.contents.clear();
264 for record in self.ids.values_mut() { record.links.remove(id); }
265 self.record(id, text);
266 if verb == "FREE" { self.counters[1] += 1; }
267 Ok(())
268 }
269 fn tree_lines(heap: &Heap, start: usize, size: usize, depth: usize, out: &mut Vec<String>) {
270 let label = format!("{}{start}+{size}", " ".repeat(depth * 2));
271 let block = &heap.blocks[heap.locate(start)];
272 if block.start == start && block.size == size {
273 out.push(format!("{label} {}", block.owner.as_ref().map_or("FREE".into(), |id| format!("USED {id}"))));
274 } else {
275 out.push(format!("{label} split"));
276 Self::tree_lines(heap, start, size / 2, depth + 1, out);
277 Self::tree_lines(heap, start + size / 2, size / 2, depth + 1, out);
278 }
279 }
280 fn run(&mut self, a: &[&str]) -> Result<(), String> {
281 let op = a[0];
282 if op == "HEAP" {
283 let size = number(a[1]);
284 if !(32..=65536).contains(&size) || size % 8 != 0 || (a[2] == "BUDDY" && !size.is_power_of_two()) {
285 return Err("BAD HEAP SIZE".into());
286 }
287 self.heap = Some(Heap::new(size, a[2]));
288 self.ids.clear(); self.counters = [0; 6]; self.peak = 0;
289 self.emit(format!("HEAP size={size} mode={}", a[2]));
290 return Ok(());
291 }
292 if op == "BENCH" {
293 let size = number(a[1]);
294 if !(32..=65536).contains(&size) || !size.is_power_of_two() { return Err("BAD HEAP SIZE".into()); }
295 let seed: u64 = a[2].parse().unwrap();
296 let ops = number(a[3]); let maxreq = number(a[4]);
297 self.emit(format!("BENCH heap={size} seed={seed} ops={ops} maxreq={maxreq}"));
298 for mode in MODES {
299 let mut heap = Heap::new(size, mode);
300 let mut live = Vec::<String>::new();
301 let (mut ok, mut fail, mut peak) = (0, 0, 0);
302 let mut x = seed;
303 for step in 0..ops {
304 x = x * 48271 % 2147483647;
305 let alloc = live.is_empty() || x % 3 != 0;
306 x = x * 48271 % 2147483647;
307 if alloc {
308 let id = step.to_string();
309 let n = 1 + x as usize % maxreq;
310 let length = heap.required(n);
311 if heap.allocate(length, 8, &id).is_some() { ok += 1; live.push(id); } else { fail += 1; }
312 } else {
313 let id = live.remove(x as usize % live.len());
314 let index = heap.owned(&id); heap.release(index);
315 }
316 peak = peak.max(heap.used());
317 }
318 self.emit(format!("BENCH {mode} ok={ok} fail={fail} peak={peak} used={} frag={} blocks={}", heap.used(), heap.fragmentation(), heap.blocks.len()));
319 }
320 return Ok(());
321 }
322 if self.heap.is_none() { return Err("NO HEAP".into()); }
323 match op {
324 "MALLOC" | "CALLOC" | "MEMALIGN" => {
325 let mut alignment = 8;
326 let n = match op {
327 "CALLOC" => number(a[2]) * number(a[3]),
328 "MEMALIGN" => {
329 if self.heap.as_ref().unwrap().buddy() { return Err("UNSUPPORTED IN BUDDY MODE".into()); }
330 alignment = number(a[2]);
331 if !(8..=4096).contains(&alignment) || !alignment.is_power_of_two() { return Err("BAD ALIGNMENT".into()); }
332 number(a[3])
333 }
334 _ => number(a[2]),
335 };
336 self.allocate_id(op, a[1], n, alignment, if op == "CALLOC" { 0 } else { -1 })?;
337 }
338 "FREE" => self.free_id(a[1], "FREE")?,
339 "REALLOC" => {
340 let id = a[1]; let n = number(a[2]);
341 if n == 0 { return self.free_id(id, "FREE"); }
342 let state = self.ids.get(id).map_or(0, |r| r.state);
343 if state == 0 { return self.allocate_id("MALLOC", id, n, 8, -1); }
344 self.live(id)?;
345 let heap = self.heap.as_mut().unwrap();
346 let index = heap.owned(id); let old = heap.blocks[index].start;
347 let length = heap.required(n);
348 let kind = if let Some(kind) = heap.resize(index, length) {
349 kind.to_string()
350 } else {
351 if heap.allocate(length, 8, id).is_none() { return Err(self.oom(id)); }
352 let old_index = heap.locate(old);
353 heap.release(old_index);
354 self.counters[3] += 1;
355 format!("moved from={}", old + 8)
356 };
357 let index = heap.owned(id);
358 let text = format!("REALLOC {id} addr={} block={} {kind}", heap.blocks[index].start + 8, heap.blocks[index].size);
359 self.ids.get_mut(id).unwrap().contents.resize(n, -1);
360 self.counters[2] += 1;
361 self.record(id, text);
362 }
363 "FILL" | "READ" | "COPY" => {
364 let id = a[1]; let offset = number(a[2]);
365 let length = number(a[if op == "COPY" { 5 } else { 3 }]);
366 self.range(id, offset, length)?;
367 if op == "COPY" {
368 let source = a[3]; let source_offset = number(a[4]);
369 self.range(source, source_offset, length)?;
370 let bytes = self.ids[source].contents[source_offset..source_offset + length].to_vec();
371 self.ids.get_mut(id).unwrap().contents[offset..offset + length].copy_from_slice(&bytes);
372 self.emit(format!("COPY {id} {offset}+{length} from {source} {source_offset}"));
373 } else if op == "FILL" {
374 let value = number(a[4]) as i16;
375 self.ids.get_mut(id).unwrap().contents[offset..offset + length].fill(value);
376 self.emit(format!("FILL {id} {offset}+{length} = {value}"));
377 } else {
378 let bytes: Vec<String> = self.ids[id].contents[offset..offset + length].iter().map(|v| if *v < 0 { "?".into() } else { v.to_string() }).collect();
379 self.emit(format!("READ {id} {offset}+{length} = {}", bytes.join(" ")));
380 }
381 }
382 "ROOT" | "UNROOT" | "LINK" | "UNLINK" => {
383 let id = a[1]; self.live(id)?;
384 if op == "ROOT" || op == "UNROOT" {
385 let record = self.ids.get_mut(id).unwrap();
386 if op == "UNROOT" && !record.root { return Err(format!("NOT ROOT {id}")); }
387 record.root = op == "ROOT";
388 self.emit(format!("{op} {id}"));
389 } else {
390 let target = a[2]; self.live(target)?;
391 let record = self.ids.get_mut(id).unwrap();
392 if op == "LINK" { record.links.insert(target.into()); }
393 else if !record.links.remove(target) { return Err(format!("NO LINK {id} {target}")); }
394 self.emit(format!("{op} {id} {target}"));
395 }
396 }
397 "GC" | "LEAKS" => {
398 let found = self.reachable();
399 let heap = self.heap.as_ref().unwrap();
400 let mut names: Vec<String> = heap.blocks.iter().filter_map(|b| b.owner.clone()).collect();
401 if op == "GC" {
402 names.retain(|id| !found.contains(id));
403 let bytes: usize = names.iter().map(|id| heap.blocks[heap.owned(id)].size).sum();
404 self.emit(format!("GC collected={} bytes={bytes}", names.len()));
405 for id in &names { self.free_id(id, "COLLECT")?; }
406 self.counters[5] += names.len();
407 } else {
408 names.sort();
409 let requested: usize = names.iter().map(|id| self.ids[id].contents.len()).sum();
410 let lost = names.iter().filter(|id| !found.contains(*id)).count();
411 let mut lines = vec![format!("LEAKS count={} requested={requested} lost={lost}", names.len())];
412 for id in &names {
413 let block = &heap.blocks[heap.owned(id)];
414 lines.push(format!("LEAK {id} addr={} block={} req={} {}", block.start + 8, block.size, self.ids[id].contents.len(), if found.contains(id) { "reachable" } else { "lost" }));
415 }
416 self.output.extend(lines);
417 }
418 }
419 "HISTORY" => {
420 let events = self.ids.get(a[1]).map_or_else(Vec::new, |r| r.history.clone());
421 self.emit(format!("HISTORY {} events={}", a[1], events.len()));
422 self.output.extend(events);
423 }
424 "SBRK" => {
425 let k = number(a[1]); let heap = self.heap.as_mut().unwrap();
426 if heap.size + k > 65536 || if heap.buddy() { k != heap.size } else { k < 16 || k % 8 != 0 } {
427 return Err("BAD SBRK SIZE".into());
428 }
429 heap.blocks.push(Block::new(heap.size, k)); heap.size += k;
430 let last = heap.blocks.len() - 1;
431 let (index, _) = heap.coalesce(last);
432 let text = format!("SBRK heap={} free={}", heap.size, heap.blocks[index].area()); self.emit(text);
433 }
434 "TRIM" => {
435 let heap = self.heap.as_mut().unwrap(); let old = heap.size;
436 while heap.blocks.len() > 1 {
437 let block = heap.blocks.last().unwrap();
438 if block.owner.is_some() || (heap.buddy() && block.size * 2 != heap.size) { break; }
439 heap.size = block.start; heap.blocks.pop();
440 if !heap.buddy() { break; }
441 }
442 if heap.rover >= heap.size { heap.rover = 0; }
443 let text = format!("TRIM heap={} released={}", heap.size, old - heap.size); self.emit(text);
444 }
445 "COMPACT" => {
446 let heap = self.heap.as_mut().unwrap();
447 if heap.buddy() { return Err("UNSUPPORTED IN BUDDY MODE".into()); }
448 heap.blocks.retain(|b| b.owner.is_some());
449 let mut address = 0; let mut moved = Vec::new();
450 for block in &mut heap.blocks {
451 if block.start != address {
452 let id = block.owner.as_ref().unwrap().clone();
453 moved.push((id.clone(), format!("MOVE {id} {} -> {}", block.start + 8, address + 8)));
454 block.start = address;
455 }
456 address += block.size;
457 }
458 let free = if address < heap.size {
459 let block = heap.register(Block::new(address, heap.size - address));
460 let area = block.area(); heap.blocks.push(block); area
461 } else { "none".into() };
462 heap.rover = 0; self.counters[3] += moved.len();
463 self.emit(format!("COMPACT moved={} free={free}", moved.len()));
464 for (id, line) in moved { self.record(&id, line); }
465 }
466 "WHO" => {
467 let address = number(a[1]); let heap = self.heap.as_ref().unwrap();
468 let detail = if address >= heap.size { "outside".into() } else {
469 let block = &heap.blocks[heap.locate(address)];
470 if let Some(id) = &block.owner {
471 if address < block.start + 8 { format!("header {id}") }
472 else {
473 let offset = address - block.start - 8;
474 format!("{} {id}+{offset}", if offset < self.ids[id].contents.len() { "data" } else { "slack" })
475 }
476 } else { format!("free {}", block.area()) }
477 };
478 self.emit(format!("WHO {address} {detail}"));
479 }
480 "DUMP" => {
481 let heap = self.heap.as_ref().unwrap();
482 let mut first = format!("DUMP mode={} heap={} blocks={}", heap.mode, heap.size, heap.blocks.len());
483 if heap.mode == "NEXT" { first += &format!(" rover={}", heap.rover); }
484 let mut lines = vec![first];
485 for block in &heap.blocks {
486 let detail = block.owner.as_ref().map_or("FREE".into(), |id| format!("USED {id} req={}", self.ids[id].contents.len()));
487 lines.push(format!("{} {} {detail}", block.start, block.size));
488 }
489 self.output.extend(lines);
490 }
491 "MAP" => {
492 let heap = self.heap.as_ref().unwrap();
493 let mut units = String::new();
494 for block in &heap.blocks {
495 units.push(if block.owner.is_some() { 'H' } else { 'F' });
496 for k in 0..block.size / 8 - 1 {
497 units.push(block.owner.as_ref().map_or('.', |id| if k * 8 < self.ids[id].contents.len() { '#' } else { '-' }));
498 }
499 }
500 let mut lines = vec![format!("MAP heap={}", heap.size)];
501 for first in (0..units.len()).step_by(64) { lines.push(format!("{:05} {}", first * 8, &units[first..(first + 64).min(units.len())])); }
502 self.output.extend(lines);
503 }
504 "FREELIST" => {
505 let heap = self.heap.as_ref().unwrap();
506 let mut groups = BTreeMap::<usize, Vec<&Block>>::new(); let mut count = 0;
507 for block in &heap.blocks {
508 if block.owner.is_none() { groups.entry(class(block.size)).or_default().push(block); count += 1; }
509 }
510 let mut lines = vec![format!("FREELIST count={count}")];
511 for (k, mut blocks) in groups {
512 if heap.mode == "SEGFIT" { blocks.sort_by_key(|b| std::cmp::Reverse(b.stamp)); }
513 lines.push(format!("{}-{}: {}", 1usize << k, (1usize << (k + 1)) - 1, blocks.iter().map(|b| b.area()).collect::<Vec<_>>().join(" ")));
514 }
515 self.output.extend(lines);
516 }
517 "TREE" => {
518 let heap = self.heap.as_ref().unwrap();
519 if !heap.buddy() { return Err("REQUIRES BUDDY MODE".into()); }
520 self.output.push(format!("TREE heap={}", heap.size));
521 Self::tree_lines(heap, 0, heap.size, 0, &mut self.output);
522 }
523 "STATS" => {
524 let heap = self.heap.as_ref().unwrap();
525 let used = heap.used();
526 let free: Vec<&Block> = heap.blocks.iter().filter(|b| b.owner.is_none()).collect();
527 let largest = free.iter().map(|b| b.size).max().unwrap_or(0);
528 let requested: usize = self.ids.values().filter(|r| r.state == 1).map(|r| r.contents.len()).sum();
529 let text = format!("STATS used={used} used_blocks={} free={} free_blocks={} largest={largest} frag={} requested={requested} peak={}", heap.blocks.len() - free.len(), heap.size - used, free.len(), heap.fragmentation(), self.peak);
530 self.emit(text);
531 let c = self.counters;
532 self.emit(format!("OPS allocs={} frees={} reallocs={} moves={} failures={} collected={}", c[0], c[1], c[2], c[3], c[4], c[5]));
533 }
534 _ => unreachable!(),
535 }
536 Ok(())
537 }
538}
539fn main() {
540 let mut input = String::new(); io::stdin().read_to_string(&mut input).unwrap();
541 let mut lines = input.lines(); let count: usize = lines.next().unwrap().parse().unwrap();
542 let mut simulator = Simulator::default();
543 for line in lines.take(count) {
544 let args: Vec<&str> = line.split_whitespace().collect();
545 if let Err(message) = simulator.run(&args) { simulator.emit(format!("ERROR {message}")); }
546 if let Some(heap) = &simulator.heap { simulator.peak = simulator.peak.max(heap.used()); }
547 }
548 println!("{}", simulator.output.join("\n"));
549}
