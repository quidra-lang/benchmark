28static int power(int n) {
29 return n > 0 && (n & (n - 1)) == 0;
30}
31static int sizeclass(int n) {
32 int k = 0;
33 while ((1 << (k + 1)) <= n)
34 ++k;
35 return k;
36}
37static int idof(const char *name) {
38 for (int i = 1; i <= idCount; ++i)
39 if (!strcmp(ids[i].name, name))
40 return i;
41 strcpy(ids[++idCount].name, name);
42 return idCount;
43}
44static void history(int id, const char *line) {
45 Id *p = &ids[id];
46 p->history = realloc(p->history, (size_t)(p->events + 1) * sizeof(char *));
47 p->history[p->events] = malloc(strlen(line) + 1);
48 strcpy(p->history[p->events++], line);
49 puts(line);
50}
51static void initialize(Heap *h, int size, int mode) {
52 memset(h, 0, sizeof(*h));
53 h->size = size;
54 h->mode = mode;
55 h->count = 1;
56 h->age = 1;
57 h->b[0] = (Block){0, size, 0, 1};
58}
59static void insert(Heap *h, int i, Block b) {
60 memmove(h->b + i + 1, h->b + i, (size_t)(h->count - i) * sizeof(Block));
61 h->b[i] = b;
62 ++h->count;
63}
64static void erase(Heap *h, int i) {
65 --h->count;
66 memmove(h->b + i, h->b + i + 1, (size_t)(h->count - i) * sizeof(Block));
67}
68static int required(const Heap *h, int n) {
69 int s = (n + 7) / 8 * 8 + 8;
70 if (h->mode == 5) {
71 int p = 32;
72 while (p < s)
73 p *= 2;
74 s = p;
75 }
76 return s;
77}
78static int locate(const Heap *h, int a) {
79 for (int i = 0; i < h->count; ++i)
80 if (a < h->b[i].start + h->b[i].size)
81 return i;
82 return -1;
83}
84static int blockof(const Heap *h, int id) {
85 for (int i = 0; i < h->count; ++i)
86 if (h->b[i].owner == id)
87 return i;
88 return -1;
89}
90static int allocate(Heap *h, int s, int alignment, int owner) {
91 int chosen = -1, place = 0, best1 = 0, best2 = 0;
92 int rover = locate(h, h->rover);
93 for (int i = 0; i < h->count; ++i) {
94 Block b = h->b[i];
95 if (b.owner)
96 continue;
97 int p = (b.start + 8 + alignment - 1) / alignment * alignment - 8;
98 if (p > b.start && p - b.start < 16)
99 p += alignment;
100 if (p + s > b.start + b.size)
101 continue;
102 int r1 = b.start, r2 = 0;
103 if (h->mode == 1)
104 r1 = (i - rover + h->count) % h->count;
105 if (h->mode == 2 || h->mode == 5) {
106 r1 = b.size;
107 r2 = b.start;
108 }
109 if (h->mode == 3) {
110 r1 = -b.size;
111 r2 = b.start;
112 }
113 if (h->mode == 4) {
114 r1 = sizeclass(b.size);
115 r2 = -b.stamp;
116 }
117 if (chosen < 0 || r1 < best1 || (r1 == best1 && r2 < best2)) {
118 chosen = i;
119 place = p;
120 best1 = r1;
121 best2 = r2;
122 }
123 }
124 if (chosen < 0)
125 return -1;
126 int i = chosen;
127 if (h->mode == 5) {
128 while (h->b[i].size > s) {
129 h->b[i].size /= 2;
130 insert(h, i + 1, (Block){h->b[i].start + h->b[i].size, h->b[i].size, 0, ++h->age});
131 }
132 } else {
133 Block old = h->b[i];
134 if (place > old.start) {
135 h->b[i] = (Block){old.start, place - old.start, 0, ++h->age};
136 insert(h, ++i, (Block){place, old.start + old.size - place, 0, 0});
137 }
138 if (h->b[i].size - s >= 16) {
139 int tail = h->b[i].size - s;
140 h->b[i].size = s;
141 insert(h, i + 1, (Block){place + s, tail, 0, ++h->age});
142 }
143 }
144 h->b[i].owner = owner;
145 if (h->mode == 1)
146 h->rover = (h->b[i].start + h->b[i].size) % h->size;
147 return i;
148}
149static int coalesce(Heap *h, int i, int *merges) {
150 *merges = 0;
151 if (h->mode == 5) {
152 while (h->b[i].size < h->size) {
153 Block b = h->b[i];
154 int other = b.start % (2 * b.size) == 0 ? i + 1 : i - 1;
155 if (other < 0 || other >= h->count || h->b[other].owner || h->b[other].size != b.size)
156 break;
157 if (other < i)
158 i = other;
159 h->b[i].size *= 2;
160 erase(h, i + 1);
161 ++*merges;
162 }
163 } else {
164 if (i + 1 < h->count && !h->b[i + 1].owner) {
165 h->b[i].size += h->b[i + 1].size;
166 erase(h, i + 1);
167 ++*merges;
168 }
169 if (i > 0 && !h->b[i - 1].owner) {
170 h->b[i - 1].size += h->b[i].size;
171 erase(h, i);
172 --i;
173 ++*merges;
174 }
175 }
176 h->b[i].stamp = ++h->age;
177 return i;
178}
179static int used(const Heap *h) {
180 int u = 0;
181 for (int i = 0; i < h->count; ++i)
182 if (h->b[i].owner)
183 u += h->b[i].size;
184 return u;
185}
186static int frag(const Heap *h) {
187 int f = 0, l = 0;
188 for (int i = 0; i < h->count; ++i)
189 if (!h->b[i].owner) {
190 f += h->b[i].size;
191 if (l < h->b[i].size)
192 l = h->b[i].size;
193 }
194 return f ? (f - l) * 100 / f : 0;
195}
196static int check(int id) {
197 if (ids[id].state == 1)
198 return 1;
199 printf("ERROR %s %s\n", ids[id].state == 2 ? "USE AFTER FREE" : "UNKNOWN ID", ids[id].name);
200 return 0;
201}
202static void release(int id, const char *verb) {
203 int i = blockof(&heap, id), size = heap.b[i].size, merges;
204 heap.b[i].owner = 0;
205 i = coalesce(&heap, i, &merges);
206 ids[id].state = 2;
207 ids[id].root = 0;
208 for (int j = 1; j <= idCount; ++j)
209 links[id][j] = links[j][id] = 0;
210 free(ids[id].data);
211 ids[id].data = NULL;
212 char line[256];
213 snprintf(line, sizeof(line), "%s %s block=%d merges=%d free=%d+%d", verb, ids[id].name, size,
214 merges, heap.b[i].start, heap.b[i].size);
215 history(id, line);
216}
217static void reachable(unsigned char *seen) {
218 memset(seen, 0, LIMIT);
219 for (int i = 1; i <= idCount; ++i)
220 seen[i] = ids[i].state == 1 && ids[i].root;
221 int change = 1;
222 while (change) {
223 change = 0;
224 for (int i = 1; i <= idCount; ++i)
225 if (seen[i]) {
226 for (int j = 1; j <= idCount; ++j)
227 if (links[i][j] && !seen[j]) {
228 seen[j] = 1;
229 change = 1;
230 }
231 }
232 }
233}
234static void tree(int start, int size, int depth) {
235 int i = locate(&heap, start);
236 printf("%*s%d+%d ", depth * 2, "", start, size);
237 if (heap.b[i].start == start && heap.b[i].size == size) {
238 if (heap.b[i].owner)
239 printf("USED %s\n", ids[heap.b[i].owner].name);
240 else
241 puts("FREE");
242 } else {
243 puts("split");
244 tree(start, size / 2, depth + 1);
245 tree(start + size / 2, size / 2, depth + 1);
246 }
247}
248static void bench(int s, int64_t seed, int ops, int maxreq) {
249 if (!power(s) || s < 32 || s > 65536) {
250 puts("ERROR BAD HEAP SIZE");
251 return;
252 }
253 printf("BENCH heap=%d seed=%lld ops=%d maxreq=%d\n", s, (long long)seed, ops, maxreq);
254 Heap *h = malloc(sizeof(Heap));
255 for (int mode = 0; mode < 6; ++mode) {
256 initialize(h, s, mode);
257 int live[LIMIT], count = 0, serial = 0, ok = 0, fail = 0, top = 0;
258 int64_t x = seed;
259 for (int step = 0; step < ops; ++step) {
260 x = x * 48271 % 2147483647;
261 int alloc = !count || x % 3 != 0;
262 x = x * 48271 % 2147483647;
263 if (alloc) {
264 int n = 1 + (int)(x % maxreq), id = ++serial;
265 if (allocate(h, required(h, n), 8, id) < 0)
266 ++fail;
267 else {
268 live[count++] = id;
269 ++ok;
270 }
271 } else {
272 int at = (int)(x % count), i = blockof(h, live[at]), merges;
273 memmove(live + at, live + at + 1, (size_t)(--count - at) * sizeof(int));
274 h->b[i].owner = 0;
275 coalesce(h, i, &merges);
276 }
277 int u = used(h);
278 if (u > top)
279 top = u;
280 }
281 printf("BENCH %s ok=%d fail=%d peak=%d used=%d frag=%d blocks=%d\n", modes[mode], ok, fail,
282 top, used(h), frag(h), h->count);
283 }
284 free(h);
285}

644 if (!b.owner)
645 printf("free %d+%d\n", b.start, b.size);
646 else if (p < b.start + 8)
647 printf("header %s\n", ids[b.owner].name);
648 else
649 printf("%s %s+%d\n", p - b.start - 8 < ids[b.owner].request ? "data" : "slack",
650 ids[b.owner].name, p - b.start - 8);
651 }
652 return;
653 }
654 if (!strcmp(op, "DUMP")) {
655 printf("DUMP mode=%s heap=%d blocks=%d", modes[heap.mode], heap.size, heap.count);
656 if (heap.mode == 1)
657 printf(" rover=%d", heap.rover);
658 putchar('\n');
659 for (int i = 0; i < heap.count; ++i) {
660 Block b = heap.b[i];
661 printf("%d %d ", b.start, b.size);
662 if (!b.owner)
663 puts("FREE");
664 else
665 printf("USED %s req=%d\n", ids[b.owner].name, ids[b.owner].request);
666 }
667 } else if (!strcmp(op, "MAP")) {
668 char map[8193];
669 int count = 0;
670 printf("MAP heap=%d\n", heap.size);
671 for (int i = 0; i < heap.count; ++i) {
672 Block b = heap.b[i];
673 map[count++] = b.owner ? 'H' : 'F';
674 for (int j = 0; j < b.size / 8 - 1; ++j)
675 map[count++] = !b.owner ? '.' : j * 8 < ids[b.owner].request ? '#' : '-';
676 }
677 for (int j = 0; j < count; j += 64)
678 printf("%05d %.*s\n", j * 8, count - j < 64 ? count - j : 64, map + j);
679 } else if (!strcmp(op, "FREELIST")) {
680 int list[BLOCKS], count = 0;
681 for (int i = 0; i < heap.count; ++i)
682 if (!heap.b[i].owner)
683 list[count++] = i;
684 printf("FREELIST count=%d\n", count);
685 for (int k = 0; k <= 16; ++k) {
686 int members[BLOCKS], n = 0;
687 for (int j = 0; j < count; ++j)
688 if (sizeclass(heap.b[list[j]].size) == k)
689 members[n++] = list[j];
690 if (!n)
691 continue;
692 if (heap.mode == 4)
693 for (int j = 0; j < n; ++j)
694 for (int l = j + 1; l < n; ++l)
695 if (heap.b[members[j]].stamp < heap.b[members[l]].stamp) {
696 int t = members[j];
697 members[j] = members[l];
698 members[l] = t;
699 }
700 printf("%d-%d:", 1 << k, (1 << (k + 1)) - 1);
701 for (int j = 0; j < n; ++j)
702 printf(" %d+%d", heap.b[members[j]].start, heap.b[members[j]].size);
703 putchar('\n');
704 }
705 } else if (!strcmp(op, "TREE")) {
706 if (heap.mode != 5)
707 puts("ERROR REQUIRES BUDDY MODE");
708 else {
709 printf("TREE heap=%d\n", heap.size);
710 tree(0, heap.size, 0);
711 }
712 } else if (!strcmp(op, "STATS")) {
713 int u = used(&heap), ub = 0, fb = 0, largest = 0, req = 0;
714 for (int i = 0; i < heap.count; ++i) {
715 Block b = heap.b[i];
716 if (b.owner) {
717 ++ub;
718 req += ids[b.owner].request;
719 } else {
720 ++fb;
721 if (b.size > largest)
722 largest = b.size;
723 }
724 }
725 printf(
726 "STATS used=%d used_blocks=%d free=%d free_blocks=%d largest=%d frag=%d requested=%d peak=%d\n",
727 u, ub, heap.size - u, fb, largest, frag(&heap), req, peak);
728 printf("OPS allocs=%d frees=%d reallocs=%d moves=%d failures=%d collected=%d\n",
729 counters[0], counters[1], counters[2], counters[3], counters[4], counters[5]);
730 }
731}
