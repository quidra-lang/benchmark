19func isPowerOfTwo(_ n: Int) -> Bool { n > 0 && (n & (n - 1)) == 0 }
20func sizeClass(_ n: Int) -> Int {
21 var result = 0
22 while (1 << (result + 1)) <= n { result += 1 }
23 return result
24}
25func fragmentationPercent(_ free: Int, _ largest: Int) -> Int {
26 free == 0 ? 0 : (free - largest) * 100 / free
27}
28func nextRandom(_ state: Int) -> Int { randomMultiplier * state % randomModulus }

30final class Block {
31 var start: Int
32 var size: Int
33 var used = false
34 var owner: String?
35 var requested = 0
36 var contents: [Int] = []
37 var stamp = 0
38
39 init(_ start: Int, _ size: Int) { self.start = start; self.size = size }
40 var end: Int { start + size }
41 var payload: Int { start + header }
42 var area: String { "\(start)+\(size)" }
43 var mapUnits: String {
44 var result = used ? "H" : "F"
45 for i in 0..<(size / unit - 1) {
46 result += !used ? "." : i * unit < requested ? "#" : "-"
47 }
48 return result
49 }
50}

62final class Heap {
63 let mode: String
64 var size: Int
65 var rover = 0
66 private var registrations = 0
67 var blocks: [Block] = []
68
69 init(_ mode: String, _ size: Int) {
70 self.mode = mode
71 self.size = size
72 blocks = [register(Block(0, size))]
73 }
74 private func register(_ block: Block) -> Block {
75 registrations += 1
76 block.stamp = registrations
77 return block
78 }
79 var buddy: Bool { mode == "BUDDY" }
80 func requiredLength(_ request: Int) -> Int {
81 let length = (request + unit - 1) / unit * unit + header
82 if !buddy { return length }
83 var power = minBuddyBlock
84 while power < length { power *= 2 }
85 return power
86 }
87 func locate(_ address: Int) -> Int {
88 var index = 0
89 while blocks[index].end <= address { index += 1 }
90 return index
91 }
92 func indexOf(_ block: Block) -> Int { blocks.firstIndex { $0 === block }! }
93 var usedBlocks: [Block] { blocks.filter { $0.used } }
94 var freeBlocks: [Block] { blocks.filter { !$0.used } }
95 var usedTotal: Int { usedBlocks.reduce(0) { $0 + $1.size } }
96 var fragmentation: Int {
97 let free = freeBlocks
98 return fragmentationPercent(free.reduce(0) { $0 + $1.size },
99 free.map(\.size).max() ?? 0)
100 }
101 func placement(_ block: Block, _ alignment: Int) -> Int {
102 var p = (block.start + header + alignment - 1) / alignment * alignment - header
103 if p > block.start && p - block.start < minBlock { p += alignment }
104 return p
105 }
106 func rank(_ index: Int, _ roverIndex: Int) -> (Int, Int) {
107 let block = blocks[index]
108 switch mode {
109 case "NEXT": return ((index - roverIndex + blocks.count) % blocks.count, 0)
110 case "BEST", "BUDDY": return (block.size, block.start)
111 case "WORST": return (-block.size, block.start)
112 case "SEGFIT": return (sizeClass(block.size), -block.stamp)
113 default: return (block.start, 0)
114 }
115 }
116 func choose(_ length: Int, _ alignment: Int) -> Choice? {
117 let roverIndex = locate(rover)
118 var selected: Choice?
119 for index in blocks.indices {
120 let block = blocks[index]
121 if block.used { continue }
122 let place = placement(block, alignment)
123 if place + length > block.end { continue }
124 let r = rank(index, roverIndex)
125 if let previous = selected {
126 if r.0 > previous.rank.0 || (r.0 == previous.rank.0 && r.1 >= previous.rank.1) {
127 continue
128 }
129 }
130 selected = Choice(index: index, place: place, rank: r)
131 }
132 return selected
133 }
134 func allocate(_ length: Int, _ alignment: Int) -> Block? {
135 guard let selected = choose(length, alignment) else { return nil }
136 let block = buddy ? halve(selected.index, length) : carve(selected.index, selected.place, length)
137 block.used = true
138 if mode == "NEXT" { rover = block.end == size ? 0 : block.end }
139 return block
140 }
141 func carve(_ index: Int, _ place: Int, _ length: Int) -> Block {
142 let previous = blocks[index]
143 var parts: [Block] = []
144 if place > previous.start {
145 parts.append(register(Block(previous.start, place - previous.start)))
146 }
147 let block = Block(place, previous.end - place)
148 parts.append(block)
149 if block.size - length >= minBlock {
150 parts.append(register(Block(place + length, block.size - length)))
151 block.size = length
152 }
153 blocks.replaceSubrange(index..<(index + 1), with: parts)
154 return block
155 }
156 func halve(_ index: Int, _ length: Int) -> Block {
157 let block = blocks[index]
158 while block.size > length {
159 block.size /= 2
160 blocks.insert(register(Block(block.end, block.size)), at: index + 1)
161 }
162 return block
163 }
164 private func mergeNext(_ index: Int) {
165 blocks[index].size += blocks.remove(at: index + 1).size
166 }
167 func coalesce(_ firstIndex: Int) -> (Block, Int) {
168 var index = firstIndex
169 var merges = 0
170 if buddy {
171 while blocks[index].size < size {
172 let block = blocks[index]
173 let partner = block.start % (2 * block.size) == 0 ? index + 1 : index - 1
174 if partner < 0 || partner >= blocks.count { break }
175 let other = blocks[partner]
176 if other.used || other.size != block.size { break }
177 index = min(index, partner)
178 mergeNext(index)
179 merges += 1
180 }
181 } else {
182 if index + 1 < blocks.count && !blocks[index + 1].used {
183 mergeNext(index)
184 merges += 1
185 }
186 if index > 0 && !blocks[index - 1].used {
187 index -= 1
188 mergeNext(index)
189 merges += 1
190 }
191 }
192 return (register(blocks[index]), merges)
193 }
194 func release(_ block: Block) -> (Block, Int) {
195 block.used = false
196 block.owner = nil
197 block.requested = 0
198 block.contents = []
199 return coalesce(indexOf(block))
200 }
201 func shrink(_ index: Int, _ length: Int) -> String {
202 let block = blocks[index]
203 if buddy {
204 if block.size == length { return "kept" }
205 _ = halve(index, length)
206 return "shrunk"
207 }
208 if block.size - length < minBlock { return "kept" }
209 blocks.insert(Block(block.start + length, block.size - length), at: index + 1)
210 block.size = length
211 _ = coalesce(index + 1)
212 return "shrunk"
213 }
214 func growIntoNext(_ index: Int, _ length: Int) -> String? {
215 let block = blocks[index]
216 if index + 1 >= blocks.count || blocks[index + 1].used { return nil }
217 let total = block.size + blocks[index + 1].size
218 if total < length { return nil }
219 blocks.remove(at: index + 1)
220 if total - length >= minBlock {
221 block.size = length
222 blocks.insert(register(Block(block.end, total - length)), at: index + 1)
223 } else {
224 block.size = total
225 }
226 return "grown"
227 }
228 func growBuddy(_ index: Int, _ length: Int) -> String? {
229 let block = blocks[index]
230 var extent = block.size, last = index
231 while extent < length {
232 last += 1
233 if block.start % (2 * extent) != 0 || last >= blocks.count { return nil }
234 let partner = blocks[last]
235 if partner.used || partner.size != extent { return nil }
236 extent *= 2
237 }
238 blocks.removeSubrange((index + 1)...last)
239 block.size = length
240 return "grown"
241 }
242 func resize(_ block: Block, _ length: Int) -> String? {
243 let index = indexOf(block)
244 if length <= block.size { return shrink(index, length) }
245 return buddy ? growBuddy(index, length) : growIntoNext(index, length)
246 }
247 func extend(_ amount: Int) -> Block {
248 blocks.append(Block(size, amount))
249 size += amount
250 return coalesce(blocks.count - 1).0
251 }
252 func trim() -> Int {
253 var released = 0
254 if buddy {
255 while !blocks.last!.used && 2 * blocks.last!.size == size {
256 released += blocks.removeLast().size
257 size /= 2
258 }
259 } else if blocks.count >= 2 && !blocks.last!.used {
260 let last = blocks.removeLast()
261 released = last.size
262 size = last.start
263 }
264 if rover >= size { rover = 0 }
265 return released
266 }
267 func compact() -> ([Move], Int) {
268 var moved: [Move] = []
269 var packed = usedBlocks
270 var address = 0
271 for block in packed {
272 if block.start != address {
273 moved.append(Move(block: block, from: block.payload))
274 block.start = address
275 }
276 address += block.size
277 }
278 if address < size { packed.append(register(Block(address, size - address))) }
279 blocks = packed
280 rover = 0
281 return (moved, address)
282 }
283}

285func experiment(_ mode: String, _ size: Int, _ seed: Int, _ operations: Int, _ maxRequest: Int) -> String {
286 let heap = Heap(mode, size)
287 var state = seed
288 var live: [Block] = []
289 var used = 0, peak = 0, successes = 0, failures = 0
290 for _ in 0..<operations {
291 state = nextRandom(state)
292 let allocating = live.isEmpty || state % 3 != 0
293 state = nextRandom(state)
294 if allocating {
295 let requested = 1 + state % maxRequest
296 if let block = heap.allocate(heap.requiredLength(requested), unit) {
297 successes += 1
298 live.append(block)
299 used += block.size
300 } else { failures += 1 }
301 } else {
302 let block = live.remove(at: state % live.count)
303 used -= block.size
304 _ = heap.release(block)
305 }
306 peak = max(peak, used)
307 }
308 return "BENCH \(mode) ok=\(successes) fail=\(failures) peak=\(peak) " +
309 "used=\(used) frag=\(heap.fragmentation) blocks=\(heap.blocks.count)"
310}

312final class Allocator {
313 var heap: Heap?
314 var allocations: [String: Block] = [:]
315 var freed: Set<String> = []
316 var roots: Set<String> = []
317 var references: [String: Set<String>] = [:]
318 var histories: [String: [String]] = [:]
319 var counters: [String: Int] = [:]
320 var peak = 0
321 var output: [String] = []
322
323 init() { reset(nil) }
324 func reset(_ replacement: Heap?) {
325 heap = replacement
326 allocations = [:]
327 freed = []
328 roots = []
329 references = [:]
330 histories = [:]
331 counters = Dictionary(uniqueKeysWithValues: counterNames.map { ($0, 0) })
332 peak = 0
333 }
334 func emit(_ parts: Any...) { output.append(parts.map { String(describing: $0) }.joined(separator: " ")) }
335 func record(_ id: String, _ line: String) {
336 emit(line)
337 histories[id, default: []].append(line)
338 }
339 func increment(_ key: String) { counters[key, default: 0] += 1 }
340 func outOfMemory(_ id: String) -> CommandError {
341 increment("failures")
342 return CommandError("OUT OF MEMORY \(id)")
343 }
344 func live(_ id: String) throws -> Block {
345 if let block = allocations[id] { return block }
346 if freed.contains(id) { throw CommandError("USE AFTER FREE \(id)") }
347 throw CommandError("UNKNOWN ID \(id)")
348 }
349 func range(_ id: String, _ offset: Int, _ length: Int) throws -> Block {
350 let block = try live(id)
351 if offset + length > block.requested { throw CommandError("OUT OF BOUNDS \(id)") }
352 return block
353 }
354 func requireFreeList() throws {
355 if heap!.buddy { throw CommandError("UNSUPPORTED IN BUDDY MODE") }
356 }
357 func reachable() -> Set<String> {
358 var found = roots
359 var pending = Array(roots)
360 while let name = pending.popLast() {
361 for target in references[name] ?? [] {
362 if found.insert(target).inserted { pending.append(target) }
363 }
364 }
365 return found
366 }
367 func allocateId(_ command: String, _ id: String, _ request: Int,
368 _ alignment: Int, _ fill: Int) throws {
369 if request == 0 { throw CommandError("BAD SIZE") }
370 if allocations[id] != nil { throw CommandError("DUPLICATE ID \(id)") }
371 guard let block = heap!.allocate(heap!.requiredLength(request), alignment) else {
372 throw outOfMemory(id)
373 }
374 block.owner = id
375 block.requested = request
376 block.contents = Array(repeating: fill, count: request)
377 allocations[id] = block
378 freed.remove(id)
379 increment("allocs")
380 record(id, "\(command) \(id) addr=\(block.payload) block=\(block.size)")
381 }
382 func releaseId(_ id: String) -> String {
383 let block = allocations.removeValue(forKey: id)!
384 let size = block.size
385 freed.insert(id)
386 roots.remove(id)
387 references.removeValue(forKey: id)
388 for key in references.keys { references[key]!.remove(id) }
389 let (result, merges) = heap!.release(block)
390 return "block=\(size) merges=\(merges) free=\(result.area)"
391 }
392 func freeId(_ id: String) throws {
393 if freed.contains(id) { throw CommandError("DOUBLE FREE \(id)") }
394 if allocations[id] == nil { throw CommandError("UNKNOWN ID \(id)") }
395 let details = releaseId(id)
396 increment("frees")
397 record(id, "FREE \(id) \(details)")
398 }
399 func reallocId(_ id: String, _ request: Int) throws {
400 if request == 0 { try freeId(id); return }
401 if freed.contains(id) { throw CommandError("USE AFTER FREE \(id)") }
402 guard var block = allocations[id] else {
403 try allocateId("MALLOC", id, request, unit, uninitialized)
404 return
405 }
406 let length = heap!.requiredLength(request)
407 var contents = Array(block.contents.prefix(request))
408 contents += Array(repeating: uninitialized, count: max(0, request - contents.count))
409 let original = block.payload
410 var kind = heap!.resize(block, length)
411 if kind == nil {
412 guard let moved = heap!.allocate(length, unit) else { throw outOfMemory(id) }
413 moved.owner = id
414 allocations[id] = moved
415 _ = heap!.release(block)
416 increment("moves")
417 block = moved
418 kind = "moved from=\(original)"
419 }
420 block.requested = request
421 block.contents = contents
422 increment("reallocs")
423 record(id, "REALLOC \(id) addr=\(block.payload) block=\(block.size) \(kind!)")
424 }
425 func treeNode(_ leaves: [Int: Block], _ start: Int, _ size: Int, _ depth: Int) {
426 let label = String(repeating: " ", count: depth * 2) + "\(start)+\(size)"
427 if let block = leaves[start], block.size == size {
428 emit(label + (block.used ? " USED \(block.owner!)" : " FREE"))
429 return
430 }
431 emit(label + " split")
432 treeNode(leaves, start, size / 2, depth + 1)
433 treeNode(leaves, start + size / 2, size / 2, depth + 1)
434 }
435 func run(_ command: String, _ a: [String]) throws {
436 if command == "HEAP" {
437 let size = Int(a[0])!, mode = a[1]
438 if size < 32 || size > maxHeap || size % unit != 0 ||
439 (mode == "BUDDY" && !isPowerOfTwo(size)) {
440 throw CommandError("BAD HEAP SIZE")
441 }
442 reset(Heap(mode, size))
443 emit("HEAP size=\(size) mode=\(mode)")
444 return
445 }
446 if command == "BENCH" {
447 let size = Int(a[0])!, seed = Int(a[1])!, operations = Int(a[2])!, maxRequest = Int(a[3])!
448 if size < 32 || size > maxHeap || !isPowerOfTwo(size) {
449 throw CommandError("BAD HEAP SIZE")
450 }
451 emit("BENCH heap=\(size) seed=\(seed) ops=\(operations) maxreq=\(maxRequest)")
452 for mode in modes { emit(experiment(mode, size, seed, operations, maxRequest)) }
453 return
454 }
455 let heap = self.heap!
456 switch command {
457 case "MALLOC":
458 try allocateId("MALLOC", a[0], Int(a[1])!, unit, uninitialized)
459 case "CALLOC":
460 try allocateId("CALLOC", a[0], Int(a[1])! * Int(a[2])!, unit, 0)
461 case "MEMALIGN":
462 try requireFreeList()
463 let alignment = Int(a[1])!
464 if !isPowerOfTwo(alignment) || alignment < unit || alignment > 4096 {
465 throw CommandError("BAD ALIGNMENT")
466 }
467 try allocateId("MEMALIGN", a[0], Int(a[2])!, alignment, uninitialized)
468 case "REALLOC":
469 try reallocId(a[0], Int(a[1])!)
470 case "FREE":
471 try freeId(a[0])
472 case "FILL":
473 let id = a[0], offset = Int(a[1])!, length = Int(a[2])!, value = Int(a[3])!
474 let block = try range(id, offset, length)
475 for i in offset..<(offset + length) { block.contents[i] = value }
476 emit("FILL \(id) \(offset)+\(length) = \(value)")
477 case "COPY":
478 let dest = a[0], doff = Int(a[1])!, source = a[2]
479 let soff = Int(a[3])!, length = Int(a[4])!
480 let dst = try range(dest, doff, length)
481 let src = try range(source, soff, length)
482 let saved = Array(src.contents[soff..<(soff + length)])
483 dst.contents.replaceSubrange(doff..<(doff + length), with: saved)
484 emit("COPY \(dest) \(doff)+\(length) from \(source) \(soff)")
485 case "READ":
486 let id = a[0], offset = Int(a[1])!, length = Int(a[2])!
487 let block = try range(id, offset, length)
488 let values = block.contents[offset..<(offset + length)].map {
489 $0 == uninitialized ? "?" : String($0)
490 }.joined(separator: " ")
491 emit("READ \(id) \(offset)+\(length) = \(values)")
492 case "ROOT":
493 _ = try live(a[0])
494 roots.insert(a[0])
495 emit("ROOT \(a[0])")
496 case "UNROOT":
497 _ = try live(a[0])
498 if !roots.contains(a[0]) { throw CommandError("NOT ROOT \(a[0])") }
499 roots.remove(a[0])
500 emit("UNROOT \(a[0])")
501 case "LINK":
502 _ = try live(a[0]); _ = try live(a[1])
503 references[a[0], default: []].insert(a[1])
504 emit("LINK \(a[0]) \(a[1])")
505 case "UNLINK":
506 _ = try live(a[0]); _ = try live(a[1])
507 if !(references[a[0]]?.contains(a[1]) ?? false) {
508 throw CommandError("NO LINK \(a[0]) \(a[1])")
509 }
510 references[a[0]]!.remove(a[1])
511 emit("UNLINK \(a[0]) \(a[1])")
512 case "GC":
513 let found = reachable()
514 let garbage = allocations.filter { !found.contains($0.key) }
515 .map { $0.value }.sorted { $0.start < $1.start }
516 emit("GC collected=\(garbage.count) bytes=\(garbage.reduce(0) { $0 + $1.size })")
517 for block in garbage {
518 let id = block.owner!
519 record(id, "COLLECT \(id) \(releaseId(id))")
520 }
521 counters["collected", default: 0] += garbage.count
522 case "SBRK":
523 let amount = Int(a[0])!
524 let valid = heap.buddy
525 ? amount == heap.size && 2 * heap.size <= maxHeap
526 : amount % unit == 0 && amount >= minBlock && heap.size + amount <= maxHeap
527 if !valid { throw CommandError("BAD SBRK SIZE") }
528 let block = heap.extend(amount)
529 emit("SBRK heap=\(heap.size) free=\(block.area)")
530 case "TRIM":
531 let released = heap.trim()
532 emit("TRIM heap=\(heap.size) released=\(released)")
533 case "COMPACT":
534 try requireFreeList()
535 let (moved, used) = heap.compact()
536 counters["moves", default: 0] += moved.count
537 let free = used < heap.size ? "\(used)+\(heap.size - used)" : "none"
538 emit("COMPACT moved=\(moved.count) free=\(free)")
539 for step in moved {
540 let id = step.block.owner!
541 record(id, "MOVE \(id) \(step.from) -> \(step.block.payload)")
542 }
543 case "WHO":
544 let address = Int(a[0])!
545 if address >= heap.size { emit("WHO \(address) outside") }
546 else {
547 let block = heap.blocks[heap.locate(address)]
548 if !block.used { emit("WHO \(address) free \(block.area)") }
549 else if address < block.payload { emit("WHO \(address) header \(block.owner!)") }
550 else {
551 let offset = address - block.payload
552 emit("WHO \(address) \(offset < block.requested ? "data" : "slack") \(block.owner!)+\(offset)")
553 }
554 }
555 case "DUMP":
556 let rover = heap.mode == "NEXT" ? " rover=\(heap.rover)" : ""
557 emit("DUMP mode=\(heap.mode) heap=\(heap.size) blocks=\(heap.blocks.count)\(rover)")
558 for block in heap.blocks {
559 if block.used {
560 emit("\(block.start) \(block.size) USED \(block.owner!) req=\(block.requested)")
561 } else { emit("\(block.start) \(block.size) FREE") }
562 }
563 case "MAP":
564 emit("MAP heap=\(heap.size)")
565 let units = heap.blocks.map(\.mapUnits).joined()
566 let chars = Array(units)
567 for first in stride(from: 0, to: chars.count, by: 64) {
568 let label = String(format: "%05d", first * unit)
569 emit("\(label) \(String(chars[first..<min(first + 64, chars.count)]))")
570 }
571 case "FREELIST":
572 var free = heap.freeBlocks
573 if heap.mode == "SEGFIT" { free.sort { $0.stamp > $1.stamp } }
574 emit("FREELIST count=\(free.count)")
575 var groups: [Int: [String]] = [:]
576 for block in free { groups[sizeClass(block.size), default: []].append(block.area) }
577 for exponent in groups.keys.sorted() {
578 let values = groups[exponent]!.joined(separator: " ")
579 emit("\(1 << exponent)-\((1 << (exponent + 1)) - 1): \(values)")
580 }
581 case "TREE":
582 if !heap.buddy { throw CommandError("REQUIRES BUDDY MODE") }
583 emit("TREE heap=\(heap.size)")
584 let leaves = Dictionary(uniqueKeysWithValues: heap.blocks.map { ($0.start, $0) })
585 treeNode(leaves, 0, heap.size, 0)
586 case "STATS":
587 let used = heap.usedBlocks, free = heap.freeBlocks
588 let totalFree = free.reduce(0) { $0 + $1.size }
589 let largest = free.map(\.size).max() ?? 0
590 let totalUsed = used.reduce(0) { $0 + $1.size }
591 let requested = used.reduce(0) { $0 + $1.requested }
592 emit("STATS used=\(totalUsed) used_blocks=\(used.count) free=\(totalFree) " +
593 "free_blocks=\(free.count) largest=\(largest) " +
594 "frag=\(fragmentationPercent(totalFree, largest)) requested=\(requested) peak=\(peak)")
595 let counts = counterNames.map { "\($0)=\(counters[$0, default: 0])" }.joined(separator: " ")
596 emit("OPS \(counts)")
597 case "LEAKS":
598 let found = reachable()
599 let names = allocations.keys.sorted()
600 let requested = names.reduce(0) { $0 + allocations[$1]!.requested }
601 let lost = names.filter { !found.contains($0) }.count
602 emit("LEAKS count=\(names.count) requested=\(requested) lost=\(lost)")
603 for id in names {
604 let block = allocations[id]!
605 emit("LEAK \(id) addr=\(block.payload) block=\(block.size) " +
606 "req=\(block.requested) \(found.contains(id) ? "reachable" : "lost")")
607 }
608 case "HISTORY":
609 let lines = histories[a[0]] ?? []
610 emit("HISTORY \(a[0]) events=\(lines.count)")
611 output.append(contentsOf: lines)
612 default:
613 fatalError("Unexpected command \(command)")
614 }
615 }
616 func execute(_ commandLine: String) {
617 let tokens = commandLine.split(separator: " ").map(String.init)
618 let command = tokens[0]
619 let args = Array(tokens.dropFirst())
620 do {
621 if heap == nil && command != "HEAP" && command != "BENCH" {
622 throw CommandError("NO HEAP")
623 }
624 try run(command, args)
625 } catch let error as CommandError {
626 emit("ERROR \(error.message)")
627 } catch { fatalError("Unexpected error: \(error)") }
628 if let heap { peak = max(peak, heap.usedTotal) }
629 }
630}
631
632let commandCount = Int(readLine()!)!
633let allocator = Allocator()
634for _ in 0..<commandCount { allocator.execute(readLine()!) }
635print(allocator.output.joined(separator: "\n"))
