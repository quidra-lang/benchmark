4local function power2(n)
5 return n > 0 and (n & (n - 1)) == 0
6end
7local function class(n)
8 local k = 0
9 while n >= 2 do
10 n = n // 2
11 k = k + 1
12 end
13 return k
14end

16Heap.__index = Heap
17function Heap:new(size, mode)
18 local h =
19 setmetatable(
20 {
21 size = size,
22 mode = mode,
23 blocks = {},
24 ids = {},
25 histories = {},
26 roots = {},
27 links = {},
28 age = 0,
29 rover = 0,
30 peak = 0,
31 ops = {0, 0, 0, 0, 0, 0}
32 },
33 self
34 )
35 h.blocks[1] = h:freeblock(0, size)
36 return h
37end
38function Heap:freeblock(a, size)
39 self.age = self.age + 1
40 return {a = a, size = size, age = self.age}
41end
42function Heap:sort()
43 table.sort(
44 self.blocks,
45 function(a, b)
46 return a.a < b.a
47 end
48 )
49end
50function Heap:index(block)
51 for i, b in ipairs(self.blocks) do
52 if b == block then
53 return i
54 end
55 end
56end
57function Heap:required(n)
58 local s = ((n + 7) // 8) * 8 + 8
59 if self.mode == "BUDDY" then
60 local p = 32
61 while p < s do
62 p = p * 2
63 end
64 return p
65 end
66 return s
67end
68function Heap:allocate(s, al)
69 local choices = {}
70 local start = 1
71 for i, b in ipairs(self.blocks) do
72 if b.a <= self.rover and self.rover < b.a + b.size then
73 start = i
74 end
75 if not b.used then
76 local p = ((b.a + 8 + al - 1) // al) * al - 8
77 if p - b.a > 0 and p - b.a < 16 then
78 p = p + al
79 end
80 if p + s <= b.a + b.size then
81 choices[#choices + 1] = {b = b, p = p, i = i}
82 end
83 end
84 end
85 if #choices == 0 then
86 return nil
87 end
88 local mode = self.mode
89 table.sort(
90 choices,
91 function(x, y)
92 local a, b = x.b, y.b
93 if mode == "NEXT" then
94 return (x.i - start + #self.blocks) % #self.blocks <
95 (y.i - start + #self.blocks) % #self.blocks
96 elseif mode == "BEST" or mode == "BUDDY" then
97 if a.size ~= b.size then
98 return a.size < b.size
99 end
100 elseif mode == "WORST" then
101 if a.size ~= b.size then
102 return a.size > b.size
103 end
104 elseif mode == "SEGFIT" then
105 if class(a.size) ~= class(b.size) then
106 return class(a.size) < class(b.size)
107 end
108 return a.age > b.age
109 end
110 return a.a < b.a
111 end
112 )
113 local c = choices[1]
114 local b, p = c.b, c.p
115 table.remove(self.blocks, c.i)
116 local used
117 if mode == "BUDDY" then
118 local z = b.size
119 while z > s do
120 z = z // 2
121 self.blocks[#self.blocks + 1] = self:freeblock(b.a + z, z)
122 end
123 used = {a = b.a, size = s, used = true}
124 else
125 if p > b.a then
126 self.blocks[#self.blocks + 1] = self:freeblock(b.a, p - b.a)
127 end
128 local z = b.a + b.size - p
129 used = {a = p, size = (z - s >= 16 and s or z), used = true}
130 if z - s >= 16 then
131 self.blocks[#self.blocks + 1] = self:freeblock(p + s, z - s)
132 end
133 end
134 self.blocks[#self.blocks + 1] = used
135 self:sort()
136 if mode == "NEXT" then
137 self.rover = (used.a + used.size) % self.size
138 end
139 return used
140end
141function Heap:release(b)
142 b.used, b.id, b.req, b.data = nil, nil, nil, nil
143 local merges = 0
144 if self.mode == "BUDDY" then
145 while b.size < self.size do
146 local buddy = b.a ~ b.size
147 local found
148 for i, c in ipairs(self.blocks) do
149 if c ~= b and not c.used and c.a == buddy and c.size == b.size then
150 found = i
151 break
152 end
153 end
154 if not found then
155 break
156 end
157 table.remove(self.blocks, found)
158 b.a, b.size = math.min(b.a, buddy), b.size * 2
159 merges = merges + 1
160 end
161 self:sort()
162 else
163 local i = self:index(b)
164 local nextb = self.blocks[i + 1]
165 if nextb and not nextb.used then
166 b.size = b.size + nextb.size
167 table.remove(self.blocks, i + 1)
168 merges = merges + 1
169 end
170 local prev = self.blocks[i - 1]
171 if prev and not prev.used then
172 b.a, b.size = prev.a, b.size + prev.size
173 table.remove(self.blocks, i - 1)
174 merges = merges + 1
175 end
176 end
177 self.age = self.age + 1
178 b.age = self.age
179 return b, merges
180end
181function Heap:record(id, line)
182 self.histories[id] = self.histories[id] or {}
183 table.insert(self.histories[id], line)
184 emit(line)
185end
186function Heap:state(id)
187 local b = self.ids[id]
188 if type(b) == "table" then
189 return b
190 end
191 emit("ERROR " .. (b == false and "USE AFTER FREE " or "UNKNOWN ID ") .. id)
192end
193function Heap:forget(id)
194 self.ids[id], self.roots[id], self.links[id] = false, nil, nil
195 for _, edges in pairs(self.links) do
196 edges[id] = nil
197 end
198end
199function Heap:freeid(id, kind)
200 local b = self.ids[id]
201 if type(b) ~= "table" then
202 emit("ERROR " .. (b == false and "DOUBLE FREE " or "UNKNOWN ID ") .. id)
203 return
204 end
205 local size = b.size
206 local f, m = self:release(b)
207 self:forget(id)
208 if kind == "FREE" then
209 self.ops[2] = self.ops[2] + 1
210 else
211 self.ops[6] = self.ops[6] + 1
212 end
213 self:record(
214 id,
215 kind ..
216 " " .. id .. " block=" .. size .. " merges=" .. m .. " free=" .. f.a .. "+" .. f.size
217 )
218end
219function Heap:malloc(id, n, al, kind)
220 if n == 0 then
221 emit("ERROR BAD SIZE")
222 return
223 end
224 if type(self.ids[id]) == "table" then
225 emit("ERROR DUPLICATE ID " .. id)
226 return
227 end
228 local b = self:allocate(self:required(n), al)
229 if not b then
230 self.ops[5] = self.ops[5] + 1
231 emit("ERROR OUT OF MEMORY " .. id)
232 return
233 end
234 b.id, b.req, b.data = id, n, {}
235 if kind == "CALLOC" then
236 for i = 1, n do
237 b.data[i] = 0
238 end
239 end
240 self.ids[id] = b
241 self.ops[1] = self.ops[1] + 1
242 self:record(id, kind .. " " .. id .. " addr=" .. (b.a + 8) .. " block=" .. b.size)
243end
244function Heap:realloc(id, n)
245 if n == 0 then
246 self:freeid(id, "FREE")
247 return
248 end
249 if self.ids[id] == nil then
250 self:malloc(id, n, 8, "MALLOC")
251 return
252 end
253 local b = self:state(id)
254 if not b then
255 return
256 end
257 local a, oldsize, oldreq, data = b.a, b.size, b.req, b.data
258 local s, kind = self:required(n), nil
259 if s <= b.size then
260 kind = "kept"
261 if self.mode == "BUDDY" then
262 if s < b.size then
263 kind = "shrunk"
264 end
265 while b.size > s do
266 b.size = b.size // 2
267 self.blocks[#self.blocks + 1] = self:freeblock(a + b.size, b.size)
268 end
269 self:sort()
270 elseif b.size - s >= 16 then
271 local i = self:index(b)
272 local z = b.size - s
273 b.size = s
274 local c = self.blocks[i + 1]
275 if c and not c.used then
276 z = z + c.size
277 table.remove(self.blocks, i + 1)
278 end
279 table.insert(self.blocks, i + 1, self:freeblock(a + s, z))
280 kind = "shrunk"
281 end
282 elseif self.mode ~= "BUDDY" then
283 local i = self:index(b)
284 local c = self.blocks[i + 1]
285 if c and not c.used and b.size + c.size >= s then
286 local z = b.size + c.size
287 table.remove(self.blocks, i + 1)
288 b.size = z - s >= 16 and s or z
289 if z - s >= 16 then
290 table.insert(self.blocks, i + 1, self:freeblock(a + s, z - s))
291 end
292 kind = "grown"
293 end
294 else
295 local absorbed = {}
296 local t = b.size
297 while t < s and a % (2 * t) == 0 do
298 local found
299 for _, c in ipairs(self.blocks) do
300 if not c.used and c.a == a + t and c.size == t then
301 found = c
302 break
303 end
304 end
305 if not found then
306 break
307 end
308 absorbed[#absorbed + 1] = found
309 t = t * 2
310 end
311 if t == s then
312 for _, c in ipairs(absorbed) do
313 table.remove(self.blocks, self:index(c))
314 end
315 b.size = s
316 kind = "grown"
317 end
318 end
319 if not kind then
320 local dest = self:allocate(s, 8)
321 if not dest then
322 self.ops[5] = self.ops[5] + 1
323 emit("ERROR OUT OF MEMORY " .. id)
324 return
325 end
326 self:release(b)
327 b = dest
328 b.id = id
329 self.ids[id] = b
330 kind = "moved"
331 self.ops[4] = self.ops[4] + 1
332 end
333 b.req, b.data = n, data
334 for i = n + 1, oldreq do
335 data[i] = nil
336 end
337 self.ops[3] = self.ops[3] + 1
338 local line = "REALLOC " .. id .. " addr=" .. (b.a + 8) .. " block=" .. b.size .. " " .. kind
339 if kind == "moved" then
340 line = line .. " from=" .. (a + 8)
341 end
342 self:record(id, line)
343end
344function Heap:reachable()
345 local seen, stack = {}, {}
346 for id in pairs(self.roots) do
347 stack[#stack + 1] = id
348 end
349 while #stack > 0 do
350 local id = table.remove(stack)
351 if not seen[id] then
352 seen[id] = true
353 for dst in pairs(self.links[id] or {}) do
354 stack[#stack + 1] = dst
355 end
356 end
357 end
358 return seen
359end
360function Heap:totals()
361 local u, f, l, ub, fb, r = 0, 0, 0, 0, 0, 0
362 for _, b in ipairs(self.blocks) do
363 if b.used then
364 u = u + b.size
365 ub = ub + 1
366 r = r + (b.req or 0)
367 else
368 f = f + b.size
369 l = math.max(l, b.size)
370 fb = fb + 1
371 end
372 end
373 return u, f, l, ub, fb, r, (f > 0 and (f - l) * 100 // f or 0)
374end

382 local h, x, live, ok, fail, peak = Heap:new(size, mode), seed, {}, 0, 0, 0
