5 size::Int
6 owner::String
7 requested::Int
8 data::Vector{Int}
9 stamp::Int
10end
11Block(a::Int, s::Int) = Block(a, s, "", 0, Int[], 0)
12used(b::Block) = !isempty(b.owner)
13area(b::Block) = "$(b.start)+$(b.size)"
14mutable struct Heap
15 size::Int
16 mode::String
17 rover::Int
18 age::Int
19 blocks::Vector{Block}
20end
21function register!(h::Heap, b::Block)
22 h.age += 1
23 b.stamp = h.age
24 b
25end
26function Heap(s::Int, m::String)
27 h = Heap(s, m, 0, 0, Block[])
28 push!(h.blocks, register!(h, Block(0, s)))
29 h
30end
31power2(x::Int) = x > 0 && (x & (x - 1)) == 0
32classof(x::Int) = ndigits(x; base=2) - 1
33function required(h::Heap, n::Int)
34 s = cld(n, 8) * 8 + 8
35 if h.mode == "BUDDY"
36 p = 32
37 while p < s
38 p *= 2
39 end
40 return p
41 end
42 s
43end
44usedtotal(h::Heap) = sum((b.size for b in h.blocks if used(b)); init=0)
45function frag(h::Heap)
46 sizes = [b.size for b in h.blocks if !used(b)]
47 f = sum(sizes)
48 f == 0 ? 0 : div((f - maximum(sizes)) * 100, f)
49end
50function halve!(h::Heap, i::Int, s::Int)
51 b = h.blocks[i]
52 while b.size > s
53 b.size = div(b.size, 2)
54 insert!(h.blocks, i + 1, register!(h, Block(b.start + b.size, b.size)))
55 end
56end
57function allocate!(h::Heap, s::Int, al::Int)
58 ri = findfirst(b -> b.start <= h.rover < b.start + b.size, h.blocks)
59 choices = Tuple{Tuple{Int,Int},Int,Int}[]
60 for (i, b) in enumerate(h.blocks)
61 used(b) && continue
62 p = cld(b.start + 8, al) * al - 8
63 if 0 < p - b.start < 16
64 p += al
65 end
66 p + s > b.start + b.size && continue
67 rank = if h.mode == "NEXT"
68 (mod(i - ri, length(h.blocks)), 0)
69 elseif h.mode in ("BEST", "BUDDY")
70 (b.size, b.start)
71 elseif h.mode == "WORST"
72 (-b.size, b.start)
73 elseif h.mode == "SEGFIT"
74 (classof(b.size), -b.stamp)
75 else
76 (b.start, 0)
77 end
78 push!(choices, (rank, i, p))
79 end
80 isempty(choices) && return nothing
81 _, i, p = minimum(choices)
82 if h.mode == "BUDDY"
83 halve!(h, i, s)
84 b = h.blocks[i]
85 else
86 old = h.blocks[i]
87 parts = Block[]
88 if p > old.start
89 push!(parts, register!(h, Block(old.start, p - old.start)))
90 end
91 b = Block(p, old.start + old.size - p)
92 push!(parts, b)
93 if b.size - s >= 16
94 push!(parts, register!(h, Block(p + s, b.size - s)))
95 b.size = s
96 end
97 splice!(h.blocks, i:i, parts)
98 end
99 b.owner = "@"
100 if h.mode == "NEXT"
101 h.rover = mod(b.start + b.size, h.size)
102 end
103 b
104end
105function coalesce!(h::Heap, i::Int)
106 merges = 0
107 if h.mode == "BUDDY"
108 while h.blocks[i].size < h.size
109 b = h.blocks[i]
110 a = xor(b.start, b.size)
111 j = findfirst(x -> !used(x) && x.start == a && x.size == b.size, h.blocks)
112 j === nothing && break
113 i = min(i, j)
114 h.blocks[i].size += h.blocks[i + 1].size
115 deleteat!(h.blocks, i + 1)
116 merges += 1
117 end
118 else
119 if i < length(h.blocks) && !used(h.blocks[i + 1])
120 h.blocks[i].size += h.blocks[i + 1].size
121 deleteat!(h.blocks, i + 1)
122 merges += 1
123 end
124 if i > 1 && !used(h.blocks[i - 1])
125 i -= 1
126 h.blocks[i].size += h.blocks[i + 1].size
127 deleteat!(h.blocks, i + 1)
128 merges += 1
129 end
130 end
131 register!(h, h.blocks[i]), merges
132end
133function release!(h::Heap, b::Block)
134 b.owner = ""
135 b.requested = 0
136 empty!(b.data)
137 coalesce!(h, findfirst(x -> x === b, h.blocks))
138end
139function resizeblock!(h::Heap, b::Block, s::Int)
140 i = findfirst(x -> x === b, h.blocks)
141 if s <= b.size
142 if h.mode == "BUDDY"
143 s == b.size && return "kept"
144 halve!(h, i, s)
145 else
146 b.size - s < 16 && return "kept"
147 insert!(h.blocks, i + 1, Block(b.start + s, b.size - s))
148 b.size = s
149 coalesce!(h, i + 1)
150 end
151 return "shrunk"
152 end
153 if h.mode == "BUDDY"
154 z = b.size
155 j = i
156 while z < s
157 j += 1
158 if mod(b.start, 2z) != 0 || j > length(h.blocks)
159 return nothing
160 end
161 p = h.blocks[j]
162 (used(p) || p.size != z) && return nothing
163 z *= 2
164 end
165 deleteat!(h.blocks, (i + 1):j)
166 b.size = s
167 return "grown"
168 end
169 if i < length(h.blocks) && !used(h.blocks[i + 1])
170 z = b.size + h.blocks[i + 1].size
171 if z >= s
172 deleteat!(h.blocks, i + 1)
173 b.size = z
174 if z - s >= 16
175 b.size = s
176 insert!(h.blocks, i + 1, register!(h, Block(b.start + s, z - s)))
177 end
178 return "grown"
179 end
180 end
181 nothing
182end
183mutable struct Allocator
184 heap::Union{Nothing,Heap}
185 live::Dict{String,Block}
186 freed::Set{String}
187 roots::Set{String}
188 refs::Dict{String,Set{String}}
189 histories::Dict{String,Vector{String}}
190 counts::Vector{Int}
191 peak::Int
192end
193Allocator(h=nothing) = Allocator(h, Dict{String,Block}(), Set{String}(), Set{String}(), Dict{String,Set{String}}(), Dict{String,Vector{String}}(), zeros(Int, 6), 0)
194struct CommandError <: Exception
195 text::String
196end
197fail(s) = throw(CommandError(s))
198function record!(a::Allocator, id::String, line::String)
199 println(line)
200 push!(get!(a.histories, id, String[]), line)
201end
202function live(a::Allocator, id::String)
203 haskey(a.live, id) && return a.live[id]
204 fail((id in a.freed ? "USE AFTER FREE " : "UNKNOWN ID ") * id)
205end
206function bounds(a::Allocator, id::String, off::Int, len::Int)
207 b = live(a, id)
208 off + len > b.requested && fail("OUT OF BOUNDS $id")
209 b
210end
211function reachable(a::Allocator)
212 seen = copy(a.roots)
213 todo = collect(seen)
214 while !isempty(todo)
215 id = pop!(todo)
216 for dst in get(a.refs, id, Set{String}())
217 if !(dst in seen)
218 push!(seen, dst)
219 push!(todo, dst)
220 end
221 end
222 end
223 seen
224end
225function malloc!(a::Allocator, word::String, id::String, n::Int, al::Int=8, value::Int=-1)
226 n == 0 && fail("BAD SIZE")
227 haskey(a.live, id) && fail("DUPLICATE ID $id")
228 b = allocate!(a.heap, required(a.heap, n), al)
229 if b === nothing
230 a.counts[5] += 1
231 fail("OUT OF MEMORY $id")
232 end
233 b.owner = id
234 b.requested = n
235 b.data = fill(value, n)
236 a.live[id] = b
237 delete!(a.freed, id)
238 a.counts[1] += 1
239 record!(a, id, "$word $id addr=$(b.start + 8) block=$(b.size)")
240end
241function releaseid!(a::Allocator, id::String)
242 b = pop!(a.live, id)
243 z = b.size
244 push!(a.freed, id)
245 delete!(a.roots, id)
246 delete!(a.refs, id)
247 for targets in values(a.refs)
248 delete!(targets, id)
249 end
250 f, m = release!(a.heap, b)
251 "block=$z merges=$m free=$(area(f))"
252end
253function freeid!(a::Allocator, id::String)
254 id in a.freed && fail("DOUBLE FREE $id")
255 haskey(a.live, id) || fail("UNKNOWN ID $id")
256 details = releaseid!(a, id)
257 a.counts[2] += 1
258 record!(a, id, "FREE $id $details")
259end
260function tree(h::Heap, start::Int, size::Int, depth::Int)
261 label = repeat(" ", 2depth) * "$start+$size"
262 i = findfirst(b -> b.start == start && b.size == size, h.blocks)
263 if i !== nothing
264 b = h.blocks[i]
265 println(label, used(b) ? " USED $(b.owner)" : " FREE")
266 else
267 println(label, " split")
268 half = div(size, 2)
269 tree(h, start, half, depth + 1)
270 tree(h, start + half, half, depth + 1)
271 end
272end
273function experiment(mode::String, size::Int, seed::Int, ops::Int, maxreq::Int)
274 h = Heap(size, mode)
275 x = seed
276 active = Block[]
277 ok = bad = peak = 0
278 for _ in 1:ops
279 x = mod(48271x, 2147483647)
280 alloc = isempty(active) || mod(x, 3) != 0
281 x = mod(48271x, 2147483647)
282 if alloc
283 b = allocate!(h, required(h, 1 + mod(x, maxreq)), 8)
284 if b === nothing
285 bad += 1
286 else
287 ok += 1
288 push!(active, b)
289 end
290 else
291 b = splice!(active, mod(x, length(active)) + 1)
292 release!(h, b)
293 end
294 peak = max(peak, usedtotal(h))
295 end
296 println("BENCH $mode ok=$ok fail=$bad peak=$peak used=$(usedtotal(h)) frag=$(frag(h)) blocks=$(length(h.blocks))")
297end
298function execute!(a::Allocator, words::Vector{String})
299 cmd = words[1]
300 v = words[2:end]
301 num(i) = parse(Int, v[i])
302 h = a.heap
303 if cmd == "HEAP"
304 s, m = num(1), v[2]
305 (s % 8 != 0 || !(32 <= s <= 65536) || (m == "BUDDY" && !power2(s))) && fail("BAD HEAP SIZE")
306 fresh = Allocator(Heap(s, m))
307 for name in fieldnames(Allocator)
308 setfield!(a, name, getfield(fresh, name))
309 end
310 println("HEAP size=$s mode=$m")
311 elseif cmd == "BENCH"
312 s, seed, ops, maxreq = num.(1:4)
313 (!power2(s) || !(32 <= s <= 65536)) && fail("BAD HEAP SIZE")
314 println("BENCH heap=$s seed=$seed ops=$ops maxreq=$maxreq")
315 for m in ("FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY")
316 experiment(m, s, seed, ops, maxreq)
317 end
318 elseif h === nothing
319 fail("NO HEAP")
320 elseif cmd in ("MALLOC", "CALLOC", "MEMALIGN")
321 al, n, value = 8, num(2), -1
322 if cmd == "CALLOC"
323 n = num(2) * num(3)
324 value = 0
325 elseif cmd == "MEMALIGN"
326 h.mode == "BUDDY" && fail("UNSUPPORTED IN BUDDY MODE")
327 al, n = num(2), num(3)
328 (!power2(al) || !(8 <= al <= 4096)) && fail("BAD ALIGNMENT")
329 end
330 malloc!(a, cmd, v[1], n, al, value)
331 elseif cmd == "FREE"
332 freeid!(a, v[1])
333 elseif cmd == "REALLOC"
334 id, n = v[1], num(2)
335 if n == 0
336 freeid!(a, id)
337 elseif id in a.freed
338 fail("USE AFTER FREE $id")
339 elseif !haskey(a.live, id)
340 malloc!(a, "MALLOC", id, n)
341 else
342 b = a.live[id]
343 data = fill(-1, n)
344 k = min(n, b.requested)
345 data[1:k] = b.data[1:k]
346 origin = b.start + 8
347 kind = resizeblock!(h, b, required(h, n))
348 if kind === nothing
349 moved = allocate!(h, required(h, n), 8)
350 if moved === nothing
351 a.counts[5] += 1
352 fail("OUT OF MEMORY $id")
353 end
354 moved.owner = id
355 a.live[id] = moved
356 release!(h, b)
357 b = moved
358 a.counts[4] += 1
359 kind = "moved from=$origin"
360 end
361 b.requested, b.data = n, data
362 a.counts[3] += 1
363 record!(a, id, "REALLOC $id addr=$(b.start + 8) block=$(b.size) $kind")
364 end
365 elseif cmd in ("FILL", "READ")
366 id, off, len = v[1], num(2), num(3)
367 b = bounds(a, id, off, len)
368 if cmd == "FILL"
369 b.data[(off + 1):(off + len)] .= num(4)
370 println("FILL $id $off+$len = $(num(4))")
371 else
372 text = join((x == -1 ? "?" : string(x) for x in b.data[(off + 1):(off + len)]), " ")
373 println("READ $id $off+$len = $text")
374 end
375 elseif cmd == "COPY"
376 dst, doff, src, soff, len = v[1], num(2), v[3], num(4), num(5)
377 d = bounds(a, dst, doff, len)
378 s = bounds(a, src, soff, len)
379 d.data[(doff + 1):(doff + len)] = copy(s.data[(soff + 1):(soff + len)])
380 println("COPY $dst $doff+$len from $src $soff")
381 elseif cmd in ("ROOT", "UNROOT")
382 id = v[1]
383 live(a, id)
384 if cmd == "ROOT"
385 push!(a.roots, id)
386 else
387 id in a.roots || fail("NOT ROOT $id")
388 delete!(a.roots, id)
389 end
390 println("$cmd $id")
391 elseif cmd in ("LINK", "UNLINK")
392 src, dst = v
393 live(a, src)
394 live(a, dst)
395 targets = get!(a.refs, src, Set{String}())
396 if cmd == "LINK"
397 push!(targets, dst)
398 else
399 dst in targets || fail("NO LINK $src $dst")
400 delete!(targets, dst)
401 end
402 println("$cmd $src $dst")
403 elseif cmd == "GC"
404 seen = reachable(a)
405 garbage = sort([b for (id, b) in a.live if !(id in seen)]; by=b -> b.start)
406 println("GC collected=$(length(garbage)) bytes=$(sum((b.size for b in garbage); init=0))")
407 for b in garbage
408 id = b.owner
409 record!(a, id, "COLLECT $id $(releaseid!(a, id))")
410 end
411 a.counts[6] += length(garbage)
412 elseif cmd == "SBRK"
413 k = num(1)
414 valid = h.mode == "BUDDY" ? k == h.size && 2h.size <= 65536 : k % 8 == 0 && k >= 16 && h.size + k <= 65536
415 valid || fail("BAD SBRK SIZE")
416 push!(h.blocks, Block(h.size, k))
417 h.size += k
418 f, _ = coalesce!(h, length(h.blocks))
419 println("SBRK heap=$(h.size) free=$(area(f))")
420 elseif cmd == "TRIM"
421 old = h.size
422 if h.mode == "BUDDY"
423 while !used(h.blocks[end]) && 2h.blocks[end].size == h.size
424 h.size -= pop!(h.blocks).size
425 end
426 elseif length(h.blocks) >= 2 && !used(h.blocks[end])
427 h.size = pop!(h.blocks).start
428 end
429 h.rover >= h.size && (h.rover = 0)
430 println("TRIM heap=$(h.size) released=$(old - h.size)")
431 elseif cmd == "COMPACT"
432 h.mode == "BUDDY" && fail("UNSUPPORTED IN BUDDY MODE")
433 packed = filter(used, h.blocks)
434 moves = Tuple{Block,Int}[]
435 pos = 0
436 for b in packed
437 if b.start != pos
438 push!(moves, (b, b.start + 8))
439 b.start = pos
440 end
441 pos += b.size
442 end
443 free = pos == h.size ? "none" : "$pos+$(h.size - pos)"
444 pos < h.size && push!(packed, register!(h, Block(pos, h.size - pos)))
445 h.blocks, h.rover = packed, 0
446 a.counts[4] += length(moves)
447 println("COMPACT moved=$(length(moves)) free=$free")
448 for (b, origin) in moves
449 record!(a, b.owner, "MOVE $(b.owner) $origin -> $(b.start + 8)")
450 end
451 elseif cmd == "WHO"
452 p = num(1)
453 if p >= h.size
454 println("WHO $p outside")
455 else
456 b = h.blocks[findfirst(b -> b.start <= p < b.start + b.size, h.blocks)]
457 if !used(b)
458 println("WHO $p free $(area(b))")
459 elseif p < b.start + 8
460 println("WHO $p header $(b.owner)")
461 else
462 d = p - b.start - 8
463 region = d < b.requested ? "data" : "slack"
464 println("WHO $p $region $(b.owner)+$d")
465 end
466 end
467 elseif cmd == "DUMP"
468 tail = h.mode == "NEXT" ? " rover=$(h.rover)" : ""
469 println("DUMP mode=$(h.mode) heap=$(h.size) blocks=$(length(h.blocks))$tail")
470 for b in h.blocks
471 println("$(b.start) $(b.size) ", used(b) ? "USED $(b.owner) req=$(b.requested)" : "FREE")
472 end
473 elseif cmd == "MAP"
474 println("MAP heap=$(h.size)")
475 chars = Char[]
476 for b in h.blocks
477 push!(chars, used(b) ? 'H' : 'F')
478 for k in 0:(div(b.size, 8) - 2)
479 push!(chars, used(b) ? (8k < b.requested ? '#' : '-') : '.')
480 end
481 end
482 for first in 1:64:length(chars)
483 @printf("%05d %s\n", (first - 1) * 8, join(chars[first:min(first + 63, end)]))
484 end
485 elseif cmd == "FREELIST"
486 free = filter(b -> !used(b), h.blocks)
487 h.mode == "SEGFIT" && sort!(free; by=b -> -b.stamp)
488 println("FREELIST count=$(length(free))")
489 for k in sort(unique(classof(b.size) for b in free))
490 println("$(2^k)-$(2^(k + 1) - 1): ", join((area(b) for b in free if classof(b.size) == k), " "))
491 end
492 elseif cmd == "TREE"
493 h.mode == "BUDDY" || fail("REQUIRES BUDDY MODE")
494 println("TREE heap=$(h.size)")
495 tree(h, 0, h.size, 0)
496 elseif cmd == "STATS"
497 u = filter(used, h.blocks)
498 f = filter(b -> !used(b), h.blocks)
499 ft = sum((b.size for b in f); init=0)
500 largest = maximum((b.size for b in f); init=0)
501 req = sum((b.requested for b in u); init=0)
502 println("STATS used=$(usedtotal(h)) used_blocks=$(length(u)) free=$ft free_blocks=$(length(f)) largest=$largest frag=$(frag(h)) requested=$req peak=$(a.peak)")
503 names = ("allocs", "frees", "reallocs", "moves", "failures", "collected")
504 println("OPS ", join(("$(names[i])=$(a.counts[i])" for i in 1:6), " "))
505 elseif cmd == "LEAKS"
506 seen = reachable(a)
507 ids = sort(collect(keys(a.live)))
508 req = sum((a.live[id].requested for id in ids); init=0)
509 lost = count(id -> !(id in seen), ids)
510 println("LEAKS count=$(length(ids)) requested=$req lost=$lost")
511 for id in ids
512 b = a.live[id]
513 status = id in seen ? "reachable" : "lost"
514 println("LEAK $id addr=$(b.start + 8) block=$(b.size) req=$(b.requested) $status")
515 end
516 elseif cmd == "HISTORY"
517 id = v[1]
518 lines = get(a.histories, id, String[])
519 println("HISTORY $id events=$(length(lines))")
520 foreach(println, lines)
521 end
522end
523function main()
524 a = Allocator()
525 n = parse(Int, readline())
526 for _ in 1:n
527 words = String.(split(readline()))
528 try
529 execute!(a, words)
530 catch e
531 e isa CommandError || rethrow()
532 println("ERROR ", e.text)
533 end
534 a.heap !== nothing && (a.peak = max(a.peak, usedtotal(a.heap)))
535 end
536end
537main()
