13bool power(int n) => n > 0 && (n & (n - 1)) == 0;
14int sizeClass(int n) => n.bitLength - 1;
15int random(int x) => x * 48271 % 2147483647;
16int sum(Iterable<int> values) => values.fold(0, (a, b) => a + b);

23class Block {
24 int start, size, requested = 0, stamp = 0;
25 bool used = false;
26 String owner = '';
27 List<int> contents = [];
28 Block(this.start, this.size);
29 int get end => start + size;
30 int get payload => start + 8;
31 String get area => '$start+$size';
32}
33
34class Heap {
35 final String mode;
36 int size, rover = 0, age = 0;
37 List<Block> blocks = [];
38 Heap(this.mode, this.size) {
39 blocks.add(register(Block(0, size)));
40 }
41 bool get buddy => mode == 'BUDDY';
42 Block register(Block b) {
43 b.stamp = ++age;
44 return b;
45 }
46
47 int required(int n) {
48 int s = (n + 7) ~/ 8 * 8 + 8;
49 if (!buddy) return s;
50 int p = 32;
51 while (p < s) {
52 p *= 2;
53 }
54 return p;
55 }
56
57 List<Block> get used => blocks.where((b) => b.used).toList();
58 List<Block> get free => blocks.where((b) => !b.used).toList();
59 int get total => sum(used.map((b) => b.size));
60 int get frag {
61 int f = sum(free.map((b) => b.size));
62 int l = free.fold(0, (v, b) => math.max(v, b.size));
63 return f == 0 ? 0 : (f - l) * 100 ~/ f;
64 }
65
66 int locate(int p) => blocks.indexWhere((b) => p < b.end);
67 Block? allocate(int s, int al) {
68 int ri = locate(rover), chosen = -1, placement = 0;
69 int bestA = 1 << 60, bestB = 1 << 60;
70 for (int i = 0; i < blocks.length; i++) {
71 Block b = blocks[i];
72 if (b.used) continue;
73 int p = (b.start + 8 + al - 1) ~/ al * al - 8;
74 if (p > b.start && p - b.start < 16) p += al;
75 if (p + s > b.end) continue;
76 int a, c;
77 switch (mode) {
78 case 'NEXT':
79 {
80 a = (i - ri + blocks.length) % blocks.length;
81 c = 0;
82 }
83 case 'BEST':
84 case 'BUDDY':
85 {
86 a = b.size;
87 c = b.start;
88 }
89 case 'WORST':
90 {
91 a = -b.size;
92 c = b.start;
93 }
94 case 'SEGFIT':
95 {
96 a = sizeClass(b.size);
97 c = -b.stamp;
98 }
99 default:
100 {
101 a = b.start;
102 c = 0;
103 }
104 }
105 if (a < bestA || a == bestA && c < bestB) {
106 chosen = i;
107 placement = p;
108 bestA = a;
109 bestB = c;
110 }
111 }
112 if (chosen < 0) return null;
113 Block b;
114 if (buddy) {
115 b = blocks[chosen];
116 halve(chosen, s);
117 } else {
118 Block old = blocks.removeAt(chosen);
119 List<Block> parts = [];
120 if (placement > old.start)
121 parts.add(register(Block(old.start, placement - old.start)));
122 b = Block(placement, old.end - placement);
123 parts.add(b);
124 if (b.size - s >= 16) {
125 parts.add(register(Block(placement + s, b.size - s)));
126 b.size = s;
127 }
128 blocks.insertAll(chosen, parts);
129 }
130 b.used = true;
131 if (mode == 'NEXT') rover = b.end == size ? 0 : b.end;
132 return b;
133 }
134
135 void halve(int i, int s) {
136 Block b = blocks[i];
137 while (b.size > s) {
138 b.size ~/= 2;
139 blocks.insert(i + 1, register(Block(b.end, b.size)));
140 }
141 }
142
143 void merge(int i) {
144 blocks[i].size += blocks.removeAt(i + 1).size;
145 }
146
147 (Block, int) coalesce(int i) {
148 int merges = 0;
149 if (buddy) {
150 while (blocks[i].size < size) {
151 Block b = blocks[i];
152 int j = b.start % (2 * b.size) == 0 ? i + 1 : i - 1;
153 if (j < 0 ||
154 j >= blocks.length ||
155 blocks[j].used ||
156 blocks[j].size != b.size)
157 break;
158 i = math.min(i, j);
159 merge(i);
160 merges++;
161 }
162 } else {
163 if (i + 1 < blocks.length && !blocks[i + 1].used) {
164 merge(i);
165 merges++;
166 }
167 if (i > 0 && !blocks[i - 1].used) {
168 i--;
169 merge(i);
170 merges++;
171 }
172 }
173 return (register(blocks[i]), merges);
174 }
175
176 (Block, int) release(Block b) {
177 b.used = false;
178 b.owner = '';
179 b.requested = 0;
180 b.contents = [];
181 return coalesce(blocks.indexOf(b));
182 }
183
184 String? resize(Block b, int s) {
185 int i = blocks.indexOf(b);
186 if (s <= b.size) {
187 if (buddy) {
188 if (s == b.size) return 'kept';
189 halve(i, s);
190 } else {
191 if (b.size - s < 16) return 'kept';
192 blocks.insert(i + 1, Block(b.start + s, b.size - s));
193 b.size = s;
194 coalesce(i + 1);
195 }
196 return 'shrunk';
197 }
198 if (buddy) {
199 int t = b.size, j = i;
200 while (t < s) {
201 j++;
202 if (b.start % (2 * t) != 0 ||
203 j >= blocks.length ||
204 blocks[j].used ||
205 blocks[j].size != t)
206 return null;
207 t *= 2;
208 }
209 blocks.removeRange(i + 1, j + 1);
210 b.size = s;
211 } else {
212 if (i + 1 >= blocks.length ||
213 blocks[i + 1].used ||
214 b.size + blocks[i + 1].size < s)
215 return null;
216 int total = b.size + blocks.removeAt(i + 1).size;
217 b.size = total - s >= 16 ? s : total;
218 if (total - s >= 16)
219 blocks.insert(i + 1, register(Block(b.end, total - s)));
220 }
221 return 'grown';
222 }
223
224 int trim() {
225 int old = size;
226 if (buddy) {
227 while (!blocks.last.used && blocks.last.size * 2 == size) {
228 blocks.removeLast();
229 size ~/= 2;
230 }
231 } else if (blocks.length > 1 && !blocks.last.used) {
232 size = blocks.removeLast().start;
233 }
234 if (rover >= size) rover = 0;
235 return old - size;
236 }
237}

239String experiment(String mode, int s, int seed, int ops, int maxreq) {
240 Heap h = Heap(mode, s);
241 List<Block> live = [];
242 int x = seed, ok = 0, fail = 0, peak = 0;
243 for (int i = 0; i < ops; i++) {
244 x = random(x);
245 bool alloc = live.isEmpty || x % 3 != 0;
246 x = random(x);
247 if (alloc) {
248 Block? b = h.allocate(h.required(1 + x % maxreq), 8);
249 if (b == null) {
250 fail++;
251 } else {
252 ok++;
253 live.add(b);
254 }
255 } else {
256 h.release(live.removeAt(x % live.length));
257 }
258 peak = math.max(peak, h.total);
259 }
260 return 'BENCH $mode ok=$ok fail=$fail peak=$peak used=${h.total} frag=${h.frag} blocks=${h.blocks.length}';
261}

263class Allocator {
264 Heap? heap;
265 final Map<String, Block> live = {};
266 final Set<String> freed = {}, roots = {};
267 final Map<String, Set<String>> links = {};
268 final Map<String, List<String>> history = {};
269 List<int> counters = List.filled(6, 0);
270 int peak = 0;
271 final List<String> out = [];
272 Never fail(String s) => throw CommandError(s);
273 void record(String id, String s) {
274 out.add(s);
275 (history[id] ??= []).add(s);
276 }
277
278 Block block(String id) =>
279 live[id] ??
280 fail(freed.contains(id) ? 'USE AFTER FREE $id' : 'UNKNOWN ID $id');
281 Block range(String id, int off, int len) {
282 Block b = block(id);
283 if (off + len > b.requested) fail('OUT OF BOUNDS $id');
284 return b;
285 }
286
287 Never oom(String id) {
288 counters[4]++;
289 fail('OUT OF MEMORY $id');
290 }
291
292 void allocate(String kind, String id, int n, [int al = 8, int fill = -1]) {
293 if (n == 0) fail('BAD SIZE');
294 if (live.containsKey(id)) fail('DUPLICATE ID $id');
295 Block b = heap!.allocate(heap!.required(n), al) ?? oom(id);
296 b.owner = id;
297 b.requested = n;
298 b.contents = List.filled(n, fill);
299 live[id] = b;
300 freed.remove(id);
301 counters[0]++;
302 record(id, '$kind $id addr=${b.payload} block=${b.size}');
303 }
304
305 String release(String id) {
306 Block b = live.remove(id)!;
307 int size = b.size;
308 freed.add(id);
309 roots.remove(id);
310 links.remove(id);
311 for (Set<String> targets in links.values) {
312 targets.remove(id);
313 }
314 final (f, merges) = heap!.release(b);
315 return 'block=$size merges=$merges free=${f.area}';
316 }
317
318 void free(String id) {
319 if (freed.contains(id)) fail('DOUBLE FREE $id');
320 if (!live.containsKey(id)) fail('UNKNOWN ID $id');
321 String detail = release(id);
322 counters[1]++;
323 record(id, 'FREE $id $detail');
324 }
325
326 void realloc(String id, int n) {
327 if (n == 0) {
328 free(id);
329 return;
330 }
331 if (freed.contains(id)) fail('USE AFTER FREE $id');
332 Block? original = live[id];
333 if (original == null) {
334 allocate('MALLOC', id, n);
335 return;
336 }
337 Block b = original;
338 List<int> contents = List.generate(
339 n,
340 (i) => i < original.requested ? original.contents[i] : -1,
341 );
342 int old = b.payload;
343 String? kind = heap!.resize(b, heap!.required(n));
344 if (kind == null) {
345 Block moved = heap!.allocate(heap!.required(n), 8) ?? oom(id);
346 moved.owner = id;
347 live[id] = moved;
348 heap!.release(b);
349 b = moved;
350 counters[3]++;
351 kind = 'moved from=$old';
352 }
353 b.requested = n;
354 b.contents = contents;
355 counters[2]++;
356 record(id, 'REALLOC $id addr=${b.payload} block=${b.size} $kind');
357 }
358
359 Set<String> reachable() {
360 Set<String> seen = {...roots};
361 List<String> stack = roots.toList();
362 while (stack.isNotEmpty) {
363 String id = stack.removeLast();
364 for (String dst in links[id] ?? <String>{}) {
365 if (seen.add(dst)) stack.add(dst);
366 }
367 }
368 return seen;
369 }
370
371 void tree(int a, int s, int depth) {
372 Block? b;
373 for (Block candidate in heap!.blocks) {
374 if (candidate.start == a && candidate.size == s) {
375 b = candidate;
376 break;
377 }
378 }
379 String prefix = ''.padLeft(depth * 2) + '$a+$s';
380 if (b != null) {
381 out.add(prefix + (b.used ? ' USED ${b.owner}' : ' FREE'));
382 } else {
383 out.add('$prefix split');
384 tree(a, s ~/ 2, depth + 1);
385 tree(a + s ~/ 2, s ~/ 2, depth + 1);
386 }
387 }
388
389 void execute(String line) {
390 List<String> t = line.split(' '), a = line.split(' ').sublist(1);
391 String cmd = t[0];
392 int n(int i) => int.parse(a[i]);
393 try {
394 if (heap == null && cmd != 'HEAP' && cmd != 'BENCH') fail('NO HEAP');
395 switch (cmd) {
396 case 'HEAP':
397 {
398 int s = n(0);
399 if (s < 32 ||
400 s > 65536 ||
401 s % 8 != 0 ||
402 a[1] == 'BUDDY' && !power(s))
403 fail('BAD HEAP SIZE');
404 heap = Heap(a[1], s);
405 live.clear();
406 freed.clear();
407 roots.clear();
408 links.clear();
409 history.clear();
410 counters = List.filled(6, 0);
411 peak = 0;
412 out.add('HEAP size=$s mode=${a[1]}');
413 }
414 case 'BENCH':
415 {
416 int s = n(0);
417 if (s < 32 || s > 65536 || !power(s)) fail('BAD HEAP SIZE');
418 out.add('BENCH heap=$s seed=${a[1]} ops=${a[2]} maxreq=${a[3]}');
419 for (String mode in modes) {
420 out.add(experiment(mode, s, n(1), n(2), n(3)));
421 }
422 }
423 case 'MALLOC':
424 {
425 allocate(cmd, a[0], n(1));
426 }
427 case 'CALLOC':
428 {
429 allocate(cmd, a[0], n(1) * n(2), 8, 0);
430 }
431 case 'MEMALIGN':
432 {
433 if (heap!.buddy) fail('UNSUPPORTED IN BUDDY MODE');
434 if (n(1) < 8 || n(1) > 4096 || !power(n(1))) fail('BAD ALIGNMENT');
435 allocate(cmd, a[0], n(2), n(1));
436 }
437 case 'REALLOC':
438 {
439 realloc(a[0], n(1));
440 }
441 case 'FREE':
442 {
443 free(a[0]);
444 }
445 case 'FILL':
446 {
447 Block b = range(a[0], n(1), n(2));
448 b.contents.fillRange(n(1), n(1) + n(2), n(3));
449 out.add('FILL ${a[0]} ${a[1]}+${a[2]} = ${a[3]}');
450 }
451 case 'COPY':
452 {
453 Block dst = range(a[0], n(1), n(4)), src = range(a[2], n(3), n(4));
454 List<int> copy = src.contents.sublist(n(3), n(3) + n(4));
455 dst.contents.setRange(n(1), n(1) + n(4), copy);
456 out.add('COPY ${a[0]} ${a[1]}+${a[4]} from ${a[2]} ${a[3]}');
457 }
458 case 'READ':
459 {
460 Block b = range(a[0], n(1), n(2));
461 String values = b.contents
462 .sublist(n(1), n(1) + n(2))
463 .map((v) => v < 0 ? '?' : '$v')
464 .join(' ');
465 out.add('READ ${a[0]} ${a[1]}+${a[2]} = $values');
466 }
467 case 'ROOT':
468 case 'UNROOT':
469 {
470 block(a[0]);
471 if (cmd == 'ROOT') {
472 roots.add(a[0]);
473 } else if (!roots.remove(a[0])) {
474 fail('NOT ROOT ${a[0]}');
475 }
476 out.add('$cmd ${a[0]}');
477 }
478 case 'LINK':
479 case 'UNLINK':
480 {
481 block(a[0]);
482 block(a[1]);
483 if (cmd == 'LINK') {
484 (links[a[0]] ??= {}).add(a[1]);
485 } else if (links[a[0]]?.remove(a[1]) != true) {
486 fail('NO LINK ${a[0]} ${a[1]}');
487 }
488 out.add('$cmd ${a[0]} ${a[1]}');
489 }
490 case 'GC':
491 {
492 Set<String> seen = reachable();
493 List<Block> garbage = heap!.used
494 .where((b) => !seen.contains(b.owner))
495 .toList();
496 out.add(
497 'GC collected=${garbage.length} bytes=${sum(garbage.map((b) => b.size))}',
498 );
499 for (Block b in garbage) {
500 String id = b.owner;
501 record(id, 'COLLECT $id ${release(id)}');
502 }
503 counters[5] += garbage.length;
504 }
505 case 'SBRK':
506 {
507 Heap h = heap!;
508 int k = n(0);
509 bool valid = h.buddy
510 ? k == h.size && h.size * 2 <= 65536
511 : k >= 16 && k % 8 == 0 && h.size + k <= 65536;
512 if (!valid) fail('BAD SBRK SIZE');
513 h.blocks.add(Block(h.size, k));
514 h.size += k;
515 Block f = h.coalesce(h.blocks.length - 1).$1;
516 out.add('SBRK heap=${h.size} free=${f.area}');
517 }
518 case 'TRIM':
519 {
520 int released = heap!.trim();
521 out.add('TRIM heap=${heap!.size} released=$released');
522 }
523 case 'COMPACT':
524 {
525 Heap h = heap!;
526 if (h.buddy) fail('UNSUPPORTED IN BUDDY MODE');
527 List<Block> used = h.used;
528 List<(Block, int)> moved = [];
529 int pos = 0;
530 for (Block b in used) {
531 if (b.start != pos) {
532 moved.add((b, b.payload));
533 b.start = pos;
534 }
535 pos += b.size;
536 }
537 if (pos < h.size) used.add(h.register(Block(pos, h.size - pos)));
538 h.blocks = used;
539 h.rover = 0;
540 counters[3] += moved.length;
541 String f = pos < h.size ? '$pos+${h.size - pos}' : 'none';
542 out.add('COMPACT moved=${moved.length} free=$f');
543 for (final (b, old) in moved) {
544 record(b.owner, 'MOVE ${b.owner} $old -> ${b.payload}');
545 }
546 }
547 case 'WHO':
548 {
549 Heap h = heap!;
550 int p = n(0);
551 String detail;
552 if (p >= h.size) {
553 detail = 'outside';
554 } else {
555 Block b = h.blocks[h.locate(p)];
556 if (!b.used) {
557 detail = 'free ${b.area}';
558 } else if (p < b.payload) {
559 detail = 'header ${b.owner}';
560 } else {
561 detail =
562 '${p - b.payload < b.requested ? 'data' : 'slack'} ${b.owner}+${p - b.payload}';
563 }
564 }
565 out.add('WHO $p $detail');
566 }
567 case 'DUMP':
568 {
569 Heap h = heap!;
570 out.add(
571 'DUMP mode=${h.mode} heap=${h.size} blocks=${h.blocks.length}${h.mode == 'NEXT' ? ' rover=${h.rover}' : ''}',
572 );
573 for (Block b in h.blocks) {
574 out.add(
575 '${b.start} ${b.size} ${b.used ? 'USED ${b.owner} req=${b.requested}' : 'FREE'}',
576 );
577 }
578 }
579 case 'MAP':
580 {
581 Heap h = heap!;
582 out.add('MAP heap=${h.size}');
583 StringBuffer units = StringBuffer();
584 for (Block b in h.blocks) {
585 units.write(b.used ? 'H' : 'F');
586 for (int i = 0; i < b.size ~/ 8 - 1; i++) {
587 units.write(
588 !b.used
589 ? '.'
590 : i * 8 < b.requested
591 ? '#'
592 : '-',
593 );
594 }
595 }
596 String all = units.toString();
597 for (int i = 0; i < all.length; i += 64) {
598 out.add(
599 '${(i * 8).toString().padLeft(5, '0')} ${all.substring(i, math.min(i + 64, all.length))}',
600 );
601 }
602 }
603 case 'FREELIST':
604 {
605 Heap h = heap!;
606 List<Block> free = h.free;
607 if (h.mode == 'SEGFIT')
608 free.sort((a, b) => b.stamp.compareTo(a.stamp));
609 out.add('FREELIST count=${free.length}');
610 Map<int, List<Block>> groups = {};
611 for (Block b in free) {
612 (groups[sizeClass(b.size)] ??= []).add(b);
613 }
614 for (int k in groups.keys.toList()..sort()) {
615 out.add(
616 '${1 << k}-${(1 << (k + 1)) - 1}: ${groups[k]!.map((b) => b.area).join(' ')}',
617 );
618 }
619 }
620 case 'TREE':
621 {
622 if (!heap!.buddy) fail('REQUIRES BUDDY MODE');
623 out.add('TREE heap=${heap!.size}');
624 tree(0, heap!.size, 0);
625 }
626 case 'STATS':
627 {
628 Heap h = heap!;
629 List<Block> used = h.used, free = h.free;
630 int largest = free.fold(0, (v, b) => math.max(v, b.size));
631 out.add(
632 'STATS used=${h.total} used_blocks=${used.length} free=${sum(free.map((b) => b.size))} free_blocks=${free.length} largest=$largest frag=${h.frag} requested=${sum(used.map((b) => b.requested))} peak=$peak',
633 );
634 out.add(
635 'OPS ${List.generate(6, (i) => '${counterNames[i]}=${counters[i]}').join(' ')}',
636 );
637 }
638 case 'LEAKS':
639 {
640 Set<String> seen = reachable();
641 out.add(
642 'LEAKS count=${live.length} requested=${sum(live.values.map((b) => b.requested))} lost=${live.keys.where((id) => !seen.contains(id)).length}',
643 );
644 for (String id in live.keys.toList()..sort()) {
645 Block b = live[id]!;
646 out.add(
647 'LEAK $id addr=${b.payload} block=${b.size} req=${b.requested} ${seen.contains(id) ? 'reachable' : 'lost'}',
648 );
649 }
650 }
651 case 'HISTORY':
652 {
653 List<String> events = history[a[0]] ?? [];
654 out.add('HISTORY ${a[0]} events=${events.length}');
655 out.addAll(events);
656 }
657 }
658 } on CommandError catch (e) {
659 out.add('ERROR ${e.message}');
660 }
661 if (heap != null) peak = math.max(peak, heap!.total);
662 }
663}

665void main() {
666 int count = int.parse(stdin.readLineSync()!);
667 Allocator allocator = Allocator();
668 for (int i = 0; i < count; i++) {
669 allocator.execute(stdin.readLineSync()!);
670 }
671 stdout.write('${allocator.out.join('\n')}\n');
672}
