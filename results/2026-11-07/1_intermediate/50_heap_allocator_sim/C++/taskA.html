35bool power_of_two(int n) {
36 return n > 0 && (n & (n - 1)) == 0;
37}
38int size_class(int n) {
39 int power = 0;
40 while ((1 << (power + 1)) <= n)
41 ++power;
42 return power;
43}
44int fragmentation_percent(int total, int largest) {
45 return total ? (total - largest) * 100 / total : 0;
46}
47int64_t next_random(int64_t state) {
48 return RANDOM_MULTIPLIER * state % RANDOM_MODULUS;
49}

51struct Block {
52 int start, size;
53 bool used = false;
54 string owner;
55 int requested = 0;
56 vector<int> contents;
57 int stamp = 0;
58
59 Block(int start, int size) : start(start), size(size) {
60 }
61 int end() const {
62 return start + size;
63 }
64 int payload() const {
65 return start + HEADER;
66 }
67 string area() const {
68 return to_string(start) + "+" + to_string(size);
69 }
70 string map_units() const {
71 string units(1, used ? 'H' : 'F');
72 for (int i = 0; i < size / UNIT - 1; ++i) {
73 units += !used ? '.' : i * UNIT < requested ? '#' : '-';
74 }
75 return units;
76 }
77};

90struct Heap {
91 string mode;
92 int size, rover = 0, registrations = 0;
93 vector<BlockRef> blocks;
94
95 Heap(string mode, int size) : mode(move(mode)), size(size) {
96 blocks.push_back(reg(make_shared<Block>(0, size)));
97 }
98 BlockRef reg(BlockRef block) {
99 block->stamp = ++registrations;
100 return block;
101 }
102 bool buddy() const {
103 return mode == "BUDDY";
104 }
105 int required_length(int request) const {
106 int length = ((request + UNIT - 1) / UNIT) * UNIT + HEADER;
107 if (!buddy())
108 return length;
109 int power = MIN_BUDDY_BLOCK;
110 while (power < length)
111 power *= 2;
112 return power;
113 }
114 int locate(int address) const {
115 int index = 0;
116 while (blocks.at(index)->end() <= address)
117 ++index;
118 return index;
119 }
120 int index_of(const BlockRef &block) const {
121 return static_cast<int>(find(blocks.begin(), blocks.end(), block) - blocks.begin());
122 }
123 vector<BlockRef> used_blocks() const {
124 vector<BlockRef> selected;
125 for (const auto &block : blocks)
126 if (block->used)
127 selected.push_back(block);
128 return selected;
129 }
130 vector<BlockRef> free_blocks() const {
131 vector<BlockRef> selected;
132 for (const auto &block : blocks)
133 if (!block->used)
134 selected.push_back(block);
135 return selected;
136 }
137 int used_total() const {
138 int total = 0;
139 for (const auto &block : blocks)
140 if (block->used)
141 total += block->size;
142 return total;
143 }
144 int fragmentation() const {
145 int total = 0, largest = 0;
146 for (const auto &block : blocks)
147 if (!block->used) {
148 total += block->size;
149 largest = max(largest, block->size);
150 }
151 return fragmentation_percent(total, largest);
152 }
153 int placement(const Block &block, int alignment) const {
154 int place = ((block.start + HEADER + alignment - 1) / alignment) * alignment - HEADER;
155 if (place > block.start && place - block.start < MIN_BLOCK)
156 place += alignment;
157 return place;
158 }
159 pair<int, int> rank(int index, int rover_index) const {
160 const Block &block = *blocks[index];
161 if (mode == "NEXT")
162 return {(index - rover_index + (int)blocks.size()) % (int)blocks.size(), 0};
163 if (mode == "BEST" || mode == "BUDDY")
164 return {block.size, block.start};
165 if (mode == "WORST")
166 return {-block.size, block.start};
167 if (mode == "SEGFIT")
168 return {size_class(block.size), -block.stamp};
169 return {block.start, 0};
170 }
171 optional<Choice> choose(int length, int alignment) const {
172 int rover_index = locate(rover);
173 optional<Choice> chosen;
174 for (int index = 0; index < (int)blocks.size(); ++index) {
175 const Block &block = *blocks[index];
176 if (block.used)
177 continue;
178 int place = placement(block, alignment);
179 if (place + length > block.end())
180 continue;
181 pair<int, int> r = rank(index, rover_index);
182 if (!chosen || r < chosen->rank)
183 chosen = Choice{r, index, place};
184 }
185 return chosen;
186 }
187 BlockRef carve(int index, int place, int length) {
188 BlockRef original = blocks[index];
189 vector<BlockRef> parts;
190 if (place > original->start)
191 parts.push_back(reg(make_shared<Block>(original->start, place - original->start)));
192 BlockRef used = make_shared<Block>(place, original->end() - place);
193 parts.push_back(used);
194 if (used->size - length >= MIN_BLOCK) {
195 parts.push_back(reg(make_shared<Block>(place + length, used->size - length)));
196 used->size = length;
197 }
198 blocks.erase(blocks.begin() + index);
199 blocks.insert(blocks.begin() + index, parts.begin(), parts.end());
200 return used;
201 }
202 BlockRef halve(int index, int length) {
203 BlockRef block = blocks[index];
204 while (block->size > length) {
205 block->size /= 2;
206 blocks.insert(blocks.begin() + index + 1,
207 reg(make_shared<Block>(block->end(), block->size)));
208 }
209 return block;
210 }
211 BlockRef allocate(int length, int alignment) {
212 optional<Choice> chosen = choose(length, alignment);
213 if (!chosen)
214 return nullptr;
215 BlockRef block =
216 buddy() ? halve(chosen->index, length) : carve(chosen->index, chosen->place, length);
217 block->used = true;
218 if (mode == "NEXT")
219 rover = block->end() == size ? 0 : block->end();
220 return block;
221 }
222 void merge_next(int index) {
223 blocks[index]->size += blocks[index + 1]->size;
224 blocks.erase(blocks.begin() + index + 1);
225 }
226 pair<BlockRef, int> coalesce(int index) {
227 int merges = 0;
228 if (buddy()) {
229 while (blocks[index]->size < size) {
230 const BlockRef block = blocks[index];
231 int partner = block->start % (2 * block->size) == 0 ? index + 1 : index - 1;
232 if (partner < 0 || partner >= (int)blocks.size())
233 break;
234 const BlockRef other = blocks[partner];
235 if (other->used || other->size != block->size)
236 break;
237 index = min(index, partner);
238 merge_next(index);
239 ++merges;
240 }
241 } else {
242 if (index + 1 < (int)blocks.size() && !blocks[index + 1]->used) {
243 merge_next(index);
244 ++merges;
245 }
246 if (index > 0 && !blocks[index - 1]->used) {
247 --index;
248 merge_next(index);
249 ++merges;
250 }
251 }
252 return {reg(blocks[index]), merges};
253 }
254 pair<BlockRef, int> release(BlockRef block) {
255 block->used = false;
256 block->owner.clear();
257 block->requested = 0;
258 block->contents.clear();
259 return coalesce(index_of(block));
260 }
261 string shrink(int index, int length) {
262 BlockRef block = blocks[index];
263 if (buddy()) {
264 if (block->size == length)
265 return "kept";
266 halve(index, length);
267 return "shrunk";
268 }
269 if (block->size - length < MIN_BLOCK)
270 return "kept";
271 blocks.insert(blocks.begin() + index + 1,
272 make_shared<Block>(block->start + length, block->size - length));
273 block->size = length;
274 coalesce(index + 1);
275 return "shrunk";
276 }
277 string grow_into_next(int index, int length) {
278 BlockRef block = blocks[index];
279 if (index + 1 >= (int)blocks.size() || blocks[index + 1]->used)
280 return "";
281 int total = block->size + blocks[index + 1]->size;
282 if (total < length)
283 return "";
284 blocks.erase(blocks.begin() + index + 1);
285 if (total - length >= MIN_BLOCK) {
286 block->size = length;
287 blocks.insert(blocks.begin() + index + 1,
288 reg(make_shared<Block>(block->end(), total - length)));
289 } else
290 block->size = total;
291 return "grown";
292 }
293 string grow_buddy(int index, int length) {
294 BlockRef block = blocks[index];
295 int size = block->size, last = index;
296 while (size < length) {
297 ++last;
298 if (block->start % (2 * size) != 0 || last >= (int)blocks.size())
299 return "";
300 BlockRef partner = blocks[last];
301 if (partner->used || partner->size != size)
302 return "";
303 size *= 2;
304 }
305 blocks.erase(blocks.begin() + index + 1, blocks.begin() + last + 1);
306 block->size = length;
307 return "grown";
308 }
309 string resize(BlockRef block, int length) {
310 int index = index_of(block);
311 if (length <= block->size)
312 return shrink(index, length);
313 return buddy() ? grow_buddy(index, length) : grow_into_next(index, length);
314 }
315 BlockRef extend(int amount) {
316 blocks.push_back(make_shared<Block>(size, amount));
317 size += amount;
318 return coalesce((int)blocks.size() - 1).first;
319 }
320 int trim() {
321 int released = 0;
322 if (buddy()) {
323 while (!blocks.back()->used && 2 * blocks.back()->size == size) {
324 released += blocks.back()->size;
325 blocks.pop_back();
326 size /= 2;
327 }
328 } else if (blocks.size() >= 2 && !blocks.back()->used) {
329 BlockRef block = blocks.back();
330 blocks.pop_back();
331 released = block->size;
332 size = block->start;
333 }
334 if (rover >= size)
335 rover = 0;
336 return released;
337 }
338 pair<vector<Moved>, int> compact() {
339 vector<BlockRef> packed = used_blocks();
340 vector<Moved> moved;
341 int address = 0;
342 for (auto &block : packed) {
343 if (block->start != address) {
344 moved.push_back({block, block->payload()});
345 block->start = address;
346 }
347 address += block->size;
348 }
349 if (address < size)
350 packed.push_back(reg(make_shared<Block>(address, size - address)));
351 blocks = move(packed);
352 rover = 0;
353 return {moved, address};
354 }
355};

357string experiment(const string &mode, int size, int64_t seed, int operations, int max_request) {
358 Heap heap(mode, size);
359 int64_t state = seed;
360 vector<BlockRef> live;
361 int used = 0, peak = 0, success = 0, failures = 0;
362 for (int operation = 0; operation < operations; ++operation) {
363 state = next_random(state);
364 bool allocating = live.empty() || state % 3 != 0;
365 state = next_random(state);
366 if (allocating) {
367 int requested = 1 + (int)(state % max_request);
368 BlockRef block = heap.allocate(heap.required_length(requested), UNIT);
369 if (!block)
370 ++failures;
371 else {
372 ++success;
373 live.push_back(block);
374 used += block->size;
375 }
376 } else {
377 size_t index = (size_t)(state % live.size());
378 BlockRef block = live[index];
379 live.erase(live.begin() + index);
380 used -= block->size;
381 heap.release(block);
382 }
383 peak = max(peak, used);
384 }
385 ostringstream out;
386 out << "BENCH " << mode << " ok=" << success << " fail=" << failures << " peak=" << peak
387 << " used=" << used << " frag=" << heap.fragmentation() << " blocks=" << heap.blocks.size();
388 return out.str();
389}

391struct Allocator {
392 unique_ptr<Heap> heap;
393 unordered_map<string, BlockRef> allocations;
394 unordered_set<string> freed, roots;
395 unordered_map<string, unordered_set<string>> references;
396 unordered_map<string, vector<string>> histories;
397 unordered_map<string, int> counters;
398 int peak = 0;
399 vector<string> output;
400
401 Allocator() {
402 reset(nullptr);
403 }
404 void reset(unique_ptr<Heap> next_heap) {
405 heap = move(next_heap);
406 allocations.clear();
407 freed.clear();
408 roots.clear();
409 references.clear();
410 histories.clear();
411 counters.clear();
412 for (const string &name : COUNTERS)
413 counters[name] = 0;
414 peak = 0;
415 }
416 template <typename... Parts> void emit(const Parts &...parts) {
417 ostringstream text;
418 bool first = true;
419 ((text << (exchange(first, false) ? "" : " ") << parts), ...);
420 output.push_back(text.str());
421 }
422 void record(const string &name, const string &line) {
423 emit(line);
424 histories[name].push_back(line);
425 }
426 CommandError out_of_memory(const string &id) {
427 ++counters["failures"];
428 return CommandError("OUT OF MEMORY " + id);
429 }
430 BlockRef live(const string &id) const {
431 auto it = allocations.find(id);
432 if (it != allocations.end())
433 return it->second;
434 if (freed.count(id))
435 throw CommandError("USE AFTER FREE " + id);
436 throw CommandError("UNKNOWN ID " + id);
437 }
438 BlockRef range(const string &id, int offset, int length) const {
439 BlockRef block = live(id);
440 if (offset + length > block->requested)
441 throw CommandError("OUT OF BOUNDS " + id);
442 return block;
443 }
444 void require_free_list() const {
445 if (heap->buddy())
446 throw CommandError("UNSUPPORTED IN BUDDY MODE");
447 }
448 unordered_set<string> reachable() const {
449 unordered_set<string> found = roots;
450 vector<string> pending(roots.begin(), roots.end());
451 while (!pending.empty()) {
452 string name = pending.back();
453 pending.pop_back();
454 auto it = references.find(name);
455 if (it == references.end())
456 continue;
457 for (const string &target : it->second) {
458 if (found.insert(target).second)
459 pending.push_back(target);
460 }
461 }
462 return found;
463 }
464 void allocate_id(const string &kind, const string &id, int requested, int alignment,
465 int initial) {
466 if (requested == 0)
467 throw CommandError("BAD SIZE");
468 if (allocations.count(id))
469 throw CommandError("DUPLICATE ID " + id);
470 BlockRef block = heap->allocate(heap->required_length(requested), alignment);
471 if (!block)
472 throw out_of_memory(id);
473 block->owner = id;
474 block->requested = requested;
475 block->contents.assign(requested, initial);
476 allocations[id] = block;
477 freed.erase(id);
478 ++counters["allocs"];
479 record(id, kind + " " + id + " addr=" + to_string(block->payload()) +
480 " block=" + to_string(block->size));
481 }
482 string release_id(const string &id) {
483 BlockRef block = allocations.at(id);
484 int size = block->size;
485 allocations.erase(id);
486 freed.insert(id);
487 roots.erase(id);
488 references.erase(id);
489 for (auto &[source, targets] : references)
490 targets.erase(id);
491 auto [free, merges] = heap->release(block);
492 return "block=" + to_string(size) + " merges=" + to_string(merges) +
493 " free=" + free->area();
494 }
495 void free_id(const string &id) {
496 if (freed.count(id))
497 throw CommandError("DOUBLE FREE " + id);
498 if (!allocations.count(id))
499 throw CommandError("UNKNOWN ID " + id);
500 string details = release_id(id);
501 ++counters["frees"];
502 record(id, "FREE " + id + " " + details);
503 }
504 void realloc_id(const string &id, int request) {
505 if (request == 0) {
506 free_id(id);
507 return;
508 }
509 if (freed.count(id))
510 throw CommandError("USE AFTER FREE " + id);
511 if (!allocations.count(id)) {
512 allocate_id("MALLOC", id, request, UNIT, UNINITIALIZED);
513 return;
514 }
515 BlockRef block = allocations.at(id);
516 int length = heap->required_length(request);
517 vector<int> contents = block->contents;
518 contents.resize(request, UNINITIALIZED);
519 int original = block->payload();
520 string kind = heap->resize(block, length);
521 if (kind.empty()) {
522 BlockRef moved = heap->allocate(length, UNIT);
523 if (!moved)
524 throw out_of_memory(id);
525 moved->owner = id;
526 allocations[id] = moved;
527 heap->release(block);
528 ++counters["moves"];
529 block = moved;
530 kind = "moved from=" + to_string(original);
531 }
532 block->requested = request;
533 block->contents = move(contents);
534 ++counters["reallocs"];
535 record(id, "REALLOC " + id + " addr=" + to_string(block->payload()) +
536 " block=" + to_string(block->size) + " " + kind);
537 }
538 void tree_node(const map<int, BlockRef> &leaves, int start, int size, int depth) {
539 string label(depth * 2, ' ');
540 label += to_string(start) + "+" + to_string(size);
541 auto it = leaves.find(start);
542 if (it != leaves.end() && it->second->size == size) {
543 BlockRef block = it->second;
544 emit(label, block->used ? "USED" : "FREE");
545 if (block->used) {
546 output.back() += " " + block->owner;
547 }
548 return;
549 }
550 emit(label, "split");
551 tree_node(leaves, start, size / 2, depth + 1);
552 tree_node(leaves, start + size / 2, size / 2, depth + 1);
553 }
554 void run(const string &name, const vector<string> &v) {
555 if (name == "HEAP") {
556 int size = stoi(v[0]);
557 const string &mode = v[1];
558 if (size % UNIT != 0 || size < 32 || size > MAX_HEAP ||
559 (mode == "BUDDY" && !power_of_two(size)))
560 throw CommandError("BAD HEAP SIZE");
561 reset(make_unique<Heap>(mode, size));
562 emit("HEAP size=" + to_string(size) + " mode=" + mode);
563 } else if (name == "MALLOC") {
564 allocate_id("MALLOC", v[0], stoi(v[1]), UNIT, UNINITIALIZED);
565 } else if (name == "CALLOC") {
566 allocate_id("CALLOC", v[0], stoi(v[1]) * stoi(v[2]), UNIT, 0);
567 } else if (name == "MEMALIGN") {
568 require_free_list();
569 int align = stoi(v[1]);
570 if (!power_of_two(align) || align < UNIT || align > 4096)
571 throw CommandError("BAD ALIGNMENT");
572 allocate_id("MEMALIGN", v[0], stoi(v[2]), align, UNINITIALIZED);
573 } else if (name == "REALLOC") {
574 realloc_id(v[0], stoi(v[1]));
575 } else if (name == "FREE") {
576 free_id(v[0]);
577 } else if (name == "FILL") {
578 const string &id = v[0];
579 int offset = stoi(v[1]), length = stoi(v[2]), value = stoi(v[3]);
580 BlockRef block = range(id, offset, length);
581 fill(block->contents.begin() + offset, block->contents.begin() + offset + length,
582 value);
583 emit("FILL", id, to_string(offset) + "+" + to_string(length), "=", value);
584 } else if (name == "COPY") {
585 const string &dest = v[0], &source = v[2];
586 int doff = stoi(v[1]), soff = stoi(v[3]), length = stoi(v[4]);
587 BlockRef destination = range(dest, doff, length);
588 BlockRef origin = range(source, soff, length);
589 vector<int> saved(origin->contents.begin() + soff,
590 origin->contents.begin() + soff + length);
591 copy(saved.begin(), saved.end(), destination->contents.begin() + doff);
592 emit("COPY", dest, to_string(doff) + "+" + to_string(length), "from", source, soff);
593 } else if (name == "READ") {
594 const string &id = v[0];
595 int offset = stoi(v[1]), length = stoi(v[2]);
596 BlockRef block = range(id, offset, length);
597 ostringstream values;
598 for (int i = 0; i < length; ++i) {
599 if (i)
600 values << ' ';
601 int value = block->contents[offset + i];
602 if (value == UNINITIALIZED)
603 values << '?';
604 else
605 values << value;
606 }
607 emit("READ", id, to_string(offset) + "+" + to_string(length), "=", values.str());
608 } else if (name == "ROOT") {
609 live(v[0]);
610 roots.insert(v[0]);
611 emit("ROOT", v[0]);
612 } else if (name == "UNROOT") {
613 live(v[0]);
614 if (!roots.count(v[0]))
615 throw CommandError("NOT ROOT " + v[0]);
616 roots.erase(v[0]);
617 emit("UNROOT", v[0]);
618 } else if (name == "LINK") {
619 live(v[0]);
620 live(v[1]);
621 references[v[0]].insert(v[1]);
622 emit("LINK", v[0], v[1]);
623 } else if (name == "UNLINK") {
624 live(v[0]);
625 live(v[1]);
626 if (!references[v[0]].count(v[1]))
627 throw CommandError("NO LINK " + v[0] + " " + v[1]);
628 references[v[0]].erase(v[1]);
629 emit("UNLINK", v[0], v[1]);
630 } else if (name == "GC") {
631 auto found = reachable();
632 vector<BlockRef> garbage;
633 for (const auto &[id, block] : allocations)
634 if (!found.count(id))
635 garbage.push_back(block);
636 sort(garbage.begin(), garbage.end(),
637 [](auto &a, auto &b) { return a->start < b->start; });
638 int bytes = 0;
639 for (const auto &block : garbage)
640 bytes += block->size;
641 emit("GC collected=" + to_string(garbage.size()) + " bytes=" + to_string(bytes));
642 for (const auto &block : garbage) {
643 string id = block->owner;
644 record(id, "COLLECT " + id + " " + release_id(id));
645 }
646 counters["collected"] += (int)garbage.size();
647 } else if (name == "SBRK") {
648 int amount = stoi(v[0]);
649 bool valid = heap->buddy() ? amount == heap->size && 2 * heap->size <= MAX_HEAP
650 : amount % UNIT == 0 && amount >= MIN_BLOCK &&
651 heap->size + amount <= MAX_HEAP;
652 if (!valid)
653 throw CommandError("BAD SBRK SIZE");
654 BlockRef free = heap->extend(amount);
655 emit("SBRK heap=" + to_string(heap->size) + " free=" + free->area());
656 } else if (name == "TRIM") {
657 int released = heap->trim();
658 emit("TRIM heap=" + to_string(heap->size) + " released=" + to_string(released));
659 } else if (name == "COMPACT") {
660 require_free_list();
661 auto [moved, used] = heap->compact();
662 counters["moves"] += (int)moved.size();
663 string free =
664 used < heap->size ? to_string(used) + "+" + to_string(heap->size - used) : "none";
665 emit("COMPACT moved=" + to_string(moved.size()) + " free=" + free);
666 for (auto &step : moved) {
667 BlockRef block = step.block;
668 record(block->owner, "MOVE " + block->owner + " " + to_string(step.from) + " -> " +
669 to_string(block->payload()));
670 }
671 } else if (name == "WHO") {
672 int address = stoi(v[0]);
673 if (address >= heap->size) {
674 emit("WHO", address, "outside");
675 return;
676 }
677 BlockRef block = heap->blocks[heap->locate(address)];
678 if (!block->used) {
679 emit("WHO", address, "free", block->area());
680 return;
681 }
682 if (address < block->payload()) {
683 emit("WHO", address, "header", block->owner);
684 return;
685 }
686 int offset = address - block->payload();
687 emit("WHO", address, offset < block->requested ? "data" : "slack",
688 block->owner + "+" + to_string(offset));
689 } else if (name == "DUMP") {
690 string line = "DUMP mode=" + heap->mode + " heap=" + to_string(heap->size) +
691 " blocks=" + to_string(heap->blocks.size());
692 if (heap->mode == "NEXT")
693 line += " rover=" + to_string(heap->rover);
694 emit(line);
695 for (auto &block : heap->blocks) {
696 if (block->used)
697 emit(block->start, block->size, "USED", block->owner,
698 "req=" + to_string(block->requested));
699 else
700 emit(block->start, block->size, "FREE");
701 }
702 } else if (name == "MAP") {
703 emit("MAP heap=" + to_string(heap->size));
704 string all;
705 for (const BlockRef &block : heap->blocks)
706 all += block->map_units();
707 for (size_t first = 0; first < all.size(); first += 64) {
708 ostringstream label;
709 label << setw(5) << setfill('0') << first * UNIT;
710 emit(label.str(), all.substr(first, 64));
711 }
712 } else if (name == "FREELIST") {
713 vector<BlockRef> free = heap->free_blocks();
714 if (heap->mode == "SEGFIT") {
715 sort(free.begin(), free.end(),
716 [](auto &a, auto &b) { return a->stamp > b->stamp; });
717 }
718 emit("FREELIST count=" + to_string(free.size()));
719 map<int, vector<string>> groups;
720 for (auto &block : free)
721 groups[size_class(block->size)].push_back(block->area());
722 for (auto &[k, areas] : groups) {
723 ostringstream line;
724 line << (1 << k) << "-" << ((1 << (k + 1)) - 1) << ":";
725 for (const string &value : areas)
726 line << " " << value;
727 emit(line.str());
728 }
729 } else if (name == "TREE") {
730 if (!heap->buddy())
731 throw CommandError("REQUIRES BUDDY MODE");
732 emit("TREE heap=" + to_string(heap->size));
733 map<int, BlockRef> leaves;
734 for (auto &block : heap->blocks)
735 leaves[block->start] = block;
736 tree_node(leaves, 0, heap->size, 0);
737 } else if (name == "STATS") {
738 auto used = heap->used_blocks(), free = heap->free_blocks();
739 int total_free = 0, largest = 0, total_used = 0, requested = 0;
740 for (auto &block : used) {
741 total_used += block->size;
742 requested += block->requested;
743 }
744 for (auto &block : free) {
745 total_free += block->size;
746 largest = max(largest, block->size);
747 }
748 emit("STATS used=" + to_string(total_used) + " used_blocks=" + to_string(used.size()) +
749 " free=" + to_string(total_free) + " free_blocks=" + to_string(free.size()) +
750 " largest=" + to_string(largest) +
751 " frag=" + to_string(fragmentation_percent(total_free, largest)) +
752 " requested=" + to_string(requested) + " peak=" + to_string(peak));
753 ostringstream line;
754 line << "OPS";
755 for (const string &key : COUNTERS)
756 line << " " << key << "=" << counters[key];
757 emit(line.str());
758 } else if (name == "LEAKS") {
759 auto found = reachable();
760 vector<string> names;
761 int lost = 0, requested = 0;
762 for (const auto &[id, block] : allocations) {
763 names.push_back(id);
764 requested += block->requested;
765 if (!found.count(id))
766 ++lost;
767 }
768 sort(names.begin(), names.end());
769 emit("LEAKS count=" + to_string(names.size()) + " requested=" + to_string(requested) +
770 " lost=" + to_string(lost));
771 for (const string &id : names) {
772 BlockRef block = allocations[id];
773 emit("LEAK " + id + " addr=" + to_string(block->payload()) +
774 " block=" + to_string(block->size) + " req=" + to_string(block->requested) +
775 " " + (found.count(id) ? "reachable" : "lost"));
776 }
777 } else if (name == "HISTORY") {
778 const auto &lines = histories[v[0]];
779 emit("HISTORY", v[0], "events=" + to_string(lines.size()));
780 output.insert(output.end(), lines.begin(), lines.end());
781 } else if (name == "BENCH") {
782 int size = stoi(v[0]), operations = stoi(v[2]), max_request = stoi(v[3]);
783 int64_t seed = stoll(v[1]);
784 if (!power_of_two(size) || size < 32 || size > MAX_HEAP)
785 throw CommandError("BAD HEAP SIZE");
786 emit("BENCH heap=" + to_string(size) + " seed=" + to_string(seed) +
787 " ops=" + to_string(operations) + " maxreq=" + to_string(max_request));
788 for (const string &mode : MODES)
789 emit(experiment(mode, size, seed, operations, max_request));
790 }
791 }
792 void execute(const string &line) {
793 istringstream words(line);
794 vector<string> tokens;
795 string word;
796 while (words >> word)
797 tokens.push_back(move(word));
798 const string command = tokens[0];
799 vector<string> args(tokens.begin() + 1, tokens.end());
800 try {
801 if (!heap && command != "HEAP" && command != "BENCH")
802 throw CommandError("NO HEAP");
803 run(command, args);
804 } catch (const CommandError &error) {
805 emit("ERROR", error.what());
806 }
807 if (heap)
808 peak = max(peak, heap->used_total());
809 }
810};

812int main() {
813 ios::sync_with_stdio(false);
814 cin.tie(nullptr);
815 int count = 0;
816 cin >> count;
817 string line;
818 getline(cin, line);
819 Allocator allocator;
820 for (int i = 0; i < count; ++i) {
821 getline(cin, line);
822 allocator.execute(line);
823 }
824 for (const string &text : allocator.output)
825 cout << text << '\n';
826}
