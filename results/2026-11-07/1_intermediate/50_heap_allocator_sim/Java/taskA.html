86 static final class Heap {
87 String mode;
88 int size;
89 int rover;
90 int registrations;
91 ArrayList<Block> blocks = new ArrayList<>();
92
93 Heap(String mode, int size) {
94 this.mode = mode;
95 this.size = size;
96 blocks.add(register(new Block(0, size)));
97 }
98 Block register(Block block) {
99 block.stamp = ++registrations;
100 return block;
101 }
102 boolean isBuddy() {
103 return mode.equals("BUDDY");
104 }
105 int requiredLength(int request) {
106 int length = (request + UNIT - 1) / UNIT * UNIT + HEADER;
107 if (!isBuddy())
108 return length;
109 int power = MIN_BUDDY_BLOCK;
110 while (power < length)
111 power *= 2;
112 return power;
113 }
114 int locate(int address) {
115 int i = 0;
116 while (blocks.get(i).end() <= address)
117 ++i;
118 return i;
119 }
120 List<Block> usedBlocks() {
121 return blocks.stream().filter(b -> b.used).toList();
122 }
123 List<Block> freeBlocks() {
124 return blocks.stream().filter(b -> !b.used).toList();
125 }
126 int usedTotal() {
127 return usedBlocks().stream().mapToInt(b -> b.size).sum();
128 }
129 int fragmentation() {
130 List<Block> free = freeBlocks();
131 int total = free.stream().mapToInt(b -> b.size).sum();
132 int largest = free.stream().mapToInt(b -> b.size).max().orElse(0);
133 return fragmentationPercent(total, largest);
134 }
135 int placement(Block block, int alignment) {
136 int p = (block.start + HEADER + alignment - 1) / alignment * alignment - HEADER;
137 if (p > block.start && p - block.start < MIN_BLOCK)
138 p += alignment;
139 return p;
140 }
141 int[] rank(int index, int roverIndex) {
142 Block block = blocks.get(index);
143 return switch (mode) {
144 case "NEXT" -> new int[] {(index - roverIndex + blocks.size()) % blocks.size(), 0};
145 case "BEST", "BUDDY" -> new int[] {block.size, block.start};
146 case "WORST" -> new int[] {-block.size, block.start};
147 case "SEGFIT" -> new int[] {sizeClass(block.size), -block.stamp};
148 default -> new int[] {block.start, 0};
149 };
150 }
151 Choice choose(int length, int alignment) {
152 int roverIndex = locate(rover);
153 Choice selected = null;
154 for (int index = 0; index < blocks.size(); ++index) {
155 Block block = blocks.get(index);
156 if (block.used)
157 continue;
158 int place = placement(block, alignment);
159 if (place + length > block.end())
160 continue;
161 int[] rank = rank(index, roverIndex);
162 if (selected == null || rank[0] < selected.first() ||
163 (rank[0] == selected.first() && rank[1] < selected.second())) {
164 selected = new Choice(index, place, rank[0], rank[1]);
165 }
166 }
167 return selected;
168 }
169 Block allocate(int length, int alignment) {
170 Choice choice = choose(length, alignment);
171 if (choice == null)
172 return null;
173 Block block = isBuddy() ? halve(choice.index(), length)
174 : carve(choice.index(), choice.place(), length);
175 block.used = true;
176 if (mode.equals("NEXT"))
177 rover = block.end() == size ? 0 : block.end();
178 return block;
179 }
180 Block carve(int index, int place, int length) {
181 Block old = blocks.get(index);
182 List<Block> parts = new ArrayList<>();
183 if (place > old.start)
184 parts.add(register(new Block(old.start, place - old.start)));
185 Block used = new Block(place, old.end() - place);
186 parts.add(used);
187 if (used.size - length >= MIN_BLOCK) {
188 parts.add(register(new Block(place + length, used.size - length)));
189 used.size = length;
190 }
191 blocks.remove(index);
192 blocks.addAll(index, parts);
193 return used;
194 }
195 Block halve(int index, int length) {
196 Block block = blocks.get(index);
197 while (block.size > length) {
198 block.size /= 2;
199 blocks.add(index + 1, register(new Block(block.end(), block.size)));
200 }
201 return block;
202 }
203 void mergeNext(int index) {
204 blocks.get(index).size += blocks.remove(index + 1).size;
205 }
206 Coalesced coalesce(int index) {
207 int merges = 0;
208 if (isBuddy()) {
209 while (blocks.get(index).size < size) {
210 Block block = blocks.get(index);
211 int partner = block.start % (2 * block.size) == 0 ? index + 1 : index - 1;
212 if (partner < 0 || partner >= blocks.size())
213 break;
214 Block other = blocks.get(partner);
215 if (other.used || other.size != block.size)
216 break;
217 index = Math.min(index, partner);
218 mergeNext(index);
219 ++merges;
220 }
221 } else {
222 if (index + 1 < blocks.size() && !blocks.get(index + 1).used) {
223 mergeNext(index);
224 ++merges;
225 }
226 if (index > 0 && !blocks.get(index - 1).used) {
227 --index;
228 mergeNext(index);
229 ++merges;
230 }
231 }
232 return new Coalesced(register(blocks.get(index)), merges);
233 }
234 Coalesced release(Block block) {
235 block.used = false;
236 block.owner = null;
237 block.requested = 0;
238 block.contents = new int[0];
239 return coalesce(blocks.indexOf(block));
240 }
241 String shrink(int index, int length) {
242 Block block = blocks.get(index);
243 if (isBuddy()) {
244 if (block.size == length)
245 return "kept";
246 halve(index, length);
247 return "shrunk";
248 }
249 if (block.size - length < MIN_BLOCK)
250 return "kept";
251 blocks.add(index + 1, new Block(block.start + length, block.size - length));
252 block.size = length;
253 coalesce(index + 1);
254 return "shrunk";
255 }
256 String growIntoNext(int index, int length) {
257 if (index + 1 == blocks.size() || blocks.get(index + 1).used)
258 return null;
259 Block block = blocks.get(index);
260 int total = block.size + blocks.get(index + 1).size;
261 if (total < length)
262 return null;
263 blocks.remove(index + 1);
264 if (total - length >= MIN_BLOCK) {
265 block.size = length;
266 blocks.add(index + 1, register(new Block(block.end(), total - length)));
267 } else
268 block.size = total;
269 return "grown";
270 }
271 String growBuddy(int index, int length) {
272 Block block = blocks.get(index);
273 int oldSize = block.size;
274 int last = index;
275 while (oldSize < length) {
276 ++last;
277 if (block.start % (2 * oldSize) != 0 || last == blocks.size())
278 return null;
279 Block partner = blocks.get(last);
280 if (partner.used || partner.size != oldSize)
281 return null;
282 oldSize *= 2;
283 }
284 blocks.subList(index + 1, last + 1).clear();
285 block.size = length;
286 return "grown";
287 }
288 String resize(Block block, int length) {
289 int index = blocks.indexOf(block);
290 if (length <= block.size)
291 return shrink(index, length);
292 return isBuddy() ? growBuddy(index, length) : growIntoNext(index, length);
293 }
294 Block extend(int amount) {
295 blocks.add(new Block(size, amount));
296 size += amount;
297 return coalesce(blocks.size() - 1).block();
298 }
299 int trim() {
300 int released = 0;
301 if (isBuddy()) {
302 while (!blocks.get(blocks.size() - 1).used &&
303 2 * blocks.get(blocks.size() - 1).size == size) {
304 released += blocks.remove(blocks.size() - 1).size;
305 size /= 2;
306 }
307 } else if (blocks.size() >= 2 && !blocks.get(blocks.size() - 1).used) {
308 Block block = blocks.remove(blocks.size() - 1);
309 released = block.size;
310 size = block.start;
311 }
312 if (rover >= size)
313 rover = 0;
314 return released;
315 }
316 Packed compact() {
317 List<Moved> moved = new ArrayList<>();
318 ArrayList<Block> packed = new ArrayList<>(usedBlocks());
319 int address = 0;
320 for (Block block : packed) {
321 if (block.start != address) {
322 moved.add(new Moved(block, block.payload()));
323 block.start = address;
324 }
325 address += block.size;
326 }
327 if (address < size)
328 packed.add(register(new Block(address, size - address)));
329 blocks = packed;
330 rover = 0;
331 return new Packed(moved, address);
332 }
333 }

335 static String experiment(String mode, int size, long seed, int operations, int maxRequest) {
336 Heap heap = new Heap(mode, size);
337 List<Block> live = new ArrayList<>();
338 long state = seed;
339 int used = 0, peak = 0, successes = 0, failures = 0;
340 for (int i = 0; i < operations; ++i) {
341 state = random(state);
342 boolean allocating = live.isEmpty() || state % 3 != 0;
343 state = random(state);
344 if (allocating) {
345 int request = 1 + (int)(state % maxRequest);
346 Block block = heap.allocate(heap.requiredLength(request), UNIT);
347 if (block == null)
348 ++failures;
349 else {
350 ++successes;
351 live.add(block);
352 used += block.size;
353 }
354 } else {
355 Block block = live.remove((int)(state % live.size()));
356 used -= block.size;
357 heap.release(block);
358 }
359 peak = Math.max(peak, used);
360 }
361 return "BENCH " + mode + " ok=" + successes + " fail=" + failures + " peak=" + peak +
362 " used=" + used + " frag=" + heap.fragmentation() + " blocks=" + heap.blocks.size();
363 }

365 static final class Allocator {
366 Heap heap;
367 Map<String, Block> allocations;
368 Set<String> freed;
369 Set<String> roots;
370 Map<String, Set<String>> references;
371 Map<String, List<String>> histories;
372 Map<String, Integer> counters;
373 int peak;
374 List<String> output = new ArrayList<>();
375
376 Allocator() {
377 reset(null);
378 }
379 void reset(Heap replacement) {
380 heap = replacement;
381 allocations = new HashMap<>();
382 freed = new HashSet<>();
383 roots = new HashSet<>();
384 references = new HashMap<>();
385 histories = new HashMap<>();
386 counters = new HashMap<>();
387 for (String name : COUNTERS)
388 counters.put(name, 0);
389 peak = 0;
390 }
391 void emit(Object... parts) {
392 StringBuilder text = new StringBuilder();
393 for (Object part : parts) {
394 if (!text.isEmpty())
395 text.append(' ');
396 text.append(part);
397 }
398 output.add(text.toString());
399 }
400 void record(String id, String line) {
401 emit(line);
402 histories.computeIfAbsent(id, key -> new ArrayList<>()).add(line);
403 }
404 void increment(String name) {
405 counters.put(name, counters.get(name) + 1);
406 }
407 CommandError outOfMemory(String id) {
408 increment("failures");
409 return new CommandError("OUT OF MEMORY " + id);
410 }
411 Block live(String id) {
412 Block block = allocations.get(id);
413 if (block != null)
414 return block;
415 if (freed.contains(id))
416 throw new CommandError("USE AFTER FREE " + id);
417 throw new CommandError("UNKNOWN ID " + id);
418 }
419 Block range(String id, int offset, int length) {
420 Block block = live(id);
421 if (offset + length > block.requested)
422 throw new CommandError("OUT OF BOUNDS " + id);
423 return block;
424 }
425 void freeListOnly() {
426 if (heap.isBuddy())
427 throw new CommandError("UNSUPPORTED IN BUDDY MODE");
428 }
429 Set<String> reachable() {
430 Set<String> reached = new HashSet<>(roots);
431 ArrayList<String> pending = new ArrayList<>(roots);
432 while (!pending.isEmpty()) {
433 String name = pending.remove(pending.size() - 1);
434 for (String target : references.getOrDefault(name, Set.of())) {
435 if (reached.add(target))
436 pending.add(target);
437 }
438 }
439 return reached;
440 }
441 void allocateId(String kind, String id, int requested, int alignment, int fill) {
442 if (requested == 0)
443 throw new CommandError("BAD SIZE");
444 if (allocations.containsKey(id))
445 throw new CommandError("DUPLICATE ID " + id);
446 Block block = heap.allocate(heap.requiredLength(requested), alignment);
447 if (block == null)
448 throw outOfMemory(id);
449 block.owner = id;
450 block.requested = requested;
451 block.contents = new int[requested];
452 Arrays.fill(block.contents, fill);
453 allocations.put(id, block);
454 freed.remove(id);
455 increment("allocs");
456 record
457 (id, kind + " " + id + " addr=" + block.payload() + " block=" + block.size);
458 }
459 String releaseId(String id) {
460 Block block = allocations.remove(id);
461 int size = block.size;
462 freed.add(id);
463 roots.remove(id);
464 references.remove(id);
465 for (Set<String> targets : references.values())
466 targets.remove(id);
467 Coalesced result = heap.release(block);
468 return "block=" + size + " merges=" + result.merges() +
469 " free=" + result.block().area();
470 }
471 void freeId(String id) {
472 if (freed.contains(id))
473 throw new CommandError("DOUBLE FREE " + id);
474 if (!allocations.containsKey(id))
475 throw new CommandError("UNKNOWN ID " + id);
476 String details = releaseId(id);
477 increment("frees");
478 record
479 (id, "FREE " + id + " " + details);
480 }
481 void realloc(String id, int requested) {
482 if (requested == 0) {
483 freeId(id);
484 return;
485 }
486 if (freed.contains(id))
487 throw new CommandError("USE AFTER FREE " + id);
488 Block block = allocations.get(id);
489 if (block == null) {
490 allocateId("MALLOC", id, requested, UNIT, UNINITIALIZED);
491 return;
492 }
493 int length = heap.requiredLength(requested);
494 int[] contents = Arrays.copyOf(block.contents, requested);
495 if (contents.length > block.contents.length)
496 Arrays.fill(contents, block.contents.length, contents.length, UNINITIALIZED);
497 int previous = block.payload();
498 String kind = heap.resize(block, length);
499 if (kind == null) {
500 Block moved = heap.allocate(length, UNIT);
501 if (moved == null)
502 throw outOfMemory(id);
503 moved.owner = id;
504 allocations.put(id, moved);
505 heap.release(block);
506 increment("moves");
507 block = moved;
508 kind = "moved from=" + previous;
509 }
510 block.requested = requested;
511 block.contents = contents;
512 increment("reallocs");
513 record
514 (id,
515 "REALLOC " + id + " addr=" + block.payload() + " block=" + block.size + " " + kind);
516 }
517 void treeNode(Map<Integer, Block> leaves, int start, int size, int depth) {
518 String label = " ".repeat(depth * 2) + start + "+" + size;
519 Block block = leaves.get(start);
520 if (block != null && block.size == size) {
521 emit(label + (block.used ? " USED " + block.owner : " FREE"));
522 return;
523 }
524 emit(label + " split");
525 treeNode(leaves, start, size / 2, depth + 1);
526 treeNode(leaves, start + size / 2, size / 2, depth + 1);
527 }
528 void run(String command, String[] args) {
529 switch (command) {
530 case "HEAP" -> {
531 int size = number(args[0]);
532 String mode = args[1];
533 if (size % UNIT != 0 || size < 32 || size > MAX_HEAP ||
534 (mode.equals("BUDDY") && !powerOfTwo(size)))
535 throw new CommandError("BAD HEAP SIZE");
536 reset(new Heap(mode, size));
537 emit("HEAP size=" + size + " mode=" + mode);
538 }
539 case "MALLOC" -> allocateId("MALLOC", args[0], number(args[1]), UNIT, UNINITIALIZED);
540 case "CALLOC" ->
541 allocateId("CALLOC", args[0], number(args[1]) * number(args[2]), UNIT, 0);
542 case "MEMALIGN" -> {
543 freeListOnly();
544 int alignment = number(args[1]);
545 if (!powerOfTwo(alignment) || alignment < UNIT || alignment > 4096)
546 throw new CommandError("BAD ALIGNMENT");
547 allocateId("MEMALIGN", args[0], number(args[2]), alignment, UNINITIALIZED);
548 }
549 case "REALLOC" -> realloc(args[0], number(args[1]));
550 case "FREE" -> freeId(args[0]);
551 case "FILL" -> {
552 String id = args[0];
553 int offset = number(args[1]), length = number(args[2]), value = number(args[3]);
554 Block block = range(id, offset, length);
555 Arrays.fill(block.contents, offset, offset + length, value);
556 emit("FILL " + id + " " + offset + "+" + length + " = " + value);
557 }
558 case "COPY" -> {
559 String dest = args[0], source = args[2];
560 int destOffset = number(args[1]), sourceOffset = number(args[3]);
561 int length = number(args[4]);
562 Block dst = range(dest, destOffset, length);
563 Block src = range(source, sourceOffset, length);
564 System.arraycopy(src.contents, sourceOffset, dst.contents, destOffset, length);
565 emit("COPY " + dest + " " + destOffset + "+" + length + " from " + source + " " +
566 sourceOffset);
567 }
568 case "READ" -> {
569 String id = args[0];
570 int offset = number(args[1]), length = number(args[2]);
571 Block block = range(id, offset, length);
572 StringBuilder values = new StringBuilder();
573 for (int i = 0; i < length; ++i) {
574 if (i > 0)
575 values.append(' ');
576 int value = block.contents[offset + i];
577 values.append(value == UNINITIALIZED ? "?" : Integer.toString(value));
578 }
579 emit("READ " + id + " " + offset + "+" + length + " = " + values);
580 }
581 case "ROOT" -> {
582 live(args[0]);
583 roots.add(args[0]);
584 emit("ROOT " + args[0]);
585 }
586 case "UNROOT" -> {
587 live(args[0]);
588 if (!roots.contains(args[0]))
589 throw new CommandError("NOT ROOT " + args[0]);
590 roots.remove(args[0]);
591 emit("UNROOT " + args[0]);
592 }
593 case "LINK" -> {
594 String source = args[0], target = args[1];
595 live(source);
596 live(target);
597 references.computeIfAbsent(source, key -> new HashSet<>()).add(target);
598 emit("LINK " + source + " " + target);
599 }
600 case "UNLINK" -> {
601 String source = args[0], target = args[1];
602 live(source);
603 live(target);
604 if (!references.getOrDefault(source, Set.of()).contains(target))
605 throw new CommandError("NO LINK " + source + " " + target);
606 references.get(source).remove(target);
607 emit("UNLINK " + source + " " + target);
608 }
609 case "GC" -> {
610 Set<String> found = reachable();
611 List<Block> garbage = allocations.entrySet()
612 .stream()
613 .filter(entry -> !found.contains(entry.getKey()))
614 .map(Map.Entry::getValue)
615 .sorted(Comparator.comparingInt(b -> b.start))
616 .toList();
617 int bytes = garbage.stream().mapToInt(b -> b.size).sum();
618 emit("GC collected=" + garbage.size() + " bytes=" + bytes);
619 for (Block block : garbage) {
620 String id = block.owner;
621 record
622 (id, "COLLECT " + id + " " + releaseId(id));
623 }
624 counters.put("collected", counters.get("collected") + garbage.size());
625 }
626 case "SBRK" -> {
627 int amount = number(args[0]);
628 boolean valid = heap.isBuddy() ? amount == heap.size && 2 * heap.size <= MAX_HEAP
629 : amount % UNIT == 0 && amount >= MIN_BLOCK &&
630 heap.size + amount <= MAX_HEAP;
631 if (!valid)
632 throw new CommandError("BAD SBRK SIZE");
633 Block free = heap.extend(amount);
634 emit("SBRK heap=" + heap.size + " free=" + free.area());
635 }
636 case "TRIM" -> {
637 int released = heap.trim();
638 emit("TRIM heap=" + heap.size + " released=" + released);
639 }
640 case "COMPACT" -> {
641 freeListOnly();
642 Packed result = heap.compact();
643 counters.put("moves", counters.get("moves") + result.moved().size());
644 String free = result.used() < heap.size
645 ? result.used() + "+" + (heap.size - result.used())
646 : "none";
647 emit("COMPACT moved=" + result.moved().size() + " free=" + free);
648 for (Moved moved : result.moved()) {
649 Block block = moved.block();
650 record
651 (block.owner,
652 "MOVE " + block.owner + " " + moved.from() + " -> " + block.payload());
653 }
654 }
655 case "WHO" -> {
656 int address = number(args[0]);
657 if (address >= heap.size) {
658 emit("WHO " + address + " outside");
659 break;
660 }
661 Block block = heap.blocks.get(heap.locate(address));
662 if (!block.used) {
663 emit("WHO " + address + " free " + block.area());
664 break;
665 }
666 if (address < block.payload()) {
667 emit("WHO " + address + " header " + block.owner);
668 break;
669 }
670 int offset = address - block.payload();
671 String part = offset < block.requested ? "data" : "slack";
672 emit("WHO " + address + " " + part + " " + block.owner + "+" + offset);
673 }
674 case "DUMP" -> {
675 String first = "DUMP mode=" + heap.mode + " heap=" + heap.size +
676 " blocks=" + heap.blocks.size();
677 if (heap.mode.equals("NEXT"))
678 first += " rover=" + heap.rover;
679 emit(first);
680 for (Block block : heap.blocks) {
681 if (block.used)
682 emit(block.start + " " + block.size + " USED " + block.owner +
683 " req=" + block.requested);
684 else
685 emit(block.start + " " + block.size + " FREE");
686 }
687 }
688 case "MAP" -> {
689 emit("MAP heap=" + heap.size);
690 StringBuilder all = new StringBuilder();
691 for (Block block : heap.blocks)
692 all.append(block.mapUnits());
693 for (int first = 0; first < all.length(); first += 64) {
694 String label = String.format("%05d", first * UNIT);
695 emit(label + " " + all.substring(first, Math.min(first + 64, all.length())));
696 }
697 }
698 case "FREELIST" -> {
699 List<Block> free = new ArrayList<>(heap.freeBlocks());
700 if (heap.mode.equals("SEGFIT"))
701 free.sort(Comparator.comparingInt((Block b) -> b.stamp).reversed());
702 emit("FREELIST count=" + free.size());
703 Map<Integer, List<String>> groups = new java.util.TreeMap<>();
704 for (Block block : free)
705 groups.computeIfAbsent(sizeClass(block.size), k -> new ArrayList<>())
706 .add(block.area());
707 for (Map.Entry<Integer, List<String>> entry : groups.entrySet()) {
708 int exp = entry.getKey();
709 emit((1 << exp) + "-" + ((1 << (exp + 1)) - 1) + ": " +
710 String.join(" ", entry.getValue()));
711 }
712 }
713 case "TREE" -> {
714 if (!heap.isBuddy())
715 throw new CommandError("REQUIRES BUDDY MODE");
716 emit("TREE heap=" + heap.size);
717 Map<Integer, Block> leaves = new HashMap<>();
718 for (Block block : heap.blocks)
719 leaves.put(block.start, block);
720 treeNode(leaves, 0, heap.size, 0);
721 }
722 case "STATS" -> {
723 List<Block> used = heap.usedBlocks(), free = heap.freeBlocks();
724 int freeTotal = free.stream().mapToInt(b -> b.size).sum();
725 int largest = free.stream().mapToInt(b -> b.size).max().orElse(0);
726 int usedTotal = used.stream().mapToInt(b -> b.size).sum();
727 int requested = used.stream().mapToInt(b -> b.requested).sum();
728 emit("STATS used=" + usedTotal + " used_blocks=" + used.size() +
729 " free=" + freeTotal + " free_blocks=" + free.size() + " largest=" + largest +
730 " frag=" + fragmentationPercent(freeTotal, largest) +
731 " requested=" + requested + " peak=" + peak);
732 List<String> ops = new ArrayList<>();
733 for (String name : COUNTERS)
734 ops.add(name + "=" + counters.get(name));
735 emit("OPS " + String.join(" ", ops));
736 }
737 case "LEAKS" -> {
738 Set<String> found = reachable();
739 Set<String> names = new TreeSet<>(allocations.keySet());
740 int requested = allocations.values().stream().mapToInt(b -> b.requested).sum();
741 int lost = 0;
742 for (String id : names)
743 if (!found.contains(id))
744 ++lost;
745 emit("LEAKS count=" + names.size() + " requested=" + requested + " lost=" + lost);
746 for (String id : names) {
747 Block block = allocations.get(id);
748 emit("LEAK " + id + " addr=" + block.payload() + " block=" + block.size +
749 " req=" + block.requested + " " +
750 (found.contains(id) ? "reachable" : "lost"));
751 }
752 }
753 case "HISTORY" -> {
754 List<String> events = histories.getOrDefault(args[0], List.of());
755 emit("HISTORY " + args[0] + " events=" + events.size());
756 output.addAll(events);
757 }
758 case "BENCH" -> {
759 int size = number(args[0]), operations = number(args[2]);
760 long seed = Long.parseLong(args[1]);
761 int maxRequest = number(args[3]);
762 if (!powerOfTwo(size) || size < 32 || size > MAX_HEAP)
763 throw new CommandError("BAD HEAP SIZE");
764 emit("BENCH heap=" + size + " seed=" + seed + " ops=" + operations +
765 " maxreq=" + maxRequest);
766 for (String mode : MODES)
767 emit(experiment(mode, size, seed, operations, maxRequest));
768 }
769 default -> throw new IllegalArgumentException("Unknown command: " + command);
770 }
771 }
772 void execute(String line) {
773 String[] tokens = line.split(" ");
774 String command = tokens[0];
775 String[] args = Arrays.copyOfRange(tokens, 1, tokens.length);
776 try {
777 if (heap == null && !command.equals("HEAP") && !command.equals("BENCH"))
778 throw new CommandError("NO HEAP");
779 run(command, args);
780 } catch (CommandError error) {
781 emit("ERROR " + error.getMessage());
782 }
783 if (heap != null)
784 peak = Math.max(peak, heap.usedTotal());
785 }
786 }

788 public static void main(String[] args) throws Exception {
789 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
790 int count = Integer.parseInt(input.readLine());
791 Allocator allocator = new Allocator();
792 for (int i = 0; i < count; ++i)
793 allocator.execute(input.readLine());
794 System.out.print(String.join("\n", allocator.output) + "\n");
795 }
