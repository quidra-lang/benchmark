6public class Program {
7 static readonly int UNIT = 8;
8 static readonly int HEADER = 8;
9 static readonly int MIN_BLOCK = 16;
10 static readonly int MIN_BUDDY_BLOCK = 32;
11 static readonly int MAX_HEAP = 65536;
12 static readonly int UNINITIALIZED = -1;
13 static readonly long RANDOM_MODULUS = 2147483647L;
14 static readonly long RANDOM_MULTIPLIER = 48271L;
15 static readonly List<string> MODES =
16 new List<string> { "FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY" };
17 static readonly List<string> COUNTERS =
18 new List<string> { "allocs", "frees", "reallocs", "moves", "failures", "collected" };

20 static int Number(string text) {
21 return int.Parse(text);
22 }
23 static bool PowerOfTwo(int n) {
24 return n > 0 && (n & (n - 1)) == 0;
25 }
26 static int SizeClass(int n) {
27 int k = 0;
28 while ((1 << (k + 1)) <= n)
29 ++k;
30 return k;
31 }
32 static int FragmentationPercent(int free, int largest) {
33 return free == 0 ? 0 : (free - largest) * 100 / free;
34 }
35 static long Random(long state) {
36 return RANDOM_MULTIPLIER * state % RANDOM_MODULUS;
37 }

44 public class Block {
45 public int start;
46 public int size;
47 public int requested;
48 public int stamp;
49 public bool used;
50 public string owner;
51 public int[] contents = new int[0];
52
53 public Block(int start, int size) {
54 this.start = start;
55 this.size = size;
56 }
57 public int End() {
58 return start + size;
59 }
60 public int Payload() {
61 return start + HEADER;
62 }
63 public string Area() {
64 return start + "+" + size;
65 }
66 public string MapUnits() {
67 int count = size / UNIT - 1;
68 StringBuilder result = new StringBuilder();
69 result.Append(used ? 'H' : 'F');
70 for (int i = 0; i < count; ++i) {
71 result.Append(!used ? '.' : i * UNIT < requested ? '#' : '-');
72 }
73 return result.ToString();
74 }

77 public record Choice(int index, int place, int first, int second);

82 public class Heap {
83 public string mode;
84 public int size;
85 public int rover;
86 public int registrations;
87 public List<Block> blocks = new();
88
89 public Heap(string mode, int size) {
90 this.mode = mode;
91 this.size = size;
92 blocks.Add(Register(new Block(0, size)));
93 }
94 public Block Register(Block block) {
95 block.stamp = ++registrations;
96 return block;
97 }
98 public bool IsBuddy() {
99 return mode.Equals("BUDDY");
100 }
101 public int RequiredLength(int request) {
102 int length = (request + UNIT - 1) / UNIT * UNIT + HEADER;
103 if (!IsBuddy())
104 return length;
105 int power = MIN_BUDDY_BLOCK;
106 while (power < length)
107 power *= 2;
108 return power;
109 }
110 public int Locate(int address) {
111 int i = 0;
112 while (blocks[i].End() <= address)
113 ++i;
114 return i;
115 }
116 public List<Block> UsedBlocks() {
117 return blocks.Where(b => b.used).ToList();
118 }
119 public List<Block> FreeBlocks() {
120 return blocks.Where(b => !b.used).ToList();
121 }
122 public int UsedTotal() {
123 return UsedBlocks().Select(b => b.size).Sum();
124 }
125 public int Fragmentation() {
126 List<Block> free = FreeBlocks();
127 int total = free.Select(b => b.size).Sum();
128 int largest = free.Select(b => b.size).DefaultIfEmpty(0).Max();
129 return FragmentationPercent(total, largest);
130 }
131 public int Placement(Block block, int alignment) {
132 int p = (block.start + HEADER + alignment - 1) / alignment * alignment - HEADER;
133 if (p > block.start && p - block.start < MIN_BLOCK)
134 p += alignment;
135 return p;
136 }
137 public int[] Rank(int index, int roverIndex) {
138 Block block = blocks[index];
139 return mode switch {
140 "NEXT" => new int[] { (index - roverIndex + blocks.Count) % blocks.Count, 0 },
141 "BEST" or "BUDDY" => new int[] { block.size, block.start },
142 "WORST" => new int[] { -block.size, block.start },
143 "SEGFIT" => new int[] { SizeClass(block.size), -block.stamp },
144 _ => new int[] { block.start, 0 },
145 };
146 }
147 public Choice Choose(int length, int alignment) {
148 int roverIndex = Locate(rover);
149 Choice selected = null;
150 for (int index = 0; index < blocks.Count; ++index) {
151 Block block = blocks[index];
152 if (block.used)
153 continue;
154 int place = Placement(block, alignment);
155 if (place + length > block.End())
156 continue;
157 int[] rank = Rank(index, roverIndex);
158 if (selected == null || rank[0] < selected.first ||
159 (rank[0] == selected.first && rank[1] < selected.second)) {
160 selected = new Choice(index, place, rank[0], rank[1]);
161 }
162 }
163 return selected;
164 }
165 public Block Allocate(int length, int alignment) {
166 Choice choice = Choose(length, alignment);
167 if (choice == null)
168 return null;
169 Block block =
170 IsBuddy() ? Halve(choice.index, length) : Carve(choice.index, choice.place, length);
171 block.used = true;
172 if (mode.Equals("NEXT"))
173 rover = block.End() == size ? 0 : block.End();
174 return block;
175 }
176 public Block Carve(int index, int place, int length) {
177 Block old = blocks[index];
178 List<Block> parts = new();
179 if (place > old.start)
180 parts.Add(Register(new Block(old.start, place - old.start)));
181 Block used = new Block(place, old.End() - place);
182 parts.Add(used);
183 if (used.size - length >= MIN_BLOCK) {
184 parts.Add(Register(new Block(place + length, used.size - length)));
185 used.size = length;
186 }
187 blocks.RemoveAt(index);
188 blocks.InsertRange(index, parts);
189 return used;
190 }
191 public Block Halve(int index, int length) {
192 Block block = blocks[index];
193 while (block.size > length) {
194 block.size /= 2;
195 blocks.Insert(index + 1, Register(new Block(block.End(), block.size)));
196 }
197 return block;
198 }
199 public void MergeNext(int index) {
200 blocks[index].size += blocks[index + 1].size;
201 blocks.RemoveAt(index + 1);
202 }
203 public Coalesced Coalesce(int index) {
204 int merges = 0;
205 if (IsBuddy()) {
206 while (blocks[index].size < size) {
207 Block block = blocks[index];
208 int partner = block.start % (2 * block.size) == 0 ? index + 1 : index - 1;
209 if (partner < 0 || partner >= blocks.Count)
210 break;
211 Block other = blocks[partner];
212 if (other.used || other.size != block.size)
213 break;
214 index = Math.Min(index, partner);
215 MergeNext(index);
216 ++merges;
217 }
218 } else {
219 if (index + 1 < blocks.Count && !blocks[index + 1].used) {
220 MergeNext(index);
221 ++merges;
222 }
223 if (index > 0 && !blocks[index - 1].used) {
224 --index;
225 MergeNext(index);
226 ++merges;
227 }
228 }
229 return new Coalesced(Register(blocks[index]), merges);
230 }
231 public Coalesced Release(Block block) {
232 block.used = false;
233 block.owner = null;
234 block.requested = 0;
235 block.contents = new int[0];
236 return Coalesce(blocks.IndexOf(block));
237 }
238 public string Shrink(int index, int length) {
239 Block block = blocks[index];
240 if (IsBuddy()) {
241 if (block.size == length)
242 return "kept";
243 Halve(index, length);
244 return "shrunk";
245 }
246 if (block.size - length < MIN_BLOCK)
247 return "kept";
248 blocks.Insert(index + 1, new Block(block.start + length, block.size - length));
249 block.size = length;
250 Coalesce(index + 1);
251 return "shrunk";
252 }
253 public string GrowIntoNext(int index, int length) {
254 if (index + 1 == blocks.Count || blocks[index + 1].used)
255 return null;
256 Block block = blocks[index];
257 int total = block.size + blocks[index + 1].size;
258 if (total < length)
259 return null;
260 blocks.RemoveAt(index + 1);
261 if (total - length >= MIN_BLOCK) {
262 block.size = length;
263 blocks.Insert(index + 1, Register(new Block(block.End(), total - length)));
264 } else
265 block.size = total;
266 return "grown";
267 }
268 public string GrowBuddy(int index, int length) {
269 Block block = blocks[index];
270 int oldSize = block.size;
271 int last = index;
272 while (oldSize < length) {
273 ++last;
274 if (block.start % (2 * oldSize) != 0 || last == blocks.Count)
275 return null;
276 Block partner = blocks[last];
277 if (partner.used || partner.size != oldSize)
278 return null;
279 oldSize *= 2;
280 }
281 blocks.RemoveRange(index + 1, last - index);
282 block.size = length;
283 return "grown";
284 }
285 public string Resize(Block block, int length) {
286 int index = blocks.IndexOf(block);
287 if (length <= block.size)
288 return Shrink(index, length);
289 return IsBuddy() ? GrowBuddy(index, length) : GrowIntoNext(index, length);
290 }
291 public Block Extend(int amount) {
292 blocks.Add(new Block(size, amount));
293 size += amount;
294 return Coalesce(blocks.Count - 1).block;
295 }
296 public int Trim() {
297 int released = 0;
298 if (IsBuddy()) {
299 while (!blocks[blocks.Count - 1].used &&
300 2 * blocks[blocks.Count - 1].size == size) {
301 released += blocks[blocks.Count - 1].size;
302 blocks.RemoveAt(blocks.Count - 1);
303 size /= 2;
304 }
305 } else if (blocks.Count >= 2 && !blocks[blocks.Count - 1].used) {
306 Block block = blocks[blocks.Count - 1];
307 blocks.RemoveAt(blocks.Count - 1);
308 released = block.size;
309 size = block.start;
310 }
311 if (rover >= size)
312 rover = 0;
313 return released;
314 }
315 public Packed Compact() {
316 List<Moved> moved = new();
317 List<Block> packed = new List<Block>(UsedBlocks());
318 int address = 0;
319 foreach (Block block in packed) {
320 if (block.start != address) {
321 moved.Add(new Moved(block, block.Payload()));
322 block.start = address;
323 }
324 address += block.size;
325 }
326 if (address < size)
327 packed.Add(Register(new Block(address, size - address)));
328 blocks = packed;
329 rover = 0;
330 return new Packed(moved, address);
331 }
332 }

334 static string Experiment(string mode, int size, long seed, int operations, int maxRequest) {
335 Heap heap = new Heap(mode, size);
336 List<Block> live = new();
337 long state = seed;
338 int used = 0, peak = 0, successes = 0, failures = 0;
339 for (int i = 0; i < operations; ++i) {
340 state = Random(state);
341 bool allocating = live.Count == 0 || state % 3 != 0;
342 state = Random(state);
343 if (allocating) {
344 int request = 1 + (int)(state % maxRequest);
345 Block block = heap.Allocate(heap.RequiredLength(request), UNIT);
346 if (block == null)
347 ++failures;
348 else {
349 ++successes;
350 live.Add(block);
351 used += block.size;
352 }
353 } else {
354 int at = (int)(state % live.Count);
355 Block block = live[at];
356 live.RemoveAt(at);
357 used -= block.size;
358 heap.Release(block);
359 }
360 peak = Math.Max(peak, used);
361 }
362 return "BENCH " + mode + " ok=" + successes + " fail=" + failures + " peak=" + peak +
363 " used=" + used + " frag=" + heap.Fragmentation() + " blocks=" + heap.blocks.Count;
364 }

366 public class Allocator {
367 public Heap heap;
368 public Dictionary<string, Block> allocations;
369 public HashSet<string> freed;
370 public HashSet<string> roots;
371 public Dictionary<string, HashSet<string>> references;
372 public Dictionary<string, List<string>> histories;
373 public Dictionary<string, int> counters;
374 public int peak;
375 public List<string> output = new();
376
377 public Allocator() {
378 Reset(null);
379 }
380 public void Reset(Heap replacement) {
381 heap = replacement;
382 allocations = new();
383 freed = new();
384 roots = new();
385 references = new();
386 histories = new();
387 counters = new();
388 foreach (string name in COUNTERS)
389 counters[name] = 0;
390 peak = 0;
391 }
392 public void Emit(string text) {
393 output.Add(text);
394 }
395 public void Record(string id, string line) {
396 Emit(line);
397 if (!histories.ContainsKey(id))
398 histories[id] = new List<string>();
399 histories[id].Add(line);
400 }
401 public void Increment(string name) {
402 counters[name] = counters[name] + 1;
403 }
404 public CommandError OutOfMemory(string id) {
405 Increment("failures");
406 return new CommandError("OUT OF MEMORY " + id);
407 }
408 public Block Live(string id) {
409 Block block = allocations.GetValueOrDefault(id);
410 if (block != null)
411 return block;
412 if (freed.Contains(id))
413 throw new CommandError("USE AFTER FREE " + id);
414 throw new CommandError("UNKNOWN ID " + id);
415 }
416 public Block Range(string id, int offset, int length) {
417 Block block = Live(id);
418 if (offset + length > block.requested)
419 throw new CommandError("OUT OF BOUNDS " + id);
420 return block;
421 }
422 public void FreeListOnly() {
423 if (heap.IsBuddy())
424 throw new CommandError("UNSUPPORTED IN BUDDY MODE");
425 }
426 public HashSet<string> Reachable() {
427 HashSet<string> reached = new HashSet<string>(roots);
428 List<string> pending = new List<string>(roots);
429 while (pending.Count != 0) {
430 string name = pending[pending.Count - 1];
431 pending.RemoveAt(pending.Count - 1);
432 foreach (string target in references.GetValueOrDefault(name,
433 new HashSet<string>())) {
434 if (reached.Add(target))
435 pending.Add(target);
436 }
437 }
438 return reached;
439 }
440 public void AllocateId(string kind, string id, int requested, int alignment, int fill) {
441 if (requested == 0)
442 throw new CommandError("BAD SIZE");
443 if (allocations.ContainsKey(id))
444 throw new CommandError("DUPLICATE ID " + id);
445 Block block = heap.Allocate(heap.RequiredLength(requested), alignment);
446 if (block == null)
447 throw OutOfMemory(id);
448 block.owner = id;
449 block.requested = requested;
450 block.contents = new int[requested];
451 Array.Fill(block.contents, fill);
452 allocations[id] = block;
453 freed.Remove(id);
454 Increment("allocs");
455 Record(id, kind + " " + id + " addr=" + block.Payload() + " block=" + block.size);
456 }
457 public string ReleaseId(string id) {
458 Block block = allocations.GetValueOrDefault(id);
459 allocations.Remove(id);
460 int size = block.size;
461 freed.Add(id);
462 roots.Remove(id);
463 references.Remove(id);
464 foreach (HashSet<string> targets in references.Values)
465 targets.Remove(id);
466 Coalesced result = heap.Release(block);
467 return "block=" + size + " merges=" + result.merges + " free=" + result.block.Area();
468 }
469 public void FreeId(string id) {
470 if (freed.Contains(id))
471 throw new CommandError("DOUBLE FREE " + id);
472 if (!allocations.ContainsKey(id))
473 throw new CommandError("UNKNOWN ID " + id);
474 string details = ReleaseId(id);
475 Increment("frees");
476 Record(id, "FREE " + id + " " + details);
477 }
478 public void Realloc(string id, int requested) {
479 if (requested == 0) {
480 FreeId(id);
481 return;
482 }
483 if (freed.Contains(id))
484 throw new CommandError("USE AFTER FREE " + id);
485 Block block = allocations.GetValueOrDefault(id);
486 if (block == null) {
487 AllocateId("MALLOC", id, requested, UNIT, UNINITIALIZED);
488 return;
489 }
490 int length = heap.RequiredLength(requested);
491 int[] contents = ResizeContents(block.contents, requested);
492 if (contents.Length > block.contents.Length)
493 Array.Fill(contents, UNINITIALIZED, block.contents.Length,
494 contents.Length - block.contents.Length);
495 int previous = block.Payload();
496 string kind = heap.Resize(block, length);
497 if (kind == null) {
498 Block moved = heap.Allocate(length, UNIT);
499 if (moved == null)
500 throw OutOfMemory(id);
501 moved.owner = id;
502 allocations[id] = moved;
503 heap.Release(block);
504 Increment("moves");
505 block = moved;
506 kind = "moved from=" + previous;
507 }
508 block.requested = requested;
509 block.contents = contents;
510 Increment("reallocs");
511 Record(id, "REALLOC " + id + " addr=" + block.Payload() + " block=" + block.size + " " +
512 kind);
513 }
514 public void TreeNode(Dictionary<int, Block> leaves, int start, int size, int depth) {
515 string label = new string(' ', depth * 2) + start + "+" + size;
516 Block block = leaves.GetValueOrDefault(start);
517 if (block != null && block.size == size) {
518 Emit(label + (block.used ? " USED " + block.owner : " FREE"));
519 return;
520 }
521 Emit(label + " split");
522 TreeNode(leaves, start, size / 2, depth + 1);
523 TreeNode(leaves, start + size / 2, size / 2, depth + 1);
524 }
525 public void Run(string command, string[] args) {
526 switch (command) {
527 case "HEAP": {
528 int size = Number(args[0]);
529 string mode = args[1];
530 if (size % UNIT != 0 || size < 32 || size > MAX_HEAP ||
531 (mode.Equals("BUDDY") && !PowerOfTwo(size)))
532 throw new CommandError("BAD HEAP SIZE");
533 Reset(new Heap(mode, size));
534 Emit("HEAP size=" + size + " mode=" + mode);
535 break;
536 }
537 case "MALLOC":
538 AllocateId("MALLOC", args[0], Number(args[1]), UNIT, UNINITIALIZED);
539 break;
540 case "CALLOC":
541 AllocateId("CALLOC", args[0], Number(args[1]) * Number(args[2]), UNIT, 0);
542 break;
543 case "MEMALIGN": {
544 FreeListOnly();
545 int alignment = Number(args[1]);
546 if (!PowerOfTwo(alignment) || alignment < UNIT || alignment > 4096)
547 throw new CommandError("BAD ALIGNMENT");
548 AllocateId("MEMALIGN", args[0], Number(args[2]), alignment, UNINITIALIZED);
549 break;
550 }
551 case "REALLOC":
552 Realloc(args[0], Number(args[1]));
553 break;
554 case "FREE":
555 FreeId(args[0]);
556 break;
557 case "FILL": {
558 string id = args[0];
559 int offset = Number(args[1]), length = Number(args[2]), value = Number(args[3]);
560 Block block = Range(id, offset, length);
561 Array.Fill(block.contents, value, offset, length);
562 Emit("FILL " + id + " " + offset + "+" + length + " = " + value);
563 break;
564 }
565 case "COPY": {
566 string dest = args[0], source = args[2];
567 int destOffset = Number(args[1]), sourceOffset = Number(args[3]);
568 int length = Number(args[4]);
569 Block dst = Range(dest, destOffset, length);
570 Block src = Range(source, sourceOffset, length);
571 Array.Copy(src.contents, sourceOffset, dst.contents, destOffset, length);
572 Emit("COPY " + dest + " " + destOffset + "+" + length + " from " + source + " " +
573 sourceOffset);
574 break;
575 }
576 case "READ": {
577 string id = args[0];
578 int offset = Number(args[1]), length = Number(args[2]);
579 Block block = Range(id, offset, length);
580 StringBuilder values = new StringBuilder();
581 for (int i = 0; i < length; ++i) {
582 if (i > 0)
583 values.Append(' ');
584 int value = block.contents[offset + i];
585 values.Append(value == UNINITIALIZED ? "?" : value.ToString());
586 }
587 Emit("READ " + id + " " + offset + "+" + length + " = " + values);
588 break;
589 }
590 case "ROOT": {
591 Live(args[0]);
592 roots.Add(args[0]);
593 Emit("ROOT " + args[0]);
594 break;
595 }
596 case "UNROOT": {
597 Live(args[0]);
598 if (!roots.Contains(args[0]))
599 throw new CommandError("NOT ROOT " + args[0]);
600 roots.Remove(args[0]);
601 Emit("UNROOT " + args[0]);
602 break;
603 }
604 case "LINK": {
605 string source = args[0], target = args[1];
606 Live(source);
607 Live(target);
608 if (!references.ContainsKey(source))
609 references[source] = new HashSet<string>();
610 references[source].Add(target);
611 Emit("LINK " + source + " " + target);
612 break;
613 }
614 case "UNLINK": {
615 string source = args[0], target = args[1];
616 Live(source);
617 Live(target);
618 if (!references.GetValueOrDefault(source, new HashSet<string>()).Contains(target))
619 throw new CommandError("NO LINK " + source + " " + target);
620 references[source].Remove(target);
621 Emit("UNLINK " + source + " " + target);
622 break;
623 }
624 case "GC": {
625 HashSet<string> found = Reachable();
626 List<Block> garbage = allocations.Where(entry => !found.Contains(entry.Key))
627 .Select(entry => entry.Value)
628 .OrderBy(b => b.start)
629 .ToList();
630 int bytes = garbage.Select(b => b.size).Sum();
631 Emit("GC collected=" + garbage.Count + " bytes=" + bytes);
632 foreach (Block block in garbage) {
633 string id = block.owner;
634 Record(id, "COLLECT " + id + " " + ReleaseId(id));
635 }
636 counters["collected"] = counters["collected"] + garbage.Count;
637 break;
638 }
639 case "SBRK": {
640 int amount = Number(args[0]);
641 bool valid = heap.IsBuddy() ? amount == heap.size && 2 * heap.size <= MAX_HEAP
642 : amount % UNIT == 0 && amount >= MIN_BLOCK &&
643 heap.size + amount <= MAX_HEAP;
644 if (!valid)
645 throw new CommandError("BAD SBRK SIZE");
646 Block free = heap.Extend(amount);
647 Emit("SBRK heap=" + heap.size + " free=" + free.Area());
648 break;
649 }
650 case "TRIM": {
651 int released = heap.Trim();
652 Emit("TRIM heap=" + heap.size + " released=" + released);
653 break;
654 }
655 case "COMPACT": {
656 FreeListOnly();
657 Packed result = heap.Compact();
658 counters["moves"] = counters["moves"] + result.moved.Count;
659 string free = result.used < heap.size
660 ? result.used + "+" + (heap.size - result.used)
661 : "none";
662 Emit("COMPACT moved=" + result.moved.Count + " free=" + free);
663 foreach (Moved moved in result.moved) {
664 Block block = moved.block;
665 Record(block.owner,
666 "MOVE " + block.owner + " " + moved.from + " -> " + block.Payload());
667 }
668 break;
669 }
670 case "WHO": {
671 int address = Number(args[0]);
672 if (address >= heap.size) {
673 Emit("WHO " + address + " outside");
674 break;
675 }
676 Block block = heap.blocks[heap.Locate(address)];
677 if (!block.used) {
678 Emit("WHO " + address + " free " + block.Area());
679 break;
680 }
681 if (address < block.Payload()) {
682 Emit("WHO " + address + " header " + block.owner);
683 break;
684 }
685 int offset = address - block.Payload();
686 string part = offset < block.requested ? "data" : "slack";
687 Emit("WHO " + address + " " + part + " " + block.owner + "+" + offset);
688 break;
689 }
690 case "DUMP": {
691 string first = "DUMP mode=" + heap.mode + " heap=" + heap.size +
692 " blocks=" + heap.blocks.Count;
693 if (heap.mode.Equals("NEXT"))
694 first += " rover=" + heap.rover;
695 Emit(first);
696 foreach (Block block in heap.blocks) {
697 if (block.used)
698 Emit(block.start + " " + block.size + " USED " + block.owner +
699 " req=" + block.requested);
700 else
701 Emit(block.start + " " + block.size + " FREE");
702 }
703 break;
704 }
705 case "MAP": {
706 Emit("MAP heap=" + heap.size);
707 StringBuilder all = new StringBuilder();
708 foreach (Block block in heap.blocks)
709 all.Append(block.MapUnits());
710 for (int first = 0; first < all.Length; first += 64) {
711 string label = (first * UNIT).ToString("D5");
712 Emit(label + " " + all.ToString(first, Math.Min(64, all.Length - first)));
713 }
714 break;
715 }
716 case "FREELIST": {
717 List<Block> free = new List<Block>(heap.FreeBlocks());
718 if (heap.mode.Equals("SEGFIT"))
719 free = free.OrderByDescending(b => b.stamp).ToList();
720 Emit("FREELIST count=" + free.Count);
721 SortedDictionary<int, List<string>> groups =
722 new SortedDictionary<int, List<string>>();
723 foreach (Block block in free) {
724 int group = SizeClass(block.size);
725 if (!groups.ContainsKey(group))
726 groups[group] = new List<string>();
727 groups[group].Add(block.Area());
728 }
729 foreach (KeyValuePair<int, List<string>> entry in groups) {
730 int exp = entry.Key;
731 Emit((1 << exp) + "-" + ((1 << (exp + 1)) - 1) + ": " +
732 string.Join(" ", entry.Value));
733 }
734 break;
735 }
736 case "TREE": {
737 if (!heap.IsBuddy())
738 throw new CommandError("REQUIRES BUDDY MODE");
739 Emit("TREE heap=" + heap.size);
740 Dictionary<int, Block> leaves = new();
741 foreach (Block block in heap.blocks)
742 leaves[block.start] = block;
743 TreeNode(leaves, 0, heap.size, 0);
744 break;
745 }
746 case "STATS": {
747 List<Block> used = heap.UsedBlocks(), free = heap.FreeBlocks();
748 int freeTotal = free.Select(b => b.size).Sum();
749 int largest = free.Select(b => b.size).DefaultIfEmpty(0).Max();
750 int usedTotal = used.Select(b => b.size).Sum();
751 int requested = used.Select(b => b.requested).Sum();
752 Emit("STATS used=" + usedTotal + " used_blocks=" + used.Count +
753 " free=" + freeTotal + " free_blocks=" + free.Count + " largest=" + largest +
754 " frag=" + FragmentationPercent(freeTotal, largest) +
755 " requested=" + requested + " peak=" + peak);
756 List<string> ops = new();
757 foreach (string name in COUNTERS)
758 ops.Add(name + "=" + counters[name]);
759 Emit("OPS " + string.Join(" ", ops));
760 break;
761 }
762 case "LEAKS": {
763 HashSet<string> found = Reachable();
764 SortedSet<string> names =
765 new SortedSet<string>(allocations.Keys, StringComparer.Ordinal);
766 int requested = allocations.Values.Select(b => b.requested).Sum();
767 int lost = 0;
768 foreach (string id in names)
769 if (!found.Contains(id))
770 ++lost;
771 Emit("LEAKS count=" + names.Count + " requested=" + requested + " lost=" + lost);
772 foreach (string id in names) {
773 Block block = allocations.GetValueOrDefault(id);
774 Emit("LEAK " + id + " addr=" + block.Payload() + " block=" + block.size +
775 " req=" + block.requested + " " +
776 (found.Contains(id) ? "reachable" : "lost"));
777 }
778 break;
779 }
780 case "HISTORY": {
781 List<string> events = histories.GetValueOrDefault(args[0], new List<string>());
782 Emit("HISTORY " + args[0] + " events=" + events.Count);
783 output.AddRange(events);
784 break;
785 }
786 case "BENCH": {
787 int size = Number(args[0]), operations = Number(args[2]);
788 long seed = long.Parse(args[1]);
789 int maxRequest = Number(args[3]);
790 if (!PowerOfTwo(size) || size < 32 || size > MAX_HEAP)
791 throw new CommandError("BAD HEAP SIZE");
792 Emit("BENCH heap=" + size + " seed=" + seed + " ops=" + operations +
793 " maxreq=" + maxRequest);
794 foreach (string mode in MODES)
795 Emit(Experiment(mode, size, seed, operations, maxRequest));
796 break;
797 }
798 default:
799 throw new ArgumentException("Unknown command: " + command);
800 }
801 }
802 public void Execute(string line) {
803 string[] tokens = line.Split(' ');
804 string command = tokens[0];
805 string[] args = tokens.Skip(1).ToArray();
806 try {
807 if (heap == null && !command.Equals("HEAP") && !command.Equals("BENCH"))
808 throw new CommandError("NO HEAP");
809 Run(command, args);
810 } catch (CommandError error) {
811 Emit("ERROR " + error.Message);
812 }
813 if (heap != null)
814 peak = Math.Max(peak, heap.UsedTotal());
815 }
816 }
