3const MODES = ["FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY"];
4const COUNTERS = ["allocs", "frees", "reallocs", "moves", "failures", "collected"];
5const UNIT = 8;
6const HEADER = 8;
7const MIN_BLOCK = 16;
8const MIN_BUDDY_BLOCK = 32;
9const MAX_HEAP = 65536;
10const UNINITIALIZED = -1;
11const RANDOM_MULTIPLIER = 48271;
12const RANDOM_MODULUS = 2147483647;

16function powerOfTwo(value) {
17 return value > 0 && (Math.log2(value) % 1) === 0;
18}
19
20function sizeClass(size) {
21 return Math.floor(Math.log2(size));
22}
23
24function fragmentationPercent(free, largest) {
25 return free > 0 ? Math.floor((free - largest) * 100 / free) : 0;
26}
27
28function nextRandom(state) {
29 return RANDOM_MULTIPLIER * state % RANDOM_MODULUS;
30}

32class Block {
33 constructor(start, size) {
34 this.start = start;
35 this.size = size;
36 this.used = false;
37 this.owner = null;
38 this.requested = 0;
39 this.contents = [];
40 this.stamp = 0;
41 }
42
43 end() { return this.start + this.size; }
44 payload() { return this.start + HEADER; }
45 area() { return `${this.start}+${this.size}`; }
46
47 mapUnits() {
48 const count = this.size / UNIT - 1;
49 if (!this.used) return "F" + ".".repeat(count);
50 let units = "H";
51 for (let unit = 0; unit < count; ++unit) {
52 units += UNIT * unit < this.requested ? "#" : "-";
53 }
54 return units;
55 }
56}

58class Heap {
59 constructor(mode, size) {
60 this.mode = mode;
61 this.size = size;
62 this.rover = 0;
63 this.registrations = 0;
64 this.blocks = [this.register(new Block(0, size))];
65 }
66
67 register(block) {
68 block.stamp = ++this.registrations;
69 return block;
70 }
71
72 isBuddy() { return this.mode === "BUDDY"; }
73
74 requiredLength(request) {
75 const length = Math.ceil(request / UNIT) * UNIT + HEADER;
76 if (!this.isBuddy()) return length;
77 let power = MIN_BUDDY_BLOCK;
78 while (power < length) power *= 2;
79 return power;
80 }
81
82 locate(address) {
83 let index = 0;
84 while (this.blocks[index].end() <= address) ++index;
85 return index;
86 }
87
88 usedBlocks() { return this.blocks.filter(block => block.used); }
89 freeBlocks() { return this.blocks.filter(block => !block.used); }
90 usedTotal() { return this.usedBlocks().reduce((sum, block) => sum + block.size, 0); }
91
92 fragmentation() {
93 const sizes = this.freeBlocks().map(block => block.size);
94 const total = sizes.reduce((sum, size) => sum + size, 0);
95 return fragmentationPercent(total, sizes.length ? Math.max(...sizes) : 0);
96 }
97
98 placement(block, alignment) {
99 let place = Math.ceil((block.start + HEADER) / alignment) * alignment - HEADER;
100 if (place > block.start && place - block.start < MIN_BLOCK) place += alignment;
101 return place;
102 }
103
104 rank(index, roverIndex) {
105 const block = this.blocks[index];
106 switch (this.mode) {
107 case "NEXT": return [(index - roverIndex + this.blocks.length) % this.blocks.length, 0];
108 case "BEST":
109 case "BUDDY": return [block.size, block.start];
110 case "WORST": return [-block.size, block.start];
111 case "SEGFIT": return [sizeClass(block.size), -block.stamp];
112 default: return [block.start, 0];
113 }
114 }
115
116 choose(length, alignment) {
117 const roverIndex = this.locate(this.rover);
118 let chosen = null;
119 for (let index = 0; index < this.blocks.length; ++index) {
120 const block = this.blocks[index];
121 if (block.used) continue;
122 const place = this.placement(block, alignment);
123 if (place + length > block.end()) continue;
124 const rank = this.rank(index, roverIndex);
125 if (chosen === null || rank[0] < chosen.rank[0] ||
126 (rank[0] === chosen.rank[0] && rank[1] < chosen.rank[1])) {
127 chosen = { rank, index, place };
128 }
129 }
130 return chosen;
131 }
132
133 allocate(length, alignment) {
134 const choice = this.choose(length, alignment);
135 if (choice === null) return null;
136 const { index, place } = choice;
137 const block = this.isBuddy() ? this.halve(index, length) : this.carve(index, place, length);
138 block.used = true;
139 if (this.mode === "NEXT") this.rover = block.end() === this.size ? 0 : block.end();
140 return block;
141 }
142
143 carve(index, place, length) {
144 const selected = this.blocks[index];
145 const parts = [];
146 if (place > selected.start) parts.push(this.register(new Block(selected.start, place - selected.start)));
147 const block = new Block(place, selected.end() - place);
148 parts.push(block);
149 if (block.size - length >= MIN_BLOCK) {
150 parts.push(this.register(new Block(place + length, block.size - length)));
151 block.size = length;
152 }
153 this.blocks.splice(index, 1, ...parts);
154 return block;
155 }
156
157 halve(index, length) {
158 const block = this.blocks[index];
159 while (block.size > length) {
160 block.size /= 2;
161 this.blocks.splice(index + 1, 0, this.register(new Block(block.end(), block.size)));
162 }
163 return block;
164 }
165
166 mergeNext(index) {
167 this.blocks[index].size += this.blocks.splice(index + 1, 1)[0].size;
168 }
169
170 buddyIndex(index) {
171 const block = this.blocks[index];
172 return block.start % (2 * block.size) === 0 ? index + 1 : index - 1;
173 }
174
175 coalesce(index) {
176 let merges = 0;
177 if (this.isBuddy()) {
178 while (this.blocks[index].size < this.size) {
179 const partner = this.buddyIndex(index);
180 const other = this.blocks[partner];
181 if (other.used || other.size !== this.blocks[index].size) break;
182 index = Math.min(index, partner);
183 this.mergeNext(index);
184 ++merges;
185 }
186 } else {
187 if (index + 1 < this.blocks.length && !this.blocks[index + 1].used) {
188 this.mergeNext(index);
189 ++merges;
190 }
191 if (index > 0 && !this.blocks[index - 1].used) {
192 --index;
193 this.mergeNext(index);
194 ++merges;
195 }
196 }
197 return [this.register(this.blocks[index]), merges];
198 }
199
200 release(block) {
201 block.used = false;
202 block.owner = null;
203 block.requested = 0;
204 block.contents = [];
205 return this.coalesce(this.blocks.indexOf(block));
206 }
207
208 shrink(index, length) {
209 const block = this.blocks[index];
210 if (this.isBuddy()) {
211 if (length === block.size) return "kept";
212 this.halve(index, length);
213 return "shrunk";
214 }
215 if (block.size - length < MIN_BLOCK) return "kept";
216 this.blocks.splice(index + 1, 0, new Block(block.start + length, block.size - length));
217 block.size = length;
218 this.coalesce(index + 1);
219 return "shrunk";
220 }
221
222 growIntoNext(index, length) {
223 const block = this.blocks[index];
224 if (index + 1 === this.blocks.length || this.blocks[index + 1].used) return null;
225 const total = block.size + this.blocks[index + 1].size;
226 if (total < length) return null;
227 this.blocks.splice(index + 1, 1);
228 if (total - length >= MIN_BLOCK) {
229 block.size = length;
230 this.blocks.splice(index + 1, 0, this.register(new Block(block.end(), total - length)));
231 } else block.size = total;
232 return "grown";
233 }
234
235 growBuddy(index, length) {
236 const block = this.blocks[index];
237 let size = block.size, last = index;
238 while (size < length) {
239 ++last;
240 if (block.start % (2 * size) !== 0 || last === this.blocks.length) return null;
241 const partner = this.blocks[last];
242 if (partner.used || partner.size !== size) return null;
243 size *= 2;
244 }
245 this.blocks.splice(index + 1, last - index);
246 block.size = length;
247 return "grown";
248 }
249
250 resize(block, length) {
251 const index = this.blocks.indexOf(block);
252 if (length <= block.size) return this.shrink(index, length);
253 return this.isBuddy() ? this.growBuddy(index, length) : this.growIntoNext(index, length);
254 }
255
256 extend(amount) {
257 this.blocks.push(new Block(this.size, amount));
258 this.size += amount;
259 return this.coalesce(this.blocks.length - 1)[0];
260 }
261
262 trim() {
263 let released = 0;
264 if (this.isBuddy()) {
265 while (!this.blocks.at(-1).used && this.blocks.at(-1).size * 2 === this.size) {
266 released += this.blocks.pop().size;
267 this.size /= 2;
268 }
269 } else if (this.blocks.length >= 2 && !this.blocks.at(-1).used) {
270 const last = this.blocks.pop();
271 released = last.size;
272 this.size = last.start;
273 }
274 if (this.rover >= this.size) this.rover = 0;
275 return released;
276 }
277
278 compact() {
279 const moved = [];
280 let address = 0;
281 const packed = this.usedBlocks();
282 for (const block of packed) {
283 if (block.start !== address) {
284 moved.push([block, block.payload()]);
285 block.start = address;
286 }
287 address += block.size;
288 }
289 if (address < this.size) packed.push(this.register(new Block(address, this.size - address)));
290 this.blocks = packed;
291 this.rover = 0;
292 return [moved, address];
293 }
294}

296function runExperiment(mode, size, seed, operations, maxRequest) {
297 const heap = new Heap(mode, size);
298 let state = seed, used = 0, peak = 0, successes = 0, failures = 0;
299 const live = [];
300 for (let operation = 0; operation < operations; ++operation) {
301 state = nextRandom(state);
302 const allocating = live.length === 0 || state % 3 !== 0;
303 state = nextRandom(state);
304 if (allocating) {
305 const size = 1 + state % maxRequest;
306 const block = heap.allocate(heap.requiredLength(size), UNIT);
307 if (block === null) ++failures;
308 else {
309 ++successes;
310 live.push(block);
311 used += block.size;
312 }
313 } else {
314 const block = live.splice(state % live.length, 1)[0];
315 used -= block.size;
316 heap.release(block);
317 }
318 peak = Math.max(peak, used);
319 }
320 return `BENCH ${mode} ok=${successes} fail=${failures} peak=${peak} used=${used} ` +
321 `frag=${heap.fragmentation()} blocks=${heap.blocks.length}`;
322}

324class Allocator {
325 constructor() {
326 this.output = [];
327 this.reset(null);
328 this.handlers = {
329 HEAP: this.createHeap, MALLOC: this.malloc, CALLOC: this.calloc,
330 MEMALIGN: this.memalign, REALLOC: this.realloc, FREE: this.free,
331 FILL: this.fill, COPY: this.copy, READ: this.read,
332 ROOT: this.root, UNROOT: this.unroot, LINK: this.link, UNLINK: this.unlink,
333 GC: this.collect, SBRK: this.sbrk, TRIM: this.trim, COMPACT: this.compact,
334 WHO: this.who, DUMP: this.dump, MAP: this.showMap, FREELIST: this.freelist,
335 TREE: this.tree, STATS: this.stats, LEAKS: this.leaks,
336 HISTORY: this.history, BENCH: this.bench
337 };
338 }
339
340 reset(heap) {
341 this.heap = heap;
342 this.allocations = new Map();
343 this.freed = new Set();
344 this.roots = new Set();
345 this.references = new Map();
346 this.histories = new Map();
347 this.counters = Object.fromEntries(COUNTERS.map(name => [name, 0]));
348 this.peak = 0;
349 }
350
351 emit(...parts) { this.output.push(parts.map(String).join(" ")); }
352
353 record(name, line) {
354 this.emit(line);
355 if (!this.histories.has(name)) this.histories.set(name, []);
356 this.histories.get(name).push(line);
357 }
358
359 outOfMemory(name) {
360 ++this.counters.failures;
361 return new CommandError(`OUT OF MEMORY ${name}`);
362 }
363
364 requireLive(name) {
365 if (this.allocations.has(name)) return this.allocations.get(name);
366 if (this.freed.has(name)) throw new CommandError(`USE AFTER FREE ${name}`);
367 throw new CommandError(`UNKNOWN ID ${name}`);
368 }
369
370 requireRange(name, offset, length) {
371 const block = this.requireLive(name);
372 if (offset + length > block.requested) throw new CommandError(`OUT OF BOUNDS ${name}`);
373 return block;
374 }
375
376 requireFreeList() {
377 if (this.heap.isBuddy()) throw new CommandError("UNSUPPORTED IN BUDDY MODE");
378 }
379
380 reachable() {
381 const found = new Set(this.roots);
382 const pending = [...this.roots];
383 while (pending.length > 0) {
384 const name = pending.pop();
385 for (const target of this.references.get(name) ?? []) {
386 if (found.has(target)) continue;
387 found.add(target);
388 pending.push(target);
389 }
390 }
391 return found;
392 }
393
394 createHeap(args) {
395 const size = Number(args[0]), mode = args[1];
396 if (size % UNIT !== 0 || size < 32 || size > MAX_HEAP ||
397 (mode === "BUDDY" && !powerOfTwo(size))) throw new CommandError("BAD HEAP SIZE");
398 this.reset(new Heap(mode, size));
399 this.emit(`HEAP size=${size} mode=${mode}`);
400 }
401
402 allocateId(word, name, request, alignment, fill) {
403 if (request === 0) throw new CommandError("BAD SIZE");
404 if (this.allocations.has(name)) throw new CommandError(`DUPLICATE ID ${name}`);
405 const block = this.heap.allocate(this.heap.requiredLength(request), alignment);
406 if (block === null) throw this.outOfMemory(name);
407 block.owner = name;
408 block.requested = request;
409 block.contents = Array(request).fill(fill);
410 this.allocations.set(name, block);
411 this.freed.delete(name);
412 ++this.counters.allocs;
413 this.record(name, `${word} ${name} addr=${block.payload()} block=${block.size}`);
414 }
415
416 malloc(args) { this.allocateId("MALLOC", args[0], Number(args[1]), UNIT, UNINITIALIZED); }
417 calloc(args) { this.allocateId("CALLOC", args[0], Number(args[1]) * Number(args[2]), UNIT, 0); }
418
419 memalign(args) {
420 const [name, alignment, request] = [args[0], Number(args[1]), Number(args[2])];
421 this.requireFreeList();
422 if (!powerOfTwo(alignment) || alignment < UNIT || alignment > 4096)
423 throw new CommandError("BAD ALIGNMENT");
424 this.allocateId("MEMALIGN", name, request, alignment, UNINITIALIZED);
425 }
426
427 releaseId(name) {
428 const block = this.allocations.get(name);
429 this.allocations.delete(name);
430 this.freed.add(name);
431 this.roots.delete(name);
432 this.references.delete(name);
433 for (const targets of this.references.values()) targets.delete(name);
434 const size = block.size;
435 const [result, merges] = this.heap.release(block);
436 return `block=${size} merges=${merges} free=${result.area()}`;
437 }
438
439 freeId(name) {
440 if (this.freed.has(name)) throw new CommandError(`DOUBLE FREE ${name}`);
441 if (!this.allocations.has(name)) throw new CommandError(`UNKNOWN ID ${name}`);
442 const details = this.releaseId(name);
443 ++this.counters.frees;
444 this.record(name, `FREE ${name} ${details}`);
445 }
446
447 free(args) { this.freeId(args[0]); }
448
449 realloc(args) {
450 const name = args[0], request = Number(args[1]);
451 if (request === 0) return this.freeId(name);
452 if (this.freed.has(name)) throw new CommandError(`USE AFTER FREE ${name}`);
453 if (!this.allocations.has(name)) {
454 this.allocateId("MALLOC", name, request, UNIT, UNINITIALIZED);
455 return;
456 }
457 let block = this.allocations.get(name);
458 const length = this.heap.requiredLength(request);
459 const contents = block.contents.slice(0, request);
460 while (contents.length < request) contents.push(UNINITIALIZED);
461 const original = block.payload();
462 let kind = this.heap.resize(block, length);
463 if (kind === null) {
464 const moved = this.heap.allocate(length, UNIT);
465 if (moved === null) throw this.outOfMemory(name);
466 moved.owner = name;
467 this.allocations.set(name, moved);
468 this.heap.release(block);
469 ++this.counters.moves;
470 block = moved;
471 kind = `moved from=${original}`;
472 }
473 block.requested = request;
474 block.contents = contents;
475 ++this.counters.reallocs;
476 this.record(name, `REALLOC ${name} addr=${block.payload()} block=${block.size} ${kind}`);
477 }
478
479 fill(args) {
480 const [name, start, count, value] = [args[0], Number(args[1]), Number(args[2]), Number(args[3])];
481 const block = this.requireRange(name, start, count);
482 block.contents.fill(value, start, start + count);
483 this.emit(`FILL ${name} ${start}+${count} = ${value}`);
484 }
485
486 copy(args) {
487 const targetName = args[0], targetOffset = Number(args[1]);
488 const sourceName = args[2], sourceOffset = Number(args[3]), count = Number(args[4]);
489 const target = this.requireRange(targetName, targetOffset, count);
490 const source = this.requireRange(sourceName, sourceOffset, count);
491 const values = source.contents.slice(sourceOffset, sourceOffset + count);
492 target.contents.splice(targetOffset, count, ...values);
493 this.emit(`COPY ${targetName} ${targetOffset}+${count} from ${sourceName} ${sourceOffset}`);
494 }
495
496 read(args) {
497 const name = args[0], start = Number(args[1]), count = Number(args[2]);
498 const block = this.requireRange(name, start, count);
499 const values = block.contents.slice(start, start + count)
500 .map(v => v === UNINITIALIZED ? "?" : String(v)).join(" ");
501 this.emit(`READ ${name} ${start}+${count} = ${values}`);
502 }
503
504 root(args) {
505 const name = args[0];
506 this.requireLive(name);
507 this.roots.add(name);
508 this.emit("ROOT", name);
509 }
510
511 unroot(args) {
512 const name = args[0];
513 this.requireLive(name);
514 if (!this.roots.has(name)) throw new CommandError(`NOT ROOT ${name}`);
515 this.roots.delete(name);
516 this.emit("UNROOT", name);
517 }
518
519 link(args) {
520 const [source, target] = args;
521 this.requireLive(source);
522 this.requireLive(target);
523 if (!this.references.has(source)) this.references.set(source, new Set());
524 this.references.get(source).add(target);
525 this.emit("LINK", source, target);
526 }
527
528 unlink(args) {
529 const [source, target] = args;
530 this.requireLive(source);
531 this.requireLive(target);
532 if (!(this.references.get(source)?.has(target)))
533 throw new CommandError(`NO LINK ${source} ${target}`);
534 this.references.get(source).delete(target);
535 this.emit("UNLINK", source, target);
536 }
537
538 collect() {
539 const found = this.reachable();
540 const garbage = [...this.allocations.entries()].filter(([name]) => !found.has(name))
541 .map(([, block]) => block).sort((a, b) => a.start - b.start);
542 this.emit(`GC collected=${garbage.length} bytes=${garbage.reduce((sum, block) => sum + block.size, 0)}`);
543 for (const block of garbage) {
544 const name = block.owner;
545 this.record(name, `COLLECT ${name} ${this.releaseId(name)}`);
546 }
547 this.counters.collected += garbage.length;
548 }
549
550 sbrk(args) {
551 const amount = Number(args[0]);
552 const heap = this.heap;
553 const valid = heap.isBuddy()
554 ? amount === heap.size && 2 * heap.size <= MAX_HEAP
555 : amount % UNIT === 0 && amount >= MIN_BLOCK && heap.size + amount <= MAX_HEAP;
556 if (!valid) throw new CommandError("BAD SBRK SIZE");
557 const result = heap.extend(amount);
558 this.emit(`SBRK heap=${heap.size} free=${result.area()}`);
559 }
560
561 trim() {
562 const released = this.heap.trim();
563 this.emit(`TRIM heap=${this.heap.size} released=${released}`);
564 }
565
566 compact() {
567 this.requireFreeList();
568 const [moved, used] = this.heap.compact();
569 this.counters.moves += moved.length;
570 const free = used < this.heap.size ? `${used}+${this.heap.size - used}` : "none";
571 this.emit(`COMPACT moved=${moved.length} free=${free}`);
572 for (const [block, from] of moved)
573 this.record(block.owner, `MOVE ${block.owner} ${from} -> ${block.payload()}`);
574 }
575
576 who(args) {
577 const address = Number(args[0]);
578 if (address >= this.heap.size) return this.emit("WHO", address, "outside");
579 const block = this.heap.blocks[this.heap.locate(address)];
580 if (!block.used) return this.emit("WHO", address, "free", block.area());
581 if (address < block.payload()) return this.emit("WHO", address, "header", block.owner);
582 const offset = address - block.payload();
583 this.emit("WHO", address, offset < block.requested ? "data" : "slack", `${block.owner}+${offset}`);
584 }
585
586 dump() {
587 const heap = this.heap;
588 this.emit(`DUMP mode=${heap.mode} heap=${heap.size} blocks=${heap.blocks.length}` +
589 (heap.mode === "NEXT" ? ` rover=${heap.rover}` : ""));
590 for (const block of heap.blocks) {
591 if (block.used) this.emit(block.start, block.size, "USED", block.owner, `req=${block.requested}`);
592 else this.emit(block.start, block.size, "FREE");
593 }
594 }
595
596 showMap() {
597 const heap = this.heap;
598 this.emit(`MAP heap=${heap.size}`);
599 const allUnits = heap.blocks.map(block => block.mapUnits()).join("");
600 for (let first = 0; first < allUnits.length; first += 64) {
601 this.emit(String(first * UNIT).padStart(5, "0"), allUnits.slice(first, first + 64));
602 }
603 }
604
605 freelist() {
606 const free = this.heap.freeBlocks();
607 if (this.heap.mode === "SEGFIT") free.sort((a, b) => b.stamp - a.stamp);
608 this.emit(`FREELIST count=${free.length}`);
609 const classes = [...new Set(free.map(block => sizeClass(block.size)))].sort((a, b) => a - b);
610 for (const exponent of classes) {
611 const areas = free.filter(block => sizeClass(block.size) === exponent).map(block => block.area());
612 this.emit(`${2 ** exponent}-${2 ** (exponent + 1) - 1}:`, ...areas);
613 }
614 }
615
616 treeNode(leaves, start, size, depth) {
617 const label = " ".repeat(depth * 2) + `${start}+${size}`;
618 const block = leaves.get(start);
619 if (block && block.size === size) {
620 this.emit(label, block.used ? "USED" : "FREE", ...(block.used ? [block.owner] : []));
621 return;
622 }
623 this.emit(label, "split");
624 const half = size / 2;
625 this.treeNode(leaves, start, half, depth + 1);
626 this.treeNode(leaves, start + half, half, depth + 1);
627 }
628
629 tree() {
630 if (!this.heap.isBuddy()) throw new CommandError("REQUIRES BUDDY MODE");
631 this.emit(`TREE heap=${this.heap.size}`);
632 const leaves = new Map(this.heap.blocks.map(block => [block.start, block]));
633 this.treeNode(leaves, 0, this.heap.size, 0);
634 }

636 stats() {
637 const used = this.heap.usedBlocks(), free = this.heap.freeBlocks();
638 const freeTotal = free.reduce((sum, block) => sum + block.size, 0);
639 const largest = free.length ? Math.max(...free.map(block => block.size)) : 0;
640 this.emit(`STATS used=${used.reduce((sum, block) => sum + block.size, 0)} used_blocks=${used.length}`,
641 `free=${freeTotal} free_blocks=${free.length} largest=${largest}`,
642 `frag=${fragmentationPercent(freeTotal, largest)}`,
643 `requested=${used.reduce((sum, block) => sum + block.requested, 0)} peak=${this.peak}`);
644 this.emit("OPS", ...COUNTERS.map(name => `${name}=${this.counters[name]}`));
645 }

647 leaks() {
648 const found = this.reachable();
649 const names = [...this.allocations.keys()].sort();
650 const requested = names.reduce((sum, name) => sum + this.allocations.get(name).requested, 0);
651 const lost = names.filter(name => !found.has(name)).length;
652 this.emit(`LEAKS count=${names.length} requested=${requested} lost=${lost}`);
653 for (const name of names) {
654 const block = this.allocations.get(name);
655 this.emit(`LEAK ${name} addr=${block.payload()} block=${block.size}`,
656 `req=${block.requested} ${found.has(name) ? "reachable" : "lost"}`);
657 }
658 }

676 try {
677 if (this.heap === null && command !== "HEAP" && command !== "BENCH")
678 throw new CommandError("NO HEAP");

680 } catch (error) {
681 if (!(error instanceof CommandError)) throw error;
682 this.emit("ERROR", error.message);
683 }
684 if (this.heap !== null) this.peak = Math.max(this.peak, this.heap.usedTotal());
