20 return $number > 0 && ($number & ($number - 1)) === 0;

38final class Block
39{
40 public bool $used = false;
41 public ?string $owner = null;
42 public int $requested = 0;
43 public array $contents = [];
44 public int $stamp = 0;
45
46 public function __construct(public int $start, public int $size) {}
47
48 public function end(): int { return $this->start + $this->size; }
49 public function payload(): int { return $this->start + HEADER; }
50 public function area(): string { return $this->start . '+' . $this->size; }
51
52 public function mapUnits(): string
53 {
54 $count = intdiv($this->size, UNIT) - 1;
55 if (!$this->used) return 'F' . str_repeat('.', $count);
56 $out = 'H';
57 for ($i = 0; $i < $count; ++$i) {
58 $out .= UNIT * $i < $this->requested ? '#' : '-';
59 }
60 return $out;
61 }

64final class Heap
65{
66 public int $rover = 0;
67 public int $registrations = 0;
68 public array $blocks;
69
70 public function __construct(public string $mode, public int $size)
71 {
72 $this->blocks = [$this->register(new Block(0, $size))];
73 }
74
75 public function register(Block $block): Block
76 {
77 $block->stamp = ++$this->registrations;
78 return $block;
79 }
80
81 public function isBuddy(): bool { return $this->mode === 'BUDDY'; }
82
83 public function requiredLength(int $request): int
84 {
85 $length = intdiv($request + UNIT - 1, UNIT) * UNIT + HEADER;
86 if (!$this->isBuddy()) return $length;
87 $power = MIN_BUDDY_BLOCK;
88 while ($power < $length) $power *= 2;
89 return $power;
90 }
91
92 public function indexOf(Block $block): int
93 {
94 return array_search($block, $this->blocks, true);
95 }
96
97 public function locate(int $address): int
98 {
99 foreach ($this->blocks as $index => $block) {
100 if ($block->end() > $address) return $index;
101 }
102 throw new LogicException('address is outside the heap');
103 }
104
105 public function usedBlocks(): array
106 {
107 return array_values(array_filter($this->blocks, static fn(Block $b): bool => $b->used));
108 }
109
110 public function freeBlocks(): array
111 {
112 return array_values(array_filter($this->blocks, static fn(Block $b): bool => !$b->used));
113 }
114
115 public function usedTotal(): int
116 {
117 return total(array_map(static fn(Block $b): int => $b->size, $this->usedBlocks()));
118 }
119
120 public function fragmentation(): int
121 {
122 $sizes = array_map(static fn(Block $b): int => $b->size, $this->freeBlocks());
123 return fragmentationPercent(total($sizes), $sizes ? max($sizes) : 0);
124 }
125
126 public function placement(Block $block, int $alignment): int
127 {
128 $place = intdiv($block->start + HEADER + $alignment - 1, $alignment)
129 * $alignment - HEADER;
130 if ($place > $block->start && $place - $block->start < MIN_BLOCK) $place += $alignment;
131 return $place;
132 }
133
134 public function rank(int $index, int $roverIndex): array
135 {
136 $block = $this->blocks[$index];
137 return match ($this->mode) {
138 'NEXT' => [($index - $roverIndex + count($this->blocks)) % count($this->blocks), 0],
139 'BEST', 'BUDDY' => [$block->size, $block->start],
140 'WORST' => [-$block->size, $block->start],
141 'SEGFIT' => [sizeClass($block->size), -$block->stamp],
142 default => [$block->start, 0]
143 };
144 }
145
146 public function choose(int $length, int $alignment): ?array
147 {
148 $roverIndex = $this->locate($this->rover);
149 $chosen = null;
150 foreach ($this->blocks as $index => $block) {
151 if ($block->used) continue;
152 $place = $this->placement($block, $alignment);
153 if ($place + $length > $block->end()) continue;
154 $rank = $this->rank($index, $roverIndex);
155 if ($chosen === null || $rank < $chosen[0]) $chosen = [$rank, $index, $place];
156 }
157 return $chosen;
158 }
159
160 public function allocate(int $length, int $alignment): ?Block
161 {
162 $choice = $this->choose($length, $alignment);
163 if ($choice === null) return null;
164 [, $index, $place] = $choice;
165 $block = $this->isBuddy()
166 ? $this->halve($index, $length) : $this->carve($index, $place, $length);
167 $block->used = true;
168 if ($this->mode === 'NEXT') {
169 $this->rover = $block->end() === $this->size ? 0 : $block->end();
170 }
171 return $block;
172 }
173
174 public function carve(int $index, int $place, int $length): Block
175 {
176 $original = $this->blocks[$index];
177 $parts = [];
178 if ($place > $original->start) {
179 $parts[] = $this->register(new Block($original->start, $place - $original->start));
180 }
181 $block = new Block($place, $original->end() - $place);
182 $parts[] = $block;
183 if ($block->size - $length >= MIN_BLOCK) {
184 $parts[] = $this->register(new Block($place + $length, $block->size - $length));
185 $block->size = $length;
186 }
187 array_splice($this->blocks, $index, 1, $parts);
188 return $block;
189 }
190
191 public function halve(int $index, int $length): Block
192 {
193 $block = $this->blocks[$index];
194 while ($block->size > $length) {
195 $block->size = intdiv($block->size, 2);
196 array_splice($this->blocks, $index + 1, 0,
197 [$this->register(new Block($block->end(), $block->size))]);
198 }
199 return $block;
200 }
201
202 public function mergeNext(int $index): void
203 {
204 $removed = array_splice($this->blocks, $index + 1, 1)[0];
205 $this->blocks[$index]->size += $removed->size;
206 }
207
208 public function buddyIndex(int $index): int
209 {
210 $block = $this->blocks[$index];
211 return $block->start % (2 * $block->size) === 0 ? $index + 1 : $index - 1;
212 }
213
214 public function coalesce(int $index): array
215 {
216 $merges = 0;
217 if ($this->isBuddy()) {
218 while ($this->blocks[$index]->size < $this->size) {
219 $partner = $this->buddyIndex($index);
220 $other = $this->blocks[$partner];
221 if ($other->used || $other->size !== $this->blocks[$index]->size) break;
222 $index = min($index, $partner);
223 $this->mergeNext($index);
224 ++$merges;
225 }
226 } else {
227 if ($index + 1 < count($this->blocks) && !$this->blocks[$index + 1]->used) {
228 $this->mergeNext($index);
229 ++$merges;
230 }
231 if ($index > 0 && !$this->blocks[$index - 1]->used) {
232 --$index;
233 $this->mergeNext($index);
234 ++$merges;
235 }
236 }
237 return [$this->register($this->blocks[$index]), $merges];
238 }
239
240 public function release(Block $block): array
241 {
242 $block->used = false;
243 $block->owner = null;
244 $block->requested = 0;
245 $block->contents = [];
246 return $this->coalesce($this->indexOf($block));
247 }
248
249 public function shrink(int $index, int $length): string
250 {
251 $block = $this->blocks[$index];
252 if ($this->isBuddy()) {
253 if ($length === $block->size) return 'kept';
254 $this->halve($index, $length);
255 return 'shrunk';
256 }
257 if ($block->size - $length < MIN_BLOCK) return 'kept';
258 array_splice($this->blocks, $index + 1, 0,
259 [new Block($block->start + $length, $block->size - $length)]);
260 $block->size = $length;
261 $this->coalesce($index + 1);
262 return 'shrunk';
263 }
264
265 public function growIntoNext(int $index, int $length): ?string
266 {
267 $block = $this->blocks[$index];
268 if ($index + 1 === count($this->blocks) || $this->blocks[$index + 1]->used) return null;
269 $total = $block->size + $this->blocks[$index + 1]->size;
270 if ($total < $length) return null;
271 array_splice($this->blocks, $index + 1, 1);
272 if ($total - $length >= MIN_BLOCK) {
273 $block->size = $length;
274 array_splice($this->blocks, $index + 1, 0,
275 [$this->register(new Block($block->end(), $total - $length))]);
276 } else $block->size = $total;
277 return 'grown';
278 }
279
280 public function growBuddy(int $index, int $length): ?string
281 {
282 $block = $this->blocks[$index];
283 $size = $block->size;
284 $last = $index;
285 while ($size < $length) {
286 ++$last;
287 if ($block->start % (2 * $size) !== 0 || $last === count($this->blocks)) return null;
288 $partner = $this->blocks[$last];
289 if ($partner->used || $partner->size !== $size) return null;
290 $size *= 2;
291 }
292 array_splice($this->blocks, $index + 1, $last - $index);
293 $block->size = $length;
294 return 'grown';
295 }
296
297 public function resize(Block $block, int $length): ?string
298 {
299 $index = $this->indexOf($block);
300 if ($length <= $block->size) return $this->shrink($index, $length);
301 return $this->isBuddy()
302 ? $this->growBuddy($index, $length) : $this->growIntoNext($index, $length);
303 }
304
305 public function extend(int $amount): Block
306 {
307 $this->blocks[] = new Block($this->size, $amount);
308 $this->size += $amount;
309 return $this->coalesce(count($this->blocks) - 1)[0];
310 }
311
312 public function trim(): int
313 {
314 $released = 0;
315 if ($this->isBuddy()) {
316 while (!$this->blocks[array_key_last($this->blocks)]->used &&
317 2 * $this->blocks[array_key_last($this->blocks)]->size === $this->size) {
318 $released += array_pop($this->blocks)->size;
319 $this->size = intdiv($this->size, 2);
320 }
321 } elseif (count($this->blocks) >= 2 && !$this->blocks[array_key_last($this->blocks)]->used) {
322 $last = array_pop($this->blocks);
323 $released = $last->size;
324 $this->size = $last->start;
325 }
326 if ($this->rover >= $this->size) $this->rover = 0;
327 return $released;
328 }
329
330 public function compact(): array
331 {
332 $moved = [];
333 $address = 0;
334 $packed = $this->usedBlocks();
335 foreach ($packed as $block) {
336 if ($block->start !== $address) {
337 $moved[] = [$block, $block->payload()];
338 $block->start = $address;
339 }
340 $address += $block->size;
341 }
342 if ($address < $this->size) {
343 $packed[] = $this->register(new Block($address, $this->size - $address));
344 }
345 $this->blocks = $packed;
346 $this->rover = 0;
347 return [$moved, $address];
348 }
349}

351function experiment(string $mode, int $size, int $seed, int $operations, int $maxRequest): string
352{
353 $heap = new Heap($mode, $size);
354 $state = $seed;
355 $live = [];
356 $used = $peak = $success = $failures = 0;
357 for ($i = 0; $i < $operations; ++$i) {
358 $state = RANDOM_MULTIPLIER * $state % RANDOM_MODULUS;
359 $allocating = !$live || $state % 3 !== 0;
360 $state = RANDOM_MULTIPLIER * $state % RANDOM_MODULUS;
361 if ($allocating) {
362 $requested = 1 + $state % $maxRequest;
363 $block = $heap->allocate($heap->requiredLength($requested), UNIT);
364 if ($block === null) ++$failures;
365 else {
366 ++$success;
367 $live[] = $block;
368 $used += $block->size;
369 }
370 } else {
371 $block = array_splice($live, $state % count($live), 1)[0];
372 $used -= $block->size;
373 $heap->release($block);
374 }
375 $peak = max($peak, $used);
376 }
377 return 'BENCH ' . $mode . ' ok=' . $success . ' fail=' . $failures .
378 ' peak=' . $peak . ' used=' . $used . ' frag=' . $heap->fragmentation() .
379 ' blocks=' . count($heap->blocks);
380}

382final class Allocator
383{
384 public array $output = [];
385 private ?Heap $heap = null;
386 private array $allocations = [];
387 private array $freed = [];
388 private array $roots = [];
389 private array $references = [];
390 private array $histories = [];
391 private array $counters = [];
392 private int $peak = 0;
393
394 private const METHODS = [
395 'HEAP' => 'createHeap', 'MALLOC' => 'malloc', 'CALLOC' => 'calloc',
396 'MEMALIGN' => 'memalign', 'REALLOC' => 'realloc', 'FREE' => 'free',
397 'FILL' => 'fill', 'COPY' => 'copy', 'READ' => 'read',
398 'ROOT' => 'root', 'UNROOT' => 'unroot', 'LINK' => 'link', 'UNLINK' => 'unlink',
399 'GC' => 'collect', 'SBRK' => 'sbrk', 'TRIM' => 'trim', 'COMPACT' => 'compact',
400 'WHO' => 'who', 'DUMP' => 'dump', 'MAP' => 'showMap', 'FREELIST' => 'freelist',
401 'TREE' => 'tree', 'STATS' => 'stats', 'LEAKS' => 'leaks',
402 'HISTORY' => 'history', 'BENCH' => 'bench'
403 ];
404
405 public function __construct() { $this->reset(null); }
406
407 private function reset(?Heap $heap): void
408 {
409 $this->heap = $heap;
410 $this->allocations = $this->freed = $this->roots = [];
411 $this->references = $this->histories = [];
412 $this->counters = array_fill_keys(COUNTERS, 0);
413 $this->peak = 0;
414 }
415
416 private function emit(string ...$parts): void { $this->output[] = implode(' ', $parts); }
417
418 private function record(string $name, string $line): void
419 {
420 $this->emit($line);
421 $this->histories[$name][] = $line;
422 }
423
424 private function outOfMemory(string $name): CommandError
425 {
426 ++$this->counters['failures'];
427 return new CommandError('OUT OF MEMORY ' . $name);
428 }
429
430 private function requireLive(string $name): Block
431 {
432 if (isset($this->allocations[$name])) return $this->allocations[$name];
433 if (isset($this->freed[$name])) throw new CommandError('USE AFTER FREE ' . $name);
434 throw new CommandError('UNKNOWN ID ' . $name);
435 }
436
437 private function requireRange(string $name, int $offset, int $length): Block
438 {
439 $block = $this->requireLive($name);
440 if ($offset + $length > $block->requested) throw new CommandError('OUT OF BOUNDS ' . $name);
441 return $block;
442 }
443
444 private function requireFreeList(): void
445 {
446 if ($this->heap->isBuddy()) throw new CommandError('UNSUPPORTED IN BUDDY MODE');
447 }
448
449 private function reachable(): array
450 {
451 $found = $this->roots;
452 $pending = array_keys($this->roots);
453 while ($pending) {
454 $name = array_pop($pending);
455 foreach ($this->references[$name] ?? [] as $target => $_) {
456 if (isset($found[$target])) continue;
457 $found[$target] = true;
458 $pending[] = $target;
459 }
460 }
461 return $found;
462 }
463
464 private function createHeap(array $args): void
465 {
466 $size = (int)$args[0];
467 $mode = $args[1];
468 if ($size % UNIT !== 0 || $size < 32 || $size > MAX_HEAP ||
469 ($mode === 'BUDDY' && !isPowerOfTwo($size))) {
470 throw new CommandError('BAD HEAP SIZE');
471 }
472 $this->reset(new Heap($mode, $size));
473 $this->emit("HEAP size=$size mode=$mode");
474 }
475
476 private function allocateId(string $word, string $name, int $request,
477 int $alignment, int $fill): void
478 {
479 if ($request === 0) throw new CommandError('BAD SIZE');
480 if (isset($this->allocations[$name])) throw new CommandError('DUPLICATE ID ' . $name);
481 $block = $this->heap->allocate($this->heap->requiredLength($request), $alignment);
482 if ($block === null) throw $this->outOfMemory($name);
483 $block->owner = $name;
484 $block->requested = $request;
485 $block->contents = array_fill(0, $request, $fill);
486 $this->allocations[$name] = $block;
487 unset($this->freed[$name]);
488 ++$this->counters['allocs'];
489 $this->record($name, "$word $name addr={$block->payload()} block={$block->size}");
490 }
491
492 private function malloc(array $args): void
493 {
494 $this->allocateId('MALLOC', $args[0], (int)$args[1], UNIT, UNINITIALIZED);
495 }
496
497 private function calloc(array $args): void
498 {
499 $this->allocateId('CALLOC', $args[0], (int)$args[1] * (int)$args[2], UNIT, 0);
500 }
501
502 private function memalign(array $args): void
503 {
504 [$name, $alignment, $request] = [$args[0], (int)$args[1], (int)$args[2]];
505 $this->requireFreeList();
506 if (!isPowerOfTwo($alignment) || $alignment < UNIT || $alignment > 4096) {
507 throw new CommandError('BAD ALIGNMENT');
508 }
509 $this->allocateId('MEMALIGN', $name, $request, $alignment, UNINITIALIZED);
510 }
511
512 private function releaseId(string $name): string
513 {
514 $block = $this->allocations[$name];
515 unset($this->allocations[$name]);
516 $size = $block->size;
517 $this->freed[$name] = true;
518 unset($this->roots[$name], $this->references[$name]);
519 foreach ($this->references as &$targets) unset($targets[$name]);
520 unset($targets);
521 [$result, $merges] = $this->heap->release($block);
522 return "block=$size merges=$merges free={$result->area()}";
523 }
524
525 private function freeId(string $name): void
526 {
527 if (isset($this->freed[$name])) throw new CommandError('DOUBLE FREE ' . $name);
528 if (!isset($this->allocations[$name])) throw new CommandError('UNKNOWN ID ' . $name);
529 $details = $this->releaseId($name);
530 ++$this->counters['frees'];
531 $this->record($name, "FREE $name $details");
532 }
533
534 private function free(array $args): void { $this->freeId($args[0]); }
535
536 private function realloc(array $args): void
537 {
538 [$name, $request] = [$args[0], (int)$args[1]];
539 if ($request === 0) { $this->freeId($name); return; }
540 if (isset($this->freed[$name])) throw new CommandError('USE AFTER FREE ' . $name);
541 if (!isset($this->allocations[$name])) {
542 $this->allocateId('MALLOC', $name, $request, UNIT, UNINITIALIZED);
543 return;
544 }
545 $block = $this->allocations[$name];
546 $length = $this->heap->requiredLength($request);
547 $contents = array_slice($block->contents, 0, $request);
548 if (count($contents) < $request) {
549 array_push($contents, ...array_fill(0, $request - count($contents), UNINITIALIZED));
550 }
551 $origin = $block->payload();
552 $kind = $this->heap->resize($block, $length);
553 if ($kind === null) {
554 $moved = $this->heap->allocate($length, UNIT);
555 if ($moved === null) throw $this->outOfMemory($name);
556 $moved->owner = $name;
557 $this->allocations[$name] = $moved;
558 $this->heap->release($block);
559 ++$this->counters['moves'];
560 $block = $moved;
561 $kind = "moved from=$origin";
562 }
563 $block->requested = $request;
564 $block->contents = $contents;
565 ++$this->counters['reallocs'];
566 $this->record($name, "REALLOC $name addr={$block->payload()} block={$block->size} $kind");
567 }

570 {
571 [$name, $offset, $length, $value] = [$args[0], (int)$args[1], (int)$args[2], (int)$args[3]];
572 $block = $this->requireRange($name, $offset, $length);
573 array_splice($block->contents, $offset, $length, array_fill(0, $length, $value));
574 $this->emit("FILL $name $offset+$length = $value");
575 }

577 private function copy(array $args): void
578 {
579 [$targetName, $targetOffset, $sourceName, $sourceOffset, $length] =
580 [$args[0], (int)$args[1], $args[2], (int)$args[3], (int)$args[4]];
581 $target = $this->requireRange($targetName, $targetOffset, $length);

586 }

595 }

635 private function collect(array $args): void
636 {
637 $reachable = $this->reachable();
638 $garbage = [];
639 foreach ($this->allocations as $name => $block) {
640 if (!isset($reachable[$name])) $garbage[] = $block;
641 }
642 usort($garbage, static fn(Block $a, Block $b): int => $a->start <=> $b->start);
643 $size = total(array_map(static fn(Block $b): int => $b->size, $garbage));
644 $this->emit('GC collected=' . count($garbage) . ' bytes=' . $size);
645 foreach ($garbage as $block) {
646 $name = $block->owner;
647 $this->record($name, "COLLECT $name " . $this->releaseId($name));
648 }
649 $this->counters['collected'] += count($garbage);
650 }

652 private function sbrk(array $args): void
653 {
654 $amount = (int)$args[0];
655 $heap = $this->heap;
656 $valid = $heap->isBuddy()
657 ? $amount === $heap->size && 2 * $heap->size <= MAX_HEAP
658 : $amount % UNIT === 0 && $amount >= MIN_BLOCK && $heap->size + $amount <= MAX_HEAP;
659 if (!$valid) throw new CommandError('BAD SBRK SIZE');
660 $result = $heap->extend($amount);
661 $this->emit("SBRK heap={$heap->size} free={$result->area()}");
662 }

670 private function compact(array $args): void
671 {
672 $this->requireFreeList();
673 [$moved, $used] = $this->heap->compact();
674 $this->counters['moves'] += count($moved);
675 $free = $used < $this->heap->size ? $used . '+' . ($this->heap->size - $used) : 'none';
676 $this->emit('COMPACT moved=' . count($moved) . ' free=' . $free);
677 foreach ($moved as [$block, $origin]) {
678 $this->record($block->owner, "MOVE {$block->owner} $origin -> {$block->payload()}");
679 }
680 }

682 private function who(array $args): void
683 {
684 $address = (int)$args[0];
685 if ($address >= $this->heap->size) { $this->emit("WHO $address outside"); return; }
686 $block = $this->heap->blocks[$this->heap->locate($address)];
687 if (!$block->used) { $this->emit("WHO $address free {$block->area()}"); return; }
688 if ($address < $block->payload()) {
689 $this->emit("WHO $address header {$block->owner}");
690 return;
691 }
692 $offset = $address - $block->payload();
693 $region = $offset < $block->requested ? 'data' : 'slack';
694 $this->emit("WHO $address $region {$block->owner}+$offset");
695 }

721 private function freelist(array $args): void
722 {
723 $free = $this->heap->freeBlocks();
724 if ($this->heap->mode === 'SEGFIT') {
725 usort($free, static fn(Block $a, Block $b): int => $b->stamp <=> $a->stamp);
726 }
727 $this->emit('FREELIST count=' . count($free));
728 $classes = [];
729 foreach ($free as $block) $classes[sizeClass($block->size)] = true;
730 ksort($classes, SORT_NUMERIC);
731 foreach (array_keys($classes) as $power) {
732 $members = [];
733 foreach ($free as $block) {
734 if (sizeClass($block->size) === $power) $members[] = $block->area();
735 }
736 $this->emit((2 ** $power) . '-' . (2 ** ($power + 1) - 1) . ':', ...$members);
737 }

740 private function treeNode(array $leaves, int $start, int $size, int $depth): void
741 {
742 $label = str_repeat(' ', 2 * $depth) . $start . '+' . $size;
743 $block = $leaves[$start] ?? null;
744 if ($block !== null && $block->size === $size) {
745 $this->emit($block->used ? "$label USED {$block->owner}" : "$label FREE");
746 return;
747 }
748 $this->emit("$label split");
749 $half = intdiv($size, 2);
750 $this->treeNode($leaves, $start, $half, $depth + 1);
751 $this->treeNode($leaves, $start + $half, $half, $depth + 1);
752 }

772 ' free=' . $freeTotal . ' free_blocks=' . count($free) . ' largest=' . $largest .
773 ' frag=' . fragmentationPercent($freeTotal, $largest) .

777 $this->emit(...$parts);

781 {
782 $reachable = $this->reachable();
783 $names = array_keys($this->allocations);
784 sort($names, SORT_STRING);
785 $requested = 0; $lost = 0;
786 foreach ($names as $name) {
787 $requested += $this->allocations[$name]->requested;
788 if (!isset($reachable[$name])) ++$lost;
789 }
790 $this->emit('LEAKS count=' . count($names) . " requested=$requested lost=$lost");
791 foreach ($names as $name) {
792 $block = $this->allocations[$name];
793 $status = isset($reachable[$name]) ? 'reachable' : 'lost';
794 $this->emit("LEAK $name addr={$block->payload()} block={$block->size} " .
795 "req={$block->requested} $status");
796 }

819 public function execute(string $line): void
820 {
821 $words = explode(' ', $line);
822 $command = array_shift($words);
823 try {
824 if ($this->heap === null && $command !== 'HEAP' && $command !== 'BENCH') {
825 throw new CommandError('NO HEAP');
826 }
827 $this->{self::METHODS[$command]}($words);
828 } catch (CommandError $error) {
829 $this->emit('ERROR', $error->getMessage());
830 }
831 if ($this->heap !== null) $this->peak = max($this->peak, $this->heap->usedTotal());
832 }
833}
