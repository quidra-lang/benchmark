45class Block:
46 def __init__(self, start, size):
47 self.start = start
48 self.size = size
49 self.used = False
50 self.owner = None
51 self.requested = 0
52 self.contents = []
53 self.stamp = 0
54
55 def end(self):
56 return self.start + self.size
57
58 def payload(self):
59 return self.start + HEADER
60
61 def area(self):
62 return f"{self.start}+{self.size}"

64 def map_units(self):
65 rest = self.size // UNIT - 1

71class Heap:
72 def __init__(self, mode, size):
73 self.mode = mode
74 self.size = size
75 self.rover = 0
76 self.registrations = 0
77 self.blocks = [self.register(Block(0, size))]
78
79 def register(self, block):
80 self.registrations += 1
81 block.stamp = self.registrations
82 return block
83
84 def is_buddy(self):
85 return self.mode == "BUDDY"
86
87 def required_length(self, request):
88 length = (request + UNIT - 1) // UNIT * UNIT + HEADER
89 if not self.is_buddy():
90 return length
91 power = MIN_BUDDY_BLOCK
92 while power < length:
93 power *= 2
94 return power
95
96 def locate(self, address):
97 index = 0
98 while self.blocks[index].end() <= address:
99 index += 1
100 return index
101
102 def used_blocks(self):
103 return [block for block in self.blocks if block.used]
104
105 def free_blocks(self):
106 return [block for block in self.blocks if not block.used]
107
108 def used_total(self):
109 return sum(block.size for block in self.used_blocks())
110
111 def fragmentation(self):
112 sizes = [block.size for block in self.free_blocks()]
113 return fragmentation_percent(sum(sizes), max(sizes, default=0))
114
115 def placement(self, block, alignment):
116 place = (block.start + HEADER + alignment - 1) // alignment * alignment - HEADER
117 if 0 < place - block.start < MIN_BLOCK:
118 place += alignment
119 return place
120
121 def rank(self, index, rover_index):
122 block = self.blocks[index]
123 if self.mode == "NEXT":
124 return (index - rover_index) % len(self.blocks), 0
125 if self.mode in ("BEST", "BUDDY"):
126 return block.size, block.start
127 if self.mode == "WORST":
128 return -block.size, block.start
129 if self.mode == "SEGFIT":
130 return size_class(block.size), -block.stamp
131 return block.start, 0
132
133 def choose(self, length, alignment):
134 rover_index = self.locate(self.rover)
135 candidates = []
136 for index, block in enumerate(self.blocks):
137 if block.used:
138 continue
139 place = self.placement(block, alignment)
140 if place + length <= block.end():
141 candidates.append((self.rank(index, rover_index), index, place))
142 return min(candidates, default=None)
143
144 def allocate(self, length, alignment):
145 choice = self.choose(length, alignment)
146 if choice is None:
147 return None
148 _, index, place = choice
149 if self.is_buddy():
150 self.halve(index, length)
151 block = self.blocks[index]
152 else:
153 block = self.carve(index, place, length)
154 block.used = True
155 if self.mode == "NEXT":
156 self.rover = 0 if block.end() == self.size else block.end()
157 return block
158
159 def carve(self, index, place, length):
160 chosen = self.blocks[index]
161 parts = []
162 if place > chosen.start:
163 parts.append(self.register(Block(chosen.start, place - chosen.start)))
164 block = Block(place, chosen.end() - place)
165 parts.append(block)
166 if block.size - length >= MIN_BLOCK:
167 parts.append(self.register(Block(place + length, block.size - length)))
168 block.size = length
169 self.blocks[index:index + 1] = parts
170 return block
171
172 def halve(self, index, length):
173 block = self.blocks[index]
174 while block.size > length:
175 block.size //= 2
176 self.blocks.insert(index + 1, self.register(Block(block.end(), block.size)))
177
178 def merge_next(self, index):
179 self.blocks[index].size += self.blocks.pop(index + 1).size
180
181 def buddy_index(self, index):
182 block = self.blocks[index]
183 if block.start % (2 * block.size) == 0:
184 return index + 1
185 return index - 1
186
187 def coalesce(self, index):
188 merges = 0
189 if self.is_buddy():
190 while self.blocks[index].size < self.size:
191 partner = self.buddy_index(index)
192 buddy = self.blocks[partner]
193 if buddy.used or buddy.size != self.blocks[index].size:
194 break
195 index = min(index, partner)
196 self.merge_next(index)
197 merges += 1
198 else:
199 if index + 1 < len(self.blocks) and not self.blocks[index + 1].used:
200 self.merge_next(index)
201 merges += 1
202 if index > 0 and not self.blocks[index - 1].used:
203 index -= 1
204 self.merge_next(index)
205 merges += 1
206 return self.register(self.blocks[index]), merges
207
208 def release(self, block):
209 block.used = False
210 block.owner = None
211 block.requested = 0
212 block.contents = []
213 return self.coalesce(self.blocks.index(block))
214
215 def shrink(self, index, length):
216 block = self.blocks[index]
217 if self.is_buddy():
218 if length == block.size:
219 return "kept"
220 self.halve(index, length)
221 return "shrunk"
222 if block.size - length < MIN_BLOCK:
223 return "kept"
224 self.blocks.insert(index + 1, Block(block.start + length, block.size - length))
225 block.size = length
226 self.coalesce(index + 1)
227 return "shrunk"
228
229 def grow_into_next(self, index, length):
230 block = self.blocks[index]
231 if index + 1 == len(self.blocks) or self.blocks[index + 1].used:
232 return None
233 total = block.size + self.blocks[index + 1].size
234 if total < length:
235 return None
236 del self.blocks[index + 1]
237 if total - length >= MIN_BLOCK:
238 block.size = length
239 self.blocks.insert(index + 1, self.register(Block(block.end(), total - length)))
240 else:
241 block.size = total
242 return "grown"
243
244 def grow_buddy(self, index, length):
245 block = self.blocks[index]
246 size = block.size
247 last = index
248 while size < length:
249 last += 1
250 if block.start % (2 * size) != 0 or last == len(self.blocks):
251 return None
252 partner = self.blocks[last]
253 if partner.used or partner.size != size:
254 return None
255 size *= 2
256 del self.blocks[index + 1:last + 1]
257 block.size = length
258 return "grown"
259
260 def resize(self, block, length):
261 index = self.blocks.index(block)
262 if length <= block.size:
263 return self.shrink(index, length)
264 if self.is_buddy():
265 return self.grow_buddy(index, length)
266 return self.grow_into_next(index, length)
267
268 def extend(self, amount):
269 self.blocks.append(Block(self.size, amount))
270 self.size += amount
271 block, _ = self.coalesce(len(self.blocks) - 1)
272 return block
273
274 def trim(self):
275 released = 0
276 if self.is_buddy():
277 while not self.blocks[-1].used and 2 * self.blocks[-1].size == self.size:
278 released += self.blocks.pop().size
279 self.size //= 2
280 elif len(self.blocks) >= 2 and not self.blocks[-1].used:
281 last = self.blocks.pop()
282 released = last.size
283 self.size = last.start
284 if self.rover >= self.size:
285 self.rover = 0
286 return released
287
288 def compact(self):
289 moved = []
290 address = 0
291 packed = self.used_blocks()
292 for block in packed:
293 if block.start != address:
294 moved.append((block, block.payload()))
295 block.start = address
296 address += block.size
297 if address < self.size:
298 packed.append(self.register(Block(address, self.size - address)))
299 self.blocks = packed
300 self.rover = 0
301 return moved, address

304def run_experiment(mode, size, seed, operations, max_request):
305 heap = Heap(mode, size)
306 state = seed
307 live = []
308 used = 0
309 peak = 0
310 successes = 0
311 failures = 0
312 for _ in range(operations):
313 state = next_random(state)
314 allocating = not live or state % 3 != 0
315 state = next_random(state)
316 if allocating:
317 block = heap.allocate(heap.required_length(1 + state % max_request), DEFAULT_ALIGNMENT)
318 if block is None:
319 failures += 1
320 else:
321 successes += 1
322 live.append(block)
323 used += block.size
324 else:
325 block = live.pop(state % len(live))
326 used -= block.size
327 heap.release(block)
328 peak = max(peak, used)
329 return (
330 f"BENCH {mode} ok={successes} fail={failures} peak={peak} used={used} "
331 f"frag={heap.fragmentation()} blocks={len(heap.blocks)}"
332 )

368 def reset(self, heap):
369 self.heap = heap
370 self.allocations = {}
371 self.freed = set()
372 self.roots = set()
373 self.references = {}
374 self.histories = {}
375 self.counters = dict.fromkeys(COUNTERS, 0)
376 self.peak = 0
377
378 def emit(self, *parts):
379 self.output.append(" ".join(str(part) for part in parts))

381 def record(self, name, line):
382 self.emit(line)
383 self.histories.setdefault(name, []).append(line)

385 def out_of_memory(self, name):
386 self.counters["failures"] += 1
387 return CommandError(f"OUT OF MEMORY {name}")

389 def require_live(self, name):
390 if name in self.allocations:
391 return self.allocations[name]
392 if name in self.freed:
393 raise CommandError(f"USE AFTER FREE {name}")
394 raise CommandError(f"UNKNOWN ID {name}")

396 def require_range(self, name, offset, length):
397 block = self.require_live(name)
398 if offset + length > block.requested:
399 raise CommandError(f"OUT OF BOUNDS {name}")
400 return block

402 def require_free_list(self):
403 if self.heap.is_buddy():
404 raise CommandError("UNSUPPORTED IN BUDDY MODE")

406 def reachable(self):
407 found = set(self.roots)
408 pending = list(self.roots)
409 while pending:
410 name = pending.pop()
411 for target in self.references.get(name, set()):
412 if target not in found:
413 found.add(target)
414 pending.append(target)
415 return found

417 def create_heap(self, arguments):
418 size, mode = int(arguments[0]), arguments[1]
419 if size % UNIT != 0 or not MIN_HEAP <= size <= MAX_HEAP:
420 raise CommandError("BAD HEAP SIZE")
421 if mode == "BUDDY" and not is_power_of_two(size):
422 raise CommandError("BAD HEAP SIZE")
423 self.reset(Heap(mode, size))
424 self.emit(f"HEAP size={size} mode={mode}")
425
426 def allocate_id(self, word, name, request, alignment, fill):
427 if request == 0:
428 raise CommandError("BAD SIZE")
429 if name in self.allocations:
430 raise CommandError(f"DUPLICATE ID {name}")
431 block = self.heap.allocate(self.heap.required_length(request), alignment)
432 if block is None:
433 raise self.out_of_memory(name)
434 block.owner = name
435 block.requested = request
436 block.contents = [fill] * request
437 self.allocations[name] = block
438 self.freed.discard(name)
439 self.counters["allocs"] += 1
440 self.record(name, f"{word} {name} addr={block.payload()} block={block.size}")
441
442 def malloc(self, arguments):
443 name, request = arguments
444 self.allocate_id("MALLOC", name, int(request), DEFAULT_ALIGNMENT, UNINITIALIZED)
445
446 def calloc(self, arguments):
447 name, count, size = arguments
448 self.allocate_id("CALLOC", name, int(count) * int(size), DEFAULT_ALIGNMENT, 0)
449
450 def memalign(self, arguments):
451 name, alignment, request = arguments[0], int(arguments[1]), int(arguments[2])
452 self.require_free_list()
453 if not is_power_of_two(alignment) or not DEFAULT_ALIGNMENT <= alignment <= MAX_ALIGNMENT:
454 raise CommandError("BAD ALIGNMENT")
455 self.allocate_id("MEMALIGN", name, request, alignment, UNINITIALIZED)
456
457 def release_id(self, name):
458 block = self.allocations.pop(name)
459 size = block.size
460 self.freed.add(name)
461 self.roots.discard(name)
462 self.references.pop(name, None)
463 for targets in self.references.values():
464 targets.discard(name)
465 result, merges = self.heap.release(block)
466 return f"block={size} merges={merges} free={result.area()}"
467
468 def free_id(self, name):
469 if name in self.freed:
470 raise CommandError(f"DOUBLE FREE {name}")
471 if name not in self.allocations:
472 raise CommandError(f"UNKNOWN ID {name}")
473 details = self.release_id(name)
474 self.counters["frees"] += 1
475 self.record(name, f"FREE {name} {details}")
476
477 def free(self, arguments):
478 self.free_id(arguments[0])
479
480 def realloc(self, arguments):
481 name, request = arguments[0], int(arguments[1])
482 if request == 0:
483 self.free_id(name)
484 return
485 if name in self.freed:
486 raise CommandError(f"USE AFTER FREE {name}")
487 if name not in self.allocations:
488 self.allocate_id("MALLOC", name, request, DEFAULT_ALIGNMENT, UNINITIALIZED)
489 return
490 block = self.allocations[name]
491 length = self.heap.required_length(request)
492 contents = block.contents[:request]
493 contents += [UNINITIALIZED] * (request - len(contents))
494 origin = block.payload()
495 kind = self.heap.resize(block, length)
496 if kind is None:
497 moved = self.heap.allocate(length, DEFAULT_ALIGNMENT)
498 if moved is None:
499 raise self.out_of_memory(name)
500 moved.owner = name
501 self.allocations[name] = moved
502 self.heap.release(block)
503 self.counters["moves"] += 1
504 block = moved
505 kind = f"moved from={origin}"
506 block.requested = request
507 block.contents = contents
508 self.counters["reallocs"] += 1
509 self.record(name, f"REALLOC {name} addr={block.payload()} block={block.size} {kind}")
510
511 def fill(self, arguments):
512 name = arguments[0]
513 offset, length, value = (int(text) for text in arguments[1:])
514 block = self.require_range(name, offset, length)
515 block.contents[offset:offset + length] = [value] * length
516 self.emit(f"FILL {name} {offset}+{length} = {value}")
517
518 def copy(self, arguments):
519 target_name, source_name = arguments[0], arguments[2]
520 target_offset, source_offset, length = int(arguments[1]), int(arguments[3]), int(arguments[4])
521 target = self.require_range(target_name, target_offset, length)
522 source = self.require_range(source_name, source_offset, length)
523 values = source.contents[source_offset:source_offset + length]
524 target.contents[target_offset:target_offset + length] = values
525 self.emit(f"COPY {target_name} {target_offset}+{length} from {source_name} {source_offset}")
526
527 def read(self, arguments):
528 name = arguments[0]
529 offset, length = int(arguments[1]), int(arguments[2])
530 block = self.require_range(name, offset, length)
531 values = " ".join(byte_text(value) for value in block.contents[offset:offset + length])
532 self.emit(f"READ {name} {offset}+{length} = {values}")

564 def collect(self, arguments):
565 reachable = self.reachable()
566 garbage = [block for name, block in self.allocations.items() if name not in reachable]
567 garbage.sort(key=lambda block: block.start)
568 self.emit(f"GC collected={len(garbage)} bytes={sum(block.size for block in garbage)}")
569 for block in garbage:
570 name = block.owner
571 self.record(name, f"COLLECT {name} {self.release_id(name)}")
572 self.counters["collected"] += len(garbage)

574 def sbrk(self, arguments):
575 amount = int(arguments[0])
576 heap = self.heap
577 if heap.is_buddy():
578 valid = amount == heap.size and 2 * heap.size <= MAX_HEAP
579 else:
580 valid = amount % UNIT == 0 and amount >= MIN_BLOCK and heap.size + amount <= MAX_HEAP
581 if not valid:
582 raise CommandError("BAD SBRK SIZE")
583 result = heap.extend(amount)
584 self.emit(f"SBRK heap={heap.size} free={result.area()}")

587 released = self.heap.trim()

590 def compact(self, arguments):
591 self.require_free_list()
592 heap = self.heap
593 moved, used = heap.compact()
594 self.counters["moves"] += len(moved)
595 free = f"{used}+{heap.size - used}" if used < heap.size else "none"
596 self.emit(f"COMPACT moved={len(moved)} free={free}")
597 for block, origin in moved:
598 self.record(block.owner, f"MOVE {block.owner} {origin} -> {block.payload()}")

600 def who(self, arguments):
601 address = int(arguments[0])
602 if address >= self.heap.size:
603 self.emit("WHO", address, "outside")
604 return
605 block = self.heap.blocks[self.heap.locate(address)]
606 if not block.used:
607 self.emit("WHO", address, "free", block.area())
608 elif address < block.payload():
609 self.emit("WHO", address, "header", block.owner)
610 else:
611 offset = address - block.payload()
612 region = "data" if offset < block.requested else "slack"
613 self.emit("WHO", address, region, f"{block.owner}+{offset}")

615 def dump(self, arguments):
616 heap = self.heap
617 header = f"DUMP mode={heap.mode} heap={heap.size} blocks={len(heap.blocks)}"
618 if heap.mode == "NEXT":
619 header += f" rover={heap.rover}"
620 self.emit(header)
621 for block in heap.blocks:
622 if block.used:
623 self.emit(block.start, block.size, "USED", block.owner, f"req={block.requested}")
624 else:
625 self.emit(block.start, block.size, "FREE")

633 def freelist(self, arguments):
634 blocks = self.heap.free_blocks()
635 if self.heap.mode == "SEGFIT":
636 blocks.sort(key=lambda block: -block.stamp)
637 self.emit(f"FREELIST count={len(blocks)}")
638 for power in sorted({size_class(block.size) for block in blocks}):
639 members = [block.area() for block in blocks if size_class(block.size) == power]
640 self.emit(f"{2 ** power}-{2 ** (power + 1) - 1}:", *members)

642 def tree_node(self, leaves, start, size, depth):
643 label = " " * (2 * depth) + f"{start}+{size}"
644 block = leaves.get(start)
645 if block is not None and block.size == size:
646 if block.used:
647 self.emit(label, "USED", block.owner)
648 else:
649 self.emit(label, "FREE")
650 return
651 self.emit(label, "split")
652 half = size // 2
653 self.tree_node(leaves, start, half, depth + 1)
654 self.tree_node(leaves, start + half, half, depth + 1)

656 def tree(self, arguments):
657 if not self.heap.is_buddy():
658 raise CommandError("REQUIRES BUDDY MODE")
659 self.emit(f"TREE heap={self.heap.size}")
660 leaves = {block.start: block for block in self.heap.blocks}
661 self.tree_node(leaves, 0, self.heap.size, 0)

663 def stats(self, arguments):
664 used = self.heap.used_blocks()
665 free = self.heap.free_blocks()
666 free_total = sum(block.size for block in free)
667 largest = max((block.size for block in free), default=0)
668 self.emit(
669 f"STATS used={sum(block.size for block in used)} used_blocks={len(used)}",
670 f"free={free_total} free_blocks={len(free)} largest={largest}",
671 f"frag={fragmentation_percent(free_total, largest)}",
672 f"requested={sum(block.requested for block in used)} peak={self.peak}",
673 )
674 self.emit("OPS", *(f"{name}={self.counters[name]}" for name in COUNTERS))

676 def leaks(self, arguments):
677 reachable = self.reachable()
678 names = sorted(self.allocations)
679 requested = sum(self.allocations[name].requested for name in names)
680 lost = sum(name not in reachable for name in names)
681 self.emit(f"LEAKS count={len(names)} requested={requested} lost={lost}")
682 for name in names:
683 block = self.allocations[name]
684 status = "reachable" if name in reachable else "lost"
685 self.emit(
686 f"LEAK {name} addr={block.payload()} block={block.size}",
687 f"req={block.requested} {status}",
688 )

696 def bench(self, arguments):
697 size, seed, operations, max_request = (int(text) for text in arguments)
698 if not is_power_of_two(size) or not MIN_HEAP <= size <= MAX_HEAP:
699 raise CommandError("BAD HEAP SIZE")
700 self.emit(f"BENCH heap={size} seed={seed} ops={operations} maxreq={max_request}")
701 for mode in MODES:
702 self.emit(run_experiment(mode, size, seed, operations, max_request))

704 def execute(self, line):
705 command, *arguments = line.split()
706 try:
707 if self.heap is None and command not in HEAPLESS_COMMANDS:
708 raise CommandError("NO HEAP")
709 self.handlers[command](arguments)
710 except CommandError as error:
711 self.emit("ERROR", error)
712 if self.heap is not None:
713 self.peak = max(self.peak, self.heap.used_total())
