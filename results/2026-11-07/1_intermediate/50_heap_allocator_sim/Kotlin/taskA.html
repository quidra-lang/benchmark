4val modes = listOf("FIRST", "NEXT", "BEST", "WORST", "SEGFIT", "BUDDY")
5val counterNames = listOf("allocs", "frees", "reallocs", "moves", "failures", "collected")

7fun power(n: Int) = n > 0 && (n and (n - 1)) == 0
8
9fun sizeClass(n: Int) = 31 - Integer.numberOfLeadingZeros(n)
10
11fun random(x: Long) = x * 48271L % 2147483647L

13class CommandError(message: String) : RuntimeException(message)

15class Block(var start: Int, var size: Int) {
16 var owner = ""
17 var requested = 0
18 var stamp = 0
19 var used = false
20 var contents = IntArray(0)
21 val end
22 get() = start + size
23
24 val payload
25 get() = start + 8
26
27 val area
28 get() = "$start+$size"
29}

31class Heap(val mode: String, var size: Int) {
32 var rover = 0
33 var age = 0
34 var blocks = mutableListOf(register(Block(0, size)))
35 val buddy
36 get() = mode == "BUDDY"
37
38 fun register(b: Block): Block {
39 b.stamp = ++age
40 return b
41 }
42
43 fun required(n: Int): Int {
44 val s = (n + 7) / 8 * 8 + 8
45 if (!buddy) return s
46 var p = 32
47 while (p < s) p *= 2
48 return p
49 }
50
51 fun used() = blocks.filter { it.used }
52
53 fun free() = blocks.filter { !it.used }
54
55 fun total() = used().sumOf { it.size }
56
57 fun frag(): Int {
58 val f = free().sumOf { it.size }
59 val l = free().maxOfOrNull { it.size } ?: 0
60 return if (f == 0) 0 else (f - l) * 100 / f
61 }
62
63 fun locate(p: Int) = blocks.indexOfFirst { p < it.end }
64
65 fun allocate(s: Int, al: Int): Block? {
66 val ri = locate(rover)
67 var chosen = -1
68 var placement = 0
69 var bestA = Int.MAX_VALUE
70 var bestB = Int.MAX_VALUE
71 for ((i, b) in blocks.withIndex()) {
72 if (b.used) continue
73 var p = (b.start + 8 + al - 1) / al * al - 8
74 if (p > b.start && p - b.start < 16) p += al
75 if (p + s > b.end) continue
76 val a =
77 when (mode) {
78 "NEXT" -> (i - ri + blocks.size) % blocks.size
79 "BEST",
80 "BUDDY" -> b.size
81 "WORST" -> -b.size
82 "SEGFIT" -> sizeClass(b.size)
83 else -> b.start
84 }
85 val c =
86 when (mode) {
87 "BEST",
88 "BUDDY",
89 "WORST" -> b.start
90 "SEGFIT" -> -b.stamp
91 else -> 0
92 }
93 if (a < bestA || a == bestA && c < bestB) {
94 chosen = i
95 placement = p
96 bestA = a
97 bestB = c
98 }
99 }
100 if (chosen < 0) return null
101 val b: Block
102 if (buddy) {
103 b = blocks[chosen]
104 halve(chosen, s)
105 } else {
106 val old = blocks.removeAt(chosen)
107 val parts = mutableListOf<Block>()
108 if (placement > old.start) parts.add(register(Block(old.start, placement - old.start)))
109 b = Block(placement, old.end - placement)
110 parts.add(b)
111 if (b.size - s >= 16) {
112 parts.add(register(Block(placement + s, b.size - s)))
113 b.size = s
114 }
115 blocks.addAll(chosen, parts)
116 }
117 b.used = true
118 if (mode == "NEXT") rover = if (b.end == size) 0 else b.end
119 return b
120 }
121
122 fun halve(i: Int, s: Int) {
123 val b = blocks[i]
124 while (b.size > s) {
125 b.size /= 2
126 blocks.add(i + 1, register(Block(b.end, b.size)))
127 }
128 }
129
130 fun merge(i: Int) {
131 blocks[i].size += blocks.removeAt(i + 1).size
132 }
133
134 fun coalesce(index: Int): Pair<Block, Int> {
135 var i = index
136 var merges = 0
137 if (buddy) {
138 while (blocks[i].size < size) {
139 val b = blocks[i]
140 val j = if (b.start % (2 * b.size) == 0) i + 1 else i - 1
141 if (j !in blocks.indices || blocks[j].used || blocks[j].size != b.size) break
142 i = min(i, j)
143 merge(i)
144 merges++
145 }
146 } else {
147 if (i + 1 < blocks.size && !blocks[i + 1].used) {
148 merge(i)
149 merges++
150 }
151 if (i > 0 && !blocks[i - 1].used) {
152 i--
153 merge(i)
154 merges++
155 }
156 }
157 return register(blocks[i]) to merges
158 }
159
160 fun release(b: Block): Pair<Block, Int> {
161 b.used = false
162 b.owner = ""
163 b.requested = 0
164 b.contents = IntArray(0)
165 return coalesce(blocks.indexOf(b))
166 }
167
168 fun resize(b: Block, s: Int): String? {
169 val i = blocks.indexOf(b)
170 if (s <= b.size) {
171 if (buddy) {
172 if (s == b.size) return "kept"
173 halve(i, s)
174 } else {
175 if (b.size - s < 16) return "kept"
176 blocks.add(i + 1, Block(b.start + s, b.size - s))
177 b.size = s
178 coalesce(i + 1)
179 }
180 return "shrunk"
181 }
182 if (buddy) {
183 var t = b.size
184 var j = i
185 while (t < s) {
186 j++
187 if (
188 b.start % (2 * t) != 0 ||
189 j >= blocks.size ||
190 blocks[j].used ||
191 blocks[j].size != t
192 )
193 return null
194 t *= 2
195 }
196 blocks.subList(i + 1, j + 1).clear()
197 b.size = s
198 } else {
199 if (i + 1 >= blocks.size || blocks[i + 1].used || b.size + blocks[i + 1].size < s)
200 return null
201 val total = b.size + blocks.removeAt(i + 1).size
202 b.size = if (total - s >= 16) s else total
203 if (total - s >= 16) blocks.add(i + 1, register(Block(b.end, total - s)))
204 }
205 return "grown"
206 }
207
208 fun trim(): Int {
209 val old = size
210 if (buddy) {
211 while (!blocks.last().used && blocks.last().size * 2 == size) {
212 blocks.removeAt(blocks.lastIndex)
213 size /= 2
214 }
215 } else if (blocks.size > 1 && !blocks.last().used)
216 size = blocks.removeAt(blocks.lastIndex).start
217 if (rover >= size) rover = 0
218 return old - size
219 }
220}

222fun experiment(mode: String, s: Int, seed: Long, ops: Int, maxreq: Int): String {
223 val h = Heap(mode, s)
224 val live = mutableListOf<Block>()
225 var x = seed
226 var ok = 0
227 var fail = 0
228 var peak = 0
229 repeat(ops) {
230 x = random(x)
231 val alloc = live.isEmpty() || x % 3 != 0L
232 x = random(x)
233 if (alloc) {
234 val b = h.allocate(h.required(1 + (x % maxreq).toInt()), 8)
235 if (b == null) fail++
236 else {
237 ok++
238 live.add(b)
239 }
240 } else h.release(live.removeAt((x % live.size).toInt()))
241 peak = max(peak, h.total())
242 }
243 return "BENCH $mode ok=$ok fail=$fail peak=$peak used=${h.total()} frag=${h.frag()} blocks=${h.blocks.size}"
244}

246class Allocator {
247 var heap: Heap? = null
248 val live = mutableMapOf<String, Block>()
249 val freed = mutableSetOf<String>()
250 val roots = mutableSetOf<String>()
251 val links = mutableMapOf<String, MutableSet<String>>()
252 val history = mutableMapOf<String, MutableList<String>>()
253 var counters = IntArray(6)
254 var peak = 0
255 val out = mutableListOf<String>()
256
257 fun fail(s: String): Nothing = throw CommandError(s)
258
259 fun record(id: String, s: String) {
260 out.add(s)
261 history.getOrPut(id) { mutableListOf() }.add(s)
262 }
263
264 fun block(id: String): Block =
265 live[id] ?: fail(if (id in freed) "USE AFTER FREE $id" else "UNKNOWN ID $id")
266
267 fun range(id: String, off: Int, len: Int): Block {
268 val b = block(id)
269 if (off + len > b.requested) fail("OUT OF BOUNDS $id")
270 return b
271 }
272
273 fun oom(id: String): Nothing {
274 counters[4]++
275 fail("OUT OF MEMORY $id")
276 }
277
278 fun allocate(kind: String, id: String, n: Int, al: Int = 8, fill: Int = -1) {
279 if (n == 0) fail("BAD SIZE")
280 if (id in live) fail("DUPLICATE ID $id")
281 val h = heap!!
282 val b = h.allocate(h.required(n), al) ?: oom(id)
283 b.owner = id
284 b.requested = n
285 b.contents = IntArray(n) { fill }
286 live[id] = b
287 freed.remove(id)
288 counters[0]++
289 record(id, "$kind $id addr=${b.payload} block=${b.size}")
290 }
291
292 fun release(id: String): String {
293 val b = live.remove(id)!!
294 val size = b.size
295 freed.add(id)
296 roots.remove(id)
297 links.remove(id)
298 links.values.forEach { it.remove(id) }
299 val (f, merges) = heap!!.release(b)
300 return "block=$size merges=$merges free=${f.area}"
301 }
302
303 fun free(id: String) {
304 if (id in freed) fail("DOUBLE FREE $id")
305 if (id !in live) fail("UNKNOWN ID $id")
306 val detail = release(id)
307 counters[1]++
308 record(id, "FREE $id $detail")
309 }
310
311 fun realloc(id: String, n: Int) {
312 if (n == 0) {
313 free(id)
314 return
315 }
316 if (id in freed) fail("USE AFTER FREE $id")
317 var b = live[id]
318 if (b == null) {
319 allocate("MALLOC", id, n)
320 return
321 }
322 val original = b
323 val contents = IntArray(n) { if (it < original.requested) original.contents[it] else -1 }
324 val old = b.payload
325 val h = heap!!
326 var kind = h.resize(b, h.required(n))
327 if (kind == null) {
328 val moved = h.allocate(h.required(n), 8) ?: oom(id)
329 moved.owner = id
330 live[id] = moved
331 h.release(b)
332 b = moved
333 counters[3]++
334 kind = "moved from=$old"
335 }
336 b.requested = n
337 b.contents = contents
338 counters[2]++
339 record(id, "REALLOC $id addr=${b.payload} block=${b.size} $kind")
340 }
341
342 fun reachable(): Set<String> {
343 val seen = roots.toMutableSet()
344 val stack = roots.toMutableList()
345 while (stack.isNotEmpty()) {
346 val id = stack.removeAt(stack.lastIndex)
347 for (dst in links[id].orEmpty()) if (seen.add(dst)) stack.add(dst)
348 }
349 return seen
350 }
351
352 fun tree(a: Int, s: Int, depth: Int) {
353 val b = heap!!.blocks.find { it.start == a && it.size == s }
354 val prefix = " ".repeat(depth * 2) + "$a+$s"
355 if (b != null) out.add(prefix + if (b.used) " USED ${b.owner}" else " FREE")
356 else {
357 out.add("$prefix split")
358 tree(a, s / 2, depth + 1)
359 tree(a + s / 2, s / 2, depth + 1)
360 }
361 }
362
363 fun execute(line: String) {
364 val t = line.split(' ')
365 val cmd = t[0]
366 val a = t.drop(1)
367 fun n(i: Int) = a[i].toInt()
368 try {
369 if (heap == null && cmd != "HEAP" && cmd != "BENCH") fail("NO HEAP")
370 val h = heap
371 when (cmd) {
372 "HEAP" -> {
373 val s = n(0)
374 if (s !in 32..65536 || s % 8 != 0 || a[1] == "BUDDY" && !power(s))
375 fail("BAD HEAP SIZE")
376 heap = Heap(a[1], s)
377 live.clear()
378 freed.clear()
379 roots.clear()
380 links.clear()
381 history.clear()
382 counters = IntArray(6)
383 peak = 0
384 out.add("HEAP size=$s mode=${a[1]}")
385 }
386 "BENCH" -> {
387 val s = n(0)
388 if (s !in 32..65536 || !power(s)) fail("BAD HEAP SIZE")
389 out.add("BENCH heap=$s seed=${a[1]} ops=${a[2]} maxreq=${a[3]}")
390 for (m in modes) out.add(experiment(m, s, a[1].toLong(), n(2), n(3)))
391 }
392 "MALLOC" -> allocate(cmd, a[0], n(1))
393 "CALLOC" -> allocate(cmd, a[0], n(1) * n(2), 8, 0)
394 "MEMALIGN" -> {
395 if (h!!.buddy) fail("UNSUPPORTED IN BUDDY MODE")
396 if (n(1) !in 8..4096 || !power(n(1))) fail("BAD ALIGNMENT")
397 allocate(cmd, a[0], n(2), n(1))
398 }
399 "REALLOC" -> realloc(a[0], n(1))
400 "FREE" -> free(a[0])
401 "FILL" -> {
402 val b = range(a[0], n(1), n(2))
403 b.contents.fill(n(3), n(1), n(1) + n(2))
404 out.add("FILL ${a[0]} ${a[1]}+${a[2]} = ${a[3]}")
405 }
406 "COPY" -> {
407 val dst = range(a[0], n(1), n(4))
408 val src = range(a[2], n(3), n(4))
409 src.contents.copyOfRange(n(3), n(3) + n(4)).copyInto(dst.contents, n(1))
410 out.add("COPY ${a[0]} ${a[1]}+${a[4]} from ${a[2]} ${a[3]}")
411 }
412 "READ" -> {
413 val b = range(a[0], n(1), n(2))
414 val values =
415 b.contents.slice(n(1) until n(1) + n(2)).joinToString(" ") {
416 if (it < 0) "?" else "$it"
417 }
418 out.add("READ ${a[0]} ${a[1]}+${a[2]} = $values")
419 }
420 "ROOT",
421 "UNROOT" -> {
422 block(a[0])
423 if (cmd == "ROOT") roots.add(a[0])
424 else if (!roots.remove(a[0])) fail("NOT ROOT ${a[0]}")
425 out.add("$cmd ${a[0]}")
426 }
427 "LINK",
428 "UNLINK" -> {
429 block(a[0])
430 block(a[1])
431 if (cmd == "LINK") links.getOrPut(a[0]) { mutableSetOf() }.add(a[1])
432 else if (links[a[0]]?.remove(a[1]) != true) fail("NO LINK ${a[0]} ${a[1]}")
433 out.add("$cmd ${a[0]} ${a[1]}")
434 }
435 "GC" -> {
436 val seen = reachable()
437 val garbage = h!!.used().filter { it.owner !in seen }
438 out.add("GC collected=${garbage.size} bytes=${garbage.sumOf { it.size }}")
439 for (b in garbage) {
440 val id = b.owner
441 record(id, "COLLECT $id ${release(id)}")
442 }
443 counters[5] += garbage.size
444 }
445 "SBRK" -> {
446 val k = n(0)
447 if (h!!.buddy) {
448 if (k != h.size || h.size * 2 > 65536) fail("BAD SBRK SIZE")
449 } else if (k < 16 || k % 8 != 0 || h.size + k > 65536) fail("BAD SBRK SIZE")
450 h.blocks.add(Block(h.size, k))
451 h.size += k
452 val f = h.coalesce(h.blocks.lastIndex).first
453 out.add("SBRK heap=${h.size} free=${f.area}")
454 }
455 "TRIM" -> {
456 val released = h!!.trim()
457 out.add("TRIM heap=${h.size} released=$released")
458 }
459 "COMPACT" -> {
460 if (h!!.buddy) fail("UNSUPPORTED IN BUDDY MODE")
461 val used = h.used().toMutableList()
462 val moved = mutableListOf<Pair<Block, Int>>()
463 var pos = 0
464 for (b in used) {
465 if (b.start != pos) {
466 moved.add(b to b.payload)
467 b.start = pos
468 }
469 pos += b.size
470 }
471 if (pos < h.size) used.add(h.register(Block(pos, h.size - pos)))
472 h.blocks = used
473 h.rover = 0
474 counters[3] += moved.size
475 out.add(
476 "COMPACT moved=${moved.size} free=" +
477 if (pos < h.size) "$pos+${h.size - pos}" else "none",
478 )
479 for ((b, old) in moved) record(b.owner, "MOVE ${b.owner} $old -> ${b.payload}")
480 }
481 "WHO" -> {
482 val p = n(0)
483 val detail =
484 if (p >= h!!.size) "outside"
485 else {
486 val b = h.blocks[h.locate(p)]
487 if (!b.used) "free ${b.area}"
488 else if (p < b.payload) "header ${b.owner}"
489 else
490 (if (p - b.payload < b.requested) "data" else "slack") +
491 " ${b.owner}+${p - b.payload}"
492 }
493 out.add("WHO $p $detail")
494 }
495 "DUMP" -> {
496 out.add(
497 "DUMP mode=${h!!.mode} heap=${h.size} blocks=${h.blocks.size}" +
498 if (h.mode == "NEXT") " rover=${h.rover}" else "",
499 )
500 for (b in h.blocks) out.add(
501 "${b.start} ${b.size} " +
502 if (b.used) "USED ${b.owner} req=${b.requested}" else "FREE",
503 )
504 }
505 "MAP" -> {
506 out.add("MAP heap=${h!!.size}")
507 val units = buildString {
508 for (b in h.blocks) {
509 append(if (b.used) 'H' else 'F')
510 repeat(b.size / 8 - 1) {
511 append(if (!b.used) '.' else if (it * 8 < b.requested) '#' else '-')
512 }
513 }
514 }
515 for (i in units.indices step 64) out.add(
516 (i * 8).toString().padStart(5, '0') +
517 " " +
518 units.substring(i, min(i + 64, units.length)),
519 )
520 }
521 "FREELIST" -> {
522 val free =
523 h!!.free().let {
524 if (h.mode == "SEGFIT") it.sortedByDescending { b -> b.stamp } else it
525 }
526 out.add("FREELIST count=${free.size}")
527 for ((k, group) in free.groupBy { sizeClass(it.size) }.toSortedMap()) out.add(
528 "${1 shl k}-${(1 shl (k + 1)) - 1}: " + group.joinToString(" ") { it.area },
529 )
530 }
531 "TREE" -> {
532 if (!h!!.buddy) fail("REQUIRES BUDDY MODE")
533 out.add("TREE heap=${h.size}")
534 tree(0, h.size, 0)
535 }
536 "STATS" -> {
537 val used = h!!.used()
538 val free = h.free()
539 out.add(
540 "STATS used=${h.total()} used_blocks=${used.size} free=${free.sumOf { it.size }} free_blocks=${free.size} largest=${free.maxOfOrNull { it.size } ?: 0} frag=${h.frag()} requested=${used.sumOf { it.requested }} peak=$peak",
541 )
542 out.add(
543 "OPS " +
544 counterNames.indices.joinToString(" ") {
545 "${counterNames[it]}=${counters[it]}"
546 },
547 )
548 }
549 "LEAKS" -> {
550 val seen = reachable()
551 out.add(
552 "LEAKS count=${live.size} requested=${live.values.sumOf { it.requested }} lost=${live.keys.count { it !in seen }}",
553 )
554 for ((id, b) in live.toSortedMap()) out.add(
555 "LEAK $id addr=${b.payload} block=${b.size} req=${b.requested} " +
556 if (id in seen) "reachable" else "lost",
557 )
558 }
559 "HISTORY" -> {
560 val events = history[a[0]].orEmpty()
561 out.add("HISTORY ${a[0]} events=${events.size}")
562 out.addAll(events)
563 }
564 }
565 } catch (e: CommandError) {
566 out.add("ERROR ${e.message}")
567 }
568 heap?.let { peak = max(peak, it.total()) }
569 }
570}

572fun main() {
573 val input = System.`in`.bufferedReader()
574 val allocator = Allocator()
575 repeat(input.readLine().toInt()) { allocator.execute(input.readLine()) }
576 print(allocator.out.joinToString("\n", postfix = "\n"))
577}
