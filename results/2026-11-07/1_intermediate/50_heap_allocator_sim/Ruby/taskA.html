4COUNTERS = %w[allocs frees reallocs moves failures collected].freeze

10UNINITIALIZED = -1
11RANDOM_MULTIPLIER = 48_271

16def power_of_two?(value)
17 value.positive? && (value & (value - 1)).zero?
18end
19
20def size_class(size)
21 size.bit_length - 1
22end
23
24def fragmentation_percent(free, largest)
25 free.positive? ? (free - largest) * 100 / free : 0
26end

28class Block
29 attr_accessor :start, :size, :used, :owner, :requested, :contents, :stamp
30
31 def initialize(start, size)
32 @start, @size = start, size
33 @used, @owner, @requested, @contents, @stamp = false, nil, 0, [], 0
34 end
35
36 def finish = @start + @size
37 def payload = @start + HEADER
38 def area = "#{@start}+#{@size}"
39
40 def map_units
41 rest = @size / UNIT - 1
42 return 'F' + '.' * rest unless @used
43 'H' + rest.times.map { |i| UNIT * i < @requested ? '#' : '-' }.join
44 end
45end

47class Heap
48 attr_accessor :mode, :size, :rover, :blocks
49
50 def initialize(mode, size)
51 @mode, @size, @rover, @registrations = mode, size, 0, 0
52 @blocks = [register(Block.new(0, size))]
53 end
54
55 def register(block)
56 @registrations += 1
57 block.stamp = @registrations
58 block
59 end
60
61 def buddy? = @mode == 'BUDDY'
62
63 def required_length(request)
64 length = ((request + UNIT - 1) / UNIT) * UNIT + HEADER
65 return length unless buddy?
66 power = MIN_BUDDY_BLOCK
67 power *= 2 while power < length
68 power
69 end
70
71 def locate(address)
72 @blocks.index { |block| block.finish > address }
73 end
74
75 def used_blocks = @blocks.select(&:used)
76 def free_blocks = @blocks.reject(&:used)
77 def used_total = used_blocks.sum(&:size)
78
79 def fragmentation
80 sizes = free_blocks.map(&:size)
81 fragmentation_percent(sizes.sum, sizes.max || 0)
82 end
83
84 def placement(block, alignment)
85 place = ((block.start + HEADER + alignment - 1) / alignment) * alignment - HEADER
86 place += alignment if place > block.start && place - block.start < MIN_BLOCK
87 place
88 end
89
90 def rank(index, rover_index)
91 block = @blocks[index]
92 case @mode
93 when 'NEXT' then [(index - rover_index) % @blocks.size, 0]
94 when 'BEST', 'BUDDY' then [block.size, block.start]
95 when 'WORST' then [-block.size, block.start]
96 when 'SEGFIT' then [size_class(block.size), -block.stamp]
97 else [block.start, 0]
98 end
99 end
100
101 def choose(length, alignment)
102 rover_index = locate(@rover)
103 options = @blocks.each_with_index.filter_map do |block, index|
104 next if block.used
105 place = placement(block, alignment)
106 [rank(index, rover_index), index, place] if place + length <= block.finish
107 end
108 options.min
109 end
110
111 def allocate(length, alignment)
112 choice = choose(length, alignment)
113 return nil unless choice
114 _, index, place = choice
115 block = buddy? ? halve(index, length) : carve(index, place, length)
116 block.used = true
117 @rover = block.finish == @size ? 0 : block.finish if @mode == 'NEXT'
118 block
119 end
120
121 def carve(index, place, length)
122 original = @blocks[index]
123 parts = []
124 parts << register(Block.new(original.start, place - original.start)) if place > original.start
125 block = Block.new(place, original.finish - place)
126 parts << block
127 if block.size - length >= MIN_BLOCK
128 parts << register(Block.new(place + length, block.size - length))
129 block.size = length
130 end
131 @blocks[index, 1] = parts
132 block
133 end
134
135 def halve(index, length)
136 block = @blocks[index]
137 while block.size > length
138 block.size /= 2
139 @blocks.insert(index + 1, register(Block.new(block.finish, block.size)))
140 end
141 block
142 end
143
144 def merge_next(index)
145 @blocks[index].size += @blocks.delete_at(index + 1).size
146 end
147
148 def buddy_index(index)
149 block = @blocks[index]
150 (block.start % (2 * block.size)).zero? ? index + 1 : index - 1
151 end
152
153 def coalesce(index)
154 merges = 0
155 if buddy?
156 while @blocks[index].size < @size
157 partner = buddy_index(index)
158 other = @blocks[partner]
159 break if other.used || other.size != @blocks[index].size
160 index = [index, partner].min
161 merge_next(index)
162 merges += 1
163 end
164 else
165 if index + 1 < @blocks.size && !@blocks[index + 1].used
166 merge_next(index)
167 merges += 1
168 end
169 if index.positive? && !@blocks[index - 1].used
170 index -= 1
171 merge_next(index)
172 merges += 1
173 end
174 end
175 [register(@blocks[index]), merges]
176 end
177
178 def release(block)
179 block.used, block.owner, block.requested, block.contents = false, nil, 0, []
180 coalesce(@blocks.index(block))
181 end
182
183 def shrink(index, length)
184 block = @blocks[index]
185 if buddy?
186 return 'kept' if length == block.size
187 halve(index, length)
188 return 'shrunk'
189 end
190 return 'kept' if block.size - length < MIN_BLOCK
191 @blocks.insert(index + 1, Block.new(block.start + length, block.size - length))
192 block.size = length
193 coalesce(index + 1)
194 'shrunk'
195 end
196
197 def grow_into_next(index, length)
198 block = @blocks[index]
199 return nil if index + 1 >= @blocks.size || @blocks[index + 1].used
200 total = block.size + @blocks[index + 1].size
201 return nil if total < length
202 @blocks.delete_at(index + 1)
203 if total - length >= MIN_BLOCK
204 block.size = length
205 @blocks.insert(index + 1, register(Block.new(block.finish, total - length)))
206 else
207 block.size = total
208 end
209 'grown'
210 end
211
212 def grow_buddy(index, length)
213 block = @blocks[index]
214 size, last = block.size, index
215 while size < length
216 last += 1
217 return nil if block.start % (2 * size) != 0 || last >= @blocks.size
218 partner = @blocks[last]
219 return nil if partner.used || partner.size != size
220 size *= 2
221 end
222 @blocks.slice!(index + 1, last - index)
223 block.size = length
224 'grown'
225 end
226
227 def resize(block, length)
228 index = @blocks.index(block)
229 return shrink(index, length) if length <= block.size
230 buddy? ? grow_buddy(index, length) : grow_into_next(index, length)
231 end
232
233 def extend(amount)
234 @blocks << Block.new(@size, amount)
235 @size += amount
236 coalesce(@blocks.size - 1)[0]
237 end
238
239 def trim
240 released = 0
241 if buddy?
242 while !@blocks[-1].used && 2 * @blocks[-1].size == @size
243 released += @blocks.pop.size
244 @size /= 2
245 end
246 elsif @blocks.size >= 2 && !@blocks[-1].used
247 last = @blocks.pop
248 released = last.size
249 @size = last.start
250 end
251 @rover = 0 if @rover >= @size
252 released
253 end
254
255 def compact
256 moved = []
257 address = 0
258 packed = used_blocks
259 packed.each do |block|
260 if block.start != address
261 moved << [block, block.payload]
262 block.start = address
263 end
264 address += block.size
265 end
266 packed << register(Block.new(address, @size - address)) if address < @size
267 @blocks = packed
268 @rover = 0
269 [moved, address]
270 end
271end

273def experiment(mode, size, seed, operations, maximum_request)
274 heap = Heap.new(mode, size)
275 state, live, used, peak, successes, failures = seed, [], 0, 0, 0, 0
276 operations.times do
277 state = RANDOM_MULTIPLIER * state % RANDOM_MODULUS
278 allocating = live.empty? || state % 3 != 0
279 state = RANDOM_MULTIPLIER * state % RANDOM_MODULUS
280 if allocating
281 block = heap.allocate(heap.required_length(1 + state % maximum_request), UNIT)
282 if block
283 successes += 1
284 live << block
285 used += block.size
286 else
287 failures += 1
288 end
289 else
290 block = live.delete_at(state % live.length)
291 used -= block.size
292 heap.release(block)
293 end
294 peak = [peak, used].max
295 end
296 "BENCH #{mode} ok=#{successes} fail=#{failures} peak=#{peak} used=#{used} " \
297 "frag=#{heap.fragmentation} blocks=#{heap.blocks.length}"
298end

300class Allocator
301 METHODS = {
302 'HEAP' => :create_heap, 'MALLOC' => :malloc, 'CALLOC' => :calloc,
303 'MEMALIGN' => :memalign, 'REALLOC' => :realloc, 'FREE' => :free,
304 'FILL' => :fill, 'COPY' => :copy, 'READ' => :read,
305 'ROOT' => :root, 'UNROOT' => :unroot, 'LINK' => :link, 'UNLINK' => :unlink,
306 'GC' => :collect, 'SBRK' => :sbrk, 'TRIM' => :trim, 'COMPACT' => :compact,
307 'WHO' => :who, 'DUMP' => :dump, 'MAP' => :show_map, 'FREELIST' => :freelist,
308 'TREE' => :tree, 'STATS' => :stats, 'LEAKS' => :leaks,
309 'HISTORY' => :history, 'BENCH' => :bench
310 }.freeze

312 attr_reader :output

314 def initialize
315 @output = []
316 reset(nil)

319 def reset(heap)
320 @heap = heap
321 @allocations, @freed, @roots, @references, @histories = {}, Set.new, Set.new, {}, {}
322 @counters = COUNTERS.to_h { |name| [name, 0] }
323 @peak = 0
324 end

327 @output << parts.join(' ')
328 end

335 def out_of_memory(name)
336 @counters['failures'] += 1
337 CommandError.new("OUT OF MEMORY #{name}")
338 end

340 def require_live(name)
341 return @allocations.fetch(name) if @allocations.key?(name)
342 raise CommandError, "USE AFTER FREE #{name}" if @freed.include?(name)
343 raise CommandError, "UNKNOWN ID #{name}"
344 end
345
346 def require_range(name, offset, length)
347 block = require_live(name)
348 raise CommandError, "OUT OF BOUNDS #{name}" if offset + length > block.requested
349 block
350 end

352 def require_free_list
353 raise CommandError, 'UNSUPPORTED IN BUDDY MODE' if @heap.buddy?
354 end

356 def reachable
357 found = @roots.dup
358 pending = @roots.to_a
359 until pending.empty?
360 name = pending.pop
361 (@references[name] || []).each do |target|
362 next if found.include?(target)
363 found.add(target)
364 pending << target
365 end
366 end
367 found
368 end

370 def create_heap(args)
371 size, mode = args[0].to_i, args[1]
372 if size % UNIT != 0 || size < 32 || size > MAX_HEAP ||
373 (mode == 'BUDDY' && !power_of_two?(size))
374 raise CommandError, 'BAD HEAP SIZE'
375 end
376 reset(Heap.new(mode, size))
377 emit("HEAP size=#{size} mode=#{mode}")
378 end
379
380 def allocate_id(word, name, request, alignment, fill)
381 raise CommandError, 'BAD SIZE' if request.zero?
382 raise CommandError, "DUPLICATE ID #{name}" if @allocations.key?(name)
383 block = @heap.allocate(@heap.required_length(request), alignment)
384 raise out_of_memory(name) unless block
385 block.owner, block.requested, block.contents = name, request, Array.new(request, fill)
386 @allocations[name] = block
387 @freed.delete(name)
388 @counters['allocs'] += 1
389 record(name, "#{word} #{name} addr=#{block.payload} block=#{block.size}")
390 end
391
392 def malloc(args)
393 allocate_id('MALLOC', args[0], args[1].to_i, UNIT, UNINITIALIZED)
394 end
395
396 def calloc(args)
397 allocate_id('CALLOC', args[0], args[1].to_i * args[2].to_i, UNIT, 0)
398 end

400 def memalign(args)
401 name, alignment, request = args[0], args[1].to_i, args[2].to_i
402 require_free_list
403 unless power_of_two?(alignment) && alignment.between?(UNIT, 4096)
404 raise CommandError, 'BAD ALIGNMENT'
405 end
406 allocate_id('MEMALIGN', name, request, alignment, UNINITIALIZED)
407 end

409 def release_id(name)
410 block = @allocations.delete(name)
411 size = block.size
412 @freed.add(name)
413 @roots.delete(name)
414 @references.delete(name)
415 @references.each_value { |targets| targets.delete(name) }
416 result, merges = @heap.release(block)
417 "block=#{size} merges=#{merges} free=#{result.area}"
418 end
419
420 def free_id(name)
421 raise CommandError, "DOUBLE FREE #{name}" if @freed.include?(name)
422 raise CommandError, "UNKNOWN ID #{name}" unless @allocations.key?(name)
423 details = release_id(name)
424 @counters['frees'] += 1
425 record(name, "FREE #{name} #{details}")
426 end
427
428 def free(args)
429 free_id(args[0])
430 end
431
432 def realloc(args)
433 name, request = args[0], args[1].to_i
434 return free_id(name) if request.zero?
435 raise CommandError, "USE AFTER FREE #{name}" if @freed.include?(name)
436 return allocate_id('MALLOC', name, request, UNIT, UNINITIALIZED) unless @allocations.key?(name)
437 block = @allocations.fetch(name)
438 length = @heap.required_length(request)
439 contents = block.contents.first(request)
440 contents += Array.new(request - contents.length, UNINITIALIZED)
441 previous = block.payload
442 kind = @heap.resize(block, length)
443 unless kind
444 moved = @heap.allocate(length, UNIT)
445 raise out_of_memory(name) unless moved
446 moved.owner = name
447 @allocations[name] = moved
448 @heap.release(block)
449 @counters['moves'] += 1
450 block = moved
451 kind = "moved from=#{previous}"
452 end
453 block.requested, block.contents = request, contents
454 @counters['reallocs'] += 1
455 record(name, "REALLOC #{name} addr=#{block.payload} block=#{block.size} #{kind}")
456 end

458 def fill(args)
459 name = args[0]
460 offset, length, value = args.drop(1).map(&:to_i)
461 block = require_range(name, offset, length)
462 block.contents[offset, length] = Array.new(length, value)
463 emit("FILL #{name} #{offset}+#{length} = #{value}")
464 end

466 def copy(args)
467 target_name, source_name = args[0], args[2]
468 target_offset, source_offset, length = args[1].to_i, args[3].to_i, args[4].to_i
469 target = require_range(target_name, target_offset, length)
470 source = require_range(source_name, source_offset, length)
471 target.contents[target_offset, length] = source.contents[source_offset, length].dup
472 emit("COPY #{target_name} #{target_offset}+#{length} from #{source_name} #{source_offset}")
473 end

475 def read(args)
476 name, offset, length = args[0], args[1].to_i, args[2].to_i
477 block = require_range(name, offset, length)
478 values = block.contents[offset, length].map { |value| value == UNINITIALIZED ? '?' : value.to_s }
479 emit("READ #{name} #{offset}+#{length} = #{values.join(' ')}")
480 end

516 def collect(_args)
517 found = reachable
518 garbage = @allocations.reject { |name, _| found.include?(name) }.values.sort_by(&:start)
519 emit("GC collected=#{garbage.length} bytes=#{garbage.sum(&:size)}")
520 garbage.each do |block|
521 name = block.owner
522 record(name, "COLLECT #{name} #{release_id(name)}")
523 end
524 @counters['collected'] += garbage.length
525 end

527 def sbrk(args)
528 amount = args[0].to_i
529 valid = if @heap.buddy?
530 amount == @heap.size && 2 * @heap.size <= MAX_HEAP
531 else
532 amount % UNIT == 0 && amount >= MIN_BLOCK && @heap.size + amount <= MAX_HEAP
533 end
534 raise CommandError, 'BAD SBRK SIZE' unless valid
535 block = @heap.extend(amount)
536 emit("SBRK heap=#{@heap.size} free=#{block.area}")
537 end

544 def compact(_args)
545 require_free_list
546 moved, used = @heap.compact
547 @counters['moves'] += moved.length
548 free = used < @heap.size ? "#{used}+#{@heap.size - used}" : 'none'
549 emit("COMPACT moved=#{moved.length} free=#{free}")
550 moved.each do |block, original|

553 end

555 def who(args)
556 address = args[0].to_i
557 return emit('WHO', address, 'outside') if address >= @heap.size
558 block = @heap.blocks[@heap.locate(address)]
559 return emit('WHO', address, 'free', block.area) unless block.used
560 return emit('WHO', address, 'header', block.owner) if address < block.payload

565 def dump(_args)
566 text = "DUMP mode=#{@heap.mode} heap=#{@heap.size} blocks=#{@heap.blocks.length}"
567 text += " rover=#{@heap.rover}" if @heap.mode == 'NEXT'

583 def freelist(_args)
584 free = @heap.free_blocks
585 free.sort_by! { |block| -block.stamp } if @heap.mode == 'SEGFIT'
586 emit("FREELIST count=#{free.length}")
587 free.map { |block| size_class(block.size) }.uniq.sort.each do |power|

589 emit("#{2**power}-#{2**(power + 1) - 1}:", *matching)

593 def tree_node(leaves, start, size, depth)
594 label = ' ' * (2 * depth) + "#{start}+#{size}"
595 block = leaves[start]
596 if block && block.size == size
597 block.used ? emit(label, 'USED', block.owner) : emit(label, 'FREE')
598 return
599 end
600 emit(label, 'split')
601 half = size / 2
602 tree_node(leaves, start, half, depth + 1)
603 tree_node(leaves, start + half, half, depth + 1)
604 end

606 def tree(_args)
607 raise CommandError, 'REQUIRES BUDDY MODE' unless @heap.buddy?
608 emit("TREE heap=#{@heap.size}")
609 leaves = @heap.blocks.to_h { |block| [block.start, block] }
610 tree_node(leaves, 0, @heap.size, 0)
611 end

613 def stats(_args)
614 used, free = @heap.used_blocks, @heap.free_blocks
615 free_total = free.sum(&:size)
616 largest = free.map(&:size).max || 0
617 emit("STATS used=#{used.sum(&:size)} used_blocks=#{used.length}",

619 "frag=#{fragmentation_percent(free_total, largest)}",
620 "requested=#{used.sum(&:requested)} peak=#{@peak}")
621 emit('OPS', *COUNTERS.map { |name| "#{name}=#{@counters[name]}" })
622 end

624 def leaks(_args)
625 found = reachable
626 names = @allocations.keys.sort
627 requested = names.sum { |name| @allocations[name].requested }
628 lost = names.count { |name| !found.include?(name) }
629 emit("LEAKS count=#{names.length} requested=#{requested} lost=#{lost}")
630 names.each do |name|
631 block = @allocations.fetch(name)
632 emit("LEAK #{name} addr=#{block.payload} block=#{block.size}",
633 "req=#{block.requested} #{found.include?(name) ? 'reachable' : 'lost'}")
634 end
635 end

644 def bench(args)
645 size, seed, operations, max_request = args.map(&:to_i)
646 unless power_of_two?(size) && size.between?(32, MAX_HEAP)
647 raise CommandError, 'BAD HEAP SIZE'
648 end
649 emit("BENCH heap=#{size} seed=#{seed} ops=#{operations} maxreq=#{max_request}")
650 MODES.each { |mode| emit(experiment(mode, size, seed, operations, max_request)) }
651 end

653 def execute(line)
654 name, *arguments = line.split
655 begin
656 if @heap.nil? && name != 'HEAP' && name != 'BENCH'
657 raise CommandError, 'NO HEAP'
658 end
659 send(METHODS.fetch(name), arguments)
660 rescue CommandError => e
661 emit('ERROR', e.message)
662 ensure
663 @peak = [@peak, @heap.used_total].max if @heap
664 end
665 end
