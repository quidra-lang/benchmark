30function logic(kind, a)
31 y=if kind in ("AND", "NAND")
32 '0' in a ? '0' : 'X' in a ? 'X' : '1'
33 elseif kind in ("OR", "NOR")
34 '1' in a ? '1' : 'X' in a ? 'X' : '0'
35 elseif kind in ("XOR", "XNOR")
36 'X' in a ? 'X' : isodd(count(==('1'), a)) ? '1' : '0'
37 elseif kind=="MUX2"
38 a[1]=='X' ? (a[2]==a[3] ? a[2] : 'X') : a[a[1]=='0' ? 2 : 3]
39 else
40 a[1]
41 end
42 kind in ("NAND", "NOR", "XNOR", "NOT") ? (y=='0' ? '1' : y=='1' ? '0' : 'X') : y
43end

136function evaluate(c, b, stuck = "", v = 'X')
137 values=Dict(s=>(s==stuck ? v : b[i]) for (i, s) in enumerate(c.inputs))
138 for i in c.order
139 g=c.gates[i]
140 values[g.output]=g.output==stuck ? v : logic(g.kind, [values[s] for s in g.inputs])
141 end
142 join(values[s] for s in c.outputs)
143end
144function arrivals(c, period)
145 late=Dict{String,Int}()
146 early=Dict{String,Int}()
147 prev=Dict{String,String}()
148 if period
149 for i in c.flops
150 g=c.gates[i]
151 late[g.output]=early[g.output]=g.delay
152 end
153 else
154 for s in c.inputs
155 late[s]=early[s]=0
156 end
157 end
158 for i in c.order
159 g=c.gates[i]
160 a=[s for s in g.inputs if haskey(late, s)]
161 isempty(a)&&continue
162 best=a[1]
163 for s in a
164 late[s]>late[best]&&(best=s)
165 end
166 late[g.output]=late[best]+g.delay
167 early[g.output]=minimum(early[s] for s in a)+g.delay
168 prev[g.output]=best
169 end
170 late, early, prev
171end

212function schedule!(s, t, x, v)
213 e=Event(t, s.sequence, x, v)
214 s.sequence+=1
215 push!(s.queue, e)
216 i=length(s.queue)
217 while i>1
218 p=i÷2
219 before(e, s.queue[p])||break
220 s.queue[i]=s.queue[p]
221 i=p
222 end
223 s.queue[i]=e
224end
225function popevent!(s)
226 result=s.queue[1]
227 last=pop!(s.queue)
228 if !isempty(s.queue)
229 i=1
230 while 2*i<=length(s.queue)
231 j=2*i
232 j+1<=length(s.queue)&&before(s.queue[j+1], s.queue[j])&&(j+=1)
233 before(s.queue[j], last)||break
234 s.queue[i]=s.queue[j]
235 i=j
236 end
237 s.queue[i]=last
238 end
239 result
240end

242function run!(s, ending)
243 total=0
244 glitches=0
245 while !isempty(s.queue)&&s.queue[1].time<ending
246 t=s.queue[1].time
247 original=Dict{String,Char}()
248 count=0
249 while !isempty(s.queue)&&s.queue[1].time==t
250 if count==1000
251 println("halted: oscillation at time $t")
252 empty!(s.queue)
253 s.time=t
254 s.halted=true
255 return
256 end
257 e=popevent!(s)
258 count+=1
259 s.processed+=1
260 old=s.values[e.signal]
261 old==e.value&&continue
262 get!(original, e.signal, old)
263 s.values[e.signal]=e.value
264 for i in s.c.readers[e.signal]
265 g=s.c.gates[i]
266 if g.kind!="DFF"
267 schedule!(
268 s,
269 t+g.delay,
270 g.output,
271 logic(g.kind, [s.values[x] for x in g.inputs]),
272 )
273 elseif e.signal==g.inputs[1]&&old=='0'&&e.value=='1'
274 schedule!(s, t+g.delay, g.output, s.values[g.inputs[2]])
275 end
276 end
277 end
278 total+=count
279 for (x, v) in original
280 if s.values[x]==v
281 glitches+=1
282 else
283 push!(s.history[x], (t, s.values[x]))
284 end
285 end
286 for x in s.watch
287 haskey(original, x)&&s.values[x]!=original[x]&&println("$t $x $(s.values[x])")
288 end
289 end
290 s.time=ending
291 println("run $ending events $total glitches $glitches pending $(length(s.queue))")
292end

307function define!(s, header, lines)

309 try
310 w=String.(split(lines[1], ' '))

312 "bad module header",

330 gates=0
331 for i in 2:length(lines)
332 line=header+i-1
333 a=String.(split(lines[i], ' '))

335 if kind in TYPES

342 hi=findfirst(==(kind), TYPES)<=6 ? 8 : lo

344 for x in vcat(inputs, [output])
345 nameok(x)||fail("bad name $x")
346 end
347 (numberok(delay)&&parse(Int, delay)<=1000)||fail("bad delay $delay")
348 n in used&&fail("duplicate name $n")
349 output in driven&&fail("multiple drivers for $output")

351 push!(driven, output)
352 push!(body, Gate(n, kind, inputs, output, parse(Int, delay)))
353 gates+=1
354 append!(reads, [(line, x) for x in inputs])
355 elseif kind=="INST"
356 length(a)>=3||fail("bad instance syntax")

358 nameok(n)||fail("bad name $n")

367 port in ports&&fail("duplicate connection $port")

369 push!(ports, port)
370 push!(connections, port=>x)
371 end
372 for p in m.inputs
373 p in ports||fail("unconnected input $p")
374 end
375 n in used&&fail("duplicate name $n")
376 for (p, x) in connections
377 if p in m.outputs
378 x in driven&&fail("multiple drivers for $x")
379 push!(driven, x)
380 end
381 end
382 push!(used, n)
383 push!(body, Instance(n, m, connections))
384 append!(reads, [(line, x) for (p, x) in connections if p in m.inputs])
385 else
386 fail("unknown statement $kind")
387 end
388 end
389 append!(reads, [(header, x) for x in outs])
390 for (n, x) in reads

393 fail("undriven signal $x")
394 end
395 end
396 m=LogicModule(name, ins, outs, body)
397 s.modules[name]=m
398 push!(s.order, m)
399 println(
400 "defined $name inputs $(length(ins)) outputs $(length(outs)) gates $gates instances $(length(body)-gates)",
401 )
402 catch e
403 e isa InputError||rethrow()

405 end
406end

681 "circuit $(sim.c.name) time $(sim.time) pending $(length(sim.queue)) events $(sim.processed) halted $(sim.halted ? "yes" : "no")",
682 )
683 println(
684 "values 0:$(count(==('0'),values(sim.values))) 1:$(count(==('1'),values(sim.values))) X:$(count(==('X'),values(sim.values)))",
685 )
686 end
687 catch e
688 e isa InputError||rethrow()
689 println("error: line $line: $(e.reason)")
690 end
691end

701 last+=1
