53bool nameOk(String s) => RegExp(r'^[a-z][a-z0-9_]{0,15}$').hasMatch(s);

56 if (!numberOk(s) || positive && s == '0') fail('bad number $s');

65String bits(int r, int n) =>
66 [for (var i = 0; i < n; i++) (r >> (n - 1 - i)) & 1].join();
67String logic(String k, List<String> a) {
68 String y;
69 if (k == 'AND' || k == 'NAND')
70 y = a.contains('0')
71 ? '0'
72 : a.contains('X')
73 ? 'X'
74 : '1';
75 else if (k == 'OR' || k == 'NOR')
76 y = a.contains('1')
77 ? '1'
78 : a.contains('X')
79 ? 'X'
80 : '0';
81 else if (k == 'XOR' || k == 'XNOR')
82 y = a.contains('X')
83 ? 'X'
84 : (a.where((x) => x == '1').length % 2).toString();
85 else if (k == 'MUX2')
86 y = a[0] == 'X' ? (a[1] == a[2] ? a[1] : 'X') : a[a[0] == '0' ? 1 : 2];
87 else
88 y = a[0];
89 return ['NAND', 'NOR', 'XNOR', 'NOT'].contains(k)
90 ? {'0': '1', '1': '0', 'X': 'X'}[y]!
91 : y;
92}

123class Circuit {
124 final String name;
125 final List<String> inputs, outputs;
126 final gates = <Gate>[], flops = <Gate>[], order = <Gate>[];
127 final driver = <String, Gate>{}, readers = <String, List<Gate>>{};
128 late final List<String> signals;
129 late final bool loop;
130 Circuit(LogicModule m)
131 : name = m.name,
132 inputs = m.inputs,
133 outputs = m.outputs {
134 expand(m, '', {});
135 signals = [...inputs, ...gates.map((g) => g.output)];
136 for (final s in signals) readers[s] = [];
137 for (final g in gates) {
138 driver[g.output] = g;
139 if (g.kind == 'DFF') flops.add(g);
140 for (final s in g.inputs.toSet()) readers[s]!.add(g);
141 }
142 final waiting = <Gate, int>{}, ready = <Gate>[];
143 for (final g in gates) {
144 if (g.kind == 'DFF') continue;
145 final n = g.inputs
146 .toSet()
147 .where((s) => driver.containsKey(s) && driver[s]!.kind != 'DFF')
148 .length;
149 waiting[g] = n;
150 if (n == 0) ready.add(g);
151 }
152 while (ready.isNotEmpty) {
153 final g = ready.removeLast();
154 order.add(g);
155 for (final r in readers[g.output]!)
156 if (r.kind != 'DFF') {
157 waiting[r] = waiting[r]! - 1;
158 if (waiting[r] == 0) ready.add(r);
159 }
160 }
161 loop = order.length != waiting.length;
162 }
163 void expand(LogicModule m, String prefix, Map<String, String> ports) {
164 String actual(String s) => ports[s] ?? prefix + s;
165 for (final x in m.body) {
166 if (x is Gate)
167 gates.add(
168 Gate(
169 prefix + x.name,
170 x.kind,
171 x.inputs.map(actual).toList(),
172 actual(x.output),
173 x.delay,
174 ),
175 );
176 else if (x is Instance)
177 expand(x.module, prefix + x.name + '.', {
178 for (final (p, s) in x.connections) p: actual(s),
179 });
180 }
181 }

183 void combinational() {
184 if (flops.isNotEmpty) fail('$name is not combinational');
185 if (loop) fail('$name has a combinational loop');
186 if (inputs.length > 8) fail('too many inputs ${inputs.length}');
187 }
188
189 String evaluate(String b, [String stuck = '', String v = 'X']) {
190 final values = {
191 for (var i = 0; i < inputs.length; i++)
192 inputs[i]: inputs[i] == stuck ? v : b[i],
193 };
194 for (final g in order)
195 values[g.output] = g.output == stuck
196 ? v
197 : logic(g.kind, g.inputs.map((s) => values[s]!).toList());
198 return outputs.map((s) => values[s]!).join();
199 }
200
201 Arrival arrivals(bool period) {
202 final late = <String, int>{},
203 early = <String, int>{},
204 previous = <String, String>{};
205 if (period) {
206 for (final g in flops) {
207 late[g.output] = g.delay;
208 early[g.output] = g.delay;
209 }
210 } else {
211 for (final s in inputs) {
212 late[s] = 0;
213 early[s] = 0;
214 }
215 }
216 for (final g in order) {
217 final a = g.inputs.where(late.containsKey).toList();
218 if (a.isEmpty) continue;
219 var best = a[0];
220 for (final s in a) if (late[s]! > late[best]!) best = s;
221 late[g.output] = late[best]! + g.delay;
222 early[g.output] =
223 a.map((s) => early[s]!).reduce((a, b) => a < b ? a : b) + g.delay;
224 previous[g.output] = best;
225 }
226 return Arrival(late, early, previous);
227 }
228}

242class Simulation {
243 final Circuit c;
244 final watch = <String>[];
245 late Map<String, String> values;
246 late Map<String, List<Change>> history;
247 final queue = <Event>[];
248 int time = 0, sequence = 0, processed = 0;
249 bool halted = false;
250 Simulation(this.c) {
251 reset();
252 }
253 void reset() {
254 values = {for (final s in c.signals) s: 'X'};
255 history = {for (final s in c.signals) s: []};
256 queue.clear();
257 time = 0;
258 sequence = 0;
259 processed = 0;
260 halted = false;
261 }
262
263 bool before(Event a, Event b) =>
264 a.time < b.time || a.time == b.time && a.sequence < b.sequence;
265 void schedule(int t, String s, String v) {
266 final e = Event(t, sequence++, s, v);
267 var i = queue.length;
268 queue.add(e);
269 while (i > 0) {
270 final p = (i - 1) ~/ 2;
271 if (!before(e, queue[p])) break;
272 queue[i] = queue[p];
273 i = p;
274 }
275 queue[i] = e;
276 }
277
278 Event pop() {
279 final result = queue[0], last = queue.removeLast();
280 if (queue.isNotEmpty) {
281 var i = 0;
282 while (2 * i + 1 < queue.length) {
283 var j = 2 * i + 1;
284 if (j + 1 < queue.length && before(queue[j + 1], queue[j])) j++;
285 if (!before(queue[j], last)) break;
286 queue[i] = queue[j];
287 i = j;
288 }
289 queue[i] = last;
290 }
291 return result;
292 }
293
294 List<String> describe(Iterable<String> a) =>
295 a.map((s) => '$s=${values[s]}').toList();
296 void run(int end) {
297 var total = 0, glitches = 0;
298 while (queue.isNotEmpty && queue[0].time < end) {
299 final t = queue[0].time, original = <String, String>{};
300 var count = 0;
301 while (queue.isNotEmpty && queue[0].time == t) {
302 if (count == 1000) {
303 emit('halted: oscillation at time $t');
304 queue.clear();
305 time = t;
306 halted = true;
307 return;
308 }
309 final e = pop();
310 count++;
311 processed++;
312 final old = values[e.signal]!;
313 if (old == e.value) continue;
314 original.putIfAbsent(e.signal, () => old);
315 values[e.signal] = e.value;
316 for (final g in c.readers[e.signal]!)
317 if (g.kind != 'DFF')
318 schedule(
319 t + g.delay,
320 g.output,
321 logic(g.kind, g.inputs.map((s) => values[s]!).toList()),
322 );
323 else if (e.signal == g.inputs[0] && old == '0' && e.value == '1')
324 schedule(t + g.delay, g.output, values[g.inputs[1]]!);
325 }
326 total += count;
327 for (final entry in original.entries) {
328 final s = entry.key;
329 if (values[s] == entry.value)
330 glitches++;
331 else
332 history[s]!.add(Change(t, values[s]!));
333 }
334 for (final s in watch)
335 if (original.containsKey(s) && values[s] != original[s])
336 emit('$t $s ${values[s]}');
337 }
338 time = end;
339 emit('run $end events $total glitches $glitches pending ${queue.length}');
340 }
341}

361 void define(int header, List<String> lines) {
362 var line = header;
363 try {
364 final w = lines[0].split(' ');
365 if (w.length < 5 ||
366 w[2] != 'IN' ||
367 w.skip(3).where((s) => s == 'OUT').length != 1 ||
368 w.last == 'OUT')
369 fail('bad module header');
370 final name = w[1];
371 if (!RegExp(r'^[A-Z][A-Z0-9_]{0,15}$').hasMatch(name))
372 fail('bad module name $name');
373 if (modules.containsKey(name)) fail('duplicate module $name');
374 final ins = w.sublist(3, w.indexOf('OUT', 3)),
375 outs = w.sublist(w.indexOf('OUT', 3) + 1),
376 seen = <String>{};
377 for (final p in [...ins, ...outs]) {
378 if (!nameOk(p)) fail('bad name $p');
379 if (!seen.add(p)) fail('duplicate port $p');
380 }
381 final driven = ins.toSet(),
382 used = <String>{},
383 body = <Item>[],
384 reads = <(int, String)>[];
385 var gates = 0;
386 for (var i = 1; i < lines.length; i++) {
387 line = header + i;
388 final a = lines[i].split(' '), kind = a[0];
389 if (types.contains(kind)) {
390 if (a.length < 5 ||
391 a[a.length - 3] != '->' ||
392 a.where((s) => s == '->').length != 1)
393 fail('bad gate syntax');
394 final n = a[1];
395 if (!nameOk(n)) fail('bad name $n');
396 final inputs = a.sublist(2, a.length - 3),
397 output = a[a.length - 2],
398 delay = a.last,
399 lo = ['NOT', 'BUF'].contains(kind)
400 ? 1
401 : kind == 'MUX2'
402 ? 3
403 : 2;
404 final hi = types.indexOf(kind) < 6 ? 8 : lo;
405 if (inputs.length < lo || inputs.length > hi)
406 fail('bad input count ${inputs.length} for $kind');
407 for (final s in [...inputs, output])
408 if (!nameOk(s)) fail('bad name $s');
409 if (!numberOk(delay) || int.parse(delay) > 1000)
410 fail('bad delay $delay');
411 if (used.contains(n)) fail('duplicate name $n');
412 if (driven.contains(output)) fail('multiple drivers for $output');
413 used.add(n);
414 driven.add(output);
415 body.add(Gate(n, kind, inputs, output, int.parse(delay)));
416 gates++;
417 reads.addAll(inputs.map((s) => (line, s)));
418 } else if (kind == 'INST') {
419 if (a.length < 3) fail('bad instance syntax');
420 final n = a[1];
421 if (!nameOk(n)) fail('bad name $n');
422 final m = lookup(a[2]),
423 connections = <(String, String)>[],
424 ports = <String>{};
425 for (final word in a.skip(3)) {
426 final p = word.split('=');
427 if (p.length != 2 || p[0].isEmpty || p[1].isEmpty)
428 fail('bad connection $word');
429 if (!m.inputs.contains(p[0]) && !m.outputs.contains(p[0]))
430 fail('unknown port ${p[0]}');
431 if (ports.contains(p[0])) fail('duplicate connection ${p[0]}');
432 if (!nameOk(p[1])) fail('bad name ${p[1]}');
433 ports.add(p[0]);
434 connections.add((p[0], p[1]));
435 }
436 for (final p in m.inputs)
437 if (!ports.contains(p)) fail('unconnected input $p');
438 if (used.contains(n)) fail('duplicate name $n');
439 for (final (p, s) in connections)
440 if (m.outputs.contains(p) && !driven.add(s))
441 fail('multiple drivers for $s');
442 used.add(n);
443 body.add(Instance(n, m, connections));
444 reads.addAll(
445 connections
446 .where((v) => m.inputs.contains(v.$1))
447 .map((v) => (line, v.$2)),
448 );
449 } else
450 fail('unknown statement $kind');
451 }
452 reads.addAll(outs.map((s) => (header, s)));
453 for (final (n, s) in reads)
454 if (!driven.contains(s)) {
455 line = n;
456 fail('undriven signal $s');
457 }
458 modules[name] = LogicModule(name, ins, outs, body);
459 emit(
460 'defined $name inputs ${ins.length} outputs ${outs.length} gates $gates instances ${body.length - gates}',
461 );
462 } on InputError catch (e) {
463 emit('error: line $line: ${e.reason}');
464 }
465 }

475 void execute(int line, List<String> w) {
476 try {
477 final cmd = w[0], a = w.sublist(1), n = counts[w[0]];
478 if (n == null) fail('unknown command $cmd');
479 if (a.length < n ||
480 !['WATCH', 'UNWATCH', 'SHOW'].contains(cmd) && a.length != n)
481 fail('wrong argument count');
482 if (simulationCommands.contains(cmd) && simulation == null)
483 fail('no circuit');
484 final s = simulation;
485 if (['SET', 'APPLY', 'PULSE', 'CLOCK', 'RUN'].contains(cmd) && s!.halted)
486 fail('simulation halted');
487 if (cmd == 'MODULES') {
488 emit('modules ${modules.length}');
489 for (final m in modules.values)
490 emit(
491 '${m.name} inputs ${m.inputs.length} outputs ${m.outputs.length} gates ${Circuit(m).gates.length}',
492 );
493 return;
494 }
495 if ([
496 'BUILD',
497 'NETLIST',
498 'HIERARCHY',
499 'STATS',
500 'TRUTHTABLE',
501 'EQUIV',
502 'FAULTS',
503 'CRITICAL',
504 'PERIOD',
505 ].contains(cmd)) {
506 final m = lookup(a[0]),
507 m2 = cmd == 'EQUIV' ? lookup(a[1]) : null,
508 c = Circuit(m);
509 if (cmd == 'BUILD') {
510 simulation = Simulation(c);
511 emit(
512 'built ${c.name} signals ${c.signals.length} gates ${c.gates.length}',
513 );
514 } else if (cmd == 'NETLIST') {
515 emit(
516 'netlist ${c.name} gates ${c.gates.length} signals ${c.signals.length}',
517 );
518 for (final g in c.gates)
519 emit(
520 '${g.name} ${g.kind} ${g.inputs.join(' ')} -> ${g.output} ${g.delay}',
521 );
522 } else if (cmd == 'HIERARCHY') {
523 emit(c.name);
524 hierarchy(m, 1);
525 } else if (cmd == 'STATS') {
526 final levels = {for (final x in c.signals) x: 0};
527 var depth = 0;
528 for (final g in c.order) {
529 final v =
530 g.inputs
531 .map((x) => levels[x]!)
532 .reduce((a, b) => a > b ? a : b) +
533 1;
534 levels[g.output] = v;
535 if (v > depth) depth = v;
536 }
537 emit(
538 'stats ${c.name} gates ${c.gates.length} signals ${c.signals.length} depth ${c.loop ? '-' : depth}',
539 );
540 final row = <Object>['types'];
541 for (final t in types) {
542 final count = c.gates.where((g) => g.kind == t).length;
543 if (count > 0) row.addAll([t, count]);
544 }
545 emit(row.join(' '));
546 var best = c.signals[0];
547 for (final x in c.signals)
548 if (c.readers[x]!.length > c.readers[best]!.length) best = x;
549 emit('max fanout $best ${c.readers[best]!.length}');
550 } else if (cmd == 'CRITICAL' || cmd == 'PERIOD') {
551 final period = cmd == 'PERIOD';
552 if (period && c.flops.isEmpty) fail('${c.name} has no flip-flops');
553 if (c.loop) fail('${c.name} has a combinational loop');
554 final arr = c.arrivals(period),
555 ends = period
556 ? c.flops.map((g) => g.inputs[1]).toList()
557 : c.outputs,
558 heading = period
559 ? 'flip-flops ${c.flops.length} period'
560 : 'critical';
561 String? best;
562 for (final x in ends) {
563 if (!period)
564 emit(
565 arr.late.containsKey(x)
566 ? '$x min ${arr.early[x]} max ${arr.late[x]}'
567 : '$x unreachable',
568 );
569 if (arr.late.containsKey(x) &&
570 (best == null || arr.late[x]! > arr.late[best]!))
571 best = x;
572 }
573 if (best == null)
574 emit('$heading none');
575 else {
576 final endpoint = best, path = [best];
577 while (arr.previous.containsKey(path.last))
578 path.add(arr.previous[path.last]!);
579 emit(
580 '$heading ${arr.late[endpoint]}: ${path.reversed.join(' -> ')}',
581 );
582 }
583 } else {
584 c.combinational();
585 final rows = [
586 for (var r = 0; r < (1 << c.inputs.length); r++)
587 bits(r, c.inputs.length),
588 ],
589 expected = rows.map(c.evaluate).toList();
590 if (cmd == 'TRUTHTABLE') {
591 emit([...c.inputs, '|', ...c.outputs].join(' '));
592 for (var r = 0; r < rows.length; r++)
593 emit(
594 [...rows[r].split(''), '|', ...expected[r].split('')].join(' '),
595 );
596 } else if (cmd == 'EQUIV') {
597 final d = Circuit(m2!);
598 d.combinational();
599 if (c.inputs.length != d.inputs.length ||
600 c.outputs.length != d.outputs.length)
601 fail('port mismatch');
602 for (var r = 0; r < rows.length; r++) {
603 final v = d.evaluate(rows[r]);
604 if (v != expected[r]) {
605 emit('not equivalent at ${rows[r]}: ${expected[r]} vs $v');
606 return;
607 }
608 }
609 emit('equivalent');
610 } else {
611 final undetected = <String>[], tests = <int>{};
612 for (final x in c.signals)
613 for (final v in ['0', '1']) {
614 var found = -1;
615 for (var r = 0; r < rows.length; r++)
616 if (c.evaluate(rows[r], x, v) != expected[r]) {
617 found = r;
618 break;
619 }
620 if (found < 0)
621 undetected.add('$x/$v');
622 else
623 tests.add(found);
624 }
625 final total = 2 * c.signals.length;
626 emit('faults $total detected ${total - undetected.length}');
627 for (final f in undetected) emit('undetected $f');
628 final sorted = tests.toList()..sort();
629 emit(['tests', ...sorted.map((r) => rows[r])].join(' '));
630 }
631 }
632 return;
633 }
634 final sim = s!;
635 if (cmd == 'RESET') {
636 sim.reset();
637 emit('reset');
638 } else if (cmd == 'SET') {
639 final t = parseTime(a[0]);
640 input(a[1]);
641 value(a[2]);
642 sim.schedule(t, a[1], a[2]);
643 emit('set ${a[1]} ${a[2]} at $t');
644 } else if (cmd == 'APPLY') {
645 final t = parseTime(a[0]), b = a[1];
646 if (b.length != sim.c.inputs.length || !RegExp(r'^[01X]*$').hasMatch(b))
647 fail('bad vector $b');
648 for (var i = 0; i < b.length; i++)
649 sim.schedule(t, sim.c.inputs[i], b[i]);
650 emit('apply $b at $t');
651 } else if (cmd == 'PULSE' || cmd == 'CLOCK') {
652 input(a[0]);
653 final t = parseTime(a[1]), width = number(a[2], true);
654 if (cmd == 'PULSE') {
655 sim.schedule(t, a[0], '1');
656 sim.schedule(t + width, a[0], '0');
657 emit('pulse ${a[0]} $t ${t + width}');
658 } else {
659 final count = number(a[3], true);
660 for (var i = 0; i < count; i++)
661 sim.schedule(t + i * width, a[0], (i % 2).toString());
662 emit('clock ${a[0]} events $count last ${t + (count - 1) * width}');
663 }
664 } else if (cmd == 'RUN') {
665 final t = number(a[0]);
666 if (t <= sim.time) fail('time $t is not after ${sim.time}');
667 sim.run(t);
668 } else if (cmd == 'WATCH' || cmd == 'UNWATCH') {
669 final seen = <String>{};
670 for (final x in a) {
671 signal(x);
672 final watched = sim.watch.contains(x);
673 if (cmd == 'WATCH'
674 ? (watched || seen.contains(x))
675 : (!watched || seen.contains(x)))
676 fail('${cmd == 'WATCH' ? 'already watched' : 'not watched'} $x');
677 seen.add(x);
678 }
679 if (cmd == 'WATCH')
680 sim.watch.addAll(a);
681 else
682 sim.watch.removeWhere(seen.contains);
683 emit('watching ${sim.watch.isEmpty ? 'nothing' : sim.watch.join(' ')}');
684 } else if (cmd == 'SHOW') {
685 for (final x in a) signal(x);
686 emit(
687 (a.isNotEmpty
688 ? sim.describe(a)
689 : [
690 ...sim.describe(sim.c.inputs),
691 '|',
692 ...sim.describe(sim.c.outputs),
693 ])
694 .join(' '),
695 );
696 } else if (cmd == 'CHECK') {
697 signal(a[0]);
698 value(a[1]);
699 emit(
700 'check ${a[0]}=${a[1]} ${sim.values[a[0]] == a[1] ? 'ok' : 'failed (${sim.values[a[0]]})'}',
701 );
702 } else if (cmd == 'PROBE') {
703 final x = a[0];
704 signal(x);
705 final g = sim.c.driver[x];
706 emit(
707 '$x=${sim.values[x]} driven by ${g == null ? 'input' : '${g.name} ${g.kind} ${sim.describe(g.inputs).join(' ')} delay ${g.delay}'}',
708 );
709 final readers = sim.c.readers[x]!;
710 emit(
711 ['fanout', readers.length, ...readers.map((g) => g.name)].join(' '),
712 );
713 } else if (cmd == 'WAVE') {
714 final first = number(a[0]), last = number(a[1]);
715 if (!(first < last && last <= sim.time)) fail('bad range $first $last');
716 if (sim.watch.isEmpty) fail('nothing watched');
717 for (final x in sim.watch) {
718 final h = sim.history[x]!, wave = StringBuffer();
719 var i = 0, v = 'X';
720 for (var t = first; t < last; t++) {
721 while (i < h.length && h[i].time <= t) v = h[i++].value;
722 wave.write(v);
723 }
724 emit('$x $wave');
725 }
726 } else if (cmd == 'HISTORY') {
727 final x = a[0];
728 signal(x);
729 final h = sim.history[x]!;
730 emit(
731 [
732 'history',
733 x,
734 'changes',
735 h.length,
736 ...h.map((v) => '${v.time}:${v.value}'),
737 ].join(' '),
738 );
739 } else if (cmd == 'ACTIVITY') {
740 final k = number(a[0], true),
741 ranked = [...sim.c.signals]
742 ..sort((x, y) {
743 final d = sim.history[y]!.length - sim.history[x]!.length;
744 return d != 0 ? d : x.compareTo(y);
745 });
746 for (final x in ranked.take(k)) emit('$x ${sim.history[x]!.length}');
747 } else if (cmd == 'STATUS') {
748 emit(
749 'circuit ${sim.c.name} time ${sim.time} pending ${sim.queue.length} events ${sim.processed} halted ${sim.halted ? 'yes' : 'no'}',
750 );
751 emit(
752 'values 0:${sim.values.values.where((v) => v == '0').length} 1:${sim.values.values.where((v) => v == '1').length} X:${sim.values.values.where((v) => v == 'X').length}',
753 );
754 }
755 } on InputError catch (e) {
756 emit('error: line $line: ${e.reason}');
757 }
758 }
759}

761void main() {
762 final lines = <String>[];
763 for (String? line; (line = stdin.readLineSync()) != null;) lines.add(line!);
764 final simulator = Simulator();
765 for (var i = 0; i < lines.length && lines[i] != 'END'; i++) {
766 final w = lines[i].split(' ');
767 if (w[0] == 'MODULE') {
768 var end = i + 1;
769 while (lines[end] != 'ENDMODULE') end++;
770 simulator.define(i + 1, lines.sublist(i, end));
771 i = end;
772 } else
773 simulator.execute(i + 1, w);
774 }
