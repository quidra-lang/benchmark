72function number(s: string, positive = false) {

80function bits(r: number, n: number) {
81 return Array.from({ length: n }, (_, i) => String((r >> (n - 1 - i)) & 1)).join("");
82}
83function logic(kind: string, a: string[]): string {
84 let y: string;
85 if (["AND", "NAND"].includes(kind)) y = a.includes("0") ? "0" : a.includes("X") ? "X" : "1";
86 else if (["OR", "NOR"].includes(kind)) y = a.includes("1") ? "1" : a.includes("X") ? "X" : "0";
87 else if (["XOR", "XNOR"].includes(kind))
88 y = a.includes("X") ? "X" : String(a.filter((x) => x === "1").length % 2);
89 else if (kind === "MUX2")
90 y = a[0] === "X" ? (a[1] === a[2] ? a[1] : "X") : a[a[0] === "0" ? 1 : 2];
91 else y = a[0];
92 return ["NAND", "NOR", "XNOR", "NOT"].includes(kind)
93 ? ({ "0": "1", "1": "0", X: "X" } as Record<string, string>)[y]
94 : y;
95}
96class Circuit {
97 constructor(module: Module) {
98 this.name = module.name;
99 this.inputs = module.inputs;
100 this.outputs = module.outputs;
101 this.gates = [];
102 const expand = (m: Module, p: string, map: Map<string, string>): void => {
103 const actual = (s: string) => map.get(s) ?? p + s;
104 for (const item of m.body) {
105 if (item.kind !== undefined)
106 this.gates.push({
107 ...item,
108 name: p + item.name,
109 inputs: item.inputs.map(actual),
110 output: actual(item.output),
111 });
112 else
113 expand(
114 item.module,
115 p + item.name + ".",
116 new Map(item.connections.map(([x, y]) => [x, actual(y)])),
117 );
118 }
119 };
120 expand(module, "", new Map());
121 this.signals = [...this.inputs, ...this.gates.map((g) => g.output)];
122 this.driver = new Map(this.gates.map((g) => [g.output, g]));
123 this.readers = new Map(this.signals.map((s) => [s, []]));
124 for (const g of this.gates) for (const s of new Set(g.inputs)) this.readers.get(s).push(g);
125 this.flops = this.gates.filter((g) => g.kind === "DFF");
126 const waiting = new Map<Gate, number>(),
127 ready: Gate[] = [];
128 for (const g of this.gates.filter((g) => g.kind !== "DFF")) {
129 const n = new Set(
130 g.inputs.filter((s) => this.driver.has(s) && this.driver.get(s).kind !== "DFF"),
131 ).size;
132 waiting.set(g, n);
133 if (!n) ready.push(g);
134 }
135 this.order = [];
136 while (ready.length) {
137 const g = ready.pop();
138 this.order.push(g);
139 for (const r of this.readers.get(g.output))
140 if (r.kind !== "DFF") {
141 waiting.set(r, waiting.get(r) - 1);
142 if (!waiting.get(r)) ready.push(r);
143 }
144 }
145 this.loop = this.order.length !== waiting.size;
146 }
147 combinational() {
148 if (this.flops.length) fail(`${this.name} is not combinational`);
149 if (this.loop) fail(`${this.name} has a combinational loop`);
150 if (this.inputs.length > 8) fail(`too many inputs ${this.inputs.length}`);
151 return this;
152 }
153 evaluate(b: string, stuck = "", v = "X") {
154 const vals = new Map(this.inputs.map((s, i) => [s, s === stuck ? v : b[i]]));
155 for (const g of this.order)
156 vals.set(
157 g.output,
158 g.output === stuck
159 ? v
160 : logic(
161 g.kind,
162 g.inputs.map((s) => vals.get(s)),
163 ),
164 );
165 return this.outputs.map((s) => vals.get(s)).join("");
166 }
167 arrivals(period: boolean) {
168 const late = new Map<string, number>(),
169 early = new Map<string, number>(),
170 prev = new Map<string, string>();
171 if (period)
172 for (const g of this.flops) {
173 late.set(g.output, g.delay);
174 early.set(g.output, g.delay);
175 }
176 else
177 for (const s of this.inputs) {
178 late.set(s, 0);
179 early.set(s, 0);
180 }
181 for (const g of this.order) {
182 const a = g.inputs.filter((s) => late.has(s));
183 if (!a.length) continue;
184 let best = a[0];
185 for (const s of a) if (late.get(s) > late.get(best)) best = s;
186 late.set(g.output, late.get(best) + g.delay);
187 early.set(g.output, Math.min(...a.map((s) => early.get(s))) + g.delay);
188 prev.set(g.output, best);
189 }
190 return { late, early, prev };
191 }
192}
193class Simulation {
194 constructor(c: Circuit) {
195 this.c = c;
196 this.watch = [];
197 this.reset();
198 }
199 reset() {
200 this.vals = new Map(this.c.signals.map((s) => [s, "X"]));
201 this.time = 0;
202 this.queue = [];
203 this.sequence = 0;
204 this.history = new Map(this.c.signals.map((s) => [s, []]));
205 this.processed = 0;
206 this.halted = false;
207 }
208 schedule(t: number, s: string, v: string) {
209 const event = { t, s, v, id: this.sequence++ };
210 let i = this.queue.length;
211 this.queue.push(event);
212 while (i > 0) {
213 const p = (i - 1) >> 1;
214 if (!this.before(event, this.queue[p])) break;
215 this.queue[i] = this.queue[p];
216 i = p;
217 }
218 this.queue[i] = event;
219 }
220 before(a: EventEntry, b: EventEntry) {
221 return a.t < b.t || (a.t === b.t && a.id < b.id);
222 }
223 pop() {
224 const result = this.queue[0],
225 last = this.queue.pop();
226 if (this.queue.length) {
227 let i = 0;
228 while (2 * i + 1 < this.queue.length) {
229 let j = 2 * i + 1;
230 if (j + 1 < this.queue.length && this.before(this.queue[j + 1], this.queue[j])) j++;
231 if (!this.before(this.queue[j], last)) break;
232 this.queue[i] = this.queue[j];
233 i = j;
234 }
235 this.queue[i] = last;
236 }
237 return result;
238 }
239 run(end: number) {
240 let total = 0,
241 glitches = 0;
242 while (this.queue.length && this.queue[0].t < end) {
243 const t = this.queue[0].t,
244 start = new Map(this.vals),
245 changed = new Set();
246 let count = 0;
247 while (this.queue.length && this.queue[0].t === t) {
248 if (count === 1000) {
249 emit(`halted: oscillation at time ${t}`);
250 this.queue = [];
251 this.time = t;
252 this.halted = true;
253 return;
254 }
255 const e = this.pop();
256 count++;
257 this.processed++;
258 const old = this.vals.get(e.s);
259 if (old === e.v) continue;
260 this.vals.set(e.s, e.v);
261 changed.add(e.s);
262 for (const g of this.c.readers.get(e.s)) {
263 if (g.kind !== "DFF")
264 this.schedule(
265 t + g.delay,
266 g.output,
267 logic(
268 g.kind,
269 g.inputs.map((s) => this.vals.get(s)),
270 ),
271 );
272 else if (e.s === g.inputs[0] && old === "0" && e.v === "1")
273 this.schedule(t + g.delay, g.output, this.vals.get(g.inputs[1]));
274 }
275 }
276 total += count;
277 for (const s of changed)
278 if (this.vals.get(s) === start.get(s)) glitches++;
279 else this.history.get(s).push([t, this.vals.get(s)]);
280 for (const s of this.watch)
281 if (this.vals.get(s) !== start.get(s)) emit(`${t} ${s} ${this.vals.get(s)}`);
282 }
283 this.time = end;
284 emit(`run ${end} events ${total} glitches ${glitches} pending ${this.queue.length}`);
285 }
286 describe(a: string[]) {
287 return a.map((s) => `${s}=${this.vals.get(s)}`);
288 }
289}

303 define(headerNo: number, lines: string[]) {
304 let lineNo = headerNo;
305 try {
306 const w = lines[0].split(" ");
307 if (
308 w.length < 5 ||
309 w[2] !== "IN" ||
310 w.slice(3).filter((s) => s === "OUT").length !== 1 ||
311 w.at(-1) === "OUT"
312 )
313 fail("bad module header");
314 const name = w[1];
315 if (!/^[A-Z][A-Z0-9_]{0,15}$/.test(name)) fail(`bad module name ${name}`);
316 if (this.modules.has(name)) fail(`duplicate module ${name}`);
317 const split = w.indexOf("OUT", 3),
318 inputs = w.slice(3, split),
319 outputs = w.slice(split + 1),
320 seen = new Set();
321 for (const s of [...inputs, ...outputs]) {
322 if (!nameOK(s)) fail(`bad name ${s}`);
323 if (seen.has(s)) fail(`duplicate port ${s}`);
324 seen.add(s);
325 }
326 const driven = new Set(inputs),
327 used = new Set<string>(),
328 body: (Gate | Instance)[] = [],
329 reads: [number, string][] = [];
330 for (let i = 1; i < lines.length; i++) {
331 lineNo = headerNo + i;
332 const a = lines[i].split(" "),
333 kind = a[0],
334 name = a[1];
335 if (TYPES.includes(kind)) {
336 if (
337 a.length < 5 ||
338 a.at(-3) !== "->" ||
339 a.filter((s) => s === "->").length !== 1
340 )
341 fail("bad gate syntax");
342 if (!nameOK(name)) fail(`bad name ${name}`);
343 const ins = a.slice(2, -3),
344 out = a.at(-2),
345 delay = a.at(-1),
346 n = ins.length;
347 const low = ["NOT", "BUF"].includes(kind) ? 1 : kind === "MUX2" ? 3 : 2,
348 high = TYPES.indexOf(kind) < 6 ? 8 : low;
349 if (n < low || n > high) fail(`bad input count ${n} for ${kind}`);
350 for (const s of [...ins, out]) if (!nameOK(s)) fail(`bad name ${s}`);
351 if (!numberOK(delay) || +delay > 1000) fail(`bad delay ${delay}`);
352 if (used.has(name)) fail(`duplicate name ${name}`);
353 if (driven.has(out)) fail(`multiple drivers for ${out}`);
354 used.add(name);
355 driven.add(out);
356 body.push({ name, kind, inputs: ins, output: out, delay: +delay });
357 for (const s of ins) reads.push([lineNo, s]);
358 } else if (kind === "INST") {
359 if (a.length < 3) fail("bad instance syntax");
360 if (!nameOK(name)) fail(`bad name ${name}`);
361 const m = this.lookup(a[2]),
362 connections: [string, string][] = [],
363 ports = new Set<string>();
364 for (const word of a.slice(3)) {
365 const p = word.split("=");
366 if (p.length !== 2 || !p[0] || !p[1]) fail(`bad connection ${word}`);
367 const [port, s] = p;
368 if (![...m.inputs, ...m.outputs].includes(port))
369 fail(`unknown port ${port}`);
370 if (ports.has(port)) fail(`duplicate connection ${port}`);
371 if (!nameOK(s)) fail(`bad name ${s}`);
372 ports.add(port);
373 connections.push([port, s]);
374 }
375 for (const p of m.inputs) if (!ports.has(p)) fail(`unconnected input ${p}`);
376 if (used.has(name)) fail(`duplicate name ${name}`);
377 for (const [p, s] of connections)
378 if (m.outputs.includes(p)) {
379 if (driven.has(s)) fail(`multiple drivers for ${s}`);
380 driven.add(s);
381 }
382 used.add(name);
383 body.push({ name, module: m, connections });
384 for (const [p, s] of connections)
385 if (m.inputs.includes(p)) reads.push([lineNo, s]);
386 } else fail(`unknown statement ${kind}`);
387 }
388 for (const s of outputs) reads.push([headerNo, s]);
389 for (const [n, s] of reads)
390 if (!driven.has(s)) {
391 lineNo = n;
392 fail(`undriven signal ${s}`);
393 }
394 this.modules.set(name, { name, inputs, outputs, body });
395 const gates = body.filter((x) => x.kind).length;
396 emit(
397 `defined ${name} inputs ${inputs.length} outputs ${outputs.length} gates ${gates} instances ${body.length - gates}`,
398 );
399 } catch (e: any) {
400 emit(`error: line ${lineNo}: ${e.message}`);
401 }
402 }
403 signal(s: string) {
404 if (!this.sim.vals.has(s)) fail(`unknown signal ${s}`);
405 }
406 input(s: string) {
407 if (!this.sim.c.inputs.includes(s)) fail(`not an input ${s}`);
408 }
409 time(w: string) {
410 const t = number(w);
411 if (t < this.sim.time) fail(`time ${t} is in the past`);
412 return t;
413 }
414 hierarchy(m: Module, depth: number): void {
415 for (const x of m.body)
416 if (x.kind === undefined) {
417 emit(" ".repeat(depth) + x.name + " " + x.module.name);
418 this.hierarchy(x.module, depth + 1);
419 }
420 }
421 execute(lineNo: number, w: string[]) {
422 try {
423 const cmd = w[0],
424 a = w.slice(1),
425 n = COUNTS[cmd];
426 if (!Object.hasOwn(COUNTS, cmd)) fail(`unknown command ${cmd}`);
427 if (a.length < n || (!["WATCH", "UNWATCH", "SHOW"].includes(cmd) && a.length !== n))
428 fail("wrong argument count");
429 if (SIM.includes(cmd) && !this.sim) fail("no circuit");
430 const s = this.sim;
431 if (["SET", "APPLY", "PULSE", "CLOCK", "RUN"].includes(cmd) && s.halted)
432 fail("simulation halted");
433 if (cmd === "MODULES") {
434 emit(`modules ${this.modules.size}`);
435 for (const m of this.modules.values())
436 emit(
437 `${m.name} inputs ${m.inputs.length} outputs ${m.outputs.length} gates ${new Circuit(m).gates.length}`,
438 );
439 return;
440 }
441 if (
442 [
443 "BUILD",
444 "NETLIST",
445 "HIERARCHY",
446 "STATS",
447 "TRUTHTABLE",
448 "EQUIV",
449 "FAULTS",
450 "CRITICAL",
451 "PERIOD",
452 ].includes(cmd)
453 ) {
454 const m = this.lookup(a[0]);
455 if (cmd === "EQUIV") this.lookup(a[1]);
456 const c = new Circuit(m);
457 if (cmd === "BUILD") {
458 this.sim = new Simulation(c);
459 emit(`built ${c.name} signals ${c.signals.length} gates ${c.gates.length}`);
460 } else if (cmd === "NETLIST") {
461 emit(`netlist ${c.name} gates ${c.gates.length} signals ${c.signals.length}`);
462 for (const g of c.gates)
463 emit(`${g.name} ${g.kind} ${g.inputs.join(" ")} -> ${g.output} ${g.delay}`);
464 } else if (cmd === "HIERARCHY") {
465 emit(m.name);
466 this.hierarchy(m, 1);
467 } else if (cmd === "STATS") {
468 const levels = new Map(
469 [...c.inputs, ...c.flops.map((g) => g.output)].map((x) => [x, 0]),
470 );
471 for (const g of c.order)
472 levels.set(g.output, Math.max(...g.inputs.map((x) => levels.get(x))) + 1);
473 emit(
474 `stats ${c.name} gates ${c.gates.length} signals ${c.signals.length} depth ${c.loop ? "-" : Math.max(...levels.values())}`,
475 );
476 emit(
477 [
478 "types",
479 ...TYPES.flatMap((t) => {
480 const n = c.gates.filter((g) => g.kind === t).length;
481 return n ? [t, String(n)] : [];
482 }),
483 ].join(" "),
484 );
485 let best = c.signals[0];
486 for (const x of c.signals)
487 if (c.readers.get(x).length > c.readers.get(best).length) best = x;
488 emit(`max fanout ${best} ${c.readers.get(best).length}`);
489 } else if (cmd === "CRITICAL" || cmd === "PERIOD") {
490 const period = cmd === "PERIOD";
491 if (period && !c.flops.length) fail(`${c.name} has no flip-flops`);
492 if (c.loop) fail(`${c.name} has a combinational loop`);
493 const { late, early, prev } = c.arrivals(period);
494 const ends = period ? c.flops.map((g) => g.inputs[1]) : c.outputs;
495 if (!period)
496 for (const x of ends)
497 emit(
498 late.has(x)
499 ? `${x} min ${early.get(x)} max ${late.get(x)}`
500 : `${x} unreachable`,
501 );
502 let best: string | null = null;
503 for (const x of ends)
504 if (late.has(x) && (best === null || late.get(x) > late.get(best)))
505 best = x;
506 const heading = period ? `flip-flops ${c.flops.length} period` : "critical";
507 if (best === null) emit(heading + " none");
508 else {
509 const path = [best];
510 while (prev.has(path.at(-1))) path.push(prev.get(path.at(-1)));
511 emit(`${heading} ${late.get(best)}: ${path.reverse().join(" -> ")}`);
512 }
513 } else {
514 c.combinational();
515 const rows = Array.from({ length: 2 ** c.inputs.length }, (_, r) =>
516 bits(r, c.inputs.length),
517 ),
518 expected = rows.map((b) => c.evaluate(b));
519 if (cmd === "TRUTHTABLE") {
520 emit([...c.inputs, "|", ...c.outputs].join(" "));
521 for (let r = 0; r < rows.length; r++)
522 emit([...rows[r], "|", ...expected[r]].join(" "));
523 } else if (cmd === "EQUIV") {
524 const d = this.circuit(a[1]).combinational();
525 if (
526 c.inputs.length !== d.inputs.length ||
527 c.outputs.length !== d.outputs.length
528 )
529 fail("port mismatch");
530 const r = rows.findIndex((b, i) => expected[i] !== d.evaluate(b));
531 emit(
532 r < 0
533 ? "equivalent"
534 : `not equivalent at ${rows[r]}: ${expected[r]} vs ${d.evaluate(rows[r])}`,
535 );
536 } else {
537 const undetected: string[] = [],
538 tests = new Set<number>();
539 for (const x of c.signals)
540 for (const v of "01") {
541 const r = rows.findIndex(
542 (b, i) => c.evaluate(b, x, v) !== expected[i],
543 );
544 if (r < 0) undetected.push(`${x}/${v}`);
545 else tests.add(r);
546 }
547 const total = 2 * c.signals.length;
548 emit(`faults ${total} detected ${total - undetected.length}`);
549 for (const f of undetected) emit("undetected " + f);
550 emit(
551 ["tests", ...[...tests].sort((x, y) => x - y).map((r) => rows[r])].join(
552 " ",
553 ),
554 );
555 }
556 }
557 return;
558 }
559 if (cmd === "RESET") {
560 s.reset();
561 emit("reset");
562 } else if (cmd === "SET") {
563 const t = this.time(a[0]);
564 this.input(a[1]);
565 value(a[2]);
566 s.schedule(t, a[1], a[2]);
567 emit(`set ${a[1]} ${a[2]} at ${t}`);
568 } else if (cmd === "APPLY") {
569 const t = this.time(a[0]),
570 b = a[1];
571 if (b.length !== s.c.inputs.length || !/^[01X]*$/.test(b)) fail(`bad vector ${b}`);
572 s.c.inputs.forEach((x, i) => s.schedule(t, x, b[i]));
573 emit(`apply ${b} at ${t}`);
574 } else if (cmd === "PULSE" || cmd === "CLOCK") {
575 this.input(a[0]);
576 const t = this.time(a[1]),
577 width = number(a[2], true);
578 if (cmd === "PULSE") {
579 s.schedule(t, a[0], "1");
580 s.schedule(t + width, a[0], "0");
581 emit(`pulse ${a[0]} ${t} ${t + width}`);
582 } else {
583 const count = number(a[3], true);
584 for (let i = 0; i < count; i++) s.schedule(t + i * width, a[0], String(i % 2));
585 emit(`clock ${a[0]} events ${count} last ${t + (count - 1) * width}`);
586 }
587 } else if (cmd === "RUN") {
588 const t = number(a[0]);
589 if (t <= s.time) fail(`time ${t} is not after ${s.time}`);
590 s.run(t);
591 } else if (cmd === "WATCH" || cmd === "UNWATCH") {
592 const seen = new Set();
593 for (const x of a) {
594 this.signal(x);
595 if (
596 cmd === "WATCH"
597 ? s.watch.includes(x) || seen.has(x)
598 : !s.watch.includes(x) || seen.has(x)
599 )
600 fail(`${cmd === "WATCH" ? "already watched" : "not watched"} ${x}`);
601 seen.add(x);
602 }
603 s.watch =
604 cmd === "WATCH" ? [...s.watch, ...a] : s.watch.filter((x) => !seen.has(x));
605 emit("watching " + (s.watch.join(" ") || "nothing"));
606 } else if (cmd === "SHOW") {
607 for (const x of a) this.signal(x);
608 emit(
609 (a.length
610 ? s.describe(a)
611 : [...s.describe(s.c.inputs), "|", ...s.describe(s.c.outputs)]
612 ).join(" "),
613 );
614 } else if (cmd === "CHECK") {
615 this.signal(a[0]);
