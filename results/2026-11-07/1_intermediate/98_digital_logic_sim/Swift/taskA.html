12func validName(_ s: String, _ upper: Bool = false) -> Bool {
13 let a = Array(s.utf8)
14 let lo: UInt8 = upper ? 65 : 97
15 let hi: UInt8 = upper ? 90 : 122
16 return (1...16).contains(a.count) && (lo...hi).contains(a[0]) && a.allSatisfy {
17 (lo...hi).contains($0) || (48...57).contains($0) || $0 == 95
18 }
19}

50func gateValue(_ kind: String, _ v: [String]) -> String {
51 var r: String
52 switch kind {
53 case "AND","NAND": r = v.contains("0") ? "0" : (v.contains("X") ? "X" : "1")
54 case "OR","NOR": r = v.contains("1") ? "1" : (v.contains("X") ? "X" : "0")
55 case "XOR","XNOR": r = v.contains("X") ? "X" : String(v.filter {
56 $0 == "1"
57 }.count % 2)
58 case "MUX2": r = v[0] == "0" ? v[1] : (v[0] == "1" ? v[2] : (v[1] == v[2] ? v[1] : "X"))
59 default: r = v[0]
60 }
61 if ["NAND","NOR","XNOR","NOT"].contains(kind) {
62 r = r == "0" ? "1" : (r == "1" ? "0" : "X")
63 }
64 return r
65}

120 func expand(_ m: Module, _ prefix: String, _ ports: [String:String]) -> [Gate] {
121 func actual(_ s: String) -> String {
122 ports[s] ?? prefix + s
123 }
124 var result: [Gate] = []
125 for item in m.body {
126 switch item {
127 case .gate(let g): result.append(Gate(name: prefix+g.name, kind: g.kind, inputs: g.inputs.map(actual), output: actual(g.output), delay: g.delay))
128 case .instance(let i): result += expand(modules[i.module]!,prefix+i.name+".",Dictionary(uniqueKeysWithValues: i.connections.map {
129 ($0.0,actual($0.1))
130 }))
131 }
132 }
133 return result
134 }
135 gates = expand(module,"",[:])
136 signals = inputs + gates.map {
137 $0.output
138 }
139 for s in signals {
140 readers[s] = []
141 }
142 for (i,g) in gates.enumerated() {
143 driver[g.output] = i
144 for s in Set(g.inputs) {
145 readers[s]!.append(i)
146 }
147 }
148 var ready = Set(inputs + dffs.map {
149 gates[$0].output
150 }), remaining = Set(gates.indices.filter {
151 gates[$0].kind != "DFF"
152 }), sorted: [Int] = []
153 while let i = remaining.sorted().first(where: {
154 gates[$0].inputs.allSatisfy {
155 ready.contains($0)
156 }
157 }) {
158 sorted.append(i)
159 ready.insert(gates[i].output)
160 remaining.remove(i)
161 }
162 if remaining.isEmpty {
163 order = sorted
164 }
165 }
166 func combinational() throws {
167 if !dffs.isEmpty {
168 try fail("\(name) is not combinational")
169 }
170 if order == nil {
171 try fail("\(name) has a combinational loop")
172 }
173 if inputs.count > 8 {
174 try fail("too many inputs \(inputs.count)")

177 func evaluate(_ b: [String], _ fault: String = "", _ stuck: String = "") -> String {
178 var v = Dictionary(uniqueKeysWithValues: zip(inputs,b).map {
179 ($0.0,$0.1)
180 })
181 if inputs.contains(fault) {
182 v[fault] = stuck
183 }
184 for i in order! {
185 let g = gates[i]
186 v[g.output] = g.output == fault ? stuck : gateValue(g.kind,g.inputs.map {
187 v[$0]!
188 })
189 }
190 return outputs.map {
191 v[$0]!
192 }.joined()
193 }
194 func arrivals(_ sources: [String:Int]) -> ([String:Int],[String:Int],[String:String]) {
195 var latest = sources
196 var earliest = sources
197 var previous: [String:String] = [:]
198 for i in order! {
199 let g = gates[i]
200 let reached = g.inputs.filter {
201 latest[$0] != nil
202 }
203 if let first = reached.first {
204 let best = reached.dropFirst().reduce(first) {
205 latest[$1]! > latest[$0]! ? $1 : $0
206 }
207 latest[g.output] = latest[best]! + g.delay
208 earliest[g.output] = reached.map {
209 earliest[$0]!
210 }.min()! + g.delay
211 previous[g.output] = best
212 }
213 }
214 return (latest,earliest,previous)
215 }
216}

226func define(_ header: [String], _ body: [(Int,[String])], _ headerLine: Int) {
227 var line = headerLine
228 do {
229 if header.count < 5 || header[2] != "IN" || header.dropFirst(3).filter({
230 $0 == "OUT"
231 }).count != 1 || header.last == "OUT" {
232 try fail("bad module header")
233 }
234 let name = header[1]
235 if !validName(name,true) {
236 try fail("bad module name \(name)")
237 }
238 if modules[name] != nil {
239 try fail("duplicate module \(name)")
240 }
241 let split = header.dropFirst(3).firstIndex(of: "OUT")!
242 let inputs = Array(header[3..<split])
243 let outputs = Array(header[(split+1)...])
244 var seen = Set<String>()
245 var driven = Set(inputs)
246 var used = Set<String>()
247 var items: [Item] = []
248 var reads: [(Int,String)] = []
249 var count = 0
250 for p in inputs + outputs {
251 if !validName(p) {
252 try fail("bad name \(p)")
253 }
254 if seen.contains(p) {
255 try fail("duplicate port \(p)")
256 }
257 seen.insert(p)
258 }
259 for (n,w) in body {
260 line = n
261 if kinds.contains(w[0]) {
262 if w.count < 5 || w[w.count-3] != "->" || w.filter({
263 $0 == "->"
264 }).count != 1 {
265 try fail("bad gate syntax")
266 }
267 let kind = w[0]
268 let gname = w[1]
269 let ins = Array(w[2..<(w.count-3)])
270 let out = w[w.count-2]
271 let delay = w.last!
272 if !validName(gname) {
273 try fail("bad name \(gname)")
274 }
275 let range = ["NOT","BUF"].contains(kind) ? 1...1 : (kind == "MUX2" ? 3...3 : (kind == "DFF" ? 2...2 : 2...8))
276 if !range.contains(ins.count) {
277 try fail("bad input count \(ins.count) for \(kind)")
278 }
279 for s in ins + [out] {
280 if !validName(s) {
281 try fail("bad name \(s)")
282 }
283 }
284 guard let d = numeric(delay), d <= 1000 else {
285 try fail("bad delay \(delay)")
286 }
287 if used.contains(gname) {
288 try fail("duplicate name \(gname)")
289 }
290 if driven.contains(out) {
291 try fail("multiple drivers for \(out)")
292 }
293 used.insert(gname)
294 driven.insert(out)
295 count += 1
296 items.append(.gate(Gate(name:gname,kind:kind,inputs:ins,output:out,delay:d)))
297 reads += ins.map {
298 (n,$0)
299 }
300 } else if w[0] == "INST" {
301 if w.count < 3 {
302 try fail("bad instance syntax")
303 }
304 let iname = w[1]
305 if !validName(iname) {
306 try fail("bad name \(iname)")
307 }
308 let m = try lookup(w[2])
309 var connections: [(String,String)] = []
310 var ports = Set<String>()
311 for word in w.dropFirst(3) {
312 let parts = word.components(separatedBy:"=")
313 if parts.count != 2 || parts[0].isEmpty || parts[1].isEmpty {
314 try fail("bad connection \(word)")
315 }
316 let p = parts[0]
317 let s = parts[1]
318 if !(m.inputs+m.outputs).contains(p) {
319 try fail("unknown port \(p)")
320 }
321 if ports.contains(p) {
322 try fail("duplicate connection \(p)")
323 }
324 if !validName(s) {
325 try fail("bad name \(s)")
326 }
327 ports.insert(p)
328 connections.append((p,s))
329 }
330 for p in m.inputs where !ports.contains(p) {
331 try fail("unconnected input \(p)")
332 }
333 if used.contains(iname) {
334 try fail("duplicate name \(iname)")
335 }
336 for (p,s) in connections where m.outputs.contains(p) {
337 if driven.contains(s) {
338 try fail("multiple drivers for \(s)")
339 }
340 driven.insert(s)
341 }
342 used.insert(iname)
343 items.append(.instance(Instance(name:iname,module:m.name,connections:connections)))
344 reads += connections.filter {
345 m.inputs.contains($0.0)
346 }.map {
347 (n,$0.1)
348 }
349 } else {
350 try fail("unknown statement \(w[0])")
351 }
352 }
353 for (n,s) in reads + outputs.map({
354 (headerLine,$0)
355 }) {
356 line = n
357 if !driven.contains(s) {
358 try fail("undriven signal \(s)")
359 }
360 }
361 modules[name] = Module(name:name,inputs:inputs,outputs:outputs,body:items)
362 moduleOrder.append(name)
363 print("defined \(name) inputs \(inputs.count) outputs \(outputs.count) gates \(count) instances \(items.count-count)")
364 } catch let e as InputError {
365 print("error: line \(line): \(e.text)")
366 } catch {
367 }
368}

404 func before(_ a: Event, _ b: Event) -> Bool {
405 a.time < b.time || a.time == b.time && a.serial < b.serial
406 }
407 func schedule(_ t: Int, _ s: String, _ v: String) {
408 queue.append(Event(time:t,serial:serial,signal:s,value:v))
409 serial += 1
410 var i = queue.count-1
411 while i > 0 {
412 let p = (i-1)/2
413 if !before(queue[i],queue[p]) {
414 break
415 }
416 queue.swapAt(i,p)
417 i = p
418 }
419 }
420 func pop() -> Event {
421 let e = queue[0]
422 let last = queue.removeLast()
423 if !queue.isEmpty {
424 queue[0] = last
425 var i = 0
426 while i*2+1 < queue.count {
427 var c = i*2+1
428 if c+1 < queue.count && before(queue[c+1],queue[c]) {
429 c += 1
430 }
431 if !before(queue[c],queue[i]) {
432 break
433 }
434 queue.swapAt(c,i)
435 i = c
436 }
437 }
438 return e
439 }
440 func run(_ end: Int) {
441 var events = 0
442 var glitches = 0
443 while let first = queue.first, first.time < end {
444 let t = first.time
445 let original = values
446 var changed = Set<String>()
447 var count = 0
448 while queue.first?.time == t {
449 if count == 1000 {
450 print("halted: oscillation at time \(t)")
451 queue = []
452 time = t
453 halted = true
454 return
455 }
456 let e = pop()
457 count += 1
458 events += 1
459 processed += 1
460 let old = values[e.signal]!
461 if old == e.value {
462 continue
463 }
464 values[e.signal] = e.value
465 changed.insert(e.signal)
466 for i in circuit.readers[e.signal]! {
467 let g = circuit.gates[i]
468 if g.kind != "DFF" {
469 schedule(t+g.delay,g.output,gateValue(g.kind,g.inputs.map {
470 values[$0]!
471 }))
472 }
473 else if e.signal == g.inputs[0] && old == "0" && e.value == "1" {
474 schedule(t+g.delay,g.output,values[g.inputs[1]]!)
475 }
476 }
477 }
478 for s in changed {
479 if values[s] == original[s] {
480 glitches += 1
481 } else {
482 history[s]!.append((t,values[s]!))
483 }
484 }
485 for s in watch where values[s] != original[s] {
486 print("\(t) \(s) \(values[s]!)")
487 }
488 }
489 time = end
490 print("run \(end) events \(events) glitches \(glitches) pending \(queue.count)")
491 }
