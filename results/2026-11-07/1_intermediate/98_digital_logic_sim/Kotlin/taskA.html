32private fun bits(r: Int, n: Int) =

35private fun logic(kind: String, a: List<Char>): Char {
36 var y =
37 when (kind) {
38 "AND",
39 "NAND" -> if ('0' in a) '0' else if ('X' in a) 'X' else '1'
40 "OR",
41 "NOR" -> if ('1' in a) '1' else if ('X' in a) 'X' else '0'
42 "XOR",
43 "XNOR" -> if ('X' in a) 'X' else if (a.count { it == '1' } % 2 == 1) '1' else '0'
44 "MUX2" ->
45 if (a[0] == 'X') if (a[1] == a[2]) a[1] else 'X' else a[if (a[0] == '0') 1 else 2]
46 else -> a[0]
47 }
48 if (kind in listOf("NAND", "NOR", "XNOR", "NOT"))
49 y =
50 when (y) {
51 '0' -> '1'
52 '1' -> '0'
53 else -> 'X'
54 }
55 return y
56}

87private class Circuit(m: Mod) {
88 val name = m.name
89 val inputs = m.inputs
90 val outputs = m.outputs
91 val gates = mutableListOf<Gate>()
92 val signals: List<String>
93 val driver = mutableMapOf<String, Gate>()
94 val readers = mutableMapOf<String, MutableList<Gate>>()
95 val flops = mutableListOf<Gate>()
96 val order = mutableListOf<Gate>()
97 val loop: Boolean
98
99 init {
100 expand(m, "", emptyMap())
101 signals = inputs + gates.map { it.output }
102 for (s in signals) readers[s] = mutableListOf()
103 for (g in gates) {
104 driver[g.output] = g
105 if (g.kind == "DFF") flops.add(g)
106 for (s in g.inputs.distinct()) readers.getValue(s).add(g)
107 }
108 val waiting = mutableMapOf<Gate, Int>()
109 val ready = ArrayDeque<Gate>()
110 for (g in gates) {
111 if (g.kind == "DFF") continue
112 val n = g.inputs.distinct().count { driver[it]?.kind?.let { k -> k != "DFF" } == true }
113 waiting[g] = n
114 if (n == 0) ready.addLast(g)
115 }
116 while (ready.isNotEmpty()) {
117 val g = ready.removeLast()
118 order.add(g)
119 for (r in readers.getValue(g.output)) if (r.kind != "DFF") {
120 val n = waiting.getValue(r) - 1
121 waiting[r] = n
122 if (n == 0) ready.addLast(r)
123 }
124 }
125 loop = order.size != waiting.size
126 }

128 private fun expand(m: Mod, prefix: String, ports: Map<String, String>) {
129 fun actual(s: String) = ports[s] ?: prefix + s
130 for (x in m.body) when (x) {
131 is Gate ->
132 gates.add(
133 Gate(
134 prefix + x.name,
135 x.kind,
136 x.inputs.map { actual(it) },
137 actual(x.output),
138 x.delay,
139 ),
140 )
141 is Instance ->
142 expand(
143 x.module,
144 prefix + x.name + ".",
145 x.connections.mapValues { actual(it.value) },
146 )
147 }
148 }

150 fun combinational() {
151 if (flops.isNotEmpty()) fail("$name is not combinational")
152 if (loop) fail("$name has a combinational loop")
153 if (inputs.size > 8) fail("too many inputs ${inputs.size}")
154 }

156 fun evaluate(b: String, stuck: String = "", v: Char = 'X'): String {
157 val values =
158 inputs.mapIndexed { i, s -> s to if (s == stuck) v else b[i] }.toMap().toMutableMap()
159 for (g in order) values[g.output] =
160 if (g.output == stuck) v else logic(g.kind, g.inputs.map { values.getValue(it) })
161 return outputs.map { values.getValue(it) }.joinToString("")
162 }
163
164 fun arrivals(period: Boolean): Arrival {
165 val late = mutableMapOf<String, Int>()
166 val early = mutableMapOf<String, Int>()
167 val prev = mutableMapOf<String, String>()
168 if (period)
169 for (g in flops) {
170 late[g.output] = g.delay
171 early[g.output] = g.delay
172 }
173 else
174 for (s in inputs) {
175 late[s] = 0
176 early[s] = 0
177 }
178 for (g in order) {
179 val a = g.inputs.filter { it in late }
180 if (a.isEmpty()) continue
181 var best = a[0]
182 for (s in a) if (late.getValue(s) > late.getValue(best)) best = s
183 late[g.output] = late.getValue(best) + g.delay
184 early[g.output] = a.minOf { early.getValue(it) } + g.delay
185 prev[g.output] = best
186 }
187 return Arrival(late, early, prev)
188 }
189}

195private class Simulation(val c: Circuit) {
196 val watch = mutableListOf<String>()
197 val values = mutableMapOf<String, Char>()
198 val history = mutableMapOf<String, MutableList<Change>>()
199 val queue = PriorityQueue(compareBy<Event> { it.time }.thenBy { it.sequence })
200 var time = 0
201 var sequence = 0L
202 var processed = 0L
203 var halted = false
204
205 init {
206 reset()
207 }
208
209 fun reset() {
210 values.clear()
211 history.clear()
212 for (s in c.signals) {
213 values[s] = 'X'
214 history[s] = mutableListOf()
215 }
216 queue.clear()
217 time = 0
218 sequence = 0
219 processed = 0
220 halted = false
221 }
222
223 fun schedule(t: Int, s: String, v: Char) {
224 queue.add(Event(t, sequence++, s, v))
225 }
226
227 fun describe(a: List<String>) = a.map { "$it=${values.getValue(it)}" }
228
229 fun run(end: Int) {
230 var total = 0
231 var glitches = 0
232 while (queue.isNotEmpty() && queue.peek().time < end) {
233 val t = queue.peek().time
234 var count = 0
235 val original = mutableMapOf<String, Char>()
236 while (queue.isNotEmpty() && queue.peek().time == t) {
237 if (count == 1000) {
238 println("halted: oscillation at time $t")
239 queue.clear()
240 time = t
241 halted = true
242 return
243 }
244 val e = queue.remove()
245 count++
246 processed++
247 val old = values.getValue(e.signal)
248 if (old == e.value) continue
249 original.putIfAbsent(e.signal, old)
250 values[e.signal] = e.value
251 for (g in c.readers.getValue(e.signal)) if (g.kind != "DFF")
252 schedule(
253 t + g.delay,
254 g.output,
255 logic(g.kind, g.inputs.map { values.getValue(it) }),
256 )
257 else if (e.signal == g.inputs[0] && old == '0' && e.value == '1')
258 schedule(t + g.delay, g.output, values.getValue(g.inputs[1]))
259 }
260 total += count
261 for ((s, v) in original) if (values[s] == v) glitches++
262 else history.getValue(s).add(Change(t, values.getValue(s)))
263 for (s in watch) if (s in original && values[s] != original[s])
264 println("$t $s ${values[s]}")
265 }
266 time = end
267 println("run $end events $total glitches $glitches pending ${queue.size}")
268 }
269}

291 fun define(header: Int, lines: List<String>) {
292 var line = header
293 try {
294 val w = lines[0].split(' ')
295 if (
296 w.size < 5 ||
297 w[2] != "IN" ||
298 w.drop(3).count { it == "OUT" } != 1 ||
299 w.last() == "OUT"
300 )
301 fail("bad module header")
302 val name = w[1]
303 if (!Regex("[A-Z][A-Z0-9_]{0,15}").matches(name)) fail("bad module name $name")
304 if (name in modules) fail("duplicate module $name")
305 val split = w.drop(3).indexOf("OUT") + 3
306 val ins = w.subList(3, split)
307 val outs = w.drop(split + 1)
308 val seen = mutableSetOf<String>()
309 for (p in ins + outs) {
310 if (!nameOK(p)) fail("bad name $p")
311 if (!seen.add(p)) fail("duplicate port $p")
312 }
313 val driven = ins.toMutableSet()
314 val used = mutableSetOf<String>()
315 val body = mutableListOf<Item>()
316 val reads = mutableListOf<Pair<Int, String>>()
317 var gates = 0
318 for (i in 1 until lines.size) {
319 line = header + i
320 val a = lines[i].split(' ')
321 val kind = a[0]
322 if (kind in types) {
323 if (a.size < 5 || a[a.size - 3] != "->" || a.count { it == "->" } != 1)
324 fail("bad gate syntax")
325 val n = a[1]
326 if (!nameOK(n)) fail("bad name $n")
327 val inputs = a.subList(2, a.size - 3)
328 val output = a[a.size - 2]
329 val delay = a.last()
330 val lo =
331 when (kind) {
332 "NOT",
333 "BUF" -> 1
334 "MUX2" -> 3
335 else -> 2
336 }
337 val hi = if (types.indexOf(kind) < 6) 8 else lo
338 if (inputs.size !in lo..hi) fail("bad input count ${inputs.size} for $kind")
339 for (s in inputs + output) if (!nameOK(s)) fail("bad name $s")
340 if (!numberOK(delay) || delay.toInt() > 1000) fail("bad delay $delay")
341 if (n in used) fail("duplicate name $n")
342 if (output in driven) fail("multiple drivers for $output")
343 used.add(n)
344 driven.add(output)
345 body.add(Gate(n, kind, inputs, output, delay.toInt()))
346 gates++
347 for (s in inputs) reads.add(line to s)
348 } else if (kind == "INST") {
349 if (a.size < 3) fail("bad instance syntax")
350 val n = a[1]
351 if (!nameOK(n)) fail("bad name $n")
352 val m = lookup(a[2])
353 val connections = linkedMapOf<String, String>()
354 for (word in a.drop(3)) {
355 val p = word.split('=')
356 if (p.size != 2 || p[0].isEmpty() || p[1].isEmpty())
357 fail("bad connection $word")
358 if (p[0] !in m.inputs && p[0] !in m.outputs) fail("unknown port ${p[0]}")
359 if (p[0] in connections) fail("duplicate connection ${p[0]}")
360 if (!nameOK(p[1])) fail("bad name ${p[1]}")
361 connections[p[0]] = p[1]
362 }
363 for (p in m.inputs) if (p !in connections) fail("unconnected input $p")
364 if (n in used) fail("duplicate name $n")
365 for ((p, s) in connections) if (p in m.outputs && !driven.add(s))
366 fail("multiple drivers for $s")
367 used.add(n)
368 body.add(Instance(n, m, connections))
369 for ((p, s) in connections) if (p in m.inputs) reads.add(line to s)
370 } else fail("unknown statement $kind")
371 }
372 for (s in outs) reads.add(header to s)
373 for ((n, s) in reads) if (s !in driven) {
374 line = n
375 fail("undriven signal $s")
376 }
377 modules[name] = Mod(name, ins, outs, body)
378 println(
379 "defined $name inputs ${ins.size} outputs ${outs.size} gates $gates instances ${body.size-gates}",
380 )
381 } catch (e: IllegalArgumentException) {
382 println("error: line $line: ${e.message}")
383 }
384 }

387 for (x in m.body) if (x is Instance) {
388 println(" ".repeat(depth) + x.name + " " + x.module.name)
389 hierarchy(x.module, depth + 1)

393 fun execute(line: Int, w: List<String>) {
394 try {
395 val cmd = w[0]
396 val a = w.drop(1)
397 val n = counts[cmd] ?: fail("unknown command $cmd")
398 if (a.size < n || (cmd !in setOf("WATCH", "UNWATCH", "SHOW") && a.size != n))
399 fail("wrong argument count")
400 if (cmd in simulationCommands && sim == null) fail("no circuit")
401 if (cmd in setOf("SET", "APPLY", "PULSE", "CLOCK", "RUN") && sim!!.halted)
402 fail("simulation halted")
403 if (cmd == "MODULES") {
404 println("modules ${modules.size}")
405 for (m in modules.values) println(
406 "${m.name} inputs ${m.inputs.size} outputs ${m.outputs.size} gates ${Circuit(m).gates.size}",
407 )
408 return
409 }
410 if (
411 cmd in
412 setOf(
413 "BUILD",
414 "NETLIST",
415 "HIERARCHY",
416 "STATS",
417 "TRUTHTABLE",
418 "EQUIV",
419 "FAULTS",
420 "CRITICAL",
421 "PERIOD",
422 )
423 ) {
424 val m = lookup(a[0])
425 val m2 = if (cmd == "EQUIV") lookup(a[1]) else null
426 val c = Circuit(m)
427 when (cmd) {
428 "BUILD" -> {
429 sim = Simulation(c)
430 println("built ${c.name} signals ${c.signals.size} gates ${c.gates.size}")
431 }
432 "NETLIST" -> {
433 println("netlist ${c.name} gates ${c.gates.size} signals ${c.signals.size}")
434 for (g in c.gates) println(
435 "${g.name} ${g.kind} ${g.inputs.joinToString(" ")} -> ${g.output} ${g.delay}",
436 )
437 }
438 "HIERARCHY" -> {
439 println(c.name)
440 hierarchy(m, 1)
441 }
442 "STATS" -> {
443 val levels = mutableMapOf<String, Int>()
444 var depth = 0
445 for (g in c.order) {
446 val v = g.inputs.maxOf { levels[it] ?: 0 } + 1
447 levels[g.output] = v
448 depth = maxOf(depth, v)
449 }
450 println(
451 "stats ${c.name} gates ${c.gates.size} signals ${c.signals.size} depth ${if(c.loop)"-" else depth}",
452 )
453 println(
454 "types" +
455 types
456 .mapNotNull { t ->
457 val count = c.gates.count { it.kind == t }
458 if (count > 0) " $t $count" else null
459 }
460 .joinToString(""),
461 )
462 var best = c.signals[0]
463 for (s in c.signals) if (
464 c.readers.getValue(s).size > c.readers.getValue(best).size
465 )
466 best = s
467 println("max fanout $best ${c.readers.getValue(best).size}")
468 }
469 "CRITICAL",
470 "PERIOD" -> {
471 val period = cmd == "PERIOD"
472 if (period && c.flops.isEmpty()) fail("${c.name} has no flip-flops")
473 if (c.loop) fail("${c.name} has a combinational loop")
474 val arr = c.arrivals(period)
475 val ends = if (period) c.flops.map { it.inputs[1] } else c.outputs
476 val heading =
477 if (period) "flip-flops ${c.flops.size} period" else "critical"
478 var best: String? = null
479 for (s in ends) {
480 if (!period)
481 println(
482 if (s in arr.late) "$s min ${arr.early[s]} max ${arr.late[s]}"
483 else "$s unreachable",
484 )
485 if (
486 s in arr.late &&
487 (best == null || arr.late.getValue(s) > arr.late.getValue(best))
488 )
489 best = s
490 }
491 if (best == null) println("$heading none")
492 else {
493 var current = best!!
494 val path = mutableListOf(current)
495 while (current in arr.previous) {
496 current = arr.previous.getValue(current)
497 path.add(current)
498 }
499 println(
500 "$heading ${arr.late[best]}: ${path.reversed().joinToString(" -> ")}",
501 )
502 }
503 }
504 else -> {
505 c.combinational()
506 val rows = (0 until (1 shl c.inputs.size)).map { bits(it, c.inputs.size) }
507 val expected = rows.map { c.evaluate(it) }
508 if (cmd == "TRUTHTABLE") {
509 println((c.inputs + "|" + c.outputs).joinToString(" "))
510 for (r in rows.indices) println(
511 (rows[r].map { it.toString() } +
512 "|" +
513 expected[r].map { it.toString() })
514 .joinToString(" "),
515 )
516 } else if (cmd == "EQUIV") {
517 val d = Circuit(m2!!)
518 d.combinational()
519 if (c.inputs.size != d.inputs.size || c.outputs.size != d.outputs.size)
520 fail("port mismatch")
521 for (r in rows.indices) {
522 val v = d.evaluate(rows[r])
523 if (v != expected[r]) {
524 println("not equivalent at ${rows[r]}: ${expected[r]} vs $v")
525 return
526 }
527 }
528 println("equivalent")
529 } else {
530 val undetected = mutableListOf<String>()
531 val tests = sortedSetOf<Int>()
532 for (s in c.signals) for (v in "01") {
533 val r =
534 rows.indices.firstOrNull {
535 c.evaluate(rows[it], s, v) != expected[it]
536 }
537 if (r == null) undetected.add("$s/$v") else tests.add(r)
538 }
539 val total = 2 * c.signals.size
540 println("faults $total detected ${total-undetected.size}")
541 for (f in undetected) println("undetected $f")
542 println((listOf("tests") + tests.map { rows[it] }).joinToString(" "))
543 }
544 }
545 }
546 return
547 }
548 val s = sim!!
549 when (cmd) {
550 "RESET" -> {
551 s.reset()
552 println("reset")
553 }
554 "SET" -> {
555 val t = time(a[0])
556 input(a[1])
557 val v = value(a[2])
558 s.schedule(t, a[1], v)
559 println("set ${a[1]} $v at $t")
560 }
561 "APPLY" -> {
562 val t = time(a[0])
563 val b = a[1]
564 if (b.length != s.c.inputs.size || b.any { it !in "01X" }) fail("bad vector $b")
565 for (i in b.indices) s.schedule(t, s.c.inputs[i], b[i])
566 println("apply $b at $t")
567 }
568 "PULSE",
569 "CLOCK" -> {
570 input(a[0])
571 val t = time(a[1])
572 val width = number(a[2], true)
573 if (cmd == "PULSE") {
574 s.schedule(t, a[0], '1')
575 s.schedule(t + width, a[0], '0')
576 println("pulse ${a[0]} $t ${t+width}")
577 } else {
578 val count = number(a[3], true)
579 for (i in 0 until count) s.schedule(
580 t + i * width,
581 a[0],
582 if (i % 2 == 0) '0' else '1',
583 )
584 println("clock ${a[0]} events $count last ${t+(count-1)*width}")
585 }
586 }
587 "RUN" -> {
588 val t = number(a[0])
589 if (t <= s.time) fail("time $t is not after ${s.time}")
590 s.run(t)
591 }
592 "WATCH",
593 "UNWATCH" -> {
594 val seen = mutableSetOf<String>()
595 for (x in a) {
596 signal(x)
597 val watched = x in s.watch
598 if (if (cmd == "WATCH") watched || x in seen else !watched || x in seen)
599 fail((if (cmd == "WATCH") "already watched " else "not watched ") + x)
600 seen.add(x)
601 }
602 if (cmd == "WATCH") s.watch.addAll(a) else s.watch.removeAll(seen)
603 println(
604 "watching " +
605 if (s.watch.isEmpty()) "nothing" else s.watch.joinToString(" "),
606 )
607 }
608 "SHOW" -> {
609 for (x in a) signal(x)
610 println(
611 (if (a.isNotEmpty()) s.describe(a)
612 else s.describe(s.c.inputs) + "|" + s.describe(s.c.outputs))
613 .joinToString(" "),
614 )
615 }
616 "CHECK" -> {
617 signal(a[0])
618 val v = value(a[1])
619 println(
620 "check ${a[0]}=$v " +
621 if (s.values[a[0]] == v) "ok" else "failed (${s.values[a[0]]})",
622 )
623 }
624 "PROBE" -> {
625 val x = a[0]
626 signal(x)
627 val g = s.c.driver[x]
628 println(
629 "$x=${s.values[x]} driven by " +
630 if (g == null) "input"
631 else
632 "${g.name} ${g.kind} ${s.describe(g.inputs).joinToString(" ")} delay ${g.delay}",
633 )
634 val readers = s.c.readers.getValue(x)
635 println(
636 (listOf("fanout", readers.size.toString()) + readers.map { it.name })
637 .joinToString(" "),
638 )
639 }
640 "WAVE" -> {
641 val first = number(a[0])
642 val last = number(a[1])
643 if (!(first < last && last <= s.time)) fail("bad range $first $last")
644 if (s.watch.isEmpty()) fail("nothing watched")
645 for (x in s.watch) {
646 val h = s.history.getValue(x)
647 var i = 0
648 var v = 'X'
649 val wave = buildString {
650 for (t in first until last) {
651 while (i < h.size && h[i].time <= t) v = h[i++].value
652 append(v)
653 }
654 }
655 println("$x $wave")
656 }
657 }
658 "HISTORY" -> {
659 val x = a[0]
660 signal(x)
661 val h = s.history.getValue(x)
662 println(
663 (listOf("history", x, "changes", h.size.toString()) +
664 h.map { "${it.time}:${it.value}" })
665 .joinToString(" "),
666 )
667 }
668 "ACTIVITY" -> {
669 val k = number(a[0], true)
670 for (x in
671 s.c.signals
672 .sortedWith(
673 compareBy<String> { -s.history.getValue(it).size }.thenBy { it },
674 )
675 .take(k)) println("$x ${s.history.getValue(x).size}")
676 }
677 "STATUS" -> {
678 println(
679 "circuit ${s.c.name} time ${s.time} pending ${s.queue.size} events ${s.processed} halted ${if(s.halted)"yes" else "no"}",
680 )

692fun main() {
693 val lines = generateSequence(::readLine).toList()
694 val simulator = Simulator()
695 var i = 0
696 while (i < lines.size && lines[i] != "END") {
697 val w = lines[i].split(' ')
698 if (w[0] == "MODULE") {
699 var end = i + 1
700 while (lines[end] != "ENDMODULE") end++
701 simulator.define(i + 1, lines.subList(i, end))
702 i = end + 1
703 } else {
704 simulator.execute(i + 1, w)
705 i++
706 }
707 }
