88char logic(const string& k, const vector<char>& a) {
89 auto contains = [&](char c) {
90 return find(a.begin(), a.end(), c) != a.end();
91 };
92 char y = a[0];
93 if (k == "AND" || k == "NAND") y = contains('0') ? '0' : contains('X') ? 'X' : '1';
94 else if (k == "OR" || k == "NOR") y = contains('1') ? '1' : contains('X') ? 'X' : '0';
95 else if (k == "XOR" || k == "XNOR") y = contains('X') ? 'X' : char('0' + count(a.begin(), a.end(), '1')%2);
96 else if (k == "MUX2") y = a[0] == 'X' ? (a[1] == a[2] ? a[1] : 'X') : a[a[0] == '0' ? 1 : 2];
97 if (k == "NAND" || k == "NOR" || k == "XNOR" || k == "NOT") y = y == '0' ? '1' : y == '1' ? '0' : 'X';
98 return y;
99}

127 void expand(const Module& m, const string& p, const map<string, string>& ports) {
128 auto actual = [&](const string& s) {
129 auto it = ports.find(s);
130 return it == ports.end() ? p + s : it->second;
131 };
132 for (const Item& x : m.body) if (!x.instance) {
133 Gate g = x.gate;
134 g.name = p + g.name;
135 for (string& s : g.inputs) s = actual(s);
136 g.output = actual(g.output);
137 gates.push_back(g);
138 } else {
139 map<string, string> mapping;
140 for (const auto&[port, s] : x.connections) mapping[port] = actual(s);
141 expand(*x.module, p + x.name + ".", mapping);
142 }
143 }
144 explicit Circuit(const Module& m) : name(m.name), inputs(m.inputs), outputs(m.outputs) {
145 expand(m, "",{});
146 signals = inputs;
147 for (size_t i = 0; i<gates.size(); i++) {
148 const Gate& g = gates[i];
149 signals.push_back(g.output);
150 driver[g.output] = int(i);
151 if (g.kind == "DFF") flops.push_back(int(i));
152 for (const string& s : set<string>(g.inputs.begin(), g.inputs.end())) readers[s].push_back(int(i));
153 }
154 for (const string& s : signals) readers.try_emplace(s);
155 vector<int> waiting(gates.size()), ready;
156 int needed = 0;
157 for (size_t i = 0; i<gates.size(); i++) {
158 const Gate& g = gates[i];
159 if (g.kind == "DFF") continue;
160 needed++;
161 for (const string& s : set<string>(g.inputs.begin(), g.inputs.end())) if (driver.count(s) && gates[driver.at(s)].kind != "DFF") waiting[i]++;
162 if (!waiting[i]) ready.push_back(int(i));
163 }
164 while (!ready.empty()) {
165 int i = ready.back();
166 ready.pop_back();
167 order.push_back(i);
168 for (int r : readers[gates[i].output]) if (gates[r].kind != "DFF" && !--waiting[r]) ready.push_back(r);
169 }
170 loop = int(order.size()) != needed;
171 }
172 void combinational() const {
173 if (!flops.empty()) fail(name + " is not combinational");
174 if (loop) fail(name + " has a combinational loop");
175 if (inputs.size()>8) fail("too many inputs " + to_string(inputs.size()));
176 }
177 string evaluate(const string& b, const string& stuck = "", char v = 'X') const {
178 map<string, char> values;
179 for (size_t i = 0; i<inputs.size(); i++) values[inputs[i]] = inputs[i] == stuck ? v : b[i];
180 for (int i : order) {
181 const Gate& g = gates[i];
182 vector<char> a;
183 for (const string& s : g.inputs) a.push_back(values.at(s));
184 values[g.output] = g.output == stuck ? v : logic(g.kind, a);
185 }
186 string result;
187 for (const string& s : outputs) result += values.at(s);
188 return result;
189 }
190 tuple<map<string, int>, map<string, int>, map<string, string>> arrivals(bool period) const {
191 map<string, int> late, early;
192 map<string, string> prev;
193 if (period) {
194 for (int i : flops) {
195 const Gate& g = gates[i];
196 late[g.output] = early[g.output] = g.delay;
197 }
198 } else for (const string& s : inputs) late[s] = early[s] = 0;
199 for (int i : order) {
200 const Gate& g = gates[i];
201 string best;
202 int minimum = 0;
203 for (const string& s : g.inputs) if (late.count(s)) {
204 if (best.empty()) {
205 best = s;
206 minimum = early[s];
207 } else if (late[s]> late[best]) best = s;
208 minimum = min(minimum, early[s]);
209 }
210 if (!best.empty()) {
211 late[g.output] = late[best] + g.delay;
212 early[g.output] = minimum + g.delay;
213 prev[g.output] = best;
214 }
215 }
216 return {late, early, prev};
217 }

230struct Simulation {
231 shared_ptr<Circuit> c;
232 map<string, char> values;
233 vector<string> watch;
234 map<string, vector<pair<int, char>>> history;
235 priority_queue<Event, vector<Event>, Later> queue;
236 int time = 0;
237 long long sequence = 0, processed = 0;
238 bool halted = false;
239 explicit Simulation(shared_ptr<Circuit> circuit) : c(circuit) {
240 reset();
241 }
242 void reset() {
243 values.clear();
244 history.clear();
245 for (const string& s : c->signals) {
246 values[s] = 'X';
247 history[s] = {};
248 }
249 queue = {};
250 time = 0;
251 sequence = processed = 0;
252 halted = false;
253 }
254 void schedule(int t, const string& s, char v) {
255 queue.push({t, sequence++, s, v});
256 }
257 vector<string> describe(const vector<string>& a) const {
258 vector<string> b;
259 for (const string& s : a) b.push_back(s + "=" + values.at(s));
260 return b;
261 }
262 void run(int end) {
263 int total = 0, glitches = 0;
264 while (!queue.empty() && queue.top().time<end) {
265 int t = queue.top().time, count = 0;
266 map<string, char> original;
267 while (!queue.empty() && queue.top().time == t) {
268 if (count == 1000) {
269 emit("halted: oscillation at time " + to_string(t));
270 queue = {};
271 time = t;
272 halted = true;
273 return;
274 }
275 Event e = queue.top();
276 queue.pop();
277 count++;
278 processed++;
279 char old = values[e.signal];
280 if (old == e.value) continue;
281 original.try_emplace(e.signal, old);
282 values[e.signal] = e.value;
283 for (int i : c->readers[e.signal]) {
284 const Gate& g = c->gates[i];
285 if (g.kind != "DFF") {
286 vector<char> a;
287 for (const string& s : g.inputs) a.push_back(values[s]);
288 schedule(t + g.delay, g.output, logic(g.kind, a));
289 } else if (e.signal == g.inputs[0] && old == '0' && e.value == '1') schedule(t + g.delay, g.output, values[g.inputs[1]]);
290 }
291 }
292 total += count;
293 for (const auto&[s, v] : original) if (values[s] == v) glitches++;
294 else history[s].push_back({t, values[s]});
295 for (const string& s : watch) if (original.count(s) && values[s] != original[s]) emit(to_string(t) + " " + s + " " + values[s]);
296 }
297 time = end;
298 emit("run " + to_string(end) + " events " + to_string(total) + " glitches " + to_string(glitches) + " pending " + to_string(queue.size()));
299 }

321 void define(int header, const vector<string>& lines) {
322 int line = header;
323 try {
324 vector<string> w = words(lines[0]);
325 if (w.size()<5 || w[2] != "IN" || count(w.begin() + 3, w.end(), "OUT") != 1 || w.back() == "OUT") fail("bad module header");
326 string name = w[1];
327 if (!regex_match(name, regex("[A-Z][A-Z0-9_]{0,15}"))) fail("bad module name " + name);
328 if (modules.count(name)) fail("duplicate module " + name);
329 auto split = find(w.begin() + 3, w.end(), "OUT");
330 auto m = make_shared<Module>();
331 m->name = name;
332 m->inputs.assign(w.begin() + 3, split);
333 m->outputs.assign(split + 1, w.end());
334 set<string> seen;
335 vector<string> ports = m->inputs;
336 ports.insert(ports.end(), m->outputs.begin(), m->outputs.end());
337 for (const string& p : ports) {
338 if (!nameOK(p)) fail("bad name " + p);
339 if (!seen.insert(p).second) fail("duplicate port " + p);
340 }
341 set<string> driven(m->inputs.begin(), m->inputs.end()), used;
342 vector<pair<int, string>> reads;
343 int gates = 0;
344 for (size_t i = 1; i<lines.size(); i++) {
345 line = header + int(i);
346 vector<string> a = words(lines[i]);
347 string kind = a[0];
348 Item item;
349 if (has(TYPES, kind)) {
350 if (a.size()<5 || a[a.size()-3] != "->" || count(a.begin(), a.end(), "->") != 1) fail("bad gate syntax");
351 string n = a[1];
352 if (!nameOK(n)) fail("bad name " + n);
353 vector<string> ins(a.begin() + 2, a.end()-3);
354 string output = a[a.size()-2], delay = a.back();
355 int low = (kind == "NOT" || kind == "BUF") ? 1 : kind == "MUX2" ? 3 : 2, high = find(TYPES.begin(), TYPES.end(), kind)-TYPES.begin()<6 ? 8 : low;
356 if (int(ins.size())<low || int(ins.size())> high) fail("bad input count " + to_string(ins.size()) + " for " + kind);
357 vector<string> names = ins;
358 names.push_back(output);
359 for (const string& s : names) if (!nameOK(s)) fail("bad name " + s);
360 if (!numberOK(delay) || stoi(delay)>1000) fail("bad delay " + delay);
361 if (used.count(n)) fail("duplicate name " + n);
362 if (driven.count(output)) fail("multiple drivers for " + output);
363 used.insert(n);
364 driven.insert(output);
365 item.gate = {n, kind, ins, output, stoi(delay)};
366 gates++;
367 for (const string& s : ins) reads.push_back({line, s});
368 } else if (kind == "INST") {
369 if (a.size()<3) fail("bad instance syntax");
370 string n = a[1];
371 if (!nameOK(n)) fail("bad name " + n);
372 auto child = lookup(a[2]);
373 set<string> connected;
374 for (size_t j = 3; j<a.size(); j++) {
375 const string& word = a[j];
376 size_t pos = word.find('=');
377 if (pos == string::npos || pos == 0 || pos + 1 == word.size() || word.find('=', pos + 1) != string::npos) fail("bad connection " + word);
378 string p = word.substr(0, pos), s = word.substr(pos + 1);
379 if (!has(child->inputs, p) && !has(child->outputs, p)) fail("unknown port " + p);
380 if (connected.count(p)) fail("duplicate connection " + p);
381 if (!nameOK(s)) fail("bad name " + s);
382 connected.insert(p);
383 item.connections.push_back({p, s});
384 }
385 for (const string& p : child->inputs) if (!connected.count(p)) fail("unconnected input " + p);
386 if (used.count(n)) fail("duplicate name " + n);
387 for (const auto&[p, s] : item.connections) if (has(child->outputs, p)) {
388 if (driven.count(s)) fail("multiple drivers for " + s);
389 driven.insert(s);
390 }
391 used.insert(n);
392 item.instance = true;
393 item.name = n;
394 item.module = child;
395 for (const auto&[p, s] : item.connections) if (has(child->inputs, p)) reads.push_back({line, s});
396 } else fail("unknown statement " + kind);
397 m->body.push_back(item);
398 }
399 for (const string& s : m->outputs) reads.push_back({header, s});
400 for (const auto&[n, s] : reads) if (!driven.count(s)) {
401 line = n;
402 fail("undriven signal " + s);
403 }
404 modules[name] = m;
405 moduleOrder.push_back(m);
406 emit("defined " + name + " inputs " + to_string(m->inputs.size()) + " outputs " + to_string(m->outputs.size()) + " gates " + to_string(gates) + " instances " + to_string(m->body.size()-gates));
407 } catch (const runtime_error& e) {
408 emit("error: line " + to_string(line) + ": " + e.what());
409 }
410 }
411 void hierarchy(const Module& m, int depth) const {
412 for (const Item& x : m.body) if (x.instance) {
413 emit(string(2*depth, ' ') + x.name + " " + x.module->name);
414 hierarchy(*x.module, depth + 1);
415 }

417 void execute(int line, const vector<string>& w) {
418 try {
419 string cmd = w[0];
420 vector<string> a(w.begin() + 1, w.end());
421 if (!COUNTS.count(cmd)) fail("unknown command " + cmd);
422 int n = COUNTS.at(cmd);
423 if (int(a.size())<n || ((cmd != "WATCH" && cmd != "UNWATCH" && cmd != "SHOW") && int(a.size()) != n)) fail("wrong argument count");
424 if (SIM.count(cmd) && !sim) fail("no circuit");
425 if ((cmd == "SET" || cmd == "APPLY" || cmd == "PULSE" || cmd == "CLOCK" || cmd == "RUN") && sim->halted) fail("simulation halted");
426 if (cmd == "MODULES") {
427 emit("modules " + to_string(modules.size()));
428 for (const auto& m : moduleOrder) emit(m->name + " inputs " + to_string(m->inputs.size()) + " outputs " + to_string(m->outputs.size()) + " gates " + to_string(Circuit(*m).gates.size()));
429 return;
430 }
431 if (has(words("BUILD NETLIST HIERARCHY STATS TRUTHTABLE EQUIV FAULTS CRITICAL PERIOD"), cmd)) {
432 auto m = lookup(a[0]);
433 shared_ptr<Module> m2;
434 if (cmd == "EQUIV") m2 = lookup(a[1]);
435 auto cp = make_shared<Circuit>(*m);
436 const Circuit& c = *cp;
437 if (cmd == "BUILD") {
438 sim = make_unique<Simulation>(cp);
439 emit("built " + c.name + " signals " + to_string(c.signals.size()) + " gates " + to_string(c.gates.size()));
440 } else if (cmd == "NETLIST") {
441 emit("netlist " + c.name + " gates " + to_string(c.gates.size()) + " signals " + to_string(c.signals.size()));
442 for (const Gate& g : c.gates) emit(g.name + " " + g.kind + " " + join(g.inputs) + " -> " + g.output + " " + to_string(g.delay));
443 } else if (cmd == "HIERARCHY") {
444 emit(c.name);
445 hierarchy(*m, 1);
446 } else if (cmd == "STATS") {
447 map<string, int> levels;
448 int depth = 0;
449 for (int i : c.order) {
450 const Gate& g = c.gates[i];
451 int v = 0;
452 for (const string& s : g.inputs) v = max(v, levels[s]);
453 levels[g.output] = v + 1;
454 depth = max(depth, v + 1);
455 }
456 emit("stats " + c.name + " gates " + to_string(c.gates.size()) + " signals " + to_string(c.signals.size()) + " depth " + (c.loop ? "-" : to_string(depth)));
457 vector<string> text = {"types"};
458 for (const string& t : TYPES) {
459 int n = 0;
460 for (const Gate& g : c.gates) n += g.kind == t;
461 if (n) {
462 text.push_back(t);
463 text.push_back(to_string(n));
464 }
465 }
466 emit(join(text));
467 string best = c.signals[0];
468 for (const string& s : c.signals) if (c.readers.at(s).size()> c.readers.at(best).size()) best = s;
469 emit("max fanout " + best + " " + to_string(c.readers.at(best).size()));
470 } else if (cmd == "CRITICAL" || cmd == "PERIOD") {
471 bool period = cmd == "PERIOD";
472 if (period && c.flops.empty()) fail(c.name + " has no flip-flops");
473 if (c.loop) fail(c.name + " has a combinational loop");
474 auto[late, early, prev] = c.arrivals(period);
475 vector<string> ends = c.outputs;
476 if (period) {
477 ends.clear();
478 for (int i : c.flops) ends.push_back(c.gates[i].inputs[1]);
479 }
480 string best, heading = period ? "flip-flops " + to_string(c.flops.size()) + " period" : "critical";
481 for (const string& s : ends) {
482 if (!period) emit(late.count(s) ? s + " min " + to_string(early[s]) + " max " + to_string(late[s]) : s + " unreachable");
483 if (late.count(s) && (best.empty() || late[s]> late[best])) best = s;
484 }
485 if (best.empty()) emit(heading + " none");
486 else {
487 vector<string> path = {best};
488 while (prev.count(path.back())) path.push_back(prev[path.back()]);
489 reverse(path.begin(), path.end());
490 emit(heading + " " + to_string(late[best]) + ": " + join(path, " -> "));
491 }
492 } else {
493 c.combinational();
494 vector<string> rows, expected;
495 for (int r = 0; r<(1<<c.inputs.size()); r++) {
496 string b = bits(r, int(c.inputs.size()));
497 rows.push_back(b);
498 expected.push_back(c.evaluate(b));
499 }
500 if (cmd == "TRUTHTABLE") {
501 vector<string> text = c.inputs;
502 text.push_back("|");
503 text.insert(text.end(), c.outputs.begin(), c.outputs.end());
504 emit(join(text));
505 for (size_t r = 0; r<rows.size(); r++) {
506 text.clear();
507 for (char x : rows[r]) text.push_back(string(1, x));
508 text.push_back("|");
509 for (char x : expected[r]) text.push_back(string(1, x));
510 emit(join(text));
511 }
512 } else if (cmd == "EQUIV") {
513 Circuit d(*m2);
514 d.combinational();
515 if (c.inputs.size() != d.inputs.size() || c.outputs.size() != d.outputs.size()) fail("port mismatch");
516 for (size_t r = 0; r<rows.size(); r++) {
517 string v = d.evaluate(rows[r]);
518 if (v != expected[r]) {
519 emit("not equivalent at " + rows[r] + ": " + expected[r] + " vs " + v);
520 return;
521 }
522 }
523 emit("equivalent");
524 } else {
525 vector<string> undetected;
526 set<int> tests;
527 for (const string& s : c.signals) for (char v : string("01")) {
528 int found = -1;
529 for (size_t r = 0; r<rows.size(); r++) if (c.evaluate(rows[r], s, v) != expected[r]) {
530 found = int(r);
531 break;
532 }
533 if (found<0) undetected.push_back(s + "/" + v);
534 else tests.insert(found);
535 }
536 int total = int(2*c.signals.size());
537 emit("faults " + to_string(total) + " detected " + to_string(total-undetected.size()));
538 for (const string& f : undetected) emit("undetected " + f);
539 vector<string> text = {"tests"};
540 for (int r : tests) text.push_back(rows[r]);
541 emit(join(text));
542 }
543 }
544 return;
545 }
546 Simulation& s = *sim;
547 if (cmd == "RESET") {
548 s.reset();
549 emit("reset");
550 } else if (cmd == "SET") {
551 int t = time(a[0]);
552 input(a[1]);
553 char v = value(a[2]);
554 s.schedule(t, a[1], v);
555 emit("set " + a[1] + " " + a[2] + " at " + to_string(t));
556 } else if (cmd == "APPLY") {
557 int t = time(a[0]);
558 const string& b = a[1];
559 if (b.size() != s.c->inputs.size() || b.find_first_not_of("01X") != string::npos) fail("bad vector " + b);
560 for (size_t i = 0; i<b.size(); i++) s.schedule(t, s.c->inputs[i], b[i]);
561 emit("apply " + b + " at " + to_string(t));
562 } else if (cmd == "PULSE" || cmd == "CLOCK") {
563 input(a[0]);
564 int t = time(a[1]), width = number(a[2], true);
565 if (cmd == "PULSE") {
566 s.schedule(t, a[0], '1');
567 s.schedule(t + width, a[0], '0');
568 emit("pulse " + a[0] + " " + to_string(t) + " " + to_string(t + width));
569 } else {
570 int count = number(a[3], true);
571 for (int i = 0; i<count; i++) s.schedule(t + i*width, a[0], char('0' + i%2));
572 emit("clock " + a[0] + " events " + to_string(count) + " last " + to_string(t + (count-1)*width));
573 }
574 } else if (cmd == "RUN") {
575 int t = number(a[0]);
576 if (t <= s.time) fail("time " + to_string(t) + " is not after " + to_string(s.time));
577 s.run(t);
578 } else if (cmd == "WATCH" || cmd == "UNWATCH") {
579 set<string> seen;
580 for (const string& x : a) {
581 signal(x);
582 bool watched = has(s.watch, x);
583 if (cmd == "WATCH" ? (watched || seen.count(x)) : (!watched || seen.count(x))) fail((cmd == "WATCH" ? "already watched " : "not watched ") + x);
584 seen.insert(x);
585 }
586 if (cmd == "WATCH") s.watch.insert(s.watch.end(), a.begin(), a.end());
587 else s.watch.erase(remove_if(s.watch.begin(), s.watch.end(), [&](const string& x) {
588 return seen.count(x);
589 }), s.watch.end());
590 emit("watching " + (s.watch.empty() ? "nothing" : join(s.watch)));
591 } else if (cmd == "SHOW") {
592 for (const string& x : a) signal(x);
593 vector<string> text = s.describe(a);
594 if (a.empty()) {
595 text = s.describe(s.c->inputs);
596 text.push_back("|");
597 vector<string> outs = s.describe(s.c->outputs);
598 text.insert(text.end(), outs.begin(), outs.end());
599 }
600 emit(join(text));
601 } else if (cmd == "CHECK") {
602 signal(a[0]);
603 char v = value(a[1]);
604 emit("check " + a[0] + "=" + a[1] + " " + (s.values[a[0]] == v ? "ok" : "failed (" + string(1, s.values[a[0]]) + ")"));
605 } else if (cmd == "PROBE") {
606 string x = a[0];
607 signal(x);
608 string text = "input";
609 if (s.c->driver.count(x)) {
610 const Gate& g = s.c->gates[s.c->driver[x]];
611 text = g.name + " " + g.kind + " " + join(s.describe(g.inputs)) + " delay " + to_string(g.delay);
612 }
613 emit(x + "=" + s.values[x] + " driven by " + text);
614 vector<string> names = {"fanout", to_string(s.c->readers[x].size())};
615 for (int i : s.c->readers[x]) names.push_back(s.c->gates[i].name);
616 emit(join(names));
617 } else if (cmd == "WAVE") {
618 int start = number(a[0]), end = number(a[1]);
619 if (!(start<end && end <= s.time)) fail("bad range " + to_string(start) + " " + to_string(end));
620 if (s.watch.empty()) fail("nothing watched");
621 for (const string& x : s.watch) {
622 const auto& h = s.history[x];
623 size_t i = 0;
624 char v = 'X';
625 string wave;
626 for (int t = start; t<end; t++) {
627 while (i<h.size() && h[i].first <= t) v = h[i++].second;
628 wave += v;
629 }
630 emit(x + " " + wave);
631 }
632 } else if (cmd == "HISTORY") {
633 string x = a[0];
634 signal(x);
635 vector<string> text = {"history", x, "changes", to_string(s.history[x].size())};
636 for (auto[t, v] : s.history[x]) text.push_back(to_string(t) + ":" + v);
637 emit(join(text));
638 } else if (cmd == "ACTIVITY") {
639 int k = number(a[0], true);
640 vector<string> ranked = s.c->signals;
641 sort(ranked.begin(), ranked.end(), [&](const string& x, const string& y) {
642 return s.history[x].size() != s.history[y].size() ? s.history[x].size()> s.history[y].size() : x<y;
643 });
644 for (int i = 0; i<k && i<int(ranked.size()); i++) emit(ranked[i] + " " + to_string(s.history[ranked[i]].size()));
645 } else if (cmd == "STATUS") {
646 emit("circuit " + s.c->name + " time " + to_string(s.time) + " pending " + to_string(s.queue.size()) + " events " + to_string(s.processed) + " halted " + (s.halted ? "yes" : "no"));
647 map<char, int> n;
648 for (const auto&[x, v] : s.values) n[v]++;
649 emit("values 0:" + to_string(n['0']) + " 1:" + to_string(n['1']) + " X:" + to_string(n['X']));
650 }
651 } catch (const runtime_error& e) {
652 emit("error: line " + to_string(line) + ": " + e.what());
653 }
654 }
655};
656int main() {
657 ios::sync_with_stdio(false);
658 cin.tie(nullptr);
659 vector<string> lines;
660 string line;
661 while (getline(cin, line)) lines.push_back(line);
662 Simulator s;
663 for (size_t i = 0; i<lines.size() && lines[i] != "END"; i++) {
664 vector<string> w = words(lines[i]);
665 if (w[0] == "MODULE") {
666 size_t end = i + 1;
667 while (lines[end] != "ENDMODULE") end++;
668 s.define(int(i) + 1, vector<string>(lines.begin() + i, lines.begin() + end));
669 i = end;
670 } else s.execute(int(i) + 1, w);
671 }
