72logic <- function(k, a) {
73 y <- if (k %in% c('AND', 'NAND')) {
74 if ('0' %in% a) {
75 '0'
76 } else if ('X' %in% a) {
77 'X'
78 } else {
79 '1'
80 }
81 } else if (k %in% c('OR', 'NOR')) {
82 if ('1' %in% a) {
83 '1'
84 } else if ('X' %in% a) {
85 'X'
86 } else {
87 '0'
88 }
89 } else if (k %in% c('XOR', 'XNOR')) {
90 if ('X' %in% a) 'X' else as.character(sum(a == '1') %% 2)
91 } else if (k == 'MUX2') {
92 if (a[1] == 'X') {
93 if (a[2] == a[3]) a[2] else 'X'
94 } else {
95 a[if (a[1] == '0') 2 else 3]
96 }
97 } else {
98 a[1]
99 }
100 if (k %in% c('NAND', 'NOR', 'XNOR', 'NOT')) {
101 switch(y, '0' = '1', '1' = '0', 'X' = 'X')
102 } else {
103 y
104 }

106circuit <- function(m) {
107 gates <- list()
108 expand <- function(m, prefix, ports) {
109 actual <- function(s) {
110 if (s %in% names(ports)) ports[[s]] else paste0(prefix, s)
111 }
112 for (x in m$body) {
113 if (!is.null(x$kind)) {
114 g <- x
115 g$name <- paste0(prefix, g$name)
116 g$inputs <- vapply(g$inputs, actual, character(1))
117 g$output <- actual(g$output)
118 gates[[length(gates) + 1]] <<- g
119 } else {
120 mapping <- vapply(
121 x$connections,
122 function(v) actual(v$signal),
123 character(1)
124 )
125 names(mapping) <- vapply(
126 x$connections,
127 function(v) v$port,
128 character(1)
129 )
130 expand(x$module, paste0(prefix, x$name, '.'), mapping)
131 }
132 }
133 }
134 expand(m, '', character())
135 signals <- c(m$inputs, vapply(gates, function(g) g$output, character(1)))
136 driver <- setNames(
137 seq_along(gates),
138 vapply(gates, function(g) g$output, character(1))
139 )
140 readers <- setNames(rep(list(integer()), length(signals)), signals)
141 flops <- which(vapply(gates, function(g) g$kind == 'DFF', logical(1)))
142 for (i in seq_along(gates)) {
143 for (s in unique(gates[[i]]$inputs)) {
144 readers[[s]] <- c(readers[[s]], i)
145 }
146 }
147 waiting <- integer(length(gates))
148 ready <- integer()
149 needed <- 0
150 for (i in seq_along(gates)) {
151 if (gates[[i]]$kind != 'DFF') {
152 needed <- needed + 1
153 for (s in unique(gates[[i]]$inputs)) {
154 if (s %in% names(driver) && gates[[driver[[s]]]]$kind != 'DFF') {
155 waiting[i] <- waiting[i] + 1
156 }
157 }
158 if (waiting[i] == 0) ready <- c(ready, i)
159 }
160 }
161 order <- integer()
162 while (length(ready)) {
163 i <- tail(ready, 1)
164 ready <- head(ready, -1)
165 order <- c(order, i)
166 for (r in readers[[gates[[i]]$output]]) {
167 if (gates[[r]]$kind != 'DFF') {
168 waiting[r] <- waiting[r] - 1
169 if (waiting[r] == 0) ready <- c(ready, r)
170 }
171 }
172 }
173 list(
174 name = m$name,
175 inputs = m$inputs,
176 outputs = m$outputs,
177 gates = gates,
178 signals = signals,
179 driver = driver,
180 readers = readers,
181 flops = flops,
182 order = order,
183 loop = length(order) != needed
184 )
185}

195evaluate <- function(c, b, stuck = '', v = 'X') {
196 values <- setNames(strsplit(b, '')[[1]], c$inputs)
197 if (stuck %in% c$inputs) {
198 values[[stuck]] <- v
199 }
200 for (i in c$order) {
201 g <- c$gates[[i]]
202 values[[g$output]] <- if (g$output == stuck) {
203 v
204 } else {
205 logic(g$kind, unname(values[g$inputs]))
206 }
207 }
208 paste0(values[c$outputs], collapse = '')

210arrivals <- function(c, period) {
211 late <- early <- numeric()
212 prev <- character()
213 if (period) {
214 for (i in c$flops) {
215 g <- c$gates[[i]]
216 late[[g$output]] <- early[[g$output]] <- g$delay
217 }
218 } else {
219 for (s in c$inputs) {
220 late[[s]] <- early[[s]] <- 0
221 }
222 }
223 for (i in c$order) {
224 g <- c$gates[[i]]
225 a <- g$inputs[g$inputs %in% names(late)]
226 if (!length(a)) {
227 next
228 }
229 best <- a[which.max(late[a])]
230 late[[g$output]] <- late[[best]] + g$delay
231 early[[g$output]] <- min(early[a]) + g$delay
232 prev[[g$output]] <- best
233 }
234 list(late = late, early = early, previous = prev)
235}

253schedule <- function(s, t, x, v) {
254 e <- list(time = t, id = s$sequence, signal = x, value = v)
255 s$sequence <- s$sequence + 1
256 i <- length(s$queue) + 1
257 while (i > 1) {
258 p <- i %/% 2
259 if (!before(e, s$queue[[p]])) {
260 break
261 }
262 s$queue[[i]] <- s$queue[[p]]
263 i <- p
264 }
265 s$queue[[i]] <- e
266}
267pop_event <- function(s) {
268 result <- s$queue[[1]]
269 last <- s$queue[[length(s$queue)]]
270 s$queue <- head(s$queue, -1)
271 if (length(s$queue)) {
272 i <- 1
273 while (2 * i <= length(s$queue)) {
274 j <- 2 * i
275 if (j + 1 <= length(s$queue) && before(s$queue[[j + 1]], s$queue[[j]])) {
276 j <- j + 1
277 }
278 if (!before(s$queue[[j]], last)) {
279 break
280 }
281 s$queue[[i]] <- s$queue[[j]]
282 i <- j
283 }
284 s$queue[[i]] <- last
285 }
286 result

294run <- function(s, end) {
295 total <- 0
296 glitches <- 0
297 while (length(s$queue) && s$queue[[1]]$time < end) {
298 t <- s$queue[[1]]$time
299 start <- s$values
300 changed <- setNames(rep(FALSE, length(start)), names(start))
301 count <- 0
302 while (length(s$queue) && s$queue[[1]]$time == t) {
303 if (count == 1000) {
304 emit(paste('halted: oscillation at time', num(t)))
305 s$queue <- list()
306 s$time <- t
307 s$halted <- TRUE
308 return(invisible(NULL))
309 }
310 e <- pop_event(s)
311 count <- count + 1
312 s$processed <- s$processed + 1
313 old <- s$values[[e$signal]]
314 if (old == e$value) {
315 next
316 }
317 changed[[e$signal]] <- TRUE
318 s$values[[e$signal]] <- e$value
319 for (i in s$c$readers[[e$signal]]) {
320 g <- s$c$gates[[i]]
321 if (g$kind != 'DFF') {
322 schedule(
323 s,
324 t + g$delay,
325 g$output,
326 logic(g$kind, unname(s$values[g$inputs]))
327 )
328 } else if (e$signal == g$inputs[1] && old == '0' && e$value == '1') {
329 schedule(s, t + g$delay, g$output, s$values[[g$inputs[2]]])
330 }
331 }
332 }
333 total <- total + count
334 for (x in names(start)[changed]) {
335 if (s$values[[x]] == start[[x]]) {
336 glitches <- glitches + 1
337 } else {
338 s$history[[x]][[length(s$history[[x]]) + 1]] <- list(
339 time = t,
340 value = s$values[[x]]
341 )
342 }
343 }
344 for (x in s$watch) {
345 if (s$values[[x]] != start[[x]]) emit(paste(num(t), x, s$values[[x]]))
346 }
347 }
348 s$time <- end
349 emit(paste(
350 'run',
351 num(end),
352 'events',
353 num(total),
354 'glitches',
355 num(glitches),
356 'pending',
357 num(length(s$queue))
358 ))
359}

385define <- function(s, header, lines) {
386 line <- header
387 tryCatch(
388 {
389 w <- strsplit(lines[1], ' ', fixed = TRUE)[[1]]
390 if (
391 length(w) < 5 ||
392 w[3] != 'IN' ||
393 sum(w[-(1:3)] == 'OUT') != 1 ||
394 tail(w, 1) == 'OUT'
395 ) {
396 fail('bad module header')
397 }
398 name <- w[2]
399 if (nchar(name) > 16 || !grepl('^[A-Z][A-Z0-9_]*$', name)) {
400 fail(paste('bad module name', name))
401 }
402 if (name %in% names(s$modules)) {
403 fail(paste('duplicate module', name))
404 }
405 split <- which(w == 'OUT' & seq_along(w) > 3)
406 ins <- if (split > 4) w[4:(split - 1)] else character()
407 outs <- w[(split + 1):length(w)]
408 seen <- character()
409 for (p in c(ins, outs)) {
410 if (!name_ok(p)) {
411 fail(paste('bad name', p))
412 }
413 if (p %in% seen) {
414 fail(paste('duplicate port', p))
415 }
416 seen <- c(seen, p)
417 }
418 driven <- ins
419 used <- character()
420 body <- reads <- list()
421 gates <- 0
422 for (i in seq_len(length(lines) - 1)) {
423 line <- header + i
424 a <- strsplit(lines[i + 1], ' ', fixed = TRUE)[[1]]
425 kind <- a[1]
426 if (kind %in% TYPES) {
427 if (
428 length(a) < 5 || a[length(a) - 2] != '->' || sum(a == '->') != 1
429 ) {
430 fail('bad gate syntax')
431 }
432 n <- a[2]
433 if (!name_ok(n)) {
434 fail(paste('bad name', n))
435 }
436 inputs <- if (length(a) > 5) a[3:(length(a) - 3)] else character()
437 output <- a[length(a) - 1]
438 delay <- tail(a, 1)
439 lo <- if (kind %in% c('NOT', 'BUF')) {
440 1
441 } else if (kind == 'MUX2') {
442 3
443 } else {
444 2
445 }
446 hi <- if (match(kind, TYPES) <= 6) 8 else lo
447 if (length(inputs) < lo || length(inputs) > hi) {
448 fail(paste('bad input count', length(inputs), 'for', kind))
449 }
450 for (x in c(inputs, output)) {
451 if (!name_ok(x)) fail(paste('bad name', x))
452 }
453 if (!number_ok(delay) || as.numeric(delay) > 1000) {
454 fail(paste('bad delay', delay))
455 }
456 if (n %in% used) {
457 fail(paste('duplicate name', n))
458 }
459 if (output %in% driven) {
460 fail(paste('multiple drivers for', output))
461 }
462 used <- c(used, n)
463 driven <- c(driven, output)
464 body[[length(body) + 1]] <- list(
465 name = n,
466 kind = kind,
467 inputs = inputs,
468 output = output,
469 delay = as.numeric(delay)
470 )
471 gates <- gates + 1
472 for (x in inputs) {
473 reads[[length(reads) + 1]] <- list(line = line, signal = x)
474 }
475 } else if (kind == 'INST') {
476 if (length(a) < 3) {
477 fail('bad instance syntax')
478 }
479 n <- a[2]
480 if (!name_ok(n)) {
481 fail(paste('bad name', n))
482 }
483 m <- lookup(s, a[3])
484 connections <- list()
485 ports <- character()
486 for (word in a[-(1:3)]) {
487 p <- regmatches(word, regexec('^([^=]+)=([^=]+)$', word))[[1]]
488 if (!length(p)) {
489 fail(paste('bad connection', word))
490 }
491 port <- p[2]
492 x <- p[3]
493 if (!(port %in% c(m$inputs, m$outputs))) {
494 fail(paste('unknown port', port))
495 }
496 if (port %in% ports) {
497 fail(paste('duplicate connection', port))
498 }
499 if (!name_ok(x)) {
500 fail(paste('bad name', x))
501 }
502 ports <- c(ports, port)
503 connections[[length(connections) + 1]] <- list(
504 port = port,
505 signal = x
506 )
507 }
508 for (p in m$inputs) {
509 if (!(p %in% ports)) fail(paste('unconnected input', p))
510 }
511 if (n %in% used) {
512 fail(paste('duplicate name', n))
513 }
514 for (v in connections) {
515 if (v$port %in% m$outputs) {
516 if (v$signal %in% driven) {
517 fail(paste('multiple drivers for', v$signal))
518 }
519 driven <- c(driven, v$signal)
520 }
521 }
522 used <- c(used, n)
523 body[[length(body) + 1]] <- list(
524 name = n,
525 module = m,
526 connections = connections
527 )
528 for (v in connections) {
529 if (v$port %in% m$inputs) {
530 reads[[length(reads) + 1]] <- list(line = line, signal = v$signal)
531 }
532 }
533 } else {
534 fail(paste('unknown statement', kind))
535 }
536 }
537 for (x in outs) {
538 reads[[length(reads) + 1]] <- list(line = header, signal = x)
539 }
540 for (r in reads) {
541 if (!(r$signal %in% driven)) {
542 line <- r$line
543 fail(paste('undriven signal', r$signal))
544 }
545 }
546 s$modules[[name]] <- list(
547 name = name,
548 inputs = ins,
549 outputs = outs,
550 body = body
551 )
552 emit(paste(
553 'defined',
554 name,
555 'inputs',
556 length(ins),
557 'outputs',
558 length(outs),
559 'gates',
560 gates,
561 'instances',
562 length(body) - gates
563 ))
564 },
565 error = function(e) {
566 emit(paste0('error: line ', line, ': ', conditionMessage(e)))
567 }
568 )
569}

578execute <- function(s, line, w) {
579 tryCatch(
580 {
581 cmd <- w[1]
582 a <- w[-1]
583 if (!(cmd %in% names(COUNTS))) {
584 fail(paste('unknown command', cmd))
585 }
586 n <- COUNTS[[cmd]]
587 if (
588 length(a) < n ||
589 (!(cmd %in% c('WATCH', 'UNWATCH', 'SHOW')) && length(a) != n)
590 ) {
591 fail('wrong argument count')
592 }
593 if (cmd %in% SIM && is.null(s$simulation)) {
594 fail('no circuit')
595 }
596 sim <- s$simulation
597 if (cmd %in% c('SET', 'APPLY', 'PULSE', 'CLOCK', 'RUN') && sim$halted) {
598 fail('simulation halted')
599 }
600 if (cmd == 'MODULES') {
601 emit(paste('modules', length(s$modules)))
602 for (m in s$modules) {
603 emit(paste(
604 m$name,
605 'inputs',
606 length(m$inputs),
607 'outputs',
608 length(m$outputs),
609 'gates',
610 length(circuit(m)$gates)
611 ))
612 }
613 return(invisible(NULL))
614 }
615 if (
616 cmd %in%
617 c(
618 'BUILD',
619 'NETLIST',
620 'HIERARCHY',
621 'STATS',
622 'TRUTHTABLE',
623 'EQUIV',
624 'FAULTS',
625 'CRITICAL',
626 'PERIOD'
627 )
628 ) {
629 m <- lookup(s, a[1])
630 if (cmd == 'EQUIV') {
631 m2 <- lookup(s, a[2])
632 }
633 c <- circuit(m)
634 if (cmd == 'BUILD') {
635 s$simulation <- new_simulation(c)
636 emit(paste(
637 'built',
638 c$name,
639 'signals',
640 length(c$signals),
641 'gates',
642 length(c$gates)
643 ))
644 } else if (cmd == 'NETLIST') {
645 emit(paste(
646 'netlist',
647 c$name,
648 'gates',
649 length(c$gates),
650 'signals',
651 length(c$signals)
652 ))
653 for (g in c$gates) {
654 emit(paste(
655 c(g$name, g$kind, g$inputs, '->', g$output, g$delay),
656 collapse = ' '
657 ))
658 }
659 } else if (cmd == 'HIERARCHY') {
660 emit(c$name)
661 hierarchy(m, 1)
662 } else if (cmd == 'STATS') {
663 levels <- setNames(rep(0, length(c$signals)), c$signals)
664 depth <- 0
665 for (i in c$order) {
666 g <- c$gates[[i]]
667 levels[[g$output]] <- max(levels[g$inputs]) + 1
668 depth <- max(depth, levels[[g$output]])
669 }
670 emit(paste(
671 'stats',
672 c$name,
673 'gates',
674 length(c$gates),
675 'signals',
676 length(c$signals),
677 'depth',
678 if (c$loop) '-' else depth
679 ))
680 row <- 'types'
681 for (t in TYPES) {
682 n <- sum(vapply(c$gates, function(g) g$kind == t, logical(1)))
683 if (n) row <- c(row, t, n)
684 }
685 emit(paste(row, collapse = ' '))
686 fanout <- vapply(c$readers, length, integer(1))
687 best <- c$signals[which.max(fanout[c$signals])]
688 emit(paste('max fanout', best, fanout[[best]]))
689 } else if (cmd %in% c('CRITICAL', 'PERIOD')) {
690 period <- cmd == 'PERIOD'
691 if (period && !length(c$flops)) {
692 fail(paste(c$name, 'has no flip-flops'))
693 }
694 if (c$loop) {
695 fail(paste(c$name, 'has a combinational loop'))
696 }
697 arr <- arrivals(c, period)
698 ends <- if (period) {
699 vapply(c$flops, function(i) c$gates[[i]]$inputs[2], character(1))
700 } else {
701 c$outputs
702 }
703 heading <- if (period) {
704 paste('flip-flops', length(c$flops), 'period')
705 } else {
706 'critical'
707 }
708 best <- NULL
709 for (x in ends) {
710 if (!period) {
711 emit(
712 if (x %in% names(arr$late)) {
713 paste(
714 x,
715 'min',
716 num(arr$early[[x]]),
717 'max',
718 num(arr$late[[x]])
719 )
720 } else {
721 paste(x, 'unreachable')
722 }
723 )
724 }
725 if (
726 x %in%
727 names(arr$late) &&
728 (is.null(best) || arr$late[[x]] > arr$late[[best]])
729 ) {
730 best <- x
731 }
732 }
733 if (is.null(best)) {
734 emit(paste(heading, 'none'))
735 } else {
736 path <- best
737 x <- best
738 while (x %in% names(arr$previous)) {
739 x <- arr$previous[[x]]
740 path <- c(x, path)
741 }
742 emit(paste0(
743 heading,
744 ' ',
745 num(arr$late[[best]]),
746 ': ',
747 paste(path, collapse = ' -> ')
748 ))
749 }
750 } else {
751 combinational(c)
752 rows <- vapply(
753 0:(2^length(c$inputs) - 1),
754 function(r) bits(r, length(c$inputs)),
755 character(1)
756 )
757 expected <- vapply(rows, function(b) evaluate(c, b), character(1))
758 if (cmd == 'TRUTHTABLE') {
759 emit(paste(c(c$inputs, '|', c$outputs), collapse = ' '))
760 for (r in seq_along(rows)) {
761 emit(paste(
762 c(
763 strsplit(rows[r], '')[[1]],
764 '|',
765 strsplit(expected[r], '')[[1]]
766 ),
767 collapse = ' '
768 ))
769 }
770 } else if (cmd == 'EQUIV') {
771 d <- circuit(m2)
772 combinational(d)
773 if (
774 length(c$inputs) != length(d$inputs) ||
775 length(c$outputs) != length(d$outputs)
776 ) {
777 fail('port mismatch')
778 }
779 different <- FALSE
780 for (r in seq_along(rows)) {
781 v <- evaluate(d, rows[r])
782 if (v != expected[r]) {
783 emit(paste0(
784 'not equivalent at ',
785 rows[r],
786 ': ',
787 expected[r],
788 ' vs ',
789 v
790 ))
791 different <- TRUE
792 break
793 }
