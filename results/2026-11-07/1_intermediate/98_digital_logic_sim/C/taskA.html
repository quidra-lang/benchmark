115static int name_valid(const char *word, int upper) {
116 size_t n = strlen(word);
117 if (!n || n > 16 || word[0] < (upper ? 'A' : 'a') || word[0] > (upper ? 'Z' : 'z'))
118 return 0;
119 for (size_t i = 1; i < n; ++i)
120 if (!((word[i] >= (upper ? 'A' : 'a') && word[i] <= (upper ? 'Z' : 'z')) ||
121 isdigit((unsigned char)word[i]) || word[i] == '_'))
122 return 0;
123 return 1;
124}

126static int is_number(const char *word) {
127 size_t n = strlen(word);
128 if (!n || n > 9 || (n > 1 && word[0] == '0'))
129 return 0;
130 for (size_t i = 0; i < n; ++i)
131 if (word[i] < '0' || word[i] > '9')
132 return 0;
133 return 1;
134}

155static void prepare_circuit(Circuit *c) {
156 c->signal_count = c->input_count + c->gate_count;
157 for (int i = 0; i < c->input_count; ++i)
158 c->signals[i] = c->inputs[i];
159 for (int i = 0; i < c->gate_count; ++i)
160 c->signals[c->input_count + i] = c->gates[i].output;
161 for (int s = 0; s < c->signal_count; ++s)
162 c->driver[s] = s < c->input_count ? -1 : s - c->input_count;
163 for (int g = 0; g < c->gate_count; ++g) {
164 c->out[g] = c->input_count + g;
165 for (int i = 0; i < c->gates[g].count; ++i)
166 c->in[g][i] = index_of(c->signals, c->signal_count, c->gates[g].inputs[i]);
167 if (c->gates[g].type == 9)
168 ++c->dff_count;
169 }
170 for (int i = 0; i < c->output_count; ++i)
171 c->output_indices[i] = index_of(c->signals, c->signal_count, c->outputs[i]);
172 for (int s = 0; s < c->signal_count; ++s) {
173 for (int g = 0; g < c->gate_count; ++g)
174 if (reads(c, g, s))
175 ++c->reader_count[s];
176 c->readers[s] = malloc((size_t)(c->reader_count[s] + 1) * sizeof(int));
177 int n = 0;
178 for (int g = 0; g < c->gate_count; ++g)
179 if (reads(c, g, s))
180 c->readers[s][n++] = g;
181 }
182 int done[500] = {0}, progress = 1;
183 while (progress) {
184 progress = 0;
185 for (int g = 0; g < c->gate_count; ++g)
186 if (!done[g] && c->gates[g].type != 9) {
187 int ready = 1;
188 for (int i = 0; i < c->gates[g].count; ++i) {
189 int source = c->driver[c->in[g][i]];
190 if (source >= 0 && c->gates[source].type != 9 && !done[source])
191 ready = 0;
192 }
193 if (ready) {
194 done[g] = 1;
195 c->order[c->order_count++] = g;
196 progress = 1;
197 }
198 }
199 }
200}

209static Circuit *flatten(Module *m) {
210 Circuit *c = calloc(1, sizeof *c);
211 c->input_count = m->input_count;
212 c->output_count = m->output_count;
213 memcpy(c->inputs, m->inputs, sizeof c->inputs);
214 memcpy(c->outputs, m->outputs, sizeof c->outputs);
215 for (int i = 0; i < m->count; ++i) {
216 Body *b = &m->body[i];
217 if (!b->instance)
218 c->gates[c->gate_count++] = b->gate;
219 else {
220 Circuit *child = modules[b->module].circuit;
221 for (int j = 0; j < child->gate_count; ++j) {
222 Gate g = child->gates[j];
223 g.name = join(b->name, ".", g.name);
224 g.output = mapped(b, g.output);
225 for (int k = 0; k < g.count; ++k)
226 g.inputs[k] = mapped(b, g.inputs[k]);
227 c->gates[c->gate_count++] = g;
228 }
229 }
230 }
231 prepare_circuit(c);
232 return c;
233}

235static void define_module(Line *lines, int first, int end) {
236 Line *header = &lines[first];
237 char **w = header->words;
238 int n = header->count, out = -1, outs = 0;
239 for (int i = 3; i < n; ++i)
240 if (!strcmp(w[i], "OUT")) {
241 out = i;
242 ++outs;
243 }
244 if (n < 5 || strcmp(w[2], "IN") || outs != 1 || out == n - 1)
245 error("bad module header");
246 if (!name_valid(w[1], 1))
247 error("bad module name %s", w[1]);
248 if (module_index(w[1]) >= 0)
249 error("duplicate module %s", w[1]);
250 Module m = {0};
251 m.name = w[1];
252 m.input_count = out - 3;
253 m.output_count = n - out - 1;
254 char *ports[100], *driven[50000], *used[1000];
255 int port_count = 0, driven_count = 0, used_count = 0;
256 for (int i = 3; i < n; ++i)
257 if (i != out) {
258 if (!name_valid(w[i], 0))
259 error("bad name %s", w[i]);
260 if (index_of(ports, port_count, w[i]) >= 0)
261 error("duplicate port %s", w[i]);
262 ports[port_count++] = w[i];
263 if (i < out) {
264 m.inputs[i - 3] = w[i];
265 driven[driven_count++] = w[i];
266 } else
267 m.outputs[i - out - 1] = w[i];
268 }
269 m.body = calloc((size_t)(end - first), sizeof *m.body);
270 for (int l = first + 1; l < end; ++l) {
271 error_line = lines[l].number;
272 w = lines[l].words;
273 n = lines[l].count;
274 Body *b = &m.body[m.count++];
275 b->line = error_line;
276 int type = gate_type(w[0]);
277 if (type >= 0) {
278 int arrows = 0;
279 for (int i = 0; i < n; ++i)
280 arrows += !strcmp(w[i], "->");
281 if (n < 5 || strcmp(w[n - 3], "->") || arrows != 1)
282 error("bad gate syntax");
283 if (!name_valid(w[1], 0))
284 error("bad name %s", w[1]);
285 int count = n - 5, minimum = type < 6 ? 2 : type == 8 ? 3 : type == 9 ? 2 : 1;
286 int maximum = type < 6 ? 8 : minimum;
287 if (count < minimum || count > maximum)
288 error("bad input count %d for %s", count, w[0]);
289 for (int i = 2; i < n - 3; ++i)
290 if (!name_valid(w[i], 0))
291 error("bad name %s", w[i]);
292 if (!name_valid(w[n - 2], 0))
293 error("bad name %s", w[n - 2]);
294 if (!is_number(w[n - 1]) || strtoll(w[n - 1], NULL, 10) > 1000)
295 error("bad delay %s", w[n - 1]);
296 if (index_of(used, used_count, w[1]) >= 0)
297 error("duplicate name %s", w[1]);
298 if (index_of(driven, driven_count, w[n - 2]) >= 0)
299 error("multiple drivers for %s", w[n - 2]);
300 b->name = w[1];
301 b->gate = (Gate){.name = w[1],
302 .output = w[n - 2],
303 .type = type,
304 .count = count,
305 .delay = atoi(w[n - 1])};
306 for (int i = 0; i < count; ++i)
307 b->gate.inputs[i] = w[i + 2];
308 driven[driven_count++] = b->gate.output;
309 ++m.gates;
310 } else if (!strcmp(w[0], "INST")) {
311 if (n < 3)
312 error("bad instance syntax");
313 if (!name_valid(w[1], 0))
314 error("bad name %s", w[1]);
315 int child = module_index(w[2]);
316 if (child < 0)
317 error("unknown module %s", w[2]);
318 Module *target = &modules[child];
319 b->instance = 1;
320 b->name = w[1];
321 b->module = child;
322 for (int i = 3; i < n; ++i) {
323 char *equal = strchr(w[i], '=');
324 if (!equal || equal == w[i] || !equal[1] || strchr(equal + 1, '='))
325 error("bad connection %s", w[i]);
326 char *port = copy(w[i]);
327 port[equal - w[i]] = 0;
328 char *signal = equal + 1;
329 if (index_of(target->inputs, target->input_count, port) < 0 &&
330 index_of(target->outputs, target->output_count, port) < 0)
331 error("unknown port %s", port);
332 for (int k = 0; k < b->count; ++k)
333 if (!strcmp(b->connections[k].port, port))
334 error("duplicate connection %s", port);
335 if (!name_valid(signal, 0))
336 error("bad name %s", signal);
337 b->connections[b->count++] = (Connection){port, signal};
338 }
339 for (int i = 0; i < target->input_count; ++i) {
340 int found = 0;
341 for (int j = 0; j < b->count; ++j)
342 if (!strcmp(target->inputs[i], b->connections[j].port))
343 found = 1;
344 if (!found)
345 error("unconnected input %s", target->inputs[i]);
346 }
347 if (index_of(used, used_count, b->name) >= 0)
348 error("duplicate name %s", b->name);
349 for (int j = 0; j < b->count; ++j)
350 if (index_of(target->outputs, target->output_count, b->connections[j].port) >= 0) {
351 char *s = b->connections[j].signal;
352 if (index_of(driven, driven_count, s) >= 0)
353 error("multiple drivers for %s", s);
354 driven[driven_count++] = s;
355 }
356 ++m.instances;
357 } else
358 error("unknown statement %s", w[0]);
359 used[used_count++] = b->name;
360 }
361 for (int i = 0; i < m.count; ++i) {
362 Body *b = &m.body[i];
363 error_line = b->line;
364 if (!b->instance) {
365 for (int j = 0; j < b->gate.count; ++j)
366 if (index_of(driven, driven_count, b->gate.inputs[j]) < 0)
367 error("undriven signal %s", b->gate.inputs[j]);
368 } else {
369 Module *target = &modules[b->module];
370 for (int j = 0; j < b->count; ++j)
371 if (index_of(target->inputs, target->input_count, b->connections[j].port) >= 0 &&
372 index_of(driven, driven_count, b->connections[j].signal) < 0)
373 error("undriven signal %s", b->connections[j].signal);
374 }
375 }
376 error_line = header->number;
377 for (int i = 0; i < m.output_count; ++i)
378 if (index_of(driven, driven_count, m.outputs[i]) < 0)
379 error("undriven signal %s", m.outputs[i]);
380 m.circuit = flatten(&m);
381 modules[module_count++] = m;
382 printf("defined %s inputs %d outputs %d gates %d instances %d\n", m.name, m.input_count,
383 m.output_count, m.gates, m.instances);
384}

386static char gate_value(Gate *g, const char *v) {
387 char result = 'X';
388 int zeros = 0, ones = 0, unknowns = 0;
389 for (int i = 0; i < g->count; ++i) {
390 zeros += v[i] == '0';
391 ones += v[i] == '1';
392 unknowns += v[i] == 'X';
393 }
394 switch (g->type) {
395 case 0:
396 case 2:
397 result = zeros ? '0' : unknowns ? 'X' : '1';
398 break;
399 case 1:
400 case 3:
401 result = ones ? '1' : unknowns ? 'X' : '0';
402 break;
403 case 4:
404 case 5:
405 result = unknowns ? 'X' : (char)('0' + ones % 2);
406 break;
407 case 6:
408 case 7:
409 result = v[0];
410 break;
411 case 8:
412 result = v[0] == '0' ? v[1] : v[0] == '1' ? v[2] : v[1] == v[2] ? v[1] : 'X';
413 break;
414 }
415 if ((g->type == 2 || g->type == 3 || g->type == 5 || g->type == 6) && result != 'X')
416 result = result == '0' ? '1' : '0';
417 return result;
418}

420static void bits(int row, int width, char *result) {
421 for (int i = 0; i < width; ++i)
422 result[i] = (char)('0' + ((row >> (width - i - 1)) & 1));
423 result[width] = 0;
424}

426static void evaluate(Circuit *c, int row, int fault, char stuck, char *result) {
427 char v[550], input[51];
428 bits(row, c->input_count, input);
429 for (int i = 0; i < c->input_count; ++i)
430 v[i] = i == fault ? stuck : input[i];
431 for (int i = 0; i < c->order_count; ++i) {
432 int g = c->order[i], output = c->out[g];
433 char args[8];
434 for (int j = 0; j < c->gates[g].count; ++j)
435 args[j] = v[c->in[g][j]];
436 v[output] = output == fault ? stuck : gate_value(&c->gates[g], args);
437 }
438 for (int i = 0; i < c->output_count; ++i)
439 result[i] = v[c->output_indices[i]];
440 result[c->output_count] = 0;
441}

473static void path(Circuit *c, int *previous, int endpoint) {
474 int reverse[550], n = 0;
475 for (int s = endpoint; s >= 0; s = previous[s])
476 reverse[n++] = s;
477 for (int i = n - 1; i >= 0; --i)
478 printf("%s%s", i == n - 1 ? "" : " -> ", c->signals[reverse[i]]);
479 putchar('\n');
480}

482static void timing(Circuit *c, const char *name, int period) {
483 if (period && !c->dff_count)
484 error("%s has no flip-flops", name);
485 if (has_loop(c))
486 error("%s has a combinational loop", name);
487 long long latest[550], earliest[550];
488 int previous[550];
489 for (int s = 0; s < c->signal_count; ++s) {
490 latest[s] = earliest[s] = -1;
491 previous[s] = -1;
492 }
493 if (!period)
494 for (int s = 0; s < c->input_count; ++s)
495 latest[s] = earliest[s] = 0;
496 if (period)
497 for (int g = 0; g < c->gate_count; ++g)
498 if (c->gates[g].type == 9)
499 latest[c->out[g]] = earliest[c->out[g]] = c->gates[g].delay;
500 for (int i = 0; i < c->order_count; ++i) {
501 int g = c->order[i], s = c->out[g];
502 for (int j = 0; j < c->gates[g].count; ++j) {
503 int source = c->in[g][j];
504 if (latest[source] < 0)
505 continue;
506 long long l = latest[source] + c->gates[g].delay,
507 e = earliest[source] + c->gates[g].delay;
508 if (l > latest[s]) {
509 latest[s] = l;
510 previous[s] = source;
511 }
512 if (earliest[s] < 0 || e < earliest[s])
513 earliest[s] = e;
514 }
515 }
516 int endpoint = -1;
517 if (period) {
518 for (int g = 0; g < c->gate_count; ++g)
519 if (c->gates[g].type == 9) {
520 int s = c->in[g][1];
521 if (latest[s] >= 0 && (endpoint < 0 || latest[s] > latest[endpoint]))
522 endpoint = s;
523 }
524 printf("flip-flops %d period ", c->dff_count);
525 if (endpoint < 0)
526 puts("none");
527 else {
528 printf("%lld: ", latest[endpoint]);
529 path(c, previous, endpoint);
530 }
531 } else {
532 for (int i = 0; i < c->output_count; ++i) {
533 int s = c->output_indices[i];
534 if (latest[s] < 0)
535 printf("%s unreachable\n", c->outputs[i]);
536 else {
537 printf("%s min %lld max %lld\n", c->outputs[i], earliest[s], latest[s]);
538 if (endpoint < 0 || latest[s] > latest[endpoint])
539 endpoint = s;
540 }
541 }
542 if (endpoint < 0)
543 puts("critical none");
544 else {
545 printf("critical %lld: ", latest[endpoint]);
546 path(c, previous, endpoint);
547 }
548 }
549}

551static void inspect(int command, char **a) {
552 if (command == 16) {
553 printf("modules %d\n", module_count);
554 for (int i = 0; i < module_count; ++i)
555 printf("%s inputs %d outputs %d gates %d\n", modules[i].name, modules[i].input_count,
556 modules[i].output_count, modules[i].circuit->gate_count);
557 return;
558 }
559 Circuit *c = get_circuit(a[0]);
560 if (command == 17) {
561 printf("netlist %s gates %d signals %d\n", a[0], c->gate_count, c->signal_count);
562 for (int i = 0; i < c->gate_count; ++i) {
563 Gate *g = &c->gates[i];
564 printf("%s %s", g->name, types[g->type]);
565 for (int j = 0; j < g->count; ++j)
566 printf(" %s", g->inputs[j]);
567 printf(" -> %s %d\n", g->output, g->delay);
568 }
569 } else if (command == 18) {
570 puts(a[0]);
571 hierarchy(module_index(a[0]), 1);
572 } else if (command == 19) {
573 int levels[550] = {0}, depth = 0, counts[10] = {0}, max_fanout = 0;
574 for (int g = 0; g < c->gate_count; ++g)
575 ++counts[c->gates[g].type];
576 for (int i = 0; i < c->order_count; ++i) {
577 int g = c->order[i], level = 0;
578 for (int j = 0; j < c->gates[g].count; ++j)
579 if (levels[c->in[g][j]] > level)
580 level = levels[c->in[g][j]];
581 levels[c->out[g]] = level + 1;
582 if (level + 1 > depth)
583 depth = level + 1;
584 }
585 for (int s = 1; s < c->signal_count; ++s)
586 if (c->reader_count[s] > c->reader_count[max_fanout])
587 max_fanout = s;
588 printf("stats %s gates %d signals %d depth ", a[0], c->gate_count, c->signal_count);
589 if (has_loop(c))
590 puts("-");
591 else
592 printf("%d\n", depth);
593 printf("types");
594 for (int i = 0; i < 10; ++i)
595 if (counts[i])
596 printf(" %s %d", types[i], counts[i]);
597 printf("\nmax fanout %s %d\n", c->signals[max_fanout], c->reader_count[max_fanout]);
598 } else if (command == 23 || command == 24)
599 timing(c, a[0], command == 24);
600 else if (command == 21) {
601 Circuit *other = get_circuit(a[1]);
602 check_combinational(c, a[0]);
603 check_combinational(other, a[1]);
604 if (c->input_count != other->input_count || c->output_count != other->output_count)
605 error("port mismatch");
606 char left[51], right[51], input[51];
607 for (int r = 0; r < (1 << c->input_count); ++r) {
608 evaluate(c, r, -1, 'X', left);
609 evaluate(other, r, -1, 'X', right);
610 if (strcmp(left, right)) {
611 bits(r, c->input_count, input);
612 printf("not equivalent at %s: %s vs %s\n", input, left, right);
613 return;
614 }
615 }
616 puts("equivalent");
617 } else {
618 check_combinational(c, a[0]);
619 char outputs[256][51], input[51];
620 int rows = 1 << c->input_count;
621 for (int r = 0; r < rows; ++r)
622 evaluate(c, r, -1, 'X', outputs[r]);
623 if (command == 20) {
624 for (int i = 0; i < c->input_count; ++i)
625 printf("%s ", c->inputs[i]);
626 putchar('|');
627 for (int i = 0; i < c->output_count; ++i)
628 printf(" %s", c->outputs[i]);
629 putchar('\n');
630 for (int r = 0; r < rows; ++r) {
631 bits(r, c->input_count, input);
632 for (int i = 0; i < c->input_count; ++i)
633 printf("%c ", input[i]);
634 putchar('|');
635 for (int i = 0; i < c->output_count; ++i)
636 printf(" %c", outputs[r][i]);
637 putchar('\n');
638 }
639 } else {
640 int found[550][2], tests[256] = {0}, detected = 0;
641 for (int s = 0; s < c->signal_count; ++s)
642 for (int v = 0; v < 2; ++v) {
643 found[s][v] = -1;
644 for (int r = 0; r < rows; ++r) {
645 char faulty[51];
646 evaluate(c, r, s, (char)('0' + v), faulty);
647 if (strcmp(faulty, outputs[r])) {
648 found[s][v] = r;
649 tests[r] = 1;
650 ++detected;
651 break;
652 }
653 }
654 }
655 printf("faults %d detected %d\n", 2 * c->signal_count, detected);
656 for (int s = 0; s < c->signal_count; ++s)
657 for (int v = 0; v < 2; ++v)
658 if (found[s][v] < 0)
659 printf("undetected %s/%d\n", c->signals[s], v);
660 printf("tests");
661 for (int r = 0; r < rows; ++r)
662 if (tests[r]) {
663 bits(r, c->input_count, input);
664 printf(" %s", input);
665 }
666 putchar('\n');
667 }
668 }
669}

671static int event_before(Event a, Event b) {
672 return a.time < b.time || (a.time == b.time && a.serial < b.serial);
673}

675static void schedule(long long time, int signal, char v) {
676 if (event_count == event_capacity) {
677 event_capacity = event_capacity ? 2 * event_capacity : 128;
678 Event *grown = realloc(events, event_capacity * sizeof *events);
679 if (!grown)
680 exit(1);
681 events = grown;
682 }
683 Event event = {time, serial++, signal, v};
684 size_t i = event_count++;
685 while (i && event_before(event, events[(i - 1) / 2])) {
686 events[i] = events[(i - 1) / 2];
687 i = (i - 1) / 2;
688 }
689 events[i] = event;
690}
691
692static Event pop_event(void) {
693 Event first = events[0], last = events[--event_count];
694 if (event_count) {
695 size_t i = 0;
696 while (2 * i + 1 < event_count) {
697 size_t child = 2 * i + 1;
698 if (child + 1 < event_count && event_before(events[child + 1], events[child]))
699 ++child;
700 if (!event_before(events[child], last))
701 break;
702 events[i] = events[child];
703 i = child;
704 }
705 events[i] = last;
706 }
707 return first;
708}

710static void record(int s, long long time) {
711 History *h = &history[s];
712 if (h->count == h->capacity) {
713 h->capacity = h->capacity ? h->capacity * 2 : 16;
714 Change *grown = realloc(h->items, h->capacity * sizeof *h->items);
715 if (!grown)
716 exit(1);
717 h->items = grown;
718 }
719 h->items[h->count++] = (Change){time, values[s]};
720}

722static void reset(void) {
723 now = processed = serial = 0;
724 event_count = 0;
725 halted = 0;
726 memset(values, 'X', sizeof values);
727 for (int s = 0; s < 550; ++s)
728 history[s].count = 0;
729}

731static void run(long long time) {
732 if (time <= now)
733 error("time %lld is not after %lld", time, now);
734 long long command_events = 0, glitches = 0;
735 Circuit *c = simulation;
736 while (event_count && events[0].time < time) {
737 long long step_time = events[0].time;
738 char before[550];
739 int changed[550] = {0}, step_count = 0;
740 memcpy(before, values, sizeof before);
741 while (event_count && events[0].time == step_time) {
742 if (step_count == 1000) {
743 printf("halted: oscillation at time %lld\n", step_time);
744 event_count = 0;
745 now = step_time;
746 halted = 1;
747 return;
748 }
749 Event e = pop_event();
750 ++step_count;
751 ++command_events;
752 ++processed;
753 int s = e.signal;
754 char old = values[s];
755 if (old == e.value)
756 continue;
757 values[s] = e.value;
758 changed[s] = 1;
759 for (int i = 0; i < c->reader_count[s]; ++i) {
760 int g = c->readers[s][i];
761 Gate *gate = &c->gates[g];
762 if (gate->type == 9) {
763 if (c->in[g][0] == s && old == '0' && e.value == '1')
764 schedule(step_time + gate->delay, c->out[g], values[c->in[g][1]]);
765 } else {
766 char args[8];
767 for (int j = 0; j < gate->count; ++j)
768 args[j] = values[c->in[g][j]];
769 schedule(step_time + gate->delay, c->out[g], gate_value(gate, args));
770 }
771 }
772 }
773 for (int s = 0; s < c->signal_count; ++s) {
774 if (values[s] != before[s])
775 record(s, step_time);
776 else if (changed[s])
777 ++glitches;
778 }
779 for (int i = 0; i < watch_count; ++i) {
780 int s = watched[i];
781 if (values[s] != before[s])
782 printf("%lld %s %c\n", step_time, c->signals[s], values[s]);
783 }
784 }
785 now = time;
786 printf("run %lld events %lld glitches %lld pending %zu\n", time, command_events, glitches,
787 event_count);
788}

792 input ? simulation->input_count : simulation->signal_count, name);
793 if (s < 0)
794 error(input ? "not an input %s" : "unknown signal %s", name);
795 return s;

814static int activity_compare(const void *a, const void *b) {
815 int x = *(const int *)a, y = *(const int *)b;
816 if (history[x].count != history[y].count)
817 return history[x].count > history[y].count ? -1 : 1;
818 return strcmp(simulation->signals[x], simulation->signals[y]);
819}

821static void execute(Line *line) {
822 const char *commands[] = {"BUILD", "RESET", "SET", "APPLY", "PULSE",
823 "CLOCK", "RUN", "WATCH", "UNWATCH", "SHOW",
824 "CHECK", "PROBE", "WAVE", "HISTORY", "ACTIVITY",
825 "STATUS", "MODULES", "NETLIST", "HIERARCHY", "STATS",
826 "TRUTHTABLE", "EQUIV", "FAULTS", "CRITICAL", "PERIOD"};
827 const int counts[] = {1, 0, 3, 2, 3, 4, 1, 1, 1, 0, 2, 1, 2,
828 1, 1, 0, 0, 1, 1, 1, 1, 2, 1, 1, 1};
829 int id = -1, n = line->count - 1;
830 char **a = line->words + 1;
831 for (int i = 0; i < 25; ++i)
832 if (!strcmp(line->words[0], commands[i]))
833 id = i;
834 if (id < 0)
835 error("unknown command %s", line->words[0]);
836 if ((id == 7 || id == 8 || id == 9) ? n < counts[id] : n != counts[id])
837 error("wrong argument count");
838 if (id >= 1 && id <= 15 && !simulation)
839 error("no circuit");
840 if (id >= 2 && id <= 6 && halted)
841 error("simulation halted");
842 if (id >= 16) {
843 inspect(id, a);
844 return;
845 }
846 if (id == 0) {
847 Circuit *c = get_circuit(a[0]);
848 simulation = c;
849 simulation_name = modules[module_index(a[0])].name;
850 watch_count = 0;
851 reset();
852 printf("built %s signals %d gates %d\n", a[0], c->signal_count, c->gate_count);
853 } else if (id == 1) {
854 reset();
855 puts("reset");
856 } else if (id == 2) {
857 long long t = stimulus_time(a[0]);
858 int s = signal_index(a[1], 1);
859 char v = value(a[2]);
860 schedule(t, s, v);
861 printf("set %s %c at %lld\n", a[1], v, t);
862 } else if (id == 3) {
863 long long t = stimulus_time(a[0]);
864 if ((int)strlen(a[1]) != simulation->input_count || strspn(a[1], "01X") != strlen(a[1]))
865 error("bad vector %s", a[1]);
866 for (int i = 0; i < simulation->input_count; ++i)
867 schedule(t, i, a[1][i]);
868 printf("apply %s at %lld\n", a[1], t);
869 } else if (id == 4) {
870 int s = signal_index(a[0], 1);
871 long long t = stimulus_time(a[1]), width = number(a[2], 1);
872 schedule(t, s, '1');
873 schedule(t + width, s, '0');
874 printf("pulse %s %lld %lld\n", a[0], t, t + width);
875 } else if (id == 5) {
876 int s = signal_index(a[0], 1);
877 long long start = stimulus_time(a[1]), half = number(a[2], 1), count = number(a[3], 1);
878 for (long long i = 0; i < count; ++i)
879 schedule(start + i * half, s, (char)('0' + i % 2));
880 printf("clock %s events %lld last %lld\n", a[0], count, start + (count - 1) * half);
881 } else if (id == 6)
882 run(number(a[0], 0));
883 else if (id == 7 || id == 8) {
884 int selected[64];
885 for (int i = 0; i < n; ++i) {
886 int s = signal_index(a[i], 0), in_watch = 0, duplicate = 0;
887 for (int j = 0; j < watch_count; ++j)
888 in_watch |= watched[j] == s;
889 for (int j = 0; j < i; ++j)
890 duplicate |= selected[j] == s;
891 if (id == 7 && (in_watch || duplicate))
892 error("already watched %s", a[i]);
893 if (id == 8 && (!in_watch || duplicate))
894 error("not watched %s", a[i]);
895 selected[i] = s;
896 }
897 if (id == 7)
898 for (int i = 0; i < n; ++i)
899 watched[watch_count++] = selected[i];
900 else {
901 int retained = 0;
902 for (int i = 0; i < watch_count; ++i) {
903 int remove = 0;
904 for (int j = 0; j < n; ++j)
905 remove |= watched[i] == selected[j];
906 if (!remove)
907 watched[retained++] = watched[i];
908 }
909 watch_count = retained;
910 }
911 show_watches();
912 } else if (id == 9) {
913 if (n) {
914 int selected[64];
915 for (int i = 0; i < n; ++i)
916 selected[i] = signal_index(a[i], 0);
917 for (int i = 0; i < n; ++i)
918 printf("%s%s=%c", i ? " " : "", a[i], values[selected[i]]);
919 } else {
920 for (int i = 0; i < simulation->input_count; ++i)
921 printf("%s=%c ", simulation->inputs[i], values[i]);
922 putchar('|');
923 for (int i = 0; i < simulation->output_count; ++i)
924 printf(" %s=%c", simulation->outputs[i], values[simulation->output_indices[i]]);
925 }
926 putchar('\n');
927 } else if (id == 10) {
928 int s = signal_index(a[0], 0);
929 char v = value(a[1]);
930 printf("check %s=%c ", a[0], v);
931 if (values[s] == v)
932 puts("ok");
933 else
934 printf("failed (%c)\n", values[s]);
935 } else if (id == 11) {
936 int s = signal_index(a[0], 0), g = simulation->driver[s];
937 printf("%s=%c driven by ", a[0], values[s]);
938 if (g < 0)
939 puts("input");
940 else {
941 Gate *gate = &simulation->gates[g];
942 printf("%s %s", gate->name, types[gate->type]);
943 for (int j = 0; j < gate->count; ++j)
944 printf(" %s=%c", gate->inputs[j], values[simulation->in[g][j]]);
945 printf(" delay %d\n", gate->delay);
946 }
947 printf("fanout %d", simulation->reader_count[s]);
948 for (int i = 0; i < simulation->reader_count[s]; ++i)
949 printf(" %s", simulation->gates[simulation->readers[s][i]].name);
950 putchar('\n');
951 } else if (id == 12) {
952 long long start = number(a[0], 0), end = number(a[1], 0);
953 if (!(start < end && end <= now))
954 error("bad range %lld %lld", start, end);
955 if (!watch_count)
956 error("nothing watched");
957 for (int i = 0; i < watch_count; ++i) {
958 int s = watched[i];
959 History *h = &history[s];
960 size_t next = 0;
961 char v = 'X';
962 printf("%s ", simulation->signals[s]);
963 for (long long t = start; t < end; ++t) {
964 while (next < h->count && h->items[next].time <= t)
965 v = h->items[next++].value;
966 putchar(v);
967 }
968 putchar('\n');
969 }
970 } else if (id == 13) {
971 int s = signal_index(a[0], 0);
972 History *h = &history[s];
973 printf("history %s changes %zu", a[0], h->count);
974 for (size_t i = 0; i < h->count; ++i)
975 printf(" %lld:%c", h->items[i].time, h->items[i].value);
976 putchar('\n');
977 } else if (id == 14) {
978 long long k = number(a[0], 1);
979 int order[550];
980 for (int i = 0; i < simulation->signal_count; ++i)
981 order[i] = i;
982 qsort(order, (size_t)simulation->signal_count, sizeof *order, activity_compare);
983 for (int i = 0; i < simulation->signal_count && i < k; ++i)
984 printf("%s %zu\n", simulation->signals[order[i]], history[order[i]].count);
985 } else {
986 int zero = 0, one = 0, unknown = 0;
987 for (int i = 0; i < simulation->signal_count; ++i) {
988 zero += values[i] == '0';
989 one += values[i] == '1';
990 unknown += values[i] == 'X';
991 }
992 printf("circuit %s time %lld pending %zu events %lld halted %s\n", simulation_name, now,
993 event_count, processed, halted ? "yes" : "no");
994 printf("values 0:%d 1:%d X:%d\n", zero, one, unknown);
995 }
996}

998int main(void) {
999 Line lines[1000];
1000 int count = 0;
1001 char buffer[128];
1002 while (count < 1000 && fgets(buffer, sizeof buffer, stdin)) {
1003 Line *line = &lines[count];
1004 line->number = count + 1;
1005 line->count = 0;
1006 char *storage = copy(buffer);
1007 for (char *word = strtok(storage, " \r\n"); word; word = strtok(NULL, " \r\n"))
1008 line->words[line->count++] = word;
1009 ++count;
1010 }
1011 int index = 0;
1012 while (index < count) {
1013 int first = index++, end = -1;
1014 Line *line = &lines[first];
1015 if (line->count == 1 && !strcmp(line->words[0], "END"))
1016 break;
1017 if (!strcmp(line->words[0], "MODULE")) {
1018 end = index;
1019 while (end < count &&
1020 !(lines[end].count == 1 && !strcmp(lines[end].words[0], "ENDMODULE")))
1021 ++end;
1022 index = end + 1;
1023 }
1024 error_line = line->number;
1025 if (!setjmp(command_end)) {
1026 if (end >= 0)
1027 define_module(lines, first, end);
1028 else
1029 execute(line);
1030 }
1031 }
1032 return 0;
