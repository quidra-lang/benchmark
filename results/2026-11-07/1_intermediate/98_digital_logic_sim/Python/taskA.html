110def gate_value(kind, values):
111 if kind in ('AND', 'NAND'):
112 result = '0' if '0' in values else 'X' if 'X' in values else '1'
113 elif kind in ('OR', 'NOR'):
114 result = '1' if '1' in values else 'X' if 'X' in values else '0'
115 elif kind in ('XOR', 'XNOR'):
116 result = 'X' if 'X' in values else str(values.count('1') % 2)
117 elif kind == 'MUX2':
118 select, first, second = values
119 if select == 'X':
120 result = first if first == second else 'X'
121 else:
122 result = first if select == '0' else second
123 else:
124 result = values[0]
125 return INVERTED[result] if kind in INVERTING_TYPES else result

156def expand(module, prefix, ports, gates):
157 def actual(signal):
158 return ports.get(signal, prefix + signal)
159
160 for item in module.body:
161 if isinstance(item, Gate):
162 inputs = [actual(signal) for signal in item.inputs]
163 output = actual(item.output)
164 gates.append(Gate(prefix + item.name, item.kind, inputs, output, item.delay))
165 else:
166 mapping = {port: actual(signal) for port, signal in item.connections.items()}
167 expand(item.module, f'{prefix}{item.name}.', mapping, gates)

170class Circuit:
171 def __init__(self, module):
172 self.name = module.name
173 self.inputs = module.inputs
174 self.outputs = module.outputs
175 self.gates = []
176 expand(module, '', {}, self.gates)
177 self.signals = self.inputs + [gate.output for gate in self.gates]
178 self.driver = {gate.output: gate for gate in self.gates}
179 self.readers = {signal: [] for signal in self.signals}
180 for gate in self.gates:
181 for signal in dict.fromkeys(gate.inputs):
182 self.readers[signal].append(gate)
183 self.flip_flops = [gate for gate in self.gates if gate.kind == 'DFF']
184 self.order = self.combinational_order()
185
186 def combinational_order(self):
187 waiting = {}
188 ready = []
189 for gate in self.gates:
190 if gate.kind == 'DFF':
191 continue
192 sources = {
193 signal
194 for signal in gate.inputs
195 if signal in self.driver and self.driver[signal].kind != 'DFF'
196 }
197 waiting[gate.output] = len(sources)
198 if not sources:
199 ready.append(gate)
200 order = []
201 while ready:
202 gate = ready.pop()
203 order.append(gate)
204 for reader in self.readers[gate.output]:
205 if reader.kind != 'DFF':
206 waiting[reader.output] -= 1
207 if waiting[reader.output] == 0:
208 ready.append(reader)
209 return order if len(order) == len(waiting) else None
210
211 def depth(self):
212 levels = dict.fromkeys(self.inputs, 0)
213 levels.update(dict.fromkeys((gate.output for gate in self.flip_flops), 0))
214 for gate in self.order:
215 levels[gate.output] = max(levels[signal] for signal in gate.inputs) + 1
216 return max(levels.values())
217
218 def evaluate(self, bits, fault=None):
219 stuck_signal, stuck_value = fault or (None, None)
220 values = dict(zip(self.inputs, bits))
221 if stuck_signal in values:
222 values[stuck_signal] = stuck_value
223 for gate in self.order:
224 if gate.output == stuck_signal:
225 values[gate.output] = stuck_value
226 else:
227 values[gate.output] = gate_value(gate.kind, [values[s] for s in gate.inputs])
228 return ''.join(values[signal] for signal in self.outputs)
229
230 def arrival_times(self, sources):
231 latest = dict(sources)
232 earliest = dict(sources)
233 previous = {}
234 for gate in self.order:
235 arrived = [signal for signal in gate.inputs if signal in latest]
236 if arrived:
237 source = max(arrived, key=latest.get)
238 latest[gate.output] = latest[source] + gate.delay
239 earliest[gate.output] = min(earliest[signal] for signal in arrived) + gate.delay
240 previous[gate.output] = source
241 return latest, earliest, previous

269def check_gate_line(words, driven, used_names):
270 if len(words) < 5 or words[-3] != '->' or words.count('->') != 1:
271 raise InputError('bad gate syntax')
272 kind, name = words[0], words[1]
273 inputs, output, delay = words[2:-3], words[-2], words[-1]
274 if not is_name(name):
275 raise InputError(f'bad name {name}')
276 low, high = INPUT_COUNTS[kind]
277 if not low <= len(inputs) <= high:
278 raise InputError(f'bad input count {len(inputs)} for {kind}')
279 for signal in inputs + [output]:
280 if not is_name(signal):
281 raise InputError(f'bad name {signal}')
282 if not is_number(delay) or int(delay) > MAX_DELAY:
283 raise InputError(f'bad delay {delay}')
284 if name in used_names:
285 raise InputError(f'duplicate name {name}')
286 if output in driven:
287 raise InputError(f'multiple drivers for {output}')
288 used_names.add(name)
289 driven.add(output)
290 return Gate(name, kind, inputs, output, int(delay))

293class Simulation:

295 self.circuit = circuit

299 def reset(self):
300 self.values = dict.fromkeys(self.circuit.signals, 'X')
301 self.time = 0
302 self.queue = []
303 self.sequence = itertools.count()
304 self.history = {signal: [] for signal in self.circuit.signals}
305 self.processed = 0
306 self.halted = False
307
308 def schedule(self, time, signal, value):
309 heapq.heappush(self.queue, (time, next(self.sequence), signal, value))
310
311 def assign(self, time, signal, value):
312 previous = self.values[signal]
313 self.values[signal] = value
314 for gate in self.circuit.readers[signal]:
315 if gate.kind != 'DFF':
316 inputs = [self.values[source] for source in gate.inputs]
317 self.schedule(time + gate.delay, gate.output, gate_value(gate.kind, inputs))
318 elif signal == gate.inputs[0] and previous == '0' and value == '1':
319 self.schedule(time + gate.delay, gate.output, self.values[gate.inputs[1]])
320
321 def step(self, time):
322 original = {}
323 events = 0
324 while self.queue and self.queue[0][0] == time:
325 if events == OSCILLATION_LIMIT:
326 return None
327 _, _, signal, value = heapq.heappop(self.queue)
328 events += 1
329 self.processed += 1
330 if self.values[signal] != value:
331 original.setdefault(signal, self.values[signal])
332 self.assign(time, signal, value)
333 glitches = 0
334 for signal, start in original.items():
335 if self.values[signal] == start:
336 glitches += 1
337 else:
338 self.history[signal].append((time, self.values[signal]))
339 for signal in self.watch:
340 if signal in original and self.values[signal] != original[signal]:
341 print(f'{time} {signal} {self.values[signal]}')
342 return events, glitches
343
344 def run(self, end):
345 events = 0
346 glitches = 0
347 while self.queue and self.queue[0][0] < end:
348 time = self.queue[0][0]
349 result = self.step(time)
350 if result is None:
351 print(f'halted: oscillation at time {time}')
352 self.queue.clear()
353 self.time = time
354 self.halted = True
355 return
356 events += result[0]
357 glitches += result[1]
358 self.time = end
359 print(f'run {end} events {events} glitches {glitches} pending {len(self.queue)}')
360
361 def waveform(self, signal, start, end):
362 changes = self.history[signal]
363 value = 'X'
364 index = 0
365 characters = []
366 for time in range(start, end):
367 while index < len(changes) and changes[index][0] <= time:
368 value = changes[index][1]
369 index += 1
370 characters.append(value)
371 return ''.join(characters)

382 def define_module(self, header_no, header, body_lines):
383 line_no = header_no
384 try:
385 name, inputs, outputs = self.check_header(header.split(' '))
386 driven = set(inputs)
387 used_names = set()
388 body = []
389 reads = []
390 for line_no, line in enumerate(body_lines, header_no + 1):
391 words = line.split(' ')
392 if words[0] in GATE_TYPES:
393 item = check_gate_line(words, driven, used_names)
394 reads.extend((line_no, signal) for signal in item.inputs)
395 elif words[0] == 'INST':
396 item = self.check_instance_line(words, driven, used_names)
397 reads.extend(
398 (line_no, signal)
399 for port, signal in item.connections.items()
400 if port in item.module.inputs
401 )
402 else:
403 raise InputError(f'unknown statement {words[0]}')
404 body.append(item)
405 reads.extend((header_no, signal) for signal in outputs)
406 for line_no, signal in reads:
407 if signal not in driven:
408 raise InputError(f'undriven signal {signal}')
409 except InputError as error:
410 print(f'error: line {line_no}: {error}')
411 return
412 self.modules[name] = Module(name, inputs, outputs, body)
413 gate_count = sum(isinstance(item, Gate) for item in body)
414 print(
415 f'defined {name} inputs {len(inputs)} outputs {len(outputs)} '
416 f'gates {gate_count} instances {len(body) - gate_count}'
417 )

419 def check_header(self, words):
420 if (
421 len(words) < 5
422 or words[2] != 'IN'
423 or words[3:].count('OUT') != 1
424 or words[-1] == 'OUT'
425 ):
426 raise InputError('bad module header')
427 name = words[1]
428 if not MODULE_NAME_PATTERN.fullmatch(name):
429 raise InputError(f'bad module name {name}')
430 if name in self.modules:
431 raise InputError(f'duplicate module {name}')
432 split = words.index('OUT', 3)
433 inputs, outputs = words[3:split], words[split + 1 :]
434 seen = set()
435 for port in inputs + outputs:
436 if not is_name(port):
437 raise InputError(f'bad name {port}')
438 if port in seen:
439 raise InputError(f'duplicate port {port}')
440 seen.add(port)
441 return name, inputs, outputs

443 def check_instance_line(self, words, driven, used_names):
444 if len(words) < 3:
445 raise InputError('bad instance syntax')
446 name, module_name = words[1], words[2]
447 if not is_name(name):
448 raise InputError(f'bad name {name}')
449 module = self.modules.get(module_name)
450 if module is None:
451 raise InputError(f'unknown module {module_name}')
452 connections = {}
453 for word in words[3:]:
454 port, _, signal = word.partition('=')
455 if word.count('=') != 1 or not port or not signal:
456 raise InputError(f'bad connection {word}')
457 if port not in module.inputs and port not in module.outputs:
458 raise InputError(f'unknown port {port}')
459 if port in connections:
460 raise InputError(f'duplicate connection {port}')
461 if not is_name(signal):
462 raise InputError(f'bad name {signal}')
463 connections[port] = signal
464 for port in module.inputs:
465 if port not in connections:
466 raise InputError(f'unconnected input {port}')
467 if name in used_names:
468 raise InputError(f'duplicate name {name}')
469 for port, signal in connections.items():
470 if port in module.outputs:
471 if signal in driven:
472 raise InputError(f'multiple drivers for {signal}')
473 driven.add(signal)
474 used_names.add(name)
475 return Instance(name, module, connections)

477 def execute(self, line_no, words):
478 name, arguments = words[0], words[1:]
479 try:
480 if name not in ARGUMENT_COUNTS:
481 raise InputError(f'unknown command {name}')
482 expected = ARGUMENT_COUNTS[name]
483 if len(arguments) < expected or (
484 name not in VARIADIC_COMMANDS and len(arguments) != expected
485 ):
486 raise InputError('wrong argument count')
487 if name in SIMULATION_COMMANDS and self.simulation is None:
488 raise InputError('no circuit')
489 if name in BLOCKED_WHEN_HALTED and self.simulation.halted:
490 raise InputError('simulation halted')
491 getattr(self, f'command_{name.lower()}')(*arguments)
492 except InputError as error:
493 print(f'error: line {line_no}: {error}')

573 def command_faults(self, name):
574 circuit = combinational_circuit(self.lookup(name))
575 width = len(circuit.inputs)
576 rows = [bit_string(row, width) for row in range(2**width)]
577 expected = [circuit.evaluate(bits) for bits in rows]

589 for fault in undetected:

593 def command_critical(self, name):
594 circuit = self.lookup(name).circuit
595 if circuit.order is None:
596 raise InputError(f'{name} has a combinational loop')

598 reached = []

602 reached.append(port)

604 print(f'{port} unreachable')

607 print(f'critical {latest[end]}: {trace_path(previous, end)}')
608 else:
609 print('critical none')

684 def command_unwatch(self, *signals):

686 for i, signal in enumerate(signals):
687 self.require_signal(signal)
688 if signal not in simulation.watch or signal in signals[:i]:
689 raise InputError(f'not watched {signal}')
690 simulation.watch = [signal for signal in simulation.watch if signal not in signals]
691 self.print_watch_list()

712 def command_probe(self, signal):
713 self.require_signal(signal)
714 simulation = self.simulation
715 circuit = simulation.circuit
716 value = simulation.values[signal]
717 if signal in circuit.inputs:
718 print(f'{signal}={value} driven by input')
719 else:
720 gate = circuit.driver[signal]
721 pins = ' '.join(simulation.describe(gate.inputs))
722 print(f'{signal}={value} driven by {gate.name} {gate.kind} {pins} delay {gate.delay}')
723 readers = circuit.readers[signal]
724 print('fanout', len(readers), *(gate.name for gate in readers))

726 def command_wave(self, start_word, end_word):
727 simulation = self.simulation
728 start = parse_number(start_word)
729 end = parse_number(end_word)
730 if not start < end <= simulation.time:
731 raise InputError(f'bad range {start} {end}')
732 if not simulation.watch:
733 raise InputError('nothing watched')
734 for signal in simulation.watch:
735 print(signal, simulation.waveform(signal, start, end))

737 def command_history(self, signal):
738 self.require_signal(signal)
739 changes = self.simulation.history[signal]

743 def command_activity(self, count_word):
744 count = parse_positive(count_word)
745 history = self.simulation.history
746 ranked = sorted(history, key=lambda signal: (-len(history[signal]), signal))
747 for signal in ranked[:count]:
748 print(signal, len(history[signal]))
