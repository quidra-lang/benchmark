49def bits(r, n) = n.times.map { |i| ((r >> (n - 1 - i)) & 1).to_s }.join
50def logic(kind, a)
51 y = case kind
52 when "AND", "NAND"
53 a.include?("0") ? "0" : a.include?("X") ? "X" : "1"
54 when "OR", "NOR"
55 a.include?("1") ? "1" : a.include?("X") ? "X" : "0"
56 when "XOR", "XNOR"
57 a.include?("X") ? "X" : (a.count("1") % 2).to_s
58 when "MUX2"
59 a[0] == "X" ? (a[1] == a[2] ? a[1] : "X") : a[a[0] == "0" ? 1 : 2]
60 else
61 a[0]
62 end
63
64 %w[NAND NOR XNOR NOT].include?(kind) ? {"0" => "1", "1" => "0", "X" => "X"}[y] : y
65end

67class Circuit
68 attr_reader :name, :inputs, :outputs, :gates, :signals, :driver, :readers, :flops, :order
69 def initialize(mod)
70 @name, @inputs, @outputs = mod.name, mod.inputs, mod.outputs
71 @gates = []
72 expand(mod, "", {})
73 @signals = @inputs + @gates.map(&:output)
74 @driver = @gates.to_h { |g| [g.output, g] }
75 @readers = @signals.to_h { |s| [s, []] }
76 @gates.each { |g| g.inputs.uniq.each { |s| @readers[s] << g } }
77 @flops = @gates.select { |g| g.kind == "DFF" }
78 waiting = {}
79 ready = []
80 @gates.each do |g|
81 next if g.kind == "DFF"
82 waiting[g.output] = g.inputs.uniq.count { |s| @driver.key?(s) && @driver[s].kind != "DFF" }
83 ready << g if waiting[g.output] == 0
84 end
85
86 @order = []
87 until ready.empty?
88 g = ready.pop
89 @order << g
90 @readers[g.output].each do |r|
91 next if r.kind == "DFF"
92 waiting[r.output] -= 1
93 ready << r if waiting[r.output] == 0
94 end
95 end
96
97 @order = nil if @order.size != waiting.size
98 end
99
100 def expand(mod, prefix, ports)
101 actual = -> (s) { ports.fetch(s, prefix + s) }
102 mod.body.each do |x|
103 if x.is_a?(Gate)
104 @gates << Gate.new(prefix + x.name, x.kind, x.inputs.map(&actual), actual.call(x.output), x.delay)
105 else
106 expand(x.mod, prefix + x.name + ".", x.connections.to_h { |p, s| [p, actual.call(s)] })
107 end
108 end
109 end
110
111 def combinational
112 raise "#{@name} is not combinational" unless @flops.empty?
113 raise "#{@name} has a combinational loop" unless @order
114 raise "too many inputs #{@inputs.size}" if @inputs.size > 8
115 self
116 end
117
118 def evaluate(b, stuck = nil, v = nil)
119 values = @inputs.zip(b.chars).to_h
120 values[stuck] = v if values.key?(stuck)
121 @order.each { |g| values[g.output] = g.output == stuck ? v : logic(g.kind, g.inputs.map { |s| values[s] }) }
122 @outputs.map { |s| values[s] }.join
123 end
124
125 def arrivals(period)
126 late = period ? @flops.to_h { |g| [g.output, g.delay] } : @inputs.to_h { |s| [s, 0] }
127 early = late.dup
128 prev = {}
129 @order.each do |g|
130 a = g.inputs.select { |s| late.key?(s) }
131 next if a.empty?
132 best = a.max_by { |s| late[s] }
133 late[g.output] = late[best] + g.delay
134 early[g.output] = a.map { |s| early[s] }.min + g.delay
135 prev[g.output] = best
136 end
137
138 [late, early, prev]
139 end
140end

142class Simulation
143 attr_accessor :watch, :values, :time, :queue, :history, :processed, :halted
144 attr_reader :circuit
145 def initialize(c)
146 @circuit = c
147 @watch = []
148 reset
149 end
150
151 def reset
152 @values = @circuit.signals.to_h { |s| [s, "X"] }
153 @time = 0
154 @queue = []
155 @sequence = 0
156 @history = @circuit.signals.to_h { |s| [s, []] }
157 @processed = 0
158 @halted = false
159 end
160
161 def schedule(t, s, v)
162 e = [t, @sequence, s, v]
163 @sequence += 1
164 i = @queue.size
165 @queue << e
166 while i > 0
167 p = (i - 1) / 2
168 break if (@queue[p][0, 2] <=> e[0, 2]) <= 0
169 @queue[i] = @queue[p]
170 i = p
171 end
172
173 @queue[i] = e
174 end
175
176 def pop
177 result = @queue[0]
178 last = @queue.pop
179 unless @queue.empty?
180 i = 0
181 while 2 * i + 1 < @queue.size
182 j = 2 * i + 1
183 j += 1 if j + 1 < @queue.size && (@queue[j + 1][0, 2] <=> @queue[j][0, 2]) < 0
184 break if (last[0, 2] <=> @queue[j][0, 2]) <= 0
185 @queue[i] = @queue[j]
186 i = j
187 end
188
189 @queue[i] = last
190 end
191
192 result
193 end
194
195 def run(ending)
196 total = 0
197 glitches = 0
198 while !@queue.empty? && @queue[0][0] < ending
199 t = @queue[0][0]
200 start = @values.dup
201 changed = []
202 count = 0
203 while !@queue.empty? && @queue[0][0] == t
204 if count == 1000
205 puts("halted: oscillation at time #{t}")
206 @queue = []
207 @time = t
208 @halted = true
209 return
210 end
211
212 _, _, s, v = pop
213 count += 1
214 @processed += 1
215 old = @values[s]
216 next if old == v
217 @values[s] = v
218 changed << s
219 @circuit.readers[s].each do |g|
220 if g.kind != "DFF"
221 schedule(t + g.delay, g.output, logic(g.kind, g.inputs.map { |x| @values[x] }))
222 elsif s == g.inputs[0] && old == "0" && v == "1"
223 schedule(t + g.delay, g.output, @values[g.inputs[1]])
224 end
225 end
226 end
227
228 total += count
229 changed.uniq.each { |s| @values[s] == start[s] ? glitches += 1 : @history[s] << [t, @values[s]] }
230 @watch.each { |s| puts("#{t} #{s} #{@values[s]}") if @values[s] != start[s] }
231 end
232
233 @time = ending
234 puts("run #{ending} events #{total} glitches #{glitches} pending #{@queue.size}")
235 end
236
237 def describe(a) = a.map { |s| "#{s}=#{@values[s]}" }
238end

240class Simulator
241 def initialize = (
242 @modules = {}
243 @sim = nil
244 )

247 def input(s) = (raise "not an input #{s}" unless @sim.circuit.inputs.include?(s))

249 t = number(w)

251 t

254 def define(header_no, lines)
255 line_no = header_no
256 begin
257 w = lines[0].split(" ")
258 raise "bad module header" unless w.size >= 5 && w[2] == "IN" && w[3..].count("OUT") == 1 && w[-1] != "OUT"
259 name = w[1]
260 raise "bad module name #{name}" unless /\A[A-Z][A-Z0-9_]{0,15}\z/.match?(name)
261 raise "duplicate module #{name}" if @modules.key?(name)
262 split = w[3..].index("OUT") + 3
263 ins = w[3...split]
264 outs = w[(split + 1)..]
265 seen = []
266 (ins + outs).each { |p|
267 raise "bad name #{p}" unless name?(p)
268 raise "duplicate port #{p}" if seen.include?(p)
269 seen << p
270 }
271 driven = ins.dup
272 used = []
273 body = []
274 reads = []
275 lines.drop(1).each_with_index do |line, i|
276 line_no = header_no + i + 1
277 a = line.split(" ")
278 kind = a[0]
279 n = a[1]
280 if TYPES.include?(kind)
281 raise "bad gate syntax" unless a.size >= 5 && a[-3] == "->" && a.count("->") == 1
282 raise "bad name #{n}" unless name?(n)
283 inputs = a[2...-3]
284 output = a[-2]
285 delay = a[-1]
286 low = %w[NOT BUF].include?(kind) ? 1 : kind == "MUX2" ? 3 : 2
287 high = TYPES.index(kind) < 6 ? 8 : low
288 raise "bad input count #{inputs.size} for #{kind}" unless (low..high).cover?(inputs.size)
289 (inputs + [output]).each { |s| raise "bad name #{s}" unless name?(s) }
290 raise "bad delay #{delay}" unless number?(delay) && delay.to_i <= 1000
291 raise "duplicate name #{n}" if used.include?(n)
292 raise "multiple drivers for #{output}" if driven.include?(output)
293 driven << output
294 used << n
295 body << Gate.new(n, kind, inputs, output, delay.to_i)
296 inputs.each { |s| reads << [line_no, s] }
297 elsif kind == "INST"
298 raise "bad instance syntax" if a.size < 3
299 raise "bad name #{n}" unless name?(n)
300 m = lookup(a[2])
301 connections = {}
302 a.drop(3).each do |word|
303 p = word.split("=", -1)
304 raise "bad connection #{word}" unless p.size == 2 && !p[0].empty? && !p[1].empty?
305 port, s = p
306 raise "unknown port #{port}" unless (m.inputs + m.outputs).include?(port)
307 raise "duplicate connection #{port}" if connections.key?(port)
308 raise "bad name #{s}" unless name?(s)
309 connections[port] = s
310 end
311
312 m.inputs.each { |p| raise "unconnected input #{p}" unless connections.key?(p) }
313 raise "duplicate name #{n}" if used.include?(n)
314 connections.each { |p, s|
315 next unless m.outputs.include?(p)
316 raise "multiple drivers for #{s}" if driven.include?(s)
317 driven << s
318 }
319 used << n
320 body << Instance.new(n, m, connections)
321 connections.each { |p, s| reads << [line_no, s] if m.inputs.include?(p) }
322 else
323 raise "unknown statement #{kind}"
324 end
325 end
326
327 outs.each { |s| reads << [header_no, s] }
328 reads.each { |n, s|
329 next if driven.include?(s)
330 line_no = n
331 raise "undriven signal #{s}"
332 }
333 @modules[name] = Mod.new(name, ins, outs, body)
334 gates = body.count { |x| x.is_a?(Gate) }
335 puts("defined #{name} inputs #{ins.size} outputs #{outs.size} gates #{gates} instances #{body.size - gates}")
336 rescue RuntimeError => e
337 puts("error: line #{line_no}: #{e.message}")
338 end
339 end

344 puts(" " * depth + "#{x.name} #{x.mod.name}")

349 def execute(line_no, w)
350 cmd = w[0]
351 a = w.drop(1)
352 raise "unknown command #{cmd}" unless COUNTS.key?(cmd)
353 unless a.size >= COUNTS[cmd] && (%w[WATCH UNWATCH SHOW].include?(cmd) || a.size == COUNTS[cmd])
354 raise "wrong argument count"
355 end
356
357 raise "no circuit" if SIM.include?(cmd) && !@sim
358 s = @sim
359 raise "simulation halted" if %w[SET APPLY PULSE CLOCK RUN].include?(cmd) && s.halted
360 case cmd
361 when "MODULES"
362 puts("modules #{@modules.size}")
363 @modules.each_value { |m|
364 puts("#{m.name} inputs #{m.inputs.size} outputs #{m.outputs.size} gates #{Circuit.new(m).gates.size}")
365 }
366 when "BUILD", "NETLIST", "HIERARCHY", "STATS", "TRUTHTABLE", "EQUIV", "FAULTS", "CRITICAL", "PERIOD"
367 m = lookup(a[0])
368 m2 = lookup(a[1]) if cmd == "EQUIV"
369 c = Circuit.new(m)
370 case cmd
371 when "BUILD"
372 @sim = Simulation.new(c)
373 puts("built #{c.name} signals #{c.signals.size} gates #{c.gates.size}")
374 when "NETLIST"
375 puts("netlist #{c.name} gates #{c.gates.size} signals #{c.signals.size}")
376 c.gates.each { |g| puts("#{g.name} #{g.kind} #{g.inputs.join(" ")} -> #{g.output} #{g.delay}") }
377 when "HIERARCHY"
378 puts(m.name)
379 hierarchy(m, 1)
380 when "STATS"
381 levels = (c.inputs + c.flops.map(&:output)).to_h { |x| [x, 0] }
382 c.order&.each { |g| levels[g.output] = g.inputs.map { |x| levels[x] }.max + 1 }
383 puts(
384 "stats #{c.name} gates #{c.gates.size} signals #{c.signals.size} depth #{c.order ? levels.values.max : "-"}"
385 )
386 puts(
387 (["types"] +
388 TYPES.flat_map { |t|
389 n = c.gates.count { |g| g.kind == t }
390 n == 0 ? [] : [t, n]
391 })
392 .join(" ")
393 )
394 best = c.signals.max_by { |x| c.readers[x].size }
395 puts("max fanout #{best} #{c.readers[best].size}")
396 when "CRITICAL", "PERIOD"
397 period = cmd == "PERIOD"
398 raise "#{c.name} has no flip-flops" if period && c.flops.empty?
399 raise "#{c.name} has a combinational loop" unless c.order
400 late, early, prev = c.arrivals(period)
401 ends = period ? c.flops.map { |g| g.inputs[1] } : c.outputs
402 ends.each { |x| puts(late.key?(x) ? "#{x} min #{early[x]} max #{late[x]}" : "#{x} unreachable") } unless period
403 reached = ends.select { |x| late.key?(x) }
404 heading = period ? "flip-flops #{c.flops.size} period" : "critical"
405 if reached.empty?
406 puts(heading + " none")
407 else
408 best = reached.max_by { |x| late[x] }
409 path = [best]
410 path << prev[path[-1]] while prev.key?(path[-1])
411 puts("#{heading} #{late[best]}: #{path.reverse.join(" -> ")}")
412 end
413 else
414 c.combinational
415 rows = (0...2 ** c.inputs.size).map { |r| bits(r, c.inputs.size) }
416 expected = rows.map { |b| c.evaluate(b) }
417 if cmd == "TRUTHTABLE"
418 puts((c.inputs + ["|"] + c.outputs).join(" "))
419 rows.each_with_index { |b, r| puts((b.chars + ["|"] + expected[r].chars).join(" ")) }
420 elsif cmd == "EQUIV"
421 d = Circuit.new(m2).combinational
422 raise "port mismatch" unless c.inputs.size == d.inputs.size && c.outputs.size == d.outputs.size
423 r = rows.each_index.find { |i| expected[i] != d.evaluate(rows[i]) }
424 puts(r ? "not equivalent at #{rows[r]}: #{expected[r]} vs #{d.evaluate(rows[r])}" : "equivalent")
425 else
426 undetected = []
427 tests = []
428 c.signals.each { |x|
429 %w[0 1].each { |v|
430 r = rows.each_index.find { |i| c.evaluate(rows[i], x, v) != expected[i] }
431 r ? tests << r : undetected << "#{x}/#{v}"
432 }
433 }
434 total = 2 * c.signals.size
435 puts("faults #{total} detected #{total - undetected.size}")
436 undetected.each { |f| puts("undetected " + f) }
437 puts((["tests"] + tests.uniq.sort.map { |r| rows[r] }).join(" "))
438 end
439 end
440
441 when "RESET"
442 s.reset
443 puts("reset")
444 when "SET"
445 t = time(a[0])
446 input(a[1])
447 value(a[2])
448 s.schedule(t, a[1], a[2])
449 puts("set #{a[1]} #{a[2]} at #{t}")
450 when "APPLY"
451 t = time(a[0])
452 b = a[1]
453 raise "bad vector #{b}" unless b.size == s.circuit.inputs.size && /\A[01X]*\z/.match?(b)
454 s.circuit.inputs.each_with_index { |x, i| s.schedule(t, x, b[i]) }
455 puts("apply #{b} at #{t}")
456 when "PULSE", "CLOCK"
457 input(a[0])
458 t = time(a[1])
459 width = number(a[2], true)
460 if cmd == "PULSE"
461 s.schedule(t, a[0], "1")
462 s.schedule(t + width, a[0], "0")
463 puts("pulse #{a[0]} #{t} #{t + width}")
464 else
465 count = number(a[3], true)
466 count.times { |i| s.schedule(t + i * width, a[0], (i % 2).to_s) }
467 puts("clock #{a[0]} events #{count} last #{t + (count - 1) * width}")
468 end
469
470 when "RUN"
471 t = number(a[0])
472 raise "time #{t} is not after #{s.time}" if t <= s.time
473 s.run(t)
474 when "WATCH", "UNWATCH"
475 seen = []
476 a.each { |x|
477 signal(x)
478 bad = cmd == "WATCH" ? s.watch.include?(x) || seen.include?(x) : !s.watch.include?(x) || seen.include?(x)
479 raise "#{cmd == "WATCH" ? "already watched" : "not watched"} #{x}" if bad
480 seen << x
481 }
482 s.watch = cmd == "WATCH" ? s.watch + a : s.watch - a
483 puts("watching " + (s.watch.empty? ? "nothing" : s.watch.join(" ")))
484 when "SHOW"
485 a.each { |x| signal(x) }
486 puts((a.empty? ? s.describe(s.circuit.inputs) + ["|"] + s.describe(s.circuit.outputs) : s.describe(a)).join(" "))
487 when "CHECK"
488 signal(a[0])
489 value(a[1])
490 puts("check #{a[0]}=#{a[1]} " + (s.values[a[0]] == a[1] ? "ok" : "failed (#{s.values[a[0]]})"))
491 when "PROBE"
492 x = a[0]
493 signal(x)
494 g = s.circuit.driver[x]
495 puts(
496 "#{x}=#{s.values[x]} driven by " +
497 (g ? "#{g.name} #{g.kind} #{s.describe(g.inputs).join(" ")} delay #{g.delay}" : "input")
498 )
499 r = s.circuit.readers[x]
500 puts((["fanout", r.size] + r.map(&:name)).join(" "))
501 when "WAVE"
502 first = number(a[0])
503 last = number(a[1])
504 raise "bad range #{first} #{last}" unless first < last && last <= s.time
505 raise "nothing watched" if s.watch.empty?
506 s.watch.each do |x|
507 h = s.history[x]
508 i = 0
509 v = "X"
510 wave = (first...last)
511 .map { |t|
512 while i < h.size && h[i][0] <= t
513 v = h[i][1]
514 i += 1
515 end
516
517 v
518 }
519 .join
520 puts(x + " " + wave)
521 end
522
523 when "HISTORY"
524 x = a[0]
525 signal(x)
526 h = s.history[x]
527 puts((["history", x, "changes", h.size] + h.map { |t, v| "#{t}:#{v}" }).join(" "))
528 when "ACTIVITY"
529 k = number(a[0], true)
530 s.circuit.signals.sort_by { |x| [-s.history[x].size, x] }.first(k).each { |x| puts("#{x} #{s.history[x].size}") }
531 when "STATUS"
532 puts(
533 "circuit #{s.circuit.name} time #{s.time} pending #{s.queue.size} events #{s.processed} halted #{s.halted ? "yes" : "no"}"
534 )
535 puts("values 0:#{s.values.values.count("0")} 1:#{s.values.values.count("1")} X:#{s.values.values.count("X")}")
536 end
537
538 rescue RuntimeError => e
539 puts("error: line #{line_no}: #{e.message}")
540 end
541end
