20 static bool NameOk(string s) => Regex.IsMatch(s, "\\A[a-z][a-z0-9_]{0,15}\\z");
21 static bool NumberOk(string s) => Regex.IsMatch(s, "\\A(?:0|[1-9][0-9]{0,8})\\z");
22 static int Number(string s, bool positive = false) {
23 if (!NumberOk(s) || (positive && s == "0"))
24 Fail("bad number " + s);
25 return int.Parse(s);
26 }
27 static char Value(string s) {
28 if (s is not("0" or "1" or "X"))
29 Fail("bad value " + s);
30 return s[0];

32 static string Bits(int r, int n) =>
33 string.Concat(Enumerable.Range(0, n).Select(i => (char)('0' + ((r >> (n - 1 - i)) & 1))));
34 static char Logic(string kind, List<char> a) {
35 char y = kind switch {
36 "AND" or "NAND" => a.Contains('0') ? '0'
37 : a.Contains('X') ? 'X'
38 : '1',
39 "OR" or "NOR" => a.Contains('1') ? '1'
40 : a.Contains('X') ? 'X'
41 : '0',
42 "XOR" or "XNOR" => a.Contains('X') ? 'X' : (char)('0' + a.Count(x => x == '1') % 2),
43 "MUX2" => a[0] == 'X' ? (a[1] == a[2] ? a[1] : 'X') : a[a[0] == '0' ? 1 : 2],
44 _ => a[0]
45 };
46 return kind is "NAND" or "NOR" or "XNOR" or "NOT" ? y == '0' ? '1'
47 : y == '1' ? '0'
48 : 'X'
49 : y;

59 sealed class Circuit {
60 public readonly string Name;
61 public readonly List<string> Inputs, Outputs, Signals;
62 public readonly List<Gate> Gates = [], Flops = [], Order = [];
63 public readonly Dictionary<string, Gate> Driver = [];
64 public readonly Dictionary<string, List<Gate>> Readers = [];
65 public readonly bool Loop;
66 public Circuit(Module m) {
67 Name = m.Name;
68 Inputs = m.Inputs;
69 Outputs = m.Outputs;
70 Expand(m, "", []);
71 Signals = [..Inputs, ..Gates.Select(g => g.Output)];
72 foreach (string s in Signals)
73 Readers[s] = [];
74 foreach (Gate g in Gates) {
75 Driver[g.Output] = g;
76 if (g.Kind == "DFF")
77 Flops.Add(g);
78 foreach (string s in g.Inputs.Distinct())
79 Readers[s].Add(g);
80 }
81 Dictionary<Gate, int> waiting = [];
82 Stack<Gate> ready = new();
83 foreach (Gate g in Gates) {
84 if (g.Kind == "DFF")
85 continue;int n=g.Inputs.Distinct().Count(s=>Driver.TryGetValue(s,out Gate? d)&&d.Kind!="DFF");
86 waiting[g] = n;
87 if (n == 0)
88 ready.Push(g);
89 }
90 while (ready.Count > 0) {
91 Gate g = ready.Pop();
92 Order.Add(g);
93 foreach (Gate r in Readers[g.Output])
94 if (r.Kind != "DFF" && --waiting[r] == 0)
95 ready.Push(r);
96 }
97 Loop = Order.Count != waiting.Count;
98 }
99 void Expand(Module m, string prefix, Dictionary<string, string> ports) {
100 string Actual(string s) => ports.GetValueOrDefault(s, prefix + s);
101 foreach (Item item in m.Body)
102 if (item is Gate g)
103 Gates.Add(new(prefix + g.Name, g.Kind, g.Inputs.Select(Actual).ToList(),
104 Actual(g.Output), g.Delay));
105 else {
106 Instance x = (Instance)item;
107 Expand(x.Module, prefix + x.Name + ".",
108 x.Connections.ToDictionary(c => c.Port, c => Actual(c.Signal)));
109 }
110 }
111 public void Combinational() {
112 if (Flops.Count > 0)
113 Fail(Name + " is not combinational");
114 if (Loop)
115 Fail(Name + " has a combinational loop");
116 if (Inputs.Count > 8)
117 Fail("too many inputs " + Inputs.Count);
118 }
119 public string Evaluate(string bits, string stuck = "", char v = 'X') {
120 Dictionary<string, char> values =
121 Inputs.Select((s, i) => (s, v: s == stuck ? v : bits[i]))
122 .ToDictionary(x => x.s, x => x.v);
123 foreach (Gate g in Order)
124 values[g.Output] =
125 g.Output == stuck ? v : Logic(g.Kind, g.Inputs.Select(s => values[s]).ToList());
126 return string.Concat(Outputs.Select(s => values[s]));
127 }
128 public Arrival Arrivals(bool period) {
129 Dictionary<string, int> late = [], early = [];
130 Dictionary<string, string> previous = [];
131 if (period) {
132 foreach (Gate g in Flops)
133 late[g.Output] = early[g.Output] = g.Delay;
134 } else
135 foreach (string s in Inputs)
136 late[s] = early[s] = 0;
137 foreach (Gate g in Order) {
138 List<string> a = g.Inputs.Where(late.ContainsKey).ToList();
139 if (a.Count == 0)
140 continue;
141 string best = a[0];
142 foreach (string s in a)
143 if (late[s] > late[best])
144 best = s;
145 late[g.Output] = late[best] + g.Delay;
146 early[g.Output] = a.Min(s => early[s]) + g.Delay;
147 previous[g.Output] = best;
148 }
149 return new(late, early, previous);
150 }

154 sealed class Simulation {
155 public Circuit Circuit;
156 public List<string> Watch = [];
157 public Dictionary<string, char> Values = [];

159 public PriorityQueue<Event, (int, long)> Queue = new();
160 public int Time;
161 public long Sequence, Processed;
162 public bool Halted;
163 public Simulation(Circuit c) {
164 Circuit = c;
165 Reset();
166 }
167 public void Reset() {
168 Values = Circuit.Signals.ToDictionary(s => s,
169 _ => 'X');
170 History = Circuit.Signals.ToDictionary(s => s,
171 _ => new List<Change>());
172 Queue.Clear();
173 Time = 0;
174 Sequence = Processed = 0;
175 Halted = false;
176 }
177 public void Schedule(int t, string s, char v) => Queue.Enqueue(new(t, s, v),
178 (t, Sequence++));
179 public List<string>
180 Describe(IEnumerable<string> a) => a.Select(s => $"{s}={Values[s]}").ToList();
181 public void Run(int end) {
182 int total = 0, glitches = 0;
183 while (Queue.TryPeek(out Event first, out _) && first.Time < end) {
184 int t = first.Time, count = 0;
185 Dictionary<string, char> original = [];
186 while (Queue.TryPeek(out Event next, out _) && next.Time == t) {
187 if (count == 1000) {
188 Emit($"halted: oscillation at time {t}");
189 Queue.Clear();
190 Time = t;
191 Halted = true;
192 return;
193 }
194 Event e = Queue.Dequeue();
195 count++;
196 Processed++;
197 char old = Values[e.Signal];
198 if (old == e.Value)
199 continue;
200 original.TryAdd(e.Signal, old);
201 Values[e.Signal] = e.Value;
202 foreach (Gate g in Circuit.Readers[e.Signal])
203 if (g.Kind != "DFF")
204 Schedule(t + g.Delay, g.Output,
205 Logic(g.Kind, g.Inputs.Select(s => Values[s]).ToList()));
206 else if (e.Signal == g.Inputs[0] && old == '0' && e.Value == '1')
207 Schedule(t + g.Delay, g.Output, Values[g.Inputs[1]]);
208 }
209 total += count;
210 foreach (var (s, v) in original)
211 if (Values[s] == v)
212 glitches++;
213 else
214 History[s].Add(new(t, Values[s]));
215 foreach (string s in Watch)
216 if (original.TryGetValue(s, out char v) && Values[s] != v)
217 Emit($"{t} {s} {Values[s]}");
218 }
219 Time = end;
220 Emit($"run {end} events {total} glitches {glitches} pending {Queue.Count}");
221 }
222 }

244 public void Define(int header, List<string> lines) {
245 int line = header;
246 try {
247 string[] w = lines[0].Split(' ');
248 if (w.Length < 5 || w[2] != "IN" || w.Skip(3).Count(s => s == "OUT") != 1 ||
249 w[^1] == "OUT")
250 Fail("bad module header");
251 string name = w[1];
252 if (!Regex.IsMatch(name, "\\A[A-Z][A-Z0-9_]{0,15}\\z"))
253 Fail("bad module name " + name);
254 if (modules.ContainsKey(name))
255 Fail("duplicate module " + name);
256 int split = Array.IndexOf(w, "OUT", 3);
257 List<string> inputs = w.Skip(3).Take(split - 3).ToList(),
258 outputs = w.Skip(split + 1).ToList();
259 HashSet<string> seen = [];
260 foreach (string p in inputs.Concat(outputs)) {
261 if (!NameOk(p))
262 Fail("bad name " + p);
263 if (!seen.Add(p))
264 Fail("duplicate port " + p);
265 }
266 HashSet<string> driven = [..inputs], used = [];
267 List<Item> body = [];
268 List<(int Line, string Signal)> reads = [];
269 int gates = 0;
270 for (int i = 1; i < lines.Count; i++) {
271 line = header + i;
272 string[] a = lines[i].Split(' ');
273 string kind = a[0];
274 if (Types.Contains(kind)) {
275 if (a.Length < 5 || a[^3] != "->" || a.Count(s => s == "->") != 1)
276 Fail("bad gate syntax");
277 string n = a[1];
278 if (!NameOk(n))
279 Fail("bad name " + n);
280 List<string> ins = a.Skip(2).Take(a.Length - 5).ToList();
281 string output = a[^2], delay = a[^1];
282 int lo = kind is "NOT" or "BUF" ? 1
283 : kind == "MUX2" ? 3
284 : 2,
285 hi = Array.IndexOf(Types, kind) < 6 ? 8 : lo;
286 if (ins.Count < lo || ins.Count > hi)
287 Fail($"bad input count {ins.Count} for {kind}");
288 foreach (string s in ins.Append(output))
289 if (!NameOk(s))
290 Fail("bad name " + s);
291 if (!NumberOk(delay) || int.Parse(delay) > 1000)
292 Fail("bad delay " + delay);
293 if (used.Contains(n))
294 Fail("duplicate name " + n);
295 if (driven.Contains(output))
296 Fail("multiple drivers for " + output);
297 used.Add(n);
298 driven.Add(output);
299 body.Add(new Gate(n, kind, ins, output, int.Parse(delay)));
300 gates++;
301 reads.AddRange(ins.Select(s => (line, s)));
302 } else if (kind == "INST") {
303 if (a.Length < 3)
304 Fail("bad instance syntax");
305 string n = a[1];
306 if (!NameOk(n))
307 Fail("bad name " + n);
308 Module m = Lookup(a[2]);
309 List<Connection> connections = [];
310 HashSet<string> ports = [];
311 foreach (string word in a.Skip(3)) {
312 string[] p = word.Split('=');
313 if (p.Length != 2 || p[0] == "" || p[1] == "")
314 Fail("bad connection " + word);
315 if (!m.Inputs.Contains(p[0]) && !m.Outputs.Contains(p[0]))
316 Fail("unknown port " + p[0]);
317 if (ports.Contains(p[0]))
318 Fail("duplicate connection " + p[0]);
319 if (!NameOk(p[1]))
320 Fail("bad name " + p[1]);
321 ports.Add(p[0]);
322 connections.Add(new(p[0], p[1]));
323 }
324 foreach (string p in m.Inputs)
325 if (!ports.Contains(p))
326 Fail("unconnected input " + p);
327 if (used.Contains(n))
328 Fail("duplicate name " + n);
329 foreach (Connection c in connections)
330 if (m.Outputs.Contains(c.Port) && !driven.Add(c.Signal))
331 Fail("multiple drivers for " + c.Signal);
332 used.Add(n);
333 body.Add(new Instance(n, m, connections));
334 reads.AddRange(connections.Where(c => m.Inputs.Contains(c.Port))
335 .Select(c => (line, c.Signal)));
336 } else
337 Fail("unknown statement " + kind);
338 }
339 reads.AddRange(outputs.Select(s => (header, s)));
340 foreach (var (n, s) in reads)
341 if (!driven.Contains(s)) {
342 line = n;
343 Fail("undriven signal " + s);
344 }
345 Module module = new(name, inputs, outputs, body);
346 modules.Add(name, module);
347 moduleOrder.Add(module);
348 Emit(
349 $"defined {name} inputs {inputs.Count} outputs {outputs.Count} gates {gates} instances {body.Count-gates}");
350 } catch (ArgumentException e) {
351 Emit($"error: line {line}: {e.Message}");
352 }
353 }

355 foreach (Instance x in m.Body.OfType<Instance>()) {
356 Emit(new string(' ', 2 * depth) + x.Name + " " + x.Module.Name);
357 Hierarchy(x.Module, depth + 1);
358 }

360 public void Execute(int line, string[] w) {
361 try {
362 string cmd = w[0];
363 List<string> a = w.Skip(1).ToList();
364 if (!Counts.TryGetValue(cmd, out int n))
365 Fail("unknown command " + cmd);
366 if (a.Count < n || (!(cmd is "WATCH" or "UNWATCH" or "SHOW") && a.Count != n))
367 Fail("wrong argument count");
368 if (SimulationCommands.Contains(cmd) && sim is null)
369 Fail("no circuit");
370 if ((cmd is "SET" or "APPLY" or "PULSE" or "CLOCK" or "RUN") && sim!.Halted)
371 Fail("simulation halted");
372 if (cmd == "MODULES") {
373 Emit($"modules {modules.Count}");
374 foreach (Module m in moduleOrder)
375 Emit(
376 $"{m.Name} inputs {m.Inputs.Count} outputs {m.Outputs.Count} gates {new Circuit(m).Gates.Count}");
377 return;
378 }
379 if (cmd is "BUILD" or "NETLIST" or "HIERARCHY" or "STATS" or "TRUTHTABLE" or
380 "EQUIV" or "FAULTS" or "CRITICAL" or "PERIOD") {
381 Module m = Lookup(a[0]);
382 Module? m2 = cmd == "EQUIV" ? Lookup(a[1]) : null;
383 Circuit c = new(m);
384 switch (cmd) {
385 case "BUILD":
386 sim = new(c);
387 Emit($"built {c.Name} signals {c.Signals.Count} gates {c.Gates.Count}");
388 break;
389 case "NETLIST":
390 Emit($"netlist {c.Name} gates {c.Gates.Count} signals {c.Signals.Count}");
391 foreach (Gate g in c.Gates)
392 Emit(
393 $"{g.Name} {g.Kind} {string.Join(' ',g.Inputs)} -> {g.Output} {g.Delay}");
394 break;
395 case "HIERARCHY":
396 Emit(c.Name);
397 Hierarchy(m, 1);
398 break;
399 case "STATS":
400 Dictionary<string, int> levels = c.Signals.ToDictionary(x => x,
401 _ => 0);
402 int depth = 0;
403 foreach (Gate g in c.Order) {
404 levels[g.Output] = g.Inputs.Max(x => levels[x]) + 1;
405 depth = Math.Max(depth, levels[g.Output]);
406 }
407 Emit(
408 $"stats {c.Name} gates {c.Gates.Count} signals {c.Signals.Count} depth {(c.Loop?"-":depth.ToString())}");
409 List<string> text = ["types"];
410 foreach (string t in Types) {
411 int count = c.Gates.Count(g => g.Kind == t);
412 if (count > 0) {
413 text.Add(t);
414 text.Add(count.ToString());
415 }
416 }
417 Emit(string.Join(' ', text));
418 string best = c.Signals[0];
419 foreach (string x in c.Signals)
420 if (c.Readers[x].Count > c.Readers[best].Count)
421 best = x;
422 Emit($"max fanout {best} {c.Readers[best].Count}");
423 break;
424 case "CRITICAL":
425 case "PERIOD":
426 bool period = cmd == "PERIOD";
427 if (period && c.Flops.Count == 0)
428 Fail(c.Name + " has no flip-flops");
429 if (c.Loop)
430 Fail(c.Name + " has a combinational loop");
431 Arrival arr = c.Arrivals(period);
432 List<string> ends =
433 period ? c.Flops.Select(g => g.Inputs[1]).ToList() : c.Outputs;
434 string heading = period ? $"flip-flops {c.Flops.Count} period" : "critical";
435 string? endpoint = null;
436 foreach (string x in ends) {
437 if (!period)
438 Emit(arr.Late.ContainsKey(x)
439 ? $"{x} min {arr.Early[x]} max {arr.Late[x]}"
440 : x + " unreachable");
441 if (arr.Late.ContainsKey(x) &&
442 (endpoint is null || arr.Late[x] > arr.Late[endpoint]))
443 endpoint = x;
444 }
445 if (endpoint is null)
446 Emit(heading + " none");
447 else {
448 List<string> path = [endpoint];while(arr.Previous.TryGetValue(path[^1],out string? previous))
449 path.Add(previous);
450 path.Reverse();
451 Emit($"{heading} {arr.Late[endpoint]}: {string.Join(" -> ",path)}");
452 }
453 break;
454 default:
455 c.Combinational();
456 List<string> rows = Enumerable.Range(0, 1 << c.Inputs.Count)
457 .Select(r => Bits(r, c.Inputs.Count))
458 .ToList(),
459 expected = rows.Select(b => c.Evaluate(b)).ToList();
460 if (cmd == "TRUTHTABLE") {
461 Emit(string.Join(' ', c.Inputs.Append("|").Concat(c.Outputs)));
462 for (int r = 0; r < rows.Count; r++)
463 Emit(string.Join(
464 ' ', rows[r]
465 .Select(x => x.ToString())
466 .Append("|")
467 .Concat(expected[r].Select(x => x.ToString()))));
468 } else if (cmd == "EQUIV") {
469 Circuit d = new(m2!);
470 d.Combinational();
471 if (c.Inputs.Count != d.Inputs.Count ||
472 c.Outputs.Count != d.Outputs.Count)
473 Fail("port mismatch");
474 for (int r = 0; r < rows.Count; r++) {
475 string v = d.Evaluate(rows[r]);
476 if (v != expected[r]) {
477 Emit($"not equivalent at {rows[r]}: {expected[r]} vs {v}");
478 return;
479 }
480 }
481 Emit("equivalent");
482 } else {
483 List<string> undetected = [];
484 SortedSet<int> tests = [];
485 foreach (string x in c.Signals)
486 foreach (char v in "01") {
487 int found = -1;
488 for (int r = 0; r < rows.Count; r++)
489 if (c.Evaluate(rows[r], x, v) != expected[r]) {
490 found = r;
491 break;
492 }
493 if (found < 0)
494 undetected.Add($"{x}/{v}");
495 else
496 tests.Add(found);
497 }
498 int total = 2 * c.Signals.Count;
499 Emit($"faults {total} detected {total-undetected.Count}");
500 foreach (string f in undetected)
501 Emit("undetected " + f);
502 Emit(string.Join(' ',
503 new[] { "tests" }.Concat(tests.Select(r => rows[r]))));
504 }
505 break;
506 }
507 return;
508 }
509 Simulation s = sim!;
510 switch (cmd) {
511 case "RESET":
512 s.Reset();
513 Emit("reset");
514 break;
515 case "SET":
516 int t = ParseTime(a[0]);
517 RequireInput(a[1]);
518 char v = Value(a[2]);
519 s.Schedule(t, a[1], v);
520 Emit($"set {a[1]} {v} at {t}");
521 break;
522 case "APPLY":
523 int applyTime = ParseTime(a[0]);
524 string b = a[1];
525 if (b.Length != s.Circuit.Inputs.Count || b.Any(x => !"01X".Contains(x)))
526 Fail("bad vector " + b);
527 for (int i = 0; i < b.Length; i++)
528 s.Schedule(applyTime, s.Circuit.Inputs[i], b[i]);
529 Emit($"apply {b} at {applyTime}");
530 break;
531 case "PULSE":
532 case "CLOCK":
533 RequireInput(a[0]);
534 int start = ParseTime(a[1]), width = Number(a[2], true);
535 if (cmd == "PULSE") {
536 s.Schedule(start, a[0], '1');
537 s.Schedule(start + width, a[0], '0');
538 Emit($"pulse {a[0]} {start} {start+width}");
539 } else {
540 int count = Number(a[3], true);
541 for (int i = 0; i < count; i++)
542 s.Schedule(start + i * width, a[0], i % 2 == 0 ? '0' : '1');
543 Emit($"clock {a[0]} events {count} last {start+(count-1)*width}");
544 }
545 break;
546 case "RUN":
547 int end = Number(a[0]);
548 if (end <= s.Time)
549 Fail($"time {end} is not after {s.Time}");
550 s.Run(end);
551 break;
552 case "WATCH":
553 case "UNWATCH":
554 HashSet<string> seen = [];
555 foreach (string x in a) {
556 RequireSignal(x);
557 bool watched = s.Watch.Contains(x);
558 if (cmd == "WATCH" ? (watched || seen.Contains(x))
559 : (!watched || seen.Contains(x)))
560 Fail((cmd == "WATCH" ? "already watched " : "not watched ") + x);
561 seen.Add(x);
562 }
563 if (cmd == "WATCH")
564 s.Watch.AddRange(a);
565 else
566 s.Watch.RemoveAll(seen.Contains);
567 Emit("watching " +
568 (s.Watch.Count == 0 ? "nothing" : string.Join(' ', s.Watch)));
569 break;
570 case "SHOW":
571 foreach (string x in a)
572 RequireSignal(x);
573 Emit(string.Join(' ', a.Count > 0
574 ? s.Describe(a)
575 : s.Describe(s.Circuit.Inputs)
576 .Append("|")
577 .Concat(s.Describe(s.Circuit.Outputs))));
578 break;
579 case "CHECK":
580 RequireSignal(a[0]);
581 char expectedValue = Value(a[1]);
582 Emit($"check {a[0]}={expectedValue} " +
583 (s.Values[a[0]] == expectedValue ? "ok" : $"failed ({s.Values[a[0]]})"));
584 break;
585 case "PROBE":
586 string signal = a[0];
587 RequireSignal(signal);
588 string driver = "input";if(s.Circuit.Driver.TryGetValue(signal,out Gate? gate))
589 driver =
590 $"{gate.Name} {gate.Kind} {string.Join(' ',s.Describe(gate.Inputs))} delay {gate.Delay}";
591 Emit($"{signal}={s.Values[signal]} driven by {driver}");
592 List<Gate> readers = s.Circuit.Readers[signal];
593 Emit(string.Join(' ', new[] { "fanout", readers.Count.ToString() }.Concat(
594 readers.Select(g => g.Name))));
595 break;
596 case "WAVE":
597 int first = Number(a[0]), last = Number(a[1]);
598 if (!(first < last && last <= s.Time))
599 Fail($"bad range {first} {last}");
600 if (s.Watch.Count == 0)
601 Fail("nothing watched");
602 foreach (string x in s.Watch) {
603 List<Change> h = s.History[x];
604 int i = 0;
605 char val = 'X';
606 StringBuilder wave = new();
607 for (int time = first; time < last; time++) {
608 while (i < h.Count && h[i].Time <= time)
609 val = h[i++].Value;
610 wave.Append(val);
611 }
612 Emit(x + " " + wave);
613 }
614 break;
615 case "HISTORY":
616 string name = a[0];
617 RequireSignal(name);
618 List<Change> history = s.History[name];
619 Emit(string.Join(
620 ' ', new[] { "history", name, "changes", history.Count.ToString() }.Concat(
621 history.Select(h => $"{h.Time}:{h.Value}"))));
622 break;
623 case "ACTIVITY":
624 int k = Number(a[0], true);
625 foreach (string x in s.Circuit.Signals
626 .OrderByDescending(x => s.History[x].Count)
627 .ThenBy(x => x, StringComparer.Ordinal)
628 .Take(k))
629 Emit($"{x} {s.History[x].Count}");
630 break;
631 case "STATUS":
632 Emit(
633 $"circuit {s.Circuit.Name} time {s.Time} pending {s.Queue.Count} events {s.Processed} halted {(s.Halted?"yes":"no")}");
634 Emit(
635 $"values 0:{s.Values.Values.Count(x=>x=='0')} 1:{s.Values.Values.Count(x=>x=='1')} X:{s.Values.Values.Count(x=>x=='X')}");
636 break;
637 }
638 } catch (ArgumentException e) {
639 Emit($"error: line {line}: {e.Message}");
640 }
641 }
642 }
643 static void Main() {
644 List<string> lines = [];
645 for (string? line; (line = Console.ReadLine()) is not null;)

649 string[] w = lines[i].Split(' ');

651 int end = i + 1;
652 while (lines[end] != "ENDMODULE")
653 end++;
654 simulator.Define(i + 1, lines.GetRange(i, end - i));
655 i = end;
656 } else
657 simulator.Execute(i + 1, w);
658 }
