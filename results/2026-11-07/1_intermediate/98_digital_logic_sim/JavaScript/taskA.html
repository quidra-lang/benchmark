51function logic(kind, a) {
52 let y;
53 if (["AND", "NAND"].includes(kind)) y = a.includes("0") ? "0" : a.includes("X") ? "X" : "1";
54 else if (["OR", "NOR"].includes(kind)) y = a.includes("1") ? "1" : a.includes("X") ? "X" : "0";
55 else if (["XOR", "XNOR"].includes(kind))
56 y = a.includes("X") ? "X" : String(a.filter((x) => x === "1").length % 2);
57 else if (kind === "MUX2")
58 y = a[0] === "X" ? (a[1] === a[2] ? a[1] : "X") : a[a[0] === "0" ? 1 : 2];
59 else y = a[0];
60 return ["NAND", "NOR", "XNOR", "NOT"].includes(kind) ? { 0: "1", 1: "0", X: "X" }[y] : y;
61}
62class Circuit {
63 constructor(module) {
64 this.name = module.name;
65 this.inputs = module.inputs;
66 this.outputs = module.outputs;
67 this.gates = [];
68 const expand = (m, p, map) => {
69 const actual = (s) => map.get(s) ?? p + s;
70 for (const item of m.body) {
71 if (item.kind)
72 this.gates.push({
73 ...item,
74 name: p + item.name,
75 inputs: item.inputs.map(actual),
76 output: actual(item.output),
77 });
78 else
79 expand(
80 item.module,
81 p + item.name + ".",
82 new Map(item.connections.map(([x, y]) => [x, actual(y)])),
83 );
84 }
85 };
86 expand(module, "", new Map());
87 this.signals = [...this.inputs, ...this.gates.map((g) => g.output)];
88 this.driver = new Map(this.gates.map((g) => [g.output, g]));
89 this.readers = new Map(this.signals.map((s) => [s, []]));
90 for (const g of this.gates) for (const s of new Set(g.inputs)) this.readers.get(s).push(g);
91 this.flops = this.gates.filter((g) => g.kind === "DFF");
92 const waiting = new Map(),
93 ready = [];
94 for (const g of this.gates.filter((g) => g.kind !== "DFF")) {
95 const n = new Set(
96 g.inputs.filter((s) => this.driver.has(s) && this.driver.get(s).kind !== "DFF"),
97 ).size;
98 waiting.set(g, n);
99 if (!n) ready.push(g);
100 }
101 this.order = [];
102 while (ready.length) {
103 const g = ready.pop();
104 this.order.push(g);
105 for (const r of this.readers.get(g.output))
106 if (r.kind !== "DFF") {
107 waiting.set(r, waiting.get(r) - 1);
108 if (!waiting.get(r)) ready.push(r);
109 }
110 }
111 this.loop = this.order.length !== waiting.size;
112 }
113 combinational() {
114 if (this.flops.length) fail(`${this.name} is not combinational`);
115 if (this.loop) fail(`${this.name} has a combinational loop`);
116 if (this.inputs.length > 8) fail(`too many inputs ${this.inputs.length}`);
117 return this;
118 }
119 evaluate(b, stuck = "", v = "X") {
120 const vals = new Map(this.inputs.map((s, i) => [s, s === stuck ? v : b[i]]));
121 for (const g of this.order)
122 vals.set(
123 g.output,
124 g.output === stuck
125 ? v
126 : logic(
127 g.kind,
128 g.inputs.map((s) => vals.get(s)),
129 ),
130 );
131 return this.outputs.map((s) => vals.get(s)).join("");
132 }
133 arrivals(period) {
134 const late = new Map(),
135 early = new Map(),
136 prev = new Map();
137 if (period)
138 for (const g of this.flops) {
139 late.set(g.output, g.delay);
140 early.set(g.output, g.delay);
141 }
142 else
143 for (const s of this.inputs) {
144 late.set(s, 0);
145 early.set(s, 0);
146 }
147 for (const g of this.order) {
148 const a = g.inputs.filter((s) => late.has(s));
149 if (!a.length) continue;
150 let best = a[0];
151 for (const s of a) if (late.get(s) > late.get(best)) best = s;
152 late.set(g.output, late.get(best) + g.delay);
153 early.set(g.output, Math.min(...a.map((s) => early.get(s))) + g.delay);
154 prev.set(g.output, best);
155 }
156 return { late, early, prev };
157 }
158}
159class Simulation {
160 constructor(c) {
161 this.c = c;
162 this.watch = [];
163 this.reset();
164 }
165 reset() {
166 this.vals = new Map(this.c.signals.map((s) => [s, "X"]));
167 this.time = 0;
168 this.queue = [];
169 this.sequence = 0;
170 this.history = new Map(this.c.signals.map((s) => [s, []]));
171 this.processed = 0;
172 this.halted = false;
173 }
174 schedule(t, s, v) {
175 const event = { t, s, v, id: this.sequence++ };
176 let i = this.queue.length;
177 this.queue.push(event);
178 while (i > 0) {
179 const p = (i - 1) >> 1;
180 if (!this.before(event, this.queue[p])) break;
181 this.queue[i] = this.queue[p];
182 i = p;
183 }
184 this.queue[i] = event;
185 }
186 before(a, b) {
187 return a.t < b.t || (a.t === b.t && a.id < b.id);
188 }
189 pop() {
190 const result = this.queue[0],
191 last = this.queue.pop();
192 if (this.queue.length) {
193 let i = 0;
194 while (2 * i + 1 < this.queue.length) {
195 let j = 2 * i + 1;
196 if (j + 1 < this.queue.length && this.before(this.queue[j + 1], this.queue[j])) j++;
197 if (!this.before(this.queue[j], last)) break;
198 this.queue[i] = this.queue[j];
199 i = j;
200 }
201 this.queue[i] = last;
202 }
203 return result;
204 }
205 run(end) {
206 let total = 0,
207 glitches = 0;
208 while (this.queue.length && this.queue[0].t < end) {
209 const t = this.queue[0].t,
210 start = new Map(this.vals),
211 changed = new Set();
212 let count = 0;
213 while (this.queue.length && this.queue[0].t === t) {
214 if (count === 1000) {
215 emit(`halted: oscillation at time ${t}`);
216 this.queue = [];
217 this.time = t;
218 this.halted = true;
219 return;
220 }
221 const e = this.pop();
222 count++;
223 this.processed++;
224 const old = this.vals.get(e.s);
225 if (old === e.v) continue;
226 this.vals.set(e.s, e.v);
227 changed.add(e.s);
228 for (const g of this.c.readers.get(e.s)) {
229 if (g.kind !== "DFF")
230 this.schedule(
231 t + g.delay,
232 g.output,
233 logic(
234 g.kind,
235 g.inputs.map((s) => this.vals.get(s)),
236 ),
237 );
238 else if (e.s === g.inputs[0] && old === "0" && e.v === "1")
239 this.schedule(t + g.delay, g.output, this.vals.get(g.inputs[1]));
240 }
241 }
242 total += count;
243 for (const s of changed)
244 if (this.vals.get(s) === start.get(s)) glitches++;
245 else this.history.get(s).push([t, this.vals.get(s)]);
246 for (const s of this.watch)
247 if (this.vals.get(s) !== start.get(s)) emit(`${t} ${s} ${this.vals.get(s)}`);
248 }
249 this.time = end;
250 emit(`run ${end} events ${total} glitches ${glitches} pending ${this.queue.length}`);
251 }
252 describe(a) {
253 return a.map((s) => `${s}=${this.vals.get(s)}`);
254 }
255}

391 n = COUNTS[cmd];
