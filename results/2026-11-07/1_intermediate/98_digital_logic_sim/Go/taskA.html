46func bits(r, n int) string {
47 b := make([]byte, n)
48 for i := range b {
49 b[i] = byte('0' + ((r >> (n - 1 - i)) & 1))
50 }
51 return string(b)
52}
53func logic(k string, a []string) string {
54 y := a[0]
55 switch k {
56 case "AND", "NAND":
57 y = "1"
58 if has(a, "0") {
59 y = "0"
60 } else if has(a, "X") {
61 y = "X"
62 }
63 case "OR", "NOR":
64 y = "0"
65 if has(a, "1") {
66 y = "1"
67 } else if has(a, "X") {
68 y = "X"
69 }
70 case "XOR", "XNOR":
71 n := 0
72 for _, v := range a {
73 if v == "1" {
74 n++
75 }
76 }
77 y = strconv.Itoa(n % 2)
78 if has(a, "X") {
79 y = "X"
80 }
81 case "MUX2":
82 if a[0] == "0" {
83 y = a[1]
84 } else if a[0] == "1" {
85 y = a[2]
86 } else if a[1] == a[2] {
87 y = a[1]
88 } else {
89 y = "X"
90 }
91 }
92 if has([]string{"NAND", "NOR", "XNOR", "NOT"}, k) {
93 if y == "0" {
94 return "1"
95 }
96 if y == "1" {
97 return "0"
98 }
99 }
100 return y
101}

131func circuit(m *Module) *Circuit {
132 c := &Circuit{name: m.name, inputs: m.inputs, outputs: m.outputs, driver: map[string]int{}, readers: map[string][]int{}}
133 var expand func(*Module, string, map[string]string)
134 expand = func(m *Module, p string, ports map[string]string) {
135 actual := func(s string) string {
136 if x, ok := ports[s]; ok {
137 return x
138 }
139 return p + s
140 }
141 for _, x := range m.body {
142 if x.gate != nil {
143 g := *x.gate
144 g.name = p + g.name
145 g.inputs = make([]string, len(x.gate.inputs))
146 for i, s := range x.gate.inputs {
147 g.inputs[i] = actual(s)
148 }
149 g.output = actual(g.output)
150 c.gates = append(c.gates, g)
151 } else {
152 mapping := map[string]string{}
153 for _, v := range x.connections {
154 mapping[v.port] = actual(v.signal)
155 }
156 expand(x.module, p+x.name+".", mapping)
157 }
158 }
159 }
160 expand(m, "", map[string]string{})
161 c.signals = append(c.signals, m.inputs...)
162 for i, g := range c.gates {
163 c.signals = append(c.signals, g.output)
164 c.driver[g.output] = i
165 if g.kind == "DFF" {
166 c.flops = append(c.flops, i)
167 }
168 seen := map[string]bool{}
169 for _, s := range g.inputs {
170 if !seen[s] {
171 c.readers[s] = append(c.readers[s], i)
172 seen[s] = true
173 }
174 }
175 }
176 waiting := make([]int, len(c.gates))
177 ready := []int{}
178 needed := 0
179 for i, g := range c.gates {
180 if g.kind == "DFF" {
181 continue
182 }
183 needed++
184 seen := map[string]bool{}
185 for _, s := range g.inputs {
186 if d, ok := c.driver[s]; ok && c.gates[d].kind != "DFF" && !seen[s] {
187 waiting[i]++
188 seen[s] = true
189 }
190 }
191 if waiting[i] == 0 {
192 ready = append(ready, i)
193 }
194 }
195 for len(ready) > 0 {
196 i := ready[len(ready)-1]
197 ready = ready[:len(ready)-1]
198 c.order = append(c.order, i)
199 for _, r := range c.readers[c.gates[i].output] {
200 if c.gates[r].kind != "DFF" {
201 waiting[r]--
202 if waiting[r] == 0 {
203 ready = append(ready, r)
204 }
205 }
206 }
207 }
208 c.loop = len(c.order) != needed
209 return c
210}
211func (c *Circuit) combinational() {
212 if len(c.flops) > 0 {
213 fail(c.name + " is not combinational")
214 }
215 if c.loop {
216 fail(c.name + " has a combinational loop")
217 }
218 if len(c.inputs) > 8 {
219 fail(fmt.Sprintf("too many inputs %d", len(c.inputs)))
220 }
221}
222func (c *Circuit) evaluate(b, stuck, v string) string {
223 values := map[string]string{}
224 for i, s := range c.inputs {
225 values[s] = b[i : i+1]
226 if s == stuck {
227 values[s] = v
228 }
229 }
230 for _, i := range c.order {
231 g := c.gates[i]
232 a := []string{}
233 for _, s := range g.inputs {
234 a = append(a, values[s])
235 }
236 values[g.output] = logic(g.kind, a)
237 if g.output == stuck {
238 values[g.output] = v
239 }
240 }
241 a := []string{}
242 for _, s := range c.outputs {
243 a = append(a, values[s])
244 }
245 return strings.Join(a, "")
246}
247func (c *Circuit) arrivals(period bool) (map[string]int, map[string]int, map[string]string) {
248 late, early, prev := map[string]int{}, map[string]int{}, map[string]string{}
249 if period {
250 for _, i := range c.flops {
251 g := c.gates[i]
252 late[g.output] = g.delay
253 early[g.output] = g.delay
254 }
255 } else {
256 for _, s := range c.inputs {
257 late[s] = 0
258 early[s] = 0
259 }
260 }
261 for _, i := range c.order {
262 g := c.gates[i]
263 best := ""
264 minimum := 0
265 for _, s := range g.inputs {
266 if t, ok := late[s]; ok {
267 if best == "" {
268 best = s
269 minimum = early[s]
270 } else if t > late[best] {
271 best = s
272 }
273 if early[s] < minimum {
274 minimum = early[s]
275 }
276 }
277 }
278 if best != "" {
279 late[g.output] = late[best] + g.delay
280 early[g.output] = minimum + g.delay
281 prev[g.output] = best
282 }
283 }
284 return late, early, prev
285}

287type Event struct {
288 time, sequence int
289 signal, value string
290}
291type EventHeap []Event
292
293func (h EventHeap) Len() int { return len(h) }
294func (h EventHeap) Less(i, j int) bool {
295 return h[i].time < h[j].time || h[i].time == h[j].time && h[i].sequence < h[j].sequence
296}
297func (h EventHeap) Swap(i, j int) { h[i], h[j] = h[j], h[i] }
298func (h *EventHeap) Push(x any) { *h = append(*h, x.(Event)) }
299func (h *EventHeap) Pop() any { a := *h; x := a[len(a)-1]; *h = a[:len(a)-1]; return x }

315func (s *Simulation) reset() {
316 s.values = map[string]string{}
317 s.history = map[string][]Change{}
318 for _, x := range s.c.signals {
319 s.values[x] = "X"
320 s.history[x] = nil
321 }
322 s.time = 0
323 s.sequence = 0
324 s.processed = 0
325 s.queue = nil
326 s.halted = false
327}
328func (s *Simulation) schedule(t int, x, v string) {
329 heap.Push(&s.queue, Event{t, s.sequence, x, v})
330 s.sequence++
331}
332func (s *Simulation) describe(a []string) []string {
333 b := []string{}
334 for _, x := range a {
335 b = append(b, x+"="+s.values[x])

337 return b

339func (s *Simulation) run(end int) {
340 total, glitches := 0, 0
341 for len(s.queue) > 0 && s.queue[0].time < end {
342 t := s.queue[0].time
343 original := map[string]string{}
344 count := 0
345 for len(s.queue) > 0 && s.queue[0].time == t {
346 if count == 1000 {
347 emit(fmt.Sprintf("halted: oscillation at time %d", t))
348 s.queue = nil
349 s.time = t
350 s.halted = true
351 return
352 }
353 e := heap.Pop(&s.queue).(Event)
354 count++
355 s.processed++
356 old := s.values[e.signal]
357 if old == e.value {
358 continue
359 }
360 if _, ok := original[e.signal]; !ok {
361 original[e.signal] = old
362 }
363 s.values[e.signal] = e.value
364 for _, i := range s.c.readers[e.signal] {
365 g := s.c.gates[i]
366 if g.kind != "DFF" {
367 a := []string{}
368 for _, x := range g.inputs {
369 a = append(a, s.values[x])
370 }
371 s.schedule(t+g.delay, g.output, logic(g.kind, a))
372 } else if e.signal == g.inputs[0] && old == "0" && e.value == "1" {
373 s.schedule(t+g.delay, g.output, s.values[g.inputs[1]])
374 }
375 }
376 }
377 total += count
378 for x, v := range original {
379 if s.values[x] == v {
380 glitches++
381 } else {
382 s.history[x] = append(s.history[x], Change{t, s.values[x]})
383 }
384 }
385 for _, x := range s.watch {
386 if v, ok := original[x]; ok && v != s.values[x] {
387 emit(fmt.Sprintf("%d %s %s", t, x, s.values[x]))
388 }
389 }
390 }
391 s.time = end
392 emit(fmt.Sprintf("run %d events %d glitches %d pending %d", end, total, glitches, len(s.queue)))
393}

425func (s *Simulator) define(header int, lines []string) {
426 line := header
427 defer func() {
428 if e := recover(); e != nil {
429 emit(fmt.Sprintf("error: line %d: %s", line, e))
430 }
431 }()
432 w := strings.Split(lines[0], " ")
433 outCount, split := 0, 0
434 for i := 3; i < len(w); i++ {
435 if w[i] == "OUT" {
436 outCount++
437 split = i
438 }
439 }
440 if len(w) < 5 || w[2] != "IN" || outCount != 1 || w[len(w)-1] == "OUT" {
441 fail("bad module header")
442 }
443 n := w[1]
444 if !moduleRE.MatchString(n) {
445 fail("bad module name " + n)
446 }
447 if _, ok := s.modules[n]; ok {
448 fail("duplicate module " + n)
449 }
450 m := &Module{name: n, inputs: w[3:split], outputs: w[split+1:]}
451 seen := map[string]bool{}
452 ports := append(append([]string{}, m.inputs...), m.outputs...)
453 for _, p := range ports {
454 if !nameRE.MatchString(p) {
455 fail("bad name " + p)
456 }
457 if seen[p] {
458 fail("duplicate port " + p)
459 }
460 seen[p] = true
461 }
462 driven, used := map[string]bool{}, map[string]bool{}
463 for _, p := range m.inputs {
464 driven[p] = true
465 }
466 type Read struct {
467 line int
468 signal string
469 }
470 reads := []Read{}
471 gates := 0
472 for i, text := range lines[1:] {
473 line = header + i + 1
474 a := strings.Split(text, " ")
475 kind := a[0]
476 if has(types, kind) {
477 arrows := 0
478 for _, x := range a {
479 if x == "->" {
480 arrows++
481 }
482 }
483 if len(a) < 5 || a[len(a)-3] != "->" || arrows != 1 {
484 fail("bad gate syntax")
485 }
486 name := a[1]
487 if !nameRE.MatchString(name) {
488 fail("bad name " + name)
489 }
490 ins := a[2 : len(a)-3]
491 output, delay := a[len(a)-2], a[len(a)-1]
492 lo, hi := 2, 8
493 if kind == "NOT" || kind == "BUF" {
494 lo = 1
495 hi = 1
496 } else if kind == "MUX2" {
497 lo = 3
498 hi = 3
499 } else if kind == "DFF" {
500 hi = 2
501 }
502 if len(ins) < lo || len(ins) > hi {
503 fail(fmt.Sprintf("bad input count %d for %s", len(ins), kind))
504 }
505 for _, x := range append(append([]string{}, ins...), output) {
506 if !nameRE.MatchString(x) {
507 fail("bad name " + x)
508 }
509 }
510 if !numberRE.MatchString(delay) {
511 fail("bad delay " + delay)
512 }
513 d, _ := strconv.Atoi(delay)
514 if d > 1000 {
515 fail("bad delay " + delay)
516 }
517 if used[name] {
518 fail("duplicate name " + name)
519 }
520 if driven[output] {
521 fail("multiple drivers for " + output)
522 }
523 used[name] = true
524 driven[output] = true
525 m.body = append(m.body, Item{gate: &Gate{name, kind, ins, output, d}})
526 gates++
527 for _, x := range ins {
528 reads = append(reads, Read{line, x})
529 }
530 } else if kind == "INST" {
531 if len(a) < 3 {
532 fail("bad instance syntax")
533 }
534 name := a[1]
535 if !nameRE.MatchString(name) {
536 fail("bad name " + name)
537 }
538 child := s.lookup(a[2])
539 connections := []Connection{}
540 seen := map[string]bool{}
541 for _, w := range a[3:] {
542 p := strings.Split(w, "=")
543 if len(p) != 2 || p[0] == "" || p[1] == "" {
544 fail("bad connection " + w)
545 }
546 port, x := p[0], p[1]
547 if !has(child.inputs, port) && !has(child.outputs, port) {
548 fail("unknown port " + port)
549 }
550 if seen[port] {
551 fail("duplicate connection " + port)
552 }
553 if !nameRE.MatchString(x) {
554 fail("bad name " + x)
555 }
556 seen[port] = true
557 connections = append(connections, Connection{port, x})
558 }
559 for _, p := range child.inputs {
560 if !seen[p] {
561 fail("unconnected input " + p)
562 }
563 }
564 if used[name] {
565 fail("duplicate name " + name)
566 }
567 for _, v := range connections {
568 if has(child.outputs, v.port) {
569 if driven[v.signal] {
570 fail("multiple drivers for " + v.signal)
571 }
572 driven[v.signal] = true
573 }
574 }
575 used[name] = true
576 m.body = append(m.body, Item{name: name, module: child, connections: connections})
577 for _, v := range connections {
578 if has(child.inputs, v.port) {
579 reads = append(reads, Read{line, v.signal})
580 }
581 }
582 } else {
583 fail("unknown statement " + kind)
584 }
585 }
586 for _, x := range m.outputs {
587 reads = append(reads, Read{header, x})
588 }
589 for _, r := range reads {
590 if !driven[r.signal] {
591 line = r.line
592 fail("undriven signal " + r.signal)
593 }
594 }
595 s.modules[n] = m
596 s.order = append(s.order, m)
597 emit(fmt.Sprintf("defined %s inputs %d outputs %d gates %d instances %d", n, len(m.inputs), len(m.outputs), gates, len(m.body)-gates))
598}
599func hierarchy(m *Module, depth int) {
600 for _, x := range m.body {
601 if x.gate == nil {
602 emit(strings.Repeat(" ", depth) + x.name + " " + x.module.name)
603 hierarchy(x.module, depth+1)
604 }
605 }
606}
607func (s *Simulator) execute(line int, w []string) {
608 defer func() {
609 if e := recover(); e != nil {
610 emit(fmt.Sprintf("error: line %d: %s", line, e))
611 }
612 }()
613 cmd, a := w[0], w[1:]
614 n, ok := counts[cmd]
615 if !ok {
616 fail("unknown command " + cmd)
617 }
618 if len(a) < n || (!has([]string{"WATCH", "UNWATCH", "SHOW"}, cmd) && len(a) != n) {
619 fail("wrong argument count")
620 }
621 simulation := strings.Fields("RESET SET APPLY PULSE CLOCK RUN WATCH UNWATCH SHOW CHECK PROBE WAVE HISTORY ACTIVITY STATUS")
622 if has(simulation, cmd) && s.sim == nil {
623 fail("no circuit")
624 }
625 sim := s.sim
626 if has([]string{"SET", "APPLY", "PULSE", "CLOCK", "RUN"}, cmd) && sim.halted {
627 fail("simulation halted")
628 }
629 if cmd == "MODULES" {
630 emit(fmt.Sprintf("modules %d", len(s.order)))
631 for _, m := range s.order {
632 emit(fmt.Sprintf("%s inputs %d outputs %d gates %d", m.name, len(m.inputs), len(m.outputs), len(circuit(m).gates)))
633 }
634 return
635 }
636 if has(strings.Fields("BUILD NETLIST HIERARCHY STATS TRUTHTABLE EQUIV FAULTS CRITICAL PERIOD"), cmd) {
637 m := s.lookup(a[0])
638 var m2 *Module
639 if cmd == "EQUIV" {
640 m2 = s.lookup(a[1])
641 }
642 c := circuit(m)
643 switch cmd {
644 case "BUILD":
645 s.sim = &Simulation{c: c}
646 s.sim.reset()
647 emit(fmt.Sprintf("built %s signals %d gates %d", c.name, len(c.signals), len(c.gates)))
648 case "NETLIST":
649 emit(fmt.Sprintf("netlist %s gates %d signals %d", c.name, len(c.gates), len(c.signals)))
650 for _, g := range c.gates {
651 emit(fmt.Sprintf("%s %s %s -> %s %d", g.name, g.kind, strings.Join(g.inputs, " "), g.output, g.delay))
652 }
653 case "HIERARCHY":
654 emit(m.name)
655 hierarchy(m, 1)
656 case "STATS":
657 levels := map[string]int{}
658 depth := 0
659 for _, i := range c.order {
660 g := c.gates[i]
661 v := 0
662 for _, x := range g.inputs {
663 if levels[x] > v {
664 v = levels[x]
665 }
666 }
667 levels[g.output] = v + 1
668 if v+1 > depth {
669 depth = v + 1
670 }
671 }
672 d := strconv.Itoa(depth)
673 if c.loop {
674 d = "-"
675 }
676 emit(fmt.Sprintf("stats %s gates %d signals %d depth %s", c.name, len(c.gates), len(c.signals), d))
677 words := []string{"types"}
678 for _, k := range types {
679 n := 0
680 for _, g := range c.gates {
681 if g.kind == k {
682 n++
683 }
684 }
685 if n > 0 {
686 words = append(words, k, strconv.Itoa(n))
687 }
688 }
689 emit(strings.Join(words, " "))
690 best := c.signals[0]
691 for _, x := range c.signals {
692 if len(c.readers[x]) > len(c.readers[best]) {
693 best = x
694 }
695 }
696 emit(fmt.Sprintf("max fanout %s %d", best, len(c.readers[best])))
697 case "CRITICAL", "PERIOD":
698 period := cmd == "PERIOD"
699 if period && len(c.flops) == 0 {
700 fail(c.name + " has no flip-flops")
701 }
702 if c.loop {
703 fail(c.name + " has a combinational loop")
704 }
705 late, early, prev := c.arrivals(period)
706 ends := c.outputs
707 heading := "critical"
708 if period {
709 ends = nil
710 for _, i := range c.flops {
711 ends = append(ends, c.gates[i].inputs[1])
712 }
713 heading = fmt.Sprintf("flip-flops %d period", len(c.flops))
714 }
715 best := ""
716 for _, x := range ends {
717 t, ok := late[x]
718 if !period {
719 if ok {
720 emit(fmt.Sprintf("%s min %d max %d", x, early[x], t))
721 } else {
722 emit(x + " unreachable")
723 }
724 }
725 if ok && (best == "" || t > late[best]) {
726 best = x
727 }
728 }
729 if best == "" {
730 emit(heading + " none")
731 } else {
732 path := []string{best}
733 for x := prev[best]; x != ""; x = prev[x] {
734 path = append(path, x)
735 }
736 for i, j := 0, len(path)-1; i < j; i, j = i+1, j-1 {
737 path[i], path[j] = path[j], path[i]
738 }
739 emit(fmt.Sprintf("%s %d: %s", heading, late[best], strings.Join(path, " -> ")))
740 }
741 default:
742 c.combinational()
743 rows, expected := []string{}, []string{}
744 for r := 0; r < 1<<len(c.inputs); r++ {
745 b := bits(r, len(c.inputs))
746 rows = append(rows, b)
747 expected = append(expected, c.evaluate(b, "", ""))
748 }
749 if cmd == "TRUTHTABLE" {
750 words := append(append(append([]string{}, c.inputs...), "|"), c.outputs...)
751 emit(strings.Join(words, " "))
752 for r, b := range rows {
753 words := []string{}
754 for _, x := range b {
755 words = append(words, string(x))
756 }
757 words = append(words, "|")
758 for _, x := range expected[r] {
759 words = append(words, string(x))
760 }
761 emit(strings.Join(words, " "))
762 }
763 } else if cmd == "EQUIV" {
764 d := circuit(m2)
765 d.combinational()
766 if len(c.inputs) != len(d.inputs) || len(c.outputs) != len(d.outputs) {
767 fail("port mismatch")
768 }
769 for r, b := range rows {
770 v := d.evaluate(b, "", "")
771 if expected[r] != v {
772 emit(fmt.Sprintf("not equivalent at %s: %s vs %s", b, expected[r], v))
773 return
774 }
775 }
776 emit("equivalent")
777 } else {
778 undetected := []string{}
779 tests := map[int]bool{}
780 for _, x := range c.signals {
781 for _, v := range []string{"0", "1"} {
782 found := -1
783 for r, b := range rows {
784 if c.evaluate(b, x, v) != expected[r] {
785 found = r
786 break
787 }
788 }
789 if found < 0 {
790 undetected = append(undetected, x+"/"+v)
791 } else {
792 tests[found] = true
793 }
794 }
795 }
796 emit(fmt.Sprintf("faults %d detected %d", 2*len(c.signals), 2*len(c.signals)-len(undetected)))
797 for _, f := range undetected {
798 emit("undetected " + f)
799 }
800 words := []string{"tests"}
801 for r, b := range rows {
802 if tests[r] {
803 words = append(words, b)
804 }
805 }
806 emit(strings.Join(words, " "))
807 }
808 }
809 return
810 }
811 switch cmd {
812 case "RESET":
813 sim.reset()
814 emit("reset")
815 case "SET":
816 t := s.time(a[0])
817 s.input(a[1])
818 value(a[2])
819 sim.schedule(t, a[1], a[2])
820 emit(fmt.Sprintf("set %s %s at %d", a[1], a[2], t))
821 case "APPLY":
822 t := s.time(a[0])
823 b := a[1]
824 if len(b) != len(sim.c.inputs) || strings.Trim(b, "01X") != "" {
825 fail("bad vector " + b)
826 }
827 for i, x := range sim.c.inputs {
828 sim.schedule(t, x, b[i:i+1])
829 }
830 emit(fmt.Sprintf("apply %s at %d", b, t))
831 case "PULSE", "CLOCK":
832 s.input(a[0])
833 t, width := s.time(a[1]), number(a[2], true)
834 if cmd == "PULSE" {
835 sim.schedule(t, a[0], "1")
836 sim.schedule(t+width, a[0], "0")
837 emit(fmt.Sprintf("pulse %s %d %d", a[0], t, t+width))
838 } else {
839 count := number(a[3], true)
840 for i := 0; i < count; i++ {
841 sim.schedule(t+i*width, a[0], strconv.Itoa(i%2))
842 }
843 emit(fmt.Sprintf("clock %s events %d last %d", a[0], count, t+(count-1)*width))
844 }
845 case "RUN":
846 t := number(a[0], false)
847 if t <= sim.time {
848 fail(fmt.Sprintf("time %d is not after %d", t, sim.time))
849 }
850 sim.run(t)
851 case "WATCH", "UNWATCH":
852 seen := map[string]bool{}
853 for _, x := range a {
854 s.signal(x)
855 watched := has(sim.watch, x)
856 if cmd == "WATCH" && (watched || seen[x]) {
857 fail("already watched " + x)
858 }
859 if cmd == "UNWATCH" && (!watched || seen[x]) {
860 fail("not watched " + x)
861 }
862 seen[x] = true
863 }
864 if cmd == "WATCH" {
865 sim.watch = append(sim.watch, a...)
866 } else {
867 remaining := []string{}
868 for _, x := range sim.watch {
869 if !seen[x] {
870 remaining = append(remaining, x)
871 }
872 }
873 sim.watch = remaining
874 }
875 text := strings.Join(sim.watch, " ")
876 if text == "" {
877 text = "nothing"
878 }
879 emit("watching " + text)
880 case "SHOW":
881 for _, x := range a {
882 s.signal(x)
883 }
884 words := sim.describe(a)
885 if len(a) == 0 {
886 words = append(append(sim.describe(sim.c.inputs), "|"), sim.describe(sim.c.outputs)...)
887 }
888 emit(strings.Join(words, " "))
889 case "CHECK":
890 s.signal(a[0])
891 value(a[1])
892 text := "ok"
893 if sim.values[a[0]] != a[1] {
894 text = "failed (" + sim.values[a[0]] + ")"
895 }
896 emit("check " + a[0] + "=" + a[1] + " " + text)
897 case "PROBE":
898 x := a[0]
899 s.signal(x)
900 text := "input"
901 if i, ok := sim.c.driver[x]; ok {
902 g := sim.c.gates[i]
903 text = fmt.Sprintf("%s %s %s delay %d", g.name, g.kind, strings.Join(sim.describe(g.inputs), " "), g.delay)
904 }
905 emit(x + "=" + sim.values[x] + " driven by " + text)
906 words := []string{"fanout", strconv.Itoa(len(sim.c.readers[x]))}
907 for _, i := range sim.c.readers[x] {
908 words = append(words, sim.c.gates[i].name)
909 }
910 emit(strings.Join(words, " "))
911 case "WAVE":
912 start, end := number(a[0], false), number(a[1], false)
913 if !(start < end && end <= sim.time) {
914 fail(fmt.Sprintf("bad range %d %d", start, end))
915 }
916 if len(sim.watch) == 0 {
917 fail("nothing watched")
918 }
919 for _, x := range sim.watch {
920 h := sim.history[x]
921 i := 0
922 v := "X"
923 var wave strings.Builder
924 for t := start; t < end; t++ {
925 for i < len(h) && h[i].time <= t {
926 v = h[i].value
927 i++
928 }
929 wave.WriteString(v)
930 }
931 emit(x + " " + wave.String())
932 }
933 case "HISTORY":
934 x := a[0]
935 s.signal(x)
936 h := sim.history[x]
937 words := []string{"history", x, "changes", strconv.Itoa(len(h))}
938 for _, v := range h {
939 words = append(words, fmt.Sprintf("%d:%s", v.time, v.value))
940 }
941 emit(strings.Join(words, " "))
942 case "ACTIVITY":
943 k := number(a[0], true)
944 ranked := append([]string{}, sim.c.signals...)
945 sort.Slice(ranked, func(i, j int) bool {
946 x, y := ranked[i], ranked[j]
947 return len(sim.history[x]) > len(sim.history[y]) || len(sim.history[x]) == len(sim.history[y]) && x < y
948 })
949 if k > len(ranked) {
950 k = len(ranked)
951 }
952 for _, x := range ranked[:k] {
953 emit(fmt.Sprintf("%s %d", x, len(sim.history[x])))
954 }
955 case "STATUS":
956 halted := "no"
957 if sim.halted {
958 halted = "yes"
959 }
960 emit(fmt.Sprintf("circuit %s time %d pending %d events %d halted %s", sim.c.name, sim.time, len(sim.queue), sim.processed, halted))
961 n := map[string]int{}
962 for _, v := range sim.values {
963 n[v]++
964 }
965 emit(fmt.Sprintf("values 0:%d 1:%d X:%d", n["0"], n["1"], n["X"]))
966 }
967}

985 } else {

987 }
988 }
