28const CHANNEL_NAMES = ["NOTE_OFF", "NOTE_ON", "POLY_PRESSURE", "CONTROL", "PROGRAM", "CHANNEL_PRESSURE", "PITCH_BEND"]
29const META_NAMES = Dict(0=>"SEQUENCE_NUMBER", 1=>"TEXT_EVENT", 2=>"COPYRIGHT", 3=>"TRACK_NAME", 4=>"INSTRUMENT_NAME", 5=>"LYRIC", 6=>"MARKER", 7=>"CUE_POINT", 32=>"CHANNEL_PREFIX", 47=>"END_OF_TRACK", 81=>"TEMPO", 84=>"SMPTE_OFFSET", 88=>"TIME_SIGNATURE", 89=>"KEY_SIGNATURE", 127=>"SEQUENCER_SPECIFIC")
30const META_CODES = Dict(name=>code for (code, name) in META_NAMES)
31const KEY_NAMES = [
32 ["Cb", "Gb", "Db", "Ab", "Eb", "Bb", "F", "C", "G", "D", "A", "E", "B", "F#", "C#"],
33 ["Ab", "Eb", "Bb", "F", "C", "G", "D", "A", "E", "B", "F#", "C#", "G#", "D#", "A#"]
34]
35number(s) = parse(Int64, s)
36channel(e) = e.status % 16 + 1
37ismeta(e, code) = e.status == 255 && e.meta == code
38ison(e) = e.status ÷ 16 == 9 && e.data[2] > 0
39isoff(e) = e.status ÷ 16 == 8 || (e.status ÷ 16 == 9 && e.data[2] == 0)
40eot(tick) = Event(tick, 255, 47, Int[])
41note_name(n) = ["C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"][n % 12 + 1] * string(n ÷ 12 - 1)
42function kind(e)
43 e.status < 240 && return CHANNEL_NAMES[e.status ÷ 16 - 7]
44 e.status == 240 && return "SYSEX"
45 e.status == 247 && return "SYSEX_ESCAPE"
46 get(META_NAMES, e.meta, "META")
47end
48function bigendian(data)
49 value = Int64(0)
50 for b in data
51 value = value * 256 + b
52 end
53 value
54end
55function bytes_be(value, size)
56 result = zeros(Int, size)
57 for i in size:-1:1
58 result[i] = mod(value, 256)
59 value = fld(value, 256)
60 end
61 result
62end
63function song_summary(s)

65 "format $(s.format) tracks $(length(s.tracks)) division $(s.division) events $count"
66end
67song_length(s) = maximum(track[end].tick for track in s.tracks)
68function ordered(s)
69 items = [(e.tick, t, i, e) for (t, track) in enumerate(s.tracks) for (i, e) in enumerate(track)]
70 sort!(items; by=x -> x[1:3])
71 [Placed(item[4], item[2]-1, i-1) for (i, item) in enumerate(items)]
72end
73function event_line(e, track)
74 fields = [string(e.tick), string(track), kind(e)]
75 d = e.data
76 if e.status < 240
77 push!(fields, string(channel(e)))
78 high = e.status ÷ 16
79 if high in (8, 9, 10)
80 append!(fields, [note_name(d[1]), string(d[2])])
81 elseif high == 14
82 push!(fields, string(d[1] + 128*d[2] - 8192))
83 else
84 append!(fields, string.(d))
85 end
86 elseif e.status == 255 && 1 <= e.meta <= 7
87 push!(fields, "\"" * String(UInt8.(d)) * "\"")
88 elseif ismeta(e, 0) || ismeta(e, 81)
89 push!(fields, string(bigendian(d)))
90 elseif ismeta(e, 32)
91 push!(fields, string(d[1]+1))
92 elseif ismeta(e, 47)
93 elseif ismeta(e, 84)
94 append!(fields, string.(d))
95 elseif ismeta(e, 88)
96 append!(fields, ["$(d[1])/$(1 << d[2])", string(d[3]), string(d[4])])
97 elseif ismeta(e, 89)
98 sf = d[1] >= 128 ? d[1]-256 : d[1]
99 append!(fields, [KEY_NAMES[d[2]+1][sf+8], d[2] == 0 ? "major" : "minor"])
100 else
101 e.status == 255 && e.meta != 127 && push!(fields, @sprintf("%02X", e.meta))
102 push!(fields, string(length(d)))
103 !isempty(d) && push!(fields, join([@sprintf("%02X", b) for b in d]))
104 end
105 join(fields, " ")
106end
107function parse_hex(data)
108 fail(pos, reason) = throw(ParseFailure(pos, reason))
109 function chunk(pos, id)
110 length(data)-pos < 8 && fail(pos, "truncated chunk")
111 String(UInt8.(data[pos+1:pos+4])) != id && fail(pos, "bad chunk id")
112 len = bigendian(data[pos+5:pos+8])
113 pos+8+len > length(data) && fail(pos, "truncated chunk")
114 pos+8, pos+8+len
115 end
116 start, finish = chunk(0, "MThd")
117 finish-start != 6 && fail(4, "bad header length")
118 f, count, division = bigendian(data[9:10]), bigendian(data[11:12]), bigendian(data[13:14])
119 f > 2 && fail(8, "bad format")
120 (count == 0 || f == 0 && count != 1) && fail(10, "bad track count")
121 division >= 32768 && fail(12, "smpte timing not supported")
122 division == 0 && fail(12, "bad division")
123 s = Song(f, division, Vector{Event}[])
124 pos = finish
125 for t in 1:count
126 pos == length(data) && fail(pos, "missing track")
127 start, finish = chunk(pos, "MTrk")
128 pos = start
129 tick = Int64(0)
130 running = 0
131 track = Event[]
132 while pos < finish
133 !isempty(track) && ismeta(track[end], 47) && fail(pos, "event after end of track")
134 event_start = pos
135 function readbyte()
136 pos >= finish && fail(event_start, "truncated event")
137 pos += 1
138 data[pos]
139 end
140 function vlq()
141 at = pos
142 value = Int64(0)
143 for i in 1:4
144 b = readbyte()
145 value = value*128 + (b & 127)
146 b < 128 && return value
147 end
148 fail(at, "variable-length value too long")
149 end
150 tick += vlq()
151 status_pos = pos
152 pos >= finish && fail(event_start, "truncated event")
153 status = data[pos+1]
154 if status < 128
155 running == 0 && fail(pos, "data byte without running status")
156 status = running
157 else
158 pos += 1
159 end
160 e = Event(tick, status, 0, Int[])
161 if status < 240
162 running = status
163 n = status ÷ 16 in (12, 13) ? 1 : 2
164 for j in 1:n
165 at = pos
166 b = readbyte()
167 b >= 128 && fail(at, "bad data byte")
168 push!(e.data, b)
169 end
170 elseif status in (240, 247, 255)
171 running = 0
172 status == 255 && (e.meta = readbyte())
173 len = vlq()
174 pos+len > finish && fail(event_start, "truncated event")
175 e.data = data[pos+1:pos+len]
176 pos += len
177 if status == 255
178 sizes = Dict(0=>2, 32=>1, 47=>0, 81=>3, 84=>5, 88=>4, 89=>2)
179 haskey(sizes, e.meta) && sizes[e.meta] != len && fail(status_pos, "bad meta length")
180 d = e.data
181 bad = e.meta == 81 && bigendian(d) == 0
182 bad |= e.meta == 32 && d[1] > 15
183 bad |= e.meta == 88 && (d[1] == 0 || d[2] > 6)
184 if e.meta == 89
185 sf = d[1] >= 128 ? d[1]-256 : d[1]
186 bad |= !( -7 <= sf <= 7 ) || d[2] > 1
187 end
188 bad && fail(status_pos, "bad meta data")
189 end
190 else
191 fail(status_pos, "unknown status byte")
192 end
193 push!(track, e)
194 end
195 (isempty(track) || !ismeta(track[end], 47)) && fail(finish, "missing end of track")
196 push!(s.tracks, track)
197 pos = finish
198 end
199 pos < length(data) && fail(pos, "extra data")
200 s
201end
202function parse_text(lines)
203 lineno = 1
204 fail(reason) = throw(ParseFailure(lineno, reason))
205 require(ok) = ok || fail("bad value")
206 h = split(lines[1])
207 (length(h) != 4 || h[1] != "HEADER") && fail("bad header")
208 f, count, division = number.(h[2:4])
209 !(0 <= f <= 2 && 1 <= count <= 65535 && 1 <= division <= 32767 && (f != 0 || count == 1)) && fail("bad header")
210 s = Song(f, division, [Event[] for _ in 1:count])
211 current = 1
212 for (i, line) in enumerate(lines[2:end])
213 lineno = i+1
214 w = split(line, ' '; limit=4)
215 tick, track, name = number(w[1]), number(w[2])+1, w[3]
216 fields = length(w) == 4 ? String.(split(w[4], ' ')) : String[]
217 index = findfirst(==(name), CHANNEL_NAMES)
218 known = haskey(META_CODES, name)
219 (isnothing(index) && !known && !(name in ("META", "SYSEX", "SYSEX_ESCAPE"))) && fail("unknown event type")
220 e = Event(tick, 255, 0, Int[])
221 function value(at, lo, hi)
222 require(at <= length(fields))
223 n = number(fields[at])
224 require(lo <= n <= hi)
225 n
226 end
227 if !isnothing(index)
228 n = index in (5, 6, 7) ? 2 : 3
229 require(length(fields) == n)
230 e.status = (index+7)*16 + value(1, 1, 16)-1
231 if index <= 3
232 pitch = findfirst(p -> note_name(p) == fields[2], 0:127)
233 require(!isnothing(pitch))
234 e.data = [pitch-1, value(3, 0, 127)]
235 elseif index == 7
236 bend = value(2, -8192, 8191)+8192
237 e.data = [bend % 128, bend ÷ 128]
238 else
239 e.data = [value(j, 0, 127) for j in 2:n]
240 end
241 elseif known && 1 <= META_CODES[name] <= 7
242 require(length(w) == 4)
243 text = w[4]
244 require(length(text) >= 2 && startswith(text, '"') && endswith(text, '"'))
245 require(!occursin('"', text[2:end-1]))
246 e.meta = META_CODES[name]
247 e.data = Int.(codeunits(text[2:end-1]))
248 elseif name in ("META", "SYSEX", "SYSEX_ESCAPE", "SEQUENCER_SPECIFIC")
249 if name == "SYSEX"
250 e.status = 240
251 elseif name == "SYSEX_ESCAPE"
252 e.status = 247
253 elseif name == "META"
254 require(length(fields) >= 2 && occursin(r"^[0-9A-F]{2}$", fields[1]))
255 e.meta = parse(Int, fields[1]; base=16)
256 require(!haskey(META_NAMES, e.meta))
257 fields = fields[2:end]
258 else
259 e.meta = 127
260 end
261 len = value(1, 0, 1000000)
262 if len == 0
263 require(length(fields) == 1)
264 else
265 require(length(fields) == 2 && length(fields[2]) == len*2)
266 require(occursin(r"^[0-9A-F]+$", fields[2]))
267 e.data = [parse(Int, fields[2][j:j+1]; base=16) for j in 1:2:length(fields[2])]
268 end
269 else
270 e.meta = META_CODES[name]
271 if e.meta == 0
272 require(length(fields) == 1)
273 e.data = bytes_be(value(1, 0, 65535), 2)
274 elseif e.meta == 32
275 require(length(fields) == 1)
276 e.data = [value(1, 1, 16)-1]
277 elseif e.meta == 47
278 require(isempty(fields))
279 elseif e.meta == 81
280 require(length(fields) == 1)
281 e.data = bytes_be(value(1, 1, 16777215), 3)
282 elseif e.meta == 84
283 require(length(fields) == 5)
284 e.data = [value(j, 0, 255) for j in 1:5]
285 elseif e.meta == 88
286 require(length(fields) == 3)
287 fraction = number.(split(fields[1], '/'))
288 nn, dd = fraction
289 exponent = findfirst(j -> dd == 1 << j, 0:6)
290 require(1 <= nn <= 255 && !isnothing(exponent))
291 e.data = [nn, exponent-1, value(2, 0, 255), value(3, 0, 255)]
292 elseif e.meta == 89
293 require(length(fields) == 2)
294 found = false
295 for mi in 0:1, sf in -7:7
296 if fields == [KEY_NAMES[mi+1][sf+8], mi == 0 ? "major" : "minor"]
297 e.data = [mod(sf, 256), mi]
298 found = true
299 end
300 end
301 require(found)
302 end
303 end
304 track > count && fail("bad track")
305 row = s.tracks[current]
306 ended = !isempty(row) && ismeta(row[end], 47)
307 track < current && fail("out of order")
308 track > current && (!ended || track > current+1) && fail("missing end of track")
309 track == current && ended && fail("event after end of track")
310 if track == current+1
311 current += 1
312 row = s.tracks[current]
313 end
314 !isempty(row) && tick < row[end].tick && fail("out of order")
315 push!(row, e)
316 end
317 lineno = length(lines)+1
318 (current < count || isempty(s.tracks[current]) || !ismeta(s.tracks[current][end], 47)) && fail("missing end of track")
319 s
320end
321function vlq(value)
322 data = [value % 128]
323 value ÷= 128
324 while value > 0
325 pushfirst!(data, (value % 128) | 128)
326 value ÷= 128
327 end
328 data
329end
330function encode(s, running, zero)
331 result = [Int.(codeunits("MThd")); bytes_be(6, 4); bytes_be(s.format, 2); bytes_be(length(s.tracks), 2); bytes_be(s.division, 2)]
332 for track in s.tracks
333 data = Int[]
334 last = Int64(0)
335 status = 0
336 for e in track
337 append!(data, vlq(e.tick-last))
338 last = e.tick
339 st, body = e.status, e.data
340 if zero && st ÷ 16 == 8
341 st += 16
342 body = [body[1], 0]
343 end
344 if st < 240
345 (!running || status != st) && push!(data, st)
346 append!(data, body)
347 status = st
348 else
349 status = 0
350 push!(data, st)
351 st == 255 && push!(data, e.meta)
352 append!(data, vlq(length(body)))
353 append!(data, body)
354 end
355 end
356 append!(result, [Int.(codeunits("MTrk")); bytes_be(length(data), 4); data])
357 end
358 result
359end
360function pairing(s)
361 queues = Dict{Tuple{Int,Int},Vector{Placed}}()
362 notes = Note[]
363 unmatched = Placed[]
364 for p in ordered(s)
365 e = p.event
366 !(ison(e) || isoff(e)) && continue
367 queue = get!(queues, (channel(e), e.data[1]), Placed[])
368 if ison(e)
369 push!(queue, p)
370 elseif isempty(queue)
371 push!(unmatched, p)
372 else
373 on = popfirst!(queue)
374 push!(notes, Note(on, p, e.tick-on.event.tick))
375 end
376 end
377 for queue in values(queues)
378 append!(unmatched, queue)
379 end
380 sort!(notes; by=n -> (n.on.event.tick, channel(n.on.event), n.on.event.data[1], n.on.order))
381 sort!(unmatched; by=p -> p.order)
382 notes, unmatched
383end
384function time_at(s, tick)
385 total, last, tempo = Int64(0), Int64(0), Int64(500000)
386 for p in ordered(s)
387 e = p.event
388 e.tick > tick && break
389 if ismeta(e, 81)
390 total += (e.tick-last)*tempo
391 last, tempo = e.tick, bigendian(e.data)
392 end
393 end
394 usec = (total+(tick-last)*tempo) ÷ s.division
395 @sprintf("%d.%03d", usec ÷ 1000, usec % 1000)
396end
397function bar_at(s, tick)
398 start, bar, nn, beat = Int64(0), Int64(1), Int64(4), s.division
399 for p in ordered(s)
400 e = p.event
401 e.tick > tick && break
402 if ismeta(e, 88)
403 size = nn*beat
404 bar += (e.tick-start+size-1) ÷ size
405 start = e.tick
406 nn = e.data[1]
407 beat = s.division*4 ÷ (1 << e.data[2])
408 end
409 end
410 x = tick-start
411 "$(bar+x ÷ (nn*beat)):$(x % (nn*beat) ÷ beat+1):$(x % beat)"
412end
413function execute(words, songs, settings)
414 cmd, a = words[1], words[2:end]
415 commands = ["DECODE", "ENCODE", "TIMING", "NOTES", "STATS", "ROLL", "TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "FILTER", "MERGE", "SPLIT", "COPY", "LIST", "RUNNING", "NOTEOFF"]
416 if !(cmd in commands)
417 println("error: unknown command")
418 return
419 elseif cmd == "LIST"
420 for name in sort(collect(keys(songs)))
421 println(name, " ", song_summary(songs[name]))
422 end
423 isempty(songs) && println("no songs")
424 return
425 elseif cmd in ("RUNNING", "NOTEOFF")
426 if !(a[1] in (cmd == "RUNNING" ? ("ON", "OFF") : ("ZERO", "KEEP")))
427 println("error: bad argument")
428 else
429 settings[cmd] = a[1]
430 println(lowercase(cmd), " ", lowercase(a[1]))
431 end
432 return
433 elseif !haskey(songs, a[1])
434 println("error: no song ", a[1])
435 return
436 end
437 s = songs[a[1]]
438 args = cmd == "COPY" ? Int64[] : number.(a[2:end])
439 valid = true
440 if cmd == "TRANSPOSE"
441 valid = -127 <= args[1] <= 127 && (length(args) == 1 || 1 <= args[2] <= 16)
442 elseif cmd == "QUANTIZE"
443 valid = 1 <= args[1] <= 1000000 && 0 <= args[2] <= 100
444 elseif cmd == "VELOCITY"
445 valid = 0 <= args[1] <= 1000 && -127 <= args[2] <= 127
446 elseif cmd == "TEMPO"
447 valid = all(x -> 1 <= x <= 1000, args)
448 elseif cmd == "FILTER"
449 valid = all(x -> 1 <= x <= 16, args)
450 elseif cmd == "ROLL"
451 valid = 0 <= args[1] <= 10000000 && 1 <= args[2] <= 10000000 && 1 <= args[3] <= 10000000 && args[2] > args[1]
452 valid && (valid = cld(args[2]-args[1], args[3]) <= 64)
453 end
454 if !valid
455 println("error: bad argument")
456 elseif cmd in ("TIMING", "STATS") && s.format == 2
457 println("error: format 2 not supported")
458 elseif cmd == "MERGE" && s.format != 1
459 println("error: not format 1")
460 elseif cmd == "SPLIT" && s.format != 0
461 println("error: not format 0")
462 elseif cmd == "COPY"
463 songs[a[2]] = deepcopy(s)
464 println("copied ", a[1], " to ", a[2])
465 elseif cmd == "DECODE"
466 println("HEADER $(s.format) $(length(s.tracks)) $(s.division)")
467 for (t, track) in enumerate(s.tracks), e in track
468 println(event_line(e, t-1))
469 end
470 elseif cmd == "ENCODE"
471 data = encode(s, settings["RUNNING"] == "ON", settings["NOTEOFF"] == "ZERO")
472 for i in 1:16:length(data)
473 println(@sprintf("%04X: ", i-1), join([@sprintf("%02X", b) for b in data[i:min(i+15, end)]], " "))
474 end
475 println(length(data), " bytes")
476 elseif cmd == "TIMING"
477 for p in ordered(s)
478 e = p.event
479 println("$(e.tick) $(p.track) $(kind(e)) $(time_at(s, e.tick)) $(bar_at(s, e.tick))")
480 end
481 tick = song_length(s)
482 println("end $tick $(time_at(s, tick)) $(bar_at(s, tick))")
483 elseif cmd in ("NOTES", "STATS", "ROLL")
484 notes, unmatched = pairing(s)
485 if cmd == "ROLL"
486 columns = cld(args[2]-args[1], args[3])
487 found = false
488 for pitch in 127:-1:0
489 row = fill('.', columns)
490 for i in eachindex(row)
491 tick = args[1]+(i-1)*args[3]
492 for n in notes
493 n.on.event.data[1] != pitch && continue
494 start = n.on.event.tick
495 if tick <= start < tick+args[3]
496 row[i] = '#'
497 break
498 elseif start <= tick < start+n.length
499 row[i] = '='
500 end
501 end
502 end
503 if any(!=('.'), row)
504 println(rpad(note_name(pitch), 4), " ", join(row))
505 found = true
506 end
507 end
508 !found && println("empty")
509 return
510 end
511 cmd == "STATS" && println(song_summary(s))
512 println("notes $(length(notes)) unmatched $(length(unmatched))")
513 if cmd == "NOTES"
514 for n in notes
515 e = n.on.event
516 println("$(e.tick) $(n.length) $(channel(e)) $(note_name(e.data[1])) $(e.data[2])")
517 end
518 for p in unmatched
519 e = p.event
520 println("unmatched $(ison(e) ? "on" : "off") $(e.tick) $(p.track) $(channel(e)) $(note_name(e.data[1]))")
521 end
522 else
523 pitches = Int[]
524 for ch in 1:16
525 group = [n.on.event.data[1] for n in notes if channel(n.on.event) == ch]
526 if !isempty(group)
527 println("channel $ch notes $(length(group)) range $(note_name(minimum(group))) $(note_name(maximum(group)))")
528 append!(pitches, group)
529 end
530 end
531 println(isempty(pitches) ? "range none" : "range $(note_name(minimum(pitches))) $(note_name(maximum(pitches)))")
532 println("length $(song_length(s)) ticks $(time_at(s, song_length(s))) ms")
533 changes = Dict{Int64,Int}()
534 for n in notes
535 if n.length > 0
536 start = n.on.event.tick
537 changes[start] = get(changes, start, 0)+1
538 changes[start+n.length] = get(changes, start+n.length, 0)-1
539 end
540 end
541 current, peak, at = 0, 0, Int64(0)
542 for tick in sort(collect(keys(changes)))
543 current += changes[tick]
544 if current > peak
545 peak, at = current, tick
546 end
547 end
548 println(peak == 0 ? "polyphony 0" : "polyphony $peak at $at")
549 end
550 elseif cmd == "TRANSPOSE"
551 kept = 0
552 dropped = String[]
553 for (t, track) in enumerate(s.tracks)
554 row = Event[]
555 for e in track
556 if e.status < 176 && (length(args) == 1 || channel(e) == args[2])
557 pitch = e.data[1]+args[1]
558 if !(0 <= pitch <= 127)
559 push!(dropped, "dropped $(e.tick) $(t-1) $(kind(e)) $(channel(e)) $(note_name(e.data[1]))")
560 continue
561 end
562 e.data[1] = pitch
563 kept += 1
564 end
565 push!(row, e)
566 end
567 s.tracks[t] = row
568 end
569 println("transposed $kept dropped $(length(dropped))")
570 foreach(println, dropped)
571 elseif cmd == "QUANTIZE"
572 notes, unmatched = pairing(s)
573 targets = [(n.on.event, n.off.event) for n in notes]
574 for p in unmatched
575 ison(p.event) && push!(targets, (p.event, p.event))
576 end
577 moved = 0
578 for (on, off) in targets
579 grid, strength = args
580 lower = on.tick ÷ grid * grid
581 target = on.tick-lower <= lower+grid-on.tick ? lower : lower+grid
582 delta = fld(2*(target-on.tick)*strength+100, 200)
583 moved += delta != 0
584 on.tick += delta
585 off !== on && (off.tick += delta)
586 end
587 for (t, track) in enumerate(s.tracks)
588 ending = track[end]
589 row = track[1:end-1]
590 sort!(row; by=e -> e.tick, alg=MergeSort)
591 !isempty(row) && (ending.tick = max(ending.tick, row[end].tick))
592 s.tracks[t] = [row; ending]
593 end
594 println("quantized $(length(targets)) moved $moved")
595 elseif cmd == "VELOCITY"
596 count, clamped = 0, 0
597 for track in s.tracks, e in track
598 if ison(e)
599 count += 1
600 v = (e.data[2]*args[1]+50) ÷ 100+args[2]
601 clamped += !(1 <= v <= 127)
602 e.data[2] = clamp(v, 1, 127)
603 end
604 end
605 println("velocity $count clamped $clamped")
606 elseif cmd == "TEMPO"
607 trial = deepcopy(s)
608 count, exists, good = 0, false, true
609 scale(v) = (2*v*args[2]+args[1]) ÷ (2*args[1])
610 for track in trial.tracks, e in track
611 if ismeta(e, 81)
612 count += 1
613 exists |= e.tick == 0
614 v = scale(bigendian(e.data))
615 good &= 1 <= v <= 16777215
616 e.data = bytes_be(v, 3)
617 end
618 end
619 inserted = scale(500000)
620 if !exists
621 good &= 1 <= inserted <= 16777215
622 pushfirst!(trial.tracks[1], Event(0, 255, 81, bytes_be(inserted, 3)))
623 end
624 if !good
625 println("error: tempo out of range")
626 else
627 songs[a[1]] = trial
628 println("scaled $count")
629 !exists && println("inserted 0 0 TEMPO $inserted")
630 end
631 elseif cmd == "FILTER"
632 count = 0
633 for (t, track) in enumerate(s.tracks)
634 row = Event[]
635 for e in track
636 if e.status < 240 && !(channel(e) in args)
637 count += 1
638 else
639 push!(row, e)
640 end
641 end
642 s.tracks[t] = row
643 end
644 println("removed $count")
645 elseif cmd == "MERGE"
646 original = length(s.tracks)
647 tick = song_length(s)
648 row = [p.event for p in ordered(s) if !ismeta(p.event, 47)]
649 push!(row, eot(tick))
650 s.tracks = [row]
651 s.format = 0
652 println("merged $original tracks events $(length(row))")
653 elseif cmd == "SPLIT"
654 tick = song_length(s)
655 rows = Dict(0=>Event[])
656 for e in s.tracks[1]
657 ismeta(e, 47) && continue
658 ch = e.status < 240 ? channel(e) : 0
659 push!(get!(rows, ch, Event[]), e)
660 end
661 s.tracks = [[rows[ch]; eot(tick)] for ch in sort(collect(keys(rows)))]
662 s.format = 1
663 println("split tracks $(length(s.tracks)) events $(sum(length, s.tracks))")
664 end
665end
666function main()
667 n = parse(Int, readline())
668 songs = Dict{String,Song}()
669 settings = Dict("RUNNING"=>"ON", "NOTEOFF"=>"KEEP")
670 for _ in 1:n
671 words = String.(split(readline()))
672 if words[1] in ("HEX", "TEXT")
673 lines = [readline() for _ in 1:parse(Int, words[3])]
674 try
675 if words[1] == "HEX"
676 data = [parse(Int, b; base=16) for line in lines for b in split(line)]
677 s = parse_hex(data)
678 else
679 s = parse_text(lines)
680 end
681 songs[words[2]] = s
682 println("loaded ", words[2], " ", song_summary(s))
683 catch e
684 e isa ParseFailure || rethrow()
685 label = words[1] == "HEX" ? "byte" : "line"
686 println("error: $label $(e.position): $(e.reason)")
687 end
688 else
689 execute(words, songs, settings)
690 end
691 end
692end
