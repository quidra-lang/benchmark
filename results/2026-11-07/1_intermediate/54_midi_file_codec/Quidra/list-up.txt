int minimum(int a, int b)
    return if a < b then a else b
int integer(string text)
    return int(text)
string hex_value(int value, width := 2)
    if width <= 0 and value <= 0
        return ""
    checked := nat(value)
    hex := checked.B16
    return if #hex < width then "0" * (width - #hex) + hex else hex
string note_name(int pitch)
    names := ["C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"]
    return names[pitch % 12] + string(pitch / 12 - 1)
int[] big_bytes(int value, int size)
    result := array(nat(size), fill = 0)
    for: j < size
        result[size - 1 - j] = value % 256
        value /= 256
    return result
int[] vlq_bytes(int value)
    result := [value % 128]
    value /= 128
    while value > 0
        result = [value % 128 + 128].concat(result)
        value /= 128
    return result
string kind_name(int code)
    if code < 240
        return ["NOTE_OFF", "NOTE_ON", "POLY_PRESSURE", "CONTROL", "PROGRAM", "CHANNEL_PRESSURE", "PITCH_BEND"][code / 16 - 8]
class Event(int tick, int code, int[] payload := [])
    bool channel_message()
        return code < 240
    int channel_number()
        return code % 16 + 1
    string fields()
        if channel_message()
            result := string(channel_number())
            family := code / 16
            if family <= 10
                return result + " " + note_name(payload[0]) + " {payload[1]}"
    string text_line(int track_number)
        string suffix := fields()
        return "{tick} {track_number} {kind_name(code)}" + (if suffix == "" then "" else " " + suffix)
class Track(Event[] events := [])
class Placed(int track_number, int event_number, Event event)
class Song(int format, int division, Track[] tracks)
int[] encode_song(Song song, bool running, bool zero_off)
    int[] result := "MThd".codepoints().concat(big_bytes(6, 4)).concat(big_bytes(song.format, 2)).concat(big_bytes(int(#song.tracks), 2)).concat(big_bytes(song.division, 2))
    for track_value := song.tracks[]
        int[] bytes := []
        previous := 0
        status := -1
        for ev := track_value.events[]
            bytes = bytes.concat(vlq_bytes(ev.tick - previous))
            previous = ev.tick
            code_value := ev.code
            payload_value := ev.payload
            if zero_off and code_value / 16 == 8
                code_value += 16
                payload_value[1] = 0
            if code_value < 240
                if not running or status != code_value
                    bytes = bytes.append(code_value)
                bytes = bytes.concat(payload_value)
                status = code_value
            else
                status = -1
                if code_value >= 256
                    bytes = bytes.append(255).append(code_value - 256)
                else
                    bytes = bytes.append(code_value)
                bytes = bytes.concat(vlq_bytes(int(#payload_value))).concat(payload_value)
        result = result.concat("MTrk".codepoints()).concat(big_bytes(int(#bytes), 4)).concat(bytes)
    return result

    string[] .command(string[] words, string[] data_lines)
        op := words[0]
        commands := ["HEX", "TEXT", "DECODE", "ENCODE", "TIMING", "NOTES", "STATS", "ROLL", "TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "FILTER", "MERGE", "SPLIT", "COPY", "LIST", "RUNNING", "NOTEOFF"]
        known := false
        for candidate := commands[]
            known = known or candidate == op
        if not known
            return ["error: unknown command"]
        int at := lookup(words[1])
        if at < 0
            return ["error: no song {words[1]}"]
        song := songs[at].song
        int[] arguments := []
        if op != "COPY"
            for: 2 <= j < #words
                arguments = arguments.append(integer(words[j]))
        valid := true
        if op == "TRANSPOSE"
            valid = -127 <= arguments[0] <= 127 and (#arguments == 1 or 1 <= arguments[1] <= 16)
        if not valid
            return ["error: bad argument"]
        if op == "DECODE"
            output = ["HEADER {song.format} {#song.tracks} {song.division}"]
            for: ti < #song.tracks
                for ev := song.tracks[ti].events[]
                    output = output.append(ev.text_line(int(ti)))
            return output
        if op == "ENCODE"
            bytes := encode_song(song, running, zero_off)
            for: 0 <= j < int(#bytes) step 16
                string[] parts := []
                for: j <= k < minimum(j + 16, int(#bytes))
                    parts = parts.append(hex_value(bytes[k]))
                output = output.append(hex_value(j, 4) + ": " + parts.join(" "))
            return output.append("{#bytes} bytes")
        if op == "TRANSPOSE" or op == "FILTER"
            changed := 0
            dropped := 0
            string[] dropped_lines := []
            for: ti < #song.tracks
                Event[] kept := []
                for ev := song.tracks[ti].events[]
                    remove := false
                    elif op == "TRANSPOSE" and ev.code / 16 >= 8 and ev.code / 16 <= 10 and (#arguments == 1 or ev.channel_number() == arguments[1])
                        pitch := ev.payload[0] + arguments[0]
                        if not (0 <= pitch <= 127)
                            remove = true
                            dropped_lines = dropped_lines.append("dropped {ev.tick} {ti} {kind_name(ev.code)} {ev.channel_number()} {note_name(ev.payload[0])}")
                        else
                            ev.payload[0] = pitch
                            changed += 1
                    if remove
                        dropped += 1
                    else
                        kept = kept.append(ev)
                song.tracks[ti].events = kept
            output = if op == "FILTER" then ["removed {dropped}"] else ["transposed {changed} dropped {dropped}"].concat(dropped_lines)
        songs[at].song = song
        return output
