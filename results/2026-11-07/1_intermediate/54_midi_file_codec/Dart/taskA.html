10class Event {
11 int tick;
12 final int code;
13 List<int> data;
14 Event(this.tick, this.code, List<int> data) : data = List.of(data);
15 Event copy() => Event(tick, code, data);
16 bool channel() => code >= 128 && code < 240;
17 int ch() => (code & 15) + 1;
18 bool on() => channel() && code >> 4 == 9 && data[1] > 0;
19 bool off() => channel() && (code >> 4 == 8 || code >> 4 == 9 && data[1] == 0);
20 bool end() => code == 303;
21 String kind() {
22 if (channel()) return channelKinds[(code >> 4) - 8];
23 if (code == 240) return 'SYSEX';
24 if (code == 247) return 'SYSEX_ESCAPE';
25 final type = code - 256;
26 return type >= 1 && type <= 7 ? textKinds[type - 1] : metaKinds[type] ?? 'META';
27 }
28}

87String description(Song s) => 'format ${s.format} tracks ${s.tracks.length} division ${s.division} events ${s.count()}';
88String eventText(Event e, int track) {
89 final head = '${e.tick} $track ${e.kind()}';
90 final d = e.data, meta = e.code - 256;
91 if (e.channel()) {
92 final String fields;
93 if ((e.code >> 4) <= 10) { fields = '${noteName(d[0])} ${d[1]}'; }
94 else if ((e.code >> 4) == 14) { fields = '${d[0] + 128 * d[1] - 8192}'; }
95 else { fields = d.join(' '); }
96 return '$head ${e.ch()} $fields';
97 }
98 if (meta >= 1 && meta <= 7) return '$head "${String.fromCharCodes(d)}"';
99 if (e.end()) return head;
100 if (e.code == 240 || e.code == 247 || meta == 127 || !metaKinds.containsKey(meta)) {
101 return '$head${e.kind() == 'META' ? ' ${hex(meta, 2)}' : ''} ${d.length}${d.isEmpty ? '' : ' ${dataHex(d)}'}';
102 }
103 if (meta == 0 || meta == 81) return '$head ${value(d)}';
104 if (meta == 32) return '$head ${d[0] + 1}';
105 if (meta == 84) return '$head ${d.join(' ')}';
106 if (meta == 88) return '$head ${d[0]}/${1 << d[1]} ${d[2]} ${d[3]}';
107 final sf = d[0] >= 128 ? d[0] - 256 : d[0];
108 return '$head ${d[1] == 0 ? '${majorKeys[sf + 7]} major' : '${minorKeys[sf + 7]} minor'}';
109}
110class Parser {
111 final List<int> data;
112 int position = 0, end, eventStart = 0;
113 Parser(this.data) : end = data.length;
114 Never error(int offset, String reason) => fail('byte $offset: $reason');
115 int read() {
116 if (position >= end) error(eventStart, 'truncated event');
117 return data[position++];
118 }
119 int vlq() {
120 final start = position;
121 int n = 0;
122 for (int i = 0; i < 4; i++) {
123 final b = read();
124 n = n * 128 + (b & 127);
125 if (b < 128) return n;
126 }
127 error(start, 'variable-length value too long');
128 }
129 int integer(int p, int n) => value(data.sublist(p, p + n));
130 int chunk(String id) {
131 final p = position;
132 if (data.length - p < 8) error(p, 'truncated chunk');
133 for (int i = 0; i < 4; i++) { if (data[p + i] != id.codeUnitAt(i)) error(p, 'bad chunk id'); }
134 final length = integer(p + 4, 4);
135 if (p + 8 + length > data.length) error(p, 'truncated chunk');
136 position += 8;
137 return length;
138 }
139 void validateMeta(int type, List<int> d, int p) {
140 final length = switch (type) { 0 || 89 => 2, 32 => 1, 47 => 0, 81 => 3, 84 => 5, 88 => 4, _ => -1 };
141 if (length >= 0 && d.length != length) error(p, 'bad meta length');
142 bool invalid = false;
143 if (type == 81) invalid = value(d) == 0;
144 if (type == 32) invalid = d[0] > 15;
145 if (type == 88) invalid = d[0] == 0 || d[1] > 6;
146 if (type == 89) {
147 final sf = d[0] >= 128 ? d[0] - 256 : d[0];
148 invalid = sf < -7 || sf > 7 || d[1] > 1;
149 }
150 if (invalid) error(p, 'bad meta data');
151 }
152 Song parse() {
153 if (chunk('MThd') != 6) error(4, 'bad header length');
154 final format = integer(8, 2), count = integer(10, 2), division = integer(12, 2);
155 if (format > 2) error(8, 'bad format');
156 if (count == 0 || format == 0 && count != 1) error(10, 'bad track count');
157 if ((division & 32768) != 0) error(12, 'smpte timing not supported');
158 if (division == 0) error(12, 'bad division');
159 position = 14;
160 final List<List<Event>> tracks = [];
161 for (int track = 0; track < count; track++) {
162 if (position == data.length) error(position, 'missing track');
163 final length = chunk('MTrk');
164 end = position + length;
165 final List<Event> events = [];
166 int tick = 0, status = -1;
167 while (position < end) {
168 if (events.isNotEmpty && events.last.end()) error(position, 'event after end of track');
169 eventStart = position;
170 tick += vlq();
171 final statusPosition = position, b = read();
172 if (b < 128) {
173 if (status < 0) error(statusPosition, 'data byte without running status');
174 position--;
175 } else { status = b; }
176 int code = status;
177 final List<int> payload = [];
178 if (status >= 128 && status <= 239) {
179 final n = (status >> 4) == 12 || (status >> 4) == 13 ? 1 : 2;
180 for (int i = 0; i < n; i++) {
181 final p = position, v = read();
182 if (v >= 128) error(p, 'bad data byte');
183 payload.add(v);
184 }
185 } else if (status == 240 || status == 247 || status == 255) {
186 final meta = status == 255;
187 if (meta) code = 256 + read();
188 final size = vlq();
189 if (position + size > end) error(eventStart, 'truncated event');
190 payload.addAll(data.sublist(position, position + size));
191 position += size;
192 if (meta) validateMeta(code - 256, payload, statusPosition);
193 status = -1;
194 } else { error(statusPosition, 'unknown status byte'); }
195 events.add(Event(tick, code, payload));
196 }
197 if (events.isEmpty || !events.last.end()) error(end, 'missing end of track');
198 tracks.add(events);
199 end = data.length;
200 }
201 if (position < data.length) error(position, 'extra data');
202 return Song(format, division, tracks);
203 }

205int bounded(String text, int low, int high) {
206 final n = int.tryParse(text);
207 if (n == null || n < low || n > high) fail('bad value');
208 return n;
209}
210void fields(List<String> f, int n) { if (f.length != n) fail('bad value'); }
211List<int> blob(List<String> f, int start) {
212 if (f.length <= start) fail('bad value');
213 final length = bounded(f[start], 0, 1000000);
214 if (f.length != start + (length == 0 ? 1 : 2)) fail('bad value');
215 if (length == 0) return [];
216 final text = f[start + 1];
217 if (text.length != 2 * length || !RegExp(r'^[0-9A-F]*$').hasMatch(text)) fail('bad value');
218 return [for (int i = 0; i < text.length; i += 2) int.parse(text.substring(i, i + 2), radix: 16)];
219}
220Event parseEvent(String line) {
221 final words = line.split(' ');
222 final tick = int.parse(words[0]), kind = words[2];
223 final prefixLength = words[0].length + words[1].length + kind.length + 2;
224 final tail = line.length > prefixLength ? line.substring(prefixLength + 1) : '';
225 final List<String> f = tail.isEmpty ? [] : tail.split(' ');
226 final channelIndex = channelKinds.indexOf(kind), textIndex = textKinds.indexOf(kind);
227 int code;
228 List<int> data = [];
229 if (channelIndex >= 0) {
230 fields(f, channelIndex >= 4 ? 2 : 3);
231 final ch = bounded(f[0], 1, 16);
232 code = (channelIndex + 8) * 16 + ch - 1;
233 if (channelIndex <= 2) { data = [noteNumber(f[1]), bounded(f[2], 0, 127)]; }
234 else if (channelIndex == 6) {
235 final bend = bounded(f[1], -8192, 8191) + 8192;
236 data = [bend % 128, bend ~/ 128];
237 } else { data = f.skip(1).map((v) => bounded(v, 0, 127)).toList(); }
238 } else if (textIndex >= 0) {
239 if (tail.length < 2 || !tail.startsWith('"') || !tail.endsWith('"') || tail.substring(1, tail.length - 1).contains('"')) fail('bad value');
240 code = 257 + textIndex;
241 data = tail.substring(1, tail.length - 1).codeUnits;
242 } else if (kind == 'SYSEX' || kind == 'SYSEX_ESCAPE') {
243 code = kind == 'SYSEX' ? 240 : 247;
244 data = blob(f, 0);
245 } else if (kind == 'META') {
246 if (f.length < 2 || !RegExp(r'^[0-9A-F]{2}$').hasMatch(f[0])) fail('bad value');
247 final type = int.parse(f[0], radix: 16);
248 if (type <= 7 || metaKinds.containsKey(type)) fail('bad value');
249 code = 256 + type;
250 data = blob(f, 1);
251 } else {
252 int? meta;
253 for (final entry in metaKinds.entries) { if (entry.value == kind) meta = entry.key; }
254 if (meta == null) fail('unknown event type');
255 code = 256 + meta;
256 switch (meta) {
257 case 0: {
258 fields(f, 1);
259 data = bytes(bounded(f[0], 0, 65535), 2);
260 }
261 case 32: {
262 fields(f, 1);
263 data = [bounded(f[0], 1, 16) - 1];
264 }
265 case 47: {
266 fields(f, 0);
267 }
268 case 81: {
269 fields(f, 1);
270 data = bytes(bounded(f[0], 1, 16777215), 3);
271 }
272 case 84: {
273 fields(f, 5);
274 data = f.map((v) => bounded(v, 0, 255)).toList();
275 }
276 case 88: {
277 fields(f, 3);
278 final signature = f[0].split('/');
279 if (signature.length != 2) fail('bad value');
280 final numerator = bounded(signature[0], 1, 255), denominator = bounded(signature[1], 1, 64);
281 if ((denominator & (denominator - 1)) != 0) fail('bad value');
282 int exponent = 0;
283 while ((1 << exponent) < denominator) { exponent++; }
284 data = [numerator, exponent, bounded(f[1], 0, 255), bounded(f[2], 0, 255)];
285 }
286 case 89: {
287 fields(f, 2);
288 if (f[1] != 'minor' && f[1] != 'major') fail('bad value');
289 final minor = f[1] == 'minor', index = (f[1] == 'minor' ? minorKeys : majorKeys).indexOf(f[0]);
290 if (index < 0) fail('bad value');
291 data = [(index - 7 + 256) % 256, minor ? 1 : 0];
292 }
293 case 127: {
294 data = blob(f, 0);
295 }
296 }
297 }
298 return Event(tick, code, data);
299}
300Song textSong(List<String> lines) {
301 final h = lines[0].split(' ');
302 late int format, count, division;
303 try {
304 if (h.length != 4 || h[0] != 'HEADER') fail('bad header');
305 format = bounded(h[1], 0, 2);
306 count = bounded(h[2], 1, 65535);
307 division = bounded(h[3], 1, 32767);
308 if (format == 0 && count != 1) fail('bad header');
309 } on Failure { fail('line 1: bad header'); }
310 final List<List<Event>> tracks = [[]];
311 int current = 0;
312 for (int i = 1; i < lines.length; i++) {
313 try {
314 final e = parseEvent(lines[i]), track = int.parse(lines[i].split(' ')[1]);
315 if (track >= count) fail('bad track');
316 List<Event> events = tracks[current];
317 final end = events.isNotEmpty && events.last.end();
318 if (track < current) fail('out of order');
319 if (track > current && (!end || track > current + 1)) fail('missing end of track');
320 if (track == current && end) fail('event after end of track');
321 if (track == current + 1) { current++; events = []; tracks.add(events); }
322 if (events.isNotEmpty && e.tick < events.last.tick) fail('out of order');
323 events.add(e);
324 } on Failure catch (error) { fail('line ${i + 1}: ${error.message}'); }
325 }
326 if (current + 1 < count || tracks[current].isEmpty || !tracks[current].last.end()) fail('line ${lines.length + 1}: missing end of track');
327 return Song(format, division, tracks);
328}
329List<int> vlq(int n) {
330 final result = [n & 127];
331 n >>= 7;
332 while (n > 0) { result.insert(0, (n & 127) | 128); n >>= 7; }
333 return result;
334}
335void encode(Song s) {
336 final List<int> result = [];
337 void chunk(List<int> target, String id, List<int> data) {
338 target.addAll(id.codeUnits);
339 target.addAll(bytes(data.length, 4));
340 target.addAll(data);
341 }
342 chunk(result, 'MThd', [...bytes(s.format, 2), ...bytes(s.tracks.length, 2), ...bytes(s.division, 2)]);
343 for (final track in s.tracks) {
344 final List<int> data = [];
345 int last = 0, previous = -1;
346 for (final e in track) {
347 data.addAll(vlq(e.tick - last));
348 last = e.tick;
349 if (e.channel()) {
350 int code = e.code;
351 List<int> payload = e.data;
352 if (zeroOff && (code >> 4) == 8) { code += 16; payload = [e.data[0], 0]; }
353 if (!running || previous != code) data.add(code);
354 data.addAll(payload);
355 previous = code;
356 } else {
357 previous = -1;
358 if (e.code < 256) { data.add(e.code); }
359 else { data.addAll([255, e.code - 256]); }
360 data.addAll(vlq(e.data.length));
361 data.addAll(e.data);
362 }
363 }
364 chunk(result, 'MTrk', data);
365 }
366 for (int i = 0; i < result.length; i += 16) {
367 emit('${hex(i, 4)}: ${result.sublist(i, i + 16 < result.length ? i + 16 : result.length).map((b) => hex(b, 2)).join(' ')}');
368 }
369 emit('${result.length} bytes');
370}
371Pairing pairing(Song s) {
372 final Map<int, Queue<Located>> queues = {};
373 final List<Note> notes = [];
374 final List<Located> unmatched = [];
375 final events = global(s);
376 for (final l in events) {
377 final e = l.event;
378 if (!e.on() && !e.off()) continue;
379 final q = queues.putIfAbsent(e.ch() * 128 + e.data[0], () => Queue());
380 if (e.on()) { q.addLast(l); }
381 else if (q.isEmpty) { unmatched.add(l); }
382 else { notes.add(Note(q.removeFirst(), l)); }
383 }
384 for (final q in queues.values) { unmatched.addAll(q); }
385 final Map<Event, int> order = {for (int i = 0; i < events.length; i++) events[i].event: i};
386 notes.sort((a, b) {
387 int c = a.start().compareTo(b.start());
388 if (c == 0) c = a.channel().compareTo(b.channel());
389 if (c == 0) c = a.pitch().compareTo(b.pitch());
390 return c == 0 ? order[a.on.event]!.compareTo(order[b.on.event]!) : c;
391 });
392 unmatched.sort((a, b) => order[a.event]!.compareTo(order[b.event]!));
393 return Pairing(notes, unmatched);
394}
395String time(Song s, int tick) {
396 int total = 0, last = 0, tempo = 500000;
397 for (final l in global(s)) {
398 if (l.event.tick <= tick && l.event.code == 337) {
399 total += (l.event.tick - last) * tempo;
400 last = l.event.tick;
401 tempo = value(l.event.data);
402 }
403 }
404 final us = (total + (tick - last) * tempo) ~/ s.division;
405 return '${us ~/ 1000}.${(us % 1000).toString().padLeft(3, '0')}';
406}
407String bar(Song s, int tick) {
408 int start = 0, number = 1, numerator = 4, beat = s.division;
409 for (final l in global(s)) {
410 if (l.event.tick <= tick && l.event.code == 344) {
411 final length = numerator * beat;
412 number += (l.event.tick - start + length - 1) ~/ length;
413 start = l.event.tick;
414 numerator = l.event.data[0];
415 beat = s.division * 4 ~/ (1 << l.event.data[1]);
416 }
417 }
418 final x = tick - start, length = numerator * beat;
419 return '${number + x ~/ length}:${x % length ~/ beat + 1}:${x % beat}';
420}

428void execute(List<String> w) {
429 final c = w[0];
430 if (c == 'HEX' || c == 'TEXT') {
431 final lines = List.generate(int.parse(w[2]), (_) => stdin.readLineSync()!);
432 final s = c == 'TEXT' ? textSong(lines) : Parser(lines.expand((line) => line.split(' ').map((b) => int.parse(b, radix: 16))).toList()).parse();
433 songs[w[1]] = s;
434 emit('loaded ${w[1]} ${description(s)}');
435 return;
436 }
437 if (!['DECODE', 'ENCODE', 'TIMING', 'NOTES', 'STATS', 'ROLL', 'TRANSPOSE', 'QUANTIZE', 'VELOCITY', 'TEMPO', 'FILTER', 'MERGE', 'SPLIT', 'COPY', 'LIST', 'RUNNING', 'NOTEOFF'].contains(c)) fail('unknown command');
438 if (c == 'LIST') {
439 if (songs.isEmpty) { emit('no songs'); }
440 else { for (final entry in songs.entries) { emit('${entry.key} ${description(entry.value)}'); } }
441 return;
442 }
443 if (c == 'RUNNING' || c == 'NOTEOFF') {
444 if (!(c == 'RUNNING' ? ['ON', 'OFF'] : ['ZERO', 'KEEP']).contains(w[1])) fail('bad argument');
445 if (c == 'RUNNING') { running = w[1] == 'ON'; }
446 else { zeroOff = w[1] == 'ZERO'; }
447 emit('${c.toLowerCase()} ${w[1].toLowerCase()}');
448 return;
449 }
450 final s = songs[w[1]];
451 if (s == null) fail('no song ${w[1]}');
452 int a = 0, b = 0, d = 0;
453 if (['TRANSPOSE', 'QUANTIZE', 'VELOCITY', 'TEMPO', 'ROLL'].contains(c)) {
454 a = int.parse(w[2]);
455 if (w.length > 3) b = int.parse(w[3]);
456 if (c == 'ROLL') d = int.parse(w[4]);
457 final valid = switch (c) {
458 'TRANSPOSE' => range(a, -127, 127) && (w.length == 3 || range(b, 1, 16)),
459 'QUANTIZE' => range(a, 1, 1000000) && range(b, 0, 100),
460 'VELOCITY' => range(a, 0, 1000) && range(b, -127, 127),
461 'TEMPO' => range(a, 1, 1000) && range(b, 1, 1000),
462 _ => range(a, 0, 10000000) && range(b, 1, 10000000) && range(d, 1, 10000000) && b > a && (b - a + d - 1) ~/ d <= 64,
463 };
464 if (!valid) fail('bad argument');
465 }
466 if (c == 'FILTER' && w.skip(2).any((v) => !range(int.parse(v), 1, 16))) fail('bad argument');
467 if ((c == 'TIMING' || c == 'STATS') && s.format == 2) fail('format 2 not supported');
468 switch (c) {
469 case 'COPY': {
470 songs[w[2]] = s.copy();
471 emit('copied ${w[1]} to ${w[2]}');
472 }
473 case 'DECODE': {
474 emit('HEADER ${s.format} ${s.tracks.length} ${s.division}');
475 for (int t = 0; t < s.tracks.length; t++) { for (final e in s.tracks[t]) { emit(eventText(e, t)); } }
476 }
477 case 'ENCODE': {
478 encode(s);
479 }
480 case 'TIMING': {
481 for (final l in global(s)) { emit('${l.event.tick} ${l.track} ${l.event.kind()} ${time(s, l.event.tick)} ${bar(s, l.event.tick)}'); }
482 emit('end ${s.length()} ${time(s, s.length())} ${bar(s, s.length())}');
483 }
484 case 'NOTES': {
485 printNotes(pairing(s));
486 }
487 case 'TRANSPOSE': {
488 int kept = 0;
489 final List<String> removed = [];
490 for (int t = 0; t < s.tracks.length; t++) {
491 final List<Event> track = [];
492 for (final e in s.tracks[t]) {
493 if (e.channel() && (e.code >> 4) <= 10 && (w.length == 3 || e.ch() == b)) {
494 final original = e.data[0], pitch = e.data[0] + a;
495 if (!range(pitch, 0, 127)) {
496 removed.add('dropped ${e.tick} $t ${e.kind()} ${e.ch()} ${noteName(original)}');
497 continue;
498 }
499 e.data[0] = pitch;
500 kept++;
501 }
502 track.add(e);
503 }
504 s.tracks[t] = track;
505 }
506 emit('transposed $kept dropped ${removed.length}');
507 for (final line in removed) { emit(line); }
508 }
509 case 'QUANTIZE': {
510 final p = pairing(s);
511 final targets = [...p.notes, ...p.unmatched.where((l) => l.event.on()).map((l) => Note(l, l))];
512 int moved = 0;
513 for (final n in targets) {
514 final tick = n.start(), lower = n.start() ~/ a * a;
515 final g = tick - lower <= lower + a - tick ? lower : lower + a;
516 final delta = floorDiv(2 * (g - tick) * b + 100, 200);
517 n.on.event.tick += delta;
518 if (!identical(n.off.event, n.on.event)) n.off.event.tick += delta;
519 if (delta != 0) moved++;
520 }
521 for (final track in s.tracks) {
522 final end = track.removeLast();
523 final Map<Event, int> order = {for (int i = 0; i < track.length; i++) track[i]: i};
524 track.sort((a, b) { final c = a.tick.compareTo(b.tick); return c == 0 ? order[a]!.compareTo(order[b]!) : c; });
525 for (final e in track) { if (e.tick > end.tick) end.tick = e.tick; }
526 track.add(end);
527 }
528 emit('quantized ${targets.length} moved $moved');
529 }
530 case 'VELOCITY': {
531 int count = 0, clamped = 0;
532 for (final l in global(s)) {
533 if (l.event.on()) {
534 final v = (l.event.data[1] * a + 50) ~/ 100 + b;
535 if (!range(v, 1, 127)) clamped++;
536 l.event.data[1] = v < 1 ? 1 : v > 127 ? 127 : v;
537 count++;
538 }
539 }
540 emit('velocity $count clamped $clamped');
541 }
542 case 'TEMPO': {
543 final edited = s.copy();
544 int count = 0;
545 bool atZero = false;
546 for (final l in global(edited)) {
547 if (l.event.code == 337) {
548 final v = (2 * value(l.event.data) * b + a) ~/ (2 * a);
549 if (!range(v, 1, 16777215)) fail('tempo out of range');
550 l.event.data = bytes(v, 3);
551 count++;
552 if (l.event.tick == 0) atZero = true;
553 }
554 }
555 final inserted = (1000000 * b + a) ~/ (2 * a);
556 if (!atZero) {
557 if (!range(inserted, 1, 16777215)) fail('tempo out of range');
558 edited.tracks[0].insert(0, Event(0, 337, bytes(inserted, 3)));
559 }
560 songs[w[1]] = edited;
561 emit('scaled $count');
562 if (!atZero) emit('inserted 0 0 TEMPO $inserted');
563 }
564 case 'FILTER': {
565 final channels = w.skip(2).map(int.parse).toSet(), before = s.count();
566 for (final track in s.tracks) { track.removeWhere((e) => e.channel() && !channels.contains(e.ch())); }
567 emit('removed ${before - s.count()}');
568 }
569 case 'MERGE': {
570 if (s.format != 1) fail('not format 1');
571 final original = s.tracks.length, length = s.length();
572 final events = global(s).where((l) => !l.event.end()).map((l) => l.event).toList();
573 events.add(Event(length, 303, []));
574 s.tracks = [events];
575 s.format = 0;
576 emit('merged $original tracks events ${s.count()}');
577 }
578 case 'SPLIT': {
579 if (s.format != 0) fail('not format 0');
580 final length = s.length();
581 final SplayTreeMap<int, List<Event>> groups = SplayTreeMap();
582 groups[0] = [];
583 for (final e in s.tracks[0]) {
584 if (!e.end()) groups.putIfAbsent(e.channel() ? e.ch() : 0, () => []).add(e);
585 }
586 s.tracks = groups.values.toList();
587 for (final track in s.tracks) { track.add(Event(length, 303, [])); }
588 s.format = 1;
589 emit('split tracks ${s.tracks.length} events ${s.count()}');
590 }
591 case 'STATS': {
592 final p = pairing(s);
593 emit(description(s));
594 emit('notes ${p.notes.length} unmatched ${p.unmatched.length}');
595 for (int ch = 1; ch <= 16; ch++) {
596 final notes = p.notes.where((n) => n.channel() == ch).toList();
597 if (notes.isNotEmpty) {
598 final ps = notes.map((n) => n.pitch()).toList()..sort();
599 emit('channel $ch notes ${notes.length} range ${noteName(ps.first)} ${noteName(ps.last)}');
600 }
601 }
602 final ps = p.notes.map((n) => n.pitch()).toList()..sort();
603 emit(ps.isEmpty ? 'range none' : 'range ${noteName(ps.first)} ${noteName(ps.last)}');
604 emit('length ${s.length()} ticks ${time(s, s.length())} ms');
605 final SplayTreeMap<int, int> changes = SplayTreeMap();
606 for (final n in p.notes) {
607 if (n.length() > 0) {
608 changes[n.start()] = (changes[n.start()] ?? 0) + 1;
609 final end = n.start() + n.length();
610 changes[end] = (changes[end] ?? 0) - 1;
611 }
612 }
613 int active = 0, maximum = 0, first = 0;
614 for (final entry in changes.entries) {
615 active += entry.value;
616 if (active > maximum) { maximum = active; first = entry.key; }
617 }
618 emit(maximum == 0 ? 'polyphony 0' : 'polyphony $maximum at $first');
619 }
620 case 'ROLL': {
621 final p = pairing(s), columns = (b - a + d - 1) ~/ d;
622 bool any = false;
623 for (int pitch = 127; pitch >= 0; pitch--) {
624 final line = StringBuffer();
625 bool visible = false;
626 final notes = p.notes.where((n) => n.pitch() == pitch).toList();
627 for (int i = 0; i < columns; i++) {
628 final tick = a + i * d;
629 final mark = notes.any((n) => tick <= n.start() && n.start() < tick + d) ? '#'
630 : notes.any((n) => n.start() <= tick && tick < n.start() + n.length()) ? '=' : '.';
631 line.write(mark);
632 if (mark != '.') visible = true;
633 }
634 if (visible) { emit('${noteName(pitch).padRight(4)} $line'); any = true; }
635 }
636 if (!any) emit('empty');
637 }
638 }
639}
640void main() {
641 final n = int.parse(stdin.readLineSync()!);
642 for (int i = 0; i < n; i++) {
643 final words = stdin.readLineSync()!.split(' ');
644 try { execute(words); }
645 on Failure catch (error) { emit('error: ${error.message}'); }
646 }
647 stdout.write(output);
648}
