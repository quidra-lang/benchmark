415function timing(s, tick) {
416 let total = 0,
417 last = 0,
418 tempo = 500000,
419 start = 0,
420 bar = 1,
421 nn = 4,
422 beat = s.division;
423 for (const l of globalEvents(s)) {

429 tempo = value(e);

432 bar += Math.ceil((e.tick - start) / (nn * beat));

439 x = tick - start;

441 time: `${Math.floor(micros / 1000)}.${String(micros % 1000).padStart(3, "0")}`,
442 bar: `${bar + Math.floor(x / (nn * beat))}:${Math.floor((x % (nn * beat)) / beat) + 1}:${x % beat}`,
443 };

643 }
644 s.tracks = s.tracks.map((t) => {
645 const end = t[t.length - 1],
646 others = t.slice(0, -1).sort((a, b) => a.tick - b.tick);
647 end.tick = others.reduce((n, e) => Math.max(n, e.tick), end.tick);
648 return others.concat(end);
649 });
650 emit(`quantized ${targets.length} moved ${moved}`);
651 break;
652 }
653 case "VELOCITY": {
654 let converted = 0,
655 clamped = 0;
656 for (const l of globalEvents(s))
657 if (isOn(l.event)) {
658 const v = Math.floor((l.event.data[1] * a[0] + 50) / 100) + a[1];
659 converted++;
660 if (v < 1 || v > 127) clamped++;
661 l.event.data[1] = Math.max(1, Math.min(127, v));
662 }
663 emit(`velocity ${converted} clamped ${clamped}`);
664 break;
665 }
666 case "TEMPO": {
667 const tempos = globalEvents(s).filter((l) => l.event.type === "TEMPO"),
668 convert = (n) => Math.floor((2 * n * a[1] + a[0]) / (2 * a[0]));
669 const values = tempos.map((l) => convert(value(l.event))),
670 insert = !tempos.some((l) => l.event.tick === 0),
671 initial = convert(500000);
672 if (
673 values.some((n) => n < 1 || n > 16777215) ||
674 (insert && (initial < 1 || initial > 16777215))
675 )
676 fail("tempo out of range");
677 tempos.forEach((l, i) => {
678 l.event.data = be(values[i], 3);
679 });
680 emit("scaled " + tempos.length);
681 if (insert) {
682 s.tracks[0].unshift(event(0, "TEMPO", be(initial, 3)));
683 emit("inserted 0 0 TEMPO " + initial);
684 }
685 break;
686 }
687 case "FILTER": {
688 const before = count(s);
689 s.tracks = s.tracks.map((t) =>
690 t.filter((e) => !channelTypes.includes(e.type) || a.includes(e.ch)),
691 );
692 emit("removed " + (before - count(s)));
693 break;
694 }
695 case "MERGE": {
696 if (s.format !== 1) fail("not format 1");
697 const n = s.tracks.length,
698 end = length(s),
699 events = globalEvents(s)
700 .filter((l) => l.event.type !== "END_OF_TRACK")
701 .map((l) => l.event);
702 s.tracks = [events.concat(event(end, "END_OF_TRACK"))];
703 s.format = 0;
704 emit(`merged ${n} tracks events ${count(s)}`);
705 break;
706 }
707 case "SPLIT": {
708 if (s.format !== 0) fail("not format 0");
709 const end = length(s),
710 events = s.tracks[0].slice(0, -1),
711 channels = Array.from(
712 new Set(events.filter((e) => channelTypes.includes(e.type)).map((e) => e.ch)),
713 ).sort((a, b) => a - b);
714 s.tracks = [events.filter((e) => !channelTypes.includes(e.type))].concat(
715 channels.map((ch) =>
716 events.filter((e) => channelTypes.includes(e.type) && e.ch === ch),
717 ),
718 );
719 s.tracks.forEach((t) => t.push(event(end, "END_OF_TRACK")));
720 s.format = 1;
721 emit(`split tracks ${s.tracks.length} events ${count(s)}`);
722 break;
723 }
724 case "STATS": {
725 const p = pairing(s);
726 emit(description(s));
727 emit(`notes ${p.notes.length} unmatched ${p.unmatched.length}`);
728 const rangeText = (notes) => {
729 let lo = 127,
730 hi = 0;
731 for (const n of notes) {
732 lo = Math.min(lo, n.on.event.data[0]);
733 hi = Math.max(hi, n.on.event.data[0]);
734 }
735 return `${noteNames[lo]} ${noteNames[hi]}`;
736 };
737 for (let ch = 1; ch <= 16; ch++) {
738 const notes = p.notes.filter((n) => n.on.event.ch === ch);
739 if (notes.length)
740 emit(`channel ${ch} notes ${notes.length} range ${rangeText(notes)}`);
741 }
742 emit("range " + (p.notes.length ? rangeText(p.notes) : "none"));
743 emit(`length ${length(s)} ticks ${timing(s, length(s)).time} ms`);
744 const changes = new Map();
745 for (const n of p.notes)
746 if (n.off.event.tick > n.on.event.tick) {
747 changes.set(n.on.event.tick, (changes.get(n.on.event.tick) ?? 0) + 1);
748 changes.set(n.off.event.tick, (changes.get(n.off.event.tick) ?? 0) - 1);
749 }
750 let active = 0,
751 maximum = 0,
752 at = 0;
753 for (const tick of Array.from(changes.keys()).sort((a, b) => a - b)) {
754 active += changes.get(tick);
755 if (active > maximum) {
756 maximum = active;
757 at = tick;
758 }
759 }
760 emit(maximum ? `polyphony ${maximum} at ${at}` : "polyphony 0");
761 break;
762 }
763 case "ROLL": {
764 const notes = pairing(s).notes;
765 let rows = 0;
766 for (let pitch = 127; pitch >= 0; pitch--) {
767 let cells = "";
768 for (let tick = a[0]; tick < a[1]; tick += a[2]) {
769 const ns = notes.filter((n) => n.on.event.data[0] === pitch);
770 cells += ns.some(
771 (n) => n.on.event.tick >= tick && n.on.event.tick < tick + a[2],
772 )
773 ? "#"
774 : ns.some((n) => n.on.event.tick <= tick && tick < n.off.event.tick)
775 ? "="
776 : ".";
777 }
778 if (/[^.]/.test(cells)) {
779 emit(noteNames[pitch].padEnd(4) + " " + cells);
780 rows++;
781 }
782 }
783 if (!rows) emit("empty");
784 break;
785 }
786 }
787}
788const input = readFileSync(0, "utf8").trimEnd().split("\n");
789let cursor = 1;
790for (let n = 0; n < Number(input[0]); n++) {
791 const words = input[cursor++].split(" "),
792 cmd = words[0];
793 try {
794 if (cmd === "HEX" || cmd === "TEXT") {
795 const data = input.slice(cursor, cursor + Number(words[2]));
796 cursor += Number(words[2]);
797 const song =
798 cmd === "HEX"
799 ? parseHex(
800 data
801 .join(" ")
802 .split(" ")
803 .map((s) => parseInt(s, 16)),
804 )
805 : parseText(data);
806 songs.set(words[1], song);
807 emit(`loaded ${words[1]} ${description(song)}`);
808 } else command(words);
809 } catch (error) {
810 emit("error: " + (error instanceof Error ? error.message : String(error)));
811 }
812}
813process.stdout.write(output.join("\n") + "\n");
