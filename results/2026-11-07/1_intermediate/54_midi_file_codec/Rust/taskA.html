4type Result<T> = std::result::Result<T, String>;
5const CHANNEL_TYPES: [&str; 7] = ["NOTE_OFF", "NOTE_ON", "POLY_PRESSURE", "CONTROL", "PROGRAM", "CHANNEL_PRESSURE", "PITCH_BEND"];
6const TEXT_TYPES: [&str; 7] = ["TEXT_EVENT", "COPYRIGHT", "TRACK_NAME", "INSTRUMENT_NAME", "LYRIC", "MARKER", "CUE_POINT"];
7const PITCHES: [&str; 12] = ["C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B"];
8const KEYS: [[&str; 15]; 2] = [
9 ["Cb", "Gb", "Db", "Ab", "Eb", "Bb", "F", "C", "G", "D", "A", "E", "B", "F#", "C#"],
10 ["Ab", "Eb", "Bb", "F", "C", "G", "D", "A", "E", "B", "F#", "C#", "G#", "D#", "A#"],
11];
12fn meta_type(meta: i32) -> &'static str {
13 match meta {
14 0 => "SEQUENCE_NUMBER", 32 => "CHANNEL_PREFIX", 47 => "END_OF_TRACK", 81 => "TEMPO",
15 84 => "SMPTE_OFFSET", 88 => "TIME_SIGNATURE", 89 => "KEY_SIGNATURE", 127 => "SEQUENCER_SPECIFIC", _ => "META",
16 }
17}
18fn inside(n: i64, low: i64, high: i64) -> bool { low <= n && n <= high }
19fn require(ok: bool) -> Result<()> { if ok { Ok(()) } else { Err("bad value".into()) } }
20fn pitch(p: u8) -> String { format!("{}{}", PITCHES[(p % 12) as usize], p as i32 / 12 - 1) }
21fn parse_pitch(name: &str) -> Option<u8> { (0..128).find(|p| pitch(*p) == name) }
22fn bytes(value: i64, count: usize) -> Vec<u8> { (0..count).map(|i| ((value >> (8 * (count - 1 - i))) & 255) as u8).collect() }
23fn blob_text(data: &[u8]) -> String {
24 format!("{}{}", data.len(), if data.is_empty() { String::new() } else { format!(" {}", data.iter().map(|x| format!("{x:02X}")).collect::<String>()) })
25}
26#[derive(Clone)]
27struct Event { tick: i64, status: u8, meta: i32, data: Vec<u8> }
28impl Event {
29 fn channel(&self) -> bool { self.status < 240 }
30 fn ch(&self) -> u8 { (self.status & 15) + 1 }
31 fn code(&self) -> u8 { self.status >> 4 }
32 fn on(&self) -> bool { self.channel() && self.code() == 9 && self.data[1] > 0 }
33 fn off(&self) -> bool { self.channel() && (self.code() == 8 || self.code() == 9 && self.data[1] == 0) }
34 fn end(&self) -> bool { self.status == 255 && self.meta == 47 }
35 fn number(&self) -> i64 { self.data.iter().fold(0, |a, b| a * 256 + *b as i64) }
36 fn kind(&self) -> &'static str {
37 if self.channel() { CHANNEL_TYPES[(self.code() - 8) as usize] }
38 else if self.status == 240 { "SYSEX" }
39 else if self.status == 247 { "SYSEX_ESCAPE" }
40 else if inside(self.meta as i64, 1, 7) { TEXT_TYPES[(self.meta - 1) as usize] }
41 else { meta_type(self.meta) }
42 }
43 fn fields(&self) -> String {
44 let d = &self.data;
45 if self.channel() {
46 let value = match self.code() {
47 8..=10 => format!("{} {}", pitch(d[0]), d[1]),
48 11 => format!("{} {}", d[0], d[1]),
49 14 => (d[0] as i32 + 128 * d[1] as i32 - 8192).to_string(),
50 _ => d[0].to_string(),
51 };
52 return format!("{} {value}", self.ch());
53 }
54 if self.status != 255 { return blob_text(d); }
55 if inside(self.meta as i64, 1, 7) { return format!("\"{}\"", String::from_utf8_lossy(d)); }
56 match self.meta {
57 0 | 81 => self.number().to_string(),
58 32 => (d[0] + 1).to_string(),
59 47 => String::new(),
60 84 => d.iter().map(u8::to_string).collect::<Vec<_>>().join(" "),
61 88 => format!("{}/{} {} {}", d[0], 1u32 << d[1], d[2], d[3]),
62 89 => format!("{} {}", KEYS[d[1] as usize][(d[0] as i8 as i32 + 7) as usize], if d[1] == 0 { "major" } else { "minor" }),
63 127 => blob_text(d),
64 _ => format!("{:02X} {}", self.meta, blob_text(d)),
65 }
66 }
67 fn line(&self, track: usize) -> String {
68 let fields = self.fields();
69 format!("{} {track} {}{}", self.tick, self.kind(), if fields.is_empty() { fields } else { format!(" {fields}") })
70 }
71}
72fn end_event(tick: i64) -> Event { Event { tick, status: 255, meta: 47, data: vec![] } }
73#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
74struct Location { track: usize, index: usize }
75#[derive(Clone)]
76struct Reference { event: Event, location: Location, order: usize }
77#[derive(Clone)]
78struct Note { on: Reference, off: Reference }
79#[derive(Clone)]
80struct Song { format: i32, division: i64, tracks: Vec<Vec<Event>> }
81impl Song {
82 fn count(&self) -> usize { self.tracks.iter().map(Vec::len).sum() }
83 fn length(&self) -> i64 { self.tracks.iter().map(|t| t.last().unwrap().tick).max().unwrap() }
84 fn summary(&self) -> String { format!("format {} tracks {} division {} events {}", self.format, self.tracks.len(), self.division, self.count()) }
85 fn global(&self) -> Vec<Reference> {
86 let mut all: Vec<Reference> = self.tracks.iter().enumerate().flat_map(|(track, events)| events.iter().enumerate().map(move |(index, event)| Reference { event: event.clone(), location: Location { track, index }, order: 0 })).collect();
87 all.sort_by_key(|r| (r.event.tick, r.location));
88 for (i, r) in all.iter_mut().enumerate() { r.order = i; }
89 all
90 }
91 fn pair(&self) -> (Vec<Note>, Vec<Reference>) {
92 let all = self.global();
93 let mut queues: BTreeMap<(u8, u8), VecDeque<Reference>> = BTreeMap::new();
94 let mut matched = BTreeSet::new();
95 let mut notes = vec![];
96 for reference in &all {
97 let e = &reference.event;
98 if !e.on() && !e.off() { continue; }
99 let queue = queues.entry((e.ch(), e.data[0])).or_default();
100 if e.on() { queue.push_back(reference.clone()); }
101 else if let Some(on) = queue.pop_front() {
102 matched.insert(on.location); matched.insert(reference.location);
103 notes.push(Note { on, off: reference.clone() });
104 }
105 }
106 notes.sort_by_key(|n| (n.on.event.tick, n.on.event.ch(), n.on.event.data[0], n.on.order));
107 let unmatched = all.into_iter().filter(|r| (r.event.on() || r.event.off()) && !matched.contains(&r.location)).collect();
108 (notes, unmatched)
109 }
110 fn position(&self, tick: i64, bars: bool) -> (String, String) {
111 let (mut sum, mut last, mut tempo) = (0, 0, 500000);
112 let (mut start, mut bar, mut nn, mut beat) = (0, 1, 4, self.division);
113 for reference in self.global().iter().filter(|r| r.event.tick <= tick) {
114 let e = &reference.event;
115 if e.status != 255 { continue; }
116 if e.meta == 81 { sum += (e.tick - last) * tempo; last = e.tick; tempo = e.number(); }
117 if e.meta == 88 && bars {
118 let size = nn * beat;
119 bar += (e.tick - start + size - 1) / size; start = e.tick;
120 nn = e.data[0] as i64; beat = self.division * 4 / (1 << e.data[1]);
121 }
122 }
123 let micro = (sum + (tick - last) * tempo) / self.division;
124 let time = format!("{}.{:03}", micro / 1000, micro % 1000);
125 let x = tick - start;
126 let bar = if bars { format!("{}:{}:{}", bar + x / (nn * beat), x % (nn * beat) / beat + 1, x % beat) } else { String::new() };
127 (time, bar)
128 }
129}
130struct Reader<'a> { data: &'a [u8], p: usize, end: usize, start: usize }
131impl Reader<'_> {
132 fn big(&self, p: usize, n: usize) -> i64 { self.data[p..p + n].iter().fold(0, |a, b| a * 256 + *b as i64) }
133 fn chunk(&self, p: usize, id: &[u8]) -> Result<usize> {
134 if self.data.len() - p < 8 { return Err(format!("byte {p}: truncated chunk")); }
135 if &self.data[p..p + 4] != id { return Err(format!("byte {p}: bad chunk id")); }
136 let end = p as i64 + 8 + self.big(p + 4, 4);
137 if end > self.data.len() as i64 { return Err(format!("byte {p}: truncated chunk")); }
138 Ok(end as usize)
139 }
140 fn byte(&mut self) -> Result<u8> {
141 if self.p >= self.end { return Err(format!("byte {}: truncated event", self.start)); }
142 let b = self.data[self.p]; self.p += 1; Ok(b)
143 }
144 fn variable(&mut self) -> Result<i64> {
145 let at = self.p;
146 let mut result = 0;
147 for _ in 0..4 {
148 let b = self.byte()?; result = result * 128 + (b & 127) as i64;
149 if b < 128 { return Ok(result); }
150 }
151 Err(format!("byte {at}: variable-length value too long"))
152 }
153}
154fn validate_meta(e: &Event, at: usize) -> Result<()> {
155 let length = match e.meta { 0 => Some(2), 32 => Some(1), 47 => Some(0), 81 => Some(3), 84 => Some(5), 88 => Some(4), 89 => Some(2), _ => None };
156 if length.is_some_and(|n| n != e.data.len()) { return Err(format!("byte {at}: bad meta length")); }
157 let d = &e.data;
158 let bad = match e.meta {
159 81 => e.number() == 0, 32 => d[0] > 15, 88 => d[0] == 0 || d[1] > 6,
160 89 => !inside(d[0] as i8 as i64, -7, 7) || d[1] > 1, _ => false,
161 };
162 if bad { Err(format!("byte {at}: bad meta data")) } else { Ok(()) }
163}
164fn parse_hex(data: &[u8]) -> Result<Song> {
165 let mut r = Reader { data, p: 0, end: 0, start: 0 };
166 if r.chunk(0, b"MThd")? != 14 { return Err("byte 4: bad header length".into()); }
167 let (format, tracks, division) = (r.big(8, 2), r.big(10, 2), r.big(12, 2));
168 if format > 2 { return Err("byte 8: bad format".into()); }
169 if tracks == 0 || format == 0 && tracks != 1 { return Err("byte 10: bad track count".into()); }
170 if division & 32768 != 0 { return Err("byte 12: smpte timing not supported".into()); }
171 if division == 0 { return Err("byte 12: bad division".into()); }
172 let mut song = Song { format: format as i32, division, tracks: vec![] };
173 r.p = 14;
174 for _ in 0..tracks {
175 if r.p == data.len() { return Err(format!("byte {}: missing track", r.p)); }
176 r.end = r.chunk(r.p, b"MTrk")?; r.p += 8;
177 let mut events: Vec<Event> = vec![];
178 let (mut tick, mut running) = (0, 0);
179 while r.p < r.end {
180 if events.last().is_some_and(Event::end) { return Err(format!("byte {}: event after end of track", r.p)); }
181 r.start = r.p; tick += r.variable()?;
182 if r.p >= r.end { return Err(format!("byte {}: truncated event", r.start)); }
183 let at = r.p;
184 let mut status = data[r.p];
185 if status < 128 {
186 if running == 0 { return Err(format!("byte {}: data byte without running status", r.p)); }
187 status = running;
188 } else { r.p += 1; }
189 let mut e = Event { tick, status, meta: -1, data: vec![] };
190 if status < 240 {
191 running = status;
192 let count = if status >> 4 == 12 || status >> 4 == 13 { 1 } else { 2 };
193 for _ in 0..count {
194 let position = r.p;
195 let b = r.byte()?;
196 if b >= 128 { return Err(format!("byte {position}: bad data byte")); }
197 e.data.push(b);
198 }
199 } else if matches!(status, 240 | 247 | 255) {
200 running = 0;
201 if status == 255 { e.meta = r.byte()? as i32; }
202 let count = r.variable()? as usize;
203 if count > r.end - r.p { return Err(format!("byte {}: truncated event", r.start)); }
204 e.data = data[r.p..r.p + count].to_vec(); r.p += count;
205 if status == 255 { validate_meta(&e, at)?; }
206 } else { return Err(format!("byte {at}: unknown status byte")); }
207 events.push(e);
208 }
209 if !events.last().is_some_and(Event::end) { return Err(format!("byte {}: missing end of track", r.end)); }
210 song.tracks.push(events);
211 }
212 if r.p != data.len() { return Err(format!("byte {}: extra data", r.p)); }
213 Ok(song)
214}
215fn num(a: &[&str], i: usize, low: i64, high: i64) -> Result<i64> {
216 a.get(i).and_then(|s| s.parse::<i64>().ok()).filter(|n| inside(*n, low, high)).ok_or_else(|| "bad value".into())
217}
218fn hex_text(s: &str) -> bool { s.bytes().all(|c| c.is_ascii_digit() || (b'A'..=b'F').contains(&c)) }
219fn blob(a: &[&str], index: usize) -> Result<Vec<u8>> {
220 let count = num(a, index, 0, 1000000)? as usize;
221 require(a.len() == index + if count == 0 { 1 } else { 2 })?;
222 if count == 0 { return Ok(vec![]); }
223 let text = a[index + 1];
224 require(text.len() == count * 2 && hex_text(text))?;
225 Ok((0..text.len()).step_by(2).map(|i| u8::from_str_radix(&text[i..i + 2], 16).unwrap()).collect())
226}
227fn text_event(tick: i64, kind: &str, fields: &str) -> Result<Event> {
228 let code = CHANNEL_TYPES.iter().position(|s| *s == kind).map(|i| i as u8 + 8);
229 let text = TEXT_TYPES.iter().position(|s| *s == kind);
230 let known_meta = (0..256).find(|m| meta_type(*m) != "META" && meta_type(*m) == kind);
231 if code.is_none() && text.is_none() && known_meta.is_none() && !matches!(kind, "META" | "SYSEX" | "SYSEX_ESCAPE") { return Err("unknown event type".into()); }
232 let a: Vec<&str> = if fields.is_empty() { vec![] } else { fields.split(' ').collect() };
233 if let Some(code) = code {
234 require(a.len() == if matches!(code, 12 | 13 | 14) { 2 } else { 3 })?;
235 let ch = num(&a, 0, 1, 16)? as u8;
236 let data = match code {
237 8..=10 => vec![parse_pitch(a[1]).ok_or("bad value")?, num(&a, 2, 0, 127)? as u8],
238 11 => vec![num(&a, 1, 0, 127)? as u8, num(&a, 2, 0, 127)? as u8],
239 14 => { let value = num(&a, 1, -8192, 8191)? + 8192; vec![(value & 127) as u8, (value >> 7) as u8] },
240 _ => vec![num(&a, 1, 0, 127)? as u8],
241 };
242 return Ok(Event { tick, status: code * 16 + ch - 1, meta: -1, data });
243 }
244 if matches!(kind, "SYSEX" | "SYSEX_ESCAPE") { return Ok(Event { tick, status: if kind == "SYSEX" { 240 } else { 247 }, meta: -1, data: blob(&a, 0)? }); }
245 if let Some(text) = text {
246 require(fields.len() >= 2 && fields.starts_with('"') && fields.ends_with('"') && !fields[1..fields.len() - 1].contains('"'))?;
247 return Ok(Event { tick, status: 255, meta: text as i32 + 1, data: fields.as_bytes()[1..fields.len() - 1].to_vec() });
248 }
249 if kind == "META" {
250 require(a.len() >= 2 && a[0].len() == 2 && hex_text(a[0]))?;
251 let meta = i32::from_str_radix(a[0], 16).unwrap();
252 require(meta_type(meta) == "META" && !inside(meta as i64, 1, 7))?;
253 return Ok(Event { tick, status: 255, meta, data: blob(&a, 1)? });
254 }
255 let meta = known_meta.unwrap();
256 if meta == 127 { return Ok(Event { tick, status: 255, meta, data: blob(&a, 0)? }); }
257 let count = match meta { 47 => 0, 84 => 5, 88 => 3, 89 => 2, _ => 1 };
258 require(a.len() == count)?;
259 let data = match meta {
260 0 => bytes(num(&a, 0, 0, 65535)?, 2),
261 32 => vec![(num(&a, 0, 1, 16)? - 1) as u8],
262 47 => vec![],
263 81 => bytes(num(&a, 0, 1, 16777215)?, 3),
264 84 => (0..5).map(|i| num(&a, i, 0, 255).map(|x| x as u8)).collect::<Result<Vec<_>>>()?,
265 88 => {
266 let signature: Vec<&str> = a[0].split('/').collect();
267 require(signature.len() == 2)?;
268 let nn = num(&signature, 0, 1, 255)?;
269 let dd = num(&signature, 1, 1, 64)?;
270 let exponent = (0..=6).find(|e| (1 << e) == dd).ok_or("bad value")?;
271 vec![nn as u8, exponent, num(&a, 1, 0, 255)? as u8, num(&a, 2, 0, 255)? as u8]
272 }
273 89 => {
274 require(matches!(a[1], "major" | "minor"))?;
275 let minor = usize::from(a[1] == "minor");
276 let key = KEYS[minor].iter().position(|s| *s == a[0]).ok_or("bad value")?;
277 vec![(key as i32 - 7) as u8, minor as u8]
278 }
279 _ => unreachable!(),
280 };
281 Ok(Event { tick, status: 255, meta, data })
282}
283fn parse_text(lines: &[String]) -> Result<Song> {
284 let h: Vec<&str> = lines[0].split(' ').collect();
285 if h.len() != 4 || h[0] != "HEADER" { return Err("line 1: bad header".into()); }
286 let header = || -> Result<(i64, i64, i64)> {
287 let f = num(&h, 1, 0, 2)?; let t = num(&h, 2, 1, 65535)?; let d = num(&h, 3, 1, 32767)?;
288 require(f != 0 || t == 1)?; Ok((f, t, d))
289 };
290 let (format, tracks, division) = header().map_err(|_| "line 1: bad header".to_string())?;
291 let mut song = Song { format: format as i32, division, tracks: vec![vec![]] };
292 let mut current = 0;
293 for (i, line) in lines.iter().enumerate().skip(1) {
294 let a: Vec<&str> = line.splitn(4, ' ').collect();
295 let tick: i64 = a[0].parse().unwrap(); let track: usize = a[1].parse().unwrap();
296 let e = text_event(tick, a[2], a.get(3).copied().unwrap_or("")).map_err(|reason| format!("line {}: {reason}", i + 1))?;
297 let bad = |reason: &str| format!("line {}: {reason}", i + 1);
298 if track >= tracks as usize { return Err(bad("bad track")); }
299 let ended = song.tracks[current].last().is_some_and(Event::end);
300 if track < current { return Err(bad("out of order")); }
301 if track > current && (!ended || track > current + 1) { return Err(bad("missing end of track")); }
302 if track == current && ended { return Err(bad("event after end of track")); }
303 if track == current + 1 { current += 1; song.tracks.push(vec![]); }
304 if song.tracks[current].last().is_some_and(|previous| tick < previous.tick) { return Err(bad("out of order")); }
305 song.tracks[current].push(e);
306 }
307 if current + 1 < tracks as usize || !song.tracks[current].last().is_some_and(Event::end) { return Err(format!("line {}: missing end of track", lines.len() + 1)); }
308 Ok(song)
309}
310fn variable(mut value: i64) -> Vec<u8> {
311 let mut result = vec![(value & 127) as u8]; value >>= 7;
312 while value > 0 { result.insert(0, ((value & 127) | 128) as u8); value >>= 7; }
313 result
314}
315fn encode(song: &Song, running: bool, zero_off: bool) {
316 let mut all = b"MThd".to_vec();
317 all.extend(bytes(6, 4)); all.extend(bytes(song.format as i64, 2)); all.extend(bytes(song.tracks.len() as i64, 2)); all.extend(bytes(song.division, 2));
318 for track in &song.tracks {
319 let mut data = vec![]; let mut previous = 0; let mut last = None;
320 for e in track {
321 data.extend(variable(e.tick - previous)); previous = e.tick;
322 if e.channel() {
323 let rewrite = zero_off && e.code() == 8;
324 let status = if rewrite { 0x90 + e.ch() - 1 } else { e.status };
325 if !running || last != Some(status) { data.push(status); }
326 if rewrite { data.extend([e.data[0], 0]); } else { data.extend(&e.data); }
327 last = Some(status);
328 } else {
329 last = None; data.push(e.status);
330 if e.status == 255 { data.push(e.meta as u8); }
331 data.extend(variable(e.data.len() as i64)); data.extend(&e.data);
332 }
333 }
334 all.extend(b"MTrk"); all.extend(bytes(data.len() as i64, 4)); all.extend(data);
335 }
336 for (i, part) in all.chunks(16).enumerate() { println!("{:04X}: {}", i * 16, part.iter().map(|x| format!("{x:02X}")).collect::<Vec<_>>().join(" ")); }
337 println!("{} bytes", all.len());
338}
339fn print_notes(song: &Song) {
340 let (notes, unmatched) = song.pair();
341 println!("notes {} unmatched {}", notes.len(), unmatched.len());
342 for note in notes { let e = note.on.event; println!("{} {} {} {} {}", e.tick, note.off.event.tick - e.tick, e.ch(), pitch(e.data[0]), e.data[1]); }
343 for r in unmatched { let e = r.event; println!("unmatched {} {} {} {} {}", if e.on() { "on" } else { "off" }, e.tick, r.location.track, e.ch(), pitch(e.data[0])); }
344}
345fn stats(song: &Song) {
346 let (notes, unmatched) = song.pair();
347 println!("{}", song.summary()); println!("notes {} unmatched {}", notes.len(), unmatched.len());
348 for ch in 1..=16 {
349 let pitches: Vec<u8> = notes.iter().filter(|n| n.on.event.ch() == ch).map(|n| n.on.event.data[0]).collect();
350 if !pitches.is_empty() { println!("channel {ch} notes {} range {} {}", pitches.len(), pitch(*pitches.iter().min().unwrap()), pitch(*pitches.iter().max().unwrap())); }
351 }
352 let pitches: Vec<u8> = notes.iter().map(|n| n.on.event.data[0]).collect();
353 if pitches.is_empty() { println!("range none"); } else { println!("range {} {}", pitch(*pitches.iter().min().unwrap()), pitch(*pitches.iter().max().unwrap())); }
354 println!("length {} ticks {} ms", song.length(), song.position(song.length(), false).0);
355 let mut changes: BTreeMap<i64, i32> = BTreeMap::new();
356 for note in notes.iter().filter(|n| n.off.event.tick > n.on.event.tick) {
357 *changes.entry(note.on.event.tick).or_default() += 1; *changes.entry(note.off.event.tick).or_default() -= 1;
358 }
359 let (mut current, mut maximum, mut at) = (0, 0, 0);
360 for (tick, change) in changes { current += change; if current > maximum { maximum = current; at = tick; } }
361 if maximum == 0 { println!("polyphony 0"); } else { println!("polyphony {maximum} at {at}"); }
362}
363struct Codec { songs: BTreeMap<String, Song>, running: bool, zero_off: bool }
364impl Codec {
365 fn execute(&mut self, a: &[&str]) -> Result<()> {
366 let cmd = a[0];
367 let known = ["DECODE", "ENCODE", "TIMING", "NOTES", "STATS", "ROLL", "TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "FILTER", "MERGE", "SPLIT", "COPY", "LIST", "RUNNING", "NOTEOFF"];
368 if !known.contains(&cmd) { return Err("unknown command".into()); }
369 if cmd == "LIST" {
370 if self.songs.is_empty() { println!("no songs"); }
371 for (name, song) in &self.songs { println!("{name} {}", song.summary()); }
372 return Ok(());
373 }
374 if matches!(cmd, "RUNNING" | "NOTEOFF") {
375 if cmd == "RUNNING" {
376 if !matches!(a[1], "ON" | "OFF") { return Err("bad argument".into()); }
377 self.running = a[1] == "ON";
378 } else {
379 if !matches!(a[1], "ZERO" | "KEEP") { return Err("bad argument".into()); }
380 self.zero_off = a[1] == "ZERO";
381 }
382 println!("{} {}", cmd.to_lowercase(), a[1].to_lowercase());
383 return Ok(());
384 }
385 let mut song = self.songs.get(a[1]).ok_or_else(|| format!("no song {}", a[1]))?.clone();
386 let first = a.get(2).and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);
387 let second = a.get(3).and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);
388 let third = a.get(4).and_then(|s| s.parse::<i64>().ok()).unwrap_or(0);
389 let bad = match cmd {
390 "TRANSPOSE" => !inside(first, -127, 127) || a.len() == 4 && !inside(second, 1, 16),
391 "QUANTIZE" => !inside(first, 1, 1000000) || !inside(second, 0, 100),
392 "VELOCITY" => !inside(first, 0, 1000) || !inside(second, -127, 127),
393 "TEMPO" => !inside(first, 1, 1000) || !inside(second, 1, 1000),
394 "FILTER" => a[2..].iter().any(|s| !inside(s.parse().unwrap(), 1, 16)),
395 "ROLL" => !inside(first, 0, 10000000) || !inside(second, 1, 10000000) || !inside(third, 1, 10000000) || second <= first || (second - first + third - 1) / third > 64,
396 _ => false,
397 };
398 if bad { return Err("bad argument".into()); }
399 if matches!(cmd, "TIMING" | "STATS") && song.format == 2 { return Err("format 2 not supported".into()); }
400 match cmd {
401 "DECODE" => {
402 println!("HEADER {} {} {}", song.format, song.tracks.len(), song.division);
403 for (tr, track) in song.tracks.iter().enumerate() { for e in track { println!("{}", e.line(tr)); } }
404 }
405 "ENCODE" => encode(&song, self.running, self.zero_off),
406 "TIMING" => {
407 for r in song.global() { let e = r.event; let p = song.position(e.tick, true); println!("{} {} {} {} {}", e.tick, r.location.track, e.kind(), p.0, p.1); }
408 let p = song.position(song.length(), true); println!("end {} {} {}", song.length(), p.0, p.1);
409 }
410 "NOTES" => print_notes(&song),
411 "COPY" => { self.songs.insert(a[2].into(), song); println!("copied {} to {}", a[1], a[2]); return Ok(()); }
412 "TRANSPOSE" => {
413 let mut kept = 0; let mut dropped = vec![];
414 for (tr, track) in song.tracks.iter_mut().enumerate() {
415 let mut result = vec![];
416 for mut e in std::mem::take(track) {
417 if e.channel() && (8..=10).contains(&e.code()) && (a.len() == 3 || e.ch() as i64 == second) {
418 let value = e.data[0] as i64 + first;
419 if !inside(value, 0, 127) { dropped.push(format!("dropped {} {tr} {} {} {}", e.tick, e.kind(), e.ch(), pitch(e.data[0]))); continue; }
420 e.data[0] = value as u8; kept += 1;
421 }
422 result.push(e);
423 }
424 *track = result;
425 }
426 println!("transposed {kept} dropped {}", dropped.len()); for line in dropped { println!("{line}"); }
427 }
428 "QUANTIZE" => {
429 let (notes, unmatched) = song.pair();
430 let mut shifts = BTreeMap::new(); let (mut targets, mut moved) = (0, 0);
431 let mut shift = |on: &Reference, off: Option<&Reference>| {
432 let t = on.event.tick; let low = t / first * first;
433 let g = if t - low <= low + first - t { low } else { low + first };
434 let delta = (2 * (g - t) * second + 100).div_euclid(200);
435 targets += 1; if delta != 0 { moved += 1; }
436 shifts.insert(on.location, delta); if let Some(off) = off { shifts.insert(off.location, delta); }
437 };
438 for note in &notes { shift(&note.on, Some(&note.off)); }
439 for r in unmatched.iter().filter(|r| r.event.on()) { shift(r, None); }
440 for (tr, track) in song.tracks.iter_mut().enumerate() {
441 let mut eot = track.pop().unwrap();
442 for (i, e) in track.iter_mut().enumerate() { e.tick += shifts.get(&Location { track: tr, index: i }).copied().unwrap_or(0); }
443 track.sort_by_key(|e| e.tick);
444 eot.tick = eot.tick.max(track.last().map_or(0, |e| e.tick)); track.push(eot);
445 }
446 println!("quantized {targets} moved {moved}");
447 }
448 "VELOCITY" => {
449 let (mut count, mut clamped) = (0, 0);
450 for e in song.tracks.iter_mut().flatten().filter(|e| e.on()) {
451 let value = (e.data[1] as i64 * first + 50) / 100 + second;
452 count += 1; if !inside(value, 1, 127) { clamped += 1; }
453 e.data[1] = value.clamp(1, 127) as u8;
454 }
455 println!("velocity {count} clamped {clamped}");
456 }
457 "TEMPO" => {
458 let (mut scaled, mut at_zero) = (0, false);
459 let scale = |value: i64| -> Result<i64> {
460 let result = (2 * value * second + first) / (2 * first);
461 if inside(result, 1, 16777215) { Ok(result) } else { Err("tempo out of range".into()) }
462 };
463 for e in song.tracks.iter_mut().flatten().filter(|e| e.status == 255 && e.meta == 81) {
464 if e.tick == 0 { at_zero = true; } scaled += 1; e.data = bytes(scale(e.number())?, 3);
465 }
466 let inserted = if !at_zero { Some(Event { tick: 0, status: 255, meta: 81, data: bytes(scale(500000)?, 3) }) } else { None };
467 if let Some(e) = &inserted { song.tracks[0].insert(0, e.clone()); }
468 println!("scaled {scaled}"); if let Some(e) = inserted { println!("inserted {}", e.line(0)); }
469 }
470 "FILTER" => {
471 let channels: BTreeSet<u8> = a[2..].iter().map(|s| s.parse().unwrap()).collect();
472 let mut removed = 0;
473 for track in &mut song.tracks { let count = track.len(); track.retain(|e| !e.channel() || channels.contains(&e.ch())); removed += count - track.len(); }
474 println!("removed {removed}");
475 }
476 "MERGE" => {
477 if song.format != 1 { return Err("not format 1".into()); }
478 let old = song.tracks.len(); let length = song.length();
479 let mut track: Vec<Event> = song.global().into_iter().filter(|r| !r.event.end()).map(|r| r.event).collect();
480 track.push(end_event(length)); song.tracks = vec![track]; song.format = 0;
481 println!("merged {old} tracks events {}", song.count());
482 }
483 "SPLIT" => {
484 if song.format != 0 { return Err("not format 0".into()); }
485 let length = song.length(); let original = song.tracks.remove(0);
486 let channels: BTreeSet<u8> = original.iter().filter(|e| e.channel()).map(Event::ch).collect();
487 let mut tracks = vec![original.iter().filter(|e| !e.channel() && !e.end()).cloned().collect::<Vec<_>>()];
488 for ch in channels { tracks.push(original.iter().filter(|e| e.channel() && e.ch() == ch).cloned().collect()); }
489 for track in &mut tracks { track.push(end_event(length)); }
490 song.tracks = tracks; song.format = 1;
491 println!("split tracks {} events {}", song.tracks.len(), song.count());
492 }
493 "STATS" => stats(&song),
494 "ROLL" => {
495 let notes = song.pair().0; let mut any = false;
496 for number in (0..128u8).rev() {
497 let mut row = String::new(); let mut at = first;
498 while at < second {
499 let onset = notes.iter().any(|n| n.on.event.data[0] == number && at <= n.on.event.tick && n.on.event.tick < at + third);
500 let held = notes.iter().any(|n| n.on.event.data[0] == number && n.on.event.tick <= at && at < n.off.event.tick);
501 row.push(if onset { '#' } else if held { '=' } else { '.' }); at += third;
502 }
503 if row.bytes().any(|c| c != b'.') { any = true; println!("{:<4} {row}", pitch(number)); }
504 }
505 if !any { println!("empty"); }
506 }
507 _ => unreachable!(),
508 }
509 self.songs.insert(a[1].into(), song);
510 Ok(())
511 }
512}
513fn main() {
514 let stdin = io::stdin(); let mut lines = stdin.lock().lines();
515 let count: usize = lines.next().unwrap().unwrap().parse().unwrap();
516 let mut codec = Codec { songs: BTreeMap::new(), running: true, zero_off: false };
517 for _ in 0..count {
518 let line = lines.next().unwrap().unwrap(); let a: Vec<&str> = line.split(' ').collect();
519 let result = if matches!(a[0], "HEX" | "TEXT") {
520 let count: usize = a[2].parse().unwrap();
521 let data_lines: Vec<String> = (0..count).map(|_| lines.next().unwrap().unwrap()).collect();
522 let parsed = if a[0] == "TEXT" { parse_text(&data_lines) }
523 else {
524 let data: Vec<u8> = data_lines.iter().flat_map(|s| s.split(' ').map(|b| u8::from_str_radix(b, 16).unwrap())).collect();
525 parse_hex(&data)
526 };
527 parsed.map(|song| { println!("loaded {} {}", a[1], song.summary()); codec.songs.insert(a[1].into(), song); })
528 } else { codec.execute(&a) };
529 if let Err(reason) = result { println!("error: {reason}"); }
530 }
