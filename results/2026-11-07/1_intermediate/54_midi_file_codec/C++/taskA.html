140int metaType(Kind kind) {
141 if (textEvent(kind))
142 return static_cast<int>(kind) - static_cast<int>(Kind::Text) + 1;
143 switch (kind) {
144 case Kind::SequenceNumber:
145 return 0x00;
146 case Kind::ChannelPrefix:
147 return 0x20;
148 case Kind::End:
149 return 0x2F;
150 case Kind::Tempo:
151 return 0x51;
152 case Kind::SmpteOffset:
153 return 0x54;
154 case Kind::TimeSignature:
155 return 0x58;
156 case Kind::KeySignature:
157 return 0x59;
158 case Kind::SequencerSpecific:
159 return 0x7F;
160 default:
161 return -1;
162 }
163}

165Kind metaKind(int type) {
166 for (int i = static_cast<int>(Kind::SequenceNumber); i < static_cast<int>(Kind::Meta); ++i) {
167 const Kind kind = static_cast<Kind>(i);
168 if (metaType(kind) == type)
169 return kind;
170 }
171 return Kind::Meta;
172}

174std::uint32_t bigEndian(const Bytes &data, std::size_t position, int count) {
175 std::uint32_t value = 0;
176 for (int i = 0; i < count; ++i)
177 value = value * 256 + data[position + i];
178 return value;
179}

181struct Chunk {
182 std::size_t begin, end;
183};
184
185Chunk readChunk(const Bytes &data, std::size_t position, std::string_view id) {
186 if (data.size() - position < 8)
187 byteError(position, "truncated chunk");
188 for (std::size_t i = 0; i < 4; ++i) {
189 if (data[position + i] != static_cast<unsigned char>(id[i]))
190 byteError(position, "bad chunk id");
191 }
192 const std::uint64_t end = position + 8ULL + bigEndian(data, position + 4, 4);
193 if (end > data.size())
194 byteError(position, "truncated chunk");
195 return {position + 8, static_cast<std::size_t>(end)};
196}

198class TrackReader {
199 const Bytes &bytes;
200 std::size_t position;
201 std::size_t end;
202 std::size_t eventStart = 0;
203
204 int take() {
205 if (position == end)
206 byteError(eventStart, "truncated event");
207 return bytes[position++];
208 }
209
210 int variable() {
211 const std::size_t start = position;
212 int value = 0;
213 for (int i = 0; i < 4; ++i) {
214 const int next = take();
215 value = value * 128 + (next & 127);
216 if (next < 128)
217 return value;
218 }
219 byteError(start, "variable-length value too long");
220 }
221
222 Bytes payload(int length) {
223 if (static_cast<std::size_t>(length) > end - position)
224 byteError(eventStart, "truncated event");
225 Bytes result(bytes.begin() + position, bytes.begin() + position + length);
226 position += length;
227 return result;
228 }

230 Event meta(int type, Bytes data, Tick tick, std::size_t statusPosition) {
231 const std::map<int, int> lengths = {{0x00, 2}, {0x20, 1}, {0x2F, 0}, {0x51, 3},
232 {0x54, 5}, {0x58, 4}, {0x59, 2}};
233 const auto required = lengths.find(type);
234 if (required != lengths.end() &&
235 data.size() != static_cast<std::size_t>(required->second)) {
236 byteError(statusPosition, "bad meta length");
237 }
238 Event event;
239 event.tick = tick;
240 event.kind = metaKind(type);
241 switch (event.kind) {
242 case Kind::SequenceNumber:
243 event.fields = {static_cast<int>(bigEndian(data, 0, 2))};
244 break;
245 case Kind::ChannelPrefix:
246 if (data[0] > 15)
247 byteError(statusPosition, "bad meta data");
248 event.fields = {data[0] + 1};
249 break;
250 case Kind::Tempo:
251 event.fields = {static_cast<int>(bigEndian(data, 0, 3))};
252 if (event.fields[0] == 0)
253 byteError(statusPosition, "bad meta data");
254 break;
255 case Kind::SmpteOffset:
256 event.fields.assign(data.begin(), data.end());
257 break;
258 case Kind::TimeSignature:
259 if (data[0] == 0 || data[1] > 6)
260 byteError(statusPosition, "bad meta data");
261 event.fields = {data[0], 1 << data[1], data[2], data[3]};
262 break;
263 case Kind::KeySignature: {
264 const int sharps = data[0] < 128 ? data[0] : data[0] - 256;
265 if (!inRange(sharps, -7, 7) || data[1] > 1)
266 byteError(statusPosition, "bad meta data");
267 event.fields = {sharps, data[1]};
268 break;
269 }
270 case Kind::Meta:
271 event.fields = {type};
272 event.data = std::move(data);
273 break;
274 default:
275 event.data = std::move(data);
276 break;
277 }
278 return event;
279 }

281 public:
282 TrackReader(const Bytes &data, Chunk chunk)
283 : bytes(data), position(chunk.begin), end(chunk.end) {
284 }

286 std::vector<Event> read() {
287 std::vector<Event> events;
288 Tick tick = 0;
289 int running = -1;
290 while (position < end) {
291 if (!events.empty() && events.back().kind == Kind::End)
292 byteError(position, "event after end of track");
293 eventStart = position;
294 tick += variable();
295 const std::size_t statusPosition = position;
296 if (position == end)
297 byteError(eventStart, "truncated event");
298 int status = bytes[position];
299 if (status < 128) {
300 if (running < 0)
301 byteError(position, "data byte without running status");
302 status = running;
303 } else {
304 ++position;
305 }
306 Event event;
307 event.tick = tick;
308 if (status < 0xF0) {
309 running = status;
310 const int group = status >> 4;
311 event.kind = static_cast<Kind>(group - 8);
312 event.channel = (status & 15) + 1;
313 const int count = group == 0xC || group == 0xD ? 1 : 2;
314 for (int i = 0; i < count; ++i) {
315 const std::size_t dataPosition = position;
316 const int value = take();
317 if (value >= 128)
318 byteError(dataPosition, "bad data byte");
319 event.fields.push_back(value);
320 }
321 if (event.kind == Kind::PitchBend) {
322 event.fields = {event.fields[0] + 128 * event.fields[1] - 8192};
323 }
324 } else if (status == 0xF0 || status == 0xF7) {
325 running = -1;
326 event.kind = status == 0xF0 ? Kind::Sysex : Kind::SysexEscape;
327 const int length = variable();
328 event.data = payload(length);
329 } else if (status == 0xFF) {
330 running = -1;
331 const int type = take();
332 const int length = variable();
333 event = meta(type, payload(length), tick, statusPosition);
334 } else {
335 byteError(statusPosition, "unknown status byte");
336 }
337 events.push_back(std::move(event));
338 }
339 if (events.empty() || events.back().kind != Kind::End)
340 byteError(end, "missing end of track");
341 return events;
342 }

345Song parseHex(const std::vector<std::string> &lines) {
346 Bytes bytes;
347 for (const std::string &line : lines) {
348 for (const std::string &value : words(line))
349 bytes.push_back(static_cast<unsigned char>(std::stoi(value, nullptr, 16)));
350 }
351 const Chunk header = readChunk(bytes, 0, "MThd");
352 if (header.end - header.begin != 6)
353 byteError(4, "bad header length");
354 Song song;
355 song.format = bigEndian(bytes, 8, 2);
356 const int count = bigEndian(bytes, 10, 2);
357 song.division = bigEndian(bytes, 12, 2);
358 if (song.format > 2)
359 byteError(8, "bad format");
360 if (count == 0 || (song.format == 0 && count != 1))
361 byteError(10, "bad track count");
362 if (song.division & 0x8000)
363 byteError(12, "smpte timing not supported");
364 if (song.division == 0)
365 byteError(12, "bad division");
366 std::size_t position = 14;
367 for (int i = 0; i < count; ++i) {
368 if (position == bytes.size())
369 byteError(position, "missing track");
370 const Chunk chunk = readChunk(bytes, position, "MTrk");
371 song.tracks.push_back(TrackReader(bytes, chunk).read());
372 position = chunk.end;
373 }
374 if (position != bytes.size())
375 byteError(position, "extra data");
376 return song;
377}

379bool uppercaseHex(const std::string &value) {
380 return std::all_of(value.begin(), value.end(), [](char ch) {
381 return (ch >= '0' && ch <= '9') || (ch >= 'A' && ch <= 'F');
382 });

385Event parseTextEvent(const std::string &line, std::size_t number, int &track) {
386 const std::size_t first = line.find(' ');
387 const std::size_t second = line.find(' ', first + 1);
388 const std::size_t third = line.find(' ', second + 1);
389 const std::string type =
390 line.substr(second + 1, third == std::string::npos ? third : third - second - 1);
391 const auto found = std::find(typeNames.begin(), typeNames.end(), type);
392 if (found == typeNames.end())
393 lineError(number, "unknown event type");
394 Event event;
395 event.tick = std::stoll(line.substr(0, first));
396 track = std::stoi(line.substr(first + 1, second - first - 1));
397 event.kind = static_cast<Kind>(found - typeNames.begin());
398 const std::string rest = third == std::string::npos ? "" : line.substr(third + 1);
399 const std::vector<std::string> fields = words(rest);
400 const auto bad = [number]() { lineError(number, "bad value"); };
401 const auto need = [&fields, &bad](std::size_t count) {
402 if (fields.size() != count)
403 bad();
404 };
405 const auto integer = [&fields, &bad](std::size_t index, int low, int high) {
406 const int value = std::stoi(fields[index]);
407 if (!inRange(value, low, high))
408 bad();
409 return value;
410 };
411 const auto blob = [&fields, &event, &bad](std::size_t offset) {
412 if (fields.size() < offset + 1)
413 bad();
414 const int length = std::stoi(fields[offset]);
415 if (!inRange(length, 0, 1000000) || fields.size() != offset + (length == 0 ? 1 : 2))
416 bad();
417 if (length > 0) {
418 const std::string &data = fields[offset + 1];
419 if (data.size() != static_cast<std::size_t>(2 * length) || !uppercaseHex(data))
420 bad();
421 for (std::size_t i = 0; i < data.size(); i += 2) {
422 event.data.push_back(
423 static_cast<unsigned char>(std::stoi(data.substr(i, 2), nullptr, 16)));
424 }
425 }
426 };
427 if (channelEvent(event.kind)) {
428 const bool single = event.kind == Kind::Program || event.kind == Kind::ChannelPressure ||
429 event.kind == Kind::PitchBend;
430 need(single ? 2 : 3);
431 event.channel = integer(0, 1, 16);
432 if (event.kind <= Kind::PolyPressure) {
433 const int pitch = parsePitch(fields[1]);
434 if (pitch < 0)
435 bad();
436 event.fields = {pitch, integer(2, 0, 127)};
437 } else if (event.kind == Kind::PitchBend) {
438 event.fields = {integer(1, -8192, 8191)};
439 } else {
440 event.fields = {integer(1, 0, 127)};
441 if (!single)
442 event.fields.push_back(integer(2, 0, 127));
443 }
444 } else if (textEvent(event.kind)) {
445 if (rest.size() < 2 || rest.front() != '"' || rest.back() != '"' ||
446 rest.substr(1, rest.size() - 2).find('"') != std::string::npos)
447 bad();
448 event.data.assign(rest.begin() + 1, rest.end() - 1);
449 } else {
450 switch (event.kind) {
451 case Kind::Sysex:
452 case Kind::SysexEscape:
453 case Kind::SequencerSpecific:
454 blob(0);
455 break;
456 case Kind::Meta:
457 if (fields.empty() || fields[0].size() != 2 || !uppercaseHex(fields[0]))
458 bad();
459 event.fields = {std::stoi(fields[0], nullptr, 16)};
460 if (metaKind(event.fields[0]) != Kind::Meta)
461 bad();
462 blob(1);
463 break;
464 case Kind::SequenceNumber:
465 need(1);
466 event.fields = {integer(0, 0, 65535)};
467 break;
468 case Kind::ChannelPrefix:
469 need(1);
470 event.fields = {integer(0, 1, 16)};
471 break;
472 case Kind::End:
473 need(0);
474 break;
475 case Kind::Tempo:
476 need(1);
477 event.fields = {integer(0, 1, 16777215)};
478 break;
479 case Kind::SmpteOffset:
480 need(5);
481 for (std::size_t i = 0; i < 5; ++i)
482 event.fields.push_back(integer(i, 0, 255));
483 break;
484 case Kind::TimeSignature: {
485 need(3);
486 const std::size_t slash = fields[0].find('/');
487 const int numerator = std::stoi(fields[0].substr(0, slash));
488 const int denominator = std::stoi(fields[0].substr(slash + 1));
489 if (!inRange(numerator, 1, 255) || !inRange(denominator, 1, 64) ||
490 (denominator & (denominator - 1)) != 0)
491 bad();
492 event.fields = {numerator, denominator, integer(1, 0, 255), integer(2, 0, 255)};
493 break;
494 }
495 case Kind::KeySignature: {
496 need(2);
497 const int mode = fields[1] == "major" ? 0 : fields[1] == "minor" ? 1 : -1;
498 if (mode < 0)
499 bad();
500 const auto key = std::find(keyNames[mode].begin(), keyNames[mode].end(), fields[0]);
501 if (key == keyNames[mode].end())
502 bad();
503 event.fields = {static_cast<int>(key - keyNames[mode].begin()) - 7, mode};
504 break;
505 }
506 default:
507 break;
508 }
509 }
510 return event;
511}

513Song parseText(const std::vector<std::string> &lines) {
514 const std::vector<std::string> header = words(lines[0]);
515 if (header.size() != 4 || header[0] != "HEADER")
516 lineError(1, "bad header");
517 Song song;
518 song.format = std::stoi(header[1]);
519 const int count = std::stoi(header[2]);
520 song.division = std::stoi(header[3]);
521 if (!inRange(song.format, 0, 2) || !inRange(count, 1, 65535) ||
522 !inRange(song.division, 1, 32767) || (song.format == 0 && count != 1))
523 lineError(1, "bad header");
524 song.tracks.resize(count);
525 int current = 0;
526 for (std::size_t i = 1; i < lines.size(); ++i) {
527 int track = 0;
528 Event event = parseTextEvent(lines[i], i + 1, track);
529 if (track >= count)
530 lineError(i + 1, "bad track");
531 const bool ended =
532 !song.tracks[current].empty() && song.tracks[current].back().kind == Kind::End;
533 if (track < current)
534 lineError(i + 1, "out of order");
535 if (track > current && (!ended || track > current + 1))
536 lineError(i + 1, "missing end of track");
537 if (track == current && ended)
538 lineError(i + 1, "event after end of track");
539 if (track > current)
540 ++current;
541 std::vector<Event> &events = song.tracks[current];
542 if (!events.empty() && event.tick < events.back().tick)
543 lineError(i + 1, "out of order");
544 events.push_back(std::move(event));
545 }
546 if (current + 1 < count || song.tracks[current].empty() ||
547 song.tracks[current].back().kind != Kind::End)
548 lineError(lines.size() + 1, "missing end of track");
549 return song;
550}

552std::string eventText(const Event &event, std::size_t track) {
553 std::ostringstream out;
554 out << event.tick << ' ' << track << ' ' << typeNames[static_cast<int>(event.kind)];
555 if (channelEvent(event.kind)) {
556 out << ' ' << event.channel;
557 for (std::size_t i = 0; i < event.fields.size(); ++i) {
558 if (i == 0 && event.kind <= Kind::PolyPressure)
559 out << ' ' << pitchName(event.fields[i]);
560 else
561 out << ' ' << event.fields[i];
562 }
563 } else if (textEvent(event.kind)) {
564 out << " \"" << std::string(event.data.begin(), event.data.end()) << '"';
565 } else if (event.kind == Kind::TimeSignature) {
566 out << ' ' << event.fields[0] << '/' << event.fields[1] << ' ' << event.fields[2] << ' '
567 << event.fields[3];
568 } else if (event.kind == Kind::KeySignature) {
569 out << ' ' << keyNames[event.fields[1]][event.fields[0] + 7] << ' '
570 << (event.fields[1] == 0 ? "major" : "minor");
571 } else if (event.kind == Kind::Sysex || event.kind == Kind::SysexEscape ||
572 event.kind == Kind::SequencerSpecific || event.kind == Kind::Meta) {
573 if (event.kind == Kind::Meta)
574 out << ' ' << hex(event.fields[0], 2);
575 out << ' ' << event.data.size();
576 if (!event.data.empty()) {
577 out << ' ';
578 for (unsigned char byte : event.data)
579 out << hex(byte, 2);
580 }
581 } else {
582 for (int field : event.fields)
583 out << ' ' << field;
584 }
585 return out.str();
586}

588struct EventRef {
589 std::size_t track, index, order;
590};
591const Event &at(const Song &song, EventRef ref) {
592 return song.tracks[ref.track][ref.index];
593}
594Event &at(Song &song, EventRef ref) {
595 return song.tracks[ref.track][ref.index];
596}

598std::vector<EventRef> globalOrder(const Song &song) {
599 std::vector<EventRef> result;
600 for (std::size_t track = 0; track < song.tracks.size(); ++track) {
601 for (std::size_t i = 0; i < song.tracks[track].size(); ++i)
602 result.push_back({track, i, 0});
603 }
604 std::stable_sort(result.begin(), result.end(), [&song](EventRef a, EventRef b) {
605 return at(song, a).tick < at(song, b).tick;
606 });
607 for (std::size_t i = 0; i < result.size(); ++i)
608 result[i].order = i;
609 return result;
610}

612struct Note {
613 Tick start, length;
614 int channel, pitch, velocity;
615 EventRef on, off;
616};
617struct Pairing {
618 std::vector<Note> notes;
619 std::vector<EventRef> unmatched;
620};
621
622Pairing pairNotes(const Song &song) {
623 Pairing result;
624 std::map<std::pair<int, int>, std::deque<EventRef>> pending;
625 for (EventRef ref : globalOrder(song)) {
626 const Event &event = at(song, ref);
627 if (noteOn(event))
628 pending[{event.channel, event.fields[0]}].push_back(ref);
629 else if (noteOff(event)) {
630 std::deque<EventRef> &queue = pending[{event.channel, event.fields[0]}];
631 if (queue.empty())
632 result.unmatched.push_back(ref);
633 else {
634 const EventRef onRef = queue.front();
635 queue.pop_front();
636 const Event &on = at(song, onRef);
637 result.notes.push_back({on.tick, event.tick - on.tick, on.channel, on.fields[0],
638 on.fields[1], onRef, ref});
639 }
640 }
641 }
642 for (const auto &entry : pending) {
643 for (EventRef ref : entry.second)
644 result.unmatched.push_back(ref);
645 }
646 std::sort(result.notes.begin(), result.notes.end(), [](const Note &a, const Note &b) {
647 return std::tie(a.start, a.channel, a.pitch, a.on.order) <
648 std::tie(b.start, b.channel, b.pitch, b.on.order);
649 });
650 std::sort(result.unmatched.begin(), result.unmatched.end(),
651 [](EventRef a, EventRef b) { return a.order < b.order; });
652 return result;
653}

655std::string milliseconds(const Song &song, Tick tick) {
656 Tick sum = 0, last = 0, tempo = 500000;
657 for (EventRef ref : globalOrder(song)) {
658 const Event &event = at(song, ref);
659 if (event.tick > tick)
660 break;
661 if (event.kind == Kind::Tempo) {
662 sum += (event.tick - last) * tempo;
663 last = event.tick;
664 tempo = event.fields[0];
665 }
666 }
667 const Tick micros = (sum + (tick - last) * tempo) / song.division;
668 std::ostringstream out;
669 out << micros / 1000 << '.' << std::setfill('0') << std::setw(3) << micros % 1000;
670 return out.str();
671}

673std::string barPosition(const Song &song, Tick tick) {
674 Tick start = 0, bar = 1, numerator = 4, beat = song.division;
675 for (EventRef ref : globalOrder(song)) {
676 const Event &event = at(song, ref);
677 if (event.tick > tick)
678 break;
679 if (event.kind == Kind::TimeSignature) {
680 const Tick barLength = numerator * beat;
681 bar += (event.tick - start + barLength - 1) / barLength;
682 start = event.tick;
683 numerator = event.fields[0];
684 beat = song.division * 4 / event.fields[1];
685 }
686 }
687 const Tick offset = tick - start;
688 return std::to_string(bar + offset / (numerator * beat)) + ':' +
689 std::to_string(offset % (numerator * beat) / beat + 1) + ':' +
690 std::to_string(offset % beat);
691}

693void appendBig(Bytes &bytes, std::uint64_t value, int count) {
694 for (int i = count - 1; i >= 0; --i)
695 bytes.push_back(static_cast<unsigned char>((value >> (8 * i)) & 255));
696}
697
698void appendVariable(Bytes &bytes, Tick value) {
699 Bytes groups{static_cast<unsigned char>(value & 127)};
700 while ((value >>= 7) != 0)
701 groups.push_back(static_cast<unsigned char>((value & 127) | 128));
702 bytes.insert(bytes.end(), groups.rbegin(), groups.rend());
703}
704
705Bytes metaData(const Event &event) {
706 Bytes data;
707 switch (event.kind) {
708 case Kind::SequenceNumber:
709 appendBig(data, event.fields[0], 2);
710 break;
711 case Kind::ChannelPrefix:
712 data.push_back(static_cast<unsigned char>(event.fields[0] - 1));
713 break;
714 case Kind::Tempo:
715 appendBig(data, event.fields[0], 3);
716 break;
717 case Kind::SmpteOffset:
718 data.assign(event.fields.begin(), event.fields.end());
719 break;
720 case Kind::TimeSignature: {
721 int exponent = 0;
722 while ((1 << exponent) != event.fields[1])
723 ++exponent;
724 data = {static_cast<unsigned char>(event.fields[0]), static_cast<unsigned char>(exponent),
725 static_cast<unsigned char>(event.fields[2]),
726 static_cast<unsigned char>(event.fields[3])};
727 break;
728 }
729 case Kind::KeySignature:
730 data = {static_cast<unsigned char>(event.fields[0]),
731 static_cast<unsigned char>(event.fields[1])};
732 break;
733 default:
734 data = event.data;
735 break;
736 }
737 return data;
738}

740Bytes encode(const Song &song, bool runningEnabled, bool zeroOff) {
741 Bytes result{'M', 'T', 'h', 'd'};
742 appendBig(result, 6, 4);
743 appendBig(result, song.format, 2);
744 appendBig(result, song.tracks.size(), 2);
745 appendBig(result, song.division, 2);
746 for (const std::vector<Event> &track : song.tracks) {
747 Bytes data;
748 Tick previous = 0;
749 int running = -1;
750 for (const Event &event : track) {
751 appendVariable(data, event.tick - previous);
752 previous = event.tick;
753 if (channelEvent(event.kind)) {
754 int group = static_cast<int>(event.kind) + 8;
755 std::vector<int> fields = event.fields;
756 if (event.kind == Kind::NoteOff && zeroOff) {
757 group = 9;
758 fields[1] = 0;
759 }
760 const int status = group * 16 + event.channel - 1;
761 if (!runningEnabled || running != status)
762 data.push_back(static_cast<unsigned char>(status));
763 running = status;
764 if (event.kind == Kind::PitchBend) {
765 const int value = fields[0] + 8192;
766 data.push_back(static_cast<unsigned char>(value % 128));
767 data.push_back(static_cast<unsigned char>(value / 128));
768 } else {
769 for (int field : fields)
770 data.push_back(static_cast<unsigned char>(field));
771 }
772 } else {
773 running = -1;
774 if (event.kind == Kind::Sysex || event.kind == Kind::SysexEscape) {
775 data.push_back(event.kind == Kind::Sysex ? 0xF0 : 0xF7);
776 appendVariable(data, event.data.size());
777 data.insert(data.end(), event.data.begin(), event.data.end());
778 } else {
779 data.push_back(0xFF);
780 data.push_back(static_cast<unsigned char>(
781 event.kind == Kind::Meta ? event.fields[0] : metaType(event.kind)));
782 const Bytes payload = metaData(event);
783 appendVariable(data, payload.size());
784 data.insert(data.end(), payload.begin(), payload.end());
785 }
786 }
787 }
788 result.insert(result.end(), {'M', 'T', 'r', 'k'});
789 appendBig(result, data.size(), 4);
790 result.insert(result.end(), data.begin(), data.end());
791 }
792 return result;
793}

807void transpose(Song &song, int shift, int channel) {
808 std::vector<std::string> dropped;
809 int kept = 0;
810 for (std::size_t t = 0; t < song.tracks.size(); ++t) {
811 std::vector<Event> track;
812 for (Event event : song.tracks[t]) {
813 if (event.kind <= Kind::PolyPressure && (channel == 0 || event.channel == channel)) {
814 const int pitch = event.fields[0] + shift;
815 if (!inRange(pitch, 0, 127)) {
816 dropped.push_back(
817 "dropped " + std::to_string(event.tick) + ' ' + std::to_string(t) + ' ' +
818 typeNames[static_cast<int>(event.kind)] + ' ' +
819 std::to_string(event.channel) + ' ' + pitchName(event.fields[0]));
820 continue;
821 }
822 event.fields[0] = pitch;
823 ++kept;
824 }
825 track.push_back(std::move(event));
826 }
827 song.tracks[t] = std::move(track);
828 }
829 std::cout << "transposed " << kept << " dropped " << dropped.size() << '\n';
830 for (const std::string &line : dropped)
831 std::cout << line << '\n';
832}

834void quantize(Song &song, Tick grid, int strength) {
835 const Pairing pairing = pairNotes(song);
836 std::size_t targets = pairing.notes.size();
837 int moved = 0;
838 const auto delta = [grid, strength](Tick start) {
839 const Tick lower = start / grid * grid;
840 const Tick upper = lower + grid;
841 const Tick target = start - lower <= upper - start ? lower : upper;
842 return floorDiv(2 * (target - start) * strength + 100, 200);
843 };
844 for (const Note &note : pairing.notes) {
845 const Tick shift = delta(note.start);
846 at(song, note.on).tick += shift;
847 at(song, note.off).tick += shift;
848 if (shift != 0)
849 ++moved;
850 }
851 for (EventRef ref : pairing.unmatched) {
852 Event &event = at(song, ref);
853 if (noteOn(event)) {
854 ++targets;
855 const Tick shift = delta(event.tick);
856 event.tick += shift;
857 if (shift != 0)
858 ++moved;
859 }
860 }
861 for (std::vector<Event> &track : song.tracks) {
862 Event ending = track.back();
863 track.pop_back();
864 std::stable_sort(track.begin(), track.end(),
865 [](const Event &a, const Event &b) { return a.tick < b.tick; });
866 if (!track.empty())
867 ending.tick = std::max(ending.tick, track.back().tick);
868 track.push_back(std::move(ending));
869 }
870 std::cout << "quantized " << targets << " moved " << moved << '\n';
871}

873void scaleTempo(Song &song, int numerator, int denominator) {
874 Song scaled = song;
875 int count = 0;
876 bool atZero = false;
877 const auto convert = [numerator, denominator](Tick value) {
878 const Tick result = (2 * value * denominator + numerator) / (2 * numerator);
879 if (!inRange(result, 1, 16777215))
880 fail("tempo out of range");
881 return static_cast<int>(result);
882 };
883 for (std::vector<Event> &track : scaled.tracks) {
884 for (Event &event : track) {
885 if (event.kind == Kind::Tempo) {
886 atZero = atZero || event.tick == 0;
887 event.fields[0] = convert(event.fields[0]);
888 ++count;
889 }
890 }
891 }
892 if (!atZero) {
893 Event inserted;
894 inserted.kind = Kind::Tempo;
895 inserted.fields = {convert(500000)};
896 scaled.tracks[0].insert(scaled.tracks[0].begin(), inserted);
897 }
898 song = std::move(scaled);
899 std::cout << "scaled " << count << '\n';
900 if (!atZero)
901 std::cout << "inserted " << eventText(song.tracks[0][0], 0) << '\n';
902}

904void printStats(const Song &song) {
905 const Pairing pairing = pairNotes(song);
906 printSummary(song);
907 std::cout << "\nnotes " << pairing.notes.size() << " unmatched " << pairing.unmatched.size()
908 << '\n';
909 int low = 128, high = -1;
910 for (int channel = 1; channel <= 16; ++channel) {
911 int count = 0, channelLow = 128, channelHigh = -1;
912 for (const Note &note : pairing.notes) {
913 if (note.channel == channel) {
914 ++count;
915 channelLow = std::min(channelLow, note.pitch);
916 channelHigh = std::max(channelHigh, note.pitch);
917 }
918 }
919 if (count > 0) {
920 std::cout << "channel " << channel << " notes " << count << " range "
921 << pitchName(channelLow) << ' ' << pitchName(channelHigh) << '\n';
922 low = std::min(low, channelLow);
923 high = std::max(high, channelHigh);
924 }
925 }
926 if (high < 0)
927 std::cout << "range none\n";
928 else
929 std::cout << "range " << pitchName(low) << ' ' << pitchName(high) << '\n';
930 std::cout << "length " << song.length() << " ticks " << milliseconds(song, song.length())
931 << " ms\n";
932 std::map<Tick, int> changes;
933 for (const Note &note : pairing.notes) {
934 if (note.length > 0) {
935 ++changes[note.start];
936 --changes[note.start + note.length];
937 }
938 }
939 int current = 0, maximum = 0;
940 Tick first = 0;
941 for (const auto &change : changes) {
942 current += change.second;
943 if (current > maximum) {
944 maximum = current;
945 first = change.first;
946 }
947 }
948 std::cout << "polyphony " << maximum;
949 if (maximum > 0)
950 std::cout << " at " << first;
951 std::cout << '\n';
952}

954void printRoll(const Song &song, Tick from, Tick to, Tick step) {
955 const Pairing pairing = pairNotes(song);
956 bool printed = false;
957 for (int pitch = 127; pitch >= 0; --pitch) {
958 std::string row;
959 for (Tick column = from; column < to; column += step) {
960 char value = '.';
961 for (const Note &note : pairing.notes) {
962 if (note.pitch != pitch)
963 continue;
964 if (column <= note.start && note.start < column + step) {
965 value = '#';
966 break;
967 }
968 if (note.start <= column && column < note.start + note.length)
969 value = '=';
970 }
971 row.push_back(value);
972 }
973 if (row.find_first_not_of('.') != std::string::npos) {
974 const std::string name = pitchName(pitch);
975 std::cout << name << std::string(4 - name.size(), ' ') << ' ' << row << '\n';
976 printed = true;
977 }
978 }
979 if (!printed)
980 std::cout << "empty\n";
981}
