1CHANNELS = %w[NOTE_OFF NOTE_ON POLY_PRESSURE CONTROL PROGRAM CHANNEL_PRESSURE PITCH_BEND].freeze
2TEXTS = %w[TEXT_EVENT COPYRIGHT TRACK_NAME INSTRUMENT_NAME LYRIC MARKER CUE_POINT].freeze
3METAS = {0=>'SEQUENCE_NUMBER',32=>'CHANNEL_PREFIX',47=>'END_OF_TRACK',81=>'TEMPO',84=>'SMPTE_OFFSET',88=>'TIME_SIGNATURE',89=>'KEY_SIGNATURE',127=>'SEQUENCER_SPECIFIC'}.merge(TEXTS.each_with_index.to_h { |s,i| [i+1,s] }).freeze
4MAJOR = %w[Cb Gb Db Ab Eb Bb F C G D A E B F# C#].freeze
5MINOR = %w[Ab Eb Bb F C G D A E B F# C# G# D# A#].freeze
6NOTES = (0..127).map { |p| %w[C C# D D# E F F# G G# A A# B][p%12] + (p/12-1).to_s }.freeze

8def integer(bytes)
9 bytes.reduce(0) { |v,b| v*256+b }
10end
11
12def big(value, count)
13 Array.new(count) { |i| (value >> (8*(count-1-i))) & 255 }
14end
15
16def vlq(value)
17 bytes = [value & 127]
18 while (value >>= 7) > 0
19 bytes.unshift((value & 127) | 128)
20 end
21 bytes
22end

24def event_type(e)
25 return CHANNELS[(e[:s] >> 4)-8] if e[:s] < 240
26 return e[:s] == 240 ? 'SYSEX' : 'SYSEX_ESCAPE' if e[:s] != 255
27 METAS[e[:m]] || 'META'
28end
29
30def event_fields(e)
31 d, s = e[:d], e[:s]
32 if s < 240
33 ch, kind = (s&15)+1, s>>4
34 return [ch, NOTES[d[0]], d[1]] if [8,9,10].include?(kind)
35 return [ch, d[0]+128*d[1]-8192] if kind == 14
36 return [ch] + d
37 end
38 if s != 255 || e[:m] == 127 || !METAS.key?(e[:m])
39 prefix = s == 255 && !METAS.key?(e[:m]) ? [format('%02X',e[:m])] : []
40 return prefix + [d.size] + (d.empty? ? [] : [d.pack('C*').unpack1('H*').upcase])
41 end
42 case e[:m]
43 when 0,81 then [integer(d)]
44 when 1..7 then ['"' + d.pack('C*') + '"']
45 when 32 then [d[0]+1]
46 when 47 then []
47 when 84 then d
48 when 88 then ["#{d[0]}/#{1<<d[1]}",d[2],d[3]]
49 when 89
50 sf = d[0] >= 128 ? d[0]-256 : d[0]
51 [(d[1] == 0 ? MAJOR : MINOR)[sf+7], d[1] == 0 ? 'major' : 'minor']
52 end
53end

55def event_line(e, track)
56 ([e[:t],track,event_type(e)] + event_fields(e)).join(' ')
57end

59def ordered(song)
60 song[:tracks].each_with_index.flat_map { |tr,i| tr.each_with_index.map { |e,j| [e,i,j] } }.sort_by { |e,i,j| [e[:t],i,j] }
61end

71class Reader
72 def initialize(bytes)
73 @bytes, @p, @limit, @start = bytes, 0, bytes.size, 0
74 end
75 def fail_at(reason, position)
76 raise "byte #{position}: #{reason}"
77 end
78 def read
79 fail_at('truncated event',@start) if @p >= @limit
80 b = @bytes[@p]
81 @p += 1
82 b
83 end
84 def variable
85 start, value = @p, 0
86 4.times do
87 b = read
88 value = value*128+(b&127)
89 return value if b < 128
90 end
91 fail_at('variable-length value too long',start)
92 end
93 def chunk(id)
94 p = @p
95 fail_at('truncated chunk',p) if @bytes.size-p < 8
96 fail_at('bad chunk id',p) unless @bytes[p,4].pack('C*') == id
97 size = integer(@bytes[p+4,4])
98 fail_at('truncated chunk',p) if p+8+size > @bytes.size
99 @p = p+8
100 size
101 end
102 def parse
103 size = chunk('MThd')
104 fail_at('bad header length',4) unless size == 6
105 f, count, division = [8,10,12].map { |i| integer(@bytes[i,2]) }
106 fail_at('bad format',8) if f > 2
107 fail_at('bad track count',10) if count == 0 || (f == 0 && count != 1)
108 fail_at('smpte timing not supported',12) if division >= 32768
109 fail_at('bad division',12) if division == 0
110 @p = 14
111 tracks = []
112 count.times do
113 fail_at('missing track',@bytes.size) if @p == @bytes.size
114 size = chunk('MTrk')
115 @limit = @p+size
116 track, tick, running, ended = [], 0, nil, false
117 while @p < @limit
118 fail_at('event after end of track',@p) if ended
119 @start = @p
120 tick += variable
121 fail_at('truncated event',@start) if @p >= @limit
122 status_position = @p
123 s = @bytes[@p]
124 if s < 128
125 fail_at('data byte without running status',@p) unless running
126 s = running
127 else
128 @p += 1
129 end
130 e = {t:tick,s:s,d:[]}
131 if s < 240
132 running = s
133 ((s>>4) == 12 || (s>>4) == 13 ? 1 : 2).times do
134 pos = @p
135 b = read
136 fail_at('bad data byte',pos) if b >= 128
137 e[:d] << b
138 end
139 elsif [240,247,255].include?(s)
140 running = nil
141 e[:m] = read if s == 255
142 n = variable
143 fail_at('truncated event',@start) if @p+n > @limit
144 e[:d] = @bytes[@p,n]
145 @p += n
146 if s == 255
147 fixed = {0=>2,32=>1,47=>0,81=>3,84=>5,88=>4,89=>2}
148 fail_at('bad meta length',status_position) if fixed.key?(e[:m]) && n != fixed[e[:m]]
149 d = e[:d]
150 bad = case e[:m]
151 when 32 then d[0] > 15
152 when 81 then integer(d) == 0
153 when 88 then d[0] == 0 || d[1] > 6
154 when 89 then !(0..7).include?(d[0]) && !(249..255).include?(d[0]) || d[1] > 1
155 else false
156 end
157 fail_at('bad meta data',status_position) if bad
158 ended = e[:m] == 47
159 end
160 else
161 fail_at('unknown status byte',status_position)
162 end
163 track << e
164 end
165 fail_at('missing end of track',@limit) unless ended
166 tracks << track
167 @limit = @bytes.size
168 end
169 fail_at('extra data',@p) if @p < @bytes.size
170 {f:f,div:division,tracks:tracks}
171 end
172end

174def valid_range(s, low, high)
175 s && s.match?(/\A-?\d+\z/) && s.to_i.between?(low,high)
176end

178def text_event(line)
179 tick, track, type, rest = line.split(' ',4)
180 known = CHANNELS + METAS.values + %w[SYSEX SYSEX_ESCAPE META]
181 raise 'unknown event type' unless known.include?(type)
182 fields = TEXTS.include?(type) ? [rest] : (rest || '').split
183 e = {t:tick.to_i,s:255,d:[]}
184 e[:m] = METAS.key(type) if METAS.value?(type)
185 good = true
186 if CHANNELS.include?(type)
187 kind = CHANNELS.index(type)+8
188 count = [12,13,14].include?(kind) ? 2 : 3
189 good = fields.size == count && valid_range(fields[0],1,16)
190 if [8,9,10].include?(kind)
191 note = NOTES.index(fields[1])
192 good &&= !note.nil? && valid_range(fields[2],0,127)
193 e[:d] = [note,fields[2].to_i]
194 elsif kind == 14
195 good &&= valid_range(fields[1],-8192,8191)
196 value = fields[1].to_i+8192
197 e[:d] = [value%128,value/128]
198 else
199 good &&= fields.drop(1).all? { |x| valid_range(x,0,127) }
200 e[:d] = fields.drop(1).map(&:to_i)
201 end
202 e[:s] = kind*16+fields[0].to_i-1
203 elsif TEXTS.include?(type)
204 good = rest && rest.size >= 2 && rest.start_with?('"') && rest.end_with?('"') && !rest[1...-1].include?('"')
205 e[:d] = good ? rest[1...-1].bytes : []
206 elsif %w[SYSEX SYSEX_ESCAPE SEQUENCER_SPECIFIC META].include?(type)
207 if type == 'META'
208 tt = fields.shift
209 good = tt && tt.match?(/\A[0-9A-F]{2}\z/) && !METAS.key?(tt.to_i(16))
210 e[:m] = tt.to_i(16) if tt
211 elsif type == 'SYSEX' || type == 'SYSEX_ESCAPE'
212 e[:s] = type == 'SYSEX' ? 240 : 247
213 end
214 good &&= valid_range(fields[0],0,1_000_000)
215 n = fields[0].to_i
216 good &&= fields.size == (n == 0 ? 1 : 2)
217 good &&= fields[1] && fields[1].size == 2*n && fields[1].match?(/\A[0-9A-F]+\z/) if n > 0
218 e[:d] = n > 0 && good ? [fields[1]].pack('H*').bytes : []
219 else
220 case type
221 when 'SEQUENCE_NUMBER','TEMPO','CHANNEL_PREFIX'
222 range = {'SEQUENCE_NUMBER'=>[0,65535],'TEMPO'=>[1,16777215],'CHANNEL_PREFIX'=>[1,16]}[type]
223 good = fields.size == 1 && valid_range(fields[0],*range)
224 v = fields[0].to_i
225 e[:d] = type == 'CHANNEL_PREFIX' ? [v-1] : big(v,type == 'TEMPO' ? 3 : 2)
226 when 'END_OF_TRACK' then good = fields.empty?
227 when 'SMPTE_OFFSET'
228 good = fields.size == 5 && fields.all? { |x| valid_range(x,0,255) }
229 e[:d] = fields.map(&:to_i)
230 when 'TIME_SIGNATURE'
231 nn, dd = (fields[0] || '').split('/').map(&:to_i)
232 good = fields.size == 3 && nn && nn.between?(1,255) && [1,2,4,8,16,32,64].include?(dd) && fields.drop(1).all? { |x| valid_range(x,0,255) }
233 e[:d] = good ? [nn, [1,2,4,8,16,32,64].index(dd),fields[1].to_i,fields[2].to_i] : []
234 when 'KEY_SIGNATURE'
235 keys = fields[1] == 'major' ? MAJOR : MINOR
236 index = keys.index(fields[0])
237 good = fields.size == 2 && %w[major minor].include?(fields[1]) && index
238 e[:d] = good ? [(index-7)%256,fields[1] == 'major' ? 0 : 1] : []
239 end
240 end
241 raise 'bad value' unless good
242 [e,track.to_i]
243end

245def parse_text(lines)
246 header = lines[0].split
247 unless header.size == 4 && header[0] == 'HEADER' && valid_range(header[1],0,2) && valid_range(header[2],1,65535) && valid_range(header[3],1,32767) && (header[1].to_i != 0 || header[2].to_i == 1)
248 raise 'line 1: bad header'
249 end
250 f, count, division = header.drop(1).map(&:to_i)
251 tracks, c = [[]], 0
252 lines.drop(1).each_with_index do |line,i|
253 begin
254 e, tr = text_event(line)
255 raise 'bad track' if tr >= count
256 ended = !tracks[c].empty? && event_type(tracks[c][-1]) == 'END_OF_TRACK'
257 raise 'out of order' if tr < c
258 raise 'missing end of track' if tr > c && (!ended || tr > c+1)
259 raise 'event after end of track' if tr == c && ended
260 if tr == c+1
261 c += 1
262 tracks << []
263 end
264 raise 'out of order' if !tracks[c].empty? && e[:t] < tracks[c][-1][:t]
265 tracks[c] << e
266 rescue RuntimeError => ex
267 raise "line #{i+2}: #{ex.message}"
268 end
269 end
270 if c+1 < count || tracks[c].empty? || event_type(tracks[c][-1]) != 'END_OF_TRACK'
271 raise "line #{lines.size+1}: missing end of track"
272 end
273 {f:f,div:division,tracks:tracks}
274end

276def pairing(song)
277 queues, notes, unmatched = {}, [], []
278 ordered(song).each_with_index do |(e,tr,_),order|
279 kind = e[:s]>>4
280 next unless kind == 8 || kind == 9
281 key = [e[:s]&15,e[:d][0]]
282 on = kind == 9 && e[:d][1] > 0
283 item = [e,tr,order,on]
284 queues[key] ||= []
285 if on
286 queues[key] << item
287 elsif queues[key].empty?
288 unmatched << item
289 else
290 first = queues[key].shift
291 notes << {on:first[0],off:e,order:first[2],start:first[0][:t],length:e[:t]-first[0][:t],ch:key[0]+1,p:key[1],v:first[0][:d][1]}
292 end
293 end
294 queues.each_value { |q| unmatched.concat(q) }
295 notes.sort_by! { |n| [n[:start],n[:ch],n[:p],n[:order]] }
296 [notes,unmatched.sort_by { |item| item[2] }]
297end

299def time_position(song, tick)
300 sum, last, tempo, st, bar, nn, beat = 0,0,500000,0,1,4,song[:div]
301 ordered(song).each do |e,_,_|
302 break if e[:t] > tick
303 if event_type(e) == 'TEMPO'
304 sum += (e[:t]-last)*tempo
305 last, tempo = e[:t], integer(e[:d])
306 elsif event_type(e) == 'TIME_SIGNATURE'
307 size = nn*beat
308 bar += (e[:t]-st+size-1)/size
309 st, nn, beat = e[:t],e[:d][0],[1,song[:div]*4/(1<<e[:d][1])].max
310 end
311 end
312 micro = (sum+(tick-last)*tempo)/song[:div]
313 x = tick-st
314 ["#{micro/1000}.#{(micro%1000).to_s.rjust(3,'0')}","#{bar+x/(nn*beat)}:#{x%(nn*beat)/beat+1}:#{x%beat}"]
315end

317def encode(song, running, noteoff)
318 bytes = 'MThd'.bytes + big(6,4) + big(song[:f],2) + big(song[:tracks].size,2) + big(song[:div],2)
319 song[:tracks].each do |track|
320 data, last, status = [],0,nil
321 track.each do |e|
322 data.concat(vlq(e[:t]-last))
323 last = e[:t]
324 s, d = e[:s], e[:d].dup
325 if noteoff == 'ZERO' && s>>4 == 8
326 s += 16
327 d[1] = 0
328 end
329 if s < 240
330 data << s unless running && status == s
331 data.concat(d)
332 status = s
333 else
334 status = nil
335 data << s
336 data << e[:m] if s == 255
337 data.concat(vlq(d.size)).concat(d)
338 end
339 end
340 bytes.concat('MTrk'.bytes).concat(big(data.size,4)).concat(data)
341 end
342 bytes.each_slice(16).with_index { |part,i| puts format('%04X: ',i*16) + part.map { |b| format('%02X',b) }.join(' ') }
343 puts "#{bytes.size} bytes"
344end

346songs, running, noteoff = {},true,'KEEP'
347known = %w[HEX TEXT DECODE ENCODE TIMING NOTES STATS ROLL TRANSPOSE QUANTIZE VELOCITY TEMPO FILTER MERGE SPLIT COPY LIST RUNNING NOTEOFF]
348STDIN.gets.to_i.times do
349 cmd, *args = STDIN.gets.chomp.split
350 begin
351 raise 'unknown command' unless known.include?(cmd)
352 name = args[0]
353 if %w[HEX TEXT].include?(cmd)
354 lines = Array.new(args[1].to_i) { STDIN.gets.chomp }
355 song = cmd == 'HEX' ? Reader.new(lines.join(' ').split.map { |s| s.to_i(16) }).parse : parse_text(lines)
356 songs[name] = song
357 puts "loaded #{name} #{summary(song)}"
358 next
359 end
360 unless %w[LIST RUNNING NOTEOFF].include?(cmd)
361 raise "no song #{name}" unless songs.key?(name)
362 end
363 song = songs[name]
364 case cmd
365 when 'LIST'
366 puts 'no songs' if songs.empty?
367 songs.keys.sort.each { |n| puts "#{n} #{summary(songs[n])}" }
368 when 'RUNNING','NOTEOFF'
369 raise 'bad argument' unless (cmd == 'RUNNING' ? %w[ON OFF] : %w[ZERO KEEP]).include?(name)
370 if cmd == 'RUNNING'
371 running = name == 'ON'
372 else
373 noteoff = name
374 end
375 puts "#{cmd.downcase} #{name.downcase}"
376 when 'COPY'
377 songs[args[1]] = Marshal.load(Marshal.dump(song))
378 puts "copied #{name} to #{args[1]}"
379 when 'DECODE'
380 puts "HEADER #{song[:f]} #{song[:tracks].size} #{song[:div]}"
381 song[:tracks].each_with_index { |tr,i| tr.each { |e| puts event_line(e,i) } }
382 when 'ENCODE' then encode(song,running,noteoff)
383 when 'TIMING'
384 raise 'format 2 not supported' if song[:f] == 2
385 ordered(song).each { |e,tr,_| puts "#{e[:t]} #{tr} #{event_type(e)} #{time_position(song,e[:t]).join(' ')}" }
386 puts "end #{length(song)} #{time_position(song,length(song)).join(' ')}"
387 when 'NOTES','STATS','ROLL'
388 if cmd == 'ROLL'
389 from, to, step = args.drop(1).map(&:to_i)
390 raise 'bad argument' unless from.between?(0,10_000_000) && to.between?(1,10_000_000) && step.between?(1,10_000_000) && to > from && (to-from+step-1)/step <= 64
391 elsif cmd == 'STATS' && song[:f] == 2
392 raise 'format 2 not supported'
393 end
394 notes, unmatched = pairing(song)
395 if cmd == 'NOTES'
396 puts "notes #{notes.size} unmatched #{unmatched.size}"
397 notes.each { |n| puts "#{n[:start]} #{n[:length]} #{n[:ch]} #{NOTES[n[:p]]} #{n[:v]}" }
398 unmatched.each { |e,tr,_,on| puts "unmatched #{on ? 'on' : 'off'} #{e[:t]} #{tr} #{(e[:s]&15)+1} #{NOTES[e[:d][0]]}" }
399 elsif cmd == 'ROLL'
400 lines = []
401 127.downto(0) do |p|
402 chars = (0...((to-from+step-1)/step)).map do |i|
403 a = from+i*step
404 group = notes.select { |n| n[:p] == p }
405 if group.any? { |n| a <= n[:start] && n[:start] < a+step } then '#'
406 elsif group.any? { |n| n[:start] <= a && a < n[:start]+n[:length] } then '='
407 else '.'
408 end
409 end.join
410 lines << NOTES[p].ljust(4)+' '+chars if chars.match?(/[#=]/)
411 end
412 puts(lines.empty? ? ['empty'] : lines)
413 else
414 puts summary(song)
415 puts "notes #{notes.size} unmatched #{unmatched.size}"
416 notes.group_by { |n| n[:ch] }.sort.each do |ch, group|
417 lo, hi = group.map { |n| n[:p] }.minmax
418 puts "channel #{ch} notes #{group.size} range #{NOTES[lo]} #{NOTES[hi]}"
419 end
420 lo, hi = notes.map { |n| n[:p] }.minmax
421 puts(notes.empty? ? 'range none' : "range #{NOTES[lo]} #{NOTES[hi]}")
422 puts "length #{length(song)} ticks #{time_position(song,length(song))[0]} ms"
423 changes = Hash.new(0)
424 notes.each do |n|
425 next if n[:length] == 0
426 changes[n[:start]] += 1
427 changes[n[:start]+n[:length]] -= 1
428 end
429 active, maximum, at = 0,0,0
430 changes.sort.each { |t,delta| active += delta; maximum, at = active,t if active > maximum }
431 puts(maximum == 0 ? 'polyphony 0' : "polyphony #{maximum} at #{at}")
432 end
433 when 'TRANSPOSE'
434 amount = args[1].to_i
435 channel = args[2]&.to_i
436 raise 'bad argument' unless amount.between?(-127,127) && (!channel || channel.between?(1,16))
437 kept, dropped = 0,[]
438 song[:tracks].each_with_index do |tr,i|
439 tr.reject! do |e|
440 target = [8,9,10].include?(e[:s]>>4) && (!channel || (e[:s]&15)+1 == channel)
441 next false unless target
442 value = e[:d][0]+amount
443 if value.between?(0,127)
444 e[:d][0] = value
445 kept += 1
446 false
447 else
448 dropped << "dropped #{e[:t]} #{i} #{event_type(e)} #{(e[:s]&15)+1} #{NOTES[e[:d][0]]}"
449 true
450 end
451 end
452 end
453 puts "transposed #{kept} dropped #{dropped.size}"
454 puts dropped
455 when 'QUANTIZE'
456 grid, strength = args.drop(1).map(&:to_i)
457 raise 'bad argument' unless grid.between?(1,1_000_000) && strength.between?(0,100)
458 notes, unmatched = pairing(song)
459 targets = notes.map { |n| [n[:on],n[:off]] } + unmatched.select { |x| x[3] }.map { |x| [x[0]] }
460 moved = 0
461 targets.each do |events|
462 t = events[0][:t]
463 lower = t/grid*grid
464 g = t-lower <= lower+grid-t ? lower : lower+grid
465 delta = (2*(g-t)*strength+100).div(200)
466 moved += 1 unless delta == 0
467 events.each { |e| e[:t] += delta }
468 end
469 song[:tracks].map! do |tr|
470 ending = tr[-1]
471 sorted = tr[0...-1].each_with_index.sort_by { |e,i| [e[:t],i] }.map(&:first)
472 ending[:t] = [ending[:t],sorted.map { |e| e[:t] }.max || 0].max
473 sorted + [ending]
474 end
475 puts "quantized #{targets.size} moved #{moved}"
476 when 'VELOCITY'
477 percent, offset = args.drop(1).map(&:to_i)
478 raise 'bad argument' unless percent.between?(0,1000) && offset.between?(-127,127)
479 count, clamped = 0,0
480 ordered(song).each do |e,_,_|
481 next unless e[:s]>>4 == 9 && e[:d][1] > 0
482 count += 1
483 v = (e[:d][1]*percent+50)/100+offset
484 clamped += 1 unless v.between?(1,127)
485 e[:d][1] = v.clamp(1,127)
486 end
487 puts "velocity #{count} clamped #{clamped}"
488 when 'TEMPO'
489 num, den = args.drop(1).map(&:to_i)
490 raise 'bad argument' unless num.between?(1,1000) && den.between?(1,1000)
491 events = ordered(song).map(&:first).select { |e| event_type(e) == 'TEMPO' }
492 values = events.map { |e| (2*integer(e[:d])*den+num)/(2*num) }
493 insert = !events.any? { |e| e[:t] == 0 }
494 new_value = (1_000_000*den+num)/(2*num)
495 raise 'tempo out of range' unless values.all? { |v| v.between?(1,16777215) } && (!insert || new_value.between?(1,16777215))
496 events.zip(values).each { |e,v| e[:d] = big(v,3) }
497 puts "scaled #{events.size}"
498 if insert
499 song[:tracks][0].unshift({t:0,s:255,m:81,d:big(new_value,3)})
500 puts "inserted 0 0 TEMPO #{new_value}"
501 end
502 when 'FILTER'
503 channels = args.drop(1).map(&:to_i)
504 raise 'bad argument' unless channels.all? { |c| c.between?(1,16) }
505 count = 0
506 song[:tracks].each { |tr| tr.reject! { |e| drop = e[:s] < 240 && !channels.include?((e[:s]&15)+1); count += 1 if drop; drop } }
507 puts "removed #{count}"
508 when 'MERGE'
509 raise 'not format 1' unless song[:f] == 1
510 count = song[:tracks].size
511 ending = {t:length(song),s:255,m:47,d:[]}
512 song[:tracks] = [ordered(song).map(&:first).reject { |e| event_type(e) == 'END_OF_TRACK' } + [ending]]
513 song[:f] = 0
514 puts "merged #{count} tracks events #{song[:tracks][0].size}"
515 when 'SPLIT'
516 raise 'not format 0' unless song[:f] == 0
517 track = song[:tracks][0]
