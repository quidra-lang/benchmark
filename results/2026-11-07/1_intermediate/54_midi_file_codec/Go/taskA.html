75func be(data []byte) int64 {
76 value := int64(0)
77 for _, b := range data {
78 value = value*256 + int64(b)
79 }
80 return value
81}

83func bytesBE(value int64, size int) []byte {
84 data := make([]byte, size)
85 for i := size - 1; i >= 0; i-- {
86 data[i] = byte(value % 256)
87 value /= 256
88 }
89 return data
90}

96func (e *Event) kind() string {
97 if e.Status < 240 {
98 return channelNames[e.Status/16-8]
99 }
100 if e.Status == 240 {
101 return "SYSEX"
102 }
103 if e.Status == 247 {
104 return "SYSEX_ESCAPE"
105 }
106 if name, ok := metaNames[e.Meta]; ok {
107 return name
108 }
109 return "META"
110}

129 c := &Song{Format: s.Format, Division: s.Division}

174func eventLine(e *Event, track int) string {
175 fields := []string{decimal(e.Tick), strconv.Itoa(track), e.kind()}
176 data := e.Data
177 if e.Status < 240 {
178 fields = append(fields, strconv.Itoa(e.channel()))
179 switch e.Status / 16 {
180 case 8, 9, 10:
181 fields = append(fields, noteName(int(data[0])), strconv.Itoa(int(data[1])))
182 case 14:
183 fields = append(fields, strconv.Itoa(int(data[0])+128*int(data[1])-8192))
184 default:
185 for _, b := range data {
186 fields = append(fields, strconv.Itoa(int(b)))
187 }
188 }
189 } else if e.Status == 255 && e.Meta >= 1 && e.Meta <= 7 {
190 fields = append(fields, "\""+string(data)+"\"")
191 } else if e.isMeta(0) || e.isMeta(81) {
192 fields = append(fields, decimal(be(data)))
193 } else if e.isMeta(32) {
194 fields = append(fields, strconv.Itoa(int(data[0])+1))
195 } else if e.isMeta(47) {
196 } else if e.isMeta(84) {
197 for _, b := range data {
198 fields = append(fields, strconv.Itoa(int(b)))
199 }
200 } else if e.isMeta(88) {
201 fields = append(fields, fmt.Sprintf("%d/%d", data[0], 1<<data[1]), strconv.Itoa(int(data[2])), strconv.Itoa(int(data[3])))
202 } else if e.isMeta(89) {
203 sf := int(int8(data[0]))
204 mode := "major"
205 if data[1] == 1 {
206 mode = "minor"
207 }
208 fields = append(fields, keyNames[data[1]][sf+7], mode)
209 } else {
210 if e.Status == 255 && e.Meta != 127 {
211 fields = append(fields, fmt.Sprintf("%02X", e.Meta))
212 }
213 fields = append(fields, strconv.Itoa(len(data)))
214 if len(data) > 0 {
215 fields = append(fields, strings.ToUpper(hex.EncodeToString(data)))
216 }
217 }
218 return strings.Join(fields, " ")
219}

221func parseHex(data []byte) (song *Song, failure *Failure) {
222 defer func() {
223 if caught := recover(); caught != nil {
224 value := caught.(Failure)
225 song = nil
226 failure = &value
227 }
228 }()
229 fail := func(pos int, reason string) {
230 panic(Failure{pos, reason})
231 }
232 chunk := func(pos int, id string) (int, int) {
233 if len(data)-pos < 8 {
234 fail(pos, "truncated chunk")
235 }
236 if string(data[pos:pos+4]) != id {
237 fail(pos, "bad chunk id")
238 }
239 length := be(data[pos+4 : pos+8])
240 if int64(pos)+8+length > int64(len(data)) {
241 fail(pos, "truncated chunk")
242 }
243 return pos + 8, pos + 8 + int(length)
244 }
245 start, finish := chunk(0, "MThd")
246 if finish-start != 6 {
247 fail(4, "bad header length")
248 }
249 format, tracks, division := int(be(data[8:10])), int(be(data[10:12])), be(data[12:14])
250 if format > 2 {
251 fail(8, "bad format")
252 }
253 if tracks == 0 || format == 0 && tracks != 1 {
254 fail(10, "bad track count")
255 }
256 if division >= 32768 {
257 fail(12, "smpte timing not supported")
258 }
259 if division == 0 {
260 fail(12, "bad division")
261 }
262 song = &Song{Format: format, Division: division}
263 position := finish
264 for t := 0; t < tracks; t++ {
265 if position == len(data) {
266 fail(len(data), "missing track")
267 }
268 start, finish = chunk(position, "MTrk")
269 position = start
270 tick := int64(0)
271 running := 0
272 track := []*Event{}
273 for position < finish {
274 if len(track) > 0 && track[len(track)-1].isMeta(47) {
275 fail(position, "event after end of track")
276 }
277 eventStart := position
278 read := func() byte {
279 if position >= finish {
280 fail(eventStart, "truncated event")
281 }
282 b := data[position]
283 position++
284 return b
285 }
286 vlq := func() int64 {
287 at := position
288 value := int64(0)
289 for i := 0; i < 4; i++ {
290 b := read()
291 value = value*128 + int64(b&127)
292 if b < 128 {
293 return value
294 }
295 }
296 fail(at, "variable-length value too long")
297 return 0
298 }
299 tick += vlq()
300 statusPos := position
301 if position >= finish {
302 fail(eventStart, "truncated event")
303 }
304 status := int(data[position])
305 if status < 128 {
306 if running == 0 {
307 fail(position, "data byte without running status")
308 }
309 status = running
310 } else {
311 position++
312 }
313 e := &Event{Tick: tick, Status: status}
314 if status < 240 {
315 running = status
316 count := 2
317 if status/16 == 12 || status/16 == 13 {
318 count = 1
319 }
320 for j := 0; j < count; j++ {
321 at := position
322 b := read()
323 if b >= 128 {
324 fail(at, "bad data byte")
325 }
326 e.Data = append(e.Data, b)
327 }
328 } else if status == 240 || status == 247 || status == 255 {
329 running = 0
330 if status == 255 {
331 e.Meta = int(read())
332 }
333 length := vlq()
334 if int64(position)+length > int64(finish) {
335 fail(eventStart, "truncated event")
336 }
337 e.Data = append([]byte{}, data[position:position+int(length)]...)
338 position += int(length)
339 if status == 255 {
340 sizes := map[int]int{0: 2, 32: 1, 47: 0, 81: 3, 84: 5, 88: 4, 89: 2}
341 if size, ok := sizes[e.Meta]; ok && size != len(e.Data) {
342 fail(statusPos, "bad meta length")
343 }
344 d := e.Data
345 bad := e.Meta == 81 && be(d) == 0 || e.Meta == 32 && d[0] > 15 || e.Meta == 88 && (d[0] == 0 || d[1] > 6) || e.Meta == 89 && (int8(d[0]) < -7 || int8(d[0]) > 7 || d[1] > 1)
346 if bad {
347 fail(statusPos, "bad meta data")
348 }
349 }
350 } else {
351 fail(statusPos, "unknown status byte")
352 }
353 track = append(track, e)
354 }
355 if len(track) == 0 || !track[len(track)-1].isMeta(47) {
356 fail(finish, "missing end of track")
357 }
358 song.Tracks = append(song.Tracks, track)
359 position = finish
360 }
361 if position < len(data) {
362 fail(position, "extra data")
363 }
364 return song, nil
365}

367func parseText(lines []string) (song *Song, failure *Failure) {
368 lineNo := 1
369 defer func() {
370 if caught := recover(); caught != nil {
371 value := Failure{lineNo, caught.(string)}
372 song = nil
373 failure = &value
374 }
375 }()
376 require := func(ok bool) {
377 if !ok {
378 panic("bad value")
379 }
380 }
381 header := strings.Fields(lines[0])
382 if len(header) != 4 || header[0] != "HEADER" {
383 panic("bad header")
384 }
385 f, t, d := integer(header[1]), integer(header[2]), integer(header[3])
386 if f < 0 || f > 2 || t < 1 || t > 65535 || d < 1 || d > 32767 || f == 0 && t != 1 {
387 panic("bad header")
388 }
389 song = &Song{Format: int(f), Division: d, Tracks: make([][]*Event, int(t))}
390 current := 0
391 code := map[string]int{}
392 for n, name := range metaNames {
393 code[name] = n
394 }
395 for i, line := range lines[1:] {
396 lineNo = i + 2
397 words := strings.SplitN(line, " ", 4)
398 tick, track, kind := integer(words[0]), int(integer(words[1])), words[2]
399 fields := []string{}
400 if len(words) == 4 {
401 fields = strings.Split(words[3], " ")
402 }
403 e := &Event{Tick: tick, Status: 255}
404 channel := -1
405 for j, name := range channelNames {
406 if name == kind {
407 channel = j
408 }
409 }
410 meta, knownMeta := code[kind]
411 if channel < 0 && !knownMeta && kind != "META" && kind != "SYSEX" && kind != "SYSEX_ESCAPE" {
412 panic("unknown event type")
413 }
414 value := func(index int, low, high int64) int64 {
415 require(index < len(fields))
416 n := integer(fields[index])
417 require(n >= low && n <= high)
418 return n
419 }
420 if channel >= 0 {
421 count := 3
422 if channel == 4 || channel == 5 || channel == 6 {
423 count = 2
424 }
425 require(len(fields) == count)
426 ch := value(0, 1, 16)
427 e.Status = (channel+8)*16 + int(ch) - 1
428 if channel <= 2 {
429 pitch := -1
430 for p := 0; p < 128; p++ {
431 if fields[1] == noteName(p) {
432 pitch = p
433 break
434 }
435 }
436 require(pitch >= 0)
437 e.Data = []byte{byte(pitch), byte(value(2, 0, 127))}
438 } else if channel == 6 {
439 bend := value(1, -8192, 8191) + 8192
440 e.Data = []byte{byte(bend % 128), byte(bend / 128)}
441 } else {
442 for j := 1; j < count; j++ {
443 e.Data = append(e.Data, byte(value(j, 0, 127)))
444 }
445 }
446 } else if knownMeta && meta >= 1 && meta <= 7 {
447 require(len(words) == 4)
448 text := words[3]
449 require(len(text) >= 2 && text[0] == '"' && text[len(text)-1] == '"')
450 require(!strings.Contains(text[1:len(text)-1], "\""))
451 e.Meta = meta
452 e.Data = []byte(text[1 : len(text)-1])
453 } else if kind == "SYSEX" || kind == "SYSEX_ESCAPE" || kind == "META" || kind == "SEQUENCER_SPECIFIC" {
454 if kind == "SYSEX" {
455 e.Status = 240
456 } else if kind == "SYSEX_ESCAPE" {
457 e.Status = 247
458 } else if kind == "META" {
459 require(len(fields) >= 2 && len(fields[0]) == 2)
460 require(strings.Trim(fields[0], "0123456789ABCDEF") == "")
461 n, _ := strconv.ParseInt(fields[0], 16, 64)
462 e.Meta = int(n)
463 _, reserved := metaNames[e.Meta]
464 require(!reserved)
465 fields = fields[1:]
466 } else {
467 e.Meta = 127
468 }
469 length := value(0, 0, 1000000)
470 if length == 0 {
471 require(len(fields) == 1)
472 } else {
473 require(len(fields) == 2 && int64(len(fields[1])) == length*2)
474 require(strings.Trim(fields[1], "0123456789ABCDEF") == "")
475 e.Data, _ = hex.DecodeString(fields[1])
476 }
477 } else {
478 e.Meta = meta
479 switch meta {
480 case 0:
481 require(len(fields) == 1)
482 e.Data = bytesBE(value(0, 0, 65535), 2)
483 case 32:
484 require(len(fields) == 1)
485 e.Data = []byte{byte(value(0, 1, 16) - 1)}
486 case 47:
487 require(len(fields) == 0)
488 case 81:
489 require(len(fields) == 1)
490 e.Data = bytesBE(value(0, 1, 16777215), 3)
491 case 84:
492 require(len(fields) == 5)
493 for j := 0; j < 5; j++ {
494 e.Data = append(e.Data, byte(value(j, 0, 255)))
495 }
496 case 88:
497 require(len(fields) == 3)
498 fraction := strings.Split(fields[0], "/")
499 require(len(fraction) == 2)
500 nn, dd := integer(fraction[0]), integer(fraction[1])
501 exponent := -1
502 for j := 0; j <= 6; j++ {
503 if dd == 1<<j {
504 exponent = j
505 }
506 }
507 require(nn >= 1 && nn <= 255 && exponent >= 0)
508 e.Data = []byte{byte(nn), byte(exponent), byte(value(1, 0, 255)), byte(value(2, 0, 255))}
509 case 89:
510 require(len(fields) == 2)
511 found := false
512 for mi, mode := range []string{"major", "minor"} {
513 for j, key := range keyNames[mi] {
514 if fields[0] == key && fields[1] == mode {
515 e.Data = []byte{byte(j - 7), byte(mi)}
516 found = true
517 }
518 }
519 }
520 require(found)
521 }
522 }
523 if track >= len(song.Tracks) {
524 panic("bad track")
525 }
526 row := song.Tracks[current]
527 ended := len(row) > 0 && row[len(row)-1].isMeta(47)
528 if track < current {
529 panic("out of order")
530 }
531 if track > current && (!ended || track > current+1) {
532 panic("missing end of track")
533 }
534 if track == current && ended {
535 panic("event after end of track")
536 }
537 if track == current+1 {
538 current++
539 row = song.Tracks[current]
540 }
541 if len(row) > 0 && tick < row[len(row)-1].Tick {
542 panic("out of order")
543 }
544 song.Tracks[current] = append(row, e)
545 }
546 lineNo = len(lines) + 1
547 row := song.Tracks[current]
548 if current+1 < len(song.Tracks) || len(row) == 0 || !row[len(row)-1].isMeta(47) {
549 panic("missing end of track")
550 }
551 return song, nil
552}

554func vlq(value int64) []byte {
555 data := []byte{byte(value % 128)}
556 for value /= 128; value > 0; value /= 128 {
557 data = append([]byte{byte(value%128) | 128}, data...)
558 }
559 return data
560}

562func (s *Song) encode(running, zero bool) []byte {
563 result := []byte("MThd")
564 result = append(result, bytesBE(6, 4)...)
565 result = append(result, bytesBE(int64(s.Format), 2)...)
566 result = append(result, bytesBE(int64(len(s.Tracks)), 2)...)
567 result = append(result, bytesBE(s.Division, 2)...)
568 for _, track := range s.Tracks {
569 data := []byte{}
570 last := int64(0)
571 status := 0
572 for _, e := range track {
573 data = append(data, vlq(e.Tick-last)...)
574 last = e.Tick
575 st := e.Status
576 body := e.Data
577 if zero && st/16 == 8 {
578 st += 16
579 body = []byte{e.Data[0], 0}
580 }
581 if st < 240 {
582 if !running || status != st {
583 data = append(data, byte(st))
584 }
585 data = append(data, body...)
586 status = st
587 } else {
588 status = 0
589 data = append(data, byte(st))
590 if st == 255 {
591 data = append(data, byte(e.Meta))
592 }
593 data = append(data, vlq(int64(len(body)))...)
594 data = append(data, body...)
595 }
596 }
597 result = append(result, []byte("MTrk")...)
598 result = append(result, bytesBE(int64(len(data)), 4)...)
599 result = append(result, data...)
600 }
601 return result
602}

604func (s *Song) pairing() ([]Note, []Placed) {
605 queues := map[[2]int][]Placed{}
606 notes := []Note{}
607 unmatched := []Placed{}
608 for _, p := range s.ordered() {
609 e := p.Event
610 if !e.on() && !e.off() {
611 continue
612 }
613 key := [2]int{e.channel(), int(e.Data[0])}
614 if e.on() {
615 queues[key] = append(queues[key], p)
616 } else if len(queues[key]) == 0 {
617 unmatched = append(unmatched, p)
618 } else {
619 on := queues[key][0]
620 queues[key] = queues[key][1:]
621 notes = append(notes, Note{on, p, e.Tick - on.Event.Tick})
622 }
623 }
624 for _, queue := range queues {
625 unmatched = append(unmatched, queue...)
626 }
627 sort.Slice(notes, func(i, j int) bool {
628 a, b := notes[i].On, notes[j].On
629 if a.Event.Tick != b.Event.Tick {
630 return a.Event.Tick < b.Event.Tick
631 }
632 if a.Event.channel() != b.Event.channel() {
633 return a.Event.channel() < b.Event.channel()
634 }
635 if a.Event.Data[0] != b.Event.Data[0] {
636 return a.Event.Data[0] < b.Event.Data[0]
637 }
638 return a.Order < b.Order
639 })
640 sort.Slice(unmatched, func(i, j int) bool {
641 return unmatched[i].Order < unmatched[j].Order
642 })
643 return notes, unmatched
644}

646func (s *Song) timeAt(tick int64) string {
647 total, last, tempo := int64(0), int64(0), int64(500000)
648 for _, p := range s.ordered() {
649 e := p.Event
650 if e.Tick > tick {
651 break
652 }
653 if e.isMeta(81) {
654 total += (e.Tick - last) * tempo
655 last = e.Tick
656 tempo = be(e.Data)
657 }
658 }
659 usec := (total + (tick-last)*tempo) / s.Division
660 return fmt.Sprintf("%d.%03d", usec/1000, usec%1000)
661}

663func (s *Song) barAt(tick int64) string {
664 start, bar, nn, beat := int64(0), int64(1), int64(4), s.Division
665 for _, p := range s.ordered() {
666 e := p.Event
667 if e.Tick > tick {
668 break
669 }
670 if e.isMeta(88) {
671 size := nn * beat
672 bar += (e.Tick - start + size - 1) / size
673 start = e.Tick
674 nn = int64(e.Data[0])
675 beat = s.Division * 4 / (1 << e.Data[1])
676 }
677 }
678 x := tick - start
679 return fmt.Sprintf("%d:%d:%d", bar+x/(nn*beat), x%(nn*beat)/beat+1, x%beat)
680}

682func floorDiv(x, y int64) int64 {
683 q := x / y
684 if x%y < 0 {
685 q--
686 }
687 return q

690func execute(words []string, songs map[string]*Song, running, zero *bool) {
691 cmd := words[0]
692 a := words[1:]
693 commands := " DECODE ENCODE TIMING NOTES STATS ROLL TRANSPOSE QUANTIZE VELOCITY TEMPO FILTER MERGE SPLIT COPY LIST RUNNING NOTEOFF "
694 if !strings.Contains(commands, " "+cmd+" ") {
695 output("error: unknown command")
696 return
697 }
698 if cmd == "LIST" {
699 names := []string{}
700 for name := range songs {
701 names = append(names, name)
702 }
703 sort.Strings(names)
704 for _, name := range names {
705 output(name + " " + songs[name].summary())
706 }
707 if len(names) == 0 {
708 output("no songs")
709 }
710 return
711 }
712 if cmd == "RUNNING" || cmd == "NOTEOFF" {
713 valid := a[0] == "ON" || a[0] == "OFF"
714 if cmd == "NOTEOFF" {
715 valid = a[0] == "ZERO" || a[0] == "KEEP"
716 }
717 if !valid {
718 output("error: bad argument")
719 return
720 }
721 if cmd == "RUNNING" {
722 *running = a[0] == "ON"
723 } else {
724 *zero = a[0] == "ZERO"
725 }
726 output(strings.ToLower(cmd) + " " + strings.ToLower(a[0]))
727 return
728 }
729 song, exists := songs[a[0]]
730 if !exists {
731 output("error: no song " + a[0])
732 return
733 }
734 args := []int64{}
735 if cmd != "COPY" {
736 for _, word := range a[1:] {
737 args = append(args, integer(word))
738 }
739 }
740 valid := true
741 between := func(x, lo, hi int64) bool {
742 return x >= lo && x <= hi
743 }
744 switch cmd {
745 case "TRANSPOSE":
746 valid = between(args[0], -127, 127)
747 if len(args) > 1 {
748 valid = valid && between(args[1], 1, 16)
749 }
750 case "QUANTIZE":
751 valid = between(args[0], 1, 1000000) && between(args[1], 0, 100)
752 case "VELOCITY":
753 valid = between(args[0], 0, 1000) && between(args[1], -127, 127)
754 case "TEMPO":
755 valid = between(args[0], 1, 1000) && between(args[1], 1, 1000)
756 case "FILTER":
757 for _, x := range args {
758 valid = valid && between(x, 1, 16)
759 }
760 case "ROLL":
761 valid = between(args[0], 0, 10000000) && between(args[1], 1, 10000000) && between(args[2], 1, 10000000) && args[1] > args[0]
762 if valid {
763 valid = (args[1]-args[0]+args[2]-1)/args[2] <= 64
764 }
765 }
766 if !valid {
767 output("error: bad argument")
768 return
769 }
770 if (cmd == "TIMING" || cmd == "STATS") && song.Format == 2 {
771 output("error: format 2 not supported")
772 return
773 }
774 if cmd == "MERGE" && song.Format != 1 {
775 output("error: not format 1")
776 return
777 }
778 if cmd == "SPLIT" && song.Format != 0 {
779 output("error: not format 0")
780 return
781 }
782 switch cmd {
783 case "COPY":
784 songs[a[1]] = song.clone()
785 output("copied " + a[0] + " to " + a[1])
786 case "DECODE":
787 output(fmt.Sprintf("HEADER %d %d %d", song.Format, len(song.Tracks), song.Division))
788 for t, track := range song.Tracks {
789 for _, e := range track {
790 output(eventLine(e, t))
791 }
792 }
793 case "ENCODE":
794 data := song.encode(*running, *zero)
795 for i := 0; i < len(data); i += 16 {
796 values := []string{}
797 for _, b := range data[i:min(i+16, len(data))] {
798 values = append(values, fmt.Sprintf("%02X", b))
799 }
800 output(fmt.Sprintf("%04X: %s", i, strings.Join(values, " ")))
801 }
802 output(fmt.Sprintf("%d bytes", len(data)))
803 case "TIMING":
804 for _, p := range song.ordered() {
805 e := p.Event
806 output(fmt.Sprintf("%d %d %s %s %s", e.Tick, p.Track, e.kind(), song.timeAt(e.Tick), song.barAt(e.Tick)))
807 }
808 t := song.length()
809 output(fmt.Sprintf("end %d %s %s", t, song.timeAt(t), song.barAt(t)))
810 case "NOTES", "STATS", "ROLL":
811 notes, unmatched := song.pairing()
812 if cmd == "ROLL" {
813 columns := int((args[1] - args[0] + args[2] - 1) / args[2])
814 any := false
815 for pitch := 127; pitch >= 0; pitch-- {
816 row := make([]byte, columns)
817 for i := range row {
818 tick := args[0] + int64(i)*args[2]
819 row[i] = '.'
820 for _, n := range notes {
821 if int(n.On.Event.Data[0]) != pitch {
822 continue
823 }
824 start := n.On.Event.Tick
825 if tick <= start && start < tick+args[2] {
826 row[i] = '#'
827 break
828 }
829 if start <= tick && tick < start+n.Length {
830 row[i] = '='
831 }
832 }
833 }
834 if strings.Trim(string(row), ".") != "" {
835 output(fmt.Sprintf("%-4s %s", noteName(pitch), row))
836 any = true
837 }
838 }
839 if !any {
840 output("empty")
841 }
842 break
843 }
844 if cmd == "STATS" {
845 output(song.summary())
846 }
847 output(fmt.Sprintf("notes %d unmatched %d", len(notes), len(unmatched)))
848 if cmd == "NOTES" {
849 for _, n := range notes {
850 e := n.On.Event
851 output(fmt.Sprintf("%d %d %d %s %d", e.Tick, n.Length, e.channel(), noteName(int(e.Data[0])), e.Data[1]))
852 }
853 for _, p := range unmatched {
854 e := p.Event
855 kind := "off"
856 if e.on() {
857 kind = "on"
858 }
859 output(fmt.Sprintf("unmatched %s %d %d %d %s", kind, e.Tick, p.Track, e.channel(), noteName(int(e.Data[0]))))
860 }
861 } else {
862 low, high := 128, -1
863 changes := map[int64]int{}
864 for channel := 1; channel <= 16; channel++ {
865 count, lo, hi := 0, 128, -1
866 for _, n := range notes {
867 e := n.On.Event
868 if e.channel() == channel {
869 count++
870 lo = min(lo, int(e.Data[0]))
871 hi = max(hi, int(e.Data[0]))
872 }
873 }
874 if count > 0 {
875 output(fmt.Sprintf("channel %d notes %d range %s %s", channel, count, noteName(lo), noteName(hi)))
876 low = min(low, lo)
877 high = max(high, hi)
878 }
879 }
880 if high < 0 {
881 output("range none")
882 } else {
883 output("range " + noteName(low) + " " + noteName(high))
884 }
885 output(fmt.Sprintf("length %d ticks %s ms", song.length(), song.timeAt(song.length())))
886 for _, n := range notes {
887 if n.Length > 0 {
888 changes[n.On.Event.Tick]++
889 changes[n.On.Event.Tick+n.Length]--
890 }
891 }
892 ticks := []int64{}
893 for tick := range changes {
894 ticks = append(ticks, tick)
895 }
896 sort.Slice(ticks, func(i, j int) bool {
897 return ticks[i] < ticks[j]
898 })
899 current, peak := 0, 0
900 at := int64(0)
901 for _, tick := range ticks {
902 current += changes[tick]
903 if current > peak {
904 peak = current
905 at = tick
906 }
907 }
908 if peak == 0 {
909 output("polyphony 0")
910 } else {
911 output(fmt.Sprintf("polyphony %d at %d", peak, at))
912 }
913 }
914 case "TRANSPOSE":
915 kept, dropped := 0, []string{}
916 for t, track := range song.Tracks {
917 row := []*Event{}
918 for _, e := range track {
919 target := e.Status < 176 && (len(args) == 1 || int64(e.channel()) == args[1])
920 if target {
921 pitch := int64(e.Data[0]) + args[0]
922 if pitch < 0 || pitch > 127 {
923 dropped = append(dropped, fmt.Sprintf("dropped %d %d %s %d %s", e.Tick, t, e.kind(), e.channel(), noteName(int(e.Data[0]))))
924 continue
925 }
926 e.Data[0] = byte(pitch)
927 kept++
928 }
929 row = append(row, e)
930 }
931 song.Tracks[t] = row
932 }
933 output(fmt.Sprintf("transposed %d dropped %d", kept, len(dropped)))
934 for _, line := range dropped {
935 output(line)
936 }
937 case "QUANTIZE":
938 notes, unmatched := song.pairing()
939 moved, targets := 0, 0
940 shift := func(on, off *Event) {
941 targets++
942 grid, strength := args[0], args[1]
943 lower := on.Tick / grid * grid
944 target := lower
945 if on.Tick-lower > lower+grid-on.Tick {
946 target += grid
947 }
948 delta := floorDiv(2*(target-on.Tick)*strength+100, 200)
949 if delta != 0 {
950 moved++
951 }
952 on.Tick += delta
953 if off != nil {
954 off.Tick += delta
955 }
956 }
957 for _, n := range notes {
958 shift(n.On.Event, n.Off.Event)
959 }
960 for _, p := range unmatched {
961 if p.Event.on() {
962 shift(p.Event, nil)
963 }
964 }
965 for t, track := range song.Tracks {
966 ending := track[len(track)-1]
967 row := track[:len(track)-1]
968 sort.SliceStable(row, func(i, j int) bool {
969 return row[i].Tick < row[j].Tick
970 })
971 if len(row) > 0 {
972 ending.Tick = max(ending.Tick, row[len(row)-1].Tick)
973 }
974 song.Tracks[t] = append(row, ending)
975 }
976 output(fmt.Sprintf("quantized %d moved %d", targets, moved))
977 case "VELOCITY":
978 count, clamped := 0, 0
979 for _, track := range song.Tracks {
980 for _, e := range track {
981 if e.on() {
982 count++
983 v := (int64(e.Data[1])*args[0]+50)/100 + args[1]
984 if v < 1 || v > 127 {
985 clamped++
986 }
987 e.Data[1] = byte(max(1, min(127, v)))
988 }
989 }
990 }
991 output(fmt.Sprintf("velocity %d clamped %d", count, clamped))
992 case "TEMPO":
993 trial := song.clone()
994 count := 0
995 exists := false
996 good := true
997 scale := func(v int64) int64 {
998 return (2*v*args[1] + args[0]) / (2 * args[0])
999 }
1000 for _, track := range trial.Tracks {
1001 for _, e := range track {
1002 if e.isMeta(81) {
1003 count++
1004 exists = exists || e.Tick == 0
1005 v := scale(be(e.Data))
1006 if v < 1 || v > 16777215 {
1007 good = false
1008 }
1009 e.Data = bytesBE(v, 3)
1010 }
1011 }
1012 }
1013 inserted := scale(500000)
1014 if !exists {
1015 if inserted < 1 || inserted > 16777215 {
1016 good = false
1017 }
1018 trial.Tracks[0] = append([]*Event{{0, 255, 81, bytesBE(inserted, 3)}}, trial.Tracks[0]...)
1019 }
1020 if !good {
1021 output("error: tempo out of range")
1022 return
1023 }
1024 songs[a[0]] = trial
1025 output(fmt.Sprintf("scaled %d", count))
1026 if !exists {
1027 output(fmt.Sprintf("inserted 0 0 TEMPO %d", inserted))
1028 }
1029 case "FILTER":
1030 allowed := map[int]bool{}
1031 for _, ch := range args {
1032 allowed[int(ch)] = true
1033 }
1034 count := 0
1035 for t, track := range song.Tracks {
1036 row := []*Event{}
1037 for _, e := range track {
1038 if e.Status < 240 && !allowed[e.channel()] {
1039 count++
1040 } else {
1041 row = append(row, e)
1042 }
1043 }
1044 song.Tracks[t] = row
1045 }
1046 output(fmt.Sprintf("removed %d", count))
1047 case "MERGE":
1048 original := len(song.Tracks)
1049 length := song.length()
1050 row := []*Event{}
1051 for _, p := range song.ordered() {
1052 if !p.Event.isMeta(47) {
1053 row = append(row, p.Event)
1054 }
1055 }
1056 row = append(row, end(length))
1057 song.Format = 0
1058 song.Tracks = [][]*Event{row}
1059 output(fmt.Sprintf("merged %d tracks events %d", original, len(row)))
1060 case "SPLIT":
1061 length := song.length()
1062 rows := map[int][]*Event{0: {}}
1063 for _, e := range song.Tracks[0] {
1064 if e.isMeta(47) {
1065 continue
1066 }
1067 channel := 0
1068 if e.Status < 240 {
1069 channel = e.channel()
1070 }
1071 rows[channel] = append(rows[channel], e)
1072 }
1073 channels := []int{}
1074 for ch := range rows {
1075 channels = append(channels, ch)
1076 }
1077 sort.Ints(channels)
1078 song.Tracks = nil
1079 for _, ch := range channels {
1080 song.Tracks = append(song.Tracks, append(rows[ch], end(length)))
1081 }
1082 song.Format = 1
1083 output(fmt.Sprintf("split tracks %d events %d", len(song.Tracks), func() int {
1084 n := 0
1085 for _, row := range song.Tracks {
1086 n += len(row)
1087 }
1088 return n
1089 }()))
1090 }
1091}
