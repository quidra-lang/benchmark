428function timing(s: Song, tick: number): { time: string; bar: string } {
429 let total = 0,
430 last = 0,
431 tempo = 500000,
432 start = 0,
433 bar = 1,
434 nn = 4,

630 `dropped ${e.tick} ${t} ${e.type} ${e.ch} ${noteNames[e.data[0]]}`,
631 );
632 return false;
633 }
634 e.data[0] = n;
635 kept++;
636 return true;
637 }),
638 );
639 emit(`transposed ${kept} dropped ${dropped.length}`);
640 dropped.forEach(emit);
641 break;
642 }
643 case "QUANTIZE": {
644 const p = pairing(s);
645 let moved = 0;
646 const targets: { on: Located; off?: Located }[] = p.notes.slice();
647 for (const l of p.unmatched) if (isOn(l.event)) targets.push({ on: l });
648 for (const n of targets) {
649 const t = n.on.event.tick,
650 lower = Math.floor(t / a[0]) * a[0],
651 g = t - lower <= lower + a[0] - t ? lower : lower + a[0];
652 const delta = Math.floor((2 * (g - t) * a[1] + 100) / 200);
653 n.on.event.tick += delta;
654 if (n.off) n.off.event.tick += delta;
655 if (delta) moved++;
656 }
657 s.tracks = s.tracks.map((t) => {
658 const end = t[t.length - 1],
659 others = t.slice(0, -1).sort((a, b) => a.tick - b.tick);
660 end.tick = others.reduce((n, e) => Math.max(n, e.tick), end.tick);
661 return others.concat(end);
662 });
663 emit(`quantized ${targets.length} moved ${moved}`);
664 break;
665 }
666 case "VELOCITY": {
667 let converted = 0,
668 clamped = 0;
669 for (const l of globalEvents(s))
670 if (isOn(l.event)) {
671 const v = Math.floor((l.event.data[1] * a[0] + 50) / 100) + a[1];
672 converted++;
673 if (v < 1 || v > 127) clamped++;
674 l.event.data[1] = Math.max(1, Math.min(127, v));
675 }
676 emit(`velocity ${converted} clamped ${clamped}`);
677 break;
678 }
679 case "TEMPO": {
680 const tempos = globalEvents(s).filter((l) => l.event.type === "TEMPO"),
681 convert = (n: number): number => Math.floor((2 * n * a[1] + a[0]) / (2 * a[0]));
682 const values = tempos.map((l) => convert(value(l.event))),
683 insert = !tempos.some((l) => l.event.tick === 0),
684 initial = convert(500000);
685 if (
686 values.some((n) => n < 1 || n > 16777215) ||
687 (insert && (initial < 1 || initial > 16777215))
688 )
689 fail("tempo out of range");
690 tempos.forEach((l, i) => {
691 l.event.data = be(values[i], 3);
692 });
693 emit("scaled " + tempos.length);
694 if (insert) {
695 s.tracks[0].unshift(event(0, "TEMPO", be(initial, 3)));
696 emit("inserted 0 0 TEMPO " + initial);
697 }
698 break;
699 }
700 case "FILTER": {
701 const before = count(s);
702 s.tracks = s.tracks.map((t) =>
703 t.filter((e) => !channelTypes.includes(e.type) || a.includes(e.ch)),
704 );
705 emit("removed " + (before - count(s)));
706 break;
707 }
708 case "MERGE": {
709 if (s.format !== 1) fail("not format 1");
710 const n = s.tracks.length,
711 end = length(s),
712 events = globalEvents(s)
713 .filter((l) => l.event.type !== "END_OF_TRACK")
714 .map((l) => l.event);
715 s.tracks = [events.concat(event(end, "END_OF_TRACK"))];
716 s.format = 0;
717 emit(`merged ${n} tracks events ${count(s)}`);
718 break;
719 }
720 case "SPLIT": {
721 if (s.format !== 0) fail("not format 0");
722 const end = length(s),
723 events = s.tracks[0].slice(0, -1),
724 channels = Array.from(
725 new Set(events.filter((e) => channelTypes.includes(e.type)).map((e) => e.ch)),
726 ).sort((a, b) => a - b);
727 s.tracks = [events.filter((e) => !channelTypes.includes(e.type))].concat(
728 channels.map((ch) =>
729 events.filter((e) => channelTypes.includes(e.type) && e.ch === ch),
730 ),
731 );
732 s.tracks.forEach((t) => t.push(event(end, "END_OF_TRACK")));
733 s.format = 1;
734 emit(`split tracks ${s.tracks.length} events ${count(s)}`);
735 break;
736 }
737 case "STATS": {
738 const p = pairing(s);
739 emit(description(s));
740 emit(`notes ${p.notes.length} unmatched ${p.unmatched.length}`);
741 const rangeText = (notes: Note[]): string => {
742 let lo = 127,
743 hi = 0;
744 for (const n of notes) {
745 lo = Math.min(lo, n.on.event.data[0]);
746 hi = Math.max(hi, n.on.event.data[0]);
747 }
748 return `${noteNames[lo]} ${noteNames[hi]}`;
749 };
750 for (let ch = 1; ch <= 16; ch++) {
751 const notes = p.notes.filter((n) => n.on.event.ch === ch);
752 if (notes.length)
753 emit(`channel ${ch} notes ${notes.length} range ${rangeText(notes)}`);
754 }
755 emit("range " + (p.notes.length ? rangeText(p.notes) : "none"));
756 emit(`length ${length(s)} ticks ${timing(s, length(s)).time} ms`);
757 const changes = new Map<number, number>();
758 for (const n of p.notes)
759 if (n.off.event.tick > n.on.event.tick) {
760 changes.set(n.on.event.tick, (changes.get(n.on.event.tick) ?? 0) + 1);
761 changes.set(n.off.event.tick, (changes.get(n.off.event.tick) ?? 0) - 1);
762 }
763 let active = 0,
764 maximum = 0,
765 at = 0;
766 for (const tick of Array.from(changes.keys()).sort((a, b) => a - b)) {
767 active += changes.get(tick)!;
768 if (active > maximum) {
769 maximum = active;
770 at = tick;
771 }
772 }
773 emit(maximum ? `polyphony ${maximum} at ${at}` : "polyphony 0");
774 break;
775 }
776 case "ROLL": {
777 const notes = pairing(s).notes;
778 let rows = 0;
779 for (let pitch = 127; pitch >= 0; pitch--) {
780 let cells = "";
781 for (let tick = a[0]; tick < a[1]; tick += a[2]) {
782 const ns = notes.filter((n) => n.on.event.data[0] === pitch);
783 cells += ns.some(
784 (n) => n.on.event.tick >= tick && n.on.event.tick < tick + a[2],
785 )
786 ? "#"
787 : ns.some((n) => n.on.event.tick <= tick && tick < n.off.event.tick)
788 ? "="
789 : ".";
790 }
791 if (/[^.]/.test(cells)) {
792 emit(noteNames[pitch].padEnd(4) + " " + cells);
793 rows++;
794 }
795 }
796 if (!rows) emit("empty");
797 break;
798 }
799 }
800}
801const input = readFileSync(0, "utf8").trimEnd().split("\n");
802let cursor = 1;
803for (let n = 0; n < Number(input[0]); n++) {
804 const words = input[cursor++].split(" "),

806 try {
807 if (cmd === "HEX" || cmd === "TEXT") {
808 const data = input.slice(cursor, cursor + Number(words[2]));
809 cursor += Number(words[2]);

811 cmd === "HEX"
812 ? parseHex(
813 data
814 .join(" ")
815 .split(" ")

817 )

821 } else command(words);
822 } catch (error) {
823 emit("error: " + (error instanceof Error ? error.message : String(error)));
824 }
825}
