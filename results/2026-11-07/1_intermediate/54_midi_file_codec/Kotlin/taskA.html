38private class Event(var tick: Long, val code: Int, var data: MutableList<Int>) {
39 fun duplicate() = Event(tick, code, data.toMutableList())
40
41 fun channel() = code in 128..239
42
43 fun ch() = (code and 15) + 1
44
45 fun on() = channel() && (code shr 4) == 9 && data[1] > 0
46
47 fun off() = channel() && ((code shr 4) == 8 || (code shr 4) == 9 && data[1] == 0)
48
49 fun end() = code == 303
50
51 fun kind(): String {
52 if (channel()) return channelKinds[(code shr 4) - 8]
53 if (code == 240) return "SYSEX"
54 if (code == 247) return "SYSEX_ESCAPE"
55 val type = code - 256
56 return if (type in 1..7) textKinds[type - 1] else metaKinds[type] ?: "META"
57 }
58}

128private fun eventText(e: Event, track: Int): String {
129 val head = "${e.tick} $track ${e.kind()}"
130 val d = e.data
131 val meta = e.code - 256
132 if (e.channel()) {
133 val fields =
134 when (e.code shr 4) {
135 8,
136 9,
137 10 -> "${noteName(d[0])} ${d[1]}"
138 14 -> (d[0] + 128 * d[1] - 8192).toString()
139 else -> d.joinToString(" ")
140 }
141 return "$head ${e.ch()} $fields"
142 }
143 if (meta in 1..7) return "$head \"${d.map { it.toChar() }.joinToString("")}\""
144 if (e.end()) return head
145 if (e.code == 240 || e.code == 247 || meta == 127 || meta !in metaKinds) {
146 return head +
147 (if (e.kind() == "META") " ${hex(meta, 2)}" else "") +
148 " ${d.size}" +
149 (if (d.isEmpty()) "" else " ${dataHex(d)}")
150 }
151 return when (meta) {
152 0,
153 81 -> "$head ${value(d)}"
154 32 -> "$head ${d[0] + 1}"
155 84 -> "$head ${d.joinToString(" ")}"
156 88 -> "$head ${d[0]}/${1 shl d[1]} ${d[2]} ${d[3]}"
157 else -> {
158 val sf = if (d[0] >= 128) d[0] - 256 else d[0]
159 "$head " + if (d[1] == 0) "${majorKeys[sf + 7]} major" else "${minorKeys[sf + 7]} minor"
160 }
161 }

164private class Parser(val data: List<Int>) {
165 var position = 0
166 var end = data.size
167 var eventStart = 0
168
169 fun error(offset: Int, reason: String): Nothing = fail("byte $offset: $reason")
170
171 fun read(): Int {
172 if (position >= end) error(eventStart, "truncated event")
173 return data[position++]
174 }
175
176 fun vlq(): Long {
177 val start = position
178 var value = 0L
179 repeat(4) {
180 val b = read()
181 value = value * 128 + (b and 127)
182 if (b < 128) return value
183 }
184 error(start, "variable-length value too long")
185 }
186
187 fun integer(p: Int, n: Int) = value(data.subList(p, p + n))
188
189 fun chunk(id: String): Int {
190 val p = position
191 if (data.size - p < 8) error(p, "truncated chunk")
192 for (i in 0..3) if (data[p + i] != id[i].code) error(p, "bad chunk id")
193 val length = integer(p + 4, 4)
194 if (p + 8L + length > data.size) error(p, "truncated chunk")
195 position += 8
196 return length.toInt()
197 }
198
199 fun validateMeta(type: Int, d: List<Int>, p: Int) {
200 val length =
201 when (type) {
202 0,
203 89 -> 2
204 32 -> 1
205 47 -> 0
206 81 -> 3
207 84 -> 5
208 88 -> 4
209 else -> -1
210 }
211 if (length >= 0 && d.size != length) error(p, "bad meta length")
212 val invalid =
213 when (type) {
214 81 -> value(d) == 0L
215 32 -> d[0] > 15
216 88 -> d[0] == 0 || d[1] > 6
217 89 -> {
218 val sf = if (d[0] >= 128) d[0] - 256 else d[0]
219 sf !in -7..7 || d[1] > 1
220 }
221 else -> false
222 }
223 if (invalid) error(p, "bad meta data")
224 }

226 fun parse(): Song {
227 if (chunk("MThd") != 6) error(4, "bad header length")
228 val format = integer(8, 2).toInt()
229 val count = integer(10, 2).toInt()
230 val division = integer(12, 2).toInt()
231 if (format > 2) error(8, "bad format")
232 if (count == 0 || format == 0 && count != 1) error(10, "bad track count")
233 if ((division and 32768) != 0) error(12, "smpte timing not supported")
234 if (division == 0) error(12, "bad division")
235 position = 14
236 val tracks = mutableListOf<MutableList<Event>>()
237 repeat(count) {
238 if (position == data.size) error(position, "missing track")
239 val length = chunk("MTrk")
240 end = position + length
241 val events = mutableListOf<Event>()
242 var tick = 0L
243 var status = -1
244 while (position < end) {
245 if (events.lastOrNull()?.end() == true) error(position, "event after end of track")
246 eventStart = position
247 tick += vlq()
248 val statusPosition = position
249 val b = read()
250 if (b < 128) {
251 if (status < 0) error(statusPosition, "data byte without running status")
252 position--
253 } else status = b
254 var code = status
255 val payload = mutableListOf<Int>()
256 when (status) {
257 in 128..239 -> {
258 val n = if ((status shr 4) == 12 || (status shr 4) == 13) 1 else 2
259 repeat(n) {
260 val p = position
261 val v = read()
262 if (v >= 128) error(p, "bad data byte")
263 payload.add(v)
264 }
265 }
266 240,
267 247,
268 255 -> {
269 val meta = status == 255
270 if (meta) code = 256 + read()
271 val size = vlq()
272 if (position + size > end) error(eventStart, "truncated event")
273 payload.addAll(data.subList(position, position + size.toInt()))
274 position += size.toInt()
275 if (meta) validateMeta(code - 256, payload, statusPosition)
276 status = -1
277 }
278 else -> error(statusPosition, "unknown status byte")
279 }
280 events.add(Event(tick, code, payload))
281 }
282 if (events.lastOrNull()?.end() != true) error(end, "missing end of track")
283 tracks.add(events)
284 end = data.size
285 }
286 if (position < data.size) error(position, "extra data")
287 return Song(format, division, tracks)
288 }
289}

291private fun bounded(text: String, low: Long, high: Long): Long {
292 val n = text.toLongOrNull() ?: fail("bad value")
293 if (n !in low..high) fail("bad value")
294 return n
295}

297private fun fields(f: List<String>, n: Int) {
298 if (f.size != n) fail("bad value")
299}

301private fun blob(f: List<String>, start: Int): MutableList<Int> {
302 if (f.size <= start) fail("bad value")
303 val length = bounded(f[start], 0, 1000000).toInt()
304 if (f.size != start + (if (length == 0) 1 else 2)) fail("bad value")
305 if (length == 0) return mutableListOf()
306 val text = f[start + 1]
307 if (text.length != 2 * length || !text.matches(Regex("[0-9A-F]*"))) fail("bad value")
308 return text.chunked(2).map { it.toInt(16) }.toMutableList()
309}

311private fun parseEvent(line: String): Event {
312 val parts = line.split(' ', limit = 4)
313 val tick = parts[0].toLong()
314 val kind = parts[2]
315 val tail = parts.getOrElse(3) { "" }
316 val f = if (tail.isEmpty()) emptyList() else tail.split(' ')
317 val channelIndex = channelKinds.indexOf(kind)
318 val textIndex = textKinds.indexOf(kind)
319 var code: Int
320 var data = mutableListOf<Int>()
321 if (channelIndex >= 0) {
322 fields(f, if (channelIndex in 4..6) 2 else 3)
323 val ch = bounded(f[0], 1, 16).toInt()
324 code = (channelIndex + 8) * 16 + ch - 1
325 when (channelIndex) {
326 0,
327 1,
328 2 -> {
329 data.add(noteNumber(f[1]))
330 data.add(bounded(f[2], 0, 127).toInt())
331 }
332 6 -> {
333 val bend = bounded(f[1], -8192, 8191).toInt() + 8192
334 data.add(bend % 128)
335 data.add(bend / 128)
336 }
337 else -> data.addAll(f.drop(1).map { bounded(it, 0, 127).toInt() })
338 }
339 } else if (textIndex >= 0) {
340 if (
341 tail.length < 2 ||
342 tail.first() != '"' ||
343 tail.last() != '"' ||
344 '"' in tail.substring(1, tail.length - 1)
345 )
346 fail("bad value")
347 code = 257 + textIndex
348 data.addAll(tail.substring(1, tail.length - 1).map { it.code })
349 } else if (kind == "SYSEX" || kind == "SYSEX_ESCAPE") {
350 code = if (kind == "SYSEX") 240 else 247
351 data = blob(f, 0)
352 } else if (kind == "META") {
353 if (f.size < 2 || !f[0].matches(Regex("[0-9A-F]{2}"))) fail("bad value")
354 val type = f[0].toInt(16)
355 if (type <= 7 || type in metaKinds) fail("bad value")
356 code = 256 + type
357 data = blob(f, 1)
358 } else {
359 val meta = metaKinds.entries.find { it.value == kind }?.key ?: fail("unknown event type")
360 code = 256 + meta
361 when (meta) {
362 0 -> {
363 fields(f, 1)
364 data = bytes(bounded(f[0], 0, 65535), 2)
365 }
366 32 -> {
367 fields(f, 1)
368 data.add(bounded(f[0], 1, 16).toInt() - 1)
369 }
370 47 -> fields(f, 0)
371 81 -> {
372 fields(f, 1)
373 data = bytes(bounded(f[0], 1, 16777215), 3)
374 }
375 84 -> {
376 fields(f, 5)
377 data.addAll(f.map { bounded(it, 0, 255).toInt() })
378 }
379 88 -> {
380 fields(f, 3)
381 val signature = f[0].split('/')
382 if (signature.size != 2) fail("bad value")
383 val numerator = bounded(signature[0], 1, 255).toInt()
384 val denominator = bounded(signature[1], 1, 64).toInt()
385 if ((denominator and (denominator - 1)) != 0) fail("bad value")
386 data.addAll(
387 listOf(
388 numerator,
389 Integer.numberOfTrailingZeros(denominator),
390 bounded(f[1], 0, 255).toInt(),
391 bounded(f[2], 0, 255).toInt(),
392 ),
393 )
394 }
395 89 -> {
396 fields(f, 2)
397 if (f[1] != "minor" && f[1] != "major") fail("bad value")
398 val minor = f[1] == "minor"
399 val index = (if (minor) minorKeys else majorKeys).indexOf(f[0])
400 if (index < 0) fail("bad value")
401 data.addAll(listOf((index - 7 + 256) % 256, if (minor) 1 else 0))
402 }
403 127 -> data = blob(f, 0)
404 }
405 }
406 return Event(tick, code, data)
407}

409private fun textSong(lines: List<String>): Song {
410 val h = lines[0].split(' ')
411 var format = 0
412 var count = 0
413 var division = 0
414 try {
415 if (h.size != 4 || h[0] != "HEADER") fail("bad header")
416 format = bounded(h[1], 0, 2).toInt()
417 count = bounded(h[2], 1, 65535).toInt()
418 division = bounded(h[3], 1, 32767).toInt()
419 if (format == 0 && count != 1) fail("bad header")
420 } catch (error: Failure) {
421 fail("line 1: bad header")
422 }
423 val tracks = mutableListOf(mutableListOf<Event>())
424 var current = 0
425 for (i in 1 until lines.size) {
426 try {
427 val e = parseEvent(lines[i])
428 val track = lines[i].split(' ', limit = 4)[1].toInt()
429 if (track >= count) fail("bad track")
430 var events = tracks[current]
431 val end = events.lastOrNull()?.end() == true
432 if (track < current) fail("out of order")
433 if (track > current && (!end || track > current + 1)) fail("missing end of track")
434 if (track == current && end) fail("event after end of track")
435 if (track == current + 1) {
436 current++
437 events = mutableListOf()
438 tracks.add(events)
439 }
440 if (events.isNotEmpty() && e.tick < events.last().tick) fail("out of order")
441 events.add(e)
442 } catch (error: Failure) {
443 fail("line ${i + 1}: ${error.message}")
444 }
445 }
446 if (current + 1 < count || tracks[current].lastOrNull()?.end() != true)
447 fail("line ${lines.size + 1}: missing end of track")
448 return Song(format, division, tracks)
449}

451private fun vlq(number: Long): List<Int> {
452 var n = number
453 val result = mutableListOf((n and 127).toInt())
454 n = n shr 7
455 while (n > 0) {
456 result.add(0, (n and 127).toInt() or 128)
457 n = n shr 7
458 }
459 return result
460}

462private fun encode(s: Song) {
463 val result = mutableListOf<Int>()
464 fun chunk(target: MutableList<Int>, id: String, data: List<Int>) {
465 target.addAll(id.map { it.code })
466 target.addAll(bytes(data.size.toLong(), 4))
467 target.addAll(data)
468 }
469 chunk(
470 result,
471 "MThd",
472 bytes(s.format.toLong(), 2) +
473 bytes(s.tracks.size.toLong(), 2) +
474 bytes(s.division.toLong(), 2),
475 )
476 for (track in s.tracks) {
477 val data = mutableListOf<Int>()
478 var last = 0L
479 var previous = -1
480 for (e in track) {
481 data.addAll(vlq(e.tick - last))
482 last = e.tick
483 if (e.channel()) {
484 var code = e.code
485 var payload: List<Int> = e.data
486 if (zeroOff && (code shr 4) == 8) {
487 code += 16
488 payload = listOf(e.data[0], 0)
489 }
490 if (!running || previous != code) data.add(code)
491 data.addAll(payload)
492 previous = code
493 } else {
494 previous = -1
495 if (e.code < 256) data.add(e.code) else data.addAll(listOf(255, e.code - 256))
496 data.addAll(vlq(e.data.size.toLong()))
497 data.addAll(e.data)
498 }
499 }
500 chunk(result, "MTrk", data)
501 }
502 for (i in result.indices step 16) emit(
503 "${hex(i, 4)}: ${result.subList(i, minOf(i + 16, result.size)).joinToString(" ") { hex(it, 2) }}",
504 )
505 emit("${result.size} bytes")
506}

508private fun pairing(s: Song): Pairing {
509 val queues = mutableMapOf<Int, ArrayDeque<Located>>()
510 val notes = mutableListOf<Note>()
511 val unmatched = mutableListOf<Located>()
512 val events = global(s)
513 for (l in events) {
514 val e = l.event
515 if (!e.on() && !e.off()) continue
516 val q = queues.getOrPut(e.ch() * 128 + e.data[0]) { ArrayDeque() }
517 if (e.on()) q.addLast(l)
518 else if (q.isEmpty()) unmatched.add(l) else notes.add(Note(q.removeFirst(), l))
519 }
520 queues.values.forEach { unmatched.addAll(it) }
521 val order = events.mapIndexed { i, l -> l.event to i }.toMap()
522 notes.sortWith(
523 compareBy<Note> { it.start() }
524 .thenBy { it.channel() }
525 .thenBy { it.pitch() }
526 .thenBy { order.getValue(it.on.event) },
527 )
528 unmatched.sortBy { order.getValue(it.event) }
529 return Pairing(notes, unmatched)
530}

532private fun time(s: Song, tick: Long): String {
533 var total = 0L
534 var last = 0L
535 var tempo = 500000L
536 for (l in global(s)) if (l.event.tick <= tick && l.event.code == 337) {
537 total += (l.event.tick - last) * tempo
538 last = l.event.tick
539 tempo = value(l.event.data)
540 }
541 val us = (total + (tick - last) * tempo) / s.division
542 return "${us / 1000}.${(us % 1000).toString().padStart(3, '0')}"
543}

545private fun bar(s: Song, tick: Long): String {
546 var start = 0L
547 var number = 1L
548 var numerator = 4L
549 var beat = s.division.toLong()
550 for (l in global(s)) if (l.event.tick <= tick && l.event.code == 344) {
551 val length = numerator * beat
552 number += (l.event.tick - start + length - 1) / length
553 start = l.event.tick
554 numerator = l.event.data[0].toLong()
555 beat = s.division * 4L / (1L shl l.event.data[1])
556 }
557 val x = tick - start
558 val length = numerator * beat
559 return "${number + x / length}:${x % length / beat + 1}:${x % beat}"
560}

572private fun execute(w: List<String>, read: () -> String) {
573 val c = w[0]
574 if (c == "HEX" || c == "TEXT") {
575 val lines = List(w[2].toInt()) { read() }
576 val s =
577 if (c == "TEXT") textSong(lines)
578 else Parser(lines.flatMap { it.split(' ').map { b -> b.toInt(16) } }).parse()
579 songs[w[1]] = s
580 emit("loaded ${w[1]} ${description(s)}")
581 return
582 }
583 if (
584 c !in
585 listOf(
586 "DECODE",
587 "ENCODE",
588 "TIMING",
589 "NOTES",
590 "STATS",
591 "ROLL",
592 "TRANSPOSE",
593 "QUANTIZE",
594 "VELOCITY",
595 "TEMPO",
596 "FILTER",
597 "MERGE",
598 "SPLIT",
599 "COPY",
600 "LIST",
601 "RUNNING",
602 "NOTEOFF",
603 )
604 )
605 fail("unknown command")
606 if (c == "LIST") {
607 if (songs.isEmpty()) emit("no songs")
608 else songs.forEach { (name, s) -> emit("$name ${description(s)}") }
609 return
610 }
611 if (c == "RUNNING" || c == "NOTEOFF") {
612 if (w[1] !in (if (c == "RUNNING") listOf("ON", "OFF") else listOf("ZERO", "KEEP")))
613 fail("bad argument")
614 if (c == "RUNNING") running = w[1] == "ON" else zeroOff = w[1] == "ZERO"
615 emit("${c.lowercase()} ${w[1].lowercase()}")
616 return
617 }
618 val s = songs[w[1]] ?: fail("no song ${w[1]}")
619 var a = 0L
620 var b = 0L
621 var d = 0L
622 if (c in listOf("TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "ROLL")) {
623 a = w[2].toLong()
624 if (w.size > 3) b = w[3].toLong()
625 if (c == "ROLL") d = w[4].toLong()
626 val valid =
627 when (c) {
628 "TRANSPOSE" -> a in -127L..127L && (w.size == 3 || b in 1L..16L)
629 "QUANTIZE" -> a in 1L..1000000L && b in 0L..100L
630 "VELOCITY" -> a in 0L..1000L && b in -127L..127L
631 "TEMPO" -> a in 1L..1000L && b in 1L..1000L
632 else ->
633 a in 0L..10000000L &&
634 b in 1L..10000000L &&
635 d in 1L..10000000L &&
636 b > a &&
637 (b - a + d - 1) / d <= 64
638 }
639 if (!valid) fail("bad argument")
640 }
641 if (c == "FILTER" && w.drop(2).any { it.toLong() !in 1L..16L }) fail("bad argument")
642 if (c in listOf("TIMING", "STATS") && s.format == 2) fail("format 2 not supported")
643 when (c) {
644 "COPY" -> {
645 songs[w[2]] = s.duplicate()
646 emit("copied ${w[1]} to ${w[2]}")
647 }
648 "DECODE" -> {
649 emit("HEADER ${s.format} ${s.tracks.size} ${s.division}")
650 s.tracks.forEachIndexed { t, events -> events.forEach { emit(eventText(it, t)) } }
651 }
652 "ENCODE" -> encode(s)
653 "TIMING" -> {
654 for (l in global(s)) emit(
655 "${l.event.tick} ${l.track} ${l.event.kind()} ${time(s, l.event.tick)} ${bar(s, l.event.tick)}",
656 )
657 emit("end ${s.length()} ${time(s, s.length())} ${bar(s, s.length())}")
658 }
659 "NOTES" -> printNotes(pairing(s))
660 "TRANSPOSE" -> {
661 var kept = 0
662 val removed = mutableListOf<String>()
663 s.tracks.forEachIndexed { t, track ->
664 val iterator = track.iterator()
665 while (iterator.hasNext()) {
666 val e = iterator.next()
667 if (
668 e.channel() && (e.code shr 4) <= 10 && (w.size == 3 || e.ch().toLong() == b)
669 ) {
670 val original = e.data[0]
671 val pitch = original + a
672 if (pitch !in 0L..127L) {
673 removed.add(
674 "dropped ${e.tick} $t ${e.kind()} ${e.ch()} ${noteName(original)}",
675 )
676 iterator.remove()
677 } else {
678 e.data[0] = pitch.toInt()
679 kept++
680 }
681 }
682 }
683 }
684 emit("transposed $kept dropped ${removed.size}")
685 removed.forEach { emit(it) }
686 }
687 "QUANTIZE" -> {
688 val p = pairing(s)
689 val targets = p.notes + p.unmatched.filter { it.event.on() }.map { Note(it, it) }
690 var moved = 0
691 for (n in targets) {
692 val tick = n.start()
693 val lower = tick / a * a
694 val g = if (tick - lower <= lower + a - tick) lower else lower + a
695 val delta = Math.floorDiv(2 * (g - tick) * b + 100, 200L)
696 n.on.event.tick += delta
697 if (n.off.event !== n.on.event) n.off.event.tick += delta
698 if (delta != 0L) moved++
699 }
700 for (track in s.tracks) {
701 val end = track.removeAt(track.lastIndex)
702 track.sortBy { it.tick }
703 end.tick = maxOf(end.tick, track.maxOfOrNull { it.tick } ?: 0L)
704 track.add(end)
705 }
706 emit("quantized ${targets.size} moved $moved")
707 }
708 "VELOCITY" -> {
709 var count = 0
710 var clamped = 0
711 for (l in global(s)) if (l.event.on()) {
712 val v = (l.event.data[1] * a + 50) / 100 + b
713 if (v !in 1L..127L) clamped++
714 l.event.data[1] = v.coerceIn(1, 127).toInt()
715 count++
716 }
717 emit("velocity $count clamped $clamped")
718 }
719 "TEMPO" -> {
720 val edited = s.duplicate()
721 var count = 0
722 var atZero = false
723 for (l in global(edited)) if (l.event.code == 337) {
724 val v = (2 * value(l.event.data) * b + a) / (2 * a)
725 if (v !in 1L..16777215L) fail("tempo out of range")
726 l.event.data = bytes(v, 3)
727 count++
728 if (l.event.tick == 0L) atZero = true
729 }
730 val inserted = (1000000 * b + a) / (2 * a)
731 if (!atZero) {
732 if (inserted !in 1L..16777215L) fail("tempo out of range")
733 edited.tracks[0].add(0, Event(0, 337, bytes(inserted, 3)))
734 }
735 songs[w[1]] = edited
736 emit("scaled $count")
737 if (!atZero) emit("inserted 0 0 TEMPO $inserted")
738 }
739 "FILTER" -> {
740 val channels = w.drop(2).map { it.toInt() }.toSet()
741 val before = s.count()
742 s.tracks.forEach { it.removeAll { e -> e.channel() && e.ch() !in channels } }
743 emit("removed ${before - s.count()}")
744 }
745 "MERGE" -> {
746 if (s.format != 1) fail("not format 1")
747 val original = s.tracks.size
748 val length = s.length()
749 val events = global(s).filter { !it.event.end() }.map { it.event }.toMutableList()
750 events.add(Event(length, 303, mutableListOf()))
751 s.tracks = mutableListOf(events)
752 s.format = 0
753 emit("merged $original tracks events ${s.count()}")
754 }
755 "SPLIT" -> {
756 if (s.format != 0) fail("not format 0")
757 val length = s.length()
758 val groups = TreeMap<Int, MutableList<Event>>()
759 groups[0] = mutableListOf()
760 for (e in s.tracks[0]) if (!e.end())
761 groups.getOrPut(if (e.channel()) e.ch() else 0) { mutableListOf() }.add(e)
762 s.tracks = groups.values.toMutableList()
763 s.tracks.forEach { it.add(Event(length, 303, mutableListOf())) }
764 s.format = 1
765 emit("split tracks ${s.tracks.size} events ${s.count()}")
766 }
767 "STATS" -> {
768 val p = pairing(s)
769 emit(description(s))
770 emit("notes ${p.notes.size} unmatched ${p.unmatched.size}")
771 for (ch in 1..16) {
772 val notes = p.notes.filter { it.channel() == ch }
773 if (notes.isNotEmpty())
774 emit(
775 "channel $ch notes ${notes.size} range ${noteName(notes.minOf { it.pitch() })} ${noteName(notes.maxOf { it.pitch() })}",
776 )
777 }
778 emit(
779 if (p.notes.isEmpty()) "range none"
780 else
781 "range ${noteName(p.notes.minOf { it.pitch() })} ${noteName(p.notes.maxOf { it.pitch() })}",
782 )
783 emit("length ${s.length()} ticks ${time(s, s.length())} ms")
784 val changes = TreeMap<Long, Int>()
785 for (n in p.notes) if (n.length() > 0) {
786 changes[n.start()] = (changes[n.start()] ?: 0) + 1
787 val end = n.start() + n.length()
788 changes[end] = (changes[end] ?: 0) - 1
789 }
790 var active = 0
791 var maximum = 0
792 var first = 0L
793 for ((tick, change) in changes) {
794 active += change
795 if (active > maximum) {
796 maximum = active
797 first = tick
798 }
799 }
800 emit(if (maximum == 0) "polyphony 0" else "polyphony $maximum at $first")
801 }
802 "ROLL" -> {
803 val p = pairing(s)
804 val columns = ((b - a + d - 1) / d).toInt()
805 var any = false
806 for (pitch in 127 downTo 0) {
807 val line = StringBuilder()
808 var visible = false
809 for (i in 0 until columns) {
810 val tick = a + i * d
811 val notes = p.notes.filter { it.pitch() == pitch }
812 val mark =
813 if (notes.any { tick <= it.start() && it.start() < tick + d }) '#'
814 else if (
815 notes.any { it.start() <= tick && tick < it.start() + it.length() }
816 )
817 '='
818 else '.'
819 line.append(mark)
820 if (mark != '.') visible = true
821 }
822 if (visible) {
823 emit("${noteName(pitch).padEnd(4)} $line")
824 any = true
825 }
826 }
827 if (!any) emit("empty")
828 }
829 }
830}
