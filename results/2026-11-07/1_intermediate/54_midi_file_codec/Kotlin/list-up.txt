private fun vlq(number: Long): List<Int> {
    var n = number
    val result = mutableListOf((n and 127).toInt())
    n = n shr 7
    while (n > 0) {
        result.add(0, (n and 127).toInt() or 128)
        n = n shr 7
    }
    return result
