51event_line <- function(e, track) {
52 fields <- c(e$tick, track, kind(e))
53 d <- e$data
54 if (e$status < 240) {
55 fields <- c(fields, channel(e))
56 high <- e$status %/% 16
57 if (high %in% c(8, 9, 10)) {
58 fields <- c(fields, note_name(d[1]), d[2])
59 } else if (high == 14) {
60 fields <- c(fields, d[1] + 128 * d[2] - 8192)
61 } else {
62 fields <- c(fields, d)
63 }
64 } else if (e$status == 255 && e$meta >= 1 && e$meta <= 7) {
65 text <- if (length(d)) rawToChar(as.raw(d)) else ""
66 fields <- c(fields, paste0('"', text, '"'))
67 } else if (is_meta(e, 0) || is_meta(e, 81)) {
68 fields <- c(fields, bigendian(d))
69 } else if (is_meta(e, 32)) {
70 fields <- c(fields, d[1] + 1)
71 } else if (is_meta(e, 47)) {
72 } else if (is_meta(e, 84)) {
73 fields <- c(fields, d)
74 } else if (is_meta(e, 88)) {
75 fields <- c(fields, paste0(d[1], "/", 2^d[2]), d[3:4])
76 } else if (is_meta(e, 89)) {
77 sf <- if (d[1] >= 128) d[1] - 256 else d[1]
78 fields <- c(fields, key_names[[d[2] + 1]][sf + 8], if (d[2] == 0) "major" else "minor")
79 } else {
80 if (e$status == 255 && e$meta != 127) fields <- c(fields, sprintf("%02X", as.integer(e$meta)))
81 fields <- c(fields, length(d))
82 if (length(d)) fields <- c(fields, paste(sprintf("%02X", as.integer(d)), collapse = ""))
83 }
84 paste(fields, collapse = " ")

92parse_hex <- function(data) {
93 chunk <- function(pos, id) {
94 if (length(data) - pos < 8) parse_failure(pos, "truncated chunk")
95 if (!identical(as.integer(data[(pos + 1):(pos + 4)]), as.integer(charToRaw(id)))) parse_failure(pos, "bad chunk id")
96 len <- bigendian(data[(pos + 5):(pos + 8)])
97 if (pos + 8 + len > length(data)) parse_failure(pos, "truncated chunk")
98 c(pos + 8, pos + 8 + len)
99 }
100 bounds <- chunk(0, "MThd")
101 if (diff(bounds) != 6) parse_failure(4, "bad header length")
102 f <- bigendian(data[9:10])
103 count <- bigendian(data[11:12])
104 division <- bigendian(data[13:14])
105 if (f > 2) parse_failure(8, "bad format")
106 if (!count || (f == 0 && count != 1)) parse_failure(10, "bad track count")
107 if (division >= 32768) parse_failure(12, "smpte timing not supported")
108 if (!division) parse_failure(12, "bad division")
109 s <- list(format = f, division = division, tracks = list())
110 pos <- bounds[2]
111 for (t in seq_len(count)) {
112 if (pos == length(data)) parse_failure(pos, "missing track")
113 bounds <- chunk(pos, "MTrk")
114 pos <- bounds[1]
115 finish <- bounds[2]
116 tick <- 0
117 running <- 0
118 track <- list()
119 while (pos < finish) {
120 if (length(track) && is_meta(track[[length(track)]], 47)) parse_failure(pos, "event after end of track")
121 event_start <- pos
122 read_byte <- function() {
123 if (pos >= finish) parse_failure(event_start, "truncated event")
124 pos <<- pos + 1
125 data[pos]
126 }
127 vlq <- function() {
128 at <- pos
129 result <- 0
130 for (i in 1:4) {
131 b <- read_byte()
132 result <- result * 128 + b %% 128
133 if (b < 128) return(result)
134 }
135 parse_failure(at, "variable-length value too long")
136 }
137 tick <- tick + vlq()
138 status_pos <- pos
139 if (pos >= finish) parse_failure(event_start, "truncated event")
140 status <- data[pos + 1]
141 if (status < 128) {
142 if (!running) parse_failure(pos, "data byte without running status")
143 status <- running
144 } else {
145 pos <- pos + 1
146 }
147 e <- event(tick, status)
148 if (status < 240) {
149 running <- status
150 n <- if (status %/% 16 %in% c(12, 13)) 1L else 2L
151 for (j in seq_len(n)) {
152 at <- pos
153 b <- read_byte()
154 if (b >= 128) parse_failure(at, "bad data byte")
155 e$data <- c(e$data, b)
156 }
157 } else if (status %in% c(240, 247, 255)) {
158 running <- 0
159 if (status == 255) e$meta <- read_byte()
160 len <- vlq()
161 if (pos + len > finish) parse_failure(event_start, "truncated event")
162 e$data <- slice(data, pos + 1, pos + len)
163 pos <- pos + len
164 if (status == 255) {
165 sizes <- c(`0` = 2, `32` = 1, `47` = 0, `81` = 3, `84` = 5, `88` = 4, `89` = 2)
166 code <- as.character(e$meta)
167 if (code %in% names(sizes) && sizes[[code]] != len) parse_failure(status_pos, "bad meta length")
168 d <- e$data
169 bad <- e$meta == 81 && bigendian(d) == 0
170 bad <- bad || (e$meta == 32 && d[1] > 15)
171 bad <- bad || (e$meta == 88 && (d[1] == 0 || d[2] > 6))
172 if (e$meta == 89) {
173 sf <- if (d[1] >= 128) d[1] - 256 else d[1]
174 bad <- bad || sf < -7 || sf > 7 || d[2] > 1
175 }
176 if (bad) parse_failure(status_pos, "bad meta data")
177 }
178 } else {
179 parse_failure(status_pos, "unknown status byte")
180 }
181 track[[length(track) + 1L]] <- e
182 }
183 if (!length(track) || !is_meta(track[[length(track)]], 47)) parse_failure(finish, "missing end of track")
184 s$tracks[[t]] <- track
185 pos <- finish
186 }
187 if (pos < length(data)) parse_failure(pos, "extra data")
188 s
189}
190parse_text <- function(lines) {
191 lineno <- 1L
192 fail <- function(reason) parse_failure(lineno, reason)
193 require <- function(ok) if (!isTRUE(ok)) fail("bad value")
194 h <- strsplit(lines[1], " ", fixed = TRUE)[[1]]
195 if (length(h) != 4L || h[1] != "HEADER") fail("bad header")
196 f <- as.numeric(h[2])
197 count <- as.numeric(h[3])
198 division <- as.numeric(h[4])
199 if (!(f >= 0 && f <= 2 && count >= 1 && count <= 65535 && division >= 1 && division <= 32767 && (f != 0 || count == 1))) fail("bad header")
200 s <- list(format = f, division = division, tracks = rep(list(list()), count))
201 current <- 1L
202 for (i in seq.int(2L, length(lines) + 1L)) {
203 if (i > length(lines)) break
204 lineno <- i
205 text <- lines[i]
206 w <- strsplit(text, " ", fixed = TRUE)[[1]]
207 tick <- as.numeric(w[1])
208 track <- as.numeric(w[2]) + 1
209 name <- w[3]
210 rest <- if (length(w) > 3L) substring(text, nchar(paste(w[1:3], collapse = " ")) + 2L) else ""
211 fields <- if (nzchar(rest)) strsplit(rest, " ", fixed = TRUE)[[1]] else character()
212 index <- match(name, channel_names)
213 meta <- match(name, meta_names)
214 known <- !is.na(meta)
215 if (is.na(index) && !known && !(name %in% c("META", "SYSEX", "SYSEX_ESCAPE"))) fail("unknown event type")
216 e <- event(tick, 255)
217 value <- function(at, lo, hi) {
218 require(at <= length(fields))
219 n <- as.numeric(fields[at])
220 require(n >= lo && n <= hi)
221 n
222 }
223 if (!is.na(index)) {
224 n <- if (index %in% c(5, 6, 7)) 2L else 3L
225 require(length(fields) == n)
226 e$status <- (index + 7) * 16 + value(1, 1, 16) - 1
227 if (index <= 3) {
228 pitch <- match(fields[2], vapply(0:127, note_name, character(1))) - 1L
229 require(!is.na(pitch))
230 e$data <- c(pitch, value(3, 0, 127))
231 } else if (index == 7) {
232 bend <- value(2, -8192, 8191) + 8192
233 e$data <- c(bend %% 128, bend %/% 128)
234 } else {
235 e$data <- vapply(2:n, function(j) value(j, 0, 127), numeric(1))
236 }
237 } else if (known && as.numeric(names(meta_names)[meta]) %in% 1:7) {
238 require(nchar(rest) >= 2L && startsWith(rest, '"') && endsWith(rest, '"'))
239 body <- substring(rest, 2L, nchar(rest) - 1L)
240 require(!grepl('"', body, fixed = TRUE))
241 e$meta <- as.numeric(names(meta_names)[meta])
242 e$data <- as.numeric(charToRaw(body))
243 } else if (name %in% c("META", "SYSEX", "SYSEX_ESCAPE", "SEQUENCER_SPECIFIC")) {
244 if (name == "SYSEX") {
245 e$status <- 240
246 } else if (name == "SYSEX_ESCAPE") {
247 e$status <- 247
248 } else if (name == "META") {
249 require(length(fields) >= 2L && grepl("^[0-9A-F]{2}$", fields[1]))
250 e$meta <- strtoi(fields[1], base = 16L)
251 require(!(as.character(e$meta) %in% names(meta_names)))
252 fields <- fields[-1]
253 } else {
254 e$meta <- 127
255 }
256 len <- value(1, 0, 1000000)
257 if (!len) {
258 require(length(fields) == 1L)
259 } else {
260 require(length(fields) == 2L && nchar(fields[2]) == len * 2)
261 require(grepl("^[0-9A-F]+$", fields[2]))
262 starts <- seq.int(1L, nchar(fields[2]), by = 2L)
263 e$data <- strtoi(substring(fields[2], starts, starts + 1L), base = 16L)
264 }
265 } else {
266 e$meta <- as.numeric(names(meta_names)[meta])
267 if (e$meta == 0) {
268 require(length(fields) == 1L)
269 e$data <- bytes_be(value(1, 0, 65535), 2L)
270 } else if (e$meta == 32) {
271 require(length(fields) == 1L)
272 e$data <- value(1, 1, 16) - 1
273 } else if (e$meta == 47) {
274 require(!length(fields))
275 } else if (e$meta == 81) {
276 require(length(fields) == 1L)
277 e$data <- bytes_be(value(1, 1, 16777215), 3L)
278 } else if (e$meta == 84) {
279 require(length(fields) == 5L)
280 e$data <- vapply(1:5, function(j) value(j, 0, 255), numeric(1))
281 } else if (e$meta == 88) {
282 require(length(fields) == 3L)
283 fraction <- as.numeric(strsplit(fields[1], "/", fixed = TRUE)[[1]])
284 exponent <- match(fraction[2], 2^(0:6)) - 1L
285 require(fraction[1] >= 1 && fraction[1] <= 255 && !is.na(exponent))
286 e$data <- c(fraction[1], exponent, value(2, 0, 255), value(3, 0, 255))
287 } else if (e$meta == 89) {
288 require(length(fields) == 2L)
289 found <- FALSE
290 for (mi in 0:1) {
291 for (sf in -7:7) {
292 if (identical(fields, c(key_names[[mi + 1L]][sf + 8L], if (!mi) "major" else "minor"))) {
293 e$data <- c(sf %% 256, mi)
294 found <- TRUE
295 }
296 }
297 }
298 require(found)
299 }
300 }
301 if (track > count) fail("bad track")
302 row <- s$tracks[[current]]
303 ended <- length(row) && is_meta(row[[length(row)]], 47)
304 if (track < current) fail("out of order")
305 if (track > current && (!ended || track > current + 1L)) fail("missing end of track")
306 if (track == current && ended) fail("event after end of track")
307 if (track == current + 1L) {
308 current <- current + 1L
309 row <- s$tracks[[current]]
310 }
311 if (length(row) && tick < row[[length(row)]]$tick) fail("out of order")
312 s$tracks[[current]][[length(row) + 1L]] <- e
313 }
314 lineno <- length(lines) + 1L
315 row <- s$tracks[[current]]
316 if (current < count || !length(row) || !is_meta(row[[length(row)]], 47)) fail("missing end of track")
317 s
318}

322 while (value > 0) {
323 data <- c(value %% 128 + 128, data)
324 value <- value %/% 128

328encode <- function(s, running, zero) {
329 data <- c(as.numeric(charToRaw("MThd")), bytes_be(6, 4L), bytes_be(s$format, 2L), bytes_be(length(s$tracks), 2L), bytes_be(s$division, 2L))
330 for (track in s$tracks) {
331 row <- numeric()
332 last <- 0
333 status <- 0
334 for (e in track) {
335 row <- c(row, vlq(e$tick - last))
336 last <- e$tick
337 st <- e$status
338 body <- e$data
339 if (zero && st %/% 16 == 8) {
340 st <- st + 16
341 body <- c(body[1], 0)
342 }
343 if (st < 240) {
344 if (!running || status != st) row <- c(row, st)
345 row <- c(row, body)
346 status <- st
347 } else {
348 status <- 0
349 row <- c(row, st)
350 if (st == 255) row <- c(row, e$meta)
351 row <- c(row, vlq(length(body)), body)
352 }
353 }
354 data <- c(data, as.numeric(charToRaw("MTrk")), bytes_be(length(row), 4L), row)
355 }
356 data

358pairing <- function(s) {
359 queues <- list()
360 notes <- list()
361 unmatched <- list()
362 for (p in ordered(s)) {
363 e <- p$event
364 if (!is_on(e) && !is_off(e)) next
365 key <- paste(channel(e), e$data[1], sep = ":")
366 queue <- queues[[key]]
367 if (is_on(e)) {
368 queue[[length(queue) + 1L]] <- p
369 queues[[key]] <- queue
370 } else if (!length(queue)) {
371 unmatched[[length(unmatched) + 1L]] <- p
372 } else {
373 on <- queue[[1L]]
374 queues[[key]] <- queue[-1L]
375 notes[[length(notes) + 1L]] <- list(on = on, off = p, length = e$tick - on$event$tick)
376 }
377 }
378 for (queue in queues) unmatched <- c(unmatched, queue)
379 if (length(notes)) {
380 notes <- notes[order(vapply(notes, function(n) n$on$event$tick, numeric(1)), vapply(notes, function(n) channel(n$on$event), numeric(1)), vapply(notes, function(n) n$on$event$data[1], numeric(1)), vapply(notes, function(n) n$on$order, numeric(1)))]
381 }
382 if (length(unmatched)) unmatched <- unmatched[order(vapply(unmatched, function(p) p$order, numeric(1)))]
383 list(notes = notes, unmatched = unmatched)
384}
385time_at <- function(s, tick) {
386 total <- 0
387 last <- 0
388 tempo <- 500000
389 for (p in ordered(s)) {
390 e <- p$event
391 if (e$tick > tick) break
392 if (is_meta(e, 81)) {
393 total <- total + (e$tick - last) * tempo
394 last <- e$tick
395 tempo <- bigendian(e$data)
396 }
397 }
398 usec <- (total + (tick - last) * tempo) %/% s$division
399 paste0(usec %/% 1000, ".", sprintf("%03d", as.integer(usec %% 1000)))
400}
401bar_at <- function(s, tick) {
402 start <- 0
403 bar <- 1
404 nn <- 4
405 beat <- s$division
406 for (p in ordered(s)) {
407 e <- p$event
408 if (e$tick > tick) break
409 if (is_meta(e, 88)) {
410 size <- nn * beat
411 bar <- bar + (e$tick - start + size - 1) %/% size
412 start <- e$tick
413 nn <- e$data[1]
414 beat <- (s$division * 4) %/% 2^e$data[2]
415 }
416 }
417 x <- tick - start
418 paste(bar + x %/% (nn * beat), x %% (nn * beat) %/% beat + 1, x %% beat, sep = ":")
419}

424execute <- function(words) {

426 a <- words[-1]
427 commands <- c("DECODE", "ENCODE", "TIMING", "NOTES", "STATS", "ROLL", "TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "FILTER", "MERGE", "SPLIT", "COPY", "LIST", "RUNNING", "NOTEOFF")
428 if (!(cmd %in% commands)) {
429 line("error: unknown command")
430 return()
431 }
432 if (cmd == "LIST") {
433 names <- sort(ls(store), method = "radix")
434 for (name in names) line(name, " ", summary_song(store[[name]]))
435 if (!length(names)) line("no songs")
436 return()
437 }
438 if (cmd %in% c("RUNNING", "NOTEOFF")) {
439 allowed <- if (cmd == "RUNNING") c("ON", "OFF") else c("ZERO", "KEEP")
440 if (!(a[1] %in% allowed)) {
441 line("error: bad argument")
442 } else {
443 if (cmd == "RUNNING") settings$running <- a[1] == "ON" else settings$zero <- a[1] == "ZERO"
444 line(tolower(cmd), " ", tolower(a[1]))

446 return()
447 }

451 }

453 args <- if (cmd == "COPY") numeric() else as.numeric(a[-1])
454 valid <- TRUE
455 if (cmd == "TRANSPOSE") valid <- args[1] >= -127 && args[1] <= 127 && (length(args) == 1L || (args[2] >= 1 && args[2] <= 16))
456 if (cmd == "QUANTIZE") valid <- args[1] >= 1 && args[1] <= 1000000 && args[2] >= 0 && args[2] <= 100
457 if (cmd == "VELOCITY") valid <- args[1] >= 0 && args[1] <= 1000 && args[2] >= -127 && args[2] <= 127
458 if (cmd == "TEMPO") valid <- all(args >= 1 & args <= 1000)
459 if (cmd == "FILTER") valid <- all(args >= 1 & args <= 16)
460 if (cmd == "ROLL") {
461 valid <- args[1] >= 0 && args[1] <= 10000000 && args[2] >= 1 && args[2] <= 10000000 && args[3] >= 1 && args[3] <= 10000000 && args[2] > args[1]
462 if (valid) valid <- (args[2] - args[1] + args[3] - 1) %/% args[3] <= 64
463 }
464 if (!valid) {
465 line("error: bad argument")
466 return()
467 }

469 line("error: format 2 not supported")
470 return()
471 }
472 if (cmd == "MERGE" && s$format != 1) {
473 line("error: not format 1")
474 return()
475 }
476 if (cmd == "SPLIT" && s$format != 0) {
477 line("error: not format 0")
478 return()
479 }
480 if (cmd == "COPY") {
481 store[[a[2]]] <- s
482 line("copied ", a[1], " to ", a[2])
483 } else if (cmd == "DECODE") {
484 line("HEADER ", s$format, " ", length(s$tracks), " ", s$division)
485 for (t in seq_along(s$tracks)) {
486 for (e in s$tracks[[t]]) line(event_line(e, t - 1L))
487 }
488 } else if (cmd == "ENCODE") {
489 data <- encode(s, settings$running, settings$zero)
490 for (i in seq.int(1L, length(data), by = 16L)) {
491 line(sprintf("%04X: ", i - 1L), paste(sprintf("%02X", as.integer(data[i:min(i + 15L, length(data))])), collapse = " "))
492 }
493 line(length(data), " bytes")
494 } else if (cmd == "TIMING") {
495 for (p in ordered(s)) {
496 e <- p$event
497 line(e$tick, " ", p$track, " ", kind(e), " ", time_at(s, e$tick), " ", bar_at(s, e$tick))
498 }
499 tick <- song_length(s)
500 line("end ", tick, " ", time_at(s, tick), " ", bar_at(s, tick))
501 } else if (cmd %in% c("NOTES", "STATS", "ROLL")) {
502 pairs <- pairing(s)
503 notes <- pairs$notes
504 unmatched <- pairs$unmatched
505 if (cmd == "ROLL") {
506 columns <- (args[2] - args[1] + args[3] - 1) %/% args[3]
507 found <- FALSE
508 for (pitch in 127:0) {
509 row <- rep(".", columns)
510 for (i in seq_along(row)) {
511 tick <- args[1] + (i - 1L) * args[3]
512 for (n in notes) {
513 if (n$on$event$data[1] != pitch) next
514 start <- n$on$event$tick
515 if (tick <= start && start < tick + args[3]) {
516 row[i] <- "#"
517 break
518 } else if (start <= tick && tick < start + n$length) {
519 row[i] <- "="
520 }
521 }
522 }
523 if (any(row != ".")) {
524 line(sprintf("%-4s", note_name(pitch)), " ", paste(row, collapse = ""))
525 found <- TRUE
526 }
527 }
528 if (!found) line("empty")
529 return()
530 }
531 if (cmd == "STATS") line(summary_song(s))
532 line("notes ", length(notes), " unmatched ", length(unmatched))
533 if (cmd == "NOTES") {
