65static int range(Integer x, Integer low, Integer high) {

68static int channel(Event e) {
69 return e.status < 240;
70}
71static int ch(Event e) {
72 return (e.status & 15) + 1;
73}
74static int on(Event e) {
75 return channel(e) && e.status >> 4 == 9 && e.data[1] > 0;
76}
77static int off(Event e) {
78 return channel(e) && (e.status >> 4 == 8 || (e.status >> 4 == 9 && e.data[1] == 0));
79}
80static int end_event(Event e) {
81 return e.status == 255 && e.meta == 47;
82}
83static Integer number(Event e) {
84 Integer n = 0;
85 for (size_t i = 0; i < e.length; ++i)
86 n = n * 256 + e.data[i];
87 return n;

97 case 81:

101 case 88:

111static const char *type(Event e) {
112 if (channel(e))
113 return channel_types[(e.status >> 4) - 8];
114 if (e.status == 240)
115 return "SYSEX";
116 if (e.status == 247)
117 return "SYSEX_ESCAPE";
118 if (range(e.meta, 1, 7))
119 return text_types[e.meta - 1];
120 return meta_type(e.meta);
121}
122static void pitch_name(int pitch, char *out) {
123 sprintf(out, "%s%d", pitches[pitch % 12], pitch / 12 - 1);

125static int parse_pitch(const char *text) {
126 char name[8];
127 for (int p = 0; p < 128; ++p) {
128 pitch_name(p, name);
129 if (!strcmp(name, text))
130 return p;
131 }
132 return -1;

134static Event event(Integer tick, int status, int meta, const unsigned char *data, size_t length) {
135 Event e = {.tick = tick, .status = status, .meta = meta, .length = length};
136 e.data = length ? malloc(length) : NULL;
137 if (length)
138 memcpy(e.data, data, length);
139 return e;
140}
141static Event number_event(Integer tick, int meta, Integer value, int length) {
142 unsigned char data[8];
143 for (int i = length - 1; i >= 0; --i) {
144 data[i] = value & 255;
145 value >>= 8;
146 }
147 return event(tick, 255, meta, data, (size_t)length);
148}
149static void append(Track *t, Event e) {
150 if (t->count == t->capacity) {
151 t->capacity = t->capacity ? t->capacity * 2 : 8;
152 t->events = realloc(t->events, t->capacity * sizeof(Event));
153 }
154 t->events[t->count++] = e;
155}
156static Song *new_song(int format, int tracks, int division) {
157 Song *s = calloc(1, sizeof(*s));
158 s->format = format;
159 s->division = division;
160 s->count = tracks;
161 s->tracks = calloc((size_t)tracks, sizeof(Track));

164static void free_tracks(Song *s) {
165 for (size_t tr = 0; tr < s->count; ++tr) {
166 for (size_t i = 0; i < s->tracks[tr].count; ++i)
167 free(s->tracks[tr].events[i].data);
168 free(s->tracks[tr].events);
169 }
170 free(s->tracks);
171}
172static Song *copy_song(const Song *s) {
173 Song *r = new_song(s->format, (int)s->count, s->division);
174 for (size_t tr = 0; tr < s->count; ++tr)
175 for (size_t i = 0; i < s->tracks[tr].count; ++i) {
176 Event e = s->tracks[tr].events[i];
177 append(&r->tracks[tr], event(e.tick, e.status, e.meta, e.data, e.length));
178 }
179 return r;
180}
181static Song *find_song(const char *name) {
182 for (Song *s = songs; s; s = s->next)
183 if (!strcmp(s->name, name))
184 return s;
185 return NULL;

187static void register_song(Song *s, const char *name) {
188 Song *old = find_song(name);
189 if (old) {
190 free_tracks(old);
191 old->format = s->format;
192 old->division = s->division;
193 old->tracks = s->tracks;
194 old->count = s->count;
195 free(s);
196 } else {
197 strcpy(s->name, name);
198 s->next = songs;
199 songs = s;
200 }
201}
202static size_t event_count(const Song *s) {
203 size_t n = 0;
204 for (size_t tr = 0; tr < s->count; ++tr)
205 n += s->tracks[tr].count;
206 return n;
207}
208static Integer song_length(const Song *s) {
209 Integer length = 0;
210 for (size_t tr = 0; tr < s->count; ++tr) {
211 Track t = s->tracks[tr];
212 if (t.events[t.count - 1].tick > length)
213 length = t.events[t.count - 1].tick;
214 }
215 return length;
216}

221static int compare_refs(const void *a, const void *b) {
222 const Ref *x = a, *y = b;
223 if (x->event->tick != y->event->tick)
224 return x->event->tick < y->event->tick ? -1 : 1;
225 if (x->track != y->track)
226 return x->track - y->track;
227 return x->index < y->index ? -1 : x->index > y->index;
228}
229static Ref *global(const Song *s, size_t *count) {
230 *count = event_count(s);
231 Ref *r = malloc(*count * sizeof(Ref));
232 size_t n = 0;
233 for (size_t tr = 0; tr < s->count; ++tr)
234 for (size_t i = 0; i < s->tracks[tr].count; ++i)
235 r[n++] = (Ref){&s->tracks[tr].events[i], (int)tr, i, 0};
236 qsort(r, n, sizeof(Ref), compare_refs);
237 for (size_t i = 0; i < n; ++i)
238 r[i].order = i;
239 return r;
240}

249static void print_event(Event e, int track) {
250 printf("%lld %d %s", (long long)e.tick, track, type(e));
251 if (channel(e)) {
252 int code = e.status >> 4;
253 printf(" %d ", ch(e));
254 if (range(code, 8, 10)) {
255 char name[8];
256 pitch_name(e.data[0], name);
257 printf("%s %d", name, e.data[1]);
258 } else if (code == 11)
259 printf("%d %d", e.data[0], e.data[1]);
260 else if (code == 14)
261 printf("%d", e.data[0] + 128 * e.data[1] - 8192);
262 else
263 printf("%d", e.data[0]);
264 } else if (e.status != 255) {
265 putchar(' ');
266 print_blob(e);
267 } else if (range(e.meta, 1, 7)) {
268 printf(" \"");
269 fwrite(e.data, 1, e.length, stdout);
270 putchar('"');
271 } else
272 switch (e.meta) {
273 case 0:
274 case 81:
275 printf(" %lld", (long long)number(e));
276 break;
277 case 32:
278 printf(" %d", e.data[0] + 1);
279 break;
280 case 47:
281 break;
282 case 84:
283 for (size_t i = 0; i < 5; ++i)
284 printf(" %d", e.data[i]);
285 break;
286 case 88:
287 printf(" %d/%d %d %d", e.data[0], 1 << e.data[1], e.data[2], e.data[3]);
288 break;
289 case 89:
290 printf(" %s %s", keys[e.data[1]][(int8_t)e.data[0] + 7], e.data[1] ? "minor" : "major");
291 break;
292 case 127:
293 putchar(' ');
294 print_blob(e);
295 break;
296 default:
297 printf(" %02X ", e.meta);
298 print_blob(e);
299 break;
300 }
301 putchar('\n');
302}
303static Integer read_big(const unsigned char *data, size_t p, int n) {
304 Integer value = 0;
305 for (int i = 0; i < n; ++i)
306 value = value * 256 + data[p + i];
307 return value;
308}
309static size_t chunk(const unsigned char *data, size_t size, size_t p, const char *id) {
310 if (size - p < 8)
311 fail("byte %zu: truncated chunk", p);
312 if (memcmp(data + p, id, 4))
313 fail("byte %zu: bad chunk id", p);
314 Integer end = (Integer)p + 8 + read_big(data, p + 4, 4);
315 if (end > (Integer)size)
316 fail("byte %zu: truncated chunk", p);
317 return (size_t)end;
318}
319static int read_byte(const unsigned char *data, size_t *p, size_t end, size_t start) {
320 if (*p >= end)
321 fail("byte %zu: truncated event", start);
322 return data[(*p)++];
323}
324static Integer read_var(const unsigned char *data, size_t *p, size_t end, size_t start) {
325 size_t at = *p;
326 Integer value = 0;
327 for (int i = 0; i < 4; ++i) {
328 int b = read_byte(data, p, end, start);
329 value = value * 128 + (b & 127);
330 if (b < 128)
331 return value;
332 }
333 fail("byte %zu: variable-length value too long", at);
334 return 0;
335}
336static void validate_meta(Event e, size_t position) {
337 int length = -1;
338 switch (e.meta) {
339 case 0:
340 length = 2;
341 break;
342 case 32:
343 length = 1;
344 break;
345 case 47:
346 length = 0;
347 break;
348 case 81:
349 length = 3;
350 break;
351 case 84:
352 length = 5;
353 break;
354 case 88:
355 length = 4;
356 break;
357 case 89:
358 length = 2;
359 break;
360 }
361 if (length >= 0 && e.length != (size_t)length)
362 fail("byte %zu: bad meta length", position);
363 unsigned char *d = e.data;
364 if ((e.meta == 81 && !number(e)) || (e.meta == 32 && d[0] > 15) ||
365 (e.meta == 88 && (!d[0] || d[1] > 6)) ||
366 (e.meta == 89 && (!range((int8_t)d[0], -7, 7) || d[1] > 1)))
367 fail("byte %zu: bad meta data", position);
368}
369static Song *parse_hex(const unsigned char *data, size_t size) {
370 size_t h = chunk(data, size, 0, "MThd");
371 if (h != 14)
372 fail("byte 4: bad header length");
373 int f = (int)read_big(data, 8, 2), tracks = (int)read_big(data, 10, 2),
374 division = (int)read_big(data, 12, 2);
375 if (f > 2)
376 fail("byte 8: bad format");
377 if (!tracks || (!f && tracks != 1))
378 fail("byte 10: bad track count");
379 if (division & 32768)
380 fail("byte 12: smpte timing not supported");
381 if (!division)
382 fail("byte 12: bad division");
383 Song *s = new_song(f, tracks, division);
384 loading_song = s;
385 size_t p = 14;
386 for (int tr = 0; tr < tracks; ++tr) {
387 if (p == size)
388 fail("byte %zu: missing track", p);
389 size_t end = chunk(data, size, p, "MTrk");
390 p += 8;
391 Integer tick = 0;
392 int running_status = 0;
393 while (p < end) {
394 Track *t = &s->tracks[tr];
395 if (t->count && end_event(t->events[t->count - 1]))
396 fail("byte %zu: event after end of track", p);
397 size_t start = p;
398 tick += read_var(data, &p, end, start);
399 if (p >= end)
400 fail("byte %zu: truncated event", start);
401 size_t position = p;
402 int status = data[p];
403 if (status < 128) {
404 if (!running_status)
405 fail("byte %zu: data byte without running status", p);
406 status = running_status;
407 } else
408 ++p;
409 Event e;
410 if (status < 240) {
411 running_status = status;
412 int code = status >> 4, length = code == 12 || code == 13 ? 1 : 2;
413 unsigned char d[2];
414 for (int i = 0; i < length; ++i) {
415 size_t at = p;
416 int b = read_byte(data, &p, end, start);
417 if (b >= 128)
418 fail("byte %zu: bad data byte", at);
419 d[i] = (unsigned char)b;
420 }
421 e = event(tick, status, -1, d, (size_t)length);
422 } else if (status == 240 || status == 247 || status == 255) {
423 running_status = 0;
424 int meta = status == 255 ? read_byte(data, &p, end, start) : -1;
425 size_t length = (size_t)read_var(data, &p, end, start);
426 if (length > end - p)
427 fail("byte %zu: truncated event", start);
428 if (status == 255) {
429 Event view = {.tick = tick,
430 .status = status,
431 .meta = meta,
432 .data = (unsigned char *)data + p,
433 .length = length};
434 validate_meta(view, position);
435 }
436 e = event(tick, status, meta, data + p, length);
437 p += length;
438 } else {
439 fail("byte %zu: unknown status byte", position);
440 return s;
441 }
442 append(&s->tracks[tr], e);
443 }
444 Track *t = &s->tracks[tr];
445 if (!t->count || !end_event(t->events[t->count - 1]))
446 fail("byte %zu: missing end of track", end);
447 }
448 if (p < size)
449 fail("byte %zu: extra data", p);
450 return s;
451}
452static int hex_digit(char c) {
453 return c >= '0' && c <= '9' ? c - '0' : c >= 'A' && c <= 'F' ? c - 'A' + 10 : -1;
454}
455static Integer text_num(char **a, int count, int i, Integer low, Integer high, int line) {
456 if (i >= count)
457 fail("line %d: bad value", line);
458 char *end;
459 Integer n = strtoll(a[i], &end, 10);
460 if (*end || !range(n, low, high))
461 fail("line %d: bad value", line);
462 return n;
463}
464static Event text_event(Integer tick, const char *kind, char *fields, int line) {
465 int text = -1, code = -1, meta = -1;
466 for (int i = 0; i < 7; ++i) {
467 if (!strcmp(kind, text_types[i]))
468 text = i;
469 if (!strcmp(kind, channel_types[i]))
470 code = i + 8;
471 }
472 if (!strcmp(kind, "META"))
473 meta = -2;
474 for (int i = 0; i < 256; ++i)
475 if (strcmp(meta_type(i), "META") && !strcmp(kind, meta_type(i)))
476 meta = i;
477 int sysex = !strcmp(kind, "SYSEX") ? 240 : !strcmp(kind, "SYSEX_ESCAPE") ? 247 : 0;
478 if (text < 0 && code < 0 && meta == -1 && !sysex)
479 fail("line %d: unknown event type", line);
480 if (text >= 0) {
481 size_t n = strlen(fields);
482 if (n < 2 || fields[0] != '"' || fields[n - 1] != '"')
483 fail("line %d: bad value", line);
484 for (size_t i = 1; i + 1 < n; ++i)
485 if (fields[i] == '"')
486 fail("line %d: bad value", line);
487 return event(tick, 255, text + 1, (unsigned char *)fields + 1, n - 2);
488 }
489 char *a[1002];
490 int count = 0;
491 for (char *p = strtok(fields, " "); p; p = strtok(NULL, " "))
492 a[count++] = p;
493 unsigned char d[8];
494 size_t length = 0;
495 if (code >= 0) {
496 int required = code == 12 || code == 13 || code == 14 ? 2 : 3;
497 if (count != required)
498 fail("line %d: bad value", line);
499 int channel = (int)text_num(a, count, 0, 1, 16, line);
500 if (code <= 10) {
501 int pitch = parse_pitch(a[1]);
502 if (pitch < 0)
503 fail("line %d: bad value", line);
504 d[0] = (unsigned char)pitch;
505 d[1] = (unsigned char)text_num(a, count, 2, 0, 127, line);
506 length = 2;
507 } else if (code == 14) {
508 int value = (int)text_num(a, count, 1, -8192, 8191, line) + 8192;
509 d[0] = value & 127;
510 d[1] = value >> 7;
511 length = 2;
512 } else {
513 d[0] = (unsigned char)text_num(a, count, 1, 0, 127, line);
514 length = 1;
515 if (code == 11) {
516 d[1] = (unsigned char)text_num(a, count, 2, 0, 127, line);
517 length = 2;
518 }
519 }
520 return event(tick, (code << 4) + channel - 1, -1, d, length);
521 }
522 if (sysex || meta == 127 || meta == -2) {
523 int index = meta == -2 ? 1 : 0;
524 if (meta == -2) {
525 if (!count || strlen(a[0]) != 2 || hex_digit(a[0][0]) < 0 || hex_digit(a[0][1]) < 0)
526 fail("line %d: bad value", line);
527 meta = hex_digit(a[0][0]) * 16 + hex_digit(a[0][1]);
528 if (strcmp(meta_type(meta), "META") || range(meta, 1, 7))
529 fail("line %d: bad value", line);
530 }
531 length = (size_t)text_num(a, count, index, 0, 1000000, line);
532 if (count != index + (length ? 2 : 1) || (length && strlen(a[index + 1]) != 2 * length))
533 fail("line %d: bad value", line);
534 for (size_t i = 0; i < length; ++i) {
535 int high = hex_digit(a[index + 1][2 * i]), low = hex_digit(a[index + 1][2 * i + 1]);
536 if (high < 0 || low < 0)
537 fail("line %d: bad value", line);
538 }
539 unsigned char *blob = length ? malloc(length) : NULL;
540 for (size_t i = 0; i < length; ++i)
541 blob[i] = hex_digit(a[index + 1][2 * i]) * 16 + hex_digit(a[index + 1][2 * i + 1]);
542 Event e = event(tick, sysex ? sysex : 255, meta, blob, length);
543 free(blob);
544 return e;
545 }
546 int required = meta == 47 ? 0 : meta == 84 ? 5 : meta == 88 ? 3 : meta == 89 ? 2 : 1;
547 if (count != required)
548 fail("line %d: bad value", line);
549 if (meta == 0 || meta == 81)
550 return number_event(
551 tick, meta,
552 text_num(a, count, 0, meta == 81 ? 1 : 0, meta == 81 ? 16777215 : 65535, line),
553 meta == 81 ? 3 : 2);
554 if (meta == 32) {
555 d[0] = (unsigned char)(text_num(a, count, 0, 1, 16, line) - 1);
556 length = 1;
557 }
558 if (meta == 84) {
559 length = 5;
560 for (int i = 0; i < 5; ++i)
561 d[i] = (unsigned char)text_num(a, count, i, 0, 255, line);
562 }
563 if (meta == 88) {
564 char *slash = strchr(a[0], '/');
565 if (!slash)
566 fail("line %d: bad value", line);
567 *slash = 0;
568 Integer nn = strtoll(a[0], NULL, 10), dd = strtoll(slash + 1, NULL, 10);
569 int exponent = -1;
570 for (int i = 0; i <= 6; ++i)
571 if ((1 << i) == dd)
572 exponent = i;
573 if (!range(nn, 1, 255) || exponent < 0)
574 fail("line %d: bad value", line);
575 d[0] = (unsigned char)nn;
576 d[1] = (unsigned char)exponent;
577 d[2] = (unsigned char)text_num(a, count, 1, 0, 255, line);
578 d[3] = (unsigned char)text_num(a, count, 2, 0, 255, line);
579 length = 4;
580 }
581 if (meta == 89) {
582 int minor = !strcmp(a[1], "minor"), key = -1;
583 if (strcmp(a[1], "major") && !minor)
584 fail("line %d: bad value", line);
585 for (int i = 0; i < 15; ++i)
586 if (!strcmp(a[0], keys[minor][i]))
587 key = i;
588 if (key < 0)
589 fail("line %d: bad value", line);
590 d[0] = (unsigned char)(key - 7);
591 d[1] = (unsigned char)minor;
592 length = 2;
593 }
594 return event(tick, 255, meta, d, length);
595}
596static void reject_text_event(Event e, int line, const char *reason) {
597 free(e.data);
598 fail("line %d: %s", line, reason);

600static Song *parse_text(char **lines, int count) {
601 char h[1002], extra;
602 long long f, tracks, division;
603 if (sscanf(lines[0], "%1001s %lld %lld %lld %c", h, &f, &tracks, &division, &extra) != 4 ||
604 strcmp(h, "HEADER") || !range(f, 0, 2) || !range(tracks, 1, 65535) ||
605 !range(division, 1, 32767) || (!f && tracks != 1))
606 fail("line 1: bad header");
607 Song *s = new_song((int)f, (int)tracks, (int)division);
608 loading_song = s;
609 int current = 0;
610 for (int i = 1; i < count; ++i) {
611 char *p = lines[i], *tail;
612 Integer tick = strtoll(p, &tail, 10);
613 p = tail + 1;
614 int tr = (int)strtol(p, &tail, 10);
615 p = tail + 1;
616 char *fields = strchr(p, ' ');
617 if (fields)
618 *fields++ = 0;
619 else
620 fields = p + strlen(p);
621 Event e = text_event(tick, p, fields, i + 1);
622 if (tr >= tracks)
623 reject_text_event(e, i + 1, "bad track");
624 Track *t = &s->tracks[current];
625 int ended = t->count && end_event(t->events[t->count - 1]);
626 if (tr < current)
627 reject_text_event(e, i + 1, "out of order");
628 if (tr > current && (!ended || tr > current + 1))
629 reject_text_event(e, i + 1, "missing end of track");
630 if (tr == current && ended)
631 reject_text_event(e, i + 1, "event after end of track");
632 if (tr == current + 1)
633 t = &s->tracks[++current];
634 if (t->count && tick < t->events[t->count - 1].tick)
635 reject_text_event(e, i + 1, "out of order");
636 append(t, e);
637 }
638 Track t = s->tracks[current];
639 if (current + 1 < tracks || !t.count || !end_event(t.events[t.count - 1]))
640 fail("line %d: missing end of track", count + 1);
641 return s;
642}
643static void byte(Buffer *b, int value) {
644 if (b->count == b->capacity) {
645 b->capacity = b->capacity ? b->capacity * 2 : 64;
646 b->data = realloc(b->data, b->capacity);
647 }
648 b->data[b->count++] = (unsigned char)value;
649}
650static void big(Buffer *b, Integer value, int n) {
651 for (int i = n - 1; i >= 0; --i)
652 byte(b, (int)(value >> (8 * i)) & 255);
653}
654static void bytes(Buffer *b, const unsigned char *data, size_t n) {
655 for (size_t i = 0; i < n; ++i)
656 byte(b, data[i]);
657}
658static void var(Buffer *b, Integer value) {
659 unsigned char data[10];
660 int n = 0;
661 data[n++] = value & 127;
662 while ((value >>= 7) > 0)
663 data[n++] = (value & 127) | 128;
664 while (n)
665 byte(b, data[--n]);
666}
667static void encode(Song *s) {
668 Buffer all = {0};
669 bytes(&all, (unsigned char *)"MThd", 4);
670 big(&all, 6, 4);
671 big(&all, s->format, 2);
672 big(&all, (Integer)s->count, 2);
673 big(&all, s->division, 2);
674 for (size_t tr = 0; tr < s->count; ++tr) {
675 Buffer data = {0};
676 Integer previous = 0;
677 int last = -1;
678 for (size_t i = 0; i < s->tracks[tr].count; ++i) {
679 Event e = s->tracks[tr].events[i];
680 var(&data, e.tick - previous);
681 previous = e.tick;
682 if (channel(e)) {
683 int status = e.status, rewrite = zero_off && e.status >> 4 == 8;
684 if (rewrite)
685 status = 0x90 + ch(e) - 1;
686 if (!running || status != last)
687 byte(&data, status);
688 byte(&data, e.data[0]);
689 if (e.length > 1)
690 byte(&data, rewrite ? 0 : e.data[1]);
691 last = status;
692 } else {
693 last = -1;
694 byte(&data, e.status);
695 if (e.status == 255)
696 byte(&data, e.meta);
697 var(&data, (Integer)e.length);
698 bytes(&data, e.data, e.length);
699 }
700 }
701 bytes(&all, (unsigned char *)"MTrk", 4);
702 big(&all, (Integer)data.count, 4);
703 bytes(&all, data.data, data.count);
704 free(data.data);
705 }
706 for (size_t i = 0; i < all.count; i += 16) {
707 printf("%04zX:", i);
708 for (size_t j = i; j < all.count && j < i + 16; ++j)
709 printf(" %02X", all.data[j]);
710 putchar('\n');
711 }
712 printf("%zu bytes\n", all.count);
713 free(all.data);
714}
715static int compare_notes(const void *a, const void *b) {
716 const Note *x = a, *y = b;
717 Event e = *x->on.event, f = *y->on.event;
718 if (e.tick != f.tick)
719 return e.tick < f.tick ? -1 : 1;
720 if (ch(e) != ch(f))
721 return ch(e) - ch(f);
722 if (e.data[0] != f.data[0])
723 return e.data[0] - f.data[0];
724 return x->on.order < y->on.order ? -1 : x->on.order > y->on.order;
725}
726static Pairing pair(Song *s) {
727 size_t n;
728 Ref *all = global(s, &n);
729 unsigned char *matched = calloc(n, 1);
730 Pairing p = {0};
731 p.notes = malloc(n * sizeof(Note));
732 p.unmatched = malloc(n * sizeof(Ref));
733 for (size_t i = 0; i < n; ++i)
734 if (off(*all[i].event))
735 for (size_t j = 0; j < i; ++j)
736 if (!matched[j] && on(*all[j].event) && ch(*all[j].event) == ch(*all[i].event) &&
737 all[j].event->data[0] == all[i].event->data[0]) {
738 matched[i] = matched[j] = 1;
739 p.notes[p.count++] = (Note){all[j], all[i]};
740 break;
741 }
742 for (size_t i = 0; i < n; ++i)
743 if (!matched[i] && (on(*all[i].event) || off(*all[i].event)))
744 p.unmatched[p.unmatched_count++] = all[i];
745 qsort(p.notes, p.count, sizeof(Note), compare_notes);
746 free(matched);
747 free(all);
748 return p;
749}
750static void free_pair(Pairing p) {
751 free(p.notes);
752 free(p.unmatched);
753}
754static void position(Song *s, Integer tick, char *time, char *bar_text) {
755 size_t n;
756 Ref *all = global(s, &n);
757 Integer sum = 0, last = 0, tempo = 500000, start = 0, bar = 1, nn = 4, beat = s->division;
758 for (size_t i = 0; i < n && all[i].event->tick <= tick; ++i) {
759 Event e = *all[i].event;
760 if (e.status != 255)
761 continue;
762 if (e.meta == 81) {
763 sum += (e.tick - last) * tempo;
764 last = e.tick;
765 tempo = number(e);
766 }
767 if (e.meta == 88 && bar_text) {
768 Integer size = nn * beat;
769 bar += (e.tick - start + size - 1) / size;
770 start = e.tick;
771 nn = e.data[0];
772 beat = s->division * 4 / (1 << e.data[1]);
773 }
774 }
775 Integer micro = (sum + (tick - last) * tempo) / s->division, x = tick - start;
776 sprintf(time, "%lld.%03lld", (long long)(micro / 1000), (long long)(micro % 1000));
777 if (bar_text)
778 sprintf(bar_text, "%lld:%lld:%lld", (long long)(bar + x / (nn * beat)),
779 (long long)(x % (nn * beat) / beat + 1), (long long)(x % beat));
780 free(all);
781}
782static void notes(Song *s) {
783 Pairing p = pair(s);
784 printf("notes %zu unmatched %zu\n", p.count, p.unmatched_count);
785 for (size_t i = 0; i < p.count; ++i) {
786 Event a = *p.notes[i].on.event, b = *p.notes[i].off.event;
787 char pitch[8];
788 pitch_name(a.data[0], pitch);
789 printf("%lld %lld %d %s %d\n", (long long)a.tick, (long long)(b.tick - a.tick), ch(a),
790 pitch, a.data[1]);
791 }
792 for (size_t i = 0; i < p.unmatched_count; ++i) {
793 Ref r = p.unmatched[i];
794 Event e = *r.event;
795 char pitch[8];
796 pitch_name(e.data[0], pitch);
797 printf("unmatched %s %lld %d %d %s\n", on(e) ? "on" : "off", (long long)e.tick, r.track,
798 ch(e), pitch);
799 }
800 free_pair(p);
801}
802static Integer floor_div(Integer n, Integer d) {
803 return n >= 0 ? n / d : -((-n + d - 1) / d);
804}
805static int compare_tick(const void *a, const void *b) {
806 const Ref *x = a, *y = b;
807 if (x->event->tick != y->event->tick)
808 return x->event->tick < y->event->tick ? -1 : 1;
809 return x->index < y->index ? -1 : x->index > y->index;
810}
811static void quantize(Song *s, Integer grid, Integer strength) {
812 Pairing p = pair(s);
813 size_t targets = 0, moved = 0;
814 for (size_t i = 0; i < p.count + p.unmatched_count; ++i) {
815 Event *a, *b = NULL;
816 if (i < p.count) {
817 a = p.notes[i].on.event;
818 b = p.notes[i].off.event;
819 } else {
820 a = p.unmatched[i - p.count].event;
821 if (!on(*a))
822 continue;
823 }
824 Integer lo = a->tick / grid * grid,
825 g = a->tick - lo <= lo + grid - a->tick ? lo : lo + grid;
826 Integer delta = floor_div(2 * (g - a->tick) * strength + 100, 200);
827 ++targets;
828 moved += delta != 0;
829 a->tick += delta;
830 if (b)
831 b->tick += delta;
832 }
833 free_pair(p);
834 for (size_t tr = 0; tr < s->count; ++tr) {
835 Track *t = &s->tracks[tr];
836 Event eot = t->events[t->count - 1];
837 size_t n = t->count - 1;
838 Ref *refs = malloc(n * sizeof(Ref));
839 for (size_t i = 0; i < n; ++i)
840 refs[i] = (Ref){&t->events[i], (int)tr, i, 0};
841 qsort(refs, n, sizeof(Ref), compare_tick);
842 Event *events = malloc(t->capacity * sizeof(Event));
843 for (size_t i = 0; i < n; ++i)
844 events[i] = *refs[i].event;
845 if (n && events[n - 1].tick > eot.tick)
846 eot.tick = events[n - 1].tick;
847 events[n] = eot;
848 free(refs);
849 free(t->events);
850 t->events = events;
851 }
852 printf("quantized %zu moved %zu\n", targets, moved);
853}
854static int compare_song_names(const void *a, const void *b) {
855 return strcmp((*(Song *const *)a)->name, (*(Song *const *)b)->name);
856}
857static void stats(Song *s) {
858 Pairing p = pair(s);
859 summary(s);
860 putchar('\n');
861 printf("notes %zu unmatched %zu\n", p.count, p.unmatched_count);
862 int minimum = 128, maximum = -1;
863 for (int c = 1; c <= 16; ++c) {
864 int low = 128, high = -1;
865 size_t count = 0;
866 for (size_t i = 0; i < p.count; ++i)
867 if (ch(*p.notes[i].on.event) == c) {
868 int pitch = p.notes[i].on.event->data[0];
869 if (pitch < low)
870 low = pitch;
871 if (pitch > high)
872 high = pitch;
873 ++count;
874 }
875 if (count) {
876 char a[8], b[8];
877 pitch_name(low, a);
878 pitch_name(high, b);
879 printf("channel %d notes %zu range %s %s\n", c, count, a, b);
880 }
881 if (low < minimum)
882 minimum = low;
883 if (high > maximum)
884 maximum = high;
885 }
886 if (!p.count)
887 puts("range none");
888 else {
889 char a[8], b[8];
890 pitch_name(minimum, a);
891 pitch_name(maximum, b);
892 printf("range %s %s\n", a, b);
893 }
894 char time[64];
895 Integer length = song_length(s);
896 position(s, length, time, NULL);
897 printf("length %lld ticks %s ms\n", (long long)length, time);
898 size_t best = 0;
899 Integer best_at = 0;
900 for (size_t i = 0; i < p.count; ++i) {
901 Integer at = p.notes[i].on.event->tick;
902 size_t current = 0;
903 for (size_t j = 0; j < p.count; ++j)
904 if (p.notes[j].on.event->tick <= at && at < p.notes[j].off.event->tick)
905 ++current;
906 if (current > best || (current == best && at < best_at)) {
907 best = current;
908 best_at = at;
909 }
910 }
911 if (best)
912 printf("polyphony %zu at %lld\n", best, (long long)best_at);
913 else
914 puts("polyphony 0");
915 free_pair(p);
916}
917static void execute(char **a, int argc) {
918 const char *cmd = a[0];
919 const char *known[] = {"DECODE", "ENCODE", "TIMING", "NOTES", "STATS", "ROLL",
920 "TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "FILTER", "MERGE",
921 "SPLIT", "COPY", "LIST", "RUNNING", "NOTEOFF"};
922 int found = 0;
923 for (size_t i = 0; i < sizeof(known) / sizeof(*known); ++i)
924 if (!strcmp(cmd, known[i]))
925 found = 1;
926 if (!found)
927 fail("unknown command");
928 if (!strcmp(cmd, "LIST")) {
929 size_t n = 0;
930 for (Song *s = songs; s; s = s->next)
931 ++n;

935 }

937 size_t i = 0;

939 sorted[i++] = s;

948 }

950 if (!strcmp(cmd, "RUNNING")) {

952 fail("bad argument");
953 running = !strcmp(a[1], "ON");
954 printf("running %s\n", running ? "on" : "off");
955 } else {
956 if (strcmp(a[1], "ZERO") && strcmp(a[1], "KEEP"))
957 fail("bad argument");
958 zero_off = !strcmp(a[1], "ZERO");
959 printf("noteoff %s\n", zero_off ? "zero" : "keep");
960 }
961 return;
962 }
963 Song *s = find_song(a[1]);
964 if (!s)
965 fail("no song %s", a[1]);

967 second = argc > 3 ? strtoll(a[3], NULL, 10) : 0,
968 third = argc > 4 ? strtoll(a[4], NULL, 10) : 0;
969 if (!strcmp(cmd, "TRANSPOSE") &&
970 (!range(first, -127, 127) || (argc == 4 && !range(second, 1, 16))))
971 fail("bad argument");
972 if (!strcmp(cmd, "QUANTIZE") && (!range(first, 1, 1000000) || !range(second, 0, 100)))
973 fail("bad argument");
974 if (!strcmp(cmd, "VELOCITY") && (!range(first, 0, 1000) || !range(second, -127, 127)))
975 fail("bad argument");
976 if (!strcmp(cmd, "TEMPO") && (!range(first, 1, 1000) || !range(second, 1, 1000)))
977 fail("bad argument");
978 if (!strcmp(cmd, "FILTER"))
979 for (int i = 2; i < argc; ++i)
980 if (!range(strtoll(a[i], NULL, 10), 1, 16))
981 fail("bad argument");
982 if (!strcmp(cmd, "ROLL") &&
983 (!range(first, 0, 10000000) || !range(second, 1, 10000000) || !range(third, 1, 10000000) ||
984 second <= first || (second - first + third - 1) / third > 64))
985 fail("bad argument");
986 if ((!strcmp(cmd, "TIMING") || !strcmp(cmd, "STATS")) && s->format == 2)
987 fail("format 2 not supported");
988 if (!strcmp(cmd, "DECODE")) {
989 printf("HEADER %d %zu %d\n", s->format, s->count, s->division);
990 for (size_t tr = 0; tr < s->count; ++tr)
991 for (size_t i = 0; i < s->tracks[tr].count; ++i)
992 print_event(s->tracks[tr].events[i], (int)tr);
993 } else if (!strcmp(cmd, "ENCODE"))
994 encode(s);
995 else if (!strcmp(cmd, "NOTES"))
996 notes(s);
997 else if (!strcmp(cmd, "STATS"))
998 stats(s);
999 else if (!strcmp(cmd, "TIMING")) {
1000 size_t count;
1001 Ref *all = global(s, &count);
1002 char time[64], bar[64];
1003 for (size_t i = 0; i < count; ++i) {
1004 Event e = *all[i].event;
1005 position(s, e.tick, time, bar);
1006 printf("%lld %d %s %s %s\n", (long long)e.tick, all[i].track, type(e), time, bar);
1007 }
1008 Integer length = song_length(s);
1009 position(s, length, time, bar);
1010 printf("end %lld %s %s\n", (long long)length, time, bar);
1011 free(all);
1012 } else if (!strcmp(cmd, "COPY")) {
1013 Song *copy = copy_song(s);
1014 register_song(copy, a[2]);
1015 printf("copied %s to %s\n", a[1], a[2]);
1016 } else if (!strcmp(cmd, "TRANSPOSE")) {
1017 size_t kept = 0, dropped = 0;
1018 Track removed = {0};
1019 int *removed_tracks = malloc(event_count(s) * sizeof(int));
1020 for (size_t tr = 0; tr < s->count; ++tr) {
1021 Track *t = &s->tracks[tr];
1022 size_t write = 0;
1023 for (size_t i = 0; i < t->count; ++i) {
1024 Event e = t->events[i];
1025 if (channel(e) && range(e.status >> 4, 8, 10) && (argc == 3 || ch(e) == second)) {
1026 Integer value = e.data[0] + first;
1027 if (!range(value, 0, 127)) {
1028 removed_tracks[dropped++] = (int)tr;
1029 append(&removed, e);
1030 continue;
1031 }
1032 e.data[0] = (unsigned char)value;
1033 ++kept;
1034 }
1035 t->events[write++] = e;
1036 }
1037 t->count = write;
1038 }
1039 printf("transposed %zu dropped %zu\n", kept, dropped);
1040 for (size_t i = 0; i < dropped; ++i) {
1041 Event e = removed.events[i];
1042 char pitch[8];
1043 pitch_name(e.data[0], pitch);
1044 printf("dropped %lld %d %s %d %s\n", (long long)e.tick, removed_tracks[i], type(e),
1045 ch(e), pitch);
1046 free(e.data);
1047 }
1048 free(removed.events);
1049 free(removed_tracks);
1050 } else if (!strcmp(cmd, "QUANTIZE"))
1051 quantize(s, first, second);
1052 else if (!strcmp(cmd, "VELOCITY")) {
1053 size_t count = 0, clamped = 0;
1054 for (size_t tr = 0; tr < s->count; ++tr)
1055 for (size_t i = 0; i < s->tracks[tr].count; ++i) {
1056 Event *e = &s->tracks[tr].events[i];
1057 if (!on(*e))
1058 continue;
1059 Integer value = (e->data[1] * first + 50) / 100 + second;
1060 ++count;
1061 clamped += !range(value, 1, 127);
1062 e->data[1] = (unsigned char)(value < 1 ? 1 : value > 127 ? 127 : value);
1063 }
1064 printf("velocity %zu clamped %zu\n", count, clamped);
1065 } else if (!strcmp(cmd, "TEMPO")) {
1066 Song *candidate = copy_song(s);
1067 int at_zero = 0;
1068 size_t scaled = 0;
1069 for (size_t tr = 0; tr < candidate->count; ++tr)
1070 for (size_t i = 0; i < candidate->tracks[tr].count; ++i) {
1071 Event *e = &candidate->tracks[tr].events[i];
1072 if (e->status != 255 || e->meta != 81)
1073 continue;
1074 Integer value = (2 * number(*e) * second + first) / (2 * first);
1075 if (!range(value, 1, 16777215)) {
1076 free_tracks(candidate);
1077 free(candidate);
1078 fail("tempo out of range");
1079 }
1080 if (!e->tick)
1081 at_zero = 1;
1082 ++scaled;
1083 free(e->data);
1084 *e = number_event(e->tick, 81, value, 3);
1085 }
1086 Integer inserted = (2 * 500000 * second + first) / (2 * first);
1087 if (!at_zero) {
1088 if (!range(inserted, 1, 16777215)) {
1089 free_tracks(candidate);
1090 free(candidate);
1091 fail("tempo out of range");
1092 }
1093 Track *t = &candidate->tracks[0];
1094 append(t, number_event(0, 81, inserted, 3));
1095 Event e = t->events[t->count - 1];
1096 memmove(t->events + 1, t->events, (t->count - 1) * sizeof(Event));
1097 t->events[0] = e;
1098 }
1099 register_song(candidate, a[1]);
1100 printf("scaled %zu\n", scaled);
1101 if (!at_zero)
1102 printf("inserted 0 0 TEMPO %lld\n", (long long)inserted);
1103 } else if (!strcmp(cmd, "FILTER")) {
1104 int allowed[17] = {0};
1105 for (int i = 2; i < argc; ++i)
1106 allowed[atoi(a[i])] = 1;
1107 size_t removed = 0;
1108 for (size_t tr = 0; tr < s->count; ++tr) {
1109 Track *t = &s->tracks[tr];
1110 size_t write = 0;
1111 for (size_t i = 0; i < t->count; ++i) {
1112 Event e = t->events[i];
1113 if (channel(e) && !allowed[ch(e)]) {
1114 ++removed;
1115 free(e.data);
1116 } else
1117 t->events[write++] = e;
1118 }
1119 t->count = write;
1120 }
1121 printf("removed %zu\n", removed);
1122 } else if (!strcmp(cmd, "MERGE")) {
1123 if (s->format != 1)
1124 fail("not format 1");
1125 Integer length = song_length(s);
1126 size_t n, old = s->count;
1127 Ref *all = global(s, &n);
1128 Track merged = {0};
1129 for (size_t i = 0; i < n; ++i) {
1130 Event e = *all[i].event;
1131 if (!end_event(e))

1134 free(e.data);

1144 s->format = 0;
1145 printf("merged %zu tracks events %zu\n", old, event_count(s));
1146 } else if (!strcmp(cmd, "SPLIT")) {
1147 if (s->format != 0)
1148 fail("not format 0");
1149 Integer length = song_length(s);
1150 Track original = s->tracks[0];

1161 Event e = original.events[i];

1172 s->tracks = tracks;
1173 s->count = count;
1174 s->format = 1;
1175 printf("split tracks %zu events %zu\n", count, event_count(s));
1176 } else if (!strcmp(cmd, "ROLL")) {
1177 Pairing p = pair(s);
1178 int any = 0;
1179 for (int pitch = 127; pitch >= 0; --pitch) {
1180 char row[65];

1182 int visible = 0;

1190 cell = '#';

1192 }

1198 }
1199 row[width] = 0;
1200 if (visible) {
1201 char name[8];
1202 pitch_name(pitch, name);
1203 printf("%-4s %s\n", name, row);
1204 any = 1;
1205 }
1206 }
1207 if (!any)
1208 puts("empty");
1209 free_pair(p);
