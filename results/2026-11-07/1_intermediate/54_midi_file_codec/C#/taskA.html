6record Event(long Tick, int Status, int Meta, byte[] Data) {
7 public bool Channel => Status < 0xF0;
8 public int Ch => (Status & 15) + 1;
9 public int Code => Status >> 4;
10 public bool On => Channel && Code == 9 && Data[1] > 0;
11 public bool Off => Channel && (Code == 8 || Code == 9 && Data[1] == 0);
12 public bool End => Status == 255 && Meta == 47;
13 public long Number => Data.Aggregate(0L, (a, b) => a * 256 + b);
14 public Event Copy() => this with { Data = (byte[])Data.Clone() };
15}
16class Song {
17 public int Format, Division;
18 public List<List<Event>> Tracks = new();
19 public int Count => Tracks.Sum(t => t.Count);
20 public long Length => Tracks.Max(t => t.Last().Tick);
21 public Song
22 Copy() => new() { Format = Format, Division = Division,
23 Tracks = Tracks.Select(t => t.Select(e => e.Copy()).ToList()).ToList() };
24 public List<(Event E, int Track, int Index)>
25 Global() => Tracks.SelectMany((t, tr) => t.Select((e, i) => (E: e, Track: tr, Index: i)))
26 .OrderBy(x => x.E.Tick)
27 .ThenBy(x => x.Track)
28 .ThenBy(x => x.Index)
29 .ToList();
30}

56 static byte[] Bytes(long value, int n) =>
57 Enumerable.Range(0, n).Select(i => (byte)(value >> (8 * (n - 1 - i)) & 255)).ToArray();
58 static string Pitch(int p) => pitchNames[p % 12] + (p / 12 - 1);

60 ParsePitch(string text) => Enumerable.Range(0, 128).FirstOrDefault(p => Pitch(p) == text, -1);
61 static string Type(Event e) => e.Channel ? channelTypes[e.Code - 8]
62 : e.Status == 240 ? "SYSEX"
63 : e.Status == 247 ? "SYSEX_ESCAPE"
64 : Range(e.Meta, 1, 7)
65 ? textTypes[e.Meta - 1]
66 : metaTypes.GetValueOrDefault(e.Meta, "META");
67 static string DataText(byte[] data) => data.Length +
68 (data.Length > 0 ? " " + Convert.ToHexString(data) : "");
69 static string Fields(Event e) {
70 byte[] d = e.Data;
71 if (e.Channel) {
72 string value = e.Code is 8 or 9 or 10 ? Pitch(d[0]) + " " + d[1]
73 : e.Code == 11 ? d[0] + " " + d[1]
74 : e.Code == 14 ? (d[0] + 128 * d[1] - 8192).ToString()
75 : d[0].ToString();
76 return e.Ch + " " + value;
77 }
78 if (e.Status != 255)
79 return DataText(d);
80 if (Range(e.Meta, 1, 7))
81 return "\"" + Encoding.ASCII.GetString(d) + "\"";
82 return e.Meta switch { 0 or
83 81 => e.Number.ToString(),
84 32 => (d[0] + 1).ToString(),
85 47 => "",
86 84 => string.Join(" ", d),
87 88 => d[0] + "/" + (1 << d[1]) + " " + d[2] + " " + d[3],
88 89 =>
89 keys[d[1]][(sbyte)d[0] + 7] + (d[1] == 0 ? " major" : " minor"),
90 127 => DataText(d),
91 _ => e.Meta.ToString("X2") + " " + DataText(d) };
92 }
93 static string Line(Event e, int track) => e.Tick + " " + track + " " + Type(e) +
94 (Fields(e) == "" ? "" : " " + Fields(e));
95 static void ValidateMeta(int meta, byte[] d, int position) {
96 int length = meta switch {
97 0 => 2,
98 32 => 1,
99 47 => 0,
100 81 => 3,
101 84 => 5,
102 88 => 4,
103 89 => 2,
104 _ => -1
105 };
106 if (length >= 0 && d.Length != length)
107 Fail("byte " + position + ": bad meta length");
108 bool bad = meta == 81 && d.All(x => x == 0) || meta == 32 && d[0] > 15 ||
109 meta == 88 && (d[0] == 0 || d[1] > 6) ||
110 meta == 89 && ((sbyte)d[0] < -7 || (sbyte)d[0] > 7 || d[1] > 1);
111 if (bad)
112 Fail("byte " + position + ": bad meta data");
113 }
114 static Song ParseHex(byte[] data) {
115 long ReadBig(int p, int n) {
116 long v = 0;
117 for (int i = 0; i < n; i++)
118 v = v * 256 + data[p + i];
119 return v;
120 }
121 int Chunk(int p, string id) {
122 if (data.Length - p < 8) {
123 Fail("byte " + p + ": truncated chunk");
124 }
125 if (Encoding.ASCII.GetString(data, p, 4) != id)
126 Fail("byte " + p + ": bad chunk id");
127 long end = p + 8L + ReadBig(p + 4, 4);
128 if (end > data.Length)
129 Fail("byte " + p + ": truncated chunk");
130 return (int)end;
131 }
132 int headerEnd = Chunk(0, "MThd");
133 if (headerEnd != 14)
134 Fail("byte 4: bad header length");
135 int f = (int)ReadBig(8, 2), tracks = (int)ReadBig(10, 2), division = (int)ReadBig(12, 2);
136 if (f > 2)
137 Fail("byte 8: bad format");
138 if (tracks == 0 || f == 0 && tracks != 1)
139 Fail("byte 10: bad track count");
140 if ((division & 32768) != 0)
141 Fail("byte 12: smpte timing not supported");
142 if (division == 0)
143 Fail("byte 12: bad division");
144 Song song = new() { Format = f, Division = division };
145 int p = 14;
146 for (int track = 0; track < tracks; track++) {
147 if (p == data.Length)
148 Fail("byte " + p + ": missing track");
149 int end = Chunk(p, "MTrk");
150 p += 8;
151 int status = 0;
152 long tick = 0;
153 var events = new List<Event>();
154 while (p < end) {
155 if (events.Count > 0 && events.Last().End)
156 Fail("byte " + p + ": event after end of track");
157 int start = p;
158 int ReadByte() {
159 if (p >= end)
160 Fail("byte " + start + ": truncated event");
161 return data[p++];
162 }
163 int Var() {
164 int at = p, value = 0;
165 for (int i = 0; i < 4; i++) {
166 int b = ReadByte();
167 value = value * 128 + (b & 127);
168 if (b < 128)
169 return value;
170 }
171 Fail("byte " + at + ": variable-length value too long");
172 return 0;
173 }
174 tick += Var();
175 if (p >= end)
176 Fail("byte " + start + ": truncated event");
177 int position = p, current;
178 if (data[p] < 128) {
179 if (status == 0)
180 Fail("byte " + p + ": data byte without running status");
181 current = status;
182 } else
183 current = ReadByte();
184 int meta = -1;
185 byte[] d;
186 if (current < 240) {
187 status = current;
188 d = new byte[(current >> 4) is 12 or 13 ? 1 : 2];
189 for (int j = 0; j < d.Length; j++) {
190 int at = p, b = ReadByte();
191 if (b >= 128)
192 Fail("byte " + at + ": bad data byte");
193 d[j] = (byte)b;
194 }
195 } else if (current is 240 or 247 or 255) {
196 status = 0;
197 if (current == 255)
198 meta = ReadByte();
199 int length = Var();
200 if (length > end - p)
201 Fail("byte " + start + ": truncated event");
202 d = data.Skip(p).Take(length).ToArray();
203 p += length;
204 if (current == 255)
205 ValidateMeta(meta, d, position);
206 } else {
207 Fail("byte " + position + ": unknown status byte");
208 return song;
209 }
210 events.Add(new(tick, current, meta, d));
211 }
212 if (events.Count == 0 || !events.Last().End)
213 Fail("byte " + end + ": missing end of track");
214 song.Tracks.Add(events);
215 }
216 if (p != data.Length)
217 Fail("byte " + p + ": extra data");
218 return song;
219 }
220 static Event TextEvent(long tick, string type, string fields) {
221 string[] a = fields == "" ? Array.Empty<string>() : fields.Split(' ');
222 void Require(bool valid) {
223 if (!valid)
224 Fail("bad value");
225 }
226 long Num(int i, long lo, long hi) {
227 if (i >= a.Length || !long.TryParse(a[i], out long v) || !Range(v, lo, hi)) {
228 Fail("bad value");
229 return 0;
230 }
231 return v;
232 }
233 byte[] Blob(int index) {
234 long length = Num(index, 0, 1000000);
235 Require(a.Length == index + (length == 0 ? 1 : 2));
236 if (length == 0)
237 return Array.Empty<byte>();
238 string text = a[index + 1];
239 Require(text.Length == length * 2 &&
240 text.All(c => c is >= '0' and <= '9' or >= 'A' and <= 'F'));
241 return Convert.FromHexString(text);
242 }
243 int code = Array.IndexOf(channelTypes, type);
244 if (code >= 0) {
245 int high = code + 8;
246 Require(a.Length == (high is 12 or 13 or 14 ? 2 : 3));
247 int ch = (int)Num(0, 1, 16);
248 byte[] d;
249 if (high is 8 or 9 or 10) {
250 int pitch = ParsePitch(a[1]);
251 Require(pitch >= 0);
252 d = new[] { (byte)pitch, (byte)Num(2, 0, 127) };
253 } else if (high == 14) {
254 int value = (int)Num(1, -8192, 8191) + 8192;
255 d = new[] { (byte)(value & 127), (byte)(value >> 7) };
256 } else
257 d = high == 11 ? new[] { (byte)Num(1, 0, 127), (byte)Num(2, 0, 127) }
258 : new[] { (byte)Num(1, 0, 127) };
259 return new(tick, (high << 4) + ch - 1, -1, d);
260 }
261 if (type is "SYSEX" or "SYSEX_ESCAPE")
262 return new(tick, type == "SYSEX" ? 240 : 247, -1, Blob(0));
263 int textIndex = Array.IndexOf(textTypes, type);
264 if (textIndex >= 0) {
265 Require(fields.Length >= 2 && fields[0] == '"' && fields[^1] == '"' &&
266 !fields[1.. ^ 1].Contains('"'));
267 return new(tick, 255, textIndex + 1, Encoding.ASCII.GetBytes(fields[1.. ^ 1]));
268 }
269 int meta =
270 metaTypes.FirstOrDefault(x => x.Value == type, new KeyValuePair<int, string>(-1, ""))
271 .Key;
272 if (type == "META") {
273 Require(a.Length >= 2 && a[0].Length == 2 &&
274 a[0].All(c => c is >= '0' and <= '9' or >= 'A' and <= 'F'));
275 meta = Convert.ToInt32(a[0], 16);
276 Require(!metaTypes.ContainsKey(meta) && !Range(meta, 1, 7));
277 return new(tick, 255, meta, Blob(1));
278 }
279 if (meta < 0) {
280 Fail("unknown event type");
281 }
282 byte[] data = Array.Empty<byte>();
283 switch (meta) {
284 case 0:
285 Require(a.Length == 1);
286 data = Bytes(Num(0, 0, 65535), 2);
287 break;
288 case 32:
289 Require(a.Length == 1);
290 data = new[] { (byte)(Num(0, 1, 16) - 1) };
291 break;
292 case 47:
293 Require(a.Length == 0);
294 break;
295 case 81:
296 Require(a.Length == 1);
297 data = Bytes(Num(0, 1, 16777215), 3);
298 break;
299 case 84:
300 Require(a.Length == 5);
301 data = Enumerable.Range(0, 5).Select(i => (byte)Num(i, 0, 255)).ToArray();
302 break;
303 case 88:
304 Require(a.Length == 3);
305 string[] signature = a[0].Split('/');
306 Require(signature.Length == 2 && long.TryParse(signature[0], out _) &&
307 long.TryParse(signature[1], out _));
308 long nn = I(signature[0]), dd = I(signature[1]);
309 Require(Range(nn, 1, 255) && new long[] { 1, 2, 4, 8, 16, 32, 64 }.Contains(dd));
310 int exponent = 0;
311 while ((1L << exponent) != dd)
312 exponent++;
313 data = new[] { (byte)nn, (byte)exponent, (byte)Num(1, 0, 255), (byte)Num(2, 0, 255) };
314 break;
315 case 89:
316 Require(a.Length == 2 && a[1] is "major" or "minor");
317 int minor = a[1] == "minor" ? 1 : 0, key = Array.IndexOf(keys[minor], a[0]);
318 Require(key >= 0);
319 data = new[] { unchecked((byte)(key - 7)), (byte)minor };
320 break;
321 case 127:
322 data = Blob(0);
323 break;
324 }
325 return new(tick, 255, meta, data);
326 }
327 static Song ParseText(string[] lines) {
328 string[] h = lines[0].Split(' ');
329 if (h.Length != 4 || h[0] != "HEADER" || !int.TryParse(h[1], out int f) ||
330 !int.TryParse(h[2], out int tracks) || !int.TryParse(h[3], out int division) ||
331 !Range(f, 0, 2) || !Range(tracks, 1, 65535) || !Range(division, 1, 32767) ||
332 f == 0 && tracks != 1) {
333 Fail("line 1: bad header");
334 return new();
335 }
336 Song s = new() { Format = f, Division = division, Tracks = new() { new() } };
337 int current = 0;
338 for (int i = 1; i < lines.Length; i++) {
339 string[] a = lines[i].Split(' ', 4);
340 long tick = I(a[0]);
341 int track = (int)I(a[1]);
342 Event e;
343 try {
344 e = TextEvent(tick, a[2], a.Length == 4 ? a[3] : "");
345 } catch (MidiError error) {
346 Fail("line " + (i + 1) + ": " + error.Message);
347 return s;
348 }
349 void Bad(string reason) => Fail("line " + (i + 1) + ": " + reason);
350 if (track >= tracks)
351 Bad("bad track");
352 bool end = s.Tracks[current].Count > 0 && s.Tracks[current].Last().End;
353 if (track < current)
354 Bad("out of order");
355 if (track > current && (!end || track > current + 1))
356 Bad("missing end of track");
357 if (track == current && end)
358 Bad("event after end of track");
359 if (track == current + 1) {
360 current++;
361 s.Tracks.Add(new());
362 }
363 if (s.Tracks[current].Count > 0 && tick < s.Tracks[current].Last().Tick)
364 Bad("out of order");
365 s.Tracks[current].Add(e);
366 }
367 if (current + 1 < tracks || s.Tracks[current].Count == 0 || !s.Tracks[current].Last().End)
368 Fail("line " + (lines.Length + 1) + ": missing end of track");
369 return s;
370 }
371 static byte[] Var(long value) {
372 List<byte> b = new() { (byte)(value & 127) };
373 while ((value >>= 7) > 0)
374 b.Insert(0, (byte)((value & 127) | 128));
375 return b.ToArray();
376 }
377 static void Encode(Song song) {
378 List<byte> all = new(Encoding.ASCII.GetBytes("MThd"));
379 all.AddRange(Bytes(6, 4));
380 all.AddRange(Bytes(song.Format, 2));
381 all.AddRange(Bytes(song.Tracks.Count, 2));
382 all.AddRange(Bytes(song.Division, 2));
383 foreach (var track in song.Tracks) {
384 List<byte> data = new();
385 long previous = 0;
386 int last = -1;
387 foreach (Event e in track) {
388 data.AddRange(Var(e.Tick - previous));
389 previous = e.Tick;
390 if (e.Channel) {
391 int status = e.Status;
392 byte[] d = e.Data;
393 if (zeroOff && e.Code == 8) {
394 status = 0x90 + e.Ch - 1;
395 d = new[] { d[0], (byte)0 };
396 }
397 if (!running || status != last)
398 data.Add((byte)status);
399 data.AddRange(d);
400 last = status;
401 } else {
402 last = -1;
403 data.Add((byte)e.Status);
404 if (e.Status == 255)
405 data.Add((byte)e.Meta);
406 data.AddRange(Var(e.Data.Length));
407 data.AddRange(e.Data);
408 }
409 }
410 all.AddRange(Encoding.ASCII.GetBytes("MTrk"));
411 all.AddRange(Bytes(data.Count, 4));
412 all.AddRange(data);
413 }
414 for (int i = 0; i < all.Count; i += 16)
415 Console.WriteLine(i.ToString("X4") + ": " +
416 string.Join(" ", all.Skip(i).Take(16).Select(x => x.ToString("X2"))));
417 Console.WriteLine(all.Count + " bytes");
418 }
419 static (List<Note> Notes, List<(Event E, int Track, int Index)> Unmatched) Pair(Song song) {
420 var all = song.Global();
421 var queues = new Dictionary<(int, int), Queue<(Event E, int Order)>>();
422 var notes = new List<Note>();
423 var paired = new HashSet<Event>(ReferenceEqualityComparer.Instance);
424 for (int i = 0; i < all.Count; i++) {
425 Event e = all[i].E;
426 if (!e.On && !e.Off)
427 continue;
428 var key = (e.Ch, (int)e.Data[0]);
429 if (!queues.ContainsKey(key))
430 queues[key] = new();
431 if (e.On)
432 queues[key].Enqueue((e, i));
433 else if (queues[key].Count > 0) {
434 var on = queues[key].Dequeue();
435 notes.Add(new(on.E, e, on.Order));
436 paired.Add(on.E);
437 paired.Add(e);
438 }
439 }
440 notes = notes.OrderBy(n => n.On.Tick)
441 .ThenBy(n => n.On.Ch)
442 .ThenBy(n => n.On.Data[0])
443 .ThenBy(n => n.Order)
444 .ToList();
445 return (notes, all.Where(x => (x.E.On || x.E.Off) && !paired.Contains(x.E)).ToList());
446 }
447 static (string Time, string Bar) Position(Song song, long tick, bool bars = true) {
448 long sum = 0, last = 0, tempo = 500000, start = 0, bar = 1, nn = 4, beat = song.Division;
449 foreach (var item in song.Global().Where(x => x.E.Tick <= tick)) {
450 Event e = item.E;
451 if (e.Status != 255)
452 continue;
453 if (e.Meta == 81) {
454 sum += (e.Tick - last) * tempo;
455 last = e.Tick;
456 tempo = e.Number;
457 }
458 if (e.Meta == 88 && bars) {
459 long size = nn * beat;
460 bar += (e.Tick - start + size - 1) / size;
461 start = e.Tick;
462 nn = e.Data[0];
463 beat = song.Division * 4 / (1 << e.Data[1]);
464 }
465 }
466 long micro = (sum + (tick - last) * tempo) / song.Division, x = tick - start;
467 return (micro / 1000 + "." + (micro % 1000).ToString("D3"),
468 bars ? (bar + x / (nn * beat)) + ":" + (x % (nn * beat) / beat + 1) + ":" + x % beat
469 : "");
470 }
471 static void Notes(Song s) {
472 var pair = Pair(s);
473 Console.WriteLine("notes " + pair.Notes.Count + " unmatched " + pair.Unmatched.Count);
474 foreach (Note n in pair.Notes)

476 Pitch(n.On.Data[0]) + " " + n.On.Data[1]);
477 foreach (var x in pair.Unmatched)
478 Console.WriteLine("unmatched " + (x.E.On ? "on" : "off") + " " + x.E.Tick + " " +
479 x.Track + " " + x.E.Ch + " " + Pitch(x.E.Data[0]));
480 }
481 static string Summary(Song s) => "format " + s.Format + " tracks " + s.Tracks.Count +
482 " division " + s.Division + " events " + s.Count;
483 static long Floor(long x, long divisor) => x >= 0 ? x / divisor
484 : -((-x + divisor - 1) / divisor);
485 static void Execute(string[] a) {
486 string cmd = a[0];
487 string[] known = { "DECODE", "ENCODE", "TIMING", "NOTES", "STATS", "ROLL",
488 "TRANSPOSE", "QUANTIZE", "VELOCITY", "TEMPO", "FILTER", "MERGE",
489 "SPLIT", "COPY", "LIST", "RUNNING", "NOTEOFF" };
490 if (!known.Contains(cmd))
491 Fail("unknown command");
492 if (cmd == "LIST") {
493 if (songs.Count == 0)
494 Console.WriteLine("no songs");
495 foreach (var item in songs)
496 Console.WriteLine(item.Key + " " + Summary(item.Value));
497 return;
498 }
499 if (cmd is "RUNNING" or "NOTEOFF") {
500 if (cmd == "RUNNING") {
501 if (a[1] is not("ON" or "OFF"))
502 Fail("bad argument");
503 running = a[1] == "ON";
504 } else {
505 if (a[1] is not("ZERO" or "KEEP"))
506 Fail("bad argument");
507 zeroOff = a[1] == "ZERO";
508 }
509 Console.WriteLine(cmd.ToLowerInvariant() + " " + a[1].ToLowerInvariant());
510 return;
511 }
512 if (!songs.ContainsKey(a[1]))
513 Fail("no song " + a[1]);
514 Song s = songs[a[1]];
515 long n = 0, second = 0, third = 0;
516 if (cmd is "TRANSPOSE" or "QUANTIZE" or "VELOCITY" or "TEMPO" or "ROLL")
517 n = I(a[2]);
518 if (cmd is "QUANTIZE" or "VELOCITY" or "TEMPO" or "ROLL")
519 second = I(a[3]);
520 if (cmd == "ROLL")
521 third = I(a[4]);
522 bool bad = cmd switch { "TRANSPOSE" =>
523 !Range(n, -127, 127) || a.Length == 4 && !Range(I(a[3]), 1, 16),
524 "QUANTIZE" => !Range(n, 1, 1000000) || !Range(second, 0, 100),
525 "VELOCITY" => !Range(n, 0, 1000) || !Range(second, -127, 127),
526 "TEMPO" => !Range(n, 1, 1000) || !Range(second, 1, 1000),
527 "FILTER" => a.Skip(2).Any(x => !Range(I(x), 1, 16)),
528 "ROLL" => !Range(n, 0, 10000000) || !Range(second, 1, 10000000) ||
529 !Range(third, 1, 10000000) || second <= n ||
530 (second - n + third - 1) / third > 64,
531 _ => false };
532 if (bad)
533 Fail("bad argument");
534 if (cmd is "TIMING" or "STATS" && s.Format == 2)
535 Fail("format 2 not supported");
536 switch (cmd) {
537 case "DECODE":
538 Console.WriteLine("HEADER " + s.Format + " " + s.Tracks.Count + " " + s.Division);
539 for (int tr = 0; tr < s.Tracks.Count; tr++)
540 foreach (Event e in s.Tracks[tr])
541 Console.WriteLine(Line(e, tr));
542 break;
543 case "ENCODE":
544 Encode(s);
545 break;
546 case "TIMING":
547 foreach (var x in s.Global()) {
548 var p = Position(s, x.E.Tick);
549 Console.WriteLine(x.E.Tick + " " + x.Track + " " + Type(x.E) + " " + p.Time + " " +
550 p.Bar);
551 }
552 var end = Position(s, s.Length);
553 Console.WriteLine("end " + s.Length + " " + end.Time + " " + end.Bar);
554 break;
555 case "NOTES":
556 Notes(s);
557 break;
558 case "COPY":
559 songs[a[2]] = s.Copy();
560 Console.WriteLine("copied " + a[1] + " to " + a[2]);
561 break;
562 case "TRANSPOSE":
563 int kept = 0;
564 var dropped = new List<string>();
565 for (int tr = 0; tr < s.Tracks.Count; tr++) {
566 var track = new List<Event>();
567 foreach (Event e in s.Tracks[tr]) {
568 if (e.Channel && e.Code is 8 or 9 or 10 && (a.Length == 3 || e.Ch == I(a[3]))) {
569 long pitch = e.Data[0] + n;
570 if (!Range(pitch, 0, 127)) {
571 dropped.Add("dropped " + e.Tick + " " + tr + " " + Type(e) + " " +
572 e.Ch + " " + Pitch(e.Data[0]));
573 continue;
574 }
575 e.Data[0] = (byte)pitch;
576 kept++;
577 }
578 track.Add(e);
579 }
580 s.Tracks[tr] = track;
581 }
582 Console.WriteLine("transposed " + kept + " dropped " + dropped.Count);
583 dropped.ForEach(Console.WriteLine);
584 break;
585 case "QUANTIZE":
586 var pairs = Pair(s);
587 var shifts = new Dictionary<Event, long>(ReferenceEqualityComparer.Instance);
588 int targets = 0, moved = 0;
589 void Shift(Event on, Event? off)
590 {
591 long lo = on.Tick / n * n, g = on.Tick - lo <= lo + n - on.Tick ? lo : lo + n;
592 long delta = Floor(2 * (g - on.Tick) * second + 100, 200);
593 targets++;
594 if (delta != 0)
595 moved++;
596 shifts[on] = delta;
597 if (off != null)
598 shifts[off] = delta;
599 }
600 foreach (Note note in pairs.Notes)
601 Shift(note.On, note.Off);
602 foreach (var x in pairs.Unmatched.Where(x => x.E.On))
603 Shift(x.E, null);
604 for (int tr = 0; tr < s.Tracks.Count; tr++) {
605 Event eot = s.Tracks[tr].Last();
606 var track =
607 s.Tracks[tr]
608 .Where(e => !e.End)
609 .Select(e => e with { Tick = e.Tick + shifts.GetValueOrDefault(e) })
610 .OrderBy(e => e.Tick)
611 .ToList();
612 track.Add(eot with { Tick = Math.Max(
613 eot.Tick, track.Count == 0 ? 0 : track.Last().Tick) });
614 s.Tracks[tr] = track;
615 }
616 Console.WriteLine("quantized " + targets + " moved " + moved);
617 break;
618 case "VELOCITY":
619 int count = 0, clamped = 0;
620 foreach (Event e in s.Tracks.SelectMany(t => t).Where(e => e.On)) {
621 long value = (e.Data[1] * n + 50) / 100 + second;
622 count++;
623 if (!Range(value, 1, 127))
624 clamped++;
625 e.Data[1] = (byte)Math.Clamp(value, 1, 127);
626 }
627 Console.WriteLine("velocity " + count + " clamped " + clamped);
628 break;
629 case "TEMPO":
630 Song candidate = s.Copy();
631 int scaled = 0;
632 bool atZero = false;
633 long Scale(long value) {
634 long result = (2 * value * second + n) / (2 * n);
635 if (!Range(result, 1, 16777215))
636 Fail("tempo out of range");
637 return result;
638 }
639 foreach (var track in candidate.Tracks)
640 for (int i = 0; i < track.Count; i++)
641 if (track[i].Status == 255 && track[i].Meta == 81) {
642 if (track[i].Tick == 0)
643 atZero = true;
644 scaled++;
645 track[i] = track[i] with { Data = Bytes(Scale(track[i].Number), 3) };
646 }
647 Event? inserted = null;
648 if (!atZero) {
649 inserted = new(0, 255, 81, Bytes(Scale(500000), 3));
650 candidate.Tracks[0].Insert(0, inserted);
651 }
652 songs[a[1]] = candidate;
653 Console.WriteLine("scaled " + scaled);
654 if (inserted != null)
655 Console.WriteLine("inserted " + Line(inserted, 0));
656 break;
657 case "FILTER":
658 var channels = a.Skip(2).Select(I).ToHashSet();
659 int removed = 0;
660 foreach (var track in s.Tracks)
661 removed += track.RemoveAll(e => e.Channel && !channels.Contains(e.Ch));
662 Console.WriteLine("removed " + removed);
663 break;
664 case "MERGE":
665 if (s.Format != 1)
666 Fail("not format 1");
667 int old = s.Tracks.Count;
668 long length = s.Length;
669 var merged = s.Global().Where(x => !x.E.End).Select(x => x.E).ToList();
670 merged.Add(new(length, 255, 47, Array.Empty<byte>()));
671 s.Tracks = new() { merged };
672 s.Format = 0;
673 Console.WriteLine("merged " + old + " tracks events " + s.Count);
674 break;
675 case "SPLIT":
676 if (s.Format != 0)
677 Fail("not format 0");
678 long lastTick = s.Length;
679 var original = s.Tracks[0];
680 var usedChannels =
681 original.Where(e => e.Channel).Select(e => e.Ch).Distinct().Order().ToArray();
682 var tracks =
683 new List<List<Event>> { original.Where(e => !e.Channel && !e.End).ToList() };
684 tracks.AddRange(
685 usedChannels.Select(ch => original.Where(e => e.Channel && e.Ch == ch).ToList()));
686 foreach (var track in tracks)
687 track.Add(new(lastTick, 255, 47, Array.Empty<byte>()));
688 s.Tracks = tracks;
689 s.Format = 1;
690 Console.WriteLine("split tracks " + tracks.Count + " events " + s.Count);
691 break;
692 case "STATS":
693 Stats(s);
694 break;
695 case "ROLL":
696 var notes = Pair(s).Notes;
697 bool any = false;
698 for (int pitch = 127; pitch >= 0; pitch--) {
699 string row = "";
700 for (long at = n; at < second; at += third)
701 row += notes.Any(note => note.On.Data[0] == pitch && at <= note.On.Tick &&
702 note.On.Tick < at + third)
703 ? '#'
704 : notes.Any(note => note.On.Data[0] == pitch && note.On.Tick <= at &&
705 at < note.Off.Tick)
706 ? '='
707 : '.';
708 if (row.All(c => c == '.'))
709 continue;
710 any = true;
711 Console.WriteLine(Pitch(pitch).PadRight(4) + " " + row);
712 }
713 if (!any)
714 Console.WriteLine("empty");
715 break;
716 }
717 }
718 static void Stats(Song s) {
719 var pair = Pair(s);
720 var notes = pair.Notes;
721 Console.WriteLine(Summary(s));
722 Console.WriteLine("notes " + notes.Count + " unmatched " + pair.Unmatched.Count);
723 foreach (var group in notes.GroupBy(n => n.On.Ch).OrderBy(g => g.Key))
724 Console.WriteLine("channel " + group.Key + " notes " + group.Count() + " range " +
725 Pitch(group.Min(n => n.On.Data[0])) + " " +
726 Pitch(group.Max(n => n.On.Data[0])));
727 Console.WriteLine(notes.Count == 0 ? "range none"
728 : "range " + Pitch(notes.Min(n => n.On.Data[0])) + " " +
729 Pitch(notes.Max(n => n.On.Data[0])));
730 Console.WriteLine("length " + s.Length + " ticks " + Position(s, s.Length, false).Time +
731 " ms");
732 var changes = new SortedDictionary<long, int>();
733 foreach (Note n in notes.Where(n => n.Off.Tick > n.On.Tick)) {
734 changes[n.On.Tick] = changes.GetValueOrDefault(n.On.Tick) + 1;
735 changes[n.Off.Tick] = changes.GetValueOrDefault(n.Off.Tick) - 1;
736 }
737 int current = 0, maximum = 0;
738 long at = 0;
739 foreach (var item in changes) {
740 current += item.Value;
741 if (current > maximum) {
742 maximum = current;
743 at = item.Key;
744 }
745 }
746 Console.WriteLine(maximum == 0 ? "polyphony 0" : "polyphony " + maximum + " at " + at);
747 }
