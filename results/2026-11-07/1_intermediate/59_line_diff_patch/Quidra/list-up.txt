class PatchHunk(nat line_number, nat old_start, nat old_count, nat new_start, nat new_count)
    BodyLine[] body := []
    nat leading_context()
        nat count := 0
        while count < #body and body[count].kind == " "
            count += 1
        return count
nat | none parse_number(string text)
    if #text < 1 or #text > 4
        return none
    for: index < #text
        if not "0123456789".contains(text[index])
            return none
    return nat(text)

class NumberPair(nat first, nat | none second)

NumberPair | none parse_number_pair(string text)
    parts := text.split(",")
    if #parts > 2
        return none
    match parse_number(parts[0])
        none
            return none
        nat first
            if #parts == 1
                return NumberPair(first, none)
            match parse_number(parts[1])
                none
                    return none
                nat second
                    return NumberPair(first, second)

class HunkRange(nat start, nat count)

HunkRange | none parse_range(string text, string sign)
    if not text.starts_with(sign)
        return none
    match parse_number_pair(text[1:])
        none
            return none
        NumberPair pair
            nat count := 1
            match pair.second
                nat second
                    count = second
                none
                    void
            if pair.first == 0 and count != 0
                return none
            return HunkRange(pair.first, count)

PatchHunk | error parse_hunk_header(string line, nat line_number)
    parts := line.split(" ")
    if #parts == 4 and parts[0] == "@@" and parts[3] == "@@"
        match parse_range(parts[1], "-")
            HunkRange old_range
                match parse_range(parts[2], "+")
                    HunkRange new_range
                        return PatchHunk(
                            line_number,
                            old_range.start,
                            old_range.count,
                            new_range.start,
                            new_range.count,
                        )
                    none
                        void
            none
                void
    return error("PATCH ERROR line {line_number}: bad hunk header")

PatchHunk[] | error parse_patch(const string[] patch_lines)
    nat index := 0
    while index < #patch_lines and not patch_lines[index].starts_with("@@")
        index += 1
    if index == #patch_lines
        return error("PATCH ERROR no hunks")
    PatchHunk[] hunks := []
    while index < #patch_lines
        hunk := try parse_hunk_header(patch_lines[index], index + 1)
        index += 1
        nat old_seen := 0
        nat new_seen := 0
        while old_seen < hunk.old_count or new_seen < hunk.new_count
            if index == #patch_lines
                return error("PATCH ERROR line {hunk.line_number}: truncated hunk")
            line := patch_lines[index]
            kind := line[0]
            in_old := kind == " " or kind == "-"
            in_new := kind == " " or kind == "+"
            old_full := in_old and old_seen == hunk.old_count
            new_full := in_new and new_seen == hunk.new_count
            if #line < 2 or not (in_old or in_new) or old_full or new_full
                return error("PATCH ERROR line {index + 1}: bad hunk line")
            hunk.body = hunk.body.append(BodyLine(kind, line[1:]))
            if in_old
                old_seen += 1
            if in_new
                new_seen += 1
            index += 1
        if hunk.leading_context() == #hunk.body
            return error("PATCH ERROR line {hunk.line_number}: no changes")
        hunks = hunks.append(hunk)
    return hunks
