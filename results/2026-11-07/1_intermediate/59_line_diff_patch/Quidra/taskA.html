60(nat, nat)[] matched_pairs(const string[] first, const string[] second)
61 rows := #first
62 columns := #second
63 nat[] empty_row := array(columns + 1, fill = 0)
64 nat[][] table := array(rows + 1, fill = empty_row)
65 for: 1 <= i <= rows
66 for: 1 <= j <= columns
67 if first[i - 1] == second[j - 1]
68 table[i][j] = table[i - 1][j - 1] + 1
69 else
70 table[i][j] = maximum(table[i - 1][j], table[i][j - 1])
71 (nat, nat)[] pairs := [(rows + 1, columns + 1)]
72 i := rows
73 j := columns
74 while i >= 1 and j >= 1
75 if first[i - 1] == second[j - 1]
76 pairs = pairs.append((i, j))
77 i -= 1
78 j -= 1
79 elif table[i - 1][j] >= table[i][j - 1]
80 i -= 1
81 else
82 j -= 1
83 pairs = pairs.append((0, 0))
84 return pairs[:: -1]

474HunkMatch | none search_hunk(
475 PatchHunk hunk,
476 const string[] lines,
477 int adjustment,
478 bool reverse,
479 nat max_fuzz,
480)
481 old, new := hunk.sequences(reverse)
482 start := int(hunk.start(reverse))
483 expected := if #old > 0 then start - 1 + adjustment else start + adjustment
484 leading := hunk.leading_context()
485 trailing := hunk.trailing_context()
486 offsets := [0, 1, -1, 2, -2, 3, -3]
487 for: 0 <= fuzz <= max_fuzz
488 head := minimum(fuzz, leading)
489 tail := minimum(fuzz, trailing)
490 string[] old_part := old[head:<#old - tail]
491 string[] new_part := new[head:<#new - tail]
492 for delta := offsets[]
493 position := expected + int(head) + delta
494 end := position + int(#old_part)
495 if position < 0 or end > int(#lines)
496 continue
497 if lines[position:<end] == old_part
498 string[] patched := lines[:<position].concat(new_part).concat(lines[end:])
499 return HunkMatch(fuzz, delta, position, patched)
500 return none

838 void | error .token_diff(const string[] arguments, bool by_word)
839 old_name := arguments[0]
840 new_name := arguments[1]
841 string[] old := try require(old_name)
842 string[] new := try require(new_name)
843 line_diff := LineDiff(old, new)
844 if #line_diff.blocks == 0
845 emit("Files {old_name} and {new_name} are identical")
846 return
847 separator := if by_word then " " else ""
848 totals := TokenTotals()
849 for block := line_diff.blocks[]
850 for: row < maximum(block.x, block.y)
851 old_number := block.p + row + 1
852 new_number := block.q + row + 1

854 old_tokens := split_tokens(old[old_number - 1], by_word)
855 new_tokens := split_tokens(new[new_number - 1], by_word)
856 text := compare_tokens(old_tokens, new_tokens, separator, &totals)
857 emit("a{old_number} b{new_number}: {text}")
858 elif row < block.x
859 old_tokens := split_tokens(old[old_number - 1], by_word)
860 totals.deleted += #old_tokens
861 emit("a{old_number}: {deletion(old_tokens, separator)}")
862 else
863 new_tokens := split_tokens(new[new_number - 1], by_word)
864 totals.inserted += #new_tokens
865 emit("b{new_number}: {insertion(new_tokens, separator)}")
866 label := if by_word then "WORDS" else "CHARS"
867 emit("{label} deleted={totals.deleted} inserted={totals.inserted} common={totals.common}")

893 void | error .patch(const string[] arguments, const string[] data)
894 name := arguments[0]
895 string[] lines := try require(name)
896 options := arguments[2:]
897 reverse := contains(options, "REVERSE")
898 dry_run := contains(options, "DRYRUN")

907 forward := hunk_applies(first, lines, reverse, max_fuzz)

910 emit("RESULT applied=0 failed=0 ignored={#hunks} lines={#lines}")
911 return
912 adjustment := 0
913 nat applied := 0
914 nat failed := 0
915 for: index < #hunks
916 hunk := hunks[index]
917 number := index + 1
918 match search_hunk(hunk, lines, adjustment, reverse, max_fuzz)

920 failed += 1
921 emit("Hunk #{number} FAILED at {hunk.start(reverse)}")
922 HunkMatch found
923 old, new := hunk.sequences(reverse)
924 lines = found.lines
925 adjustment += found.delta + int(#new) - int(#old)
926 applied += 1
927 emit(found.report(number))
928 emit("RESULT applied={applied} failed={failed} ignored=0 lines={#lines}")
