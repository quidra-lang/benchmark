26 var key = line

33fun matchedPairs(first: List<String>, second: List<String>): List<Pair<Int, Int>> {
34 val rows = first.size
35 val columns = second.size
36 val table = Array(rows + 1) { IntArray(columns + 1) }
37 for (i in 1..rows) {
38 for (j in 1..columns) {
39 table[i][j] = if (first[i - 1] == second[j - 1]) {
40 table[i - 1][j - 1] + 1
41 } else {
42 maxOf(table[i - 1][j], table[i][j - 1])
43 }
44 }
45 }
46 val pairs = mutableListOf(rows + 1 to columns + 1)
47 var i = rows
48 var j = columns
49 while (i >= 1 && j >= 1) {
50 when {
51 first[i - 1] == second[j - 1] -> {
52 pairs.add(i to j)
53 i--
54 j--
55 }
56 table[i - 1][j] >= table[i][j - 1] -> i--
57 else -> j--
58 }
59 }
60 pairs.add(0 to 0)
61 return pairs.asReversed()
62}

64data class ChangeBlock(val p: Int, val q: Int, val x: Int, val y: Int) {
65 val letter: Char
66 get() = when {
67 x >= 1 && y >= 1 -> 'c'
68 y == 0 -> 'd'
69 else -> 'a'
70 }
71}

73fun changeBlocks(first: List<String>, second: List<String>): List<ChangeBlock> =
74 matchedPairs(first, second).zipWithNext().mapNotNull { (current, next) ->
75 val (i, j) = current
76 val (nextI, nextJ) = next
77 ChangeBlock(i, j, nextI - i - 1, nextJ - j - 1).takeIf { it.x > 0 || it.y > 0 }
78 }

80data class Step(val oldIndex: Int, val newIndex: Int, val block: ChangeBlock?)

82fun walk(blocks: List<ChangeBlock>, oldStart: Int, newStart: Int, oldEnd: Int): List<Step> {
83 val steps = mutableListOf<Step>()
84 var oldIndex = oldStart
85 var newIndex = newStart
86 for (block in blocks) {
87 while (oldIndex < block.p) {
88 steps.add(Step(oldIndex++, newIndex++, null))
89 }
90 steps.add(Step(oldIndex, newIndex, block))
91 oldIndex = block.p + block.x
92 newIndex = block.q + block.y
93 }
94 while (oldIndex < oldEnd) {
95 steps.add(Step(oldIndex++, newIndex++, null))
96 }
97 return steps
98}

103fun statGraph(inserted: Int, deleted: Int): String {
104 val total = inserted + deleted
105 var plus = inserted
106 var minus = deleted
107 if (total > GRAPH_WIDTH) {
108 plus = GRAPH_WIDTH * inserted / total
109 minus = GRAPH_WIDTH * deleted / total
110 if (inserted >= 1 && plus == 0) plus = 1
111 if (deleted >= 1 && minus == 0) minus = 1
112 }
113 return "+".repeat(plus) + "-".repeat(minus)
114}

116class Hunk(val blocks: List<ChangeBlock>, context: Int, oldTotal: Int) {
117 val oldStart: Int
118 val oldCount: Int
119 val newStart: Int
120 val newCount: Int
121
122 init {
123 val first = blocks.first()
124 val last = blocks.last()
125 val before = minOf(context, first.p)
126 val after = minOf(context, oldTotal - last.p - last.x)
127 oldStart = first.p - before
128 oldCount = last.p + last.x + after - oldStart
129 newStart = first.q - before
130 newCount = last.q + last.y + after - newStart
131 }
132
133 fun steps() = walk(blocks, oldStart, newStart, oldStart + oldCount)
134}

136class LineDiff(
137 private val oldLines: List<String>,
138 private val newLines: List<String>,
139 options: List<String> = emptyList(),
140) {
141 val blocks = changeBlocks(
142 oldLines.map { comparisonKey(it, options) },

145
146 fun deleted(block: ChangeBlock) = oldLines.subList(block.p, block.p + block.x)
147
148 fun inserted(block: ChangeBlock) = newLines.subList(block.q, block.q + block.y)

150 fun hunks(context: Int): List<Hunk> {
151 val groups = mutableListOf<MutableList<ChangeBlock>>()
152 for (block in blocks) {
153 val previous = groups.lastOrNull()?.last()
154 if (previous != null && block.p - (previous.p + previous.x) <= 2 * context) {
155 groups.last().add(block)
156 } else {
157 groups.add(mutableListOf(block))
158 }
159 }
160 return groups.map { Hunk(it, context, oldLines.size) }
161 }

163 fun normal(): List<String> = buildList {
164 for (block in blocks) {
165 add(rangeR(block.p, block.x) + block.letter + rangeR(block.q, block.y))
166 addAll(deleted(block).map { "< $it" })
167 if (block.letter == 'c') add("---")

192 fun ifdef(macro: String): List<String> = buildList {
193 for ((oldIndex, _, block) in walk(blocks, 0, 0, oldLines.size)) {

195 block == null -> add(oldLines[oldIndex])

206 add("#endif /* ! $macro */")

210 addAll(inserted(block))
211 add("#endif /* $macro */")
212 }
213 }

217 fun unified(context: Int, oldName: String, newName: String): List<String> = buildList {
218 add("--- $oldName")
219 add("+++ $newName")
220 for (hunk in hunks(context)) {
221 val oldRange = rangeU(hunk.oldStart, hunk.oldCount)
222 val newRange = rangeU(hunk.newStart, hunk.newCount)
223 add("@@ -$oldRange +$newRange @@")
224 for ((oldIndex, _, block) in hunk.steps()) {
225 if (block == null) {
226 add(" " + oldLines[oldIndex])
227 } else {
228 addAll(deleted(block).map { "-$it" })
229 addAll(inserted(block).map { "+$it" })
230 }
231 }
232 }
233 }

235 fun context(context: Int, oldName: String, newName: String): List<String> = buildList {
236 add("*** $oldName")
237 add("--- $newName")
238 for (hunk in hunks(context)) {
239 add("*".repeat(15))
240 add("*** ${rangeR(hunk.oldStart, hunk.oldCount)} ****")
241 if (hunk.blocks.any { it.x >= 1 }) {
242 for ((oldIndex, _, block) in hunk.steps()) {
243 if (block == null) {
244 add(" " + oldLines[oldIndex])
245 } else {
246 val marker = if (block.y >= 1) "! " else "- "
247 addAll(deleted(block).map { marker + it })
248 }
249 }
250 }
251 add("--- ${rangeR(hunk.newStart, hunk.newCount)} ----")
252 if (hunk.blocks.any { it.y >= 1 }) {
253 for ((_, newIndex, block) in hunk.steps()) {
254 if (block == null) {
255 add(" " + newLines[newIndex])
256 } else {
257 val marker = if (block.x >= 1) "! " else "+ "
258 addAll(inserted(block).map { marker + it })
259 }
260 }
261 }
262 }
263 }
264
265 fun sideBySide(width: Int): List<String> = buildList {
266 for ((oldIndex, newIndex, block) in walk(blocks, 0, 0, oldLines.size)) {
267 if (block == null) {
268 add(sideRow(oldLines[oldIndex], ' ', newLines[newIndex], width))
269 continue
270 }
271 for (row in 0 until maxOf(block.x, block.y)) {
272 val line = when {
273 row < block.x && row < block.y ->
274 sideRow(oldLines[block.p + row], '|', newLines[block.q + row], width)
275 row < block.x -> sideRow(oldLines[block.p + row], '<', "", width)
276 else -> sideRow("", '>', newLines[block.q + row], width)
277 }
278 add(line)
279 }
280 }
281 }
282}

293fun deletion(tokens: List<String>, separator: String) = tokens.joinToString(separator, "[-", "-]")
294
295fun insertion(tokens: List<String>, separator: String) = tokens.joinToString(separator, "{+", "+}")

297fun compareTokens(
298 oldTokens: List<String>,
299 newTokens: List<String>,
300 separator: String,
301 totals: TokenTotals,
302): String {
303 val parts = mutableListOf<String>()
304 for ((oldIndex, _, block) in walk(changeBlocks(oldTokens, newTokens), 0, 0, oldTokens.size)) {
305 if (block == null) {
306 parts.add(oldTokens[oldIndex])
307 totals.common++
308 continue
309 }
310 if (block.x >= 1) {
311 parts.add(deletion(oldTokens.subList(block.p, block.p + block.x), separator))
312 }
313 if (block.y >= 1) {
314 parts.add(insertion(newTokens.subList(block.q, block.q + block.y), separator))
315 }
316 totals.deleted += block.x
317 totals.inserted += block.y
318 }
319 return parts.joinToString(separator)
320}

344class HunkMatch(val fuzz: Int, val delta: Int, val position: Int, val lines: List<String>) {
345 fun report(number: Int): String = buildString {
346 append("Hunk #$number succeeded at ${position + 1}")
347 if (fuzz >= 1) append(" with fuzz $fuzz")
348 if (delta == 1 || delta == -1) {
349 append(" (offset $delta line)")
350 } else if (delta != 0) {
351 append(" (offset $delta lines)")
352 }
353 }
354}

356fun parseRange(text: String, sign: Char): LineRange? {
357 if (!text.startsWith(sign)) return null
358 val match = NUMBER_PAIR.matchEntire(text.substring(1)) ?: return null
359 val start = match.groupValues[1].toInt()
360 val count = match.groups[2]?.value?.toInt() ?: 1
361 if (start == 0 && count != 0) return null
362 return LineRange(start, count)
363}

365fun parseHunkHeader(line: String, lineNumber: Int): PatchHunk {
366 val parts = line.split(" ")
367 if (parts.size == 4 && parts[0] == "@@" && parts[3] == "@@") {
368 val oldRange = parseRange(parts[1], '-')
369 val newRange = parseRange(parts[2], '+')
370 if (oldRange != null && newRange != null) return PatchHunk(lineNumber, oldRange, newRange)
371 }
372 throw CommandError("PATCH ERROR line $lineNumber: bad hunk header")
373}
374
375fun parsePatch(patchLines: List<String>): List<PatchHunk> {
376 var index = patchLines.indexOfFirst { it.startsWith("@@") }
377 if (index < 0) throw CommandError("PATCH ERROR no hunks")
378 val hunks = mutableListOf<PatchHunk>()
379 while (index < patchLines.size) {
380 val hunk = parseHunkHeader(patchLines[index], index + 1)
381 index++
382 val oldCount = hunk.oldRange.count
383 val newCount = hunk.newRange.count
384 var oldSeen = 0
385 var newSeen = 0
386 while (oldSeen < oldCount || newSeen < newCount) {
387 if (index == patchLines.size) {
388 throw CommandError("PATCH ERROR line ${hunk.lineNumber}: truncated hunk")
389 }
390 val line = patchLines[index]
391 val kind = line[0]
392 val inOld = kind == ' ' || kind == '-'
393 val inNew = kind == ' ' || kind == '+'
394 if (line.length < 2 ||
395 !(inOld || inNew) ||
396 (inOld && oldSeen == oldCount) ||
397 (inNew && newSeen == newCount)
398 ) {
399 throw CommandError("PATCH ERROR line ${index + 1}: bad hunk line")
400 }
401 hunk.body.add(BodyLine(kind, line.substring(1)))
402 if (inOld) oldSeen++
403 if (inNew) newSeen++
404 index++
405 }
406 if (hunk.body.all { it.kind == ' ' }) {
407 throw CommandError("PATCH ERROR line ${hunk.lineNumber}: no changes")
408 }
409 hunks.add(hunk)
410 }
411 return hunks
412}

414fun searchHunk(
415 hunk: PatchHunk,
416 lines: List<String>,
417 adjustment: Int,
418 reverse: Boolean,
419 maxFuzz: Int,
420): HunkMatch? {
421 val (oldLines, newLines) = hunk.sequences(reverse)
422 val start = hunk.start(reverse)
423 val expected = if (oldLines.isEmpty()) start + adjustment else start - 1 + adjustment
424 val leading = hunk.leadingContext()
425 val trailing = hunk.trailingContext()
426 for (fuzz in 0..maxFuzz) {
427 val head = minOf(fuzz, leading)
428 val tail = minOf(fuzz, trailing)
429 val oldPart = oldLines.subList(head, oldLines.size - tail)
430 val newPart = newLines.subList(head, newLines.size - tail)
431 for (delta in OFFSETS) {
432 val position = expected + head + delta
433 val end = position + oldPart.size
434 if (position < 0 || end > lines.size) continue
435 if (lines.subList(position, end) == oldPart) {
436 val result = lines.subList(0, position) + newPart + lines.subList(end, lines.size)
437 return HunkMatch(fuzz, delta, position, result)
438 }
439 }
440 }
441 return null
442}

444data class EditCommand(val instruction: Char, val first: Int, val last: Int)
445
446fun parseEditCommand(line: String): EditCommand? {
447 val instruction = line.last()
448 val match = NUMBER_PAIR.matchEntire(line.dropLast(1))
449 if (instruction !in "acd" || match == null) return null
450 val second = match.groups[2]?.value?.toInt()
451 if (instruction == 'a' && second != null) return null
452 val first = match.groupValues[1].toInt()
453 return EditCommand(instruction, first, second ?: first)
454}

456fun runEditScript(original: List<String>, script: List<String>): Pair<List<String>, Int> {
457 val lines = original.toMutableList()
458 var executed = 0
459 var index = 0
460 while (index < script.size) {
461 val lineNumber = index + 1
462 val (instruction, first, last) = parseEditCommand(script[index])
463 ?: throw CommandError("EDIT ERROR line $lineNumber: bad command")
464 val valid = if (instruction == 'a') {
465 first <= lines.size
466 } else {
467 first in 1..last && last <= lines.size
468 }
469 if (!valid) throw CommandError("EDIT ERROR line $lineNumber: bad range")
470 index++
471 var text = emptyList<String>()
472 if (instruction == 'a' || instruction == 'c') {
473 var end = index
474 while (end < script.size && script[end] != ".") {
475 end++
476 }
477 if (end == script.size) {
478 throw CommandError("EDIT ERROR line $lineNumber: missing terminator")
479 }
480 text = script.subList(index, end)
481 index = end + 1
482 }
483 if (instruction == 'a') {
484 lines.addAll(first, text)
485 } else {
486 lines.subList(first - 1, last).clear()
487 lines.addAll(first - 1, text)
488 }
489 executed++
490 }
491 return lines to executed
492}

517fun matchTable(first: List<String>, second: List<String>): IntArray {
518 val matches = IntArray(first.size + 2)
519 val pairs = matchedPairs(first, second)
520 for ((i, j) in pairs.subList(1, pairs.size - 1)) {
521 matches[i] = j
522 }

526fun threeWayChunks(base: List<String>, ours: List<String>, theirs: List<String>): List<Chunk> {
527 val oursMatch = matchTable(base, ours)
528 val theirsMatch = matchTable(base, theirs)
529 val chunks = mutableListOf<Chunk>()
530 var x = 0
531 var y = 0
532 var z = 0
533 while (x < base.size || y < ours.size || z < theirs.size) {
534 if (x < base.size && oursMatch[x + 1] == y + 1 && theirsMatch[x + 1] == z + 1) {
535 val line = base.subList(x, x + 1)
536 chunks.add(Chunk(ChunkKind.STABLE, x, y, z, line, line, line))
537 x++
538 y++
539 z++
540 continue
541 }
542 var anchor = x + 1
543 while (anchor <= base.size && (oursMatch[anchor] == 0 || theirsMatch[anchor] == 0)) {
544 anchor++
545 }
546 val (nextX, nextY, nextZ) = if (anchor <= base.size) {
547 Triple(anchor - 1, oursMatch[anchor] - 1, theirsMatch[anchor] - 1)
548 } else {
549 Triple(base.size, ours.size, theirs.size)
550 }
551 val basePart = base.subList(x, nextX)
552 val oursPart = ours.subList(y, nextY)
553 val theirsPart = theirs.subList(z, nextZ)
554 val kind = chunkKind(basePart, oursPart, theirsPart)
555 chunks.add(Chunk(kind, x, y, z, basePart, oursPart, theirsPart))
556 x = nextX
557 y = nextY
558 z = nextZ
559 }
560 return chunks
561}

585fun conflictLines(
586 chunk: Chunk,
587 mode: String,
588 baseName: String,
589 oursName: String,
590 theirsName: String,
591): List<String> {
592 if (mode == "OURS") return chunk.ours
593 if (mode == "THEIRS") return chunk.theirs
594 if (mode == "DIFF3") {
595 return listOf("<<<<<<< $oursName") + chunk.ours + "||||||| $baseName" + chunk.base +
596 SEPARATOR + chunk.theirs + ">>>>>>> $theirsName"
597 }
598 val ours = chunk.ours
599 val theirs = chunk.theirs
600 val head = commonPrefixLength(ours, theirs)
601 val tail = commonSuffixLength(ours, theirs, minOf(ours.size, theirs.size) - head)
602 val oursMiddle = ours.subList(head, ours.size - tail)
603 val theirsMiddle = theirs.subList(head, theirs.size - tail)
604 val middle = if (mode == "UNION") {
605 oursMiddle + theirsMiddle
606 } else {
607 listOf("<<<<<<< $oursName") + oursMiddle + SEPARATOR + theirsMiddle + ">>>>>>> $theirsName"
608 }
609 return ours.subList(0, head) + middle + ours.subList(ours.size - tail, ours.size)
610}

612enum class Marker { START, BASE, SEPARATOR, END }
613
614private val MARKER_PREFIXES =
615 mapOf("<<<<<<<" to Marker.START, "|||||||" to Marker.BASE, ">>>>>>>" to Marker.END)

617fun markerKind(line: String): Marker? {
618 if (line == SEPARATOR) return Marker.SEPARATOR
619 val marker = MARKER_PREFIXES[line.take(7)]
620 return if (marker != null && (line.length == 7 || line[7] == ' ')) marker else null
621}

623fun nextMarker(lines: List<String>, index: Int): Int {
624 var next = index + 1
625 while (next < lines.size && markerKind(lines[next]) == null) {
626 next++
627 }

631class ConflictRegion(
632 val start: Int,
633 val end: Int,
634 val ours: List<String>,
635 val base: List<String>?,
636 val theirs: List<String>,
637) {
638 fun part(side: String): List<String> = when (side) {
639 "OURS" -> ours
640 "THEIRS" -> theirs
641 "BASE" -> base!!
642 else -> ours + theirs
643 }
644}
645
646fun regionAt(lines: List<String>, start: Int): ConflictRegion? {
647 if (markerKind(lines[start]) != Marker.START) return null
648 var middle = nextMarker(lines, start)
649 var baseMarker: Int? = null
650 if (middle < lines.size && markerKind(lines[middle]) == Marker.BASE) {
651 baseMarker = middle
652 middle = nextMarker(lines, middle)
653 }
654 if (middle == lines.size || markerKind(lines[middle]) != Marker.SEPARATOR) return null
655 val end = nextMarker(lines, middle)
656 if (end == lines.size || markerKind(lines[end]) != Marker.END) return null
657 val ours = lines.subList(start + 1, baseMarker ?: middle)
658 val base = baseMarker?.let { lines.subList(it + 1, middle) }
659 return ConflictRegion(start + 1, end + 1, ours, base, lines.subList(middle + 1, end))
660}
661
662fun conflictRegions(lines: List<String>): List<ConflictRegion> {
663 val regions = mutableListOf<ConflictRegion>()
664 var index = 0
665 while (index < lines.size) {
666 val region = regionAt(lines, index)
667 if (region == null) {
668 index++
669 } else {
670 regions.add(region)
671 index = region.end
672 }
673 }
674 return regions
675}

677class Workspace {
678 private val files = mutableMapOf<String, List<String>>()
679 private val history = mutableMapOf<String, MutableList<List<String>?>>()
680 val output = StringBuilder()
681
682 fun execute(words: List<String>, data: List<String>) {
683 val arguments = words.drop(1)
684 try {
685 when (words[0]) {
686 "FILE" -> loadFile(arguments, data)
687 "COPY" -> copy(arguments)
688 "PRINT" -> printFile(arguments)
689 "FILES" -> listFiles()
690 "DIFF" -> diff(arguments)
691 "WORDDIFF" -> tokenDiff(arguments, true)
692 "CHARDIFF" -> tokenDiff(arguments, false)
693 "STAT" -> stat(arguments)
694 "PATCH" -> patch(arguments, data)
695 "EDIT" -> edit(arguments, data)
696 "DIFF3" -> diff3(arguments)
697 "MERGE" -> merge(arguments)
698 "CONFLICTS" -> conflicts(arguments)
699 "RESOLVE" -> resolve(arguments)
700 "UNDO" -> undo(arguments)
701 }
702 } catch (error: CommandError) {
703 emit(error.message)
704 }
705 }
706
707 private fun emit(line: String) {
708 output.append(line).append('\n')
709 }
710
711 private fun emitAll(lines: List<String>) = lines.forEach(::emit)
712
713 private fun requireFiles(vararg names: String) {
714 for (name in names) {
715 if (name !in files) throw CommandError("ERROR no such file $name")
716 }
717 }
718
719 private fun save(name: String, lines: List<String>) {
720 history.getOrPut(name) { mutableListOf() }.add(files[name])
721 files[name] = lines.toList()
722 }
723
724 private fun loadFile(arguments: List<String>, data: List<String>) {
725 val name = arguments[0]
726 save(name, data)
727 emit("FILE $name lines=${data.size}")
728 }
729
730 private fun copy(arguments: List<String>) {
731 val (source, target) = arguments
732 requireFiles(source)
733 val lines = files.getValue(source)
734 save(target, lines)
735 emit("COPY $source $target lines=${lines.size}")
736 }
737
738 private fun printFile(arguments: List<String>) {
739 val name = arguments[0]
740 requireFiles(name)
741 val lines = files.getValue(name)
742 emit("PRINT $name lines=${lines.size}")
743 lines.forEachIndexed { index, line -> emit("${(index + 1).toString().padStart(3)}| $line") }
744 }
745
746 private fun listFiles() {
747 emit("FILES count=${files.size}")
748 for (name in files.keys.sorted()) {
749 emit("$name lines=${files.getValue(name).size} undo=${history.getValue(name).size}")
750 }
751 }
752
753 private fun undo(arguments: List<String>) {
754 val name = arguments[0]
755 val states = history[name]
756 if (states.isNullOrEmpty()) throw CommandError("ERROR nothing to undo $name")
757 val state = states.removeAt(states.lastIndex)
758 if (state == null) {
759 files.remove(name)
760 emit("UNDO $name removed")
761 } else {
762 files[name] = state
763 emit("UNDO $name lines=${state.size}")
764 }
765 }
766
767 private fun diff(arguments: List<String>) {
768 val (oldName, newName, form) = arguments
769 requireFiles(oldName, newName)
770 val hasParameter = form in FORMATS_WITH_ARGUMENT
771 val parameter = if (hasParameter) arguments[3] else ""
772 val options = arguments.drop(if (hasParameter) 4 else 3)
773 val diff = LineDiff(files.getValue(oldName), files.getValue(newName), options)
774 if (diff.blocks.isEmpty()) {
775 emit("Files $oldName and $newName are identical")
776 return
777 }
778 when (form) {
779 "BRIEF" -> emit("Files $oldName and $newName differ")
780 "NORMAL" -> emitAll(diff.normal())
781 "ED" -> emitAll(diff.ed())
782 "RCS" -> emitAll(diff.rcs())
783 "IFDEF" -> emitAll(diff.ifdef(parameter))
784 "UNIFIED" -> emitAll(diff.unified(parameter.toInt(), oldName, newName))
785 "CONTEXT" -> emitAll(diff.context(parameter.toInt(), oldName, newName))
786 else -> emitAll(diff.sideBySide(parameter.toInt()))
787 }
788 }
789
790 private fun tokenDiff(arguments: List<String>, byWord: Boolean) {
791 val (oldName, newName) = arguments
792 requireFiles(oldName, newName)
793 val oldLines = files.getValue(oldName)
794 val newLines = files.getValue(newName)
795 val diff = LineDiff(oldLines, newLines)
796 if (diff.blocks.isEmpty()) {
797 emit("Files $oldName and $newName are identical")
798 return
799 }
800 val separator = if (byWord) " " else ""
801 val totals = TokenTotals()
802 for (block in diff.blocks) {
803 for (row in 0 until maxOf(block.x, block.y)) {
804 val oldNumber = block.p + row + 1
805 val newNumber = block.q + row + 1
806 if (row < block.x && row < block.y) {
807 val oldTokens = splitTokens(oldLines[oldNumber - 1], byWord)
808 val newTokens = splitTokens(newLines[newNumber - 1], byWord)
809 val text = compareTokens(oldTokens, newTokens, separator, totals)
810 emit("a$oldNumber b$newNumber: $text")
811 } else if (row < block.x) {
812 val oldTokens = splitTokens(oldLines[oldNumber - 1], byWord)
813 totals.deleted += oldTokens.size
814 emit("a$oldNumber: ${deletion(oldTokens, separator)}")
815 } else {
816 val newTokens = splitTokens(newLines[newNumber - 1], byWord)
817 totals.inserted += newTokens.size
818 emit("b$newNumber: ${insertion(newTokens, separator)}")
819 }
820 }
821 }
822 val label = if (byWord) "WORDS" else "CHARS"
823 emit("$label deleted=${totals.deleted} inserted=${totals.inserted} common=${totals.common}")
824 }

826 private fun stat(arguments: List<String>) {
827 val (oldName, newName) = arguments
828 requireFiles(oldName, newName)
829 val oldLines = files.getValue(oldName)
830 val newLines = files.getValue(newName)
831 val diff = LineDiff(oldLines, newLines, arguments.drop(2))
832 val deleted = diff.blocks.sumOf { it.x }
833 val inserted = diff.blocks.sumOf { it.y }
834 val total = deleted + inserted
835 val unchanged = oldLines.size - deleted
836 val graph = if (total > 0) " " + statGraph(inserted, deleted) else ""
837 emit("$oldName -> $newName | $total$graph")
838 val lineTotal = oldLines.size + newLines.size
839 val similarity = if (lineTotal > 0) 200 * unchanged / lineTotal else 100
840 emit(
841 "changes=${diff.blocks.size} insertions=$inserted deletions=$deleted " +
842 "unchanged=$unchanged similarity=$similarity%",
843 )
844 }

846 private fun patch(arguments: List<String>, data: List<String>) {
847 val name = arguments[0]
848 requireFiles(name)
849 val options = arguments.drop(2)
850 val reverse = "REVERSE" in options
851 val dryRun = "DRYRUN" in options
852 val fuzzIndex = options.indexOf("FUZZ")
853 val maxFuzz = if (fuzzIndex >= 0) options[fuzzIndex + 1].toInt() else 0
854 val hunks = parsePatch(data)
855 emit((if (dryRun) "checking file " else "patching file ") + name)
856 var lines = files.getValue(name)
857 val first = hunks.first()
858 if (searchHunk(first, lines, 0, reverse, maxFuzz) == null &&
859 searchHunk(first, lines, 0, !reverse, maxFuzz) != null
860 ) {
861 emit("Reversed (or previously applied) patch detected! Skipping patch.")
862 emit("RESULT applied=0 failed=0 ignored=${hunks.size} lines=${lines.size}")
863 return
864 }
865 var adjustment = 0
866 var applied = 0
867 var failed = 0
868 for ((index, hunk) in hunks.withIndex()) {
869 val match = searchHunk(hunk, lines, adjustment, reverse, maxFuzz)
870 if (match == null) {
871 failed++
872 emit("Hunk #${index + 1} FAILED at ${hunk.start(reverse)}")
873 continue
874 }
875 val (oldLines, newLines) = hunk.sequences(reverse)
876 lines = match.lines
877 adjustment += match.delta + newLines.size - oldLines.size
878 applied++
879 emit(match.report(index + 1))
880 }
881 emit("RESULT applied=$applied failed=$failed ignored=0 lines=${lines.size}")
882 if (!dryRun && applied >= 1) save(name, lines)
883 }

893 private fun threeWay(baseName: String, oursName: String, theirsName: String): List<Chunk> {
894 requireFiles(baseName, oursName, theirsName)
895 return threeWayChunks(
896 files.getValue(baseName),
897 files.getValue(oursName),
898 files.getValue(theirsName),
899 )
900 }
901
902 private fun diff3(arguments: List<String>) {
903 val (baseName, oursName, theirsName) = arguments
904 val chunks = threeWay(baseName, oursName, theirsName)
905 val changes = chunks.filter { it.kind != ChunkKind.STABLE }
906 for (chunk in changes) {
907 emit("==== ${chunk.kind.name.lowercase()} ${chunk.location()}")
908 chunk.base.forEach { emit("O $it") }
909 if (chunk.kind != ChunkKind.B) chunk.ours.forEach { emit("A $it") }
910 if (chunk.kind == ChunkKind.B || chunk.kind == ChunkKind.CONFLICT) {
911 chunk.theirs.forEach { emit("B $it") }
912 }
913 }
914 emit("DIFF3 chunks=${changes.size} ${kindCounts(chunks)}")
915 }
916
917 private fun merge(arguments: List<String>) {
918 val (baseName, oursName, theirsName, target) = arguments
919 val mode = arguments.getOrElse(4) { "MARKERS" }
920 val chunks = threeWay(baseName, oursName, theirsName)
921 val result = mutableListOf<String>()
922 val conflicts = mutableListOf<Chunk>()
923 for (chunk in chunks) {
924 when (chunk.kind) {
925 ChunkKind.B -> result.addAll(chunk.theirs)
926 ChunkKind.CONFLICT -> {
927 result.addAll(conflictLines(chunk, mode, baseName, oursName, theirsName))
928 conflicts.add(chunk)
929 }
930 else -> result.addAll(chunk.ours)
931 }
932 }
933 emit("MERGE $target lines=${result.size} ${kindCounts(chunks)}")
934 conflicts.forEachIndexed { index, chunk ->
935 emit("CONFLICT ${index + 1} ${chunk.location()}")
936 }
937 save(target, result)
938 }

940 private fun conflicts(arguments: List<String>) {
941 val name = arguments[0]
942 requireFiles(name)
943 val regions = conflictRegions(files.getValue(name))
944 emit("CONFLICTS $name count=${regions.size}")
945 regions.forEachIndexed { index, region ->
946 val base = region.base?.size?.toString() ?: "none"
947 emit(
948 "${index + 1}: lines ${region.start}-${region.end} ours=${region.ours.size} " +
949 "base=$base theirs=${region.theirs.size}",
950 )
951 }
952 }
953
954 private fun resolve(arguments: List<String>) {
955 val (name, numberText, side) = arguments
956 requireFiles(name)
957 val number = numberText.toInt()
958 val lines = files.getValue(name)
959 val regions = conflictRegions(lines)
960 if (regions.size < number) throw CommandError("ERROR no conflict $number in $name")
961 val region = regions[number - 1]
962 if (side == "BASE" && region.base == null) {
963 throw CommandError("ERROR no base section in conflict $number")
964 }
965 val content = lines.subList(0, region.start - 1) + region.part(side) +
966 lines.subList(region.end, lines.size)
967 val remaining = conflictRegions(content).size
968 emit("RESOLVE $name $number $side lines=${content.size} remaining=$remaining")
969 save(name, content)
970 }
971}

973fun main() {
974 val lines = generateSequence(::readlnOrNull).toList()
975 val workspace = Workspace()
976 var position = 1
977 repeat(lines[0].toInt()) {
978 val words = lines[position].split(" ")
979 position++
980 var data = emptyList<String>()
981 if (words[0] in DATA_COMMANDS) {
982 val count = words[2].toInt()
983 data = lines.subList(position, position + count)
984 position += count
985 }
986 workspace.execute(words, data)
987 }
988 print(workspace.output)
989}
