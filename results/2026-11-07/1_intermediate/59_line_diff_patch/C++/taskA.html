65Lines split(const std::string& text, char delimiter) {
66 Lines parts;
67 std::size_t start = 0;
68 while (true) {
69 const std::size_t end = text.find(delimiter, start);
70 parts.push_back(text.substr(start, end - start));
71 if (end == std::string::npos) return parts;
72 start = end + 1;
73 }

113std::vector<std::pair<int, int>> matchedPairs(const Lines& first, const Lines& second) {
114 const int rows = length(first);
115 const int columns = length(second);
116 std::vector<std::vector<int>> table(rows + 1, std::vector<int>(columns + 1, 0));
117 for (int i = 1; i <= rows; ++i) {
118 for (int j = 1; j <= columns; ++j) {
119 if (first[i - 1] == second[j - 1]) {
120 table[i][j] = table[i - 1][j - 1] + 1;
121 } else {
122 table[i][j] = std::max(table[i - 1][j], table[i][j - 1]);
123 }
124 }
125 }
126 std::vector<std::pair<int, int>> pairs = {{rows + 1, columns + 1}};
127 int i = rows;
128 int j = columns;
129 while (i >= 1 && j >= 1) {
130 if (first[i - 1] == second[j - 1]) {
131 pairs.emplace_back(i, j);
132 --i;
133 --j;
134 } else if (table[i - 1][j] >= table[i][j - 1]) {
135 --i;
136 } else {
137 --j;
138 }
139 }
140 pairs.emplace_back(0, 0);
141 std::reverse(pairs.begin(), pairs.end());
142 return pairs;
143}

145struct ChangeBlock {
146 int p;
147 int q;
148 int x;
149 int y;
150
151 char letter() const {
152 if (x >= 1 && y >= 1) return 'c';
153 return y == 0 ? 'd' : 'a';
154 }
155};

157std::vector<ChangeBlock> changeBlocks(const Lines& first, const Lines& second) {
158 const std::vector<std::pair<int, int>> pairs = matchedPairs(first, second);
159 std::vector<ChangeBlock> blocks;
160 for (std::size_t k = 0; k + 1 < pairs.size(); ++k) {
161 const auto [i, j] = pairs[k];
162 const auto [nextI, nextJ] = pairs[k + 1];
163 if (nextI - i > 1 || nextJ - j > 1) {
164 blocks.push_back({i, j, nextI - i - 1, nextJ - j - 1});
165 }
166 }
167 return blocks;
168}

170struct Step {
171 int oldIndex;
172 int newIndex;
173 const ChangeBlock* block;
174};
175
176std::vector<Step> walk(const std::vector<ChangeBlock>& blocks, int oldStart, int newStart,
177 int oldEnd) {
178 std::vector<Step> steps;
179 int oldIndex = oldStart;
180 int newIndex = newStart;
181 for (const ChangeBlock& block : blocks) {
182 while (oldIndex < block.p) {
183 steps.push_back({oldIndex++, newIndex++, nullptr});
184 }
185 steps.push_back({oldIndex, newIndex, &block});
186 oldIndex = block.p + block.x;
187 newIndex = block.q + block.y;
188 }
189 while (oldIndex < oldEnd) {
190 steps.push_back({oldIndex++, newIndex++, nullptr});
191 }
192 return steps;
193}

195std::string sideRow(const std::string& left, char symbol, const std::string& right, int width) {
196 std::string row = left.substr(0, width);
197 row.resize(width, ' ');
198 row += concat(' ', symbol, ' ', right.substr(0, width));
199 row.erase(row.find_last_not_of(' ') + 1);
200 return row;
201}

203std::string statGraph(int inserted, int deleted) {
204 const int total = inserted + deleted;
205 int plus = inserted;
206 int minus = deleted;
207 if (total > kGraphWidth) {
208 plus = kGraphWidth * inserted / total;
209 minus = kGraphWidth * deleted / total;
210 if (inserted >= 1 && plus == 0) plus = 1;
211 if (deleted >= 1 && minus == 0) minus = 1;
212 }
213 return std::string(plus, '+') + std::string(minus, '-');
214}

216struct Hunk {
217 std::vector<ChangeBlock> blocks;
218 int oldStart;
219 int oldCount;
220 int newStart;
221 int newCount;
222
223 Hunk(std::vector<ChangeBlock> group, int context, int oldTotal) : blocks(std::move(group)) {
224 const ChangeBlock& first = blocks.front();
225 const ChangeBlock& last = blocks.back();
226 const int before = std::min(context, first.p);
227 const int after = std::min(context, oldTotal - last.p - last.x);
228 oldStart = first.p - before;
229 oldCount = last.p + last.x + after - oldStart;
230 newStart = first.q - before;
231 newCount = last.q + last.y + after - newStart;
232 }
233
234 std::vector<Step> steps() const {
235 return walk(blocks, oldStart, newStart, oldStart + oldCount);
236 }
237};

239class LineDiff {
240public:
241 const Lines oldLines;
242 const Lines newLines;
243 const std::vector<ChangeBlock> blocks;
244
245 LineDiff(const Lines& oldLines, const Lines& newLines, const Lines& options = {})
246 : oldLines(oldLines),
247 newLines(newLines),
248 blocks(
249 changeBlocks(comparisonKeys(oldLines, options), comparisonKeys(newLines, options))) {}
250
251 Lines deleted(const ChangeBlock& block) const {
252 return slice(oldLines, block.p, block.p + block.x);
253 }
254
255 Lines inserted(const ChangeBlock& block) const {
256 return slice(newLines, block.q, block.q + block.y);
257 }
258
259 std::vector<Hunk> hunks(int context) const {
260 std::vector<std::vector<ChangeBlock>> groups;
261 for (const ChangeBlock& block : blocks) {
262 if (!groups.empty()) {
263 const ChangeBlock& previous = groups.back().back();
264 if (block.p - (previous.p + previous.x) <= 2 * context) {
265 groups.back().push_back(block);
266 continue;
267 }
268 }
269 groups.push_back({block});
270 }
271 std::vector<Hunk> result;
272 for (std::vector<ChangeBlock>& group : groups) {
273 result.emplace_back(std::move(group), context, length(oldLines));
274 }
275 return result;
276 }
277
278 Lines normal() const {
279 Lines lines;
280 for (const ChangeBlock& block : blocks) {
281 const char letter = block.letter();
282 lines.push_back(rangeR(block.p, block.x) + letter + rangeR(block.q, block.y));
283 appendPrefixed(lines, "< ", deleted(block));
284 if (letter == 'c') lines.push_back("---");
285 appendPrefixed(lines, "> ", inserted(block));
286 }
287 return lines;
288 }
289
290 Lines ed() const {
291 Lines lines;
292 for (auto block = blocks.rbegin(); block != blocks.rend(); ++block) {
293 const char letter = block->letter();
294 lines.push_back(rangeR(block->p, block->x) + letter);
295 if (letter != 'd') {
296 append(lines, inserted(*block));
297 lines.push_back(".");
298 }
299 }
300 return lines;
301 }
302
303 Lines rcs() const {
304 Lines lines;
305 for (const ChangeBlock& block : blocks) {
306 if (block.x >= 1) lines.push_back(concat('d', block.p + 1, ' ', block.x));
307 if (block.y >= 1) {
308 lines.push_back(concat('a', block.p + block.x, ' ', block.y));
309 append(lines, inserted(block));
310 }
311 }
312 return lines;
313 }

315 Lines ifdef(const std::string& macro) const {
316 Lines lines;
317 for (const Step& step : walk(blocks, 0, 0, length(oldLines))) {
318 if (step.block == nullptr) {
319 lines.push_back(oldLines[step.oldIndex]);
320 continue;
321 }
322 const ChangeBlock& block = *step.block;
323 if (block.x >= 1 && block.y >= 1) {
324 lines.push_back("#ifndef " + macro);
325 append(lines, deleted(block));
326 lines.push_back("#else /* " + macro + " */");
327 append(lines, inserted(block));
328 lines.push_back("#endif /* " + macro + " */");
329 } else if (block.y == 0) {
330 lines.push_back("#ifndef " + macro);
331 append(lines, deleted(block));
332 lines.push_back("#endif /* ! " + macro + " */");
333 } else {
334 lines.push_back("#ifdef " + macro);
335 append(lines, inserted(block));
336 lines.push_back("#endif /* " + macro + " */");
337 }
338 }
339 return lines;
340 }

342 Lines unified(int context, const std::string& oldName, const std::string& newName) const {
343 Lines lines = {"--- " + oldName, "+++ " + newName};
344 for (const Hunk& hunk : hunks(context)) {
345 const std::string oldRange = rangeU(hunk.oldStart, hunk.oldCount);
346 const std::string newRange = rangeU(hunk.newStart, hunk.newCount);
347 lines.push_back("@@ -" + oldRange + " +" + newRange + " @@");
348 for (const Step& step : hunk.steps()) {
349 if (step.block == nullptr) {
350 lines.push_back(" " + oldLines[step.oldIndex]);
351 } else {
352 appendPrefixed(lines, "-", deleted(*step.block));
353 appendPrefixed(lines, "+", inserted(*step.block));
354 }
355 }
356 }
357 return lines;
358 }
359
360 Lines context(int context, const std::string& oldName, const std::string& newName) const {
361 Lines lines = {"*** " + oldName, "--- " + newName};
362 for (const Hunk& hunk : hunks(context)) {
363 lines.push_back(std::string(15, '*'));
364 lines.push_back("*** " + rangeR(hunk.oldStart, hunk.oldCount) + " ****");
365 const bool anyDeleted =
366 std::any_of(hunk.blocks.begin(), hunk.blocks.end(),
367 [](const ChangeBlock& block) { return block.x >= 1; });
368 if (anyDeleted) {
369 for (const Step& step : hunk.steps()) {
370 if (step.block == nullptr) {
371 lines.push_back(" " + oldLines[step.oldIndex]);
372 } else {
373 appendPrefixed(lines, step.block->y >= 1 ? "! " : "- ",
374 deleted(*step.block));
375 }
376 }
377 }
378 lines.push_back("--- " + rangeR(hunk.newStart, hunk.newCount) + " ----");
379 const bool anyInserted =
380 std::any_of(hunk.blocks.begin(), hunk.blocks.end(),
381 [](const ChangeBlock& block) { return block.y >= 1; });
382 if (anyInserted) {
383 for (const Step& step : hunk.steps()) {
384 if (step.block == nullptr) {
385 lines.push_back(" " + newLines[step.newIndex]);
386 } else {
387 appendPrefixed(lines, step.block->x >= 1 ? "! " : "+ ",
388 inserted(*step.block));
389 }
390 }
391 }
392 }
393 return lines;
394 }

396 Lines sideBySide(int width) const {
397 Lines lines;
398 for (const Step& step : walk(blocks, 0, 0, length(oldLines))) {
399 if (step.block == nullptr) {
400 lines.push_back(
401 sideRow(oldLines[step.oldIndex], ' ', newLines[step.newIndex], width));
402 continue;
403 }
404 const ChangeBlock& block = *step.block;
405 for (int row = 0; row < std::max(block.x, block.y); ++row) {
406 if (row < block.x && row < block.y) {
407 lines.push_back(
408 sideRow(oldLines[block.p + row], '|', newLines[block.q + row], width));
409 } else if (row < block.x) {
410 lines.push_back(sideRow(oldLines[block.p + row], '<', "", width));
411 } else {
412 lines.push_back(sideRow("", '>', newLines[block.q + row], width));
413 }
414 }
415 }
416 return lines;
417 }

426Lines splitTokens(const std::string& line, bool byWord) {
427 Lines tokens;
428 if (byWord) {
429 std::istringstream stream(line);
430 std::string word;
431 while (stream >> word) {
432 tokens.push_back(word);
433 }
434 } else {
435 for (char ch : line) {
436 tokens.emplace_back(1, ch);
437 }
438 }
439 return tokens;
440}

450std::string compareTokens(const Lines& oldTokens, const Lines& newTokens,
451 const std::string& separator, TokenTotals& totals) {
452 const std::vector<ChangeBlock> blocks = changeBlocks(oldTokens, newTokens);
453 Lines parts;
454 for (const Step& step : walk(blocks, 0, 0, length(oldTokens))) {
455 if (step.block == nullptr) {
456 parts.push_back(oldTokens[step.oldIndex]);
457 ++totals.common;
458 continue;
459 }
460 const ChangeBlock& block = *step.block;
461 if (block.x >= 1)
462 parts.push_back(deletion(slice(oldTokens, block.p, block.p + block.x), separator));
463 if (block.y >= 1)
464 parts.push_back(insertion(slice(newTokens, block.q, block.q + block.y), separator));
465 totals.deleted += block.x;
466 totals.inserted += block.y;
467 }
468 return join(parts, separator);
469}

476struct PatchHunk {
477 int lineNumber;
478 int oldStart;
479 int oldCount;
480 int newStart;
481 int newCount;
482 std::vector<BodyLine> body;
483
484 std::pair<Lines, Lines> sequences(bool reverse) const {
485 const char oldKind = reverse ? '+' : '-';
486 const char newKind = reverse ? '-' : '+';
487 Lines oldLines;
488 Lines newLines;
489 for (const BodyLine& line : body) {
490 if (line.kind == ' ' || line.kind == oldKind) oldLines.push_back(line.text);
491 if (line.kind == ' ' || line.kind == newKind) newLines.push_back(line.text);
492 }
493 return {oldLines, newLines};
494 }
495
496 int start(bool reverse) const { return reverse ? newStart : oldStart; }
497
498 int leadingContext() const {
499 int count = 0;
500 while (count < static_cast<int>(body.size()) && body[count].kind == ' ') {
501 ++count;
502 }
503 return count;
504 }
505
506 int trailingContext() const {
507 int count = 0;
508 while (count < static_cast<int>(body.size()) && body[body.size() - 1 - count].kind == ' ') {
509 ++count;
510 }
511 return count;
512 }

533std::optional<std::pair<int, int>> parseRange(const std::string& text, char sign) {
534 if (text.empty() || text[0] != sign) return std::nullopt;
535 const std::string numbers = text.substr(1);
536 std::smatch match;
537 if (!std::regex_match(numbers, match, kNumberPair)) return std::nullopt;
538 const int start = std::stoi(match[1].str());
539 const int count = match[2].matched ? std::stoi(match[2].str()) : 1;
540 if (start == 0 && count != 0) return std::nullopt;
541 return std::make_pair(start, count);
542}

544PatchHunk parseHunkHeader(const std::string& line, int lineNumber) {
545 const Lines parts = split(line, ' ');
546 std::optional<std::pair<int, int>> oldRange;
547 std::optional<std::pair<int, int>> newRange;
548 if (parts.size() == 4 && parts[0] == "@@" && parts[3] == "@@") {
549 oldRange = parseRange(parts[1], '-');
550 newRange = parseRange(parts[2], '+');
551 }
552 if (!oldRange || !newRange)
553 throw CommandError(concat("PATCH ERROR line ", lineNumber, ": bad hunk header"));
554 return {lineNumber, oldRange->first, oldRange->second, newRange->first, newRange->second, {}};
555}

557std::vector<PatchHunk> parsePatch(const Lines& patchLines) {
558 int index = 0;
559 while (index < length(patchLines) && patchLines[index].rfind("@@", 0) != 0) {
560 ++index;
561 }
562 if (index == length(patchLines)) throw CommandError("PATCH ERROR no hunks");
563 std::vector<PatchHunk> hunks;
564 while (index < length(patchLines)) {
565 PatchHunk hunk = parseHunkHeader(patchLines[index], index + 1);
566 ++index;
567 int oldSeen = 0;
568 int newSeen = 0;
569 while (oldSeen < hunk.oldCount || newSeen < hunk.newCount) {
570 if (index == length(patchLines)) {
571 throw CommandError(
572 concat("PATCH ERROR line ", hunk.lineNumber, ": truncated hunk"));
573 }
574 const std::string& line = patchLines[index];
575 const char kind = line[0];
576 const bool inOld = kind == ' ' || kind == '-';
577 const bool inNew = kind == ' ' || kind == '+';
578 if (line.size() < 2 || !(inOld || inNew) || (inOld && oldSeen == hunk.oldCount) ||
579 (inNew && newSeen == hunk.newCount)) {
580 throw CommandError(concat("PATCH ERROR line ", index + 1, ": bad hunk line"));
581 }
582 hunk.body.push_back({kind, line.substr(1)});
583 if (inOld) ++oldSeen;
584 if (inNew) ++newSeen;
585 ++index;
586 }
587 const bool onlyContext = std::all_of(hunk.body.begin(), hunk.body.end(),
588 [](const BodyLine& line) { return line.kind == ' '; });
589 if (onlyContext)
590 throw CommandError(concat("PATCH ERROR line ", hunk.lineNumber, ": no changes"));
591 hunks.push_back(std::move(hunk));
592 }
593 return hunks;
594}

596std::optional<HunkMatch> searchHunk(const PatchHunk& hunk, const Lines& lines, int adjustment,
597 bool reverse, int maxFuzz) {
598 const auto [oldLines, newLines] = hunk.sequences(reverse);
599 const int start = hunk.start(reverse);
600 const int expected = oldLines.empty() ? start + adjustment : start - 1 + adjustment;
601 const int leading = hunk.leadingContext();
602 const int trailing = hunk.trailingContext();
603 for (int fuzz = 0; fuzz <= maxFuzz; ++fuzz) {
604 const int head = std::min(fuzz, leading);
605 const int tail = std::min(fuzz, trailing);
606 const Lines oldPart = slice(oldLines, head, length(oldLines) - tail);
607 const Lines newPart = slice(newLines, head, length(newLines) - tail);
608 for (int delta : kOffsets) {
609 const int position = expected + head + delta;
610 const int end = position + length(oldPart);
611 if (position < 0 || end > length(lines)) continue;
612 if (std::equal(oldPart.begin(), oldPart.end(), lines.begin() + position)) {
613 Lines result = slice(lines, 0, position);
614 append(result, newPart);
615 append(result, slice(lines, end, length(lines)));
616 return HunkMatch{fuzz, delta, position, result};
617 }
618 }
619 }
620 return std::nullopt;
621}

629std::optional<EditCommand> parseEditCommand(const std::string& line) {
630 const char instruction = line.back();
631 const std::string address = line.substr(0, line.size() - 1);
632 std::smatch match;
633 if ((instruction != 'a' && instruction != 'c' && instruction != 'd') ||
634 !std::regex_match(address, match, kNumberPair)) {
635 return std::nullopt;
636 }
637 if (instruction == 'a' && match[2].matched) return std::nullopt;
638 const int first = std::stoi(match[1].str());
639 const int last = match[2].matched ? std::stoi(match[2].str()) : first;
640 return EditCommand{instruction, first, last};
641}

643std::pair<Lines, int> runEditScript(Lines lines, const Lines& script) {
644 int executed = 0;
645 int index = 0;
646 while (index < length(script)) {
647 const int lineNumber = index + 1;
648 const std::optional<EditCommand> command = parseEditCommand(script[index]);
649 if (!command) throw CommandError(concat("EDIT ERROR line ", lineNumber, ": bad command"));
650 const auto [instruction, first, last] = *command;
651 const bool valid = instruction == 'a'
652 ? first <= length(lines)
653 : 1 <= first && first <= last && last <= length(lines);
654 if (!valid) throw CommandError(concat("EDIT ERROR line ", lineNumber, ": bad range"));
655 ++index;
656 Lines text;
657 if (instruction == 'a' || instruction == 'c') {
658 int end = index;
659 while (end < length(script) && script[end] != ".") {
660 ++end;
661 }
662 if (end == length(script)) {
663 throw CommandError(concat("EDIT ERROR line ", lineNumber, ": missing terminator"));
664 }
665 text = slice(script, index, end);
666 index = end + 1;
667 }
668 if (instruction == 'a') {
669 lines.insert(lines.begin() + first, text.begin(), text.end());
670 } else {
671 lines.erase(lines.begin() + first - 1, lines.begin() + last);
672 lines.insert(lines.begin() + first - 1, text.begin(), text.end());
673 }
674 ++executed;
675 }
676 return {lines, executed};
677}

696struct Chunk {
697 ChunkKind kind;
698 int baseStart;
699 int oursStart;
700 int theirsStart;
701 Lines base;
702 Lines ours;
703 Lines theirs;
704
705 std::string location() const {
706 return "base " + rangeM(baseStart, length(base)) + " a " + rangeM(oursStart, length(ours)) +
707 " b " + rangeM(theirsStart, length(theirs));
708 }
709};
710
711ChunkKind chunkKind(const Lines& base, const Lines& ours, const Lines& theirs) {
712 if (ours == base) return ChunkKind::B;
713 if (theirs == base) return ChunkKind::A;
714 if (ours == theirs) return ChunkKind::Both;
715 return ChunkKind::Conflict;
716}
717
718std::vector<int> matchTable(const Lines& first, const Lines& second) {
719 std::vector<int> matches(first.size() + 2, 0);
720 const std::vector<std::pair<int, int>> pairs = matchedPairs(first, second);
721 for (std::size_t k = 1; k + 1 < pairs.size(); ++k) {
722 matches[pairs[k].first] = pairs[k].second;
723 }
724 return matches;
725}
726
727std::vector<Chunk> threeWayChunks(const Lines& base, const Lines& ours, const Lines& theirs) {
728 const std::vector<int> oursMatch = matchTable(base, ours);
729 const std::vector<int> theirsMatch = matchTable(base, theirs);
730 std::vector<Chunk> chunks;
731 int x = 0;
732 int y = 0;
733 int z = 0;
734 while (x < length(base) || y < length(ours) || z < length(theirs)) {
735 if (x < length(base) && oursMatch[x + 1] == y + 1 && theirsMatch[x + 1] == z + 1) {
736 chunks.push_back({ChunkKind::Stable, x, y, z, slice(base, x, x + 1),
737 slice(ours, y, y + 1), slice(theirs, z, z + 1)});
738 ++x;
739 ++y;
740 ++z;
741 continue;
742 }
743 int anchor = x + 1;
744 while (anchor <= length(base) && (oursMatch[anchor] == 0 || theirsMatch[anchor] == 0)) {
745 ++anchor;
746 }
747 int nextX = length(base);
748 int nextY = length(ours);
749 int nextZ = length(theirs);
750 if (anchor <= length(base)) {
751 nextX = anchor - 1;
752 nextY = oursMatch[anchor] - 1;
753 nextZ = theirsMatch[anchor] - 1;
754 }
755 Lines basePart = slice(base, x, nextX);
756 Lines oursPart = slice(ours, y, nextY);
757 Lines theirsPart = slice(theirs, z, nextZ);
758 const ChunkKind kind = chunkKind(basePart, oursPart, theirsPart);
759 chunks.push_back(
760 {kind, x, y, z, std::move(basePart), std::move(oursPart), std::move(theirsPart)});
761 x = nextX;
762 y = nextY;
763 z = nextZ;
764 }
765 return chunks;
766}
767
768std::string kindCounts(const std::vector<Chunk>& chunks) {
769 std::map<ChunkKind, int> counts;
770 for (const Chunk& chunk : chunks) {
771 ++counts[chunk.kind];
772 }
773 return concat("a=", counts[ChunkKind::A], " b=", counts[ChunkKind::B],
774 " both=", counts[ChunkKind::Both], " conflicts=", counts[ChunkKind::Conflict]);
775}

777int commonPrefixLength(const Lines& first, const Lines& second) {
778 int size = 0;
779 while (size < length(first) && size < length(second) && first[size] == second[size]) {
780 ++size;
781 }
782 return size;
783}
784
785int commonSuffixLength(const Lines& first, const Lines& second, int limit) {
786 int size = 0;
787 while (size < limit && first[first.size() - 1 - size] == second[second.size() - 1 - size]) {
788 ++size;
789 }
790 return size;
791}

793Lines conflictLines(const Chunk& chunk, const std::string& mode, const std::string& baseName,
794 const std::string& oursName, const std::string& theirsName) {
795 if (mode == "OURS") return chunk.ours;
796 if (mode == "THEIRS") return chunk.theirs;
797 Lines lines;
798 if (mode == "DIFF3") {
799 lines.push_back("<<<<<<< " + oursName);
800 append(lines, chunk.ours);
801 lines.push_back("||||||| " + baseName);
802 append(lines, chunk.base);
803 lines.push_back(kSeparator);
804 append(lines, chunk.theirs);
805 lines.push_back(">>>>>>> " + theirsName);
806 return lines;
807 }
808 const Lines& ours = chunk.ours;
809 const Lines& theirs = chunk.theirs;
810 const int head = commonPrefixLength(ours, theirs);
811 const int tail =
812 commonSuffixLength(ours, theirs, std::min(length(ours), length(theirs)) - head);
813 const Lines oursMiddle = slice(ours, head, length(ours) - tail);
814 const Lines theirsMiddle = slice(theirs, head, length(theirs) - tail);
815 lines = slice(ours, 0, head);
816 if (mode == "UNION") {
817 append(lines, oursMiddle);
818 append(lines, theirsMiddle);
819 } else {
820 lines.push_back("<<<<<<< " + oursName);
821 append(lines, oursMiddle);
822 lines.push_back(kSeparator);
823 append(lines, theirsMiddle);
824 lines.push_back(">>>>>>> " + theirsName);
825 }
826 append(lines, slice(ours, length(ours) - tail, length(ours)));
827 return lines;
828}

830enum class Marker { None, Start, Base, Separator, End };
831
832Marker markerKind(const std::string& line) {
833 static const std::map<std::string, Marker> prefixes = {
834 {"<<<<<<<", Marker::Start}, {"|||||||", Marker::Base}, {">>>>>>>", Marker::End}};
835 if (line == kSeparator) return Marker::Separator;
836 const auto found = prefixes.find(line.substr(0, 7));
837 if (found != prefixes.end() && (line.size() == 7 || line[7] == ' ')) return found->second;
838 return Marker::None;
839}
840
841int nextMarker(const Lines& lines, int index) {
842 ++index;
843 while (index < length(lines) && markerKind(lines[index]) == Marker::None) {
844 ++index;
845 }
846 return index;
847}
848
849struct ConflictRegion {
850 int start;
851 int end;
852 Lines ours;
853 std::optional<Lines> base;
854 Lines theirs;
855
856 Lines part(const std::string& side) const {
857 if (side == "OURS") return ours;
858 if (side == "THEIRS") return theirs;
859 if (side == "BASE") return *base;
860 Lines both = ours;
861 append(both, theirs);
862 return both;
863 }
864};
865
866std::optional<ConflictRegion> regionAt(const Lines& lines, int start) {
867 if (markerKind(lines[start]) != Marker::Start) return std::nullopt;
868 int middle = nextMarker(lines, start);
869 std::optional<int> baseMarker;
870 if (middle < length(lines) && markerKind(lines[middle]) == Marker::Base) {
871 baseMarker = middle;
872 middle = nextMarker(lines, middle);
873 }
874 if (middle == length(lines) || markerKind(lines[middle]) != Marker::Separator)
875 return std::nullopt;
876 const int end = nextMarker(lines, middle);
877 if (end == length(lines) || markerKind(lines[end]) != Marker::End) return std::nullopt;
878 ConflictRegion region{start + 1, end + 1, {}, std::nullopt, slice(lines, middle + 1, end)};
879 if (baseMarker) {
880 region.ours = slice(lines, start + 1, *baseMarker);
881 region.base = slice(lines, *baseMarker + 1, middle);
882 } else {
883 region.ours = slice(lines, start + 1, middle);
884 }
885 return region;
886}
887
888std::vector<ConflictRegion> conflictRegions(const Lines& lines) {
889 std::vector<ConflictRegion> regions;
890 int index = 0;
891 while (index < length(lines)) {
892 std::optional<ConflictRegion> region = regionAt(lines, index);
893 if (region) {
894 index = region->end;
895 regions.push_back(std::move(*region));
896 } else {
897 ++index;
898 }
899 }
900 return regions;
901}

904public:
905 void execute(const Lines& words, const Lines& data) {
906 const std::string& command = words[0];
907 const Lines arguments(words.begin() + 1, words.end());
908 try {
909 if (command == "FILE") {
910 loadFile(arguments, data);
911 } else if (command == "COPY") {
912 copy(arguments);
913 } else if (command == "PRINT") {
914 printFile(arguments);
915 } else if (command == "FILES") {
916 listFiles();
917 } else if (command == "DIFF") {
918 diff(arguments);
919 } else if (command == "WORDDIFF") {
920 tokenDiff(arguments, true);
921 } else if (command == "CHARDIFF") {
922 tokenDiff(arguments, false);
923 } else if (command == "STAT") {
924 stat(arguments);
925 } else if (command == "PATCH") {
926 patch(arguments, data);
927 } else if (command == "EDIT") {
928 edit(arguments, data);
929 } else if (command == "DIFF3") {
930 diff3(arguments);
931 } else if (command == "MERGE") {
932 merge(arguments);
933 } else if (command == "CONFLICTS") {
934 conflicts(arguments);
935 } else if (command == "RESOLVE") {
936 resolve(arguments);
937 } else if (command == "UNDO") {

1020 void diff(const Lines& arguments) {
1021 const std::string& oldName = arguments[0];
1022 const std::string& newName = arguments[1];
1023 const std::string& form = arguments[2];
1024 require({oldName, newName});
1025 const bool hasParameter = contains(kFormatsWithArgument, form);
1026 const std::string parameter = hasParameter ? arguments[3] : "";
1027 const Lines options(arguments.begin() + (hasParameter ? 4 : 3), arguments.end());
1028 const LineDiff diff(files_.at(oldName), files_.at(newName), options);
1029 if (diff.blocks.empty()) {
1030 emit("Files ", oldName, " and ", newName, " are identical");
1031 } else if (form == "BRIEF") {
1032 emit("Files ", oldName, " and ", newName, " differ");
1033 } else if (form == "NORMAL") {
1034 emitAll(diff.normal());
1035 } else if (form == "ED") {
1036 emitAll(diff.ed());
1037 } else if (form == "RCS") {
1038 emitAll(diff.rcs());
1039 } else if (form == "IFDEF") {
1040 emitAll(diff.ifdef(parameter));
1041 } else if (form == "UNIFIED") {
1042 emitAll(diff.unified(std::stoi(parameter), oldName, newName));
1043 } else if (form == "CONTEXT") {
1044 emitAll(diff.context(std::stoi(parameter), oldName, newName));
1045 } else {
1046 emitAll(diff.sideBySide(std::stoi(parameter)));
1047 }
1048 }

1050 void tokenDiff(const Lines& arguments, bool byWord) {
1051 const std::string& oldName = arguments[0];
1052 const std::string& newName = arguments[1];
1053 require({oldName, newName});
1054 const Lines& oldLines = files_.at(oldName);
1055 const Lines& newLines = files_.at(newName);
1056 const LineDiff diff(oldLines, newLines);
1057 if (diff.blocks.empty()) {
1058 emit("Files ", oldName, " and ", newName, " are identical");
1059 return;
1060 }
1061 const std::string separator = byWord ? " " : "";
1062 TokenTotals totals;
1063 for (const ChangeBlock& block : diff.blocks) {
1064 for (int row = 0; row < std::max(block.x, block.y); ++row) {
1065 const int oldNumber = block.p + row + 1;
1066 const int newNumber = block.q + row + 1;
1067 if (row < block.x && row < block.y) {
1068 const Lines oldTokens = splitTokens(oldLines[oldNumber - 1], byWord);
1069 const Lines newTokens = splitTokens(newLines[newNumber - 1], byWord);
1070 const std::string text = compareTokens(oldTokens, newTokens, separator, totals);
1071 emit('a', oldNumber, " b", newNumber, ": ", text);
1072 } else if (row < block.x) {
1073 const Lines oldTokens = splitTokens(oldLines[oldNumber - 1], byWord);
1074 totals.deleted += length(oldTokens);
1075 emit('a', oldNumber, ": ", deletion(oldTokens, separator));
1076 } else {
1077 const Lines newTokens = splitTokens(newLines[newNumber - 1], byWord);
1078 totals.inserted += length(newTokens);
1079 emit('b', newNumber, ": ", insertion(newTokens, separator));
1080 }
1081 }
1082 }
1083 emit(byWord ? "WORDS" : "CHARS", " deleted=", totals.deleted, " inserted=", totals.inserted,
1084 " common=", totals.common);
1085 }

1087 void stat(const Lines& arguments) {
1088 const std::string& oldName = arguments[0];
1089 const std::string& newName = arguments[1];
1090 require({oldName, newName});
1091 const Lines& oldLines = files_.at(oldName);
1092 const Lines& newLines = files_.at(newName);
1093 const LineDiff diff(oldLines, newLines, Lines(arguments.begin() + 2, arguments.end()));
1094 int deleted = 0;
1095 int inserted = 0;
1096 for (const ChangeBlock& block : diff.blocks) {
1097 deleted += block.x;
1098 inserted += block.y;
1099 }
1100 const int total = deleted + inserted;
1101 const int unchanged = length(oldLines) - deleted;
1102 std::string summary = concat(oldName, " -> ", newName, " | ", total);
1103 if (total > 0) summary += " " + statGraph(inserted, deleted);
1104 emit(summary);
1105 const int lineTotal = length(oldLines) + length(newLines);
1106 const int similarity = lineTotal > 0 ? 200 * unchanged / lineTotal : 100;
1107 emit("changes=", diff.blocks.size(), " insertions=", inserted, " deletions=", deleted,
1108 " unchanged=", unchanged, " similarity=", similarity, '%');
1109 }

1111 void patch(const Lines& arguments, const Lines& data) {
1112 const std::string& name = arguments[0];
1113 require({name});
1114 const Lines options(arguments.begin() + 2, arguments.end());
1115 const bool reverse = contains(options, "REVERSE");
1116 const bool dryRun = contains(options, "DRYRUN");
1117 const auto fuzzOption = std::find(options.begin(), options.end(), "FUZZ");
1118 const int maxFuzz = fuzzOption == options.end() ? 0 : std::stoi(*(fuzzOption + 1));
1119 const std::vector<PatchHunk> hunks = parsePatch(data);
1120 emit(dryRun ? "checking file " : "patching file ", name);
1121 Lines lines = files_.at(name);
1122 const PatchHunk& first = hunks.front();
1123 if (!searchHunk(first, lines, 0, reverse, maxFuzz) &&
1124 searchHunk(first, lines, 0, !reverse, maxFuzz)) {
1125 emit("Reversed (or previously applied) patch detected! Skipping patch.");
1126 emit("RESULT applied=0 failed=0 ignored=", hunks.size(), " lines=", lines.size());
1127 return;
1128 }
1129 int adjustment = 0;
1130 int applied = 0;
1131 int failed = 0;
1132 for (std::size_t index = 0; index < hunks.size(); ++index) {
1133 const PatchHunk& hunk = hunks[index];
1134 const std::optional<HunkMatch> match =
1135 searchHunk(hunk, lines, adjustment, reverse, maxFuzz);
1136 if (!match) {
1137 ++failed;
1138 emit("Hunk #", index + 1, " FAILED at ", hunk.start(reverse));
1139 continue;
1140 }
1141 const auto [oldLines, newLines] = hunk.sequences(reverse);
1142 lines = match->lines;
1143 adjustment += match->delta + length(newLines) - length(oldLines);
1144 ++applied;
1145 emit(match->report(static_cast<int>(index) + 1));
1146 }
1147 emit("RESULT applied=", applied, " failed=", failed, " ignored=0 lines=", lines.size());
1148 if (!dryRun && applied >= 1) save(name, lines);
1149 }

1165 void diff3(const Lines& arguments) {
1166 const std::vector<Chunk> chunks = threeWay(arguments[0], arguments[1], arguments[2]);
1167 int changes = 0;
1168 for (const Chunk& chunk : chunks) {
1169 if (chunk.kind == ChunkKind::Stable) continue;
1170 ++changes;
1171 emit("==== ", kindName(chunk.kind), ' ', chunk.location());
1172 for (const std::string& line : chunk.base) {
1173 emit("O ", line);
1174 }
1175 if (chunk.kind != ChunkKind::B) {
1176 for (const std::string& line : chunk.ours) {
1177 emit("A ", line);
1178 }
1179 }
1180 if (chunk.kind == ChunkKind::B || chunk.kind == ChunkKind::Conflict) {
1181 for (const std::string& line : chunk.theirs) {
1182 emit("B ", line);
1183 }
1184 }
1185 }
1186 emit("DIFF3 chunks=", changes, ' ', kindCounts(chunks));
1187 }

1189 void merge(const Lines& arguments) {
1190 const std::string& baseName = arguments[0];
1191 const std::string& oursName = arguments[1];
1192 const std::string& theirsName = arguments[2];
1193 const std::string& target = arguments[3];
1194 const std::string mode = arguments.size() == 5 ? arguments[4] : "MARKERS";
1195 const std::vector<Chunk> chunks = threeWay(baseName, oursName, theirsName);
1196 Lines result;
1197 std::vector<const Chunk*> conflicts;
1198 for (const Chunk& chunk : chunks) {
1199 if (chunk.kind == ChunkKind::B) {
1200 append(result, chunk.theirs);
1201 } else if (chunk.kind == ChunkKind::Conflict) {
1202 append(result, conflictLines(chunk, mode, baseName, oursName, theirsName));
1203 conflicts.push_back(&chunk);
1204 } else {
1205 append(result, chunk.ours);
1206 }
1207 }
1208 emit("MERGE ", target, " lines=", result.size(), ' ', kindCounts(chunks));
1209 for (std::size_t index = 0; index < conflicts.size(); ++index) {
1210 emit("CONFLICT ", index + 1, ' ', conflicts[index]->location());
1211 }
1212 save(target, result);
1213 }

1215 void conflicts(const Lines& arguments) {
1216 const std::string& name = arguments[0];
1217 require({name});
1218 const std::vector<ConflictRegion> regions = conflictRegions(files_.at(name));
1219 emit("CONFLICTS ", name, " count=", regions.size());
1220 for (std::size_t index = 0; index < regions.size(); ++index) {
1221 const ConflictRegion& region = regions[index];
1222 const std::string base = region.base ? std::to_string(region.base->size()) : "none";
1223 emit(index + 1, ": lines ", region.start, '-', region.end, " ours=", region.ours.size(),
1224 " base=", base, " theirs=", region.theirs.size());
1225 }
1226 }

1228 void resolve(const Lines& arguments) {
1229 const std::string& name = arguments[0];
1230 const std::string& side = arguments[2];
1231 require({name});
1232 const int number = std::stoi(arguments[1]);
1233 const Lines& lines = files_.at(name);
1234 const std::vector<ConflictRegion> regions = conflictRegions(lines);
1235 if (static_cast<int>(regions.size()) < number) {
1236 throw CommandError(concat("ERROR no conflict ", number, " in ", name));
1237 }
1238 const ConflictRegion& region = regions[number - 1];
1239 if (side == "BASE" && !region.base) {
1240 throw CommandError(concat("ERROR no base section in conflict ", number));
1241 }
1242 Lines content = slice(lines, 0, region.start - 1);
1243 append(content, region.part(side));
1244 append(content, slice(lines, region.end, length(lines)));
1245 const std::size_t remaining = conflictRegions(content).size();
1246 emit("RESOLVE ", name, ' ', number, ' ', side, " lines=", content.size(),
1247 " remaining=", remaining);
1248 save(name, content);
1249 }

1254 Lines input;
1255 std::string line;
1256 while (std::getline(std::cin, line)) {
1257 input.push_back(line);
1258 }
1259 Workspace workspace;
1260 std::size_t position = 1;
1261 const int commandCount = std::stoi(input[0]);
1262 for (int command = 0; command < commandCount; ++command) {
1263 const Lines words = split(input[position++], ' ');
1264 Lines data;
1265 if (contains(kDataCommands, words[0])) {
1266 const int count = std::stoi(words[2]);
1267 data.assign(input.begin() + position, input.begin() + position + count);
1268 position += count;
1269 }
1270 workspace.execute(words, data);
1271 }
1272 return 0;
