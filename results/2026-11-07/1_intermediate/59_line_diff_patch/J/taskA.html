801resolve =: 3 : 0
802 'name numberText side' =. y
803 lines =. contentOf name
804 number =. ". numberText
805 regions =. conflictRegions lines
806 if. number > # regions do. fail 'ERROR no conflict ' , (num number) , ' in ' , name end.
807 'g b h r' =. (number - 1) { regions
808 if. (side -: 'BASE') *. b < 0 do. fail 'ERROR no base section in conflict ' , num number end.
809 ours =. lines section (g + 1) , (b < 0) { b , h
810 theirs =. lines section (h + 1) , r
811 select. side
812 case. 'OURS' do. part =. ours
813 case. 'THEIRS' do. part =. theirs
814 case. 'BASE' do. part =. lines section (b + 1) , h
815 case. 'BOTH' do. part =. ours , theirs
816 end.
817 content =. (g {. lines) , part , (r + 1) }. lines
818 remaining =. # conflictRegions content
819 summary =. 'RESOLVE ' , name , ' ' , (num number) , ' ' , side
820 emit summary , ' lines=' , (num # content) , ' remaining=' , num remaining
