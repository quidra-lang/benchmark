67function matchedPairs(first: string[], second: string[]): [number, number][] {
68 const rows = first.length;
69 const columns = second.length;
70 const table = Array.from({ length: rows + 1 }, () => new Array<number>(columns + 1).fill(0));
71 for (let i = 1; i <= rows; i++) {
72 for (let j = 1; j <= columns; j++) {
73 if (first[i - 1] === second[j - 1]) {
74 table[i][j] = table[i - 1][j - 1] + 1;
75 } else {
76 table[i][j] = Math.max(table[i - 1][j], table[i][j - 1]);
77 }
78 }
79 }
80 const pairs: [number, number][] = [[rows + 1, columns + 1]];
81 let i = rows;
82 let j = columns;
83 while (i >= 1 && j >= 1) {
84 if (first[i - 1] === second[j - 1]) {
85 pairs.push([i, j]);
86 i--;
87 j--;
88 } else if (table[i - 1][j] >= table[i][j - 1]) {
89 i--;
90 } else {
91 j--;
92 }
93 }
94 pairs.push([0, 0]);
95 return pairs.reverse();
96}

98class ChangeBlock {
99 readonly p: number;
100 readonly q: number;
101 readonly x: number;
102 readonly y: number;
103
104 constructor(p: number, q: number, x: number, y: number) {
105 this.p = p;
106 this.q = q;
107 this.x = x;
108 this.y = y;
109 }
110
111 letter(): string {
112 if (this.x >= 1 && this.y >= 1) {
113 return 'c';
114 }
115 return this.y === 0 ? 'd' : 'a';
116 }

119function changeBlocks(first: string[], second: string[]): ChangeBlock[] {
120 const pairs = matchedPairs(first, second);
121 const blocks: ChangeBlock[] = [];
122 for (let index = 0; index + 1 < pairs.length; index++) {
123 const [i, j] = pairs[index];
124 const [nextI, nextJ] = pairs[index + 1];
125 if (nextI - i > 1 || nextJ - j > 1) {
126 blocks.push(new ChangeBlock(i, j, nextI - i - 1, nextJ - j - 1));
127 }
128 }
129 return blocks;
130}

132function walk(blocks: ChangeBlock[], oldStart: number, newStart: number, oldEnd: number): Step[] {
133 const steps: Step[] = [];
134 let oldIndex = oldStart;
135 let newIndex = newStart;
136 for (const block of blocks) {
137 while (oldIndex < block.p) {
138 steps.push({ oldIndex, newIndex, block: null });
139 oldIndex++;
140 newIndex++;
141 }
142 steps.push({ oldIndex, newIndex, block });
143 oldIndex = block.p + block.x;
144 newIndex = block.q + block.y;
145 }
146 while (oldIndex < oldEnd) {
147 steps.push({ oldIndex, newIndex, block: null });
148 oldIndex++;
149 newIndex++;
150 }
151 return steps;
152}

154function sideRow(left: string, symbol: string, right: string, width: number): string {
155 return `${left.slice(0, width).padEnd(width)} ${symbol} ${right.slice(0, width)}`.trimEnd();

158function statGraph(inserted: number, deleted: number): string {
159 const total = inserted + deleted;
160 let plus = inserted;
161 let minus = deleted;
162 if (total > GRAPH_WIDTH) {
163 plus = Math.floor((GRAPH_WIDTH * inserted) / total);
164 minus = Math.floor((GRAPH_WIDTH * deleted) / total);
165 if (inserted >= 1 && plus === 0) {
166 plus = 1;
167 }
168 if (deleted >= 1 && minus === 0) {
169 minus = 1;
170 }
171 }
172 return '+'.repeat(plus) + '-'.repeat(minus);
173}

175class Hunk {
176 readonly blocks: ChangeBlock[];
177 readonly oldStart: number;
178 readonly oldCount: number;
179 readonly newStart: number;
180 readonly newCount: number;
181
182 constructor(blocks: ChangeBlock[], context: number, oldTotal: number) {
183 const first = blocks[0];
184 const last = blocks[blocks.length - 1];
185 const before = Math.min(context, first.p);
186 const after = Math.min(context, oldTotal - last.p - last.x);
187 this.blocks = blocks;
188 this.oldStart = first.p - before;
189 this.oldCount = last.p + last.x + after - this.oldStart;
190 this.newStart = first.q - before;
191 this.newCount = last.q + last.y + after - this.newStart;
192 }
193
194 steps(): Step[] {
195 return walk(this.blocks, this.oldStart, this.newStart, this.oldStart + this.oldCount);
196 }

199class LineDiff {
200 readonly oldLines: string[];
201 readonly newLines: string[];
202 readonly blocks: ChangeBlock[];
203
204 constructor(oldLines: string[], newLines: string[], options: string[] = []) {
205 this.oldLines = oldLines;
206 this.newLines = newLines;
207 this.blocks = changeBlocks(
208 oldLines.map((line) => comparisonKey(line, options)),
209 newLines.map((line) => comparisonKey(line, options)),
210 );
211 }
212
213 deleted(block: ChangeBlock): string[] {
214 return this.oldLines.slice(block.p, block.p + block.x);
215 }
216
217 inserted(block: ChangeBlock): string[] {
218 return this.newLines.slice(block.q, block.q + block.y);
219 }
220
221 hunks(context: number): Hunk[] {
222 const groups: ChangeBlock[][] = [];
223 for (const block of this.blocks) {
224 const group = groups.at(-1);
225 if (group !== undefined) {
226 const previous = group[group.length - 1];
227 if (block.p - (previous.p + previous.x) <= 2 * context) {
228 group.push(block);
229 continue;
230 }
231 }
232 groups.push([block]);
233 }
234 return groups.map((group) => new Hunk(group, context, this.oldLines.length));
235 }
236
237 normal(): string[] {
238 const lines: string[] = [];
239 for (const block of this.blocks) {
240 const letter = block.letter();
241 lines.push(rangeR(block.p, block.x) + letter + rangeR(block.q, block.y));
242 lines.push(...this.deleted(block).map((line) => `< ${line}`));
243 if (letter === 'c') {
244 lines.push('---');
245 }
246 lines.push(...this.inserted(block).map((line) => `> ${line}`));
247 }
248 return lines;
249 }
250
251 ed(): string[] {
252 const lines: string[] = [];
253 for (const block of this.blocks.toReversed()) {
254 const letter = block.letter();
255 lines.push(rangeR(block.p, block.x) + letter);
256 if (letter !== 'd') {
257 lines.push(...this.inserted(block), '.');
258 }
259 }
260 return lines;
261 }
262
263 rcs(): string[] {
264 const lines: string[] = [];
265 for (const block of this.blocks) {
266 if (block.x >= 1) {
267 lines.push(`d${block.p + 1} ${block.x}`);
268 }
269 if (block.y >= 1) {
270 lines.push(`a${block.p + block.x} ${block.y}`, ...this.inserted(block));
271 }
272 }
273 return lines;
274 }

276 ifdef(macro: string): string[] {
277 const lines: string[] = [];
278 for (const { oldIndex, block } of walk(this.blocks, 0, 0, this.oldLines.length)) {
279 if (block === null) {
280 lines.push(this.oldLines[oldIndex]);
281 } else if (block.x >= 1 && block.y >= 1) {
282 lines.push(
283 `#ifndef ${macro}`,
284 ...this.deleted(block),
285 `#else /* ${macro} */`,
286 ...this.inserted(block),
287 `#endif /* ${macro} */`,
288 );
289 } else if (block.y === 0) {
290 lines.push(`#ifndef ${macro}`, ...this.deleted(block), `#endif /* ! ${macro} */`);
291 } else {
292 lines.push(`#ifdef ${macro}`, ...this.inserted(block), `#endif /* ${macro} */`);
293 }
294 }
295 return lines;
296 }

298 unified(context: number, oldName: string, newName: string): string[] {
299 const lines = [`--- ${oldName}`, `+++ ${newName}`];
300 for (const hunk of this.hunks(context)) {
301 const oldRange = rangeU(hunk.oldStart, hunk.oldCount);
302 const newRange = rangeU(hunk.newStart, hunk.newCount);
303 lines.push(`@@ -${oldRange} +${newRange} @@`);
304 for (const { oldIndex, block } of hunk.steps()) {
305 if (block === null) {
306 lines.push(` ${this.oldLines[oldIndex]}`);
307 } else {
308 lines.push(...this.deleted(block).map((line) => `-${line}`));
309 lines.push(...this.inserted(block).map((line) => `+${line}`));
310 }
311 }
312 }
313 return lines;
314 }
315
316 context(context: number, oldName: string, newName: string): string[] {
317 const lines = [`*** ${oldName}`, `--- ${newName}`];
318 for (const hunk of this.hunks(context)) {
319 lines.push('*'.repeat(15), `*** ${rangeR(hunk.oldStart, hunk.oldCount)} ****`);
320 if (hunk.blocks.some((block) => block.x >= 1)) {
321 for (const { oldIndex, block } of hunk.steps()) {
322 if (block === null) {
323 lines.push(` ${this.oldLines[oldIndex]}`);
324 } else {
325 const marker = block.y >= 1 ? '! ' : '- ';
326 lines.push(...this.deleted(block).map((line) => marker + line));
327 }
328 }
329 }
330 lines.push(`--- ${rangeR(hunk.newStart, hunk.newCount)} ----`);
331 if (hunk.blocks.some((block) => block.y >= 1)) {
332 for (const { newIndex, block } of hunk.steps()) {
333 if (block === null) {
334 lines.push(` ${this.newLines[newIndex]}`);
335 } else {
336 const marker = block.x >= 1 ? '! ' : '+ ';
337 lines.push(...this.inserted(block).map((line) => marker + line));
338 }
339 }
340 }
341 }
342 return lines;
343 }
344
345 sideBySide(width: number): string[] {
346 const lines: string[] = [];
347 for (const { oldIndex, newIndex, block } of walk(this.blocks, 0, 0, this.oldLines.length)) {
348 if (block === null) {
349 lines.push(sideRow(this.oldLines[oldIndex], ' ', this.newLines[newIndex], width));
350 continue;
351 }
352 for (let row = 0; row < Math.max(block.x, block.y); row++) {
353 const left = row < block.x ? this.oldLines[block.p + row] : '';
354 const right = row < block.y ? this.newLines[block.q + row] : '';
355 let symbol = '>';
356 if (row < block.x && row < block.y) {
357 symbol = '|';
358 } else if (row < block.x) {
359 symbol = '<';
360 }
361 lines.push(sideRow(left, symbol, right, width));
362 }
363 }
364 return lines;
365 }

382function insertion(tokens: string[], separator: string): string {
383 return `{+${tokens.join(separator)}+}`;

386function compareTokens(
387 oldTokens: string[],
388 newTokens: string[],
389 separator: string,
390 totals: TokenTotals,
391): string {
392 const parts: string[] = [];
393 const blocks = changeBlocks(oldTokens, newTokens);
394 for (const { oldIndex, block } of walk(blocks, 0, 0, oldTokens.length)) {
395 if (block === null) {
396 parts.push(oldTokens[oldIndex]);
397 totals.common++;
398 continue;
399 }
400 if (block.x >= 1) {
401 parts.push(deletion(oldTokens.slice(block.p, block.p + block.x), separator));
402 }
403 if (block.y >= 1) {
404 parts.push(insertion(newTokens.slice(block.q, block.q + block.y), separator));
405 }
406 totals.deleted += block.x;
407 totals.inserted += block.y;
408 }
409 return parts.join(separator);
410}

412class PatchHunk {
413 readonly lineNumber: number;
414 readonly oldStart: number;
415 readonly oldCount: number;
416 readonly newStart: number;
417 readonly newCount: number;
418 readonly body: BodyLine[] = [];
419
420 constructor(
421 lineNumber: number,
422 oldStart: number,
423 oldCount: number,
424 newStart: number,
425 newCount: number,
426 ) {
427 this.lineNumber = lineNumber;
428 this.oldStart = oldStart;
429 this.oldCount = oldCount;
430 this.newStart = newStart;
431 this.newCount = newCount;
432 }
433
434 sequences(reverse: boolean): [string[], string[]] {
435 const oldKinds = reverse ? ' +' : ' -';
436 const newKinds = reverse ? ' -' : ' +';
437 const oldLines = this.body.filter(({ kind }) => oldKinds.includes(kind));
438 const newLines = this.body.filter(({ kind }) => newKinds.includes(kind));
439 return [oldLines.map(({ text }) => text), newLines.map(({ text }) => text)];
440 }
441
442 start(reverse: boolean): number {
443 return reverse ? this.newStart : this.oldStart;
444 }
445
446 leadingContext(): number {
447 let count = 0;
448 while (count < this.body.length && this.body[count].kind === ' ') {
449 count++;
450 }
451 return count;
452 }
453
454 trailingContext(): number {
455 let count = 0;
456 while (count < this.body.length && this.body[this.body.length - 1 - count].kind === ' ') {
457 count++;
458 }
459 return count;
460 }
461}

463class HunkMatch {
464 readonly fuzz: number;
465 readonly delta: number;
466 readonly position: number;
467 readonly lines: string[];
468
469 constructor(fuzz: number, delta: number, position: number, lines: string[]) {
470 this.fuzz = fuzz;
471 this.delta = delta;

478 if (this.fuzz >= 1) {
479 text += ` with fuzz ${this.fuzz}`;
480 }
481 if (Math.abs(this.delta) === 1) {

490function parseRange(text: string, sign: string): Range | null {
491 if (!text.startsWith(sign)) {
492 return null;
493 }
494 const match = NUMBER_PAIR.exec(text.slice(1));
495 if (match === null) {
496 return null;
497 }
498 const start = Number(match[1]);
499 const count = match[2] === undefined ? 1 : Number(match[2]);
500 if (start === 0 && count !== 0) {
501 return null;
502 }
503 return { start, count };
504}

506function parseHunkHeader(line: string, lineNumber: number): PatchHunk {
507 const parts = line.split(' ');
508 let oldRange: Range | null = null;
509 let newRange: Range | null = null;
510 if (parts.length === 4 && parts[0] === '@@' && parts[3] === '@@') {
511 oldRange = parseRange(parts[1], '-');
512 newRange = parseRange(parts[2], '+');
513 }
514 if (oldRange === null || newRange === null) {
515 throw new CommandError(`PATCH ERROR line ${lineNumber}: bad hunk header`);
516 }
517 return new PatchHunk(lineNumber, oldRange.start, oldRange.count, newRange.start, newRange.count);
518}

520function parsePatch(patchLines: string[]): PatchHunk[] {
521 let index = patchLines.findIndex((line) => line.startsWith('@@'));
522 if (index === -1) {
523 throw new CommandError('PATCH ERROR no hunks');
524 }
525 const hunks: PatchHunk[] = [];
526 while (index < patchLines.length) {
527 const hunk = parseHunkHeader(patchLines[index], index + 1);
528 index++;
529 let oldSeen = 0;
530 let newSeen = 0;
531 while (oldSeen < hunk.oldCount || newSeen < hunk.newCount) {
532 if (index === patchLines.length) {
533 throw new CommandError(`PATCH ERROR line ${hunk.lineNumber}: truncated hunk`);
534 }
535 const line = patchLines[index];
536 const kind = line[0];
537 const inOld = kind === ' ' || kind === '-';
538 const inNew = kind === ' ' || kind === '+';
539 if (
540 line.length < 2 ||
541 !(inOld || inNew) ||
542 (inOld && oldSeen === hunk.oldCount) ||
543 (inNew && newSeen === hunk.newCount)
544 ) {
545 throw new CommandError(`PATCH ERROR line ${index + 1}: bad hunk line`);
546 }
547 hunk.body.push({ kind, text: line.slice(1) });
548 if (inOld) {
549 oldSeen++;
550 }
551 if (inNew) {
552 newSeen++;
553 }
554 index++;
555 }
556 if (hunk.body.every(({ kind }) => kind === ' ')) {
557 throw new CommandError(`PATCH ERROR line ${hunk.lineNumber}: no changes`);
558 }
559 hunks.push(hunk);
560 }
561 return hunks;
562}

564function searchHunk(
565 hunk: PatchHunk,
566 lines: string[],
567 adjustment: number,
568 reverse: boolean,
569 maxFuzz: number,
570): HunkMatch | null {
571 const [oldLines, newLines] = hunk.sequences(reverse);
572 const start = hunk.start(reverse);
573 const expected = oldLines.length > 0 ? start - 1 + adjustment : start + adjustment;
574 const leading = hunk.leadingContext();
575 const trailing = hunk.trailingContext();
576 for (let fuzz = 0; fuzz <= maxFuzz; fuzz++) {
577 const head = Math.min(fuzz, leading);
578 const tail = Math.min(fuzz, trailing);
579 const oldPart = oldLines.slice(head, oldLines.length - tail);
580 const newPart = newLines.slice(head, newLines.length - tail);
581 for (const delta of OFFSETS) {
582 const position = expected + head + delta;
583 const end = position + oldPart.length;
584 if (position < 0 || end > lines.length) {
585 continue;
586 }
587 if (equalLines(lines.slice(position, end), oldPart)) {
588 const result = [...lines.slice(0, position), ...newPart, ...lines.slice(end)];
589 return new HunkMatch(fuzz, delta, position, result);
590 }
591 }
592 }
593 return null;
594}

596function parseEditCommand(line: string): EditCommand | null {
597 const instruction = line[line.length - 1];
598 const match = NUMBER_PAIR.exec(line.slice(0, -1));
599 if (!['a', 'c', 'd'].includes(instruction) || match === null) {
600 return null;
601 }
602 if (instruction === 'a' && match[2] !== undefined) {
603 return null;
604 }
605 const first = Number(match[1]);
606 const last = match[2] === undefined ? first : Number(match[2]);
607 return { instruction, first, last };
608}

610function runEditScript(
611 original: string[],
612 script: string[],
613): { lines: string[]; executed: number } {
614 const lines = [...original];
615 let executed = 0;
616 let index = 0;
617 while (index < script.length) {
618 const lineNumber = index + 1;
619 const command = parseEditCommand(script[index]);
620 if (command === null) {
621 throw new CommandError(`EDIT ERROR line ${lineNumber}: bad command`);
622 }
623 const { instruction, first, last } = command;
624 const valid =
625 instruction === 'a'
626 ? first <= lines.length
627 : 1 <= first && first <= last && last <= lines.length;
628 if (!valid) {
629 throw new CommandError(`EDIT ERROR line ${lineNumber}: bad range`);
630 }
631 index++;
632 let text: string[] = [];
633 if (instruction === 'a' || instruction === 'c') {
634 let end = index;
635 while (end < script.length && script[end] !== '.') {
636 end++;
637 }
638 if (end === script.length) {
639 throw new CommandError(`EDIT ERROR line ${lineNumber}: missing terminator`);
640 }
641 text = script.slice(index, end);
642 index = end + 1;
643 }
644 if (instruction === 'a') {
645 lines.splice(first, 0, ...text);
646 } else if (instruction === 'c') {
647 lines.splice(first - 1, last - first + 1, ...text);
648 } else {
649 lines.splice(first - 1, last - first + 1);
650 }
651 executed++;
652 }
653 return { lines, executed };
654}

692function chunkKind(base: string[], ours: string[], theirs: string[]): ChunkKind {
693 if (equalLines(ours, base)) {
694 return 'b';
695 }

706 const matches = new Array<number>(first.length + 2).fill(0);
707 for (const [i, j] of matchedPairs(first, second).slice(1, -1)) {

713function threeWayChunks(base: string[], ours: string[], theirs: string[]): Chunk[] {
714 const oursMatch = matchTable(base, ours);
715 const theirsMatch = matchTable(base, theirs);
716 const chunks: Chunk[] = [];
717 let x = 0;
718 let y = 0;
719 let z = 0;
720 while (x < base.length || y < ours.length || z < theirs.length) {
721 if (x < base.length && oursMatch[x + 1] === y + 1 && theirsMatch[x + 1] === z + 1) {
722 const [baseLine, oursLine, theirsLine] = [
723 base.slice(x, x + 1),
724 ours.slice(y, y + 1),
725 theirs.slice(z, z + 1),
726 ];
727 chunks.push(new Chunk('stable', x, y, z, baseLine, oursLine, theirsLine));
728 x++;
729 y++;
730 z++;
731 continue;
732 }
733 let anchor = x + 1;
734 while (anchor <= base.length && (oursMatch[anchor] === 0 || theirsMatch[anchor] === 0)) {
735 anchor++;
736 }
737 const [nextX, nextY, nextZ] =
738 anchor <= base.length
739 ? [anchor - 1, oursMatch[anchor] - 1, theirsMatch[anchor] - 1]
740 : [base.length, ours.length, theirs.length];
741 const basePart = base.slice(x, nextX);
742 const oursPart = ours.slice(y, nextY);
743 const theirsPart = theirs.slice(z, nextZ);
744 const kind = chunkKind(basePart, oursPart, theirsPart);
745 chunks.push(new Chunk(kind, x, y, z, basePart, oursPart, theirsPart));
746 [x, y, z] = [nextX, nextY, nextZ];
747 }
748 return chunks;
749}

751function kindCounts(chunks: Chunk[]): string {
752 const counts: Record<string, number> = { a: 0, b: 0, both: 0, conflict: 0 };
753 for (const chunk of chunks) {
754 if (chunk.kind !== 'stable') {
755 counts[chunk.kind]++;
756 }
757 }
758 return `a=${counts.a} b=${counts.b} both=${counts.both} conflicts=${counts.conflict}`;
759}

761function commonPrefixLength(first: string[], second: string[]): number {
762 let length = 0;
763 while (length < first.length && length < second.length && first[length] === second[length]) {
764 length++;
765 }
766 return length;
767}

769function commonSuffixLength(first: string[], second: string[], limit: number): number {
770 let length = 0;
771 while (
772 length < limit &&
773 first[first.length - 1 - length] === second[second.length - 1 - length]
774 ) {
775 length++;
776 }
777 return length;
778}

780function conflictLines(
781 chunk: Chunk,
782 mode: string,
783 baseName: string,
784 oursName: string,
785 theirsName: string,
786): string[] {
787 const { base, ours, theirs } = chunk;
788 if (mode === 'OURS') {
789 return ours;
790 }
791 if (mode === 'THEIRS') {
792 return theirs;
793 }
794 if (mode === 'DIFF3') {
795 return [
796 `<<<<<<< ${oursName}`,
797 ...ours,
798 `||||||| ${baseName}`,
799 ...base,
800 SEPARATOR,
801 ...theirs,
802 `>>>>>>> ${theirsName}`,
803 ];
804 }
805 const head = commonPrefixLength(ours, theirs);
806 const tail = commonSuffixLength(ours, theirs, Math.min(ours.length, theirs.length) - head);
807 const oursMiddle = ours.slice(head, ours.length - tail);
808 const theirsMiddle = theirs.slice(head, theirs.length - tail);
809 const middle =
810 mode === 'UNION'
811 ? [...oursMiddle, ...theirsMiddle]
812 : [
813 `<<<<<<< ${oursName}`,
814 ...oursMiddle,
815 SEPARATOR,
816 ...theirsMiddle,
817 `>>>>>>> ${theirsName}`,
818 ];
819 return [...ours.slice(0, head), ...middle, ...ours.slice(ours.length - tail)];
820}

822function markerKind(line: string): MarkerKind | null {
823 if (line === SEPARATOR) {
824 return 'separator';
825 }
826 const kind = MARKER_PREFIXES.get(line.slice(0, 7));
827 if (kind !== undefined && (line.length === 7 || line[7] === ' ')) {
828 return kind;
829 }
830 return null;
831}
832
833function nextMarker(lines: string[], index: number): number {
834 let next = index + 1;
835 while (next < lines.length && markerKind(lines[next]) === null) {
836 next++;
837 }
838 return next;
839}

841class ConflictRegion {
842 readonly start: number;
843 readonly end: number;
844 readonly ours: string[];
845 readonly base: string[] | null;
846 readonly theirs: string[];
847
848 constructor(start: number, end: number, ours: string[], base: string[] | null, theirs: string[]) {
849 this.start = start;
850 this.end = end;
851 this.ours = ours;
852 this.base = base;
853 this.theirs = theirs;
854 }
855
856 part(side: string): string[] {
857 if (side === 'OURS') {
858 return this.ours;
859 }
860 if (side === 'THEIRS') {
861 return this.theirs;
862 }
863 if (side === 'BASE') {
864 return this.base ?? [];
865 }
866 return [...this.ours, ...this.theirs];
867 }
868}
869
870function regionAt(lines: string[], start: number): ConflictRegion | null {
871 if (markerKind(lines[start]) !== 'start') {
872 return null;
873 }
874 let middle = nextMarker(lines, start);
875 let baseMarker: number | null = null;
876 if (middle < lines.length && markerKind(lines[middle]) === 'base') {
877 baseMarker = middle;
878 middle = nextMarker(lines, middle);
879 }
880 if (middle === lines.length || markerKind(lines[middle]) !== 'separator') {
881 return null;
882 }
883 const end = nextMarker(lines, middle);
884 if (end === lines.length || markerKind(lines[end]) !== 'end') {
885 return null;
886 }
887 const theirs = lines.slice(middle + 1, end);
888 if (baseMarker === null) {
889 return new ConflictRegion(start + 1, end + 1, lines.slice(start + 1, middle), null, theirs);
890 }
891 const ours = lines.slice(start + 1, baseMarker);
892 const base = lines.slice(baseMarker + 1, middle);
893 return new ConflictRegion(start + 1, end + 1, ours, base, theirs);
894}
895
896function conflictRegions(lines: string[]): ConflictRegion[] {
897 const regions: ConflictRegion[] = [];
898 let index = 0;
899 while (index < lines.length) {
900 const region = regionAt(lines, index);
901 if (region === null) {
902 index++;
903 } else {
904 regions.push(region);
905 index = region.end;
906 }
907 }
908 return regions;
909}

911class Workspace {
912 readonly output: string[] = [];

995 private diff(args: string[]): void {
996 const [oldName, newName, form] = args;
997 const [oldLines, newLines] = this.contents(oldName, newName);
998 const hasParameter = FORMATS_WITH_ARGUMENT.has(form);
999 const parameter = hasParameter ? args[3] : '';
1000 const diff = new LineDiff(oldLines, newLines, args.slice(hasParameter ? 4 : 3));
1001 if (diff.blocks.length === 0) {
1002 this.emit(`Files ${oldName} and ${newName} are identical`);
1003 return;
1004 }
1005 switch (form) {
1006 case 'BRIEF':
1007 this.emit(`Files ${oldName} and ${newName} differ`);
1008 break;
1009 case 'NORMAL':
1010 this.emit(...diff.normal());
1011 break;
1012 case 'ED':
1013 this.emit(...diff.ed());
1014 break;
1015 case 'RCS':
1016 this.emit(...diff.rcs());
1017 break;
1018 case 'IFDEF':
1019 this.emit(...diff.ifdef(parameter));
1020 break;
1021 case 'UNIFIED':
1022 this.emit(...diff.unified(Number(parameter), oldName, newName));
1023 break;
1024 case 'CONTEXT':
1025 this.emit(...diff.context(Number(parameter), oldName, newName));
1026 break;
1027 default:
1028 this.emit(...diff.sideBySide(Number(parameter)));
1029 }
1030 }

1032 private tokenDiff([oldName, newName]: string[], byWord: boolean): void {
1033 const [oldLines, newLines] = this.contents(oldName, newName);
1034 const diff = new LineDiff(oldLines, newLines);
1035 if (diff.blocks.length === 0) {
1036 this.emit(`Files ${oldName} and ${newName} are identical`);
1037 return;
1038 }
1039 const separator = byWord ? ' ' : '';
1040 const totals = new TokenTotals();
1041 for (const block of diff.blocks) {
1042 for (let row = 0; row < Math.max(block.x, block.y); row++) {
1043 const oldNumber = block.p + row + 1;
1044 const newNumber = block.q + row + 1;
1045 if (row < block.x && row < block.y) {
1046 const oldTokens = splitTokens(oldLines[oldNumber - 1], byWord);
1047 const newTokens = splitTokens(newLines[newNumber - 1], byWord);
1048 const text = compareTokens(oldTokens, newTokens, separator, totals);
1049 this.emit(`a${oldNumber} b${newNumber}: ${text}`);
1050 } else if (row < block.x) {
1051 const oldTokens = splitTokens(oldLines[oldNumber - 1], byWord);
1052 totals.deleted += oldTokens.length;
1053 this.emit(`a${oldNumber}: ${deletion(oldTokens, separator)}`);
1054 } else {
1055 const newTokens = splitTokens(newLines[newNumber - 1], byWord);
1056 totals.inserted += newTokens.length;
1057 this.emit(`b${newNumber}: ${insertion(newTokens, separator)}`);
1058 }
1059 }
1060 }
1061 const label = byWord ? 'WORDS' : 'CHARS';
1062 this.emit(
1063 `${label} deleted=${totals.deleted} inserted=${totals.inserted} common=${totals.common}`,
1064 );
1065 }

1067 private stat([oldName, newName, ...options]: string[]): void {
1068 const [oldLines, newLines] = this.contents(oldName, newName);
1069 const diff = new LineDiff(oldLines, newLines, options);
1070 const deleted = diff.blocks.reduce((sum, block) => sum + block.x, 0);
1071 const inserted = diff.blocks.reduce((sum, block) => sum + block.y, 0);
1072 const total = deleted + inserted;
1073 const unchanged = oldLines.length - deleted;
1074 let summary = `${oldName} -> ${newName} | ${total}`;
1075 if (total > 0) {
1076 summary += ` ${statGraph(inserted, deleted)}`;
1077 }
1078 this.emit(summary);
1079 const lineTotal = oldLines.length + newLines.length;
1080 const similarity = lineTotal > 0 ? Math.floor((200 * unchanged) / lineTotal) : 100;
1081 this.emit(
1082 `changes=${diff.blocks.length} insertions=${inserted} deletions=${deleted}` +
1083 ` unchanged=${unchanged} similarity=${similarity}%`,
1084 );
1085 }

1087 private patch([name, , ...options]: string[], data: string[]): void {
1088 let [lines] = this.contents(name);
1089 const reverse = options.includes('REVERSE');
1090 const dryRun = options.includes('DRYRUN');
1091 const fuzzIndex = options.indexOf('FUZZ');
1092 const maxFuzz = fuzzIndex === -1 ? 0 : Number(options[fuzzIndex + 1]);
1093 const hunks = parsePatch(data);
1094 this.emit(`${dryRun ? 'checking' : 'patching'} file ${name}`);
1095 const [first] = hunks;
1096 if (
1097 searchHunk(first, lines, 0, reverse, maxFuzz) === null &&
1098 searchHunk(first, lines, 0, !reverse, maxFuzz) !== null
1099 ) {
1100 this.emit('Reversed (or previously applied) patch detected! Skipping patch.');
1101 this.emit(`RESULT applied=0 failed=0 ignored=${hunks.length} lines=${lines.length}`);
1102 return;
1103 }
1104 let adjustment = 0;
1105 let applied = 0;
1106 let failed = 0;
1107 for (const [index, hunk] of hunks.entries()) {
1108 const number = index + 1;
1109 const match = searchHunk(hunk, lines, adjustment, reverse, maxFuzz);
1110 if (match === null) {
1111 failed++;
1112 this.emit(`Hunk #${number} FAILED at ${hunk.start(reverse)}`);
1113 continue;
1114 }
1115 const [oldLines, newLines] = hunk.sequences(reverse);
1116 lines = match.lines;
1117 adjustment += match.delta + newLines.length - oldLines.length;
1118 applied++;
1119 this.emit(match.report(number));
1120 }
1121 this.emit(`RESULT applied=${applied} failed=${failed} ignored=0 lines=${lines.length}`);
1122 if (!dryRun && applied >= 1) {
1123 this.save(name, lines);
1124 }
1125 }

1134 private threeWay(baseName: string, oursName: string, theirsName: string): Chunk[] {
1135 const [base, ours, theirs] = this.contents(baseName, oursName, theirsName);
1136 return threeWayChunks(base, ours, theirs);

1139 private diff3([baseName, oursName, theirsName]: string[]): void {
1140 const chunks = this.threeWay(baseName, oursName, theirsName);
1141 const changes = chunks.filter((chunk) => chunk.kind !== 'stable');
1142 for (const chunk of changes) {
1143 this.emit(`==== ${chunk.kind} ${chunk.location()}`);
1144 this.emit(...chunk.base.map((line) => `O ${line}`));
1145 if (chunk.kind !== 'b') {
1146 this.emit(...chunk.ours.map((line) => `A ${line}`));
1147 }
1148 if (chunk.kind === 'b' || chunk.kind === 'conflict') {
1149 this.emit(...chunk.theirs.map((line) => `B ${line}`));
1150 }
1151 }
1152 this.emit(`DIFF3 chunks=${changes.length} ${kindCounts(chunks)}`);
1153 }
1154
1155 private merge([baseName, oursName, theirsName, target, mode = 'MARKERS']: string[]): void {
1156 const chunks = this.threeWay(baseName, oursName, theirsName);
1157 const result: string[] = [];
1158 const conflicts: Chunk[] = [];
1159 for (const chunk of chunks) {
1160 if (chunk.kind === 'b') {
1161 result.push(...chunk.theirs);
1162 } else if (chunk.kind === 'conflict') {
1163 result.push(...conflictLines(chunk, mode, baseName, oursName, theirsName));
1164 conflicts.push(chunk);
1165 } else {
1166 result.push(...chunk.ours);
1167 }
1168 }
1169 this.emit(`MERGE ${target} lines=${result.length} ${kindCounts(chunks)}`);
1170 conflicts.forEach((chunk, index) => {
1171 this.emit(`CONFLICT ${index + 1} ${chunk.location()}`);
1172 });
1173 this.save(target, result);
1174 }

1176 private conflicts([name]: string[]): void {
1177 const [lines] = this.contents(name);
1178 const regions = conflictRegions(lines);
1179 this.emit(`CONFLICTS ${name} count=${regions.length}`);
1180 regions.forEach((region, index) => {
1181 const base = region.base === null ? 'none' : region.base.length;
1182 this.emit(
1183 `${index + 1}: lines ${region.start}-${region.end} ours=${region.ours.length}` +
1184 ` base=${base} theirs=${region.theirs.length}`,
1185 );
1186 });
1187 }
1188
1189 private resolve([name, numberText, side]: string[]): void {
1190 const [lines] = this.contents(name);
1191 const number = Number(numberText);
1192 const regions = conflictRegions(lines);
1193 if (regions.length < number) {
1194 throw new CommandError(`ERROR no conflict ${number} in ${name}`);
1195 }
1196 const region = regions[number - 1];
1197 if (side === 'BASE' && region.base === null) {
1198 throw new CommandError(`ERROR no base section in conflict ${number}`);
1199 }
1200 const content = [
1201 ...lines.slice(0, region.start - 1),
1202 ...region.part(side),
1203 ...lines.slice(region.end),
1204 ];
1205 const remaining = conflictRegions(content).length;
1206 this.emit(`RESOLVE ${name} ${number} ${side} lines=${content.length} remaining=${remaining}`);
1207 this.save(name, content);
1208 }
1209
1210 execute([command, ...args]: string[], data: string[]): void {
1211 try {
1212 this.handlers[command](args, data);
1213 } catch (error) {
1214 if (!(error instanceof CommandError)) {
1215 throw error;
1216 }
1217 this.emit(error.message);
1218 }
1219 }
