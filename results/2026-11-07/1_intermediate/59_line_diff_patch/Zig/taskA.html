77 ignore_case: bool = false,
78 ignore_white: bool = false,

82 return .{

90fn comparisonKeys(allocator: Allocator, lines: Lines, options: CompareOptions) !Lines {
91 const keys = try allocator.alloc([]const u8, lines.len);
92 for (keys, lines) |*key, line| {
93 var text: std.ArrayList(u8) = .empty;
94 for (line, 0..) |c, i| {
95 const repeated = i > 0 and line[i - 1] == ' ';
96 const skipped = options.ignore_white or (options.ignore_space and repeated);
97 if (c == ' ' and skipped) continue;
98 try text.append(allocator, if (options.ignore_case) std.ascii.toLower(c) else c);
99 }
100 key.* = text.items;
101 }
102 return keys;
103}

105const Pair = struct { old: usize, new: usize };
106
107fn matchedPairs(allocator: Allocator, first: Lines, second: Lines) ![]Pair {
108 const columns = second.len + 1;
109 const table = try allocator.alloc(usize, (first.len + 1) * columns);
110 @memset(table, 0);
111 for (1..first.len + 1) |i| {
112 for (1..second.len + 1) |j| {
113 table[i * columns + j] = if (std.mem.eql(u8, first[i - 1], second[j - 1]))
114 table[(i - 1) * columns + j - 1] + 1
115 else
116 @max(table[(i - 1) * columns + j], table[i * columns + j - 1]);
117 }
118 }
119 var pairs: std.ArrayList(Pair) = .empty;
120 try pairs.append(allocator, .{ .old = first.len + 1, .new = second.len + 1 });
121 var i = first.len;
122 var j = second.len;
123 while (i >= 1 and j >= 1) {
124 if (std.mem.eql(u8, first[i - 1], second[j - 1])) {
125 try pairs.append(allocator, .{ .old = i, .new = j });
126 i -= 1;
127 j -= 1;
128 } else if (table[(i - 1) * columns + j] >= table[i * columns + j - 1]) {
129 i -= 1;
130 } else {
131 j -= 1;
132 }
133 }
134 try pairs.append(allocator, .{ .old = 0, .new = 0 });
135 std.mem.reverse(Pair, pairs.items);
136 return pairs.items;
137}

139const ChangeBlock = struct {
140 p: usize,
141 q: usize,
142 x: usize,
143 y: usize,
144
145 fn letter(self: ChangeBlock) u8 {
146 if (self.x >= 1 and self.y >= 1) return 'c';
147 return if (self.y == 0) 'd' else 'a';
148 }
149};
150
151fn changeBlocks(allocator: Allocator, first: Lines, second: Lines) ![]ChangeBlock {
152 const pairs = try matchedPairs(allocator, first, second);
153 var blocks: std.ArrayList(ChangeBlock) = .empty;
154 for (pairs[0 .. pairs.len - 1], pairs[1..]) |pair, next| {
155 if (next.old - pair.old > 1 or next.new - pair.new > 1) {
156 try blocks.append(allocator, .{
157 .p = pair.old,
158 .q = pair.new,
159 .x = next.old - pair.old - 1,
160 .y = next.new - pair.new - 1,
161 });
162 }
163 }
164 return blocks.items;
165}

167const Step = struct { old: usize, new: usize, block: ?ChangeBlock };
168
169fn walk(
170 allocator: Allocator,
171 blocks: []const ChangeBlock,
172 old_start: usize,
173 new_start: usize,
174 old_end: usize,
175) ![]Step {
176 var steps: std.ArrayList(Step) = .empty;
177 var old_index = old_start;
178 var new_index = new_start;
179 for (blocks) |block| {
180 while (old_index < block.p) {
181 try steps.append(allocator, .{ .old = old_index, .new = new_index, .block = null });
182 old_index += 1;
183 new_index += 1;
184 }
185 try steps.append(allocator, .{ .old = old_index, .new = new_index, .block = block });
186 old_index = block.p + block.x;
187 new_index = block.q + block.y;
188 }
189 while (old_index < old_end) {
190 try steps.append(allocator, .{ .old = old_index, .new = new_index, .block = null });
191 old_index += 1;
192 new_index += 1;
193 }
194 return steps.items;
195}

197fn writeSideRow(
198 out: *std.Io.Writer,
199 left: []const u8,
200 symbol: u8,
201 right: []const u8,
202 width: usize,
203) !void {
204 var buffer: [2 * max_side_width + 3]u8 = undefined;
205 const left_part = left[0..@min(width, left.len)];
206 const right_part = right[0..@min(width, right.len)];
207 @memcpy(buffer[0..left_part.len], left_part);
208 @memset(buffer[left_part.len..width], ' ');
209 buffer[width] = ' ';
210 buffer[width + 1] = symbol;
211 buffer[width + 2] = ' ';
212 const length = width + 3 + right_part.len;
213 @memcpy(buffer[width + 3 .. length], right_part);
214 try out.print("{s}\n", .{std.mem.trimEnd(u8, buffer[0..length], " ")});
215}

217fn writeStatGraph(out: *std.Io.Writer, inserted: usize, deleted: usize) !void {
218 const total = inserted + deleted;
219 var plus = inserted;
220 var minus = deleted;
221 if (total > graph_width) {
222 plus = graph_width * inserted / total;
223 minus = graph_width * deleted / total;
224 if (inserted >= 1 and plus == 0) plus = 1;
225 if (deleted >= 1 and minus == 0) minus = 1;
226 }
227 try out.splatByteAll('+', plus);
228 try out.splatByteAll('-', minus);
229}

231const Hunk = struct {
232 blocks: []const ChangeBlock,
233 old_start: usize,
234 old_count: usize,
235 new_start: usize,
236 new_count: usize,
237
238 fn init(blocks: []const ChangeBlock, context: usize, old_total: usize) Hunk {
239 const first = blocks[0];
240 const last = blocks[blocks.len - 1];
241 const before = @min(context, first.p);
242 const after = @min(context, old_total - last.p - last.x);
243 const old_start = first.p - before;
244 const new_start = first.q - before;
245 return .{
246 .blocks = blocks,
247 .old_start = old_start,
248 .old_count = last.p + last.x + after - old_start,
249 .new_start = new_start,
250 .new_count = last.q + last.y + after - new_start,
251 };
252 }
253
254 fn steps(self: Hunk, allocator: Allocator) ![]Step {
255 const old_end = self.old_start + self.old_count;
256 return walk(allocator, self.blocks, self.old_start, self.new_start, old_end);
257 }
258};

260const LineDiff = struct {
261 old: Lines,
262 new: Lines,
263 blocks: []const ChangeBlock,
264
265 fn init(allocator: Allocator, old: Lines, new: Lines, options: CompareOptions) !LineDiff {
266 const old_keys = try comparisonKeys(allocator, old, options);
267 const new_keys = try comparisonKeys(allocator, new, options);
268 const blocks = try changeBlocks(allocator, old_keys, new_keys);
269 return .{ .old = old, .new = new, .blocks = blocks };
270 }
271
272 fn deleted(self: LineDiff, block: ChangeBlock) Lines {
273 return self.old[block.p .. block.p + block.x];
274 }
275
276 fn inserted(self: LineDiff, block: ChangeBlock) Lines {
277 return self.new[block.q .. block.q + block.y];
278 }
279
280 fn hunks(self: LineDiff, allocator: Allocator, context: usize) ![]Hunk {
281 var groups: std.ArrayList(Hunk) = .empty;
282 var first: usize = 0;
283 for (1..self.blocks.len + 1) |index| {
284 const last = self.blocks[index - 1];
285 const ends_group = index == self.blocks.len or
286 self.blocks[index].p - (last.p + last.x) > 2 * context;
287 if (ends_group) {
288 const group = self.blocks[first..index];
289 try groups.append(allocator, Hunk.init(group, context, self.old.len));
290 first = index;
291 }
292 }
293 return groups.items;
294 }

296 fn writeNormal(self: LineDiff, out: *std.Io.Writer) !void {
297 for (self.blocks) |block| {
298 const letter = block.letter();
299 const old_range = rangeR(block.p, block.x);
300 const new_range = rangeR(block.q, block.y);
301 try out.print("{f}{c}{f}\n", .{ old_range, letter, new_range });
302 try writeLines(out, "< ", self.deleted(block));
303 if (letter == 'c') try out.writeAll("---\n");
304 try writeLines(out, "> ", self.inserted(block));
305 }
306 }

308 fn writeEd(self: LineDiff, out: *std.Io.Writer) !void {
309 var index = self.blocks.len;
310 while (index > 0) {
311 index -= 1;
312 const block = self.blocks[index];
313 const letter = block.letter();
314 try out.print("{f}{c}\n", .{ rangeR(block.p, block.x), letter });
315 if (letter != 'd') {
316 try writeLines(out, "", self.inserted(block));
317 try out.writeAll(".\n");
318 }
319 }
320 }

325 if (block.y >= 1) {

332 fn writeIfdef(
333 self: LineDiff,
334 allocator: Allocator,
335 out: *std.Io.Writer,
336 macro: []const u8,
337 ) !void {
338 for (try walk(allocator, self.blocks, 0, 0, self.old.len)) |step| {
339 const block = step.block orelse {
340 try out.print("{s}\n", .{self.old[step.old]});
341 continue;
342 };
343 if (block.x >= 1 and block.y >= 1) {
344 try out.print("#ifndef {s}\n", .{macro});
345 try writeLines(out, "", self.deleted(block));
346 try out.print("#else /* {s} */\n", .{macro});
347 try writeLines(out, "", self.inserted(block));
348 try out.print("#endif /* {s} */\n", .{macro});
349 } else if (block.y == 0) {
350 try out.print("#ifndef {s}\n", .{macro});
351 try writeLines(out, "", self.deleted(block));
352 try out.print("#endif /* ! {s} */\n", .{macro});
353 } else {
354 try out.print("#ifdef {s}\n", .{macro});
355 try writeLines(out, "", self.inserted(block));
356 try out.print("#endif /* {s} */\n", .{macro});
357 }
358 }

361 fn writeUnified(
362 self: LineDiff,
363 allocator: Allocator,
364 out: *std.Io.Writer,
365 context: usize,
366 old_name: []const u8,
367 new_name: []const u8,
368 ) !void {
369 try out.print("--- {s}\n+++ {s}\n", .{ old_name, new_name });
370 for (try self.hunks(allocator, context)) |hunk| {
371 try out.print("@@ -{f} +{f} @@\n", .{
372 rangeU(hunk.old_start, hunk.old_count),
373 rangeU(hunk.new_start, hunk.new_count),
374 });
375 for (try hunk.steps(allocator)) |step| {
376 if (step.block) |block| {
377 try writeLines(out, "-", self.deleted(block));
378 try writeLines(out, "+", self.inserted(block));
379 } else {
380 try out.print(" {s}\n", .{self.old[step.old]});
381 }
382 }
383 }
384 }

386 fn writeContext(
387 self: LineDiff,
388 allocator: Allocator,
389 out: *std.Io.Writer,
390 context: usize,
391 old_name: []const u8,
392 new_name: []const u8,
393 ) !void {
394 try out.print("*** {s}\n--- {s}\n", .{ old_name, new_name });
395 for (try self.hunks(allocator, context)) |hunk| {
396 const steps = try hunk.steps(allocator);
397 var deletions = false;
398 var insertions = false;
399 for (hunk.blocks) |block| {
400 deletions = deletions or block.x >= 1;
401 insertions = insertions or block.y >= 1;
402 }
403 const old_range = rangeR(hunk.old_start, hunk.old_count);
404 try out.print("***************\n*** {f} ****\n", .{old_range});
405 if (deletions) {
406 for (steps) |step| {
407 if (step.block) |block| {
408 try writeLines(out, if (block.y >= 1) "! " else "- ", self.deleted(block));
409 } else {
410 try out.print(" {s}\n", .{self.old[step.old]});
411 }
412 }
413 }
414 try out.print("--- {f} ----\n", .{rangeR(hunk.new_start, hunk.new_count)});
415 if (insertions) {
416 for (steps) |step| {
417 if (step.block) |block| {
418 try writeLines(out, if (block.x >= 1) "! " else "+ ", self.inserted(block));
419 } else {
420 try out.print(" {s}\n", .{self.new[step.new]});
421 }
422 }
423 }
424 }
425 }

427 fn writeSideBySide(
428 self: LineDiff,
429 allocator: Allocator,
430 out: *std.Io.Writer,
431 width: usize,
432 ) !void {
433 for (try walk(allocator, self.blocks, 0, 0, self.old.len)) |step| {
434 const block = step.block orelse {
435 try writeSideRow(out, self.old[step.old], ' ', self.new[step.new], width);
436 continue;
437 };
438 for (0..@max(block.x, block.y)) |row| {
439 const left = if (row < block.x) self.old[block.p + row] else "";
440 const right = if (row < block.y) self.new[block.q + row] else "";
441 const symbol: u8 = if (row >= block.y) '<' else if (row >= block.x) '>' else '|';
442 try writeSideRow(out, left, symbol, right, width);
443 }
444 }

448const TokenTotals = struct {
449 deleted: usize = 0,
450 inserted: usize = 0,
451 common: usize = 0,
452};
453
454fn splitTokens(allocator: Allocator, line: []const u8, by_word: bool) !Lines {
455 var tokens: std.ArrayList([]const u8) = .empty;
456 if (by_word) {
457 var words = std.mem.tokenizeScalar(u8, line, ' ');
458 while (words.next()) |word| try tokens.append(allocator, word);
459 } else {
460 for (0..line.len) |i| try tokens.append(allocator, line[i .. i + 1]);
461 }
462 return tokens.items;
463}
464
465fn deletion(allocator: Allocator, tokens: Lines, separator: []const u8) ![]const u8 {
466 const joined = try std.mem.join(allocator, separator, tokens);
467 return std.mem.concat(allocator, u8, &.{ "[-", joined, "-]" });
468}
469
470fn insertion(allocator: Allocator, tokens: Lines, separator: []const u8) ![]const u8 {
471 const joined = try std.mem.join(allocator, separator, tokens);
472 return std.mem.concat(allocator, u8, &.{ "{+", joined, "+}" });
473}
474
475fn compareTokens(
476 allocator: Allocator,
477 old_tokens: Lines,
478 new_tokens: Lines,
479 separator: []const u8,
480 totals: *TokenTotals,
481) ![]const u8 {
482 var parts: std.ArrayList([]const u8) = .empty;
483 const blocks = try changeBlocks(allocator, old_tokens, new_tokens);
484 for (try walk(allocator, blocks, 0, 0, old_tokens.len)) |step| {
485 const block = step.block orelse {
486 try parts.append(allocator, old_tokens[step.old]);
487 totals.common += 1;
488 continue;
489 };
490 if (block.x >= 1) {
491 const tokens = old_tokens[block.p .. block.p + block.x];
492 try parts.append(allocator, try deletion(allocator, tokens, separator));
493 }
494 if (block.y >= 1) {
495 const tokens = new_tokens[block.q .. block.q + block.y];
496 try parts.append(allocator, try insertion(allocator, tokens, separator));
497 }
498 totals.deleted += block.x;
499 totals.inserted += block.y;
500 }
501 return std.mem.join(allocator, separator, parts.items);
502}

508const PatchHunk = struct {

516 fn sequences(self: PatchHunk, allocator: Allocator, reverse: bool) !Sequences {
517 const removed: u8 = if (reverse) '+' else '-';
518 const added: u8 = if (reverse) '-' else '+';
519 var old: std.ArrayList([]const u8) = .empty;
520 var new: std.ArrayList([]const u8) = .empty;
521 for (self.body) |line| {
522 if (line.kind != added) try old.append(allocator, line.text);
523 if (line.kind != removed) try new.append(allocator, line.text);
524 }
525 return .{ .old = old.items, .new = new.items };
526 }

529 return if (reverse) self.new_start else self.old_start;

533 var count: usize = 0;

535 return count;
536 }

538 fn trailingContext(self: PatchHunk) usize {
539 var count: usize = 0;
540 while (count < self.body.len and self.body[self.body.len - 1 - count].kind == ' ') {
541 count += 1;

565const NumberPair = struct { first: usize, second: ?usize };
566
567fn parseNumber(text: []const u8) ?usize {
568 if (text.len < 1 or text.len > 4) return null;
569 for (text) |c| {
570 if (!std.ascii.isDigit(c)) return null;
571 }
572 return std.fmt.parseInt(usize, text, 10) catch null;
573}
574
575fn parseNumberPair(text: []const u8) ?NumberPair {
576 var parts = std.mem.splitScalar(u8, text, ',');
577 const first = parseNumber(parts.next().?) orelse return null;
578 const second_text = parts.next() orelse return .{ .first = first, .second = null };
579 if (parts.next() != null) return null;
580 return .{ .first = first, .second = parseNumber(second_text) orelse return null };
581}

583const HunkRange = struct { start: usize, count: usize };
584
585fn parseRange(text: []const u8, sign: u8) ?HunkRange {
586 if (text.len == 0 or text[0] != sign) return null;
587 const pair = parseNumberPair(text[1..]) orelse return null;
588 const count = pair.second orelse 1;
589 if (pair.first == 0 and count != 0) return null;
590 return .{ .start = pair.first, .count = count };
591}

593fn parseHunkHeader(
594 allocator: Allocator,
595 out: *std.Io.Writer,
596 line: []const u8,
597 line_number: usize,
598) Error!PatchHunk {
599 const parts = try splitWords(allocator, line);
600 if (parts.len == 4 and std.mem.eql(u8, parts[0], "@@") and std.mem.eql(u8, parts[3], "@@")) {
601 if (parseRange(parts[1], '-')) |old_range| {
602 if (parseRange(parts[2], '+')) |new_range| {
603 return .{
604 .line_number = line_number,
605 .old_start = old_range.start,
606 .old_count = old_range.count,
607 .new_start = new_range.start,
608 .new_count = new_range.count,
609 };
610 }
611 }
612 }
613 return fail(out, "PATCH ERROR line {d}: bad hunk header", .{line_number});
614}

616fn parsePatch(allocator: Allocator, out: *std.Io.Writer, patch_lines: Lines) Error![]PatchHunk {
617 var index: usize = 0;
618 while (index < patch_lines.len and !std.mem.startsWith(u8, patch_lines[index], "@@")) {
619 index += 1;
620 }
621 if (index == patch_lines.len) return fail(out, "PATCH ERROR no hunks", .{});
622 var hunks: std.ArrayList(PatchHunk) = .empty;
623 while (index < patch_lines.len) {
624 var hunk = try parseHunkHeader(allocator, out, patch_lines[index], index + 1);
625 index += 1;
626 var body: std.ArrayList(BodyLine) = .empty;
627 var old_seen: usize = 0;
628 var new_seen: usize = 0;
629 while (old_seen < hunk.old_count or new_seen < hunk.new_count) {
630 if (index == patch_lines.len)
631 return fail(out, "PATCH ERROR line {d}: truncated hunk", .{hunk.line_number});
632 const line = patch_lines[index];
633 const kind = line[0];
634 const in_old = kind == ' ' or kind == '-';
635 const in_new = kind == ' ' or kind == '+';
636 const old_full = in_old and old_seen == hunk.old_count;
637 const new_full = in_new and new_seen == hunk.new_count;
638 if (line.len < 2 or !(in_old or in_new) or old_full or new_full)
639 return fail(out, "PATCH ERROR line {d}: bad hunk line", .{index + 1});
640 try body.append(allocator, .{ .kind = kind, .text = line[1..] });
641 if (in_old) old_seen += 1;
642 if (in_new) new_seen += 1;
643 index += 1;
644 }
645 hunk.body = body.items;
646 for (hunk.body) |line| {
647 if (line.kind != ' ') break;
648 } else return fail(out, "PATCH ERROR line {d}: no changes", .{hunk.line_number});
649 try hunks.append(allocator, hunk);
650 }
651 return hunks.items;
652}

654fn searchHunk(
655 allocator: Allocator,
656 hunk: PatchHunk,
657 lines: Lines,
658 adjustment: isize,
659 reverse: bool,
660 max_fuzz: usize,
661) !?HunkMatch {
662 const sequences = try hunk.sequences(allocator, reverse);
663 const old = sequences.old;
664 const new = sequences.new;
665 const start: isize = @intCast(hunk.start(reverse));
666 const expected = if (old.len > 0) start - 1 + adjustment else start + adjustment;
667 const leading = hunk.leadingContext();
668 const trailing = hunk.trailingContext();
669 for (0..max_fuzz + 1) |fuzz| {
670 const head = @min(fuzz, leading);
671 const tail = @min(fuzz, trailing);
672 const old_part = old[head .. old.len - tail];
673 const new_part = new[head .. new.len - tail];
674 for (offsets) |delta| {
675 const signed_position = expected + @as(isize, @intCast(head)) + delta;
676 if (signed_position < 0) continue;
677 const position: usize = @intCast(signed_position);
678 const end = position + old_part.len;
679 if (end > lines.len or !linesEqual(lines[position..end], old_part)) continue;
680 return .{
681 .fuzz = fuzz,
682 .delta = delta,
683 .position = position,
684 .lines = try std.mem.concat(allocator, []const u8, &.{
685 lines[0..position],
686 new_part,
687 lines[end..],
688 }),
689 };
690 }
691 }
692 return null;
693}

709const EditResult = struct { lines: Lines, executed: usize };
710
711fn runEditScript(
712 allocator: Allocator,
713 out: *std.Io.Writer,
714 original: Lines,
715 script: Lines,
716) Error!EditResult {
717 var lines: std.ArrayList([]const u8) = .empty;
718 try lines.appendSlice(allocator, original);
719 var executed: usize = 0;
720 var index: usize = 0;
721 while (index < script.len) {
722 const line_number = index + 1;
723 const command = parseEditCommand(script[index]) orelse
724 return fail(out, "EDIT ERROR line {d}: bad command", .{line_number});
725 const line_count = lines.items.len;
726 const valid = if (command.instruction == 'a')
727 command.first <= line_count
728 else
729 1 <= command.first and command.first <= command.last and command.last <= line_count;
730 if (!valid) return fail(out, "EDIT ERROR line {d}: bad range", .{line_number});
731 index += 1;
732 var text: Lines = &.{};
733 if (command.instruction != 'd') {
734 var end = index;
735 while (end < script.len and !std.mem.eql(u8, script[end], ".")) end += 1;
736 if (end == script.len)
737 return fail(out, "EDIT ERROR line {d}: missing terminator", .{line_number});
738 text = script[index..end];
739 index = end + 1;
740 }
741 if (command.instruction == 'a') {
742 try lines.insertSlice(allocator, command.first, text);
743 } else {
744 const count = command.last - command.first + 1;
745 try lines.replaceRange(allocator, command.first - 1, count, text);
746 }
747 executed += 1;
748 }
749 return .{ .lines = lines.items, .executed = executed };
750}

789fn matchTable(allocator: Allocator, first: Lines, second: Lines) ![]usize {
790 const matches = try allocator.alloc(usize, first.len + 2);
791 @memset(matches, 0);
792 const pairs = try matchedPairs(allocator, first, second);

797fn threeWayChunks(allocator: Allocator, base: Lines, ours: Lines, theirs: Lines) ![]Chunk {
798 const ours_match = try matchTable(allocator, base, ours);
799 const theirs_match = try matchTable(allocator, base, theirs);
800 var chunks: std.ArrayList(Chunk) = .empty;
801 var x: usize = 0;
802 var y: usize = 0;
803 var z: usize = 0;
804 while (x < base.len or y < ours.len or z < theirs.len) {
805 if (x < base.len and ours_match[x + 1] == y + 1 and theirs_match[x + 1] == z + 1) {
806 try chunks.append(allocator, .{
807 .kind = .stable,
808 .base_start = x,
809 .ours_start = y,
810 .theirs_start = z,
811 .base = base[x .. x + 1],
812 .ours = ours[y .. y + 1],
813 .theirs = theirs[z .. z + 1],
814 });
815 x += 1;
816 y += 1;
817 z += 1;
818 continue;
819 }
820 var anchor = x + 1;
821 while (anchor <= base.len and (ours_match[anchor] == 0 or theirs_match[anchor] == 0)) {
822 anchor += 1;
823 }
824 var next_x = base.len;
825 var next_y = ours.len;
826 var next_z = theirs.len;
827 if (anchor <= base.len) {
828 next_x = anchor - 1;
829 next_y = ours_match[anchor] - 1;
830 next_z = theirs_match[anchor] - 1;
831 }
832 const base_part = base[x..next_x];
833 const ours_part = ours[y..next_y];
834 const theirs_part = theirs[z..next_z];
835 try chunks.append(allocator, .{
836 .kind = chunkKind(base_part, ours_part, theirs_part),
837 .base_start = x,
838 .ours_start = y,
839 .theirs_start = z,
840 .base = base_part,
841 .ours = ours_part,
842 .theirs = theirs_part,
843 });
844 x = next_x;
845 y = next_y;
846 z = next_z;
847 }
848 return chunks.items;
849}

901fn conflictLines(
902 allocator: Allocator,
903 chunk: Chunk,
904 mode: []const u8,
905 base_name: []const u8,
906 ours_name: []const u8,
907 theirs_name: []const u8,
908) !Lines {
909 if (std.mem.eql(u8, mode, "OURS")) return chunk.ours;
910 if (std.mem.eql(u8, mode, "THEIRS")) return chunk.theirs;
911 const start_marker = try std.fmt.allocPrint(allocator, "<<<<<<< {s}", .{ours_name});
912 const end_marker = try std.fmt.allocPrint(allocator, ">>>>>>> {s}", .{theirs_name});
913 if (std.mem.eql(u8, mode, "DIFF3")) {
914 const base_marker = try std.fmt.allocPrint(allocator, "||||||| {s}", .{base_name});
915 return std.mem.concat(allocator, []const u8, &.{
916 &.{start_marker},
917 chunk.ours,
918 &.{base_marker},
919 chunk.base,
920 &.{separator_marker},
921 chunk.theirs,
922 &.{end_marker},
923 });
924 }
925 const ours = chunk.ours;
926 const theirs = chunk.theirs;
927 const head = commonPrefixLength(ours, theirs);
928 const tail = commonSuffixLength(ours, theirs, @min(ours.len, theirs.len) - head);
929 const ours_middle = ours[head .. ours.len - tail];
930 const theirs_middle = theirs[head .. theirs.len - tail];
931 const middle = if (std.mem.eql(u8, mode, "UNION"))
932 try std.mem.concat(allocator, []const u8, &.{ ours_middle, theirs_middle })
933 else
934 try std.mem.concat(allocator, []const u8, &.{
935 &.{start_marker},
936 ours_middle,
937 &.{separator_marker},
938 theirs_middle,
939 &.{end_marker},
940 });
941 return std.mem.concat(allocator, []const u8, &.{
942 ours[0..head],
943 middle,
944 ours[ours.len - tail ..],
945 });
946}

948const Marker = enum { none, start, base, separator, end };

950fn markerKind(line: []const u8) Marker {
951 if (std.mem.eql(u8, line, separator_marker)) return .separator;
952 if (line.len < 7 or (line.len > 7 and line[7] != ' ')) return .none;
953 const prefix = line[0..7];

955 if (std.mem.eql(u8, prefix, "|||||||")) return .base;
956 if (std.mem.eql(u8, prefix, ">>>>>>>")) return .end;
957 return .none;
958}

961 var next = index + 1;
962 while (next < lines.len and markerKind(lines[next]) == .none) next += 1;

966const ConflictRegion = struct {
967 start: usize,
968 end: usize,
969 ours: Lines,
970 base: ?Lines,
971 theirs: Lines,
972
973 fn part(self: ConflictRegion, allocator: Allocator, side: []const u8) !Lines {
974 if (std.mem.eql(u8, side, "OURS")) return self.ours;
975 if (std.mem.eql(u8, side, "THEIRS")) return self.theirs;
976 if (std.mem.eql(u8, side, "BASE")) return self.base.?;
977 return std.mem.concat(allocator, []const u8, &.{ self.ours, self.theirs });
978 }

981fn regionAt(lines: Lines, start: usize) ?ConflictRegion {
982 if (markerKind(lines[start]) != .start) return null;
983 var middle = nextMarker(lines, start);
984 var base_marker: ?usize = null;
985 if (middle < lines.len and markerKind(lines[middle]) == .base) {
986 base_marker = middle;
987 middle = nextMarker(lines, middle);
988 }
989 if (middle == lines.len or markerKind(lines[middle]) != .separator) return null;
990 const end = nextMarker(lines, middle);
991 if (end == lines.len or markerKind(lines[end]) != .end) return null;
992 return .{
993 .start = start + 1,
994 .end = end + 1,
995 .ours = lines[start + 1 .. base_marker orelse middle],
996 .base = if (base_marker) |marker| lines[marker + 1 .. middle] else null,
997 .theirs = lines[middle + 1 .. end],
998 };
999}

1002 var regions: std.ArrayList(ConflictRegion) = .empty;
1003 var index: usize = 0;
1004 while (index < lines.len) {
1005 if (regionAt(lines, index)) |region| {
1006 try regions.append(allocator, region);
1007 index = region.end;
1008 } else {
1009 index += 1;
1010 }
1011 }
1012 return regions.items;
1013}

1113 const old_name = arguments[0];
1114 const new_name = arguments[1];

1138 } else if (std.mem.eql(u8, form, "CONTEXT")) {
1139 const context = try std.fmt.parseInt(usize, parameter, 10);

1147 fn tokenDiff(
1148 self: *Workspace,
1149 allocator: Allocator,
1150 arguments: Lines,
1151 by_word: bool,
1152 ) Error!void {
1153 const old_name = arguments[0];
1154 const new_name = arguments[1];
1155 const old = try self.require(old_name);
1156 const new = try self.require(new_name);
1157 const line_diff = try LineDiff.init(allocator, old, new, .{});
1158 if (line_diff.blocks.len == 0) {
1159 try self.out.print("Files {s} and {s} are identical\n", .{ old_name, new_name });
1160 return;
1161 }
1162 const separator: []const u8 = if (by_word) " " else "";
1163 var totals: TokenTotals = .{};
1164 for (line_diff.blocks) |block| {
1165 for (0..@max(block.x, block.y)) |row| {
1166 const old_number = block.p + row + 1;
1167 const new_number = block.q + row + 1;
1168 if (row < block.x and row < block.y) {
1169 const old_tokens = try splitTokens(allocator, old[old_number - 1], by_word);
1170 const new_tokens = try splitTokens(allocator, new[new_number - 1], by_word);
1171 const text = try compareTokens(
1172 allocator,
1173 old_tokens,
1174 new_tokens,
1175 separator,
1176 &totals,
1177 );
1178 try self.out.print("a{d} b{d}: {s}\n", .{ old_number, new_number, text });
1179 } else if (row < block.x) {
1180 const old_tokens = try splitTokens(allocator, old[old_number - 1], by_word);
1181 totals.deleted += old_tokens.len;
1182 const text = try deletion(allocator, old_tokens, separator);
1183 try self.out.print("a{d}: {s}\n", .{ old_number, text });
1184 } else {
1185 const new_tokens = try splitTokens(allocator, new[new_number - 1], by_word);
1186 totals.inserted += new_tokens.len;
1187 const text = try insertion(allocator, new_tokens, separator);
1188 try self.out.print("b{d}: {s}\n", .{ new_number, text });
1189 }
1190 }
1191 }
1192 const label = if (by_word) "WORDS" else "CHARS";
1193 try self.out.print("{s} deleted={d} inserted={d} common={d}\n", .{
1194 label,
1195 totals.deleted,
1196 totals.inserted,
1197 totals.common,
1198 });
1199 }

1231 const name = arguments[0];
1232 var lines = try self.require(name);
1233 const options = arguments[2..];
1234 const reverse = contains(options, "REVERSE");
1235 const dry_run = contains(options, "DRYRUN");
1236 var max_fuzz: usize = 0;
1237 for (options, 0..) |option, index| {
1238 if (std.mem.eql(u8, option, "FUZZ"))
1239 max_fuzz = try std.fmt.parseInt(usize, options[index + 1], 10);
1240 }
1241 const hunks = try parsePatch(allocator, self.out, data);
1242 try self.out.print("{s} file {s}\n", .{ if (dry_run) "checking" else "patching", name });
1243 const first = hunks[0];
1244 if ((try searchHunk(allocator, first, lines, 0, reverse, max_fuzz)) == null and
1245 (try searchHunk(allocator, first, lines, 0, !reverse, max_fuzz)) != null)
1246 {
1247 try self.out.writeAll("Reversed (or previously applied) patch detected! Skipping patch.\n");
1248 try self.out.print(
1249 "RESULT applied=0 failed=0 ignored={d} lines={d}\n",
1250 .{ hunks.len, lines.len },
1251 );
1252 return;
1253 }
1254 var adjustment: isize = 0;
1255 var applied: usize = 0;
1256 var failed: usize = 0;
1257 for (hunks, 1..) |hunk, number| {
1258 const found = try searchHunk(allocator, hunk, lines, adjustment, reverse, max_fuzz);
1259 const match = found orelse {
1260 failed += 1;
1261 try self.out.print("Hunk #{d} FAILED at {d}\n", .{ number, hunk.start(reverse) });
1262 continue;
1263 };
1264 const sequences = try hunk.sequences(allocator, reverse);
1265 lines = match.lines;
1266 adjustment += match.delta + @as(isize, @intCast(sequences.new.len)) -
1267 @as(isize, @intCast(sequences.old.len));
1268 applied += 1;
1269 try match.writeReport(self.out, number);
1270 }
1271 try self.out.print(
1272 "RESULT applied={d} failed={d} ignored=0 lines={d}\n",
1273 .{ applied, failed, lines.len },
1274 );
1275 if (!dry_run and applied >= 1) try self.save(name, lines);
1276 }

1278 fn edit(self: *Workspace, allocator: Allocator, arguments: Lines, data: Lines) Error!void {
1279 const name = arguments[0];
1280 const result = try runEditScript(allocator, self.out, try self.require(name), data);
1281 try self.out.print(
1282 "EDIT {s} commands={d} lines={d}\n",
1283 .{ name, result.executed, result.lines.len },
1284 );
1285 try self.save(name, result.lines);
1286 }

1288 fn threeWay(self: *Workspace, allocator: Allocator, arguments: Lines) Error![]Chunk {
1289 const base = try self.require(arguments[0]);
1290 const ours = try self.require(arguments[1]);
1291 const theirs = try self.require(arguments[2]);
1292 return threeWayChunks(allocator, base, ours, theirs);
1293 }
1294
1295 fn diff3(self: *Workspace, allocator: Allocator, arguments: Lines) Error!void {
1296 const chunks = try self.threeWay(allocator, arguments);
1297 var changes: usize = 0;
1298 for (chunks) |chunk| {
1299 if (chunk.kind == .stable) continue;
1300 changes += 1;
1301 try self.out.print("==== {s} {f}\n", .{ @tagName(chunk.kind), chunk.location() });
1302 try writeLines(self.out, "O ", chunk.base);
1303 if (chunk.kind != .b) try writeLines(self.out, "A ", chunk.ours);
1304 if (chunk.kind == .b or chunk.kind == .conflict)
1305 try writeLines(self.out, "B ", chunk.theirs);
1306 }
1307 try self.out.print("DIFF3 chunks={d} {f}\n", .{ changes, KindCounts.of(chunks) });
1308 }

1310 fn merge(self: *Workspace, allocator: Allocator, arguments: Lines) Error!void {
1311 const base_name = arguments[0];
1312 const ours_name = arguments[1];
1313 const theirs_name = arguments[2];
1314 const target = arguments[3];
1315 const mode = if (arguments.len == 5) arguments[4] else "MARKERS";
1316 const chunks = try self.threeWay(allocator, arguments);
1317 var result: std.ArrayList([]const u8) = .empty;
1318 var conflict_chunks: std.ArrayList(Chunk) = .empty;
1319 for (chunks) |chunk| {
1320 switch (chunk.kind) {
1321 .b => try result.appendSlice(allocator, chunk.theirs),
1322 .conflict => {
1323 const lines = try conflictLines(
1324 allocator,
1325 chunk,
1326 mode,
1327 base_name,
1328 ours_name,
1329 theirs_name,
1330 );
1331 try result.appendSlice(allocator, lines);
1332 try conflict_chunks.append(allocator, chunk);
1333 },
1334 else => try result.appendSlice(allocator, chunk.ours),
1335 }
1336 }
1337 try self.out.print(
1338 "MERGE {s} lines={d} {f}\n",
1339 .{ target, result.items.len, KindCounts.of(chunks) },
1340 );
1341 for (conflict_chunks.items, 1..) |chunk, number| {
1342 try self.out.print("CONFLICT {d} {f}\n", .{ number, chunk.location() });
1343 }
1344 try self.save(target, result.items);
1345 }
1346
1347 fn conflicts(self: *Workspace, allocator: Allocator, arguments: Lines) Error!void {
1348 const name = arguments[0];
1349 const regions = try conflictRegions(allocator, try self.require(name));
1350 try self.out.print("CONFLICTS {s} count={d}\n", .{ name, regions.len });
1351 for (regions, 1..) |region, number| {
1352 try self.out.print(
1353 "{d}: lines {d}-{d} ours={d} base=",
1354 .{ number, region.start, region.end, region.ours.len },
1355 );
1356 if (region.base) |base| {
1357 try self.out.print("{d}", .{base.len});
1358 } else {
1359 try self.out.writeAll("none");
1360 }
1361 try self.out.print(" theirs={d}\n", .{region.theirs.len});
1362 }
1363 }
1364
1365 fn resolve(self: *Workspace, allocator: Allocator, arguments: Lines) Error!void {
1366 const name = arguments[0];
1367 const side = arguments[2];
1368 const lines = try self.require(name);
1369 const number = try std.fmt.parseInt(usize, arguments[1], 10);
1370 const regions = try conflictRegions(allocator, lines);
1371 if (regions.len < number)
1372 return fail(self.out, "ERROR no conflict {d} in {s}", .{ number, name });
1373 const region = regions[number - 1];
1374 if (std.mem.eql(u8, side, "BASE") and region.base == null)
1375 return fail(self.out, "ERROR no base section in conflict {d}", .{number});
1376 const content = try std.mem.concat(allocator, []const u8, &.{
1377 lines[0 .. region.start - 1],
1378 try region.part(allocator, side),
1379 lines[region.end..],
1380 });
1381 const remaining = (try conflictRegions(allocator, content)).len;
1382 try self.out.print(
1383 "RESOLVE {s} {d} {s} lines={d} remaining={d}\n",
1384 .{ name, number, side, content.len, remaining },
1385 );
1386 try self.save(name, content);
1387 }
1388
1389 fn execute(self: *Workspace, allocator: Allocator, words: Lines, data: Lines) Error!void {
1390 const command = words[0];
1391 const arguments = words[1..];
1392 if (std.mem.eql(u8, command, "FILE")) return self.loadFile(arguments, data);
1393 if (std.mem.eql(u8, command, "COPY")) return self.copy(arguments);
1394 if (std.mem.eql(u8, command, "PRINT")) return self.printFile(arguments);
1395 if (std.mem.eql(u8, command, "FILES")) return self.listFiles();
1396 if (std.mem.eql(u8, command, "DIFF")) return self.diff(allocator, arguments);
1397 if (std.mem.eql(u8, command, "WORDDIFF")) return self.tokenDiff(allocator, arguments, true);
1398 if (std.mem.eql(u8, command, "CHARDIFF"))
1399 return self.tokenDiff(allocator, arguments, false);
1400 if (std.mem.eql(u8, command, "STAT")) return self.stat(allocator, arguments);
1401 if (std.mem.eql(u8, command, "PATCH")) return self.patch(allocator, arguments, data);
1402 if (std.mem.eql(u8, command, "EDIT")) return self.edit(allocator, arguments, data);
1403 if (std.mem.eql(u8, command, "DIFF3")) return self.diff3(allocator, arguments);
1404 if (std.mem.eql(u8, command, "MERGE")) return self.merge(allocator, arguments);
1405 if (std.mem.eql(u8, command, "CONFLICTS")) return self.conflicts(allocator, arguments);
1406 if (std.mem.eql(u8, command, "RESOLVE")) return self.resolve(allocator, arguments);
1407 return self.undo(arguments);
1408 }
1409};

1411pub fn main(init: std.process.Init) !void {
1412 const storage = init.arena.allocator();
1413 var input_buffer: [4096]u8 = undefined;
1414 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
1415 var output_buffer: [8192]u8 = undefined;
1416 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1417 var lines: std.ArrayList([]const u8) = .empty;
1418 while (try input.interface.takeDelimiter('\n')) |line| {
1419 try lines.append(storage, try storage.dupe(u8, line));
1420 }
1421 var workspace: Workspace = .{ .storage = storage, .out = &output.interface };
1422 var scratch = std.heap.ArenaAllocator.init(init.gpa);
1423 defer scratch.deinit();
1424 const command_count = try std.fmt.parseInt(usize, lines.items[0], 10);
1425 var position: usize = 1;
1426 for (0..command_count) |_| {
1427 _ = scratch.reset(.retain_capacity);
1428 const allocator = scratch.allocator();
1429 const words = try splitWords(allocator, lines.items[position]);
1430 position += 1;
1431 var data: Lines = &.{};
1432 if (contains(&data_commands, words[0])) {
1433 const count = try std.fmt.parseInt(usize, words[2], 10);
1434 data = lines.items[position .. position + count];
1435 position += count;
1436 }
1437 workspace.execute(allocator, words, data) catch |err| switch (err) {
1438 error.CommandFailed => {},
1439 else => return err,
1440 };
1441 }
1442 try output.interface.flush();
1443}
