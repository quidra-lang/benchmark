63function matchedPairs(array $first, array $second): array
64{
65 $rows = count($first);
66 $columns = count($second);
67 $table = array_fill(0, $rows + 1, array_fill(0, $columns + 1, 0));
68 for ($i = 1; $i <= $rows; $i++) {
69 for ($j = 1; $j <= $columns; $j++) {
70 if ($first[$i - 1] === $second[$j - 1]) {
71 $table[$i][$j] = $table[$i - 1][$j - 1] + 1;
72 } else {
73 $table[$i][$j] = max($table[$i - 1][$j], $table[$i][$j - 1]);
74 }
75 }
76 }
77 $pairs = [[$rows + 1, $columns + 1]];
78 $i = $rows;
79 $j = $columns;
80 while ($i >= 1 && $j >= 1) {
81 if ($first[$i - 1] === $second[$j - 1]) {
82 $pairs[] = [$i, $j];
83 $i--;
84 $j--;
85 } elseif ($table[$i - 1][$j] >= $table[$i][$j - 1]) {
86 $i--;
87 } else {
88 $j--;
89 }
90 }
91 $pairs[] = [0, 0];
92 return array_reverse($pairs);
93}

114function changeBlocks(array $first, array $second): array
115{
116 $pairs = matchedPairs($first, $second);
117 $blocks = [];
118 for ($index = 0; $index + 1 < count($pairs); $index++) {
119 [$i, $j] = $pairs[$index];
120 [$nextI, $nextJ] = $pairs[$index + 1];
121 if ($nextI - $i > 1 || $nextJ - $j > 1) {
122 $blocks[] = new ChangeBlock($i, $j, $nextI - $i - 1, $nextJ - $j - 1);
123 }
124 }
125 return $blocks;
126}

128function walk(array $blocks, int $oldStart, int $newStart, int $oldEnd): array
129{
130 $steps = [];
131 $oldIndex = $oldStart;
132 $newIndex = $newStart;
133 foreach ($blocks as $block) {
134 while ($oldIndex < $block->p) {
135 $steps[] = [$oldIndex, $newIndex, null];
136 $oldIndex++;
137 $newIndex++;
138 }
139 $steps[] = [$oldIndex, $newIndex, $block];
140 $oldIndex = $block->p + $block->x;
141 $newIndex = $block->q + $block->y;
142 }
143 while ($oldIndex < $oldEnd) {
144 $steps[] = [$oldIndex, $newIndex, null];
145 $oldIndex++;
146 $newIndex++;
147 }
148 return $steps;
149}

157function statGraph(int $inserted, int $deleted): string
158{
159 $total = $inserted + $deleted;
160 if ($total <= GRAPH_WIDTH) {
161 $plus = $inserted;
162 $minus = $deleted;
163 } else {
164 $plus = intdiv(GRAPH_WIDTH * $inserted, $total);
165 $minus = intdiv(GRAPH_WIDTH * $deleted, $total);
166 if ($inserted >= 1 && $plus === 0) {
167 $plus = 1;
168 }
169 if ($deleted >= 1 && $minus === 0) {
170 $minus = 1;
171 }
172 }
173 return str_repeat('+', $plus) . str_repeat('-', $minus);

281 {

299 $lines[] = "#endif /* $macro */";

305 public function unified(int $context, string $oldName, string $newName): array
306 {
307 $lines = ["--- $oldName", "+++ $newName"];
308 foreach ($this->hunks($context) as $hunk) {
309 $oldRange = rangeU($hunk->oldStart, $hunk->oldCount);
310 $newRange = rangeU($hunk->newStart, $hunk->newCount);
311 $lines[] = "@@ -$oldRange +$newRange @@";
312 foreach ($hunk->steps() as [$oldIndex, , $block]) {
313 if ($block === null) {
314 $lines[] = ' ' . $this->old[$oldIndex];
315 } else {
316 array_push($lines, ...prefixed('-', $this->deleted($block)));
317 array_push($lines, ...prefixed('+', $this->inserted($block)));
318 }
319 }
320 }
321 return $lines;
322 }

324 public function context(int $context, string $oldName, string $newName): array
325 {
326 $lines = ["*** $oldName", "--- $newName"];
327 foreach ($this->hunks($context) as $hunk) {
328 $lines[] = str_repeat('*', 15);
329 $lines[] = '*** ' . rangeR($hunk->oldStart, $hunk->oldCount) . ' ****';
330 if (array_sum(array_column($hunk->blocks, 'x')) > 0) {
331 foreach ($hunk->steps() as [$oldIndex, , $block]) {
332 if ($block === null) {
333 $lines[] = ' ' . $this->old[$oldIndex];
334 } else {
335 $marker = $block->y >= 1 ? '! ' : '- ';
336 array_push($lines, ...prefixed($marker, $this->deleted($block)));
337 }
338 }
339 }
340 $lines[] = '--- ' . rangeR($hunk->newStart, $hunk->newCount) . ' ----';
341 if (array_sum(array_column($hunk->blocks, 'y')) > 0) {
342 foreach ($hunk->steps() as [, $newIndex, $block]) {
343 if ($block === null) {
344 $lines[] = ' ' . $this->new[$newIndex];
345 } else {
346 $marker = $block->x >= 1 ? '! ' : '+ ';
347 array_push($lines, ...prefixed($marker, $this->inserted($block)));
348 }
349 }
350 }
351 }
352 return $lines;
353 }

355 public function sideBySide(int $width): array
356 {
357 $lines = [];
358 foreach (walk($this->blocks, 0, 0, count($this->old)) as [$oldIndex, $newIndex, $block]) {
359 if ($block === null) {
360 $lines[] = sideRow($this->old[$oldIndex], ' ', $this->new[$newIndex], $width);
361 continue;
362 }
363 for ($row = 0; $row < max($block->x, $block->y); $row++) {
364 if ($row < $block->x && $row < $block->y) {
365 $left = $this->old[$block->p + $row];
366 $symbol = '|';
367 $right = $this->new[$block->q + $row];
368 } elseif ($row < $block->x) {
369 $left = $this->old[$block->p + $row];
370 $symbol = '<';
371 $right = '';
372 } else {
373 $left = '';
374 $symbol = '>';
375 $right = $this->new[$block->q + $row];
376 }
377 $lines[] = sideRow($left, $symbol, $right, $width);
378 }
379 }
380 return $lines;
381 }

406function compareTokens(
407 array $oldTokens,
408 array $newTokens,
409 string $separator,
410 TokenTotals $totals,
411): string {
412 $parts = [];
413 $blocks = changeBlocks($oldTokens, $newTokens);
414 foreach (walk($blocks, 0, 0, count($oldTokens)) as [$oldIndex, , $block]) {
415 if ($block === null) {
416 $parts[] = $oldTokens[$oldIndex];
417 $totals->common++;
418 continue;
419 }
420 if ($block->x >= 1) {
421 $parts[] = deletion(array_slice($oldTokens, $block->p, $block->x), $separator);
422 }
423 if ($block->y >= 1) {
424 $parts[] = insertion(array_slice($newTokens, $block->q, $block->y), $separator);
425 }
426 $totals->deleted += $block->x;
427 $totals->inserted += $block->y;
428 }
429 return implode($separator, $parts);
430}

432final class PatchHunk
433{
434 public array $body = [];
435
436 public function __construct(
437 public readonly int $lineNumber,
438 public readonly int $oldStart,
439 public readonly int $oldCount,
440 public readonly int $newStart,
441 public readonly int $newCount,
442 ) {
443 }
444
445 public function sequences(bool $reverse): array
446 {
447 $oldKinds = $reverse ? [' ', '+'] : [' ', '-'];
448 $newKinds = $reverse ? [' ', '-'] : [' ', '+'];
449 $old = [];
450 $new = [];
451 foreach ($this->body as [$kind, $text]) {
452 if (in_array($kind, $oldKinds, true)) {
453 $old[] = $text;
454 }
455 if (in_array($kind, $newKinds, true)) {
456 $new[] = $text;
457 }
458 }
459 return [$old, $new];
460 }

468 {
469 $count = 0;
470 while ($count < count($this->body) && $this->body[$count][0] === ' ') {
471 $count++;
472 }
473 return $count;
474 }

476 public function trailingContext(): int
477 {
478 $count = 0;
479 $size = count($this->body);
480 while ($count < $size && $this->body[$size - 1 - $count][0] === ' ') {
481 $count++;
482 }
483 return $count;
484 }

512function parseRange(string $text, string $sign): ?array
513{
514 if (!str_starts_with($text, $sign) || preg_match(NUMBER_PAIR, substr($text, 1), $match) !== 1) {
515 return null;
516 }
517 $start = (int) $match[1];
518 $count = isset($match[2]) ? (int) $match[2] : 1;
519 if ($start === 0 && $count !== 0) {
520 return null;
521 }
522 return [$start, $count];

525function parseHunkHeader(string $line, int $lineNumber): PatchHunk
526{
527 $parts = explode(' ', $line);
528 $oldRange = null;
529 $newRange = null;
530 if (count($parts) === 4 && $parts[0] === '@@' && $parts[3] === '@@') {
531 $oldRange = parseRange($parts[1], '-');
532 $newRange = parseRange($parts[2], '+');

540function parsePatch(array $patchLines): array
541{
542 $total = count($patchLines);
543 $index = 0;
544 while ($index < $total && !str_starts_with($patchLines[$index], '@@')) {
545 $index++;
546 }
547 if ($index === $total) {
548 throw new CommandError('PATCH ERROR no hunks');
549 }
550 $hunks = [];
551 while ($index < $total) {
552 $hunk = parseHunkHeader($patchLines[$index], $index + 1);
553 $index++;
554 $oldSeen = 0;
555 $newSeen = 0;
556 while ($oldSeen < $hunk->oldCount || $newSeen < $hunk->newCount) {
557 if ($index === $total) {
558 throw new CommandError("PATCH ERROR line {$hunk->lineNumber}: truncated hunk");
559 }
560 $line = $patchLines[$index];
561 $kind = $line[0];
562 $inOld = $kind === ' ' || $kind === '-';
563 $inNew = $kind === ' ' || $kind === '+';
564 if (
565 strlen($line) < 2
566 || !($inOld || $inNew)
567 || ($inOld && $oldSeen === $hunk->oldCount)
568 || ($inNew && $newSeen === $hunk->newCount)
569 ) {
570 throw new CommandError('PATCH ERROR line ' . ($index + 1) . ': bad hunk line');
571 }
572 $hunk->body[] = [$kind, substr($line, 1)];
573 if ($inOld) {
574 $oldSeen++;
575 }
576 if ($inNew) {
577 $newSeen++;
578 }
579 $index++;
580 }
581 $kinds = array_column($hunk->body, 0);
582 if (!in_array('-', $kinds, true) && !in_array('+', $kinds, true)) {
583 throw new CommandError("PATCH ERROR line {$hunk->lineNumber}: no changes");
584 }
585 $hunks[] = $hunk;
586 }
587 return $hunks;
588}

590function searchHunk(
591 PatchHunk $hunk,
592 array $lines,
593 int $adjustment,
594 bool $reverse,
595 int $maxFuzz,
596): ?HunkMatch {
597 [$old, $new] = $hunk->sequences($reverse);
598 $start = $hunk->start($reverse);
599 $expected = $old !== [] ? $start - 1 + $adjustment : $start + $adjustment;
600 $leading = $hunk->leadingContext();
601 $trailing = $hunk->trailingContext();
602 for ($fuzz = 0; $fuzz <= $maxFuzz; $fuzz++) {
603 $head = min($fuzz, $leading);
604 $tail = min($fuzz, $trailing);
605 $oldPart = array_slice($old, $head, count($old) - $head - $tail);
606 $newPart = array_slice($new, $head, count($new) - $head - $tail);
607 foreach (OFFSETS as $delta) {
608 $position = $expected + $head + $delta;
609 $length = count($oldPart);
610 if ($position < 0 || $position + $length > count($lines)) {
611 continue;
612 }
613 if (array_slice($lines, $position, $length) === $oldPart) {
614 array_splice($lines, $position, $length, $newPart);
615 return new HunkMatch($fuzz, $delta, $position, $lines);
616 }
617 }
618 }
619 return null;
620}

622function parseEditCommand(string $line): ?array
623{
624 $instruction = substr($line, -1);

627 return null;
628 }
629 if ($instruction === 'a' && isset($match[2])) {
630 return null;

633 $last = isset($match[2]) ? (int) $match[2] : $first;

637function runEditScript(array $lines, array $script): array
638{
639 $executed = 0;
640 $index = 0;
641 while ($index < count($script)) {
642 $lineNumber = $index + 1;
643 $command = parseEditCommand($script[$index]);
644 if ($command === null) {
645 throw new CommandError("EDIT ERROR line $lineNumber: bad command");
646 }
647 [$instruction, $first, $last] = $command;
648 $valid = $instruction === 'a'
649 ? $first <= count($lines)
650 : 1 <= $first && $first <= $last && $last <= count($lines);
651 if (!$valid) {
652 throw new CommandError("EDIT ERROR line $lineNumber: bad range");
653 }
654 $index++;
655 $text = [];
656 if ($instruction !== 'd') {
657 $end = $index;
658 while ($end < count($script) && $script[$end] !== '.') {
659 $end++;
660 }
661 if ($end === count($script)) {
662 throw new CommandError("EDIT ERROR line $lineNumber: missing terminator");
663 }
664 $text = array_slice($script, $index, $end - $index);
665 $index = $end + 1;
666 }
667 if ($instruction === 'a') {
668 array_splice($lines, $first, 0, $text);
669 } else {
670 array_splice($lines, $first - 1, $last - $first + 1, $text);
671 }
672 $executed++;
673 }
674 return [$lines, $executed];
675}

721function threeWayChunks(array $base, array $ours, array $theirs): array
722{
723 $oursMatch = matchTable($base, $ours);
724 $theirsMatch = matchTable($base, $theirs);
725 $chunks = [];
726 $x = 0;
727 $y = 0;
728 $z = 0;
729 while ($x < count($base) || $y < count($ours) || $z < count($theirs)) {
730 if ($x < count($base) && $oursMatch[$x + 1] === $y + 1 && $theirsMatch[$x + 1] === $z + 1) {
731 $chunks[] = new Chunk('stable', $x, $y, $z, [$base[$x]], [$ours[$y]], [$theirs[$z]]);
732 $x++;
733 $y++;
734 $z++;
735 continue;
736 }
737 $anchor = $x + 1;
738 while ($anchor <= count($base) && ($oursMatch[$anchor] === 0 || $theirsMatch[$anchor] === 0)) {
739 $anchor++;
740 }
741 if ($anchor <= count($base)) {
742 $nextX = $anchor - 1;
743 $nextY = $oursMatch[$anchor] - 1;
744 $nextZ = $theirsMatch[$anchor] - 1;
745 } else {
746 $nextX = count($base);
747 $nextY = count($ours);
748 $nextZ = count($theirs);
749 }
750 $basePart = array_slice($base, $x, $nextX - $x);
751 $oursPart = array_slice($ours, $y, $nextY - $y);
752 $theirsPart = array_slice($theirs, $z, $nextZ - $z);
753 $kind = chunkKind($basePart, $oursPart, $theirsPart);
754 $chunks[] = new Chunk($kind, $x, $y, $z, $basePart, $oursPart, $theirsPart);
755 [$x, $y, $z] = [$nextX, $nextY, $nextZ];
756 }
757 return $chunks;
758}

764 if ($chunk->kind !== 'stable') {
765 $counts[$chunk->kind]++;
766 }
767 }
768 return vsprintf('a=%d b=%d both=%d conflicts=%d', $counts);

773 $length = 0;

781{
782 $length = 0;

784 $length < $limit

792function conflictLines(
793 Chunk $chunk,
794 string $mode,
795 string $baseName,
796 string $oursName,
797 string $theirsName,
798): array {
799 if ($mode === 'OURS') {
800 return $chunk->ours;
801 }
802 if ($mode === 'THEIRS') {
803 return $chunk->theirs;
804 }
805 if ($mode === 'DIFF3') {
806 return [
807 "<<<<<<< $oursName",
808 ...$chunk->ours,
809 "||||||| $baseName",
810 ...$chunk->base,
811 SEPARATOR,
812 ...$chunk->theirs,
813 ">>>>>>> $theirsName",
814 ];
815 }
816 $ours = $chunk->ours;
817 $theirs = $chunk->theirs;
818 $head = commonPrefixLength($ours, $theirs);
819 $tail = commonSuffixLength($ours, $theirs, min(count($ours), count($theirs)) - $head);
820 $oursMiddle = array_slice($ours, $head, count($ours) - $head - $tail);
821 $theirsMiddle = array_slice($theirs, $head, count($theirs) - $head - $tail);
822 $middle = $mode === 'UNION'
823 ? [...$oursMiddle, ...$theirsMiddle]
824 : ["<<<<<<< $oursName", ...$oursMiddle, SEPARATOR, ...$theirsMiddle, ">>>>>>> $theirsName"];
825 return [...array_slice($ours, 0, $head), ...$middle, ...array_slice($ours, count($ours) - $tail)];
826}

828function markerKind(string $line): ?string
829{
830 if ($line === SEPARATOR) {
831 return 'separator';
832 }
833 $kind = MARKER_PREFIXES[substr($line, 0, 7)] ?? null;
834 if ($kind !== null && (strlen($line) === 7 || $line[7] === ' ')) {
835 return $kind;
836 }
837 return null;
838}

844 $index++;

849final readonly class ConflictRegion
850{
851 public function __construct(
852 public int $start,
853 public int $end,
854 public array $ours,
855 public ?array $base,
856 public array $theirs,
857 ) {
858 }
859
860 public function part(string $side): array
861 {
862 return match ($side) {
863 'OURS' => $this->ours,
864 'THEIRS' => $this->theirs,
865 'BASE' => $this->base,
866 default => [...$this->ours, ...$this->theirs],
867 };
868 }
869}
870
871function regionAt(array $lines, int $start): ?ConflictRegion
872{
873 if (markerKind($lines[$start]) !== 'start') {
874 return null;
875 }
876 $middle = nextMarker($lines, $start);
877 $baseMarker = null;
878 if ($middle < count($lines) && markerKind($lines[$middle]) === 'base') {
879 $baseMarker = $middle;
880 $middle = nextMarker($lines, $middle);
881 }
882 if ($middle === count($lines) || markerKind($lines[$middle]) !== 'separator') {
883 return null;
884 }
885 $end = nextMarker($lines, $middle);
886 if ($end === count($lines) || markerKind($lines[$end]) !== 'end') {
887 return null;
888 }
889 if ($baseMarker === null) {
890 $ours = array_slice($lines, $start + 1, $middle - $start - 1);
891 $base = null;
892 } else {
893 $ours = array_slice($lines, $start + 1, $baseMarker - $start - 1);
894 $base = array_slice($lines, $baseMarker + 1, $middle - $baseMarker - 1);
895 }
896 $theirs = array_slice($lines, $middle + 1, $end - $middle - 1);
897 return new ConflictRegion($start + 1, $end + 1, $ours, $base, $theirs);
898}
899
900function conflictRegions(array $lines): array
901{
902 $regions = [];
903 $index = 0;
904 while ($index < count($lines)) {
905 $region = regionAt($lines, $index);
906 if ($region === null) {
907 $index++;
908 } else {
909 $regions[] = $region;
910 $index = $region->end;
911 }
912 }
913 return $regions;
914}

1052 private function tokenDiff(array $arguments, bool $byWord): void
1053 {
1054 [$oldName, $newName] = $arguments;
1055 $this->requireFiles($oldName, $newName);
1056 $old = $this->files[$oldName];
1057 $new = $this->files[$newName];
1058 $diff = new LineDiff($old, $new);
1059 if ($diff->blocks === []) {
1060 $this->emit("Files $oldName and $newName are identical");
1061 return;
1062 }
1063 $separator = $byWord ? ' ' : '';
1064 $totals = new TokenTotals();
1065 foreach ($diff->blocks as $block) {
1066 for ($row = 0; $row < max($block->x, $block->y); $row++) {
1067 $oldNumber = $block->p + $row + 1;
1068 $newNumber = $block->q + $row + 1;
1069 if ($row < $block->x && $row < $block->y) {
1070 $oldTokens = splitTokens($old[$oldNumber - 1], $byWord);
1071 $newTokens = splitTokens($new[$newNumber - 1], $byWord);
1072 $text = compareTokens($oldTokens, $newTokens, $separator, $totals);
1073 $this->emit("a$oldNumber b$newNumber: $text");
1074 } elseif ($row < $block->x) {
1075 $oldTokens = splitTokens($old[$oldNumber - 1], $byWord);
1076 $totals->deleted += count($oldTokens);
1077 $this->emit("a$oldNumber: " . deletion($oldTokens, $separator));
1078 } else {
1079 $newTokens = splitTokens($new[$newNumber - 1], $byWord);
1080 $totals->inserted += count($newTokens);
1081 $this->emit("b$newNumber: " . insertion($newTokens, $separator));
1082 }
1083 }
1084 }
1085 $this->emit(sprintf(
1086 '%s deleted=%d inserted=%d common=%d',
1087 $byWord ? 'WORDS' : 'CHARS',
1088 $totals->deleted,
1089 $totals->inserted,
1090 $totals->common,
1091 ));
1092 }

1094 private function stat(array $arguments): void
1095 {
1096 [$oldName, $newName] = $arguments;
1097 $this->requireFiles($oldName, $newName);
1098 $old = $this->files[$oldName];
1099 $new = $this->files[$newName];
1100 $diff = new LineDiff($old, $new, array_slice($arguments, 2));
1101 $deleted = array_sum(array_column($diff->blocks, 'x'));
1102 $inserted = array_sum(array_column($diff->blocks, 'y'));
1103 $total = $deleted + $inserted;
1104 $unchanged = count($old) - $deleted;
1105 $summary = "$oldName -> $newName | $total";

1107 $summary .= ' ' . statGraph($inserted, $deleted);
1108 }
1109 $this->emit($summary);
1110 $lineTotal = count($old) + count($new);
1111 $similarity = $lineTotal > 0 ? intdiv(200 * $unchanged, $lineTotal) : 100;
1112 $this->emit(sprintf(
1113 'changes=%d insertions=%d deletions=%d unchanged=%d similarity=%d%%',
1114 count($diff->blocks),

1119 ));

1122 private function patch(array $arguments, array $data): void
1123 {
1124 $name = $arguments[0];
1125 $this->requireFiles($name);
1126 $options = array_slice($arguments, 2);
1127 $reverse = in_array('REVERSE', $options, true);
1128 $dryRun = in_array('DRYRUN', $options, true);
1129 $fuzzIndex = array_search('FUZZ', $options, true);
1130 $maxFuzz = $fuzzIndex === false ? 0 : (int) $options[$fuzzIndex + 1];
1131 $hunks = parsePatch($data);
1132 $this->emit(($dryRun ? 'checking file ' : 'patching file ') . $name);
1133 $lines = $this->files[$name];
1134 $first = $hunks[0];
1135 if (
1136 searchHunk($first, $lines, 0, $reverse, $maxFuzz) === null
1137 && searchHunk($first, $lines, 0, !$reverse, $maxFuzz) !== null
1138 ) {
1139 $this->emit('Reversed (or previously applied) patch detected! Skipping patch.');
1140 $this->emit('RESULT applied=0 failed=0 ignored=' . count($hunks) . ' lines=' . count($lines));
1141 return;
1142 }
1143 $adjustment = 0;
1144 $applied = 0;
1145 $failed = 0;
1146 foreach ($hunks as $index => $hunk) {
1147 $number = $index + 1;
1148 $match = searchHunk($hunk, $lines, $adjustment, $reverse, $maxFuzz);
1149 if ($match === null) {
1150 $failed++;
1151 $this->emit("Hunk #$number FAILED at " . $hunk->start($reverse));
1152 continue;
1153 }
1154 [$old, $new] = $hunk->sequences($reverse);
1155 $lines = $match->lines;
1156 $adjustment += $match->delta + count($new) - count($old);
1157 $applied++;
1158 $this->emit($match->report($number));
1159 }
1160 $this->emit("RESULT applied=$applied failed=$failed ignored=0 lines=" . count($lines));
1161 if (!$dryRun && $applied >= 1) {
1162 $this->save($name, $lines);
1163 }
1164 }

1186 {

1189 $changes = array_filter($chunks, fn (Chunk $chunk): bool => $chunk->kind !== 'stable');

1203 private function merge(array $arguments): void
1204 {
1205 [$baseName, $oursName, $theirsName, $target] = $arguments;
1206 $mode = $arguments[4] ?? 'MARKERS';
1207 $chunks = $this->threeWay($baseName, $oursName, $theirsName);
1208 $result = [];
1209 $conflicts = [];

1211 if ($chunk->kind === 'b') {
1212 array_push($result, ...$chunk->theirs);

1218 array_push($result, ...$chunk->ours);
1219 }
1220 }

1222 foreach ($conflicts as $index => $chunk) {
1223 $this->emit('CONFLICT ' . ($index + 1) . ' ' . $chunk->location());

1266 ];
