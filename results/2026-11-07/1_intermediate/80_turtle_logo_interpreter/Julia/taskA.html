123namevalid(s) = occursin(r"\A[A-Z][A-Z0-9_]*\z", s)
124function classify(s, line)
125 occursin(r"\A[0-9]+\z", s) && return Token(:number, s, line, s)
126 namevalid(s) && return Token(:word, s, line, s)
127 s[1] in (':', '"') &&
128 namevalid(s[2:end]) &&
129 return Token(s[1]==':' ? :variable : :quoted, s, line, s[2:end])
130 logoerror(line, "bad token '$s'")
131end
132function tokenize(text, line)
133 source=uppercase(first(split(text, ';'; limit = 2)))
134 result=Token[]
135 i=1
136 delimiters=" []()+-*/=<>"
137 while i<=length(source)
138 c=source[i]
139 following=i<length(source) ? source[i+1] : '\0'
140 if c==' '
141 i+=1
142 elseif c=='-'
143 sign=(i==1 || source[i-1] in " [(") && following!='\0' && following!=' '
144 push!(result, Token(sign ? :sign : :symbol, "-", line, ""))
145 i+=1
146 elseif c in "<>"
147 pair=string(c, following)
148 spelling=pair in ("<=", "<>", ">=") ? pair : string(c)
149 push!(result, Token(:symbol, spelling, line, ""))
150 i+=length(spelling)
151 elseif c in delimiters
152 push!(result, Token(:symbol, string(c), line, ""))
153 i+=1
154 else
155 start=i
156 while i<=length(source) && !(source[i] in delimiters)
157 i+=1
158 end
159 push!(result, classify(source[start:(i-1)], line))
160 end
161 end
162 result
163end
164isword(t, s) = t.kind==:word && t.spelling==s
165function extract(source)
166 procedures=Dict{String,Procedure}()
167 order=String[]
168 top=Token[]
169 current=nothing
170 for (line, text) in source
171 tokens=tokenize(text, line)
172 isempty(tokens) && continue
173 if isword(tokens[1], "TO")
174 current!==nothing && logoerror(line, "TO inside definition of $(current.name)")
175 length(tokens)>=2 && tokens[2].kind==:word || logoerror(line, "bad procedure header")
176 name=tokens[2].spelling
177 name in PRIMITIVES && logoerror(line, "$name is a primitive")
178 haskey(procedures, name) && logoerror(line, "$name is already defined")
179 parameters=String[]
180 for token in tokens[3:end]
181 token.kind==:variable || logoerror(line, "bad procedure header")
182 token.value in parameters && logoerror(line, "duplicate input $(token.spelling)")
183 push!(parameters, token.value)
184 end
185 current=Procedure(name, parameters, line, Token[], ListNode(CallNode[]))
186 procedures[name]=current
187 push!(order, name)
188 elseif length(tokens)==1 && isword(tokens[1], "END")
189 current===nothing && logoerror(line, "END without TO")
190 current=nothing
191 elseif current===nothing
192 append!(top, tokens)
193 else
194 append!(current.tokens, tokens)
195 end
196 end
197 current!==nothing && logoerror(current.line, "missing END for $(current.name)")
198 procedures, order, top
199end
200mutable struct Parser

206Parser(tokens, procedures, inprocedure) = Parser(tokens, procedures, inprocedure, 1)
207peek(p) = p.position<=length(p.tokens) ? p.tokens[p.position] : nothing
208function advance!(p)
209 t=p.tokens[p.position]
210 p.position+=1
211 t
212end
213nextis(p, spellings) = peek(p)!==nothing && peek(p).kind!=:sign && peek(p).spelling in spellings
214unknown(p, t) = t.kind==:word && !(t.spelling in PRIMITIVES) && !haskey(p.procedures, t.spelling)
215function inputtoken(p, owner)
216 t=peek(p)
217 (t===nothing || t.spelling in ("]", ")")) &&
218 logoerror(owner.line, "not enough inputs to $(owner.spelling)")
219 unknown(p, t) && logoerror(t.line, "I don't know how to $(t.spelling)")
220 t
221end
222function parsesequence!(p)
223 commands=CallNode[]
224 while peek(p)!==nothing
225 t=peek(p)
226 t.spelling=="]" && logoerror(t.line, "unexpected ]")
227 push!(commands, parsecommand!(p))
228 end
229 ListNode(commands)
230end
231function parselist!(p)
232 opening=advance!(p)
233 commands=CallNode[]
234 while !nextis(p, ("]",))
235 peek(p)===nothing && logoerror(opening.line, "missing ]")
236 push!(commands, parsecommand!(p))
237 end
238 advance!(p)
239 ListNode(commands)
240end
241function parseinput!(p, owner, kind)
242 t=inputtoken(p, owner)
243 kind=='L' && t.spelling=="[" && return parselist!(p)
244 if kind in ('Q', 'P') && t.kind==:quoted
245 advance!(p)
246 return QuotedNode(t.value, t.line)
247 end
248 kind in ('E', 'P') && return parseexpression!(p, owner)
249 reject(t.line, owner.spelling, t.spelling)
250end
251function parsecall!(p, t, n)
252 inputs=LogoNode[]
253 for _ in 1:n
254 push!(inputs, parseexpression!(p, t))
255 end
256 CallNode(t.spelling, t.spelling, t.line, inputs)
257end
258function parsecommand!(p)
259 t=advance!(p)
260 if t.spelling=="(" && nextis(p, ("PRINT", "PR"))
261 word=advance!(p)
262 inputs=LogoNode[]
263 while !nextis(p, (")",))
264 (peek(p)===nothing || nextis(p, ("]",))) && logoerror(t.line, "missing )")
265 push!(inputs, parseinput!(p, word, 'P'))
266 end
267 advance!(p)
268 return CallNode("PRINT", word.spelling, t.line, inputs)
269 end
270 if t.kind==:word
271 name=get(ALIASES, t.spelling, t.spelling)
272 name in ("TO", "END") && logoerror(t.line, "misplaced $name")
273 if haskey(COMMANDS, name)
274 name in ("STOP", "OUTPUT", "LOCAL", "LOCALMAKE") &&
275 !p.inprocedure &&
276 logoerror(t.line, "can only use $(t.spelling) inside a procedure")
277 inputs=LogoNode[]
278 for kind in COMMANDS[name]
279 push!(inputs, parseinput!(p, t, kind))
280 end
281 if name in ("TRACE", "UNTRACE") && !haskey(p.procedures, inputs[1].value)
282 logoerror(inputs[1].line, "I don't know how to $(inputs[1].value)")
283 end
284 return CallNode(name, t.spelling, t.line, inputs)
285 end
286 haskey(p.procedures, name) && return parsecall!(p, t, length(p.procedures[name].parameters))
287 unknown(p, t) && logoerror(t.line, "I don't know how to $name")
288 end
289 logoerror(t.line, "you don't say what to do with $(t.spelling)")
290end
291function parseexpression!(p, owner, level = 1)
292 level>3 && return parseterm!(p, owner)
293 left=parseexpression!(p, owner, level+1)
294 while nextis(p, OPLEVELS[level])
295 op=advance!(p)
296 right=parseexpression!(p, op, level+1)
297 left=BinaryNode(op.spelling, left, right, op.line)
298 end
299 left
300end
301function parseterm!(p, owner)
302 t=inputtoken(p, owner)
303 advance!(p)
304 t.kind==:number && return NumberNode(parse(Int, t.value))
305 t.kind==:variable && return VariableNode(t.value, t.line)
306 t.spelling=="-" && return NegationNode(parseterm!(p, t))
307 if t.spelling=="("
308 node=parseexpression!(p, t)
309 nextis(p, (")",)) || logoerror(t.line, "missing )")
310 advance!(p)
311 return node
312 end
313 haskey(FUNCTIONS, t.spelling) && return parsecall!(p, t, FUNCTIONS[t.spelling])
314 haskey(p.procedures, t.spelling) &&
315 return parsecall!(p, t, length(p.procedures[t.spelling].parameters))
316 reject(t.line, owner.spelling, t.spelling)
317end

352function lookup(r, node)
353 for env in Iterators.reverse(r.environments)
354 haskey(env.variables, node.name) && return env.variables[node.name]
355 end
356 logoerror(node.line, "$(node.name) has no value")
357end
358function make!(r, name, value)
359 for env in Iterators.reverse(r.environments[2:end])
360 if haskey(env.variables, name)
361 env.variables[name]=value
362 return
363 end
364 end
365 r.environments[1].variables[name]=value
366end
367function callprocedure!(r, call)
368 procedure=r.procedures[call.name]
369 values=Int[]
370 for arg in call.inputs
371 push!(values, evaluate!(r, arg))
372 end
373 depth=length(r.environments)-1
374 depth>=100 && logoerror(call.line, "recursion too deep")
375 traced=procedure.name in r.traced
376 indent=repeat(" ", depth)
377 traced && println(indent, "( ", join([procedure.name; string.(values)], ' '), " )")
378 push!(
379 r.environments,
380 LogoEnvironment(procedure.name, Dict(zip(procedure.parameters, values)), nothing),
381 )
382 result=nothing
383 try
384 runlist!(r, procedure.body)
385 catch signal
386 signal isa ProcedureExit || rethrow()
387 result=signal.value
388 end
389 pop!(r.environments)
390 traced && println(indent, procedure.name, result===nothing ? " stops" : " outputs $result")
391 result
392end
393function evaluate!(r, node)
394 node isa NumberNode && return node.value
395 node isa VariableNode && return lookup(r, node)
396 node isa NegationNode && return -evaluate!(r, node.operand)
397 if node isa BinaryNode
398 a=evaluate!(r, node.left)
399 b=evaluate!(r, node.right)
400 op=node.operator
401 if op in ("/", "MOD")
402 b==0 && reject(node.line, op, 0)
403 q=div(a, b)
404 return op=="/" ? q : a-b*q
405 end
406 op=="+" && return a+b
407 op=="-" && return a-b
408 op=="*" && return a*b
409 return Int(
410 op=="=" ? a==b :
411 op=="<>" ? a!=b : op=="<" ? a<b : op==">" ? a>b : op=="<=" ? a<=b : a>=b,
412 )
413 end
414 if haskey(r.procedures, node.name)
415 value=callprocedure!(r, node)
416 value===nothing && logoerror(node.line, "$(node.name) did not output")
417 return value
418 end
419 args=Int[]
420 for arg in node.inputs
421 push!(args, evaluate!(r, arg))
422 end
423 name=node.name
424 name=="XCOR" && return r.x
425 name=="YCOR" && return r.y
426 name=="HEADING" && return r.heading
427 if name=="REPCOUNT"
428 isempty(r.repcounts) && logoerror(node.line, "REPCOUNT outside REPEAT")
429 return r.repcounts[end]
430 end
431 if name=="RANDOM"
432 args[1]<1 && reject(node.line, node.spelling, args[1])
433 r.randomstate=mod(48271*r.randomstate, 2147483647)
434 return mod(r.randomstate, args[1])
435 end
436 name=="ABS" && return abs(args[1])
437 name=="NOT" && return Int(args[1]==0)
438 name=="AND" && return Int(args[1]!=0 && args[2]!=0)
439 name=="OR" && return Int(args[1]!=0 || args[2]!=0)
440 name=="MAX" ? max(args...) : min(args...)
441end
442printitem!(r, node) = node isa QuotedNode ? node.value : string(evaluate!(r, node))
443function applypen!(r, x, y)
444 if r.pen=="DOWN"
445 r.canvas[(x, y)]=r.character
446 elseif r.pen=="ERASE"
447 delete!(r.canvas, (x, y))
448 end
449end
450function forward!(r, steps)
451 dx, dy=DIRECTIONS[div(r.heading, 45)+1]
452 steps<0 && ((dx, dy, steps)=(-dx, -dy, -steps))
453 applypen!(r, r.x, r.y)
454 for _ in 1:steps
455 r.x+=dx
456 r.y+=dy
457 applypen!(r, r.x, r.y)
458 end
459end
460function moveto!(r, x, y)
461 dx=x-r.x
462 dy=y-r.y
463 s=max(abs(dx), abs(dy))
464 if s==0
465 applypen!(r, r.x, r.y)
466 else
467 for i in 0:s
468 applypen!(r, r.x+fld(2i*dx+s, 2s), r.y+fld(2i*dy+s, 2s))
469 end
470 end
471 r.x=x
472 r.y=y
473end
474function canvasbounds(r)
475 (
476 minimum(p[1] for p in keys(r.canvas)),
477 minimum(p[2] for p in keys(r.canvas)),
478 maximum(p[1] for p in keys(r.canvas)),
479 maximum(p[2] for p in keys(r.canvas)),
480 )
481end
482function fillcanvas!(r)
483 (isempty(r.canvas) || haskey(r.canvas, (r.x, r.y))) && return
484 x, y, u, v=canvasbounds(r)
485 x<=r.x<=u && y<=r.y<=v || return
486 pending=[(r.x, r.y)]
487 r.canvas[(r.x, r.y)]=r.character
488 while !isempty(pending)
489 a, b=pop!(pending)
490 for (p, q) in ((a-1, b), (a+1, b), (a, b-1), (a, b+1))
491 if x<=p<=u && y<=q<=v && !haskey(r.canvas, (p, q))
492 r.canvas[(p, q)]=r.character
493 push!(pending, (p, q))
494 end
495 end
496 end
497end
498function writecanvas(r)
499 if isempty(r.canvas)
500 println("CANVAS 0 0 0")
501 return
502 end
503 x, y, u, v=canvasbounds(r)
504 println("CANVAS $(u-x+1) $(v-y+1) $(length(r.canvas))")
505 for row in v:-1:y
506 println(rstrip(join(get(r.canvas, (col, row), ' ') for col in x:u), ' '))
507 end
508end
509function runlist!(r, list)
510 for command in list.commands
511 execute!(r, command)
512 end
513end
514function execute!(r, command)
515 name=command.name
516 inputs=command.inputs
517 if haskey(r.procedures, name)
518 value=callprocedure!(r, command)
519 value!==nothing && logoerror(command.line, "you don't say what to do with $value")
520 return
521 end
522 if name in ("FD", "BK")
523 n=evaluate!(r, inputs[1])
524 forward!(r, name=="FD" ? n : -n)
525 elseif name in ("RT", "LT", "SETH")
526 angle=evaluate!(r, inputs[1])
527 angle%45!=0 && reject(command.line, command.spelling, angle)
528 r.heading=mod(name=="SETH" ? angle : r.heading+(name=="RT" ? angle : -angle), 360)
529 elseif name in ("PU", "PD", "PE")
530 r.pen=name=="PU" ? "UP" : name=="PD" ? "DOWN" : "ERASE"
531 elseif name=="SETPEN"
532 code=evaluate!(r, inputs[1])
533 33<=code<=126 || reject(command.line, command.spelling, code)
534 r.character=Char(code)
535 elseif name=="SETXY"
536 x=evaluate!(r, inputs[1])
537 y=evaluate!(r, inputs[2])
538 moveto!(r, x, y)
539 elseif name=="SETX"
540 x=evaluate!(r, inputs[1])
541 moveto!(r, x, r.y)
542 elseif name=="SETY"
543 y=evaluate!(r, inputs[1])
544 moveto!(r, r.x, y)
545 elseif name=="HOME"
546 moveto!(r, 0, 0)
547 r.heading=0
548 elseif name=="CS"
549 empty!(r.canvas)
550 r.x=r.y=r.heading=0
551 elseif name=="CLEAN"
552 empty!(r.canvas)
553 elseif name=="FILL"
554 fillcanvas!(r)
555 elseif name=="LABEL"
556 text=printitem!(r, inputs[1])
557 dx, dy=DIRECTIONS[div(r.heading, 45)+1]
558 for (i, c) in enumerate(text)
559 r.canvas[(r.x+(i-1)*dx, r.y+(i-1)*dy)]=c
560 end
561 elseif name=="SNAPSHOT"
562 writecanvas(r)
563 elseif name=="REPEAT"
564 n=evaluate!(r, inputs[1])
565 for k in (isempty(inputs[2].commands) ? (1:0) : (1:n))
566 push!(r.repcounts, k)
567 try
568 runlist!(r, inputs[2])
569 finally
570 pop!(r.repcounts)
571 end
572 end
573 elseif name=="FOR"
574 variable=inputs[1].value
575 a=evaluate!(r, inputs[2])
576 b=evaluate!(r, inputs[3])
577 step=a<=b ? 1 : -1
578 if isempty(inputs[4].commands)
579 make!(r, variable, b)
580 else
581 for value in a:step:b
582 make!(r, variable, value)
583 runlist!(r, inputs[4])
584 end
585 end
586 elseif name=="WHILE"
587 while evaluate!(r, inputs[1])!=0
588 runlist!(r, inputs[2])
589 end
590 elseif name=="IF"
591 evaluate!(r, inputs[1])!=0 && runlist!(r, inputs[2])
592 elseif name=="IFELSE"
593 value=evaluate!(r, inputs[1])
594 runlist!(r, inputs[value!=0 ? 2 : 3])
595 elseif name=="TEST"
596 value=evaluate!(r, inputs[1])
597 r.environments[end].test=value!=0
598 elseif name in ("IFTRUE", "IFFALSE")
599 test=r.environments[end].test
600 test===nothing && logoerror(command.line, "$(command.spelling) without TEST")
601 test==(name=="IFTRUE") && runlist!(r, inputs[1])
602 elseif name=="STOP"
603 throw(ProcedureExit(nothing))
604 elseif name=="OUTPUT"
605 throw(ProcedureExit(evaluate!(r, inputs[1])))
606 elseif name=="MAKE"
607 value=evaluate!(r, inputs[2])
608 make!(r, inputs[1].value, value)
609 elseif name=="LOCAL"
610 r.environments[end].variables[inputs[1].value]=0
611 elseif name=="LOCALMAKE"
612 value=evaluate!(r, inputs[2])
613 r.environments[end].variables[inputs[1].value]=value
614 elseif name=="TRACE"
615 push!(r.traced, inputs[1].value)
616 elseif name=="UNTRACE"
617 delete!(r.traced, inputs[1].value)
618 elseif name=="SEED"
619 value=evaluate!(r, inputs[1])
620 value<1 && reject(command.line, command.spelling, value)
621 r.randomstate=value
622 elseif name=="PRINT"
623 texts=String[]
624 for item in inputs
625 push!(texts, printitem!(r, item))
626 end
627 println(join(texts, ' '))
628 end
629end
630function runprogram(name, source)
631 println("PROGRAM $name")
632 local procedures, order, top, commands
633 try
634 procedures, order, top=extract(source)
635 for name in order
636 procedure=procedures[name]
637 procedure.body=parsesequence!(Parser(procedure.tokens, procedures, true))
638 end
639 commands=parsesequence!(Parser(top, procedures, false))
640 catch e
641 e isa LogoError || rethrow()
642 println("ERROR line $(e.line): $(e.reason)")
643 return
644 end
645 runtime=Interpreter(procedures)
646 try
647 runlist!(runtime, commands)
648 println(
649 "TURTLE $(runtime.x) $(runtime.y) $(runtime.heading) $(runtime.pen) $(runtime.character)",
650 )
651 writecanvas(runtime)
652 catch e
653 e isa LogoError || rethrow()
654 println("ERROR line $(e.line): $(e.reason)")
655 calls=[env.procedure for env in Iterators.reverse(runtime.environments[2:end])]
656 at=1
657 while at<=length(calls)
658 finish=at+1
659 while finish<=length(calls) && calls[finish]==calls[at]
660 finish+=1
661 end
662 count=finish-at
663 println(" in ", calls[at], count==1 ? "" : " x$count")
664 at=finish
665 end
666 end
667end
668function main()
669 name=""
670 source=Tuple{Int,String}[]
671 for (number, line) in enumerate(eachline())
672 if line=="#EOF"
673 runprogram(name, source)
674 break
675 elseif startswith(line, "#PROGRAM ")
676 !isempty(name) && runprogram(name, source)
677 name=line[10:end]
678 source=Tuple{Int,String}[]
679 else
680 push!(source, (number, line))
681 end
682 end
683end
