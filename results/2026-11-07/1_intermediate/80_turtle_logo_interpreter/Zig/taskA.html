77fn findPrimitive(name: Text) ?Primitive {
78 for (primitives) |item| {
79 var iterator = std.mem.tokenizeScalar(u8, item.names, ' ');
80 while (iterator.next()) |alias| if (equal(name, alias)) return item;
81 }
82 return null;
83}
84fn isName(text: Text) bool {
85 if (text.len == 0 or text[0] < 'A' or text[0] > 'Z') return false;
86 for (text) |c| if (!(c >= 'A' and c <= 'Z' or std.ascii.isDigit(c) or c == '_')) return false;
87 return true;
88}
89fn numeric(text: Text) bool {
90 if (text.len == 0) return false;
91 for (text) |c| if (!std.ascii.isDigit(c)) return false;
92 return true;
93}
94const SourceLine = struct { text: Text, number: usize };
95fn lex(gpa: Allocator, line: SourceLine) Error![]Token {
96 const end = std.mem.findScalar(u8, line.text, ';') orelse line.text.len;
97 const text = try gpa.dupe(u8, line.text[0..end]);
98 for (text) |*c| c.* = std.ascii.toUpper(c.*);
99 var result: std.ArrayList(Token) = .empty;
100 var i: usize = 0;
101 while (i < text.len) {
102 if (text[i] == ' ') {
103 i += 1;
104 continue;
105 }
106 const start = i;
107 var t: Token = .{ .line = line.number };
108 const c = text[i];
109 if (std.mem.findScalar(u8, "[]()+*/=", c) != null) {
110 i += 1;
111 } else if (c == '<' or c == '>') {
112 i += 1;
113 if (i < text.len and (text[i] == '=' or c == '<' and text[i] == '>')) i += 1;
114 } else if (c == '-') {
115 t.sign = (i == 0 or std.mem.findScalar(u8, " [(", text[i - 1]) != null) and i + 1 < text.len and text[i + 1] != ' ';
116 i += 1;
117 } else {
118 while (i < text.len and std.mem.findScalar(u8, " []()+-*/=<>", text[i]) == null) i += 1;
119 const word = text[start..i];
120 if (numeric(word)) {
121 t.kind = .number;
122 t.value = std.fmt.parseInt(Int, word, 10) catch unreachable;
123 } else if (word[0] == ':' and isName(word[1..])) {
124 t.kind = .variable;
125 t.name = word[1..];
126 } else if (word[0] == '"' and isName(word[1..])) {
127 t.kind = .quoted;
128 t.name = word[1..];
129 } else if (isName(word)) {
130 t.kind = .word;
131 t.name = word;
132 } else return bad(gpa, t, "bad token '{s}'", .{word});
133 }
134 t.text = text[start..i];
135 try result.append(gpa, t);
136 }
137 return result.items;
138}

140const Expression = struct { kind: ExpressionKind, token: Token, args: []const *Expression = &.{} };
141const Input = union(enum) { expression: *Expression, quoted: Token, list: []const *Command };
142const Command = struct { token: Token, kind: Text, inputs: []const Input = &.{} };
143const Procedure = struct { name: Text, line: usize, parameters: []const Text, tokens: std.ArrayList(Token) = .empty, body: []const *Command = &.{} };

145const Parser = struct {
146 gpa: Allocator,
147 tokens: []const Token,
148 procedures: ProcedureMap,
149 inside: bool,
150 position: usize = 0,
151 fn peek(self: *Parser) ?Token {
152 return if (self.position < self.tokens.len) self.tokens[self.position] else null;
153 }
154 fn take(self: *Parser) Token {
155 const t = self.tokens[self.position];
156 self.position += 1;
157 return t;
158 }
159 fn unknown(self: *Parser, t: Token) bool {
160 return t.kind == .word and findPrimitive(t.text) == null and !self.procedures.contains(t.text);
161 }
162 fn check(self: *Parser, owner: Token) Error!Token {
163 const t = self.peek() orelse return bad(self.gpa, owner, "not enough inputs to {s}", .{owner.text});
164 if (equal(t.text, "]") or equal(t.text, ")")) return bad(self.gpa, owner, "not enough inputs to {s}", .{owner.text});
165 if (self.unknown(t)) return bad(self.gpa, t, "I don't know how to {s}", .{t.text});
166 return t;
167 }
168 fn input(self: *Parser, kind: u8, owner: Token) Error!Input {
169 const t = try self.check(owner);
170 if (kind == 'L') {
171 if (!equal(t.text, "[")) return bad(self.gpa, t, "{s} doesn't like {s} as input", .{ owner.text, t.text });
172 const opening = self.take();
173 var commands: std.ArrayList(*Command) = .empty;
174 while (true) {
175 const next = self.peek() orelse return bad(self.gpa, opening, "missing ]", .{});
176 if (equal(next.text, "]")) {
177 _ = self.take();
178 break;
179 }
180 try commands.append(self.gpa, try self.command());
181 }
182 return .{ .list = commands.items };
183 }
184 if (kind == 'Q' or kind == 'P' and t.kind == .quoted) {
185 if (t.kind != .quoted) return bad(self.gpa, t, "{s} doesn't like {s} as input", .{ owner.text, t.text });
186 return .{ .quoted = self.take() };
187 }
188 return .{ .expression = try self.comparison(owner) };
189 }
190 fn node(self: *Parser, kind: ExpressionKind, t: Token, args: []const *Expression) Error!*Expression {
191 const expression = try self.gpa.create(Expression);
192 expression.* = .{ .kind = kind, .token = t, .args = args };
193 return expression;
194 }
195 fn binary(self: *Parser, t: Token, a: *Expression, b: *Expression) Error!*Expression {
196 const args = try self.gpa.alloc(*Expression, 2);
197 args[0] = a;
198 args[1] = b;
199 return self.node(.binary, t, args);
200 }
201 fn term(self: *Parser, owner: Token) Error!*Expression {
202 const t = try self.check(owner);
203 _ = self.take();
204 if (t.kind == .number) return self.node(.number, t, &.{});
205 if (t.kind == .variable) return self.node(.variable, t, &.{});
206 if (equal(t.text, "-")) {
207 const args = try self.gpa.alloc(*Expression, 1);
208 args[0] = try self.term(t);
209 return self.node(.unary, t, args);
210 }
211 if (equal(t.text, "(")) {
212 const value = try self.comparison(t);
213 const next = self.peek();
214 if (next == null or !equal(next.?.text, ")")) return bad(self.gpa, t, "missing )", .{});
215 _ = self.take();
216 return value;
217 }
218 var count: usize = 0;
219 var kind: ExpressionKind = undefined;
220 if (findPrimitive(t.text)) |primitive| {
221 if (!primitive.function) return bad(self.gpa, t, "{s} doesn't like {s} as input", .{ owner.text, t.text });
222 kind = .function;
223 count = primitive.inputs.len;
224 } else if (t.kind == .word and self.procedures.contains(t.text)) {
225 kind = .procedure;
226 count = self.procedures.get(t.text).?.parameters.len;
227 } else return bad(self.gpa, t, "{s} doesn't like {s} as input", .{ owner.text, t.text });
228 const args = try self.gpa.alloc(*Expression, count);
229 for (args) |*arg| arg.* = (try self.input('E', t)).expression;
230 return self.node(kind, t, args);
231 }
232 fn product(self: *Parser, owner: Token) Error!*Expression {
233 var result = try self.term(owner);
234 while (self.peek()) |t| {
235 if (!oneOf(t.text, &.{ "*", "/", "MOD" })) break;
236 _ = self.take();
237 result = try self.binary(t, result, try self.term(t));
238 }
239 return result;
240 }
241 fn sum(self: *Parser, owner: Token) Error!*Expression {
242 var result = try self.product(owner);
243 while (self.peek()) |t| {
244 if (!(equal(t.text, "+") or equal(t.text, "-") and !t.sign)) break;
245 _ = self.take();
246 result = try self.binary(t, result, try self.product(t));
247 }
248 return result;
249 }
250 fn comparison(self: *Parser, owner: Token) Error!*Expression {
251 var result = try self.sum(owner);
252 while (self.peek()) |t| {
253 if (!oneOf(t.text, &.{ "=", "<>", "<", ">", "<=", ">=" })) break;
254 _ = self.take();
255 result = try self.binary(t, result, try self.sum(t));
256 }
257 return result;
258 }
259 fn command(self: *Parser) Error!*Command {
260 const t = self.take();
261 if (equal(t.text, "]")) return bad(self.gpa, t, "unexpected ]", .{});
262 if (equal(t.text, "(")) {
263 if (self.peek()) |next| {
264 if (next.kind == .word and oneOf(next.text, &.{ "PRINT", "PR" })) {
265 const owner = self.take();
266 var inputs: std.ArrayList(Input) = .empty;
267 while (true) {
268 const nextToken = self.peek() orelse return bad(self.gpa, t, "missing )", .{});
269 if (equal(nextToken.text, "]")) return bad(self.gpa, t, "missing )", .{});
270 if (equal(nextToken.text, ")")) {
271 _ = self.take();
272 break;
273 }
274 try inputs.append(self.gpa, try self.input('P', owner));
275 }
276 const result = try self.gpa.create(Command);
277 result.* = .{ .token = owner, .kind = "multiple", .inputs = inputs.items };
278 return result;
279 }
280 }
281 }
282 if (t.kind == .word and oneOf(t.text, &.{ "TO", "END" })) return bad(self.gpa, t, "misplaced {s}", .{t.text});
283 var kind: Text = undefined;
284 var inputs: std.ArrayList(Input) = .empty;
285 if (t.kind == .word and findPrimitive(t.text) != null and !findPrimitive(t.text).?.function and !equal(t.text, "MOD")) {
286 const primitive = findPrimitive(t.text).?;
287 kind = primitive.canonical;
288 if (!self.inside and oneOf(kind, &.{ "STOP", "OUTPUT", "LOCAL", "LOCALMAKE" })) return bad(self.gpa, t, "can only use {s} inside a procedure", .{t.text});
289 for (primitive.inputs) |inputKind| try inputs.append(self.gpa, try self.input(inputKind, t));
290 if (oneOf(kind, &.{ "TRACE", "UNTRACE" })) {
291 const quoted = inputs.items[0].quoted;
292 if (!self.procedures.contains(quoted.name)) return bad(self.gpa, quoted, "I don't know how to {s}", .{quoted.name});
293 }
294 } else if (t.kind == .word and self.procedures.contains(t.text)) {
295 kind = "procedure";
296 for (0..self.procedures.get(t.text).?.parameters.len) |_| try inputs.append(self.gpa, try self.input('E', t));
297 } else if (self.unknown(t)) {
298 return bad(self.gpa, t, "I don't know how to {s}", .{t.text});
299 } else return bad(self.gpa, t, "you don't say what to do with {s}", .{t.text});
300 const result = try self.gpa.create(Command);
301 result.* = .{ .token = t, .kind = kind, .inputs = inputs.items };
302 return result;
303 }
304 fn sequence(self: *Parser) Error![]*Command {
305 var commands: std.ArrayList(*Command) = .empty;
306 while (self.peek() != null) try commands.append(self.gpa, try self.command());
307 return commands.items;
308 }
309};

318const Runtime = struct {

321 out: *Builder,

334 fn deinit(self: *Runtime) void {
335 for (self.environments.items) |environment| {
336 environment.variables.deinit(self.gpa);
337 self.gpa.destroy(environment);
338 }
339 self.environments.deinit(self.gpa);
340 self.calls.deinit(self.gpa);

345 fn current(self: *Runtime) *Environment {
346 return self.environments.items[self.environments.items.len - 1];

348 fn assign(self: *Runtime, name: Text, value: Int) Error!void {
349 var i = self.environments.items.len;
350 while (i > 1) {
351 i -= 1;
352 const environment = self.environments.items[i];
353 if (environment.variables.contains(name)) {
354 try environment.variables.put(self.gpa, name, value);
355 return;
356 }
357 }
358 try self.environments.items[0].variables.put(self.gpa, name, value);
359 }
360 fn evaluate(self: *Runtime, expression: *const Expression) Error!Int {
361 const t = expression.token;
362 if (expression.kind == .number) return t.value;
363 if (expression.kind == .variable) {
364 var i = self.environments.items.len;
365 while (i > 0) {
366 i -= 1;
367 if (self.environments.items[i].variables.get(t.name)) |value| return value;
368 }
369 return bad(self.errors, t, "{s} has no value", .{t.name});
370 }
371 if (expression.kind == .unary) return -(try self.evaluate(expression.args[0]));
372 const values = try self.gpa.alloc(Int, expression.args.len);
373 defer self.gpa.free(values);
374 for (expression.args, 0..) |argument, i| values[i] = try self.evaluate(argument);
375 if (expression.kind == .procedure) {
376 return (try self.call(t, values)) orelse return bad(self.errors, t, "{s} did not output", .{t.text});
377 }
378 if (expression.kind == .binary) {
379 const a = values[0];
380 const b = values[1];
381 if (equal(t.text, "+")) return a + b;
382 if (equal(t.text, "-")) return a - b;
383 if (equal(t.text, "*")) return a * b;
384 if (equal(t.text, "/") or equal(t.text, "MOD")) {
385 if (b == 0) return bad(self.errors, t, "{s} doesn't like 0 as input", .{t.text});
386 return if (equal(t.text, "/")) @divTrunc(a, b) else @rem(a, b);
387 }
388 if (equal(t.text, "=")) return truth(a == b);
389 if (equal(t.text, "<>")) return truth(a != b);
390 if (equal(t.text, "<")) return truth(a < b);
391 if (equal(t.text, ">")) return truth(a > b);
392 if (equal(t.text, "<=")) return truth(a <= b);
393 return truth(a >= b);
394 }
395 if (equal(t.text, "XCOR")) return self.x;
396 if (equal(t.text, "YCOR")) return self.y;
397 if (equal(t.text, "HEADING")) return self.heading;
398 if (equal(t.text, "REPCOUNT")) {
399 if (self.repetitions.items.len == 0) return bad(self.errors, t, "REPCOUNT outside REPEAT", .{});
400 return self.repetitions.items[self.repetitions.items.len - 1];
401 }
402 if (equal(t.text, "RANDOM")) {
403 if (values[0] < 1) return bad(self.errors, t, "RANDOM doesn't like {d} as input", .{values[0]});
404 self.random = @mod(48271 * self.random, 2147483647);
405 return @mod(self.random, values[0]);
406 }
407 if (equal(t.text, "ABS")) return abs(values[0]);
408 if (equal(t.text, "NOT")) return truth(values[0] == 0);
409 if (equal(t.text, "AND")) return truth(values[0] != 0 and values[1] != 0);
410 if (equal(t.text, "OR")) return truth(values[0] != 0 or values[1] != 0);
411 if (equal(t.text, "MAX")) return @max(values[0], values[1]);
412 return @min(values[0], values[1]);
413 }
414 fn call(self: *Runtime, t: Token, values: []const Int) Error!?Int {
415 if (self.calls.items.len == 100) return bad(self.errors, t, "recursion too deep", .{});
416 const procedure = self.procedures.get(t.text).?;
417 const traced = self.traced.contains(procedure.name);
418 const depth = self.calls.items.len;
419 if (traced) {
420 for (0..depth * 2) |_| try self.out.add(" ");
421 try self.out.print("( {s}", .{procedure.name});
422 for (values) |value| try self.out.print(" {d}", .{value});
423 try self.out.add(" )\n");
424 }
425 const environment = try self.gpa.create(Environment);
426 environment.* = .{};
427 for (procedure.parameters, values) |name, value| try environment.variables.put(self.gpa, name, value);
428 try self.environments.append(self.gpa, environment);
429 try self.calls.append(self.gpa, procedure.name);
430 const signal = try self.sequence(procedure.body);
431 _ = self.environments.pop();
432 _ = self.calls.pop();
433 environment.variables.deinit(self.gpa);
434 self.gpa.destroy(environment);
435 if (traced) {
436 for (0..depth * 2) |_| try self.out.add(" ");
437 if (signal != null and signal.?.output) {
438 try self.out.line("{s} outputs {d}", .{ procedure.name, signal.?.value });
439 } else try self.out.line("{s} stops", .{procedure.name});
440 }
441 return if (signal != null and signal.?.output) signal.?.value else null;
442 }
443 fn apply(self: *Runtime, p: Point) Error!void {
444 if (equal(self.pen, "DOWN")) {
445 try self.canvas.put(self.gpa, p, self.character);
446 } else if (equal(self.pen, "ERASE")) {
447 _ = self.canvas.remove(p);
448 }
449 }
450 fn forward(self: *Runtime, amount: Int) Error!void {
451 var steps = amount;
452 var delta = directions[u(@divTrunc(self.heading, 45))];
453 if (steps < 0) {
454 steps = -steps;
455 delta.x = -delta.x;
456 delta.y = -delta.y;
457 }
458 try self.apply(.{ .x = self.x, .y = self.y });
459 var i: Int = 0;
460 while (i < steps) : (i += 1) {
461 self.x += delta.x;
462 self.y += delta.y;
463 try self.apply(.{ .x = self.x, .y = self.y });
464 }
465 }
466 fn move(self: *Runtime, x: Int, y: Int) Error!void {
467 const x0 = self.x;
468 const y0 = self.y;
469 const dx = x - x0;
470 const dy = y - y0;
471 const steps = @max(abs(dx), abs(dy));
472 if (steps == 0) {
473 try self.apply(.{ .x = x0, .y = y0 });
474 } else {
475 var i: Int = 0;
476 while (i <= steps) : (i += 1) try self.apply(.{ .x = x0 + @divFloor(2 * i * dx + steps, 2 * steps), .y = y0 + @divFloor(2 * i * dy + steps, 2 * steps) });
477 }
478 self.x = x;
479 self.y = y;
480 }
481 fn canvasBounds(self: *Runtime) Bounds {
482 var result: ?Bounds = null;
483 var iterator = self.canvas.keyIterator();
484 while (iterator.next()) |p| {
485 if (result == null) result = .{ .x1 = p.x, .y1 = p.y, .x2 = p.x, .y2 = p.y };
486 result.?.x1 = @min(result.?.x1, p.x);
487 result.?.y1 = @min(result.?.y1, p.y);
488 result.?.x2 = @max(result.?.x2, p.x);
489 result.?.y2 = @max(result.?.y2, p.y);
490 }
491 return result.?;
492 }
493 fn fill(self: *Runtime) Error!void {
494 const start: Point = .{ .x = self.x, .y = self.y };
495 if (self.canvas.count() == 0 or self.canvas.contains(start)) return;
496 const b = self.canvasBounds();
497 if (self.x < b.x1 or self.x > b.x2 or self.y < b.y1 or self.y > b.y2) return;
498 var queue: std.ArrayList(Point) = .empty;
499 defer queue.deinit(self.gpa);
500 try queue.append(self.gpa, start);
501 try self.canvas.put(self.gpa, start, self.character);
502 var i: usize = 0;
503 while (i < queue.items.len) : (i += 1) {
504 const p = queue.items[i];
505 for ([_]Point{ .{ .x = 1, .y = 0 }, .{ .x = -1, .y = 0 }, .{ .x = 0, .y = 1 }, .{ .x = 0, .y = -1 } }) |delta| {
506 const q: Point = .{ .x = p.x + delta.x, .y = p.y + delta.y };
507 if (q.x < b.x1 or q.x > b.x2 or q.y < b.y1 or q.y > b.y2 or self.canvas.contains(q)) continue;
508 try self.canvas.put(self.gpa, q, self.character);
509 try queue.append(self.gpa, q);
510 }
511 }
512 }
513 fn picture(self: *Runtime) Error!void {
514 if (self.canvas.count() == 0) {
515 try self.out.line("CANVAS 0 0 0", .{});
516 return;
517 }
518 const b = self.canvasBounds();
519 try self.out.line("CANVAS {d} {d} {d}", .{ b.x2 - b.x1 + 1, b.y2 - b.y1 + 1, self.canvas.count() });
520 const row = try self.gpa.alloc(u8, u(b.x2 - b.x1 + 1));
521 defer self.gpa.free(row);
522 var y = b.y2;
523 while (y >= b.y1) : (y -= 1) {
524 for (row, 0..) |*c, i| c.* = self.canvas.get(.{ .x = b.x1 + si(i), .y = y }) orelse ' ';
525 try self.out.line("{s}", .{std.mem.trimEnd(u8, row, " ")});
526 }
527 }
528 fn item(self: *Runtime, value: Input) Error!Text {
529 return switch (value) {
530 .quoted => |quoted| try self.gpa.dupe(u8, quoted.name),
531 .expression => |expression| try std.fmt.allocPrint(self.gpa, "{d}", .{try self.evaluate(expression)}),
532 .list => unreachable,
533 };
534 }
535 fn sequence(self: *Runtime, commands: []const *Command) Error!?Signal {
536 for (commands) |command| if (try self.execute(command)) |signal| return signal;
537 return null;
538 }
539 fn execute(self: *Runtime, command: *const Command) Error!?Signal {
540 const kind = command.kind;
541 const t = command.token;
542 const inputs = command.inputs;
543 if (equal(kind, "multiple")) {
544 var line: Builder = .{ .gpa = self.gpa };
545 defer line.deinit();
546 for (inputs, 0..) |value, i| {
547 const text = try self.item(value);
548 defer self.gpa.free(text);
549 if (i > 0) try line.add(" ");
550 try line.add(text);
551 }
552 try self.out.line("{s}", .{line.bytes.items});
553 return null;
554 }
555 if (equal(kind, "WHILE")) {
556 while (try self.evaluate(inputs[0].expression) != 0) {
557 if (try self.sequence(inputs[1].list)) |signal| return signal;
558 }
559 return null;
560 }
561 const values = try self.gpa.alloc(Int, inputs.len);
562 defer self.gpa.free(values);
563 for (inputs, 0..) |value, i| values[i] = switch (value) {
564 .expression => |expression| try self.evaluate(expression),
565 else => 0,
566 };
567 if (equal(kind, "procedure")) {
568 if (try self.call(t, values)) |value| return bad(self.errors, t, "you don't say what to do with {d}", .{value});
569 return null;
570 }
571 if (equal(kind, "FD")) {
572 try self.forward(values[0]);
573 } else if (equal(kind, "BK")) {
574 try self.forward(-values[0]);
575 } else if (oneOf(kind, &.{ "RT", "LT", "SETH" })) {
576 var angle = values[0];
577 if (@rem(angle, 45) != 0) return bad(self.errors, t, "{s} doesn't like {d} as input", .{ t.text, angle });
578 if (equal(kind, "RT")) {
579 angle += self.heading;
580 } else if (equal(kind, "LT")) {
581 angle = self.heading - angle;
582 }
583 self.heading = @mod(angle, 360);
584 } else if (equal(kind, "PU")) {
585 self.pen = "UP";
586 } else if (equal(kind, "PD")) {
587 self.pen = "DOWN";
588 } else if (equal(kind, "PE")) {
589 self.pen = "ERASE";
590 } else if (equal(kind, "SETPEN")) {
591 if (values[0] < 33 or values[0] > 126) return bad(self.errors, t, "SETPEN doesn't like {d} as input", .{values[0]});
592 self.character = @intCast(values[0]);
593 } else if (equal(kind, "SETXY")) {
594 try self.move(values[0], values[1]);
595 } else if (equal(kind, "SETX")) {
596 try self.move(values[0], self.y);
597 } else if (equal(kind, "SETY")) {
598 try self.move(self.x, values[0]);
599 } else if (equal(kind, "HOME")) {
600 try self.move(0, 0);
601 self.heading = 0;
602 } else if (equal(kind, "CS")) {
603 self.canvas.clearRetainingCapacity();
604 self.x = 0;
605 self.y = 0;
606 self.heading = 0;
607 } else if (equal(kind, "CLEAN")) {
608 self.canvas.clearRetainingCapacity();
609 } else if (equal(kind, "FILL")) {
610 try self.fill();
611 } else if (equal(kind, "LABEL") or equal(kind, "PRINT")) {
612 const text = switch (inputs[0]) {
613 .quoted => |quoted| try self.gpa.dupe(u8, quoted.name),
614 else => try std.fmt.allocPrint(self.gpa, "{d}", .{values[0]}),
615 };
616 defer self.gpa.free(text);
617 if (equal(kind, "PRINT")) {
618 try self.out.line("{s}", .{text});
619 } else {
620 const delta = directions[u(@divTrunc(self.heading, 45))];
621 for (text, 0..) |c, i| try self.canvas.put(self.gpa, .{ .x = self.x + si(i) * delta.x, .y = self.y + si(i) * delta.y }, c);
622 }
623 } else if (equal(kind, "SNAPSHOT")) {
624 try self.picture();
625 } else if (equal(kind, "REPEAT")) {
626 if (inputs[1].list.len == 0) return null;
627 var n: Int = 1;
628 while (n <= values[0]) : (n += 1) {
629 try self.repetitions.append(self.gpa, n);
630 const signal = self.sequence(inputs[1].list) catch |err| {
631 _ = self.repetitions.pop();
632 return err;
633 };
634 _ = self.repetitions.pop();
635 if (signal) |value| return value;
636 }
637 } else if (equal(kind, "FOR")) {
638 if (inputs[3].list.len == 0) {
639 try self.assign(inputs[0].quoted.name, values[2]);
640 return null;
641 }
642 const a = values[1];
643 const b = values[2];
644 const step: Int = if (a <= b) 1 else -1;
645 var value = a;
646 while (true) {
647 try self.assign(inputs[0].quoted.name, value);
648 if (try self.sequence(inputs[3].list)) |signal| return signal;
649 if (value == b) break;
650 value += step;
651 }
652 } else if (equal(kind, "IF")) {
653 if (values[0] != 0) return self.sequence(inputs[1].list);
654 } else if (equal(kind, "IFELSE")) {
655 return self.sequence(inputs[if (values[0] != 0) @as(usize, 1) else 2].list);
656 } else if (equal(kind, "TEST")) {
657 self.current().test_result = values[0] != 0;
658 } else if (equal(kind, "IFTRUE") or equal(kind, "IFFALSE")) {
659 const value = self.current().test_result orelse return bad(self.errors, t, "{s} without TEST", .{t.text});
660 if (value == equal(kind, "IFTRUE")) return self.sequence(inputs[0].list);
661 } else if (equal(kind, "STOP")) {
662 return Signal{};
663 } else if (equal(kind, "OUTPUT")) {
664 return Signal{ .output = true, .value = values[0] };
665 } else if (equal(kind, "MAKE")) {
666 try self.assign(inputs[0].quoted.name, values[1]);
667 } else if (equal(kind, "LOCAL")) {
668 try self.current().variables.put(self.gpa, inputs[0].quoted.name, 0);
669 } else if (equal(kind, "LOCALMAKE")) {
670 try self.current().variables.put(self.gpa, inputs[0].quoted.name, values[1]);
671 } else if (equal(kind, "TRACE")) {
672 try self.traced.put(self.gpa, inputs[0].quoted.name, {});
673 } else if (equal(kind, "UNTRACE")) {
674 _ = self.traced.remove(inputs[0].quoted.name);
675 } else if (equal(kind, "SEED")) {
676 if (values[0] < 1) return bad(self.errors, t, "SEED doesn't like {d} as input", .{values[0]});
677 self.random = values[0];
678 }
679 return null;
680 }
681};
682const Program = struct {
683 gpa: Allocator,
684 out: *Builder,
685 procedures: ProcedureMap = .empty,
686 machine: ?*Runtime = null,
687 fn process(self: *Program, lines: []const SourceLine) Error!void {
688 var ordered: std.ArrayList(*Procedure) = .empty;
689 var top: std.ArrayList(Token) = .empty;
690 var current: ?*Procedure = null;
691 for (lines) |line| {
692 const tokens = try lex(self.gpa, line);
693 if (tokens.len == 0) continue;
694 const first = tokens[0];
695 if (first.kind == .word and equal(first.text, "TO")) {
696 if (current) |procedure| return bad(self.gpa, first, "TO inside definition of {s}", .{procedure.name});
697 if (tokens.len < 2 or tokens[1].kind != .word) return bad(self.gpa, first, "bad procedure header", .{});
698 const name = tokens[1].text;
699 if (findPrimitive(name) != null) return bad(self.gpa, first, "{s} is a primitive", .{name});
700 if (self.procedures.contains(name)) return bad(self.gpa, first, "{s} is already defined", .{name});
701 var parameters: std.ArrayList(Text) = .empty;
702 var seen: std.StringHashMapUnmanaged(void) = .empty;
703 for (tokens[2..]) |t| {
704 if (t.kind != .variable) return bad(self.gpa, first, "bad procedure header", .{});
705 if (seen.contains(t.name)) return bad(self.gpa, first, "duplicate input {s}", .{t.text});
706 try seen.put(self.gpa, t.name, {});
707 try parameters.append(self.gpa, t.name);
708 }
709 const procedure = try self.gpa.create(Procedure);
710 procedure.* = .{ .name = name, .line = first.line, .parameters = parameters.items };
711 try self.procedures.put(self.gpa, name, procedure);
712 try ordered.append(self.gpa, procedure);
713 current = procedure;
714 } else if (tokens.len == 1 and first.kind == .word and equal(first.text, "END")) {
715 if (current == null) return bad(self.gpa, first, "END without TO", .{});
716 current = null;
717 } else if (current) |procedure| {
718 try procedure.tokens.appendSlice(self.gpa, tokens);
719 } else {
720 try top.appendSlice(self.gpa, tokens);
721 }
722 }
723 if (current) |procedure| return bad(self.gpa, .{ .line = procedure.line }, "missing END for {s}", .{procedure.name});
724 for (ordered.items) |procedure| {
725 var parser: Parser = .{ .gpa = self.gpa, .tokens = procedure.tokens.items, .procedures = self.procedures, .inside = true };
726 procedure.body = try parser.sequence();
727 }
728 var parser: Parser = .{ .gpa = self.gpa, .tokens = top.items, .procedures = self.procedures, .inside = false };
729 const commands = try parser.sequence();
730 const machine = try self.gpa.create(Runtime);
731 machine.* = .{ .gpa = std.heap.page_allocator, .errors = self.gpa, .out = self.out, .procedures = self.procedures };
732 self.machine = machine;
733 const environment = try machine.gpa.create(Environment);
734 environment.* = .{};
735 try machine.environments.append(machine.gpa, environment);
736 _ = try machine.sequence(commands);
737 try self.out.line("TURTLE {d} {d} {d} {s} {c}", .{ machine.x, machine.y, machine.heading, machine.pen, machine.character });
738 try machine.picture();
739 }
740};
741fn runProgram(name: Text, lines: []const SourceLine, out: *Builder) Error!void {
742 try out.line("PROGRAM {s}", .{name});
743 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
744 defer arena.deinit();
745 var program: Program = .{ .gpa = arena.allocator(), .out = out };
746 defer {
747 if (program.machine) |machine| machine.deinit();
748 }
749 program.process(lines) catch |err| {
750 if (err != error.Invalid) return err;
751 try out.line("ERROR line {d}: {s}", .{ failureLine, failureReason });
752 if (program.machine) |machine| {
753 var i = machine.calls.items.len;
754 while (i > 0) {
755 i -= 1;
756 const procedureName = machine.calls.items[i];
757 var count: usize = 1;
758 while (i > 0 and equal(machine.calls.items[i - 1], procedureName)) {
759 i -= 1;
760 count += 1;
761 }
762 if (count == 1) {
763 try out.line(" in {s}", .{procedureName});
764 } else try out.line(" in {s} x{d}", .{ procedureName, count });
765 }
766 }
767 };
768}
769pub fn main(init: std.process.Init) !void {
770 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
771 defer arena.deinit();
772 const gpa = arena.allocator();
773 var inputBuffer: [8192]u8 = undefined;
774 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
775 const text = try input.interface.allocRemaining(gpa, .unlimited);
776 var lines = std.mem.splitScalar(u8, text, '\n');
777 var out: Builder = .{ .gpa = gpa };
778 var name: ?Text = null;
779 var source: std.ArrayList(SourceLine) = .empty;
780 var lineNumber: usize = 0;
781 while (lines.next()) |raw| {
782 lineNumber += 1;
783 const line = std.mem.trimEnd(u8, raw, "\r");
784 if (equal(line, "#EOF")) {
785 try runProgram(name.?, source.items, &out);
786 break;
787 }
788 if (std.mem.startsWith(u8, line, "#PROGRAM ")) {
789 if (name) |previous| try runProgram(previous, source.items, &out);
790 name = line[9..];
791 source.clearRetainingCapacity();
792 } else try source.append(gpa, .{ .text = line, .number = lineNumber });
793 }
794 var outputBuffer: [8192]u8 = undefined;
795 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
796 try output.interface.writeAll(out.bytes.items);
797 try output.interface.flush();
798}
