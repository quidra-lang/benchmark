29func name(_ s: String) -> Bool {
30 let b = Array(s.utf8)
31 return !b.isEmpty && b[0] >= 65 && b[0] <= 90 && b.allSatisfy {
32 $0 >= 65 && $0 <= 90 || $0 >= 48 && $0 <= 57 || $0 == 95
33 }
34}

36func lex(_ text: String, _ line: Int) throws -> [Token] {
37 let s = text.components(separatedBy: ";")[0].uppercased()
38 let b = Array(s.utf8)
39 var out = [Token]()
40 var i = 0
41 let symbols = Array("[]()+-*/=<>".utf8)
42 let delimiters = Array(" []()+-*/=<>".utf8)
43 while i < b.count {
44 let c = b[i]
45 if c == 32 {
46 i += 1
47 continue
48 }
49 if symbols.contains(c) {
50 let start = i
51 let kind = c == 45 && (i == 0 || [UInt8(32), 91, 40].contains(b[i - 1])) && i + 1 < b.count && b[i + 1] != 32 ? "sign" : "symbol"
52 if c == 60 && i + 1 < b.count && [UInt8(61), 62].contains(b[i + 1]) || c == 62 && i + 1 < b.count && b[i + 1] == 61 {
53 i += 1
54 }
55 i += 1
56 out.append(Token(text: String(decoding: b[start..<i], as: UTF8.self), kind: kind, line: line))
57 continue
58 }
59 let start = i
60 while i < b.count && !delimiters.contains(b[i]) {
61 i += 1
62 }
63 let word = String(decoding: b[start..<i], as: UTF8.self)
64 let kind: String
65 if word.utf8.allSatisfy( {
66 $0 >= 48 && $0 <= 57
67 }) {
68 kind = "number"
69 } else if word.hasPrefix(":") && name(String(word.dropFirst())) {
70 kind = "variable"
71 } else if word.hasPrefix("\"") && name(String(word.dropFirst())) {
72 kind = "quote"
73 } else if name(word) {
74 kind = "word"
75 } else {
76 throw Problem(line: line, reason: "bad token '\(word)'")
77 }
78 out.append(Token(text: word, kind: kind, line: line))
79 }
80 return out
81}

137 func unknown(_ t: Token) -> Bool {
138 t.kind == "word" && !primitive(t.text) && procedures[t.text] == nil
139 }
140
141 func next(_ owner: Token) throws -> Token {
142 if at == tokens.count || tokens[at].text == "]" || tokens[at].text == ")" {
143 try fail(owner, "not enough inputs to \(owner.text)")
144 }
145 let t = tokens[at]
146 if unknown(t) {
147 try fail(t, "I don't know how to \(t.text)")
148 }
149 return t
150 }
151
152 func rejected(_ owner: Token, _ t: Token) throws -> Never {
153 try fail(t, "\(owner.text) doesn't like \(t.text) as input")
154 }
155
156 func expression(_ owner: Token) throws -> Expr {
157 try binary(owner, 0)
158 }
159
160 func binary(_ owner: Token, _ level: Int) throws -> Expr {
161 if level == 3 {
162 return try term(owner)
163 }
164 let operators = level == 0 ? ["=", "<>", "<", ">", "<=", ">="] : level == 1 ? ["+", "-"] : ["*", "/", "MOD"]
165 var left = try binary(owner, level + 1)
166 while at < tokens.count && operators.contains(tokens[at].text) && tokens[at].kind != "sign" {
167 let t = tokens[at]
168 at += 1
169 let right = try binary(t, level + 1)
170 left = Expr(t, t.text, [left, right])
171 }
172 return left
173 }
174
175 func expressions(_ owner: Token, _ count: Int) throws -> [Expr] {
176 try (0..<count).map {
177 _ in try expression(owner)
178 }
179 }
180
181 func term(_ owner: Token) throws -> Expr {
182 let t = try next(owner)
183 at += 1
184 if t.kind == "number" {
185 return Expr(t, "NUMBER")
186 }
187 if t.kind == "variable" {
188 return Expr(t, "VARIABLE")
189 }
190 if t.text == "-" {
191 return Expr(t, "NEG", [try term(t)])
192 }
193 if t.text == "(" {
194 let e = try expression(t)
195 if at == tokens.count || tokens[at].text != ")" {
196 try fail(t, "missing )")
197 }
198 at += 1
199 return e
200 }
201 if t.kind == "word", let count = functions[t.text] {
202 return Expr(t, "FUNCTION", try expressions(t, count))
203 }
204 if t.kind == "word", let p = procedures[t.text] {
205 return Expr(t, "CALL", try expressions(t, p.params.count))
206 }
207 try rejected(owner, t)
208 }
209
210 func item(_ owner: Token) throws -> Item {
211 let t = try next(owner)
212 if t.kind == "quote" {
213 at += 1
214 return .word(String(t.text.dropFirst()))
215 }
216 return .expression(try expression(owner))
217 }
218
219 func list(_ owner: Token) throws -> [Command] {
220 let open = try next(owner)
221 if open.text != "[" {
222 try rejected(owner, open)
223 }
224 at += 1
225 var out = [Command]()
226 while at < tokens.count && tokens[at].text != "]" {
227 out.append(try command())
228 }
229 if at == tokens.count {
230 try fail(open, "missing ]")
231 }
232 at += 1
233 return out
234 }
235
236 func command() throws -> Command {
237 let t = tokens[at]
238 at += 1
239 if t.text == "]" {
240 try fail(t, "unexpected ]")
241 }
242 if t.text == "(" && at < tokens.count && tokens[at].kind == "word" && canonical(tokens[at].text) == "PRINT" {
243 let owner = tokens[at]
244 at += 1
245 var result = Command(t, "MULTIPRINT")
246 while at == tokens.count || tokens[at].text != ")" {
247 if at == tokens.count || tokens[at].text == "]" {
248 try fail(t, "missing )")
249 }
250 result.items.append(try item(owner))
251 }
252 at += 1
253 return result
254 }
255 if t.kind == "word" && ["TO", "END"].contains(t.text) {
256 try fail(t, "misplaced \(t.text)")
257 }
258 let op = canonical(t.text)
259 if t.kind == "word", let signature = signatures[op] {
260 if !inside && ["STOP", "OUTPUT", "LOCAL", "LOCALMAKE"].contains(op) {
261 try fail(t, "can only use \(t.text) inside a procedure")
262 }
263 var result = Command(t, op)
264 for kind in signature {
265 switch kind {
266 case "E":
267 result.exprs.append(try expression(t))
268 case "L":
269 result.lists.append(try list(t))
270 case "P":
271 result.items.append(try item(t))
272 default:
273 let q = try next(t)
274 if q.kind != "quote" {
275 try rejected(t, q)
276 }
277 at += 1
278 let word = String(q.text.dropFirst())
279 result.words.append(word)
280 if ["TRACE", "UNTRACE"].contains(op) && procedures[word] == nil {
281 try fail(q, "I don't know how to \(word)")
282 }
283 }
284 }
285 return result
286 }
287 if t.kind == "word", let p = procedures[t.text] {
288 var result = Command(t, "CALL")
289 result.exprs = try expressions(t, p.params.count)
290 return result
291 }
292 if unknown(t) {
293 try fail(t, "I don't know how to \(t.text)")
294 }
295 try fail(t, "you don't say what to do with \(t.text)")
296 }
297
298 func sequence() throws -> [Command] {
299 var out = [Command]()
300 while at < tokens.count {
301 out.append(try command())
302 }
303 return out
304 }
305}

345 func variable(_ t: Token) throws -> Int {
346 let name = String(t.text.dropFirst())
347 for env in envs.reversed() {
348 if let v = env.values[name] {
349 return v

355 func make(_ name: String, _ value: Int) {
356 for i in (1..<envs.count).reversed() {
357 if envs[i].values[name] != nil {
358 envs[i].values[name] = value
359 return
360 }
361 }
362 envs[0].values[name] = value
363 }

365 func invoke(_ t: Token, _ values: [Int]) throws -> Int? {
366 if calls.count == 100 {
367 try fail(t, "recursion too deep")
368 }
369 let p = procedures[t.text]!
370 let trace = traced.contains(t.text)
371 let indent = String(repeating: " ", count: calls.count)
372 if trace {
373 output.append(indent + "( " + t.text + values.map {
374 " \($0)"
375 }.joined() + " )")
376 }
377 envs.append(Environment(Dictionary(uniqueKeysWithValues: zip(p.params, values))))
378 calls.append(t.text)
379 let value: Int?
380 do {
381 try execute(p.body)
382 value = nil
383 } catch Flow.stop {
384 value = nil
385 } catch let Flow.output(v) {
386 value = v
387 }
388 envs.removeLast()
389 calls.removeLast()
390 if trace {
391 output.append(indent + t.text + (value.map {
392 " outputs \($0)"
393 } ?? " stops"))
394 }
395 return value
396 }

398 func evaluate(_ e: Expr) throws -> Int {
399 let t = e.token
400 if e.op == "NUMBER" {
401 return Int(t.text)!
402 }
403 if e.op == "VARIABLE" {
404 return try variable(t)
405 }
406 let a = try e.args.map {
407 try evaluate($0)
408 }
409 if e.op == "NEG" {
410 return -a[0]
411 }
412 if e.op == "CALL" {
413 guard let v = try invoke(t, a) else {
414 try fail(t, "\(t.text) did not output")
415 }
416 return v
417 }
418 if e.op == "FUNCTION" {
419 switch t.text {
420 case "XCOR":
421 return x
422 case "YCOR":
423 return y
424 case "HEADING":
425 return heading
426 case "REPCOUNT":
427 guard let n = repetitions.last else {
428 try fail(t, "REPCOUNT outside REPEAT")
429 }
430 return n
431 case "RANDOM":
432 if a[0] <= 0 {
433 try fail(t, "RANDOM doesn't like \(a[0]) as input")
434 }
435 random = 48271 * random % 2147483647
436 return random % a[0]
437 case "ABS":
438 return abs(a[0])
439 case "NOT":
440 return a[0] == 0 ? 1 : 0
441 case "AND":
442 return a[0] != 0 && a[1] != 0 ? 1 : 0
443 case "OR":
444 return a[0] != 0 || a[1] != 0 ? 1 : 0
445 case "MAX":
446 return max(a[0], a[1])
447 default:
448 return min(a[0], a[1])
449 }
450 }
451 if (e.op == "/" || e.op == "MOD") && a[1] == 0 {
452 try fail(t, "\(t.text) doesn't like 0 as input")
453 }
454 switch e.op {
455 case "+":
456 return a[0] + a[1]
457 case "-":
458 return a[0] - a[1]
459 case "*":
460 return a[0] * a[1]
461 case "/":
462 return a[0] / a[1]
463 case "MOD":
464 return a[0] % a[1]
465 case "=":
466 return a[0] == a[1] ? 1 : 0
467 case "<>":
468 return a[0] != a[1] ? 1 : 0
469 case "<":
470 return a[0] < a[1] ? 1 : 0
471 case ">":
472 return a[0] > a[1] ? 1 : 0
473 case "<=":
474 return a[0] <= a[1] ? 1 : 0
475 default:
476 return a[0] >= a[1] ? 1 : 0
477 }
478 }

490 let p = Point(x: x, y: y)
491 if pen == "DOWN" {
492 canvas[p] = character
493 } else if pen == "ERASE" {
494 canvas.removeValue(forKey: p)
495 }

498 func move(_ n: Int) {
499 let (dx, dy) = directions[heading / 45]
500 let sign = n < 0 ? -1 : 1
501 apply(x, y)
502 for _ in 0..<abs(n) {
503 x += sign * dx
504 y += sign * dy
505 apply(x, y)
506 }
507 }

509 func floor(_ n: Int, _ d: Int) -> Int {
510 n / d - (n < 0 && n % d != 0 ? 1 : 0)
511 }

513 func straight(_ xx: Int, _ yy: Int) {
514 let dx = xx - x
515 let dy = yy - y
516 let size = max(abs(dx), abs(dy))
517 if size == 0 {
518 apply(x, y)
519 } else {
520 for i in 0...size {
521 apply(x + floor(2 * i * dx + size, 2 * size), y + floor(2 * i * dy + size, 2 * size))
522 }
523 }
524 x = xx
525 y = yy
526 }

528 func bounds() -> (Int, Int, Int, Int)? {
529 if canvas.isEmpty {
530 return nil
531 }
532 return (canvas.keys.map(\.x).min()!, canvas.keys.map(\.y).min()!, canvas.keys.map(\.x).max()!, canvas.keys.map(\.y).max()!)
533 }

535 func fill() {
536 guard let (x1, y1, x2, y2) = bounds(), canvas[Point(x: x, y: y)] == nil, x >= x1, x <= x2, y >= y1, y <= y2 else {
537 return
538 }
539 var queue = [Point(x: x, y: y)]
540 var at = 0
541 canvas[queue[0]] = character
542 while at < queue.count {
543 let p = queue[at]
544 at += 1
545 for (dx, dy) in [(0, 1), (1, 0), (0, -1), (-1, 0)] {
546 let q = Point(x: p.x + dx, y: p.y + dy)
547 if q.x >= x1 && q.x <= x2 && q.y >= y1 && q.y <= y2 && canvas[q] == nil {
548 canvas[q] = character
549 queue.append(q)
550 }
551 }
552 }
553 }

555 func picture() {
556 guard let (x1, y1, x2, y2) = bounds() else {
557 output.append("CANVAS 0 0 0")
558 return
559 }
560 output.append("CANVAS \(x2 - x1 + 1) \(y2 - y1 + 1) \(canvas.count)")
561 for yy in (y1...y2).reversed() {
562 var line = String((x1...x2).map {
563 canvas[Point(x: $0, y: yy)] ?? " "
564 })
565 while line.last == " " {
566 line.removeLast()
567 }
568 output.append(line)
569 }

578 func command(_ c: Command) throws {
579 let t = c.token
580 let op = c.op
581 if op == "WHILE" {
582 while try evaluate(c.exprs[0]) != 0 {
583 try execute(c.lists[0])
584 }
585 return
586 }
587 if op == "PRINT" || op == "MULTIPRINT" {
588 let values = try c.items.map {
589 try item($0)
590 }
591 output.append(values.joined(separator: " "))
592 return
593 }
594 if op == "LABEL" {
595 let text = try item(c.items[0])
596 let (dx, dy) = directions[heading / 45]
597 for (i, ch) in text.enumerated() {
598 canvas[Point(x: x + i * dx, y: y + i * dy)] = ch
599 }
600 return
601 }
602 let a = try c.exprs.map {
603 try evaluate($0)
604 }
605 switch op {
606 case "CALL":
607 if let v = try invoke(t, a) {
608 try fail(t, "you don't say what to do with \(v)")
609 }
610 case "FD":
611 move(a[0])
612 case "BK":
613 move(-a[0])
614 case "RT", "LT", "SETH":
615 let angle = a[0]
616 if angle % 45 != 0 {
617 try fail(t, "\(t.text) doesn't like \(angle) as input")
618 }
619 let h = op == "SETH" ? angle : heading + (op == "RT" ? angle : -angle)
620 heading = (h % 360 + 360) % 360
621 case "PU":
622 pen = "UP"
623 case "PD":
624 pen = "DOWN"
625 case "PE":
626 pen = "ERASE"
627 case "SETPEN":
628 if !(33...126).contains(a[0]) {
629 try fail(t, "SETPEN doesn't like \(a[0]) as input")
630 }
631 character = Character(UnicodeScalar(a[0])!)
632 case "SETXY":
633 straight(a[0], a[1])
634 case "SETX":
635 straight(a[0], y)
636 case "SETY":
637 straight(x, a[0])
638 case "HOME":
639 straight(0, 0)
640 heading = 0
641 case "CS":
642 canvas = [:]
643 x = 0
644 y = 0
645 heading = 0
646 case "CLEAN":
647 canvas = [:]
648 case "FILL":
649 fill()
650 case "SNAPSHOT":
651 picture()
652 case "MAKE":
653 make(c.words[0], a[0])
654 case "LOCAL":
655 envs.last!.values[c.words[0]] = 0
656 case "LOCALMAKE":
657 envs.last!.values[c.words[0]] = a[0]
658 case "TRACE":
659 traced.insert(c.words[0])
660 case "UNTRACE":
661 traced.remove(c.words[0])
662 case "SEED":
663 if a[0] <= 0 {
664 try fail(t, "SEED doesn't like \(a[0]) as input")
665 }
666 random = a[0]
667 case "STOP":
668 throw Flow.stop
669 case "OUTPUT":
670 throw Flow.output(a[0])
671 case "REPEAT":
672 if c.lists[0].isEmpty {
673 return
674 }
675 if a[0] > 0 {
676 for i in 1...a[0] {
677 repetitions.append(i)
678 do {
679 defer {
680 repetitions.removeLast()
681 }
682 try execute(c.lists[0])
683 }
684 }
685 }
686 case "FOR":
687 if c.lists[0].isEmpty {
688 make(c.words[0], a[1])
689 return
690 }
691 let step = a[0] <= a[1] ? 1 : -1
692 var i = a[0]
693 while true {
694 make(c.words[0], i)
695 try execute(c.lists[0])
696 if i == a[1] {
697 break
698 }
699 i += step
700 }
701 case "IF":
702 if a[0] != 0 {
703 try execute(c.lists[0])
704 }
705 case "IFELSE":
706 try execute(c.lists[a[0] != 0 ? 0 : 1])
707 case "TEST":
708 envs.last!.test = a[0] != 0
709 case "IFTRUE", "IFFALSE":
710 guard let test = envs.last!.test else {
711 try fail(t, "\(t.text) without TEST")
712 }
713 if test == (op == "IFTRUE") {
714 try execute(c.lists[0])
715 }
716 default:
717 break
718 }
719 }

721 func errors() {
722 var i = calls.count
723 while i > 0 {
724 let p = calls[i - 1]
725 var count = 0
726 while i > 0 && calls[i - 1] == p {
727 i -= 1
728 count += 1
729 }
730 output.append(" in \(p)" + (count == 1 ? "" : " x\(count)"))
731 }
732 }

735func process(_ source: [(Int, String)]) -> [String] {
736 do {
737 var procedures = [String: Procedure]()
738 var order = [Procedure]()
739 var top = [Token]()
740 var current: Procedure?
741 for (line, text) in source {
742 let tokens = try lex(text, line)
743 if tokens.isEmpty {
744 continue
745 }
746 let start = tokens[0]
747 if start.kind == "word" && start.text == "TO" {
748 if let p = current {
749 try fail(start, "TO inside definition of \(p.token.text)")
750 }
751 if tokens.count < 2 || tokens[1].kind != "word" {
752 try fail(start, "bad procedure header")
753 }
754 let name = tokens[1]
755 if primitive(name.text) {
756 try fail(start, "\(name.text) is a primitive")
757 }
758 if procedures[name.text] != nil {
759 try fail(start, "\(name.text) is already defined")
760 }
761 var params = [String]()
762 for t in tokens.dropFirst(2) {
763 if t.kind != "variable" {
764 try fail(start, "bad procedure header")
765 }
766 let p = String(t.text.dropFirst())
767 if params.contains(p) {
768 try fail(start, "duplicate input \(t.text)")
769 }
770 params.append(p)
771 }
772 let p = Procedure(name, params)
773 procedures[name.text] = p
774 order.append(p)
775 current = p
776 } else if tokens.count == 1 && start.kind == "word" && start.text == "END" {
777 if current == nil {
