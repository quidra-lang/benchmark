24fn canonical(s: &str) -> &str {
25 match s {
26 "FORWARD" => "FD",
27 "BACK" => "BK",
28 "RIGHT" => "RT",
29 "LEFT" => "LT",
30 "SETHEADING" => "SETH",
31 "PENUP" => "PU",
32 "PENDOWN" => "PD",
33 "PENERASE" => "PE",
34 "CLEARSCREEN" => "CS",
35 "IFT" => "IFTRUE",
36 "IFF" => "IFFALSE",
37 "OP" => "OUTPUT",
38 "PR" => "PRINT",
39 _ => s,
40 }
41}
42
43fn signature(s: &str) -> Option<&'static str> {
44 match canonical(s) {
45 "FD" | "BK" | "RT" | "LT" | "SETH" | "SETPEN" | "SETX" | "SETY" | "TEST" | "OUTPUT"
46 | "SEED" => Some("E"),
47 "PU" | "PD" | "PE" | "HOME" | "CS" | "CLEAN" | "FILL" | "SNAPSHOT" | "STOP" => Some(""),
48 "SETXY" => Some("EE"),
49 "LABEL" | "PRINT" => Some("P"),
50 "REPEAT" | "WHILE" | "IF" => Some("EL"),
51 "IFELSE" => Some("ELL"),
52 "FOR" => Some("QEEL"),
53 "IFTRUE" | "IFFALSE" => Some("L"),
54 "MAKE" | "LOCALMAKE" => Some("QE"),
55 "LOCAL" | "TRACE" | "UNTRACE" => Some("Q"),
56 _ => None,
57 }
58}
59
60fn inputs(s: &str) -> Option<usize> {
61 match s {
62 "XCOR" | "YCOR" | "HEADING" | "REPCOUNT" => Some(0),
63 "RANDOM" | "ABS" | "NOT" => Some(1),
64 "AND" | "OR" | "MAX" | "MIN" => Some(2),
65 _ => None,
66 }
67}
68
69fn primitive(s: &str) -> bool {
70 signature(s).is_some() || inputs(s).is_some() || ["TO", "END", "MOD"].contains(&s)
71}
72
73fn name(s: &str) -> bool {
74 !s.is_empty()
75 && s.as_bytes()[0].is_ascii_uppercase()
76 && s.bytes()
77 .all(|c| c.is_ascii_uppercase() || c.is_ascii_digit() || c == b'_')
78}

80fn lex(text: &str, line: usize) -> Result<Vec<Token>, Problem> {
81 let s = text.split(';').next().unwrap().to_ascii_uppercase();
82 let b = s.as_bytes();
83 let mut i = 0;
84 let mut out = Vec::new();
85 while i < b.len() {
86 let c = b[i];
87 if c == b' ' {
88 i += 1;
89 continue;
90 }
91 if b"[]()+-*/=<>".contains(&c) {
92 let start = i;
93 let kind = if c == b'-'
94 && (i == 0 || b" [(".contains(&b[i - 1]))
95 && i + 1 < b.len()
96 && b[i + 1] != b' '
97 {
98 "sign"
99 } else {
100 "symbol"
101 };
102 if (c == b'<' && i + 1 < b.len() && b"=>".contains(&b[i + 1]))
103 || (c == b'>' && i + 1 < b.len() && b[i + 1] == b'=')
104 {
105 i += 1;
106 }
107 i += 1;
108 out.push(Token {
109 text: s[start..i].into(),
110 kind,
111 line,
112 });
113 continue;
114 }
115 let start = i;
116 while i < b.len() && !b" []()+-*/=<>".contains(&b[i]) {
117 i += 1;
118 }
119 let word = &s[start..i];
120 let kind = if word.bytes().all(|c| c.is_ascii_digit()) {
121 "number"
122 } else if word.starts_with(':') && name(&word[1..]) {
123 "variable"
124 } else if word.starts_with('"') && name(&word[1..]) {
125 "quote"
126 } else if name(word) {
127 "word"
128 } else {
129 return Err(Problem {
130 line,
131 reason: format!("bad token '{}'", word),
132 });
133 };
134 out.push(Token {
135 text: word.into(),
136 kind,
137 line,
138 });
139 }
140 Ok(out)
141}

173struct Parser<'a> {
174 tokens: &'a [Token],
175 procedures: &'a BTreeMap<String, Procedure>,
176 inside: bool,
177 at: usize,
178}
179
180impl Parser<'_> {
181 fn unknown(&self, t: &Token) -> bool {
182 t.kind == "word" && !primitive(&t.text) && !self.procedures.contains_key(&t.text)
183 }
184
185 fn next(&self, owner: &Token) -> Result<Token, Problem> {
186 let t = self
187 .tokens
188 .get(self.at)
189 .ok_or_else(|| problem(owner, format!("not enough inputs to {}", owner.text)))?;
190 if t.text == "]" || t.text == ")" {
191 return Err(problem(
192 owner,
193 format!("not enough inputs to {}", owner.text),
194 ));
195 }
196 if self.unknown(t) {
197 return Err(problem(t, format!("I don't know how to {}", t.text)));
198 }
199 Ok(t.clone())
200 }
201
202 fn rejected(owner: &Token, t: &Token) -> Problem {
203 problem(
204 t,
205 format!("{} doesn't like {} as input", owner.text, t.text),
206 )
207 }
208
209 fn expression(&mut self, owner: &Token) -> Result<Expr, Problem> {
210 self.binary(owner, 0)
211 }
212
213 fn binary(&mut self, owner: &Token, level: usize) -> Result<Expr, Problem> {
214 if level == 3 {
215 return self.term(owner);
216 }
217 let operators: &[&str] = match level {
218 0 => &["=", "<>", "<", ">", "<=", ">="],
219 1 => &["+", "-"],
220 _ => &["*", "/", "MOD"],
221 };
222 let mut left = self.binary(owner, level + 1)?;
223 while self.at < self.tokens.len()
224 && operators.contains(&self.tokens[self.at].text.as_str())
225 && self.tokens[self.at].kind != "sign"
226 {
227 let t = self.tokens[self.at].clone();
228 self.at += 1;
229 let right = self.binary(&t, level + 1)?;
230 left = Expr {
231 op: t.text.clone(),
232 token: t,
233 args: vec![left, right],
234 };
235 }
236 Ok(left)
237 }
238
239 fn expressions(&mut self, owner: &Token, count: usize) -> Result<Vec<Expr>, Problem> {
240 (0..count).map(|_| self.expression(owner)).collect()
241 }
242
243 fn term(&mut self, owner: &Token) -> Result<Expr, Problem> {
244 let t = self.next(owner)?;
245 self.at += 1;
246 let (op, args) = if t.kind == "number" {
247 ("NUMBER", Vec::new())
248 } else if t.kind == "variable" {
249 ("VARIABLE", Vec::new())
250 } else if t.text == "-" {
251 ("NEG", vec![self.term(&t)?])
252 } else if t.text == "(" {
253 let e = self.expression(&t)?;
254 if self.tokens.get(self.at).map(|t| t.text.as_str()) != Some(")") {
255 return Err(problem(&t, "missing )"));
256 }
257 self.at += 1;
258 return Ok(e);
259 } else if t.kind == "word" && inputs(&t.text).is_some() {
260 ("FUNCTION", self.expressions(&t, inputs(&t.text).unwrap())?)
261 } else if t.kind == "word" && self.procedures.contains_key(&t.text) {
262 (
263 "CALL",
264 self.expressions(&t, self.procedures[&t.text].params.len())?,
265 )
266 } else {
267 return Err(Self::rejected(owner, &t));
268 };
269 Ok(Expr {
270 token: t,
271 op: op.into(),
272 args,
273 })
274 }
275
276 fn item(&mut self, owner: &Token) -> Result<Item, Problem> {
277 let t = self.next(owner)?;
278 if t.kind == "quote" {
279 self.at += 1;
280 Ok(Item::Word(t.text[1..].into()))
281 } else {
282 Ok(Item::Expression(self.expression(owner)?))
283 }
284 }
285
286 fn list(&mut self, owner: &Token) -> Result<Vec<Command>, Problem> {
287 let open = self.next(owner)?;
288 if open.text != "[" {
289 return Err(Self::rejected(owner, &open));
290 }
291 self.at += 1;
292 let mut out = Vec::new();
293 while self.at < self.tokens.len() && self.tokens[self.at].text != "]" {
294 out.push(self.command()?);
295 }
296 if self.at == self.tokens.len() {
297 return Err(problem(&open, "missing ]"));
298 }
299 self.at += 1;
300 Ok(out)
301 }
302
303 fn command(&mut self) -> Result<Command, Problem> {
304 let t = self.tokens[self.at].clone();
305 self.at += 1;
306 if t.text == "]" {
307 return Err(problem(&t, "unexpected ]"));
308 }
309 let mut result = Command {
310 token: t.clone(),
311 op: String::new(),
312 exprs: Vec::new(),
313 words: Vec::new(),
314 lists: Vec::new(),
315 items: Vec::new(),
316 };
317 if t.text == "("
318 && self
319 .tokens
320 .get(self.at)
321 .is_some_and(|t| t.kind == "word" && canonical(&t.text) == "PRINT")
322 {
323 let owner = self.tokens[self.at].clone();
324 self.at += 1;
325 while self.tokens.get(self.at).map(|t| t.text.as_str()) != Some(")") {
326 if self.at == self.tokens.len() || self.tokens[self.at].text == "]" {
327 return Err(problem(&t, "missing )"));
328 }
329 result.items.push(self.item(&owner)?);
330 }
331 self.at += 1;
332 result.op = "MULTIPRINT".into();
333 return Ok(result);
334 }
335 if t.kind == "word" && ["TO", "END"].contains(&t.text.as_str()) {
336 return Err(problem(&t, format!("misplaced {}", t.text)));
337 }
338 if t.kind == "word" {
339 if let Some(sig) = signature(&t.text) {
340 let op = canonical(&t.text);
341 if !self.inside && ["STOP", "OUTPUT", "LOCAL", "LOCALMAKE"].contains(&op) {
342 return Err(problem(
343 &t,
344 format!("can only use {} inside a procedure", t.text),
345 ));
346 }
347 for kind in sig.chars() {
348 match kind {
349 'E' => result.exprs.push(self.expression(&t)?),
350 'L' => result.lists.push(self.list(&t)?),
351 'P' => result.items.push(self.item(&t)?),
352 'Q' => {
353 let q = self.next(&t)?;
354 if q.kind != "quote" {
355 return Err(Self::rejected(&t, &q));
356 }
357 self.at += 1;
358 if ["TRACE", "UNTRACE"].contains(&op)
359 && !self.procedures.contains_key(&q.text[1..])
360 {
361 return Err(problem(
362 &q,
363 format!("I don't know how to {}", &q.text[1..]),
364 ));
365 }
366 result.words.push(q.text[1..].into());
367 }
368 _ => unreachable!(),
369 }
370 }
371 result.op = op.into();
372 return Ok(result);
373 }
374 if let Some(p) = self.procedures.get(&t.text) {
375 result.exprs = self.expressions(&t, p.params.len())?;
376 result.op = "CALL".into();
377 return Ok(result);
378 }
379 }
380 if self.unknown(&t) {
381 return Err(problem(&t, format!("I don't know how to {}", t.text)));
382 }
383 Err(problem(
384 &t,
385 format!("you don't say what to do with {}", t.text),
386 ))
387 }
388
389 fn sequence(&mut self) -> Result<Vec<Command>, Problem> {
390 let mut out = Vec::new();
391 while self.at < self.tokens.len() {
392 out.push(self.command()?);
393 }
394 Ok(out)
395 }
396}

398struct Environment {
399 values: BTreeMap<String, i64>,
400 test: Option<bool>,
401}
402
403enum Signal {
404 Failure(Problem),
405 Stop,
406 Output(i64),
407}
408type Running<T> = Result<T, Signal>;
409
410fn runtime_error(t: &Token, reason: impl Into<String>) -> Signal {
411 Signal::Failure(problem(t, reason))
412}
413
414struct Runtime<'a> {
415 procedures: &'a BTreeMap<String, Procedure>,
416 output: Vec<String>,
417 x: i64,
418 y: i64,
419 heading: i64,
420 pen: &'static str,
421 character: char,
422 random: i64,
423 canvas: BTreeMap<(i64, i64), char>,
424 envs: Vec<Environment>,
425 calls: Vec<String>,
426 repetitions: Vec<i64>,
427 traced: BTreeSet<String>,
428}
429
430impl Runtime<'_> {
431 fn variable(&self, t: &Token) -> Running<i64> {
432 for env in self.envs.iter().rev() {
433 if let Some(&v) = env.values.get(&t.text[1..]) {
434 return Ok(v);
435 }
436 }
437 Err(runtime_error(t, format!("{} has no value", &t.text[1..])))
438 }
439
440 fn make(&mut self, name: &str, value: i64) {
441 for i in (1..self.envs.len()).rev() {
442 if self.envs[i].values.contains_key(name) {
443 self.envs[i].values.insert(name.into(), value);
444 return;
445 }
446 }
447 self.envs[0].values.insert(name.into(), value);
448 }
449
450 fn invoke(&mut self, t: &Token, values: &[i64]) -> Running<Option<i64>> {
451 if self.calls.len() == 100 {
452 return Err(runtime_error(t, "recursion too deep"));
453 }
454 let p = &self.procedures[&t.text];
455 let trace = self.traced.contains(&t.text);
456 let indent = " ".repeat(self.calls.len());
457 if trace {
458 self.output.push(format!(
459 "{}( {}{} )",
460 indent,
461 t.text,
462 values.iter().map(|v| format!(" {}", v)).collect::<String>()
463 ));
464 }
465 let body = p.body.clone();
466 self.envs.push(Environment {
467 values: p
468 .params
469 .iter()
470 .cloned()
471 .zip(values.iter().copied())
472 .collect(),
473 test: None,
474 });
475 self.calls.push(t.text.clone());
476 let value = match self.execute(&body) {
477 Ok(()) | Err(Signal::Stop) => None,
478 Err(Signal::Output(v)) => Some(v),
479 Err(e) => return Err(e),
480 };
481 self.envs.pop();
482 self.calls.pop();
483 if trace {
484 self.output.push(format!(
485 "{}{}{}",
486 indent,
487 t.text,
488 value.map_or(" stops".into(), |v| format!(" outputs {}", v))
489 ));
490 }
491 Ok(value)
492 }
493
494 fn evaluate(&mut self, e: &Expr) -> Running<i64> {
495 let t = &e.token;
496 if e.op == "NUMBER" {
497 return Ok(t.text.parse().unwrap());
498 }
499 if e.op == "VARIABLE" {
500 return self.variable(t);
501 }
502 let mut a = Vec::new();
503 for arg in &e.args {
504 a.push(self.evaluate(arg)?);
505 }
506 if e.op == "NEG" {
507 return Ok(-a[0]);
508 }
509 if e.op == "CALL" {
510 return self
511 .invoke(t, &a)?
512 .ok_or_else(|| runtime_error(t, format!("{} did not output", t.text)));
513 }
514 if e.op == "FUNCTION" {
515 return Ok(match t.text.as_str() {
516 "XCOR" => self.x,
517 "YCOR" => self.y,
518 "HEADING" => self.heading,
519 "REPCOUNT" => *self
520 .repetitions
521 .last()
522 .ok_or_else(|| runtime_error(t, "REPCOUNT outside REPEAT"))?,
523 "RANDOM" => {
524 if a[0] <= 0 {
525 return Err(runtime_error(
526 t,
527 format!("RANDOM doesn't like {} as input", a[0]),
528 ));
529 }
530 self.random = 48271 * self.random % 2147483647;
531 self.random % a[0]
532 }
533 "ABS" => a[0].abs(),
534 "NOT" => i64::from(a[0] == 0),
535 "AND" => i64::from(a[0] != 0 && a[1] != 0),
536 "OR" => i64::from(a[0] != 0 || a[1] != 0),
537 "MAX" => a[0].max(a[1]),
538 "MIN" => a[0].min(a[1]),
539 _ => unreachable!(),
540 });
541 }
542 if (e.op == "/" || e.op == "MOD") && a[1] == 0 {
543 return Err(runtime_error(
544 t,
545 format!("{} doesn't like 0 as input", t.text),
546 ));
547 }
548 Ok(match e.op.as_str() {
549 "+" => a[0] + a[1],
550 "-" => a[0] - a[1],
551 "*" => a[0] * a[1],
552 "/" => a[0] / a[1],
553 "MOD" => a[0] % a[1],
554 "=" => i64::from(a[0] == a[1]),
555 "<>" => i64::from(a[0] != a[1]),
556 "<" => i64::from(a[0] < a[1]),
557 ">" => i64::from(a[0] > a[1]),
558 "<=" => i64::from(a[0] <= a[1]),
559 ">=" => i64::from(a[0] >= a[1]),
560 _ => unreachable!(),
561 })
562 }
563
564 fn item(&mut self, item: &Item) -> Running<String> {
565 match item {
566 Item::Word(word) => Ok(word.clone()),
567 Item::Expression(e) => Ok(self.evaluate(e)?.to_string()),
568 }
569 }

571 fn direction(&self) -> (i64, i64) {
572 [
573 (0, 1),
574 (1, 1),
575 (1, 0),
576 (1, -1),
577 (0, -1),
578 (-1, -1),
579 (-1, 0),
580 (-1, 1),
581 ][(self.heading / 45) as usize]
582 }

584 fn apply(&mut self, x: i64, y: i64) {
585 if self.pen == "DOWN" {
586 self.canvas.insert((x, y), self.character);
587 } else if self.pen == "ERASE" {
588 self.canvas.remove(&(x, y));
589 }
590 }
591
592 fn move_forward(&mut self, n: i64) {
593 let (dx, dy) = self.direction();
594 let sign = if n < 0 { -1 } else { 1 };
595 self.apply(self.x, self.y);
596 for _ in 0..n.abs() {
597 self.x += sign * dx;
598 self.y += sign * dy;
599 self.apply(self.x, self.y);
600 }
601 }
602
603 fn straight(&mut self, x: i64, y: i64) {
604 let (dx, dy) = (x - self.x, y - self.y);
605 let s = dx.abs().max(dy.abs());
606 if s == 0 {
607 self.apply(self.x, self.y);
608 } else {
609 for i in 0..=s {
610 self.apply(
611 self.x + (2 * i * dx + s).div_euclid(2 * s),
612 self.y + (2 * i * dy + s).div_euclid(2 * s),
613 );
614 }
615 }
616 self.x = x;
617 self.y = y;
618 }

620 fn bounds(&self) -> Option<(i64, i64, i64, i64)> {
621 if self.canvas.is_empty() {
622 return None;
623 }
624 Some((
625 self.canvas.keys().map(|p| p.0).min().unwrap(),
626 self.canvas.keys().map(|p| p.1).min().unwrap(),
627 self.canvas.keys().map(|p| p.0).max().unwrap(),
628 self.canvas.keys().map(|p| p.1).max().unwrap(),
629 ))
630 }

632 fn fill(&mut self) {
633 let Some((x1, y1, x2, y2)) = self.bounds() else {
634 return;
635 };
636 if self.canvas.contains_key(&(self.x, self.y))
637 || self.x < x1
638 || self.x > x2
639 || self.y < y1
640 || self.y > y2
641 {
642 return;
643 }
644 let mut queue = VecDeque::from(vec![(self.x, self.y)]);
645 self.canvas.insert((self.x, self.y), self.character);
646 while let Some((x, y)) = queue.pop_front() {
647 for (dx, dy) in [(0, 1), (1, 0), (0, -1), (-1, 0)].iter().copied() {
648 let p = (x + dx, y + dy);
649 if x1 <= p.0 && p.0 <= x2 && y1 <= p.1 && p.1 <= y2 && !self.canvas.contains_key(&p)
650 {
651 self.canvas.insert(p, self.character);
652 queue.push_back(p);
653 }
654 }
655 }
656 }
657
658 fn picture(&mut self) {
659 let Some((x1, y1, x2, y2)) = self.bounds() else {
660 self.output.push("CANVAS 0 0 0".into());
661 return;
662 };
663 self.output.push(format!(
664 "CANVAS {} {} {}",
665 x2 - x1 + 1,
666 y2 - y1 + 1,
667 self.canvas.len()
668 ));
669 for y in (y1..=y2).rev() {
670 let line: String = (x1..=x2)
671 .map(|x| self.canvas.get(&(x, y)).copied().unwrap_or(' '))
672 .collect();
673 self.output.push(line.trim_end_matches(' ').into());
674 }
675 }

678 for c in commands {
679 self.command(c)?;
680 }
681 Ok(())

684 fn command(&mut self, c: &Command) -> Running<()> {
685 let t = &c.token;
686 let op = c.op.as_str();
687 if op == "WHILE" {
688 while self.evaluate(&c.exprs[0])? != 0 {
689 self.execute(&c.lists[0])?;
690 }
691 return Ok(());
692 }
693 if op == "PRINT" || op == "MULTIPRINT" {
694 let mut values = Vec::new();
695 for item in &c.items {
696 values.push(self.item(item)?);
697 }
698 self.output.push(values.join(" "));
699 return Ok(());
700 }
701 if op == "LABEL" {
702 let text = self.item(&c.items[0])?;
703 let (dx, dy) = self.direction();
704 for (i, ch) in text.chars().enumerate() {
705 self.canvas
706 .insert((self.x + i as i64 * dx, self.y + i as i64 * dy), ch);
707 }
708 return Ok(());
709 }
710 let mut a = Vec::new();
711 for e in &c.exprs {
712 a.push(self.evaluate(e)?);
713 }
714 match op {
715 "CALL" => {
716 if let Some(v) = self.invoke(t, &a)? {
717 return Err(runtime_error(
718 t,
719 format!("you don't say what to do with {}", v),
720 ));
721 }
722 }
723 "FD" => self.move_forward(a[0]),
724 "BK" => self.move_forward(-a[0]),
725 "RT" | "LT" | "SETH" => {
726 let angle = a[0];
727 if angle % 45 != 0 {
728 return Err(runtime_error(
729 t,
730 format!("{} doesn't like {} as input", t.text, angle),
731 ));
732 }
733 self.heading = (if op == "SETH" {
734 angle
735 } else {
736 self.heading + if op == "RT" { angle } else { -angle }
737 })
738 .rem_euclid(360);
739 }
740 "PU" => self.pen = "UP",
741 "PD" => self.pen = "DOWN",
742 "PE" => self.pen = "ERASE",
743 "SETPEN" => {
744 if !(33..=126).contains(&a[0]) {
745 return Err(runtime_error(
746 t,
747 format!("SETPEN doesn't like {} as input", a[0]),
748 ));
749 }
750 self.character = a[0] as u8 as char;
751 }
752 "SETXY" => self.straight(a[0], a[1]),
753 "SETX" => self.straight(a[0], self.y),
754 "SETY" => self.straight(self.x, a[0]),
755 "HOME" => {
756 self.straight(0, 0);
757 self.heading = 0;
758 }
759 "CS" => {
760 self.canvas.clear();
761 self.x = 0;
762 self.y = 0;
763 self.heading = 0;
764 }
765 "CLEAN" => self.canvas.clear(),
766 "FILL" => self.fill(),
767 "SNAPSHOT" => self.picture(),
768 "MAKE" => self.make(&c.words[0], a[0]),
769 "LOCAL" => {
770 self.envs
771 .last_mut()
772 .unwrap()
773 .values
774 .insert(c.words[0].clone(), 0);
775 }
776 "LOCALMAKE" => {
777 self.envs
778 .last_mut()
779 .unwrap()
780 .values
781 .insert(c.words[0].clone(), a[0]);
782 }
783 "TRACE" => {
784 self.traced.insert(c.words[0].clone());
785 }
786 "UNTRACE" => {
787 self.traced.remove(&c.words[0]);
788 }
789 "SEED" => {
790 if a[0] <= 0 {
791 return Err(runtime_error(
792 t,
793 format!("SEED doesn't like {} as input", a[0]),
794 ));
795 }
796 self.random = a[0];
797 }
798 "STOP" => return Err(Signal::Stop),
799 "OUTPUT" => return Err(Signal::Output(a[0])),
800 "REPEAT" => {
801 if c.lists[0].is_empty() {
802 return Ok(());
803 }
804 for i in 1..=a[0] {
805 self.repetitions.push(i);
806 let result = self.execute(&c.lists[0]);
807 self.repetitions.pop();
808 result?;
809 }
810 }
811 "FOR" => {
812 if c.lists[0].is_empty() {
813 self.make(&c.words[0], a[1]);
814 return Ok(());
815 }
816 let step = if a[0] <= a[1] { 1 } else { -1 };
817 let mut i = a[0];
818 loop {
819 self.make(&c.words[0], i);
820 self.execute(&c.lists[0])?;
821 if i == a[1] {
822 break;
823 }
824 i += step;
825 }
826 }
827 "IF" => {
828 if a[0] != 0 {
829 self.execute(&c.lists[0])?;
830 }
831 }
832 "IFELSE" => self.execute(&c.lists[if a[0] != 0 { 0 } else { 1 }])?,
833 "TEST" => self.envs.last_mut().unwrap().test = Some(a[0] != 0),
834 "IFTRUE" | "IFFALSE" => {
835 let test = self
836 .envs
837 .last()
838 .unwrap()
839 .test
840 .ok_or_else(|| runtime_error(t, format!("{} without TEST", t.text)))?;
841 if test == (op == "IFTRUE") {
842 self.execute(&c.lists[0])?;
843 }
844 }
845 _ => unreachable!(),
846 }
847 Ok(())
848 }

850 fn errors(&mut self) {
851 let mut i = self.calls.len();
852 while i > 0 {
853 let p = &self.calls[i - 1];
854 let mut count = 0;
855 while i > 0 && self.calls[i - 1] == *p {
856 count += 1;
857 i -= 1;
858 }
859 self.output.push(format!(
860 " in {}{}",
861 p,
862 if count == 1 {
863 String::new()
864 } else {
865 format!(" x{}", count)
866 }
867 ));
868 }
869 }

872fn prepare(
873 source: &[(usize, &str)],
874) -> Result<(BTreeMap<String, Procedure>, Vec<Command>), Problem> {
875 let mut procedures: BTreeMap<String, Procedure> = BTreeMap::new();
876 let mut order = Vec::new();
877 let mut current: Option<String> = None;
878 let mut top = Vec::new();
879 for &(line, text) in source {
880 let tokens = lex(text, line)?;
881 if tokens.is_empty() {
882 continue;
883 }
884 let start = &tokens[0];
885 if start.kind == "word" && start.text == "TO" {
886 if let Some(p) = &current {
887 return Err(problem(start, format!("TO inside definition of {}", p)));
888 }
889 if tokens.len() < 2 || tokens[1].kind != "word" {
890 return Err(problem(start, "bad procedure header"));
891 }
892 let name = &tokens[1];
893 if primitive(&name.text) {
894 return Err(problem(start, format!("{} is a primitive", name.text)));
895 }
896 if procedures.contains_key(&name.text) {
897 return Err(problem(start, format!("{} is already defined", name.text)));
898 }
899 let mut params = Vec::new();
900 for t in &tokens[2..] {
901 if t.kind != "variable" {
902 return Err(problem(start, "bad procedure header"));
903 }
904 let input = t.text[1..].to_string();
905 if params.contains(&input) {
906 return Err(problem(start, format!("duplicate input {}", t.text)));
907 }
908 params.push(input);
909 }
910 current = Some(name.text.clone());
911 order.push(name.text.clone());
912 procedures.insert(
913 name.text.clone(),
914 Procedure {
915 token: name.clone(),
916 params,
917 raw: Vec::new(),
918 body: Rc::new(Vec::new()),
919 },
920 );
921 } else if tokens.len() == 1 && start.kind == "word" && start.text == "END" {
922 if current.is_none() {
923 return Err(problem(start, "END without TO"));
924 }
925 current = None;
926 } else if let Some(p) = &current {
927 procedures.get_mut(p).unwrap().raw.extend(tokens);
928 } else {
929 top.extend(tokens);
930 }
931 }
932 if let Some(p) = current {
933 return Err(problem(
934 &procedures[&p].token,
935 format!("missing END for {}", p),
936 ));
937 }
938 for p in order {
939 let raw = procedures[&p].raw.clone();
940 let body = Parser {
941 tokens: &raw,
942 procedures: &procedures,
943 inside: true,
944 at: 0,
945 }
946 .sequence()?;
947 procedures.get_mut(&p).unwrap().body = Rc::new(body);
948 }
949 let body = Parser {
950 tokens: &top,
951 procedures: &procedures,
952 inside: false,
953 at: 0,
954 }
955 .sequence()?;
956 Ok((procedures, body))
957}

959fn main() {
960 let mut input = String::new();
961 io::stdin().read_to_string(&mut input).unwrap();
962 let lines: Vec<&str> = input.lines().collect();
963 let mut output = Vec::new();
964 let mut at = 0;
965 while at < lines.len() && lines[at] != "#EOF" {
966 output.push(format!("PROGRAM {}", &lines[at][9..]));
967 at += 1;
968 let mut source = Vec::new();
969 while at < lines.len() && lines[at] != "#EOF" && !lines[at].starts_with("#PROGRAM ") {
970 source.push((at + 1, lines[at]));
971 at += 1;
972 }
973 match prepare(&source) {
974 Err(e) => output.push(format!("ERROR line {}: {}", e.line, e.reason)),
975 Ok((procedures, body)) => {
976 let mut runtime = Runtime {
977 procedures: &procedures,
978 output: Vec::new(),
979 x: 0,
980 y: 0,
981 heading: 0,
982 pen: "DOWN",
983 character: '*',
984 random: 1,
985 canvas: BTreeMap::new(),
986 envs: vec![Environment {
987 values: BTreeMap::new(),
988 test: None,
989 }],
990 calls: Vec::new(),
991 repetitions: Vec::new(),
992 traced: BTreeSet::new(),
993 };
994 match runtime.execute(&body) {
995 Ok(()) => {
996 runtime.output.push(format!(
997 "TURTLE {} {} {} {} {}",
998 runtime.x, runtime.y, runtime.heading, runtime.pen, runtime.character
999 ));
1000 runtime.picture();
1001 }
1002 Err(Signal::Failure(e)) => {
1003 runtime
1004 .output
1005 .push(format!("ERROR line {}: {}", e.line, e.reason));
1006 runtime.errors();
1007 }
1008 _ => unreachable!(),
1009 }
1010 output.extend(runtime.output);
1011 }
1012 }
1013 }
1014 for line in output {
1015 println!("{}", line);
1016 }
1017}
