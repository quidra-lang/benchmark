28lex <- function(line, number) {
29 semicolon <- regexpr(";", line, fixed = TRUE)[1]
30 if (semicolon > 0) line <- substr(line, 1, semicolon - 1L)
31 text <- chartr("abcdefghijklmnopqrstuvwxyz", "ABCDEFGHIJKLMNOPQRSTUVWXYZ", line)
32 chars <- strsplit(text, "", fixed = TRUE)[[1]]
33 result <- list()
34 i <- 1L
35 delimiters <- c(" ", "[", "]", "(", ")", "+", "-", "*", "/", "=", "<", ">")
36 while (i <= length(chars)) {
37 char <- chars[i]
38 if (char == " ") {
39 i <- i + 1L
40 } else if (char %in% delimiters[-1]) {
41 next_char <- if (i < length(chars)) chars[i + 1L] else ""
42 spelling <- char
43 if ((char == "<" && next_char %in% c("=", ">")) || (char == ">" && next_char == "=")) spelling <- paste0(spelling, next_char)
44 sign <- char == "-" && (i == 1L || chars[i - 1L] %in% c(" ", "[", "(")) && nzchar(next_char) && next_char != " "
45 result[[length(result) + 1L]] <- new_token("symbol", spelling, number, sign)
46 i <- i + nchar(spelling)
47 } else {
48 start <- i
49 while (i <= length(chars) && !(chars[i] %in% delimiters)) i <- i + 1L
50 spelling <- paste(chars[seq.int(start, i - 1L)], collapse = "")
51 kind <- if (grepl("\\A[0-9]+\\z", spelling, perl = TRUE)) "number" else if (grepl("\\A:[A-Z][A-Z0-9_]*\\z", spelling, perl = TRUE)) "variable" else if (grepl("\\A\"[A-Z][A-Z0-9_]*\\z", spelling, perl = TRUE)) "quote" else if (grepl("\\A[A-Z][A-Z0-9_]*\\z", spelling, perl = TRUE)) "word" else NULL
52 if (is.null(kind)) logo_error(number, paste0("bad token '", spelling, "'"))
53 result[[length(result) + 1L]] <- new_token(kind, spelling, number)
54 }
55 }
56 result
57}

59parse <- function(tokens, procedures, top) {
60 index <- 1L
61 peek <- function() if (index <= length(tokens)) tokens[[index]] else NULL
62 read <- function() {
63 token <- tokens[[index]]
64 index <<- index + 1L
65 token
66 }
67 error <- function(token, reason) logo_error(token$line, reason)
68 unknown <- function(token) token$kind == "word" && !primitive(token$text) && is.null(procedures[[token$text]])
69 need <- function(owner) {
70 token <- peek()
71 if (is.null(token) || token$text %in% c("]", ")")) error(owner, paste0("not enough inputs to ", owner$text))
72 if (unknown(token)) error(token, paste0("I don't know how to ", token$text))
73 token
74 }
75 read_input <- function(kind, owner) {
76 token <- need(owner)
77 if (kind == "L") {
78 if (token$text != "[") error(token, paste0(owner$text, " doesn't like ", token$text, " as input"))
79 opening <- read()
80 result <- list()
81 repeat {
82 if (is.null(peek())) error(opening, "missing ]")
83 if (peek()$text == "]") break
84 result[[length(result) + 1L]] <- command()
85 }
86 read()
87 return(result)
88 }
89 if (kind == "Q") {
90 if (token$kind != "quote") error(token, paste0(owner$text, " doesn't like ", token$text, " as input"))
91 return(read())
92 }
93 if (kind == "P" && token$kind == "quote") return(list(kind = "quote", value = substring(read()$text, 2)))
94 expression(owner)
95 }
96 expressions <- function(count, owner) lapply(seq_len(count), function(i) expression(owner))
97 command <- function() {
98 token <- read()
99 if (token$text == "]") error(token, "unexpected ]")
100 if (token$text == "(" && !is.null(peek()) && peek()$text %in% c("PRINT", "PR")) {
101 owner <- read()
102 args <- list()
103 repeat {
104 if (is.null(peek()) || peek()$text == "]") error(token, "missing )")
105 if (peek()$text == ")") break
106 args[[length(args) + 1L]] <- read_input("P", owner)
107 }
108 read()
109 return(list(name = "MULTIPRINT", token = owner, args = args))
110 }
111 if (token$kind == "word" && token$text %in% c("TO", "END")) error(token, paste0("misplaced ", token$text))
112 if (token$kind == "word" && token$text %in% names(commands)) {
113 name <- commands[[token$text]][1]
114 if (top && token$text %in% c("STOP", "OUTPUT", "OP", "LOCAL", "LOCALMAKE")) error(token, paste0("can only use ", token$text, " inside a procedure"))
115 kinds <- strsplit(commands[[token$text]][2], "", fixed = TRUE)[[1]]
116 args <- lapply(kinds, function(kind) read_input(kind, token))
117 if (name %in% c("TRACE", "UNTRACE") && is.null(procedures[[substring(args[[1]]$text, 2)]])) error(args[[1]], paste0("I don't know how to ", substring(args[[1]]$text, 2)))
118 return(list(name = name, token = token, args = args))
119 }
120 if (token$kind == "word" && !is.null(procedures[[token$text]])) return(list(name = "call", token = token, args = expressions(length(procedures[[token$text]]$parameters), token)))
121 if (unknown(token)) error(token, paste0("I don't know how to ", token$text))
122 error(token, paste0("you don't say what to do with ", token$text))
123 }
124 binary <- function(owner, lower, operators) {
125 left <- lower(owner)
126 repeat {
127 next_token <- peek()
128 if (is.null(next_token) || !(next_token$text %in% operators) || (next_token$text == "-" && next_token$sign)) break
129 operator <- read()
130 right <- lower(operator)
131 left <- list(kind = "binary", token = operator, left = left, right = right)
132 }
133 left
134 }
135 expression <- function(owner) binary(owner, parse_sum, c("=", "<>", "<", ">", "<=", ">="))
136 parse_sum <- function(owner) binary(owner, product, c("+", "-"))
137 product <- function(owner) binary(owner, term, c("*", "/", "MOD"))
138 term <- function(owner) {
139 token <- need(owner)
140 read()
141 if (token$kind == "number") return(list(kind = "number", value = as.numeric(token$text)))
142 if (token$kind == "variable") return(list(kind = "variable", token = token))
143 if (token$text == "-") return(list(kind = "unary", value = term(token)))
144 if (token$text == "(") {
145 value <- expression(token)
146 if (is.null(peek()) || peek()$text != ")") error(token, "missing )")
147 read()
148 return(value)
149 }
150 if (token$kind == "word" && token$text %in% names(functions)) return(list(kind = "function", token = token, args = expressions(functions[[token$text]], token)))
151 if (token$kind == "word" && !is.null(procedures[[token$text]])) return(list(kind = "call", token = token, args = expressions(length(procedures[[token$text]]$parameters), token)))
152 error(token, paste0(owner$text, " doesn't like ", token$text, " as input"))
153 }
154 result <- list()
155 while (!is.null(peek())) result[[length(result) + 1L]] <- command()
156 result
157}

165make_turtle <- function(procedures) {
166 state <- new.env(parent = emptyenv())
167 state$x <- state$y <- state$heading <- 0
168 state$pen <- "DOWN"
169 state$character <- "*"
170 state$canvas <- new.env(hash = TRUE, parent = emptyenv())
171 state$environments <- list(new_environment())
172 state$calls <- character()
173 state$repetitions <- numeric()
174 state$traced <- character()
175 state$random <- 1
176 directions <- list(c(0, 1), c(1, 1), c(1, 0), c(1, -1), c(0, -1), c(-1, -1), c(-1, 0), c(-1, 1))
177 error <- function(token, reason) logo_error(token$line, reason, state$calls)
178 value <- function(node) {
179 kind <- node$kind
180 if (kind %in% c("number", "quote")) return(node$value)
181 if (kind == "variable") {
182 name <- substring(node$token$text, 2)
183 for (i in rev(seq_along(state$environments))) {
184 result <- state$environments[[i]]$variables[[name]]
185 if (!is.null(result)) return(result)
186 }
187 error(node$token, paste0(name, " has no value"))
188 }
189 if (kind == "unary") return(-value(node$value))
190 if (kind == "binary") {
191 a <- value(node$left)
192 b <- value(node$right)
193 operator <- node$token$text
194 if (operator %in% c("/", "MOD") && b == 0) error(node$token, paste0(operator, " doesn't like 0 as input"))
195 return(switch(operator,
196 "+" = a + b, "-" = a - b, "*" = a * b, "/" = trunc(a / b), MOD = a - b * trunc(a / b),
197 "=" = as.numeric(a == b), "<>" = as.numeric(a != b), "<" = as.numeric(a < b),
198 ">" = as.numeric(a > b), "<=" = as.numeric(a <= b), ">=" = as.numeric(a >= b)))
199 }
200 if (kind == "call") {
201 result <- call(node$token, node$args)
202 if (is.null(result)) error(node$token, paste0(node$token$text, " did not output"))
203 return(result)
204 }
205 args <- lapply(node$args, value)
206 token <- node$token
207 a <- if (length(args) >= 1) args[[1]] else NULL
208 b <- if (length(args) >= 2) args[[2]] else NULL
209 if (token$text == "XCOR") return(state$x)
210 if (token$text == "YCOR") return(state$y)
211 if (token$text == "HEADING") return(state$heading)
212 if (token$text == "REPCOUNT") {
213 if (length(state$repetitions) == 0) error(token, "REPCOUNT outside REPEAT")
214 return(tail(state$repetitions, 1))
215 }
216 if (token$text == "RANDOM") {
217 if (a <= 0) error(token, paste0("RANDOM doesn't like ", a, " as input"))
218 state$random <- (48271 * state$random) %% 2147483647
219 return(state$random %% a)
220 }
221 if (token$text == "ABS") return(abs(a))
222 if (token$text == "NOT") return(as.numeric(a == 0))
223 if (token$text == "AND") return(as.numeric(a != 0 && b != 0))
224 if (token$text == "OR") return(as.numeric(a != 0 || b != 0))
225 if (token$text == "MAX") return(max(a, b))
226 if (token$text == "MIN") return(min(a, b))
227 }
228 call <- function(token, arguments) {
229 inputs <- lapply(arguments, value)
230 if (length(state$calls) == 100) error(token, "recursion too deep")
231 procedure <- procedures[[token$text]]
232 traced <- token$text %in% state$traced
233 indent <- paste(rep(" ", length(state$calls)), collapse = "")
234 if (traced) emit(indent, "( ", paste(c(token$text, unlist(inputs, use.names = FALSE)), collapse = " "), " )")
235 state$calls <- c(state$calls, token$text)
236 variables <- setNames(inputs, procedure$parameters)
237 state$environments[[length(state$environments) + 1L]] <- new_environment(variables)
238 result <- tryCatch({
239 execute(procedure$body)
240 NULL
241 }, logo_return = function(signal) signal$value, finally = {
242 state$environments <- head(state$environments, -1L)
243 state$calls <- head(state$calls, -1L)
244 })
245 if (traced) emit(indent, token$text, if (is.null(result)) " stops" else paste0(" outputs ", result))
246 result
247 }

271 direction <- directions[[state$heading %/% 45 + 1L]]
272 if (number < 0) direction <- -direction
273 apply_pen(state$x, state$y)
274 for (i in seq_len(abs(number))) {
275 state$x <- state$x + direction[1]
276 state$y <- state$y + direction[2]

280 move_to <- function(x, y) {
281 dx <- x - state$x
282 dy <- y - state$y
283 steps <- max(abs(dx), abs(dy))
284 if (steps == 0) {
285 apply_pen(state$x, state$y)
286 } else {
287 for (i in 0:steps) apply_pen(state$x + floor((2 * i * dx + steps) / (2 * steps)), state$y + floor((2 * i * dy + steps) / (2 * steps)))
288 }
289 state$x <- x
290 state$y <- y
291 }
292 canvas_bounds <- function() {
293 names <- ls(envir = state$canvas, all.names = TRUE)
294 points <- do.call(rbind, lapply(names, function(name) as.numeric(strsplit(name, ",", fixed = TRUE)[[1]])))
295 c(min(points[, 1]), min(points[, 2]), max(points[, 1]), max(points[, 2]))
296 }
297 fill <- function() {
298 if (length(ls(envir = state$canvas, all.names = TRUE)) == 0 || marked(state$x, state$y)) return(invisible(NULL))
299 box <- canvas_bounds()
300 if (state$x < box[1] || state$x > box[3] || state$y < box[2] || state$y > box[4]) return(invisible(NULL))
301 queue <- list(c(state$x, state$y))
302 mark(state$x, state$y, state$character)
303 head <- 1L
304 while (head <= length(queue)) {
305 current <- queue[[head]]
306 x <- current[1]
307 y <- current[2]
308 head <- head + 1L
309 for (point in list(c(x - 1, y), c(x + 1, y), c(x, y - 1), c(x, y + 1))) {
310 a <- point[1]
311 b <- point[2]
312 if (a < box[1] || a > box[3] || b < box[2] || b > box[4] || marked(a, b)) next
313 mark(a, b, state$character)
314 queue[[length(queue) + 1L]] <- point
315 }
316 }
317 invisible(NULL)

337 execute <- function(sequence) {
338 for (command in sequence) {
339 token <- command$token
340 args <- command$args
341 name <- command$name
342 if (name == "call") {
343 result <- call(token, args)
344 if (!is.null(result)) error(token, paste0("you don't say what to do with ", result))
345 next
346 }
347 if (name == "WHILE") {
348 while (value(args[[1]]) != 0) execute(args[[2]])
349 next
350 }
351 values <- lapply(args, function(arg) {
352 if (inherits(arg, "logo_token")) substring(arg$text, 2) else if (is.null(arg$kind)) arg else value(arg)
353 })
354 a <- if (length(values) >= 1) values[[1]] else NULL
355 b <- if (length(values) >= 2) values[[2]] else NULL
356 c <- if (length(values) >= 3) values[[3]] else NULL
357 d <- if (length(values) >= 4) values[[4]] else NULL
358 if (name == "FD") move(a)
359 if (name == "BK") move(-a)
360 if (name %in% c("RT", "LT", "SETH")) {
361 if (a %% 45 != 0) error(token, paste0(token$text, " doesn't like ", a, " as input"))
362 heading <- if (name == "SETH") a else state$heading + if (name == "LT") -a else a
363 state$heading <- heading %% 360
364 }
365 if (name == "PU") state$pen <- "UP"
366 if (name == "PD") state$pen <- "DOWN"
367 if (name == "PE") state$pen <- "ERASE"
368 if (name == "SETPEN") {
369 if (a < 33 || a > 126) error(token, paste0("SETPEN doesn't like ", a, " as input"))
370 state$character <- rawToChar(as.raw(a))
371 }
372 if (name == "SETXY") move_to(a, b)
373 if (name == "SETX") move_to(a, state$y)
374 if (name == "SETY") move_to(state$x, a)
375 if (name == "HOME") {
376 move_to(0, 0)
377 state$heading <- 0
378 }
379 if (name == "CS") {
380 state$canvas <- new.env(hash = TRUE, parent = emptyenv())
381 state$x <- state$y <- state$heading <- 0
382 }
383 if (name == "CLEAN") state$canvas <- new.env(hash = TRUE, parent = emptyenv())
384 if (name == "FILL") fill()
385 if (name == "LABEL") {
386 direction <- directions[[state$heading %/% 45 + 1L]]
387 chars <- strsplit(as.character(a), "", fixed = TRUE)[[1]]
388 for (i in seq_along(chars)) mark(state$x + (i - 1L) * direction[1], state$y + (i - 1L) * direction[2], chars[i])
389 }
390 if (name == "SNAPSHOT") snapshot()
391 if (name == "REPEAT" && a > 0 && length(b) > 0L) {
392 i <- 1
393 while (i <= a) {
394 repeat_body(i, b)
395 i <- i + 1
396 }
397 }
398 if (name == "FOR") {
399 if (length(d) == 0L) {
400 make(a, c)
401 } else {
402 step <- if (b <= c) 1 else -1
403 number <- b
404 repeat {
405 make(a, number)
406 execute(d)
407 if (number == c) break
408 number <- number + step
409 }
410 }
411 }
412 if (name == "IF" && a != 0) execute(b)
413 if (name == "IFELSE") execute(if (a == 0) c else b)
414 if (name == "TEST") state$environments[[length(state$environments)]]$test <- a != 0
415 if (name %in% c("IFTRUE", "IFFALSE")) {
416 test <- state$environments[[length(state$environments)]]$test
417 if (is.null(test)) error(token, paste0(token$text, " without TEST"))
418 if (test == (name == "IFTRUE")) execute(a)
419 }
420 if (name == "STOP") logo_return(NULL)
421 if (name == "OUTPUT") logo_return(a)
422 if (name == "MAKE") make(a, b)
423 if (name == "LOCAL") state$environments[[length(state$environments)]]$variables[[a]] <- 0
424 if (name == "LOCALMAKE") state$environments[[length(state$environments)]]$variables[[a]] <- b
425 if (name == "TRACE") state$traced <- unique(c(state$traced, a))
426 if (name == "UNTRACE") state$traced <- state$traced[state$traced != a]
427 if (name == "SEED") {
428 if (a <= 0) error(token, paste0("SEED doesn't like ", a, " as input"))
429 state$random <- a
430 }
431 if (name == "PRINT") emit(a)
432 if (name == "MULTIPRINT") emit(paste(unlist(values, use.names = FALSE), collapse = " "))
433 }
434 invisible(NULL)
435 }
436 finish <- function() {
437 emit("TURTLE ", state$x, " ", state$y, " ", state$heading, " ", state$pen, " ", state$character)
438 snapshot()
439 }
440 list(execute = execute, finish = finish)

443run_program <- function(name, source) {
444 emit("PROGRAM ", name)
445 tryCatch({
446 procedures <- list()
447 top <- list()
448 current <- NULL
449 for (line in source) {
450 number <- line$number
451 tokens <- lex(line$text, number)
452 if (length(tokens) == 0) next
453 if (tokens[[1]]$kind == "word" && tokens[[1]]$text == "TO") {
454 if (!is.null(current)) logo_error(number, paste0("TO inside definition of ", current$name))
455 if (length(tokens) < 2 || tokens[[2]]$kind != "word") logo_error(number, "bad procedure header")
456 procedure_name <- tokens[[2]]$text
457 if (primitive(procedure_name)) logo_error(number, paste0(procedure_name, " is a primitive"))
458 if (!is.null(procedures[[procedure_name]])) logo_error(number, paste0(procedure_name, " is already defined"))
459 parameters <- character()
460 for (token in tail(tokens, -2L)) {
461 if (token$kind != "variable") logo_error(number, "bad procedure header")
462 parameter <- substring(token$text, 2)
463 if (parameter %in% parameters) logo_error(number, paste0("duplicate input ", token$text))
464 parameters <- c(parameters, parameter)
465 }
466 current <- new.env(parent = emptyenv())
467 current$name <- procedure_name
468 current$parameters <- parameters
469 current$line <- number
470 current$tokens <- list()
471 procedures[[procedure_name]] <- current
472 } else if (length(tokens) == 1 && tokens[[1]]$kind == "word" && tokens[[1]]$text == "END") {
473 if (is.null(current)) logo_error(number, "END without TO")
474 current <- NULL
475 } else if (is.null(current)) {
476 top <- c(top, tokens)
477 } else {
478 current$tokens <- c(current$tokens, tokens)
479 }
480 }
481 if (!is.null(current)) logo_error(current$line, paste0("missing END for ", current$name))
482 for (procedure in procedures) procedure$body <- parse(procedure$tokens, procedures, FALSE)
483 body <- parse(top, procedures, TRUE)
484 turtle <- make_turtle(procedures)
485 turtle$execute(body)
486 turtle$finish()
487 }, logo_error = function(error) {
488 emit("ERROR line ", error$line, ": ", conditionMessage(error))
489 calls <- rev(error$calls)
490 i <- 1L
491 while (i <= length(calls)) {
