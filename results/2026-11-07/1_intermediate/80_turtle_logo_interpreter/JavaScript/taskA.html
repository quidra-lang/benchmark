107function lex(source, line) {
108 const text = source.split(";")[0].toUpperCase(),
109 tokens = [];
110 for (let i = 0; i < text.length;) {
111 if (text[i] === " ") {
112 i++;
113 continue;
114 }
115 const start = i,
116 first = text[i];
117 if ("[]()+-*/=<>".includes(first)) {
118 i++;
119 if (
120 (first === "<" && ["=", ">"].includes(text[i])) ||
121 (first === ">" && text[i] === "=")
122 )
123 i++;
124 tokens.push({
125 kind: "symbol",
126 text: text.slice(start, i),
127 line,
128 sign:
129 first === "-" &&
130 (start === 0 || " [(".includes(text[start - 1])) &&
131 i < text.length &&
132 text[i] !== " ",
133 });
134 } else {
135 while (i < text.length && !" []()+-*/=<>".includes(text[i])) i++;
136 const word = text.slice(start, i);
137 let kind;
138 if (/^[0-9]+$/.test(word)) kind = "number";
139 else if (/^:[A-Z][A-Z0-9_]*$/.test(word)) kind = "variable";
140 else if (/^"[A-Z][A-Z0-9_]*$/.test(word)) kind = "quote";
141 else if (/^[A-Z][A-Z0-9_]*$/.test(word)) kind = "word";
142 else throw new LogoError(line, `bad token '${word}'`);
143 tokens.push({ kind, text: word, line, sign: false });
144 }
145 }
146 return tokens;
147}
148function extract(lines) {
149 const procedures = new Map(),
150 top = [];
151 let current = null;
152 for (const source of lines) {
153 const tokens = lex(source.text, source.line);
154 if (!tokens.length) continue;
155 const first = tokens[0];
156 if (first.kind === "word" && first.text === "TO") {
157 if (current !== null) fail(first, "TO inside definition of " + current.name);
158 if (tokens.length < 2 || tokens[1].kind !== "word") fail(first, "bad procedure header");
159 const name = tokens[1].text;
160 if (primitives.has(name)) fail(first, name + " is a primitive");
161 if (procedures.has(name)) fail(first, name + " is already defined");
162 const parameters = [];
163 for (const token of tokens.slice(2)) {
164 if (token.kind !== "variable") fail(first, "bad procedure header");
165 if (parameters.includes(token.text.slice(1)))
166 fail(first, "duplicate input " + token.text);
167 parameters.push(token.text.slice(1));
168 }
169 current = { name, parameters, token: first, tokens: [], body: [] };
170 procedures.set(name, current);
171 } else if (tokens.length === 1 && first.kind === "word" && first.text === "END") {
172 if (current === null) fail(first, "END without TO");
173 current = null;
174 } else (current === null ? top : current.tokens).push(...tokens);
175 }
176 if (current !== null) fail(current.token, "missing END for " + current.name);
177 return { procedures, top };

179class Parser {
180 constructor(tokens, procedures, inside) {
181 this.tokens = tokens;
182 this.procedures = procedures;
183 this.inside = inside;
184 this.at = 0;
185 }
186 peek() {
187 return this.tokens[this.at];
188 }
189 unknown(token) {
190 return (
191 token.kind === "word" && !primitives.has(token.text) && !this.procedures.has(token.text)
192 );
193 }
194 inputToken(owner) {
195 const token = this.peek();
196 if (token === undefined || token.text === "]" || token.text === ")")
197 fail(owner, "not enough inputs to " + owner.text);
198 if (this.unknown(token)) fail(token, "I don't know how to " + token.text);
199 return token;
200 }
201 input(kind, owner) {
202 const token = this.inputToken(owner);
203 if (kind === "L") {
204 if (token.text !== "[")
205 fail(token, `${owner.text} doesn't like ${token.text} as input`);
206 this.at++;
207 const commands = [];
208 while (this.peek() !== undefined && this.peek().text !== "]")
209 commands.push(this.command());
210 if (this.peek() === undefined) fail(token, "missing ]");
211 this.at++;
212 return { kind: "list", commands };
213 }
214 if (kind === "Q" || (kind === "P" && token.kind === "quote")) {
215 if (token.kind !== "quote")
216 fail(token, `${owner.text} doesn't like ${token.text} as input`);
217 this.at++;
218 return { kind: "quote", value: token.text.slice(1) };
219 }
220 return this.expression(owner);
221 }
222 expression(owner) {
223 return this.binary(owner, 0);
224 }
225 binary(owner, level) {
226 if (level === 3) return this.term(owner);
227 let left = this.binary(owner, level + 1);
228 const operators =
229 level === 0
230 ? ["=", "<>", "<", ">", "<=", ">="]
231 : level === 1
232 ? ["+", "-"]
233 : ["*", "/", "MOD"];
234 while (
235 this.peek() !== undefined &&
236 operators.includes(this.peek().text) &&
237 !this.peek().sign
238 ) {
239 const token = this.tokens[this.at++],
240 right = this.binary(token, level + 1);
241 left = { kind: "binary", token, left, right };
242 }
243 return left;
244 }
245 term(owner) {
246 const token = this.inputToken(owner);
247 this.at++;
248 if (token.kind === "number") return { kind: "number", value: Number(token.text), token };
249 if (token.kind === "variable")
250 return { kind: "variable", name: token.text.slice(1), token };
251 if (token.text === "-") return { kind: "unary", value: this.term(token), token };
252 if (token.text === "(") {
253 const value = this.expression(token);
254 if (this.peek() === undefined || this.peek().text !== ")") fail(token, "missing )");
255 this.at++;
256 return value;
257 }
258 if (
259 token.kind === "word" &&
260 (functionInputs.has(token.text) || this.procedures.has(token.text))
261 ) {
262 const count =
263 functionInputs.get(token.text) ?? this.procedures.get(token.text).parameters.length;
264 const inputs = Array.from({ length: count }, () => this.expression(token));
265 return { kind: functionInputs.has(token.text) ? "function" : "call", token, inputs };
266 }
267 fail(token, `${owner.text} doesn't like ${token.text} as input`);
268 }
269 command() {
270 const token = this.tokens[this.at++];
271 if (token.text === "]") fail(token, "unexpected ]");
272 const next = this.peek();
273 if (
274 token.text === "(" &&
275 next !== undefined &&
276 next.kind === "word" &&
277 ["PRINT", "PR"].includes(next.text)
278 ) {
279 this.at++;
280 const inputs = [];
281 while (this.peek() === undefined || this.peek().text !== ")") {
282 if (this.peek() === undefined || this.peek().text === "]") fail(token, "missing )");
283 inputs.push(this.input("P", next));
284 }
285 this.at++;
286 return { token: next, name: "MULTIPRINT", inputs };
287 }
288 if (token.kind === "word" && ["TO", "END"].includes(token.text))
289 fail(token, "misplaced " + token.text);
290 if (token.kind === "word" && commandInputs.has(token.text)) {
291 if (!this.inside && ["STOP", "OUTPUT", "OP", "LOCAL", "LOCALMAKE"].includes(token.text))
292 fail(token, "can only use " + token.text + " inside a procedure");
293 const inputs = [...commandInputs.get(token.text)].map((kind) =>
294 this.input(kind, token),
295 );
296 if (["TRACE", "UNTRACE"].includes(token.text) && !this.procedures.has(inputs[0].value))
297 fail(this.tokens[this.at - 1], "I don't know how to " + inputs[0].value);
298 return { token, name: aliases.get(token.text) ?? token.text, inputs };
299 }
300 if (token.kind === "word" && this.procedures.has(token.text)) {
301 return {
302 token,
303 name: "CALL",
304 inputs: Array.from(
305 { length: this.procedures.get(token.text).parameters.length },
306 () => this.expression(token),
307 ),
308 };
309 }
310 if (this.unknown(token)) fail(token, "I don't know how to " + token.text);
311 fail(token, "you don't say what to do with " + token.text);
312 }
313 sequence() {
314 const commands = [];
315 while (this.peek() !== undefined) commands.push(this.command());
316 return commands;
317 }

319class Turtle {
320 constructor(procedures, output) {
321 this.procedures = procedures;
322 this.output = output;
323 this.x = 0;
324 this.y = 0;
325 this.heading = 0;
326 this.pen = "DOWN";
327 this.character = "*";
328 this.canvas = new Map();
329 this.environments = [{ variables: new Map(), test: null }];
330 this.active = [];
331 this.repetitions = [];
332 this.traced = new Set();
333 this.random = 1;
334 }
335 current() {
336 return this.environments.at(-1);
337 }
338 make(name, value) {
339 for (let i = this.environments.length - 1; i > 0; i--)
340 if (this.environments[i].variables.has(name)) {
341 this.environments[i].variables.set(name, value);
342 return;
343 }
344 this.environments[0].variables.set(name, value);
345 }
346 key(x, y) {
347 return x + "," + y;
348 }
349 apply(x, y) {
350 const key = this.key(x, y);
351 if (this.pen === "DOWN") this.canvas.set(key, this.character);
352 else if (this.pen === "ERASE") this.canvas.delete(key);
353 }
354 move(x, y) {
355 const x0 = this.x,
356 y0 = this.y,
357 dx = x - x0,
358 dy = y - y0,
359 steps = Math.max(Math.abs(dx), Math.abs(dy));
360 if (!steps) this.apply(x0, y0);
361 else
362 for (let i = 0; i <= steps; i++)
363 this.apply(
364 x0 + Math.floor((2 * i * dx + steps) / (2 * steps)),
365 y0 + Math.floor((2 * i * dy + steps) / (2 * steps)),
366 );
367 this.x = x;
368 this.y = y;
369 }
370 extent() {
371 let x1 = Infinity,
372 y1 = Infinity,
373 x2 = -Infinity,
374 y2 = -Infinity;
375 for (const key of this.canvas.keys()) {
376 const [x, y] = key.split(",").map(Number);
377 x1 = Math.min(x1, x);
378 y1 = Math.min(y1, y);
379 x2 = Math.max(x2, x);
380 y2 = Math.max(y2, y);
381 }
382 return { x1, y1, x2, y2 };
383 }
384 picture() {
385 if (!this.canvas.size) {
386 this.output.push("CANVAS 0 0 0");
387 return;
388 }
389 const { x1, y1, x2, y2 } = this.extent();
390 this.output.push(`CANVAS ${x2 - x1 + 1} ${y2 - y1 + 1} ${this.canvas.size}`);
391 for (let y = y2; y >= y1; y--) {
392 let row = "";
393 for (let x = x1; x <= x2; x++) row += this.canvas.get(this.key(x, y)) ?? " ";
394 this.output.push(row.replace(/ +$/, ""));
395 }
396 }
397 fill() {
398 if (!this.canvas.size || this.canvas.has(this.key(this.x, this.y))) return;
399 const { x1, y1, x2, y2 } = this.extent();
400 if (this.x < x1 || this.x > x2 || this.y < y1 || this.y > y2) return;
401 const queue = [[this.x, this.y]];
402 this.canvas.set(this.key(this.x, this.y), this.character);
403 for (let i = 0; i < queue.length; i++) {
404 const [x, y] = queue[i];
405 for (const [a, b] of [
406 [x - 1, y],
407 [x + 1, y],
408 [x, y - 1],
409 [x, y + 1],
410 ])
411 if (a >= x1 && a <= x2 && b >= y1 && b <= y2 && !this.canvas.has(this.key(a, b))) {
412 this.canvas.set(this.key(a, b), this.character);
413 queue.push([a, b]);
414 }
415 }
416 }
417 call(token, inputs) {
418 const values = inputs.map((input) => this.evaluate(input));
419 if (this.active.length >= 100) fail(token, "recursion too deep");
420 const traced = this.traced.has(token.text),
421 indent = " ".repeat(2 * this.active.length),
422 procedure = this.procedures.get(token.text);
423 if (traced)
424 this.output.push(
425 indent + "( " + token.text + (values.length ? " " + values.join(" ") : "") + " )",
426 );
427 this.active.push(token.text);
428 this.environments.push({
429 variables: new Map(procedure.parameters.map((name, i) => [name, values[i]])),
430 test: null,
431 });
432 const result = this.sequence(procedure.body);
433 this.environments.pop();
434 this.active.pop();
435 const value = result === null ? null : result.value;
436 if (traced)
437 this.output.push(
438 indent + token.text + (value === null ? " stops" : " outputs " + value),
439 );
440 return value;
441 }
442 evaluate(expression) {
443 const token = expression.token;
444 if (expression.kind === "number") return expression.value;
445 if (expression.kind === "variable") {
446 for (let i = this.environments.length - 1; i >= 0; i--)
447 if (this.environments[i].variables.has(expression.name))
448 return this.environments[i].variables.get(expression.name);
449 fail(token, expression.name + " has no value");
450 }
451 if (expression.kind === "unary") return -this.evaluate(expression.value);
452 if (expression.kind === "binary") {
453 const a = this.evaluate(expression.left),
454 b = this.evaluate(expression.right);
455 if ((token.text === "/" || token.text === "MOD") && b === 0)
456 fail(token, token.text + " doesn't like 0 as input");
457 switch (token.text) {
458 case "+":
459 return a + b;
460 case "-":
461 return a - b;
462 case "*":
463 return a * b;
464 case "/":
465 return Math.trunc(a / b);
466 case "MOD":
467 return a % b;
468 case "=":
469 return Number(a === b);
470 case "<>":
471 return Number(a !== b);
472 case "<":
473 return Number(a < b);
474 case ">":
475 return Number(a > b);
476 case "<=":
477 return Number(a <= b);
478 default:
479 return Number(a >= b);
480 }
481 }
482 if (expression.kind === "call") {
483 const value = this.call(token, expression.inputs);
484 if (value === null) fail(token, token.text + " did not output");
485 return value;
486 }
487 const values = expression.inputs.map((input) => this.evaluate(input)),
488 a = values[0],
489 b = values[1];
490 switch (token.text) {
491 case "XCOR":
492 return this.x;
493 case "YCOR":
494 return this.y;
495 case "HEADING":
496 return this.heading;
497 case "REPCOUNT":
498 if (!this.repetitions.length) fail(token, "REPCOUNT outside REPEAT");
499 return this.repetitions.at(-1);
500 case "RANDOM":
501 if (a < 1) fail(token, `RANDOM doesn't like ${a} as input`);
502 this.random = (48271 * this.random) % 2147483647;
503 return this.random % a;
504 case "ABS":
505 return Math.abs(a);
506 case "NOT":
507 return Number(a === 0);
508 case "AND":
509 return Number(a !== 0 && b !== 0);
510 case "OR":
511 return Number(a !== 0 || b !== 0);
512 case "MAX":
513 return Math.max(a, b);
514 default:
515 return Math.min(a, b);
516 }
517 }
518 item(input) {
519 return input.kind === "quote" ? input.value : String(this.evaluate(input));
520 }
521 sequence(commands) {
522 for (const command of commands) {
523 const result = this.command(command);
524 if (result !== null) return result;
525 }
526 return null;
527 }
528 command(command) {
529 const { name, token, inputs } = command;
530 if (name === "CALL") {
531 const value = this.call(token, inputs);
532 if (value !== null) fail(token, "you don't say what to do with " + value);
533 return null;
534 }
535 if (name === "PRINT" || name === "MULTIPRINT") {
536 const values = inputs.map((input) => this.item(input));
537 this.output.push(values.join(" "));
538 return null;
539 }
540 if (name === "LABEL") {
541 const text = this.item(inputs[0]),
542 [dx, dy] = directions[this.heading / 45];
543 for (let i = 0; i < text.length; i++)
544 this.canvas.set(this.key(this.x + i * dx, this.y + i * dy), text[i]);
545 return null;
546 }
547 if (name === "WHILE") {
548 while (this.evaluate(inputs[0]) !== 0) {
549 const result = this.sequence(inputs[1].commands);
550 if (result !== null) return result;
551 }
552 return null;
553 }
554 const values = inputs.map((input) =>
555 input.kind === "quote"
556 ? input.value
557 : input.kind === "list"
558 ? input.commands
559 : this.evaluate(input),
560 );
561 const a = values[0],
562 b = values[1];
563 switch (name) {
564 case "FD":
565 case "BK": {
566 const amount = name === "BK" ? -a : a,
567 [dx, dy] = directions[this.heading / 45],
568 sign = amount < 0 ? -1 : 1;
569 this.apply(this.x, this.y);
570 for (let i = 0; i < Math.abs(amount); i++) {
571 this.x += sign * dx;
572 this.y += sign * dy;
573 this.apply(this.x, this.y);
574 }
575 break;
576 }
577 case "RT":
578 case "LT":
579 case "SETH":
580 if (a % 45 !== 0) fail(token, `${token.text} doesn't like ${a} as input`);
581 this.heading =
582 (((name === "SETH" ? a : this.heading + (name === "LT" ? -a : a)) % 360) +
583 360) %
584 360;
585 break;
586 case "PU":
587 this.pen = "UP";
588 break;
589 case "PD":
590 this.pen = "DOWN";
591 break;
592 case "PE":
593 this.pen = "ERASE";
594 break;
595 case "SETPEN":
596 if (a < 33 || a > 126) fail(token, `SETPEN doesn't like ${a} as input`);
597 this.character = String.fromCharCode(a);
598 break;
599 case "SETXY":
600 this.move(a, b);
601 break;
602 case "SETX":
603 this.move(a, this.y);
604 break;
605 case "SETY":
606 this.move(this.x, a);
607 break;
608 case "HOME":
609 this.move(0, 0);
610 this.heading = 0;
611 break;
612 case "CS":
613 this.canvas.clear();
614 this.x = 0;
615 this.y = 0;
616 this.heading = 0;
617 break;
618 case "CLEAN":
619 this.canvas.clear();
620 break;
621 case "FILL":
622 this.fill();
623 break;
624 case "SNAPSHOT":
625 this.picture();
626 break;
627 case "MAKE":
628 this.make(a, b);
629 break;
630 case "LOCAL":
631 this.current().variables.set(a, 0);
632 break;
633 case "LOCALMAKE":
634 this.current().variables.set(a, b);
635 break;
636 case "TRACE":
637 this.traced.add(a);
638 break;
639 case "UNTRACE":
640 this.traced.delete(a);
641 break;
642 case "SEED":
643 if (a < 1) fail(token, `SEED doesn't like ${a} as input`);
644 this.random = a;
645 break;
646 case "STOP":
647 return { value: null };
648 case "OUTPUT":
649 return { value: a };
650 case "REPEAT":
651 if (b.length === 0) break;
652 for (let k = 1; k <= a; k++) {
653 this.repetitions.push(k);
654 let result;
655 try {
656 result = this.sequence(b);
657 } finally {
658 this.repetitions.pop();
659 }
660 if (result !== null) return result;
661 }
662 break;
663 case "FOR": {
664 const finish = values[2],
665 step = b <= finish ? 1 : -1;
666 if (values[3].length === 0) {
667 this.make(a, finish);
668 break;
669 }
670 for (let value = b; ; value += step) {
671 this.make(a, value);
672 const result = this.sequence(values[3]);
673 if (result !== null) return result;
674 if (value === finish) break;
675 }
676 break;
677 }
678 case "IF":
679 if (a !== 0) return this.sequence(b);
680 break;
681 case "IFELSE":
682 return this.sequence(a !== 0 ? b : values[2]);
683 case "TEST":
684 this.current().test = a !== 0;
685 break;
686 case "IFTRUE":
687 case "IFFALSE":
688 if (this.current().test === null) fail(token, token.text + " without TEST");
689 if (this.current().test === (name === "IFTRUE")) return this.sequence(a);
690 break;
691 }
692 return null;
693 }

695function run(name, lines, output) {
696 output.push("PROGRAM " + name);
697 let turtle = null;
698 try {
699 const { procedures, top } = extract(lines);
700 for (const procedure of procedures.values())
701 procedure.body = new Parser(procedure.tokens, procedures, true).sequence();
702 const commands = new Parser(top, procedures, false).sequence();
703 turtle = new Turtle(procedures, output);
704 turtle.sequence(commands);
705 output.push(
706 `TURTLE ${turtle.x} ${turtle.y} ${turtle.heading} ${turtle.pen} ${turtle.character}`,
707 );
708 turtle.picture();
709 } catch (error) {
710 if (!(error instanceof LogoError)) throw error;
711 output.push(`ERROR line ${error.line}: ${error.message}`);
712 if (turtle !== null)
713 for (let i = turtle.active.length - 1; i >= 0;) {
714 const name = turtle.active[i];
715 let count = 0;
716 while (i >= 0 && turtle.active[i] === name) {
717 i--;
718 count++;
719 }
720 output.push(" in " + name + (count > 1 ? " x" + count : ""));
721 }
722 }
