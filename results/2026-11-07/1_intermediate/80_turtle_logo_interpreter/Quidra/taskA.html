123Token[] | error tokenize(string source, nat line)
124 text := upper_ascii(source.split(";")[0])
125 delimiters := " []()+-*/=<>"
126 Token[] result := []
127 nat i := 0
128 while i < #text
129 character := text[i]
130 following := if i + 1 < #text then text[i + 1] else ""
131 if character == " "
132 i += 1
133 elif character == "-"
134 starts_term := i == 0 or " [(".contains(text[i - 1])
135 sign := starts_term and following != "" and following != " "
136 result = result.append(Token(if sign then 5 else 4, "-", line))
137 i += 1
138 elif character == "<" or character == ">"
139 pair := character + following
140 spelling := if has_text(["<=", "<>", ">="], pair) then pair else character
141 result = result.append(Token(4, spelling, line))
142 i += #spelling
143 elif delimiters.contains(character)
144 result = result.append(Token(4, character, line))
145 i += 1
146 else
147 start := i
148 while i < #text and not delimiters.contains(text[i])
149 i += 1
150 result = result.append(try classify(text[start:<i], line))
151 return result
