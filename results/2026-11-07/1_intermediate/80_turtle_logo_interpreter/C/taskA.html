370static Tokens lex(const char *source, int line) {
371 const char *comment = strchr(source, ';');
372 char *text = part(source, comment ? (size_t)(comment - source) : strlen(source));
373 for (char *p = text; *p; p++) if (*p >= 'a' && *p <= 'z') *p = (char)(*p - 'a' + 'A');
374 Tokens tokens = {0};
375 for (int i = 0; text[i];) {
376 char c = text[i];
377 if (c == ' ') {
378 i++;
379 continue;
380 }
381 if (strchr("[]()+*/=", c)) {
382 add_token(&tokens, (Token){SYMBOL, part(text + i, 1), line});
383 i++;
384 } else if (c == '<' || c == '>') {
385 int length = text[i + 1] && (text[i + 1] == '=' || (c == '<' && text[i + 1] == '>')) ? 2 : 1;
386 add_token(&tokens, (Token){SYMBOL, part(text + i, (size_t)length), line});
387 i += length;
388 } else if (c == '-') {
389 bool sign = (i == 0 || strchr(" [(", text[i - 1])) && text[i + 1] && text[i + 1] != ' ';
390 add_token(&tokens, (Token){sign ? SIGN_MINUS : OPERATOR_MINUS, "-", line});
391 i++;
392 } else {
393 int start = i;
394 while (text[i] && !strchr(" []()+-*/=<>", text[i])) i++;
395 char *word = part(text + start, (size_t)(i - start));
396 TokenKind kind;
397 bool numeric = true;
398 for (char *p = word; *p; p++) if (*p < '0' || *p > '9') numeric = false;
399 if (numeric) kind = NUMBER;
400 else if (*word == ':' && name(word + 1)) kind = VARIABLE;
401 else if (*word == '"' && name(word + 1)) kind = QUOTED;
402 else if (name(word)) kind = WORD;
403 else {
404 error(line, "bad token '%s'", word);
405 return tokens;
406 }
407 add_token(&tokens, (Token){kind, word, line});
408 }
409 }
410 return tokens;
411}

468static Command *block(Parser *parser, Token owner) {
469 Token opener = input(parser, owner);
470 if (strcmp(opener.text, "[")) reject(owner, opener);
471 parser->position++;
472 Command *first = NULL, *last = NULL;
473 while (peek(parser) && !is(parser, "]")) append_command(&first, &last, parse_command(parser));
474 if (!peek(parser)) error(opener.line, "missing ]");
475 parser->position++;
476 return first;
477}

502static Command *parse_command(Parser *parser) {
503 Token token = parser->tokens->items[parser->position++];
504 if (!strcmp(token.text, "(") && peek(parser) && peek(parser)->kind == WORD && (is(parser, "PRINT") || is(parser, "PR"))) {
505 Token print = parser->tokens->items[parser->position++];
506 Command *command = new_command(print, MULTIPRINT);
507 while (!is(parser, ")")) {
508 if (!peek(parser) || is(parser, "]")) error(token.line, "missing )");
509 add_print(command, print_item(parser, print));
510 }
511 parser->position++;
512 return command;
513 }
514 if (token.kind == WORD && (!strcmp(token.text, "TO") || !strcmp(token.text, "END"))) error(token.line, "misplaced %s", token.text);
515 const Primitive *p = token.kind == WORD ? primitive(token.text) : NULL;
516 if (p && !p->function && p->code != MOD) {
517 if (!parser->inside && (p->code == STOP || p->code == OUTPUT || p->code == LOCAL || p->code == LOCALMAKE)) error(token.line, "can only use %s inside a procedure", token.text);
518 Command *command = new_command(token, p->code);
519 for (const char *kind = p->inputs; *kind; kind++) switch (*kind) {
520 case 'E':
521 add_argument(&command->expressions, expression(parser, token));
522 break;
523 case 'L':
524 command->blocks[command->block_count++] = block(parser, token);
525 break;
526 case 'Q':
527 command->quoted = quoted(parser, token);
528 break;
529 case 'P':
530 add_print(command, print_item(parser, token));
531 break;
532 }
533 if ((p->code == TRACE || p->code == UNTRACE) && !procedure(command->quoted)) error(parser->tokens->items[parser->position - 1].line, "I don't know how to %s", command->quoted);
534 return command;
535 }
536 Procedure *proc = token.kind == WORD ? procedure(token.text) : NULL;
537 if (proc) {
538 Command *command = new_command(token, CALL);
539 command->procedure = proc;
540 for (int i = 0; i < proc->parameters.count; i++) add_argument(&command->expressions, expression(parser, token));
541 return command;
542 }
543 if (unknown(token)) error(token.line, "I don't know how to %s", token.text);
544 error(token.line, "you don't say what to do with %s", token.text);
545 return NULL;
546}

548static Expression *term(Parser *parser, Token owner) {
549 Token token = input(parser, owner);
550 parser->position++;
551 if (token.kind == NUMBER || token.kind == VARIABLE) return new_expression(token, token.kind == NUMBER ? E_NUMBER : E_VARIABLE);
552 if (!strcmp(token.text, "-")) {
553 Expression *result = new_expression(token, E_NEGATE);
554 add_argument(&result->arguments, term(parser, token));
555 return result;
556 }
557 if (!strcmp(token.text, "(")) {
558 Expression *result = expression(parser, token);
559 if (!is(parser, ")")) error(token.line, "missing )");
560 parser->position++;
561 return result;
562 }
563 if (token.kind == WORD) {
564 const Primitive *p = primitive(token.text);
565 Procedure *proc = procedure(token.text);
566 if ((p && p->function) || proc) {
567 Expression *result = new_expression(token, p ? E_FUNCTION : E_CALL);
568 result->procedure = proc;
569 if (p) result->code = p->code;
570 int count = p ? (int)strlen(p->inputs) : proc->parameters.count;
571 for (int i = 0; i < count; i++) add_argument(&result->arguments, expression(parser, token));
572 return result;
573 }
574 }
575 reject(owner, token);
576 return NULL;
577}

586static Expression *product(Parser *parser, Token owner) {
587 Expression *result = term(parser, owner);
588 while (is(parser, "*") || is(parser, "/") || is(parser, "MOD")) {
589 Token operation = parser->tokens->items[parser->position++];
590 result = binary(result, operation, term(parser, operation));
591 }
592 return result;
593}

595static Expression *sum(Parser *parser, Token owner) {
596 Expression *result = product(parser, owner);
597 while (is(parser, "+") || (is(parser, "-") && peek(parser)->kind == OPERATOR_MINUS)) {
598 Token operation = parser->tokens->items[parser->position++];
599 result = binary(result, operation, product(parser, operation));
600 }
601 return result;

604static Expression *expression(Parser *parser, Token owner) {
605 Expression *result = sum(parser, owner);
606 while (is(parser, "=") || is(parser, "<>") || is(parser, "<") || is(parser, ">") || is(parser, "<=") || is(parser, ">=")) {
607 Token operation = parser->tokens->items[parser->position++];
608 result = binary(result, operation, sum(parser, operation));
609 }
610 return result;
611}

618static void set_variable(Environment *environment, char *name, long long value) {
619 int index = variable_index(environment, name);

621 environment->variables[index].value = value;
622 return;

627 if (!next) exit(EXIT_FAILURE);
628 environment->variables = next;

652static Value call(Token token, Procedure *proc, const long long *values) {
653 int depth = environment_count - 1;
654 if (depth == 100) error(token.line, "recursion too deep");
655 bool trace = proc->traced;
656 if (trace) {
657 for (int i = 0; i < 2 * depth; i++) putchar(' ');
658 printf("( %s", token.text);
659 for (int i = 0; i < proc->parameters.count; i++) printf(" %lld", values[i]);
660 puts(" )");
661 }
662 Environment *environment = &environments[environment_count++];
663 *environment = (Environment){0};
664 environment->procedure = proc;
665 environment->test = -1;
666 for (int i = 0; i < proc->parameters.count; i++) set_variable(environment, proc->parameters.items[i], values[i]);
667 Flow flow = execute(proc->commands);
668 free(environment->variables);
669 *environment = (Environment){0};
670 environment_count--;
671 Value result = {flow.kind == 2, flow.value};
672 if (trace) {
673 for (int i = 0; i < 2 * depth; i++) putchar(' ');
674 printf("%s", token.text);
675 if (result.present) printf(" outputs %lld\n", result.value);
676 else puts(" stops");
677 }
678 return result;
679}

681static long long evaluate(Expression *e) {
682 Token token = e->token;
683 if (e->kind == E_NUMBER) return atoll(token.text);
684 if (e->kind == E_VARIABLE) return variable(token);
685 if (e->kind == E_NEGATE) return -evaluate(e->arguments.items[0]);
686 if (e->kind == E_BINARY) {
687 long long a = evaluate(e->arguments.items[0]);
688 long long b = evaluate(e->arguments.items[1]);
689 const char *op = token.text;
690 if ((!strcmp(op, "/") || !strcmp(op, "MOD")) && !b) error(token.line, "%s doesn't like 0 as input", op);
691 if (!strcmp(op, "+")) return a + b;
692 if (!strcmp(op, "-")) return a - b;
693 if (!strcmp(op, "*")) return a * b;
694 if (!strcmp(op, "/")) return a / b;
695 if (!strcmp(op, "MOD")) return a % b;
696 if (!strcmp(op, "=")) return a == b;
697 if (!strcmp(op, "<>")) return a != b;
698 if (!strcmp(op, "<")) return a < b;
699 if (!strcmp(op, ">")) return a > b;
700 if (!strcmp(op, "<=")) return a <= b;
701 return a >= b;
702 }
703 RuntimeAllocation *allocation = e->arguments.count ? runtime_allocate((size_t)e->arguments.count * sizeof(long long)) : NULL;
704 long long *values = allocation ? allocation->pointer : NULL;
705 for (int i = 0; i < e->arguments.count; i++) values[i] = evaluate(e->arguments.items[i]);
706 long long a = e->arguments.count ? values[0] : 0, b = e->arguments.count > 1 ? values[1] : 0, result = 0;
707 if (e->kind == E_CALL) {
708 Value value = call(token, e->procedure, values);
709 if (!value.present) error(token.line, "%s did not output", token.text);
710 result = value.value;
711 } else switch (e->code) {
712 case XCOR:
713 result = turtle_x;
714 break;
715 case YCOR:
716 result = turtle_y;
717 break;
718 case HEADING:
719 result = heading;
720 break;
721 case REPCOUNT:
722 if (!repetition_count) error(token.line, "REPCOUNT outside REPEAT");
723 result = repetitions[repetition_count - 1];
724 break;
725 case RANDOM:
726 if (a < 1) error(token.line, "RANDOM doesn't like %lld as input", a);
727 random_state = random_state * 48271 % 2147483647;
728 result = random_state % a;
729 break;
730 case ABS:
731 result = a < 0 ? -a : a;
732 break;
733 case NOT:
734 result = a == 0;
735 break;
736 case AND:
737 result = a != 0 && b != 0;
738 break;
739 case OR:
740 result = a != 0 || b != 0;
741 break;
742 case MAXIMUM:
743 result = a > b ? a : b;
744 break;
745 case MINIMUM:
746 result = a < b ? a : b;
747 break;
748 }
749 runtime_release(allocation);
750 return result;
751}

753static void set_cell(int x, int y, unsigned char mark) {
754 int xx = x + COORDINATE_OFFSET, yy = y + COORDINATE_OFFSET;
755 unsigned char old = canvas[yy][xx];
756 if (!old && mark) {
757 if (marked_count == marked_capacity) {
758 marked_capacity = marked_capacity ? marked_capacity * 2 : 1024;
759 int *next = realloc(marked_positions, (size_t)marked_capacity * sizeof *next);
760 if (!next) exit(EXIT_FAILURE);
761 marked_positions = next;
762 }
763 marked_positions[marked_count++] = yy * CANVAS_SIDE + xx;
764 row_counts[yy]++;
765 column_counts[xx]++;
766 if (!mark_count) {
767 min_x = max_x = x;
768 min_y = max_y = y;
769 } else {
770 if (x < min_x) min_x = x;
771 if (x > max_x) max_x = x;
772 if (y < min_y) min_y = y;
773 if (y > max_y) max_y = y;
774 }
775 mark_count++;
776 }
777 canvas[yy][xx] = mark;
778 if (old && !mark) {
779 row_counts[yy]--;
780 column_counts[xx]--;
781 mark_count--;
782 if (mark_count) {
783 while (!column_counts[min_x + COORDINATE_OFFSET]) min_x++;
784 while (!column_counts[max_x + COORDINATE_OFFSET]) max_x--;
785 while (!row_counts[min_y + COORDINATE_OFFSET]) min_y++;
786 while (!row_counts[max_y + COORDINATE_OFFSET]) max_y--;
787 }
788 }
789}

808static void forward(long long n) {
809 int dx = direction_x[heading / 45], dy = direction_y[heading / 45];
810 if (n < 0) {
811 dx = -dx;
812 dy = -dy;
813 n = -n;
814 }
815 apply(turtle_x, turtle_y);
816 for (long long i = 0; i < n; i++) {
817 turtle_x += dx;
818 turtle_y += dy;
819 apply(turtle_x, turtle_y);
820 }
821}

828static void move(int x, int y) {
829 int old_x = turtle_x, old_y = turtle_y, dx = x - old_x, dy = y - old_y, steps = abs(dx) > abs(dy) ? abs(dx) : abs(dy);
830 if (!steps) apply(old_x, old_y);
831 else for (int i = 0; i <= steps; i++) apply(old_x + (int)floor_divide(2LL * i * dx + steps, 2LL * steps), old_y + (int)floor_divide(2LL * i * dy + steps, 2LL * steps));
832 turtle_x = x;
833 turtle_y = y;
834}

836static void fill(void) {
837 if (!mark_count || canvas[turtle_y + COORDINATE_OFFSET][turtle_x + COORDINATE_OFFSET]) return;
838 int x1 = min_x, x2 = max_x, y1 = min_y, y2 = max_y;
839 if (turtle_x < x1 || turtle_x > x2 || turtle_y < y1 || turtle_y > y2) return;
840 size_t capacity = (size_t)(x2 - x1 + 1) * (y2 - y1 + 1);
841 Point *queue = malloc(capacity * sizeof *queue);
842 if (!queue) exit(EXIT_FAILURE);
843 size_t head = 0, tail = 0;
844 queue[tail++] = (Point){turtle_x, turtle_y};
845 set_cell(turtle_x, turtle_y, pen_character);
846 const int dx[] = {-1, 1, 0, 0}, dy[] = {0, 0, -1, 1};
847 while (head < tail) {
848 Point point = queue[head++];
849 for (int d = 0; d < 4; d++) {
850 int x = point.x + dx[d], y = point.y + dy[d];
851 if (x >= x1 && x <= x2 && y >= y1 && y <= y2 && !canvas[y + COORDINATE_OFFSET][x + COORDINATE_OFFSET]) {
852 set_cell(x, y, pen_character);
853 queue[tail++] = (Point){x, y};
854 }
855 }
856 }
857 free(queue);
858}

884static void append(Buffer *buffer, const char *text) {
885 size_t length = strlen(text);
886 if (buffer->length + length + 1 > buffer->capacity) {
887 size_t capacity = buffer->capacity ? buffer->capacity * 2 : 64;
888 while (capacity < buffer->length + length + 1) capacity *= 2;
889 if (!buffer->allocation) buffer->allocation = runtime_allocate(capacity);
890 else {
891 void *next = realloc(buffer->allocation->pointer, capacity);
892 if (!next) exit(EXIT_FAILURE);
893 buffer->allocation->pointer = next;
894 }
895 buffer->text = buffer->allocation->pointer;
896 buffer->capacity = capacity;
897 }
898 memcpy(buffer->text + buffer->length, text, length + 1);
899 buffer->length += length;
900}

912static Flow execute_one(Command *command) {
913 Token token = command->token;
914 Flow result = {0};
915 if (command->code == WHILE) {
916 while (evaluate(command->expressions.items[0])) {
917 result = execute(command->blocks[0]);
918 if (result.kind) return result;
919 }
920 return result;
921 }
922 RuntimeAllocation *allocation = command->expressions.count ? runtime_allocate((size_t)command->expressions.count * sizeof(long long)) : NULL;
923 long long *values = allocation ? allocation->pointer : NULL;
924 for (int i = 0; i < command->expressions.count; i++) values[i] = evaluate(command->expressions.items[i]);
925 long long a = command->expressions.count ? values[0] : 0, b = command->expressions.count > 1 ? values[1] : 0;
926 Environment *current = &environments[environment_count - 1];
927 switch (command->code) {
928 case FD:
929 forward(a);
930 break;
931 case BK:
932 forward(-a);
933 break;
934 case RT:
935 case LT:
936 case SETH:
937 {
938 if (a % 45) error(token.line, "%s doesn't like %lld as input", token.text, a);
939 long long angle = command->code == SETH ? a : command->code == RT ? heading + a : heading - a;
940 heading = (int)((angle % 360 + 360) % 360);
941 break;
942 }
943 case PU:
944 pen = 1;
945 break;
946 case PD:
947 pen = 0;
948 break;
949 case PE:
950 pen = 2;
951 break;
952 case SETPEN:
953 if (a < 33 || a > 126) error(token.line, "SETPEN doesn't like %lld as input", a);
954 pen_character = (unsigned char)a;
955 break;
956 case SETXY:
957 move((int)a, (int)b);
958 break;
959 case SETX:
960 move((int)a, turtle_y);
961 break;
962 case SETY:
963 move(turtle_x, (int)a);
964 break;
965 case HOME:
966 move(0, 0);
967 heading = 0;
968 break;
969 case CS:
970 clear_canvas();
971 turtle_x = turtle_y = heading = 0;
972 break;
973 case CLEAN:
974 clear_canvas();
975 break;
976 case FILL:
977 fill();
978 break;
979 case LABEL:
980 {
981 char number[32];
982 const char *text = item(command->prints, number);
983 int dx = direction_x[heading / 45], dy = direction_y[heading / 45];
984 for (int i = 0; text[i]; i++) set_cell(turtle_x + i * dx, turtle_y + i * dy, (unsigned char)text[i]);
985 break;
986 }
987 case SNAPSHOT:
988 picture();
989 break;
990 case REPEAT:
991 if (command->blocks[0]) for (long long i = 1; i <= a; i++) {
992 push_repetition(i);
993 result = execute(command->blocks[0]);
994 repetition_count--;
995 if (result.kind) break;
996 }
997 break;
998 case FOR:
999 if (!command->blocks[0]) make(command->quoted, b);
1000 else for (long long i = a;; i += a <= b ? 1 : -1) {
1001 make(command->quoted, i);
1002 result = execute(command->blocks[0]);
1003 if (result.kind || i == b) break;
1004 }
1005 break;
1006 case IF:
1007 if (a) result = execute(command->blocks[0]);
1008 break;
1009 case IFELSE:
1010 result = execute(command->blocks[a ? 0 : 1]);
1011 break;
1012 case TEST:
1013 current->test = a != 0;
1014 break;
1015 case IFTRUE:
1016 case IFFALSE:
1017 if (current->test < 0) error(token.line, "%s without TEST", token.text);
1018 if (current->test == (command->code == IFTRUE)) result = execute(command->blocks[0]);
1019 break;
1020 case STOP:
1021 result.kind = 1;
1022 break;
1023 case OUTPUT:
1024 result = (Flow){2, a};
1025 break;
1026 case MAKE:
1027 make(command->quoted, a);
1028 break;
1029 case LOCAL:
1030 set_variable(current, command->quoted, 0);
1031 break;
1032 case LOCALMAKE:
1033 set_variable(current, command->quoted, a);
1034 break;
1035 case TRACE:
1036 procedure(command->quoted)->traced = true;
1037 break;
1038 case UNTRACE:
1039 procedure(command->quoted)->traced = false;
1040 break;
1041 case SEED:
1042 if (a < 1) error(token.line, "SEED doesn't like %lld as input", a);
1043 random_state = a;
1044 break;
1045 case PRINT:
1046 case MULTIPRINT:
1047 {
1048 Buffer buffer = {0};
1049 bool first = true;
1050 for (PrintItem *print = command->prints; print; print = print->next) {
1051 if (!first) append(&buffer, " ");
1052 char number[32];
1053 const char *text = item(print, number);
1054 append(&buffer, text);
1055 first = false;
1056 }
1057 puts(buffer.text ? buffer.text : "");
1058 runtime_release(buffer.allocation);
1059 break;
1060 }
1061 case CALL:
1062 {
1063 Value value = call(token, command->procedure, values);
1064 if (value.present) error(token.line, "you don't say what to do with %lld", value.value);
1065 break;
1066 }
1067 }
1068 runtime_release(allocation);
1069 return result;
1070}

1072static Flow execute(Command *commands) {
1073 for (Command *command = commands; command; command = command->next) {
1074 Flow result = execute_one(command);
1075 if (result.kind) return result;
1076 }
1077 return (Flow){0};
1078}

1080static void run_program(const char *program_name, char **source, int count, int first_line) {
1081 printf("PROGRAM %s\n", program_name);
1082 procedures = last_procedure = NULL;
1083 running = false;
1084 environment_count = 0;
1085 if (setjmp(program_error) == 0) {
1086 Tokens top = {0};
1087 Procedure *current = NULL;
1088 for (int i = 0; i < count; i++) {
1089 int line = first_line + i;
1090 Tokens tokens = lex(source[i], line);
1091 if (!tokens.count) continue;
1092 Token first = tokens.items[0];
1093 if (first.kind == WORD && !strcmp(first.text, "TO")) {
1094 if (current) error(line, "TO inside definition of %s", current->token.text);
1095 if (tokens.count < 2 || tokens.items[1].kind != WORD) error(line, "bad procedure header");
1096 Token proc_name = tokens.items[1];
1097 if (primitive(proc_name.text)) error(line, "%s is a primitive", proc_name.text);
1098 if (procedure(proc_name.text)) error(line, "%s is already defined", proc_name.text);
1099 Procedure *proc = allocate(sizeof *proc);
1100 proc->token = proc_name;
1101 for (int j = 2; j < tokens.count; j++) {
1102 Token parameter = tokens.items[j];
1103 if (parameter.kind != VARIABLE) error(line, "bad procedure header");
1104 for (int k = 0; k < proc->parameters.count; k++) if (!strcmp(proc->parameters.items[k], parameter.text + 1)) error(line, "duplicate input %s", parameter.text);
1105 add_string(&proc->parameters, parameter.text + 1);
1106 }
1107 if (last_procedure) last_procedure->next = proc;
1108 else procedures = proc;
1109 last_procedure = proc;
1110 current = proc;
1111 } else if (tokens.count == 1 && first.kind == WORD && !strcmp(first.text, "END")) {
1112 if (!current) error(line, "END without TO");
1113 current = NULL;
1114 } else {
1115 Tokens *body = current ? &current->tokens : &top;
1116 for (int j = 0; j < tokens.count; j++) add_token(body, tokens.items[j]);
1117 }
1118 }
1119 if (current) error(current->token.line, "missing END for %s", current->token.text);
1120 for (Procedure *proc = procedures; proc; proc = proc->next) {
1121 Parser parser = {&proc->tokens, 0, true};
1122 proc->commands = sequence(&parser);
1123 }
1124 Parser parser = {&top, 0, false};
1125 Command *commands = sequence(&parser);
1126 clear_canvas();
1127 turtle_x = turtle_y = heading = pen = 0;
1128 pen_character = '*';
1129 random_state = 1;
1130 repetition_count = 0;
1131 environments[0] = (Environment){0};
1132 environments[0].test = -1;
1133 environment_count = 1;
1134 running = true;
1135 execute(commands);
1136 const char *pens[] = {"DOWN", "UP", "ERASE"};
1137 printf("TURTLE %d %d %d %s %c\n", turtle_x, turtle_y, heading, pens[pen], pen_character);
1138 picture();
1139 }
1140 for (int i = 0; i < environment_count; i++) {
1141 free(environments[i].variables);
1142 environments[i] = (Environment){0};
1143 }
1144 environment_count = 0;
1145 while (runtime_allocations) runtime_release(runtime_allocations);
1146 clear_syntax();
1147 running = false;
1148}

1150static char *read_line(void) {
1151 size_t count = 0, capacity = 128;
1152 char *line = malloc(capacity);
1153 int ch;
1154 if (!line) exit(EXIT_FAILURE);
1155 while ((ch = getchar()) != EOF && ch != '\n') {
1156 if (count + 1 == capacity) {
1157 capacity *= 2;
1158 char *next = realloc(line, capacity);
1159 if (!next) exit(EXIT_FAILURE);
1160 line = next;
1161 }
1162 line[count++] = (char)ch;
1163 }
1164 if (ch == EOF && !count) {
1165 free(line);
1166 return NULL;
1167 }
1168 line[count] = '\0';
1169 return line;
1170}

1172int main(void) {
1173 char **lines = NULL, *line;
1174 int count = 0, capacity = 0;
1175 while ((line = read_line())) {
1176 if (count == capacity) {
1177 capacity = capacity ? 2 * capacity : 64;
1178 char **next = realloc(lines, (size_t)capacity * sizeof *next);
1179 if (!next) exit(EXIT_FAILURE);
1180 lines = next;

1185 const char *program_name = lines[i] + 9;

1187 while (strcmp(lines[i], "#EOF") && strncmp(lines[i], "#PROGRAM ", 9)) i++;
1188 run_program(program_name, lines + first, i - first, first + 1);
1189 }
1190 clear_canvas();
1191 free(marked_positions);
1192 free(repetitions);
1193 for (int i = 0; i < count; i++) free(lines[i]);
1194 free(lines);
