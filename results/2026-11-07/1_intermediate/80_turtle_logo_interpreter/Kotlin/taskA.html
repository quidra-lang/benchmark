6private val signatures = mapOf(

14private val aliases = mapOf("FORWARD" to "FD", "BACK" to "BK", "RIGHT" to "RT", "LEFT" to "LT", "SETHEADING" to "SETH", "PENUP" to "PU", "PENDOWN" to "PD", "PENERASE" to "PE", "CLEARSCREEN" to "CS", "IFT" to "IFTRUE", "IFF" to "IFFALSE", "OP" to "OUTPUT", "PR" to "PRINT")

17private fun canonical(s: String) = aliases[s] ?: s

21private fun name(s: String) = s.isNotEmpty() && s[0] in 'A'..'Z' && s.all {

25private fun lex(text: String, line: Int): List<Token> {
26 val s = text.substringBefore(';').uppercase()
27 val out = mutableListOf<Token>()
28 var i = 0
29 while (i < s.length) {
30 val c = s[i]
31 if (c == ' ') {
32 i++
33 continue
34 }
35 if (c in "[]()+-*/=<>") {
36 var word = c.toString()
37 var kind = "symbol"
38 if (c == '-' && (i == 0 || s[i - 1] in " [(") && i + 1 < s.length && s[i + 1] != ' ') kind = "sign"
39 if (c == '<' && i + 1 < s.length && s[i + 1] in "=>" || c == '>' && i + 1 < s.length && s[i + 1] == '=') {
40 word += s[i + 1]
41 i++
42 }
43 out.add(Token(word, kind, line))
44 i++
45 continue
46 }
47 val start = i
48 while (i < s.length && s[i] !in " []()+-*/=<>") i++
49 val word = s.substring(start, i)
50 val kind = when {
51 word.all {
52 it in '0'..'9'
53 } -> "number"
54 word.startsWith(':') && name(word.drop(1)) -> "variable"
55 word.startsWith('"') && name(word.drop(1)) -> "quote"
56 name(word) -> "word"
57 else -> throw Problem(line, "bad token '$word'")
58 }
59 out.add(Token(word, kind, line))
60 }
61 return out
62}

75private class Parser(val tokens: List<Token>, val procedures: Map<String, Procedure>, val inside: Boolean) {
76 var at = 0
77
78 fun unknown(t: Token) = t.kind == "word" && !primitive(t.text) && t.text !in procedures
79
80 fun next(owner: Token): Token {
81 val t = tokens.getOrNull(at)
82 if (t == null || t.text == "]" || t.text == ")") fail(owner, "not enough inputs to ${owner.text}")
83 if (unknown(t)) fail(t, "I don't know how to ${t.text}")
84 return t
85 }
86
87 fun rejected(owner: Token, t: Token): Nothing = fail(t, "${owner.text} doesn't like ${t.text} as input")
88
89 fun expression(owner: Token): Expr = binary(owner, 0)
90
91 fun binary(owner: Token, level: Int): Expr {
92 if (level == 3) return term(owner)
93 val operators = when (level) {
94 0 -> setOf("=", "<>", "<", ">", "<=", ">=")
95 1 -> setOf("+", "-")
96 else -> setOf("*", "/", "MOD")
97 }
98 var left = binary(owner, level + 1)
99 while (at < tokens.size && tokens[at].text in operators && tokens[at].kind != "sign") {
100 val op = tokens[at++]
101 val right = binary(op, level + 1)
102 left = Expr(op, op.text, listOf(left, right))
103 }
104 return left
105 }
106
107 fun term(owner: Token): Expr {
108 val t = next(owner)
109 at++
110 return when {
111 t.kind == "number" -> Expr(t, "NUMBER")
112 t.kind == "variable" -> Expr(t, "VARIABLE")
113 t.text == "-" -> Expr(t, "NEG", listOf(term(t)))
114 t.text == "(" -> {
115 val e = expression(t)
116 if (tokens.getOrNull(at)?.text != ")") fail(t, "missing )")
117 at++
118 e
119 }
120 t.kind == "word" && t.text in functions -> Expr(t, "FUNCTION", List(functions.getValue(t.text)) {
121 expression(t)
122 })
123 t.kind == "word" && t.text in procedures -> Expr(t, "CALL", List(procedures.getValue(t.text).params.size) {
124 expression(t)
125 })
126 else -> rejected(owner, t)
127 }
128 }
129
130 fun item(owner: Token): Item {
131 val t = next(owner)
132 return if (t.kind == "quote") {
133 at++
134 Item(word = t.text.drop(1))
135 } else Item(expr = expression(owner))
136 }
137
138 fun list(owner: Token): List<Command> {
139 val open = next(owner)
140 if (open.text != "[") rejected(owner, open)
141 at++
142 val out = mutableListOf<Command>()
143 while (at < tokens.size && tokens[at].text != "]") out.add(command())
144 if (at == tokens.size) fail(open, "missing ]")
145 at++
146 return out
147 }
148
149 fun command(): Command {
150 val t = tokens[at++]
151 if (t.text == "]") fail(t, "unexpected ]")
152 if (t.text == "(" && tokens.getOrNull(at)?.let {
153 it.kind == "word" && canonical(it.text) == "PRINT"
154 } == true) {
155 val owner = tokens[at++]
156 val items = mutableListOf<Item>()
157 while (tokens.getOrNull(at)?.text != ")") {
158 if (at == tokens.size || tokens[at].text == "]") fail(t, "missing )")
159 items.add(item(owner))
160 }
161 at++
162 return Command(t, "MULTIPRINT", items = items)
163 }
164 if (t.kind == "word" && t.text in setOf("TO", "END")) fail(t, "misplaced ${t.text}")
165 val op = canonical(t.text)
166 if (t.kind == "word" && op in signatures) {
167 if (!inside && op in setOf("STOP", "OUTPUT", "LOCAL", "LOCALMAKE")) fail(t, "can only use ${t.text} inside a procedure")
168 val exprs = mutableListOf<Expr>()
169 val words = mutableListOf<String>()
170 val lists = mutableListOf<List<Command>>()
171 val items = mutableListOf<Item>()
172 for (kind in signatures.getValue(op)) when (kind) {
173 'E' -> exprs.add(expression(t))
174 'P' -> items.add(item(t))
175 'L' -> lists.add(list(t))
176 'Q' -> {
177 val q = next(t)
178 if (q.kind != "quote") rejected(t, q)
179 at++
180 words.add(q.text.drop(1))
181 if (op in setOf("TRACE", "UNTRACE") && q.text.drop(1) !in procedures) fail(q, "I don't know how to ${q.text.drop(1)}")
182 }
183 }
184 return Command(t, op, exprs, words, lists, items)
185 }
186 if (t.kind == "word" && t.text in procedures) return Command(t, "CALL", List(procedures.getValue(t.text).params.size) {
187 expression(t)
188 })
189 if (unknown(t)) fail(t, "I don't know how to ${t.text}")
190 fail(t, "you don't say what to do with ${t.text}")
191 }
192
193 fun sequence(): List<Command> {
194 val out = mutableListOf<Command>()
195 while (at < tokens.size) out.add(command())
196 return out
197 }
198}

206private class Runtime(val procedures: Map<String, Procedure>, val output: MutableList<String>) {

220 fun variable(t: Token): Long {
221 val name = t.text.drop(1)
222 for (env in envs.asReversed()) env.values[name]?.let {
223 return it
224 }
225 fail(t, "$name has no value")
226 }

228 fun make(name: String, value: Long) {
229 for (i in envs.lastIndex downTo 1) if (name in envs[i].values) {
230 envs[i].values[name] = value
231 return
232 }
233 envs[0].values[name] = value
234 }

236 fun invoke(t: Token, values: List<Long>): Long? {
237 if (calls.size == 100) fail(t, "recursion too deep")
238 val p = procedures.getValue(t.text)
239 val trace = t.text in traced
240 val indent = " ".repeat(calls.size)
241 if (trace) output.add(indent + "( " + t.text + values.joinToString("") {
242 " $it"
243 } + " )")
244 envs.add(Environment(p.params.zip(values).toMap().toMutableMap()))
245 calls.add(t.text)
246 var value: Long? = null
247 try {
248 execute(p.body)
249 } catch (flow: Flow) {
250 value = flow.value
251 }
252 envs.removeAt(envs.lastIndex)
253 calls.removeAt(calls.lastIndex)
254 if (trace) output.add(indent + t.text + if (value == null) " stops" else " outputs $value")
255 return value
256 }

258 fun evaluate(e: Expr): Long {
259 val t = e.token
260 if (e.op == "NUMBER") return t.text.toLong()
261 if (e.op == "VARIABLE") return variable(t)
262 val a = e.args.map(::evaluate)
263 if (e.op == "NEG") return -a[0]
264 if (e.op == "CALL") return invoke(t, a) ?: fail(t, "${t.text} did not output")
265 if (e.op == "FUNCTION") return when (t.text) {
266 "XCOR" -> x.toLong()
267 "YCOR" -> y.toLong()
268 "HEADING" -> heading.toLong()
269 "REPCOUNT" -> repetitions.lastOrNull() ?: fail(t, "REPCOUNT outside REPEAT")
270 "RANDOM" -> {
271 if (a[0] <= 0) fail(t, "RANDOM doesn't like ${a[0]} as input")
272 random = 48271 * random % 2147483647
273 random % a[0]
274 }
275 "ABS" -> kotlin.math.abs(a[0])
276 "NOT" -> if (a[0] == 0L) 1L else 0L
277 "AND" -> if (a[0] != 0L && a[1] != 0L) 1L else 0L
278 "OR" -> if (a[0] != 0L || a[1] != 0L) 1L else 0L
279 "MAX" -> maxOf(a[0], a[1])
280 else -> minOf(a[0], a[1])
281 }
282 if ((e.op == "/" || e.op == "MOD") && a[1] == 0L) fail(t, "${t.text} doesn't like 0 as input")
283 return when (e.op) {
284 "+" -> a[0] + a[1]
285 "-" -> a[0] - a[1]
286 "*" -> a[0] * a[1]
287 "/" -> a[0] / a[1]
288 "MOD" -> a[0] % a[1]
289 "=" -> if (a[0] == a[1]) 1L else 0L
290 "<>" -> if (a[0] != a[1]) 1L else 0L
291 "<" -> if (a[0] < a[1]) 1L else 0L
292 ">" -> if (a[0] > a[1]) 1L else 0L
293 "<=" -> if (a[0] <= a[1]) 1L else 0L
294 else -> if (a[0] >= a[1]) 1L else 0L
295 }
296 }

305 fun move(n: Long) {
306 val d = directions[heading / 45]
307 val sign = if (n < 0) -1 else 1
308 apply(x, y)
309 repeat(kotlin.math.abs(n).toInt()) {
310 x += sign * d.first
311 y += sign * d.second
312 apply(x, y)
313 }
314 }

321 fun straight(xx: Int, yy: Int) {
322 val dx = xx - x
323 val dy = yy - y
324 val size = maxOf(kotlin.math.abs(dx), kotlin.math.abs(dy))
325 if (size == 0) apply(x, y) else for (i in 0..size) apply(x + floor(2L * i * dx + size, 2L * size).toInt(), y + floor(2L * i * dy + size, 2L * size).toInt())
326 x = xx
327 y = yy
328 }

330 fun fill() {
331 if (canvas.isEmpty() || Point(x, y) in canvas) return
332 val x1 = canvas.keys.minOf {
333 it.x
334 }
335 val x2 = canvas.keys.maxOf {
336 it.x
337 }
338 val y1 = canvas.keys.minOf {
339 it.y
340 }
341 val y2 = canvas.keys.maxOf {
342 it.y
343 }
344 if (x !in x1..x2 || y !in y1..y2) return
345 val queue = ArrayDeque<Point>()
346 queue.add(Point(x, y))
347 canvas[Point(x, y)] = character
348 while (queue.isNotEmpty()) {
349 val p = queue.removeFirst()
350 for ((dx, dy) in listOf(0 to 1, 1 to 0, 0 to -1, -1 to 0)) {
351 val q = Point(p.x + dx, p.y + dy)
352 if (q.x in x1..x2 && q.y in y1..y2 && q !in canvas) {
353 canvas[q] = character
354 queue.add(q)
355 }
356 }
357 }

362 output.add("CANVAS 0 0 0")

366 it.x
367 }
368 val x2 = canvas.keys.maxOf {

387 fun command(c: Command) {
388 val t = c.token
389 val op = c.op
390 if (op == "WHILE") {
391 while (evaluate(c.exprs[0]) != 0L) execute(c.lists[0])
392 return
393 }
394 if (op == "PRINT" || op == "MULTIPRINT") {
395 val values = c.items.map(::item)
396 output.add(values.joinToString(" "))
397 return
398 }
399 if (op == "LABEL") {
400 val text = item(c.items[0])
401 val d = directions[heading / 45]
402 for ((i, ch) in text.withIndex()) canvas[Point(x + i * d.first, y + i * d.second)] = ch
403 return
404 }
405 val a = c.exprs.map(::evaluate)
406 when (op) {
407 "CALL" -> invoke(t, a)?.let {
408 fail(t, "you don't say what to do with $it")
409 }
410 "FD" -> move(a[0])
411 "BK" -> move(-a[0])
412 "RT", "LT", "SETH" -> {
413 val angle = a[0]
414 if (angle % 45 != 0L) fail(t, "${t.text} doesn't like $angle as input")
415 val h = if (op == "SETH") angle else heading + if (op == "RT") angle else -angle
416 heading = ((h % 360 + 360) % 360).toInt()
417 }
418 "PU" -> pen = "UP"
419 "PD" -> pen = "DOWN"
420 "PE" -> pen = "ERASE"
421 "SETPEN" -> {
422 if (a[0] !in 33L..126L) fail(t, "SETPEN doesn't like ${a[0]} as input")
423 character = a[0].toInt().toChar()
424 }
425 "SETXY" -> straight(a[0].toInt(), a[1].toInt())
426 "SETX" -> straight(a[0].toInt(), y)
427 "SETY" -> straight(x, a[0].toInt())
428 "HOME" -> {
429 straight(0, 0)
430 heading = 0
431 }
432 "CS" -> {
433 canvas.clear()
434 x = 0
435 y = 0
436 heading = 0
437 }
438 "CLEAN" -> canvas.clear()
439 "FILL" -> fill()
440 "SNAPSHOT" -> picture()
441 "MAKE" -> make(c.words[0], a[0])
442 "LOCAL" -> envs.last().values[c.words[0]] = 0
443 "LOCALMAKE" -> envs.last().values[c.words[0]] = a[0]
444 "TRACE" -> traced.add(c.words[0])
445 "UNTRACE" -> traced.remove(c.words[0])
446 "SEED" -> {
447 if (a[0] <= 0) fail(t, "SEED doesn't like ${a[0]} as input")
448 random = a[0]
449 }
450 "STOP" -> throw Flow(null)
451 "OUTPUT" -> throw Flow(a[0])
452 "REPEAT" -> {
453 if (c.lists[0].isEmpty()) return
454 var i = 1L
455 while (i <= a[0]) {
456 repetitions.add(i)
457 try {
458 execute(c.lists[0])
459 } finally {
460 repetitions.removeAt(repetitions.lastIndex)
461 }
462 i++
463 }
464 }
465 "FOR" -> {
466 if (c.lists[0].isEmpty()) {
467 make(c.words[0], a[1])
468 return
469 }
470 val step = if (a[0] <= a[1]) 1L else -1L
471 var i = a[0]
472 while (true) {
473 make(c.words[0], i)
474 execute(c.lists[0])
475 if (i == a[1]) break
476 i += step
477 }
478 }
479 "IF" -> if (a[0] != 0L) execute(c.lists[0])
480 "IFELSE" -> execute(c.lists[if (a[0] != 0L) 0 else 1])
481 "TEST" -> envs.last().test = a[0] != 0L
482 "IFTRUE", "IFFALSE" -> {
483 val test = envs.last().test ?: fail(t, "${t.text} without TEST")
484 if (test == (op == "IFTRUE")) execute(c.lists[0])
485 }
486 }
487 }

503private fun process(source: List<Pair<Int, String>>, output: MutableList<String>) {
504 val procedures = linkedMapOf<String, Procedure>()
505 val top = mutableListOf<Token>()
506 var current: Procedure? = null
507 try {
508 for ((line, text) in source) {
509 val tokens = lex(text, line)
510 if (tokens.isEmpty()) continue
511 if (tokens[0].kind == "word" && tokens[0].text == "TO") {
512 val start = tokens[0]
513 current?.let {
514 fail(start, "TO inside definition of ${it.token.text}")
515 }
516 if (tokens.size < 2 || tokens[1].kind != "word") fail(start, "bad procedure header")
517 val name = tokens[1]
518 if (primitive(name.text)) fail(start, "${name.text} is a primitive")
519 if (name.text in procedures) fail(start, "${name.text} is already defined")
520 val params = mutableListOf<String>()
521 for (t in tokens.drop(2)) {
522 if (t.kind != "variable") fail(start, "bad procedure header")
523 if (t.text.drop(1) in params) fail(start, "duplicate input ${t.text}")
524 params.add(t.text.drop(1))
525 }
526 current = Procedure(Token(name.text, "word", line), params)
527 procedures[name.text] = current!!
528 } else if (tokens.size == 1 && tokens[0].kind == "word" && tokens[0].text == "END") {
529 if (current == null) fail(tokens[0], "END without TO")
530 current = null
531 } else if (current != null) current!!.raw.addAll(tokens) else top.addAll(tokens)
532 }
533 current?.let {
534 fail(it.token, "missing END for ${it.token.text}")
535 }
536 for (p in procedures.values) p.body = Parser(p.raw, procedures, true).sequence()
537 val body = Parser(top, procedures, false).sequence()
538 val runtime = Runtime(procedures, output)
539 try {
540 runtime.execute(body)
541 output.add("TURTLE ${runtime.x} ${runtime.y} ${runtime.heading} ${runtime.pen} ${runtime.character}")
542 runtime.picture()
543 } catch (e: Problem) {
544 output.add("ERROR line ${e.line}: ${e.reason}")
545 runtime.errors()
546 }
547 } catch (e: Problem) {
548 output.add("ERROR line ${e.line}: ${e.reason}")
549 }
550}

552fun main() {
553 val lines = generateSequence(::readLine).toList()
554 val output = mutableListOf<String>()
555 var at = 0
556 while (at < lines.size && lines[at] != "#EOF") {
557 output.add("PROGRAM ${lines[at].drop(9)}")
558 at++
559 val source = mutableListOf<Pair<Int, String>>()
560 while (at < lines.size && lines[at] != "#EOF" && !lines[at].startsWith("#PROGRAM ")) {
561 source.add(at + 1 to lines[at])
562 at++
563 }
564 process(source, output)
565 }
566 for (line in output) println(line)
567}
