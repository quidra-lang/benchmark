137 static boolean name(String value) {
138 if (value.isEmpty() || value.charAt(0) < 'A' || value.charAt(0) > 'Z') return false;
139 for (int i = 1; i < value.length(); i++) {
140 char c = value.charAt(i);
141 if (!(c >= 'A' && c <= 'Z' || c >= '0' && c <= '9' || c == '_')) return false;
142 }
143 return true;
144 }

146 static List<Token> lex(String source, int line) {
147 int comment = source.indexOf(';');
148 if (comment >= 0) source = source.substring(0, comment);
149 StringBuilder upper = new StringBuilder();
150 for (char c : source.toCharArray()) upper.append(c >= 'a' && c <= 'z' ? (char)(c - 'a' + 'A') : c);
151 String text = upper.toString();
152 List<Token> tokens = new ArrayList<>();
153 for (int i = 0; i < text.length();) {
154 char c = text.charAt(i);
155 if (c == ' ') {
156 i++;
157 continue;
158 }
159 if ("[]()+*/=".indexOf(c) >= 0) {
160 tokens.add(new Token("symbol", Character.toString(c), line));
161 i++;
162 } else if (c == '<' || c == '>') {
163 int length = i + 1 < text.length() && (text.charAt(i + 1) == '=' || c == '<' && text.charAt(i + 1) == '>') ? 2 : 1;
164 tokens.add(new Token("symbol", text.substring(i, i + length), line));
165 i += length;
166 } else if (c == '-') {
167 boolean sign = (i == 0 || " [(".indexOf(text.charAt(i - 1)) >= 0) && i + 1 < text.length() && text.charAt(i + 1) != ' ';
168 tokens.add(new Token(sign ? "sign" : "minus", "-", line));
169 i++;
170 } else {
171 int start = i;
172 while (i < text.length() && " []()+-*/=<>".indexOf(text.charAt(i)) < 0) i++;
173 String word = text.substring(start, i);
174 String kind;
175 if (word.chars().allMatch(x -> x >= '0' && x <= '9')) kind = "number";
176 else if (word.startsWith(":") && name(word.substring(1))) kind = "variable";
177 else if (word.startsWith("\"") && name(word.substring(1))) kind = "quoted";
178 else if (name(word)) kind = "word";
179 else throw new LogoError(line, "bad token '" + word + "'");
180 tokens.add(new Token(kind, word, line));
181 }
182 }
183 return tokens;
184 }

198 Token peek() {
199 return position < tokens.size() ? tokens.get(position) : null;
200 }
201
202 boolean is(String text) {
203 return peek() != null && peek().text.equals(text);
204 }
205
206 boolean unknown(Token token) {
207 return token.kind.equals("word") && !PRIMITIVES.containsKey(token.text) && !procedures.containsKey(token.text);
208 }
209
210 Token input(Token owner) {
211 Token next = peek();
212 if (next == null || next.text.equals("]") || next.text.equals(")")) throw new LogoError(owner.line, "not enough inputs to " + owner.text);
213 if (unknown(next)) throw new LogoError(next.line, "I don't know how to " + next.text);
214 return next;
215 }
216
217 void reject(Token owner, Token next) {
218 throw new LogoError(next.line, owner.text + " doesn't like " + next.text + " as input");
219 }
220
221 List<Command> sequence() {
222 List<Command> result = new ArrayList<>();
223 while (peek() != null) {
224 if (is("]")) throw new LogoError(peek().line, "unexpected ]");
225 result.add(command());
226 }
227 return result;
228 }
229
230 List<Command> block(Token owner) {
231 Token opener = input(owner);
232 if (!opener.text.equals("[")) reject(owner, opener);
233 position++;
234 List<Command> result = new ArrayList<>();
235 while (peek() != null && !is("]")) result.add(command());
236 if (peek() == null) throw new LogoError(opener.line, "missing ]");
237 position++;
238 return result;
239 }
240
241 String quoted(Token owner) {
242 Token next = input(owner);
243 if (!next.kind.equals("quoted")) reject(owner, next);
244 position++;
245 return next.text.substring(1);
246 }
247
248 PrintItem printItem(Token owner) {
249 Token next = input(owner);
250 if (next.kind.equals("quoted")) {
251 position++;
252 return new PrintItem(next.text.substring(1), null);
253 }
254 return new PrintItem(null, expression(owner));
255 }
256
257 Command command() {
258 Token token = tokens.get(position++);
259 if (token.text.equals("(") && peek() != null && peek().kind.equals("word") && (is("PRINT") || is("PR"))) {
260 Token print = tokens.get(position++);
261 Command command = new Command(print, "MULTIPRINT");
262 while (!is(")")) {
263 if (peek() == null || is("]")) throw new LogoError(token.line, "missing )");
264 command.prints.add(printItem(print));
265 }
266 position++;
267 return command;
268 }
269 if (token.kind.equals("word") && (token.text.equals("TO") || token.text.equals("END"))) throw new LogoError(token.line, "misplaced " + token.text);
270 Primitive primitive = token.kind.equals("word") ? PRIMITIVES.get(token.text) : null;
271 if (primitive != null && !primitive.function && !primitive.name.equals("MOD")) {
272 if (!inside && Set.of("STOP", "OUTPUT", "LOCAL", "LOCALMAKE").contains(primitive.name)) throw new LogoError(token.line, "can only use " + token.text + " inside a procedure");
273 Command command = new Command(token, primitive.name);
274 for (char kind : primitive.inputs.toCharArray()) switch (kind) {
275 case 'E' -> command.expressions.add(expression(token));
276 case 'L' -> command.blocks.add(block(token));
277 case 'Q' -> command.quoted = quoted(token);
278 case 'P' -> command.prints.add(printItem(token));
279 default -> throw new IllegalStateException();
280 }
281 if ((primitive.name.equals("TRACE") || primitive.name.equals("UNTRACE")) && !procedures.containsKey(command.quoted)) {
282 Token word = tokens.get(position - 1);
283 throw new LogoError(word.line, "I don't know how to " + command.quoted);
284 }
285 return command;
286 }
287 if (token.kind.equals("word") && procedures.containsKey(token.text)) {
288 Command command = new Command(token, "CALL");
289 for (int i = 0; i < procedures.get(token.text).parameters.size(); i++) command.expressions.add(expression(token));
290 return command;
291 }
292 if (unknown(token)) throw new LogoError(token.line, "I don't know how to " + token.text);
293 throw new LogoError(token.line, "you don't say what to do with " + token.text);
294 }
295
296 Expression expression(Token owner) {
297 return comparison(owner);
298 }
299
300 Expression binary(Expression left, Token operator, Expression right) {
301 Expression result = new Expression(operator, "binary");
302 result.arguments.add(left);
303 result.arguments.add(right);
304 return result;
305 }
306
307 Expression comparison(Token owner) {
308 Expression result = sum(owner);
309 while (peek() != null && List.of("=", "<>", "<", ">", "<=", ">=").contains(peek().text)) {
310 Token operator = tokens.get(position++);
311 result = binary(result, operator, sum(operator));
312 }
313 return result;
314 }
315
316 Expression sum(Token owner) {
317 Expression result = product(owner);
318 while (peek() != null && (is("+") || is("-") && peek().kind.equals("minus"))) {
319 Token operator = tokens.get(position++);
320 result = binary(result, operator, product(operator));
321 }
322 return result;
323 }
324
325 Expression product(Token owner) {
326 Expression result = term(owner);
327 while (peek() != null && (is("*") || is("/") || is("MOD"))) {
328 Token operator = tokens.get(position++);
329 result = binary(result, operator, term(operator));
330 }
331 return result;
332 }
333
334 Expression term(Token owner) {
335 Token token = input(owner);
336 position++;
337 if (token.kind.equals("number") || token.kind.equals("variable")) return new Expression(token, token.kind);
338 if (token.text.equals("-")) {
339 Expression result = new Expression(token, "negate");
340 result.arguments.add(term(token));
341 return result;
342 }
343 if (token.text.equals("(")) {
344 Expression result = expression(token);
345 if (!is(")")) throw new LogoError(token.line, "missing )");
346 position++;
347 return result;
348 }
349 if (token.kind.equals("word")) {
350 Primitive primitive = PRIMITIVES.get(token.text);
351 if (primitive != null && primitive.function || procedures.containsKey(token.text)) {
352 Expression result = new Expression(token, primitive != null ? "function" : "call");
353 int count = primitive != null ? primitive.inputs.length() : procedures.get(token.text).parameters.size();
354 for (int i = 0; i < count; i++) result.arguments.add(expression(token));
355 return result;
356 }
357 }
358 reject(owner, token);
359 return null;
360 }
361 }

413 long evaluate(Expression expression) {
414 Token token = expression.token;
415 switch (expression.kind) {
416 case "number":
417 return Long.parseLong(token.text);
418 case "variable":
419 return variable(token);
420 case "negate":
421 return -evaluate(expression.arguments.get(0));
422 case "binary":
423 {
424 long a = evaluate(expression.arguments.get(0)), b = evaluate(expression.arguments.get(1));
425 if ((token.text.equals("/") || token.text.equals("MOD")) && b == 0) error(token, token.text + " doesn't like 0 as input");
426 return switch (token.text) {
427 case "+" -> a + b;
428 case "-" -> a - b;
429 case "*" -> a * b;
430 case "/" -> a / b;
431 case "MOD" -> a % b;
432 case "=" -> a == b ? 1 : 0;
433 case "<>" -> a != b ? 1 : 0;
434 case "<" -> a < b ? 1 : 0;
435 case ">" -> a > b ? 1 : 0;
436 case "<=" -> a <= b ? 1 : 0;
437 case ">=" -> a >= b ? 1 : 0;
438 default -> throw new IllegalStateException();
439 };
440 }
441 default:
442 {
443 List<Long> values = new ArrayList<>();
444 for (Expression argument : expression.arguments) values.add(evaluate(argument));
445 if (expression.kind.equals("call")) {
446 Long value = call(token, values);
447 if (value == null) error(token, token.text + " did not output");
448 return value;
449 }
450 long a = values.isEmpty() ? 0 : values.get(0), b = values.size() < 2 ? 0 : values.get(1);
451 return switch (token.text) {
452 case "XCOR" -> x;
453 case "YCOR" -> y;
454 case "HEADING" -> heading;
455 case "REPCOUNT" -> {
456 if (repetitions.isEmpty()) error(token, "REPCOUNT outside REPEAT");
457 yield repetitions.getLast();
458 }
459 case "RANDOM" -> {
460 if (a < 1) error(token, "RANDOM doesn't like " + a + " as input");
461 random = random * 48271 % 2147483647;
462 yield random % a;
463 }
464 case "ABS" -> Math.abs(a);
465 case "NOT" -> a == 0 ? 1 : 0;
466 case "AND" -> a != 0 && b != 0 ? 1 : 0;
467 case "OR" -> a != 0 || b != 0 ? 1 : 0;
468 case "MAX" -> Math.max(a, b);
469 case "MIN" -> Math.min(a, b);
470 default -> throw new IllegalStateException();
471 };
472 }
473 }
474 }

476 Long call(Token token, List<Long> values) {
477 int depth = environments.size() - 1;
478 if (depth == 100) error(token, "recursion too deep");
479 Procedure procedure = procedures.get(token.text);
480 boolean trace = traced.contains(token.text);
481 String indent = " ".repeat(depth);
482 if (trace) {
483 List<String> parts = new ArrayList<>();
484 parts.add(token.text);
485 for (long value : values) parts.add(Long.toString(value));
486 System.out.println(indent + "( " + String.join(" ", parts) + " )");
487 }
488 Environment environment = new Environment(procedure);
489 for (int i = 0; i < values.size(); i++) environment.variables.put(procedure.parameters.get(i), values.get(i));
490 environments.add(environment);
491 Long result = null;
492 try {
493 execute(procedure.commands);
494 } catch (Flow flow) {
495 result = flow.value;
496 }
497 environments.removeLast();
498 if (trace) System.out.println(indent + token.text + (result == null ? " stops" : " outputs " + result));
499 return result;
500 }

523 void move(int targetX, int targetY) {
524 int oldX = x, oldY = y, dx = targetX - x, dy = targetY - y, steps = Math.max(Math.abs(dx), Math.abs(dy));
525 if (steps == 0) apply(x, y);
526 else for (int i = 0; i <= steps; i++) apply(oldX + (int)Math.floorDiv(2L * i * dx + steps, 2L * steps), oldY + (int)Math.floorDiv(2L * i * dy + steps, 2L * steps));
527 x = targetX;
528 y = targetY;
529 }

531 void fill() {
532 if (canvas.isEmpty() || canvas.containsKey(new Point(x, y))) return;
533 int x1 = canvas.keySet().stream().mapToInt(Point::x).min().orElseThrow(), x2 = canvas.keySet().stream().mapToInt(Point::x).max().orElseThrow(), y1 = canvas.keySet().stream().mapToInt(Point::y).min().orElseThrow(), y2 = canvas.keySet().stream().mapToInt(Point::y).max().orElseThrow();
534 if (x < x1 || x > x2 || y < y1 || y > y2) return;
535 int[] dx = {-1, 1, 0, 0}, dy = {0, 0, -1, 1};
536 ArrayDeque<Point> queue = new ArrayDeque<>();
537 Point first = new Point(x, y);
538 canvas.put(first, penCharacter);
539 queue.add(first);
540 while (!queue.isEmpty()) {
541 Point point = queue.remove();
542 for (int d = 0; d < 4; d++) {
543 Point next = new Point(point.x + dx[d], point.y + dy[d]);
544 if (next.x >= x1 && next.x <= x2 && next.y >= y1 && next.y <= y2 && !canvas.containsKey(next)) {
545 canvas.put(next, penCharacter);
546 queue.add(next);
547 }
548 }
549 }

576 void execute(Command command) {
577 Token token = command.token;
578 String name = command.name;
579 if (name.equals("WHILE")) { while (evaluate(command.expressions.get(0)) != 0) execute(command.blocks.get(0));
580 return;
581 }
582 List<Long> values = new ArrayList<>();
583 for (Expression expression : command.expressions) values.add(evaluate(expression));
584 long a = values.isEmpty() ? 0 : values.get(0), b = values.size() < 2 ? 0 : values.get(1);
585 Environment current = environments.getLast();
586 switch (name) {
587 case "FD" -> forward(a);
588 case "BK" -> forward(-a);
589 case "RT", "LT", "SETH" -> {
590 if (a % 45 != 0) error(token, token.text + " doesn't like " + a + " as input");
591 long angle = name.equals("SETH") ? a : name.equals("RT") ? heading + a : heading - a;
592 heading = (int)((angle % 360 + 360) % 360);
593 }
594 case "PU" -> pen = "UP";
595 case "PD" -> pen = "DOWN";
596 case "PE" -> pen = "ERASE";
597 case "SETPEN" -> {
598 if (a < 33 || a > 126) error(token, "SETPEN doesn't like " + a + " as input");
599 penCharacter = (char)a;
600 }
601 case "SETXY" -> move((int)a, (int)b);
602 case "SETX" -> move((int)a, y);
603 case "SETY" -> move(x, (int)a);
604 case "HOME" -> {
605 move(0, 0);
606 heading = 0;
607 }
608 case "CS" -> {
609 canvas.clear();
610 x = y = heading = 0;
611 }
612 case "CLEAN" -> canvas.clear();
613 case "FILL" -> fill();
614 case "LABEL" -> {
615 String text = item(command.prints.get(0));
616 int dx = DX[heading / 45], dy = DY[heading / 45];
617 for (int i = 0; i < text.length(); i++) canvas.put(new Point(x + i * dx, y + i * dy), text.charAt(i));
618 }
619 case "SNAPSHOT" -> picture();
620 case "REPEAT" -> {
621 if (!command.blocks.get(0).isEmpty()) for (long i = 1; i <= a; i++) {
622 repetitions.add(i);
623 try {
624 execute(command.blocks.get(0));
625 } finally {
626 repetitions.removeLast();
627 }
628 }
629 }
630 case "FOR" -> {
631 List<Command> body = command.blocks.get(0);
632 if (body.isEmpty()) make(command.quoted, b);
633 else for (long i = a;; i += a <= b ? 1 : -1) {
634 make(command.quoted, i);
635 execute(body);
636 if (i == b) break;
637 }
638 }
639 case "IF" -> {
640 if (a != 0) execute(command.blocks.get(0));
641 }
642 case "IFELSE" -> execute(command.blocks.get(a != 0 ? 0 : 1));
643 case "TEST" -> current.test = a != 0;
644 case "IFTRUE", "IFFALSE" -> {
645 if (current.test == null) error(token, token.text + " without TEST");
646 if (current.test == name.equals("IFTRUE")) execute(command.blocks.get(0));
647 }
648 case "STOP" -> throw new Flow(null);
649 case "OUTPUT" -> throw new Flow(a);
650 case "MAKE" -> make(command.quoted, a);
651 case "LOCAL" -> current.variables.put(command.quoted, 0L);
652 case "LOCALMAKE" -> current.variables.put(command.quoted, a);
653 case "TRACE" -> traced.add(command.quoted);
654 case "UNTRACE" -> traced.remove(command.quoted);
655 case "SEED" -> {
656 if (a < 1) error(token, "SEED doesn't like " + a + " as input");
657 random = a;
658 }
659 case "PRINT", "MULTIPRINT" -> {
660 List<String> items = new ArrayList<>();
661 for (PrintItem item : command.prints) items.add(item(item));
662 System.out.println(String.join(" ", items));
663 }
664 case "CALL" -> {
665 Long result = call(token, values);
666 if (result != null) error(token, "you don't say what to do with " + result);
667 }
668 default -> throw new IllegalStateException();
669 }
670 }
671 }

673 static void run(String name, List<String> source, int firstLine) {
674 System.out.println("PROGRAM " + name);
675 try {
676 LinkedHashMap<String, Procedure> procedures = new LinkedHashMap<>();
677 List<Token> top = new ArrayList<>();
678 Procedure current = null;
679 for (int i = 0; i < source.size(); i++) {
680 int line = firstLine + i;
681 List<Token> tokens = lex(source.get(i), line);
682 if (tokens.isEmpty()) continue;
683 Token first = tokens.get(0);
684 if (first.kind.equals("word") && first.text.equals("TO")) {
685 if (current != null) throw new LogoError(line, "TO inside definition of " + current.token.text);
686 if (tokens.size() < 2 || !tokens.get(1).kind.equals("word")) throw new LogoError(line, "bad procedure header");
687 Token procedureName = tokens.get(1);
688 if (PRIMITIVES.containsKey(procedureName.text)) throw new LogoError(line, procedureName.text + " is a primitive");
689 if (procedures.containsKey(procedureName.text)) throw new LogoError(line, procedureName.text + " is already defined");
690 Procedure procedure = new Procedure(procedureName);
691 Set<String> seen = new HashSet<>();
692 for (int j = 2; j < tokens.size(); j++) {
693 Token input = tokens.get(j);
694 if (!input.kind.equals("variable")) throw new LogoError(line, "bad procedure header");
695 if (!seen.add(input.text.substring(1))) throw new LogoError(line, "duplicate input " + input.text);
696 procedure.parameters.add(input.text.substring(1));
697 }
698 procedures.put(procedureName.text, procedure);
699 current = procedure;
700 } else if (tokens.size() == 1 && first.kind.equals("word") && first.text.equals("END")) {
701 if (current == null) throw new LogoError(line, "END without TO");
702 current = null;
703 } else if (current != null) current.tokens.addAll(tokens);
704 else top.addAll(tokens);
705 }
706 if (current != null) throw new LogoError(current.token.line, "missing END for " + current.token.text);
707 for (Procedure procedure : procedures.values()) procedure.commands = new Parser(procedure.tokens, procedures, true).sequence();
708 List<Command> commands = new Parser(top, procedures, false).sequence();
709 Runner runner = new Runner(procedures);
710 runner.execute(commands);
711 System.out.println("TURTLE " + runner.x + " " + runner.y + " " + runner.heading + " " + runner.pen + " " + runner.penCharacter);
712 runner.picture();
713 } catch (LogoError error) {
714 System.out.println("ERROR line " + error.line + ": " + error.reason);
715 for (int i = 0; i < error.stack.size();) {
716 String procedure = error.stack.get(i);
717 int end = i + 1;
718 while (end < error.stack.size() && error.stack.get(end).equals(procedure)) end++;
719 System.out.println(" in " + procedure + (end - i == 1 ? "" : " x" + (end - i)));
720 i = end;
721 }
722 }
723 }
724 public static void main(String[] args) throws Exception {
725 BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
726 List<String> lines = new ArrayList<>();
727 String line;
728 while ((line = reader.readLine()) != null) lines.add(line);
729 for (int i = 0; i < lines.size() && !lines.get(i).equals("#EOF");) {
730 String name = lines.get(i).substring(9);
731 int first = ++i;
732 List<String> source = new ArrayList<>();
733 while (!lines.get(i).equals("#EOF") && !lines.get(i).startsWith("#PROGRAM ")) source.add(lines.get(i++));
734 run(name, source, first + 1);
735 }
736 }
