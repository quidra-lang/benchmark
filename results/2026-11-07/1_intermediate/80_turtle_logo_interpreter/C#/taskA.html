115 static bool Name(string value) => value.Length > 0 && value[0] is >= 'A' and <= 'Z' && value.Skip(1).All(c => c is >= 'A' and <= 'Z' or >= '0' and <= '9' or '_');

117 static List<Token> Lex(string source, int line)
118 {
119 int comment = source.IndexOf(';');
120 if (comment >= 0) source = source[..comment];
121 string text = string.Concat(source.Select(c => c is >= 'a' and <= 'z' ? (char)(c - 'a' + 'A') : c));
122 List<Token> tokens = new();
123 for (int i = 0; i < text.Length;)
124 {
125 char c = text[i];
126 if (c == ' ')
127 {
128 i++;
129 continue;
130 }
131 if ("[]()+*/=".Contains(c))
132 {
133 tokens.Add(new("symbol", c.ToString(), line));
134 i++;
135 }
136 else if (c is '<' or '>')
137 {
138 int length = i + 1 < text.Length && (text[i + 1] == '=' || c == '<' && text[i + 1] == '>') ? 2 : 1;
139 tokens.Add(new("symbol", text.Substring(i, length), line));
140 i += length;
141 }
142 else if (c == '-')
143 {
144 bool sign = (i == 0 || " [(".Contains(text[i - 1])) && i + 1 < text.Length && text[i + 1] != ' ';
145 tokens.Add(new(sign ? "sign" : "minus", "-", line));
146 i++;
147 }
148 else
149 {
150 int start = i;
151 while (i < text.Length && !" []()+-*/=<>".Contains(text[i])) i++;
152 string word = text[start..i], kind;
153 if (word.All(x => x is >= '0' and <= '9')) kind = "number";
154 else if (word.StartsWith(':') && Name(word[1..])) kind = "variable";
155 else if (word.StartsWith('"') && Name(word[1..])) kind = "quoted";
156 else if (Name(word)) kind = "word";
157 else throw new LogoError(line, $"bad token '{word}'");
158 tokens.Add(new(kind, word, line));
159 }
160 }
161 return tokens;
162 }

164 class Parser(List<Token> tokens, Dictionary<string, Procedure> procedures, bool inside)
165 {
166 int position;
167
168 Token? Peek() => position < tokens.Count ? tokens[position] : null;
169
170 bool Is(string text) => Peek()?.Text == text;
171
172 bool Unknown(Token token) => token.Kind == "word" && !Primitives.ContainsKey(token.Text) && !procedures.ContainsKey(token.Text);
173
174 Token Input(Token owner)
175 {
176 Token? next = Peek();
177 if (next == null || next.Text is "]" or ")") throw new LogoError(owner.Line, "not enough inputs to " + owner.Text);
178 if (Unknown(next)) throw new LogoError(next.Line, "I don't know how to " + next.Text);
179 return next;
180 }
181
182 static void Reject(Token owner, Token next) => throw new LogoError(next.Line, $"{owner.Text} doesn't like {next.Text} as input");
183
184 public List<Command> Sequence()
185 {
186 List<Command> result = new();
187 while (Peek() != null)
188 {
189 if (Is("]")) throw new LogoError(Peek()!.Line, "unexpected ]");
190 result.Add(ParseCommand());
191 }
192 return result;
193 }
194
195 List<Command> Block(Token owner)
196 {
197 Token opener = Input(owner);
198 if (opener.Text != "[") Reject(owner, opener);
199 position++;
200 List<Command> result = new();
201 while (Peek() != null && !Is("]")) result.Add(ParseCommand());
202 if (Peek() == null) throw new LogoError(opener.Line, "missing ]");
203 position++;
204 return result;
205 }
206
207 string Quoted(Token owner)
208 {
209 Token next = Input(owner);
210 if (next.Kind != "quoted") Reject(owner, next);
211 position++;
212 return next.Text[1..];
213 }
214
215 PrintItem ParsePrintItem(Token owner)
216 {
217 Token next = Input(owner);
218 if (next.Kind == "quoted")
219 {
220 position++;
221 return new(next.Text[1..], null);
222 }
223 return new(null, ParseExpression(owner));
224 }
225
226 Command ParseCommand()
227 {
228 Token token = tokens[position++];
229 if (token.Text == "(" && Peek()?.Kind == "word" && (Is("PRINT") || Is("PR")))
230 {
231 Token print = tokens[position++];
232 Command command = new(print, "MULTIPRINT");
233 while (!Is(")"))
234 {
235 if (Peek() == null || Is("]")) throw new LogoError(token.Line, "missing )");
236 command.Prints.Add(ParsePrintItem(print));
237 }
238 position++;
239 return command;
240 }
241 if (token.Kind == "word" && (token.Text is "TO" or "END")) throw new LogoError(token.Line, "misplaced " + token.Text);
242 Primitive? primitive = token.Kind == "word" ? Primitives.GetValueOrDefault(token.Text) : null;
243 if (primitive != null && !primitive.Function && primitive.Name != "MOD")
244 {
245 if (!inside && (primitive.Name is "STOP" or "OUTPUT" or "LOCAL" or "LOCALMAKE")) throw new LogoError(token.Line, $"can only use {token.Text} inside a procedure");
246 Command command = new(token, primitive.Name);
247 foreach (char kind in primitive.Inputs) switch (kind)
248 {
249 case 'E':
250 command.Expressions.Add(ParseExpression(token));
251 break;
252 case 'L':
253 command.Blocks.Add(Block(token));
254 break;
255 case 'Q':
256 command.Quoted = Quoted(token);
257 break;
258 case 'P':
259 command.Prints.Add(ParsePrintItem(token));
260 break;
261 }
262 if ((primitive.Name is "TRACE" or "UNTRACE") && !procedures.ContainsKey(command.Quoted!)) throw new LogoError(tokens[position - 1].Line, "I don't know how to " + command.Quoted);
263 return command;
264 }
265 if (token.Kind == "word" && procedures.TryGetValue(token.Text, out Procedure? procedure))
266 {
267 Command command = new(token, "CALL");
268 for (int i = 0; i < procedure.Parameters.Count; i++) command.Expressions.Add(ParseExpression(token));
269 return command;
270 }
271 if (Unknown(token)) throw new LogoError(token.Line, "I don't know how to " + token.Text);
272 throw new LogoError(token.Line, "you don't say what to do with " + token.Text);
273 }
274
275 Expression ParseExpression(Token owner) => Comparison(owner);
276
277 static Expression Binary(Expression left, Token operation, Expression right)
278 {
279 Expression result = new(operation, "binary");
280 result.Arguments.Add(left);
281 result.Arguments.Add(right);
282 return result;
283 }
284
285 Expression Comparison(Token owner)
286 {
287 Expression result = Sum(owner);
288 while (Peek()?.Text is "=" or "<>" or "<" or ">" or "<=" or ">=")
289 {
290 Token operation = tokens[position++];
291 result = Binary(result, operation, Sum(operation));
292 }
293 return result;
294 }
295
296 Expression Sum(Token owner)
297 {
298 Expression result = Product(owner);
299 while (Is("+") || Is("-") && Peek()!.Kind == "minus")
300 {
301 Token operation = tokens[position++];
302 result = Binary(result, operation, Product(operation));
303 }
304 return result;
305 }
306
307 Expression Product(Token owner)
308 {
309 Expression result = Term(owner);
310 while (Peek()?.Text is "*" or "/" or "MOD")
311 {
312 Token operation = tokens[position++];
313 result = Binary(result, operation, Term(operation));
314 }
315 return result;
316 }
317
318 Expression Term(Token owner)
319 {
320 Token token = Input(owner);
321 position++;
322 if (token.Kind is "number" or "variable") return new(token, token.Kind);
323 if (token.Text == "-")
324 {
325 Expression result = new(token, "negate");
326 result.Arguments.Add(Term(token));
327 return result;
328 }
329 if (token.Text == "(")
330 {
331 Expression result = ParseExpression(token);
332 if (!Is(")")) throw new LogoError(token.Line, "missing )");
333 position++;
334 return result;
335 }
336 if (token.Kind == "word")
337 {
338 Primitive? primitive = Primitives.GetValueOrDefault(token.Text);
339 if (primitive?.Function == true || procedures.ContainsKey(token.Text))
340 {
341 Expression result = new(token, primitive != null ? "function" : "call");
342 int count = primitive != null ? primitive.Inputs.Length : procedures[token.Text].Parameters.Count;
343 for (int i = 0; i < count; i++) result.Arguments.Add(ParseExpression(token));
344 return result;
345 }
346 }
347 Reject(owner, token);
348 throw new InvalidOperationException();
349 }

374 long Variable(Token token)
375 {
376 string name = token.Text[1..];
377 for (int i = environments.Count - 1; i >= 0; i--) if (environments[i].Variables.TryGetValue(name, out long value)) return value;
378 Error(token, name + " has no value");
379 return 0;
380 }

382 void Make(string name, long value)
383 {
384 for (int i = environments.Count - 1; i > 0; i--) if (environments[i].Variables.ContainsKey(name))
385 {
386 environments[i].Variables[name] = value;
387 return;
388 }
389 environments[0].Variables[name] = value;

392 long Evaluate(Expression expression)
393 {
394 Token token = expression.Token;
395 switch (expression.Kind)
396 {
397 case "number":
398 return long.Parse(token.Text, CultureInfo.InvariantCulture);
399 case "variable":
400 return Variable(token);
401 case "negate":
402 return -Evaluate(expression.Arguments[0]);
403 case "binary":
404 {
405 long a = Evaluate(expression.Arguments[0]), b = Evaluate(expression.Arguments[1]);
406 if ((token.Text is "/" or "MOD") && b == 0) Error(token, token.Text + " doesn't like 0 as input");
407 return token.Text switch
408 {
409 "+" => a + b,
410 "-" => a - b,
411 "*" => a * b,
412 "/" => a / b,
413 "MOD" => a % b,
414 "=" => a == b ? 1 : 0,
415 "<>" => a != b ? 1 : 0,
416 "<" => a < b ? 1 : 0,
417 ">" => a > b ? 1 : 0,
418 "<=" => a <= b ? 1 : 0,
419 ">=" => a >= b ? 1 : 0,
420 _ => throw new InvalidOperationException()
421 };
422 }
423 default:
424 {
425 List<long> values = new();
426 foreach (Expression argument in expression.Arguments) values.Add(Evaluate(argument));
427 if (expression.Kind == "call")
428 {
429 long? value = Call(token, values);
430 if (!value.HasValue) Error(token, token.Text + " did not output");
431 return value.Value;
432 }
433 long a = values.Count == 0 ? 0 : values[0], b = values.Count < 2 ? 0 : values[1];
434 switch (token.Text)
435 {
436 case "XCOR":
437 return X;
438 case "YCOR":
439 return Y;
440 case "HEADING":
441 return Heading;
442 case "REPCOUNT":
443 if (repetitions.Count == 0) Error(token, "REPCOUNT outside REPEAT");
444 return repetitions[^1];
445 case "RANDOM":
446 if (a < 1) Error(token, $"RANDOM doesn't like {a} as input");
447 random = random * 48271 % 2147483647;
448 return random % a;
449 case "ABS":
450 return Math.Abs(a);
451 case "NOT":
452 return a == 0 ? 1 : 0;
453 case "AND":
454 return a != 0 && b != 0 ? 1 : 0;
455 case "OR":
456 return a != 0 || b != 0 ? 1 : 0;
457 case "MAX":
458 return Math.Max(a, b);
459 case "MIN":
460 return Math.Min(a, b);
461 default:
462 throw new InvalidOperationException();
463 }
464 }
465 }
466 }

468 long? Call(Token token, List<long> values)
469 {
470 int depth = environments.Count - 1;
471 if (depth == 100) Error(token, "recursion too deep");
472 Procedure procedure = procedures[token.Text];
473 bool trace = traced.Contains(token.Text);
474 string indent = new(' ', 2 * depth);
475 if (trace) Console.WriteLine(indent + "( " + token.Text + (values.Count == 0 ? "" : " " + string.Join(" ", values)) + " )");
476 Environment environment = new(procedure);
477 for (int i = 0; i < values.Count; i++) environment.Variables[procedure.Parameters[i]] = values[i];
478 environments.Add(environment);
479 long? result = null;
480 try
481 {
482 Execute(procedure.Commands);
483 }
484 catch (Flow flow)
485 {
486 result = flow.Value;
487 }
488 environments.RemoveAt(environments.Count - 1);
489 if (trace) Console.WriteLine(indent + token.Text + (result.HasValue ? $" outputs {result.Value}" : " stops"));
490 return result;
491 }

500 void Forward(long n)
501 {
502 int dx = Dx[Heading / 45], dy = Dy[Heading / 45];
503 if (n < 0)
504 {
505 dx = -dx;
506 dy = -dy;
507 n = -n;
508 }
509 Apply(X, Y);
510 for (long i = 0; i < n; i++)
511 {
512 X += dx;
513 Y += dy;
514 Apply(X, Y);
515 }

518 static long FloorDivide(long numerator, long denominator)
519 {
520 long value = numerator / denominator;
521 return numerator < 0 && numerator % denominator != 0 ? value - 1 : value;
522 }

524 void Move(int targetX, int targetY)
525 {
526 int oldX = X, oldY = Y, dx = targetX - X, dy = targetY - Y, steps = Math.Max(Math.Abs(dx), Math.Abs(dy));
527 if (steps == 0) Apply(X, Y);
528 else for (int i = 0; i <= steps; i++) Apply(oldX + (int)FloorDivide(2L * i * dx + steps, 2L * steps), oldY + (int)FloorDivide(2L * i * dy + steps, 2L * steps));
529 X = targetX;
530 Y = targetY;
531 }

533 void Fill()
534 {
535 if (canvas.Count == 0 || canvas.ContainsKey(new(X, Y))) return;
536 int x1 = canvas.Keys.Min(p => p.X), x2 = canvas.Keys.Max(p => p.X), y1 = canvas.Keys.Min(p => p.Y), y2 = canvas.Keys.Max(p => p.Y);
537 if (X < x1 || X > x2 || Y < y1 || Y > y2) return;
538 int[] dx = { -1, 1, 0, 0 }, dy = { 0, 0, -1, 1 };
539 Queue<Point> queue = new();
540 Point first = new(X, Y);
541 canvas[first] = PenCharacter;
542 queue.Enqueue(first);
543 while (queue.Count > 0)
544 {
545 Point point = queue.Dequeue();
546 for (int d = 0; d < 4; d++)
547 {
548 Point next = new(point.X + dx[d], point.Y + dy[d]);
549 if (next.X >= x1 && next.X <= x2 && next.Y >= y1 && next.Y <= y2 && !canvas.ContainsKey(next))
550 {
551 canvas[next] = PenCharacter;
552 queue.Enqueue(next);
553 }
554 }
555 }

579 foreach (Command command in commands) Execute(command);

581
582 void Execute(Command command)
583 {
584 Token token = command.Token;
585 string name = command.Name;
586 if (name == "WHILE")
587 { while (Evaluate(command.Expressions[0]) != 0) Execute(command.Blocks[0]);
588 return;
589 }
590 List<long> values = new();
591 foreach (Expression expression in command.Expressions) values.Add(Evaluate(expression));
592 long a = values.Count == 0 ? 0 : values[0], b = values.Count < 2 ? 0 : values[1];
593 Environment current = environments[^1];
594 switch (name)
595 {
596 case "FD":
597 Forward(a);
598 break;
599 case "BK":
600 Forward(-a);
601 break;
602 case "RT":
603 case "LT":
604 case "SETH":
605 if (a % 45 != 0) Error(token, $"{token.Text} doesn't like {a} as input");
606 long angle = name == "SETH" ? a : name == "RT" ? Heading + a : Heading - a;
607 Heading = (int)((angle % 360 + 360) % 360);
608 break;
609 case "PU":
610 Pen = "UP";
611 break;
612 case "PD":
613 Pen = "DOWN";
614 break;
615 case "PE":
616 Pen = "ERASE";
617 break;
618 case "SETPEN":
619 if (a < 33 || a > 126) Error(token, $"SETPEN doesn't like {a} as input");
620 PenCharacter = (char)a;
621 break;
622 case "SETXY":
623 Move((int)a, (int)b);
624 break;
625 case "SETX":
626 Move((int)a, Y);
627 break;
628 case "SETY":
629 Move(X, (int)a);
630 break;
631 case "HOME":
632 Move(0, 0);
633 Heading = 0;
634 break;
635 case "CS":
636 canvas.Clear();
637 X = Y = Heading = 0;
638 break;
639 case "CLEAN":
640 canvas.Clear();
641 break;
642 case "FILL":
643 Fill();
644 break;
645 case "LABEL":
646 {
647 string text = Item(command.Prints[0]);
648 int dx = Dx[Heading / 45], dy = Dy[Heading / 45];
649 for (int i = 0; i < text.Length; i++) canvas[new(X + i * dx, Y + i * dy)] = text[i];
650 break;
651 }
652 case "SNAPSHOT":
653 Picture();
654 break;
655 case "REPEAT":
656 if (command.Blocks[0].Count > 0) for (long i = 1; i <= a; i++)
657 {
658 repetitions.Add(i);
659 try
660 {
661 Execute(command.Blocks[0]);
662 }
663 finally
664 {
665 repetitions.RemoveAt(repetitions.Count - 1);
666 }
667 }
668 break;
669 case "FOR":
670 {
671 List<Command> body = command.Blocks[0];
672 if (body.Count == 0) Make(command.Quoted!, b);
673 else for (long i = a;; i += a <= b ? 1 : -1)
674 {
675 Make(command.Quoted!, i);
676 Execute(body);
677 if (i == b) break;
678 }
679 break;
680 }
681 case "IF":
682 if (a != 0) Execute(command.Blocks[0]);
683 break;
684 case "IFELSE":
685 Execute(command.Blocks[a != 0 ? 0 : 1]);
686 break;
687 case "TEST":
688 current.Test = a != 0;
689 break;
690 case "IFTRUE":
691 case "IFFALSE":
692 if (!current.Test.HasValue) Error(token, token.Text + " without TEST");
693 if (current.Test.Value == (name == "IFTRUE")) Execute(command.Blocks[0]);
694 break;
695 case "STOP":
696 throw new Flow(null);
697 case "OUTPUT":
698 throw new Flow(a);
699 case "MAKE":
700 Make(command.Quoted!, a);
701 break;
702 case "LOCAL":
703 current.Variables[command.Quoted!] = 0;
704 break;
705 case "LOCALMAKE":
706 current.Variables[command.Quoted!] = a;
707 break;
708 case "TRACE":
709 traced.Add(command.Quoted!);
710 break;
711 case "UNTRACE":
712 traced.Remove(command.Quoted!);
713 break;
714 case "SEED":
715 if (a < 1) Error(token, $"SEED doesn't like {a} as input");
716 random = a;
717 break;
718 case "PRINT":
719 case "MULTIPRINT":
720 {
721 List<string> items = new();
722 foreach (PrintItem item in command.Prints) items.Add(Item(item));
723 Console.WriteLine(string.Join(" ", items));
724 break;
725 }
726 case "CALL":
727 {
728 long? result = Call(token, values);
729 if (result.HasValue) Error(token, $"you don't say what to do with {result.Value}");
730 break;
731 }
732 }
733 }

736 static void Run(string name, List<string> source, int firstLine)
737 {
738 Console.WriteLine("PROGRAM " + name);
739 try
740 {
741 Dictionary<string, Procedure> procedures = new(StringComparer.Ordinal);
742 List<Procedure> defined = new();
743 List<Token> top = new();
744 Procedure? current = null;
745 for (int i = 0; i < source.Count; i++)
746 {
747 int line = firstLine + i;
748 List<Token> tokens = Lex(source[i], line);
749 if (tokens.Count == 0) continue;
750 Token first = tokens[0];
751 if (first.Kind == "word" && first.Text == "TO")
752 {
753 if (current != null) throw new LogoError(line, "TO inside definition of " + current.Token.Text);
754 if (tokens.Count < 2 || tokens[1].Kind != "word") throw new LogoError(line, "bad procedure header");
755 Token procedureName = tokens[1];
756 if (Primitives.ContainsKey(procedureName.Text)) throw new LogoError(line, procedureName.Text + " is a primitive");
757 if (procedures.ContainsKey(procedureName.Text)) throw new LogoError(line, procedureName.Text + " is already defined");
758 Procedure procedure = new(procedureName);
759 HashSet<string> seen = new(StringComparer.Ordinal);
760 foreach (Token input in tokens.Skip(2))
761 {
762 if (input.Kind != "variable") throw new LogoError(line, "bad procedure header");
763 if (!seen.Add(input.Text[1..])) throw new LogoError(line, "duplicate input " + input.Text);
764 procedure.Parameters.Add(input.Text[1..]);
765 }
766 procedures[procedureName.Text] = procedure;
767 defined.Add(procedure);
768 current = procedure;
769 }
770 else if (tokens.Count == 1 && first.Kind == "word" && first.Text == "END")
771 {
772 if (current == null) throw new LogoError(line, "END without TO");
773 current = null;
774 }
775 else if (current != null) current.Tokens.AddRange(tokens);
776 else top.AddRange(tokens);
777 }
778 if (current != null) throw new LogoError(current.Token.Line, "missing END for " + current.Token.Text);
779 foreach (Procedure procedure in defined) procedure.Commands = new Parser(procedure.Tokens, procedures, true).Sequence();
780 List<Command> commands = new Parser(top, procedures, false).Sequence();
781 Runner runner = new(procedures);
782 runner.Execute(commands);
783 Console.WriteLine($"TURTLE {runner.X} {runner.Y} {runner.Heading} {runner.Pen} {runner.PenCharacter}");
784 runner.Picture();
785 }
786 catch (LogoError error)
787 {
788 Console.WriteLine($"ERROR line {error.Line}: {error.Reason}");
789 for (int i = 0; i < error.Stack.Count;)
790 {
791 string procedure = error.Stack[i];
792 int end = i + 1;
793 while (end < error.Stack.Count && error.Stack[end] == procedure) end++;
794 Console.WriteLine(" in " + procedure + (end - i == 1 ? "" : " x" + (end - i)));
795 i = end;
796 }
797 }
798 }

800 static void Main()
801 {
802 CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
803 List<string> lines = new();
804 string? line;
805 while ((line = Console.ReadLine()) != null) lines.Add(line);
806 for (int i = 0; i < lines.Count && lines[i] != "#EOF";)
807 {
808 string name = lines[i][9..];
809 int first = ++i;
810 List<string> source = new();
811 while (lines[i] != "#EOF" && !lines[i].StartsWith("#PROGRAM ", StringComparison.Ordinal)) source.Add(lines[i++]);
812 Run(name, source, first + 1);
813 }
814 }
