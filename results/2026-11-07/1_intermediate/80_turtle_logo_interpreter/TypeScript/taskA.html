161function lex(source: string, line: number): Token[] {
162 const text = source.split(";")[0].toUpperCase();
163 const tokens: Token[] = [];
164 for (let i = 0; i < text.length;) {
165 if (text[i] === " ") {
166 i++;
167 continue;
168 }
169 const start = i,
170 first = text[i];
171 if ("[]()+-*/=<>".includes(first)) {
172 i++;
173 if (
174 (first === "<" && ["=", ">"].includes(text[i])) ||
175 (first === ">" && text[i] === "=")
176 )
177 i++;
178 tokens.push({
179 kind: "symbol",
180 text: text.slice(start, i),
181 line,
182 sign:
183 first === "-" &&
184 (start === 0 || " [(".includes(text[start - 1])) &&
185 i < text.length &&
186 text[i] !== " ",
187 });
188 } else {
189 while (i < text.length && !" []()+-*/=<>".includes(text[i])) i++;
190 const word = text.slice(start, i);
191 let kind: string;
192 if (/^[0-9]+$/.test(word)) kind = "number";
193 else if (/^:[A-Z][A-Z0-9_]*$/.test(word)) kind = "variable";
194 else if (/^"[A-Z][A-Z0-9_]*$/.test(word)) kind = "quote";
195 else if (/^[A-Z][A-Z0-9_]*$/.test(word)) kind = "word";
196 else throw new LogoError(line, `bad token '${word}'`);
197 tokens.push({ kind, text: word, line, sign: false });
198 }
199 }
200 return tokens;
201}
202function extract(lines: SourceLine[]): { procedures: Map<string, Procedure>; top: Token[] } {
203 const procedures = new Map<string, Procedure>();
204 const top: Token[] = [];
205 let current: Procedure | null = null;
206 for (const source of lines) {
207 const tokens = lex(source.text, source.line);
208 if (!tokens.length) continue;
209 const first = tokens[0];
210 if (first.kind === "word" && first.text === "TO") {
211 if (current !== null) fail(first, "TO inside definition of " + current.name);
212 if (tokens.length < 2 || tokens[1].kind !== "word") fail(first, "bad procedure header");
213 const name = tokens[1].text;
214 if (primitives.has(name)) fail(first, name + " is a primitive");
215 if (procedures.has(name)) fail(first, name + " is already defined");
216 const parameters: string[] = [];
217 for (const token of tokens.slice(2)) {
218 if (token.kind !== "variable") fail(first, "bad procedure header");
219 if (parameters.includes(token.text.slice(1)))
220 fail(first, "duplicate input " + token.text);
221 parameters.push(token.text.slice(1));
222 }
223 current = { name, parameters, token: first, tokens: [], body: [] };
224 procedures.set(name, current);
225 } else if (tokens.length === 1 && first.kind === "word" && first.text === "END") {
226 if (current === null) fail(first, "END without TO");
227 current = null;
228 } else (current === null ? top : current.tokens).push(...tokens);
229 }
230 if (current !== null) fail(current.token, "missing END for " + current.name);
231 return { procedures, top };
232}
233class Parser {
234 tokens: Token[];
235 procedures: Map<string, Procedure>;
236 inside: boolean;
237 at: number;
238 constructor(tokens: Token[], procedures: Map<string, Procedure>, inside: boolean) {
239 this.tokens = tokens;
240 this.procedures = procedures;
241 this.inside = inside;
242 this.at = 0;
243 }
244 peek(): Token | undefined {
245 return this.tokens[this.at];
246 }
247 unknown(token: Token): boolean {
248 return (
249 token.kind === "word" && !primitives.has(token.text) && !this.procedures.has(token.text)
250 );
251 }
252 inputToken(owner: Token): Token {
253 const token = this.peek();
254 if (token === undefined || token.text === "]" || token.text === ")")
255 fail(owner, "not enough inputs to " + owner.text);
256 if (this.unknown(token)) fail(token, "I don't know how to " + token.text);
257 return token;
258 }
259 input(kind: string, owner: Token): Input {
260 const token = this.inputToken(owner);
261 if (kind === "L") {
262 if (token.text !== "[")
263 fail(token, `${owner.text} doesn't like ${token.text} as input`);
264 this.at++;
265 const commands: Command[] = [];
266 while (this.peek() !== undefined && this.peek()!.text !== "]")
267 commands.push(this.command());
268 if (this.peek() === undefined) fail(token, "missing ]");
269 this.at++;
270 return { kind: "list", commands };
271 }
272 if (kind === "Q" || (kind === "P" && token.kind === "quote")) {
273 if (token.kind !== "quote")
274 fail(token, `${owner.text} doesn't like ${token.text} as input`);
275 this.at++;
276 return { kind: "quote", value: token.text.slice(1) };
277 }
278 return this.expression(owner);
279 }
280 expression(owner: Token): Expression {
281 return this.binary(owner, 0);
282 }
283 binary(owner: Token, level: number): Expression {
284 if (level === 3) return this.term(owner);
285 let left = this.binary(owner, level + 1);
286 const operators =
287 level === 0
288 ? ["=", "<>", "<", ">", "<=", ">="]
289 : level === 1
290 ? ["+", "-"]
291 : ["*", "/", "MOD"];
292 while (
293 this.peek() !== undefined &&
294 operators.includes(this.peek()!.text) &&
295 !this.peek()!.sign
296 ) {
297 const token = this.tokens[this.at++],
298 right = this.binary(token, level + 1);
299 left = { kind: "binary", token, left, right };
300 }
301 return left;
302 }
303 term(owner: Token): Expression {
304 const token = this.inputToken(owner);
305 this.at++;
306 if (token.kind === "number") return { kind: "number", value: Number(token.text), token };
307 if (token.kind === "variable")
308 return { kind: "variable", name: token.text.slice(1), token };
309 if (token.text === "-") return { kind: "unary", value: this.term(token), token };
310 if (token.text === "(") {
311 const value = this.expression(token);
312 if (this.peek() === undefined || this.peek()!.text !== ")") fail(token, "missing )");
313 this.at++;
314 return value;
315 }
316 if (
317 token.kind === "word" &&
318 (functionInputs.has(token.text) || this.procedures.has(token.text))
319 ) {
320 const count =
321 functionInputs.get(token.text) ??
322 this.procedures.get(token.text)!.parameters.length;
323 const inputs = Array.from({ length: count }, () => this.expression(token));
324 if (functionInputs.has(token.text)) return { kind: "function", token, inputs };
325 return { kind: "call", token, inputs };
326 }
327 fail(token, `${owner.text} doesn't like ${token.text} as input`);
328 }
329 command(): Command {
330 const token = this.tokens[this.at++];
331 if (token.text === "]") fail(token, "unexpected ]");
332 const next = this.peek();
333 if (
334 token.text === "(" &&
335 next !== undefined &&
336 next.kind === "word" &&
337 ["PRINT", "PR"].includes(next.text)
338 ) {
339 this.at++;
340 const inputs: PrintItem[] = [];
341 while (this.peek() === undefined || this.peek()!.text !== ")") {
342 if (this.peek() === undefined || this.peek()!.text === "]")
343 fail(token, "missing )");
344 inputs.push(this.input("P", next) as PrintItem);
345 }
346 this.at++;
347 return { token: next, name: "MULTIPRINT", inputs };
348 }
349 if (token.kind === "word" && ["TO", "END"].includes(token.text))
350 fail(token, "misplaced " + token.text);
351 if (token.kind === "word" && commandInputs.has(token.text)) {
352 if (!this.inside && ["STOP", "OUTPUT", "OP", "LOCAL", "LOCALMAKE"].includes(token.text))
353 fail(token, "can only use " + token.text + " inside a procedure");
354 const inputs = [...commandInputs.get(token.text)!].map((kind) =>
355 this.input(kind, token),
356 );
357 if (
358 ["TRACE", "UNTRACE"].includes(token.text) &&
359 !this.procedures.has((inputs[0] as QuotedWord).value)
360 )
361 fail(
362 this.tokens[this.at - 1],
363 "I don't know how to " + (inputs[0] as QuotedWord).value,
364 );
365 return { token, name: aliases.get(token.text) ?? token.text, inputs };
366 }
367 if (token.kind === "word" && this.procedures.has(token.text)) {
368 return {
369 token,
370 name: "CALL",
371 inputs: Array.from(
372 { length: this.procedures.get(token.text)!.parameters.length },
373 () => this.expression(token),
374 ),
375 };
376 }
377 if (this.unknown(token)) fail(token, "I don't know how to " + token.text);
378 fail(token, "you don't say what to do with " + token.text);
379 }
380 sequence(): Command[] {
381 const commands: Command[] = [];
382 while (this.peek() !== undefined) commands.push(this.command());
383 return commands;
384 }
385}
386class Turtle {
387 procedures: Map<string, Procedure>;
388 output: string[];
389 x: number;
390 y: number;
391 heading: number;
392 pen: string;
393 character: string;
394 canvas: Map<string, string>;
395 environments: Environment[];
396 active: string[];
397 repetitions: number[];
398 traced: Set<string>;
399 random: number;
400 constructor(procedures: Map<string, Procedure>, output: string[]) {
401 this.procedures = procedures;
402 this.output = output;
403 this.x = 0;
404 this.y = 0;
405 this.heading = 0;
406 this.pen = "DOWN";
407 this.character = "*";
408 this.canvas = new Map();
409 this.environments = [{ variables: new Map(), test: null }];
410 this.active = [];
411 this.repetitions = [];
412 this.traced = new Set();
413 this.random = 1;
414 }
415 current(): Environment {
416 return this.environments.at(-1)!;
417 }
418 make(name: string, value: number): void {
419 for (let i = this.environments.length - 1; i > 0; i--)
420 if (this.environments[i].variables.has(name)) {
421 this.environments[i].variables.set(name, value);
422 return;
423 }
424 this.environments[0].variables.set(name, value);
425 }
426 key(x: number, y: number): string {
427 return x + "," + y;
428 }
429 apply(x: number, y: number): void {
430 const key = this.key(x, y);
431 if (this.pen === "DOWN") this.canvas.set(key, this.character);
432 else if (this.pen === "ERASE") this.canvas.delete(key);
433 }
434 move(x: number, y: number): void {
435 const x0 = this.x,
436 y0 = this.y,
437 dx = x - x0,
438 dy = y - y0,
439 steps = Math.max(Math.abs(dx), Math.abs(dy));
440 if (!steps) this.apply(x0, y0);
441 else
442 for (let i = 0; i <= steps; i++)
443 this.apply(
444 x0 + Math.floor((2 * i * dx + steps) / (2 * steps)),
445 y0 + Math.floor((2 * i * dy + steps) / (2 * steps)),
446 );
447 this.x = x;
448 this.y = y;
449 }
450 extent(): Extent {
451 let x1 = Infinity,
452 y1 = Infinity,
453 x2 = -Infinity,
454 y2 = -Infinity;
455 for (const key of this.canvas.keys()) {
456 const [x, y] = key.split(",").map(Number);
457 x1 = Math.min(x1, x);
458 y1 = Math.min(y1, y);
459 x2 = Math.max(x2, x);
460 y2 = Math.max(y2, y);
461 }
462 return { x1, y1, x2, y2 };
463 }
464 picture(): void {
465 if (!this.canvas.size) {
466 this.output.push("CANVAS 0 0 0");
467 return;
468 }
469 const { x1, y1, x2, y2 } = this.extent();
470 this.output.push(`CANVAS ${x2 - x1 + 1} ${y2 - y1 + 1} ${this.canvas.size}`);
471 for (let y = y2; y >= y1; y--) {
472 let row = "";
473 for (let x = x1; x <= x2; x++) row += this.canvas.get(this.key(x, y)) ?? " ";
474 this.output.push(row.replace(/ +$/, ""));
475 }
476 }
477 fill(): void {
478 if (!this.canvas.size || this.canvas.has(this.key(this.x, this.y))) return;
479 const { x1, y1, x2, y2 } = this.extent();
480 if (this.x < x1 || this.x > x2 || this.y < y1 || this.y > y2) return;
481 const queue = [[this.x, this.y]];
482 this.canvas.set(this.key(this.x, this.y), this.character);
483 for (let i = 0; i < queue.length; i++) {
484 const [x, y] = queue[i];
485 for (const [a, b] of [
486 [x - 1, y],
487 [x + 1, y],
488 [x, y - 1],
489 [x, y + 1],
490 ])
491 if (a >= x1 && a <= x2 && b >= y1 && b <= y2 && !this.canvas.has(this.key(a, b))) {
492 this.canvas.set(this.key(a, b), this.character);
493 queue.push([a, b]);
494 }
495 }
496 }
497 call(token: Token, inputs: Expression[]): number | null {
498 const values = inputs.map((input) => this.evaluate(input));
499 if (this.active.length >= 100) fail(token, "recursion too deep");
500 const traced = this.traced.has(token.text),
501 indent = " ".repeat(2 * this.active.length),
502 procedure = this.procedures.get(token.text)!;
503 if (traced)
504 this.output.push(
505 indent + "( " + token.text + (values.length ? " " + values.join(" ") : "") + " )",
506 );
507 this.active.push(token.text);
508 this.environments.push({
509 variables: new Map(procedure.parameters.map((name, i) => [name, values[i]])),
510 test: null,
511 });
512 const result = this.sequence(procedure.body);
513 this.environments.pop();
514 this.active.pop();
515 const value = result === null ? null : result.value;
516 if (traced)
517 this.output.push(
518 indent + token.text + (value === null ? " stops" : " outputs " + value),
519 );
520 return value;
521 }
522 evaluate(expression: Expression): number {
523 const token = expression.token;
524 if (expression.kind === "number") return expression.value;
525 if (expression.kind === "variable") {
526 for (let i = this.environments.length - 1; i >= 0; i--)
527 if (this.environments[i].variables.has(expression.name))
528 return this.environments[i].variables.get(expression.name)!;
529 fail(token, expression.name + " has no value");
530 }
531 if (expression.kind === "unary") return -this.evaluate(expression.value);
532 if (expression.kind === "binary") {
533 const a = this.evaluate(expression.left),
534 b = this.evaluate(expression.right);
535 if ((token.text === "/" || token.text === "MOD") && b === 0)
536 fail(token, token.text + " doesn't like 0 as input");
537 switch (token.text) {
538 case "+":
539 return a + b;
540 case "-":
541 return a - b;
542 case "*":
543 return a * b;
544 case "/":
545 return Math.trunc(a / b);
546 case "MOD":
547 return a % b;
548 case "=":
549 return Number(a === b);
550 case "<>":
551 return Number(a !== b);
552 case "<":
553 return Number(a < b);
554 case ">":
555 return Number(a > b);
556 case "<=":
557 return Number(a <= b);
558 default:
559 return Number(a >= b);
560 }
561 }
562 if (expression.kind === "call") {
563 const value = this.call(token, expression.inputs);
564 if (value === null) fail(token, token.text + " did not output");
565 return value;
566 }
567 const values = expression.inputs.map((input) => this.evaluate(input)),
568 a = values[0],
569 b = values[1];
570 switch (token.text) {
571 case "XCOR":
572 return this.x;
573 case "YCOR":
574 return this.y;
575 case "HEADING":
576 return this.heading;
577 case "REPCOUNT":
578 if (!this.repetitions.length) fail(token, "REPCOUNT outside REPEAT");
579 return this.repetitions.at(-1)!;
580 case "RANDOM":
581 if (a < 1) fail(token, `RANDOM doesn't like ${a} as input`);
582 this.random = (48271 * this.random) % 2147483647;
583 return this.random % a;
584 case "ABS":
585 return Math.abs(a);
586 case "NOT":
587 return Number(a === 0);
588 case "AND":
589 return Number(a !== 0 && b !== 0);
590 case "OR":
591 return Number(a !== 0 || b !== 0);
592 case "MAX":
593 return Math.max(a, b);
594 default:
595 return Math.min(a, b);
596 }
597 }
598 item(input: PrintItem): string {
599 return input.kind === "quote" ? input.value : String(this.evaluate(input));
600 }
601 sequence(commands: Command[]): Flow | null {
602 for (const command of commands) {
603 const result = this.command(command);
604 if (result !== null) return result;
605 }
606 return null;
607 }
608 command(command: Command): Flow | null {
609 const { name, token, inputs } = command;
610 if (name === "CALL") {
611 const value = this.call(token, inputs as Expression[]);
612 if (value !== null) fail(token, "you don't say what to do with " + value);
613 return null;
614 }
615 if (name === "PRINT" || name === "MULTIPRINT") {
616 const values = inputs.map((input) => this.item(input as PrintItem));
617 this.output.push(values.join(" "));
618 return null;
619 }
620 if (name === "LABEL") {
621 const text = this.item(inputs[0] as PrintItem),
622 [dx, dy] = directions[this.heading / 45];
623 for (let i = 0; i < text.length; i++)
624 this.canvas.set(this.key(this.x + i * dx, this.y + i * dy), text[i]);
625 return null;
626 }
627 if (name === "WHILE") {
628 while (this.evaluate(inputs[0] as Expression) !== 0) {
629 const result = this.sequence((inputs[1] as CommandList).commands);
630 if (result !== null) return result;
631 }
632 return null;
633 }
634 const values = inputs.map((input) =>
635 input.kind === "quote"
636 ? input.value
637 : input.kind === "list"
638 ? input.commands
639 : this.evaluate(input),
640 );
641 const a = values[0] as number,
642 b = values[1] as number;
643 switch (name) {
644 case "FD":
645 case "BK": {
646 const amount = name === "BK" ? -a : a,
647 [dx, dy] = directions[this.heading / 45],
648 sign = amount < 0 ? -1 : 1;
649 this.apply(this.x, this.y);
650 for (let i = 0; i < Math.abs(amount); i++) {
651 this.x += sign * dx;
652 this.y += sign * dy;
653 this.apply(this.x, this.y);
654 }
655 break;
656 }
657 case "RT":
658 case "LT":
659 case "SETH":
660 if (a % 45 !== 0) fail(token, `${token.text} doesn't like ${a} as input`);
661 this.heading =
662 (((name === "SETH" ? a : this.heading + (name === "LT" ? -a : a)) % 360) +
663 360) %
664 360;
665 break;
666 case "PU":
667 this.pen = "UP";
668 break;
669 case "PD":
670 this.pen = "DOWN";
671 break;
672 case "PE":
673 this.pen = "ERASE";
674 break;
675 case "SETPEN":
676 if (a < 33 || a > 126) fail(token, `SETPEN doesn't like ${a} as input`);
677 this.character = String.fromCharCode(a);
678 break;
679 case "SETXY":
680 this.move(a, b);
681 break;
682 case "SETX":
683 this.move(a, this.y);
684 break;
685 case "SETY":
686 this.move(this.x, a);
687 break;
688 case "HOME":
689 this.move(0, 0);
690 this.heading = 0;
691 break;
692 case "CS":
693 this.canvas.clear();
694 this.x = 0;
695 this.y = 0;
696 this.heading = 0;
697 break;
698 case "CLEAN":
699 this.canvas.clear();
700 break;
701 case "FILL":
702 this.fill();
703 break;
704 case "SNAPSHOT":
705 this.picture();
706 break;
707 case "MAKE":
708 this.make(values[0] as string, b);
709 break;
710 case "LOCAL":
711 this.current().variables.set(values[0] as string, 0);
712 break;
713 case "LOCALMAKE":
714 this.current().variables.set(values[0] as string, b);
715 break;
716 case "TRACE":
717 this.traced.add(values[0] as string);
718 break;
719 case "UNTRACE":
720 this.traced.delete(values[0] as string);
721 break;
722 case "SEED":
723 if (a < 1) fail(token, `SEED doesn't like ${a} as input`);
724 this.random = a;
725 break;
726 case "STOP":
727 return { value: null };
728 case "OUTPUT":
729 return { value: a };
730 case "REPEAT":
731 if ((values[1] as Command[]).length === 0) break;
732 for (let k = 1; k <= a; k++) {
733 this.repetitions.push(k);
734 let result: Flow | null = null;
735 try {
736 result = this.sequence(values[1] as Command[]);
737 } finally {
738 this.repetitions.pop();
739 }
740 if (result !== null) return result;
741 }
742 break;
743 case "FOR": {
744 const finish = values[2] as number,
745 step = b <= finish ? 1 : -1;
746 if ((values[3] as Command[]).length === 0) {
747 this.make(values[0] as string, finish);
748 break;
749 }
750 for (let value = b; ; value += step) {
751 this.make(values[0] as string, value);
752 const result = this.sequence(values[3] as Command[]);
753 if (result !== null) return result;
754 if (value === finish) break;
755 }
756 break;
757 }
758 case "IF":
759 if (a !== 0) return this.sequence(values[1] as Command[]);
760 break;
761 case "IFELSE":
762 return this.sequence((a !== 0 ? values[1] : values[2]) as Command[]);
763 case "TEST":
764 this.current().test = a !== 0;
765 break;
766 case "IFTRUE":
767 case "IFFALSE":
768 if (this.current().test === null) fail(token, token.text + " without TEST");
769 if (this.current().test === (name === "IFTRUE"))
770 return this.sequence(values[0] as Command[]);
771 break;
772 }
773 return null;
774 }
775}
776function run(name: string, lines: SourceLine[], output: string[]): void {
777 output.push("PROGRAM " + name);
778 let turtle: Turtle | null = null;
779 try {
780 const { procedures, top } = extract(lines);
781 for (const procedure of procedures.values())
782 procedure.body = new Parser(procedure.tokens, procedures, true).sequence();
783 const commands = new Parser(top, procedures, false).sequence();
784 turtle = new Turtle(procedures, output);
785 turtle.sequence(commands);
786 output.push(
787 `TURTLE ${turtle.x} ${turtle.y} ${turtle.heading} ${turtle.pen} ${turtle.character}`,
788 );
789 turtle.picture();
790 } catch (error) {
791 if (!(error instanceof LogoError)) throw error;
792 output.push(`ERROR line ${error.line}: ${error.message}`);
793 if (turtle !== null)
794 for (let i = turtle.active.length - 1; i >= 0;) {
795 const name = turtle.active[i];
796 let count = 0;
797 while (i >= 0 && turtle.active[i] === name) {
798 i--;
799 count++;
800 }
801 output.push(" in " + name + (count > 1 ? " x" + count : ""));
802 }
803 }
804}
805const input = fs.readFileSync(0, "utf8").split(/\r?\n/);
806const output: string[] = [];
807let name = "";
808let source: SourceLine[] = [];
809for (let i = 0; i < input.length; i++) {
810 if (input[i] === "#EOF" || input[i].startsWith("#PROGRAM ")) {
811 if (name !== "") run(name, source, output);
812 if (input[i] === "#EOF") break;
813 name = input[i].slice(9);
814 source = [];
815 } else source.push({ text: input[i], line: i + 1 });
816}
817process.stdout.write(output.join("\n") + "\n");
