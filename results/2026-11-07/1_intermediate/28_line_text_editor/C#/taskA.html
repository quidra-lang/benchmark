35static class Parser
36{
37 public static bool Digit(char c) => c is >= '0' and <= '9';
38 static bool Digits(string text) => text.Length > 0 && text.All(Digit);
39 static int Number(string text) => int.Parse(text, CultureInfo.InvariantCulture);
40 static bool PrintSuffix(char c) => c is 'p' or 'n' or 'l';
41 static bool SortForm(string text) => text == "sort" || text.StartsWith("sort ", StringComparison.Ordinal);
42
43 public static Delimited ReadDelimited(string text, ref int position, char delimiter)
44 {
45 int start = position;
46 var value = new StringBuilder();
47 var replacement = new List<char?>();
48 while (position < text.Length)
49 {
50 char c = text[position++];
51 if (c == delimiter)
52 return new(value.ToString(), replacement.ToArray(), text[start..(position - 1)]);
53 bool escaped = c == '\\';
54 if (escaped)
55 {
56 if (position == text.Length) break;
57 c = text[position++];
58 }
59 value.Append(c);
60 replacement.Add(c == '&' && !escaped ? null : c);
61 }
62 throw new CommandError("missing delimiter");
63 }
64
65 public static List<Element> Scan(string text, ref int position)
66 {
67 var elements = new List<Element>();
68 while (position < text.Length)
69 {
70 char c = text[position];
71 if (Digit(c))
72 {
73 int start = position++;
74 while (position < text.Length && Digit(text[position])) position++;
75 elements.Add(new(ElementKind.Number, Number(text[start..position])));
76 }
77 else if (c is '+' or '-')
78 {
79 int start = ++position;
80 while (position < text.Length && Digit(text[position])) position++;
81 int value = position == start ? 1 : Number(text[start..position]);
82 elements.Add(new(ElementKind.Offset, c == '+' ? value : -value));
83 }
84 else if (c == '\'')
85 {
86 position++;
87 elements.Add(new(ElementKind.Mark, Symbol: position < text.Length ? text[position++] : '\0'));
88 }
89 else if (c is '/' or '?')
90 {
91 position++;
92 string pattern = ReadDelimited(text, ref position, c).Text;
93 elements.Add(new(c == '/' ? ElementKind.Forward : ElementKind.Backward, Text: pattern));
94 }
95 else
96 {
97 ElementKind? kind = c switch
98 {
99 '.' => ElementKind.Current, '$' => ElementKind.Last,
100 ',' or ';' => ElementKind.Separator, '%' => ElementKind.All, _ => null
101 };
102 if (kind is null) break;
103 elements.Add(new(kind.Value, Symbol: c));
104 position++;
105 }
106 }
107 return elements;
108 }

110 static bool Expression(IReadOnlyList<Element> elements)
111 {
112 for (int i = 0; i < elements.Count; i++)
113 {
114 ElementKind kind = elements[i].Kind;
115 if (kind == ElementKind.Offset) continue;
116 if (i != 0 || kind is ElementKind.Separator or ElementKind.All) return false;
117 }
118 return true;
119 }

121 public static Address Form(List<Element> elements)
122 {
123 if (elements.Count == 1 && elements[0].Kind == ElementKind.All)
124 return new([new(ElementKind.Number, 1)], [new(ElementKind.Last)], ',');
125 int[] separators = Enumerable.Range(0, elements.Count)
126 .Where(i => elements[i].Kind == ElementKind.Separator).ToArray();
127 if (separators.Length > 1) throw new CommandError("bad address");
128 if (separators.Length == 0)
129 {
130 if (!Expression(elements)) throw new CommandError("bad address");
131 return new(elements.ToArray());
132 }
133 int split = separators[0];
134 Element[] left = elements.Take(split).ToArray(), right = elements.Skip(split + 1).ToArray();
135 if (!Expression(left) || !Expression(right)) throw new CommandError("bad address");
136 return new(left, right, elements[split].Symbol);
137 }

139 static int OptionalCount(string argument)
140 {
141 if (argument.Length == 0) return 0;
142 if (!Digits(argument) || Number(argument) == 0) throw new CommandError("invalid suffix");
143 return Number(argument);
144 }
145
146 static char Suffix(string argument)
147 {
148 if (argument.Length == 0) return '\0';
149 if (argument.Length != 1 || !PrintSuffix(argument[0])) throw new CommandError("invalid suffix");
150 return argument[0];
151 }

153 static string FileName(string argument)
154 {
155 if (argument.Length is < 2 or > 9 || argument[0] != ' ' ||
156 !argument[1..].All(c => c is >= 'a' and <= 'z' || Digit(c)))
157 throw new CommandError("invalid file name");
158 return argument[1..];
159 }
160
161 static Command Substitution(string argument)
162 {
163 var command = new Command(CommandKind.Substitute) { Count = 1 };
164 string flags = argument;
165 if (!argument.All(c => c is 'g' or 'p' or 'n' or 'l' || Digit(c)))
166 {
167 char delimiter = argument[0];
168 if (delimiter is ' ' or '\\') throw new CommandError("missing delimiter");
169 int position = 1;
170 command.Pattern = ReadDelimited(argument, ref position, delimiter).Text;
171 Delimited replacement = ReadDelimited(argument, ref position, delimiter);
172 if (replacement.Raw != "%") command.Replacement = replacement.Replacement;
173 flags = argument[position..];
174 }
175 bool hasCount = false;
176 for (int position = 0; position < flags.Length;)
177 {
178 char c = flags[position++];
179 if (c == 'g' && !command.All) command.All = true;
180 else if (PrintSuffix(c) && command.Suffix == '\0') command.Suffix = c;
181 else if (Digit(c) && !hasCount)
182 {
183 int start = position - 1;
184 while (position < flags.Length && Digit(flags[position])) position++;
185 command.Count = Number(flags[start..position]);
186 if (command.Count == 0) throw new CommandError("invalid suffix");
187 hasCount = true;
188 }
189 else throw new CommandError("invalid suffix");
190 }
191 return command;
192 }

194 public static Command Parse(string text)
195 {
196 if (SortForm(text))
197 {
198 string options = text.Length == 4 ? "" : text[5..];
199 if (text.Length > 4 && options.Length == 0 || options.Any(c => !"ruin".Contains(c)) ||
200 options.Distinct().Count() != options.Length || options.Contains('i') && options.Contains('n'))
201 throw new CommandError("invalid suffix");
202 return new(CommandKind.Sort) { Options = options };
203 }
204 char first = text.Length == 0 ? '\0' : text[0];
205 string argument = text.Length == 0 ? "" : text[1..];
206 CommandKind? plain = first switch
207 {
208 '\0' => CommandKind.Empty, 'a' => CommandKind.Append, 'i' => CommandKind.Insert,
209 'c' => CommandKind.Change, 'p' => CommandKind.Print, 'n' => CommandKind.Numbered,
210 'l' => CommandKind.Listed, '=' => CommandKind.Number, '#' => CommandKind.Count,
211 'y' => CommandKind.Yank, 'K' => CommandKind.Marks, 'h' => CommandKind.History,
212 'f' => CommandKind.Files, _ => null
213 };
214 if (plain is not null)
215 {
216 if (argument.Length != 0) throw new CommandError("invalid suffix");
217 return new(plain.Value);
218 }
219 switch (first)
220 {
221 case 'd': case 'x': case 'j': case 'J':
222 return new(first switch
223 {
224 'd' => CommandKind.Delete, 'x' => CommandKind.Put, 'j' => CommandKind.Join,
225 _ => CommandKind.JoinSpaced
226 }) { Suffix = Suffix(argument) };
227 case 'z': case '>': case '<': case 'u':
228 return new(first switch
229 {
230 'z' => CommandKind.Window, '>' => CommandKind.Indent, '<' => CommandKind.Unindent,
231 _ => CommandKind.Undo
232 }) { Count = OptionalCount(argument) };
233 case '~':
234 if (argument != "" && argument != "u" && argument != "l") throw new CommandError("invalid suffix");
235 return new(CommandKind.Case) { CaseMode = argument.Length == 0 ? '\0' : argument[0] };
236 case 'r':
237 return argument.Length == 0 || Digits(argument)
238 ? new(CommandKind.Redo) { Count = OptionalCount(argument) }
239 : new(CommandKind.Read) { Name = FileName(argument) };
240 case 'w':
241 return argument.Length == 0 ? new(CommandKind.WritePrint)
242 : new(CommandKind.Write) { Name = FileName(argument) };
243 case 'W': case 'e':
244 return new(first == 'W' ? CommandKind.WriteAppend : CommandKind.Edit) { Name = FileName(argument) };
245 case 'k':
246 if (argument.Length != 1 || argument[0] is < 'a' or > 'z') throw new CommandError("invalid mark");
247 return new(CommandKind.Mark) { Name = argument };
248 case 'm': case 't':
249 int position = 0;
250 List<Element> destination = Scan(argument, ref position);
251 if (destination.Count == 0 || !Expression(destination)) throw new CommandError("invalid destination");
252 return new(first == 'm' ? CommandKind.Move : CommandKind.Copy)
253 {
254 Destination = destination.ToArray(), Suffix = Suffix(argument[position..])
255 };
256 case 's': return Substitution(argument);
257 case 'g': case 'v':
258 if (argument.Length == 0 || argument[0] is ' ' or '\\') throw new CommandError("missing delimiter");
259 int offset = 1;
260 string pattern = ReadDelimited(argument, ref offset, argument[0]).Text;
261 string subcommand = offset == argument.Length ? "p" : argument[offset..];
262 if (!"pnldmtjJs><~aicx".Contains(subcommand[0]) || SortForm(subcommand))
263 throw new CommandError("invalid global command");
264 return new(first == 'g' ? CommandKind.Global : CommandKind.InverseGlobal)
265 {
266 Pattern = pattern, Subcommand = Parse(subcommand)
267 };
268 default: throw new CommandError("unknown command");
269 }
270 }
271
272 public static bool HasText(string line)
273 {
274 try
275 {
276 int position = 0;
277 Scan(line, ref position);
278 string command = line[position..];
279 if (command is "a" or "i" or "c") return true;
280 if (command.Length < 2 || command[0] is not ('g' or 'v') || command[1] is ' ' or '\\') return false;
281 position = 2;
282 ReadDelimited(command, ref position, command[1]);
283 return command[position..] is "a" or "i" or "c";
284 }
285 catch (CommandError) { return false; }
286 }
287}

315 static bool NoAddress(CommandKind kind) => kind is CommandKind.Undo or CommandKind.Redo or
316 CommandKind.History or CommandKind.Marks or CommandKind.Files or CommandKind.Edit;
317 static bool OneAddress(CommandKind kind) => kind is CommandKind.Append or CommandKind.Insert or
318 CommandKind.Put or CommandKind.Number or CommandKind.Read or CommandKind.Mark or CommandKind.Window;
319 static bool Modifies(Command command) => command.Kind switch
320 {
321 CommandKind.Append or CommandKind.Insert or CommandKind.Change or CommandKind.Delete or
322 CommandKind.Put or CommandKind.Join or CommandKind.JoinSpaced or CommandKind.Move or CommandKind.Copy or
323 CommandKind.Substitute or CommandKind.Indent or CommandKind.Unindent or CommandKind.Case or
324 CommandKind.Sort or CommandKind.Read or CommandKind.Edit => true,
325 CommandKind.Global or CommandKind.InverseGlobal => Modifies(command.Subcommand!),
326 _ => false
327 };

329 string Pattern(string search)
330 {
331 if (search.Length != 0) state.Pattern = search;
332 return state.Pattern ?? throw new CommandError("no previous pattern");
333 }
334
335 int Evaluate(Element[] elements)
336 {
337 int value = state.Current;
338 foreach (Element element in elements)
339 {
340 switch (element.Kind)
341 {
342 case ElementKind.Offset: value += element.Value; break;
343 case ElementKind.Number: value = element.Value; break;
344 case ElementKind.Current: value = state.Current; break;
345 case ElementKind.Last: value = state.Lines.Count; break;
346 case ElementKind.Mark:
347 if (element.Symbol is < 'a' or > 'z') throw new CommandError("invalid mark");
348 long? id = state.Marks[element.Symbol - 'a'];
349 value = id is null ? 0 : state.Lines.FindIndex(line => line.Id == id.Value) + 1;
350 if (value == 0) throw new CommandError("mark not set");
351 break;
352 case ElementKind.Forward: case ElementKind.Backward:
353 string pattern = Pattern(element.Text);
354 int size = state.Lines.Count;
355 bool forward = element.Kind == ElementKind.Forward;
356 int index = forward ? state.Current : state.Current - 2;
357 value = 0;
358 for (int checkedLines = 0; checkedLines < size; checkedLines++)
359 {
360 if (index >= size) index = 0;
361 if (index < 0) index = size - 1;
362 if (state.Lines[index].Text.Contains(pattern, StringComparison.Ordinal))
363 {
364 value = index + 1;
365 break;
366 }
367 index += forward ? 1 : -1;
368 }
369 if (value == 0) throw new CommandError("no match");
370 break;
371 }
372 }
373 return value;
374 }
375
376 int[] Evaluate(Address address)
377 {
378 if (address.Count == 0) return [];
379 if (address.Count == 1) return [Evaluate(address.Left)];
380 int left = address.Left.Length == 0 ? address.Separator == ',' ? 1 : state.Current : Evaluate(address.Left);
381 if (address.Separator == ';')
382 {
383 if (left < 0 || left > state.Lines.Count) throw new CommandError("address out of range");
384 state.Current = left;
385 }
386 int right = address.Right!.Length == 0 ? state.Lines.Count : Evaluate(address.Right);
387 return [left, right];
388 }
389
390 (int First, int Last) Range(Command command, int[] addresses)
391 {
392 CommandKind kind = command.Kind;
393 int size = state.Lines.Count, a, b;
394 if (NoAddress(kind)) return (0, 0);
395 if (OneAddress(kind))
396 {
397 a = addresses.Length != 0 ? addresses[0] : kind switch
398 {
399 CommandKind.Number or CommandKind.Read => size,
400 CommandKind.Window => state.Current + 1, _ => state.Current
401 };
402 int minimum = kind is CommandKind.Mark or CommandKind.Window ? 1 : 0;
403 if (a < minimum || a > size) throw new CommandError("address out of range");
404 return (a, a);
405 }
406 if (addresses.Length != 0)
407 {
408 a = addresses[0];
409 b = addresses[^1];
410 }
411 else if (kind is CommandKind.WritePrint or CommandKind.Write or CommandKind.WriteAppend or CommandKind.Count)
412 return (1, size);
413 else if (kind is CommandKind.Global or CommandKind.InverseGlobal or CommandKind.Sort) (a, b) = (1, size);
414 else if (kind is CommandKind.Join or CommandKind.JoinSpaced) (a, b) = (state.Current, state.Current + 1);
415 else (a, b) = (state.Current, state.Current);
416 if (a < 1 || b < 1 || a > size || b > size) throw new CommandError("address out of range");
417 if (a > b || a == b && kind is (CommandKind.Join or CommandKind.JoinSpaced))
418 throw new CommandError("invalid range");
419 return (a, b);
420 }

422 public void Process(string line, string[] text, State before)
423 {
424 int position = 0;
425 Address address = Parser.Form(Parser.Scan(line, ref position));
426 Command command = Parser.Parse(line[position..]);
427 if (NoAddress(command.Kind) && address.Count != 0) throw new CommandError("unexpected address");
428 if (OneAddress(command.Kind) && address.Count == 2) throw new CommandError("too many addresses");
429 var (a, b) = Range(command, Evaluate(address));
430 Execute(command, a, b, text);
431 if (state.Lines.Count > 200) throw new CommandError("buffer full");
432 if (Modifies(command))
433 {
434 state.Undo.Add(new(line, before.Lines.ToArray(), before.Current));
435 state.Redo.Clear();
436 }
437 }

439 void Print(int number, char form = 'p')
440 {
441 string text = state.Lines[number - 1].Text;
442 if (form == 'n') Output.Add($"{number,3} {text}");
443 else if (form == 'l')
444 {
445 text = text.Replace("\\", "\\\\", StringComparison.Ordinal) + "$";
446 while (text.Length > 40)
447 {
448 Output.Add(text[..39] + "\\");
449 text = text[39..];
450 }
451 Output.Add(text);
452 }
453 else Output.Add(text);
454 }

480 static int LeadingInteger(string text)
481 {
482 int position = text[0] == '-' ? 1 : 0, start = position;
483 while (position < text.Length && Parser.Digit(text[position])) position++;
484 return position == start ? 0 : int.Parse(text[..position], CultureInfo.InvariantCulture);

487 void Sort(Command command, int a, int b)
488 {
489 List<Line> lines = state.Lines.GetRange(a - 1, b - a + 1);
490 bool reverse = command.Options.Contains('r'), numeric = command.Options.Contains('n');
491 string Key(Line line) => command.Options.Contains('i') ? line.Text.ToLowerInvariant() : line.Text;
492 IEnumerable<Line> ordered = numeric
493 ? reverse ? lines.OrderByDescending(line => LeadingInteger(line.Text)) : lines.OrderBy(line => LeadingInteger(line.Text))
494 : reverse ? lines.OrderByDescending(Key, StringComparer.Ordinal) : lines.OrderBy(Key, StringComparer.Ordinal);
495 var result = new List<Line>();
496 foreach (Line line in ordered)
497 {
498 if (command.Options.Contains('u') && result.Count != 0 &&
499 (numeric ? LeadingInteger(result[^1].Text) == LeadingInteger(line.Text) : Key(result[^1]) == Key(line))) continue;
500 result.Add(line);
501 }
502 state.Lines.RemoveRange(a - 1, b - a + 1);
503 state.Lines.InsertRange(a - 1, result);
504 state.Current = a + result.Count - 1;
505 }

507 int Substitute(Command command, int a, int b, bool global)
508 {
509 string pattern = Pattern(command.Pattern);
510 char?[] replacement = command.Replacement ?? state.Replacement ?? throw new CommandError("no previous substitution");
511 state.Replacement = replacement;
512 var expanded = new StringBuilder();
513 foreach (char? part in replacement)
514 if (part is null) expanded.Append(pattern); else expanded.Append(part.Value);
515 string value = expanded.ToString();
516 int substituted = 0;
517 for (int number = a; number <= b; number++)
518 {
519 Line line = state.Lines[number - 1];
520 var result = new StringBuilder();
521 int position = 0, occurrences = 0, match;
522 while ((match = line.Text.IndexOf(pattern, position, StringComparison.Ordinal)) >= 0)
523 {
524 result.Append(line.Text, position, match - position);
525 occurrences++;
526 result.Append(occurrences == command.Count || command.All && occurrences >= command.Count ? value : pattern);
527 position = match + pattern.Length;
528 }
529 if (occurrences < command.Count) continue;
530 result.Append(line.Text, position, line.Text.Length - position);
531 string changed = result.ToString();
532 if (changed.Length > 80) throw new CommandError("line too long");
533 if (changed.Length == 0 || changed[^1] == ' ') throw new CommandError("bad line");
534 state.Lines[number - 1] = line with { Text = changed };
535 state.Current = number;
536 substituted++;
537 }
538 if (substituted == 0 && !global) throw new CommandError("no match");
539 if (substituted != 0 && command.Suffix != '\0') Print(state.Current, command.Suffix);
540 return substituted;
541 }
542
543 int Execute(Command command, int a, int b, string[] text, bool global = false)
544 {
545 CommandKind kind = command.Kind;
546 switch (kind)
547 {
548 case CommandKind.Empty: Print(b); state.Current = b; break;
549 case CommandKind.Print: case CommandKind.Numbered: case CommandKind.Listed:
550 char form = kind == CommandKind.Print ? 'p' : kind == CommandKind.Numbered ? 'n' : 'l';
551 for (int i = a; i <= b; i++) Print(i, form);
552 state.Current = b;
553 break;
554 case CommandKind.Window:
555 if (command.Count != 0) state.DisplayCount = command.Count;
556 int end = Math.Min(state.Lines.Count, a + state.DisplayCount - 1);
557 for (int i = a; i <= end; i++) Print(i);
558 state.Current = end;
559 break;
560 case CommandKind.Number: Output.Add(a.ToString(CultureInfo.InvariantCulture)); break;
561 case CommandKind.Count:
562 string[] counted = Contents(a, b);
563 Output.Add($"count lines={counted.Length} words={counted.Sum(line => line.Split(' ', StringSplitOptions.RemoveEmptyEntries).Length)} chars={counted.Sum(line => line.Length)}");
564 break;
565 case CommandKind.WritePrint:
566 for (int i = a; i <= b; i++) Print(i);
567 Output.Add($"lines={b - a + 1} bytes={Bytes(Contents(a, b))}");
568 break;
569 case CommandKind.Append: Insert(a, text); break;
570 case CommandKind.Insert: Insert(Math.Max(a, 1) - 1, text); break;
571 case CommandKind.Change: case CommandKind.Delete:
572 state.Cut = Contents(a, b);
573 state.Lines.RemoveRange(a - 1, b - a + 1);
574 if (kind == CommandKind.Change && text.Length != 0) Insert(a - 1, text);
575 else state.Current = Math.Min(a, state.Lines.Count);
576 break;
577 case CommandKind.Yank: state.Cut = Contents(a, b); break;
578 case CommandKind.Put:
579 if (state.Cut.Length == 0) throw new CommandError("nothing to put");
580 Insert(a, state.Cut);
581 break;
582 case CommandKind.Move: case CommandKind.Copy:
583 int destination = Evaluate(command.Destination);
584 if (destination < 0 || destination > state.Lines.Count) throw new CommandError("address out of range");
585 if (kind == CommandKind.Copy) Insert(destination, Contents(a, b));
586 else
587 {
588 if (a <= destination && destination < b) throw new CommandError("invalid destination");
589 List<Line> moved = state.Lines.GetRange(a - 1, b - a + 1);
590 state.Lines.RemoveRange(a - 1, moved.Count);
591 int position = destination < a ? destination : destination - moved.Count;
592 state.Lines.InsertRange(position, moved);
593 state.Current = position + moved.Count;
594 }
595 break;
596 case CommandKind.Join: case CommandKind.JoinSpaced:
597 string[] pieces = Contents(a, b);
598 if (kind == CommandKind.JoinSpaced)
599 for (int i = 1; i < pieces.Length; i++) pieces[i] = pieces[i].TrimStart(' ');
600 string joined = string.Join(kind == CommandKind.Join ? "" : " ", pieces);
601 if (joined.Length > 80) throw new CommandError("line too long");
602 state.Lines[a - 1] = state.Lines[a - 1] with { Text = joined };
603 state.Lines.RemoveRange(a, b - a);
604 state.Current = a;
605 break;
606 case CommandKind.Indent: case CommandKind.Unindent: case CommandKind.Case:
607 int width = command.Count == 0 ? 4 : command.Count;
608 for (int i = a - 1; i < b; i++)
609 {
610 Line line = state.Lines[i];
611 string changed = line.Text;
612 if (kind == CommandKind.Indent)
613 {
614 if (changed.Length + width > 80) throw new CommandError("line too long");
615 changed = new string(' ', width) + changed;
616 }
617 else if (kind == CommandKind.Unindent)
618 {
619 int removed = 0;
620 while (removed < width && removed < changed.Length && changed[removed] == ' ') removed++;
621 changed = changed[removed..];
622 }
623 else if (command.CaseMode == 'u') changed = changed.ToUpperInvariant();
624 else if (command.CaseMode == 'l') changed = changed.ToLowerInvariant();
625 else changed = new string(changed.Select(c => c is >= 'A' and <= 'Z' ? (char)(c + 32) :
626 c is >= 'a' and <= 'z' ? (char)(c - 32) : c).ToArray());
627 state.Lines[i] = line with { Text = changed };
628 }
629 state.Current = b;
630 break;
631 case CommandKind.Sort: Sort(command, a, b); break;
632 case CommandKind.Substitute: return Substitute(command, a, b, global);
633 case CommandKind.Mark: state.Marks[command.Name[0] - 'a'] = state.Lines[a - 1].Id; break;
634 case CommandKind.Marks:
635 var marks = new List<string>();
636 for (int i = 0; i < 26; i++)
637 {
638 int number = state.Marks[i] is long id ? state.Lines.FindIndex(line => line.Id == id) + 1 : 0;
639 if (number != 0) marks.Add($"{(char)('a' + i)}={number}");
640 }
641 Output.Add("marks " + (marks.Count == 0 ? "none" : string.Join(' ', marks)));
642 break;
643 case CommandKind.Undo: case CommandKind.Redo:
644 History(kind == CommandKind.Undo, command.Count == 0 ? 1 : command.Count);
645 break;
646 case CommandKind.History:
647 Output.Add($"history undo={state.Undo.Count} redo={state.Redo.Count}");
648 for (int i = 0; i < state.Undo.Count; i++) Output.Add($"undo {i + 1}: {state.Undo[^(i + 1)].Command}");
649 for (int i = 0; i < state.Redo.Count; i++) Output.Add($"redo {i + 1}: {state.Redo[^(i + 1)].Command}");
650 break;
651 case CommandKind.Global: case CommandKind.InverseGlobal:
652 string pattern = Pattern(command.Pattern);
653 long[] selected = state.Lines.GetRange(a - 1, b - a + 1)
654 .Where(line => line.Text.Contains(pattern, StringComparison.Ordinal) == (kind == CommandKind.Global))
655 .Select(line => line.Id).ToArray();
656 if (selected.Length == 0) throw new CommandError("no match");
657 int substituted = 0;
658 Command subcommand = command.Subcommand!;
659 foreach (long id in selected)
660 {
661 int index = state.Lines.FindIndex(line => line.Id == id);
662 if (index < 0) continue;
663 state.Current = index + 1;
664 var (first, last) = Range(subcommand, []);
665 substituted += Execute(subcommand, first, last, text, true);
666 }
667 if (subcommand.Kind == CommandKind.Substitute && substituted == 0) throw new CommandError("no match");
668 break;
669 case CommandKind.Write: case CommandKind.WriteAppend: case CommandKind.Read: case CommandKind.Edit:
670 if (kind is CommandKind.Write or CommandKind.WriteAppend)
671 {
672 string[] contents = Contents(a, b);
673 state.Files[command.Name] = kind == CommandKind.WriteAppend && state.Files.TryGetValue(command.Name, out string[]? old)
674 ? [.. old, .. contents] : contents;
675 }
676 else
677 {
678 if (!state.Files.TryGetValue(command.Name, out string[]? contents)) throw new CommandError("no such file");
679 if (kind == CommandKind.Edit) state.Lines.Clear();
680 Insert(kind == CommandKind.Edit ? 0 : a, contents);
681 }
682 string[] file = state.Files[command.Name];
683 Output.Add($"file {command.Name} lines={file.Length} bytes={Bytes(file)}");
684 break;
685 case CommandKind.Files:
686 Output.Add("files " + (state.Files.Count == 0 ? "none" : string.Join(' ',
687 state.Files.OrderBy(pair => pair.Key, StringComparer.Ordinal).Select(pair => $"{pair.Key}={pair.Value.Length}"))));
688 break;
689 }
690 if (command.Suffix != '\0' && state.Current >= 1) Print(state.Current, command.Suffix);
691 return 0;
692 }
693}
