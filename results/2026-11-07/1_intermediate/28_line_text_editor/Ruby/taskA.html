22 when :undo, :redo, :history, :marks, :files, :edit then 0

38module Parsing
39 SIMPLE = { 'a' => :append, 'i' => :insert, 'c' => :change, 'p' => :print, 'n' => :number,
40 'l' => :list, '=' => :equals, '#' => :count, 'y' => :yank, 'K' => :marks,
41 'h' => :history, 'f' => :files }.freeze
42 BASES = %i[number current last mark forward backward].freeze

46 def digit?(c) = c && c >= '0' && c <= '9'
47 def lower?(c) = c && c >= 'a' && c <= 'z'
48 def digits?(s) = s.match?(/\A[0-9]+\z/)
49
50 def delimited(s, position, delimiter)
51 text = String.new
52 atoms = []
53 while position < s.length
54 character = s[position]
55 position += 1
56 return Delimited.new(text.freeze, atoms.freeze, position) if character == delimiter
57
58 escaped = character == '\\'
59 if escaped
60 raise CommandError, 'missing delimiter' if position == s.length
61 character = s[position]
62 position += 1
63 end
64 text << character
65 atoms << Atom.new(character.freeze, character == '&' && !escaped)
66 end
67 raise CommandError, 'missing delimiter'
68 end

70 def lex_addresses(s)
71 tokens = []
72 position = 0
73 while position < s.length
74 c = s[position]
75 if digit?(c)
76 start = position
77 position += 1 while digit?(s[position])
78 token = Token.new(:number, s[start...position].to_i(10))
79 elsif c == '+' || c == '-'
80 position += 1
81 start = position
82 position += 1 while digit?(s[position])
83 value = start == position ? 1 : s[start...position].to_i(10)
84 token = Token.new(:offset, c == '-' ? -value : value)
85 elsif '.$,;%'.include?(c)
86 kind = { '.' => :current, '$' => :last, ',' => :comma, ';' => :semicolon, '%' => :percent }.fetch(c)
87 token = Token.new(kind, nil)
88 position += 1
89 elsif c == "'"
90 position += 1
91 name = s[position]
92 position += 1 if name
93 token = Token.new(:mark, name)
94 elsif c == '/' || c == '?'
95 search = delimited(s, position + 1, c)
96 position = search.position
97 token = Token.new(c == '/' ? :forward : :backward, search.text)
98 else
99 break
100 end
101 tokens << token
102 end
103 Lexed.new(tokens.freeze, position)
104 end

106 def with_text?(line)
107 part = line[lex_addresses(line).position..]
108 return true if %w[a i c].include?(part)
109 if part.length >= 2 && 'gv'.include?(part[0]) && !" \\".include?(part[1])
110 rest = part[delimited(part, 2, part[1]).position..]
111 return %w[a i c].include?(rest)
112 end
113 false
114 rescue CommandError
115 false
116 end

118 def valid_address?(tokens)
119 tokens.each_with_index.all? { |token, i| token.kind == :offset || i.zero? && BASES.include?(token.kind) }
120 end

122 def addresses(tokens)
123 return Addresses.new(0, [], [], false) if tokens.empty?
124 if tokens.length == 1 && tokens[0].kind == :percent
125 return Addresses.new(2, [Token.new(:number, 1)], [Token.new(:last, nil)], false)
126 end
127 separators = tokens.each_index.select { |i| %i[comma semicolon].include?(tokens[i].kind) }
128 raise CommandError, 'bad address' if separators.length > 1
129 separator = separators.first
130 left = separator ? tokens[0...separator] : tokens
131 right = separator ? tokens[separator + 1..] : []
132 raise CommandError, 'bad address' unless valid_address?(left) && valid_address?(right)
133 Addresses.new(separator ? 2 : 1, left, right, separator && tokens[separator].kind == :semicolon)
134 end

136 def print_suffix(s)
137 return nil if s.empty?

142 def positive(s)
143 return 0 if s.empty?

145 s.to_i(10)

149 raise CommandError, 'invalid file name' unless s.match?(/\A [a-z0-9]{1,8}\z/)

153 def sort_form?(s) = s == 'sort' || s.start_with?('sort ')

155 def command(part)
156 c = Command.new
157 if sort_form?(part)
158 c.kind = :sort
159 c.options = part == 'sort' ? '' : part[5..]
160 if part != 'sort'
161 valid = c.options.match?(/\A[ruin]+\z/) && c.options.chars.uniq.length == c.options.length
162 valid &&= !(c.options.include?('i') && c.options.include?('n'))
163 raise CommandError, 'invalid suffix' unless valid
164 end
165 return c.freeze
166 end
167 return c.freeze if part.empty?
168 first, arg = part[0], part[1..]
169 if SIMPLE.key?(first)
170 c.kind = SIMPLE.fetch(first)
171 raise CommandError, 'invalid suffix' unless arg.empty?
172 return c.freeze
173 end
174 case first
175 when 'd', 'x', 'j', 'J'
176 c.kind = { 'd' => :delete, 'x' => :put, 'j' => :join, 'J' => :join_space }.fetch(first)
177 c.suffix = print_suffix(arg)
178 when 'z', '>', '<', 'u'
179 c.kind = { 'z' => :page, '>' => :indent, '<' => :unindent, 'u' => :undo }.fetch(first)
180 c.number = positive(arg)
181 when '~'
182 c.kind = :case
183 raise CommandError, 'invalid suffix' unless arg.empty? || %w[u l].include?(arg)
184 c.option = arg.empty? ? nil : arg
185 when 'r'
186 if arg.empty? || digits?(arg)
187 c.kind, c.number = :redo, positive(arg)
188 else
189 c.kind, c.name = :read, filename(arg)
190 end
191 when 'w', 'W', 'e'
192 if first == 'w' && arg.empty?
193 c.kind = :write_print
194 else
195 c.kind = { 'w' => :write, 'W' => :write_append, 'e' => :edit }.fetch(first)
196 c.name = filename(arg)
197 end
198 when 'k'
199 c.kind = :mark
200 raise CommandError, 'invalid mark' unless arg.length == 1 && lower?(arg)
201 c.option = arg
202 when 'm', 't'
203 c.kind = first == 'm' ? :move : :copy
204 destination = lex_addresses(arg)
205 raise CommandError, 'invalid destination' if destination.tokens.empty? || !valid_address?(destination.tokens)
206 c.destination, c.suffix = destination.tokens, print_suffix(arg[destination.position..])
207 when 's'
208 c.kind, c.number = :substitute, 1
209 position = 0
210 unless arg.match?(/\A[gpnl0-9]*\z/)
211 delimiter = arg[0]
212 raise CommandError, 'missing delimiter' if " \\".include?(delimiter)
213 search = delimited(arg, 1, delimiter)
214 replacement = delimited(arg, search.position, delimiter)
215 c.search = search.text
216 c.replacement = replacement.atoms unless arg[search.position...replacement.position - 1] == '%'
217 position = replacement.position
218 end
219 counted = false
220 while position < arg.length
221 flag = arg[position]
222 position += 1
223 if flag == 'g'
224 raise CommandError, 'invalid suffix' if c.all
225 c.all = true
226 elsif 'pnl'.include?(flag)
227 raise CommandError, 'invalid suffix' if c.suffix
228 c.suffix = flag
229 elsif digit?(flag)
230 raise CommandError, 'invalid suffix' if counted
231 counted = true
232 start = position - 1
233 position += 1 while digit?(arg[position])
234 c.number = arg[start...position].to_i(10)
235 raise CommandError, 'invalid suffix' if c.number.zero?
236 else
237 raise CommandError, 'invalid suffix'
238 end
239 end
240 when 'g', 'v'
241 c.kind = first == 'g' ? :global : :inverse
242 raise CommandError, 'missing delimiter' if arg.empty? || " \\".include?(arg[0])
243 search = delimited(arg, 1, arg[0])
244 c.search = search.text
245 body = arg[search.position..]
246 body = 'p' if body.empty?
247 raise CommandError, 'invalid global command' unless 'pnldmtjJs><~aicx'.include?(body[0]) && !sort_form?(body)
248 c.sub = command(body)
249 else
250 raise CommandError, 'unknown command'
251 end
252 c.freeze
253 end
254end

283 def resolve_pattern(search)
284 @pattern = search unless search.empty?
285 raise CommandError, 'no previous pattern' unless @pattern
286 @pattern

289 def address(tokens)
290 value = @current
291 tokens.each do |token|
292 case token.kind
293 when :offset then value += token.value
294 when :number then value = token.value
295 when :current then value = @current
296 when :last then value = @buffer.length
297 when :mark
298 raise CommandError, 'invalid mark' unless Parsing.lower?(token.value)
299 value = locate(@marks[token.value.ord - 'a'.ord])
300 raise CommandError, 'mark not set' unless value
301 when :forward, :backward
302 pattern = resolve_pattern(token.value)
303 size = @buffer.length
304 start = token.kind == :backward && @current.zero? ? size + 1 : @current
305 value = nil
306 1.upto(size) do |step|
307 index = token.kind == :forward ? (start + step - 1) % size : (start - step - 1) % size
308 if @buffer[index].text.include?(pattern)
309 value = index + 1
310 break
311 end
312 end
313 raise CommandError, 'no match' unless value
314 end
315 end
316 value
317 end

319 def evaluate(addresses)
320 return LineRange.new(0, 0) if addresses.count.zero?
321 a = addresses.left.empty? ? (addresses.semicolon ? @current : 1) : address(addresses.left)
322 b = 0
323 if addresses.count == 2
324 if addresses.semicolon
325 raise CommandError, 'address out of range' unless a.between?(0, @buffer.length)
326 @current = a
327 end
328 b = addresses.right.empty? ? @buffer.length : address(addresses.right)
329 end
330 LineRange.new(a, b)
331 end

333 def checked_range(c, count, values)
334 a, b, size = values.first, values.last, @buffer.length
335 return LineRange.new(0, 0) if c.address_kind.zero?
336 if c.address_kind == 1
337 if count.zero?
338 a = case c.kind
339 when :equals, :read then size
340 when :page then @current + 1
341 else @current
342 end
343 end
344 minimum = %i[mark page].include?(c.kind) ? 1 : 0
345 raise CommandError, 'address out of range' unless a.between?(minimum, size)
346 return LineRange.new(a, a)
347 end
348 if count.zero?
349 case c.kind
350 when :write, :write_append, :write_print, :count then return LineRange.new(1, size)
351 when :global, :inverse, :sort then a, b = 1, size
352 when :join, :join_space then a, b = @current, @current + 1
353 else a = b = @current
354 end
355 elsif count == 1
356 b = a
357 end
358 raise CommandError, 'address out of range' unless a.between?(1, size) && b.between?(1, size)
359 raise CommandError, 'invalid range' if a > b || a == b && %i[join join_space].include?(c.kind)
360 LineRange.new(a, b)
361 end

363 def print_line(number, form, output)
364 text = @buffer[number - 1].text
365 case form
366 when 'n'
367 output << format('%3d %s', number, text)
368 when 'l'
369 escaped = text.gsub('\\') { '\\\\' } + '$'
370 position = 0
371 while escaped.length - position > 40
372 output << escaped[position, 39] + '\\'
373 position += 39
374 end
375 output << escaped[position..]
376 else
377 output << text
378 end
379 end

396 def substitute(c, range, output, in_global)
397 pattern = resolve_pattern(c.search)
398 used = c.replacement || @replacement
399 raise CommandError, 'no previous substitution' unless used
400 @replacement = used
401 expanded = used.map { |atom| atom.matched ? pattern : atom.character }.join
402 last = nil
403 range.first.upto(range.last) do |i|
404 old = @buffer[i - 1].text
405 result = String.new
406 position = occurrence = 0
407 changed = false
408 while (start = old.index(pattern, position))
409 result << old[position...start]
410 occurrence += 1
411 if occurrence == c.number || c.all && occurrence > c.number
412 result << expanded
413 changed = true
414 else
415 result << pattern
416 end
417 position = start + pattern.length
418 end
419 result << old[position..]
420 next unless changed
421 raise CommandError, 'line too long' if result.length > 80
422 raise CommandError, 'bad line' if result.empty? || result.end_with?(' ')
423 set_text(i - 1, result)
424 last = i
425 end
426 raise CommandError, 'no match' unless last || in_global
427 if last
428 @current = last
429 print_line(last, c.suffix, output) if c.suffix
430 end
431 !last.nil?
432 end

434 def leading_integer(text)
435 prefix = text[/\A-?[0-9]+/]
436 prefix ? prefix.to_i(10) : 0

439 def sort_lines(c, range)
440 decorated = @buffer[range.first - 1...range.last].each_with_index.map do |line, index|
441 key = if c.options.include?('n')
442 leading_integer(line.text)
443 elsif c.options.include?('i')
444 line.text.downcase
445 else
446 line.text
447 end
448 [key, index, line]
449 end
450 decorated.sort! do |left, right|
451 order = left[0] <=> right[0]
452 order = -order if c.options.include?('r')
453 order.zero? ? left[1] <=> right[1] : order
454 end
455 decorated = decorated.uniq { |entry| entry[0] } if c.options.include?('u')
456 kept = decorated.map(&:last)
457 @buffer[range.first - 1, range.last - range.first + 1] = kept
458 @current = range.first + kept.length - 1
459 end

463 def file_command(c, range, output)
464 file = @files[c.name]
465 if %i[write write_append].include?(c.kind)
466 saved = c.kind == :write_append && file ? file : []
467 file = (saved + contents(range)).freeze
468 @files[c.name] = file
469 else

478 end
479 output << "file #{c.name} lines=#{file.length} bytes=#{file_bytes(file)}"
480 end

482 def transfer_history(is_redo, count)
483 from, to = is_redo ? [@redo, @undo] : [@undo, @redo]
484 raise CommandError, (is_redo ? 'nothing to redo' : 'nothing to undo') if from.length < count
485 count.times do
486 record = from.pop
487 to << HistoryRecord.new(record.command, snapshot)
488 @buffer, @current = record.state.buffer.dup, record.state.current
489 end
490 end

492 def execute(c, range, text, output, in_global: false)
493 a, b = range.first, range.last
494 case c.kind
495 when :empty
496 print_line(b, 'p', output)
497 @current = b
498 when :print, :number, :list
499 form = { print: 'p', number: 'n', list: 'l' }.fetch(c.kind)
500 a.upto(b) { |i| print_line(i, form, output) }
501 @current = b
502 when :page
503 @display = c.number unless c.number.zero?
504 last = [a + @display - 1, @buffer.length].min
505 a.upto(last) { |i| print_line(i, 'p', output) }
506 @current = last
507 when :equals
508 output << a.to_s
509 when :count, :write_print
510 lines = @buffer[a - 1...b]
511 chars = lines.sum { |line| line.text.length }
512 if c.kind == :count
513 words = lines.sum { |line| line.text.scan(/[^ ]+/).length }
514 output << "count lines=#{lines.length} words=#{words} chars=#{chars}"
515 else
516 a.upto(b) { |i| print_line(i, 'p', output) }
517 output << "lines=#{lines.length} bytes=#{chars + lines.length}"
518 end
519 when :append, :insert
520 after = c.kind == :append ? a : [a, 1].max - 1
521 insert(after, text)
522 @current = after + text.length
523 when :change, :delete
524 @cut = contents(range)
525 @buffer.slice!(a - 1, b - a + 1)
526 insert(a - 1, text) if c.kind == :change
527 @current = c.kind == :change && !text.empty? ? a + text.length - 1 : [a, @buffer.length].min
528 when :yank
529 @cut = contents(range)
530 when :put
531 raise CommandError, 'nothing to put' if @cut.empty?
532 insert(a, @cut)
533 @current = a + @cut.length
534 when :move, :copy
535 destination = address(c.destination)
536 raise CommandError, 'address out of range' unless destination.between?(0, @buffer.length)
537 raise CommandError, 'invalid destination' if c.kind == :move && a <= destination && destination < b
538 moved = @buffer[a - 1...b]
539 if c.kind == :move
540 @buffer.slice!(a - 1, b - a + 1)
541 destination -= moved.length if destination >= b
542 else
543 moved = moved.map do |line|
544 copy = Line.new(@next_id, line.text)
545 @next_id += 1
546 copy
547 end
548 end
549 @buffer.insert(destination, *moved)
550 @current = destination + moved.length
551 when :join, :join_space
552 pieces = @buffer[a - 1...b].map(&:text)
553 if c.kind == :join_space
554 pieces = [pieces[0]] + pieces.drop(1).map { |line| line.sub(/\A +/, '') }
555 end
556 joined = pieces.join(c.kind == :join_space ? ' ' : '')
557 raise CommandError, 'line too long' if joined.length > 80
558 set_text(a - 1, joined)
559 @buffer.slice!(a, b - a)
560 @current = a
561 when :indent, :unindent, :case
562 width = c.number.zero? ? 4 : c.number
563 a.upto(b) do |i|
564 old = @buffer[i - 1].text
565 value = case c.kind
566 when :indent
567 raise CommandError, 'line too long' if old.length + width > 80
568 ' ' * width + old
569 when :unindent
570 old[[width, old[/\A */].length].min..]
571 else
572 case c.option
573 when 'u' then old.upcase
574 when 'l' then old.downcase
575 else old.swapcase
576 end
577 end
578 set_text(i - 1, value)
579 end
580 @current = b
581 when :sort
582 sort_lines(c, range)
583 when :substitute
584 return substitute(c, range, output, in_global)
585 when :mark
586 @marks[c.option.ord - 'a'.ord] = @buffer[a - 1].id
587 when :marks
588 marks = @marks.each_with_index.filter_map do |id, index|
589 position = locate(id)
590 "#{('a'.ord + index).chr}=#{position}" if position
591 end
592 output << "marks #{marks.empty? ? 'none' : marks.join(' ')}"
593 when :undo, :redo
594 transfer_history(c.kind == :redo, c.number.zero? ? 1 : c.number)
595 when :history
596 output << "history undo=#{@undo.length} redo=#{@redo.length}"
597 [['undo', @undo], ['redo', @redo]].each do |name, records|
598 records.reverse_each.with_index { |record, i| output << "#{name} #{i + 1}: #{record.command}" }
599 end
600 when :read, :write, :write_append, :edit
601 file_command(c, range, output)
602 when :files
603 files = @files.keys.sort.map { |name| "#{name}=#{@files[name].length}" }
604 output << "files #{files.empty? ? 'none' : files.join(' ')}"
605 when :global, :inverse
606 pattern = resolve_pattern(c.search)
607 selected = @buffer[a - 1...b].select { |line| line.text.include?(pattern) == (c.kind == :global) }.map(&:id)
608 raise CommandError, 'no match' if selected.empty?
609 substituted = false
610 selected.each do |id|
611 position = locate(id)
612 next unless position
613 @current = position
614 subrange = checked_range(c.sub, 0, LineRange.new(0, 0))
615 changed = execute(c.sub, subrange, text, output, in_global: true)
616 substituted ||= changed
617 end
618 raise CommandError, 'no match' if c.sub.kind == :substitute && !substituted
619 end
620 print_line(@current, c.suffix, output) if c.suffix && @current.positive?
621 false
622 end
623end

625class Editor
626 def initialize
627 @state = EditorState.new

630 def apply(line, text)
631 lexed = Parsing.lex_addresses(line)
632 addresses = Parsing.addresses(lexed.tokens)
633 command = Parsing.command(line[lexed.position..])
634 raise CommandError, 'unexpected address' if command.address_kind.zero? && addresses.count.positive?
635 raise CommandError, 'too many addresses' if command.address_kind == 1 && addresses.count == 2
636 trial = @state.dup
637 values = trial.evaluate(addresses)
638 range = trial.checked_range(command, addresses.count, values)
639 output = []
640 trial.execute(command, range, text, output)
641 raise CommandError, 'buffer full' if trial.buffer.length > 200
642 if command.modifying?
643 trial.undo << HistoryRecord.new(line, @state.snapshot)
644 trial.redo.clear
645 end
646 @state = trial
647 output
648 end
649end

651editor = Editor.new
652while (line = STDIN.gets)
653 line = line.chomp.freeze
654 break if line == 'Q'
655 text = []
656 if Parsing.with_text?(line)
657 while (entered = STDIN.gets)
658 entered = entered.chomp.freeze
659 break if entered == '.'
660 text << entered
661 end
662 end
663 begin
664 editor.apply(line, text).each { |message| puts message }
665 rescue CommandError => error
666 puts "? #{error.message}"
667 end
