9#[derive(Clone)]
10struct Atom {
11 byte: u8,
12 matched: bool,
13}

15type Replacement = Vec<Atom>;

17fn replacement_text(atoms: &[Atom]) -> String {
18 String::from_utf8(atoms.iter().map(|a| a.byte).collect()).unwrap()
19}

21fn delimited(s: &str, mut position: usize, delimiter: u8) -> Result<(Replacement, usize)> {
22 let bytes = s.as_bytes();
23 let mut atoms = Vec::new();
24 while position < bytes.len() {
25 let mut byte = bytes[position];
26 position += 1;
27 if byte == delimiter {
28 return Ok((atoms, position));
29 }
30 let escaped = byte == b'\\';
31 if escaped {
32 byte = *bytes.get(position).ok_or("missing delimiter")?;
33 position += 1;
34 }
35 atoms.push(Atom { byte, matched: byte == b'&' && !escaped });
36 }
37 Err("missing delimiter")
38}

40#[derive(Clone)]
41enum Token {
42 Number(i32),
43 Offset(i32),
44 Current,
45 Last,
46 Separator(u8),
47 Percent,
48 Mark(Option<u8>),
49 Search { backward: bool, text: String },
50}

52fn number(bytes: &[u8], position: &mut usize) -> i32 {
53 let mut value = 0;
54 while let Some(&byte) = bytes.get(*position).filter(|b| b.is_ascii_digit()) {
55 value = value * 10 + i32::from(byte - b'0');
56 *position += 1;
57 }
58 value
59}
60
61fn lex_addresses(s: &str) -> Result<(Vec<Token>, usize)> {
62 let bytes = s.as_bytes();
63 let mut position = 0;
64 let mut tokens = Vec::new();
65 while let Some(&byte) = bytes.get(position) {
66 let token = match byte {
67 b'0'..=b'9' => Token::Number(number(bytes, &mut position)),
68 b'+' | b'-' => {
69 position += 1;
70 let value = if bytes.get(position).is_some_and(u8::is_ascii_digit) {
71 number(bytes, &mut position)
72 } else { 1 };
73 Token::Offset(if byte == b'-' { -value } else { value })
74 }
75 b'.' => { position += 1; Token::Current }
76 b'$' => { position += 1; Token::Last }
77 b',' | b';' => { position += 1; Token::Separator(byte) }
78 b'%' => { position += 1; Token::Percent }
79 b'\'' => {
80 position += 1;
81 let name = bytes.get(position).copied();
82 if name.is_some() { position += 1; }
83 Token::Mark(name)
84 }
85 b'/' | b'?' => {
86 let (atoms, end) = delimited(s, position + 1, byte)?;
87 position = end;
88 Token::Search { backward: byte == b'?', text: replacement_text(&atoms) }
89 }
90 _ => break,
91 };
92 tokens.push(token);
93 }
94 Ok((tokens, position))
95}

97fn with_text(line: &str) -> bool {
98 let Ok((_, end)) = lex_addresses(line) else { return false };
99 let part = &line[end..];
100 if matches!(part, "a" | "i" | "c") { return true; }
101 let bytes = part.as_bytes();
102 if bytes.len() >= 2 && matches!(bytes[0], b'g' | b'v') && !matches!(bytes[1], b' ' | b'\\') {
103 if let Ok((_, end)) = delimited(part, 2, bytes[1]) {
104 return matches!(&part[end..], "a" | "i" | "c");
105 }
106 }
107 false
108}

110fn valid_address(tokens: &[Token]) -> bool {
111 tokens.iter().enumerate().all(|(i, token)| match token {
112 Token::Offset(_) => true,
113 Token::Number(_) | Token::Current | Token::Last | Token::Mark(_) | Token::Search { .. } => i == 0,
114 _ => false,
115 })
116}

118enum Addresses {
119 None,
120 One(Vec<Token>),
121 Two { left: Vec<Token>, right: Vec<Token>, semicolon: bool },
122}
123
124impl Addresses {
125 fn parse(tokens: Vec<Token>) -> Result<Self> {
126 if tokens.is_empty() { return Ok(Self::None); }
127 if matches!(tokens.as_slice(), [Token::Percent]) {
128 return Ok(Self::Two { left: vec![Token::Number(1)], right: vec![Token::Last], semicolon: false });
129 }
130 let separators: Vec<usize> = tokens.iter().enumerate()
131 .filter_map(|(i, t)| matches!(t, Token::Separator(_)).then_some(i)).collect();
132 if separators.len() > 1 { return Err("bad address"); }
133 if let Some(&index) = separators.first() {
134 let left = tokens[..index].to_vec();
135 let right = tokens[index + 1..].to_vec();
136 if !valid_address(&left) || !valid_address(&right) { return Err("bad address"); }
137 Ok(Self::Two { left, right, semicolon: matches!(tokens[index], Token::Separator(b';')) })
138 } else {
139 if !valid_address(&tokens) { return Err("bad address"); }
140 Ok(Self::One(tokens))
141 }
142 }
143
144 fn count(&self) -> usize {
145 match self { Self::None => 0, Self::One(_) => 1, Self::Two { .. } => 2 }
146 }
147}

174fn suffix(s: &str) -> Result<Option<u8>> {
175 match s.as_bytes() {
176 [] => Ok(None),
177 [byte @ (b'p' | b'n' | b'l')] => Ok(Some(*byte)),
178 _ => Err("invalid suffix"),
179 }
180}
181
182fn positive(s: &str) -> Result<Option<usize>> {
183 if s.is_empty() { return Ok(None); }
184 if !s.bytes().all(|b| b.is_ascii_digit()) { return Err("invalid suffix"); }
185 let value: usize = s.parse().unwrap();
186 if value == 0 { return Err("invalid suffix"); }
187 Ok(Some(value))
188}

190fn filename(s: &str) -> Result<String> {
191 let name = s.strip_prefix(' ').ok_or("invalid file name")?;
192 if name.is_empty() || name.len() > 8 || !name.bytes().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit()) {
193 return Err("invalid file name");
194 }
195 Ok(name.to_owned())
196}

198fn sort_form(s: &str) -> bool { s == "sort" || s.starts_with("sort ") }
199
200impl Command {
201 fn parse(part: &str) -> Result<Self> {
202 if sort_form(part) {
203 let mut options = SortOptions::default();
204 if part == "sort " { return Err("invalid suffix"); }
205 for byte in part.get(5..).unwrap_or("").bytes() {
206 let flag = match byte {
207 b'r' => &mut options.reverse, b'u' => &mut options.unique,
208 b'i' => &mut options.insensitive, b'n' => &mut options.numeric,
209 _ => return Err("invalid suffix"),
210 };
211 if *flag { return Err("invalid suffix"); }
212 *flag = true;
213 }
214 if options.insensitive && options.numeric { return Err("invalid suffix"); }
215 return Ok(Self::Sort(options));
216 }
217 let Some(&first) = part.as_bytes().first() else { return Ok(Self::Empty) };
218 let arg = &part[1..];
219 if b"aicpnl=#yKhf".contains(&first) {
220 if !arg.is_empty() { return Err("invalid suffix"); }
221 return Ok(match first {
222 b'a' => Self::Append, b'i' => Self::Insert, b'c' => Self::Change,
223 b'p' | b'n' | b'l' => Self::Print(first), b'=' => Self::Equals, b'#' => Self::Count,
224 b'y' => Self::Yank, b'K' => Self::Marks, b'h' => Self::History, _ => Self::Files,
225 });
226 }
227 Ok(match first {
228 b'd' => Self::Delete(suffix(arg)?),
229 b'x' => Self::Put(suffix(arg)?),
230 b'j' | b'J' => Self::Join { spaced: first == b'J', suffix: suffix(arg)? },
231 b'z' => Self::Page(positive(arg)?),
232 b'>' | b'<' => Self::Indent { remove: first == b'<', width: positive(arg)?.unwrap_or(4) },
233 b'u' => Self::Undo { redo: false, count: positive(arg)?.unwrap_or(1) },
234 b'r' if arg.bytes().all(|b| b.is_ascii_digit()) => Self::Undo { redo: true, count: positive(arg)?.unwrap_or(1) },
235 b'r' => Self::Read(filename(arg)?),
236 b'w' if arg.is_empty() => Self::WritePrint,
237 b'w' | b'W' => Self::Write { name: filename(arg)?, append: first == b'W' },
238 b'e' => Self::Edit(filename(arg)?),
239 b'~' => Self::Case(match arg { "" => None, "u" => Some(b'u'), "l" => Some(b'l'), _ => return Err("invalid suffix") }),
240 b'k' => {
241 if arg.len() != 1 || !arg.as_bytes()[0].is_ascii_lowercase() { return Err("invalid mark"); }
242 Self::Mark(usize::from(arg.as_bytes()[0] - b'a'))
243 }
244 b'm' | b't' => {
245 let (destination, end) = lex_addresses(arg)?;
246 if destination.is_empty() || !valid_address(&destination) { return Err("invalid destination"); }
247 Self::Move { copy: first == b't', destination, suffix: suffix(&arg[end..])? }
248 }
249 b's' => {
250 let repeated = arg.bytes().all(|b| b"gpnl".contains(&b) || b.is_ascii_digit());
251 let (search, replacement, mut position) = if repeated {
252 (String::new(), None, 0)
253 } else {
254 let delimiter = arg.as_bytes()[0];
255 if matches!(delimiter, b' ' | b'\\') { return Err("missing delimiter"); }
256 let (search, start) = delimited(arg, 1, delimiter)?;
257 let (replacement, end) = delimited(arg, start, delimiter)?;
258 let previous = &arg[start..end - 1] == "%";
259 (replacement_text(&search), if previous { None } else { Some(replacement) }, end)
260 };
261 let mut value = Substitution { search, replacement, all: false, count: 1, suffix: None };
262 let mut counted = false;
263 let bytes = arg.as_bytes();
264 while let Some(&byte) = bytes.get(position) {
265 position += 1;
266 match byte {
267 b'g' => {
268 if value.all { return Err("invalid suffix"); }
269 value.all = true;
270 }
271 b'p' | b'n' | b'l' => {
272 if value.suffix.is_some() { return Err("invalid suffix"); }
273 value.suffix = Some(byte);
274 }
275 b'0'..=b'9' => {
276 if counted { return Err("invalid suffix"); }
277 counted = true;
278 position -= 1;
279 value.count = number(bytes, &mut position) as usize;
280 if value.count == 0 { return Err("invalid suffix"); }
281 }
282 _ => return Err("invalid suffix"),
283 }
284 }
285 Self::Substitute(value)
286 }
287 b'g' | b'v' => {
288 let delimiter = *arg.as_bytes().first().ok_or("missing delimiter")?;
289 if matches!(delimiter, b' ' | b'\\') { return Err("missing delimiter"); }
290 let (search, end) = delimited(arg, 1, delimiter)?;
291 let body = if end == arg.len() { "p" } else { &arg[end..] };
292 if !b"pnldmtjJs><~aicx".contains(&body.as_bytes()[0]) || sort_form(body) {
293 return Err("invalid global command");
294 }
295 Self::Global { inverse: first == b'v', search: replacement_text(&search), sub: Box::new(Self::parse(body)?) }
296 }
297 _ => return Err("unknown command"),
298 })
299 }

301 fn address_kind(&self) -> usize {
302 match self {
303 Self::Undo { .. } | Self::History | Self::Marks | Self::Files | Self::Edit(_) => 0,
304 Self::Append | Self::Insert | Self::Put(_) | Self::Equals | Self::Read(_) | Self::Mark(_) | Self::Page(_) => 1,
305 _ => 2,
306 }
307 }

309 fn modifying(&self) -> bool {
310 match self {
311 Self::Append | Self::Insert | Self::Change | Self::Delete(_) | Self::Put(_) | Self::Join { .. } |
312 Self::Move { .. } | Self::Substitute(_) | Self::Indent { .. } | Self::Case(_) | Self::Sort(_) |
313 Self::Read(_) | Self::Edit(_) => true,
314 Self::Global { sub, .. } => sub.modifying(),
315 _ => false,
316 }
317 }

369 fn address(&mut self, tokens: &[Token]) -> Result<i32> {
370 let mut value = self.current as i32;
371 for token in tokens {
372 match token {
373 Token::Offset(n) => value += n,
374 Token::Number(n) => value = *n,
375 Token::Current => value = self.current as i32,
376 Token::Last => value = self.buffer.len() as i32,
377 Token::Mark(name) => {
378 let name = name.filter(u8::is_ascii_lowercase).ok_or("invalid mark")?;
379 value = self.marks[usize::from(name - b'a')].and_then(|id| self.locate(id))
380 .ok_or("mark not set")? as i32;
381 }
382 Token::Search { backward, text } => {
383 let pattern = self.pattern(text)?;
384 let size = self.buffer.len() as i32;
385 let start = if *backward && self.current == 0 { size + 1 } else { self.current as i32 };
386 value = (1..=size).find_map(|step| {
387 let index = if *backward { (start - step - 1).rem_euclid(size) } else { (start + step - 1) % size };
388 self.buffer[index as usize].text.contains(&pattern).then_some(index + 1)
389 }).ok_or("no match")?;
390 }
391 _ => unreachable!(),
392 }
393 }
394 Ok(value)
395 }

397 fn evaluate(&mut self, addresses: &Addresses) -> Result<(i32, i32)> {
398 match addresses {
399 Addresses::None => Ok((0, 0)),
400 Addresses::One(tokens) => Ok((self.address(tokens)?, 0)),
401 Addresses::Two { left, right, semicolon } => {
402 let a = if left.is_empty() { if *semicolon { self.current as i32 } else { 1 } } else { self.address(left)? };
403 if *semicolon {
404 if a < 0 || a > self.buffer.len() as i32 { return Err("address out of range"); }
405 self.current = a as usize;
406 }
407 let b = if right.is_empty() { self.buffer.len() as i32 } else { self.address(right)? };
408 Ok((a, b))
409 }
410 }
411 }

413 fn checked_range(&self, command: &Command, count: usize, (mut a, mut b): (i32, i32)) -> Result<Range> {
414 let size = self.buffer.len() as i32;
415 match command.address_kind() {
416 0 => return Ok(Range { first: 0, last: 0 }),
417 1 => {
418 if count == 0 {
419 a = match command {
420 Command::Equals | Command::Read(_) => size,
421 Command::Page(_) => self.current as i32 + 1,
422 _ => self.current as i32,
423 };
424 }
425 let minimum = if matches!(command, Command::Mark(_) | Command::Page(_)) { 1 } else { 0 };
426 if a < minimum || a > size { return Err("address out of range"); }
427 return Ok(Range { first: a as usize, last: a as usize });
428 }
429 _ => {}
430 }
431 if count == 0 {
432 match command {
433 Command::WritePrint | Command::Write { .. } | Command::Count => return Ok(Range { first: 1, last: size as usize }),
434 Command::Global { .. } | Command::Sort(_) => { a = 1; b = size; }
435 Command::Join { .. } => { a = self.current as i32; b = a + 1; }
436 _ => { a = self.current as i32; b = a; }
437 }
438 } else if count == 1 { b = a; }
439 if a < 1 || a > size || b < 1 || b > size { return Err("address out of range"); }
440 if a > b || a == b && matches!(command, Command::Join { .. }) { return Err("invalid range"); }
441 Ok(Range { first: a as usize, last: b as usize })
442 }

444 fn print_line(&self, number: usize, form: u8, output: &mut String) {
445 let text = &self.buffer[number - 1].text;
446 match form {
447 b'n' => writeln!(output, "{number:>3} {text}").unwrap(),
448 b'l' => {
449 let escaped = text.replace('\\', "\\\\") + "$";
450 let mut remaining = escaped.as_str();
451 while remaining.len() > 40 {
452 writeln!(output, "{}\\", &remaining[..39]).unwrap();
453 remaining = &remaining[39..];
454 }
455 writeln!(output, "{remaining}").unwrap();
456 }
457 _ => writeln!(output, "{text}").unwrap(),
458 }
459 }

476 fn substitute(&mut self, sub: &Substitution, range: Range, output: &mut String, in_global: bool) -> Result<bool> {
477 let pattern = self.pattern(&sub.search)?;
478 let replacement = sub.replacement.clone().or_else(|| self.replacement.clone()).ok_or("no previous substitution")?;
479 self.replacement = Some(replacement.clone());
480 let mut expanded = String::new();
481 for atom in replacement {
482 if atom.matched { expanded.push_str(&pattern); }
483 else { expanded.push(char::from(atom.byte)); }
484 }
485 let mut last = None;
486 for i in range.first..=range.last {
487 let old = &self.buffer[i - 1].text;
488 let mut result = String::new();
489 let mut position = 0;
490 let mut changed = false;
491 for (index, (start, matched)) in old.match_indices(&pattern).enumerate() {
492 result.push_str(&old[position..start]);
493 if index + 1 == sub.count || sub.all && index + 1 > sub.count {
494 result.push_str(&expanded);
495 changed = true;
496 } else { result.push_str(matched); }
497 position = start + pattern.len();
498 }
499 result.push_str(&old[position..]);
500 if changed {
501 if result.len() > 80 { return Err("line too long"); }
502 if result.is_empty() || result.ends_with(' ') { return Err("bad line"); }
503 self.lines_mut()[i - 1].text = result;
504 last = Some(i);
505 }
506 }
507 if let Some(last) = last {
508 self.current = last;
509 if let Some(suffix) = sub.suffix { self.print_line(last, suffix, output); }
510 } else if !in_global { return Err("no match"); }
511 Ok(last.is_some())
512 }

514 fn transfer_history(&mut self, redo: bool, count: usize) -> Result<()> {
515 if (if redo { self.redo.len() } else { self.undo.len() }) < count {
516 return Err(if redo { "nothing to redo" } else { "nothing to undo" });
517 }
518 for _ in 0..count {
519 let record = if redo { self.redo.pop().unwrap() } else { self.undo.pop().unwrap() };
520 let replacement = Rc::new(Record { command: record.command.clone(), state: self.snapshot() });
521 if redo { self.undo.push(replacement); } else { self.redo.push(replacement); }
522 self.buffer = Rc::clone(&record.state.buffer);
523 self.current = record.state.current;
524 }
525 Ok(())
526 }

528 fn execute(&mut self, command: &Command, range: Range, text: &[String], output: &mut String, in_global: bool) -> Result<bool> {
529 let Range { first: a, last: b } = range;
530 match command {
531 Command::Empty => { self.print_line(b, b'p', output); self.current = b; }
532 Command::Print(form) => {
533 for i in a..=b { self.print_line(i, *form, output); }
534 self.current = b;
535 }
536 Command::Page(count) => {
537 if let Some(count) = count { self.display = *count; }
538 let last = (a + self.display - 1).min(self.buffer.len());
539 for i in a..=last { self.print_line(i, b'p', output); }
540 self.current = last;
541 }
542 Command::Equals => writeln!(output, "{a}").unwrap(),
543 Command::Count | Command::WritePrint => {
544 let lines = &self.buffer[a - 1..b];
545 let chars: usize = lines.iter().map(|line| line.text.len()).sum();
546 if matches!(command, Command::Count) {
547 let words: usize = lines.iter().map(|line| line.text.split(' ').filter(|word| !word.is_empty()).count()).sum();
548 writeln!(output, "count lines={} words={words} chars={chars}", lines.len()).unwrap();
549 } else {
550 for i in a..=b { self.print_line(i, b'p', output); }
551 writeln!(output, "lines={} bytes={}", lines.len(), chars + lines.len()).unwrap();
552 }
553 }
554 Command::Append | Command::Insert => {
555 let after = if matches!(command, Command::Append) { a } else { a.saturating_sub(1) };
556 self.insert(after, text);
557 self.current = after + text.len();
558 }
559 Command::Change | Command::Delete(_) => {
560 self.cut = Rc::new(self.contents(range));
561 self.lines_mut().drain(a - 1..b);
562 if matches!(command, Command::Change) { self.insert(a - 1, text); }
563 self.current = if matches!(command, Command::Change) && !text.is_empty() {
564 a + text.len() - 1
565 } else { a.min(self.buffer.len()) };
566 }
567 Command::Yank => self.cut = Rc::new(self.contents(range)),
568 Command::Put(_) => {
569 if self.cut.is_empty() { return Err("nothing to put"); }
570 let cut = Rc::clone(&self.cut);
571 self.insert(a, &cut);
572 self.current = a + cut.len();
573 }
574 Command::Move { copy, destination, .. } => {
575 let destination = self.address(destination)?;
576 if destination < 0 || destination > self.buffer.len() as i32 { return Err("address out of range"); }
577 let mut destination = destination as usize;
578 if !*copy && a <= destination && destination < b { return Err("invalid destination"); }
579 let mut moved = self.buffer[a - 1..b].to_vec();
580 if *copy {
581 for line in &mut moved { line.id = self.next_id; self.next_id += 1; }
582 } else {
583 self.lines_mut().drain(a - 1..b);
584 if destination >= b { destination -= moved.len(); }
585 }
586 self.current = destination + moved.len();
587 self.lines_mut().splice(destination..destination, moved);
588 }
589 Command::Join { spaced, .. } => {
590 let mut joined = self.buffer[a - 1].text.clone();
591 for line in &self.buffer[a..b] {
592 if *spaced { joined.push(' '); joined.push_str(line.text.trim_start_matches(' ')); }
593 else { joined.push_str(&line.text); }
594 }
595 if joined.len() > 80 { return Err("line too long"); }
596 self.lines_mut()[a - 1].text = joined;
597 self.lines_mut().drain(a..b);
598 self.current = a;
599 }
600 Command::Indent { remove, width } => {
601 for line in &mut self.lines_mut()[a - 1..b] {
602 if *remove {
603 let count = line.text.bytes().take_while(|b| *b == b' ').count().min(*width);
604 line.text.drain(..count);
605 } else {
606 if line.text.len() + *width > 80 { return Err("line too long"); }
607 line.text.insert_str(0, &" ".repeat(*width));
608 }
609 }
610 self.current = b;
611 }
612 Command::Case(option) => {
613 for line in &mut self.lines_mut()[a - 1..b] {
614 line.text = match option {
615 Some(b'u') => line.text.to_ascii_uppercase(),
616 Some(b'l') => line.text.to_ascii_lowercase(),
617 _ => line.text.bytes().map(|byte| char::from(if byte.is_ascii_lowercase() {
618 byte.to_ascii_uppercase()
619 } else { byte.to_ascii_lowercase() })).collect(),
620 };
621 }
622 self.current = b;
623 }
624 Command::Sort(options) => {
625 let mut lines = self.buffer[a - 1..b].to_vec();
626 lines.sort_by(|left, right| {
627 let order = compare_lines(left, right, options);
628 if options.reverse { order.reverse() } else { order }
629 });
630 if options.unique { lines.dedup_by(|left, right| compare_lines(left, right, options) == Ordering::Equal); }
631 self.current = a + lines.len() - 1;
632 self.lines_mut().splice(a - 1..b, lines);
633 }
634 Command::Substitute(sub) => return self.substitute(sub, range, output, in_global),
635 Command::Mark(name) => self.marks[*name] = Some(self.buffer[a - 1].id),
636 Command::Marks => {
637 output.push_str("marks");
638 let mut any = false;
639 for (name, mark) in self.marks.iter().enumerate() {
640 if let Some(position) = mark.and_then(|id| self.locate(id)) {
641 write!(output, " {}={position}", char::from(b'a' + name as u8)).unwrap();
642 any = true;
643 }
644 }
645 output.push_str(if any { "\n" } else { " none\n" });
646 }
647 Command::Undo { redo, count } => self.transfer_history(*redo, *count)?,
648 Command::History => {
649 writeln!(output, "history undo={} redo={}", self.undo.len(), self.redo.len()).unwrap();
650 for (name, records) in [("undo", &self.undo), ("redo", &self.redo)] {
651 for (i, record) in records.iter().rev().enumerate() {
652 writeln!(output, "{name} {}: {}", i + 1, record.command).unwrap();
653 }
654 }
655 }
656 Command::Read(name) | Command::Edit(name) => {
657 let file = self.files.get(name).cloned().ok_or("no such file")?;
658 let after = if matches!(command, Command::Edit(_)) { self.lines_mut().clear(); 0 } else { a };
659 self.insert(after, &file);
660 self.current = after + file.len();
661 writeln!(output, "file {name} lines={} bytes={}", file.len(), file_bytes(&file)).unwrap();
662 }
663 Command::Write { name, append } => {
664 let mut file = if *append { self.files.get(name).map(|file| (**file).clone()).unwrap_or_default() } else { Vec::new() };
665 file.extend(self.contents(range));
666 writeln!(output, "file {name} lines={} bytes={}", file.len(), file_bytes(&file)).unwrap();
667 self.files.insert(name.clone(), Rc::new(file));
668 }
669 Command::Files => {
670 output.push_str("files");
671 for (name, file) in &self.files { write!(output, " {name}={}", file.len()).unwrap(); }
672 output.push_str(if self.files.is_empty() { " none\n" } else { "\n" });
673 }
674 Command::Global { inverse, search, sub } => {
675 let pattern = self.pattern(search)?;
676 let selected: Vec<u64> = self.buffer[a - 1..b].iter()
677 .filter(|line| line.text.contains(&pattern) != *inverse).map(|line| line.id).collect();
678 if selected.is_empty() { return Err("no match"); }
679 let mut substituted = false;
680 for id in selected {
681 if let Some(position) = self.locate(id) {
682 self.current = position;
683 let range = self.checked_range(sub, 0, (0, 0))?;
684 let changed = self.execute(sub, range, text, output, true)?;
685 substituted |= changed;
686 }
687 }
688 if matches!(sub.as_ref(), Command::Substitute(_)) && !substituted { return Err("no match"); }
689 }
690 }
691 if let Some(suffix) = command.print_suffix() {
692 if self.current > 0 { self.print_line(self.current, suffix, output); }
693 }
694 Ok(false)
695 }
696
697 fn apply(&mut self, line: &str, text: &[String]) -> Result<String> {
698 let (tokens, end) = lex_addresses(line)?;
699 let addresses = Addresses::parse(tokens)?;
700 let command = Command::parse(&line[end..])?;
701 let count = addresses.count();
702 if command.address_kind() == 0 && count > 0 { return Err("unexpected address"); }
703 if command.address_kind() == 1 && count == 2 { return Err("too many addresses"); }
704 let mut trial = self.clone();
705 let values = trial.evaluate(&addresses)?;
706 let range = trial.checked_range(&command, count, values)?;
707 let mut output = String::new();
708 trial.execute(&command, range, text, &mut output, false)?;
709 if trial.buffer.len() > 200 { return Err("buffer full"); }
710 if command.modifying() {
711 trial.undo.push(Rc::new(Record { command: line.to_owned(), state: self.snapshot() }));
712 trial.redo.clear();
713 }
714 *self = trial;
715 Ok(output)
716 }
717}

719fn leading_integer(s: &str) -> i64 {
720 let (negative, digits) = if let Some(rest) = s.strip_prefix('-') { (true, rest) } else { (false, s) };
721 let value = digits.bytes().take_while(u8::is_ascii_digit).fold(0_i64, |n, b| n * 10 + i64::from(b - b'0'));
722 if negative { -value } else { value }
723}
724
725fn compare_lines(a: &Line, b: &Line, options: &SortOptions) -> Ordering {
726 if options.numeric { leading_integer(&a.text).cmp(&leading_integer(&b.text)) }
727 else if options.insensitive { a.text.to_ascii_lowercase().cmp(&b.text.to_ascii_lowercase()) }
728 else { a.text.cmp(&b.text) }
729}

731fn file_bytes(file: &[String]) -> usize { file.iter().map(|line| line.len() + 1).sum() }

733fn main() -> io::Result<()> {
734 let mut input = String::new();
735 io::stdin().read_to_string(&mut input)?;
736 let mut lines = input.lines();
737 let mut editor = Editor::new();
738 let stdout = io::stdout();
739 let mut output = io::BufWriter::new(stdout.lock());
740 while let Some(line) = lines.next() {
741 if line == "Q" { break; }
742 let mut text = Vec::new();
743 if with_text(line) {
744 for entered in lines.by_ref() {
745 if entered == "." { break; }
746 text.push(entered.to_owned());
747 }
748 }
749 match editor.apply(line, &text) {
750 Ok(response) => output.write_all(response.as_bytes())?,
751 Err(reason) => writeln!(output, "? {reason}")?,
752 }
753 }
754 output.flush()
755}
