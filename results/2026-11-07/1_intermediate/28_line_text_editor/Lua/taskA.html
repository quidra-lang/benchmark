45function Cursor:delimited(delimiter)
46 local start, content, replacement = self.position, {}, {}
47 while self.position <= #self.text do
48 local c = self.text:sub(self.position, self.position)
49 self.position = self.position + 1
50 if c == delimiter then
51 return {text = table.concat(content), replacement = replacement,
52 raw = self.text:sub(start, self.position - 2)}
53 end
54 local escaped = c == "\\"
55 if escaped then
56 if self.position > #self.text then break end
57 c = self.text:sub(self.position, self.position)
58 self.position = self.position + 1
59 end
60 content[#content + 1] = c
61 if c == "&" and not escaped then replacement[#replacement + 1] = false
62 else replacement[#replacement + 1] = c end
63 end
64 fail("missing delimiter")
65end

67function Cursor:address()
68 local elements = {}
69 while self.position <= #self.text do
70 local c = self.text:sub(self.position, self.position)
71 local element
72 if digit(c) then
73 local start = self.position
74 repeat self.position = self.position + 1
75 until not digit(self.text:sub(self.position, self.position))
76 element = {kind = "number", value = tonumber(self.text:sub(start, self.position - 1), 10)}
77 elseif c == "+" or c == "-" then
78 self.position = self.position + 1
79 local start = self.position
80 while digit(self.text:sub(self.position, self.position)) do self.position = self.position + 1 end
81 local value = start == self.position and 1 or tonumber(self.text:sub(start, self.position - 1), 10)
82 element = {kind = "offset", value = c == "+" and value or -value}
83 elseif c == "'" then
84 self.position = self.position + 1
85 element = {kind = "mark", name = self.text:sub(self.position, self.position)}
86 if self.position <= #self.text then self.position = self.position + 1 end
87 elseif c == "/" or c == "?" then
88 self.position = self.position + 1
89 element = {kind = c, text = self:delimited(c).text}
90 elseif c == "." or c == "$" or c == "," or c == ";" or c == "%" then
91 element = {kind = c}
92 self.position = self.position + 1
93 else
94 break
95 end
96 elements[#elements + 1] = element
97 end
98 return elements
99end

101local function expression(elements)
102 for i, element in ipairs(elements) do
103 if element.kind ~= "offset" and
104 (i ~= 1 or element.kind == "," or element.kind == ";" or element.kind == "%") then return false end
105 end
106 return true
107end

109local function address_form(elements)
110 if #elements == 1 and elements[1].kind == "%" then
111 return {count = 2, left = {{kind = "number", value = 1}}, right = {{kind = "$"}}, separator = ","}
112 end
113 local separator
114 for i, element in ipairs(elements) do
115 if element.kind == "," or element.kind == ";" then
116 if separator ~= nil then fail("bad address") end
117 separator = i
118 end
119 end
120 if separator == nil then
121 if not expression(elements) then fail("bad address") end
122 return {count = #elements == 0 and 0 or 1, left = elements}
123 end
124 local left, right = slice(elements, 1, separator - 1), slice(elements, separator + 1, #elements)
125 if not expression(left) or not expression(right) then fail("bad address") end
126 return {count = 2, left = left, right = right, separator = elements[separator].kind}
127end

149local function substitution(argument)
150 local result, flags = command("s"), argument
151 result.count = 1
152 if argument:find("[^gpnl0-9]") then
153 local delimiter = argument:sub(1, 1)
154 if delimiter == " " or delimiter == "\\" then fail("missing delimiter") end
155 local cursor = Cursor.new(argument, 2)
156 result.pattern = cursor:delimited(delimiter).text
157 local replacement = cursor:delimited(delimiter)
158 if replacement.raw ~= "%" then result.replacement = replacement.replacement end
159 flags = cursor:rest()
160 end
161 local has_count, position = false, 1
162 while position <= #flags do
163 local c = flags:sub(position, position)
164 position = position + 1
165 if c == "g" and not result.all then result.all = true
166 elseif print_suffix(c) and result.suffix == "" then result.suffix = c
167 elseif digit(c) and not has_count then
168 local start = position - 1
169 while digit(flags:sub(position, position)) do position = position + 1 end
170 result.count = tonumber(flags:sub(start, position - 1), 10)
171 if result.count == 0 then fail("invalid suffix") end
172 has_count = true
173 else fail("invalid suffix") end
174 end
175 return result
176end

178local plain_commands = { [""] = true, a = true, i = true, c = true, p = true, n = true, l = true,
179 ["="] = true, ["#"] = true, y = true, K = true, h = true, f = true }

181local function parse_command(text)
182 if sort_form(text) then
183 local result = command("sort")
184 result.options = {}
185 local options = text == "sort" and "" or text:sub(6)
186 if text ~= "sort" and options == "" then fail("invalid suffix") end
187 for c in options:gmatch(".") do
188 if not (c == "r" or c == "u" or c == "i" or c == "n") or result.options[c] then fail("invalid suffix") end
189 result.options[c] = true
190 end
191 if result.options.i and result.options.n then fail("invalid suffix") end
192 return result
193 end
194 local first, argument = text:sub(1, 1), text:sub(2)
195 local result = command(first)
196 if plain_commands[first] then
197 if argument ~= "" then fail("invalid suffix") end
198 elseif first == "d" or first == "x" or first == "j" or first == "J" then
199 result.suffix = suffix(argument)
200 elseif first == "z" or first == ">" or first == "<" or first == "u" then
201 result.count = optional_count(argument)
202 elseif first == "~" then
203 if argument ~= "" and argument ~= "u" and argument ~= "l" then fail("invalid suffix") end
204 result.case_mode = argument
205 elseif first == "r" then
206 if argument == "" or digits(argument) then
207 result.kind, result.count = "redo", optional_count(argument)
208 else
209 result.kind, result.name = "read", file_name(argument)
210 end
211 elseif first == "w" then
212 if argument == "" then result.kind = "write_print"
213 else result.kind, result.name = "write", file_name(argument) end
214 elseif first == "W" or first == "e" then
215 result.name = file_name(argument)
216 elseif first == "k" then
217 if not argument:match("^[a-z]$") then fail("invalid mark") end
218 result.name = argument
219 elseif first == "m" or first == "t" then
220 local cursor = Cursor.new(argument)
221 result.destination = cursor:address()
222 if #result.destination == 0 or not expression(result.destination) then fail("invalid destination") end
223 result.suffix = suffix(cursor:rest())
224 elseif first == "s" then
225 return substitution(argument)
226 elseif first == "g" or first == "v" then
227 local delimiter = argument:sub(1, 1)
228 if delimiter == "" or delimiter == " " or delimiter == "\\" then fail("missing delimiter") end
229 local cursor = Cursor.new(argument, 2)
230 result.pattern = cursor:delimited(delimiter).text
231 local subcommand = cursor:rest()
232 if subcommand == "" then subcommand = "p" end
233 if not ("pnldmtjJs><~aicx"):find(subcommand:sub(1, 1), 1, true) or sort_form(subcommand) then
234 fail("invalid global command")
235 end
236 result.subcommand = parse_command(subcommand)
237 else
238 fail("unknown command")
239 end
240 return result
241end

243local function has_text(line)
244 local ok, result = pcall(function()
245 local cursor = Cursor.new(line)
246 cursor:address()
247 local part = cursor:rest()
248 if part == "a" or part == "i" or part == "c" then return true end
249 local first, delimiter = part:sub(1, 1), part:sub(2, 2)
250 if (first ~= "g" and first ~= "v") or delimiter == "" or delimiter == " " or delimiter == "\\" then return false end
251 cursor = Cursor.new(part, 3)
252 cursor:delimited(delimiter)
253 part = cursor:rest()
254 return part == "a" or part == "i" or part == "c"
255 end)
256 if ok then return result end
257 if type(result) == "table" and result.command_error then return false end
258 error(result, 0)
259end

277local function ascii_compare(a, b)
278 for i = 1, math.min(#a, #b) do
279 if a:byte(i) ~= b:byte(i) then return a:byte(i) < b:byte(i) and -1 or 1 end
280 end
281 return #a < #b and -1 or #a > #b and 1 or 0
282end

284local function change_case(text, mode)
285 local result = {}
286 for i = 1, #text do
287 local c = text:byte(i)
288 if c >= 65 and c <= 90 and mode ~= "u" then c = c + 32
289 elseif c >= 97 and c <= 122 and mode ~= "l" then c = c - 32 end
290 result[i] = string.char(c)
291 end
292 return table.concat(result)
293end

313function Editor:evaluate(elements)
314 local state, value = self.state, self.state.current
315 for _, element in ipairs(elements) do
316 local kind = element.kind
317 if kind == "offset" then value = value + element.value
318 elseif kind == "number" then value = element.value
319 elseif kind == "." then value = state.current
320 elseif kind == "$" then value = #state.lines
321 elseif kind == "mark" then
322 if not element.name:match("^[a-z]$") then fail("invalid mark") end
323 value = self:locate(state.marks[element.name])
324 if value == nil then fail("mark not set") end
325 elseif kind == "/" or kind == "?" then
326 local pattern = self:resolve_pattern(element.text)
327 local step, index, found = kind == "/" and 1 or -1, state.current, false
328 for _ = 1, #state.lines do
329 index = index + step
330 if index < 1 then index = #state.lines elseif index > #state.lines then index = 1 end
331 if state.lines[index].text:find(pattern, 1, true) then value, found = index, true; break end
332 end
333 if not found then fail("no match") end
334 end
335 end
336 return value
337end

339function Editor:range(kind, address)
340 local state, count = self.state, address.count
341 if no_address[kind] then return 0, 0 end
342 local a, b
343 if count >= 1 then
344 if count == 2 and #address.left == 0 then a = address.separator == "," and 1 or state.current
345 else a = self:evaluate(address.left) end
346 if count == 2 then
347 if address.separator == ";" then
348 if a < 0 or a > #state.lines then fail("address out of range") end
349 state.current = a
350 end
351 b = #address.right == 0 and #state.lines or self:evaluate(address.right)
352 else b = a end
353 end
354 if one_address[kind] then
355 if count == 0 then
356 if kind == "=" or kind == "read" then a = #state.lines
357 elseif kind == "z" then a = state.current + 1
358 else a = state.current end
359 end
360 local minimum = (kind == "k" or kind == "z") and 1 or 0
361 if a < minimum or a > #state.lines then fail("address out of range") end
362 return a, a
363 end
364 local whole = kind == "write_print" or kind == "write" or kind == "W" or kind == "#"
365 if count == 0 then
366 if whole or kind == "g" or kind == "v" or kind == "sort" then a, b = 1, #state.lines
367 elseif kind == "j" or kind == "J" then a, b = state.current, state.current + 1
368 else a, b = state.current, state.current end
369 if whole then return a, b end
370 end
371 if a < 1 or a > #state.lines or b < 1 or b > #state.lines then fail("address out of range") end
372 if a > b or a == b and (kind == "j" or kind == "J") then fail("invalid range") end
373 return a, b
374end

380function Editor:print_line(number, form)
381 local text = self.state.lines[number].text
382 if form == "n" then self:emit(string.format("%3d %s", number, text))
383 elseif form == "l" then
384 local escaped = {}
385 for i = 1, #text do
386 local c = text:sub(i, i)
387 escaped[#escaped + 1] = c == "\\" and "\\\\" or c
388 end
389 text = table.concat(escaped) .. "$"
390 while #text > 40 do
391 self:emit(text:sub(1, 39) .. "\\")
392 text = text:sub(40)
393 end
394 self:emit(text)
395 else self:emit(text) end
396end

427function Editor:substitute(cmd, a, b, in_global)
428 local state, pattern = self.state, self:resolve_pattern(cmd.pattern)
429 local replacement = cmd.replacement or state.replacement
430 if replacement == nil then fail("no previous substitution") end
431 state.replacement = replacement
432 local expanded = {}
433 for _, part in ipairs(replacement) do expanded[#expanded + 1] = part == false and pattern or part end
434 local text, changed = table.concat(expanded), 0
435 for i = a, b do
436 local original, pieces = state.lines[i].text, {}
437 local position, occurrence, substituted = 1, 0, false
438 while true do
439 local found = original:find(pattern, position, true)
440 if found == nil then break end
441 occurrence = occurrence + 1
442 pieces[#pieces + 1] = original:sub(position, found - 1)
443 if occurrence == cmd.count or cmd.all and occurrence > cmd.count then
444 pieces[#pieces + 1] = text
445 substituted = true
446 else pieces[#pieces + 1] = pattern end
447 position = found + #pattern
448 end
449 if substituted then
450 pieces[#pieces + 1] = original:sub(position)
451 local result = table.concat(pieces)
452 if #result > 80 then fail("line too long") end
453 if result == "" or result:sub(-1) == " " then fail("bad line") end
454 state.lines[i] = {id = state.lines[i].id, text = result}
455 state.current, changed = i, changed + 1
456 end
457 end
458 if changed == 0 and not in_global then fail("no match") end
459 if changed > 0 and cmd.suffix ~= "" then self:print_line(state.current, cmd.suffix) end
460 return changed
461end

463function Editor:sort_lines(cmd, a, b)
464 local decorated = {}
465 for i = a, b do
466 local line = self.state.lines[i]
467 local key = line.text
468 if cmd.options.n then key = tonumber(key:match("^%-?[0-9]+") or "0", 10)
469 elseif cmd.options.i then key = change_case(key, "l") end
470 decorated[#decorated + 1] = {line = line, key = key, index = i}
471 end
472 table.sort(decorated, function(left, right)
473 local order
474 if cmd.options.n then order = left.key < right.key and -1 or left.key > right.key and 1 or 0
475 else order = ascii_compare(left.key, right.key) end
476 if cmd.options.r then order = -order end
477 if order == 0 then return left.index < right.index end
478 return order < 0
479 end)
480 local sorted, previous = {}, nil
481 for _, item in ipairs(decorated) do
482 if not cmd.options.u or #sorted == 0 or item.key ~= previous then sorted[#sorted + 1] = item.line end
483 previous = item.key
484 end
485 self:splice(a - 1, b - a + 1, sorted)
486 self.state.current = a + #sorted - 1
487end

489function Editor:execute(cmd, a, b, text, in_global)
490 local state, kind = self.state, cmd.kind
491 if kind == "" then
492 self:print_line(b, "p")
493 state.current = b
494 elseif kind == "p" or kind == "n" or kind == "l" then
495 for i = a, b do self:print_line(i, kind) end
496 state.current = b
497 elseif kind == "z" then
498 if cmd.count ~= nil then state.display_count = cmd.count end
499 local last = math.min(a + state.display_count - 1, #state.lines)
500 for i = a, last do self:print_line(i, "p") end
501 state.current = last
502 elseif kind == "=" then self:emit(tostring(a))
503 elseif kind == "#" then
504 local words, chars = 0, 0
505 for i = a, b do
506 local content = state.lines[i].text
507 chars = chars + #content
508 for _ in content:gmatch("[^ ]+") do words = words + 1 end
509 end
510 self:emit("count lines=" .. (b - a + 1) .. " words=" .. words .. " chars=" .. chars)
511 elseif kind == "write_print" then
512 for i = a, b do self:print_line(i, "p") end
513 self:emit("lines=" .. (b - a + 1) .. " bytes=" .. bytes(self:contents(a, b)))
514 elseif kind == "a" or kind == "i" then
515 local position = kind == "a" and a or math.max(a, 1) - 1
516 self:splice(position, 0, self:new_lines(text))
517 state.current = position + #text
518 elseif kind == "c" or kind == "d" then
519 state.cut = self:contents(a, b)
520 self:splice(a - 1, b - a + 1, kind == "c" and self:new_lines(text) or {})
521 state.current = kind == "c" and #text > 0 and a + #text - 1 or math.min(a, #state.lines)
522 elseif kind == "y" then state.cut = self:contents(a, b)
523 elseif kind == "x" then
524 if #state.cut == 0 then fail("nothing to put") end
525 self:splice(a, 0, self:new_lines(state.cut))
526 state.current = a + #state.cut
527 elseif kind == "m" or kind == "t" then
528 local destination = self:evaluate(cmd.destination)
529 if destination < 0 or destination > #state.lines then fail("address out of range") end
530 if kind == "m" and a <= destination and destination < b then fail("invalid destination") end
531 local block = kind == "m" and slice(state.lines, a, b) or self:new_lines(self:contents(a, b))
532 if kind == "m" then
533 self:splice(a - 1, b - a + 1, {})
534 if destination >= b then destination = destination - #block end
535 end
536 self:splice(destination, 0, block)
537 state.current = destination + #block
538 elseif kind == "j" or kind == "J" then
539 local parts = {state.lines[a].text}
540 for i = a + 1, b do
541 local part = state.lines[i].text
542 if kind == "J" then part = part:gsub("^ +", "") end
543 parts[#parts + 1] = part
544 end
545 local joined = table.concat(parts, kind == "J" and " " or "")
546 if #joined > 80 then fail("line too long") end
547 self:splice(a - 1, b - a + 1, {{id = state.lines[a].id, text = joined}})
548 state.current = a
549 elseif kind == ">" or kind == "<" or kind == "~" then
550 for i = a, b do
551 local line, width = state.lines[i], cmd.count or 4
552 local content = line.text
553 if kind == ">" then content = string.rep(" ", width) .. content
554 elseif kind == "<" then content = content:sub(math.min(width, #(content:match("^ *"))) + 1)
555 else content = change_case(content, cmd.case_mode) end
556 if #content > 80 then fail("line too long") end
557 state.lines[i] = {id = line.id, text = content}
558 end
559 state.current = b
560 elseif kind == "sort" then self:sort_lines(cmd, a, b)
561 elseif kind == "s" then return self:substitute(cmd, a, b, in_global)
562 elseif kind == "k" then state.marks[cmd.name] = state.lines[a].id
563 elseif kind == "K" then
564 local marks = {}
565 for code = 97, 122 do
566 local name = string.char(code)
567 local index = self:locate(state.marks[name])
568 if index ~= nil then marks[#marks + 1] = name .. "=" .. index end
569 end
570 self:emit("marks " .. (#marks == 0 and "none" or table.concat(marks, " ")))
571 elseif kind == "u" or kind == "redo" then
572 local source = kind == "u" and state.undo or state.redo
573 local target = kind == "u" and state.redo or state.undo
574 local count = cmd.count or 1
575 if #source < count then fail(kind == "u" and "nothing to undo" or "nothing to redo") end
576 for _ = 1, count do
577 local record = table.remove(source)
578 target[#target + 1] = {command = record.command, lines = copy(state.lines), current = state.current}
579 state.lines, state.current = copy(record.lines), record.current
580 end
581 elseif kind == "h" then
582 self:emit("history undo=" .. #state.undo .. " redo=" .. #state.redo)
583 for i = #state.undo, 1, -1 do self:emit("undo " .. (#state.undo - i + 1) .. ": " .. state.undo[i].command) end
584 for i = #state.redo, 1, -1 do self:emit("redo " .. (#state.redo - i + 1) .. ": " .. state.redo[i].command) end
585 elseif kind == "g" or kind == "v" then
586 local pattern, selected = self:resolve_pattern(cmd.pattern), {}
587 for i = a, b do
588 local matches = state.lines[i].text:find(pattern, 1, true) ~= nil
589 if matches == (kind == "g") then selected[#selected + 1] = state.lines[i].id end
590 end
591 if #selected == 0 then fail("no match") end
592 local changed = 0
593 for _, id in ipairs(selected) do
594 local index = self:locate(id)
595 if index ~= nil then
596 state.current = index
597 local first, last = self:range(cmd.subcommand.kind, {count = 0, left = {}})
598 changed = changed + self:execute(cmd.subcommand, first, last, text, true)
599 end
600 end
601 if cmd.subcommand.kind == "s" and changed == 0 then fail("no match") end
602 elseif kind == "write" or kind == "W" then
603 local contents = self:contents(a, b)
604 if kind == "W" and state.files[cmd.name] ~= nil then
605 local combined = copy(state.files[cmd.name])
606 for _, line in ipairs(contents) do combined[#combined + 1] = line end
607 contents = combined
608 end
609 state.files[cmd.name] = contents
610 self:emit("file " .. cmd.name .. " lines=" .. #contents .. " bytes=" .. bytes(contents))
611 elseif kind == "read" or kind == "e" then
612 local contents = state.files[cmd.name]
613 if contents == nil then fail("no such file") end
614 local entered = self:new_lines(contents)
615 if kind == "e" then state.lines, state.current = entered, #entered
616 else self:splice(a, 0, entered); state.current = a + #entered end
617 self:emit("file " .. cmd.name .. " lines=" .. #contents .. " bytes=" .. bytes(contents))
618 elseif kind == "f" then
619 local names, parts = {}, {}
620 for name in pairs(state.files) do names[#names + 1] = name end
621 table.sort(names, function(left, right) return ascii_compare(left, right) < 0 end)
622 for _, name in ipairs(names) do parts[#parts + 1] = name .. "=" .. #state.files[name] end
623 self:emit("files " .. (#parts == 0 and "none" or table.concat(parts, " ")))
624 end
625 if cmd.suffix ~= "" and state.current >= 1 then self:print_line(state.current, cmd.suffix) end
626 return 0
627end
628
629local state = {lines = {}, current = 0, marks = {}, cut = {}, display_count = 5,
630 undo = {}, redo = {}, files = {}, next_id = 1}
631
632while true do
633 local line = io.read("l")
634 if line == nil or line == "Q" then break end
635 local text = {}
636 if has_text(line) then
637 while true do
638 local content = io.read("l")
639 if content == "." then break end
640 text[#text + 1] = content
641 end
642 end
643 local ok, result = pcall(function()
644 local cursor = Cursor.new(line)
645 local address = address_form(cursor:address())
646 local cmd = parse_command(cursor:rest())
647 if no_address[cmd.kind] and address.count > 0 then fail("unexpected address") end
648 if one_address[cmd.kind] and address.count == 2 then fail("too many addresses") end
649 local next_state = copy_state(state)
650 local editor = Editor.new(next_state)
651 local a, b = editor:range(cmd.kind, address)
652 editor:execute(cmd, a, b, text, false)
653 if #next_state.lines > 200 then fail("buffer full") end
654 if modifies(cmd) then
655 next_state.undo[#next_state.undo + 1] = {command = line, lines = copy(state.lines), current = state.current}
656 next_state.redo = {}
657 end
658 return {state = next_state, output = editor.output}
659 end)
660 if ok then
661 state = result.state
662 for _, output in ipairs(result.output) do io.write(output, "\n") end
663 elseif type(result) == "table" and result.command_error then io.write("? ", result.reason, "\n")
664 else error(result, 0) end
665end
