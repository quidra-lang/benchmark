20[[noreturn]] void fail(const std::string& reason) { throw EditorError(reason); }
21bool digit(char ch) { return ch >= '0' && ch <= '9'; }
22bool lower(char ch) { return ch >= 'a' && ch <= 'z'; }
23bool upper(char ch) { return ch >= 'A' && ch <= 'Z'; }
24bool contains(std::string_view choices, char ch) { return choices.find(ch) != std::string_view::npos; }
25bool digits(const std::string& text) { return std::all_of(text.begin(), text.end(), digit); }

31struct Delimited {
32 std::string text;
33 Replacement replacement;
34 std::string raw;
35 std::size_t next;
36};
37
38Delimited readDelimited(const std::string& source, std::size_t start, char delimiter) {
39 std::string text;
40 Replacement replacement;
41 for (std::size_t i = start; i < source.size(); ++i) {
42 char ch = source[i];
43 if (ch == '\\') {
44 if (++i == source.size()) fail("missing delimiter");
45 ch = source[i];
46 text += ch;
47 replacement.emplace_back(ch);
48 } else if (ch == delimiter) {
49 return {text, replacement, source.substr(start, i - start), i + 1};
50 } else {
51 text += ch;
52 replacement.push_back(ch == '&' ? std::nullopt : std::optional<char>(ch));
53 }
54 }
55 fail("missing delimiter");
56}

71AddressScan scanAddress(const std::string& source) {
72 Address elements;
73 std::size_t i = 0;
74 while (i < source.size()) {
75 const char ch = source[i];
76 if (digit(ch) || ch == '+' || ch == '-') {
77 const std::size_t start = i++;
78 while (i < source.size() && digit(source[i])) ++i;
79 elements.push_back({digit(ch) ? ElementKind::number : ElementKind::offset, source.substr(start, i - start)});
80 } else if (contains(".$,;%", ch)) {
81 ElementKind kind = ElementKind::all;
82 if (ch == '.') kind = ElementKind::current;
83 else if (ch == '$') kind = ElementKind::last;
84 else if (ch == ',') kind = ElementKind::comma;
85 else if (ch == ';') kind = ElementKind::semicolon;
86 elements.push_back({kind, ""});
87 ++i;
88 } else if (ch == '\'') {
89 ++i;
90 elements.push_back({ElementKind::mark, i < source.size() ? source.substr(i++, 1) : ""});
91 } else if (ch == '/' || ch == '?') {
92 const Delimited found = readDelimited(source, i + 1, ch);
93 elements.push_back({ch == '/' ? ElementKind::forward : ElementKind::backward, found.text});
94 i = found.next;
95 } else break;
96 }
97 return {elements, source.substr(i)};
98}

100bool singleAddress(const Address& elements) {
101 if (elements.empty()) return false;
102 for (std::size_t i = 0; i < elements.size(); ++i) {
103 const ElementKind kind = elements[i].kind;
104 if (kind == ElementKind::offset) continue;
105 if (i != 0 || kind == ElementKind::comma || kind == ElementKind::semicolon || kind == ElementKind::all) return false;
106 }
107 return true;
108}

110struct AddressForm {
111 int count = 0;
112 Address left, right;
113 ElementKind separator = ElementKind::comma;
114};
115
116AddressForm addressForm(const Address& elements) {
117 if (elements.empty()) return {};
118 if (elements.size() == 1 && elements[0].kind == ElementKind::all) {
119 return {2, {{ElementKind::number, "1"}}, {{ElementKind::last, ""}}, ElementKind::comma};
120 }
121 std::optional<std::size_t> separator;
122 for (std::size_t i = 0; i < elements.size(); ++i) {
123 if (elements[i].kind != ElementKind::comma && elements[i].kind != ElementKind::semicolon) continue;
124 if (separator) fail("bad address");
125 separator = i;
126 }
127 if (!separator) {
128 if (!singleAddress(elements)) fail("bad address");
129 return {1, elements, {}, ElementKind::comma};
130 }
131 Address left(elements.begin(), elements.begin() + *separator);
132 Address right(elements.begin() + *separator + 1, elements.end());
133 if ((!left.empty() && !singleAddress(left)) || (!right.empty() && !singleAddress(right))) fail("bad address");
134 return {2, left, right, elements[*separator].kind};
135}

170std::string fileName(const std::string& argument) {
171 if (argument.size() < 2 || argument.size() > 9 || argument[0] != ' '
172 || !std::all_of(argument.begin() + 1, argument.end(), [](char ch) { return lower(ch) || digit(ch); })) {
173 fail("invalid file name");
174 }
175 return argument.substr(1);
176}

178Command parseSubstitution(const std::string& argument) {
179 Command command(Kind::substitute);
180 std::string flags = argument;
181 if (!std::all_of(argument.begin(), argument.end(), [](char ch) { return digit(ch) || contains("gpnl", ch); })) {
182 const char delimiter = argument[0];
183 if (delimiter == ' ' || delimiter == '\\') fail("missing delimiter");
184 const Delimited search = readDelimited(argument, 1, delimiter);
185 const Delimited replacement = readDelimited(argument, search.next, delimiter);
186 command.pattern = search.text;
187 if (replacement.raw != "%") command.replacement = replacement.replacement;
188 flags = argument.substr(replacement.next);
189 }
190 bool counted = false;
191 for (std::size_t i = 0; i < flags.size();) {
192 const char ch = flags[i++];
193 if (ch == 'g') {
194 if (command.allMatches) fail("invalid suffix");
195 command.allMatches = true;
196 } else if (contains("pnl", ch)) {
197 if (command.suffix != '\0') fail("invalid suffix");
198 command.suffix = ch;
199 } else if (digit(ch)) {
200 if (counted) fail("invalid suffix");
201 const std::size_t start = i - 1;
202 while (i < flags.size() && digit(flags[i])) ++i;
203 command.occurrence = std::stoi(flags.substr(start, i - start));
204 if (command.occurrence == 0) fail("invalid suffix");
205 counted = true;
206 } else fail("invalid suffix");
207 }
208 return command;
209}

211bool sortForm(const std::string& source) { return source == "sort" || source.compare(0, 5, "sort ") == 0; }

213Command parseCommand(const std::string& source) {
214 if (sortForm(source)) {
215 Command command(Kind::sort);
216 command.argument = source == "sort" ? "" : source.substr(5);
217 if (source != "sort") {
218 std::string seen;
219 if (command.argument.empty()) fail("invalid suffix");
220 for (char ch : command.argument) {
221 if (!contains("ruin", ch) || contains(seen, ch)) fail("invalid suffix");
222 seen += ch;
223 }
224 if (contains(seen, 'i') && contains(seen, 'n')) fail("invalid suffix");
225 }
226 return command;
227 }
228 const char ch = source.empty() ? '\0' : source[0];
229 const std::string argument = source.empty() ? "" : source.substr(1);
230 static const std::map<char, Kind> plain = {
231 {'\0', Kind::bare}, {'a', Kind::append}, {'i', Kind::insert}, {'c', Kind::change},
232 {'p', Kind::print}, {'n', Kind::numbered}, {'l', Kind::list}, {'=', Kind::position},
233 {'#', Kind::count}, {'y', Kind::yank}, {'K', Kind::marks}, {'h', Kind::history}, {'f', Kind::files}
234 };
235 const auto plainCommand = plain.find(ch);
236 if (plainCommand != plain.end()) {
237 if (!argument.empty()) fail("invalid suffix");
238 return Command(plainCommand->second);
239 }
240 if (contains("dxjJ", ch)) {
241 Command command(ch == 'd' ? Kind::erase : ch == 'x' ? Kind::put : ch == 'j' ? Kind::join : Kind::joinWords);
242 command.suffix = printSuffix(argument);
243 return command;
244 }
245 if (contains("z><u", ch)) {
246 Command command(ch == 'z' ? Kind::scroll : ch == '>' ? Kind::indent : ch == '<' ? Kind::unindent : Kind::undo);
247 command.amount = positiveArgument(argument);
248 return command;
249 }
250 if (ch == '~') {
251 if (!argument.empty() && argument != "u" && argument != "l") fail("invalid suffix");
252 Command command(Kind::letterCase);
253 command.argument = argument;
254 return command;
255 }
256 if (ch == 'r' && digits(argument)) {
257 Command command(Kind::redo);
258 command.amount = positiveArgument(argument);
259 return command;
260 }
261 if (ch == 'w' && argument.empty()) return Command(Kind::writePrint);
262 if (contains("rwWe", ch)) {
263 Command command(ch == 'r' ? Kind::read : ch == 'w' ? Kind::write : ch == 'W' ? Kind::writeAppend : Kind::edit);
264 command.argument = fileName(argument);
265 return command;
266 }
267 if (ch == 'k') {
268 if (argument.size() != 1 || !lower(argument[0])) fail("invalid mark");
269 Command command(Kind::mark);
270 command.argument = argument;
271 return command;
272 }
273 if (ch == 'm' || ch == 't') {
274 const AddressScan destination = scanAddress(argument);
275 if (!singleAddress(destination.elements)) fail("invalid destination");
276 Command command(ch == 'm' ? Kind::move : Kind::copy);
277 command.destination = destination.elements;
278 command.suffix = printSuffix(destination.command);
279 return command;
280 }
281 if (ch == 's') return parseSubstitution(argument);
282 if (ch == 'g' || ch == 'v') {
283 if (argument.empty() || argument[0] == ' ' || argument[0] == '\\') fail("missing delimiter");
284 const Delimited search = readDelimited(argument, 1, argument[0]);
285 const std::string subcommand = search.next == argument.size() ? "p" : argument.substr(search.next);
286 if (!contains("pnldmtjJs><~aicx", subcommand[0]) || sortForm(subcommand)) fail("invalid global command");
287 Command command(ch == 'g' ? Kind::global : Kind::inverseGlobal);
288 command.pattern = search.text;
289 command.subcommand = std::make_unique<Command>(parseCommand(subcommand));
290 return command;
291 }
292 fail("unknown command");
293}

295bool hasText(const std::string& source) {
296 try {
297 const std::string command = scanAddress(source).command;
298 if (command.size() == 1 && contains("aic", command[0])) return true;
299 if (command.size() < 2 || !contains("gv", command[0]) || command[1] == ' ' || command[1] == '\\') return false;
300 const Delimited search = readDelimited(command, 2, command[1]);
301 return search.next + 1 == command.size() && contains("aic", command[search.next]);
302 } catch (const EditorError&) { return false; }
303}

305int addressKind(Kind kind) {
306 switch (kind) {
307 case Kind::undo: case Kind::redo: case Kind::history: case Kind::marks: case Kind::files: case Kind::edit:
308 return 0;
309 case Kind::append: case Kind::insert: case Kind::put: case Kind::position: case Kind::read: case Kind::mark: case Kind::scroll:
310 return 1;
311 default: return 2;
312 }
313}
314
315bool modifying(const Command& command) {
316 switch (command.kind) {
317 case Kind::global: case Kind::inverseGlobal:
318 return command.subcommand->kind != Kind::print && command.subcommand->kind != Kind::numbered && command.subcommand->kind != Kind::list;
319 case Kind::append: case Kind::insert: case Kind::change: case Kind::erase: case Kind::put:
320 case Kind::join: case Kind::joinWords: case Kind::move: case Kind::copy: case Kind::substitute:
321 case Kind::indent: case Kind::unindent: case Kind::letterCase: case Kind::sort: case Kind::read: case Kind::edit:
322 return true;
323 default: return false;
324 }
325}

397 int address(const Address& elements) {
398 int value = state.current;
399 std::size_t firstOffset = 0;
400 const Element& base = elements[0];
401 if (base.kind != ElementKind::offset) {
402 firstOffset = 1;
403 if (base.kind == ElementKind::number) value = std::stoi(base.word);
404 else if (base.kind == ElementKind::current) value = state.current;
405 else if (base.kind == ElementKind::last) value = last();
406 else if (base.kind == ElementKind::mark) {
407 if (base.word.size() != 1 || !lower(base.word[0])) fail("invalid mark");
408 value = lineNumber(state.marks[base.word[0] - 'a']);
409 if (value == 0) fail("mark not set");
410 } else {
411 const std::string search = pattern(base.word);
412 value = 0;
413 for (int step = 1; step <= last(); ++step) {
414 const int pivot = state.current == 0 && base.kind == ElementKind::backward ? last() + 1 : state.current;
415 const int index = base.kind == ElementKind::forward
416 ? (pivot + step - 1) % last() : (pivot - step - 1 + last()) % last();
417 if (state.lines[index].text.find(search) != std::string::npos) { value = index + 1; break; }
418 }
419 if (value == 0) fail("no match");
420 }
421 }
422 for (std::size_t i = firstOffset; i < elements.size(); ++i) {
423 const std::string& word = elements[i].word;
424 const int magnitude = word.size() == 1 ? 1 : std::stoi(word.substr(1));
425 value += word[0] == '+' ? magnitude : -magnitude;
426 }
427 return value;
428 }

430 std::vector<int> addresses(const AddressForm& form) {
431 if (form.count == 0) return {};
432 if (form.count == 1) return {address(form.left)};
433 const int left = form.left.empty() ? (form.separator == ElementKind::comma ? 1 : state.current) : address(form.left);
434 if (form.separator == ElementKind::semicolon) {
435 if (left < 0 || left > last()) fail("address out of range");
436 state.current = left;
437 }
438 return {left, form.right.empty() ? last() : address(form.right)};
439 }

441 std::pair<int, int> range(const Command& command, const std::vector<int>& values) const {
442 const Kind kind = command.kind;
443 const int category = addressKind(kind);
444 if (category == 0) return {0, 0};
445 if (category == 1) {
446 const int fallback = kind == Kind::position || kind == Kind::read ? last() : state.current + (kind == Kind::scroll ? 1 : 0);
447 const int a = values.empty() ? fallback : values[0];
448 const int minimum = kind == Kind::mark || kind == Kind::scroll ? 1 : 0;
449 if (a < minimum || a > last()) fail("address out of range");
450 return {a, a};
451 }
452 int a, b;
453 if (!values.empty()) { a = values.front(); b = values.back(); }
454 else if (kind == Kind::writePrint || kind == Kind::write || kind == Kind::writeAppend || kind == Kind::count) return {1, last()};
455 else if (kind == Kind::global || kind == Kind::inverseGlobal || kind == Kind::sort) { a = 1; b = last(); }
456 else { a = state.current; b = state.current + (kind == Kind::join || kind == Kind::joinWords ? 1 : 0); }
457 if (a < 1 || b < 1 || a > last() || b > last()) fail("address out of range");
458 if (a > b || (a == b && (kind == Kind::join || kind == Kind::joinWords))) fail("invalid range");
459 return {a, b};
460 }

462 void print(int number, char form) {
463 const std::string& text = state.lines[number - 1].text;
464 if (form == 'p') emit(text);
465 else if (form == 'n') {
466 const std::string label = std::to_string(number);
467 emit(std::string(label.size() < 3 ? 3 - label.size() : 0, ' ') + label + " " + text);
468 } else {
469 std::string escaped;
470 for (char ch : text) { escaped += ch; if (ch == '\\') escaped += ch; }
471 escaped += '$';
472 while (escaped.size() > 40) {
473 emit(escaped.substr(0, 39) + "\\");
474 escaped.erase(0, 39);
475 }
476 emit(escaped);
477 }
478 }

484 int substitute(const Command& command, int a, int b, bool insideGlobal) {
485 const std::string search = pattern(command.pattern);
486 const std::optional<Replacement> replacement = command.replacement ? command.replacement : state.replacement;
487 if (!replacement) fail("no previous substitution");
488 state.replacement = replacement;
489 std::string expanded;
490 for (const std::optional<char>& part : *replacement) expanded += part ? std::string(1, *part) : search;
491 int changed = 0;
492 for (int i = a - 1; i < b; ++i) {
493 Line& line = state.lines[i];
494 std::size_t position = 0;
495 int occurrence = 0;
496 std::string result;
497 bool substituted = false;
498 for (;;) {
499 const std::size_t found = line.text.find(search, position);
500 if (found == std::string::npos) break;
501 ++occurrence;
502 const bool replace = occurrence == command.occurrence || (command.allMatches && occurrence > command.occurrence);
503 result += line.text.substr(position, found - position) + (replace ? expanded : search);
504 substituted = substituted || replace;
505 position = found + search.size();
506 }
507 if (!substituted) continue;
508 result += line.text.substr(position);
509 if (result.size() > 80) fail("line too long");
510 if (result.empty() || result.back() == ' ') fail("bad line");
511 line.text = std::move(result);
512 state.current = i + 1;
513 ++changed;
514 }
515 if (changed == 0 && !insideGlobal) fail("no match");
516 if (changed > 0) suffix(command);
517 return changed;
518 }

520 void global(const Command& command, int a, int b, const Text& text) {
521 const std::string search = pattern(command.pattern);
522 std::vector<std::uint64_t> selected;
523 for (int i = a - 1; i < b; ++i) {
524 const Line& line = state.lines[i];
525 if ((line.text.find(search) != std::string::npos) == (command.kind == Kind::global)) selected.push_back(line.id);
526 }
527 if (selected.empty()) fail("no match");
528 int changed = 0;
529 for (std::uint64_t id : selected) {
530 const int number = lineNumber(id);
531 if (number == 0) continue;
532 state.current = number;
533 const auto [start, end] = range(*command.subcommand, {});
534 changed += execute(*command.subcommand, start, end, text, true);
535 }
536 if (command.subcommand->kind == Kind::substitute && changed == 0) fail("no match");
537 }
538
539 int execute(const Command& command, int a, int b, const Text& text, bool insideGlobal = false) {
540 const Kind kind = command.kind;
541 if (kind == Kind::substitute) return substitute(command, a, b, insideGlobal);
542 if (kind == Kind::global || kind == Kind::inverseGlobal) global(command, a, b, text);
543 else if (kind == Kind::bare || kind == Kind::print || kind == Kind::numbered || kind == Kind::list) {
544 const char form = kind == Kind::numbered ? 'n' : kind == Kind::list ? 'l' : 'p';
545 for (int i = kind == Kind::bare ? b : a; i <= b; ++i) print(i, form);
546 state.current = b;
547 } else if (kind == Kind::scroll) {
548 if (command.amount) state.displayCount = *command.amount;
549 const int end = std::min(a + state.displayCount - 1, last());
550 for (int i = a; i <= end; ++i) print(i, 'p');
551 state.current = end;
552 } else if (kind == Kind::position) emit(std::to_string(a));
553 else if (kind == Kind::count) {
554 const Text text = contents(a, b);
555 int words = 0, characters = 0;
556 for (const std::string& line : text) {
557 characters += static_cast<int>(line.size());
558 bool inWord = false;
559 for (char ch : line) {
560 if (ch != ' ' && !inWord) ++words;
561 inWord = ch != ' ';
562 }
563 }
564 emit("count lines=" + std::to_string(text.size()) + " words=" + std::to_string(words) + " chars=" + std::to_string(characters));
565 } else if (kind == Kind::writePrint) {
566 const Text text = contents(a, b);
567 for (const std::string& line : text) emit(line);
568 emit("lines=" + std::to_string(text.size()) + " bytes=" + std::to_string(byteCount(text)));
569 } else if (kind == Kind::append || kind == Kind::insert) {
570 const int position = kind == Kind::append ? a : std::max(a, 1) - 1;
571 const Buffer inserted = newLines(text);
572 replace(position, 0, inserted);
573 state.current = position + static_cast<int>(inserted.size());
574 } else if (kind == Kind::change || kind == Kind::erase) {
575 state.cut = contents(a, b);
576 const Buffer inserted = kind == Kind::change ? newLines(text) : Buffer{};
577 replace(a - 1, b - a + 1, inserted);
578 state.current = inserted.empty() ? std::min(a, last()) : a + static_cast<int>(inserted.size()) - 1;
579 suffix(command);
580 } else if (kind == Kind::yank) state.cut = contents(a, b);
581 else if (kind == Kind::put) {
582 if (state.cut.empty()) fail("nothing to put");
583 const Buffer inserted = newLines(state.cut);
584 replace(a, 0, inserted);
585 state.current = a + static_cast<int>(inserted.size());
586 suffix(command);
587 } else if (kind == Kind::move || kind == Kind::copy) {
588 const int destination = address(command.destination), count = b - a + 1;
589 if (destination < 0 || destination > last()) fail("address out of range");
590 if (kind == Kind::move) {
591 if (a <= destination && destination < b) fail("invalid destination");
592 const Buffer moved(state.lines.begin() + a - 1, state.lines.begin() + b);
593 replace(a - 1, count, {});
594 const int position = destination < a ? destination : destination - count;
595 replace(position, 0, moved);
596 state.current = position + count;
597 } else {
598 const Buffer copied = newLines(contents(a, b));
599 replace(destination, 0, copied);
600 state.current = destination + count;
601 }
602 suffix(command);
603 } else if (kind == Kind::join || kind == Kind::joinWords) {
604 std::string joined = state.lines[a - 1].text;
605 for (int i = a; i < b; ++i) {
606 const std::string& line = state.lines[i].text;
607 if (kind == Kind::join) joined += line;
608 else joined += " " + line.substr(line.find_first_not_of(' '));
609 }
610 if (joined.size() > 80) fail("line too long");
611 const Buffer joinedLine = {{state.lines[a - 1].id, joined}};
612 replace(a - 1, b - a + 1, joinedLine);
613 state.current = a;
614 suffix(command);
615 } else if (kind == Kind::indent || kind == Kind::unindent || kind == Kind::letterCase) {
616 const int width = command.amount.value_or(4);
617 for (int i = a - 1; i < b; ++i) {
618 std::string& value = state.lines[i].text;
619 if (kind == Kind::indent) {
620 if (value.size() + width > 80) fail("line too long");
621 value.insert(0, width, ' ');
622 } else if (kind == Kind::unindent) {
623 const std::size_t spaces = value.find_first_not_of(' ');
624 value.erase(0, std::min(static_cast<std::size_t>(width), spaces));
625 } else {
626 for (char& ch : value) {
627 if (lower(ch) && command.argument != "l") ch = static_cast<char>(ch - ('a' - 'A'));
628 else if (upper(ch) && command.argument != "u") ch = lowercase(ch);
629 }
630 }
631 }
632 state.current = b;
633 } else if (kind == Kind::sort) {
634 struct KeyedLine { Line line; std::variant<int, std::string> key; };
635 std::vector<KeyedLine> sorted;
636 const bool numeric = contains(command.argument, 'n');
637 for (int i = a - 1; i < b; ++i) {
638 const Line& line = state.lines[i];
639 if (numeric) {
640 const std::size_t start = line.text[0] == '-' ? 1 : 0;
641 std::size_t end = start;
642 while (end < line.text.size() && digit(line.text[end])) ++end;
643 sorted.push_back({line, end == start ? 0 : std::stoi(line.text.substr(0, end))});
644 } else {
645 std::string key = line.text;
646 if (contains(command.argument, 'i')) std::transform(key.begin(), key.end(), key.begin(), lowercase);
647 sorted.push_back({line, key});
648 }
649 }
650 const bool reverse = contains(command.argument, 'r');
651 std::stable_sort(sorted.begin(), sorted.end(), [reverse](const KeyedLine& x, const KeyedLine& y) {
652 return reverse ? y.key < x.key : x.key < y.key;
653 });
654 if (contains(command.argument, 'u')) {
655 sorted.erase(std::unique(sorted.begin(), sorted.end(), [](const KeyedLine& x, const KeyedLine& y) {
656 return x.key == y.key;
657 }), sorted.end());
658 }
659 Buffer lines;
660 for (const KeyedLine& item : sorted) lines.push_back(item.line);
661 replace(a - 1, b - a + 1, lines);
662 state.current = a + static_cast<int>(lines.size()) - 1;
663 } else if (kind == Kind::mark) state.marks[command.argument[0] - 'a'] = state.lines[a - 1].id;
664 else if (kind == Kind::marks) {
665 std::string text = "marks";
666 for (int i = 0; i < 26; ++i) {
667 const int number = lineNumber(state.marks[i]);
668 if (number > 0) text += " " + std::string(1, static_cast<char>('a' + i)) + "=" + std::to_string(number);
669 }
670 emit(text == "marks" ? "marks none" : text);
671 } else if (kind == Kind::undo || kind == Kind::redo) {
672 std::vector<Record>& source = kind == Kind::undo ? state.undo : state.redo;
673 std::vector<Record>& target = kind == Kind::undo ? state.redo : state.undo;
674 const int count = command.amount.value_or(1);
675 if (source.size() < static_cast<std::size_t>(count)) fail(kind == Kind::undo ? "nothing to undo" : "nothing to redo");
676 for (int i = 0; i < count; ++i) {
677 const Record record = source.back();
678 source.pop_back();
679 target.push_back(snapshot(record.command));
680 state.lines = *record.lines;
681 state.current = record.current;
682 }
683 } else if (kind == Kind::history) {
684 emit("history undo=" + std::to_string(state.undo.size()) + " redo=" + std::to_string(state.redo.size()));
685 for (const std::string name : {"undo", "redo"}) {
686 const std::vector<Record>& history = name == "undo" ? state.undo : state.redo;
687 int number = 0;
688 for (auto it = history.rbegin(); it != history.rend(); ++it) emit(name + " " + std::to_string(++number) + ": " + it->command);
689 }
690 } else if (kind == Kind::write || kind == Kind::writeAppend || kind == Kind::read || kind == Kind::edit) {
691 const auto found = state.files.find(command.argument);
692 std::shared_ptr<const Text> file;
693 if (kind == Kind::write || kind == Kind::writeAppend) {
694 Text written = kind == Kind::writeAppend && found != state.files.end() ? *found->second : Text{};
695 const Text added = contents(a, b);
696 written.insert(written.end(), added.begin(), added.end());
697 file = std::make_shared<const Text>(std::move(written));
698 state.files[command.argument] = file;
699 } else {
700 if (found == state.files.end()) fail("no such file");
701 file = found->second;
702 const Buffer inserted = newLines(*file);
703 if (kind == Kind::edit) { state.lines = inserted; state.current = last(); }
704 else { replace(a, 0, inserted); state.current = a + static_cast<int>(inserted.size()); }
705 }
706 emit("file " + command.argument + " lines=" + std::to_string(file->size()) + " bytes=" + std::to_string(byteCount(*file)));
707 } else if (kind == Kind::files) {
708 std::string text = "files";
709 for (const auto& [name, file] : state.files) text += " " + name + "=" + std::to_string(file->size());
710 emit(text == "files" ? "files none" : text);
711 }
712 return 0;
713 }
714
715public:
716 Text process(const std::string& source, const Text& text) {
717 State previous = state;
718 const Record before = snapshot(source);
719 output.clear();
720 try {
721 const AddressScan scanned = scanAddress(source);
722 const AddressForm form = addressForm(scanned.elements);
723 const Command command = parseCommand(scanned.command);
724 const int category = addressKind(command.kind);
725 if (category == 0 && form.count > 0) fail("unexpected address");
726 if (category == 1 && form.count > 1) fail("too many addresses");
727 const std::vector<int> values = addresses(form);
728 const auto [a, b] = range(command, values);
729 execute(command, a, b, text);
730 if (last() > 200) fail("buffer full");
731 if (modifying(command)) {
732 state.undo.push_back(before);
733 state.redo.clear();
734 }
735 return output;
736 } catch (const EditorError& error) {
737 state = std::move(previous);
738 return {"? " + std::string(error.what())};
739 }
740 }
741};

753 }

755 }
