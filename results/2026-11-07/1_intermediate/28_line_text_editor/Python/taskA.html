84def read_delimited(text, start, delimiter):
85 pieces = [""]
86 position = start
87 while position < len(text) and text[position] != delimiter:
88 char = text[position]
89 if char == "\\":
90 if position + 1 == len(text):
91 raise EditorError("missing delimiter")
92 pieces[-1] += text[position + 1]
93 position += 2
94 elif char == "&":
95 pieces.append("")
96 position += 1
97 else:
98 pieces[-1] += char
99 position += 1
100 if position == len(text):
101 raise EditorError("missing delimiter")
102 return pieces, position + 1

105def digits_end(text, start):
106 end = start
107 while end < len(text) and text[end].isdigit():
108 end += 1
109 return end

112def scan_elements(text):
113 elements = []
114 position = 0
115 while position < len(text):
116 char = text[position]
117 if char.isdigit():
118 end = digits_end(text, position)
119 elements.append(Element("number", int(text[position:end])))
120 elif char in "+-":
121 end = digits_end(text, position + 1)
122 amount = int(text[position + 1:end]) if end > position + 1 else 1
123 elements.append(Element("offset", amount if char == "+" else -amount))
124 elif char in ".$%":
125 end = position + 1
126 elements.append(Element(char))
127 elif char in ",;":
128 end = position + 1
129 elements.append(Element("separator", text=char))
130 elif char == "'":
131 end = min(position + 2, len(text))
132 elements.append(Element("mark", text=text[position + 1:end]))
133 elif char in "/?":
134 pieces, end = read_delimited(text, position + 1, char)
135 kind = "forward" if char == "/" else "backward"
136 elements.append(Element(kind, text="&".join(pieces)))
137 else:
138 break
139 position = end
140 return elements, position

143def is_single_address(elements):
144 if not elements:
145 return False
146 offsets = elements[1:] if elements[0].kind in BASE_KINDS else elements
147 return all(element.kind == "offset" for element in offsets)

150def address_form(elements):
151 if not elements:
152 return Addresses(0)
153 if len(elements) == 1 and elements[0].kind == "%":
154 return Addresses(2, [Element("number", 1)], ",", [Element("$")])
155 separators = [index for index, element in enumerate(elements) if element.kind == "separator"]
156 if len(separators) > 1:
157 raise EditorError("bad address")
158 if not separators:
159 if not is_single_address(elements):
160 raise EditorError("bad address")
161 return Addresses(1, elements)
162 index = separators[0]
163 left, right = elements[:index], elements[index + 1:]
164 if (left and not is_single_address(left)) or (right and not is_single_address(right)):
165 raise EditorError("bad address")
166 return Addresses(2, left, elements[index].text, right)

195def parse_sort(text):
196 command = Command("sort")
197 if text.startswith("sort "):
198 options = text[5:]
199 if (not options or any(option not in SORT_OPTIONS for option in options)
200 or len(set(options)) != len(options) or ("i" in options and "n" in options)):
201 raise EditorError("invalid suffix")
202 command.options = options
203 return command

206def parse_transfer(command, argument):
207 elements, position = scan_elements(argument)
208 if not is_single_address(elements):
209 raise EditorError("invalid destination")

214def parse_flags(command, flags):
215 has_count = False
216 position = 0
217 while position < len(flags):
218 char = flags[position]
219 if char == "g" and not command.every:
220 command.every = True
221 position += 1
222 elif char in FORMS and not command.suffix:
223 command.suffix = char
224 position += 1
225 elif char.isdigit() and not has_count:
226 end = digits_end(flags, position)
227 command.count = int(flags[position:end])
228 if command.count == 0:
229 raise EditorError("invalid suffix")
230 has_count = True
231 position = end
232 else:
233 raise EditorError("invalid suffix")

236def parse_substitute(command, argument):
237 if all(char in FLAG_CHARACTERS for char in argument):
238 flags = argument
239 else:
240 delimiter = argument[0]
241 if delimiter in (" ", "\\"):
242 raise EditorError("missing delimiter")
243 pieces, middle = read_delimited(argument, 1, delimiter)
244 replacement, end = read_delimited(argument, middle, delimiter)
245 command.search = "&".join(pieces)
246 if argument[middle:end - 1] != "%":
247 command.replacement = replacement
248 flags = argument[end:]
249 parse_flags(command, flags)

252def parse_global(command, argument):
253 if argument == "" or argument[0] in (" ", "\\"):
254 raise EditorError("missing delimiter")
255 pieces, end = read_delimited(argument, 1, argument[0])
256 command.search = "&".join(pieces)
257 text = argument[end:] or "p"
258 if text[0] not in GLOBAL_SUBCOMMANDS or is_sort_form(text):
259 raise EditorError("invalid global command")
260 command.subcommand = parse_command(text)

263def parse_command(text):
264 if is_sort_form(text):
265 return parse_sort(text)
266 name, argument = text[:1], text[1:]
267 if name == "r":
268 name = "redo" if argument == "" or argument.isdigit() else "read"
269 elif name == "w" and argument:
270 name = "write"
271 command = Command(name)
272 if name in PLAIN_COMMANDS:
273 if argument:
274 raise EditorError("invalid suffix")
275 elif name in SUFFIX_COMMANDS:
276 command.suffix = print_suffix(argument)
277 elif name in COUNT_COMMANDS:
278 command.number = count_argument(argument)
279 elif name == "~":
280 if argument not in ("", "u", "l"):
281 raise EditorError("invalid suffix")
282 command.options = argument
283 elif name in FILE_COMMANDS:
284 command.file_name = file_name(argument)
285 elif name == "k":
286 if len(argument) != 1 or argument not in string.ascii_lowercase:
287 raise EditorError("invalid mark")
288 command.mark = argument
289 elif name in ("m", "t"):
290 parse_transfer(command, argument)
291 elif name == "s":
292 parse_substitute(command, argument)
293 elif name in ("g", "v"):
294 parse_global(command, argument)
295 else:
296 raise EditorError("unknown command")
297 return command

300def has_text(line):
301 try:
302 _, position = scan_elements(line)
303 command = line[position:]
304 if command in TEXT_COMMANDS:
305 return True
306 if len(command) < 2 or command[0] not in ("g", "v") or command[1] in (" ", "\\"):
307 return False
308 _, end = read_delimited(command, 2, command[1])
309 except EditorError:
310 return False
311 return command[end:] in TEXT_COMMANDS

314def read_commands(lines):
315 commands = []
316 index = 0
317 while index < len(lines) and lines[index] != "Q":
318 line = lines[index]
319 index += 1
320 text = []
321 if has_text(line):
322 while index < len(lines) and lines[index] != ".":
323 text.append(lines[index])
324 index += 1
325 index += 1
326 commands.append((line, text))
327 return commands

334def leading_integer(text):
335 sign = 1
336 start = 0
337 if text.startswith("-"):
338 sign = -1
339 start = 1
340 end = digits_end(text, start)
341 if end == start:
342 return 0
343 return sign * int(text[start:end])

354def substitute_text(text, pattern, replacement, count, every):
355 result = ""
356 position = 0
357 found = 0
358 index = text.find(pattern)
359 while index >= 0:
360 found += 1
361 if found >= count:
362 result += text[position:index] + replacement
363 else:
364 result += text[position:index + len(pattern)]
365 position = index + len(pattern)
366 if found >= count and not every:
367 break
368 index = text.find(pattern, position)
369 if found < count:
370 return None
371 return result + text[position:]

429 def show(self, number, form):
430 text = self.lines[number - 1].text
431 if form == "p":

446 def use_pattern(self, text):
447 if text:
448 self.pattern = text
449 elif self.pattern is None:
450 raise EditorError("no previous pattern")

453 def search(self, text, forward):
454 pattern = self.use_pattern(text)
455 total = len(self.lines)
456 if forward:
457 order = list(range(self.current + 1, total + 1)) + list(range(1, self.current + 1))
458 else:
459 order = list(range(self.current - 1, 0, -1)) + list(range(total, self.current - 1, -1))
460 for number in order:
461 if number >= 1 and pattern in self.lines[number - 1].text:
462 return number
463 raise EditorError("no match")

466 if len(name) != 1 or name not in string.ascii_lowercase:

473 def base_value(self, element):
474 if element.kind == "number":
475 return element.value
476 if element.kind == ".":
477 return self.current
478 if element.kind == "$":
479 return len(self.lines)
480 if element.kind == "mark":
481 return self.mark_line(element.text)
482 return self.search(element.text, element.kind == "forward")

484 def address_value(self, elements):
485 value = self.current
486 offsets = elements
487 if elements[0].kind in BASE_KINDS:
488 value = self.base_value(elements[0])
489 offsets = elements[1:]
490 for element in offsets:
491 value += element.value
492 return value

494 def address_values(self, addresses):
495 if addresses.count == 0:
496 return []
497 if addresses.count == 1:
498 return [self.address_value(addresses.left)]
499 if addresses.left:
500 left = self.address_value(addresses.left)
501 elif addresses.separator == ",":
502 left = 1
503 else:
504 left = self.current
505 if addresses.separator == ";":
506 if not 0 <= left <= len(self.lines):
507 raise EditorError("address out of range")
508 self.current = left
509 right = self.address_value(addresses.right) if addresses.right else len(self.lines)
510 return [left, right]

512 def command_range(self, command, values):
513 name = command.name
514 total = len(self.lines)
515 if name in NO_ADDRESS_COMMANDS:
516 return 0, 0
517 if name in ONE_ADDRESS_COMMANDS:
518 if values:
519 value = values[0]
520 elif name in ("=", "read"):
521 value = total
522 elif name == "z":
523 value = self.current + 1
524 else:
525 value = self.current
526 lowest = 1 if name in ("k", "z") else 0
527 if not lowest <= value <= total:
528 raise EditorError("address out of range")
529 return value, value
530 if values:
531 first, last = values[0], values[-1]
532 elif name in ("g", "v", "sort"):
533 first, last = 1, total
534 elif name in WHOLE_BUFFER_COMMANDS:
535 return 1, total
536 elif name in ("j", "J"):
537 first, last = self.current, self.current + 1
538 else:
539 first, last = self.current, self.current
540 if not (1 <= first <= total and 1 <= last <= total):
541 raise EditorError("address out of range")
542 if first > last or (first == last and name in ("j", "J")):
543 raise EditorError("invalid range")
544 return first, last

546 def process(self, line, text):
547 before = Record(line, list(self.lines), self.current)
548 elements, position = scan_elements(line)
549 addresses = address_form(elements)
550 command = parse_command(line[position:])
551 if command.name in NO_ADDRESS_COMMANDS and addresses.count > 0:
552 raise EditorError("unexpected address")
553 if command.name in ONE_ADDRESS_COMMANDS and addresses.count == 2:
554 raise EditorError("too many addresses")
555 values = self.address_values(addresses)
556 first, last = self.command_range(command, values)
557 self.run(command, first, last, text)
558 if len(self.lines) > MAX_LINES:
559 raise EditorError("buffer full")
560 if command.is_modifying():
561 self.undo_history.append(before)
562 self.redo_history = []

676 def move_lines(self, command, first, last, text):
677 target = self.destination(command)
678 if first <= target < last:
679 raise EditorError("invalid destination")
680 block = self.lines[first - 1:last]
681 del self.lines[first - 1:last]
682 after = target if target < first else target - len(block)
683 self.lines[after:after] = block
684 self.current = after + len(block)

687 def copy_lines(self, command, first, last, text):

693 def join_lines(self, command, first, last, text):
694 texts = self.texts(first, last)
695 if command.name == "J":
696 joined = " ".join([texts[0]] + [line.lstrip(" ") for line in texts[1:]])
697 else:
698 joined = "".join(texts)
699 if len(joined) > MAX_LINE_LENGTH:
700 raise EditorError("line too long")
701 self.set_text(first, joined)
702 del self.lines[first:last]
703 self.current = first
704 self.show_suffix(command.suffix)

706 def shift_width(self, command):

709 def indent_lines(self, command, first, last, text):
710 width = self.shift_width(command)
711 for number in range(first, last + 1):
712 shifted = " " * width + self.lines[number - 1].text
713 if len(shifted) > MAX_LINE_LENGTH:
714 raise EditorError("line too long")
715 self.set_text(number, shifted)
716 self.current = last

718 def unindent_lines(self, command, first, last, text):
719 width = self.shift_width(command)
720 for number in range(first, last + 1):
721 line = self.lines[number - 1].text
722 spaces = len(line) - len(line.lstrip(" "))
723 self.set_text(number, line[min(width, spaces):])
724 self.current = last

726 def convert_case(self, command, first, last, text):
727 for number in range(first, last + 1):
728 line = self.lines[number - 1].text
729 if command.options == "u":
730 line = line.upper()
731 elif command.options == "l":
732 line = line.lower()
733 else:
734 line = line.swapcase()
735 self.set_text(number, line)
736 self.current = last

738 def sort_lines(self, command, first, last, text):
739 options = command.options
740 block = self.lines[first - 1:last]
741 block.sort(key=lambda line: sort_key(line.text, options), reverse="r" in options)
742 if "u" in options:
743 kept = []
744 for line in block:
745 if not kept or sort_key(line.text, options) != sort_key(kept[-1].text, options):
746 kept.append(line)
747 block = kept
748 self.lines[first - 1:last] = block
749 self.current = first + len(block) - 1

751 def substitute_lines(self, command, first, last):
752 pattern = self.use_pattern(command.search)
753 if command.replacement is not None:
754 self.replacement = command.replacement
755 elif self.replacement is None:
756 raise EditorError("no previous substitution")
757 replacement = pattern.join(self.replacement)
758 substituted = 0
759 for number in range(first, last + 1):
760 result = substitute_text(
761 self.lines[number - 1].text, pattern, replacement, command.count, command.every
762 )
763 if result is None:
764 continue
765 if len(result) > MAX_LINE_LENGTH:
766 raise EditorError("line too long")
767 if result == "" or result.endswith(" "):
768 raise EditorError("bad line")
769 self.set_text(number, result)
770 self.current = number
771 substituted += 1
772 if substituted:
773 self.show_suffix(command.suffix)
774 return substituted

777 if self.substitute_lines(command, first, last) == 0:
778 raise EditorError("no match")

791 def travel(self, command, source, target, message):
792 steps = 1 if command.number is None else command.number
793 if len(source) < steps:
794 raise EditorError(message)
795 for _ in range(steps):
796 record = source.pop()
797 target.append(Record(record.command, list(self.lines), self.current))
798 self.lines = list(record.lines)
799 self.current = record.current

810 for number, record in enumerate(reversed(history), 1):
811 self.emit(f"{kind} {number}: {record.command}")

813 def global_command(self, command, first, last, text):
814 pattern = self.use_pattern(command.search)
815 wanted = command.name == "g"
816 selected = [
817 line.identity for line in self.lines[first - 1:last]
818 if (pattern in line.text) == wanted
819 ]
820 if not selected:
821 raise EditorError("no match")
822 subcommand = command.subcommand
823 substituted = 0
824 for identity in selected:
825 number = self.line_number(identity)
826 if number == 0:
827 continue
828 self.current = number
829 sub_first, sub_last = self.command_range(subcommand, [])
830 if subcommand.name == "s":
831 substituted += self.substitute_lines(subcommand, sub_first, sub_last)
832 else:
833 self.run(subcommand, sub_first, sub_last, text)
834 if subcommand.name == "s" and substituted == 0:
835 raise EditorError("no match")
