61function digits_end(text::String, position::Int)
62 while position <= lastindex(text) && digit(text[position])
63 position += 1
64 end
65 position

68function delimited(text::String, position::Int, delimiter::Char)
69 start = position
70 atoms = Atom[]
71 while position <= lastindex(text)
72 ch = text[position]
73 escaped = ch == '\\'
74 if escaped
75 position += 1
76 position <= lastindex(text) || fail("missing delimiter")
77 ch = text[position]
78 elseif ch == delimiter
79 return Delimited(join(atom.character for atom in atoms), atoms, part(text, start, position - 1), position + 1)
80 end
81 push!(atoms, Atom(ch, !escaped && ch == '&'))
82 position += 1
83 end
84 fail("missing delimiter")
85end

87function scan_address(text::String)
88 tokens = Token[]
89 position = 1
90 while position <= lastindex(text)
91 ch = text[position]
92 if digit(ch)
93 stop = digits_end(text, position)
94 token = Token(kind = NumberToken, value = parse(Int, part(text, position, stop - 1)))
95 position = stop
96 elseif ch == '+' || ch == '-'
97 stop = digits_end(text, position + 1)
98 magnitude = stop == position + 1 ? 1 : parse(Int, part(text, position + 1, stop - 1))
99 token = Token(kind = OffsetToken, value = ch == '+' ? magnitude : -magnitude)
100 position = stop
101 elseif ch in ".\$,;%"
102 kinds = (CurrentToken, LastToken, CommaToken, SemicolonToken, WholeToken)
103 token = Token(kind = kinds[findfirst(==(ch), ".\$,;%")])
104 position += 1
105 elseif ch == '\''
106 position += 1
107 name = position <= lastindex(text) ? text[position] : '\0'
108 token = Token(kind = MarkToken, name = name)
109 position += name == '\0' ? 0 : 1
110 elseif ch == '/' || ch == '?'
111 found = delimited(text, position + 1, ch)
112 token = Token(kind = ch == '/' ? ForwardToken : BackwardToken, search = found.text)
113 position = found.next
114 else
115 break
116 end
117 push!(tokens, token)
118 end
119 Scanned(tokens, part(text, position))
120end

122function valid_address(tokens::Vector{Token})
123 bases = (NumberToken, CurrentToken, LastToken, MarkToken, ForwardToken, BackwardToken)
124 all(token.kind == OffsetToken || index == 1 && token.kind in bases for (index, token) in enumerate(tokens))
125end

127function address_form(tokens::Vector{Token})
128 isempty(tokens) && return Addresses(count = 0)
129 if length(tokens) == 1 && tokens[1].kind == WholeToken
130 return Addresses(count = 2, left = [Token(kind = NumberToken, value = 1)], right = [Token(kind = LastToken)])
131 end
132 separators = findall(token -> token.kind in (CommaToken, SemicolonToken), tokens)
133 length(separators) <= 1 || fail("bad address")
134 if isempty(separators)
135 valid_address(tokens) || fail("bad address")
136 return Addresses(count = 1, left = tokens)
137 end
138 separator = separators[1]
139 left = tokens[1:separator-1]
140 right = tokens[separator+1:end]
141 valid_address(left) && valid_address(right) || fail("bad address")
142 Addresses(count = 2, left = left, right = right, semicolon = tokens[separator].kind == SemicolonToken)
143end

165function parse_substitution(argument::String)
166 command = Command(kind = SubstituteText, amount = 1)
167 flags = argument
168 if !all(ch -> ch in "gpnl" || digit(ch), argument)
169 delimiter = argument[1]
170 delimiter in " \\" && fail("missing delimiter")
171 search = delimited(argument, 2, delimiter)
172 replacement = delimited(argument, search.next, delimiter)
173 command.search = search.text
174 replacement.raw == "%" || (command.replacement = replacement.atoms)
175 flags = part(argument, replacement.next)
176 end
177 position = 1
178 counted = false
179 while position <= lastindex(flags)
180 ch = flags[position]
181 position += 1
182 if ch == 'g'
183 command.all && fail("invalid suffix")
184 command.all = true
185 elseif ch in "pnl"
186 isnothing(command.suffix) || fail("invalid suffix")
187 command.suffix = ch
188 elseif digit(ch)
189 counted && fail("invalid suffix")
190 counted = true
191 start = position - 1
192 position = digits_end(flags, position)
193 command.amount = parse(Int, part(flags, start, position - 1))
194 command.amount > 0 || fail("invalid suffix")
195 else
196 fail("invalid suffix")
197 end
198 end
199 command
200end

202function parse_command(text::String)
203 if sort_form(text)
204 options = text == "sort" ? "" : part(text, 6)
205 if text != "sort"
206 !isempty(options) && all(ch -> ch in "ruin", options) && length(Set(options)) == length(options) || fail("invalid suffix")
207 'i' in options && 'n' in options && fail("invalid suffix")
208 end
209 return Command(kind = SortText, options = options)
210 end
211 isempty(text) && return Command(kind = EmptyCommand)
212 first = text[1]
213 argument = part(text, 2)
214 if haskey(SIMPLE, first)
215 isempty(argument) || fail("invalid suffix")
216 return Command(kind = SIMPLE[first])
217 elseif first in "dxjJ"
218 kinds = (DeleteText, PutText, JoinText, JoinWords)
219 return Command(kind = kinds[findfirst(==(first), "dxjJ")], suffix = print_suffix(argument))
220 elseif first in "z><u"
221 kinds = (PageText, IndentText, OutdentText, UndoChange)
222 return Command(kind = kinds[findfirst(==(first), "z><u")], amount = positive(argument))
223 elseif first == '~'
224 argument in ("", "u", "l") || fail("invalid suffix")
225 return Command(kind = ConvertCase, option = isempty(argument) ? '\0' : argument[1])
226 elseif first == 'r'
227 all(digit, argument) && return Command(kind = RedoChange, amount = positive(argument))
228 return Command(kind = ReadFile, name = file_name(argument))
229 elseif first == 'w'
230 isempty(argument) && return Command(kind = PrintWrite)
231 return Command(kind = WriteFile, name = file_name(argument))
232 elseif first == 'W' || first == 'e'
233 return Command(kind = first == 'W' ? AppendFile : EditFile, name = file_name(argument))
234 elseif first == 'k'
235 length(argument) == 1 && lower(argument[1]) || fail("invalid mark")
236 return Command(kind = MarkLine, option = argument[1])
237 elseif first == 'm' || first == 't'
238 destination = scan_address(argument)
239 !isempty(destination.tokens) && valid_address(destination.tokens) || fail("invalid destination")
240 return Command(kind = first == 'm' ? MoveText : CopyText, destination = destination.tokens, suffix = print_suffix(destination.tail))
241 elseif first == 's'
242 return parse_substitution(argument)
243 elseif first == 'g' || first == 'v'
244 !isempty(argument) && !(argument[1] in " \\") || fail("missing delimiter")
245 search = delimited(argument, 2, argument[1])
246 body = part(argument, search.next)
247 isempty(body) && (body = "p")
248 body[1] in "pnldmtjJs><~aicx" && !sort_form(body) || fail("invalid global command")
249 return Command(kind = first == 'g' ? GlobalCommand : InverseCommand, search = search.text, sub = parse_command(body))
250 end
251 fail("unknown command")
252end

254function with_text(text::String)
255 try
256 tail = scan_address(text).tail
257 tail in ("a", "i", "c") && return true
258 length(tail) >= 2 && tail[1] in "gv" && !(tail[2] in " \\") || return false
259 search = delimited(tail, 3, tail[2])
260 part(tail, search.next) in ("a", "i", "c")
261 catch error
262 error isa CommandError || rethrow()
263 false
264 end
265end

316function address_value!(state::State, tokens::Vector{Token})
317 value = state.current
318 for token in tokens
319 if token.kind == OffsetToken
320 value += token.value
321 elseif token.kind == NumberToken
322 value = token.value
323 elseif token.kind == CurrentToken
324 value = state.current
325 elseif token.kind == LastToken
326 value = length(state.buffer)
327 elseif token.kind == MarkToken
328 lower(token.name) || fail("invalid mark")
329 value = locate(state, state.marks[Int(token.name - 'a') + 1])
330 value != 0 || fail("mark not set")
331 else
332 pattern = choose_pattern!(state, token.search)
333 size = length(state.buffer)
334 pivot = token.kind == BackwardToken && state.current == 0 ? size + 1 : state.current
335 value = 0
336 for step in 1:size
337 index = token.kind == ForwardToken ? mod(pivot + step - 1, size) + 1 : mod(pivot - step - 1, size) + 1
338 if occursin(pattern, state.buffer[index].text)
339 value = index
340 break
341 end
342 end
343 value != 0 || fail("no match")
344 end
345 end
346 value
347end

361function checked_range(state::State, kind::CommandKind, values::Vector{Int})
362 arity = address_kind(kind)
363 size = length(state.buffer)
364 arity == 0 && return LineRange(0, 0)
365 if arity == 1
366 first = isempty(values) ? (kind in (PrintEqual, ReadFile) ? size : state.current + (kind == PageText)) : values[1]
367 minimum = kind in (MarkLine, PageText) ? 1 : 0
368 minimum <= first <= size || fail("address out of range")
369 return LineRange(first, first)
370 end
371 if !isempty(values)
372 first = values[1]
373 last = length(values) == 1 ? first : values[2]
374 elseif kind in (PrintWrite, WriteFile, AppendFile, CountText)
375 return LineRange(1, size)
376 elseif kind in (GlobalCommand, InverseCommand, SortText)
377 first, last = 1, size
378 else
379 first = state.current
380 last = first + (kind in (JoinText, JoinWords))
381 end
382 1 <= first <= size && 1 <= last <= size || fail("address out of range")
383 first <= last && (first != last || !(kind in (JoinText, JoinWords))) || fail("invalid range")
384 LineRange(first, last)
385end

396function print_line!(output::Vector{String}, state::State, number::Int, form::Char)
397 text = state.buffer[number].text
398 if form == 'n'
399 push!(output, lpad(string(number), 3) * " " * text)
400 elseif form == 'l'
401 escaped = replace(text, "\\" => "\\\\") * "\$"
402 while length(escaped) > 40
403 push!(output, escaped[1:39] * "\\")
404 escaped = escaped[40:end]
405 end
406 push!(output, escaped)
407 else
408 push!(output, text)
409 end
410end

415byte_count(text::Vector{String}) = sum(line -> length(line) + 1, text; init = 0)
416word_count(text::String) = length(split(text, ' '; keepempty = false))
417leading_spaces(text::String) = length(text) - length(lstrip(text, ' '))
418function letter_case(text::String, option::Char)
419 option == 'u' && return uppercase(text)
420 option == 'l' && return lowercase(text)
421 join(lower(ch) ? uppercase(ch) : ('A' <= ch <= 'Z' ? lowercase(ch) : ch) for ch in text)
422end
423function leading_integer(text::String)
424 start = text[1] == '-' ? 2 : 1
425 stop = digits_end(text, start)
426 start == stop && return 0
427 magnitude = parse(Int, text[start:stop-1])
428 start == 2 ? -magnitude : magnitude
429end

431function substitute!(state::State, command::Command, range::LineRange, output::Vector{String}, inside::Bool)
432 pattern = choose_pattern!(state, command.search)
433 replacement = isnothing(command.replacement) ? state.replacement : command.replacement
434 isnothing(replacement) && fail("no previous substitution")
435 state.replacement = replacement
436 expanded = join(atom.matched ? pattern : string(atom.character) for atom in replacement)
437 changed = 0
438 for number in range.first:range.last
439 original = state.buffer[number].text
440 result = IOBuffer()
441 position, occurrence = 1, 0
442 replaced = false
443 while true
444 found = findnext(pattern, original, position)
445 isnothing(found) && break
446 print(result, part(original, position, first(found) - 1))
447 occurrence += 1
448 if occurrence == command.amount || command.all && occurrence > command.amount
449 print(result, expanded)
450 replaced = true
451 else
452 print(result, pattern)
453 end
454 position = last(found) + 1
455 end
456 if replaced
457 print(result, part(original, position))
458 text = String(take!(result))
459 length(text) <= 80 || fail("line too long")
460 !isempty(text) && !endswith(text, ' ') || fail("bad line")
461 state.buffer[number] = Line(state.buffer[number].id, text)
462 state.current = number
463 changed += 1
464 end
465 end
466 changed != 0 || inside || fail("no match")
467 changed > 0 && suffix_line!(output, state, command.suffix)
468 changed
469end

471function transfer_history!(state::State, redo::Bool, count::Int)
472 source = redo ? state.redo : state.undo
473 target = redo ? state.undo : state.redo
474 length(source) >= count || fail(redo ? "nothing to redo" : "nothing to undo")
475 for _ in 1:count
476 saved = pop!(source)
477 push!(target, snapshot(state, saved.source))
478 state.buffer = copy(saved.buffer)
479 state.current = saved.current
480 end
481end

503function execute!(state::State, command::Command, range::LineRange, text::Vector{String}, output::Vector{String}; inside::Bool = false)
504 kind = command.kind
505 a, b = range.first, range.last
506 if kind == EmptyCommand
507 print_line!(output, state, b, 'p')
508 state.current = b
509 elseif kind in (PrintText, PrintNumbers, PrintList)
510 form = kind == PrintText ? 'p' : (kind == PrintNumbers ? 'n' : 'l')
511 for number in a:b
512 print_line!(output, state, number, form)
513 end
514 state.current = b
515 elseif kind == PageText
516 command.amount == 0 || (state.display = command.amount)
517 stop = min(length(state.buffer), a + state.display - 1)
518 for number in a:stop
519 print_line!(output, state, number, 'p')
520 end
521 state.current = stop
522 elseif kind == PrintEqual
523 push!(output, string(a))
524 elseif kind == CountText
525 selected = contents(state, range)
526 words = sum(word_count, selected; init = 0)
527 chars = sum(length, selected; init = 0)
528 push!(output, "count lines=$(length(selected)) words=$words chars=$chars")
529 elseif kind == PrintWrite
530 selected = contents(state, range)
531 append!(output, selected)
532 push!(output, "lines=$(length(selected)) bytes=$(byte_count(selected))")
533 elseif kind in (AppendText, InsertText)
534 position = kind == AppendText ? a : max(a, 1) - 1
535 replace_lines!(state, position, 0, new_lines!(state, text))
536 state.current = position + length(text)
537 elseif kind in (ChangeText, DeleteText)
538 state.cut = contents(state, range)
539 inserted = kind == ChangeText ? new_lines!(state, text) : Line[]
540 replace_lines!(state, a - 1, b - a + 1, inserted)
541 state.current = isempty(inserted) ? min(a, length(state.buffer)) : a + length(inserted) - 1
542 elseif kind == YankText
543 state.cut = contents(state, range)
544 elseif kind == PutText
545 isempty(state.cut) && fail("nothing to put")
546 replace_lines!(state, a, 0, new_lines!(state, state.cut))
547 state.current = a + length(state.cut)
548 elseif kind in (MoveText, CopyText)
549 destination = address_value!(state, command.destination)
550 0 <= destination <= length(state.buffer) || fail("address out of range")
551 kind == MoveText && a <= destination < b && fail("invalid destination")
552 selected = state.buffer[a:b]
553 position = destination
554 if kind == MoveText
555 replace_lines!(state, a - 1, b - a + 1, Line[])
556 destination >= b && (position -= length(selected))
557 else
558 selected = new_lines!(state, [line.text for line in selected])
559 end
560 replace_lines!(state, position, 0, selected)
561 state.current = position + length(selected)
562 elseif kind in (JoinText, JoinWords)
563 selected = contents(state, range)
564 if kind == JoinWords
565 selected[2:end] = lstrip.(selected[2:end], ' ')
566 end
567 joined = join(selected, kind == JoinText ? "" : " ")
568 length(joined) <= 80 || fail("line too long")
569 replace_lines!(state, a - 1, b - a + 1, [Line(state.buffer[a].id, joined)])
570 state.current = a
571 elseif kind in (IndentText, OutdentText, ConvertCase)
572 width = command.amount == 0 ? 4 : command.amount
573 for number in a:b
574 original = state.buffer[number]
575 if kind == IndentText
576 length(original.text) + width <= 80 || fail("line too long")
577 changed = repeat(" ", width) * original.text
578 elseif kind == OutdentText
579 changed = part(original.text, min(width, leading_spaces(original.text)) + 1)
580 else
581 changed = letter_case(original.text, command.option)
582 end
583 state.buffer[number] = Line(original.id, changed)
584 end
585 state.current = b
586 elseif kind == SortText
587 selected = state.buffer[a:b]
588 sort_keys = 'n' in command.options ? leading_integer.(getfield.(selected, :text)) :
589 ('i' in command.options ? lowercase.(getfield.(selected, :text)) : getfield.(selected, :text))
590 ordered = sortperm(sort_keys; rev = 'r' in command.options)
591 if 'u' in command.options
592 ordered = [index for (position, index) in enumerate(ordered) if position == 1 || sort_keys[index] != sort_keys[ordered[position-1]]]
593 end
594 replace_lines!(state, a - 1, b - a + 1, selected[ordered])
595 state.current = a + length(ordered) - 1
596 elseif kind == SubstituteText
597 return substitute!(state, command, range, output, inside)
598 elseif kind == MarkLine
599 state.marks[Int(command.option - 'a') + 1] = state.buffer[a].id
600 elseif kind == ListMarks
601 pieces = String[]
602 for index in eachindex(state.marks)
603 position = locate(state, state.marks[index])
604 position > 0 && push!(pieces, "$(Char(Int('a') + index - 1))=$position")
605 end
606 push!(output, "marks " * (isempty(pieces) ? "none" : join(pieces, ' ')))
607 elseif kind in (UndoChange, RedoChange)
608 transfer_history!(state, kind == RedoChange, command.amount == 0 ? 1 : command.amount)
609 elseif kind == ListHistory
610 push!(output, "history undo=$(length(state.undo)) redo=$(length(state.redo))")
611 for (name, history) in (("undo", state.undo), ("redo", state.redo))
612 for (index, record) in enumerate(Iterators.reverse(history))
613 push!(output, "$name $index: $(record.source)")
614 end
615 end
616 elseif kind in (ReadFile, WriteFile, AppendFile, EditFile)
617 file_command!(state, command, range, output)
618 elseif kind == ListFiles
619 names = sort!(collect(keys(state.files)))
620 pieces = ["$name=$(length(state.files[name]))" for name in names]
621 push!(output, "files " * (isempty(pieces) ? "none" : join(pieces, ' ')))
622 elseif kind in (GlobalCommand, InverseCommand)
623 pattern = choose_pattern!(state, command.search)
624 selected = [line.id for line in state.buffer[a:b] if occursin(pattern, line.text) == (kind == GlobalCommand)]
625 isempty(selected) && fail("no match")
626 subcommand = command.sub::Command
627 changed = 0
628 for identity in selected
629 position = locate(state, identity)
630 position == 0 && continue
631 state.current = position
632 subrange = checked_range(state, subcommand.kind, Int[])
633 changed += execute!(state, subcommand, subrange, text, output; inside = true)
634 end
635 subcommand.kind == SubstituteText && changed == 0 && fail("no match")
636 end
637 kind in (DeleteText, PutText, JoinText, JoinWords, MoveText, CopyText) && suffix_line!(output, state, command.suffix)
638 0
639end

641function apply!(editor::Editor, source::String, text::Vector{String})
642 scanned = scan_address(source)
643 form = address_form(scanned.tokens)
644 command = parse_command(scanned.tail)
645 arity = address_kind(command.kind)
646 arity == 0 && form.count > 0 && fail("unexpected address")
647 arity == 1 && form.count == 2 && fail("too many addresses")
648 trial = fork(editor.state)
649 values = address_values!(trial, form)
650 range = checked_range(trial, command.kind, values)
651 output = String[]
652 execute!(trial, command, range, text, output)
653 length(trial.buffer) <= 200 || fail("buffer full")
654 if modifying(command)
655 push!(trial.undo, snapshot(editor.state, source))
656 empty!(trial.redo)
657 end
658 editor.state = trial
659 output
660end

664 while !eof(stdin)

666 source == "Q" && break
667 text = String[]

669 while true

671 entered == "." && break
