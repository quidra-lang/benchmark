62const BASIC_KINDS = new Set<string>(['', 'a', 'i', 'c', 'p', 'n', 'l', '=', '#', 'y', 'K', 'h', 'f']);
63function isBasicKind(kind: string): kind is BasicKind { return BASIC_KINDS.has(kind); }
64function isPrintForm(value: string): value is PrintForm { return value === 'p' || value === 'n' || value === 'l'; }
65const sortForm = (source: string): boolean => source === 'sort' || source.startsWith('sort ');

67function delimited(source: string, start: number, delimiter: string): Delimited {
68 let text = '';
69 const replacement: (string | null)[] = [];
70 for (let i = start; i < source.length; i++) {
71 let ch = source[i];
72 if (ch === '\\') {
73 if (++i === source.length) fail('missing delimiter');
74 ch = source[i];
75 text += ch;
76 replacement.push(ch);
77 } else if (ch === delimiter) {
78 return { text, replacement, raw: source.slice(start, i), next: i + 1 };
79 } else {
80 text += ch;
81 replacement.push(ch === '&' ? null : ch);
82 }
83 }
84 return fail('missing delimiter');
85}

87function scanAddress(source: string): Scanned {
88 const elements: AddressElement[] = [];
89 let i = 0;
90 while (i < source.length) {
91 const ch = source[i];
92 if (isDigit(ch) || ch === '+' || ch === '-') {
93 const start = i++;
94 while (isDigit(source[i])) i++;
95 elements.push({ kind: isDigit(ch) ? 'number' : 'offset', word: source.slice(start, i) });
96 } else if (ch === '.' || ch === '$' || ch === ',' || ch === ';' || ch === '%') {
97 elements.push({ kind: ch });
98 i++;
99 } else if (ch === "'") {
100 elements.push({ kind: 'mark', word: source[i + 1] ?? '' });
101 i += Math.min(2, source.length - i);
102 } else if (ch === '/' || ch === '?') {
103 const found = delimited(source, i + 1, ch);
104 elements.push({ kind: ch, word: found.text });
105 i = found.next;
106 } else break;
107 }
108 return { elements, command: source.slice(i) };
109}

111function singleAddress(elements: readonly AddressElement[]): elements is Address {
112 const bases = new Set<string>(['number', '.', '$', 'mark', '/', '?']);
113 return elements.length > 0 && elements.every((item, i) =>
114 item.kind === 'offset' || i === 0 && bases.has(item.kind));
115}

117function optionalAddress(elements: readonly AddressElement[]): Address | null {
118 if (elements.length === 0) return null;
119 if (!singleAddress(elements)) fail('bad address');
120 return elements;
121}

123function addressForm(elements: readonly AddressElement[]): AddressForm {
124 if (elements.length === 0) return { count: 0 };
125 if (elements.length === 1 && elements[0].kind === '%') {
126 return { count: 2, left: [{ kind: 'number', word: '1' }], separator: ',', right: [{ kind: '$' }] };
127 }
128 const separators = elements.flatMap((item, i) => item.kind === ',' || item.kind === ';' ? [i] : []);
129 if (separators.length > 1) fail('bad address');
130 if (separators.length === 0) {
131 if (!singleAddress(elements)) fail('bad address');
132 return { count: 1, left: elements };
133 }
134 const index = separators[0];
135 const left = optionalAddress(elements.slice(0, index));
136 const right = optionalAddress(elements.slice(index + 1));
137 return { count: 2, left, right, separator: elements[index].kind === ';' ? ';' : ',' };
138}

140function positiveArgument(argument: string): number | null {
141 if (argument === '') return null;
142 if (!/^\d+$/.test(argument) || Number(argument) === 0) fail('invalid suffix');
143 return Number(argument);
144}

146function printSuffix(argument: string): PrintSuffix {
147 if (argument === '' || isPrintForm(argument)) return argument;
148 return fail('invalid suffix');
149}

151function fileName(argument: string): string {
152 if (!/^ [a-z0-9]{1,8}$/.test(argument)) fail('invalid file name');
153 return argument.slice(1);
154}

156function parseSubstitution(argument: string): Substitution {
157 let pattern = '', flags = argument;
158 let replacement: Replacement | null = null;
159 if (!/^[gpnl0-9]*$/.test(argument)) {
160 const delimiter = argument[0];
161 if (delimiter === ' ' || delimiter === '\\') fail('missing delimiter');
162 const search = delimited(argument, 1, delimiter);
163 const replace = delimited(argument, search.next, delimiter);
164 pattern = search.text;
165 replacement = replace.raw === '%' ? null : replace.replacement;
166 flags = argument.slice(replace.next);
167 }
168 let global = false, count = 1, counted = false;
169 let suffix: PrintSuffix = '';
170 for (let i = 0; i < flags.length;) {
171 const ch = flags[i++];
172 if (ch === 'g') {
173 if (global) fail('invalid suffix');
174 global = true;
175 } else if (isPrintForm(ch)) {
176 if (suffix !== '') fail('invalid suffix');
177 suffix = ch;
178 } else if (isDigit(ch)) {
179 if (counted) fail('invalid suffix');
180 const start = i - 1;
181 while (isDigit(flags[i])) i++;
182 count = Number(flags.slice(start, i));
183 if (count === 0) fail('invalid suffix');
184 counted = true;
185 } else fail('invalid suffix');
186 }
187 return { kind: 's', pattern, replacement, global, suffix, count };
188}

190function parseOrdinary(source: string): OrdinaryCommand {
191 if (sortForm(source)) {
192 const options = source === 'sort' ? '' : source.slice(5);
193 if (source !== 'sort' && (!/^[ruin]+$/.test(options) || new Set(options).size !== options.length
194 || options.includes('i') && options.includes('n'))) fail('invalid suffix');
195 return { kind: 'sort', options };
196 }
197 const kind = source[0] ?? '', argument = source.slice(1);
198 if (isBasicKind(kind)) {
199 if (argument !== '') fail('invalid suffix');
200 return { kind };
201 }
202 if (kind === 'd' || kind === 'x' || kind === 'j' || kind === 'J') return { kind, suffix: printSuffix(argument) };
203 if (kind === 'z' || kind === '>' || kind === '<' || kind === 'u') return { kind, amount: positiveArgument(argument) };
204 if (kind === '~') {
205 if (argument !== '' && argument !== 'u' && argument !== 'l') fail('invalid suffix');
206 return { kind, letterCase: argument };
207 }
208 if (kind === 'r') {
209 return /^\d*$/.test(argument)
210 ? { kind: 'redo', amount: positiveArgument(argument) }
211 : { kind: 'read', name: fileName(argument) };
212 }
213 if (kind === 'w') return argument === '' ? { kind } : { kind: 'write', name: fileName(argument) };
214 if (kind === 'W' || kind === 'e') return { kind, name: fileName(argument) };
215 if (kind === 'k') {
216 if (!/^[a-z]$/.test(argument)) fail('invalid mark');
217 return { kind, mark: argument };
218 }
219 if (kind === 'm' || kind === 't') {
220 const destination = scanAddress(argument);
221 if (!singleAddress(destination.elements)) fail('invalid destination');
222 return { kind, destination: destination.elements, suffix: printSuffix(destination.command) };
223 }
224 if (kind === 's') return parseSubstitution(argument);
225 return fail('unknown command');
226}

228function parseCommand(source: string): Command {
229 const kind = source[0];
230 if (kind === 'g' || kind === 'v') {
231 const argument = source.slice(1);
232 if (argument === '' || argument[0] === ' ' || argument[0] === '\\') fail('missing delimiter');
233 const search = delimited(argument, 1, argument[0]);
234 const subcommand = argument.slice(search.next) || 'p';
235 if (!'pnldmtjJs><~aicx'.includes(subcommand[0]) || sortForm(subcommand)) fail('invalid global command');
236 return { kind, pattern: search.text, subcommand: parseOrdinary(subcommand) };
237 }
238 return parseOrdinary(source);
239}

241function hasText(source: string): boolean {
242 try {
243 const command = scanAddress(source).command;
244 if (/^[aic]$/.test(command)) return true;
245 if (!/^[gv]/.test(command) || command.length < 2 || command[1] === ' ' || command[1] === '\\') return false;
246 const search = delimited(command, 2, command[1]);
247 return /^[aic]$/.test(command.slice(search.next));
248 } catch (error) {
249 if (!(error instanceof EditorError)) throw error;
250 return false;
251 }
252}

254function addressKind(command: Command): 0 | 1 | 2 {
255 if (['u', 'redo', 'h', 'K', 'f', 'e'].includes(command.kind)) return 0;
256 if (['a', 'i', 'x', '=', 'read', 'k', 'z'].includes(command.kind)) return 1;
257 return 2;
258}

260function modifying(command: Command): boolean {
261 if (command.kind === 'g' || command.kind === 'v') return !['p', 'n', 'l'].includes(command.subcommand.kind);
262 return ['a', 'i', 'c', 'd', 'x', 'j', 'J', 'm', 't', 's', '>', '<', '~', 'sort', 'read', 'e'].includes(command.kind);
263}

319 private pattern(text: string): string {
320 if (text !== '') this.state.pattern = text;
321 if (this.state.pattern === null) fail('no previous pattern');
322 return this.state.pattern;

325 private address(elements: Address): number {
326 const state = this.state, last = state.lines.length;
327 let value = state.current;
328 const base = elements[0];
329 if (base.kind !== 'offset') {
330 if (base.kind === 'number') value = Number(base.word);
331 else if (base.kind === '.') value = state.current;
332 else if (base.kind === '$') value = last;
333 else if (base.kind === 'mark') {
334 if (!/^[a-z]$/.test(base.word)) fail('invalid mark');
335 value = state.lines.findIndex(line => line.id === state.marks.get(base.word)) + 1;
336 if (value === 0) fail('mark not set');
337 } else {
338 const pattern = this.pattern(base.word);
339 value = 0;
340 for (let step = 1; step <= last; step++) {
341 const pivot = state.current === 0 && base.kind === '?' ? last + 1 : state.current;
342 const index = base.kind === '/' ? (pivot + step - 1) % last : (pivot - step - 1 + last) % last;
343 if (state.lines[index].text.includes(pattern)) { value = index + 1; break; }
344 }
345 if (value === 0) fail('no match');
346 }
347 }
348 for (const element of elements) {
349 if (element.kind !== 'offset') continue;
350 const magnitude = element.word.length === 1 ? 1 : Number(element.word.slice(1));
351 value += element.word[0] === '+' ? magnitude : -magnitude;
352 }
353 return value;
354 }

356 private addresses(form: AddressForm): number[] {
357 if (form.count === 0) return [];
358 if (form.count === 1) return [this.address(form.left)];
359 const left = form.left === null ? (form.separator === ',' ? 1 : this.state.current) : this.address(form.left);
360 if (form.separator === ';') {
361 if (left < 0 || left > this.state.lines.length) fail('address out of range');
362 this.state.current = left;
363 }
364 return [left, form.right === null ? this.state.lines.length : this.address(form.right)];
365 }

367 private range(command: Command, values: readonly number[]): LineRange {
368 const state = this.state, last = state.lines.length, kind = command.kind;
369 const category = addressKind(command);
370 if (category === 0) return [0, 0];
371 if (category === 1) {
372 const fallback = ['=', 'read'].includes(kind) ? last : state.current + (kind === 'z' ? 1 : 0);
373 const a = values.length ? values[0] : fallback;
374 const minimum = kind === 'k' || kind === 'z' ? 1 : 0;
375 if (a < minimum || a > last) fail('address out of range');
376 return [a, a];
377 }
378 let a: number, b: number;
379 if (values.length) [a, b] = [values[0], values[values.length - 1]];
380 else if (['w', 'write', 'W', '#'].includes(kind)) return [1, last];
381 else if (['g', 'v', 'sort'].includes(kind)) [a, b] = [1, last];
382 else [a, b] = [state.current, state.current + (kind === 'j' || kind === 'J' ? 1 : 0)];
383 if (a < 1 || b < 1 || a > last || b > last) fail('address out of range');
384 if (a > b || a === b && (kind === 'j' || kind === 'J')) fail('invalid range');
385 return [a, b];
386 }

388 private print(number: number, form: PrintForm): void {
389 const text = this.state.lines[number - 1].text;
390 if (form === 'p') this.emit(text);
391 else if (form === 'n') this.emit(`${String(number).padStart(3, ' ')} ${text}`);
392 else {
393 let escaped = text.replaceAll('\\', '\\\\') + '$';
394 while (escaped.length > 40) {
395 this.emit(escaped.slice(0, 39) + '\\');
396 escaped = escaped.slice(39);
397 }
398 this.emit(escaped);
399 }
400 }

402 private suffix(suffix: PrintSuffix): void {
403 if (suffix !== '' && this.state.current > 0) this.print(this.state.current, suffix);
404 }

406 private substitute(command: Substitution, a: number, b: number, insideGlobal: boolean): number {
407 const state = this.state, pattern = this.pattern(command.pattern);
408 const replacement = command.replacement ?? state.replacement;
409 if (replacement === null) fail('no previous substitution');
410 state.replacement = replacement;
411 const expanded = replacement.map(part => part === null ? pattern : part).join('');
412 let changed = 0;
413 for (let i = a - 1; i < b; i++) {
414 const line = state.lines[i];
415 let position = 0, occurrence = 0, result = '', substituted = false;
416 for (;;) {
417 const found = line.text.indexOf(pattern, position);
418 if (found < 0) break;
419 occurrence++;
420 const replace = occurrence === command.count || command.global && occurrence > command.count;
421 result += line.text.slice(position, found) + (replace ? expanded : pattern);
422 substituted ||= replace;
423 position = found + pattern.length;
424 }
425 if (!substituted) continue;
426 result += line.text.slice(position);
427 if (result.length > 80) fail('line too long');
428 if (result === '' || result.endsWith(' ')) fail('bad line');
429 state.lines[i] = { id: line.id, text: result };
430 state.current = i + 1;
431 changed++;
432 }
433 if (changed === 0 && !insideGlobal) fail('no match');
434 if (changed > 0) this.suffix(command.suffix);
435 return changed;
436 }

438 private global(command: GlobalCommand, a: number, b: number, text: readonly string[]): void {
439 const state = this.state, pattern = this.pattern(command.pattern);
440 const selected = state.lines.slice(a - 1, b)
441 .filter(line => line.text.includes(pattern) === (command.kind === 'g')).map(line => line.id);
442 if (selected.length === 0) fail('no match');
443 let changed = 0;
444 for (const id of selected) {
445 const index = state.lines.findIndex(line => line.id === id);
446 if (index < 0) continue;
447 state.current = index + 1;
448 const [start, end] = this.range(command.subcommand, []);
449 changed += this.execute(command.subcommand, start, end, text, true);
450 }
451 if (command.subcommand.kind === 's' && changed === 0) fail('no match');
452 }

454 private execute(command: Command, a: number, b: number, text: readonly string[], insideGlobal = false): number {
455 const state = this.state, kind = command.kind;
456 if (kind === 's') return this.substitute(command, a, b, insideGlobal);
457 if (kind === 'g' || kind === 'v') this.global(command, a, b, text);
458 else if (kind === '' || kind === 'p' || kind === 'n' || kind === 'l') {
459 for (let i = kind === '' ? b : a; i <= b; i++) this.print(i, kind || 'p');
460 state.current = b;
461 } else if (kind === 'z') {
462 if (command.amount !== null) state.displayCount = command.amount;
463 const end = Math.min(a + state.displayCount - 1, state.lines.length);
464 for (let i = a; i <= end; i++) this.print(i, 'p');
465 state.current = end;
466 } else if (kind === '=') this.emit(String(a));
467 else if (kind === '#') {
468 const contents = this.contents(a, b);
469 const words = contents.reduce((n, line) => n + line.split(' ').filter(Boolean).length, 0);
470 const chars = contents.reduce((n, line) => n + line.length, 0);
471 this.emit(`count lines=${contents.length} words=${words} chars=${chars}`);
472 } else if (kind === 'w') {
473 const contents = this.contents(a, b);
474 for (const line of contents) this.emit(line);
475 this.emit(`lines=${contents.length} bytes=${contents.reduce((n, line) => n + line.length + 1, 0)}`);
476 } else if (kind === 'a' || kind === 'i') {
477 const position = kind === 'a' ? a : Math.max(a, 1) - 1;
478 this.replace(position, 0, this.newLines(text));
479 state.current = position + text.length;
480 } else if (kind === 'c' || kind === 'd') {
481 state.cut = this.contents(a, b);
482 const inserted = kind === 'c' ? this.newLines(text) : [];
483 this.replace(a - 1, b - a + 1, inserted);
484 state.current = inserted.length ? a + inserted.length - 1 : Math.min(a, state.lines.length);
485 if (kind === 'd') this.suffix(command.suffix);
486 } else if (kind === 'y') state.cut = this.contents(a, b);
487 else if (kind === 'x') {
488 if (state.cut.length === 0) fail('nothing to put');
489 this.replace(a, 0, this.newLines(state.cut));
490 state.current = a + state.cut.length;
491 this.suffix(command.suffix);
492 } else if (kind === 'm' || kind === 't') {
493 const destination = this.address(command.destination), count = b - a + 1;
494 if (destination < 0 || destination > state.lines.length) fail('address out of range');
495 if (kind === 'm') {
496 if (a <= destination && destination < b) fail('invalid destination');
497 const moved = state.lines.splice(a - 1, count);
498 const position = destination < a ? destination : destination - count;
499 this.replace(position, 0, moved);
500 state.current = position + count;
501 } else {
502 const copied = this.newLines(this.contents(a, b));
503 this.replace(destination, 0, copied);
504 state.current = destination + count;
505 }
506 this.suffix(command.suffix);
507 } else if (kind === 'j' || kind === 'J') {
508 const contents = this.contents(a, b);
509 const joined = kind === 'j' ? contents.join('') : contents.map((line, i) => i ? line.replace(/^ +/, '') : line).join(' ');
510 if (joined.length > 80) fail('line too long');
511 this.replace(a - 1, b - a + 1, [{ id: state.lines[a - 1].id, text: joined }]);
512 state.current = a;
513 this.suffix(command.suffix);
514 } else if (kind === '>' || kind === '<' || kind === '~') {
515 const width = kind === '~' ? 4 : command.amount ?? 4;
516 for (let i = a - 1; i < b; i++) {
517 const line = state.lines[i];
518 let value = line.text;
519 if (kind === '>') {
520 if (value.length + width > 80) fail('line too long');
521 value = ' '.repeat(width) + value;
522 } else if (kind === '<') value = value.slice(Math.min(width, /^ */.exec(value)![0].length));
523 else {
524 const letterCase = command.letterCase;
525 value = value.replace(/[A-Za-z]/g, ch => letterCase === 'u' ? ch.toUpperCase()
526 : letterCase === 'l' ? ch.toLowerCase() : ch === ch.toUpperCase() ? ch.toLowerCase() : ch.toUpperCase());
527 }
528 state.lines[i] = { id: line.id, text: value };
529 }
530 state.current = b;
531 } else if (kind === 'sort') {
532 const options = command.options;
533 const key = (line: Line): string | number => options.includes('n') ? Number(/^-?\d+/.exec(line.text)?.[0] ?? 0)
534 : options.includes('i') ? line.text.toLowerCase() : line.text;
535 const direction = options.includes('r') ? -1 : 1;
536 let sorted = state.lines.slice(a - 1, b).map(line => ({ line, key: key(line) }));
537 sorted.sort((x, y) => direction * (x.key < y.key ? -1 : x.key > y.key ? 1 : 0));
538 if (options.includes('u')) sorted = sorted.filter((item, i, all) => i === 0 || item.key !== all[i - 1].key);
539 this.replace(a - 1, b - a + 1, sorted.map(item => item.line));
540 state.current = a + sorted.length - 1;
541 } else if (kind === 'k') state.marks.set(command.mark, state.lines[a - 1].id);
542 else if (kind === 'K') {
543 const marks = [...state.marks].sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0).flatMap(([name, id]) => {
544 const index = state.lines.findIndex(line => line.id === id);
545 return index < 0 ? [] : [`${name}=${index + 1}`];
546 });
547 this.emit('marks ' + (marks.join(' ') || 'none'));
548 } else if (kind === 'u' || kind === 'redo') {
549 const source = kind === 'u' ? state.undo : state.redo;
550 const target = kind === 'u' ? state.redo : state.undo;
551 const count = command.amount ?? 1;
552 if (source.length < count) fail(kind === 'u' ? 'nothing to undo' : 'nothing to redo');
553 for (let i = 0; i < count; i++) {
554 const record = source.pop()!;
555 target.push(this.snapshot(record.command));
556 state.lines = record.lines.slice();
557 state.current = record.current;
558 }
559 } else if (kind === 'h') {
560 this.emit(`history undo=${state.undo.length} redo=${state.redo.length}`);
561 for (const [name, history] of [['undo', state.undo], ['redo', state.redo]] as const) {
562 for (let i = history.length - 1; i >= 0; i--) this.emit(`${name} ${history.length - i}: ${history[i].command}`);
563 }
564 } else if (kind === 'write' || kind === 'W' || kind === 'read' || kind === 'e') {
565 let file = state.files.get(command.name);
566 if (kind === 'write' || kind === 'W') {
567 file = (kind === 'W' ? file ?? [] : []).concat(this.contents(a, b));
568 state.files.set(command.name, file);
569 } else {
570 if (file === undefined) fail('no such file');
571 const inserted = this.newLines(file);
572 if (kind === 'e') { state.lines = inserted; state.current = inserted.length; }
573 else { this.replace(a, 0, inserted); state.current = a + inserted.length; }
574 }
575 this.emit(`file ${command.name} lines=${file.length} bytes=${file.reduce((n, line) => n + line.length + 1, 0)}`);
576 } else if (kind === 'f') {
577 const files = [...state.files.keys()].sort().map(name => `${name}=${state.files.get(name)!.length}`);
578 this.emit('files ' + (files.join(' ') || 'none'));
579 }
580 return 0;
581 }

583 process(source: string, text: readonly string[]): string[] {
584 const previous = this.state;
585 const before = this.snapshot(source);
586 this.state = copyState(previous);
587 this.output = [];
588 try {
589 const split = scanAddress(source);
590 const form = addressForm(split.elements);
591 const command = parseCommand(split.command);
592 const category = addressKind(command);
593 if (category === 0 && form.count > 0) fail('unexpected address');
594 if (category === 1 && form.count > 1) fail('too many addresses');
595 const [a, b] = this.range(command, this.addresses(form));
596 this.execute(command, a, b, text);
597 if (this.state.lines.length > 200) fail('buffer full');
598 if (modifying(command)) {
599 this.state.undo.push(before);
600 this.state.redo = [];
601 }
602 return this.output;
603 } catch (error) {
604 this.state = previous;
605 this.output = [];
606 if (!(error instanceof EditorError)) throw error;
607 return ['? ' + error.message];
608 }
609 }
