10func isDigit(_ ch: Character) -> Bool {
11 ch >= "0" && ch <= "9"
12}
13func isLower(_ ch: Character) -> Bool {
14 ch >= "a" && ch <= "z"

17 ch >= "A" && ch <= "Z"

20typealias Replacement = [Character?]

22struct Delimited {
23 let text: String
24 let replacement: Replacement
25 let raw: String
26 let next: Int
27}
28
29func readDelimited(_ source: [Character], from start: Int, delimiter: Character) throws -> Delimited {
30 var text = "", replacement: Replacement = []
31 var i = start
32 while i < source.count {
33 let ch = source[i]
34 if ch == "\\" {
35 i += 1
36 guard i < source.count else {
37 throw EditorError("missing delimiter")
38 }
39 text.append(source[i])
40 replacement.append(source[i])
41 } else if ch == delimiter {
42 return Delimited(text: text, replacement: replacement, raw: String(source[start..<i]), next: i + 1)
43 } else {
44 text.append(ch)
45 replacement.append(ch == "&" ? nil : ch)
46 }
47 i += 1
48 }
49 throw EditorError("missing delimiter")
50}

52enum Element {
53 case number(Int), offset(Int), current, last, separator(Character), all, mark(String), search(String, forward: Bool)
54}

56struct AddressScan {
57 let elements: [Element]
58 let command: String
59}
60
61func scanAddress(_ text: String) throws -> AddressScan {
62 let source = Array(text)
63 var elements: [Element] = []
64 var i = 0
65 while i < source.count {
66 let ch = source[i]
67 if isDigit(ch) || ch == "+" || ch == "-" {
68 let start = i
69 i += 1
70 while i < source.count && isDigit(source[i]) {
71 i += 1
72 }
73 if isDigit(ch) {
74 elements.append(.number(Int(String(source[start..<i]))!))
75 } else {
76 let magnitude = i == start + 1 ? 1 : Int(String(source[(start + 1)..<i]))!
77 elements.append(.offset(ch == "+" ? magnitude : -magnitude))
78 }
79 } else if ".$,;%".contains(ch) {
80 switch ch {
81 case ".":
82 elements.append(.current)
83 case "$":
84 elements.append(.last)
85 case ",", ";":
86 elements.append(.separator(ch))
87 default:
88 elements.append(.all)
89 }
90 i += 1
91 } else if ch == "'" {
92 i += 1
93 elements.append(.mark(i < source.count ? String(source[i]) : ""))
94 if i < source.count {
95 i += 1
96 }
97 } else if ch == "/" || ch == "?" {
98 let found = try readDelimited(source, from: i + 1, delimiter: ch)
99 elements.append(.search(found.text, forward: ch == "/"))
100 i = found.next
101 } else {
102 break
103 }
104 }
105 return AddressScan(elements: elements, command: String(source[i...]))
106}

108func singleAddress(_ elements: [Element]) -> Bool {
109 if elements.isEmpty {
110 return false
111 }
112 for (i, element) in elements.enumerated() {
113 switch element {
114 case .offset:
115 continue
116 case .separator, .all:
117 return false
118 default:
119 if i != 0 {
120 return false
121 }
122 }
123 }
124 return true
125}

127struct AddressForm {
128 var count = 0
129 var left: [Element] = []
130 var right: [Element] = []
131 var separator: Character = ","
132}
133
134func addressForm(_ elements: [Element]) throws -> AddressForm {
135 if elements.isEmpty {
136 return AddressForm()
137 }
138 if elements.count == 1, case .all = elements[0] {
139 return AddressForm(count: 2, left: [.number(1)], right: [.last])
140 }
141 var separator: Int?
142 for (i, element) in elements.enumerated() {
143 if case .separator = element {
144 if separator != nil {
145 throw EditorError("bad address")
146 }
147 separator = i
148 }
149 }
150 guard let index = separator else {
151 guard singleAddress(elements) else {
152 throw EditorError("bad address")
153 }
154 return AddressForm(count: 1, left: elements)
155 }
156 let left = Array(elements[..<index]), right = Array(elements[(index + 1)...])
157 guard (left.isEmpty || singleAddress(left)) && (right.isEmpty || singleAddress(right)) else {
158 throw EditorError("bad address")
159 }
160 if case let .separator(ch) = elements[index] {
161 return AddressForm(count: 2, left: left, right: right, separator: ch)
162 }
163 preconditionFailure("Expected separator")
164}

188func positiveArgument(_ argument: String) throws -> Int? {
189 if argument.isEmpty {
190 return nil
191 }
192 guard argument.allSatisfy(isDigit), let value = Int(argument),
193 value > 0
194 else {
195 throw EditorError("invalid suffix")
196 }
197 return value

200func printSuffix(_ argument: String) throws -> Character? {
201 if argument.isEmpty {
202 return nil
203 }
204 guard argument.count == 1, let ch = argument.first, "pnl".contains(ch) else {
205 throw EditorError("invalid suffix")

207 return ch

210func fileName(_ argument: String) throws -> String {
211 guard (2...9).contains(argument.count), argument.first == " ",
212 argument.dropFirst().allSatisfy({ isLower($0) || isDigit($0) })
213 else {
214 throw EditorError("invalid file name")
215 }
216 return String(argument.dropFirst())
217}

219func parseSubstitution(_ argument: String) throws -> Command {
220 let command = Command(.substitute)
221 var flags = Array(argument)
222 if !argument.allSatisfy({ isDigit($0) || "gpnl".contains($0) }) {
223 let source = Array(argument), delimiter = argument.first!
224 if delimiter == " " || delimiter == "\\" {
225 throw EditorError("missing delimiter")
226 }
227 let search = try readDelimited(source, from: 1, delimiter: delimiter)
228 let replacement = try readDelimited(source, from: search.next, delimiter: delimiter)
229 command.pattern = search.text
230 if replacement.raw != "%" {
231 command.replacement = replacement.replacement
232 }
233 flags = Array(source[replacement.next...])
234 }
235 var i = 0, counted = false
236 while i < flags.count {
237 let ch = flags[i]
238 i += 1
239 if ch == "g" {
240 if command.allMatches {
241 throw EditorError("invalid suffix")
242 }
243 command.allMatches = true
244 } else if "pnl".contains(ch) {
245 if command.suffix != nil {
246 throw EditorError("invalid suffix")
247 }
248 command.suffix = ch
249 } else if isDigit(ch) {
250 if counted {
251 throw EditorError("invalid suffix")
252 }
253 let start = i - 1
254 while i < flags.count && isDigit(flags[i]) {
255 i += 1
256 }
257 command.occurrence = Int(String(flags[start..<i]))!
258 if command.occurrence == 0 {
259 throw EditorError("invalid suffix")
260 }
261 counted = true
262 } else {
263 throw EditorError("invalid suffix")
264 }
265 }
266 return command
267}

269func sortForm(_ source: String) -> Bool {
270 source == "sort" || source.hasPrefix("sort ")
271}

273func parseCommand(_ source: String) throws -> Command {
274 if sortForm(source) {
275 let command = Command(.sort)
276 command.argument = source == "sort" ? "" : String(source.dropFirst(5))
277 if source != "sort" {
278 let options = command.argument
279 guard !options.isEmpty, options.allSatisfy({ "ruin".contains($0) }), Set(options).count == options.count,
280 !(options.contains("i") && options.contains("n"))
281 else {
282 throw EditorError("invalid suffix")
283 }
284 }
285 return command
286 }
287 guard let ch = source.first else {
288 return Command(.bare)
289 }
290 let argument = String(source.dropFirst())
291 let plain: [Character: Kind] = [
292 "a": .append, "i": .insert, "c": .change, "p": .print, "n": .numbered, "l": .list,
293 "=": .position, "#": .count, "y": .yank, "K": .marks, "h": .history, "f": .files
294 ]
295 if let kind = plain[ch] {
296 if !argument.isEmpty {
297 throw EditorError("invalid suffix")
298 }
299 return Command(kind)
300 }
301 if "dxjJ".contains(ch) {
302 let command = Command(ch == "d" ? .erase : ch == "x" ? .put : ch == "j" ? .join : .joinWords)
303 command.suffix = try printSuffix(argument)
304 return command
305 }
306 if "z><u".contains(ch) {
307 let command = Command(ch == "z" ? .scroll : ch == ">" ? .indent : ch == "<" ? .unindent : .undo)
308 command.amount = try positiveArgument(argument)
309 return command
310 }
311 if ch == "~" {
312 guard argument.isEmpty || argument == "u" || argument == "l" else {
313 throw EditorError("invalid suffix")
314 }
315 let command = Command(.letterCase)
316 command.argument = argument
317 return command
318 }
319 if ch == "r" && argument.allSatisfy(isDigit) {
320 let command = Command(.redo)
321 command.amount = try positiveArgument(argument)
322 return command
323 }
324 if ch == "w" && argument.isEmpty {
325 return Command(.writePrint)
326 }
327 if "rwWe".contains(ch) {
328 let command = Command(ch == "r" ? .read : ch == "w" ? .write : ch == "W" ? .writeAppend : .edit)
329 command.argument = try fileName(argument)
330 return command
331 }
332 if ch == "k" {
333 guard argument.count == 1, let mark = argument.first, isLower(mark) else {
334 throw EditorError("invalid mark")
335 }
336 let command = Command(.mark)
337 command.argument = argument
338 return command
339 }
340 if ch == "m" || ch == "t" {
341 let destination = try scanAddress(argument)
342 guard singleAddress(destination.elements) else {
343 throw EditorError("invalid destination")
344 }
345 let command = Command(ch == "m" ? .move : .copy)
346 command.destination = destination.elements
347 command.suffix = try printSuffix(destination.command)
348 return command
349 }
350 if ch == "s" {
351 return try parseSubstitution(argument)
352 }
353 if ch == "g" || ch == "v" {
354 guard let delimiter = argument.first, delimiter != " ",
355 delimiter != "\\"
356 else {
357 throw EditorError("missing delimiter")
358 }
359 let chars = Array(argument)
360 let search = try readDelimited(chars, from: 1, delimiter: delimiter)
361 let subcommand = search.next == chars.count ? "p" : String(chars[search.next...])
362 guard "pnldmtjJs><~aicx".contains(subcommand.first!),
363 !sortForm(subcommand)
364 else {
365 throw EditorError("invalid global command")
366 }
367 let command = Command(ch == "g" ? .global : .inverseGlobal)
368 command.pattern = search.text
369 command.subcommand = try parseCommand(subcommand)
370 return command
371 }
372 throw EditorError("unknown command")
373}

375func hasText(_ source: String) -> Bool {
376 do {
377 let command = Array(try scanAddress(source).command)
378 if command.count == 1 && "aic".contains(command[0]) {
379 return true
380 }
381 guard command.count >= 2, "gv".contains(command[0]), command[1] != " ", command[1] != "\\" else {
382 return false
383 }
384 let search = try readDelimited(command, from: 2, delimiter: command[1])
385 return search.next + 1 == command.count && "aic".contains(command[search.next])
386 } catch { return false }
387}

389func addressKind(_ kind: Kind) -> Int {
390 switch kind {
391 case .undo, .redo, .history, .marks, .files, .edit:
392 return 0
393 case .append, .insert, .put, .position, .read, .mark, .scroll:
394 return 1
395 default:
396 return 2
397 }
398}

400func modifying(_ command: Command) -> Bool {
401 switch command.kind {
402 case .global, .inverseGlobal:
403 return ![Kind.print, .numbered, .list].contains(command.subcommand!.kind)
404 case .append, .insert, .change, .erase, .put, .join, .joinWords, .move, .copy, .substitute,
405 .indent, .unindent, .letterCase, .sort, .read, .edit:
406 return true
407 default:
408 return false
409 }

440 static func < (lhs: SortKey, rhs: SortKey) -> Bool {
441 switch (lhs, rhs) {
442 case let (.number(a), .number(b)):
443 return a < b
444 case let (.text(a), .text(b)):

447 return true

464final class Editor {
465 private var state = EditorState()
466 private var output: [String] = []
467 private var last: Int {
468 state.lines.count
469 }
470 private func emit(_ text: String) {
471 output.append(text)
472 }
473
474 private func newLines(_ text: [String]) -> [Line] {
475 var lines: [Line] = []
476 for value in text {
477 lines.append(Line(id: state.nextID, text: value))
478 state.nextID += 1
479 }
480 return lines
481 }
482
483 private func snapshot(_ command: String) -> Record {
484 Record(
485 command: command,
486 lines: state.lines,
487 current: state.current
488 )
489 }
490 private func contents(_ a: Int, _ b: Int) -> [String] {
491 state.lines[(a - 1)..<b].map(\.text)
492 }
493
494 private func replace(at position: Int, count: Int, with inserted: [Line]) {
495 state.lines.replaceSubrange(position..<(position + count), with: inserted)
496 }
497
498 private func lineNumber(_ id: Int) -> Int {
499 state.lines.firstIndex(where: { $0.id == id }).map { $0 + 1 } ?? 0
500 }
501
502 private func pattern(_ text: String) throws -> String {
503 if !text.isEmpty {
504 state.pattern = text
505 }
506 guard let pattern = state.pattern else {
507 throw EditorError("no previous pattern")
508 }
509 return pattern
510 }
511
512 private func address(_ elements: [Element]) throws -> Int {
513 var value = state.current, firstOffset = 1
514 switch elements[0] {
515 case let .number(number):
516 value = number
517 case .current:
518 value = state.current
519 case .last:
520 value = last
521 case let .mark(name):
522 guard name.count == 1, let mark = name.first, isLower(mark) else {
523 throw EditorError("invalid mark")
524 }
525 value = state.marks[mark].map(lineNumber) ?? 0
526 if value == 0 {
527 throw EditorError("mark not set")
528 }
529 case let .search(text, forward):
530 let search = try pattern(text)
531 value = 0
532 for offset in 0..<last {
533 let step = offset + 1
534 let pivot = state.current == 0 && !forward ? last + 1 : state.current
535 let index = forward ? (pivot + step - 1) % last : (pivot - step - 1 + last) % last
536 if state.lines[index].text.contains(search) {
537 value = index + 1
538 break
539 }
540 }
541 if value == 0 {
542 throw EditorError("no match")
543 }
544 case .offset:
545 firstOffset = 0
546 default:
547 preconditionFailure("Expected a single address")
548 }
549 for element in elements.dropFirst(firstOffset) {
550 if case let .offset(delta) = element {
551 value += delta
552 }
553 }
554 return value
555 }
556
557 private func addresses(_ form: AddressForm) throws -> [Int] {
558 if form.count == 0 {
559 return []
560 }
561 if form.count == 1 {
562 return [try address(form.left)]
563 }
564 let left = try (form.left.isEmpty ? (form.separator == "," ? 1 : state.current) : address(form.left))
565 if form.separator == ";" {
566 if left < 0 || left > last {
567 throw EditorError("address out of range")
568 }
569 state.current = left
570 }
571 let right = try (form.right.isEmpty ? last : address(form.right))
572 return [left, right]
573 }
574
575 private func range(_ command: Command, values: [Int]) throws -> (Int, Int) {
576 let kind = command.kind, category = addressKind(kind)
577 if category == 0 {
578 return (0, 0)
579 }
580 if category == 1 {
581 let fallback = kind == .position || kind == .read ? last : state.current + (kind == .scroll ? 1 : 0)
582 let a = values.first ?? fallback
583 let minimum = kind == .mark || kind == .scroll ? 1 : 0
584 if a < minimum || a > last {
585 throw EditorError("address out of range")
586 }
587 return (a, a)
588 }
589 let a: Int, b: Int
590 if let first = values.first {
591 a = first
592 b = values.last!
593 } else if [Kind.writePrint, .write, .writeAppend, .count].contains(kind) {
594 return (1, last)
595 } else if [Kind.global, .inverseGlobal, .sort].contains(kind) {
596 a = 1
597 b = last
598 } else {
599 a = state.current
600 b = state.current + (kind == .join || kind == .joinWords ? 1 : 0)
601 }
602 if a < 1 || b < 1 || a > last || b > last {
603 throw EditorError("address out of range")
604 }
605 if a > b || a == b && (kind == .join || kind == .joinWords) {
606 throw EditorError("invalid range")
607 }
608 return (a, b)
609 }

611 private func printLine(_ number: Int, form: Character) {
612 let text = state.lines[number - 1].text
613 if form == "p" {
614 emit(text)
615 } else if form == "n" {
616 let label = String(number)
617 emit(String(repeating: " ", count: max(0, 3 - label.count)) + label + " " + text)
618 } else {
619 var escaped = text.replacingOccurrences(of: "\\", with: "\\\\") + "$"
620 while escaped.count > 40 {
621 emit(String(escaped.prefix(39)) + "\\")
622 escaped = String(escaped.dropFirst(39))
623 }
624 emit(escaped)
625 }
626 }

628 private func suffix(_ command: Command) {
629 if let form = command.suffix, state.current > 0 {
630 printLine(state.current, form: form)
631 }
632 }

634 private func substitute(_ command: Command, from a: Int, through b: Int, insideGlobal: Bool) throws -> Int {
635 let search = try pattern(command.pattern)
636 guard let replacement = command.replacement ?? state.replacement
637 else {
638 throw EditorError("no previous substitution")
639 }
640 state.replacement = replacement
641 let expanded = replacement.map { part in part.map { String($0) } ?? search }.joined()
642 var changed = 0
643 for i in (a - 1)..<b {
644 let line = state.lines[i]
645 var position = line.text.startIndex, occurrence = 0, result = "", substituted = false
646 while let found = line.text.range(of: search, range: position..<line.text.endIndex) {
647 occurrence += 1
648 let replace = occurrence == command.occurrence || command.allMatches && occurrence > command.occurrence
649 result += String(line.text[position..<found.lowerBound]) + (replace ? expanded : search)
650 substituted = substituted || replace
651 position = found.upperBound
652 }
653 if !substituted {
654 continue
655 }
656 result += String(line.text[position...])
657 if result.utf8.count > 80 {
658 throw EditorError("line too long")
659 }
660 if result.isEmpty || result.last == " " {
661 throw EditorError("bad line")
662 }
663 state.lines[i].text = result
664 state.current = i + 1
665 changed += 1
666 }
667 if changed == 0 && !insideGlobal {
668 throw EditorError("no match")
669 }
670 if changed > 0 {
671 suffix(command)
672 }
673 return changed
674 }

676 private func global(_ command: Command, from a: Int, through b: Int, text: [String]) throws {
677 let search = try pattern(command.pattern)
678 let selected = state.lines[(a - 1)..<b].filter { $0.text.contains(search) == (command.kind == .global) }
679 .map(\.id)
680 if selected.isEmpty {
681 throw EditorError("no match")
682 }
683 let subcommand = command.subcommand!
684 var changed = 0
685 for id in selected {
686 let number = lineNumber(id)
687 if number == 0 {
688 continue
689 }
690 state.current = number
691 let (start, end) = try range(subcommand, values: [])
692 changed += try execute(subcommand, from: start, through: end, text: text, insideGlobal: true)
693 }
694 if subcommand.kind == .substitute && changed == 0 {
695 throw EditorError("no match")
696 }
697 }

699 private func execute(_ command: Command, from a: Int, through b: Int, text: [String],
700 insideGlobal: Bool = false) throws -> Int
701 {
702 let kind = command.kind
703 if kind == .substitute {
704 return try substitute(command, from: a, through: b, insideGlobal: insideGlobal)
705 }
706 if kind == .global || kind == .inverseGlobal {
707 try global(command, from: a, through: b, text: text)
708 } else if [Kind.bare, .print, .numbered, .list].contains(kind) {
709 let form: Character = kind == .numbered ? "n" : kind == .list ? "l" : "p"
710 for i in (kind == .bare ? b : a)...b {
711 printLine(i, form: form)
712 }
713 state.current = b
714 } else if kind == .scroll {
715 if let amount = command.amount {
716 state.displayCount = amount
717 }
718 let end = min(a + state.displayCount - 1, last)
719 for i in a...end {
720 printLine(i, form: "p")
721 }
722 state.current = end
723 } else if kind == .position {
724 emit(String(a))
725 } else if kind == .count {
726 let text = contents(a, b)
727 let words = text.reduce(0) { $0 + $1.split(separator: " ").count }
728 let chars = text.reduce(0) { $0 + $1.utf8.count }
729 emit("count lines=\(text.count) words=\(words) chars=\(chars)")
730 } else if kind == .writePrint {
731 let text = contents(a, b)
732 for line in text {
733 emit(line)
734 }
735 emit("lines=\(text.count) bytes=\(byteCount(text))")
736 } else if kind == .append || kind == .insert {
737 let position = kind == .append ? a : max(a, 1) - 1
738 let inserted = newLines(text)
739 replace(at: position, count: 0, with: inserted)
740 state.current = position + inserted.count
741 } else if kind == .change || kind == .erase {
742 state.cut = contents(a, b)
743 let inserted = kind == .change ? newLines(text) : []
744 replace(at: a - 1, count: b - a + 1, with: inserted)
745 state.current = inserted.isEmpty ? min(a, last) : a + inserted.count - 1
746 suffix(command)
747 } else if kind == .yank {
748 state.cut = contents(a, b)
749 } else if kind == .put {
750 if state.cut.isEmpty {
751 throw EditorError("nothing to put")
752 }
753 let inserted = newLines(state.cut)
754 replace(at: a, count: 0, with: inserted)
755 state.current = a + inserted.count
756 suffix(command)
757 } else if kind == .move || kind == .copy {
758 let destination = try address(command.destination), count = b - a + 1
759 if destination < 0 || destination > last {
760 throw EditorError("address out of range")
761 }
762 if kind == .move {
763 if a <= destination && destination < b {
764 throw EditorError("invalid destination")
765 }
766 let moved = Array(state.lines[(a - 1)..<b])
767 replace(at: a - 1, count: count, with: [])
768 let position = destination < a ? destination : destination - count
769 replace(at: position, count: 0, with: moved)
770 state.current = position + count
771 } else {
772 let copied = newLines(contents(a, b))
773 replace(at: destination, count: 0, with: copied)
774 state.current = destination + count
775 }
776 suffix(command)
777 } else if kind == .join || kind == .joinWords {
778 let text = contents(a, b)
779 let joined = kind == .join ? text.joined()
780 : text.enumerated().map { i, line in i == 0 ? line : String(line.drop(while: { $0 == " " })) }
781 .joined(separator: " ")
782 if joined.utf8.count > 80 {
783 throw EditorError("line too long")
784 }
785 let line = Line(id: state.lines[a - 1].id, text: joined)
786 replace(at: a - 1, count: b - a + 1, with: [line])
787 state.current = a
788 suffix(command)
789 } else if kind == .indent || kind == .unindent || kind == .letterCase {
790 let width = command.amount ?? 4
791 for i in (a - 1)..<b {
792 var value = state.lines[i].text
793 if kind == .indent {
794 if value.utf8.count + width > 80 {
795 throw EditorError("line too long")
796 }
797 value = String(repeating: " ", count: width) + value
798 } else if kind == .unindent {
799 let spaces = value.prefix(while: { $0 == " " }).count
800 value = String(value.dropFirst(min(width, spaces)))
801 } else if command.argument == "u" {
802 value = value.uppercased()
803 } else if command.argument == "l" {
804 value = value.lowercased()
805 } else {
806 value = String(value.map { ch in
807 isLower(ch) ? Character(String(ch).uppercased()) : isUpper(ch) ?
808 Character(String(ch).lowercased()) : ch
809 })
810 }
811 state.lines[i].text = value
812 }
813 state.current = b
814 } else if kind == .sort {
815 let options = command.argument
816 var sorted: [KeyedLine] = []
817 for i in (a - 1)..<b {
818 let line = state.lines[i]
819 let key: SortKey
820 if options.contains("n") {
821 let negative = line.text.first == "-"
822 let digits = line.text.dropFirst(negative ? 1 : 0).prefix(while: isDigit)
823 let number = digits.isEmpty ? 0 : Int(String(digits))! * (negative ? -1 : 1)
824 key = .number(number)
825 } else {
826 key = .text(options.contains("i") ? line.text.lowercased() : line.text)
827 }
828 sorted.append(KeyedLine(line: line, key: key, position: i))
829 }
830 sorted.sort { x, y in
831 if x.key == y.key {
832 return x.position < y.position
833 }
834 return options.contains("r") ? y.key < x.key : x.key < y.key
835 }
836 var lines: [Line] = [], previous: SortKey?
837 for item in sorted {
838 if options.contains("u") && previous == item.key {
839 continue
840 }
841 lines.append(item.line)
842 previous = item.key
843 }
844 replace(at: a - 1, count: b - a + 1, with: lines)
845 state.current = a + lines.count - 1
846 } else if kind == .mark {
847 state.marks[command.argument.first!] = state.lines[a - 1].id
848 } else if kind == .marks {
849 var marks: [String] = []
850 for name in state.marks.keys.sorted() {
851 let number = lineNumber(state.marks[name]!)
852 if number > 0 {
853 marks.append("\(name)=\(number)")
854 }
855 }
856 emit("marks " + (marks.isEmpty ? "none" : marks.joined(separator: " ")))
857 } else if kind == .undo || kind == .redo {
858 let count = command.amount ?? 1
859 if count > (kind == .undo ? state.undo.count : state.redo.count) {
860 throw EditorError(kind == .undo ? "nothing to undo" : "nothing to redo")
861 }
862 for _ in 0..<count {
863 let record = kind == .undo ? state.undo.removeLast() : state.redo.removeLast()
864 let reverse = snapshot(record.command)
865 if kind == .undo {
866 state.redo.append(reverse)
867 } else {
868 state.undo.append(reverse)
869 }
870 state.lines = record.lines
871 state.current = record.current
872 }
873 } else if kind == .history {
874 emit("history undo=\(state.undo.count) redo=\(state.redo.count)")
875 for (name, history) in [("undo", state.undo), ("redo", state.redo)] {
876 for (i, record) in history.reversed().enumerated() {
877 emit("\(name) \(i + 1): \(record.command)")
878 }
879 }
880 } else if [Kind.write, .writeAppend, .read, .edit].contains(kind) {
881 let file: [String]
882 if kind == .write || kind == .writeAppend {
883 file = (kind == .writeAppend ? state.files[command.argument] ?? [] : []) + contents(a, b)
884 state.files[command.argument] = file
885 } else {
886 guard let stored = state.files[command.argument] else {
887 throw EditorError("no such file")
888 }
889 file = stored
890 let inserted = newLines(file)
891 if kind == .edit {
892 state.lines = inserted
893 state.current = inserted.count
894 } else {
895 replace(at: a, count: 0, with: inserted)
896 state.current = a + inserted.count
897 }
898 }
899 emit("file \(command.argument) lines=\(file.count) bytes=\(byteCount(file))")
900 } else if kind == .files {
901 let files = state.files.keys.sorted().map { "\($0)=\(state.files[$0]!.count)" }
902 emit("files " + (files.isEmpty ? "none" : files.joined(separator: " ")))
903 }
904 return 0
905 }

907 func process(_ source: String, text: [String]) -> [String] {
908 let previous = state
909 let before = snapshot(source)
910 output = []
911 do {
912 let scanned = try scanAddress(source)
913 let form = try addressForm(scanned.elements)
914 let command = try parseCommand(scanned.command)
915 let category = addressKind(command.kind)
916 if category == 0 && form.count > 0 {
917 throw EditorError("unexpected address")
918 }
919 if category == 1 && form.count > 1 {
920 throw EditorError("too many addresses")
921 }
922 let values = try addresses(form)
923 let (a, b) = try range(command, values: values)
924 _ = try execute(command, from: a, through: b, text: text)
925 if last > 200 {
926 throw EditorError("buffer full")
927 }
928 if modifying(command) {
929 state.undo.append(before)
930 state.redo = []
931 }
932 return output
933 } catch let error as EditorError {
934 state = previous
935 return ["? " + error.reason]
936 } catch { fatalError("Unexpected error: \(error)") }
937 }
938}
939
940let editor = Editor()
941var result: [String] = []
942while let command = readLine(), command != "Q" {
943 var text: [String] = []
944 if hasText(command) {
945 while let line = readLine(), line != "." {
946 text.append(line)
947 }
948 }
949 result.append(contentsOf: editor.process(command, text: text))
950}
951if !result.isEmpty {
952 FileHandle.standardOutput.write(Data((result.joined(separator: "\n") + "\n").utf8))
953}
