43class Cursor {
44 final String text;
45 int position;
46 Cursor(this.text, [this.position = 0]);
47 bool get ended => position == text.length;
48 String get rest => text.substring(position);
49}
50
51bool digit(String c) => c.isNotEmpty && c.codeUnitAt(0) >= 48 && c.codeUnitAt(0) <= 57;
52bool digits(String text) => text.isNotEmpty && text.split('').every(digit);
53bool printSuffix(String c) => c == 'p' || c == 'n' || c == 'l';
54bool sortForm(String text) => text == 'sort' || text.startsWith('sort ');
55
56({String text, List<String?> replacement, String raw}) delimited(Cursor cursor, String delimiter) {
57 final start = cursor.position;
58 final value = StringBuffer();
59 final replacement = <String?>[];
60 while (!cursor.ended) {
61 var c = cursor.text[cursor.position++];
62 if (c == delimiter) {
63 return (
64 text: value.toString(), replacement: replacement,
65 raw: cursor.text.substring(start, cursor.position - 1),
66 );
67 }
68 final escaped = c == '\\';
69 if (escaped) {
70 if (cursor.ended) break;
71 c = cursor.text[cursor.position++];
72 }
73 value.write(c);
74 replacement.add(c == '&' && !escaped ? null : c);
75 }
76 throw CommandError('missing delimiter');
77}

79List<Element> scanAddress(Cursor cursor) {
80 final elements = <Element>[];
81 while (!cursor.ended) {
82 final c = cursor.text[cursor.position];
83 if (digit(c)) {
84 final start = cursor.position++;
85 while (!cursor.ended && digit(cursor.text[cursor.position])) cursor.position++;
86 elements.add(Element(ElementKind.number, value: int.parse(cursor.text.substring(start, cursor.position))));
87 } else if (c == '+' || c == '-') {
88 final start = ++cursor.position;
89 while (!cursor.ended && digit(cursor.text[cursor.position])) cursor.position++;
90 final value = cursor.position == start ? 1 : int.parse(cursor.text.substring(start, cursor.position));
91 elements.add(Element(ElementKind.offset, value: c == '+' ? value : -value));
92 } else if (c == "'") {
93 cursor.position++;
94 elements.add(Element(ElementKind.mark, symbol: cursor.ended ? '' : cursor.text[cursor.position++]));
95 } else if (c == '/' || c == '?') {
96 cursor.position++;
97 final pattern = delimited(cursor, c).text;
98 elements.add(Element(c == '/' ? ElementKind.forward : ElementKind.backward, text: pattern));
99 } else {
100 final kind = switch (c) {
101 '.' => ElementKind.current, r'$' => ElementKind.last,
102 ',' || ';' => ElementKind.separator, '%' => ElementKind.all, _ => null,
103 };
104 if (kind == null) break;
105 elements.add(Element(kind, symbol: c));
106 cursor.position++;
107 }
108 }
109 return elements;
110}

112bool expression(List<Element> elements) {
113 for (var i = 0; i < elements.length; i++) {
114 final kind = elements[i].kind;
115 if (kind == ElementKind.offset) continue;
116 if (i != 0 || kind == ElementKind.separator || kind == ElementKind.all) return false;
117 }
118 return true;
119}

121Address addressForm(List<Element> elements) {
122 if (elements.length == 1 && elements.first.kind == ElementKind.all) {
123 return const Address([Element(ElementKind.number, value: 1)], [Element(ElementKind.last)], ',');
124 }
125 final separators = [
126 for (var i = 0; i < elements.length; i++)
127 if (elements[i].kind == ElementKind.separator) i,
128 ];
129 if (separators.length > 1) throw CommandError('bad address');
130 if (separators.isEmpty) {
131 if (!expression(elements)) throw CommandError('bad address');
132 return Address(elements);
133 }
134 final split = separators.single;
135 final left = elements.sublist(0, split), right = elements.sublist(split + 1);
136 if (!expression(left) || !expression(right)) throw CommandError('bad address');
137 return Address(left, right, elements[split].symbol);
138}

140int optionalCount(String argument) {
141 if (argument.isEmpty) return 0;
142 if (!digits(argument) || int.parse(argument) == 0) throw CommandError('invalid suffix');
143 return int.parse(argument);

146String suffix(String argument) {
147 if (argument.isNotEmpty && !printSuffix(argument)) throw CommandError('invalid suffix');
148 return argument;

151String fileName(String argument) {
152 if (!RegExp(r'^ [a-z0-9]{1,8}$').hasMatch(argument)) throw CommandError('invalid file name');
153 return argument.substring(1);

156Command substitution(String argument) {
157 final command = Command(Kind.substitute)..count = 1;
158 var flags = argument;
159 if (!argument.split('').every((c) => c == 'g' || printSuffix(c) || digit(c))) {
160 final delimiter = argument[0];
161 if (delimiter == ' ' || delimiter == '\\') throw CommandError('missing delimiter');
162 final cursor = Cursor(argument, 1);
163 command.pattern = delimited(cursor, delimiter).text;
164 final replacement = delimited(cursor, delimiter);
165 if (replacement.raw != '%') command.replacement = replacement.replacement;
166 flags = cursor.rest;
167 }
168 var hasCount = false;
169 var position = 0;
170 while (position < flags.length) {
171 final c = flags[position++];
172 if (c == 'g' && !command.all) {
173 command.all = true;
174 } else if (printSuffix(c) && command.suffix.isEmpty) {
175 command.suffix = c;
176 } else if (digit(c) && !hasCount) {
177 final start = position - 1;
178 while (position < flags.length && digit(flags[position])) position++;
179 command.count = int.parse(flags.substring(start, position));
180 if (command.count == 0) throw CommandError('invalid suffix');
181 hasCount = true;
182 } else {
183 throw CommandError('invalid suffix');
184 }
185 }
186 return command;
187}

189Command parseCommand(String text) {
190 if (sortForm(text)) {
191 final options = text.length == 4 ? '' : text.substring(5);
192 if ((text.length > 4 && options.isEmpty) ||
193 options.split('').any((c) => !'ruin'.contains(c)) ||
194 options.split('').toSet().length != options.length ||
195 (options.contains('i') && options.contains('n'))) {
196 throw CommandError('invalid suffix');
197 }
198 return Command(Kind.sort)..options = options;
199 }
200 final first = text.isEmpty ? '' : text[0];
201 final argument = text.isEmpty ? '' : text.substring(1);
202 const plain = {
203 '': Kind.empty, 'a': Kind.append, 'i': Kind.insert, 'c': Kind.change,
204 'p': Kind.print, 'n': Kind.numbered, 'l': Kind.listed, '=': Kind.number,
205 '#': Kind.count, 'y': Kind.yank, 'K': Kind.marks, 'h': Kind.history, 'f': Kind.files,
206 };
207 final simple = plain[first];
208 if (simple != null) {
209 if (argument.isNotEmpty) throw CommandError('invalid suffix');
210 return Command(simple);
211 }
212 switch (first) {
213 case 'd': case 'x': case 'j': case 'J':
214 final kind = switch (first) {
215 'd' => Kind.delete, 'x' => Kind.put, 'j' => Kind.join, _ => Kind.joinSpaced,
216 };
217 return Command(kind)..suffix = suffix(argument);
218 case 'z': case '>': case '<': case 'u':
219 final kind = switch (first) {
220 'z' => Kind.window, '>' => Kind.indent, '<' => Kind.unindent, _ => Kind.undo,
221 };
222 return Command(kind)..count = optionalCount(argument);
223 case '~':
224 if (argument != '' && argument != 'u' && argument != 'l') throw CommandError('invalid suffix');
225 return Command(Kind.changeCase)..caseMode = argument;
226 case 'r':
227 if (argument.isEmpty || digits(argument)) return Command(Kind.redo)..count = optionalCount(argument);
228 return Command(Kind.read)..name = fileName(argument);
229 case 'w':
230 if (argument.isEmpty) return Command(Kind.writePrint);
231 return Command(Kind.write)..name = fileName(argument);
232 case 'W': case 'e':
233 return Command(first == 'W' ? Kind.writeAppend : Kind.edit)..name = fileName(argument);
234 case 'k':
235 if (!RegExp(r'^[a-z]$').hasMatch(argument)) throw CommandError('invalid mark');
236 return Command(Kind.mark)..name = argument;
237 case 'm': case 't':
238 final cursor = Cursor(argument);
239 final destination = scanAddress(cursor);
240 if (destination.isEmpty || !expression(destination)) throw CommandError('invalid destination');
241 return Command(first == 'm' ? Kind.move : Kind.copy)
242 ..destination = destination
243 ..suffix = suffix(cursor.rest);
244 case 's': return substitution(argument);
245 case 'g': case 'v':
246 if (argument.isEmpty || argument[0] == ' ' || argument[0] == '\\') throw CommandError('missing delimiter');
247 final cursor = Cursor(argument, 1);
248 final pattern = delimited(cursor, argument[0]).text;
249 final subcommand = cursor.ended ? 'p' : cursor.rest;
250 if (!'pnldmtjJs><~aicx'.contains(subcommand[0]) || sortForm(subcommand)) {
251 throw CommandError('invalid global command');
252 }
253 return Command(first == 'g' ? Kind.global : Kind.inverseGlobal)
254 ..pattern = pattern
255 ..subcommand = parseCommand(subcommand);
256 default: throw CommandError('unknown command');
257 }
258}

260bool hasText(String line) {
261 try {
262 final cursor = Cursor(line);
263 scanAddress(cursor);
264 final command = cursor.rest;
265 if (command == 'a' || command == 'i' || command == 'c') return true;
266 if (command.length < 2 || (command[0] != 'g' && command[0] != 'v') ||
267 command[1] == ' ' || command[1] == '\\') return false;
268 final global = Cursor(command, 2);
269 delimited(global, command[1]);
270 return global.rest == 'a' || global.rest == 'i' || global.rest == 'c';
271 } on CommandError {
272 return false;
273 }
274}

315bool noAddress(Kind kind) => {Kind.undo, Kind.redo, Kind.history, Kind.marks, Kind.files, Kind.edit}.contains(kind);
316bool oneAddress(Kind kind) => {Kind.append, Kind.insert, Kind.put, Kind.number, Kind.read, Kind.mark, Kind.window}.contains(kind);
317bool modifies(Command command) => switch (command.kind) {
318 Kind.append || Kind.insert || Kind.change || Kind.delete || Kind.put || Kind.join ||
319 Kind.joinSpaced || Kind.move || Kind.copy || Kind.substitute || Kind.indent ||
320 Kind.unindent || Kind.changeCase || Kind.sort || Kind.read || Kind.edit => true,
321 Kind.global || Kind.inverseGlobal => modifies(command.subcommand!),
322 _ => false,
323};

330 String pattern(String search) {
331 if (search.isNotEmpty) state.pattern = search;
332 return state.pattern ?? (throw CommandError('no previous pattern'));
333 }
334
335 int evaluate(List<Element> elements) {
336 var value = state.current;
337 for (final element in elements) {
338 switch (element.kind) {
339 case ElementKind.offset: value += element.value;
340 case ElementKind.number: value = element.value;
341 case ElementKind.current: value = state.current;
342 case ElementKind.last: value = state.lines.length;
343 case ElementKind.mark:
344 if (!RegExp(r'^[a-z]$').hasMatch(element.symbol)) throw CommandError('invalid mark');
345 final id = state.marks[element.symbol.codeUnitAt(0) - 97];
346 value = id == null ? 0 : state.lines.indexWhere((line) => line.id == id) + 1;
347 if (value == 0) throw CommandError('mark not set');
348 case ElementKind.forward: case ElementKind.backward:
349 final search = pattern(element.text);
350 final size = state.lines.length;
351 final forward = element.kind == ElementKind.forward;
352 var index = forward ? state.current : state.current - 2;
353 value = 0;
354 for (var checked = 0; checked < size; checked++) {
355 if (index >= size) index = 0;
356 if (index < 0) index = size - 1;
357 if (state.lines[index].text.contains(search)) {
358 value = index + 1;
359 break;
360 }
361 index += forward ? 1 : -1;
362 }
363 if (value == 0) throw CommandError('no match');
364 case ElementKind.separator: case ElementKind.all:
365 throw StateError('invalid address form');
366 }
367 }
368 return value;
369 }
370
371 List<int> evaluateAddress(Address address) {
372 if (address.count == 0) return [];
373 if (address.count == 1) return [evaluate(address.left)];
374 final left = address.left.isEmpty ? (address.separator == ',' ? 1 : state.current) : evaluate(address.left);
375 if (address.separator == ';') {
376 if (left < 0 || left > state.lines.length) throw CommandError('address out of range');
377 state.current = left;
378 }
379 final right = address.right!.isEmpty ? state.lines.length : evaluate(address.right!);
380 return [left, right];
381 }
382
383 (int, int) range(Command command, List<int> addresses) {
384 final kind = command.kind;
385 final size = state.lines.length;
386 if (noAddress(kind)) return (0, 0);
387 if (oneAddress(kind)) {
388 final a = addresses.isNotEmpty ? addresses.first : switch (kind) {
389 Kind.number || Kind.read => size, Kind.window => state.current + 1, _ => state.current,
390 };
391 final minimum = kind == Kind.mark || kind == Kind.window ? 1 : 0;
392 if (a < minimum || a > size) throw CommandError('address out of range');
393 return (a, a);
394 }
395 int a, b;
396 if (addresses.isNotEmpty) {
397 a = addresses.first;
398 b = addresses.last;
399 } else if ({Kind.writePrint, Kind.write, Kind.writeAppend, Kind.count}.contains(kind)) {
400 return (1, size);
401 } else if ({Kind.global, Kind.inverseGlobal, Kind.sort}.contains(kind)) {
402 a = 1;
403 b = size;
404 } else {
405 a = state.current;
406 b = a + (kind == Kind.join || kind == Kind.joinSpaced ? 1 : 0);
407 }
408 if (a < 1 || b < 1 || a > size || b > size) throw CommandError('address out of range');
409 if (a > b || a == b && (kind == Kind.join || kind == Kind.joinSpaced)) throw CommandError('invalid range');
410 return (a, b);
411 }

413 void process(String line, List<String> text, EditorState before) {
414 final cursor = Cursor(line);
415 final address = addressForm(scanAddress(cursor));
416 final command = parseCommand(cursor.rest);
417 if (noAddress(command.kind) && address.count != 0) throw CommandError('unexpected address');
418 if (oneAddress(command.kind) && address.count == 2) throw CommandError('too many addresses');
419 final (a, b) = range(command, evaluateAddress(address));
420 execute(command, a, b, text);
421 if (state.lines.length > 200) throw CommandError('buffer full');
422 if (modifies(command)) {
423 state.undo.add(HistoryRecord(line, [...before.lines], before.current));
424 state.redo.clear();
425 }
426 }

428 void printLine(int number, [String form = 'p']) {
429 var text = state.lines[number - 1].text;
430 if (form == 'n') {
431 output.add('${number.toString().padLeft(3)} $text');
432 } else if (form == 'l') {
433 text = text.replaceAll('\\', '\\\\') + r'$';
434 while (text.length > 40) {
435 output.add('${text.substring(0, 39)}\\');
436 text = text.substring(39);
437 }
438 output.add(text);
439 } else {
440 output.add(text);
441 }
442 }

444 List<String> contents(int a, int b) => state.lines.sublist(a - 1, b).map((line) => line.text).toList();
445 int bytes(Iterable<String> lines) => lines.fold(0, (sum, line) => sum + line.length + 1);

447 void insert(int position, Iterable<String> text) {
448 final lines = text.map((value) => Line(state.nextId++, value)).toList();
449 state.lines.insertAll(position, lines);
450 state.current = position + lines.length;
451 }

453 void history(bool undo, int count) {
454 final source = undo ? state.undo : state.redo, target = undo ? state.redo : state.undo;
455 if (source.length < count) throw CommandError(undo ? 'nothing to undo' : 'nothing to redo');
456 for (var i = 0; i < count; i++) {
457 final record = source.removeLast();
458 target.add(HistoryRecord(record.command, [...state.lines], state.current));
459 state.lines = [...record.lines];
460 state.current = record.current;
461 }

464 int leadingInteger(String text) {
465 final match = RegExp(r'^-?[0-9]+').firstMatch(text);
466 return match == null ? 0 : int.parse(match[0]!);

469 void sort(Command command, int a, int b) {
470 final lines = state.lines.sublist(a - 1, b);
471 final numeric = command.options.contains('n'), reverse = command.options.contains('r');
472 String key(Line line) => command.options.contains('i') ? line.text.toLowerCase() : line.text;
473 int compare(Line left, Line right) => numeric
474 ? leadingInteger(left.text).compareTo(leadingInteger(right.text)) : key(left).compareTo(key(right));
475 final ordered = [for (var i = 0; i < lines.length; i++) (i, lines[i])];
476 ordered.sort((left, right) {
477 var comparison = compare(left.$2, right.$2);
478 if (reverse) comparison = -comparison;
479 return comparison == 0 ? left.$1.compareTo(right.$1) : comparison;
480 });
481 final result = <Line>[];
482 for (final (_, line) in ordered) {
483 if (command.options.contains('u') && result.isNotEmpty && compare(result.last, line) == 0) continue;
484 result.add(line);
485 }
486 state.lines.replaceRange(a - 1, b, result);
487 state.current = a + result.length - 1;
488 }

490 int substitute(Command command, int a, int b, bool global) {
491 final search = pattern(command.pattern);
492 final replacement = command.replacement ?? state.replacement ?? (throw CommandError('no previous substitution'));
493 state.replacement = replacement;
494 final expanded = StringBuffer();
495 for (final part in replacement) expanded.write(part ?? search);
496 final value = expanded.toString();
497 var substituted = 0;
498 for (var number = a; number <= b; number++) {
499 final line = state.lines[number - 1];
500 final result = StringBuffer();
501 var position = 0, occurrences = 0;
502 while (true) {
503 final match = line.text.indexOf(search, position);
504 if (match < 0) break;
505 result.write(line.text.substring(position, match));
506 occurrences++;
507 result.write(occurrences == command.count || command.all && occurrences >= command.count ? value : search);
508 position = match + search.length;
509 }
510 if (occurrences < command.count) continue;
511 result.write(line.text.substring(position));
512 final changed = result.toString();
513 if (changed.length > 80) throw CommandError('line too long');
514 if (changed.isEmpty || changed.endsWith(' ')) throw CommandError('bad line');
515 state.lines[number - 1] = Line(line.id, changed);
516 state.current = number;
517 substituted++;
518 }
519 if (substituted == 0 && !global) throw CommandError('no match');
520 if (substituted != 0 && command.suffix.isNotEmpty) printLine(state.current, command.suffix);
521 return substituted;
522 }

524 int execute(Command command, int a, int b, List<String> text, [bool global = false]) {
525 final kind = command.kind;
526 switch (kind) {
527 case Kind.empty:
528 printLine(b);
529 state.current = b;
530 case Kind.print: case Kind.numbered: case Kind.listed:
531 final form = kind == Kind.print ? 'p' : kind == Kind.numbered ? 'n' : 'l';
532 for (var i = a; i <= b; i++) printLine(i, form);
533 state.current = b;
534 case Kind.window:
535 if (command.count != 0) state.displayCount = command.count;
536 final end = min(state.lines.length, a + state.displayCount - 1);
537 for (var i = a; i <= end; i++) printLine(i);
538 state.current = end;
539 case Kind.number: output.add('$a');
540 case Kind.count:
541 final counted = contents(a, b);
542 final words = counted.fold<int>(0, (sum, line) => sum + line.split(' ').where((word) => word.isNotEmpty).length);
543 final chars = counted.fold<int>(0, (sum, line) => sum + line.length);
544 output.add('count lines=${counted.length} words=$words chars=$chars');
545 case Kind.writePrint:
546 for (var i = a; i <= b; i++) printLine(i);
547 output.add('lines=${b - a + 1} bytes=${bytes(contents(a, b))}');
548 case Kind.append: insert(a, text);
549 case Kind.insert: insert(max(a, 1) - 1, text);
550 case Kind.change: case Kind.delete:
551 state.cut = contents(a, b);
552 state.lines.removeRange(a - 1, b);
553 if (kind == Kind.change && text.isNotEmpty) {
554 insert(a - 1, text);
555 } else {
556 state.current = min(a, state.lines.length);
557 }
558 case Kind.yank: state.cut = contents(a, b);
559 case Kind.put:
560 if (state.cut.isEmpty) throw CommandError('nothing to put');
561 insert(a, state.cut);
562 case Kind.move: case Kind.copy:
563 final destination = evaluate(command.destination);
564 if (destination < 0 || destination > state.lines.length) throw CommandError('address out of range');
565 if (kind == Kind.copy) {
566 insert(destination, contents(a, b));
567 } else {
568 if (a <= destination && destination < b) throw CommandError('invalid destination');
569 final moved = state.lines.sublist(a - 1, b);
570 state.lines.removeRange(a - 1, b);
571 final position = destination < a ? destination : destination - moved.length;
572 state.lines.insertAll(position, moved);
573 state.current = position + moved.length;
574 }
575 case Kind.join: case Kind.joinSpaced:
576 final pieces = contents(a, b);
577 if (kind == Kind.joinSpaced) {
578 for (var i = 1; i < pieces.length; i++) pieces[i] = pieces[i].trimLeft();
579 }
580 final joined = pieces.join(kind == Kind.join ? '' : ' ');
581 if (joined.length > 80) throw CommandError('line too long');
582 state.lines[a - 1] = Line(state.lines[a - 1].id, joined);
583 state.lines.removeRange(a, b);
584 state.current = a;
585 case Kind.indent: case Kind.unindent: case Kind.changeCase:
586 final width = command.count == 0 ? 4 : command.count;
587 for (var i = a - 1; i < b; i++) {
588 final line = state.lines[i];
589 var changed = line.text;
590 if (kind == Kind.indent) {
591 if (changed.length + width > 80) throw CommandError('line too long');
592 changed = ' ' * width + changed;
593 } else if (kind == Kind.unindent) {
594 var removed = 0;
595 while (removed < width && removed < changed.length && changed[removed] == ' ') removed++;
596 changed = changed.substring(removed);
597 } else if (command.caseMode == 'u') {
598 changed = changed.toUpperCase();
599 } else if (command.caseMode == 'l') {
600 changed = changed.toLowerCase();
601 } else {
602 changed = String.fromCharCodes(changed.codeUnits.map((c) => c >= 65 && c <= 90 ? c + 32 :
603 c >= 97 && c <= 122 ? c - 32 : c));
604 }
605 state.lines[i] = Line(line.id, changed);
606 }
607 state.current = b;
608 case Kind.sort: sort(command, a, b);
609 case Kind.substitute: return substitute(command, a, b, global);
610 case Kind.mark: state.marks[command.name.codeUnitAt(0) - 97] = state.lines[a - 1].id;
611 case Kind.marks:
612 final marks = <String>[];
613 for (var i = 0; i < 26; i++) {
614 final id = state.marks[i];
615 final number = id == null ? 0 : state.lines.indexWhere((line) => line.id == id) + 1;
616 if (number != 0) marks.add('${String.fromCharCode(97 + i)}=$number');
617 }
618 output.add('marks ${marks.isEmpty ? 'none' : marks.join(' ')}');
619 case Kind.undo: case Kind.redo: history(kind == Kind.undo, command.count == 0 ? 1 : command.count);
620 case Kind.history:
621 output.add('history undo=${state.undo.length} redo=${state.redo.length}');
622 for (var i = 0; i < state.undo.length; i++) output.add('undo ${i + 1}: ${state.undo[state.undo.length - 1 - i].command}');
623 for (var i = 0; i < state.redo.length; i++) output.add('redo ${i + 1}: ${state.redo[state.redo.length - 1 - i].command}');
624 case Kind.global: case Kind.inverseGlobal:
625 final search = pattern(command.pattern);
626 final selected = state.lines.sublist(a - 1, b)
627 .where((line) => line.text.contains(search) == (kind == Kind.global)).map((line) => line.id).toList();
628 if (selected.isEmpty) throw CommandError('no match');
629 final subcommand = command.subcommand!;
630 var substituted = 0;
631 for (final id in selected) {
632 final index = state.lines.indexWhere((line) => line.id == id);
633 if (index < 0) continue;
634 state.current = index + 1;
635 final (first, last) = range(subcommand, []);
636 substituted += execute(subcommand, first, last, text, true);
637 }
638 if (subcommand.kind == Kind.substitute && substituted == 0) throw CommandError('no match');
639 case Kind.write: case Kind.writeAppend: case Kind.read: case Kind.edit:
640 if (kind == Kind.write || kind == Kind.writeAppend) {
641 final written = contents(a, b);
642 state.files[command.name] = kind == Kind.writeAppend ? [...?state.files[command.name], ...written] : written;
643 } else {
644 final read = state.files[command.name];
645 if (read == null) throw CommandError('no such file');
646 if (kind == Kind.edit) state.lines.clear();
647 insert(kind == Kind.edit ? 0 : a, read);
648 }
649 final file = state.files[command.name]!;
650 output.add('file ${command.name} lines=${file.length} bytes=${bytes(file)}');
651 case Kind.files:
652 final names = state.files.keys.toList()..sort();
653 output.add('files ${names.isEmpty ? 'none' : names.map((name) => '$name=${state.files[name]!.length}').join(' ')}');
654 }
655 if (command.suffix.isNotEmpty && state.current >= 1) printLine(state.current, command.suffix);
656 return 0;
657 }
658}

660void main() {
661 var state = EditorState();
662 while (true) {
663 final line = stdin.readLineSync();
664 if (line == null || line == 'Q') break;
665 final text = <String>[];
666 if (hasText(line)) {
667 while (true) {
668 final value = stdin.readLineSync();
669 if (value == null || value == '.') break;
670 text.add(value);
671 }
672 }
673 try {
674 final next = EditorState.copy(state);
675 final editor = Editor(next)..process(line, text, state);
676 state = next;
677 if (editor.output.isNotEmpty) stdout.writeln(editor.output.join('\n'));
678 } on CommandError catch (error) {
679 stdout.writeln('? ${error.reason}');
680 }
681 }
