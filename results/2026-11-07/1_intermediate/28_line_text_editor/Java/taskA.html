22 static boolean digit(char c) { return c >= '0' && c <= '9'; }
23 static boolean lower(char c) { return c >= 'a' && c <= 'z'; }
24 static boolean upper(char c) { return c >= 'A' && c <= 'Z'; }
25 static boolean digits(String s) { return !s.isEmpty() && s.chars().allMatch(c -> digit((char)c)); }
26 static boolean oneOf(char c, String set) { return set.indexOf(c) >= 0; }

28 record Atom(char character, boolean matched) {}
29 record Delimited(String text, List<Atom> atoms, int end) {}

31 static Delimited delimited(String s, int position, char delimiter) {
32 StringBuilder text = new StringBuilder();
33 List<Atom> atoms = new ArrayList<>();
34 while (position < s.length()) {
35 char character = s.charAt(position++);
36 if (character == delimiter) return new Delimited(text.toString(), List.copyOf(atoms), position);
37 boolean escaped = character == '\\';
38 if (escaped) {
39 require(position < s.length(), "missing delimiter");
40 character = s.charAt(position++);
41 }
42 text.append(character);
43 atoms.add(new Atom(character, character == '&' && !escaped));
44 }
45 throw new CommandError("missing delimiter");
46 }

48 record Token(char kind, int number, char name, String search) {
49 Token(char kind) { this(kind, 0, '\0', ""); }
50 Token(char kind, int number) { this(kind, number, '\0', ""); }
51 }
52 record Lexed(List<Token> tokens, int end) {}
53
54 static Lexed lexAddresses(String s) {
55 List<Token> tokens = new ArrayList<>();
56 int position = 0;
57 while (position < s.length()) {
58 char c = s.charAt(position);
59 Token token;
60 if (digit(c)) {
61 int start = position;
62 while (position < s.length() && digit(s.charAt(position))) position++;
63 token = new Token('N', Integer.parseInt(s.substring(start, position)));
64 } else if (c == '+' || c == '-') {
65 int start = ++position;
66 while (position < s.length() && digit(s.charAt(position))) position++;
67 int value = position == start ? 1 : Integer.parseInt(s.substring(start, position));
68 token = new Token('O', c == '-' ? -value : value);
69 } else if (oneOf(c, ".$,;%")) {
70 position++;
71 token = new Token(c);
72 } else if (c == '\'') {
73 position++;
74 char name = position < s.length() ? s.charAt(position++) : '\0';
75 token = new Token(c, 0, name, "");
76 } else if (c == '/' || c == '?') {
77 Delimited search = delimited(s, position + 1, c);
78 position = search.end;
79 token = new Token(c, 0, '\0', search.text);
80 } else break;
81 tokens.add(token);
82 }
83 return new Lexed(tokens, position);
84 }

86 static boolean withText(String line) {
87 try {
88 String part = line.substring(lexAddresses(line).end);
89 if (part.equals("a") || part.equals("i") || part.equals("c")) return true;
90 if (part.length() >= 2 && oneOf(part.charAt(0), "gv") && !oneOf(part.charAt(1), " \\")) {
91 int end = delimited(part, 2, part.charAt(1)).end;
92 String rest = part.substring(end);
93 return rest.equals("a") || rest.equals("i") || rest.equals("c");
94 }
95 } catch (CommandError ignored) {
96 return false;
97 }
98 return false;
99 }

101 static boolean validAddress(List<Token> tokens) {
102 for (int i = 0; i < tokens.size(); i++) {
103 char kind = tokens.get(i).kind;
104 if (kind != 'O' && (i != 0 || !oneOf(kind, "N.$'/?"))) return false;
105 }
106 return true;
107 }

109 record Addresses(int count, List<Token> left, List<Token> right, boolean semicolon) {
110 static Addresses parse(List<Token> tokens) {
111 if (tokens.isEmpty()) return new Addresses(0, List.of(), List.of(), false);
112 if (tokens.size() == 1 && tokens.getFirst().kind == '%')
113 return new Addresses(2, List.of(new Token('N', 1)), List.of(new Token('$')), false);
114 int separator = -1;
115 for (int i = 0; i < tokens.size(); i++) {
116 if (oneOf(tokens.get(i).kind, ",;")) {
117 require(separator < 0, "bad address");
118 separator = i;
119 }
120 }
121 List<Token> left = separator < 0 ? tokens : tokens.subList(0, separator);
122 List<Token> right = separator < 0 ? List.of() : tokens.subList(separator + 1, tokens.size());
123 require(validAddress(left) && validAddress(right), "bad address");
124 return new Addresses(separator < 0 ? 1 : 2, left, right,
125 separator >= 0 && tokens.get(separator).kind == ';');
126 }

167 static Command parse(String part) {
168 Command c = new Command();
169 if (sortForm(part)) {
170 c.kind = Kind.SORT;
171 require(!part.equals("sort "), "invalid suffix");
172 String options = part.length() > 4 ? part.substring(5) : "";
173 for (int i = 0; i < options.length(); i++) {
174 switch (options.charAt(i)) {
175 case 'r' -> { require(!c.reverse, "invalid suffix"); c.reverse = true; }
176 case 'u' -> { require(!c.unique, "invalid suffix"); c.unique = true; }
177 case 'i' -> { require(!c.insensitive, "invalid suffix"); c.insensitive = true; }
178 case 'n' -> { require(!c.numeric, "invalid suffix"); c.numeric = true; }
179 default -> throw new CommandError("invalid suffix");
180 }
181 }
182 require(!c.insensitive || !c.numeric, "invalid suffix");
183 return c;
184 }
185 if (part.isEmpty()) return c;
186 char first = part.charAt(0);
187 String arg = part.substring(1);
188 String simple = "aicpnl=#yKhf";
189 Kind[] kinds = {Kind.APPEND, Kind.INSERT, Kind.CHANGE, Kind.PRINT, Kind.NUMBER, Kind.LIST,
190 Kind.EQUALS, Kind.COUNT, Kind.YANK, Kind.MARKS, Kind.HISTORY, Kind.FILES};
191 int index = simple.indexOf(first);
192 if (index >= 0) {
193 c.kind = kinds[index];
194 require(arg.isEmpty(), "invalid suffix");
195 return c;
196 }
197 switch (first) {
198 case 'd', 'x', 'j', 'J' -> {
199 c.kind = switch (first) { case 'd' -> Kind.DELETE; case 'x' -> Kind.PUT;
200 case 'j' -> Kind.JOIN; default -> Kind.JOIN_SPACE; };
201 c.suffix = printSuffix(arg);
202 }
203 case 'z', '>', '<', 'u' -> {
204 c.kind = switch (first) { case 'z' -> Kind.PAGE; case '>' -> Kind.INDENT;
205 case '<' -> Kind.UNINDENT; default -> Kind.UNDO; };
206 c.number = positive(arg);
207 }
208 case '~' -> {
209 c.kind = Kind.CASE;
210 require(arg.isEmpty() || arg.equals("u") || arg.equals("l"), "invalid suffix");
211 c.option = arg.isEmpty() ? '\0' : arg.charAt(0);
212 }
213 case 'r' -> {
214 if (arg.isEmpty() || digits(arg)) { c.kind = Kind.REDO; c.number = positive(arg); }
215 else { c.kind = Kind.READ; c.name = filename(arg); }
216 }
217 case 'w', 'W', 'e' -> {
218 if (first == 'w' && arg.isEmpty()) c.kind = Kind.WRITE_PRINT;
219 else {
220 c.kind = first == 'w' ? Kind.WRITE : first == 'W' ? Kind.WRITE_APPEND : Kind.EDIT;
221 c.name = filename(arg);
222 }
223 }
224 case 'k' -> {
225 c.kind = Kind.MARK;
226 require(arg.length() == 1 && lower(arg.charAt(0)), "invalid mark");
227 c.option = arg.charAt(0);
228 }
229 case 'm', 't' -> {
230 c.kind = first == 'm' ? Kind.MOVE : Kind.COPY;
231 Lexed destination = lexAddresses(arg);
232 require(!destination.tokens.isEmpty() && validAddress(destination.tokens), "invalid destination");
233 c.destination = destination.tokens;
234 c.suffix = printSuffix(arg.substring(destination.end));
235 }
236 case 's' -> {
237 c.kind = Kind.SUBSTITUTE;
238 c.number = 1;
239 boolean repeated = arg.chars().allMatch(ch -> oneOf((char)ch, "gpnl") || digit((char)ch));
240 int position = 0;
241 if (!repeated) {
242 char delimiter = arg.charAt(0);
243 require(!oneOf(delimiter, " \\"), "missing delimiter");
244 Delimited search = delimited(arg, 1, delimiter);
245 Delimited replacement = delimited(arg, search.end, delimiter);
246 c.search = search.text;
247 if (!arg.substring(search.end, replacement.end - 1).equals("%")) c.replacement = replacement.atoms;
248 position = replacement.end;
249 }
250 boolean counted = false;
251 while (position < arg.length()) {
252 char flag = arg.charAt(position++);
253 if (flag == 'g') {
254 require(!c.all, "invalid suffix");
255 c.all = true;
256 } else if (oneOf(flag, "pnl")) {
257 require(c.suffix == '\0', "invalid suffix");
258 c.suffix = flag;
259 } else if (digit(flag)) {
260 require(!counted, "invalid suffix");
261 counted = true;
262 int start = position - 1;
263 while (position < arg.length() && digit(arg.charAt(position))) position++;
264 c.number = Integer.parseInt(arg.substring(start, position));
265 require(c.number > 0, "invalid suffix");
266 } else throw new CommandError("invalid suffix");
267 }
268 }
269 case 'g', 'v' -> {
270 c.kind = first == 'g' ? Kind.GLOBAL : Kind.INVERSE;
271 require(!arg.isEmpty() && !oneOf(arg.charAt(0), " \\"), "missing delimiter");
272 Delimited search = delimited(arg, 1, arg.charAt(0));
273 c.search = search.text;
274 String body = arg.substring(search.end);
275 if (body.isEmpty()) body = "p";
276 require(oneOf(body.charAt(0), "pnldmtjJs><~aicx") && !sortForm(body), "invalid global command");
277 c.sub = parse(body);
278 }
279 default -> throw new CommandError("unknown command");
280 }
281 return c;
282 }

347 int address(List<Token> tokens) {
348 int value = current;
349 for (Token token : tokens) {
350 switch (token.kind) {
351 case 'O' -> value += token.number;
352 case 'N' -> value = token.number;
353 case '.' -> value = current;
354 case '$' -> value = buffer.size();
355 case '\'' -> {
356 require(lower(token.name), "invalid mark");
357 value = locate(marks[token.name - 'a']);
358 require(value != 0, "mark not set");
359 }
360 case '/', '?' -> {
361 String pattern = resolvePattern(token.search);
362 int size = buffer.size(), start = token.kind == '?' && current == 0 ? size + 1 : current;
363 value = 0;
364 for (int step = 1; step <= size; step++) {
365 int index = token.kind == '/' ? (start + step - 1) % size : Math.floorMod(start - step - 1, size);
366 if (buffer.get(index).text.contains(pattern)) { value = index + 1; break; }
367 }
368 require(value != 0, "no match");
369 }
370 default -> throw new AssertionError();
371 }
372 }
373 return value;
374 }

376 Range evaluate(Addresses addresses) {
377 if (addresses.count == 0) return new Range(0, 0);
378 int a = addresses.left.isEmpty() ? (addresses.semicolon ? current : 1) : address(addresses.left);
379 int b = 0;
380 if (addresses.count == 2) {
381 if (addresses.semicolon) {
382 require(a >= 0 && a <= buffer.size(), "address out of range");
383 current = a;
384 }
385 b = addresses.right.isEmpty() ? buffer.size() : address(addresses.right);
386 }
387 return new Range(a, b);
388 }

390 Range checkedRange(Command c, int count, Range values) {
391 int size = buffer.size(), a = values.first, b = values.last;
392 if (c.addressKind() == 0) return new Range(0, 0);
393 if (c.addressKind() == 1) {
394 if (count == 0) a = switch (c.kind) {
395 case EQUALS, READ -> size;
396 case PAGE -> current + 1;
397 default -> current;
398 };
399 int minimum = c.kind == Kind.MARK || c.kind == Kind.PAGE ? 1 : 0;
400 require(a >= minimum && a <= size, "address out of range");
401 return new Range(a, a);
402 }
403 if (count == 0) {
404 switch (c.kind) {
405 case WRITE, WRITE_APPEND, WRITE_PRINT, COUNT -> { return new Range(1, size); }
406 case GLOBAL, INVERSE, SORT -> { a = 1; b = size; }
407 case JOIN, JOIN_SPACE -> { a = current; b = current + 1; }
408 default -> { a = current; b = current; }
409 }
410 } else if (count == 1) b = a;
411 require(a >= 1 && a <= size && b >= 1 && b <= size, "address out of range");
412 require(a <= b && (a != b || c.kind != Kind.JOIN && c.kind != Kind.JOIN_SPACE), "invalid range");
413 return new Range(a, b);
414 }

416 void printLine(int number, char form, StringBuilder output) {
417 String text = buffer.get(number - 1).text;
418 if (form == 'n') emit(output, "%3d %s\n", number, text);
419 else if (form == 'l') {
420 String escaped = text.replace("\\", "\\\\") + "$";
421 int position = 0;
422 while (escaped.length() - position > 40) {
423 emit(output, "%s\\\n", escaped.substring(position, position + 39));
424 position += 39;
425 }
426 emit(output, "%s\n", escaped.substring(position));
427 } else emit(output, "%s\n", text);
428 }

442 boolean substitute(Command c, Range range, StringBuilder output, boolean inGlobal) {
443 String pattern = resolvePattern(c.search);
444 List<Atom> used = c.replacement != null ? c.replacement : replacement;
445 require(used != null, "no previous substitution");
446 replacement = used;
447 StringBuilder expanded = new StringBuilder();
448 for (Atom atom : used) {
449 if (atom.matched) expanded.append(pattern);
450 else expanded.append(atom.character);
451 }
452 int last = 0;
453 for (int i = range.first; i <= range.last; i++) {
454 String old = buffer.get(i - 1).text;
455 StringBuilder result = new StringBuilder();
456 int position = 0, occurrence = 0, start;
457 boolean changed = false;
458 while ((start = old.indexOf(pattern, position)) >= 0) {
459 result.append(old, position, start);
460 occurrence++;
461 if (occurrence == c.number || c.all && occurrence > c.number) {
462 result.append(expanded);
463 changed = true;
464 } else result.append(pattern);
465 position = start + pattern.length();
466 }
467 result.append(old, position, old.length());
468 if (changed) {
469 require(result.length() <= 80, "line too long");
470 require(!result.isEmpty() && result.charAt(result.length() - 1) != ' ', "bad line");
471 setText(i - 1, result.toString());
472 last = i;
473 }
474 }
475 require(last != 0 || inGlobal, "no match");
476 if (last != 0) {
477 current = last;
478 if (c.suffix != '\0') printLine(last, c.suffix, output);
479 }
480 return last != 0;
481 }

483 void sortLines(Command c, Range range) {
484 List<Line> lines = new ArrayList<>(buffer.subList(range.first - 1, range.last));
485 Comparator<Line> key = c.numeric
486 ? Comparator.comparingLong((Line line) -> leadingInteger(line.text))
487 : Comparator.comparing((Line line) -> c.insensitive ? line.text.toLowerCase(Locale.ROOT) : line.text);
488 lines.sort(c.reverse ? key.reversed() : key);
489 if (c.unique) {
490 List<Line> unique = new ArrayList<>();
491 for (Line line : lines)
492 if (unique.isEmpty() || key.compare(unique.getLast(), line) != 0) unique.add(line);
493 lines = unique;
494 }
495 buffer.subList(range.first - 1, range.last).clear();
496 buffer.addAll(range.first - 1, lines);
497 current = range.first + lines.size() - 1;
498 }

500 void fileCommand(Command c, Range range, StringBuilder output) {
501 List<String> file = files.get(c.name);
502 if (c.kind == Kind.WRITE || c.kind == Kind.WRITE_APPEND) {
503 List<String> changed = new ArrayList<>();
504 if (c.kind == Kind.WRITE_APPEND && file != null) changed.addAll(file);
505 changed.addAll(contents(range));
506 file = List.copyOf(changed);
507 files.put(c.name, file);
508 } else {
509 require(file != null, "no such file");
510 int after = range.first;
511 if (c.kind == Kind.EDIT) { buffer.clear(); after = 0; }
512 insert(after, file);
513 current = after + file.size();
514 }
515 emit(output, "file %s lines=%d bytes=%d\n", c.name, file.size(), fileBytes(file));
516 }

518 void transferHistory(boolean isRedo, int count) {
519 ArrayList<HistoryRecord> from = isRedo ? redo : undo, to = isRedo ? undo : redo;
520 require(from.size() >= count, isRedo ? "nothing to redo" : "nothing to undo");

522 HistoryRecord record = from.removeLast();
523 to.add(new HistoryRecord(record.command, snapshot()));

525 current = record.state.current;
526 }

529 boolean execute(Command c, Range range, List<String> text, StringBuilder output, boolean inGlobal) {
530 int a = range.first, b = range.last;
531 switch (c.kind) {
532 case EMPTY -> { printLine(b, 'p', output); current = b; }
533 case PRINT, NUMBER, LIST -> {
534 char form = c.kind == Kind.PRINT ? 'p' : c.kind == Kind.NUMBER ? 'n' : 'l';
535 for (int i = a; i <= b; i++) printLine(i, form, output);
536 current = b;
537 }
538 case PAGE -> {
539 if (c.number != 0) display = c.number;
540 int last = Math.min(a + display - 1, buffer.size());
541 for (int i = a; i <= last; i++) printLine(i, 'p', output);
542 current = last;
543 }
544 case EQUALS -> emit(output, "%d\n", a);
545 case COUNT, WRITE_PRINT -> {
546 int words = 0, chars = 0;
547 for (int i = a; i <= b; i++) {
548 String value = buffer.get(i - 1).text;
549 chars += value.length();
550 for (int j = 0; j < value.length(); j++)
551 if (value.charAt(j) != ' ' && (j == 0 || value.charAt(j - 1) == ' ')) words++;
552 if (c.kind == Kind.WRITE_PRINT) printLine(i, 'p', output);
553 }
554 int count = b - a + 1;
555 if (c.kind == Kind.COUNT) emit(output, "count lines=%d words=%d chars=%d\n", count, words, chars);
556 else emit(output, "lines=%d bytes=%d\n", count, chars + count);
557 }
558 case APPEND, INSERT -> {
559 int after = c.kind == Kind.APPEND ? a : Math.max(a, 1) - 1;
560 insert(after, text);
561 current = after + text.size();
562 }
563 case CHANGE, DELETE -> {
564 cut = contents(range);
565 buffer.subList(a - 1, b).clear();
566 if (c.kind == Kind.CHANGE) insert(a - 1, text);
567 current = c.kind == Kind.CHANGE && !text.isEmpty() ? a + text.size() - 1 : Math.min(a, buffer.size());
568 }
569 case YANK -> cut = contents(range);
570 case PUT -> {
571 require(!cut.isEmpty(), "nothing to put");
572 insert(a, cut);
573 current = a + cut.size();
574 }
575 case MOVE, COPY -> {
576 int destination = address(c.destination);
577 require(destination >= 0 && destination <= buffer.size(), "address out of range");
578 require(c.kind != Kind.MOVE || destination < a || destination >= b, "invalid destination");
579 List<Line> moved = new ArrayList<>(buffer.subList(a - 1, b));
580 if (c.kind == Kind.MOVE) {
581 buffer.subList(a - 1, b).clear();
582 if (destination >= b) destination -= moved.size();
583 } else {
584 for (int i = 0; i < moved.size(); i++) moved.set(i, new Line(nextId++, moved.get(i).text));
585 }
586 buffer.addAll(destination, moved);
587 current = destination + moved.size();
588 }
589 case JOIN, JOIN_SPACE -> {
590 StringBuilder joined = new StringBuilder(buffer.get(a - 1).text);
591 for (int i = a + 1; i <= b; i++) {
592 String value = buffer.get(i - 1).text;
593 if (c.kind == Kind.JOIN_SPACE) {
594 joined.append(' ');
595 int start = 0;
596 while (value.charAt(start) == ' ') start++;
597 value = value.substring(start);
598 }
599 joined.append(value);
600 }
601 require(joined.length() <= 80, "line too long");
602 setText(a - 1, joined.toString());
603 buffer.subList(a, b).clear();
604 current = a;
605 }
606 case INDENT, UNINDENT, CASE -> {
607 int width = c.number == 0 ? 4 : c.number;
608 for (int i = a; i <= b; i++) {
609 String value = buffer.get(i - 1).text;
610 if (c.kind == Kind.INDENT) {
611 require(value.length() + width <= 80, "line too long");
612 value = " ".repeat(width) + value;
613 } else if (c.kind == Kind.UNINDENT) {
614 int removed = 0;
615 while (removed < width && value.charAt(removed) == ' ') removed++;
616 value = value.substring(removed);
617 } else {
618 char[] converted = value.toCharArray();
619 for (int j = 0; j < converted.length; j++) {
620 if (lower(converted[j]) && c.option != 'l') converted[j] = (char)(converted[j] - 'a' + 'A');
621 else if (upper(converted[j]) && c.option != 'u') converted[j] = (char)(converted[j] - 'A' + 'a');
622 }
623 value = new String(converted);
624 }
625 setText(i - 1, value);
626 }
627 current = b;
628 }
629 case SORT -> sortLines(c, range);
630 case SUBSTITUTE -> { return substitute(c, range, output, inGlobal); }
631 case MARK -> marks[c.option - 'a'] = buffer.get(a - 1).id;
632 case MARKS -> {
633 output.append("marks");
634 boolean any = false;
635 for (int i = 0; i < marks.length; i++) {
636 int position = locate(marks[i]);
637 if (position != 0) { emit(output, " %c=%d", (char)('a' + i), position); any = true; }
638 }
639 output.append(any ? "\n" : " none\n");
640 }
641 case UNDO, REDO -> transferHistory(c.kind == Kind.REDO, c.number == 0 ? 1 : c.number);
642 case HISTORY -> {
643 emit(output, "history undo=%d redo=%d\n", undo.size(), redo.size());
644 for (int side = 0; side < 2; side++) {
645 List<HistoryRecord> records = side == 0 ? undo : redo;
646 for (int i = records.size() - 1; i >= 0; i--)
647 emit(output, "%s %d: %s\n", side == 0 ? "undo" : "redo", records.size() - i, records.get(i).command);
648 }
649 }
650 case READ, WRITE, WRITE_APPEND, EDIT -> fileCommand(c, range, output);
651 case FILES -> {
652 output.append("files");
653 for (Map.Entry<String, List<String>> file : files.entrySet()) emit(output, " %s=%d", file.getKey(), file.getValue().size());
654 output.append(files.isEmpty() ? " none\n" : "\n");
655 }
656 case GLOBAL, INVERSE -> {
657 String pattern = resolvePattern(c.search);
658 List<Long> selected = new ArrayList<>();
659 for (Line line : buffer.subList(a - 1, b))
660 if (line.text.contains(pattern) == (c.kind == Kind.GLOBAL)) selected.add(line.id);
661 require(!selected.isEmpty(), "no match");
662 boolean substituted = false;
663 for (long id : selected) {
664 int position = locate(id);
665 if (position != 0) {
666 current = position;
667 Range subrange = checkedRange(c.sub, 0, new Range(0, 0));
668 boolean changed = execute(c.sub, subrange, text, output, true);
669 substituted |= changed;
670 }
671 }
672 require(c.sub.kind != Kind.SUBSTITUTE || substituted, "no match");
673 }
674 }
675 if (c.suffix != '\0' && current >= 1) printLine(current, c.suffix, output);
676 return false;
677 }
678 }

680 static long leadingInteger(String text) {
681 boolean negative = text.charAt(0) == '-';
682 int position = negative ? 1 : 0;
683 long value = 0;
684 while (position < text.length() && digit(text.charAt(position))) value = value * 10 + text.charAt(position++) - '0';
685 return negative ? -value : value;
686 }

694 static final class Editor {
695 State state = new State();
696
697 String apply(String line, List<String> text) {
698 Lexed lexed = lexAddresses(line);
699 Addresses addresses = Addresses.parse(lexed.tokens);
700 Command command = Command.parse(line.substring(lexed.end));
701 require(command.addressKind() != 0 || addresses.count == 0, "unexpected address");
702 require(command.addressKind() != 1 || addresses.count != 2, "too many addresses");
703 State trial = state.copy();
704 Range values = trial.evaluate(addresses);
705 Range range = trial.checkedRange(command, addresses.count, values);
706 StringBuilder output = new StringBuilder();
707 trial.execute(command, range, text, output, false);
708 require(trial.buffer.size() <= 200, "buffer full");
709 if (command.modifying()) {
710 trial.undo.add(new HistoryRecord(line, state.snapshot()));
711 trial.redo.clear();
712 }
713 state = trial;
714 return output.toString();
715 }
716 }

718 public static void main(String[] args) throws Exception {
719 BufferedReader input = new BufferedReader(new InputStreamReader(System.in, StandardCharsets.US_ASCII));
720 BufferedWriter output = new BufferedWriter(new OutputStreamWriter(System.out, StandardCharsets.US_ASCII));
721 Editor editor = new Editor();
722 String line;
723 while ((line = input.readLine()) != null && !line.equals("Q")) {
724 List<String> text = new ArrayList<>();
725 if (withText(line)) {
726 String entered;
727 while ((entered = input.readLine()) != null && !entered.equals(".")) text.add(entered);
728 }

730 output.write(editor.apply(line, text));
731 } catch (CommandError error) {
732 output.write("? " + error.getMessage() + "\n");
733 }
734 }
735 output.flush();
