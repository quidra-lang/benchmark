7private fun digit(c: Char) = c in '0'..'9'
8private fun lower(c: Char) = c in 'a'..'z'
9private fun upper(c: Char) = c in 'A'..'Z'
10private fun digits(s: String) = s.isNotEmpty() && s.all(::digit)

12private data class Atom(val character: Char, val matched: Boolean)
13private data class Delimited(val text: String, val atoms: List<Atom>, val end: Int)
14
15private fun delimited(s: String, start: Int, delimiter: Char): Delimited {
16 var position = start
17 val text = StringBuilder()
18 val atoms = mutableListOf<Atom>()
19 while (position < s.length) {
20 var character = s[position++]
21 if (character == delimiter) return Delimited(text.toString(), atoms.toList(), position)
22 val escaped = character == '\\'
23 if (escaped) {
24 ensure(position < s.length, "missing delimiter")
25 character = s[position++]
26 }
27 text.append(character)
28 atoms.add(Atom(character, character == '&' && !escaped))
29 }
30 throw CommandError("missing delimiter")
31}

33private data class Token(val kind: Char, val number: Int = 0, val name: Char? = null, val search: String = "")
34private data class Lexed(val tokens: List<Token>, val end: Int)
35
36private fun lexAddresses(s: String): Lexed {
37 val tokens = mutableListOf<Token>()
38 var position = 0
39 while (position < s.length) {
40 val c = s[position]
41 val token = when {
42 digit(c) -> {
43 val start = position
44 while (position < s.length && digit(s[position])) position++
45 Token('N', s.substring(start, position).toInt())
46 }
47 c == '+' || c == '-' -> {
48 val start = ++position
49 while (position < s.length && digit(s[position])) position++
50 val value = if (position == start) 1 else s.substring(start, position).toInt()
51 Token('O', if (c == '-') -value else value)
52 }
53 c in ".\$,;%" -> { position++; Token(c) }
54 c == '\'' -> {
55 position++
56 val name = s.getOrNull(position)
57 if (name != null) position++
58 Token(c, name = name)
59 }
60 c == '/' || c == '?' -> {
61 val search = delimited(s, position + 1, c)
62 position = search.end
63 Token(c, search = search.text)
64 }
65 else -> break
66 }
67 tokens.add(token)
68 }
69 return Lexed(tokens, position)
70}

72private fun withText(line: String): Boolean = try {
73 val part = line.substring(lexAddresses(line).end)
74 when {
75 part == "a" || part == "i" || part == "c" -> true
76 part.length >= 2 && part[0] in "gv" && part[1] !in " \\" -> {
77 val rest = part.substring(delimited(part, 2, part[1]).end)
78 rest == "a" || rest == "i" || rest == "c"
79 }
80 else -> false
81 }
82} catch (_: CommandError) {
83 false
84}

86private fun validAddress(tokens: List<Token>) = tokens.withIndex().all { (i, token) ->
87 token.kind == 'O' || i == 0 && token.kind in "N.\$'/?"
88}

90private data class Addresses(val count: Int, val left: List<Token>, val right: List<Token>, val semicolon: Boolean) {
91 companion object {
92 fun parse(tokens: List<Token>): Addresses {
93 if (tokens.isEmpty()) return Addresses(0, emptyList(), emptyList(), false)
94 if (tokens.size == 1 && tokens[0].kind == '%')
95 return Addresses(2, listOf(Token('N', 1)), listOf(Token('$')), false)
96 val separators = tokens.indices.filter { tokens[it].kind in ",;" }
97 ensure(separators.size <= 1, "bad address")
98 val separator = separators.firstOrNull()
99 val left = if (separator == null) tokens else tokens.subList(0, separator)
100 val right = if (separator == null) emptyList() else tokens.subList(separator + 1, tokens.size)
101 ensure(validAddress(left) && validAddress(right), "bad address")
102 return Addresses(if (separator == null) 1 else 2, left, right, separator != null && tokens[separator].kind == ';')
103 }
104 }
105}

135private class Command(var kind: Kind = Kind.EMPTY) {
136 var number = 0
137 var suffix: Char? = null
138 var option: Char? = null
139 var name = ""
140 var search = ""
141 var destination: List<Token> = emptyList()
142 var replacement: List<Atom>? = null
143 var all = false
144 var reverse = false
145 var unique = false
146 var insensitive = false
147 var numeric = false
148 var sub: Command? = null

150 companion object {
151 fun parse(part: String): Command {
152 val c = Command()
153 if (sortForm(part)) {
154 c.kind = Kind.SORT
155 ensure(part != "sort ", "invalid suffix")
156 val options = if (part.length > 4) part.substring(5) else ""
157 for (option in options) {
158 when (option) {
159 'r' -> { ensure(!c.reverse, "invalid suffix"); c.reverse = true }
160 'u' -> { ensure(!c.unique, "invalid suffix"); c.unique = true }
161 'i' -> { ensure(!c.insensitive, "invalid suffix"); c.insensitive = true }
162 'n' -> { ensure(!c.numeric, "invalid suffix"); c.numeric = true }
163 else -> throw CommandError("invalid suffix")
164 }
165 }
166 ensure(!c.insensitive || !c.numeric, "invalid suffix")
167 return c
168 }
169 if (part.isEmpty()) return c
170 val first = part[0]
171 val arg = part.substring(1)
172 val kinds = listOf(Kind.APPEND, Kind.INSERT, Kind.CHANGE, Kind.PRINT, Kind.NUMBER, Kind.LIST,
173 Kind.EQUALS, Kind.COUNT, Kind.YANK, Kind.MARKS, Kind.HISTORY, Kind.FILES)
174 val index = "aicpnl=#yKhf".indexOf(first)
175 if (index >= 0) {
176 c.kind = kinds[index]
177 ensure(arg.isEmpty(), "invalid suffix")
178 return c
179 }
180 when (first) {
181 'd', 'x', 'j', 'J' -> {
182 c.kind = when (first) { 'd' -> Kind.DELETE; 'x' -> Kind.PUT; 'j' -> Kind.JOIN; else -> Kind.JOIN_SPACE }
183 c.suffix = printSuffix(arg)
184 }
185 'z', '>', '<', 'u' -> {
186 c.kind = when (first) { 'z' -> Kind.PAGE; '>' -> Kind.INDENT; '<' -> Kind.UNINDENT; else -> Kind.UNDO }
187 c.number = positive(arg)
188 }
189 '~' -> {
190 c.kind = Kind.CASE
191 ensure(arg.isEmpty() || arg == "u" || arg == "l", "invalid suffix")
192 c.option = arg.firstOrNull()
193 }
194 'r' -> {
195 if (arg.isEmpty() || digits(arg)) { c.kind = Kind.REDO; c.number = positive(arg) }
196 else { c.kind = Kind.READ; c.name = filename(arg) }
197 }
198 'w', 'W', 'e' -> {
199 if (first == 'w' && arg.isEmpty()) c.kind = Kind.WRITE_PRINT
200 else {
201 c.kind = when (first) { 'w' -> Kind.WRITE; 'W' -> Kind.WRITE_APPEND; else -> Kind.EDIT }
202 c.name = filename(arg)
203 }
204 }
205 'k' -> {
206 c.kind = Kind.MARK
207 ensure(arg.length == 1 && lower(arg[0]), "invalid mark")
208 c.option = arg[0]
209 }
210 'm', 't' -> {
211 c.kind = if (first == 'm') Kind.MOVE else Kind.COPY
212 val destination = lexAddresses(arg)
213 ensure(destination.tokens.isNotEmpty() && validAddress(destination.tokens), "invalid destination")
214 c.destination = destination.tokens
215 c.suffix = printSuffix(arg.substring(destination.end))
216 }
217 's' -> {
218 c.kind = Kind.SUBSTITUTE
219 c.number = 1
220 val repeated = arg.all { it in "gpnl" || digit(it) }
221 var position = 0
222 if (!repeated) {
223 val delimiter = arg[0]
224 ensure(delimiter !in " \\", "missing delimiter")
225 val search = delimited(arg, 1, delimiter)
226 val replacement = delimited(arg, search.end, delimiter)
227 c.search = search.text
228 if (arg.substring(search.end, replacement.end - 1) != "%") c.replacement = replacement.atoms
229 position = replacement.end
230 }
231 var counted = false
232 while (position < arg.length) {
233 val flag = arg[position++]
234 when {
235 flag == 'g' -> { ensure(!c.all, "invalid suffix"); c.all = true }
236 flag in "pnl" -> { ensure(c.suffix == null, "invalid suffix"); c.suffix = flag }
237 digit(flag) -> {
238 ensure(!counted, "invalid suffix")
239 counted = true
240 val start = position - 1
241 while (position < arg.length && digit(arg[position])) position++
242 c.number = arg.substring(start, position).toInt()
243 ensure(c.number > 0, "invalid suffix")
244 }
245 else -> throw CommandError("invalid suffix")
246 }
247 }
248 }
249 'g', 'v' -> {
250 c.kind = if (first == 'g') Kind.GLOBAL else Kind.INVERSE
251 ensure(arg.isNotEmpty() && arg[0] !in " \\", "missing delimiter")
252 val search = delimited(arg, 1, arg[0])
253 c.search = search.text
254 val body = arg.substring(search.end).ifEmpty { "p" }
255 ensure(body[0] in "pnldmtjJs><~aicx" && !sortForm(body), "invalid global command")
256 c.sub = parse(body)
257 }
258 else -> throw CommandError("unknown command")
259 }
260 return c
261 }
262 }

266 Kind.APPEND, Kind.INSERT, Kind.PUT, Kind.EQUALS, Kind.READ, Kind.MARK, Kind.PAGE -> 1

302 fun resolvePattern(search: String): String {
303 if (search.isNotEmpty()) pattern = search

307 fun address(tokens: List<Token>): Int {
308 var value = current
309 for (token in tokens) {
310 when (token.kind) {
311 'O' -> value += token.number
312 'N' -> value = token.number
313 '.' -> value = current
314 '$' -> value = buffer.size
315 '\'' -> {
316 val name = token.name?.takeIf(::lower) ?: throw CommandError("invalid mark")
317 value = locate(marks[name - 'a'])
318 ensure(value != 0, "mark not set")
319 }
320 '/', '?' -> {
321 val pattern = resolvePattern(token.search)
322 val size = buffer.size
323 val start = if (token.kind == '?' && current == 0) size + 1 else current
324 value = 0
325 for (step in 1..size) {
326 val index = if (token.kind == '/') (start + step - 1) % size else (start - step - 1).mod(size)
327 if (pattern in buffer[index].text) { value = index + 1; break }
328 }
329 ensure(value != 0, "no match")
330 }
331 else -> error("Invalid address token")
332 }
333 }
334 return value
335 }

337 fun evaluate(addresses: Addresses): Range {
338 if (addresses.count == 0) return Range(0, 0)
339 val a = if (addresses.left.isEmpty()) { if (addresses.semicolon) current else 1 } else address(addresses.left)
340 var b = 0
341 if (addresses.count == 2) {
342 if (addresses.semicolon) {
343 ensure(a in 0..buffer.size, "address out of range")
344 current = a
345 }
346 b = if (addresses.right.isEmpty()) buffer.size else address(addresses.right)
347 }
348 return Range(a, b)
349 }

351 fun checkedRange(c: Command, count: Int, values: Range): Range {
352 val size = buffer.size
353 var a = values.first
354 var b = values.last
355 if (c.addressKind() == 0) return Range(0, 0)
356 if (c.addressKind() == 1) {
357 if (count == 0) a = when (c.kind) {
358 Kind.EQUALS, Kind.READ -> size
359 Kind.PAGE -> current + 1
360 else -> current
361 }
362 val minimum = if (c.kind == Kind.MARK || c.kind == Kind.PAGE) 1 else 0
363 ensure(a in minimum..size, "address out of range")
364 return Range(a, a)
365 }
366 if (count == 0) {
367 when (c.kind) {
368 Kind.WRITE, Kind.WRITE_APPEND, Kind.WRITE_PRINT, Kind.COUNT -> return Range(1, size)
369 Kind.GLOBAL, Kind.INVERSE, Kind.SORT -> { a = 1; b = size }
370 Kind.JOIN, Kind.JOIN_SPACE -> { a = current; b = current + 1 }
371 else -> { a = current; b = current }
372 }
373 } else if (count == 1) b = a
374 ensure(a in 1..size && b in 1..size, "address out of range")
375 ensure(a <= b && (a != b || c.kind != Kind.JOIN && c.kind != Kind.JOIN_SPACE), "invalid range")
376 return Range(a, b)
377 }

379 fun printLine(number: Int, form: Char, output: StringBuilder) {
380 val text = buffer[number - 1].text
381 when (form) {
382 'n' -> output.appendLine("${number.toString().padStart(3)} $text")
383 'l' -> {
384 val escaped = text.replace("\\", "\\\\") + '$'
385 var position = 0
386 while (escaped.length - position > 40) {
387 output.append(escaped, position, position + 39).appendLine('\\')
388 position += 39
389 }
390 output.appendLine(escaped.substring(position))
391 }
392 else -> output.appendLine(text)
393 }
394 }

405 fun substitute(c: Command, range: Range, output: StringBuilder, inGlobal: Boolean): Boolean {
406 val pattern = resolvePattern(c.search)
407 val used = c.replacement ?: replacement ?: throw CommandError("no previous substitution")
408 replacement = used
409 val expanded = buildString {
410 for (atom in used) {
411 if (atom.matched) append(pattern) else append(atom.character)
412 }
413 }
414 var last = 0
415 for (i in range.first..range.last) {
416 val old = buffer[i - 1].text
417 val result = StringBuilder()
418 var position = 0
419 var occurrence = 0
420 var changed = false
421 while (true) {
422 val start = old.indexOf(pattern, position)
423 if (start < 0) break
424 result.append(old, position, start)
425 occurrence++
426 if (occurrence == c.number || c.all && occurrence > c.number) {
427 result.append(expanded)
428 changed = true
429 } else result.append(pattern)
430 position = start + pattern.length
431 }
432 result.append(old, position, old.length)
433 if (changed) {
434 ensure(result.length <= 80, "line too long")
435 ensure(result.isNotEmpty() && result.last() != ' ', "bad line")
436 setText(i - 1, result.toString())
437 last = i
438 }
439 }
440 ensure(last != 0 || inGlobal, "no match")
441 if (last != 0) {
442 current = last
443 c.suffix?.let { printLine(last, it, output) }
444 }
445 return last != 0
446 }

448 fun sortLines(c: Command, range: Range) {
449 val key: Comparator<Line> = if (c.numeric) compareBy { leadingInteger(it.text) }
450 else compareBy { if (c.insensitive) it.text.lowercase() else it.text }
451 val sorted = buffer.subList(range.first - 1, range.last).sortedWith(if (c.reverse) key.reversed() else key)
452 val kept = if (c.unique) {
453 val unique = mutableListOf<Line>()
454 for (line in sorted) if (unique.isEmpty() || key.compare(unique.last(), line) != 0) unique.add(line)
455 unique
456 } else sorted
457 buffer.subList(range.first - 1, range.last).clear()
458 buffer.addAll(range.first - 1, kept)
459 current = range.first + kept.size - 1
460 }

462 fun fileCommand(c: Command, range: Range, output: StringBuilder) {
463 val file = if (c.kind == Kind.WRITE || c.kind == Kind.WRITE_APPEND) {
464 val saved = if (c.kind == Kind.WRITE_APPEND) files[c.name].orEmpty() else emptyList()

470 current = after + saved.size

472 }

476 fun transferHistory(isRedo: Boolean, count: Int) {
477 val from = if (isRedo) redo else undo
478 val to = if (isRedo) undo else redo
479 ensure(from.size >= count, if (isRedo) "nothing to redo" else "nothing to undo")
480 repeat(count) {
481 val record = from.removeAt(from.lastIndex)
482 to.add(HistoryRecord(record.command, snapshot()))
483 buffer = record.state.buffer.toMutableList()
484 current = record.state.current
485 }

488 fun execute(c: Command, range: Range, text: List<String>, output: StringBuilder, inGlobal: Boolean): Boolean {
489 val (a, b) = range
490 when (c.kind) {
491 Kind.EMPTY -> { printLine(b, 'p', output); current = b }
492 Kind.PRINT, Kind.NUMBER, Kind.LIST -> {
493 val form = when (c.kind) { Kind.PRINT -> 'p'; Kind.NUMBER -> 'n'; else -> 'l' }
494 for (i in a..b) printLine(i, form, output)
495 current = b
496 }
497 Kind.PAGE -> {
498 if (c.number != 0) display = c.number
499 val last = minOf(a + display - 1, buffer.size)
500 for (i in a..last) printLine(i, 'p', output)
501 current = last
502 }
503 Kind.EQUALS -> output.appendLine(a)
504 Kind.COUNT, Kind.WRITE_PRINT -> {
505 var words = 0
506 var chars = 0
507 for (i in a..b) {
508 val value = buffer[i - 1].text
509 chars += value.length
510 for (j in value.indices) if (value[j] != ' ' && (j == 0 || value[j - 1] == ' ')) words++
511 if (c.kind == Kind.WRITE_PRINT) printLine(i, 'p', output)
512 }
513 val count = b - a + 1
514 if (c.kind == Kind.COUNT) output.appendLine("count lines=$count words=$words chars=$chars")
515 else output.appendLine("lines=$count bytes=${chars + count}")
516 }
517 Kind.APPEND, Kind.INSERT -> {
518 val after = if (c.kind == Kind.APPEND) a else maxOf(a, 1) - 1
519 insert(after, text)
520 current = after + text.size
521 }
522 Kind.CHANGE, Kind.DELETE -> {
523 cut = contents(range)
524 buffer.subList(a - 1, b).clear()
525 if (c.kind == Kind.CHANGE) insert(a - 1, text)
526 current = if (c.kind == Kind.CHANGE && text.isNotEmpty()) a + text.size - 1 else minOf(a, buffer.size)
527 }
528 Kind.YANK -> cut = contents(range)
529 Kind.PUT -> {
530 ensure(cut.isNotEmpty(), "nothing to put")
531 insert(a, cut)
532 current = a + cut.size
533 }
534 Kind.MOVE, Kind.COPY -> {
535 var destination = address(c.destination)
536 ensure(destination in 0..buffer.size, "address out of range")
537 ensure(c.kind != Kind.MOVE || destination < a || destination >= b, "invalid destination")
538 val moved = buffer.subList(a - 1, b).toMutableList()
539 if (c.kind == Kind.MOVE) {
540 buffer.subList(a - 1, b).clear()
541 if (destination >= b) destination -= moved.size
542 } else {
543 for (i in moved.indices) moved[i] = moved[i].copy(id = nextId++)
544 }
545 buffer.addAll(destination, moved)
546 current = destination + moved.size
547 }
548 Kind.JOIN, Kind.JOIN_SPACE -> {
549 val joined = buildString {
550 append(buffer[a - 1].text)
551 for (i in a + 1..b) {
552 val value = buffer[i - 1].text
553 if (c.kind == Kind.JOIN_SPACE) { append(' '); append(value.trimStart(' ')) }
554 else append(value)
555 }
556 }
557 ensure(joined.length <= 80, "line too long")
558 setText(a - 1, joined)
559 buffer.subList(a, b).clear()
560 current = a
561 }
562 Kind.INDENT, Kind.UNINDENT, Kind.CASE -> {
563 val width = if (c.number == 0) 4 else c.number
564 for (i in a..b) {
565 val old = buffer[i - 1].text
566 val value = when (c.kind) {
567 Kind.INDENT -> {
568 ensure(old.length + width <= 80, "line too long")
569 " ".repeat(width) + old
570 }
571 Kind.UNINDENT -> old.drop(minOf(width, old.takeWhile { it == ' ' }.length))
572 else -> buildString {
573 for (character in old) append(when {
574 lower(character) && c.option != 'l' -> character.uppercaseChar()
575 upper(character) && c.option != 'u' -> character.lowercaseChar()
576 else -> character
577 })
578 }
579 }
580 setText(i - 1, value)
581 }
582 current = b
583 }
584 Kind.SORT -> sortLines(c, range)
585 Kind.SUBSTITUTE -> return substitute(c, range, output, inGlobal)
586 Kind.MARK -> marks[c.option!! - 'a'] = buffer[a - 1].id
587 Kind.MARKS -> {
588 output.append("marks")
589 var any = false
590 for (name in marks.indices) {
591 val position = locate(marks[name])
592 if (position != 0) { output.append(" ${'a' + name}=$position"); any = true }
593 }
594 output.appendLine(if (any) "" else " none")
595 }
596 Kind.UNDO, Kind.REDO -> transferHistory(c.kind == Kind.REDO, if (c.number == 0) 1 else c.number)
597 Kind.HISTORY -> {
598 output.appendLine("history undo=${undo.size} redo=${redo.size}")
599 for ((name, records) in listOf("undo" to undo, "redo" to redo)) {
600 for ((i, record) in records.asReversed().withIndex()) output.appendLine("$name ${i + 1}: ${record.command}")
601 }
602 }
603 Kind.READ, Kind.WRITE, Kind.WRITE_APPEND, Kind.EDIT -> fileCommand(c, range, output)
604 Kind.FILES -> {
605 output.append("files")
606 for (name in files.keys.sorted()) output.append(" $name=${files.getValue(name).size}")
607 output.appendLine(if (files.isEmpty()) " none" else "")
608 }
609 Kind.GLOBAL, Kind.INVERSE -> {
610 val pattern = resolvePattern(c.search)
611 val selected = buffer.subList(a - 1, b).filter { (pattern in it.text) == (c.kind == Kind.GLOBAL) }.map { it.id }
612 ensure(selected.isNotEmpty(), "no match")
613 val sub = c.sub!!
614 var substituted = false
615 for (id in selected) {
616 val position = locate(id)
617 if (position != 0) {
618 current = position
619 val subrange = checkedRange(sub, 0, Range(0, 0))
620 val changed = execute(sub, subrange, text, output, true)
621 substituted = substituted || changed
622 }
623 }
624 ensure(sub.kind != Kind.SUBSTITUTE || substituted, "no match")
625 }
626 }
627 c.suffix?.let { if (current >= 1) printLine(current, it, output) }
628 return false
629 }
630}

632private fun leadingInteger(text: String): Long {
633 val negative = text.startsWith('-')
634 var position = if (negative) 1 else 0
635 var value = 0L
636 while (position < text.length && digit(text[position])) value = value * 10 + (text[position++] - '0')
637 return if (negative) -value else value
638}

642private class Editor {
643 private var state = State()
644
645 fun apply(line: String, text: List<String>): String {
646 val lexed = lexAddresses(line)
647 val addresses = Addresses.parse(lexed.tokens)
648 val command = Command.parse(line.substring(lexed.end))
649 ensure(command.addressKind() != 0 || addresses.count == 0, "unexpected address")
650 ensure(command.addressKind() != 1 || addresses.count != 2, "too many addresses")
651 val trial = state.fork()
652 val values = trial.evaluate(addresses)
653 val range = trial.checkedRange(command, addresses.count, values)
654 val output = StringBuilder()
655 trial.execute(command, range, text, output, false)
656 ensure(trial.buffer.size <= 200, "buffer full")
657 if (command.modifying()) {
658 trial.undo.add(HistoryRecord(line, state.snapshot()))
659 trial.redo.clear()
660 }
661 state = trial
662 return output.toString()
663 }
664}
