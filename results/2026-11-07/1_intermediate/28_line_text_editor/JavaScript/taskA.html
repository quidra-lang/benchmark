5class EditorError extends Error {}
6function fail(reason) { throw new EditorError(reason); }
7const isDigit = ch => ch !== undefined && ch >= '0' && ch <= '9';

9function delimited(source, start, delimiter) {
10 let text = '';
11 const replacement = [];
12 for (let i = start; i < source.length; i++) {
13 let ch = source[i];
14 if (ch === '\\') {
15 if (++i === source.length) fail('missing delimiter');
16 ch = source[i];
17 text += ch;
18 replacement.push(ch);
19 } else if (ch === delimiter) {
20 return { text, replacement, raw: source.slice(start, i), next: i + 1 };
21 } else {
22 text += ch;
23 replacement.push(ch === '&' ? null : ch);
24 }
25 }
26 fail('missing delimiter');
27}

29function scanAddress(source) {
30 const elements = [];
31 let i = 0;
32 while (i < source.length) {
33 const ch = source[i];
34 if (isDigit(ch) || ch === '+' || ch === '-') {
35 const start = i++;
36 while (isDigit(source[i])) i++;
37 const word = source.slice(start, i);
38 elements.push({ kind: isDigit(ch) ? 'number' : 'offset', word });
39 } else if ('.$,;%'.includes(ch)) {
40 elements.push({ kind: ch });
41 i++;
42 } else if (ch === "'") {
43 elements.push({ kind: 'mark', word: source[i + 1] ?? '' });
44 i += Math.min(2, source.length - i);
45 } else if (ch === '/' || ch === '?') {
46 const found = delimited(source, i + 1, ch);
47 elements.push({ kind: ch, word: found.text });
48 i = found.next;
49 } else break;
50 }
51 return { elements, command: source.slice(i) };
52}

54function singleAddress(elements) {
55 const bases = new Set(['number', '.', '$', 'mark', '/', '?']);
56 return elements.length > 0 && elements.every((item, i) =>
57 item.kind === 'offset' || i === 0 && bases.has(item.kind));
58}

60function addressForm(elements) {
61 if (elements.length === 0) return { count: 0 };
62 if (elements.length === 1 && elements[0].kind === '%') {
63 return { count: 2, left: [{ kind: 'number', word: '1' }], separator: ',', right: [{ kind: '$' }] };
64 }
65 const separators = elements.flatMap((item, i) => ',;'.includes(item.kind) ? [i] : []);
66 if (separators.length > 1) fail('bad address');
67 if (separators.length === 0) {
68 if (!singleAddress(elements)) fail('bad address');
69 return { count: 1, left: elements };
70 }
71 const index = separators[0];
72 const left = elements.slice(0, index), right = elements.slice(index + 1);
73 if (left.length && !singleAddress(left) || right.length && !singleAddress(right)) fail('bad address');
74 return { count: 2, left, right, separator: elements[index].kind };
75}

77function positiveArgument(argument) {
78 if (argument === '') return null;
79 if (!/^\d+$/.test(argument) || Number(argument) === 0) fail('invalid suffix');
80 return Number(argument);

83function printSuffix(argument) {
84 if (!/^[pnl]?$/.test(argument)) fail('invalid suffix');
85 return argument;

88function fileName(argument) {
89 if (!/^ [a-z0-9]{1,8}$/.test(argument)) fail('invalid file name');
90 return argument.slice(1);

93function parseSubstitution(argument) {
94 let pattern = '', replacement = null, flags = argument;
95 if (!/^[gpnl0-9]*$/.test(argument)) {
96 const delimiter = argument[0];
97 if (delimiter === ' ' || delimiter === '\\') fail('missing delimiter');
98 const search = delimited(argument, 1, delimiter);
99 const replace = delimited(argument, search.next, delimiter);
100 pattern = search.text;
101 replacement = replace.raw === '%' ? null : replace.replacement;
102 flags = argument.slice(replace.next);
103 }
104 let global = false, suffix = '', count = 1, counted = false;
105 for (let i = 0; i < flags.length;) {
106 const ch = flags[i++];
107 if (ch === 'g') {
108 if (global) fail('invalid suffix');
109 global = true;
110 } else if ('pnl'.includes(ch)) {
111 if (suffix !== '') fail('invalid suffix');
112 suffix = ch;
113 } else if (isDigit(ch)) {
114 if (counted) fail('invalid suffix');
115 const start = i - 1;
116 while (isDigit(flags[i])) i++;
117 count = Number(flags.slice(start, i));
118 if (count === 0) fail('invalid suffix');
119 counted = true;
120 } else fail('invalid suffix');
121 }
122 return { kind: 's', pattern, replacement, global, suffix, count };
123}

125function parseCommand(source) {
126 if (source === 'sort' || source.startsWith('sort ')) {
127 const options = source === 'sort' ? '' : source.slice(5);
128 if (source !== 'sort' && (!/^[ruin]+$/.test(options) || new Set(options).size !== options.length
129 || options.includes('i') && options.includes('n'))) fail('invalid suffix');
130 return { kind: 'sort', options };
131 }
132 const kind = source[0] ?? '', argument = source.slice(1);
133 if (kind === '' || 'aicpnl=#yKhf'.includes(kind)) {
134 if (argument !== '') fail('invalid suffix');
135 return { kind };
136 }
137 if ('dxjJ'.includes(kind)) return { kind, suffix: printSuffix(argument) };
138 if ('z><u'.includes(kind)) return { kind, amount: positiveArgument(argument) };
139 if (kind === '~') {
140 if (!/^[ul]?$/.test(argument)) fail('invalid suffix');
141 return { kind, letterCase: argument };
142 }
143 if (kind === 'r') {
144 return /^\d*$/.test(argument)
145 ? { kind: 'redo', amount: positiveArgument(argument) }
146 : { kind: 'read', name: fileName(argument) };
147 }
148 if (kind === 'w') return argument === '' ? { kind } : { kind: 'write', name: fileName(argument) };
149 if (kind === 'W' || kind === 'e') return { kind, name: fileName(argument) };
150 if (kind === 'k') {
151 if (!/^[a-z]$/.test(argument)) fail('invalid mark');
152 return { kind, mark: argument };
153 }
154 if (kind === 'm' || kind === 't') {
155 const destination = scanAddress(argument);
156 if (!singleAddress(destination.elements)) fail('invalid destination');
157 return { kind, destination: destination.elements, suffix: printSuffix(destination.command) };
158 }
159 if (kind === 's') return parseSubstitution(argument);
160 if (kind === 'g' || kind === 'v') {
161 if (argument === '' || argument[0] === ' ' || argument[0] === '\\') fail('missing delimiter');
162 const search = delimited(argument, 1, argument[0]);
163 const subcommand = argument.slice(search.next) || 'p';
164 if (!'pnldmtjJs><~aicx'.includes(subcommand[0]) || subcommand === 'sort' || subcommand.startsWith('sort ')) {
165 fail('invalid global command');
166 }
167 return { kind, pattern: search.text, subcommand: parseCommand(subcommand) };
168 }
169 fail('unknown command');
170}

172function hasText(source) {
173 try {
174 const command = scanAddress(source).command;
175 if (/^[aic]$/.test(command)) return true;
176 if (!/^[gv]/.test(command) || command.length < 2 || command[1] === ' ' || command[1] === '\\') return false;
177 const search = delimited(command, 2, command[1]);
178 return /^[aic]$/.test(command.slice(search.next));
179 } catch (error) {
180 if (!(error instanceof EditorError)) throw error;
181 return false;
182 }
183}

185function addressKind(command) {
186 if (['u', 'redo', 'h', 'K', 'f', 'e'].includes(command.kind)) return 0;
187 if (['a', 'i', 'x', '=', 'read', 'k', 'z'].includes(command.kind)) return 1;
188 return 2;
189}

191function modifying(command) {
192 if (command.kind === 'g' || command.kind === 'v') return !['p', 'n', 'l'].includes(command.subcommand.kind);
193 return ['a', 'i', 'c', 'd', 'x', 'j', 'J', 'm', 't', 's', '>', '<', '~', 'sort', 'read', 'e'].includes(command.kind);
194}

196function copyState(state) {
197 return { ...state, lines: state.lines.slice(), marks: new Map(state.marks), cut: state.cut.slice(),
198 undo: state.undo.slice(), redo: state.redo.slice(), files: new Map(state.files) };

201class Editor {
202 constructor() {
203 this.state = { lines: [], current: 0, marks: new Map(), cut: [], pattern: null, replacement: null,
204 displayCount: 5, undo: [], redo: [], files: new Map(), nextId: 1 };
205 this.output = [];
206 }
207
208 newLines(text) { return text.map(value => ({ id: this.state.nextId++, text: value })); }
209 replace(position, count, inserted) {
210 const lines = this.state.lines;
211 this.state.lines = lines.slice(0, position).concat(inserted, lines.slice(position + count));
212 }
213 snapshot(command) { return { command, lines: this.state.lines.slice(), current: this.state.current }; }
214 contents(a, b) { return this.state.lines.slice(a - 1, b).map(line => line.text); }
215 emit(text) { this.output.push(text); }
216
217 pattern(text) {
218 if (text !== '') this.state.pattern = text;
219 if (this.state.pattern === null) fail('no previous pattern');
220 return this.state.pattern;
221 }
222
223 address(elements) {
224 const state = this.state, last = state.lines.length;
225 let value = state.current, firstOffset = 0;
226 const base = elements[0];
227 if (base.kind !== 'offset') {
228 firstOffset = 1;
229 if (base.kind === 'number') value = Number(base.word);
230 else if (base.kind === '.') value = state.current;
231 else if (base.kind === '$') value = last;
232 else if (base.kind === 'mark') {
233 if (!/^[a-z]$/.test(base.word)) fail('invalid mark');
234 value = state.lines.findIndex(line => line.id === state.marks.get(base.word)) + 1;
235 if (value === 0) fail('mark not set');
236 } else {
237 const pattern = this.pattern(base.word);
238 value = 0;
239 for (let step = 1; step <= last; step++) {
240 const pivot = state.current === 0 && base.kind === '?' ? last + 1 : state.current;
241 const index = base.kind === '/' ? (pivot + step - 1) % last : (pivot - step - 1 + last) % last;
242 if (state.lines[index].text.includes(pattern)) { value = index + 1; break; }
243 }
244 if (value === 0) fail('no match');
245 }
246 }
247 for (const offset of elements.slice(firstOffset)) {
248 const magnitude = offset.word.length === 1 ? 1 : Number(offset.word.slice(1));
249 value += offset.word[0] === '+' ? magnitude : -magnitude;
250 }
251 return value;
252 }
253
254 addresses(form) {
255 if (form.count === 0) return [];
256 if (form.count === 1) return [this.address(form.left)];
257 const left = form.left.length ? this.address(form.left) : form.separator === ',' ? 1 : this.state.current;
258 if (form.separator === ';') {
259 if (left < 0 || left > this.state.lines.length) fail('address out of range');
260 this.state.current = left;
261 }
262 return [left, form.right.length ? this.address(form.right) : this.state.lines.length];
263 }
264
265 range(command, values) {
266 const state = this.state, last = state.lines.length, kind = command.kind;
267 const category = addressKind(command);
268 if (category === 0) return [0, 0];
269 if (category === 1) {
270 const fallback = ['=', 'read'].includes(kind) ? last : state.current + (kind === 'z' ? 1 : 0);
271 const a = values.length ? values[0] : fallback;
272 const minimum = kind === 'k' || kind === 'z' ? 1 : 0;
273 if (a < minimum || a > last) fail('address out of range');
274 return [a, a];
275 }
276 let a, b;
277 if (values.length) [a, b] = [values[0], values.at(-1)];
278 else if (['w', 'write', 'W', '#'].includes(kind)) return [1, last];
279 else if (['g', 'v', 'sort'].includes(kind)) [a, b] = [1, last];
280 else [a, b] = [state.current, state.current + (kind === 'j' || kind === 'J' ? 1 : 0)];
281 if (a < 1 || b < 1 || a > last || b > last) fail('address out of range');
282 if (a > b || a === b && (kind === 'j' || kind === 'J')) fail('invalid range');
283 return [a, b];
284 }

286 print(number, form) {
287 const text = this.state.lines[number - 1].text;
288 if (form === 'p') this.emit(text);
289 else if (form === 'n') this.emit(`${String(number).padStart(3, ' ')} ${text}`);
290 else {
291 let escaped = text.replaceAll('\\', '\\\\') + '$';
292 while (escaped.length > 40) {
293 this.emit(escaped.slice(0, 39) + '\\');
294 escaped = escaped.slice(39);
295 }
296 this.emit(escaped);
297 }
298 }
299
300 suffix(command) {
301 if (command.suffix && this.state.current > 0) this.print(this.state.current, command.suffix);
302 }
303
304 substitute(command, a, b, insideGlobal) {
305 const state = this.state, pattern = this.pattern(command.pattern);
306 const replacement = command.replacement ?? state.replacement;
307 if (replacement === null) fail('no previous substitution');
308 state.replacement = replacement;
309 const expanded = replacement.map(part => part === null ? pattern : part).join('');
310 let changed = 0;
311 for (let i = a - 1; i < b; i++) {
312 const line = state.lines[i];
313 let position = 0, occurrence = 0, result = '', substituted = false;
314 for (;;) {
315 const found = line.text.indexOf(pattern, position);
316 if (found < 0) break;
317 occurrence++;
318 const replace = occurrence === command.count || command.global && occurrence > command.count;
319 result += line.text.slice(position, found) + (replace ? expanded : pattern);
320 substituted ||= replace;
321 position = found + pattern.length;
322 }
323 if (!substituted) continue;
324 result += line.text.slice(position);
325 if (result.length > 80) fail('line too long');
326 if (result === '' || result.endsWith(' ')) fail('bad line');
327 state.lines[i] = { id: line.id, text: result };
328 state.current = i + 1;
329 changed++;
330 }
331 if (changed === 0 && !insideGlobal) fail('no match');
332 if (changed > 0) this.suffix(command);
333 return changed;
334 }
335
336 global(command, a, b, text) {
337 const state = this.state, pattern = this.pattern(command.pattern);
338 const selected = state.lines.slice(a - 1, b)
339 .filter(line => line.text.includes(pattern) === (command.kind === 'g')).map(line => line.id);
340 if (selected.length === 0) fail('no match');
341 let changed = 0;
342 for (const id of selected) {
343 const index = state.lines.findIndex(line => line.id === id);
344 if (index < 0) continue;
345 state.current = index + 1;
346 const [start, end] = this.range(command.subcommand, []);
347 changed += this.execute(command.subcommand, start, end, text, true);
348 }
349 if (command.subcommand.kind === 's' && changed === 0) fail('no match');
350 }
351
352 execute(command, a, b, text, insideGlobal = false) {
353 const state = this.state, kind = command.kind;
354 if (kind === 's') return this.substitute(command, a, b, insideGlobal);
355 if (kind === 'g' || kind === 'v') this.global(command, a, b, text);
356 else if (kind === '' || ['p', 'n', 'l'].includes(kind)) {
357 for (let i = kind === '' ? b : a; i <= b; i++) this.print(i, kind || 'p');
358 state.current = b;
359 } else if (kind === 'z') {
360 if (command.amount !== null) state.displayCount = command.amount;
361 const end = Math.min(a + state.displayCount - 1, state.lines.length);
362 for (let i = a; i <= end; i++) this.print(i, 'p');
363 state.current = end;
364 } else if (kind === '=') this.emit(String(a));
365 else if (kind === '#') {
366 const contents = this.contents(a, b);
367 const words = contents.reduce((n, line) => n + line.split(' ').filter(Boolean).length, 0);
368 const chars = contents.reduce((n, line) => n + line.length, 0);
369 this.emit(`count lines=${contents.length} words=${words} chars=${chars}`);
370 } else if (kind === 'w') {
371 const contents = this.contents(a, b);
372 for (const line of contents) this.emit(line);
373 this.emit(`lines=${contents.length} bytes=${contents.reduce((n, line) => n + line.length + 1, 0)}`);
374 } else if (kind === 'a' || kind === 'i') {
375 const position = kind === 'a' ? a : Math.max(a, 1) - 1;
376 this.replace(position, 0, this.newLines(text));
377 state.current = position + text.length;
378 } else if (kind === 'c' || kind === 'd') {
379 state.cut = this.contents(a, b);
380 const inserted = kind === 'c' ? this.newLines(text) : [];
381 this.replace(a - 1, b - a + 1, inserted);
382 state.current = inserted.length ? a + inserted.length - 1 : Math.min(a, state.lines.length);
383 this.suffix(command);
384 } else if (kind === 'y') state.cut = this.contents(a, b);
385 else if (kind === 'x') {
386 if (state.cut.length === 0) fail('nothing to put');
387 this.replace(a, 0, this.newLines(state.cut));
388 state.current = a + state.cut.length;
389 this.suffix(command);
390 } else if (kind === 'm' || kind === 't') {
391 const destination = this.address(command.destination), count = b - a + 1;
392 if (destination < 0 || destination > state.lines.length) fail('address out of range');
393 if (kind === 'm') {
394 if (a <= destination && destination < b) fail('invalid destination');
395 const moved = state.lines.splice(a - 1, count);
396 const position = destination < a ? destination : destination - count;
397 this.replace(position, 0, moved);
398 state.current = position + count;
399 } else {
400 const copied = this.newLines(this.contents(a, b));
401 this.replace(destination, 0, copied);
402 state.current = destination + count;
403 }
404 this.suffix(command);
405 } else if (kind === 'j' || kind === 'J') {
406 const contents = this.contents(a, b);
407 const joined = kind === 'j' ? contents.join('') : contents.map((line, i) => i ? line.replace(/^ +/, '') : line).join(' ');
408 if (joined.length > 80) fail('line too long');
409 state.lines.splice(a - 1, b - a + 1, { id: state.lines[a - 1].id, text: joined });
410 state.current = a;
411 this.suffix(command);
412 } else if (kind === '>' || kind === '<' || kind === '~') {
413 const width = command.amount ?? 4;
414 for (let i = a - 1; i < b; i++) {
415 const line = state.lines[i];
416 let value = line.text;
417 if (kind === '>') {
418 if (value.length + width > 80) fail('line too long');
419 value = ' '.repeat(width) + value;
420 } else if (kind === '<') value = value.slice(Math.min(width, /^ */.exec(value)[0].length));
421 else value = value.replace(/[A-Za-z]/g, ch => command.letterCase === 'u' ? ch.toUpperCase()
422 : command.letterCase === 'l' ? ch.toLowerCase() : ch === ch.toUpperCase() ? ch.toLowerCase() : ch.toUpperCase());
423 state.lines[i] = { id: line.id, text: value };
424 }
425 state.current = b;
426 } else if (kind === 'sort') {
427 const options = command.options;
428 const key = line => options.includes('n') ? Number(/^-?\d+/.exec(line.text)?.[0] ?? 0)
429 : options.includes('i') ? line.text.toLowerCase() : line.text;
430 const direction = options.includes('r') ? -1 : 1;
431 let sorted = state.lines.slice(a - 1, b).map(line => ({ line, key: key(line) }));
432 sorted.sort((x, y) => direction * (x.key < y.key ? -1 : x.key > y.key ? 1 : 0));
433 if (options.includes('u')) sorted = sorted.filter((item, i, all) => i === 0 || item.key !== all[i - 1].key);
434 this.replace(a - 1, b - a + 1, sorted.map(item => item.line));
435 state.current = a + sorted.length - 1;
436 } else if (kind === 'k') state.marks.set(command.mark, state.lines[a - 1].id);
437 else if (kind === 'K') {
438 const marks = [...state.marks].sort(([a], [b]) => a < b ? -1 : a > b ? 1 : 0).flatMap(([name, id]) => {
439 const index = state.lines.findIndex(line => line.id === id);
440 return index < 0 ? [] : [`${name}=${index + 1}`];
441 });
442 this.emit('marks ' + (marks.join(' ') || 'none'));
443 } else if (kind === 'u' || kind === 'redo') {
444 const source = kind === 'u' ? state.undo : state.redo;
445 const target = kind === 'u' ? state.redo : state.undo;
446 const count = command.amount ?? 1;
447 if (source.length < count) fail(kind === 'u' ? 'nothing to undo' : 'nothing to redo');
448 for (let i = 0; i < count; i++) {
449 const record = source.pop();
450 target.push(this.snapshot(record.command));
451 state.lines = record.lines.slice();
452 state.current = record.current;
453 }
454 } else if (kind === 'h') {
455 this.emit(`history undo=${state.undo.length} redo=${state.redo.length}`);
456 for (const [name, history] of [['undo', state.undo], ['redo', state.redo]]) {
457 for (let i = history.length - 1; i >= 0; i--) this.emit(`${name} ${history.length - i}: ${history[i].command}`);
458 }
459 } else if (['write', 'W', 'read', 'e'].includes(kind)) {
460 let file = state.files.get(command.name);
461 if (kind === 'write' || kind === 'W') {
462 file = (kind === 'W' ? file ?? [] : []).concat(this.contents(a, b));
463 state.files.set(command.name, file);
464 } else {
465 if (file === undefined) fail('no such file');
466 const inserted = this.newLines(file);
467 if (kind === 'e') { state.lines = inserted; state.current = inserted.length; }
468 else { this.replace(a, 0, inserted); state.current = a + inserted.length; }
469 }
470 this.emit(`file ${command.name} lines=${file.length} bytes=${file.reduce((n, line) => n + line.length + 1, 0)}`);
471 } else if (kind === 'f') {
472 const files = [...state.files.keys()].sort().map(name => `${name}=${state.files.get(name).length}`);
473 this.emit('files ' + (files.join(' ') || 'none'));
474 }
475 return 0;
476 }

478 process(source, text) {
479 const previous = this.state;
480 const before = this.snapshot(source);
481 this.state = copyState(previous);
482 this.output = [];
483 try {
484 const split = scanAddress(source);
485 const form = addressForm(split.elements);
486 const command = parseCommand(split.command);
487 const category = addressKind(command);
488 if (category === 0 && form.count > 0) fail('unexpected address');
489 if (category === 1 && form.count > 1) fail('too many addresses');
490 const [a, b] = this.range(command, this.addresses(form));
491 this.execute(command, a, b, text);
492 if (this.state.lines.length > 200) fail('buffer full');
493 if (modifying(command)) {
494 this.state.undo.push(before);
495 this.state.redo = [];
496 }
497 return this.output;
498 } catch (error) {
499 this.state = previous;
500 if (!(error instanceof EditorError)) throw error;
501 return ['? ' + error.message];
502 }
503 }
504}

506const input = fs.readFileSync(0, 'utf8').split('\n');
507const editor = new Editor(), output = [];
508for (let i = 0; input[i] !== 'Q'; i++) {
509 const command = input[i], text = [];
510 if (hasText(command)) {
511 while (input[++i] !== '.') text.push(input[i]);
512 }
513 output.push(...editor.process(command, text));
514}
515if (output.length) process.stdout.write(output.join('\n') + '\n');
