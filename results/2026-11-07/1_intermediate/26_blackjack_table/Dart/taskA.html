10int integer(String word, int minimum, int maximum) {
11 if (word.isEmpty || word.length > 10 || word.length > 1 && word[0] == '0') {
12 fail('bad value $word');
13 }
14 var value = 0;
15 for (final digit in word.codeUnits) {
16 if (digit < 48 || digit > 57) fail('bad value $word');
17 value = value * 10 + digit - 48;
18 }
19 if (value < minimum || value > maximum) fail('bad value $word');
20 return value;
21}

23String choice(String word, List<String> values) {
24 if (!values.contains(word)) fail('bad value $word');
25 return word;
26}

42class Card {
43 static const ranks = 'A23456789TJQK';
44 static const suits = 'CDHS';
45 final int code;
46 const Card(this.code);
47 int get rank => code % 13;
48 int get suit => code ~/ 13;
49 int get points => rank >= 9 ? 10 : rank + 1;
50 bool get black => suit == 0 || suit == 3;
51 @override
52 String toString() => '${ranks[rank]}${suits[suit]}';
53}
54
55({int value, bool soft}) total(List<Card> cards) {
56 var sum = 0;
57 var ace = false;
58 for (final card in cards) {
59 sum += card.points;
60 ace = ace || card.rank == 0;
61 }
62 final soft = ace && sum + 10 <= 21;
63 return (value: sum + (soft ? 10 : 0), soft: soft);
64}
65
66bool blackjack(List<Card> cards, {bool split = false}) =>
67 !split && cards.length == 2 && total(cards).value == 21;
68
69String handText(List<Card> cards, {bool split = false}) {
70 final t = total(cards);
71 final status = blackjack(cards, split: split) ? 'blackjack'
72 : t.value > 21 ? 'bust ${t.value}' : t.soft ? 'soft ${t.value}' : '${t.value}';
73 return '${cards.join(' ')} ($status)';
74}

76class Rules {
77 bool hitSoft17 = false;
78 int penetration = 75, minimum = 10, maximum = 500;
79 bool sixToFive = false, doubleAfterSplit = true;
80 DoubleRule doubling = DoubleRule.any;
81 int hands = 4;
82 bool resplitAces = false, surrender = true;
83 int charlie = 0, sideMaximum = 25;
84
85 Rules();
86 Rules.from(Rules other) {
87 hitSoft17 = other.hitSoft17;
88 penetration = other.penetration;
89 minimum = other.minimum;
90 maximum = other.maximum;
91 sixToFive = other.sixToFive;
92 doubleAfterSplit = other.doubleAfterSplit;
93 doubling = other.doubling;
94 hands = other.hands;
95 resplitAces = other.resplitAces;
96 surrender = other.surrender;
97 charlie = other.charlie;
98 sideMaximum = other.sideMaximum;
99 }

101 Rules changed(String name, String word) {
102 final next = Rules.from(this);
103 switch (name) {
104 case 'DEALER': next.hitSoft17 = choice(word, ['S17', 'H17']) == 'H17';
105 case 'PEN': next.penetration = integer(word, 50, 90);
106 case 'MIN': next.minimum = integer(word, 1, 10000);
107 case 'MAX': next.maximum = integer(word, 1, 10000);
108 case 'BJ': next.sixToFive = choice(word, ['3:2', '6:5']) == '6:5';
109 case 'DAS': next.doubleAfterSplit = choice(word, ['ON', 'OFF']) == 'ON';
110 case 'DOUBLE':
111 next.doubling = switch (choice(word, ['ANY', '9-11', '10-11'])) {
112 'ANY' => DoubleRule.any, '9-11' => DoubleRule.hard9, _ => DoubleRule.hard10,
113 };
114 case 'HANDS': next.hands = integer(word, 1, 4);
115 case 'RSA': next.resplitAces = choice(word, ['ON', 'OFF']) == 'ON';
116 case 'SURRENDER': next.surrender = choice(word, ['ON', 'OFF']) == 'ON';
117 case 'CHARLIE': next.charlie = int.parse(choice(word, ['0', '5', '6', '7']));
118 case 'SIDEMAX': next.sideMaximum = integer(word, 0, 1000);
119 default: fail('unknown rule $name');
120 }
121 return next;
122 }
123
124 @override
125 String toString() {
126 final doubleWord = switch (doubling) {
127 DoubleRule.any => 'ANY', DoubleRule.hard9 => '9-11', DoubleRule.hard10 => '10-11',
128 };
129 return 'rules DEALER ${hitSoft17 ? 'H17' : 'S17'} PEN $penetration MIN $minimum MAX $maximum'
130 ' BJ ${sixToFive ? '6:5' : '3:2'} DAS ${onOff(doubleAfterSplit)} DOUBLE $doubleWord'
131 ' HANDS $hands RSA ${onOff(resplitAces)} SURRENDER ${onOff(surrender)}'
132 ' CHARLIE $charlie SIDEMAX $sideMaximum';
133 }
134}

170 static const commands = {
171 'SHOE', 'RULE', 'RULES', 'PLAYER', 'LEAVE', 'REBUY', 'BET', 'SIDE', 'DEAL', 'INSURANCE',
172 'HIT', 'STAND', 'DOUBLE', 'SPLIT', 'SURRENDER', 'HINT', 'AUTO', 'STATUS', 'COUNT', 'STATS',
173 };
174 static const actions = {
175 'HIT': PlayerAction.hit, 'STAND': PlayerAction.stand, 'DOUBLE': PlayerAction.doubleDown,
176 'SPLIT': PlayerAction.split, 'SURRENDER': PlayerAction.surrender,

178 Rules rules = Rules();
179 final List<Player?> seats = List.filled(7, null);
180 List<Player> participants = [];
181 Phase phase = Phase.idle;
182 int round = 0, dealerBlackjacks = 0, dealerBusts = 0;
183 int decks = 0, drawn = 0, running = 0, insuranceTurn = 0, random = 1;
184 List<Card> shoe = [];
185 final List<Card> dealer = [];
186 bool holeRevealed = false;
187 final List<String> output = [];

189 Iterable<Player> get players => seats.whereType<Player>();

193 for (final player in players) {

199 void shuffle() {
200 shoe = List.generate(52 * decks, (i) => Card(i % 52));
201 for (var i = shoe.length - 1; i >= 1; i--) {
202 random = random * 48271 % 2147483647;
203 final j = random % (i + 1);
204 final saved = shoe[i];
205 shoe[i] = shoe[j];
206 shoe[j] = saved;
207 }
208 drawn = running = 0;
209 }

211 void countCard(Card card) {
212 if (card.points >= 2 && card.points <= 6) {
213 running++;
214 } else if (card.points == 1 || card.points == 10) {
215 running--;
216 }
217 }

219 Card draw({bool faceUp = true}) {
220 final card = shoe[drawn++];
221 if (faceUp) countCard(card);
222 return card;
223 }

227 countCard(dealer[1]);

233 Turn? current() {
234 for (final player in participants) {
235 for (var i = 0; i < player.hands.length; i++) {
236 if (!player.hands[i].finished) return (player: player, index: i, hand: player.hands[i]);
237 }

239 return null;

242 void finishCheck(Player player, Hand hand) {
243 final value = total(hand.cards).value;
244 if (value > 21) {
245 hand.reason = Finish.bust;
246 hand.finished = true;
247 } else if (rules.charlie != 0 && hand.cards.length == rules.charlie) {
248 hand.reason = Finish.charlie;
249 hand.finished = true;
250 } else if (value == 21) {
251 hand.finished = true;
252 } else if (hand.splitAces && hand.cards.length == 2) {
253 hand.finished = !(hand.cards.every((c) => c.rank == 0) && rules.resplitAces && player.hands.length < rules.hands);
254 }
255 }

257 void settle({bool dealerNatural = false}) {
258 final dealerTotal = total(dealer).value;
259 for (final player in participants) {
260 for (var i = 0; i < player.hands.length; i++) {
261 final hand = player.hands[i];
262 final value = total(hand.cards).value;
263 final result = dealerNatural ? (hand.natural ? Outcome.push : Outcome.lose)
264 : hand.natural ? Outcome.blackjack : switch (hand.reason) {
265 Finish.bust => Outcome.bust, Finish.surrender => Outcome.surrender, Finish.charlie => Outcome.charlie,
266 null => dealerTotal > 21 || value > dealerTotal ? Outcome.win : value == dealerTotal ? Outcome.push : Outcome.lose,
267 };
268 final profit = switch (result) {
269 Outcome.blackjack => rules.sixToFive ? hand.bet * 6 ~/ 5 : hand.bet * 3 ~/ 2,
270 Outcome.win || Outcome.charlie => hand.bet,
271 Outcome.push => 0,
272 Outcome.surrender => hand.bet ~/ 2 - hand.bet,
273 Outcome.lose || Outcome.bust => -hand.bet,
274 };
275 player.bankroll += hand.bet + profit;
276 player.net += profit;
277 player.stats.hands++;
278 if (result == Outcome.blackjack || result == Outcome.win || result == Outcome.charlie) {
279 player.stats.wins++;
280 } else if (result == Outcome.push) {
281 player.stats.pushes++;
282 } else {
283 player.stats.losses++;
284 }
285 if (result == Outcome.blackjack) player.stats.blackjacks++;
286 if (result == Outcome.bust) player.stats.busts++;
287 if (result == Outcome.surrender) player.stats.surrenders++;
288 emit('${player.handName(i)} ${result.name} ${signed(profit)}');
289 }
290 emit('${player.name} bankroll ${player.bankroll}');
291 }
292 for (final player in players) {
293 player.bet = player.side = 0;
294 }
295 phase = Phase.idle;
296 }

298 void dealerTurn() {
299 revealDealer();
300 final mustPlay = participants.any((p) => p.hands.any((h) => h.reason == null && !h.natural));
301 var drew = false;
302 if (mustPlay) {
303 while (true) {
304 final t = total(dealer);
305 if (!(t.value <= 16 || t.value == 17 && t.soft && rules.hitSoft17)) break;
306 dealer.add(draw());
307 drew = true;
308 emit('dealer hits: ${handText(dealer)}');
309 }
310 }
311 if (drew && total(dealer).value > 21) dealerBusts++;
312 settle();
313 }
314
315 void advance() {
316 while (true) {
317 final turn = current();
318 if (turn == null) {
319 dealerTurn();
320 return;
321 }
322 if (turn.hand.cards.length != 1) return;
323 turn.hand.cards.add(draw());
324 emit('${turn.player.handName(turn.index)}: ${turn.hand}');
325 finishCheck(turn.player, turn.hand);
326 if (!turn.hand.finished) return;
327 }
328 }

330 void peek() {
331 final natural = blackjack(dealer);
332 emit('dealer peeks: ${natural ? 'blackjack' : 'no blackjack'}');
333 for (final player in participants.where((p) => p.insurance != 0)) {
334 final profit = player.insurance * (natural ? 2 : -1);
335 player.bankroll += player.insurance + profit;
336 player.net += profit;
337 emit('${player.name} insurance ${signed(profit)}');
338 }
339 if (natural) {
340 dealerBlackjacks++;
341 revealDealer();
342 settle(dealerNatural: true);
343 } else {
344 phase = Phase.play;
345 advance();
346 }
347 }

349 void deal() {
350 if (drawn >= cut) {
351 shuffle();
352 emit('shuffle');
353 }
354 emit('round ${++round}');
355 participants = players.where((p) => p.bet != 0).toList();
356 for (final player in participants) {
357 player.bankroll -= player.bet + player.side;
358 player.insurance = 0;
359 player.stats.rounds++;
360 player.hands.clear();
361 player.hands.add(Hand([draw()], player.bet));
362 }
363 dealer.clear();
364 dealer.add(draw());
365 for (final player in participants) {
366 final hand = player.hands[0];
367 hand.cards.add(draw());
368 hand.finished = hand.natural;
369 }
370 dealer.add(draw(faceUp: false));
371 holeRevealed = false;
372 for (final player in participants) {
373 emit('${player.handName(0)}: ${player.hands[0]}');
374 }
375 emit('dealer: ${dealer[0]} ??');
376 for (final player in participants.where((p) => p.side != 0)) {
377 final first = player.hands[0].cards[0], second = player.hands[0].cards[1];
378 final (result, multiplier) = first.rank != second.rank ? ('lose', -1)
379 : first.suit == second.suit ? ('perfect', 25) : first.black == second.black ? ('colored', 12) : ('mixed', 6);
380 final profit = player.side * multiplier;
381 player.bankroll += player.side + profit;
382 player.net += profit;
383 emit('${player.name} side $result ${signed(profit)}');
384 }
385 if (dealer[0].points == 1) {
386 phase = Phase.insurance;
387 insuranceTurn = 0;
388 emit('insurance offered');
389 } else if (dealer[0].points == 10) {
390 peek();
391 } else {
392 phase = Phase.play;
393 advance();
394 }
395 }

397 String? actionError(PlayerAction action, Player player, Hand hand) {
398 switch (action) {
399 case PlayerAction.hit:
400 if (hand.splitAces) return 'split aces';
401 case PlayerAction.doubleDown:
402 if (hand.splitAces) return 'split aces';
403 if (hand.cards.length != 2) return 'two cards only';
404 final t = total(hand.cards);
405 if (rules.doubling != DoubleRule.any && (t.soft || t.value < (rules.doubling == DoubleRule.hard9 ? 9 : 10) || t.value > 11)
406 || !rules.doubleAfterSplit && hand.split) return 'double not allowed';
407 if (player.bankroll < hand.bet) return 'insufficient funds';
408 case PlayerAction.split:
409 if (hand.cards.length != 2) return 'two cards only';
410 if (hand.cards[0].points != hand.cards[1].points) return 'not a pair';
411 if (player.hands.length >= rules.hands) return 'hand limit';
412 if (player.bankroll < hand.bet) return 'insufficient funds';
413 case PlayerAction.surrender:
414 if (!rules.surrender || hand.split) return 'surrender not allowed';
415 if (hand.cards.length != 2) return 'two cards only';
416 case PlayerAction.stand: break;
417 }
418 return null;
419 }

421 void perform(PlayerAction action) {
422 final turn = current()!;
423 final player = turn.player;
424 final hand = turn.hand;
425 final name = player.handName(turn.index);
426 if (action == PlayerAction.split) {
427 final first = hand.cards[0], second = hand.cards[1];
428 final aces = hand.splitAces || first.points == 1;
429 player.bankroll -= hand.bet;
430 player.stats.splits++;
431 emit('$name splits $first $second');
432 final replacement = Hand([first], hand.bet, split: true, splitAces: aces);
433 player.hands[turn.index] = replacement;
434 player.hands.insert(turn.index + 1, Hand([second], hand.bet, split: true, splitAces: aces));
435 replacement.cards.add(draw());
436 emit('$name: $replacement');
437 finishCheck(player, replacement);
438 } else if (action == PlayerAction.stand || action == PlayerAction.surrender) {
439 hand.finished = true;
440 if (action == PlayerAction.surrender) hand.reason = Finish.surrender;
441 emit('$name ${action == PlayerAction.stand ? 'stands' : 'surrenders'}: $hand');
442 } else {
443 if (action == PlayerAction.doubleDown) {
444 player.bankroll -= hand.bet;
445 hand.bet *= 2;
446 player.stats.doubles++;
447 }
448 hand.cards.add(draw());
449 emit('$name ${action == PlayerAction.doubleDown ? 'doubles' : 'hits'}: $hand');
450 finishCheck(player, hand);
451 if (action == PlayerAction.doubleDown) hand.finished = true;
452 }
453 advance();
454 }

456 PlayerAction recommend() {
457 final turn = current()!;
458 final player = turn.player;
459 final hand = turn.hand;
460 bool possible(PlayerAction action) => actionError(action, player, hand) == null;
461 final up = dealer[0].points == 1 ? 11 : dealer[0].points;
462 if (hand.splitAces) return possible(PlayerAction.split) ? PlayerAction.split : PlayerAction.stand;
463 if (hand.cards.length == 2 && hand.cards[0].points == hand.cards[1].points && possible(PlayerAction.split)) {
464 final pair = hand.cards[0].points;
465 if (pair == 1 || pair == 8 || pair == 9 && ![7, 10, 11].contains(up)
466 || [2, 3, 7].contains(pair) && up <= 7 || pair == 6 && up <= 6 || pair == 4 && (up == 5 || up == 6)) {
467 return PlayerAction.split;
468 }
469 }
470 final t = total(hand.cards);
471 var symbol = 'H';
472 if (t.soft) {
473 if (t.value >= 19) {
474 symbol = 'S';
475 } else if (t.value == 18) {
476 symbol = up >= 3 && up <= 6 ? 'Ds' : [2, 7, 8].contains(up) ? 'S' : 'H';
477 } else if (t.value == 17 && up >= 3 && up <= 6 || [15, 16].contains(t.value) && up >= 4 && up <= 6
478 || [13, 14].contains(t.value) && (up == 5 || up == 6)) {
479 symbol = 'D';
480 }
481 } else {
482 if (t.value >= 17) {
483 symbol = 'S';
484 } else if (t.value >= 13) {
485 symbol = up <= 6 ? 'S' : t.value == 16 && up >= 9 || t.value == 15 && up == 10 ? 'R' : 'H';
486 } else if (t.value == 12) {
487 symbol = up >= 4 && up <= 6 ? 'S' : 'H';
488 } else if (t.value == 11 && up <= 10 || t.value == 10 && up <= 9 || t.value == 9 && up >= 3 && up <= 6) {
489 symbol = 'D';
490 }
491 }
492 return switch (symbol) {
493 'S' => PlayerAction.stand,
494 'D' => possible(PlayerAction.doubleDown) ? PlayerAction.doubleDown : PlayerAction.hit,
495 'Ds' => possible(PlayerAction.doubleDown) ? PlayerAction.doubleDown : PlayerAction.stand,
496 'R' => possible(PlayerAction.surrender) ? PlayerAction.surrender : PlayerAction.hit,
497 _ => PlayerAction.hit,
498 };
499 }

521 List<String> execute(List<String> words) {
522 output.clear();
523 final command = words[0];
524 if (!commands.contains(command)) fail('unknown command $command');
525 final first = words.length > 1 ? words[1] : '', second = words.length > 2 ? words[2] : '';
526 final action = actions[command];
527 Player? target;
528 var amount = 0, seed = 0;
529 Rules? proposed;
530
531 if (command == 'SHOE') {
532 amount = integer(first, 1, 8);
533 seed = integer(second, 1, 2147483646);
534 } else if (command == 'RULE') {
535 proposed = rules.changed(first, second);
536 } else if (command == 'PLAYER') {
537 if (first.isEmpty || first.length > 8 || first.codeUnits.any((c) => c < 65 || c > 90)) fail('bad name $first');
538 amount = integer(second, 1, 1000000);
539 } else if (action != null || ['LEAVE', 'REBUY', 'BET', 'SIDE', 'INSURANCE', 'AUTO'].contains(command)) {
540 target = find(first);
541 if (['REBUY', 'BET', 'SIDE', 'INSURANCE'].contains(command)) {
542 amount = integer(second, command == 'REBUY' ? 1 : 0, 1000000);
543 }
544 }
545 if (['SHOE', 'RULE', 'PLAYER', 'LEAVE', 'REBUY', 'BET', 'SIDE', 'DEAL'].contains(command)) {
546 if (phase != Phase.idle) fail('round in progress');
547 } else if (command == 'INSURANCE') {
548 if (phase != Phase.insurance) fail('no insurance');
549 } else if (action != null || command == 'AUTO' || command == 'HINT') {
550 if (phase == Phase.idle) fail('no round');
551 if (phase == Phase.insurance) fail('insurance pending');
552 }
553 if (action != null || command == 'AUTO') {
554 final turn = current()!;
555 if (turn.player != target) fail('waiting for ${turn.player.handName(turn.index)}');
556 if (action != null) {
557 final error = actionError(action, turn.player, turn.hand);
558 if (error != null) fail(error);
559 perform(action);
560 } else {
561 while (phase == Phase.play && current()!.player == target) {
562 perform(recommend());
563 }
564 }
565 return output;
566 }
567 switch (command) {
568 case 'SHOE':
569 decks = amount;
570 random = seed;
571 shuffle();
572 emit('shoe $decks decks ${shoe.length} cards cut $cut');
573 case 'RULE':
574 final next = proposed!;
575 if (next.minimum > next.maximum) fail('min above max');
576 rules = next;
577 emit('rule $first $second');
578 case 'RULES': emit('$rules');
579 case 'PLAYER':
580 if (players.any((p) => p.name == first)) fail('player exists');
581 final seat = seats.indexWhere((p) => p == null);
582 if (seat < 0) fail('table full');
583 seats[seat] = Player(first, seat + 1, amount);
584 emit('$first sits at seat ${seat + 1} with $amount');
585 case 'LEAVE':
586 final player = target!;
587 emit('${player.name} leaves with ${player.bankroll} net ${signed(player.net)}');
588 seats[player.seat - 1] = null;
589 case 'REBUY':
590 final player = target!;
591 player.bankroll += amount;
592 emit('${player.name} rebuys $amount bankroll ${player.bankroll}');
593 case 'BET':
594 final player = target!;
595 if (amount != 0) {
596 if (amount < rules.minimum || amount > rules.maximum) fail('bet out of limits');
597 if (amount + player.side > player.bankroll) fail('insufficient funds');
598 }
599 player.bet = amount;
600 if (amount == 0) player.side = 0;
601 emit(amount == 0 ? '${player.name} bet cleared' : '${player.name} bets $amount');
602 case 'SIDE':
603 final player = target!;
604 if (amount != 0) {
605 if (player.bet == 0) fail('no bet');
606 if (amount > rules.sideMaximum) fail('side out of limits');
607 if (player.bet + amount > player.bankroll) fail('insufficient funds');
608 }
609 player.side = amount;
610 emit(amount == 0 ? '${player.name} side cleared' : '${player.name} side $amount');
611 case 'DEAL':
612 if (decks == 0) fail('no shoe');
613 if (!players.any((p) => p.bet != 0)) fail('no bets');
614 deal();
615 case 'INSURANCE':
616 final player = target!;
617 if (participants[insuranceTurn] != player) fail('waiting for ${participants[insuranceTurn].name}');
618 if (amount > player.bet ~/ 2) fail('insurance out of limits');
619 if (amount > player.bankroll) fail('insufficient funds');
620 player.bankroll -= amount;
621 player.insurance = amount;
622 emit(amount == 0 ? '${player.name} declines insurance' : '${player.name} insures $amount');
623 if (++insuranceTurn == participants.length) peek();
624 case 'HINT':
625 final turn = current()!;
626 emit('hint ${turn.player.handName(turn.index)}: ${recommend().text}');
627 case 'STATUS': status();
628 case 'COUNT':
629 if (decks == 0) fail('no shoe');
630 final remaining = shoe.length - drawn;
631 final hundredths = running.abs() * 5200 ~/ remaining;
632 final count = '${running < 0 ? '-' : ''}${hundredths ~/ 100}.${(hundredths % 100).toString().padLeft(2, '0')}';
633 emit('count running $running true $count remaining $remaining');
634 case 'STATS':
635 if (players.isEmpty) fail('no players');
636 for (final player in players) {
637 emit('stats ${player.name} ${player.stats} net ${signed(player.net)}');
638 }
639 emit('stats dealer rounds $round blackjacks $dealerBlackjacks busts $dealerBusts');
640 }
641 return output;
642 }
