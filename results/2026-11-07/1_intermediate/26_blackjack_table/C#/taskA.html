14readonly record struct Card(int Code)
15{
16 public int Rank => Code % 13;
17 public int Suit => Code / 13;
18 public int Points => Math.Min(Rank + 1, 10);
19 public bool Black => Suit is 0 or 3;
20 public override string ToString() => $"{"A23456789TJQK"[Rank]}{"CDHS"[Suit]}";
21}

23static class Format
24{
25 public static long Integer(string word, long minimum, long maximum)
26 {
27 if (word.Length is < 1 or > 10 || word.Length > 1 && word[0] == '0')
28 throw new CommandError($"bad value {word}");
29 long value = 0;
30 foreach (char ch in word)
31 {
32 if (ch is < '0' or > '9') throw new CommandError($"bad value {word}");
33 value = value * 10 + ch - '0';
34 }
35 if (value < minimum || value > maximum) throw new CommandError($"bad value {word}");
36 return value;
37 }
38
39 public static string Choice(string word, params string[] choices) => choices.Contains(word)
40 ? word : throw new CommandError($"bad value {word}");
41 public static string Signed(long value) => (value >= 0 ? "+" : "") + value;
42 public static string Switch(bool enabled) => enabled ? "ON" : "OFF";
43 public static string Action(PlayerAction action) => action.ToString().ToLowerInvariant();
44
45 public static (int Value, bool Soft) Total(IReadOnlyList<Card> cards)
46 {
47 int sum = 0;
48 bool ace = false;
49 foreach (Card card in cards) { sum += card.Points; ace |= card.Rank == 0; }
50 bool soft = ace && sum + 10 <= 21;
51 return (sum + (soft ? 10 : 0), soft);
52 }
53
54 public static bool Blackjack(IReadOnlyList<Card> cards, bool split = false) =>
55 !split && cards.Count == 2 && Total(cards).Value == 21;
56
57 public static string Hand(IReadOnlyList<Card> cards, bool split = false)
58 {
59 var (value, soft) = Total(cards);
60 string status = Blackjack(cards, split) ? "blackjack"
61 : value > 21 ? $"bust {value}" : soft ? $"soft {value}" : value.ToString();
62 return $"{string.Join(" ", cards)} ({status})";
63 }
64}

66sealed record Rules
67{
68 public bool HitSoft17 { get; init; }
69 public int Penetration { get; init; } = 75;
70 public int Minimum { get; init; } = 10;
71 public int Maximum { get; init; } = 500;
72 public bool SixToFive { get; init; }
73 public bool DoubleAfterSplit { get; init; } = true;
74 public DoubleRule Doubling { get; init; } = DoubleRule.Any;
75 public int Hands { get; init; } = 4;
76 public bool ResplitAces { get; init; }
77 public bool Surrender { get; init; } = true;
78 public int Charlie { get; init; }
79 public int SideMaximum { get; init; } = 25;
80
81 public Rules Changed(string name, string word) => name switch
82 {
83 "DEALER" => this with { HitSoft17 = Format.Choice(word, "S17", "H17") == "H17" },
84 "PEN" => this with { Penetration = (int)Format.Integer(word, 50, 90) },
85 "MIN" => this with { Minimum = (int)Format.Integer(word, 1, 10000) },
86 "MAX" => this with { Maximum = (int)Format.Integer(word, 1, 10000) },
87 "BJ" => this with { SixToFive = Format.Choice(word, "3:2", "6:5") == "6:5" },
88 "DAS" => this with { DoubleAfterSplit = Format.Choice(word, "ON", "OFF") == "ON" },
89 "DOUBLE" => this with
90 {
91 Doubling = Format.Choice(word, "ANY", "9-11", "10-11") switch
92 {
93 "ANY" => DoubleRule.Any, "9-11" => DoubleRule.Hard9, _ => DoubleRule.Hard10
94 }
95 },
96 "HANDS" => this with { Hands = (int)Format.Integer(word, 1, 4) },
97 "RSA" => this with { ResplitAces = Format.Choice(word, "ON", "OFF") == "ON" },
98 "SURRENDER" => this with { Surrender = Format.Choice(word, "ON", "OFF") == "ON" },
99 "CHARLIE" => this with { Charlie = Format.Choice(word, "0", "5", "6", "7")[0] - '0' },
100 "SIDEMAX" => this with { SideMaximum = (int)Format.Integer(word, 0, 1000) },
101 _ => throw new CommandError($"unknown rule {name}")
102 };
103
104 public override string ToString()
105 {
106 string doubling = Doubling switch { DoubleRule.Any => "ANY", DoubleRule.Hard9 => "9-11", _ => "10-11" };
107 return $"rules DEALER {(HitSoft17 ? "H17" : "S17")} PEN {Penetration} MIN {Minimum} MAX {Maximum}"
108 + $" BJ {(SixToFive ? "6:5" : "3:2")} DAS {Format.Switch(DoubleAfterSplit)} DOUBLE {doubling}"
109 + $" HANDS {Hands} RSA {Format.Switch(ResplitAces)} SURRENDER {Format.Switch(Surrender)}"
110 + $" CHARLIE {Charlie} SIDEMAX {SideMaximum}";
111 }
112}

144sealed class Table
145{
146 static readonly HashSet<string> Commands = new(StringComparer.Ordinal)
147 {
148 "SHOE", "RULE", "RULES", "PLAYER", "LEAVE", "REBUY", "BET", "SIDE", "DEAL", "INSURANCE",
149 "HIT", "STAND", "DOUBLE", "SPLIT", "SURRENDER", "HINT", "AUTO", "STATUS", "COUNT", "STATS"
150 };
151 Rules rules = new();
152 readonly Player?[] seats = new Player?[7];
153 List<Player> participants = [];
154 Phase phase = Phase.Idle;
155 int round, dealerBlackjacks, dealerBusts;
156 int decks, drawn, running, insuranceTurn;
157 long random = 1;
158 Card[] shoe = [];
159 readonly List<Card> dealer = [];
160 bool holeRevealed;
161 readonly List<string> output = [];
162
163 IEnumerable<Player> Players => seats.OfType<Player>();
164 int Cut => shoe.Length * rules.Penetration / 100;
165 Player Find(string name) => Players.FirstOrDefault(p => p.Name == name)
166 ?? throw new CommandError($"unknown player {name}");
167 void Emit(string line) => output.Add(line);

169 void Shuffle()
170 {
171 shoe = Enumerable.Range(0, 52 * decks).Select(i => new Card(i % 52)).ToArray();
172 for (int i = shoe.Length - 1; i >= 1; i--)
173 {
174 random = random * 48271 % 2147483647;
175 int j = (int)(random % (i + 1));
176 (shoe[i], shoe[j]) = (shoe[j], shoe[i]);
177 }
178 drawn = running = 0;
179 }
180
181 void CountCard(Card card) => running += card.Points is >= 2 and <= 6 ? 1 : card.Points is 1 or 10 ? -1 : 0;
182 Card Draw(bool faceUp = true)
183 {
184 Card card = shoe[drawn++];
185 if (faceUp) CountCard(card);
186 return card;
187 }
188
189 void RevealDealer()
190 {
191 if (!holeRevealed) { CountCard(dealer[1]); holeRevealed = true; }
192 Emit($"dealer: {Format.Hand(dealer)}");
193 }
194
195 (Player Player, int Index, Hand Hand)? Current()
196 {
197 foreach (Player player in participants)
198 for (int i = 0; i < player.Hands.Count; i++)
199 if (!player.Hands[i].Finished) return (player, i, player.Hands[i]);
200 return null;
201 }
202
203 void FinishCheck(Player player, Hand hand)
204 {
205 int value = Format.Total(hand.Cards).Value;
206 if (value > 21) { hand.Reason = Finish.Bust; hand.Finished = true; }
207 else if (rules.Charlie != 0 && hand.Cards.Count == rules.Charlie)
208 {
209 hand.Reason = Finish.Charlie;
210 hand.Finished = true;
211 }
212 else if (value == 21) hand.Finished = true;
213 else if (hand.SplitAces && hand.Cards.Count == 2)
214 hand.Finished = !(hand.Cards.All(c => c.Rank == 0) && rules.ResplitAces && player.Hands.Count < rules.Hands);
215 }
216
217 void Settle(bool dealerNatural = false)
218 {
219 int dealerTotal = Format.Total(dealer).Value;
220 foreach (Player player in participants)
221 {
222 for (int i = 0; i < player.Hands.Count; i++)
223 {
224 Hand hand = player.Hands[i];
225 int value = Format.Total(hand.Cards).Value;
226 Outcome result = dealerNatural ? (hand.Natural ? Outcome.Push : Outcome.Lose)
227 : hand.Natural ? Outcome.Blackjack : hand.Reason switch
228 {
229 Finish.Bust => Outcome.Bust, Finish.Surrender => Outcome.Surrender, Finish.Charlie => Outcome.Charlie,
230 _ => dealerTotal > 21 || value > dealerTotal ? Outcome.Win : value == dealerTotal ? Outcome.Push : Outcome.Lose
231 };
232 long profit = result switch
233 {
234 Outcome.Blackjack => rules.SixToFive ? hand.Bet * 6 / 5 : hand.Bet * 3 / 2,
235 Outcome.Win or Outcome.Charlie => hand.Bet,
236 Outcome.Push => 0,
237 Outcome.Surrender => hand.Bet / 2 - hand.Bet,
238 _ => -hand.Bet
239 };
240 player.Bankroll += hand.Bet + profit;
241 player.Net += profit;
242 player.Stats.Hands++;
243 if (result is Outcome.Blackjack or Outcome.Win or Outcome.Charlie) player.Stats.Wins++;
244 else if (result == Outcome.Push) player.Stats.Pushes++;
245 else player.Stats.Losses++;
246 if (result == Outcome.Blackjack) player.Stats.Blackjacks++;
247 if (result == Outcome.Bust) player.Stats.Busts++;
248 if (result == Outcome.Surrender) player.Stats.Surrenders++;
249 Emit($"{player.HandName(i)} {result.ToString().ToLowerInvariant()} {Format.Signed(profit)}");
250 }
251 Emit($"{player.Name} bankroll {player.Bankroll}");
252 }
253 foreach (Player player in Players) player.Bet = player.Side = 0;
254 phase = Phase.Idle;
255 }
256
257 void DealerTurn()
258 {
259 RevealDealer();
260 bool mustPlay = participants.Any(p => p.Hands.Any(h => h.Reason is null && !h.Natural));
261 bool drew = false;
262 if (mustPlay)
263 {
264 while (true)
265 {
266 var (value, soft) = Format.Total(dealer);
267 if (!(value <= 16 || value == 17 && soft && rules.HitSoft17)) break;
268 dealer.Add(Draw());
269 drew = true;
270 Emit($"dealer hits: {Format.Hand(dealer)}");
271 }
272 }
273 if (drew && Format.Total(dealer).Value > 21) dealerBusts++;
274 Settle();
275 }
276
277 void Advance()
278 {
279 while (Current() is { } turn)
280 {
281 if (turn.Hand.Cards.Count != 1) return;
282 turn.Hand.Cards.Add(Draw());
283 Emit($"{turn.Player.HandName(turn.Index)}: {turn.Hand}");
284 FinishCheck(turn.Player, turn.Hand);
285 if (!turn.Hand.Finished) return;
286 }
287 DealerTurn();
288 }
289
290 void Peek()
291 {
292 bool natural = Format.Blackjack(dealer);
293 Emit($"dealer peeks: {(natural ? "blackjack" : "no blackjack")}");
294 foreach (Player player in participants.Where(p => p.Insurance != 0))
295 {
296 long profit = player.Insurance * (natural ? 2 : -1);
297 player.Bankroll += player.Insurance + profit;
298 player.Net += profit;
299 Emit($"{player.Name} insurance {Format.Signed(profit)}");
300 }
301 if (natural)
302 {
303 dealerBlackjacks++;
304 RevealDealer();
305 Settle(true);
306 }
307 else { phase = Phase.Play; Advance(); }
308 }
309
310 void Deal()
311 {
312 if (drawn >= Cut) { Shuffle(); Emit("shuffle"); }
313 Emit($"round {++round}");
314 participants = Players.Where(p => p.Bet != 0).ToList();
315 foreach (Player player in participants)
316 {
317 player.Bankroll -= player.Bet + player.Side;
318 player.Insurance = 0;
319 player.Stats.Rounds++;
320 player.Hands.Clear();
321 player.Hands.Add(new Hand([Draw()], player.Bet));
322 }
323 dealer.Clear();
324 dealer.Add(Draw());
325 foreach (Player player in participants)
326 {
327 Hand hand = player.Hands[0];
328 hand.Cards.Add(Draw());
329 hand.Finished = hand.Natural;
330 }
331 dealer.Add(Draw(false));
332 holeRevealed = false;
333 foreach (Player player in participants) Emit($"{player.HandName(0)}: {player.Hands[0]}");
334 Emit($"dealer: {dealer[0]} ??");
335 foreach (Player player in participants.Where(p => p.Side != 0))
336 {
337 Card first = player.Hands[0].Cards[0], second = player.Hands[0].Cards[1];
338 var (result, multiplier) = first.Rank != second.Rank ? ("lose", -1)
339 : first.Suit == second.Suit ? ("perfect", 25) : first.Black == second.Black ? ("colored", 12) : ("mixed", 6);
340 long profit = player.Side * multiplier;
341 player.Bankroll += player.Side + profit;
342 player.Net += profit;
343 Emit($"{player.Name} side {result} {Format.Signed(profit)}");
344 }
345 if (dealer[0].Points == 1)
346 {
347 phase = Phase.Insurance;
348 insuranceTurn = 0;
349 Emit("insurance offered");
350 }
351 else if (dealer[0].Points == 10) Peek();
352 else { phase = Phase.Play; Advance(); }
353 }

355 string? ActionError(PlayerAction action, Player player, Hand hand)
356 {
357 switch (action)
358 {
359 case PlayerAction.Hit:
360 if (hand.SplitAces) return "split aces";
361 break;
362 case PlayerAction.Double:
363 if (hand.SplitAces) return "split aces";
364 if (hand.Cards.Count != 2) return "two cards only";
365 var (value, soft) = Format.Total(hand.Cards);
366 if (rules.Doubling != DoubleRule.Any && (soft || value < (rules.Doubling == DoubleRule.Hard9 ? 9 : 10) || value > 11)
367 || !rules.DoubleAfterSplit && hand.Split) return "double not allowed";
368 if (player.Bankroll < hand.Bet) return "insufficient funds";
369 break;
370 case PlayerAction.Split:
371 if (hand.Cards.Count != 2) return "two cards only";
372 if (hand.Cards[0].Points != hand.Cards[1].Points) return "not a pair";
373 if (player.Hands.Count >= rules.Hands) return "hand limit";
374 if (player.Bankroll < hand.Bet) return "insufficient funds";
375 break;
376 case PlayerAction.Surrender:
377 if (!rules.Surrender || hand.Split) return "surrender not allowed";
378 if (hand.Cards.Count != 2) return "two cards only";
379 break;
380 }
381 return null;
382 }
383
384 void Perform(PlayerAction action)
385 {
386 var (player, index, hand) = Current()!.Value;
387 string name = player.HandName(index);
388 if (action == PlayerAction.Split)
389 {
390 Card first = hand.Cards[0], second = hand.Cards[1];
391 bool aces = hand.SplitAces || first.Points == 1;
392 player.Bankroll -= hand.Bet;
393 player.Stats.Splits++;
394 Emit($"{name} splits {first} {second}");
395 var replacement = new Hand([first], hand.Bet, true, aces);
396 player.Hands[index] = replacement;
397 player.Hands.Insert(index + 1, new Hand([second], hand.Bet, true, aces));
398 replacement.Cards.Add(Draw());
399 Emit($"{name}: {replacement}");
400 FinishCheck(player, replacement);
401 }
402 else if (action is PlayerAction.Stand or PlayerAction.Surrender)
403 {
404 hand.Finished = true;
405 if (action == PlayerAction.Surrender) hand.Reason = Finish.Surrender;
406 Emit($"{name} {(action == PlayerAction.Stand ? "stands" : "surrenders")}: {hand}");
407 }
408 else
409 {
410 if (action == PlayerAction.Double)
411 {
412 player.Bankroll -= hand.Bet;
413 hand.Bet *= 2;
414 player.Stats.Doubles++;
415 }
416 hand.Cards.Add(Draw());
417 Emit($"{name} {(action == PlayerAction.Double ? "doubles" : "hits")}: {hand}");
418 FinishCheck(player, hand);
419 if (action == PlayerAction.Double) hand.Finished = true;
420 }
421 Advance();
422 }
423
424 PlayerAction Recommend()
425 {
426 var (player, _, hand) = Current()!.Value;
427 bool Possible(PlayerAction action) => ActionError(action, player, hand) is null;
428 int up = dealer[0].Points == 1 ? 11 : dealer[0].Points;
429 if (hand.SplitAces) return Possible(PlayerAction.Split) ? PlayerAction.Split : PlayerAction.Stand;
430 if (hand.Cards.Count == 2 && hand.Cards[0].Points == hand.Cards[1].Points && Possible(PlayerAction.Split))
431 {
432 int pair = hand.Cards[0].Points;
433 if (pair is 1 or 8 || pair == 9 && up is not (7 or 10 or 11)
434 || pair is 2 or 3 or 7 && up <= 7 || pair == 6 && up <= 6 || pair == 4 && up is 5 or 6)
435 return PlayerAction.Split;
436 }
437 var (value, soft) = Format.Total(hand.Cards);
438 string symbol = "H";
439 if (soft)
440 {
441 if (value >= 19) symbol = "S";
442 else if (value == 18) symbol = up is >= 3 and <= 6 ? "Ds" : up is 2 or 7 or 8 ? "S" : "H";
443 else if (value == 17 && up is >= 3 and <= 6 || value is 15 or 16 && up is >= 4 and <= 6
444 || value is 13 or 14 && up is 5 or 6) symbol = "D";
445 }
446 else
447 {
448 if (value >= 17) symbol = "S";
449 else if (value >= 13) symbol = up <= 6 ? "S" : value == 16 && up >= 9 || value == 15 && up == 10 ? "R" : "H";
450 else if (value == 12) symbol = up is >= 4 and <= 6 ? "S" : "H";
451 else if (value == 11 && up <= 10 || value == 10 && up <= 9 || value == 9 && up is >= 3 and <= 6) symbol = "D";
452 }
453 return symbol switch
454 {
455 "S" => PlayerAction.Stand,
456 "D" => Possible(PlayerAction.Double) ? PlayerAction.Double : PlayerAction.Hit,
457 "Ds" => Possible(PlayerAction.Double) ? PlayerAction.Double : PlayerAction.Stand,
458 "R" => Possible(PlayerAction.Surrender) ? PlayerAction.Surrender : PlayerAction.Hit,
459 _ => PlayerAction.Hit
460 };
461 }

471 }

474 foreach (Player player in Players)
475 {
476 Emit($"seat {player.Seat} {player.Name} bankroll {player.Bankroll} bet {player.Bet} side {player.Side}");
477 if (phase != Phase.Idle && participants.Contains(player))

480 }

483 public IReadOnlyList<string> Execute(string[] words)
484 {
485 output.Clear();
486 string command = words[0];
487 if (!Commands.Contains(command)) throw new CommandError($"unknown command {command}");
488 string first = words.Length > 1 ? words[1] : "", second = words.Length > 2 ? words[2] : "";
489 PlayerAction? action = command switch
490 {
491 "HIT" => PlayerAction.Hit, "STAND" => PlayerAction.Stand, "DOUBLE" => PlayerAction.Double,
492 "SPLIT" => PlayerAction.Split, "SURRENDER" => PlayerAction.Surrender, _ => null
493 };
494 Player? target = null;
495 long amount = 0, seed = 0;
496 Rules? proposed = null;
497
498 if (command == "SHOE")
499 {
500 amount = Format.Integer(first, 1, 8);
501 seed = Format.Integer(second, 1, 2147483646);
502 }
503 else if (command == "RULE") proposed = rules.Changed(first, second);
504 else if (command == "PLAYER")
505 {
506 if (first.Length is < 1 or > 8 || first.Any(c => c is < 'A' or > 'Z')) throw new CommandError($"bad name {first}");
507 amount = Format.Integer(second, 1, 1000000);
508 }
509 else if (action is not null || command is "LEAVE" or "REBUY" or "BET" or "SIDE" or "INSURANCE" or "AUTO")
510 {
511 target = Find(first);
512 if (command is "REBUY" or "BET" or "SIDE" or "INSURANCE")
513 amount = Format.Integer(second, command == "REBUY" ? 1 : 0, 1000000);
514 }
515 if (command is "SHOE" or "RULE" or "PLAYER" or "LEAVE" or "REBUY" or "BET" or "SIDE" or "DEAL")
516 {
517 if (phase != Phase.Idle) throw new CommandError("round in progress");
518 }
519 else if (command == "INSURANCE")
520 {
521 if (phase != Phase.Insurance) throw new CommandError("no insurance");
522 }
523 else if (action is not null || command is "AUTO" or "HINT")
524 {
525 if (phase == Phase.Idle) throw new CommandError("no round");
526 if (phase == Phase.Insurance) throw new CommandError("insurance pending");
527 }
528 if (action is not null || command == "AUTO")
529 {
530 var turn = Current()!.Value;
531 if (turn.Player != target) throw new CommandError($"waiting for {turn.Player.HandName(turn.Index)}");
532 if (action is { } chosen)
533 {
534 string? error = ActionError(chosen, turn.Player, turn.Hand);
535 if (error is not null) throw new CommandError(error);
536 Perform(chosen);
537 }
538 else while (phase == Phase.Play && Current()!.Value.Player == target) Perform(Recommend());
539 return output;
540 }
541 switch (command)
542 {
543 case "SHOE":
544 decks = (int)amount;
545 random = seed;
546 Shuffle();
547 Emit($"shoe {decks} decks {shoe.Length} cards cut {Cut}");
548 break;
549 case "RULE":
550 if (proposed!.Minimum > proposed.Maximum) throw new CommandError("min above max");
551 rules = proposed;
552 Emit($"rule {first} {second}");
553 break;
554 case "RULES": Emit(rules.ToString()); break;
555 case "PLAYER":
556 {
557 if (Players.Any(p => p.Name == first)) throw new CommandError("player exists");
558 int seat = Array.FindIndex(seats, p => p is null);
559 if (seat < 0) throw new CommandError("table full");
560 seats[seat] = new Player(first, seat + 1, amount);
561 Emit($"{first} sits at seat {seat + 1} with {amount}");
562 break;
563 }
564 case "LEAVE":
565 Emit($"{target!.Name} leaves with {target.Bankroll} net {Format.Signed(target.Net)}");
566 seats[target.Seat - 1] = null;
567 break;
568 case "REBUY":
569 target!.Bankroll += amount;
570 Emit($"{target.Name} rebuys {amount} bankroll {target.Bankroll}");
571 break;
572 case "BET":
573 if (amount != 0)
574 {
575 if (amount < rules.Minimum || amount > rules.Maximum) throw new CommandError("bet out of limits");
576 if (amount + target!.Side > target.Bankroll) throw new CommandError("insufficient funds");
577 }
578 target!.Bet = amount;
579 if (amount == 0) target.Side = 0;
580 Emit(amount == 0 ? $"{target.Name} bet cleared" : $"{target.Name} bets {amount}");
581 break;
582 case "SIDE":
583 if (amount != 0)
584 {
585 if (target!.Bet == 0) throw new CommandError("no bet");
586 if (amount > rules.SideMaximum) throw new CommandError("side out of limits");
587 if (target.Bet + amount > target.Bankroll) throw new CommandError("insufficient funds");
588 }
589 target!.Side = amount;
590 Emit(amount == 0 ? $"{target.Name} side cleared" : $"{target.Name} side {amount}");
591 break;
592 case "DEAL":
593 if (decks == 0) throw new CommandError("no shoe");
594 if (!Players.Any(p => p.Bet != 0)) throw new CommandError("no bets");
595 Deal();
596 break;
597 case "INSURANCE":
598 if (participants[insuranceTurn] != target) throw new CommandError($"waiting for {participants[insuranceTurn].Name}");
599 if (amount > target!.Bet / 2) throw new CommandError("insurance out of limits");
600 if (amount > target.Bankroll) throw new CommandError("insufficient funds");
601 target.Bankroll -= amount;
602 target.Insurance = amount;
603 Emit(amount == 0 ? $"{target.Name} declines insurance" : $"{target.Name} insures {amount}");
604 if (++insuranceTurn == participants.Count) Peek();
605 break;
606 case "HINT":
607 {
608 var turn = Current()!.Value;
609 Emit($"hint {turn.Player.HandName(turn.Index)}: {Format.Action(Recommend())}");
610 break;
611 }
612 case "STATUS": Status(); break;
613 case "COUNT":
614 {
615 if (decks == 0) throw new CommandError("no shoe");
616 int remaining = shoe.Length - drawn;
617 int hundredths = Math.Abs(running) * 5200 / remaining;
618 string count = $"{(running < 0 ? "-" : "")}{hundredths / 100}.{hundredths % 100:D2}";
619 Emit($"count running {running} true {count} remaining {remaining}");
620 break;
621 }
622 case "STATS":
623 if (!Players.Any()) throw new CommandError("no players");
624 foreach (Player player in Players) Emit($"stats {player.Name} {player.Stats} net {Format.Signed(player.Net)}");
625 Emit($"stats dealer rounds {round} blackjacks {dealerBlackjacks} busts {dealerBusts}");
626 break;
627 }
628 return output;
629 }
