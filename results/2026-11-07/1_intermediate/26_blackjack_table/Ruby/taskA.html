4def integer(word, minimum, maximum)
5 raise CommandError, "bad value #{word}" unless /\A(?:0|[1-9][0-9]{0,9})\z/.match?(word)
6 value = word.to_i
7 raise CommandError, "bad value #{word}" unless value.between?(minimum, maximum)
8 value
9end

12 raise CommandError, "bad value #{word}" unless choices.include?(word)
13 word
14end

19Rules = Data.define(:hit_soft17, :penetration, :minimum, :maximum, :six_to_five, :double_after_split,
20 :doubling, :hands, :resplit_aces, :surrender, :charlie, :side_maximum) do
21 def self.defaults
22 new(hit_soft17: false, penetration: 75, minimum: 10, maximum: 500, six_to_five: false,
23 double_after_split: true, doubling: 'ANY', hands: 4, resplit_aces: false,
24 surrender: true, charlie: 0, side_maximum: 25)
25 end
26
27 def changed(name, word)
28 case name
29 when 'DEALER' then with(hit_soft17: choice(word, 'S17', 'H17') == 'H17')
30 when 'PEN' then with(penetration: integer(word, 50, 90))
31 when 'MIN' then with(minimum: integer(word, 1, 10000))
32 when 'MAX' then with(maximum: integer(word, 1, 10000))
33 when 'BJ' then with(six_to_five: choice(word, '3:2', '6:5') == '6:5')
34 when 'DAS' then with(double_after_split: on(word))
35 when 'DOUBLE' then with(doubling: choice(word, 'ANY', '9-11', '10-11'))
36 when 'HANDS' then with(hands: integer(word, 1, 4))
37 when 'RSA' then with(resplit_aces: on(word))
38 when 'SURRENDER' then with(surrender: on(word))
39 when 'CHARLIE' then with(charlie: choice(word, '0', '5', '6', '7').to_i)
40 when 'SIDEMAX' then with(side_maximum: integer(word, 0, 1000))
41 else raise CommandError, "unknown rule #{name}"
42 end
43 end

45 def to_s
46 format('rules DEALER %s PEN %d MIN %d MAX %d BJ %s DAS %s DOUBLE %s HANDS %d RSA %s SURRENDER %s CHARLIE %d SIDEMAX %d',
47 hit_soft17 ? 'H17' : 'S17', penetration, minimum, maximum, six_to_five ? '6:5' : '3:2',
48 on_off(double_after_split), doubling, hands, on_off(resplit_aces), on_off(surrender), charlie, side_maximum)
49 end

52Card = Data.define(:code) do
53 def rank = code % 13
54 def suit = code / 13
55 def point = [rank + 1, 10].min
56 def black? = suit == 0 || suit == 3
57 def hilo
58 case point
59 when 2..6 then 1
60 when 1, 10 then -1
61 else 0
62 end
63 end
64 def to_s = "#{'A23456789TJQK'[rank]}#{'CDHS'[suit]}"
65end

69class Hand
70 attr_reader :cards, :split, :split_aces
71 attr_accessor :bet, :finished, :reason
72
73 def initialize(cards = [], bet: 0, split: false, split_aces: false)
74 @cards = cards.dup
75 @bet, @split, @split_aces = bet, split, split_aces
76 @finished = false
77 @reason = nil
78 end
79
80 def total
81 sum = @cards.sum(&:point)
82 soft = @cards.any? { |card| card.rank.zero? } && sum + 10 <= 21
83 Total.new(sum + (soft ? 10 : 0), soft)
84 end
85
86 def blackjack? = !@split && @cards.length == 2 && total.value == 21
87
88 def to_s
89 t = total
90 status = if blackjack? then 'blackjack'
91 elsif t.value > 21 then "bust #{t.value}"
92 elsif t.soft then "soft #{t.value}"
93 else t.value.to_s
94 end
95 "#{@cards.join(' ')} (#{status})"
96 end
97
98 def outcome(dealer, dealer_natural)
99 return blackjack? ? :push : :lose if dealer_natural
100 return :blackjack if blackjack?
101 return @reason if @reason
102 return :win if dealer > 21 || total.value > dealer
103 total.value == dealer ? :push : :lose
104 end
105end

107def hand_profit(result, bet, rules)
108 case result
109 when :blackjack then rules.six_to_five ? bet * 6 / 5 : bet * 3 / 2
110 when :win, :charlie then bet
111 when :push then 0
112 when :surrender then bet / 2 - bet
113 else -bet
114 end
115end

157class Shoe
158 attr_reader :decks, :cards, :drawn
159 attr_accessor :running
160
161 def initialize(decks, seed)
162 @decks, @random = decks, seed
163 shuffle
164 end
165
166 def shuffle
167 @cards = Array.new(52 * @decks) { |i| Card.new(i % 52) }
168 (@cards.length - 1).downto(1) do |i|
169 @random = @random * 48_271 % 2_147_483_647
170 j = @random % (i + 1)
171 @cards[i], @cards[j] = @cards[j], @cards[i]
172 end
173 @drawn = @running = 0
174 end
175
176 def cut(penetration) = @cards.length * penetration / 100
177
178 def draw(visible = true)
179 card = @cards[@drawn]
180 @drawn += 1
181 @running += card.hilo if visible
182 card
183 end
184end

190class Table
191 COMMANDS = %w[SHOE RULE RULES PLAYER LEAVE REBUY BET SIDE DEAL INSURANCE HIT STAND DOUBLE SPLIT SURRENDER HINT AUTO STATUS COUNT STATS].freeze
192 ACTIONS = { 'HIT' => :hit, 'STAND' => :stand, 'DOUBLE' => :double, 'SPLIT' => :split, 'SURRENDER' => :surrender }.freeze
193 IDLE_COMMANDS = %w[SHOE RULE PLAYER LEAVE REBUY BET SIDE DEAL].freeze
194 NAMED_COMMANDS = %w[LEAVE REBUY BET SIDE INSURANCE AUTO].freeze
195 AMOUNT_COMMANDS = %w[REBUY BET SIDE INSURANCE].freeze
196
197 def initialize
198 @rules = Rules.defaults
199 @seats = Array.new(7)
200 @participants = []
201 @output = []
202 @shoe = nil
203 @state = :idle
204 @round = @dealer_blackjacks = @dealer_busts = @insurance_turn = 0
205 @dealer = Hand.new
206 @hole_visible = false
207 end
208
209 def execute(words)
210 @output = []
211 command, first, second = words
212 raise CommandError, "unknown command #{command}" unless COMMANDS.include?(command)
213 action = ACTIONS[command]
214 target = proposed = nil
215 amount = seed = 0
216 case command
217 when 'SHOE'
218 amount = integer(first, 1, 8)
219 seed = integer(second, 1, 2_147_483_646)
220 when 'RULE'
221 proposed = @rules.changed(first, second)
222 when 'PLAYER'
223 raise CommandError, "bad name #{first}" unless /\A[A-Z]{1,8}\z/.match?(first)
224 amount = integer(second, 1, 1_000_000)
225 else
226 if action || NAMED_COMMANDS.include?(command)
227 target = find(first)
228 raise CommandError, "unknown player #{first}" unless target
229 amount = integer(second, command == 'REBUY' ? 1 : 0, 1_000_000) if AMOUNT_COMMANDS.include?(command)
230 end
231 end
232 if IDLE_COMMANDS.include?(command)
233 raise CommandError, 'round in progress' unless @state == :idle
234 elsif command == 'INSURANCE'
235 raise CommandError, 'no insurance' unless @state == :insurance
236 elsif action || command == 'AUTO' || command == 'HINT'
237 raise CommandError, 'no round' if @state == :idle
238 raise CommandError, 'insurance pending' if @state == :insurance
239 end
240 if action || command == 'AUTO'
241 turn = current
242 raise CommandError, "waiting for #{turn.name}" unless turn.player.equal?(target)
243 if action
244 problem = action_error(action, target, turn.hand)
245 raise CommandError, problem if problem
246 perform(action)
247 else
248 perform(recommend) while @state == :play && current.player.equal?(target)
249 end
250 return @output
251 end
252 case command
253 when 'SHOE'
254 @shoe = Shoe.new(amount, seed)
255 emit('shoe %d decks %d cards cut %d', @shoe.decks, @shoe.cards.length, @shoe.cut(@rules.penetration))
256 when 'RULE'
257 raise CommandError, 'min above max' if proposed.minimum > proposed.maximum
258 @rules = proposed
259 emit('rule %s %s', first, second)
260 when 'RULES'
261 emit('%s', @rules)
262 when 'PLAYER'
263 raise CommandError, 'player exists' if find(first)
264 seat = @seats.index(nil)
265 raise CommandError, 'table full' if seat.nil?
266 @seats[seat] = Player.new(first, seat, amount)
267 emit('%s sits at seat %d with %d', first, seat + 1, amount)
268 when 'LEAVE'
269 emit('%s leaves with %d net %+d', target.name, target.bankroll, target.net)
270 @seats[target.seat] = nil
271 when 'REBUY'
272 target.bankroll += amount
273 emit('%s rebuys %d bankroll %d', target.name, amount, target.bankroll)
274 when 'BET'
275 unless amount.zero?
276 raise CommandError, 'bet out of limits' unless amount.between?(@rules.minimum, @rules.maximum)
277 raise CommandError, 'insufficient funds' if amount + target.side > target.bankroll
278 end
279 target.bet = amount
280 if amount.zero?
281 target.side = 0
282 emit('%s bet cleared', target.name)
283 else
284 emit('%s bets %d', target.name, amount)
285 end
286 when 'SIDE'
287 unless amount.zero?
288 raise CommandError, 'no bet' if target.bet.zero?
289 raise CommandError, 'side out of limits' if amount > @rules.side_maximum
290 raise CommandError, 'insufficient funds' if target.bet + amount > target.bankroll
291 end
292 target.side = amount
293 if amount.zero? then emit('%s side cleared', target.name)
294 else emit('%s side %d', target.name, amount)
295 end
296 when 'DEAL'
297 raise CommandError, 'no shoe' unless @shoe
298 raise CommandError, 'no bets' unless @seats.compact.any? { |player| !player.bet.zero? }
299 deal
300 when 'INSURANCE'
301 waiting = @participants[@insurance_turn]
302 raise CommandError, "waiting for #{waiting.name}" unless target.equal?(waiting)
303 raise CommandError, 'insurance out of limits' if amount > target.bet / 2
304 raise CommandError, 'insufficient funds' if amount > target.bankroll
305 target.bankroll -= amount
306 target.insurance = amount
307 if amount.zero? then emit('%s declines insurance', target.name)
308 else emit('%s insures %d', target.name, amount)
309 end
310 @insurance_turn += 1
311 peek if @insurance_turn == @participants.length
312 when 'HINT'
313 emit('hint %s: %s', current.name, recommend)
314 when 'STATUS'
315 status
316 when 'COUNT'
317 raise CommandError, 'no shoe' unless @shoe
318 remaining = @shoe.cards.length - @shoe.drawn
319 scaled = @shoe.running.abs * 5200 / remaining
320 emit('count running %d true %s%d.%02d remaining %d', @shoe.running, @shoe.running.negative? ? '-' : '', scaled / 100, scaled % 100, remaining)
321 when 'STATS'
322 players = @seats.compact
323 raise CommandError, 'no players' if players.empty?
324 players.each { |player| emit('stats %s %s net %+d', player.name, player.stats, player.net) }
325 emit('stats dealer rounds %d blackjacks %d busts %d', @round, @dealer_blackjacks, @dealer_busts)
326 end
327 @output
328 end

352 def finish_check(player, hand)
353 value = hand.total.value
354 if value > 21
355 hand.finished = true
356 hand.reason = :bust
357 elsif !@rules.charlie.zero? && hand.cards.length == @rules.charlie
358 hand.finished = true
359 hand.reason = :charlie
360 elsif value == 21
361 hand.finished = true
362 elsif hand.split_aces && hand.cards.length == 2
363 hand.finished = !(hand.cards.all? { |card| card.rank.zero? } && @rules.resplit_aces && player.hands.length < @rules.hands)
364 end
365 end

367 def settle(dealer_natural)
368 dealer_total = @dealer.total.value
369 @participants.each do |player|
370 player.hands.each_with_index do |hand, index|
371 result = hand.outcome(dealer_total, dealer_natural)
372 profit = hand_profit(result, hand.bet, @rules)
373 player.bankroll += hand.bet + profit
374 player.net += profit
375 player.stats.record(result)
376 emit('%s %s %+d', player.hand_name(index), result, profit)
377 end
378 emit('%s bankroll %d', player.name, player.bankroll)
379 end
380 @seats.compact.each { |player| player.bet = player.side = 0 }
381 @state = :idle
382 end

384 def dealer_turn
385 reveal
386 needs_draw = @participants.any? { |player| player.hands.any? { |hand| hand.reason.nil? && !hand.blackjack? } }
387 drew = false
388 if needs_draw
389 loop do
390 total = @dealer.total
391 break unless total.value <= 16 || total.value == 17 && total.soft && @rules.hit_soft17
392 @dealer.cards << @shoe.draw
393 drew = true
394 emit('dealer hits: %s', @dealer)
395 end
396 end
397 @dealer_busts += 1 if drew && @dealer.total.value > 21
398 settle(false)
399 end

401 def advance
402 loop do
403 turn = current
404 unless turn
405 dealer_turn
406 return
407 end
408 return unless turn.hand.cards.length == 1
409 turn.hand.cards << @shoe.draw
410 emit('%s: %s', turn.name, turn.hand)
411 finish_check(turn.player, turn.hand)
412 return unless turn.hand.finished
413 end
414 end

416 def peek
417 natural = @dealer.blackjack?
418 emit('dealer peeks: %s', natural ? 'blackjack' : 'no blackjack')
419 @participants.each do |player|
420 next if player.insurance.zero?
421 profit = player.insurance * (natural ? 2 : -1)
422 player.bankroll += player.insurance + profit
423 player.net += profit
424 emit('%s insurance %+d', player.name, profit)
425 end
426 if natural
427 @dealer_blackjacks += 1
428 reveal
429 settle(true)
430 else
431 @state = :play
432 advance
433 end
434 end

436 def deal
437 if @shoe.drawn >= @shoe.cut(@rules.penetration)
438 @shoe.shuffle
439 emit('shuffle')
440 end
441 @round += 1
442 emit('round %d', @round)
443 @participants = @seats.compact.reject { |player| player.bet.zero? }
444 @participants.each do |player|
445 player.bankroll -= player.bet + player.side
446 player.insurance = 0
447 player.stats.rounds += 1
448 player.hands.replace([Hand.new(bet: player.bet)])
449 end
450 @dealer = Hand.new
451 @hole_visible = false
452 @participants.each { |player| player.hands[0].cards << @shoe.draw }
453 @dealer.cards << @shoe.draw
454 @participants.each do |player|
455 hand = player.hands[0]
456 hand.cards << @shoe.draw
457 hand.finished = hand.blackjack?
458 end
459 @dealer.cards << @shoe.draw(false)
460 @participants.each { |player| emit('%s: %s', player.hand_name(0), player.hands[0]) }
461 emit('dealer: %s ??', @dealer.cards[0])
462 @participants.each do |player|
463 next if player.side.zero?
464 first, second = player.hands[0].cards
465 result, multiplier = if first.rank != second.rank then [:lose, -1]
466 elsif first.suit == second.suit then [:perfect, 25]
467 elsif first.black? == second.black? then [:colored, 12]
468 else [:mixed, 6]
469 end
470 profit = player.side * multiplier
471 player.bankroll += player.side + profit
472 player.net += profit
473 emit('%s side %s %+d', player.name, result, profit)
474 end
475 case @dealer.cards[0].point
476 when 1
477 @state = :insurance
478 @insurance_turn = 0
479 emit('insurance offered')
480 when 10 then peek
481 else
482 @state = :play
483 advance
484 end
485 end

487 def action_error(action, player, hand)
488 case action
489 when :hit
490 return 'split aces' if hand.split_aces
491 when :double
492 return 'split aces' if hand.split_aces
493 return 'two cards only' unless hand.cards.length == 2
494 total = hand.total
495 minimum = @rules.doubling == '10-11' ? 10 : 9
496 if @rules.doubling != 'ANY' && (total.soft || !total.value.between?(minimum, 11)) || !@rules.double_after_split && hand.split
497 return 'double not allowed'
498 end
499 return 'insufficient funds' if player.bankroll < hand.bet
500 when :split
501 return 'two cards only' unless hand.cards.length == 2
502 return 'not a pair' unless hand.cards[0].point == hand.cards[1].point
503 return 'hand limit' if player.hands.length >= @rules.hands
504 return 'insufficient funds' if player.bankroll < hand.bet
505 when :surrender
506 return 'surrender not allowed' if !@rules.surrender || hand.split
507 return 'two cards only' unless hand.cards.length == 2
508 end
509 nil
510 end

512 def recommend
513 turn = current
514 hand = turn.hand
515 possible = ->(action) { action_error(action, turn.player, hand).nil? }
516 up = @dealer.cards[0].point
517 up = 11 if up == 1
518 return possible.call(:split) ? :split : :stand if hand.split_aces
519 if possible.call(:split)
520 pair = hand.cards[0].point
521 if pair == 1 || pair == 8 || pair == 9 && ![7, 10, 11].include?(up) ||
522 [2, 3, 7].include?(pair) && up <= 7 || pair == 6 && up <= 6 || pair == 4 && [5, 6].include?(up)
523 return :split
524 end
525 end
526 total = hand.total
527 value = total.value
528 symbol = :h
529 if total.soft
530 if value >= 19
531 symbol = :s
532 elsif value == 18
533 symbol = up.between?(3, 6) ? :ds : [2, 7, 8].include?(up) ? :s : :h
534 elsif value == 17 && up.between?(3, 6) || [15, 16].include?(value) && up.between?(4, 6) || [13, 14].include?(value) && up.between?(5, 6)
535 symbol = :d
536 end
537 else
538 if value >= 17
539 symbol = :s
540 elsif value >= 13
541 symbol = up <= 6 ? :s : value == 16 && up >= 9 || value == 15 && up == 10 ? :r : :h
542 elsif value == 12
543 symbol = up.between?(4, 6) ? :s : :h
544 elsif value == 11 && up <= 10 || value == 10 && up <= 9 || value == 9 && up.between?(3, 6)
545 symbol = :d
546 end
547 end
548 case symbol
549 when :s then :stand
550 when :d then possible.call(:double) ? :double : :hit
551 when :ds then possible.call(:double) ? :double : :stand
552 when :r then possible.call(:surrender) ? :surrender : :hit
553 else :hit
554 end
555 end

557 def perform(action)
558 turn = current
559 player, hand, name = turn.player, turn.hand, turn.name
560 case action
561 when :split
562 first, second = hand.cards
563 aces = hand.split_aces || first.rank.zero?
564 player.bankroll -= hand.bet
565 player.stats.splits += 1
566 emit('%s splits %s %s', name, first, second)
567 replacement = Hand.new([first], bet: hand.bet, split: true, split_aces: aces)
568 player.hands[turn.index] = replacement
569 player.hands.insert(turn.index + 1, Hand.new([second], bet: hand.bet, split: true, split_aces: aces))
570 replacement.cards << @shoe.draw
571 emit('%s: %s', name, replacement)
572 finish_check(player, replacement)
573 when :stand, :surrender
574 hand.finished = true
575 hand.reason = :surrender if action == :surrender
576 emit('%s %s: %s', name, action == :stand ? 'stands' : 'surrenders', hand)
577 when :hit, :double
578 if action == :double
579 player.bankroll -= hand.bet
580 hand.bet *= 2
581 player.stats.doubles += 1
582 end
583 hand.cards << @shoe.draw
584 emit('%s %s: %s', name, action == :double ? 'doubles' : 'hits', hand)
585 finish_check(player, hand)
586 hand.finished = true if action == :double
587 end
588 advance
589 end

592 case @state
593 when :idle then emit('table idle round %d', @round)
594 when :insurance then emit('table insurance round %d next %s', @round, @participants[@insurance_turn].name)
595 when :play then emit('table play round %d turn %s', @round, current.name)

598 @seats.compact.each do |player|
599 emit('seat %d %s bankroll %d bet %d side %d', player.seat + 1, player.name, player.bankroll, player.bet, player.side)
600 next if @state == :idle || !@participants.include?(player)
601 player.hands.each_with_index { |hand, index| emit('%s: %s bet %d', player.hand_name(index), hand, hand.bet) }
602 end
603 end
