26bool one_of(std::string_view word, std::initializer_list<std::string_view> choices) {
27 return std::find(choices.begin(), choices.end(), word) != choices.end();

30std::int64_t integer(const std::string& word, std::int64_t minimum, std::int64_t maximum) {
31 if (word.empty() || word.size() > 10 || (word.size() > 1 && word.front() == '0'))
32 fail("bad value " + word);
33 std::int64_t value = 0;
34 for (char ch : word) {
35 if (ch < '0' || ch > '9') fail("bad value " + word);
36 value = value * 10 + ch - '0';
37 }
38 if (value < minimum || value > maximum) fail("bad value " + word);
39 return value;
40}

46int rank(Card card) { return card % 13; }
47int suit(Card card) { return card / 13; }
48int points(Card card) { return std::min(rank(card) + 1, 10); }
49bool black(Card card) { return suit(card) == 0 || suit(card) == 3; }

51std::string card_text(Card card) {
52 return std::string(1, "A23456789TJQK"[rank(card)]) + "CDHS"[suit(card)];
53}

55struct Total { int value; bool soft; };
56
57Total total(const std::vector<Card>& cards) {
58 int sum = 0;
59 bool ace = false;
60 for (Card card : cards) {
61 sum += points(card);
62 ace = ace || rank(card) == 0;
63 }
64 const bool soft = ace && sum + 10 <= 21;
65 return {sum + (soft ? 10 : 0), soft};
66}
67
68bool blackjack(const std::vector<Card>& cards, bool split = false) {
69 return !split && cards.size() == 2 && total(cards).value == 21;
70}
71
72std::string hand_text(const std::vector<Card>& cards, bool split = false) {
73 const Total t = total(cards);
74 std::ostringstream out;
75 for (std::size_t i = 0; i < cards.size(); ++i) {
76 if (i) out << ' ';
77 out << card_text(cards[i]);
78 }
79 out << " (";
80 if (blackjack(cards, split)) out << "blackjack";
81 else {
82 if (t.value > 21) out << "bust ";
83 else if (t.soft) out << "soft ";
84 out << t.value;
85 }
86 out << ')';
87 return out.str();
88}

117struct Rules {
118 bool hit_soft17 = false;
119 int penetration = 75;
120 int minimum = 10;
121 int maximum = 500;
122 bool six_to_five = false;
123 bool double_after_split = true;
124 DoubleRule doubling = DoubleRule::Any;
125 int hands = 4;
126 bool resplit_aces = false;
127 bool surrender = true;
128 int charlie = 0;
129 int side_maximum = 25;
130
131 Rules changed(const std::string& name, const std::string& word) const {
132 Rules next = *this;
133 auto choice = [&](std::initializer_list<std::string_view> allowed) {
134 if (!one_of(word, allowed)) fail("bad value " + word);
135 };
136 if (name == "DEALER") {
137 choice({"S17", "H17"});
138 next.hit_soft17 = word == "H17";
139 } else if (name == "PEN") next.penetration = static_cast<int>(integer(word, 50, 90));
140 else if (name == "MIN") next.minimum = static_cast<int>(integer(word, 1, 10000));
141 else if (name == "MAX") next.maximum = static_cast<int>(integer(word, 1, 10000));
142 else if (name == "BJ") {
143 choice({"3:2", "6:5"});
144 next.six_to_five = word == "6:5";
145 } else if (name == "DAS") {
146 choice({"ON", "OFF"});
147 next.double_after_split = word == "ON";
148 } else if (name == "DOUBLE") {
149 choice({"ANY", "9-11", "10-11"});
150 next.doubling = word == "ANY" ? DoubleRule::Any
151 : word == "9-11" ? DoubleRule::Hard9 : DoubleRule::Hard10;
152 } else if (name == "HANDS") next.hands = static_cast<int>(integer(word, 1, 4));
153 else if (name == "RSA") {
154 choice({"ON", "OFF"});
155 next.resplit_aces = word == "ON";
156 } else if (name == "SURRENDER") {
157 choice({"ON", "OFF"});
158 next.surrender = word == "ON";
159 } else if (name == "CHARLIE") {
160 choice({"0", "5", "6", "7"});
161 next.charlie = word.front() - '0';
162 } else if (name == "SIDEMAX") next.side_maximum = static_cast<int>(integer(word, 0, 1000));
163 else fail("unknown rule " + name);
164 return next;
165 }
166
167 std::string text() const {
168 auto on = [](bool value) { return value ? "ON" : "OFF"; };
169 std::ostringstream out;
170 out << "rules DEALER " << (hit_soft17 ? "H17" : "S17")
171 << " PEN " << penetration << " MIN " << minimum << " MAX " << maximum
172 << " BJ " << (six_to_five ? "6:5" : "3:2") << " DAS " << on(double_after_split)
173 << " DOUBLE " << (doubling == DoubleRule::Any ? "ANY" : doubling == DoubleRule::Hard9 ? "9-11" : "10-11")
174 << " HANDS " << hands << " RSA " << on(resplit_aces) << " SURRENDER " << on(surrender)
175 << " CHARLIE " << charlie << " SIDEMAX " << side_maximum;
176 return out.str();
177 }

229 void shuffle() {
230 shoe.resize(52 * decks);
231 for (std::size_t i = 0; i < shoe.size(); ++i) shoe[i] = static_cast<int>(i % 52);
232 for (int i = static_cast<int>(shoe.size()) - 1; i >= 1; --i) {
233 random = random * 48271 % 2147483647;
234 std::swap(shoe[i], shoe[static_cast<std::size_t>(random % (i + 1))]);
235 }
236 drawn = running = 0;

267 void finish_check(Player& player, Hand& hand) const {
268 const int value = total(hand.cards).value;
269 if (value > 21) {
270 hand.reason = Finish::Bust;
271 hand.finished = true;
272 } else if (rules.charlie != 0 && hand.cards.size() == static_cast<std::size_t>(rules.charlie)) {
273 hand.reason = Finish::Charlie;
274 hand.finished = true;
275 } else if (value == 21) hand.finished = true;
276 else if (hand.split_aces && hand.cards.size() == 2) {
277 hand.finished = !(rank(hand.cards[0]) == 0 && rank(hand.cards[1]) == 0
278 && rules.resplit_aces && player.hands.size() < static_cast<std::size_t>(rules.hands));
279 }

282 void settle(bool dealer_natural = false) {
283 const int dealer_total = total(dealer).value;
284 for (int seat : participants) {
285 Player& player = *seats[seat];
286 for (std::size_t i = 0; i < player.hands.size(); ++i) {
287 const Hand& hand = player.hands[i];
288 const bool natural = blackjack(hand.cards, hand.split);
289 Result result;
290 if (dealer_natural) result = natural ? Result::Push : Result::Lose;
291 else if (natural) result = Result::Blackjack;
292 else if (hand.reason == Finish::Bust) result = Result::Bust;
293 else if (hand.reason == Finish::Surrender) result = Result::Surrender;
294 else if (hand.reason == Finish::Charlie) result = Result::Charlie;
295 else {
296 const int value = total(hand.cards).value;
297 result = dealer_total > 21 || value > dealer_total ? Result::Win
298 : value == dealer_total ? Result::Push : Result::Lose;
299 }
300 Money profit;
301 if (result == Result::Blackjack)
302 profit = rules.six_to_five ? hand.bet * 6 / 5 : hand.bet * 3 / 2;
303 else if (result == Result::Win || result == Result::Charlie) profit = hand.bet;
304 else if (result == Result::Push) profit = 0;
305 else if (result == Result::Surrender) profit = hand.bet / 2 - hand.bet;
306 else profit = -hand.bet;
307 player.bankroll += hand.bet + profit;
308 player.net += profit;
309 ++player.stats.hands;
310 if (result == Result::Blackjack || result == Result::Win || result == Result::Charlie) ++player.stats.wins;
311 else if (result == Result::Push) ++player.stats.pushes;
312 else ++player.stats.losses;
313 if (result == Result::Blackjack) ++player.stats.blackjacks;
314 if (result == Result::Bust) ++player.stats.busts;
315 if (result == Result::Surrender) ++player.stats.surrenders;
316 std::cout << player.hand_name(i) << ' ' << result_name(result) << ' ' << signed_number(profit) << '\n';
317 }
318 std::cout << player.name << " bankroll " << player.bankroll << '\n';
319 }
320 for (auto& player : seats) if (player) player->bet = player->side = 0;
321 phase = Phase::Idle;
322 }

324 void dealer_turn() {
325 reveal_dealer();
326 bool must_play = false;
327 for (int seat : participants)
328 for (const Hand& hand : seats[seat]->hands)
329 must_play = must_play || (hand.reason == Finish::None && !blackjack(hand.cards, hand.split));
330 bool drew = false;
331 if (must_play) {
332 for (;;) {
333 const Total t = total(dealer);
334 if (!(t.value <= 16 || (t.value == 17 && t.soft && rules.hit_soft17))) break;
335 dealer.push_back(draw());
336 drew = true;
337 std::cout << "dealer hits: " << hand_text(dealer) << '\n';
338 }
339 }
340 if (drew && total(dealer).value > 21) ++dealer_busts;
341 settle();
342 }

356 finish_check(player, hand);

382 void deal() {
383 if (drawn >= cut()) {
384 shuffle();
385 std::cout << "shuffle\n";
386 }
387 std::cout << "round " << ++round << '\n';
388 participants.clear();
389 for (int i = 0; i < 7; ++i) if (seats[i] && seats[i]->bet) participants.push_back(i);
390 for (int seat : participants) {
391 Player& player = *seats[seat];
392 player.bankroll -= player.bet + player.side;
393 player.insurance = 0;
394 ++player.stats.rounds;
395 player.hands = {Hand{{draw()}, player.bet}};
396 }
397 dealer = {draw()};
398 for (int seat : participants) {
399 Hand& hand = seats[seat]->hands[0];
400 hand.cards.push_back(draw());
401 hand.finished = blackjack(hand.cards);
402 }
403 dealer.push_back(draw(false));
404 hole_revealed = false;
405 for (int seat : participants) {
406 const Player& player = *seats[seat];
407 std::cout << player.hand_name(0) << ": " << hand_text(player.hands[0].cards) << '\n';
408 }
409 std::cout << "dealer: " << card_text(dealer[0]) << " ??\n";
410 for (int seat : participants) {
411 Player& player = *seats[seat];
412 if (!player.side) continue;
413 const Card a = player.hands[0].cards[0], b = player.hands[0].cards[1];
414 std::string result = "lose";
415 int multiplier = -1;
416 if (rank(a) == rank(b)) {
417 if (suit(a) == suit(b)) { result = "perfect"; multiplier = 25; }
418 else if (black(a) == black(b)) { result = "colored"; multiplier = 12; }
419 else { result = "mixed"; multiplier = 6; }
420 }
421 const Money profit = player.side * multiplier;
422 player.bankroll += player.side + profit;
423 player.net += profit;
424 std::cout << player.name << " side " << result << ' ' << signed_number(profit) << '\n';
425 }
426 if (points(dealer[0]) == 1) {
427 phase = Phase::Insurance;
428 insurance_turn = 0;
429 std::cout << "insurance offered\n";
430 } else if (points(dealer[0]) == 10) peek();
431 else {
432 phase = Phase::Play;
433 advance();
434 }

437 std::optional<std::string> action_error(Action action, const Player& player, const Hand& hand) const {
438 if (action == Action::Hit) {
439 if (hand.split_aces) return "split aces";
440 } else if (action == Action::Double) {
441 if (hand.split_aces) return "split aces";
442 if (hand.cards.size() != 2) return "two cards only";
443 const Total t = total(hand.cards);
444 if ((rules.doubling != DoubleRule::Any
445 && (t.soft || t.value < (rules.doubling == DoubleRule::Hard9 ? 9 : 10) || t.value > 11))
446 || (!rules.double_after_split && hand.split)) return "double not allowed";
447 if (player.bankroll < hand.bet) return "insufficient funds";
448 } else if (action == Action::Split) {
449 if (hand.cards.size() != 2) return "two cards only";
450 if (points(hand.cards[0]) != points(hand.cards[1])) return "not a pair";
451 if (player.hands.size() >= static_cast<std::size_t>(rules.hands)) return "hand limit";
452 if (player.bankroll < hand.bet) return "insufficient funds";
453 } else if (action == Action::Surrender) {
454 if (!rules.surrender || hand.split) return "surrender not allowed";
455 if (hand.cards.size() != 2) return "two cards only";
456 }
457 return std::nullopt;
458 }

460 void perform(Action action) {
461 const Turn turn = current().value();
462 Player& player = *seats[turn.seat];
463 const std::string name = player.hand_name(turn.hand);
464 if (action == Action::Split) {
465 const Hand old = player.hands[turn.hand];
466 const Card first = old.cards[0], second = old.cards[1];
467 const bool aces = old.split_aces || points(first) == 1;
468 player.bankroll -= old.bet;
469 ++player.stats.splits;
470 std::cout << name << " splits " << card_text(first) << ' ' << card_text(second) << '\n';
471 player.hands[turn.hand] = Hand{{first}, old.bet, true, aces};
472 player.hands.insert(player.hands.begin() + static_cast<std::ptrdiff_t>(turn.hand + 1),
473 Hand{{second}, old.bet, true, aces});
474 Hand& hand = player.hands[turn.hand];
475 hand.cards.push_back(draw());
476 std::cout << name << ": " << hand_text(hand.cards, true) << '\n';
477 finish_check(player, hand);
478 } else {
479 Hand& hand = player.hands[turn.hand];
480 if (action == Action::Stand || action == Action::Surrender) {
481 hand.finished = true;
482 if (action == Action::Surrender) hand.reason = Finish::Surrender;
483 std::cout << name << (action == Action::Stand ? " stands: " : " surrenders: ")
484 << hand_text(hand.cards, hand.split) << '\n';
485 } else {
486 if (action == Action::Double) {
487 player.bankroll -= hand.bet;
488 hand.bet *= 2;
489 ++player.stats.doubles;
490 }
491 hand.cards.push_back(draw());
492 std::cout << name << (action == Action::Double ? " doubles: " : " hits: ")
493 << hand_text(hand.cards, hand.split) << '\n';
494 finish_check(player, hand);
495 if (action == Action::Double) hand.finished = true;
496 }
497 }
498 advance();
499 }

501 Action recommend() const {
502 const Turn turn = current().value();
503 const Player& player = *seats[turn.seat];
504 const Hand& hand = player.hands[turn.hand];
505 auto possible = [&](Action action) { return !action_error(action, player, hand); };
506 const int u = points(dealer[0]) == 1 ? 11 : points(dealer[0]);
507 if (hand.split_aces) return possible(Action::Split) ? Action::Split : Action::Stand;
508 if (hand.cards.size() == 2 && points(hand.cards[0]) == points(hand.cards[1]) && possible(Action::Split)) {
509 const int v = points(hand.cards[0]);
510 if (v == 1 || v == 8 || (v == 9 && u != 7 && u != 10 && u != 11)
511 || ((v == 2 || v == 3 || v == 7) && u <= 7) || (v == 6 && u <= 6)
512 || (v == 4 && (u == 5 || u == 6))) return Action::Split;
513 }
514 const Total t = total(hand.cards);
515 enum class Symbol { H, S, D, Ds, R };
516 Symbol symbol = Symbol::H;
517 if (t.soft) {
518 if (t.value >= 19) symbol = Symbol::S;
519 else if (t.value == 18)
520 symbol = u >= 3 && u <= 6 ? Symbol::Ds : u == 2 || u == 7 || u == 8 ? Symbol::S : Symbol::H;
521 else if ((t.value == 17 && u >= 3 && u <= 6)
522 || ((t.value == 15 || t.value == 16) && u >= 4 && u <= 6)
523 || ((t.value == 13 || t.value == 14) && (u == 5 || u == 6))) symbol = Symbol::D;
524 } else {
525 if (t.value >= 17) symbol = Symbol::S;
526 else if (t.value >= 13)
527 symbol = u <= 6 ? Symbol::S
528 : (t.value == 16 && u >= 9) || (t.value == 15 && u == 10) ? Symbol::R : Symbol::H;
529 else if (t.value == 12) symbol = u >= 4 && u <= 6 ? Symbol::S : Symbol::H;
530 else if ((t.value == 11 && u <= 10) || (t.value == 10 && u <= 9)
531 || (t.value == 9 && u >= 3 && u <= 6)) symbol = Symbol::D;
532 }
533 if (symbol == Symbol::S) return Action::Stand;
534 if (symbol == Symbol::D || symbol == Symbol::Ds)
535 return possible(Action::Double) ? Action::Double : symbol == Symbol::Ds ? Action::Stand : Action::Hit;
536 if (symbol == Symbol::R) return possible(Action::Surrender) ? Action::Surrender : Action::Hit;
537 return Action::Hit;
538 }

565 void command(const std::vector<std::string>& words) {
566 const std::string& command = words[0];
567 if (!one_of(command, {"SHOE", "RULE", "RULES", "PLAYER", "LEAVE", "REBUY", "BET", "SIDE", "DEAL",
568 "INSURANCE", "HIT", "STAND", "DOUBLE", "SPLIT", "SURRENDER", "HINT", "AUTO", "STATUS", "COUNT", "STATS"}))
569 fail("unknown command " + command);
570 const std::string first = words.size() > 1 ? words[1] : "";
571 const std::string second = words.size() > 2 ? words[2] : "";
572 const auto action = parse_action(command);
573 int seat = -1;
574 std::int64_t value = 0, seed = 0;
575 std::optional<Rules> proposed_rules;
576
577 if (command == "SHOE") {
578 value = integer(first, 1, 8);
579 seed = integer(second, 1, 2147483646);
580 } else if (command == "RULE") proposed_rules = rules.changed(first, second);
581 else if (command == "PLAYER") {
582 if (first.empty() || first.size() > 8
583 || !std::all_of(first.begin(), first.end(), [](char c) { return c >= 'A' && c <= 'Z'; }))
584 fail("bad name " + first);
585 value = integer(second, 1, 1000000);
586 } else if (action || one_of(command, {"LEAVE", "REBUY", "BET", "SIDE", "INSURANCE", "AUTO"})) {
587 seat = find_player(first);
588 if (seat < 0) fail("unknown player " + first);
589 if (one_of(command, {"REBUY", "BET", "SIDE", "INSURANCE"}))
590 value = integer(second, command == "REBUY" ? 1 : 0, 1000000);
591 }
592 if (one_of(command, {"SHOE", "RULE", "PLAYER", "LEAVE", "REBUY", "BET", "SIDE", "DEAL"})) {
593 if (phase != Phase::Idle) fail("round in progress");
594 } else if (command == "INSURANCE") {
595 if (phase != Phase::Insurance) fail("no insurance");
596 } else if (action || command == "AUTO" || command == "HINT") {
597 if (phase == Phase::Idle) fail("no round");
598 if (phase == Phase::Insurance) fail("insurance pending");
599 }
600
601 if (action || command == "AUTO") {
602 const Turn turn = current().value();
603 if (turn.seat != seat) fail("waiting for " + seats[turn.seat]->hand_name(turn.hand));
604 if (action) {
605 if (auto error = action_error(*action, *seats[seat], seats[seat]->hands[turn.hand])) fail(*error);
606 perform(*action);
607 } else {
608 while (phase == Phase::Play && current()->seat == seat) perform(recommend());
609 }
610 } else if (command == "SHOE") {
611 decks = static_cast<int>(value);
612 random = seed;
613 shuffle();
614 std::cout << "shoe " << decks << " decks " << shoe.size() << " cards cut " << cut() << '\n';
615 } else if (command == "RULE") {
616 if (proposed_rules->minimum > proposed_rules->maximum) fail("min above max");
617 rules = *proposed_rules;
618 std::cout << "rule " << first << ' ' << second << '\n';
619 } else if (command == "RULES") std::cout << rules.text() << '\n';
620 else if (command == "PLAYER") {
621 if (find_player(first) >= 0) fail("player exists");
622 const auto empty = std::find_if(seats.begin(), seats.end(), [](const auto& player) { return !player; });
623 if (empty == seats.end()) fail("table full");
624 const int number = static_cast<int>(empty - seats.begin()) + 1;
625 *empty = Player{first, number, value};
626 std::cout << first << " sits at seat " << number << " with " << value << '\n';
627 } else if (command == "LEAVE") {
628 const Player& player = *seats[seat];
629 std::cout << player.name << " leaves with " << player.bankroll << " net " << signed_number(player.net) << '\n';
630 seats[seat].reset();
631 } else if (command == "REBUY") {
632 Player& player = *seats[seat];
633 player.bankroll += value;
634 std::cout << player.name << " rebuys " << value << " bankroll " << player.bankroll << '\n';
635 } else if (command == "BET") {
636 Player& player = *seats[seat];
637 if (value != 0) {
638 if (value < rules.minimum || value > rules.maximum) fail("bet out of limits");
639 if (value + player.side > player.bankroll) fail("insufficient funds");
640 }
641 player.bet = value;
642 if (value == 0) {
643 player.side = 0;
644 std::cout << player.name << " bet cleared\n";
645 } else std::cout << player.name << " bets " << value << '\n';
646 } else if (command == "SIDE") {
647 Player& player = *seats[seat];
648 if (value != 0) {
649 if (player.bet == 0) fail("no bet");
650 if (value > rules.side_maximum) fail("side out of limits");
651 if (player.bet + value > player.bankroll) fail("insufficient funds");
652 }
653 player.side = value;
654 if (value == 0) std::cout << player.name << " side cleared\n";
655 else std::cout << player.name << " side " << value << '\n';
656 } else if (command == "DEAL") {
657 if (decks == 0) fail("no shoe");
658 if (std::none_of(seats.begin(), seats.end(), [](const auto& player) { return player && player->bet != 0; }))
659 fail("no bets");
660 deal();
661 } else if (command == "INSURANCE") {
662 if (seat != participants[insurance_turn]) fail("waiting for " + seats[participants[insurance_turn]]->name);
663 Player& player = *seats[seat];
664 if (value > player.bet / 2) fail("insurance out of limits");
665 if (value > player.bankroll) fail("insufficient funds");
666 player.bankroll -= value;
667 player.insurance = value;
668 if (value == 0) std::cout << player.name << " declines insurance\n";
669 else std::cout << player.name << " insures " << value << '\n';
670 if (++insurance_turn == participants.size()) peek();
671 } else if (command == "HINT") {
672 const Turn turn = current().value();
673 std::cout << "hint " << seats[turn.seat]->hand_name(turn.hand) << ": " << action_name(recommend()) << '\n';
674 } else if (command == "STATUS") status();
675 else if (command == "COUNT") {
676 if (decks == 0) fail("no shoe");
677 const int remaining = static_cast<int>(shoe.size()) - drawn;
678 const int hundredths = (running < 0 ? -running : running) * 5200 / remaining;
679 std::ostringstream count;
680 if (running < 0) count << '-';
681 count << hundredths / 100 << '.' << std::setw(2) << std::setfill('0') << hundredths % 100;
682 std::cout << "count running " << running << " true " << count.str() << " remaining " << remaining << '\n';
683 } else if (command == "STATS") {
684 if (std::none_of(seats.begin(), seats.end(), [](const auto& player) { return player.has_value(); })) fail("no players");
685 for (const auto& entry : seats) {
686 if (!entry) continue;
687 const Player& player = *entry;
688 const Statistics& s = player.stats;
689 std::cout << "stats " << player.name << " rounds " << s.rounds << " hands " << s.hands
690 << " wins " << s.wins << " losses " << s.losses << " pushes " << s.pushes
691 << " blackjacks " << s.blackjacks << " busts " << s.busts << " surrenders " << s.surrenders
692 << " doubles " << s.doubles << " splits " << s.splits << " net " << signed_number(player.net) << '\n';
693 }
694 std::cout << "stats dealer rounds " << round << " blackjacks " << dealer_blackjacks << " busts " << dealer_busts << '\n';
695 }
696 }
697};
