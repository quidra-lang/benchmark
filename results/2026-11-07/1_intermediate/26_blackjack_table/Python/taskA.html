84 if first.rank != second.rank:
85 return "lose"
86 if first.suit == second.suit:

93def hand_profit(result, bet, blackjack_rule):
94 if result == "blackjack":
95 return bet * 3 // 2 if blackjack_rule == "3:2" else bet * 6 // 5
96 if result in ("win", "charlie"):
97 return bet
98 if result == "push":
99 return 0
100 if result == "surrender":
101 return bet // 2 - bet
102 return -bet

105def pair_split_wanted(value, upcard):
106 if value in (1, 8):
107 return True
108 if value == 9:
109 return upcard not in (7, 10, 11)
110 if value in (2, 3, 7):
111 return upcard <= 7
112 if value == 6:
113 return upcard <= 6
114 if value == 4:
115 return upcard in (5, 6)
116 return False

119def soft_symbol(total, upcard):
120 if total >= 19:
121 return "S"
122 if total == 18:
123 if 3 <= upcard <= 6:
124 return "Ds"
125 return "S" if upcard in (2, 7, 8) else "H"
126 if total == 17:
127 return "D" if 3 <= upcard <= 6 else "H"
128 if total >= 15:
129 return "D" if 4 <= upcard <= 6 else "H"
130 if total >= 13:
131 return "D" if upcard in (5, 6) else "H"
132 return "H"
133
134
135def hard_symbol(total, upcard):
136 if total >= 17:
137 return "S"
138 if total >= 13:
139 if upcard <= 6:
140 return "S"
141 if (total == 16 and upcard >= 9) or (total == 15 and upcard == 10):
142 return "R"
143 return "H"
144 if total == 12:
145 return "S" if 4 <= upcard <= 6 else "H"
146 if total == 11:
147 return "D" if upcard <= 10 else "H"
148 if total == 10:
149 return "D" if upcard <= 9 else "H"
150 if total == 9:
151 return "D" if 3 <= upcard <= 6 else "H"
152 return "H"

155class Card:
156 def __init__(self, rank, suit):
157 self.rank = rank
158 self.suit = suit
159 self.code = rank + suit
160 self.value = min(RANKS.index(rank) + 1, 10)
161
162 def is_black(self):
163 return self.suit in BLACK_SUITS
164
165 def hi_lo(self):
166 if 2 <= self.value <= 6:
167 return 1
168 if 7 <= self.value <= 9:
169 return 0
170 return -1

173class Shoe:
174 def __init__(self, decks, seed):
175 self.decks = decks
176 self.size = 52 * decks
177 self.state = seed
178 self.cards = []
179 self.drawn = 0
180 self.running = 0
181 self.shuffle()
182
183 def next_random(self):
184 self.state = MULTIPLIER * self.state % MODULUS
185 return self.state
186
187 def shuffle(self):
188 self.cards = [
189 Card(rank, suit)
190 for _ in range(self.decks) for suit in SUITS for rank in RANKS
191 ]
192 for i in range(self.size - 1, 0, -1):
193 j = self.next_random() % (i + 1)
194 self.cards[i], self.cards[j] = self.cards[j], self.cards[i]
195 self.drawn = 0
196 self.running = 0
197
198 def draw(self, face_up=True):
199 card = self.cards[self.drawn]
200 self.drawn += 1
201 if face_up:
202 self.running += card.hi_lo()
203 return card
204
205 def reveal(self, card):
206 self.running += card.hi_lo()
207
208 def cut(self, penetration):
209 return self.size * penetration // 100
210
211 def remaining(self):
212 return self.size - self.drawn

215class Hand:
216 def __init__(self, bet, split=False, split_aces=False):
217 self.cards = []
218 self.bet = bet
219 self.split = split
220 self.split_aces = split_aces
221 self.finished = False
222 self.reason = None
223
224 def add(self, card):
225 self.cards.append(card)
226
227 def total(self):
228 points = sum(card.value for card in self.cards)
229 if any(card.rank == "A" for card in self.cards) and points + 10 <= 21:
230 return points + 10, True
231 return points, False
232
233 def is_bust(self):
234 return self.total()[0] > 21
235
236 def is_blackjack(self):
237 return not self.split and len(self.cards) == 2 and self.total()[0] == 21
238
239 def finish(self, reason):
240 self.finished = True
241 self.reason = reason
242
243 def display(self):
244 total, soft = self.total()
245 if self.is_blackjack():
246 status = "blackjack"
247 elif total > 21:
248 status = f"bust {total}"
249 elif soft:
250 status = f"soft {total}"
251 else:
252 status = str(total)
253 return " ".join(card.code for card in self.cards) + f" ({status})"

368 def can_resplit_aces(self, player, hand):
369 return (
370 all(card.rank == "A" for card in hand.cards)

372 and len(player.hands) < self.rule_number("HANDS")

375 def auto_finish(self, player, hand):
376 total = hand.total()[0]
377 charlie = self.rule_number("CHARLIE")
378 if total > 21:
379 hand.finish("bust")
380 elif charlie != 0 and len(hand.cards) == charlie:
381 hand.finish("charlie")
382 elif total == 21:
383 hand.finish("stand")
384 elif hand.split_aces and len(hand.cards) == 2 and not self.can_resplit_aces(player, hand):
385 hand.finish("stand")

387 def double_allowed(self, hand):
388 total, soft = hand.total()
389 rule = self.rules["DOUBLE"]
390 if rule == "9-11" and (soft or not 9 <= total <= 11):
391 return False
392 if rule == "10-11" and (soft or not 10 <= total <= 11):
393 return False
394 return not (self.rules["DAS"] == "OFF" and hand.split)

396 def action_error(self, action, player, hand):
397 if action == "hit":
398 if hand.split_aces:
399 return "split aces"
400 elif action == "double":
401 if hand.split_aces:
402 return "split aces"
403 if len(hand.cards) != 2:
404 return "two cards only"
405 if not self.double_allowed(hand):
406 return "double not allowed"
407 if player.bankroll < hand.bet:
408 return "insufficient funds"
409 elif action == "split":
410 if len(hand.cards) != 2:
411 return "two cards only"
412 if hand.cards[0].value != hand.cards[1].value:
413 return "not a pair"
414 if len(player.hands) >= self.rule_number("HANDS"):
415 return "hand limit"
416 if player.bankroll < hand.bet:
417 return "insufficient funds"
418 elif action == "surrender":
419 if self.rules["SURRENDER"] == "OFF" or hand.split:
420 return "surrender not allowed"
421 if len(hand.cards) != 2:
422 return "two cards only"
423 return None

425 def possible(self, action, player, hand):
426 return self.action_error(action, player, hand) is None
427
428 def recommendation(self, player, hand):
429 if hand.split_aces:
430 return "split" if self.possible("split", player, hand) else "stand"
431 upcard = self.upcard_value()
432 if (
433 len(hand.cards) == 2
434 and hand.cards[0].value == hand.cards[1].value
435 and self.possible("split", player, hand)
436 and pair_split_wanted(hand.cards[0].value, upcard)
437 ):
438 return "split"
439 total, soft = hand.total()
440 symbol = soft_symbol(total, upcard) if soft else hard_symbol(total, upcard)
441 preferred, fallback = SYMBOL_ACTIONS[symbol]
442 return preferred if self.possible(preferred, player, hand) else fallback

444 def split_hand(self, player, index):
445 hand = player.hands[index]
446 first, second = hand.cards
447 player.bankroll -= hand.bet
448 player.stats["splits"] += 1
449 self.emit(f"{player.hand_name(index)} splits {first.code} {second.code}")
450 aces = first.rank == "A"
451 hand.cards = [first]
452 hand.split = True
453 hand.split_aces = aces
454 new_hand = Hand(hand.bet, True, aces)
455 new_hand.add(second)
456 player.hands.insert(index + 1, new_hand)
457 hand.add(self.shoe.draw())
458 self.emit(f"{player.hand_name(index)}: {hand.display()}")
459 self.auto_finish(player, hand)

461 def perform(self, action, player, index):
462 hand = player.hands[index]
463 name = player.hand_name(index)
464 if action == "hit":
465 hand.add(self.shoe.draw())
466 self.emit(f"{name} hits: {hand.display()}")
467 self.auto_finish(player, hand)
468 elif action == "stand":
469 hand.finish("stand")
470 self.emit(f"{name} stands: {hand.display()}")
471 elif action == "double":
472 player.bankroll -= hand.bet
473 hand.bet *= 2
474 player.stats["doubles"] += 1
475 hand.add(self.shoe.draw())
476 self.emit(f"{name} doubles: {hand.display()}")
477 self.auto_finish(player, hand)
478 if not hand.finished:
479 hand.finish("stand")
480 elif action == "split":
481 self.split_hand(player, index)
482 else:
483 hand.finish("surrender")
484 self.emit(f"{name} surrenders: {hand.display()}")
485 self.advance()

487 def advance(self):
488 while True:
489 current = self.current_hand()
490 if current is None:
491 self.dealer_turn()
492 return
493 player, index = current
494 hand = player.hands[index]
495 if len(hand.cards) >= 2:
496 return
497 hand.add(self.shoe.draw())
498 self.emit(f"{player.hand_name(index)}: {hand.display()}")
499 self.auto_finish(player, hand)

505 def dealer_hits(self):
506 total, soft = self.dealer.total()
507 return total <= 16 or (total == 17 and soft and self.rules["DEALER"] == "H17")

509 def dealer_turn(self):
510 self.reveal_dealer()
511 if any(
512 hand.reason not in DECIDED_REASONS and not hand.is_blackjack()
513 for player in self.participants for hand in player.hands
514 ):
515 drew = False
516 while self.dealer_hits():
517 self.dealer.add(self.shoe.draw())
518 self.emit("dealer hits:", self.dealer.display())
519 drew = True
520 if drew and self.dealer.is_bust():
521 self.dealer_busts += 1
522 self.settle()

524 def hand_result(self, hand, dealer_total):
525 if self.dealer_blackjack:
526 return "push" if hand.is_blackjack() else "lose"
527 if hand.is_blackjack():
528 return "blackjack"
529 if hand.reason in DECIDED_REASONS:
530 return hand.reason
531 total = hand.total()[0]
532 if dealer_total > 21 or total > dealer_total:
533 return "win"
534 if total == dealer_total:
535 return "push"
536 return "lose"

538 def settle(self):
539 dealer_total = self.dealer.total()[0]
540 for player in self.participants:
541 for index, hand in enumerate(player.hands):
542 result = self.hand_result(hand, dealer_total)
543 profit = hand_profit(result, hand.bet, self.rules["BJ"])
544 player.bankroll += hand.bet + profit
545 player.net += profit
546 player.record(result)
547 self.emit(player.hand_name(index), result, signed(profit))
548 self.emit(player.name, "bankroll", player.bankroll)
549 for player in self.seated():
550 player.bet = 0
551 player.side = 0
552 self.participants = []
553 self.state = "idle"

555 def peek(self):
556 self.dealer_blackjack = self.dealer.is_blackjack()
557 if self.dealer_blackjack:
558 self.emit("dealer peeks: blackjack")
559 else:
560 self.emit("dealer peeks: no blackjack")
561 for player in self.participants:
562 if player.insurance != 0:
563 if self.dealer_blackjack:
564 profit = player.insurance * 2
565 else:
566 profit = -player.insurance
567 player.bankroll += player.insurance + profit
568 player.net += profit
569 self.emit(player.name, "insurance", signed(profit))
570 if self.dealer_blackjack:
571 self.dealer_blackjacks += 1
572 self.reveal_dealer()
573 self.settle()
574 else:
575 self.state = "play"
576 self.advance()

687 def deal(self, arguments):
688 self.require_idle()
689 self.require_shoe()
690 participants = [player for player in self.seated() if player.bet != 0]
691 if not participants:
692 raise CommandError("no bets")
693 if self.shoe.drawn >= self.shoe.cut(self.rule_number("PEN")):
694 self.shoe.shuffle()
695 self.emit("shuffle")
696 self.round_number += 1
697 self.emit("round", self.round_number)
698 self.participants = participants
699 self.dealer_blackjack = False
700 for player in participants:
701 player.bankroll -= player.bet + player.side
702 player.insurance = 0
703 player.stats["rounds"] += 1
704 self.deal_cards()
705 for player in participants:
706 hand = player.hands[0]
707 if hand.is_blackjack():
708 hand.finish("blackjack")
709 self.emit(f"{player.hand_name(0)}: {hand.display()}")
710 upcard = self.dealer.cards[0]
711 self.emit("dealer:", upcard.code, "??")
712 for player in participants:
713 if player.side != 0:
714 self.settle_side(player)
715 if upcard.rank == "A":
716 self.state = "insurance"
717 self.insurance_turn = 0
718 self.emit("insurance offered")
719 elif upcard.value == 10:
720 self.peek()
721 else:
722 self.state = "play"
723 self.advance()

725 def insure(self, arguments):
726 player = self.require_player(arguments[0])
727 amount = integer_value(arguments[1], 0, 1000000)
728 if self.state != "insurance":
729 raise CommandError("no insurance")
730 turn = self.participants[self.insurance_turn]
731 if player is not turn:
732 raise CommandError(f"waiting for {turn.name}")
733 if amount > player.bet // 2:
734 raise CommandError("insurance out of limits")
735 if amount > player.bankroll:
736 raise CommandError("insufficient funds")
737 player.bankroll -= amount
738 player.insurance = amount
739 if amount == 0:
740 self.emit(player.name, "declines insurance")
741 else:
742 self.emit(player.name, "insures", amount)
743 self.insurance_turn += 1
744 if self.insurance_turn == len(self.participants):
745 self.peek()
