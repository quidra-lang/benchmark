31function integerWord(word: string, low: number, high: number): number {
32 if (!/^(0|[1-9][0-9]{0,9})$/.test(word)) fail(`bad value ${word}`);
33 const value = Number(word);
34 if (value < low || value > high) fail(`bad value ${word}`);
35 return value;
36}
37function choice<T extends string>(word: string, choices: readonly T[]): T {
38 if (!(choices as readonly string[]).includes(word)) fail(`bad value ${word}`);
39 return word as T;

41function isRuleName(word: string): word is RuleName {

44function newRules(): Rules {
45 return { DEALER: 'S17', PEN: 75, MIN: 10, MAX: 500, BJ: '3:2', DAS: true,
46 DOUBLE: 'ANY', HANDS: 4, RSA: false, SURRENDER: true, CHARLIE: 0, SIDEMAX: 25 };

48function changeRule(original: Rules, name: RuleName, word: string): Rules {
49 const next = { ...original };
50 switch (name) {
51 case 'DEALER': next.DEALER = choice(word, ['S17', 'H17']); break;
52 case 'PEN': next.PEN = integerWord(word, 50, 90); break;
53 case 'MIN': next.MIN = integerWord(word, 1, 10000); break;
54 case 'MAX': next.MAX = integerWord(word, 1, 10000); break;
55 case 'BJ': next.BJ = choice(word, ['3:2', '6:5']); break;
56 case 'DAS': next.DAS = choice(word, ['ON', 'OFF']) === 'ON'; break;
57 case 'DOUBLE': next.DOUBLE = choice(word, ['ANY', '9-11', '10-11']); break;
58 case 'HANDS': next.HANDS = integerWord(word, 1, 4); break;
59 case 'RSA': next.RSA = choice(word, ['ON', 'OFF']) === 'ON'; break;
60 case 'SURRENDER': next.SURRENDER = choice(word, ['ON', 'OFF']) === 'ON'; break;
61 case 'CHARLIE': {
62 const value = integerWord(word, 0, 7);
63 if (value !== 0 && value !== 5 && value !== 6 && value !== 7) fail(`bad value ${word}`);
64 next.CHARLIE = value;
65 break;
66 }
67 case 'SIDEMAX': next.SIDEMAX = integerWord(word, 0, 1000); break;
68 }
69 return next;
70}

75const point = (card: number): number => Math.min(card % 13 + 1, 10);
76const cardText = (card: number): string => 'A23456789TJQK'[card % 13] + 'CDHS'[Math.floor(card / 13)];
77interface Total { readonly value: number; readonly soft: boolean }
78class Hand {
79 finish: Finish = 'pending';
80 constructor(
81 readonly cards: number[],
82 public bet: number,
83 readonly split = false,
84 readonly splitAces = false,
85 ) {}
86 total(): Total {
87 const raw = this.cards.reduce((sum, card) => sum + point(card), 0);
88 const soft = this.cards.some(card => card % 13 === 0) && raw + 10 <= 21;
89 return { value: raw + (soft ? 10 : 0), soft };
90 }
91 blackjack(): boolean { return !this.split && this.cards.length === 2 && this.total().value === 21; }
92 text(): string {
93 const total = this.total();
94 const status = this.blackjack() ? 'blackjack' : total.value > 21 ? `bust ${total.value}`
95 : total.soft ? `soft ${total.value}` : String(total.value);
96 return `${this.cards.map(cardText).join(' ')} (${status})`;
97 }
98 finishCheck(rules: Rules, handCount: number): void {
99 const total = this.total().value;
100 if (total > 21) this.finish = 'bust';
101 else if (rules.CHARLIE !== 0 && this.cards.length === rules.CHARLIE) this.finish = 'charlie';
102 else if (total === 21) this.finish = 'ordinary';
103 else if (this.splitAces && this.cards.length === 2) {
104 const again = this.cards.every(card => card % 13 === 0) && rules.RSA && handCount < rules.HANDS;
105 this.finish = again ? 'pending' : 'ordinary';
106 }
107 }
108 outcome(dealer: number, dealerBlackjack: boolean): Outcome {
109 if (dealerBlackjack) return this.blackjack() ? 'push' : 'lose';
110 if (this.blackjack()) return 'blackjack';
111 if (this.finish === 'bust' || this.finish === 'surrender' || this.finish === 'charlie') return this.finish;
112 return dealer > 21 || this.total().value > dealer ? 'win' : this.total().value === dealer ? 'push' : 'lose';
113 }
114}
115class Statistics {
116 rounds = 0;
117 hands = 0;
118 wins = 0;
119 losses = 0;
120 pushes = 0;
121 blackjacks = 0;
122 busts = 0;
123 surrenders = 0;
124 doubles = 0;

127 this.hands++;
128 if (result === 'blackjack' || result === 'win' || result === 'charlie') this.wins++;
129 else if (result === 'push') this.pushes++;
130 else this.losses++;
131 if (result === 'blackjack') this.blackjacks++;
132 if (result === 'bust') this.busts++;
133 if (result === 'surrender') this.surrenders++;
134 }

147class Shoe {
148 cards: number[] = [];
149 drawn = 0;
150 running = 0;
151 constructor(readonly decks: number, private randomState: bigint) { this.shuffle(); }
152 shuffle(): void {
153 this.cards = Array.from({ length: 52 * this.decks }, (_, index) => index % 52);
154 for (let index = this.cards.length - 1; index >= 1; index--) {
155 this.randomState = this.randomState * 48271n % 2147483647n;
156 const other = Number(this.randomState % BigInt(index + 1));
157 [this.cards[index], this.cards[other]] = [this.cards[other], this.cards[index]];
158 }
159 this.drawn = this.running = 0;
160 }
161 cut(penetration: number): number { return Math.floor(this.cards.length * penetration / 100); }
162 static hilo(card: number): number {
163 const value = point(card);
164 return value >= 2 && value <= 6 ? 1 : value === 1 || value === 10 ? -1 : 0;
165 }
166 draw(visible: boolean): number {
167 const card = this.cards[this.drawn++];
168 if (visible) this.running += Shoe.hilo(card);
169 return card;
170 }

185 private rules = newRules();
186 private readonly seats: (Player | null)[] = Array(7).fill(null);
187 private participants: Player[] = [];
188 private shoe: Shoe | null = null;
189 private state: TableState = { kind: 'idle' };
190 private round = 0;
191 private dealer = new Hand([], 0);
192 private holeVisible = false;
193 private dealerBlackjacks = 0;
194 private dealerBusts = 0;

196 private seated(): Player[] { return this.seats.filter((player): player is Player => player !== null); }
197 private find(name: string): Player | undefined { return this.seated().find(player => player.name === name); }
198 private lookup(name: string): Player { return this.find(name) ?? fail(`unknown player ${name}`); }
199 private requireShoe(): Shoe { return this.shoe ?? fail('no shoe'); }
200 private requireTurn(): Turn {
201 if (this.state.kind === 'idle') fail('no round');
202 if (this.state.kind === 'insurance') fail('insurance pending');

205 private handLine(turn: Turn, verb = ''): void {
206 emit(`${turn.player.handName(turn.index)}${verb}: ${turn.hand.text()}`);
207 }
208 private revealDealer(): void {
209 if (!this.holeVisible) {
210 this.requireShoe().running += Shoe.hilo(this.dealer.cards[1]);
211 this.holeVisible = true;
212 }
213 emit(`dealer: ${this.dealer.text()}`);

215 private profit(bet: number, result: Outcome): number {
216 switch (result) {
217 case 'blackjack': return this.rules.BJ === '3:2' ? Math.floor(bet * 3 / 2) : Math.floor(bet * 6 / 5);
218 case 'win': case 'charlie': return bet;
219 case 'push': return 0;
220 case 'surrender': return Math.floor(bet / 2) - bet;
221 case 'lose': case 'bust': return -bet;
222 }
223 }
224 private settle(dealerBlackjack: boolean): void {
225 const dealer = this.dealer.total().value;
226 for (const player of this.participants) {
227 player.hands.forEach((hand, index) => {
228 const result = hand.outcome(dealer, dealerBlackjack);
229 const profit = this.profit(hand.bet, result);
230 player.bankroll += hand.bet + profit;
231 player.net += profit;
232 player.stats.record(result);
233 emit(player.handName(index), result, signed(profit));
234 });
235 emit(player.name, 'bankroll', player.bankroll);
236 }
237 for (const player of this.seated()) player.bet = player.side = 0;
238 this.state = { kind: 'idle' };
239 }
240 private dealerTurn(): void {
241 this.revealDealer();
242 const required = this.participants.some(player => player.hands.some(hand => hand.finish === 'ordinary' && !hand.blackjack()));
243 let drew = false;
244 if (required) for (;;) {
245 const total = this.dealer.total();
246 if (!(total.value < 17 || total.value === 17 && total.soft && this.rules.DEALER === 'H17')) break;
247 this.dealer.cards.push(this.requireShoe().draw(true));
248 emit(`dealer hits: ${this.dealer.text()}`);
249 drew = true;
250 }
251 if (drew && this.dealer.total().value > 21) this.dealerBusts++;
252 this.settle(false);
253 }
254 private firstUnfinished(): Turn | null {
255 for (const player of this.participants) {
256 const index = player.hands.findIndex(hand => hand.finish === 'pending');
257 if (index >= 0) return { player, index, hand: player.hands[index] };
258 }
259 return null;
260 }
261 private advance(): void {
262 for (;;) {
263 const turn = this.firstUnfinished();
264 if (turn === null) { this.dealerTurn(); return; }
265 if (turn.hand.cards.length === 1) {
266 turn.hand.cards.push(this.requireShoe().draw(true));
267 this.handLine(turn);
268 turn.hand.finishCheck(this.rules, turn.player.hands.length);
269 if (turn.hand.finish !== 'pending') continue;
270 }
271 this.state = { kind: 'play', turn };
272 return;
273 }
274 }
275 private peek(): void {
276 const natural = this.dealer.blackjack();
277 emit('dealer peeks:', natural ? 'blackjack' : 'no blackjack');
278 for (const player of this.participants) {
279 if (player.insurance === 0) continue;
280 const profit = natural ? player.insurance * 2 : -player.insurance;
281 player.bankroll += player.insurance + profit;
282 player.net += profit;
283 emit(player.name, 'insurance', signed(profit));
284 }
285 if (natural) {
286 this.dealerBlackjacks++;
287 this.revealDealer();
288 this.settle(true);
289 } else this.advance();
290 }
291 private deal(): void {
292 const shoe = this.requireShoe();
293 if (shoe.drawn >= shoe.cut(this.rules.PEN)) { shoe.shuffle(); emit('shuffle'); }
294 this.round++;
295 emit('round', this.round);
296 this.participants = this.seated().filter(player => player.bet !== 0);
297 for (const player of this.participants) {
298 player.bankroll -= player.bet + player.side;
299 player.insurance = 0;
300 player.stats.rounds++;
301 player.hands = [new Hand([], player.bet)];
302 }
303 for (const player of this.participants) player.hands[0].cards.push(shoe.draw(true));
304 this.dealer = new Hand([shoe.draw(true)], 0);
305 for (const player of this.participants) {
306 const hand = player.hands[0];
307 hand.cards.push(shoe.draw(true));
308 if (hand.blackjack()) hand.finish = 'ordinary';
309 }
310 this.dealer.cards.push(shoe.draw(false));
311 this.holeVisible = false;
312 for (const player of this.participants) this.handLine({ player, index: 0, hand: player.hands[0] });
313 emit('dealer:', cardText(this.dealer.cards[0]), '??');
314 for (const player of this.participants) {
315 if (player.side === 0) continue;
316 const [first, second] = player.hands[0].cards;
317 const sameColor = [0, 3].includes(Math.floor(first / 13)) === [0, 3].includes(Math.floor(second / 13));
318 const multiplier = first % 13 !== second % 13 ? -1 : first === second ? 25 : sameColor ? 12 : 6;
319 const result = multiplier === 25 ? 'perfect' : multiplier === 12 ? 'colored' : multiplier === 6 ? 'mixed' : 'lose';
320 const profit = player.side * multiplier;
321 player.bankroll += player.side + profit;
322 player.net += profit;
323 emit(player.name, 'side', result, signed(profit));
324 }
325 const up = point(this.dealer.cards[0]);
326 if (up === 1) { this.state = { kind: 'insurance', next: 0 }; emit('insurance offered'); }
327 else if (up === 10) this.peek();
328 else this.advance();
329 }
330 private actionProblem(turn: Turn, action: Action): string | null {
331 const { player, hand } = turn;
332 if (action === 'HIT') return hand.splitAces ? 'split aces' : null;
333 if (action === 'DOUBLE') {
334 if (hand.splitAces) return 'split aces';
335 if (hand.cards.length !== 2) return 'two cards only';
336 const total = hand.total();
337 const low = this.rules.DOUBLE === '9-11' ? 9 : 10;
338 const limited = this.rules.DOUBLE !== 'ANY' && (total.soft || total.value < low || total.value > 11);
339 if (limited || !this.rules.DAS && hand.split) return 'double not allowed';
340 if (player.bankroll < hand.bet) return 'insufficient funds';
341 } else if (action === 'SPLIT') {
342 if (hand.cards.length !== 2) return 'two cards only';
343 if (point(hand.cards[0]) !== point(hand.cards[1])) return 'not a pair';
344 if (player.hands.length >= this.rules.HANDS) return 'hand limit';
345 if (player.bankroll < hand.bet) return 'insufficient funds';
346 } else if (action === 'SURRENDER') {
347 if (!this.rules.SURRENDER || hand.split) return 'surrender not allowed';
348 if (hand.cards.length !== 2) return 'two cards only';
349 }
350 return null;
351 }
352 private recommend(turn: Turn): Action {
353 const hand = turn.hand;
354 const canSplit = this.actionProblem(turn, 'SPLIT') === null;
355 if (hand.splitAces) return canSplit ? 'SPLIT' : 'STAND';
356 const upcard = point(this.dealer.cards[0]);
357 const up = upcard === 1 ? 11 : upcard;
358 if (canSplit) {
359 const pair = point(hand.cards[0]);
360 const preferred = [1, 8].includes(pair) || pair === 9 && ![7, 10, 11].includes(up)
361 || [2, 3, 7].includes(pair) && up <= 7 || pair === 6 && up <= 6 || pair === 4 && [5, 6].includes(up);
362 if (preferred) return 'SPLIT';
363 }
364 const total = hand.total();
365 const value = total.value;
366 const canDouble = this.actionProblem(turn, 'DOUBLE') === null;
367 if (total.soft) {
368 if (value >= 19) return 'STAND';
369 if (value === 18) {
370 if (up >= 3 && up <= 6) return canDouble ? 'DOUBLE' : 'STAND';
371 return [2, 7, 8].includes(up) ? 'STAND' : 'HIT';
372 }
373 const wantsDouble = value === 17 && up >= 3 && up <= 6 || [15, 16].includes(value) && up >= 4 && up <= 6
374 || [13, 14].includes(value) && [5, 6].includes(up);
375 return wantsDouble && canDouble ? 'DOUBLE' : 'HIT';
376 }
377 if (value >= 17) return 'STAND';
378 if (value >= 13) {
379 if (up <= 6) return 'STAND';
380 if ((value === 16 && up >= 9 || value === 15 && up === 10) && this.actionProblem(turn, 'SURRENDER') === null) return 'SURRENDER';
381 return 'HIT';
382 }
383 if (value === 12) return up >= 4 && up <= 6 ? 'STAND' : 'HIT';
384 const wantsDouble = value === 11 && up <= 10 || value === 10 && up <= 9 || value === 9 && up >= 3 && up <= 6;
385 return wantsDouble && canDouble ? 'DOUBLE' : 'HIT';
386 }
387 private perform(turn: Turn, action: Action): void {
388 const { player, hand, index } = turn;
389 if (action === 'SPLIT') {
390 player.bankroll -= hand.bet;
391 player.stats.splits++;
392 emit(player.handName(index), 'splits', cardText(hand.cards[0]), cardText(hand.cards[1]));
393 const aces = hand.splitAces || point(hand.cards[0]) === 1;
394 const first = new Hand([hand.cards[0]], hand.bet, true, aces);
395 const second = new Hand([hand.cards[1]], hand.bet, true, aces);
396 player.hands.splice(index, 1, first, second);
397 first.cards.push(this.requireShoe().draw(true));
398 this.handLine({ player, index, hand: first });
399 first.finishCheck(this.rules, player.hands.length);
400 } else if (action === 'STAND' || action === 'SURRENDER') {
401 hand.finish = action === 'STAND' ? 'ordinary' : 'surrender';
402 this.handLine(turn, action === 'STAND' ? ' stands' : ' surrenders');
403 } else {
404 if (action === 'DOUBLE') {
405 player.bankroll -= hand.bet;
406 hand.bet *= 2;
407 player.stats.doubles++;
408 }
409 hand.cards.push(this.requireShoe().draw(true));
410 this.handLine(turn, action === 'DOUBLE' ? ' doubles' : ' hits');
411 hand.finishCheck(this.rules, player.hands.length);
412 if (action === 'DOUBLE' && hand.finish === 'pending') hand.finish = 'ordinary';
413 }
414 this.advance();
415 }
416 private status(): void {
417 switch (this.state.kind) {
418 case 'idle': emit('table idle round', this.round); break;
419 case 'insurance': emit('table insurance round', this.round, 'next', this.participants[this.state.next].name); break;
420 case 'play': emit('table play round', this.round, 'turn', this.state.turn.player.handName(this.state.turn.index)); break;
421 }
422 if (this.state.kind !== 'idle') emit('dealer:', cardText(this.dealer.cards[0]), '??');
423 for (const player of this.seated()) {
424 emit('seat', player.seat + 1, player.name, 'bankroll', player.bankroll, 'bet', player.bet, 'side', player.side);
425 if (this.state.kind !== 'idle' && this.participants.includes(player)) {
426 player.hands.forEach((hand, index) => emit(`${player.handName(index)}: ${hand.text()} bet ${hand.bet}`));
427 }
428 }

431 private parse(words: string[]): Command {
432 const op = words[0];
433 switch (op) {
434 case 'SHOE': {
435 const decks = integerWord(words[1], 1, 8);
436 const seed = BigInt(integerWord(words[2], 1, 2147483646));
437 return { kind: op, decks, seed };
438 }
439 case 'RULE': {
440 const name = words[1];
441 if (!isRuleName(name)) fail(`unknown rule ${name}`);
442 return { kind: op, name, rules: changeRule(this.rules, name, words[2]) };
443 }
444 case 'PLAYER': {
445 const name = words[1];
446 if (!/^[A-Z]{1,8}$/.test(name)) fail(`bad name ${name}`);
447 const bankroll = integerWord(words[2], 1, 1000000);
448 return { kind: op, name, bankroll };
449 }
450 case 'LEAVE': case 'AUTO': return { kind: op, player: this.lookup(words[1]) };
451 case 'REBUY': case 'BET': case 'SIDE': case 'INSURANCE': {
452 const player = this.lookup(words[1]);
453 const amount = integerWord(words[2], op === 'REBUY' ? 1 : 0, 1000000);
454 return { kind: op, player, amount };
455 }
456 case 'HIT': case 'STAND': case 'DOUBLE': case 'SPLIT': case 'SURRENDER':
457 return { kind: 'action', action: op, player: this.lookup(words[1]) };
458 case 'RULES': case 'DEAL': case 'HINT': case 'STATUS': case 'COUNT': case 'STATS': return { kind: op };
459 default: return fail(`unknown command ${op}`);
460 }
461 }
462 execute(words: string[]): void {
463 const command = this.parse(words);
464 if (IDLE_COMMANDS.has(command.kind) && this.state.kind !== 'idle') fail('round in progress');
465 if (command.kind === 'INSURANCE' && this.state.kind !== 'insurance') fail('no insurance');
466 if (command.kind === 'action' || command.kind === 'AUTO') {
467 const turn = this.requireTurn();
468 if (turn.player !== command.player) fail(`waiting for ${turn.player.handName(turn.index)}`);
469 if (command.kind === 'action') {
470 const problem = this.actionProblem(turn, command.action);
471 if (problem !== null) fail(problem);
472 this.perform(turn, command.action);
473 } else {
474 while (this.state.kind === 'play' && this.state.turn.player === command.player) {
475 const current = this.state.turn;
476 this.perform(current, this.recommend(current));
477 }
478 }
479 return;
480 }
481 switch (command.kind) {
482 case 'SHOE':
483 this.shoe = new Shoe(command.decks, command.seed);
484 emit('shoe', command.decks, 'decks', this.shoe.cards.length, 'cards cut', this.shoe.cut(this.rules.PEN));
485 break;
486 case 'RULE':
487 if (command.rules.MIN > command.rules.MAX) fail('min above max');
488 this.rules = command.rules;
489 emit('rule', command.name, ruleText(this.rules[command.name]));
490 break;
491 case 'RULES': emit('rules', RULE_NAMES.map(name => `${name} ${ruleText(this.rules[name])}`).join(' ')); break;
492 case 'PLAYER': {
493 if (this.find(command.name)) fail('player exists');
494 const seat = this.seats.indexOf(null);
495 if (seat < 0) fail('table full');
496 this.seats[seat] = new Player(command.name, seat, command.bankroll);
497 emit(command.name, 'sits at seat', seat + 1, 'with', command.bankroll);
498 break;
499 }
500 case 'LEAVE':
501 emit(command.player.name, 'leaves with', command.player.bankroll, 'net', signed(command.player.net));
502 this.seats[command.player.seat] = null;
503 break;
504 case 'REBUY':
505 command.player.bankroll += command.amount;
506 emit(command.player.name, 'rebuys', command.amount, 'bankroll', command.player.bankroll);
507 break;
508 case 'BET': {
509 const { player, amount } = command;
510 if (amount !== 0) {
511 if (amount < this.rules.MIN || amount > this.rules.MAX) fail('bet out of limits');
512 if (amount + player.side > player.bankroll) fail('insufficient funds');
513 }
514 player.bet = amount;
515 if (amount === 0) { player.side = 0; emit(player.name, 'bet cleared'); }
516 else emit(player.name, 'bets', amount);
517 break;
518 }
519 case 'SIDE': {
520 const { player, amount } = command;
521 if (amount !== 0) {
522 if (player.bet === 0) fail('no bet');
523 if (amount > this.rules.SIDEMAX) fail('side out of limits');
524 if (player.bet + amount > player.bankroll) fail('insufficient funds');
525 }
526 player.side = amount;
527 if (amount === 0) emit(player.name, 'side cleared');
528 else emit(player.name, 'side', amount);
529 break;
530 }
531 case 'DEAL':
532 this.requireShoe();
533 if (!this.seated().some(player => player.bet !== 0)) fail('no bets');
534 this.deal();
535 break;
536 case 'INSURANCE': {
537 const state = this.state as Extract<TableState, { kind: 'insurance' }>;
538 const { player, amount } = command;
539 const waiting = this.participants[state.next];
540 if (player !== waiting) fail(`waiting for ${waiting.name}`);
541 if (amount > Math.floor(player.bet / 2)) fail('insurance out of limits');
542 if (amount > player.bankroll) fail('insufficient funds');
543 player.bankroll -= amount;
544 player.insurance = amount;
545 if (amount === 0) emit(player.name, 'declines insurance');
546 else emit(player.name, 'insures', amount);
547 state.next++;
548 if (state.next === this.participants.length) this.peek();
549 break;
550 }
551 case 'HINT': {
552 const turn = this.requireTurn();
553 emit(`hint ${turn.player.handName(turn.index)}: ${this.recommend(turn).toLowerCase()}`);
554 break;
555 }
556 case 'STATUS': this.status(); break;
557 case 'COUNT': {
558 const shoe = this.requireShoe();
559 const remaining = shoe.cards.length - shoe.drawn;
560 const scaled = Math.floor(Math.abs(shoe.running) * 5200 / remaining);
561 const sign = shoe.running < 0 ? '-' : '';
562 const text = `${sign}${Math.floor(scaled / 100)}.${String(scaled % 100).padStart(2, '0')}`;
563 emit('count running', shoe.running, 'true', text, 'remaining', remaining);
564 break;
565 }
566 case 'STATS': {
567 const players = this.seated();
568 if (players.length === 0) fail('no players');
569 for (const player of players) emit('stats', player.name, player.stats.text(), 'net', signed(player.net));
570 emit('stats dealer rounds', this.round, 'blackjacks', this.dealerBlackjacks, 'busts', this.dealerBusts);
571 break;
572 }
573 }
574 }
575}

578const table = new Table();
579for (const line of input.slice(1, Number(input[0]) + 1)) {

581 catch (problem) {

583 emit('error:', problem.message);
