14function integerWord(string $word, int $low, int $high): int
15{
16 if (preg_match('/\A(?:0|[1-9][0-9]{0,9})\z/', $word) !== 1) throw new CommandError("bad value $word");
17 $value = (int)$word;
18 if ($value < $low || $value > $high) throw new CommandError("bad value $word");
19 return $value;
20}
21function choiceWord(string $word, array $choices): string
22{
23 if (!in_array($word, $choices, true)) throw new CommandError("bad value $word");
24 return $word;
25}
26function signed(int $value): string { return ($value >= 0 ? '+' : '') . $value; }
27function switchText(bool $value): string { return $value ? 'ON' : 'OFF'; }

29final class Rules
30{
31 public string $dealer = 'S17', $blackjack = '3:2', $doubling = 'ANY';
32 public int $penetration = 75, $minimum = 10, $maximum = 500, $hands = 4, $charlie = 0, $sideMaximum = 25;
33 public bool $doubleAfterSplit = true, $resplitAces = false, $surrender = true;
34
35 public function changed(string $name, string $word): self
36 {
37 $next = clone $this;
38 switch ($name) {
39 case 'DEALER': $next->dealer = choiceWord($word, ['S17', 'H17']); break;
40 case 'PEN': $next->penetration = integerWord($word, 50, 90); break;
41 case 'MIN': $next->minimum = integerWord($word, 1, 10000); break;
42 case 'MAX': $next->maximum = integerWord($word, 1, 10000); break;
43 case 'BJ': $next->blackjack = choiceWord($word, ['3:2', '6:5']); break;
44 case 'DAS': $next->doubleAfterSplit = choiceWord($word, ['ON', 'OFF']) === 'ON'; break;
45 case 'DOUBLE': $next->doubling = choiceWord($word, ['ANY', '9-11', '10-11']); break;
46 case 'HANDS': $next->hands = integerWord($word, 1, 4); break;
47 case 'RSA': $next->resplitAces = choiceWord($word, ['ON', 'OFF']) === 'ON'; break;
48 case 'SURRENDER': $next->surrender = choiceWord($word, ['ON', 'OFF']) === 'ON'; break;
49 case 'CHARLIE':
50 $next->charlie = integerWord($word, 0, 7);
51 if (!in_array($next->charlie, [0, 5, 6, 7], true)) throw new CommandError("bad value $word");
52 break;
53 case 'SIDEMAX': $next->sideMaximum = integerWord($word, 0, 1000); break;
54 default: throw new CommandError("unknown rule $name");
55 }
56 return $next;
57 }
58 public function fields(): array
59 {
60 return [
61 'DEALER' => $this->dealer, 'PEN' => $this->penetration, 'MIN' => $this->minimum,
62 'MAX' => $this->maximum, 'BJ' => $this->blackjack, 'DAS' => switchText($this->doubleAfterSplit),
63 'DOUBLE' => $this->doubling, 'HANDS' => $this->hands, 'RSA' => switchText($this->resplitAces),
64 'SURRENDER' => switchText($this->surrender), 'CHARLIE' => $this->charlie, 'SIDEMAX' => $this->sideMaximum,
65 ];
66 }

70{
71 public function __construct(public int $code) {}
72 public function rank(): int { return $this->code % 13; }
73 public function point(): int { return min(10, $this->rank() + 1); }
74 public function black(): bool { return in_array(intdiv($this->code, 13), [0, 3], true); }
75 public function count(): int

77 $point = $this->point();
78 return $point >= 2 && $point <= 6 ? 1 : ($point >= 7 && $point <= 9 ? 0 : -1);

86final class Hand
87{
88 public ?Finish $finish = null;
89 public function __construct(public array $cards = [], public int $bet = 0, public bool $split = false, public bool $splitAces = false) {}
90 public function total(): Total
91 {
92 $sum = 0;
93 $ace = false;
94 foreach ($this->cards as $card) {
95 $sum += $card->point();
96 $ace = $ace || $card->rank() === 0;
97 }
98 $soft = $ace && $sum + 10 <= 21;
99 return new Total($sum + ($soft ? 10 : 0), $soft);
100 }
101 public function blackjack(): bool { return !$this->split && count($this->cards) === 2 && $this->total()->value === 21; }
102 public function __toString(): string
103 {
104 $total = $this->total();
105 $status = $this->blackjack() ? 'blackjack' : ($total->value > 21 ? "bust $total->value" : ($total->soft ? "soft $total->value" : (string)$total->value));
106 return implode(' ', $this->cards) . " ($status)";
107 }
108 public function check(Rules $rules, int $handCount): void
109 {
110 $total = $this->total()->value;
111 if ($total > 21) $this->finish = Finish::Bust;
112 elseif ($rules->charlie !== 0 && count($this->cards) === $rules->charlie) $this->finish = Finish::Charlie;
113 elseif ($total === 21) $this->finish = Finish::Ordinary;
114 elseif ($this->splitAces && count($this->cards) === 2) {
115 $again = $this->cards[0]->rank() === 0 && $this->cards[1]->rank() === 0 && $rules->resplitAces && $handCount < $rules->hands;
116 if (!$again) $this->finish = Finish::Ordinary;
117 }
118 }

129final class Table
130{
131 private Rules $rules;
132 private array $players;
133 private Phase $phase = Phase::Idle;
134 private int $round = 0, $decks = 0, $random = 1, $drawn = 0, $running = 0;
135 private array $shoe = [], $participants = [], $output = [];
136 private int $insuranceTurn = 0, $currentSeat = -1, $currentHand = 0;
137 private Hand $dealer;
138 private bool $holeVisible = false, $dealerNatural = false;
139 private int $dealerBlackjacks = 0, $dealerBusts = 0;
140
141 public function __construct()
142 {
143 $this->rules = new Rules();
144 $this->players = array_fill(0, 7, null);
145 $this->dealer = new Hand();
146 }
147 private function say(string $line): void { $this->output[] = $line; }
148 private function findPlayer(string $name): int
149 {
150 foreach ($this->players as $seat => $player) if ($player !== null && $player->name === $name) return $seat;
151 return -1;
152 }
153 private function lookup(string $name): int
154 {
155 $seat = $this->findPlayer($name);
156 if ($seat < 0) throw new CommandError("unknown player $name");
157 return $seat;
158 }
159 private function player(int $seat): Player { return $this->players[$seat]; }
160 private function current(): Hand { return $this->player($this->currentSeat)->hands[$this->currentHand]; }
161 private function currentName(): string { return $this->player($this->currentSeat)->handName($this->currentHand); }
162 private function cut(): int { return intdiv(count($this->shoe) * $this->rules->penetration, 100); }
163 private function shuffle(): void
164 {
165 $this->shoe = [];
166 for ($i = 0; $i < 52 * $this->decks; $i++) $this->shoe[] = new Card($i % 52);
167 for ($i = count($this->shoe) - 1; $i > 0; $i--) {
168 $this->random = 48271 * $this->random % 2147483647;
169 $j = $this->random % ($i + 1);
170 [$this->shoe[$i], $this->shoe[$j]] = [$this->shoe[$j], $this->shoe[$i]];
171 }
172 $this->drawn = $this->running = 0;
173 }
174 private function draw(bool $visible = true): Card
175 {
176 $card = $this->shoe[$this->drawn++];
177 if ($visible) $this->running += $card->count();
178 return $card;
179 }
180 private function revealDealer(): void
181 {
182 if (!$this->holeVisible) {
183 $this->running += $this->dealer->cards[1]->count();
184 $this->holeVisible = true;
185 }
186 $this->say('dealer: ' . $this->dealer);
187 }
188 private function result(Hand $hand): Outcome
189 {
190 if ($this->dealerNatural) return $hand->blackjack() ? Outcome::Push : Outcome::Lose;
191 if ($hand->blackjack()) return Outcome::Blackjack;
192 if ($hand->finish === Finish::Bust) return Outcome::Bust;
193 if ($hand->finish === Finish::Surrender) return Outcome::Surrender;
194 if ($hand->finish === Finish::Charlie) return Outcome::Charlie;
195 $own = $hand->total()->value;
196 $dealer = $this->dealer->total()->value;
197 return $dealer > 21 || $own > $dealer ? Outcome::Win : ($own === $dealer ? Outcome::Push : Outcome::Lose);
198 }
199 private function profit(int $bet, Outcome $outcome): int
200 {
201 return match ($outcome) {
202 Outcome::Blackjack => $this->rules->blackjack === '3:2' ? intdiv($bet * 3, 2) : intdiv($bet * 6, 5),
203 Outcome::Win, Outcome::Charlie => $bet,
204 Outcome::Push => 0,
205 Outcome::Surrender => intdiv($bet, 2) - $bet,
206 Outcome::Lose, Outcome::Bust => -$bet,
207 };
208 }
209 private function settle(): void
210 {
211 foreach ($this->participants as $seat) {
212 $player = $this->player($seat);
213 foreach ($player->hands as $index => $hand) {
214 $outcome = $this->result($hand);
215 $profit = $this->profit($hand->bet, $outcome);
216 $player->bankroll += $hand->bet + $profit;
217 $player->net += $profit;
218 $player->stats['hands']++;
219 $field = match ($outcome) {
220 Outcome::Blackjack, Outcome::Win, Outcome::Charlie => 'wins',
221 Outcome::Push => 'pushes',
222 default => 'losses',
223 };
224 $player->stats[$field]++;
225 $special = match ($outcome) {
226 Outcome::Blackjack => 'blackjacks', Outcome::Bust => 'busts', Outcome::Surrender => 'surrenders', default => null,
227 };
228 if ($special !== null) $player->stats[$special]++;
229 $this->say($player->handName($index) . " {$outcome->value} " . signed($profit));
230 }
231 $this->say("$player->name bankroll $player->bankroll");
232 }
233 foreach ($this->players as $player) if ($player !== null) $player->bet = $player->side = 0;
234 $this->phase = Phase::Idle;
235 }
236 private function dealerTurn(): void
237 {
238 $this->revealDealer();
239 $required = false;
240 foreach ($this->participants as $seat) {
241 foreach ($this->player($seat)->hands as $hand) {
242 if ($hand->finish === Finish::Ordinary && !$hand->blackjack()) $required = true;
243 }
244 }
245 $drew = false;
246 if ($required) {
247 while (true) {
248 $total = $this->dealer->total();
249 if ($total->value > 16 && !($total->value === 17 && $total->soft && $this->rules->dealer === 'H17')) break;
250 $this->dealer->cards[] = $this->draw();
251 $this->say('dealer hits: ' . $this->dealer);
252 $drew = true;
253 }
254 }
255 if ($drew && $this->dealer->total()->value > 21) $this->dealerBusts++;
256 $this->settle();
257 }
258 private function advanceHands(): void
259 {
260 while (true) {
261 $next = null;
262 foreach ($this->participants as $seat) {
263 foreach ($this->player($seat)->hands as $index => $hand) {
264 if ($hand->finish === null) { $next = [$seat, $index]; break 2; }
265 }
266 }
267 if ($next === null) { $this->dealerTurn(); return; }
268 [$this->currentSeat, $this->currentHand] = $next;
269 $hand = $this->current();
270 if (count($hand->cards) !== 1) return;
271 $hand->cards[] = $this->draw();
272 $this->say($this->currentName() . ': ' . $hand);
273 $hand->check($this->rules, count($this->player($this->currentSeat)->hands));
274 if ($hand->finish === null) return;
275 }
276 }
277 private function peek(): void
278 {
279 $this->dealerNatural = $this->dealer->blackjack();
280 $this->say('dealer peeks: ' . ($this->dealerNatural ? 'blackjack' : 'no blackjack'));
281 foreach ($this->participants as $seat) {
282 $player = $this->player($seat);
283 if ($player->insurance === 0) continue;
284 $profit = $this->dealerNatural ? $player->insurance * 2 : -$player->insurance;
285 $player->bankroll += $player->insurance + $profit;
286 $player->net += $profit;
287 $this->say("$player->name insurance " . signed($profit));
288 }
289 if ($this->dealerNatural) {
290 $this->dealerBlackjacks++;
291 $this->revealDealer();
292 $this->settle();
293 } else {
294 $this->phase = Phase::Play;
295 $this->advanceHands();
296 }
297 }
298 private function deal(): void
299 {
300 if ($this->drawn >= $this->cut()) { $this->shuffle(); $this->say('shuffle'); }
301 $this->round++;
302 $this->say("round $this->round");
303 $this->participants = [];
304 foreach ($this->players as $seat => $player) {
305 if ($player === null || $player->bet === 0) continue;
306 $this->participants[] = $seat;
307 $player->bankroll -= $player->bet + $player->side;
308 $player->insurance = 0;
309 $player->stats['rounds']++;
310 $player->hands = [new Hand(bet: $player->bet)];
311 }
312 foreach ($this->participants as $seat) $this->player($seat)->hands[0]->cards[] = $this->draw();
313 $this->dealer = new Hand([$this->draw()]);
314 foreach ($this->participants as $seat) {
315 $hand = $this->player($seat)->hands[0];
316 $hand->cards[] = $this->draw();
317 if ($hand->blackjack()) $hand->finish = Finish::Ordinary;
318 }
319 $this->dealer->cards[] = $this->draw(false);
320 $this->holeVisible = $this->dealerNatural = false;
321 foreach ($this->participants as $seat) {
322 $player = $this->player($seat);
323 $this->say($player->handName(0) . ': ' . $player->hands[0]);
324 }
325 $this->say('dealer: ' . $this->dealer->cards[0] . ' ??');
326 foreach ($this->participants as $seat) {
327 $player = $this->player($seat);
328 if ($player->side === 0) continue;
329 [$first, $second] = $player->hands[0]->cards;
330 [$result, $multiplier] = match (true) {
331 $first->rank() !== $second->rank() => ['lose', -1],
332 $first->code === $second->code => ['perfect', 25],
333 $first->black() === $second->black() => ['colored', 12],
334 default => ['mixed', 6],
335 };
336 $profit = $player->side * $multiplier;
337 $player->bankroll += $player->side + $profit;
338 $player->net += $profit;
339 $this->say("$player->name side $result " . signed($profit));
340 }
341 $up = $this->dealer->cards[0]->point();
342 if ($up === 1) {
343 $this->phase = Phase::Insurance;
344 $this->insuranceTurn = 0;
345 $this->say('insurance offered');
346 } elseif ($up === 10) $this->peek();
347 else { $this->phase = Phase::Play; $this->advanceHands(); }
348 }
349 private function actionProblem(Action $action): ?string
350 {
351 $player = $this->player($this->currentSeat);
352 $hand = $this->current();
353 if ($action === Action::Hit) return $hand->splitAces ? 'split aces' : null;
354 if ($action === Action::DoubleDown) {
355 if ($hand->splitAces) return 'split aces';
356 if (count($hand->cards) !== 2) return 'two cards only';
357 $total = $hand->total();
358 $limited = $this->rules->doubling !== 'ANY' && ($total->soft || $total->value > 11 || $total->value < ($this->rules->doubling === '9-11' ? 9 : 10));
359 if ($limited || (!$this->rules->doubleAfterSplit && $hand->split)) return 'double not allowed';
360 if ($player->bankroll < $hand->bet) return 'insufficient funds';
361 } elseif ($action === Action::Split) {
362 if (count($hand->cards) !== 2) return 'two cards only';
363 if ($hand->cards[0]->point() !== $hand->cards[1]->point()) return 'not a pair';
364 if (count($player->hands) >= $this->rules->hands) return 'hand limit';
365 if ($player->bankroll < $hand->bet) return 'insufficient funds';
366 } elseif ($action === Action::Surrender) {
367 if (!$this->rules->surrender || $hand->split) return 'surrender not allowed';
368 if (count($hand->cards) !== 2) return 'two cards only';
369 }
370 return null;
371 }
372 private function recommendation(): Action
373 {
374 $hand = $this->current();
375 $canSplit = $this->actionProblem(Action::Split) === null;
376 if ($hand->splitAces) return $canSplit ? Action::Split : Action::Stand;
377 $up = $this->dealer->cards[0]->point();
378 if ($up === 1) $up = 11;
379 if ($canSplit) {
380 $point = $hand->cards[0]->point();
381 $wanted = in_array($point, [1, 8], true)
382 || ($point === 9 && !in_array($up, [7, 10, 11], true))
383 || (in_array($point, [2, 3, 7], true) && $up <= 7)
384 || ($point === 6 && $up <= 6) || ($point === 4 && in_array($up, [5, 6], true));
385 if ($wanted) return Action::Split;
386 }
387 $total = $hand->total();
388 $value = $total->value;
389 $canDouble = $this->actionProblem(Action::DoubleDown) === null;
390 if ($total->soft) {
391 if ($value >= 19) return Action::Stand;
392 if ($value === 18) {
393 if ($up >= 3 && $up <= 6) return $canDouble ? Action::DoubleDown : Action::Stand;
394 return in_array($up, [2, 7, 8], true) ? Action::Stand : Action::Hit;
395 }
396 $wanted = ($value === 17 && $up >= 3 && $up <= 6)
397 || (in_array($value, [15, 16], true) && $up >= 4 && $up <= 6)
398 || (in_array($value, [13, 14], true) && in_array($up, [5, 6], true));
399 return $wanted && $canDouble ? Action::DoubleDown : Action::Hit;
400 }
401 if ($value >= 17) return Action::Stand;
402 if ($value >= 13) {
403 if ($up <= 6) return Action::Stand;
404 if ((($value === 16 && $up >= 9) || ($value === 15 && $up === 10)) && $this->actionProblem(Action::Surrender) === null) return Action::Surrender;
405 return Action::Hit;
406 }
407 if ($value === 12) return $up >= 4 && $up <= 6 ? Action::Stand : Action::Hit;
408 $wanted = ($value === 11 && $up <= 10) || ($value === 10 && $up <= 9) || ($value === 9 && $up >= 3 && $up <= 6);
409 return $wanted && $canDouble ? Action::DoubleDown : Action::Hit;
410 }
411 private function act(Action $action): void
412 {
413 $player = $this->player($this->currentSeat);
414 $index = $this->currentHand;
415 $hand = $this->current();
416 $name = $player->handName($index);
417 if ($action === Action::Split) {
418 $player->bankroll -= $hand->bet;
419 $player->stats['splits']++;
420 $this->say($name . ' splits ' . implode(' ', $hand->cards));
421 $aces = $hand->splitAces || $hand->cards[0]->point() === 1;
422 $first = new Hand([$hand->cards[0]], $hand->bet, true, $aces);
423 $second = new Hand([$hand->cards[1]], $hand->bet, true, $aces);
424 array_splice($player->hands, $index, 1, [$first, $second]);
425 $first->cards[] = $this->draw();
426 $this->say($name . ': ' . $first);
427 $first->check($this->rules, count($player->hands));
428 } elseif ($action === Action::Stand || $action === Action::Surrender) {
429 $hand->finish = $action === Action::Stand ? Finish::Ordinary : Finish::Surrender;
430 $verb = $action === Action::Stand ? 'stands' : 'surrenders';
431 $this->say("$name $verb: " . $hand);
432 } else {
433 if ($action === Action::DoubleDown) {
434 $player->bankroll -= $hand->bet;
435 $hand->bet *= 2;
436 $player->stats['doubles']++;
437 }
438 $hand->cards[] = $this->draw();
439 $verb = $action === Action::DoubleDown ? 'doubles' : 'hits';
440 $this->say("$name $verb: " . $hand);
441 $hand->check($this->rules, count($player->hands));
442 if ($action === Action::DoubleDown && $hand->finish === null) $hand->finish = Finish::Ordinary;
443 }
444 $this->advanceHands();
445 }
446 private function status(): void
447 {
448 $line = "table {$this->phase->value} round $this->round";
449 if ($this->phase === Phase::Insurance) $line .= ' next ' . $this->player($this->participants[$this->insuranceTurn])->name;
450 elseif ($this->phase === Phase::Play) $line .= ' turn ' . $this->currentName();
451 $this->say($line);
452 if ($this->phase !== Phase::Idle) $this->say('dealer: ' . $this->dealer->cards[0] . ' ??');
453 foreach ($this->players as $seat => $player) {
454 if ($player === null) continue;
455 $this->say('seat ' . ($seat + 1) . " $player->name bankroll $player->bankroll bet $player->bet side $player->side");
456 if ($this->phase !== Phase::Idle && in_array($seat, $this->participants, true)) {
457 foreach ($player->hands as $index => $hand) $this->say($player->handName($index) . ': ' . $hand . " bet $hand->bet");
458 }
459 }
460 }
461 private function statistics(): void
462 {
463 foreach ($this->players as $player) {
464 if ($player === null) continue;
465 $line = "stats $player->name";
466 foreach ($player->stats as $name => $value) $line .= " $name $value";
467 $this->say($line . ' net ' . signed($player->net));

471 public function execute(array $words): array
472 {
473 $this->output = [];
474 $op = $words[0];
475 if (!in_array($op, COMMANDS, true)) throw new CommandError("unknown command $op");
476 $seat = -1;
477 $amount = $seed = 0;
478 $proposed = null;
479 if ($op === 'SHOE') {
480 $amount = integerWord($words[1], 1, 8);
481 $seed = integerWord($words[2], 1, 2147483646);
482 } elseif ($op === 'RULE') $proposed = $this->rules->changed($words[1], $words[2]);
483 elseif ($op === 'PLAYER') {
484 if (preg_match('/\A[A-Z]{1,8}\z/', $words[1]) !== 1) throw new CommandError("bad name $words[1]");
485 $amount = integerWord($words[2], 1, 1000000);
486 } elseif (in_array($op, PLAYER_COMMANDS, true)) {
487 $seat = $this->lookup($words[1]);
488 if (in_array($op, ['REBUY', 'BET', 'SIDE', 'INSURANCE'], true)) $amount = integerWord($words[2], $op === 'REBUY' ? 1 : 0, 1000000);
489 }
490 if (in_array($op, IDLE_COMMANDS, true) && $this->phase !== Phase::Idle) throw new CommandError('round in progress');
491 if ($op === 'INSURANCE' && $this->phase !== Phase::Insurance) throw new CommandError('no insurance');
492 $action = Action::tryFrom($op);
493 $actionCommand = $action !== null || $op === 'AUTO';
494 if ($actionCommand || $op === 'HINT') {
495 if ($this->phase === Phase::Idle) throw new CommandError('no round');
496 if ($this->phase === Phase::Insurance) throw new CommandError('insurance pending');
497 if ($actionCommand && $seat !== $this->currentSeat) throw new CommandError('waiting for ' . $this->currentName());
498 }
499 if ($actionCommand) {
500 if ($op === 'AUTO') {
501 while ($this->phase === Phase::Play && $this->currentSeat === $seat) $this->act($this->recommendation());
502 } else {
503 $problem = $this->actionProblem($action);
504 if ($problem !== null) throw new CommandError($problem);
505 $this->act($action);
506 }
507 return $this->output;
508 }
509 switch ($op) {
510 case 'SHOE':
511 $this->decks = $amount;
512 $this->random = $seed;
513 $this->shuffle();
514 $this->say("shoe $amount decks " . count($this->shoe) . ' cards cut ' . $this->cut());
515 break;
516 case 'RULE':
517 if ($proposed->minimum > $proposed->maximum) throw new CommandError('min above max');
518 $this->rules = $proposed;
519 $this->say("rule $words[1] " . $this->rules->fields()[$words[1]]);
520 break;
521 case 'RULES':
522 $line = 'rules';
523 foreach ($this->rules->fields() as $name => $value) $line .= " $name $value";
524 $this->say($line);
525 break;
526 case 'PLAYER':
527 if ($this->findPlayer($words[1]) >= 0) throw new CommandError('player exists');
528 $empty = array_search(null, $this->players, true);
529 if ($empty === false) throw new CommandError('table full');
530 $this->players[$empty] = new Player($words[1], $amount);
531 $this->say("$words[1] sits at seat " . ($empty + 1) . " with $amount");
532 break;
533 case 'LEAVE':
534 $player = $this->player($seat);
535 $this->say("$player->name leaves with $player->bankroll net " . signed($player->net));
536 $this->players[$seat] = null;
537 break;
538 case 'REBUY':
539 $player = $this->player($seat);
540 $player->bankroll += $amount;
541 $this->say("$player->name rebuys $amount bankroll $player->bankroll");
542 break;
543 case 'BET':
544 $player = $this->player($seat);
545 if ($amount !== 0) {
546 if ($amount < $this->rules->minimum || $amount > $this->rules->maximum) throw new CommandError('bet out of limits');
547 if ($amount + $player->side > $player->bankroll) throw new CommandError('insufficient funds');
548 }
549 $player->bet = $amount;
550 if ($amount === 0) { $player->side = 0; $this->say("$player->name bet cleared"); }
551 else $this->say("$player->name bets $amount");
552 break;
553 case 'SIDE':
554 $player = $this->player($seat);
555 if ($amount !== 0) {
556 if ($player->bet === 0) throw new CommandError('no bet');
557 if ($amount > $this->rules->sideMaximum) throw new CommandError('side out of limits');
558 if ($player->bet + $amount > $player->bankroll) throw new CommandError('insufficient funds');
559 }
560 $player->side = $amount;
561 $this->say($amount === 0 ? "$player->name side cleared" : "$player->name side $amount");
562 break;
563 case 'DEAL':
564 if ($this->decks === 0) throw new CommandError('no shoe');
565 if (!array_filter($this->players, static fn(?Player $player): bool => $player !== null && $player->bet !== 0)) throw new CommandError('no bets');
566 $this->deal();
567 break;
568 case 'INSURANCE':
569 $waiting = $this->participants[$this->insuranceTurn];
570 if ($seat !== $waiting) throw new CommandError('waiting for ' . $this->player($waiting)->name);
571 $player = $this->player($seat);
572 if ($amount > intdiv($player->bet, 2)) throw new CommandError('insurance out of limits');
573 if ($amount > $player->bankroll) throw new CommandError('insufficient funds');
574 $player->bankroll -= $amount;
575 $player->insurance = $amount;
576 $this->say($amount === 0 ? "$player->name declines insurance" : "$player->name insures $amount");
577 if (++$this->insuranceTurn === count($this->participants)) $this->peek();
578 break;
579 case 'HINT': $this->say('hint ' . $this->currentName() . ': ' . strtolower($this->recommendation()->value)); break;
580 case 'STATUS': $this->status(); break;
581 case 'COUNT':
582 if ($this->decks === 0) throw new CommandError('no shoe');
583 $remaining = count($this->shoe) - $this->drawn;
584 $value = intdiv(abs($this->running) * 5200, $remaining);
585 $text = ($this->running < 0 ? '-' : '') . intdiv($value, 100) . '.' . sprintf('%02d', $value % 100);
586 $this->say("count running $this->running true $text remaining $remaining");
587 break;
588 case 'STATS':
589 if (!array_filter($this->players, static fn(?Player $player): bool => $player !== null)) throw new CommandError('no players');
590 $this->statistics();
591 break;
592 }
593 return $this->output;
594 }
