    public function total(): Total
    {
        $sum = 0;
        $ace = false;
        foreach ($this->cards as $card) {
            $sum += $card->point();
            $ace = $ace || $card->rank() === 0;
        }
        $soft = $ace && $sum + 10 <= 21;
        return new Total($sum + ($soft ? 10 : 0), $soft);
    }

        return implode(' ', $this->cards) . " ($status)";
