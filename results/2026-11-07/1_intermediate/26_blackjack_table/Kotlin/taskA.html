3private class CommandError(message: String) : RuntimeException(message)
4private fun fail(message: String): Nothing = throw CommandError(message)
5private fun format(pattern: String, vararg values: Any?): String = String.format(Locale.ROOT, pattern, *values)

7private fun integer(word: String, minimum: Long, maximum: Long): Long {
8 if (word.isEmpty() || word.length > 10 || word.length > 1 && word[0] == '0') fail("bad value $word")
9 var value = 0L
10 for (digit in word) {
11 if (digit !in '0'..'9') fail("bad value $word")
12 value = value * 10 + (digit - '0')
13 }
14 if (value !in minimum..maximum) fail("bad value $word")
15 return value
16}

18private fun choice(word: String, vararg choices: String): String {
19 if (word !in choices) fail("bad value $word")
20 return word
21}

23private fun on(word: String) = choice(word, "ON", "OFF") == "ON"
24private fun onOff(value: Boolean) = if (value) "ON" else "OFF"

26private enum class State { IDLE, INSURANCE, PLAY }

28private enum class Action {
29 HIT, STAND, DOUBLE, SPLIT, SURRENDER;
30 val text: String get() = name.lowercase(Locale.ROOT)
31}
32private enum class DoubleRule(val text: String) { ANY("ANY"), HARD_9("9-11"), HARD_10("10-11") }
33private enum class Outcome {
34 BLACKJACK, WIN, CHARLIE, PUSH, LOSE, BUST, SURRENDER;
35 val text: String get() = name.lowercase(Locale.ROOT)
36 fun profit(bet: Long, rules: Rules): Long = when (this) {
37 BLACKJACK -> if (rules.sixToFive) bet * 6 / 5 else bet * 3 / 2
38 WIN, CHARLIE -> bet
39 PUSH -> 0L
40 SURRENDER -> bet / 2 - bet
41 LOSE, BUST -> -bet
42 }
43}

45private enum class Command {

47 HIT, STAND, DOUBLE, SPLIT, SURRENDER, HINT, AUTO, STATUS, COUNT, STATS;
48
49 val action: Action? get() = when (this) {
50 HIT -> Action.HIT
51 STAND -> Action.STAND
52 DOUBLE -> Action.DOUBLE
53 SPLIT -> Action.SPLIT
54 SURRENDER -> Action.SURRENDER
55 else -> null
56 }
57 val namedPlayer: Boolean get() = when (this) {
58 LEAVE, REBUY, BET, SIDE, INSURANCE, HIT, STAND, DOUBLE, SPLIT, SURRENDER, AUTO -> true
59 else -> false
60 }
61 val idleOnly: Boolean get() = when (this) {
62 SHOE, RULE, PLAYER, LEAVE, REBUY, BET, SIDE, DEAL -> true
63 else -> false
64 }

70private data class Rules(
71 val hitSoft17: Boolean = false,
72 val penetration: Int = 75,
73 val minimum: Long = 10,
74 val maximum: Long = 500,
75 val sixToFive: Boolean = false,
76 val doubleAfterSplit: Boolean = true,
77 val doubling: DoubleRule = DoubleRule.ANY,
78 val hands: Int = 4,
79 val resplitAces: Boolean = false,
80 val surrender: Boolean = true,
81 val charlie: Int = 0,
82 val sideMaximum: Long = 25,
83) {
84 fun changed(name: String, word: String): Rules = when (name) {
85 "DEALER" -> copy(hitSoft17 = choice(word, "S17", "H17") == "H17")
86 "PEN" -> copy(penetration = integer(word, 50, 90).toInt())
87 "MIN" -> copy(minimum = integer(word, 1, 10000))
88 "MAX" -> copy(maximum = integer(word, 1, 10000))
89 "BJ" -> copy(sixToFive = choice(word, "3:2", "6:5") == "6:5")
90 "DAS" -> copy(doubleAfterSplit = on(word))
91 "DOUBLE" -> copy(doubling = when (choice(word, "ANY", "9-11", "10-11")) {
92 "ANY" -> DoubleRule.ANY
93 "9-11" -> DoubleRule.HARD_9
94 else -> DoubleRule.HARD_10
95 })
96 "HANDS" -> copy(hands = integer(word, 1, 4).toInt())
97 "RSA" -> copy(resplitAces = on(word))
98 "SURRENDER" -> copy(surrender = on(word))
99 "CHARLIE" -> copy(charlie = choice(word, "0", "5", "6", "7").toInt())
100 "SIDEMAX" -> copy(sideMaximum = integer(word, 0, 1000))
101 else -> fail("unknown rule $name")
102 }
103
104 override fun toString(): String = format(
105 "rules DEALER %s PEN %d MIN %d MAX %d BJ %s DAS %s DOUBLE %s HANDS %d RSA %s SURRENDER %s CHARLIE %d SIDEMAX %d",
106 if (hitSoft17) "H17" else "S17", penetration, minimum, maximum, if (sixToFive) "6:5" else "3:2",
107 onOff(doubleAfterSplit), doubling.text, hands, onOff(resplitAces), onOff(surrender), charlie, sideMaximum,
108 )
109}

111private data class Card(val code: Int) {
112 val rank: Int get() = code % 13
113 val suit: Int get() = code / 13
114 val point: Int get() = minOf(rank + 1, 10)
115 val black: Boolean get() = suit == 0 || suit == 3
116 val hilo: Int get() = when (point) { in 2..6 -> 1; 1, 10 -> -1; else -> 0 }
117 override fun toString() = "${"A23456789TJQK"[rank]}${"CDHS"[suit]}"
118}

120private data class Total(val value: Int, val soft: Boolean)
121
122private class Hand(
123 cards: List<Card> = emptyList(),
124 var bet: Long = 0,
125 val split: Boolean = false,
126 val splitAces: Boolean = false,
127) {
128 val cards = cards.toMutableList()
129 var finished = false
130 var reason: Finish? = null
131 val total: Total get() {
132 val sum = cards.sumOf { it.point }
133 val soft = cards.any { it.rank == 0 } && sum + 10 <= 21
134 return Total(sum + if (soft) 10 else 0, soft)
135 }
136 val natural: Boolean get() = !split && cards.size == 2 && total.value == 21
137
138 override fun toString(): String {
139 val total = total
140 val status = when {
141 natural -> "blackjack"
142 total.value > 21 -> "bust ${total.value}"
143 total.soft -> "soft ${total.value}"
144 else -> total.value.toString()
145 }
146 return "${cards.joinToString(" ")} ($status)"
147 }
148
149 fun outcome(dealer: Int, dealerNatural: Boolean): Outcome {
150 if (dealerNatural) return if (natural) Outcome.PUSH else Outcome.LOSE
151 if (natural) return Outcome.BLACKJACK
152 return when (reason) {
153 Finish.BUST -> Outcome.BUST
154 Finish.SURRENDER -> Outcome.SURRENDER
155 Finish.CHARLIE -> Outcome.CHARLIE
156 null -> when {
157 dealer > 21 || total.value > dealer -> Outcome.WIN
158 total.value == dealer -> Outcome.PUSH
159 else -> Outcome.LOSE
160 }
161 }
162 }
163}

165private class Statistics {
166 var rounds = 0
167 var hands = 0
168 var wins = 0
169 var losses = 0
170 var pushes = 0
171 var blackjacks = 0
172 var busts = 0
173 var surrenders = 0
174 var doubles = 0
175 var splits = 0
176
177 fun record(result: Outcome) {
178 hands++
179 when (result) {
180 Outcome.BLACKJACK, Outcome.WIN, Outcome.CHARLIE -> wins++
181 Outcome.PUSH -> pushes++
182 else -> losses++
183 }
184 when (result) {
185 Outcome.BLACKJACK -> blackjacks++
186 Outcome.BUST -> busts++
187 Outcome.SURRENDER -> surrenders++
188 else -> Unit
189 }
190 }
191
192 override fun toString() = "rounds $rounds hands $hands wins $wins losses $losses pushes $pushes" +
193 " blackjacks $blackjacks busts $busts surrenders $surrenders doubles $doubles splits $splits"
194}

206private class Shoe(val decks: Int, seed: Long) {
207 val cards = Array(52 * decks) { Card(it % 52) }
208 var drawn = 0
209 var running = 0
210 private var random = seed
211 init { shuffle() }
212
213 fun shuffle() {
214 for (i in cards.indices) cards[i] = Card(i % 52)
215 for (i in cards.lastIndex downTo 1) {
216 random = random * 48271 % 2147483647
217 val j = (random % (i + 1)).toInt()
218 val saved = cards[i]
219 cards[i] = cards[j]
220 cards[j] = saved
221 }
222 drawn = 0
223 running = 0
224 }
225
226 fun cut(penetration: Int) = cards.size * penetration / 100
227 fun draw(visible: Boolean): Card {
228 val card = cards[drawn++]
229 if (visible) running += card.hilo
230 return card
231 }
232}

238private class Table {
239 private var rules = Rules()
240 private val seats = arrayOfNulls<Player>(7)
241 private val participants = mutableListOf<Player>()
242 private val output = mutableListOf<String>()
243 private var shoe: Shoe? = null
244 private var state = State.IDLE
245 private var round = 0
246 private var dealerBlackjacks = 0
247 private var dealerBusts = 0
248 private var insuranceTurn = 0
249 private var dealer = Hand()
250 private var holeVisible = false
251
252 private fun emit(pattern: String, vararg values: Any?) { output += format(pattern, *values) }
253 private fun find(name: String) = seats.firstOrNull { it?.name == name }
254 private fun draw(visible: Boolean = true) = shoe!!.draw(visible)
255 private fun current(): Turn? {
256 for (player in participants) {
257 for ((index, hand) in player.hands.withIndex()) if (!hand.finished) return Turn(player, index, hand)
258 }
259 return null

262 private fun reveal() {
263 if (!holeVisible) {
264 shoe!!.running += dealer.cards[1].hilo
265 holeVisible = true
266 }
267 emit("dealer: %s", dealer)
268 }
269
270 private fun finishCheck(player: Player, hand: Hand) {
271 val value = hand.total.value
272 when {
273 value > 21 -> { hand.finished = true; hand.reason = Finish.BUST }
274 rules.charlie != 0 && hand.cards.size == rules.charlie -> { hand.finished = true; hand.reason = Finish.CHARLIE }
275 value == 21 -> hand.finished = true
276 hand.splitAces && hand.cards.size == 2 -> {
277 hand.finished = !(hand.cards.all { it.rank == 0 } && rules.resplitAces && player.hands.size < rules.hands)
278 }
279 }
280 }
281
282 private fun settle(dealerNatural: Boolean) {
283 val dealerTotal = dealer.total.value
284 for (player in participants) {
285 for ((index, hand) in player.hands.withIndex()) {
286 val result = hand.outcome(dealerTotal, dealerNatural)
287 val profit = result.profit(hand.bet, rules)
288 player.bankroll += hand.bet + profit
289 player.net += profit
290 player.stats.record(result)
291 emit("%s %s %+d", player.handName(index), result.text, profit)
292 }
293 emit("%s bankroll %d", player.name, player.bankroll)
294 }
295 for (player in seats.filterNotNull()) { player.bet = 0; player.side = 0 }
296 state = State.IDLE
297 }
298
299 private fun dealerTurn() {
300 reveal()
301 val needsDraw = participants.any { player -> player.hands.any { it.reason == null && !it.natural } }
302 var drew = false
303 if (needsDraw) {
304 while (true) {
305 val total = dealer.total
306 if (!(total.value <= 16 || total.value == 17 && total.soft && rules.hitSoft17)) break
307 dealer.cards += draw()
308 drew = true
309 emit("dealer hits: %s", dealer)
310 }
311 }
312 if (drew && dealer.total.value > 21) dealerBusts++
313 settle(false)
314 }
315
316 private fun advance() {
317 while (true) {
318 val turn = current()
319 if (turn == null) { dealerTurn(); return }
320 if (turn.hand.cards.size != 1) return
321 turn.hand.cards += draw()
322 emit("%s: %s", turn.name, turn.hand)
323 finishCheck(turn.player, turn.hand)
324 if (!turn.hand.finished) return
325 }
326 }

328 private fun peek() {
329 val natural = dealer.natural
330 emit("dealer peeks: %s", if (natural) "blackjack" else "no blackjack")
331 for (player in participants) {
332 if (player.insurance == 0L) continue
333 val profit = player.insurance * if (natural) 2 else -1
334 player.bankroll += player.insurance + profit
335 player.net += profit
336 emit("%s insurance %+d", player.name, profit)
337 }
338 if (natural) {
339 dealerBlackjacks++
340 reveal()
341 settle(true)
342 } else {
343 state = State.PLAY
344 advance()
345 }
346 }

348 private fun deal() {
349 val pile = shoe!!
350 if (pile.drawn >= pile.cut(rules.penetration)) { pile.shuffle(); emit("shuffle") }
351 emit("round %d", ++round)
352 participants.clear()
353 for (player in seats.filterNotNull()) {
354 if (player.bet == 0L) continue
355 participants += player
356 player.bankroll -= player.bet + player.side
357 player.insurance = 0
358 player.stats.rounds++
359 player.hands.clear()
360 player.hands += Hand(bet = player.bet)
361 }
362 dealer = Hand()
363 holeVisible = false
364 for (player in participants) player.hands[0].cards += draw()
365 dealer.cards += draw()
366 for (player in participants) {
367 val hand = player.hands[0]
368 hand.cards += draw()
369 hand.finished = hand.natural
370 }
371 dealer.cards += draw(false)
372 for (player in participants) emit("%s: %s", player.handName(0), player.hands[0])
373 emit("dealer: %s ??", dealer.cards[0])
374 for (player in participants) {
375 if (player.side == 0L) continue
376 val (first, second) = player.hands[0].cards
377 val (result, multiplier) = when {
378 first.rank != second.rank -> "lose" to -1
379 first.suit == second.suit -> "perfect" to 25
380 first.black == second.black -> "colored" to 12
381 else -> "mixed" to 6
382 }
383 val profit = player.side * multiplier
384 player.bankroll += player.side + profit
385 player.net += profit
386 emit("%s side %s %+d", player.name, result, profit)
387 }
388 when (dealer.cards[0].point) {
389 1 -> { state = State.INSURANCE; insuranceTurn = 0; emit("insurance offered") }
390 10 -> peek()
391 else -> { state = State.PLAY; advance() }
392 }
393 }

395 private fun actionProblem(action: Action, player: Player, hand: Hand): String? {
396 when (action) {
397 Action.HIT -> if (hand.splitAces) return "split aces"
398 Action.DOUBLE -> {
399 if (hand.splitAces) return "split aces"
400 if (hand.cards.size != 2) return "two cards only"
401 val total = hand.total
402 val minimum = if (rules.doubling == DoubleRule.HARD_10) 10 else 9
403 if (rules.doubling != DoubleRule.ANY && (total.soft || total.value !in minimum..11)
404 || !rules.doubleAfterSplit && hand.split) return "double not allowed"
405 if (player.bankroll < hand.bet) return "insufficient funds"
406 }
407 Action.SPLIT -> {
408 if (hand.cards.size != 2) return "two cards only"
409 if (hand.cards[0].point != hand.cards[1].point) return "not a pair"
410 if (player.hands.size >= rules.hands) return "hand limit"
411 if (player.bankroll < hand.bet) return "insufficient funds"
412 }
413 Action.SURRENDER -> {
414 if (!rules.surrender || hand.split) return "surrender not allowed"
415 if (hand.cards.size != 2) return "two cards only"
416 }
417 Action.STAND -> Unit
418 }
419 return null
420 }

422 private fun recommend(): Action {
423 val turn = current()!!
424 val hand = turn.hand
425 fun possible(action: Action) = actionProblem(action, turn.player, hand) == null
426 val up = if (dealer.cards[0].point == 1) 11 else dealer.cards[0].point
427 if (hand.splitAces) return if (possible(Action.SPLIT)) Action.SPLIT else Action.STAND
428 if (possible(Action.SPLIT)) {
429 val pair = hand.cards[0].point
430 if (pair == 1 || pair == 8 || pair == 9 && up !in listOf(7, 10, 11)
431 || pair in listOf(2, 3, 7) && up <= 7 || pair == 6 && up <= 6 || pair == 4 && up in 5..6) return Action.SPLIT
432 }
433 val total = hand.total
434 val value = total.value
435 val symbol = if (total.soft) when {
436 value >= 19 -> "S"
437 value == 18 -> when (up) { in 3..6 -> "Ds"; 2, 7, 8 -> "S"; else -> "H" }
438 value == 17 && up in 3..6 || value in 15..16 && up in 4..6 || value in 13..14 && up in 5..6 -> "D"
439 else -> "H"
440 } else when {
441 value >= 17 -> "S"
442 value >= 13 -> when {
443 up <= 6 -> "S"
444 value == 16 && up >= 9 || value == 15 && up == 10 -> "R"
445 else -> "H"
446 }
447 value == 12 -> if (up in 4..6) "S" else "H"
448 value == 11 && up <= 10 || value == 10 && up <= 9 || value == 9 && up in 3..6 -> "D"
449 else -> "H"
450 }
451 return when (symbol) {
452 "S" -> Action.STAND
453 "D" -> if (possible(Action.DOUBLE)) Action.DOUBLE else Action.HIT
454 "Ds" -> if (possible(Action.DOUBLE)) Action.DOUBLE else Action.STAND
455 "R" -> if (possible(Action.SURRENDER)) Action.SURRENDER else Action.HIT
456 else -> Action.HIT
457 }
458 }

460 private fun perform(action: Action) {
461 val turn = current()!!
462 val player = turn.player
463 val hand = turn.hand
464 val name = turn.name
465 when (action) {
466 Action.SPLIT -> {
467 val (first, second) = hand.cards
468 val aces = hand.splitAces || first.rank == 0
469 player.bankroll -= hand.bet
470 player.stats.splits++
471 emit("%s splits %s %s", name, first, second)
472 val replacement = Hand(listOf(first), hand.bet, split = true, splitAces = aces)
473 player.hands[turn.index] = replacement
474 player.hands.add(turn.index + 1, Hand(listOf(second), hand.bet, split = true, splitAces = aces))
475 replacement.cards += draw()
476 emit("%s: %s", name, replacement)
477 finishCheck(player, replacement)
478 }
479 Action.STAND, Action.SURRENDER -> {
480 hand.finished = true
481 if (action == Action.SURRENDER) hand.reason = Finish.SURRENDER
482 emit("%s %s: %s", name, if (action == Action.STAND) "stands" else "surrenders", hand)
483 }
484 Action.HIT, Action.DOUBLE -> {
485 if (action == Action.DOUBLE) {
486 player.bankroll -= hand.bet
487 hand.bet *= 2
488 player.stats.doubles++
489 }
490 hand.cards += draw()
491 emit("%s %s: %s", name, if (action == Action.DOUBLE) "doubles" else "hits", hand)
492 finishCheck(player, hand)
493 if (action == Action.DOUBLE) hand.finished = true
494 }
495 }
496 advance()
497 }

499 private fun status() {
500 when (state) {
501 State.IDLE -> emit("table idle round %d", round)
502 State.INSURANCE -> emit("table insurance round %d next %s", round, participants[insuranceTurn].name)
503 State.PLAY -> emit("table play round %d turn %s", round, current()!!.name)
504 }
505 if (state != State.IDLE) emit("dealer: %s ??", dealer.cards[0])
506 for (player in seats.filterNotNull()) {
507 emit("seat %d %s bankroll %d bet %d side %d", player.seat + 1, player.name, player.bankroll, player.bet, player.side)
508 if (state != State.IDLE && player in participants) {
509 for ((index, hand) in player.hands.withIndex()) emit("%s: %s bet %d", player.handName(index), hand, hand.bet)
510 }
511 }
512 }

514 fun execute(words: List<String>): List<String> {
515 output.clear()
516 val command = Command.parse(words[0])
517 val action = command.action
518 var target: Player? = null
519 var amount = 0L
520 var seed = 0L
521 var proposed: Rules? = null
522 when (command) {
523 Command.SHOE -> { amount = integer(words[1], 1, 8); seed = integer(words[2], 1, 2147483646) }
524 Command.RULE -> proposed = rules.changed(words[1], words[2])
525 Command.PLAYER -> {
526 val name = words[1]
527 if (name.isEmpty() || name.length > 8 || name.any { it !in 'A'..'Z' }) fail("bad name $name")
528 amount = integer(words[2], 1, 1000000)
529 }
530 else -> if (command.namedPlayer) {
531 target = find(words[1]) ?: fail("unknown player ${words[1]}")
532 if (command in listOf(Command.REBUY, Command.BET, Command.SIDE, Command.INSURANCE)) {
533 amount = integer(words[2], if (command == Command.REBUY) 1 else 0, 1000000)
534 }
535 }
536 }
537 if (command.idleOnly && state != State.IDLE) fail("round in progress")
538 if (command == Command.INSURANCE && state != State.INSURANCE) fail("no insurance")
539 if (action != null || command == Command.AUTO || command == Command.HINT) {
540 if (state == State.IDLE) fail("no round")
541 if (state == State.INSURANCE) fail("insurance pending")
542 }
543 if (action != null || command == Command.AUTO) {
544 val turn = current()!!
545 if (turn.player !== target) fail("waiting for ${turn.name}")
546 if (action != null) {
547 actionProblem(action, turn.player, turn.hand)?.let { fail(it) }
548 perform(action)
549 } else {
550 while (state == State.PLAY && current()!!.player === target) perform(recommend())
551 }
552 return output
553 }
554 when (command) {
555 Command.SHOE -> {
556 val pile = Shoe(amount.toInt(), seed)
557 shoe = pile
558 emit("shoe %d decks %d cards cut %d", pile.decks, pile.cards.size, pile.cut(rules.penetration))
559 }
560 Command.RULE -> {
561 val next = proposed!!
562 if (next.minimum > next.maximum) fail("min above max")
563 rules = next
564 emit("rule %s %s", words[1], words[2])
565 }
566 Command.RULES -> emit("%s", rules)
567 Command.PLAYER -> {
568 if (find(words[1]) != null) fail("player exists")
569 val seat = seats.indexOfFirst { it == null }
570 if (seat < 0) fail("table full")
571 seats[seat] = Player(words[1], seat, amount)
572 emit("%s sits at seat %d with %d", words[1], seat + 1, amount)
573 }
574 Command.LEAVE -> {
575 val player = target!!
576 emit("%s leaves with %d net %+d", player.name, player.bankroll, player.net)
577 seats[player.seat] = null
578 }
579 Command.REBUY -> {
580 val player = target!!
581 player.bankroll += amount
582 emit("%s rebuys %d bankroll %d", player.name, amount, player.bankroll)
583 }
584 Command.BET -> {
585 val player = target!!
586 if (amount != 0L) {
587 if (amount !in rules.minimum..rules.maximum) fail("bet out of limits")
588 if (amount + player.side > player.bankroll) fail("insufficient funds")
589 }
590 player.bet = amount
591 if (amount == 0L) { player.side = 0; emit("%s bet cleared", player.name) }
592 else emit("%s bets %d", player.name, amount)
593 }
594 Command.SIDE -> {
595 val player = target!!
596 if (amount != 0L) {
597 if (player.bet == 0L) fail("no bet")
598 if (amount > rules.sideMaximum) fail("side out of limits")
599 if (player.bet + amount > player.bankroll) fail("insufficient funds")
600 }
601 player.side = amount
602 if (amount == 0L) emit("%s side cleared", player.name)
603 else emit("%s side %d", player.name, amount)
604 }
605 Command.DEAL -> {
606 if (shoe == null) fail("no shoe")
607 if (seats.none { it != null && it.bet != 0L }) fail("no bets")
608 deal()
609 }
610 Command.INSURANCE -> {
611 val player = target!!
612 val waiting = participants[insuranceTurn]
613 if (player !== waiting) fail("waiting for ${waiting.name}")
614 if (amount > player.bet / 2) fail("insurance out of limits")
615 if (amount > player.bankroll) fail("insufficient funds")
616 player.bankroll -= amount
617 player.insurance = amount
618 if (amount == 0L) emit("%s declines insurance", player.name)
619 else emit("%s insures %d", player.name, amount)
620 if (++insuranceTurn == participants.size) peek()
621 }
622 Command.HINT -> emit("hint %s: %s", current()!!.name, recommend().text)
623 Command.STATUS -> status()
624 Command.COUNT -> {
625 val pile = shoe ?: fail("no shoe")
626 val remaining = pile.cards.size - pile.drawn
627 val scaled = kotlin.math.abs(pile.running.toLong()) * 5200 / remaining
628 emit("count running %d true %s%d.%02d remaining %d", pile.running, if (pile.running < 0) "-" else "", scaled / 100, scaled % 100, remaining)
629 }
630 Command.STATS -> {
631 val players = seats.filterNotNull()
632 if (players.isEmpty()) fail("no players")
633 for (player in players) emit("stats %s %s net %+d", player.name, player.stats, player.net)
634 emit("stats dealer rounds %d blackjacks %d busts %d", round, dealerBlackjacks, dealerBusts)
635 }
636 else -> Unit
637 }
638 return output
639 }
