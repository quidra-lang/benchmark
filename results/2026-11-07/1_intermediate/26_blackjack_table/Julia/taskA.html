115function integer_word(word::AbstractString, low::Integer, high::Integer)::Int64
116 occursin(r"^(0|[1-9][0-9]{0,9})$", word) || fail("bad value $word")
117 value = parse(Int64, word)
118 low <= value <= high || fail("bad value $word")
119 value
120end
121function parse_switch(word::AbstractString)
122 word in ("ON", "OFF") || fail("bad value $word")
123 word == "ON"
124end
125function change_rule(original::Rules, name::AbstractString, word::AbstractString)
126 name in RULE_NAMES || fail("unknown rule $name")
127 changed = deepcopy(original)
128 if name == "DEALER"
129 word in ("S17", "H17") || fail("bad value $word")
130 changed.dealer = word == "H17" ? H17 : S17
131 elseif name == "PEN"
132 changed.pen = integer_word(word, 50, 90)
133 elseif name == "MIN"
134 changed.minimum = integer_word(word, 1, 10000)
135 elseif name == "MAX"
136 changed.maximum = integer_word(word, 1, 10000)
137 elseif name == "BJ"
138 word in ("3:2", "6:5") || fail("bad value $word")
139 changed.bj = word == "3:2" ? ThreeToTwo : SixToFive
140 elseif name == "DAS"
141 changed.das = parse_switch(word)
142 elseif name == "DOUBLE"
143 word in ("ANY", "9-11", "10-11") || fail("bad value $word")
144 changed.doubling = word == "ANY" ? AnyDouble : word == "9-11" ? Hard9 : Hard10
145 elseif name == "HANDS"
146 changed.hands = integer_word(word, 1, 4)
147 elseif name == "RSA"
148 changed.rsa = parse_switch(word)
149 elseif name == "SURRENDER"
150 changed.surrender = parse_switch(word)
151 elseif name == "CHARLIE"
152 value = integer_word(word, 0, 7)
153 value in (0, 5, 6, 7) || fail("bad value $word")
154 changed.charlie = value
155 else
156 changed.side_maximum = integer_word(word, 0, 1000)
157 end
158 changed
159end
160switch_text(value::Bool) = value ? "ON" : "OFF"
161function rule_text(rules::Rules, name::AbstractString)
162 name == "DEALER" && return string(rules.dealer)
163 name == "PEN" && return string(rules.pen)
164 name == "MIN" && return string(rules.minimum)
165 name == "MAX" && return string(rules.maximum)
166 name == "BJ" && return rules.bj == ThreeToTwo ? "3:2" : "6:5"
167 name == "DAS" && return switch_text(rules.das)
168 name == "DOUBLE" && return rules.doubling == AnyDouble ? "ANY" : rules.doubling == Hard9 ? "9-11" : "10-11"
169 name == "HANDS" && return string(rules.hands)
170 name == "RSA" && return switch_text(rules.rsa)
171 name == "SURRENDER" && return switch_text(rules.surrender)
172 name == "CHARLIE" && return string(rules.charlie)
173 string(rules.side_maximum)
174end

176point(card::Int) = min(mod(card, 13) + 1, 10)
177card_text(card::Int) = string(RANKS[mod(card, 13) + 1], SUITS[div(card, 13) + 1])
178function hand_total(hand::Hand)
179 value = sum(point, hand.cards; init = 0)
180 soft = any(card -> mod(card, 13) == 0, hand.cards) && value + 10 <= 21
181 (value = soft ? value + 10 : value, soft = soft)
182end
183blackjack(hand::Hand) = !hand.split && length(hand.cards) == 2 && hand_total(hand).value == 21
184function hand_text(hand::Hand)
185 score = hand_total(hand)
186 status = blackjack(hand) ? "blackjack" : score.value > 21 ? "bust $(score.value)" : score.soft ? "soft $(score.value)" : string(score.value)
187 join(card_text.(hand.cards), ' ') * " ($status)"
188end
189hand_name(player::Player, index::Int) = "$(player.name)[$index]"
190function hand_line(table::Table, seat::Int, index::Int, verb::String = "")
191 player = player_at(table, seat)
192 emit("$(hand_name(player, index))$verb: $(hand_text(player.hands[index]))")
193end
194cut(shoe::Shoe, rules::Rules) = (length(shoe.cards) * rules.pen) ÷ 100
195function shuffle!(shoe::Shoe)
196 shoe.cards = mod.(0:(52 * shoe.decks - 1), 52)
197 for index in length(shoe.cards):-1:2
198 shoe.random_state = 48271 * shoe.random_state % 2147483647
199 other = Int(shoe.random_state % index) + 1
200 shoe.cards[index], shoe.cards[other] = shoe.cards[other], shoe.cards[index]
201 end
202 shoe.drawn = shoe.running = 0
203end
204function hilo(card::Int)
205 value = point(card)
206 2 <= value <= 6 ? 1 : 7 <= value <= 9 ? 0 : -1
207end
208function draw!(table::Table, visible::Bool)
209 shoe = shoe_of(table)
210 shoe.drawn += 1
211 card = shoe.cards[shoe.drawn]
212 visible && (shoe.running += hilo(card))
213 card
214end
215function reveal_dealer!(table::Table)
216 if !table.hole_visible
217 shoe_of(table).running += hilo(table.dealer.cards[2])
218 table.hole_visible = true
219 end
220 emit("dealer: $(hand_text(table.dealer))")
221end
222function finish_hand!(hand::Hand, rules::Rules, hand_count::Int)
223 value = hand_total(hand).value
224 if value > 21
225 hand.finish = Bust
226 elseif rules.charlie != 0 && length(hand.cards) == rules.charlie
227 hand.finish = Charlie
228 elseif value == 21
229 hand.finish = Ordinary
230 elseif hand.split_aces && length(hand.cards) == 2
231 again = all(card -> mod(card, 13) == 0, hand.cards) && rules.rsa && hand_count < rules.hands
232 hand.finish = again ? Pending : Ordinary
233 end
234end
235function hand_result(table::Table, hand::Hand)::Outcome
236 table.dealer_natural && return blackjack(hand) ? Push : Lose
237 blackjack(hand) && return Blackjack
238 hand.finish == Bust && return Busted
239 hand.finish == Surrendered && return SurrenderLoss
240 hand.finish == Charlie && return CharlieWin
241 player_total = hand_total(hand).value
242 dealer_total = hand_total(table.dealer).value
243 dealer_total > 21 || player_total > dealer_total ? Win : player_total == dealer_total ? Push : Lose
244end
245function hand_profit(rules::Rules, bet::Int64, result::Outcome)::Int64
246 result == Blackjack && return rules.bj == ThreeToTwo ? (bet * 3) ÷ 2 : (bet * 6) ÷ 5
247 result in (Win, CharlieWin) && return bet
248 result == Push && return 0
249 result == SurrenderLoss && return bet ÷ 2 - bet
250 -bet
251end
252function record!(stats::Statistics, result::Outcome)
253 stats.hands += 1
254 if result in (Blackjack, Win, CharlieWin)
255 stats.wins += 1
256 elseif result == Push
257 stats.pushes += 1
258 else
259 stats.losses += 1
260 end
261 result == Blackjack && (stats.blackjacks += 1)
262 result == Busted && (stats.busts += 1)
263 result == SurrenderLoss && (stats.surrenders += 1)
264end
265function settle!(table::Table)
266 for seat in table.participants
267 player = player_at(table, seat)
268 for (index, hand) in enumerate(player.hands)
269 result = hand_result(table, hand)
270 profit = hand_profit(table.rules, hand.bet, result)
271 player.bankroll += hand.bet + profit
272 player.net += profit
273 record!(player.stats, result)
274 emit(hand_name(player, index), RESULT_NAMES[result], signed_text(profit))
275 end
276 emit(player.name, "bankroll", player.bankroll)
277 end
278 for seat in seated(table)
279 player = player_at(table, seat)
280 player.bet = player.side = 0
281 end
282 table.phase = Idle
283 table.current = nothing
284end
285function dealer_turn!(table::Table)
286 reveal_dealer!(table)
287 required = any(hand.finish == Ordinary && !blackjack(hand) for seat in table.participants for hand in player_at(table, seat).hands)
288 drew = false
289 if required
290 while true
291 score = hand_total(table.dealer)
292 (score.value < 17 || (score.value == 17 && score.soft && table.rules.dealer == H17)) || break
293 push!(table.dealer.cards, draw!(table, true))
294 emit("dealer hits: $(hand_text(table.dealer))")
295 drew = true
296 end
297 end
298 drew && hand_total(table.dealer).value > 21 && (table.dealer_busts += 1)
299 settle!(table)
300end
301function advance_hands!(table::Table)
302 while true
303 table.current = nothing
304 for seat in table.participants
305 index = findfirst(hand -> hand.finish == Pending, player_at(table, seat).hands)
306 if !isnothing(index)
307 table.current = Turn(seat, index)
308 break
309 end
310 end
311 if isnothing(table.current)
312 dealer_turn!(table)
313 return
314 end
315 turn = current_turn(table)
316 player = player_at(table, turn.seat)
317 hand = player.hands[turn.index]
318 length(hand.cards) == 1 || return
319 push!(hand.cards, draw!(table, true))
320 hand_line(table, turn.seat, turn.index)
321 finish_hand!(hand, table.rules, length(player.hands))
322 hand.finish == Pending && return
323 end
324end
325function peek!(table::Table)
326 table.dealer_natural = blackjack(table.dealer)
327 emit("dealer peeks:", table.dealer_natural ? "blackjack" : "no blackjack")
328 for seat in table.participants
329 player = player_at(table, seat)
330 player.insurance == 0 && continue
331 profit = table.dealer_natural ? player.insurance * 2 : -player.insurance
332 player.bankroll += player.insurance + profit
333 player.net += profit
334 emit(player.name, "insurance", signed_text(profit))
335 end
336 if table.dealer_natural
337 table.dealer_blackjacks += 1
338 reveal_dealer!(table)
339 settle!(table)
340 else
341 table.phase = Playing
342 advance_hands!(table)
343 end
344end
345function deal!(table::Table)
346 shoe = shoe_of(table)
347 if shoe.drawn >= cut(shoe, table.rules)
348 shuffle!(shoe)
349 emit("shuffle")
350 end
351 table.round += 1
352 emit("round", table.round)
353 table.participants = [seat for seat in seated(table) if player_at(table, seat).bet != 0]
354 for seat in table.participants
355 player = player_at(table, seat)
356 player.bankroll -= player.bet + player.side
357 player.insurance = 0
358 player.stats.rounds += 1
359 player.hands = [Hand(bet = player.bet)]
360 end
361 for seat in table.participants
362 push!(player_at(table, seat).hands[1].cards, draw!(table, true))
363 end
364 table.dealer = Hand(cards = [draw!(table, true)])
365 for seat in table.participants
366 hand = player_at(table, seat).hands[1]
367 push!(hand.cards, draw!(table, true))
368 blackjack(hand) && (hand.finish = Ordinary)
369 end
370 push!(table.dealer.cards, draw!(table, false))
371 table.hole_visible = table.dealer_natural = false
372 table.current = nothing
373 for seat in table.participants
374 hand_line(table, seat, 1)
375 end
376 emit("dealer:", card_text(table.dealer.cards[1]), "??")
377 for seat in table.participants
378 player = player_at(table, seat)
379 player.side == 0 && continue
380 first, second = player.hands[1].cards
381 same_color = (div(first, 13) in (0, 3)) == (div(second, 13) in (0, 3))
382 multiplier = mod(first, 13) != mod(second, 13) ? -1 : first == second ? 25 : same_color ? 12 : 6
383 result = multiplier == 25 ? "perfect" : multiplier == 12 ? "colored" : multiplier == 6 ? "mixed" : "lose"
384 profit = player.side * multiplier
385 player.bankroll += player.side + profit
386 player.net += profit
387 emit(player.name, "side", result, signed_text(profit))
388 end
389 up = point(table.dealer.cards[1])
390 if up == 1
391 table.phase = Insurance
392 table.insurance_turn = 1
393 emit("insurance offered")
394 elseif up == 10
395 peek!(table)
396 else
397 table.phase = Playing
398 advance_hands!(table)
399 end
400end

402function action_problem(table::Table, action::Action)::Union{Nothing,String}
403 turn = current_turn(table)
404 player = player_at(table, turn.seat)
405 hand = player.hands[turn.index]
406 if action == Hit
407 return hand.split_aces ? "split aces" : nothing
408 elseif action == DoubleDown
409 hand.split_aces && return "split aces"
410 length(hand.cards) == 2 || return "two cards only"
411 score = hand_total(hand)
412 low = table.rules.doubling == Hard9 ? 9 : 10
413 limited = table.rules.doubling != AnyDouble && (score.soft || score.value < low || score.value > 11)
414 (limited || (!table.rules.das && hand.split)) && return "double not allowed"
415 player.bankroll < hand.bet && return "insufficient funds"
416 elseif action == SplitHand
417 length(hand.cards) == 2 || return "two cards only"
418 point(hand.cards[1]) == point(hand.cards[2]) || return "not a pair"
419 length(player.hands) >= table.rules.hands && return "hand limit"
420 player.bankroll < hand.bet && return "insufficient funds"
421 elseif action == Surrender
422 (!table.rules.surrender || hand.split) && return "surrender not allowed"
423 length(hand.cards) == 2 || return "two cards only"
424 end
425 nothing
426end
427function recommendation(table::Table)::Action
428 turn = current_turn(table)
429 hand = player_at(table, turn.seat).hands[turn.index]
430 can_split = isnothing(action_problem(table, SplitHand))
431 hand.split_aces && return can_split ? SplitHand : Stand
432 up = point(table.dealer.cards[1])
433 up == 1 && (up = 11)
434 if can_split
435 value = point(hand.cards[1])
436 wanted = value in (1, 8) || (value == 9 && !(up in (7, 10, 11))) || (value in (2, 3, 7) && up <= 7) || (value == 6 && up <= 6) || (value == 4 && up in (5, 6))
437 wanted && return SplitHand
438 end
439 score = hand_total(hand)
440 value = score.value
441 can_double = isnothing(action_problem(table, DoubleDown))
442 if score.soft
443 value >= 19 && return Stand
444 if value == 18
445 3 <= up <= 6 && return can_double ? DoubleDown : Stand
446 return up in (2, 7, 8) ? Stand : Hit
447 end
448 wanted = (value == 17 && 3 <= up <= 6) || (value in (15, 16) && 4 <= up <= 6) || (value in (13, 14) && up in (5, 6))
449 return wanted && can_double ? DoubleDown : Hit
450 end
451 value >= 17 && return Stand
452 if value >= 13
453 up <= 6 && return Stand
454 if ((value == 16 && up >= 9) || (value == 15 && up == 10)) && isnothing(action_problem(table, Surrender))
455 return Surrender
456 end
457 return Hit
458 end
459 value == 12 && return 4 <= up <= 6 ? Stand : Hit
460 wanted = (value == 11 && up <= 10) || (value == 10 && up <= 9) || (value == 9 && 3 <= up <= 6)
461 wanted && can_double ? DoubleDown : Hit
462end
463function perform_action!(table::Table, action::Action)
464 turn = current_turn(table)
465 player = player_at(table, turn.seat)
466 hand = player.hands[turn.index]
467 name = hand_name(player, turn.index)
468 if action == SplitHand
469 player.bankroll -= hand.bet
470 player.stats.splits += 1
471 emit(name, "splits", card_text.(hand.cards)...)
472 aces = hand.split_aces || point(hand.cards[1]) == 1
473 first = Hand(cards = [hand.cards[1]], bet = hand.bet, split = true, split_aces = aces)
474 second = Hand(cards = [hand.cards[2]], bet = hand.bet, split = true, split_aces = aces)
475 player.hands[turn.index] = first
476 insert!(player.hands, turn.index + 1, second)
477 push!(first.cards, draw!(table, true))
478 hand_line(table, turn.seat, turn.index)
479 finish_hand!(first, table.rules, length(player.hands))
480 elseif action == Stand || action == Surrender
481 hand.finish = action == Stand ? Ordinary : Surrendered
482 hand_line(table, turn.seat, turn.index, action == Stand ? " stands" : " surrenders")
483 else
484 if action == DoubleDown
485 player.bankroll -= hand.bet
486 hand.bet *= 2
487 player.stats.doubles += 1
488 end
489 push!(hand.cards, draw!(table, true))
490 hand_line(table, turn.seat, turn.index, action == DoubleDown ? " doubles" : " hits")
491 finish_hand!(hand, table.rules, length(player.hands))
492 action == DoubleDown && hand.finish == Pending && (hand.finish = Ordinary)
493 end
494 advance_hands!(table)
495end

526function command!(table::Table, tokens::AbstractVector{<:AbstractString})
527 op = tokens[1]
528 op in COMMANDS || fail("unknown command $op")
529 seat = 0
530 amount = seed = Int64(0)
531 proposed = table.rules
532 if op == "SHOE"
533 amount = integer_word(tokens[2], 1, 8)
534 seed = integer_word(tokens[3], 1, 2147483646)
535 elseif op == "RULE"
536 proposed = change_rule(table.rules, tokens[2], tokens[3])
537 elseif op == "PLAYER"
538 occursin(r"^[A-Z]{1,8}$", tokens[2]) || fail("bad name $(tokens[2])")
539 amount = integer_word(tokens[3], 1, 1000000)
540 elseif op in PLAYER_COMMANDS
541 seat = lookup_player(table, tokens[2])
542 if op in ("REBUY", "BET", "SIDE", "INSURANCE")
543 amount = integer_word(tokens[3], op == "REBUY" ? 1 : 0, 1000000)
544 end
545 end
546 op in IDLE_COMMANDS && table.phase != Idle && fail("round in progress")
547 op == "INSURANCE" && table.phase != Insurance && fail("no insurance")
548 action_command = haskey(ACTIONS, op) || op == "AUTO"
549 if action_command || op == "HINT"
550 table.phase == Idle && fail("no round")
551 table.phase == Insurance && fail("insurance pending")
552 turn = current_turn(table)
553 action_command && seat != turn.seat && fail("waiting for $(hand_name(player_at(table, turn.seat), turn.index))")
554 end
555 if action_command
556 if op == "AUTO"
557 while table.phase == Playing && current_turn(table).seat == seat
558 perform_action!(table, recommendation(table))
559 end
560 else
561 action = ACTIONS[op]
562 problem = action_problem(table, action)
563 isnothing(problem) || fail(problem)
564 perform_action!(table, action)
565 end
566 return
567 end
568 if op == "SHOE"
569 table.shoe = Shoe(decks = Int(amount), random_state = seed)
570 shuffle!(shoe_of(table))
571 emit("shoe", amount, "decks", length(shoe_of(table).cards), "cards cut", cut(shoe_of(table), table.rules))
572 elseif op == "RULE"
573 proposed.minimum > proposed.maximum && fail("min above max")
574 table.rules = proposed
575 emit("rule", tokens[2], rule_text(table.rules, tokens[2]))
576 elseif op == "RULES"
577 emit("rules", join(("$name $(rule_text(table.rules, name))" for name in RULE_NAMES), ' '))
578 elseif op == "PLAYER"
579 !isnothing(find_player(table, tokens[2])) && fail("player exists")
580 empty_seat = findfirst(isnothing, table.players)
581 isnothing(empty_seat) && fail("table full")
582 table.players[empty_seat] = Player(name = String(tokens[2]), bankroll = amount)
583 emit(tokens[2], "sits at seat", empty_seat, "with", amount)
584 elseif op == "LEAVE"
585 player = player_at(table, seat)
586 emit(player.name, "leaves with", player.bankroll, "net", signed_text(player.net))
587 table.players[seat] = nothing
588 elseif op == "REBUY"
589 player = player_at(table, seat)
590 player.bankroll += amount
591 emit(player.name, "rebuys", amount, "bankroll", player.bankroll)
592 elseif op == "BET"
593 player = player_at(table, seat)
594 if amount != 0
595 table.rules.minimum <= amount <= table.rules.maximum || fail("bet out of limits")
596 amount + player.side > player.bankroll && fail("insufficient funds")
597 end
598 player.bet = amount
599 if amount == 0
600 player.side = 0
601 emit(player.name, "bet cleared")
602 else
603 emit(player.name, "bets", amount)
604 end
605 elseif op == "SIDE"
606 player = player_at(table, seat)
607 if amount != 0
608 player.bet == 0 && fail("no bet")
609 amount > table.rules.side_maximum && fail("side out of limits")
610 player.bet + amount > player.bankroll && fail("insufficient funds")
611 end
612 player.side = amount
613 amount == 0 ? emit(player.name, "side cleared") : emit(player.name, "side", amount)
614 elseif op == "DEAL"
615 isnothing(table.shoe) && fail("no shoe")
616 any(player_at(table, index).bet != 0 for index in seated(table)) || fail("no bets")
617 deal!(table)
618 elseif op == "INSURANCE"
619 waiting = table.participants[table.insurance_turn]
620 seat == waiting || fail("waiting for $(player_at(table, waiting).name)")
621 player = player_at(table, seat)
622 amount > player.bet ÷ 2 && fail("insurance out of limits")
623 amount > player.bankroll && fail("insufficient funds")
624 player.bankroll -= amount
625 player.insurance = amount
626 amount == 0 ? emit(player.name, "declines insurance") : emit(player.name, "insures", amount)
627 table.insurance_turn += 1
628 table.insurance_turn > length(table.participants) && peek!(table)
629 elseif op == "HINT"
630 turn = current_turn(table)
631 emit("hint $(hand_name(player_at(table, turn.seat), turn.index)): $(ACTION_NAMES[recommendation(table)])")
632 elseif op == "STATUS"
633 status_output(table)
634 elseif op == "COUNT"
635 isnothing(table.shoe) && fail("no shoe")
636 shoe = shoe_of(table)
637 remaining = length(shoe.cards) - shoe.drawn
638 scaled = (abs(shoe.running) * 5200) ÷ remaining
639 sign = shoe.running < 0 ? "-" : ""
640 decimals = lpad(string(mod(scaled, 100)), 2, '0')
641 emit("count running", shoe.running, "true", "$sign$(scaled ÷ 100).$decimals", "remaining", remaining)
642 elseif op == "STATS"
643 isempty(seated(table)) && fail("no players")
644 statistics_output(table)
645 end
646end

652 tokens = split(readline(), ' ')
