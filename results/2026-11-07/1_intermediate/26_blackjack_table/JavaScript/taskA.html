3const RANKS = 'A23456789TJQK';
4const SUITS = 'CDHS';
5const RULE_DEFINITIONS = [
6 ['DEALER', 'S17', ['S17', 'H17']],
7 ['PEN', 75, [50, 90]],
8 ['MIN', 10, [1, 10000]],
9 ['MAX', 500, [1, 10000]],
10 ['BJ', '3:2', ['3:2', '6:5']],
11 ['DAS', 'ON', ['ON', 'OFF']],
12 ['DOUBLE', 'ANY', ['ANY', '9-11', '10-11']],
13 ['HANDS', 4, [1, 4]],
14 ['RSA', 'OFF', ['ON', 'OFF']],
15 ['SURRENDER', 'ON', ['ON', 'OFF']],
16 ['CHARLIE', '0', ['0', '5', '6', '7']],
17 ['SIDEMAX', 25, [0, 1000]],
18];
19const COMMANDS = new Set([
20 'SHOE', 'RULE', 'RULES', 'PLAYER', 'LEAVE', 'REBUY', 'BET', 'SIDE',
21 'DEAL', 'INSURANCE', 'HIT', 'STAND', 'DOUBLE', 'SPLIT', 'SURRENDER',
22 'HINT', 'AUTO', 'STATUS', 'COUNT', 'STATS',
23]);
24const ACTIONS = new Set(['HIT', 'STAND', 'DOUBLE', 'SPLIT', 'SURRENDER']);
25const STAT_NAMES = [
26 'rounds', 'hands', 'wins', 'losses', 'pushes', 'blackjacks', 'busts',
27 'surrenders', 'doubles', 'splits',
28];

30class CommandError extends Error {}
31function fail(reason) { throw new CommandError(reason); }
32
33function integer(word, minimum, maximum) {
34 if (!/^(?:0|[1-9][0-9]{0,9})$/.test(word)) fail(`bad value ${word}`);
35 const value = Number(word);
36 if (value < minimum || value > maximum) fail(`bad value ${word}`);
37 return value;
38}
39
40const rank = card => card % 13;
41const suit = card => Math.floor(card / 13);
42const points = card => Math.min(rank(card) + 1, 10);
43const cardText = card => RANKS[rank(card)] + SUITS[suit(card)];
44const signed = value => (value >= 0 ? '+' : '') + value;
45
46function total(cards) {
47 const sum = cards.reduce((n, card) => n + points(card), 0);
48 const soft = cards.some(card => rank(card) === 0) && sum + 10 <= 21;
49 return { value: sum + (soft ? 10 : 0), soft };
50}
51
52function blackjack(cards, split = false) {
53 return !split && cards.length === 2 && total(cards).value === 21;
54}
55
56function handText(cards, split = false) {
57 const { value, soft } = total(cards);
58 const status = blackjack(cards, split) ? 'blackjack'
59 : value > 21 ? `bust ${value}` : soft ? `soft ${value}` : String(value);
60 return `${cards.map(cardText).join(' ')} (${status})`;
61}

63function makeHand(cards, bet, split = false, splitAces = false) {
64 return {
65 cards, bet, split, splitAces,
66 finished: blackjack(cards, split), reason: '',
67 };

70class Player {
71 constructor(name, seat, bankroll) {
72 this.name = name;
73 this.seat = seat;
74 this.bankroll = bankroll;
75 this.bet = 0;
76 this.side = 0;
77 this.insurance = 0;
78 this.hands = [];
79 this.net = 0;
80 this.stats = Object.fromEntries(STAT_NAMES.map(name => [name, 0]));
81 }
82
83 handName(index) { return `${this.name}[${index + 1}]`; }
84}

86class Table {
87 constructor() {
88 this.rules = Object.fromEntries(RULE_DEFINITIONS.map(([name, value]) => [name, value]));
89 this.seats = Array(7).fill(null);
90 this.participants = [];
91 this.state = 'idle';
92 this.round = 0;
93 this.dealerBlackjacks = 0;
94 this.dealerBusts = 0;
95 this.decks = 0;
96 this.random = 1;
97 this.shoe = [];
98 this.drawn = 0;
99 this.running = 0;
100 this.dealer = [];
101 this.holeRevealed = false;
102 this.insuranceTurn = 0;
103 this.output = [];
104 }
105
106 emit(line) { this.output.push(line); }
107 players() { return this.seats.filter(player => player !== null); }
108 cut() { return Math.floor(this.shoe.length * this.rules.PEN / 100); }
109
110 player(name) {
111 const player = this.seats.find(player => player !== null && player.name === name);
112 if (!player) fail(`unknown player ${name}`);
113 return player;
114 }
115
116 shuffle() {
117 this.shoe = Array.from({ length: 52 * this.decks }, (_, index) => index % 52);
118 for (let i = this.shoe.length - 1; i > 0; --i) {
119 this.random = this.random * 48271 % 2147483647;
120 const j = this.random % (i + 1);
121 [this.shoe[i], this.shoe[j]] = [this.shoe[j], this.shoe[i]];
122 }
123 this.drawn = 0;
124 this.running = 0;
125 }
126
127 countCard(card) {
128 const value = points(card);
129 this.running += value >= 2 && value <= 6 ? 1 : value === 1 || value === 10 ? -1 : 0;
130 }
131
132 draw(faceUp = true) {
133 const card = this.shoe[this.drawn++];
134 if (faceUp) this.countCard(card);
135 return card;
136 }
137
138 revealDealer() {
139 if (!this.holeRevealed) {
140 this.countCard(this.dealer[1]);
141 this.holeRevealed = true;
142 }
143 this.emit(`dealer: ${handText(this.dealer)}`);
144 }
145
146 current() {
147 for (const player of this.participants) {
148 const index = player.hands.findIndex(hand => !hand.finished);
149 if (index !== -1) return { player, index, hand: player.hands[index] };
150 }
151 return null;
152 }
153
154 finishCheck(player, hand) {
155 const value = total(hand.cards).value;
156 if (value > 21) {
157 hand.reason = 'bust';
158 hand.finished = true;
159 } else if (this.rules.CHARLIE !== '0' && hand.cards.length === Number(this.rules.CHARLIE)) {
160 hand.reason = 'charlie';
161 hand.finished = true;
162 } else if (value === 21) {
163 hand.finished = true;
164 } else if (hand.splitAces && hand.cards.length === 2) {
165 hand.finished = !(hand.cards.every(card => rank(card) === 0)
166 && this.rules.RSA === 'ON' && player.hands.length < this.rules.HANDS);
167 }
168 }
169
170 settle(dealerNatural = false) {
171 const dealerTotal = total(this.dealer).value;
172 for (const player of this.participants) {
173 player.hands.forEach((hand, index) => {
174 let result;
175 if (dealerNatural) result = blackjack(hand.cards, hand.split) ? 'push' : 'lose';
176 else if (blackjack(hand.cards, hand.split)) result = 'blackjack';
177 else if (hand.reason) result = hand.reason;
178 else {
179 const value = total(hand.cards).value;
180 result = dealerTotal > 21 || value > dealerTotal ? 'win'
181 : value === dealerTotal ? 'push' : 'lose';
182 }
183 const bet = hand.bet;
184 let profit;
185 if (result === 'blackjack') {
186 profit = this.rules.BJ === '3:2' ? Math.floor(bet * 3 / 2) : Math.floor(bet * 6 / 5);
187 } else if (result === 'win' || result === 'charlie') profit = bet;
188 else if (result === 'push') profit = 0;
189 else if (result === 'surrender') profit = Math.floor(bet / 2) - bet;
190 else profit = -bet;
191 player.bankroll += bet + profit;
192 player.net += profit;
193 player.stats.hands++;
194 if (['blackjack', 'charlie', 'win'].includes(result)) player.stats.wins++;
195 else if (result === 'push') player.stats.pushes++;
196 else player.stats.losses++;
197 if (result === 'blackjack') player.stats.blackjacks++;
198 if (result === 'bust') player.stats.busts++;
199 if (result === 'surrender') player.stats.surrenders++;
200 this.emit(`${player.handName(index)} ${result} ${signed(profit)}`);
201 });
202 this.emit(`${player.name} bankroll ${player.bankroll}`);
203 }
204 for (const player of this.players()) {
205 player.bet = 0;
206 player.side = 0;
207 }
208 this.state = 'idle';
209 }
210
211 dealerTurn() {
212 this.revealDealer();
213 const mustPlay = this.participants.some(player => player.hands.some(hand =>
214 !['bust', 'surrender', 'charlie'].includes(hand.reason) && !blackjack(hand.cards, hand.split)));
215 let drew = false;
216 if (mustPlay) {
217 while (true) {
218 const { value, soft } = total(this.dealer);
219 if (!(value <= 16 || value === 17 && soft && this.rules.DEALER === 'H17')) break;
220 this.dealer.push(this.draw());
221 drew = true;
222 this.emit(`dealer hits: ${handText(this.dealer)}`);
223 }
224 }
225 if (drew && total(this.dealer).value > 21) this.dealerBusts++;
226 this.settle();
227 }
228
229 advance() {
230 while (true) {
231 const turn = this.current();
232 if (!turn) {
233 this.dealerTurn();
234 return;
235 }
236 const { player, index, hand } = turn;
237 if (hand.cards.length !== 1) return;
238 hand.cards.push(this.draw());
239 this.emit(`${player.handName(index)}: ${handText(hand.cards, hand.split)}`);
240 this.finishCheck(player, hand);
241 if (!hand.finished) return;
242 }
243 }
244
245 peek() {
246 const natural = blackjack(this.dealer);
247 this.emit(`dealer peeks: ${natural ? 'blackjack' : 'no blackjack'}`);
248 for (const player of this.participants) {
249 if (player.insurance === 0) continue;
250 const profit = player.insurance * (natural ? 2 : -1);
251 player.bankroll += player.insurance + profit;
252 player.net += profit;
253 this.emit(`${player.name} insurance ${signed(profit)}`);
254 }
255 if (natural) {
256 this.dealerBlackjacks++;
257 this.revealDealer();
258 this.settle(true);
259 } else {
260 this.state = 'play';
261 this.advance();
262 }
263 }
264
265 deal() {
266 if (this.drawn >= this.cut()) {
267 this.shuffle();
268 this.emit('shuffle');
269 }
270 this.emit(`round ${++this.round}`);
271 this.participants = this.players().filter(player => player.bet !== 0);
272 for (const player of this.participants) {
273 player.bankroll -= player.bet + player.side;
274 player.insurance = 0;
275 player.stats.rounds++;
276 player.hands = [makeHand([this.draw()], player.bet)];
277 }
278 this.dealer = [this.draw()];
279 for (const player of this.participants) {
280 const hand = player.hands[0];
281 hand.cards.push(this.draw());
282 hand.finished = blackjack(hand.cards);
283 }
284 this.dealer.push(this.draw(false));
285 this.holeRevealed = false;
286 for (const player of this.participants) {
287 this.emit(`${player.handName(0)}: ${handText(player.hands[0].cards)}`);
288 }
289 this.emit(`dealer: ${cardText(this.dealer[0])} ??`);
290 for (const player of this.participants) {
291 if (player.side === 0) continue;
292 const [a, b] = player.hands[0].cards;
293 let result = 'lose', multiplier = -1;
294 if (rank(a) === rank(b)) {
295 if (suit(a) === suit(b)) { result = 'perfect'; multiplier = 25; }
296 else if ([0, 3].includes(suit(a)) === [0, 3].includes(suit(b))) {
297 result = 'colored'; multiplier = 12;
298 } else { result = 'mixed'; multiplier = 6; }
299 }
300 const profit = player.side * multiplier;
301 player.bankroll += player.side + profit;
302 player.net += profit;
303 this.emit(`${player.name} side ${result} ${signed(profit)}`);
304 }
305 if (points(this.dealer[0]) === 1) {
306 this.state = 'insurance';
307 this.insuranceTurn = 0;
308 this.emit('insurance offered');
309 } else if (points(this.dealer[0]) === 10) this.peek();
310 else {
311 this.state = 'play';
312 this.advance();
313 }
314 }
315
316 actionError(action, player, hand) {
317 if (action === 'HIT') return hand.splitAces ? 'split aces' : null;
318 if (action === 'DOUBLE') {
319 if (hand.splitAces) return 'split aces';
320 if (hand.cards.length !== 2) return 'two cards only';
321 const { value, soft } = total(hand.cards);
322 if (this.rules.DOUBLE !== 'ANY'
323 && (soft || value < (this.rules.DOUBLE === '9-11' ? 9 : 10) || value > 11)
324 || this.rules.DAS === 'OFF' && hand.split) return 'double not allowed';
325 if (player.bankroll < hand.bet) return 'insufficient funds';
326 } else if (action === 'SPLIT') {
327 if (hand.cards.length !== 2) return 'two cards only';
328 if (points(hand.cards[0]) !== points(hand.cards[1])) return 'not a pair';
329 if (player.hands.length >= this.rules.HANDS) return 'hand limit';
330 if (player.bankroll < hand.bet) return 'insufficient funds';
331 } else if (action === 'SURRENDER') {
332 if (this.rules.SURRENDER === 'OFF' || hand.split) return 'surrender not allowed';
333 if (hand.cards.length !== 2) return 'two cards only';
334 }
335 return null;
336 }
337
338 action(action) {
339 const { player, index, hand } = this.current();
340 const name = player.handName(index);
341 if (action === 'SPLIT') {
342 const [first, second] = hand.cards;
343 player.bankroll -= hand.bet;
344 player.stats.splits++;
345 this.emit(`${name} splits ${cardText(first)} ${cardText(second)}`);
346 const aces = hand.splitAces || points(first) === 1;
347 const replacement = makeHand([first], hand.bet, true, aces);
348 player.hands.splice(index, 1, replacement, makeHand([second], hand.bet, true, aces));
349 replacement.cards.push(this.draw());
350 this.emit(`${name}: ${handText(replacement.cards, true)}`);
351 this.finishCheck(player, replacement);
352 } else if (action === 'STAND' || action === 'SURRENDER') {
353 hand.finished = true;
354 if (action === 'SURRENDER') hand.reason = 'surrender';
355 this.emit(`${name} ${action === 'STAND' ? 'stands' : 'surrenders'}: ${handText(hand.cards, hand.split)}`);
356 } else {
357 if (action === 'DOUBLE') {
358 player.bankroll -= hand.bet;
359 hand.bet *= 2;
360 player.stats.doubles++;
361 }
362 hand.cards.push(this.draw());
363 this.emit(`${name} ${action === 'DOUBLE' ? 'doubles' : 'hits'}: ${handText(hand.cards, hand.split)}`);
364 this.finishCheck(player, hand);
365 if (action === 'DOUBLE') hand.finished = true;
366 }
367 this.advance();
368 }

370 recommend() {
371 const { player, hand } = this.current();
372 const possible = action => this.actionError(action, player, hand) === null;
373 const upcard = points(this.dealer[0]);
374 const u = upcard === 1 ? 11 : upcard;
375 if (hand.splitAces) return possible('SPLIT') ? 'SPLIT' : 'STAND';
376 if (hand.cards.length === 2 && points(hand.cards[0]) === points(hand.cards[1]) && possible('SPLIT')) {
377 const v = points(hand.cards[0]);
378 if (v === 1 || v === 8 || v === 9 && ![7, 10, 11].includes(u)
379 || [2, 3, 7].includes(v) && u <= 7 || v === 6 && u <= 6 || v === 4 && (u === 5 || u === 6)) {
380 return 'SPLIT';
381 }
382 }
383 const { value: t, soft } = total(hand.cards);
384 let symbol = 'H';
385 if (soft) {
386 if (t >= 19) symbol = 'S';
387 else if (t === 18) symbol = u >= 3 && u <= 6 ? 'Ds' : [2, 7, 8].includes(u) ? 'S' : 'H';
388 else if (t === 17 && u >= 3 && u <= 6 || (t === 15 || t === 16) && u >= 4 && u <= 6
389 || (t === 13 || t === 14) && (u === 5 || u === 6)) symbol = 'D';
390 } else {
391 if (t >= 17) symbol = 'S';
392 else if (t >= 13) {
393 symbol = u <= 6 ? 'S' : t === 16 && u >= 9 || t === 15 && u === 10 ? 'R' : 'H';
394 } else if (t === 12) symbol = u >= 4 && u <= 6 ? 'S' : 'H';
395 else if (t === 11 && u <= 10 || t === 10 && u <= 9 || t === 9 && u >= 3 && u <= 6) symbol = 'D';
396 }
397 if (symbol === 'S') return 'STAND';
398 if (symbol === 'D' || symbol === 'Ds') return possible('DOUBLE') ? 'DOUBLE' : symbol === 'Ds' ? 'STAND' : 'HIT';
399 if (symbol === 'R') return possible('SURRENDER') ? 'SURRENDER' : 'HIT';
400 return 'HIT';
401 }

403 status() {
404 let line = `table ${this.state} round ${this.round}`;
405 if (this.state === 'insurance') line += ` next ${this.participants[this.insuranceTurn].name}`;
406 else if (this.state === 'play') {
407 const { player, index } = this.current();
408 line += ` turn ${player.handName(index)}`;
409 }
410 this.emit(line);
411 if (this.state !== 'idle') this.emit(`dealer: ${cardText(this.dealer[0])} ??`);
412 for (const player of this.players()) {
413 this.emit(`seat ${player.seat} ${player.name} bankroll ${player.bankroll} bet ${player.bet} side ${player.side}`);
414 if (this.state !== 'idle' && this.participants.includes(player)) {
415 player.hands.forEach((hand, index) => {
416 this.emit(`${player.handName(index)}: ${handText(hand.cards, hand.split)} bet ${hand.bet}`);
417 });
418 }
419 }
420 }

422 command(line) {
423 const [command, first, second] = line.split(' ');
424 if (!COMMANDS.has(command)) fail(`unknown command ${command}`);
425
426 let player, value, seed;
427 if (command === 'SHOE') {
428 value = integer(first, 1, 8);
429 seed = integer(second, 1, 2147483646);
430 } else if (command === 'RULE') {
431 const definition = RULE_DEFINITIONS.find(([name]) => name === first);
432 if (!definition) fail(`unknown rule ${first}`);
433 const allowed = definition[2];
434 if (typeof allowed[0] === 'number') value = integer(second, allowed[0], allowed[1]);
435 else {
436 if (!allowed.includes(second)) fail(`bad value ${second}`);
437 value = second;
438 }
439 } else if (command === 'PLAYER') {
440 if (!/^[A-Z]{1,8}$/.test(first)) fail(`bad name ${first}`);
441 value = integer(second, 1, 1000000);
442 } else if (['LEAVE', 'REBUY', 'BET', 'SIDE', 'INSURANCE', 'AUTO'].includes(command) || ACTIONS.has(command)) {
443 player = this.player(first);
444 if (['REBUY', 'BET', 'SIDE', 'INSURANCE'].includes(command)) {
445 value = integer(second, command === 'REBUY' ? 1 : 0, 1000000);
446 }
447 }
448 if (['SHOE', 'RULE', 'PLAYER', 'LEAVE', 'REBUY', 'BET', 'SIDE', 'DEAL'].includes(command)) {
449 if (this.state !== 'idle') fail('round in progress');
450 } else if (command === 'INSURANCE') {
451 if (this.state !== 'insurance') fail('no insurance');
452 } else if (ACTIONS.has(command) || command === 'AUTO' || command === 'HINT') {
453 if (this.state === 'idle') fail('no round');
454 if (this.state === 'insurance') fail('insurance pending');
455 }
456
457 if (ACTIONS.has(command) || command === 'AUTO') {
458 const turn = this.current();
459 if (turn.player !== player) fail(`waiting for ${turn.player.handName(turn.index)}`);
460 if (command === 'AUTO') {
461 while (this.state === 'play' && this.current().player === player) this.action(this.recommend());
462 } else {
463 const error = this.actionError(command, player, turn.hand);
464 if (error) fail(error);
465 this.action(command);
466 }
467 return;
468 }
469 switch (command) {
470 case 'SHOE':
471 this.decks = value;
472 this.random = seed;
473 this.shuffle();
474 this.emit(`shoe ${value} decks ${this.shoe.length} cards cut ${this.cut()}`);
475 break;
476 case 'RULE':
477 if ((first === 'MIN' ? value : this.rules.MIN) > (first === 'MAX' ? value : this.rules.MAX)) fail('min above max');
478 this.rules[first] = value;
479 this.emit(`rule ${first} ${value}`);
480 break;
481 case 'RULES':
482 this.emit('rules ' + RULE_DEFINITIONS.map(([name]) => `${name} ${this.rules[name]}`).join(' '));
483 break;
484 case 'PLAYER': {
485 if (this.seats.some(player => player !== null && player.name === first)) fail('player exists');
486 const seat = this.seats.indexOf(null);
487 if (seat === -1) fail('table full');
488 this.seats[seat] = new Player(first, seat + 1, value);
489 this.emit(`${first} sits at seat ${seat + 1} with ${value}`);
490 break;
491 }
492 case 'LEAVE':
493 this.emit(`${player.name} leaves with ${player.bankroll} net ${signed(player.net)}`);
494 this.seats[player.seat - 1] = null;
495 break;
496 case 'REBUY':
497 player.bankroll += value;
498 this.emit(`${player.name} rebuys ${value} bankroll ${player.bankroll}`);
499 break;
500 case 'BET':
501 if (value !== 0) {
502 if (value < this.rules.MIN || value > this.rules.MAX) fail('bet out of limits');
503 if (value + player.side > player.bankroll) fail('insufficient funds');
504 }
505 player.bet = value;
506 if (value === 0) player.side = 0;
507 this.emit(value === 0 ? `${player.name} bet cleared` : `${player.name} bets ${value}`);
508 break;
509 case 'SIDE':
510 if (value !== 0) {
511 if (player.bet === 0) fail('no bet');
512 if (value > this.rules.SIDEMAX) fail('side out of limits');
513 if (player.bet + value > player.bankroll) fail('insufficient funds');
514 }
515 player.side = value;
516 this.emit(value === 0 ? `${player.name} side cleared` : `${player.name} side ${value}`);
517 break;
518 case 'DEAL':
519 if (this.decks === 0) fail('no shoe');
520 if (!this.players().some(player => player.bet !== 0)) fail('no bets');
521 this.deal();
522 break;
523 case 'INSURANCE':
524 if (this.participants[this.insuranceTurn] !== player) fail(`waiting for ${this.participants[this.insuranceTurn].name}`);
525 if (value > Math.floor(player.bet / 2)) fail('insurance out of limits');
526 if (value > player.bankroll) fail('insufficient funds');
527 player.bankroll -= value;
528 player.insurance = value;
529 this.emit(value === 0 ? `${player.name} declines insurance` : `${player.name} insures ${value}`);
530 if (++this.insuranceTurn === this.participants.length) this.peek();
531 break;
532 case 'HINT': {
533 const { player, index } = this.current();
534 this.emit(`hint ${player.handName(index)}: ${this.recommend().toLowerCase()}`);
535 break;
536 }
537 case 'STATUS':
538 this.status();
539 break;
540 case 'COUNT': {
541 if (this.decks === 0) fail('no shoe');
542 const remaining = this.shoe.length - this.drawn;
543 const hundredths = Math.floor(Math.abs(this.running) * 5200 / remaining);
544 const count = `${this.running < 0 ? '-' : ''}${Math.floor(hundredths / 100)}.${String(hundredths % 100).padStart(2, '0')}`;
545 this.emit(`count running ${this.running} true ${count} remaining ${remaining}`);
546 break;
547 }
548 case 'STATS':
549 if (this.players().length === 0) fail('no players');
550 for (const player of this.players()) {
551 this.emit(`stats ${player.name} ${STAT_NAMES.map(name => `${name} ${player.stats[name]}`).join(' ')} net ${signed(player.net)}`);
552 }
553 this.emit(`stats dealer rounds ${this.round} blackjacks ${this.dealerBlackjacks} busts ${this.dealerBusts}`);
554 break;
555 }
556 }
557}

560const table = new Table();
561const count = Number(lines[0]);
562for (let i = 1; i <= count; ++i) {
563 try {
564 table.command(lines[i]);
565 } catch (error) {
566 if (!(error instanceof CommandError)) throw error;
567 table.emit(`error: ${error.message}`);
