8func integer(_ word: String, _ range: ClosedRange<Int>) throws -> Int {
9 let bytes = Array(word.utf8)
10 guard !bytes.isEmpty, bytes.count <= 10,
11 bytes.count == 1 || bytes[0] != 48,
12 bytes.allSatisfy({ (48...57).contains($0) }),
13 let value = Int64(word), value >= Int64(range.lowerBound), value <= Int64(range.upperBound)
14 else { throw CommandError("bad value \(word)") }
15 return Int(value)
16}

18func signed(_ value: Int) -> String { (value >= 0 ? "+" : "") + String(value) }
19func onOff(_ value: Bool) -> String { value ? "ON" : "OFF" }
20
21func switchValue(_ word: String) throws -> Bool {
22 guard word == "ON" || word == "OFF" else { throw CommandError("bad value \(word)") }
23 return word == "ON"
24}

26struct Card {
27 static let ranks = Array("A23456789TJQK")
28 static let suits = Array("CDHS")
29 let code: Int
30 var rank: Int { code % 13 }
31 var suit: Int { code / 13 }
32 var points: Int { min(rank + 1, 10) }
33 var isBlack: Bool { suit == 0 || suit == 3 }
34 var text: String { String(Card.ranks[rank]) + String(Card.suits[suit]) }
35}
36
37func total(_ cards: [Card]) -> (value: Int, soft: Bool) {
38 let sum = cards.reduce(0) { $0 + $1.points }
39 let soft = cards.contains(where: { $0.rank == 0 }) && sum + 10 <= 21
40 return (sum + (soft ? 10 : 0), soft)
41}
42
43func blackjack(_ cards: [Card], split: Bool = false) -> Bool {
44 !split && cards.count == 2 && total(cards).value == 21
45}
46
47func handText(_ cards: [Card], split: Bool = false) -> String {
48 let t = total(cards)
49 let status: String
50 if blackjack(cards, split: split) { status = "blackjack" }
51 else if t.value > 21 { status = "bust \(t.value)" }
52 else if t.soft { status = "soft \(t.value)" }
53 else { status = String(t.value) }
54 return cards.map(\.text).joined(separator: " ") + " (\(status))"
55}

57enum Phase: String { case idle, insurance, play }
58enum Finish: String { case bust, surrender, charlie }
59enum Outcome: String { case blackjack, win, charlie, push, lose, bust, surrender }
60enum DealerRule: String { case s17 = "S17", h17 = "H17" }
61enum Payout: String { case threeToTwo = "3:2", sixToFive = "6:5" }
62enum DoubleRule: String { case any = "ANY", hard9 = "9-11", hard10 = "10-11" }
63enum Action: String {
64 case hit = "HIT", stand = "STAND", double = "DOUBLE", split = "SPLIT", surrender = "SURRENDER"
65}

67struct Rules {
68 var dealer = DealerRule.s17
69 var penetration = 75
70 var minimum = 10
71 var maximum = 500
72 var payout = Payout.threeToTwo
73 var doubleAfterSplit = true
74 var doubling = DoubleRule.any
75 var hands = 4
76 var resplitAces = false
77 var surrender = true
78 var charlie = 0
79 var sideMaximum = 25
80
81 func changed(_ name: String, to word: String) throws -> Rules {
82 var next = self
83 switch name {
84 case "DEALER":
85 guard let value = DealerRule(rawValue: word) else { throw CommandError("bad value \(word)") }
86 next.dealer = value
87 case "PEN": next.penetration = try integer(word, 50...90)
88 case "MIN": next.minimum = try integer(word, 1...10000)
89 case "MAX": next.maximum = try integer(word, 1...10000)
90 case "BJ":
91 guard let value = Payout(rawValue: word) else { throw CommandError("bad value \(word)") }
92 next.payout = value
93 case "DAS": next.doubleAfterSplit = try switchValue(word)
94 case "DOUBLE":
95 guard let value = DoubleRule(rawValue: word) else { throw CommandError("bad value \(word)") }
96 next.doubling = value
97 case "HANDS": next.hands = try integer(word, 1...4)
98 case "RSA": next.resplitAces = try switchValue(word)
99 case "SURRENDER": next.surrender = try switchValue(word)
100 case "CHARLIE":
101 guard ["0", "5", "6", "7"].contains(word) else { throw CommandError("bad value \(word)") }
102 next.charlie = Int(word)!
103 case "SIDEMAX": next.sideMaximum = try integer(word, 0...1000)
104 default: throw CommandError("unknown rule \(name)")
105 }
106 return next
107 }
108
109 var text: String {
110 "rules DEALER \(dealer.rawValue) PEN \(penetration) MIN \(minimum) MAX \(maximum)"
111 + " BJ \(payout.rawValue) DAS \(onOff(doubleAfterSplit)) DOUBLE \(doubling.rawValue)"
112 + " HANDS \(hands) RSA \(onOff(resplitAces)) SURRENDER \(onOff(surrender))"
113 + " CHARLIE \(charlie) SIDEMAX \(sideMaximum)"
114 }
115}

165final class Table {
166 static let commands: Set<String> = [
167 "SHOE", "RULE", "RULES", "PLAYER", "LEAVE", "REBUY", "BET", "SIDE", "DEAL", "INSURANCE",
168 "HIT", "STAND", "DOUBLE", "SPLIT", "SURRENDER", "HINT", "AUTO", "STATUS", "COUNT", "STATS"
169 ]
170 var rules = Rules()
171 var seats: [Player?] = Array(repeating: nil, count: 7)
172 var participants: [Player] = []
173 var phase = Phase.idle
174 var round = 0, dealerBlackjacks = 0, dealerBusts = 0
175 var decks = 0, drawn = 0, running = 0
176 var random: Int64 = 1
177 var shoe: [Card] = []
178 var dealer: [Card] = []
179 var holeRevealed = false
180 var insuranceTurn = 0
181 var output: [String] = []
182
183 var players: [Player] { seats.compactMap { $0 } }
184 var cut: Int { shoe.count * rules.penetration / 100 }
185 func emit(_ line: String) { output.append(line) }

187 func player(named name: String) throws -> Player {

194 func shuffle() {
195 shoe = (0..<(52 * decks)).map { Card(code: $0 % 52) }
196 for i in stride(from: shoe.count - 1, through: 1, by: -1) {
197 random = random * 48271 % 2147483647
198 shoe.swapAt(i, Int(random % Int64(i + 1)))
199 }
200 drawn = 0
201 running = 0
202 }
203
204 func countCard(_ card: Card) {
205 if (2...6).contains(card.points) { running += 1 }
206 else if card.points == 1 || card.points == 10 { running -= 1 }
207 }
208
209 func draw(faceUp: Bool = true) -> Card {
210 let card = shoe[drawn]
211 drawn += 1
212 if faceUp { countCard(card) }
213 return card
214 }

216 func revealDealer() {
217 if !holeRevealed {
218 countCard(dealer[1])
219 holeRevealed = true
220 }
221 emit("dealer: \(handText(dealer))")

224 func current() -> (player: Player, index: Int, hand: Hand)? {
225 for player in participants {
226 if let index = player.hands.firstIndex(where: { !$0.finished }) {
227 return (player, index, player.hands[index])
228 }
229 }
230 return nil

233 func finishCheck(_ hand: Hand, player: Player) {
234 let value = total(hand.cards).value
235 if value > 21 {
236 hand.reason = .bust
237 hand.finished = true
238 } else if rules.charlie != 0 && hand.cards.count == rules.charlie {
239 hand.reason = .charlie
240 hand.finished = true
241 } else if value == 21 {
242 hand.finished = true
243 } else if hand.splitAces && hand.cards.count == 2 {
244 hand.finished = !(hand.cards.allSatisfy { $0.rank == 0 }
245 && rules.resplitAces && player.hands.count < rules.hands)
246 }
247 }

249 func settle(dealerNatural: Bool = false) {
250 let dealerTotal = total(dealer).value
251 for player in participants {
252 for (index, hand) in player.hands.enumerated() {
253 let result: Outcome
254 if dealerNatural { result = hand.natural ? .push : .lose }
255 else if hand.natural { result = .blackjack }
256 else if let reason = hand.reason { result = Outcome(rawValue: reason.rawValue)! }
257 else {
258 let value = total(hand.cards).value
259 result = dealerTotal > 21 || value > dealerTotal ? .win : value == dealerTotal ? .push : .lose
260 }
261 let profit: Int
262 switch result {
263 case .blackjack: profit = rules.payout == .threeToTwo ? hand.bet * 3 / 2 : hand.bet * 6 / 5
264 case .win, .charlie: profit = hand.bet
265 case .push: profit = 0
266 case .surrender: profit = hand.bet / 2 - hand.bet
267 case .lose, .bust: profit = -hand.bet
268 }
269 player.bankroll += hand.bet + profit
270 player.net += profit
271 player.stats.hands += 1
272 if result == .blackjack || result == .win || result == .charlie { player.stats.wins += 1 }
273 else if result == .push { player.stats.pushes += 1 }
274 else { player.stats.losses += 1 }
275 if result == .blackjack { player.stats.blackjacks += 1 }
276 if result == .bust { player.stats.busts += 1 }
277 if result == .surrender { player.stats.surrenders += 1 }
278 emit("\(player.handName(index)) \(result.rawValue) \(signed(profit))")
279 }
280 emit("\(player.name) bankroll \(player.bankroll)")
281 }
282 for player in players {
283 player.bet = 0
284 player.side = 0
285 }
286 phase = .idle
287 }

289 func dealerTurn() {
290 revealDealer()
291 let mustPlay = participants.contains { player in
292 player.hands.contains { $0.reason == nil && !$0.natural }
293 }
294 var drew = false
295 if mustPlay {
296 while true {
297 let t = total(dealer)
298 if !(t.value <= 16 || t.value == 17 && t.soft && rules.dealer == .h17) { break }
299 let card = draw()
300 dealer.append(card)
301 drew = true
302 emit("dealer hits: \(handText(dealer))")
303 }
304 }
305 if drew && total(dealer).value > 21 { dealerBusts += 1 }
306 settle()
307 }

309 func advance() {
310 while let turn = current() {
311 if turn.hand.cards.count != 1 { return }
312 let card = draw()
313 turn.hand.cards.append(card)
314 emit("\(turn.player.handName(turn.index)): \(turn.hand.text)")
315 finishCheck(turn.hand, player: turn.player)
316 if !turn.hand.finished { return }
317 }
318 dealerTurn()
319 }

321 func peek() {
322 let natural = blackjack(dealer)
323 emit("dealer peeks: \(natural ? "blackjack" : "no blackjack")")
324 for player in participants where player.insurance != 0 {
325 let profit = player.insurance * (natural ? 2 : -1)
326 player.bankroll += player.insurance + profit
327 player.net += profit
328 emit("\(player.name) insurance \(signed(profit))")
329 }
330 if natural {
331 dealerBlackjacks += 1
332 revealDealer()
333 settle(dealerNatural: true)
334 } else {
335 phase = .play
336 advance()
337 }
338 }

340 func deal() {
341 if drawn >= cut {
342 shuffle()
343 emit("shuffle")
344 }
345 round += 1
346 emit("round \(round)")
347 participants = players.filter { $0.bet != 0 }
348 for player in participants {
349 player.bankroll -= player.bet + player.side
350 player.insurance = 0
351 player.stats.rounds += 1
352 player.hands = [Hand([draw()], bet: player.bet)]
353 }
354 dealer = [draw()]
355 for player in participants {
356 let hand = player.hands[0]
357 let card = draw()
358 hand.cards.append(card)
359 hand.finished = hand.natural
360 }
361 let hole = draw(faceUp: false)
362 dealer.append(hole)
363 holeRevealed = false
364 for player in participants { emit("\(player.handName(0)): \(player.hands[0].text)") }
365 emit("dealer: \(dealer[0].text) ??")
366 for player in participants where player.side != 0 {
367 let cards = player.hands[0].cards
368 let a = cards[0], b = cards[1]
369 let result: String, multiplier: Int
370 if a.rank != b.rank { result = "lose"; multiplier = -1 }
371 else if a.suit == b.suit { result = "perfect"; multiplier = 25 }
372 else if a.isBlack == b.isBlack { result = "colored"; multiplier = 12 }
373 else { result = "mixed"; multiplier = 6 }
374 let profit = player.side * multiplier
375 player.bankroll += player.side + profit
376 player.net += profit
377 emit("\(player.name) side \(result) \(signed(profit))")
378 }
379 if dealer[0].points == 1 {
380 phase = .insurance
381 insuranceTurn = 0
382 emit("insurance offered")
383 } else if dealer[0].points == 10 { peek() }
384 else {
385 phase = .play
386 advance()
387 }
388 }

390 func actionError(_ action: Action, player: Player, hand: Hand) -> String? {
391 switch action {
392 case .hit:
393 if hand.splitAces { return "split aces" }
394 case .double:
395 if hand.splitAces { return "split aces" }
396 if hand.cards.count != 2 { return "two cards only" }
397 let t = total(hand.cards)
398 if rules.doubling != .any && (t.soft || t.value < (rules.doubling == .hard9 ? 9 : 10) || t.value > 11)
399 || !rules.doubleAfterSplit && hand.split { return "double not allowed" }
400 if player.bankroll < hand.bet { return "insufficient funds" }
401 case .split:
402 if hand.cards.count != 2 { return "two cards only" }
403 if hand.cards[0].points != hand.cards[1].points { return "not a pair" }
404 if player.hands.count >= rules.hands { return "hand limit" }
405 if player.bankroll < hand.bet { return "insufficient funds" }
406 case .surrender:
407 if !rules.surrender || hand.split { return "surrender not allowed" }
408 if hand.cards.count != 2 { return "two cards only" }
409 case .stand: break
410 }
411 return nil
412 }

414 func perform(_ action: Action) {
415 let turn = current()!
416 let player = turn.player, hand = turn.hand
417 let name = player.handName(turn.index)
418 switch action {
419 case .split:
420 let first = hand.cards[0], second = hand.cards[1]
421 let aces = hand.splitAces || first.points == 1
422 player.bankroll -= hand.bet
423 player.stats.splits += 1
424 emit("\(name) splits \(first.text) \(second.text)")
425 let replacement = Hand([first], bet: hand.bet, split: true, splitAces: aces)
426 player.hands[turn.index] = replacement
427 player.hands.insert(Hand([second], bet: hand.bet, split: true, splitAces: aces), at: turn.index + 1)
428 let card = draw()
429 replacement.cards.append(card)
430 emit("\(name): \(replacement.text)")
431 finishCheck(replacement, player: player)
432 case .stand, .surrender:
433 hand.finished = true
434 if action == .surrender { hand.reason = .surrender }
435 emit("\(name) \(action == .stand ? "stands" : "surrenders"): \(hand.text)")
436 case .hit, .double:
437 if action == .double {
438 player.bankroll -= hand.bet
439 hand.bet *= 2
440 player.stats.doubles += 1
441 }
442 let card = draw()
443 hand.cards.append(card)
444 emit("\(name) \(action == .double ? "doubles" : "hits"): \(hand.text)")
445 finishCheck(hand, player: player)
446 if action == .double { hand.finished = true }
447 }
448 advance()
449 }

451 func recommend() -> Action {
452 let turn = current()!
453 let player = turn.player, hand = turn.hand
454 func possible(_ action: Action) -> Bool { actionError(action, player: player, hand: hand) == nil }
455 let u = dealer[0].points == 1 ? 11 : dealer[0].points
456 if hand.splitAces { return possible(.split) ? .split : .stand }
457 if hand.cards.count == 2 && hand.cards[0].points == hand.cards[1].points && possible(.split) {
458 let v = hand.cards[0].points
459 if v == 1 || v == 8 || v == 9 && ![7, 10, 11].contains(u)
460 || [2, 3, 7].contains(v) && u <= 7 || v == 6 && u <= 6 || v == 4 && (u == 5 || u == 6) {
461 return .split
462 }
463 }
464 let t = total(hand.cards)
465 enum Symbol { case h, s, d, ds, r }
466 var symbol = Symbol.h
467 if t.soft {
468 if t.value >= 19 { symbol = .s }
469 else if t.value == 18 { symbol = (3...6).contains(u) ? .ds : [2, 7, 8].contains(u) ? .s : .h }
470 else if t.value == 17 && (3...6).contains(u)
471 || [15, 16].contains(t.value) && (4...6).contains(u)
472 || [13, 14].contains(t.value) && [5, 6].contains(u) { symbol = .d }
473 } else {
474 if t.value >= 17 { symbol = .s }
475 else if t.value >= 13 {
476 symbol = u <= 6 ? .s : t.value == 16 && u >= 9 || t.value == 15 && u == 10 ? .r : .h
477 } else if t.value == 12 { symbol = (4...6).contains(u) ? .s : .h }
478 else if t.value == 11 && u <= 10 || t.value == 10 && u <= 9 || t.value == 9 && (3...6).contains(u) {
479 symbol = .d
480 }
481 }
482 switch symbol {
483 case .s: return .stand
484 case .h: return .hit
485 case .d: return possible(.double) ? .double : .hit
486 case .ds: return possible(.double) ? .double : .stand
487 case .r: return possible(.surrender) ? .surrender : .hit
488 }
489 }

494 else if phase == .play {
495 let turn = current()!
496 line += " turn \(turn.player.handName(turn.index))"

499 if phase != .idle { emit("dealer: \(dealer[0].text) ??") }

501 emit("seat \(player.seat) \(player.name) bankroll \(player.bankroll) bet \(player.bet) side \(player.side)")

504 emit("\(player.handName(index)): \(hand.text) bet \(hand.bet)")
505 }
506 }
507 }

510 func command(_ line: String) throws {
511 let words = line.split(separator: " ").map(String.init)
512 let command = words[0]
513 guard Table.commands.contains(command) else { throw CommandError("unknown command \(command)") }
514 let first = words.count > 1 ? words[1] : ""
515 let second = words.count > 2 ? words[2] : ""
516 let action = Action(rawValue: command)
517 var target: Player?
518 var value = 0
519 var seed: Int64 = 1
520 var proposedRules: Rules?
521
522 if command == "SHOE" {
523 value = try integer(first, 1...8)
524 seed = Int64(try integer(second, 1...2147483646))
525 } else if command == "RULE" { proposedRules = try rules.changed(first, to: second) }
526 else if command == "PLAYER" {
527 let bytes = Array(first.utf8)
528 guard (1...8).contains(bytes.count), bytes.allSatisfy({ (65...90).contains($0) }) else {
529 throw CommandError("bad name \(first)")
530 }
531 value = try integer(second, 1...1000000)
532 } else if action != nil || ["LEAVE", "REBUY", "BET", "SIDE", "INSURANCE", "AUTO"].contains(command) {
533 target = try player(named: first)
534 if ["REBUY", "BET", "SIDE", "INSURANCE"].contains(command) {
535 value = try integer(second, (command == "REBUY" ? 1 : 0)...1000000)
536 }
537 }
538 if ["SHOE", "RULE", "PLAYER", "LEAVE", "REBUY", "BET", "SIDE", "DEAL"].contains(command) {
539 guard phase == .idle else { throw CommandError("round in progress") }
540 } else if command == "INSURANCE" {
541 guard phase == .insurance else { throw CommandError("no insurance") }
542 } else if action != nil || command == "AUTO" || command == "HINT" {
543 if phase == .idle { throw CommandError("no round") }
544 if phase == .insurance { throw CommandError("insurance pending") }
545 }
546
547 if action != nil || command == "AUTO" {
548 let turn = current()!, player = target!
549 guard turn.player === player else { throw CommandError("waiting for \(turn.player.handName(turn.index))") }
550 if let action {
551 if let error = actionError(action, player: player, hand: turn.hand) { throw CommandError(error) }
552 perform(action)
553 } else {
554 while phase == .play && current()!.player === player { perform(recommend()) }
555 }
556 return
557 }
558 switch command {
559 case "SHOE":
560 decks = value
561 random = seed
562 shuffle()
563 emit("shoe \(decks) decks \(shoe.count) cards cut \(cut)")
564 case "RULE":
565 let next = proposedRules!
566 guard next.minimum <= next.maximum else { throw CommandError("min above max") }
567 rules = next
568 emit("rule \(first) \(second)")
569 case "RULES": emit(rules.text)
570 case "PLAYER":
571 if players.contains(where: { $0.name == first }) { throw CommandError("player exists") }
572 guard let seat = seats.firstIndex(where: { $0 == nil }) else { throw CommandError("table full") }
573 seats[seat] = Player(name: first, seat: seat + 1, bankroll: value)
574 emit("\(first) sits at seat \(seat + 1) with \(value)")
575 case "LEAVE":
576 let player = target!
577 emit("\(player.name) leaves with \(player.bankroll) net \(signed(player.net))")
578 seats[player.seat - 1] = nil
579 case "REBUY":
580 let player = target!
581 player.bankroll += value
582 emit("\(player.name) rebuys \(value) bankroll \(player.bankroll)")
583 case "BET":
584 let player = target!
585 if value != 0 {
586 guard (rules.minimum...rules.maximum).contains(value) else { throw CommandError("bet out of limits") }
587 guard value + player.side <= player.bankroll else { throw CommandError("insufficient funds") }
588 }
589 player.bet = value
590 if value == 0 { player.side = 0 }
591 emit(value == 0 ? "\(player.name) bet cleared" : "\(player.name) bets \(value)")
592 case "SIDE":
593 let player = target!
594 if value != 0 {
595 if player.bet == 0 { throw CommandError("no bet") }
596 if value > rules.sideMaximum { throw CommandError("side out of limits") }
597 if player.bet + value > player.bankroll { throw CommandError("insufficient funds") }
598 }
599 player.side = value
600 emit(value == 0 ? "\(player.name) side cleared" : "\(player.name) side \(value)")
601 case "DEAL":
602 if decks == 0 { throw CommandError("no shoe") }
603 guard players.contains(where: { $0.bet != 0 }) else { throw CommandError("no bets") }
604 deal()
605 case "INSURANCE":
606 let player = target!
607 guard participants[insuranceTurn] === player else {
608 throw CommandError("waiting for \(participants[insuranceTurn].name)")
609 }
610 if value > player.bet / 2 { throw CommandError("insurance out of limits") }
611 if value > player.bankroll { throw CommandError("insufficient funds") }
612 player.bankroll -= value
613 player.insurance = value
614 emit(value == 0 ? "\(player.name) declines insurance" : "\(player.name) insures \(value)")
615 insuranceTurn += 1
616 if insuranceTurn == participants.count { peek() }
617 case "HINT":
618 let turn = current()!
619 emit("hint \(turn.player.handName(turn.index)): \(recommend().rawValue.lowercased())")
620 case "STATUS": status()
621 case "COUNT":
622 if decks == 0 { throw CommandError("no shoe") }
623 let remaining = shoe.count - drawn
624 let hundredths = abs(running) * 5200 / remaining
625 let fraction = String(hundredths % 100)
626 let count = (running < 0 ? "-" : "") + String(hundredths / 100)
627 + "." + (fraction.count == 1 ? "0" : "") + fraction
628 emit("count running \(running) true \(count) remaining \(remaining)")
