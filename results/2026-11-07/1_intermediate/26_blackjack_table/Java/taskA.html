20 static long integer(String word, long minimum, long maximum) {
21 if (word.isEmpty() || word.length() > 10 || word.length() > 1 && word.charAt(0) == '0') {
22 throw error("bad value " + word);
23 }
24 long value = 0;
25 for (int i = 0; i < word.length(); i++) {
26 char digit = word.charAt(i);
27 if (digit < '0' || digit > '9') throw error("bad value " + word);
28 value = value * 10 + digit - '0';
29 }
30 if (value < minimum || value > maximum) throw error("bad value " + word);
31 return value;
32 }

34 static String choice(String word, String... choices) {
35 for (String value : choices) if (value.equals(word)) return word;
36 throw error("bad value " + word);
37 }

128 Rules changed(String name, String word) {
129 var next = new Rules(this);
130 switch (name) {
131 case "DEALER" -> next.hitSoft17 = choice(word, "S17", "H17").equals("H17");
132 case "PEN" -> next.penetration = (int) integer(word, 50, 90);
133 case "MIN" -> next.minimum = integer(word, 1, 10000);
134 case "MAX" -> next.maximum = integer(word, 1, 10000);
135 case "BJ" -> next.sixToFive = choice(word, "3:2", "6:5").equals("6:5");
136 case "DAS" -> next.doubleAfterSplit = on(word);
137 case "DOUBLE" -> next.doubling = switch (choice(word, "ANY", "9-11", "10-11")) {
138 case "ANY" -> DoubleRule.ANY;
139 case "9-11" -> DoubleRule.HARD_9;
140 default -> DoubleRule.HARD_10;
141 };
142 case "HANDS" -> next.hands = (int) integer(word, 1, 4);
143 case "RSA" -> next.resplitAces = on(word);
144 case "SURRENDER" -> next.surrender = on(word);
145 case "CHARLIE" -> next.charlie = Integer.parseInt(choice(word, "0", "5", "6", "7"));
146 case "SIDEMAX" -> next.sideMaximum = integer(word, 0, 1000);
147 default -> throw error("unknown rule " + name);
148 }
149 return next;
150 }

152 @Override public String toString() {
153 return format("rules DEALER %s PEN %d MIN %d MAX %d BJ %s DAS %s DOUBLE %s HANDS %d RSA %s SURRENDER %s CHARLIE %d SIDEMAX %d",
154 hitSoft17 ? "H17" : "S17", penetration, minimum, maximum, sixToFive ? "6:5" : "3:2",
155 onOff(doubleAfterSplit), doubling.text, hands, onOff(resplitAces), onOff(surrender), charlie, sideMaximum);
156 }

159 record Card(int code) {
160 int rank() { return code % 13; }
161 int suit() { return code / 13; }
162 int point() { return Math.min(rank() + 1, 10); }
163 boolean black() { return suit() == 0 || suit() == 3; }
164 int hilo() {
165 int value = point();
166 return value >= 2 && value <= 6 ? 1 : value == 1 || value == 10 ? -1 : 0;
167 }
168 @Override public String toString() {
169 return "" + "A23456789TJQK".charAt(rank()) + "CDHS".charAt(suit());
170 }
171 }

176 final List<Card> cards;

178 final boolean split, splitAces;

180 Finish reason;

182 Hand(List<Card> cards, long bet, boolean split, boolean splitAces) {
183 this.cards = new ArrayList<>(cards);
184 this.bet = bet;
185 this.split = split;
186 this.splitAces = splitAces;

189 Total total() {
190 int sum = 0;
191 boolean ace = false;
192 for (Card card : cards) {
193 sum += card.point();
194 ace |= card.rank() == 0;
195 }
196 boolean soft = ace && sum + 10 <= 21;
197 return new Total(sum + (soft ? 10 : 0), soft);
198 }
199
200 boolean blackjack() { return !split && cards.size() == 2 && total().value() == 21; }
201
202 @Override public String toString() {
203 var result = new StringBuilder();
204 for (Card card : cards) {
205 if (!result.isEmpty()) result.append(' ');
206 result.append(card);
207 }
208 Total total = total();
209 String status = blackjack() ? "blackjack" : total.value() > 21 ? "bust " + total.value()
210 : total.soft() ? "soft " + total.value() : Integer.toString(total.value());
211 return result.append(" (").append(status).append(')').toString();
212 }
213
214 Outcome outcome(int dealer, boolean dealerBlackjack) {
215 if (dealerBlackjack) return blackjack() ? Outcome.PUSH : Outcome.LOSE;
216 if (blackjack()) return Outcome.BLACKJACK;
217 if (reason != null) {
218 return switch (reason) {
219 case BUST -> Outcome.BUST;
220 case SURRENDER -> Outcome.SURRENDER;
221 case CHARLIE -> Outcome.CHARLIE;
222 };
223 }
224 int value = total().value();
225 return dealer > 21 || value > dealer ? Outcome.WIN : value == dealer ? Outcome.PUSH : Outcome.LOSE;
226 }

268 static final class Shoe {
269 final int decks;
270 final Card[] cards;
271 int drawn, running;
272 long random;
273
274 Shoe(int decks, long seed) {
275 this.decks = decks;
276 cards = new Card[52 * decks];
277 random = seed;
278 shuffle();
279 }
280
281 void shuffle() {
282 for (int i = 0; i < cards.length; i++) cards[i] = new Card(i % 52);
283 for (int i = cards.length - 1; i >= 1; i--) {
284 random = random * 48271 % 2147483647;
285 int j = (int) (random % (i + 1));
286 Card card = cards[i];
287 cards[i] = cards[j];
288 cards[j] = card;
289 }
290 drawn = running = 0;
291 }
292
293 int cut(int penetration) { return cards.length * penetration / 100; }
294 Card draw(boolean visible) {
295 Card card = cards[drawn++];
296 if (visible) running += card.hilo();
297 return card;
298 }
299 }

341 void finishCheck(Player player, Hand hand) {
342 int value = hand.total().value();
343 if (value > 21) {
344 hand.finished = true;
345 hand.reason = Finish.BUST;
346 } else if (rules.charlie != 0 && hand.cards.size() == rules.charlie) {
347 hand.finished = true;
348 hand.reason = Finish.CHARLIE;
349 } else if (value == 21) {
350 hand.finished = true;
351 } else if (hand.splitAces && hand.cards.size() == 2) {
352 hand.finished = !(hand.cards.get(0).rank() == 0 && hand.cards.get(1).rank() == 0
353 && rules.resplitAces && player.hands.size() < rules.hands);
354 }
355 }

357 void settle(boolean dealerBlackjack) {
358 int dealerTotal = dealer.total().value();
359 for (Player player : participants) {
360 for (int i = 0; i < player.hands.size(); i++) {
361 Hand hand = player.hands.get(i);
362 Outcome result = hand.outcome(dealerTotal, dealerBlackjack);
363 long profit = result.profit(hand.bet, rules);
364 player.bankroll += hand.bet + profit;
365 player.net += profit;
366 player.stats.record(result);
367 emit("%s %s %+d", player.handName(i), result.text(), profit);
368 }
369 emit("%s bankroll %d", player.name, player.bankroll);
370 }
371 for (Player player : seats) if (player != null) player.bet = player.side = 0;
372 state = State.IDLE;
373 }

375 void dealerTurn() {
376 reveal();
377 boolean needsDraw = false;
378 for (Player player : participants) {
379 for (Hand hand : player.hands) {
380 if (hand.reason == null && !hand.blackjack()) needsDraw = true;
381 }
382 }
383 boolean drew = false;
384 if (needsDraw) {
385 while (true) {
386 Total total = dealer.total();
387 if (!(total.value() <= 16 || total.value() == 17 && total.soft() && rules.hitSoft17)) break;
388 dealer.cards.add(shoe.draw(true));
389 drew = true;
390 emit("dealer hits: %s", dealer);
391 }
392 }
393 if (drew && dealer.total().value() > 21) dealerBusts++;
394 settle(false);
395 }

398 while (true) {
399 Turn turn = current();
400 if (turn == null) {
401 dealerTurn();

405 turn.hand().cards.add(shoe.draw(true));

407 finishCheck(turn.player(), turn.hand());

413 boolean natural = dealer.blackjack();
414 emit("dealer peeks: %s", natural ? "blackjack" : "no blackjack");
415 for (Player player : participants) {
416 if (player.insurance == 0) continue;
417 long profit = player.insurance * (natural ? 2 : -1);
418 player.bankroll += player.insurance + profit;

432 void deal() {
433 if (shoe.drawn >= shoe.cut(rules.penetration)) {
434 shoe.shuffle();
435 emit("shuffle");
436 }
437 emit("round %d", ++round);
438 participants.clear();
439 for (Player player : seats) {
440 if (player == null || player.bet == 0) continue;
441 participants.add(player);
442 player.bankroll -= player.bet + player.side;
443 player.insurance = 0;
444 player.stats.rounds++;
445 player.hands.clear();
446 player.hands.add(new Hand(List.of(), player.bet, false, false));
447 }
448 dealer = new Hand(List.of(), 0, false, false);
449 holeVisible = false;
450 for (Player player : participants) player.hands.getFirst().cards.add(shoe.draw(true));
451 dealer.cards.add(shoe.draw(true));
452 for (Player player : participants) {
453 Hand hand = player.hands.getFirst();
454 hand.cards.add(shoe.draw(true));
455 hand.finished = hand.blackjack();
456 }
457 dealer.cards.add(shoe.draw(false));
458 for (Player player : participants) emit("%s: %s", player.handName(0), player.hands.getFirst());
459 emit("dealer: %s ??", dealer.cards.getFirst());
460 for (Player player : participants) {
461 if (player.side == 0) continue;
462 Card first = player.hands.getFirst().cards.get(0), second = player.hands.getFirst().cards.get(1);
463 String result = "lose";
464 long multiplier = -1;
465 if (first.rank() == second.rank()) {
466 if (first.suit() == second.suit()) { result = "perfect"; multiplier = 25; }
467 else if (first.black() == second.black()) { result = "colored"; multiplier = 12; }
468 else { result = "mixed"; multiplier = 6; }
469 }
470 long profit = player.side * multiplier;
471 player.bankroll += player.side + profit;
472 player.net += profit;
473 emit("%s side %s %+d", player.name, result, profit);
474 }
475 switch (dealer.cards.getFirst().point()) {
476 case 1 -> {
477 state = State.INSURANCE;
478 insuranceTurn = 0;
479 emit("insurance offered");
480 }
481 case 10 -> peek();
482 default -> { state = State.PLAY; advance(); }
483 }
484 }

486 String actionProblem(Action action, Player player, Hand hand) {
487 switch (action) {
488 case HIT -> { if (hand.splitAces) return "split aces"; }
489 case DOUBLE -> {
490 if (hand.splitAces) return "split aces";
491 if (hand.cards.size() != 2) return "two cards only";
492 Total total = hand.total();
493 int minimum = rules.doubling == DoubleRule.HARD_10 ? 10 : 9;
494 if (rules.doubling != DoubleRule.ANY && (total.soft() || total.value() < minimum || total.value() > 11)
495 || !rules.doubleAfterSplit && hand.split) return "double not allowed";
496 if (player.bankroll < hand.bet) return "insufficient funds";
497 }
498 case SPLIT -> {
499 if (hand.cards.size() != 2) return "two cards only";
500 if (hand.cards.get(0).point() != hand.cards.get(1).point()) return "not a pair";
501 if (player.hands.size() >= rules.hands) return "hand limit";
502 if (player.bankroll < hand.bet) return "insufficient funds";
503 }
504 case SURRENDER -> {
505 if (!rules.surrender || hand.split) return "surrender not allowed";
506 if (hand.cards.size() != 2) return "two cards only";
507 }
508 case STAND -> {}
509 }
510 return null;
511 }

513 Action recommend() {
514 Turn turn = current();
515 Player player = turn.player();
516 Hand hand = turn.hand();
517 int up = dealer.cards.getFirst().point();
518 if (up == 1) up = 11;
519 boolean canSplit = actionProblem(Action.SPLIT, player, hand) == null;
520 if (hand.splitAces) return canSplit ? Action.SPLIT : Action.STAND;
521 if (canSplit) {
522 int pair = hand.cards.getFirst().point();
523 if (pair == 1 || pair == 8 || pair == 9 && up != 7 && up != 10 && up != 11
524 || (pair == 2 || pair == 3 || pair == 7) && up <= 7 || pair == 6 && up <= 6
525 || pair == 4 && (up == 5 || up == 6)) return Action.SPLIT;
526 }
527 Total total = hand.total();
528 int value = total.value();
529 String symbol = "H";
530 if (total.soft()) {
531 if (value >= 19) symbol = "S";
532 else if (value == 18) symbol = up >= 3 && up <= 6 ? "Ds" : up == 2 || up == 7 || up == 8 ? "S" : "H";
533 else if (value == 17 && up >= 3 && up <= 6 || (value == 15 || value == 16) && up >= 4 && up <= 6
534 || (value == 13 || value == 14) && (up == 5 || up == 6)) symbol = "D";
535 } else {
536 if (value >= 17) symbol = "S";
537 else if (value >= 13) symbol = up <= 6 ? "S" : value == 16 && up >= 9 || value == 15 && up == 10 ? "R" : "H";
538 else if (value == 12) symbol = up >= 4 && up <= 6 ? "S" : "H";
539 else if (value == 11 && up <= 10 || value == 10 && up <= 9 || value == 9 && up >= 3 && up <= 6) symbol = "D";
540 }
541 return switch (symbol) {
542 case "S" -> Action.STAND;
543 case "D" -> actionProblem(Action.DOUBLE, player, hand) == null ? Action.DOUBLE : Action.HIT;
544 case "Ds" -> actionProblem(Action.DOUBLE, player, hand) == null ? Action.DOUBLE : Action.STAND;
545 case "R" -> actionProblem(Action.SURRENDER, player, hand) == null ? Action.SURRENDER : Action.HIT;
546 default -> Action.HIT;
547 };
548 }

550 void perform(Action action) {
551 Turn turn = current();
552 Player player = turn.player();
553 Hand hand = turn.hand();
554 String name = turn.name();
555 switch (action) {
556 case SPLIT -> {
557 Card first = hand.cards.get(0), second = hand.cards.get(1);
558 boolean aces = hand.splitAces || first.rank() == 0;
559 player.bankroll -= hand.bet;
560 player.stats.splits++;
561 emit("%s splits %s %s", name, first, second);
562 var replacement = new Hand(List.of(first), hand.bet, true, aces);
563 player.hands.set(turn.index(), replacement);
564 player.hands.add(turn.index() + 1, new Hand(List.of(second), hand.bet, true, aces));
565 replacement.cards.add(shoe.draw(true));
566 emit("%s: %s", name, replacement);
567 finishCheck(player, replacement);
568 }
569 case STAND, SURRENDER -> {
570 hand.finished = true;
571 if (action == Action.SURRENDER) hand.reason = Finish.SURRENDER;
572 emit("%s %s: %s", name, action == Action.STAND ? "stands" : "surrenders", hand);
573 }
574 case HIT, DOUBLE -> {
575 if (action == Action.DOUBLE) {
576 player.bankroll -= hand.bet;
577 hand.bet *= 2;
578 player.stats.doubles++;
579 }
580 hand.cards.add(shoe.draw(true));
581 emit("%s %s: %s", name, action == Action.DOUBLE ? "doubles" : "hits", hand);
582 finishCheck(player, hand);
583 if (action == Action.DOUBLE) hand.finished = true;
584 }
585 }
586 advance();
587 }

608 List<String> execute(String[] words) {
609 output.clear();
610 Command command = Command.parse(words[0]);
611 Action action = command.action();
612 Player player = null;
613 long amount = 0, seed = 0;
614 Rules proposed = null;
615 switch (command) {
616 case SHOE -> { amount = integer(words[1], 1, 8); seed = integer(words[2], 1, 2147483646); }
617 case RULE -> proposed = rules.changed(words[1], words[2]);
618 case PLAYER -> {
619 String name = words[1];
620 if (name.isEmpty() || name.length() > 8) throw error("bad name " + name);
621 for (int i = 0; i < name.length(); i++) {
622 if (name.charAt(i) < 'A' || name.charAt(i) > 'Z') throw error("bad name " + name);
623 }
624 amount = integer(words[2], 1, 1000000);
625 }
626 default -> {
627 if (command.namedPlayer()) {
628 player = find(words[1]);
629 if (player == null) throw error("unknown player " + words[1]);
630 if (command == Command.REBUY || command == Command.BET || command == Command.SIDE || command == Command.INSURANCE) {
631 amount = integer(words[2], command == Command.REBUY ? 1 : 0, 1000000);
632 }
633 }
634 }
635 }
636 if (command.idleOnly() && state != State.IDLE) throw error("round in progress");
637 if (command == Command.INSURANCE && state != State.INSURANCE) throw error("no insurance");
638 if (action != null || command == Command.AUTO || command == Command.HINT) {
639 if (state == State.IDLE) throw error("no round");
640 if (state == State.INSURANCE) throw error("insurance pending");
641 }
642 if (action != null || command == Command.AUTO) {
643 Turn turn = current();
644 if (turn.player() != player) throw error("waiting for " + turn.name());
645 if (action != null) {
646 String problem = actionProblem(action, player, turn.hand());
647 if (problem != null) throw error(problem);
648 perform(action);
649 } else {
650 while (state == State.PLAY && current().player() == player) perform(recommend());
651 }
652 return output;
653 }
654 switch (command) {
655 case SHOE -> {
656 shoe = new Shoe((int) amount, seed);
657 emit("shoe %d decks %d cards cut %d", shoe.decks, shoe.cards.length, shoe.cut(rules.penetration));
658 }
659 case RULE -> {
660 if (proposed.minimum > proposed.maximum) throw error("min above max");
661 rules = proposed;
662 emit("rule %s %s", words[1], words[2]);
663 }
664 case RULES -> emit("%s", rules);
665 case PLAYER -> {
666 if (find(words[1]) != null) throw error("player exists");
667 int seat = 0;
668 while (seat < seats.length && seats[seat] != null) seat++;
669 if (seat == seats.length) throw error("table full");
670 seats[seat] = new Player(words[1], seat, amount);
671 emit("%s sits at seat %d with %d", words[1], seat + 1, amount);
672 }
673 case LEAVE -> {
674 emit("%s leaves with %d net %+d", player.name, player.bankroll, player.net);
675 seats[player.seat] = null;
676 }
677 case REBUY -> {
678 player.bankroll += amount;
679 emit("%s rebuys %d bankroll %d", player.name, amount, player.bankroll);
680 }
681 case BET -> {
682 if (amount != 0) {
683 if (amount < rules.minimum || amount > rules.maximum) throw error("bet out of limits");
684 if (amount + player.side > player.bankroll) throw error("insufficient funds");
685 }
686 player.bet = amount;
687 if (amount == 0) { player.side = 0; emit("%s bet cleared", player.name); }
688 else emit("%s bets %d", player.name, amount);
689 }
690 case SIDE -> {
691 if (amount != 0) {
692 if (player.bet == 0) throw error("no bet");
693 if (amount > rules.sideMaximum) throw error("side out of limits");
694 if (player.bet + amount > player.bankroll) throw error("insufficient funds");
695 }
696 player.side = amount;
697 if (amount == 0) emit("%s side cleared", player.name);
698 else emit("%s side %d", player.name, amount);
699 }
700 case DEAL -> {
701 if (shoe == null) throw error("no shoe");
702 boolean hasBet = false;
703 for (Player seated : seats) if (seated != null && seated.bet != 0) hasBet = true;
704 if (!hasBet) throw error("no bets");
705 deal();
706 }
707 case INSURANCE -> {
708 if (player != participants.get(insuranceTurn)) throw error("waiting for " + participants.get(insuranceTurn).name);
709 if (amount > player.bet / 2) throw error("insurance out of limits");
710 if (amount > player.bankroll) throw error("insufficient funds");
711 player.bankroll -= amount;
712 player.insurance = amount;
713 if (amount == 0) emit("%s declines insurance", player.name);
714 else emit("%s insures %d", player.name, amount);
715 if (++insuranceTurn == participants.size()) peek();
716 }
717 case HINT -> emit("hint %s: %s", current().name(), recommend().text());
718 case STATUS -> status();
719 case COUNT -> {
720 if (shoe == null) throw error("no shoe");
721 int remaining = shoe.cards.length - shoe.drawn;
722 long scaled = Math.abs((long) shoe.running) * 5200 / remaining;
723 emit("count running %d true %s%d.%02d remaining %d", shoe.running, shoe.running < 0 ? "-" : "", scaled / 100, scaled % 100, remaining);
724 }
725 case STATS -> {
726 boolean any = false;
727 for (Player seated : seats) if (seated != null) any = true;
728 if (!any) throw error("no players");
729 for (Player seated : seats) {
730 if (seated != null) emit("stats %s %s net %+d", seated.name, seated.stats, seated.net);
731 }
732 emit("stats dealer rounds %d blackjacks %d busts %d", round, dealerBlackjacks, dealerBusts);
733 }
734 default -> {}
735 }
736 return output;
737 }
738 }

754 }
