131 fn add(hand: *Hand, card: Card) void {
132 hand.cards[hand.len] = card;
133 hand.len += 1;
134 }
135 fn total(hand: *const Hand) Total {
136 var sum: i64 = 0;
137 var ace = false;
138 for (hand.cards[0..hand.len]) |card| {
139 sum += card.points();
140 ace = ace or card.rank() == 0;
141 }
142 const soft = ace and sum + 10 <= 21;
143 return .{ .value = sum + (if (soft) @as(i64, 10) else 0), .soft = soft };
144 }
145 fn natural(hand: *const Hand) bool {
146 return !hand.split and hand.len == 2 and hand.total().value == 21;
147 }
148 pub fn format(hand: Hand, out: *Writer) Writer.Error!void {
149 for (hand.cards[0..hand.len], 0..) |card, i| {
150 if (i > 0) try out.writeByte(' ');
151 try out.print("{f}", .{card});
152 }
153 try out.writeAll(" (");
154 const t = hand.total();
155 if (hand.natural()) try out.writeAll("blackjack") else {
156 if (t.value > 21) try out.writeAll("bust ") else if (t.soft) try out.writeAll("soft ");
157 try out.print("{d}", .{t.value});
158 }
159 try out.writeByte(')');
160 }

256 fn cut(table: *const Table) usize {
257 return table.shoe_len * table.rules.penetration / 100;

259 fn shuffle(table: *Table) void {
260 table.shoe_len = 52 * table.decks;
261 for (0..table.shoe_len) |i| table.shoe[i] = .{ .code = @intCast(i % 52) };
262 var i = table.shoe_len - 1;
263 while (i > 0) : (i -= 1) {
264 table.random = table.random * 48271 % 2147483647;
265 const j: usize = @intCast(table.random % (i + 1));
266 if (i != j) std.mem.swap(Card, &table.shoe[i], &table.shoe[j]);
267 }
268 table.drawn = 0;
269 table.running = 0;
270 }
271 fn countCard(table: *Table, card: Card) void {
272 const value = card.points();
273 table.running += if (value >= 2 and value <= 6) @as(i64, 1) else if (value == 1 or value == 10) -1 else 0;
274 }
275 fn draw(table: *Table, face_up: bool) Card {
276 const card = table.shoe[table.drawn];
277 table.drawn += 1;
278 if (face_up) table.countCard(card);
279 return card;
280 }

288 fn current(table: *const Table) ?Turn {
289 for (table.participants[0..table.participant_count]) |seat| {
290 const p = table.owner(seat);

295 fn finishCheck(table: *const Table, p: *const Player, hand: *Hand) void {
296 const value = hand.total().value;
297 if (value > 21) {
298 hand.reason = .bust;
299 hand.finished = true;
300 } else if (table.rules.charlie != 0 and hand.len == table.rules.charlie) {
301 hand.reason = .charlie;
302 hand.finished = true;
303 } else if (value == 21) {
304 hand.finished = true;
305 } else if (hand.split_aces and hand.len == 2) {
306 hand.finished = !(hand.cards[0].rank() == 0 and hand.cards[1].rank() == 0 and table.rules.resplit_aces and p.hand_count < table.rules.hands);
307 }
308 }
309 fn settle(table: *Table, natural: bool, out: *Writer) Writer.Error!void {
310 const dealer_total = table.dealer.total().value;
311 for (table.participants[0..table.participant_count]) |seat| {
312 const p = table.player(seat);
313 for (p.hands[0..p.hand_count], 0..) |hand, i| {
314 const value = hand.total().value;
315 const result: Outcome = if (natural) (if (hand.natural()) .push else .lose) else if (hand.natural()) .blackjack else if (hand.reason) |reason| switch (reason) {
316 .bust => .bust,
317 .surrender => .surrender,
318 .charlie => .charlie,
319 } else if (dealer_total > 21 or value > dealer_total) .win else if (value == dealer_total) .push else .lose;
320 const profit: i64 = switch (result) {
321 .blackjack => if (table.rules.six_to_five) @divTrunc(hand.bet * 6, 5) else @divTrunc(hand.bet * 3, 2),
322 .win, .charlie => hand.bet,
323 .push => 0,
324 .surrender => @divTrunc(hand.bet, 2) - hand.bet,
325 .lose, .bust => -hand.bet,
326 };
327 p.bankroll += hand.bet + profit;
328 p.net += profit;
329 p.stats.hands += 1;
330 switch (result) {
331 .blackjack, .win, .charlie => p.stats.wins += 1,
332 .push => p.stats.pushes += 1,
333 else => p.stats.losses += 1,
334 }
335 if (result == .blackjack) p.stats.blackjacks += 1;
336 if (result == .bust) p.stats.busts += 1;
337 if (result == .surrender) p.stats.surrenders += 1;
338 try line(out, "{f} {s} {f}", .{ p.label(i), @tagName(result), signed(profit) });
339 }
340 try line(out, "{f} bankroll {d}", .{ p.name, p.bankroll });
341 }
342 for (0..7) |seat| if (table.seats[seat] != null) {
343 const p = table.player(seat);
344 p.bet = 0;
345 p.side = 0;
346 };
347 table.phase = .idle;
348 }
349 fn dealerTurn(table: *Table, out: *Writer) Writer.Error!void {
350 try table.revealDealer(out);
351 var required = false;
352 for (table.participants[0..table.participant_count]) |seat| {
353 const p = table.owner(seat);
354 for (p.hands[0..p.hand_count]) |hand| required = required or (hand.reason == null and !hand.natural());
355 }
356 var drew = false;
357 if (required) while (true) {
358 const t = table.dealer.total();
359 if (!(t.value <= 16 or t.value == 17 and t.soft and table.rules.hit_soft17)) break;
360 table.dealer.add(table.draw(true));
361 drew = true;
362 try line(out, "dealer hits: {f}", .{table.dealer});
363 };
364 if (drew and table.dealer.total().value > 21) table.dealer_busts += 1;
365 try table.settle(false, out);
366 }
367 fn advanceHands(table: *Table, out: *Writer) Writer.Error!void {
368 while (table.current()) |turn| {
369 const p = table.player(turn.player);
370 const hand = &p.hands[turn.hand];
371 if (hand.len != 1) return;
372 hand.add(table.draw(true));
373 try line(out, "{f}: {f}", .{ p.label(turn.hand), hand.* });
374 table.finishCheck(p, hand);
375 if (!hand.finished) return;
376 }
377 try table.dealerTurn(out);
378 }
379 fn peek(table: *Table, out: *Writer) Writer.Error!void {
380 const natural = table.dealer.natural();
381 try line(out, "dealer peeks: {s}", .{if (natural) "blackjack" else "no blackjack"});
382 for (table.participants[0..table.participant_count]) |seat| {
383 const p = table.player(seat);
384 if (p.insurance == 0) continue;
385 const profit = p.insurance * (if (natural) @as(i64, 2) else -1);
386 p.bankroll += p.insurance + profit;
387 p.net += profit;
388 try line(out, "{f} insurance {f}", .{ p.name, signed(profit) });
389 }
390 if (natural) {
391 table.dealer_blackjacks += 1;
392 try table.revealDealer(out);
393 try table.settle(true, out);
394 } else {
395 table.phase = .play;
396 try table.advanceHands(out);
397 }
398 }
399 fn deal(table: *Table, out: *Writer) Writer.Error!void {
400 if (table.drawn >= table.cut()) {
401 table.shuffle();
402 try line(out, "shuffle", .{});
403 }
404 table.round += 1;
405 try line(out, "round {d}", .{table.round});
406 table.participant_count = 0;
407 for (table.seats, 0..) |slot, seat| if (slot) |value| {
408 if (value.bet != 0) {
409 table.participants[table.participant_count] = seat;
410 table.participant_count += 1;
411 }
412 };
413 for (table.participants[0..table.participant_count]) |seat| {
414 const p = table.player(seat);
415 p.bankroll -= p.bet + p.side;
416 p.insurance = 0;
417 p.stats.rounds += 1;
418 p.hand_count = 1;
419 p.hands[0] = .{ .bet = p.bet };
420 p.hands[0].add(table.draw(true));
421 }
422 table.dealer = .{};
423 table.dealer.add(table.draw(true));
424 for (table.participants[0..table.participant_count]) |seat| {
425 const hand = &table.player(seat).hands[0];
426 hand.add(table.draw(true));
427 hand.finished = hand.natural();
428 }
429 table.dealer.add(table.draw(false));
430 table.hole_revealed = false;
431 for (table.participants[0..table.participant_count]) |seat| {
432 const p = table.owner(seat);
433 try line(out, "{f}: {f}", .{ p.label(0), p.hands[0] });
434 }
435 try line(out, "dealer: {f} ??", .{table.dealer.cards[0]});
436 for (table.participants[0..table.participant_count]) |seat| {
437 const p = table.player(seat);
438 if (p.side == 0) continue;
439 const first = p.hands[0].cards[0];
440 const second = p.hands[0].cards[1];
441 const result: []const u8 = if (first.rank() != second.rank()) "lose" else if (first.suit() == second.suit()) "perfect" else if (first.black() == second.black()) "colored" else "mixed";
442 const multiplier: i64 = if (first.rank() != second.rank()) -1 else if (first.suit() == second.suit()) 25 else if (first.black() == second.black()) 12 else 6;
443 const profit = p.side * multiplier;
444 p.bankroll += p.side + profit;
445 p.net += profit;
446 try line(out, "{f} side {s} {f}", .{ p.name, result, signed(profit) });
447 }
448 const up = table.dealer.cards[0].points();
449 if (up == 1) {
450 table.phase = .insurance;
451 table.insurance_turn = 0;
452 try line(out, "insurance offered", .{});
453 } else if (up == 10) try table.peek(out) else {
454 table.phase = .play;
455 try table.advanceHands(out);
456 }
457 }
458 fn actionError(table: *const Table, action: Action, p: *const Player, hand: *const Hand) ?[]const u8 {
459 switch (action) {
460 .hit => if (hand.split_aces) return "split aces",
461 .double => {
462 if (hand.split_aces) return "split aces";
463 if (hand.len != 2) return "two cards only";
464 const t = hand.total();
465 if (table.rules.doubling != .any and (t.soft or t.value < (if (table.rules.doubling == .hard9) @as(i64, 9) else 10) or t.value > 11) or !table.rules.double_after_split and hand.split) return "double not allowed";
466 if (p.bankroll < hand.bet) return "insufficient funds";
467 },
468 .split => {
469 if (hand.len != 2) return "two cards only";
470 if (hand.cards[0].points() != hand.cards[1].points()) return "not a pair";
471 if (p.hand_count >= table.rules.hands) return "hand limit";
472 if (p.bankroll < hand.bet) return "insufficient funds";
473 },
474 .surrender => {
475 if (!table.rules.surrender or hand.split) return "surrender not allowed";
476 if (hand.len != 2) return "two cards only";
477 },
478 .stand => {},
479 }
480 return null;
481 }
482 fn perform(table: *Table, action: Action, out: *Writer) Writer.Error!void {
483 const turn = table.current().?;
484 const p = table.player(turn.player);
485 const label = p.label(turn.hand);
486 if (action == .split) {
487 const old = p.hands[turn.hand];
488 const first = old.cards[0];
489 const second = old.cards[1];
490 const aces = old.split_aces or first.points() == 1;
491 p.bankroll -= old.bet;
492 p.stats.splits += 1;
493 try line(out, "{f} splits {f} {f}", .{ label, first, second });
494 var i = p.hand_count;
495 while (i > turn.hand + 1) : (i -= 1) p.hands[i] = p.hands[i - 1];
496 p.hand_count += 1;
497 p.hands[turn.hand] = .{ .bet = old.bet, .split = true, .split_aces = aces };
498 p.hands[turn.hand].add(first);
499 p.hands[turn.hand + 1] = .{ .bet = old.bet, .split = true, .split_aces = aces };
500 p.hands[turn.hand + 1].add(second);
501 const hand = &p.hands[turn.hand];
502 hand.add(table.draw(true));
503 try line(out, "{f}: {f}", .{ label, hand.* });
504 table.finishCheck(p, hand);
505 } else {
506 const hand = &p.hands[turn.hand];
507 if (action == .stand or action == .surrender) {
508 hand.finished = true;
509 if (action == .surrender) hand.reason = .surrender;
510 try line(out, "{f} {s}: {f}", .{ label, if (action == .stand) "stands" else "surrenders", hand.* });
511 } else {
512 if (action == .double) {
513 p.bankroll -= hand.bet;
514 hand.bet *= 2;
515 p.stats.doubles += 1;
516 }
517 hand.add(table.draw(true));
518 try line(out, "{f} {s}: {f}", .{ label, if (action == .double) "doubles" else "hits", hand.* });
519 table.finishCheck(p, hand);
520 if (action == .double) hand.finished = true;
521 }
522 }
523 try table.advanceHands(out);
524 }
525 fn recommend(table: *const Table) Action {
526 const turn = table.current().?;
527 const p = table.owner(turn.player);
528 const hand = &p.hands[turn.hand];
529 const up: i64 = if (table.dealer.cards[0].points() == 1) 11 else table.dealer.cards[0].points();
530 const can_split = table.actionError(.split, p, hand) == null;
531 if (hand.split_aces) return if (can_split) .split else .stand;
532 if (hand.len == 2 and hand.cards[0].points() == hand.cards[1].points() and can_split) {
533 const pair = hand.cards[0].points();
534 if (pair == 1 or pair == 8 or pair == 9 and up != 7 and up != 10 and up != 11 or (pair == 2 or pair == 3 or pair == 7) and up <= 7 or pair == 6 and up <= 6 or pair == 4 and (up == 5 or up == 6)) return .split;
535 }
536 const t = hand.total();
537 const Symbol = enum { H, S, D, Ds, R };
538 var symbol: Symbol = .H;
539 if (t.soft) {
540 if (t.value >= 19) symbol = .S else if (t.value == 18) {
541 symbol = if (up >= 3 and up <= 6) .Ds else if (up == 2 or up == 7 or up == 8) .S else .H;
542 } else if (t.value == 17 and up >= 3 and up <= 6 or (t.value == 15 or t.value == 16) and up >= 4 and up <= 6 or (t.value == 13 or t.value == 14) and (up == 5 or up == 6)) symbol = .D;
543 } else {
544 if (t.value >= 17) symbol = .S else if (t.value >= 13) {
545 symbol = if (up <= 6) .S else if (t.value == 16 and up >= 9 or t.value == 15 and up == 10) .R else .H;
546 } else if (t.value == 12) symbol = if (up >= 4 and up <= 6) .S else .H else if (t.value == 11 and up <= 10 or t.value == 10 and up <= 9 or t.value == 9 and up >= 3 and up <= 6) symbol = .D;
547 }
548 return switch (symbol) {
549 .S => .stand,
550 .H => .hit,
551 .D => if (table.actionError(.double, p, hand) == null) .double else .hit,
552 .Ds => if (table.actionError(.double, p, hand) == null) .double else .stand,
553 .R => if (table.actionError(.surrender, p, hand) == null) .surrender else .hit,
554 };
555 }
556 fn status(table: *const Table, out: *Writer) Writer.Error!void {
557 try out.print("table {s} round {d}", .{ @tagName(table.phase), table.round });
558 if (table.phase == .insurance) try out.print(" next {f}", .{table.owner(table.participants[table.insurance_turn]).name}) else if (table.phase == .play) {
559 const turn = table.current().?;
560 try out.print(" turn {f}", .{table.owner(turn.player).label(turn.hand)});
561 }
562 try out.writeByte('\n');
563 if (table.phase != .idle) try line(out, "dealer: {f} ??", .{table.dealer.cards[0]});
564 for (table.seats, 0..) |slot, seat| if (slot) |p| {
565 try line(out, "seat {d} {f} bankroll {d} bet {d} side {d}", .{ seat + 1, p.name, p.bankroll, p.bet, p.side });
566 if (table.phase != .idle) for (table.participants[0..table.participant_count]) |participant| {
567 if (participant == seat) {
568 for (p.hands[0..p.hand_count], 0..) |hand, i| try line(out, "{f}: {f} bet {d}", .{ p.label(i), hand, hand.bet });
569 break;
570 }
571 };
572 };
573 }
574};

576const Program = struct {
577 table: Table = .{},
578 scratch: Allocator,
579 out: *Writer,
580 error_message: []const u8 = "",
581
582 fn fail(program: *Program, comptime fmt: []const u8, args: anytype) Error {
583 program.error_message = std.fmt.allocPrint(program.scratch, fmt, args) catch return error.OutOfMemory;
584 return error.BadCommand;
585 }
586 fn integer(program: *Program, text: []const u8, minimum: i64, maximum: i64) Error!i64 {
587 if (text.len < 1 or text.len > 10 or text.len > 1 and text[0] == '0') return program.fail("bad value {s}", .{text});
588 var value: i64 = 0;
589 for (text) |ch| {
590 if (ch < '0' or ch > '9') return program.fail("bad value {s}", .{text});
591 value = value * 10 + ch - '0';
592 }
593 if (value < minimum or value > maximum) return program.fail("bad value {s}", .{text});
594 return value;
595 }
596 fn option(program: *Program, text: []const u8, choices: []const []const u8) Error!usize {
597 for (choices, 0..) |word, i| if (std.mem.eql(u8, text, word)) return i;
598 return program.fail("bad value {s}", .{text});
599 }
600 fn changedRules(program: *Program, name: []const u8, text: []const u8) Error!Rules {
601 const rule = std.meta.stringToEnum(RuleName, name) orelse return program.fail("unknown rule {s}", .{name});
602 var next = program.table.rules;
603 switch (rule) {
604 .DEALER => next.hit_soft17 = (try program.option(text, &.{ "S17", "H17" })) == 1,
605 .PEN => next.penetration = @intCast(try program.integer(text, 50, 90)),
606 .MIN => next.minimum = try program.integer(text, 1, 10000),
607 .MAX => next.maximum = try program.integer(text, 1, 10000),
608 .BJ => next.six_to_five = (try program.option(text, &.{ "3:2", "6:5" })) == 1,
609 .DAS => next.double_after_split = (try program.option(text, &.{ "OFF", "ON" })) == 1,
610 .DOUBLE => next.doubling = @fromBackingInt(@intCast(try program.option(text, &.{ "ANY", "9-11", "10-11" }))),
611 .HANDS => next.hands = @intCast(try program.integer(text, 1, 4)),
612 .RSA => next.resplit_aces = (try program.option(text, &.{ "OFF", "ON" })) == 1,
613 .SURRENDER => next.surrender = (try program.option(text, &.{ "OFF", "ON" })) == 1,
614 .CHARLIE => {
615 const choices = [_]usize{ 0, 5, 6, 7 };
616 next.charlie = choices[try program.option(text, &.{ "0", "5", "6", "7" })];
617 },
618 .SIDEMAX => next.side_maximum = try program.integer(text, 0, 1000),
619 }
620 return next;
621 }
622 fn execute(program: *Program, words: []const []const u8) Error!void {
623 const command = std.meta.stringToEnum(Command, words[0]) orelse return program.fail("unknown command {s}", .{words[0]});
624 const first = if (words.len > 1) words[1] else "";
625 const second = if (words.len > 2) words[2] else "";
626 const table = &program.table;
627 const out = program.out;
628 const action = command.action();
629 var target: ?usize = null;
630 var value: i64 = 0;
631 var seed: u64 = 0;
632 var proposed = table.rules;
633
634 if (command == .SHOE) {
635 value = try program.integer(first, 1, 8);
636 seed = @intCast(try program.integer(second, 1, 2147483646));
637 } else if (command == .RULE) {
638 proposed = try program.changedRules(first, second);
639 } else if (command == .PLAYER) {
640 if (first.len < 1 or first.len > 8) return program.fail("bad name {s}", .{first});
641 for (first) |ch| if (ch < 'A' or ch > 'Z') return program.fail("bad name {s}", .{first});
642 value = try program.integer(second, 1, 1000000);
643 } else if (command.namedPlayer()) {
644 target = table.find(first) orelse return program.fail("unknown player {s}", .{first});
645 if (command.hasAmount()) value = try program.integer(second, if (command == .REBUY) 1 else 0, 1000000);
646 }
647 if (command.idleOnly()) {
648 if (table.phase != .idle) return program.fail("round in progress", .{});
649 } else if (command == .INSURANCE) {
650 if (table.phase != .insurance) return program.fail("no insurance", .{});
651 } else if (action != null or command == .AUTO or command == .HINT) {
652 if (table.phase == .idle) return program.fail("no round", .{});
653 if (table.phase == .insurance) return program.fail("insurance pending", .{});
654 }
655 if (action != null or command == .AUTO) {
656 const turn = table.current().?;
657 const p = table.owner(turn.player);
658 if (turn.player != target.?) return program.fail("waiting for {f}", .{p.label(turn.hand)});
659 if (action) |selected| {
660 if (table.actionError(selected, p, &p.hands[turn.hand])) |problem| return program.fail("{s}", .{problem});
661 try table.perform(selected, out);
662 } else while (table.phase == .play and table.current().?.player == target.?) {
663 try table.perform(table.recommend(), out);
664 }
665 return;
666 }
667 switch (command) {
668 .SHOE => {
669 table.decks = @intCast(value);
670 table.random = seed;
671 table.shuffle();
672 try line(out, "shoe {d} decks {d} cards cut {d}", .{ table.decks, table.shoe_len, table.cut() });
673 },
674 .RULE => {
675 if (proposed.minimum > proposed.maximum) return program.fail("min above max", .{});
676 table.rules = proposed;
677 try line(out, "rule {s} {s}", .{ first, second });
678 },
679 .RULES => try line(out, "{f}", .{table.rules}),
680 .PLAYER => {
681 if (table.find(first) != null) return program.fail("player exists", .{});
682 var empty: ?usize = null;
683 for (table.seats, 0..) |slot, seat| if (slot == null) {
684 empty = seat;
685 break;
686 };
687 const seat = empty orelse return program.fail("table full", .{});
688 table.seats[seat] = .{ .name = Name.init(first), .bankroll = value };
689 try line(out, "{f} sits at seat {d} with {d}", .{ table.owner(seat).name, seat + 1, value });
690 },
691 .LEAVE => {
692 const p = table.owner(target.?);
693 try line(out, "{f} leaves with {d} net {f}", .{ p.name, p.bankroll, signed(p.net) });
694 table.seats[target.?] = null;
695 },
696 .REBUY => {
697 const p = table.player(target.?);
698 p.bankroll += value;
699 try line(out, "{f} rebuys {d} bankroll {d}", .{ p.name, value, p.bankroll });
700 },
701 .BET => {
702 const p = table.player(target.?);
703 if (value != 0) {
704 if (value < table.rules.minimum or value > table.rules.maximum) return program.fail("bet out of limits", .{});
705 if (value + p.side > p.bankroll) return program.fail("insufficient funds", .{});
706 }
707 p.bet = value;
708 if (value == 0) {
709 p.side = 0;
710 try line(out, "{f} bet cleared", .{p.name});
711 } else try line(out, "{f} bets {d}", .{ p.name, value });
712 },
713 .SIDE => {
714 const p = table.player(target.?);
715 if (value != 0) {
716 if (p.bet == 0) return program.fail("no bet", .{});
717 if (value > table.rules.side_maximum) return program.fail("side out of limits", .{});
718 if (p.bet + value > p.bankroll) return program.fail("insufficient funds", .{});
719 }
720 p.side = value;
721 if (value == 0) try line(out, "{f} side cleared", .{p.name}) else try line(out, "{f} side {d}", .{ p.name, value });
722 },
723 .DEAL => {
724 if (table.decks == 0) return program.fail("no shoe", .{});
725 var bets = false;
726 for (table.seats) |slot| if (slot) |p| {
727 bets = bets or p.bet != 0;
728 };
729 if (!bets) return program.fail("no bets", .{});
730 try table.deal(out);
731 },
732 .INSURANCE => {
733 const waiting = table.participants[table.insurance_turn];
734 if (target.? != waiting) return program.fail("waiting for {f}", .{table.owner(waiting).name});
735 const p = table.player(target.?);
736 if (value > @divTrunc(p.bet, 2)) return program.fail("insurance out of limits", .{});
737 if (value > p.bankroll) return program.fail("insufficient funds", .{});
738 p.bankroll -= value;
739 p.insurance = value;
740 if (value == 0) try line(out, "{f} declines insurance", .{p.name}) else try line(out, "{f} insures {d}", .{ p.name, value });
741 table.insurance_turn += 1;
742 if (table.insurance_turn == table.participant_count) try table.peek(out);
743 },
744 .HINT => {
745 const turn = table.current().?;
746 try line(out, "hint {f}: {s}", .{ table.owner(turn.player).label(turn.hand), @tagName(table.recommend()) });
747 },
748 .STATUS => try table.status(out),
749 .COUNT => {
750 if (table.decks == 0) return program.fail("no shoe", .{});
751 const remaining = table.shoe_len - table.drawn;
752 const magnitude: u64 = @intCast(if (table.running < 0) -table.running else table.running);
753 const hundredths = magnitude * 5200 / remaining;
754 try line(out, "count running {d} true {s}{d}.{d:0>2} remaining {d}", .{ table.running, if (table.running < 0) "-" else "", hundredths / 100, hundredths % 100, remaining });
755 },
756 .STATS => {
757 var any = false;
758 for (table.seats) |slot| if (slot != null) {
759 any = true;
760 };
761 if (!any) return program.fail("no players", .{});
762 for (table.seats) |slot| if (slot) |p| try line(out, "stats {f} {f} net {f}", .{ p.name, p.stats, signed(p.net) });
763 try line(out, "stats dealer rounds {d} blackjacks {d} busts {d}", .{ table.round, table.dealer_blackjacks, table.dealer_busts });
764 },
765 .HIT, .STAND, .DOUBLE, .SPLIT, .SURRENDER, .AUTO => unreachable,
766 }
767 }
768};

770fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
771 var buffer: Writer.Allocating = .init(gpa);
772 defer buffer.deinit();
773 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
774 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
775 const text = try buffer.toOwnedSlice();
776 var words: std.ArrayList([]const u8) = .empty;
777 var parts = std.mem.tokenizeAny(u8, text, " \r");
778 while (parts.next()) |word| try words.append(gpa, word);
779 return try words.toOwnedSlice(gpa);
780}

782pub fn main(init: std.process.Init) !void {
783 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
784 defer scratch.deinit();
785 var input_buffer: [4096]u8 = undefined;
786 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
787 var output_buffer: [4096]u8 = undefined;
788 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
789 var program: Program = .{ .scratch = scratch.allocator(), .out = &output.interface };
790 const count = try std.fmt.parseInt(usize, (try readWords(scratch.allocator(), &input.interface))[0], 10);
791 for (0..count) |_| {
792 _ = scratch.reset(.retain_capacity);
793 const words = try readWords(scratch.allocator(), &input.interface);
794 var response: Writer.Allocating = .init(scratch.allocator());
795 defer response.deinit();
796 program.out = &response.writer;
797 program.execute(words) catch |err| {
798 if (err != error.BadCommand) return err;
799 try output.interface.print("error: {s}\n", .{program.error_message});
800 continue;
801 };
802 try output.interface.writeAll(response.written());
803 }
804 try output.interface.flush();
805}
