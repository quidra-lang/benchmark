101static bool integer_word(const char *word, int low, int high, int *value) {
102 size_t length = strlen(word);
103 if (length == 0 || length > 10 || (length > 1 && word[0] == '0')) return false;
104 int64_t number = 0;
105 for (size_t i = 0; i < length; i++) {
106 if (word[i] < '0' || word[i] > '9') return false;
107 number = number * 10 + word[i] - '0';
108 }
109 if (number < low || number > high) return false;
110 *value = (int)number;
111 return true;
112}

143static Total total(const Hand *hand) {
144 int sum = 0;
145 bool ace = false;
146 for (int i = 0; i < hand->count; i++) {
147 sum += point(hand->cards[i]);
148 ace = ace || hand->cards[i].rank == 0;
149 }
150 bool soft = ace && sum + 10 <= 21;
151 return (Total){ sum + (soft ? 10 : 0), soft };
152}

180static void shuffle(Shoe *shoe) {
181 for (int i = 0; i < shoe->size; i++) shoe->cards[i] = (Card){ i % 13, i / 13 % 4 };
182 for (int i = shoe->size - 1; i > 0; i--) {
183 shoe->random = 48271 * shoe->random % 2147483647;
184 int j = (int)(shoe->random % (i + 1));
185 Card temporary = shoe->cards[i];
186 shoe->cards[i] = shoe->cards[j];
187 shoe->cards[j] = temporary;
188 }
189 shoe->drawn = shoe->running = 0;
190}

207static bool parse_rule(Rule rule, const char *word, int *value) {
208 const char *const dealers[] = { "S17", "H17" };
209 const char *const blackjacks[] = { "3:2", "6:5" };
210 const char *const doubles[] = { "ANY", "9-11", "10-11" };
211 switch (rule) {
212 case R_DEALER: *value = lookup(word, dealers, 2); return *value >= 0;
213 case R_BJ: *value = lookup(word, blackjacks, 2); return *value >= 0;
214 case R_DOUBLE: *value = lookup(word, doubles, 3); return *value >= 0;
215 case R_DAS: case R_RSA: case R_SURRENDER:
216 if (strcmp(word, "ON") == 0) { *value = 1; return true; }
217 if (strcmp(word, "OFF") == 0) { *value = 0; return true; }
218 return false;
219 case R_PEN: return integer_word(word, 50, 90, value);
220 case R_MIN: case R_MAX: return integer_word(word, 1, 10000, value);
221 case R_HANDS: return integer_word(word, 1, 4, value);
222 case R_SIDEMAX: return integer_word(word, 0, 1000, value);
223 case R_CHARLIE:
224 return integer_word(word, 0, 7, value) && (*value == 0 || *value >= 5);
225 default: return false;
226 }
227}

244 }

247static Text rule_text(const Rules *rules, Rule rule) {
248 Text text = { "" };
249 const char *word = NULL;
250 int value = 0;
251 switch (rule) {
252 case R_DEALER: word = rules->dealer == S17 ? "S17" : "H17"; break;
253 case R_PEN: value = rules->pen; break;
254 case R_MIN: value = rules->min; break;
255 case R_MAX: value = rules->max; break;
256 case R_BJ: word = rules->bj == THREE_TWO ? "3:2" : "6:5"; break;
257 case R_DAS: word = rules->das ? "ON" : "OFF"; break;
258 case R_DOUBLE: word = rules->doubling == ANY ? "ANY" : rules->doubling == NINE_ELEVEN ? "9-11" : "10-11"; break;
259 case R_HANDS: value = rules->hands; break;
260 case R_RSA: word = rules->rsa ? "ON" : "OFF"; break;
261 case R_SURRENDER: word = rules->surrender ? "ON" : "OFF"; break;
262 case R_CHARLIE: value = rules->charlie; break;
263 case R_SIDEMAX: value = rules->side_max; break;
264 default: break;
265 }
266 if (word) snprintf(text.text, sizeof text.text, "%s", word);
267 else snprintf(text.text, sizeof text.text, "%d", value);
268 return text;
269}

271static void automatic_finish(const Table *table, const Player *player, Hand *hand) {
272 Total score = total(hand);
273 if (score.value > 21) { hand->finished = true; hand->reason = FINISH_BUST; }
274 else if (table->rules.charlie && hand->count == table->rules.charlie) {
275 hand->finished = true; hand->reason = FINISH_CHARLIE;
276 } else if (score.value == 21) hand->finished = true;
277 else if (hand->split_aces && hand->count == 2) {
278 bool resplit = hand->cards[0].rank == 0 && hand->cards[1].rank == 0 &&
279 table->rules.rsa && player->hand_count < table->rules.hands;
280 hand->finished = !resplit;
281 }
282}

284static Result hand_result(const Table *table, const Hand *hand) {
285 if (table->dealer_blackjack) return blackjack(hand) ? PUSH : LOSE;
286 if (blackjack(hand)) return BLACKJACK;
287 if (hand->reason == FINISH_BUST) return BUST;
288 if (hand->reason == FINISH_SURRENDER) return SURRENDERED;
289 if (hand->reason == FINISH_CHARLIE) return CHARLIE;
290 int player_total = total(hand).value, dealer_total = total(&table->dealer).value;
291 if (dealer_total > 21 || player_total > dealer_total) return WIN;
292 return player_total == dealer_total ? PUSH : LOSE;
293}

295static Money profit(const Rules *rules, Money bet, Result result) {
296 switch (result) {
297 case BLACKJACK: return rules->bj == THREE_TWO ? bet * 3 / 2 : bet * 6 / 5;
298 case WIN: case CHARLIE: return bet;
299 case PUSH: return 0;
300 case SURRENDERED: return bet / 2 - bet;
301 default: return -bet;
302 }
303}

305static void settle(Table *table) {
306 for (int seat = 0; seat < SEATS; seat++) {
307 Player *player = &table->players[seat];
308 if (!player->participating) continue;
309 for (int index = 0; index < player->hand_count; index++) {
310 const Hand *hand = &player->hands[index];
311 Result result = hand_result(table, hand);
312 Money gain = profit(&table->rules, hand->bet, result);
313 player->bankroll += hand->bet + gain;
314 player->net += gain;
315 player->stats.hands++;
316 if (result == BLACKJACK || result == WIN || result == CHARLIE) player->stats.wins++;
317 else if (result == PUSH) player->stats.pushes++;
318 else player->stats.losses++;
319 if (result == BLACKJACK) player->stats.blackjacks++;
320 if (result == BUST) player->stats.busts++;
321 if (result == SURRENDERED) player->stats.surrenders++;
322 printf("%s %s %+" PRId64 "\n", hand_name(player, index).text, result_names[result], gain);
323 }
324 printf("%s bankroll %" PRId64 "\n", player->name, player->bankroll);
325 }
326 for (int seat = 0; seat < SEATS; seat++) {
327 table->players[seat].bet = table->players[seat].side = 0;
328 table->players[seat].participating = false;
329 }
330 table->state = IDLE;
331}

333static void dealer_turn(Table *table) {
334 reveal(table);
335 dealer_line(table, false);
336 bool needs_draw = false;
337 for (int seat = 0; seat < SEATS; seat++) {
338 const Player *player = &table->players[seat];
339 if (!player->participating) continue;
340 for (int i = 0; i < player->hand_count; i++) {
341 const Hand *hand = &player->hands[i];
342 if (hand->reason == FINISH_NONE && !blackjack(hand)) needs_draw = true;
343 }
344 }
345 bool drew = false;
346 if (needs_draw) {
347 for (;;) {
348 Total score = total(&table->dealer);
349 if (!(score.value <= 16 || (score.value == 17 && score.soft && table->rules.dealer == H17))) break;
350 add_card(&table->dealer, draw(table, true));
351 dealer_line(table, true);
352 drew = true;
353 }
354 }
355 if (drew && total(&table->dealer).value > 21) table->dealer_busts++;
356 settle(table);
357}

359static void next_hand(Table *table) {
360 for (;;) {
361 bool found = false;
362 for (int seat = 0; seat < SEATS && !found; seat++) {
363 Player *player = &table->players[seat];
364 if (!player->participating) continue;
365 for (int index = 0; index < player->hand_count; index++) {
366 if (!player->hands[index].finished) {
367 table->current_seat = seat;
368 table->current_hand = index;
369 found = true;
370 break;
371 }
372 }
373 }
374 if (!found) { dealer_turn(table); return; }
375 Player *player = &table->players[table->current_seat];
376 Hand *hand = &player->hands[table->current_hand];
377 if (hand->count != 1) return;
378 add_card(hand, draw(table, true));
379 hand_line(player, table->current_hand, "");
380 automatic_finish(table, player, hand);
381 if (!hand->finished) return;
382 }
383}

385static void peek(Table *table) {
386 table->dealer_blackjack = blackjack(&table->dealer);
387 printf("dealer peeks: %s\n", table->dealer_blackjack ? "blackjack" : "no blackjack");
388 if (table->dealer_blackjack) table->dealer_blackjacks++;
389 for (int seat = 0; seat < SEATS; seat++) {
390 Player *player = &table->players[seat];
391 if (!player->participating || player->insurance == 0) continue;
392 Money gain = table->dealer_blackjack ? player->insurance * 2 : -player->insurance;
393 player->bankroll += player->insurance + gain;
394 player->net += gain;
395 printf("%s insurance %+" PRId64 "\n", player->name, gain);
396 }
397 if (table->dealer_blackjack) {
398 reveal(table);
399 dealer_line(table, false);
400 settle(table);
401 } else {
402 table->state = PLAY;
403 next_hand(table);
404 }
405}

407static void deal(Table *table) {
408 if (table->shoe.drawn >= table->shoe.size * table->rules.pen / 100) {
409 shuffle(&table->shoe);
410 puts("shuffle");
411 }
412 table->round++;
413 printf("round %d\n", table->round);
414 table->dealer = (Hand){0};
415 table->hole_visible = table->dealer_blackjack = false;
416 for (int seat = 0; seat < SEATS; seat++) {
417 Player *player = &table->players[seat];
418 player->participating = player->seated && player->bet != 0;
419 player->hand_count = 0;
420 if (player->participating) {
421 player->bankroll -= player->bet + player->side;
422 player->insurance = 0;
423 player->stats.rounds++;
424 player->hand_count = 1;
425 player->hands[0] = (Hand){ .bet = player->bet };
426 add_card(&player->hands[0], draw(table, true));
427 }
428 }
429 add_card(&table->dealer, draw(table, true));
430 for (int seat = 0; seat < SEATS; seat++) {
431 Player *player = &table->players[seat];
432 if (player->participating) {
433 add_card(&player->hands[0], draw(table, true));
434 player->hands[0].finished = blackjack(&player->hands[0]);
435 }
436 }
437 add_card(&table->dealer, draw(table, false));
438 for (int seat = 0; seat < SEATS; seat++) {
439 if (table->players[seat].participating) hand_line(&table->players[seat], 0, "");
440 }
441 printf("dealer: %s ??\n", card_text(table->dealer.cards[0]).text);
442 for (int seat = 0; seat < SEATS; seat++) {
443 Player *player = &table->players[seat];
444 if (!player->participating || player->side == 0) continue;
445 Card a = player->hands[0].cards[0], b = player->hands[0].cards[1];
446 int multiplier = a.rank != b.rank ? -1 : a.suit == b.suit ? 25 : black(a) == black(b) ? 12 : 6;
447 const char *result = multiplier == 25 ? "perfect" : multiplier == 12 ? "colored" : multiplier == 6 ? "mixed" : "lose";
448 Money gain = player->side * multiplier;
449 player->bankroll += player->side + gain;
450 player->net += gain;
451 printf("%s side %s %+" PRId64 "\n", player->name, result, gain);
452 }
453 int upcard = point(table->dealer.cards[0]);
454 if (upcard == 1) {
455 table->state = INSURANCE;
456 table->insurance_seat = 0;
457 while (!table->players[table->insurance_seat].participating) table->insurance_seat++;
458 puts("insurance offered");
459 } else if (upcard == 10) peek(table);
460 else { table->state = PLAY; next_hand(table); }
461}

463static const char *action_problem(const Table *table, Action action) {
464 const Player *player = &table->players[table->current_seat];
465 const Hand *hand = &player->hands[table->current_hand];
466 Total score = total(hand);
467 if (action == HIT) return hand->split_aces ? "split aces" : NULL;
468 if (action == DOUBLE) {
469 if (hand->split_aces) return "split aces";
470 if (hand->count != 2) return "two cards only";
471 if ((!table->rules.das && hand->split) ||
472 (table->rules.doubling != ANY && (score.soft || score.value > 11 ||
473 score.value < (table->rules.doubling == NINE_ELEVEN ? 9 : 10)))) return "double not allowed";
474 if (player->bankroll < hand->bet) return "insufficient funds";
475 } else if (action == SPLIT) {
476 if (hand->count != 2) return "two cards only";
477 if (point(hand->cards[0]) != point(hand->cards[1])) return "not a pair";
478 if (player->hand_count >= table->rules.hands) return "hand limit";
479 if (player->bankroll < hand->bet) return "insufficient funds";
480 } else if (action == SURRENDER) {
481 if (!table->rules.surrender || hand->split) return "surrender not allowed";
482 if (hand->count != 2) return "two cards only";
483 }
484 return NULL;
485}

487static Action recommendation(const Table *table) {
488 const Player *player = &table->players[table->current_seat];
489 const Hand *hand = &player->hands[table->current_hand];
490 int up = point(table->dealer.cards[0]);
491 if (up == 1) up = 11;
492 bool split_possible = action_problem(table, SPLIT) == NULL;
493 if (hand->split_aces) return split_possible ? SPLIT : STAND;
494 if (split_possible) {
495 int value = point(hand->cards[0]);
496 bool preferred = value == 1 || value == 8 ||
497 (value == 9 && up != 7 && up != 10 && up != 11) ||
498 ((value == 2 || value == 3 || value == 7) && up <= 7) ||
499 (value == 6 && up <= 6) || (value == 4 && (up == 5 || up == 6));
500 if (preferred) return SPLIT;
501 }
502 Total score = total(hand);
503 int value = score.value;
504 bool can_double = action_problem(table, DOUBLE) == NULL;
505 if (score.soft) {
506 if (value >= 19) return STAND;
507 if (value == 18) {
508 if (up >= 3 && up <= 6) return can_double ? DOUBLE : STAND;
509 return up == 2 || up == 7 || up == 8 ? STAND : HIT;
510 }
511 bool wants_double = (value == 17 && up >= 3 && up <= 6) ||
512 ((value == 15 || value == 16) && up >= 4 && up <= 6) ||
513 ((value == 13 || value == 14) && (up == 5 || up == 6));
514 return wants_double && can_double ? DOUBLE : HIT;
515 }
516 if (value >= 17) return STAND;
517 if (value >= 13) {
518 if (up <= 6) return STAND;
519 if (((value == 16 && up >= 9) || (value == 15 && up == 10)) && action_problem(table, SURRENDER) == NULL) return SURRENDER;
520 return HIT;
521 }
522 if (value == 12) return up >= 4 && up <= 6 ? STAND : HIT;
523 bool wants_double = (value == 11 && up <= 10) || (value == 10 && up <= 9) ||
524 (value == 9 && up >= 3 && up <= 6);
525 return wants_double && can_double ? DOUBLE : HIT;
526}

528static void perform_action(Table *table, Action action) {
529 Player *player = &table->players[table->current_seat];
530 int index = table->current_hand;
531 Hand *hand = &player->hands[index];
532 if (action == HIT || action == DOUBLE) {
533 if (action == DOUBLE) {
534 player->bankroll -= hand->bet;
535 hand->bet *= 2;
536 player->stats.doubles++;
537 }
538 add_card(hand, draw(table, true));
539 hand_line(player, index, action == HIT ? " hits" : " doubles");
540 automatic_finish(table, player, hand);
541 if (action == DOUBLE) hand->finished = true;
542 } else if (action == STAND || action == SURRENDER) {
543 hand->finished = true;
544 if (action == SURRENDER) hand->reason = FINISH_SURRENDER;
545 hand_line(player, index, action == STAND ? " stands" : " surrenders");
546 } else {
547 Card first = hand->cards[0], second = hand->cards[1];
548 Money bet = hand->bet;
549 bool aces = hand->split_aces || first.rank == 0;
550 player->bankroll -= bet;
551 player->stats.splits++;
552 printf("%s splits %s %s\n", hand_name(player, index).text, card_text(first).text, card_text(second).text);
553 memmove(&player->hands[index + 2], &player->hands[index + 1],
554 (size_t)(player->hand_count - index - 1) * sizeof(Hand));
555 player->hands[index] = (Hand){ .bet = bet, .split = true, .split_aces = aces };
556 player->hands[index + 1] = (Hand){ .bet = bet, .split = true, .split_aces = aces };
557 add_card(&player->hands[index], first);
558 add_card(&player->hands[index + 1], second);
559 player->hand_count++;
560 add_card(&player->hands[index], draw(table, true));
561 hand_line(player, index, "");
562 automatic_finish(table, player, &player->hands[index]);
563 }
564 next_hand(table);
565}

567static void show_status(const Table *table) {
568 if (table->state == IDLE) printf("table idle round %d\n", table->round);
569 else if (table->state == INSURANCE) printf("table insurance round %d next %s\n", table->round, table->players[table->insurance_seat].name);
570 else printf("table play round %d turn %s\n", table->round, hand_name(&table->players[table->current_seat], table->current_hand).text);

576 seat + 1, player->name, player->bankroll, player->bet, player->side);
577 if (table->state != IDLE && player->participating) {
578 for (int i = 0; i < player->hand_count; i++) {
579 printf("%s: ", hand_name(player, i).text);
580 show_hand(&player->hands[i]);
581 printf(" bet %" PRId64 "\n", player->hands[i].bet);
582 }
583 }
584 }

598static bool named_player(Command command) {
599 return command == C_LEAVE || command == C_REBUY || command == C_BET || command == C_SIDE ||
600 command == C_INSURANCE || (command >= C_HIT && command <= C_SURRENDER) || command == C_AUTO;
601}
602
603static bool idle_only(Command command) {
604 return command == C_SHOE || command == C_RULE || command == C_PLAYER || command == C_LEAVE ||
605 command == C_REBUY || command == C_BET || command == C_SIDE || command == C_DEAL;
606}

608static void execute(Table *table, char *words[]) {
609 int found = lookup(words[0], command_names, COMMAND_COUNT);
610 if (found < 0) { error_line("unknown command %s", words[0]); return; }
611 Command command = (Command)found;
612 int seat = -1, first = 0, second = 0, rule = -1;
613 if (command == C_SHOE) {
614 if (!integer_word(words[1], 1, 8, &first)) { error_line("bad value %s", words[1]); return; }
615 if (!integer_word(words[2], 1, 2147483646, &second)) { error_line("bad value %s", words[2]); return; }
616 } else if (command == C_RULE) {
617 rule = lookup(words[1], rule_names, RULE_COUNT);
618 if (rule < 0) { error_line("unknown rule %s", words[1]); return; }
619 if (!parse_rule((Rule)rule, words[2], &first)) { error_line("bad value %s", words[2]); return; }
620 } else if (command == C_PLAYER) {
621 if (!valid_name(words[1])) { error_line("bad name %s", words[1]); return; }
622 if (!integer_word(words[2], 1, 1000000, &first)) { error_line("bad value %s", words[2]); return; }
623 } else if (named_player(command)) {
624 seat = find_player(table, words[1]);
625 if (seat < 0) { error_line("unknown player %s", words[1]); return; }
626 if (command == C_REBUY || command == C_BET || command == C_SIDE || command == C_INSURANCE) {
627 if (!integer_word(words[2], command == C_REBUY ? 1 : 0, 1000000, &first)) {
628 error_line("bad value %s", words[2]); return;
629 }
630 }
631 }
632 if (idle_only(command) && table->state != IDLE) { error_line("round in progress"); return; }
633 if (command == C_INSURANCE && table->state != INSURANCE) { error_line("no insurance"); return; }
634 bool action_command = (command >= C_HIT && command <= C_SURRENDER) || command == C_AUTO;
635 if (action_command || command == C_HINT) {
636 if (table->state == IDLE) { error_line("no round"); return; }
637 if (table->state == INSURANCE) { error_line("insurance pending"); return; }
638 if (action_command && seat != table->current_seat) {
639 error_line("waiting for %s", hand_name(&table->players[table->current_seat], table->current_hand).text);
640 return;
641 }
642 }
643 Player *player = seat < 0 ? NULL : &table->players[seat];
644 switch (command) {
645 case C_SHOE:
646 table->shoe.decks = first;
647 table->shoe.size = first * 52;
648 table->shoe.random = second;
649 shuffle(&table->shoe);
650 printf("shoe %d decks %d cards cut %d\n", first, table->shoe.size, table->shoe.size * table->rules.pen / 100);
651 break;
652 case C_RULE: {
653 Rules changed = table->rules;
654 set_rule(&changed, (Rule)rule, first);
655 if (changed.min > changed.max) { error_line("min above max"); return; }
656 table->rules = changed;
657 printf("rule %s %s\n", rule_names[rule], rule_text(&changed, (Rule)rule).text);
658 break;
659 }
660 case C_RULES:
661 fputs("rules", stdout);
662 for (int i = 0; i < RULE_COUNT; i++) printf(" %s %s", rule_names[i], rule_text(&table->rules, (Rule)i).text);
663 putchar('\n');
664 break;
665 case C_PLAYER:
666 if (find_player(table, words[1]) >= 0) { error_line("player exists"); return; }
667 for (seat = 0; seat < SEATS && table->players[seat].seated; seat++) {}
668 if (seat == SEATS) { error_line("table full"); return; }
669 table->players[seat] = (Player){ .seated = true, .bankroll = first };
670 strcpy(table->players[seat].name, words[1]);
671 printf("%s sits at seat %d with %d\n", words[1], seat + 1, first);
672 break;
673 case C_LEAVE:
674 printf("%s leaves with %" PRId64 " net %+" PRId64 "\n", player->name, player->bankroll, player->net);
675 *player = (Player){0};
676 break;
677 case C_REBUY:
678 player->bankroll += first;
679 printf("%s rebuys %d bankroll %" PRId64 "\n", player->name, first, player->bankroll);
680 break;
681 case C_BET:
682 if (first == 0) {
683 player->bet = player->side = 0;
684 printf("%s bet cleared\n", player->name);
685 } else {
686 if (first < table->rules.min || first > table->rules.max) { error_line("bet out of limits"); return; }
687 if (first + player->side > player->bankroll) { error_line("insufficient funds"); return; }
688 player->bet = first;
689 printf("%s bets %d\n", player->name, first);
690 }
691 break;
692 case C_SIDE:
693 if (first == 0) {
694 player->side = 0;
695 printf("%s side cleared\n", player->name);
696 } else {
697 if (player->bet == 0) { error_line("no bet"); return; }
698 if (first > table->rules.side_max) { error_line("side out of limits"); return; }
699 if (player->bet + first > player->bankroll) { error_line("insufficient funds"); return; }
700 player->side = first;
701 printf("%s side %d\n", player->name, first);
702 }
703 break;
704 case C_DEAL: {
705 if (table->shoe.size == 0) { error_line("no shoe"); return; }
706 bool bets = false;
707 for (int i = 0; i < SEATS; i++) if (table->players[i].seated && table->players[i].bet != 0) bets = true;
708 if (!bets) { error_line("no bets"); return; }
709 deal(table);
710 break;
711 }
712 case C_INSURANCE:
713 if (seat != table->insurance_seat) { error_line("waiting for %s", table->players[table->insurance_seat].name); return; }
714 if (first > player->bet / 2) { error_line("insurance out of limits"); return; }
715 if (first > player->bankroll) { error_line("insufficient funds"); return; }
716 player->bankroll -= first;
717 player->insurance = first;
718 if (first == 0) printf("%s declines insurance\n", player->name);
719 else printf("%s insures %d\n", player->name, first);
720 do { table->insurance_seat++; }
721 while (table->insurance_seat < SEATS && !table->players[table->insurance_seat].participating);
722 if (table->insurance_seat == SEATS) peek(table);
723 break;
724 case C_HIT: case C_STAND: case C_DOUBLE: case C_SPLIT: case C_SURRENDER: {
725 Action action = (Action)(command - C_HIT);
726 const char *problem = action_problem(table, action);
727 if (problem) { error_line("%s", problem); return; }
728 perform_action(table, action);
729 break;
730 }
731 case C_HINT:
732 printf("hint %s: %s\n", hand_name(&table->players[table->current_seat], table->current_hand).text, action_names[recommendation(table)]);
733 break;
734 case C_AUTO:
735 while (table->state == PLAY && table->current_seat == seat) perform_action(table, recommendation(table));
736 break;
737 case C_STATUS: show_status(table); break;
738 case C_COUNT: {
739 if (table->shoe.size == 0) { error_line("no shoe"); return; }
740 int remaining = table->shoe.size - table->shoe.drawn;
741 int running = table->shoe.running;
742 int64_t scaled = (int64_t)abs(running) * 5200 / remaining;
743 printf("count running %d true %s%" PRId64 ".%02" PRId64 " remaining %d\n", running, running < 0 ? "-" : "", scaled / 100, scaled % 100, remaining);
744 break;
745 }
746 case C_STATS: {
747 bool seated = false;
748 for (int i = 0; i < SEATS; i++) seated = seated || table->players[i].seated;
749 if (!seated) { error_line("no players"); return; }
750 show_stats(table);
751 break;
752 }
753 default: break;
754 }
755}

757int main(void) {
758 Table table = {
759 .rules = { .dealer = S17, .pen = 75, .min = 10, .max = 500, .bj = THREE_TWO,
760 .das = true, .doubling = ANY, .hands = 4, .rsa = false,
761 .surrender = true, .charlie = 0, .side_max = 25 }
762 };
763 char line[128];
764 if (!fgets(line, sizeof line, stdin)) return 0;
765 int commands = atoi(line);
766 for (int i = 0; i < commands; i++) {
767 if (!fgets(line, sizeof line, stdin)) break;
768 char *words[4] = {0};
769 int count = 0;
770 char *word = strtok(line, " \r\n");
771 while (word && count < 4) {
772 words[count++] = word;
773 word = strtok(NULL, " \r\n");
774 }
775 execute(&table, words);
776 }
777 return 0;
