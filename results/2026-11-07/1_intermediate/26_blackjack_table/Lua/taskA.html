2 local result = {}
3 for word in words:gmatch("%S+") do result[word] = true end
4 return result
5end

7local actions = set("HIT STAND DOUBLE SPLIT SURRENDER")
8local idle_commands = set("SHOE RULE PLAYER LEAVE REBUY BET SIDE DEAL")
9local player_commands = set("LEAVE REBUY BET SIDE INSURANCE HIT STAND DOUBLE SPLIT SURRENDER AUTO")
10local amount_commands = set("REBUY BET SIDE INSURANCE")
11local rule_names = { "DEALER", "PEN", "MIN", "MAX", "BJ", "DAS", "DOUBLE", "HANDS", "RSA", "SURRENDER", "CHARLIE", "SIDEMAX" }
12local stat_names = { "rounds", "hands", "wins", "losses", "pushes", "blackjacks", "busts", "surrenders", "doubles", "splits" }
13local defaults = { DEALER = "S17", PEN = 75, MIN = 10, MAX = 500, BJ = "3:2", DAS = "ON", DOUBLE = "ANY",
14 HANDS = 4, RSA = "OFF", SURRENDER = "ON", CHARLIE = 0, SIDEMAX = 25 }
15local rule_choices = { DEALER = set("S17 H17"), BJ = set("3:2 6:5"), DOUBLE = set("ANY 9-11 10-11"),
16 DAS = set("ON OFF"), RSA = set("ON OFF"), SURRENDER = set("ON OFF") }
17local rule_bounds = { PEN = { 50, 90 }, MIN = { 1, 10000 }, MAX = { 1, 10000 }, HANDS = { 1, 4 },
18 CHARLIE = { 0, 7 }, SIDEMAX = { 0, 1000 } }

20local function fail(message) error({ command_error = true, message = message }, 0) end
21local function copy(values)
22 local result = {}
23 for key, value in pairs(values) do result[key] = value end
24 return result
25end
26local function integer_word(word, low, high)
27 if #word > 10 or not word:match("^[0-9]+$") or (#word > 1 and word:sub(1, 1) == "0") then fail("bad value " .. word) end
28 local value = tonumber(word)
29 if value < low or value > high then fail("bad value " .. word) end
30 return value
31end
32local function rule_value(name, word)
33 if defaults[name] == nil then fail("unknown rule " .. name) end
34 if rule_choices[name] then
35 if not rule_choices[name][word] then fail("bad value " .. word) end
36 return word
37 end
38 local bounds = rule_bounds[name]
39 local value = integer_word(word, bounds[1], bounds[2])
40 if name == "CHARLIE" and value ~= 0 and value < 5 then fail("bad value " .. word) end
41 return value
42end
43local function signed(value) return (value >= 0 and "+" or "") .. value end
44local function point(card) return math.min(10, card % 13 + 1) end
45local function card_text(card) return ("A23456789TJQK"):sub(card % 13 + 1, card % 13 + 1) .. ("CDHS"):sub(card // 13 + 1, card // 13 + 1) end
46local function black(card) return card // 13 == 0 or card // 13 == 3 end
47local function hilo(card)
48 local value = point(card)
49 return value >= 2 and value <= 6 and 1 or (value >= 7 and value <= 9 and 0 or -1)
50end

53Hand.__index = Hand
54function Hand.new(cards, bet, split, split_aces)
55 return setmetatable({ cards = cards or {}, bet = bet or 0, split = split or false, split_aces = split_aces or false }, Hand)
56end
57function Hand:total()
58 local value, ace = 0, false
59 for _, card in ipairs(self.cards) do
60 value = value + point(card)
61 ace = ace or card % 13 == 0
62 end
63 local soft = ace and value + 10 <= 21
64 return value + (soft and 10 or 0), soft
65end
66function Hand:blackjack() return not self.split and #self.cards == 2 and self:total() == 21 end
67function Hand:text()
68 local value, soft = self:total()
69 local status = self:blackjack() and "blackjack" or (value > 21 and "bust " .. value or (soft and "soft " .. value or tostring(value)))
70 local cards = {}
71 for _, card in ipairs(self.cards) do cards[#cards + 1] = card_text(card) end
72 return table.concat(cards, " ") .. " (" .. status .. ")"
73end
74function Hand:check(rules, hand_count)
75 local value = self:total()
76 if value > 21 then self.finish = "bust"
77 elseif rules.CHARLIE ~= 0 and #self.cards == rules.CHARLIE then self.finish = "charlie"
78 elseif value == 21 then self.finish = "ordinary"
79 elseif self.split_aces and #self.cards == 2 then
80 local again = self.cards[1] % 13 == 0 and self.cards[2] % 13 == 0 and rules.RSA == "ON" and hand_count < rules.HANDS
81 if not again then self.finish = "ordinary" end
82 end
83end

115function Game:current() return self.players[self.current_seat].hands[self.current_hand] end
116function Game:current_name() return self.players[self.current_seat]:hand_name(self.current_hand) end
117function Game:hand_line(player, index, verb)
118 self:emit(player:hand_name(index) .. (verb and " " .. verb or "") .. ": " .. player.hands[index]:text())

120function Game:cut() return (#self.shoe * self.rules.PEN) // 100 end
121function Game:shuffle()
122 self.shoe = {}
123 for card = 0, self.decks * 52 - 1 do self.shoe[#self.shoe + 1] = card % 52 end
124 for index = #self.shoe, 2, -1 do
125 self.random = 48271 * self.random % 2147483647
126 local other = self.random % index + 1
127 self.shoe[index], self.shoe[other] = self.shoe[other], self.shoe[index]
128 end
129 self.drawn, self.running = 0, 0
130end
131function Game:draw(visible)
132 self.drawn = self.drawn + 1
133 local card = self.shoe[self.drawn]
134 if visible then self.running = self.running + hilo(card) end
135 return card
136end
137function Game:reveal_dealer()
138 if not self.hole_visible then
139 self.running = self.running + hilo(self.dealer.cards[2])
140 self.hole_visible = true
141 end
142 self:emit("dealer: " .. self.dealer:text())
143end
144function Game:result(hand)
145 if self.dealer_natural then return hand:blackjack() and "push" or "lose" end
146 if hand:blackjack() then return "blackjack" end
147 if hand.finish == "bust" or hand.finish == "surrender" or hand.finish == "charlie" then return hand.finish end
148 local own = hand:total()
149 local dealer = self.dealer:total()
150 if dealer > 21 or own > dealer then return "win" end
151 return own == dealer and "push" or "lose"
152end
153function Game:profit(bet, result)
154 if result == "blackjack" then return self.rules.BJ == "3:2" and (bet * 3) // 2 or (bet * 6) // 5 end
155 if result == "win" or result == "charlie" then return bet end
156 if result == "push" then return 0 end
157 if result == "surrender" then return bet // 2 - bet end
158 return -bet
159end
160function Game:settle()
161 for _, seat in ipairs(self.participants) do
162 local player = self.players[seat]
163 for index, hand in ipairs(player.hands) do
164 local result = self:result(hand)
165 local profit = self:profit(hand.bet, result)
166 player.bankroll = player.bankroll + hand.bet + profit
167 player.net = player.net + profit
168 local stats = player.stats
169 stats.hands = stats.hands + 1
170 local field = (result == "blackjack" or result == "win" or result == "charlie") and "wins" or (result == "push" and "pushes" or "losses")
171 stats[field] = stats[field] + 1
172 local special = result == "blackjack" and "blackjacks" or (result == "bust" and "busts" or (result == "surrender" and "surrenders" or nil))
173 if special then stats[special] = stats[special] + 1 end
174 self:emit(player:hand_name(index), result, signed(profit))
175 end
176 self:emit(player.name, "bankroll", player.bankroll)
177 end
178 for _, player in ipairs(self.players) do if player then player.bet, player.side = 0, 0 end end
179 self.phase, self.current_seat, self.current_hand = "idle", nil, nil
180end
181function Game:dealer_turn()
182 self:reveal_dealer()
183 local required = false
184 for _, seat in ipairs(self.participants) do
185 for _, hand in ipairs(self.players[seat].hands) do
186 if hand.finish == "ordinary" and not hand:blackjack() then required = true end
187 end
188 end
189 local drew = false
190 if required then
191 while true do
192 local value, soft = self.dealer:total()
193 if value > 16 and not (value == 17 and soft and self.rules.DEALER == "H17") then break end
194 self.dealer.cards[#self.dealer.cards + 1] = self:draw(true)
195 self:emit("dealer hits: " .. self.dealer:text())
196 drew = true
197 end
198 end
199 if drew and self.dealer:total() > 21 then self.dealer_busts = self.dealer_busts + 1 end
200 self:settle()
201end
202function Game:next_hand()
203 for _, seat in ipairs(self.participants) do
204 for index, hand in ipairs(self.players[seat].hands) do if not hand.finish then return seat, index end end
205 end
206end
207function Game:advance_hands()
208 while true do
209 local seat, index = self:next_hand()
210 if not seat then self:dealer_turn(); return end
211 self.current_seat, self.current_hand = seat, index
212 local player = self.players[seat]
213 local hand = player.hands[index]
214 if #hand.cards ~= 1 then return end
215 hand.cards[2] = self:draw(true)
216 self:hand_line(player, index)
217 hand:check(self.rules, #player.hands)
218 if not hand.finish then return end
219 end
220end
221function Game:peek()
222 self.dealer_natural = self.dealer:blackjack()
223 self:emit("dealer peeks:", self.dealer_natural and "blackjack" or "no blackjack")
224 for _, seat in ipairs(self.participants) do
225 local player = self.players[seat]
226 if player.insurance ~= 0 then
227 local profit = self.dealer_natural and player.insurance * 2 or -player.insurance
228 player.bankroll = player.bankroll + player.insurance + profit
229 player.net = player.net + profit
230 self:emit(player.name, "insurance", signed(profit))
231 end
232 end
233 if self.dealer_natural then
234 self.dealer_blackjacks = self.dealer_blackjacks + 1
235 self:reveal_dealer()
236 self:settle()
237 else
238 self.phase = "play"
239 self:advance_hands()
240 end
241end
242function Game:deal()
243 if self.drawn >= self:cut() then self:shuffle(); self:emit("shuffle") end
244 self.round = self.round + 1
245 self:emit("round", self.round)
246 self.participants = {}
247 for seat, player in ipairs(self.players) do
248 if player and player.bet ~= 0 then
249 self.participants[#self.participants + 1] = seat
250 player.bankroll = player.bankroll - player.bet - player.side
251 player.insurance = 0
252 player.stats.rounds = player.stats.rounds + 1
253 player.hands = { Hand.new({}, player.bet) }
254 end
255 end
256 for _, seat in ipairs(self.participants) do self.players[seat].hands[1].cards[1] = self:draw(true) end
257 self.dealer = Hand.new({ self:draw(true) })
258 for _, seat in ipairs(self.participants) do
259 local hand = self.players[seat].hands[1]
260 hand.cards[2] = self:draw(true)
261 if hand:blackjack() then hand.finish = "ordinary" end
262 end
263 self.dealer.cards[2] = self:draw(false)
264 self.hole_visible, self.dealer_natural = false, false
265 self.current_seat, self.current_hand = nil, nil
266 for _, seat in ipairs(self.participants) do self:hand_line(self.players[seat], 1) end
267 self:emit("dealer:", card_text(self.dealer.cards[1]), "??")
268 for _, seat in ipairs(self.participants) do
269 local player = self.players[seat]
270 if player.side ~= 0 then
271 local first, second = player.hands[1].cards[1], player.hands[1].cards[2]
272 local result, multiplier
273 if first % 13 ~= second % 13 then result, multiplier = "lose", -1
274 elseif first == second then result, multiplier = "perfect", 25
275 elseif black(first) == black(second) then result, multiplier = "colored", 12
276 else result, multiplier = "mixed", 6 end
277 local profit = player.side * multiplier
278 player.bankroll = player.bankroll + player.side + profit
279 player.net = player.net + profit
280 self:emit(player.name, "side", result, signed(profit))
281 end
282 end
283 local up = point(self.dealer.cards[1])
284 if up == 1 then
285 self.phase, self.insurance_turn = "insurance", 1
286 self:emit("insurance offered")
287 elseif up == 10 then self:peek()
288 else self.phase = "play"; self:advance_hands() end
289end
290function Game:action_problem(action)
291 local player, hand = self.players[self.current_seat], self:current()
292 if action == "HIT" then return hand.split_aces and "split aces" or nil end
293 if action == "DOUBLE" then
294 if hand.split_aces then return "split aces" end
295 if #hand.cards ~= 2 then return "two cards only" end
296 local value, soft = hand:total()
297 local low = self.rules.DOUBLE == "9-11" and 9 or 10
298 local limited = self.rules.DOUBLE ~= "ANY" and (soft or value < low or value > 11)
299 if limited or (self.rules.DAS == "OFF" and hand.split) then return "double not allowed" end
300 if player.bankroll < hand.bet then return "insufficient funds" end
301 elseif action == "SPLIT" then
302 if #hand.cards ~= 2 then return "two cards only" end
303 if point(hand.cards[1]) ~= point(hand.cards[2]) then return "not a pair" end
304 if #player.hands >= self.rules.HANDS then return "hand limit" end
305 if player.bankroll < hand.bet then return "insufficient funds" end
306 elseif action == "SURRENDER" then
307 if self.rules.SURRENDER == "OFF" or hand.split then return "surrender not allowed" end
308 if #hand.cards ~= 2 then return "two cards only" end
309 end
310end
311function Game:recommendation()
312 local hand = self:current()
313 local can_split = not self:action_problem("SPLIT")
314 if hand.split_aces then return can_split and "SPLIT" or "STAND" end
315 local up = point(self.dealer.cards[1])
316 if up == 1 then up = 11 end
317 if can_split then
318 local value = point(hand.cards[1])
319 local wanted = value == 1 or value == 8 or (value == 9 and up ~= 7 and up ~= 10 and up ~= 11)
320 or ((value == 2 or value == 3 or value == 7) and up <= 7) or (value == 6 and up <= 6)
321 or (value == 4 and (up == 5 or up == 6))
322 if wanted then return "SPLIT" end
323 end
324 local value, soft = hand:total()
325 local can_double = not self:action_problem("DOUBLE")
326 if soft then
327 if value >= 19 then return "STAND" end
328 if value == 18 then
329 if up >= 3 and up <= 6 then return can_double and "DOUBLE" or "STAND" end
330 return (up == 2 or up == 7 or up == 8) and "STAND" or "HIT"
331 end
332 local wanted = (value == 17 and up >= 3 and up <= 6) or ((value == 15 or value == 16) and up >= 4 and up <= 6)
333 or ((value == 13 or value == 14) and (up == 5 or up == 6))
334 return wanted and can_double and "DOUBLE" or "HIT"
335 end
336 if value >= 17 then return "STAND" end
337 if value >= 13 then
338 if up <= 6 then return "STAND" end
339 if ((value == 16 and up >= 9) or (value == 15 and up == 10)) and not self:action_problem("SURRENDER") then return "SURRENDER" end
340 return "HIT"
341 end
342 if value == 12 then return up >= 4 and up <= 6 and "STAND" or "HIT" end
343 local wanted = (value == 11 and up <= 10) or (value == 10 and up <= 9) or (value == 9 and up >= 3 and up <= 6)
344 return wanted and can_double and "DOUBLE" or "HIT"
345end
346function Game:act(action)
347 local player, index = self.players[self.current_seat], self.current_hand
348 local hand = player.hands[index]
349 if action == "SPLIT" then
350 player.bankroll = player.bankroll - hand.bet
351 player.stats.splits = player.stats.splits + 1
352 self:emit(player:hand_name(index), "splits", card_text(hand.cards[1]), card_text(hand.cards[2]))
353 local aces = hand.split_aces or point(hand.cards[1]) == 1
354 local first = Hand.new({ hand.cards[1] }, hand.bet, true, aces)
355 local second = Hand.new({ hand.cards[2] }, hand.bet, true, aces)
356 player.hands[index] = first
357 table.insert(player.hands, index + 1, second)
358 first.cards[2] = self:draw(true)
359 self:hand_line(player, index)
360 first:check(self.rules, #player.hands)
361 elseif action == "STAND" or action == "SURRENDER" then
362 hand.finish = action == "STAND" and "ordinary" or "surrender"
363 self:hand_line(player, index, action == "STAND" and "stands" or "surrenders")
364 else
365 if action == "DOUBLE" then
366 player.bankroll = player.bankroll - hand.bet
367 hand.bet = hand.bet * 2
368 player.stats.doubles = player.stats.doubles + 1
369 end
370 hand.cards[#hand.cards + 1] = self:draw(true)
371 self:hand_line(player, index, action == "DOUBLE" and "doubles" or "hits")
372 hand:check(self.rules, #player.hands)
373 if action == "DOUBLE" and not hand.finish then hand.finish = "ordinary" end
374 end
375 self:advance_hands()
376end
377function Game:status()
378 local headline = "table " .. self.phase .. " round " .. self.round
379 if self.phase == "insurance" then headline = headline .. " next " .. self.players[self.participants[self.insurance_turn]].name
380 elseif self.phase == "play" then headline = headline .. " turn " .. self:current_name() end
381 self:emit(headline)
382 if self.phase ~= "idle" then self:emit("dealer:", card_text(self.dealer.cards[1]), "??") end
383 local participating = {}
384 for _, seat in ipairs(self.participants) do participating[seat] = true end
385 for seat, player in ipairs(self.players) do
386 if player then
387 self:emit("seat", seat, player.name, "bankroll", player.bankroll, "bet", player.bet, "side", player.side)
388 if self.phase ~= "idle" and participating[seat] then
389 for index, hand in ipairs(player.hands) do self:emit(player:hand_name(index) .. ": " .. hand:text(), "bet", hand.bet) end
390 end
391 end
392 end
393end
394function Game:statistics()
395 for _, player in ipairs(self.players) do
396 if player then
397 local fields = { "stats", player.name }
398 for _, name in ipairs(stat_names) do fields[#fields + 1], fields[#fields + 2] = name, tostring(player.stats[name]) end
399 fields[#fields + 1], fields[#fields + 2] = "net", signed(player.net)
400 self:emit(table.concat(fields, " "))
401 end
402 end
403 self:emit("stats dealer rounds", self.round, "blackjacks", self.dealer_blackjacks, "busts", self.dealer_busts)
404end
405function Game:command(words)
406 self.output = {}
407 local op = words[1]
408 if not commands[op] then fail("unknown command " .. op) end
409 local seat, amount, seed, proposed
410 if op == "SHOE" then
411 amount = integer_word(words[2], 1, 8)
412 seed = integer_word(words[3], 1, 2147483646)
413 elseif op == "RULE" then proposed = rule_value(words[2], words[3])
414 elseif op == "PLAYER" then
415 if #words[2] > 8 or not words[2]:match("^[A-Z]+$") then fail("bad name " .. words[2]) end
416 amount = integer_word(words[3], 1, 1000000)
417 elseif player_commands[op] then
418 seat = self:lookup(words[2])
419 if amount_commands[op] then amount = integer_word(words[3], op == "REBUY" and 1 or 0, 1000000) end
420 end
421 if idle_commands[op] and self.phase ~= "idle" then fail("round in progress") end
422 if op == "INSURANCE" and self.phase ~= "insurance" then fail("no insurance") end
423 local action_command = actions[op] or op == "AUTO"
424 if action_command or op == "HINT" then
425 if self.phase == "idle" then fail("no round") end
426 if self.phase == "insurance" then fail("insurance pending") end
427 if action_command and seat ~= self.current_seat then fail("waiting for " .. self:current_name()) end
428 end
429 if action_command then
430 if op == "AUTO" then
431 while self.phase == "play" and self.current_seat == seat do self:act(self:recommendation()) end
432 else
433 local problem = self:action_problem(op)
434 if problem then fail(problem) end
435 self:act(op)
436 end
437 return self.output
438 end
439 if op == "SHOE" then
440 self.decks, self.random = amount, seed
441 self:shuffle()
442 self:emit("shoe", amount, "decks", #self.shoe, "cards cut", self:cut())
443 elseif op == "RULE" then
444 local changed = copy(self.rules)
445 changed[words[2]] = proposed
446 if changed.MIN > changed.MAX then fail("min above max") end
447 self.rules = changed
448 self:emit("rule", words[2], proposed)
449 elseif op == "RULES" then
450 local fields = { "rules" }
451 for _, name in ipairs(rule_names) do fields[#fields + 1], fields[#fields + 2] = name, tostring(self.rules[name]) end
452 self:emit(table.concat(fields, " "))
453 elseif op == "PLAYER" then
454 if self:find_player(words[2]) then fail("player exists") end
455 local empty
456 for index, player in ipairs(self.players) do if not player then empty = index; break end end
457 if not empty then fail("table full") end
458 self.players[empty] = Player.new(words[2], amount)
459 self:emit(words[2], "sits at seat", empty, "with", amount)
460 elseif op == "LEAVE" then
461 local player = self.players[seat]
462 self:emit(player.name, "leaves with", player.bankroll, "net", signed(player.net))
463 self.players[seat] = false
464 elseif op == "REBUY" then
465 local player = self.players[seat]
466 player.bankroll = player.bankroll + amount
467 self:emit(player.name, "rebuys", amount, "bankroll", player.bankroll)
468 elseif op == "BET" then
469 local player = self.players[seat]
470 if amount ~= 0 then
471 if amount < self.rules.MIN or amount > self.rules.MAX then fail("bet out of limits") end
472 if amount + player.side > player.bankroll then fail("insufficient funds") end
473 end
474 player.bet = amount
475 if amount == 0 then player.side = 0; self:emit(player.name, "bet cleared")
476 else self:emit(player.name, "bets", amount) end
477 elseif op == "SIDE" then
478 local player = self.players[seat]
479 if amount ~= 0 then
480 if player.bet == 0 then fail("no bet") end
481 if amount > self.rules.SIDEMAX then fail("side out of limits") end
482 if player.bet + amount > player.bankroll then fail("insufficient funds") end
483 end
484 player.side = amount
485 if amount == 0 then self:emit(player.name, "side cleared") else self:emit(player.name, "side", amount) end
486 elseif op == "DEAL" then
487 if self.decks == 0 then fail("no shoe") end
488 local has_bet = false
489 for _, player in ipairs(self.players) do if player and player.bet ~= 0 then has_bet = true end end
490 if not has_bet then fail("no bets") end
491 self:deal()
492 elseif op == "INSURANCE" then
493 local waiting = self.participants[self.insurance_turn]
494 if seat ~= waiting then fail("waiting for " .. self.players[waiting].name) end
495 local player = self.players[seat]
496 if amount > player.bet // 2 then fail("insurance out of limits") end
497 if amount > player.bankroll then fail("insufficient funds") end
498 player.bankroll = player.bankroll - amount
499 player.insurance = amount
500 if amount == 0 then self:emit(player.name, "declines insurance") else self:emit(player.name, "insures", amount) end
501 self.insurance_turn = self.insurance_turn + 1
