82 command := Command(word)

129func integer(word string, low, high int64) (int64, error) {
130 if len(word) == 0 || len(word) > 10 || len(word) > 1 && word[0] == '0' {
131 return 0, fmt.Errorf("bad value %s", word)
132 }
133 var value int64
134 for i := range len(word) {
135 if word[i] < '0' || word[i] > '9' {
136 return 0, fmt.Errorf("bad value %s", word)
137 }
138 value = value*10 + int64(word[i]-'0')
139 }
140 if value < low || value > high {
141 return 0, fmt.Errorf("bad value %s", word)
142 }
143 return value, nil
144}

160type Rules struct {

171func defaultRules() Rules {
172 return Rules{pen: 75, minimum: 10, maximum: 500, das: true, doubling: doubleAny,
173 hands: 4, surrender: true, sideMaximum: 25}
174}

176func (rules Rules) changed(name, word string) (Rules, error) {
177 var err error
178 switch name {
179 case "DEALER":
180 if word != "S17" && word != "H17" {
181 return rules, fmt.Errorf("bad value %s", word)
182 }
183 rules.hitSoft17 = word == "H17"
184 case "BJ":
185 if word != "3:2" && word != "6:5" {
186 return rules, fmt.Errorf("bad value %s", word)
187 }
188 rules.sixToFive = word == "6:5"
189 case "DOUBLE":
190 value := DoubleRule(word)
191 if value != doubleAny && value != doubleHard9 && value != doubleHard10 {
192 return rules, fmt.Errorf("bad value %s", word)
193 }
194 rules.doubling = value
195 case "DAS":
196 rules.das, err = parseSwitch(word)
197 case "RSA":
198 rules.rsa, err = parseSwitch(word)
199 case "SURRENDER":
200 rules.surrender, err = parseSwitch(word)
201 case "PEN":
202 value, failure := integer(word, 50, 90)
203 rules.pen, err = int(value), failure
204 case "MIN":
205 value, failure := integer(word, 1, 10000)
206 rules.minimum, err = Money(value), failure
207 case "MAX":
208 value, failure := integer(word, 1, 10000)
209 rules.maximum, err = Money(value), failure
210 case "HANDS":
211 value, failure := integer(word, 1, 4)
212 rules.hands, err = int(value), failure
213 case "CHARLIE":
214 value, failure := integer(word, 0, 7)
215 if failure != nil {
216 return rules, failure
217 }
218 if value != 0 && value < 5 {
219 return rules, fmt.Errorf("bad value %s", word)
220 }
221 rules.charlie = int(value)
222 case "SIDEMAX":
223 value, failure := integer(word, 0, 1000)
224 rules.sideMaximum, err = Money(value), failure
225 default:
226 return rules, fmt.Errorf("unknown rule %s", name)
227 }
228 return rules, err
229}

231func (rules Rules) String() string {
232 dealer, bj := "S17", "3:2"
233 if rules.hitSoft17 {
234 dealer = "H17"
235 }
236 if rules.sixToFive {
237 bj = "6:5"
238 }
239 return fmt.Sprintf("rules DEALER %s PEN %d MIN %d MAX %d BJ %s DAS %s DOUBLE %s HANDS %d RSA %s SURRENDER %s CHARLIE %d SIDEMAX %d",
240 dealer, rules.pen, rules.minimum, rules.maximum, bj, switchName(rules.das), rules.doubling,
241 rules.hands, switchName(rules.rsa), switchName(rules.surrender), rules.charlie, rules.sideMaximum)
242}

246func (card Card) point() int { return min(card.rank+1, 10) }
247func (card Card) black() bool { return card.suit == 0 || card.suit == 3 }
248func (card Card) String() string {

251func (card Card) hilo() int {
252 value := card.point()
253 if value >= 2 && value <= 6 {
254 return 1
255 }
256 if value == 1 || value == 10 {
257 return -1
258 }
259 return 0

274func (hand *Hand) total() Total {
275 sum, ace := 0, false
276 for _, card := range hand.cards {
277 sum += card.point()
278 ace = ace || card.rank == 0
279 }
280 soft := ace && sum+10 <= 21
281 if soft {
282 sum += 10
283 }
284 return Total{sum, soft}
285}

287func (hand *Hand) blackjack() bool {
288 return !hand.split && len(hand.cards) == 2 && hand.total().value == 21
289}

291func (hand *Hand) String() string {
292 cards := make([]string, len(hand.cards))
293 for i, card := range hand.cards {
294 cards[i] = card.String()
295 }
296 total := hand.total()
297 status := strconv.Itoa(total.value)
298 if hand.blackjack() {
299 status = "blackjack"
300 } else if total.value > 21 {
301 status = "bust " + status
302 } else if total.soft {
303 status = "soft " + status
304 }
305 return strings.Join(cards, " ") + " (" + status + ")"
306}

308func (hand *Hand) outcome(dealer int, dealerBlackjack bool) Outcome {
309 if dealerBlackjack {
310 if hand.blackjack() {
311 return pushOutcome
312 }
313 return loseOutcome
314 }
315 if hand.blackjack() {
316 return blackjackOutcome
317 }
318 switch hand.reason {
319 case bustFinish:
320 return bustOutcome
321 case surrenderFinish:
322 return surrenderOutcome
323 case charlieFinish:
324 return charlieOutcome
325 }
326 value := hand.total().value
327 if dealer > 21 || value > dealer {
328 return winOutcome
329 }
330 if value == dealer {
331 return pushOutcome
332 }
333 return loseOutcome
334}

336func (result Outcome) profit(bet Money, rules Rules) Money {
337 switch result {
338 case blackjackOutcome:
339 if rules.sixToFive {
340 return bet * 6 / 5
341 }
342 return bet * 3 / 2
343 case winOutcome, charlieOutcome:
344 return bet
345 case pushOutcome:
346 return 0
347 case surrenderOutcome:
348 return bet/2 - bet
349 default:
350 return -bet
351 }
352}

404 shoe := &Shoe{decks: decks, random: seed}

409func (shoe *Shoe) shuffle() {
410 shoe.cards = make([]Card, 52*shoe.decks)
411 for i := range shoe.cards {
412 shoe.cards[i] = Card{i % 13, i / 13 % 4}
413 }
414 for i := len(shoe.cards) - 1; i > 0; i-- {
415 shoe.random = shoe.random * 48271 % 2147483647
416 j := int(shoe.random % int64(i+1))
417 shoe.cards[i], shoe.cards[j] = shoe.cards[j], shoe.cards[i]
418 }
419 shoe.drawn, shoe.running = 0, 0
420}

422func (shoe *Shoe) cut(pen int) int { return len(shoe.cards) * pen / 100 }
423func (shoe *Shoe) draw(visible bool) Card {
424 card := shoe.cards[shoe.drawn]
425 shoe.drawn++
426 if visible {
427 shoe.running += card.hilo()
428 }
429 return card
430}

495func (table *Table) finishCheck(player *Player, hand *Hand) {
496 value := hand.total().value
497 switch {
498 case value > 21:
499 hand.finished, hand.reason = true, bustFinish
500 case table.rules.charlie != 0 && len(hand.cards) == table.rules.charlie:
501 hand.finished, hand.reason = true, charlieFinish
502 case value == 21:
503 hand.finished = true
504 case hand.splitAces && len(hand.cards) == 2:
505 resplit := hand.cards[0].rank == 0 && hand.cards[1].rank == 0 && table.rules.rsa && len(player.hands) < table.rules.hands
506 hand.finished = !resplit
507 }
508}

510func (table *Table) settle(dealerBlackjack bool) {
511 dealer := table.dealer.total().value
512 for _, player := range table.participants {
513 for index, hand := range player.hands {
514 result := hand.outcome(dealer, dealerBlackjack)
515 profit := result.profit(hand.bet, table.rules)
516 player.bankroll += hand.bet + profit
517 player.net += profit
518 player.stats.record(result)
519 table.emit("%s %s %+d", player.handName(index), result, profit)
520 }
521 table.emit("%s bankroll %d", player.name, player.bankroll)
522 }
523 for _, player := range table.seated() {
524 player.bet, player.side = 0, 0
525 }
526 table.state = idle
527}

529func (table *Table) dealerTurn() {
530 table.reveal()
531 needsDraw := false
532 for _, player := range table.participants {
533 for _, hand := range player.hands {
534 if hand.reason == ordinary && !hand.blackjack() {
535 needsDraw = true
536 }
537 }
538 }
539 drew := false
540 if needsDraw {
541 for {
542 total := table.dealer.total()
543 if !(total.value <= 16 || total.value == 17 && total.soft && table.rules.hitSoft17) {
544 break
545 }
546 table.dealer.cards = append(table.dealer.cards, table.shoe.draw(true))
547 drew = true
548 table.emit("dealer hits: %s", &table.dealer)
549 }
550 }
551 if drew && table.dealer.total().value > 21 {
552 table.dealerBusts++
553 }
554 table.settle(false)
555}

557func (table *Table) advance() {
558 for {
559 turn, ok := table.current()
560 if !ok {
561 table.dealerTurn()
562 return
563 }
564 if len(turn.hand.cards) != 1 {
565 return
566 }
567 turn.hand.cards = append(turn.hand.cards, table.shoe.draw(true))
568 table.emit("%s: %s", turn.name(), turn.hand)
569 table.finishCheck(turn.player, turn.hand)
570 if !turn.hand.finished {
571 return
572 }
573 }
574}

576func (table *Table) peek() {
577 natural := table.dealer.blackjack()
578 if natural {
579 table.emit("dealer peeks: blackjack")
580 } else {
581 table.emit("dealer peeks: no blackjack")
582 }
583 for _, player := range table.participants {
584 if player.insurance == 0 {
585 continue
586 }
587 profit := -player.insurance
588 if natural {
589 profit = player.insurance * 2
590 }
591 player.bankroll += player.insurance + profit
592 player.net += profit
593 table.emit("%s insurance %+d", player.name, profit)
594 }
595 if natural {
596 table.dealerBlackjacks++
597 table.reveal()
598 table.settle(true)
599 } else {
600 table.state = play
601 table.advance()
602 }
603}

605func (table *Table) deal() {
606 if table.shoe.drawn >= table.shoe.cut(table.rules.pen) {
607 table.shoe.shuffle()
608 table.emit("shuffle")
609 }
610 table.round++
611 table.emit("round %d", table.round)
612 table.participants = nil
613 for _, player := range table.seated() {
614 if player.bet != 0 {
615 table.participants = append(table.participants, player)
616 player.bankroll -= player.bet + player.side
617 player.insurance = 0
618 player.stats.rounds++
619 player.hands = []*Hand{{bet: player.bet}}
620 }
621 }
622 table.dealer = Hand{}
623 table.holeVisible = false
624 for _, player := range table.participants {
625 player.hands[0].cards = append(player.hands[0].cards, table.shoe.draw(true))
626 }
627 table.dealer.cards = append(table.dealer.cards, table.shoe.draw(true))
628 for _, player := range table.participants {
629 hand := player.hands[0]
630 hand.cards = append(hand.cards, table.shoe.draw(true))
631 hand.finished = hand.blackjack()
632 }
633 table.dealer.cards = append(table.dealer.cards, table.shoe.draw(false))
634 for _, player := range table.participants {
635 table.emit("%s: %s", player.handName(0), player.hands[0])
636 }
637 table.emit("dealer: %s ??", table.dealer.cards[0])
638 for _, player := range table.participants {
639 if player.side == 0 {
640 continue
641 }
642 a, b := player.hands[0].cards[0], player.hands[0].cards[1]
643 result, multiplier := "lose", Money(-1)
644 if a.rank == b.rank {
645 switch {
646 case a.suit == b.suit:
647 result, multiplier = "perfect", 25
648 case a.black() == b.black():
649 result, multiplier = "colored", 12
650 default:
651 result, multiplier = "mixed", 6
652 }
653 }
654 profit := player.side * multiplier
655 player.bankroll += player.side + profit
656 player.net += profit
657 table.emit("%s side %s %+d", player.name, result, profit)
658 }
659 switch table.dealer.cards[0].point() {
660 case 1:
661 table.state = insurance
662 table.insuranceTurn = 0
663 table.emit("insurance offered")
664 case 10:
665 table.peek()
666 default:
667 table.state = play
668 table.advance()
669 }
670}

672func (table *Table) actionProblem(action Action, player *Player, hand *Hand) string {
673 switch action {
674 case hitAction:
675 if hand.splitAces {
676 return "split aces"
677 }
678 case doubleAction:
679 if hand.splitAces {
680 return "split aces"
681 }
682 if len(hand.cards) != 2 {
683 return "two cards only"
684 }
685 total := hand.total()
686 low := 9
687 if table.rules.doubling == doubleHard10 {
688 low = 10
689 }
690 if table.rules.doubling != doubleAny && (total.soft || total.value < low || total.value > 11) || !table.rules.das && hand.split {
691 return "double not allowed"
692 }
693 if player.bankroll < hand.bet {
694 return "insufficient funds"
695 }
696 case splitAction:
697 if len(hand.cards) != 2 {
698 return "two cards only"
699 }
700 if hand.cards[0].point() != hand.cards[1].point() {
701 return "not a pair"
702 }
703 if len(player.hands) >= table.rules.hands {
704 return "hand limit"
705 }
706 if player.bankroll < hand.bet {
707 return "insufficient funds"
708 }
709 case surrenderAction:
710 if !table.rules.surrender || hand.split {
711 return "surrender not allowed"
712 }
713 if len(hand.cards) != 2 {
714 return "two cards only"
715 }
716 }
717 return ""
718}

720func (table *Table) recommend() Action {
721 turn, _ := table.current()
722 hand, player := turn.hand, turn.player
723 possible := func(action Action) bool { return table.actionProblem(action, player, hand) == "" }
724 up := table.dealer.cards[0].point()
725 if up == 1 {
726 up = 11
727 }
728 if hand.splitAces {
729 if possible(splitAction) {
730 return splitAction
731 }
732 return standAction
733 }
734 if possible(splitAction) {
735 pair := hand.cards[0].point()
736 if pair == 1 || pair == 8 || pair == 9 && up != 7 && up != 10 && up != 11 ||
737 (pair == 2 || pair == 3 || pair == 7) && up <= 7 || pair == 6 && up <= 6 || pair == 4 && (up == 5 || up == 6) {
738 return splitAction
739 }
740 }
741 total := hand.total()
742 value, symbol := total.value, "H"
743 if total.soft {
744 switch {
745 case value >= 19:
746 symbol = "S"
747 case value == 18:
748 if up >= 3 && up <= 6 {
749 symbol = "Ds"
750 } else if up == 2 || up == 7 || up == 8 {
751 symbol = "S"
752 }
753 case value == 17 && up >= 3 && up <= 6,
754 (value == 15 || value == 16) && up >= 4 && up <= 6,
755 (value == 13 || value == 14) && (up == 5 || up == 6):
756 symbol = "D"
757 }
758 } else {
759 switch {
760 case value >= 17:
761 symbol = "S"
762 case value >= 13:
763 if up <= 6 {
764 symbol = "S"
765 } else if value == 16 && up >= 9 || value == 15 && up == 10 {
766 symbol = "R"
767 }
768 case value == 12:
769 if up >= 4 && up <= 6 {
770 symbol = "S"
771 }
772 case value == 11 && up <= 10, value == 10 && up <= 9, value == 9 && up >= 3 && up <= 6:
773 symbol = "D"
774 }
775 }
776 switch symbol {
777 case "S":
778 return standAction
779 case "D", "Ds":
780 if possible(doubleAction) {
781 return doubleAction
782 }
783 if symbol == "Ds" {
784 return standAction
785 }
786 case "R":
787 if possible(surrenderAction) {
788 return surrenderAction
789 }
790 }
791 return hitAction
792}

794func (table *Table) perform(action Action) {
795 turn, _ := table.current()
796 player, hand, name := turn.player, turn.hand, turn.name()
797 switch action {
798 case splitAction:
799 first, second := hand.cards[0], hand.cards[1]
800 aces := hand.splitAces || first.rank == 0
801 player.bankroll -= hand.bet
802 player.stats.splits++
803 table.emit("%s splits %s %s", name, first, second)
804 firstHand := &Hand{cards: []Card{first}, bet: hand.bet, split: true, splitAces: aces}
805 secondHand := &Hand{cards: []Card{second}, bet: hand.bet, split: true, splitAces: aces}
806 player.hands[turn.index] = firstHand
807 player.hands = slices.Insert(player.hands, turn.index+1, secondHand)
808 firstHand.cards = append(firstHand.cards, table.shoe.draw(true))
809 table.emit("%s: %s", name, firstHand)
810 table.finishCheck(player, firstHand)
811 case standAction, surrenderAction:
812 hand.finished = true
813 verb := "stands"
814 if action == surrenderAction {
815 hand.reason, verb = surrenderFinish, "surrenders"
816 }
817 table.emit("%s %s: %s", name, verb, hand)
818 case hitAction, doubleAction:
819 verb := "hits"
820 if action == doubleAction {
821 player.bankroll -= hand.bet
822 hand.bet *= 2
823 player.stats.doubles++
824 verb = "doubles"
825 }
826 hand.cards = append(hand.cards, table.shoe.draw(true))
827 table.emit("%s %s: %s", name, verb, hand)
828 table.finishCheck(player, hand)
829 if action == doubleAction {
830 hand.finished = true
831 }
832 }
833 table.advance()
834}

836func (table *Table) status() {
837 switch table.state {
838 case idle:
839 table.emit("table idle round %d", table.round)
840 case insurance:
841 table.emit("table insurance round %d next %s", table.round, table.participants[table.insuranceTurn].name)
842 case play:
843 turn, _ := table.current()
844 table.emit("table play round %d turn %s", table.round, turn.name())
845 }
846 if table.state != idle {
847 table.emit("dealer: %s ??", table.dealer.cards[0])
848 }
849 for _, player := range table.seated() {
850 table.emit("seat %d %s bankroll %d bet %d side %d", player.seat+1, player.name, player.bankroll, player.bet, player.side)
851 if table.state != idle && slices.Contains(table.participants, player) {
852 for index, hand := range player.hands {
853 table.emit("%s: %s bet %d", player.handName(index), hand, hand.bet)
854 }
855 }
856 }
857}

859func (table *Table) execute(words []string) ([]string, error) {
860 table.output = nil
861 command, err := parseCommand(words[0])
862 if err != nil {
863 return nil, err
864 }
865 var player *Player
866 var amount, seed int64
867 var proposed Rules
868 switch command {
869 case shoeCommand:
870 amount, err = integer(words[1], 1, 8)
871 if err != nil {
872 return nil, err
873 }
874 seed, err = integer(words[2], 1, 2147483646)
875 case ruleCommand:
876 proposed, err = table.rules.changed(words[1], words[2])
877 case playerCommand:
878 name := words[1]
879 if len(name) == 0 || len(name) > 8 {
880 return nil, fmt.Errorf("bad name %s", name)
881 }
882 for i := range len(name) {
883 if name[i] < 'A' || name[i] > 'Z' {
884 return nil, fmt.Errorf("bad name %s", name)
885 }
886 }
887 amount, err = integer(words[2], 1, 1000000)
888 default:
889 if command.namedPlayer() {
890 player = table.find(words[1])
891 if player == nil {
892 return nil, fmt.Errorf("unknown player %s", words[1])
893 }
894 switch command {
895 case rebuyCommand:
896 amount, err = integer(words[2], 1, 1000000)
897 case betCommand, sideCommand, insuranceCommand:
898 amount, err = integer(words[2], 0, 1000000)
899 }
900 }
901 }
902 if err != nil {
903 return nil, err
904 }
905 if command.idleOnly() && table.state != idle {
906 return nil, errors.New("round in progress")
907 }
908 if command == insuranceCommand && table.state != insurance {
909 return nil, errors.New("no insurance")
910 }
911 action := command.action()
912 if action != "" || command == autoCommand || command == hintCommand {
913 if table.state == idle {
914 return nil, errors.New("no round")
915 }
916 if table.state == insurance {
917 return nil, errors.New("insurance pending")
918 }
919 }
920 if action != "" || command == autoCommand {
921 turn, _ := table.current()
922 if turn.player != player {
923 return nil, fmt.Errorf("waiting for %s", turn.name())
924 }
925 if action != "" {
926 if problem := table.actionProblem(action, player, turn.hand); problem != "" {
927 return nil, errors.New(problem)
928 }
929 table.perform(action)
930 } else {
931 for table.state == play {
932 turn, _ := table.current()
933 if turn.player != player {
934 break
935 }
936 table.perform(table.recommend())
937 }
938 }
939 return table.output, nil
940 }
941 switch command {
942 case shoeCommand:
943 table.shoe = newShoe(int(amount), seed)
944 table.emit("shoe %d decks %d cards cut %d", amount, len(table.shoe.cards), table.shoe.cut(table.rules.pen))
945 case ruleCommand:
946 if proposed.minimum > proposed.maximum {
947 return nil, errors.New("min above max")
948 }
949 table.rules = proposed
950 table.emit("rule %s %s", words[1], words[2])
951 case rulesCommand:
952 table.emit("%s", table.rules)
953 case playerCommand:
954 if table.find(words[1]) != nil {
955 return nil, errors.New("player exists")
956 }
957 seat := slices.Index(table.seats[:], (*Player)(nil))
958 if seat < 0 {
959 return nil, errors.New("table full")
960 }
961 table.seats[seat] = &Player{name: words[1], seat: seat, bankroll: Money(amount)}
962 table.emit("%s sits at seat %d with %d", words[1], seat+1, amount)
963 case leaveCommand:
964 table.emit("%s leaves with %d net %+d", player.name, player.bankroll, player.net)
965 table.seats[player.seat] = nil
966 case rebuyCommand:
967 player.bankroll += Money(amount)
968 table.emit("%s rebuys %d bankroll %d", player.name, amount, player.bankroll)
969 case betCommand:
970 if amount != 0 {
971 if Money(amount) < table.rules.minimum || Money(amount) > table.rules.maximum {
972 return nil, errors.New("bet out of limits")
973 }
974 if Money(amount)+player.side > player.bankroll {
975 return nil, errors.New("insufficient funds")
976 }
977 }
978 player.bet = Money(amount)
979 if amount == 0 {
980 player.side = 0
981 table.emit("%s bet cleared", player.name)
982 } else {
983 table.emit("%s bets %d", player.name, amount)
984 }
985 case sideCommand:
986 if amount != 0 {
987 if player.bet == 0 {
988 return nil, errors.New("no bet")
989 }
990 if Money(amount) > table.rules.sideMaximum {
991 return nil, errors.New("side out of limits")
992 }
993 if player.bet+Money(amount) > player.bankroll {
994 return nil, errors.New("insufficient funds")
995 }
996 }
997 player.side = Money(amount)
998 if amount == 0 {
999 table.emit("%s side cleared", player.name)
1000 } else {
1001 table.emit("%s side %d", player.name, amount)
1002 }
1003 case dealCommand:
1004 if table.shoe == nil {
1005 return nil, errors.New("no shoe")
1006 }
1007 hasBet := false
1008 for _, player := range table.seated() {
1009 hasBet = hasBet || player.bet != 0
1010 }
1011 if !hasBet {
1012 return nil, errors.New("no bets")
1013 }
1014 table.deal()
1015 case insuranceCommand:
1016 if player != table.participants[table.insuranceTurn] {
1017 return nil, fmt.Errorf("waiting for %s", table.participants[table.insuranceTurn].name)
1018 }
1019 if Money(amount) > player.bet/2 {
1020 return nil, errors.New("insurance out of limits")
1021 }
1022 if Money(amount) > player.bankroll {
1023 return nil, errors.New("insufficient funds")
1024 }
1025 player.bankroll -= Money(amount)
1026 player.insurance = Money(amount)
1027 if amount == 0 {
1028 table.emit("%s declines insurance", player.name)
1029 } else {
1030 table.emit("%s insures %d", player.name, amount)
1031 }
1032 table.insuranceTurn++
1033 if table.insuranceTurn == len(table.participants) {
1034 table.peek()
1035 }
1036 case hintCommand:
1037 turn, _ := table.current()
1038 table.emit("hint %s: %s", turn.name(), table.recommend())
1039 case statusCommand:
1040 table.status()
1041 case countCommand:
1042 if table.shoe == nil {
1043 return nil, errors.New("no shoe")
1044 }
1045 remaining := len(table.shoe.cards) - table.shoe.drawn
1046 running := table.shoe.running
1047 magnitude, sign := running, ""
1048 if running < 0 {
1049 magnitude, sign = -running, "-"
1050 }
1051 scaled := int64(magnitude) * 5200 / int64(remaining)
1052 table.emit("count running %d true %s%d.%02d remaining %d", running, sign, scaled/100, scaled%100, remaining)
1053 case statsCommand:
1054 players := table.seated()
1055 if len(players) == 0 {
1056 return nil, errors.New("no players")
1057 }
1058 for _, player := range players {
1059 table.emit("stats %s %s net %+d", player.name, player.stats, player.net)
1060 }
1061 table.emit("stats dealer rounds %d blackjacks %d busts %d", table.round, table.dealerBlackjacks, table.dealerBusts)
1062 }
1063 return table.output, nil
1064}

1066func main() {
1067 input := bufio.NewReader(os.Stdin)
1068 output := bufio.NewWriter(os.Stdout)
1069 defer output.Flush()
1070 line, _ := input.ReadString('\n')

1072 table := Table{rules: defaultRules()}

1078 } else {

1081 }
1082 }
1083 }
