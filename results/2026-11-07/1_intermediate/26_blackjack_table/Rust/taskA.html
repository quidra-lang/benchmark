10impl Card {

12 fn black(self) -> bool { self.suit == 0 || self.suit == 3 }
13 fn hilo(self) -> i32 { match self.point() { 2..=6 => 1, 7..=9 => 0, _ => -1 } }

22#[derive(Clone, Copy)]
23struct Total { value: i32, soft: bool }
24
25#[derive(Clone, Copy, PartialEq, Eq)]
26enum Finish { Ordinary, Bust, Surrender, Charlie }
27
28#[derive(Default)]
29struct Hand {
30 cards: Vec<Card>,
31 bet: Money,
32 split: bool,
33 split_aces: bool,
34 finish: Option<Finish>,
35}
36
37impl Hand {
38 fn new(bet: Money) -> Self { Self { bet, ..Self::default() } }
39 fn total(&self) -> Total {
40 let sum: i32 = self.cards.iter().map(|card| card.point()).sum();
41 let soft = self.cards.iter().any(|card| card.rank == 0) && sum + 10 <= 21;
42 Total { value: sum + if soft { 10 } else { 0 }, soft }
43 }
44 fn blackjack(&self) -> bool { !self.split && self.cards.len() == 2 && self.total().value == 21 }
45 fn automatic_finish(&mut self, rules: Rules, hands: usize) {
46 let total = self.total().value;
47 self.finish = if total > 21 { Some(Finish::Bust) }
48 else if rules.charlie != 0 && self.cards.len() == rules.charlie { Some(Finish::Charlie) }
49 else if total == 21 { Some(Finish::Ordinary) }
50 else if self.split_aces && self.cards.len() == 2 &&
51 !(self.cards.iter().all(|card| card.rank == 0) && rules.rsa && hands < rules.hands) {
52 Some(Finish::Ordinary)
53 } else { None };
54 }
55 fn outcome(&self, dealer: i32, dealer_blackjack: bool) -> Outcome {
56 if dealer_blackjack { return if self.blackjack() { Outcome::Push } else { Outcome::Lose }; }
57 if self.blackjack() { return Outcome::Blackjack; }
58 match self.finish {
59 Some(Finish::Bust) => Outcome::Bust,
60 Some(Finish::Surrender) => Outcome::Surrender,
61 Some(Finish::Charlie) => Outcome::Charlie,
62 _ if dealer > 21 || self.total().value > dealer => Outcome::Win,
63 _ if self.total().value == dealer => Outcome::Push,
64 _ => Outcome::Lose,
65 }
66 }
67}
68
69impl Display for Hand {
70 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
71 for (index, card) in self.cards.iter().enumerate() {
72 if index != 0 { write!(f, " ")?; }
73 write!(f, "{card}")?;
74 }
75 let total = self.total();
76 if self.blackjack() { write!(f, " (blackjack)") }
77 else if total.value > 21 { write!(f, " (bust {})", total.value) }
78 else if total.soft { write!(f, " (soft {})", total.value) }
79 else { write!(f, " ({})", total.value) }
80 }
81}

114fn integer(word: &str, low: i64, high: i64) -> CommandResult<i64> {
115 let bytes = word.as_bytes();
116 if bytes.is_empty() || bytes.len() > 10 || (bytes.len() > 1 && bytes[0] == b'0') ||
117 !bytes.iter().all(u8::is_ascii_digit) {
118 return Err(format!("bad value {word}"));
119 }
120 let value = word.parse::<i64>().unwrap();
121 if value < low || value > high { Err(format!("bad value {word}")) } else { Ok(value) }
122}
123
124fn switch(word: &str) -> CommandResult<bool> {
125 match word { "ON" => Ok(true), "OFF" => Ok(false), _ => Err(format!("bad value {word}")) }
126}
127fn switch_name(value: bool) -> &'static str { if value { "ON" } else { "OFF" } }
128
129impl Rules {
130 fn changed(self, name: &str, word: &str) -> CommandResult<Self> {
131 let mut changed = self;
132 match name {
133 "DEALER" => changed.dealer = match word {
134 "S17" => DealerRule::S17, "H17" => DealerRule::H17,
135 _ => return Err(format!("bad value {word}")),
136 },
137 "PEN" => changed.pen = integer(word, 50, 90)? as usize,
138 "MIN" => changed.min = integer(word, 1, 10000)?,
139 "MAX" => changed.max = integer(word, 1, 10000)?,
140 "BJ" => changed.bj = match word {
141 "3:2" => BlackjackPay::ThreeTwo, "6:5" => BlackjackPay::SixFive,
142 _ => return Err(format!("bad value {word}")),
143 },
144 "DAS" => changed.das = switch(word)?,
145 "DOUBLE" => changed.doubling = match word {
146 "ANY" => DoubleRule::Any, "9-11" => DoubleRule::NineEleven, "10-11" => DoubleRule::TenEleven,
147 _ => return Err(format!("bad value {word}")),
148 },
149 "HANDS" => changed.hands = integer(word, 1, 4)? as usize,
150 "RSA" => changed.rsa = switch(word)?,
151 "SURRENDER" => changed.surrender = switch(word)?,
152 "CHARLIE" => {
153 let value = integer(word, 0, 7)?;
154 if value != 0 && value < 5 { return Err(format!("bad value {word}")); }
155 changed.charlie = value as usize;
156 }
157 "SIDEMAX" => changed.side_max = integer(word, 0, 1000)?,
158 _ => return Err(format!("unknown rule {name}")),
159 }
160 Ok(changed)
161 }
162}

164impl Display for Rules {
165 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
166 let dealer = if self.dealer == DealerRule::S17 { "S17" } else { "H17" };
167 let bj = if self.bj == BlackjackPay::ThreeTwo { "3:2" } else { "6:5" };
168 let doubling = match self.doubling { DoubleRule::Any => "ANY", DoubleRule::NineEleven => "9-11", DoubleRule::TenEleven => "10-11" };
169 write!(f, "DEALER {dealer} PEN {} MIN {} MAX {} BJ {bj} DAS {} DOUBLE {doubling} HANDS {} RSA {} SURRENDER {} CHARLIE {} SIDEMAX {}",
170 self.pen, self.min, self.max, switch_name(self.das), self.hands, switch_name(self.rsa),
171 switch_name(self.surrender), self.charlie, self.side_max)
172 }

175#[derive(Clone, Copy, PartialEq, Eq)]
176enum Outcome { Blackjack, Win, Lose, Push, Bust, Surrender, Charlie }
177
178impl Outcome {
179 fn name(self) -> &'static str {
180 match self { Self::Blackjack => "blackjack", Self::Win => "win", Self::Lose => "lose",
181 Self::Push => "push", Self::Bust => "bust", Self::Surrender => "surrender", Self::Charlie => "charlie" }
182 }
183 fn profit(self, bet: Money, rules: Rules) -> Money {
184 match self {
185 Self::Blackjack => if rules.bj == BlackjackPay::ThreeTwo { bet * 3 / 2 } else { bet * 6 / 5 },
186 Self::Win | Self::Charlie => bet,
187 Self::Push => 0,
188 Self::Surrender => bet / 2 - bet,
189 Self::Lose | Self::Bust => -bet,
190 }
191 }
192}

194#[derive(Default)]
195struct Stats {
196 rounds: i32, hands: i32, wins: i32, losses: i32, pushes: i32,
197 blackjacks: i32, busts: i32, surrenders: i32, doubles: i32, splits: i32,
198}
199
200impl Stats {
201 fn record(&mut self, result: Outcome) {
202 self.hands += 1;
203 match result {
204 Outcome::Blackjack | Outcome::Win | Outcome::Charlie => self.wins += 1,
205 Outcome::Push => self.pushes += 1,
206 _ => self.losses += 1,
207 }
208 match result {
209 Outcome::Blackjack => self.blackjacks += 1,
210 Outcome::Bust => self.busts += 1,
211 Outcome::Surrender => self.surrenders += 1,
212 _ => {}
213 }
214 }
215}

237struct Shoe {
238 cards: Vec<Card>,
239 decks: usize,
240 drawn: usize,
241 running: i32,
242 random: i64,
243}
244
245impl Shoe {
246 fn new(decks: usize, seed: i64) -> Self {
247 let mut shoe = Self { cards: Vec::new(), decks, drawn: 0, running: 0, random: seed };
248 shoe.shuffle();
249 shoe
250 }
251 fn shuffle(&mut self) {
252 self.cards = (0..self.decks * 52).map(|i| Card { rank: (i % 13) as u8, suit: (i / 13 % 4) as u8 }).collect();
253 for i in (1..self.cards.len()).rev() {
254 self.random = 48271 * self.random % 2147483647;
255 self.cards.swap(i, (self.random % (i + 1) as i64) as usize);
256 }
257 self.drawn = 0;
258 self.running = 0;
259 }
260 fn cut(&self, pen: usize) -> usize { self.cards.len() * pen / 100 }
261 fn draw(&mut self, visible: bool) -> Card {
262 let card = self.cards[self.drawn];
263 self.drawn += 1;
264 if visible { self.running += card.hilo(); }
265 card
266 }
267}

309struct Table {
310 rules: Rules,
311 players: [Option<Player>; 7],
312 shoe: Option<Shoe>,
313 state: State,
314 round: i32,
315 insurance_seat: usize,
316 current_seat: usize,
317 current_hand: usize,
318 dealer: Hand,
319 hole_visible: bool,
320 dealer_blackjack: bool,

325impl Table {
326 fn player(&self, seat: usize) -> &Player { self.players[seat].as_ref().unwrap() }
327 fn player_mut(&mut self, seat: usize) -> &mut Player { self.players[seat].as_mut().unwrap() }
328 fn find_player(&self, name: &str) -> Option<usize> {
329 self.players.iter().position(|player| player.as_ref().is_some_and(|player| player.name == name))
330 }
331 fn participants(&self) -> Vec<usize> {
332 self.players.iter().enumerate().filter_map(|(seat, player)|
333 player.as_ref().filter(|player| player.participating).map(|_| seat)).collect()
334 }
335 fn current_name(&self) -> String { self.player(self.current_seat).hand_name(self.current_hand) }
336 fn hand_line(&self, seat: usize, index: usize, verb: &str) -> String {
337 let player = self.player(seat);
338 format!("{}{}: {}", player.hand_name(index), verb, player.hands[index])
339 }
340 fn add_player_card(&mut self, seat: usize, index: usize) {
341 let card = self.shoe.as_mut().unwrap().draw(true);
342 self.player_mut(seat).hands[index].cards.push(card);
343 }
344 fn add_dealer_card(&mut self, visible: bool) {
345 let card = self.shoe.as_mut().unwrap().draw(visible);
346 self.dealer.cards.push(card);
347 }
348 fn reveal(&mut self) {
349 if !self.hole_visible {
350 self.shoe.as_mut().unwrap().running += self.dealer.cards[1].hilo();
351 self.hole_visible = true;
352 }
353 }
354 fn automatic_finish(&mut self, seat: usize, index: usize) {
355 let rules = self.rules;
356 let player = self.player_mut(seat);
357 let hands = player.hands.len();
358 player.hands[index].automatic_finish(rules, hands);
359 }
360 fn settle(&mut self, output: &mut Vec<String>) {
361 let dealer = self.dealer.total().value;
362 let peek_blackjack = self.dealer_blackjack;
363 let rules = self.rules;
364 for seat in self.participants() {
365 let player = self.player_mut(seat);
366 for index in 0..player.hands.len() {
367 let hand = &player.hands[index];
368 let result = hand.outcome(dealer, peek_blackjack);
369 let bet = hand.bet;
370 let profit = result.profit(bet, rules);
371 player.bankroll += bet + profit;
372 player.net += profit;
373 player.stats.record(result);
374 output.push(format!("{} {} {profit:+}", player.hand_name(index), result.name()));
375 }
376 output.push(format!("{} bankroll {}", player.name, player.bankroll));
377 }
378 for player in self.players.iter_mut().flatten() {
379 player.bet = 0;
380 player.side = 0;
381 player.participating = false;
382 }
383 self.state = State::Idle;
384 }
385 fn dealer_turn(&mut self, output: &mut Vec<String>) {
386 self.reveal();
387 output.push(format!("dealer: {}", self.dealer));
388 let needs_draw = self.players.iter().flatten().filter(|player| player.participating)
389 .flat_map(|player| &player.hands).any(|hand| hand.finish == Some(Finish::Ordinary) && !hand.blackjack());
390 let mut drew = false;
391 if needs_draw {
392 loop {
393 let total = self.dealer.total();
394 if !(total.value <= 16 || (total.value == 17 && total.soft && self.rules.dealer == DealerRule::H17)) { break; }
395 self.add_dealer_card(true);
396 output.push(format!("dealer hits: {}", self.dealer));
397 drew = true;
398 }
399 }
400 if drew && self.dealer.total().value > 21 { self.dealer_busts += 1; }
401 self.settle(output);
402 }
403 fn next_hand(&mut self, output: &mut Vec<String>) {
404 loop {
405 let next = self.players.iter().enumerate().find_map(|(seat, player)| {
406 player.as_ref().filter(|player| player.participating).and_then(|player|
407 player.hands.iter().position(|hand| hand.finish.is_none()).map(|index| (seat, index)))
408 });
409 let Some((seat, index)) = next else { self.dealer_turn(output); return; };
410 self.current_seat = seat;
411 self.current_hand = index;
412 if self.player(seat).hands[index].cards.len() != 1 { return; }
413 self.add_player_card(seat, index);
414 output.push(self.hand_line(seat, index, ""));
415 self.automatic_finish(seat, index);
416 if self.player(seat).hands[index].finish.is_none() { return; }
417 }
418 }
419 fn peek(&mut self, output: &mut Vec<String>) {
420 self.dealer_blackjack = self.dealer.blackjack();
421 output.push(format!("dealer peeks: {}", if self.dealer_blackjack { "blackjack" } else { "no blackjack" }));
422 if self.dealer_blackjack { self.dealer_blackjacks += 1; }
423 let dealer_blackjack = self.dealer_blackjack;
424 for seat in self.participants() {
425 let player = self.player_mut(seat);
426 if player.insurance == 0 { continue; }
427 let profit = if dealer_blackjack { player.insurance * 2 } else { -player.insurance };
428 player.bankroll += player.insurance + profit;
429 player.net += profit;
430 output.push(format!("{} insurance {profit:+}", player.name));
431 }
432 if self.dealer_blackjack {
433 self.reveal();
434 output.push(format!("dealer: {}", self.dealer));
435 self.settle(output);
436 } else {
437 self.state = State::Play;
438 self.next_hand(output);
439 }
440 }
441 fn deal(&mut self, output: &mut Vec<String>) {
442 let shoe = self.shoe.as_mut().unwrap();
443 if shoe.drawn >= shoe.cut(self.rules.pen) {
444 shoe.shuffle();
445 output.push("shuffle".to_owned());
446 }
447 self.round += 1;
448 output.push(format!("round {}", self.round));
449 self.dealer = Hand::default();
450 self.hole_visible = false;
451 self.dealer_blackjack = false;
452 for player in self.players.iter_mut().flatten() {
453 player.participating = player.bet != 0;
454 player.hands.clear();
455 if player.participating {
456 player.bankroll -= player.bet + player.side;
457 player.insurance = 0;
458 player.stats.rounds += 1;
459 player.hands.push(Hand::new(player.bet));
460 }
461 }
462 let participants = self.participants();
463 for &seat in &participants { self.add_player_card(seat, 0); }
464 self.add_dealer_card(true);
465 for &seat in &participants {
466 self.add_player_card(seat, 0);
467 let hand = &mut self.player_mut(seat).hands[0];
468 if hand.blackjack() { hand.finish = Some(Finish::Ordinary); }
469 }
470 self.add_dealer_card(false);
471 for &seat in &participants { output.push(self.hand_line(seat, 0, "")); }
472 output.push(format!("dealer: {} ??", self.dealer.cards[0]));
473 for &seat in &participants {
474 let player = self.player_mut(seat);
475 if player.side == 0 { continue; }
476 let a = player.hands[0].cards[0];
477 let b = player.hands[0].cards[1];
478 let (result, multiplier) = if a.rank != b.rank { ("lose", -1) }
479 else if a.suit == b.suit { ("perfect", 25) }
480 else if a.black() == b.black() { ("colored", 12) } else { ("mixed", 6) };
481 let profit = player.side * multiplier;
482 player.bankroll += player.side + profit;
483 player.net += profit;
484 output.push(format!("{} side {result} {profit:+}", player.name));
485 }
486 match self.dealer.cards[0].point() {
487 1 => {
488 self.state = State::Insurance;
489 self.insurance_seat = participants[0];
490 output.push("insurance offered".to_owned());
491 }
492 10 => self.peek(output),
493 _ => { self.state = State::Play; self.next_hand(output); }
494 }
495 }
496 fn action_problem(&self, action: Action) -> Option<&'static str> {
497 let player = self.player(self.current_seat);
498 let hand = &player.hands[self.current_hand];
499 let total = hand.total();
500 match action {
501 Action::Hit if hand.split_aces => return Some("split aces"),
502 Action::Double => {
503 if hand.split_aces { return Some("split aces"); }
504 if hand.cards.len() != 2 { return Some("two cards only"); }
505 let allowed = match self.rules.doubling {
506 DoubleRule::Any => true,
507 DoubleRule::NineEleven => !total.soft && (9..=11).contains(&total.value),
508 DoubleRule::TenEleven => !total.soft && (10..=11).contains(&total.value),
509 };
510 if !allowed || (!self.rules.das && hand.split) { return Some("double not allowed"); }
511 if player.bankroll < hand.bet { return Some("insufficient funds"); }
512 }
513 Action::Split => {
514 if hand.cards.len() != 2 { return Some("two cards only"); }
515 if hand.cards[0].point() != hand.cards[1].point() { return Some("not a pair"); }
516 if player.hands.len() >= self.rules.hands { return Some("hand limit"); }
517 if player.bankroll < hand.bet { return Some("insufficient funds"); }
518 }
519 Action::Surrender => {
520 if !self.rules.surrender || hand.split { return Some("surrender not allowed"); }
521 if hand.cards.len() != 2 { return Some("two cards only"); }
522 }
523 _ => {}
524 }
525 None
526 }
527 fn recommendation(&self) -> Action {
528 let hand = &self.player(self.current_seat).hands[self.current_hand];
529 let mut up = self.dealer.cards[0].point();
530 if up == 1 { up = 11; }
531 let can_split = self.action_problem(Action::Split).is_none();
532 if hand.split_aces { return if can_split { Action::Split } else { Action::Stand }; }
533 if can_split {
534 let preferred = match hand.cards[0].point() {
535 1 | 8 => true, 9 => !matches!(up, 7 | 10 | 11),
536 2 | 3 | 7 => up <= 7, 6 => up <= 6, 4 => up == 5 || up == 6, _ => false,
537 };
538 if preferred { return Action::Split; }
539 }
540 let total = hand.total();
541 let value = total.value;
542 let can_double = self.action_problem(Action::Double).is_none();
543 if total.soft {
544 if value >= 19 { return Action::Stand; }
545 if value == 18 {
546 if (3..=6).contains(&up) { return if can_double { Action::Double } else { Action::Stand }; }
547 return if matches!(up, 2 | 7 | 8) { Action::Stand } else { Action::Hit };
548 }
549 let wants_double = match value {
550 17 => (3..=6).contains(&up), 15 | 16 => (4..=6).contains(&up),
551 13 | 14 => up == 5 || up == 6, _ => false,
552 };
553 return if wants_double && can_double { Action::Double } else { Action::Hit };
554 }
555 if value >= 17 { return Action::Stand; }
556 if value >= 13 {
557 if up <= 6 { return Action::Stand; }
558 if ((value == 16 && up >= 9) || (value == 15 && up == 10)) && self.action_problem(Action::Surrender).is_none() { return Action::Surrender; }
559 return Action::Hit;
560 }
561 if value == 12 { return if (4..=6).contains(&up) { Action::Stand } else { Action::Hit }; }
562 let wants_double = (value == 11 && up <= 10) || (value == 10 && up <= 9) || (value == 9 && (3..=6).contains(&up));
563 if wants_double && can_double { Action::Double } else { Action::Hit }
564 }
565 fn perform_action(&mut self, action: Action, output: &mut Vec<String>) {
566 let seat = self.current_seat;
567 let index = self.current_hand;
568 match action {
569 Action::Hit | Action::Double => {
570 if action == Action::Double {
571 let player = self.player_mut(seat);
572 player.bankroll -= player.hands[index].bet;
573 player.hands[index].bet *= 2;
574 player.stats.doubles += 1;
575 }
576 self.add_player_card(seat, index);
577 output.push(self.hand_line(seat, index, if action == Action::Hit { " hits" } else { " doubles" }));
578 self.automatic_finish(seat, index);
579 if action == Action::Double { self.player_mut(seat).hands[index].finish.get_or_insert(Finish::Ordinary); }
580 }
581 Action::Stand | Action::Surrender => {
582 self.player_mut(seat).hands[index].finish = Some(if action == Action::Stand { Finish::Ordinary } else { Finish::Surrender });
583 output.push(self.hand_line(seat, index, if action == Action::Stand { " stands" } else { " surrenders" }));
584 }
585 Action::Split => {
586 let player = self.player_mut(seat);
587 let old = &player.hands[index];
588 let (first, second, bet, aces) = (old.cards[0], old.cards[1], old.bet, old.split_aces || old.cards[0].rank == 0);
589 player.bankroll -= bet;
590 player.stats.splits += 1;
591 output.push(format!("{} splits {first} {second}", player.hand_name(index)));
592 player.hands[index] = Hand { cards: vec![first], bet, split: true, split_aces: aces, finish: None };
593 player.hands.insert(index + 1, Hand { cards: vec![second], bet, split: true, split_aces: aces, finish: None });
594 self.add_player_card(seat, index);
595 output.push(self.hand_line(seat, index, ""));
596 self.automatic_finish(seat, index);
597 }
598 }
599 self.next_hand(output);
600 }
601 fn status(&self, output: &mut Vec<String>) {
602 output.push(match self.state {
603 State::Idle => format!("table idle round {}", self.round),
604 State::Insurance => format!("table insurance round {} next {}", self.round, self.player(self.insurance_seat).name),
605 State::Play => format!("table play round {} turn {}", self.round, self.current_name()),
606 });
607 if self.state != State::Idle { output.push(format!("dealer: {} ??", self.dealer.cards[0])); }
608 for (seat, player) in self.players.iter().enumerate() {
609 let Some(player) = player else { continue; };
610 output.push(format!("seat {} {} bankroll {} bet {} side {}", seat + 1, player.name, player.bankroll, player.bet, player.side));
611 if self.state != State::Idle && player.participating {
612 for (index, hand) in player.hands.iter().enumerate() {
613 output.push(format!("{}: {hand} bet {}", player.hand_name(index), hand.bet));
614 }
615 }
616 }
617 }
618 fn stats(&self, output: &mut Vec<String>) {
619 for player in self.players.iter().flatten() {
620 let s = &player.stats;
621 output.push(format!("stats {} rounds {} hands {} wins {} losses {} pushes {} blackjacks {} busts {} surrenders {} doubles {} splits {} net {:+}",
622 player.name, s.rounds, s.hands, s.wins, s.losses, s.pushes, s.blackjacks, s.busts, s.surrenders, s.doubles, s.splits, player.net));
623 }
624 output.push(format!("stats dealer rounds {} blackjacks {} busts {}", self.round, self.dealer_blackjacks, self.dealer_busts));
625 }
626 fn execute(&mut self, words: &[&str]) -> CommandResult<Vec<String>> {
627 let command = Command::parse(words[0])?;
628 let mut seat = None;
629 let mut amount = 0;
630 let mut seed = 0;
631 let mut changed_rules = None;
632 match command {
633 Command::Shoe => { amount = integer(words[1], 1, 8)?; seed = integer(words[2], 1, 2147483646)?; }
634 Command::Rule => changed_rules = Some(self.rules.changed(words[1], words[2])?),
635 Command::Player => {
636 if words[1].is_empty() || words[1].len() > 8 || !words[1].bytes().all(|byte| byte.is_ascii_uppercase()) {
637 return Err(format!("bad name {}", words[1]));
638 }
639 amount = integer(words[2], 1, 1000000)?;
640 }
641 _ if command.named_player() => {
642 seat = Some(self.find_player(words[1]).ok_or_else(|| format!("unknown player {}", words[1]))?);
643 if matches!(command, Command::Rebuy | Command::Bet | Command::Side | Command::Insurance) {
644 amount = integer(words[2], if command == Command::Rebuy { 1 } else { 0 }, 1000000)?;
645 }
646 }
647 _ => {}
648 }
649 if command.idle_only() && self.state != State::Idle { return Err("round in progress".to_owned()); }
650 if command == Command::Insurance && self.state != State::Insurance { return Err("no insurance".to_owned()); }
651 if matches!(command, Command::Action(_) | Command::Auto | Command::Hint) {
652 match self.state {
653 State::Idle => return Err("no round".to_owned()),
654 State::Insurance => return Err("insurance pending".to_owned()),
655 State::Play => {}
656 }
657 if command != Command::Hint && seat != Some(self.current_seat) { return Err(format!("waiting for {}", self.current_name())); }
658 }
659 let mut output = Vec::new();
660 match command {
661 Command::Shoe => {
662 let shoe = Shoe::new(amount as usize, seed);
663 output.push(format!("shoe {} decks {} cards cut {}", shoe.decks, shoe.cards.len(), shoe.cut(self.rules.pen)));
664 self.shoe = Some(shoe);
665 }
666 Command::Rule => {
667 let rules = changed_rules.unwrap();
668 if rules.min > rules.max { return Err("min above max".to_owned()); }
669 self.rules = rules;
670 output.push(format!("rule {} {}", words[1], words[2]));
671 }
672 Command::Rules => output.push(format!("rules {}", self.rules)),
673 Command::Player => {
674 if self.find_player(words[1]).is_some() { return Err("player exists".to_owned()); }
675 let seat = self.players.iter().position(Option::is_none).ok_or("table full")?;
676 self.players[seat] = Some(Player::new(words[1], amount));
677 output.push(format!("{} sits at seat {} with {amount}", words[1], seat + 1));
678 }
679 Command::Leave => {
680 let player = self.players[seat.unwrap()].take().unwrap();
681 output.push(format!("{} leaves with {} net {:+}", player.name, player.bankroll, player.net));
682 }
683 Command::Rebuy => {
684 let player = self.player_mut(seat.unwrap());
685 player.bankroll += amount;
686 output.push(format!("{} rebuys {amount} bankroll {}", player.name, player.bankroll));
687 }
688 Command::Bet => {
689 let rules = self.rules;
690 let player = self.player_mut(seat.unwrap());
691 if amount == 0 {
692 player.bet = 0;
693 player.side = 0;
694 output.push(format!("{} bet cleared", player.name));
695 } else {
696 if amount < rules.min || amount > rules.max { return Err("bet out of limits".to_owned()); }
697 if amount + player.side > player.bankroll { return Err("insufficient funds".to_owned()); }
698 player.bet = amount;
699 output.push(format!("{} bets {amount}", player.name));
700 }
701 }
702 Command::Side => {
703 let side_max = self.rules.side_max;
704 let player = self.player_mut(seat.unwrap());
705 if amount == 0 {
706 player.side = 0;
707 output.push(format!("{} side cleared", player.name));
708 } else {
709 if player.bet == 0 { return Err("no bet".to_owned()); }
710 if amount > side_max { return Err("side out of limits".to_owned()); }
711 if player.bet + amount > player.bankroll { return Err("insufficient funds".to_owned()); }
712 player.side = amount;
713 output.push(format!("{} side {amount}", player.name));
714 }
715 }
716 Command::Deal => {
717 if self.shoe.is_none() { return Err("no shoe".to_owned()); }
718 if !self.players.iter().flatten().any(|player| player.bet != 0) { return Err("no bets".to_owned()); }
719 self.deal(&mut output);
720 }
721 Command::Insurance => {
722 let seat = seat.unwrap();
723 if seat != self.insurance_seat { return Err(format!("waiting for {}", self.player(self.insurance_seat).name)); }
724 let player = self.player_mut(seat);
725 if amount > player.bet / 2 { return Err("insurance out of limits".to_owned()); }
726 if amount > player.bankroll { return Err("insufficient funds".to_owned()); }
727 player.bankroll -= amount;
728 player.insurance = amount;
729 output.push(if amount == 0 { format!("{} declines insurance", player.name) } else { format!("{} insures {amount}", player.name) });
730 if let Some(next) = self.participants().into_iter().find(|&next| next > seat) { self.insurance_seat = next; }
731 else { self.peek(&mut output); }
732 }
733 Command::Action(action) => {
734 if let Some(problem) = self.action_problem(action) { return Err(problem.to_owned()); }
735 self.perform_action(action, &mut output);
736 }
737 Command::Hint => output.push(format!("hint {}: {}", self.current_name(), self.recommendation().name())),
738 Command::Auto => {
739 while self.state == State::Play && self.current_seat == seat.unwrap() {
740 self.perform_action(self.recommendation(), &mut output);
741 }
742 }
743 Command::Status => self.status(&mut output),
744 Command::Count => {
745 let shoe = self.shoe.as_ref().ok_or("no shoe")?;
746 let remaining = shoe.cards.len() - shoe.drawn;
747 let scaled = i64::from(shoe.running.abs()) * 5200 / remaining as i64;
748 output.push(format!("count running {} true {}{}.{:02} remaining {remaining}", shoe.running,
749 if shoe.running < 0 { "-" } else { "" }, scaled / 100, scaled % 100));
750 }
751 Command::Stats => {
752 if self.players.iter().all(Option::is_none) { return Err("no players".to_owned()); }
753 self.stats(&mut output);
754 }
755 }
756 Ok(output)
757 }
758}

760fn main() {
761 let mut input = String::new();
762 io::stdin().read_to_string(&mut input).unwrap();
763 let mut lines = input.lines();
764 let count: usize = lines.next().unwrap().parse().unwrap();
765 let mut table = Table::default();
766 let stdout = io::stdout();
767 let mut output = io::BufWriter::new(stdout.lock());
768 for line in lines.take(count) {
769 let words: Vec<&str> = line.split_ascii_whitespace().collect();
770 match table.execute(&words) {
771 Ok(lines) => for line in lines { writeln!(output, "{line}").unwrap(); },
772 Err(error) => writeln!(output, "error: {error}").unwrap(),
773 }
774 }
775}
