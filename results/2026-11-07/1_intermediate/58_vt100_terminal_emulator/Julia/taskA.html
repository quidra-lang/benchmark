69function reset!(t::Terminal)
70 t.screens = [[blankrow(t) for _ in 1:t.height] for _ in 1:2]
71 t.row,t.column,t.top,t.bottom,t.display,t.shape = 1,1,1,t.height,1,1
72 t.pending = false
73 t.attr = Attributes()
74 t.g0,t.g1,t.gl = false,false,false
75 t.modes = Dict(name=>initial(name) for name in MODE_NAMES)
76 t.saved = Union{Nothing,SavedCursor}[nothing,nothing]
77 t.tabs = Set(9:8:t.width)
78 empty!(t.scrollback)
79 t.title,t.lastchar = nothing,nothing
80 t.state = GROUND
81 clearsequence!(t)
82end
83rowtext(row::Vector{Cell}) = replace(join(cell.char for cell in row),r" +$"=>"")
84function scrollup!(t::Terminal,top::Int,bottom::Int; history::Bool=false)
85 rows = t.screens[t.display]
86 removed = rows[top]
87 for r in top:bottom-1
88 rows[r] = rows[r+1]
89 end
90 rows[bottom] = blankrow(t)
91 if history && t.display == 1 && top == 1
92 push!(t.scrollback,rowtext(removed))
93 length(t.scrollback) > 100 && popfirst!(t.scrollback)
94 end
95end
96function scrolldown!(t::Terminal,top::Int,bottom::Int)
97 rows = t.screens[t.display]
98 for r in bottom:-1:top+1
99 rows[r] = rows[r-1]
100 end
101 rows[top] = blankrow(t)
102end
103function linefeed!(t::Terminal)
104 if t.row == t.bottom
105 scrollup!(t,t.top,t.bottom; history=true)
106 elseif t.row < t.height
107 t.row += 1
108 end
109end
110function reversefeed!(t::Terminal)
111 if t.row == t.top
112 scrolldown!(t,t.top,t.bottom)
113 elseif t.row > 1
114 t.row -= 1
115 end
116end
117function tab!(t::Terminal,reverse::Bool=false)
118 if reverse
119 earlier = [stop for stop in t.tabs if stop < t.column]
120 t.column = isempty(earlier) ? 1 : maximum(earlier)
121 else
122 later = [stop for stop in t.tabs if stop > t.column]
123 t.column = isempty(later) ? t.width : minimum(later)
124 end
125end

139end

150 row = t.screens[t.display][t.row]
151 for c in t.column:t.width-n
152 row[c] = row[c+n]
153 end
154 for c in t.width-n+1:t.width

157end
158function writechar!(t::Terminal,c::Char)
159 t.printed += 1
160 t.lastchar = c
161 if t.pending
162 t.pending = false
163 t.column = 1
164 linefeed!(t)
165 end
166 t.modes["IRM"] && insertcells!(t,1)
167 t.screens[t.display][t.row][t.column] = Cell(c,t.attr)
168 if t.column < t.width
169 t.column += 1
170 elseif t.modes["DECAWM"]
171 t.pending = true
172 end
173end
174function printchar!(t::Terminal,c::Char)
175 graphics = t.gl ? t.g1 : t.g0
176 if graphics
177 replacement = Dict('`'=>'*','a'=>'#','j'=>'+','k'=>'+','l'=>'+','m'=>'+','n'=>'+',
178 't'=>'+','u'=>'+','v'=>'+','w'=>'+','q'=>'-','x'=>'|','~'=>'o')
179 c = get(replacement,c,c)
180 end
181 writechar!(t,c)
182end
183function control!(t::Terminal,b::Int)

185 t.bells += 1
186 elseif b == 8
187 t.pending = false
188 t.column = max(t.column-1,1)
189 elseif b == 9
190 t.pending = false
191 tab!(t)
192 elseif b in (10,11,12)
193 t.pending = false
194 linefeed!(t)
195 t.modes["LNM"] && (t.column = 1)
196 elseif b == 13
197 t.pending = false
198 t.column = 1
199 elseif b == 14
200 t.gl = true
201 elseif b == 15
202 t.gl = false
203 end
204end
205function executeesc!(t::Terminal,key::String)

207 savecursor!(t)
208 elseif key == "8"
209 restorecursor!(t)
210 elseif key == "D"
211 t.pending = false
212 linefeed!(t)
213 elseif key == "E"

215 t.column = 1
216 linefeed!(t)
217 elseif key == "M"
218 t.pending = false
219 reversefeed!(t)
220 elseif key == "H"
221 push!(t.tabs,t.column)
222 elseif key == "c"
223 reset!(t)
224 elseif key in ("=",">")

226 elseif key == "\\"

228 elseif key in ("(0","(B")

231 t.g1 = key == ")0"
232 elseif key == "#8"

235 t.top,t.bottom,t.row,t.column = 1,t.height,1,1

237 else
238 t.ignored[1] += 1

240 end

242end

251function palette(number::Int)
252 names = ["black","red","green","yellow","blue","magenta","cyan","white"]
253 number < 8 && return names[number+1]
254 number < 16 && return "bright-"*names[number-7]
255 return string(number)
256end
257function sgr!(t::Terminal,params::Vector{Int})
258 values = isempty(params) ? [0] : params
259 flags,fg,bg = t.attr.flags,t.attr.fg,t.attr.bg
260 on = Dict(1=>0,2=>1,3=>2,4=>3,5=>4,7=>5,8=>6,9=>7)
261 off = Dict(23=>2,24=>3,25=>4,27=>5,28=>6,29=>7)
262 index = 1
263 while index <= length(values)
264 v = values[index]
265 if v == 0
266 flags,fg,bg = 0,"",""
267 elseif haskey(on,v)
268 flags |= 1 << on[v]
269 elseif v == 22
270 flags &= ~3
271 elseif haskey(off,v)
272 flags &= ~(1 << off[v])
273 elseif 30 <= v <= 37
274 fg = palette(v-30)
275 elseif 90 <= v <= 97
276 fg = palette(v-82)
277 elseif 40 <= v <= 47
278 bg = palette(v-40)
279 elseif 100 <= v <= 107
280 bg = palette(v-92)
281 elseif v == 39
282 fg = ""
283 elseif v == 49
284 bg = ""
285 elseif v in (38,48)
286 color = ""
287 if index+2 <= length(values) && values[index+1] == 5 && values[index+2] <= 255
288 color = palette(values[index+2])
289 index += 2
290 elseif index+4 <= length(values) && values[index+1] == 2 && maximum(values[index+2:index+4]) <= 255
291 color = "#"*join(lpad(string(value; base=16),2,'0') for value in values[index+2:index+4])
292 index += 4
293 else
294 return false
295 end
296 if v == 38
297 fg = color
298 else
299 bg = color
300 end
301 else
302 return false
303 end
304 index += 1
305 end
306 t.attr = Attributes(flags,fg,bg)
307 return true
308end
309function setmodes!(t::Terminal,params::Vector{Int},dec::Bool,enabled::Bool)
310 values = isempty(params) ? [0] : params
311 allowed = dec ? (1,5,6,7,25,47,1047,1048,1049,2004) : (4,20)
312 all(v -> v in allowed,values) || return false
313 for v in values
314 if !dec
315 t.modes[ANSI_MODES[v]] = enabled
316 elseif haskey(DEC_MODES,v)
317 (v == 6 || v == 7 && !enabled) && (t.pending = false)
318 t.modes[DEC_MODES[v]] = enabled
319 v == 6 && home!(t)
320 elseif v == 1048
321 enabled ? savecursor!(t) : restorecursor!(t)
322 else
323 t.pending = false
324 if enabled && t.display == 1
325 v == 1049 && savecursor!(t)
326 t.display = 2
327 v == 1049 && clearscreen!(t,2)
328 elseif !enabled && t.display == 2
329 v == 1047 && clearscreen!(t,2)
330 t.display = 1
331 v == 1049 && restorecursor!(t)
332 end
333 end
334 end
335 return true
336end

355function csi!(t::Terminal,key::String,params::Vector{Int})
356 p1 = isempty(params) ? 0 : params[1]
357 p2 = length(params) < 2 ? 0 : params[2]
358 n = max(p1,1)
359 conditions = Dict("J"=>(0,1,2,3),"K"=>(0,1,2),"g"=>(0,3)," q"=>(0,1,2,3,4,5,6),
360 "n"=>(5,6),"?n"=>(6,),"c"=>(0,),">c"=>(0,),"t"=>(18,))
361 haskey(conditions,key) && !(p1 in conditions[key]) && return 1
362 length(key) == 1 && first(key) in "ABCDEFGHIJKLMPSTXZ`adef@" && (t.pending = false)
363 if key == "A"
364 moveup!(t,n)
365 elseif key in ("B","e")
366 movedown!(t,n)
367 elseif key in ("C","a")
368 t.column = min(t.column+n,t.width)
369 elseif key == "D"
370 t.column = max(t.column-n,1)
371 elseif key == "E"
372 movedown!(t,n)
373 t.column = 1
374 elseif key == "F"
375 moveup!(t,n)
376 t.column = 1
377 elseif key in ("G","`")
378 t.column = min(n,t.width)
379 elseif key in ("H","f")
380 t.row,t.column = address(t,n),min(max(p2,1),t.width)
381 elseif key == "d"
382 t.row = address(t,n)
383 elseif key in ("I","Z")
384 for _ in 1:n
385 tab!(t,key == "Z")
386 end
387 elseif key == "J"
388 if p1 == 0
389 erase!(t,t.row,t.column,t.width)
390 for r in t.row+1:t.height
391 erase!(t,r,1,t.width)
392 end
393 elseif p1 == 1
394 for r in 1:t.row-1
395 erase!(t,r,1,t.width)
396 end
397 erase!(t,t.row,1,t.column)
398 elseif p1 == 2
399 clearscreen!(t,t.display)
400 else
401 empty!(t.scrollback)
402 end
403 elseif key == "K"
404 firstcol = p1 in (1,2) ? 1 : t.column
405 lastcol = p1 == 1 ? t.column : t.width
406 erase!(t,t.row,firstcol,lastcol)
407 elseif key == "X"
408 erase!(t,t.row,t.column,min(t.column+n-1,t.width))
409 elseif key == "@"
410 insertcells!(t,min(n,t.width-t.column+1))
411 elseif key == "P"
412 deletecells!(t,min(n,t.width-t.column+1))
413 elseif key in ("L","M")
414 if t.top <= t.row <= t.bottom
415 for _ in 1:min(n,t.bottom-t.row+1)
416 key == "L" ? scrolldown!(t,t.row,t.bottom) : scrollup!(t,t.row,t.bottom)
417 end
418 t.column = 1
419 end
420 elseif key in ("S","T")
421 for _ in 1:min(n,t.bottom-t.top+1)
422 key == "S" ? scrollup!(t,t.top,t.bottom; history=true) : scrolldown!(t,t.top,t.bottom)
423 end
424 elseif key == "b"
425 if t.lastchar !== nothing
426 c = t.lastchar
427 for _ in 1:n
428 writechar!(t,c)
429 end
430 end
431 elseif key == "g"
432 p1 == 0 ? delete!(t.tabs,t.column) : empty!(t.tabs)
433 elseif key == "m"
434 sgr!(t,params) || return 1
435 elseif key in ("h","l","?h","?l")
436 setmodes!(t,params,startswith(key,"?"),endswith(key,"h")) || return 1
437 elseif key == " q"
438 t.shape = n
439 elseif key == "r"
440 bottom = min(p2 == 0 ? t.height : p2,t.height)
441 n < bottom || return 4
442 t.pending = false
443 t.top,t.bottom = n,bottom
444 home!(t)
445 elseif key == "s"
446 savecursor!(t)
447 elseif key == "u"
448 restorecursor!(t)
449 elseif key == "!p"
450 t.pending = false
451 for name in ("IRM","DECCKM","DECOM","DECAWM","DECTCEM","DECKPAM")
452 t.modes[name] = initial(name)
453 end
454 t.top,t.bottom = 1,t.height
455 t.attr = Attributes()
456 t.g0,t.g1,t.gl = false,false,false
457 t.saved[t.display] = nothing
458 elseif key in ("n","?n")
459 r = t.modes["DECOM"] ? max(t.row-t.top+1,1) : t.row
460 text = key == "?n" ? "^[[?$r;$(t.column);1R" : p1 == 5 ? "^[[0n" : "^[[$r;$(t.column)R"
461 reply!(t,text)
462 elseif key == "c"
463 reply!(t,"^[[?1;2c")
464 elseif key == ">c"
465 reply!(t,"^[[>0;95;0c")
466 elseif key == "t"
467 reply!(t,"^[[8;$(t.height);$(t.width)t")
468 elseif key in ("\$p","?\$p")
469 state,prefix = 0,""
470 if key == "\$p"
471 haskey(ANSI_MODES,p1) && (state = modestate(t.modes[ANSI_MODES[p1]]))
472 else
473 prefix = "?"
474 if haskey(DEC_MODES,p1)
475 state = modestate(t.modes[DEC_MODES[p1]])
476 elseif p1 in (47,1047,1049)
477 state = modestate(t.display == 2)
478 end
479 end
480 reply!(t,"^[[$prefix$p1;$state\$y")
481 else
482 return 1
483 end
484 return 0
485end
486function executecsi!(t::Terminal,finalchar::Char)
487 fields = isempty(t.parameters) ? SubString{String}[] : split(t.parameters,';'; keepempty=true)
488 if length(fields) > 16
489 t.ignored[4] += 1
490 return
491 end
492 params = Int[]
493 for field in fields
494 value = 0
495 for digit in field
496 value = min(9999,value*10+Int(digit-'0'))
497 end
498 push!(params,value)
499 end
500 reason = csi!(t,t.marker*t.intermediates*string(finalchar),params)
501 if reason == 0
502 t.sequences += 1
503 else
504 t.ignored[reason] += 1
505 end
506end
507isintermediate(b::Int) = 0x20 <= b <= 0x2f
508isfinal(b::Int) = 0x40 <= b <= 0x7e
509function receive!(t::Terminal,b::Int)
510 t.bytes += 1
511 if b < 0x20
512 if b == 27
513 if t.state == OSC_STRING
514 executeosc!(t)
515 elseif t.state != GROUND
516 t.ignored[3] += 1
517 end
518 clearsequence!(t)
519 t.state = ESCAPE
520 elseif b in (24,26)
521 if t.state != GROUND
522 t.ignored[3] += 1
523 t.state = GROUND
524 end
525 elseif t.state == OSC_STRING
526 if b == 7
527 executeosc!(t)
528 t.state = GROUND
529 end
530 else
531 control!(t,b)
532 end
533 return
534 end
535 c = Char(b)
536 if t.state == GROUND
537 printchar!(t,c)
538 elseif t.state in (ESCAPE,ESC_INTERMEDIATE)
539 if isintermediate(b)
540 t.intermediates *= string(c)
541 t.state = ESC_INTERMEDIATE
542 elseif t.state == ESCAPE && c == '['
543 t.state = CSI_ENTRY
544 elseif t.state == ESCAPE && c == ']'
545 t.state = OSC_STRING
546 else
547 executeesc!(t,t.intermediates*string(c))
548 t.state = GROUND
549 end
550 elseif t.state == OSC_STRING
551 t.osc *= string(c)
552 elseif t.state == CSI_IGNORE
553 if isfinal(b)
554 t.ignored[2] += 1
555 t.state = GROUND
556 end
557 elseif t.state == CSI_INTERMEDIATE
558 if isintermediate(b)
559 t.intermediates *= string(c)
560 elseif isfinal(b)
561 executecsi!(t,c)
562 t.state = GROUND
563 else
564 t.state = CSI_IGNORE
565 end
566 elseif t.state == CSI_ENTRY && 0x3c <= b <= 0x3f
567 t.marker = string(c)
568 t.state = CSI_PARAM
569 elseif '0' <= c <= '9' || c == ';'
570 t.parameters *= string(c)
571 t.state = CSI_PARAM
572 elseif isintermediate(b)
573 t.intermediates *= string(c)
574 t.state = CSI_INTERMEDIATE
575 elseif isfinal(b)
576 executecsi!(t,c)
577 t.state = GROUND
578 else
579 t.state = CSI_IGNORE
580 end
581end
582function keystring(t::Terminal,name::String)
583 cursor = Dict("UP"=>"A","DOWN"=>"B","RIGHT"=>"C","LEFT"=>"D","HOME"=>"H","END"=>"F")
584 haskey(cursor,name) && return (t.modes["DECCKM"] ? "^[O" : "^[[")*cursor[name]
585 fixed = Dict("INSERT"=>"^[[2~","DELETE"=>"^[[3~","PAGEUP"=>"^[[5~","PAGEDOWN"=>"^[[6~",
586 "F1"=>"^[OP","F2"=>"^[OQ","F3"=>"^[OR","F4"=>"^[OS","F5"=>"^[[15~","F6"=>"^[[17~",
587 "F7"=>"^[[18~","F8"=>"^[[19~","F9"=>"^[[20~","F10"=>"^[[21~","F11"=>"^[[23~","F12"=>"^[[24~",
588 "TAB"=>"^I","BACKSPACE"=>"^?","ESCAPE"=>"^[")
589 haskey(fixed,name) && return fixed[name]
590 enter = t.modes["LNM"] ? "^M^J" : "^M"
591 name == "ENTER" && return enter
592 if name == "KPENTER"
593 return t.modes["DECKPAM"] ? "^[OM" : enter
594 end
595 digit = parse(Int,name[3:end])
596 return t.modes["DECKPAM"] ? "^[O"*string(Char(0x70+digit)) : string(digit)
597end

599 name = t.display == 1 ? "main" : "alt"
600 println("$label $name cursor $(t.row),$(t.column) scrollback $(length(t.scrollback))",t.pending ? " pending" : "")
601 for row in t.screens[t.display]

603 end
604end
605function describe(attr::Attributes)
606 parts = [name for (bit,name) in enumerate(FLAG_NAMES) if attr.flags & (1 << (bit-1)) != 0]
607 !isempty(attr.fg) && push!(parts,"fg="*attr.fg)
608 !isempty(attr.bg) && push!(parts,"bg="*attr.bg)
609 return join(parts," ")
610end
611function finish!(t::Terminal)
612 if t.state != GROUND
613 t.ignored[3] += 1
614 t.state = GROUND
615 end
616 screenoutput(t,"FINAL")
617 println("TITLE ",t.title === nothing ? "none" : "\""*t.title*"\"")
618 modes = [name*"="*(t.modes[name] ? "on" : "off") for name in MODE_NAMES if t.modes[name] != initial(name)]
619 t.g0 && push!(modes,"G0=graphics")
620 t.g1 && push!(modes,"G1=graphics")
621 t.gl && push!(modes,"GL=G1")
622 t.shape != 1 && push!(modes,"CURSOR=$(t.shape)")
623 println("MODES ",isempty(modes) ? "default" : join(modes," "))
624 println("REGION $(t.top)-$(t.bottom)")
625 println("TABS ",isempty(t.tabs) ? "none" : join(sort(collect(t.tabs)),","))
626 println("SCROLLBACK ",length(t.scrollback))
627 for text in t.scrollback[max(1,length(t.scrollback)-4):end]
628 println("|",text)
629 end
630 runs = String[]
631 for (r,row) in enumerate(t.screens[t.display])
632 start = 1
633 while start <= t.width

650function main()
651 lines = readlines()
652 width,height = parse.(Int,split(lines[1]))
653 terminal = Terminal(width=width,height=height)
654 reset!(terminal)
655 caret = Dict('@'=>0,'G'=>7,'H'=>8,'I'=>9,'J'=>10,'K'=>11,'L'=>12,'M'=>13,'N'=>14,'O'=>15,'X'=>24,'Z'=>26,'['=>27,'^'=>94)
656 for line in lines[2:end]
657 line == "%%END" && break
658 if line == "%%DUMP"
659 terminal.dumps += 1
660 screenoutput(terminal,"DUMP $(terminal.dumps)")
661 elseif startswith(line,"%%KEY ")
662 name = line[7:end]
663 println("KEY $name $(keystring(terminal,name))")
664 elseif startswith(line,"%%PASTE ")
665 sent = replace(line[9:end],"^"=>"^^")
666 terminal.modes["BRACKETED"] && (sent = "^[[200~"*sent*"^[[201~")
667 println("PASTE ",sent)
668 else
669 index = 1
670 while index <= length(line)
671 if line[index] == '^'
672 receive!(terminal,caret[line[index+1]])
673 index += 2
674 else
675 receive!(terminal,Int(line[index]))
676 index += 1
677 end
678 end
679 end
680 end
681 finish!(terminal)
682end
