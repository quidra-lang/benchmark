344const Color = union(enum) {
345 palette: u8,
346 rgb: [3]u8,
347
348 pub fn format(self: Color, writer: *std.Io.Writer) std.Io.Writer.Error!void {
349 switch (self) {
350 .palette => |number| {
351 if (number < 8) {
352 try writer.writeAll(color_names[number]);
353 } else if (number < 16) {
354 try writer.print("bright-{s}", .{color_names[number - 8]});
355 } else {
356 try writer.print("{d}", .{number});
357 }
358 },
359 .rgb => |components| try writer.print("#{x:0>2}{x:0>2}{x:0>2}", .{
360 components[0],
361 components[1],
362 components[2],
363 }),
364 }
365 }
366};
367
368fn extendedColor(rest: []const usize) Ignored!struct { Color, usize } {
369 if (rest.len >= 2 and rest[0] == 5 and rest[1] <= 255) {
370 return .{ Color{ .palette = @intCast(rest[1]) }, 2 };
371 }
372 if (rest.len >= 4 and rest[0] == 2 and std.mem.max(usize, rest[1..4]) <= 255) {
373 return .{ Color{ .rgb = .{ @intCast(rest[1]), @intCast(rest[2]), @intCast(rest[3]) } }, 4 };
374 }
375 return error.Unsupported;
376}

468 }

472 }
473
474 fn erase(self: *Screen, row: usize, first: usize, last: usize) void {
475 @memset(self.rows[row - 1][first - 1 .. last], Cell{});
476 }
477
478 fn eraseRows(self: *Screen, first: usize, last: usize) void {
479 @memset(self.rows[first - 1 .. last], blank_row);

481
482 fn scrollUp(self: *Screen, top: usize, bottom: usize) Row {
483 const removed = self.rows[top - 1];
484 std.mem.copyForwards(Row, self.rows[top - 1 .. bottom - 1], self.rows[top..bottom]);
485 self.rows[bottom - 1] = blank_row;
486 return removed;
487 }
488
489 fn scrollDown(self: *Screen, top: usize, bottom: usize) void {
490 std.mem.copyBackwards(Row, self.rows[top..bottom], self.rows[top - 1 .. bottom - 1]);
491 self.rows[top - 1] = blank_row;
492 }
493
494 fn insertCells(self: *Screen, row: usize, column: usize, count: usize) void {
495 const cells = self.rows[row - 1][0..self.width];
496 const start = column - 1;
497 std.mem.copyBackwards(Cell, cells[start + count ..], cells[start .. self.width - count]);
498 @memset(cells[start .. start + count], Cell{});
499 }
500
501 fn deleteCells(self: *Screen, row: usize, column: usize, count: usize) void {
502 const cells = self.rows[row - 1][0..self.width];
503 const start = column - 1;
504 std.mem.copyForwards(Cell, cells[start .. self.width - count], cells[start + count ..]);
505 @memset(cells[self.width - count ..], Cell{});
506 }

597 fn receive(self: *Terminal, byte: u8) Error!void {
598 self.stream_bytes += 1;
599 if (byte < 0x20) return self.receiveControl(byte);
600 switch (self.state) {
601 .ground => self.printChar(byte),
602 .escape, .escape_intermediate => try self.escapeByte(byte),
603 .osc_string => try self.osc.append(self.gpa, byte),
604 .csi_ignore => if (isFinal(byte)) {
605 self.ignored.malformed += 1;
606 self.state = .ground;
607 },
608 .csi_intermediate => try self.csiIntermediateByte(byte),
609 .csi_entry => if (isPrivateMarker(byte)) {
610 self.marker = byte;
611 self.state = .csi_param;
612 } else {
613 try self.csiParamByte(byte);
614 },
615 .csi_param => try self.csiParamByte(byte),
616 }
617 }
618
619 fn receiveControl(self: *Terminal, byte: u8) Error!void {
620 if (byte == esc) {
621 if (self.state == .osc_string) {
622 try self.executeOsc();
623 } else if (self.state != .ground) {
624 self.ignored.canceled += 1;
625 }
626 self.clearSequence();
627 self.state = .escape;
628 } else if (byte == can or byte == sub) {
629 if (self.state != .ground) {
630 self.ignored.canceled += 1;
631 self.state = .ground;
632 }
633 } else if (self.state == .osc_string) {
634 if (byte == bel) {
635 try self.executeOsc();
636 self.state = .ground;
637 }
638 } else {
639 self.executeControl(byte);
640 }
641 }
642
643 fn escapeByte(self: *Terminal, byte: u8) Error!void {
644 if (isIntermediate(byte)) {
645 try self.intermediates.append(self.gpa, byte);
646 self.state = .escape_intermediate;
647 } else if (self.state == .escape and byte == '[') {
648 self.state = .csi_entry;
649 } else if (self.state == .escape and byte == ']') {
650 self.state = .osc_string;
651 } else {
652 const key = try self.sequenceKey(byte);
653 defer self.gpa.free(key);
654 self.executeEsc(key);
655 self.state = .ground;
656 }
657 }
658
659 fn csiParamByte(self: *Terminal, byte: u8) Error!void {
660 if (isParameter(byte)) {
661 try self.parameters.append(self.gpa, byte);
662 self.state = .csi_param;
663 } else if (isIntermediate(byte)) {
664 try self.intermediates.append(self.gpa, byte);
665 self.state = .csi_intermediate;
666 } else if (isFinal(byte)) {
667 try self.executeCsi(byte);
668 self.state = .ground;
669 } else {
670 self.state = .csi_ignore;
671 }
672 }
673
674 fn csiIntermediateByte(self: *Terminal, byte: u8) Error!void {
675 if (isIntermediate(byte)) {
676 try self.intermediates.append(self.gpa, byte);
677 } else if (isFinal(byte)) {
678 try self.executeCsi(byte);
679 self.state = .ground;
680 } else {
681 self.state = .csi_ignore;
682 }
683 }
684
685 fn sequenceKey(self: *Terminal, final: u8) std.mem.Allocator.Error![]u8 {
686 var key: std.ArrayList(u8) = .empty;
687 errdefer key.deinit(self.gpa);
688 if (self.marker) |marker| try key.append(self.gpa, marker);
689 try key.appendSlice(self.gpa, self.intermediates.items);
690 try key.append(self.gpa, final);
691 return key.toOwnedSlice(self.gpa);
692 }
693
694 fn executeControl(self: *Terminal, byte: u8) void {
695 switch (byte) {
696 bel => self.bells += 1,
697 bs => {
698 self.pending_wrap = false;
699 self.column = @max(self.column - 1, 1);
700 },
701 ht => {
702 self.pending_wrap = false;
703 self.tabMove(1);
704 },
705 lf, vt, ff => {
706 self.pending_wrap = false;
707 self.lineFeed();
708 if (self.modes.get(.lnm)) self.column = 1;
709 },
710 cr => {
711 self.pending_wrap = false;
712 self.column = 1;
713 },
714 so => self.gl = .g1,
715 si => self.gl = .g0,
716 else => {},
717 }
718 }
719
720 fn printChar(self: *Terminal, char: u8) void {
721 const charset = if (self.gl == .g1) self.g1 else self.g0;
722 self.write(if (charset == .line_drawing) lineDrawing(char) else char);
723 }
724
725 fn write(self: *Terminal, char: u8) void {
726 self.printed += 1;
727 self.last_char = char;
728 if (self.pending_wrap) {
729 self.pending_wrap = false;
730 self.column = 1;
731 self.lineFeed();
732 }
733 const screen = self.displayedScreen();
734 if (self.modes.get(.irm)) screen.insertCells(self.row, self.column, 1);
735 screen.put(self.row, self.column, .{ .char = char, .attributes = self.attributes });

905 fn runCsi(self: *Terminal, key: []const u8) (Ignored || Error)!void {
906 var values: [max_fields]usize = undefined;
907 var field_count: usize = 0;
908 if (self.parameters.items.len > 0) {
909 var fields = std.mem.splitScalar(u8, self.parameters.items, ';');
910 while (fields.next()) |field| : (field_count += 1) {
911 if (field_count == max_fields) return error.Invalid;
912 values[field_count] = fieldValue(field);
913 }
914 }
915 const params = values[0..field_count];
916 const command = csi_commands.get(key) orelse return error.Unsupported;
917 if (!command.accepts(parameter(params, 0))) return error.Unsupported;
918 if (command.clearsPendingWrap()) self.pending_wrap = false;
919 switch (command) {
920 .cursor_up => self.moveUp(amount(params)),
921 .cursor_down => self.moveDown(amount(params)),
922 .cursor_forward => self.column = @min(self.column + amount(params), self.width),
923 .cursor_backward => self.column = @max(self.column -| amount(params), 1),
924 .cursor_next_line => {
925 self.moveDown(amount(params));
926 self.column = 1;
927 },
928 .cursor_previous_line => {
929 self.moveUp(amount(params));
930 self.column = 1;
931 },
932 .cursor_column => self.column = @min(amount(params), self.width),
933 .cursor_position => {
934 self.row = self.addressRow(amount(params));
935 self.column = @min(@max(parameter(params, 1), 1), self.width);
936 },
937 .cursor_row => self.row = self.addressRow(amount(params)),
938 .forward_tabulation => self.tabMove(amount(params)),
939 .backward_tabulation => self.reverseTabMove(amount(params)),
940 .erase_in_display => self.eraseInDisplay(parameter(params, 0)),
941 .erase_in_line => self.eraseInLine(parameter(params, 0)),
942 .erase_characters => {
943 const last = @min(self.column + amount(params) - 1, self.width);
944 self.displayedScreen().erase(self.row, self.column, last);
945 },
946 .insert_characters => {
947 const count = @min(amount(params), self.width - self.column + 1);
948 self.displayedScreen().insertCells(self.row, self.column, count);
949 },
950 .delete_characters => {
951 const count = @min(amount(params), self.width - self.column + 1);
952 self.displayedScreen().deleteCells(self.row, self.column, count);
953 },
954 .insert_lines => self.insertLines(amount(params)),
955 .delete_lines => self.deleteLines(amount(params)),
956 .scroll_up => {
957 const count = @min(amount(params), self.bottom - self.top + 1);
958 for (0..count) |_| self.scrollRegionUp();
959 },
960 .scroll_down => {
961 const count = @min(amount(params), self.bottom - self.top + 1);
962 for (0..count) |_| self.scrollRegionDown();
963 },
964 .repeat_character => if (self.last_char) |char| {
965 for (0..amount(params)) |_| self.write(char);
966 },
967 .clear_tab_stops => if (parameter(params, 0) == 0) {
968 self.tab_stops[self.column] = false;
969 } else {
970 @memset(&self.tab_stops, false);
971 },
972 .select_graphic_rendition => try self.selectGraphicRendition(params),
973 .set_ansi_modes => try self.setAnsiModes(params, true),
974 .reset_ansi_modes => try self.setAnsiModes(params, false),
975 .set_dec_modes => try self.setDecModes(params, true),
976 .reset_dec_modes => try self.setDecModes(params, false),
977 .set_cursor_shape => self.cursor_shape = amount(params),
978 .set_scroll_region => try self.setScrollRegion(params),
979 .save_cursor => self.saveCursor(),
980 .restore_cursor => self.restoreCursor(),
981 .soft_reset => self.softReset(),
982 .device_status_report => if (parameter(params, 0) == 5) {
983 try self.reply("^[[0n", .{});
984 } else {
985 try self.reply("^[[{d};{d}R", .{ self.reportedRow(), self.column });
986 },
987 .extended_cursor_report => {
988 try self.reply("^[[?{d};{d};1R", .{ self.reportedRow(), self.column });
989 },
990 .device_attributes => try self.reply("^[[?1;2c", .{}),
991 .secondary_device_attributes => try self.reply("^[[>0;95;0c", .{}),
992 .window_report => try self.reply("^[[8;{d};{d}t", .{ self.height, self.width }),
993 .request_ansi_mode => try self.requestAnsiMode(parameter(params, 0)),
994 .request_dec_mode => try self.requestDecMode(parameter(params, 0)),
995 }
996 }

1281 fn finish(self: *Terminal) Error!void {
1282 if (self.state != .ground) {
1283 self.ignored.canceled += 1;
1284 self.state = .ground;
1285 }
1286 const out = self.out;
1287 try out.writeAll("FINAL");
1288 try self.writeScreen();
1289 if (self.title) |title| {
1290 try out.print("TITLE \"{s}\"\n", .{title});
1291 } else {
1292 try out.writeAll("TITLE none\n");
1293 }
1294 var buffer: [256]u8 = undefined;
1295 var changed: std.Io.Writer = .fixed(&buffer);
1296 try self.writeChangedModes(&changed);
1297 const modes: []const u8 = if (changed.end == 0) " default" else changed.buffered();
1298 try out.print("MODES{s}\n", .{modes});
1299 try out.print("REGION {d}-{d}\nTABS", .{ self.top, self.bottom });
1300 var separator: u8 = ' ';
1301 for (1..self.width + 1) |column| {
1302 if (self.tab_stops[column]) {
1303 try out.print("{c}{d}", .{ separator, column });
1304 separator = ',';
1305 }
1306 }
1307 if (separator == ' ') try out.writeAll(" none");
1308 try out.print("\nSCROLLBACK {d}\n", .{self.scrollback_len});
1309 const first_shown = self.scrollback_len -| scrollback_shown;
1310 for (self.scrollback[first_shown..self.scrollback_len]) |*line| {
1311 try out.print("|{s}\n", .{line.text()});
1312 }
1313 var runs = try self.attributeRuns();
1314 defer runs.deinit(self.gpa);
1315 try out.print("ATTRS {d}\n", .{runs.items.len});
1316 for (runs.items) |run| {
1317 try out.print("row {d} col {d}-{d}{f}\n", .{
1318 run.row,
1319 run.first,
1320 run.last,
1321 run.attributes,
1322 });
1323 }

1341pub fn main(init: std.process.Init) !void {
1342 var input_buffer: [4096]u8 = undefined;
1343 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
1344 var output_buffer: [8192]u8 = undefined;
1345 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1346 const reader = &input.interface;
1347 const first = (try reader.takeDelimiter('\n')).?;
1348 var sizes = std.mem.splitScalar(u8, first, ' ');
1349 const width = try std.fmt.parseInt(usize, sizes.next().?, 10);
1350 const height = try std.fmt.parseInt(usize, sizes.next().?, 10);
1351 var terminal = Terminal.init(init.gpa, &output.interface, width, height);
1352 defer terminal.deinit();
1353 while (try reader.takeDelimiter('\n')) |line| {
1354 if (std.mem.eql(u8, line, "%%END")) break;
1355 if (std.mem.eql(u8, line, "%%DUMP")) {
1356 try terminal.dump();
1357 } else if (std.mem.startsWith(u8, line, "%%KEY ")) {
1358 try terminal.pressKey(line["%%KEY ".len..]);
1359 } else if (std.mem.startsWith(u8, line, "%%PASTE ")) {
1360 try terminal.paste(line["%%PASTE ".len..]);
1361 } else {
1362 var index: usize = 0;
1363 while (index < line.len) : (index += 1) {
1364 if (line[index] == '^') {
1365 index += 1;
1366 try terminal.receive(caretByte(line[index]));
1367 } else {
1368 try terminal.receive(line[index]);
1369 }
1370 }
1371 }
1372 }
1373 try terminal.finish();
1374 try output.interface.flush();
1375}
