122fun decodeStream(line: String): String = buildString {
123 var index = 0
124 while (index < line.length) {
125 if (line[index] == '^') {
126 append(CARET_BYTES.getValue(line[index + 1]))
127 index += 2
128 } else {
129 append(line[index])
130 index++
131 }
132 }
133}

151 val rest = values.drop(start)

339 fun receive(ch: Char) {
340 streamBytes++
341 if (ch < ' ') {
342 receiveControl(ch)
343 return
344 }
345 when (state) {
346 ParserState.GROUND -> printChar(ch)
347 ParserState.ESCAPE, ParserState.ESCAPE_INTERMEDIATE -> escapeByte(ch)
348 ParserState.OSC_STRING -> osc += ch
349 ParserState.CSI_IGNORE -> {
350 if (isFinal(ch)) {
351 ignore(Reason.MALFORMED)
352 state = ParserState.GROUND
353 }
354 }
355 ParserState.CSI_INTERMEDIATE -> csiIntermediateByte(ch)
356 ParserState.CSI_ENTRY -> {
357 if (isPrivateMarker(ch)) {
358 marker = ch.toString()
359 state = ParserState.CSI_PARAM
360 } else {
361 csiParamByte(ch)
362 }
363 }
364 ParserState.CSI_PARAM -> csiParamByte(ch)
365 }
366 }

368 private fun receiveControl(ch: Char) {
369 when {
370 ch == ESC -> {
371 if (state == ParserState.OSC_STRING) {
372 executeOsc()
373 } else if (state != ParserState.GROUND) {
374 ignore(Reason.CANCELED)
375 }
376 clearSequence()
377 state = ParserState.ESCAPE
378 }
379 ch == CAN || ch == SUB -> {
380 if (state != ParserState.GROUND) {
381 ignore(Reason.CANCELED)
382 state = ParserState.GROUND
383 }
384 }
385 state == ParserState.OSC_STRING -> {
386 if (ch == BEL) {
387 executeOsc()
388 state = ParserState.GROUND
389 }
390 }
391 else -> executeControl(ch)
392 }
393 }
394
395 private fun escapeByte(ch: Char) {
396 when {
397 isIntermediate(ch) -> {
398 intermediates += ch
399 state = ParserState.ESCAPE_INTERMEDIATE
400 }
401 state == ParserState.ESCAPE && ch == '[' -> state = ParserState.CSI_ENTRY
402 state == ParserState.ESCAPE && ch == ']' -> state = ParserState.OSC_STRING
403 else -> {
404 executeEsc(intermediates + ch)
405 state = ParserState.GROUND
406 }
407 }
408 }

410 private fun csiParamByte(ch: Char) {
411 when {
412 isParameter(ch) -> {
413 parameters += ch
414 state = ParserState.CSI_PARAM
415 }
416 isIntermediate(ch) -> {
417 intermediates += ch
418 state = ParserState.CSI_INTERMEDIATE
419 }
420 isFinal(ch) -> {
421 executeCsi(ch)
422 state = ParserState.GROUND
423 }
424 else -> state = ParserState.CSI_IGNORE
425 }
426 }

428 private fun csiIntermediateByte(ch: Char) {
429 when {
430 isIntermediate(ch) -> intermediates += ch
431 isFinal(ch) -> {
432 executeCsi(ch)
433 state = ParserState.GROUND
434 }
435 else -> state = ParserState.CSI_IGNORE
436 }

570 when (key) {
571 "7" -> saveCursor()
572 "8" -> restoreCursor()
573 "D" -> {
574 pendingWrap = false
575 lineFeed()
576 }
577 "E" -> {
578 pendingWrap = false
579 column = 1

581 }
582 "M" -> {

610 private fun executeCsi(finalByte: Char) {
611 val key = marker + intermediates + finalByte
612 val fields = if (parameters.isEmpty()) emptyList() else parameters.split(";")
613 try {
614 ensure(fields.size <= MAX_FIELDS, Reason.INVALID)
615 val params = fields.map(::fieldValue)
616 val allowed = CSI_CONDITIONS[key]
617 ensure(allowed == null || parameter(params, 0) in allowed)
618 if (key.length == 1 && key in WRAP_CLEARING_CSI) {
619 pendingWrap = false
620 }
621 dispatchCsi(key, params)
622 } catch (ignoredSequence: SequenceIgnored) {
623 ignore(ignoredSequence.reason)
624 return
625 }
626 sequences++
627 }

629 private fun dispatchCsi(key: String, params: List<Int>) {
630 when (key) {
631 "A" -> moveUp(amount(params))
632 "B", "e" -> moveDown(amount(params))
633 "C", "a" -> column = minOf(column + amount(params), width)
634 "D" -> column = maxOf(column - amount(params), 1)
635 "E" -> {
636 moveDown(amount(params))
637 column = 1
638 }
639 "F" -> {
640 moveUp(amount(params))
641 column = 1
642 }
643 "G", "`" -> column = minOf(amount(params), width)
644 "H", "f" -> {
645 row = addressRow(amount(params))
646 column = parameter(params, 1).coerceIn(1, width)
647 }
648 "d" -> row = addressRow(amount(params))
649 "I" -> tabMove(amount(params))
650 "Z" -> reverseTabMove(amount(params))
651 "J" -> eraseInDisplay(parameter(params, 0))
652 "K" -> eraseInLine(parameter(params, 0))
653 "X" -> screen.erase(row, column, minOf(column + amount(params) - 1, width))
654 "@" -> screen.insertCells(row, column, minOf(amount(params), width - column + 1))
655 "P" -> screen.deleteCells(row, column, minOf(amount(params), width - column + 1))
656 "L" -> insertLines(amount(params))
657 "M" -> deleteLines(amount(params))
658 "S" -> repeat(minOf(amount(params), bottom - top + 1)) { scrollRegionUp() }
659 "T" -> repeat(minOf(amount(params), bottom - top + 1)) { scrollRegionDown() }
660 "b" -> lastChar?.let { char -> repeat(amount(params)) { write(char) } }
661 "g" -> if (parameter(params, 0) == 0) tabStops.remove(column) else tabStops.clear()
662 "m" -> selectGraphicRendition(params)
663 "h", "l" -> setAnsiModes(params, key == "h")
664 "?h", "?l" -> setDecModes(params, key == "?h")
665 " q" -> cursorShape = amount(params)
666 "r" -> setScrollRegion(params)
667 "s" -> saveCursor()
668 "u" -> restoreCursor()
669 "!p" -> softReset()
670 "n" -> deviceStatusReport(parameter(params, 0))
671 "?n" -> reply("^[[?${reportedRow()};$column;1R")
672 "c" -> reply("^[[?1;2c")
673 ">c" -> reply("^[[>0;95;0c")
674 "t" -> reply("^[[8;$height;${width}t")
675 "\$p" -> requestAnsiMode(parameter(params, 0))
676 "?\$p" -> requestDecMode(parameter(params, 0))
677 else -> throw SequenceIgnored(Reason.UNSUPPORTED)
678 }
679 }

726 private fun selectGraphicRendition(params: List<Int>) {
727 val values = params.ifEmpty { listOf(0) }
728 var (flags, foreground, background) = attributes
729 var index = 0
730 while (index < values.size) {
731 val value = values[index]
732 when (value) {
733 0 -> {
734 flags = 0
735 foreground = null
736 background = null
737 }
738 in SGR_FLAGS_ON -> flags = flags or (1 shl SGR_FLAGS_ON.getValue(value))
739 in SGR_FLAGS_OFF -> {
740 for (flag in SGR_FLAGS_OFF.getValue(value)) {
741 flags = flags and (1 shl flag).inv()
742 }
743 }
744 in 30..37 -> foreground = paletteColor(value - 30)
745 in 90..97 -> foreground = paletteColor(value - 82)
746 in 40..47 -> background = paletteColor(value - 40)
747 in 100..107 -> background = paletteColor(value - 92)
748 39 -> foreground = null
749 49 -> background = null
750 38, 48 -> {
751 val (color, used) = extendedColor(values, index + 1)
752 if (value == 38) foreground = color else background = color
753 index += used
754 }
755 else -> throw SequenceIgnored(Reason.UNSUPPORTED)
756 }
757 index++
758 }
759 attributes = Attributes(flags, foreground, background)
760 }

856 private fun keyString(name: String): String {
857 CURSOR_KEYS[name]?.let { suffix ->
858 return (if (Mode.DECCKM in modes) "^[O" else "^[[") + suffix
859 }
860 FIXED_KEYS[name]?.let { return it }
861 val enter = if (Mode.LNM in modes) "^M^J" else "^M"
862 if (name == "ENTER") {
863 return enter
864 }
865 if (name == "KPENTER") {
866 return if (Mode.DECKPAM in modes) "^[OM" else enter
867 }
868 val digit = name.substring(2).toInt()
869 return if (Mode.DECKPAM in modes) "^[O${Char(0x70 + digit)}" else digit.toString()
870 }

908 private fun attributeRuns(): List<String> {
909 val runs = mutableListOf<String>()
910 for ((index, cells) in screen.rows.withIndex()) {
911 var start = 0
912 for (next in 1..width) {
913 if (next < width && cells[next].attributes == cells[start].attributes) {
914 continue
915 }
916 val run = cells[start].attributes
917 if (!run.isDefault()) {
918 runs.add("row ${index + 1} col ${start + 1}-$next ${run.describe()}")
919 }
920 start = next
921 }
922 }
923 return runs
924 }

926 fun finish() {
927 if (state != ParserState.GROUND) {
928 ignore(Reason.CANCELED)
929 state = ParserState.GROUND
930 }
931 screenOutput("FINAL")
932 emit("TITLE " + (title?.let { "\"$it\"" } ?: "none"))
933 emit("MODES " + changedModes().ifEmpty { listOf("default") }.joinToString(" "))
934 emit("REGION $top-$bottom")
935 emit("TABS " + if (tabStops.isEmpty()) "none" else tabStops.joinToString(","))
936 emit("SCROLLBACK ${scrollback.size}")
937 scrollback.takeLast(SCROLLBACK_SHOWN).forEach { emit("|$it") }
938 val runs = attributeRuns()
939 emit("ATTRS ${runs.size}")
940 runs.forEach(this::emit)
941 emit("BELL $bells")
942 emit("REPLIES $replies")
943 emit("STATS bytes=$streamBytes printed=$printed sequences=$sequences")
944 val counts = Reason.entries.joinToString(" ") {
945 "${it.name.lowercase()}=${ignored[it.ordinal]}"
946 }
947 emit("IGNORED ${ignored.sum()} $counts")
948 }

951fun main() {
952 val (width, height) = readln().split(" ").map(String::toInt)
953 val terminal = Terminal(width, height)
954 for (line in generateSequence(::readlnOrNull).takeWhile { it != "%%END" }) {
955 when {
956 line == "%%DUMP" -> terminal.dump()
957 line.startsWith(KEY_PREFIX) -> terminal.pressKey(line.removePrefix(KEY_PREFIX))
958 line.startsWith(PASTE_PREFIX) -> terminal.paste(line.removePrefix(PASTE_PREFIX))
959 else -> decodeStream(line).forEach(terminal::receive)
960 }
961 }
962 terminal.finish()
963 print(terminal.output)
964}
