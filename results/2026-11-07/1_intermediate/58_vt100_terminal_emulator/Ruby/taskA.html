1class Terminal
2 FLAGS = %w[bold dim italic underline blink reverse hidden strike]
3 INITIAL = {'IRM'=>false, 'LNM'=>false, 'DECCKM'=>false, 'DECSCNM'=>false,
4 'DECOM'=>false, 'DECAWM'=>true, 'DECTCEM'=>true, 'DECKPAM'=>false, 'BRACKETED'=>false}
5 ANSI = {4=>'IRM', 20=>'LNM'}
6 DEC = {1=>'DECCKM', 5=>'DECSCNM', 6=>'DECOM', 7=>'DECAWM', 25=>'DECTCEM', 2004=>'BRACKETED'}
7 DEFAULT = [0, 'default', 'default'].freeze

9 def initialize(width, height)
10 @w, @h = width, height
11 @bytes = @printed = @sequences = @bell = @replies = @dumps = 0

13 reset

54 def scroll(up, count = 1)
55 count.times do
56 if up
57 removed = grid.delete_at(@top - 1)
58 grid.insert(@bottom - 1, blank_row)
59 if @screen == 0 && @top == 1
60 @scrollback << row_text(removed)
61 @scrollback.shift if @scrollback.size > 100
62 end
63 else
64 grid.delete_at(@bottom - 1)
65 grid.insert(@top - 1, blank_row)
66 end
67 end
68 end

121 @printed += 1
122 @last = char
123 if @pending
124 @pending = false
125 @col = 1

141 def print_char(char)
142 if (@gl == 0 ? @g0 : @g1) == 1
143 char = {'`'=>'*', 'a'=>'#', 'j'=>'+', 'k'=>'+', 'l'=>'+', 'm'=>'+', 'n'=>'+',
144 't'=>'+', 'u'=>'+', 'v'=>'+', 'w'=>'+', 'q'=>'-', 'x'=>'|', '~'=>'o'}.fetch(char, char)
145 end
146 write(char)
147 end

149 def control(code)
150 @pending = false if [8, 9, 10, 11, 12, 13].include?(code)
151 case code
152 when 7 then @bell += 1
153 when 8 then @col = [@col - 1, 1].max
154 when 9 then tab
155 when 10, 11, 12
156 line_feed
157 @col = 1 if @modes['LNM']
158 when 13 then @col = 1
159 when 14 then @gl = 1
160 when 15 then @gl = 0
161 end
162 end

164 def escape(key)
165 known = ['7', '8', 'D', 'E', 'M', 'H', 'c', '=', '>', '\\', '(0', '(B', ')0', ')B', '#8']
166 return 'unsupported' unless known.include?(key)
167 @pending = false if ['D', 'E', 'M', 'c', '#8'].include?(key)
168 case key
169 when '7' then save
170 when '8' then restore
171 when 'D' then line_feed
172 when 'E' then @col = 1; line_feed
173 when 'M' then reverse_feed
174 when 'H' then @tabs = (@tabs + [@col]).uniq.sort
175 when 'c' then reset
176 when '=' then @modes['DECKPAM'] = true
177 when '>' then @modes['DECKPAM'] = false
178 when '(0' then @g0 = 1
179 when '(B' then @g0 = 0
180 when ')0' then @g1 = 1
181 when ')B' then @g1 = 0
182 when '#8'
183 @screens[@screen] = Array.new(@h) { Array.new(@w) { ['E', DEFAULT] } }
184 @top, @bottom = 1, @h
185 @modes['DECOM'] = false
186 @row = @col = 1
187 end
188 nil
189 end

191 def palette(number)
192 colors = %w[black red green yellow blue magenta cyan white]
193 return colors[number] if number < 8
194 return 'bright-' + colors[number - 8] if number < 16
195 number.to_s
196 end

198 def sgr(params)
199 params = [0] if params.empty?
200 attrs = @attrs.dup
201 on = {1=>0, 2=>1, 3=>2, 4=>3, 5=>4, 7=>5, 8=>6, 9=>7}
202 off = {23=>2, 24=>3, 25=>4, 27=>5, 28=>6, 29=>7}
203 i = 0
204 while i < params.size
205 value = params[i]
206 if value == 0
207 attrs = DEFAULT.dup
208 elsif on.key?(value)
209 attrs[0] |= 1 << on[value]
210 elsif value == 22
211 attrs[0] &= ~3
212 elsif off.key?(value)
213 attrs[0] &= ~(1 << off[value])
214 elsif (30..37).cover?(value)
215 attrs[1] = palette(value - 30)
216 elsif (90..97).cover?(value)
217 attrs[1] = palette(value - 82)
218 elsif (40..47).cover?(value)
219 attrs[2] = palette(value - 40)
220 elsif (100..107).cover?(value)
221 attrs[2] = palette(value - 92)
222 elsif value == 39 || value == 49
223 attrs[value == 39 ? 1 : 2] = 'default'
224 elsif value == 38 || value == 48
225 color = nil
226 if params[i + 1] == 5 && i + 2 < params.size && params[i + 2] <= 255
227 color = palette(params[i + 2])
228 i += 2
229 elsif params[i + 1] == 2 && i + 4 < params.size && params[i + 2, 3].all? { |v| v <= 255 }
230 color = format('#%02x%02x%02x', *params[i + 2, 3])
231 i += 4
232 end
233 return 'unsupported' unless color
234 attrs[value == 38 ? 1 : 2] = color
235 else
236 return 'unsupported'
237 end
238 i += 1
239 end
240 @attrs = attrs
241 nil
242 end

244 def modes(key, params)
245 params = [0] if params.empty?
246 private_mode = key.start_with?('?')
247 allowed = private_mode ? [1, 5, 6, 7, 25, 47, 1047, 1048, 1049, 2004] : [4, 20]
248 return 'unsupported' unless params.all? { |p| allowed.include?(p) }
249 setting = key.end_with?('h')
250 params.each do |p|
251 if !private_mode
252 @modes[ANSI[p]] = setting
253 elsif p == 6
254 @pending = false
255 @modes['DECOM'] = setting
256 home
257 elsif [47, 1047, 1049].include?(p)
258 @pending = false
259 target = setting ? 1 : 0
260 next if @screen == target
261 save if setting && p == 1049
262 @screens[1] = blank_screen if !setting && p == 1047
263 @screen = target
264 @screens[1] = blank_screen if setting && p == 1049
265 restore if !setting && p == 1049
266 elsif p == 1048
267 setting ? save : restore
268 else
269 @pending = false if p == 7 && !setting
270 @modes[DEC[p]] = setting
271 end
272 end
273 nil
274 end

285 def csi(final)
286 fields = @params.empty? ? [] : @params.split(';', -1)
287 return 'invalid' if fields.size > 16
288 params = fields.map do |field|
289 value = 0
290 field.each_byte { |byte| value = [9999, value * 10 + byte - 48].min }
291 value
292 end
293 p1, p2 = params[0] || 0, params[1] || 0
294 n = [p1, 1].max
295 key = @private + @intermediate + final
296 known = ['A','B','e','C','a','D','E','F','G','`','H','f','d','I','Z','J','K','X','@','P','L','M','S','T','b','g','m','h','l','?h','?l',' q','r','s','u','!p','n','?n','c','>c','t','$p','?$p']
297 return 'unsupported' unless known.include?(key)
298 bad = (key == 'J' && p1 > 3) || (key == 'K' && p1 > 2) ||
299 (key == 'g' && ![0, 3].include?(p1)) || (key == ' q' && p1 > 6) ||
300 (key == 'n' && ![5, 6].include?(p1)) || (key == '?n' && p1 != 6) ||
301 (['c', '>c'].include?(key) && p1 != 0) || (key == 't' && p1 != 18)
302 return 'unsupported' if bad
303 return sgr(params) if key == 'm'
304 return modes(key, params) if ['h','l','?h','?l'].include?(key)
305 if key == 'r'
306 top = p1 == 0 ? 1 : p1
307 bottom = [p2 == 0 ? @h : p2, @h].min
308 return 'invalid' if top >= bottom
309 @pending = false
310 @top, @bottom = top, bottom
311 home
312 return nil
313 end
314 @pending = false if ['A','B','C','D','E','F','G','H','I','J','K','L','M','P','S','T','X','Z','`','a','d','e','f','@','!p'].include?(key)
315 case key
316 when 'A','F'
317 @row = [@row - n, @row >= @top ? @top : 1].max
318 @col = 1 if key == 'F'
319 when 'B','e','E'
320 @row = [@row + n, @row <= @bottom ? @bottom : @h].min
321 @col = 1 if key == 'E'
322 when 'C','a' then @col = [@col + n, @w].min
323 when 'D' then @col = [@col - n, 1].max
324 when 'G','`' then @col = [n, @w].min
325 when 'H','f' then @row, @col = address(n), [[p2, 1].max, @w].min
326 when 'd' then @row = address(n)
327 when 'I','Z' then n.times { tab(key == 'Z') }
328 when 'J'
329 if p1 == 3
330 @scrollback.clear
331 else
332 @h.times do |r|
333 (1..@w).each do |c|
334 position = r * @w + c
335 cursor = (@row - 1) * @w + @col
336 erase(r + 1, c, c) if p1 == 2 || (p1 == 0 && position >= cursor) || (p1 == 1 && position <= cursor)
337 end
338 end
339 end
340 when 'K' then erase(@row, p1 == 0 ? @col : 1, p1 == 1 ? @col : @w)
341 when 'X' then erase(@row, @col, [@col + n - 1, @w].min)
342 when '@','P'
343 cells = grid[@row - 1]
344 k = [n, @w - @col + 1].min
345 if key == '@'
346 cells.insert(@col - 1, *Array.new(k) { [' ', DEFAULT] })
347 cells.slice!(@w..-1)
348 else
349 cells.slice!(@col - 1, k)
350 k.times { cells << [' ', DEFAULT] }
351 end
352 when 'L','M'
353 if @row.between?(@top, @bottom)
354 k = [n, @bottom - @row + 1].min
355 k.times do
356 if key == 'L'
357 grid.delete_at(@bottom - 1)
358 grid.insert(@row - 1, blank_row)
359 else
360 grid.delete_at(@row - 1)
361 grid.insert(@bottom - 1, blank_row)
362 end
363 end
364 @col = 1
365 end
366 when 'S','T' then scroll(key == 'S', [n, @bottom - @top + 1].min)
367 when 'b' then n.times { write(@last) } unless @last.nil?
368 when 'g' then p1 == 3 ? @tabs.clear : @tabs.delete(@col)
369 when ' q' then @shape = n
370 when 's' then save
371 when 'u' then restore
372 when '!p'
373 %w[IRM DECCKM DECOM DECAWM DECTCEM DECKPAM].each { |mode| @modes[mode] = INITIAL[mode] }
374 @top, @bottom = 1, @h
375 @attrs = DEFAULT.dup
376 @g0 = @g1 = @gl = 0
377 @saves[@screen] = nil
378 when 'n'
379 r = @modes['DECOM'] ? [@row - @top + 1, 1].max : @row
380 reply(p1 == 5 ? '^[[0n' : "^[[#{r};#{@col}R")
381 when '?n'
382 r = @modes['DECOM'] ? [@row - @top + 1, 1].max : @row
383 reply("^[[?#{r};#{@col};1R")
384 when 'c' then reply('^[[?1;2c')
385 when '>c' then reply('^[[>0;95;0c')
386 when 't' then reply("^[[8;#{@h};#{@w}t")
387 when '$p','?$p'
388 mode = key == '$p' ? ANSI[p1] : DEC[p1]
389 status = mode ? (@modes[mode] ? 1 : 2) : 0
390 status = @screen == 1 ? 1 : 2 if key == '?$p' && [47, 1047, 1049].include?(p1)
391 reply("^[[#{key == '?$p' ? '?' : ''}#{p1};#{status}$y")
392 end
393 nil
394 end

413 def byte(code)
414 @bytes += 1
415 if code == 27
416 if @parser == 'osc'
417 execute_osc
418 elsif @parser != 'ground'
419 @ignored['canceled'] += 1
420 end
421 parser_start('escape')
422 return
423 end
424 if code == 24 || code == 26
425 if @parser != 'ground'
426 @ignored['canceled'] += 1
427 @parser = 'ground'
428 end
429 return
430 end
431 if code < 32
432 if @parser == 'osc'
433 if code == 7
434 execute_osc
435 @parser = 'ground'
436 end
437 else
438 control(code)
439 end
440 return
441 end
442 char = code.chr
443 intermediate = code.between?(32, 47)
444 final = code >= 64
445 case @parser
446 when 'ground' then print_char(char)
447 when 'escape'
448 if intermediate
449 @intermediate += char
450 @parser = 'escape intermediate'
451 elsif char == '['
452 @parser = 'entry'
453 elsif char == ']'
454 @parser = 'osc'
455 else
456 finish_sequence(escape(@intermediate + char))
457 @parser = 'ground'
458 end
459 when 'escape intermediate'
460 if intermediate
461 @intermediate += char
462 else
463 finish_sequence(escape(@intermediate + char))
464 @parser = 'ground'
465 end
466 when 'entry','param'
467 if @parser == 'entry' && code.between?(60, 63)
468 @private = char
469 @parser = 'param'
470 elsif code.between?(48, 57) || char == ';'
471 @params += char
472 @parser = 'param'
473 elsif intermediate
474 @intermediate += char
475 @parser = 'csi intermediate'
476 elsif final
477 finish_sequence(csi(char))
478 @parser = 'ground'
479 else
480 @parser = 'ignore'
481 end
482 when 'csi intermediate'
483 if intermediate
484 @intermediate += char
485 elsif final
486 finish_sequence(csi(char))
487 @parser = 'ground'
488 else
489 @parser = 'ignore'
490 end
491 when 'ignore'
492 if final
493 @ignored['malformed'] += 1
494 @parser = 'ground'
495 end
496 when 'osc' then @osc += char
497 end
498 end

502 grid.each { |cells| puts '|' + row_text(cells) }

505 def key(name)
506 arrows = {'UP'=>'A','DOWN'=>'B','RIGHT'=>'C','LEFT'=>'D','HOME'=>'H','END'=>'F'}
507 fixed = {'INSERT'=>'^[[2~','DELETE'=>'^[[3~','PAGEUP'=>'^[[5~','PAGEDOWN'=>'^[[6~',
508 'F1'=>'^[OP','F2'=>'^[OQ','F3'=>'^[OR','F4'=>'^[OS','F5'=>'^[[15~','F6'=>'^[[17~',
509 'F7'=>'^[[18~','F8'=>'^[[19~','F9'=>'^[[20~','F10'=>'^[[21~','F11'=>'^[[23~','F12'=>'^[[24~',
510 'TAB'=>'^I','BACKSPACE'=>'^?','ESCAPE'=>'^['}
511 enter = @modes['LNM'] ? '^M^J' : '^M'
512 text = if arrows.key?(name)
513 (@modes['DECCKM'] ? '^[O' : '^[[') + arrows[name]
514 elsif name == 'ENTER'
515 enter
516 elsif name == 'KPENTER'
517 @modes['DECKPAM'] ? '^[OM' : enter
518 elsif name.match?(/\AKP\d\z/)
519 @modes['DECKPAM'] ? '^[O' + (112 + name[2].to_i).chr : name[2]
520 else fixed[name]
521 end
522 puts "KEY #{name} #{text}"
523 end

525 def finish
526 @ignored['canceled'] += 1 if @parser != 'ground'
527 dump('FINAL')
528 puts @title.nil? ? 'TITLE none' : "TITLE \"#{@title}\""
529 modes = INITIAL.filter_map { |name, initial| "#{name}=#{@modes[name] ? 'on' : 'off'}" if @modes[name] != initial }
530 modes << 'G0=graphics' if @g0 == 1
531 modes << 'G1=graphics' if @g1 == 1
532 modes << 'GL=G1' if @gl == 1
533 modes << "CURSOR=#{@shape}" if @shape != 1
534 puts 'MODES ' + (modes.empty? ? 'default' : modes.join(' '))
535 puts "REGION #{@top}-#{@bottom}"
536 puts 'TABS ' + (@tabs.empty? ? 'none' : @tabs.sort.join(','))
537 puts "SCROLLBACK #{@scrollback.size}"
538 @scrollback.last(5).each { |line| puts '|' + line }
539 runs = []
540 grid.each_with_index do |cells, r|
541 c = 0
542 while c < @w
543 attrs = cells[c][1]
544 last = c
545 last += 1 while last + 1 < @w && cells[last + 1][1] == attrs
546 if attrs != DEFAULT
547 names = FLAGS.each_with_index.filter_map { |name, i| name if attrs[0] & (1 << i) != 0 }
548 names << "fg=#{attrs[1]}" if attrs[1] != 'default'
549 names << "bg=#{attrs[2]}" if attrs[2] != 'default'
550 runs << "row #{r + 1} col #{c + 1}-#{last + 1} #{names.join(' ')}"
551 end
552 c = last + 1
553 end
554 end
555 puts "ATTRS #{runs.size}"
556 puts runs
557 puts "BELL #{@bell}", "REPLIES #{@replies}", "STATS bytes=#{@bytes} printed=#{@printed} sequences=#{@sequences}"
558 puts "IGNORED #{@ignored.values.sum} " + @ignored.map { |name, value| "#{name}=#{value}" }.join(' ')
559 end

561 def line(text)
562 if text == '%%END'
563 finish
564 elsif text == '%%DUMP'
565 @dumps += 1
566 dump("DUMP #{@dumps}")
567 elsif text.start_with?('%%KEY ')
568 key(text[6..])
569 elsif text.start_with?('%%PASTE ')
570 pasted = text[8..].gsub('^', '^^')
571 pasted = '^[[200~' + pasted + '^[[201~' if @modes['BRACKETED']
572 puts 'PASTE ' + pasted
573 else
574 codes = {'@'=>0,'G'=>7,'H'=>8,'I'=>9,'J'=>10,'K'=>11,'L'=>12,'M'=>13,'N'=>14,'O'=>15,'X'=>24,'Z'=>26,'['=>27,'^'=>94}
575 i = 0
576 while i < text.size
577 if text[i] == '^'
578 byte(codes[text[i + 1]])
579 i += 2
580 else
581 byte(text.getbyte(i))
582 i += 1
583 end
584 end
585 end
586 end
587end
