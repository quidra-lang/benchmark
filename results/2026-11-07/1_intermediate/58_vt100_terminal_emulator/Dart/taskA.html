247List<int> decodeStream(String line) {
248 final data = <int>[];
249 var index = 0;
250 while (index < line.length) {
251 if (line[index] == '^') {
252 data.add(caretBytes[line[index + 1]]!);
253 index += 2;
254 } else {
255 data.add(line.codeUnitAt(index));
256 index++;
257 }
258 }
259 return data;
260}

294(String, int) extendedColor(List<int> values, int start) {
295 final rest = values.sublist(start);
296 if (rest.length >= 2 && rest[0] == 5 && rest[1] <= 255) {
297 return (paletteColor(rest[1]), 2);

299 if (rest.length >= 4 &&
300 rest[0] == 2 &&

465 void receive(int byte) {
466 streamBytes++;
467 if (byte < 0x20) {
468 receiveControl(byte);
469 return;
470 }
471 switch (state) {
472 case ParserState.ground:
473 printChar(String.fromCharCode(byte));
474 case ParserState.escape || ParserState.escapeIntermediate:
475 escapeByte(byte);
476 case ParserState.oscString:
477 osc += String.fromCharCode(byte);
478 case ParserState.csiIgnore:
479 if (isFinal(byte)) {
480 ignore(Reason.malformed);
481 state = ParserState.ground;
482 }
483 case ParserState.csiIntermediate:
484 csiIntermediateByte(byte);
485 case ParserState.csiEntry when isPrivateMarker(byte):
486 marker = String.fromCharCode(byte);
487 state = ParserState.csiParam;
488 case ParserState.csiEntry || ParserState.csiParam:
489 csiParamByte(byte);
490 }
491 }
492
493 void receiveControl(int byte) {
494 if (byte == esc) {
495 if (state == ParserState.oscString) {
496 executeOsc();
497 } else if (state != ParserState.ground) {
498 ignore(Reason.canceled);
499 }
500 clearSequence();
501 state = ParserState.escape;
502 } else if (byte == can || byte == sub) {
503 if (state != ParserState.ground) {
504 ignore(Reason.canceled);
505 state = ParserState.ground;
506 }
507 } else if (state == ParserState.oscString) {
508 if (byte == bel) {
509 executeOsc();
510 state = ParserState.ground;
511 }
512 } else {
513 executeControl(byte);
514 }
515 }
516
517 void escapeByte(int byte) {
518 final char = String.fromCharCode(byte);
519 if (isIntermediate(byte)) {
520 intermediates += char;
521 state = ParserState.escapeIntermediate;
522 } else if (state == ParserState.escape && char == '[') {
523 state = ParserState.csiEntry;
524 } else if (state == ParserState.escape && char == ']') {
525 state = ParserState.oscString;
526 } else {
527 executeEsc(intermediates + char);
528 state = ParserState.ground;
529 }
530 }
531
532 void csiParamByte(int byte) {
533 final char = String.fromCharCode(byte);
534 if (isParameter(byte)) {
535 parameters += char;
536 state = ParserState.csiParam;
537 } else if (isIntermediate(byte)) {
538 intermediates += char;
539 state = ParserState.csiIntermediate;
540 } else if (isFinal(byte)) {
541 executeCsi(char);
542 state = ParserState.ground;
543 } else {
544 state = ParserState.csiIgnore;
545 }
546 }
547
548 void csiIntermediateByte(int byte) {
549 final char = String.fromCharCode(byte);
550 if (isIntermediate(byte)) {
551 intermediates += char;
552 } else if (isFinal(byte)) {
553 executeCsi(char);
554 state = ParserState.ground;
555 } else {
556 state = ParserState.csiIgnore;
557 }
558 }

560 void executeControl(int byte) {

562 case bel:
563 bells++;
564 case bs:

567 case ht:

570 case lf || vt || ff:

575 }
576 case cr:
577 pendingWrap = false;
578 column = 1;
579 case so:
580 gl = 1;
581 case si:
582 gl = 0;
583 }
584 }

705 void executeEsc(String key) {
706 switch (key) {
707 case '7':
708 saveCursor();
709 case '8':
710 restoreCursor();
711 case 'D':
712 pendingWrap = false;
713 lineFeed();
714 case 'E':
715 pendingWrap = false;
716 column = 1;
717 lineFeed();
718 case 'M':
719 pendingWrap = false;
720 reverseLineFeed();
721 case 'H':
722 tabStops.add(column);
723 case 'c':
724 reset();
725 case '=' || '>':
726 setMode(Mode.deckpam, key == '=');
727 case '\\':
728 break;
729 case '(0' || '(B':
730 g0 = charsets[key[1]]!;
731 case ')0' || ')B':
732 g1 = charsets[key[1]]!;
733 case '#8':
734 screenAlignment();
735 default:
736 ignore(Reason.unsupported);
737 return;
738 }
739 sequences++;
740 }

751 void executeCsi(String finalByte) {
752 final key = marker + intermediates + finalByte;
753 final fields = parameters.isEmpty ? <String>[] : parameters.split(';');
754 try {
755 require(fields.length <= maxFields, Reason.invalid);
756 final params = fields.map(fieldValue).toList();
757 final allowed = csiConditions[key];
758 require(allowed == null || allowed.contains(parameter(params, 0)));
759 if (key.length == 1 && wrapClearingCsi.contains(key)) {
760 pendingWrap = false;
761 }
762 dispatchCsi(key, params);
763 } on SequenceIgnored catch (ignoredSequence) {
764 ignore(ignoredSequence.reason);
765 return;
766 }
767 sequences++;
768 }

770 void dispatchCsi(String key, List<int> params) {
771 switch (key) {
772 case 'A':
773 moveUp(amount(params));
774 case 'B' || 'e':
775 moveDown(amount(params));
776 case 'C' || 'a':
777 column = min(column + amount(params), width);
778 case 'D':
779 column = max(column - amount(params), 1);
780 case 'E':
781 moveDown(amount(params));
782 column = 1;
783 case 'F':
784 moveUp(amount(params));
785 column = 1;
786 case 'G' || '`':
787 column = min(amount(params), width);
788 case 'H' || 'f':
789 row = addressRow(amount(params));
790 column = min(max(parameter(params, 1), 1), width);
791 case 'd':
792 row = addressRow(amount(params));
793 case 'I':
794 tabMove(amount(params));
795 case 'Z':
796 reverseTabMove(amount(params));
797 case 'J':
798 eraseInDisplay(parameter(params, 0));
799 case 'K':
800 eraseInLine(parameter(params, 0));
801 case 'X':
802 screen.erase(row, column, min(column + amount(params) - 1, width));
803 case '@':
804 screen.insertCells(
805 row,
806 column,
807 min(amount(params), width - column + 1),
808 );
809 case 'P':
810 screen.deleteCells(
811 row,
812 column,
813 min(amount(params), width - column + 1),
814 );
815 case 'L':
816 insertLines(amount(params));
817 case 'M':
818 deleteLines(amount(params));
819 case 'S':
820 scrollUp(amount(params));
821 case 'T':
822 scrollDown(amount(params));
823 case 'b':
824 repeatCharacter(amount(params));
825 case 'g':
826 clearTabStops(parameter(params, 0));
827 case 'm':
828 selectGraphicRendition(params);
829 case 'h' || 'l':
830 setAnsiModes(params, key == 'h');
831 case '?h' || '?l':
832 setDecModes(params, key == '?h');
833 case ' q':
834 cursorShape = amount(params);
835 case 'r':
836 setScrollRegion(params);
837 case 's':
838 saveCursor();
839 case 'u':
840 restoreCursor();
841 case '!p':
842 softReset();
843 case 'n':
844 deviceStatusReport(parameter(params, 0));
845 case '?n':
846 reply('^[[?${reportedRow()};$column;1R');
847 case 'c':
848 reply('^[[?1;2c');
849 case '>c':
850 reply('^[[>0;95;0c');
851 case 't':
852 reply('^[[8;$height;${width}t');
853 case r'$p':
854 requestAnsiMode(parameter(params, 0));
855 case r'?$p':
856 requestDecMode(parameter(params, 0));
857 default:
858 throw SequenceIgnored(Reason.unsupported);
859 }
860 }

1217 .toList();
