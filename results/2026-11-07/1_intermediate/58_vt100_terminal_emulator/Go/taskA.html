55var modeNames = []string{"IRM", "LNM", "DECCKM", "DECSCNM", "DECOM", "DECAWM", "DECTCEM", "DECKPAM", "BRACKETED"}
56var flagNames = []string{"bold", "dim", "italic", "underline", "blink", "reverse", "hidden", "strike"}
57var ansiModes = map[int]string{4: "IRM", 20: "LNM"}
58var decModes = map[int]string{1: "DECCKM", 5: "DECSCNM", 6: "DECOM", 7: "DECAWM", 25: "DECTCEM", 2004: "BRACKETED"}

61func initial(name string) bool {
62 return name == "DECAWM" || name == "DECTCEM"
63}
64func (t *Terminal) emit(line string) {
65 fmt.Fprintln(t.Out, line)

67func (t *Terminal) blankRow() []Cell {
68 row := make([]Cell, t.W)
69 for c := range row {
70 row[c] = blank
71 }
72 return row
73}
74func (t *Terminal) clearScreen(display int) {
75 t.Screens[display] = make([][]Cell, t.H)
76 for r := 0; r < t.H; r++ {
77 t.Screens[display][r] = t.blankRow()
78 }
79}
80func (t *Terminal) reset() {
81 t.clearScreen(0)
82 t.clearScreen(1)
83 t.Row, t.Col, t.Top, t.Bottom, t.Display, t.Shape = 1, 1, 1, t.H, 0, 1
84 t.Pending = false
85 t.Attr = Attributes{}
86 t.G0, t.G1, t.GL = false, false, false
87 t.Modes = make(map[string]bool)
88 for _, name := range modeNames {
89 t.Modes[name] = initial(name)
90 }
91 t.Saved = [2]*SavedCursor{}
92 t.Tabs = make([]bool, t.W+1)
93 for c := 9; c <= t.W; c += 8 {
94 t.Tabs[c] = true
95 }
96 t.Scrollback = nil
97 t.Title = nil
98 t.HasLast = false
99 t.State = ground
100 t.clearSequence()
101}
102func (t *Terminal) clearSequence() {
103 t.Marker, t.Params, t.Intermediate, t.OSC = "", "", "", ""
104}
105func rowText(row []Cell) string {
106 bytes := make([]byte, len(row))
107 for c, cell := range row {
108 bytes[c] = cell.Char
109 }
110 return strings.TrimRight(string(bytes), " ")
111}
112func (t *Terminal) scrollUp(top, bottom int, history bool) {
113 rows := t.Screens[t.Display]
114 removed := rows[top-1]
115 copy(rows[top-1:bottom-1], rows[top:bottom])
116 rows[bottom-1] = t.blankRow()
117 if history && t.Display == 0 && top == 1 {
118 t.Scrollback = append(t.Scrollback, rowText(removed))
119 if len(t.Scrollback) > 100 {
120 t.Scrollback = t.Scrollback[1:]
121 }
122 }
123}
124func (t *Terminal) scrollDown(top, bottom int) {
125 rows := t.Screens[t.Display]
126 copy(rows[top:bottom], rows[top-1:bottom-1])
127 rows[top-1] = t.blankRow()
128}
129func (t *Terminal) lineFeed() {
130 if t.Row == t.Bottom {
131 t.scrollUp(t.Top, t.Bottom, true)
132 } else if t.Row < t.H {
133 t.Row++
134 }
135}
136func (t *Terminal) reverseFeed() {
137 if t.Row == t.Top {
138 t.scrollDown(t.Top, t.Bottom)
139 } else if t.Row > 1 {
140 t.Row--
141 }
142}
143func (t *Terminal) tab(reverse bool) {
144 if reverse {
145 for c := t.Col - 1; c >= 1; c-- {
146 if t.Tabs[c] {
147 t.Col = c
148 return
149 }
150 }
151 t.Col = 1
152 } else {
153 for c := t.Col + 1; c <= t.W; c++ {
154 if t.Tabs[c] {
155 t.Col = c
156 return
157 }
158 }
159 t.Col = t.W
160 }
161}
162func (t *Terminal) address(v int) int {
163 if t.Modes["DECOM"] {
164 return min(t.Top+v-1, t.Bottom)
165 }
166 return min(v, t.H)
167}
168func (t *Terminal) home() {
169 t.Row, t.Col = t.address(1), 1

171func (t *Terminal) save() {
172 t.Saved[t.Display] = &SavedCursor{t.Row, t.Col, t.Attr, t.Modes["DECOM"], t.G0, t.G1, t.GL}
173}
174func (t *Terminal) restore() {
175 t.Pending = false
176 s := t.Saved[t.Display]
177 if s == nil {
178 s = &SavedCursor{Row: 1, Col: 1}
179 }
180 t.Row, t.Col, t.Attr = s.Row, s.Col, s.Attr
181 t.Modes["DECOM"], t.G0, t.G1, t.GL = s.Origin, s.G0, s.G1, s.GL
182}

184 row := t.Screens[t.Display][t.Row-1]
185 copy(row[t.Col-1+count:], row[t.Col-1:t.W-count])
186 for c := t.Col - 1; c < t.Col-1+count; c++ {
187 row[c] = blank
188 }
189}
190func (t *Terminal) deleteCells(count int) {
191 row := t.Screens[t.Display][t.Row-1]
192 copy(row[t.Col-1:], row[t.Col-1+count:])
193 for c := t.W - count; c < t.W; c++ {
194 row[c] = blank
195 }
196}
197func (t *Terminal) write(c byte) {
198 t.Printed++
199 t.Last = c
200 t.HasLast = true
201 if t.Pending {
202 t.Pending = false
203 t.Col = 1
204 t.lineFeed()
205 }
206 if t.Modes["IRM"] {
207 t.insertCells(1)
208 }
209 t.Screens[t.Display][t.Row-1][t.Col-1] = Cell{c, t.Attr}
210 if t.Col < t.W {
211 t.Col++
212 } else if t.Modes["DECAWM"] {
213 t.Pending = true
214 }
215}
216func (t *Terminal) printChar(c byte) {
217 graphics := t.G0
218 if t.GL {
219 graphics = t.G1
220 }
221 if graphics {
222 switch c {
223 case '`':
224 c = '*'
225 case 'a':
226 c = '#'
227 case 'j', 'k', 'l', 'm', 'n', 't', 'u', 'v', 'w':
228 c = '+'
229 case 'q':
230 c = '-'
231 case 'x':
232 c = '|'
233 case '~':
234 c = 'o'
235 }
236 }
237 t.write(c)
238}
239func (t *Terminal) control(c byte) {
240 switch c {
241 case 7:
242 t.Bells++
243 case 8:
244 t.Pending = false
245 t.Col = max(t.Col-1, 1)
246 case 9:
247 t.Pending = false
248 t.tab(false)
249 case 10, 11, 12:
250 t.Pending = false
251 t.lineFeed()
252 if t.Modes["LNM"] {
253 t.Col = 1
254 }
255 case 13:
256 t.Pending = false
257 t.Col = 1
258 case 14:
259 t.GL = true
260 case 15:
261 t.GL = false
262 }
263}
264func (t *Terminal) executeESC(key string) {
265 switch key {
266 case "7":
267 t.save()
268 case "8":
269 t.restore()
270 case "D":
271 t.Pending = false
272 t.lineFeed()
273 case "E":
274 t.Pending = false
275 t.Col = 1
276 t.lineFeed()
277 case "M":
278 t.Pending = false
279 t.reverseFeed()
280 case "H":
281 t.Tabs[t.Col] = true
282 case "c":
283 t.reset()
284 case "=":
285 t.Modes["DECKPAM"] = true
286 case ">":
287 t.Modes["DECKPAM"] = false
288 case "\\":
289 case "(0":
290 t.G0 = true
291 case "(B":
292 t.G0 = false
293 case ")0":
294 t.G1 = true
295 case ")B":
296 t.G1 = false
297 case "#8":
298 t.Pending = false
299 for _, row := range t.Screens[t.Display] {
300 for c := range row {
301 row[c] = Cell{Char: 'E'}
302 }
303 }
304 t.Top, t.Bottom, t.Row, t.Col = 1, t.H, 1, 1
305 t.Modes["DECOM"] = false
306 default:
307 t.Ignored[0]++
308 return
309 }
310 t.Sequences++
311}
312func (t *Terminal) executeOSC() {
313 if strings.HasPrefix(t.OSC, "0;") || strings.HasPrefix(t.OSC, "2;") {
314 title := t.OSC[2:]
315 t.Title = &title
316 t.Sequences++
317 } else {
318 t.Ignored[0]++
319 }
320}
321func palette(number int) string {
322 names := []string{"black", "red", "green", "yellow", "blue", "magenta", "cyan", "white"}
323 if number < 8 {
324 return names[number]
325 }
326 if number < 16 {
327 return "bright-" + names[number-8]
328 }
329 return strconv.Itoa(number)
330}
331func (t *Terminal) sgr(params []int) bool {
332 if len(params) == 0 {
333 params = []int{0}
334 }
335 a := t.Attr
336 on := map[int]uint{1: 0, 2: 1, 3: 2, 4: 3, 5: 4, 7: 5, 8: 6, 9: 7}
337 off := map[int]uint{23: 2, 24: 3, 25: 4, 27: 5, 28: 6, 29: 7}
338 for i := 0; i < len(params); i++ {
339 v := params[i]
340 if v == 0 {
341 a = Attributes{}
342 } else if bit, ok := on[v]; ok {
343 a.Flags |= 1 << bit
344 } else if v == 22 {
345 a.Flags &^= 3
346 } else if bit, ok := off[v]; ok {
347 a.Flags &^= 1 << bit
348 } else {
349 switch {
350 case v >= 30 && v <= 37:
351 a.FG = palette(v - 30)
352 case v >= 90 && v <= 97:
353 a.FG = palette(v - 82)
354 case v >= 40 && v <= 47:
355 a.BG = palette(v - 40)
356 case v >= 100 && v <= 107:
357 a.BG = palette(v - 92)
358 case v == 39:
359 a.FG = ""
360 case v == 49:
361 a.BG = ""
362 case v == 38 || v == 48:
363 color := ""
364 if i+2 < len(params) && params[i+1] == 5 && params[i+2] <= 255 {
365 color = palette(params[i+2])
366 i += 2
367 } else if i+4 < len(params) && params[i+1] == 2 && params[i+2] <= 255 && params[i+3] <= 255 && params[i+4] <= 255 {
368 color = fmt.Sprintf("#%02x%02x%02x", params[i+2], params[i+3], params[i+4])
369 i += 4
370 } else {
371 return false
372 }
373 if v == 38 {
374 a.FG = color
375 } else {
376 a.BG = color
377 }
378 default:
379 return false
380 }
381 }
382 }
383 t.Attr = a
384 return true
385}
386func (t *Terminal) setModes(params []int, dec, enabled bool) bool {
387 if len(params) == 0 {
388 params = []int{0}
389 }
390 for _, v := range params {
391 if dec {
392 if _, ok := decModes[v]; !ok && v != 47 && v != 1047 && v != 1048 && v != 1049 {
393 return false
394 }
395 } else if _, ok := ansiModes[v]; !ok {
396 return false
397 }
398 }
399 for _, v := range params {
400 if !dec {
401 t.Modes[ansiModes[v]] = enabled
402 continue
403 }
404 if name, ok := decModes[v]; ok {
405 if v == 6 || v == 7 && !enabled {
406 t.Pending = false
407 }
408 t.Modes[name] = enabled
409 if v == 6 {
410 t.home()
411 }
412 } else if v == 1048 {
413 if enabled {
414 t.save()
415 } else {
416 t.restore()
417 }
418 } else {
419 t.Pending = false
420 if enabled && t.Display == 0 {
421 if v == 1049 {
422 t.save()
423 }
424 t.Display = 1
425 if v == 1049 {
426 t.clearScreen(1)
427 }
428 } else if !enabled && t.Display == 1 {
429 if v == 1047 {
430 t.clearScreen(1)
431 }
432 t.Display = 0
433 if v == 1049 {
434 t.restore()
435 }
436 }
437 }
438 }
439 return true
440}

470func (t *Terminal) csi(key string, params []int) int {
471 p1, p2 := 0, 0
472 if len(params) > 0 {
473 p1 = params[0]
474 }
475 if len(params) > 1 {
476 p2 = params[1]
477 }
478 n := max(p1, 1)
479 switch key {
480 case "J":
481 if p1 > 3 {
482 return 0
483 }
484 case "K":
485 if p1 > 2 {
486 return 0
487 }
488 case "g":
489 if p1 != 0 && p1 != 3 {
490 return 0
491 }
492 case " q":
493 if p1 > 6 {
494 return 0
495 }
496 case "n":
497 if p1 != 5 && p1 != 6 {
498 return 0
499 }
500 case "?n":
501 if p1 != 6 {
502 return 0
503 }
504 case "c", ">c":
505 if p1 != 0 {
506 return 0
507 }
508 case "t":
509 if p1 != 18 {
510 return 0
511 }
512 }
513 if len(key) == 1 && strings.Contains("ABCDEFGHIJKLMPSTXZ`adef@", key) {
514 t.Pending = false
515 }
516 switch key {
517 case "A":
518 t.moveUp(n)
519 case "B", "e":
520 t.moveDown(n)
521 case "C", "a":
522 t.Col = min(t.Col+n, t.W)
523 case "D":
524 t.Col = max(t.Col-n, 1)
525 case "E":
526 t.moveDown(n)
527 t.Col = 1
528 case "F":
529 t.moveUp(n)
530 t.Col = 1
531 case "G", "`":
532 t.Col = min(n, t.W)
533 case "H", "f":
534 t.Row = t.address(n)
535 t.Col = min(max(p2, 1), t.W)
536 case "d":
537 t.Row = t.address(n)
538 case "I", "Z":
539 for i := 0; i < n; i++ {
540 t.tab(key == "Z")
541 }
542 case "J":
543 switch p1 {
544 case 0:
545 t.erase(t.Row, t.Col, t.W)
546 for r := t.Row + 1; r <= t.H; r++ {
547 t.erase(r, 1, t.W)
548 }
549 case 1:
550 for r := 1; r < t.Row; r++ {
551 t.erase(r, 1, t.W)
552 }
553 t.erase(t.Row, 1, t.Col)
554 case 2:
555 t.clearScreen(t.Display)
556 case 3:
557 t.Scrollback = nil
558 }
559 case "K":
560 first, last := t.Col, t.W
561 if p1 == 1 || p1 == 2 {
562 first = 1
563 }
564 if p1 == 1 {
565 last = t.Col
566 }
567 t.erase(t.Row, first, last)
568 case "X":
569 t.erase(t.Row, t.Col, min(t.Col+n-1, t.W))
570 case "@":
571 t.insertCells(min(n, t.W-t.Col+1))
572 case "P":
573 t.deleteCells(min(n, t.W-t.Col+1))
574 case "L", "M":
575 if t.Row >= t.Top && t.Row <= t.Bottom {
576 for i := 0; i < min(n, t.Bottom-t.Row+1); i++ {
577 if key == "L" {
578 t.scrollDown(t.Row, t.Bottom)
579 } else {
580 t.scrollUp(t.Row, t.Bottom, false)
581 }
582 }
583 t.Col = 1
584 }
585 case "S", "T":
586 for i := 0; i < min(n, t.Bottom-t.Top+1); i++ {
587 if key == "S" {
588 t.scrollUp(t.Top, t.Bottom, true)
589 } else {
590 t.scrollDown(t.Top, t.Bottom)
591 }
592 }
593 case "b":
594 if t.HasLast {
595 c := t.Last
596 for i := 0; i < n; i++ {
597 t.write(c)
598 }
599 }
600 case "g":
601 if p1 == 0 {
602 t.Tabs[t.Col] = false
603 } else {
604 t.Tabs = make([]bool, t.W+1)
605 }
606 case "m":
607 if !t.sgr(params) {
608 return 0
609 }
610 case "h", "l", "?h", "?l":
611 if !t.setModes(params, key[0] == '?', strings.HasSuffix(key, "h")) {
612 return 0
613 }
614 case " q":
615 t.Shape = n
616 case "r":
617 bottom := p2
618 if bottom == 0 {
619 bottom = t.H
620 }
621 bottom = min(bottom, t.H)
622 if n >= bottom {
623 return 3
624 }
625 t.Pending = false
626 t.Top, t.Bottom = n, bottom
627 t.home()
628 case "s":
629 t.save()
630 case "u":
631 t.restore()
632 case "!p":
633 t.Pending = false
634 for _, name := range []string{"IRM", "DECCKM", "DECOM", "DECAWM", "DECTCEM", "DECKPAM"} {
635 t.Modes[name] = initial(name)
636 }
637 t.Top, t.Bottom = 1, t.H
638 t.Attr = Attributes{}
639 t.G0, t.G1, t.GL = false, false, false
640 t.Saved[t.Display] = nil
641 case "n", "?n":
642 row := t.Row
643 if t.Modes["DECOM"] {
644 row = max(t.Row-t.Top+1, 1)
645 }
646 if key == "?n" {
647 t.reply(fmt.Sprintf("^[[?%d;%d;1R", row, t.Col))
648 } else if p1 == 5 {
649 t.reply("^[[0n")
650 } else {
651 t.reply(fmt.Sprintf("^[[%d;%dR", row, t.Col))
652 }
653 case "c":
654 t.reply("^[[?1;2c")
655 case ">c":
656 t.reply("^[[>0;95;0c")
657 case "t":
658 t.reply(fmt.Sprintf("^[[8;%d;%dt", t.H, t.W))
659 case "$p", "?$p":
660 state, prefix := 0, ""
661 if key == "$p" {
662 if name, ok := ansiModes[p1]; ok {
663 state = modeState(t.Modes[name])
664 }
665 } else {
666 prefix = "?"
667 if name, ok := decModes[p1]; ok {
668 state = modeState(t.Modes[name])
669 } else if p1 == 47 || p1 == 1047 || p1 == 1049 {
670 state = modeState(t.Display == 1)
671 }
672 }
673 t.reply(fmt.Sprintf("^[[%s%d;%d$y", prefix, p1, state))
674 default:
675 return 0
676 }
677 return -1
678}
679func (t *Terminal) executeCSI(final byte) {
680 fields := []string{}
681 if t.Params != "" {
682 fields = strings.Split(t.Params, ";")
683 }
684 if len(fields) > 16 {
685 t.Ignored[3]++
686 return
687 }
688 params := make([]int, len(fields))
689 for i, field := range fields {
690 for _, digit := range field {
691 params[i] = min(9999, params[i]*10+int(digit-'0'))
692 }
693 }
694 reason := t.csi(t.Marker+t.Intermediate+string(final), params)
695 if reason < 0 {
696 t.Sequences++
697 } else {
698 t.Ignored[reason]++
699 }
700}
701func intermediate(c byte) bool {
702 return c >= 0x20 && c <= 0x2f
703}
704func final(c byte) bool {
705 return c >= 0x40 && c <= 0x7e
706}
707func (t *Terminal) receive(c byte) {
708 t.Bytes++
709 if c < 0x20 {
710 switch c {
711 case 27:
712 if t.State == oscString {
713 t.executeOSC()
714 } else if t.State != ground {
715 t.Ignored[2]++
716 }
717 t.clearSequence()
718 t.State = escape
719 case 24, 26:
720 if t.State != ground {
721 t.Ignored[2]++
722 t.State = ground
723 }
724 default:
725 if t.State == oscString {
726 if c == 7 {
727 t.executeOSC()
728 t.State = ground
729 }
730 } else {
731 t.control(c)
732 }
733 }
734 return
735 }
736 switch t.State {
737 case ground:
738 t.printChar(c)
739 case escape, escapeIntermediate:
740 if intermediate(c) {
741 t.Intermediate += string(c)
742 t.State = escapeIntermediate
743 } else if t.State == escape && c == '[' {
744 t.State = csiEntry
745 } else if t.State == escape && c == ']' {
746 t.State = oscString
747 } else {
748 t.executeESC(t.Intermediate + string(c))
749 t.State = ground
750 }
751 case oscString:
752 t.OSC += string(c)
753 case csiIgnore:
754 if final(c) {
755 t.Ignored[1]++
756 t.State = ground
757 }
758 case csiIntermediate:
759 if intermediate(c) {
760 t.Intermediate += string(c)
761 } else if final(c) {
762 t.executeCSI(c)
763 t.State = ground
764 } else {
765 t.State = csiIgnore
766 }
767 default:
768 if t.State == csiEntry && c >= 0x3c && c <= 0x3f {
769 t.Marker = string(c)
770 t.State = csiParam
771 } else if c >= '0' && c <= '9' || c == ';' {
772 t.Params += string(c)
773 t.State = csiParam
774 } else if intermediate(c) {
775 t.Intermediate += string(c)
776 t.State = csiIntermediate
777 } else if final(c) {
778 t.executeCSI(c)
779 t.State = ground
780 } else {
781 t.State = csiIgnore
782 }
783 }
784}
785func (t *Terminal) key(name string) string {
786 cursor := map[string]string{"UP": "A", "DOWN": "B", "RIGHT": "C", "LEFT": "D", "HOME": "H", "END": "F"}
787 if suffix, ok := cursor[name]; ok {
788 if t.Modes["DECCKM"] {
789 return "^[O" + suffix
790 }
791 return "^[[" + suffix
792 }
793 fixed := map[string]string{
794 "INSERT": "^[[2~", "DELETE": "^[[3~", "PAGEUP": "^[[5~", "PAGEDOWN": "^[[6~",
795 "F1": "^[OP", "F2": "^[OQ", "F3": "^[OR", "F4": "^[OS",
796 "F5": "^[[15~", "F6": "^[[17~", "F7": "^[[18~", "F8": "^[[19~",
797 "F9": "^[[20~", "F10": "^[[21~", "F11": "^[[23~", "F12": "^[[24~",
798 "TAB": "^I", "BACKSPACE": "^?", "ESCAPE": "^[",
799 }
800 if value, ok := fixed[name]; ok {
801 return value
802 }
803 enter := "^M"
804 if t.Modes["LNM"] {
805 enter += "^J"
806 }
807 if name == "ENTER" {
808 return enter
809 }
810 if name == "KPENTER" {
811 if t.Modes["DECKPAM"] {
812 return "^[OM"
813 }
814 return enter
815 }
816 digit := name[2] - '0'
817 if t.Modes["DECKPAM"] {
818 return "^[O" + string(byte(0x70)+digit)
819 }
820 return string(name[2])
821}
822func (t *Terminal) screenOutput(label string) {
823 name := "main"
824 if t.Display == 1 {
825 name = "alt"
826 }
827 line := fmt.Sprintf("%s %s cursor %d,%d scrollback %d", label, name, t.Row, t.Col, len(t.Scrollback))
828 if t.Pending {
829 line += " pending"
830 }
831 t.emit(line)
832 for _, row := range t.Screens[t.Display] {
833 t.emit("|" + rowText(row))
834 }
835}
836func describe(a Attributes) string {
837 parts := []string{}
838 for bit, name := range flagNames {
839 if a.Flags&(1<<uint(bit)) != 0 {
840 parts = append(parts, name)
841 }
842 }
843 if a.FG != "" {
844 parts = append(parts, "fg="+a.FG)
845 }
846 if a.BG != "" {
847 parts = append(parts, "bg="+a.BG)
848 }
849 return strings.Join(parts, " ")
850}
851func (t *Terminal) finish() {
852 if t.State != ground {
853 t.Ignored[2]++
854 t.State = ground
855 }
856 t.screenOutput("FINAL")
857 if t.Title == nil {
858 t.emit("TITLE none")
859 } else {
860 t.emit("TITLE \"" + *t.Title + "\"")
861 }
862 modes := []string{}
863 for _, name := range modeNames {
864 if t.Modes[name] != initial(name) {
865 value := "off"
866 if t.Modes[name] {
867 value = "on"
868 }
869 modes = append(modes, name+"="+value)
870 }
871 }
872 if t.G0 {
873 modes = append(modes, "G0=graphics")
874 }
875 if t.G1 {
876 modes = append(modes, "G1=graphics")
877 }
878 if t.GL {
879 modes = append(modes, "GL=G1")
880 }
881 if t.Shape != 1 {
882 modes = append(modes, fmt.Sprintf("CURSOR=%d", t.Shape))
883 }
884 if len(modes) == 0 {
885 modes = append(modes, "default")
886 }
887 t.emit("MODES " + strings.Join(modes, " "))
888 t.emit(fmt.Sprintf("REGION %d-%d", t.Top, t.Bottom))
889 tabs := []string{}
890 for c := 1; c <= t.W; c++ {
891 if t.Tabs[c] {
892 tabs = append(tabs, strconv.Itoa(c))
893 }
894 }
895 if len(tabs) == 0 {
896 tabs = append(tabs, "none")
897 }
898 t.emit("TABS " + strings.Join(tabs, ","))
899 t.emit(fmt.Sprintf("SCROLLBACK %d", len(t.Scrollback)))
900 for _, line := range t.Scrollback[max(0, len(t.Scrollback)-5):] {
901 t.emit("|" + line)
902 }
903 runs := []string{}
904 for r, row := range t.Screens[t.Display] {
905 for start := 0; start < t.W; {
906 end := start + 1
907 for end < t.W && row[end].Attr == row[start].Attr {
908 end++
909 }
910 if row[start].Attr != (Attributes{}) {
911 runs = append(runs, fmt.Sprintf("row %d col %d-%d %s", r+1, start+1, end, describe(row[start].Attr)))
912 }
913 start = end
914 }
915 }
916 t.emit(fmt.Sprintf("ATTRS %d", len(runs)))
917 for _, line := range runs {
918 t.emit(line)
919 }
920 t.emit(fmt.Sprintf("BELL %d", t.Bells))
921 t.emit(fmt.Sprintf("REPLIES %d", t.Replies))
922 t.emit(fmt.Sprintf("STATS bytes=%d printed=%d sequences=%d", t.Bytes, t.Printed, t.Sequences))
923 v := t.Ignored
924 t.emit(fmt.Sprintf("IGNORED %d unsupported=%d malformed=%d canceled=%d invalid=%d", v[0]+v[1]+v[2]+v[3], v[0], v[1], v[2], v[3]))
925}
926func main() {
927 input := bufio.NewScanner(os.Stdin)
928 input.Buffer(make([]byte, 256), 256)
929 input.Scan()
930 dimensions := strings.Fields(input.Text())
931 w, _ := strconv.Atoi(dimensions[0])
932 h, _ := strconv.Atoi(dimensions[1])
933 output := bufio.NewWriter(os.Stdout)
934 defer output.Flush()
935 t := Terminal{W: w, H: h, Out: output}
936 t.reset()
937 caret := map[byte]byte{'@': 0, 'G': 7, 'H': 8, 'I': 9, 'J': 10, 'K': 11, 'L': 12, 'M': 13, 'N': 14, 'O': 15, 'X': 24, 'Z': 26, '[': 27, '^': 94}
938 for input.Scan() {
939 line := input.Text()
940 if line == "%%END" {
941 break
942 }
943 if line == "%%DUMP" {
944 t.Dumps++
945 t.screenOutput(fmt.Sprintf("DUMP %d", t.Dumps))
946 } else if strings.HasPrefix(line, "%%KEY ") {
947 name := line[6:]
948 t.emit("KEY " + name + " " + t.key(name))
949 } else if strings.HasPrefix(line, "%%PASTE ") {
950 sent := strings.ReplaceAll(line[8:], "^", "^^")
951 if t.Modes["BRACKETED"] {
952 sent = "^[[200~" + sent + "^[[201~"
953 }
954 t.emit("PASTE " + sent)
955 } else {
956 for i := 0; i < len(line); i++ {
957 c := line[i]
958 if c == '^' {
959 i++
960 c = caret[line[i]]
961 }
962 t.receive(c)
963 }
964 }
965 }
966 t.finish()
967}
