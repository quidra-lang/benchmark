179 static String decodeStream(String line) {
180 StringBuilder data = new StringBuilder();
181 for (int index = 0; index < line.length(); index++) {
182 char ch = line.charAt(index);
183 if (ch == '^') {
184 index++;
185 data.append(CARET_BYTES.get(line.charAt(index)));
186 } else {
187 data.append(ch);
188 }
189 }
190 return data.toString();
191 }

417 void receive(char ch) {
418 streamBytes++;
419 if (ch < 0x20) {
420 receiveControl(ch);
421 return;
422 }
423 switch (state) {
424 case GROUND -> printChar(ch);
425 case ESCAPE, ESCAPE_INTERMEDIATE -> escapeByte(ch);
426 case OSC_STRING -> osc.append(ch);
427 case CSI_IGNORE -> {
428 if (isFinal(ch)) {
429 ignore(Reason.MALFORMED);
430 state = State.GROUND;
431 }
432 }
433 case CSI_INTERMEDIATE -> csiIntermediateByte(ch);
434 case CSI_ENTRY -> {
435 if (isPrivateMarker(ch)) {
436 marker = String.valueOf(ch);
437 state = State.CSI_PARAM;
438 } else {
439 csiParamByte(ch);
440 }
441 }
442 case CSI_PARAM -> csiParamByte(ch);
443 }
444 }
445
446 private void receiveControl(char ch) {
447 if (ch == ESC) {
448 if (state == State.OSC_STRING) {
449 executeOsc();
450 } else if (state != State.GROUND) {
451 ignore(Reason.CANCELED);
452 }
453 clearSequence();
454 state = State.ESCAPE;
455 } else if (ch == CAN || ch == SUB) {
456 if (state != State.GROUND) {
457 ignore(Reason.CANCELED);
458 state = State.GROUND;
459 }
460 } else if (state == State.OSC_STRING) {
461 if (ch == BEL) {
462 executeOsc();
463 state = State.GROUND;
464 }
465 } else {
466 executeControl(ch);
467 }
468 }
469
470 private void escapeByte(char ch) {
471 if (isIntermediate(ch)) {
472 intermediates.append(ch);
473 state = State.ESCAPE_INTERMEDIATE;
474 } else if (state == State.ESCAPE && ch == '[') {
475 state = State.CSI_ENTRY;
476 } else if (state == State.ESCAPE && ch == ']') {
477 state = State.OSC_STRING;
478 } else {
479 executeEsc(intermediates.toString() + ch);
480 state = State.GROUND;
481 }
482 }
483
484 private void csiParamByte(char ch) {
485 if (isParameter(ch)) {
486 parameters.append(ch);
487 state = State.CSI_PARAM;
488 } else if (isIntermediate(ch)) {
489 intermediates.append(ch);
490 state = State.CSI_INTERMEDIATE;
491 } else if (isFinal(ch)) {
492 executeCsi(ch);
493 state = State.GROUND;
494 } else {
495 state = State.CSI_IGNORE;
496 }
497 }
498
499 private void csiIntermediateByte(char ch) {
500 if (isIntermediate(ch)) {
501 intermediates.append(ch);
502 } else if (isFinal(ch)) {
503 executeCsi(ch);
504 state = State.GROUND;
505 } else {
506 state = State.CSI_IGNORE;
507 }
508 }

510 private void executeControl(char ch) {
511 switch (ch) {
512 case BEL -> bells++;
513 case BS -> {
514 pendingWrap = false;
515 column = Math.max(column - 1, 1);
516 }
517 case HT -> {
518 pendingWrap = false;
519 tabMove(1);
520 }
521 case LF, VT, FF -> {
522 pendingWrap = false;
523 lineFeed();
524 if (modes.contains(Mode.LNM)) {
525 column = 1;
526 }
527 }
528 case CR -> {
529 pendingWrap = false;
530 column = 1;
531 }
532 case SO -> gl = 1;
533 case SI -> gl = 0;
534 }
535 }

650 private void executeEsc(String key) {
651 switch (key) {
652 case "7" -> saveCursor();
653 case "8" -> restoreCursor();
654 case "D" -> {
655 pendingWrap = false;
656 lineFeed();
657 }
658 case "E" -> {
659 pendingWrap = false;
660 column = 1;
661 lineFeed();
662 }
663 case "M" -> {
664 pendingWrap = false;
665 reverseLineFeed();
666 }
667 case "H" -> tabStops.add(column);
668 case "c" -> reset();
669 case "=", ">" -> setMode(Mode.DECKPAM, key.equals("="));
670 case "\\" -> {
671 }
672 case "(0", "(B" -> g0 = CHARSETS.get(key.charAt(1));
673 case ")0", ")B" -> g1 = CHARSETS.get(key.charAt(1));
674 case "#8" -> screenAlignment();
675 default -> {
676 ignore(Reason.UNSUPPORTED);
677 return;
678 }
679 }
680 sequences++;
681 }

693 private void executeCsi(char finalByte) {
694 String key = marker + intermediates + finalByte;
695 String text = parameters.toString();
696 String[] fields = text.isEmpty() ? new String[0] : text.split(";", -1);
697 try {
698 require(fields.length <= MAX_FIELDS, Reason.INVALID);
699 int[] params = Arrays.stream(fields).mapToInt(Main::fieldValue).toArray();
700 Set<Integer> allowed = CSI_CONDITIONS.get(key);
701 require(allowed == null || allowed.contains(parameter(params, 0)));
702 if (key.length() == 1 && WRAP_CLEARING_CSI.contains(key)) {
703 pendingWrap = false;
704 }
705 dispatchCsi(key, params);
706 } catch (SequenceIgnored sequence) {
707 ignore(sequence.reason);
708 return;
709 }
710 sequences++;
711 }

713 private void dispatchCsi(String key, int[] params) {
714 switch (key) {
715 case "A" -> moveUp(amount(params));
716 case "B", "e" -> moveDown(amount(params));
717 case "C", "a" -> column = Math.min(column + amount(params), width);
718 case "D" -> column = Math.max(column - amount(params), 1);
719 case "E" -> {
720 moveDown(amount(params));
721 column = 1;
722 }
723 case "F" -> {
724 moveUp(amount(params));
725 column = 1;
726 }
727 case "G", "`" -> column = Math.min(amount(params), width);
728 case "H", "f" -> {
729 row = addressRow(amount(params));
730 column = Math.min(Math.max(parameter(params, 1), 1), width);
731 }
732 case "d" -> row = addressRow(amount(params));
733 case "I" -> tabMove(amount(params));
734 case "Z" -> reverseTabMove(amount(params));
735 case "J" -> eraseInDisplay(parameter(params, 0));
736 case "K" -> eraseInLine(parameter(params, 0));
737 case "X" -> eraseCharacters(amount(params));
738 case "@" -> insertCharacters(amount(params));
739 case "P" -> deleteCharacters(amount(params));
740 case "L" -> insertLines(amount(params));
741 case "M" -> deleteLines(amount(params));
742 case "S" -> scrollUp(amount(params));
743 case "T" -> scrollDown(amount(params));
744 case "b" -> repeatCharacter(amount(params));
745 case "g" -> clearTabStops(parameter(params, 0));
746 case "m" -> selectGraphicRendition(params);
747 case "h", "l" -> setAnsiModes(params, key.equals("h"));
748 case "?h", "?l" -> setDecModes(params, key.equals("?h"));
749 case " q" -> cursorShape = amount(params);
750 case "r" -> setScrollRegion(params);
751 case "s" -> saveCursor();
752 case "u" -> restoreCursor();
753 case "!p" -> softReset();
754 case "n" -> deviceStatusReport(parameter(params, 0));
755 case "?n" -> reply("^[[?" + reportedRow() + ";" + column + ";1R");
756 case "c" -> reply("^[[?1;2c");

761 default -> throw new SequenceIgnored(Reason.UNSUPPORTED);

859 private void selectGraphicRendition(int[] params) {
860 int[] values = params.length == 0 ? new int[] {0} : params;
861 int flags = attributes.flags();
862 String foreground = attributes.foreground();
863 String background = attributes.background();
864 for (int index = 0; index < values.length; index++) {
865 int value = values[index];
866 if (value == 0) {
867 flags = 0;
868 foreground = null;
869 background = null;
870 } else if (SGR_FLAGS_ON.containsKey(value)) {
871 flags |= 1 << SGR_FLAGS_ON.get(value);
872 } else if (SGR_FLAGS_OFF.containsKey(value)) {
873 for (int flag : SGR_FLAGS_OFF.get(value)) {
874 flags &= ~(1 << flag);
875 }
876 } else if (value >= 30 && value <= 37) {
877 foreground = paletteColor(value - 30);
878 } else if (value >= 90 && value <= 97) {
879 foreground = paletteColor(value - 82);
880 } else if (value >= 40 && value <= 47) {
881 background = paletteColor(value - 40);
882 } else if (value >= 100 && value <= 107) {
883 background = paletteColor(value - 92);
884 } else if (value == 39) {
885 foreground = null;
886 } else if (value == 49) {
887 background = null;
888 } else {
889 require(value == 38 || value == 48);
890 ExtendedColor color = extendedColor(values, index + 1);
891 if (value == 38) {
892 foreground = color.name();
893 } else {
894 background = color.name();
895 }
896 index += color.length();
897 }
898 }
899 attributes = new Attributes(flags, foreground, background);
900 }

1020 private String keyString(String name) {
1021 Character cursorKey = CURSOR_KEYS.get(name);
1022 if (cursorKey != null) {
1023 return (modes.contains(Mode.DECCKM) ? "^[O" : "^[[") + cursorKey;
1024 }
1025 String fixedKey = FIXED_KEYS.get(name);
1026 if (fixedKey != null) {
1027 return fixedKey;
1028 }
1029 String enter = modes.contains(Mode.LNM) ? "^M^J" : "^M";
1030 if (name.equals("ENTER")) {
1031 return enter;
1032 }
1033 if (name.equals("KPENTER")) {
1034 return modes.contains(Mode.DECKPAM) ? "^[OM" : enter;
1035 }
1036 int digit = Integer.parseInt(name.substring(2));
1037 if (modes.contains(Mode.DECKPAM)) {
1038 return "^[O" + (char) (0x70 + digit);
1039 }
1040 return String.valueOf(digit);
1041 }

1094 private List<String> attributeRuns() {
1095 List<String> runs = new ArrayList<>();
1096 List<Cell[]> rows = screen().rows;
1097 for (int number = 1; number <= height; number++) {
1098 Cell[] cells = rows.get(number - 1);
1099 int start = 0;
1100 for (int next = 1; next <= width; next++) {
1101 Attributes run = cells[start].attributes();
1102 if (next < width && cells[next].attributes().equals(run)) {
1103 continue;
1104 }
1105 if (!run.isDefault()) {
1106 runs.add("row " + number + " col " + (start + 1) + "-" + next + " "
1107 + run.describe());
1108 }
1109 start = next;

1115 void finish() {
1116 if (state != State.GROUND) {
1117 ignore(Reason.CANCELED);
1118 state = State.GROUND;
1119 }
1120 screenOutput("FINAL");
1121 emit("TITLE " + (title == null ? "none" : "\"" + title + "\""));
1122 List<String> changed = changedModes();
1123 emit("MODES " + (changed.isEmpty() ? "default" : String.join(" ", changed)));
1124 emit("REGION " + top + "-" + bottom);
1125 List<String> stops = tabStops.stream().map(String::valueOf).toList();
1126 emit("TABS " + (stops.isEmpty() ? "none" : String.join(",", stops)));
1127 emit("SCROLLBACK " + scrollback.size());
1128 int first = Math.max(scrollback.size() - SCROLLBACK_SHOWN, 0);
1129 for (String text : scrollback.subList(first, scrollback.size())) {
1130 emit("|" + text);
1131 }
1132 List<String> runs = attributeRuns();
1133 emit("ATTRS " + runs.size());
1134 runs.forEach(this::emit);
1135 emit("BELL " + bells);
1136 emit("REPLIES " + replies);
1137 emit("STATS bytes=" + streamBytes + " printed=" + printed + " sequences=" + sequences);
1138 List<String> counts = Arrays.stream(Reason.values())
1139 .map(reason -> reason.name().toLowerCase(Locale.ROOT) + "="
1140 + ignored[reason.ordinal()])
1141 .toList();
1142 emit("IGNORED " + Arrays.stream(ignored).sum() + " " + String.join(" ", counts));
1143 }

1146 public static void main(String[] args) throws IOException {
1147 BufferedReader reader = new BufferedReader(new InputStreamReader(System.in));
1148 String[] size = reader.readLine().split(" ");
1149 Terminal terminal = new Terminal(Integer.parseInt(size[0]), Integer.parseInt(size[1]));
1150 String line;
1151 while ((line = reader.readLine()) != null && !line.equals("%%END")) {
1152 if (line.equals("%%DUMP")) {
1153 terminal.dump();
1154 } else if (line.startsWith(KEY_PREFIX)) {
1155 terminal.pressKey(line.substring(KEY_PREFIX.length()));
1156 } else if (line.startsWith(PASTE_PREFIX)) {
1157 terminal.paste(line.substring(PASTE_PREFIX.length()));
1158 } else {
1159 for (char ch : decodeStream(line).toCharArray()) {
1160 terminal.receive(ch);
1161 }
1162 }
1163 }
1164 terminal.finish();
1165 System.out.print(terminal.output());
1166 }
