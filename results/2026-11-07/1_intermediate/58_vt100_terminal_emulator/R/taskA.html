41scroll_up <- function(t, top, bottom, history = FALSE) {
42 removed <- t$screens[[t$display]][[top]]

44 t$screens[[t$display]][[bottom]] <- blank_row(t)
45 if (history && t$display == 1L && top == 1L) {

201sgr <- function(t, params) {
202 values <- if (length(params)) params else 0L
203 attr <- t$attr
204 on <- c(`1` = 0L, `2` = 1L, `3` = 2L, `4` = 3L, `5` = 4L, `7` = 5L, `8` = 6L, `9` = 7L)
205 off <- c(`23` = 2L, `24` = 3L, `25` = 4L, `27` = 5L, `28` = 6L, `29` = 7L)
206 index <- 1L
207 while (index <= length(values)) {
208 value <- values[index]
209 name <- as.character(value)
210 if (value == 0L) {
211 attr <- default_attr
212 } else if (name %in% names(on)) {
213 attr$flags <- bitwOr(attr$flags, bitwShiftL(1L, on[[name]]))
214 } else if (value == 22L) {
215 attr$flags <- bitwAnd(attr$flags, bitwNot(3L))
216 } else if (name %in% names(off)) {
217 attr$flags <- bitwAnd(attr$flags, bitwNot(bitwShiftL(1L, off[[name]])))
218 } else if (value >= 30L && value <= 37L) {
219 attr$fg <- palette(value - 30L)
220 } else if (value >= 90L && value <= 97L) {
221 attr$fg <- palette(value - 82L)
222 } else if (value >= 40L && value <= 47L) {
223 attr$bg <- palette(value - 40L)
224 } else if (value >= 100L && value <= 107L) {
225 attr$bg <- palette(value - 92L)
226 } else if (value == 39L) {
227 attr$fg <- ""
228 } else if (value == 49L) {
229 attr$bg <- ""
230 } else if (value %in% c(38L, 48L)) {
231 if (index + 2L <= length(values) && values[index + 1L] == 5L && values[index + 2L] <= 255L) {
232 color <- palette(values[index + 2L])
233 index <- index + 2L
234 } else if (index + 4L <= length(values) && values[index + 1L] == 2L && all(values[index + 2:4] <= 255L)) {
235 color <- paste0("#", paste(sprintf("%02x", values[index + 2:4]), collapse = ""))
236 index <- index + 4L
237 } else {
238 return(FALSE)
239 }
240 if (value == 38L) attr$fg <- color else attr$bg <- color
241 } else {
242 return(FALSE)
243 }
244 index <- index + 1L
245 }
246 t$attr <- attr
247 TRUE
248}

254 name <- as.character(value)

257 } else if (name %in% names(dec_modes)) {

262 if (enabled) save_cursor(t) else restore_cursor(t)

264 t$pending <- FALSE

266 if (value == 1049L) save_cursor(t)
267 t$display <- 2L
268 if (value == 1049L) clear_screen(t, 2L)
269 } else if (!enabled && t$display == 2L) {
270 if (value == 1047L) clear_screen(t, 2L)
271 t$display <- 1L
272 if (value == 1049L) restore_cursor(t)
273 }
274 }
275 }
276 TRUE

288csi <- function(t, key, params) {
289 p1 <- if (length(params)) params[1] else 0L

291 n <- max(p1, 1L)
292 conditions <- list(J = 0:3, K = 0:2, g = c(0L, 3L), ` q` = 0:6, n = c(5L, 6L), `?n` = 6L, c = 0L, `>c` = 0L, t = 18L)
293 if (key %in% names(conditions) && !(p1 %in% conditions[[key]])) return(1L)
294 wrap_keys <- strsplit("ABCDEFGHIJKLMPSTXZ`adef@", "", fixed = TRUE)[[1]]
295 if (key %in% wrap_keys) t$pending <- FALSE

302 } else if (key == "D") {

307 } else if (key == "F") {

310 } else if (key %in% c("G", "`")) {
311 t$column <- min(n, t$width)
312 } else if (key %in% c("H", "f")) {
313 t$row <- address(t, n)
314 t$column <- min(max(p2, 1L), t$width)
315 } else if (key == "d") {
316 t$row <- address(t, n)
317 } else if (key %in% c("I", "Z")) {
318 for (i in seq_len(n)) tab_move(t, key == "Z")
319 } else if (key == "J") {
320 if (p1 == 0L) {
321 erase(t, t$row, t$column, t$width)
322 for (row in span(t$row + 1L, t$height)) erase(t, row, 1L, t$width)
323 } else if (p1 == 1L) {
324 for (row in span(1L, t$row - 1L)) erase(t, row, 1L, t$width)
325 erase(t, t$row, 1L, t$column)
326 } else if (p1 == 2L) {
327 clear_screen(t, t$display)
328 } else {
329 t$scrollback <- character()
330 }
331 } else if (key == "K") {
332 erase(t, t$row, if (p1 %in% c(1L, 2L)) 1L else t$column, if (p1 == 1L) t$column else t$width)
333 } else if (key == "X") {
334 erase(t, t$row, t$column, min(t$column + n - 1L, t$width))
335 } else if (key == "@") {
336 insert_cells(t, min(n, t$width - t$column + 1L))
337 } else if (key == "P") {
338 delete_cells(t, min(n, t$width - t$column + 1L))
339 } else if (key %in% c("L", "M")) {
340 if (t$row >= t$top && t$row <= t$bottom) {
341 for (i in seq_len(min(n, t$bottom - t$row + 1L))) {
342 if (key == "L") scroll_down(t, t$row, t$bottom) else scroll_up(t, t$row, t$bottom)
343 }
344 t$column <- 1L
345 }
346 } else if (key %in% c("S", "T")) {
347 for (i in seq_len(min(n, t$bottom - t$top + 1L))) {
348 if (key == "S") scroll_up(t, t$top, t$bottom, TRUE) else scroll_down(t, t$top, t$bottom)
349 }
350 } else if (key == "b") {
351 if (!is.null(t$last)) {
352 char <- t$last
353 for (i in seq_len(n)) write_char(t, char)
354 }
355 } else if (key == "g") {
356 t$tabs <- if (p1 == 0L) setdiff(t$tabs, t$column) else integer()
357 } else if (key == "m") {
358 if (!sgr(t, params)) return(1L)
359 } else if (key %in% c("h", "l", "?h", "?l")) {
360 if (!set_modes(t, params, startsWith(key, "?"), endsWith(key, "h"))) return(1L)
361 } else if (key == " q") {
362 t$shape <- n
363 } else if (key == "r") {
364 bottom <- min(if (p2 == 0L) t$height else p2, t$height)
365 if (n >= bottom) return(4L)
366 t$pending <- FALSE
367 t$top <- n
368 t$bottom <- bottom
369 home(t)
370 } else if (key == "s") {
371 save_cursor(t)
372 } else if (key == "u") {
373 restore_cursor(t)
374 } else if (key == "!p") {
375 t$pending <- FALSE
376 for (name in c("IRM", "DECCKM", "DECOM", "DECAWM", "DECTCEM", "DECKPAM")) t$modes[[name]] <- initial(name)
377 t$top <- 1L
378 t$bottom <- t$height
379 t$attr <- default_attr
380 t$g0 <- FALSE
381 t$g1 <- FALSE
382 t$gl <- FALSE
383 t$saved[t$display] <- list(NULL)
384 } else if (key %in% c("n", "?n")) {
385 row <- if (t$modes[["DECOM"]]) max(t$row - t$top + 1L, 1L) else t$row
386 text <- if (key == "?n") paste0("^[[?", row, ";", t$column, ";1R") else if (p1 == 5L) "^[[0n" else paste0("^[[", row, ";", t$column, "R")
387 reply(t, text)
388 } else if (key == "c") {
389 reply(t, "^[[?1;2c")
390 } else if (key == ">c") {
391 reply(t, "^[[>0;95;0c")
392 } else if (key == "t") {
393 reply(t, paste0("^[[8;", t$height, ";", t$width, "t"))

395 state <- 0L
396 prefix <- ""
397 name <- as.character(p1)
398 if (key == "$p") {
399 if (name %in% names(ansi_modes)) state <- mode_state(t$modes[[ansi_modes[[name]]]])

401 prefix <- "?"

403 }

406 return(1L)
407 }
408 0L

427receive <- function(t, byte) {
428 t$bytes <- t$bytes + 1L
429 if (byte < 32L) {
430 if (byte == 27L) {
431 if (t$state == "osc") execute_osc(t) else if (t$state != "ground") t$ignored[3] <- t$ignored[3] + 1L
432 clear_sequence(t)
433 t$state <- "escape"
434 } else if (byte %in% c(24L, 26L)) {
435 if (t$state != "ground") {
436 t$ignored[3] <- t$ignored[3] + 1L
437 t$state <- "ground"
438 }
439 } else if (t$state == "osc") {
440 if (byte == 7L) {
441 execute_osc(t)
442 t$state <- "ground"
443 }
444 } else {
445 control(t, byte)
446 }
447 return(invisible(NULL))
448 }
449 char <- intToUtf8(byte)
450 if (t$state == "ground") {
451 print_char(t, char)
452 } else if (t$state %in% c("escape", "escape_intermediate")) {
453 if (is_intermediate(byte)) {
454 t$intermediates <- paste0(t$intermediates, char)
455 t$state <- "escape_intermediate"
456 } else if (t$state == "escape" && char == "[") {
457 t$state <- "csi_entry"
458 } else if (t$state == "escape" && char == "]") {
459 t$state <- "osc"
460 } else {
461 execute_esc(t, paste0(t$intermediates, char))
462 t$state <- "ground"
463 }
464 } else if (t$state == "osc") {
465 t$osc <- paste0(t$osc, char)
466 } else if (t$state == "csi_ignore") {
467 if (is_final(byte)) {
468 t$ignored[2] <- t$ignored[2] + 1L
469 t$state <- "ground"
470 }
471 } else if (t$state == "csi_intermediate") {
472 if (is_intermediate(byte)) {
473 t$intermediates <- paste0(t$intermediates, char)
474 } else if (is_final(byte)) {
475 execute_csi(t, char)
476 t$state <- "ground"
477 } else {
478 t$state <- "csi_ignore"
479 }
480 } else if (t$state == "csi_entry" && byte >= 60L && byte <= 63L) {
481 t$marker <- char
482 t$state <- "csi_param"
483 } else if (byte >= 48L && byte <= 57L || byte == 59L) {
484 t$parameters <- paste0(t$parameters, char)
485 t$state <- "csi_param"
486 } else if (is_intermediate(byte)) {
487 t$intermediates <- paste0(t$intermediates, char)
488 t$state <- "csi_intermediate"
489 } else if (is_final(byte)) {
490 execute_csi(t, char)
491 t$state <- "ground"
492 } else {
493 t$state <- "csi_ignore"
494 }

510screen_output <- function(t, label) {
511 cat(label, if (t$display == 1L) "main" else "alt", "cursor", paste0(t$row, ",", t$column), "scrollback", length(t$scrollback), sep = " ")
512 cat(if (t$pending) " pending\n" else "\n")
513 for (row in t$screens[[t$display]]) cat("|", row_text(row), "\n", sep = "")
514}
515describe <- function(attr) {
516 parts <- flag_names[vapply(0:7, function(bit) bitwAnd(attr$flags, bitwShiftL(1L, bit)) != 0L, logical(1))]
517 if (nzchar(attr$fg)) parts <- c(parts, paste0("fg=", attr$fg))
518 if (nzchar(attr$bg)) parts <- c(parts, paste0("bg=", attr$bg))
519 paste(parts, collapse = " ")
520}
521finish <- function(t) {
522 if (t$state != "ground") {
523 t$ignored[3] <- t$ignored[3] + 1L
524 t$state <- "ground"
525 }
526 screen_output(t, "FINAL")
527 cat("TITLE ", if (is.null(t$title)) "none" else paste0('"', t$title, '"'), "\n", sep = "")
528 modes <- character()
529 for (name in mode_names) if (t$modes[[name]] != initial(name)) modes <- c(modes, paste0(name, "=", if (t$modes[[name]]) "on" else "off"))
530 if (t$g0) modes <- c(modes, "G0=graphics")
531 if (t$g1) modes <- c(modes, "G1=graphics")
532 if (t$gl) modes <- c(modes, "GL=G1")
533 if (t$shape != 1L) modes <- c(modes, paste0("CURSOR=", t$shape))
534 cat("MODES ", if (length(modes)) paste(modes, collapse = " ") else "default", "\n", sep = "")
535 cat("REGION ", t$top, "-", t$bottom, "\n", sep = "")
536 cat("TABS ", if (length(t$tabs)) paste(sort(t$tabs), collapse = ",") else "none", "\n", sep = "")
537 cat("SCROLLBACK ", length(t$scrollback), "\n", sep = "")
538 for (text in tail(t$scrollback, 5L)) cat("|", text, "\n", sep = "")
539 runs <- character()
540 for (r in seq_len(t$height)) {
541 row <- t$screens[[t$display]][[r]]
542 start <- 1L
543 while (start <= t$width) {
544 last <- start
545 while (last < t$width && identical(row[[last + 1L]]$attr, row[[start]]$attr)) last <- last + 1L
546 if (!identical(row[[start]]$attr, default_attr)) runs <- c(runs, paste0("row ", r, " col ", start, "-", last, " ", describe(row[[start]]$attr)))
547 start <- last + 1L
548 }
549 }
550 cat("ATTRS ", length(runs), "\n", sep = "")
551 for (line in runs) cat(line, "\n", sep = "")
552 cat("BELL ", t$bells, "\n", sep = "")
553 cat("REPLIES ", t$replies, "\n", sep = "")
554 cat("STATS bytes=", t$bytes, " printed=", t$printed, " sequences=", t$sequences, "\n", sep = "")
555 v <- t$ignored
556 cat("IGNORED ", sum(v), " unsupported=", v[1], " malformed=", v[2], " canceled=", v[3], " invalid=", v[4], "\n", sep = "")
557}
558main <- function() {
559 lines <- readLines(file("stdin"), warn = FALSE)
560 dimensions <- as.integer(strsplit(lines[1], " ", fixed = TRUE)[[1]])
561 terminal <- new.env(parent = emptyenv())
562 terminal$width <- dimensions[1]
563 terminal$height <- dimensions[2]
564 terminal$bytes <- 0L
565 terminal$printed <- 0L
566 terminal$sequences <- 0L
567 terminal$bells <- 0L
568 terminal$replies <- 0L
569 terminal$dumps <- 0L
570 terminal$ignored <- integer(4)
571 reset(terminal)
572 caret <- setNames(c(0L, 7L, 8L, 9L, 10L, 11L, 12L, 13L, 14L, 15L, 24L, 26L, 27L, 94L), c("@", "G", "H", "I", "J", "K", "L", "M", "N", "O", "X", "Z", "[", "^"))
573 for (line in lines[-1L]) {
574 if (line == "%%END") break
575 if (line == "%%DUMP") {
576 terminal$dumps <- terminal$dumps + 1L
577 screen_output(terminal, paste("DUMP", terminal$dumps))
578 } else if (startsWith(line, "%%KEY ")) {
579 name <- substring(line, 7L)
580 cat("KEY ", name, " ", key_string(terminal, name), "\n", sep = "")
581 } else if (startsWith(line, "%%PASTE ")) {
582 sent <- gsub("^", "^^", substring(line, 9L), fixed = TRUE)
583 if (terminal$modes[["BRACKETED"]]) sent <- paste0("^[[200~", sent, "^[[201~")
584 cat("PASTE ", sent, "\n", sep = "")
585 } else {
586 bytes <- utf8ToInt(line)
587 index <- 1L
588 while (index <= length(bytes)) {
589 if (bytes[index] == 94L) {
590 receive(terminal, caret[[intToUtf8(bytes[index + 1L])]])
591 index <- index + 2L
592 } else {
593 receive(terminal, bytes[index])
594 index <- index + 1L
595 }
596 }
597 }
598 }
599 finish(terminal)
600}
601main()
