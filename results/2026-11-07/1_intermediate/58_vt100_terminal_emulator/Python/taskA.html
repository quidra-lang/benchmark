116def decode_stream(line):
117 data = []
118 index = 0
119 while index < len(line):
120 if line[index] == "^":
121 data.append(CARET_BYTES[line[index + 1]])
122 index += 2
123 else:
124 data.append(ord(line[index]))
125 index += 1
126 return data

193 parts = [name for index, name in enumerate(FLAG_NAMES) if self.flags >> index & 1]

238 cell = Cell(char, DEFAULT_ATTRIBUTES)

242 self.rows[row - 1][column - 1] = cell

244 def erase(self, row, first, last):
245 cells = self.rows[row - 1]
246 for column in range(first, last + 1):
247 cells[column - 1] = BLANK_CELL

249 def erase_rows(self, first, last):
250 for row in range(first, last + 1):
251 self.rows[row - 1] = self.blank_row()
252
253 def scroll_up(self, top, bottom):
254 removed = self.rows.pop(top - 1)
255 self.rows.insert(bottom - 1, self.blank_row())

257
258 def scroll_down(self, top, bottom):
259 self.rows.pop(bottom - 1)

263 cells = self.rows[row - 1]

268 cells = self.rows[row - 1]

369 def receive(self, byte):
370 self.stream_bytes += 1
371 if byte < 0x20:
372 self.receive_control(byte)
373 elif self.state == GROUND:
374 self.print_char(chr(byte))
375 elif self.state in (ESCAPE, ESCAPE_INTERMEDIATE):
376 self.escape_byte(byte)
377 elif self.state == OSC_STRING:
378 self.osc += chr(byte)
379 elif self.state == CSI_IGNORE:
380 if is_final(byte):
381 self.ignored[MALFORMED] += 1
382 self.state = GROUND
383 elif self.state == CSI_INTERMEDIATE:
384 self.csi_intermediate_byte(byte)
385 elif self.state == CSI_ENTRY and is_private_marker(byte):
386 self.marker = chr(byte)
387 self.state = CSI_PARAM
388 else:
389 self.csi_param_byte(byte)
390
391 def receive_control(self, byte):
392 if byte == ESC:
393 if self.state == OSC_STRING:
394 self.execute_osc()
395 elif self.state != GROUND:
396 self.ignored[CANCELED] += 1
397 self.clear_sequence()
398 self.state = ESCAPE
399 elif byte in (CAN, SUB):
400 if self.state != GROUND:
401 self.ignored[CANCELED] += 1
402 self.state = GROUND
403 elif self.state == OSC_STRING:
404 if byte == BEL:
405 self.execute_osc()
406 self.state = GROUND
407 else:
408 self.execute_control(byte)
409
410 def escape_byte(self, byte):
411 char = chr(byte)
412 if is_intermediate(byte):
413 self.intermediates += char
414 self.state = ESCAPE_INTERMEDIATE
415 elif self.state == ESCAPE and char == "[":
416 self.state = CSI_ENTRY
417 elif self.state == ESCAPE and char == "]":
418 self.state = OSC_STRING
419 else:
420 self.execute_esc(self.intermediates + char)
421 self.state = GROUND
422
423 def csi_param_byte(self, byte):
424 char = chr(byte)
425 if is_parameter(byte):
426 self.parameters += char
427 self.state = CSI_PARAM
428 elif is_intermediate(byte):
429 self.intermediates += char
430 self.state = CSI_INTERMEDIATE
431 elif is_final(byte):
432 self.execute_csi(char)
433 self.state = GROUND
434 else:
435 self.state = CSI_IGNORE
436
437 def csi_intermediate_byte(self, byte):
438 char = chr(byte)
439 if is_intermediate(byte):
440 self.intermediates += char
441 elif is_final(byte):
442 self.execute_csi(char)
443 self.state = GROUND
444 else:
445 self.state = CSI_IGNORE

447 def execute_control(self, byte):
448 if byte == BEL:
449 self.bells += 1
450 elif byte == BS:
451 self.pending_wrap = False
452 self.column = max(self.column - 1, 1)
453 elif byte == HT:
454 self.pending_wrap = False
455 self.tab_move(1)
456 elif byte in (LF, VT, FF):
457 self.pending_wrap = False
458 self.line_feed()
459 if self.modes["LNM"]:
460 self.column = 1
461 elif byte == CR:
462 self.pending_wrap = False
463 self.column = 1
464 elif byte == SO:
465 self.gl = "G1"
466 elif byte == SI:
467 self.gl = "G0"

562 def execute_esc(self, key):
563 if key == "7":
564 self.save_cursor()
565 elif key == "8":
566 self.restore_cursor()
567 elif key == "D":
568 self.pending_wrap = False
569 self.line_feed()
570 elif key == "E":
571 self.pending_wrap = False
572 self.column = 1
573 self.line_feed()
574 elif key == "M":
575 self.pending_wrap = False
576 self.reverse_line_feed()
577 elif key == "H":
578 self.tab_stops.add(self.column)
579 elif key == "c":
580 self.reset()
581 elif key in ("=", ">"):
582 self.modes["DECKPAM"] = key == "="
583 elif key == "\\":
584 pass
585 elif key in ("(0", "(B"):
586 self.g0 = CHARSETS[key[1]]
587 elif key in (")0", ")B"):
588 self.g1 = CHARSETS[key[1]]
589 elif key == "#8":
590 self.screen_alignment()
591 else:
592 self.ignored[UNSUPPORTED] += 1
593 return

603 def execute_csi(self, final):
604 key = self.marker + self.intermediates + final
605 fields = self.parameters.split(";") if self.parameters else []
606 try:
607 require(len(fields) <= MAX_FIELDS, INVALID)
608 require(key in self.csi_handlers)
609 params = [field_value(field) for field in fields]
610 if key in CSI_CONDITIONS:
611 require(parameter(params, 0) in CSI_CONDITIONS[key])
612 if key in WRAP_CLEARING_CSI:
613 self.pending_wrap = False
614 self.csi_handlers[key](params)
615 except SequenceIgnored as ignored:
616 self.ignored[ignored.reason] += 1
617 else:
618 self.sequences += 1

729 def select_graphic_rendition(self, params):
730 values = params or [0]
731 attributes = self.attributes.copy()
732 index = 0
733 while index < len(values):
734 value = values[index]
735 if value == 0:
736 attributes = Attributes()
737 elif value in SGR_FLAGS_ON:
738 attributes.flags |= 1 << SGR_FLAGS_ON[value]
739 elif value in SGR_FLAGS_OFF:
740 for flag in SGR_FLAGS_OFF[value]:
741 attributes.flags &= ~(1 << flag)
742 elif 30 <= value <= 37:
743 attributes.foreground = palette_color(value - 30)
744 elif 90 <= value <= 97:
745 attributes.foreground = palette_color(value - 82)
746 elif 40 <= value <= 47:
747 attributes.background = palette_color(value - 40)
748 elif 100 <= value <= 107:
749 attributes.background = palette_color(value - 92)
750 elif value == 39:
751 attributes.foreground = None
752 elif value == 49:
753 attributes.background = None
754 else:
755 require(value in (38, 48))
756 color, used = extended_color(values, index + 1)
757 if value == 38:
758 attributes.foreground = color
759 else:
760 attributes.background = color
761 index += used
762 index += 1
763 self.attributes = attributes

928 def attribute_runs(self):
929 runs = []
930 for row, cells in enumerate(self.screen().rows, 1):
931 start = 0
932 for column in range(1, self.width + 1):
933 if column < self.width and cells[column].attributes == cells[start].attributes:
934 continue

937 runs.append(f"row {row} col {start + 1}-{column} {attributes.describe()}")

941 def finish(self):
942 if self.state != GROUND:
943 self.ignored[CANCELED] += 1
944 self.state = GROUND
945 self.screen_output("FINAL")
946 self.emit("TITLE", "none" if self.title is None else f'"{self.title}"')
947 self.emit("MODES", *(self.changed_modes() or ["default"]))
948 self.emit("REGION", f"{self.top}-{self.bottom}")
949 stops = sorted(self.tab_stops)
950 self.emit("TABS", ",".join(str(stop) for stop in stops) if stops else "none")
951 self.emit("SCROLLBACK", len(self.scrollback))
952 for text in self.scrollback[-SCROLLBACK_SHOWN:]:
953 self.output.append("|" + text)
954 runs = self.attribute_runs()
955 self.emit("ATTRS", len(runs))
956 self.output.extend(runs)
957 self.emit("BELL", self.bells)
958 self.emit("REPLIES", self.replies)
959 self.emit(
960 "STATS", f"bytes={self.stream_bytes}", f"printed={self.printed}",
961 f"sequences={self.sequences}",
962 )
963 counts = [f"{reason}={count}" for reason, count in self.ignored.items()]
964 self.emit("IGNORED", sum(self.ignored.values()), *counts)

967def main():
968 lines = sys.stdin.read().splitlines()
969 width, height = (int(value) for value in lines[0].split())
970 terminal = Terminal(width, height)
971 for line in lines[1:]:
972 if line == "%%END":
973 break
974 if line == "%%DUMP":
975 terminal.dump()
976 elif line.startswith("%%KEY "):
977 terminal.press_key(line[len("%%KEY "):])
978 elif line.startswith("%%PASTE "):
979 terminal.paste(line[len("%%PASTE "):])
980 else:
981 for byte in decode_stream(line):
982 terminal.receive(byte)
983 terminal.finish()
984 sys.stdout.write("\n".join(terminal.output) + "\n")
