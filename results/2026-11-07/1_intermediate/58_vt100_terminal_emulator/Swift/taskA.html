19final class Terminal {
20 let width: Int, height: Int
21 let irm = 0, lnm = 1, ckm = 2, scnm = 3, origin = 4, wrap = 5, visible = 6, keypad = 7, bracketed = 8
22 var screens: [[[Cell]]] = []
23 var screen = 0, row = 1, col = 1, top = 1, bottom = 1, shape = 1
24 var pending = false, g0 = false, g1 = false, gl = false
25 var modes = Array(repeating: false, count: 9)
26 var tabs = Set<Int>()
27 var saved: [Saved?] = [nil, nil]
28 var scrollback: [String] = []
29 var title: String?
30 var last: Character?
31 var attributes = Attributes()
32 var parser = ParserState.ground
33 var marker = "", parameters = "", intermediate = "", osc = ""
34 var bytes = 0, printed = 0, sequences = 0, bells = 0, replies = 0, dumps = 0

37 init(_ width: Int, _ height: Int) {
38 self.width = width
39 self.height = height
40 reset()
41 }
42 func blankRow() -> [Cell] {
43 Array(repeating: Cell(), count: width)
44 }
45 func blankScreen() -> [[Cell]] {
46 Array(repeating: blankRow(), count: height)
47 }
48 func reset() {
49 screens = [blankScreen(), blankScreen()]
50 screen = 0
51 row = 1
52 col = 1
53 top = 1
54 bottom = height
55 shape = 1
56 pending = false
57 attributes = Attributes()
58 modes = Array(repeating: false, count: 9)
59 modes[wrap] = true
60 modes[visible] = true
61 g0 = false
62 g1 = false
63 gl = false
64 saved = [nil, nil]
65 tabs = Set(stride(from: 9, through: width, by: 8))
66 scrollback = []
67 title = nil
68 last = nil
69 parser = .ground
70 marker = ""
71 parameters = ""
72 intermediate = ""
73 osc = ""
74 }
75 func rowText(_ cells: [Cell]) -> String {
76 var chars = cells.map(\.character)
77 while chars.last == " " {
78 chars.removeLast()
79 }
80 return String(chars)
81 }
82 func erase(_ r: Int, _ first: Int, _ end: Int) {
83 if first > end {
84 return
85 }
86 for c in first...end {
87 screens[screen][r - 1][c - 1] = Cell()
88 }
89 }
90 func scroll(_ up: Bool, _ count: Int) {
91 for _ in 0..<count {
92 if up {
93 if screen == 0 && top == 1 {
94 scrollback.append(rowText(screens[0][top - 1]))
95 if scrollback.count > 100 {
96 scrollback.removeFirst()
97 }
98 }
99 for r in top..<bottom {
100 screens[screen][r - 1] = screens[screen][r]
101 }
102 screens[screen][bottom - 1] = blankRow()
103 } else {
104 for r in stride(from: bottom, through: top + 1, by: -1) {
105 screens[screen][r - 1] = screens[screen][r - 2]
106 }
107 screens[screen][top - 1] = blankRow()
108 }
109 }
110 }
111 func lineFeed() {
112 if row == bottom {
113 scroll(true, 1)
114 } else if row < height {
115 row += 1
116 }
117 }
118 func tab(_ reverse: Bool) {
119 col = reverse ? (tabs.filter { $0 < col }.max() ?? 1) : (tabs.filter { $0 > col }.min() ?? width)
120 }
121 func address(_ n: Int) -> Int {
122 modes[origin] ? min(top + n - 1, bottom) : min(n, height)
123 }
124 func home() {
125 row = address(1)
126 col = 1
127 }
128 func save() {
129 saved[screen] = Saved(
130 row: row,
131 col: col,
132 attributes: attributes,
133 origin: modes[origin],
134 g0: g0,
135 g1: g1,
136 gl: gl
137 )
138 }
139 func restore() {
140 if let s = saved[screen] {
141 row = s.row
142 col = s.col
143 attributes = s.attributes
144 modes[origin] = s.origin
145 g0 = s.g0
146 g1 = s.g1
147 gl = s.gl
148 } else {
149 row = 1
150 col = 1
151 attributes = Attributes()
152 modes[origin] = false
153 g0 = false
154 g1 = false
155 gl = false
156 }
157 pending = false
158 }
159 func write(_ ch: Character) {
160 printed += 1
161 last = ch
162 if pending {
163 pending = false
164 col = 1
165 lineFeed()
166 }
167 if modes[irm] {
168 for c in stride(from: width - 1, through: col, by: -1) {
169 screens[screen][row - 1][c] = screens[screen][row - 1][c - 1]
170 }
171 }
172 screens[screen][row - 1][col - 1] = Cell(character: ch, attributes: attributes)
173 if col < width {
174 col += 1
175 } else if modes[wrap] {
176 pending = true
177 }
178 }
179 func printCharacter(_ character: Character) {
180 var ch = character
181 if gl ? g1 : g0 {
182 switch ch {
183 case "`":
184 ch = "*"
185 case "a":
186 ch = "#"
187 case "q":
188 ch = "-"
189 case "x":
190 ch = "|"
191 case "~":
192 ch = "o"
193 case "j", "k", "l", "m", "n", "t", "u", "v", "w":
194 ch = "+"
195 default:
196 break
197 }
198 }
199 write(ch)
200 }
201 func control(_ b: UInt8) {
202 if b >= 8 && b <= 13 {
203 pending = false
204 }
205 switch b {
206 case 7:
207 bells += 1
208 case 8:
209 col = max(1, col - 1)
210 case 9:
211 tab(false)
212 case 10...12:
213 lineFeed()
214 if modes[lnm] {
215 col = 1
216 }
217 case 13:
218 col = 1
219 case 14:
220 gl = true
221 case 15:
222 gl = false
223 default:
224 break
225 }
226 }
227 func palette(_ n: Int) -> String {
228 let names = ["black", "red", "green", "yellow", "blue", "magenta", "cyan", "white"]
229 if n < 8 {
230 return names[n]
231 }
232 if n < 16 {
233 return "bright-" + names[n - 8]
234 }
235 return String(n)
236 }
237 func sgr(_ p: [Int]) -> Bool {
238 var a = attributes, i = 0
239 while i < p.count {
240 let v = p[i]
241 switch v {
242 case 0:
243 a = Attributes()
244 case 1...5:
245 a.flags |= 1 << (v - 1)
246 case 7...9:
247 a.flags |= 1 << (v - 2)
248 case 22:
249 a.flags &= ~3
250 case 23...25:
251 a.flags &= ~(1 << (v - 21))
252 case 27...29:
253 a.flags &= ~(1 << (v - 22))
254 case 30...37:
255 a.fg = palette(v - 30)
256 case 90...97:
257 a.fg = palette(v - 82)
258 case 40...47:
259 a.bg = palette(v - 40)
260 case 100...107:
261 a.bg = palette(v - 92)
262 case 39:
263 a.fg = "default"
264 case 49:
265 a.bg = "default"
266 case 38, 48:
267 let color: String
268 if i + 2 < p.count && p[i + 1] == 5 && p[i + 2] <= 255 {
269 color = palette(p[i + 2])
270 i += 2
271 } else if i + 4 < p.count && p[i + 1] == 2 && p[(i + 2)...(i + 4)].allSatisfy({ $0 <= 255 }) {
272 color = String(format: "#%02x%02x%02x", p[i + 2], p[i + 3], p[i + 4])
273 i += 4
274 } else {
275 return false
276 }
277 if v == 38 {
278 a.fg = color
279 } else {
280 a.bg = color
281 }
282 default:
283 return false
284 }
285 i += 1
286 }
287 attributes = a
288 return true
289 }
290 func modeIndex(_ p: Int, _ dec: Bool) -> Int? {
291 if dec {
292 switch p {
293 case 1:
294 return ckm
295 case 5:
296 return scnm
297 case 6:
298 return origin
299 case 7:
300 return wrap
301 case 25:
302 return visible
303 case 2004:
304 return bracketed
305 default:
306 return nil
307 }
308 }
309 return p == 4 ? irm : p == 20 ? lnm : nil
310 }
311 func setModes(_ p: [Int], _ dec: Bool, _ on: Bool) -> Bool {
312 if p
313 .contains(where: { modeIndex($0, dec) == nil && !(dec && [47, 1047, 1048, 1049].contains($0)) })
314 {
315 return false
316 }
317 for v in p {
318 if let index = modeIndex(v, dec) {
319 modes[index] = on
320 if index == origin {
321 pending = false
322 home()
323 }
324 if index == wrap && !on {
325 pending = false
326 }
327 } else if v == 1048 {
328 if on {
329 save()
330 } else {
331 restore()
332 }
333 } else {
334 pending = false
335 if on && screen == 0 {
336 if v == 1049 {
337 save()
338 }
339 screen = 1
340 if v == 1049 {
341 screens[1] = blankScreen()
342 }
343 } else if !on && screen == 1 {
344 if v == 1047 {
345 screens[1] = blankScreen()
346 }
347 screen = 0
348 if v == 1049 {
349 restore()
350 }
351 }
352 }
353 }
354 return true
355 }
356 func executeOsc() {
357 if osc.hasPrefix("0;") || osc.hasPrefix("2;") {
358 title = String(osc.dropFirst(2))
359 sequences += 1
360 } else {
361 ignored[0] += 1
362 }
363 }
364 func escape(_ final: Character) {
365 let key = intermediate + String(final)
366 switch key {
367 case "7":
368 save()
369 case "8":
370 restore()
371 case "D", "E":
372 pending = false
373 if final == "E" {
374 col = 1
375 }
376 lineFeed()
377 case "M":
378 pending = false
379 if row == top {
380 scroll(false, 1)
381 } else {
382 row = max(1, row - 1)
383 }
384 case "H":
385 tabs.insert(col)
386 case "c":
387 reset()
388 case "=":
389 modes[keypad] = true
390 case ">":
391 modes[keypad] = false
392 case "\\":
393 break
394 case "(0":
395 g0 = true
396 case "(B":
397 g0 = false
398 case ")0":
399 g1 = true
400 case ")B":
401 g1 = false
402 case "#8":
403 screens[screen] = Array(repeating: Array(repeating: Cell(character: "E"), count: width), count: height)
404 top = 1
405 bottom = height
406 modes[origin] = false
407 row = 1
408 col = 1
409 pending = false
410 default:
411 ignored[0] += 1
412 return
413 }
414 sequences += 1
415 }
416 func reply(_ s: String) {
417 replies += 1
418 print("REPLY " + s)
419 }
420 func csi(_ final: Character) {
421 let fields = parameters.isEmpty ? [] : parameters.split(separator: ";", omittingEmptySubsequences: false)
422 if fields.count >= 17 {
423 ignored[3] += 1
424 return
425 }
426 var p = fields.map { field in field.utf8.reduce(0) { min(9999, $0 * 10 + Int($1) - 48) } }
427 if p.isEmpty {
428 p = [0]
429 }
430 let first = p[0], second = p.count > 1 ? p[1] : 0, n = max(first, 1)
431 let key = marker + intermediate + String(final)
432 if (key == "J" && first > 3) || (key == "K" && first > 2) || (key == "g" && first != 0 && first != 3)
433 || (key == " q" && first > 6) || (key == "n" && first != 5 && first != 6) || (key == "?n" && first != 6)
434 || (["c", ">c"].contains(key) && first != 0) || (key == "t" && first != 18)
435 {
436 ignored[0] += 1
437 return
438 }
439 if key.count == 1 && "ABCDEFGHIJKLMPSTXZ`adef@".contains(final) {
440 pending = false
441 }
442 var supported = true
443 switch key {
444 case "A", "F":
445 row = max(row - n, row >= top ? top : 1)
446 if key == "F" {
447 col = 1
448 }
449 case "B", "e", "E":
450 row = min(row + n, row <= bottom ? bottom : height)
451 if key == "E" {
452 col = 1
453 }
454 case "C", "a":
455 col = min(col + n, width)
456 case "D":
457 col = max(col - n, 1)
458 case "G", "`":
459 col = min(n, width)
460 case "H", "f":
461 row = address(n)
462 col = min(max(second, 1), width)
463 case "d":
464 row = address(n)
465 case "I", "Z":
466 for _ in 0..<n {
467 tab(key == "Z")
468 }
469 case "J":
470 if first == 3 {
471 scrollback = []
472 } else {
473 for r in 1...height {
474 if first == 2 || (first == 0 && r > row) || (first == 1 && r < row) {
475 erase(r, 1, width)
476 } else if r == row {
477 erase(r, first == 0 ? col : 1, first == 1 ? col : width)
478 }
479 }
480 }
481 case "K":
482 erase(row, first == 0 ? col : 1, first == 1 ? col : width)
483 case "X":
484 erase(row, col, min(col + n - 1, width))
485 case "@", "P":
486 let amount = min(n, width - col + 1)
487 if key == "@" {
488 for c in stride(from: width - 1, through: col - 1 + amount, by: -1) {
489 screens[screen][row - 1][c] = screens[screen][row - 1][c - amount]
490 }
491 erase(row, col, col + amount - 1)
492 } else {
493 for c in (col - 1)..<(width - amount) {
494 screens[screen][row - 1][c] = screens[screen][row - 1][c + amount]
495 }
496 erase(row, width - amount + 1, width)
497 }
498 case "L", "M":
499 if row >= top && row <= bottom {
500 let amount = min(n, bottom - row + 1)
501 if key == "L" {
502 for r in stride(from: bottom, through: row + amount, by: -1) {
503 screens[screen][r - 1] = screens[screen][r - amount - 1]
504 }
505 for r in row..<(row + amount) {
506 screens[screen][r - 1] = blankRow()
507 }
508 } else {
509 if row <= bottom - amount {
510 for r in row...(bottom - amount) {
511 screens[screen][r - 1] = screens[screen][r + amount - 1]
512 }
513 }
514 for r in (bottom - amount + 1)...bottom {
515 screens[screen][r - 1] = blankRow()
516 }
517 }
518 col = 1
519 }
520 case "S", "T":
521 scroll(key == "S", min(n, bottom - top + 1))
522 case "b":
523 if let ch = last {
524 for _ in 0..<n {
525 write(ch)
526 }
527 }
528 case "g":
529 if first == 0 {
530 tabs.remove(col)
531 } else {
532 tabs = []
533 }
534 case "m":
535 supported = sgr(p)
536 case "h", "l", "?h", "?l":
537 supported = setModes(p, key.hasPrefix("?"), final == "h")
538 case " q":
539 shape = n
540 case "r":
541 let t = first == 0 ? 1 : first, b = min(second == 0 ? height : second, height)
542 if t >= b {
543 ignored[3] += 1
544 return
545 }
546 pending = false
547 top = t
548 bottom = b
549 home()
550 case "s":
551 save()
552 case "u":
553 restore()
554 case "!p":
555 pending = false
556 modes[irm] = false
557 modes[ckm] = false
558 modes[origin] = false
559 modes[keypad] = false
560 modes[wrap] = true
561 modes[visible] = true
562 top = 1
563 bottom = height
564 attributes = Attributes()
565 g0 = false
566 g1 = false
567 gl = false
568 saved[screen] = nil
569 case "n":
570 reply(first == 5 ? "^[[0n" : "^[[\(modes[origin] ? max(row - top + 1, 1) : row);\(col)R")
571 case "?n":
572 reply("^[[?\(modes[origin] ? max(row - top + 1, 1) : row);\(col);1R")
573 case "c":
574 reply("^[[?1;2c")
575 case ">c":
576 reply("^[[>0;95;0c")
577 case "t":
578 reply("^[[8;\(height);\(width)t")
579 case "$p", "?$p":
580 let dec = key.hasPrefix("?")
581 let status: Int
582 if let index = modeIndex(first, dec) {
583 status = modes[index] ? 1 : 2
584 } else if dec && [47, 1047, 1049].contains(first) {
585 status = screen == 1 ? 1 : 2
586 } else {
587 status = 0
588 }
589 reply("^[[\(dec ? "?" : "")\(first);\(status)$y")
590 default:
591 supported = false
592 }
593 if supported {
594 sequences += 1
595 } else {
596 ignored[0] += 1
597 }
598 }
599 func feed(_ b: UInt8) {
600 bytes += 1
601 if b == 27 {
602 if parser == .osc {
603 executeOsc()
604 } else if parser != .ground {
605 ignored[2] += 1
606 }
607 marker = ""
608 parameters = ""
609 intermediate = ""
610 osc = ""
611 parser = .escape
612 return
613 }
614 if b == 24 || b == 26 {
615 if parser != .ground {
616 ignored[2] += 1
617 }
618 parser = .ground
619 return
620 }
621 if b < 32 {
622 if parser == .osc {
623 if b == 7 {
624 executeOsc()
625 parser = .ground
626 }
627 } else {
628 control(b)
629 }
630 return
631 }
632 let c = Character(UnicodeScalar(b))
633 if parser == .csiEntry {
634 parser = .csiParam
635 if b >= 60 && b <= 63 {
636 marker = String(c)
637 return
638 }
639 }
640 switch parser {
641 case .ground:
642 printCharacter(c)
643 case .osc:
644 osc.append(c)
645 case .escape:
646 if b <= 47 {
647 intermediate.append(c)
648 parser = .escapeIntermediate
649 } else if c == "[" {
650 parser = .csiEntry
651 } else if c == "]" {
652 parser = .osc
653 } else {
654 escape(c)
655 parser = .ground
656 }
657 case .escapeIntermediate:
658 if b <= 47 {
659 intermediate.append(c)
660 } else {
661 escape(c)
662 parser = .ground
663 }
664 case .csiParam:
665 if (b >= 48 && b <= 57) || c == ";" {
666 parameters.append(c)
667 } else if b <= 47 {
668 intermediate.append(c)
669 parser = .csiIntermediate
670 } else if b < 64 {
671 parser = .csiIgnore
672 } else {
673 csi(c)
674 parser = .ground
675 }
676 case .csiIntermediate:
677 if b <= 47 {
678 intermediate.append(c)
679 } else if b < 64 {
680 parser = .csiIgnore
681 } else {
682 csi(c)
683 parser = .ground
684 }
685 case .csiIgnore:
686 if b >= 64 {
687 ignored[1] += 1
688 parser = .ground
689 }
690 case .csiEntry:
691 break
692 }
693 }
694 func dump(_ label: String) {
695 print(
696 "\(label) \(screen == 0 ? "main" : "alt") cursor \(row),\(col) scrollback \(scrollback.count)\(pending ? " pending" : "")"
697 )
698 for cells in screens[screen] {
699 print("|" + rowText(cells))
700 }
701 }
702 func key(_ name: String) -> String {
703 let nav = ["UP", "DOWN", "RIGHT", "LEFT", "HOME", "END"]
704 if let index = nav.firstIndex(of: name) {
705 return (modes[ckm] ? "^[O" : "^[[") + String(Array("ABCDHF")[index])
706 }
707 if name.hasPrefix("F") {
708 let f = Int(name.dropFirst())!
709 if f <= 4 {
710 return "^[O" + String(Character(UnicodeScalar(80 + f - 1)!))
711 }
712 return "^[[\([15, 17, 18, 19, 20, 21, 23, 24][f - 5])~"
713 }
714 if name.hasPrefix("KP") && name.count == 3 {
715 let digit = name.last!
716 return modes[keypad] ? "^[O" + String(Character(UnicodeScalar(112 + Int(String(digit))!)!)) : String(digit)
717 }
718 switch name {
719 case "INSERT":
720 return "^[[2~"
721 case "DELETE":
722 return "^[[3~"
723 case "PAGEUP":
724 return "^[[5~"
725 case "PAGEDOWN":
726 return "^[[6~"
727 case "TAB":
728 return "^I"
729 case "BACKSPACE":
730 return "^?"
731 case "ESCAPE":
732 return "^["
733 case "KPENTER" where modes[keypad]:
734 return "^[OM"
735 default:
736 return modes[lnm] ? "^M^J" : "^M"
737 }
738 }
739 func directive(_ line: String) {
740 if line == "%%DUMP" {
741 dumps += 1
742 dump("DUMP \(dumps)")
743 } else if line.hasPrefix("%%KEY ") {
744 let name = String(line.dropFirst(6))
745 print("KEY \(name) \(key(name))")
746 } else {
747 let s = String(line.dropFirst(8)).replacingOccurrences(of: "^", with: "^^")
748 print("PASTE " + (modes[bracketed] ? "^[[200~" + s + "^[[201~" : s))
749 }
750 }
751 func attributeText(_ a: Attributes) -> String {
752 let names = ["bold", "dim", "italic", "underline", "blink", "reverse", "hidden", "strike"]
753 var parts = (0..<8).filter { a.flags & (1 << $0) != 0 }.map { names[$0] }
754 if a.fg != "default" {
755 parts.append("fg=" + a.fg)
756 }
757 if a.bg != "default" {
758 parts.append("bg=" + a.bg)
759 }
760 return parts.joined(separator: " ")
761 }
762 func finish() {
763 if parser != .ground {
764 ignored[2] += 1
765 }
766 dump("FINAL")
767 print(title.map { "TITLE \"\($0)\"" } ?? "TITLE none")
768 let names = ["IRM", "LNM", "DECCKM", "DECSCNM", "DECOM", "DECAWM", "DECTCEM", "DECKPAM", "BRACKETED"]
769 var changes: [String] = []
770 for i in 0..<9
771 where modes[i] != (i == wrap || i == visible)
772 {
773 changes.append(names[i] + (modes[i] ? "=on" : "=off"))
774 }
775 if g0 {
776 changes.append("G0=graphics")
777 }
778 if g1 {
779 changes.append("G1=graphics")
780 }
781 if gl {
782 changes.append("GL=G1")
783 }
784 if shape != 1 {
785 changes.append("CURSOR=\(shape)")
786 }
787 print("MODES " + (changes.isEmpty ? "default" : changes.joined(separator: " ")))
788 print("REGION \(top)-\(bottom)")
789 print("TABS " + (tabs.isEmpty ? "none" : tabs.sorted().map(String.init).joined(separator: ",")))
790 print("SCROLLBACK \(scrollback.count)")
791 for s in scrollback.suffix(5) {
792 print("|" + s)
793 }
794 var runs: [String] = []
795 for r in 0..<height {
796 var c = 0
797 while c < width {
798 let a = screens[screen][r][c].attributes
799 var end = c + 1
800 while end < width && screens[screen][r][end].attributes == a {
801 end += 1
802 }
803 if a != Attributes() {
804 runs.append("row \(r + 1) col \(c + 1)-\(end) \(attributeText(a))")
805 }
806 c = end
807 }
808 }
809 print("ATTRS \(runs.count)")
810 for s in runs {
811 print(s)
812 }
813 print("BELL \(bells)\nREPLIES \(replies)\nSTATS bytes=\(bytes) printed=\(printed) sequences=\(sequences)")
814 print(
815 "IGNORED \(ignored.reduce(0, +)) unsupported=\(ignored[0]) malformed=\(ignored[1]) canceled=\(ignored[2]) invalid=\(ignored[3])"
816 )
817 }
818}
819
820let size = readLine()!.split(separator: " ").map { Int($0)! }
821let terminal = Terminal(size[0], size[1])
822while let line = readLine() {
823 if line == "%%END" {
824 terminal.finish()
825 break
826 }
827 if line.hasPrefix("%%") {
828 terminal.directive(line)
829 continue
830 }
831 let data = Array(line.utf8)
832 var i = 0
833 while i < data.count {
834 var b = data[i]
835 if b == 94 {
836 i += 1
837 b = data[i]
838 if b != 94 {
839 b -= 64
840 }
841 }
842 terminal.feed(b)
843 i += 1
844 }
845}
