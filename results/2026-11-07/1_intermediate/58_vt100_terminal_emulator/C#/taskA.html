188 static string Palette(int n) {
189 string[] colors = { "black", "red", "green", "yellow", "blue", "magenta", "cyan", "white" };
190 return n < 8 ? colors[n] : n < 16 ? "bright-" + colors[n - 8] : n.ToString();

192 bool Sgr(int[] p) {
193 Attributes a = attributes;
194 for (int i = 0; i < p.Length; ++i) {
195 int v = p[i], bit = v >= 1 && v <= 5 ? v - 1 : v >= 7 && v <= 9 ? v - 2 : -1;
196 if (v == 0)
197 a = Default;
198 else if (bit >= 0)
199 a = a with { Flags = a.Flags | (1 << bit) };
200 else if (v == 22)
201 a = a with { Flags = a.Flags & ~3 };
202 else if (v >= 23 && v <= 25)
203 a = a with { Flags = a.Flags & ~(1 << (v - 21)) };
204 else if (v >= 27 && v <= 29)
205 a = a with { Flags = a.Flags & ~(1 << (v - 22)) };
206 else if (v >= 30 && v <= 37)
207 a = a with { Fg = Palette(v - 30) };
208 else if (v >= 90 && v <= 97)
209 a = a with { Fg = Palette(v - 82) };
210 else if (v >= 40 && v <= 47)
211 a = a with { Bg = Palette(v - 40) };
212 else if (v >= 100 && v <= 107)
213 a = a with { Bg = Palette(v - 92) };
214 else if (v == 39)
215 a = a with { Fg = "default" };
216 else if (v == 49)
217 a = a with { Bg = "default" };
218 else if (v == 38 || v == 48) {
219 string color;
220 if (i + 2 < p.Length && p[i + 1] == 5 && p[i + 2] <= 255) {
221 color = Palette(p[i + 2]);
222 i += 2;
223 } else if (i + 4 < p.Length && p[i + 1] == 2 && p[i + 2] <= 255 &&
224 p[i + 3] <= 255 && p[i + 4] <= 255) {
225 color = $"#{p[i+2]:x2}{p[i+3]:x2}{p[i+4]:x2}";
226 i += 4;
227 } else
228 return false;
229 a = v == 38 ? a with { Fg = color } : a with { Bg = color };
230 } else
231 return false;
232 }
233 attributes = a;
234 return true;
235 }
236 static int ModeIndex(int p, bool dec) => dec ? p switch
237 { 1 => CKM, 5 => SCNM, 6 => ORIGIN, 7 => WRAP, 25 => VISIBLE, 2004 => BRACKETED, _ => -1 }
238 : p switch { 4 => IRM, 20 => LNM, _ => -1 };
239 bool SetModes(int[] p, bool dec, bool on) {
240 if (p.Any(v => ModeIndex(v, dec) < 0 &&
241 !(dec && (v == 47 || v == 1047 || v == 1048 || v == 1049))))
242 return false;
243 foreach (int v in p) {
244 int index = ModeIndex(v, dec);
245 if (index >= 0) {
246 modes[index] = on;
247 if (index == ORIGIN) {
248 pending = false;
249 Home();
250 }
251 if (index == WRAP && !on)
252 pending = false;
253 } else if (v == 1048) {
254 if (on)
255 Save();
256 else
257 Restore();
258 } else {
259 pending = false;
260 if (on && screen == 0) {
261 if (v == 1049)
262 Save();
263 screen = 1;
264 if (v == 1049)
265 screens[1] = BlankScreen();
266 } else if (!on && screen == 1) {
267 if (v == 1047)
268 screens[1] = BlankScreen();
269 screen = 0;
270 if (v == 1049)
271 Restore();
272 }
273 }
274 }
275 return true;

354 void Csi(char final) {
355 string[] fields =
356 parameters.Length == 0 ? Array.Empty<string>() : parameters.ToString().Split(';');
357 if (fields.Length >= 17) {
358 ++ignored[3];
359 return;
360 }
361 int[] raw = fields
362 .Select(f => {
363 int v = 0;
364 foreach (char c in f)
365 v = Math.Min(9999, v * 10 + c - '0');
366 return v;
367 })
368 .ToArray();
369 int[] p = raw.Length == 0 ? new[] { 0 } : raw;
370 int first = p[0], second = p.Length > 1 ? p[1] : 0, n = Math.Max(first, 1);
371 string k = marker + intermediate.ToString() + final;
372 if ((k == "J" && first > 3) || (k == "K" && first > 2) ||
373 (k == "g" && first != 0 && first != 3) || (k == " q" && first > 6) ||
374 (k == "n" && first != 5 && first != 6) || (k == "?n" && first != 6) ||
375 ((k == "c" || k == ">c") && first != 0) || (k == "t" && first != 18)) {
376 ++ignored[0];
377 return;
378 }
379 if (k.Length == 1 && "ABCDEFGHIJKLM PSTXZ`adef@".Contains(final) && final != ' ')
380 pending = false;
381 bool supported = true;
382 switch (k) {
383 case "A":
384 case "F":
385 row = Math.Max(row - n, row >= top ? top : 1);
386 if (k == "F")
387 col = 1;
388 break;
389 case "B":
390 case "e":
391 case "E":
392 row = Math.Min(row + n, row <= bottom ? bottom : height);
393 if (k == "E")
394 col = 1;
395 break;
396 case "C":
397 case "a":
398 col = Math.Min(col + n, width);
399 break;
400 case "D":
401 col = Math.Max(col - n, 1);
402 break;
403 case "G":
404 case "`":
405 col = Math.Min(n, width);
406 break;
407 case "H":
408 case "f":
409 row = Address(n);
410 col = Math.Min(Math.Max(second, 1), width);
411 break;
412 case "d":
413 row = Address(n);
414 break;
415 case "I":
416 case "Z":
417 for (int i = 0; i < n; ++i)
418 Tab(k == "Z");
419 break;
420 case "J":
421 if (first == 3)
422 scrollback.Clear();
423 else
424 for (int r = 1; r <= height; ++r) {
425 if (first == 2 || (first == 0 && r > row) || (first == 1 && r < row))
426 Erase(r, 1, width);
427 else if (r == row)
428 Erase(r, first == 0 ? col : 1, first == 1 ? col : width);
429 }
430 break;
431 case "K":
432 Erase(row, first == 0 ? col : 1, first == 1 ? col : width);
433 break;
434 case "X":
435 Erase(row, col, Math.Min(col + n - 1, width));
436 break;
437 case "@":
438 case "P":
439 int amount = Math.Min(n, width - col + 1);
440 if (k == "@") {
441 for (int c = width - 1; c >= col - 1 + amount; --c)
442 Screen[row - 1][c] = Screen[row - 1][c - amount];
443 Erase(row, col, col + amount - 1);
444 } else {
445 for (int c = col - 1; c < width - amount; ++c)
446 Screen[row - 1][c] = Screen[row - 1][c + amount];
447 Erase(row, width - amount + 1, width);
448 }
449 break;
450 case "L":
451 case "M":
452 if (row >= top && row <= bottom) {
453 int lines = Math.Min(n, bottom - row + 1);
454 if (k == "L") {
455 for (int r = bottom; r >= row + lines; --r)
456 Screen[r - 1] = Screen[r - lines - 1];
457 for (int r = row; r < row + lines; ++r)
458 Screen[r - 1] = BlankRow();
459 } else {
460 for (int r = row; r <= bottom - lines; ++r)
461 Screen[r - 1] = Screen[r + lines - 1];
462 for (int r = bottom - lines + 1; r <= bottom; ++r)
463 Screen[r - 1] = BlankRow();
464 }
465 col = 1;
466 }
467 break;
468 case "S":
469 case "T":
470 Scroll(k == "S", Math.Min(n, bottom - top + 1));
471 break;
472 case "b":
473 if (last >= 0)
474 for (int i = 0; i < n; ++i)
475 Write((char)last);
476 break;
477 case "g":
478 if (first == 0)
479 tabs.Remove(col);
480 else
481 tabs.Clear();
482 break;
483 case "m":
484 supported = Sgr(p);
485 break;
486 case "h":
487 case "l":
488 case "?h":
489 case "?l":
490 supported = SetModes(p, k[0] == '?', final == 'h');
491 break;
492 case " q":
493 shape = n;
494 break;
495 case "r":
496 int newTop = first == 0 ? 1 : first,
497 newBottom = Math.Min(second == 0 ? height : second, height);
498 if (newTop >= newBottom) {
499 ++ignored[3];
500 return;
501 }
502 pending = false;
503 top = newTop;
504 bottom = newBottom;
505 Home();
506 break;
507 case "s":
508 Save();
509 break;
510 case "u":
511 Restore();
512 break;
513 case "!p":
514 pending = false;
515 modes[IRM] = modes[CKM] = modes[ORIGIN] = modes[KEYPAD] = false;
516 modes[WRAP] = modes[VISIBLE] = true;
517 top = 1;
518 bottom = height;
519 attributes = Default;
520 g0 = g1 = gl = false;
521 saved[screen] = null;
522 break;
523 case "n":
524 Reply(first == 5 ? "^[[0n"
525 : $"^[[{(modes[ORIGIN] ? Math.Max(row-top+1, 1) : row)};{col}R");
526 break;
527 case "?n":
528 Reply($"^[[?{(modes[ORIGIN] ? Math.Max(row-top+1, 1) : row)};{col};1R");
529 break;
530 case "c":
531 Reply("^[[?1;2c");
532 break;
533 case ">c":
534 Reply("^[[>0;95;0c");
535 break;
536 case "t":
537 Reply($"^[[8;{height};{width}t");
538 break;
539 case "$p":
540 case "?$p":
541 bool dec = k[0] == '?';
542 int index = ModeIndex(first, dec);
543 int status = index >= 0 ? (modes[index] ? 1 : 2)
544 : dec && (first == 47 || first == 1047 || first == 1049)
545 ? (screen == 1 ? 1 : 2)
546 : 0;
547 Reply($"^[[{(dec ? "?" : "")}{first};{status}$y");
548 break;
549 default:
550 supported = false;
551 break;
552 }
553 if (supported)
554 ++sequences;
555 else
556 ++ignored[0];
557 }
558 public void Feed(int b) {
559 ++bytes;
560 if (b == 27) {
561 if (state == ParseState.Osc)
562 Osc();
563 else if (state != ParseState.Ground)
564 ++ignored[2];
565 marker = "";
566 intermediate.Clear();
567 parameters.Clear();
568 osc.Clear();
569 state = ParseState.Escape;
570 return;
571 }
572 if (b == 24 || b == 26) {
573 if (state != ParseState.Ground)
574 ++ignored[2];
575 state = ParseState.Ground;
576 return;
577 }
578 if (b < 32) {
579 if (state == ParseState.Osc) {
580 if (b == 7) {
581 Osc();
582 state = ParseState.Ground;
583 }
584 } else
585 Control(b);
586 return;
587 }
588 char c = (char)b;
589 if (state == ParseState.CsiEntry) {
590 state = ParseState.CsiParam;
591 if (b >= 60 && b <= 63) {
592 marker = c.ToString();
593 return;
594 }
595 }
596 switch (state) {
597 case ParseState.Ground:
598 Print(c);
599 break;
600 case ParseState.Osc:
601 osc.Append(c);
602 break;
603 case ParseState.Escape:
604 if (b <= 47) {
605 intermediate.Append(c);
606 state = ParseState.EscapeIntermediate;
607 } else if (c == '[')
608 state = ParseState.CsiEntry;
609 else if (c == ']')
610 state = ParseState.Osc;
611 else {
612 Escape(c);
613 state = ParseState.Ground;
614 }
615 break;
616 case ParseState.EscapeIntermediate:
617 if (b <= 47)
618 intermediate.Append(c);
619 else {
620 Escape(c);
621 state = ParseState.Ground;
622 }
623 break;
624 case ParseState.CsiParam:
625 if ((c >= '0' && c <= '9') || c == ';')
626 parameters.Append(c);
627 else if (b <= 47) {
628 intermediate.Append(c);
629 state = ParseState.CsiIntermediate;
630 } else if (b < 64)
631 state = ParseState.CsiIgnore;
632 else {
633 Csi(c);
634 state = ParseState.Ground;
635 }
636 break;
637 case ParseState.CsiIntermediate:
638 if (b <= 47)
639 intermediate.Append(c);
640 else if (b < 64)
641 state = ParseState.CsiIgnore;
642 else {
643 Csi(c);
644 state = ParseState.Ground;
645 }
646 break;
647 case ParseState.CsiIgnore:
648 if (b >= 64) {
649 ++ignored[1];
650 state = ParseState.Ground;
651 }
652 break;
653 }

661 string Key(string name) {
662 string[] nav = { "UP", "DOWN", "RIGHT", "LEFT", "HOME", "END" };
663 int i = Array.IndexOf(nav, name);
664 if (i >= 0)
665 return (modes[CKM] ? "^[O" : "^[[") + "ABCDHF"[i];
666 if (name.StartsWith("F")) {
667 int f = int.Parse(name[1..]);
668 int[] codes = { 15, 17, 18, 19, 20, 21, 23, 24 };
669 return f <= 4 ? "^[O" + (char)('P' + f - 1) : $"^[[{codes[f-5]}~";
670 }
671 if (name.StartsWith("KP") && name.Length == 3)
672 return modes[KEYPAD] ? "^[O" + (char)('p' + name[2] - '0') : name[2].ToString();
673 return name switch { "INSERT" => "^[[2~",
674 "DELETE" => "^[[3~",
675 "PAGEUP" => "^[[5~",
676 "PAGEDOWN" => "^[[6~",
677 "TAB" => "^I",
678 "BACKSPACE" => "^?",
679 "ESCAPE" => "^[",
680 "KPENTER" when modes[KEYPAD] => "^[OM",
681 _ => modes[LNM] ? "^M^J" : "^M" };
682 }

694 static string AttrText(Attributes a) {
695 List<string> parts = new();
696 for (int i = 0; i < 8; ++i)
697 if ((a.Flags & (1 << i)) != 0)
698 parts.Add(FlagNames[i]);
699 if (a.Fg != "default")
700 parts.Add("fg=" + a.Fg);
701 if (a.Bg != "default")
702 parts.Add("bg=" + a.Bg);
703 return string.Join(" ", parts);
704 }
705 public void Finish() {
706 if (state != ParseState.Ground)
707 ++ignored[2];
708 Dump("FINAL");
709 Console.WriteLine(title == null ? "TITLE none" : $"TITLE \"{title}\"");
710 List<string> changed = new();
711 for (int i = 0; i < 9; ++i)
712 if (modes[i] != (i == WRAP || i == VISIBLE))
713 changed.Add(ModeNames[i] + (modes[i] ? "=on" : "=off"));
714 if (g0)
715 changed.Add("G0=graphics");
716 if (g1)
717 changed.Add("G1=graphics");
718 if (gl)
719 changed.Add("GL=G1");
720 if (shape != 1)
721 changed.Add("CURSOR=" + shape);
722 Console.WriteLine("MODES " + (changed.Count == 0 ? "default" : string.Join(" ", changed)));
723 Console.WriteLine($"REGION {top}-{bottom}");
724 Console.WriteLine("TABS " + (tabs.Count == 0 ? "none" : string.Join(",", tabs)));
725 Console.WriteLine("SCROLLBACK " + scrollback.Count);
726 foreach (string s in scrollback.Skip(Math.Max(0, scrollback.Count - 5)))
727 Console.WriteLine("|" + s);
728 List<string> runs = new();
729 for (int r = 0; r < height; ++r)
730 for (int c = 0; c < width;) {
731 Attributes a = Screen[r][c].Attributes;
732 int end = c + 1;
733 while (end < width && Screen[r][end].Attributes == a)
734 ++end;
735 if (a != Default)
736 runs.Add($"row {r+1} col {c+1}-{end} {AttrText(a)}");
737 c = end;
738 }
739 Console.WriteLine("ATTRS " + runs.Count);
740 foreach (string s in runs)
