147std::string join(const std::vector<std::string> &items, const std::string &separator) {
148 std::string result;
149 for (std::size_t index = 0; index < items.size(); ++index) {
150 if (index > 0) {
151 result += separator;
152 }
153 result += items[index];
154 }
155 return result;
156}

158std::string decodeStream(const std::string &line) {
159 std::string data;
160 for (std::size_t index = 0; index < line.size(); ++index) {
161 if (line[index] == '^') {
162 data += CARET_BYTES.at(line[++index]);
163 } else {
164 data += line[index];
165 }
166 }
167 return data;
168}

170Params parseParameters(const std::string &text) {
171 Params params;
172 if (text.empty()) {
173 return params;
174 }
175 params.push_back(0);
176 for (char ch : text) {
177 if (ch == ';') {
178 params.push_back(0);
179 } else {
180 params.back() = std::min(params.back() * 10 + (ch - '0'), MAX_PARAMETER);
181 }
182 }
183 return params;
184}

208std::pair<std::string, std::size_t> extendedColor(const Params &values, std::size_t start) {
209 std::size_t rest = values.size() - start;
210 if (rest >= 2 && values[start] == 5 && values[start + 1] <= 255) {
211 return {paletteColor(values[start + 1]), 2};
212 }
213 if (rest >= 4 && values[start] == 2 &&
214 std::max({values[start + 1], values[start + 2], values[start + 3]}) <= 255) {

217 throw SequenceIgnored{UNSUPPORTED};

222struct Attributes {
223 unsigned flags = 0;
224 std::optional<std::string> foreground;
225 std::optional<std::string> background;
226
227 bool operator==(const Attributes &other) const {
228 return flags == other.flags && foreground == other.foreground &&
229 background == other.background;
230 }
231
232 bool isDefault() const { return flags == 0 && !foreground && !background; }
233
234 std::string describe() const {
235 std::vector<std::string> parts;
236 for (std::size_t index = 0; index < FLAG_NAMES.size(); ++index) {
237 if (flags >> index & 1) {
238 parts.push_back(FLAG_NAMES[index]);
239 }
240 }
241 if (foreground) {
242 parts.push_back("fg=" + *foreground);
243 }
244 if (background) {
245 parts.push_back("bg=" + *background);
246 }
247 return join(parts, " ");
248 }

333 void receive(char byte) {
334 ++streamBytes;
335 if (byte < 0x20) {
336 receiveControl(byte);
337 } else if (state == State::Ground) {
338 printChar(byte);
339 } else if (state == State::Escape || state == State::EscapeIntermediate) {
340 escapeByte(byte);
341 } else if (state == State::OscString) {
342 osc += byte;
343 } else if (state == State::CsiIgnore) {
344 if (isFinal(byte)) {
345 ++ignored[MALFORMED];
346 state = State::Ground;
347 }
348 } else if (state == State::CsiIntermediate) {
349 csiIntermediateByte(byte);
350 } else if (state == State::CsiEntry && isPrivateMarker(byte)) {
351 marker = std::string(1, byte);
352 state = State::CsiParam;
353 } else {
354 csiParamByte(byte);
355 }
356 }

361 std::string sent;

363 sent += ch == '^' ? "^^" : std::string(1, ch);

376 void finish() {
377 if (state != State::Ground) {
378 ++ignored[CANCELED];
379 }
380 screenOutput("FINAL");
381 emit("TITLE " + (title ? "\"" + *title + "\"" : std::string("none")));
382 std::vector<std::string> changed = changedModes();
383 emit("MODES " + (changed.empty() ? std::string("default") : join(changed, " ")));
384 emit("REGION " + std::to_string(top) + "-" + std::to_string(bottom));
385 std::vector<std::string> stops;
386 for (int stop : tabStops) {
387 stops.push_back(std::to_string(stop));
388 }
389 emit("TABS " + (stops.empty() ? std::string("none") : join(stops, ",")));
390 emit("SCROLLBACK " + std::to_string(scrollback.size()));
391 std::size_t shown = std::min(scrollback.size(), SCROLLBACK_SHOWN);
392 for (std::size_t index = scrollback.size() - shown; index < scrollback.size(); ++index) {
393 emit("|" + scrollback[index]);
394 }
395 std::vector<std::string> runs = attributeRuns();
396 emit("ATTRS " + std::to_string(runs.size()));
397 for (const std::string &run : runs) {
398 emit(run);
399 }
400 emit("BELL " + std::to_string(bells));
401 emit("REPLIES " + std::to_string(replies));
402 emit("STATS bytes=" + std::to_string(streamBytes) + " printed=" + std::to_string(printed) +
403 " sequences=" + std::to_string(sequences));
404 std::string counts;
405 for (int reason = 0; reason < REASON_COUNT; ++reason) {
406 counts += " " + REASON_NAMES[reason] + "=" + std::to_string(ignored[reason]);
407 }
408 int total = std::accumulate(ignored.begin(), ignored.end(), 0);
409 emit("IGNORED " + std::to_string(total) + counts);
410 }

487 void receiveControl(char byte) {
488 if (byte == ESC) {

499 state = State::Ground;
500 }
501 } else if (state == State::OscString) {

505 }

507 executeControl(byte);

511 void escapeByte(char byte) {
512 if (isIntermediate(byte)) {
513 intermediates += byte;
514 state = State::EscapeIntermediate;
515 } else if (state == State::Escape && byte == '[') {
516 state = State::CsiEntry;
517 } else if (state == State::Escape && byte == ']') {
518 state = State::OscString;
519 } else {
520 executeEsc(intermediates + byte);

522 }

551 void executeControl(char byte) {
552 switch (byte) {
553 case BEL:
554 ++bells;
555 break;
556 case BS:
557 pendingWrap = false;
558 column = std::max(column - 1, 1);
559 break;
560 case HT:
561 pendingWrap = false;
562 tabMove(1);
563 break;
564 case LF:
565 case VT:
566 case FF:
567 pendingWrap = false;
568 lineFeed();
569 if (modes[LNM]) {
570 column = 1;
571 }
572 break;
573 case CR:
574 pendingWrap = false;
575 column = 1;
576 break;
577 case SO:
578 gl = 1;
579 break;
580 case SI:
581 gl = 0;
582 break;
583 }
584 }

588 if (charset == Charset::LineDrawing) {
589 auto replacement = LINE_DRAWING_CHARS.find(ch);
590 if (replacement != LINE_DRAWING_CHARS.end()) {
591 ch = replacement->second;
592 }
593 }
594 write(ch);
595 }

608 screen().put(row, column, Cell{ch, attributes});
609 if (column < width) {

611 } else if (modes[DECAWM]) {
612 pendingWrap = true;
613 }

646 auto later = tabStops.upper_bound(column);

651 void reverseTabMove(int count) {
652 for (; count > 0 && column > 1; --count) {
653 auto notEarlier = tabStops.lower_bound(column);
654 column = notEarlier != tabStops.begin() ? *std::prev(notEarlier) : 1;
655 }

696 void executeEsc(const std::string &key) {
697 if (key == "7") {
698 saveCursor();
699 } else if (key == "8") {
700 restoreCursor();
701 } else if (key == "D") {
702 pendingWrap = false;
703 lineFeed();
704 } else if (key == "E") {
705 pendingWrap = false;
706 column = 1;
707 lineFeed();
708 } else if (key == "M") {
709 pendingWrap = false;
710 reverseLineFeed();
711 } else if (key == "H") {
712 tabStops.insert(column);
713 } else if (key == "c") {
714 reset();
715 } else if (key == "=" || key == ">") {
716 modes[DECKPAM] = key == "=";
717 } else if (key == "\\") {
718 } else if (key == "(0" || key == "(B") {
719 g0 = key[1] == '0' ? Charset::LineDrawing : Charset::Ascii;
720 } else if (key == ")0" || key == ")B") {
721 g1 = key[1] == '0' ? Charset::LineDrawing : Charset::Ascii;
722 } else if (key == "#8") {
723 screenAlignment();
724 } else {
725 ++ignored[UNSUPPORTED];
726 return;
727 }
728 ++sequences;
729 }

733 title = osc.substr(2);

736 ++ignored[UNSUPPORTED];
737 }

740 void executeCsi(char finalByte) {
741 std::string key = marker + intermediates + finalByte;
742 Params params = parseParameters(parameters);
743 try {
744 require(params.size() <= MAX_FIELDS, INVALID);
745 auto handler = CSI_HANDLERS.find(key);
746 require(handler != CSI_HANDLERS.end());
747 auto condition = CSI_CONDITIONS.find(key);
748 if (condition != CSI_CONDITIONS.end()) {
749 require(condition->second.count(parameter(params, 0)) > 0);
750 }
751 if (key.size() == 1 && WRAP_CLEARING_CSI.find(key) != std::string::npos) {
752 pendingWrap = false;
753 }
754 handler->second(*this, params);
755 } catch (const SequenceIgnored &ignoredSequence) {
756 ++ignored[ignoredSequence.reason];
757 return;
758 }
759 ++sequences;
760 }

917 if (value == 38) {
918 updated.foreground = color;

920 updated.background = color;
