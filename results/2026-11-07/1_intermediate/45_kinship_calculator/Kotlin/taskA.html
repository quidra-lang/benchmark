31private data class Fraction private constructor(val numerator: Long, val denominator: Long) {
32 operator fun plus(other: Fraction): Fraction {
33 val common = maxOf(denominator, other.denominator)
34 return of(numerator * (common / denominator) +
35 other.numerator * (common / other.denominator), common)
36 }
37
38 fun half() = of(numerator, denominator * 2)
39 override fun toString() = "$numerator/$denominator"
40
41 companion object {
42 val ZERO = Fraction(0, 1)
43 val ONE = Fraction(1, 1)
44 val HALF = Fraction(1, 2)
45
46 fun of(numerator: Long, denominator: Long): Fraction {
47 var a = numerator
48 var b = denominator
49 while (b != 0L) {
50 val remainder = a % b
51 a = b
52 b = remainder
53 }
54 return Fraction(numerator / a, denominator / a)
55 }
56 }
57}

79private fun ordinal(number: Int): String {
80 val words = listOf("first", "second", "third", "fourth", "fifth", "sixth",
81 "seventh", "eighth", "ninth", "tenth")
82 if (number <= 10) return words[number - 1]
83 val suffix = if (number % 100 in 11..13) "th" else when (number % 10) {
84 1 -> "st"
85 2 -> "nd"
86 3 -> "rd"
87 else -> "th"
88 }
89 return "$number$suffix"
90}

92private fun term(da: Int, db: Int, sex: String): String = when {
93 da == 0 && db == 0 -> "self"
94 da == 0 -> generationPrefix(db) + sexWord(sex, "father", "mother", "parent")
95 db == 0 -> generationPrefix(da) + sexWord(sex, "son", "daughter", "child")
96 da == 1 && db == 1 -> sexWord(sex, "brother", "sister", "sibling")
97 da == 1 -> generationPrefix(db - 1) + sexWord(sex, "uncle", "aunt", "pibling")
98 db == 1 -> generationPrefix(da - 1) + sexWord(sex, "nephew", "niece", "nibling")
99 else -> {
100 val removal = when (val distance = abs(da - db)) {
101 0 -> ""
102 1 -> " once removed"
103 2 -> " twice removed"
104 else -> " $distance times removed"
105 }
106 ordinal(min(da, db) - 1) + " cousin" + removal
107 }
108}

143 private fun currentMarriage(id: String) = marriages.firstOrNull {

149 private fun partners(id: String): Set<String> {
150 val latest = mutableMapOf<String, Marriage>()
151 for (marriage in marriages) {
152 if (marriage.contains(id)) latest[marriage.other(id)] = marriage
153 }
154 return latest.filterValues { it.status != "divorced" }.keys

157 private fun siblings(a: String, b: String) =
158 a != b && person(a).parents.any { it in person(b).parents }
159
160 private fun ancestors(id: String): Map<String, Int> = ancestorCache.getOrPut(id) {
161 val distance = mutableMapOf(id to 0)
162 val queue = ArrayDeque<String>()
163 queue.add(id)
164 while (queue.isNotEmpty()) {
165 val current = queue.removeFirst()
166 for (parent in person(current).parents) {
167 if (parent !in distance) {
168 distance[parent] = distance.getValue(current) + 1
169 queue.add(parent)
170 }
171 }
172 }
173 distance
174 }
175
176 private fun descendants(id: String): Map<String, Int> {
177 val distance = mutableMapOf(id to 0)
178 val queue = ArrayDeque<String>()
179 queue.add(id)
180 while (queue.isNotEmpty()) {
181 val current = queue.removeFirst()
182 for (child in children(current)) {
183 if (child !in distance) {
184 distance[child] = distance.getValue(current) + 1
185 queue.add(child)
186 }
187 }
188 }
189 distance.remove(id)
190 return distance
191 }

193 private fun apex(a: String, b: String): Apex? {
194 val fromA = ancestors(a)
195 val fromB = ancestors(b)
196 if (a in fromB) return Apex(a, 0, fromB.getValue(a))
197 if (b in fromA) return Apex(b, fromA.getValue(b), 0)
198 val common = fromA.keys.intersect(fromB.keys)
199 val chosen = common.minWithOrNull(compareBy<String> {
200 fromA.getValue(it) + fromB.getValue(it)
201 }.thenBy { fromA.getValue(it) }.thenBy { it }) ?: return null
202 val da = fromA.getValue(chosen)
203 val db = fromB.getValue(chosen)
204 val shares = common.count { fromA.getValue(it) == da && fromB.getValue(it) == db }
205 return Apex(chosen, da, db, shares, true)
206 }

208 private fun bloodTerm(a: String, b: String, sex: String = person(a).sex): String {
209 val top = apex(a, b)!!
210 val prefix = when {
211 !top.collateral -> ""
212 top.shares == 1 -> "half-"
213 top.shares >= 4 -> "double "
214 else -> ""
215 }
216 return prefix + term(top.da, top.db, sex)
217 }

219 private fun relationships(a: String, b: String, sex: String = person(a).sex): List<String> {
220 if (a == b) return listOf("self")
221 val result = mutableListOf<String>()
222 val parentsA = person(a).parents
223 val parentsB = person(b).parents
224 val partnersA = partners(a)
225 val partnersB = partners(b)
226 val marriage = lastMarriage(a, b)
227 if (apex(a, b) != null) result.add(bloodTerm(a, b, sex))
228 when (marriage?.status) {
229 "current" -> result.add(sexWord(sex, "husband", "wife", "spouse"))
230 "widowed" -> result.add(if (marriage.died == a)
231 sexWord(sex, "late husband", "late wife", "late spouse")
232 else sexWord(sex, "widower", "widow", "surviving spouse"))
233 "divorced" -> result.add(sexWord(sex, "ex-husband", "ex-wife", "ex-spouse"))
234 }
235 if (marriage == null && children(a).any { b in person(it).parents }) {
236 result.add("co-parent")
237 }
238 if (a !in parentsB && parentsB.any { it in partnersA }) {
239 result.add(sexWord(sex, "stepfather", "stepmother", "stepparent"))
240 }
241 if (b !in parentsA && parentsA.any { it in partnersB }) {
242 result.add(sexWord(sex, "stepson", "stepdaughter", "stepchild"))
243 }
244 if (parentsA.none { it in parentsB } &&
245 parentsA.any { p -> parentsB.any { q -> q in partners(p) } }) {
246 result.add(sexWord(sex, "stepbrother", "stepsister", "stepsibling"))
247 }
248 if (a !in ancestors(b) && ancestors(b).any { (id, g) -> g == 2 && id in partnersA }) {
249 result.add(sexWord(sex, "stepgrandfather", "stepgrandmother", "stepgrandparent"))
250 }
251 if (b !in ancestors(a) && ancestors(a).any { (id, g) -> g == 2 && id in partnersB }) {
252 result.add(sexWord(sex, "stepgrandson", "stepgranddaughter", "stepgrandchild"))
253 }
254 if (partnersB.any { a in person(it).parents }) {
255 result.add(sexWord(sex, "father-in-law", "mother-in-law", "parent-in-law"))
256 }
257 if (partnersA.any { b in person(it).parents }) {
258 result.add(sexWord(sex, "son-in-law", "daughter-in-law", "child-in-law"))
259 }
260 if (partnersB.any { siblings(a, it) } || partnersA.any { siblings(it, b) }) {
261 result.add(sexWord(sex, "brother-in-law", "sister-in-law", "sibling-in-law"))
262 }
263 return result
264 }
265
266 private fun relationshipString(a: String, b: String) =
267 relationships(a, b).joinToString(", ").ifEmpty { "no relation" }
268
269 private fun kinship(a: String, b: String): Fraction {
270 val key = if (a <= b) a to b else b to a
271 return kinshipCache.getOrPut(key) {
272 if (a == b) {
273 val parents = person(a).parents.toList()
274 if (parents.size == 2) (Fraction.ONE + kinship(parents[0], parents[1])).half()
275 else Fraction.HALF
276 } else {
277 val x = if (person(a).birth >= person(b).birth) a else b
278 val y = if (x == a) b else a
279 person(x).parents.fold(Fraction.ZERO) { total, parent ->
280 total + kinship(parent, y)
281 }.half()
282 }
283 }
284 }
285
286 private fun inbreeding(id: String): Fraction {
287 val parents = person(id).parents.toList()
288 return if (parents.size == 2) kinship(parents[0], parents[1]) else Fraction.ZERO
289 }

291 private fun upwardPath(id: String, top: String): List<String> {
292 val result = mutableListOf(id)
293 var current = id
294 while (current != top) {
295 val remaining = ancestors(current).getValue(top)
296 current = person(current).parents.sorted().first {
297 ancestors(it)[top] == remaining - 1
298 }
299 result.add(current)
300 }
301 return result
302 }

304 fun execute(line: String): List<String> {
305 ancestorCache.clear()
306 kinshipCache.clear()
307 val words = line.split(' ')
308 val command = words[0]
309 val twoIds = command in setOf("PARENT", "UNPARENT", "SPOUSE", "DIVORCE",
310 "RELATION", "PATH", "COMMON", "KINSHIP")
311 val oneId = command in setOf("DEATH", "REMOVE", "INFO", "RELATIVES", "WHO",
312 "ANCESTORS", "DESCENDANTS", "COUSINS", "SIBLINGS", "GENERATIONS",
313 "TREE", "PEDIGREE", "TIMELINE", "FIND")
314 val checkedIds = if (twoIds) 2 else if (oneId) 1 else 0
315 for (index in 1..checkedIds) {
316 if (words[index] !in people) throw CommandError("UNKNOWN PERSON ${words[index]}")
317 }
318 val modifying = command in setOf("PERSON", "DEATH", "REMOVE", "PARENT",
319 "UNPARENT", "SPOUSE", "DIVORCE")
320 val before = if (modifying) snapshot() else null
321 val output = mutableListOf<String>()
322 when (command) {
323 "PERSON" -> {
324 val id = words[1]
325 if (id in people) throw CommandError("DUPLICATE PERSON $id")
326 people[id] = Person(id, words[2], words[3].toInt(), words[4])
327 output.add("ADDED $id ${words[4]} ${words[2]} ${words[3]}")
328 }
329 "DEATH" -> {
330 val id = words[1]
331 val year = words[2].toInt()
332 val p = person(id)
333 if (p.death != null) throw CommandError("ALREADY DECEASED $id")
334 if (year < p.birth || marriages.any {
335 it.contains(id) && (year < it.start || (it.end != null && year < it.end!!))
336 }) throw CommandError("BAD YEAR $id")
337 val child = children(id).sorted().firstOrNull { person(it).birth > year + 1 }
338 if (child != null) throw CommandError("AFTER DEATH $child $id")
339 val current = currentMarriage(id)
340 p.death = year
341 if (current != null) {
342 current.end = year
343 current.status = "widowed"
344 current.died = id
345 }
346 output.add("DIED $id $year" + (current?.let { " spouse=${it.other(id)}" } ?: ""))
347 }
348 "REMOVE" -> {
349 val id = words[1]
350 for (p in people.values) p.parents.remove(id)
351 marriages.removeAll { it.contains(id) }
352 people.remove(id)
353 output.add("REMOVED $id")
354 }
355 "PARENT" -> {
356 val child = words[1]
357 val parent = words[2]
358 val c = person(child)
359 val p = person(parent)
360 if (child == parent) throw CommandError("SAME PERSON $child")
361 if (parent in c.parents) throw CommandError("ALREADY PARENT $child $parent")
362 if (c.parents.size == 2) throw CommandError("TOO MANY PARENTS $child")
363 if (child in ancestors(parent)) throw CommandError("CYCLE $child $parent")
364 if (p.birth + 12 > c.birth) throw CommandError("AGE GAP $child $parent")
365 if (p.death != null && p.death!! + 1 < c.birth) {
366 throw CommandError("AFTER DEATH $child $parent")
367 }
368 c.parents.add(parent)
369 output.add("LINKED $child $parent ${sexWord(p.sex, "father", "mother", "parent")}")
370 }
371 "UNPARENT" -> {
372 val child = words[1]
373 val parent = words[2]
374 if (parent !in person(child).parents) throw CommandError("NOT PARENT $child $parent")
375 person(child).parents.remove(parent)
376 output.add("UNLINKED $child $parent")
377 }
378 "SPOUSE" -> {
379 val a = words[1]
380 val b = words[2]
381 val year = words[3].toInt()
382 val ids = listOf(a, b)
383 if (a == b) throw CommandError("SAME PERSON $a")
384 for (id in ids) if (person(id).death != null) throw CommandError("DECEASED $id")
385 for (id in ids) if (spouse(id) != null) throw CommandError("ALREADY MARRIED $id")
386 if (a in ancestors(b) || b in ancestors(a) || siblings(a, b)) {
387 throw CommandError("CLOSE KIN $a $b")
388 }
389 for (id in ids) if (year < person(id).birth + 16) throw CommandError("TOO YOUNG $id")
390 for (id in ids) {
391 if (marriages.any { it.contains(id) && it.end != null && it.end!! >= year }) {
392 throw CommandError("BAD YEAR $id")
393 }
394 }
395 marriages.add(Marriage(a, b, year))
396 output.add("MARRIED $a $b $year")
397 }
398 "DIVORCE" -> {
399 val a = words[1]
400 val b = words[2]
401 val year = words[3].toInt()
402 if (spouse(a) != b) throw CommandError("NOT MARRIED $a $b")
403 val marriage = currentMarriage(a)!!
404 if (year <= marriage.start) throw CommandError("BAD YEAR $a")
405 marriage.end = year
406 marriage.status = "divorced"
407 output.add("DIVORCED $a $b $year")
408 }
409 "UNDO" -> {
410 if (history.isEmpty()) throw CommandError("NOTHING TO UNDO")
411 val entry = history.removeAt(history.lastIndex)
412 people = entry.state.people
413 marriages = entry.state.marriages
414 output.add("UNDONE ${entry.line}")
415 }
416 "INFO" -> {
417 val id = words[1]
418 val p = person(id)
419 output.add("INFO $id ${p.name} ${p.sex} ${p.lifeYears()}")
420 output.add(" parents: ${personList(p.parents)}")
421 for (marriage in marriages.filter { it.contains(id) }) {
422 output.add(" marriage: ${marriage.other(id)} ${marriage.start}-${marriage.end ?: ""} ${marriage.status}")
423 }
424 output.add(" children: ${personList(children(id))}")
425 output.add(" siblings: ${personList(people.keys.filter { siblings(id, it) })}")
426 output.add(" inbreeding: ${inbreeding(id)}")
427 }
428 "RELATION" -> output.add("RELATION ${words[1]} ${words[2]}: ${relationshipString(words[1], words[2])}")
429 "RELATIVES" -> {
430 val id = words[1]
431 val rows = people.keys.sorted().filter { it != id }.mapNotNull { other ->
432 val terms = relationships(other, id)
433 if (terms.isEmpty()) null else " $other: ${terms.joinToString(", ")}"
434 }
435 output.add("RELATIVES $id ${rows.size}")
436 output.addAll(rows)
437 }
438 "WHO" -> {
439 val id = words[1]
440 val query = words.drop(2).joinToString(" ")
441 val rows = people.keys.sorted().filter { it != id }.mapNotNull { other ->
442 val actual = relationships(other, id)
443 val neutral = relationships(other, id, "X")
444 val index = actual.indices.firstOrNull { actual[it] == query || neutral[it] == query }
445 index?.let { " $other: ${actual[it]}" }
446 }
447 output.add("WHO $id $query ${rows.size}")
448 output.addAll(rows)
449 }
450 "PATH" -> {
451 val a = words[1]
452 val b = words[2]
453 val top = apex(a, b)
454 if (top == null) output.add("PATH $a $b none") else {
455 val path = upwardPath(a, top.id) + upwardPath(b, top.id).dropLast(1).reversed()
456 output.add("PATH $a $b ${top.da + top.db}: ${path.joinToString(" ")}")
457 }
458 }
459 "COMMON" -> {
460 val a = words[1]
461 val b = words[2]
462 val fromA = ancestors(a)
463 val fromB = ancestors(b)
464 val common = fromA.keys.intersect(fromB.keys)
465 val nearest = common.filter { c -> children(c).none { it in common } }.sortedWith(
466 compareBy<String> { fromA.getValue(it) + fromB.getValue(it) }.thenBy { it }
467 )
468 output.add("COMMON $a $b ${nearest.size}")
469 for (id in nearest) output.add(" $id ${fromA.getValue(id)} ${fromB.getValue(id)}")
470 }
471 "ANCESTORS", "DESCENDANTS" -> {
472 val id = words[1]
473 val upward = command == "ANCESTORS"
474 val distances = if (upward) ancestors(id).filterKeys { it != id } else descendants(id)
475 val ids = distances.keys.sortedWith(compareBy<String> { distances.getValue(it) }.thenBy { it })
476 output.add("$command $id ${ids.size}")
477 for (other in ids) {
478 val generation = distances.getValue(other)
479 val label = if (upward) term(0, generation, person(other).sex)
480 else term(generation, 0, person(other).sex)
481 output.add(" $generation $other $label")
482 }
483 }
484 "COUSINS" -> {
485 val id = words[1]
486 val degree = words[2].toInt()
487 val matches = people.keys.filter { it != id }.mapNotNull { other ->
488 val top = apex(other, id)
489 if (top != null && top.da >= 2 && top.db >= 2 && min(top.da, top.db) - 1 == degree)
490 other to abs(top.da - top.db) else null
491 }.sortedWith(compareBy<Pair<String, Int>> { it.second }.thenBy { it.first })
492 output.add("COUSINS $id $degree ${matches.size}")
493 for ((other, _) in matches) output.add(" $other ${bloodTerm(other, id)}")
494 }
495 "SIBLINGS" -> {
496 val id = words[1]
497 val classes = mutableMapOf<String, String>()
498 for (other in people.keys.filter { it != id }) {
499 val shared = person(id).parents.intersect(person(other).parents).size
500 val kind = when {
501 shared == 2 -> "full"
502 shared == 1 -> "half"
503 person(id).parents.any { p -> person(other).parents.any { it in partners(p) } } -> "step"
504 else -> null
505 }
506 if (kind != null) classes[other] = kind
507 }
508 val ordered = ageOrder(classes.keys)
509 val earlier = ageOrder(classes.filterValues { it != "step" }.keys + id).indexOf(id)
510 output.add("SIBLINGS $id ${classes.size} order=${earlier + 1}")
511 for (other in ordered) output.add(" $other ${classes.getValue(other)} ${person(other).birth}")
512 }
513 "GENERATIONS" -> {
514 val id = words[1]
515 val groups = descendants(id).entries.groupBy({ it.value }, { it.key }).toSortedMap()
516 output.add("GENERATIONS $id ${groups.size}")
517 for ((generation, ids) in groups) {
518 val births = ids.map { person(it).birth }
519 output.add(" $generation ${ids.size} ${ids.count { person(it).death == null }} ${births.min()}-${births.max()}")
520 }
521 }
522 "TREE" -> {
523 val id = words[1]
524 val visited = mutableSetOf<String>()
525 output.add("TREE $id ${descendants(id).size}")
526 fun visit(current: String, depth: Int) {
527 val indent = " ".repeat(depth + 1)
528 if (!visited.add(current)) {
529 output.add("$indent$current (see above)")
530 return
531 }
532 val p = person(current)
533 output.add("$indent$current ${p.name} ${p.lifeYears()}" +
534 (spouse(current)?.let { " = $it" } ?: ""))
535 for (child in ageOrder(children(current))) visit(child, depth + 1)
536 }
537 visit(id, 0)
538 }
539 "PEDIGREE" -> {
540 val id = words[1]
541 val depth = words[2].toInt()
542 val rows = mutableListOf<String>()
543 val distinct = mutableSetOf<String>()
544 fun enter(current: String, number: Int, generation: Int) {
545 val p = person(current)
546 distinct.add(current)
547 rows.add("${" ".repeat(generation + 1)}$number $current ${p.name} ${p.lifeYears()}")
548 if (generation < depth) {
549 val parents = p.parents.sortedWith(compareBy<String> {
550 when (person(it).sex) { "M" -> 0; "F" -> 1; else -> 2 }
551 }.thenBy { it })
552 for ((index, parent) in parents.withIndex()) enter(parent, 2 * number + index, generation + 1)
553 }
554 }
555 enter(id, 1, 0)
556 output.add("PEDIGREE $id $depth entries=${rows.size} distinct=${distinct.size}")
557 output.addAll(rows)
558 }
559 "TIMELINE" -> {
560 val id = words[1]
561 val p = person(id)
562 val events = mutableListOf(Event(p.birth, 0, "", "born"))
563 for (marriage in marriages.filter { it.contains(id) }) {
564 val other = marriage.other(id)
565 events.add(Event(marriage.start, 1, other, "married $other"))
566 if (marriage.status == "divorced") {
567 events.add(Event(marriage.end!!, 3, other, "divorced $other"))
568 } else if (marriage.status == "widowed" && marriage.died == other) {
569 events.add(Event(marriage.end!!, 4, other, "widowed $other"))
570 }
571 }
572 for (child in children(id)) events.add(Event(person(child).birth, 2, child, "child $child"))
573 fun loss(other: String, kind: String) {
574 val year = person(other).death ?: return
575 if (year >= p.birth && (p.death == null || year <= p.death!!)) {
576 events.add(Event(year, 5, other, "lost $kind $other"))
577 }
578 }
579 for (parent in p.parents) loss(parent, "parent")
580 for (child in children(id)) loss(child, "child")
581 p.death?.let { events.add(Event(it, 6, "", "died")) }
582 output.add("TIMELINE $id ${events.size}")
583 for (event in events.sortedWith(compareBy<Event> { it.year }.thenBy { it.rank }.thenBy { it.counterpart })) {
584 output.add(" ${event.year} ${event.text}")
585 }
586 }
587 "CENSUS" -> {
588 val year = words[1].toInt()
589 val residents = ageOrder(people.keys.filter {
590 val p = person(it)
591 p.birth <= year && (p.death == null || p.death!! > year)
592 })
593 output.add("CENSUS $year ${residents.size}")
594 for (id in residents) {
595 val marriage = marriages.lastOrNull { it.contains(id) && it.start <= year }
596 val status = when {
597 marriage == null -> "single"
598 marriage.end == null || marriage.end!! > year -> "married ${marriage.other(id)}"
599 else -> marriage.status
600 }
601 output.add(" $id ${year - person(id).birth} $status")
602 }
603 }
604 "KINSHIP" -> output.add("KINSHIP ${words[1]} ${words[2]} ${kinship(words[1], words[2])}")
605 "FIND" -> {
606 val id = words[1]
607 val route = words[2]
608 var found = setOf(id)
609 for (step in route.split('.')) {
610 val requiredSex = when (step) {
611 "father", "son", "brother", "husband" -> "M"
612 "mother", "daughter", "sister", "wife" -> "F"
613 else -> null
614 }
615 found = found.flatMap { current ->
616 when (step) {
617 "parent", "father", "mother" -> person(current).parents.toList()
618 "child", "son", "daughter" -> children(current)
619 "sibling", "brother", "sister" -> people.keys.filter { siblings(current, it) }
620 else -> listOfNotNull(spouse(current))
621 }
622 }.filter { requiredSex == null || person(it).sex == requiredSex }.toSet()
623 }
624 found = found - id
625 output.add("FIND $id $route ${found.size}")
626 for (other in found.sorted()) output.add(" $other: ${relationshipString(other, id)}")
627 }
628 "TERM" -> output.add("TERM ${words[1]} ${words[2]} ${words[3]} ${term(words[1].toInt(), words[2].toInt(), words[3])}")
629 "STATS" -> {
630 val lengths = mutableMapOf<String, Int>()
631 fun length(id: String): Int = lengths.getOrPut(id) {
632 1 + (person(id).parents.maxOfOrNull { length(it) } ?: 0)
633 }
634 val generations = people.keys.maxOfOrNull { length(it) } ?: 0
635 output.add("STATS persons=${people.size} living=${people.values.count { it.death == null }} " +
636 "links=${people.values.sumOf { it.parents.size }} couples=${marriages.count { it.status == "current" }} " +
637 "founders=${people.values.count { it.parents.isEmpty() }} generations=$generations")
638 }
639 "AUDIT" -> {
640 val rows = mutableListOf<String>()
641 for (marriage in marriages) {
642 if (apex(marriage.first, marriage.second) != null) {
643 rows.add(" marriage ${marriage.first} ${marriage.second} ${bloodTerm(marriage.first, marriage.second)}")
644 }
645 }
646 for (id in people.keys.sorted()) {
647 val coefficient = inbreeding(id)
648 if (coefficient.numerator != 0L) rows.add(" inbred $id $coefficient")
649 }
650 output.add("AUDIT ${rows.size}")
651 output.addAll(rows)
652 }
653 }
654 if (before != null) history.add(UndoEntry(line, before))
655 return output
656 }
657}
