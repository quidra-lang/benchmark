61def ordinal(number):
62 if number <= len(ORDINALS):
63 return ORDINALS[number - 1]
64 if number % 100 in (11, 12, 13):
65 suffix = "th"
66 elif number % 10 == 1:
67 suffix = "st"
68 elif number % 10 == 2:
69 suffix = "nd"
70 elif number % 10 == 3:
71 suffix = "rd"
72 else:
73 suffix = "th"
74 return f"{number}{suffix}"

77def removal(distance):
78 if distance == 0:
79 return ""
80 if distance == 1:
81 return " once removed"
82 if distance == 2:
83 return " twice removed"
84 return f" {distance} times removed"

87def kinship_term(da, db, sex):
88 if da == 0 and db == 0:
89 return "self"
90 if da == 0:
91 return generation_prefix(db) + gendered("parent", sex)
92 if db == 0:
93 return generation_prefix(da) + gendered("child", sex)
94 if da == 1 and db == 1:
95 return gendered("sibling", sex)
96 if da == 1:
97 return generation_prefix(db - 1) + gendered("pibling", sex)
98 if db == 1:
99 return generation_prefix(da - 1) + gendered("nibling", sex)
100 return ordinal(min(da, db) - 1) + " cousin" + removal(abs(da - db))

246 def generation_distances(self, start, upward):
247 distances = {start: 0}
248 frontier = [start]
249 while frontier:
250 following = []
251 for x in frontier:
252 for y in self.parents(x) if upward else self.children(x):
253 if y not in distances:
254 distances[y] = distances[x] + 1
255 following.append(y)
256 frontier = following
257 return distances

269 def apex(self, a, b):
270 up_a = self.generation_distances(a, True)
271 up_b = self.generation_distances(b, True)
272 if a in up_b:
273 return Apex(a, 0, up_b[a], 0)
274 if b in up_a:
275 return Apex(b, up_a[b], 0, 0)
276 best = min(
277 ((up_a[c] + up_b[c], up_a[c], c) for c in up_a if c in up_b),
278 default=None,
279 )
280 if best is None:
281 return None
282 total, da, c = best
283 db = total - da
284 shares = sum(1 for y in up_a if y in up_b and up_a[y] == da and up_b[y] == db)
285 return Apex(c, da, db, shares)

287 def relationship(self, a, b, sex):
288 if a == b:
289 return ["self"]
290 terms = []
291 apex = self.apex(a, b)
292 if apex is not None:
293 terms.append(apex.blood_term(sex))
294 record = self.last_record(a, b)
295 if record is None:
296 if self.have_common_child(a, b):
297 terms.append("co-parent")
298 elif record.status == "current":
299 terms.append(gendered("spouse", sex))
300 elif record.status == "widowed":
301 terms.append(gendered("late" if record.died == a else "survivor", sex))
302 else:
303 terms.append(gendered("ex", sex))
304 parents_a = self.parents(a)
305 parents_b = self.parents(b)
306 partners_a = self.partners(a)
307 partners_b = self.partners(b)
308 if a not in parents_b and any(p in partners_a for p in parents_b):
309 terms.append(gendered("stepparent", sex))
310 if b not in parents_a and any(p in partners_b for p in parents_a):
311 terms.append(gendered("stepchild", sex))
312 if self.common_parents(a, b) == 0 and self.step_related(parents_a, parents_b):
313 terms.append(gendered("stepsibling", sex))
314 if not self.is_ancestor(a, b) and any(g in partners_a for g in self.grandparents(b)):
315 terms.append(gendered("stepgrandparent", sex))
316 if not self.is_ancestor(b, a) and any(g in partners_b for g in self.grandparents(a)):
317 terms.append(gendered("stepgrandchild", sex))
318 if any(a in self.parents(q) for q in partners_b):
319 terms.append(gendered("parent-in-law", sex))
320 if any(b in self.parents(q) for q in partners_a):
321 terms.append(gendered("child-in-law", sex))
322 sibling_of_partner = any(self.are_siblings(a, q) for q in partners_b)
323 partner_of_sibling = any(self.are_siblings(q, b) for q in partners_a)
324 if sibling_of_partner or partner_of_sibling:
325 terms.append(gendered("sibling-in-law", sex))
326 return terms

332 def kinship_coefficient(self, a, b, memo):
333 key = (min(a, b), max(a, b))
334 if key in memo:
335 return memo[key]
336 if a == b:
337 value = (SCALE + self.inbreeding_coefficient(a, memo)) // 2
338 else:
339 x, y = (a, b) if self.persons[a].birth >= self.persons[b].birth else (b, a)
340 value = sum(self.kinship_coefficient(p, y, memo) for p in self.parents(x)) // 2
341 memo[key] = value
342 return value

344 def inbreeding_coefficient(self, x, memo):
345 parents = self.parents(x)
346 if len(parents) < 2:
347 return 0
348 return self.kinship_coefficient(parents[0], parents[1], memo)

350 def lineage_length(self, x, memo):
351 if x not in memo:
352 heights = [self.lineage_length(p, memo) for p in self.parents(x)]
353 memo[x] = 1 + max(heights, default=0)
354 return memo[x]

356 def upward_path(self, start, depths):
357 path = [start]
358 while depths[path[-1]] > 0:
359 current = path[-1]
360 path.append(min(
361 p for p in self.parents(current)
362 if p in depths and depths[p] == depths[current] - 1
363 ))
364 return path

672 groups = {}
673 for g, y in family.lineage(x, False):
674 groups.setdefault(g, []).append(family.persons[y])
675 self.emit(f"GENERATIONS {x} {len(groups)}")
676 for g in sorted(groups):
677 members = groups[g]
678 births = [person.birth for person in members]
679 living = sum(person.death is None for person in members)
680 self.emit(f" {g} {len(members)} {living} {min(births)}-{max(births)}")

682 def visit(self, x, depth, lines, visited):
683 indent = " " * (2 * (depth + 1))
684 if x in visited:
685 lines.append(f"{indent}{x} (see above)")
686 return
687 visited.add(x)
688 family = self.family
689 person = family.persons[x]
690 line = f"{indent}{x} {person.name} {life_years(person)}"
691 spouse = family.spouse(x)
692 if spouse is not None:
693 line += f" = {spouse}"
694 lines.append(line)
695 for child in sorted(family.children(x), key=family.age_key):
696 self.visit(child, depth + 1, lines, visited)

698 def tree(self, arguments):
699 x = arguments[0]
700 family = self.family
701 family.person(x)
702 lines = []
703 self.visit(x, 0, lines, set())
704 self.emit(f"TREE {x} {len(family.lineage(x, False))}")
705 self.output.extend(lines)

707 def enter(self, x, number, generation, depth, lines, entered):
708 family = self.family
709 person = family.persons[x]
710 indent = " " * (2 * (generation + 1))
711 lines.append(f"{indent}{number} {x} {person.name} {life_years(person)}")
712 entered.add(x)
713 if generation < depth:
714 parents = sorted(person.parents, key=lambda p: (SEX_INDEX[family.sex(p)], p))
715 for index, parent in enumerate(parents):
716 self.enter(parent, 2 * number + index, generation + 1, depth, lines, entered)

718 def pedigree(self, arguments):
719 x, depth_text = arguments
720 depth = int(depth_text)
721 self.family.person(x)
722 lines = []
723 entered = set()
724 self.enter(x, 1, 0, depth, lines, entered)
725 self.emit(f"PEDIGREE {x} {depth} entries={len(lines)} distinct={len(entered)}")
726 self.output.extend(lines)

728 def timeline(self, arguments):
729 x = arguments[0]
730 family = self.family
731 person = family.person(x)
732 events = [(person.birth, 0, "", "born")]
733 for record in family.records_of(x):
734 other = record.other(x)
735 events.append((record.start, 1, other, f"married {other}"))
736 if record.status == "divorced":
737 events.append((record.end, 3, other, f"divorced {other}"))
738 elif record.status == "widowed" and record.died == other:
739 events.append((record.end, 4, other, f"widowed {other}"))
740 children = family.children(x)
741 for child in children:
742 events.append((family.persons[child].birth, 2, child, f"child {child}"))
743 for label, relatives in (("parent", person.parents), ("child", children)):
744 for other in relatives:
745 death = family.persons[other].death
746 if death is None or death < person.birth:
747 continue
748 if person.death is None or death <= person.death:
749 events.append((death, 5, other, f"lost {label} {other}"))
750 if person.death is not None:
751 events.append((person.death, 6, "", "died"))
752 events.sort(key=lambda event: event[:3])
753 self.emit(f"TIMELINE {x} {len(events)}")
754 for year, _, _, text in events:
755 self.emit(f" {year} {text}")

757 def census(self, arguments):
758 year = int(arguments[0])
759 family = self.family
760 residents = [
761 x for x, person in family.persons.items()
762 if person.birth <= year and (person.death is None or person.death > year)
763 ]
764 residents.sort(key=family.age_key)
765 self.emit(f"CENSUS {year} {len(residents)}")
766 for x in residents:
767 age = year - family.persons[x].birth
768 self.emit(f" {x} {age} {family.census_status(x, year)}")

778 def find(self, arguments):
779 x, route = arguments
780 family = self.family
781 family.person(x)
782 current = {x}
783 for step in route.split("."):
784 kind, sex = STEPS[step]
785 collected = set()
786 for y in current:
787 for z in family.step_targets(y, kind):
788 if sex is None or family.sex(z) == sex:
789 collected.add(z)
790 current = collected
791 current.discard(x)
792 self.emit(f"FIND {x} {route} {len(current)}")
793 for y in sorted(current):
794 self.emit(f" {y}: {family.relationship_text(y, x)}")

800 def stats(self, arguments):
801 family = self.family
802 persons = family.persons.values()
803 living = sum(person.death is None for person in persons)
804 links = sum(len(person.parents) for person in persons)

809 self.emit(

814 def audit(self, arguments):
815 family = self.family
816 lines = []
817 for record in family.marriages:
818 apex = family.apex(record.first, record.second)
819 if apex is not None:
820 term = apex.blood_term(family.sex(record.first))
821 lines.append(f" marriage {record.first} {record.second} {term}")
822 memo = {}
823 for x in sorted(family.persons):
824 value = family.inbreeding_coefficient(x, memo)
825 if value != 0:
826 lines.append(f" inbred {x} {fraction_text(value)}")
