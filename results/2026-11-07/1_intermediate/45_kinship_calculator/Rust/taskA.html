6#[derive(Clone, Copy, PartialEq, Eq)]
7struct Fraction {
8 numerator: u64,
9 denominator: u64,
10}
11
12impl Fraction {
13 fn new(numerator: u64, denominator: u64) -> Self {
14 let (mut a, mut b) = (numerator, denominator);
15 while b != 0 {
16 (a, b) = (b, a % b);
17 }
18 Self {
19 numerator: numerator / a,
20 denominator: denominator / a,
21 }
22 }
23
24 fn half(self) -> Self {
25 Self::new(self.numerator, self.denominator * 2)
26 }
27}
28
29impl Add for Fraction {
30 type Output = Self;
31
32 fn add(self, other: Self) -> Self {
33 let denominator = self.denominator.max(other.denominator);
34 Self::new(
35 self.numerator * (denominator / self.denominator)
36 + other.numerator * (denominator / other.denominator),
37 denominator,
38 )
39 }
40}
41
42impl fmt::Display for Fraction {
43 fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
44 write!(f, "{}/{}", self.numerator, self.denominator)
45 }
46}

115fn pair(a: &str, b: &str) -> (String, String) {
116 if a <= b { (a.to_owned(), b.to_owned()) } else { (b.to_owned(), a.to_owned()) }

127fn grand_prefix(generation: usize) -> String {
128 if generation == 1 {
129 String::new()
130 } else {
131 "great-".repeat(generation - 2) + "grand"
132 }
133}

135fn ordinal(number: usize) -> String {
136 let names = [
137 "first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth",
138 "ninth", "tenth",
139 ];
140 if number <= 10 {
141 return names[number - 1].to_owned();
142 }
143 let suffix = if (11..=13).contains(&(number % 100)) {
144 "th"
145 } else {
146 match number % 10 {
147 1 => "st",
148 2 => "nd",
149 3 => "rd",
150 _ => "th",
151 }
152 };
153 format!("{number}{suffix}")
154}
155
156fn term(da: usize, db: usize, sex: char) -> String {
157 if da == 0 && db == 0 {
158 return "self".to_owned();
159 }
160 if da == 0 {
161 return grand_prefix(db) + gender(sex, "father", "mother", "parent");
162 }
163 if db == 0 {
164 return grand_prefix(da) + gender(sex, "son", "daughter", "child");
165 }
166 if da == 1 && db == 1 {
167 return gender(sex, "brother", "sister", "sibling").to_owned();
168 }
169 if da == 1 {
170 return grand_prefix(db - 1) + gender(sex, "uncle", "aunt", "pibling");
171 }
172 if db == 1 {
173 return grand_prefix(da - 1) + gender(sex, "nephew", "niece", "nibling");
174 }
175 let ending = match da.abs_diff(db) {
176 0 => String::new(),
177 1 => " once removed".to_owned(),
178 2 => " twice removed".to_owned(),
179 n => format!(" {n} times removed"),
180 };
181 format!("{} cousin{ending}", ordinal(da.min(db) - 1))
182}

191struct Context<'a> {
192 state: &'a State,
193 ancestors: BTreeMap<String, BTreeMap<String, usize>>,
194 children: BTreeMap<String, BTreeSet<String>>,
195 latest: BTreeMap<(String, String), usize>,
196 spouses: BTreeMap<String, String>,
197 partners: BTreeMap<String, BTreeSet<String>>,
198 kinship: HashMap<(String, String), Fraction>,
199}
200
201impl<'a> Context<'a> {
202 fn new(state: &'a State) -> Self {
203 let mut ancestors = BTreeMap::new();
204 let mut children: BTreeMap<String, BTreeSet<String>> = state
205 .people.keys().map(|id| (id.clone(), BTreeSet::new())).collect();
206 for (id, person) in &state.people {
207 for parent in &person.parents {
208 children.get_mut(parent).unwrap().insert(id.clone());
209 }
210 let mut distances = BTreeMap::from([(id.clone(), 0)]);
211 let mut queue = VecDeque::from([id.clone()]);
212 while let Some(current) = queue.pop_front() {
213 let distance = distances[&current] + 1;
214 for parent in &state.people[&current].parents {
215 if !distances.contains_key(parent) {
216 distances.insert(parent.clone(), distance);
217 queue.push_back(parent.clone());
218 }
219 }
220 }
221 ancestors.insert(id.clone(), distances);
222 }
223 let mut latest = BTreeMap::new();
224 let mut spouses = BTreeMap::new();
225 for (i, marriage) in state.marriages.iter().enumerate() {
226 latest.insert(pair(&marriage.first, &marriage.second), i);
227 if marriage.status == Status::Current {
228 spouses.insert(marriage.first.clone(), marriage.second.clone());
229 spouses.insert(marriage.second.clone(), marriage.first.clone());
230 }
231 }
232 let mut partners: BTreeMap<String, BTreeSet<String>> = state
233 .people.keys().map(|id| (id.clone(), BTreeSet::new())).collect();
234 for &i in latest.values() {
235 let marriage = &state.marriages[i];
236 if marriage.status != Status::Divorced {
237 partners.get_mut(&marriage.first).unwrap().insert(marriage.second.clone());
238 partners.get_mut(&marriage.second).unwrap().insert(marriage.first.clone());
239 }
240 }
241 Self { state, ancestors, children, latest, spouses, partners, kinship: HashMap::new() }
242 }

244 fn check(&self, ids: &[&str]) -> Result<(), String> {
245 for &id in ids {
246 if !self.state.people.contains_key(id) {
247 return Err(format!("UNKNOWN PERSON {id}"));
248 }

250 Ok(())
251 }

255 ids

263 fn siblings(&self, a: &str, b: &str) -> bool {
264 a != b && !self.state.people[a].parents.is_disjoint(&self.state.people[b].parents)
265 }

267 fn descendants(&self, id: &str) -> BTreeMap<String, usize> {
268 self.ancestors.iter().filter_map(|(person, distances)| {
269 if person == id { None } else { distances.get(id).map(|&g| (person.clone(), g)) }
270 }).collect()
271 }

273 fn apex(&self, a: &str, b: &str) -> Option<Apex> {
274 let aa = &self.ancestors[a];
275 let ab = &self.ancestors[b];
276 if let Some(&db) = ab.get(a) {
277 return Some(Apex { id: a.to_owned(), da: 0, db, shares: 0 });
278 }
279 if let Some(&da) = aa.get(b) {
280 return Some(Apex { id: b.to_owned(), da, db: 0, shares: 0 });
281 }
282 let (sum, da, id) = aa.iter().filter_map(|(id, &da)| {
283 ab.get(id).map(|&db| (da + db, da, id.clone()))
284 }).min()?;
285 let db = sum - da;
286 let shares = aa.iter().filter(|(id, distance)| **distance == da && ab.get(*id) == Some(&db)).count();
287 Some(Apex { id, da, db, shares })
288 }
289
290 fn blood_term(&self, a: &str, b: &str, sex: char) -> Option<String> {
291 if a == b { return None; }
292 let apex = self.apex(a, b)?;
293 let prefix = if apex.shares == 1 { "half-" } else if apex.shares >= 4 { "double " } else { "" };
294 Some(prefix.to_owned() + &term(apex.da, apex.db, sex))
295 }

297 fn relationship(&self, a: &str, b: &str, sex: char) -> Vec<String> {
298 if a == b { return vec!["self".to_owned()]; }
299 let pa = &self.state.people[a].parents;
300 let pb = &self.state.people[b].parents;
301 let partners_a = &self.partners[a];
302 let partners_b = &self.partners[b];
303 let mut result = Vec::new();
304 if let Some(blood) = self.blood_term(a, b, sex) { result.push(blood); }
305 if let Some(&i) = self.latest.get(&pair(a, b)) {
306 let marriage = &self.state.marriages[i];
307 let name = match marriage.status {
308 Status::Current => gender(sex, "husband", "wife", "spouse"),
309 Status::Divorced => gender(sex, "ex-husband", "ex-wife", "ex-spouse"),
310 Status::Widowed if marriage.died.as_deref() == Some(a) => gender(sex, "late husband", "late wife", "late spouse"),
311 Status::Widowed => gender(sex, "widower", "widow", "surviving spouse"),
312 };
313 result.push(name.to_owned());
314 } else if !self.children[a].is_disjoint(&self.children[b]) {
315 result.push("co-parent".to_owned());
316 }
317 if !pb.contains(a) && !pb.is_disjoint(partners_a) {
318 result.push(gender(sex, "stepfather", "stepmother", "stepparent").to_owned());
319 }
320 if !pa.contains(b) && !pa.is_disjoint(partners_b) {
321 result.push(gender(sex, "stepson", "stepdaughter", "stepchild").to_owned());
322 }
323 if pa.is_disjoint(pb) && pa.iter().any(|p| !self.partners[p].is_disjoint(pb)) {
324 result.push(gender(sex, "stepbrother", "stepsister", "stepsibling").to_owned());
325 }
326 if !self.ancestors[b].contains_key(a)
327 && self.ancestors[b].iter().any(|(g, &distance)| distance == 2 && partners_a.contains(g))
328 {
329 result.push(gender(sex, "stepgrandfather", "stepgrandmother", "stepgrandparent").to_owned());
330 }
331 if !self.ancestors[a].contains_key(b)
332 && self.ancestors[a].iter().any(|(g, &distance)| distance == 2 && partners_b.contains(g))
333 {
334 result.push(gender(sex, "stepgrandson", "stepgranddaughter", "stepgrandchild").to_owned());
335 }
336 if partners_b.iter().any(|p| self.state.people[p].parents.contains(a)) {
337 result.push(gender(sex, "father-in-law", "mother-in-law", "parent-in-law").to_owned());
338 }
339 if partners_a.iter().any(|p| self.state.people[p].parents.contains(b)) {
340 result.push(gender(sex, "son-in-law", "daughter-in-law", "child-in-law").to_owned());
341 }
342 if partners_b.iter().any(|p| self.siblings(a, p)) || partners_a.iter().any(|p| self.siblings(p, b)) {
343 result.push(gender(sex, "brother-in-law", "sister-in-law", "sibling-in-law").to_owned());
344 }
345 result
346 }

348 fn relationship_string(&self, a: &str, b: &str) -> String {
349 let terms = self.relationship(a, b, self.state.people[a].sex);
350 if terms.is_empty() { "no relation".to_owned() } else { terms.join(", ") }
351 }

353 fn phi(&mut self, a: &str, b: &str) -> Fraction {
354 let key = pair(a, b);
355 if let Some(&value) = self.kinship.get(&key) { return value; }
356 let value = if a == b {
357 (Fraction::new(1, 1) + self.inbreeding(a)).half()
358 } else {
359 let (x, y) = if self.state.people[a].birth >= self.state.people[b].birth { (a, b) } else { (b, a) };
360 let parents: Vec<_> = self.state.people[x].parents.iter().cloned().collect();
361 let mut sum = Fraction::new(0, 1);
362 for parent in parents { sum = sum + self.phi(&parent, y); }
363 sum.half()
364 };
365 self.kinship.insert(key, value);
366 value
367 }
368
369 fn inbreeding(&mut self, id: &str) -> Fraction {
370 let parents: Vec<_> = self.state.people[id].parents.iter().cloned().collect();
371 if parents.len() == 2 { self.phi(&parents[0], &parents[1]) } else { Fraction::new(0, 1) }
372 }

374 fn upward_path(&self, from: &str, apex: &str) -> Vec<String> {
375 let mut result = vec![from.to_owned()];
376 while result.last().unwrap() != apex {
377 let current = result.last().unwrap();
378 let distance = self.ancestors[current][apex] - 1;
379 let parent = self.state.people[current].parents.iter().find(|p| self.ancestors[*p].get(apex) == Some(&distance)).unwrap().clone();
380 result.push(parent);
381 }
382 result
383 }

385 fn tree_visit(&self, id: &str, depth: usize, visited: &mut BTreeSet<String>, lines: &mut Vec<String>) {
386 let indent = " ".repeat(2 * (depth + 1));
387 if !visited.insert(id.to_owned()) {
388 lines.push(format!("{indent}{id} (see above)"));
389 return;
390 }
391 let person = &self.state.people[id];
392 let spouse = self.spouses.get(id).map(|s| format!(" = {s}")).unwrap_or_default();
393 lines.push(format!("{indent}{id} {} {}{spouse}", person.name, person.life()));
394 for child in self.age_order(self.children[id].iter().cloned().collect()) {
395 self.tree_visit(&child, depth + 1, visited, lines);
396 }
397 }

399 fn pedigree_enter(&self, id: &str, number: usize, generation: usize, depth: usize,
400 entered: &mut BTreeSet<String>, lines: &mut Vec<String>) {
401 entered.insert(id.to_owned());
402 let person = &self.state.people[id];
403 lines.push(format!("{}{number} {id} {} {}", " ".repeat(2 * (generation + 1)), person.name, person.life()));
404 if generation < depth {
405 let mut parents: Vec<_> = person.parents.iter().cloned().collect();
406 parents.sort_by_key(|p| {
407 let rank = match self.state.people[p].sex { 'M' => 0, 'F' => 1, _ => 2 };
408 (rank, p.clone())
409 });
410 for (i, parent) in parents.iter().enumerate() {
411 self.pedigree_enter(parent, 2 * number + i, generation + 1, depth, entered, lines);
412 }
413 }
414 }
415
416 fn longest_lineage(&self, id: &str, memo: &mut BTreeMap<String, usize>) -> usize {
417 if let Some(&length) = memo.get(id) { return length; }
418 let length = 1 + self.state.people[id].parents.iter().map(|p| self.longest_lineage(p, memo)).max().unwrap_or(0);
419 memo.insert(id.to_owned(), length);
420 length
421 }

424#[derive(Default)]
425struct Family {
426 state: State,
427 history: Vec<UndoRecord>,
428}
429
430impl Family {
431 fn remember(&mut self, command: &str) {
432 self.history.push(UndoRecord { command: command.to_owned(), state: self.state.clone() });
433 }
434
435 fn command(&mut self, line: &str) -> Result<Vec<String>, String> {
436 let words: Vec<_> = line.split(' ').collect();
437 let command = words[0];
438 let mut ctx = Context::new(&self.state);
439 match command {
440 "PERSON" => {
441 let id = words[1];
442 if self.state.people.contains_key(id) { return Err(format!("DUPLICATE PERSON {id}")); }
443 drop(ctx);
444 self.remember(line);
445 self.state.people.insert(id.to_owned(), Person {
446 sex: words[2].chars().next().unwrap(), birth: words[3].parse().unwrap(),
447 name: words[4].to_owned(), death: None, parents: BTreeSet::new(),
448 });
449 Ok(vec![format!("ADDED {id} {} {} {}", words[4], words[2], words[3])])
450 }
451 "DEATH" => {
452 let id = words[1];
453 let year: i32 = words[2].parse().unwrap();
454 ctx.check(&[id])?;
455 let person = &self.state.people[id];
456 if person.death.is_some() { return Err(format!("ALREADY DECEASED {id}")); }
457 if year < person.birth || self.state.marriages.iter().any(|m| m.contains(id) && (m.start > year || m.end.is_some_and(|end| end > year))) {
458 return Err(format!("BAD YEAR {id}"));
459 }
460 if let Some(child) = ctx.children[id].iter().find(|c| self.state.people[*c].birth > year + 1) {
461 return Err(format!("AFTER DEATH {child} {id}"));
462 }
463 let spouse = ctx.spouses.get(id).cloned();
464 let index = spouse.as_ref().map(|s| ctx.latest[&pair(id, s)]);
465 drop(ctx);
466 self.remember(line);
467 self.state.people.get_mut(id).unwrap().death = Some(year);
468 if let Some(i) = index {
469 let marriage = &mut self.state.marriages[i];
470 marriage.end = Some(year);
471 marriage.status = Status::Widowed;
472 marriage.died = Some(id.to_owned());
473 }
474 Ok(vec![format!("DIED {id} {year}{}", spouse.map(|s| format!(" spouse={s}")).unwrap_or_default())])
475 }
476 "REMOVE" => {
477 let id = words[1];
478 ctx.check(&[id])?;
479 drop(ctx);
480 self.remember(line);
481 self.state.people.remove(id);
482 for person in self.state.people.values_mut() { person.parents.remove(id); }
483 self.state.marriages.retain(|m| !m.contains(id));
484 Ok(vec![format!("REMOVED {id}")])
485 }
486 "PARENT" => {
487 let (child, parent) = (words[1], words[2]);
488 ctx.check(&[child, parent])?;
489 if child == parent { return Err(format!("SAME PERSON {child}")); }
490 if self.state.people[child].parents.contains(parent) { return Err(format!("ALREADY PARENT {child} {parent}")); }
491 if self.state.people[child].parents.len() == 2 { return Err(format!("TOO MANY PARENTS {child}")); }
492 if ctx.ancestors[parent].contains_key(child) { return Err(format!("CYCLE {child} {parent}")); }
493 if self.state.people[parent].birth + 12 > self.state.people[child].birth { return Err(format!("AGE GAP {child} {parent}")); }
494 if self.state.people[parent].death.is_some_and(|death| death + 1 < self.state.people[child].birth) {
495 return Err(format!("AFTER DEATH {child} {parent}"));
496 }
497 let title = gender(self.state.people[parent].sex, "father", "mother", "parent");
498 drop(ctx);
499 self.remember(line);
500 self.state.people.get_mut(child).unwrap().parents.insert(parent.to_owned());
501 Ok(vec![format!("LINKED {child} {parent} {title}")])
502 }
503 "UNPARENT" => {
504 let (child, parent) = (words[1], words[2]);
505 ctx.check(&[child, parent])?;
506 if !self.state.people[child].parents.contains(parent) { return Err(format!("NOT PARENT {child} {parent}")); }
507 drop(ctx);
508 self.remember(line);
509 self.state.people.get_mut(child).unwrap().parents.remove(parent);
510 Ok(vec![format!("UNLINKED {child} {parent}")])
511 }
512 "SPOUSE" => {
513 let (a, b) = (words[1], words[2]);
514 let year: i32 = words[3].parse().unwrap();
515 ctx.check(&[a, b])?;
516 if a == b { return Err(format!("SAME PERSON {a}")); }
517 for id in [a, b] {
518 if self.state.people[id].death.is_some() { return Err(format!("DECEASED {id}")); }
519 }
520 for id in [a, b] {
521 if ctx.spouses.contains_key(id) { return Err(format!("ALREADY MARRIED {id}")); }
522 }
523 if ctx.ancestors[a].contains_key(b) || ctx.ancestors[b].contains_key(a) || ctx.siblings(a, b) {
524 return Err(format!("CLOSE KIN {a} {b}"));
525 }
526 for id in [a, b] {
527 if year < self.state.people[id].birth + 16 { return Err(format!("TOO YOUNG {id}")); }
528 }
529 for id in [a, b] {
530 if self.state.marriages.iter().any(|m| m.contains(id) && m.end.is_some_and(|end| end >= year)) {
531 return Err(format!("BAD YEAR {id}"));
532 }
533 }
534 drop(ctx);
535 self.remember(line);
536 self.state.marriages.push(Marriage {
537 first: a.to_owned(), second: b.to_owned(), start: year, end: None,
538 status: Status::Current, died: None,
539 });
540 Ok(vec![format!("MARRIED {a} {b} {year}")])
541 }
542 "DIVORCE" => {
543 let (a, b) = (words[1], words[2]);
544 let year: i32 = words[3].parse().unwrap();
545 ctx.check(&[a, b])?;
546 if ctx.spouses.get(a).map(String::as_str) != Some(b) { return Err(format!("NOT MARRIED {a} {b}")); }
547 let index = ctx.latest[&pair(a, b)];
548 if year <= self.state.marriages[index].start { return Err(format!("BAD YEAR {a}")); }
549 drop(ctx);
550 self.remember(line);
551 self.state.marriages[index].end = Some(year);
552 self.state.marriages[index].status = Status::Divorced;
553 Ok(vec![format!("DIVORCED {a} {b} {year}")])
554 }
555 "UNDO" => {
556 drop(ctx);
557 let previous = self.history.pop().ok_or("NOTHING TO UNDO")?;
558 self.state = previous.state;
559 Ok(vec![format!("UNDONE {}", previous.command)])
560 }
561 "INFO" => {
562 let id = words[1];
563 ctx.check(&[id])?;
564 let person = &self.state.people[id];
565 let mut lines = vec![
566 format!("INFO {id} {} {} {}", person.name, person.sex, person.life()),
567 format!(" parents: {}", ctx.person_list(person.parents.iter().cloned().collect())),
568 ];
569 for marriage in &self.state.marriages {
570 if marriage.contains(id) {
571 lines.push(format!(" marriage: {} {}-{} {}", marriage.other(id), marriage.start,
572 marriage.end.map(|year| year.to_string()).unwrap_or_default(), marriage.status.as_str()));
573 }
574 }
575 lines.push(format!(" children: {}", ctx.person_list(ctx.children[id].iter().cloned().collect())));
576 let siblings = self.state.people.keys().filter(|p| ctx.siblings(p, id)).cloned().collect();
577 lines.push(format!(" siblings: {}", ctx.person_list(siblings)));
578 lines.push(format!(" inbreeding: {}", ctx.inbreeding(id)));
579 Ok(lines)
580 }
581 "RELATION" => {
582 let (a, b) = (words[1], words[2]);
583 ctx.check(&[a, b])?;
584 Ok(vec![format!("RELATION {a} {b}: {}", ctx.relationship_string(a, b))])
585 }
586 "RELATIVES" | "WHO" => {
587 let id = words[1];
588 ctx.check(&[id])?;
589 let query = words[2..].join(" ");
590 let mut lines = Vec::new();
591 for (person, value) in &self.state.people {
592 if person == id { continue; }
593 let terms = ctx.relationship(person, id, value.sex);
594 if command == "RELATIVES" {
595 if !terms.is_empty() { lines.push(format!(" {person}: {}", terms.join(", "))); }
596 } else {
597 let neutral = ctx.relationship(person, id, 'X');
598 if let Some(i) = terms.iter().zip(&neutral).position(|(actual, generic)| *actual == query || *generic == query) {
599 lines.push(format!(" {person}: {}", terms[i]));
600 }
601 }
602 }
603 let header = if command == "WHO" { format!("WHO {id} {query} {}", lines.len()) }
604 else { format!("RELATIVES {id} {}", lines.len()) };
605 lines.insert(0, header);
606 Ok(lines)
607 }
608 "PATH" => {
609 let (a, b) = (words[1], words[2]);
610 ctx.check(&[a, b])?;
611 if let Some(apex) = ctx.apex(a, b) {
612 let mut path = ctx.upward_path(a, &apex.id);
613 let mut right = ctx.upward_path(b, &apex.id);
614 right.pop();
615 path.extend(right.into_iter().rev());
616 Ok(vec![format!("PATH {a} {b} {}: {}", apex.da + apex.db, path.join(" "))])
617 } else {
618 Ok(vec![format!("PATH {a} {b} none")])
619 }
620 }
621 "COMMON" => {
622 let (a, b) = (words[1], words[2]);
623 ctx.check(&[a, b])?;
624 let aa = &ctx.ancestors[a];
625 let ab = &ctx.ancestors[b];
626 let common: BTreeSet<_> = aa.keys().filter(|p| ab.contains_key(*p)).cloned().collect();
627 let mut nearest: Vec<_> = common.iter().filter(|p| ctx.children[*p].is_disjoint(&common)).cloned().collect();
628 nearest.sort_by_key(|p| (aa[p] + ab[p], p.clone()));
629 let mut lines = vec![format!("COMMON {a} {b} {}", nearest.len())];
630 lines.extend(nearest.iter().map(|p| format!(" {p} {} {}", aa[p], ab[p])));
631 Ok(lines)
632 }
633 "ANCESTORS" | "DESCENDANTS" => {
634 let id = words[1];
635 ctx.check(&[id])?;
636 let mut distances = if command == "ANCESTORS" { ctx.ancestors[id].clone() } else { ctx.descendants(id) };
637 distances.remove(id);
638 let mut entries: Vec<_> = distances.into_iter().map(|(person, g)| (g, person)).collect();
639 entries.sort();
640 let mut lines = vec![format!("{command} {id} {}", entries.len())];
641 for (g, person) in entries {
642 let sex = self.state.people[&person].sex;
643 let title = if command == "ANCESTORS" { term(0, g, sex) } else { term(g, 0, sex) };
644 lines.push(format!(" {g} {person} {title}"));
645 }
646 Ok(lines)
647 }
648 "COUSINS" => {
649 let id = words[1];
650 let degree: usize = words[2].parse().unwrap();
651 ctx.check(&[id])?;
652 let mut matches = Vec::new();
653 for person in self.state.people.keys() {
654 if person == id { continue; }
655 if let Some(apex) = ctx.apex(person, id) {
656 if apex.da >= 2 && apex.db >= 2 && apex.da.min(apex.db) - 1 == degree {
657 matches.push((apex.da.abs_diff(apex.db), person.clone()));
658 }
659 }
660 }
661 matches.sort();
662 let mut lines = vec![format!("COUSINS {id} {degree} {}", matches.len())];
663 for (_, person) in matches {
664 lines.push(format!(" {person} {}", ctx.blood_term(&person, id, self.state.people[&person].sex).unwrap()));
665 }
666 Ok(lines)
667 }
668 "SIBLINGS" => {
669 let id = words[1];
670 ctx.check(&[id])?;
671 let parents = &self.state.people[id].parents;
672 let mut matches = BTreeMap::new();
673 let mut order = 1;
674 for (person, value) in &self.state.people {
675 if person == id { continue; }
676 let common = parents.intersection(&value.parents).count();
677 let kind = match common {
678 2 => "full",
679 1 => "half",
680 _ if parents.iter().any(|p| !ctx.partners[p].is_disjoint(&value.parents)) => "step",
681 _ => continue,
682 };
683 matches.insert(person.clone(), kind);
684 if common > 0 && (value.birth, person.as_str()) < (self.state.people[id].birth, id) { order += 1; }
685 }
686 let mut lines = vec![format!("SIBLINGS {id} {} order={order}", matches.len())];
687 for person in ctx.age_order(matches.keys().cloned().collect()) {
688 lines.push(format!(" {person} {} {}", matches[&person], self.state.people[&person].birth));
689 }
690 Ok(lines)
691 }
692 "GENERATIONS" => {
693 let id = words[1];
694 ctx.check(&[id])?;
695 let mut groups: BTreeMap<usize, Vec<&Person>> = BTreeMap::new();
696 for (person, g) in ctx.descendants(id) { groups.entry(g).or_default().push(&self.state.people[&person]); }
697 let mut lines = vec![format!("GENERATIONS {id} {}", groups.len())];
698 for (g, people) in groups {
699 let living = people.iter().filter(|p| p.death.is_none()).count();
700 let low = people.iter().map(|p| p.birth).min().unwrap();
701 let high = people.iter().map(|p| p.birth).max().unwrap();
702 lines.push(format!(" {g} {} {living} {low}-{high}", people.len()));
703 }
704 Ok(lines)
705 }
706 "TREE" => {
707 let id = words[1];
708 ctx.check(&[id])?;
709 let mut lines = vec![format!("TREE {id} {}", ctx.descendants(id).len())];
710 ctx.tree_visit(id, 0, &mut BTreeSet::new(), &mut lines);
711 Ok(lines)
712 }
713 "PEDIGREE" => {
714 let id = words[1];
715 let depth: usize = words[2].parse().unwrap();
716 ctx.check(&[id])?;
717 let mut lines = Vec::new();
718 let mut entered = BTreeSet::new();
719 ctx.pedigree_enter(id, 1, 0, depth, &mut entered, &mut lines);
720 lines.insert(0, format!("PEDIGREE {id} {depth} entries={} distinct={}", lines.len(), entered.len()));
721 Ok(lines)
722 }
723 "TIMELINE" => {
724 let id = words[1];
725 ctx.check(&[id])?;
726 let person = &self.state.people[id];
727 let mut events = vec![(person.birth, 0, String::new(), "born".to_owned())];
728 for marriage in &self.state.marriages {
729 if !marriage.contains(id) { continue; }
730 let other = marriage.other(id);
731 events.push((marriage.start, 1, other.to_owned(), format!("married {other}")));
732 if marriage.status == Status::Divorced {
733 events.push((marriage.end.unwrap(), 3, other.to_owned(), format!("divorced {other}")));
734 }
735 if marriage.status == Status::Widowed && marriage.died.as_deref() == Some(other) {
736 events.push((marriage.end.unwrap(), 4, other.to_owned(), format!("widowed {other}")));
737 }
738 }
739 for child in &ctx.children[id] {
740 events.push((self.state.people[child].birth, 2, child.clone(), format!("child {child}")));
741 }
742 for (relatives, label) in [(&person.parents, "parent"), (&ctx.children[id], "child")] {
743 for other in relatives {
744 if let Some(death) = self.state.people[other].death {
745 if death >= person.birth && person.death.is_none_or(|end| death <= end) {
746 events.push((death, 5, other.clone(), format!("lost {label} {other}")));
747 }
748 }
749 }
750 }
751 if let Some(death) = person.death { events.push((death, 6, String::new(), "died".to_owned())); }
752 events.sort_by(|a, b| (&a.0, &a.1, &a.2).cmp(&(&b.0, &b.1, &b.2)));
753 let mut lines = vec![format!("TIMELINE {id} {}", events.len())];
754 lines.extend(events.into_iter().map(|(year, _, _, text)| format!(" {year} {text}")));
755 Ok(lines)
756 }
757 "CENSUS" => {
758 let year: i32 = words[1].parse().unwrap();
759 let residents = ctx.age_order(self.state.people.iter().filter_map(|(id, person)| {
760 (person.birth <= year && person.death.is_none_or(|death| death > year)).then(|| id.clone())
761 }).collect());
762 let mut lines = vec![format!("CENSUS {year} {}", residents.len())];
763 for id in residents {
764 let status = if let Some(marriage) = self.state.marriages.iter().rev().find(|m| m.contains(&id) && m.start <= year) {
765 if marriage.end.is_none_or(|end| end > year) { format!("married {}", marriage.other(&id)) }
766 else { marriage.status.as_str().to_owned() }
767 } else { "single".to_owned() };
768 lines.push(format!(" {id} {} {status}", year - self.state.people[&id].birth));
769 }
770 Ok(lines)
771 }
772 "KINSHIP" => {
773 let (a, b) = (words[1], words[2]);
774 ctx.check(&[a, b])?;
775 Ok(vec![format!("KINSHIP {a} {b} {}", ctx.phi(a, b))])
776 }
777 "FIND" => {
778 let id = words[1];
779 let route = words[2];
780 ctx.check(&[id])?;
781 let mut found = BTreeSet::from([id.to_owned()]);
782 for step in route.split('.') {
783 let sex = match step {
784 "father" | "son" | "brother" | "husband" => Some('M'),
785 "mother" | "daughter" | "sister" | "wife" => Some('F'),
786 _ => None,
787 };
788 let mut next = BTreeSet::new();
789 for person in found {
790 let candidates: Vec<String> = match step {
791 "parent" | "father" | "mother" => self.state.people[&person].parents.iter().cloned().collect(),
792 "child" | "son" | "daughter" => ctx.children[&person].iter().cloned().collect(),
793 "sibling" | "brother" | "sister" => self.state.people.keys().filter(|p| ctx.siblings(&person, p)).cloned().collect(),
794 _ => ctx.spouses.get(&person).cloned().into_iter().collect(),
795 };
796 for candidate in candidates {
797 if sex.is_none_or(|s| self.state.people[&candidate].sex == s) { next.insert(candidate); }
798 }
799 }
800 found = next;
801 }
802 found.remove(id);
803 let mut lines = vec![format!("FIND {id} {route} {}", found.len())];
804 lines.extend(found.into_iter().map(|person| format!(" {person}: {}", ctx.relationship_string(&person, id))));
805 Ok(lines)
806 }
807 "TERM" => {
808 let da: usize = words[1].parse().unwrap();
809 let db: usize = words[2].parse().unwrap();
810 let sex = words[3].chars().next().unwrap();
811 Ok(vec![format!("TERM {da} {db} {sex} {}", term(da, db, sex))])
812 }
813 "STATS" => {
814 let mut memo = BTreeMap::new();
815 let generations = self.state.people.keys().map(|id| ctx.longest_lineage(id, &mut memo)).max().unwrap_or(0);
816 let living = self.state.people.values().filter(|p| p.death.is_none()).count();
817 let links: usize = self.state.people.values().map(|p| p.parents.len()).sum();
818 let couples = self.state.marriages.iter().filter(|m| m.status == Status::Current).count();
819 let founders = self.state.people.values().filter(|p| p.parents.is_empty()).count();
820 Ok(vec![format!("STATS persons={} living={living} links={links} couples={couples} founders={founders} generations={generations}", self.state.people.len())])
821 }
822 "AUDIT" => {
823 let mut lines = Vec::new();
824 for marriage in &self.state.marriages {
825 if let Some(blood) = ctx.blood_term(&marriage.first, &marriage.second, self.state.people[&marriage.first].sex) {
826 lines.push(format!(" marriage {} {} {blood}", marriage.first, marriage.second));
827 }
828 }
829 for id in self.state.people.keys() {
830 let coefficient = ctx.inbreeding(id);
831 if coefficient.numerator != 0 { lines.push(format!(" inbred {id} {coefficient}")); }
832 }
833 lines.insert(0, format!("AUDIT {}", lines.len()));
834 Ok(lines)
835 }
836 _ => unreachable!(),
837 }
838 }
839}

841fn main() {
842 let mut input = String::new();
843 io::stdin().read_to_string(&mut input).unwrap();
844 let mut lines = input.lines();
845 let count: usize = lines.next().unwrap().parse().unwrap();
846 let mut family = Family::default();
847 let mut output = Vec::new();
848 for line in lines.take(count) {
849 match family.command(line) {
850 Ok(lines) => output.extend(lines),
851 Err(reason) => output.push(format!("ERROR {reason}")),
852 }
853 }
854 if !output.is_empty() { println!("{}", output.join("\n")); }
855}
