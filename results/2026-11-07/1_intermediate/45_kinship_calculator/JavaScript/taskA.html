3const SCALE = 2 ** 37;
4const SEX_INDEX = { M: 0, F: 1, X: 2 };
5const ORDINALS = ['first', 'second', 'third', 'fourth', 'fifth',
6 'sixth', 'seventh', 'eighth', 'ninth', 'tenth'];
7const TERMS = {
8 parent: ['father', 'mother', 'parent'],
9 child: ['son', 'daughter', 'child'],
10 sibling: ['brother', 'sister', 'sibling'],
11 pibling: ['uncle', 'aunt', 'pibling'],
12 nibling: ['nephew', 'niece', 'nibling'],
13 spouse: ['husband', 'wife', 'spouse'],
14 late: ['late husband', 'late wife', 'late spouse'],
15 survivor: ['widower', 'widow', 'surviving spouse'],
16 ex: ['ex-husband', 'ex-wife', 'ex-spouse'],
17 stepparent: ['stepfather', 'stepmother', 'stepparent'],
18 stepchild: ['stepson', 'stepdaughter', 'stepchild'],
19 stepsibling: ['stepbrother', 'stepsister', 'stepsibling'],
20 stepgrandparent: ['stepgrandfather', 'stepgrandmother', 'stepgrandparent'],
21 stepgrandchild: ['stepgrandson', 'stepgranddaughter', 'stepgrandchild'],
22 parentInLaw: ['father-in-law', 'mother-in-law', 'parent-in-law'],
23 childInLaw: ['son-in-law', 'daughter-in-law', 'child-in-law'],
24 siblingInLaw: ['brother-in-law', 'sister-in-law', 'sibling-in-law'],
25};
26const STEPS = {
27 parent: ['parent', null], father: ['parent', 'M'], mother: ['parent', 'F'],
28 child: ['child', null], son: ['child', 'M'], daughter: ['child', 'F'],
29 sibling: ['sibling', null], brother: ['sibling', 'M'], sister: ['sibling', 'F'],
30 spouse: ['spouse', null], husband: ['spouse', 'M'], wife: ['spouse', 'F'],
31};
32const MODIFYING = new Set(['PERSON', 'DEATH', 'REMOVE', 'PARENT', 'UNPARENT', 'SPOUSE', 'DIVORCE']);

34class CommandError extends Error {}

36function compareIds(a, b) {
37 return a < b ? -1 : a > b ? 1 : 0;
38}
39
40function gendered(kind, sex) {
41 return TERMS[kind][SEX_INDEX[sex]];
42}

44function generationPrefix(g) {
45 return g === 1 ? '' : 'great-'.repeat(g - 2) + 'grand';
46}
47
48function term(da, db, sex) {
49 if (da === 0 && db === 0) return 'self';
50 if (da === 0) return generationPrefix(db) + gendered('parent', sex);
51 if (db === 0) return generationPrefix(da) + gendered('child', sex);
52 if (da === 1 && db === 1) return gendered('sibling', sex);
53 if (da === 1) return generationPrefix(db - 1) + gendered('pibling', sex);
54 if (db === 1) return generationPrefix(da - 1) + gendered('nibling', sex);
55 const k = Math.min(da, db) - 1;
56 const suffix = [11, 12, 13].includes(k % 100) ? 'th'
57 : ({ 1: 'st', 2: 'nd', 3: 'rd' }[k % 10] ?? 'th');
58 const ordinal = k <= 10 ? ORDINALS[k - 1] : `${k}${suffix}`;
59 const r = Math.abs(da - db);
60 const removed = r === 0 ? '' : r === 1 ? ' once removed'
61 : r === 2 ? ' twice removed' : ` ${r} times removed`;
62 return `${ordinal} cousin${removed}`;
63}

65function bloodTerm(apex, sex) {
66 const prefix = apex.shares === 1 ? 'half-' : apex.shares >= 4 ? 'double ' : '';
67 return prefix + term(apex.da, apex.db, sex);
68}

70function life(person) {
71 return `${person.birth}-${person.death ?? ''}`;
72}

74function fraction(value) {
75 let a = value;
76 let b = SCALE;
77 while (b !== 0) [a, b] = [b, a % b];
78 return `${value / a}/${SCALE / a}`;
79}

81function contains(record, x) {
82 return record.first === x || record.second === x;
83}

85function other(record, x) {
86 return record.first === x ? record.second : record.first;

89class Family {
90 persons = new Map();
91 marriages = [];
92 distanceCache = new Map();
93 coefficientCache = new Map();
94
95 copy() {
96 const family = new Family();
97 family.persons = new Map([...this.persons].map(([x, p]) =>
98 [x, { ...p, parents: [...p.parents] }]));
99 family.marriages = this.marriages.map(m => ({ ...m }));
100 return family;
101 }
102
103 invalidate() {
104 this.distanceCache.clear();
105 this.coefficientCache.clear();
106 }
107
108 person(x) {
109 const person = this.persons.get(x);
110 if (!person) throw new CommandError(`UNKNOWN PERSON ${x}`);
111 return person;
112 }
113
114 ageCompare(a, b) {
115 return this.person(a).birth - this.person(b).birth || compareIds(a, b);
116 }
117
118 ageOrder(ids) {
119 return [...ids].sort((a, b) => this.ageCompare(a, b));
120 }
121
122 personList(ids) {
123 return this.ageOrder(ids).join(' ') || '-';
124 }
125
126 parents(x) {
127 return this.person(x).parents;
128 }
129
130 children(x) {
131 return [...this.persons.keys()].filter(y => this.parents(y).includes(x));
132 }
133
134 commonParents(a, b) {
135 return this.parents(a).filter(p => this.parents(b).includes(p)).length;
136 }
137
138 areSiblings(a, b) {
139 return a !== b && this.commonParents(a, b) > 0;
140 }
141
142 siblings(x) {
143 return [...this.persons.keys()].filter(y => this.areSiblings(x, y));
144 }
145
146 records(x) {
147 return this.marriages.filter(m => contains(m, x));
148 }
149
150 current(x) {
151 return this.marriages.find(m => m.status === 'current' && contains(m, x));
152 }
153
154 spouse(x) {
155 const record = this.current(x);
156 return record ? other(record, x) : null;
157 }
158
159 last(a, b) {
160 return this.marriages.findLast(m => contains(m, a) && contains(m, b));
161 }
162
163 partners(x) {
164 const latest = new Map(this.records(x).map(m => [other(m, x), m]));
165 return new Set([...latest].filter(([, m]) => m.status !== 'divorced').map(([y]) => y));
166 }
167
168 stepRelated(a, b) {
169 return this.parents(a).some(p => this.parents(b).some(q => this.partners(p).has(q)));
170 }
171
172 distances(start, upward) {
173 const key = `${upward}:${start}`;
174 if (this.distanceCache.has(key)) return this.distanceCache.get(key);
175 const distances = new Map([[start, 0]]);
176 const queue = [start];
177 for (let i = 0; i < queue.length; ++i) {
178 const x = queue[i];
179 for (const y of upward ? this.parents(x) : this.children(x)) {
180 if (!distances.has(y)) {
181 distances.set(y, distances.get(x) + 1);
182 queue.push(y);
183 }
184 }
185 }
186 this.distanceCache.set(key, distances);
187 return distances;
188 }
189
190 ancestor(a, b) {
191 return a !== b && this.distances(b, true).has(a);
192 }
193
194 grandparents(x) {
195 return [...this.distances(x, true)].filter(([, g]) => g === 2).map(([y]) => y);
196 }
197
198 apex(a, b) {
199 const upA = this.distances(a, true);
200 const upB = this.distances(b, true);
201 if (upB.has(a)) return { person: a, da: 0, db: upB.get(a), shares: 0 };
202 if (upA.has(b)) return { person: b, da: upA.get(b), db: 0, shares: 0 };
203 const common = [...upA.keys()].filter(c => upB.has(c));
204 common.sort((c, d) => (upA.get(c) + upB.get(c)) - (upA.get(d) + upB.get(d))
205 || upA.get(c) - upA.get(d) || compareIds(c, d));
206 if (common.length === 0) return null;
207 const person = common[0];
208 const da = upA.get(person);
209 const db = upB.get(person);
210 const shares = common.filter(c => upA.get(c) === da && upB.get(c) === db).length;
211 return { person, da, db, shares };
212 }
213
214 relationship(a, b, sex = this.person(a).sex) {
215 if (a === b) return ['self'];
216 const terms = [];
217 const apex = this.apex(a, b);
218 if (apex) terms.push(bloodTerm(apex, sex));
219 const record = this.last(a, b);
220 if (!record) {
221 if ([...this.persons.values()].some(p => p.parents.includes(a) && p.parents.includes(b)))
222 terms.push('co-parent');
223 } else if (record.status === 'current') terms.push(gendered('spouse', sex));
224 else if (record.status === 'widowed')
225 terms.push(gendered(record.died === a ? 'late' : 'survivor', sex));
226 else terms.push(gendered('ex', sex));
227 const pa = this.parents(a);
228 const pb = this.parents(b);
229 const qa = this.partners(a);
230 const qb = this.partners(b);
231 if (!pb.includes(a) && pb.some(p => qa.has(p))) terms.push(gendered('stepparent', sex));
232 if (!pa.includes(b) && pa.some(p => qb.has(p))) terms.push(gendered('stepchild', sex));
233 if (this.commonParents(a, b) === 0 && this.stepRelated(a, b))
234 terms.push(gendered('stepsibling', sex));
235 if (!this.ancestor(a, b) && this.grandparents(b).some(g => qa.has(g)))
236 terms.push(gendered('stepgrandparent', sex));
237 if (!this.ancestor(b, a) && this.grandparents(a).some(g => qb.has(g)))
238 terms.push(gendered('stepgrandchild', sex));
239 if ([...qb].some(q => this.parents(q).includes(a))) terms.push(gendered('parentInLaw', sex));
240 if ([...qa].some(q => this.parents(q).includes(b))) terms.push(gendered('childInLaw', sex));
241 if ([...qb].some(q => this.areSiblings(a, q)) || [...qa].some(q => this.areSiblings(q, b)))
242 terms.push(gendered('siblingInLaw', sex));
243 return terms;
244 }
245
246 relationshipText(a, b) {
247 return this.relationship(a, b).join(', ') || 'no relation';
248 }
249
250 coefficient(a, b) {
251 const key = [a, b].sort(compareIds).join(':');
252 if (this.coefficientCache.has(key)) return this.coefficientCache.get(key);
253 let value;
254 if (a === b) value = (SCALE + this.inbreeding(a)) / 2;
255 else {
256 const [x, y] = this.person(a).birth >= this.person(b).birth ? [a, b] : [b, a];
257 value = this.parents(x).reduce((sum, p) => sum + this.coefficient(p, y), 0) / 2;
258 }
259 this.coefficientCache.set(key, value);
260 return value;
261 }
262
263 inbreeding(x) {
264 const parents = this.parents(x);
265 return parents.length === 2 ? this.coefficient(parents[0], parents[1]) : 0;
266 }
267
268 lineageLength(x, memo) {
269 if (!memo.has(x)) memo.set(x, 1 + Math.max(0, ...this.parents(x).map(p => this.lineageLength(p, memo))));
270 return memo.get(x);
271 }
272}

274class Calculator {
275 family = new Family();
276 history = [];
277
278 execute(line) {
279 const [command, ...args] = line.split(' ');
280 const saved = MODIFYING.has(command) ? this.family.copy() : null;
281 try {
282 const output = this.dispatch(command, args);
283 if (saved) {
284 this.history.push({ line, family: saved });
285 this.family.invalidate();
286 }
287 return output;
288 } catch (error) {
289 if (!(error instanceof CommandError)) throw error;
290 return [`ERROR ${error.message}`];
291 }
292 }
293
294 dispatch(command, args) {
295 const f = this.family;
296 const [x, y] = args;
297 const year = Number(args[2]);
298 switch (command) {
299 case 'PERSON': {
300 if (f.persons.has(x)) throw new CommandError(`DUPLICATE PERSON ${x}`);
301 f.persons.set(x, { sex: y, birth: year, name: args[3], death: null, parents: [] });
302 return [`ADDED ${x} ${args[3]} ${y} ${year}`];
303 }
304 case 'DEATH': {
305 const p = f.person(x);
306 const death = Number(y);
307 if (p.death !== null) throw new CommandError(`ALREADY DECEASED ${x}`);
308 if (death < p.birth || f.records(x).some(m => death < m.start || (m.end !== null && death < m.end)))
309 throw new CommandError(`BAD YEAR ${x}`);
310 const late = f.children(x).filter(c => f.person(c).birth > death + 1).sort(compareIds);
311 if (late.length) throw new CommandError(`AFTER DEATH ${late[0]} ${x}`);
312 p.death = death;
313 const record = f.current(x);
314 if (!record) return [`DIED ${x} ${death}`];
315 record.end = death;
316 record.status = 'widowed';
317 record.died = x;
318 return [`DIED ${x} ${death} spouse=${other(record, x)}`];
319 }
320 case 'REMOVE':
321 f.person(x);
322 for (const p of f.persons.values()) p.parents = p.parents.filter(p => p !== x);
323 f.marriages = f.marriages.filter(m => !contains(m, x));
324 f.persons.delete(x);
325 return [`REMOVED ${x}`];
326 case 'PARENT': {
327 const child = f.person(x);
328 const parent = f.person(y);
329 if (x === y) throw new CommandError(`SAME PERSON ${x}`);
330 if (child.parents.includes(y)) throw new CommandError(`ALREADY PARENT ${x} ${y}`);
331 if (child.parents.length === 2) throw new CommandError(`TOO MANY PARENTS ${x}`);
332 if (f.ancestor(x, y)) throw new CommandError(`CYCLE ${x} ${y}`);
333 if (parent.birth + 12 > child.birth) throw new CommandError(`AGE GAP ${x} ${y}`);
334 if (parent.death !== null && parent.death + 1 < child.birth)
335 throw new CommandError(`AFTER DEATH ${x} ${y}`);
336 child.parents.push(y);
337 return [`LINKED ${x} ${y} ${gendered('parent', parent.sex)}`];
338 }
339 case 'UNPARENT': {
340 const child = f.person(x);
341 f.person(y);
342 if (!child.parents.includes(y)) throw new CommandError(`NOT PARENT ${x} ${y}`);
343 child.parents = child.parents.filter(p => p !== y);
344 return [`UNLINKED ${x} ${y}`];
345 }
346 case 'SPOUSE': {
347 f.person(x);
348 f.person(y);
349 if (x === y) throw new CommandError(`SAME PERSON ${x}`);
350 for (const a of [x, y]) if (f.person(a).death !== null) throw new CommandError(`DECEASED ${a}`);
351 for (const a of [x, y]) if (f.spouse(a) !== null) throw new CommandError(`ALREADY MARRIED ${a}`);
352 if (f.ancestor(x, y) || f.ancestor(y, x) || f.areSiblings(x, y))
353 throw new CommandError(`CLOSE KIN ${x} ${y}`);
354 for (const a of [x, y]) if (year < f.person(a).birth + 16) throw new CommandError(`TOO YOUNG ${a}`);
355 for (const a of [x, y])
356 if (f.records(a).some(m => m.end !== null && m.end >= year)) throw new CommandError(`BAD YEAR ${a}`);
357 f.marriages.push({ first: x, second: y, start: year, end: null, status: 'current', died: null });
358 return [`MARRIED ${x} ${y} ${year}`];
359 }
360 case 'DIVORCE': {
361 f.person(x);
362 f.person(y);
363 if (f.spouse(x) !== y) throw new CommandError(`NOT MARRIED ${x} ${y}`);
364 const record = f.current(x);
365 if (year <= record.start) throw new CommandError(`BAD YEAR ${x}`);
366 record.end = year;
367 record.status = 'divorced';
368 return [`DIVORCED ${x} ${y} ${year}`];
369 }
370 case 'UNDO': {
371 const saved = this.history.pop();
372 if (!saved) throw new CommandError('NOTHING TO UNDO');
373 this.family = saved.family;
374 return [`UNDONE ${saved.line}`];
375 }
376 case 'TERM':
377 return [`TERM ${x} ${y} ${args[2]} ${term(Number(x), Number(y), args[2])}`];
378 case 'STATS': {
379 const people = [...f.persons.values()];
380 const memo = new Map();
381 return [`STATS persons=${people.length} living=${people.filter(p => p.death === null).length}`
382 + ` links=${people.reduce((sum, p) => sum + p.parents.length, 0)}`
383 + ` couples=${f.marriages.filter(m => m.status === 'current').length}`
384 + ` founders=${people.filter(p => p.parents.length === 0).length}`
385 + ` generations=${Math.max(0, ...[...f.persons.keys()].map(a => f.lineageLength(a, memo)))}`];
386 }
387 case 'AUDIT': {
388 const lines = [];
389 for (const m of f.marriages) {
390 const apex = f.apex(m.first, m.second);
391 if (apex) lines.push(` marriage ${m.first} ${m.second} ${bloodTerm(apex, f.person(m.first).sex)}`);
392 }
393 for (const a of [...f.persons.keys()].sort(compareIds))
394 if (f.inbreeding(a) !== 0) lines.push(` inbred ${a} ${fraction(f.inbreeding(a))}`);
395 return [`AUDIT ${lines.length}`, ...lines];
396 }
397 case 'CENSUS': {
398 const censusYear = Number(x);
399 const residents = f.ageOrder([...f.persons.keys()].filter(a => {
400 const p = f.person(a);
401 return p.birth <= censusYear && (p.death === null || p.death > censusYear);
402 }));
403 const lines = residents.map(a => {
404 const m = f.records(a).findLast(m => m.start <= censusYear);
405 const status = !m ? 'single' : m.end === null || m.end > censusYear
406 ? `married ${other(m, a)}` : m.status;
407 return ` ${a} ${censusYear - f.person(a).birth} ${status}`;
408 });
409 return [`CENSUS ${censusYear} ${lines.length}`, ...lines];
410 }
411 }
412 const person = f.person(x);
413 if (['RELATION', 'PATH', 'COMMON', 'KINSHIP'].includes(command)) f.person(y);
414 const ids = [...f.persons.keys()].sort(compareIds);
415 switch (command) {
416 case 'INFO':
417 return [`INFO ${x} ${person.name} ${person.sex} ${life(person)}`,
418 ` parents: ${f.personList(person.parents)}`,
419 ...f.records(x).map(m => ` marriage: ${other(m, x)} ${m.start}-${m.end ?? ''} ${m.status}`),
420 ` children: ${f.personList(f.children(x))}`,
421 ` siblings: ${f.personList(f.siblings(x))}`,
422 ` inbreeding: ${fraction(f.inbreeding(x))}`];
423 case 'RELATION':
424 return [`RELATION ${x} ${y}: ${f.relationshipText(x, y)}`];
425 case 'KINSHIP':
426 return [`KINSHIP ${x} ${y} ${fraction(f.coefficient(x, y))}`];
427 case 'RELATIVES': {
428 const lines = [];
429 for (const a of ids) {
430 if (a === x) continue;
431 const terms = f.relationship(a, x);
432 if (terms.length) lines.push(` ${a}: ${terms.join(', ')}`);
433 }
434 return [`RELATIVES ${x} ${lines.length}`, ...lines];
435 }
436 case 'WHO': {
437 const query = args.slice(1).join(' ');
438 const lines = [];
439 for (const a of ids) {
440 if (a === x) continue;
441 const terms = f.relationship(a, x);
442 const neutral = f.relationship(a, x, 'X');
443 const i = terms.findIndex((t, i) => t === query || neutral[i] === query);
444 if (i >= 0) lines.push(` ${a}: ${terms[i]}`);
445 }
446 return [`WHO ${x} ${query} ${lines.length}`, ...lines];
447 }
448 case 'PATH': {
449 const apex = f.apex(x, y);
450 if (!apex) return [`PATH ${x} ${y} none`];
451 const apexPerson = apex.person;
452 const depths = f.distances(apexPerson, false);
453 function upward(start) {
454 const path = [start];
455 while (path.at(-1) !== apexPerson) {
456 const current = path.at(-1);
457 path.push(f.parents(current).filter(p => depths.get(p) === depths.get(current) - 1).sort(compareIds)[0]);
458 }
459 return path;
460 }
461 const route = [...upward(x), ...upward(y).slice(0, -1).reverse()];
462 return [`PATH ${x} ${y} ${apex.da + apex.db}: ${route.join(' ')}`];
463 }
464 case 'COMMON': {
465 const a = f.distances(x, true);
466 const b = f.distances(y, true);
467 const common = new Set([...a.keys()].filter(c => b.has(c)));
468 const nearest = [...common].filter(c => !f.children(c).some(child => common.has(child)));
469 nearest.sort((c, d) => (a.get(c) + b.get(c)) - (a.get(d) + b.get(d)) || compareIds(c, d));
470 return [`COMMON ${x} ${y} ${nearest.length}`, ...nearest.map(c => ` ${c} ${a.get(c)} ${b.get(c)}`)];
471 }
472 case 'ANCESTORS':
473 case 'DESCENDANTS': {
474 const upward = command === 'ANCESTORS';
475 const entries = [...f.distances(x, upward)].filter(([a]) => a !== x);
476 entries.sort(([a, g], [b, h]) => g - h || compareIds(a, b));
477 return [`${command} ${x} ${entries.length}`, ...entries.map(([a, g]) =>
478 ` ${g} ${a} ${term(upward ? 0 : g, upward ? g : 0, f.person(a).sex)}`)];
479 }
480 case 'COUSINS': {
481 const matches = [];
482 for (const a of ids) {
483 if (a === x) continue;
484 const apex = f.apex(a, x);
485 if (apex && apex.da >= 2 && apex.db >= 2 && Math.min(apex.da, apex.db) - 1 === Number(y))
486 matches.push({ id: a, apex });
487 }
488 matches.sort((a, b) => Math.abs(a.apex.da - a.apex.db) - Math.abs(b.apex.da - b.apex.db) || compareIds(a.id, b.id));
489 return [`COUSINS ${x} ${y} ${matches.length}`, ...matches.map(a =>
490 ` ${a.id} ${bloodTerm(a.apex, f.person(a.id).sex)}`)];
491 }
492 case 'SIBLINGS': {
493 const lines = [];
494 let earlier = 0;
495 for (const a of f.ageOrder(ids)) {
496 if (a === x) continue;
497 const common = f.commonParents(x, a);
498 const kind = common === 2 ? 'full' : common === 1 ? 'half' : f.stepRelated(x, a) ? 'step' : null;
499 if (!kind) continue;
500 if (common > 0 && f.ageCompare(a, x) < 0) ++earlier;
501 lines.push(` ${a} ${kind} ${f.person(a).birth}`);
502 }
503 return [`SIBLINGS ${x} ${lines.length} order=${earlier + 1}`, ...lines];
504 }
505 case 'GENERATIONS': {
506 const groups = new Map();
507 for (const [a, g] of f.distances(x, false)) {
508 if (a === x) continue;
509 if (!groups.has(g)) groups.set(g, []);
510 groups.get(g).push(f.person(a));
511 }
512 const lines = [...groups].sort(([g], [h]) => g - h).map(([g, people]) => {
513 const births = people.map(p => p.birth);
514 return ` ${g} ${people.length} ${people.filter(p => p.death === null).length} ${Math.min(...births)}-${Math.max(...births)}`;
515 });
516 return [`GENERATIONS ${x} ${groups.size}`, ...lines];
517 }
518 case 'TREE': {
519 const lines = [];
520 const visited = new Set();
521 function visit(a, depth) {
522 const indent = ' '.repeat(2 * (depth + 1));
523 if (visited.has(a)) { lines.push(`${indent}${a} (see above)`); return; }
524 visited.add(a);
525 const p = f.person(a);
526 const spouse = f.spouse(a);
527 lines.push(`${indent}${a} ${p.name} ${life(p)}${spouse === null ? '' : ` = ${spouse}`}`);
528 for (const child of f.ageOrder(f.children(a))) visit(child, depth + 1);
529 }
530 visit(x, 0);
531 return [`TREE ${x} ${f.distances(x, false).size - 1}`, ...lines];
532 }
533 case 'PEDIGREE': {
534 const depth = Number(y);
535 const lines = [];
536 const entered = new Set();
537 function enter(a, number, generation) {
538 const p = f.person(a);
539 lines.push(`${' '.repeat(2 * (generation + 1))}${number} ${a} ${p.name} ${life(p)}`);
540 entered.add(a);
541 if (generation >= depth) return;
542 const parents = [...p.parents].sort((a, b) => SEX_INDEX[f.person(a).sex] - SEX_INDEX[f.person(b).sex] || compareIds(a, b));
543 parents.forEach((parent, i) => enter(parent, 2 * number + i, generation + 1));
544 }
545 enter(x, 1, 0);
546 return [`PEDIGREE ${x} ${y} entries=${lines.length} distinct=${entered.size}`, ...lines];
547 }
548 case 'TIMELINE': {
549 const events = [{ year: person.birth, rank: 0, counterpart: '', text: 'born' }];
550 for (const m of f.records(x)) {
551 const a = other(m, x);
552 events.push({ year: m.start, rank: 1, counterpart: a, text: `married ${a}` });
553 if (m.status === 'divorced') events.push({ year: m.end, rank: 3, counterpart: a, text: `divorced ${a}` });
554 else if (m.status === 'widowed' && m.died === a)
555 events.push({ year: m.end, rank: 4, counterpart: a, text: `widowed ${a}` });
556 }
557 for (const a of f.children(x)) events.push({ year: f.person(a).birth, rank: 2, counterpart: a, text: `child ${a}` });
558 for (const [label, relatives] of [['parent', person.parents], ['child', f.children(x)]]) {
559 for (const a of relatives) {
560 const death = f.person(a).death;
561 if (death !== null && death >= person.birth && (person.death === null || death <= person.death))
562 events.push({ year: death, rank: 5, counterpart: a, text: `lost ${label} ${a}` });
563 }
564 }
565 if (person.death !== null) events.push({ year: person.death, rank: 6, counterpart: '', text: 'died' });
566 events.sort((a, b) => a.year - b.year || a.rank - b.rank || compareIds(a.counterpart, b.counterpart));
567 return [`TIMELINE ${x} ${events.length}`, ...events.map(e => ` ${e.year} ${e.text}`)];
568 }
569 case 'FIND': {
570 let current = new Set([x]);
571 for (const step of y.split('.')) {
572 const [kind, sex] = STEPS[step];
573 const next = new Set();
574 for (const a of current) {
575 const targets = kind === 'parent' ? f.parents(a) : kind === 'child' ? f.children(a)
576 : kind === 'sibling' ? f.siblings(a) : f.spouse(a) === null ? [] : [f.spouse(a)];
577 for (const b of targets) if (sex === null || f.person(b).sex === sex) next.add(b);
578 }
579 current = next;
580 }
581 current.delete(x);
582 return [`FIND ${x} ${y} ${current.size}`, ...[...current].sort(compareIds).map(a => ` ${a}: ${f.relationshipText(a, x)}`)];
583 }
584 }
585 }
