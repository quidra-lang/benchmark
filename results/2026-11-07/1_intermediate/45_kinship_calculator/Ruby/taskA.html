1SEX_INDEX = { 'M' => 0, 'F' => 1, 'X' => 2 }.freeze
2ORDINALS = %w[first second third fourth fifth sixth seventh eighth ninth tenth].freeze
3TERMS = {
4 'parent' => %w[father mother parent], 'child' => %w[son daughter child],
5 'sibling' => %w[brother sister sibling], 'pibling' => %w[uncle aunt pibling],
6 'nibling' => %w[nephew niece nibling], 'spouse' => %w[husband wife spouse],
7 'late' => ['late husband', 'late wife', 'late spouse'],
8 'survivor' => ['widower', 'widow', 'surviving spouse'],
9 'ex' => %w[ex-husband ex-wife ex-spouse],
10 'stepparent' => %w[stepfather stepmother stepparent],
11 'stepchild' => %w[stepson stepdaughter stepchild],
12 'stepsibling' => %w[stepbrother stepsister stepsibling],
13 'stepgrandparent' => %w[stepgrandfather stepgrandmother stepgrandparent],
14 'stepgrandchild' => %w[stepgrandson stepgranddaughter stepgrandchild],
15 'parent-in-law' => %w[father-in-law mother-in-law parent-in-law],
16 'child-in-law' => %w[son-in-law daughter-in-law child-in-law],
17 'sibling-in-law' => %w[brother-in-law sister-in-law sibling-in-law]
18}.freeze
19STEPS = {
20 'parent' => ['parent', nil], 'father' => ['parent', 'M'], 'mother' => ['parent', 'F'],
21 'child' => ['child', nil], 'son' => ['child', 'M'], 'daughter' => ['child', 'F'],
22 'sibling' => ['sibling', nil], 'brother' => ['sibling', 'M'], 'sister' => ['sibling', 'F'],
23 'spouse' => ['spouse', nil], 'husband' => ['spouse', 'M'], 'wife' => ['spouse', 'F']
24}.freeze
25MODIFYING = %w[PERSON DEATH REMOVE PARENT UNPARENT SPOUSE DIVORCE].freeze
26
27class CommandError < StandardError; end
28
29def gendered(kind, sex)
30 TERMS[kind][SEX_INDEX[sex]]
31end
32
33def generation_prefix(generations)
34 generations == 1 ? '' : 'great-' * (generations - 2) + 'grand'
35end
36
37def ordinal(number)
38 return ORDINALS[number - 1] if number <= 10
39
40 suffix = if [11, 12, 13].include?(number % 100)
41 'th'
42 else
43 { 1 => 'st', 2 => 'nd', 3 => 'rd' }.fetch(number % 10, 'th')
44 end
45 "#{number}#{suffix}"
46end
47
48def kinship_term(da, db, sex)
49 return 'self' if da.zero? && db.zero?
50 return generation_prefix(db) + gendered('parent', sex) if da.zero?
51 return generation_prefix(da) + gendered('child', sex) if db.zero?
52 return gendered('sibling', sex) if da == 1 && db == 1
53 return generation_prefix(db - 1) + gendered('pibling', sex) if da == 1
54 return generation_prefix(da - 1) + gendered('nibling', sex) if db == 1
55
56 distance = (da - db).abs
57 removal = case distance
58 when 0 then ''
59 when 1 then ' once removed'
60 when 2 then ' twice removed'
61 else " #{distance} times removed"
62 end
63 ordinal([da, db].min - 1) + ' cousin' + removal
64end
65
66def fraction_text(value)
67 "#{value.numerator}/#{value.denominator}"
68end

70Person = Struct.new(:id, :sex, :birth, :name, :death, :parents) do
71 def life_years
72 "#{birth}-#{death}"
73 end
74end
75
76Marriage = Struct.new(:first, :second, :start, :finish, :status, :died) do
77 def contains?(id)
78 id == first || id == second
79 end
80
81 def other(id)
82 id == first ? second : first
83 end
84end
85
86Apex = Struct.new(:person, :da, :db, :shares) do
87 def blood_term(sex)
88 prefix = if shares == 1
89 'half-'
90 elsif shares >= 4
91 'double '
92 else
93 ''
94 end
95 prefix + kinship_term(da, db, sex)
96 end
97end

99class Family
100 attr_reader :persons
101 attr_accessor :marriages
102
103 def initialize
104 @persons = {}
105 @marriages = []
106 reset_cache
107 end
108
109 def reset_cache
110 @distances = {}
111 @children = {}
112 @records = {}
113 @partners = {}
114 @coefficients = {}
115 end
116
117 def person(id)
118 @persons.fetch(id) { raise CommandError, "UNKNOWN PERSON #{id}" }
119 end
120
121 def sex(id)
122 @persons[id].sex
123 end
124
125 def age_key(id)
126 [@persons[id].birth, id]
127 end
128
129 def person_list(ids)
130 ids.empty? ? '-' : ids.sort_by { |id| age_key(id) }.join(' ')
131 end
132
133 def parents(id)
134 @persons[id].parents
135 end
136
137 def children(id)
138 @children[id] ||= @persons.keys.select { |child| parents(child).include?(id) }
139 end
140
141 def common_parents(a, b)
142 (parents(a) & parents(b)).length
143 end
144
145 def siblings?(a, b)
146 a != b && common_parents(a, b).positive?
147 end
148
149 def siblings(id)
150 @persons.keys.select { |other| siblings?(id, other) }
151 end
152
153 def common_child?(a, b)
154 @persons.each_value.any? { |person| person.parents.include?(a) && person.parents.include?(b) }
155 end
156
157 def records_of(id)
158 @records[id] ||= @marriages.select { |record| record.contains?(id) }
159 end
160
161 def current_record(id)
162 records_of(id).find { |record| record.status == 'current' }
163 end
164
165 def spouse(id)
166 current_record(id)&.other(id)
167 end
168
169 def last_record(a, b)
170 records_of(a).reverse.find { |record| record.contains?(b) }
171 end
172
173 def partners(id)
174 @partners[id] ||= begin
175 latest = {}
176 records_of(id).each { |record| latest[record.other(id)] = record }
177 latest.keys.reject { |other| latest[other].status == 'divorced' }
178 end
179 end
180
181 def step_related?(parents_a, parents_b)
182 parents_a.any? { |parent| (partners(parent) & parents_b).any? }
183 end
184
185 def generation_distances(start, upward)
186 @distances[[start, upward]] ||= begin
187 distances = { start => 0 }
188 queue = [start]
189 index = 0
190 while index < queue.length
191 id = queue[index]
192 index += 1
193 neighbors = upward ? parents(id) : children(id)
194 neighbors.each do |other|
195 next if distances.key?(other)
196
197 distances[other] = distances[id] + 1
198 queue << other
199 end
200 end
201 distances
202 end
203 end
204
205 def lineage(id, upward)
206 generation_distances(id, upward).filter_map do |other, distance|
207 [distance, other] unless other == id
208 end.sort
209 end
210
211 def ancestor?(a, b)
212 a != b && generation_distances(b, true).key?(a)
213 end
214
215 def grandparents(id)
216 generation_distances(id, true).filter_map { |other, distance| other if distance == 2 }
217 end
218
219 def apex(a, b)
220 up_a = generation_distances(a, true)
221 up_b = generation_distances(b, true)
222 return Apex.new(a, 0, up_b[a], 0) if up_b.key?(a)
223 return Apex.new(b, up_a[b], 0, 0) if up_a.key?(b)
224
225 common = up_a.keys & up_b.keys
226 return nil if common.empty?
227
228 chosen = common.min_by { |id| [up_a[id] + up_b[id], up_a[id], id] }
229 da = up_a[chosen]
230 db = up_b[chosen]
231 shares = common.count { |id| up_a[id] == da && up_b[id] == db }
232 Apex.new(chosen, da, db, shares)
233 end
234
235 def relationship(a, b, term_sex = sex(a))
236 return ['self'] if a == b
237
238 terms = []
239 apex = apex(a, b)
240 terms << apex.blood_term(term_sex) if apex
241 record = last_record(a, b)
242 if record.nil?
243 terms << 'co-parent' if common_child?(a, b)
244 elsif record.status == 'current'
245 terms << gendered('spouse', term_sex)
246 elsif record.status == 'widowed'
247 terms << gendered(record.died == a ? 'late' : 'survivor', term_sex)
248 else
249 terms << gendered('ex', term_sex)
250 end
251 parents_a = parents(a)
252 parents_b = parents(b)
253 partners_a = partners(a)
254 partners_b = partners(b)
255 if !parents_b.include?(a) && (partners_a & parents_b).any?
256 terms << gendered('stepparent', term_sex)
257 end
258 if !parents_a.include?(b) && (partners_b & parents_a).any?
259 terms << gendered('stepchild', term_sex)
260 end
261 if common_parents(a, b).zero? && step_related?(parents_a, parents_b)
262 terms << gendered('stepsibling', term_sex)
263 end
264 if !ancestor?(a, b) && (partners_a & grandparents(b)).any?
265 terms << gendered('stepgrandparent', term_sex)
266 end
267 if !ancestor?(b, a) && (partners_b & grandparents(a)).any?
268 terms << gendered('stepgrandchild', term_sex)
269 end
270 terms << gendered('parent-in-law', term_sex) if partners_b.any? { |other| parents(other).include?(a) }
271 terms << gendered('child-in-law', term_sex) if partners_a.any? { |other| parents(other).include?(b) }
272 if partners_b.any? { |other| siblings?(a, other) } || partners_a.any? { |other| siblings?(other, b) }
273 terms << gendered('sibling-in-law', term_sex)
274 end
275 terms
276 end
277
278 def relationship_text(a, b)
279 terms = relationship(a, b)
280 terms.empty? ? 'no relation' : terms.join(', ')
281 end
282
283 def kinship(a, b)
284 key = [a, b].sort
285 @coefficients[key] ||= if a == b
286 (1 + inbreeding(a)) / 2
287 else
288 x, y = @persons[a].birth >= @persons[b].birth ? [a, b] : [b, a]
289 parents(x).sum(Rational(0)) { |parent| kinship(parent, y) } / 2
290 end
291 end
292
293 def inbreeding(id)
294 ps = parents(id)
295 ps.length == 2 ? kinship(ps[0], ps[1]) : Rational(0)
296 end
297
298 def lineage_length(id, memo)
299 memo[id] ||= 1 + (parents(id).map { |parent| lineage_length(parent, memo) }.max || 0)
300 end
301
302 def upward_path(start, apex)
303 route = [start]
304 until route.last == apex
305 current = route.last
306 distance = generation_distances(current, true)[apex]
307 candidates = parents(current).select do |parent|
308 generation_distances(parent, true)[apex] == distance - 1
309 end
310 route << candidates.min
311 end
312 route
313 end

315 def census_status(id, year)
316 record = records_of(id).reverse.find { |marriage| marriage.start <= year }
317 return 'single' unless record
318 return "married #{record.other(id)}" if record.finish.nil? || record.finish > year
319
320 record.status
321 end

323 def step_targets(id, kind)
324 case kind
325 when 'parent' then parents(id)
326 when 'child' then children(id)
327 when 'sibling' then siblings(id)
328 when 'spouse' then [spouse(id)].compact
329 end
330 end
331end

333class KinshipCalculator
334 def initialize
335 @family = Family.new
336 @history = []
337 end

339 def execute(line)
340 command, *args = line.split(' ')
341 @family.reset_cache
342 saved = Marshal.dump(@family) if MODIFYING.include?(command)
343 lines = process(command, args)
344 @history << [line, saved] if MODIFYING.include?(command)
345 lines
346 rescue CommandError => error
347 ["ERROR #{error.message}"]
348 end
349
350 def process(command, args)
351 case command
352 when 'PERSON' then add_person(*args)
353 when 'DEATH' then record_death(*args)
354 when 'REMOVE' then remove_person(*args)
355 when 'PARENT' then link_parent(*args)
356 when 'UNPARENT' then unlink_parent(*args)
357 when 'SPOUSE' then marry(*args)
358 when 'DIVORCE' then divorce(*args)
359 when 'UNDO'
360 raise CommandError, 'NOTHING TO UNDO' if @history.empty?
361
362 line, saved = @history.pop
363 @family = Marshal.load(saved)
364 ["UNDONE #{line}"]
365 when 'INFO' then info(*args)
366 when 'RELATION'
367 a, b = checked(*args)
368 ["RELATION #{a} #{b}: #{@family.relationship_text(a, b)}"]
369 when 'RELATIVES' then relatives(*args)
370 when 'WHO' then who(args.shift, args.join(' '))
371 when 'PATH' then path(*args)
372 when 'COMMON' then common(*args)
373 when 'ANCESTORS' then lineage(args.first, true)
374 when 'DESCENDANTS' then lineage(args.first, false)
375 when 'COUSINS' then cousins(*args)
376 when 'SIBLINGS' then siblings(*args)
377 when 'GENERATIONS' then generations(*args)
378 when 'TREE' then tree(*args)
379 when 'PEDIGREE' then pedigree(*args)
380 when 'TIMELINE' then timeline(*args)
381 when 'CENSUS' then census(*args)
382 when 'KINSHIP'
383 a, b = checked(*args)
384 ["KINSHIP #{a} #{b} #{fraction_text(@family.kinship(a, b))}"]
385 when 'FIND' then find(*args)
386 when 'TERM'
387 da, db, sex = args
388 ["TERM #{da} #{db} #{sex} #{kinship_term(da.to_i, db.to_i, sex)}"]
389 when 'STATS' then stats
390 when 'AUDIT' then audit
391 end
392 end

399 def add_person(id, sex, birth, name)
400 raise CommandError, "DUPLICATE PERSON #{id}" if @family.persons.key?(id)
401
402 @family.persons[id] = Person.new(id, sex, birth.to_i, name, nil, [])
403 ["ADDED #{id} #{name} #{sex} #{birth}"]
404 end

406 def record_death(id, year)
407 year = year.to_i
408 person = @family.person(id)
409 raise CommandError, "ALREADY DECEASED #{id}" if person.death
410
411 years = [person.birth] + @family.records_of(id).flat_map { |record| [record.start, record.finish].compact }
412 raise CommandError, "BAD YEAR #{id}" if year < years.max
413
414 late = @family.children(id).select { |child| @family.persons[child].birth > year + 1 }.min
415 raise CommandError, "AFTER DEATH #{late} #{id}" if late
416
417 person.death = year
418 record = @family.current_record(id)
419 return ["DIED #{id} #{year}"] unless record
420
421 record.finish = year
422 record.status = 'widowed'
423 record.died = id
424 ["DIED #{id} #{year} spouse=#{record.other(id)}"]
425 end

427 def remove_person(id)
428 checked(id)
429 @family.persons.each_value { |person| person.parents.delete(id) }
430 @family.marriages.reject! { |record| record.contains?(id) }
431 @family.persons.delete(id)
432 ["REMOVED #{id}"]
433 end

435 def link_parent(child_id, parent_id)
436 child = @family.person(child_id)
437 parent = @family.person(parent_id)
438 raise CommandError, "SAME PERSON #{child_id}" if child_id == parent_id
439 raise CommandError, "ALREADY PARENT #{child_id} #{parent_id}" if child.parents.include?(parent_id)
440 raise CommandError, "TOO MANY PARENTS #{child_id}" if child.parents.length == 2
441 raise CommandError, "CYCLE #{child_id} #{parent_id}" if @family.ancestor?(child_id, parent_id)
442 raise CommandError, "AGE GAP #{child_id} #{parent_id}" if parent.birth + 12 > child.birth
443 if parent.death && parent.death + 1 < child.birth
444 raise CommandError, "AFTER DEATH #{child_id} #{parent_id}"
445 end
446 child.parents << parent_id
447 ["LINKED #{child_id} #{parent_id} #{gendered('parent', parent.sex)}"]
448 end
449
450 def unlink_parent(child_id, parent_id)
451 checked(child_id, parent_id)
452 parents = @family.parents(child_id)
453 raise CommandError, "NOT PARENT #{child_id} #{parent_id}" unless parents.include?(parent_id)
454
455 parents.delete(parent_id)
456 ["UNLINKED #{child_id} #{parent_id}"]
457 end

459 def marry(a, b, year)
460 checked(a, b)
461 year = year.to_i
462 raise CommandError, "SAME PERSON #{a}" if a == b
463
464 [a, b].each { |id| raise CommandError, "DECEASED #{id}" if @family.persons[id].death }
465 [a, b].each { |id| raise CommandError, "ALREADY MARRIED #{id}" if @family.spouse(id) }
466 if @family.ancestor?(a, b) || @family.ancestor?(b, a) || @family.siblings?(a, b)
467 raise CommandError, "CLOSE KIN #{a} #{b}"
468 end
469 [a, b].each do |id|
470 raise CommandError, "TOO YOUNG #{id}" if year < @family.persons[id].birth + 16
471 end
472 [a, b].each do |id|
473 if @family.records_of(id).any? { |record| record.finish && record.finish >= year }
474 raise CommandError, "BAD YEAR #{id}"
475 end
476 end
477 @family.marriages << Marriage.new(a, b, year, nil, 'current', nil)
478 ["MARRIED #{a} #{b} #{year}"]
479 end

481 def divorce(a, b, year)
482 checked(a, b)
483 year = year.to_i
484 raise CommandError, "NOT MARRIED #{a} #{b}" unless @family.spouse(a) == b
485
486 record = @family.current_record(a)
487 raise CommandError, "BAD YEAR #{a}" if year <= record.start
488
489 record.finish = year
490 record.status = 'divorced'
491 ["DIVORCED #{a} #{b} #{year}"]
492 end

521 next if other == id
522

532 checked(a, b)
533 apex = @family.apex(a, b)
534 return ["PATH #{a} #{b} none"] unless apex
535
536 route = @family.upward_path(a, apex.person) + @family.upward_path(b, apex.person)[0...-1].reverse

540 def common(a, b)
541 checked(a, b)
542 up_a = @family.generation_distances(a, true)
543 up_b = @family.generation_distances(b, true)
544 shared = up_a.keys & up_b.keys
545 nearest = shared.reject { |id| (@family.children(id) & shared).any? }
546 nearest.sort_by! { |id| [up_a[id] + up_b[id], id] }
547 ["COMMON #{a} #{b} #{nearest.length}"] + nearest.map { |id| " #{id} #{up_a[id]} #{up_b[id]}" }
548 end

550 def lineage(id, upward)
551 checked(id)
552 entries = @family.lineage(id, upward)
553 command = upward ? 'ANCESTORS' : 'DESCENDANTS'
554 lines = entries.map do |distance, other|
555 da, db = upward ? [0, distance] : [distance, 0]
556 " #{distance} #{other} #{kinship_term(da, db, @family.sex(other))}"
557 end
558 ["#{command} #{id} #{entries.length}"] + lines
559 end

561 def cousins(id, degree)
562 checked(id)
563 degree = degree.to_i
564 entries = @family.persons.keys.filter_map do |other|
565 next if other == id
566
567 apex = @family.apex(other, id)
568 next unless apex && apex.da >= 2 && apex.db >= 2 && [apex.da, apex.db].min - 1 == degree
569
570 [(apex.da - apex.db).abs, other, apex.blood_term(@family.sex(other))]
571 end.sort
572 ["COUSINS #{id} #{degree} #{entries.length}"] + entries.map { |_, other, term| " #{other} #{term}" }
573 end
574
575 def siblings(id)
576 checked(id)
577 earlier = 0
578 lines = @family.persons.keys.sort_by { |other| @family.age_key(other) }.filter_map do |other|
579 next if other == id
580
581 shared = @family.common_parents(id, other)
582 kind = if shared == 2
583 'full'
584 elsif shared == 1
585 'half'
586 elsif @family.step_related?(@family.parents(id), @family.parents(other))
587 'step'
588 end
589 next unless kind
590
591 if kind != 'step' && (@family.age_key(other) <=> @family.age_key(id)).negative?
592 earlier += 1
593 end
594 " #{other} #{kind} #{@family.persons[other].birth}"
595 end
596 ["SIBLINGS #{id} #{lines.length} order=#{earlier + 1}"] + lines
597 end

599 def generations(id)
600 checked(id)
601 groups = @family.lineage(id, false).group_by(&:first)
602 lines = groups.keys.sort.map do |distance|
603 people = groups[distance].map { |_, other| @family.persons[other] }
604 births = people.map(&:birth)
605 " #{distance} #{people.length} #{people.count { |person| person.death.nil? }} #{births.min}-#{births.max}"
606 end
607 ["GENERATIONS #{id} #{groups.length}"] + lines
608 end

610 def tree(id)
611 checked(id)
612 lines = []
613 visit(id, 0, lines, {})
614 ["TREE #{id} #{@family.lineage(id, false).length}"] + lines
615 end

617 def visit(id, depth, lines, visited)
618 indent = ' ' * (depth + 1)
619 if visited[id]
620 lines << "#{indent}#{id} (see above)"
621 return
622 end
623 visited[id] = true
624 person = @family.persons[id]
625 line = "#{indent}#{id} #{person.name} #{person.life_years}"
626 spouse = @family.spouse(id)
627 line += " = #{spouse}" if spouse
628 lines << line
629 @family.children(id).sort_by { |child| @family.age_key(child) }.each do |child|
630 visit(child, depth + 1, lines, visited)
631 end
632 end

634 def pedigree(id, depth)
635 checked(id)
636 depth = depth.to_i
637 lines = []
638 entered = {}
639 enter(id, 1, 0, depth, lines, entered)
640 ["PEDIGREE #{id} #{depth} entries=#{lines.length} distinct=#{entered.length}"] + lines
641 end

643 def enter(id, number, generation, depth, lines, entered)
644 person = @family.persons[id]
645 lines << "#{' ' * (generation + 1)}#{number} #{id} #{person.name} #{person.life_years}"
646 entered[id] = true
647 return unless generation < depth
648
649 parents = person.parents.sort_by { |parent| [SEX_INDEX[@family.sex(parent)], parent] }
650 parents.each_with_index do |parent, index|
651 enter(parent, 2 * number + index, generation + 1, depth, lines, entered)
652 end
653 end

655 def timeline(id)
656 person = @family.person(id)
657 events = [[person.birth, 0, '', 'born']]
658 @family.records_of(id).each do |record|
659 other = record.other(id)
660 events << [record.start, 1, other, "married #{other}"]
661 if record.status == 'divorced'
662 events << [record.finish, 3, other, "divorced #{other}"]
663 elsif record.status == 'widowed' && record.died == other
664 events << [record.finish, 4, other, "widowed #{other}"]
665 end
666 end
667 children = @family.children(id)
668 children.each { |child| events << [@family.persons[child].birth, 2, child, "child #{child}"] }
669 [['parent', person.parents], ['child', children]].each do |label, relatives|
670 relatives.each do |other|
671 death = @family.persons[other].death
672 next unless death && death >= person.birth && (person.death.nil? || death <= person.death)

674 events << [death, 5, other, "lost #{label} #{other}"]
675 end
676 end
677 events << [person.death, 6, '', 'died'] if person.death
678 events.sort_by! { |event| event[0, 3] }
679 ["TIMELINE #{id} #{events.length}"] + events.map { |year, _, _, text| " #{year} #{text}" }
