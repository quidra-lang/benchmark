38gendered(kind, sex) = TERMS[kind][SEX_INDEX[sex]]
39generation_prefix(g) = g == 1 ? "" : repeat("great-", g - 2) * "grand"

41function ordinal(k)
42 k <= 10 && return ORDINALS[k]
43 suffix = k % 100 in (11, 12, 13) ? "th" :
44 k % 10 == 1 ? "st" : k % 10 == 2 ? "nd" : k % 10 == 3 ? "rd" : "th"
45 return "$k$suffix"
46end

48function kinship_term(da, db, sex)
49 da == 0 && db == 0 && return "self"
50 da == 0 && return generation_prefix(db) * gendered("parent", sex)
51 db == 0 && return generation_prefix(da) * gendered("child", sex)
52 da == 1 && db == 1 && return gendered("sibling", sex)
53 da == 1 && return generation_prefix(db - 1) * gendered("pibling", sex)
54 db == 1 && return generation_prefix(da - 1) * gendered("nibling", sex)
55 r = abs(da - db)
56 removal = r == 0 ? "" : r == 1 ? " once removed" :
57 r == 2 ? " twice removed" : " $r times removed"
58 return ordinal(min(da, db) - 1) * " cousin" * removal
59end

87function person(f, x)
88 haskey(f.persons, x) || fail("UNKNOWN PERSON $x")
89 return f.persons[x]
90end
91parents(f, x) = f.persons[x].parents
92children(f, x) = [y for (y, p) in f.persons if x in p.parents]
93common_parents(f, x, y) = count(p -> p in parents(f, y), parents(f, x))
94are_siblings(f, x, y) = x != y && common_parents(f, x, y) > 0
95siblings(f, x) = [y for y in keys(f.persons) if are_siblings(f, x, y)]
96age_key(f, x) = (f.persons[x].birth, x)
97age_order(f, ids) = sort!(collect(ids); by=x -> age_key(f, x))

99function person_list(f, ids)
100 ordered = age_order(f, ids)
101 return isempty(ordered) ? "-" : join(ordered, ' ')
102end

104records_of(f, x) = [m for m in f.marriages if involves(m, x)]
105function current_record(f, x)
106 for m in f.marriages
107 m.status == "current" && involves(m, x) && return m
108 end
109 return nothing
110end
111function spouse(f, x)
112 m = current_record(f, x)
113 return isnothing(m) ? nothing : other(m, x)
114end
115function last_record(f, a, b)
116 for m in Iterators.reverse(f.marriages)
117 involves(m, a) && involves(m, b) && return m
118 end
119 return nothing
120end
121function partners(f, x)
122 latest = Dict{String,Marriage}()
123 for m in f.marriages
124 involves(m, x) && (latest[other(m, x)] = m)
125 end
126 return Set(y for (y, m) in latest if m.status != "divorced")
127end
128step_related(f, px, py) = any(p -> any(q -> q in partners(f, p), py), px)

130function generation_distances(f, start, upward)
131 distances = Dict(start => 0)
132 queue = [start]
133 next = 1
134 while next <= length(queue)
135 x = queue[next]
136 next += 1
137 for y in (upward ? parents(f, x) : children(f, x))
138 if !haskey(distances, y)
139 distances[y] = distances[x] + 1
140 push!(queue, y)
141 end
142 end
143 end
144 return distances
145end
146lineage(f, x, up) = sort!([(g, y) for (y, g) in generation_distances(f, x, up) if y != x])
147is_ancestor(f, a, b) = a != b && haskey(generation_distances(f, b, true), a)
148grandparents(f, x) = [y for (y, g) in generation_distances(f, x, true) if g == 2]

150struct Apex
151 id::String
152 da::Int
153 db::Int
154 shares::Int
155end
156function apex(f, a, b)
157 ua = generation_distances(f, a, true)
158 ub = generation_distances(f, b, true)
159 haskey(ub, a) && return Apex(a, 0, ub[a], 0)
160 haskey(ua, b) && return Apex(b, ua[b], 0, 0)
161 shared = [c for c in keys(ua) if haskey(ub, c)]
162 isempty(shared) && return nothing
163 c = first(sort!(shared; by=c -> (ua[c] + ub[c], ua[c], c)))
164 da, db = ua[c], ub[c]
165 shares = count(y -> ua[y] == da && ub[y] == db, shared)
166 return Apex(c, da, db, shares)
167end
168function blood_term(ap, sex)
169 prefix = ap.shares == 1 ? "half-" : ap.shares >= 4 ? "double " : ""
170 return prefix * kinship_term(ap.da, ap.db, sex)
171end

173function relationship(f, a, b, sex=f.persons[a].sex)
174 a == b && return ["self"]
175 terms = String[]
176 ap = apex(f, a, b)
177 !isnothing(ap) && push!(terms, blood_term(ap, sex))
178 m = last_record(f, a, b)
179 if isnothing(m)
180 any(p -> a in p.parents && b in p.parents, values(f.persons)) && push!(terms, "co-parent")
181 elseif m.status == "current"
182 push!(terms, gendered("spouse", sex))
183 elseif m.status == "widowed"
184 push!(terms, gendered(m.died == a ? "late" : "survivor", sex))
185 else
186 push!(terms, gendered("ex", sex))
187 end
188 pa, pb = parents(f, a), parents(f, b)
189 qa, qb = partners(f, a), partners(f, b)
190 a ∉ pb && any(p -> p in qa, pb) && push!(terms, gendered("stepparent", sex))
191 b ∉ pa && any(p -> p in qb, pa) && push!(terms, gendered("stepchild", sex))
192 common_parents(f, a, b) == 0 && step_related(f, pa, pb) &&
193 push!(terms, gendered("stepsibling", sex))
194 !is_ancestor(f, a, b) && any(g -> g in qa, grandparents(f, b)) &&
195 push!(terms, gendered("stepgrandparent", sex))
196 !is_ancestor(f, b, a) && any(g -> g in qb, grandparents(f, a)) &&
197 push!(terms, gendered("stepgrandchild", sex))
198 any(q -> a in parents(f, q), qb) && push!(terms, gendered("parent-in-law", sex))
199 any(q -> b in parents(f, q), qa) && push!(terms, gendered("child-in-law", sex))
200 if any(q -> are_siblings(f, a, q), qb) || any(q -> are_siblings(f, q, b), qa)
201 push!(terms, gendered("sibling-in-law", sex))
202 end
203 return terms
204end
205function relationship_text(f, a, b)
206 terms = relationship(f, a, b)
207 return isempty(terms) ? "no relation" : join(terms, ", ")
208end

210function kinship(f, a, b, memo)
211 key = a <= b ? (a, b) : (b, a)
212 haskey(memo, key) && return memo[key]
213 if a == b
214 value = (SCALE + inbreeding(f, a, memo)) ÷ 2
215 else
216 x, y = f.persons[a].birth >= f.persons[b].birth ? (a, b) : (b, a)
217 total = 0
218 for p in parents(f, x)
219 total += kinship(f, p, y, memo)
220 end
221 value = total ÷ 2
222 end
223 memo[key] = value
224 return value
225end
226function inbreeding(f, x, memo)
227 ps = parents(f, x)
228 return length(ps) == 2 ? kinship(f, ps[1], ps[2], memo) : 0
229end
230coefficient_memo() = Dict{Tuple{String,String},Int}()
231function fraction_text(value)
232 d = gcd(value, SCALE)
233 return string(value ÷ d, '/', SCALE ÷ d)
234end
235function lineage_length(f, x, memo)
236 if !haskey(memo, x)
237 height = 0
238 for p in parents(f, x)
239 height = max(height, lineage_length(f, p, memo))
240 end
241 memo[x] = height + 1
242 end
243 return memo[x]
244end
245function upward_path(f, start, depths)
246 path = [start]
247 while depths[last(path)] > 0
248 x = last(path)
249 candidates = [p for p in parents(f, x) if haskey(depths, p) && depths[p] == depths[x] - 1]
250 push!(path, minimum(candidates))
251 end
252 return path
253end
254function census_status(f, x, year)
255 last = nothing
256 for m in records_of(f, x)
257 m.start <= year && (last = m)
258 end
259 isnothing(last) && return "single"
260 if isnothing(last.finish) || last.finish > year
261 return "married $(other(last, x))"
262 end
263 return last.status
264end
265function step_targets(f, x, kind)
266 kind == "parent" && return parents(f, x)
267 kind == "child" && return children(f, x)
268 kind == "sibling" && return siblings(f, x)
269 s = spouse(f, x)
270 return isnothing(s) ? String[] : [s]
271end

273function tree_visit!(lines, visited, f, x, depth)
274 indent = repeat(" ", depth + 1)
275 if x in visited
276 push!(lines, "$indent$x (see above)")
277 return
278 end
279 push!(visited, x)
280 p = f.persons[x]
281 s = spouse(f, x)
282 push!(lines, "$indent$x $(p.name) $(life_years(p))" * (isnothing(s) ? "" : " = $s"))
283 for child in age_order(f, children(f, x))
284 tree_visit!(lines, visited, f, child, depth + 1)
285 end
286end
287function pedigree_enter!(lines, entered, f, x, k, g, depth)
288 p = f.persons[x]
289 push!(lines, "$(repeat(" ", g + 1))$k $x $(p.name) $(life_years(p))")
290 push!(entered, x)
291 if g < depth
292 ps = sort(parents(f, x); by=y -> (SEX_INDEX[f.persons[y].sex], y))
293 for (i, parent) in enumerate(ps)
294 pedigree_enter!(lines, entered, f, parent, 2k + i - 1, g + 1, depth)
295 end
296 end

301 history::Vector{Tuple{String,Family}}

303Calculator() = Calculator(Family(), Tuple{String,Family}[])

305function process!(calc, cmd, args)
306 f = calc.family
307 lines = String[]
308 if cmd == "PERSON"
309 x, sex, birth, name = args
310 haskey(f.persons, x) && fail("DUPLICATE PERSON $x")
311 f.persons[x] = Person(sex, parse(Int, birth), name, nothing, String[])
312 push!(lines, "ADDED $x $name $sex $birth")
313 elseif cmd == "DEATH"
314 x = args[1]
315 year = parse(Int, args[2])
316 p = person(f, x)
317 !isnothing(p.death) && fail("ALREADY DECEASED $x")
318 year < p.birth && fail("BAD YEAR $x")
319 for m in records_of(f, x)
320 (year < m.start || (!isnothing(m.finish) && year < m.finish)) && fail("BAD YEAR $x")
321 end
322 late = sort!([c for c in children(f, x) if f.persons[c].birth > year + 1])
323 !isempty(late) && fail("AFTER DEATH $(first(late)) $x")
324 p.death = year
325 m = current_record(f, x)
326 if isnothing(m)
327 push!(lines, "DIED $x $year")
328 else
329 m.finish, m.status, m.died = year, "widowed", x
330 push!(lines, "DIED $x $year spouse=$(other(m, x))")
331 end
332 elseif cmd == "REMOVE"
333 x = args[1]
334 person(f, x)
335 for p in values(f.persons)
336 filter!(y -> y != x, p.parents)
337 end
338 filter!(m -> !involves(m, x), f.marriages)
339 delete!(f.persons, x)
340 push!(lines, "REMOVED $x")
341 elseif cmd == "PARENT"
342 child, parent = args
343 c, p = person(f, child), person(f, parent)
344 child == parent && fail("SAME PERSON $child")
345 parent in c.parents && fail("ALREADY PARENT $child $parent")
346 length(c.parents) == 2 && fail("TOO MANY PARENTS $child")
347 is_ancestor(f, child, parent) && fail("CYCLE $child $parent")
348 p.birth + 12 > c.birth && fail("AGE GAP $child $parent")
349 !isnothing(p.death) && p.death + 1 < c.birth && fail("AFTER DEATH $child $parent")
350 push!(c.parents, parent)
351 push!(lines, "LINKED $child $parent $(gendered("parent", p.sex))")
352 elseif cmd == "UNPARENT"
353 child, parent = args
354 c = person(f, child)
355 person(f, parent)
356 parent ∉ c.parents && fail("NOT PARENT $child $parent")
357 filter!(p -> p != parent, c.parents)
358 push!(lines, "UNLINKED $child $parent")
359 elseif cmd == "SPOUSE"
360 a, b = args[1:2]
361 year = parse(Int, args[3])
362 person(f, a)
363 person(f, b)
364 a == b && fail("SAME PERSON $a")
365 for x in (a, b)
366 !isnothing(f.persons[x].death) && fail("DECEASED $x")
367 end
368 for x in (a, b)
369 !isnothing(spouse(f, x)) && fail("ALREADY MARRIED $x")
370 end
371 (is_ancestor(f, a, b) || is_ancestor(f, b, a) || are_siblings(f, a, b)) &&
372 fail("CLOSE KIN $a $b")
373 for x in (a, b)
374 year < f.persons[x].birth + 16 && fail("TOO YOUNG $x")
375 end
376 for x in (a, b)
377 any(m -> !isnothing(m.finish) && m.finish >= year, records_of(f, x)) && fail("BAD YEAR $x")
378 end
379 push!(f.marriages, Marriage(a, b, year, nothing, "current", nothing))
380 push!(lines, "MARRIED $a $b $year")
381 elseif cmd == "DIVORCE"
382 a, b = args[1:2]
383 year = parse(Int, args[3])
384 person(f, a)
385 person(f, b)
386 spouse(f, a) != b && fail("NOT MARRIED $a $b")
387 m = current_record(f, a)
388 year <= m.start && fail("BAD YEAR $a")
389 m.finish, m.status = year, "divorced"
390 push!(lines, "DIVORCED $a $b $year")
391 elseif cmd == "UNDO"
392 isempty(calc.history) && fail("NOTHING TO UNDO")
393 line, previous = pop!(calc.history)
394 calc.family = previous
395 push!(lines, "UNDONE $line")
396 elseif cmd == "INFO"
397 x = args[1]
398 p = person(f, x)
399 push!(lines, "INFO $x $(p.name) $(p.sex) $(life_years(p))",
400 " parents: $(person_list(f, p.parents))")
401 for m in records_of(f, x)
402 finish = isnothing(m.finish) ? "" : string(m.finish)
403 push!(lines, " marriage: $(other(m, x)) $(m.start)-$finish $(m.status)")
404 end
405 push!(lines, " children: $(person_list(f, children(f, x)))",
406 " siblings: $(person_list(f, siblings(f, x)))",
407 " inbreeding: $(fraction_text(inbreeding(f, x, coefficient_memo())))")
408 elseif cmd == "RELATION" || cmd == "KINSHIP" || cmd == "PATH" || cmd == "COMMON"
409 a, b = args
410 person(f, a)
411 person(f, b)
412 if cmd == "RELATION"
413 push!(lines, "RELATION $a $b: $(relationship_text(f, a, b))")
414 elseif cmd == "KINSHIP"
415 push!(lines, "KINSHIP $a $b $(fraction_text(kinship(f, a, b, coefficient_memo())))")
416 elseif cmd == "PATH"
417 ap = apex(f, a, b)
418 if isnothing(ap)
419 push!(lines, "PATH $a $b none")
420 else
421 depths = generation_distances(f, ap.id, false)
422 left, right = upward_path(f, a, depths), upward_path(f, b, depths)
423 route = vcat(left, reverse(right[1:end-1]))
424 push!(lines, "PATH $a $b $(ap.da + ap.db): $(join(route, ' '))")
425 end
426 else
427 ua, ub = generation_distances(f, a, true), generation_distances(f, b, true)
428 shared = Set(c for c in keys(ua) if haskey(ub, c))
429 nearest = [c for c in shared if !any(child -> child in shared, children(f, c))]
430 sort!(nearest; by=c -> (ua[c] + ub[c], c))
431 push!(lines, "COMMON $a $b $(length(nearest))")
432 for c in nearest
433 push!(lines, " $c $(ua[c]) $(ub[c])")
434 end
435 end
436 elseif cmd == "RELATIVES" || cmd == "WHO"
437 x = args[1]
438 person(f, x)
439 query = cmd == "WHO" ? join(args[2:end], ' ') : ""
440 entries = String[]
441 for y in sort!(collect(keys(f.persons)))
442 y == x && continue
443 terms = relationship(f, y, x)
444 if cmd == "RELATIVES"
445 !isempty(terms) && push!(entries, " $y: $(join(terms, ", "))")
446 else
447 neutral = relationship(f, y, x, "X")
448 for i in eachindex(terms)
449 if query == terms[i] || query == neutral[i]
450 push!(entries, " $y: $(terms[i])")
451 break
452 end
453 end
454 end
455 end
456 push!(lines, cmd == "WHO" ? "WHO $x $query $(length(entries))" : "RELATIVES $x $(length(entries))")
457 append!(lines, entries)
458 elseif cmd == "ANCESTORS" || cmd == "DESCENDANTS"
459 x = args[1]
460 person(f, x)
461 entries = lineage(f, x, cmd == "ANCESTORS")
462 push!(lines, "$cmd $x $(length(entries))")
463 for (g, y) in entries
464 term = cmd == "ANCESTORS" ? kinship_term(0, g, f.persons[y].sex) : kinship_term(g, 0, f.persons[y].sex)
465 push!(lines, " $g $y $term")
466 end
467 elseif cmd == "COUSINS"
468 x = args[1]
469 degree = parse(Int, args[2])
470 person(f, x)
471 matches = Tuple{Int,String,String}[]
472 for y in keys(f.persons)
473 y == x && continue
474 ap = apex(f, y, x)
475 if !isnothing(ap) && min(ap.da, ap.db) >= 2 && min(ap.da, ap.db) - 1 == degree
476 push!(matches, (abs(ap.da - ap.db), y, blood_term(ap, f.persons[y].sex)))
477 end
478 end
479 sort!(matches)
480 push!(lines, "COUSINS $x $degree $(length(matches))")
481 for (_, y, term) in matches
482 push!(lines, " $y $term")
483 end
484 elseif cmd == "SIBLINGS"
485 x = args[1]
486 person(f, x)
487 entries = String[]
488 earlier = 0
489 for y in age_order(f, keys(f.persons))
490 y == x && continue
491 shared = common_parents(f, x, y)
492 kind = shared == 2 ? "full" : shared == 1 ? "half" :
493 step_related(f, parents(f, x), parents(f, y)) ? "step" : ""
494 isempty(kind) && continue
495 if kind != "step" && isless(age_key(f, y), age_key(f, x))
496 earlier += 1
497 end
498 push!(entries, " $y $kind $(f.persons[y].birth)")
499 end
500 push!(lines, "SIBLINGS $x $(length(entries)) order=$(earlier + 1)")
501 append!(lines, entries)
502 elseif cmd == "GENERATIONS"
503 x = args[1]
504 person(f, x)
505 groups = Dict{Int,Vector{Person}}()
506 for (g, y) in lineage(f, x, false)
507 push!(get!(groups, g, Person[]), f.persons[y])
508 end
509 push!(lines, "GENERATIONS $x $(length(groups))")
510 for g in sort!(collect(keys(groups)))
511 members = groups[g]
512 births = [p.birth for p in members]
513 living = count(p -> isnothing(p.death), members)
514 push!(lines, " $g $(length(members)) $living $(minimum(births))-$(maximum(births))")
515 end
516 elseif cmd == "TREE"
517 x = args[1]
518 person(f, x)
519 push!(lines, "TREE $x $(length(lineage(f, x, false)))")
520 tree_visit!(lines, Set{String}(), f, x, 0)
521 elseif cmd == "PEDIGREE"
522 x = args[1]
523 depth = parse(Int, args[2])
524 person(f, x)
525 entries, entered = String[], Set{String}()
526 pedigree_enter!(entries, entered, f, x, 1, 0, depth)
527 push!(lines, "PEDIGREE $x $depth entries=$(length(entries)) distinct=$(length(entered))")
528 append!(lines, entries)
529 elseif cmd == "TIMELINE"
530 x = args[1]
531 p = person(f, x)
532 events = Tuple{Int,Int,String,String}[(p.birth, 0, "", "born")]
533 for m in records_of(f, x)
534 q = other(m, x)
535 push!(events, (m.start, 1, q, "married $q"))
536 if m.status == "divorced"
537 push!(events, (m.finish, 3, q, "divorced $q"))
538 elseif m.status == "widowed" && m.died == q
539 push!(events, (m.finish, 4, q, "widowed $q"))
540 end
541 end
542 cs = children(f, x)
543 for c in cs
544 push!(events, (f.persons[c].birth, 2, c, "child $c"))
545 end
546 for (label, relatives) in (("parent", p.parents), ("child", cs))
547 for y in relatives
548 death = f.persons[y].death
549 if !isnothing(death) && death >= p.birth && (isnothing(p.death) || death <= p.death)
550 push!(events, (death, 5, y, "lost $label $y"))
551 end
552 end
553 end
554 !isnothing(p.death) && push!(events, (p.death, 6, "", "died"))
555 sort!(events; by=e -> (e[1], e[2], e[3]))
556 push!(lines, "TIMELINE $x $(length(events))")
557 for (year, _, _, text) in events
558 push!(lines, " $year $text")
559 end
560 elseif cmd == "CENSUS"
561 year = parse(Int, args[1])
562 residents = [x for (x, p) in f.persons if p.birth <= year && (isnothing(p.death) || p.death > year)]
563 push!(lines, "CENSUS $year $(length(residents))")
564 for x in age_order(f, residents)
565 push!(lines, " $x $(year - f.persons[x].birth) $(census_status(f, x, year))")
566 end
567 elseif cmd == "FIND"
568 x, route = args
569 person(f, x)
570 current = Set([x])
571 for step in split(route, '.')
572 kind, sex = STEPS[step]
573 following = Set{String}()
574 for y in current, z in step_targets(f, y, kind)
575 (isnothing(sex) || f.persons[z].sex == sex) && push!(following, z)
576 end
577 current = following
578 end
579 delete!(current, x)
580 push!(lines, "FIND $x $route $(length(current))")
581 for y in sort!(collect(current))
582 push!(lines, " $y: $(relationship_text(f, y, x))")
583 end
584 elseif cmd == "TERM"
585 da, db = parse(Int, args[1]), parse(Int, args[2])
586 sex = args[3]
587 push!(lines, "TERM $da $db $sex $(kinship_term(da, db, sex))")
588 elseif cmd == "STATS"
589 living = count(p -> isnothing(p.death), values(f.persons))
590 links = sum((length(p.parents) for p in values(f.persons)); init=0)
591 couples = count(m -> m.status == "current", f.marriages)
592 founders = count(p -> isempty(p.parents), values(f.persons))
593 memo = Dict{String,Int}()
594 longest = maximum((lineage_length(f, x, memo) for x in keys(f.persons)); init=0)
595 push!(lines, "STATS persons=$(length(f.persons)) living=$living links=$links couples=$couples founders=$founders generations=$longest")
596 elseif cmd == "AUDIT"
597 entries = String[]
598 for m in f.marriages
599 ap = apex(f, m.first, m.second)
600 !isnothing(ap) && push!(entries, " marriage $(m.first) $(m.second) $(blood_term(ap, f.persons[m.first].sex))")
601 end
602 memo = coefficient_memo()
603 for x in sort!(collect(keys(f.persons)))
604 value = inbreeding(f, x, memo)
605 value != 0 && push!(entries, " inbred $x $(fraction_text(value))")
606 end
607 push!(lines, "AUDIT $(length(entries))")
608 append!(lines, entries)
609 end
610 return lines
611end

613function execute!(calc, line)
614 words = String.(split(line, ' '))
615 cmd, args = first(words), words[2:end]
616 try
617 saved = cmd in MODIFYING ? deepcopy(calc.family) : nothing
618 lines = process!(calc, cmd, args)
619 cmd in MODIFYING && push!(calc.history, (line, saved))
620 return lines
621 catch err
622 err isa CommandError || rethrow()
623 return ["ERROR $(err.reason)"]
624 end
625end
