3type Sex = 'M' | 'F' | 'X';
4type StepKind = 'parent' | 'child' | 'sibling' | 'spouse';

33const SCALE = 2 ** 37;
34const SEX_INDEX: Record<Sex, number> = { M: 0, F: 1, X: 2 };
35const ORDINAL_SUFFIXES: Record<number, string> = { 1: 'st', 2: 'nd', 3: 'rd' };
36const ORDINALS = ['first', 'second', 'third', 'fourth', 'fifth',
37 'sixth', 'seventh', 'eighth', 'ninth', 'tenth'];
38const TERMS = {
39 parent: ['father', 'mother', 'parent'],
40 child: ['son', 'daughter', 'child'],
41 sibling: ['brother', 'sister', 'sibling'],
42 pibling: ['uncle', 'aunt', 'pibling'],
43 nibling: ['nephew', 'niece', 'nibling'],
44 spouse: ['husband', 'wife', 'spouse'],
45 late: ['late husband', 'late wife', 'late spouse'],
46 survivor: ['widower', 'widow', 'surviving spouse'],
47 ex: ['ex-husband', 'ex-wife', 'ex-spouse'],
48 stepparent: ['stepfather', 'stepmother', 'stepparent'],
49 stepchild: ['stepson', 'stepdaughter', 'stepchild'],
50 stepsibling: ['stepbrother', 'stepsister', 'stepsibling'],
51 stepgrandparent: ['stepgrandfather', 'stepgrandmother', 'stepgrandparent'],
52 stepgrandchild: ['stepgrandson', 'stepgranddaughter', 'stepgrandchild'],
53 parentInLaw: ['father-in-law', 'mother-in-law', 'parent-in-law'],
54 childInLaw: ['son-in-law', 'daughter-in-law', 'child-in-law'],
55 siblingInLaw: ['brother-in-law', 'sister-in-law', 'sibling-in-law'],
56};
57const STEPS: Record<string, [StepKind, Sex | null]> = {
58 parent: ['parent', null], father: ['parent', 'M'], mother: ['parent', 'F'],
59 child: ['child', null], son: ['child', 'M'], daughter: ['child', 'F'],
60 sibling: ['sibling', null], brother: ['sibling', 'M'], sister: ['sibling', 'F'],
61 spouse: ['spouse', null], husband: ['spouse', 'M'], wife: ['spouse', 'F'],
62};
63const MODIFYING = new Set(['PERSON', 'DEATH', 'REMOVE', 'PARENT', 'UNPARENT', 'SPOUSE', 'DIVORCE']);
64
65class CommandError extends Error {}

67function compareIds(a: string, b: string) {
68 return a < b ? -1 : a > b ? 1 : 0;
69}
70
71function gendered(kind: keyof typeof TERMS, sex: Sex) {
72 return TERMS[kind][SEX_INDEX[sex]];
73}
74
75function generationPrefix(g: number) {
76 return g === 1 ? '' : 'great-'.repeat(g - 2) + 'grand';
77}
78
79function term(da: number, db: number, sex: Sex) {
80 if (da === 0 && db === 0) return 'self';
81 if (da === 0) return generationPrefix(db) + gendered('parent', sex);
82 if (db === 0) return generationPrefix(da) + gendered('child', sex);
83 if (da === 1 && db === 1) return gendered('sibling', sex);
84 if (da === 1) return generationPrefix(db - 1) + gendered('pibling', sex);
85 if (db === 1) return generationPrefix(da - 1) + gendered('nibling', sex);
86 const k = Math.min(da, db) - 1;
87 const suffix = [11, 12, 13].includes(k % 100) ? 'th'
88 : (ORDINAL_SUFFIXES[k % 10] ?? 'th');
89 const ordinal = k <= 10 ? ORDINALS[k - 1] : `${k}${suffix}`;
90 const r = Math.abs(da - db);
91 const removed = r === 0 ? '' : r === 1 ? ' once removed'
92 : r === 2 ? ' twice removed' : ` ${r} times removed`;
93 return `${ordinal} cousin${removed}`;
94}
95
96function bloodTerm(apex: Apex, sex: Sex) {
97 const prefix = apex.shares === 1 ? 'half-' : apex.shares >= 4 ? 'double ' : '';
98 return prefix + term(apex.da, apex.db, sex);
99}
100
101function life(person: Person) {
102 return `${person.birth}-${person.death ?? ''}`;
103}
104
105function fraction(value: number) {
106 let a = value;
107 let b = SCALE;
108 while (b !== 0) [a, b] = [b, a % b];
109 return `${value / a}/${SCALE / a}`;
110}
111
112function contains(record: Marriage, x: string) {
113 return record.first === x || record.second === x;
114}
115
116function other(record: Marriage, x: string) {
117 return record.first === x ? record.second : record.first;
118}

120class Family {
121 persons = new Map<string, Person>();
122 marriages: Marriage[] = [];
123 distanceCache = new Map<string, Map<string, number>>();
124 coefficientCache = new Map<string, number>();
125
126 copy() {
127 const family = new Family();
128 family.persons = new Map<string, Person>([...this.persons].map(([x, p]) =>
129 [x, { ...p, parents: [...p.parents] }]));
130 family.marriages = this.marriages.map(m => ({ ...m }));
131 return family;
132 }
133
134 invalidate() {
135 this.distanceCache.clear();
136 this.coefficientCache.clear();
137 }
138
139 person(x: string) {
140 const person = this.persons.get(x);
141 if (!person) throw new CommandError(`UNKNOWN PERSON ${x}`);
142 return person;
143 }
144
145 ageCompare(a: string, b: string) {
146 return this.person(a).birth - this.person(b).birth || compareIds(a, b);
147 }
148
149 ageOrder(ids: Iterable<string>) {
150 return [...ids].sort((a, b) => this.ageCompare(a, b));
151 }
152
153 personList(ids: Iterable<string>) {
154 return this.ageOrder(ids).join(' ') || '-';
155 }
156
157 parents(x: string) {
158 return this.person(x).parents;
159 }
160
161 children(x: string) {
162 return [...this.persons.keys()].filter(y => this.parents(y).includes(x));
163 }
164
165 commonParents(a: string, b: string) {
166 return this.parents(a).filter(p => this.parents(b).includes(p)).length;
167 }
168
169 areSiblings(a: string, b: string) {
170 return a !== b && this.commonParents(a, b) > 0;
171 }
172
173 siblings(x: string) {
174 return [...this.persons.keys()].filter(y => this.areSiblings(x, y));
175 }
176
177 records(x: string) {
178 return this.marriages.filter(m => contains(m, x));
179 }
180
181 current(x: string) {
182 return this.marriages.find(m => m.status === 'current' && contains(m, x));
183 }
184
185 spouse(x: string) {
186 const record = this.current(x);
187 return record ? other(record, x) : null;
188 }
189
190 last(a: string, b: string) {
191 return this.marriages.findLast(m => contains(m, a) && contains(m, b));
192 }
193
194 partners(x: string) {
195 const latest = new Map(this.records(x).map(m => [other(m, x), m]));
196 return new Set([...latest].filter(([, m]) => m.status !== 'divorced').map(([y]) => y));
197 }
198
199 stepRelated(a: string, b: string) {
200 return this.parents(a).some(p => this.parents(b).some(q => this.partners(p).has(q)));
201 }
202
203 distances(start: string, upward: boolean): Map<string, number> {
204 const key = `${upward}:${start}`;
205 if (this.distanceCache.has(key)) return this.distanceCache.get(key)!;
206 const distances = new Map([[start, 0]]);
207 const queue = [start];
208 for (let i = 0; i < queue.length; ++i) {
209 const x = queue[i];
210 for (const y of upward ? this.parents(x) : this.children(x)) {
211 if (!distances.has(y)) {
212 distances.set(y, distances.get(x)! + 1);
213 queue.push(y);
214 }
215 }
216 }
217 this.distanceCache.set(key, distances);
218 return distances;
219 }
220
221 ancestor(a: string, b: string) {
222 return a !== b && this.distances(b, true).has(a);
223 }
224
225 grandparents(x: string) {
226 return [...this.distances(x, true)].filter(([, g]) => g === 2).map(([y]) => y);
227 }
228
229 apex(a: string, b: string): Apex | null {
230 const upA = this.distances(a, true);
231 const upB = this.distances(b, true);
232 if (upB.has(a)) return { person: a, da: 0, db: upB.get(a)!, shares: 0 };
233 if (upA.has(b)) return { person: b, da: upA.get(b)!, db: 0, shares: 0 };
234 const common = [...upA.keys()].filter(c => upB.has(c));
235 common.sort((c, d) => (upA.get(c)! + upB.get(c)!) - (upA.get(d)! + upB.get(d)!)
236 || upA.get(c)! - upA.get(d)! || compareIds(c, d));
237 if (common.length === 0) return null;
238 const person = common[0];
239 const da = upA.get(person)!;
240 const db = upB.get(person)!;
241 const shares = common.filter(c => upA.get(c)! === da && upB.get(c)! === db).length;
242 return { person, da, db, shares };
243 }
244
245 relationship(a: string, b: string, sex: Sex = this.person(a).sex) {
246 if (a === b) return ['self'];
247 const terms = [];
248 const apex = this.apex(a, b);
249 if (apex) terms.push(bloodTerm(apex, sex));
250 const record = this.last(a, b);
251 if (!record) {
252 if ([...this.persons.values()].some(p => p.parents.includes(a) && p.parents.includes(b)))
253 terms.push('co-parent');
254 } else if (record.status === 'current') terms.push(gendered('spouse', sex));
255 else if (record.status === 'widowed')
256 terms.push(gendered(record.died === a ? 'late' : 'survivor', sex));
257 else terms.push(gendered('ex', sex));
258 const pa = this.parents(a);
259 const pb = this.parents(b);
260 const qa = this.partners(a);
261 const qb = this.partners(b);
262 if (!pb.includes(a) && pb.some(p => qa.has(p))) terms.push(gendered('stepparent', sex));
263 if (!pa.includes(b) && pa.some(p => qb.has(p))) terms.push(gendered('stepchild', sex));
264 if (this.commonParents(a, b) === 0 && this.stepRelated(a, b))
265 terms.push(gendered('stepsibling', sex));
266 if (!this.ancestor(a, b) && this.grandparents(b).some(g => qa.has(g)))
267 terms.push(gendered('stepgrandparent', sex));
268 if (!this.ancestor(b, a) && this.grandparents(a).some(g => qb.has(g)))
269 terms.push(gendered('stepgrandchild', sex));
270 if ([...qb].some(q => this.parents(q).includes(a))) terms.push(gendered('parentInLaw', sex));
271 if ([...qa].some(q => this.parents(q).includes(b))) terms.push(gendered('childInLaw', sex));
272 if ([...qb].some(q => this.areSiblings(a, q)) || [...qa].some(q => this.areSiblings(q, b)))
273 terms.push(gendered('siblingInLaw', sex));
274 return terms;
275 }
276
277 relationshipText(a: string, b: string) {
278 return this.relationship(a, b).join(', ') || 'no relation';
279 }
280
281 coefficient(a: string, b: string): number {
282 const key = [a, b].sort(compareIds).join(':');
283 if (this.coefficientCache.has(key)) return this.coefficientCache.get(key)!;
284 let value: number;
285 if (a === b) value = (SCALE + this.inbreeding(a)) / 2;
286 else {
287 const [x, y] = this.person(a).birth >= this.person(b).birth ? [a, b] : [b, a];
288 value = this.parents(x).reduce((sum, p) => sum + this.coefficient(p, y), 0) / 2;
289 }
290 this.coefficientCache.set(key, value);
291 return value;
292 }
293
294 inbreeding(x: string): number {
295 const parents = this.parents(x);
296 return parents.length === 2 ? this.coefficient(parents[0], parents[1]) : 0;
297 }
298
299 lineageLength(x: string, memo: Map<string, number>): number {
300 if (!memo.has(x)) memo.set(x, 1 + Math.max(0, ...this.parents(x).map(p => this.lineageLength(p, memo))));
301 return memo.get(x)!;
302 }
303}

305class Calculator {
306 family = new Family();
307 history: { line: string; family: Family }[] = [];
308
309 execute(line: string): string[] {
310 const [command, ...args] = line.split(' ');
311 const saved = MODIFYING.has(command) ? this.family.copy() : null;
312 try {
313 const output = this.dispatch(command, args);
314 if (saved) {
315 this.history.push({ line, family: saved });
316 this.family.invalidate();
317 }
318 return output;
319 } catch (error) {
320 if (!(error instanceof CommandError)) throw error;
321 return [`ERROR ${error.message}`];
322 }
323 }
324
325 dispatch(command: string, args: string[]): string[] {
326 const f = this.family;
327 const [x, y] = args;
328 const year = Number(args[2]);
329 switch (command) {
330 case 'PERSON': {
331 if (f.persons.has(x)) throw new CommandError(`DUPLICATE PERSON ${x}`);
332 f.persons.set(x, { sex: y as Sex, birth: year, name: args[3], death: null, parents: [] });
333 return [`ADDED ${x} ${args[3]} ${y} ${year}`];
334 }
335 case 'DEATH': {
336 const p = f.person(x);
337 const death = Number(y);
338 if (p.death !== null) throw new CommandError(`ALREADY DECEASED ${x}`);
339 if (death < p.birth || f.records(x).some(m => death < m.start || (m.end !== null && death < m.end)))
340 throw new CommandError(`BAD YEAR ${x}`);
341 const late = f.children(x).filter(c => f.person(c).birth > death + 1).sort(compareIds);
342 if (late.length) throw new CommandError(`AFTER DEATH ${late[0]} ${x}`);
343 p.death = death;
344 const record = f.current(x);
345 if (!record) return [`DIED ${x} ${death}`];
346 record.end = death;
347 record.status = 'widowed';
348 record.died = x;
349 return [`DIED ${x} ${death} spouse=${other(record, x)}`];
350 }
351 case 'REMOVE':
352 f.person(x);
353 for (const p of f.persons.values()) p.parents = p.parents.filter(p => p !== x);
354 f.marriages = f.marriages.filter(m => !contains(m, x));
355 f.persons.delete(x);
356 return [`REMOVED ${x}`];
357 case 'PARENT': {
358 const child = f.person(x);
359 const parent = f.person(y);
360 if (x === y) throw new CommandError(`SAME PERSON ${x}`);
361 if (child.parents.includes(y)) throw new CommandError(`ALREADY PARENT ${x} ${y}`);
362 if (child.parents.length === 2) throw new CommandError(`TOO MANY PARENTS ${x}`);
363 if (f.ancestor(x, y)) throw new CommandError(`CYCLE ${x} ${y}`);
364 if (parent.birth + 12 > child.birth) throw new CommandError(`AGE GAP ${x} ${y}`);
365 if (parent.death !== null && parent.death + 1 < child.birth)
366 throw new CommandError(`AFTER DEATH ${x} ${y}`);
367 child.parents.push(y);
368 return [`LINKED ${x} ${y} ${gendered('parent', parent.sex)}`];
369 }
370 case 'UNPARENT': {
371 const child = f.person(x);
372 f.person(y);
373 if (!child.parents.includes(y)) throw new CommandError(`NOT PARENT ${x} ${y}`);
374 child.parents = child.parents.filter(p => p !== y);
375 return [`UNLINKED ${x} ${y}`];
376 }
377 case 'SPOUSE': {
378 f.person(x);
379 f.person(y);
380 if (x === y) throw new CommandError(`SAME PERSON ${x}`);
381 for (const a of [x, y]) if (f.person(a).death !== null) throw new CommandError(`DECEASED ${a}`);
382 for (const a of [x, y]) if (f.spouse(a) !== null) throw new CommandError(`ALREADY MARRIED ${a}`);
383 if (f.ancestor(x, y) || f.ancestor(y, x) || f.areSiblings(x, y))
384 throw new CommandError(`CLOSE KIN ${x} ${y}`);
385 for (const a of [x, y]) if (year < f.person(a).birth + 16) throw new CommandError(`TOO YOUNG ${a}`);
386 for (const a of [x, y])
387 if (f.records(a).some(m => m.end !== null && m.end >= year)) throw new CommandError(`BAD YEAR ${a}`);
388 f.marriages.push({ first: x, second: y, start: year, end: null, status: 'current', died: null });
389 return [`MARRIED ${x} ${y} ${year}`];
390 }
391 case 'DIVORCE': {
392 f.person(x);
393 f.person(y);
394 if (f.spouse(x) !== y) throw new CommandError(`NOT MARRIED ${x} ${y}`);
395 const record = f.current(x)!;
396 if (year <= record.start) throw new CommandError(`BAD YEAR ${x}`);
397 record.end = year;
398 record.status = 'divorced';
399 return [`DIVORCED ${x} ${y} ${year}`];
400 }
401 case 'UNDO': {
402 const saved = this.history.pop();
403 if (!saved) throw new CommandError('NOTHING TO UNDO');
404 this.family = saved.family;
405 return [`UNDONE ${saved.line}`];
406 }
407 case 'TERM':
408 return [`TERM ${x} ${y} ${args[2]} ${term(Number(x), Number(y), args[2] as Sex)}`];
409 case 'STATS': {
410 const people = [...f.persons.values()];
411 const memo = new Map<string, number>();
412 return [`STATS persons=${people.length} living=${people.filter(p => p.death === null).length}`
413 + ` links=${people.reduce((sum, p) => sum + p.parents.length, 0)}`
414 + ` couples=${f.marriages.filter(m => m.status === 'current').length}`
415 + ` founders=${people.filter(p => p.parents.length === 0).length}`
416 + ` generations=${Math.max(0, ...[...f.persons.keys()].map(a => f.lineageLength(a, memo)))}`];
417 }
418 case 'AUDIT': {
419 const lines: string[] = [];
420 for (const m of f.marriages) {
421 const apex = f.apex(m.first, m.second);
422 if (apex) lines.push(` marriage ${m.first} ${m.second} ${bloodTerm(apex, f.person(m.first).sex)}`);
423 }
424 for (const a of [...f.persons.keys()].sort(compareIds))
425 if (f.inbreeding(a) !== 0) lines.push(` inbred ${a} ${fraction(f.inbreeding(a))}`);
426 return [`AUDIT ${lines.length}`, ...lines];
427 }
428 case 'CENSUS': {
429 const censusYear = Number(x);
430 const residents = f.ageOrder([...f.persons.keys()].filter(a => {
431 const p = f.person(a);
432 return p.birth <= censusYear && (p.death === null || p.death > censusYear);
433 }));
434 const lines = residents.map(a => {
435 const m = f.records(a).findLast(m => m.start <= censusYear);
436 const status = !m ? 'single' : m.end === null || m.end > censusYear
437 ? `married ${other(m, a)}` : m.status;
438 return ` ${a} ${censusYear - f.person(a).birth} ${status}`;
439 });
440 return [`CENSUS ${censusYear} ${lines.length}`, ...lines];
441 }
442 }
443 const person = f.person(x);
444 if (['RELATION', 'PATH', 'COMMON', 'KINSHIP'].includes(command)) f.person(y);
445 const ids = [...f.persons.keys()].sort(compareIds);
446 switch (command) {
447 case 'INFO':
448 return [`INFO ${x} ${person.name} ${person.sex} ${life(person)}`,
449 ` parents: ${f.personList(person.parents)}`,
450 ...f.records(x).map(m => ` marriage: ${other(m, x)} ${m.start}-${m.end ?? ''} ${m.status}`),
451 ` children: ${f.personList(f.children(x))}`,
452 ` siblings: ${f.personList(f.siblings(x))}`,
453 ` inbreeding: ${fraction(f.inbreeding(x))}`];
454 case 'RELATION':
455 return [`RELATION ${x} ${y}: ${f.relationshipText(x, y)}`];
456 case 'KINSHIP':
457 return [`KINSHIP ${x} ${y} ${fraction(f.coefficient(x, y))}`];
458 case 'RELATIVES': {
459 const lines: string[] = [];
460 for (const a of ids) {
461 if (a === x) continue;
462 const terms = f.relationship(a, x);
463 if (terms.length) lines.push(` ${a}: ${terms.join(', ')}`);
464 }
465 return [`RELATIVES ${x} ${lines.length}`, ...lines];
466 }
467 case 'WHO': {
468 const query = args.slice(1).join(' ');
469 const lines: string[] = [];
470 for (const a of ids) {
471 if (a === x) continue;
472 const terms = f.relationship(a, x);
473 const neutral = f.relationship(a, x, 'X');
474 const i = terms.findIndex((t, i) => t === query || neutral[i] === query);
475 if (i >= 0) lines.push(` ${a}: ${terms[i]}`);
476 }
477 return [`WHO ${x} ${query} ${lines.length}`, ...lines];
478 }
479 case 'PATH': {
480 const apex = f.apex(x, y);
481 if (!apex) return [`PATH ${x} ${y} none`];
482 const apexPerson = apex.person;
483 const depths = f.distances(apexPerson, false);
484 function upward(start: string): string[] {
485 const path = [start];
486 while (path.at(-1) !== apexPerson) {
487 const current = path.at(-1)!;
488 path.push(f.parents(current).filter(p => depths.get(p) === depths.get(current)! - 1).sort(compareIds)[0]);
489 }
490 return path;
491 }
492 const route = [...upward(x), ...upward(y).slice(0, -1).reverse()];
493 return [`PATH ${x} ${y} ${apex.da + apex.db}: ${route.join(' ')}`];
494 }
495 case 'COMMON': {
496 const a = f.distances(x, true);
497 const b = f.distances(y, true);
498 const common = new Set([...a.keys()].filter(c => b.has(c)));
499 const nearest = [...common].filter(c => !f.children(c).some(child => common.has(child)));
500 nearest.sort((c, d) => (a.get(c)! + b.get(c)!) - (a.get(d)! + b.get(d)!) || compareIds(c, d));
501 return [`COMMON ${x} ${y} ${nearest.length}`, ...nearest.map(c => ` ${c} ${a.get(c)!} ${b.get(c)!}`)];
502 }
503 case 'ANCESTORS':
504 case 'DESCENDANTS': {
505 const upward = command === 'ANCESTORS';
506 const entries = [...f.distances(x, upward)].filter(([a]) => a !== x);
507 entries.sort(([a, g], [b, h]) => g - h || compareIds(a, b));
508 return [`${command} ${x} ${entries.length}`, ...entries.map(([a, g]) =>
509 ` ${g} ${a} ${term(upward ? 0 : g, upward ? g : 0, f.person(a).sex)}`)];
510 }
511 case 'COUSINS': {
512 const matches: { id: string; apex: Apex }[] = [];
513 for (const a of ids) {
514 if (a === x) continue;
515 const apex = f.apex(a, x);
516 if (apex && apex.da >= 2 && apex.db >= 2 && Math.min(apex.da, apex.db) - 1 === Number(y))
517 matches.push({ id: a, apex });
518 }
519 matches.sort((a, b) => Math.abs(a.apex.da - a.apex.db) - Math.abs(b.apex.da - b.apex.db) || compareIds(a.id, b.id));
520 return [`COUSINS ${x} ${y} ${matches.length}`, ...matches.map(a =>
521 ` ${a.id} ${bloodTerm(a.apex, f.person(a.id).sex)}`)];
522 }
523 case 'SIBLINGS': {
524 const lines: string[] = [];
525 let earlier = 0;
526 for (const a of f.ageOrder(ids)) {
527 if (a === x) continue;
528 const common = f.commonParents(x, a);
529 const kind = common === 2 ? 'full' : common === 1 ? 'half' : f.stepRelated(x, a) ? 'step' : null;
530 if (!kind) continue;
531 if (common > 0 && f.ageCompare(a, x) < 0) ++earlier;
532 lines.push(` ${a} ${kind} ${f.person(a).birth}`);
533 }
534 return [`SIBLINGS ${x} ${lines.length} order=${earlier + 1}`, ...lines];
535 }
536 case 'GENERATIONS': {
537 const groups = new Map<number, Person[]>();
538 for (const [a, g] of f.distances(x, false)) {

549 case 'TREE': {

553 const indent = ' '.repeat(2 * (depth + 1));

555 visited.add(a);

558 lines.push(`${indent}${a} ${p.name} ${life(p)}${spouse === null ? '' : ` = ${spouse}`}`);
559 for (const child of f.ageOrder(f.children(a))) visit(child, depth + 1);
560 }
561 visit(x, 0);

574 parents.forEach((parent, i) => enter(parent, 2 * number + i, generation + 1));

579 case 'TIMELINE': {
580 const events: FamilyEvent[] = [{ year: person.birth, rank: 0, counterpart: '', text: 'born' }];

583 events.push({ year: m.start, rank: 1, counterpart: a, text: `married ${a}` });
584 if (m.status === 'divorced') events.push({ year: m.end!, rank: 3, counterpart: a, text: `divorced ${a}` });
585 else if (m.status === 'widowed' && m.died === a)
586 events.push({ year: m.end!, rank: 4, counterpart: a, text: `widowed ${a}` });

588 for (const a of f.children(x)) events.push({ year: f.person(a).birth, rank: 2, counterpart: a, text: `child ${a}` });
589 for (const [label, relatives] of [['parent', person.parents], ['child', f.children(x)]] as const) {
590 for (const a of relatives) {
591 const death = f.person(a).death;
592 if (death !== null && death >= person.birth && (person.death === null || death <= person.death))
593 events.push({ year: death, rank: 5, counterpart: a, text: `lost ${label} ${a}` });
594 }
595 }
596 if (person.death !== null) events.push({ year: person.death, rank: 6, counterpart: '', text: 'died' });
597 events.sort((a, b) => a.year - b.year || a.rank - b.rank || compareIds(a.counterpart, b.counterpart));
598 return [`TIMELINE ${x} ${events.length}`, ...events.map(e => ` ${e.year} ${e.text}`)];
599 }
600 case 'FIND': {

602 for (const step of y.split('.')) {
603 const [kind, sex] = STEPS[step];
604 const next = new Set<string>();
605 for (const a of current) {

608 for (const b of targets) if (sex === null || f.person(b).sex === sex) next.add(b);
609 }
610 current = next;
611 }
612 current.delete(x);
613 return [`FIND ${x} ${y} ${current.size}`, ...[...current].sort(compareIds).map(a => ` ${a}: ${f.relationshipText(a, x)}`)];
614 }
