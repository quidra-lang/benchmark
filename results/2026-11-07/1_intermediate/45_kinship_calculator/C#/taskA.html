11readonly record struct Fraction(long Numerator, int Power) {
12 public static Fraction Make(long numerator, int power) {
13 if (numerator == 0)
14 return new(0, 0);
15 while (power > 0 && numerator % 2 == 0) {
16 numerator /= 2;
17 power--;
18 }
19 return new(numerator, power);
20 }
21 public static Fraction operator +(Fraction a, Fraction b) {
22 int power = Math.Max(a.Power, b.Power);
23 return Make((a.Numerator << (power - a.Power)) + (b.Numerator << (power - b.Power)), power);
24 }
25 public Fraction Half() => Make(Numerator, Power + 1);
26 public override string ToString() => $"{Numerator}/{1L << Power}";
27}

69static class Terms {
70 public static string Gender(Sex sex, string male, string female,
71 string neutral) => sex == Sex.M ? male
72 : sex == Sex.F ? female
73 : neutral;
74
75 static string Grand(int generation) =>
76 generation == 1 ? "" : string.Concat(Enumerable.Repeat("great-", generation - 2)) + "grand";
77
78 static string Ordinal(int value) {
79 string[] first = [
80 "first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth", "ninth",
81 "tenth"
82 ];
83 if (value <= 10)
84 return first[value - 1];
85 string suffix = value % 100 is 11 or 12 or 13 ? "th"
86 : ((value % 10) switch {
87 1 => "st",
88 2 => "nd",
89 3 => "rd",
90 _ => "th"
91 });
92 return value + suffix;
93 }
94
95 public static string Build(int a, int b, Sex sex) {
96 if (a == 0 && b == 0)
97 return "self";
98 if (a == 0)
99 return Grand(b) + Gender(sex, "father", "mother", "parent");
100 if (b == 0)
101 return Grand(a) + Gender(sex, "son", "daughter", "child");
102 if (a == 1 && b == 1)
103 return Gender(sex, "brother", "sister", "sibling");
104 if (a == 1)
105 return Grand(b - 1) + Gender(sex, "uncle", "aunt", "pibling");
106 if (b == 1)
107 return Grand(a - 1) + Gender(sex, "nephew", "niece", "nibling");
108 int removal = Math.Abs(a - b);
109 string suffix = removal switch {
110 0 => "",
111 1 => " once removed",
112 2 => " twice removed",
113 _ => $" {removal} times removed"
114 };
115 return Ordinal(Math.Min(a, b) - 1) + " cousin" + suffix;
116 }
117}

138 static (string, string) Pair(string a, string b) => string.CompareOrdinal(a, b) <= 0 ? (a, b)
139 : (b, a);
140 public Person Person(string id) => family.People[id];

161 Parents(a).Any(p => Parents(b).Any(q => Partner(p, q)));

163 Dictionary<string, int> Distances(string start, bool upward) {
164 Dictionary<string, Dictionary<string, int>> cache = upward ? ancestors : descendants;
165 if (cache.TryGetValue(start, out Dictionary<string, int>? saved))
166 return saved;
167 Dictionary<string, int> distances = new(StringComparer.Ordinal) { [start] = 0 };
168 Queue<string> pending = new();
169 pending.Enqueue(start);
170 while (pending.TryDequeue(out string? current))
171 {
172 foreach (string next in upward ? Parents(current) : Children(current))
173 if (distances.TryAdd(next, distances[current] + 1))
174 pending.Enqueue(next);
175 }
176 cache[start] = distances;
177 return distances;
178 }
179 public Dictionary<string, int> Up(string id) => Distances(id, true);
180 public Dictionary<string, int> Down(string id) => Distances(id, false);
181 public bool Ancestor(string a, string b) => a != b && Up(b).ContainsKey(a);

183 public Apex? ApexOf(string a, string b) {
184 Dictionary<string, int> da = Up(a), db = Up(b);
185 if (db.TryGetValue(a, out int fromB))
186 return new(a, 0, fromB);
187 if (da.TryGetValue(b, out int fromA))
188 return new(b, fromA, 0);
189 string? apex = da.Keys.Where(db.ContainsKey)
190 .OrderBy(id => da[id] + db[id])
191 .ThenBy(id => da[id])
192 .ThenBy(id => id, StringComparer.Ordinal)
193 .FirstOrDefault();
194 if (apex is null)
195 return null;
196 int share =
197 da.Keys.Count(id => db.ContainsKey(id) && da[id] == da[apex] && db[id] == db[apex]);
198 return new(apex, da[apex], db[apex], share);
199 }

201 public string Blood(string a, Apex apex, Sex? sex = null) {
202 string prefix = apex.Collateral ? apex.Share == 1 ? "half-"
203 : apex.Share >= 4 ? "double "
204 : ""
205 : "";
206 return prefix + Terms.Build(apex.A, apex.B, sex ?? Person(a).Sex);
207 }

209 public List<string> Relations(string a, string b, Sex? overrideSex = null) {
210 if (a == b)
211 return ["self"];
212 Sex sex = overrideSex ?? Person(a).Sex;
213 List<string> result = [];
214 string Word(string male, string female, string neutral) =>
215 Terms.Gender(sex, male, female, neutral);
216 if (ApexOf(a, b) is Apex apex)
217 result.Add(Blood(a, apex, sex));
218 Marriage? marriage = Last(a, b);
219 if (marriage?.Status == MarriageStatus.Current)
220 result.Add(Word("husband", "wife", "spouse"));
221 if (marriage?.Status == MarriageStatus.Widowed)
222 result.Add(marriage.Died == a ? Word("late husband", "late wife", "late spouse")
223 : Word("widower", "widow", "surviving spouse"));
224 if (marriage?.Status == MarriageStatus.Divorced)
225 result.Add(Word("ex-husband", "ex-wife", "ex-spouse"));
226 if (marriage is null && Children(a).Any(child => Person(child).Parents.Contains(b)))
227 result.Add("co-parent");
228 if (!Person(b).Parents.Contains(a) && Parents(b).Any(p => Partner(a, p)))
229 result.Add(Word("stepfather", "stepmother", "stepparent"));
230 if (!Person(a).Parents.Contains(b) && Parents(a).Any(p => Partner(b, p)))
231 result.Add(Word("stepson", "stepdaughter", "stepchild"));
232 if (StepSiblings(a, b))
233 result.Add(Word("stepbrother", "stepsister", "stepsibling"));
234 if (!Ancestor(a, b) && Up(b).Any(item => item.Value == 2 && Partner(a, item.Key)))
235 result.Add(Word("stepgrandfather", "stepgrandmother", "stepgrandparent"));
236 if (!Ancestor(b, a) && Up(a).Any(item => item.Value == 2 && Partner(b, item.Key)))
237 result.Add(Word("stepgrandson", "stepgranddaughter", "stepgrandchild"));
238 if (Partners(b).Any(p => Person(p).Parents.Contains(a)))
239 result.Add(Word("father-in-law", "mother-in-law", "parent-in-law"));
240 if (Partners(a).Any(p => Person(p).Parents.Contains(b)))
241 result.Add(Word("son-in-law", "daughter-in-law", "child-in-law"));
242 if (Partners(b).Any(p => Siblings(a, p)) || Partners(a).Any(p => Siblings(p, b)))
243 result.Add(Word("brother-in-law", "sister-in-law", "sibling-in-law"));
244 return result;
245 }
246 public string Relation(string a, string b) {
247 List<string> terms = Relations(a, b);
248 return terms.Count == 0 ? "no relation" : string.Join(", ", terms);
249 }

251 public Fraction Phi(string a, string b) {
252 (string, string) key = Pair(a, b);
253 if (coefficients.TryGetValue(key, out Fraction value))
254 return value;
255 if (a == b)
256 value = (new Fraction(1, 0) + Inbreeding(a)).Half();
257 else {
258 if (Person(a).Birth < Person(b).Birth)
259 (a, b) = (b, a);
260 Fraction sum = default;
261 foreach (string parent in Parents(a))
262 sum += Phi(parent, b);
263 value = sum.Half();
264 }
265 coefficients[key] = value;
266 return value;
267 }
268 public Fraction Inbreeding(string id) {
269 string[] parents = Parents(id).ToArray();
270 return parents.Length == 2 ? Phi(parents[0], parents[1]) : default;
271 }
272}

288 List<string> Modify(string line, string[] args, Analysis graph) {
289 string command = args[0], id = args[1];
290 if (command == "PERSON") {
291 Reject(family.People.ContainsKey(id), "DUPLICATE PERSON " + id);
292 Remember(line);
293 Person person = new(id, Enum.Parse<Sex>(args[2]), int.Parse(args[3]), args[4]);
294 family.People[id] = person;
295 return [$"ADDED {id} {person.Name} {person.Sex} {person.Birth}"];
296 }
297 Check(id);
298 Person first = family.People[id];
299 if (command == "REMOVE") {
300 Remember(line);
301 family.People.Remove(id);
302 foreach (Person person in family.People.Values)
303 person.Parents.Remove(id);
304 family.Marriages.RemoveAll(m => m.Contains(id));
305 return ["REMOVED " + id];
306 }
307 if (command == "DEATH") {
308 int year = int.Parse(args[2]);
309 Reject(first.Death is not null, "ALREADY DECEASED " + id);
310 Reject(
311 year < first.Birth ||
312 family.Marriages.Any(m => m.Contains(id) && (m.Start > year || m.End > year)),
313 "BAD YEAR " + id);
314 string? child =
315 graph.Children(id).FirstOrDefault(c => graph.Person(c).Birth > year + 1);
316 if (child is not null)
317 throw new CommandError($"AFTER DEATH {child} {id}");
318 Marriage? current = graph.Current(id);
319 Remember(line);
320 first.Death = year;
321 if (current is not null) {
322 int index = family.Marriages.IndexOf(current);
323 family.Marriages[index] =
324 current with { End = year, Status = MarriageStatus.Widowed, Died = id };
325 }
326 return [$"DIED {id} {year}" + (current is null ? "" : " spouse=" + current.Other(id))];
327 }
328 string other = args[2];
329 Check(other);
330 Person second = family.People[other];
331 if (command == "PARENT") {
332 Reject(id == other, "SAME PERSON " + id);
333 Reject(first.Parents.Contains(other), $"ALREADY PARENT {id} {other}");
334 Reject(first.Parents.Count == 2, "TOO MANY PARENTS " + id);
335 Reject(graph.Ancestor(id, other), $"CYCLE {id} {other}");
336 Reject(second.Birth + 12 > first.Birth, $"AGE GAP {id} {other}");
337 Reject(second.Death + 1 < first.Birth, $"AFTER DEATH {id} {other}");
338 Remember(line);
339 first.Parents.Add(other);
340 return
341 [$"LINKED {id} {other} " + Terms.Gender(second.Sex, "father", "mother", "parent")];
342 }
343 if (command == "UNPARENT") {
344 Reject(!first.Parents.Contains(other), $"NOT PARENT {id} {other}");
345 Remember(line);
346 first.Parents.Remove(other);
347 return [$"UNLINKED {id} {other}"];
348 }
349 int marriageYear = int.Parse(args[3]);
350 if (command == "SPOUSE") {
351 Reject(id == other, "SAME PERSON " + id);
352 foreach (Person person in new[] { first, second })
353 Reject(person.Death is not null, "DECEASED " + person.Id);
354 foreach (Person person in new[] { first, second })
355 Reject(graph.Spouse(person.Id) is not null, "ALREADY MARRIED " + person.Id);
356 Reject(graph.Ancestor(id, other) || graph.Ancestor(other, id) ||
357 graph.Siblings(id, other),
358 $"CLOSE KIN {id} {other}");
359 foreach (Person person in new[] { first, second })
360 Reject(marriageYear < person.Birth + 16, "TOO YOUNG " + person.Id);
361 foreach (Person person in new[] { first, second })
362 Reject(family.Marriages.Any(m => m.Contains(person.Id) && m.End >= marriageYear),
363 "BAD YEAR " + person.Id);
364 Remember(line);
365 family.Marriages.Add(new(id, other, marriageYear));
366 return [$"MARRIED {id} {other} {marriageYear}"];
367 }
368 Marriage? marriage = graph.Current(id);
369 Reject(marriage is null || marriage.Other(id) != other, $"NOT MARRIED {id} {other}");
370 Reject(marriageYear <= marriage!.Start, "BAD YEAR " + id);
371 Remember(line);
372 family.Marriages[family.Marriages.IndexOf(marriage)] =
373 marriage with { End = marriageYear, Status = MarriageStatus.Divorced };
374 return [$"DIVORCED {id} {other} {marriageYear}"];
375 }

393 static List<string> Path(string a, string b, Analysis graph) {
394 if (graph.ApexOf(a, b) is not Apex apex)
395 return [$"PATH {a} {b} none"];
396 Dictionary<string, int> distances = graph.Down(apex.Id);
397 List<string> Upward(string start) {
398 List<string> path = [start];
399 while (path[^1] != apex.Id) {
400 string current = path[^1];
401 path.Add(
402 graph.Parents(current)
403 .Where(p => distances.GetValueOrDefault(p, -1) == distances[current] - 1)
404 .OrderBy(p => p, StringComparer.Ordinal)
405 .First());
406 }
407 return path;
408 }
409 List<string> result = Upward(a), tail = Upward(b);
410 result.AddRange(tail.Take(tail.Count - 1).Reverse());
411 return [$"PATH {a} {b} {apex.A + apex.B}: " + string.Join(" ", result)];
412 }

441 static List<string> Tree(string id, Analysis graph) {
442 List<string> lines = [$"TREE {id} {graph.Down(id).Count - 1}"];
443 HashSet<string> seen = new(StringComparer.Ordinal);
444 void Visit(string current, int depth) {
445 string prefix = new(' ', 2 * (depth + 1));
446 if (!seen.Add(current)) {
447 lines.Add(prefix + current + " (see above)");
448 return;
449 }
450 Person person = graph.Person(current);
451 string? spouse = graph.Spouse(current);
452 lines.Add($"{prefix}{current} {person.Name} {person.Life}" +
453 (spouse is null ? "" : " = " + spouse));
454 foreach (string child in graph.ByAge(graph.Children(current)))
455 Visit(child, depth + 1);
456 }
457 Visit(id, 0);
458 return lines;
459 }

461 static List<string> Pedigree(string id, int depth, Analysis graph) {
462 List<string> lines = [];
463 HashSet<string> distinct = new(StringComparer.Ordinal);
464 void Enter(string current, int number, int generation) {
465 Person person = graph.Person(current);
466 distinct.Add(current);
467 lines.Add(new string(' ', 2 * (generation + 1)) +
468 $"{number} {current} {person.Name} {person.Life}");
469 if (generation >= depth)
470 return;
471 string[] parents = graph.Parents(current)
472 .OrderBy(p => graph.Person(p).Sex)
473 .ThenBy(p => p, StringComparer.Ordinal)
474 .ToArray();
475 for (int i = 0; i < parents.Length; i++)
476 Enter(parents[i], 2 * number + i, generation + 1);
477 }
478 Enter(id, 1, 0);
479 lines.Insert(0, $"PEDIGREE {id} {depth} entries={lines.Count} distinct={distinct.Count}");
480 return lines;
481 }

560 static List<string> Find(string id, string route, Analysis graph) {
561 HashSet<string> current = new(StringComparer.Ordinal) { id };
562 foreach (string step in route.Split('.')) {
563 string kind = step switch {
564 "father" or "mother" => "parent",
565 "son" or "daughter" => "child",
566 "brother" or "sister" => "sibling",
567 "husband" or "wife" => "spouse",
568 _ => step,
569 };
570 Sex? sex = step switch {
571 "father" or "son" or "brother" or "husband" => Sex.M,
572 "mother" or "daughter" or "sister" or "wife" => Sex.F,
573 _ => null,
574 };
575 HashSet<string> next = new(StringComparer.Ordinal);
576 foreach (string person in current) {
577 IEnumerable<string> candidates = kind switch {
578 "parent" => graph.Parents(person),
579 "child" => graph.Children(person),
580 "sibling" => graph.Ids.Where(other => graph.Siblings(person, other)),
581 _ => graph.Spouse(person) is string spouse ? [spouse] : [],
582 };
583 foreach (string other in candidates)
584 if (sex is null || graph.Person(other).Sex == sex)
585 next.Add(other);
586 }
587 current = next;
588 }
589 current.Remove(id);
590 List<string> lines = [$"FIND {id} {route} {current.Count}"];
591 lines.AddRange(current.OrderBy(other => other, StringComparer.Ordinal)
592 .Select(other => $" {other}: {graph.Relation(other, id)}"));
593 return lines;
594 }

596 public List<string> Execute(string line) {
597 string[] args = line.Split(' ');
598 string command = args[0];
599 if (command == "UNDO") {
600 if (!history.TryPop(out var previous))
601 throw new CommandError("NOTHING TO UNDO");
602 family = previous.State;
603 return ["UNDONE " + previous.Line];
604 }
605 Analysis graph = new(family);
606 if (command is "PERSON" or "DEATH" or "REMOVE" or "PARENT" or "UNPARENT" or "SPOUSE" or
607 "DIVORCE")
608 return Modify(line, args, graph);
609 if (command == "TERM") {
610 int a = int.Parse(args[1]), b = int.Parse(args[2]);
611 Sex sex = Enum.Parse<Sex>(args[3]);
612 return [$"TERM {a} {b} {sex} {Terms.Build(a, b, sex)}"];
613 }
614 if (command == "STATS")
615 return Stats(graph);
616 if (command == "AUDIT")
617 return Audit(graph);
618 if (command == "CENSUS")
619 return Census(int.Parse(args[1]), graph);
620 string id = args[1];
621 Check(id);
622 if (command is "RELATION" or "PATH" or "COMMON" or "KINSHIP") {
623 string other = args[2];
624 Check(other);
625 return command switch {
626 "RELATION" => [$"RELATION {id} {other}: {graph.Relation(id, other)}"],
627 "PATH" => Path(id, other, graph),
628 "COMMON" => Common(id, other, graph),
629 _ => [$"KINSHIP {id} {other} {graph.Phi(id, other)}"],
630 };
631 }
632 if (command == "INFO")
633 return Info(id, graph);
634 if (command == "TREE")
635 return Tree(id, graph);
636 if (command == "PEDIGREE")
637 return Pedigree(id, int.Parse(args[2]), graph);
638 if (command == "TIMELINE")
639 return Timeline(id, graph);
640 if (command == "GENERATIONS")
641 return Generations(id, graph);
642 if (command == "FIND")
643 return Find(id, args[2], graph);
644 if (command == "RELATIVES" || command == "WHO") {
645 string query = command == "WHO" ? string.Join(" ", args.Skip(2)) : "";
646 List<string> lines = [];
647 foreach (string other in graph.Ids.Where(other => other != id)) {
648 List<string> actual = graph.Relations(other, id);
649 if (command == "RELATIVES") {
650 if (actual.Count > 0)
651 lines.Add($" {other}: " + string.Join(", ", actual));
652 } else {
653 List<string> neutral = graph.Relations(other, id, Sex.X);
654 int match =
655 Enumerable.Range(0, actual.Count)
656 .FirstOrDefault(i => actual[i] == query || neutral[i] == query, -1);
657 if (match >= 0)
658 lines.Add($" {other}: {actual[match]}");
659 }
660 }
661 lines.Insert(0, command == "WHO" ? $"WHO {id} {query} {lines.Count}"
662 : $"RELATIVES {id} {lines.Count}");
663 return lines;
664 }
665 if (command is "ANCESTORS" or "DESCENDANTS") {
666 bool upward = command == "ANCESTORS";
667 Dictionary<string, int> distances = upward ? graph.Up(id) : graph.Down(id);
668 List<string> lines = [$"{command} {id} {distances.Count - 1}"];
669 foreach ((string other, int distance) in distances.Where(pair => pair.Value > 0)
670 .OrderBy(pair => pair.Value)
671 .ThenBy(pair => pair.Key, StringComparer.Ordinal))
672 lines.Add($" {distance} {other} " + Terms.Build(upward ? 0 : distance,
673 upward ? distance : 0,
674 graph.Person(other).Sex));
675 return lines;
676 }
677 if (command == "COUSINS") {
678 int degree = int.Parse(args[2]);
679 List<(string Id, Apex Apex)> matches = [];
680 foreach (string other in graph.Ids.Where(other => other != id))
681 if (graph.ApexOf(other, id) is Apex apex && apex.A >= 2 && apex.B >= 2 &&
682 Math.Min(apex.A, apex.B) - 1 == degree)
683 matches.Add((other, apex));
684 List<string> lines = [$"COUSINS {id} {degree} {matches.Count}"];
685 lines.AddRange(matches.OrderBy(item => Math.Abs(item.Apex.A - item.Apex.B))
686 .ThenBy(item => item.Id, StringComparer.Ordinal)
687 .Select(item => $" {item.Id} {graph.Blood(item.Id, item.Apex)}"));
688 return lines;
689 }
690 List<(string Id, string Kind)> siblings = [];
691 int before = 0;
692 Person subject = graph.Person(id);
693 foreach (string other in graph.ByAge(graph.Ids.Where(other => other != id))) {
694 int common = graph.SharedParents(id, other);
695 string kind = common == 2 ? "full"
696 : common == 1 ? "half"
697 : graph.StepSiblings(id, other) ? "step"
698 : "";
699 if (kind.Length == 0)
700 continue;
701 siblings.Add((other, kind));
702 Person person = graph.Person(other);
703 if (common > 0 &&
704 (person.Birth < subject.Birth ||
705 (person.Birth == subject.Birth && string.CompareOrdinal(other, id) < 0)))
706 before++;
707 }
708 List<string> result = [$"SIBLINGS {id} {siblings.Count} order={before + 1}"];
709 result.AddRange(
710 siblings.Select(item => $" {item.Id} {item.Kind} {graph.Person(item.Id).Birth}"));
711 return result;
712 }
713}
