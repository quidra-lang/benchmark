90 private record Fraction(long numerator, long denominator) {
91 static final Fraction ZERO = new Fraction(0, 1);
92 static final Fraction ONE = new Fraction(1, 1);
93 static final Fraction HALF = new Fraction(1, 2);
94
95 Fraction {
96 long a = numerator;
97 long b = denominator;
98 while (b != 0) {
99 long remainder = a % b;
100 a = b;
101 b = remainder;
102 }
103 numerator /= a;
104 denominator /= a;
105 }
106
107 Fraction add(Fraction other) {
108 long common = Math.max(denominator, other.denominator);
109 return new Fraction(numerator * (common / denominator)
110 + other.numerator * (common / other.denominator), common);
111 }
112
113 Fraction half() {
114 return new Fraction(numerator, denominator * 2);
115 }
116
117 @Override
118 public String toString() {
119 return numerator + "/" + denominator;
120 }
121 }

146 private static String ordinal(int number) {
147 String[] words = {"first", "second", "third", "fourth", "fifth", "sixth",
148 "seventh", "eighth", "ninth", "tenth"};
149 if (number <= 10) return words[number - 1];
150 String suffix = number % 100 >= 11 && number % 100 <= 13 ? "th"
151 : switch (number % 10) {
152 case 1 -> "st";
153 case 2 -> "nd";
154 case 3 -> "rd";
155 default -> "th";
156 };
157 return number + suffix;
158 }

160 private static String term(int da, int db, String sex) {
161 if (da == 0 && db == 0) return "self";
162 if (da == 0) return generationPrefix(db) + sexWord(sex, "father", "mother", "parent");
163 if (db == 0) return generationPrefix(da) + sexWord(sex, "son", "daughter", "child");
164 if (da == 1 && db == 1) return sexWord(sex, "brother", "sister", "sibling");
165 if (da == 1) return generationPrefix(db - 1) + sexWord(sex, "uncle", "aunt", "pibling");
166 if (db == 1) return generationPrefix(da - 1) + sexWord(sex, "nephew", "niece", "nibling");
167 int distance = Math.abs(da - db);
168 String removal = switch (distance) {
169 case 0 -> "";
170 case 1 -> " once removed";
171 case 2 -> " twice removed";
172 default -> " " + distance + " times removed";
173 };
174 return ordinal(Math.min(da, db) - 1) + " cousin" + removal;
175 }

260 private Map<String, Integer> ancestors(String id) {
261 Map<String, Integer> cached = ancestorCache.get(id);
262 if (cached != null) return cached;
263 Map<String, Integer> distance = new HashMap<>();
264 ArrayDeque<String> queue = new ArrayDeque<>();
265 distance.put(id, 0);
266 queue.add(id);
267 while (!queue.isEmpty()) {
268 String current = queue.removeFirst();
269 for (String parent : person(current).parents) {
270 if (!distance.containsKey(parent)) {
271 distance.put(parent, distance.get(current) + 1);
272 queue.addLast(parent);
273 }
274 }
275 }
276 ancestorCache.put(id, distance);
277 return distance;
278 }

280 private Map<String, Integer> descendants(String id) {
281 Map<String, Integer> distance = new HashMap<>();
282 ArrayDeque<String> queue = new ArrayDeque<>();
283 distance.put(id, 0);
284 queue.add(id);
285 while (!queue.isEmpty()) {
286 String current = queue.removeFirst();
287 for (String child : children(current)) {
288 if (!distance.containsKey(child)) {
289 distance.put(child, distance.get(current) + 1);
290 queue.addLast(child);
291 }
292 }
293 }
294 distance.remove(id);
295 return distance;
296 }

298 private Apex apex(String a, String b) {
299 Map<String, Integer> fromA = ancestors(a);
300 Map<String, Integer> fromB = ancestors(b);
301 if (fromB.containsKey(a)) return new Apex(a, 0, fromB.get(a), 0, false);
302 if (fromA.containsKey(b)) return new Apex(b, fromA.get(b), 0, 0, false);
303 Set<String> common = new HashSet<>(fromA.keySet());
304 common.retainAll(fromB.keySet());
305 String chosen = common.stream().min(Comparator.<String>comparingInt(
306 id -> fromA.get(id) + fromB.get(id)).thenComparingInt(fromA::get)
307 .thenComparing(id -> id)).orElse(null);
308 if (chosen == null) return null;
309 int da = fromA.get(chosen);
310 int db = fromB.get(chosen);
311 int shares = 0;
312 for (String id : common) {
313 if (fromA.get(id) == da && fromB.get(id) == db) shares++;
314 }
315 return new Apex(chosen, da, db, shares, true);
316 }

321 : top.shares() >= 4 ? "double " : "";
322 return prefix + term(top.da(), top.db(), sex);

329 private List<String> relationships(String a, String b, String sex) {
330 if (a.equals(b)) return List.of("self");
331 List<String> result = new ArrayList<>();
332 Set<String> parentsA = person(a).parents;
333 Set<String> parentsB = person(b).parents;
334 Set<String> partnersA = partners(a);
335 Set<String> partnersB = partners(b);
336 Marriage marriage = lastMarriage(a, b);
337 if (apex(a, b) != null) result.add(bloodTerm(a, b, sex));
338 if (marriage != null) {
339 switch (marriage.status) {
340 case "current" -> result.add(sexWord(sex, "husband", "wife", "spouse"));
341 case "widowed" -> result.add(marriage.died.equals(a)
342 ? sexWord(sex, "late husband", "late wife", "late spouse")
343 : sexWord(sex, "widower", "widow", "surviving spouse"));
344 case "divorced" -> result.add(sexWord(sex, "ex-husband", "ex-wife", "ex-spouse"));
345 }
346 }
347 if (marriage == null && children(a).stream().anyMatch(id -> person(id).parents.contains(b))) {
348 result.add("co-parent");
349 }
350 if (!parentsB.contains(a) && parentsB.stream().anyMatch(partnersA::contains)) {
351 result.add(sexWord(sex, "stepfather", "stepmother", "stepparent"));
352 }
353 if (!parentsA.contains(b) && parentsA.stream().anyMatch(partnersB::contains)) {
354 result.add(sexWord(sex, "stepson", "stepdaughter", "stepchild"));
355 }
356 if (parentsA.stream().noneMatch(parentsB::contains)
357 && parentsA.stream().anyMatch(p -> parentsB.stream().anyMatch(partners(p)::contains))) {
358 result.add(sexWord(sex, "stepbrother", "stepsister", "stepsibling"));
359 }
360 if (!ancestors(b).containsKey(a) && ancestors(b).entrySet().stream()
361 .anyMatch(entry -> entry.getValue() == 2 && partnersA.contains(entry.getKey()))) {
362 result.add(sexWord(sex, "stepgrandfather", "stepgrandmother", "stepgrandparent"));
363 }
364 if (!ancestors(a).containsKey(b) && ancestors(a).entrySet().stream()
365 .anyMatch(entry -> entry.getValue() == 2 && partnersB.contains(entry.getKey()))) {
366 result.add(sexWord(sex, "stepgrandson", "stepgranddaughter", "stepgrandchild"));
367 }
368 if (partnersB.stream().anyMatch(id -> person(id).parents.contains(a))) {
369 result.add(sexWord(sex, "father-in-law", "mother-in-law", "parent-in-law"));
370 }
371 if (partnersA.stream().anyMatch(id -> person(id).parents.contains(b))) {
372 result.add(sexWord(sex, "son-in-law", "daughter-in-law", "child-in-law"));
373 }
374 if (partnersB.stream().anyMatch(id -> siblings(a, id))
375 || partnersA.stream().anyMatch(id -> siblings(id, b))) {
376 result.add(sexWord(sex, "brother-in-law", "sister-in-law", "sibling-in-law"));
377 }
378 return result;
379 }

390 private Fraction kinship(String a, String b) {
391 Pair key = a.compareTo(b) <= 0 ? new Pair(a, b) : new Pair(b, a);
392 Fraction cached = kinshipCache.get(key);
393 if (cached != null) return cached;
394 Fraction result;
395 if (a.equals(b)) {
396 List<String> parents = new ArrayList<>(person(a).parents);
397 result = parents.size() == 2
398 ? Fraction.ONE.add(kinship(parents.get(0), parents.get(1))).half() : Fraction.HALF;
399 } else {
400 String x = person(a).birth >= person(b).birth ? a : b;
401 String y = x.equals(a) ? b : a;
402 Fraction total = Fraction.ZERO;
403 for (String parent : person(x).parents) total = total.add(kinship(parent, y));
404 result = total.half();
405 }
406 kinshipCache.put(key, result);
407 return result;
408 }

415 private List<String> upwardPath(String id, String top) {
416 List<String> result = new ArrayList<>();
417 String current = id;
418 result.add(current);
419 while (!current.equals(top)) {
420 int remaining = ancestors(current).get(top);
421 current = person(current).parents.stream().sorted()
422 .filter(parent -> Integer.valueOf(remaining - 1).equals(ancestors(parent).get(top)))
423 .findFirst().orElseThrow();
424 result.add(current);
425 }
426 return result;
427 }

429 private int longestChain(String id, Map<String, Integer> lengths) {
430 Integer cached = lengths.get(id);
431 if (cached != null) return cached;
432 int result = 1;
433 for (String parent : person(id).parents) result = Math.max(result, 1 + longestChain(parent, lengths));
434 lengths.put(id, result);
435 return result;
436 }

438 private void tree(String id, int depth, Set<String> visited, List<String> output) {
439 String indent = " ".repeat(depth + 1);
440 if (!visited.add(id)) {
441 output.add(indent + id + " (see above)");
442 return;
443 }
444 Person person = person(id);
445 String spouse = spouse(id);
446 output.add(indent + id + " " + person.name + " " + person.lifeYears()
447 + (spouse == null ? "" : " = " + spouse));
448 for (String child : ageOrder(children(id))) tree(child, depth + 1, visited, output);
449 }

451 private void pedigree(String id, int number, int generation, int depth,
452 List<String> rows, Set<String> distinct) {
453 Person person = person(id);
454 distinct.add(id);
455 rows.add(" ".repeat(generation + 1) + number + " " + id + " " + person.name + " " + person.lifeYears());
456 if (generation >= depth) return;
457 List<String> parents = new ArrayList<>(person.parents);
458 parents.sort(Comparator.<String>comparingInt(parent -> switch (person(parent).sex) {
459 case "M" -> 0;
460 case "F" -> 1;
461 default -> 2;
462 }).thenComparing(parent -> parent));
463 for (int index = 0; index < parents.size(); index++) {
464 pedigree(parents.get(index), 2 * number + index, generation + 1, depth, rows, distinct);
465 }
466 }

475 List<String> execute(String line) {
476 ancestorCache.clear();
477 kinshipCache.clear();
478 String[] words = line.split(" ");
479 String command = words[0];
480 int checkedIds = TWO_IDS.contains(command) ? 2 : ONE_ID.contains(command) ? 1 : 0;
481 for (int index = 1; index <= checkedIds; index++) {
482 if (!people.containsKey(words[index])) throw new CommandError("UNKNOWN PERSON " + words[index]);
483 }
484 Snapshot before = MODIFYING.contains(command) ? snapshot() : null;
485 List<String> output = new ArrayList<>();
486 switch (command) {
487 case "PERSON" -> {
488 String id = words[1];
489 if (people.containsKey(id)) throw new CommandError("DUPLICATE PERSON " + id);
490 people.put(id, new Person(id, words[2], Integer.parseInt(words[3]), words[4]));
491 output.add("ADDED " + id + " " + words[4] + " " + words[2] + " " + words[3]);
492 }
493 case "DEATH" -> {
494 String id = words[1];
495 int year = Integer.parseInt(words[2]);
496 Person person = person(id);
497 if (person.death != null) throw new CommandError("ALREADY DECEASED " + id);
498 if (year < person.birth || marriages.stream().anyMatch(marriage -> marriage.contains(id)
499 && (year < marriage.start || marriage.end != null && year < marriage.end))) {
500 throw new CommandError("BAD YEAR " + id);
501 }
502 String child = children(id).stream().sorted().filter(c -> person(c).birth > year + 1)
503 .findFirst().orElse(null);
504 if (child != null) throw new CommandError("AFTER DEATH " + child + " " + id);
505 Marriage marriage = currentMarriage(id);
506 person.death = year;
507 if (marriage != null) {
508 marriage.end = year;
509 marriage.status = "widowed";
510 marriage.died = id;
511 }
512 output.add("DIED " + id + " " + year + (marriage == null ? "" : " spouse=" + marriage.other(id)));
513 }
514 case "REMOVE" -> {
515 String id = words[1];
516 for (Person person : people.values()) person.parents.remove(id);
517 marriages.removeIf(marriage -> marriage.contains(id));
518 people.remove(id);
519 output.add("REMOVED " + id);
520 }
521 case "PARENT" -> {
522 String child = words[1];
523 String parent = words[2];
524 Person c = person(child);
525 Person p = person(parent);
526 if (child.equals(parent)) throw new CommandError("SAME PERSON " + child);
527 if (c.parents.contains(parent)) throw new CommandError("ALREADY PARENT " + child + " " + parent);
528 if (c.parents.size() == 2) throw new CommandError("TOO MANY PARENTS " + child);
529 if (ancestors(parent).containsKey(child)) throw new CommandError("CYCLE " + child + " " + parent);
530 if (p.birth + 12 > c.birth) throw new CommandError("AGE GAP " + child + " " + parent);
531 if (p.death != null && p.death + 1 < c.birth) throw new CommandError("AFTER DEATH " + child + " " + parent);
532 c.parents.add(parent);
533 output.add("LINKED " + child + " " + parent + " " + sexWord(p.sex, "father", "mother", "parent"));
534 }
535 case "UNPARENT" -> {
536 String child = words[1];
537 String parent = words[2];
538 if (!person(child).parents.contains(parent)) throw new CommandError("NOT PARENT " + child + " " + parent);
539 person(child).parents.remove(parent);
540 output.add("UNLINKED " + child + " " + parent);
541 }
542 case "SPOUSE" -> {
543 String a = words[1];
544 String b = words[2];
545 int year = Integer.parseInt(words[3]);
546 List<String> ids = List.of(a, b);
547 if (a.equals(b)) throw new CommandError("SAME PERSON " + a);
548 for (String id : ids) if (person(id).death != null) throw new CommandError("DECEASED " + id);
549 for (String id : ids) if (spouse(id) != null) throw new CommandError("ALREADY MARRIED " + id);
550 if (ancestors(b).containsKey(a) || ancestors(a).containsKey(b) || siblings(a, b)) {
551 throw new CommandError("CLOSE KIN " + a + " " + b);
552 }
553 for (String id : ids) if (year < person(id).birth + 16) throw new CommandError("TOO YOUNG " + id);
554 for (String id : ids) {
555 if (marriages.stream().anyMatch(marriage -> marriage.contains(id)
556 && marriage.end != null && marriage.end >= year)) throw new CommandError("BAD YEAR " + id);
557 }
558 marriages.add(new Marriage(a, b, year));
559 output.add("MARRIED " + a + " " + b + " " + year);
560 }
561 case "DIVORCE" -> {
562 String a = words[1];
563 String b = words[2];
564 int year = Integer.parseInt(words[3]);
565 if (!b.equals(spouse(a))) throw new CommandError("NOT MARRIED " + a + " " + b);
566 Marriage marriage = currentMarriage(a);
567 if (year <= marriage.start) throw new CommandError("BAD YEAR " + a);
568 marriage.end = year;
569 marriage.status = "divorced";
570 output.add("DIVORCED " + a + " " + b + " " + year);
571 }
572 case "UNDO" -> {
573 if (history.isEmpty()) throw new CommandError("NOTHING TO UNDO");
574 UndoEntry entry = history.remove(history.size() - 1);
575 people = entry.state().people();
576 marriages = entry.state().marriages();
577 output.add("UNDONE " + entry.line());
578 }
579 case "INFO" -> {
580 String id = words[1];
581 Person person = person(id);
582 output.add("INFO " + id + " " + person.name + " " + person.sex + " " + person.lifeYears());
583 output.add(" parents: " + personList(person.parents));
584 for (Marriage marriage : marriages) {
585 if (marriage.contains(id)) output.add(" marriage: " + marriage.other(id) + " "
586 + marriage.start + "-" + (marriage.end == null ? "" : marriage.end) + " " + marriage.status);
587 }
588 output.add(" children: " + personList(children(id)));
589 output.add(" siblings: " + personList(people.keySet().stream().filter(other -> siblings(id, other)).toList()));
590 output.add(" inbreeding: " + inbreeding(id));
591 }
592 case "RELATION" -> output.add("RELATION " + words[1] + " " + words[2] + ": "
593 + relationshipString(words[1], words[2]));
594 case "RELATIVES" -> {
595 String id = words[1];
596 List<String> rows = new ArrayList<>();
597 for (String other : idOrder()) {
598 if (other.equals(id)) continue;
599 List<String> terms = relationships(other, id);
600 if (!terms.isEmpty()) rows.add(" " + other + ": " + String.join(", ", terms));
601 }
602 output.add("RELATIVES " + id + " " + rows.size());
603 output.addAll(rows);
604 }
605 case "WHO" -> {
606 String id = words[1];
607 String query = String.join(" ", Arrays.copyOfRange(words, 2, words.length));
608 List<String> rows = new ArrayList<>();
609 for (String other : idOrder()) {
610 if (other.equals(id)) continue;
611 List<String> actual = relationships(other, id);
612 List<String> neutral = relationships(other, id, "X");
613 for (int index = 0; index < actual.size(); index++) {
614 if (actual.get(index).equals(query) || neutral.get(index).equals(query)) {
615 rows.add(" " + other + ": " + actual.get(index));
616 break;
617 }
618 }
619 }
620 output.add("WHO " + id + " " + query + " " + rows.size());
621 output.addAll(rows);
622 }
623 case "PATH" -> {
624 String a = words[1];
625 String b = words[2];
626 Apex top = apex(a, b);
627 if (top == null) output.add("PATH " + a + " " + b + " none");
628 else {
629 List<String> path = upwardPath(a, top.id());
630 List<String> tail = upwardPath(b, top.id());
631 for (int index = tail.size() - 2; index >= 0; index--) path.add(tail.get(index));
632 output.add("PATH " + a + " " + b + " " + (top.da() + top.db()) + ": " + String.join(" ", path));
633 }
634 }
635 case "COMMON" -> {
636 String a = words[1];
637 String b = words[2];
638 Map<String, Integer> fromA = ancestors(a);
639 Map<String, Integer> fromB = ancestors(b);
640 Set<String> common = new HashSet<>(fromA.keySet());
641 common.retainAll(fromB.keySet());
642 List<String> nearest = common.stream().filter(id -> children(id).stream().noneMatch(common::contains))
643 .sorted(Comparator.<String>comparingInt(id -> fromA.get(id) + fromB.get(id))
644 .thenComparing(id -> id)).toList();
645 output.add("COMMON " + a + " " + b + " " + nearest.size());
646 for (String id : nearest) output.add(" " + id + " " + fromA.get(id) + " " + fromB.get(id));
647 }
648 case "ANCESTORS", "DESCENDANTS" -> {
649 String id = words[1];
650 boolean upward = command.equals("ANCESTORS");
651 Map<String, Integer> distances = new HashMap<>(upward ? ancestors(id) : descendants(id));
652 distances.remove(id);
653 List<String> ids = new ArrayList<>(distances.keySet());
654 ids.sort(Comparator.<String>comparingInt(distances::get).thenComparing(other -> other));
655 output.add(command + " " + id + " " + ids.size());
656 for (String other : ids) {
657 int generation = distances.get(other);
658 String label = upward ? term(0, generation, person(other).sex)
659 : term(generation, 0, person(other).sex);
660 output.add(" " + generation + " " + other + " " + label);
661 }
662 }
663 case "COUSINS" -> {
664 String id = words[1];
665 int degree = Integer.parseInt(words[2]);
666 List<Cousin> matches = new ArrayList<>();
667 for (String other : people.keySet()) {
668 if (other.equals(id)) continue;
669 Apex top = apex(other, id);
670 if (top != null && top.da() >= 2 && top.db() >= 2
671 && Math.min(top.da(), top.db()) - 1 == degree) {
672 matches.add(new Cousin(other, Math.abs(top.da() - top.db())));
673 }
674 }
675 matches.sort(Comparator.comparingInt(Cousin::removal).thenComparing(Cousin::id));
676 output.add("COUSINS " + id + " " + degree + " " + matches.size());
677 for (Cousin match : matches) output.add(" " + match.id() + " " + bloodTerm(match.id(), id));
678 }
679 case "SIBLINGS" -> {
680 String id = words[1];
681 Map<String, String> classes = new HashMap<>();
682 for (String other : people.keySet()) {
683 if (other.equals(id)) continue;
684 Set<String> shared = new HashSet<>(person(id).parents);
685 shared.retainAll(person(other).parents);
686 String kind;
687 if (shared.size() == 2) kind = "full";
688 else if (shared.size() == 1) kind = "half";
689 else if (person(id).parents.stream().anyMatch(p -> person(other).parents.stream()
690 .anyMatch(partners(p)::contains))) kind = "step";
691 else continue;
692 classes.put(other, kind);
693 }
694 long earlier = classes.entrySet().stream().filter(entry -> !entry.getValue().equals("step")
695 && ageComparator().compare(entry.getKey(), id) < 0).count();
696 output.add("SIBLINGS " + id + " " + classes.size() + " order=" + (earlier + 1));
697 for (String other : ageOrder(classes.keySet())) {
698 output.add(" " + other + " " + classes.get(other) + " " + person(other).birth);
699 }
700 }
701 case "GENERATIONS" -> {
702 String id = words[1];
703 Map<Integer, List<String>> groups = new TreeMap<>();
704 descendants(id).forEach((other, generation) ->
705 groups.computeIfAbsent(generation, ignored -> new ArrayList<>()).add(other));
706 output.add("GENERATIONS " + id + " " + groups.size());
707 groups.forEach((generation, ids) -> {
708 int minimum = ids.stream().mapToInt(other -> person(other).birth).min().orElseThrow();
709 int maximum = ids.stream().mapToInt(other -> person(other).birth).max().orElseThrow();
710 long living = ids.stream().filter(other -> person(other).death == null).count();
711 output.add(" " + generation + " " + ids.size() + " " + living + " " + minimum + "-" + maximum);
712 });
713 }
714 case "TREE" -> {
715 String id = words[1];
716 output.add("TREE " + id + " " + descendants(id).size());
717 tree(id, 0, new HashSet<>(), output);
718 }
719 case "PEDIGREE" -> {
720 String id = words[1];
721 int depth = Integer.parseInt(words[2]);
722 List<String> rows = new ArrayList<>();
723 Set<String> distinct = new HashSet<>();
724 pedigree(id, 1, 0, depth, rows, distinct);
725 output.add("PEDIGREE " + id + " " + depth + " entries=" + rows.size() + " distinct=" + distinct.size());
726 output.addAll(rows);
727 }
728 case "TIMELINE" -> {
729 String id = words[1];
730 Person person = person(id);
731 List<Event> events = new ArrayList<>();
732 events.add(new Event(person.birth, 0, "", "born"));
733 for (Marriage marriage : marriages) {
734 if (!marriage.contains(id)) continue;
735 String other = marriage.other(id);
736 events.add(new Event(marriage.start, 1, other, "married " + other));
737 if (marriage.status.equals("divorced")) {
738 events.add(new Event(marriage.end, 3, other, "divorced " + other));
739 } else if (marriage.status.equals("widowed") && marriage.died.equals(other)) {
740 events.add(new Event(marriage.end, 4, other, "widowed " + other));
741 }
742 }
743 for (String child : children(id)) events.add(new Event(person(child).birth, 2, child, "child " + child));
744 for (String parent : person.parents) loss(parent, "parent", person, events);
745 for (String child : children(id)) loss(child, "child", person, events);
746 if (person.death != null) events.add(new Event(person.death, 6, "", "died"));
747 events.sort(Comparator.comparingInt(Event::year).thenComparingInt(Event::rank).thenComparing(Event::counterpart));
748 output.add("TIMELINE " + id + " " + events.size());
749 for (Event event : events) output.add(" " + event.year() + " " + event.text());
750 }
751 case "CENSUS" -> {
752 int year = Integer.parseInt(words[1]);
753 List<String> residents = ageOrder(people.keySet().stream().filter(id -> person(id).birth <= year
754 && (person(id).death == null || person(id).death > year)).toList());
755 output.add("CENSUS " + year + " " + residents.size());
756 for (String id : residents) {
757 Marriage latest = null;
758 for (Marriage marriage : marriages) {
759 if (marriage.contains(id) && marriage.start <= year) latest = marriage;
760 }
761 String status = latest == null ? "single"
762 : latest.end == null || latest.end > year ? "married " + latest.other(id) : latest.status;
763 output.add(" " + id + " " + (year - person(id).birth) + " " + status);
764 }
765 }
766 case "KINSHIP" -> output.add("KINSHIP " + words[1] + " " + words[2] + " " + kinship(words[1], words[2]));
767 case "FIND" -> {
768 String id = words[1];
769 String route = words[2];
770 Set<String> found = new HashSet<>(Set.of(id));
771 for (String step : route.split("\\.")) {
772 String requiredSex = switch (step) {
773 case "father", "son", "brother", "husband" -> "M";
774 case "mother", "daughter", "sister", "wife" -> "F";
775 default -> null;
776 };
777 Set<String> next = new HashSet<>();
778 for (String current : found) {
779 Collection<String> candidates = switch (step) {
780 case "parent", "father", "mother" -> person(current).parents;
781 case "child", "son", "daughter" -> children(current);
782 case "sibling", "brother", "sister" -> people.keySet().stream()
783 .filter(other -> siblings(current, other)).toList();
784 default -> spouse(current) == null ? List.of() : List.of(spouse(current));
785 };
786 for (String candidate : candidates) {
787 if (requiredSex == null || person(candidate).sex.equals(requiredSex)) next.add(candidate);
788 }
789 }
790 found = next;
791 }
792 found.remove(id);
793 List<String> ordered = new ArrayList<>(found);
794 ordered.sort(Comparator.naturalOrder());
795 output.add("FIND " + id + " " + route + " " + ordered.size());
796 for (String other : ordered) output.add(" " + other + ": " + relationshipString(other, id));
797 }
798 case "TERM" -> output.add("TERM " + words[1] + " " + words[2] + " " + words[3] + " "
799 + term(Integer.parseInt(words[1]), Integer.parseInt(words[2]), words[3]));
800 case "STATS" -> {
801 long living = people.values().stream().filter(person -> person.death == null).count();
802 int links = people.values().stream().mapToInt(person -> person.parents.size()).sum();
803 long couples = marriages.stream().filter(marriage -> marriage.status.equals("current")).count();
804 long founders = people.values().stream().filter(person -> person.parents.isEmpty()).count();
805 Map<String, Integer> lengths = new HashMap<>();
806 int generations = people.keySet().stream().mapToInt(id -> longestChain(id, lengths)).max().orElse(0);
807 output.add("STATS persons=" + people.size() + " living=" + living + " links=" + links
808 + " couples=" + couples + " founders=" + founders + " generations=" + generations);
809 }
810 case "AUDIT" -> {
811 List<String> rows = new ArrayList<>();
812 for (Marriage marriage : marriages) {
813 if (apex(marriage.first, marriage.second) != null) {
814 rows.add(" marriage " + marriage.first + " " + marriage.second + " "
815 + bloodTerm(marriage.first, marriage.second));
816 }
817 }
818 for (String id : idOrder()) {
819 Fraction coefficient = inbreeding(id);
820 if (coefficient.numerator() != 0) rows.add(" inbred " + id + " " + coefficient);
821 }
822 output.add("AUDIT " + rows.size());
823 output.addAll(rows);
824 }
825 }
826 if (before != null) history.add(new UndoEntry(line, before));
827 return output;
828 }
829 }
