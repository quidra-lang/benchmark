20final class Fraction {
21 final int numerator;
22 final int power;
23 const Fraction._(this.numerator, this.power);
24 static const zero = Fraction._(0, 0);
25 static const one = Fraction._(1, 0);
26
27 factory Fraction(int numerator, int power) {
28 if (numerator == 0) return zero;
29 while (power > 0 && numerator.isEven) {
30 numerator ~/= 2;
31 power--;
32 }
33 return Fraction._(numerator, power);
34 }
35
36 Fraction operator +(Fraction other) {
37 final exponent = math.max(power, other.power);
38 return Fraction((numerator << (exponent - power)) +
39 (other.numerator << (exponent - other.power)), exponent);
40 }
41
42 Fraction half() => Fraction(numerator, power + 1);
43 @override
44 String toString() => '$numerator/${1 << power}';
45}

105String grand(int generation) =>
106 generation == 1 ? '' : List.filled(generation - 2, 'great-').join() + 'grand';
107
108String ordinal(int value) {
109 const words = ['first', 'second', 'third', 'fourth', 'fifth',
110 'sixth', 'seventh', 'eighth', 'ninth', 'tenth'];
111 if (value <= 10) return words[value - 1];
112 final suffix = switch (value % 100) {
113 11 || 12 || 13 => 'th',
114 _ => switch (value % 10) { 1 => 'st', 2 => 'nd', 3 => 'rd', _ => 'th' },
115 };
116 return '$value$suffix';
117}
118
119String term(int a, int b, Sex sex) {
120 if (a == 0 && b == 0) return 'self';
121 if (a == 0) return grand(b) + gender(sex, 'father', 'mother', 'parent');
122 if (b == 0) return grand(a) + gender(sex, 'son', 'daughter', 'child');
123 if (a == 1 && b == 1) return gender(sex, 'brother', 'sister', 'sibling');
124 if (a == 1) return grand(b - 1) + gender(sex, 'uncle', 'aunt', 'pibling');
125 if (b == 1) return grand(a - 1) + gender(sex, 'nephew', 'niece', 'nibling');
126 final removal = (a - b).abs();
127 final suffix = switch (removal) {
128 0 => '', 1 => ' once removed', 2 => ' twice removed',
129 _ => ' $removal times removed',
130 };
131 return '${ordinal(math.min(a, b) - 1)} cousin$suffix';
132}

134final class Analysis {
135 final Family family;
136 final children = <String, List<String>>{};
137 final _up = <String, Map<String, int>>{};
138 final _down = <String, Map<String, int>>{};
139 final _coefficients = <(String, String), Fraction>{};

142 Analysis(this.family) {
143 for (final id in ids) {
144 children[id] = [];
145 }
146 for (final person in family.people.values) {
147 for (final parent in person.parents) {
148 children[parent]!.add(person.id);
149 }
150 }
151 for (final marriage in family.marriages) {
152 _last[pair(marriage.first, marriage.second)] = marriage;
153 }

156 static (String, String) pair(String a, String b) =>
157 a.compareTo(b) <= 0 ? (a, b) : (b, a);
158 Iterable<String> get ids => family.people.keys;
159 Person person(String id) => family.people[id]!;
160 Set<String> parents(String id) => person(id).parents;
161 int compareAge(String a, String b) {
162 final difference = person(a).birth.compareTo(person(b).birth);
163 return difference != 0 ? difference : a.compareTo(b);
164 }
165 List<String> byAge(Iterable<String> values) =>
166 values.toList()..sort(compareAge);
167 String personList(Iterable<String> values) {
168 final result = byAge(values).join(' ');
169 return result.isEmpty ? '-' : result;
170 }
171 Marriage? current(String id) {
172 for (final marriage in family.marriages) {
173 if (marriage.status == MarriageStatus.current && marriage.contains(id)) {
174 return marriage;
175 }
176 }
177 return null;

179 String? spouse(String id) => current(id)?.other(id);
180 Marriage? last(String a, String b) => _last[pair(a, b)];
181 bool partner(String a, String b) {
182 final marriage = last(a, b);
183 return marriage != null && marriage.status != MarriageStatus.divorced;
184 }
185 Iterable<String> partners(String id) => ids.where((other) => partner(id, other));
186 int sharedParents(String a, String b) =>
187 parents(a).where(parents(b).contains).length;
188 bool siblings(String a, String b) => a != b && sharedParents(a, b) > 0;
189 bool stepSiblings(String a, String b) => sharedParents(a, b) == 0 &&
190 parents(a).any((p) => parents(b).any((q) => partner(p, q)));

192 Map<String, int> distances(String start, bool upward) {
193 final cache = upward ? _up : _down;
194 final saved = cache[start];
195 if (saved != null) return saved;
196 final result = <String, int>{start: 0};
197 final pending = Queue<String>()..add(start);
198 while (pending.isNotEmpty) {
199 final current = pending.removeFirst();
200 for (final next in upward ? parents(current) : children[current]!) {
201 if (!result.containsKey(next)) {
202 result[next] = result[current]! + 1;
203 pending.add(next);
204 }
205 }
206 }
207 return cache[start] = result;
208 }
209 Map<String, int> up(String id) => distances(id, true);
210 Map<String, int> down(String id) => distances(id, false);
211 bool ancestor(String a, String b) => a != b && up(b).containsKey(a);
212
213 Apex? apex(String a, String b) {
214 final first = up(a), second = up(b);
215 if (second.containsKey(a)) return Apex(a, 0, second[a]!);
216 if (first.containsKey(b)) return Apex(b, first[b]!, 0);
217 final common = first.keys.where(second.containsKey).toList();
218 if (common.isEmpty) return null;
219 common.sort((x, y) {
220 var order = (first[x]! + second[x]!).compareTo(first[y]! + second[y]!);
221 if (order == 0) order = first[x]!.compareTo(first[y]!);
222 return order != 0 ? order : x.compareTo(y);
223 });
224 final id = common.first, da = first[id]!, db = second[id]!;
225 final share = common.where((c) => first[c] == da && second[c] == db).length;
226 return Apex(id, da, db, share);
227 }

229 String blood(Apex apex, Sex sex) {
230 final prefix = !apex.collateral ? '' :
231 apex.share == 1 ? 'half-' : apex.share >= 4 ? 'double ' : '';
232 return prefix + term(apex.a, apex.b, sex);
233 }

235 List<String> relations(String a, String b, [Sex? overrideSex]) {
236 if (a == b) return ['self'];
237 final sex = overrideSex ?? person(a).sex;
238 String word(String m, String f, String x) => gender(sex, m, f, x);
239 final result = <String>[];
240 final common = apex(a, b);
241 if (common != null) result.add(blood(common, sex));
242 final marriage = last(a, b);
243 if (marriage != null) {
244 result.add(switch (marriage.status) {
245 MarriageStatus.current => word('husband', 'wife', 'spouse'),
246 MarriageStatus.divorced => word('ex-husband', 'ex-wife', 'ex-spouse'),
247 MarriageStatus.widowed => marriage.died == a
248 ? word('late husband', 'late wife', 'late spouse')
249 : word('widower', 'widow', 'surviving spouse'),
250 });
251 } else if (children[a]!.any((c) => parents(c).contains(b))) {
252 result.add('co-parent');
253 }
254 if (!parents(b).contains(a) && parents(b).any((p) => partner(a, p))) {
255 result.add(word('stepfather', 'stepmother', 'stepparent'));
256 }
257 if (!parents(a).contains(b) && parents(a).any((p) => partner(b, p))) {
258 result.add(word('stepson', 'stepdaughter', 'stepchild'));
259 }
260 if (stepSiblings(a, b)) result.add(word('stepbrother', 'stepsister', 'stepsibling'));
261 if (!ancestor(a, b) && up(b).entries.any((p) => p.value == 2 && partner(a, p.key))) {
262 result.add(word('stepgrandfather', 'stepgrandmother', 'stepgrandparent'));
263 }
264 if (!ancestor(b, a) && up(a).entries.any((p) => p.value == 2 && partner(b, p.key))) {
265 result.add(word('stepgrandson', 'stepgranddaughter', 'stepgrandchild'));
266 }
267 if (partners(b).any((p) => parents(p).contains(a))) {
268 result.add(word('father-in-law', 'mother-in-law', 'parent-in-law'));
269 }
270 if (partners(a).any((p) => parents(p).contains(b))) {
271 result.add(word('son-in-law', 'daughter-in-law', 'child-in-law'));
272 }
273 if (partners(b).any((p) => siblings(a, p)) || partners(a).any((p) => siblings(p, b))) {
274 result.add(word('brother-in-law', 'sister-in-law', 'sibling-in-law'));
275 }
276 return result;
277 }
278
279 String relation(String a, String b) {
280 final result = relations(a, b);
281 return result.isEmpty ? 'no relation' : result.join(', ');
282 }
283
284 Fraction phi(String a, String b) {
285 final key = pair(a, b), saved = _coefficients[pair(a, b)];
286 if (saved != null) return saved;
287 Fraction result;
288 if (a == b) {
289 result = (Fraction.one + inbreeding(a)).half();
290 } else {
291 if (person(a).birth < person(b).birth) {
292 final temporary = a;
293 a = b;
294 b = temporary;
295 }
296 var sum = Fraction.zero;
297 for (final parent in parents(a)) {
298 sum = sum + phi(parent, b);
299 }
300 result = sum.half();
301 }
302 return _coefficients[key] = result;
303 }
304 Fraction inbreeding(String id) {
305 final values = parents(id).toList();
306 return values.length == 2 ? phi(values[0], values[1]) : Fraction.zero;
307 }
308}

322 List<String> modify(String line, List<String> args, Analysis graph) {
323 final command = args[0], id = args[1];
324 if (command == 'PERSON') {
325 reject(family.people.containsKey(id), 'DUPLICATE PERSON $id');
326 remember(line);
327 final person = Person(id, Sex.parse(args[2]), int.parse(args[3]), args[4]);
328 family.people[id] = person;
329 return ['ADDED $id ${person.name} ${person.sex.code} ${person.birth}'];
330 }
331 check(id);
332 final first = graph.person(id);
333 if (command == 'REMOVE') {
334 remember(line);
335 family.people.remove(id);
336 for (final person in family.people.values) {
337 person.parents.remove(id);
338 }
339 family.marriages.removeWhere((m) => m.contains(id));
340 return ['REMOVED $id'];
341 }
342 if (command == 'DEATH') {
343 final year = int.parse(args[2]);
344 reject(first.death != null, 'ALREADY DECEASED $id');
345 reject(year < first.birth || family.marriages.any((m) => m.contains(id) &&
346 (m.start > year || m.end != null && m.end! > year)), 'BAD YEAR $id');
347 for (final child in graph.children[id]!) {
348 if (graph.person(child).birth > year + 1) throw CommandError('AFTER DEATH $child $id');
349 }
350 final current = graph.current(id);
351 remember(line);
352 first.death = year;
353 if (current != null) {
354 family.marriages[family.marriages.indexOf(current)] =
355 current.finish(year, MarriageStatus.widowed, id);
356 }
357 return ['DIED $id $year${current == null ? '' : ' spouse=${current.other(id)}'}'];
358 }
359 final other = args[2];
360 check(other);
361 final second = graph.person(other);
362 if (command == 'PARENT') {
363 reject(id == other, 'SAME PERSON $id');
364 reject(first.parents.contains(other), 'ALREADY PARENT $id $other');
365 reject(first.parents.length == 2, 'TOO MANY PARENTS $id');
366 reject(graph.ancestor(id, other), 'CYCLE $id $other');
367 reject(second.birth + 12 > first.birth, 'AGE GAP $id $other');
368 reject(second.death != null && second.death! + 1 < first.birth, 'AFTER DEATH $id $other');
369 remember(line);
370 first.parents.add(other);
371 return ['LINKED $id $other ${gender(second.sex, 'father', 'mother', 'parent')}'];
372 }
373 if (command == 'UNPARENT') {
374 reject(!first.parents.contains(other), 'NOT PARENT $id $other');
375 remember(line);
376 first.parents.remove(other);
377 return ['UNLINKED $id $other'];
378 }
379 final year = int.parse(args[3]);
380 if (command == 'SPOUSE') {
381 reject(id == other, 'SAME PERSON $id');
382 for (final person in [first, second]) {
383 reject(person.death != null, 'DECEASED ${person.id}');
384 }
385 for (final person in [first, second]) {
386 reject(graph.spouse(person.id) != null, 'ALREADY MARRIED ${person.id}');
387 }
388 reject(graph.ancestor(id, other) || graph.ancestor(other, id) ||
389 graph.siblings(id, other), 'CLOSE KIN $id $other');
390 for (final person in [first, second]) {
391 reject(year < person.birth + 16, 'TOO YOUNG ${person.id}');
392 }
393 for (final person in [first, second]) {
394 reject(family.marriages.any((m) => m.contains(person.id) &&
395 m.end != null && m.end! >= year), 'BAD YEAR ${person.id}');
396 }
397 remember(line);
398 family.marriages.add(Marriage(id, other, year));
399 return ['MARRIED $id $other $year'];
400 }
401 final marriage = graph.current(id);
402 if (marriage == null || marriage.other(id) != other) {
403 throw CommandError('NOT MARRIED $id $other');
404 }
405 reject(year <= marriage.start, 'BAD YEAR $id');
406 remember(line);
407 family.marriages[family.marriages.indexOf(marriage)] =
408 marriage.finish(year, MarriageStatus.divorced);
409 return ['DIVORCED $id $other $year'];
410 }

425 List<String> path(String a, String b, Analysis graph) {
426 final apex = graph.apex(a, b);
427 if (apex == null) return ['PATH $a $b none'];
428 final distance = graph.down(apex.id);
429 List<String> upward(String id) {
430 final result = [id];
431 while (id != apex.id) {
432 final candidates = graph.parents(id)
433 .where((p) => distance[p] == distance[id]! - 1).toList()..sort();
434 id = candidates.first;
435 result.add(id);
436 }
437 return result;
438 }
439 final first = upward(a), second = upward(b)..removeLast();
440 first.addAll(second.reversed);
441 return ['PATH $a $b ${apex.a + apex.b}: ${first.join(' ')}'];
442 }

457 final lines = ['TREE $id ${graph.down(id).length - 1}'];
458 final visited = <String>{};

460 final prefix = ' ' * (2 * (depth + 1));

464 }

466 lines.add('$prefix$current ${person.name} ${person.life}'

476 List<String> pedigree(String id, int depth, Analysis graph) {
477 final lines = <String>[], distinct = <String>{};
478 void enter(String current, int number, int generation) {
479 final person = graph.person(current);
480 distinct.add(current);

495 List<String> timeline(String id, Analysis graph) {
496 final person = graph.person(id);
497 final events = <(int, int, String, String)>[(person.birth, 0, '', 'born')];
498 for (final marriage in family.marriages.where((m) => m.contains(id))) {
499 final other = marriage.other(id);
500 events.add((marriage.start, 1, other, 'married $other'));
501 if (marriage.status == MarriageStatus.divorced) {
502 events.add((marriage.end!, 3, other, 'divorced $other'));
503 } else if (marriage.status == MarriageStatus.widowed && marriage.died == other) {
504 events.add((marriage.end!, 4, other, 'widowed $other'));
505 }
506 }
507 for (final child in graph.children[id]!) {
508 events.add((graph.person(child).birth, 2, child, 'child $child'));

510 void lost(Iterable<String> relatives, String kind) {
511 for (final other in relatives) {
512 final year = graph.person(other).death;
513 if (year != null && year >= person.birth &&
514 (person.death == null || year <= person.death!)) {
515 events.add((year, 5, other, 'lost $kind $other'));
516 }
517 }
518 }
519 lost(person.parents, 'parent');
520 lost(graph.children[id]!, 'child');
521 if (person.death != null) events.add((person.death!, 6, '', 'died'));
522 events.sort((a, b) {
523 var order = a.$1.compareTo(b.$1);
524 if (order == 0) order = a.$2.compareTo(b.$2);
525 return order != 0 ? order : a.$3.compareTo(b.$3);
526 });
527 return ['TIMELINE $id ${events.length}',
528 for (final event in events) ' ${event.$1} ${event.$4}'];
529 }

533 final person = graph.person(id);

537 for (final id in residents) {

539 for (final marriage in family.marriages) {

543 last.end == null || last.end! > year ? 'married ${last.other(id)}' : last.status.name;
544 lines.add(' $id ${year - graph.person(id).birth} $status');
545 }
546 return lines;

549 List<String> find(String id, String route, Analysis graph) {
550 var found = <String>{id};
551 for (final step in route.split('.')) {
552 final next = <String>{};
553 final sex = switch (step) {
554 'father' || 'son' || 'brother' || 'husband' => Sex.male,
555 'mother' || 'daughter' || 'sister' || 'wife' => Sex.female,
556 _ => null,
557 };
558 for (final current in found) {
559 Iterable<String> candidates;
560 switch (step) {
561 case 'parent' || 'father' || 'mother':
562 candidates = graph.parents(current);
563 case 'child' || 'son' || 'daughter':
564 candidates = graph.children[current]!;
565 case 'sibling' || 'brother' || 'sister':
566 candidates = graph.ids.where((other) => graph.siblings(current, other));
567 default:
568 final spouse = graph.spouse(current);
569 candidates = spouse == null ? const [] : [spouse];
570 }
571 next.addAll(candidates.where((other) => sex == null || graph.person(other).sex == sex));
572 }
573 found = next;
574 }
575 found.remove(id);
576 final ordered = found.toList()..sort();
577 return ['FIND $id $route ${ordered.length}',
578 for (final other in ordered) ' $other: ${graph.relation(other, id)}'];
579 }

581 List<String> execute(String line) {
582 final args = line.split(' '), command = args[0];
583 if (command == 'UNDO') {
584 reject(history.isEmpty, 'NOTHING TO UNDO');
585 final (previous, state) = history.removeLast();
586 family = state;
587 return ['UNDONE $previous'];
588 }
589 final graph = Analysis(family);
590 if (const {'PERSON', 'DEATH', 'REMOVE', 'PARENT', 'UNPARENT', 'SPOUSE', 'DIVORCE'}
591 .contains(command)) return modify(line, args, graph);
592 if (command == 'TERM') {
593 return ['$line ${term(int.parse(args[1]), int.parse(args[2]), Sex.parse(args[3]))}'];
594 }
595 if (command == 'CENSUS') return census(int.parse(args[1]), graph);
596 if (command == 'STATS') {
597 final lengths = <String, int>{};
598 var longest = 0, links = 0, living = 0, founders = 0;
599 for (final id in graph.byAge(graph.ids)) {
600 final person = graph.person(id);
601 var length = 1;
602 for (final parent in person.parents) {
603 length = math.max(length, lengths[parent]! + 1);
604 }
605 lengths[id] = length;
606 longest = math.max(longest, length);
607 links += person.parents.length;
608 if (person.parents.isEmpty) founders++;
609 if (person.death == null) living++;
610 }
611 final couples = family.marriages.where((m) => m.status == MarriageStatus.current).length;
612 return ['STATS persons=${family.people.length} living=$living links=$links '
613 'couples=$couples founders=$founders generations=$longest'];
614 }
615 if (command == 'AUDIT') {
616 final lines = <String>[];
617 for (final marriage in family.marriages) {
618 final apex = graph.apex(marriage.first, marriage.second);
619 if (apex != null) {
620 lines.add(' marriage ${marriage.first} ${marriage.second} '
621 '${graph.blood(apex, graph.person(marriage.first).sex)}');
622 }
623 }
624 for (final id in graph.ids) {
625 final value = graph.inbreeding(id);
626 if (value.numerator != 0) lines.add(' inbred $id $value');
627 }
628 return ['AUDIT ${lines.length}', ...lines];
629 }
630 final id = args[1];
631 check(id);
632 if (const {'RELATION', 'PATH', 'COMMON', 'KINSHIP'}.contains(command)) {
633 final other = args[2];
634 check(other);
635 return switch (command) {
636 'RELATION' => ['RELATION $id $other: ${graph.relation(id, other)}'],
637 'PATH' => path(id, other, graph),
638 'COMMON' => common(id, other, graph),
639 _ => ['KINSHIP $id $other ${graph.phi(id, other)}'],
640 };
641 }
642 switch (command) {
643 case 'INFO': return info(id, graph);
644 case 'TREE': return tree(id, graph);
645 case 'PEDIGREE': return pedigree(id, int.parse(args[2]), graph);
646 case 'TIMELINE': return timeline(id, graph);
647 case 'FIND': return find(id, args[2], graph);
648 case 'RELATIVES':
649 case 'WHO':
650 final query = args.skip(2).join(' '), lines = <String>[];
651 for (final other in graph.ids.where((other) => other != id)) {
652 final terms = graph.relations(other, id);
653 if (command == 'RELATIVES') {
654 if (terms.isNotEmpty) lines.add(' $other: ${terms.join(', ')}');
655 } else {
656 final neutral = graph.relations(other, id, Sex.other);
657 for (var i = 0; i < terms.length; i++) {
658 if (terms[i] == query || neutral[i] == query) {
659 lines.add(' $other: ${terms[i]}');
660 break;
661 }
662 }
663 }
664 }
665 return ['${command == 'WHO' ? 'WHO $id $query' : 'RELATIVES $id'} ${lines.length}', ...lines];
666 case 'ANCESTORS':
667 case 'DESCENDANTS':
668 final upward = command == 'ANCESTORS';
669 final distances = upward ? graph.up(id) : graph.down(id);
670 final others = distances.keys.where((other) => other != id).toList()
671 ..sort((a, b) {
672 final order = distances[a]!.compareTo(distances[b]!);
673 return order != 0 ? order : a.compareTo(b);
674 });
675 return ['$command $id ${others.length}', for (final other in others)
676 ' ${distances[other]} $other ${term(upward ? 0 : distances[other]!, upward ? distances[other]! : 0, graph.person(other).sex)}'];
677 case 'COUSINS':
678 final degree = int.parse(args[2]), matches = <(String, Apex)>[];
679 for (final other in graph.ids.where((other) => other != id)) {
680 final apex = graph.apex(other, id);
681 if (apex != null && apex.a >= 2 && apex.b >= 2 &&
682 math.min(apex.a, apex.b) - 1 == degree) matches.add((other, apex));
683 }
684 matches.sort((a, b) {
685 final order = (a.$2.a - a.$2.b).abs().compareTo((b.$2.a - b.$2.b).abs());
686 return order != 0 ? order : a.$1.compareTo(b.$1);
687 });
688 return ['COUSINS $id $degree ${matches.length}', for (final (other, apex) in matches)
689 ' $other ${graph.blood(apex, graph.person(other).sex)}'];
690 case 'GENERATIONS':
691 final groups = SplayTreeMap<int, List<Person>>();
692 for (final entry in graph.down(id).entries) {
693 if (entry.value > 0) {
694 groups.putIfAbsent(entry.value, () => []).add(graph.person(entry.key));
695 }
696 }
697 return ['GENERATIONS $id ${groups.length}', for (final entry in groups.entries)
698 ' ${entry.key} ${entry.value.length} ${entry.value.where((p) => p.death == null).length} '
699 '${entry.value.map((p) => p.birth).reduce(math.min)}-${entry.value.map((p) => p.birth).reduce(math.max)}'];
700 default:
701 final rows = <String>[];
702 var order = 1;
703 for (final other in graph.byAge(graph.ids.where((other) => other != id))) {
704 final shared = graph.sharedParents(id, other);
705 final kind = shared == 2 ? 'full' : shared == 1 ? 'half' :
706 graph.stepSiblings(id, other) ? 'step' : '';
707 if (kind.isEmpty) continue;
708 if (shared > 0 && graph.compareAge(other, id) < 0) order++;
709 rows.add(' $other $kind ${graph.person(other).birth}');
710 }
711 return ['SIBLINGS $id ${rows.length} order=$order', ...rows];
712 }
713 }
714}
