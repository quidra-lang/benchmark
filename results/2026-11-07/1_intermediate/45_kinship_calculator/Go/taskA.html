34func sexIndex(sex string) int {

36 return 0

38 if sex == "F" {
39 return 1

48func generationPrefix(g int) string {
49 if g == 1 {
50 return ""
51 }
52 return strings.Repeat("great-", g-2) + "grand"
53}

55func ordinal(k int) string {
56 words := []string{"first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth", "ninth", "tenth"}
57 if k <= 10 {
58 return words[k-1]
59 }
60 suffix := "th"
61 if k%100 < 11 || k%100 > 13 {
62 switch k % 10 {
63 case 1:
64 suffix = "st"
65 case 2:
66 suffix = "nd"
67 case 3:
68 suffix = "rd"
69 }
70 }
71 return strconv.Itoa(k) + suffix
72}

86func kinshipTerm(da, db int, sex string) string {
87 switch {
88 case da == 0 && db == 0:
89 return "self"
90 case da == 0:
91 return generationPrefix(db) + gendered("parent", sex)
92 case db == 0:
93 return generationPrefix(da) + gendered("child", sex)
94 case da == 1 && db == 1:
95 return gendered("sibling", sex)
96 case da == 1:
97 return generationPrefix(db-1) + gendered("pibling", sex)
98 case db == 1:
99 return generationPrefix(da-1) + gendered("nibling", sex)
100 }
101 removal := ""
102 switch r := abs(da - db); r {
103 case 0:
104 case 1:
105 removal = " once removed"
106 case 2:
107 removal = " twice removed"
108 default:
109 removal = fmt.Sprintf(" %d times removed", r)
110 }
111 return ordinal(min(da, db)-1) + " cousin" + removal
112}

114func fraction(value int64) string {
115 a, b := value, scale
116 for b != 0 {
117 a, b = b, a%b
118 }
119 return fmt.Sprintf("%d/%d", value/a, scale/a)
120}

312func (f *Family) partners(x string) map[string]bool {
313 latest := make(map[string]*Marriage)

317 result := make(map[string]bool)
318 for y, m := range latest {

326func (f *Family) stepRelated(a, b string) bool {
327 for _, p := range f.Persons[a].Parents {
328 partners := f.partners(p)
329 for _, q := range f.Persons[b].Parents {
330 if partners[q] {
331 return true
332 }

335 return false

338func (f *Family) distances(start string, upward bool) map[string]int {
339 key := fmt.Sprintf("%t:%s", upward, start)
340 if cached, ok := f.Distances[key]; ok {
341 return cached
342 }
343 distances := map[string]int{start: 0}
344 queue := []string{start}
345 for i := 0; i < len(queue); i++ {
346 x := queue[i]
347 var neighbors []string
348 if upward {
349 neighbors = f.Persons[x].Parents
350 } else {
351 neighbors = f.children(x)
352 }
353 for _, y := range neighbors {
354 if _, seen := distances[y]; !seen {
355 distances[y] = distances[x] + 1
356 queue = append(queue, y)
357 }
358 }
359 }
360 f.Distances[key] = distances
361 return distances
362}

379func (f *Family) apex(a, b string) *Apex {
380 upA, upB := f.distances(a, true), f.distances(b, true)
381 if g, ok := upB[a]; ok {
382 return &Apex{Person: a, DB: g}
383 }
384 if g, ok := upA[b]; ok {
385 return &Apex{Person: b, DA: g}
386 }
387 var best *Apex
388 for c, da := range upA {
389 db, ok := upB[c]
390 if !ok {
391 continue
392 }
393 if best == nil || da+db < best.DA+best.DB ||
394 (da+db == best.DA+best.DB && (da < best.DA || (da == best.DA && c < best.Person))) {
395 best = &Apex{Person: c, DA: da, DB: db}
396 }
397 }
398 if best != nil {
399 for c, da := range upA {
400 if db, ok := upB[c]; ok && da == best.DA && db == best.DB {
401 best.Shares++
402 }
403 }
404 }
405 return best
406}

408func (f *Family) relationship(a, b, sex string) []string {
409 if a == b {
410 return []string{"self"}
411 }
412 var result []string
413 if apex := f.apex(a, b); apex != nil {
414 result = append(result, apex.bloodTerm(sex))
415 }
416 m := f.last(a, b)
417 if m == nil {
418 for _, p := range f.Persons {
419 if has(p.Parents, a) && has(p.Parents, b) {
420 result = append(result, "co-parent")
421 break
422 }
423 }
424 } else {
425 kind := "ex"
426 if m.Status == "current" {
427 kind = "spouse"
428 } else if m.Status == "widowed" {
429 kind = "survivor"
430 if m.Died == a {
431 kind = "late"
432 }
433 }
434 result = append(result, gendered(kind, sex))
435 }
436 pa, pb := f.Persons[a].Parents, f.Persons[b].Parents
437 qa, qb := f.partners(a), f.partners(b)
438 if !has(pb, a) {
439 for _, p := range pb {
440 if qa[p] {
441 result = append(result, gendered("stepparent", sex))
442 break
443 }
444 }
445 }
446 if !has(pa, b) {
447 for _, p := range pa {
448 if qb[p] {
449 result = append(result, gendered("stepchild", sex))
450 break
451 }
452 }
453 }
454 if f.commonParents(a, b) == 0 && f.stepRelated(a, b) {
455 result = append(result, gendered("stepsibling", sex))
456 }
457 if !f.ancestor(a, b) {
458 for _, g := range f.grandparents(b) {
459 if qa[g] {
460 result = append(result, gendered("stepgrandparent", sex))
461 break
462 }
463 }
464 }
465 if !f.ancestor(b, a) {
466 for _, g := range f.grandparents(a) {
467 if qb[g] {
468 result = append(result, gendered("stepgrandchild", sex))
469 break
470 }
471 }
472 }
473 for q := range qb {
474 if has(f.Persons[q].Parents, a) {
475 result = append(result, gendered("parent-in-law", sex))
476 break
477 }
478 }
479 for q := range qa {
480 if has(f.Persons[q].Parents, b) {
481 result = append(result, gendered("child-in-law", sex))
482 break
483 }
484 }
485 siblingInLaw := false
486 for q := range qb {
487 if f.areSiblings(a, q) {
488 siblingInLaw = true
489 }
490 }
491 for q := range qa {
492 if f.areSiblings(q, b) {
493 siblingInLaw = true
494 }
495 }
496 if siblingInLaw {
497 result = append(result, gendered("sibling-in-law", sex))
498 }
499 return result
500}

510func (f *Family) coefficient(a, b string) int64 {
511 key := [2]string{a, b}
512 if a > b {
513 key = [2]string{b, a}
514 }
515 if value, ok := f.Coefficients[key]; ok {
516 return value
517 }
518 var value int64
519 if a == b {
520 value = (scale + f.inbreeding(a)) / 2
521 } else {
522 x, y := a, b
523 if f.Persons[b].Birth > f.Persons[a].Birth {
524 x, y = b, a
525 }
526 for _, p := range f.Persons[x].Parents {
527 value += f.coefficient(p, y)
528 }
529 value /= 2
530 }
531 f.Coefficients[key] = value
532 return value
533}

535func (f *Family) inbreeding(x string) int64 {
536 parents := f.Persons[x].Parents
537 if len(parents) != 2 {
538 return 0
539 }
540 return f.coefficient(parents[0], parents[1])
541}

543func (f *Family) lineageLength(x string, memo map[string]int) int {
544 if length, ok := memo[x]; ok {
545 return length
546 }
547 length := 1
548 for _, p := range f.Persons[x].Parents {
549 length = max(length, 1+f.lineageLength(p, memo))
550 }
551 memo[x] = length
552 return length
553}

555func (f *Family) modify(command string, args []string) (string, error) {
556 x := args[0]
557 if command == "PERSON" {
558 if f.Persons[x] != nil {
559 return "", fmt.Errorf("DUPLICATE PERSON %s", x)
560 }
561 f.Persons[x] = &Person{Sex: args[1], Birth: number(args[2]), Name: args[3]}
562 return fmt.Sprintf("ADDED %s %s %s %s", x, args[3], args[1], args[2]), nil
563 }
564 if f.Persons[x] == nil {
565 return "", fmt.Errorf("UNKNOWN PERSON %s", x)
566 }
567 if command == "PARENT" || command == "UNPARENT" || command == "SPOUSE" || command == "DIVORCE" {
568 if f.Persons[args[1]] == nil {
569 return "", fmt.Errorf("UNKNOWN PERSON %s", args[1])
570 }
571 }
572 p := f.Persons[x]
573 switch command {
574 case "DEATH":
575 year := number(args[1])
576 if p.Death != 0 {
577 return "", fmt.Errorf("ALREADY DECEASED %s", x)
578 }
579 bad := year < p.Birth
580 for _, m := range f.records(x) {
581 if year < m.Start || (m.End != 0 && year < m.End) {
582 bad = true
583 }
584 }
585 if bad {
586 return "", fmt.Errorf("BAD YEAR %s", x)
587 }
588 children := f.children(x)
589 sort.Strings(children)
590 for _, c := range children {
591 if f.Persons[c].Birth > year+1 {
592 return "", fmt.Errorf("AFTER DEATH %s %s", c, x)
593 }
594 }
595 p.Death = year
596 m := f.current(x)
597 if m == nil {
598 return fmt.Sprintf("DIED %s %d", x, year), nil
599 }
600 m.End, m.Status, m.Died = year, "widowed", x
601 return fmt.Sprintf("DIED %s %d spouse=%s", x, year, m.other(x)), nil
602 case "REMOVE":
603 for _, person := range f.Persons {
604 person.Parents = without(person.Parents, x)
605 }
606 var records []*Marriage
607 for _, m := range f.Marriages {
608 if !m.contains(x) {
609 records = append(records, m)
610 }
611 }
612 f.Marriages = records
613 delete(f.Persons, x)
614 return "REMOVED " + x, nil
615 case "PARENT":
616 y := args[1]
617 parent := f.Persons[y]
618 if x == y {
619 return "", fmt.Errorf("SAME PERSON %s", x)
620 }
621 if has(p.Parents, y) {
622 return "", fmt.Errorf("ALREADY PARENT %s %s", x, y)
623 }
624 if len(p.Parents) == 2 {
625 return "", fmt.Errorf("TOO MANY PARENTS %s", x)
626 }
627 if f.ancestor(x, y) {
628 return "", fmt.Errorf("CYCLE %s %s", x, y)
629 }
630 if parent.Birth+12 > p.Birth {
631 return "", fmt.Errorf("AGE GAP %s %s", x, y)
632 }
633 if parent.Death != 0 && parent.Death+1 < p.Birth {
634 return "", fmt.Errorf("AFTER DEATH %s %s", x, y)
635 }
636 p.Parents = append(p.Parents, y)
637 return fmt.Sprintf("LINKED %s %s %s", x, y, gendered("parent", parent.Sex)), nil
638 case "UNPARENT":
639 y := args[1]
640 if !has(p.Parents, y) {
641 return "", fmt.Errorf("NOT PARENT %s %s", x, y)
642 }
643 p.Parents = without(p.Parents, y)
644 return fmt.Sprintf("UNLINKED %s %s", x, y), nil
645 case "SPOUSE":
646 y, year := args[1], number(args[2])
647 if x == y {
648 return "", fmt.Errorf("SAME PERSON %s", x)
649 }
650 for _, a := range []string{x, y} {
651 if f.Persons[a].Death != 0 {
652 return "", fmt.Errorf("DECEASED %s", a)
653 }
654 }
655 for _, a := range []string{x, y} {
656 if f.spouse(a) != "" {
657 return "", fmt.Errorf("ALREADY MARRIED %s", a)
658 }
659 }
660 if f.ancestor(x, y) || f.ancestor(y, x) || f.areSiblings(x, y) {
661 return "", fmt.Errorf("CLOSE KIN %s %s", x, y)
662 }
663 for _, a := range []string{x, y} {
664 if year < f.Persons[a].Birth+16 {
665 return "", fmt.Errorf("TOO YOUNG %s", a)
666 }
667 }
668 for _, a := range []string{x, y} {
669 for _, m := range f.records(a) {
670 if m.End != 0 && m.End >= year {
671 return "", fmt.Errorf("BAD YEAR %s", a)
672 }
673 }
674 }
675 f.Marriages = append(f.Marriages, &Marriage{First: x, Second: y, Start: year, Status: "current"})
676 return fmt.Sprintf("MARRIED %s %s %d", x, y, year), nil
677 case "DIVORCE":
678 y, year := args[1], number(args[2])
679 if f.spouse(x) != y {
680 return "", fmt.Errorf("NOT MARRIED %s %s", x, y)
681 }
682 m := f.current(x)
683 if year <= m.Start {
684 return "", fmt.Errorf("BAD YEAR %s", x)
685 }
686 m.End, m.Status = year, "divorced"
687 return fmt.Sprintf("DIVORCED %s %s %d", x, y, year), nil
688 }
689 panic("unknown modifying command")
690}

692type LineageEntry struct {
693 ID string
694 Distance int
695}

697func (f *Family) lineage(x string, upward bool) []LineageEntry {
698 var result []LineageEntry
699 for y, g := range f.distances(x, upward) {
700 if y != x {
701 result = append(result, LineageEntry{y, g})
702 }
703 }
704 sort.Slice(result, func(i, j int) bool {
705 return result[i].Distance < result[j].Distance || (result[i].Distance == result[j].Distance && result[i].ID < result[j].ID)
706 })
707 return result
708}

710func (f *Family) query(command string, args []string) ([]string, error) {
711 switch command {
712 case "TERM":
713 return []string{fmt.Sprintf("TERM %s %s %s %s", args[0], args[1], args[2], kinshipTerm(number(args[0]), number(args[1]), args[2]))}, nil
714 case "STATS":
715 living, links, couples, founders, longest := 0, 0, 0, 0, 0
716 memo := make(map[string]int)
717 for x, p := range f.Persons {
718 if p.Death == 0 {
719 living++
720 }
721 links += len(p.Parents)
722 if len(p.Parents) == 0 {
723 founders++
724 }
725 longest = max(longest, f.lineageLength(x, memo))
726 }
727 for _, m := range f.Marriages {
728 if m.Status == "current" {
729 couples++
730 }
731 }
732 return []string{fmt.Sprintf("STATS persons=%d living=%d links=%d couples=%d founders=%d generations=%d", len(f.Persons), living, links, couples, founders, longest)}, nil
733 case "AUDIT":
734 var lines []string
735 for _, m := range f.Marriages {
736 if apex := f.apex(m.First, m.Second); apex != nil {
737 lines = append(lines, fmt.Sprintf(" marriage %s %s %s", m.First, m.Second, apex.bloodTerm(f.Persons[m.First].Sex)))
738 }
739 }
740 for _, x := range f.ids() {
741 if value := f.inbreeding(x); value != 0 {
742 lines = append(lines, fmt.Sprintf(" inbred %s %s", x, fraction(value)))
743 }
744 }
745 return append([]string{fmt.Sprintf("AUDIT %d", len(lines))}, lines...), nil
746 case "CENSUS":
747 year := number(args[0])
748 var residents []string
749 for x, p := range f.Persons {
750 if p.Birth <= year && (p.Death == 0 || p.Death > year) {
751 residents = append(residents, x)
752 }
753 }
754 lines := []string{fmt.Sprintf("CENSUS %d %d", year, len(residents))}
755 for _, x := range f.ageOrder(residents) {
756 var last *Marriage
757 for _, m := range f.records(x) {
758 if m.Start <= year {
759 last = m
760 }
761 }
762 status := "single"
763 if last != nil {
764 if last.End == 0 || last.End > year {
765 status = "married " + last.other(x)
766 } else {
767 status = last.Status
768 }
769 }
770 lines = append(lines, fmt.Sprintf(" %s %d %s", x, year-f.Persons[x].Birth, status))
771 }
772 return lines, nil
773 }
774 x := args[0]
775 if f.Persons[x] == nil {
776 return nil, fmt.Errorf("UNKNOWN PERSON %s", x)
777 }
778 if command == "RELATION" || command == "PATH" || command == "COMMON" || command == "KINSHIP" {
779 if f.Persons[args[1]] == nil {
780 return nil, fmt.Errorf("UNKNOWN PERSON %s", args[1])
781 }
782 }
783 p := f.Persons[x]
784 switch command {
785 case "INFO":
786 lines := []string{fmt.Sprintf("INFO %s %s %s %s", x, p.Name, p.Sex, p.life()), " parents: " + f.personList(p.Parents)}
787 for _, m := range f.records(x) {
788 end := ""
789 if m.End != 0 {
790 end = strconv.Itoa(m.End)
791 }
792 lines = append(lines, fmt.Sprintf(" marriage: %s %d-%s %s", m.other(x), m.Start, end, m.Status))
793 }
794 lines = append(lines, " children: "+f.personList(f.children(x)), " siblings: "+f.personList(f.siblings(x)), " inbreeding: "+fraction(f.inbreeding(x)))
795 return lines, nil
796 case "RELATION":
797 return []string{fmt.Sprintf("RELATION %s %s: %s", x, args[1], f.relationshipText(x, args[1]))}, nil
798 case "KINSHIP":
799 return []string{fmt.Sprintf("KINSHIP %s %s %s", x, args[1], fraction(f.coefficient(x, args[1])))}, nil
800 case "RELATIVES", "WHO":
801 var lines []string
802 query := strings.Join(args[1:], " ")
803 for _, y := range f.ids() {
804 if y == x {
805 continue
806 }
807 actual := f.relationship(y, x, f.Persons[y].Sex)
808 if command == "RELATIVES" {
809 if len(actual) > 0 {
810 lines = append(lines, fmt.Sprintf(" %s: %s", y, strings.Join(actual, ", ")))
811 }
812 } else {
813 neutral := f.relationship(y, x, "X")
814 for i, t := range actual {
815 if t == query || neutral[i] == query {
816 lines = append(lines, fmt.Sprintf(" %s: %s", y, t))
817 break
818 }
819 }
820 }
821 }
822 header := fmt.Sprintf("RELATIVES %s %d", x, len(lines))
823 if command == "WHO" {
824 header = fmt.Sprintf("WHO %s %s %d", x, query, len(lines))
825 }
826 return append([]string{header}, lines...), nil
827 case "PATH":
828 y := args[1]
829 apex := f.apex(x, y)
830 if apex == nil {
831 return []string{fmt.Sprintf("PATH %s %s none", x, y)}, nil
832 }
833 depths := f.distances(apex.Person, false)
834 upward := func(start string) []string {
835 path := []string{start}
836 for current := start; current != apex.Person; {
837 var choices []string
838 for _, parent := range f.Persons[current].Parents {
839 if d, ok := depths[parent]; ok && d == depths[current]-1 {
840 choices = append(choices, parent)
841 }
842 }
843 sort.Strings(choices)
844 current = choices[0]
845 path = append(path, current)
846 }
847 return path
848 }
849 route, descent := upward(x), upward(y)
850 for i := len(descent) - 2; i >= 0; i-- {
851 route = append(route, descent[i])
852 }
853 return []string{fmt.Sprintf("PATH %s %s %d: %s", x, y, apex.DA+apex.DB, strings.Join(route, " "))}, nil
854 case "COMMON":
855 y := args[1]
856 a, b := f.distances(x, true), f.distances(y, true)
857 shared := make(map[string]bool)
858 for c := range a {
859 if _, ok := b[c]; ok {
860 shared[c] = true
861 }
862 }
863 var nearest []string
864 for c := range shared {
865 ok := true
866 for _, child := range f.children(c) {
867 if shared[child] {
868 ok = false
869 break
870 }
871 }
872 if ok {
873 nearest = append(nearest, c)
874 }
875 }
876 sort.Slice(nearest, func(i, j int) bool {
877 c, d := nearest[i], nearest[j]
878 return a[c]+b[c] < a[d]+b[d] || (a[c]+b[c] == a[d]+b[d] && c < d)
879 })
880 lines := []string{fmt.Sprintf("COMMON %s %s %d", x, y, len(nearest))}
881 for _, c := range nearest {
882 lines = append(lines, fmt.Sprintf(" %s %d %d", c, a[c], b[c]))
883 }
884 return lines, nil
885 case "ANCESTORS", "DESCENDANTS":
886 upward := command == "ANCESTORS"
887 entries := f.lineage(x, upward)
888 lines := []string{fmt.Sprintf("%s %s %d", command, x, len(entries))}
889 for _, e := range entries {
890 da, db := e.Distance, 0
891 if upward {
892 da, db = 0, e.Distance
893 }
894 lines = append(lines, fmt.Sprintf(" %d %s %s", e.Distance, e.ID, kinshipTerm(da, db, f.Persons[e.ID].Sex)))
895 }
896 return lines, nil
897 case "COUSINS":
898 type Match struct {
899 ID string
900 Apex *Apex
901 }
902 var matches []Match
903 for y := range f.Persons {
904 if x == y {
905 continue
906 }
907 apex := f.apex(y, x)
908 if apex != nil && apex.DA >= 2 && apex.DB >= 2 && min(apex.DA, apex.DB)-1 == number(args[1]) {
909 matches = append(matches, Match{y, apex})
910 }
911 }
912 sort.Slice(matches, func(i, j int) bool {
913 a, b := matches[i], matches[j]
914 da, db := abs(a.Apex.DA-a.Apex.DB), abs(b.Apex.DA-b.Apex.DB)
915 return da < db || (da == db && a.ID < b.ID)
916 })
917 lines := []string{fmt.Sprintf("COUSINS %s %s %d", x, args[1], len(matches))}
918 for _, m := range matches {
919 lines = append(lines, fmt.Sprintf(" %s %s", m.ID, m.Apex.bloodTerm(f.Persons[m.ID].Sex)))
920 }
921 return lines, nil
922 case "SIBLINGS":
923 var lines []string
924 earlier := 0
925 for _, y := range f.ageOrder(f.ids()) {
926 if y == x {
927 continue
928 }
929 common := f.commonParents(x, y)
930 kind := ""
931 if common == 2 {
932 kind = "full"
933 } else if common == 1 {
934 kind = "half"
935 } else if f.stepRelated(x, y) {
936 kind = "step"
937 }
938 if kind == "" {
939 continue
940 }
941 if common > 0 && f.ageLess(y, x) {
942 earlier++
943 }
944 lines = append(lines, fmt.Sprintf(" %s %s %d", y, kind, f.Persons[y].Birth))
945 }
946 return append([]string{fmt.Sprintf("SIBLINGS %s %d order=%d", x, len(lines), earlier+1)}, lines...), nil
947 case "GENERATIONS":
948 groups := make(map[int][]string)
949 for _, e := range f.lineage(x, false) {
950 groups[e.Distance] = append(groups[e.Distance], e.ID)
951 }
952 var generations []int
953 for g := range groups {
954 generations = append(generations, g)
955 }
956 sort.Ints(generations)
957 lines := []string{fmt.Sprintf("GENERATIONS %s %d", x, len(groups))}
958 for _, g := range generations {
959 members := groups[g]
960 lo, hi, living := 2025, 1800, 0
961 for _, y := range members {
962 p := f.Persons[y]
963 lo, hi = min(lo, p.Birth), max(hi, p.Birth)
964 if p.Death == 0 {
965 living++
966 }
967 }
968 lines = append(lines, fmt.Sprintf(" %d %d %d %d-%d", g, len(members), living, lo, hi))
969 }
970 return lines, nil
971 case "TREE":
972 lines := []string{fmt.Sprintf("TREE %s %d", x, len(f.lineage(x, false)))}
973 visited := make(map[string]bool)
974 var visit func(string, int)
975 visit = func(y string, depth int) {
976 indent := strings.Repeat(" ", 2*(depth+1))
977 if visited[y] {
978 lines = append(lines, indent+y+" (see above)")
979 return
980 }
981 visited[y] = true
982 p := f.Persons[y]
983 line := fmt.Sprintf("%s%s %s %s", indent, y, p.Name, p.life())
984 if spouse := f.spouse(y); spouse != "" {
985 line += " = " + spouse
986 }
987 lines = append(lines, line)
988 for _, child := range f.ageOrder(f.children(y)) {
989 visit(child, depth+1)
990 }
991 }
992 visit(x, 0)
993 return lines, nil
994 case "PEDIGREE":
995 var entries []string
996 entered := make(map[string]bool)
997 depth := number(args[1])
998 var enter func(string, int, int)
999 enter = func(y string, k, g int) {
1000 p := f.Persons[y]
1001 entries = append(entries, fmt.Sprintf("%s%d %s %s %s", strings.Repeat(" ", 2*(g+1)), k, y, p.Name, p.life()))
1002 entered[y] = true
1003 if g >= depth {
1004 return
1005 }
1006 parents := append([]string(nil), p.Parents...)
1007 sort.Slice(parents, func(i, j int) bool {
1008 a, b := parents[i], parents[j]
1009 sa, sb := sexIndex(f.Persons[a].Sex), sexIndex(f.Persons[b].Sex)
1010 return sa < sb || (sa == sb && a < b)
1011 })
1012 for i, parent := range parents {
1013 enter(parent, 2*k+i, g+1)
1014 }
1015 }
1016 enter(x, 1, 0)
1017 return append([]string{fmt.Sprintf("PEDIGREE %s %d entries=%d distinct=%d", x, depth, len(entries), len(entered))}, entries...), nil
1018 case "TIMELINE":
1019 type Event struct {
1020 Year, Rank int
1021 Other, Text string
1022 }
1023 events := []Event{{p.Birth, 0, "", "born"}}
1024 for _, m := range f.records(x) {
1025 y := m.other(x)
1026 events = append(events, Event{m.Start, 1, y, "married " + y})
1027 if m.Status == "divorced" {
1028 events = append(events, Event{m.End, 3, y, "divorced " + y})
1029 } else if m.Status == "widowed" && m.Died == y {
1030 events = append(events, Event{m.End, 4, y, "widowed " + y})
1031 }
1032 }
1033 children := f.children(x)
1034 for _, y := range children {
1035 events = append(events, Event{f.Persons[y].Birth, 2, y, "child " + y})
1036 }
1037 for _, relatives := range []struct {
1038 Label string
1039 IDs []string
1040 }{{"parent", p.Parents}, {"child", children}} {
1041 for _, y := range relatives.IDs {
1042 death := f.Persons[y].Death
1043 if death != 0 && death >= p.Birth && (p.Death == 0 || death <= p.Death) {
1044 events = append(events, Event{death, 5, y, "lost " + relatives.Label + " " + y})
1045 }
1046 }
1047 }
1048 if p.Death != 0 {
1049 events = append(events, Event{p.Death, 6, "", "died"})
1050 }
1051 sort.Slice(events, func(i, j int) bool {
1052 a, b := events[i], events[j]
1053 if a.Year != b.Year {
1054 return a.Year < b.Year
1055 }
1056 if a.Rank != b.Rank {
1057 return a.Rank < b.Rank
1058 }
1059 return a.Other < b.Other
1060 })
1061 lines := []string{fmt.Sprintf("TIMELINE %s %d", x, len(events))}
1062 for _, e := range events {
1063 lines = append(lines, fmt.Sprintf(" %d %s", e.Year, e.Text))
1064 }
1065 return lines, nil
1066 case "FIND":
1067 steps := map[string][2]string{
1068 "parent": {"parent", ""},
1069 "father": {"parent", "M"},
1070 "mother": {"parent", "F"},
1071 "child": {"child", ""},
1072 "son": {"child", "M"},
1073 "daughter": {"child", "F"},
1074 "sibling": {"sibling", ""},
1075 "brother": {"sibling", "M"},
1076 "sister": {"sibling", "F"},
1077 "spouse": {"spouse", ""},
1078 "husband": {"spouse", "M"},
1079 "wife": {"spouse", "F"},
1080 }
1081 current := map[string]bool{x: true}
1082 for _, step := range strings.Split(args[1], ".") {
1083 kind, sex := steps[step][0], steps[step][1]
1084 next := make(map[string]bool)
1085 for y := range current {
1086 var targets []string
1087 switch kind {
1088 case "parent":
1089 targets = f.Persons[y].Parents
1090 case "child":
1091 targets = f.children(y)
1092 case "sibling":
1093 targets = f.siblings(y)
1094 case "spouse":
1095 if s := f.spouse(y); s != "" {
1096 targets = []string{s}
1097 }
1098 }
1099 for _, z := range targets {
1100 if sex == "" || f.Persons[z].Sex == sex {
1101 next[z] = true
1102 }
1103 }
1104 }
1105 current = next
1106 }
1107 delete(current, x)
1108 var ids []string
1109 for y := range current {
1110 ids = append(ids, y)
1111 }
1112 sort.Strings(ids)
1113 lines := []string{fmt.Sprintf("FIND %s %s %d", x, args[1], len(ids))}
1114 for _, y := range ids {
1115 lines = append(lines, fmt.Sprintf(" %s: %s", y, f.relationshipText(y, x)))
1116 }
1117 return lines, nil
1118 }
1119 panic("unknown query")
1120}
