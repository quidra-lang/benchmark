102distances <- function(start, upward = TRUE) {
103 key <- paste0(if (upward) "up:" else "down:", start)
104 if (exists(key, distance_cache, inherits = FALSE)) return(distance_cache[[key]])
105 result <- setNames(0L, start)
106 queue <- start
107 position <- 1L
108 while (position <= length(queue)) {
109 x <- queue[position]
110 position <- position + 1L
111 neighbors <- if (upward) parents_of(x) else children_of(x)
112 for (y in neighbors) if (!(y %in% names(result))) {
113 result[y] <- result[[x]] + 1L
114 queue <- c(queue, y)
115 }
116 }
117 distance_cache[[key]] <- result
118 result
119}

125apex <- function(a, b) {
126 da <- distances(a)
127 db <- distances(b)
128 if (a %in% names(db)) return(list(id = a, da = 0L, db = db[[a]], shares = 0L))
129 if (b %in% names(da)) return(list(id = b, da = da[[b]], db = 0L, shares = 0L))
130 shared <- intersect(names(da), names(db))
131 if (!length(shared)) return(NULL)
132 best <- shared[order(da[shared] + db[shared], da[shared], shared, method = "radix")][1]
133 shares <- sum(da[shared] == da[[best]] & db[shared] == db[[best]])
134 list(id = best, da = da[[best]], db = db[[best]], shares = shares)
135}

140relationship <- function(a, b, sex = sex_of(a)) {
141 if (a == b) return("self")
142 result <- character()
143 top <- apex(a, b)
144 if (!is.null(top)) result <- c(result, blood_term(top, sex))
145 record <- last_record(a, b)
146 if (is.null(record)) {
147 common_child <- any(vapply(persons, function(p) a %in% p$parents && b %in% p$parents, logical(1)))
148 if (common_child) result <- c(result, "co-parent")
149 } else if (record$status == "current") result <- c(result, gendered("spouse", sex))
150 else if (record$status == "widowed") result <- c(result, gendered(if (record$died == a) "late" else "survivor", sex))
151 else result <- c(result, gendered("ex", sex))
152 pa <- parents_of(a)
153 pb <- parents_of(b)
154 partners_a <- partners_of(a)
155 partners_b <- partners_of(b)
156 if (!(a %in% pb) && any(pb %in% partners_a)) result <- c(result, gendered("stepparent", sex))
157 if (!(b %in% pa) && any(pa %in% partners_b)) result <- c(result, gendered("stepchild", sex))
158 if (common_parents(a, b) == 0L && step_related(pa, pb)) result <- c(result, gendered("stepsibling", sex))
159 if (!is_ancestor(a, b) && any(grandparents_of(b) %in% partners_a)) result <- c(result, gendered("stepgrandparent", sex))
160 if (!is_ancestor(b, a) && any(grandparents_of(a) %in% partners_b)) result <- c(result, gendered("stepgrandchild", sex))
161 if (any(vapply(partners_b, function(q) a %in% parents_of(q), logical(1)))) result <- c(result, gendered("parent-in-law", sex))
162 if (any(vapply(partners_a, function(q) b %in% parents_of(q), logical(1)))) result <- c(result, gendered("child-in-law", sex))
163 sibling_of_partner <- any(vapply(partners_b, function(q) are_siblings(a, q), logical(1)))
164 partner_of_sibling <- any(vapply(partners_a, function(q) are_siblings(q, b), logical(1)))
165 if (sibling_of_partner || partner_of_sibling) result <- c(result, gendered("sibling-in-law", sex))
166 result
167}

173coefficient_scale <- 2^37
174phi <- function(a, b) {
175 key <- paste(id_order(c(a, b)), collapse = ":")
176 if (exists(key, coefficient_cache, inherits = FALSE)) return(coefficient_cache[[key]])
177 if (a == b) value <- (coefficient_scale + inbreeding(a)) / 2
178 else {
179 if (persons[[a]]$birth >= persons[[b]]$birth) {
180 x <- a
181 y <- b
182 } else {
183 x <- b
184 y <- a
185 }
186 value <- sum(vapply(parents_of(x), function(p) phi(p, y), numeric(1))) / 2
187 }
188 coefficient_cache[[key]] <- value
189 value

211upward_path <- function(x, top) {
212 path <- x
213 while (x != top) {
214 target_distance <- distances(x)[[top]] - 1L
215 candidates <- parents_of(x)
216 candidates <- candidates[vapply(candidates, function(p) {
217 d <- distances(p)
218 top %in% names(d) && d[[top]] == target_distance
219 }, logical(1))]
220 x <- id_order(candidates)[1]
221 path <- c(path, x)
222 }
223 path

238 x <- if (length(words) > 1L) words[2] else ""

247 if (command == "DEATH") {
248 year <- as.integer(words[3])
249 person <- persons[[x]]

251 years <- person$birth
252 for (m in records_of(x)) years <- c(years, m$start, if (!is.na(m$end)) m$end)
253 if (year < max(years)) fail("BAD YEAR", x)
254 children <- id_order(children_of(x))
255 for (child in children) if (persons[[child]]$birth > year + 1L) fail("AFTER DEATH", child, x)
256 persons[[x]]$death <<- year

263 }

265 for (id in names(persons)) persons[[id]]$parents <<- setdiff(parents_of(id), x)

272 if (y %in% parents_of(x)) fail("ALREADY PARENT", x, y)
273 if (length(parents_of(x)) == 2L) fail("TOO MANY PARENTS", x)
274 if (is_ancestor(x, y)) fail("CYCLE", x, y)

276 if (!is.na(persons[[y]]$death) && persons[[y]]$death + 1L < persons[[x]]$birth) fail("AFTER DEATH", x, y)

279 }

474 }

485 current <- x
486 for (step in steps) {
487 kind <- if (step %in% c("parent", "father", "mother")) "parent" else if (step %in% c("child", "son", "daughter")) "child" else if (step %in% c("sibling", "brother", "sister")) "sibling" else "spouse"
488 sex <- if (step %in% c("father", "son", "brother", "husband")) "M" else if (step %in% c("mother", "daughter", "sister", "wife")) "F" else ""
489 collected <- character()
490 for (id in current) {
491 targets <- switch(kind, parent = parents_of(id), child = children_of(id), sibling = siblings_of(id), spouse = spouse_of(id))
492 if (nzchar(sex)) targets <- targets[vapply(targets, function(target) sex_of(target) == sex, logical(1))]
