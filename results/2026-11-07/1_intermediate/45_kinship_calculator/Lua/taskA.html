28local function generation_prefix(g)
29 if g == 1 then return "" end
30 return string.rep("great-", g - 2) .. "grand"
31end

33local function ordinal(k)
34 if k <= 10 then return ordinals[k] end
35 local suffix = "th"
36 if k % 100 < 11 or k % 100 > 13 then
37 suffix = ({[1] = "st", [2] = "nd", [3] = "rd"})[k % 10] or "th"
38 end
39 return k .. suffix
40end

42local function kinship_term(da, db, sex)
43 if da == 0 and db == 0 then return "self" end
44 if da == 0 then return generation_prefix(db) .. gendered("parent", sex) end
45 if db == 0 then return generation_prefix(da) .. gendered("child", sex) end
46 if da == 1 and db == 1 then return gendered("sibling", sex) end
47 if da == 1 then return generation_prefix(db - 1) .. gendered("pibling", sex) end
48 if db == 1 then return generation_prefix(da - 1) .. gendered("nibling", sex) end
49 local distance = math.abs(da - db)
50 local removal = ""
51 if distance == 1 then removal = " once removed"
52 elseif distance == 2 then removal = " twice removed"
53 elseif distance >= 3 then removal = " " .. distance .. " times removed" end
54 return ordinal(math.min(da, db) - 1) .. " cousin" .. removal
55end

215local function distances(start, upward)
216 local key = (upward and "up:" or "down:") .. start
217 if distance_cache[key] then return distance_cache[key] end
218 local result, queue = {[start] = 0}, {start}
219 local position = 1
220 while position <= #queue do
221 local x = queue[position]
222 position = position + 1
223 local neighbors = upward and parents_of(x) or children_of(x)
224 for _, y in ipairs(neighbors) do
225 if result[y] == nil then
226 result[y] = result[x] + 1
227 queue[#queue + 1] = y
228 end
229 end
230 end
231 distance_cache[key] = result
232 return result
233end

247local function apex(a, b)
248 local da, db = distances(a, true), distances(b, true)
249 if db[a] then return {id = a, da = 0, db = db[a], shares = 0} end
250 if da[b] then return {id = b, da = da[b], db = 0, shares = 0} end
251 local best
252 for id, first_distance in pairs(da) do
253 if db[id] then
254 if not best or first_distance + db[id] < da[best] + db[best]
255 or (first_distance + db[id] == da[best] + db[best]
256 and (first_distance < da[best] or (first_distance == da[best] and id < best))) then
257 best = id
258 end
259 end
260 end
261 if not best then return nil end
262 local shares = 0
263 for id, first_distance in pairs(da) do
264 if first_distance == da[best] and db[id] == db[best] then shares = shares + 1 end
265 end
266 return {id = best, da = da[best], db = db[best], shares = shares}
267end

274local function relationship(a, b, sex)
275 if a == b then return {"self"} end
276 sex = sex or persons[a].sex
277 local result = {}
278 local function add(kind) result[#result + 1] = gendered(kind, sex) end
279 local top = apex(a, b)
280 if top then result[#result + 1] = blood_term(top, sex) end
281 local record = last_record(a, b)
282 if not record then
283 for _, person in pairs(persons) do
284 if has(person.parents, a) and has(person.parents, b) then
285 result[#result + 1] = "co-parent"
286 break
287 end
288 end
289 elseif record.status == "current" then add("spouse")
290 elseif record.status == "widowed" then add(record.died == a and "late" or "survivor")
291 else add("ex") end
292 local pa, pb = parents_of(a), parents_of(b)
293 local partners_a, partners_b = partners_of(a), partners_of(b)
294 if not has(pb, a) then
295 for _, p in ipairs(pb) do
296 if partners_a[p] then add("stepparent"); break end
297 end
298 end
299 if not has(pa, b) then
300 for _, p in ipairs(pa) do
301 if partners_b[p] then add("stepchild"); break end
302 end
303 end
304 if common_parents(a, b) == 0 and step_related(pa, pb) then add("stepsibling") end
305 if not is_ancestor(a, b) then
306 for _, p in ipairs(grandparents_of(b)) do
307 if partners_a[p] then add("stepgrandparent"); break end
308 end
309 end
310 if not is_ancestor(b, a) then
311 for _, p in ipairs(grandparents_of(a)) do
312 if partners_b[p] then add("stepgrandchild"); break end
313 end
314 end
315 for p in pairs(partners_b) do
316 if has(parents_of(p), a) then add("parent-in-law"); break end
317 end
318 for p in pairs(partners_a) do
319 if has(parents_of(p), b) then add("child-in-law"); break end
320 end
321 local in_law = false
322 for p in pairs(partners_b) do
323 if are_siblings(a, p) then in_law = true; break end
324 end
325 if not in_law then
326 for p in pairs(partners_a) do
327 if are_siblings(p, b) then in_law = true; break end
328 end
329 end
330 if in_law then add("sibling-in-law") end
331 return result
332end

339local coefficient_scale = 1 << 37
340local phi
341local function inbreeding(x)
342 local parents = parents_of(x)
343 if #parents < 2 then return 0 end
344 return phi(parents[1], parents[2])
345end
346
347phi = function(a, b)
348 local key = a < b and a .. ":" .. b or b .. ":" .. a
349 if coefficient_cache[key] ~= nil then return coefficient_cache[key] end
350 local value
351 if a == b then value = (coefficient_scale + inbreeding(a)) // 2
352 else
353 local x, y = a, b
354 if persons[x].birth < persons[y].birth then x, y = y, x end
355 value = 0
356 for _, parent in ipairs(parents_of(x)) do value = value + phi(parent, y) end
357 value = value // 2
358 end
359 coefficient_cache[key] = value
360 return value
361end
362
363local function fraction_text(value)
364 if value == 0 then return "0/1" end
365 local denominator = coefficient_scale
366 while denominator > 1 and value % 2 == 0 do
367 value, denominator = value // 2, denominator // 2
368 end
369 return value .. "/" .. denominator
370end

372local function lineage_length(x)
373 if height_cache[x] then return height_cache[x] end
374 local height = 0
375 for _, parent in ipairs(parents_of(x)) do height = math.max(height, lineage_length(parent)) end
376 height_cache[x] = height + 1
377 return height + 1
378end

380local function upward_path(x, top)
381 local result = {x}
382 while x ~= top do
383 local required = distances(x, true)[top] - 1
384 local best
385 for _, parent in ipairs(parents_of(x)) do
386 if distances(parent, true)[top] == required and (not best or parent < best) then best = parent end
387 end
388 x = best
389 result[#result + 1] = x
390 end
391 return result
392end

427function handlers.PERSON(words)
428 local x, sex, birth, name = words[2], words[3], tonumber(words[4]), words[5]
429 if persons[x] then fail("DUPLICATE PERSON", x) end

434function handlers.DEATH(words)
435 local x, year = words[2], tonumber(words[3])
436 local person = persons[x]
437 if person.death then fail("ALREADY DECEASED", x) end
438 if year < person.birth then fail("BAD YEAR", x) end
439 for _, record in ipairs(records_of(x)) do
440 if year < record.start or (record.finish and year < record.finish) then fail("BAD YEAR", x) end
441 end
442 local children = copy_array(children_of(x))
443 table.sort(children)
444 for _, child in ipairs(children) do
445 if persons[child].birth > year + 1 then fail("AFTER DEATH", child, x) end
446 end
447 person.death = year
448 local record = current_record(x)
449 if not record then return one_line("DIED", x, year) end
450 record.finish, record.status, record.died = year, "widowed", x
451 return one_line("DIED", x, year, "spouse=" .. other_person(record, x))
452end

471function handlers.PARENT(words)
472 local child, parent = words[2], words[3]
473 if child == parent then fail("SAME PERSON", child) end
474 if has(parents_of(child), parent) then fail("ALREADY PARENT", child, parent) end
475 if #parents_of(child) == 2 then fail("TOO MANY PARENTS", child) end
476 if is_ancestor(child, parent) then fail("CYCLE", child, parent) end
477 if persons[parent].birth + 12 > persons[child].birth then fail("AGE GAP", child, parent) end
478 if persons[parent].death and persons[parent].death + 1 < persons[child].birth then fail("AFTER DEATH", child, parent) end
479 local parents = parents_of(child)
480 parents[#parents + 1] = parent
481 return one_line("LINKED", child, parent, gendered("parent", persons[parent].sex))
482end

484function handlers.UNPARENT(words)
485 local child, parent = words[2], words[3]
486 if not has(parents_of(child), parent) then fail("NOT PARENT", child, parent) end
487 for i, id in ipairs(parents_of(child)) do
488 if id == parent then table.remove(parents_of(child), i); break end
489 end

493function handlers.SPOUSE(words)
494 local a, b, year = words[2], words[3], tonumber(words[4])
495 if a == b then fail("SAME PERSON", a) end
496 for _, x in ipairs({a, b}) do
497 if persons[x].death then fail("DECEASED", x) end
498 end
499 for _, x in ipairs({a, b}) do
500 if spouse_of(x) then fail("ALREADY MARRIED", x) end
501 end
502 if is_ancestor(a, b) or is_ancestor(b, a) or are_siblings(a, b) then fail("CLOSE KIN", a, b) end
503 for _, x in ipairs({a, b}) do
504 if year < persons[x].birth + 16 then fail("TOO YOUNG", x) end
505 end
506 for _, x in ipairs({a, b}) do
507 for _, record in ipairs(records_of(x)) do
508 if record.finish and record.finish >= year then fail("BAD YEAR", x) end
509 end
510 end
511 marriages[#marriages + 1] = {first = a, second = b, start = year, status = "current"}
512 return one_line("MARRIED", a, b, year)
513end

515function handlers.DIVORCE(words)

518 local record = current_record(a)
519 if year <= record.start then fail("BAD YEAR", a) end
520 record.finish, record.status = year, "divorced"
521 return one_line("DIVORCED", a, b, year)

589function handlers.COMMON(words)
590 local a, b = words[2], words[3]
591 local da, db, shared = distances(a, true), distances(b, true), {}
592 for id in pairs(da) do
593 if db[id] then shared[id] = true end
594 end
595 local nearest = {}
596 for id in pairs(shared) do
597 local is_nearest = true
598 for _, child in ipairs(children_of(id)) do
599 if shared[child] then is_nearest = false; break end
600 end
601 if is_nearest then nearest[#nearest + 1] = id end
602 end
603 table.sort(nearest, function(x, y)
604 local sx, sy = da[x] + db[x], da[y] + db[y]
605 return sx < sy or (sx == sy and x < y)
606 end)
607 local lines = {"COMMON " .. a .. " " .. b .. " " .. #nearest}
608 for _, id in ipairs(nearest) do lines[#lines + 1] = " " .. id .. " " .. da[id] .. " " .. db[id] end
609 return lines
610end

612local function lineage(words, upward)
613 local x = words[2]
614 local d, ids = distances(x, upward), {}
615 for id in pairs(d) do
616 if id ~= x then ids[#ids + 1] = id end
617 end
618 table.sort(ids, function(a, b) return d[a] < d[b] or (d[a] == d[b] and a < b) end)
619 local lines = {words[1] .. " " .. x .. " " .. #ids}
620 for _, id in ipairs(ids) do
621 local term = upward and kinship_term(0, d[id], persons[id].sex) or kinship_term(d[id], 0, persons[id].sex)
622 lines[#lines + 1] = " " .. d[id] .. " " .. id .. " " .. term
623 end
624 return lines
625end

629function handlers.COUSINS(words)
630 local x, degree, matches = words[2], tonumber(words[3]), {}
631 for y in pairs(persons) do
632 if y ~= x then
633 local top = apex(y, x)
634 if top and top.da >= 2 and top.db >= 2 and math.min(top.da, top.db) - 1 == degree then
635 matches[#matches + 1] = {id = y, removal = math.abs(top.da - top.db), term = blood_term(top, persons[y].sex)}
636 end
637 end
638 end
639 table.sort(matches, function(a, b)
640 return a.removal < b.removal or (a.removal == b.removal and a.id < b.id)
641 end)
642 local lines = {"COUSINS " .. x .. " " .. degree .. " " .. #matches}
643 for _, match in ipairs(matches) do lines[#lines + 1] = " " .. match.id .. " " .. match.term end
644 return lines
645end

647function handlers.SIBLINGS(words)
648 local x, lines, earlier = words[2], {}, 0
649 for _, y in ipairs(age_order(keys(persons))) do
650 if y ~= x then
651 local common = common_parents(x, y)
652 local kind
653 if common == 2 then kind = "full"
654 elseif common == 1 then kind = "half"
655 elseif step_related(parents_of(x), parents_of(y)) then kind = "step" end
656 if kind then
657 if kind ~= "step" and age_less(y, x) then earlier = earlier + 1 end
658 lines[#lines + 1] = " " .. y .. " " .. kind .. " " .. persons[y].birth
659 end
660 end
661 end
662 table.insert(lines, 1, "SIBLINGS " .. x .. " " .. #lines .. " order=" .. (earlier + 1))
663 return lines
664end

666function handlers.GENERATIONS(words)
667 local x, groups = words[2], {}
668 for id, generation in pairs(distances(x, false)) do
669 if id ~= x then
670 local group = groups[generation]
671 if not group then group = {}; groups[generation] = group end
672 group[#group + 1] = id
673 end
674 end
675 local generations = keys(groups)
676 table.sort(generations)
677 local lines = {"GENERATIONS " .. x .. " " .. #generations}
678 for _, generation in ipairs(generations) do
679 local members, living, minimum, maximum = groups[generation], 0, math.huge, -math.huge
680 for _, id in ipairs(members) do
681 local person = persons[id]
682 if not person.death then living = living + 1 end
683 minimum, maximum = math.min(minimum, person.birth), math.max(maximum, person.birth)
684 end
685 lines[#lines + 1] = " " .. generation .. " " .. #members .. " " .. living .. " " .. minimum .. "-" .. maximum
686 end
687 return lines
688end

690function handlers.TREE(words)
691 local x, lines, visited = words[2], {}, {}
692 local function visit(id, depth)
693 local indent = string.rep(" ", depth + 1)
694 if visited[id] then lines[#lines + 1] = indent .. id .. " (see above)"; return end
695 visited[id] = true
696 local person = persons[id]
697 local line = indent .. id .. " " .. person.name .. " " .. life_years(person)
698 local spouse = spouse_of(id)
699 if spouse then line = line .. " = " .. spouse end
700 lines[#lines + 1] = line
701 for _, child in ipairs(age_order(children_of(id))) do visit(child, depth + 1) end
702 end
703 visit(x, 0)
704 table.insert(lines, 1, "TREE " .. x .. " " .. (#keys(distances(x, false)) - 1))
705 return lines

708function handlers.PEDIGREE(words)
709 local x, depth, lines, entered = words[2], tonumber(words[3]), {}, {}
710 local function enter(id, number, generation)
711 local person = persons[id]
712 lines[#lines + 1] = string.rep(" ", generation + 1) .. number .. " " .. id .. " " .. person.name .. " " .. life_years(person)
713 entered[id] = true
714 if generation < depth then
715 local parents = copy_array(parents_of(id))
716 table.sort(parents, function(a, b)
717 local sa, sb = sex_index[persons[a].sex], sex_index[persons[b].sex]
718 return sa < sb or (sa == sb and a < b)
719 end)
720 for i, parent in ipairs(parents) do enter(parent, 2 * number + i - 1, generation + 1) end
721 end
722 end
723 enter(x, 1, 0)
724 table.insert(lines, 1, "PEDIGREE " .. x .. " " .. depth .. " entries=" .. #lines .. " distinct=" .. #keys(entered))
725 return lines

728function handlers.TIMELINE(words)
729 local x, events = words[2], {}
730 local person = persons[x]
731 local function event(year, rank, counterpart, text)
732 events[#events + 1] = {year = year, rank = rank, counterpart = counterpart, text = text}
733 end
734 event(person.birth, 0, "", "born")
735 for _, record in ipairs(records_of(x)) do
736 local other = other_person(record, x)
737 event(record.start, 1, other, "married " .. other)
738 if record.status == "divorced" then event(record.finish, 3, other, "divorced " .. other)
739 elseif record.status == "widowed" and record.died == other then event(record.finish, 4, other, "widowed " .. other) end
740 end
741 for _, child in ipairs(children_of(x)) do event(persons[child].birth, 2, child, "child " .. child) end
742 for _, group in ipairs({{label = "parent", ids = parents_of(x)}, {label = "child", ids = children_of(x)}}) do
743 for _, other in ipairs(group.ids) do
744 local death = persons[other].death
745 if death and death >= person.birth and (not person.death or death <= person.death) then
746 event(death, 5, other, "lost " .. group.label .. " " .. other)
747 end
748 end
749 end
750 if person.death then event(person.death, 6, "", "died") end
751 table.sort(events, function(a, b)
752 if a.year ~= b.year then return a.year < b.year end
753 if a.rank ~= b.rank then return a.rank < b.rank end
754 return a.counterpart < b.counterpart
755 end)
756 local lines = {"TIMELINE " .. x .. " " .. #events}
757 for _, e in ipairs(events) do lines[#lines + 1] = " " .. e.year .. " " .. e.text end
758 return lines
759end
