7struct Fraction: CustomStringConvertible {
8 let numerator: Int64
9 let denominator: Int64
10
11 init(_ numerator: Int64, _ denominator: Int64 = 1) {
12 var a = numerator
13 var b = denominator
14 while b != 0 { (a, b) = (b, a % b) }
15 self.numerator = numerator / a
16 self.denominator = denominator / a
17 }
18
19 static func + (a: Fraction, b: Fraction) -> Fraction {
20 let denominator = max(a.denominator, b.denominator)
21 return Fraction(a.numerator * (denominator / a.denominator) + b.numerator * (denominator / b.denominator), denominator)
22 }
23
24 var half: Fraction { Fraction(numerator, denominator * 2) }
25 var description: String { "\(numerator)/\(denominator)" }
26}

80func grandPrefix(_ generation: Int) -> String {
81 if generation == 1 { return "" }
82 return String(repeating: "great-", count: generation - 2) + "grand"
83}

85func ordinal(_ number: Int) -> String {
86 let words = ["first", "second", "third", "fourth", "fifth", "sixth", "seventh", "eighth", "ninth", "tenth"]
87 if number <= 10 { return words[number - 1] }
88 let suffix: String
89 if (11...13).contains(number % 100) { suffix = "th" }
90 else { suffix = [1: "st", 2: "nd", 3: "rd"][number % 10] ?? "th" }
91 return "\(number)" + suffix
92}

94func term(_ da: Int, _ db: Int, _ sex: String) -> String {
95 if da == 0 && db == 0 { return "self" }
96 if da == 0 { return grandPrefix(db) + gender(sex, "father", "mother", "parent") }
97 if db == 0 { return grandPrefix(da) + gender(sex, "son", "daughter", "child") }
98 if da == 1 && db == 1 { return gender(sex, "brother", "sister", "sibling") }
99 if da == 1 { return grandPrefix(db - 1) + gender(sex, "uncle", "aunt", "pibling") }
100 if db == 1 { return grandPrefix(da - 1) + gender(sex, "nephew", "niece", "nibling") }
101 let removed = abs(da - db)
102 let ending: String
103 switch removed {
104 case 0: ending = ""
105 case 1: ending = " once removed"
106 case 2: ending = " twice removed"
107 default: ending = " \(removed) times removed"
108 }
109 return ordinal(min(da, db) - 1) + " cousin" + ending
110}

112final class Family {
113 var state = FamilyState()
114 var history: [UndoRecord] = []
115 var ancestorCache: [String: [String: Int]] = [:]
116 var kinshipCache: [PersonPair: Fraction] = [:]
117 var lastMarriages: [PersonPair: Int] = [:]
118 var spouses: [String: String] = [:]
119 var partnerCache: [String: Set<String>] = [:]

121 func prepareQueries() {
122 ancestorCache.removeAll(keepingCapacity: true)
123 kinshipCache.removeAll(keepingCapacity: true)
124 partnerCache.removeAll(keepingCapacity: true)
125 lastMarriages.removeAll(keepingCapacity: true)
126 spouses.removeAll(keepingCapacity: true)
127 for (i, marriage) in state.marriages.enumerated() {
128 lastMarriages[PersonPair(marriage.first, marriage.second)] = i
129 if marriage.status == "current" {
130 spouses[marriage.first] = marriage.second
131 spouses[marriage.second] = marriage.first
132 }
133 }
134 }

137 for id in ids where state.people[id] == nil { throw CommandError(reason: "UNKNOWN PERSON " + id) }
138 }

142 func ageBefore(_ a: String, _ b: String) -> Bool {
143 let x = state.people[a]!.birth

153 func children(_ id: String) -> [String] {
154 state.people.keys.filter { state.people[$0]!.parents.contains(id) }
155 }

157 func siblings(_ a: String, _ b: String) -> Bool {
158 a != b && !state.people[a]!.parents.isDisjoint(with: state.people[b]!.parents)
159 }

161 func ancestors(_ id: String) -> [String: Int] {
162 if let cached = ancestorCache[id] { return cached }
163 var distances = [id: 0]
164 var queue = [id]
165 var cursor = 0
166 while cursor < queue.count {
167 let person = queue[cursor]
168 cursor += 1
169 let distance = distances[person]! + 1
170 for parent in state.people[person]!.parents where distances[parent] == nil {
171 distances[parent] = distance
172 queue.append(parent)
173 }
174 }
175 ancestorCache[id] = distances
176 return distances
177 }
178
179 func descendants(_ id: String) -> [String: Int] {
180 var result: [String: Int] = [:]
181 for person in state.people.keys where person != id {
182 if let distance = ancestors(person)[id] { result[person] = distance }
183 }
184 return result
185 }
186
187 func apex(_ a: String, _ b: String) -> Apex? {
188 let aa = ancestors(a)
189 let ab = ancestors(b)
190 if let db = ab[a] { return Apex(id: a, da: 0, db: db, shares: 0) }
191 if let da = aa[b] { return Apex(id: b, da: da, db: 0, shares: 0) }
192 let common = aa.keys.filter { ab[$0] != nil }
193 guard let best = common.min(by: {
194 let leftSum = aa[$0]! + ab[$0]!
195 let rightSum = aa[$1]! + ab[$1]!
196 if leftSum != rightSum { return leftSum < rightSum }
197 if aa[$0]! != aa[$1]! { return aa[$0]! < aa[$1]! }
198 return $0 < $1
199 }) else { return nil }
200 let da = aa[best]!
201 let db = ab[best]!
202 let shares = common.filter { aa[$0] == da && ab[$0] == db }.count
203 return Apex(id: best, da: da, db: db, shares: shares)
204 }

206 func bloodTerm(_ a: String, _ b: String, sex: String? = nil) -> String? {
207 guard a != b, let apex = apex(a, b) else { return nil }
208 let prefix = apex.shares == 1 ? "half-" : apex.shares >= 4 ? "double " : ""
209 return prefix + term(apex.da, apex.db, sex ?? state.people[a]!.sex)
210 }

212 func partners(_ id: String) -> Set<String> {
213 if let cached = partnerCache[id] { return cached }
214 var result: Set<String> = []
215 for (pair, i) in lastMarriages where pair.first == id || pair.second == id {
216 let marriage = state.marriages[i]
217 if marriage.status != "divorced" { result.insert(marriage.other(id)) }
218 }
219 partnerCache[id] = result
220 return result
221 }

223 func relationship(_ a: String, _ b: String, sex: String? = nil) -> [String] {
224 if a == b { return ["self"] }
225 let s = sex ?? state.people[a]!.sex
226 let parentsA = state.people[a]!.parents
227 let parentsB = state.people[b]!.parents
228 let partnersA = partners(a)
229 let partnersB = partners(b)
230 let aa = ancestors(a)
231 let ab = ancestors(b)
232 var terms: [String] = []
233 if let blood = bloodTerm(a, b, sex: s) { terms.append(blood) }
234 if let i = lastMarriages[PersonPair(a, b)] {
235 let marriage = state.marriages[i]
236 switch marriage.status {
237 case "current": terms.append(gender(s, "husband", "wife", "spouse"))
238 case "widowed":
239 terms.append(marriage.died == a ? gender(s, "late husband", "late wife", "late spouse") : gender(s, "widower", "widow", "surviving spouse"))
240 default: terms.append(gender(s, "ex-husband", "ex-wife", "ex-spouse"))
241 }
242 } else if children(a).contains(where: { state.people[$0]!.parents.contains(b) }) {
243 terms.append("co-parent")
244 }
245 if !parentsB.contains(a) && parentsB.contains(where: { partners($0).contains(a) }) {
246 terms.append(gender(s, "stepfather", "stepmother", "stepparent"))
247 }
248 if !parentsA.contains(b) && parentsA.contains(where: { partners($0).contains(b) }) {
249 terms.append(gender(s, "stepson", "stepdaughter", "stepchild"))
250 }
251 if parentsA.isDisjoint(with: parentsB) && parentsA.contains(where: { !partners($0).isDisjoint(with: parentsB) }) {
252 terms.append(gender(s, "stepbrother", "stepsister", "stepsibling"))
253 }
254 if ab[a] == nil && ab.contains(where: { $0.value == 2 && partners($0.key).contains(a) }) {
255 terms.append(gender(s, "stepgrandfather", "stepgrandmother", "stepgrandparent"))
256 }
257 if aa[b] == nil && aa.contains(where: { $0.value == 2 && partners($0.key).contains(b) }) {
258 terms.append(gender(s, "stepgrandson", "stepgranddaughter", "stepgrandchild"))
259 }
260 if partnersB.contains(where: { state.people[$0]!.parents.contains(a) }) {
261 terms.append(gender(s, "father-in-law", "mother-in-law", "parent-in-law"))
262 }
263 if partnersA.contains(where: { state.people[$0]!.parents.contains(b) }) {
264 terms.append(gender(s, "son-in-law", "daughter-in-law", "child-in-law"))
265 }
266 if partnersB.contains(where: { siblings(a, $0) }) || partnersA.contains(where: { siblings($0, b) }) {
267 terms.append(gender(s, "brother-in-law", "sister-in-law", "sibling-in-law"))
268 }
269 return terms
270 }

272 func relationshipString(_ a: String, _ b: String) -> String {
273 let terms = relationship(a, b)
274 return terms.isEmpty ? "no relation" : terms.joined(separator: ", ")
275 }

277 func phi(_ a: String, _ b: String) -> Fraction {
278 let pair = PersonPair(a, b)
279 if let value = kinshipCache[pair] { return value }
280 let value: Fraction
281 if a == b {
282 let parents = Array(state.people[a]!.parents)
283 value = parents.count == 2 ? (Fraction(1) + phi(parents[0], parents[1])).half : Fraction(1, 2)
284 } else {
285 let x = state.people[a]!.birth >= state.people[b]!.birth ? a : b
286 let y = x == a ? b : a
287 var sum = Fraction(0)
288 for parent in state.people[x]!.parents { sum = sum + phi(parent, y) }
289 value = sum.half
290 }
291 kinshipCache[pair] = value
292 return value
293 }

295 func inbreeding(_ id: String) -> Fraction {
296 let parents = Array(state.people[id]!.parents)
297 return parents.count == 2 ? phi(parents[0], parents[1]) : Fraction(0)
298 }

300 func upwardPath(_ from: String, to apex: String) -> [String] {
301 var result = [from]
302 var current = from
303 while current != apex {
304 let targetDistance = ancestors(current)[apex]! - 1
305 current = state.people[current]!.parents.sorted().first { ancestors($0)[apex] == targetDistance }!
306 result.append(current)
307 }
308 return result
309 }

311 func command(_ line: String) throws -> [String] {
312 prepareQueries()
313 let words = line.components(separatedBy: " ")
314 let command = words[0]
315 switch command {
316 case "PERSON":
317 let id = words[1]
318 guard state.people[id] == nil else { throw CommandError(reason: "DUPLICATE PERSON " + id) }
319 remember(line)
320 state.people[id] = Person(sex: words[2], birth: Int(words[3])!, name: words[4])
321 return ["ADDED \(id) \(words[4]) \(words[2]) \(words[3])"]
322 case "DEATH":
323 let id = words[1]
324 let year = Int(words[2])!
325 try check(id)
326 let person = state.people[id]!
327 guard person.death == nil else { throw CommandError(reason: "ALREADY DECEASED " + id) }
328 if year < person.birth || state.marriages.contains(where: { $0.contains(id) && ($0.start > year || ($0.end != nil && $0.end! > year)) }) {
329 throw CommandError(reason: "BAD YEAR " + id)
330 }
331 if let child = children(id).sorted().first(where: { state.people[$0]!.birth > year + 1 }) {
332 throw CommandError(reason: "AFTER DEATH \(child) \(id)")
333 }
334 remember(line)
335 state.people[id]!.death = year
336 if let spouse = spouses[id] {
337 let i = lastMarriages[PersonPair(id, spouse)]!
338 state.marriages[i].end = year
339 state.marriages[i].status = "widowed"
340 state.marriages[i].died = id
341 return ["DIED \(id) \(year) spouse=\(spouse)"]
342 }
343 return ["DIED \(id) \(year)"]
344 case "REMOVE":
345 let id = words[1]
346 try check(id)
347 remember(line)
348 let ids = Array(state.people.keys)
349 for person in ids { state.people[person]!.parents.remove(id) }
350 state.people.removeValue(forKey: id)
351 state.marriages.removeAll { $0.contains(id) }
352 return ["REMOVED " + id]
353 case "PARENT":
354 let child = words[1]
355 let parent = words[2]
356 try check(child, parent)
357 guard child != parent else { throw CommandError(reason: "SAME PERSON " + child) }
358 guard !state.people[child]!.parents.contains(parent) else { throw CommandError(reason: "ALREADY PARENT \(child) \(parent)") }
359 guard state.people[child]!.parents.count < 2 else { throw CommandError(reason: "TOO MANY PARENTS " + child) }
360 guard ancestors(parent)[child] == nil else { throw CommandError(reason: "CYCLE \(child) \(parent)") }
361 guard state.people[parent]!.birth + 12 <= state.people[child]!.birth else { throw CommandError(reason: "AGE GAP \(child) \(parent)") }
362 if let death = state.people[parent]!.death, death + 1 < state.people[child]!.birth {
363 throw CommandError(reason: "AFTER DEATH \(child) \(parent)")
364 }
365 remember(line)
366 state.people[child]!.parents.insert(parent)
367 return ["LINKED \(child) \(parent) " + gender(state.people[parent]!.sex, "father", "mother", "parent")]
368 case "UNPARENT":
369 let child = words[1]
370 let parent = words[2]
371 try check(child, parent)
372 guard state.people[child]!.parents.contains(parent) else { throw CommandError(reason: "NOT PARENT \(child) \(parent)") }
373 remember(line)
374 state.people[child]!.parents.remove(parent)
375 return ["UNLINKED \(child) \(parent)"]
376 case "SPOUSE":
377 let a = words[1]
378 let b = words[2]
379 let year = Int(words[3])!
380 try check(a, b)
381 guard a != b else { throw CommandError(reason: "SAME PERSON " + a) }
382 for id in [a, b] where state.people[id]!.death != nil { throw CommandError(reason: "DECEASED " + id) }
383 for id in [a, b] where spouses[id] != nil { throw CommandError(reason: "ALREADY MARRIED " + id) }
384 if ancestors(a)[b] != nil || ancestors(b)[a] != nil || siblings(a, b) { throw CommandError(reason: "CLOSE KIN \(a) \(b)") }
385 for id in [a, b] where year < state.people[id]!.birth + 16 { throw CommandError(reason: "TOO YOUNG " + id) }
386 for id in [a, b] {
387 if state.marriages.contains(where: { $0.contains(id) && $0.end != nil && $0.end! >= year }) {
388 throw CommandError(reason: "BAD YEAR " + id)
389 }
390 }
391 remember(line)
392 state.marriages.append(Marriage(first: a, second: b, start: year))
393 return ["MARRIED \(a) \(b) \(year)"]
394 case "DIVORCE":
395 let a = words[1]
396 let b = words[2]
397 let year = Int(words[3])!
398 try check(a, b)
399 guard spouses[a] == b else { throw CommandError(reason: "NOT MARRIED \(a) \(b)") }
400 let i = lastMarriages[PersonPair(a, b)]!
401 guard year > state.marriages[i].start else { throw CommandError(reason: "BAD YEAR " + a) }
402 remember(line)
403 state.marriages[i].end = year
404 state.marriages[i].status = "divorced"
405 return ["DIVORCED \(a) \(b) \(year)"]
406 case "UNDO":
407 guard let previous = history.popLast() else { throw CommandError(reason: "NOTHING TO UNDO") }
408 state = previous.state
409 return ["UNDONE " + previous.command]
410 case "INFO":
411 let id = words[1]
412 try check(id)
413 let person = state.people[id]!
414 var lines = ["INFO \(id) \(person.name) \(person.sex) \(person.life)", " parents: " + personList(person.parents)]
415 for marriage in state.marriages where marriage.contains(id) {
416 lines.append(" marriage: \(marriage.other(id)) \(marriage.start)-\(marriage.end.map(String.init) ?? "") \(marriage.status)")
417 }
418 lines.append(" children: " + personList(children(id)))
419 lines.append(" siblings: " + personList(state.people.keys.filter { siblings($0, id) }))
420 lines.append(" inbreeding: \(inbreeding(id))")
421 return lines
422 case "RELATION":
423 let a = words[1]
424 let b = words[2]
425 try check(a, b)
426 return ["RELATION \(a) \(b): " + relationshipString(a, b)]
427 case "RELATIVES", "WHO":
428 let id = words[1]
429 try check(id)
430 let query = words.dropFirst(2).joined(separator: " ")
431 var lines: [String] = []
432 for person in state.people.keys.sorted() where person != id {
433 let terms = relationship(person, id)
434 if command == "RELATIVES" {
435 if !terms.isEmpty { lines.append(" \(person): " + terms.joined(separator: ", ")) }
436 } else {
437 let neutral = relationship(person, id, sex: "X")
438 if let index = terms.indices.first(where: { terms[$0] == query || neutral[$0] == query }) {
439 lines.append(" \(person): " + terms[index])
440 }
441 }
442 }
443 let title = command == "WHO" ? "WHO \(id) \(query)" : "RELATIVES \(id)"
444 return [title + " \(lines.count)"] + lines
445 case "PATH":
446 let a = words[1]
447 let b = words[2]
448 try check(a, b)
449 guard let apex = apex(a, b) else { return ["PATH \(a) \(b) none"] }
450 let path = upwardPath(a, to: apex.id) + Array(upwardPath(b, to: apex.id).dropLast().reversed())
451 return ["PATH \(a) \(b) \(apex.da + apex.db): " + path.joined(separator: " ")]
452 case "COMMON":
453 let a = words[1]
454 let b = words[2]
455 try check(a, b)
456 let aa = ancestors(a)
457 let ab = ancestors(b)
458 let common = Set(aa.keys.filter { ab[$0] != nil })
459 let nearest = common.filter { Set(children($0)).isDisjoint(with: common) }.sorted {
460 let left = aa[$0]! + ab[$0]!
461 let right = aa[$1]! + ab[$1]!
462 return left != right ? left < right : $0 < $1
463 }
464 return ["COMMON \(a) \(b) \(nearest.count)"] + nearest.map { " \($0) \(aa[$0]!) \(ab[$0]!)" }
465 case "ANCESTORS", "DESCENDANTS":
466 let id = words[1]
467 try check(id)
468 var distances = command == "ANCESTORS" ? ancestors(id) : descendants(id)
469 distances.removeValue(forKey: id)
470 let sorted = distances.keys.sorted {
471 distances[$0]! != distances[$1]! ? distances[$0]! < distances[$1]! : $0 < $1
472 }
473 return ["\(command) \(id) \(sorted.count)"] + sorted.map { person in
474 let generation = distances[person]!
475 let title = command == "ANCESTORS" ? term(0, generation, state.people[person]!.sex) : term(generation, 0, state.people[person]!.sex)
476 return " \(generation) \(person) \(title)"
477 }
478 case "COUSINS":
479 let id = words[1]
480 let degree = Int(words[2])!
481 try check(id)
482 var matches: [(String, Apex)] = []
483 for person in state.people.keys where person != id {
484 if let apex = apex(person, id), apex.da >= 2, apex.db >= 2, min(apex.da, apex.db) - 1 == degree {
485 matches.append((person, apex))
486 }
487 }
488 matches.sort {
489 let left = abs($0.1.da - $0.1.db)
490 let right = abs($1.1.da - $1.1.db)
491 return left != right ? left < right : $0.0 < $1.0
492 }
493 return ["COUSINS \(id) \(degree) \(matches.count)"] + matches.map { " \($0.0) \(bloodTerm($0.0, id)!)" }
494 case "SIBLINGS":
495 let id = words[1]
496 try check(id)
497 let parents = state.people[id]!.parents
498 var matches: [(String, String)] = []
499 var order = 1
500 for person in state.people.keys where person != id {
501 let otherParents = state.people[person]!.parents
502 let count = parents.intersection(otherParents).count
503 let kind: String
504 if count == 2 { kind = "full" }
505 else if count == 1 { kind = "half" }
506 else if parents.contains(where: { !partners($0).isDisjoint(with: otherParents) }) { kind = "step" }
507 else { continue }
508 matches.append((person, kind))
509 if count > 0 && ageBefore(person, id) { order += 1 }
510 }
511 matches.sort { ageBefore($0.0, $1.0) }
512 return ["SIBLINGS \(id) \(matches.count) order=\(order)"] + matches.map { " \($0.0) \($0.1) \(state.people[$0.0]!.birth)" }
513 case "GENERATIONS":
514 let id = words[1]
515 try check(id)
516 var groups: [Int: [String]] = [:]
517 for (person, generation) in descendants(id) { groups[generation, default: []].append(person) }
518 return ["GENERATIONS \(id) \(groups.count)"] + groups.keys.sorted().map { generation in
519 let people = groups[generation]!.map { state.people[$0]! }
520 let years = people.map(\.birth)
521 return " \(generation) \(people.count) \(people.filter { $0.death == nil }.count) \(years.min()!)-\(years.max()!)"
522 }
523 case "TREE":
524 let id = words[1]
525 try check(id)
526 var lines = ["TREE \(id) \(descendants(id).count)"]
527 var visited: Set<String> = []
528 func visit(_ person: String, depth: Int) {
529 let indent = String(repeating: " ", count: 2 * (depth + 1))
530 if !visited.insert(person).inserted { lines.append(indent + person + " (see above)"); return }
531 let value = state.people[person]!
532 let spouse = spouses[person].map { " = " + $0 } ?? ""
533 lines.append(indent + "\(person) \(value.name) \(value.life)" + spouse)
534 for child in children(person).sorted(by: ageBefore) { visit(child, depth: depth + 1) }
535 }
536 visit(id, depth: 0)
537 return lines
538 case "PEDIGREE":
539 let id = words[1]
540 let depth = Int(words[2])!
541 try check(id)
542 var lines: [String] = []
543 var entered: Set<String> = []
544 let sexOrder = ["M": 0, "F": 1, "X": 2]
545 func enter(_ person: String, number: Int, generation: Int) {
546 entered.insert(person)
547 let value = state.people[person]!
548 lines.append(String(repeating: " ", count: 2 * (generation + 1)) + "\(number) \(person) \(value.name) \(value.life)")
549 if generation < depth {
550 let parents = value.parents.sorted {
551 let left = sexOrder[state.people[$0]!.sex]!
552 let right = sexOrder[state.people[$1]!.sex]!
553 return left != right ? left < right : $0 < $1
554 }
555 for (i, parent) in parents.enumerated() { enter(parent, number: 2 * number + i, generation: generation + 1) }
556 }
557 }
558 enter(id, number: 1, generation: 0)
559 return ["PEDIGREE \(id) \(depth) entries=\(lines.count) distinct=\(entered.count)"] + lines
560 case "TIMELINE":
561 let id = words[1]
562 try check(id)
563 let person = state.people[id]!
