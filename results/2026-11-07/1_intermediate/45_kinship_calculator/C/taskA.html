63static int compare_people(const void *left, const void *right) {
64 int a = *(const int *)left, b = *(const int *)right;
65 Person *pa = &sort_family->persons[a], *pb = &sort_family->persons[b];
66 if (sort_mode == 1 && pa->birth != pb->birth)
67 return pa->birth < pb->birth ? -1 : 1;
68 if (sort_mode == 2 && sort_scores[a] != sort_scores[b])
69 return sort_scores[a] < sort_scores[b] ? -1 : 1;
70 if (sort_mode == 3 && sex_rank(pa->sex) != sex_rank(pb->sex))
71 return sex_rank(pa->sex) - sex_rank(pb->sex);
72 return strcmp(pa->id, pb->id);
73}

105static int has_parent(Family *family, int child, int parent) {
106 Person *person = &family->persons[child];
107 for (int i = 0; i < person->parent_count; ++i)
108 if (person->parents[i] == parent)

122 int count = 0;

165static void distances(Family *family, int start, int upward, int result[]) {
166 int queue[MAX_PERSONS], head = 0, tail = 0;
167 for (int i = 0; i < MAX_PERSONS; ++i)
168 result[i] = -1;
169 result[start] = 0;
170 queue[tail++] = start;
171 while (head < tail) {
172 int current = queue[head++], next[MAX_PERSONS], count;
173 if (upward) {
174 count = family->persons[current].parent_count;
175 for (int i = 0; i < count; ++i)
176 next[i] = family->persons[current].parents[i];
177 } else {
178 count = children(family, current, next);
179 }
180 for (int i = 0; i < count; ++i) {
181 int target = next[i];
182 if (result[target] < 0) {
183 result[target] = result[current] + 1;
184 queue[tail++] = target;
185 }
186 }
187 }
188}

190static int ancestor(Family *family, int a, int b) {
191 int depths[MAX_PERSONS];
192 distances(family, b, 1, depths);
193 return a != b && depths[a] >= 0;
194}

196static Apex apex(Family *family, int a, int b) {
197 int da[MAX_PERSONS], db[MAX_PERSONS];
198 distances(family, a, 1, da);
199 distances(family, b, 1, db);
200 if (db[a] >= 0)
201 return (Apex){a, 0, db[a], 0};
202 if (da[b] >= 0)
203 return (Apex){b, da[b], 0, 0};
204 Apex result = {-1, 0, 0, 0};
205 for (int i = 0; i < MAX_PERSONS; ++i) {
206 if (da[i] < 0 || db[i] < 0)
207 continue;
208 if (result.person < 0 || da[i] + db[i] < result.da + result.db ||
209 (da[i] + db[i] == result.da + result.db &&
210 (da[i] < result.da ||
211 (da[i] == result.da &&
212 strcmp(family->persons[i].id, family->persons[result.person].id) < 0)))) {
213 result.person = i;
214 result.da = da[i];
215 result.db = db[i];
216 }
217 }
218 if (result.person >= 0)
219 for (int i = 0; i < MAX_PERSONS; ++i)
220 if (da[i] == result.da && db[i] == result.db)
221 ++result.shares;
222 return result;
223}

234static void kinship_term(int da, int db, char sex, char output[]) {
235 char prefix[TERM_SIZE];
236 if (da == 0 && db == 0) {
237 strcpy(output, "self");
238 } else if (da == 0) {
239 generation_prefix(db, prefix);
240 snprintf(output, TERM_SIZE, "%s%s", prefix, gendered(sex, "father", "mother", "parent"));
241 } else if (db == 0) {
242 generation_prefix(da, prefix);
243 snprintf(output, TERM_SIZE, "%s%s", prefix, gendered(sex, "son", "daughter", "child"));
244 } else if (da == 1 && db == 1) {
245 strcpy(output, gendered(sex, "brother", "sister", "sibling"));
246 } else if (da == 1) {
247 generation_prefix(db - 1, prefix);
248 snprintf(output, TERM_SIZE, "%s%s", prefix, gendered(sex, "uncle", "aunt", "pibling"));
249 } else if (db == 1) {
250 generation_prefix(da - 1, prefix);
251 snprintf(output, TERM_SIZE, "%s%s", prefix, gendered(sex, "nephew", "niece", "nibling"));
252 } else {
253 char ordinal[32], removal[48] = "";
254 char *ordinals[] = {"first", "second", "third", "fourth", "fifth",
255 "sixth", "seventh", "eighth", "ninth", "tenth"};
256 int degree = (da < db ? da : db) - 1, difference = abs(da - db);
257 if (degree <= 10) {
258 strcpy(ordinal, ordinals[degree - 1]);
259 } else {
260 char *suffix = "th";
261 if (degree % 100 < 11 || degree % 100 > 13) {
262 if (degree % 10 == 1)
263 suffix = "st";
264 else if (degree % 10 == 2)
265 suffix = "nd";
266 else if (degree % 10 == 3)
267 suffix = "rd";
268 }
269 snprintf(ordinal, sizeof(ordinal), "%d%s", degree, suffix);
270 }
271 if (difference == 1)
272 strcpy(removal, " once removed");
273 else if (difference == 2)
274 strcpy(removal, " twice removed");
275 else if (difference >= 3)
276 snprintf(removal, sizeof(removal), " %d times removed", difference);
277 snprintf(output, TERM_SIZE, "%s cousin%s", ordinal, removal);
278 }
279}

281static void blood_term(Apex common, char sex, char output[]) {

284 snprintf(output, TERM_SIZE, "%s%s",
285 common.shares == 1 ? "half-"
286 : common.shares >= 4 ? "double "
287 : "",
288 term);
289}

295static int step_siblings(Family *family, int a, int b) {
296 for (int i = 0; i < family->persons[a].parent_count; ++i)
297 for (int j = 0; j < family->persons[b].parent_count; ++j)
298 if (partners(family, family->persons[a].parents[i], family->persons[b].parents[j]))
299 return 1;
300 return 0;
301}

303static void relationship(Family *family, int a, int b, char sex, Relationships *result) {
304 result->count = 0;
305 if (a == b) {
306 add_relationship(result, "self");
307 return;
308 }
309 Apex common = apex(family, a, b);
310 char term[TERM_SIZE];
311 if (common.person >= 0) {
312 blood_term(common, sex, term);
313 add_relationship(result, term);
314 }
315 int last = last_record(family, a, b);
316 if (last >= 0) {
317 Marriage *record = &family->marriages[last];
318 if (record->status == CURRENT)
319 add_relationship(result, gendered(sex, "husband", "wife", "spouse"));
320 else if (record->status == DIVORCED)
321 add_relationship(result, gendered(sex, "ex-husband", "ex-wife", "ex-spouse"));
322 else if (record->died == a)
323 add_relationship(result, gendered(sex, "late husband", "late wife", "late spouse"));
324 else
325 add_relationship(result, gendered(sex, "widower", "widow", "surviving spouse"));
326 } else {
327 for (int i = 0; i < MAX_PERSONS; ++i)
328 if (family->persons[i].active && has_parent(family, i, a) && has_parent(family, i, b)) {
329 add_relationship(result, "co-parent");
330 break;
331 }
332 }
333 Person *pa = &family->persons[a], *pb = &family->persons[b];
334 int step_parent = 0, step_child = 0, grand_parent = 0, grand_child = 0;
335 for (int i = 0; i < pb->parent_count; ++i)
336 step_parent |= partners(family, a, pb->parents[i]);
337 for (int i = 0; i < pa->parent_count; ++i)
338 step_child |= partners(family, b, pa->parents[i]);
339 if (!has_parent(family, b, a) && step_parent)
340 add_relationship(result, gendered(sex, "stepfather", "stepmother", "stepparent"));
341 if (!has_parent(family, a, b) && step_child)
342 add_relationship(result, gendered(sex, "stepson", "stepdaughter", "stepchild"));
343 if (common_parents(family, a, b) == 0 && step_siblings(family, a, b))
344 add_relationship(result, gendered(sex, "stepbrother", "stepsister", "stepsibling"));
345 int up_a[MAX_PERSONS], up_b[MAX_PERSONS];
346 distances(family, a, 1, up_a);
347 distances(family, b, 1, up_b);
348 for (int i = 0; i < MAX_PERSONS; ++i) {
349 if (up_b[i] == 2 && partners(family, a, i))
350 grand_parent = 1;
351 if (up_a[i] == 2 && partners(family, b, i))
352 grand_child = 1;
353 }
354 if (!ancestor(family, a, b) && grand_parent)
355 add_relationship(result,
356 gendered(sex, "stepgrandfather", "stepgrandmother", "stepgrandparent"));
357 if (!ancestor(family, b, a) && grand_child)
358 add_relationship(result,
359 gendered(sex, "stepgrandson", "stepgranddaughter", "stepgrandchild"));
360 int parent_law = 0, child_law = 0, sibling_law = 0;
361 for (int i = 0; i < MAX_PERSONS; ++i) {
362 if (!family->persons[i].active)
363 continue;
364 if (partners(family, b, i)) {
365 parent_law |= has_parent(family, i, a);
366 sibling_law |= siblings(family, a, i);
367 }
368 if (partners(family, a, i)) {
369 child_law |= has_parent(family, i, b);
370 sibling_law |= siblings(family, i, b);
371 }
372 }
373 if (parent_law)
374 add_relationship(result, gendered(sex, "father-in-law", "mother-in-law", "parent-in-law"));
375 if (child_law)
376 add_relationship(result, gendered(sex, "son-in-law", "daughter-in-law", "child-in-law"));
377 if (sibling_law)
378 add_relationship(result,
379 gendered(sex, "brother-in-law", "sister-in-law", "sibling-in-law"));
380}

393static int64_t coefficient(Family *family, int a, int b) {
394 if (phi_memo[a][b] >= 0)
395 return phi_memo[a][b];
396 int64_t result = 0;
397 Person *first = &family->persons[a], *second = &family->persons[b];
398 if (a == b) {
399 int64_t inbred = first->parent_count == 2
400 ? coefficient(family, first->parents[0], first->parents[1])
401 : 0;
402 result = (SCALE + inbred) / 2;
403 } else {
404 int x = first->birth >= second->birth ? a : b, y = x == a ? b : a;
405 for (int i = 0; i < family->persons[x].parent_count; ++i)
406 result += coefficient(family, family->persons[x].parents[i], y);
407 result /= 2;
408 }
409 phi_memo[a][b] = phi_memo[b][a] = result;
410 return result;
411}

413static int64_t inbreeding(Family *family, int id) {
414 Person *person = &family->persons[id];
415 return person->parent_count == 2 ? coefficient(family, person->parents[0], person->parents[1])
416 : 0;

443static int modify(Family *family, char *words[]) {
444 char *command = words[0], *aid = words[1];
445 if (strcmp(command, "PERSON") == 0) {
446 if (find_person(family, aid) >= 0) {
447 printf("ERROR DUPLICATE PERSON %s\n", aid);
448 return 0;
449 }
450 int slot = 0;
451 while (family->persons[slot].active)
452 ++slot;
453 Person person = {0};
454 person.active = 1;
455 strcpy(person.id, aid);
456 person.sex = words[2][0];
457 person.birth = atoi(words[3]);
458 strcpy(person.name, words[4]);
459 family->persons[slot] = person;
460 printf("ADDED %s %s %c %d\n", aid, person.name, person.sex, person.birth);
461 return 1;
462 }
463 int a = checked_person(family, aid);
464 if (a < 0)
465 return 0;
466 Person *first = &family->persons[a];
467 if (strcmp(command, "DEATH") == 0) {
468 int year = atoi(words[2]);
469 if (first->death) {
470 printf("ERROR ALREADY DECEASED %s\n", aid);
471 return 0;
472 }
473 int bad_year = year < first->birth;
474 for (int i = 0; i < family->record_count; ++i) {
475 Marriage *record = &family->marriages[i];
476 if (contains(record, a) && (year < record->start || year < record->end))
477 bad_year = 1;
478 }
479 if (bad_year) {
480 printf("ERROR BAD YEAR %s\n", aid);
481 return 0;
482 }
483 int offspring[MAX_PERSONS], count = children(family, a, offspring);
484 order_people(family, offspring, count, 0, NULL);
485 for (int i = 0; i < count; ++i)
486 if (family->persons[offspring[i]].birth > year + 1) {
487 printf("ERROR AFTER DEATH %s %s\n", family->persons[offspring[i]].id, aid);
488 return 0;
489 }
490 first->death = year;
491 int index = current_record(family, a);
492 printf("DIED %s %d", aid, year);
493 if (index >= 0) {
494 Marriage *record = &family->marriages[index];
495 record->end = year;
496 record->status = WIDOWED;
497 record->died = a;
498 printf(" spouse=%s", family->persons[other(record, a)].id);
499 }
500 putchar('\n');
501 return 1;
502 }
503 if (strcmp(command, "REMOVE") == 0) {
504 for (int i = 0; i < MAX_PERSONS; ++i) {
505 Person *person = &family->persons[i];
506 int kept = 0;
507 for (int j = 0; j < person->parent_count; ++j)
508 if (person->parents[j] != a)
509 person->parents[kept++] = person->parents[j];
510 person->parent_count = kept;
511 }
512 int kept = 0;
513 for (int i = 0; i < family->record_count; ++i)
514 if (!contains(&family->marriages[i], a))
515 family->marriages[kept++] = family->marriages[i];
516 family->record_count = kept;
517 first->active = 0;
518 printf("REMOVED %s\n", aid);
519 return 1;
520 }
521 char *bid = words[2];
522 int b = checked_person(family, bid);
523 if (b < 0)
524 return 0;
525 Person *second = &family->persons[b];
526 if (strcmp(command, "PARENT") == 0) {
527 char *reason = NULL;
528 if (a == b)
529 reason = "SAME PERSON";
530 else if (has_parent(family, a, b))
531 reason = "ALREADY PARENT";
532 else if (first->parent_count == 2)
533 reason = "TOO MANY PARENTS";
534 else if (ancestor(family, a, b))
535 reason = "CYCLE";
536 else if (second->birth + 12 > first->birth)
537 reason = "AGE GAP";
538 else if (second->death && second->death + 1 < first->birth)
539 reason = "AFTER DEATH";
540 if (reason) {
541 printf("ERROR %s %s", reason, aid);
542 if (strcmp(reason, "SAME PERSON") != 0 && strcmp(reason, "TOO MANY PARENTS") != 0)
543 printf(" %s", bid);
544 putchar('\n');
545 return 0;
546 }
547 first->parents[first->parent_count++] = b;
548 printf("LINKED %s %s %s\n", aid, bid, gendered(second->sex, "father", "mother", "parent"));
549 return 1;
550 }
551 if (strcmp(command, "UNPARENT") == 0) {
552 if (!has_parent(family, a, b)) {
553 printf("ERROR NOT PARENT %s %s\n", aid, bid);
554 return 0;
555 }
556 int kept = 0;
557 for (int i = 0; i < first->parent_count; ++i)
558 if (first->parents[i] != b)
559 first->parents[kept++] = first->parents[i];
560 first->parent_count = kept;
561 printf("UNLINKED %s %s\n", aid, bid);
562 return 1;
563 }
564 int year = atoi(words[3]);
565 if (strcmp(command, "SPOUSE") == 0) {
566 if (a == b) {
567 printf("ERROR SAME PERSON %s\n", aid);
568 return 0;
569 }
570 int couple[] = {a, b};
571 for (int i = 0; i < 2; ++i)
572 if (family->persons[couple[i]].death) {
573 printf("ERROR DECEASED %s\n", family->persons[couple[i]].id);
574 return 0;
575 }
576 for (int i = 0; i < 2; ++i)
577 if (spouse(family, couple[i]) >= 0) {
578 printf("ERROR ALREADY MARRIED %s\n", family->persons[couple[i]].id);
579 return 0;
580 }
581 if (ancestor(family, a, b) || ancestor(family, b, a) || siblings(family, a, b)) {
582 printf("ERROR CLOSE KIN %s %s\n", aid, bid);
583 return 0;
584 }
585 for (int i = 0; i < 2; ++i)
586 if (year < family->persons[couple[i]].birth + 16) {
587 printf("ERROR TOO YOUNG %s\n", family->persons[couple[i]].id);
588 return 0;
589 }
590 for (int i = 0; i < 2; ++i)
591 for (int j = 0; j < family->record_count; ++j) {
592 Marriage *record = &family->marriages[j];
593 if (contains(record, couple[i]) && record->end && record->end >= year) {
594 printf("ERROR BAD YEAR %s\n", family->persons[couple[i]].id);
595 return 0;
596 }
597 }
598 family->marriages[family->record_count++] = (Marriage){a, b, year, 0, CURRENT, -1};
599 printf("MARRIED %s %s %d\n", aid, bid, year);
600 return 1;
601 }
602 if (spouse(family, a) != b) {
603 printf("ERROR NOT MARRIED %s %s\n", aid, bid);
604 return 0;
605 }
606 Marriage *record = &family->marriages[current_record(family, a)];
607 if (year <= record->start) {
608 printf("ERROR BAD YEAR %s\n", aid);
609 return 0;
610 }
611 record->end = year;
612 record->status = DIVORCED;
613 printf("DIVORCED %s %s %d\n", aid, bid, year);
614 return 1;
615}

617static void tree_visit(Family *family, int id, int depth, int visited[]) {
618 printf("%*s%s", 2 * (depth + 1), "", family->persons[id].id);
619 if (visited[id]) {
620 printf(" (see above)\n");
621 return;
622 }
623 visited[id] = 1;
624 printf(" %s ", family->persons[id].name);
625 print_life(&family->persons[id]);
626 int partner = spouse(family, id);
627 if (partner >= 0)
628 printf(" = %s", family->persons[partner].id);

636static void pedigree_visit(Family *family, int id, int number, int generation, int depth, int emit,
637 int *entries, int seen[]) {
638 ++*entries;
639 seen[id] = 1;
640 if (emit) {
641 printf("%*s%d %s %s ", 2 * (generation + 1), "", number, family->persons[id].id,
642 family->persons[id].name);
643 print_life(&family->persons[id]);
644 putchar('\n');
645 }
646 if (generation == depth)
647 return;
648 int up[2], count = family->persons[id].parent_count;
649 for (int i = 0; i < count; ++i)
650 up[i] = family->persons[id].parents[i];
651 order_people(family, up, count, 3, NULL);
652 for (int i = 0; i < count; ++i)
653 pedigree_visit(family, up[i], 2 * number + i, generation + 1, depth, emit, entries, seen);
654}

656static int lineage_length(Family *family, int id, int memo[]) {
657 if (memo[id])
658 return memo[id];
659 int result = 1;
660 for (int i = 0; i < family->persons[id].parent_count; ++i) {
661 int length = 1 + lineage_length(family, family->persons[id].parents[i], memo);
662 if (length > result)
663 result = length;
664 }
665 return memo[id] = result;
666}

668static int compare_events(const void *left, const void *right) {
669 const Event *a = left, *b = right;
670 if (a->year != b->year)
671 return a->year < b->year ? -1 : 1;
672 if (a->rank != b->rank)
673 return a->rank - b->rank;
674 if (a->counterpart < 0 || b->counterpart < 0)
675 return 0;
676 return strcmp(sort_family->persons[a->counterpart].id, sort_family->persons[b->counterpart].id);
677}

679static void timeline(Family *family, int id) {
680 Event events[2600];
681 int count = 0;
682 Person *person = &family->persons[id];
683 events[count++] = (Event){person->birth, 0, -1, "born"};
684 for (int i = 0; i < family->record_count; ++i) {
685 Marriage *record = &family->marriages[i];
686 if (!contains(record, id))
687 continue;
688 int counterpart = other(record, id);
689 events[count] = (Event){record->start, 1, counterpart, ""};
690 snprintf(events[count++].text, sizeof(events[0].text), "married %s",
691 family->persons[counterpart].id);
692 if (record->status == DIVORCED ||
693 (record->status == WIDOWED && record->died == counterpart)) {
694 events[count] =
695 (Event){record->end, record->status == DIVORCED ? 3 : 4, counterpart, ""};
696 snprintf(events[count++].text, sizeof(events[0].text), "%s %s",
697 record->status == DIVORCED ? "divorced" : "widowed",
698 family->persons[counterpart].id);
699 }
700 }
701 int offspring[MAX_PERSONS], child_count = children(family, id, offspring);
702 for (int i = 0; i < child_count; ++i) {
703 int child = offspring[i];
704 events[count] = (Event){family->persons[child].birth, 2, child, ""};
705 snprintf(events[count++].text, sizeof(events[0].text), "child %s",
706 family->persons[child].id);
707 }
708 for (int group = 0; group < 2; ++group) {
709 int relative_count = group ? child_count : person->parent_count;
710 for (int i = 0; i < relative_count; ++i) {
711 int relative = group ? offspring[i] : person->parents[i];
712 int death = family->persons[relative].death;
713 if (death && death >= person->birth && (!person->death || death <= person->death)) {
714 events[count] = (Event){death, 5, relative, ""};
715 snprintf(events[count++].text, sizeof(events[0].text), "lost %s %s",
716 group ? "child" : "parent", family->persons[relative].id);
717 }
718 }
719 }
720 if (person->death)
721 events[count++] = (Event){person->death, 6, -1, "died"};
722 sort_family = family;
723 qsort(events, (size_t)count, sizeof(events[0]), compare_events);
724 printf("TIMELINE %s %d\n", person->id, count);
725 for (int i = 0; i < count; ++i)
726 printf(" %d %s\n", events[i].year, events[i].text);
727}
