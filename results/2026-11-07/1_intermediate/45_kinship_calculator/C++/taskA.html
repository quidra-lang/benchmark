26string gendered(const string &kind, char sex) {
27 static const std::map<string, std::array<string, 3>> terms = {
28 {"parent", {"father", "mother", "parent"}},
29 {"child", {"son", "daughter", "child"}},
30 {"sibling", {"brother", "sister", "sibling"}},
31 {"pibling", {"uncle", "aunt", "pibling"}},
32 {"nibling", {"nephew", "niece", "nibling"}},
33 {"spouse", {"husband", "wife", "spouse"}},
34 {"late", {"late husband", "late wife", "late spouse"}},
35 {"survivor", {"widower", "widow", "surviving spouse"}},
36 {"ex", {"ex-husband", "ex-wife", "ex-spouse"}},
37 {"stepparent", {"stepfather", "stepmother", "stepparent"}},
38 {"stepchild", {"stepson", "stepdaughter", "stepchild"}},
39 {"stepsibling", {"stepbrother", "stepsister", "stepsibling"}},
40 {"stepgrandparent", {"stepgrandfather", "stepgrandmother", "stepgrandparent"}},
41 {"stepgrandchild", {"stepgrandson", "stepgranddaughter", "stepgrandchild"}},
42 {"parent-in-law", {"father-in-law", "mother-in-law", "parent-in-law"}},
43 {"child-in-law", {"son-in-law", "daughter-in-law", "child-in-law"}},
44 {"sibling-in-law", {"brother-in-law", "sister-in-law", "sibling-in-law"}},
45 };
46 return terms.at(kind)[sex_index(sex)];
47}

49string generation_prefix(int g) {
50 if (g == 1)
51 return "";
52 string prefix;
53 for (int i = 2; i < g; ++i)
54 prefix += "great-";
55 return prefix + "grand";
56}

58string ordinal(int k) {
59 static const std::array<string, 10> words = {"first", "second", "third", "fourth", "fifth",
60 "sixth", "seventh", "eighth", "ninth", "tenth"};
61 if (k <= 10)
62 return words[k - 1];
63 string suffix = "th";
64 if (k % 100 < 11 || k % 100 > 13) {
65 if (k % 10 == 1)
66 suffix = "st";
67 if (k % 10 == 2)
68 suffix = "nd";
69 if (k % 10 == 3)
70 suffix = "rd";
71 }
72 return std::to_string(k) + suffix;
73}

75string term(int da, int db, char sex) {
76 if (da == 0 && db == 0)
77 return "self";
78 if (da == 0)
79 return generation_prefix(db) + gendered("parent", sex);
80 if (db == 0)
81 return generation_prefix(da) + gendered("child", sex);
82 if (da == 1 && db == 1)
83 return gendered("sibling", sex);
84 if (da == 1)
85 return generation_prefix(db - 1) + gendered("pibling", sex);
86 if (db == 1)
87 return generation_prefix(da - 1) + gendered("nibling", sex);
88 int removed = std::abs(da - db);
89 string removal;
90 if (removed == 1)
91 removal = " once removed";
92 else if (removed == 2)
93 removal = " twice removed";
94 else if (removed >= 3)
95 removal = " " + std::to_string(removed) + " times removed";
96 return ordinal(std::min(da, db) - 1) + " cousin" + removal;
97}

101 return std::to_string(value / divisor) + "/" + std::to_string(scale / divisor);

275 const std::map<string, int> &distances(const string &start, bool upward) {
276 std::pair<string, bool> key{start, upward};
277 auto cached = distance_cache.find(key);
278 if (cached != distance_cache.end())
279 return cached->second;
280 std::map<string, int> result{{start, 0}};
281 vector<string> queue{start};
282 for (std::size_t i = 0; i < queue.size(); ++i) {
283 string x = queue[i];
284 vector<string> adjacent = upward ? persons.at(x).parents : children(x);
285 for (const string &y : adjacent) {
286 if (result.count(y))
287 continue;
288 result[y] = result.at(x) + 1;
289 queue.push_back(y);
290 }
291 }
292 return distance_cache.emplace(key, std::move(result)).first->second;
293 }

300 vector<string> result;

302 if (g == 2)
303 result.push_back(y);
304 return result;
305 }

307 Apex apex(const string &a, const string &b) {
308 const std::map<string, int> &up_a = distances(a, true);
309 const std::map<string, int> &up_b = distances(b, true);
310 if (up_b.count(a))
311 return {a, 0, up_b.at(a), 0};
312 if (up_a.count(b))
313 return {b, up_a.at(b), 0, 0};
314 Apex best;
315 for (const auto &[c, da] : up_a) {
316 auto it = up_b.find(c);
317 if (it == up_b.end())
318 continue;
319 int db = it->second;
320 if (!best || std::make_tuple(da + db, da, c) <
321 std::make_tuple(best.da + best.db, best.da, best.person))
322 best = {c, da, db, 0};
323 }
324 if (best)
325 for (const auto &[c, da] : up_a) {
326 auto it = up_b.find(c);
327 if (it != up_b.end() && da == best.da && it->second == best.db)
328 ++best.shares;
329 }
330 return best;
331 }

333 vector<string> relationship(const string &a, const string &b, char sex) {
334 if (a == b)
335 return {"self"};
336 vector<string> result;
337 Apex top = apex(a, b);
338 if (top)
339 result.push_back(top.blood_term(sex));
340 const Marriage *m = last(a, b);
341 if (!m) {
342 for (const auto &[id, p] : persons)
343 if (has(p.parents, a) && has(p.parents, b)) {
344 result.push_back("co-parent");
345 break;
346 }
347 } else {
348 string kind = "ex";
349 if (m->status == "current")
350 kind = "spouse";
351 else if (m->status == "widowed")
352 kind = m->died == a ? "late" : "survivor";
353 result.push_back(gendered(kind, sex));
354 }
355 const vector<string> &pa = persons.at(a).parents;
356 const vector<string> &pb = persons.at(b).parents;
357 std::set<string> qa = partners(a), qb = partners(b);
358 if (!has(pb, a))
359 for (const string &p : pb)
360 if (qa.count(p)) {
361 result.push_back(gendered("stepparent", sex));
362 break;
363 }
364 if (!has(pa, b))
365 for (const string &p : pa)
366 if (qb.count(p)) {
367 result.push_back(gendered("stepchild", sex));
368 break;
369 }
370 if (common_parents(a, b) == 0 && step_related(a, b))
371 result.push_back(gendered("stepsibling", sex));
372 if (!ancestor(a, b))
373 for (const string &g : grandparents(b))
374 if (qa.count(g)) {
375 result.push_back(gendered("stepgrandparent", sex));
376 break;
377 }
378 if (!ancestor(b, a))
379 for (const string &g : grandparents(a))
380 if (qb.count(g)) {
381 result.push_back(gendered("stepgrandchild", sex));
382 break;
383 }
384 for (const string &q : qb)
385 if (has(persons.at(q).parents, a)) {
386 result.push_back(gendered("parent-in-law", sex));
387 break;
388 }
389 for (const string &q : qa)
390 if (has(persons.at(q).parents, b)) {
391 result.push_back(gendered("child-in-law", sex));
392 break;
393 }
394 bool in_law = false;
395 for (const string &q : qb)
396 if (are_siblings(a, q))
397 in_law = true;
398 for (const string &q : qa)
399 if (are_siblings(q, b))
400 in_law = true;
401 if (in_law)
402 result.push_back(gendered("sibling-in-law", sex));
403 return result;
404 }

407 vector<string> result = relationship(a, b, persons.at(a).sex);

411 std::int64_t coefficient(const string &a, const string &b) {
412 std::pair<string, string> key{std::min(a, b), std::max(a, b)};
413 auto cached = coefficient_cache.find(key);
414 if (cached != coefficient_cache.end())
415 return cached->second;
416 std::int64_t value = 0;
417 if (a == b)
418 value = (scale + inbreeding(a)) / 2;
419 else {
420 string x = a, y = b;
421 if (persons.at(b).birth > persons.at(a).birth)
422 std::swap(x, y);
423 for (const string &p : persons.at(x).parents)
424 value += coefficient(p, y);
425 value /= 2;
426 }
427 coefficient_cache[key] = value;
428 return value;
429 }

431 std::int64_t inbreeding(const string &x) {
432 const vector<string> &parents = persons.at(x).parents;
433 return parents.size() == 2 ? coefficient(parents[0], parents[1]) : 0;
434 }

436 int lineage_length(const string &x, std::map<string, int> &memo) {
437 auto it = memo.find(x);
438 if (it != memo.end())
439 return it->second;
440 int length = 1;
441 for (const string &p : persons.at(x).parents)
442 length = std::max(length, 1 + lineage_length(p, memo));
443 memo[x] = length;
444 return length;
445 }

447 vector<std::pair<int, string>> lineage(const string &x, bool upward) {
448 vector<std::pair<int, string>> result;
449 for (const auto &[y, g] : distances(x, upward))
450 if (y != x)
451 result.emplace_back(g, y);
452 std::sort(result.begin(), result.end());
453 return result;
454 }

460string Family::modify(const string &command, const vector<string> &args) {
461 const string &x = args[0];
462 if (command == "PERSON") {
463 if (persons.count(x))
464 throw std::runtime_error("DUPLICATE PERSON " + x);
465 persons.emplace(x, Person{args[1][0], std::stoi(args[2]), args[3]});
466 return "ADDED " + x + " " + args[3] + " " + args[1] + " " + args[2];
467 }
468 Person &p = person(x);
469 if (command == "PARENT" || command == "UNPARENT" || command == "SPOUSE" || command == "DIVORCE")
470 person(args[1]);
471 if (command == "DEATH") {
472 int year = std::stoi(args[1]);
473 if (p.death)
474 throw std::runtime_error("ALREADY DECEASED " + x);
475 bool bad = year < p.birth;
476 for (const Marriage &m : marriages)
477 if (m.contains(x) && (year < m.start || (m.end && year < m.end)))
478 bad = true;
479 if (bad)
480 throw std::runtime_error("BAD YEAR " + x);
481 for (const string &child : children(x))
482 if (persons.at(child).birth > year + 1)
483 throw std::runtime_error("AFTER DEATH " + child + " " + x);
484 p.death = year;
485 Marriage *m = current(x);
486 string output = "DIED " + x + " " + std::to_string(year);
487 if (m) {
488 m->end = year;
489 m->status = "widowed";
490 m->died = x;
491 output += " spouse=" + m->other(x);
492 }
493 return output;
494 }
495 if (command == "REMOVE") {
496 for (auto &[id, person] : persons) {
497 vector<string> &parents = person.parents;
498 parents.erase(std::remove(parents.begin(), parents.end(), x), parents.end());
499 }
500 marriages.erase(std::remove_if(marriages.begin(), marriages.end(),
501 [&x](const Marriage &m) { return m.contains(x); }),
502 marriages.end());
503 persons.erase(x);
504 return "REMOVED " + x;
505 }
506 const string &y = args[1];
507 if (command == "PARENT") {
508 Person &parent = persons.at(y);
509 if (x == y)
510 throw std::runtime_error("SAME PERSON " + x);
511 if (has(p.parents, y))
512 throw std::runtime_error("ALREADY PARENT " + x + " " + y);
513 if (p.parents.size() == 2)
514 throw std::runtime_error("TOO MANY PARENTS " + x);
515 if (ancestor(x, y))
516 throw std::runtime_error("CYCLE " + x + " " + y);
517 if (parent.birth + 12 > p.birth)
518 throw std::runtime_error("AGE GAP " + x + " " + y);
519 if (parent.death && parent.death + 1 < p.birth)
520 throw std::runtime_error("AFTER DEATH " + x + " " + y);
521 p.parents.push_back(y);
522 return "LINKED " + x + " " + y + " " + gendered("parent", parent.sex);
523 }
524 if (command == "UNPARENT") {
525 if (!has(p.parents, y))
526 throw std::runtime_error("NOT PARENT " + x + " " + y);
527 p.parents.erase(std::find(p.parents.begin(), p.parents.end(), y));
528 return "UNLINKED " + x + " " + y;
529 }
530 int year = std::stoi(args[2]);
531 if (command == "SPOUSE") {
532 if (x == y)
533 throw std::runtime_error("SAME PERSON " + x);
534 for (const string &a : {x, y})
535 if (persons.at(a).death)
536 throw std::runtime_error("DECEASED " + a);
537 for (const string &a : {x, y})
538 if (!spouse(a).empty())
539 throw std::runtime_error("ALREADY MARRIED " + a);
540 if (ancestor(x, y) || ancestor(y, x) || are_siblings(x, y))
541 throw std::runtime_error("CLOSE KIN " + x + " " + y);
542 for (const string &a : {x, y})
543 if (year < persons.at(a).birth + 16)
544 throw std::runtime_error("TOO YOUNG " + a);
545 for (const string &a : {x, y})
546 for (const Marriage &m : marriages)
547 if (m.contains(a) && m.end && m.end >= year)
548 throw std::runtime_error("BAD YEAR " + a);
549 marriages.push_back(Marriage{x, y, year});
550 return "MARRIED " + x + " " + y + " " + std::to_string(year);
551 }
552 if (spouse(x) != y)
553 throw std::runtime_error("NOT MARRIED " + x + " " + y);
554 Marriage *m = current(x);
555 if (year <= m->start)
556 throw std::runtime_error("BAD YEAR " + x);
557 m->end = year;
558 m->status = "divorced";
559 return "DIVORCED " + x + " " + y + " " + std::to_string(year);
560}

562vector<string> Family::query(const string &command, const vector<string> &args) {
563 if (command == "TERM")
564 return {"TERM " + join(args, " ") + " " +
565 term(std::stoi(args[0]), std::stoi(args[1]), args[2][0])};
566 if (command == "STATS") {
567 int living = 0, links = 0, couples = 0, founders = 0, longest = 0;
568 std::map<string, int> memo;
569 for (const auto &[x, p] : persons) {
570 if (!p.death)
571 ++living;
572 links += static_cast<int>(p.parents.size());
573 if (p.parents.empty())
574 ++founders;
575 longest = std::max(longest, lineage_length(x, memo));
576 }
577 for (const Marriage &m : marriages)
578 if (m.status == "current")
579 ++couples;
580 return {"STATS persons=" + std::to_string(persons.size()) +
581 " living=" + std::to_string(living) + " links=" + std::to_string(links) +
582 " couples=" + std::to_string(couples) + " founders=" + std::to_string(founders) +
583 " generations=" + std::to_string(longest)};
584 }
585 if (command == "AUDIT") {
586 vector<string> lines;
587 for (const Marriage &m : marriages) {
588 Apex top = apex(m.first, m.second);
589 if (top)
590 lines.push_back(" marriage " + m.first + " " + m.second + " " +
591 top.blood_term(persons.at(m.first).sex));
592 }
593 for (const auto &[x, p] : persons) {
594 std::int64_t value = inbreeding(x);
595 if (value)
596 lines.push_back(" inbred " + x + " " + fraction(value));
597 }
598 lines.insert(lines.begin(), "AUDIT " + std::to_string(lines.size()));
599 return lines;
600 }
601 if (command == "CENSUS") {
602 int year = std::stoi(args[0]);
603 vector<string> residents;
604 for (const auto &[x, p] : persons)
605 if (p.birth <= year && (!p.death || p.death > year))
606 residents.push_back(x);
607 vector<string> lines{"CENSUS " + args[0] + " " + std::to_string(residents.size())};
608 for (const string &x : age_order(residents)) {
609 const Marriage *last = nullptr;
610 for (const Marriage &m : marriages)
611 if (m.contains(x) && m.start <= year)
612 last = &m;
613 string status = "single";
614 if (last)
615 status =
616 !last->end || last->end > year ? "married " + last->other(x) : last->status;
617 lines.push_back(" " + x + " " + std::to_string(year - persons.at(x).birth) + " " +
618 status);
619 }
620 return lines;
621 }
622 const string &x = args[0];
623 Person &p = person(x);
624 if (command == "RELATION" || command == "KINSHIP" || command == "PATH" || command == "COMMON")
625 person(args[1]);
626 if (command == "INFO") {
627 vector<string> lines{"INFO " + x + " " + p.name + " " + p.sex + " " + p.life(),
628 " parents: " + person_list(p.parents)};
629 for (const Marriage &m : marriages)
630 if (m.contains(x))
631 lines.push_back(" marriage: " + m.other(x) + " " + std::to_string(m.start) + "-" +
632 (m.end ? std::to_string(m.end) : "") + " " + m.status);
633 lines.push_back(" children: " + person_list(children(x)));
634 lines.push_back(" siblings: " + person_list(siblings(x)));
635 lines.push_back(" inbreeding: " + fraction(inbreeding(x)));
636 return lines;
637 }
638 if (command == "RELATION")
639 return {"RELATION " + x + " " + args[1] + ": " + relationship_text(x, args[1])};
640 if (command == "KINSHIP")
641 return {"KINSHIP " + x + " " + args[1] + " " + fraction(coefficient(x, args[1]))};
642 if (command == "RELATIVES" || command == "WHO") {
643 vector<string> lines;
644 string query = join(vector<string>(args.begin() + 1, args.end()), " ");
645 for (const auto &[y, py] : persons) {
646 if (y == x)
647 continue;
648 vector<string> actual = relationship(y, x, py.sex);
649 if (command == "RELATIVES") {
650 if (!actual.empty())
651 lines.push_back(" " + y + ": " + join(actual, ", "));
652 } else {
653 vector<string> neutral = relationship(y, x, 'X');
654 for (std::size_t i = 0; i < actual.size(); ++i)
655 if (actual[i] == query || neutral[i] == query) {
656 lines.push_back(" " + y + ": " + actual[i]);
657 break;
658 }
659 }
660 }
661 string header = command + " " + x + (command == "WHO" ? " " + query : "") + " " +
662 std::to_string(lines.size());
663 lines.insert(lines.begin(), header);
664 return lines;
665 }
666 if (command == "PATH") {
667 const string &y = args[1];
668 Apex top = apex(x, y);
669 if (!top)
670 return {"PATH " + x + " " + y + " none"};
671 const std::map<string, int> &depths = distances(top.person, false);
672 auto upward = [&](string current) {
673 vector<string> path{current};
674 while (current != top.person) {
675 vector<string> choices;
676 for (const string &parent : persons.at(current).parents)
677 if (depths.count(parent) && depths.at(parent) == depths.at(current) - 1)
678 choices.push_back(parent);
679 current = *std::min_element(choices.begin(), choices.end());
680 path.push_back(current);
681 }
682 return path;
683 };
684 vector<string> route = upward(x), descent = upward(y);
685 for (int i = static_cast<int>(descent.size()) - 2; i >= 0; --i)
686 route.push_back(descent[i]);
687 return {"PATH " + x + " " + y + " " + std::to_string(top.da + top.db) + ": " +
688 join(route, " ")};
689 }
690 if (command == "COMMON") {
691 const string &y = args[1];
692 const std::map<string, int> &a = distances(x, true);
693 const std::map<string, int> &b = distances(y, true);
694 std::set<string> shared;
695 for (const auto &[c, g] : a)
696 if (b.count(c))
697 shared.insert(c);
698 vector<string> nearest;
699 for (const string &c : shared) {
700 vector<string> offspring = children(c);
701 if (std::none_of(offspring.begin(), offspring.end(),
702 [&](const string &child) { return shared.count(child); }))
703 nearest.push_back(c);
704 }
705 std::sort(nearest.begin(), nearest.end(), [&](const string &c, const string &d) {
706 return std::make_pair(a.at(c) + b.at(c), c) < std::make_pair(a.at(d) + b.at(d), d);
707 });
708 vector<string> lines{"COMMON " + x + " " + y + " " + std::to_string(nearest.size())};
709 for (const string &c : nearest)
710 lines.push_back(" " + c + " " + std::to_string(a.at(c)) + " " +
711 std::to_string(b.at(c)));
712 return lines;
713 }
714 if (command == "ANCESTORS" || command == "DESCENDANTS") {
715 bool upward = command == "ANCESTORS";
716 vector<std::pair<int, string>> entries = lineage(x, upward);
717 vector<string> lines{command + " " + x + " " + std::to_string(entries.size())};
718 for (const auto &[g, y] : entries)
719 lines.push_back(" " + std::to_string(g) + " " + y + " " +
720 term(upward ? 0 : g, upward ? g : 0, persons.at(y).sex));
721 return lines;
722 }
723 if (command == "COUSINS") {
724 vector<std::tuple<int, string, string>> matches;
725 for (const auto &[y, py] : persons) {
726 if (x == y)
727 continue;
728 Apex top = apex(y, x);
729 if (top && top.da >= 2 && top.db >= 2 &&
730 std::min(top.da, top.db) - 1 == std::stoi(args[1]))
731 matches.emplace_back(std::abs(top.da - top.db), y, top.blood_term(py.sex));
732 }
733 std::sort(matches.begin(), matches.end());
734 vector<string> lines{"COUSINS " + x + " " + args[1] + " " + std::to_string(matches.size())};
735 for (const auto &[removed, y, blood] : matches)
736 lines.push_back(" " + y + " " + blood);
737 return lines;
738 }
739 if (command == "SIBLINGS") {
740 vector<string> lines;
741 int earlier = 0;
742 for (const string &y : age_order(ids())) {
743 if (x == y)
744 continue;
745 int common = common_parents(x, y);
746 string kind = common == 2 ? "full"
747 : common == 1 ? "half"
748 : step_related(x, y) ? "step"
749 : "";
750 if (kind.empty())
751 continue;
752 if (common > 0 && age_less(y, x))
753 ++earlier;
754 lines.push_back(" " + y + " " + kind + " " + std::to_string(persons.at(y).birth));
755 }
756 lines.insert(lines.begin(), "SIBLINGS " + x + " " + std::to_string(lines.size()) +
757 " order=" + std::to_string(earlier + 1));
758 return lines;
759 }
760 if (command == "GENERATIONS") {
761 std::map<int, vector<string>> groups;
762 for (const auto &[g, y] : lineage(x, false))
763 groups[g].push_back(y);
764 vector<string> lines{"GENERATIONS " + x + " " + std::to_string(groups.size())};
765 for (const auto &[g, members] : groups) {
766 int low = 2025, high = 1800, living = 0;
767 for (const string &y : members) {
768 const Person &py = persons.at(y);
769 low = std::min(low, py.birth);
770 high = std::max(high, py.birth);
771 if (!py.death)
772 ++living;
773 }
774 lines.push_back(" " + std::to_string(g) + " " + std::to_string(members.size()) + " " +
775 std::to_string(living) + " " + std::to_string(low) + "-" +
776 std::to_string(high));
777 }
778 return lines;
779 }
780 if (command == "TREE") {
781 vector<string> lines{"TREE " + x + " " + std::to_string(lineage(x, false).size())};
782 std::set<string> visited;
783 std::function<void(const string &, int)> visit = [&](const string &y, int depth) {
784 string indent(2 * (depth + 1), ' ');
785 if (!visited.insert(y).second) {
786 lines.push_back(indent + y + " (see above)");
787 return;
788 }
789 const Person &py = persons.at(y);
790 string line = indent + y + " " + py.name + " " + py.life();
791 string partner = spouse(y);
792 if (!partner.empty())
793 line += " = " + partner;
794 lines.push_back(line);
795 for (const string &child : age_order(children(y)))
796 visit(child, depth + 1);
797 };
798 visit(x, 0);
799 return lines;
800 }
801 if (command == "PEDIGREE") {
802 int depth = std::stoi(args[1]);
803 vector<string> lines;
804 std::set<string> entered;
805 std::function<void(const string &, int, int)> enter = [&](const string &y, int number,
806 int generation) {
807 const Person &py = persons.at(y);
808 lines.push_back(string(2 * (generation + 1), ' ') + std::to_string(number) + " " + y +
809 " " + py.name + " " + py.life());
810 entered.insert(y);
811 if (generation >= depth)
812 return;
813 vector<string> parents = py.parents;
814 std::sort(parents.begin(), parents.end(), [&](const string &a, const string &b) {
815 return std::make_pair(sex_index(persons.at(a).sex), a) <
816 std::make_pair(sex_index(persons.at(b).sex), b);
817 });
818 for (std::size_t i = 0; i < parents.size(); ++i)
819 enter(parents[i], 2 * number + static_cast<int>(i), generation + 1);
820 };
821 enter(x, 1, 0);
822 lines.insert(lines.begin(), "PEDIGREE " + x + " " + args[1] +
823 " entries=" + std::to_string(lines.size()) +
824 " distinct=" + std::to_string(entered.size()));
825 return lines;
826 }
827 if (command == "TIMELINE") {
828 using Event = std::tuple<int, int, string, string>;
829 vector<Event> events{{p.birth, 0, "", "born"}};
830 for (const Marriage &m : marriages)
831 if (m.contains(x)) {
832 string y = m.other(x);
833 events.emplace_back(m.start, 1, y, "married " + y);
834 if (m.status == "divorced")
835 events.emplace_back(m.end, 3, y, "divorced " + y);
836 else if (m.status == "widowed" && m.died == y)
837 events.emplace_back(m.end, 4, y, "widowed " + y);
838 }
839 vector<string> offspring = children(x);
840 for (const string &y : offspring)
841 events.emplace_back(persons.at(y).birth, 2, y, "child " + y);
842 for (const auto &[label, relatives] : vector<std::pair<string, vector<string>>>{
843 {"parent", p.parents}, {"child", offspring}}) {
844 for (const string &y : relatives) {
845 int death = persons.at(y).death;
846 if (death && death >= p.birth && (!p.death || death <= p.death))
847 events.emplace_back(death, 5, y, "lost " + label + " " + y);
848 }
849 }
850 if (p.death)
851 events.emplace_back(p.death, 6, "", "died");
852 std::sort(events.begin(), events.end());
853 vector<string> lines{"TIMELINE " + x + " " + std::to_string(events.size())};
854 for (const auto &[year, rank, counterpart, text] : events)
855 lines.push_back(" " + std::to_string(year) + " " + text);
856 return lines;
857 }
858 if (command == "FIND") {
859 static const std::map<string, std::pair<string, char>> steps = {
860 {"parent", {"parent", 0}}, {"father", {"parent", 'M'}},
861 {"mother", {"parent", 'F'}}, {"child", {"child", 0}},
862 {"son", {"child", 'M'}}, {"daughter", {"child", 'F'}},
863 {"sibling", {"sibling", 0}}, {"brother", {"sibling", 'M'}},
864 {"sister", {"sibling", 'F'}}, {"spouse", {"spouse", 0}},
865 {"husband", {"spouse", 'M'}}, {"wife", {"spouse", 'F'}}};
866 std::set<string> current{x};
867 std::istringstream route(args[1]);
868 string step;
869 while (std::getline(route, step, '.')) {
870 const auto &[kind, sex] = steps.at(step);
871 std::set<string> next;
872 for (const string &y : current) {
873 vector<string> targets;
874 if (kind == "parent")
875 targets = persons.at(y).parents;
876 else if (kind == "child")
877 targets = children(y);
878 else if (kind == "sibling")
879 targets = siblings(y);
880 else {
881 string partner = spouse(y);
882 if (!partner.empty())
883 targets.push_back(partner);
884 }
885 for (const string &z : targets)
886 if (!sex || persons.at(z).sex == sex)
887 next.insert(z);
888 }
889 current = std::move(next);
890 }
891 current.erase(x);
892 vector<string> lines{"FIND " + x + " " + args[1] + " " + std::to_string(current.size())};
893 for (const string &y : current)
894 lines.push_back(" " + y + ": " + relationship_text(y, x));
895 return lines;
896 }
897 throw std::logic_error("unknown command");
898}
