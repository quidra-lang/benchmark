37function gendered(string $kind, string $sex): string
38{
39 return TERMS[$kind][SEX_INDEX[$sex]];
40}

42function generationPrefix(int $g): string
43{
44 return $g === 1 ? '' : str_repeat('great-', $g - 2) . 'grand';
45}
46
47function ordinal(int $number): string
48{
49 if ($number <= 10) {
50 return ORDINALS[$number - 1];
51 }
52 $suffix = in_array($number % 100, [11, 12, 13], true)
53 ? 'th' : ([1 => 'st', 2 => 'nd', 3 => 'rd'][$number % 10] ?? 'th');
54 return $number . $suffix;
55}
56
57function kinshipTerm(int $da, int $db, string $sex): string
58{
59 if ($da === 0 && $db === 0) {
60 return 'self';
61 }
62 if ($da === 0) {
63 return generationPrefix($db) . gendered('parent', $sex);
64 }
65 if ($db === 0) {
66 return generationPrefix($da) . gendered('child', $sex);
67 }
68 if ($da === 1 && $db === 1) {
69 return gendered('sibling', $sex);
70 }
71 if ($da === 1) {
72 return generationPrefix($db - 1) . gendered('pibling', $sex);
73 }
74 if ($db === 1) {
75 return generationPrefix($da - 1) . gendered('nibling', $sex);
76 }
77 $distance = abs($da - $db);
78 $removal = match ($distance) {
79 0 => '', 1 => ' once removed', 2 => ' twice removed',
80 default => " $distance times removed",
81 };
82 return ordinal(min($da, $db) - 1) . ' cousin' . $removal;
83}
84
85function bloodTerm(array $apex, string $sex): string
86{
87 $prefix = $apex['shares'] === 1 ? 'half-' : ($apex['shares'] >= 4 ? 'double ' : '');
88 return $prefix . kinshipTerm($apex['da'], $apex['db'], $sex);
89}

91function lifeYears(array $person): string

93 return $person['birth'] . '-' . ($person['death'] ?? '');

96function fractionText(int $value): string
97{
98 $a = $value;
99 $b = SCALE;
100 while ($b !== 0) {
101 [$a, $b] = [$b, $a % $b];
102 }
103 return intdiv($value, $a) . '/' . intdiv(SCALE, $a);
104}

106function containsPerson(array $record, string $id): bool
107{
108 return $record['first'] === $id || $record['second'] === $id;

111function otherPerson(array $record, string $id): string
112{
113 return $record['first'] === $id ? $record['second'] : $record['first'];

194 public function currentIndex(string $id): ?int
195 {
196 foreach ($this->marriages as $index => $record) {

198 return $index;

206 $index = $this->currentIndex($id);
207 return $index === null ? null : otherPerson($this->marriages[$index], $id);

210 public function lastRecord(string $a, string $b): ?array
211 {
212 foreach (array_reverse($this->recordsOf($a)) as $record) {
213 if (containsPerson($record, $b)) {
214 return $record;
215 }
216 }
217 return null;
218 }
219
220 public function partners(string $id): array
221 {
222 if (!isset($this->partners[$id])) {
223 $latest = [];
224 foreach ($this->recordsOf($id) as $record) {
225 $latest[otherPerson($record, $id)] = $record;
226 }
227 $this->partners[$id] = array_keys(array_filter($latest, fn($record) => $record['status'] !== 'divorced'));
228 }
229 return $this->partners[$id];
230 }
231
232 public function stepRelated(string $a, string $b): bool
233 {
234 foreach ($this->parents($a) as $parent) {
235 if (array_intersect($this->partners($parent), $this->parents($b)) !== []) {
236 return true;
237 }
238 }
239 return false;
240 }

242 public function distances(string $start, bool $upward): array
243 {
244 $key = ($upward ? 'up:' : 'down:') . $start;
245 if (isset($this->distances[$key])) {
246 return $this->distances[$key];
247 }
248 $distances = [$start => 0];
249 $queue = [$start];
250 for ($index = 0; $index < count($queue); ++$index) {
251 $id = $queue[$index];
252 foreach ($upward ? $this->parents($id) : $this->children($id) as $other) {
253 if (!isset($distances[$other])) {
254 $distances[$other] = $distances[$id] + 1;
255 $queue[] = $other;
256 }
257 }
258 }
259 return $this->distances[$key] = $distances;
260 }
261
262 public function lineage(string $id, bool $upward): array
263 {
264 $entries = [];
265 foreach ($this->distances($id, $upward) as $other => $distance) {
266 if ($other !== $id) {
267 $entries[] = [$distance, $other];
268 }
269 }
270 usort($entries, fn($a, $b) => ($a[0] <=> $b[0]) ?: strcmp($a[1], $b[1]));
271 return $entries;
272 }
273
274 public function ancestor(string $a, string $b): bool
275 {
276 return $a !== $b && isset($this->distances($b, true)[$a]);
277 }
278
279 public function grandparents(string $id): array
280 {
281 return array_keys(array_filter($this->distances($id, true), fn($distance) => $distance === 2));
282 }
283
284 public function apex(string $a, string $b): ?array
285 {
286 $upA = $this->distances($a, true);
287 $upB = $this->distances($b, true);
288 if (isset($upB[$a])) {
289 return ['person' => $a, 'da' => 0, 'db' => $upB[$a], 'shares' => 0];
290 }
291 if (isset($upA[$b])) {
292 return ['person' => $b, 'da' => $upA[$b], 'db' => 0, 'shares' => 0];
293 }
294 $common = array_keys(array_intersect_key($upA, $upB));
295 if ($common === []) {
296 return null;
297 }
298 usort($common, fn($c, $d) => (($upA[$c] + $upB[$c]) <=> ($upA[$d] + $upB[$d]))
299 ?: ($upA[$c] <=> $upA[$d]) ?: strcmp($c, $d));
300 $chosen = $common[0];
301 $da = $upA[$chosen];
302 $db = $upB[$chosen];
303 $shares = count(array_filter($common, fn($id) => $upA[$id] === $da && $upB[$id] === $db));
304 return ['person' => $chosen, 'da' => $da, 'db' => $db, 'shares' => $shares];
305 }

307 public function relationship(string $a, string $b, ?string $sex = null): array
308 {
309 if ($a === $b) {
310 return ['self'];
311 }
312 $sex ??= $this->sex($a);
313 $terms = [];
314 $apex = $this->apex($a, $b);
315 if ($apex !== null) {
316 $terms[] = bloodTerm($apex, $sex);
317 }
318 $record = $this->lastRecord($a, $b);
319 if ($record === null) {
320 foreach ($this->persons as $person) {
321 if (in_array($a, $person['parents'], true) && in_array($b, $person['parents'], true)) {
322 $terms[] = 'co-parent';
323 break;
324 }
325 }
326 } elseif ($record['status'] === 'current') {
327 $terms[] = gendered('spouse', $sex);
328 } elseif ($record['status'] === 'widowed') {
329 $terms[] = gendered($record['died'] === $a ? 'late' : 'survivor', $sex);
330 } else {
331 $terms[] = gendered('ex', $sex);
332 }
333 $pa = $this->parents($a);
334 $pb = $this->parents($b);
335 $qa = $this->partners($a);
336 $qb = $this->partners($b);
337 if (!in_array($a, $pb, true) && array_intersect($qa, $pb) !== []) {
338 $terms[] = gendered('stepparent', $sex);
339 }
340 if (!in_array($b, $pa, true) && array_intersect($qb, $pa) !== []) {
341 $terms[] = gendered('stepchild', $sex);
342 }
343 if ($this->commonParents($a, $b) === 0 && $this->stepRelated($a, $b)) {
344 $terms[] = gendered('stepsibling', $sex);
345 }
346 if (!$this->ancestor($a, $b) && array_intersect($qa, $this->grandparents($b)) !== []) {
347 $terms[] = gendered('stepgrandparent', $sex);
348 }
349 if (!$this->ancestor($b, $a) && array_intersect($qb, $this->grandparents($a)) !== []) {
350 $terms[] = gendered('stepgrandchild', $sex);
351 }
352 if (array_any($qb, fn($partner) => in_array($a, $this->parents($partner), true))) {
353 $terms[] = gendered('parent-in-law', $sex);
354 }
355 if (array_any($qa, fn($partner) => in_array($b, $this->parents($partner), true))) {
356 $terms[] = gendered('child-in-law', $sex);
357 }
358 if (array_any($qb, fn($partner) => $this->areSiblings($a, $partner))
359 || array_any($qa, fn($partner) => $this->areSiblings($partner, $b))) {
360 $terms[] = gendered('sibling-in-law', $sex);
361 }
362 return $terms;
363 }

365 public function relationshipText(string $a, string $b): string
366 {
367 $terms = $this->relationship($a, $b);
368 return $terms === [] ? 'no relation' : implode(', ', $terms);
369 }

371 public function coefficient(string $a, string $b): int
372 {
373 $key = strcmp($a, $b) <= 0 ? "$a:$b" : "$b:$a";
374 if (isset($this->coefficients[$key])) {
375 return $this->coefficients[$key];
376 }
377 if ($a === $b) {
378 $value = intdiv(SCALE + $this->inbreeding($a), 2);
379 } else {
380 [$x, $y] = $this->persons[$a]['birth'] >= $this->persons[$b]['birth'] ? [$a, $b] : [$b, $a];
381 $sum = 0;
382 foreach ($this->parents($x) as $parent) {
383 $sum += $this->coefficient($parent, $y);
384 }
385 $value = intdiv($sum, 2);
386 }
387 return $this->coefficients[$key] = $value;
388 }
389
390 public function inbreeding(string $id): int
391 {
392 $parents = $this->parents($id);
393 return count($parents) === 2 ? $this->coefficient($parents[0], $parents[1]) : 0;
394 }

396 public function lineageLength(string $id, array &$memo): int
397 {
398 if (!isset($memo[$id])) {
399 $longest = 0;
400 foreach ($this->parents($id) as $parent) {
401 $longest = max($longest, $this->lineageLength($parent, $memo));
402 }
403 $memo[$id] = $longest + 1;
404 }
405 return $memo[$id];
406 }

408 public function upwardPath(string $start, string $apex): array
409 {
410 $path = [$start];
411 while (end($path) !== $apex) {
412 $current = end($path);
413 $distance = $this->distances($current, true)[$apex];
414 $candidates = array_values(array_filter($this->parents($current),
415 fn($parent) => ($this->distances($parent, true)[$apex] ?? -1) === $distance - 1));
416 sort($candidates, SORT_STRING);
417 $path[] = $candidates[0];
418 }
419 return $path;
420 }

443class KinshipCalculator
444{
445 private Family $family;
446 private array $history = [];
447
448 public function __construct()
449 {
450 $this->family = new Family();
451 }
452
453 private function checked(string ...$ids): void
454 {
455 foreach ($ids as $id) {
456 $this->family->person($id);
457 }
458 }
459
460 public function execute(string $line): array
461 {
462 $words = explode(' ', $line);
463 $command = array_shift($words);
464 $this->family->resetCache();
465 $modifying = in_array($command, MODIFYING, true);
466 $saved = $modifying ? [$this->family->persons, $this->family->marriages] : null;
467 try {
468 $lines = $this->process($command, $words);
469 if ($modifying) {
470 $this->history[] = [$line, $saved];
471 }
472 return $lines;
473 } catch (CommandError $error) {
474 return ['ERROR ' . $error->getMessage()];
475 }
476 }
477
478 private function process(string $command, array $args): array
479 {
480 switch ($command) {
481 case 'PERSON': return $this->addPerson(...$args);
482 case 'DEATH': return $this->recordDeath(...$args);
483 case 'REMOVE': return $this->removePerson(...$args);
484 case 'PARENT': return $this->linkParent(...$args);
485 case 'UNPARENT': return $this->unlinkParent(...$args);
486 case 'SPOUSE': return $this->marry(...$args);
487 case 'DIVORCE': return $this->divorce(...$args);
488 case 'UNDO':
489 if ($this->history === []) {
490 throw new CommandError('NOTHING TO UNDO');
491 }
492 [$line, $saved] = array_pop($this->history);
493 [$this->family->persons, $this->family->marriages] = $saved;
494 return ["UNDONE $line"];
495 case 'INFO': return $this->info($args[0]);
496 case 'RELATION':
497 [$a, $b] = $args;
498 $this->checked($a, $b);
499 return ["RELATION $a $b: " . $this->family->relationshipText($a, $b)];
500 case 'RELATIVES': return $this->relatives($args[0]);
501 case 'WHO':
502 $id = array_shift($args);
503 return $this->who($id, implode(' ', $args));
504 case 'PATH': return $this->path(...$args);
505 case 'COMMON': return $this->common(...$args);
506 case 'ANCESTORS': return $this->lineage($args[0], true);
507 case 'DESCENDANTS': return $this->lineage($args[0], false);
508 case 'COUSINS': return $this->cousins($args[0], (int)$args[1]);
509 case 'SIBLINGS': return $this->siblings($args[0]);
510 case 'GENERATIONS': return $this->generations($args[0]);
511 case 'TREE': return $this->tree($args[0]);
512 case 'PEDIGREE': return $this->pedigree($args[0], (int)$args[1]);
513 case 'TIMELINE': return $this->timeline($args[0]);
514 case 'CENSUS': return $this->census((int)$args[0]);
515 case 'KINSHIP':
516 [$a, $b] = $args;
517 $this->checked($a, $b);
518 return ["KINSHIP $a $b " . fractionText($this->family->coefficient($a, $b))];
519 case 'FIND': return $this->find(...$args);
520 case 'TERM':
521 [$da, $db, $sex] = $args;
522 return ["TERM $da $db $sex " . kinshipTerm((int)$da, (int)$db, $sex)];
523 case 'STATS': return $this->stats();
524 case 'AUDIT': return $this->audit();
525 }
526 return [];
527 }

529 private function addPerson(string $id, string $sex, string $birth, string $name): array
530 {
531 if (isset($this->family->persons[$id])) {
532 throw new CommandError("DUPLICATE PERSON $id");
533 }
534 $this->family->persons[$id] = ['sex' => $sex, 'birth' => (int)$birth,
535 'name' => $name, 'death' => null, 'parents' => []];
536 return ["ADDED $id $name $sex $birth"];
537 }
538
539 private function recordDeath(string $id, string $yearText): array
540 {
541 $year = (int)$yearText;
542 $person = $this->family->person($id);
543 if ($person['death'] !== null) {
544 throw new CommandError("ALREADY DECEASED $id");
545 }
546 $years = [$person['birth']];
547 foreach ($this->family->recordsOf($id) as $record) {
548 $years[] = $record['start'];
549 if ($record['end'] !== null) {
550 $years[] = $record['end'];
551 }
552 }
553 if ($year < max($years)) {
554 throw new CommandError("BAD YEAR $id");
555 }
556 $late = array_values(array_filter($this->family->children($id),
557 fn($child) => $this->family->persons[$child]['birth'] > $year + 1));
558 sort($late, SORT_STRING);
559 if ($late !== []) {
560 throw new CommandError("AFTER DEATH {$late[0]} $id");
561 }
562 $this->family->persons[$id]['death'] = $year;
563 $index = $this->family->currentIndex($id);
564 if ($index === null) {
565 return ["DIED $id $year"];
566 }
567 $other = otherPerson($this->family->marriages[$index], $id);
568 $this->family->marriages[$index]['end'] = $year;
569 $this->family->marriages[$index]['status'] = 'widowed';
570 $this->family->marriages[$index]['died'] = $id;
571 return ["DIED $id $year spouse=$other"];
572 }
573
574 private function removePerson(string $id): array
575 {
576 $this->checked($id);
577 foreach (array_keys($this->family->persons) as $other) {
578 $this->family->persons[$other]['parents'] = array_values(array_diff($this->family->parents($other), [$id]));
579 }
580 $this->family->marriages = array_values(array_filter($this->family->marriages,
581 fn($record) => !containsPerson($record, $id)));
582 unset($this->family->persons[$id]);
583 return ["REMOVED $id"];
584 }
585
586 private function linkParent(string $childId, string $parentId): array
587 {
588 $child = $this->family->person($childId);
589 $parent = $this->family->person($parentId);
590 if ($childId === $parentId) {
591 throw new CommandError("SAME PERSON $childId");
592 }
593 if (in_array($parentId, $child['parents'], true)) {
594 throw new CommandError("ALREADY PARENT $childId $parentId");
595 }
596 if (count($child['parents']) === 2) {
597 throw new CommandError("TOO MANY PARENTS $childId");
598 }
599 if ($this->family->ancestor($childId, $parentId)) {
600 throw new CommandError("CYCLE $childId $parentId");
601 }
602 if ($parent['birth'] + 12 > $child['birth']) {
603 throw new CommandError("AGE GAP $childId $parentId");
604 }
605 if ($parent['death'] !== null && $parent['death'] + 1 < $child['birth']) {
606 throw new CommandError("AFTER DEATH $childId $parentId");
607 }
608 $this->family->persons[$childId]['parents'][] = $parentId;
609 return ["LINKED $childId $parentId " . gendered('parent', $parent['sex'])];
610 }
611
612 private function unlinkParent(string $childId, string $parentId): array
613 {
614 $this->checked($childId, $parentId);
615 $parents = $this->family->parents($childId);
616 if (!in_array($parentId, $parents, true)) {
617 throw new CommandError("NOT PARENT $childId $parentId");
618 }
619 $this->family->persons[$childId]['parents'] = array_values(array_diff($parents, [$parentId]));
620 return ["UNLINKED $childId $parentId"];
621 }

623 private function marry(string $a, string $b, string $yearText): array
624 {
625 $this->checked($a, $b);
626 $year = (int)$yearText;
627 if ($a === $b) {
628 throw new CommandError("SAME PERSON $a");
629 }
630 foreach ([$a, $b] as $id) {
631 if ($this->family->persons[$id]['death'] !== null) {
632 throw new CommandError("DECEASED $id");
633 }
634 }
635 foreach ([$a, $b] as $id) {
636 if ($this->family->spouse($id) !== null) {
637 throw new CommandError("ALREADY MARRIED $id");
638 }
639 }
640 if ($this->family->ancestor($a, $b) || $this->family->ancestor($b, $a) || $this->family->areSiblings($a, $b)) {
641 throw new CommandError("CLOSE KIN $a $b");
642 }
643 foreach ([$a, $b] as $id) {
644 if ($year < $this->family->persons[$id]['birth'] + 16) {
645 throw new CommandError("TOO YOUNG $id");
646 }
647 }
648 foreach ([$a, $b] as $id) {
649 if (array_any($this->family->recordsOf($id), fn($record) => $record['end'] !== null && $record['end'] >= $year)) {
650 throw new CommandError("BAD YEAR $id");
651 }
652 }
653 $this->family->marriages[] = ['first' => $a, 'second' => $b, 'start' => $year,
654 'end' => null, 'status' => 'current', 'died' => null];
655 return ["MARRIED $a $b $year"];
656 }

729 private function path(string $a, string $b): array
730 {
731 $this->checked($a, $b);
732 $apex = $this->family->apex($a, $b);
733 if ($apex === null) {
734 return ["PATH $a $b none"];

737 $down = $this->family->upwardPath($b, $apex['person']);
738 array_pop($down);

740 return ["PATH $a $b " . ($apex['da'] + $apex['db']) . ': ' . implode(' ', $route)];

769 private function cousins(string $id, int $degree): array
770 {
771 $this->checked($id);
772 $entries = [];
773 foreach (array_keys($this->family->persons) as $other) {
774 if ($other === $id) {
775 continue;
776 }
777 $apex = $this->family->apex($other, $id);
778 if ($apex !== null && $apex['da'] >= 2 && $apex['db'] >= 2 && min($apex['da'], $apex['db']) - 1 === $degree) {
779 $entries[] = [abs($apex['da'] - $apex['db']), $other, bloodTerm($apex, $this->family->sex($other))];
780 }
781 }
782 usort($entries, fn($a, $b) => ($a[0] <=> $b[0]) ?: strcmp($a[1], $b[1]));
783 $lines = array_map(fn($entry) => " {$entry[1]} {$entry[2]}", $entries);
784 return array_merge(["COUSINS $id $degree " . count($entries)], $lines);

787 private function siblings(string $id): array
788 {
789 $this->checked($id);
790 $lines = [];
791 $earlier = 0;
792 foreach ($this->family->ageOrder(array_keys($this->family->persons)) as $other) {
793 if ($other === $id) {
794 continue;
795 }
796 $shared = $this->family->commonParents($id, $other);
797 $kind = $shared === 2 ? 'full' : ($shared === 1 ? 'half' : ($this->family->stepRelated($id, $other) ? 'step' : null));
798 if ($kind === null) {
799 continue;
800 }
801 if ($kind !== 'step' && $this->family->ageCompare($other, $id) < 0) {
802 ++$earlier;
803 }
804 $lines[] = " $other $kind {$this->family->persons[$other]['birth']}";
805 }
806 return array_merge(["SIBLINGS $id " . count($lines) . ' order=' . ($earlier + 1)], $lines);
807 }

836 {

879 private function timeline(string $id): array
880 {
881 $person = $this->family->person($id);
882 $events = [[$person['birth'], 0, '', 'born']];
883 foreach ($this->family->recordsOf($id) as $record) {
884 $other = otherPerson($record, $id);
885 $events[] = [$record['start'], 1, $other, "married $other"];
886 if ($record['status'] === 'divorced') {
887 $events[] = [$record['end'], 3, $other, "divorced $other"];
888 } elseif ($record['status'] === 'widowed' && $record['died'] === $other) {
889 $events[] = [$record['end'], 4, $other, "widowed $other"];
890 }
891 }
892 $children = $this->family->children($id);
893 foreach ($children as $child) {
894 $events[] = [$this->family->persons[$child]['birth'], 2, $child, "child $child"];
895 }
896 foreach (['parent' => $person['parents'], 'child' => $children] as $label => $relatives) {
897 foreach ($relatives as $other) {
898 $death = $this->family->persons[$other]['death'];
899 if ($death !== null && $death >= $person['birth'] && ($person['death'] === null || $death <= $person['death'])) {
900 $events[] = [$death, 5, $other, "lost $label $other"];
901 }
902 }
903 }
904 if ($person['death'] !== null) {
905 $events[] = [$person['death'], 6, '', 'died'];
906 }
907 usort($events, fn($a, $b) => ($a[0] <=> $b[0]) ?: ($a[1] <=> $b[1]) ?: strcmp($a[2], $b[2]));
908 $lines = array_map(fn($event) => " {$event[0]} {$event[3]}", $events);
909 return array_merge(["TIMELINE $id " . count($events)], $lines);
910 }
