95 case none
96 case number(Int)
97 case text(String)

100struct Condition {
101 let field: String
102 let operation: String
103 let value: QueryValue
104
105 func matches(_ issue: Issue, tracker: Tracker) -> Bool {
106 let actual = tracker.fieldValue(issue, field: field)
107 if operation == "~", case .text(let text) = actual, case .text(let part) = value {
108 return text.contains(part)
109 }
110 if operation == "=" {
111 return actual == value
112 }
113 if operation == "!=" {
114 return actual != value
115 }
116 guard case .number(let a) = actual, case .number(let b) = value else {
117 return false
118 }
119 switch operation {
120 case "<":
121 return a < b
122 case "<=":
123 return a <= b
124 case ">":
125 return a > b
126 default:
127 return a >= b
128 }
129 }
130}
131
132indirect enum QueryNode {
133 case condition(Condition)
134 case not(QueryNode)
135 case and(QueryNode, QueryNode)
136 case or(QueryNode, QueryNode)
137
138 func matches(_ issue: Issue, tracker: Tracker) -> Bool {
139 switch self {
140 case .condition(let condition):
141 return condition.matches(issue, tracker: tracker)
142 case .not(let node):
143 return !node.matches(issue, tracker: tracker)
144 case .and(let a, let b):
145 return a.matches(issue, tracker: tracker) && b.matches(issue, tracker: tracker)
146 case .or(let a, let b):
147 return a.matches(issue, tracker: tracker) || b.matches(issue, tracker: tracker)
148 }
149 }
150}

152final class QueryParser {
153 let tokens: [String]
154 let conditions: [Condition?]
155 var position = 0
156 init(tokens: [String], conditions: [Condition?]) {
157 self.tokens = tokens
158 self.conditions = conditions
159 }
160
161 func accept(_ token: String) -> Bool {
162 if position < tokens.count && tokens[position] == token {
163 position += 1
164 return true
165 }
166 return false
167 }
168
169 func expression() -> QueryNode {
170 var node = term()
171 while accept("OR") {
172 node = .or(node, term())
173 }
174 return node
175 }
176
177 func term() -> QueryNode {
178 var node = factor()
179 while accept("AND") {
180 node = .and(node, factor())
181 }
182 return node
183 }
184
185 func factor() -> QueryNode {
186 if accept("NOT") {
187 return .not(factor())
188 }
189 if accept("(") {
190 let node = expression()
191 _ = accept(")")
192 return node
193 }
194 let node = QueryNode.condition(conditions[position]!)
195 position += 1
196 return node
197 }
198}

200func queryTokens(_ query: String) -> [String] {
201 var tokens: [String] = []
202 var current = ""
203 for character in query {
204 if character == " " || character == "(" || character == ")" {
205 if !current.isEmpty {
206 tokens.append(current)
207 current = ""
208 }
209 if character != " " {
210 tokens.append(String(character))
211 }
212 } else {
213 current.append(character)
214 }
215 }
216 if !current.isEmpty {
217 tokens.append(current)
218 }
219 return tokens
220}

284 func notifications(_ before: [Int: Bool]) -> [String] {
285 before.keys.sorted().compactMap { number in

287 return after == before[number]! ? nil : "\(after ? "BLOCKED" : "UNBLOCKED") \(number)"
288 }

290 func changeState(_ issue: Issue, to target: String) -> [String] {
291 let before = Dictionary(uniqueKeysWithValues: issue.blocks.map { ($0, blocked(issues[$0]!)) })
292 let line = "MOVE \(issue.number) \(issue.state) \(target)"
293 record(issue, "MOVE \(issue.state) \(target)")
294 issue.state = target
295 return [line] + notifications(before)
296 }
297 func reaches(_ start: Int, _ goal: Int) -> Bool {
298 var pending = [start]
299 var visited: Set<Int> = []
300 while let number = pending.popLast() {
301 if number == goal {
302 return true
303 }
304 if visited.insert(number).inserted {
305 pending.append(contentsOf: issues[number]!.blocks)
306 }
307 }
308 return false
309 }

311 func fieldValue(_ issue: Issue, field: String) -> QueryValue {
312 switch field {
313 case "id":
314 return .number(issue.number)
315 case "type":
316 return .text(issue.kind)
317 case "status":
318 return .text(issue.state)
319 case "priority":
320 return .number(issue.level)
321 case "assignee":
322 return issue.assignee.map(QueryValue.text) ?? .none
323 case "reviewer":
324 return issue.reviewer.map(QueryValue.text) ?? .none
325 case "estimate":
326 return issue.estimate.map(QueryValue.number) ?? .none
327 case "sprint":
328 return issue.sprint.map(QueryValue.text) ?? .none
329 case "comments":
330 return .number(issue.comments)
331 case "blocked":
332 return .text(yesNo(blocked(issue)))
333 case "done":
334 return .text(yesNo(issue.done))
335 default:
336 return .text(issue.title.lowercased())
337 }

340 func condition(_ token: String) throws -> Condition {
341 let bytes = Array(token.utf8)
342 var end = 0
343 while end < bytes.count && (97...122).contains(bytes[end]) {
344 end += 1
345 }
346 let field = String(decoding: bytes[..<end], as: UTF8.self)
347 let rest = String(decoding: bytes[end...], as: UTF8.self)
348 guard !field.isEmpty,
349 let operation = [">=", "<=", "!=", "=", "<", ">", "~"].first(where: { rest.hasPrefix($0) })
350 else {
351 throw CommandError(reason: "syntax near " + token)
352 }
353 let text = String(rest.dropFirst(operation.count))
354 guard !text.isEmpty else {
355 throw CommandError(reason: "syntax near " + token)
356 }
357 guard let allowed = fieldOperators[field] else {
358 throw CommandError(reason: "unknown field " + field)
359 }
360 guard allowed.contains(operation) else {
361 throw CommandError(reason: "bad operator " + token)
362 }
363 let value: QueryValue
364 if text == "-" && ["assignee", "reviewer", "estimate", "sprint"].contains(field) && equalityOperators
365 .contains(operation)
366 {
367 value = .none
368 } else if ["id", "estimate", "comments"].contains(field), (1...6).contains(text.count),
369 text.utf8.allSatisfy({ (48...57).contains($0) })
370 {
371 value = .number(Int(text)!)
372 } else if field == "priority", let level = priorities.firstIndex(of: text) {
373 value = .number(level)
374 } else if field == "type" && issueTypes.contains(text) {
375 value = .text(text)
376 } else if field == "status" && allStates.contains(text) {
377 value = .text(text)
378 } else if ["assignee", "reviewer"].contains(field) && users.contains(text) {
379 value = .text(text)
380 } else if field == "sprint" && sprints[text] != nil {
381 value = .text(text)
382 } else if ["blocked", "done"].contains(field) && ["yes", "no"].contains(text) {
383 value = .text(text)
384 } else if field == "title" {
385 value = .text(text.lowercased())
386 } else {
387 throw CommandError(reason: "bad value " + token)
388 }
389 return Condition(field: field, operation: operation, value: value)
390 }

392 func query(_ text: String) throws -> [String] {
393 let tokens = queryTokens(text)
394 var conditions: [Condition?] = []
395 var expectingOperand = true
396 var depth = 0
397 for token in tokens {
398 var parsed: Condition?
399 if expectingOperand {
400 if token == "(" {
401 depth += 1
402 } else if ["AND", "OR", ")"].contains(token) {
403 throw CommandError(reason: "syntax near " + token)
404 } else if token != "NOT" {
405 parsed = try condition(token)
406 expectingOperand = false
407 }
408 } else if token == "AND" || token == "OR" {
409 expectingOperand = true
410 } else if token == ")" && depth > 0 {
411 depth -= 1
412 } else {
413 throw CommandError(reason: "syntax near " + token)
414 }
415 conditions.append(parsed)
416 }
417 guard !expectingOperand && depth == 0 else {
418 throw CommandError(reason: "syntax at end")
419 }
420 let tree = QueryParser(tokens: tokens, conditions: conditions).expression()
421 let matches = issues.values.filter { tree.matches($0, tracker: self) }.sorted {
422 $0.level != $1.level ? $0.level > $1.level : $0.number < $1.number
423 }
424 return ["QUERY \(matches.count)"] + matches.map {
425 "\($0.number) \($0.kind) \($0.priority) \($0.state) \($0.assignee ?? "-") \($0.estimateText) \($0.title)"
426 }
427 }

429 func sprintCommand(_ words: [String]) throws -> [String] {
430 let action = words[1]
431 let name = words[2]
432 if action == "NEW" {
433 guard sprints[name] == nil else {
434 throw CommandError(reason: "sprint exists")
435 }
436 let sprint = Sprint(name)
437 sprints[name] = sprint
438 sprintOrder.append(sprint)
439 return ["SPRINT NEW " + name]
440 }
441 let sprint = try requireSprint(name)
442 if action == "ADD" || action == "REMOVE" {
443 let issue = try requireIssue(words[3])
444 guard sprint.phase != "ENDED" else {
445 throw CommandError(reason: "sprint ended")
446 }
447 if action == "ADD" {
448 try requireNotDone(issue)
449 if let old = issue.sprint,
450 sprints[old]!.phase != "ENDED"
451 {
452 throw CommandError(reason: "in sprint " + old)
453 }
454 guard issue.state != "BACKLOG" else {
455 throw CommandError(reason: "not ready")
456 }
457 issue.sprint = name
458 } else {
459 guard issue.sprint == name else {
460 throw CommandError(reason: "not in sprint")
461 }
462 try requireNotDone(issue)
463 issue.sprint = nil
464 }
465 record(issue, "SPRINT \(action) \(name)")
466 return ["SPRINT \(action) \(name) \(issue.number) \(summary(sprint))"]
467 }
468 if action == "START" {
469 guard sprint.phase == "PLANNED" else {
470 throw CommandError(reason: "sprint not planned")
471 }
472 if let active = activeSprint {
473 throw CommandError(reason: "sprint active " + active.name)
474 }
475 guard !members(sprint).isEmpty else {
476 throw CommandError(reason: "sprint empty")
477 }
478 sprint.phase = "ACTIVE"
479 sprint.start = day
480 let points = remaining(sprint)
481 sprint.records[day] = points
482 return ["SPRINT START \(name) day \(day) \(summary(sprint)) remaining \(points)"]
483 }
484 guard sprint.phase == "ACTIVE" else {
485 throw CommandError(reason: "sprint not active")
486 }
487 sprint.records[day] = remaining(sprint)
488 let members = members(sprint)
489 let finished = members.filter { $0.done }
490 let returned = members.filter { !$0.done }
491 sprint.velocity = finished.reduce(0) { $0 + $1.points }
492 sprint.phase = "ENDED"
493 sprint.end = day
494 var lines =
495 [
496 "SPRINT END \(name) day \(day) velocity \(sprint.velocity) done \(finished.count) returned \(returned.count)"
497 ]
498 for issue in returned {
499 issue.sprint = nil
500 record(issue, "SPRINT RETURN " + name)
501 lines.append("RETURNED \(issue.number)")
502 }
503 return lines
504 }

506 func command(_ line: String) throws -> [String] {
507 let words = line.components(separatedBy: " ")
508 switch words[0] {
509 case "DAY":
510 let target = Int(words[1])!
511 guard target >= day else {
512 throw CommandError(reason: "day backwards")
513 }
514 day = target
515 return ["DAY \(day)"]
516 case "USER":
517 guard users.insert(words[1]).inserted else {
518 throw CommandError(reason: "user exists")
519 }
520 return ["USER " + words[1]]
521 case "CREATE":
522 let number = Int(words[1])!
523 let kind = words[2]
524 let priority = words[3]
525 guard issues[number] == nil else {
526 throw CommandError(reason: "issue exists")
527 }
528 guard issueTypes.contains(kind) else {
529 throw CommandError(reason: "bad type")
530 }
531 guard priorities.contains(priority) else {
532 throw CommandError(reason: "bad priority")
533 }
534 let issue = Issue(
535 number: number,
536 kind: kind,
537 priority: priority,
538 title: words.dropFirst(4).joined(separator: " ")
539 )
540 issues[number] = issue
541 record(issue, "CREATE \(kind) \(priority)")
542 return ["CREATE \(number) \(kind) \(priority) \(issue.state)"]
543 case "PRIORITY":
544 let issue = try requireIssue(words[1])
545 let priority = words[2]
546 guard priorities.contains(priority) else {
547 throw CommandError(reason: "bad priority")
548 }
549 try requireNotDone(issue)
550 guard priority != issue.priority else {
551 throw CommandError(reason: "no change")
552 }
553 let old = issue.priority
554 issue.priority = priority
555 record(issue, "PRIORITY \(old) \(priority)")
556 return ["PRIORITY \(issue.number) \(old) \(priority)"]
557 case "ASSIGN", "REVIEWER":
558 let issue = try requireIssue(words[1])
559 let user = try requireUser(words[2])
560 try requireNotDone(issue)
561 if words[0] == "ASSIGN" {
562 guard user != issue.assignee else {
563 throw CommandError(reason: "no change")
564 }
565 guard user != nil || !["IN_PROGRESS", "REVIEW"].contains(issue.state)
566 else {
567 throw CommandError(reason: "assignee required")
568 }
569 if issue
570 .state == "IN_PROGRESS" && wip(user) >= 3
571 {
572 throw CommandError(reason: "wip limit " + words[2])
573 }
574 if issue.state == "REVIEW" && user == issue
575 .reviewer
576 {
577 throw CommandError(reason: "reviewer is assignee")
578 }
579 issue.assignee = user
580 } else {
581 guard user != issue.reviewer else {
582 throw CommandError(reason: "no change")
583 }
584 if issue.state == "REVIEW" && user == nil {
585 throw CommandError(reason: "reviewer required")
586 }
587 if issue.state == "REVIEW" && user == issue
588 .assignee
589 {
590 throw CommandError(reason: "reviewer is assignee")
591 }
592 issue.reviewer = user
593 }
594 record(issue, "\(words[0]) \(words[2])")
595 return ["\(words[0]) \(issue.number) \(words[2])"]
596 case "ESTIMATE":
597 let issue = try requireIssue(words[1])
598 let points = Int(words[2])!
599 guard estimates.contains(points) else {
600 throw CommandError(reason: "bad estimate")
601 }
602 try requireNotDone(issue)
603 guard issue.estimate != points else {
604 throw CommandError(reason: "no change")
605 }
606 let old = issue.estimateText
607 issue.estimate = points
608 record(issue, "ESTIMATE \(old) \(points)")
609 return ["ESTIMATE \(issue.number) \(old) \(points)"]
610 case "COMMENT":
611 let issue = try requireIssue(words[1])
612 _ = try requireUser(words[2])
613 issue.comments += 1
614 record(issue, "COMMENT \(words[2]) " + words.dropFirst(3).joined(separator: " "))
615 return ["COMMENT \(issue.number) \(issue.comments)"]
616 case "MOVE":
617 let issue = try requireIssue(words[1])
618 let target = words[2]
619 if let duplicate = issue.duplicate {
620 throw CommandError(reason: "duplicate of \(duplicate)")
621 }
622 guard workflows[issue.kind]!.contains(target) else {
623 throw CommandError(reason: "bad state")
624 }
625 guard transitions[issue.kind]!.contains(issue.state + ">" + target)
626 else {
627 throw CommandError(reason: "cannot move \(issue.state) to \(target)")
628 }
629 if target == "IN_PROGRESS" {
630 if issue
631 .kind == "STORY" &&
632 (activeSprint == nil || issue.sprint != activeSprint!.name)
633 {
634 throw CommandError(reason: "not in active sprint")
635 }
636 guard let user = issue.assignee else {
637 throw CommandError(reason: "no assignee")
638 }
639 if wip(user) >= 3 {
640 throw CommandError(reason: "wip limit " + user)
641 }
642 }
643 if target == "REVIEW" {
644 guard issue.reviewer != nil else {
645 throw CommandError(reason: "no reviewer")
646 }
647 guard issue.reviewer != issue.assignee else {
648 throw CommandError(reason: "reviewer is assignee")
649 }
650 }
651 if ["DONE", "RESOLVED"].contains(target),
652 let blocker = unfinishedBlockers(issue).first
653 {
654 throw CommandError(reason: "blocked by \(blocker)")
655 }
656 if target == "READY" && issue.estimate == nil {
657 throw CommandError(reason: "no estimate")
658 }
659 return changeState(issue, to: target)
660 case "LINK", "UNLINK":
661 let relation = words[2]
662 let unlink = words[0] == "UNLINK"
663 guard relation == "BLOCKS" || (!unlink && relation == "DUPLICATES")
664 else {
665 throw CommandError(reason: "bad link")
666 }
667 let a = try requireIssue(words[1])
668 let b = try requireIssue(words[3])
669 if unlink {
670 guard a.blocks.contains(b.number) else {
671 throw CommandError(reason: "no link")
672 }
673 } else {
674 guard a.number != b.number else {
675 throw CommandError(reason: "self link")
676 }
677 if relation == "BLOCKS" {
678 guard !a.blocks.contains(b.number) else {
679 throw CommandError(reason: "link exists")
680 }
681 guard !reaches(b.number, a.number) else {
682 throw CommandError(reason: "cycle")
683 }
684 try requireNotDone(b)
685 } else {
686 guard a.kind == b.kind else {
687 throw CommandError(reason: "type mismatch")
688 }
689 try requireNotDone(a)
690 guard b.duplicate == nil else {
691 throw CommandError(reason: "target is duplicate")
692 }
693 a.duplicate = b.number
694 b.duplicates.insert(a.number)
695 record(a, "LINK DUPLICATES \(b.number)")
696 record(b, "LINK DUPLICATED_BY \(a.number)")
697 var lines = ["LINK \(a.number) DUPLICATES \(b.number)"]
698 lines += changeState(a, to: a.kind == "BUG" ? "CLOSED" : "DONE")
699 if let sprint = a.sprint, sprints[sprint]!.phase != "ENDED" {
700 a.sprint = nil
701 record(a, "SPRINT REMOVE " + sprint)
702 }
703 return lines
704 }
705 }
706 let before = [b.number: blocked(b)]
707 if unlink {
708 a.blocks.remove(b.number)
709 b.blockers.remove(a.number)
710 } else {
711 a.blocks.insert(b.number)
712 b.blockers.insert(a.number)
713 }
714 record(a, "\(words[0]) BLOCKS \(b.number)")
715 record(b, "\(words[0]) BLOCKED_BY \(a.number)")
716 return ["\(words[0]) \(a.number) BLOCKS \(b.number)"] + notifications(before)
717 case "SPRINT":
718 return try sprintCommand(words)
719 case "QUERY":
720 return try query(words.dropFirst().joined(separator: " "))
721 case "SHOW":
722 let issue = try requireIssue(words[1])
723 return [
724 "ISSUE \(issue.number) \(issue.kind) \(issue.priority) \(issue.state) blocked \(yesNo(blocked(issue)))",
725 "TITLE " + issue.title,
726 "PEOPLE assignee \(issue.assignee ?? "-") reviewer \(issue.reviewer ?? "-")",
727 "PLAN estimate \(issue.estimateText) sprint \(issue.sprint ?? "-") comments \(issue.comments)",
728 "LINKS blocks \(numberList(issue.blocks)) blocked_by \(numberList(issue.blockers)) duplicate_of \(issue.duplicate.map(String.init) ?? "-") duplicated_by \(numberList(issue.duplicates))"
729 ]
730 case "HISTORY":
731 let issue = try requireIssue(words[1])
732 return ["HISTORY \(issue.number) \(issue.history.count)"] + issue.history
733 case "BOARD":
734 var lines = ["BOARD"]
735 for kind in issueTypes {
736 lines

738 " \(state) \(issues.values.filter { $0.kind == kind && $0.state == state }.count)"
739 }.joined())

741 let done = issues.values.filter { $0.done }.count
742 lines.append("TOTAL \(issues.count) open \(issues.count - done) done \(done)")
743 return lines
744 case "WORKLOAD":
745 var lines = ["WORKLOAD \(users.count)"]
746 for user in users.sorted() {
747 let assigned = issues.values.filter { !$0.done && $0.assignee == user }
748 let progress = assigned.filter { $0.state == "IN_PROGRESS" }.count
749 let review = assigned.filter { $0.state == "REVIEW" }.count

753 "\(user) assigned \(assigned.count) in_progress \(progress) review \(review) reviewing \(reviewing) points \(assigned.reduce(0) { $0 + $1.points })"

755 }
756 return lines
757 case "SPRINTS":
758 return ["SPRINTS \(sprintOrder.count)"] + sprintOrder.map { sprint in
759 if sprint.phase == "PLANNED" {
760 return "\(sprint.name) PLANNED \(summary(sprint))"
761 }
762 if sprint
763 .phase ==
764 "ACTIVE"

766 return "\(sprint.name) ACTIVE start \(sprint.start!) \(summary(sprint)) remaining \(remaining(sprint))"
767 }
768 return "\(sprint.name) ENDED start \(sprint.start!) end \(sprint.end!) velocity \(sprint.velocity)"
769 }
770 case "BURNDOWN":
771 let sprint = try requireSprint(words[1])
772 guard sprint.phase != "PLANNED" else {
773 throw CommandError(reason: "sprint not started")
774 }
775 let start = sprint.start!
776 let last = sprint.phase == "ACTIVE" ? day : sprint.end!
777 var lines = ["BURNDOWN \(sprint.name) \(sprint.phase) days \(last - start + 1)"]
778 var points = 0
779 for date in start...last {
780 if let record = sprint.records[date] {
781 points = record
782 }
783 lines.append("day \(date) \(points)")
784 }
785 return lines
786 default:
787 return []
788 }
