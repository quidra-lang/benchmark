197Condition | error condition(const Tracker tracker, string token)
198 end := 0
199 while end < int(#token) and "abcdefghijklmnopqrstuvwxyz".contains(token[end])
200 end += 1
201 field := token[0:<end]
202 rest := token[end:]
203 operation := ""
204 for candidate := [">=", "<=", "!=", "=", "<", ">", "~"][]
205 if rest.starts_with(candidate)
206 operation = candidate
207 break
208 value := rest[#operation:]
209 if field == "" or operation == "" or value == ""
210 return error("syntax near " + token)
211 if not one_of("id type status priority assignee reviewer estimate sprint comments blocked done title", field)
212 return error("unknown field " + field)
213 equality := operation == "=" or operation == "!="
214 ordered := one_of("= != < <= > >=", operation)
215 if (field == "title" and operation != "~") or (one_of("id priority estimate comments", field) and not ordered) or (not one_of("title id priority estimate comments", field) and not equality)
216 return error("bad operator " + token)
217 result := Condition(field, operation, value)
218 if value == "-" and one_of("assignee reviewer estimate sprint", field) and equality
219 return result
220 if one_of("id estimate comments", field) and numeric_text(value)
221 result.number = int(value)
222 return result
223 if field == "priority" and priority_level(value) >= 0
224 result.number = priority_level(value)
225 return result
226 if (field == "type" and one_of("BUG TASK STORY", value)) or (field == "status" and one_of("OPEN TRIAGED IN_PROGRESS REVIEW RESOLVED CLOSED REOPENED DONE BACKLOG READY", value)) or (one_of("assignee reviewer", field) and tracker.users.has(value)) or (field == "sprint" and tracker.sprints.has(value)) or (one_of("blocked done", field) and one_of("yes no", value))
227 return result
228 if field == "title"
229 result.value = lowercase(value)
230 return result
231 return error("bad value " + token)

250map<int, Condition> | error check_query(const Tracker tracker, const string[] tokens)
251 map<int, Condition> result
252 operand := true
253 depth := 0
254 for: index < #tokens
255 token := tokens[index]
256 if operand
257 if token == "("
258 depth += 1
259 elif one_of("AND OR )", token)
260 return error("syntax near " + token)
261 elif token != "NOT"
262 parsed := try condition(tracker, token)
263 .result.set(int(index), parsed)
264 operand = false
265 elif token == "AND" or token == "OR"
266 operand = true
267 elif token == ")" and depth > 0
268 depth -= 1
269 else
270 return error("syntax near " + token)
271 if operand or depth != 0
272 return error("syntax at end")
