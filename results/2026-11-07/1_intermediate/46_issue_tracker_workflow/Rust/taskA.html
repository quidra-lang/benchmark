4const TYPES: &[&str] = &["BUG", "TASK", "STORY"];
5const PRIORITIES: &[&str] = &["LOW", "MEDIUM", "HIGH", "CRITICAL"];
6const ALL_STATES: &[&str] = &["OPEN", "TRIAGED", "IN_PROGRESS", "REVIEW", "RESOLVED", "CLOSED", "REOPENED", "DONE", "BACKLOG", "READY"];
7const EQUALITY: &[&str] = &["=", "!="];
8const ORDERED: &[&str] = &["=", "!=", "<", "<=", ">", ">="];

10fn workflow(kind: &str) -> &'static [&'static str] {
11 match kind {
12 "BUG" => &["OPEN", "TRIAGED", "IN_PROGRESS", "REVIEW", "RESOLVED", "CLOSED", "REOPENED"],
13 "TASK" => &["OPEN", "IN_PROGRESS", "DONE"],
14 "STORY" => &["BACKLOG", "READY", "IN_PROGRESS", "REVIEW", "DONE"],
15 _ => unreachable!(),
16 }
17}
18
19fn can_move(kind: &str, from: &str, to: &str) -> bool {
20 let edges: &[&str] = match kind {
21 "BUG" => &["OPEN>TRIAGED", "TRIAGED>IN_PROGRESS", "IN_PROGRESS>REVIEW", "REVIEW>IN_PROGRESS", "REVIEW>RESOLVED", "RESOLVED>CLOSED", "RESOLVED>REOPENED", "CLOSED>REOPENED", "REOPENED>IN_PROGRESS"],
22 "TASK" => &["OPEN>IN_PROGRESS", "IN_PROGRESS>OPEN", "IN_PROGRESS>DONE"],
23 "STORY" => &["BACKLOG>READY", "READY>BACKLOG", "READY>IN_PROGRESS", "IN_PROGRESS>REVIEW", "REVIEW>IN_PROGRESS", "REVIEW>DONE"],
24 _ => unreachable!(),
25 };
26 edges.contains(&format!("{from}>{to}").as_str())
27}

29fn yes_no(value: bool) -> &'static str { if value { "yes" } else { "no" } }
30fn display_number(value: Option<i32>) -> String { value.map(|n| n.to_string()).unwrap_or_else(|| "-".to_owned()) }
31fn number_list(numbers: &BTreeSet<i32>) -> String {
32 if numbers.is_empty() { "-".to_owned() } else { numbers.iter().map(i32::to_string).collect::<Vec<_>>().join(",") }
33}

82#[derive(Clone, PartialEq, Eq)]
83enum QueryValue {
84 None,
85 Number(i32),
86 Text(String),
87}
88
89#[derive(Clone)]
90struct Condition {
91 field: String,
92 operation: String,
93 value: QueryValue,
94}
95
96impl Condition {
97 fn matches(&self, tracker: &Tracker, issue: &Issue) -> bool {
98 let actual = tracker.field_value(issue, &self.field);
99 match self.operation.as_str() {
100 "=" => actual == self.value,
101 "!=" => actual != self.value,
102 "~" => match (&actual, &self.value) { (QueryValue::Text(a), QueryValue::Text(b)) => a.contains(b.as_str()), _ => false },
103 operation => match (&actual, &self.value) {
104 (QueryValue::Number(a), QueryValue::Number(b)) => match operation {
105 "<" => a < b, "<=" => a <= b, ">" => a > b, _ => a >= b,
106 },
107 _ => false,
108 },
109 }
110 }
111}
112
113enum QueryNode {
114 Condition(Condition),
115 Not(Box<QueryNode>),
116 And(Box<QueryNode>, Box<QueryNode>),
117 Or(Box<QueryNode>, Box<QueryNode>),
118}
119
120impl QueryNode {
121 fn matches(&self, tracker: &Tracker, issue: &Issue) -> bool {
122 match self {
123 Self::Condition(condition) => condition.matches(tracker, issue),
124 Self::Not(node) => !node.matches(tracker, issue),
125 Self::And(a, b) => a.matches(tracker, issue) && b.matches(tracker, issue),
126 Self::Or(a, b) => a.matches(tracker, issue) || b.matches(tracker, issue),
127 }
128 }
129}
130
131struct QueryParser<'a> {
132 tokens: &'a [String],
133 conditions: Vec<Option<Condition>>,
134 position: usize,
135}
136
137impl QueryParser<'_> {
138 fn accept(&mut self, token: &str) -> bool {
139 if self.tokens.get(self.position).is_some_and(|t| t == token) {
140 self.position += 1;
141 true
142 } else { false }
143 }
144 fn expression(&mut self) -> QueryNode {
145 let mut node = self.term();
146 while self.accept("OR") { node = QueryNode::Or(Box::new(node), Box::new(self.term())); }
147 node
148 }
149 fn term(&mut self) -> QueryNode {
150 let mut node = self.factor();
151 while self.accept("AND") { node = QueryNode::And(Box::new(node), Box::new(self.factor())); }
152 node
153 }
154 fn factor(&mut self) -> QueryNode {
155 if self.accept("NOT") { return QueryNode::Not(Box::new(self.factor())); }
156 if self.accept("(") {
157 let node = self.expression();
158 self.accept(")");
159 return node;
160 }
161 let condition = self.conditions[self.position].take().unwrap();
162 self.position += 1;
163 QueryNode::Condition(condition)
164 }
165}

167fn query_tokens(text: &str) -> Vec<String> {
168 let mut tokens = Vec::new();
169 let mut current = String::new();
170 for character in text.chars() {
171 if matches!(character, ' ' | '(' | ')') {
172 if !current.is_empty() { tokens.push(std::mem::take(&mut current)); }
173 if character != ' ' { tokens.push(character.to_string()); }
174 } else { current.push(character); }
175 }
176 if !current.is_empty() { tokens.push(current); }
177 tokens
178}

180#[derive(Default)]
181struct Tracker {
182 day: i32,
183 users: BTreeSet<String>,
184 issues: BTreeMap<i32, Issue>,
185 sprints: Vec<Sprint>,
186 sprint_indices: BTreeMap<String, usize>,
187}
188
189impl Tracker {
190 fn require_issue(&self, text: &str) -> Result<i32, String> {
191 let number = text.parse().unwrap();
192 if self.issues.contains_key(&number) { Ok(number) } else { Err("unknown issue".to_owned()) }
193 }
194 fn require_user(&self, text: &str) -> Result<Option<String>, String> {
195 if text == "-" { Ok(None) }
196 else if self.users.contains(text) { Ok(Some(text.to_owned())) }
197 else { Err("unknown user".to_owned()) }
198 }
199 fn require_sprint(&self, name: &str) -> Result<usize, String> {
200 self.sprint_indices.get(name).copied().ok_or_else(|| "unknown sprint".to_owned())
201 }
202 fn require_not_done(&self, number: i32) -> Result<(), String> {
203 if self.issues[&number].done() { Err("issue done".to_owned()) } else { Ok(()) }
204 }
205 fn record(&mut self, number: i32, content: String) {
206 self.issues.get_mut(&number).unwrap().history.push(format!("day {} {content}", self.day));
207 }
208 fn unfinished_blockers(&self, number: i32) -> Vec<i32> {
209 self.issues[&number].blockers.iter().copied().filter(|n| !self.issues[n].done()).collect()
210 }
211 fn blocked(&self, number: i32) -> bool { !self.issues[&number].done() && !self.unfinished_blockers(number).is_empty() }
212 fn wip(&self, user: Option<&str>) -> usize {
213 self.issues.values().filter(|i| i.assignee.as_deref() == user && i.state == "IN_PROGRESS").count()
214 }
215 fn active_sprint(&self) -> Option<usize> { self.sprints.iter().position(|s| s.phase == "ACTIVE") }
216 fn members(&self, index: usize) -> Vec<i32> {
217 self.issues.values().filter(|i| i.sprint.as_deref() == Some(self.sprints[index].name.as_str())).map(|i| i.number).collect()
218 }
219 fn remaining(&self, index: usize) -> i32 {
220 self.members(index).iter().filter(|n| !self.issues[*n].done()).map(|n| self.issues[n].points()).sum()
221 }
222 fn summary(&self, index: usize) -> String {
223 let members = self.members(index);
224 let points: i32 = members.iter().map(|n| self.issues[n].points()).sum();
225 format!("issues {} points {points}", members.len())
226 }
227 fn notify(&self, before: &BTreeMap<i32, bool>) -> Vec<String> {
228 before.iter().filter_map(|(&number, &old)| {
229 let after = self.blocked(number);
230 (after != old).then(|| format!("{} {number}", if after { "BLOCKED" } else { "UNBLOCKED" }))
231 }).collect()
232 }
233 fn change_state(&mut self, number: i32, target: &str) -> Vec<String> {
234 let before = self.issues[&number].blocks.iter().map(|&n| (n, self.blocked(n))).collect();
235 let old = self.issues[&number].state.clone();
236 self.record(number, format!("MOVE {old} {target}"));
237 self.issues.get_mut(&number).unwrap().state = target.to_owned();
238 let mut lines = vec![format!("MOVE {number} {old} {target}")];
239 lines.extend(self.notify(&before));
240 lines
241 }
242 fn reaches(&self, start: i32, goal: i32) -> bool {
243 let mut pending = vec![start];
244 let mut seen = BTreeSet::new();
245 while let Some(number) = pending.pop() {
246 if number == goal { return true; }
247 if seen.insert(number) { pending.extend(self.issues[&number].blocks.iter().copied()); }
248 }
249 false
250 }

252 fn field_value(&self, issue: &Issue, field: &str) -> QueryValue {
253 match field {
254 "id" => QueryValue::Number(issue.number),
255 "type" => QueryValue::Text(issue.kind.clone()),
256 "status" => QueryValue::Text(issue.state.clone()),
257 "priority" => QueryValue::Number(issue.level()),
258 "assignee" => issue.assignee.clone().map(QueryValue::Text).unwrap_or(QueryValue::None),
259 "reviewer" => issue.reviewer.clone().map(QueryValue::Text).unwrap_or(QueryValue::None),
260 "estimate" => issue.estimate.map(QueryValue::Number).unwrap_or(QueryValue::None),
261 "sprint" => issue.sprint.clone().map(QueryValue::Text).unwrap_or(QueryValue::None),
262 "comments" => QueryValue::Number(issue.comments),
263 "blocked" => QueryValue::Text(yes_no(self.blocked(issue.number)).to_owned()),
264 "done" => QueryValue::Text(yes_no(issue.done()).to_owned()),
265 _ => QueryValue::Text(issue.title.to_ascii_lowercase()),
266 }
267 }

269 fn condition(&self, token: &str) -> Result<Condition, String> {
270 let length = token.bytes().take_while(|b| b.is_ascii_lowercase()).count();
271 let field = &token[..length];
272 let rest = &token[length..];
273 let operation = [">=", "<=", "!=", "=", "<", ">", "~"].iter().copied().find(|op| rest.starts_with(*op));
274 let Some(operation) = operation else { return Err(format!("syntax near {token}")); };
275 let text = &rest[operation.len()..];
276 if field.is_empty() || text.is_empty() { return Err(format!("syntax near {token}")); }
277 let allowed = match field {
278 "id" | "priority" | "estimate" | "comments" => ORDERED,
279 "type" | "status" | "assignee" | "reviewer" | "sprint" | "blocked" | "done" => EQUALITY,
280 "title" => &["~"],
281 _ => return Err(format!("unknown field {field}")),
282 };
283 if !allowed.contains(&operation) { return Err(format!("bad operator {token}")); }
284 let value = if text == "-" && matches!(field, "assignee" | "reviewer" | "estimate" | "sprint") && EQUALITY.contains(&operation) {
285 QueryValue::None
286 } else if matches!(field, "id" | "estimate" | "comments") && (1..=6).contains(&text.len()) && text.bytes().all(|b| b.is_ascii_digit()) {
287 QueryValue::Number(text.parse().unwrap())
288 } else if field == "priority" && PRIORITIES.contains(&text) {
289 QueryValue::Number(PRIORITIES.iter().position(|&p| p == text).unwrap() as i32)
290 } else if (field == "type" && TYPES.contains(&text)) || (field == "status" && ALL_STATES.contains(&text))
291 || (matches!(field, "assignee" | "reviewer") && self.users.contains(text))
292 || (field == "sprint" && self.sprint_indices.contains_key(text))
293 || (matches!(field, "blocked" | "done") && matches!(text, "yes" | "no")) {
294 QueryValue::Text(text.to_owned())
295 } else if field == "title" {
296 QueryValue::Text(text.to_ascii_lowercase())
297 } else { return Err(format!("bad value {token}")); };
298 Ok(Condition { field: field.to_owned(), operation: operation.to_owned(), value })
299 }

301 fn query(&self, text: &str) -> Result<Vec<String>, String> {
302 let tokens = query_tokens(text);
303 let mut conditions = Vec::new();
304 let mut expecting_operand = true;
305 let mut depth = 0;
306 for token in &tokens {
307 let mut condition = None;
308 if expecting_operand {
309 match token.as_str() {
310 "(" => depth += 1,
311 "AND" | "OR" | ")" => return Err(format!("syntax near {token}")),
312 "NOT" => {},
313 _ => { condition = Some(self.condition(token)?); expecting_operand = false; }
314 }
315 } else {
316 match token.as_str() {
317 "AND" | "OR" => expecting_operand = true,
318 ")" if depth > 0 => depth -= 1,
319 _ => return Err(format!("syntax near {token}")),
320 }
321 }
322 conditions.push(condition);
323 }
324 if expecting_operand || depth != 0 { return Err("syntax at end".to_owned()); }
325 let tree = QueryParser { tokens: &tokens, conditions, position: 0 }.expression();
326 let mut matches: Vec<_> = self.issues.values().filter(|i| tree.matches(self, i)).collect();
327 matches.sort_by_key(|i| (-i.level(), i.number));
328 let mut lines = vec![format!("QUERY {}", matches.len())];
329 lines.extend(matches.into_iter().map(|i| format!("{} {} {} {} {} {} {}", i.number, i.kind, i.priority, i.state, i.assignee.as_deref().unwrap_or("-"), display_number(i.estimate), i.title)));
330 Ok(lines)
331 }

333 fn sprint_command(&mut self, words: &[&str]) -> Result<Vec<String>, String> {
334 let (action, name) = (words[1], words[2]);
335 if action == "NEW" {
336 if self.sprint_indices.contains_key(name) { return Err("sprint exists".to_owned()); }
337 self.sprint_indices.insert(name.to_owned(), self.sprints.len());
338 self.sprints.push(Sprint::new(name));
339 return Ok(vec![format!("SPRINT NEW {name}")]);
340 }
341 let index = self.require_sprint(name)?;
342 if matches!(action, "ADD" | "REMOVE") {
343 let number = self.require_issue(words[3])?;
344 if self.sprints[index].phase == "ENDED" { return Err("sprint ended".to_owned()); }
345 if action == "ADD" {
346 self.require_not_done(number)?;
347 let issue = &self.issues[&number];
348 if let Some(old) = &issue.sprint {
349 if self.sprints[self.sprint_indices[old]].phase != "ENDED" { return Err(format!("in sprint {old}")); }
350 }
351 if issue.state == "BACKLOG" { return Err("not ready".to_owned()); }
352 self.issues.get_mut(&number).unwrap().sprint = Some(name.to_owned());
353 } else {
354 if self.issues[&number].sprint.as_deref() != Some(name) { return Err("not in sprint".to_owned()); }
355 self.require_not_done(number)?;
356 self.issues.get_mut(&number).unwrap().sprint = None;
357 }
358 self.record(number, format!("SPRINT {action} {name}"));
359 return Ok(vec![format!("SPRINT {action} {name} {number} {}", self.summary(index))]);
360 }
361 if action == "START" {
362 if self.sprints[index].phase != "PLANNED" { return Err("sprint not planned".to_owned()); }
363 if let Some(active) = self.active_sprint() { return Err(format!("sprint active {}", self.sprints[active].name)); }
364 if self.members(index).is_empty() { return Err("sprint empty".to_owned()); }
365 self.sprints[index].phase = "ACTIVE".to_owned();
366 self.sprints[index].start = Some(self.day);
367 let remaining = self.remaining(index);
368 self.sprints[index].records.insert(self.day, remaining);
369 return Ok(vec![format!("SPRINT START {name} day {} {} remaining {remaining}", self.day, self.summary(index))]);
370 }
371 if self.sprints[index].phase != "ACTIVE" { return Err("sprint not active".to_owned()); }
372 let remaining = self.remaining(index);
373 self.sprints[index].records.insert(self.day, remaining);
374 let (finished, returned): (Vec<_>, Vec<_>) = self.members(index).into_iter().partition(|n| self.issues[n].done());
375 let velocity: i32 = finished.iter().map(|n| self.issues[n].points()).sum();
376 self.sprints[index].velocity = velocity;
377 self.sprints[index].phase = "ENDED".to_owned();
378 self.sprints[index].end = Some(self.day);
379 let mut lines = vec![format!("SPRINT END {name} day {} velocity {velocity} done {} returned {}", self.day, finished.len(), returned.len())];
380 for number in returned {
381 self.issues.get_mut(&number).unwrap().sprint = None;
382 self.record(number, format!("SPRINT RETURN {name}"));
383 lines.push(format!("RETURNED {number}"));
384 }
385 Ok(lines)
386 }

388 fn command(&mut self, line: &str) -> Result<Vec<String>, String> {
389 let words: Vec<_> = line.split(' ').collect();
390 let command = words[0];
391 match command {
392 "DAY" => {
393 let day = words[1].parse().unwrap();
394 if day < self.day { return Err("day backwards".to_owned()); }
395 self.day = day;
396 Ok(vec![format!("DAY {day}")])
397 }
398 "USER" => {
399 if !self.users.insert(words[1].to_owned()) { return Err("user exists".to_owned()); }
400 Ok(vec![format!("USER {}", words[1])])
401 }
402 "CREATE" => {
403 let number = words[1].parse().unwrap();
404 let (kind, priority) = (words[2], words[3]);
405 if self.issues.contains_key(&number) { return Err("issue exists".to_owned()); }
406 if !TYPES.contains(&kind) { return Err("bad type".to_owned()); }
407 if !PRIORITIES.contains(&priority) { return Err("bad priority".to_owned()); }
408 let issue = Issue::new(number, kind, priority, words[4..].join(" "));
409 let state = issue.state.clone();
410 self.issues.insert(number, issue);
411 self.record(number, format!("CREATE {kind} {priority}"));
412 Ok(vec![format!("CREATE {number} {kind} {priority} {state}")])
413 }
414 "PRIORITY" => {
415 let number = self.require_issue(words[1])?;
416 let priority = words[2];
417 if !PRIORITIES.contains(&priority) { return Err("bad priority".to_owned()); }
418 self.require_not_done(number)?;
419 let old = self.issues[&number].priority.clone();
420 if old == priority { return Err("no change".to_owned()); }
421 self.issues.get_mut(&number).unwrap().priority = priority.to_owned();
422 self.record(number, format!("PRIORITY {old} {priority}"));
423 Ok(vec![format!("PRIORITY {number} {old} {priority}")])
424 }
425 "ASSIGN" | "REVIEWER" => {
426 let number = self.require_issue(words[1])?;
427 let user = self.require_user(words[2])?;
428 self.require_not_done(number)?;
429 let issue = &self.issues[&number];
430 if command == "ASSIGN" {
431 if user == issue.assignee { return Err("no change".to_owned()); }
432 if user.is_none() && matches!(issue.state.as_str(), "IN_PROGRESS" | "REVIEW") { return Err("assignee required".to_owned()); }
433 if issue.state == "IN_PROGRESS" && self.wip(user.as_deref()) >= 3 { return Err(format!("wip limit {}", words[2])); }
434 if issue.state == "REVIEW" && user == issue.reviewer { return Err("reviewer is assignee".to_owned()); }
435 self.issues.get_mut(&number).unwrap().assignee = user;
436 } else {
437 if user == issue.reviewer { return Err("no change".to_owned()); }
438 if issue.state == "REVIEW" && user.is_none() { return Err("reviewer required".to_owned()); }
439 if issue.state == "REVIEW" && user == issue.assignee { return Err("reviewer is assignee".to_owned()); }
440 self.issues.get_mut(&number).unwrap().reviewer = user;
441 }
442 self.record(number, format!("{command} {}", words[2]));
443 Ok(vec![format!("{command} {number} {}", words[2])])
444 }
445 "ESTIMATE" => {
446 let number = self.require_issue(words[1])?;
447 let points = words[2].parse().unwrap();
448 if ![0, 1, 2, 3, 5, 8, 13, 21].contains(&points) { return Err("bad estimate".to_owned()); }
449 self.require_not_done(number)?;
450 let old = self.issues[&number].estimate;
451 if old == Some(points) { return Err("no change".to_owned()); }
452 self.issues.get_mut(&number).unwrap().estimate = Some(points);
453 self.record(number, format!("ESTIMATE {} {points}", display_number(old)));
454 Ok(vec![format!("ESTIMATE {number} {} {points}", display_number(old))])
455 }
456 "COMMENT" => {
457 let number = self.require_issue(words[1])?;
458 self.require_user(words[2])?;
459 self.issues.get_mut(&number).unwrap().comments += 1;
460 self.record(number, format!("COMMENT {} {}", words[2], words[3..].join(" ")));
461 Ok(vec![format!("COMMENT {number} {}", self.issues[&number].comments)])
462 }
463 "MOVE" => {
464 let number = self.require_issue(words[1])?;
465 let target = words[2];
466 let issue = &self.issues[&number];
467 if let Some(duplicate) = issue.duplicate { return Err(format!("duplicate of {duplicate}")); }
468 if !workflow(&issue.kind).contains(&target) { return Err("bad state".to_owned()); }
469 if !can_move(&issue.kind, &issue.state, target) { return Err(format!("cannot move {} to {target}", issue.state)); }
470 if target == "IN_PROGRESS" {
471 if issue.kind == "STORY" && self.active_sprint().is_none_or(|i| issue.sprint.as_deref() != Some(self.sprints[i].name.as_str())) {
472 return Err("not in active sprint".to_owned());
473 }
474 let user = issue.assignee.as_deref().ok_or("no assignee")?;
475 if self.wip(Some(user)) >= 3 { return Err(format!("wip limit {user}")); }
476 }
477 if target == "REVIEW" {
478 if issue.reviewer.is_none() { return Err("no reviewer".to_owned()); }
479 if issue.reviewer == issue.assignee { return Err("reviewer is assignee".to_owned()); }
480 }
481 if matches!(target, "DONE" | "RESOLVED") {
482 if let Some(blocker) = self.unfinished_blockers(number).first() { return Err(format!("blocked by {blocker}")); }
483 }
484 if target == "READY" && issue.estimate.is_none() { return Err("no estimate".to_owned()); }
485 Ok(self.change_state(number, target))
486 }
487 "LINK" | "UNLINK" => {
488 let relation = words[2];
489 let unlink = command == "UNLINK";
490 if relation != "BLOCKS" && (unlink || relation != "DUPLICATES") { return Err("bad link".to_owned()); }
491 let a = self.require_issue(words[1])?;
492 let b = self.require_issue(words[3])?;
493 if unlink {
494 if !self.issues[&a].blocks.contains(&b) { return Err("no link".to_owned()); }
495 } else {
496 if a == b { return Err("self link".to_owned()); }
497 if relation == "BLOCKS" {
498 if self.issues[&a].blocks.contains(&b) { return Err("link exists".to_owned()); }
499 if self.reaches(b, a) { return Err("cycle".to_owned()); }
500 self.require_not_done(b)?;
501 } else {
502 if self.issues[&a].kind != self.issues[&b].kind { return Err("type mismatch".to_owned()); }
503 self.require_not_done(a)?;
504 if self.issues[&b].duplicate.is_some() { return Err("target is duplicate".to_owned()); }
505 let target_state = if self.issues[&a].kind == "BUG" { "CLOSED" } else { "DONE" };
506 self.issues.get_mut(&a).unwrap().duplicate = Some(b);
507 self.issues.get_mut(&b).unwrap().duplicates.insert(a);
508 self.record(a, format!("LINK DUPLICATES {b}"));
509 self.record(b, format!("LINK DUPLICATED_BY {a}"));
510 let mut lines = vec![format!("LINK {a} DUPLICATES {b}")];
511 lines.extend(self.change_state(a, target_state));
512 if let Some(old) = self.issues[&a].sprint.clone() {
513 if self.sprints[self.sprint_indices[&old]].phase != "ENDED" {
514 self.issues.get_mut(&a).unwrap().sprint = None;
515 self.record(a, format!("SPRINT REMOVE {old}"));
516 }
517 }
518 return Ok(lines);
519 }
520 }
521 let before = BTreeMap::from([(b, self.blocked(b))]);
522 if unlink {
523 self.issues.get_mut(&a).unwrap().blocks.remove(&b);
524 self.issues.get_mut(&b).unwrap().blockers.remove(&a);
525 } else {
526 self.issues.get_mut(&a).unwrap().blocks.insert(b);
527 self.issues.get_mut(&b).unwrap().blockers.insert(a);
528 }
529 self.record(a, format!("{command} BLOCKS {b}"));
530 self.record(b, format!("{command} BLOCKED_BY {a}"));
531 let mut lines = vec![format!("{command} {a} BLOCKS {b}")];
532 lines.extend(self.notify(&before));
533 Ok(lines)
534 }
535 "SPRINT" => self.sprint_command(&words),
536 "QUERY" => self.query(&words[1..].join(" ")),
537 "SHOW" => {
538 let number = self.require_issue(words[1])?;
539 let issue = &self.issues[&number];
540 Ok(vec![
541 format!("ISSUE {number} {} {} {} blocked {}", issue.kind, issue.priority, issue.state, yes_no(self.blocked(number))),
542 format!("TITLE {}", issue.title),
543 format!("PEOPLE assignee {} reviewer {}", issue.assignee.as_deref().unwrap_or("-"), issue.reviewer.as_deref().unwrap_or("-")),
544 format!("PLAN estimate {} sprint {} comments {}", display_number(issue.estimate), issue.sprint.as_deref().unwrap_or("-"), issue.comments),
545 format!("LINKS blocks {} blocked_by {} duplicate_of {} duplicated_by {}", number_list(&issue.blocks), number_list(&issue.blockers), display_number(issue.duplicate), number_list(&issue.duplicates)),
546 ])
547 }
548 "HISTORY" => {
549 let number = self.require_issue(words[1])?;
550 let issue = &self.issues[&number];
551 let mut lines = vec![format!("HISTORY {number} {}", issue.history.len())];
552 lines.extend(issue.history.iter().cloned());
553 Ok(lines)
554 }
555 "BOARD" => {
556 let mut lines = vec!["BOARD".to_owned()];
557 for &kind in TYPES {
558 let mut line = kind.to_owned();
559 for &state in workflow(kind) {
560 let count = self.issues.values().filter(|i| i.kind == kind && i.state == state).count();
561 line.push_str(&format!(" {state} {count}"));
562 }
563 lines.push(line);
564 }
565 let done = self.issues.values().filter(|i| i.done()).count();
566 lines.push(format!("TOTAL {} open {} done {done}", self.issues.len(), self.issues.len() - done));
567 Ok(lines)
568 }
569 "WORKLOAD" => {
570 let mut lines = vec![format!("WORKLOAD {}", self.users.len())];
571 for user in &self.users {
572 let assigned: Vec<_> = self.issues.values().filter(|i| !i.done() && i.assignee.as_ref() == Some(user)).collect();
573 let progress = assigned.iter().filter(|i| i.state == "IN_PROGRESS").count();
574 let review = assigned.iter().filter(|i| i.state == "REVIEW").count();
575 let reviewing = self.issues.values().filter(|i| i.reviewer.as_ref() == Some(user) && i.state == "REVIEW").count();
576 let points: i32 = assigned.iter().map(|i| i.points()).sum();
577 lines.push(format!("{user} assigned {} in_progress {progress} review {review} reviewing {reviewing} points {points}", assigned.len()));
578 }
579 Ok(lines)
580 }
581 "SPRINTS" => {
582 let mut lines = vec![format!("SPRINTS {}", self.sprints.len())];
583 for (index, sprint) in self.sprints.iter().enumerate() {
584 lines.push(match sprint.phase.as_str() {
585 "PLANNED" => format!("{} PLANNED {}", sprint.name, self.summary(index)),
586 "ACTIVE" => format!("{} ACTIVE start {} {} remaining {}", sprint.name, sprint.start.unwrap(), self.summary(index), self.remaining(index)),
587 _ => format!("{} ENDED start {} end {} velocity {}", sprint.name, sprint.start.unwrap(), sprint.end.unwrap(), sprint.velocity),
588 });
589 }
590 Ok(lines)
591 }
592 "BURNDOWN" => {
593 let index = self.require_sprint(words[1])?;
594 let sprint = &self.sprints[index];
595 if sprint.phase == "PLANNED" { return Err("sprint not started".to_owned()); }
596 let start = sprint.start.unwrap();
597 let last = if sprint.phase == "ACTIVE" { self.day } else { sprint.end.unwrap() };
598 let mut lines = vec![format!("BURNDOWN {} {} days {}", sprint.name, sprint.phase, last - start + 1)];
599 let mut points = 0;
600 for day in start..=last {
601 if let Some(&recorded) = sprint.records.get(&day) { points = recorded; }
602 lines.push(format!("day {day} {points}"));
603 }
604 Ok(lines)
605 }
606 _ => unreachable!(),
607 }
608 }
609
610 fn execute(&mut self, line: &str) -> Vec<String> {
611 let lines = self.command(line).unwrap_or_else(|error| vec![format!("ERROR {error}")]);
612 if let Some(index) = self.active_sprint() {
613 let points = self.remaining(index);
614 self.sprints[index].records.insert(self.day, points);
615 }
616 lines
617 }
618}

620fn main() {
621 let mut input = String::new();
622 io::stdin().read_to_string(&mut input).unwrap();
623 let mut lines = input.lines();
624 let count: usize = lines.next().unwrap().parse().unwrap();
625 let mut tracker = Tracker { day: 1, ..Tracker::default() };
626 let mut output = Vec::new();
627 for line in lines.take(count) { output.extend(tracker.execute(line)); }
628 if !output.is_empty() { println!("{}", output.join("\n")); }
