114 private static final class QueryParser {
115 final List<String> tokens;
116 final List<Predicate<Issue>> conditions;
117 int position;
118
119 QueryParser(List<String> tokens, List<Predicate<Issue>> conditions) {
120 this.tokens = tokens;
121 this.conditions = conditions;
122 }
123
124 boolean accept(String token) {
125 if (position >= tokens.size() || !tokens.get(position).equals(token))
126 return false;
127 position++;
128 return true;
129 }
130
131 Predicate<Issue> expression() {
132 Predicate<Issue> node = term();
133 while (accept("OR"))
134 node = node.or(term());
135 return node;
136 }
137
138 Predicate<Issue> term() {
139 Predicate<Issue> node = factor();
140 while (accept("AND"))
141 node = node.and(factor());
142 return node;
143 }
144
145 Predicate<Issue> factor() {
146 if (accept("NOT"))
147 return factor().negate();
148 if (accept("(")) {
149 Predicate<Issue> node = expression();
150 accept(")");
151 return node;
152 }
153 return conditions.get(position++);
154 }
155 }

249 boolean reaches(int start, int goal) {
250 ArrayDeque<Integer> pending = new ArrayDeque<>();
251 Set<Integer> visited = new HashSet<>();
252 pending.add(start);
253 while (!pending.isEmpty()) {
254 int number = pending.removeFirst();

256 return true;
257 if (visited.add(number))

259 }
260 return false;
261 }

263 boolean compare(Integer actual, Integer expected, String operator) {
264 if (operator.equals("="))
265 return Objects.equals(actual, expected);
266 if (operator.equals("!="))
267 return !Objects.equals(actual, expected);
268 if (actual == null)
269 return false;
270 return switch (operator) {
271 case "<" -> actual < expected;
272 case "<=" -> actual <= expected;
273 case ">" -> actual > expected;
274 default -> actual >= expected;
275 };
276 }
277
278 Predicate<Issue> condition(String token) {
279 int length = 0;
280 while (length < token.length() && token.charAt(length) >= 'a' &&
281 token.charAt(length) <= 'z')
282 length++;
283 String field = token.substring(0, length);
284 String rest = token.substring(length);
285 String operator = null;
286 for (String candidate : List.of(">=", "<=", "!=", "=", "<", ">", "~")) {
287 if (rest.startsWith(candidate)) {
288 operator = candidate;
289 break;
290 }
291 }
292 String text = operator == null ? "" : rest.substring(operator.length());
293 if (field.isEmpty() || operator == null || text.isEmpty())
294 throw new CommandError("syntax near " + token);
295 Set<String> operators = FIELD_OPERATORS.get(field);
296 if (operators == null)
297 throw new CommandError("unknown field " + field);
298 if (!operators.contains(operator))
299 throw new CommandError("bad operator " + token);
300 boolean none = text.equals("-") &&
301 Set.of("assignee", "reviewer", "estimate", "sprint").contains(field) &&
302 EQUALITY.contains(operator);
303 Integer numeric = null;
304 String string = none ? null : text;
305 boolean valid;
306 switch (field) {
307 case "id", "estimate", "comments" -> {
308 valid = none || text.matches("[0-9]{1,6}");
309 if (valid && !none)
310 numeric = Integer.valueOf(text);
311 }
312 case "priority" -> {
313 valid = PRIORITIES.contains(text);
314 if (valid)
315 numeric = PRIORITIES.indexOf(text);
316 }
317 case "type" -> valid = TYPES.contains(text);
318 case "status" ->
319 valid = WORKFLOWS.values().stream().anyMatch(states -> states.contains(text));
320 case "assignee", "reviewer" -> valid = none || users.contains(text);
321 case "sprint" -> valid = none || sprints.containsKey(text);
322 case "blocked", "done" -> valid = text.equals("yes") || text.equals("no");
323 case "title" -> {
324 string = text.toLowerCase(Locale.ROOT);
325 valid = true;
326 }
327 default -> valid = false;
328 }
329 if (!valid)
330 throw new CommandError("bad value " + token);
331 Integer expectedNumber = numeric;
332 String expectedString = string;
333 String comparison = operator;
334 return issue -> {
335 if (Set.of("id", "estimate", "comments", "priority").contains(field)) {
336 Integer actual = switch (field) {
337 case "id" -> issue.number;
338 case "estimate" -> issue.estimate;
339 case "comments" -> issue.comments;
340 default -> issue.level();
341 };
342 return compare(actual, expectedNumber, comparison);
343 }
344 String actual = switch (field) {
345 case "type" -> issue.type;
346 case "status" -> issue.state;
347 case "assignee" -> issue.assignee;
348 case "reviewer" -> issue.reviewer;
349 case "sprint" -> issue.sprint;
350 case "blocked" -> yesNo(blocked(issue));
351 case "done" -> yesNo(issue.done());
352 default -> issue.title.toLowerCase(Locale.ROOT);
353 };
354 return switch (comparison) {
355 case "~" -> actual.contains(expectedString);
356 case "=" -> Objects.equals(actual, expectedString);
357 default -> !Objects.equals(actual, expectedString);
358 };
359 };
360 }

362 void query(String text, List<String> output) {
363 List<String> tokens = new ArrayList<>();
364 Matcher matcher = Pattern.compile("[()]|[^ ()]+").matcher(text);
365 while (matcher.find())
366 tokens.add(matcher.group());
367 List<Predicate<Issue>> conditions = new ArrayList<>();
368 boolean operand = true;
369 int depth = 0;
370 for (String token : tokens) {
371 Predicate<Issue> predicate = null;
372 if (operand) {
373 switch (token) {
374 case "NOT" -> {
375 }
376 case "(" -> depth++;
377 case "AND", "OR", ")" -> throw new CommandError("syntax near " + token);
378 default -> {
379 predicate = condition(token);
380 operand = false;
381 }
382 }
383 } else {
384 switch (token) {
385 case "AND", "OR" -> operand = true;
386 case ")" -> {
387 if (depth == 0)
388 throw new CommandError("syntax near " + token);
389 depth--;
390 }
391 default -> throw new CommandError("syntax near " + token);
392 }
393 }
394 conditions.add(predicate);
395 }
396 if (operand || depth != 0)
397 throw new CommandError("syntax at end");
398 Predicate<Issue> predicate = new QueryParser(tokens, conditions).expression();
399 List<Issue> matches = issues.values()
400 .stream()
401 .filter(predicate)
402 .sorted(Comparator.comparingInt(Issue::level)
403 .reversed()
404 .thenComparingInt(i -> i.number))
405 .toList();
406 output.add("QUERY " + matches.size());
407 for (Issue issue : matches)
408 output.add(issue.number + " " + issue.type + " " + issue.priority + " " +
409 issue.state + " " + display(issue.assignee) + " " +
410 display(issue.estimate) + " " + issue.title);
411 }

413 void sprintCommand(String[] words, List<String> output) {
414 String action = words[1];
415 String name = words[2];
416 if (action.equals("NEW")) {
417 if (sprints.containsKey(name))
418 throw new CommandError("sprint exists");
419 sprints.put(name, new Sprint(name));
420 output.add("SPRINT NEW " + name);
421 return;
422 }
423 Sprint sprint = sprint(name);
424 switch (action) {
425 case "ADD", "REMOVE" -> {
426 Issue issue = issue(words[3]);
427 if (sprint.phase.equals("ENDED"))
428 throw new CommandError("sprint ended");
429 if (action.equals("ADD")) {
430 notDone(issue);
431 if (issue.sprint != null && !sprints.get(issue.sprint).phase.equals("ENDED"))
432 throw new CommandError("in sprint " + issue.sprint);
433 if (issue.state.equals("BACKLOG"))
434 throw new CommandError("not ready");
435 issue.sprint = name;
436 } else {
437 if (!name.equals(issue.sprint))
438 throw new CommandError("not in sprint");
439 notDone(issue);
440 issue.sprint = null;
441 }
442 record
443 (issue, "SPRINT " + action + " " + name);
444 output.add("SPRINT " + action + " " + name + " " + issue.number + " " +
445 summary(sprint));
446 }
447 case "START" -> {
448 if (!sprint.phase.equals("PLANNED"))
449 throw new CommandError("sprint not planned");
450 Sprint active = activeSprint();
451 if (active != null)
452 throw new CommandError("sprint active " + active.name);
453 if (members(sprint).isEmpty())
454 throw new CommandError("sprint empty");
455 sprint.phase = "ACTIVE";
456 sprint.start = day;
457 sprint.records.put(day, remaining(sprint));
458 output.add("SPRINT START " + name + " day " + day + " " + summary(sprint) +
459 " remaining " + remaining(sprint));
460 }
461 case "END" -> {
462 if (!sprint.phase.equals("ACTIVE"))
463 throw new CommandError("sprint not active");
464 sprint.records.put(day, remaining(sprint));
465 List<Issue> members = members(sprint);
466 List<Issue> finished = members.stream().filter(Issue::done).toList();
467 List<Issue> returned = members.stream().filter(i -> !i.done()).toList();
468 sprint.velocity = finished.stream().mapToInt(Issue::points).sum();
469 sprint.phase = "ENDED";
470 sprint.end = day;
471 output.add("SPRINT END " + name + " day " + day + " velocity " + sprint.velocity +
472 " done " + finished.size() + " returned " + returned.size());
473 for (Issue issue : returned) {
474 issue.sprint = null;
475 record
476 (issue, "SPRINT RETURN " + name);
477 output.add("RETURNED " + issue.number);
478 }
479 }
480 }

483 void process(String line, List<String> output) {
484 String[] words = line.split(" ");
485 String command = words[0];
486 switch (command) {
487 case "DAY" -> {
488 int next = Integer.parseInt(words[1]);
489 if (next < day)
490 throw new CommandError("day backwards");
491 day = next;
492 output.add("DAY " + day);
493 }
494 case "USER" -> {
495 if (!users.add(words[1]))
496 throw new CommandError("user exists");
497 output.add("USER " + words[1]);
498 }
499 case "CREATE" -> {
500 int number = Integer.parseInt(words[1]);
501 if (issues.containsKey(number))
502 throw new CommandError("issue exists");
503 if (!TYPES.contains(words[2]))
504 throw new CommandError("bad type");
505 if (!PRIORITIES.contains(words[3]))
506 throw new CommandError("bad priority");
507 Issue issue =
508 new Issue(number, words[2], words[3],
509 String.join(" ", Arrays.copyOfRange(words, 4, words.length)));
510 issues.put(number, issue);
511 record
512 (issue, "CREATE " + issue.type + " " + issue.priority);
513 output.add("CREATE " + number + " " + issue.type + " " + issue.priority + " " +
514 issue.state);
515 }
516 case "PRIORITY" -> {
517 Issue issue = issue(words[1]);
518 String priority = words[2];
519 if (!PRIORITIES.contains(priority))
520 throw new CommandError("bad priority");
521 notDone(issue);
522 if (priority.equals(issue.priority))
523 throw new CommandError("no change");
524 String old = issue.priority;
525 issue.priority = priority;
526 record
527 (issue, "PRIORITY " + old + " " + priority);
528 output.add("PRIORITY " + issue.number + " " + old + " " + priority);
529 }
530 case "ASSIGN", "REVIEWER" -> {
531 Issue issue = issue(words[1]);
532 String spec = words[2];
533 String user = user(spec);
534 notDone(issue);
535 String old = command.equals("ASSIGN") ? issue.assignee : issue.reviewer;
536 if (Objects.equals(old, user))
537 throw new CommandError("no change");
538 if (command.equals("ASSIGN")) {
539 if (user == null && Set.of("IN_PROGRESS", "REVIEW").contains(issue.state))
540 throw new CommandError("assignee required");
541 if (issue.state.equals("IN_PROGRESS") && wip(user) >= 3)
542 throw new CommandError("wip limit " + spec);
543 if (issue.state.equals("REVIEW") && Objects.equals(user, issue.reviewer))
544 throw new CommandError("reviewer is assignee");
545 issue.assignee = user;
546 } else {
547 if (user == null && issue.state.equals("REVIEW"))
548 throw new CommandError("reviewer required");
549 if (issue.state.equals("REVIEW") && Objects.equals(user, issue.assignee))
550 throw new CommandError("reviewer is assignee");
551 issue.reviewer = user;
552 }
553 record
554 (issue, command + " " + spec);
555 output.add(command + " " + issue.number + " " + spec);
556 }
557 case "ESTIMATE" -> {
558 Issue issue = issue(words[1]);
559 int points = Integer.parseInt(words[2]);
560 if (!Set.of(0, 1, 2, 3, 5, 8, 13, 21).contains(points))
561 throw new CommandError("bad estimate");
562 notDone(issue);
563 if (Objects.equals(issue.estimate, points))
564 throw new CommandError("no change");
565 String old = display(issue.estimate);
566 issue.estimate = points;
567 record
568 (issue, "ESTIMATE " + old + " " + points);
569 output.add("ESTIMATE " + issue.number + " " + old + " " + points);
570 }
571 case "COMMENT" -> {
572 Issue issue = issue(words[1]);
573 user(words[2]);
574 issue.comments++;
575 record
576 (issue, "COMMENT " + words[2] + " " +
577 String.join(" ", Arrays.copyOfRange(words, 3, words.length)));
578 output.add("COMMENT " + issue.number + " " + issue.comments);
579 }
580 case "MOVE" -> {
581 Issue issue = issue(words[1]);
582 String target = words[2];
583 if (issue.duplicate != null)
584 throw new CommandError("duplicate of " + issue.duplicate);
585 if (!WORKFLOWS.get(issue.type).contains(target))
586 throw new CommandError("bad state");
587 if (!TRANSITIONS.get(issue.type).contains(issue.state + ">" + target))
588 throw new CommandError("cannot move " + issue.state + " to " + target);
589 if (target.equals("IN_PROGRESS")) {
590 Sprint active = activeSprint();
591 if (issue.type.equals("STORY") &&
592 (active == null || !active.name.equals(issue.sprint)))
593 throw new CommandError("not in active sprint");
594 if (issue.assignee == null)
595 throw new CommandError("no assignee");
596 if (wip(issue.assignee) >= 3)
597 throw new CommandError("wip limit " + issue.assignee);
598 }
599 if (target.equals("REVIEW")) {
600 if (issue.reviewer == null)
601 throw new CommandError("no reviewer");
602 if (issue.reviewer.equals(issue.assignee))
603 throw new CommandError("reviewer is assignee");
604 }
605 if (Set.of("DONE", "RESOLVED").contains(target) && !unfinished(issue).isEmpty()) {
606 throw new CommandError("blocked by " + unfinished(issue).get(0));
607 }
608 if (target.equals("READY") && issue.estimate == null)
609 throw new CommandError("no estimate");
610 changeState(issue, target, output);
611 }
612 case "LINK", "UNLINK" -> {
613 String relation = words[2];
614 if (!(command.equals("LINK") ? Set.of("BLOCKS", "DUPLICATES") : Set.of("BLOCKS"))
615 .contains(relation))
616 throw new CommandError("bad link");
617 Issue source = issue(words[1]);
618 Issue target = issue(words[3]);
619 if (command.equals("UNLINK")) {
620 if (!source.blocks.contains(target.number))
621 throw new CommandError("no link");
622 Map<Integer, Boolean> before = blockedBefore(List.of(target.number));
623 source.blocks.remove(target.number);
624 target.blockers.remove(source.number);
625 record
626 (source, "UNLINK BLOCKS " + target.number);
627 record
628 (target, "UNLINK BLOCKED_BY " + source.number);
629 output.add("UNLINK " + source.number + " BLOCKS " + target.number);
630 notify(before, output);
631 } else {
632 if (source == target)
633 throw new CommandError("self link");
634 if (relation.equals("BLOCKS")) {
635 if (source.blocks.contains(target.number))
636 throw new CommandError("link exists");
637 if (reaches(target.number, source.number))
638 throw new CommandError("cycle");
639 notDone(target);
640 Map<Integer, Boolean> before = blockedBefore(List.of(target.number));
641 source.blocks.add(target.number);
642 target.blockers.add(source.number);
643 record
644 (source, "LINK BLOCKS " + target.number);
645 record
646 (target, "LINK BLOCKED_BY " + source.number);
647 output.add("LINK " + source.number + " BLOCKS " + target.number);
648 notify(before, output);
649 } else {
650 if (!source.type.equals(target.type))
651 throw new CommandError("type mismatch");
652 notDone(source);
653 if (target.duplicate != null)
654 throw new CommandError("target is duplicate");
655 source.duplicate = target.number;
656 target.duplicates.add(source.number);
657 record
658 (source, "LINK DUPLICATES " + target.number);
659 record
660 (target, "LINK DUPLICATED_BY " + source.number);
661 output.add("LINK " + source.number + " DUPLICATES " + target.number);
662 changeState(source, source.type.equals("BUG") ? "CLOSED" : "DONE", output);
663 if (source.sprint != null &&
664 !sprints.get(source.sprint).phase.equals("ENDED")) {
665 record
666 (source, "SPRINT REMOVE " + source.sprint);
667 source.sprint = null;
668 }
669 }
670 }
671 }
672 case "SPRINT" -> sprintCommand(words, output);
673 case "QUERY" ->
674 query(String.join(" ", Arrays.copyOfRange(words, 1, words.length)), output);
675 case "SHOW" -> {
676 Issue issue = issue(words[1]);
677 output.add("ISSUE " + issue.number + " " + issue.type + " " + issue.priority + " " +
678 issue.state + " blocked " + yesNo(blocked(issue)));
679 output.add("TITLE " + issue.title);
680 output.add("PEOPLE assignee " + display(issue.assignee) + " reviewer " +
681 display(issue.reviewer));
682 output.add("PLAN estimate " + display(issue.estimate) + " sprint " +
683 display(issue.sprint) + " comments " + issue.comments);
684 output.add("LINKS blocks " + numberList(issue.blocks) + " blocked_by " +
685 numberList(issue.blockers) + " duplicate_of " +
686 display(issue.duplicate) + " duplicated_by " +
687 numberList(issue.duplicates));
688 }
689 case "HISTORY" -> {
690 Issue issue = issue(words[1]);
691 output.add("HISTORY " + issue.number + " " + issue.history.size());
692 output.addAll(issue.history);
693 }
694 case "BOARD" -> {
695 output.add("BOARD");
696 for (String type : TYPES) {
697 StringBuilder row = new StringBuilder(type);
698 for (String state : WORKFLOWS.get(type))
699 row.append(' ').append(state).append(' ').append(
700 issues.values()
701 .stream()
702 .filter(i -> i.type.equals(type) && i.state.equals(state))
703 .count());
704 output.add(row.toString());
705 }
706 long done = issues.values().stream().filter(Issue::done).count();
707 output.add("TOTAL " + issues.size() + " open " + (issues.size() - done) + " done " +
708 done);
709 }
710 case "WORKLOAD" -> {
711 output.add("WORKLOAD " + users.size());
712 for (String user : users.stream().sorted().toList()) {
713 List<Issue> assigned = issues.values()
714 .stream()
715 .filter(i -> user.equals(i.assignee) && !i.done())
716 .toList();
717 output.add(
718 user + " assigned " + assigned.size() + " in_progress " +
719 assigned.stream().filter(i -> i.state.equals("IN_PROGRESS")).count() +
720 " review " +
721 assigned.stream().filter(i -> i.state.equals("REVIEW")).count() +
722 " reviewing " +
723 issues.values()
724 .stream()
725 .filter(i -> user.equals(i.reviewer) && i.state.equals("REVIEW"))
726 .count() +
727 " points " + assigned.stream().mapToInt(Issue::points).sum());
728 }
729 }
730 case "SPRINTS" -> {
731 output.add("SPRINTS " + sprints.size());
732 for (Sprint sprint : sprints.values())
733 output.add(switch (sprint.phase) {
734 case "PLANNED" -> sprint.name + " PLANNED " + summary(sprint);
735 case "ACTIVE" ->
736 sprint.name + " ACTIVE start " + sprint.start + " " + summary(sprint) +
737 " remaining " + remaining(sprint);
738 default ->
739 sprint.name + " ENDED start " + sprint.start + " end " + sprint.end +
740 " velocity " + sprint.velocity;
741 });
742 }
743 case "BURNDOWN" -> {
744 Sprint sprint = sprint(words[1]);
745 if (sprint.phase.equals("PLANNED"))
746 throw new CommandError("sprint not started");
747 int last = sprint.phase.equals("ACTIVE") ? day : sprint.end;
748 output.add("BURNDOWN " + sprint.name + " " + sprint.phase + " days " +
749 (last - sprint.start + 1));
750 int value = 0;
751 for (int recordDay = sprint.start; recordDay <= last; recordDay++) {
752 value = sprint.records.getOrDefault(recordDay, value);
753 output.add("day " + recordDay + " " + value);
754 }
755 }
756 }
757 }
758
759 List<String> execute(String line) {
760 List<String> output = new ArrayList<>();
761 try {
762 process(line, output);
763 } catch (CommandError error) {
764 output.add("ERROR " + error.getMessage());
765 }
766 Sprint active = activeSprint();
767 if (active != null)
768 active.records.put(day, remaining(active));
769 return output;
770 }
771 }
