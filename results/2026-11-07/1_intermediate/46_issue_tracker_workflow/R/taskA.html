84reaches <- function(start, goal) {
85 pending <- start
86 visited <- character()
87 while (length(pending)) {
88 id <- pending[length(pending)]
89 pending <- head(pending, -1L)
90 if (id == goal) return(TRUE)
91 if (id %in% visited) next
92 visited <- c(visited, id)
93 pending <- c(pending, issues[[id]]$blocks)
94 }
95 FALSE

97lower_ascii <- function(text) chartr("ABCDEFGHIJKLMNOPQRSTUVWXYZ", "abcdefghijklmnopqrstuvwxyz", text)
98condition <- function(token) {
99 field <- sub("^([a-z]*).*", "\\1", token)
100 rest <- substring(token, nchar(field) + 1L)
101 operator <- ""
102 for (op in c(">=", "<=", "!=", "=", "<", ">", "~")) if (startsWith(rest, op)) {
103 operator <- op
104 break
105 }
106 value <- substring(rest, nchar(operator) + 1L)
107 if (field == "" || operator == "" || value == "") fail(paste("syntax near", token))
108 if (!(field %in% names(field_operators))) fail(paste("unknown field", field))
109 if (!(operator %in% field_operators[[field]])) fail(paste("bad operator", token))
110 valid <- FALSE
111 if (field %in% c("assignee", "reviewer")) valid <- value == "-" || value %in% users
112 if (field == "sprint") valid <- value == "-" || value %in% names(sprints)
113 if (field %in% c("id", "estimate", "comments") && grepl("^[0-9]{1,6}$", value)) {
114 value <- as.integer(value)
115 valid <- TRUE
116 } else if (field == "estimate" && value == "-" && operator %in% equality_operators) {
117 value <- NA_integer_
118 valid <- TRUE
119 }
120 if (field == "priority" && value %in% priorities) {
121 value <- match(value, priorities)
122 valid <- TRUE
123 }
124 if (field == "type") valid <- value %in% types
125 if (field == "status") valid <- value %in% unlist(workflows, use.names = FALSE)
126 if (field %in% c("blocked", "done")) valid <- value %in% c("yes", "no")
127 if (field == "title") {
128 value <- lower_ascii(value)
129 valid <- TRUE
130 }
131 if (!valid) fail(paste("bad value", token))
132 list(field = field, operator = operator, value = value)
133}
134parse_query <- function(text) {
135 tokens <- regmatches(text, gregexpr("[()]|[^ ()]+", text))[[1]]
136 conditions <- vector("list", length(tokens))
137 operand <- TRUE
138 depth <- 0L
139 for (i in seq_along(tokens)) {
140 token <- tokens[i]
141 if (operand) {
142 if (token == "(") depth <- depth + 1L
143 else if (token %in% c("AND", "OR", ")")) fail(paste("syntax near", token))
144 else if (token != "NOT") {
145 conditions[[i]] <- condition(token)
146 operand <- FALSE
147 }
148 } else if (token %in% c("AND", "OR")) operand <- TRUE
149 else if (token == ")" && depth > 0L) depth <- depth - 1L
150 else fail(paste("syntax near", token))
151 }
152 if (operand || depth != 0L) fail("syntax at end")
153 position <- 1L
154 accept <- function(token) {
155 if (position <= length(tokens) && tokens[position] == token) {
156 position <<- position + 1L
157 return(TRUE)
158 }
159 FALSE
160 }
161 factor <- function() {
162 if (accept("NOT")) return(list(op = "NOT", child = factor()))
163 if (accept("(")) {
164 node <- expression()
165 accept(")")
166 return(node)
167 }
168 node <- list(op = "condition", value = conditions[[position]])
169 position <<- position + 1L
170 node
171 }
172 term <- function() {
173 node <- factor()
174 while (accept("AND")) node <- list(op = "AND", left = node, right = factor())
175 node
176 }
177 expression <- function() {
178 node <- term()
179 while (accept("OR")) node <- list(op = "OR", left = node, right = term())
180 node
181 }
182 expression()
183}
184matches_query <- function(node, id) {
185 if (node$op == "NOT") return(!matches_query(node$child, id))
186 if (node$op == "AND") return(matches_query(node$left, id) && matches_query(node$right, id))
187 if (node$op == "OR") return(matches_query(node$left, id) || matches_query(node$right, id))
188 c <- node$value
189 issue <- issues[[id]]
190 actual <- switch(c$field, id = as.integer(id), type = issue$kind, status = issue$state,
191 priority = match(issue$priority, priorities), assignee = issue$assignee,
192 reviewer = issue$reviewer, estimate = issue$estimate, sprint = issue$sprint,
193 comments = issue$comments, blocked = yes_no(is_blocked(id)), done = yes_no(is_done(id)),
194 title = lower_ascii(issue$title))
195 if (c$operator == "~") return(grepl(c$value, actual, fixed = TRUE))
196 equal <- if (is.na(actual) || is.na(c$value)) is.na(actual) && is.na(c$value) else actual == c$value
197 if (c$operator == "=") return(equal)
198 if (c$operator == "!=") return(!equal)
199 if (is.na(actual)) return(FALSE)
200 switch(c$operator, "<" = actual < c$value, "<=" = actual <= c$value,
201 ">" = actual > c$value, ">=" = actual >= c$value)
202}
203process_sprint <- function(words) {
204 command <- words[2]
205 name <- words[3]
206 if (command == "NEW") {
207 if (name %in% names(sprints)) fail("sprint exists")
208 sprints[[name]] <<- list(phase = "PLANNED", start = NA_integer_, end = NA_integer_, velocity = 0L, records = integer())
209 return(paste("SPRINT NEW", name))
210 }
211 require_sprint(name)
212 if (command %in% c("ADD", "REMOVE")) {
213 id <- require_issue(words[4])
214 if (sprints[[name]]$phase == "ENDED") fail("sprint ended")
215 if (command == "ADD") {
216 not_done(id)
217 old <- issues[[id]]$sprint
218 if (old != "-" && sprints[[old]]$phase != "ENDED") fail(paste("in sprint", old))
219 if (issues[[id]]$state == "BACKLOG") fail("not ready")
220 issues[[id]]$sprint <<- name
221 } else {
222 if (issues[[id]]$sprint != name) fail("not in sprint")
223 not_done(id)
224 issues[[id]]$sprint <<- "-"
225 }
226 record(id, paste("SPRINT", command, name))
227 return(paste("SPRINT", command, name, id, sprint_summary(name)))
228 }
229 if (command == "START") {
230 if (sprints[[name]]$phase != "PLANNED") fail("sprint not planned")
231 active <- active_sprint()
232 if (active != "-") fail(paste("sprint active", active))
233 if (!length(members(name))) fail("sprint empty")
234 sprints[[name]]$phase <<- "ACTIVE"
235 sprints[[name]]$start <<- day
236 remaining <- remaining_points(name)
237 sprints[[name]]$records[as.character(day)] <<- remaining
238 return(paste("SPRINT START", name, "day", day, sprint_summary(name), "remaining", remaining))
239 }
240 if (sprints[[name]]$phase != "ACTIVE") fail("sprint not active")
241 sprints[[name]]$records[as.character(day)] <<- remaining_points(name)
242 ids <- members(name)
243 finished <- ids[vapply(ids, is_done, logical(1))]
244 returned <- ids[!vapply(ids, is_done, logical(1))]
245 velocity <- sum(vapply(finished, points, integer(1)))
246 sprints[[name]]$velocity <<- velocity
247 sprints[[name]]$phase <<- "ENDED"
248 sprints[[name]]$end <<- day
249 result <- paste("SPRINT END", name, "day", day, "velocity", velocity, "done", length(finished), "returned", length(returned))
250 for (id in returned) {
251 issues[[id]]$sprint <<- "-"
252 record(id, paste("SPRINT RETURN", name))
253 result <- c(result, paste("RETURNED", id))
254 }
255 result
256}
257process_command <- function(line) {
258 words <- strsplit(line, " ", fixed = TRUE)[[1]]
259 command <- words[1]
260 if (command == "DAY") {
261 value <- as.integer(words[2])
262 if (value < day) fail("day backwards")
263 day <<- value
264 return(paste("DAY", day))
265 }
266 if (command == "USER") {
267 if (words[2] %in% users) fail("user exists")
268 users <<- c(users, words[2])
269 return(paste("USER", words[2]))
270 }
271 if (command == "CREATE") {
272 id <- words[2]
273 kind <- words[3]
274 priority <- words[4]
275 if (id %in% names(issues)) fail("issue exists")
276 if (!(kind %in% types)) fail("bad type")
277 if (!(priority %in% priorities)) fail("bad priority")
278 issues[[id]] <<- list(kind = kind, priority = priority, title = paste(words[-seq_len(4L)], collapse = " "),
279 state = workflows[[kind]][1], assignee = "-", reviewer = "-", estimate = NA_integer_,
280 sprint = "-", comments = 0L, blocks = character(), blockers = character(), duplicate = "-",
281 duplicates = character(), history = character())
282 record(id, paste("CREATE", kind, priority))
283 return(paste("CREATE", id, kind, priority, issues[[id]]$state))
284 }
285 if (command == "SPRINT") return(process_sprint(words))
286 if (command == "QUERY") {
287 tree <- parse_query(paste(words[-1L], collapse = " "))
288 ids <- ordered_ids()
289 ids <- ids[vapply(ids, function(id) matches_query(tree, id), logical(1))]
290 levels <- vapply(ids, function(id) match(issues[[id]]$priority, priorities), integer(1))
291 ids <- ids[order(-levels, as.integer(ids))]
292 lines <- vapply(ids, function(id) {
293 i <- issues[[id]]
294 paste(id, i$kind, i$priority, i$state, i$assignee, display_estimate(i$estimate), i$title)
295 }, character(1))
296 return(c(paste("QUERY", length(ids)), lines))
297 }
298 if (command == "BOARD") {
299 result <- "BOARD"
300 for (kind in types) {
301 parts <- kind
302 for (state in workflows[[kind]]) parts <- c(parts, state, sum(vapply(issues, function(i) i$kind == kind && i$state == state, logical(1))))
303 result <- c(result, paste(parts, collapse = " "))
304 }
305 finished <- sum(vapply(ordered_ids(), is_done, logical(1)))
306 return(c(result, paste("TOTAL", length(issues), "open", length(issues) - finished, "done", finished)))
307 }
308 if (command == "WORKLOAD") {
309 result <- paste("WORKLOAD", length(users))
310 for (user in sort(users, method = "radix")) {
311 assigned <- ordered_ids()[vapply(ordered_ids(), function(id) issues[[id]]$assignee == user && !is_done(id), logical(1))]
312 progress <- sum(vapply(assigned, function(id) issues[[id]]$state == "IN_PROGRESS", logical(1)))
313 review <- sum(vapply(assigned, function(id) issues[[id]]$state == "REVIEW", logical(1)))
314 reviewing <- sum(vapply(issues, function(i) i$reviewer == user && i$state == "REVIEW", logical(1)))
315 result <- c(result, paste(user, "assigned", length(assigned), "in_progress", progress, "review", review, "reviewing", reviewing, "points", sum(vapply(assigned, points, integer(1)))))
316 }
317 return(result)
318 }
319 if (command == "SPRINTS") {
320 result <- paste("SPRINTS", length(sprints))
321 for (name in names(sprints)) {
322 s <- sprints[[name]]
323 line <- if (s$phase == "PLANNED") paste(name, s$phase, sprint_summary(name)) else if (s$phase == "ACTIVE") paste(name, s$phase, "start", s$start, sprint_summary(name), "remaining", remaining_points(name)) else paste(name, s$phase, "start", s$start, "end", s$end, "velocity", s$velocity)
324 result <- c(result, line)
325 }
326 return(result)
327 }
328 if (command == "BURNDOWN") {
329 name <- require_sprint(words[2])
330 s <- sprints[[name]]
331 if (s$phase == "PLANNED") fail("sprint not started")
332 last <- if (s$phase == "ACTIVE") day else s$end
333 result <- paste("BURNDOWN", name, s$phase, "days", last - s$start + 1L)
334 value <- 0L
335 for (d in seq.int(s$start, last)) {
336 key <- as.character(d)
337 if (key %in% names(s$records)) value <- s$records[[key]]
338 result <- c(result, paste("day", d, value))
339 }
340 return(result)
341 }
342 if (command %in% c("LINK", "UNLINK")) {
343 relation <- words[3]
344 allowed <- if (command == "LINK") c("BLOCKS", "DUPLICATES") else "BLOCKS"
345 if (!(relation %in% allowed)) fail("bad link")
346 a <- require_issue(words[2])
347 b <- require_issue(words[4])
348 if (command == "UNLINK") {
349 if (!(b %in% issues[[a]]$blocks)) fail("no link")
350 before <- blocked_states(b)
351 issues[[a]]$blocks <<- setdiff(issues[[a]]$blocks, b)
352 issues[[b]]$blockers <<- setdiff(issues[[b]]$blockers, a)
353 record(a, paste("UNLINK BLOCKS", b))
354 record(b, paste("UNLINK BLOCKED_BY", a))
355 return(c(paste("UNLINK", a, "BLOCKS", b), notifications(before)))
356 }
357 if (a == b) fail("self link")
358 if (relation == "BLOCKS") {
359 if (b %in% issues[[a]]$blocks) fail("link exists")
360 if (reaches(b, a)) fail("cycle")
361 not_done(b)
362 before <- blocked_states(b)
363 issues[[a]]$blocks <<- union(issues[[a]]$blocks, b)
364 issues[[b]]$blockers <<- union(issues[[b]]$blockers, a)
365 record(a, paste("LINK BLOCKS", b))
366 record(b, paste("LINK BLOCKED_BY", a))
367 return(c(paste("LINK", a, relation, b), notifications(before)))
368 }
369 if (issues[[a]]$kind != issues[[b]]$kind) fail("type mismatch")
370 not_done(a)
371 if (issues[[b]]$duplicate != "-") fail("target is duplicate")
372 issues[[a]]$duplicate <<- b
373 issues[[b]]$duplicates <<- union(issues[[b]]$duplicates, a)
374 record(a, paste("LINK DUPLICATES", b))
375 record(b, paste("LINK DUPLICATED_BY", a))
376 result <- c(paste("LINK", a, relation, b), change_state(a, if (issues[[a]]$kind == "BUG") "CLOSED" else "DONE"))
377 old <- issues[[a]]$sprint
378 if (old != "-" && sprints[[old]]$phase != "ENDED") {
379 issues[[a]]$sprint <<- "-"
380 record(a, paste("SPRINT REMOVE", old))
381 }
382 return(result)
383 }
384 id <- require_issue(words[2])
385 i <- issues[[id]]
386 value <- if (length(words) >= 3L) words[3] else ""
387 if (command == "SHOW") return(c(
388 paste("ISSUE", id, i$kind, i$priority, i$state, "blocked", yes_no(is_blocked(id))),
389 paste("TITLE", i$title), paste("PEOPLE assignee", i$assignee, "reviewer", i$reviewer),
390 paste("PLAN estimate", display_estimate(i$estimate), "sprint", i$sprint, "comments", i$comments),
391 paste("LINKS blocks", number_list(i$blocks), "blocked_by", number_list(i$blockers), "duplicate_of", i$duplicate, "duplicated_by", number_list(i$duplicates))))
392 if (command == "HISTORY") return(c(paste("HISTORY", id, length(i$history)), i$history))
393 if (command == "COMMENT") {
394 require_user(value)
395 issues[[id]]$comments <<- i$comments + 1L
396 record(id, paste("COMMENT", value, paste(words[-seq_len(3L)], collapse = " ")))
397 return(paste("COMMENT", id, i$comments + 1L))
398 }
399 if (command == "PRIORITY") {
400 if (!(value %in% priorities)) fail("bad priority")
401 not_done(id)
402 if (value == i$priority) fail("no change")
403 issues[[id]]$priority <<- value
404 record(id, paste("PRIORITY", i$priority, value))
405 return(paste("PRIORITY", id, i$priority, value))
406 }
407 if (command %in% c("ASSIGN", "REVIEWER")) {
408 require_user(value)
409 not_done(id)
410 current <- if (command == "ASSIGN") i$assignee else i$reviewer
411 if (value == current) fail("no change")
412 if (command == "ASSIGN") {
413 if (value == "-" && i$state %in% c("IN_PROGRESS", "REVIEW")) fail("assignee required")
414 if (i$state == "IN_PROGRESS" && wip_count(value) >= 3L) fail(paste("wip limit", value))
415 if (i$state == "REVIEW" && value == i$reviewer) fail("reviewer is assignee")
416 issues[[id]]$assignee <<- value
417 } else {
418 if (value == "-" && i$state == "REVIEW") fail("reviewer required")
419 if (i$state == "REVIEW" && value == i$assignee) fail("reviewer is assignee")
420 issues[[id]]$reviewer <<- value
421 }
422 record(id, paste(command, value))
423 return(paste(command, id, value))
424 }
425 if (command == "ESTIMATE") {
426 value <- as.integer(value)
427 if (!(value %in% estimates)) fail("bad estimate")
428 not_done(id)
429 if (!is.na(i$estimate) && value == i$estimate) fail("no change")
430 previous <- display_estimate(i$estimate)
431 issues[[id]]$estimate <<- value
432 record(id, paste("ESTIMATE", previous, value))
433 return(paste("ESTIMATE", id, previous, value))
434 }
435 if (i$duplicate != "-") fail(paste("duplicate of", i$duplicate))
436 if (!(value %in% workflows[[i$kind]])) fail("bad state")
437 if (!(paste(i$state, value) %in% transitions[[i$kind]])) fail(paste("cannot move", i$state, "to", value))
438 if (value == "IN_PROGRESS") {
439 active <- active_sprint()
440 if (i$kind == "STORY" && (active == "-" || i$sprint != active)) fail("not in active sprint")
441 if (i$assignee == "-") fail("no assignee")
442 if (wip_count(i$assignee) >= 3L) fail(paste("wip limit", i$assignee))
443 }
444 if (value == "REVIEW") {
445 if (i$reviewer == "-") fail("no reviewer")
446 if (i$reviewer == i$assignee) fail("reviewer is assignee")
447 }
448 if (value %in% c("DONE", "RESOLVED")) {
449 blockers <- unfinished_blockers(id)
450 if (length(blockers)) fail(paste("blocked by", blockers[1]))
451 }
452 if (value == "READY" && is.na(i$estimate)) fail("no estimate")
453 change_state(id, value)
454}
455input <- readLines(file("stdin"), warn = FALSE)
456count <- as.integer(input[1])
457output <- character()
458for (i in seq_len(count)) {
459 lines <- tryCatch(process_command(input[i + 1L]), command_error = function(e) paste("ERROR", e$message))
460 output <- c(output, lines)
461 active <- active_sprint()
462 if (active != "-") sprints[[active]]$records[as.character(day)] <- remaining_points(active)
463}
464writeLines(output)
