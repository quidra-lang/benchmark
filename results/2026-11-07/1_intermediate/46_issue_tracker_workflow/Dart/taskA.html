3const types = ['BUG', 'TASK', 'STORY'];
4const priorities = ['LOW', 'MEDIUM', 'HIGH', 'CRITICAL'];
5const workflows = {
6 'BUG': ['OPEN', 'TRIAGED', 'IN_PROGRESS', 'REVIEW', 'RESOLVED', 'CLOSED', 'REOPENED'],
7 'TASK': ['OPEN', 'IN_PROGRESS', 'DONE'],
8 'STORY': ['BACKLOG', 'READY', 'IN_PROGRESS', 'REVIEW', 'DONE'],
9};
10const transitions = {
11 'BUG': {'OPEN>TRIAGED', 'TRIAGED>IN_PROGRESS', 'IN_PROGRESS>REVIEW', 'REVIEW>IN_PROGRESS',
12 'REVIEW>RESOLVED', 'RESOLVED>CLOSED', 'RESOLVED>REOPENED', 'CLOSED>REOPENED', 'REOPENED>IN_PROGRESS'},
13 'TASK': {'OPEN>IN_PROGRESS', 'IN_PROGRESS>OPEN', 'IN_PROGRESS>DONE'},
14 'STORY': {'BACKLOG>READY', 'READY>BACKLOG', 'READY>IN_PROGRESS', 'IN_PROGRESS>REVIEW',
15 'REVIEW>IN_PROGRESS', 'REVIEW>DONE'},
16};
17const equalityOperators = {'=', '!='};
18const orderedOperators = {'=', '!=', '<', '<=', '>', '>='};
19const fieldOperators = {
20 'id': orderedOperators, 'type': equalityOperators, 'status': equalityOperators,
21 'priority': orderedOperators, 'assignee': equalityOperators, 'reviewer': equalityOperators,
22 'estimate': orderedOperators, 'sprint': equalityOperators, 'comments': orderedOperators,
23 'blocked': equalityOperators, 'done': equalityOperators, 'title': {'~'},
24};

73typedef IssuePredicate = bool Function(Issue);

75class QueryParser {
76 final List<String> tokens;
77 final List<IssuePredicate?> conditions;
78 int position = 0;
79 QueryParser(this.tokens, this.conditions);
80
81 bool accept(String token) {
82 if (position >= tokens.length || tokens[position] != token) return false;
83 position++;
84 return true;
85 }
86
87 IssuePredicate expression() {
88 var node = term();
89 while (accept('OR')) {
90 final left = node;
91 final right = term();
92 node = (issue) => left(issue) || right(issue);
93 }
94 return node;
95 }
96
97 IssuePredicate term() {
98 var node = factor();
99 while (accept('AND')) {
100 final left = node;
101 final right = factor();
102 node = (issue) => left(issue) && right(issue);
103 }
104 return node;
105 }
106
107 IssuePredicate factor() {
108 if (accept('NOT')) {
109 final operand = factor();
110 return (issue) => !operand(issue);
111 }
112 if (accept('(')) {
113 final node = expression();
114 accept(')');
115 return node;
116 }
117 return conditions[position++]!;
118 }
119}

121class Tracker {
122 int day = 1;
123 final users = <String>{};
124 final issues = <int, Issue>{};
125 final sprints = <String, Sprint>{};
126
127 Issue requireIssue(String text) =>
128 issues[int.parse(text)] ?? (throw CommandError('unknown issue'));
129 Sprint requireSprint(String name) =>
130 sprints[name] ?? (throw CommandError('unknown sprint'));
131 String? requireUser(String spec) {
132 if (spec == '-') return null;
133 if (!users.contains(spec)) throw CommandError('unknown user');
134 return spec;
135 }
136 void requireNotDone(Issue issue) {
137 if (issue.done) throw CommandError('issue done');
138 }
139 List<Issue> orderedIssues() => issues.values.toList()
140 ..sort((a, b) => a.number.compareTo(b.number));
141 List<int> unfinished(Issue issue) => issue.blockers
142 .where((number) => !issues[number]!.done).toList()..sort();
143 bool blocked(Issue issue) => !issue.done && unfinished(issue).isNotEmpty;
144 int wip(String? user) => issues.values
145 .where((issue) => issue.assignee == user && issue.state == 'IN_PROGRESS').length;
146 Sprint? activeSprint() {
147 for (final sprint in sprints.values) {
148 if (sprint.phase == 'ACTIVE') return sprint;
149 }
150 return null;
151 }
152 List<Issue> members(Sprint sprint) => orderedIssues()
153 .where((issue) => issue.sprint == sprint.name).toList();
154 int remaining(Sprint sprint) => members(sprint).where((issue) => !issue.done)
155 .fold(0, (sum, issue) => sum + issue.points);
156 String summary(Sprint sprint) {
157 final current = members(sprint);
158 final points = current.fold<int>(0, (sum, issue) => sum + issue.points);
159 return 'issues ${current.length} points $points';
160 }
161 void record(Issue issue, String content) => issue.history.add('day $day $content');
162 Map<int, bool> blockedBefore(Iterable<int> numbers) => {
163 for (final number in numbers) number: blocked(issues[number]!),
164 };
165 void notify(Map<int, bool> before, List<String> output) {
166 final numbers = before.keys.toList()..sort();
167 for (final number in numbers) {
168 final after = blocked(issues[number]!);
169 if (after != before[number]) output.add('${after ? 'BLOCKED' : 'UNBLOCKED'} $number');
170 }
171 }
172 void changeState(Issue issue, String target, List<String> output) {
173 final before = blockedBefore(issue.blocks);
174 output.add('MOVE ${issue.number} ${issue.state} $target');
175 record(issue, 'MOVE ${issue.state} $target');
176 issue.state = target;
177 notify(before, output);
178 }
179 bool reaches(int start, int goal) {
180 final pending = [start];
181 final visited = <int>{};
182 while (pending.isNotEmpty) {
183 final number = pending.removeLast();
184 if (number == goal) return true;
185 if (visited.add(number)) pending.addAll(issues[number]!.blocks);
186 }
187 return false;
188 }

190 bool compare(int? actual, int? expected, String operation) {
191 if (operation == '=') return actual == expected;
192 if (operation == '!=') return actual != expected;
193 if (actual == null) return false;
194 return switch (operation) {
195 '<' => actual < expected!,
196 '<=' => actual <= expected!,
197 '>' => actual > expected!,
198 _ => actual >= expected!,
199 };
200 }
201
202 IssuePredicate condition(String token) {
203 var length = 0;
204 while (length < token.length && token.codeUnitAt(length) >= 97 && token.codeUnitAt(length) <= 122) {
205 length++;
206 }
207 final field = token.substring(0, length);
208 final rest = token.substring(length);
209 String? operation;
210 for (final candidate in ['>=', '<=', '!=', '=', '<', '>', '~']) {
211 if (rest.startsWith(candidate)) { operation = candidate; break; }
212 }
213 final text = operation == null ? '' : rest.substring(operation.length);
214 if (field.isEmpty || operation == null || text.isEmpty) throw CommandError('syntax near $token');
215 final operators = fieldOperators[field];
216 if (operators == null) throw CommandError('unknown field $field');
217 if (!operators.contains(operation)) throw CommandError('bad operator $token');
218 final none = text == '-' && const {'assignee', 'reviewer', 'estimate', 'sprint'}.contains(field)
219 && equalityOperators.contains(operation);
220 int? numeric;
221 String? string = none ? null : text;
222 bool valid;
223 switch (field) {
224 case 'id':
225 case 'estimate':
226 case 'comments':
227 valid = none || RegExp(r'^[0-9]{1,6}$').hasMatch(text);
228 if (valid && !none) numeric = int.parse(text);
229 break;
230 case 'priority':
231 valid = priorities.contains(text);
232 if (valid) numeric = priorities.indexOf(text);
233 break;
234 case 'type': valid = types.contains(text); break;
235 case 'status': valid = workflows.values.any((states) => states.contains(text)); break;
236 case 'assignee':
237 case 'reviewer': valid = none || users.contains(text); break;
238 case 'sprint': valid = none || sprints.containsKey(text); break;
239 case 'blocked':
240 case 'done': valid = text == 'yes' || text == 'no'; break;
241 case 'title': string = text.toLowerCase(); valid = true; break;
242 default: valid = false;
243 }
244 if (!valid) throw CommandError('bad value $token');
245 final expectedNumber = numeric;
246 final expectedString = string;
247 final comparison = operation;
248 return (issue) {
249 if (const {'id', 'estimate', 'comments', 'priority'}.contains(field)) {
250 final int? actual = switch (field) {
251 'id' => issue.number,
252 'estimate' => issue.estimate,
253 'comments' => issue.comments,
254 _ => issue.level,
255 };
256 return compare(actual, expectedNumber, comparison);
257 }
258 final String? actual = switch (field) {
259 'type' => issue.type,
260 'status' => issue.state,
261 'assignee' => issue.assignee,
262 'reviewer' => issue.reviewer,
263 'sprint' => issue.sprint,
264 'blocked' => yesNo(blocked(issue)),
265 'done' => yesNo(issue.done),
266 _ => issue.title.toLowerCase(),
267 };
268 return switch (comparison) {
269 '~' => actual!.contains(expectedString!),
270 '=' => actual == expectedString,
271 _ => actual != expectedString,
272 };
273 };
274 }

276 void query(String text, List<String> output) {
277 final tokens = RegExp(r'[()]|[^ ()]+').allMatches(text).map((match) => match.group(0)!).toList();
278 final conditions = <IssuePredicate?>[];
279 var operand = true;
280 var depth = 0;
281 for (final token in tokens) {
282 IssuePredicate? predicate;
283 if (operand) {
284 if (token == 'NOT') {
285 } else if (token == '(') {
286 depth++;
287 } else if (const {'AND', 'OR', ')'}.contains(token)) {
288 throw CommandError('syntax near $token');
289 } else {
290 predicate = condition(token);
291 operand = false;
292 }
293 } else if (token == 'AND' || token == 'OR') {
294 operand = true;
295 } else if (token == ')' && depth > 0) {
296 depth--;
297 } else {
298 throw CommandError('syntax near $token');
299 }
300 conditions.add(predicate);
301 }
302 if (operand || depth != 0) throw CommandError('syntax at end');
303 final predicate = QueryParser(tokens, conditions).expression();
304 final matches = issues.values.where(predicate).toList()..sort((a, b) {
305 final priority = b.level.compareTo(a.level);
306 return priority != 0 ? priority : a.number.compareTo(b.number);
307 });
308 output.add('QUERY ${matches.length}');
309 for (final issue in matches) {
310 output.add('${issue.number} ${issue.type} ${issue.priority} ${issue.state} '
311 '${display(issue.assignee)} ${display(issue.estimate)} ${issue.title}');
312 }
313 }

315 void sprintCommand(List<String> words, List<String> output) {
316 final action = words[1];
317 final name = words[2];
318 if (action == 'NEW') {
319 if (sprints.containsKey(name)) throw CommandError('sprint exists');
320 sprints[name] = Sprint(name);
321 output.add('SPRINT NEW $name');
322 return;
323 }
324 final sprint = requireSprint(name);
325 switch (action) {
326 case 'ADD':
327 case 'REMOVE':
328 final issue = requireIssue(words[3]);
329 if (sprint.phase == 'ENDED') throw CommandError('sprint ended');
330 if (action == 'ADD') {
331 requireNotDone(issue);
332 if (issue.sprint != null && sprints[issue.sprint]!.phase != 'ENDED') {
333 throw CommandError('in sprint ${issue.sprint}');
334 }
335 if (issue.state == 'BACKLOG') throw CommandError('not ready');
336 issue.sprint = name;
337 } else {
338 if (issue.sprint != name) throw CommandError('not in sprint');
339 requireNotDone(issue);
340 issue.sprint = null;
341 }
342 record(issue, 'SPRINT $action $name');
343 output.add('SPRINT $action $name ${issue.number} ${summary(sprint)}');
344 break;
345 case 'START':
346 if (sprint.phase != 'PLANNED') throw CommandError('sprint not planned');
347 final active = activeSprint();
348 if (active != null) throw CommandError('sprint active ${active.name}');
349 if (members(sprint).isEmpty) throw CommandError('sprint empty');
350 sprint.phase = 'ACTIVE';
351 sprint.start = day;
352 sprint.records[day] = remaining(sprint);
353 output.add('SPRINT START $name day $day ${summary(sprint)} remaining ${remaining(sprint)}');
354 break;
355 case 'END':
356 if (sprint.phase != 'ACTIVE') throw CommandError('sprint not active');
357 sprint.records[day] = remaining(sprint);
358 final current = members(sprint);
359 final finished = current.where((issue) => issue.done).toList();
360 final returned = current.where((issue) => !issue.done).toList();
361 sprint.velocity = finished.fold(0, (sum, issue) => sum + issue.points);
362 sprint.phase = 'ENDED';
363 sprint.end = day;
364 output.add('SPRINT END $name day $day velocity ${sprint.velocity} done ${finished.length} returned ${returned.length}');
365 for (final issue in returned) {
366 issue.sprint = null;
367 record(issue, 'SPRINT RETURN $name');
368 output.add('RETURNED ${issue.number}');
369 }
370 break;
371 }
372 }

374 void process(String line, List<String> output) {
375 final words = line.split(' ');
376 final command = words[0];
377 switch (command) {
378 case 'DAY': {
379 final next = int.parse(words[1]);
380 if (next < day) throw CommandError('day backwards');
381 day = next;
382 output.add('DAY $day');
383 break;
384 }
385 case 'USER':
386 if (!users.add(words[1])) throw CommandError('user exists');
387 output.add('USER ${words[1]}');
388 break;
389 case 'CREATE': {
390 final number = int.parse(words[1]);
391 if (issues.containsKey(number)) throw CommandError('issue exists');
392 if (!types.contains(words[2])) throw CommandError('bad type');
393 if (!priorities.contains(words[3])) throw CommandError('bad priority');
394 final issue = Issue(number, words[2], words[3], words.sublist(4).join(' '));
395 issues[number] = issue;
396 record(issue, 'CREATE ${issue.type} ${issue.priority}');
397 output.add('CREATE $number ${issue.type} ${issue.priority} ${issue.state}');
398 break;
399 }
400 case 'PRIORITY': {
401 final issue = requireIssue(words[1]);
402 final priority = words[2];
403 if (!priorities.contains(priority)) throw CommandError('bad priority');
404 requireNotDone(issue);
405 if (priority == issue.priority) throw CommandError('no change');
406 final old = issue.priority;
407 issue.priority = priority;
408 record(issue, 'PRIORITY $old $priority');
409 output.add('PRIORITY ${issue.number} $old $priority');
410 break;
411 }
412 case 'ASSIGN':
413 case 'REVIEWER': {
414 final issue = requireIssue(words[1]);
415 final spec = words[2];
416 final user = requireUser(spec);
417 requireNotDone(issue);
418 final old = command == 'ASSIGN' ? issue.assignee : issue.reviewer;
419 if (old == user) throw CommandError('no change');
420 if (command == 'ASSIGN') {
421 if (user == null && const {'IN_PROGRESS', 'REVIEW'}.contains(issue.state)) throw CommandError('assignee required');
422 if (issue.state == 'IN_PROGRESS' && wip(user) >= 3) throw CommandError('wip limit $spec');
423 if (issue.state == 'REVIEW' && user == issue.reviewer) throw CommandError('reviewer is assignee');
424 issue.assignee = user;
425 } else {
426 if (user == null && issue.state == 'REVIEW') throw CommandError('reviewer required');
427 if (issue.state == 'REVIEW' && user == issue.assignee) throw CommandError('reviewer is assignee');
428 issue.reviewer = user;
429 }
430 record(issue, '$command $spec');
431 output.add('$command ${issue.number} $spec');
432 break;
433 }
434 case 'ESTIMATE': {
435 final issue = requireIssue(words[1]);
436 final points = int.parse(words[2]);
437 if (!const {0, 1, 2, 3, 5, 8, 13, 21}.contains(points)) throw CommandError('bad estimate');
438 requireNotDone(issue);
439 if (issue.estimate == points) throw CommandError('no change');
440 final old = display(issue.estimate);
441 issue.estimate = points;
442 record(issue, 'ESTIMATE $old $points');
443 output.add('ESTIMATE ${issue.number} $old $points');
444 break;
445 }
446 case 'COMMENT': {
447 final issue = requireIssue(words[1]);
448 requireUser(words[2]);
449 issue.comments++;
450 record(issue, 'COMMENT ${words[2]} ${words.sublist(3).join(' ')}');
451 output.add('COMMENT ${issue.number} ${issue.comments}');
452 break;
453 }
454 case 'MOVE': {
455 final issue = requireIssue(words[1]);
456 final target = words[2];
457 if (issue.duplicate != null) throw CommandError('duplicate of ${issue.duplicate}');
458 if (!workflows[issue.type]!.contains(target)) throw CommandError('bad state');
459 if (!transitions[issue.type]!.contains('${issue.state}>$target')) {
460 throw CommandError('cannot move ${issue.state} to $target');
461 }
462 if (target == 'IN_PROGRESS') {
463 final active = activeSprint();
464 if (issue.type == 'STORY' && (active == null || issue.sprint != active.name)) throw CommandError('not in active sprint');
465 if (issue.assignee == null) throw CommandError('no assignee');
466 if (wip(issue.assignee) >= 3) throw CommandError('wip limit ${issue.assignee}');
467 }
468 if (target == 'REVIEW') {
469 if (issue.reviewer == null) throw CommandError('no reviewer');
470 if (issue.reviewer == issue.assignee) throw CommandError('reviewer is assignee');
471 }
472 final blockers = unfinished(issue);
473 if (const {'DONE', 'RESOLVED'}.contains(target) && blockers.isNotEmpty) throw CommandError('blocked by ${blockers.first}');
474 if (target == 'READY' && issue.estimate == null) throw CommandError('no estimate');
475 changeState(issue, target, output);
476 break;
477 }
478 case 'LINK':
479 case 'UNLINK': {
480 final relation = words[2];
481 if (!(command == 'LINK' ? {'BLOCKS', 'DUPLICATES'} : {'BLOCKS'}).contains(relation)) throw CommandError('bad link');
482 final source = requireIssue(words[1]);
483 final target = requireIssue(words[3]);
484 if (command == 'UNLINK') {
485 if (!source.blocks.contains(target.number)) throw CommandError('no link');
486 final before = blockedBefore([target.number]);
487 source.blocks.remove(target.number);
488 target.blockers.remove(source.number);
489 record(source, 'UNLINK BLOCKS ${target.number}');
490 record(target, 'UNLINK BLOCKED_BY ${source.number}');
491 output.add('UNLINK ${source.number} BLOCKS ${target.number}');
492 notify(before, output);
493 } else {
494 if (identical(source, target)) throw CommandError('self link');
495 if (relation == 'BLOCKS') {
496 if (source.blocks.contains(target.number)) throw CommandError('link exists');
497 if (reaches(target.number, source.number)) throw CommandError('cycle');
498 requireNotDone(target);
499 final before = blockedBefore([target.number]);
500 source.blocks.add(target.number);
501 target.blockers.add(source.number);
502 record(source, 'LINK BLOCKS ${target.number}');
503 record(target, 'LINK BLOCKED_BY ${source.number}');
504 output.add('LINK ${source.number} BLOCKS ${target.number}');
505 notify(before, output);
506 } else {
507 if (source.type != target.type) throw CommandError('type mismatch');
508 requireNotDone(source);
509 if (target.duplicate != null) throw CommandError('target is duplicate');
510 source.duplicate = target.number;
511 target.duplicates.add(source.number);
512 record(source, 'LINK DUPLICATES ${target.number}');
513 record(target, 'LINK DUPLICATED_BY ${source.number}');
514 output.add('LINK ${source.number} DUPLICATES ${target.number}');
515 changeState(source, source.type == 'BUG' ? 'CLOSED' : 'DONE', output);
516 if (source.sprint != null && sprints[source.sprint]!.phase != 'ENDED') {
517 record(source, 'SPRINT REMOVE ${source.sprint}');
518 source.sprint = null;
519 }
520 }
521 }
522 break;
523 }
524 case 'SPRINT': sprintCommand(words, output); break;
525 case 'QUERY': query(words.sublist(1).join(' '), output); break;
526 case 'SHOW': {
527 final issue = requireIssue(words[1]);
528 output.add('ISSUE ${issue.number} ${issue.type} ${issue.priority} ${issue.state} blocked ${yesNo(blocked(issue))}');
529 output.add('TITLE ${issue.title}');
530 output.add('PEOPLE assignee ${display(issue.assignee)} reviewer ${display(issue.reviewer)}');
531 output.add('PLAN estimate ${display(issue.estimate)} sprint ${display(issue.sprint)} comments ${issue.comments}');
532 output.add('LINKS blocks ${numberList(issue.blocks)} blocked_by ${numberList(issue.blockers)} '
533 'duplicate_of ${display(issue.duplicate)} duplicated_by ${numberList(issue.duplicates)}');
534 break;
535 }
536 case 'HISTORY': {
537 final issue = requireIssue(words[1]);
538 output.add('HISTORY ${issue.number} ${issue.history.length}');
539 output.addAll(issue.history);
540 break;
541 }
542 case 'BOARD': {
543 output.add('BOARD');
544 for (final type in types) {
545 final row = StringBuffer(type);
546 for (final state in workflows[type]!) {
547 row.write(' $state ${issues.values.where((issue) => issue.type == type && issue.state == state).length}');
548 }
549 output.add(row.toString());
550 }
551 final done = issues.values.where((issue) => issue.done).length;
552 output.add('TOTAL ${issues.length} open ${issues.length - done} done $done');
553 break;
554 }
555 case 'WORKLOAD': {
556 output.add('WORKLOAD ${users.length}');
557 final orderedUsers = users.toList()..sort();
558 for (final user in orderedUsers) {
559 final assigned = issues.values.where((issue) => issue.assignee == user && !issue.done).toList();
560 final inProgress = assigned.where((issue) => issue.state == 'IN_PROGRESS').length;
561 final review = assigned.where((issue) => issue.state == 'REVIEW').length;
562 final reviewing = issues.values.where((issue) => issue.reviewer == user && issue.state == 'REVIEW').length;
563 final points = assigned.fold<int>(0, (sum, issue) => sum + issue.points);
564 output.add('$user assigned ${assigned.length} in_progress $inProgress review $review reviewing $reviewing points $points');
565 }
566 break;
567 }
568 case 'SPRINTS':
569 output.add('SPRINTS ${sprints.length}');
570 for (final sprint in sprints.values) {
571 output.add(switch (sprint.phase) {
572 'PLANNED' => '${sprint.name} PLANNED ${summary(sprint)}',
573 'ACTIVE' => '${sprint.name} ACTIVE start ${sprint.start} ${summary(sprint)} remaining ${remaining(sprint)}',
574 _ => '${sprint.name} ENDED start ${sprint.start} end ${sprint.end} velocity ${sprint.velocity}',
575 });
576 }
577 break;
578 case 'BURNDOWN': {
579 final sprint = requireSprint(words[1]);
580 if (sprint.phase == 'PLANNED') throw CommandError('sprint not started');
581 final last = sprint.phase == 'ACTIVE' ? day : sprint.end;
582 output.add('BURNDOWN ${sprint.name} ${sprint.phase} days ${last - sprint.start + 1}');
583 var value = 0;
584 for (var recordDay = sprint.start; recordDay <= last; recordDay++) {
585 value = sprint.records[recordDay] ?? value;
586 output.add('day $recordDay $value');
587 }
588 break;
589 }
590 }
591 }
592
593 List<String> execute(String line) {
594 final output = <String>[];
595 try {
596 process(line, output);
597 } on CommandError catch (error) {
598 output.add('ERROR ${error.message}');
599 }
600 final active = activeSprint();
601 if (active != null) active.records[day] = remaining(active);
602 return output;
603 }
604}

606void main() {
607 final tracker = Tracker();
608 final count = int.parse(stdin.readLineSync()!);
609 for (var index = 0; index < count; index++) {
610 for (final line in tracker.execute(stdin.readLineSync()!)) {
611 stdout.writeln(line);
612 }
613 }
