6static class Rules
7{
8 public static readonly string[] Types = { "BUG", "TASK", "STORY" };
9 public static readonly string[] Priorities = { "LOW", "MEDIUM", "HIGH", "CRITICAL" };
10 public static readonly Dictionary<string, string[]> Workflows = new()
11 {
12 ["BUG"] = new[] { "OPEN", "TRIAGED", "IN_PROGRESS", "REVIEW", "RESOLVED", "CLOSED", "REOPENED" },
13 ["TASK"] = new[] { "OPEN", "IN_PROGRESS", "DONE" },
14 ["STORY"] = new[] { "BACKLOG", "READY", "IN_PROGRESS", "REVIEW", "DONE" }
15 };
16 public static readonly Dictionary<string, HashSet<string>> Transitions = new()
17 {
18 ["BUG"] = new() { "OPEN>TRIAGED", "TRIAGED>IN_PROGRESS", "IN_PROGRESS>REVIEW", "REVIEW>IN_PROGRESS", "REVIEW>RESOLVED", "RESOLVED>CLOSED", "RESOLVED>REOPENED", "CLOSED>REOPENED", "REOPENED>IN_PROGRESS" },
19 ["TASK"] = new() { "OPEN>IN_PROGRESS", "IN_PROGRESS>OPEN", "IN_PROGRESS>DONE" },
20 ["STORY"] = new() { "BACKLOG>READY", "READY>BACKLOG", "READY>IN_PROGRESS", "IN_PROGRESS>REVIEW", "REVIEW>IN_PROGRESS", "REVIEW>DONE" }
21 };
22 public static readonly HashSet<string> AllStates = Workflows.Values.SelectMany(x => x).ToHashSet();
23 public static readonly HashSet<int> Estimates = new() { 0, 1, 2, 3, 5, 8, 13, 21 };
24 public static readonly string[] Equality = { "=", "!=" };
25 public static readonly string[] Ordered = { "=", "!=", "<", "<=", ">", ">=" };
26 public static readonly Dictionary<string, string[]> FieldOperators = new()
27 {
28 ["id"] = Ordered, ["type"] = Equality, ["status"] = Equality, ["priority"] = Ordered,
29 ["assignee"] = Equality, ["reviewer"] = Equality, ["estimate"] = Ordered, ["sprint"] = Equality,
30 ["comments"] = Ordered, ["blocked"] = Equality, ["done"] = Equality, ["title"] = new[] { "~" }
31 };
32 public static bool IsDone(string state) => state is "RESOLVED" or "CLOSED" or "DONE";
33 public static string Display(object? value) => value?.ToString() ?? "-";
34 public static string YesNo(bool value) => value ? "yes" : "no";
35 public static string NumberList(SortedSet<int> values) => values.Count == 0 ? "-" : string.Join(",", values);
36}

85sealed record Condition(string Field, string Operation, object? Value)
86{
87 public bool Matches(Tracker tracker, Issue issue)
88 {
89 object? actual = tracker.FieldValue(issue, Field);
90 if (Operation == "~") return ((string)actual!).Contains((string)Value!, StringComparison.Ordinal);
91 if (Operation == "=") return Equals(actual, Value);
92 if (Operation == "!=") return !Equals(actual, Value);
93 if (actual is not int a || Value is not int b) return false;
94 return Operation switch { "<" => a < b, "<=" => a <= b, ">" => a > b, _ => a >= b };
95 }
96}
97
98abstract record QueryNode
99{
100 public abstract bool Matches(Tracker tracker, Issue issue);
101}
102sealed record ConditionNode(Condition Condition) : QueryNode
103{
104 public override bool Matches(Tracker tracker, Issue issue) => Condition.Matches(tracker, issue);
105}
106sealed record NotNode(QueryNode Operand) : QueryNode
107{
108 public override bool Matches(Tracker tracker, Issue issue) => !Operand.Matches(tracker, issue);
109}
110sealed record BinaryNode(string Operation, QueryNode Left, QueryNode Right) : QueryNode
111{
112 public override bool Matches(Tracker tracker, Issue issue) => Operation == "AND"
113 ? Left.Matches(tracker, issue) && Right.Matches(tracker, issue)
114 : Left.Matches(tracker, issue) || Right.Matches(tracker, issue);
115}

117sealed class QueryParser
118{
119 readonly List<string> tokens;
120 readonly List<Condition?> conditions;
121 int position;
122 public QueryParser(List<string> tokens, List<Condition?> conditions)
123 {
124 this.tokens = tokens;
125 this.conditions = conditions;
126 }
127 bool Accept(string token)
128 {
129 if (position >= tokens.Count || tokens[position] != token) return false;
130 position++;
131 return true;
132 }
133 public QueryNode Expression()
134 {
135 QueryNode node = Term();
136 while (Accept("OR")) node = new BinaryNode("OR", node, Term());
137 return node;
138 }
139 QueryNode Term()
140 {
141 QueryNode node = Factor();
142 while (Accept("AND")) node = new BinaryNode("AND", node, Factor());
143 return node;
144 }
145 QueryNode Factor()
146 {
147 if (Accept("NOT")) return new NotNode(Factor());
148 if (Accept("("))
149 {
150 QueryNode node = Expression();
151 Accept(")");
152 return node;
153 }
154 return new ConditionNode(conditions[position++]!);
155 }
156}

158sealed class Tracker
159{
160 int day = 1;
161 readonly SortedSet<string> users = new(StringComparer.Ordinal);
162 readonly SortedDictionary<int, Issue> issues = new();
163 readonly Dictionary<string, Sprint> sprints = new(StringComparer.Ordinal);
164 readonly List<Sprint> sprintOrder = new();
165 Sprint? ActiveSprint => sprintOrder.FirstOrDefault(s => s.Phase == "ACTIVE");
166
167 Issue RequireIssue(string text) => issues.TryGetValue(int.Parse(text), out Issue? issue)
168 ? issue : throw new CommandError("unknown issue");
169 string? RequireUser(string text)
170 {
171 if (text == "-") return null;
172 if (!users.Contains(text)) throw new CommandError("unknown user");
173 return text;
174 }
175 Sprint RequireSprint(string name) => sprints.TryGetValue(name, out Sprint? sprint)
176 ? sprint : throw new CommandError("unknown sprint");
177 static void RequireNotDone(Issue issue)
178 {
179 if (issue.Done) throw new CommandError("issue done");
180 }
181 void Record(Issue issue, string content) => issue.History.Add($"day {day} {content}");
182 List<int> UnfinishedBlockers(Issue issue) => issue.Blockers.Where(n => !issues[n].Done).ToList();
183 bool Blocked(Issue issue) => !issue.Done && issue.Blockers.Any(n => !issues[n].Done);
184 int Wip(string? user) => issues.Values.Count(i => i.Assignee == user && i.State == "IN_PROGRESS");
185 List<Issue> Members(Sprint sprint) => issues.Values.Where(i => i.Sprint == sprint.Name).ToList();
186 int Remaining(Sprint sprint) => Members(sprint).Where(i => !i.Done).Sum(i => i.Points);
187 string Summary(Sprint sprint)
188 {
189 List<Issue> members = Members(sprint);
190 return $"issues {members.Count} points {members.Sum(i => i.Points)}";
191 }
192 List<string> Notify(Dictionary<int, bool> before)
193 {
194 var lines = new List<string>();
195 foreach (int number in before.Keys.OrderBy(n => n))
196 {
197 bool after = Blocked(issues[number]);
198 if (before[number] != after) lines.Add($"{(after ? "BLOCKED" : "UNBLOCKED")} {number}");
199 }
200 return lines;
201 }
202 List<string> ChangeState(Issue issue, string target)
203 {
204 var before = issue.Blocks.ToDictionary(n => n, n => Blocked(issues[n]));
205 var lines = new List<string> { $"MOVE {issue.Number} {issue.State} {target}" };
206 Record(issue, $"MOVE {issue.State} {target}");
207 issue.State = target;
208 lines.AddRange(Notify(before));
209 return lines;
210 }
211 bool Reaches(int start, int goal)
212 {
213 var pending = new Stack<int>();
214 var seen = new HashSet<int>();
215 pending.Push(start);
216 while (pending.Count > 0)
217 {
218 int number = pending.Pop();
219 if (number == goal) return true;
220 if (seen.Add(number)) foreach (int child in issues[number].Blocks) pending.Push(child);
221 }
222 return false;
223 }

225 public object? FieldValue(Issue issue, string field) => field switch
226 {
227 "id" => issue.Number, "type" => issue.Kind, "status" => issue.State, "priority" => issue.Level,
228 "assignee" => issue.Assignee, "reviewer" => issue.Reviewer, "estimate" => issue.Estimate,
229 "sprint" => issue.Sprint, "comments" => issue.Comments, "blocked" => Rules.YesNo(Blocked(issue)),
230 "done" => Rules.YesNo(issue.Done), _ => issue.Title.ToLowerInvariant()
231 };

233 Condition ParseCondition(string token)
234 {
235 int length = 0;
236 while (length < token.Length && token[length] >= 'a' && token[length] <= 'z') length++;
237 string field = token[..length];
238 string rest = token[length..];
239 string? operation = new[] { ">=", "<=", "!=", "=", "<", ">", "~" }.FirstOrDefault(op => rest.StartsWith(op, StringComparison.Ordinal));
240 if (field.Length == 0 || operation == null || rest.Length == operation.Length) throw new CommandError($"syntax near {token}");
241 string text = rest[operation.Length..];
242 if (!Rules.FieldOperators.TryGetValue(field, out string[]? allowed)) throw new CommandError($"unknown field {field}");
243 if (!allowed.Contains(operation)) throw new CommandError($"bad operator {token}");
244 object? value;
245 if (text == "-" && new[] { "assignee", "reviewer", "estimate", "sprint" }.Contains(field) && Rules.Equality.Contains(operation)) value = null;
246 else if (new[] { "id", "estimate", "comments" }.Contains(field) && text.Length is >= 1 and <= 6 && text.All(c => c >= '0' && c <= '9')) value = int.Parse(text);
247 else if (field == "priority" && Rules.Priorities.Contains(text)) value = Array.IndexOf(Rules.Priorities, text);
248 else if (field == "type" && Rules.Types.Contains(text)) value = text;
249 else if (field == "status" && Rules.AllStates.Contains(text)) value = text;
250 else if ((field is "assignee" or "reviewer") && users.Contains(text)) value = text;
251 else if (field == "sprint" && sprints.ContainsKey(text)) value = text;
252 else if ((field is "blocked" or "done") && (text is "yes" or "no")) value = text;
253 else if (field == "title") value = text.ToLowerInvariant();
254 else throw new CommandError($"bad value {token}");
255 return new Condition(field, operation, value);
256 }

258 List<string> Query(string text)
259 {
260 var tokens = new List<string>();
261 int position = 0;
262 while (position < text.Length)
263 {
264 if (text[position] == ' ') { position++; continue; }
265 if (text[position] is '(' or ')') { tokens.Add(text[position++].ToString()); continue; }
266 int start = position;
267 while (position < text.Length && text[position] != ' ' && text[position] != '(' && text[position] != ')') position++;
268 tokens.Add(text[start..position]);
269 }
270 bool expectingOperand = true;
271 int depth = 0;
272 var conditions = new List<Condition?>();
273 foreach (string token in tokens)
274 {
275 Condition? condition = null;
276 if (expectingOperand)
277 {
278 if (token == "(") depth++;
279 else if (token is "AND" or "OR" or ")") throw new CommandError($"syntax near {token}");
280 else if (token != "NOT") { condition = ParseCondition(token); expectingOperand = false; }
281 }
282 else if (token is "AND" or "OR") expectingOperand = true;
283 else if (token == ")" && depth > 0) depth--;
284 else throw new CommandError($"syntax near {token}");
285 conditions.Add(condition);
286 }
287 if (expectingOperand || depth != 0) throw new CommandError("syntax at end");
288 QueryNode tree = new QueryParser(tokens, conditions).Expression();
289 List<Issue> matches = issues.Values.Where(i => tree.Matches(this, i)).OrderByDescending(i => i.Level).ThenBy(i => i.Number).ToList();
290 var lines = new List<string> { $"QUERY {matches.Count}" };
291 lines.AddRange(matches.Select(i => $"{i.Number} {i.Kind} {i.Priority} {i.State} {Rules.Display(i.Assignee)} {Rules.Display(i.Estimate)} {i.Title}"));
292 return lines;
293 }

295 List<string> SprintCommand(string[] words)
296 {
297 string action = words[1], name = words[2];
298 if (action == "NEW")
299 {
300 if (sprints.ContainsKey(name)) throw new CommandError("sprint exists");
301 var created = new Sprint(name);
302 sprints.Add(name, created);
303 sprintOrder.Add(created);
304 return new() { $"SPRINT NEW {name}" };
305 }
306 Sprint sprint = RequireSprint(name);
307 if (action is "ADD" or "REMOVE")
308 {
309 Issue issue = RequireIssue(words[3]);
310 if (sprint.Phase == "ENDED") throw new CommandError("sprint ended");
311 if (action == "ADD")
312 {
313 RequireNotDone(issue);
314 if (issue.Sprint != null && sprints[issue.Sprint].Phase != "ENDED") throw new CommandError($"in sprint {issue.Sprint}");
315 if (issue.State == "BACKLOG") throw new CommandError("not ready");
316 issue.Sprint = name;
317 }
318 else
319 {
320 if (issue.Sprint != name) throw new CommandError("not in sprint");
321 RequireNotDone(issue);
322 issue.Sprint = null;
323 }
324 Record(issue, $"SPRINT {action} {name}");
325 return new() { $"SPRINT {action} {name} {issue.Number} {Summary(sprint)}" };
326 }
327 if (action == "START")
328 {
329 if (sprint.Phase != "PLANNED") throw new CommandError("sprint not planned");
330 if (ActiveSprint is Sprint active) throw new CommandError($"sprint active {active.Name}");
331 if (Members(sprint).Count == 0) throw new CommandError("sprint empty");
332 sprint.Phase = "ACTIVE";
333 sprint.Start = day;
334 int remaining = Remaining(sprint);
335 sprint.Records[day] = remaining;
336 return new() { $"SPRINT START {name} day {day} {Summary(sprint)} remaining {remaining}" };
337 }
338 if (sprint.Phase != "ACTIVE") throw new CommandError("sprint not active");
339 sprint.Records[day] = Remaining(sprint);
340 List<Issue> members = Members(sprint);
341 List<Issue> finished = members.Where(i => i.Done).ToList();
342 List<Issue> returned = members.Where(i => !i.Done).ToList();
343 sprint.Velocity = finished.Sum(i => i.Points);
344 sprint.Phase = "ENDED";
345 sprint.End = day;
346 var lines = new List<string> { $"SPRINT END {name} day {day} velocity {sprint.Velocity} done {finished.Count} returned {returned.Count}" };
347 foreach (Issue issue in returned)
348 {
349 issue.Sprint = null;
350 Record(issue, $"SPRINT RETURN {name}");
351 lines.Add($"RETURNED {issue.Number}");
352 }
353 return lines;
354 }

356 List<string> Command(string line)
357 {
358 string[] words = line.Split(' ');
359 string command = words[0];
360 switch (command)
361 {
362 case "DAY":
363 int targetDay = int.Parse(words[1]);
364 if (targetDay < day) throw new CommandError("day backwards");
365 day = targetDay;
366 return new() { $"DAY {day}" };
367 case "USER":
368 if (!users.Add(words[1])) throw new CommandError("user exists");
369 return new() { $"USER {words[1]}" };
370 case "CREATE":
371 {
372 int number = int.Parse(words[1]);
373 string kind = words[2], priority = words[3];
374 if (issues.ContainsKey(number)) throw new CommandError("issue exists");
375 if (!Rules.Types.Contains(kind)) throw new CommandError("bad type");
376 if (!Rules.Priorities.Contains(priority)) throw new CommandError("bad priority");
377 var issue = new Issue(number, kind, priority, string.Join(" ", words.Skip(4)));
378 issues.Add(number, issue);
379 Record(issue, $"CREATE {kind} {priority}");
380 return new() { $"CREATE {number} {kind} {priority} {issue.State}" };
381 }
382 case "PRIORITY":
383 {
384 Issue issue = RequireIssue(words[1]);
385 string priority = words[2];
386 if (!Rules.Priorities.Contains(priority)) throw new CommandError("bad priority");
387 RequireNotDone(issue);
388 if (priority == issue.Priority) throw new CommandError("no change");
389 string old = issue.Priority;
390 issue.Priority = priority;
391 Record(issue, $"PRIORITY {old} {priority}");
392 return new() { $"PRIORITY {issue.Number} {old} {priority}" };
393 }
394 case "ASSIGN":
395 case "REVIEWER":
396 {
397 Issue issue = RequireIssue(words[1]);
398 string? user = RequireUser(words[2]);
399 RequireNotDone(issue);
400 if (command == "ASSIGN")
401 {
402 if (user == issue.Assignee) throw new CommandError("no change");
403 if (user == null && (issue.State is "IN_PROGRESS" or "REVIEW")) throw new CommandError("assignee required");
404 if (issue.State == "IN_PROGRESS" && Wip(user) >= 3) throw new CommandError($"wip limit {words[2]}");
405 if (issue.State == "REVIEW" && user == issue.Reviewer) throw new CommandError("reviewer is assignee");
406 issue.Assignee = user;
407 }
408 else
409 {
410 if (user == issue.Reviewer) throw new CommandError("no change");
411 if (issue.State == "REVIEW" && user == null) throw new CommandError("reviewer required");
412 if (issue.State == "REVIEW" && user == issue.Assignee) throw new CommandError("reviewer is assignee");
413 issue.Reviewer = user;
414 }
415 Record(issue, $"{command} {words[2]}");
416 return new() { $"{command} {issue.Number} {words[2]}" };
417 }
418 case "ESTIMATE":
419 {
420 Issue issue = RequireIssue(words[1]);
421 int points = int.Parse(words[2]);
422 if (!Rules.Estimates.Contains(points)) throw new CommandError("bad estimate");
423 RequireNotDone(issue);
424 if (issue.Estimate == points) throw new CommandError("no change");
425 string old = Rules.Display(issue.Estimate);
426 issue.Estimate = points;
427 Record(issue, $"ESTIMATE {old} {points}");
428 return new() { $"ESTIMATE {issue.Number} {old} {points}" };
429 }
430 case "COMMENT":
431 {
432 Issue issue = RequireIssue(words[1]);
433 RequireUser(words[2]);
434 issue.Comments++;
435 Record(issue, $"COMMENT {words[2]} {string.Join(" ", words.Skip(3))}");
436 return new() { $"COMMENT {issue.Number} {issue.Comments}" };
437 }
438 case "MOVE":
439 {
440 Issue issue = RequireIssue(words[1]);
441 string target = words[2];
442 if (issue.Duplicate != null) throw new CommandError($"duplicate of {issue.Duplicate}");
443 if (!Rules.Workflows[issue.Kind].Contains(target)) throw new CommandError("bad state");
444 if (!Rules.Transitions[issue.Kind].Contains($"{issue.State}>{target}")) throw new CommandError($"cannot move {issue.State} to {target}");
445 if (target == "IN_PROGRESS")
446 {
447 if (issue.Kind == "STORY" && (ActiveSprint == null || issue.Sprint != ActiveSprint.Name)) throw new CommandError("not in active sprint");
448 if (issue.Assignee == null) throw new CommandError("no assignee");
449 if (Wip(issue.Assignee) >= 3) throw new CommandError($"wip limit {issue.Assignee}");
450 }
451 if (target == "REVIEW")
452 {
453 if (issue.Reviewer == null) throw new CommandError("no reviewer");
454 if (issue.Reviewer == issue.Assignee) throw new CommandError("reviewer is assignee");
455 }
456 if (target is "DONE" or "RESOLVED")
457 {
458 List<int> blockers = UnfinishedBlockers(issue);
459 if (blockers.Count > 0) throw new CommandError($"blocked by {blockers[0]}");
460 }
461 if (target == "READY" && issue.Estimate == null) throw new CommandError("no estimate");
462 return ChangeState(issue, target);
463 }
464 case "LINK":
465 case "UNLINK":
466 {
467 string relation = words[2];
468 bool unlink = command == "UNLINK";
469 if (relation != "BLOCKS" && (unlink || relation != "DUPLICATES")) throw new CommandError("bad link");
470 Issue a = RequireIssue(words[1]), b = RequireIssue(words[3]);
471 if (unlink)
472 {
473 if (!a.Blocks.Contains(b.Number)) throw new CommandError("no link");
474 }
475 else
476 {
477 if (a.Number == b.Number) throw new CommandError("self link");
478 if (relation == "BLOCKS")
479 {
480 if (a.Blocks.Contains(b.Number)) throw new CommandError("link exists");
481 if (Reaches(b.Number, a.Number)) throw new CommandError("cycle");
482 RequireNotDone(b);
483 }
484 else
485 {
486 if (a.Kind != b.Kind) throw new CommandError("type mismatch");
487 RequireNotDone(a);
488 if (b.Duplicate != null) throw new CommandError("target is duplicate");
489 a.Duplicate = b.Number;
490 b.Duplicates.Add(a.Number);
491 Record(a, $"LINK DUPLICATES {b.Number}");
492 Record(b, $"LINK DUPLICATED_BY {a.Number}");
493 var lines = new List<string> { $"LINK {a.Number} DUPLICATES {b.Number}" };
494 lines.AddRange(ChangeState(a, a.Kind == "BUG" ? "CLOSED" : "DONE"));
495 if (a.Sprint != null && sprints[a.Sprint].Phase != "ENDED")
496 {
497 string old = a.Sprint;
498 a.Sprint = null;
499 Record(a, $"SPRINT REMOVE {old}");
500 }
501 return lines;
502 }
503 }
504 var before = new Dictionary<int, bool> { [b.Number] = Blocked(b) };
505 if (unlink) { a.Blocks.Remove(b.Number); b.Blockers.Remove(a.Number); }
506 else { a.Blocks.Add(b.Number); b.Blockers.Add(a.Number); }
507 Record(a, $"{command} BLOCKS {b.Number}");
508 Record(b, $"{command} BLOCKED_BY {a.Number}");
509 var result = new List<string> { $"{command} {a.Number} BLOCKS {b.Number}" };
510 result.AddRange(Notify(before));
511 return result;
512 }
513 case "SPRINT": return SprintCommand(words);
514 case "QUERY": return Query(string.Join(" ", words.Skip(1)));
515 case "SHOW":
516 {
517 Issue issue = RequireIssue(words[1]);
518 return new()
519 {
520 $"ISSUE {issue.Number} {issue.Kind} {issue.Priority} {issue.State} blocked {Rules.YesNo(Blocked(issue))}",
521 $"TITLE {issue.Title}",
522 $"PEOPLE assignee {Rules.Display(issue.Assignee)} reviewer {Rules.Display(issue.Reviewer)}",
523 $"PLAN estimate {Rules.Display(issue.Estimate)} sprint {Rules.Display(issue.Sprint)} comments {issue.Comments}",
524 $"LINKS blocks {Rules.NumberList(issue.Blocks)} blocked_by {Rules.NumberList(issue.Blockers)} duplicate_of {Rules.Display(issue.Duplicate)} duplicated_by {Rules.NumberList(issue.Duplicates)}"
525 };
526 }
527 case "HISTORY":
528 {
529 Issue issue = RequireIssue(words[1]);
530 var lines = new List<string> { $"HISTORY {issue.Number} {issue.History.Count}" };
531 lines.AddRange(issue.History);
532 return lines;
533 }
534 case "BOARD":
535 {
536 var lines = new List<string> { "BOARD" };
537 foreach (string kind in Rules.Types)
538 lines.Add(kind + string.Concat(Rules.Workflows[kind].Select(state => $" {state} {issues.Values.Count(i => i.Kind == kind && i.State == state)}")));
539 int done = issues.Values.Count(i => i.Done);
540 lines.Add($"TOTAL {issues.Count} open {issues.Count - done} done {done}");
541 return lines;
542 }
543 case "WORKLOAD":
544 {
545 var lines = new List<string> { $"WORKLOAD {users.Count}" };
546 foreach (string user in users)
547 {
548 List<Issue> assigned = issues.Values.Where(i => !i.Done && i.Assignee == user).ToList();
549 int progress = assigned.Count(i => i.State == "IN_PROGRESS");
550 int review = assigned.Count(i => i.State == "REVIEW");
551 int reviewing = issues.Values.Count(i => i.Reviewer == user && i.State == "REVIEW");
552 lines.Add($"{user} assigned {assigned.Count} in_progress {progress} review {review} reviewing {reviewing} points {assigned.Sum(i => i.Points)}");
553 }
554 return lines;
555 }
556 case "SPRINTS":
557 {
558 var lines = new List<string> { $"SPRINTS {sprintOrder.Count}" };
559 foreach (Sprint sprint in sprintOrder)
560 lines.Add(sprint.Phase switch
561 {
562 "PLANNED" => $"{sprint.Name} PLANNED {Summary(sprint)}",
563 "ACTIVE" => $"{sprint.Name} ACTIVE start {sprint.Start} {Summary(sprint)} remaining {Remaining(sprint)}",
564 _ => $"{sprint.Name} ENDED start {sprint.Start} end {sprint.End} velocity {sprint.Velocity}"
565 });
566 return lines;
567 }
568 case "BURNDOWN":
569 {
570 Sprint sprint = RequireSprint(words[1]);
571 if (sprint.Phase == "PLANNED") throw new CommandError("sprint not started");
572 int start = sprint.Start!.Value, last = sprint.Phase == "ACTIVE" ? day : sprint.End!.Value;
573 var lines = new List<string> { $"BURNDOWN {sprint.Name} {sprint.Phase} days {last - start + 1}" };
574 int points = 0;
575 for (int date = start; date <= last; date++)
576 {
577 if (sprint.Records.TryGetValue(date, out int recorded)) points = recorded;
578 lines.Add($"day {date} {points}");
579 }
580 return lines;
581 }
582 default: throw new InvalidOperationException("Unknown command");
583 }
584 }
585
586 public List<string> Execute(string line)
587 {
588 List<string> lines;
589 try { lines = Command(line); }
590 catch (CommandError error) { lines = new() { $"ERROR {error.Message}" }; }
591 if (ActiveSprint is Sprint sprint) sprint.Records[day] = Remaining(sprint);
592 return lines;
593 }
594}
