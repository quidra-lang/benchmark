71def tokenize(query):
72 tokens = []
73 current = ""
74 for character in query:
75 if character in " ()":
76 if current:
77 tokens.append(current)
78 current = ""
79 if character != " ":
80 tokens.append(character)
81 else:
82 current += character
83 if current:
84 tokens.append(current)
85 return tokens

88def split_condition(token):
89 length = 0
90 while length < len(token) and "a" <= token[length] <= "z":
91 length += 1
92 rest = token[length:]
93 for operator in CONDITION_OPERATORS:
94 if rest.startswith(operator):
95 return token[:length], operator, rest[len(operator):]
96 return token[:length], None, ""

103def compare(actual, operator, expected):
104 if operator == "=":
105 return actual == expected
106 if operator == "!=":
107 return actual != expected
108 if actual is None:
109 return False
110 if operator == "<":
111 return actual < expected
112 if operator == "<=":
113 return actual <= expected
114 if operator == ">":
115 return actual > expected
116 return actual >= expected

178class BinaryNode:
179 def __init__(self, operator, left, right):
180 self.operator = operator
181 self.left = left
182 self.right = right
183
184 def matches(self, tracker, issue):
185 if self.operator == "AND":
186 return self.left.matches(tracker, issue) and self.right.matches(tracker, issue)
187 return self.left.matches(tracker, issue) or self.right.matches(tracker, issue)

190class QueryParser:
191 def __init__(self, tokens, conditions):
192 self.tokens = tokens
193 self.conditions = conditions
194 self.position = 0
195
196 def accept(self, token):
197 if self.position < len(self.tokens) and self.tokens[self.position] == token:
198 self.position += 1
199 return True
200 return False
201
202 def expression(self):
203 node = self.term()
204 while self.accept("OR"):
205 node = BinaryNode("OR", node, self.term())
206 return node
207
208 def term(self):
209 node = self.factor()
210 while self.accept("AND"):
211 node = BinaryNode("AND", node, self.factor())
212 return node
213
214 def factor(self):
215 if self.accept("NOT"):
216 return NotNode(self.factor())
217 if self.accept("("):
218 node = self.expression()
219 self.accept(")")
220 return node
221 condition = self.conditions[self.position]
222 self.position += 1
223 return condition

315 def reaches(self, start, goal):
316 pending = [start.number]
317 visited = set()
318 while pending:
319 number = pending.pop()
320 if number == goal.number:
321 return True
322 if number in visited:
323 continue
324 visited.add(number)
325 pending.extend(self.issues[number].blocks)
326 return False

422 def move(self, arguments):
423 issue = self.require_issue(arguments[0])
424 target = arguments[1]
425 if issue.duplicate is not None:
426 raise CommandError(f"duplicate of {issue.duplicate}")
427 if target not in WORKFLOWS[issue.kind]:
428 raise CommandError("bad state")
429 if (issue.state, target) not in TRANSITIONS[issue.kind]:
430 raise CommandError(f"cannot move {issue.state} to {target}")
431 if target == "IN_PROGRESS":
432 active = self.active_sprint()
433 if issue.kind == "STORY" and (active is None or issue.sprint != active.name):
434 raise CommandError("not in active sprint")
435 if issue.assignee is None:
436 raise CommandError("no assignee")
437 if self.wip_count(issue.assignee) >= WIP_LIMIT:
438 raise CommandError(f"wip limit {issue.assignee}")
439 if target == "REVIEW":
440 if issue.reviewer is None:
441 raise CommandError("no reviewer")
442 if issue.reviewer == issue.assignee:
443 raise CommandError("reviewer is assignee")
444 if target in ("DONE", "RESOLVED"):
445 blockers = self.unfinished_blockers(issue)
446 if blockers:
447 raise CommandError(f"blocked by {blockers[0]}")
448 if target == "READY" and issue.estimate is None:
449 raise CommandError("no estimate")
450 self.change_state(issue, target)

452 def link(self, arguments):
453 relation = arguments[1]
454 if relation not in ("BLOCKS", "DUPLICATES"):
455 raise CommandError("bad link")
456 source = self.require_issue(arguments[0])
457 target = self.require_issue(arguments[2])
458 if source is target:
459 raise CommandError("self link")

462 else:

465 def link_blocks(self, source, target):
466 if target.number in source.blocks:
467 raise CommandError("link exists")
468 if self.reaches(target, source):
469 raise CommandError("cycle")
470 self.require_not_done(target)
471 before = self.blocked_states([target.number])
472 source.blocks.add(target.number)
473 target.blockers.add(source.number)
474 self.record(source, f"LINK BLOCKS {target.number}")
475 self.record(target, f"LINK BLOCKED_BY {source.number}")
476 self.emit("LINK", source.number, "BLOCKS", target.number)
477 self.notify(before)
478
479 def link_duplicates(self, source, target):
480 if source.kind != target.kind:
481 raise CommandError("type mismatch")
482 self.require_not_done(source)
483 if target.duplicate is not None:
484 raise CommandError("target is duplicate")
485 source.duplicate = target.number
486 target.duplicates.add(source.number)
487 self.record(source, f"LINK DUPLICATES {target.number}")
488 self.record(target, f"LINK DUPLICATED_BY {source.number}")
489 self.emit("LINK", source.number, "DUPLICATES", target.number)
490 self.change_state(source, DUPLICATE_STATES[source.kind])
491 if source.sprint is not None and self.sprints[source.sprint].phase != "ENDED":
492 self.record(source, f"SPRINT REMOVE {source.sprint}")
493 source.sprint = None

562 sprint = self.require_sprint(arguments[0])
563 if sprint.phase != "ACTIVE":
564 raise CommandError("sprint not active")

568 returned = [issue for issue in members if not issue.done()]

573 "SPRINT", "END", sprint.name, "day", sprint.end,

616 def condition_value(self, token, field, operator, text):
617 if text == "-" and field in NONE_FIELDS and operator in EQUALITY_OPERATORS:
618 return None
619 if field in ("id", "estimate", "comments") and is_number(text):
620 return int(text)
621 if field == "priority" and text in PRIORITIES:
622 return PRIORITIES.index(text)
623 if field == "type" and text in TYPES:
624 return text
625 if field == "status" and text in ALL_STATES:
626 return text
627 if field in ("assignee", "reviewer") and text in self.users:
628 return text
629 if field == "sprint" and text in self.sprints:
630 return text
631 if field in ("blocked", "done") and text in ("yes", "no"):
632 return text
633 if field == "title":
634 return text.lower()
635 raise CommandError(f"bad value {token}")

637 def condition(self, token):
638 field, operator, text = split_condition(token)
639 if not field or operator is None or not text:
640 raise CommandError(f"syntax near {token}")
641 if field not in FIELD_OPERATORS:
642 raise CommandError(f"unknown field {field}")
643 if operator not in FIELD_OPERATORS[field]:
644 raise CommandError(f"bad operator {token}")
645 return Condition(field, operator, self.condition_value(token, field, operator, text))

647 def check_query(self, tokens):
648 conditions = []
649 expecting_operand = True
650 depth = 0
651 for token in tokens:
652 condition = None
653 if expecting_operand:
654 if token == "(":
655 depth += 1
656 elif token in ("AND", "OR", ")"):
657 raise CommandError(f"syntax near {token}")
658 elif token != "NOT":
659 condition = self.condition(token)
660 expecting_operand = False
661 elif token in ("AND", "OR"):
662 expecting_operand = True
663 elif token == ")" and depth > 0:
664 depth -= 1
665 else:
666 raise CommandError(f"syntax near {token}")
667 conditions.append(condition)
668 if expecting_operand or depth != 0:
669 raise CommandError("syntax at end")
670 return conditions

672 def query(self, arguments):
673 tokens = tokenize(" ".join(arguments))
674 conditions = self.check_query(tokens)
675 tree = QueryParser(tokens, conditions).expression()
676 matches = [issue for issue in self.issues.values() if tree.matches(self, issue)]
677 matches.sort(key=lambda issue: (-issue.level(), issue.number))
678 self.emit("QUERY", len(matches))
679 for issue in matches:
680 self.emit(
681 issue.number, issue.kind, issue.priority, issue.state,
682 display(issue.assignee), display(issue.estimate), issue.title,
683 )

710 def board(self, arguments):
711 issues = list(self.issues.values())
712 self.emit("BOARD")
713 for kind in TYPES:
714 parts = [kind]
715 for state in WORKFLOWS[kind]:
716 count = sum(issue.kind == kind and issue.state == state for issue in issues)
717 parts.extend((state, count))
718 self.emit(*parts)
719 finished = sum(issue.done() for issue in issues)
720 self.emit("TOTAL", len(issues), "open", len(issues) - finished, "done", finished)

722 def workload(self, arguments):
723 issues = list(self.issues.values())
724 self.emit("WORKLOAD", len(self.users))
725 for user in sorted(self.users):
726 assigned = [issue for issue in issues if issue.assignee == user and not issue.done()]
727 in_progress = sum(issue.state == "IN_PROGRESS" for issue in assigned)
728 review = sum(issue.state == "REVIEW" for issue in assigned)
729 reviewing = sum(
730 issue.reviewer == user and issue.state == "REVIEW" for issue in issues
731 )
732 points = sum(issue.points() for issue in assigned)
733 self.emit(
734 user, "assigned", len(assigned), "in_progress", in_progress,
735 "review", review, "reviewing", reviewing, "points", points,
736 )

738 def list_sprints(self, arguments):
739 self.emit("SPRINTS", len(self.sprints))
740 for sprint in self.sprints.values():
741 if sprint.phase == "PLANNED":
742 self.emit(sprint.name, "PLANNED", self.summary(sprint))
743 elif sprint.phase == "ACTIVE":
744 self.emit(
745 sprint.name, "ACTIVE", "start", sprint.start, self.summary(sprint),
746 "remaining", self.remaining_points(sprint),
747 )
748 else:
749 self.emit(
750 sprint.name, "ENDED", "start", sprint.start, "end", sprint.end,
751 "velocity", sprint.velocity,
752 )

754 def burndown(self, arguments):
755 sprint = self.require_sprint(arguments[0])
756 if sprint.phase == "PLANNED":
757 raise CommandError("sprint not started")
758 last = self.day if sprint.phase == "ACTIVE" else sprint.end
759 self.emit("BURNDOWN", sprint.name, sprint.phase, "days", last - sprint.start + 1)
760 value = 0
761 for day in range(sprint.start, last + 1):
762 if day in sprint.records:
763 value = sprint.records[day]
764 self.emit("day", day, value)

766 def execute(self, line):
767 command, *arguments = line.split(" ")
768 handlers = {
769 "DAY": self.set_day,
770 "USER": self.add_user,
771 "CREATE": self.create,
772 "PRIORITY": self.set_priority,
773 "ASSIGN": self.assign,
774 "REVIEWER": self.set_reviewer,
775 "ESTIMATE": self.set_estimate,
776 "COMMENT": self.comment,
777 "MOVE": self.move,
778 "LINK": self.link,
779 "UNLINK": self.unlink,
780 "SPRINT": self.sprint,
781 "QUERY": self.query,
782 "SHOW": self.show,
783 "HISTORY": self.show_history,
784 "BOARD": self.board,
785 "WORKLOAD": self.workload,
786 "SPRINTS": self.list_sprints,
787 "BURNDOWN": self.burndown,
788 }
789 try:
790 handlers[command](arguments)
791 except CommandError as error:
792 self.emit("ERROR", error)
793 active = self.active_sprint()
794 if active is not None:
795 active.records[self.day] = self.remaining_points(active)
