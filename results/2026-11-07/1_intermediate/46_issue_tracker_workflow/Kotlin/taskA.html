3private val TYPES = listOf("BUG", "TASK", "STORY")
4private val PRIORITIES = listOf("LOW", "MEDIUM", "HIGH", "CRITICAL")
5private val WORKFLOWS = mapOf(
6 "BUG" to listOf("OPEN", "TRIAGED", "IN_PROGRESS", "REVIEW", "RESOLVED", "CLOSED", "REOPENED"),
7 "TASK" to listOf("OPEN", "IN_PROGRESS", "DONE"),
8 "STORY" to listOf("BACKLOG", "READY", "IN_PROGRESS", "REVIEW", "DONE")
9)
10private val TRANSITIONS = mapOf(
11 "BUG" to setOf("OPEN>TRIAGED", "TRIAGED>IN_PROGRESS", "IN_PROGRESS>REVIEW", "REVIEW>IN_PROGRESS",
12 "REVIEW>RESOLVED", "RESOLVED>CLOSED", "RESOLVED>REOPENED", "CLOSED>REOPENED", "REOPENED>IN_PROGRESS"),
13 "TASK" to setOf("OPEN>IN_PROGRESS", "IN_PROGRESS>OPEN", "IN_PROGRESS>DONE"),
14 "STORY" to setOf("BACKLOG>READY", "READY>BACKLOG", "READY>IN_PROGRESS", "IN_PROGRESS>REVIEW",
15 "REVIEW>IN_PROGRESS", "REVIEW>DONE")
16)
17private val EQUALITY = setOf("=", "!=")
18private val ORDERED = EQUALITY + setOf("<", "<=", ">", ">=")
19private val FIELD_OPERATORS = mapOf(
20 "id" to ORDERED, "type" to EQUALITY, "status" to EQUALITY, "priority" to ORDERED,
21 "assignee" to EQUALITY, "reviewer" to EQUALITY, "estimate" to ORDERED,
22 "sprint" to EQUALITY, "comments" to ORDERED, "blocked" to EQUALITY,
23 "done" to EQUALITY, "title" to setOf("~")
24)

26private class CommandError(message: String) : RuntimeException(message)
27private fun display(value: Any?) = value?.toString() ?: "-"
28private fun yesNo(value: Boolean) = if (value) "yes" else "no"
29private fun numberList(numbers: Collection<Int>) = numbers.sorted().joinToString(",").ifEmpty { "-" }

31private class Issue(val number: Int, val type: String, var priority: String, val title: String) {
32 var state = WORKFLOWS.getValue(type).first()
33 var assignee: String? = null
34 var reviewer: String? = null
35 var estimate: Int? = null
36 var sprint: String? = null
37 var comments = 0
38 val blocks = mutableSetOf<Int>()
39 val blockers = mutableSetOf<Int>()
40 var duplicate: Int? = null
41 val duplicates = mutableSetOf<Int>()
42 val history = mutableListOf<String>()
43 fun done() = state in setOf("RESOLVED", "CLOSED", "DONE")
44 fun points() = estimate ?: 0
45 fun level() = PRIORITIES.indexOf(priority)
46}

48private class Sprint(val name: String) {
49 var phase = "PLANNED"
50 var start = 0
51 var end = 0
52 var velocity = 0
53 val records = mutableMapOf<Int, Int>()
54}

56private class QueryParser(
57 private val tokens: List<String>,
58 private val conditions: List<((Issue) -> Boolean)?>
59) {
60 private var position = 0
61 private fun accept(token: String): Boolean {
62 if (tokens.getOrNull(position) != token) return false
63 position++
64 return true
65 }
66
67 fun expression(): (Issue) -> Boolean {
68 var node = term()
69 while (accept("OR")) {
70 val left = node
71 val right = term()
72 node = { left(it) || right(it) }
73 }
74 return node
75 }
76
77 private fun term(): (Issue) -> Boolean {
78 var node = factor()
79 while (accept("AND")) {
80 val left = node
81 val right = factor()
82 node = { left(it) && right(it) }
83 }
84 return node
85 }
86
87 private fun factor(): (Issue) -> Boolean {
88 if (accept("NOT")) {
89 val operand = factor()
90 return { !operand(it) }
91 }
92 if (accept("(")) {
93 val node = expression()
94 accept(")")
95 return node
96 }
97 return conditions[position++]!!
98 }
99}

101private class Tracker {
102 private var day = 1
103 private val users = mutableSetOf<String>()
104 private val issues = mutableMapOf<Int, Issue>()
105 private val sprints = linkedMapOf<String, Sprint>()
106
107 private fun issue(text: String) = issues[text.toInt()] ?: throw CommandError("unknown issue")
108 private fun sprint(name: String) = sprints[name] ?: throw CommandError("unknown sprint")
109 private fun user(spec: String): String? {
110 if (spec == "-") return null
111 if (spec !in users) throw CommandError("unknown user")
112 return spec
113 }
114 private fun notDone(issue: Issue) {
115 if (issue.done()) throw CommandError("issue done")
116 }
117 private fun orderedIssues() = issues.values.sortedBy { it.number }
118 private fun unfinished(issue: Issue) = issue.blockers.filter { !issues.getValue(it).done() }.sorted()
119 private fun blocked(issue: Issue) = !issue.done() && unfinished(issue).isNotEmpty()
120 private fun wip(user: String?) = issues.values.count { it.assignee == user && it.state == "IN_PROGRESS" }
121 private fun activeSprint() = sprints.values.firstOrNull { it.phase == "ACTIVE" }
122 private fun members(sprint: Sprint) = orderedIssues().filter { it.sprint == sprint.name }
123 private fun remaining(sprint: Sprint) = members(sprint).filter { !it.done() }.sumOf { it.points() }
124 private fun summary(sprint: Sprint): String {
125 val members = members(sprint)
126 return "issues ${members.size} points ${members.sumOf { it.points() }}"
127 }
128 private fun record(issue: Issue, content: String) { issue.history.add("day $day $content") }
129 private fun blockedBefore(numbers: Collection<Int>) = numbers.associateWith { blocked(issues.getValue(it)) }
130 private fun notify(before: Map<Int, Boolean>, output: MutableList<String>) {
131 for (number in before.keys.sorted()) {
132 val after = blocked(issues.getValue(number))
133 if (after != before.getValue(number)) output.add("${if (after) "BLOCKED" else "UNBLOCKED"} $number")
134 }
135 }
136 private fun changeState(issue: Issue, target: String, output: MutableList<String>) {
137 val before = blockedBefore(issue.blocks)
138 output.add("MOVE ${issue.number} ${issue.state} $target")
139 record(issue, "MOVE ${issue.state} $target")
140 issue.state = target
141 notify(before, output)
142 }
143 private fun reaches(start: Int, goal: Int): Boolean {
144 val pending = ArrayDeque<Int>()
145 val visited = mutableSetOf<Int>()
146 pending.add(start)
147 while (pending.isNotEmpty()) {
148 val number = pending.removeFirst()
149 if (number == goal) return true
150 if (visited.add(number)) pending.addAll(issues.getValue(number).blocks)
151 }
152 return false
153 }

155 private fun compare(actual: Int?, expected: Int?, operator: String): Boolean = when (operator) {
156 "=" -> actual == expected
157 "!=" -> actual != expected
158 "<" -> actual != null && actual < expected!!
159 "<=" -> actual != null && actual <= expected!!
160 ">" -> actual != null && actual > expected!!
161 else -> actual != null && actual >= expected!!
162 }
163
164 private fun condition(token: String): (Issue) -> Boolean {
165 val field = token.takeWhile { it in 'a'..'z' }
166 val rest = token.substring(field.length)
167 val operator = listOf(">=", "<=", "!=", "=", "<", ">", "~").firstOrNull { rest.startsWith(it) }
168 val text = operator?.let { rest.substring(it.length) } ?: ""
169 if (field.isEmpty() || operator == null || text.isEmpty()) throw CommandError("syntax near $token")
170 val operators = FIELD_OPERATORS[field] ?: throw CommandError("unknown field $field")
171 if (operator !in operators) throw CommandError("bad operator $token")
172 val none = text == "-" && field in setOf("assignee", "reviewer", "estimate", "sprint") && operator in EQUALITY
173 var numeric: Int? = null
174 var string: String? = if (none) null else text
175 val valid = when (field) {
176 "id", "estimate", "comments" -> if (none) true else {
177 if (Regex("[0-9]{1,6}").matches(text)) { numeric = text.toInt(); true } else false
178 }
179 "priority" -> if (text in PRIORITIES) { numeric = PRIORITIES.indexOf(text); true } else false
180 "type" -> text in TYPES
181 "status" -> WORKFLOWS.values.any { text in it }
182 "assignee", "reviewer" -> none || text in users
183 "sprint" -> none || text in sprints
184 "blocked", "done" -> text == "yes" || text == "no"
185 "title" -> { string = text.lowercase(); true }
186 else -> false
187 }
188 if (!valid) throw CommandError("bad value $token")
189 val expectedNumber = numeric
190 val expectedString = string
191 return { issue ->
192 if (field in setOf("id", "estimate", "comments", "priority")) {
193 val actual = when (field) {
194 "id" -> issue.number
195 "estimate" -> issue.estimate
196 "comments" -> issue.comments
197 else -> issue.level()
198 }
199 compare(actual, expectedNumber, operator)
200 } else {
201 val actual = when (field) {
202 "type" -> issue.type
203 "status" -> issue.state
204 "assignee" -> issue.assignee
205 "reviewer" -> issue.reviewer
206 "sprint" -> issue.sprint
207 "blocked" -> yesNo(blocked(issue))
208 "done" -> yesNo(issue.done())
209 else -> issue.title.lowercase()
210 }
211 when (operator) {
212 "~" -> actual!!.contains(expectedString!!)
213 "=" -> actual == expectedString
214 else -> actual != expectedString
215 }
216 }
217 }
218 }

220 private fun query(text: String, output: MutableList<String>) {
221 val tokens = Regex("[()]|[^ ()]+").findAll(text).map { it.value }.toList()
222 val conditions = mutableListOf<((Issue) -> Boolean)?>()
223 var operand = true
224 var depth = 0
225 for (token in tokens) {
226 var predicate: ((Issue) -> Boolean)? = null
227 if (operand) when (token) {
228 "NOT" -> Unit
229 "(" -> depth++
230 "AND", "OR", ")" -> throw CommandError("syntax near $token")
231 else -> { predicate = condition(token); operand = false }
232 } else when (token) {
233 "AND", "OR" -> operand = true
234 ")" -> if (depth > 0) depth-- else throw CommandError("syntax near $token")
235 else -> throw CommandError("syntax near $token")
236 }
237 conditions.add(predicate)
238 }
239 if (operand || depth != 0) throw CommandError("syntax at end")
240 val predicate = QueryParser(tokens, conditions).expression()
241 val matches = issues.values.filter(predicate).sortedWith(compareByDescending<Issue> { it.level() }.thenBy { it.number })
242 output.add("QUERY ${matches.size}")
243 for (issue in matches) output.add("${issue.number} ${issue.type} ${issue.priority} ${issue.state} " +
244 "${display(issue.assignee)} ${display(issue.estimate)} ${issue.title}")
245 }

247 private fun sprintCommand(words: List<String>, output: MutableList<String>) {
248 val action = words[1]
249 val name = words[2]
250 if (action == "NEW") {
251 if (name in sprints) throw CommandError("sprint exists")
252 sprints[name] = Sprint(name)
253 output.add("SPRINT NEW $name")
254 return
255 }
256 val sprint = sprint(name)
257 when (action) {
258 "ADD", "REMOVE" -> {
259 val issue = issue(words[3])
260 if (sprint.phase == "ENDED") throw CommandError("sprint ended")
261 if (action == "ADD") {
262 notDone(issue)
263 if (issue.sprint != null && sprints.getValue(issue.sprint!!).phase != "ENDED") {
264 throw CommandError("in sprint ${issue.sprint}")
265 }
266 if (issue.state == "BACKLOG") throw CommandError("not ready")
267 issue.sprint = name
268 } else {
269 if (issue.sprint != name) throw CommandError("not in sprint")
270 notDone(issue)
271 issue.sprint = null
272 }
273 record(issue, "SPRINT $action $name")
274 output.add("SPRINT $action $name ${issue.number} ${summary(sprint)}")
275 }
276 "START" -> {
277 if (sprint.phase != "PLANNED") throw CommandError("sprint not planned")
278 activeSprint()?.let { throw CommandError("sprint active ${it.name}") }
279 if (members(sprint).isEmpty()) throw CommandError("sprint empty")
280 sprint.phase = "ACTIVE"
281 sprint.start = day
282 sprint.records[day] = remaining(sprint)
283 output.add("SPRINT START $name day $day ${summary(sprint)} remaining ${remaining(sprint)}")
284 }
285 "END" -> {
286 if (sprint.phase != "ACTIVE") throw CommandError("sprint not active")
287 sprint.records[day] = remaining(sprint)
288 val members = members(sprint)
289 val finished = members.filter { it.done() }
290 val returned = members.filter { !it.done() }
291 sprint.velocity = finished.sumOf { it.points() }
292 sprint.phase = "ENDED"
293 sprint.end = day
294 output.add("SPRINT END $name day $day velocity ${sprint.velocity} done ${finished.size} returned ${returned.size}")
295 for (issue in returned) {
296 issue.sprint = null
297 record(issue, "SPRINT RETURN $name")
298 output.add("RETURNED ${issue.number}")
299 }
300 }
301 }
302 }

304 private fun process(line: String, output: MutableList<String>) {
305 val words = line.split(' ')
306 when (val command = words[0]) {
307 "DAY" -> {
308 val next = words[1].toInt()
309 if (next < day) throw CommandError("day backwards")
310 day = next
311 output.add("DAY $day")
312 }
313 "USER" -> {
314 if (!users.add(words[1])) throw CommandError("user exists")
315 output.add("USER ${words[1]}")
316 }
317 "CREATE" -> {
318 val number = words[1].toInt()
319 if (number in issues) throw CommandError("issue exists")
320 if (words[2] !in TYPES) throw CommandError("bad type")
321 if (words[3] !in PRIORITIES) throw CommandError("bad priority")
322 val issue = Issue(number, words[2], words[3], words.drop(4).joinToString(" "))
323 issues[number] = issue
324 record(issue, "CREATE ${issue.type} ${issue.priority}")
325 output.add("CREATE $number ${issue.type} ${issue.priority} ${issue.state}")
326 }
327 "PRIORITY" -> {
328 val issue = issue(words[1])
329 val priority = words[2]
330 if (priority !in PRIORITIES) throw CommandError("bad priority")
331 notDone(issue)
332 if (priority == issue.priority) throw CommandError("no change")
333 val old = issue.priority
334 issue.priority = priority
335 record(issue, "PRIORITY $old $priority")
336 output.add("PRIORITY ${issue.number} $old $priority")
337 }
338 "ASSIGN", "REVIEWER" -> {
339 val issue = issue(words[1])
340 val spec = words[2]
341 val user = user(spec)
342 notDone(issue)
343 val old = if (command == "ASSIGN") issue.assignee else issue.reviewer
344 if (old == user) throw CommandError("no change")
345 if (command == "ASSIGN") {
346 if (user == null && issue.state in setOf("IN_PROGRESS", "REVIEW")) throw CommandError("assignee required")
347 if (issue.state == "IN_PROGRESS" && wip(user) >= 3) throw CommandError("wip limit $spec")
348 if (issue.state == "REVIEW" && user == issue.reviewer) throw CommandError("reviewer is assignee")
349 issue.assignee = user
350 } else {
351 if (user == null && issue.state == "REVIEW") throw CommandError("reviewer required")
352 if (issue.state == "REVIEW" && user == issue.assignee) throw CommandError("reviewer is assignee")
353 issue.reviewer = user
354 }
355 record(issue, "$command $spec")
356 output.add("$command ${issue.number} $spec")
357 }
358 "ESTIMATE" -> {
359 val issue = issue(words[1])
360 val points = words[2].toInt()
361 if (points !in setOf(0, 1, 2, 3, 5, 8, 13, 21)) throw CommandError("bad estimate")
362 notDone(issue)
363 if (issue.estimate == points) throw CommandError("no change")
364 val old = display(issue.estimate)
365 issue.estimate = points
366 record(issue, "ESTIMATE $old $points")
367 output.add("ESTIMATE ${issue.number} $old $points")
368 }
369 "COMMENT" -> {
370 val issue = issue(words[1])
371 user(words[2])
372 issue.comments++
373 record(issue, "COMMENT ${words[2]} ${words.drop(3).joinToString(" ")}")
374 output.add("COMMENT ${issue.number} ${issue.comments}")
375 }
376 "MOVE" -> {
377 val issue = issue(words[1])
378 val target = words[2]
379 if (issue.duplicate != null) throw CommandError("duplicate of ${issue.duplicate}")
380 if (target !in WORKFLOWS.getValue(issue.type)) throw CommandError("bad state")
381 if ("${issue.state}>$target" !in TRANSITIONS.getValue(issue.type)) {
382 throw CommandError("cannot move ${issue.state} to $target")
383 }
384 if (target == "IN_PROGRESS") {
385 val active = activeSprint()
386 if (issue.type == "STORY" && (active == null || issue.sprint != active.name)) throw CommandError("not in active sprint")
387 if (issue.assignee == null) throw CommandError("no assignee")
388 if (wip(issue.assignee) >= 3) throw CommandError("wip limit ${issue.assignee}")
389 }
390 if (target == "REVIEW") {
391 if (issue.reviewer == null) throw CommandError("no reviewer")
392 if (issue.reviewer == issue.assignee) throw CommandError("reviewer is assignee")
393 }
394 if (target in setOf("DONE", "RESOLVED")) {
395 unfinished(issue).firstOrNull()?.let { throw CommandError("blocked by $it") }
396 }
397 if (target == "READY" && issue.estimate == null) throw CommandError("no estimate")
398 changeState(issue, target, output)
399 }
400 "LINK", "UNLINK" -> {
401 val relation = words[2]
402 if (relation !in (if (command == "LINK") setOf("BLOCKS", "DUPLICATES") else setOf("BLOCKS"))) throw CommandError("bad link")
403 val source = issue(words[1])
404 val target = issue(words[3])
405 if (command == "UNLINK") {
406 if (target.number !in source.blocks) throw CommandError("no link")
407 val before = blockedBefore(listOf(target.number))
408 source.blocks.remove(target.number)
409 target.blockers.remove(source.number)
410 record(source, "UNLINK BLOCKS ${target.number}")
411 record(target, "UNLINK BLOCKED_BY ${source.number}")
412 output.add("UNLINK ${source.number} BLOCKS ${target.number}")
413 notify(before, output)
414 } else {
415 if (source === target) throw CommandError("self link")
416 if (relation == "BLOCKS") {
417 if (target.number in source.blocks) throw CommandError("link exists")
418 if (reaches(target.number, source.number)) throw CommandError("cycle")
419 notDone(target)
420 val before = blockedBefore(listOf(target.number))
421 source.blocks.add(target.number)
422 target.blockers.add(source.number)
423 record(source, "LINK BLOCKS ${target.number}")
424 record(target, "LINK BLOCKED_BY ${source.number}")
425 output.add("LINK ${source.number} BLOCKS ${target.number}")
426 notify(before, output)
427 } else {
428 if (source.type != target.type) throw CommandError("type mismatch")
429 notDone(source)
430 if (target.duplicate != null) throw CommandError("target is duplicate")
431 source.duplicate = target.number
432 target.duplicates.add(source.number)
433 record(source, "LINK DUPLICATES ${target.number}")
434 record(target, "LINK DUPLICATED_BY ${source.number}")
435 output.add("LINK ${source.number} DUPLICATES ${target.number}")
436 changeState(source, if (source.type == "BUG") "CLOSED" else "DONE", output)
437 if (source.sprint != null && sprints.getValue(source.sprint!!).phase != "ENDED") {
438 record(source, "SPRINT REMOVE ${source.sprint}")
439 source.sprint = null
440 }
441 }
442 }
443 }
444 "SPRINT" -> sprintCommand(words, output)
445 "QUERY" -> query(words.drop(1).joinToString(" "), output)
446 "SHOW" -> {
447 val issue = issue(words[1])
448 output.add("ISSUE ${issue.number} ${issue.type} ${issue.priority} ${issue.state} blocked ${yesNo(blocked(issue))}")
449 output.add("TITLE ${issue.title}")
450 output.add("PEOPLE assignee ${display(issue.assignee)} reviewer ${display(issue.reviewer)}")
451 output.add("PLAN estimate ${display(issue.estimate)} sprint ${display(issue.sprint)} comments ${issue.comments}")
452 output.add("LINKS blocks ${numberList(issue.blocks)} blocked_by ${numberList(issue.blockers)} " +
453 "duplicate_of ${display(issue.duplicate)} duplicated_by ${numberList(issue.duplicates)}")
454 }
455 "HISTORY" -> {
456 val issue = issue(words[1])
457 output.add("HISTORY ${issue.number} ${issue.history.size}")
458 output.addAll(issue.history)
459 }
460 "BOARD" -> {
461 output.add("BOARD")
462 for (type in TYPES) output.add(type + WORKFLOWS.getValue(type).joinToString("") { state ->
463 " $state ${issues.values.count { it.type == type && it.state == state }}"
464 })
465 val done = issues.values.count { it.done() }
466 output.add("TOTAL ${issues.size} open ${issues.size - done} done $done")
467 }
468 "WORKLOAD" -> {
469 output.add("WORKLOAD ${users.size}")
470 for (user in users.sorted()) {
471 val assigned = issues.values.filter { it.assignee == user && !it.done() }
472 output.add("$user assigned ${assigned.size} in_progress ${assigned.count { it.state == "IN_PROGRESS" }} " +
473 "review ${assigned.count { it.state == "REVIEW" }} reviewing ${issues.values.count { it.reviewer == user && it.state == "REVIEW" }} " +
474 "points ${assigned.sumOf { it.points() }}")
475 }
476 }
477 "SPRINTS" -> {
478 output.add("SPRINTS ${sprints.size}")
479 for (sprint in sprints.values) output.add(when (sprint.phase) {
480 "PLANNED" -> "${sprint.name} PLANNED ${summary(sprint)}"
481 "ACTIVE" -> "${sprint.name} ACTIVE start ${sprint.start} ${summary(sprint)} remaining ${remaining(sprint)}"
482 else -> "${sprint.name} ENDED start ${sprint.start} end ${sprint.end} velocity ${sprint.velocity}"
483 })
484 }
485 "BURNDOWN" -> {
486 val sprint = sprint(words[1])
487 if (sprint.phase == "PLANNED") throw CommandError("sprint not started")
488 val last = if (sprint.phase == "ACTIVE") day else sprint.end
489 output.add("BURNDOWN ${sprint.name} ${sprint.phase} days ${last - sprint.start + 1}")
490 var value = 0
491 for (recordDay in sprint.start..last) {
492 value = sprint.records[recordDay] ?: value
493 output.add("day $recordDay $value")
494 }
495 }
496 }
497 }
498
499 fun execute(line: String): List<String> {
500 val output = mutableListOf<String>()
501 try { process(line, output) } catch (error: CommandError) { output.add("ERROR ${error.message}") }
502 activeSprint()?.let { it.records[day] = remaining(it) }
503 return output
504 }
505}

507fun main() {
508 val input = System.`in`.bufferedReader()
509 val output = System.out.bufferedWriter()
510 val tracker = Tracker()
511 repeat(input.readLine().toInt()) {
512 for (line in tracker.execute(input.readLine())) {
513 output.write(line)
514 output.newLine()
515 }
516 }
517 output.flush()
518}
