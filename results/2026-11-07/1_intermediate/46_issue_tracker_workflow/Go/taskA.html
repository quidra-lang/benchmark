280func (t *Tracker) reaches(start, goal int) bool {
281 pending := []int{start}
282 visited := make(map[int]bool)
283 for len(pending) > 0 {
284 id := pending[len(pending)-1]
285 pending = pending[:len(pending)-1]
286 if id == goal {
287 return true
288 }
289 if visited[id] {
290 continue
291 }
292 visited[id] = true
293 for next := range t.Issues[id].Blocks {
294 pending = append(pending, next)
295 }
296 }
297 return false
298}

311 Child QueryNode

313
314func (n *NotNode) matches(t *Tracker, issue *Issue) bool {
315 return !n.Child.matches(t, issue)
316}

318type BinaryNode struct {
319 Operator string
320 Left, Right QueryNode
321}
322
323func (n *BinaryNode) matches(t *Tracker, issue *Issue) bool {
324 if n.Operator == "AND" {
325 return n.Left.matches(t, issue) && n.Right.matches(t, issue)
326 }
327 return n.Left.matches(t, issue) || n.Right.matches(t, issue)
328}

330func (c *Condition) matches(t *Tracker, issue *Issue) bool {
331 var actualText string
332 actualNumber := 0
333 numeric, none := false, false
334 switch c.Field {
335 case "id":
336 actualNumber, numeric = issue.ID, true
337 case "type":
338 actualText = issue.Kind
339 case "status":
340 actualText = issue.State
341 case "priority":
342 actualNumber, numeric = issue.level(), true
343 case "assignee":
344 actualText, none = issue.Assignee, issue.Assignee == "-"
345 case "reviewer":
346 actualText, none = issue.Reviewer, issue.Reviewer == "-"
347 case "estimate":
348 actualNumber, numeric, none = issue.Estimate, true, issue.Estimate < 0
349 case "sprint":
350 actualText, none = issue.Sprint, issue.Sprint == "-"
351 case "comments":
352 actualNumber, numeric = issue.Comments, true
353 case "blocked":
354 actualText = yesNo(t.blocked(issue))
355 case "done":
356 actualText = yesNo(issue.done())
357 case "title":
358 return strings.Contains(strings.ToLower(issue.Title), c.Text)
359 }
360 equal := none && c.None
361 if !none && !c.None {
362 if numeric {
363 equal = actualNumber == c.Number
364 } else {
365 equal = actualText == c.Text
366 }
367 }
368 switch c.Operator {
369 case "=":
370 return equal
371 case "!=":
372 return !equal
373 }
374 if none {
375 return false
376 }
377 switch c.Operator {
378 case "<":
379 return actualNumber < c.Number
380 case "<=":
381 return actualNumber <= c.Number
382 case ">":
383 return actualNumber > c.Number
384 default:
385 return actualNumber >= c.Number
386 }
387}

389func digits(text string) bool {
390 if len(text) < 1 || len(text) > 6 {
391 return false
392 }
393 for i := range len(text) {
394 if text[i] < '0' || text[i] > '9' {
395 return false
396 }
397 }
398 return true
399}

401func (t *Tracker) condition(token string) (*Condition, error) {
402 position := 0
403 for position < len(token) && token[position] >= 'a' && token[position] <= 'z' {
404 position++
405 }
406 field, rest := token[:position], token[position:]
407 operator := ""
408 for _, candidate := range []string{">=", "<=", "!=", "=", "<", ">", "~"} {
409 if strings.HasPrefix(rest, candidate) {
410 operator = candidate
411 break
412 }
413 }
414 value := rest[len(operator):]
415 if field == "" || operator == "" || value == "" {
416 return nil, fmt.Errorf("syntax near %s", token)
417 }
418 operators, known := fieldOperators[field]
419 if !known {
420 return nil, fmt.Errorf("unknown field %s", field)
421 }
422 if !has(operators, operator) {
423 return nil, fmt.Errorf("bad operator %s", token)
424 }
425 c := &Condition{Field: field, Operator: operator, Text: value}
426 valid := false
427 switch field {
428 case "id", "comments", "estimate":
429 if digits(value) {
430 c.Number = number(value)
431 valid = true
432 } else if field == "estimate" && value == "-" && has(equalityOperators, operator) {
433 c.None, valid = true, true
434 }
435 case "priority":
436 for level, priority := range priorities {
437 if value == priority {
438 c.Number, valid = level, true
439 }
440 }
441 case "type":
442 valid = has(issueTypes, value)
443 case "status":
444 for _, states := range workflows {
445 valid = valid || has(states, value)
446 }
447 case "assignee", "reviewer":
448 c.None = value == "-"
449 valid = c.None || t.Users[value]
450 case "sprint":
451 c.None = value == "-"
452 valid = c.None || t.Sprints[value] != nil
453 case "blocked", "done":
454 valid = value == "yes" || value == "no"
455 case "title":
456 c.Text, valid = strings.ToLower(value), true
457 }
458 if !valid {
459 return nil, fmt.Errorf("bad value %s", token)
460 }
461 return c, nil
462}

464func tokenize(query string) []string {
465 var tokens []string
466 start := 0
467 for position := 0; position < len(query); position++ {
468 c := query[position]
469 if c == ' ' || c == '(' || c == ')' {
470 if start < position {
471 tokens = append(tokens, query[start:position])
472 }
473 if c != ' ' {
474 tokens = append(tokens, string(c))
475 }
476 start = position + 1
477 }
478 }
479 if start < len(query) {
480 tokens = append(tokens, query[start:])
481 }
482 return tokens
483}

485type QueryParser struct {
486 Tokens []string
487 Conditions []QueryNode
488 Position int
489}
490
491func (p *QueryParser) accept(token string) bool {
492 if p.Position < len(p.Tokens) && p.Tokens[p.Position] == token {
493 p.Position++
494 return true
495 }
496 return false
497}
498
499func (p *QueryParser) factor() QueryNode {
500 if p.accept("NOT") {
501 return &NotNode{p.factor()}
502 }
503 if p.accept("(") {
504 node := p.expression()
505 p.accept(")")
506 return node
507 }
508 node := p.Conditions[p.Position]
509 p.Position++
510 return node
511}
512
513func (p *QueryParser) term() QueryNode {
514 node := p.factor()
515 for p.accept("AND") {
516 node = &BinaryNode{"AND", node, p.factor()}
517 }
518 return node
519}
520
521func (p *QueryParser) expression() QueryNode {
522 node := p.term()
523 for p.accept("OR") {
524 node = &BinaryNode{"OR", node, p.term()}
525 }
526 return node
527}

529func (t *Tracker) parseQuery(query string) (QueryNode, error) {
530 tokens := tokenize(query)
531 conditions := make([]QueryNode, len(tokens))
532 operand, depth := true, 0
533 for position, token := range tokens {
534 if operand {
535 switch token {
536 case "NOT":
537 case "(":
538 depth++
539 case "AND", "OR", ")":
540 return nil, fmt.Errorf("syntax near %s", token)
541 default:
542 condition, err := t.condition(token)
543 if err != nil {
544 return nil, err
545 }
546 conditions[position] = condition
547 operand = false
548 }
549 } else {
550 switch token {
551 case "AND", "OR":
552 operand = true
553 case ")":
554 if depth == 0 {
555 return nil, fmt.Errorf("syntax near %s", token)
556 }
557 depth--
558 default:
559 return nil, fmt.Errorf("syntax near %s", token)
560 }
561 }
562 }
563 if operand || depth != 0 {
564 return nil, fmt.Errorf("syntax at end")
565 }
566 parser := &QueryParser{Tokens: tokens, Conditions: conditions}
567 return parser.expression(), nil
568}

570func (t *Tracker) processSprint(args []string) ([]string, error) {
571 command, name := args[0], args[1]
572 if command == "NEW" {
573 if t.Sprints[name] != nil {
574 return nil, fmt.Errorf("sprint exists")
575 }
576 t.Sprints[name] = &Sprint{Name: name, Phase: "PLANNED", Records: make(map[int]int)}
577 t.SprintOrder = append(t.SprintOrder, name)
578 return []string{"SPRINT NEW " + name}, nil
579 }
580 sprint, err := t.requireSprint(name)
581 if err != nil {
582 return nil, err
583 }
584 if command == "ADD" || command == "REMOVE" {
585 issue, err := t.requireIssue(args[2])
586 if err != nil {
587 return nil, err
588 }
589 if sprint.Phase == "ENDED" {
590 return nil, fmt.Errorf("sprint ended")
591 }
592 if command == "ADD" {
593 if issue.done() {
594 return nil, fmt.Errorf("issue done")
595 }
596 if issue.Sprint != "-" && t.Sprints[issue.Sprint].Phase != "ENDED" {
597 return nil, fmt.Errorf("in sprint %s", issue.Sprint)
598 }
599 if issue.State == "BACKLOG" {
600 return nil, fmt.Errorf("not ready")
601 }
602 issue.Sprint = name
603 } else {
604 if issue.Sprint != name {
605 return nil, fmt.Errorf("not in sprint")
606 }
607 if issue.done() {
608 return nil, fmt.Errorf("issue done")
609 }
610 issue.Sprint = "-"
611 }
612 t.record(issue, "SPRINT "+command+" "+name)
613 return []string{fmt.Sprintf("SPRINT %s %s %d %s", command, name, issue.ID, t.summary(sprint))}, nil
614 }
615 if command == "START" {
616 if sprint.Phase != "PLANNED" {
617 return nil, fmt.Errorf("sprint not planned")
618 }
619 if active := t.activeSprint(); active != nil {
620 return nil, fmt.Errorf("sprint active %s", active.Name)
621 }
622 if len(t.members(sprint)) == 0 {
623 return nil, fmt.Errorf("sprint empty")
624 }
625 sprint.Phase, sprint.Start = "ACTIVE", t.Day
626 sprint.Records[t.Day] = t.remaining(sprint)
627 return []string{fmt.Sprintf("SPRINT START %s day %d %s remaining %d", name, t.Day, t.summary(sprint), t.remaining(sprint))}, nil
628 }
629 if sprint.Phase != "ACTIVE" {
630 return nil, fmt.Errorf("sprint not active")
631 }
632 sprint.Records[t.Day] = t.remaining(sprint)
633 var finished, returned []*Issue
634 for _, issue := range t.members(sprint) {
635 if issue.done() {
636 finished = append(finished, issue)
637 } else {
638 returned = append(returned, issue)
639 }
640 }
641 for _, issue := range finished {
642 sprint.Velocity += issue.points()
643 }
644 sprint.Phase, sprint.End = "ENDED", t.Day
645 lines := []string{fmt.Sprintf("SPRINT END %s day %d velocity %d done %d returned %d", name, t.Day, sprint.Velocity, len(finished), len(returned))}
646 for _, issue := range returned {
647 issue.Sprint = "-"
648 t.record(issue, "SPRINT RETURN "+name)
649 lines = append(lines, fmt.Sprintf("RETURNED %d", issue.ID))
650 }
651 return lines, nil
652}

654func (t *Tracker) process(line string) ([]string, error) {
655 words := strings.Split(line, " ")
656 command, args := words[0], words[1:]
657 switch command {
658 case "DAY":
659 day := number(args[0])
660 if day < t.Day {
661 return nil, fmt.Errorf("day backwards")
662 }
663 t.Day = day
664 return []string{fmt.Sprintf("DAY %d", day)}, nil
665 case "USER":
666 user := args[0]
667 if t.Users[user] {
668 return nil, fmt.Errorf("user exists")
669 }
670 t.Users[user] = true
671 return []string{"USER " + user}, nil
672 case "CREATE":
673 id, kind, priority := number(args[0]), args[1], args[2]
674 if t.Issues[id] != nil {
675 return nil, fmt.Errorf("issue exists")
676 }
677 if !has(issueTypes, kind) {
678 return nil, fmt.Errorf("bad type")
679 }
680 if !has(priorities, priority) {
681 return nil, fmt.Errorf("bad priority")
682 }
683 issue := &Issue{
684 ID: id, Kind: kind, Priority: priority, Title: strings.Join(args[3:], " "),
685 State: workflows[kind][0], Assignee: "-", Reviewer: "-", Estimate: -1, Sprint: "-",
686 Blocks: make(map[int]bool), Blockers: make(map[int]bool), Duplicates: make(map[int]bool),
687 }
688 t.Issues[id] = issue
689 t.record(issue, "CREATE "+kind+" "+priority)
690 return []string{fmt.Sprintf("CREATE %d %s %s %s", id, kind, priority, issue.State)}, nil
691 case "SPRINT":
692 return t.processSprint(args)
693 case "QUERY":
694 tree, err := t.parseQuery(strings.Join(args, " "))
695 if err != nil {
696 return nil, err
697 }
698 var found []*Issue
699 for _, issue := range t.Issues {
700 if tree.matches(t, issue) {
701 found = append(found, issue)
702 }
703 }
704 sort.Slice(found, func(i, j int) bool {
705 return found[i].level() > found[j].level() || (found[i].level() == found[j].level() && found[i].ID < found[j].ID)
706 })
707 lines := []string{fmt.Sprintf("QUERY %d", len(found))}
708 for _, issue := range found {
709 lines = append(lines, fmt.Sprintf("%d %s %s %s %s %s %s", issue.ID, issue.Kind, issue.Priority, issue.State, issue.Assignee, estimateText(issue.Estimate), issue.Title))
710 }
711 return lines, nil
712 case "BOARD":
713 lines := []string{"BOARD"}
714 for _, kind := range issueTypes {
715 parts := []string{kind}
716 for _, state := range workflows[kind] {
717 count := 0
718 for _, issue := range t.Issues {
719 if issue.Kind == kind && issue.State == state {
720 count++
721 }
722 }
723 parts = append(parts, state, strconv.Itoa(count))
724 }
725 lines = append(lines, strings.Join(parts, " "))
726 }
727 finished := 0
728 for _, issue := range t.Issues {
729 if issue.done() {
730 finished++
731 }
732 }
733 lines = append(lines, fmt.Sprintf("TOTAL %d open %d done %d", len(t.Issues), len(t.Issues)-finished, finished))
734 return lines, nil
735 case "WORKLOAD":
736 var users []string
737 for user := range t.Users {
738 users = append(users, user)
739 }
740 sort.Strings(users)
741 lines := []string{fmt.Sprintf("WORKLOAD %d", len(users))}
742 for _, user := range users {
743 assigned, progress, review, reviewing, points := 0, 0, 0, 0, 0
744 for _, issue := range t.Issues {
745 if issue.Assignee == user && !issue.done() {
746 assigned++
747 points += issue.points()
748 if issue.State == "IN_PROGRESS" {
749 progress++
750 }
751 if issue.State == "REVIEW" {
752 review++
753 }
754 }
755 if issue.Reviewer == user && issue.State == "REVIEW" {
756 reviewing++
757 }
758 }
759 lines = append(lines, fmt.Sprintf("%s assigned %d in_progress %d review %d reviewing %d points %d", user, assigned, progress, review, reviewing, points))
760 }
761 return lines, nil
762 case "SPRINTS":
763 lines := []string{fmt.Sprintf("SPRINTS %d", len(t.Sprints))}
764 for _, name := range t.SprintOrder {
765 s := t.Sprints[name]
766 switch s.Phase {
767 case "PLANNED":
768 lines = append(lines, fmt.Sprintf("%s PLANNED %s", name, t.summary(s)))
769 case "ACTIVE":
770 lines = append(lines, fmt.Sprintf("%s ACTIVE start %d %s remaining %d", name, s.Start, t.summary(s), t.remaining(s)))
771 case "ENDED":
772 lines = append(lines, fmt.Sprintf("%s ENDED start %d end %d velocity %d", name, s.Start, s.End, s.Velocity))
773 }
774 }
775 return lines, nil
776 case "BURNDOWN":
777 sprint, err := t.requireSprint(args[0])
778 if err != nil {
779 return nil, err
780 }
781 if sprint.Phase == "PLANNED" {
782 return nil, fmt.Errorf("sprint not started")
783 }
784 last := sprint.End
785 if sprint.Phase == "ACTIVE" {
786 last = t.Day
787 }
788 lines := []string{fmt.Sprintf("BURNDOWN %s %s days %d", sprint.Name, sprint.Phase, last-sprint.Start+1)}
789 value := 0
790 for day := sprint.Start; day <= last; day++ {
791 if recorded, ok := sprint.Records[day]; ok {
792 value = recorded
793 }
794 lines = append(lines, fmt.Sprintf("day %d %d", day, value))
795 }
796 return lines, nil
797 case "LINK", "UNLINK":
798 relation := args[1]
799 if relation != "BLOCKS" && (command == "UNLINK" || relation != "DUPLICATES") {
800 return nil, fmt.Errorf("bad link")
801 }
802 a, err := t.requireIssue(args[0])
803 if err != nil {
804 return nil, err
805 }
806 b, err := t.requireIssue(args[2])
807 if err != nil {
808 return nil, err
809 }
810 if command == "UNLINK" {
811 if !a.Blocks[b.ID] {
812 return nil, fmt.Errorf("no link")
813 }
814 before := map[int]bool{b.ID: t.blocked(b)}
815 delete(a.Blocks, b.ID)
816 delete(b.Blockers, a.ID)
817 t.record(a, fmt.Sprintf("UNLINK BLOCKS %d", b.ID))
818 t.record(b, fmt.Sprintf("UNLINK BLOCKED_BY %d", a.ID))
819 return append([]string{fmt.Sprintf("UNLINK %d BLOCKS %d", a.ID, b.ID)}, t.notifications(before)...), nil
820 }
821 if a == b {
822 return nil, fmt.Errorf("self link")
823 }
824 if relation == "BLOCKS" {
825 if a.Blocks[b.ID] {
826 return nil, fmt.Errorf("link exists")
827 }
828 if t.reaches(b.ID, a.ID) {
829 return nil, fmt.Errorf("cycle")
830 }
831 if b.done() {
832 return nil, fmt.Errorf("issue done")
833 }
834 before := map[int]bool{b.ID: t.blocked(b)}
835 a.Blocks[b.ID], b.Blockers[a.ID] = true, true
836 t.record(a, fmt.Sprintf("LINK BLOCKS %d", b.ID))
837 t.record(b, fmt.Sprintf("LINK BLOCKED_BY %d", a.ID))
838 return append([]string{fmt.Sprintf("LINK %d BLOCKS %d", a.ID, b.ID)}, t.notifications(before)...), nil
839 }
840 if a.Kind != b.Kind {
841 return nil, fmt.Errorf("type mismatch")
842 }
843 if a.done() {
844 return nil, fmt.Errorf("issue done")
845 }
846 if b.Duplicate != 0 {
847 return nil, fmt.Errorf("target is duplicate")
848 }
849 a.Duplicate, b.Duplicates[a.ID] = b.ID, true
850 t.record(a, fmt.Sprintf("LINK DUPLICATES %d", b.ID))
851 t.record(b, fmt.Sprintf("LINK DUPLICATED_BY %d", a.ID))
852 target := "DONE"
853 if a.Kind == "BUG" {
854 target = "CLOSED"
855 }
856 lines := append([]string{fmt.Sprintf("LINK %d DUPLICATES %d", a.ID, b.ID)}, t.changeState(a, target)...)
857 if a.Sprint != "-" && t.Sprints[a.Sprint].Phase != "ENDED" {
858 t.record(a, "SPRINT REMOVE "+a.Sprint)
859 a.Sprint = "-"
860 }
861 return lines, nil
862 }
863 issue, err := t.requireIssue(args[0])
864 if err != nil {
865 return nil, err
866 }
867 switch command {
868 case "SHOW":
869 duplicate := "-"
870 if issue.Duplicate != 0 {
871 duplicate = strconv.Itoa(issue.Duplicate)
872 }
873 return []string{
874 fmt.Sprintf("ISSUE %d %s %s %s blocked %s", issue.ID, issue.Kind, issue.Priority, issue.State, yesNo(t.blocked(issue))),
875 "TITLE " + issue.Title,
876 fmt.Sprintf("PEOPLE assignee %s reviewer %s", issue.Assignee, issue.Reviewer),
877 fmt.Sprintf("PLAN estimate %s sprint %s comments %d", estimateText(issue.Estimate), issue.Sprint, issue.Comments),
878 fmt.Sprintf("LINKS blocks %s blocked_by %s duplicate_of %s duplicated_by %s", numberList(issue.Blocks), numberList(issue.Blockers), duplicate, numberList(issue.Duplicates)),
879 }, nil
880 case "HISTORY":
881 return append([]string{fmt.Sprintf("HISTORY %d %d", issue.ID, len(issue.History))}, issue.History...), nil
882 case "COMMENT":
883 if err := t.requireUser(args[1]); err != nil {
884 return nil, err
885 }
886 issue.Comments++
887 t.record(issue, "COMMENT "+args[1]+" "+strings.Join(args[2:], " "))
888 return []string{fmt.Sprintf("COMMENT %d %d", issue.ID, issue.Comments)}, nil
889 case "PRIORITY":
890 target := args[1]
891 if !has(priorities, target) {
892 return nil, fmt.Errorf("bad priority")
893 }
894 if issue.done() {
895 return nil, fmt.Errorf("issue done")
896 }
897 if target == issue.Priority {
898 return nil, fmt.Errorf("no change")
899 }
900 previous := issue.Priority
901 issue.Priority = target
902 t.record(issue, "PRIORITY "+previous+" "+target)
903 return []string{fmt.Sprintf("PRIORITY %d %s %s", issue.ID, previous, target)}, nil
904 case "ASSIGN", "REVIEWER":
905 target := args[1]
906 if err := t.requireUser(target); err != nil {
907 return nil, err
908 }
909 if issue.done() {
910 return nil, fmt.Errorf("issue done")
911 }
912 current := issue.Reviewer
913 if command == "ASSIGN" {
914 current = issue.Assignee
915 }
916 if target == current {
917 return nil, fmt.Errorf("no change")
918 }
919 if command == "ASSIGN" {
920 if target == "-" && (issue.State == "IN_PROGRESS" || issue.State == "REVIEW") {
921 return nil, fmt.Errorf("assignee required")
922 }
923 if issue.State == "IN_PROGRESS" && t.wipCount(target) >= 3 {
924 return nil, fmt.Errorf("wip limit %s", target)
925 }
926 if issue.State == "REVIEW" && target == issue.Reviewer {
927 return nil, fmt.Errorf("reviewer is assignee")
928 }
929 issue.Assignee = target
930 } else {
931 if target == "-" && issue.State == "REVIEW" {
932 return nil, fmt.Errorf("reviewer required")
933 }
934 if issue.State == "REVIEW" && target == issue.Assignee {
935 return nil, fmt.Errorf("reviewer is assignee")
936 }
937 issue.Reviewer = target
938 }
939 t.record(issue, command+" "+target)
940 return []string{fmt.Sprintf("%s %d %s", command, issue.ID, target)}, nil
941 case "ESTIMATE":
942 value := number(args[1])
943 if !estimates[value] {
944 return nil, fmt.Errorf("bad estimate")
945 }
946 if issue.done() {
947 return nil, fmt.Errorf("issue done")
948 }
949 if issue.Estimate == value {
950 return nil, fmt.Errorf("no change")
951 }
952 previous := estimateText(issue.Estimate)
953 issue.Estimate = value
954 t.record(issue, fmt.Sprintf("ESTIMATE %s %d", previous, value))
955 return []string{fmt.Sprintf("ESTIMATE %d %s %d", issue.ID, previous, value)}, nil
956 case "MOVE":
957 target := args[1]
958 if issue.Duplicate != 0 {
959 return nil, fmt.Errorf("duplicate of %d", issue.Duplicate)
960 }
961 if !has(workflows[issue.Kind], target) {
962 return nil, fmt.Errorf("bad state")
963 }
964 if !has(transitions[issue.Kind], issue.State+" "+target) {
965 return nil, fmt.Errorf("cannot move %s to %s", issue.State, target)
966 }
967 if target == "IN_PROGRESS" {
968 active := t.activeSprint()
969 if issue.Kind == "STORY" && (active == nil || issue.Sprint != active.Name) {
970 return nil, fmt.Errorf("not in active sprint")
971 }
972 if issue.Assignee == "-" {
973 return nil, fmt.Errorf("no assignee")
974 }
975 if t.wipCount(issue.Assignee) >= 3 {
976 return nil, fmt.Errorf("wip limit %s", issue.Assignee)
977 }
978 }
979 if target == "REVIEW" {
980 if issue.Reviewer == "-" {
981 return nil, fmt.Errorf("no reviewer")
982 }
983 if issue.Reviewer == issue.Assignee {
984 return nil, fmt.Errorf("reviewer is assignee")
985 }
986 }
987 if target == "DONE" || target == "RESOLVED" {
988 if blockers := t.unfinishedBlockers(issue); len(blockers) > 0 {
989 return nil, fmt.Errorf("blocked by %d", blockers[0])
990 }
991 }
992 if target == "READY" && issue.Estimate < 0 {
993 return nil, fmt.Errorf("no estimate")
994 }
995 return t.changeState(issue, target), nil
996 }
997 panic("unknown command")
998}

1000func (t *Tracker) execute(line string) []string {
1001 lines, err := t.process(line)
1002 if err != nil {
1003 lines = []string{"ERROR " + err.Error()}
1004 }
1005 if active := t.activeSprint(); active != nil {
1006 active.Records[t.Day] = t.remaining(active)
1007 }
1008 return lines
1009}
