3PRIORITIES = %w[LOW MEDIUM HIGH CRITICAL].freeze
4WORKFLOWS = {
5 'BUG' => %w[OPEN TRIAGED IN_PROGRESS REVIEW RESOLVED CLOSED REOPENED],
6 'TASK' => %w[OPEN IN_PROGRESS DONE],
7 'STORY' => %w[BACKLOG READY IN_PROGRESS REVIEW DONE]
8}.freeze
9TRANSITIONS = {
10 'BUG' => %w[OPEN:TRIAGED TRIAGED:IN_PROGRESS IN_PROGRESS:REVIEW REVIEW:IN_PROGRESS
11 REVIEW:RESOLVED RESOLVED:CLOSED RESOLVED:REOPENED CLOSED:REOPENED REOPENED:IN_PROGRESS],
12 'TASK' => %w[OPEN:IN_PROGRESS IN_PROGRESS:OPEN IN_PROGRESS:DONE],
13 'STORY' => %w[BACKLOG:READY READY:BACKLOG READY:IN_PROGRESS IN_PROGRESS:REVIEW REVIEW:IN_PROGRESS REVIEW:DONE]
14}.freeze
15DONE_STATES = %w[RESOLVED CLOSED DONE].freeze
16ESTIMATES = [0, 1, 2, 3, 5, 8, 13, 21].freeze
17EQUALITY = %w[= !=].freeze
18ORDERED = %w[= != < <= > >=].freeze
19FIELD_OPERATORS = {
20 'id' => ORDERED, 'type' => EQUALITY, 'status' => EQUALITY, 'priority' => ORDERED,
21 'assignee' => EQUALITY, 'reviewer' => EQUALITY, 'estimate' => ORDERED,
22 'sprint' => EQUALITY, 'comments' => ORDERED, 'blocked' => EQUALITY,
23 'done' => EQUALITY, 'title' => ['~']
24}.freeze

74class Tracker
75 def initialize
76 @day = 1
77 @users = Set.new
78 @issues = {}
79 @sprints = {}
80 end

82 def execute(line)
83 @output = []
84 command, *args = line.split(' ')
85 begin
86 process(command, args)
87 rescue CommandError => error
88 emit('ERROR', error.message)
89 end
90 active = active_sprint
91 active.records[@day] = remaining(active) if active
92 @output
93 end
94
95 def emit(*parts)
96 @output << parts.join(' ')
97 end

107 spec

118 def ordered_issues
119 @issues.values.sort_by(&:number)

122 def unfinished(issue)
123 issue.blockers.sort.reject { |number| @issues[number].done? }
124 end
125
126 def blocked?(issue)
127 !issue.done? && !unfinished(issue).empty?
128 end
129
130 def wip(user)
131 @issues.values.count { |issue| issue.assignee == user && issue.state == 'IN_PROGRESS' }
132 end

134 def active_sprint
135 @sprints.values.find { |sprint| sprint.phase == 'ACTIVE' }
136 end

138 def members(sprint)
139 ordered_issues.select { |issue| issue.sprint == sprint.name }
140 end

142 def remaining(sprint)
143 members(sprint).reject(&:done?).sum(&:points)
144 end

146 def summary(sprint)
147 issues = members(sprint)
148 "issues #{issues.length} points #{issues.sum(&:points)}"
149 end

151 def record(issue, text)
152 issue.history << "day #{@day} #{text}"

155 def blocked_states(numbers)
156 numbers.to_h { |number| [number, blocked?(@issues[number])] }
157 end
158
159 def notify(before)
160 before.keys.sort.each do |number|
161 after = blocked?(@issues[number])
162 emit(after ? 'BLOCKED' : 'UNBLOCKED', number) if before[number] != after
163 end
164 end

166 def change_state(issue, target)
167 before = blocked_states(issue.blocks)
168 emit('MOVE', issue.number, issue.state, target)
169 record(issue, "MOVE #{issue.state} #{target}")
170 issue.state = target
171 notify(before)
172 end

174 def reaches?(start, goal)
175 pending = [start.number]
176 visited = Set.new
177 until pending.empty?
178 number = pending.pop
179 return true if number == goal.number
180 next if visited.include?(number)
181
182 visited << number
183 pending.concat(@issues[number].blocks.to_a)
184 end
185 false
186 end

188 def process(command, args)
189 case command
190 when 'DAY'
191 day = args[0].to_i
192 raise CommandError, 'day backwards' if day < @day
193
194 @day = day
195 emit('DAY', day)
196 when 'USER'
197 raise CommandError, 'user exists' if @users.include?(args[0])
198
199 @users << args[0]
200 emit('USER', args[0])
201 when 'CREATE'
202 number, kind, priority = args[0].to_i, args[1], args[2]
203 raise CommandError, 'issue exists' if @issues.key?(number)
204 raise CommandError, 'bad type' unless WORKFLOWS.key?(kind)
205 raise CommandError, 'bad priority' unless PRIORITIES.include?(priority)
206
207 created = Issue.new(number, kind, priority, args.drop(3).join(' '))
208 @issues[number] = created
209 record(created, "CREATE #{kind} #{priority}")
210 emit('CREATE', number, kind, priority, created.state)
211 when 'PRIORITY'
212 item = issue(args[0])
213 priority = args[1]
214 raise CommandError, 'bad priority' unless PRIORITIES.include?(priority)
215 not_done(item)
216 raise CommandError, 'no change' if item.priority == priority
217
218 old = item.priority
219 item.priority = priority
220 record(item, "PRIORITY #{old} #{priority}")
221 emit('PRIORITY', item.number, old, priority)
222 when 'ASSIGN', 'REVIEWER'
223 set_person(command, args)
224 when 'ESTIMATE'
225 item = issue(args[0])
226 points = args[1].to_i
227 raise CommandError, 'bad estimate' unless ESTIMATES.include?(points)
228 not_done(item)
229 raise CommandError, 'no change' if item.estimate == points
230
231 old = display(item.estimate)
232 item.estimate = points
233 record(item, "ESTIMATE #{old} #{points}")
234 emit('ESTIMATE', item.number, old, points)
235 when 'COMMENT'
236 item = issue(args[0])
237 author = user(args[1])
238 item.comments += 1
239 record(item, "COMMENT #{author} #{args.drop(2).join(' ')}")
240 emit('COMMENT', item.number, item.comments)
241 when 'MOVE' then move(args)
242 when 'LINK', 'UNLINK' then link(command, args)
243 when 'SPRINT' then sprint_command(args)
244 when 'QUERY' then query(args.join(' '))
245 when 'SHOW' then show(issue(args[0]))
246 when 'HISTORY'
247 item = issue(args[0])
248 emit('HISTORY', item.number, item.history.length)
249 @output.concat(item.history)
250 when 'BOARD' then board
251 when 'WORKLOAD' then workload
252 when 'SPRINTS' then list_sprints
253 when 'BURNDOWN' then burndown(sprint(args[0]))
254 end
255 end

257 def set_person(command, args)
258 item = issue(args[0])
259 chosen = user(args[1])
260 not_done(item)
261 if command == 'ASSIGN'
262 raise CommandError, 'no change' if item.assignee == chosen
263 if chosen.nil? && %w[IN_PROGRESS REVIEW].include?(item.state)
264 raise CommandError, 'assignee required'
265 end
266 raise CommandError, "wip limit #{args[1]}" if item.state == 'IN_PROGRESS' && wip(chosen) >= 3
267 if item.state == 'REVIEW' && chosen == item.reviewer
268 raise CommandError, 'reviewer is assignee'
269 end
270 item.assignee = chosen
271 else
272 raise CommandError, 'no change' if item.reviewer == chosen
273 raise CommandError, 'reviewer required' if chosen.nil? && item.state == 'REVIEW'
274 if item.state == 'REVIEW' && chosen == item.assignee
275 raise CommandError, 'reviewer is assignee'
276 end
277 item.reviewer = chosen
278 end
279 record(item, "#{command} #{args[1]}")
280 emit(command, item.number, args[1])
281 end

283 def move(args)
284 item = issue(args[0])
285 target = args[1]
286 raise CommandError, "duplicate of #{item.duplicate}" if item.duplicate
287 raise CommandError, 'bad state' unless WORKFLOWS[item.kind].include?(target)
288 unless TRANSITIONS[item.kind].include?("#{item.state}:#{target}")
289 raise CommandError, "cannot move #{item.state} to #{target}"
290 end
291 if target == 'IN_PROGRESS'
292 active = active_sprint
293 if item.kind == 'STORY' && (active.nil? || item.sprint != active.name)
294 raise CommandError, 'not in active sprint'
295 end
296 raise CommandError, 'no assignee' if item.assignee.nil?
297 raise CommandError, "wip limit #{item.assignee}" if wip(item.assignee) >= 3
298 elsif target == 'REVIEW'
299 raise CommandError, 'no reviewer' if item.reviewer.nil?
300 raise CommandError, 'reviewer is assignee' if item.reviewer == item.assignee
301 end
302 blockers = unfinished(item)
303 if %w[DONE RESOLVED].include?(target) && !blockers.empty?
304 raise CommandError, "blocked by #{blockers.first}"
305 end
306 raise CommandError, 'no estimate' if target == 'READY' && item.estimate.nil?
307
308 change_state(item, target)
309 end

311 def link(command, args)
312 relation = args[1]
313 allowed = command == 'LINK' ? %w[BLOCKS DUPLICATES] : ['BLOCKS']
314 raise CommandError, 'bad link' unless allowed.include?(relation)
315
316 source, target = issue(args[0]), issue(args[2])
317 if command == 'UNLINK'
318 raise CommandError, 'no link' unless source.blocks.include?(target.number)
319
320 before = blocked_states([target.number])
321 source.blocks.delete(target.number)
322 target.blockers.delete(source.number)
323 record(source, "UNLINK BLOCKS #{target.number}")
324 record(target, "UNLINK BLOCKED_BY #{source.number}")
325 emit('UNLINK', source.number, 'BLOCKS', target.number)
326 notify(before)
327 return
328 end
329 raise CommandError, 'self link' if source == target
330 if relation == 'BLOCKS'
331 raise CommandError, 'link exists' if source.blocks.include?(target.number)
332 raise CommandError, 'cycle' if reaches?(target, source)
333 not_done(target)
334 before = blocked_states([target.number])
335 source.blocks << target.number
336 target.blockers << source.number
337 record(source, "LINK BLOCKS #{target.number}")
338 record(target, "LINK BLOCKED_BY #{source.number}")
339 emit('LINK', source.number, relation, target.number)
340 notify(before)
341 else
342 raise CommandError, 'type mismatch' if source.kind != target.kind
343 not_done(source)
344 raise CommandError, 'target is duplicate' if target.duplicate
345
346 source.duplicate = target.number
347 target.duplicates << source.number
348 record(source, "LINK DUPLICATES #{target.number}")
349 record(target, "LINK DUPLICATED_BY #{source.number}")
350 emit('LINK', source.number, relation, target.number)
351 change_state(source, source.kind == 'BUG' ? 'CLOSED' : 'DONE')
352 if source.sprint && @sprints[source.sprint].phase != 'ENDED'
353 record(source, "SPRINT REMOVE #{source.sprint}")
354 source.sprint = nil
355 end
356 end
357 end

359 def sprint_command(args)
360 action, name = args[0], args[1]
361 if action == 'NEW'
362 raise CommandError, 'sprint exists' if @sprints.key?(name)
363
364 @sprints[name] = Sprint.new(name)
365 emit('SPRINT', 'NEW', name)
366 return
367 end
368 plan = sprint(name)
369 case action
370 when 'ADD', 'REMOVE'
371 item = issue(args[2])
372 raise CommandError, 'sprint ended' if plan.phase == 'ENDED'
373 if action == 'ADD'
374 not_done(item)
375 if item.sprint && @sprints[item.sprint].phase != 'ENDED'
376 raise CommandError, "in sprint #{item.sprint}"
377 end
378 raise CommandError, 'not ready' if item.state == 'BACKLOG'
379
380 item.sprint = name
381 else
382 raise CommandError, 'not in sprint' if item.sprint != name
383 not_done(item)
384 item.sprint = nil
385 end
386 record(item, "SPRINT #{action} #{name}")
387 emit('SPRINT', action, name, item.number, summary(plan))
388 when 'START'
389 raise CommandError, 'sprint not planned' if plan.phase != 'PLANNED'
390 active = active_sprint
391 raise CommandError, "sprint active #{active.name}" if active
392 raise CommandError, 'sprint empty' if members(plan).empty?
393
394 plan.phase, plan.start = 'ACTIVE', @day
395 plan.records[@day] = remaining(plan)
396 emit('SPRINT', 'START', name, 'day', @day, summary(plan), 'remaining', remaining(plan))
397 when 'END'
398 raise CommandError, 'sprint not active' if plan.phase != 'ACTIVE'
399
400 plan.records[@day] = remaining(plan)
401 finished, returned = members(plan).partition(&:done?)
402 plan.velocity = finished.sum(&:points)
403 plan.phase, plan.finish = 'ENDED', @day
404 emit('SPRINT', 'END', name, 'day', @day, 'velocity', plan.velocity,
405 'done', finished.length, 'returned', returned.length)
406 returned.each do |item|
407 item.sprint = nil
408 record(item, "SPRINT RETURN #{name}")
409 emit('RETURNED', item.number)
410 end
411 end
412 end

414 def condition(token)
415 match = /\A([a-z]*)(>=|<=|!=|=|<|>|~)(.*)\z/.match(token)
416 unless match && !match[1].empty? && !match[3].empty?
417 raise CommandError, "syntax near #{token}"
418 end
419 field, operator, text = match.captures
420 raise CommandError, "unknown field #{field}" unless FIELD_OPERATORS.key?(field)
421 raise CommandError, "bad operator #{token}" unless FIELD_OPERATORS[field].include?(operator)
422
423 value = if text == '-' && %w[assignee reviewer estimate sprint].include?(field) && EQUALITY.include?(operator)
424 nil
425 elsif %w[id estimate comments].include?(field) && /\A[0-9]{1,6}\z/.match?(text)
426 text.to_i
427 elsif field == 'priority' && PRIORITIES.include?(text)
428 PRIORITIES.index(text)
429 elsif field == 'type' && WORKFLOWS.key?(text)
430 text
431 elsif field == 'status' && WORKFLOWS.values.flatten.include?(text)
432 text
433 elsif %w[assignee reviewer].include?(field) && @users.include?(text)
434 text
435 elsif field == 'sprint' && @sprints.key?(text)
436 text
437 elsif %w[blocked done].include?(field) && %w[yes no].include?(text)
438 text
439 elsif field == 'title'
440 text.downcase
441 else
442 raise CommandError, "bad value #{token}"
443 end
444 [field, operator, value]
445 end

447 def compile_query(text)
448 tokens = text.scan(/[()]|[^ ()]+/)
449 conditions = {}
450 operand, depth = true, 0
451 tokens.each_with_index do |token, index|
452 if operand
453 if token == '('
454 depth += 1
455 elsif %w[AND OR )].include?(token)
456 raise CommandError, "syntax near #{token}"
457 elsif token != 'NOT'
458 conditions[index] = condition(token)
459 operand = false
460 end
461 elsif %w[AND OR].include?(token)
462 operand = true
463 elsif token == ')' && depth.positive?
464 depth -= 1
465 else
466 raise CommandError, "syntax near #{token}"
467 end
468 end
469 raise CommandError, 'syntax at end' if operand || !depth.zero?
470
471 output, operators = [], []
472 precedence = { 'OR' => 1, 'AND' => 2, 'NOT' => 3 }
473 tokens.each_with_index do |token, index|
474 if conditions.key?(index)
475 output << conditions[index]
476 elsif token == '(' || token == 'NOT'
477 operators << token
478 elsif token == ')'
479 output << operators.pop until operators.last == '('
480 operators.pop
481 else
482 while !operators.empty? && operators.last != '(' && precedence[operators.last] >= precedence[token]
483 output << operators.pop
484 end
485 operators << token
486 end
487 end
488 output.concat(operators.reverse)
489 end

491 def field_value(item, field)
492 case field
493 when 'id' then item.number
494 when 'type' then item.kind
495 when 'status' then item.state
496 when 'priority' then item.level
497 when 'blocked' then yes_no(blocked?(item))
498 when 'done' then yes_no(item.done?)
499 when 'title' then item.title.downcase
500 else item.public_send(field)
501 end
502 end
503
504 def matches?(item, program)
505 stack = []
506 program.each do |node|
507 if node == 'NOT'
508 stack << !stack.pop
509 elsif node == 'AND' || node == 'OR'
510 right, left = stack.pop, stack.pop
511 stack << (node == 'AND' ? left && right : left || right)
512 else
513 field, operator, expected = node
514 actual = field_value(item, field)
515 result = case operator
516 when '=' then actual == expected
517 when '!=' then actual != expected
518 when '~' then actual.include?(expected)
519 else
520 !actual.nil? && case operator
521 when '<' then actual < expected
522 when '<=' then actual <= expected
523 when '>' then actual > expected
524 when '>=' then actual >= expected
525 end
526 end
527 stack << result
528 end
529 end
530 stack.last
531 end

533 def query(text)
534 program = compile_query(text)
535 matches = @issues.values.select { |item| matches?(item, program) }.sort_by { |item| [-item.level, item.number] }
536 emit('QUERY', matches.length)
537 matches.each do |item|
538 emit(item.number, item.kind, item.priority, item.state, display(item.assignee), display(item.estimate), item.title)
539 end
540 end

542 def show(item)
543 emit('ISSUE', item.number, item.kind, item.priority, item.state, 'blocked', yes_no(blocked?(item)))
544 emit('TITLE', item.title)
545 emit('PEOPLE', 'assignee', display(item.assignee), 'reviewer', display(item.reviewer))
546 emit('PLAN', 'estimate', display(item.estimate), 'sprint', display(item.sprint), 'comments', item.comments)
547 emit('LINKS', 'blocks', number_list(item.blocks), 'blocked_by', number_list(item.blockers),
548 'duplicate_of', display(item.duplicate), 'duplicated_by', number_list(item.duplicates))

551 def board
552 emit('BOARD')
553 WORKFLOWS.each do |kind, states|
554 parts = [kind]
555 states.each { |state| parts.concat([state, @issues.values.count { |item| item.kind == kind && item.state == state }]) }
556 emit(*parts)
557 end
558 finished = @issues.values.count(&:done?)
559 emit('TOTAL', @issues.length, 'open', @issues.length - finished, 'done', finished)
560 end

562 def workload
563 emit('WORKLOAD', @users.length)
564 @users.sort.each do |name|
565 assigned = @issues.values.select { |item| item.assignee == name && !item.done? }
566 emit(name, 'assigned', assigned.length, 'in_progress', assigned.count { |item| item.state == 'IN_PROGRESS' },
567 'review', assigned.count { |item| item.state == 'REVIEW' }, 'reviewing',
568 @issues.values.count { |item| item.reviewer == name && item.state == 'REVIEW' }, 'points', assigned.sum(&:points))
569 end
570 end

572 def list_sprints
573 emit('SPRINTS', @sprints.length)
574 @sprints.each_value do |plan|
575 case plan.phase
576 when 'PLANNED' then emit(plan.name, plan.phase, summary(plan))
577 when 'ACTIVE' then emit(plan.name, plan.phase, 'start', plan.start, summary(plan), 'remaining', remaining(plan))
578 else emit(plan.name, plan.phase, 'start', plan.start, 'end', plan.finish, 'velocity', plan.velocity)
579 end
580 end
581 end

583 def burndown(plan)
584 raise CommandError, 'sprint not started' if plan.phase == 'PLANNED'
585
586 last = plan.phase == 'ACTIVE' ? @day : plan.finish
587 emit('BURNDOWN', plan.name, plan.phase, 'days', last - plan.start + 1)
588 value = 0
589 (plan.start..last).each do |day|
590 value = plan.records[day] if plan.records.key?(day)
591 emit('day', day, value)
592 end
593 end
