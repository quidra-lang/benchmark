  def record(issue, text)
    issue.history << "day #{@day} #{text}"
    when 'HISTORY'
      item = issue(args[0])
      emit('HISTORY', item.number, item.history.length)
      @output.concat(item.history)
    if command == 'ASSIGN'
      item.assignee = chosen
    record(item, "#{command} #{args[1]}")
