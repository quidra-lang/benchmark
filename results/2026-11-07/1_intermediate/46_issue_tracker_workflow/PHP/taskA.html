4const PRIORITIES = ['LOW', 'MEDIUM', 'HIGH', 'CRITICAL'];
5const WORKFLOWS = [
6 'BUG' => ['OPEN', 'TRIAGED', 'IN_PROGRESS', 'REVIEW', 'RESOLVED', 'CLOSED', 'REOPENED'],
7 'TASK' => ['OPEN', 'IN_PROGRESS', 'DONE'],
8 'STORY' => ['BACKLOG', 'READY', 'IN_PROGRESS', 'REVIEW', 'DONE'],
9];
10const TRANSITIONS = [
11 'BUG' => ['OPEN:TRIAGED', 'TRIAGED:IN_PROGRESS', 'IN_PROGRESS:REVIEW', 'REVIEW:IN_PROGRESS',
12 'REVIEW:RESOLVED', 'RESOLVED:CLOSED', 'RESOLVED:REOPENED', 'CLOSED:REOPENED', 'REOPENED:IN_PROGRESS'],
13 'TASK' => ['OPEN:IN_PROGRESS', 'IN_PROGRESS:OPEN', 'IN_PROGRESS:DONE'],
14 'STORY' => ['BACKLOG:READY', 'READY:BACKLOG', 'READY:IN_PROGRESS', 'IN_PROGRESS:REVIEW', 'REVIEW:IN_PROGRESS', 'REVIEW:DONE'],
15];
16const DONE_STATES = ['RESOLVED', 'CLOSED', 'DONE'];
17const ESTIMATES = [0, 1, 2, 3, 5, 8, 13, 21];
18const EQUALITY = ['=', '!='];
19const ORDERED = ['=', '!=', '<', '<=', '>', '>='];
20const FIELD_OPERATORS = [
21 'id' => ORDERED, 'type' => EQUALITY, 'status' => EQUALITY, 'priority' => ORDERED,
22 'assignee' => EQUALITY, 'reviewer' => EQUALITY, 'estimate' => ORDERED, 'sprint' => EQUALITY,
23 'comments' => ORDERED, 'blocked' => EQUALITY, 'done' => EQUALITY, 'title' => ['~'],
24];

99 public function execute(string $line): array
100 {
101 $this->output = [];
102 $args = explode(' ', $line);
103 $command = array_shift($args);
104 try {
105 $this->process($command, $args);
106 } catch (CommandError $error) {
107 $this->emit('ERROR', $error->getMessage());
108 }
109 $active = $this->activeSprint();
110 if ($active !== null) {
111 $active->records[$this->day] = $this->remaining($active);
112 }
113 return $this->output;
114 }

228 $this->emit('MOVE', $issue->number, $issue->state, $target);

234 private function reaches(Issue $start, Issue $goal): bool
235 {
236 $pending = [$start->number];
237 $visited = [];
238 while ($pending !== []) {
239 $number = array_pop($pending);
240 if ($number === $goal->number) {
241 return true;
242 }
243 if (isset($visited[$number])) {
244 continue;
245 }
246 $visited[$number] = true;
247 array_push($pending, ...array_keys($this->issues[$number]->blocks));
248 }
249 return false;
250 }

252 private function process(string $command, array $args): void
253 {
254 switch ($command) {
255 case 'DAY':
256 $day = (int)$args[0];
257 if ($day < $this->day) {
258 throw new CommandError('day backwards');
259 }
260 $this->day = $day;
261 $this->emit('DAY', $day);
262 break;
263 case 'USER':
264 if (isset($this->users[$args[0]])) {
265 throw new CommandError('user exists');
266 }
267 $this->users[$args[0]] = true;
268 $this->emit('USER', $args[0]);
269 break;
270 case 'CREATE':
271 [$text, $kind, $priority] = $args;
272 $number = (int)$text;
273 if (isset($this->issues[$number])) {
274 throw new CommandError('issue exists');
275 }
276 if (!array_key_exists($kind, WORKFLOWS)) {
277 throw new CommandError('bad type');
278 }
279 if (!in_array($priority, PRIORITIES, true)) {
280 throw new CommandError('bad priority');
281 }
282 $issue = new Issue($number, $kind, $priority, implode(' ', array_slice($args, 3)));
283 $this->issues[$number] = $issue;
284 $this->record($issue, "CREATE $kind $priority");
285 $this->emit('CREATE', $number, $kind, $priority, $issue->state);
286 break;
287 case 'PRIORITY':
288 $issue = $this->issue($args[0]);
289 $priority = $args[1];
290 if (!in_array($priority, PRIORITIES, true)) {
291 throw new CommandError('bad priority');
292 }
293 $this->notDone($issue);
294 if ($issue->priority === $priority) {
295 throw new CommandError('no change');
296 }
297 $old = $issue->priority;
298 $issue->priority = $priority;
299 $this->record($issue, "PRIORITY $old $priority");
300 $this->emit('PRIORITY', $issue->number, $old, $priority);
301 break;
302 case 'ASSIGN': case 'REVIEWER':
303 $this->setPerson($command, $args);
304 break;
305 case 'ESTIMATE':
306 $issue = $this->issue($args[0]);
307 $points = (int)$args[1];
308 if (!in_array($points, ESTIMATES, true)) {
309 throw new CommandError('bad estimate');
310 }
311 $this->notDone($issue);
312 if ($issue->estimate === $points) {
313 throw new CommandError('no change');
314 }
315 $old = display($issue->estimate);
316 $issue->estimate = $points;
317 $this->record($issue, "ESTIMATE $old $points");
318 $this->emit('ESTIMATE', $issue->number, $old, $points);
319 break;
320 case 'COMMENT':
321 $issue = $this->issue($args[0]);
322 $user = $this->user($args[1]);
323 ++$issue->comments;
324 $this->record($issue, "COMMENT $user " . implode(' ', array_slice($args, 2)));
325 $this->emit('COMMENT', $issue->number, $issue->comments);
326 break;
327 case 'MOVE': $this->move($args); break;
328 case 'LINK': case 'UNLINK': $this->link($command, $args); break;
329 case 'SPRINT': $this->sprintCommand($args); break;
330 case 'QUERY': $this->query(implode(' ', $args)); break;
331 case 'SHOW': $this->show($this->issue($args[0])); break;
332 case 'HISTORY':
333 $issue = $this->issue($args[0]);
334 $this->emit('HISTORY', $issue->number, count($issue->history));
335 array_push($this->output, ...$issue->history);
336 break;
337 case 'BOARD': $this->board(); break;
338 case 'WORKLOAD': $this->workload(); break;
339 case 'SPRINTS': $this->listSprints(); break;
340 case 'BURNDOWN': $this->burndown($this->sprint($args[0])); break;
341 }
342 }

344 private function setPerson(string $command, array $args): void
345 {
346 $issue = $this->issue($args[0]);
347 $chosen = $this->user($args[1]);
348 $this->notDone($issue);
349 if ($command === 'ASSIGN') {
350 if ($issue->assignee === $chosen) {
351 throw new CommandError('no change');
352 }
353 if ($chosen === null && in_array($issue->state, ['IN_PROGRESS', 'REVIEW'], true)) {
354 throw new CommandError('assignee required');
355 }
356 if ($issue->state === 'IN_PROGRESS' && $this->wip($chosen) >= 3) {
357 throw new CommandError("wip limit {$args[1]}");
358 }
359 if ($issue->state === 'REVIEW' && $chosen === $issue->reviewer) {
360 throw new CommandError('reviewer is assignee');
361 }
362 $issue->assignee = $chosen;
363 } else {
364 if ($issue->reviewer === $chosen) {
365 throw new CommandError('no change');
366 }
367 if ($chosen === null && $issue->state === 'REVIEW') {
368 throw new CommandError('reviewer required');
369 }
370 if ($issue->state === 'REVIEW' && $chosen === $issue->assignee) {
371 throw new CommandError('reviewer is assignee');
372 }
373 $issue->reviewer = $chosen;
374 }
375 $this->record($issue, "$command {$args[1]}");
376 $this->emit($command, $issue->number, $args[1]);
377 }

379 private function move(array $args): void
380 {
381 $issue = $this->issue($args[0]);
382 $target = $args[1];
383 if ($issue->duplicate !== null) {
384 throw new CommandError("duplicate of {$issue->duplicate}");
385 }
386 if (!in_array($target, WORKFLOWS[$issue->kind], true)) {
387 throw new CommandError('bad state');
388 }
389 if (!in_array("{$issue->state}:$target", TRANSITIONS[$issue->kind], true)) {
390 throw new CommandError("cannot move {$issue->state} to $target");
391 }
392 if ($target === 'IN_PROGRESS') {
393 $active = $this->activeSprint();
394 if ($issue->kind === 'STORY' && ($active === null || $issue->sprint !== $active->name)) {
395 throw new CommandError('not in active sprint');
396 }
397 if ($issue->assignee === null) {
398 throw new CommandError('no assignee');
399 }
400 if ($this->wip($issue->assignee) >= 3) {
401 throw new CommandError("wip limit {$issue->assignee}");
402 }
403 } elseif ($target === 'REVIEW') {
404 if ($issue->reviewer === null) {
405 throw new CommandError('no reviewer');
406 }
407 if ($issue->reviewer === $issue->assignee) {
408 throw new CommandError('reviewer is assignee');
409 }
410 }
411 $blockers = $this->unfinished($issue);
412 if (in_array($target, ['DONE', 'RESOLVED'], true) && $blockers !== []) {
413 throw new CommandError("blocked by {$blockers[0]}");
414 }
415 if ($target === 'READY' && $issue->estimate === null) {
416 throw new CommandError('no estimate');
417 }
418 $this->changeState($issue, $target);
419 }

421 private function link(string $command, array $args): void
422 {
423 $relation = $args[1];
424 $allowed = $command === 'LINK' ? ['BLOCKS', 'DUPLICATES'] : ['BLOCKS'];
425 if (!in_array($relation, $allowed, true)) {
426 throw new CommandError('bad link');
427 }
428 $source = $this->issue($args[0]);
429 $target = $this->issue($args[2]);
430 if ($command === 'UNLINK') {
431 if (!isset($source->blocks[$target->number])) {
432 throw new CommandError('no link');
433 }
434 $before = $this->blockedStates([$target->number]);
435 unset($source->blocks[$target->number], $target->blockers[$source->number]);
436 $this->record($source, "UNLINK BLOCKS {$target->number}");
437 $this->record($target, "UNLINK BLOCKED_BY {$source->number}");
438 $this->emit('UNLINK', $source->number, 'BLOCKS', $target->number);
439 $this->notify($before);
440 return;
441 }
442 if ($source === $target) {
443 throw new CommandError('self link');
444 }
445 if ($relation === 'BLOCKS') {
446 if (isset($source->blocks[$target->number])) {
447 throw new CommandError('link exists');
448 }
449 if ($this->reaches($target, $source)) {
450 throw new CommandError('cycle');
451 }
452 $this->notDone($target);
453 $before = $this->blockedStates([$target->number]);
454 $source->blocks[$target->number] = true;
455 $target->blockers[$source->number] = true;
456 $this->record($source, "LINK BLOCKS {$target->number}");
457 $this->record($target, "LINK BLOCKED_BY {$source->number}");
458 $this->emit('LINK', $source->number, $relation, $target->number);
459 $this->notify($before);
460 } else {
461 if ($source->kind !== $target->kind) {
462 throw new CommandError('type mismatch');
463 }
464 $this->notDone($source);
465 if ($target->duplicate !== null) {
466 throw new CommandError('target is duplicate');
467 }
468 $source->duplicate = $target->number;
469 $target->duplicates[$source->number] = true;
470 $this->record($source, "LINK DUPLICATES {$target->number}");
471 $this->record($target, "LINK DUPLICATED_BY {$source->number}");
472 $this->emit('LINK', $source->number, $relation, $target->number);
473 $this->changeState($source, $source->kind === 'BUG' ? 'CLOSED' : 'DONE');
474 if ($source->sprint !== null && $this->sprints[$source->sprint]->phase !== 'ENDED') {
475 $this->record($source, "SPRINT REMOVE {$source->sprint}");
476 $source->sprint = null;
477 }
478 }
479 }

481 private function sprintCommand(array $args): void
482 {
483 [$action, $name] = $args;
484 if ($action === 'NEW') {
485 if (isset($this->sprints[$name])) {
486 throw new CommandError('sprint exists');
487 }
488 $this->sprints[$name] = new Sprint($name);
489 $this->emit('SPRINT', 'NEW', $name);
490 return;
491 }
492 $sprint = $this->sprint($name);
493 if ($action === 'ADD' || $action === 'REMOVE') {
494 $issue = $this->issue($args[2]);
495 if ($sprint->phase === 'ENDED') {
496 throw new CommandError('sprint ended');
497 }
498 if ($action === 'ADD') {
499 $this->notDone($issue);
500 if ($issue->sprint !== null && $this->sprints[$issue->sprint]->phase !== 'ENDED') {
501 throw new CommandError("in sprint {$issue->sprint}");
502 }
503 if ($issue->state === 'BACKLOG') {
504 throw new CommandError('not ready');
505 }
506 $issue->sprint = $name;
507 } else {
508 if ($issue->sprint !== $name) {
509 throw new CommandError('not in sprint');
510 }
511 $this->notDone($issue);
512 $issue->sprint = null;
513 }
514 $this->record($issue, "SPRINT $action $name");
515 $this->emit('SPRINT', $action, $name, $issue->number, $this->summary($sprint));
516 } elseif ($action === 'START') {
517 if ($sprint->phase !== 'PLANNED') {
518 throw new CommandError('sprint not planned');
519 }
520 $active = $this->activeSprint();
521 if ($active !== null) {
522 throw new CommandError("sprint active {$active->name}");
523 }
524 if ($this->members($sprint) === []) {
525 throw new CommandError('sprint empty');
526 }
527 $sprint->phase = 'ACTIVE';
528 $sprint->start = $this->day;
529 $sprint->records[$this->day] = $this->remaining($sprint);
530 $this->emit('SPRINT', 'START', $name, 'day', $this->day, $this->summary($sprint), 'remaining', $this->remaining($sprint));
531 } else {
532 if ($sprint->phase !== 'ACTIVE') {
533 throw new CommandError('sprint not active');
534 }
535 $sprint->records[$this->day] = $this->remaining($sprint);
536 $members = $this->members($sprint);
537 $finished = array_values(array_filter($members, fn($issue) => $issue->done()));
538 $returned = array_values(array_filter($members, fn($issue) => !$issue->done()));
539 $sprint->velocity = array_sum(array_map(fn($issue) => $issue->points(), $finished));
540 $sprint->phase = 'ENDED';
541 $sprint->end = $this->day;
542 $this->emit('SPRINT', 'END', $name, 'day', $this->day, 'velocity', $sprint->velocity,
543 'done', count($finished), 'returned', count($returned));
544 foreach ($returned as $issue) {
545 $issue->sprint = null;
546 $this->record($issue, "SPRINT RETURN $name");
547 $this->emit('RETURNED', $issue->number);
548 }
549 }

552 private function condition(string $token): array
553 {
554 if (!preg_match('/\A([a-z]*)(>=|<=|!=|=|<|>|~)(.*)\z/', $token, $parts) || $parts[1] === '' || $parts[3] === '') {
555 throw new CommandError("syntax near $token");
556 }
557 [, $field, $operator, $text] = $parts;
558 if (!array_key_exists($field, FIELD_OPERATORS)) {
559 throw new CommandError("unknown field $field");
560 }
561 if (!in_array($operator, FIELD_OPERATORS[$field], true)) {
562 throw new CommandError("bad operator $token");
563 }
564 if ($text === '-' && in_array($field, ['assignee', 'reviewer', 'estimate', 'sprint'], true) && in_array($operator, EQUALITY, true)) {
565 $value = null;
566 } elseif (in_array($field, ['id', 'estimate', 'comments'], true) && preg_match('/\A[0-9]{1,6}\z/', $text)) {
567 $value = (int)$text;
568 } elseif ($field === 'priority' && in_array($text, PRIORITIES, true)) {
569 $value = array_search($text, PRIORITIES, true);
570 } elseif ($field === 'type' && array_key_exists($text, WORKFLOWS)) {
571 $value = $text;
572 } elseif ($field === 'status' && array_any(WORKFLOWS, fn($states) => in_array($text, $states, true))) {
573 $value = $text;
574 } elseif (in_array($field, ['assignee', 'reviewer'], true) && isset($this->users[$text])) {
575 $value = $text;
576 } elseif ($field === 'sprint' && isset($this->sprints[$text])) {
577 $value = $text;
578 } elseif (in_array($field, ['blocked', 'done'], true) && in_array($text, ['yes', 'no'], true)) {
579 $value = $text;
580 } elseif ($field === 'title') {
581 $value = strtolower($text);
582 } else {
583 throw new CommandError("bad value $token");
584 }
585 return [$field, $operator, $value];
586 }

588 private function compileQuery(string $text): array
589 {
590 preg_match_all('/[()]|[^ ()]+/', $text, $matches);
591 $tokens = $matches[0];
592 $conditions = [];
593 $operand = true;
594 $depth = 0;
595 foreach ($tokens as $index => $token) {
596 if ($operand) {
597 if ($token === '(') {
598 ++$depth;
599 } elseif (in_array($token, ['AND', 'OR', ')'], true)) {
600 throw new CommandError("syntax near $token");
601 } elseif ($token !== 'NOT') {
602 $conditions[$index] = $this->condition($token);
603 $operand = false;
604 }
605 } elseif ($token === 'AND' || $token === 'OR') {
606 $operand = true;
607 } elseif ($token === ')' && $depth > 0) {
608 --$depth;
609 } else {
610 throw new CommandError("syntax near $token");
611 }
612 }
613 if ($operand || $depth !== 0) {
614 throw new CommandError('syntax at end');
615 }
616 $output = [];
617 $operators = [];
618 $precedence = ['OR' => 1, 'AND' => 2, 'NOT' => 3];
619 foreach ($tokens as $index => $token) {
620 if (isset($conditions[$index])) {
621 $output[] = $conditions[$index];
622 } elseif ($token === '(' || $token === 'NOT') {
623 $operators[] = $token;
624 } elseif ($token === ')') {
625 while (end($operators) !== '(') {
626 $output[] = array_pop($operators);
627 }
628 array_pop($operators);
629 } else {
630 while ($operators !== [] && end($operators) !== '(' && $precedence[end($operators)] >= $precedence[$token]) {
631 $output[] = array_pop($operators);
632 }
633 $operators[] = $token;
634 }
635 }
636 return array_merge($output, array_reverse($operators));
637 }

639 private function fieldValue(Issue $issue, string $field): mixed
640 {
641 return match ($field) {
642 'id' => $issue->number, 'type' => $issue->kind, 'status' => $issue->state,
643 'priority' => $issue->level(), 'blocked' => yesNo($this->blocked($issue)),
644 'done' => yesNo($issue->done()), 'title' => strtolower($issue->title),
645 default => $issue->$field,
646 };
647 }

649 private function matches(Issue $issue, array $program): bool
650 {
651 $stack = [];
652 foreach ($program as $node) {
653 if ($node === 'NOT') {
654 $stack[] = !array_pop($stack);
655 } elseif ($node === 'AND' || $node === 'OR') {
656 $right = array_pop($stack);
657 $left = array_pop($stack);
658 $stack[] = $node === 'AND' ? $left && $right : $left || $right;
659 } else {
660 [$field, $operator, $expected] = $node;
661 $actual = $this->fieldValue($issue, $field);
662 $stack[] = match ($operator) {
663 '=' => $actual === $expected, '!=' => $actual !== $expected,
664 '~' => str_contains($actual, $expected),
665 '<' => $actual !== null && $actual < $expected,
666 '<=' => $actual !== null && $actual <= $expected,
667 '>' => $actual !== null && $actual > $expected,
668 '>=' => $actual !== null && $actual >= $expected,
669 };
670 }
671 }
672 return $stack[0];
673 }

675 private function query(string $text): void
676 {
677 $program = $this->compileQuery($text);
