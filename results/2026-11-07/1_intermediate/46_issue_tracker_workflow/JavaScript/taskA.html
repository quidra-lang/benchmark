71function compare(actual, operator, expected) {
72 if (operator === "=") return actual === expected;
73 if (operator === "!=") return actual !== expected;
74 if (actual === null || expected === null) return false;
75 if (operator === "<") return actual < expected;
76 if (operator === "<=") return actual <= expected;
77 if (operator === ">") return actual > expected;
78 return actual >= expected;
79}

124function parseQuery(tokens, conditions) {
125 let position = 0;
126 function factor() {
127 if (tokens[position] === "NOT") {
128 ++position;
129 const operand = factor();
130 return (issue) => !operand(issue);
131 }
132 if (tokens[position] === "(") {
133 ++position;
134 const node = expression();
135 ++position;
136 return node;
137 }
138 return conditions[position++];
139 }
140 function term() {
141 let node = factor();
142 while (tokens[position] === "AND") {
143 ++position;
144 const left = node;
145 const right = factor();
146 node = (issue) => left(issue) && right(issue);
147 }
148 return node;
149 }
150 function expression() {
151 let node = term();
152 while (tokens[position] === "OR") {
153 ++position;
154 const left = node;
155 const right = term();
156 node = (issue) => left(issue) || right(issue);
157 }
158 return node;
159 }
160 return expression();
161}

250 reaches(start, goal) {
251 const pending = [start];
252 const seen = new Set();
253 while (pending.length) {
254 const n = pending.pop();
255 if (n === goal) return true;
256 if (seen.has(n)) continue;
257 seen.add(n);
258 pending.push(...this.issues.get(n).blocks);
259 }
260 return false;
261 }

263 fieldValue(issue, field) {
264 switch (field) {
265 case "id":
266 return issue.number;
267 case "type":

269 case "status":
270 return issue.state;
271 case "priority":
272 return issue.level();
273 case "assignee":
274 return issue.assignee;
275 case "reviewer":
276 return issue.reviewer;
277 case "estimate":

279 case "sprint":

292 conditionValue(token, field, operator, value) {
293 if (value === "-" && NONE_FIELDS.has(field) && EQUALITY.includes(operator)) return null;
294 if (["id", "estimate", "comments"].includes(field) && isNumber(value)) return Number(value);
295 if (field === "priority" && PRIORITIES.includes(value)) return PRIORITIES.indexOf(value);
296 if (field === "type" && TYPES.includes(value)) return value;
297 if (field === "status" && ALL_STATES.has(value)) return value;
298 if (["assignee", "reviewer"].includes(field) && this.users.has(value)) return value;
299 if (field === "sprint" && this.sprints.has(value)) return value;
300 if (["blocked", "done"].includes(field) && ["yes", "no"].includes(value)) return value;
301 if (field === "title") return value.toLowerCase();
302 throw new CommandError(`bad value ${token}`);
303 }

305 condition(token) {
306 const match = /^([a-z]*)(>=|<=|!=|=|<|>|~)(.*)$/.exec(token);
307 if (!match || !match[1] || !match[3]) throw new CommandError(`syntax near ${token}`);
308 const [, field, operator, text] = match;
309 const operators = FIELD_OPERATORS.get(field);
310 if (!operators) throw new CommandError(`unknown field ${field}`);
311 if (!operators.includes(operator)) throw new CommandError(`bad operator ${token}`);
312 const expected = this.conditionValue(token, field, operator, text);
313 return (issue) => {
314 const actual = this.fieldValue(issue, field);
315 return operator === "~"
316 ? actual.includes(expected)
317 : compare(actual, operator, expected);
318 };
319 }

321 query(text) {
322 const tokens = text.match(/[()]|[^ ()]+/g) ?? [];
323 const conditions = [];
324 let operand = true;
325 let depth = 0;
326 for (const token of tokens) {
327 let condition = null;
328 if (operand) {
329 if (token === "(") ++depth;
330 else if (["AND", "OR", ")"].includes(token))
331 throw new CommandError(`syntax near ${token}`);
332 else if (token !== "NOT") {
333 condition = this.condition(token);
334 operand = false;
335 }
336 } else if (["AND", "OR"].includes(token)) operand = true;
337 else if (token === ")" && depth > 0) --depth;
338 else throw new CommandError(`syntax near ${token}`);
339 conditions.push(condition);
340 }
341 if (operand || depth !== 0) throw new CommandError("syntax at end");
342 const matches = [...this.issues.values()].filter(parseQuery(tokens, conditions));
343 matches.sort((a, b) => b.level() - a.level() || a.number - b.number);
344 this.emit("QUERY", matches.length);
345 for (const i of matches)
346 this.emit(
347 i.number,
348 i.type,
349 i.priority,
350 i.state,
351 display(i.assignee),
352 display(i.estimate),
353 i.title,
354 );
355 }

358 this.output = [];

360 const [command, ...args] = line.split(" ");

368 return this.output;
369 }

371 dispatch(command, args) {
372 const [x, y, z] = args;
373 switch (command) {
374 case "DAY": {
375 const day = Number(x);
376 if (day < this.day) throw new CommandError("day backwards");
377 this.day = day;
378 this.emit("DAY", day);
379 return;
380 }
381 case "USER":
382 if (this.users.has(x)) throw new CommandError("user exists");
383 this.users.add(x);
384 this.emit("USER", x);
385 return;
386 case "CREATE": {
387 const n = Number(x);
388 if (this.issues.has(n)) throw new CommandError("issue exists");
389 if (!TYPES.includes(y)) throw new CommandError("bad type");
390 if (!PRIORITIES.includes(z)) throw new CommandError("bad priority");
391 const issue = new Issue(n, y, z, args.slice(3).join(" "));
392 this.issues.set(n, issue);
393 this.record(issue, `CREATE ${y} ${z}`);
394 this.emit("CREATE", n, y, z, issue.state);
395 return;
396 }
397 case "QUERY":
398 this.query(args.join(" "));
399 return;
400 case "SPRINT":
401 this.sprintCommand(args);
402 return;
403 case "BOARD": {
404 this.emit("BOARD");
405 const issues = [...this.issues.values()];
406 for (const type of TYPES) {
407 const parts = [type];
408 for (const state of WORKFLOWS[type])
409 parts.push(
410 state,
411 String(
412 issues.filter((i) => i.type === type && i.state === state).length,
413 ),
414 );
415 this.emit(...parts);
416 }
417 const done = issues.filter((i) => i.done()).length;
418 this.emit("TOTAL", issues.length, "open", issues.length - done, "done", done);
419 return;
420 }
421 case "WORKLOAD":
422 this.emit("WORKLOAD", this.users.size);
423 for (const user of [...this.users].sort()) {
424 const issues = [...this.issues.values()];
425 const assigned = issues.filter((i) => i.assignee === user && !i.done());
426 this.emit(
427 user,
428 "assigned",
429 assigned.length,
430 "in_progress",
431 assigned.filter((i) => i.state === "IN_PROGRESS").length,
432 "review",
433 assigned.filter((i) => i.state === "REVIEW").length,
434 "reviewing",
435 issues.filter((i) => i.reviewer === user && i.state === "REVIEW").length,
436 "points",
437 assigned.reduce((sum, i) => sum + i.points(), 0),
438 );
439 }
440 return;
441 case "SPRINTS":
442 this.emit("SPRINTS", this.sprints.size);
443 for (const s of this.sprints.values()) {
444 if (s.phase === "PLANNED") this.emit(s.name, s.phase, this.summary(s));
445 else if (s.phase === "ACTIVE")
446 this.emit(
447 s.name,
448 s.phase,
449 "start",
450 s.start,
451 this.summary(s),
452 "remaining",
453 this.remaining(s),
454 );
455 else
456 this.emit(
457 s.name,
458 s.phase,
459 "start",
460 s.start,
461 "end",
462 s.end,
463 "velocity",
464 s.velocity,
465 );
466 }
467 return;
468 case "BURNDOWN": {
469 const s = this.sprint(x);
470 if (s.phase === "PLANNED") throw new CommandError("sprint not started");
471 const last = s.phase === "ACTIVE" ? this.day : s.end;
472 this.emit("BURNDOWN", x, s.phase, "days", last - s.start + 1);
473 let value = 0;
474 for (let day = s.start; day <= last; ++day) {
475 if (s.records.has(day)) value = s.records.get(day);
476 this.emit("day", day, value);
477 }
478 return;
479 }
480 case "LINK":
481 case "UNLINK":
482 this.link(command, args);
483 return;
484 }
485 const i = this.issue(x);
486 switch (command) {
487 case "PRIORITY": {
488 if (!PRIORITIES.includes(y)) throw new CommandError("bad priority");
489 this.notDone(i);
490 if (i.priority === y) throw new CommandError("no change");
491 const before = i.priority;
492 i.priority = y;
493 this.record(i, `PRIORITY ${before} ${y}`);
494 this.emit("PRIORITY", i.number, before, y);
495 return;
496 }
497 case "ASSIGN": {
498 const user = this.user(y);
499 this.notDone(i);
500 if (i.assignee === user) throw new CommandError("no change");
501 if (user === null && ["IN_PROGRESS", "REVIEW"].includes(i.state))
502 throw new CommandError("assignee required");
503 if (i.state === "IN_PROGRESS" && this.wip(user) >= 3)
504 throw new CommandError(`wip limit ${y}`);
505 if (i.state === "REVIEW" && user === i.reviewer)
506 throw new CommandError("reviewer is assignee");
507 i.assignee = user;
508 this.record(i, `ASSIGN ${y}`);
509 this.emit("ASSIGN", i.number, y);
510 return;
511 }
512 case "REVIEWER": {
513 const user = this.user(y);
514 this.notDone(i);
515 if (i.reviewer === user) throw new CommandError("no change");
516 if (user === null && i.state === "REVIEW")
517 throw new CommandError("reviewer required");
518 if (i.state === "REVIEW" && user === i.assignee)
519 throw new CommandError("reviewer is assignee");
520 i.reviewer = user;
521 this.record(i, `REVIEWER ${y}`);
522 this.emit("REVIEWER", i.number, y);
523 return;
524 }
525 case "ESTIMATE": {
526 const points = Number(y);
527 if (!ESTIMATES.has(points)) throw new CommandError("bad estimate");
528 this.notDone(i);
529 if (i.estimate === points) throw new CommandError("no change");
530 const before = display(i.estimate);
531 i.estimate = points;
532 this.record(i, `ESTIMATE ${before} ${points}`);
533 this.emit("ESTIMATE", i.number, before, points);
534 return;
535 }
536 case "COMMENT":
537 this.user(y);
538 ++i.comments;
539 this.record(i, `COMMENT ${y} ${args.slice(2).join(" ")}`);
540 this.emit("COMMENT", i.number, i.comments);
541 return;
542 case "MOVE": {
543 if (i.duplicate !== null) throw new CommandError(`duplicate of ${i.duplicate}`);
544 if (!WORKFLOWS[i.type].includes(y)) throw new CommandError("bad state");
545 if (!TRANSITIONS[i.type].includes(`${i.state}:${y}`))
546 throw new CommandError(`cannot move ${i.state} to ${y}`);
547 if (y === "IN_PROGRESS") {
548 const active = this.active();
549 if (i.type === "STORY" && (!active || i.sprint !== active.name))
550 throw new CommandError("not in active sprint");
551 if (i.assignee === null) throw new CommandError("no assignee");
552 if (this.wip(i.assignee) >= 3)
553 throw new CommandError(`wip limit ${i.assignee}`);
554 }
555 if (y === "REVIEW") {
556 if (i.reviewer === null) throw new CommandError("no reviewer");
557 if (i.reviewer === i.assignee) throw new CommandError("reviewer is assignee");
558 }
559 if (y === "DONE" || y === "RESOLVED") {
560 const blockers = this.unfinishedBlockers(i);
561 if (blockers.length) throw new CommandError(`blocked by ${blockers[0]}`);
562 }
563 if (y === "READY" && i.estimate === null) throw new CommandError("no estimate");
564 this.changeState(i, y);
565 return;
566 }
567 case "SHOW":
568 this.emit(
569 "ISSUE",
570 i.number,
571 i.type,
572 i.priority,
573 i.state,
574 "blocked",
575 yesNo(this.blocked(i)),
576 );
577 this.emit("TITLE", i.title);
578 this.emit(
579 "PEOPLE",
580 "assignee",
581 display(i.assignee),
582 "reviewer",
583 display(i.reviewer),
584 );
585 this.emit(
586 "PLAN",
587 "estimate",
588 display(i.estimate),
589 "sprint",
590 display(i.sprint),
591 "comments",
592 i.comments,
593 );
594 this.emit(
595 "LINKS",
596 "blocks",
597 numberList(i.blocks),
598 "blocked_by",
599 numberList(i.blockers),
600 "duplicate_of",
601 display(i.duplicate),
602 "duplicated_by",
603 numberList(i.duplicates),
604 );
605 return;
606 case "HISTORY":
607 this.emit("HISTORY", i.number, i.history.length);
608 for (const entry of i.history) this.emit(entry);
609 return;
610 }
611 }

613 link(command, args) {
614 const relation = args[1];
615 if (
616 command === "UNLINK"
617 ? relation !== "BLOCKS"
618 : !["BLOCKS", "DUPLICATES"].includes(relation)
619 )
620 throw new CommandError("bad link");
621 const a = this.issue(args[0]);
622 const b = this.issue(args[2]);
623 if (command === "UNLINK") {
624 if (!a.blocks.has(b.number)) throw new CommandError("no link");
625 const before = this.blockedStates([b.number]);
626 a.blocks.delete(b.number);
627 b.blockers.delete(a.number);
628 this.record(a, `UNLINK BLOCKS ${b.number}`);
629 this.record(b, `UNLINK BLOCKED_BY ${a.number}`);
630 this.emit("UNLINK", a.number, "BLOCKS", b.number);
631 this.notify(before);
632 } else {
633 if (a === b) throw new CommandError("self link");
634 if (relation === "BLOCKS") {
635 if (a.blocks.has(b.number)) throw new CommandError("link exists");
636 if (this.reaches(b.number, a.number)) throw new CommandError("cycle");
637 this.notDone(b);
638 const before = this.blockedStates([b.number]);
639 a.blocks.add(b.number);
640 b.blockers.add(a.number);
641 this.record(a, `LINK BLOCKS ${b.number}`);
642 this.record(b, `LINK BLOCKED_BY ${a.number}`);
643 this.emit("LINK", a.number, "BLOCKS", b.number);
644 this.notify(before);
645 } else {
646 if (a.type !== b.type) throw new CommandError("type mismatch");
647 this.notDone(a);
648 if (b.duplicate !== null) throw new CommandError("target is duplicate");
649 a.duplicate = b.number;
650 b.duplicates.add(a.number);
651 this.record(a, `LINK DUPLICATES ${b.number}`);
652 this.record(b, `LINK DUPLICATED_BY ${a.number}`);
653 this.emit("LINK", a.number, "DUPLICATES", b.number);
654 this.changeState(a, a.type === "BUG" ? "CLOSED" : "DONE");
655 if (a.sprint !== null && this.sprints.get(a.sprint).phase !== "ENDED") {
656 this.record(a, `SPRINT REMOVE ${a.sprint}`);
657 a.sprint = null;
658 }
659 }
660 }

663 sprintCommand(args) {
664 const [operation, name, issueNumber] = args;
665 if (operation === "NEW") {
666 if (this.sprints.has(name)) throw new CommandError("sprint exists");
667 this.sprints.set(name, new Sprint(name));
668 this.emit("SPRINT", "NEW", name);
669 return;
670 }
671 const s = this.sprint(name);
672 if (operation === "ADD" || operation === "REMOVE") {
673 const i = this.issue(issueNumber);
674 if (s.phase === "ENDED") throw new CommandError("sprint ended");
675 if (operation === "ADD") {
676 this.notDone(i);
677 if (i.sprint !== null && this.sprints.get(i.sprint).phase !== "ENDED")
678 throw new CommandError(`in sprint ${i.sprint}`);
679 if (i.state === "BACKLOG") throw new CommandError("not ready");
680 i.sprint = name;
681 } else {
682 if (i.sprint !== name) throw new CommandError("not in sprint");
683 this.notDone(i);
684 i.sprint = null;
685 }
686 this.record(i, `SPRINT ${operation} ${name}`);
687 this.emit("SPRINT", operation, name, i.number, this.summary(s));
688 } else if (operation === "START") {
689 if (s.phase !== "PLANNED") throw new CommandError("sprint not planned");
690 const active = this.active();
691 if (active) throw new CommandError(`sprint active ${active.name}`);
692 if (this.members(s).length === 0) throw new CommandError("sprint empty");
693 s.phase = "ACTIVE";
694 s.start = this.day;
695 const remaining = this.remaining(s);
696 s.records.set(this.day, remaining);
697 this.emit(
698 "SPRINT",
699 "START",
700 name,
701 "day",
702 this.day,
703 this.summary(s),
704 "remaining",
705 remaining,
706 );
707 } else {
708 if (s.phase !== "ACTIVE") throw new CommandError("sprint not active");
709 s.records.set(this.day, this.remaining(s));
710 const members = this.members(s);
711 const done = members.filter((i) => i.done());
712 const returned = members.filter((i) => !i.done());
713 s.velocity = done.reduce((sum, i) => sum + i.points(), 0);
714 s.phase = "ENDED";
715 s.end = this.day;
716 this.emit(
717 "SPRINT",
718 "END",
719 name,
720 "day",
721 this.day,
