1const TYPES = ("BUG", "TASK", "STORY")
2const PRIORITIES = ("LOW", "MEDIUM", "HIGH", "CRITICAL")
3const WORKFLOWS = Dict(
4 "BUG" => ["OPEN", "TRIAGED", "IN_PROGRESS", "REVIEW", "RESOLVED", "CLOSED", "REOPENED"],
5 "TASK" => ["OPEN", "IN_PROGRESS", "DONE"],
6 "STORY" => ["BACKLOG", "READY", "IN_PROGRESS", "REVIEW", "DONE"],
7)
8const TRANSITIONS = Dict(
9 "BUG" => Set([("OPEN", "TRIAGED"), ("TRIAGED", "IN_PROGRESS"), ("IN_PROGRESS", "REVIEW"),
10 ("REVIEW", "IN_PROGRESS"), ("REVIEW", "RESOLVED"), ("RESOLVED", "CLOSED"),
11 ("RESOLVED", "REOPENED"), ("CLOSED", "REOPENED"), ("REOPENED", "IN_PROGRESS")]),
12 "TASK" => Set([("OPEN", "IN_PROGRESS"), ("IN_PROGRESS", "OPEN"), ("IN_PROGRESS", "DONE")]),
13 "STORY" => Set([("BACKLOG", "READY"), ("READY", "BACKLOG"), ("READY", "IN_PROGRESS"),
14 ("IN_PROGRESS", "REVIEW"), ("REVIEW", "IN_PROGRESS"), ("REVIEW", "DONE")]),
15)
16const DONE_STATES = ("RESOLVED", "CLOSED", "DONE")
17const ESTIMATES = (0, 1, 2, 3, 5, 8, 13, 21)
18const EQUALITY = ("=", "!=")
19const ORDERING = ("=", "!=", "<", "<=", ">", ">=")
20const FIELD_OPERATORS = Dict(
21 "id" => ORDERING, "type" => EQUALITY, "status" => EQUALITY, "priority" => ORDERING,
22 "assignee" => EQUALITY, "reviewer" => EQUALITY, "estimate" => ORDERING,
23 "sprint" => EQUALITY, "comments" => ORDERING, "blocked" => EQUALITY, "done" => EQUALITY,
24 "title" => ("~",),
25)
26struct CommandError <: Exception
27 reason::String
28end
29fail(reason) = throw(CommandError(reason))
30display_value(value) = isnothing(value) ? "-" : string(value)
31yes_no(value) = value ? "yes" : "no"
32number_list(ids) = isempty(ids) ? "-" : join(sort!(collect(ids)), ',')

34mutable struct Issue
35 number::Int
36 kind::String
37 priority::String
38 title::String
39 state::String
40 assignee::String
41 reviewer::String
42 estimate::Union{Nothing,Int}
43 sprint::String
44 comments::Int
45 blocks::Set{Int}
46 blockers::Set{Int}
47 duplicate::Union{Nothing,Int}
48 duplicates::Set{Int}
49 history::Vector{String}
50end
51Issue(n, kind, priority, title) = Issue(n, kind, priority, title, WORKFLOWS[kind][1],
52 "-", "-", nothing, "-", 0, Set{Int}(), Set{Int}(), nothing, Set{Int}(), String[])
53done(i::Issue) = i.state in DONE_STATES
54points(i::Issue) = isnothing(i.estimate) ? 0 : i.estimate
55level(i::Issue) = findfirst(==(i.priority), PRIORITIES)

57mutable struct Sprint
58 name::String
59 phase::String
60 start::Int
61 finish::Int
62 velocity::Int
63 records::Dict{Int,Int}
64end
65Sprint(name) = Sprint(name, "PLANNED", 0, 0, 0, Dict{Int,Int}())
66mutable struct Tracker
67 day::Int
68 users::Set{String}
69 issues::Dict{Int,Issue}
70 sprints::Dict{String,Sprint}
71 sprint_order::Vector{String}
72end
73Tracker() = Tracker(1, Set{String}(), Dict{Int,Issue}(), Dict{String,Sprint}(), String[])

75function require_issue(t, id)
76 n = parse(Int, id)
77 haskey(t.issues, n) || fail("unknown issue")
78 return t.issues[n]
79end
80function require_user(t, user)
81 user != "-" && user ∉ t.users && fail("unknown user")
82 return user
83end
84function require_sprint(t, name)
85 haskey(t.sprints, name) || fail("unknown sprint")
86 return t.sprints[name]
87end
88not_done(i) = done(i) && fail("issue done")
89ordered_issues(t) = sort!(collect(values(t.issues)); by=i -> i.number)
90unfinished_blockers(t, i) = sort!([id for id in i.blockers if !done(t.issues[id])])
91blocked(t, i) = !done(i) && !isempty(unfinished_blockers(t, i))
92wip_count(t, user) = count(i -> i.assignee == user && i.state == "IN_PROGRESS", values(t.issues))
93function active_sprint(t)
94 for name in t.sprint_order
95 t.sprints[name].phase == "ACTIVE" && return t.sprints[name]
96 end
97 return nothing
98end
99members(t, s) = [i for i in ordered_issues(t) if i.sprint == s.name]
100remaining(t, s) = sum((points(i) for i in members(t, s) if !done(i)); init=0)
101function sprint_summary(t, s)
102 ids = members(t, s)
103 return "issues $(length(ids)) points $(sum((points(i) for i in ids); init=0))"
104end
105record!(t, i, content) = push!(i.history, "day $(t.day) $content")
106blocked_states(t, ids) = Dict(id => blocked(t, t.issues[id]) for id in ids)
107function notifications(t, before)
108 lines = String[]
109 for id in sort!(collect(keys(before)))
110 after = blocked(t, t.issues[id])
111 if after != before[id]
112 push!(lines, "$(after ? "BLOCKED" : "UNBLOCKED") $id")
113 end
114 end
115 return lines
116end
117function change_state!(t, i, target)
118 before = blocked_states(t, i.blocks)
119 previous = i.state
120 record!(t, i, "MOVE $previous $target")
121 i.state = target
122 return vcat(["MOVE $(i.number) $previous $target"], notifications(t, before))
123end
124function reaches(t, start, goal)
125 pending = [start]
126 visited = Set{Int}()
127 while !isempty(pending)
128 id = pop!(pending)
129 id == goal && return true
130 id in visited && continue
131 push!(visited, id)
132 append!(pending, t.issues[id].blocks)
133 end
134 return false
135end

137abstract type QueryNode end
138struct QueryCondition <: QueryNode
139 field::String
140 operator::String
141 value::Union{Nothing,Int,String}
142end
143struct NotNode <: QueryNode
144 child::QueryNode
145end
146struct BinaryNode <: QueryNode
147 operator::String
148 left::QueryNode
149 right::QueryNode
150end
151function condition(t, token)
152 m = match(r"^([a-z]*)(>=|<=|!=|=|<|>|~)(.*)$", token)
153 isnothing(m) && fail("syntax near $token")
154 field, op, text = String.(m.captures)
155 (isempty(field) || isempty(text)) && fail("syntax near $token")
156 haskey(FIELD_OPERATORS, field) || fail("unknown field $field")
157 op in FIELD_OPERATORS[field] || fail("bad operator $token")
158 value = if field in ("id", "estimate", "comments") && occursin(r"^[0-9]{1,6}$", text)
159 parse(Int, text)
160 elseif field == "estimate" && text == "-" && op in EQUALITY
161 nothing
162 elseif field == "priority" && text in PRIORITIES
163 findfirst(==(text), PRIORITIES)
164 elseif field == "type" && text in TYPES
165 text
166 elseif field == "status" && any(states -> text in states, values(WORKFLOWS))
167 text
168 elseif field in ("assignee", "reviewer") && (text == "-" || text in t.users)
169 text
170 elseif field == "sprint" && (text == "-" || haskey(t.sprints, text))
171 text
172 elseif field in ("blocked", "done") && text in ("yes", "no")
173 text
174 elseif field == "title"
175 lowercase(text)
176 else
177 fail("bad value $token")
178 end
179 return QueryCondition(field, op, value)
180end
181mutable struct QueryParser
182 tokens::Vector{String}
183 conditions::Vector{Union{Nothing,QueryCondition}}
184 position::Int
185end
186function accept!(p, token)
187 if p.position <= length(p.tokens) && p.tokens[p.position] == token
188 p.position += 1
189 return true
190 end
191 return false
192end
193function parse_factor!(p)
194 accept!(p, "NOT") && return NotNode(parse_factor!(p))
195 if accept!(p, "(")
196 node = parse_expression!(p)
197 accept!(p, ")")
198 return node
199 end
200 node = p.conditions[p.position]
201 p.position += 1
202 return node
203end
204function parse_term!(p)
205 node = parse_factor!(p)
206 while accept!(p, "AND")
207 node = BinaryNode("AND", node, parse_factor!(p))
208 end
209 return node
210end
211function parse_expression!(p)
212 node = parse_term!(p)
213 while accept!(p, "OR")
214 node = BinaryNode("OR", node, parse_term!(p))
215 end
216 return node
217end
218function parse_query(t, text)
219 tokens = [String(m.match) for m in eachmatch(r"[()]|[^ ()]+", text)]
220 conditions = Union{Nothing,QueryCondition}[nothing for _ in tokens]
221 operand, depth = true, 0
222 for (i, token) in enumerate(tokens)
223 if operand
224 if token == "("
225 depth += 1
226 elseif token in ("AND", "OR", ")")
227 fail("syntax near $token")
228 elseif token != "NOT"
229 conditions[i] = condition(t, token)
230 operand = false
231 end
232 elseif token in ("AND", "OR")
233 operand = true
234 elseif token == ")" && depth > 0
235 depth -= 1
236 else
237 fail("syntax near $token")
238 end
239 end
240 (operand || depth != 0) && fail("syntax at end")
241 return parse_expression!(QueryParser(tokens, conditions, 1))
242end
243function field_value(t, i, field)
244 field == "id" && return i.number
245 field == "type" && return i.kind
246 field == "status" && return i.state
247 field == "priority" && return level(i)
248 field == "assignee" && return i.assignee
249 field == "reviewer" && return i.reviewer
250 field == "estimate" && return i.estimate
251 field == "sprint" && return i.sprint
252 field == "comments" && return i.comments
253 field == "blocked" && return yes_no(blocked(t, i))
254 field == "done" && return yes_no(done(i))
255 return lowercase(i.title)
256end
257matches(n::NotNode, t, i) = !matches(n.child, t, i)
258matches(n::BinaryNode, t, i) = n.operator == "AND" ?
259 matches(n.left, t, i) && matches(n.right, t, i) : matches(n.left, t, i) || matches(n.right, t, i)
260function matches(n::QueryCondition, t, i)
261 actual = field_value(t, i, n.field)
262 n.operator == "~" && return occursin(n.value, actual)
263 n.operator == "=" && return actual == n.value
264 n.operator == "!=" && return actual != n.value
265 isnothing(actual) && return false
266 n.operator == "<" && return actual < n.value
267 n.operator == "<=" && return actual <= n.value
268 n.operator == ">" && return actual > n.value
269 return actual >= n.value
270end

272function process_sprint!(t, args)
273 cmd, name = args[1:2]
274 if cmd == "NEW"
275 haskey(t.sprints, name) && fail("sprint exists")
276 t.sprints[name] = Sprint(name)
277 push!(t.sprint_order, name)
278 return ["SPRINT NEW $name"]
279 end
280 s = require_sprint(t, name)
281 if cmd in ("ADD", "REMOVE")
282 i = require_issue(t, args[3])
283 s.phase == "ENDED" && fail("sprint ended")
284 if cmd == "ADD"
285 not_done(i)
286 i.sprint != "-" && t.sprints[i.sprint].phase != "ENDED" && fail("in sprint $(i.sprint)")
287 i.state == "BACKLOG" && fail("not ready")
288 i.sprint = name
289 else
290 i.sprint != name && fail("not in sprint")
291 not_done(i)
292 i.sprint = "-"
293 end
294 record!(t, i, "SPRINT $cmd $name")
295 return ["SPRINT $cmd $name $(i.number) $(sprint_summary(t, s))"]
296 elseif cmd == "START"
297 s.phase != "PLANNED" && fail("sprint not planned")
298 active = active_sprint(t)
299 !isnothing(active) && fail("sprint active $(active.name)")
300 isempty(members(t, s)) && fail("sprint empty")
301 s.phase, s.start = "ACTIVE", t.day
302 s.records[t.day] = remaining(t, s)
303 return ["SPRINT START $name day $(t.day) $(sprint_summary(t, s)) remaining $(remaining(t, s))"]
304 end
305 s.phase != "ACTIVE" && fail("sprint not active")
306 s.records[t.day] = remaining(t, s)
307 ids = members(t, s)
308 finished, returned = filter(done, ids), filter(i -> !done(i), ids)
309 s.velocity = sum((points(i) for i in finished); init=0)
310 s.phase, s.finish = "ENDED", t.day
311 lines = ["SPRINT END $name day $(t.day) velocity $(s.velocity) done $(length(finished)) returned $(length(returned))"]
312 for i in returned
313 i.sprint = "-"
314 record!(t, i, "SPRINT RETURN $name")
315 push!(lines, "RETURNED $(i.number)")
316 end
317 return lines
318end
319function process_command!(t, line)
320 words = String.(split(line, ' '))
321 cmd, args = words[1], words[2:end]
322 if cmd == "DAY"
323 day = parse(Int, args[1])
324 day < t.day && fail("day backwards")
325 t.day = day
326 return ["DAY $day"]
327 elseif cmd == "USER"
328 user = args[1]
329 user in t.users && fail("user exists")
330 push!(t.users, user)
331 return ["USER $user"]
332 elseif cmd == "CREATE"
333 id, kind, priority = parse(Int, args[1]), args[2], args[3]
334 haskey(t.issues, id) && fail("issue exists")
335 kind in TYPES || fail("bad type")
336 priority in PRIORITIES || fail("bad priority")
337 i = Issue(id, kind, priority, join(args[4:end], ' '))
338 t.issues[id] = i
339 record!(t, i, "CREATE $kind $priority")
340 return ["CREATE $id $kind $priority $(i.state)"]
341 elseif cmd == "SPRINT"
342 return process_sprint!(t, args)
343 elseif cmd == "QUERY"
344 tree = parse_query(t, join(args, ' '))
345 found = [i for i in values(t.issues) if matches(tree, t, i)]
346 sort!(found; by=i -> (-level(i), i.number))
347 lines = ["QUERY $(length(found))"]
348 for i in found
349 push!(lines, "$(i.number) $(i.kind) $(i.priority) $(i.state) $(i.assignee) $(display_value(i.estimate)) $(i.title)")
350 end
351 return lines
352 elseif cmd == "BOARD"
353 lines = ["BOARD"]
354 for kind in TYPES
355 parts = [kind]
356 for state in WORKFLOWS[kind]
357 count_issue = count(i -> i.kind == kind && i.state == state, values(t.issues))
358 push!(parts, state, string(count_issue))
359 end
360 push!(lines, join(parts, ' '))
361 end
362 finished = count(done, values(t.issues))
363 push!(lines, "TOTAL $(length(t.issues)) open $(length(t.issues) - finished) done $finished")
364 return lines
365 elseif cmd == "WORKLOAD"
366 lines = ["WORKLOAD $(length(t.users))"]
367 for user in sort!(collect(t.users))
368 assigned = [i for i in values(t.issues) if i.assignee == user && !done(i)]
369 progress = count(i -> i.state == "IN_PROGRESS", assigned)
370 review = count(i -> i.state == "REVIEW", assigned)
371 reviewing = count(i -> i.reviewer == user && i.state == "REVIEW", values(t.issues))
372 total = sum((points(i) for i in assigned); init=0)
373 push!(lines, "$user assigned $(length(assigned)) in_progress $progress review $review reviewing $reviewing points $total")
374 end
375 return lines
376 elseif cmd == "SPRINTS"
377 lines = ["SPRINTS $(length(t.sprints))"]
378 for name in t.sprint_order
379 s = t.sprints[name]
380 if s.phase == "PLANNED"
381 push!(lines, "$name PLANNED $(sprint_summary(t, s))")
382 elseif s.phase == "ACTIVE"
383 push!(lines, "$name ACTIVE start $(s.start) $(sprint_summary(t, s)) remaining $(remaining(t, s))")
384 else
385 push!(lines, "$name ENDED start $(s.start) end $(s.finish) velocity $(s.velocity)")
386 end
387 end
388 return lines
389 elseif cmd == "BURNDOWN"
390 s = require_sprint(t, args[1])
391 s.phase == "PLANNED" && fail("sprint not started")
392 last = s.phase == "ACTIVE" ? t.day : s.finish
393 lines = ["BURNDOWN $(s.name) $(s.phase) days $(last - s.start + 1)"]
394 value = 0
395 for day in s.start:last
396 value = get(s.records, day, value)
397 push!(lines, "day $day $value")
398 end
399 return lines
400 elseif cmd == "LINK" || cmd == "UNLINK"
401 relation = args[2]
402 relation in (cmd == "LINK" ? ("BLOCKS", "DUPLICATES") : ("BLOCKS",)) || fail("bad link")
403 a, b = require_issue(t, args[1]), require_issue(t, args[3])
404 if cmd == "UNLINK"
405 b.number in a.blocks || fail("no link")
406 before = blocked_states(t, [b.number])
407 delete!(a.blocks, b.number)
408 delete!(b.blockers, a.number)
409 record!(t, a, "UNLINK BLOCKS $(b.number)")
410 record!(t, b, "UNLINK BLOCKED_BY $(a.number)")
411 return vcat(["UNLINK $(a.number) BLOCKS $(b.number)"], notifications(t, before))
412 end
413 a === b && fail("self link")
414 if relation == "BLOCKS"
415 b.number in a.blocks && fail("link exists")
416 reaches(t, b.number, a.number) && fail("cycle")
417 not_done(b)
418 before = blocked_states(t, [b.number])
419 push!(a.blocks, b.number)
420 push!(b.blockers, a.number)
421 record!(t, a, "LINK BLOCKS $(b.number)")
422 record!(t, b, "LINK BLOCKED_BY $(a.number)")
423 return vcat(["LINK $(a.number) BLOCKS $(b.number)"], notifications(t, before))
424 end
425 a.kind != b.kind && fail("type mismatch")
426 not_done(a)
427 !isnothing(b.duplicate) && fail("target is duplicate")
428 a.duplicate = b.number
429 push!(b.duplicates, a.number)
430 record!(t, a, "LINK DUPLICATES $(b.number)")
431 record!(t, b, "LINK DUPLICATED_BY $(a.number)")
432 lines = vcat(["LINK $(a.number) DUPLICATES $(b.number)"], change_state!(t, a, a.kind == "BUG" ? "CLOSED" : "DONE"))
433 if a.sprint != "-" && t.sprints[a.sprint].phase != "ENDED"
434 record!(t, a, "SPRINT REMOVE $(a.sprint)")
435 a.sprint = "-"
436 end
437 return lines
438 end
439 i = require_issue(t, args[1])
440 id = i.number
441 if cmd == "SHOW"
442 return ["ISSUE $id $(i.kind) $(i.priority) $(i.state) blocked $(yes_no(blocked(t, i)))",
443 "TITLE $(i.title)", "PEOPLE assignee $(i.assignee) reviewer $(i.reviewer)",
444 "PLAN estimate $(display_value(i.estimate)) sprint $(i.sprint) comments $(i.comments)",
445 "LINKS blocks $(number_list(i.blocks)) blocked_by $(number_list(i.blockers)) duplicate_of $(display_value(i.duplicate)) duplicated_by $(number_list(i.duplicates))"]
446 elseif cmd == "HISTORY"
447 return vcat(["HISTORY $id $(length(i.history))"], i.history)
448 elseif cmd == "COMMENT"
449 user = require_user(t, args[2])
450 i.comments += 1
451 record!(t, i, "COMMENT $user $(join(args[3:end], ' '))")
452 return ["COMMENT $id $(i.comments)"]
453 end
454 target = args[2]
455 if cmd == "PRIORITY"
456 target in PRIORITIES || fail("bad priority")
457 not_done(i)
458 target == i.priority && fail("no change")
459 old = i.priority
460 i.priority = target
461 record!(t, i, "PRIORITY $old $target")
462 return ["PRIORITY $id $old $target"]
463 elseif cmd == "ASSIGN" || cmd == "REVIEWER"
464 require_user(t, target)
465 not_done(i)
466 target == (cmd == "ASSIGN" ? i.assignee : i.reviewer) && fail("no change")
467 if cmd == "ASSIGN"
468 target == "-" && i.state in ("IN_PROGRESS", "REVIEW") && fail("assignee required")
469 i.state == "IN_PROGRESS" && wip_count(t, target) >= 3 && fail("wip limit $target")
470 i.state == "REVIEW" && target == i.reviewer && fail("reviewer is assignee")
471 i.assignee = target
472 else
473 target == "-" && i.state == "REVIEW" && fail("reviewer required")
474 i.state == "REVIEW" && target == i.assignee && fail("reviewer is assignee")
475 i.reviewer = target
476 end
477 record!(t, i, "$cmd $target")
478 return ["$cmd $id $target"]
479 elseif cmd == "ESTIMATE"
480 value = parse(Int, target)
481 value in ESTIMATES || fail("bad estimate")
482 not_done(i)
483 value == i.estimate && fail("no change")
484 old = display_value(i.estimate)
485 i.estimate = value
486 record!(t, i, "ESTIMATE $old $value")
487 return ["ESTIMATE $id $old $value"]
488 end
489 !isnothing(i.duplicate) && fail("duplicate of $(i.duplicate)")
490 target in WORKFLOWS[i.kind] || fail("bad state")
491 (i.state, target) in TRANSITIONS[i.kind] || fail("cannot move $(i.state) to $target")
492 if target == "IN_PROGRESS"
493 active = active_sprint(t)
494 i.kind == "STORY" && (isnothing(active) || i.sprint != active.name) && fail("not in active sprint")
495 i.assignee == "-" && fail("no assignee")
496 wip_count(t, i.assignee) >= 3 && fail("wip limit $(i.assignee)")
497 elseif target == "REVIEW"
498 i.reviewer == "-" && fail("no reviewer")
499 i.reviewer == i.assignee && fail("reviewer is assignee")
500 end
501 if target in ("DONE", "RESOLVED")
502 blockers = unfinished_blockers(t, i)
503 !isempty(blockers) && fail("blocked by $(first(blockers))")
504 end
505 target == "READY" && isnothing(i.estimate) && fail("no estimate")
506 return change_state!(t, i, target)
507end
508function execute!(t, line)
509 result = try
510 process_command!(t, line)
511 catch err
512 err isa CommandError || rethrow()
513 ["ERROR $(err.reason)"]
514 end
515 active = active_sprint(t)
516 !isnothing(active) && (active.records[t.day] = remaining(t, active))
517 return result
518end

520 tracker = Tracker()
521 count = parse(Int, readline())
522 for _ in 1:count
523 for line in execute!(tracker, readline())
524 println(line)
525 end
526 end
