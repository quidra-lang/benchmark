13fn equal(a: Text, b: Text) bool {
14 return std.mem.eql(u8, a, b);
15}
16fn number(text: Text) u32 {
17 return std.fmt.parseInt(u32, text, 10) catch unreachable;
18}
19fn line(out: *Writer, comptime format: Text, arguments: anytype) Writer.Error!void {
20 try out.print(format ++ "\n", arguments);
21}
22fn lower(gpa: Allocator, text: Text) Allocator.Error!Text {
23 const result = try gpa.dupe(u8, text);
24 for (result) |*character| character.* = std.ascii.toLower(character.*);
25 return result;
26}
27fn workflow(kind: Kind) []const State {
28 return switch (kind) {
29 .BUG => &.{ .OPEN, .TRIAGED, .IN_PROGRESS, .REVIEW, .RESOLVED, .CLOSED, .REOPENED },
30 .TASK => &.{ .OPEN, .IN_PROGRESS, .DONE },
31 .STORY => &.{ .BACKLOG, .READY, .IN_PROGRESS, .REVIEW, .DONE },
32 };
33}
34fn transition(kind: Kind, from: State, target: State) bool {
35 const edges: []const [2]State = switch (kind) {
36 .BUG => &.{ .{ .OPEN, .TRIAGED }, .{ .TRIAGED, .IN_PROGRESS }, .{ .IN_PROGRESS, .REVIEW }, .{ .REVIEW, .IN_PROGRESS }, .{ .REVIEW, .RESOLVED }, .{ .RESOLVED, .CLOSED }, .{ .RESOLVED, .REOPENED }, .{ .CLOSED, .REOPENED }, .{ .REOPENED, .IN_PROGRESS } },
37 .TASK => &.{ .{ .OPEN, .IN_PROGRESS }, .{ .IN_PROGRESS, .OPEN }, .{ .IN_PROGRESS, .DONE } },
38 .STORY => &.{ .{ .BACKLOG, .READY }, .{ .READY, .BACKLOG }, .{ .READY, .IN_PROGRESS }, .{ .IN_PROGRESS, .REVIEW }, .{ .REVIEW, .IN_PROGRESS }, .{ .REVIEW, .DONE } },
39 };
40 for (edges) |edge| if (edge[0] == from and edge[1] == target) return true;
41 return false;
42}

103 var result: [100]usize = @splat(0);
104 for (0..tracker.issues.items.len) |index| result[index] = index;

114 fn unfinishedBlocker(tracker: *const Tracker, target: usize) ?usize {
115 var best: ?usize = null;
116 for (tracker.issues.items, 0..) |issue, index| {

118 if (best == null or issue.number < tracker.issues.items[best.?].number) best = index;

122 fn blocked(tracker: *const Tracker, index: usize) bool {

126 var count: usize = 0;
127 for (tracker.issues.items) |issue| {
128 if (issue.assignee == user and issue.state == .IN_PROGRESS) count += 1;
129 }
130 return count;
131 }
132 fn active(tracker: *const Tracker) ?usize {
133 for (tracker.sprints.items, 0..) |sprint, index| if (sprint.phase == .ACTIVE) return index;

135 }
136 fn summary(tracker: *const Tracker, sprint: usize) Summary {
137 var result: Summary = .{};
138 for (tracker.issues.items) |issue| {
139 if (issue.sprint != sprint) continue;
140 result.count += 1;
141 result.total += issue.points();
142 if (issue.done()) result.done += 1 else result.remaining += issue.points();
143 }
144 return result;
145 }
146 fn reaches(tracker: *const Tracker, start: usize, goal: usize) bool {
147 var queue: [100]usize = undefined;
148 var seen: [100]bool = @splat(false);
149 var head: usize = 0;
150 var tail: usize = 1;
151 queue[0] = start;
152 seen[start] = true;
153 while (head < tail) : (head += 1) {
154 const current = queue[head];
155 if (current == goal) return true;
156 for (0..tracker.issues.items.len) |index| {
157 if (!tracker.issues.items[current].blocks[index] or seen[index]) continue;
158 seen[index] = true;
159 queue[tail] = index;
160 tail += 1;
161 }
162 }
163 return false;
164 }

167fn compareNumber(actual: ?u32, operation: Operation, expected: ?u32) bool {
168 if (operation == .eq) return actual == expected;
169 if (operation == .ne) return actual != expected;
170 const left = actual orelse return false;
171 const right = expected.?;
172 return switch (operation) {
173 .lt => left < right,
174 .le => left <= right,
175 .gt => left > right,
176 .ge => left >= right,
177 else => unreachable,
178 };
179}

181fn conditionMatches(tracker: *const Tracker, index: usize, condition: Condition) bool {
182 const issue = tracker.issues.items[index];
183 const actual_number: ?u32 = switch (condition.field) {
184 .id => issue.number,
185 .priority => @backingInt(issue.priority),
186 .estimate => issue.estimate,
187 .comments => issue.comments,
188 else => null,
189 };
190 switch (condition.field) {
191 .id, .priority, .estimate, .comments => return compareNumber(actual_number, condition.operation, condition.expected),
192 else => {},
193 }
194 if (condition.field == .title) {
195 if (condition.value.len > issue.title.len) return false;
196 for (0..issue.title.len - condition.value.len + 1) |offset| {
197 var match = true;
198 for (condition.value, 0..) |character, position| {
199 if (std.ascii.toLower(issue.title[offset + position]) != character) {
200 match = false;
201 break;
202 }
203 }
204 if (match) return true;
205 }
206 return false;
207 }
208 const actual: Text = switch (condition.field) {
209 .kind => @tagName(issue.kind),
210 .status => @tagName(issue.state),
211 .assignee => tracker.assigneeName(issue.assignee),
212 .reviewer => tracker.assigneeName(issue.reviewer),
213 .sprint => tracker.sprintName(issue.sprint),
214 .blocked => if (tracker.blocked(index)) "yes" else "no",
215 .done => if (issue.done()) "yes" else "no",
216 else => unreachable,
217 };
218 return if (condition.operation == .eq) equal(actual, condition.value) else !equal(actual, condition.value);
219}

221const Parser = struct {
222 tracker: *const Tracker,
223 query: *const Query,
224 issue: usize,
225 position: usize = 0,
226
227 fn factor(parser: *Parser) bool {
228 const position = parser.position;
229 const token = parser.query.tokens[position];
230 parser.position += 1;
231 if (equal(token, "NOT")) return !parser.factor();
232 if (equal(token, "(")) {
233 const result = parser.expression();
234 parser.position += 1;
235 return result;
236 }
237 return conditionMatches(parser.tracker, parser.issue, parser.query.conditions[position].?);
238 }
239 fn term(parser: *Parser) bool {
240 var result = parser.factor();
241 while (parser.position < parser.query.count and equal(parser.query.tokens[parser.position], "AND")) {
242 parser.position += 1;
243 const right = parser.factor();
244 result = result and right;
245 }
246 return result;
247 }
248 fn expression(parser: *Parser) bool {
249 var result = parser.term();
250 while (parser.position < parser.query.count and equal(parser.query.tokens[parser.position], "OR")) {
251 parser.position += 1;
252 const right = parser.term();
253 result = result or right;
254 }
255 return result;
256 }

282 fn record(session: *Session, index: usize, comptime format: Text, arguments: anytype) Error!void {

288 fn notify(session: *Session, index: usize, before: bool) Error!void {
289 const after = session.tracker.blocked(index);
290 if (before != after) try line(session.out, "{s} {d}", .{ if (after) "BLOCKED" else "UNBLOCKED", session.tracker.issues.items[index].number });
291 }
292 fn changeState(session: *Session, index: usize, target: State) Error!void {
293 const tracker = session.tracker;
294 const ids = tracker.ordered();
295 var before: [100]bool = @splat(false);
296 for (0..tracker.issues.items.len) |affected| before[affected] = tracker.blocked(affected);
297 const issue = &tracker.issues.items[index];
298 try line(session.out, "MOVE {d} {s} {s}", .{ issue.number, @tagName(issue.state), @tagName(target) });
299 try session.record(index, "MOVE {s} {s}", .{ @tagName(issue.state), @tagName(target) });
300 issue.state = target;
301 for (ids[0..tracker.issues.items.len]) |affected| if (issue.blocks[affected]) try session.notify(affected, before[affected]);
302 }

310 fn printLinks(session: *Session, index: usize, kind: enum { blocks, blockers, duplicates }) Error!void {
311 const tracker = session.tracker;
312 const ids = tracker.ordered();
313 var first = true;
314 for (ids[0..tracker.issues.items.len]) |other_index| {
315 const included = switch (kind) {
316 .blocks => tracker.issues.items[index].blocks[other_index],
317 .blockers => tracker.issues.items[other_index].blocks[index],
318 .duplicates => tracker.issues.items[other_index].duplicate == index,
319 };
320 if (!included) continue;
321 if (!first) try session.out.writeByte(',');
322 try session.out.print("{d}", .{tracker.issues.items[other_index].number});
323 first = false;
324 }
325 if (first) try session.out.writeAll("-");
326 }
327 fn checkCondition(session: *Session, token: Text) Error!Condition {
328 var end: usize = 0;
329 while (end < token.len and std.ascii.isLower(token[end])) : (end += 1) {}
330 const field_name = token[0..end];
331 const rest = token[end..];
332 const operators = [_]Text{ ">=", "<=", "!=", "=", "<", ">", "~" };
333 const operations = [_]Operation{ .ge, .le, .ne, .eq, .lt, .gt, .contains };
334 var operator_index: ?usize = null;
335 for (operators, 0..) |candidate, index| {
336 if (std.mem.startsWith(u8, rest, candidate)) {
337 operator_index = index;
338 break;
339 }
340 }
341 const selected = operator_index orelse return session.failFormat("syntax near {s}", .{token});
342 const value = rest[operators[selected].len..];
343 if (field_name.len == 0 or value.len == 0) return session.failFormat("syntax near {s}", .{token});
344 const field: Field = if (equal(field_name, "type")) .kind else std.meta.stringToEnum(Field, field_name) orelse return session.failFormat("unknown field {s}", .{field_name});
345 if (equal(field_name, "kind")) return session.failFormat("unknown field {s}", .{field_name});
346 const operation = operations[selected];
347 const equality = operation == .eq or operation == .ne;
348 const ordered = field == .id or field == .priority or field == .estimate or field == .comments;
349 if ((field == .title and operation != .contains) or (ordered and operation == .contains) or (!ordered and field != .title and !equality)) return session.failFormat("bad operator {s}", .{token});
350 var result: Condition = .{ .field = field, .operation = operation, .value = value };
351 if (equality and equal(value, "-") and (field == .assignee or field == .reviewer or field == .estimate or field == .sprint)) return result;
352 if (field == .id or field == .estimate or field == .comments) {
353 var valid = value.len >= 1 and value.len <= 6;
354 for (value) |character| valid = valid and std.ascii.isDigit(character);
355 if (valid) {
356 result.expected = number(value);
357 return result;
358 }
359 }
360 if (field == .priority) {
361 if (std.meta.stringToEnum(Priority, value)) |priority| {
362 result.expected = @backingInt(priority);
363 return result;
364 }
365 }
366 if ((field == .kind and std.meta.stringToEnum(Kind, value) != null) or
367 (field == .status and std.meta.stringToEnum(State, value) != null) or
368 ((field == .assignee or field == .reviewer) and session.tracker.userIndex(value) != null) or
369 (field == .sprint and session.tracker.sprintIndex(value) != null) or
370 ((field == .blocked or field == .done) and (equal(value, "yes") or equal(value, "no")))) return result;
371 if (field == .title) {
372 result.value = try lower(session.tracker.gpa, value);
373 return result;
374 }
375 return session.failFormat("bad value {s}", .{token});
376 }
377 fn checkQuery(session: *Session, text: Text) Error!Query {
378 var query: Query = .{};
379 var position: usize = 0;
380 while (position < text.len) {
381 if (text[position] == ' ') {
382 position += 1;
383 continue;
384 }
385 const start = position;
386 if (text[position] == '(' or text[position] == ')') position += 1 else {
387 while (position < text.len and text[position] != ' ' and text[position] != '(' and text[position] != ')') : (position += 1) {}
388 }
389 query.tokens[query.count] = text[start..position];
390 query.count += 1;
391 }
392 var operand = true;
393 var depth: usize = 0;
394 for (query.tokens[0..query.count], 0..) |token, index| {
395 if (operand) {
396 if (equal(token, "(")) depth += 1 else if (equal(token, "AND") or equal(token, "OR") or equal(token, ")")) return session.failFormat("syntax near {s}", .{token}) else if (!equal(token, "NOT")) {
397 query.conditions[index] = try session.checkCondition(token);
398 operand = false;
399 }
400 } else if (equal(token, "AND") or equal(token, "OR")) operand = true else if (equal(token, ")") and depth > 0) depth -= 1 else return session.failFormat("syntax near {s}", .{token});
401 }
402 if (operand or depth != 0) return session.fail("syntax at end");
403 return query;
404 }
405 fn queryIssues(session: *Session, text: Text) Error!void {
406 const parsed = try session.checkQuery(text);
407 const tracker = session.tracker;
408 const ids = tracker.ordered();
409 var matches: [100]usize = undefined;
410 var count: usize = 0;
411 var rank: u32 = 4;
412 while (rank > 0) {
413 rank -= 1;
414 for (ids[0..tracker.issues.items.len]) |index| {
415 if (@backingInt(tracker.issues.items[index].priority) != rank) continue;
416 var parser: Parser = .{ .tracker = tracker, .query = &parsed, .issue = index };
417 if (parser.expression()) {
418 matches[count] = index;
419 count += 1;
420 }
421 }
422 }
423 try line(session.out, "QUERY {d}", .{count});
424 for (matches[0..count]) |index| {
425 const issue = tracker.issues.items[index];
426 try session.out.print("{d} {s} {s} {s} {s} ", .{ issue.number, @tagName(issue.kind), @tagName(issue.priority), @tagName(issue.state), tracker.assigneeName(issue.assignee) });
427 try session.printOptional(issue.estimate);
428 try line(session.out, " {s}", .{issue.title});
429 }
430 }
431 fn display(session: *Session, command: Text, words: []const Text) Error!void {
432 const tracker = session.tracker;
433 if (equal(command, "BOARD")) {
434 try line(session.out, "BOARD", .{});
435 inline for (.{ Kind.BUG, Kind.TASK, Kind.STORY }) |kind| {
436 try session.out.print("{s}", .{@tagName(kind)});
437 for (workflow(kind)) |state| {
438 var count: usize = 0;
439 for (tracker.issues.items) |issue| {
440 if (issue.kind == kind and issue.state == state) count += 1;
441 }
442 try session.out.print(" {s} {d}", .{ @tagName(state), count });
443 }
444 try session.out.writeByte('\n');
445 }
446 var finished: usize = 0;
447 for (tracker.issues.items) |issue| {
448 if (issue.done()) finished += 1;
449 }
450 try line(session.out, "TOTAL {d} open {d} done {d}", .{ tracker.issues.items.len, tracker.issues.items.len - finished, finished });
451 } else if (equal(command, "WORKLOAD")) {
452 var users: [20]Text = undefined;
453 @memcpy(users[0..tracker.users.items.len], tracker.users.items);
454 std.sort.heap(Text, users[0..tracker.users.items.len], {}, struct {
455 fn less(_: void, a: Text, b: Text) bool {
456 return std.mem.order(u8, a, b) == .lt;
457 }
458 }.less);
459 try line(session.out, "WORKLOAD {d}", .{tracker.users.items.len});
460 for (users[0..tracker.users.items.len]) |user| {
461 const index = tracker.userIndex(user).?;
462 var assigned: u32 = 0;
463 var progressing: u32 = 0;
464 var review: u32 = 0;
465 var reviewing: u32 = 0;
466 var total: u32 = 0;
467 for (tracker.issues.items) |issue| {
468 if (issue.assignee == index and !issue.done()) {
469 assigned += 1;
470 total += issue.points();
471 if (issue.state == .IN_PROGRESS) progressing += 1;
472 if (issue.state == .REVIEW) review += 1;
473 }
474 if (issue.reviewer == index and issue.state == .REVIEW) reviewing += 1;
475 }
476 try line(session.out, "{s} assigned {d} in_progress {d} review {d} reviewing {d} points {d}", .{ user, assigned, progressing, review, reviewing, total });
477 }
478 } else if (equal(command, "SPRINTS")) {
479 try line(session.out, "SPRINTS {d}", .{tracker.sprints.items.len});
480 for (tracker.sprints.items, 0..) |sprint, index| {
481 try session.out.print("{s} {s}", .{ sprint.name, @tagName(sprint.phase) });
482 if (sprint.phase == .ENDED) try session.out.print(" start {d} end {d} velocity {d}", .{ sprint.start, sprint.end, sprint.velocity }) else {
483 if (sprint.phase == .ACTIVE) try session.out.print(" start {d}", .{sprint.start});
484 try session.out.writeByte(' ');
485 try session.printSummary(index);
486 if (sprint.phase == .ACTIVE) try session.out.print(" remaining {d}", .{tracker.summary(index).remaining});
487 }
488 try session.out.writeByte('\n');
489 }
490 } else if (equal(command, "BURNDOWN")) {
491 const index = try session.requireSprint(words[1]);
492 const sprint = tracker.sprints.items[index];
493 if (sprint.phase == .PLANNED) return session.fail("sprint not started");
494 const last = if (sprint.phase == .ACTIVE) tracker.day else sprint.end;
495 try line(session.out, "BURNDOWN {s} {s} days {d}", .{ sprint.name, @tagName(sprint.phase), last - sprint.start + 1 });
496 var day = sprint.start;
497 var value: u32 = 0;
498 while (day <= last) : (day += 1) {
499 if (sprint.records[day]) |recorded| value = recorded;
500 try line(session.out, "day {d} {d}", .{ day, value });
501 }
502 } else {
503 const index = try session.requireIssue(words[1]);
504 const issue = tracker.issues.items[index];
505 if (equal(command, "HISTORY")) {
506 try line(session.out, "HISTORY {d} {d}", .{ issue.number, issue.history.items.len });
507 for (issue.history.items) |entry| try line(session.out, "{s}", .{entry});
508 return;
509 }
510 try line(session.out, "ISSUE {d} {s} {s} {s} blocked {s}", .{ issue.number, @tagName(issue.kind), @tagName(issue.priority), @tagName(issue.state), if (tracker.blocked(index)) "yes" else "no" });
511 try line(session.out, "TITLE {s}", .{issue.title});
512 try line(session.out, "PEOPLE assignee {s} reviewer {s}", .{ tracker.assigneeName(issue.assignee), tracker.assigneeName(issue.reviewer) });
513 try session.out.writeAll("PLAN estimate ");
514 try session.printOptional(issue.estimate);
515 try line(session.out, " sprint {s} comments {d}", .{ tracker.sprintName(issue.sprint), issue.comments });
516 try session.out.writeAll("LINKS blocks ");
517 try session.printLinks(index, .blocks);
518 try session.out.writeAll(" blocked_by ");
519 try session.printLinks(index, .blockers);
520 try session.out.writeAll(" duplicate_of ");
521 try session.printOptional(if (issue.duplicate) |target| tracker.issues.items[target].number else null);
522 try session.out.writeAll(" duplicated_by ");
523 try session.printLinks(index, .duplicates);
524 try session.out.writeByte('\n');
525 }
526 }
527 fn sprintCommand(session: *Session, words: []const Text) Error!void {
528 const tracker = session.tracker;
529 const action = words[1];
530 const name = words[2];
531 if (equal(action, "NEW")) {
532 if (tracker.sprintIndex(name) != null) return session.fail("sprint exists");
533 try tracker.sprints.append(tracker.gpa, .{ .name = name });
534 try line(session.out, "SPRINT NEW {s}", .{name});
535 return;
536 }
537 const index = try session.requireSprint(name);
538 const sprint = &tracker.sprints.items[index];
539 if (equal(action, "ADD") or equal(action, "REMOVE")) {
540 const issue_index = try session.requireIssue(words[3]);
541 const issue = &tracker.issues.items[issue_index];
542 if (sprint.phase == .ENDED) return session.fail("sprint ended");
543 if (equal(action, "ADD")) {
544 if (issue.done()) return session.fail("issue done");
545 if (issue.sprint) |previous| if (tracker.sprints.items[previous].phase != .ENDED) return session.failFormat("in sprint {s}", .{tracker.sprints.items[previous].name});
546 if (issue.state == .BACKLOG) return session.fail("not ready");
547 issue.sprint = index;
548 } else {
549 if (issue.sprint != index) return session.fail("not in sprint");
550 if (issue.done()) return session.fail("issue done");
551 issue.sprint = null;
552 }
553 try session.record(issue_index, "SPRINT {s} {s}", .{ action, name });
554 try session.out.print("SPRINT {s} {s} {d} ", .{ action, name, issue.number });
555 try session.printSummary(index);
556 try session.out.writeByte('\n');
557 } else if (equal(action, "START")) {
558 if (sprint.phase != .PLANNED) return session.fail("sprint not planned");
559 if (tracker.active()) |active| return session.failFormat("sprint active {s}", .{tracker.sprints.items[active].name});
560 const summary = tracker.summary(index);
561 if (summary.count == 0) return session.fail("sprint empty");
562 sprint.phase = .ACTIVE;
563 sprint.start = tracker.day;
564 sprint.records[tracker.day] = summary.remaining;
565 try line(session.out, "SPRINT START {s} day {d} issues {d} points {d} remaining {d}", .{ name, tracker.day, summary.count, summary.total, summary.remaining });
566 } else {
567 if (sprint.phase != .ACTIVE) return session.fail("sprint not active");
568 const summary = tracker.summary(index);
569 sprint.records[tracker.day] = summary.remaining;
570 sprint.velocity = summary.total - summary.remaining;
571 sprint.phase = .ENDED;
572 sprint.end = tracker.day;
573 try line(session.out, "SPRINT END {s} day {d} velocity {d} done {d} returned {d}", .{ name, tracker.day, sprint.velocity, summary.done, summary.count - summary.done });
574 const ids = tracker.ordered();
575 for (ids[0..tracker.issues.items.len]) |issue_index| {
576 const issue = &tracker.issues.items[issue_index];
577 if (issue.sprint != index or issue.done()) continue;
578 issue.sprint = null;
579 try session.record(issue_index, "SPRINT RETURN {s}", .{name});
580 try line(session.out, "RETURNED {d}", .{issue.number});
581 }
582 }
583 }
584 fn link(session: *Session, command: Text, words: []const Text) Error!void {
585 const tracker = session.tracker;
586 const unlink = equal(command, "UNLINK");
587 const blocks = equal(words[2], "BLOCKS");
588 if ((unlink and !blocks) or (!blocks and !equal(words[2], "DUPLICATES"))) return session.fail("bad link");
589 const a = try session.requireIssue(words[1]);
590 const b = try session.requireIssue(words[3]);
591 const source = &tracker.issues.items[a];
592 const target = &tracker.issues.items[b];
593 if (!unlink and a == b) return session.fail("self link");
594 const before = tracker.blocked(b);
595 if (unlink) {
596 if (!source.blocks[b]) return session.fail("no link");
597 source.blocks[b] = false;
598 } else if (blocks) {
599 if (source.blocks[b]) return session.fail("link exists");
600 if (tracker.reaches(b, a)) return session.fail("cycle");
601 if (target.done()) return session.fail("issue done");
602 source.blocks[b] = true;
603 } else {
604 if (source.kind != target.kind) return session.fail("type mismatch");
605 if (source.done()) return session.fail("issue done");
606 if (target.duplicate != null) return session.fail("target is duplicate");
607 source.duplicate = b;
608 }
609 try session.record(a, "{s} {s} {d}", .{ command, words[2], target.number });
610 try session.record(b, "{s} {s} {d}", .{ command, if (blocks) "BLOCKED_BY" else "DUPLICATED_BY", source.number });
611 try line(session.out, "{s} {d} {s} {d}", .{ command, source.number, words[2], target.number });
612 if (blocks) try session.notify(b, before) else {
613 try session.changeState(a, if (source.kind == .BUG) .CLOSED else .DONE);
614 if (source.sprint) |sprint| {
615 if (tracker.sprints.items[sprint].phase != .ENDED) {
616 try session.record(a, "SPRINT REMOVE {s}", .{tracker.sprints.items[sprint].name});
617 source.sprint = null;
618 }
619 }
620 }
621 }
622 fn execute(session: *Session, source: Text) Error!void {
623 const tracker = session.tracker;
624 var words: std.ArrayList(Text) = .empty;
625 var split = std.mem.splitScalar(u8, source, ' ');
626 while (split.next()) |word| try words.append(tracker.gpa, word);
627 const args = words.items;
628 const command = args[0];
629 if (equal(command, "QUERY")) return session.queryIssues(source[6..]);
630 if (equal(command, "SHOW") or equal(command, "HISTORY") or equal(command, "BOARD") or equal(command, "WORKLOAD") or equal(command, "SPRINTS") or equal(command, "BURNDOWN")) return session.display(command, args);
631 if (equal(command, "DAY")) {
632 const day = number(args[1]);
633 if (day < tracker.day) return session.fail("day backwards");
634 tracker.day = day;
635 try line(session.out, "DAY {d}", .{day});
636 return;
637 }
638 if (equal(command, "USER")) {
639 if (tracker.userIndex(args[1]) != null) return session.fail("user exists");
640 try tracker.users.append(tracker.gpa, args[1]);
641 try line(session.out, "USER {s}", .{args[1]});
642 return;
643 }
644 if (equal(command, "SPRINT")) return session.sprintCommand(args);
645 if (equal(command, "CREATE")) {
646 const id = number(args[1]);
647 if (tracker.issueIndex(id) != null) return session.fail("issue exists");
648 const kind = std.meta.stringToEnum(Kind, args[2]) orelse return session.fail("bad type");
649 const priority = std.meta.stringToEnum(Priority, args[3]) orelse return session.fail("bad priority");
650 const title = try std.mem.join(tracker.gpa, " ", args[4..]);
651 const index = tracker.issues.items.len;
652 const state = workflow(kind)[0];
653 try tracker.issues.append(tracker.gpa, .{ .number = id, .kind = kind, .priority = priority, .title = title, .state = state });
654 try session.record(index, "CREATE {s} {s}", .{ args[2], args[3] });
655 try line(session.out, "CREATE {d} {s} {s} {s}", .{ id, args[2], args[3], @tagName(state) });
656 return;
657 }
658 if (equal(command, "LINK") or equal(command, "UNLINK")) return session.link(command, args);
659 const index = try session.requireIssue(args[1]);
660 const issue = &tracker.issues.items[index];
661 if (equal(command, "COMMENT")) {
662 _ = try session.requireUser(args[2]);
663 issue.comments += 1;
664 const text = try std.mem.join(tracker.gpa, " ", args[3..]);
665 try session.record(index, "COMMENT {s} {s}", .{ args[2], text });
666 try line(session.out, "COMMENT {d} {d}", .{ issue.number, issue.comments });
667 } else if (equal(command, "PRIORITY")) {
668 const priority = std.meta.stringToEnum(Priority, args[2]) orelse return session.fail("bad priority");
669 if (issue.done()) return session.fail("issue done");
670 if (issue.priority == priority) return session.fail("no change");
671 const previous = issue.priority;
672 issue.priority = priority;
673 try session.record(index, "PRIORITY {s} {s}", .{ @tagName(previous), args[2] });
674 try line(session.out, "PRIORITY {d} {s} {s}", .{ issue.number, @tagName(previous), args[2] });
675 } else if (equal(command, "ASSIGN") or equal(command, "REVIEWER")) {
676 const user = try session.requireUser(args[2]);
677 if (issue.done()) return session.fail("issue done");
678 const assign = equal(command, "ASSIGN");
679 if (user == (if (assign) issue.assignee else issue.reviewer)) return session.fail("no change");
680 if (assign) {
681 if (user == null and (issue.state == .IN_PROGRESS or issue.state == .REVIEW)) return session.fail("assignee required");
682 if (issue.state == .IN_PROGRESS and tracker.wip(user) >= 3) return session.failFormat("wip limit {s}", .{args[2]});
683 if (issue.state == .REVIEW and user == issue.reviewer) return session.fail("reviewer is assignee");
684 issue.assignee = user;
685 } else {
686 if (user == null and issue.state == .REVIEW) return session.fail("reviewer required");
687 if (issue.state == .REVIEW and user == issue.assignee) return session.fail("reviewer is assignee");
688 issue.reviewer = user;
689 }
690 try session.record(index, "{s} {s}", .{ command, args[2] });
691 try line(session.out, "{s} {d} {s}", .{ command, issue.number, args[2] });
692 } else if (equal(command, "ESTIMATE")) {
693 const estimate = number(args[2]);
694 if (estimate != 0 and estimate != 1 and estimate != 2 and estimate != 3 and estimate != 5 and estimate != 8 and estimate != 13 and estimate != 21) return session.fail("bad estimate");
695 if (issue.done()) return session.fail("issue done");
696 if (issue.estimate == estimate) return session.fail("no change");
697 const previous = if (issue.estimate) |value| try std.fmt.allocPrint(tracker.gpa, "{d}", .{value}) else "-";
698 issue.estimate = estimate;
699 try session.record(index, "ESTIMATE {s} {d}", .{ previous, estimate });
700 try line(session.out, "ESTIMATE {d} {s} {d}", .{ issue.number, previous, estimate });
701 } else {
702 if (issue.duplicate) |target| return session.failFormat("duplicate of {d}", .{tracker.issues.items[target].number});
703 const target = std.meta.stringToEnum(State, args[2]) orelse return session.fail("bad state");
704 if (std.mem.indexOfScalar(State, workflow(issue.kind), target) == null) return session.fail("bad state");
705 if (!transition(issue.kind, issue.state, target)) return session.failFormat("cannot move {s} to {s}", .{ @tagName(issue.state), args[2] });
706 if (target == .IN_PROGRESS) {
707 const active = tracker.active();
708 if (issue.kind == .STORY and (active == null or issue.sprint != active)) return session.fail("not in active sprint");
709 if (issue.assignee == null) return session.fail("no assignee");
710 if (tracker.wip(issue.assignee) >= 3) return session.failFormat("wip limit {s}", .{tracker.assigneeName(issue.assignee)});
711 }
712 if (target == .REVIEW) {
713 if (issue.reviewer == null) return session.fail("no reviewer");
714 if (issue.reviewer == issue.assignee) return session.fail("reviewer is assignee");
715 }
716 if (target == .DONE or target == .RESOLVED) {
717 if (tracker.unfinishedBlocker(index)) |blocker| return session.failFormat("blocked by {d}", .{tracker.issues.items[blocker].number});
718 }
719 if (target == .READY and issue.estimate == null) return session.fail("no estimate");
720 try session.changeState(index, target);
721 }
722 }
723};

725fn readLine(reader: *std.Io.Reader, gpa: Allocator) !?Text {
726 var buffer: Writer.Allocating = .init(gpa);
727 defer buffer.deinit();
728 _ = try reader.streamDelimiterEnding(&buffer.writer, '\n');
729 if (reader.bufferedLen() > 0) reader.toss(1) else if (buffer.written().len == 0) return null;
730 return try buffer.toOwnedSlice();
731}

733pub fn main(init: std.process.Init) !void {
734 var storage: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
735 defer storage.deinit();
736 const gpa = storage.allocator();
737 var tracker: Tracker = .{ .gpa = gpa };
738 var input_buffer: [4096]u8 = undefined;
739 var output_buffer: [4096]u8 = undefined;
740 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
741 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
742 const first = std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r");
743 const count = number(first);
744 for (0..count) |_| {
745 const source = std.mem.trim(u8, (try readLine(&input.interface, gpa)).?, "\r");
746 var session: Session = .{ .tracker = &tracker, .out = &output.interface };
747 session.execute(source) catch |err| switch (err) {
748 error.Invalid => try line(&output.interface, "ERROR {s}", .{session.problem}),
749 else => return err,
750 };
751 if (tracker.active()) |active| tracker.sprints.items[active].records[tracker.day] = tracker.summary(active).remaining;
752 try output.interface.flush();
753 }
754}
