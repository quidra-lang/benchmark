1local priorities = {"LOW", "MEDIUM", "HIGH", "CRITICAL"}
2local kinds = {"BUG", "TASK", "STORY"}
3local workflows = {
4 BUG = {"OPEN", "TRIAGED", "IN_PROGRESS", "REVIEW", "RESOLVED", "CLOSED", "REOPENED"},
5 TASK = {"OPEN", "IN_PROGRESS", "DONE"},
6 STORY = {"BACKLOG", "READY", "IN_PROGRESS", "REVIEW", "DONE"},
7}
8local transitions = {
9 BUG = {"OPEN:TRIAGED", "TRIAGED:IN_PROGRESS", "IN_PROGRESS:REVIEW", "REVIEW:IN_PROGRESS",
10 "REVIEW:RESOLVED", "RESOLVED:CLOSED", "RESOLVED:REOPENED", "CLOSED:REOPENED", "REOPENED:IN_PROGRESS"},
11 TASK = {"OPEN:IN_PROGRESS", "IN_PROGRESS:OPEN", "IN_PROGRESS:DONE"},
12 STORY = {"BACKLOG:READY", "READY:BACKLOG", "READY:IN_PROGRESS", "IN_PROGRESS:REVIEW", "REVIEW:IN_PROGRESS", "REVIEW:DONE"},
13}
14local done_states = {RESOLVED = true, CLOSED = true, DONE = true}
15local estimates = {[0] = true, [1] = true, [2] = true, [3] = true, [5] = true, [8] = true, [13] = true, [21] = true}
16local equality = {"=", "!="}
17local ordered = {"=", "!=", "<", "<=", ">", ">="}
18local field_operators = {
19 id = ordered, type = equality, status = equality, priority = ordered,
20 assignee = equality, reviewer = equality, estimate = ordered, sprint = equality,
21 comments = ordered, blocked = equality, done = equality, title = {"~"},
22}
23local none_fields = {assignee = true, reviewer = true, estimate = true, sprint = true}

193function Tracker:reaches(start, goal)
194 local pending, visited = {start.number}, {}
195 while #pending > 0 do
196 local number = table.remove(pending)
197 if number == goal.number then return true end
198 if not visited[number] then
199 visited[number] = true
200 for blocked in pairs(self.issues[number].blocks) do pending[#pending + 1] = blocked end
201 end
202 end
203 return false
204end

206function Tracker:set_person(command, args)
207 local issue = self:issue(args[1])
208 local user = self:user(args[2])
209 self:not_done(issue)
210 if command == "ASSIGN" then
211 if issue.assignee == user then fail("no change") end
212 if user == nil and (issue.state == "IN_PROGRESS" or issue.state == "REVIEW") then fail("assignee required") end
213 if issue.state == "IN_PROGRESS" and self:wip(user) >= 3 then fail("wip limit " .. args[2]) end
214 if issue.state == "REVIEW" and user == issue.reviewer then fail("reviewer is assignee") end
215 issue.assignee = user
216 else
217 if issue.reviewer == user then fail("no change") end
218 if user == nil and issue.state == "REVIEW" then fail("reviewer required") end
219 if issue.state == "REVIEW" and user == issue.assignee then fail("reviewer is assignee") end
220 issue.reviewer = user
221 end
222 self:record(issue, command .. " " .. args[2])
223 self:emit(command, issue.number, args[2])
224end

226function Tracker:move(args)
227 local issue, target = self:issue(args[1]), args[2]
228 if issue.duplicate ~= nil then fail("duplicate of " .. issue.duplicate) end
229 if not contains(workflows[issue.kind], target) then fail("bad state") end
230 if not contains(transitions[issue.kind], issue.state .. ":" .. target) then
231 fail("cannot move " .. issue.state .. " to " .. target)
232 end
233 if target == "IN_PROGRESS" then
234 local active = self:active_sprint()
235 if issue.kind == "STORY" and (not active or issue.sprint ~= active.name) then fail("not in active sprint") end
236 if issue.assignee == nil then fail("no assignee") end
237 if self:wip(issue.assignee) >= 3 then fail("wip limit " .. issue.assignee) end
238 elseif target == "REVIEW" then
239 if issue.reviewer == nil then fail("no reviewer") end
240 if issue.reviewer == issue.assignee then fail("reviewer is assignee") end
241 end
242 if target == "DONE" or target == "RESOLVED" then
243 local blockers = self:unfinished(issue)
244 if #blockers > 0 then fail("blocked by " .. blockers[1]) end
245 end
246 if target == "READY" and issue.estimate == nil then fail("no estimate") end
247 self:change_state(issue, target)
248end

250function Tracker:link(command, args)
251 local relation = args[2]
252 if relation ~= "BLOCKS" and (command == "UNLINK" or relation ~= "DUPLICATES") then fail("bad link") end
253 local source = self:issue(args[1])
254 local target = self:issue(args[3])
255 if command == "UNLINK" then
256 if not source.blocks[target.number] then fail("no link") end
257 local before = self:blocked_states({target.number})
258 source.blocks[target.number], target.blockers[source.number] = nil, nil
259 self:record(source, "UNLINK BLOCKS " .. target.number)
260 self:record(target, "UNLINK BLOCKED_BY " .. source.number)
261 self:emit("UNLINK", source.number, "BLOCKS", target.number)
262 self:notify(before)
263 return
264 end
265 if source == target then fail("self link") end
266 if relation == "BLOCKS" then
267 if source.blocks[target.number] then fail("link exists") end
268 if self:reaches(target, source) then fail("cycle") end
269 self:not_done(target)
270 local before = self:blocked_states({target.number})
271 source.blocks[target.number], target.blockers[source.number] = true, true
272 self:record(source, "LINK BLOCKS " .. target.number)
273 self:record(target, "LINK BLOCKED_BY " .. source.number)
274 self:emit("LINK", source.number, relation, target.number)
275 self:notify(before)
276 else
277 if source.kind ~= target.kind then fail("type mismatch") end
278 self:not_done(source)
279 if target.duplicate ~= nil then fail("target is duplicate") end
280 source.duplicate = target.number
281 target.duplicates[source.number] = true
282 self:record(source, "LINK DUPLICATES " .. target.number)
283 self:record(target, "LINK DUPLICATED_BY " .. source.number)
284 self:emit("LINK", source.number, relation, target.number)
285 self:change_state(source, source.kind == "BUG" and "CLOSED" or "DONE")
286 if source.sprint ~= nil and self.sprints[source.sprint].phase ~= "ENDED" then
287 self:record(source, "SPRINT REMOVE " .. source.sprint)
288 source.sprint = nil
289 end
290 end
291end

293function Tracker:sprint_command(args)
294 local action, name = args[1], args[2]
295 if action == "NEW" then
296 if self.sprints[name] then fail("sprint exists") end
297 local sprint = {name = name, phase = "PLANNED", velocity = 0, records = {}}
298 self.sprints[name] = sprint
299 self.sprint_order[#self.sprint_order + 1] = sprint
300 self:emit("SPRINT", "NEW", name)
301 return
302 end
303 local sprint = self:sprint(name)
304 if action == "ADD" or action == "REMOVE" then
305 local issue = self:issue(args[3])
306 if sprint.phase == "ENDED" then fail("sprint ended") end
307 if action == "ADD" then
308 self:not_done(issue)
309 if issue.sprint ~= nil and self.sprints[issue.sprint].phase ~= "ENDED" then fail("in sprint " .. issue.sprint) end
310 if issue.state == "BACKLOG" then fail("not ready") end
311 issue.sprint = name
312 else
313 if issue.sprint ~= name then fail("not in sprint") end
314 self:not_done(issue)
315 issue.sprint = nil
316 end
317 self:record(issue, "SPRINT " .. action .. " " .. name)
318 self:emit("SPRINT", action, name, issue.number, self:summary(sprint))
319 elseif action == "START" then
320 if sprint.phase ~= "PLANNED" then fail("sprint not planned") end
321 local active = self:active_sprint()
322 if active then fail("sprint active " .. active.name) end
323 if #self:members(sprint) == 0 then fail("sprint empty") end
324 sprint.phase, sprint.start = "ACTIVE", self.day
325 sprint.records[self.day] = self:remaining(sprint)
326 self:emit("SPRINT", "START", name, "day", self.day, self:summary(sprint), "remaining", self:remaining(sprint))
327 else
328 if sprint.phase ~= "ACTIVE" then fail("sprint not active") end
329 sprint.records[self.day] = self:remaining(sprint)
330 local finished, returned, velocity = {}, {}, 0
331 for _, issue in ipairs(self:members(sprint)) do
332 if done(issue) then
333 finished[#finished + 1] = issue
334 velocity = velocity + points(issue)
335 else
336 returned[#returned + 1] = issue
337 end
338 end
339 sprint.velocity, sprint.phase, sprint.finish = velocity, "ENDED", self.day
340 self:emit("SPRINT", "END", name, "day", self.day, "velocity", velocity, "done", #finished, "returned", #returned)
341 for _, issue in ipairs(returned) do
342 issue.sprint = nil
343 self:record(issue, "SPRINT RETURN " .. name)
344 self:emit("RETURNED", issue.number)
345 end
346 end
347end

349function Tracker:condition(token)
350 local field = token:match("^[a-z]*")
351 local rest = token:sub(#field + 1)
352 local operator
353 for _, candidate in ipairs({">=", "<=", "!=", "=", "<", ">", "~"}) do
354 if rest:sub(1, #candidate) == candidate then
355 operator = candidate
356 break
357 end
358 end
359 local text = operator and rest:sub(#operator + 1) or ""
360 if field == "" or not operator or text == "" then fail("syntax near " .. token) end
361 if not field_operators[field] then fail("unknown field " .. field) end
362 if not contains(field_operators[field], operator) then fail("bad operator " .. token) end
363 local value
364 if text == "-" and none_fields[field] and contains(equality, operator) then
365 value = nil
366 elseif (field == "id" or field == "estimate" or field == "comments") and #text <= 6 and text:match("^[0-9]+$") then
367 value = tonumber(text)
368 elseif field == "priority" and contains(priorities, text) then
369 for index, priority in ipairs(priorities) do if priority == text then value = index end end
370 elseif field == "type" and workflows[text] then
371 value = text
372 elseif field == "status" then
373 for _, states in pairs(workflows) do if contains(states, text) then value = text end end
374 if value == nil then fail("bad value " .. token) end
375 elseif (field == "assignee" or field == "reviewer") and self.users[text] then
376 value = text
377 elseif field == "sprint" and self.sprints[text] then
378 value = text
379 elseif (field == "blocked" or field == "done") and (text == "yes" or text == "no") then
380 value = text
381 elseif field == "title" then
382 value = text:lower()
383 else
384 fail("bad value " .. token)
385 end
386 return {field = field, operator = operator, value = value}
387end

389function Tracker:compile_query(text)
390 local tokens, position = {}, 1
391 while position <= #text do
392 local character = text:sub(position, position)
393 if character == " " then
394 position = position + 1
395 elseif character == "(" or character == ")" then
396 tokens[#tokens + 1] = character
397 position = position + 1
398 else
399 local token = text:sub(position):match("^[^ ()]+")
400 tokens[#tokens + 1] = token
401 position = position + #token
402 end
403 end
404 local conditions, operand, depth = {}, true, 0
405 for index, token in ipairs(tokens) do
406 if operand then
407 if token == "(" then
408 depth = depth + 1
409 elseif token == "AND" or token == "OR" or token == ")" then
410 fail("syntax near " .. token)
411 elseif token ~= "NOT" then
412 conditions[index] = self:condition(token)
413 operand = false
414 end
415 elseif token == "AND" or token == "OR" then
416 operand = true
417 elseif token == ")" and depth > 0 then
418 depth = depth - 1
419 else
420 fail("syntax near " .. token)
421 end
422 end
423 if operand or depth ~= 0 then fail("syntax at end") end
424 local output, operators = {}, {}
425 local precedence = {OR = 1, AND = 2, NOT = 3}
426 for index, token in ipairs(tokens) do
427 if conditions[index] then
428 output[#output + 1] = conditions[index]
429 elseif token == "(" or token == "NOT" then
430 operators[#operators + 1] = token
431 elseif token == ")" then
432 while operators[#operators] ~= "(" do output[#output + 1] = table.remove(operators) end
433 table.remove(operators)
434 else
435 while #operators > 0 and operators[#operators] ~= "(" and precedence[operators[#operators]] >= precedence[token] do
436 output[#output + 1] = table.remove(operators)
437 end
438 operators[#operators + 1] = token
439 end
440 end
441 while #operators > 0 do output[#output + 1] = table.remove(operators) end
442 return output
443end

445function Tracker:field_value(issue, field)
446 if field == "id" then return issue.number end
447 if field == "type" then return issue.kind end
448 if field == "status" then return issue.state end
449 if field == "priority" then return level(issue) end
450 if field == "blocked" then return yes_no(self:blocked(issue)) end
451 if field == "done" then return yes_no(done(issue)) end
452 if field == "title" then return issue.title:lower() end
453 return issue[field]
454end

456function Tracker:matches(issue, program)
457 local stack = {}
458 for _, node in ipairs(program) do
459 if node == "NOT" then
460 local value = table.remove(stack)
461 stack[#stack + 1] = not value
462 elseif node == "AND" or node == "OR" then
463 local right, left = table.remove(stack), table.remove(stack)
464 if node == "AND" then
465 stack[#stack + 1] = left and right
466 else
467 stack[#stack + 1] = left or right
468 end
469 else
470 local actual = self:field_value(issue, node.field)
471 local expected, operator = node.value, node.operator
472 local value
473 if operator == "=" then value = actual == expected
474 elseif operator == "!=" then value = actual ~= expected
475 elseif operator == "~" then value = actual:find(expected, 1, true) ~= nil
476 elseif actual == nil then value = false
477 elseif operator == "<" then value = actual < expected
478 elseif operator == "<=" then value = actual <= expected
479 elseif operator == ">" then value = actual > expected
480 else value = actual >= expected end
481 stack[#stack + 1] = value
482 end
483 end
484 return stack[1]
485end

487function Tracker:query(text)
488 local program, matches = self:compile_query(text), {}
489 for _, issue in pairs(self.issues) do
490 if self:matches(issue, program) then matches[#matches + 1] = issue end
491 end
492 table.sort(matches, function(a, b)
493 if level(a) ~= level(b) then return level(a) > level(b) end
494 return a.number < b.number
495 end)
496 self:emit("QUERY", #matches)
497 for _, issue in ipairs(matches) do
498 self:emit(issue.number, issue.kind, issue.priority, issue.state, display(issue.assignee), display(issue.estimate), issue.title)
499 end
500end

511function Tracker:board()
512 self:emit("BOARD")
513 local total, finished = 0, 0
514 for _, issue in pairs(self.issues) do
515 total = total + 1
516 if done(issue) then finished = finished + 1 end
517 end
518 for _, kind in ipairs(kinds) do
519 local parts = {kind}
520 for _, state in ipairs(workflows[kind]) do
521 local count = 0
522 for _, issue in pairs(self.issues) do
523 if issue.kind == kind and issue.state == state then count = count + 1 end
524 end
525 parts[#parts + 1], parts[#parts + 2] = state, count
526 end
527 self:emit(table.unpack(parts))
528 end
529 self:emit("TOTAL", total, "open", total - finished, "done", finished)
530end

532function Tracker:workload()
533 local users = sorted_keys(self.users)
534 self:emit("WORKLOAD", #users)
535 for _, user in ipairs(users) do
536 local assigned, progress, review, reviewing, total = 0, 0, 0, 0, 0
537 for _, issue in pairs(self.issues) do
538 if issue.assignee == user and not done(issue) then
539 assigned = assigned + 1
540 total = total + points(issue)
541 if issue.state == "IN_PROGRESS" then progress = progress + 1 end
542 if issue.state == "REVIEW" then review = review + 1 end
543 end
544 if issue.reviewer == user and issue.state == "REVIEW" then reviewing = reviewing + 1 end
545 end
546 self:emit(user, "assigned", assigned, "in_progress", progress, "review", review, "reviewing", reviewing, "points", total)
547 end
548end

563function Tracker:burndown(sprint)
564 if sprint.phase == "PLANNED" then fail("sprint not started") end
565 local last = sprint.phase == "ACTIVE" and self.day or sprint.finish
566 self:emit("BURNDOWN", sprint.name, sprint.phase, "days", last - sprint.start + 1)
567 local value = 0
568 for day = sprint.start, last do
569 if sprint.records[day] ~= nil then value = sprint.records[day] end
570 self:emit("day", day, value)

574function Tracker:process(command, args)
575 if command == "DAY" then
576 local day = tonumber(args[1])
577 if day < self.day then fail("day backwards") end
578 self.day = day
579 self:emit("DAY", day)
580 elseif command == "USER" then
581 if self.users[args[1]] then fail("user exists") end
582 self.users[args[1]] = true
583 self:emit("USER", args[1])
584 elseif command == "CREATE" then
585 local number, kind, priority = tonumber(args[1]), args[2], args[3]
586 if self.issues[number] then fail("issue exists") end
587 if not workflows[kind] then fail("bad type") end
588 if not contains(priorities, priority) then fail("bad priority") end
589 local issue = {number = number, kind = kind, priority = priority, title = table.concat(args, " ", 4),
590 state = workflows[kind][1], comments = 0, blocks = {}, blockers = {}, duplicates = {}, history = {}}
591 self.issues[number] = issue
592 self:record(issue, "CREATE " .. kind .. " " .. priority)
593 self:emit("CREATE", number, kind, priority, issue.state)
594 elseif command == "PRIORITY" then
595 local issue, priority = self:issue(args[1]), args[2]
596 if not contains(priorities, priority) then fail("bad priority") end
597 self:not_done(issue)
598 if issue.priority == priority then fail("no change") end
599 local old = issue.priority
600 issue.priority = priority
601 self:record(issue, "PRIORITY " .. old .. " " .. priority)
602 self:emit("PRIORITY", issue.number, old, priority)
603 elseif command == "ASSIGN" or command == "REVIEWER" then
604 self:set_person(command, args)
605 elseif command == "ESTIMATE" then
606 local issue, estimate = self:issue(args[1]), tonumber(args[2])
607 if not estimates[estimate] then fail("bad estimate") end
608 self:not_done(issue)
609 if issue.estimate == estimate then fail("no change") end
610 local old = display(issue.estimate)
611 issue.estimate = estimate
612 self:record(issue, "ESTIMATE " .. old .. " " .. estimate)
613 self:emit("ESTIMATE", issue.number, old, estimate)
614 elseif command == "COMMENT" then
615 local issue = self:issue(args[1])
616 local user = self:user(args[2])
617 issue.comments = issue.comments + 1
618 self:record(issue, "COMMENT " .. user .. " " .. table.concat(args, " ", 3))
619 self:emit("COMMENT", issue.number, issue.comments)
620 elseif command == "MOVE" then self:move(args)
621 elseif command == "LINK" or command == "UNLINK" then self:link(command, args)
622 elseif command == "SPRINT" then self:sprint_command(args)
623 elseif command == "QUERY" then self:query(table.concat(args, " "))
624 elseif command == "SHOW" then self:show(self:issue(args[1]))
625 elseif command == "HISTORY" then
626 local issue = self:issue(args[1])
627 self:emit("HISTORY", issue.number, #issue.history)
628 for _, line in ipairs(issue.history) do self:emit(line) end
629 elseif command == "BOARD" then self:board()
630 elseif command == "WORKLOAD" then self:workload()
631 elseif command == "SPRINTS" then self:list_sprints()
632 elseif command == "BURNDOWN" then self:burndown(self:sprint(args[1])) end
633end

635function Tracker:execute(line)
636 self.output = {}
637 local words = split(line)
638 local command = table.remove(words, 1)
639 local ok, err = pcall(function() self:process(command, words) end)
640 if not ok then
641 if type(err) ~= "table" or not err.message then error(err, 0) end
642 self:emit("ERROR", err.message)
643 end
644 local active = self:active_sprint()
645 if active then active.records[self.day] = self:remaining(active) end
646 return self.output
647end
