95static int compare_issues(const void *a, const void *b) {
96 int left = order_tracker->issues[*(const int *)a].number;
97 int right = order_tracker->issues[*(const int *)b].number;
98 return (left > right) - (left < right);
99}

173static void record(Tracker *tracker, int index, char *format, ...) {
174 Issue *issue = &tracker->issues[index];
175 char content[128], line[160];
176 va_list arguments;
177 va_start(arguments, format);
178 vsnprintf(content, sizeof(content), format, arguments);
179 va_end(arguments);
180 snprintf(line, sizeof(line), "day %d %s", tracker->day, content);
181 if (issue->history_count == issue->history_capacity) {
182 int capacity = issue->history_capacity ? issue->history_capacity * 2 : 8;
183 char **grown = realloc(issue->history, (size_t)capacity * sizeof(*grown));
184 if (!grown)
185 exit(EXIT_FAILURE);
186 issue->history = grown;
187 issue->history_capacity = capacity;
188 }
189 char *copy = malloc(strlen(line) + 1);
190 if (!copy)
191 exit(EXIT_FAILURE);
192 strcpy(copy, line);
193 issue->history[issue->history_count++] = copy;
194}

206 for (int i = 0; i < tracker->issue_count; ++i)
207 before[i] = blocked(tracker, i);

213 notify(tracker, ids[i], before[ids[i]]);
214}

216static int reaches(Tracker *tracker, int start, int goal) {
217 int queue[100], head = 0, tail = 0, seen[100] = {0};
218 queue[tail++] = start;
219 seen[start] = 1;
220 while (head < tail) {
221 int current = queue[head++];
222 if (current == goal)
223 return 1;
224 for (int i = 0; i < tracker->issue_count; ++i)
225 if (tracker->issues[current].blocks[i] && !seen[i]) {
226 seen[i] = 1;
227 queue[tail++] = i;
228 }
229 }
230 return 0;
231}

233static int transition(int type, int from, int target) {
234 static int edges[][3] = {
235 {BUG, OPEN, TRIAGED}, {BUG, TRIAGED, IN_PROGRESS}, {BUG, IN_PROGRESS, REVIEW},
236 {BUG, REVIEW, IN_PROGRESS}, {BUG, REVIEW, RESOLVED}, {BUG, RESOLVED, CLOSED},
237 {BUG, RESOLVED, REOPENED}, {BUG, CLOSED, REOPENED}, {BUG, REOPENED, IN_PROGRESS},
238 {TASK, OPEN, IN_PROGRESS}, {TASK, IN_PROGRESS, OPEN}, {TASK, IN_PROGRESS, DONE},
239 {STORY, BACKLOG, READY}, {STORY, READY, BACKLOG}, {STORY, READY, IN_PROGRESS},
240 {STORY, IN_PROGRESS, REVIEW}, {STORY, REVIEW, IN_PROGRESS}, {STORY, REVIEW, DONE}};
241 for (size_t i = 0; i < sizeof(edges) / sizeof(edges[0]); ++i)
242 if (edges[i][0] == type && edges[i][1] == from && edges[i][2] == target)
243 return 1;
244 return 0;
245}

253static int numeric_text(char *text) {
254 size_t size = strlen(text);
255 if (!size || size > 6)
256 return 0;
257 for (size_t i = 0; i < size; ++i)
258 if (!isdigit((unsigned char)text[i]))
259 return 0;
260 return 1;
261}

263static int check_condition(Tracker *tracker, char *token, Condition *result) {
264 char field[101], *operations[] = {">=", "<=", "!=", "=", "<", ">", "~"};
265 int length = 0;
266 while ('a' <= token[length] && token[length] <= 'z')
267 ++length;
268 memcpy(field, token, (size_t)length);
269 field[length] = '\0';
270 char *rest = token + length, *operation = NULL;
271 for (int i = 0; i < 7; ++i)
272 if (strncmp(rest, operations[i], strlen(operations[i])) == 0) {
273 operation = operations[i];
274 break;
275 }
276 char *value = operation ? rest + strlen(operation) : rest;
277 if (!length || !operation || !*value) {
278 printf("ERROR syntax near %s\n", token);
279 return 0;
280 }
281 char *fields[] = {"id", "type", "status", "priority", "assignee", "reviewer",
282 "estimate", "sprint", "comments", "blocked", "done", "title"};
283 int field_id = name_index(fields, 12, field);
284 if (field_id < 0) {
285 printf("ERROR unknown field %s\n", field);
286 return 0;
287 }
288 int equality = strcmp(operation, "=") == 0 || strcmp(operation, "!=") == 0;
289 int ordered =
290 field_id == ID || field_id == PRIORITY || field_id == ESTIMATE || field_id == COMMENTS;
291 if ((field_id == TITLE && strcmp(operation, "~") != 0) ||
292 (ordered && strcmp(operation, "~") == 0) || (!ordered && field_id != TITLE && !equality)) {
293 printf("ERROR bad operator %s\n", token);
294 return 0;
295 }
296 result->field = field_id;
297 result->number = -1;
298 strcpy(result->operation, operation);
299 strcpy(result->value, value);
300 if (strcmp(value, "-") == 0 && equality &&
301 (field_id == ASSIGNEE || field_id == REVIEWER || field_id == ESTIMATE ||
302 field_id == SPRINT))
303 return 1;
304 if ((field_id == ID || field_id == ESTIMATE || field_id == COMMENTS) && numeric_text(value)) {
305 result->number = atoi(value);
306 return 1;
307 }
308 if (field_id == PRIORITY && (result->number = name_index(priorities, 4, value)) >= 0)
309 return 1;
310 if ((field_id == TYPE && name_index(type_names, 3, value) >= 0) ||
311 (field_id == STATUS && name_index(state_names, 10, value) >= 0) ||
312 ((field_id == ASSIGNEE || field_id == REVIEWER) && user_index(tracker, value) >= 0) ||
313 (field_id == SPRINT && sprint_index(tracker, value) >= 0) ||
314 ((field_id == BLOCKED || field_id == FINISHED) &&
315 (strcmp(value, "yes") == 0 || strcmp(value, "no") == 0)))
316 return 1;
317 if (field_id == TITLE) {
318 lowercase(value, result->value);
319 return 1;
320 }
321 printf("ERROR bad value %s\n", token);
322 return 0;
323}

325static int check_query(Tracker *tracker, char *text, Query *query) {
326 query->count = query->position = 0;
327 for (int position = 0; text[position];) {
328 if (text[position] == ' ') {
329 ++position;
330 continue;
331 }
332 char *token = query->tokens[query->count++];
333 int size = 0;
334 if (text[position] == '(' || text[position] == ')')
335 token[size++] = text[position++];
336 else
337 while (text[position] && text[position] != ' ' && text[position] != '(' &&
338 text[position] != ')')
339 token[size++] = text[position++];
340 token[size] = '\0';
341 }
342 int operand = 1, depth = 0;
343 for (int i = 0; i < query->count; ++i) {
344 char *token = query->tokens[i];
345 if (operand) {
346 if (strcmp(token, "(") == 0)
347 ++depth;
348 else if (strcmp(token, "AND") == 0 || strcmp(token, "OR") == 0 ||
349 strcmp(token, ")") == 0) {
350 printf("ERROR syntax near %s\n", token);
351 return 0;
352 } else if (strcmp(token, "NOT") != 0) {
353 if (!check_condition(tracker, token, &query->conditions[i]))
354 return 0;
355 operand = 0;
356 }
357 } else if (strcmp(token, "AND") == 0 || strcmp(token, "OR") == 0)
358 operand = 1;
359 else if (strcmp(token, ")") == 0 && depth > 0)
360 --depth;
361 else {
362 printf("ERROR syntax near %s\n", token);
363 return 0;
364 }
365 }
366 if (operand || depth) {
367 puts("ERROR syntax at end");
368 return 0;
369 }
370 return 1;
371}

373static int matches(Tracker *tracker, int id, Condition *condition) {
374 Issue *issue = &tracker->issues[id];
375 int field = condition->field;
376 if (field == ID || field == PRIORITY || field == ESTIMATE || field == COMMENTS) {
377 int actual = field == ID ? issue->number
378 : field == PRIORITY ? issue->priority
379 : field == ESTIMATE ? issue->estimate
380 : issue->comments;
381 int expected = condition->number;
382 char *operation = condition->operation;
383 if (strcmp(operation, "=") == 0)
384 return actual == expected;
385 if (strcmp(operation, "!=") == 0)
386 return actual != expected;
387 if (actual < 0)
388 return 0;
389 if (strcmp(operation, "<") == 0)
390 return actual < expected;
391 if (strcmp(operation, "<=") == 0)
392 return actual <= expected;
393 if (strcmp(operation, ">") == 0)
394 return actual > expected;
395 return actual >= expected;
396 }
397 char title[41], *actual;
398 if (field == TYPE)
399 actual = type_names[issue->type];
400 else if (field == STATUS)
401 actual = state_names[issue->state];
402 else if (field == ASSIGNEE)
403 actual = user_name(tracker, issue->assignee);
404 else if (field == REVIEWER)
405 actual = user_name(tracker, issue->reviewer);
406 else if (field == SPRINT)
407 actual = sprint_name(tracker, issue->sprint);
408 else if (field == BLOCKED)
409 actual = blocked(tracker, id) ? "yes" : "no";
410 else if (field == FINISHED)
411 actual = done(issue) ? "yes" : "no";
412 else {
413 lowercase(issue->title, title);
414 actual = title;
415 }
416 if (field == TITLE)
417 return strstr(actual, condition->value) != NULL;
418 int equal = strcmp(actual, condition->value) == 0;
419 return strcmp(condition->operation, "=") == 0 ? equal : !equal;
420}

424static int query_factor(Tracker *tracker, Query *query, int issue) {
425 char *token = query->tokens[query->position++];
426 if (strcmp(token, "NOT") == 0)
427 return !query_factor(tracker, query, issue);
428 if (strcmp(token, "(") == 0) {
429 int result = query_expression(tracker, query, issue);
430 ++query->position;
431 return result;
432 }
433 return matches(tracker, issue, &query->conditions[query->position - 1]);
434}

436static int query_term(Tracker *tracker, Query *query, int issue) {
437 int result = query_factor(tracker, query, issue);
438 while (query->position < query->count && strcmp(query->tokens[query->position], "AND") == 0) {
439 ++query->position;
440 int right = query_factor(tracker, query, issue);
441 result = result && right;
442 }
443 return result;
444}

446static int query_expression(Tracker *tracker, Query *query, int issue) {
447 int result = query_term(tracker, query, issue);
448 while (query->position < query->count && strcmp(query->tokens[query->position], "OR") == 0) {
449 ++query->position;
450 int right = query_term(tracker, query, issue);
451 result = result || right;
452 }
453 return result;
454}

490static void execute(Tracker *tracker, char *words[], int word_count) {
491 char *command = words[0];
492 int ids[100];
493 ordered_issues(tracker, ids);
494 if (strcmp(command, "DAY") == 0) {
495 int day = atoi(words[1]);
496 if (day < tracker->day)
497 REJECT("day backwards");
498 tracker->day = day;
499 printf("DAY %d\n", day);
500 return;
501 }
502 if (strcmp(command, "USER") == 0) {
503 if (user_index(tracker, words[1]) >= 0)
504 REJECT("user exists");
505 strcpy(tracker->users[tracker->user_count++], words[1]);
506 printf("USER %s\n", words[1]);
507 return;
508 }
509 if (strcmp(command, "CREATE") == 0) {
510 int number = atoi(words[1]), type = name_index(type_names, 3, words[2]),
511 priority = name_index(priorities, 4, words[3]);
512 if (issue_index(tracker, number) >= 0)
513 REJECT("issue exists");
514 if (type < 0)
515 REJECT("bad type");
516 if (priority < 0)
517 REJECT("bad priority");
518 int index = tracker->issue_count++;
519 Issue *issue = &tracker->issues[index];
520 issue->number = number;
521 issue->type = type;
522 issue->priority = priority;
523 issue->state = workflows[type][0];
524 issue->assignee = issue->reviewer = issue->estimate = issue->sprint = issue->duplicate = -1;
525 join_words(words, 4, word_count, issue->title);
526 record(tracker, index, "CREATE %s %s", words[2], words[3]);
527 printf("CREATE %d %s %s %s\n", number, words[2], words[3], state_names[issue->state]);
528 return;
529 }
530 if (strcmp(command, "QUERY") == 0) {
531 char text[101];
532 Query query;
533 join_words(words, 1, word_count, text);
534 if (!check_query(tracker, text, &query))
535 return;
536 int result[100], count = 0;
537 for (int priority = 3; priority >= 0; --priority)
538 for (int i = 0; i < tracker->issue_count; ++i) {
539 int id = ids[i];
540 query.position = 0;
541 if (tracker->issues[id].priority == priority &&
542 query_expression(tracker, &query, id))
543 result[count++] = id;
544 }
545 printf("QUERY %d\n", count);
546 for (int i = 0; i < count; ++i) {
547 Issue *issue = &tracker->issues[result[i]];
548 printf("%d %s %s %s %s ", issue->number, type_names[issue->type],
549 priorities[issue->priority], state_names[issue->state],
550 user_name(tracker, issue->assignee));
551 print_optional(issue->estimate);
552 printf(" %s\n", issue->title);
553 }
554 return;
555 }
556 if (strcmp(command, "BOARD") == 0) {
557 puts("BOARD");
558 int finished = 0;
559 for (int type = 0; type < 3; ++type) {
560 printf("%s", type_names[type]);
561 for (int state = 0; state < state_counts[type]; ++state) {
562 int count = 0;
563 for (int i = 0; i < tracker->issue_count; ++i)
564 count += tracker->issues[i].type == type &&
565 tracker->issues[i].state == workflows[type][state];
566 printf(" %s %d", state_names[workflows[type][state]], count);
567 }
568 putchar('\n');
569 }
570 for (int i = 0; i < tracker->issue_count; ++i)
571 finished += done(&tracker->issues[i]);
572 printf("TOTAL %d open %d done %d\n", tracker->issue_count, tracker->issue_count - finished,
573 finished);
574 return;
575 }
576 if (strcmp(command, "WORKLOAD") == 0) {
577 int users[20];
578 for (int i = 0; i < tracker->user_count; ++i)
579 users[i] = i;
580 for (int i = 1; i < tracker->user_count; ++i)
581 for (int j = i;
582 j > 0 && strcmp(tracker->users[users[j]], tracker->users[users[j - 1]]) < 0; --j) {
583 int temporary = users[j];
584 users[j] = users[j - 1];
585 users[j - 1] = temporary;
586 }
587 printf("WORKLOAD %d\n", tracker->user_count);
588 for (int i = 0; i < tracker->user_count; ++i) {
589 int user = users[i], assigned = 0, progressing = 0, review = 0, reviewing = 0,
590 total = 0;
591 for (int j = 0; j < tracker->issue_count; ++j) {
592 Issue *issue = &tracker->issues[j];
593 if (issue->assignee == user && !done(issue)) {
594 ++assigned;
595 total += points(issue);
596 progressing += issue->state == IN_PROGRESS;
597 review += issue->state == REVIEW;
598 }
599 reviewing += issue->reviewer == user && issue->state == REVIEW;
600 }
601 printf("%s assigned %d in_progress %d review %d reviewing %d points %d\n",
602 tracker->users[user], assigned, progressing, review, reviewing, total);
603 }
604 return;
605 }
606 if (strcmp(command, "SPRINTS") == 0) {
607 printf("SPRINTS %d\n", tracker->sprint_count);
608 for (int i = 0; i < tracker->sprint_count; ++i) {
609 Sprint *sprint = &tracker->sprints[i];
610 printf("%s %s", sprint->name, phase_names[sprint->phase]);
611 if (sprint->phase == ENDED)
612 printf(" start %d end %d velocity %d", sprint->start, sprint->end,
613 sprint->velocity);
614 else {
615 if (sprint->phase == ACTIVE)
616 printf(" start %d", sprint->start);
617 putchar(' ');
618 print_summary(tracker, i);
619 if (sprint->phase == ACTIVE)
620 printf(" remaining %d", remaining(tracker, i));
621 }
622 putchar('\n');
623 }
624 return;
625 }
626 if (strcmp(command, "BURNDOWN") == 0 || strcmp(command, "SPRINT") == 0) {
627 int burndown = strcmp(command, "BURNDOWN") == 0;
628 char *name = words[burndown ? 1 : 2], *action = burndown ? "" : words[1];
629 int index = sprint_index(tracker, name);
630 if (!burndown && strcmp(action, "NEW") == 0) {
631 if (index >= 0)
632 REJECT("sprint exists");
633 Sprint *sprint = &tracker->sprints[tracker->sprint_count++];
634 strcpy(sprint->name, name);
635 for (int i = 0; i < 366; ++i)
636 sprint->records[i] = -1;
637 printf("SPRINT NEW %s\n", name);
638 return;
639 }
640 if (index < 0)
641 REJECT("unknown sprint");
642 Sprint *sprint = &tracker->sprints[index];
643 if (burndown) {
644 if (sprint->phase == PLANNED)
645 REJECT("sprint not started");
646 int last = sprint->phase == ACTIVE ? tracker->day : sprint->end, value = 0;
647 printf("BURNDOWN %s %s days %d\n", name, phase_names[sprint->phase],
648 last - sprint->start + 1);
649 for (int day = sprint->start; day <= last; ++day) {
650 if (sprint->records[day] >= 0)
651 value = sprint->records[day];
652 printf("day %d %d\n", day, value);
653 }
654 } else if (strcmp(action, "ADD") == 0 || strcmp(action, "REMOVE") == 0) {
655 int id = issue_index(tracker, atoi(words[3]));
656 if (id < 0)
657 REJECT("unknown issue");
658 Issue *issue = &tracker->issues[id];
659 if (sprint->phase == ENDED)
660 REJECT("sprint ended");
661 if (strcmp(action, "ADD") == 0) {
662 if (done(issue))
663 REJECT("issue done");
664 if (issue->sprint >= 0 && tracker->sprints[issue->sprint].phase != ENDED)
665 REJECT("in sprint %s", tracker->sprints[issue->sprint].name);
666 if (issue->state == BACKLOG)
667 REJECT("not ready");
668 issue->sprint = index;
669 } else {
670 if (issue->sprint != index)
671 REJECT("not in sprint");
672 if (done(issue))
673 REJECT("issue done");
674 issue->sprint = -1;
675 }
676 record(tracker, id, "SPRINT %s %s", action, name);
677 printf("SPRINT %s %s %d ", action, name, issue->number);
678 print_summary(tracker, index);
679 putchar('\n');
680 } else if (strcmp(action, "START") == 0) {
681 if (sprint->phase != PLANNED)
682 REJECT("sprint not planned");
683 int active = active_sprint(tracker), count = 0;
684 if (active >= 0)
685 REJECT("sprint active %s", tracker->sprints[active].name);
686 for (int i = 0; i < tracker->issue_count; ++i)
687 count += tracker->issues[i].sprint == index;
688 if (!count)
689 REJECT("sprint empty");
690 sprint->phase = ACTIVE;
691 sprint->start = tracker->day;
692 sprint->records[tracker->day] = remaining(tracker, index);
693 printf("SPRINT START %s day %d ", name, tracker->day);
694 print_summary(tracker, index);
695 printf(" remaining %d\n", remaining(tracker, index));
696 } else {
697 if (sprint->phase != ACTIVE)
698 REJECT("sprint not active");
699 sprint->records[tracker->day] = remaining(tracker, index);
700 int finished = 0, returned = 0;
701 for (int i = 0; i < tracker->issue_count; ++i)
702 if (tracker->issues[i].sprint == index) {
703 if (done(&tracker->issues[i])) {
704 ++finished;
705 sprint->velocity += points(&tracker->issues[i]);
706 } else
707 ++returned;
708 }
709 sprint->phase = ENDED;
710 sprint->end = tracker->day;
711 printf("SPRINT END %s day %d velocity %d done %d returned %d\n", name, tracker->day,
712 sprint->velocity, finished, returned);
713 for (int i = 0; i < tracker->issue_count; ++i) {
714 int id = ids[i];
715 if (tracker->issues[id].sprint == index && !done(&tracker->issues[id])) {
716 tracker->issues[id].sprint = -1;
717 record(tracker, id, "SPRINT RETURN %s", name);
718 printf("RETURNED %d\n", tracker->issues[id].number);
719 }
720 }
721 }
722 return;
723 }
724 if (strcmp(command, "LINK") == 0 || strcmp(command, "UNLINK") == 0) {
725 int unlink = strcmp(command, "UNLINK") == 0, blocks = strcmp(words[2], "BLOCKS") == 0;
726 if ((!blocks && unlink) || (!blocks && strcmp(words[2], "DUPLICATES") != 0))
727 REJECT("bad link");
728 int a = issue_index(tracker, atoi(words[1])), b = issue_index(tracker, atoi(words[3]));
729 if (a < 0 || b < 0)
730 REJECT("unknown issue");
731 Issue *source = &tracker->issues[a], *target = &tracker->issues[b];
732 if (!unlink && a == b)
733 REJECT("self link");
734 int before = blocked(tracker, b);
735 if (unlink) {
736 if (!source->blocks[b])
737 REJECT("no link");
738 source->blocks[b] = 0;
739 } else if (blocks) {
740 if (source->blocks[b])
741 REJECT("link exists");
742 if (reaches(tracker, b, a))
743 REJECT("cycle");
744 if (done(target))
745 REJECT("issue done");
746 source->blocks[b] = 1;
747 } else {
748 if (source->type != target->type)
749 REJECT("type mismatch");
750 if (done(source))
751 REJECT("issue done");
752 if (target->duplicate >= 0)
753 REJECT("target is duplicate");
754 source->duplicate = b;
755 }
756 record(tracker, a, "%s %s %d", command, words[2], target->number);
757 record(tracker, b, "%s %s %d", command, blocks ? "BLOCKED_BY" : "DUPLICATED_BY",
758 source->number);
759 printf("%s %d %s %d\n", command, source->number, words[2], target->number);
760 if (blocks)
761 notify(tracker, b, before);
762 else {
763 change_state(tracker, a, source->type == BUG ? CLOSED : DONE);
764 if (source->sprint >= 0 && tracker->sprints[source->sprint].phase != ENDED) {
765 record(tracker, a, "SPRINT REMOVE %s", tracker->sprints[source->sprint].name);
766 source->sprint = -1;
767 }
768 }
769 return;
770 }
771 int index = issue_index(tracker, atoi(words[1]));
772 if (index < 0)
773 REJECT("unknown issue");
774 Issue *issue = &tracker->issues[index];
775 int number = issue->number;
776 if (strcmp(command, "SHOW") == 0) {
777 printf("ISSUE %d %s %s %s blocked %s\nTITLE %s\n", number, type_names[issue->type],
778 priorities[issue->priority], state_names[issue->state],
779 blocked(tracker, index) ? "yes" : "no", issue->title);
780 printf("PEOPLE assignee %s reviewer %s\nPLAN estimate ",
781 user_name(tracker, issue->assignee), user_name(tracker, issue->reviewer));
782 print_optional(issue->estimate);
783 printf(" sprint %s comments %d\nLINKS blocks ", sprint_name(tracker, issue->sprint),
784 issue->comments);
785 print_links(tracker, index, 0);
786 printf(" blocked_by ");
787 print_links(tracker, index, 1);
788 printf(" duplicate_of ");
789 print_optional(issue->duplicate < 0 ? -1 : tracker->issues[issue->duplicate].number);
790 printf(" duplicated_by ");
791 print_links(tracker, index, 2);
792 putchar('\n');
793 } else if (strcmp(command, "HISTORY") == 0) {
794 printf("HISTORY %d %d\n", number, issue->history_count);
795 for (int i = 0; i < issue->history_count; ++i)
796 puts(issue->history[i]);
797 } else if (strcmp(command, "COMMENT") == 0) {
798 if (user_index(tracker, words[2]) < 0)
799 REJECT("unknown user");
800 char text[61];
801 join_words(words, 3, word_count, text);
802 ++issue->comments;
803 record(tracker, index, "COMMENT %s %s", words[2], text);
804 printf("COMMENT %d %d\n", number, issue->comments);
805 } else if (strcmp(command, "PRIORITY") == 0) {
806 int priority = name_index(priorities, 4, words[2]);
807 if (priority < 0)
808 REJECT("bad priority");
809 if (done(issue))
810 REJECT("issue done");
811 if (priority == issue->priority)
812 REJECT("no change");
813 printf("PRIORITY %d %s %s\n", number, priorities[issue->priority], words[2]);
814 record(tracker, index, "PRIORITY %s %s", priorities[issue->priority], words[2]);
815 issue->priority = priority;
816 } else if (strcmp(command, "ASSIGN") == 0 || strcmp(command, "REVIEWER") == 0) {
817 int user = user_index(tracker, words[2]), assign = strcmp(command, "ASSIGN") == 0;
818 if (user < 0 && strcmp(words[2], "-") != 0)
819 REJECT("unknown user");
820 if (done(issue))
821 REJECT("issue done");
822 if (user == (assign ? issue->assignee : issue->reviewer))
823 REJECT("no change");
824 if (assign) {
825 if (user < 0 && (issue->state == IN_PROGRESS || issue->state == REVIEW))
826 REJECT("assignee required");
827 if (issue->state == IN_PROGRESS && wip(tracker, user) >= 3)
828 REJECT("wip limit %s", words[2]);
829 if (issue->state == REVIEW && user == issue->reviewer)
830 REJECT("reviewer is assignee");
831 issue->assignee = user;
832 } else {
833 if (user < 0 && issue->state == REVIEW)
834 REJECT("reviewer required");
835 if (issue->state == REVIEW && user == issue->assignee)
836 REJECT("reviewer is assignee");
837 issue->reviewer = user;
838 }
839 record(tracker, index, "%s %s", command, words[2]);
840 printf("%s %d %s\n", command, number, words[2]);
841 } else if (strcmp(command, "ESTIMATE") == 0) {
842 int estimate = atoi(words[2]);
843 if (estimate != 0 && estimate != 1 && estimate != 2 && estimate != 3 && estimate != 5 &&
844 estimate != 8 && estimate != 13 && estimate != 21)
845 REJECT("bad estimate");
846 if (done(issue))
847 REJECT("issue done");
848 if (estimate == issue->estimate)
849 REJECT("no change");
850 char previous[16];
851 if (issue->estimate < 0)
852 strcpy(previous, "-");
853 else
854 snprintf(previous, sizeof(previous), "%d", issue->estimate);
855 issue->estimate = estimate;
856 record(tracker, index, "ESTIMATE %s %d", previous, estimate);
857 printf("ESTIMATE %d %s %d\n", number, previous, estimate);
858 } else {
859 int target = name_index(state_names, 10, words[2]), valid = 0;
860 if (issue->duplicate >= 0)
861 REJECT("duplicate of %d", tracker->issues[issue->duplicate].number);
862 for (int i = 0; i < state_counts[issue->type]; ++i)
863 valid |= workflows[issue->type][i] == target;
864 if (!valid)
865 REJECT("bad state");
866 if (!transition(issue->type, issue->state, target))
867 REJECT("cannot move %s to %s", state_names[issue->state], words[2]);
868 if (target == IN_PROGRESS) {
869 int active = active_sprint(tracker);
870 if (issue->type == STORY && (active < 0 || issue->sprint != active))
871 REJECT("not in active sprint");
872 if (issue->assignee < 0)
873 REJECT("no assignee");
874 if (wip(tracker, issue->assignee) >= 3)
875 REJECT("wip limit %s", user_name(tracker, issue->assignee));
876 }
877 if (target == REVIEW) {
878 if (issue->reviewer < 0)
879 REJECT("no reviewer");
880 if (issue->reviewer == issue->assignee)
881 REJECT("reviewer is assignee");
882 }
883 int blocker = unfinished_blocker(tracker, index);
884 if ((target == DONE || target == RESOLVED) && blocker >= 0)
885 REJECT("blocked by %d", tracker->issues[blocker].number);
886 if (target == READY && issue->estimate < 0)
887 REJECT("no estimate");
888 change_state(tracker, index, target);
889 }
890}

892int main(void) {
893 Tracker *tracker = calloc(1, sizeof(*tracker));
894 if (!tracker)
895 return 1;
896 tracker->day = 1;
897 char line[256];
898 if (!fgets(line, sizeof(line), stdin)) {
899 free(tracker);
900 return 0;
901 }
902 int count = atoi(line);
903 for (int command = 0; command < count; ++command) {
904 if (!fgets(line, sizeof(line), stdin))
905 break;
906 line[strcspn(line, "\r\n")] = '\0';
907 char *words[128], *token = strtok(line, " ");
908 int word_count = 0;
909 while (token) {
910 words[word_count++] = token;
911 token = strtok(NULL, " ");
912 }
913 execute(tracker, words, word_count);
914 int active = active_sprint(tracker);
915 if (active >= 0)
916 tracker->sprints[active].records[tracker->day] = remaining(tracker, active);
917 }
918 for (int i = 0; i < tracker->issue_count; ++i) {
919 for (int j = 0; j < tracker->issues[i].history_count; ++j)
920 free(tracker->issues[i].history[j]);
921 free(tracker->issues[i].history);
922 }
923 free(tracker);
924 return 0;
925}
