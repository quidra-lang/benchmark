32static Text fraction(int64_t numerator, int64_t denominator) {
33 int64_t a = numerator, b = denominator;
34 while (b) { int64_t rest = a % b; a = b; b = rest; }
35 numerator /= a;
36 denominator /= a;
37 Text text;
38 if (denominator == 1) snprintf(text.text, sizeof text.text, "%" PRId64, numerator);
39 else snprintf(text.text, sizeof text.text, "%" PRId64 "/%" PRId64, numerator, denominator);
40 return text;
41}

43static Text average(int64_t sum, int count) {
44 int64_t value = count ? 100 * sum / count : 0;
45 Text text;
46 snprintf(text.text, sizeof text.text, "%" PRId64 ".%02" PRId64, value / 100, value % 100);
47 return text;
48}

50static bool held(int mask, int die) { return (mask & (1 << (4 - die))) != 0; }

52static Text pattern_text(int mask) {
53 Text text;
54 for (int i = 0; i < DICE; i++) text.text[i] = held(mask, i) ? '1' : '0';
55 text.text[DICE] = '\0';
56 return text;
57}

59static Text face_text(const int *faces, int mask) {
60 Text text = {""};
61 int length = 0;
62 if (faces) for (int i = 0; i < DICE; i++) if (held(mask, i)) {
63 if (length) text.text[length++] = ' ';
64 text.text[length++] = (char)('0' + faces[i]);
65 }
66 if (!length) text.text[length++] = '-';
67 text.text[length] = '\0';
68 return text;

92static int normal_score(int category, const int counts[FACES]) {
93 if (category < 6) return (category + 1) * counts[category];
94 int total = 0, maximum = 0, bits = 0;
95 bool pair = false, triple = false;
96 for (int i = 0; i < FACES; i++) {
97 total += (i + 1) * counts[i];
98 if (counts[i] > maximum) maximum = counts[i];
99 if (counts[i]) bits |= 1 << i;
100 pair |= counts[i] == 2;
101 triple |= counts[i] == 3;
102 }
103 switch (category) {
104 case 6: return maximum >= 3 ? total : 0;
105 case 7: return maximum >= 4 ? total : 0;
106 case 8: return pair && triple ? 25 : 0;
107 case 9: return (bits & 15) == 15 || (bits & 30) == 30 || (bits & 60) == 60 ? 30 : 0;
108 case 10: return bits == 31 || bits == 62 ? 40 : 0;
109 case 11: return maximum == 5 ? 50 : 0;
110 default: return total;
111 }
112}

129 int total = player->yahtzee_bonus + upper_bonus(player);

131 return total;

140typedef struct { int points[CATEGORIES], allowed, best, yahtzee_face; bool joker; } Candidates;
141
142static Candidates evaluate(const Player *player, const int counts[FACES], JokerRule rule) {
143 Candidates result = {.best = -1, .yahtzee_face = -1};
144 for (int i = 0; i < FACES; i++) if (counts[i] == 5) result.yahtzee_face = i;
145 for (int c = 0; c < CATEGORIES; c++) if (player->scores[c] < 0) result.allowed |= 1 << c;
146 result.joker = rule != OFF && result.yahtzee_face >= 0 && player->scores[11] >= 0;
147 if (result.joker && rule == FORCED) {
148 int upper = 1 << result.yahtzee_face;
149 if (result.allowed & upper) result.allowed = upper;
150 else if (result.allowed & 0x1fc0) result.allowed &= 0x1fc0;
151 }
152 for (int c = 0; c < CATEGORIES; c++) {
153 int points = player->scores[c] >= 0 ? -1 : !(result.allowed & (1 << c)) ? 0
154 : result.joker && c >= 8 && c <= 10 ? c == 8 ? 25 : c == 9 ? 30 : 40 : normal_score(c, counts);
155 result.points[c] = points;
156 if (points >= 0 && (result.best < 0 || points > result.points[result.best])) result.best = c;
157 }
158 return result;
159}

165static void make_outcomes(int size, int face, int remaining, int code, int divisor) {
166 if (face == FACES) {
167 if (!remaining) outcomes[size][outcome_count[size]++] = (Outcome){code, factorial[size] / divisor};
168 return;
169 }
170 for (int n = 0; n <= remaining; n++) make_outcomes(size, face + 1, remaining - n, code + n * powers[face], divisor * factorial[n]);
171}

227static bool roll(Game *game) {
228 if (!check_turn(game)) return false;
229 Turn *turn = &game->turn;
230 if (turn->rolls == game->max_rolls) return fail("no rolls left");
231 if (turn->hold == 31) return fail("all dice held");
232 game->redo_count = 0;
233 for (int i = 0; i < DICE; i++) {
234 if (turn->rolls && held(turn->hold, i)) continue;
235 if (game->loaded_head < game->loaded_tail) turn->faces[i] = game->loaded[game->loaded_head++];
236 else {
237 game->random_state = 48271 * game->random_state % 2147483647;
238 turn->faces[i] = (int)(game->random_state % 6) + 1;
239 }
240 game->face_counts[turn->faces[i] - 1]++;
241 }
242 turn->rolls++;
243 if (game->verbose) printf("%s roll %d: %s\n", game->players[turn->seat].name, turn->rolls, face_text(turn->faces, 31).text);
244 return true;
245}

247static bool hold(Game *game, int mask) {

255static bool keep(Game *game, const int *values, int count) {
256 if (!check_rolls_left(game)) return false;
257 int mask = 0;
258 for (int j = 0; j < count; j++) {
259 int die = -1;
260 for (int i = 0; i < DICE; i++) if (!held(mask, i) && game->turn.faces[i] == values[j]) { die = i; break; }
261 if (die < 0) return fail("not showing %d", values[j]);
262 mask |= 1 << (4 - die);
263 }
264 return hold(game, mask);
265}

267static bool score(Game *game, int category, bool preview) {
268 if (!check_rolled(game)) return false;
269 Player *player = &game->players[game->turn.seat];
270 if (player->scores[category] >= 0) return fail("%s already used", categories[category]);
271 int counts[FACES];
272 face_counts(game->turn.faces, counts);
273 Candidates candidates = evaluate(player, counts, game->joker);
274 if (!(candidates.allowed & (1 << category))) {
275 return fail("joker requires %s", player->scores[candidates.yahtzee_face] < 0 ? categories[candidates.yahtzee_face] : "lower box");
276 }
277 int points = candidates.points[category];
278 int bonus = candidates.yahtzee_face >= 0 && player->scores[11] == 50 ? 100 : 0;
279 int upper = upper_subtotal(player);
280 int gained_upper = category < 6 && upper < 63 && upper + points >= 63 ? 35 : 0;
281 int total = player_total(player) + points + bonus + gained_upper;
282 if (game->verbose) {
283 printf("%s %s %d in %s", player->name, preview ? "would score" : "scores", points, categories[category]);
284 if (candidates.joker) printf(" joker");
285 if (bonus) printf(" ybonus 100");
286 if (gained_upper) printf(" ubonus 35");
287 printf(" total %d\n", total);
288 }
289 if (preview) return true;
290 game->redo_count = 0;
291 game->entries[game->entry_count++] = (ScoreEntry){game->turn, category, points, bonus, candidates.joker};
292 player->scores[category] = points;
293 player->yahtzee_bonus += bonus;
294 if (game->entry_count == CATEGORIES * game->player_count) game->turn = (Turn){.finished = true};
295 else game->turn = (Turn){.seat = (game->turn.seat + 1) % game->player_count};
296 if (game->verbose) {
297 if (game->turn.finished) puts("game over");
298 else {
299 Player *next = &game->players[game->turn.seat];
300 printf("turn %d %s\n", player_round(next), next->name);
301 }
302 }
303 return true;
304}

306static bool undo(Game *game) {
307 if (!check_game(game)) return false;
308 if (!game->entry_count) return fail("nothing to undo");
309 ScoreEntry entry = game->entries[--game->entry_count];
310 game->redo[game->redo_count++] = (RedoEntry){game->turn, entry};
311 Player *player = &game->players[entry.before.seat];
312 player->scores[entry.category] = -1;
313 player->yahtzee_bonus -= entry.bonus;
314 game->turn = entry.before;
315 printf("undo %s %s %d\n", player->name, categories[entry.category], entry.points);
316 return true;
317}

319static bool redo(Game *game) {
320 if (!check_game(game)) return false;
321 if (!game->redo_count) return fail("nothing to redo");
322 RedoEntry saved = game->redo[--game->redo_count];
323 ScoreEntry entry = saved.entry;
324 Player *player = &game->players[entry.before.seat];
325 player->scores[entry.category] = entry.points;
326 player->yahtzee_bonus += entry.bonus;
327 game->entries[game->entry_count++] = entry;
328 game->turn = saved.after;
329 printf("redo %s %s %d\n", player->name, categories[entry.category], entry.points);
330 return true;
331}

333static int bot_hold(const Game *game, Strategy strategy) {
334 const int *faces = game->turn.faces;
335 int counts[FACES];
336 face_counts(faces, counts);
337 int favorite = frequent(counts), mask = 0;
338 if (strategy == STRAIGHT) {
339 int best_start = 0, best_length = 0;
340 for (int start = 0; start < FACES; start++) {
341 int end = start;
342 while (end < FACES && counts[end]) end++;
343 if (end - start >= best_length) { best_start = start; best_length = end - start; }
344 }
345 if (best_length >= 3) {
346 for (int face = best_start + 1; face <= best_start + best_length; face++) {
347 for (int i = 0; i < DICE; i++) if (faces[i] == face) { mask |= 1 << (4 - i); break; }
348 }
349 return mask;
350 }
351 }
352 for (int i = 0; i < DICE; i++) if (strategy == HIGH ? faces[i] >= 5 : faces[i] == favorite + 1) mask |= 1 << (4 - i);
353 return mask;
354}

356static void bot_turn(Game *game) {
357 Player *player = &game->players[game->turn.seat];
358 Strategy strategy = player->strategy;
359 if (!game->turn.rolls) roll(game);
360 while (game->turn.rolls < game->max_rolls) {
361 int mask = bot_hold(game, strategy);
362 if (mask == 31) break;
363 hold(game, mask);
364 roll(game);
365 }
366 int counts[FACES];
367 face_counts(game->turn.faces, counts);
368 Candidates candidates = evaluate(player, counts, game->joker);
369 int favorite = frequent(counts);
370 int category = strategy == UPPER && counts[favorite] >= 3 && (candidates.allowed & (1 << favorite)) ? favorite : candidates.best;
371 score(game, category, false);
372}

375 for (int i = 0; i < game->player_count; i++) {
376 int j = i;
377 while (j > 0 && player_total(&game->players[i]) > player_total(&game->players[order[j - 1]])) { order[j] = order[j - 1]; j--; }
378 order[j] = i;

382typedef struct {

388static int64_t expected(Adviser *adviser, int code, int depth);

390static void search_holds(Adviser *adviser, const int counts[FACES], int face, int code, int depth, int64_t *best) {
391 if (face == FACES) {
392 int64_t value = expected(adviser, code, depth);
393 if (value > *best) *best = value;
394 return;
395 }
396 for (int n = 0; n <= counts[face]; n++) search_holds(adviser, counts, face + 1, code + n * powers[face], depth, best);
397}

399static int64_t advise_value(Adviser *adviser, int code, int depth) {
400 int64_t *memo = &adviser->values[depth][code];
401 if (*memo >= 0) return *memo;
402 int counts[FACES];
403 decode(code, counts);
404 if (!depth) {
405 Candidates candidates = evaluate(adviser->player, counts, adviser->joker);
406 *memo = candidates.points[candidates.best];
407 } else {
408 int64_t best = -1;
409 search_holds(adviser, counts, 0, 0, depth, &best);
410 *memo = best;
411 }
412 return *memo;
413}

415static int64_t expected(Adviser *adviser, int code, int depth) {
416 int64_t *memo = &adviser->expectations[depth][code];
417 if (*memo >= 0) return *memo;
418 int counts[FACES], rerolled = DICE;
419 decode(code, counts);
420 for (int i = 0; i < FACES; i++) rerolled -= counts[i];
421 int64_t sum = 0;
422 for (int i = 0; i < outcome_count[rerolled]; i++) {
423 Outcome outcome = outcomes[rerolled][i];
424 sum += outcome.weight * advise_value(adviser, code + outcome.code, depth - 1);
425 }
426 *memo = sum * powers[DICE - rerolled];
427 return *memo;
428}

430static bool advise(Game *game) {
431 if (!check_rolls_left(game)) return false;
432 Adviser *adviser = malloc(sizeof *adviser);
433 if (!adviser) exit(EXIT_FAILURE);
434 adviser->player = &game->players[game->turn.seat];
435 adviser->joker = game->joker;
436 for (int depth = 0; depth < 4; depth++) for (int code = 0; code < CODES; code++) {
437 adviser->values[depth][code] = adviser->expectations[depth][code] = -1;
438 }
439 int depth = game->max_rolls - game->turn.rolls, chosen = 0;
440 int64_t best = -1, denominator = 1;
441 for (int i = 0; i < depth; i++) denominator *= 7776;
442 for (int mask = 0; mask < 32; mask++) {
443 int64_t value = expected(adviser, held_code(game->turn.faces, mask), depth);
444 if (value > best) { best = value; chosen = mask; }
445 }
446 printf("advise %s: %s expect %s\n", pattern_text(chosen).text, face_text(game->turn.faces, chosen).text, fraction(best, denominator).text);
447 free(adviser);
448 return true;
449}

451static bool odds(Game *game, int mask) {
452 if (!check_rolls_left(game)) return false;
453 int code = held_code(game->turn.faces, mask), counts[FACES], rerolled = DICE;
454 decode(code, counts);
455 for (int i = 0; i < FACES; i++) rerolled -= counts[i];
456 int64_t sums[CATEGORIES] = {0}, chances[CATEGORIES] = {0}, best = 0;
457 Player *player = &game->players[game->turn.seat];
458 for (int i = 0; i < outcome_count[rerolled]; i++) {
459 Outcome outcome = outcomes[rerolled][i];
460 decode(code + outcome.code, counts);
461 Candidates candidates = evaluate(player, counts, game->joker);
462 for (int c = 0; c < CATEGORIES; c++) if (candidates.points[c] >= 0) {
463 sums[c] += outcome.weight * candidates.points[c];
464 if (candidates.points[c] > 0) chances[c] += outcome.weight;
465 }
466 best += outcome.weight * candidates.points[candidates.best];
467 }
468 printf("odds %s outcomes %d\n", pattern_text(mask).text, powers[rerolled]);
469 for (int c = 0; c < CATEGORIES; c++) if (player->scores[c] < 0) {
470 printf("%s expect %s chance %s\n", categories[c], fraction(sums[c], powers[rerolled]).text, fraction(chances[c], powers[rerolled]).text);
471 }
472 printf("best expect %s\n", fraction(best, powers[rerolled]).text);
473 return true;
474}

555static void simulate(const Table *table, int64_t seed, int count, const Strategy *bots, int players) {
556 int wins[MAX_PLAYERS] = {0}, best[MAX_PLAYERS] = {0}, bonus[MAX_PLAYERS] = {0}, yahtzee[MAX_PLAYERS] = {0};
557 int64_t sums[MAX_PLAYERS] = {0}, state = seed;
558 char *names[] = {"A", "B", "C", "D", "E", "F"};
559 for (int n = 0; n < count; n++) {
560 Game *game = new_game(state, players, names, table->rolls, table->joker, false);
561 for (int i = 0; i < players; i++) game->players[i].strategy = bots[i];
562 while (!game->turn.finished) bot_turn(game);
563 state = game->random_state;
564 int order[MAX_PLAYERS];
565 ranking(game, order);
566 wins[order[0]]++;
567 for (int i = 0; i < players; i++) {
568 Player *player = &game->players[i];
569 int total = player_total(player);
570 sums[i] += total;
571 if (total > best[i]) best[i] = total;
572 bonus[i] += upper_bonus(player) == 35;
573 yahtzee[i] += player->scores[11] == 50;
574 }
575 free(game);
576 }
577 printf("sim seed %" PRId64 " games %d players %d\n", seed, count, players);
578 for (int i = 0; i < players; i++) printf("seat %d %s wins %d average %s best %d bonus %d yahtzee %d\n",
579 i + 1, strategies[bots[i]], wins[i], average(sums[i], count).text, best[i], bonus[i], yahtzee[i]);
580}

582static bool execute(Table *table, char **words, int count) {
583 const char *command = words[0];
584 Game *game = table->game;
585 if (strcmp(command, "RULE") == 0) {
586 if (strcmp(words[1], "ROLLS") != 0 && strcmp(words[1], "JOKER") != 0) return fail("unknown rule %s", words[1]);
587 int rolls = table->rolls;
588 JokerRule joker = table->joker;
589 if (strcmp(words[1], "ROLLS") == 0) {
590 int64_t value;
591 if (!positive(words[2], 4, &value)) return false;
592 rolls = (int)value;
593 } else if (strcmp(words[2], "FORCED") == 0) joker = FORCED;
594 else if (strcmp(words[2], "FREE") == 0) joker = FREE;
595 else if (strcmp(words[2], "OFF") == 0) joker = OFF;
596 else return fail("bad value %s", words[2]);
597 if (game) return fail("game in progress");
598 table->rolls = rolls;
599 table->joker = joker;
600 printf("rule %s %s\n", words[1], words[2]);
601 } else if (strcmp(command, "GAME") == 0) {
602 int64_t seed;
603 if (!positive(words[1], 2147483646, &seed)) return false;
604 for (int i = 2; i < count; i++) if (!valid_name(words[i])) return false;
605 if (count - 2 > MAX_PLAYERS) return fail("too many players");
606 if (game) return fail("game in progress");
607 for (int i = 2; i < count; i++) for (int j = 2; j < i; j++) if (strcmp(words[i], words[j]) == 0) return fail("duplicate name %s", words[i]);
608 table->game = new_game(seed, count - 2, words + 2, table->rolls, table->joker, true);
609 printf("game seed %" PRId64 " players", seed);
610 for (int i = 2; i < count; i++) printf(" %s", words[i]);
611 printf("\nturn 1 %s\n", words[2]);
612 } else if (strcmp(command, "BOT") == 0) {
613 if (!valid_name(words[1])) return false;
614 Strategy strategy;
615 if (!parse_strategy(words[2], &strategy) || !check_game(game)) return false;
616 int player = -1;
617 for (int i = 0; i < game->player_count; i++) if (strcmp(game->players[i].name, words[1]) == 0) { player = i; break; }
618 if (player < 0) return fail("no player %s", words[1]);
619 game->players[player].strategy = strategy;
620 printf("bot %s %s\n", words[1], words[2]);
621 } else if (strcmp(command, "LOAD") == 0 || strcmp(command, "KEEP") == 0) {
622 int values[20];
623 for (int i = 1; i < count; i++) {
624 int64_t value;
625 if (!positive(words[i], 6, &value)) return false;
626 values[i - 1] = (int)value;
627 }
628 if (!check_game(game)) return false;
629 if (strcmp(command, "KEEP") == 0) return keep(game, values, count - 1);
630 for (int i = 0; i < count - 1; i++) game->loaded[game->loaded_tail++] = values[i];
631 printf("loaded %d\n", game->loaded_tail - game->loaded_head);
632 } else if (strcmp(command, "ROLL") == 0) return roll(game);
633 else if (strcmp(command, "HOLD") == 0 || strcmp(command, "ODDS") == 0) {
634 int mask;
635 if (!parse_pattern(words[1], &mask)) return false;
636 return strcmp(command, "HOLD") == 0 ? hold(game, mask) : odds(game, mask);
637 } else if (strcmp(command, "SCORE") == 0 || strcmp(command, "WHATIF") == 0) {
638 int category = category_index(words[1]);
639 if (category < 0) return fail("bad category %s", words[1]);
640 return score(game, category, strcmp(command, "WHATIF") == 0);
641 } else if (strcmp(command, "UNDO") == 0) return undo(game);
642 else if (strcmp(command, "REDO") == 0) return redo(game);
643 else if (strcmp(command, "SUGGEST") == 0) {
644 if (!check_rolled(game)) return false;
645 int counts[FACES];
646 face_counts(game->turn.faces, counts);
647 Candidates candidates = evaluate(&game->players[game->turn.seat], counts, game->joker);
648 printf("suggest");
649 for (int c = 0; c < CATEGORIES; c++) if (candidates.points[c] >= 0) printf(" %s %d", categories[c], candidates.points[c]);
650 printf(" best %s %d\n", categories[candidates.best], candidates.points[candidates.best]);
651 } else if (strcmp(command, "ADVISE") == 0) return advise(game);
652 else if (strcmp(command, "AUTO") == 0) {
653 int64_t turns;
654 if (!positive(words[1], 100, &turns) || !check_turn(game)) return false;
655 game->redo_count = 0;
656 for (int i = 0; i < turns && !game->turn.finished; i++) bot_turn(game);
657 } else if (strcmp(command, "STATUS") == 0) {
658 if (!check_game(game)) return false;
659 if (game->turn.finished) puts("status game over");
660 else {
661 Player *player = &game->players[game->turn.seat];
662 printf("status round %d player %s rolls %d dice %s holds %s\n", player_round(player), player->name, game->turn.rolls,
663 face_text(game->turn.rolls ? game->turn.faces : NULL, 31).text, pattern_text(game->turn.hold).text);
664 }
665 } else if (strcmp(command, "CARD") == 0) {
666 if (!check_game(game)) return false;
667 card_row(game, "CATEGORY", -1);
668 for (int c = 0; c < CATEGORIES; c++) card_row(game, categories[c], c);
669 card_row(game, "UPPER", 13);
670 card_row(game, "BONUS", 14);
671 card_row(game, "YBONUS", 15);
672 card_row(game, "TOTAL", 16);
673 } else if (strcmp(command, "BONUS") == 0) {
674 if (!check_game(game)) return false;
675 for (int i = 0; i < game->player_count; i++) {
676 Player *player = &game->players[i];
677 int upper = upper_subtotal(player), maximum = upper;
678 for (int c = 0; c < 6; c++) if (player->scores[c] < 0) maximum += 5 * (c + 1);
679 printf("bonus %s upper %d need %d max %d %s\n", player->name, upper, upper < 63 ? 63 - upper : 0,
680 maximum, upper >= 63 ? "made" : maximum < 63 ? "lost" : "open");
681 }
682 } else if (strcmp(command, "STATS") == 0) {
683 if (!check_game(game)) return false;
684 for (int i = 0; i < game->player_count; i++) {
685 int turns = 0, rolls = 0, zeros = 0, yahtzees = 0, sum = 0;
686 for (int j = 0; j < game->entry_count; j++) {
687 ScoreEntry *entry = &game->entries[j];
688 if (entry->before.seat != i) continue;
689 turns++;
690 rolls += entry->before.rolls;
691 zeros += entry->points == 0;
692 sum += entry->points;
693 int counts[FACES];
694 face_counts(entry->before.faces, counts);
695 yahtzees += counts[frequent(counts)] == 5;
696 }
697 printf("stats %s turns %d rolls %d zeros %d yahtzees %d average %s\n", game->players[i].name, turns, rolls, zeros, yahtzees, average(sum, turns).text);
698 }
699 } else if (strcmp(command, "LOG") == 0) {
700 if (!check_game(game)) return false;
701 for (int i = 0; i < game->entry_count; i++) {
702 ScoreEntry *entry = &game->entries[i];
703 printf("log %d %s %s %s %d", i + 1, game->players[entry->before.seat].name, face_text(entry->before.faces, 31).text,
704 categories[entry->category], entry->points);
705 if (entry->joker) printf(" joker");
706 if (entry->bonus) printf(" ybonus 100");
707 putchar('\n');
708 }
709 if (!game->entry_count) puts("no scores");
710 } else if (strcmp(command, "FACES") == 0) {
711 if (!check_game(game)) return false;
712 int64_t total = 0, squares = 0;
713 printf("faces");
714 for (int i = 0; i < FACES; i++) {
715 int64_t count = game->face_counts[i];
716 total += count;
717 squares += count * count;
718 printf(" %" PRId64, count);
719 }
720 printf(" total %" PRId64 " chi2 %s\n", total, fraction(6 * squares - total * total, total ? total : 1).text);
721 } else if (strcmp(command, "STANDINGS") == 0 || strcmp(command, "END") == 0) {
722 if (!check_game(game)) return false;
723 int order[MAX_PLAYERS];
724 ranking(game, order);
725 bool end = strcmp(command, "END") == 0;
726 if (end) puts(game->turn.finished ? "end final" : "end abandoned");
727 int leader = player_total(&game->players[order[0]]);
728 for (int i = 0; i < game->player_count; i++) {
729 Player *player = &game->players[order[i]];
730 int total = player_total(player);
731 if (!end) printf("standing %d %s %d behind %d\n", i + 1, player->name, total, leader - total);
732 else {
733 printf("rank %d %s %d\n", i + 1, player->name, total);
734 if (game->turn.finished) {
735 Record *record = record_for(table, player->name);
736 record->games++;
737 record->wins += i == 0;
738 record->sum += total;
739 if (total > record->best) record->best = total;
740 }
741 }
742 }
743 if (end) { free(game); table->game = NULL; }
744 } else if (strcmp(command, "RECORDS") == 0) {
745 Record *order[2400];
746 for (int i = 0; i < table->record_count; i++) order[i] = &table->records[i];
747 qsort(order, table->record_count, sizeof *order, compare_records);
748 for (int i = 0; i < table->record_count; i++) {
749 Record *record = order[i];
750 printf("record %s games %d wins %d best %d average %s\n", record->name, record->games, record->wins,
751 record->best, average(record->sum, record->games).text);
752 }
753 if (!table->record_count) puts("no records");
754 } else if (strcmp(command, "SIM") == 0) {
755 int64_t seed, games;
756 if (!positive(words[1], 2147483646, &seed) || !positive(words[2], 50, &games)) return false;
757 Strategy bots[8];
758 for (int i = 3; i < count; i++) if (!parse_strategy(words[i], &bots[i - 3])) return false;
759 if (count - 3 > MAX_PLAYERS) return fail("too many players");
760 simulate(table, seed, (int)games, bots, count - 3);
761 } else return fail("unknown command %s", command);
762 return true;
763}
