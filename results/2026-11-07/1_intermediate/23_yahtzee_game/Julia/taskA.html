49face_words(tokens::Vector{String}) = Int[integer_word(token, 1, 6) for token in tokens]
50faces_text(dice) = dice === nothing || isempty(dice) ? "-" : join(dice, " ")
51pattern_text(holds::Holds) = join(mark ? "1" : "0" for mark in holds)
52held_faces(dice::Dice, holds::Holds) = [dice[index] for index in 1:5 if holds[index]]
53face_counts(dice)::Counts = ntuple(face -> count(==(face), dice), 6)
54most_frequent(counts::Counts)::Int = findlast(==(maximum(counts)), counts)

56function fraction(numerator::Int64, denominator::Int64)::String
57 divisor = gcd(numerator, denominator)
58 top, bottom = div(numerator, divisor), div(denominator, divisor)
59 bottom == 1 ? string(top) : "$top/$bottom"
60end

81upper_subtotal(player::Player)::Int64 = sum(something(score, Int64(0)) for score in view(player.boxes, 1:6))
82upper_bonus(player::Player)::Int64 = upper_subtotal(player) >= 63 ? 35 : 0
83player_total(player::Player)::Int64 = sum(something(score, Int64(0)) for score in player.boxes) + upper_bonus(player) + player.yahtzee_bonus
84player_round(player::Player)::Int = count(score -> score !== nothing, player.boxes) + 1

86function normal_score(category::Category, counts::Counts)::Int64
87 index = Int(category)
88 index <= 6 && return index * counts[index]
89 total = sum(face * counts[face] for face in 1:6)
90 largest = maximum(counts)
91 if category == THREEKIND
92 return largest >= 3 ? total : 0
93 elseif category == FOURKIND
94 return largest >= 4 ? total : 0
95 elseif category == FULLHOUSE
96 return 3 in counts && 2 in counts ? 25 : 0
97 elseif category == SMALL || category == LARGE
98 run_length = category == SMALL ? 4 : 5
99 has_run = any(start -> all(face -> counts[face] > 0, start:(start + run_length - 1)), 1:(7 - run_length))
100 return has_run ? (category == SMALL ? 30 : 40) : 0
101 elseif category == YAHTZEE
102 return largest == 5 ? 50 : 0
103 end
104 total
105end

107struct CandidateScores
108 joker::Bool
109 allowed::BitVector
110 points::Vector{Int64}
111 best::Category
112 requirement::String
113end
114
115function candidates(player::Player, counts::Counts, joker_rule::JokerRule)::CandidateScores
116 allowed = BitVector(isnothing.(player.boxes))
117 joker = joker_rule != OFF && 5 in counts && player.boxes[Int(YAHTZEE)] !== nothing
118 requirement = ""
119 if joker && joker_rule == FORCED
120 face = most_frequent(counts)
121 if player.boxes[face] === nothing
122 fill!(allowed, false)
123 allowed[face] = true
124 requirement = CATEGORY_NAMES[face]
125 elseif any(view(allowed, 7:13))
126 allowed[1:6] .= false
127 requirement = "lower box"
128 end
129 end
130 points = zeros(Int64, 13)
131 best_category = ONES
132 best_points = Int64(-1)
133 for category in instances(Category)
134 index = Int(category)
135 allowed[index] || continue
136 score = normal_score(category, counts)
137 if joker
138 category == FULLHOUSE && (score = 25)
139 category == SMALL && (score = 30)
140 category == LARGE && (score = 40)
141 end
142 points[index] = score
143 if score > best_points
144 best_points = score
145 best_category = category
146 end
147 end
148 CandidateScores(joker, allowed, points, best_category, requirement)
149end

260function score_text(game::Game, entry::ScoreEntry; hypothetical::Bool=false)::String

285function bot_holds(strategy::BotStrategy, dice::Dice)::Holds
286 strategy == HIGH && return ntuple(index -> dice[index] >= 5, 5)
287 counts = face_counts(dice)
288 if strategy == STRAIGHT
289 best_start, best_length = 0, 0
290 for start in 1:6
291 run_length = 0
292 while start + run_length <= 6 && counts[start + run_length] > 0
293 run_length += 1
294 end
295 if run_length >= best_length
296 best_start, best_length = start, run_length
297 end
298 end
299 if best_length >= 3
300 marks = falses(5)
301 for face in best_start:(best_start + best_length - 1)
302 marks[findfirst(==(face), dice)] = true
303 end
304 return Tuple(marks)
305 end
306 end
307 target = most_frequent(counts)
308 ntuple(index -> dice[index] == target, 5)
309end

311function bot_turn!(game::Game)
312 game.turn.rolls == 0 && roll_dice!(game)
313 strategy = game.players[game.turn.player].strategy
314 while game.turn.rolls < game.rules.rolls
315 holds = bot_holds(strategy, game.turn.dice)
316 all(holds) && break
317 set_holds!(game, holds)
318 roll_dice!(game)
319 end
320 counts = face_counts(game.turn.dice)
321 options = candidates(game.players[game.turn.player], counts, game.rules.joker)
322 face = most_frequent(counts)
323 category = strategy == UPPER && counts[face] >= 3 && options.allowed[face] ? Category(face) : options.best
324 enter_score!(game, category)
325end

327struct Outcome
328 counts::Counts
329 weight::Int64
330end
331
332function distribution(total::Int)::Vector{Outcome}
333 outcomes = Outcome[]
334 counts = zeros(Int, 6)
335 function visit(remaining::Int, face::Int)
336 if face == 6
337 counts[face] = remaining
338 weight = div(factorial(total), prod(factorial(count) for count in counts))
339 push!(outcomes, Outcome(Tuple(counts), weight))
340 return
341 end
342 for frequency in 0:remaining
343 counts[face] = frequency
344 visit(remaining - frequency, face + 1)
345 end
346 end
347 visit(total, 1)
348 outcomes
349end

351combined(held::Counts, rolled::Counts)::Counts = ntuple(face -> held[face] + rolled[face], 6)

353struct RollTables
354 outcomes::Vector{Vector{Outcome}}
355 full::Vector{Counts}
356 held::Vector{Counts}
357 held_index::Dict{Counts,Int}
358 next_states::Vector{Vector{Int}}
359 weights::Vector{Vector{Int64}}
360 subsets::Vector{Vector{Int}}
361end
362
363function RollTables()
364 outcomes = distribution.(0:5)
365 full = [outcome.counts for outcome in outcomes[6]]
366 held = [outcome.counts for group in outcomes for outcome in group]
367 full_index = Dict(state => index for (index, state) in enumerate(full))
368 held_index = Dict(state => index for (index, state) in enumerate(held))
369 next_states, weights = Vector{Int}[], Vector{Int64}[]
370 for state in held
371 kept = sum(state)
372 rolled = outcomes[6 - kept]
373 push!(next_states, [full_index[combined(state, outcome.counts)] for outcome in rolled])
374 push!(weights, Int64[outcome.weight * 6^kept for outcome in rolled])
375 end
376 subsets = [findall(state -> all(state[face] <= dice[face] for face in 1:6), held) for dice in full]
377 RollTables(outcomes, full, held, held_index, next_states, weights, subsets)
378end

380function print_odds(game::Game, tables::RollTables, holds::Holds)
381 current = game.turn::Turn
382 player = game.players[current.player]
383 held = face_counts(held_faces(current.dice, holds))
384 rolled = 5 - sum(held)
385 denominator = Int64(6)^rolled
386 totals, chances = zeros(Int64, 13), zeros(Int64, 13)
387 best_total = Int64(0)
388 for outcome in tables.outcomes[rolled + 1]
389 options = candidates(player, combined(held, outcome.counts), game.rules.joker)
390 totals .+= outcome.weight .* options.points
391 chances .+= outcome.weight .* (options.points .> 0)
392 best_total += outcome.weight * options.points[Int(options.best)]
393 end
394 emit("odds", pattern_text(holds), "outcomes", denominator)
395 for category in instances(Category)
396 index = Int(category)
397 if player.boxes[index] === nothing
398 emit(category, "expect", fraction(totals[index], denominator), "chance", fraction(chances[index], denominator))
399 end
400 end
401 emit("best expect", fraction(best_total, denominator))
402end

404function print_advice(game::Game, tables::RollTables)
405 current = game.turn::Turn
406 player = game.players[current.player]
407 values = Int64[]
408 for counts in tables.full
409 options = candidates(player, counts, game.rules.joker)
410 push!(values, options.points[Int(options.best)])
411 end
412 remaining = game.rules.rolls - current.rolls
413 expected = zeros(Int64, length(tables.held))
414 for depth in 1:remaining
415 for index in eachindex(tables.held)
416 expected[index] = sum(weight * values[state] for (weight, state) in zip(tables.weights[index], tables.next_states[index]))
417 end
418 if depth < remaining
419 values = Int64[maximum(expected[subset]) for subset in tables.subsets]
420 end
421 end
422 choices = Int64[expected[tables.held_index[face_counts(held_faces(current.dice, holds))]] for holds in PATTERNS]
423 chosen = argmax(choices)
424 holds = PATTERNS[chosen]
425 emit("advise", pattern_text(holds) * ":", faces_text(held_faces(current.dice, holds)),
426 "expect", fraction(choices[chosen], Int64(7776)^remaining))
427end

543function game_command!(program::Program, tokens::Vector{String})
544 op = tokens[1]
545 if op == "BOT"
546 name = player_name(tokens[2])
547 strategy = strategy_word(tokens[3])
548 elseif op == "LOAD" || op == "KEEP"
549 faces = face_words(tokens[2:end])
550 elseif op == "HOLD" || op == "ODDS"
551 holds = hold_pattern(tokens[2])
552 elseif op == "SCORE" || op == "WHATIF"
553 category = category_word(tokens[2])
554 elseif op == "AUTO"
555 turn_count = Int(integer_word(tokens[2], 1, 100))
556 end
557 game = require_game(program)
558 if op == "BOT"
559 seat = findfirst(player -> player.name == name, game.players)
560 seat === nothing && fail("no player $name")
561 game.players[seat].strategy = strategy
562 emit("bot", name, strategy)
563 elseif op == "LOAD"
564 append!(game.loaded, faces)
565 emit("loaded", length(game.loaded) - game.loaded_head + 1)
566 elseif op == "ROLL"
567 turn_check(game)
568 game.turn.rolls == game.rules.rolls && fail("no rolls left")
569 all(game.turn.holds) && fail("all dice held")
570 roll_dice!(game)
571 elseif op == "HOLD"
572 rolls_left_check(game)
573 set_holds!(game, holds)
574 elseif op == "KEEP"
575 rolls_left_check(game)
576 marks = falses(5)
577 for face in faces
578 die = findfirst(index -> !marks[index] && game.turn.dice[index] == face, 1:5)
579 die === nothing && fail("not showing $face")
580 marks[die] = true
581 end
582 set_holds!(game, Tuple(marks))
583 elseif op == "SCORE" || op == "WHATIF"
584 score_check(game, category)
585 if op == "WHATIF"
586 emit(score_text(game, score_details(game, category); hypothetical=true))
587 else
588 enter_score!(game, category)
589 end
590 elseif op == "UNDO"
591 isempty(game.entries) && fail("nothing to undo")
592 entry = pop!(game.entries)
593 push!(game.redo, RedoEntry(game.turn, entry))
594 player = game.players[entry.turn.player]
595 player.boxes[Int(entry.category)] = nothing
596 player.yahtzee_bonus -= entry.yahtzee_bonus
597 game.turn = entry.turn
598 emit("undo", player.name, entry.category, entry.points)
599 elseif op == "REDO"
600 isempty(game.redo) && fail("nothing to redo")
601 saved = pop!(game.redo)
602 entry = saved.score
603 player = game.players[entry.turn.player]
604 player.boxes[Int(entry.category)] = entry.points
605 player.yahtzee_bonus += entry.yahtzee_bonus
606 push!(game.entries, entry)
607 game.turn = saved.resume
608 emit("redo", player.name, entry.category, entry.points)
609 elseif op == "SUGGEST"
610 rolled_check(game)
611 player = game.players[game.turn.player]
612 options = candidates(player, face_counts(game.turn.dice), game.rules.joker)
613 parts = String["suggest"]
614 for box in instances(Category)
615 if player.boxes[Int(box)] === nothing
616 append!(parts, [string(box), string(options.points[Int(box)])])
617 end
618 end
619 emit(join(parts, " "), "best", options.best, options.points[Int(options.best)])
620 elseif op == "ODDS"
621 rolls_left_check(game)
622 print_odds(game, program.tables, holds)
623 elseif op == "ADVISE"
624 rolls_left_check(game)
625 print_advice(game, program.tables)
626 elseif op == "AUTO"
627 turn_check(game)
628 empty!(game.redo)
629 for _ in 1:turn_count
630 game.turn === nothing && break
631 bot_turn!(game)
632 end
633 elseif op == "STATUS"
634 if game.turn === nothing
635 emit("status game over")
636 else
637 current = game.turn
638 player = game.players[current.player]
639 emit("status round", player_round(player), "player", player.name, "rolls", current.rolls,
640 "dice", faces_text(current.dice), "holds", pattern_text(current.holds))
641 end
642 elseif op == "CARD"
643 print_card(game)
644 elseif op == "BONUS"
645 for player in game.players
646 upper = upper_subtotal(player)
647 maximum_upper = upper + sum((5 * face for face in 1:6 if player.boxes[face] === nothing); init=Int64(0))
648 status = upper >= 63 ? "made" : maximum_upper < 63 ? "lost" : "open"
649 emit("bonus", player.name, "upper", upper, "need", max(63 - upper, 0), "max", maximum_upper, status)
650 end
651 elseif op == "STATS"
652 print_stats(game)
653 elseif op == "LOG"
654 isempty(game.entries) && emit("no scores")
655 for (index, entry) in enumerate(game.entries)
656 text = "log $index $(game.players[entry.turn.player].name) $(faces_text(entry.turn.dice)) $(entry.category) $(entry.points)"
657 entry.joker && (text *= " joker")
658 entry.yahtzee_bonus > 0 && (text *= " ybonus 100")
659 emit(text)
660 end
661 elseif op == "FACES"
662 total = sum(game.face_totals)
663 chi = total == 0 ? "0" : fraction(6 * sum(value -> value * value, game.face_totals) - total * total, total)
664 emit("faces", faces_text(game.face_totals), "total", total, "chi2", chi)
665 elseif op == "STANDINGS" || op == "END"
666 order = ranking(game)
667 finished = game.turn === nothing
668 op == "END" && emit("end", finished ? "final" : "abandoned")
669 leader = player_total(game.players[order[1]])
670 for (rank, seat) in enumerate(order)
671 player = game.players[seat]
672 if op == "STANDINGS"
673 emit("standing", rank, player.name, player_total(player), "behind", leader - player_total(player))
674 else
675 emit("rank", rank, player.name, player_total(player))
676 end
677 end
678 if op == "END"
679 finished && record_game!(program.records, game)
680 program.game = nothing
681 end
682 end
683 nothing
684end

686function execute!(program::Program, tokens::Vector{String})
687 op = tokens[1]
688 op in COMMANDS || fail("unknown command $op")
689 if op == "RULE"
690 rule, value = tokens[2], tokens[3]
691 rule in ("ROLLS", "JOKER") || fail("unknown rule $rule")
692 if rule == "ROLLS"
693 changed = Rules(Int(integer_word(value, 1, 4)), program.rules.joker)
694 else
695 haskey(JOKER_RULES, value) || fail("bad value $value")
696 changed = Rules(program.rules.rolls, JOKER_RULES[value])
697 end
698 program.game !== nothing && fail("game in progress")
699 program.rules = changed
700 emit("rule", rule, value)
701 elseif op == "GAME"
702 seed = integer_word(tokens[2], 1, 2147483646)
703 names = tokens[3:end]
704 foreach(player_name, names)
705 length(names) > 6 && fail("too many players")
706 program.game !== nothing && fail("game in progress")
707 seen = Set{String}()
708 for name in names
709 name in seen && fail("duplicate name $name")
710 push!(seen, name)
711 end
712 program.game = Game(seed, Player.(names), program.rules)
713 emit("game seed", seed, "players", join(names, " "))
714 emit("turn 1", names[1])
715 elseif op == "RECORDS"
716 print_records(program.records)
717 elseif op == "SIM"
718 seed = integer_word(tokens[2], 1, 2147483646)
719 sample_count = Int(integer_word(tokens[3], 1, 50))
720 strategies = strategy_word.(tokens[4:end])
721 length(strategies) > 6 && fail("too many players")
722 simulate(seed, sample_count, strategies, program.rules)
723 else
724 game_command!(program, tokens)
725 end
726 nothing
727end

729function main()
730 command_count = parse(Int, readline())
731 program = Program()
732 for _ in 1:command_count
733 tokens = String.(split(readline(), ' '))
734 try
735 execute!(program, tokens)
736 catch problem
737 problem isa CommandError || rethrow()
738 emit("error:", problem.reason)
739 end
740 end
741end
