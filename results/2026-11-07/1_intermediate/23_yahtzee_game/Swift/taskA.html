34func integerWord(_ word: String, minimum: Int64, maximum: Int64) throws -> Int64 {
35 let bytes = Array(word.utf8)
36 let first = bytes[0] == 45 ? 1 : 0
37 guard first < bytes.count, word != "-0",
38 bytes.count - first == 1 || bytes[first] != 48 else {
39 throw CommandError.reason("bad value \(word)")
40 }
41 var value: Int64 = 0
42 for byte in bytes[first...] {
43 guard byte >= 48 && byte <= 57 else { throw CommandError.reason("bad value \(word)") }
44 value = value * 10 + Int64(byte - 48)
45 }
46 if first == 1 { value = -value }
47 guard value >= minimum && value <= maximum else { throw CommandError.reason("bad value \(word)") }
48 return value
49}

77func fraction(_ numerator: Int64, _ denominator: Int64) -> String {
78 var a = numerator
79 var b = denominator
80 while b != 0 { (a, b) = (b, a % b) }
81 let top = numerator / a
82 let bottom = denominator / a
83 return bottom == 1 ? "\(top)" : "\(top)/\(bottom)"
84}

100 var counts = Array(repeating: 0, count: 6)

105func mostFrequent(_ counts: Counts) -> Int {
106 var best = 0
107 for face in 1..<6 where counts[face] >= counts[best] { best = face }
108 return best

111func isYahtzee(_ counts: Counts) -> Bool { counts.contains(5) }

113func normalScore(_ category: Category, _ counts: Counts) -> Int {
114 if category.isUpper { return (category.rawValue + 1) * counts[category.rawValue] }
115 let sum = counts.enumerated().reduce(0) { $0 + ($1.offset + 1) * $1.element }
116 let largest = counts.max()!
117 switch category {
118 case .threeKind: return largest >= 3 ? sum : 0
119 case .fourKind: return largest >= 4 ? sum : 0
120 case .fullHouse: return counts.contains(3) && counts.contains(2) ? 25 : 0
121 case .small, .large:
122 let length = category == .small ? 4 : 5
123 for start in 0...(6 - length) {
124 if (start..<(start + length)).allSatisfy({ counts[$0] > 0 }) {
125 return category == .small ? 30 : 40
126 }
127 }
128 return 0
129 case .yahtzee: return largest == 5 ? 50 : 0
130 default: return sum
131 }
132}

134final class Player {
135 let name: String
136 var strategy = Strategy.kind
137 var boxes = Array<Int?>(repeating: nil, count: 13)
138 var yahtzeeBonus = 0
139
140 init(_ name: String) { self.name = name }
141 var upper: Int { boxes.prefix(6).compactMap { $0 }.reduce(0, +) }
142 var upperBonus: Int { upper >= 63 ? 35 : 0 }
143 var total: Int { boxes.compactMap { $0 }.reduce(0, +) + upperBonus + yahtzeeBonus }
144 var round: Int { boxes.filter { $0 != nil }.count + 1 }
145}

147struct Candidates {
148 var allowed = Array(repeating: false, count: 13)
149 var points = Array(repeating: 0, count: 13)
150 var best = Category.ones
151 var joker = false
152 var requirement = ""
153}
154
155func candidates(_ player: Player, _ counts: Counts, joker: Joker) -> Candidates {
156 var result = Candidates()
157 result.allowed = player.boxes.map { $0 == nil }
158 result.joker = joker != .off && isYahtzee(counts) && player.boxes[Category.yahtzee.rawValue] != nil
159 if result.joker && joker == .forced {
160 let face = mostFrequent(counts)
161 if player.boxes[face] == nil {
162 result.allowed = Array(repeating: false, count: 13)
163 result.allowed[face] = true
164 result.requirement = Category(rawValue: face)!.name
165 } else if result.allowed[6...].contains(true) {
166 for index in 0..<6 { result.allowed[index] = false }
167 result.requirement = "lower box"
168 }
169 }
170 var bestScore = -1
171 for category in Category.allCases where result.allowed[category.rawValue] {
172 var points = normalScore(category, counts)
173 if result.joker {
174 if category == .fullHouse { points = 25 }
175 if category == .small { points = 30 }
176 if category == .large { points = 40 }
177 }
178 result.points[category.rawValue] = points
179 if points > bestScore { bestScore = points; result.best = category }
180 }
181 return result
182}

184func botHolds(_ strategy: Strategy, _ dice: [Int]) -> Holds {
185 if strategy == .high { return dice.map { $0 >= 5 } }
186 let counts = faceCounts(dice)
187 if strategy == .straight {
188 var bestStart = 0
189 var bestLength = 0
190 for start in 0..<6 {
191 var length = 0
192 while start + length < 6 && counts[start + length] > 0 { length += 1 }
193 if length >= bestLength { bestStart = start; bestLength = length }
194 }
195 if bestLength >= 3 {
196 var holds = Array(repeating: false, count: 5)
197 for face in (bestStart + 1)...(bestStart + bestLength) {
198 holds[dice.firstIndex(of: face)!] = true
199 }
200 return holds
201 }
202 }
203 let target = mostFrequent(counts) + 1
204 return dice.map { $0 == target }
205}

207struct Outcome {
208 let counts: Counts
209 let weight: Int64
210}
211
212struct Rerolls {
213 let outcomes: [[Outcome]]
214
215 private static func distribution(_ total: Int) -> [Outcome] {
216 var outcomes: [Outcome] = []
217 var counts = Array(repeating: 0, count: 6)
218 func visit(_ remaining: Int, face: Int) {
219 if face == 5 {
220 counts[face] = remaining
221 let divisor = counts.reduce(1) { $0 * factorial[$1] }
222 outcomes.append(Outcome(counts: counts, weight: Int64(factorial[total] / divisor)))
223 return
224 }
225 for count in 0...remaining {
226 counts[face] = count
227 visit(remaining - count, face: face + 1)
228 }
229 }
230 visit(total, face: 0)
231 return outcomes
232 }
233
234 init() { outcomes = (0...5).map(Self.distribution) }
235}

237func combined(_ held: Counts, _ rolled: Counts) -> Counts {
238 zip(held, rolled).map { $0.0 + $0.1 }
239}

241struct Advice {
242 let holds: Holds
243 let numerator: Int64
244 let denominator: Int64
245}
246
247func advise(_ player: Player, dice: [Int], rollsLeft: Int, joker: Joker, rerolls: Rerolls) -> Advice {
248 var values: [Counts: Int64] = [:]
249 for outcome in rerolls.outcomes[5] {
250 let scores = candidates(player, outcome.counts, joker: joker)
251 values[outcome.counts] = Int64(scores.points[scores.best.rawValue])
252 }
253 var denominator: Int64 = 1
254 var expectations: [Counts: Int64] = [:]
255 for depth in 1...rollsLeft {
256 expectations.removeAll(keepingCapacity: true)
257 for kept in 0...5 {
258 let rolled = 5 - kept
259 for held in rerolls.outcomes[kept] {
260 var sum: Int64 = 0
261 for outcome in rerolls.outcomes[rolled] {
262 sum += outcome.weight * values[combined(held.counts, outcome.counts)]!
263 }
264 expectations[held.counts] = sum * (powersOfSix[5] / powersOfSix[rolled])
265 }
266 }
267 denominator *= powersOfSix[5]
268 if depth == rollsLeft { break }
269 var next: [Counts: Int64] = [:]
270 for outcome in rerolls.outcomes[5] {
271 var best: Int64 = 0
272 for (held, expected) in expectations {
273 if zip(held, outcome.counts).allSatisfy({ $0.0 <= $0.1 }) { best = max(best, expected) }
274 }
275 next[outcome.counts] = best
276 }
277 values = next
278 }
279 var best = Advice(holds: Array(repeating: false, count: 5), numerator: -1, denominator: denominator)
280 for mask in 0..<32 {
281 let holds = (0..<5).map { mask & (1 << (4 - $0)) != 0 }
282 let expected = expectations[faceCounts(dice, selected: holds)]!
283 if expected > best.numerator { best = Advice(holds: holds, numerator: expected, denominator: denominator) }
284 }
285 return best
286}

311 private let output: ((String) -> Void)?
312 var state: Int64
313 var loaded: [Int] = []
314 var loadedHead = 0

322 self.players = names.map(Player.init)
323 self.rules = rules
324 self.output = output

327 var loadedCount: Int { loaded.count - loadedHead }

330 private func rollDie() -> Int {
331 let face: Int
332 if loadedHead < loaded.count {
333 face = loaded[loadedHead]
334 loadedHead += 1
335 } else {
336 state = 48271 * state % 2147483647
337 face = Int(state % 6) + 1
338 }
339 faceTotals[face - 1] += 1
340 return face
341 }

343 func turnCheck() throws {

345 }
346 func rolledCheck() throws {
347 try turnCheck()

353 }
354 func ranking() -> [Int] {
355 players.indices.sorted {
356 players[$0].total != players[$1].total ? players[$0].total > players[$1].total : $0 < $1
357 }

359 var nextTurnText: String {
360 guard let turn else { return "game over" }

365 func roll() {
366 redo.removeAll()
367 var current = turn!
368 for die in 0..<5 where current.rolls == 0 || !current.holds[die] { current.dice[die] = rollDie() }
369 current.rolls += 1
370 turn = current

372 }

377 current.holds = holds
378 turn = current
379 emit("\(players[current.player].name) holds \(patternText(holds)): \(facesText(current.dice, selected: holds))")

382 func scoreDetails(_ category: Category) -> ScoreEntry {
383 let current = turn!
384 let player = players[current.player]
385 let counts = faceCounts(current.dice)
386 let scores = candidates(player, counts, joker: rules.joker)
387 let bonus = isYahtzee(counts) && player.boxes[Category.yahtzee.rawValue] == 50 ? 100 : 0
388 return ScoreEntry(turn: current, category: category, points: scores.points[category.rawValue],
389 yahtzeeBonus: bonus, joker: scores.joker)
390 }

392 func checkedScore(_ category: Category) throws -> ScoreEntry {
393 try rolledCheck()
394 let current = turn!
395 let player = players[current.player]
396 guard player.boxes[category.rawValue] == nil else { throw CommandError.reason("\(category.name) already used") }
397 let scores = candidates(player, faceCounts(current.dice), joker: rules.joker)
398 guard scores.allowed[category.rawValue] else { throw CommandError.reason("joker requires \(scores.requirement)") }
399 return scoreDetails(category)

402 func scoreText(_ entry: ScoreEntry, hypothetical: Bool) -> String {
403 let player = players[entry.turn.player]
404 let upperAfter = player.upper + (entry.category.isUpper ? entry.points : 0)
405 let addedUpper = upperAfter >= 63 && player.upperBonus == 0 ? 35 : 0
406 var line = "\(player.name) \(hypothetical ? "would score" : "scores") \(entry.points) in \(entry.category.name)"
407 if entry.joker { line += " joker" }
408 if entry.yahtzeeBonus != 0 { line += " ybonus 100" }
409 if addedUpper != 0 { line += " ubonus 35" }
410 return line + " total \(player.total + entry.points + entry.yahtzeeBonus + addedUpper)"
411 }
412
413 func score(_ category: Category) {
414 let entry = scoreDetails(category)
415 let line = scoreText(entry, hypothetical: false)
416 let player = players[entry.turn.player]
417 player.boxes[category.rawValue] = entry.points
418 player.yahtzeeBonus += entry.yahtzeeBonus
419 entries.append(entry)
420 redo.removeAll()
421 turn = entries.count == 13 * players.count ? nil : Turn(player: (entry.turn.player + 1) % players.count)
422 emit(line)
423 emit(nextTurnText)
424 }

426 func botTurn() {
427 if turn!.rolls == 0 { roll() }
428 let strategy = players[turn!.player].strategy
429 while turn!.rolls < rules.rolls {
430 let holds = botHolds(strategy, turn!.dice)
431 if holds.allSatisfy({ $0 }) { break }
432 hold(holds)
433 roll()
434 }
435 let counts = faceCounts(turn!.dice)
436 let scores = candidates(players[turn!.player], counts, joker: rules.joker)
437 let face = mostFrequent(counts)
438 let category = strategy == .upper && counts[face] >= 3 && scores.allowed[face] ? Category(rawValue: face)! : scores.best
439 score(category)
440 }

442 func undoScore() throws {
443 guard let entry = entries.popLast() else { throw CommandError.reason("nothing to undo") }
444 players[entry.turn.player].boxes[entry.category.rawValue] = nil
445 players[entry.turn.player].yahtzeeBonus -= entry.yahtzeeBonus
446 redo.append(RedoEntry(resume: turn, score: entry))
447 turn = entry.turn
448 emit("undo \(players[entry.turn.player].name) \(entry.category.name) \(entry.points)")
449 }
450
451 func redoScore() throws {
452 guard let saved = redo.popLast() else { throw CommandError.reason("nothing to redo") }
453 let entry = saved.score
454 players[entry.turn.player].boxes[entry.category.rawValue] = entry.points
455 players[entry.turn.player].yahtzeeBonus += entry.yahtzeeBonus
456 entries.append(entry)
457 turn = saved.resume
458 emit("redo \(players[entry.turn.player].name) \(entry.category.name) \(entry.points)")
459 }
460}

493 private func odds(_ table: Game, holds: Holds) {
494 let turn = table.turn!
495 let player = table.players[turn.player]
496 let held = faceCounts(turn.dice, selected: holds)
497 let rolled = 5 - held.reduce(0, +)
498 let outcomes = powersOfSix[rolled]
499 var sums = Array<Int64>(repeating: 0, count: 13)
500 var positive = Array<Int64>(repeating: 0, count: 13)
501 var bestSum: Int64 = 0
502 for outcome in rerolls.outcomes[rolled] {
503 let scores = candidates(player, combined(held, outcome.counts), joker: rules.joker)
504 for index in 0..<13 {
505 sums[index] += outcome.weight * Int64(scores.points[index])
506 if scores.points[index] > 0 { positive[index] += outcome.weight }
507 }
508 bestSum += outcome.weight * Int64(scores.points[scores.best.rawValue])
509 }
510 emit("odds", patternText(holds), "outcomes", outcomes)
511 for category in Category.allCases where player.boxes[category.rawValue] == nil {
512 let index = category.rawValue
513 emit(category.name, "expect", fraction(sums[index], outcomes), "chance", fraction(positive[index], outcomes))
514 }
515 emit("best expect", fraction(bestSum, outcomes))
516 }

527 else { cell = String([player.upper, player.upperBonus, player.yahtzeeBonus, player.total][row - 13]) }
528 let width = max(4, player.name.count)
529 line += " " + String(repeating: " ", count: max(0, width - cell.count)) + cell
530 }
531 emit(line)

548 private func simulate(seed: Int64, count: Int, strategies: [Strategy]) {
549 var totals = Array(repeating: SimulationTotal(), count: strategies.count)
550 let names = Array(["A", "B", "C", "D", "E", "F"].prefix(strategies.count))
551 var state = seed
552 for _ in 0..<count {
553 let simulation = Game(seed: state, names: names, rules: rules)
554 for seat in strategies.indices { simulation.players[seat].strategy = strategies[seat] }
555 while simulation.turn != nil { simulation.botTurn() }
556 state = simulation.state
557 totals[simulation.ranking()[0]].wins += 1
558 for seat in strategies.indices {
559 let player = simulation.players[seat]
560 totals[seat].sum += Int64(player.total)
561 totals[seat].best = max(totals[seat].best, player.total)
562 if player.upperBonus == 35 { totals[seat].bonus += 1 }
563 if player.boxes[Category.yahtzee.rawValue] == 50 { totals[seat].yahtzee += 1 }
564 }
565 }
566 emit("sim seed", seed, "games", count, "players", strategies.count)
567 for seat in strategies.indices {
568 let total = totals[seat]
569 emit("seat", seat + 1, strategies[seat].rawValue, "wins", total.wins,
570 "average", average(total.sum, Int64(count)), "best", total.best, "bonus", total.bonus, "yahtzee", total.yahtzee)
571 }
572 }

574 private func execute(_ words: [String]) throws {
575 switch words[0] {
576 case "RULE":
577 guard words[1] == "ROLLS" || words[1] == "JOKER" else { throw CommandError.reason("unknown rule \(words[1])") }
578 var changed = rules
579 if words[1] == "ROLLS" {
580 changed.rolls = Int(try integerWord(words[2], minimum: 1, maximum: 4))
581 } else {
582 guard let joker = Joker(rawValue: words[2]) else { throw CommandError.reason("bad value \(words[2])") }
583 changed.joker = joker
584 }
585 guard game == nil else { throw CommandError.reason("game in progress") }
586 rules = changed
587 emit("rule", words[1], words[2])
588 case "GAME":
589 let seed = try integerWord(words[1], minimum: 1, maximum: 2147483646)
590 let names = Array(words.dropFirst(2))
591 for name in names { try playerName(name) }
592 guard names.count <= 6 else { throw CommandError.reason("too many players") }
593 guard game == nil else { throw CommandError.reason("game in progress") }
594 var seen: Set<String> = []
595 for name in names {
596 guard seen.insert(name).inserted else { throw CommandError.reason("duplicate name \(name)") }
597 }
598 game = Game(seed: seed, names: names, rules: rules, output: { [unowned self] line in self.output.append(line) })
599 emit("game seed", seed, "players", names.joined(separator: " "))
600 emit(game!.nextTurnText)
601 case "BOT":
602 try playerName(words[1])
603 let strategy = try strategyWord(words[2])
604 let table = try requireGame()
605 guard let player = table.players.first(where: { $0.name == words[1] }) else {
606 throw CommandError.reason("no player \(words[1])")
607 }
608 player.strategy = strategy
609 emit("bot", words[1], words[2])
610 case "LOAD", "KEEP":
611 let faces = try words.dropFirst().map { Int(try integerWord($0, minimum: 1, maximum: 6)) }
612 let table = try requireGame()
613 if words[0] == "LOAD" {
614 table.loaded.append(contentsOf: faces)
615 emit("loaded", table.loadedCount)
616 } else {
617 try table.rollsLeftCheck()
618 var holds = Array(repeating: false, count: 5)
619 for face in faces {
620 guard let die = (0..<5).first(where: { !holds[$0] && table.turn!.dice[$0] == face }) else {
621 throw CommandError.reason("not showing \(face)")
622 }
623 holds[die] = true
624 }
625 table.hold(holds)
626 }
627 case "ROLL":
628 let table = try requireGame()
629 try table.turnCheck()
630 guard table.turn!.rolls != rules.rolls else { throw CommandError.reason("no rolls left") }
631 guard !table.turn!.holds.allSatisfy({ $0 }) else { throw CommandError.reason("all dice held") }
632 table.roll()
633 case "HOLD", "ODDS":
634 let holds = try holdPattern(words[1])
635 let table = try requireGame()
636 try table.rollsLeftCheck()
637 if words[0] == "HOLD" { table.hold(holds) }
638 else { odds(table, holds: holds) }
639 case "SCORE", "WHATIF":
640 let category = try categoryWord(words[1])
641 let table = try requireGame()
642 let entry = try table.checkedScore(category)
643 if words[0] == "SCORE" { table.score(category) }
644 else { emit(table.scoreText(entry, hypothetical: true)) }
645 case "UNDO":
646 try requireGame().undoScore()
647 case "REDO":
648 try requireGame().redoScore()
649 case "SUGGEST":
650 let table = try requireGame()
651 try table.rolledCheck()
652 let player = table.players[table.turn!.player]
653 let scores = candidates(player, faceCounts(table.turn!.dice), joker: rules.joker)
654 var line = "suggest"
655 for category in Category.allCases where player.boxes[category.rawValue] == nil {
656 line += " \(category.name) \(scores.points[category.rawValue])"
657 }
658 emit(line, "best", scores.best.name, scores.points[scores.best.rawValue])
659 case "ADVISE":
660 let table = try requireGame()
661 try table.rollsLeftCheck()
662 let turn = table.turn!
663 let best = advise(table.players[turn.player], dice: turn.dice, rollsLeft: rules.rolls - turn.rolls,
664 joker: rules.joker, rerolls: rerolls)
665 emit("advise", patternText(best.holds) + ":", facesText(turn.dice, selected: best.holds),
666 "expect", fraction(best.numerator, best.denominator))
667 case "AUTO":
668 let count = Int(try integerWord(words[1], minimum: 1, maximum: 100))
669 let table = try requireGame()
670 try table.turnCheck()
671 table.redo.removeAll()
672 for _ in 0..<count {
673 if table.turn == nil { break }
674 table.botTurn()
675 }
676 case "STATUS":
677 let table = try requireGame()
678 if let turn = table.turn {
679 let player = table.players[turn.player]
680 emit("status round", player.round, "player", player.name, "rolls", turn.rolls,
681 "dice", turn.rolls == 0 ? "-" : facesText(turn.dice), "holds", patternText(turn.holds))
682 } else { emit("status game over") }
683 case "CARD":
684 card(try requireGame())
685 case "BONUS":
686 let table = try requireGame()
687 for player in table.players {
688 let upper = player.upper
689 let maximum = upper + (0..<6).filter { player.boxes[$0] == nil }.reduce(0) { $0 + 5 * ($1 + 1) }
690 let status = upper >= 63 ? "made" : maximum < 63 ? "lost" : "open"
691 emit("bonus", player.name, "upper", upper, "need", max(63 - upper, 0), "max", maximum, status)
692 }
693 case "STATS":
694 statistics(try requireGame())
695 case "LOG":
696 let table = try requireGame()
697 if table.entries.isEmpty { emit("no scores") }
698 for (index, entry) in table.entries.enumerated() {
699 var suffix = ""
700 if entry.joker { suffix += " joker" }
701 if entry.yahtzeeBonus != 0 { suffix += " ybonus 100" }
702 emit("log", index + 1, table.players[entry.turn.player].name, facesText(entry.turn.dice),
703 entry.category.name, "\(entry.points)" + suffix)
704 }
705 case "FACES":
706 let table = try requireGame()
707 let total = table.faceTotals.reduce(0, +)
708 let squares = table.faceTotals.reduce(0) { $0 + $1 * $1 }
709 emit("faces", table.faceTotals.map(String.init).joined(separator: " "), "total", total,
710 "chi2", total == 0 ? "0" : fraction(6 * squares - total * total, total))
711 case "STANDINGS", "END":
712 let table = try requireGame()
713 let ranking = table.ranking()
714 let finished = table.turn == nil
715 if words[0] == "END" { emit("end", finished ? "final" : "abandoned") }
716 let best = table.players[ranking[0]].total
717 for (rank, seat) in ranking.enumerated() {
718 let player = table.players[seat]
719 if words[0] == "STANDINGS" {
720 emit("standing", rank + 1, player.name, player.total, "behind", best - player.total)
721 } else {
722 emit("rank", rank + 1, player.name, player.total)
723 if finished {
724 var record = records[player.name, default: Record()]
725 record.games += 1
726 if rank == 0 { record.wins += 1 }
727 record.sum += Int64(player.total)
728 record.best = max(record.best, player.total)
729 records[player.name] = record
730 }
731 }
732 }
733 if words[0] == "END" { game = nil }
734 case "RECORDS":
735 let names = records.keys.sorted {
736 let first = records[$0]!
737 let second = records[$1]!
738 if first.wins != second.wins { return first.wins > second.wins }
739 if first.best != second.best { return first.best > second.best }
740 return $0 < $1
741 }
742 if names.isEmpty { emit("no records") }
743 for name in names {
744 let record = records[name]!
745 emit("record", name, "games", record.games, "wins", record.wins, "best", record.best, "average", average(record.sum, record.games))
746 }
747 case "SIM":
748 let seed = try integerWord(words[1], minimum: 1, maximum: 2147483646)
749 let count = Int(try integerWord(words[2], minimum: 1, maximum: 50))
750 let strategies = try words.dropFirst(3).map(strategyWord)
751 guard strategies.count <= 6 else { throw CommandError.reason("too many players") }
752 simulate(seed: seed, count: count, strategies: strategies)
753 default:
754 throw CommandError.reason("unknown command \(words[0])")
755 }
756 }

759 do { try execute(line.split(separator: " ").map(String.init)) }
760 catch CommandError.reason(let reason) { emit("error:", reason) }
761 catch { fatalError("Unexpected command error: \(error)") }
762 }
