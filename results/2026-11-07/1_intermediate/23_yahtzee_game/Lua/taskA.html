67local function ascii_less(a, b)
68 for i = 1, math.min(#a, #b) do
69 local first, second = a:byte(i), b:byte(i)
70 if first ~= second then
71 return first < second
72 end
73 end

77local function fraction(numerator, denominator)
78 local a, b = numerator, denominator
79 while b ~= 0 do
80 a, b = b, a % b
81 end
82 numerator, denominator = numerator // a, denominator // a
83 return denominator == 1 and tostring(numerator) or
84 string.format("%d/%d", numerator, denominator)
85end

161local function normal_score(category, counts)
162 if category <= 6 then
163 return category * counts[category]
164 end
165 local total, largest, bits, pair, triple = 0, 0, 0, false, false
166 for face, count in ipairs(counts) do
167 total = total + face * count
168 largest = math.max(largest, count)
169 if count > 0 then
170 bits = bits | (1 << (face - 1))
171 end
172 pair, triple = pair or count == 2, triple or count == 3
173 end
174 if category == 7 then
175 return largest >= 3 and total or 0
176 elseif category == 8 then
177 return largest >= 4 and total or 0
178 elseif category == 9 then
179 return pair and triple and 25 or 0
180 elseif category == 10 then
181 return ((bits & 15) == 15 or (bits & 30) == 30 or (bits & 60) == 60) and 30 or 0
182 elseif category == 11 then
183 return (bits == 31 or bits == 62) and 40 or 0
184 elseif category == 12 then
185 return largest == 5 and 50 or 0
186 else
187 return total
188 end
189end

223local function evaluate(player, counts, rule)
224 local allowed = 0
225 for c = 1, 13 do
226 if player.scores[c] == nil then
227 allowed = allowed | (1 << (c - 1))
228 end
229 end
230 local joker = rule ~= "OFF" and yahtzee(counts) and player.scores[12] ~= nil
231 if joker and rule == "FORCED" then
232 local upper = frequent(counts)
233 if (allowed & (1 << (upper - 1))) ~= 0 then
234 allowed = 1 << (upper - 1)
235 elseif (allowed & lower_mask) ~= 0 then
236 allowed = allowed & lower_mask
237 end
238 end
239 local points, best = {}, nil
240 for c = 1, 13 do
241 points[c] = -1
242 if player.scores[c] == nil then
243 if (allowed & (1 << (c - 1))) == 0 then
244 points[c] = 0
245 elseif joker and c >= 9 and c <= 11 then
246 points[c] = ({25, 30, 40})[c - 8]
247 else
248 points[c] = normal_score(c, counts)
249 end
250 if not best or points[c] > points[best] then
251 best = c
252 end
253 end
254 end
255 return {points = points, allowed = allowed, joker = joker, best = best}
256end

258local function make_outcomes()
259 local result, factorial = {}, {1, 1, 2, 6, 24, 120}
260 for size = 0, 5 do
261 result[size] = {}
262 local function add(face, remaining, code, divisor)
263 if face == 7 then
264 if remaining == 0 then
265 local list = result[size]
266 list[#list + 1] = {code = code, weight = factorial[size + 1] // divisor}
267 end
268 return
269 end
270 for count = 0, remaining do
271 add(
272 face + 1,
273 remaining - count,
274 code + count * powers[face],
275 divisor * factorial[count + 1]
276 )
277 end
278 end
279 add(1, size, 0, 1)
280 end
281 return result
282end

337function Game:roll()
338 self:check_turn()
339 local turn = self.turn
340 if turn.rolls == self.max_rolls then
341 fail("no rolls left")
342 end
343 if turn.hold == 31 then
344 fail("all dice held")
345 end
346 self.redo_list = {}
347 local faces = {}
348 for i = 1, 5 do
349 if turn.rolls > 0 and held(turn.hold, i) then
350 faces[i] = turn.faces[i]
351 else
352 if self.loaded_head <= #self.loaded then
353 faces[i] = self.loaded[self.loaded_head]
354 self.loaded_head = self.loaded_head + 1
355 else
356 self.random_state = 48271 * self.random_state % 2147483647
357 faces[i] = self.random_state % 6 + 1
358 end
359 self.face_totals[faces[i]] = self.face_totals[faces[i]] + 1
360 end
361 end
362 self.turn = {seat = turn.seat, rolls = turn.rolls + 1, faces = faces, hold = turn.hold}
363 self:emit("%s roll %d: %s", self:current().name, self.turn.rolls, faces_text(faces))
364end
365function Game:hold(mask)
366 self:check_rolls_left()
367 self.redo_list = {}
368 local turn = self.turn
369 self.turn = {seat = turn.seat, rolls = turn.rolls, faces = turn.faces, hold = mask}
370 self:emit(
371 "%s holds %s: %s",
372 self:current().name,
373 pattern_text(mask),
374 faces_text(turn.faces, mask)
375 )
376end
377function Game:keep(values)
378 self:check_rolls_left()
379 local mask = 0
380 for _, value in ipairs(values) do
381 local die
382 for i = 1, 5 do
383 if not held(mask, i) and self.turn.faces[i] == value then
384 die = i
385 break
386 end
387 end
388 if not die then
389 fail("not showing %d", value)
390 end
391 mask = mask | (1 << (5 - die))
392 end
393 self:hold(mask)
394end
395function Game:score(category, preview)
396 self:check_rolled()
397 local player = self:current()
398 if player.scores[category] ~= nil then
399 fail("%s already used", categories[category])
400 end
401 local counts = face_counts(self.turn.faces)
402 local candidates = evaluate(player, counts, self.joker)
403 if (candidates.allowed & (1 << (category - 1))) == 0 then
404 local upper = frequent(counts)
405 fail("joker requires %s", player.scores[upper] == nil and categories[upper] or "lower box")
406 end
407 local points = candidates.points[category]
408 local bonus = yahtzee(counts) and player.scores[12] == 50 and 100 or 0
409 local upper_bonus =
410 category <= 6 and player:upper() < 63 and player:upper() + points >= 63 and 35 or 0
411 local line =
412 string.format(
413 "%s %s %d in %s",
414 player.name,
415 preview and "would score" or "scores",
416 points,
417 categories[category]
418 )
419 if candidates.joker then
420 line = line .. " joker"
421 end
422 if bonus ~= 0 then
423 line = line .. " ybonus 100"
424 end
425 if upper_bonus ~= 0 then
426 line = line .. " ubonus 35"
427 end
428 line = line .. string.format(" total %d", player:total() + points + bonus + upper_bonus)
429 if preview then
430 self:emit("%s", line)
431 return
432 end
433 self.redo_list = {}
434 local entry = {
435 before = self.turn,
436 category = category,
437 points = points,
438 bonus = bonus,
439 joker = candidates.joker
440 }
441 self.entries[#self.entries + 1] = entry
442 player.scores[category] = points
443 player.yahtzee_bonus = player.yahtzee_bonus + bonus
444 if #self.entries == 13 * #self.players then
445 self.turn = nil
446 else
447 self.turn = new_turn(entry.before.seat % #self.players + 1)
448 end
449 self:emit("%s", line)
450 if self.turn then
451 self:emit("turn %d %s", self:current():round(), self:current().name)
452 else
453 self:emit("game over")
454 end
455end
456function Game:undo()
457 if #self.entries == 0 then
458 fail("nothing to undo")
459 end
460 local entry = table.remove(self.entries)
461 self.redo_list[#self.redo_list + 1] = {after = self.turn, entry = entry}
462 local player = self.players[entry.before.seat]
463 player.scores[entry.category] = nil
464 player.yahtzee_bonus = player.yahtzee_bonus - entry.bonus
465 self.turn = entry.before
466 self:emit("undo %s %s %d", player.name, categories[entry.category], entry.points)
467end
468function Game:redo()
469 if #self.redo_list == 0 then
470 fail("nothing to redo")
471 end
472 local saved = table.remove(self.redo_list)
473 local entry = saved.entry
474 local player = self.players[entry.before.seat]
475 player.scores[entry.category] = entry.points
476 player.yahtzee_bonus = player.yahtzee_bonus + entry.bonus
477 self.entries[#self.entries + 1] = entry
478 self.turn = saved.after
479 self:emit("redo %s %s %d", player.name, categories[entry.category], entry.points)
480end
481function Game:bot_hold(strategy)
482 local faces, counts = self.turn.faces, face_counts(self.turn.faces)
483 local target, mask = frequent(counts), 0
484 if strategy == "STRAIGHT" then
485 local best_start, best_length = 1, 0
486 for start = 1, 6 do
487 local last = start
488 while last <= 6 and counts[last] > 0 do
489 last = last + 1
490 end
491 if last - start >= best_length then
492 best_start, best_length = start, last - start
493 end
494 end
495 if best_length >= 3 then
496 for face = best_start, best_start + best_length - 1 do
497 for i = 1, 5 do
498 if faces[i] == face then
499 mask = mask | (1 << (5 - i))
500 break
501 end
502 end
503 end
504 return mask
505 end
506 end
507 for i = 1, 5 do
508 local keep =
509 strategy == "HIGH" and faces[i] >= 5 or strategy ~= "HIGH" and faces[i] == target
510 if keep then
511 mask = mask | (1 << (5 - i))
512 end
513 end
514 return mask
515end
516function Game:bot_turn()
517 local strategy = self:current().strategy
518 if self.turn.rolls == 0 then
519 self:roll()
520 end
521 while self.turn.rolls < self.max_rolls do
522 local mask = self:bot_hold(strategy)
523 if mask == 31 then
524 break
525 end
526 self:hold(mask)
527 self:roll()
528 end
529 local counts = face_counts(self.turn.faces)
530 local candidates = evaluate(self:current(), counts, self.joker)
531 local target = frequent(counts)
532 local category =
533 strategy == "UPPER" and counts[target] >= 3 and
534 (candidates.allowed & (1 << (target - 1))) ~= 0 and
535 target or
536 candidates.best
537 self:score(category, false)
538end
539function Game:ranking()
540 local seats = {}
541 for i = 1, #self.players do
542 seats[i] = i
543 end
544 table.sort(
545 seats,
546 function(a, b)
547 local first, second = self.players[a]:total(), self.players[b]:total()
548 if first ~= second then
549 return first > second
550 end
551 return a < b
552 end
553 )
554 return seats
555end

557local Adviser = {}
558Adviser.__index = Adviser
559function Adviser.new(player, joker)
560 return setmetatable({player = player, joker = joker, values = {}, expectations = {}}, Adviser)
561end
562function Adviser:value(code, depth)
563 local key = depth * 46656 + code
564 if self.values[key] ~= nil then
565 return self.values[key]
566 end
567 local counts, best = decode(code), -1
568 if depth == 0 then
569 local candidates = evaluate(self.player, counts, self.joker)
570 best = candidates.points[candidates.best]
571 else
572 best = self:choose(counts, depth, 1, 0)
573 end
574 self.values[key] = best
575 return best
576end
577function Adviser:choose(counts, depth, face, kept)
578 if face == 7 then
579 return self:expected(kept, depth)
580 end
581 local best = -1
582 for count = 0, counts[face] do
583 best = math.max(best, self:choose(counts, depth, face + 1, kept + count * powers[face]))
584 end
585 return best
586end
587function Adviser:expected(kept, depth)
588 local key = depth * 46656 + kept
589 if self.expectations[key] ~= nil then
590 return self.expectations[key]
591 end
592 local rerolled, sum = 5, 0
593 for _, count in ipairs(decode(kept)) do
594 rerolled = rerolled - count
595 end
596 for _, outcome in ipairs(outcomes[rerolled]) do
597 sum = sum + outcome.weight * self:value(kept + outcome.code, depth - 1)
598 end
599 local result = sum * powers[6 - rerolled]
600 self.expectations[key] = result
601 return result
602end

618function Program:odds(game, mask)
619 game:check_rolls_left()
620 local kept = held_code(game.turn.faces, mask)
621 local rerolled = 5
622 for _, count in ipairs(decode(kept)) do
623 rerolled = rerolled - count
624 end
625 local count, sums, chances, best = powers[rerolled + 1], {}, {}, 0
626 for c = 1, 13 do
627 sums[c], chances[c] = 0, 0
628 end
629 for _, outcome in ipairs(outcomes[rerolled]) do
630 local candidates = evaluate(game:current(), decode(kept + outcome.code), game.joker)
631 for c = 1, 13 do
632 if candidates.points[c] >= 0 then
633 sums[c] = sums[c] + outcome.weight * candidates.points[c]
634 if candidates.points[c] > 0 then
635 chances[c] = chances[c] + outcome.weight
636 end
637 end
638 end
639 best = best + outcome.weight * candidates.points[candidates.best]
640 end
641 self:emit("odds %s outcomes %d", pattern_text(mask), count)
642 for c = 1, 13 do
643 if game:current().scores[c] == nil then
644 self:emit(
645 "%s expect %s chance %s",
646 categories[c],
647 fraction(sums[c], count),
648 fraction(chances[c], count)
649 )
650 end
651 end
652 self:emit("best expect %s", fraction(best, count))
653end
654function Program:advise(game)
655 game:check_rolls_left()
656 local depth = game.max_rolls - game.turn.rolls
657 local chosen, best, denominator = 0, -1, 1
658 for _ = 1, depth do
659 denominator = denominator * 7776
660 end
661 local adviser = Adviser.new(game:current(), game.joker)
662 for mask = 0, 31 do
663 local value = adviser:expected(held_code(game.turn.faces, mask), depth)
664 if value > best then
665 best, chosen = value, mask
666 end
667 end
668 self:emit(
669 "advise %s: %s expect %s",
670 pattern_text(chosen),
671 faces_text(game.turn.faces, chosen),
672 fraction(best, denominator)
673 )
674end

722function Program:simulate(seed, count, bots)
723 local names, totals = {}, {}
724 for i = 1, #bots do
725 names[i], totals[i] =
726 string.char(64 + i),
727 {wins = 0, sum = 0, best = 0, bonus = 0, yahtzee = 0}
728 end
729 local state = seed
730 for _ = 1, count do
731 local game = Game.new(state, names, self.rolls, self.joker)
732 for i, strategy in ipairs(bots) do
733 game.players[i].strategy = strategy
734 end
735 while game.turn do
736 game:bot_turn()
737 end
738 state = game.random_state
739 local winner = game:ranking()[1]
740 totals[winner].wins = totals[winner].wins + 1
741 for i, player in ipairs(game.players) do
742 local total = totals[i]
743 total.sum, total.best = total.sum + player:total(), math.max(total.best, player:total())
744 if player:upper_bonus() == 35 then
745 total.bonus = total.bonus + 1
746 end
747 if player.scores[12] == 50 then
748 total.yahtzee = total.yahtzee + 1
749 end
750 end
751 end
752 self:emit("sim seed %d games %d players %d", seed, count, #bots)
753 for i, strategy in ipairs(bots) do
754 local total = totals[i]
755 self:emit(
756 "seat %d %s wins %d average %s best %d bonus %d yahtzee %d",
757 i,
758 strategy,
759 total.wins,
760 average(total.sum, count),
761 total.best,
762 total.bonus,
763 total.yahtzee
764 )
765 end

767function Program:execute(words)
768 local command = words[1]
769 if command == "RULE" then
770 if words[2] ~= "ROLLS" and words[2] ~= "JOKER" then
771 fail("unknown rule %s", words[2])
772 end
773 local rolls, joker = self.rolls, self.joker
774 if words[2] == "ROLLS" then
775 rolls = positive(words[3], 4)
776 elseif not jokers[words[3]] then
777 fail("bad value %s", words[3])
778 else
779 joker = words[3]
780 end
781 if self.game then
782 fail("game in progress")
783 end
784 self.rolls, self.joker = rolls, joker
785 self:emit("rule %s %s", words[2], words[3])
786 elseif command == "GAME" then
787 local seed = positive(words[2], 2147483646)
788 local names = {}
789 for i = 3, #words do
790 check_name(words[i])
791 names[#names + 1] = words[i]
792 end
793 if #names > 6 then
794 fail("too many players")
795 end
796 if self.game then
797 fail("game in progress")
798 end
799 local seen = {}
800 for _, name in ipairs(names) do
801 if seen[name] then
802 fail("duplicate name %s", name)
803 end
804 seen[name] = true
805 end
806 self.game =
807 Game.new(
808 seed,
809 names,
810 self.rolls,
811 self.joker,
812 function(line)
813 self.output[#self.output + 1] = line
814 end
815 )
816 self:emit("game seed %d players %s", seed, table.concat(names, " "))
817 self:emit("turn 1 %s", names[1])
