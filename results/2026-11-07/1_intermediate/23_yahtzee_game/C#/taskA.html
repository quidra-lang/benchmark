21 static List<Outcome>[] MakeOutcomes()
22 {
23 int[] factorial = [1, 1, 2, 6, 24, 120];
24 var result = new List<Outcome>[6];
25 for (int size = 0; size <= 5; size++)
26 {
27 result[size] = [];
28 void Add(int face, int remaining, int code, int divisor)
29 {
30 if (face == 6)
31 {
32 if (remaining == 0) result[size].Add(new(code, factorial[size] / divisor));
33 return;
34 }
35 for (int count = 0; count <= remaining; count++)
36 Add(face + 1, remaining - count, code + count * Powers[face], divisor * factorial[count]);
37 }
38 Add(0, size, 0, 1);
39 }
40 return result;
41 }

57 public static bool Held(int mask, int die) => (mask & (1 << (4 - die))) != 0;

66 {
67 int code = 0;
68 for (int i = 0; i < 5; i++) if (Held(mask, i)) code += Powers[faces[i] - 1];

72 public static int Frequent(int[] counts)
73 {
74 int face = 0;
75 for (int i = 1; i < 6; i++) if (counts[i] >= counts[face]) face = i;
76 return face;

79 public static int Normal(int category, int[] counts)
80 {
81 if (category < 6) return (category + 1) * counts[category];
82 int total = 0, maximum = 0, bits = 0;
83 for (int i = 0; i < 6; i++)
84 {
85 total += (i + 1) * counts[i];
86 maximum = Math.Max(maximum, counts[i]);
87 if (counts[i] > 0) bits |= 1 << i;
88 }
89 return category switch
90 {
91 6 => maximum >= 3 ? total : 0,
92 7 => maximum >= 4 ? total : 0,
93 8 => counts.Contains(3) && counts.Contains(2) ? 25 : 0,
94 9 => (bits & 15) == 15 || (bits & 30) == 30 || (bits & 60) == 60 ? 30 : 0,
95 10 => bits == 31 || bits == 62 ? 40 : 0,
96 11 => maximum == 5 ? 50 : 0,
97 _ => total,
98 };
99 }

101 public readonly record struct Candidates(int[] Points, int Allowed, bool Joker, int Best);
102
103 public static Candidates Evaluate(Player player, int[] counts, JokerRule rule)
104 {
105 int allowed = 0;
106 for (int c = 0; c < 13; c++) if (player.Scores[c] is null) allowed |= 1 << c;
107 bool joker = rule != JokerRule.Off && counts.Contains(5) && player.Scores[11] is not null;
108 if (joker && rule == JokerRule.Forced)
109 {
110 int upper = Array.IndexOf(counts, 5);
111 if ((allowed & (1 << upper)) != 0) allowed = 1 << upper;
112 else if ((allowed & LowerMask) != 0) allowed &= LowerMask;
113 }
114 var points = new int[13];
115 int best = -1;
116 for (int c = 0; c < 13; c++)
117 {
118 points[c] = player.Scores[c] is not null ? -1 : (allowed & (1 << c)) == 0 ? 0
119 : joker && (c is >= 8 and <= 10) ? c == 8 ? 25 : c == 9 ? 30 : 40 : Normal(c, counts);
120 if (points[c] >= 0 && (best < 0 || points[c] > points[best])) best = c;
121 }
122 return new(points, allowed, joker, best);
123 }

125 public static string Fraction(long numerator, long denominator)
126 {
127 long a = numerator, b = denominator;
128 while (b != 0) (a, b) = (b, a % b);
129 numerator /= a;
130 denominator /= a;
131 return denominator == 1 ? numerator.ToString() : $"{numerator}/{denominator}";
132 }
133
134 public static string Average(long sum, long count)
135 {
136 long value = count == 0 ? 0 : 100 * sum / count;
137 return $"{value / 100}.{value % 100:00}";
138 }

142{

147 public int Upper => Scores.Take(6).Sum(value => value ?? 0);
148 public int UpperBonus => Upper >= 63 ? 35 : 0;

157sealed class Game(long seed, Player[] players, int rolls, JokerRule joker, Action<string>? output)
158{
159 public Player[] Players { get; } = players;
160 public long RandomState { get; private set; } = seed;
161 public int MaxRolls { get; } = rolls;
162 public JokerRule Joker { get; } = joker;
163 public Queue<int> Loaded { get; } = new();
164 public long[] FaceCounts { get; } = new long[6];
165 public List<ScoreEntry> Entries { get; } = [];
166 public Stack<RedoEntry> RedoList { get; } = new();
167 public Turn State { get; private set; } = new();
168 public Player Current => Players[State.Seat];
169 public void Emit(string line) => output?.Invoke(line);
170
171 public void CheckTurn()
172 {
173 if (State.Finished) throw new CommandError("game over");
174 }
175
176 public void CheckRolled()
177 {
178 CheckTurn();
179 if (State.Rolls == 0) throw new CommandError("not rolled");
180 }
181
182 public void CheckRollsLeft()
183 {
184 CheckRolled();
185 if (State.Rolls == MaxRolls) throw new CommandError("no rolls left");
186 }
187
188 public void Roll()
189 {
190 CheckTurn();
191 if (State.Rolls == MaxRolls) throw new CommandError("no rolls left");
192 if (State.Hold == 31) throw new CommandError("all dice held");
193 RedoList.Clear();
194 var faces = State.Faces is null ? new int[5] : (int[])State.Faces.Clone();
195 for (int i = 0; i < 5; i++)
196 {
197 if (State.Rolls > 0 && Dice.Held(State.Hold, i)) continue;
198 if (Loaded.Count > 0) faces[i] = Loaded.Dequeue();
199 else { RandomState = 48271 * RandomState % 2147483647; faces[i] = (int)(RandomState % 6) + 1; }
200 FaceCounts[faces[i] - 1]++;
201 }
202 State = State with { Rolls = State.Rolls + 1, Faces = faces };
203 Emit($"{Current.Name} roll {State.Rolls}: {Dice.Faces(faces)}");
204 }
205
206 public void Hold(int mask)
207 {
208 CheckRollsLeft();
209 RedoList.Clear();
210 State = State with { Hold = mask };
211 Emit($"{Current.Name} holds {Dice.Pattern(mask)}: {Dice.Faces(State.Faces, mask)}");
212 }
213
214 public void Keep(int[] values)
215 {
216 CheckRollsLeft();
217 int mask = 0;
218 foreach (int value in values)
219 {
220 int die = -1;
221 for (int i = 0; i < 5; i++)
222 if (!Dice.Held(mask, i) && State.Faces![i] == value) { die = i; break; }
223 if (die < 0) throw new CommandError($"not showing {value}");
224 mask |= 1 << (4 - die);
225 }
226 Hold(mask);
227 }
228
229 public void Score(int category, bool preview = false)
230 {
231 CheckRolled();
232 var player = Current;
233 if (player.Scores[category] is not null) throw new CommandError($"{Dice.Categories[category]} already used");
234 int[] counts = Dice.Counts(State.Faces!);
235 var candidates = Dice.Evaluate(player, counts, Joker);
236 if ((candidates.Allowed & (1 << category)) == 0)
237 {
238 int upper = Array.IndexOf(counts, 5);
239 throw new CommandError($"joker requires {(player.Scores[upper] is null ? Dice.Categories[upper] : "lower box")}");
240 }
241 int points = candidates.Points[category];
242 int bonus = counts.Contains(5) && player.Scores[11] == 50 ? 100 : 0;
243 int upperBonus = category < 6 && player.Upper < 63 && player.Upper + points >= 63 ? 35 : 0;
244 string line = $"{player.Name} {(preview ? "would score" : "scores")} {points} in {Dice.Categories[category]}";
245 if (candidates.Joker) line += " joker";
246 if (bonus != 0) line += " ybonus 100";
247 if (upperBonus != 0) line += " ubonus 35";
248 line += $" total {player.Total + points + bonus + upperBonus}";
249 if (preview) { Emit(line); return; }
250 RedoList.Clear();
251 Entries.Add(new(State, category, points, bonus, candidates.Joker));
252 player.Scores[category] = points;
253 player.YahtzeeBonus += bonus;
254 State = Entries.Count == 13 * Players.Length ? new(Finished: true) : new((State.Seat + 1) % Players.Length);
255 Emit(line);
256 Emit(State.Finished ? "game over" : $"turn {Current.Round} {Current.Name}");
257 }
258
259 public void Undo()
260 {
261 if (Entries.Count == 0) throw new CommandError("nothing to undo");
262 var entry = Entries[^1];
263 Entries.RemoveAt(Entries.Count - 1);
264 RedoList.Push(new(State, entry));
265 var player = Players[entry.Before.Seat];
266 player.Scores[entry.Category] = null;
267 player.YahtzeeBonus -= entry.Bonus;
268 State = entry.Before;
269 Emit($"undo {player.Name} {Dice.Categories[entry.Category]} {entry.Points}");
270 }
271
272 public void Redo()
273 {
274 if (RedoList.Count == 0) throw new CommandError("nothing to redo");
275 var saved = RedoList.Pop();
276 var entry = saved.Entry;
277 var player = Players[entry.Before.Seat];
278 player.Scores[entry.Category] = entry.Points;
279 player.YahtzeeBonus += entry.Bonus;
280 Entries.Add(entry);
281 State = saved.After;
282 Emit($"redo {player.Name} {Dice.Categories[entry.Category]} {entry.Points}");
283 }

285 int BotHold(Strategy strategy)
286 {
287 int[] faces = State.Faces!, counts = Dice.Counts(faces);
288 int frequent = Dice.Frequent(counts), mask = 0;
289 if (strategy == Strategy.Straight)
290 {
291 int bestStart = 0, bestLength = 0;
292 for (int start = 0; start < 6; start++)
293 {
294 int end = start;
295 while (end < 6 && counts[end] > 0) end++;
296 if (end - start >= bestLength) { bestStart = start; bestLength = end - start; }
297 }
298 if (bestLength >= 3)
299 {
300 for (int face = bestStart + 1; face <= bestStart + bestLength; face++)
301 mask |= 1 << (4 - Array.IndexOf(faces, face));
302 return mask;
303 }
304 }
305 for (int i = 0; i < 5; i++)
306 if (strategy == Strategy.High ? faces[i] >= 5 : faces[i] == frequent + 1) mask |= 1 << (4 - i);
307 return mask;
308 }
309
310 public void BotTurn()
311 {
312 var strategy = Current.Strategy;
313 if (State.Rolls == 0) Roll();
314 while (State.Rolls < MaxRolls)
315 {
316 int mask = BotHold(strategy);
317 if (mask == 31) break;
318 Hold(mask);
319 Roll();
320 }
321 int[] counts = Dice.Counts(State.Faces!);
322 var candidates = Dice.Evaluate(Current, counts, Joker);
323 int frequent = Dice.Frequent(counts);
324 int category = strategy == Strategy.Upper && counts[frequent] >= 3 && (candidates.Allowed & (1 << frequent)) != 0
325 ? frequent : candidates.Best;
326 Score(category);
327 }

329 public int[] Ranking() => Enumerable.Range(0, Players.Length).OrderByDescending(i => Players[i].Total).ThenBy(i => i).ToArray();

332sealed class Adviser(Player player, JokerRule rule)
333{
334 readonly Dictionary<(int Depth, int Code), long> values = [];
335 readonly Dictionary<(int Depth, int Code), long> expectations = [];
336
337 long Value(int code, int depth)
338 {
339 if (values.TryGetValue((depth, code), out long cached)) return cached;
340 int[] counts = Dice.Decode(code);
341 long best;
342 if (depth == 0)
343 {
344 var candidates = Dice.Evaluate(player, counts, rule);
345 best = candidates.Points[candidates.Best];
346 }
347 else
348 {
349 best = -1;
350 void Search(int face, int held)
351 {
352 if (face == 6) { best = Math.Max(best, Expected(held, depth)); return; }
353 for (int n = 0; n <= counts[face]; n++) Search(face + 1, held + n * Dice.Powers[face]);
354 }
355 Search(0, 0);
356 }
357 values[(depth, code)] = best;
358 return best;
359 }
360
361 public long Expected(int held, int depth)
362 {
363 if (expectations.TryGetValue((depth, held), out long cached)) return cached;
364 int rerolled = 5 - Dice.Decode(held).Sum();
365 long sum = 0;
366 foreach (var outcome in Dice.Outcomes[rerolled]) sum += outcome.Weight * Value(held + outcome.Code, depth - 1);
367 long result = sum * Dice.Powers[5 - rerolled];
368 expectations[(depth, held)] = result;
369 return result;
370 }
371}

391 if (word[0] < '1' || word[0] > '9' || word.Any(c => c < '0' || c > '9')) throw new CommandError($"bad value {word}");

420 void Odds(Game table, int mask)
421 {
422 table.CheckRollsLeft();
423 int held = Dice.HeldCode(table.State.Faces!, mask), rerolled = 5 - Dice.Decode(held).Sum();
424 long outcomes = Dice.Powers[rerolled], best = 0;
425 var sums = new long[13];
426 var chances = new long[13];
427 foreach (var outcome in Dice.Outcomes[rerolled])
428 {
429 var candidates = Dice.Evaluate(table.Current, Dice.Decode(held + outcome.Code), table.Joker);
430 for (int c = 0; c < 13; c++)
431 {
432 if (candidates.Points[c] < 0) continue;
433 sums[c] += outcome.Weight * candidates.Points[c];
434 if (candidates.Points[c] > 0) chances[c] += outcome.Weight;
435 }
436 best += outcome.Weight * candidates.Points[candidates.Best];
437 }
438 Emit($"odds {Dice.Pattern(mask)} outcomes {outcomes}");
439 for (int c = 0; c < 13; c++)
440 if (table.Current.Scores[c] is null) Emit($"{Dice.Categories[c]} expect {Dice.Fraction(sums[c], outcomes)} chance {Dice.Fraction(chances[c], outcomes)}");
441 Emit($"best expect {Dice.Fraction(best, outcomes)}");
442 }

444 void Advise(Game table)
445 {
446 table.CheckRollsLeft();
447 int depth = table.MaxRolls - table.State.Rolls, chosen = 0;
448 long best = -1, denominator = 1;
449 for (int i = 0; i < depth; i++) denominator *= 7776;
450 var adviser = new Adviser(table.Current, table.Joker);
451 for (int mask = 0; mask < 32; mask++)
452 {
453 long value = adviser.Expected(Dice.HeldCode(table.State.Faces!, mask), depth);
454 if (value > best) { best = value; chosen = mask; }
455 }
456 Emit($"advise {Dice.Pattern(chosen)}: {Dice.Faces(table.State.Faces, chosen)} expect {Dice.Fraction(best, denominator)}");
457 }

475 void Simulate(long seed, int count, Strategy[] strategies)
476 {
477 int size = strategies.Length;
478 var wins = new int[size];
479 var sums = new long[size];
480 var best = new int[size];
481 var bonuses = new int[size];
482 var yahtzees = new int[size];
483 long state = seed;
484 for (int n = 0; n < count; n++)
485 {
486 var players = strategies.Select((strategy, i) => new Player($"BOT{i + 1}", strategy)).ToArray();
487 var table = new Game(state, players, rolls, joker, null);
488 while (!table.State.Finished) table.BotTurn();
489 state = table.RandomState;
490 wins[table.Ranking()[0]]++;
491 for (int i = 0; i < size; i++)
492 {
493 sums[i] += players[i].Total;
494 best[i] = Math.Max(best[i], players[i].Total);
495 if (players[i].UpperBonus == 35) bonuses[i]++;
496 if (players[i].Scores[11] == 50) yahtzees[i]++;
497 }
498 }
499 Emit($"sim seed {seed} games {count} players {size}");
500 for (int i = 0; i < size; i++) Emit($"seat {i + 1} {Dice.Strategies[(int)strategies[i]]} wins {wins[i]} average {Dice.Average(sums[i], count)} best {best[i]} bonus {bonuses[i]} yahtzee {yahtzees[i]}");
501 }

503 void Execute(string[] words)
504 {
505 switch (words[0])
506 {
507 case "RULE":
508 {
509 if (words[1] != "ROLLS" && words[1] != "JOKER") throw new CommandError($"unknown rule {words[1]}");
510 int newRolls = rolls;
511 JokerRule newJoker = joker;
512 if (words[1] == "ROLLS") newRolls = (int)Positive(words[2], 4);
513 else newJoker = words[2] switch { "FORCED" => JokerRule.Forced, "FREE" => JokerRule.Free, "OFF" => JokerRule.Off, _ => throw new CommandError($"bad value {words[2]}") };
514 if (game is not null) throw new CommandError("game in progress");
515 rolls = newRolls; joker = newJoker;
516 Emit($"rule {words[1]} {words[2]}");
517 break;
518 }
519 case "GAME":
520 {
521 long seed = Positive(words[1], 2147483646);
522 string[] names = words[2..];
523 foreach (string name in names) Name(name);
524 if (names.Length > 6) throw new CommandError("too many players");
525 if (game is not null) throw new CommandError("game in progress");
526 var seen = new HashSet<string>(StringComparer.Ordinal);
527 foreach (string name in names) if (!seen.Add(name)) throw new CommandError($"duplicate name {name}");
528 game = new(seed, names.Select(name => new Player(name)).ToArray(), rolls, joker, Emit);
529 Emit($"game seed {seed} players {string.Join(" ", names)}");
530 Emit($"turn 1 {names[0]}");
531 break;
532 }
533 case "BOT":
534 {
535 Name(words[1]);
536 var strategy = BotStrategy(words[2]);
537 var table = Table();
538 var player = table.Players.FirstOrDefault(p => p.Name == words[1]) ?? throw new CommandError($"no player {words[1]}");
539 player.Strategy = strategy;
540 Emit($"bot {words[1]} {words[2]}");
541 break;
542 }
543 case "LOAD":
544 case "KEEP":
545 {
546 int[] values = words.Skip(1).Select(word => (int)Positive(word, 6)).ToArray();
547 var table = Table();
548 if (words[0] == "KEEP") table.Keep(values);
549 else { foreach (int value in values) table.Loaded.Enqueue(value); Emit($"loaded {table.Loaded.Count}"); }
550 break;
551 }
552 case "ROLL": Table().Roll(); break;
553 case "HOLD":
554 case "ODDS":
555 {
556 int mask = Pattern(words[1]);
557 var table = Table();
558 if (words[0] == "HOLD") table.Hold(mask); else Odds(table, mask);
559 break;
560 }
561 case "SCORE":
562 case "WHATIF":
563 {
564 int category = Category(words[1]);
565 Table().Score(category, words[0] == "WHATIF");
566 break;
567 }
568 case "UNDO": Table().Undo(); break;
569 case "REDO": Table().Redo(); break;
570 case "SUGGEST":
571 {
572 var table = Table();
573 table.CheckRolled();
574 var candidates = Dice.Evaluate(table.Current, Dice.Counts(table.State.Faces!), table.Joker);
575 string line = "suggest";
576 for (int c = 0; c < 13; c++) if (candidates.Points[c] >= 0) line += $" {Dice.Categories[c]} {candidates.Points[c]}";
577 Emit($"{line} best {Dice.Categories[candidates.Best]} {candidates.Points[candidates.Best]}");
578 break;
579 }
580 case "ADVISE": Advise(Table()); break;
581 case "AUTO":
582 {
583 int count = (int)Positive(words[1], 100);
584 var table = Table();
585 table.CheckTurn();
586 table.RedoList.Clear();
587 for (int i = 0; i < count && !table.State.Finished; i++) table.BotTurn();
588 break;
589 }
590 case "STATUS":
591 {
592 var table = Table();
593 var turn = table.State;
594 Emit(turn.Finished ? "status game over" : $"status round {table.Current.Round} player {table.Current.Name} rolls {turn.Rolls} dice {Dice.Faces(turn.Faces)} holds {Dice.Pattern(turn.Hold)}");
595 break;
596 }
597 case "CARD": Card(Table()); break;
598 case "BONUS":
599 {
600 foreach (var player in Table().Players)
601 {
602 int maximum = player.Upper;
603 for (int c = 0; c < 6; c++) if (player.Scores[c] is null) maximum += 5 * (c + 1);
604 string status = player.Upper >= 63 ? "made" : maximum < 63 ? "lost" : "open";
605 Emit($"bonus {player.Name} upper {player.Upper} need {Math.Max(63 - player.Upper, 0)} max {maximum} {status}");
606 }
607 break;
608 }
609 case "STATS":
610 {
611 var table = Table();
612 for (int i = 0; i < table.Players.Length; i++)
613 {
614 var entries = table.Entries.Where(entry => entry.Before.Seat == i).ToArray();
615 Emit($"stats {table.Players[i].Name} turns {entries.Length} rolls {entries.Sum(e => e.Before.Rolls)} zeros {entries.Count(e => e.Points == 0)} yahtzees {entries.Count(e => Dice.Counts(e.Before.Faces!).Contains(5))} average {Dice.Average(entries.Sum(e => e.Points), entries.Length)}");
616 }
617 break;
618 }
619 case "LOG":
620 {
621 var table = Table();
622 for (int i = 0; i < table.Entries.Count; i++)
623 {
624 var entry = table.Entries[i];
625 string line = $"log {i + 1} {table.Players[entry.Before.Seat].Name} {Dice.Faces(entry.Before.Faces)} {Dice.Categories[entry.Category]} {entry.Points}";
626 if (entry.Joker) line += " joker";
627 if (entry.Bonus != 0) line += " ybonus 100";
628 Emit(line);
629 }
630 if (table.Entries.Count == 0) Emit("no scores");
631 break;
632 }
633 case "FACES":
634 {
635 long[] counts = Table().FaceCounts;
636 long total = counts.Sum(), numerator = 6 * counts.Sum(c => c * c) - total * total;
637 Emit($"faces {string.Join(" ", counts)} total {total} chi2 {Dice.Fraction(numerator, total == 0 ? 1 : total)}");
638 break;
639 }
640 case "STANDINGS":
641 case "END":
642 {
643 var table = Table();
644 int[] ranking = table.Ranking();
645 bool end = words[0] == "END";
646 if (end) Emit(table.State.Finished ? "end final" : "end abandoned");
647 for (int rank = 0; rank < ranking.Length; rank++)
648 {
649 var player = table.Players[ranking[rank]];
650 if (!end) Emit($"standing {rank + 1} {player.Name} {player.Total} behind {table.Players[ranking[0]].Total - player.Total}");
651 else
652 {
653 Emit($"rank {rank + 1} {player.Name} {player.Total}");
654 if (table.State.Finished)
655 {
656 if (!records.TryGetValue(player.Name, out var record)) { record = new(); records.Add(player.Name, record); }
657 record.Games++; record.Sum += player.Total; record.Best = Math.Max(record.Best, player.Total);
658 if (rank == 0) record.Wins++;
659 }
660 }
661 }
662 if (end) game = null;
663 break;
664 }
665 case "RECORDS":
666 {
667 foreach (var item in records.OrderByDescending(item => item.Value.Wins).ThenByDescending(item => item.Value.Best).ThenBy(item => item.Key, StringComparer.Ordinal))
668 Emit($"record {item.Key} games {item.Value.Games} wins {item.Value.Wins} best {item.Value.Best} average {Dice.Average(item.Value.Sum, item.Value.Games)}");
669 if (records.Count == 0) Emit("no records");
670 break;
671 }
672 case "SIM":
673 {
674 long seed = Positive(words[1], 2147483646);
675 int count = (int)Positive(words[2], 50);
676 var strategies = words.Skip(3).Select(BotStrategy).ToArray();
677 if (strategies.Length > 6) throw new CommandError("too many players");
678 Simulate(seed, count, strategies);
679 break;
680 }
681 default: throw new CommandError($"unknown command {words[0]}");
682 }
