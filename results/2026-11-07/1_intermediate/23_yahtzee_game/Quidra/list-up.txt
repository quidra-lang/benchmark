string[] category_names()
    return ["ONES", "TWOS", "THREES", "FOURS", "FIVES", "SIXES", "THREEKIND", "FOURKIND", "FULLHOUSE", "SMALL", "LARGE", "YAHTZEE", "CHANCE"]

bool includes(const string[] values, string target)
    for value := values[]
        if value == target
            return true
    return false

int | error integer_word(string text, int minimum, int maximum)
    start := if text[0] == "-" then 1 else 0
    if start == int(#text) or text == "-0"
        return error("bad value " + text)
    if int(#text) - start > 1 and text[start] == "0"
        return error("bad value " + text)
    for: start <= index < int(#text)
        if not "0123456789".contains(text[index])
            return error("bad value " + text)
    value := int(text)
    if value < minimum or value > maximum
        return error("bad value " + text)
    return value

string | error player_name(string text)
    if #text > 8
        return error("bad name " + text)
    for point := text.codepoints()[]
        if point < 65 or point > 90
            return error("bad name " + text)
    return text
int | error category_word(string text)
    names := category_names()
    for index := names.indices()[]
        if names[index] == text
            return int(index)
    return error("bad category " + text)

bool[] | error hold_pattern(string text)
    if #text != 5
        return error("bad pattern " + text)
    bool[] marks := []
    for: 0 <= index < 5
        if text[index] != "0" and text[index] != "1"
            return error("bad pattern " + text)
        marks = marks.append(text[index] == "1")
    return marks

int[] | error face_words(const string[] tokens)
    int[] faces := []
    for token := tokens[]
        face := try integer_word(token, 1, 6)
        faces = faces.append(face)
    return faces
string faces_text(const int[] dice)
    return if #dice == 0 then "-" else [for face := dice[] then "{face}"].join(" ")

string pattern_text(const bool[] marks)
    return [for mark := marks[] then if mark then "1" else "0"].join("")

int[] held_faces(const int[] dice, const bool[] marks)
    return [for index := dice.indices()[] if marks[index] then dice[index]]

int held_count(const bool[] marks)
    selected := [for mark := marks[] if mark then 1]
    return selected.sum()

int[] face_counts(const int[] dice)
    counts := array(6, fill = 0)
    for face := dice[]
        counts[face - 1] += 1
    return counts
class Rules()
    rolls := 3
    joker := "FORCED"

class Player(string name, strategy := "KIND")
    boxes := array(13, fill = -1)
    yahtzee_bonus := 0

int upper_subtotal(const Player player)
    return [for score := player.boxes[0:<6][] if score >= 0 then score].sum()

int player_total(const Player player)
    return [for score := player.boxes[] if score >= 0 then score].sum() + player.yahtzee_bonus + (if upper_subtotal(player) >= 63 then 35 else 0)

int player_round(const Player player)
    return int(#[for score := player.boxes[] if score >= 0 then score]) + 1

class CandidateScores()
    joker := false
    allowed := array(13, fill = false)
    points := array(13, fill = 0)
    best := 0
    requirement := ""

CandidateScores candidates(const Player player, const int[] counts, string joker_rule)
    result := CandidateScores()
    result.allowed = [for score := player.boxes[] then score < 0]
    largest := counts.max()
    total := (counts * [1, 2, 3, 4, 5, 6]).sum()
    result.joker = joker_rule != "OFF" and largest == 5 and player.boxes[11] >= 0
    if result.joker and joker_rule == "FORCED"
    best_points := -1
    for: 0 <= box < 13
        if not result.allowed[box]
            continue
        score := 0
        if box < 6
            score = (box + 1) * counts[box]
        result.points[box] = score
        if score > best_points
            best_points = score
            result.best = box
    return result

class Turn(int player)
    rolls := 0
    int[] dice := []
    holds := array(5, fill = false)

class ScoreEntry(Turn turn, int category, int points, int yahtzee_bonus, bool joker)
class RedoEntry(Turn resume, ScoreEntry score)

class Game(int state, Player[] players, Rules rules, silent := false)
    turn := Turn(0)
    int[] loaded := []
    face_totals := array(6, fill = 0)
    ScoreEntry[] entries := []
    RedoEntry[] redo := []

void game_emit(const Game table, string text)
    if not table.silent
        emit(text)

void | error turn_check(const Game table)
    if table.turn.player < 0
        return error("game over")
    return void

void | error rolled_check(const Game table)
    try turn_check(table)
    if table.turn.rolls == 0
        return error("not rolled")
    return void

void | error rolls_left_check(const Game table)
    try rolled_check(table)
    if table.turn.rolls == table.rules.rolls
        return error("no rolls left")
    return void

string turn_text(const Game table)
    if table.turn.player < 0
        return "game over"
    player := table.players[table.turn.player]
    return "turn {player_round(player)} {player.name}"

void roll_dice(Game &table)
    if table.turn.rolls == 0
        table.turn.dice = array(5, fill = 0)
    for: 0 <= die < 5
        if table.turn.holds[die]
            continue
        int face
        if #table.loaded > 0
            face = table.loaded[0]
            table.loaded = table.loaded[1:]
        else
            table.state = table.state * 48271 % 2147483647
            face = table.state % 6 + 1
        table.turn.dice[die] = face
        table.face_totals[face - 1] += 1
    table.turn.rolls += 1
    table.redo = []
    game_emit(table, "{table.players[table.turn.player].name} roll {table.turn.rolls}: {faces_text(table.turn.dice)}")

void set_holds(Game &table, const bool[] marks)
    table.turn.holds = marks
    table.redo = []
    game_emit(table, "{table.players[table.turn.player].name} holds {pattern_text(marks)}: {faces_text(held_faces(table.turn.dice, marks))}")

ScoreEntry score_details(const Game table, int category)
    player := table.players[table.turn.player]
    counts := face_counts(table.turn.dice)
    options := candidates(player, counts, table.rules.joker)
    bonus := if counts.max() == 5 and player.boxes[11] == 50 then 100 else 0
    return ScoreEntry(table.turn, category, options.points[category], bonus, options.joker)

void | error score_check(const Game table, int category)
    try rolled_check(table)
    player := table.players[table.turn.player]
    if player.boxes[category] >= 0
        return error(category_names()[category] + " already used")
    options := candidates(player, face_counts(table.turn.dice), table.rules.joker)
    if not options.allowed[category]
        return error("joker requires " + options.requirement)
    return void

string score_text(const Game table, const ScoreEntry entry, hypothetical := false)
    player := table.players[entry.turn.player]
    upper := upper_subtotal(player)
    after := upper + (if entry.category < 6 then entry.points else 0)
    added_upper := if upper < 63 and after >= 63 then 35 else 0
    verb := if hypothetical then "would score" else "scores"
    text := "{player.name} {verb} {entry.points} in {category_names()[entry.category]}"
    if entry.joker
        text += " joker"
    if entry.yahtzee_bonus > 0
        text += " ybonus 100"
    if added_upper > 0
        text += " ubonus 35"
    return text + " total {player_total(player) + entry.points + entry.yahtzee_bonus + added_upper}"

void enter_score(Game &table, int category)
    entry := score_details(table, category)
    game_emit(table, score_text(table, entry))
    table.players[entry.turn.player].boxes[category] = entry.points
    table.players[entry.turn.player].yahtzee_bonus += entry.yahtzee_bonus
    table.entries = table.entries.append(entry)
    table.redo = []
    next := if #table.entries == 13 * #table.players then -1 else (entry.turn.player + 1) % int(#table.players)
    table.turn = Turn(next)
    game_emit(table, turn_text(table))
class Program(RollTables tables)
    rules := Rules()
    Game | none game := none
Game | error require_game(const Program program)
    match program.game
        Game table
            return table
        none
            return error("no game")

void | error game_command(Program &program, const string[] tokens)
    op := tokens[0]
    name := ""
    strategy := ""
    count := 0
    category := 0
    int[] faces := []
    bool[] marks := []
    elif op == "LOAD" or op == "KEEP"
        faces = try face_words(tokens[1:])
    elif op == "HOLD" or op == "ODDS"
        marks = try hold_pattern(tokens[1])
    elif op == "SCORE" or op == "WHATIF"
        category = try category_word(tokens[1])
    table := try require_game(program)
    if op == "BOT"
    elif op == "LOAD"
        table.loaded = table.loaded.concat(faces)
        emit("loaded {#table.loaded}")
    elif op == "ROLL"
        try turn_check(table)
        if table.turn.rolls == table.rules.rolls
            return error("no rolls left")
        if held_count(table.turn.holds) == 5
            return error("all dice held")
        roll_dice(&table)
    elif op == "HOLD"
        try rolls_left_check(table)
        set_holds(&table, marks)
    elif op == "SCORE" or op == "WHATIF"
        try score_check(table, category)
        if op == "WHATIF"
            emit(score_text(table, score_details(table, category), true))
        else
            enter_score(&table, category)
    elif op == "UNDO"
        if #table.entries == 0
            return error("nothing to undo")
        entry := table.entries[-1]
        table.entries = table.entries[0:<int(#table.entries) - 1]
        table.redo = table.redo.append(RedoEntry(table.turn, entry))
        table.players[entry.turn.player].boxes[entry.category] = -1
        table.players[entry.turn.player].yahtzee_bonus -= entry.yahtzee_bonus
        table.turn = entry.turn
        emit("undo {table.players[entry.turn.player].name} {category_names()[entry.category]} {entry.points}")
    elif op == "REDO"
        if #table.redo == 0
            return error("nothing to redo")
        saved := table.redo[-1]
        table.redo = table.redo[0:<int(#table.redo) - 1]
        entry := saved.score
        table.players[entry.turn.player].boxes[entry.category] = entry.points
        table.players[entry.turn.player].yahtzee_bonus += entry.yahtzee_bonus
        table.entries = table.entries.append(entry)
        table.turn = saved.resume
        emit("redo {table.players[entry.turn.player].name} {category_names()[entry.category]} {entry.points}")
    program.game = table
    return void

void | error command(Program &program, const string[] tokens)
    op := tokens[0]
    commands := ["RULE", "GAME", "BOT", "LOAD", "ROLL", "HOLD", "KEEP", "SCORE", "WHATIF", "UNDO", "REDO", "SUGGEST", "ODDS", "ADVISE", "AUTO", "STATUS", "CARD", "BONUS", "STATS", "LOG", "FACES", "STANDINGS", "END", "RECORDS", "SIM"]
    if not includes(commands, op)
        return error("unknown command " + op)
    if op == "RULE"
    elif op == "GAME"
        seed := try integer_word(tokens[1], 1, 2147483646)
        names := tokens[2:]
        for name := names[]
            try player_name(name)
        if #names > 6
            return error("too many players")
        match program.game
            Game
                return error("game in progress")
            none
                string[] seen := []
                for name := names[]
                    if includes(seen, name)
                        return error("duplicate name " + name)
                    seen = seen.append(name)
        players := [for name := names[] then Player(name)]
        program.game = Game(seed, players, program.rules)
        emit("game seed {seed} players {names.join(" ")}")
        emit("turn 1 {names[0]}")
    else
        try game_command(&program, tokens)
    return void
