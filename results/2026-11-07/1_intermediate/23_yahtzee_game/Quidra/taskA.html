130CandidateScores candidates(const Player player, const int[] counts, string joker_rule)
131 result := CandidateScores()
132 result.allowed = [for score := player.boxes[] then score < 0]
133 largest := counts.max()
134 total := (counts * [1, 2, 3, 4, 5, 6]).sum()
135 result.joker = joker_rule != "OFF" and largest == 5 and player.boxes[11] >= 0
136 if result.joker and joker_rule == "FORCED"
137 face := most_frequent(counts)
138 if player.boxes[face] < 0
139 result.allowed = array(13, fill = false)
140 result.allowed[face] = true
141 result.requirement = category_names()[face]
142 elif #[for available := result.allowed[6:][] if available then available] > 0
143 for: 0 <= index < 6
144 result.allowed[index] = false
145 result.requirement = "lower box"
146 best_points := -1
147 for: 0 <= box < 13
148 if not result.allowed[box]
149 continue
150 score := 0
151 if box < 6
152 score = (box + 1) * counts[box]
153 elif box == 6 or box == 7
154 score = if largest >= box - 3 then total else 0
155 elif box == 8
156 score = if result.joker or (largest == 3 and #[for frequency := counts[] if frequency == 2 then frequency] > 0) then 25 else 0
157 elif box == 9 or box == 10
158 length := box - 5
159 for: 0 <= start <= 6 - length
160 if #[for frequency := counts[start:<start + length][] if frequency > 0 then frequency] == nat(length)
161 score = if box == 9 then 30 else 40
162 if result.joker
163 score = if box == 9 then 30 else 40
164 elif box == 11
165 score = if largest == 5 then 50 else 0
166 else
167 score = total
168 result.points[box] = score
169 if score > best_points
170 best_points = score
171 result.best = box
172 return result

541void | error game_command(Program &program, const string[] tokens)
542 op := tokens[0]
543 name := ""
544 strategy := ""
545 count := 0
546 category := 0
547 int[] faces := []
548 bool[] marks := []
549 if op == "BOT"
550 name = try player_name(tokens[1])
551 strategy = try strategy_word(tokens[2])
552 elif op == "LOAD" or op == "KEEP"
553 faces = try face_words(tokens[1:])
554 elif op == "HOLD" or op == "ODDS"
555 marks = try hold_pattern(tokens[1])
556 elif op == "SCORE" or op == "WHATIF"
557 category = try category_word(tokens[1])
558 elif op == "AUTO"
559 count = try integer_word(tokens[1], 1, 100)
560 table := try require_game(program)
561 if op == "BOT"
562 seat := -1
563 for index := table.players.indices()[]
564 if table.players[index].name == name
565 seat = int(index)
566 if seat < 0
567 return error("no player " + name)
568 table.players[seat].strategy = strategy
569 emit("bot {name} {strategy}")
570 elif op == "LOAD"
571 table.loaded = table.loaded.concat(faces)
572 emit("loaded {#table.loaded}")
573 elif op == "ROLL"
574 try turn_check(table)
575 if table.turn.rolls == table.rules.rolls
576 return error("no rolls left")
577 if held_count(table.turn.holds) == 5
578 return error("all dice held")
579 roll_dice(&table)
580 elif op == "HOLD"
581 try rolls_left_check(table)
582 set_holds(&table, marks)
583 elif op == "KEEP"
584 try rolls_left_check(table)
585 marks = array(5, fill = false)
586 for face := faces[]
587 chosen := -1
588 for: 0 <= die < 5
589 if not marks[die] and table.turn.dice[die] == face
590 chosen = die
591 break
592 if chosen < 0
593 return error("not showing {face}")
594 marks[chosen] = true
595 set_holds(&table, marks)
596 elif op == "SCORE" or op == "WHATIF"
597 try score_check(table, category)
598 if op == "WHATIF"
599 emit(score_text(table, score_details(table, category), true))
600 else
601 enter_score(&table, category)
602 elif op == "UNDO"
603 if #table.entries == 0
604 return error("nothing to undo")
605 entry := table.entries[-1]
606 table.entries = table.entries[0:<int(#table.entries) - 1]
607 table.redo = table.redo.append(RedoEntry(table.turn, entry))
608 table.players[entry.turn.player].boxes[entry.category] = -1
609 table.players[entry.turn.player].yahtzee_bonus -= entry.yahtzee_bonus
610 table.turn = entry.turn
611 emit("undo {table.players[entry.turn.player].name} {category_names()[entry.category]} {entry.points}")
612 elif op == "REDO"
613 if #table.redo == 0
614 return error("nothing to redo")
615 saved := table.redo[-1]
616 table.redo = table.redo[0:<int(#table.redo) - 1]
617 entry := saved.score
618 table.players[entry.turn.player].boxes[entry.category] = entry.points
619 table.players[entry.turn.player].yahtzee_bonus += entry.yahtzee_bonus
620 table.entries = table.entries.append(entry)
621 table.turn = saved.resume
622 emit("redo {table.players[entry.turn.player].name} {category_names()[entry.category]} {entry.points}")
623 elif op == "SUGGEST"
