40Integer integerWord(const std::string& word, Integer minimum, Integer maximum) {
41 std::size_t first = word[0] == '-' ? 1 : 0;
42 if (first == word.size() || word == "-0" ||
43 (word.size() - first > 1 && word[first] == '0')) {
44 throw CommandError("bad value " + word);
45 }
46 Integer value = 0;
47 for (std::size_t i = first; i < word.size(); ++i) {
48 if (word[i] < '0' || word[i] > '9') throw CommandError("bad value " + word);
49 value = value * 10 + word[i] - '0';
50 }
51 if (first != 0) value = -value;
52 if (value < minimum || value > maximum) throw CommandError("bad value " + word);
53 return value;
54}

81std::string fraction(Integer numerator, Integer denominator) {
82 Integer divisor = std::gcd(numerator, denominator);
83 numerator /= divisor;
84 denominator /= divisor;
85 return std::to_string(numerator) + (denominator == 1 ? "" : "/" + std::to_string(denominator));
86}

123bool isYahtzee(const Counts& counts) { return *std::max_element(counts.begin(), counts.end()) == 5; }

125int normalScore(Category category, const Counts& counts) {
126 if (category <= Sixes) return (category + 1) * counts[category];
127 int sum = 0;
128 int largest = *std::max_element(counts.begin(), counts.end());
129 for (int face = 0; face < 6; ++face) sum += (face + 1) * counts[face];
130 switch (category) {
131 case ThreeKind: return largest >= 3 ? sum : 0;
132 case FourKind: return largest >= 4 ? sum : 0;
133 case FullHouse:
134 return std::find(counts.begin(), counts.end(), 3) != counts.end() &&
135 std::find(counts.begin(), counts.end(), 2) != counts.end() ? 25 : 0;
136 case Small:
137 case Large: {
138 int length = category == Small ? 4 : 5;
139 for (int start = 0; start <= 6 - length; ++start) {
140 bool present = true;
141 for (int offset = 0; offset < length; ++offset) present = present && counts[start + offset] > 0;
142 if (present) return category == Small ? 30 : 40;
143 }
144 return 0;
145 }
146 case Yahtzee: return largest == 5 ? 50 : 0;
147 default: return sum;
148 }
149}

153struct Player {
154 std::string name;
155 Strategy strategy = Strategy::Kind;
156 std::array<std::optional<int>, CategoryCount> boxes{};
157 int yahtzeeBonus = 0;
158
159 int upper() const {
160 int sum = 0;
161 for (int i = Ones; i <= Sixes; ++i) sum += boxes[i].value_or(0);
162 return sum;
163 }
164 int upperBonus() const { return upper() >= 63 ? 35 : 0; }
165 int total() const {
166 int sum = upperBonus() + yahtzeeBonus;
167 for (const std::optional<int>& box : boxes) sum += box.value_or(0);
168 return sum;
169 }
170 int round() const {
171 return 1 + static_cast<int>(std::count_if(boxes.begin(), boxes.end(), [](const std::optional<int>& box) { return box.has_value(); }));
172 }

175struct Candidates {
176 std::array<bool, CategoryCount> allowed{};
177 std::array<int, CategoryCount> points{};
178 Category best = Ones;
179 bool joker = false;

183Candidates candidates(const Player& player, const Counts& counts, Joker rule) {
184 Candidates result;
185 for (int i = 0; i < CategoryCount; ++i) result.allowed[i] = !player.boxes[i].has_value();
186 result.joker = rule != Joker::Off && isYahtzee(counts) && player.boxes[Yahtzee].has_value();
187 if (result.joker && rule == Joker::Forced) {
188 int face = mostFrequent(counts);
189 if (!player.boxes[face].has_value()) {
190 result.allowed.fill(false);
191 result.allowed[face] = true;
192 result.requirement = categoryNames[face];
193 } else {
194 bool lowerOpen = false;
195 for (int i = ThreeKind; i < CategoryCount; ++i) lowerOpen = lowerOpen || result.allowed[i];
196 if (lowerOpen) {
197 for (int i = Ones; i <= Sixes; ++i) result.allowed[i] = false;
198 result.requirement = "lower box";
199 }
200 }
201 }
202 int bestScore = -1;
203 for (int i = 0; i < CategoryCount; ++i) {
204 if (!result.allowed[i]) continue;
205 Category category = static_cast<Category>(i);
206 int points = normalScore(category, counts);
207 if (result.joker) {
208 if (category == FullHouse) points = 25;
209 if (category == Small) points = 30;
210 if (category == Large) points = 40;
211 }
212 result.points[i] = points;
213 if (points > bestScore) { bestScore = points; result.best = category; }
214 }
215 return result;
216}

218Holds botHolds(Strategy strategy, const Dice& dice) {
219 Holds holds{};
220 Counts counts = faceCounts(dice);
221 if (strategy == Strategy::High) {
222 for (int i = 0; i < 5; ++i) holds[i] = dice[i] >= 5;
223 return holds;
224 }
225 if (strategy == Strategy::Straight) {
226 int bestStart = 0, bestLength = 0;
227 for (int start = 0; start < 6; ++start) {
228 int length = 0;
229 while (start + length < 6 && counts[start + length] > 0) ++length;
230 if (length >= bestLength) { bestStart = start; bestLength = length; }
231 }
232 if (bestLength >= 3) {
233 for (int face = bestStart + 1; face <= bestStart + bestLength; ++face) {
234 for (int i = 0; i < 5; ++i) if (dice[i] == face) { holds[i] = true; break; }
235 }
236 return holds;
237 }
238 }
239 int target = mostFrequent(counts) + 1;
240 for (int i = 0; i < 5; ++i) holds[i] = dice[i] == target;
241 return holds;
242}

246class Rerolls {
247 void enumerate(int remaining, int face, int total, Counts& counts) {
248 if (face == 5) {
249 counts[face] = remaining;
250 int divisor = 1;
251 for (int count : counts) divisor *= factorial[count];
252 outcomes[total].push_back({counts, factorial[total] / divisor});
253 return;
254 }
255 for (int count = 0; count <= remaining; ++count) {
256 counts[face] = count;
257 enumerate(remaining - count, face + 1, total, counts);
258 }
259 }
260public:
261 std::array<std::vector<Outcome>, 6> outcomes;
262 Rerolls() {
263 Counts counts{};
264 for (int total = 0; total <= 5; ++total) enumerate(total, 0, total, counts);
265 }
266};

268Counts combined(const Counts& held, const Counts& rolled) {
269 Counts result{};
270 for (int i = 0; i < 6; ++i) result[i] = held[i] + rolled[i];
271 return result;
272}

274struct Advice { Holds holds; Integer numerator; Integer denominator; };
275
276Advice advise(const Player& player, const Dice& dice, int rollsLeft, Joker joker, const Rerolls& rerolls) {
277 std::map<Counts, Integer> values;
278 for (const Outcome& outcome : rerolls.outcomes[5]) {
279 Candidates scores = candidates(player, outcome.counts, joker);
280 values[outcome.counts] = scores.points[scores.best];
281 }
282 Integer denominator = 1;
283 std::map<Counts, Integer> expectations;
284 for (int depth = 1; depth <= rollsLeft; ++depth) {
285 expectations.clear();
286 for (int kept = 0; kept <= 5; ++kept) {
287 int rolled = 5 - kept;
288 for (const Outcome& held : rerolls.outcomes[kept]) {
289 Integer sum = 0;
290 for (const Outcome& outcome : rerolls.outcomes[rolled]) {
291 sum += outcome.weight * values.at(combined(held.counts, outcome.counts));
292 }
293 expectations[held.counts] = sum * (powersOfSix[5] / powersOfSix[rolled]);
294 }
295 }
296 denominator *= powersOfSix[5];
297 if (depth == rollsLeft) break;
298 for (auto& [counts, value] : values) {
299 Integer best = 0;
300 for (const auto& [held, expected] : expectations) {
301 bool available = true;
302 for (int face = 0; face < 6; ++face) available = available && held[face] <= counts[face];
303 if (available) best = std::max(best, expected);
304 }
305 value = best;
306 }
307 }
308 Advice best{{}, -1, denominator};
309 for (int mask = 0; mask < 32; ++mask) {
310 Holds holds{};
311 for (int i = 0; i < 5; ++i) holds[i] = (mask & (1 << (4 - i))) != 0;
312 Integer expected = expectations.at(faceCounts(dice, holds));
313 if (expected > best.numerator) best = {holds, expected, denominator};
314 }
315 return best;
316}

338 void emit(const std::string& line) { if (output != nullptr) output->push_back(line); }
339 int rollDie() {
340 int face;
341 if (!loaded.empty()) { face = loaded.front(); loaded.pop_front(); }
342 else { state = 48271 * state % 2147483647; face = static_cast<int>(state % 6) + 1; }
343 ++faceTotals[face - 1];
344 return face;
345 }

363 turnCheck();

370 std::vector<int> ranking() const {

372 std::iota(seats.begin(), seats.end(), 0);
373 std::stable_sort(seats.begin(), seats.end(), [&](int a, int b) { return players[a].total() > players[b].total(); });

376 std::string nextTurnText() const {
377 if (!turn.has_value()) return "game over";
378 const Player& player = players[turn->player];
379 return "turn " + std::to_string(player.round()) + " " + player.name;
380 }
381 void roll() {
382 redo.clear();
383 for (int i = 0; i < 5; ++i) if (turn->rolls == 0 || !turn->holds[i]) turn->dice[i] = rollDie();
384 ++turn->rolls;
385 emit(players[turn->player].name + " roll " + std::to_string(turn->rolls) + ": " + facesText(turn->dice));
386 }
387 void hold(const Holds& holds) {
388 redo.clear();
389 turn->holds = holds;
390 emit(players[turn->player].name + " holds " + patternText(holds) + ": " + facesText(turn->dice, holds));
391 }
392 ScoreEntry scoreDetails(Category category) const {
393 const Player& player = players[turn->player];
394 Counts counts = faceCounts(turn->dice);
395 Candidates scores = candidates(player, counts, rules.joker);
396 int bonus = isYahtzee(counts) && player.boxes[Yahtzee] == 50 ? 100 : 0;
397 return {*turn, category, scores.points[category], bonus, scores.joker};
398 }
399 ScoreEntry checkedScore(Category category) const {
400 rolledCheck();
401 const Player& player = players[turn->player];
402 if (player.boxes[category].has_value()) throw CommandError(categoryNames[category] + " already used");
403 Candidates scores = candidates(player, faceCounts(turn->dice), rules.joker);
404 if (!scores.allowed[category]) throw CommandError("joker requires " + scores.requirement);
405 return scoreDetails(category);
406 }
407 std::string scoreText(const ScoreEntry& entry, bool hypothetical) const {
408 const Player& player = players[entry.turn.player];
409 int upperAfter = player.upper() + (entry.category <= Sixes ? entry.points : 0);
410 int addedUpper = upperAfter >= 63 && player.upperBonus() == 0 ? 35 : 0;
411 std::string line = player.name + (hypothetical ? " would score " : " scores ") + std::to_string(entry.points) + " in " + categoryNames[entry.category];
412 if (entry.joker) line += " joker";
413 if (entry.yahtzeeBonus != 0) line += " ybonus 100";
414 if (addedUpper != 0) line += " ubonus 35";
415 return line + " total " + std::to_string(player.total() + entry.points + entry.yahtzeeBonus + addedUpper);
416 }
417 void score(Category category) {
418 ScoreEntry entry = scoreDetails(category);
419 std::string line = scoreText(entry, false);
420 Player& player = players[entry.turn.player];
421 player.boxes[category] = entry.points;
422 player.yahtzeeBonus += entry.yahtzeeBonus;
423 entries.push_back(entry);
424 redo.clear();
425 if (entries.size() == 13 * players.size()) turn.reset();
426 else turn = Turn{(entry.turn.player + 1) % static_cast<int>(players.size())};
427 emit(line);
428 emit(nextTurnText());
429 }
430 void botTurn() {
431 if (turn->rolls == 0) roll();
432 Strategy strategy = players[turn->player].strategy;
433 while (turn->rolls < rules.rolls) {
434 Holds holds = botHolds(strategy, turn->dice);
435 if (std::all_of(holds.begin(), holds.end(), [](bool held) { return held; })) break;
436 hold(holds);
437 roll();
438 }
439 Counts counts = faceCounts(turn->dice);
440 Candidates scores = candidates(players[turn->player], counts, rules.joker);
441 Category category = scores.best;
442 int face = mostFrequent(counts);
443 if (strategy == Strategy::Upper && counts[face] >= 3 && scores.allowed[face]) category = static_cast<Category>(face);
444 score(category);
445 }
446 void undoScore() {
447 if (entries.empty()) throw CommandError("nothing to undo");
448 ScoreEntry entry = entries.back();
449 entries.pop_back();
450 players[entry.turn.player].boxes[entry.category].reset();
451 players[entry.turn.player].yahtzeeBonus -= entry.yahtzeeBonus;
452 redo.push_back({turn, entry});
453 turn = entry.turn;
454 emit("undo " + players[entry.turn.player].name + " " + categoryNames[entry.category] + " " + std::to_string(entry.points));
455 }
456 void redoScore() {
457 if (redo.empty()) throw CommandError("nothing to redo");
458 RedoEntry saved = redo.back();
459 redo.pop_back();
460 const ScoreEntry& entry = saved.score;
461 players[entry.turn.player].boxes[entry.category] = entry.points;
462 players[entry.turn.player].yahtzeeBonus += entry.yahtzeeBonus;
463 entries.push_back(entry);
464 turn = saved.resume;
465 emit("redo " + players[entry.turn.player].name + " " + categoryNames[entry.category] + " " + std::to_string(entry.points));
466 }

491 void odds(Game& table, const Holds& holds) {
492 const Turn& turn = *table.turn;
493 const Player& player = table.players[turn.player];
494 Counts held = faceCounts(turn.dice, holds);
495 int rolled = 5 - std::accumulate(held.begin(), held.end(), 0);
496 Integer outcomes = powersOfSix[rolled];
497 std::array<Integer, CategoryCount> sums{}, positive{};
498 Integer bestSum = 0;
499 for (const Outcome& outcome : rerolls.outcomes[rolled]) {
500 Candidates scores = candidates(player, combined(held, outcome.counts), rules.joker);
501 for (int i = 0; i < CategoryCount; ++i) {
502 sums[i] += outcome.weight * scores.points[i];
503 if (scores.points[i] > 0) positive[i] += outcome.weight;
504 }
505 bestSum += outcome.weight * scores.points[scores.best];
506 }
507 emit("odds", patternText(holds), "outcomes", outcomes);
508 for (int i = 0; i < CategoryCount; ++i) {
509 if (!player.boxes[i].has_value()) emit(categoryNames[i], "expect", fraction(sums[i], outcomes), "chance", fraction(positive[i], outcomes));
510 }
511 emit("best expect", fraction(bestSum, outcomes));
512 }

514 void card(const Game& table) {

517 std::array<std::string, 4>{"UPPER", "BONUS", "YBONUS", "TOTAL"}[row - CategoryCount];

526 value = std::to_string(values[row - CategoryCount]);
527 }
528 line << ' ' << std::right << std::setw(std::max(4, static_cast<int>(player.name.size()))) << value;

550 void simulate(Integer seed, int count, const std::vector<Strategy>& strategies) {
551 std::vector<SimulationTotal> totals(strategies.size());
552 std::vector<std::string> names;
553 for (int seat = 0; seat < static_cast<int>(strategies.size()); ++seat) names.emplace_back(1, static_cast<char>('A' + seat));
554 Integer state = seed;
555 for (int round = 0; round < count; ++round) {
556 Game simulation(state, names, rules);
557 for (std::size_t seat = 0; seat < strategies.size(); ++seat) simulation.players[seat].strategy = strategies[seat];
558 while (simulation.turn.has_value()) simulation.botTurn();
559 state = simulation.state;
560 ++totals[simulation.ranking()[0]].wins;
561 for (std::size_t seat = 0; seat < strategies.size(); ++seat) {
562 const Player& player = simulation.players[seat];
563 SimulationTotal& total = totals[seat];
564 total.sum += player.total();
565 total.best = std::max(total.best, player.total());
566 total.bonus += player.upperBonus() == 35;
567 total.yahtzee += player.boxes[Yahtzee] == 50;
568 }
569 }
570 emit("sim seed", seed, "games", count, "players", strategies.size());
571 for (std::size_t seat = 0; seat < strategies.size(); ++seat) {
572 const SimulationTotal& total = totals[seat];
573 emit("seat", seat + 1, strategyNames[static_cast<int>(strategies[seat])], "wins", total.wins,
574 "average", average(total.sum, count), "best", total.best, "bonus", total.bonus, "yahtzee", total.yahtzee);
575 }
576 }

578 void execute(const std::vector<std::string>& words) {
579 const std::string& command = words[0];
580 if (command == "RULE") {
581 if (words[1] != "ROLLS" && words[1] != "JOKER") throw CommandError("unknown rule " + words[1]);
582 Rules changed = rules;
583 if (words[1] == "ROLLS") changed.rolls = static_cast<int>(integerWord(words[2], 1, 4));
584 else if (words[2] == "FORCED") changed.joker = Joker::Forced;
585 else if (words[2] == "FREE") changed.joker = Joker::Free;
586 else if (words[2] == "OFF") changed.joker = Joker::Off;
587 else throw CommandError("bad value " + words[2]);
588 if (game != nullptr) throw CommandError("game in progress");
589 rules = changed;
590 emit("rule", words[1], words[2]);
591 } else if (command == "GAME") {
592 Integer seed = integerWord(words[1], 1, 2147483646);
593 std::vector<std::string> names(words.begin() + 2, words.end());
594 for (const std::string& name : names) playerName(name);
595 if (names.size() > 6) throw CommandError("too many players");
596 if (game != nullptr) throw CommandError("game in progress");
597 std::set<std::string> seen;
598 for (const std::string& name : names) if (!seen.insert(name).second) throw CommandError("duplicate name " + name);
599 game = std::make_unique<Game>(seed, names, rules, &output);
600 std::string line = "game seed " + std::to_string(seed) + " players";
601 for (const std::string& name : names) line += " " + name;
602 emit(line);
603 emit(game->nextTurnText());
604 } else if (command == "BOT") {
605 playerName(words[1]);
606 Strategy strategy = strategyWord(words[2]);
607 Game& table = requireGame();
608 auto player = std::find_if(table.players.begin(), table.players.end(), [&](const Player& value) { return value.name == words[1]; });
609 if (player == table.players.end()) throw CommandError("no player " + words[1]);
610 player->strategy = strategy;
611 emit("bot", words[1], words[2]);
612 } else if (command == "LOAD" || command == "KEEP") {
613 std::vector<int> faces;
614 for (std::size_t i = 1; i < words.size(); ++i) faces.push_back(static_cast<int>(integerWord(words[i], 1, 6)));
615 Game& table = requireGame();
616 if (command == "LOAD") {
617 table.loaded.insert(table.loaded.end(), faces.begin(), faces.end());
618 emit("loaded", table.loaded.size());
619 } else {
620 table.rollsLeftCheck();
621 Holds holds{};
622 for (int face : faces) {
623 int die = 0;
624 while (die < 5 && (holds[die] || table.turn->dice[die] != face)) ++die;
625 if (die == 5) throw CommandError("not showing " + std::to_string(face));
626 holds[die] = true;
627 }
628 table.hold(holds);
629 }
630 } else if (command == "ROLL") {
631 Game& table = requireGame();
632 table.turnCheck();
633 if (table.turn->rolls == rules.rolls) throw CommandError("no rolls left");
634 if (std::all_of(table.turn->holds.begin(), table.turn->holds.end(), [](bool held) { return held; })) throw CommandError("all dice held");
635 table.roll();
636 } else if (command == "HOLD" || command == "ODDS") {
637 Holds holds = holdPattern(words[1]);
638 Game& table = requireGame();
639 table.rollsLeftCheck();
640 if (command == "HOLD") table.hold(holds);
641 else odds(table, holds);
642 } else if (command == "SCORE" || command == "WHATIF") {
643 Category category = categoryWord(words[1]);
644 Game& table = requireGame();
645 ScoreEntry entry = table.checkedScore(category);
646 if (command == "SCORE") table.score(category);
647 else emit(table.scoreText(entry, true));
648 } else if (command == "UNDO") {
649 requireGame().undoScore();
650 } else if (command == "REDO") {
651 requireGame().redoScore();
652 } else if (command == "SUGGEST") {
653 Game& table = requireGame();
654 table.rolledCheck();
655 const Player& player = table.players[table.turn->player];
656 Candidates scores = candidates(player, faceCounts(table.turn->dice), rules.joker);
657 std::string line = "suggest";
658 for (int i = 0; i < CategoryCount; ++i) {
659 if (!player.boxes[i].has_value()) line += " " + categoryNames[i] + " " + std::to_string(scores.points[i]);
660 }
661 emit(line, "best", categoryNames[scores.best], scores.points[scores.best]);
662 } else if (command == "ADVISE") {
663 Game& table = requireGame();
664 table.rollsLeftCheck();
665 const Turn& turn = *table.turn;
666 Advice best = advise(table.players[turn.player], turn.dice, rules.rolls - turn.rolls, rules.joker, rerolls);
667 emit("advise", patternText(best.holds) + ":", facesText(turn.dice, best.holds), "expect", fraction(best.numerator, best.denominator));
668 } else if (command == "AUTO") {
669 int count = static_cast<int>(integerWord(words[1], 1, 100));
670 Game& table = requireGame();
671 table.turnCheck();
672 table.redo.clear();
673 for (int i = 0; i < count && table.turn.has_value(); ++i) table.botTurn();
674 } else if (command == "STATUS") {
675 const Game& table = requireGame();
676 if (!table.turn.has_value()) emit("status game over");
677 else {
678 const Turn& turn = *table.turn;
679 const Player& player = table.players[turn.player];
680 emit("status round", player.round(), "player", player.name, "rolls", turn.rolls,
681 "dice", turn.rolls == 0 ? "-" : facesText(turn.dice), "holds", patternText(turn.holds));
682 }
683 } else if (command == "CARD") {
684 card(requireGame());
685 } else if (command == "BONUS") {
686 const Game& table = requireGame();
687 for (const Player& player : table.players) {
688 int upper = player.upper(), maximum = upper;
689 for (int i = Ones; i <= Sixes; ++i) if (!player.boxes[i].has_value()) maximum += 5 * (i + 1);
690 emit("bonus", player.name, "upper", upper, "need", std::max(63 - upper, 0), "max", maximum,
691 upper >= 63 ? "made" : maximum < 63 ? "lost" : "open");
692 }
693 } else if (command == "STATS") {
694 statistics(requireGame());
695 } else if (command == "LOG") {
696 const Game& table = requireGame();
697 if (table.entries.empty()) emit("no scores");
698 for (std::size_t i = 0; i < table.entries.size(); ++i) {
699 const ScoreEntry& entry = table.entries[i];
700 std::string suffix;
701 if (entry.joker) suffix += " joker";
702 if (entry.yahtzeeBonus != 0) suffix += " ybonus 100";
703 emit("log", i + 1, table.players[entry.turn.player].name, facesText(entry.turn.dice),
704 categoryNames[entry.category], std::to_string(entry.points) + suffix);
705 }
706 } else if (command == "FACES") {
707 const Game& table = requireGame();
708 Integer total = 0, squares = 0;
709 std::string line = "faces";
710 for (Integer count : table.faceTotals) { total += count; squares += count * count; line += " " + std::to_string(count); }
711 emit(line, "total", total, "chi2", total == 0 ? "0" : fraction(6 * squares - total * total, total));
712 } else if (command == "STANDINGS" || command == "END") {
713 const Game& table = requireGame();
714 std::vector<int> ranking = table.ranking();
715 bool finished = !table.turn.has_value();
716 if (command == "END") emit("end", finished ? "final" : "abandoned");
717 int best = table.players[ranking[0]].total();
718 for (std::size_t rank = 0; rank < ranking.size(); ++rank) {
719 const Player& player = table.players[ranking[rank]];
720 if (command == "STANDINGS") emit("standing", rank + 1, player.name, player.total(), "behind", best - player.total());
721 else {
722 emit("rank", rank + 1, player.name, player.total());
723 if (finished) {
724 Record& record = records[player.name];
725 ++record.games;
726 record.wins += rank == 0;
727 record.sum += player.total();
728 record.best = std::max(record.best, player.total());
729 }
730 }
731 }
732 if (command == "END") game.reset();
733 } else if (command == "RECORDS") {
734 std::vector<std::string> names;
735 for (const auto& [name, record] : records) names.push_back(name);
736 std::sort(names.begin(), names.end(), [&](const std::string& a, const std::string& b) {
737 const Record& first = records.at(a);
738 const Record& second = records.at(b);
739 if (first.wins != second.wins) return first.wins > second.wins;
740 if (first.best != second.best) return first.best > second.best;
741 return a < b;
742 });
743 if (names.empty()) emit("no records");
744 for (const std::string& name : names) {
745 const Record& record = records.at(name);
746 emit("record", name, "games", record.games, "wins", record.wins, "best", record.best, "average", average(record.sum, record.games));
747 }
748 } else if (command == "SIM") {
749 Integer seed = integerWord(words[1], 1, 2147483646);
750 int count = static_cast<int>(integerWord(words[2], 1, 50));
751 std::vector<Strategy> strategies;
752 for (std::size_t i = 3; i < words.size(); ++i) strategies.push_back(strategyWord(words[i]));
753 if (strategies.size() > 6) throw CommandError("too many players");
754 simulate(seed, count, strategies);
755 } else {
756 throw CommandError("unknown command " + command);
757 }
758 }
759
760public:
761 void process(const std::string& line) {
762 std::istringstream input(line);
763 std::vector<std::string> words;
764 for (std::string word; input >> word;) words.push_back(word);
765 try { execute(words); }
766 catch (const CommandError& error) { emit("error:", error.what()); }
767 }
768 void print() const { for (const std::string& line : output) std::cout << line << '\n'; }
769};

772 std::ios::sync_with_stdio(false);
773 std::cin.tie(nullptr);

775 std::cin >> count;
776 std::string line;
777 std::getline(std::cin, line);
778 Program program;
779 for (int i = 0; i < count; ++i) { std::getline(std::cin, line); program.process(line); }
780 program.print();
781}
