112int normalScore(int category, List<int> counts) {
113 if (category < 6) return (category + 1) * counts[category];
114 var total = 0, largest = 0, bits = 0;
115 for (var i = 0; i < 6; i++) {
116 total += (i + 1) * counts[i];
117 largest = max(largest, counts[i]);
118 if (counts[i] > 0) bits |= 1 << i;
119 }
120 return switch (category) {
121 6 => largest >= 3 ? total : 0,
122 7 => largest >= 4 ? total : 0,
123 8 => counts.contains(3) && counts.contains(2) ? 25 : 0,
124 9 => (bits & 15) == 15 || (bits & 30) == 30 || (bits & 60) == 60 ? 30 : 0,
125 10 => bits == 31 || bits == 62 ? 40 : 0,
126 11 => largest == 5 ? 50 : 0,
127 _ => total,
128 };
129}

131class Player {
132 final String name;
133 Strategy strategy;
134 final scores = List<int?>.filled(13, null);
135 int yahtzeeBonus = 0;
136 Player(this.name, [this.strategy = Strategy.kind]);
137 int get upper => sum(scores.take(6).map((value) => value ?? 0));
138 int get upperBonus => upper >= 63 ? 35 : 0;
139 int get total => sum(scores.map((value) => value ?? 0)) + upperBonus + yahtzeeBonus;
140 int get round => scores.where((value) => value != null).length + 1;
141}

143typedef Candidates = ({List<int> points, int allowed, bool joker, int best});
144
145Candidates evaluate(Player player, List<int> counts, JokerRule rule) {
146 var allowed = 0;
147 for (var c = 0; c < 13; c++) {
148 if (player.scores[c] == null) allowed |= 1 << c;
149 }
150 final joker = rule != JokerRule.off && counts.contains(5) && player.scores[11] != null;
151 if (joker && rule == JokerRule.forced) {
152 final upper = counts.indexOf(5);
153 if ((allowed & (1 << upper)) != 0) {
154 allowed = 1 << upper;
155 } else if ((allowed & lowerMask) != 0) {
156 allowed &= lowerMask;
157 }
158 }
159 final points = List.filled(13, -1);
160 var best = -1;
161 for (var c = 0; c < 13; c++) {
162 if (player.scores[c] != null) continue;
163 points[c] = (allowed & (1 << c)) == 0 ? 0
164 : joker && c >= 8 && c <= 10 ? const [25, 30, 40][c - 8] : normalScore(c, counts);
165 if (best < 0 || points[c] > points[best]) best = c;
166 }
167 return (points: points, allowed: allowed, joker: joker, best: best);
168}

170typedef Outcome = ({int code, int weight});
171
172List<List<Outcome>> makeOutcomes() {
173 const factorial = [1, 1, 2, 6, 24, 120];
174 final result = List.generate(6, (_) => <Outcome>[]);
175 for (var size = 0; size <= 5; size++) {
176 void add(int face, int remaining, int code, int divisor) {
177 if (face == 6) {
178 if (remaining == 0) result[size].add((code: code, weight: factorial[size] ~/ divisor));
179 return;
180 }
181 for (var n = 0; n <= remaining; n++) {
182 add(face + 1, remaining - n, code + n * powers[face], divisor * factorial[n]);
183 }
184 }
185 add(0, size, 0, 1);
186 }
187 return result;
188}

228 void checkRolled() {
229 checkTurn();

233 void checkRollsLeft() {
234 checkRolled();
235 if (state.rolls == maxRolls) throw CommandError('no rolls left');

238 void roll() {
239 checkTurn();
240 if (state.rolls == maxRolls) throw CommandError('no rolls left');
241 if (state.hold == 31) throw CommandError('all dice held');
242 redoList.clear();
243 final faces = state.faces == null ? List.filled(5, 0) : List<int>.of(state.faces!);
244 for (var i = 0; i < 5; i++) {
245 if (state.rolls > 0 && isHeld(state.hold, i)) continue;
246 if (loaded.isNotEmpty) {
247 faces[i] = loaded.removeFirst();
248 } else {
249 randomState = 48271 * randomState % 2147483647;
250 faces[i] = randomState % 6 + 1;
251 }
252 faceTotals[faces[i] - 1]++;
253 }
254 state = Turn(seat: state.seat, rolls: state.rolls + 1, faces: faces, hold: state.hold);
255 emit('${current.name} roll ${state.rolls}: ${facesText(faces)}');
256 }

258 void hold(int mask) {
259 checkRollsLeft();
260 redoList.clear();
261 state = Turn(seat: state.seat, rolls: state.rolls, faces: state.faces, hold: mask);
262 emit('${current.name} holds ${patternText(mask)}: ${facesText(state.faces, mask)}');
263 }
264
265 void keep(List<int> values) {
266 checkRollsLeft();
267 var mask = 0;
268 for (final value in values) {
269 var die = -1;
270 for (var i = 0; i < 5; i++) {
271 if (!isHeld(mask, i) && state.faces![i] == value) { die = i; break; }
272 }
273 if (die < 0) throw CommandError('not showing $value');
274 mask |= 1 << (4 - die);
275 }
276 hold(mask);
277 }

279 void score(int category, {bool preview = false}) {
280 checkRolled();
281 final player = current;
282 if (player.scores[category] != null) throw CommandError('${categories[category]} already used');
283 final counts = faceCounts(state.faces!);
284 final candidates = evaluate(player, counts, joker);
285 if ((candidates.allowed & (1 << category)) == 0) {
286 final upper = counts.indexOf(5);
287 throw CommandError('joker requires ${player.scores[upper] == null ? categories[upper] : 'lower box'}');
288 }
289 final points = candidates.points[category];
290 final bonus = counts.contains(5) && player.scores[11] == 50 ? 100 : 0;
291 final upperBonus = category < 6 && player.upper < 63 && player.upper + points >= 63 ? 35 : 0;
292 var line = '${player.name} ${preview ? 'would score' : 'scores'} $points in ${categories[category]}';
293 if (candidates.joker) line += ' joker';
294 if (bonus != 0) line += ' ybonus 100';
295 if (upperBonus != 0) line += ' ubonus 35';
296 line += ' total ${player.total + points + bonus + upperBonus}';
297 if (preview) { emit(line); return; }
298 redoList.clear();
299 entries.add(ScoreEntry(state, category, points, bonus, candidates.joker));
300 player.scores[category] = points;
301 player.yahtzeeBonus += bonus;
302 state = entries.length == 13 * players.length ? Turn(finished: true) : Turn(seat: (state.seat + 1) % players.length);
303 emit(line);
304 emit(state.finished ? 'game over' : 'turn ${current.round} ${current.name}');
305 }

307 void undo() {
308 if (entries.isEmpty) throw CommandError('nothing to undo');
309 final entry = entries.removeLast();
310 redoList.add((after: state, entry: entry));
311 final player = players[entry.before.seat];
312 player.scores[entry.category] = null;
313 player.yahtzeeBonus -= entry.bonus;
314 state = entry.before;
315 emit('undo ${player.name} ${categories[entry.category]} ${entry.points}');
316 }
317
318 void redo() {
319 if (redoList.isEmpty) throw CommandError('nothing to redo');
320 final saved = redoList.removeLast();
321 final entry = saved.entry;
322 final player = players[entry.before.seat];
323 player.scores[entry.category] = entry.points;
324 player.yahtzeeBonus += entry.bonus;
325 entries.add(entry);
326 state = saved.after;
327 emit('redo ${player.name} ${categories[entry.category]} ${entry.points}');
328 }

330 int botHold(Strategy strategy) {
331 final faces = state.faces!, counts = faceCounts(faces);
332 final frequent = mostFrequent(counts);
333 var mask = 0;
334 if (strategy == Strategy.straight) {
335 var bestStart = 0, bestLength = 0;
336 for (var start = 0; start < 6; start++) {
337 var end = start;
338 while (end < 6 && counts[end] > 0) end++;
339 if (end - start >= bestLength) { bestStart = start; bestLength = end - start; }
340 }
341 if (bestLength >= 3) {
342 for (var face = bestStart + 1; face <= bestStart + bestLength; face++) {
343 mask |= 1 << (4 - faces.indexOf(face));
344 }
345 return mask;
346 }
347 }
348 for (var i = 0; i < 5; i++) {
349 if (strategy == Strategy.high ? faces[i] >= 5 : faces[i] == frequent + 1) mask |= 1 << (4 - i);
350 }
351 return mask;
352 }
353
354 void botTurn() {
355 final strategy = current.strategy;
356 if (state.rolls == 0) roll();
357 while (state.rolls < maxRolls) {
358 final mask = botHold(strategy);
359 if (mask == 31) break;
360 hold(mask);
361 roll();
362 }
363 final counts = faceCounts(state.faces!);
364 final candidates = evaluate(current, counts, joker);
365 final frequent = mostFrequent(counts);
366 final category = strategy == Strategy.upper && counts[frequent] >= 3 && (candidates.allowed & (1 << frequent)) != 0
367 ? frequent : candidates.best;
368 score(category);
369 }

371 List<int> ranking() => List.generate(players.length, (i) => i)
372 ..sort((a, b) {
373 final difference = players[b].total.compareTo(players[a].total);
374 return difference == 0 ? a.compareTo(b) : difference;
375 });

378class Adviser {
379 final Player player;
380 final JokerRule rule;
381 final values = <(int, int), int>{};
382 final expectations = <(int, int), int>{};
383 Adviser(this.player, this.rule);
384
385 int value(int code, int depth) {
386 final key = (depth, code);
387 final cached = values[key];
388 if (cached != null) return cached;
389 final counts = decode(code);
390 var best = -1;
391 if (depth == 0) {
392 final candidates = evaluate(player, counts, rule);
393 best = candidates.points[candidates.best];
394 } else {
395 void search(int face, int held) {
396 if (face == 6) { best = max(best, expected(held, depth)); return; }
397 for (var n = 0; n <= counts[face]; n++) search(face + 1, held + n * powers[face]);
398 }
399 search(0, 0);
400 }
401 values[key] = best;
402 return best;
403 }
404
405 int expected(int held, int depth) {
406 final key = (depth, held);
407 final cached = expectations[key];
408 if (cached != null) return cached;
409 final rerolled = 5 - sum(decode(held));
410 var total = 0;
411 for (final outcome in outcomes[rerolled]) total += outcome.weight * value(held + outcome.code, depth - 1);
412 final result = total * powers[5 - rerolled];
413 expectations[key] = result;
414 return result;
415 }
416}

435 void odds(Game table, int mask) {
436 table.checkRollsLeft();
437 final held = heldCode(table.state.faces!, mask), rerolled = 5 - sum(decode(held));
438 final count = powers[rerolled];
439 final totals = List.filled(13, 0), chances = List.filled(13, 0);
440 var best = 0;
441 for (final outcome in outcomes[rerolled]) {
442 final candidates = evaluate(table.current, decode(held + outcome.code), table.joker);
443 for (var c = 0; c < 13; c++) {
444 if (candidates.points[c] < 0) continue;
445 totals[c] += outcome.weight * candidates.points[c];
446 if (candidates.points[c] > 0) chances[c] += outcome.weight;
447 }
448 best += outcome.weight * candidates.points[candidates.best];
449 }
450 emit('odds ${patternText(mask)} outcomes $count');
451 for (var c = 0; c < 13; c++) {
452 if (table.current.scores[c] == null) emit('${categories[c]} expect ${fraction(totals[c], count)} chance ${fraction(chances[c], count)}');
453 }
454 emit('best expect ${fraction(best, count)}');
455 }

457 void advise(Game table) {
458 table.checkRollsLeft();
459 final depth = table.maxRolls - table.state.rolls;
460 var chosen = 0, best = -1, denominator = 1;
461 for (var i = 0; i < depth; i++) denominator *= 7776;
462 final adviser = Adviser(table.current, table.joker);
463 for (var mask = 0; mask < 32; mask++) {
464 final value = adviser.expected(heldCode(table.state.faces!, mask), depth);
465 if (value > best) { best = value; chosen = mask; }
466 }
467 emit('advise ${patternText(chosen)}: ${facesText(table.state.faces, chosen)} expect ${fraction(best, denominator)}');
468 }

484 void simulate(int seed, int count, List<Strategy> strategies) {
485 final totals = List.generate(strategies.length, (_) => SimulationTotal());
486 var state = seed;
487 for (var n = 0; n < count; n++) {
488 final players = List.generate(strategies.length, (i) => Player('BOT${i + 1}', strategies[i]));
489 final table = Game(state, players, rolls, joker);
490 while (!table.state.finished) table.botTurn();
491 state = table.randomState;
492 totals[table.ranking().first].wins++;
493 for (var i = 0; i < players.length; i++) {
494 final player = players[i];
495 final total = totals[i];
496 total.sum += player.total;
497 total.best = max(total.best, player.total);
498 if (player.upperBonus == 35) total.bonus++;
499 if (player.scores[11] == 50) total.yahtzee++;
500 }
501 }
502 emit('sim seed $seed games $count players ${strategies.length}');
503 for (var i = 0; i < strategies.length; i++) {
504 final total = totals[i];
505 emit('seat ${i + 1} ${strategies[i].label} wins ${total.wins} average ${average(total.sum, count)} best ${total.best} bonus ${total.bonus} yahtzee ${total.yahtzee}');
506 }

509 void execute(List<String> words) {
510 switch (words[0]) {
511 case 'RULE':
512 if (words[1] != 'ROLLS' && words[1] != 'JOKER') throw CommandError('unknown rule ${words[1]}');
513 var newRolls = rolls;
514 var newJoker = joker;
515 if (words[1] == 'ROLLS') {
516 newRolls = positive(words[2], 4);
517 } else {
518 newJoker = switch (words[2]) {
519 'FORCED' => JokerRule.forced, 'FREE' => JokerRule.free, 'OFF' => JokerRule.off,
520 _ => throw CommandError('bad value ${words[2]}'),
521 };
522 }
523 if (game != null) throw CommandError('game in progress');
524 rolls = newRolls; joker = newJoker;
525 emit('rule ${words[1]} ${words[2]}');
526 return;
527 case 'GAME':
528 final seed = positive(words[1], 2147483646);
529 final names = words.sublist(2);
530 for (final name in names) checkName(name);
531 if (names.length > 6) throw CommandError('too many players');
532 if (game != null) throw CommandError('game in progress');
533 final seen = <String>{};
534 for (final name in names) {
535 if (!seen.add(name)) throw CommandError('duplicate name $name');
536 }
537 game = Game(seed, names.map((name) => Player(name)).toList(), rolls, joker, emit);
538 emit('game seed $seed players ${names.join(' ')}');
539 emit('turn 1 ${names.first}');
540 return;
541 case 'BOT':
542 checkName(words[1]);
543 final strategy = parseStrategy(words[2]);
544 final table = this.table();
545 final seat = table.players.indexWhere((player) => player.name == words[1]);
546 if (seat < 0) throw CommandError('no player ${words[1]}');
547 table.players[seat].strategy = strategy;
548 emit('bot ${words[1]} ${words[2]}');
549 return;
550 case 'LOAD':
551 case 'KEEP':
552 final values = words.skip(1).map((word) => positive(word, 6)).toList();
553 final table = this.table();
554 if (words[0] == 'KEEP') {
555 table.keep(values);
556 } else {
557 table.loaded.addAll(values);
558 emit('loaded ${table.loaded.length}');
559 }
560 return;
561 case 'ROLL': table().roll(); return;
562 case 'HOLD':
563 case 'ODDS':
564 final mask = parsePattern(words[1]);
565 final table = this.table();
566 if (words[0] == 'HOLD') { table.hold(mask); } else { odds(table, mask); }
567 return;
568 case 'SCORE':
569 case 'WHATIF':
570 final category = parseCategory(words[1]);
571 table().score(category, preview: words[0] == 'WHATIF');
572 return;
573 case 'UNDO': table().undo(); return;
574 case 'REDO': table().redo(); return;
575 case 'SUGGEST':
576 final table = this.table();
577 table.checkRolled();
578 final candidates = evaluate(table.current, faceCounts(table.state.faces!), table.joker);
579 var line = 'suggest';
580 for (var c = 0; c < 13; c++) {
581 if (candidates.points[c] >= 0) line += ' ${categories[c]} ${candidates.points[c]}';
582 }
583 emit('$line best ${categories[candidates.best]} ${candidates.points[candidates.best]}');
584 return;
585 case 'ADVISE': advise(table()); return;
586 case 'AUTO':
587 final count = positive(words[1], 100);
588 final table = this.table();
589 table.checkTurn();
590 table.redoList.clear();
591 for (var i = 0; i < count && !table.state.finished; i++) table.botTurn();
592 return;
593 case 'STATUS':
594 final table = this.table();
595 final turn = table.state;
596 emit(turn.finished ? 'status game over'
597 : 'status round ${table.current.round} player ${table.current.name} rolls ${turn.rolls} dice ${facesText(turn.faces)} holds ${patternText(turn.hold)}');
598 return;
599 case 'CARD': card(table()); return;
600 case 'BONUS':
601 for (final player in table().players) {
602 var maximum = player.upper;
603 for (var c = 0; c < 6; c++) {
604 if (player.scores[c] == null) maximum += 5 * (c + 1);
605 }
606 final status = player.upper >= 63 ? 'made' : maximum < 63 ? 'lost' : 'open';
607 emit('bonus ${player.name} upper ${player.upper} need ${max(63 - player.upper, 0)} max $maximum $status');
608 }
609 return;
610 case 'STATS':
611 final table = this.table();
612 for (var i = 0; i < table.players.length; i++) {
613 final entries = table.entries.where((entry) => entry.before.seat == i).toList();
614 emit('stats ${table.players[i].name} turns ${entries.length} rolls ${sum(entries.map((e) => e.before.rolls))} zeros ${entries.where((e) => e.points == 0).length} yahtzees ${entries.where((e) => faceCounts(e.before.faces!).contains(5)).length} average ${average(sum(entries.map((e) => e.points)), entries.length)}');
615 }
616 return;
617 case 'LOG':
618 final table = this.table();
619 for (var i = 0; i < table.entries.length; i++) {
620 final entry = table.entries[i];
621 var line = 'log ${i + 1} ${table.players[entry.before.seat].name} ${facesText(entry.before.faces)} ${categories[entry.category]} ${entry.points}';
622 if (entry.joker) line += ' joker';
623 if (entry.bonus != 0) line += ' ybonus 100';
624 emit(line);
625 }
626 if (table.entries.isEmpty) emit('no scores');
627 return;
628 case 'FACES':
629 final counts = table().faceTotals;
630 final total = sum(counts);
631 final numerator = 6 * sum(counts.map((c) => c * c)) - total * total;
632 emit('faces ${counts.join(' ')} total $total chi2 ${fraction(numerator, total == 0 ? 1 : total)}');
633 return;
634 case 'STANDINGS':
635 case 'END':
636 final table = this.table();
637 final ranking = table.ranking();
638 final end = words[0] == 'END';
639 if (end) emit(table.state.finished ? 'end final' : 'end abandoned');
640 for (var rank = 0; rank < ranking.length; rank++) {
641 final player = table.players[ranking[rank]];
642 if (!end) {
643 emit('standing ${rank + 1} ${player.name} ${player.total} behind ${table.players[ranking.first].total - player.total}');
644 } else {
645 emit('rank ${rank + 1} ${player.name} ${player.total}');
646 if (table.state.finished) {
647 final record = records.putIfAbsent(player.name, GameRecord.new);
648 record.games++; record.sum += player.total; record.best = max(record.best, player.total);
649 if (rank == 0) record.wins++;
650 }
651 }
652 }
653 if (end) game = null;
654 return;
655 case 'RECORDS':
656 final names = records.keys.toList()..sort((a, b) {
657 final first = records[a]!, second = records[b]!;
658 if (first.wins != second.wins) return second.wins.compareTo(first.wins);
659 if (first.best != second.best) return second.best.compareTo(first.best);
660 return a.compareTo(b);
661 });
662 for (final name in names) {
663 final record = records[name]!;
664 emit('record $name games ${record.games} wins ${record.wins} best ${record.best} average ${average(record.sum, record.games)}');
665 }
666 if (names.isEmpty) emit('no records');
667 return;
668 case 'SIM':
669 final seed = positive(words[1], 2147483646), count = positive(words[2], 50);
670 final strategies = words.skip(3).map(parseStrategy).toList();
671 if (strategies.length > 6) throw CommandError('too many players');
672 simulate(seed, count, strategies);
673 return;
674 default: throw CommandError('unknown command ${words[0]}');
675 }
676 }
677}

683 final words = stdin.readLineSync()!.split(' ');
