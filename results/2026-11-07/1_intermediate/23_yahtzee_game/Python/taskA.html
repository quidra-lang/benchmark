87def fraction_text(numerator, denominator):
88 divisor = gcd(numerator, denominator)
89 numerator //= divisor
90 denominator //= divisor
91 return str(numerator) if denominator == 1 else f"{numerator}/{denominator}"

107def normal_score(category, faces):
108 if category in UPPER_CATEGORIES:
109 face = UPPER_CATEGORIES.index(category) + 1
110 return face * faces.count(face)
111 counts = [faces.count(face) for face in FACE_VALUES]
112 if category == "THREEKIND":
113 return sum(faces) if max(counts) >= 3 else 0
114 if category == "FOURKIND":
115 return sum(faces) if max(counts) >= 4 else 0
116 if category == "FULLHOUSE":
117 return 25 if 3 in counts and 2 in counts else 0
118 if category == "SMALL":
119 return 30 if any(has_run(faces, start, 4) for start in (1, 2, 3)) else 0
120 if category == "LARGE":
121 return 40 if any(has_run(faces, start, 5) for start in (1, 2)) else 0
122 if category == "YAHTZEE":
123 return YAHTZEE_SCORE if max(counts) == DICE else 0
124 return sum(faces)

127def best_candidate(candidates):
128 return max(candidates, key=lambda candidate: candidate[1])

131def most_frequent_face(faces):
132 return max(FACE_VALUES, key=lambda face: (faces.count(face), face))

135def longest_run(faces):
136 best_start, best_length = 0, 0
137 for start in FACE_VALUES:
138 length = 0
139 while start + length in faces:
140 length += 1
141 if length > 0 and length >= best_length:
142 best_start, best_length = start, length
143 return best_start, best_length

146def bot_holds(strategy, faces):
147 if strategy == "HIGH":
148 return [face >= 5 for face in faces]
149 if strategy == "STRAIGHT":
150 start, length = longest_run(faces)
151 if length >= 3:
152 holds = [False] * DICE
153 for value in range(start, start + length):
154 holds[faces.index(value)] = True
155 return holds
156 target = most_frequent_face(faces)
157 return [face == target for face in faces]

160def reroll_outcomes(count):
161 outcomes = []
162 for rolled in combinations_with_replacement(FACE_VALUES, count):
163 weight = factorial(count)
164 for face in FACE_VALUES:
165 weight //= factorial(rolled.count(face))
166 outcomes.append((rolled, weight))
167 return outcomes

179class DiceRoller:
180 def __init__(self, seed):
181 self.state = seed
182 self.loaded = deque()
183 self.counts = [0] * len(FACE_VALUES)

185 def roll_die(self):
186 if self.loaded:
187 face = self.loaded.popleft()
188 else:
189 self.state = MULTIPLIER * self.state % MODULUS
190 face = self.state % len(FACE_VALUES) + 1
191 self.counts[face - 1] += 1
192 return face

261 def roll(self):
262 turn = self.turn
263 if turn.rolls == 0:
264 turn.faces = [self.roller.roll_die() for _ in range(DICE)]

279 def joker_situation(self, player, faces):

285 def allowed_categories(self, player, faces, joker):
286 open_categories = player.open_categories()
287 if not joker or self.rules.joker == "FREE":
288 return open_categories

290 if player.boxes[upper] is None:
291 return [upper]
292 lower = [category for category in open_categories if category in LOWER_CATEGORIES]

295 def points_to_enter(self, faces, category, joker):
296 if joker and category in JOKER_POINTS:
297 return JOKER_POINTS[category]

300 def candidates(self, player, faces):
301 joker = self.joker_situation(player, faces)
302 allowed = self.allowed_categories(player, faces, joker)
303 result = []
304 for category in player.open_categories():
305 score = self.points_to_enter(faces, category, joker) if category in allowed else 0
306 result.append((category, score))

312 def prepare_entry(self, category):
313 player = self.current_player()
314 faces = self.turn.faces
315 if player.boxes[category] is not None:
316 raise CommandError(f"{category} already used")
317 joker = self.joker_situation(player, faces)
318 if category not in self.allowed_categories(player, faces, joker):
319 upper = UPPER_CATEGORIES[faces[0] - 1]
320 if player.boxes[upper] is None:
321 raise CommandError(f"joker requires {upper}")
322 raise CommandError("joker requires lower box")
323 points = self.points_to_enter(faces, category, joker)
324 bonus = 0
325 if is_yahtzee(faces) and player.boxes["YAHTZEE"] == YAHTZEE_SCORE:
326 bonus = YAHTZEE_BONUS
327 return ScoreEntry(player, category, points, bonus, joker, self.turn)

329 def score_line(self, entry, verb):
330 player = entry.player
331 parts = [player.name, verb, entry.points, "in", entry.category]
332 if entry.joker:
333 parts.append("joker")
334 if entry.bonus:
335 parts.extend(("ybonus", entry.bonus))
336 subtotal = player.upper_subtotal()
337 if entry.category in UPPER_CATEGORIES:
338 subtotal += entry.points
339 gained = 0
340 if player.upper_bonus() == 0 and subtotal >= UPPER_TARGET:
341 gained = UPPER_BONUS
342 parts.extend(("ubonus", gained))
343 parts.extend(("total", player.total() + entry.points + entry.bonus + gained))
344 return " ".join(str(part) for part in parts)

346 def enter(self, entry):
347 entry.player.boxes[entry.category] = entry.points
348 entry.player.yahtzee_bonus += entry.bonus
349 self.entries.append(entry)
350
351 def score(self, category):
352 entry = self.prepare_entry(category)
353 line = self.score_line(entry, "scores")
354 self.enter(entry)
355 self.redo_list.clear()
356 if all(player.used_count() == len(CATEGORIES) for player in self.players):
357 self.turn = None
358 return [line, "game over"]
359 self.turn = Turn((self.turn.seat + 1) % len(self.players))
360 player = self.current_player()
361 return [line, f"turn {player.round()} {player.name}"]

363 def undo(self):
364 entry = self.entries.pop()
365 entry.player.boxes[entry.category] = None
366 entry.player.yahtzee_bonus -= entry.bonus
367 self.redo_list.append((self.turn, entry))
368 self.turn = entry.turn.copy()
369 return entry

371 def redo(self):
372 turn, entry = self.redo_list.pop()
373 self.enter(entry)
374 self.turn = turn

377 def bot_category(self):
378 player = self.current_player()
379 faces = self.turn.faces
380 if player.strategy == "UPPER":
381 face = most_frequent_face(faces)
382 category = UPPER_CATEGORIES[face - 1]
383 allowed = self.allowed_categories(player, faces, self.joker_situation(player, faces))
384 if faces.count(face) >= 3 and category in allowed:
385 return category
386 return self.best(player, faces)[0]

388 def bot_turn(self):
389 lines = []
390 if self.turn.rolls == 0:
391 lines.append(self.roll())
392 while self.turn.rolls < self.rules.rolls:
393 holds = bot_holds(self.current_player().strategy, self.turn.faces)
394 if all(holds):
395 break
396 lines.append(self.set_holds(holds))
397 lines.append(self.roll())
398 lines.extend(self.score(self.bot_category()))
399 return lines

402class Advisor:
403 def __init__(self, game, player):
404 self.game = game
405 self.player = player
406 self.values = {}
407 self.expectations = {}
408
409 def value(self, hand, level):
410 key = (hand, level)
411 if key not in self.values:
412 if level == 0:
413 result = self.game.best(self.player, list(hand))[1]
414 else:
415 result = max(
416 self.expectation(tuple(held_faces(hand, holds)), level)
417 for holds in HOLD_PATTERNS
418 )
419 self.values[key] = result
420 return self.values[key]
421
422 def expectation(self, held, level):
423 key = (held, level)
424 if key not in self.expectations:
425 total = 0
426 for rolled, weight in REROLLS[DICE - len(held)]:
427 total += weight * self.value(tuple(sorted(held + rolled)), level - 1)
428 self.expectations[key] = total * len(FACE_VALUES) ** len(held)
429 return self.expectations[key]
430
431 def best_hold(self, faces, levels):
432 best_holds, best_value = None, -1
433 for holds in HOLD_PATTERNS:
434 value = self.expectation(tuple(sorted(held_faces(faces, holds))), levels)
435 if value > best_value:
436 best_holds, best_value = holds, value
437 return best_holds, best_value

647 def odds(self, arguments):
648 holds = hold_pattern(arguments[0])
649 game = self.require_rolls_left()
650 player = game.current_player()
651 held = held_faces(game.turn.faces, holds)
652 open_categories = player.open_categories()
653 expected = dict.fromkeys(open_categories, 0)
654 chances = dict.fromkeys(open_categories, 0)
655 best_total = 0
656 for rolled, weight in REROLLS[DICE - len(held)]:
657 candidates = game.candidates(player, held + list(rolled))
658 for category, score in candidates:
659 expected[category] += weight * score
660 if score > 0:
661 chances[category] += weight
662 best_total += weight * best_candidate(candidates)[1]
663 outcomes = len(FACE_VALUES) ** (DICE - len(held))
664 self.emit(f"odds {pattern_text(holds)} outcomes {outcomes}")
665 for category in open_categories:
666 self.emit(
667 f"{category} expect {fraction_text(expected[category], outcomes)}"
668 f" chance {fraction_text(chances[category], outcomes)}"
669 )
670 self.emit(f"best expect {fraction_text(best_total, outcomes)}")

713 rows.append(("BONUS", [str(player.upper_bonus()) for player in players]))
714 rows.append(("YBONUS", [str(player.yahtzee_bonus) for player in players]))
715 rows.append(("TOTAL", [str(player.total()) for player in players]))
716 for heading, cells in rows:
717 line = f"{heading:<9}"
718 for cell, width in zip(cells, widths):
719 line += f" {cell:>{width}}"
720 self.emit(line)

722 def bonus(self, arguments):
723 game = self.require_game()
724 for player in game.players:
725 subtotal = player.upper_subtotal()
726 need = max(UPPER_TARGET - subtotal, 0)
727 maximum = subtotal
728 for face, category in zip(FACE_VALUES, UPPER_CATEGORIES):
729 if player.boxes[category] is None:
730 maximum += DICE * face
731 if subtotal >= UPPER_TARGET:
732 state = "made"
733 elif maximum < UPPER_TARGET:
734 state = "lost"
735 else:
736 state = "open"
737 self.emit(f"bonus {player.name} upper {subtotal} need {need} max {maximum} {state}")

808 def simulate(self, arguments):
809 seed = integer_word(arguments[0], 1, MODULUS - 1)
810 count = integer_word(arguments[1], 1, MAX_SIM_GAMES)
811 strategies = [strategy_name(text) for text in arguments[2:]]
812 if len(strategies) > MAX_PLAYERS:
813 raise CommandError("too many players")
814 roller = DiceRoller(seed)
815 summaries = [SeatSummary(strategy) for strategy in strategies]
816 for _ in range(count):
817 players = [Player(str(seat), strategy) for seat, strategy in enumerate(strategies, 1)]
818 game = Game(players, self.rules, roller)
819 while not game.finished():
820 game.bot_turn()
821 winner = game.ranking()[0]
822 for player, summary in zip(players, summaries):
823 summary.add(player, player is winner)
824 self.emit(f"sim seed {seed} games {count} players {len(strategies)}")
825 for seat, summary in enumerate(summaries, 1):
826 self.emit(
827 f"seat {seat} {summary.strategy} wins {summary.wins}"
828 f" average {average_text(summary.total, count)} best {summary.best}"
829 f" bonus {summary.bonus} yahtzee {summary.yahtzee}"
830 )

832 def execute(self, line):
833 command, *arguments = line.split()
834 try:
835 if command not in self.handlers:
836 raise CommandError(f"unknown command {command}")
837 self.handlers[command](arguments)
838 except CommandError as error:
839 self.emit(f"error: {error}")
