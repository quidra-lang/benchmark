15fn equal(a: []const u8, b: []const u8) bool {
16 return std.mem.eql(u8, a, b);
17}
18fn dieBit(die: usize) Mask {
19 return @as(Mask, 1) << @intCast(4 - die);
20}
21fn held(mask: Mask, die: usize) bool {
22 return mask & dieBit(die) != 0;
23}
24fn categoryBit(category: usize) u13 {
25 return @as(u13, 1) << @intCast(category);
26}
27fn countsOf(dice: Dice) Counts {
28 var result: Counts = @splat(0);
29 for (dice) |face| result[face - 1] += 1;
30 return result;
31}
32fn decode(code: usize) Counts {
33 var value = code;
34 var result: Counts = undefined;
35 for (&result) |*count| {
36 count.* = @intCast(value % 6);
37 value /= 6;
38 }
39 return result;
40}
41fn heldCode(dice: Dice, mask: Mask) usize {
42 var result: usize = 0;
43 for (dice, 0..) |face, i| if (held(mask, i)) {
44 result += powers[face - 1];
45 };
46 return result;
47}
48fn mostFrequent(counts: Counts) usize {
49 var best: usize = 0;
50 for (1..6) |i| if (counts[i] >= counts[best]) {
51 best = i;
52 };
53 return best;
54}
55fn yahtzee(counts: Counts) bool {
56 for (counts) |count| if (count == 5) return true;
57 return false;
58}
59fn normalScore(category: Category, counts: Counts) i32 {
60 const c: usize = @backingInt(category);
61 if (c < 6) return @intCast((c + 1) * counts[c]);
62 var total: i32 = 0;
63 var largest: u8 = 0;
64 var bits: u6 = 0;
65 var pair = false;
66 var triple = false;
67 for (counts, 0..) |count, i| {
68 total += @as(i32, @intCast(i + 1)) * count;
69 largest = @max(largest, count);
70 if (count > 0) bits |= @as(u6, 1) << @intCast(i);
71 pair = pair or count == 2;
72 triple = triple or count == 3;
73 }
74 return switch (category) {
75 .THREEKIND => if (largest >= 3) total else 0,
76 .FOURKIND => if (largest >= 4) total else 0,
77 .FULLHOUSE => if (pair and triple) 25 else 0,
78 .SMALL => if (bits & 15 == 15 or bits & 30 == 30 or bits & 60 == 60) 30 else 0,
79 .LARGE => if (bits == 31 or bits == 62) 40 else 0,
80 .YAHTZEE => if (largest == 5) 50 else 0,
81 else => total,
82 };
83}

85const Player = struct {
86 name: []const u8,
87 strategy: Strategy = .KIND,
88 scores: [13]?i32 = @splat(null),
89 yahtzee_bonus: i32 = 0,
90 fn upper(player: Player) i32 {
91 var result: i32 = 0;
92 for (player.scores[0..6]) |score| result += score orelse 0;
93 return result;
94 }
95 fn upperBonus(player: Player) i32 {
96 return if (player.upper() >= 63) 35 else 0;
97 }
98 fn total(player: Player) i32 {
99 var result = player.upperBonus() + player.yahtzee_bonus;
100 for (player.scores) |score| result += score orelse 0;
101 return result;
102 }
103 fn round(player: Player) usize {
104 var result: usize = 1;
105 for (player.scores) |score| if (score != null) {
106 result += 1;
107 };
108 return result;
109 }

112const Candidates = struct {
113 points: [13]i32 = @splat(-1),
114 allowed: u13 = 0,
115 joker: bool = false,
116 best: Category = .ONES,
117};
118fn evaluate(player: *const Player, counts: Counts, rule: Joker) Candidates {
119 var result: Candidates = .{};
120 for (player.scores, 0..) |score, i| if (score == null) {
121 result.allowed |= categoryBit(i);
122 };
123 result.joker = rule != .OFF and yahtzee(counts) and player.scores[11] != null;
124 if (result.joker and rule == .FORCED) {
125 const upper = mostFrequent(counts);
126 if (result.allowed & categoryBit(upper) != 0) result.allowed = categoryBit(upper) else if (result.allowed & lower_mask != 0) result.allowed &= lower_mask;
127 }
128 var best_score: i32 = -1;
129 for (categories, 0..) |category, i| {
130 if (player.scores[i] != null) continue;
131 result.points[i] = if (result.allowed & categoryBit(i) == 0) 0 else if (result.joker) switch (category) {
132 .FULLHOUSE => 25,
133 .SMALL => 30,
134 .LARGE => 40,
135 else => normalScore(category, counts),
136 } else normalScore(category, counts);
137 if (result.points[i] > best_score) {
138 best_score = result.points[i];
139 result.best = category;
140 }
141 }
142 return result;
143}

153const Faces = struct {
154 dice: ?Dice,
155 mask: Mask = 31,
156 pub fn format(value: Faces, out: *std.Io.Writer) std.Io.Writer.Error!void {
157 var first = true;
158 if (value.dice) |dice| for (dice, 0..) |face, i| {
159 if (!held(value.mask, i)) continue;
160 if (!first) try out.writeByte(' ');
161 try out.print("{d}", .{face});
162 first = false;
163 };
164 if (first) try out.writeByte('-');
165 }

167const Fraction = struct {
168 numerator: i64,
169 denominator: i64,
170 pub fn format(value: Fraction, out: *std.Io.Writer) std.Io.Writer.Error!void {
171 var a = value.numerator;
172 var b = value.denominator;
173 while (b != 0) {
174 const remainder = @mod(a, b);
175 a = b;
176 b = remainder;
177 }
178 const numerator = @divExact(value.numerator, a);
179 const denominator = @divExact(value.denominator, a);
180 try out.print("{d}", .{numerator});
181 if (denominator != 1) try out.print("/{d}", .{denominator});
182 }
183};

205const Outcome = struct { code: usize, weight: i64 };
206const Outcomes = struct {
207 items: [6]std.ArrayList(Outcome) = @splat(.empty),
208 fn init(result: *Outcomes, gpa: Allocator) Allocator.Error!void {
209 for (0..6) |size| try result.add(gpa, 0, size, size, 0, 1);
210 }
211 fn add(result: *Outcomes, gpa: Allocator, face: usize, remaining: usize, size: usize, code: usize, divisor: i64) Allocator.Error!void {
212 const factorial = [_]i64{ 1, 1, 2, 6, 24, 120 };
213 if (face == 6) {
214 if (remaining == 0) try result.items[size].append(gpa, .{ .code = code, .weight = @divExact(factorial[size], divisor) });
215 return;
216 }
217 for (0..remaining + 1) |count| try result.add(gpa, face + 1, remaining - count, size, code + count * powers[face], divisor * factorial[count]);
218 }

220const Adviser = struct {
221 gpa: Allocator,
222 player: *const Player,
223 joker: Joker,
224 outcomes: *const Outcomes,
225 values: std.AutoHashMapUnmanaged(usize, i64) = .empty,
226 expectations: std.AutoHashMapUnmanaged(usize, i64) = .empty,
227 fn value(adviser: *Adviser, code: usize, depth: usize) Allocator.Error!i64 {
228 const key = depth * 46656 + code;
229 if (adviser.values.get(key)) |cached| return cached;
230 const counts = decode(code);
231 const best: i64 = if (depth == 0) blk: {
232 const scores = evaluate(adviser.player, counts, adviser.joker);
233 break :blk scores.points[@backingInt(scores.best)];
234 } else try adviser.choose(counts, depth, 0, 0);
235 try adviser.values.put(adviser.gpa, key, best);
236 return best;
237 }
238 fn choose(adviser: *Adviser, counts: Counts, depth: usize, face: usize, kept: usize) Allocator.Error!i64 {
239 if (face == 6) return adviser.expected(kept, depth);
240 var best: i64 = -1;
241 for (0..@as(usize, counts[face]) + 1) |count| best = @max(best, try adviser.choose(counts, depth, face + 1, kept + count * powers[face]));
242 return best;
243 }
244 fn expected(adviser: *Adviser, kept: usize, depth: usize) Allocator.Error!i64 {
245 const key = depth * 46656 + kept;
246 if (adviser.expectations.get(key)) |cached| return cached;
247 var rerolled: usize = 5;
248 for (decode(kept)) |count| rerolled -= count;
249 var sum: i64 = 0;
250 for (adviser.outcomes.items[rerolled].items) |outcome| sum += outcome.weight * (try adviser.value(kept + outcome.code, depth - 1));
251 const result = sum * @as(i64, @intCast(powers[5 - rerolled]));
252 try adviser.expectations.put(adviser.gpa, key, result);
253 return result;
254 }

289 fn checkRolled(game: *Game) Error!void {
290 try game.checkTurn();

293 fn checkRollsLeft(game: *Game) Error!void {
294 try game.checkRolled();
295 if (game.turn.?.rolls == game.rules.rolls) return game.program.fail("no rolls left", .{});
296 }
297 fn roll(game: *Game) Error!void {
298 try game.checkTurn();
299 var turn = game.turn.?;
300 if (turn.rolls == game.rules.rolls) return game.program.fail("no rolls left", .{});
301 if (turn.hold == 31) return game.program.fail("all dice held", .{});
302 game.redo_count = 0;
303 for (&turn.dice, 0..) |*face, i| {
304 if (turn.rolls > 0 and held(turn.hold, i)) continue;
305 if (game.loaded_head < game.loaded.items.len) {
306 face.* = game.loaded.items[game.loaded_head];
307 game.loaded_head += 1;
308 } else {
309 game.random_state = @mod(48271 * game.random_state, 2147483647);
310 face.* = @intCast(@mod(game.random_state, 6) + 1);
311 }
312 game.face_totals[face.* - 1] += 1;
313 }
314 turn.rolls += 1;
315 game.turn = turn;
316 try game.emit("{s} roll {d}: {f}", .{ game.current().name, turn.rolls, faces(turn.dice, 31) });
317 }
318 fn hold(game: *Game, mask: Mask) Error!void {
319 try game.checkRollsLeft();
320 game.redo_count = 0;
321 game.turn.?.hold = mask;
322 try game.emit("{s} holds {f}: {f}", .{ game.current().name, pattern(mask), faces(game.turn.?.dice, mask) });
323 }
324 fn keep(game: *Game, values: []const u8) Error!void {
325 try game.checkRollsLeft();
326 var mask: Mask = 0;
327 for (values) |value| {
328 var selected: ?usize = null;
329 for (game.turn.?.dice, 0..) |face, i| if (!held(mask, i) and face == value) {
330 selected = i;
331 break;
332 };
333 const die = selected orelse return game.program.fail("not showing {d}", .{value});
334 mask |= dieBit(die);
335 }
336 try game.hold(mask);
337 }
338 fn score(game: *Game, category: Category, preview: bool) Error!void {
339 try game.checkRolled();
340 const player = game.current();
341 const c = @backingInt(category);
342 if (player.scores[c] != null) return game.program.fail("{s} already used", .{@tagName(category)});
343 const counts = countsOf(game.turn.?.dice);
344 const scores = evaluate(player, counts, game.rules.joker);
345 if (scores.allowed & categoryBit(c) == 0) {
346 const upper = mostFrequent(counts);
347 return game.program.fail("joker requires {s}", .{if (player.scores[upper] == null) @tagName(categories[upper]) else "lower box"});
348 }
349 const points = scores.points[c];
350 const bonus: i32 = if (yahtzee(counts) and (player.scores[11] orelse 0) == 50) 100 else 0;
351 const upper_bonus: i32 = if (c < 6 and player.upper() < 63 and player.upper() + points >= 63) 35 else 0;
352 const total = player.total() + points + bonus + upper_bonus;
353 const entry: ScoreEntry = .{ .before = game.turn.?, .category = category, .points = points, .bonus = bonus, .joker = scores.joker };
354 if (!preview) {
355 game.redo_count = 0;
356 game.entries[game.entry_count] = entry;
357 game.entry_count += 1;
358 player.scores[c] = points;
359 player.yahtzee_bonus += bonus;
360 game.turn = if (game.entry_count == 13 * game.player_count) null else .{ .seat = (entry.before.seat + 1) % game.player_count };
361 }
362 try game.emit("{s} {s} {d} in {s}{s}{s}{s} total {d}", .{
363 player.name, if (preview) "would score" else "scores", points, @tagName(category),
364 if (scores.joker) " joker" else "", if (bonus != 0) " ybonus 100" else "", if (upper_bonus != 0) " ubonus 35" else "", total,
365 });
366 if (!preview) {
367 if (game.turn == null) try game.emit("game over", .{}) else try game.emit("turn {d} {s}", .{ game.current().round(), game.current().name });
368 }
369 }
370 fn undo(game: *Game) Error!void {
371 if (game.entry_count == 0) return game.program.fail("nothing to undo", .{});
372 game.entry_count -= 1;
373 const entry = game.entries[game.entry_count];
374 game.redo_entries[game.redo_count] = .{ .after = game.turn, .entry = entry };
375 game.redo_count += 1;
376 const player = &game.players[entry.before.seat];
377 player.scores[@backingInt(entry.category)] = null;
378 player.yahtzee_bonus -= entry.bonus;
379 game.turn = entry.before;
380 try game.emit("undo {s} {s} {d}", .{ player.name, @tagName(entry.category), entry.points });
381 }
382 fn redo(game: *Game) Error!void {
383 if (game.redo_count == 0) return game.program.fail("nothing to redo", .{});
384 game.redo_count -= 1;
385 const saved = game.redo_entries[game.redo_count];
386 const entry = saved.entry;
387 const player = &game.players[entry.before.seat];
388 player.scores[@backingInt(entry.category)] = entry.points;
389 player.yahtzee_bonus += entry.bonus;
390 game.entries[game.entry_count] = entry;
391 game.entry_count += 1;
392 game.turn = saved.after;
393 try game.emit("redo {s} {s} {d}", .{ player.name, @tagName(entry.category), entry.points });
394 }
395 fn botHold(game: *Game, strategy: Strategy) Mask {
396 const dice = game.turn.?.dice;
397 const counts = countsOf(dice);
398 var mask: Mask = 0;
399 if (strategy == .STRAIGHT) {
400 var best_start: usize = 0;
401 var best_length: usize = 0;
402 for (0..6) |start| {
403 var end = start;
404 while (end < 6 and counts[end] > 0) : (end += 1) {}
405 if (end - start >= best_length) {
406 best_start = start;
407 best_length = end - start;
408 }
409 }
410 if (best_length >= 3) {
411 for (best_start + 1..best_start + best_length + 1) |value| {
412 for (dice, 0..) |face, i| if (face == value) {
413 mask |= dieBit(i);
414 break;
415 };
416 }
417 return mask;
418 }
419 }
420 const frequent = mostFrequent(counts) + 1;
421 for (dice, 0..) |face, i| if (if (strategy == .HIGH) face >= 5 else face == frequent) {
422 mask |= dieBit(i);
423 };
424 return mask;
425 }
426 fn botTurn(game: *Game) Error!void {
427 const strategy = game.current().strategy;
428 if (game.turn.?.rolls == 0) try game.roll();
429 while (game.turn.?.rolls < game.rules.rolls) {
430 const mask = game.botHold(strategy);
431 if (mask == 31) break;
432 try game.hold(mask);
433 try game.roll();
434 }
435 const counts = countsOf(game.turn.?.dice);
436 const scores = evaluate(game.current(), counts, game.rules.joker);
437 const frequent = mostFrequent(counts);
438 const category = if (strategy == .UPPER and counts[frequent] >= 3 and scores.allowed & categoryBit(frequent) != 0) categories[frequent] else scores.best;
439 try game.score(category, false);
440 }
441 fn ranking(game: *const Game) [6]usize {
442 var result: [6]usize = undefined;
443 for (0..game.player_count) |i| result[i] = i;
444 std.mem.sort(usize, result[0..game.player_count], game, struct {
445 fn before(table: *const Game, a: usize, b: usize) bool {
446 const first = table.players[a].total();
447 const second = table.players[b].total();
448 return if (first != second) first > second else a < b;
449 }
450 }.before);
451 return result;
452 }

497 fn odds(program: *Program, game: *Game, mask: Mask) Error!void {
498 try game.checkRollsLeft();
499 const kept = heldCode(game.turn.?.dice, mask);
500 var rerolled: usize = 5;
501 for (decode(kept)) |count| rerolled -= count;
502 const outcome_count: i64 = @intCast(powers[rerolled]);
503 var sums: [13]i64 = @splat(0);
504 var chances: [13]i64 = @splat(0);
505 var best: i64 = 0;
506 for (program.outcomes.items[rerolled].items) |outcome| {
507 const scores = evaluate(game.current(), decode(kept + outcome.code), game.rules.joker);
508 for (scores.points, 0..) |points, i| {
509 if (points < 0) continue;
510 sums[i] += outcome.weight * points;
511 if (points > 0) chances[i] += outcome.weight;
512 }
513 best += outcome.weight * scores.points[@backingInt(scores.best)];
514 }
515 try program.emit("odds {f} outcomes {d}", .{ pattern(mask), outcome_count });
516 for (categories, 0..) |c, i| if (game.current().scores[i] == null) {
517 try program.emit("{s} expect {f} chance {f}", .{ @tagName(c), fraction(sums[i], outcome_count), fraction(chances[i], outcome_count) });
518 };
519 try program.emit("best expect {f}", .{fraction(best, outcome_count)});
520 }
521 fn advise(program: *Program, game: *Game) Error!void {
522 try game.checkRollsLeft();
523 const turn = game.turn.?;
524 const depth = game.rules.rolls - turn.rolls;
525 var temporary = std.heap.ArenaAllocator.init(std.heap.page_allocator);
526 defer temporary.deinit();
527 var adviser: Adviser = .{ .gpa = temporary.allocator(), .player = game.current(), .joker = game.rules.joker, .outcomes = &program.outcomes };
528 var best: i64 = -1;
529 var chosen: Mask = 0;
530 var denominator: i64 = 1;
531 for (0..depth) |_| denominator *= 7776;
532 for (0..32) |i| {
533 const mask: Mask = @intCast(i);
534 const value = try adviser.expected(heldCode(turn.dice, mask), depth);
535 if (value > best) {
536 best = value;
537 chosen = mask;
538 }
539 }
540 try program.emit("advise {f}: {f} expect {f}", .{ pattern(chosen), faces(turn.dice, chosen), fraction(best, denominator) });
541 }
542 fn card(program: *Program, game: *Game) Error!void {
543 const totals = [_][]const u8{ "UPPER", "BONUS", "YBONUS", "TOTAL" };
544 for (0..18) |row| {
545 const heading = if (row == 0) "CATEGORY" else if (row <= 13) @tagName(categories[row - 1]) else totals[row - 14];
546 try program.out.writeAll(heading);
547 try program.out.splatByteAll(' ', 9 - heading.len);
548 for (game.players[0..game.player_count]) |player| {
549 var buffer: [32]u8 = undefined;
550 const cell = if (row == 0) player.name else blk: {
551 const value: ?i32 = if (row <= 13) player.scores[row - 1] else switch (row) {
552 14 => player.upper(),
553 15 => player.upperBonus(),
554 16 => player.yahtzee_bonus,
555 else => player.total(),
556 };
557 break :blk if (value) |number| std.fmt.bufPrint(&buffer, "{d}", .{number}) catch unreachable else "-";
558 };
559 const width = @max(@as(usize, 4), player.name.len);
560 try program.out.splatByteAll(' ', 1 + width - @min(width, cell.len));
561 try program.out.writeAll(cell);
562 }
563 try program.out.writeByte('\n');
564 }
565 }
566 fn simulate(program: *Program, seed: i64, count: usize, strategies: []const Strategy) Error!void {
567 var totals: [6]SimulationTotal = @splat(.{});
568 const names = [_][]const u8{ "A", "B", "C", "D", "E", "F" };
569 var state = seed;
570 for (0..count) |_| {
571 var simulation = Game.init(program, state, names[0..strategies.len]);
572 simulation.quiet = true;
573 for (strategies, 0..) |bot, i| simulation.players[i].strategy = bot;
574 while (simulation.turn != null) try simulation.botTurn();
575 state = simulation.random_state;
576 const order = simulation.ranking();
577 totals[order[0]].wins += 1;
578 for (simulation.players[0..strategies.len], 0..) |player, i| {
579 totals[i].sum += player.total();
580 totals[i].best = @max(totals[i].best, player.total());
581 if (player.upperBonus() == 35) totals[i].bonus += 1;
582 if ((player.scores[11] orelse 0) == 50) totals[i].yahtzee += 1;
583 }
584 }
585 try program.emit("sim seed {d} games {d} players {d}", .{ seed, count, strategies.len });
586 for (strategies, 0..) |bot, i| {
587 const total = totals[i];
588 try program.emit("seat {d} {s} wins {d} average {f} best {d} bonus {d} yahtzee {d}", .{ i + 1, @tagName(bot), total.wins, average(total.sum, count), total.best, total.bonus, total.yahtzee });
589 }
590 }
591 const Command = enum { RULE, GAME, BOT, LOAD, ROLL, HOLD, KEEP, SCORE, WHATIF, UNDO, REDO, SUGGEST, ODDS, ADVISE, AUTO, STATUS, CARD, BONUS, STATS, LOG, FACES, STANDINGS, END, RECORDS, SIM };
592 fn execute(program: *Program, words: []const []const u8) Error!void {
593 const command = std.meta.stringToEnum(Command, words[0]) orelse return program.fail("unknown command {s}", .{words[0]});
594 switch (command) {
595 .RULE => {
596 if (!equal(words[1], "ROLLS") and !equal(words[1], "JOKER")) return program.fail("unknown rule {s}", .{words[1]});
597 var changed = program.rules;
598 if (equal(words[1], "ROLLS")) changed.rolls = @intCast(try program.positive(words[2], 4)) else changed.joker = std.meta.stringToEnum(Joker, words[2]) orelse return program.fail("bad value {s}", .{words[2]});
599 if (program.game != null) return program.fail("game in progress", .{});
600 program.rules = changed;
601 try program.emit("rule {s} {s}", .{ words[1], words[2] });
602 },
603 .GAME => {
604 const seed = try program.positive(words[1], 2147483646);
605 const names = words[2..];
606 for (names) |word| try program.checkName(word);
607 if (names.len > 6) return program.fail("too many players", .{});
608 if (program.game != null) return program.fail("game in progress", .{});
609 for (names, 0..) |word, i| for (names[0..i]) |previous| if (equal(word, previous)) return program.fail("duplicate name {s}", .{word});
610 const game = try program.gpa.create(Game);
611 game.* = Game.init(program, seed, names);
612 program.game = game;
613 try program.out.print("game seed {d} players", .{seed});
614 for (names) |word| try program.out.print(" {s}", .{word});
615 try program.out.writeByte('\n');
616 try program.emit("turn 1 {s}", .{names[0]});
617 },
618 .BOT => {
619 try program.checkName(words[1]);
620 const bot = try program.parseStrategy(words[2]);
621 const game = try program.table();
622 const player: *Player = blk: {
623 for (game.players[0..game.player_count]) |*candidate| if (equal(candidate.name, words[1])) break :blk candidate;
624 return program.fail("no player {s}", .{words[1]});
625 };
626 player.strategy = bot;
627 try program.emit("bot {s} {s}", .{ words[1], words[2] });
628 },
629 .LOAD, .KEEP => {
630 var values: [20]u8 = undefined;
631 const count = words.len - 1;
632 for (words[1..], 0..) |word, i| values[i] = @intCast(try program.positive(word, 6));
633 const game = try program.table();
634 if (command == .KEEP) try game.keep(values[0..count]) else {
635 try game.loaded.appendSlice(program.gpa, values[0..count]);
636 try program.emit("loaded {d}", .{game.loaded.items.len - game.loaded_head});
637 }
638 },
639 .ROLL => try (try program.table()).roll(),
640 .HOLD, .ODDS => {
641 const mask = try program.parsePattern(words[1]);
642 const game = try program.table();
643 if (command == .HOLD) try game.hold(mask) else try program.odds(game, mask);
644 },
645 .SCORE, .WHATIF => {
646 const category = try program.parseCategory(words[1]);
647 try (try program.table()).score(category, command == .WHATIF);
648 },
649 .UNDO => try (try program.table()).undo(),
650 .REDO => try (try program.table()).redo(),
651 .SUGGEST => {
652 const game = try program.table();
653 try game.checkRolled();
654 const scores = evaluate(game.current(), countsOf(game.turn.?.dice), game.rules.joker);
655 try program.out.writeAll("suggest");
656 for (categories, 0..) |category, i| if (scores.points[i] >= 0) try program.out.print(" {s} {d}", .{ @tagName(category), scores.points[i] });
657 try program.out.print(" best {s} {d}\n", .{ @tagName(scores.best), scores.points[@backingInt(scores.best)] });
658 },
659 .ADVISE => try program.advise(try program.table()),
660 .AUTO => {
661 const count: usize = @intCast(try program.positive(words[1], 100));
662 const game = try program.table();
663 try game.checkTurn();
664 game.redo_count = 0;
665 for (0..count) |_| {
666 if (game.turn == null) break;
667 try game.botTurn();
668 }
669 },
670 .STATUS => {
671 const game = try program.table();
672 if (game.turn) |turn| try program.emit("status round {d} player {s} rolls {d} dice {f} holds {f}", .{ game.current().round(), game.current().name, turn.rolls, faces(if (turn.rolls == 0) null else turn.dice, 31), pattern(turn.hold) }) else try program.emit("status game over", .{});
673 },
674 .CARD => try program.card(try program.table()),
675 .BONUS => {
676 const game = try program.table();
677 for (game.players[0..game.player_count]) |player| {
678 const upper = player.upper();
679 var maximum = upper;
680 for (player.scores[0..6], 0..) |score, i| if (score == null) {
681 maximum += @intCast(5 * (i + 1));
682 };
683 const status = if (upper >= 63) "made" else if (maximum < 63) "lost" else "open";
684 try program.emit("bonus {s} upper {d} need {d} max {d} {s}", .{ player.name, upper, @max(63 - upper, 0), maximum, status });
685 }
686 },
687 .STATS => {
688 const game = try program.table();
689 for (game.players[0..game.player_count], 0..) |player, i| {
690 var turns: usize = 0;
691 var rolls: usize = 0;
692 var zeros: usize = 0;
693 var yahtzees: usize = 0;
694 var sum: i64 = 0;
695 for (game.entries[0..game.entry_count]) |entry| {
696 if (entry.before.seat != i) continue;
697 turns += 1;
698 rolls += entry.before.rolls;
699 zeros += @intFromBool(entry.points == 0);
700 yahtzees += @intFromBool(yahtzee(countsOf(entry.before.dice)));
701 sum += entry.points;
702 }
703 try program.emit("stats {s} turns {d} rolls {d} zeros {d} yahtzees {d} average {f}", .{ player.name, turns, rolls, zeros, yahtzees, average(sum, turns) });
704 }
705 },
706 .LOG => {
707 const game = try program.table();
708 for (game.entries[0..game.entry_count], 0..) |entry, i| {
709 try program.emit("log {d} {s} {f} {s} {d}{s}{s}", .{ i + 1, game.players[entry.before.seat].name, faces(entry.before.dice, 31), @tagName(entry.category), entry.points, if (entry.joker) " joker" else "", if (entry.bonus != 0) " ybonus 100" else "" });
710 }
711 if (game.entry_count == 0) try program.emit("no scores", .{});
712 },
713 .FACES => {
714 const game = try program.table();
715 var total: i64 = 0;
716 var squares: i64 = 0;
717 try program.out.writeAll("faces");
718 for (game.face_totals) |count| {
719 total += count;
720 squares += count * count;
721 try program.out.print(" {d}", .{count});
722 }
723 try program.out.print(" total {d} chi2 {f}\n", .{ total, fraction(6 * squares - total * total, if (total == 0) 1 else total) });
724 },
725 .STANDINGS, .END => {
726 const game = try program.table();
727 const ranking = game.ranking();
728 const finished = game.turn == null;
729 if (command == .END) try program.emit("end {s}", .{if (finished) "final" else "abandoned"});
730 for (ranking[0..game.player_count], 0..) |seat, rank| {
731 const player = game.players[seat];
732 if (command == .STANDINGS) try program.emit("standing {d} {s} {d} behind {d}", .{ rank + 1, player.name, player.total(), game.players[ranking[0]].total() - player.total() }) else {
733 try program.emit("rank {d} {s} {d}", .{ rank + 1, player.name, player.total() });
734 if (finished) {
735 var record = program.records.get(player.name) orelse Record{};
736 record.games += 1;
737 record.wins += @intFromBool(rank == 0);
738 record.sum += player.total();
739 record.best = @max(record.best, player.total());
740 try program.records.put(program.gpa, player.name, record);
741 }
742 }
743 }
744 if (command == .END) program.game = null;
745 },
746 .RECORDS => {
747 var names: std.ArrayList([]const u8) = .empty;
748 var iterator = program.records.iterator();
749 while (iterator.next()) |item| try names.append(program.gpa, item.key_ptr.*);
750 std.mem.sort([]const u8, names.items, program, struct {
751 fn before(context: *Program, a: []const u8, b: []const u8) bool {
752 const first = context.records.get(a).?;
753 const second = context.records.get(b).?;
754 if (first.wins != second.wins) return first.wins > second.wins;
755 if (first.best != second.best) return first.best > second.best;
756 return std.mem.lessThan(u8, a, b);
757 }
758 }.before);
759 for (names.items) |word| {
760 const record = program.records.get(word).?;
761 try program.emit("record {s} games {d} wins {d} best {d} average {f}", .{ word, record.games, record.wins, record.best, average(record.sum, record.games) });
762 }
763 if (names.items.len == 0) try program.emit("no records", .{});
764 },
765 .SIM => {
766 const seed = try program.positive(words[1], 2147483646);
767 const count: usize = @intCast(try program.positive(words[2], 50));
768 var strategies: [8]Strategy = undefined;
769 for (words[3..], 0..) |word, i| strategies[i] = try program.parseStrategy(word);
770 const size = words.len - 3;
771 if (size > 6) return program.fail("too many players", .{});
772 try program.simulate(seed, count, strategies[0..size]);
773 },
774 }
775 }
776};

778fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
779 var buffer: std.Io.Writer.Allocating = .init(gpa);
780 defer buffer.deinit();
781 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
782 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
783 const line = try buffer.toOwnedSlice();
784 var words: std.ArrayList([]const u8) = .empty;
785 var parts = std.mem.tokenizeAny(u8, line, " \r");
786 while (parts.next()) |part| try words.append(gpa, part);
787 return try words.toOwnedSlice(gpa);
788}

790pub fn main(init: std.process.Init) !void {
791 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
792 defer arena.deinit();
793 const gpa = arena.allocator();
794 var input_buffer: [4096]u8 = undefined;
795 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
796 var output_buffer: [4096]u8 = undefined;
797 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
798 const count = std.fmt.parseInt(usize, (try readWords(gpa, &input.interface))[0], 10) catch unreachable;
799 var program: Program = .{ .gpa = gpa, .out = &output.interface };
800 try program.outcomes.init(gpa);
801 for (0..count) |_| {
802 const words = try readWords(gpa, &input.interface);
803 var response: std.Io.Writer.Allocating = .init(gpa);
804 defer response.deinit();
805 program.out = &response.writer;
806 program.execute(words) catch |err| {
807 if (err != error.BadCommand) return err;
808 try output.interface.print("error: {s}\n", .{program.error_message});
809 continue;
810 };
811 try output.interface.writeAll(response.written());
812 }
813 try output.interface.flush();
814}
