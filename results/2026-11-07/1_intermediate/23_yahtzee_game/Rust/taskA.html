4const CATEGORIES: [&str; 13] = [
5 "ONES", "TWOS", "THREES", "FOURS", "FIVES", "SIXES", "THREEKIND",
6 "FOURKIND", "FULLHOUSE", "SMALL", "LARGE", "YAHTZEE", "CHANCE",
7];
8const POWERS: [usize; 6] = [1, 6, 36, 216, 1296, 7776];
9const FACTORIAL: [i64; 6] = [1, 1, 2, 6, 24, 120];
10const CODES: usize = 46656;
11type CommandResult<T = ()> = Result<T, String>;

13#[derive(Clone, Copy, PartialEq, Eq)]
14enum Strategy { Kind, Straight, High, Upper }
15
16impl Strategy {
17 fn parse(word: &str) -> CommandResult<Self> {
18 match word {
19 "KIND" => Ok(Self::Kind),
20 "STRAIGHT" => Ok(Self::Straight),
21 "HIGH" => Ok(Self::High),
22 "UPPER" => Ok(Self::Upper),
23 _ => Err(format!("bad strategy {word}")),
24 }
25 }
26
27 fn name(self) -> &'static str {
28 match self {
29 Self::Kind => "KIND", Self::Straight => "STRAIGHT",
30 Self::High => "HIGH", Self::Upper => "UPPER",
31 }
32 }
33}

35#[derive(Clone, Copy, PartialEq, Eq)]
36enum JokerRule { Forced, Free, Off }

38#[derive(Clone, Copy)]
39struct Rules { rolls: usize, joker: JokerRule }

41fn positive(word: &str, maximum: i64) -> CommandResult<i64> {
42 if !matches!(word.as_bytes()[0], b'1'..=b'9') || !word.bytes().all(|b| b.is_ascii_digit()) {
43 return Err(format!("bad value {word}"));
44 }
45 let value = word.parse::<i64>().unwrap();
46 if value > maximum { Err(format!("bad value {word}")) } else { Ok(value) }
47}
48
49fn valid_name(word: &str) -> CommandResult {
50 if word.len() > 8 || !word.bytes().all(|b| b.is_ascii_uppercase()) {
51 Err(format!("bad name {word}"))
52 } else { Ok(()) }
53}
54
55fn parse_pattern(word: &str) -> CommandResult<u8> {
56 if word.len() != 5 || !word.bytes().all(|b| b == b'0' || b == b'1') {
57 return Err(format!("bad pattern {word}"));
58 }
59 Ok(word.bytes().fold(0, |mask, b| mask * 2 + b - b'0'))
60}
61
62fn fraction(mut numerator: i64, mut denominator: i64) -> String {
63 let (mut a, mut b) = (numerator, denominator);
64 while b != 0 { (a, b) = (b, a % b); }
65 numerator /= a;
66 denominator /= a;
67 if denominator == 1 { numerator.to_string() } else { format!("{numerator}/{denominator}") }
68}
69
70fn average(sum: i64, count: usize) -> String {
71 let value = if count == 0 { 0 } else { 100 * sum / count as i64 };
72 format!("{}.{:02}", value / 100, value % 100)
73}
74
75fn held(mask: u8, die: usize) -> bool { mask & (1 << (4 - die)) != 0 }
76
77fn pattern_text(mask: u8) -> String { format!("{mask:05b}") }
78
79fn face_text(faces: &[usize; 5], mask: u8) -> String {
80 let selected: Vec<_> = faces.iter().enumerate()
81 .filter(|(i, _)| held(mask, *i)).map(|(_, face)| face.to_string()).collect();
82 if selected.is_empty() { "-".into() } else { selected.join(" ") }
83}
84
85fn face_counts(faces: &[usize; 5]) -> [usize; 6] {
86 let mut counts = [0; 6];
87 for &face in faces { counts[face - 1] += 1; }
88 counts
89}
90
91fn decode(mut code: usize) -> [usize; 6] {
92 std::array::from_fn(|_| { let count = code % 6; code /= 6; count })
93}
94
95fn held_code(faces: &[usize; 5], mask: u8) -> usize {
96 faces.iter().enumerate().filter(|(i, _)| held(mask, *i))
97 .map(|(_, face)| POWERS[face - 1]).sum()
98}
99
100fn frequent(counts: &[usize; 6]) -> usize {
101 (0..6).max_by_key(|&i| (counts[i], i)).unwrap()
102}
103
104fn normal_score(category: usize, counts: &[usize; 6]) -> i32 {
105 if category < 6 { return ((category + 1) * counts[category]) as i32; }
106 let sum = counts.iter().enumerate().map(|(i, n)| (i + 1) * n).sum::<usize>() as i32;
107 let maximum = *counts.iter().max().unwrap();
108 let bits = (0..6).filter(|&i| counts[i] != 0).fold(0, |mask, i| mask | (1 << i));
109 match category {
110 6 => if maximum >= 3 { sum } else { 0 },
111 7 => if maximum >= 4 { sum } else { 0 },
112 8 => if counts.contains(&2) && counts.contains(&3) { 25 } else { 0 },
113 9 => if bits & 15 == 15 || bits & 30 == 30 || bits & 60 == 60 { 30 } else { 0 },
114 10 => if bits == 31 || bits == 62 { 40 } else { 0 },
115 11 => if maximum == 5 { 50 } else { 0 },
116 _ => sum,
117 }
118}

132 fn upper(&self) -> i32 { self.scores[..6].iter().flatten().sum() }
133 fn upper_bonus(&self) -> i32 { if self.upper() >= 63 { 35 } else { 0 } }
134 fn total(&self) -> i32 { self.scores.iter().flatten().sum::<i32>() + self.upper_bonus() + self.yahtzee_bonus }
135 fn round(&self) -> usize { self.scores.iter().filter(|s| s.is_some()).count() + 1 }
136}

138struct Candidates {
139 points: [Option<i32>; 13],
140 allowed: u16,
141 best: usize,
142 yahtzee_face: Option<usize>,
143 joker: bool,
144}
145
146fn evaluate(player: &Player, counts: &[usize; 6], rule: JokerRule) -> Candidates {
147 let yahtzee_face = counts.iter().position(|&n| n == 5);
148 let joker = rule != JokerRule::Off && yahtzee_face.is_some() && player.scores[11].is_some();
149 let mut allowed = (0..13).filter(|&c| player.scores[c].is_none()).fold(0, |mask, c| mask | (1 << c));
150 if joker && rule == JokerRule::Forced {
151 let upper = 1 << yahtzee_face.unwrap();
152 if allowed & upper != 0 { allowed = upper; }
153 else if allowed & 0x1fc0 != 0 { allowed &= 0x1fc0; }
154 }
155 let mut result = Candidates { points: [None; 13], allowed, best: 0, yahtzee_face, joker };
156 let mut best_score = -1;
157 for c in 0..13 {
158 if player.scores[c].is_some() { continue; }
159 let points = if allowed & (1 << c) == 0 { 0 }
160 else if joker && (8..=10).contains(&c) { [25, 30, 40][c - 8] }
161 else { normal_score(c, counts) };
162 result.points[c] = Some(points);
163 if points > best_score { best_score = points; result.best = c; }
164 }
165 result
166}

168#[derive(Clone, Copy)]
169struct Outcome { code: usize, weight: i64 }
170
171fn make_outcomes() -> [Vec<Outcome>; 6] {
172 fn fill(out: &mut Vec<Outcome>, size: usize, face: usize, left: usize, code: usize, divisor: i64) {
173 if face == 6 {
174 if left == 0 { out.push(Outcome { code, weight: FACTORIAL[size] / divisor }); }
175 return;
176 }
177 for n in 0..=left { fill(out, size, face + 1, left - n, code + n * POWERS[face], divisor * FACTORIAL[n]); }
178 }
179 std::array::from_fn(|size| {
180 let mut out = Vec::new();
181 fill(&mut out, size, 0, size, 0, 1);
182 out
183 })
184}

215 fn check_turn(&self) -> CommandResult<Turn> { self.turn.ok_or_else(|| "game over".into()) }
216
217 fn rolled(&self) -> CommandResult<Turn> {
218 let turn = self.check_turn()?;
219 if turn.rolls == 0 { Err("not rolled".into()) } else { Ok(turn) }
220 }
221
222 fn rolls_left(&self) -> CommandResult<Turn> {
223 let turn = self.rolled()?;
224 if turn.rolls == self.rules.rolls { Err("no rolls left".into()) } else { Ok(turn) }
225 }
226
227 fn roll(&mut self) -> CommandResult {
228 let mut turn = self.check_turn()?;
229 if turn.rolls == self.rules.rolls { return Err("no rolls left".into()); }
230 if turn.hold == 31 { return Err("all dice held".into()); }
231 self.redo.clear();
232 for i in 0..5 {
233 if turn.rolls != 0 && held(turn.hold, i) { continue; }
234 turn.faces[i] = if let Some(face) = self.loaded.pop_front() { face } else {
235 self.random_state = 48271 * self.random_state % 2147483647;
236 (self.random_state % 6 + 1) as usize
237 };
238 self.face_counts[turn.faces[i] - 1] += 1;
239 }
240 turn.rolls += 1;
241 self.turn = Some(turn);
242 if self.verbose { println!("{} roll {}: {}", self.players[turn.seat].name, turn.rolls, face_text(&turn.faces, 31)); }
243 Ok(())
244 }
245
246 fn hold(&mut self, mask: u8) -> CommandResult {
247 let mut turn = self.rolls_left()?;
248 self.redo.clear();
249 turn.hold = mask;
250 self.turn = Some(turn);
251 if self.verbose { println!("{} holds {}: {}", self.players[turn.seat].name, pattern_text(mask), face_text(&turn.faces, mask)); }
252 Ok(())
253 }
254
255 fn keep(&mut self, values: &[usize]) -> CommandResult {
256 let turn = self.rolls_left()?;
257 let mut mask = 0;
258 for &value in values {
259 let die = (0..5).find(|&i| !held(mask, i) && turn.faces[i] == value)
260 .ok_or_else(|| format!("not showing {value}"))?;
261 mask |= 1 << (4 - die);
262 }
263 self.hold(mask)
264 }
265
266 fn score(&mut self, category: usize, preview: bool) -> CommandResult {
267 let turn = self.rolled()?;
268 let player = &self.players[turn.seat];
269 if player.scores[category].is_some() { return Err(format!("{} already used", CATEGORIES[category])); }
270 let candidates = evaluate(player, &face_counts(&turn.faces), self.rules.joker);
271 if candidates.allowed & (1 << category) == 0 {
272 let face = candidates.yahtzee_face.unwrap();
273 return Err(format!("joker requires {}", if player.scores[face].is_none() { CATEGORIES[face] } else { "lower box" }));
274 }
275 let points = candidates.points[category].unwrap();
276 let bonus = if candidates.yahtzee_face.is_some() && player.scores[11] == Some(50) { 100 } else { 0 };
277 let upper = player.upper();
278 let gained_upper = if category < 6 && upper < 63 && upper + points >= 63 { 35 } else { 0 };
279 if self.verbose {
280 print!("{} {} {} in {}", player.name, if preview { "would score" } else { "scores" }, points, CATEGORIES[category]);
281 if candidates.joker { print!(" joker"); }
282 if bonus != 0 { print!(" ybonus 100"); }
283 if gained_upper != 0 { print!(" ubonus 35"); }
284 println!(" total {}", player.total() + points + bonus + gained_upper);
285 }
286 if preview { return Ok(()); }
287 self.redo.clear();
288 self.entries.push(ScoreEntry { before: turn, category, points, bonus, joker: candidates.joker });
289 self.players[turn.seat].scores[category] = Some(points);
290 self.players[turn.seat].yahtzee_bonus += bonus;
291 self.turn = if self.entries.len() == 13 * self.players.len() { None }
292 else { Some(Turn { seat: (turn.seat + 1) % self.players.len(), ..Turn::default() }) };
293 if self.verbose {
294 if let Some(next) = self.turn { println!("turn {} {}", self.players[next.seat].round(), self.players[next.seat].name); }
295 else { println!("game over"); }
296 }
297 Ok(())
298 }
299
300 fn undo(&mut self) -> CommandResult {
301 let entry = self.entries.pop().ok_or_else(|| "nothing to undo".to_string())?;
302 self.redo.push(RedoEntry { after: self.turn, entry });
303 let player = &mut self.players[entry.before.seat];
304 player.scores[entry.category] = None;
305 player.yahtzee_bonus -= entry.bonus;
306 self.turn = Some(entry.before);
307 println!("undo {} {} {}", player.name, CATEGORIES[entry.category], entry.points);
308 Ok(())
309 }

311 fn redo(&mut self) -> CommandResult {
312 let saved = self.redo.pop().ok_or_else(|| "nothing to redo".to_string())?;
313 let entry = saved.entry;
314 let player = &mut self.players[entry.before.seat];
315 player.scores[entry.category] = Some(entry.points);
316 player.yahtzee_bonus += entry.bonus;
317 self.entries.push(entry);
318 self.turn = saved.after;
319 println!("redo {} {} {}", player.name, CATEGORIES[entry.category], entry.points);
320 Ok(())
321 }

323 fn bot_hold(&self, strategy: Strategy) -> u8 {
324 let faces = self.turn.unwrap().faces;
325 let counts = face_counts(&faces);
326 let favorite = frequent(&counts) + 1;
327 let mut mask = 0;
328 if strategy == Strategy::Straight {
329 let (mut best_start, mut best_length) = (0, 0);
330 for start in 0..6 {
331 let mut end = start;
332 while end < 6 && counts[end] != 0 { end += 1; }
333 if end - start >= best_length { best_start = start; best_length = end - start; }
334 }
335 if best_length >= 3 {
336 for face in best_start + 1..=best_start + best_length {
337 let die = faces.iter().position(|&value| value == face).unwrap();
338 mask |= 1 << (4 - die);
339 }
340 return mask;
341 }
342 }
343 for (i, &face) in faces.iter().enumerate() {
344 if if strategy == Strategy::High { face >= 5 } else { face == favorite } { mask |= 1 << (4 - i); }
345 }
346 mask
347 }
348
349 fn bot_turn(&mut self) {
350 let strategy = self.players[self.turn.unwrap().seat].strategy;
351 if self.turn.unwrap().rolls == 0 { self.roll().unwrap(); }
352 while self.turn.unwrap().rolls < self.rules.rolls {
353 let mask = self.bot_hold(strategy);
354 if mask == 31 { break; }
355 self.hold(mask).unwrap();
356 self.roll().unwrap();
357 }
358 let turn = self.turn.unwrap();
359 let counts = face_counts(&turn.faces);
360 let candidates = evaluate(&self.players[turn.seat], &counts, self.rules.joker);
361 let favorite = frequent(&counts);
362 let category = if strategy == Strategy::Upper && counts[favorite] >= 3 && candidates.allowed & (1 << favorite) != 0 {
363 favorite
364 } else { candidates.best };
365 self.score(category, false).unwrap();
366 }

374 fn odds(&self, mask: u8, outcomes: &[Vec<Outcome>; 6]) -> CommandResult {
375 let turn = self.rolls_left()?;
376 let player = &self.players[turn.seat];
377 let code = held_code(&turn.faces, mask);
378 let rerolled = 5 - mask.count_ones() as usize;
379 let (mut sums, mut chances, mut best) = ([0; 13], [0; 13], 0);
380 for outcome in &outcomes[rerolled] {
381 let candidates = evaluate(player, &decode(code + outcome.code), self.rules.joker);
382 for c in 0..13 {
383 if let Some(points) = candidates.points[c] {
384 sums[c] += outcome.weight * i64::from(points);
385 if points > 0 { chances[c] += outcome.weight; }
386 }
387 }
388 best += outcome.weight * i64::from(candidates.points[candidates.best].unwrap());
389 }
390 let denominator = POWERS[rerolled] as i64;
391 println!("odds {} outcomes {}", pattern_text(mask), denominator);
392 for c in 0..13 {
393 if player.scores[c].is_none() {
394 println!("{} expect {} chance {}", CATEGORIES[c], fraction(sums[c], denominator), fraction(chances[c], denominator));
395 }
396 }
397 println!("best expect {}", fraction(best, denominator));
398 Ok(())
399 }

401 fn advise(&self, outcomes: &[Vec<Outcome>; 6]) -> CommandResult {
402 let turn = self.rolls_left()?;
403 let depth = self.rules.rolls - turn.rolls;
404 let mut adviser = Adviser {
405 player: &self.players[turn.seat], joker: self.rules.joker, outcomes,
406 values: vec![vec![-1; CODES]; depth + 1], expectations: vec![vec![-1; CODES]; depth + 1],
407 };
408 let (mut best, mut chosen) = (-1, 0);
409 for mask in 0..32 {
410 let value = adviser.expected(held_code(&turn.faces, mask), depth);
411 if value > best { best = value; chosen = mask; }
412 }
413 println!("advise {}: {} expect {}", pattern_text(chosen), face_text(&turn.faces, chosen), fraction(best, 7776_i64.pow(depth as u32)));
414 Ok(())
415 }

453 let sum = entries.iter().map(|e| i64::from(e.points)).sum();

461 if entry.bonus != 0 { print!(" ybonus 100"); }

476 for (rank, i) in order.into_iter().enumerate() {
477 let player = &self.players[i];
478 println!("standing {} {} {} behind {}", rank + 1, player.name, player.total(), leader - player.total());
479 }
480 }
481 _ => unreachable!(),
482 }
483 }

486struct Adviser<'a> {
487 player: &'a Player,
488 joker: JokerRule,
489 outcomes: &'a [Vec<Outcome>; 6],
490 values: Vec<Vec<i64>>,
491 expectations: Vec<Vec<i64>>,
492}
493
494impl Adviser<'_> {
495 fn value(&mut self, code: usize, depth: usize) -> i64 {
496 if self.values[depth][code] >= 0 { return self.values[depth][code]; }
497 let counts = decode(code);
498 let value = if depth == 0 {
499 let candidates = evaluate(self.player, &counts, self.joker);
500 i64::from(candidates.points[candidates.best].unwrap())
501 } else { self.search_holds(&counts, 0, 0, depth) };
502 self.values[depth][code] = value;
503 value
504 }
505
506 fn search_holds(&mut self, counts: &[usize; 6], face: usize, code: usize, depth: usize) -> i64 {
507 if face == 6 { return self.expected(code, depth); }
508 let mut best = -1_i64;
509 for n in 0..=counts[face] { best = best.max(self.search_holds(counts, face + 1, code + n * POWERS[face], depth)); }
510 best
511 }
512
513 fn expected(&mut self, code: usize, depth: usize) -> i64 {
514 if self.expectations[depth][code] >= 0 { return self.expectations[depth][code]; }
515 let kept: usize = decode(code).iter().sum();
516 let mut sum = 0;
517 for i in 0..self.outcomes[5 - kept].len() {
518 let outcome = self.outcomes[5 - kept][i];
519 sum += outcome.weight * self.value(code + outcome.code, depth - 1);
520 }
521 let value = sum * POWERS[kept] as i64;
522 self.expectations[depth][code] = value;
523 value
524 }
525}

536 fn simulate(&self, seed: i64, count: usize, bots: &[Strategy]) {
537 let (mut wins, mut sums, mut best, mut bonus, mut yahtzee) = ([0; 6], [0_i64; 6], [0_i32; 6], [0; 6], [0; 6]);
538 let names = ["A", "B", "C", "D", "E", "F"];
539 let mut state = seed;
540 for _ in 0..count {
541 let mut game = Game::new(state, &names[..bots.len()], self.rules, false);
542 for (player, &strategy) in game.players.iter_mut().zip(bots) { player.strategy = strategy; }
543 while game.turn.is_some() { game.bot_turn(); }
544 state = game.random_state;
545 wins[game.ranking()[0]] += 1;
546 for (i, player) in game.players.iter().enumerate() {
547 let total = player.total();
548 sums[i] += i64::from(total);
549 best[i] = best[i].max(total);
550 bonus[i] += usize::from(player.upper_bonus() == 35);
551 yahtzee[i] += usize::from(player.scores[11] == Some(50));
552 }
553 }
554 println!("sim seed {} games {} players {}", seed, count, bots.len());
555 for (i, strategy) in bots.iter().enumerate() {
556 println!("seat {} {} wins {} average {} best {} bonus {} yahtzee {}", i + 1, strategy.name(), wins[i], average(sums[i], count), best[i], bonus[i], yahtzee[i]);
557 }
558 }

560 fn execute(&mut self, words: &[&str], outcomes: &[Vec<Outcome>; 6]) -> CommandResult {
561 match words[0] {
562 "RULE" => {
563 let mut rules = self.rules;
564 match words[1] {
565 "ROLLS" => rules.rolls = positive(words[2], 4)? as usize,
566 "JOKER" => rules.joker = match words[2] {
567 "FORCED" => JokerRule::Forced, "FREE" => JokerRule::Free, "OFF" => JokerRule::Off,
568 word => return Err(format!("bad value {word}")),
569 },
570 word => return Err(format!("unknown rule {word}")),
571 }
572 if self.game.is_some() { return Err("game in progress".into()); }
573 self.rules = rules;
574 println!("rule {} {}", words[1], words[2]);
575 }
576 "GAME" => {
577 let seed = positive(words[1], 2147483646)?;
578 let names = &words[2..];
579 for name in names { valid_name(name)?; }
580 if names.len() > 6 { return Err("too many players".into()); }
581 if self.game.is_some() { return Err("game in progress".into()); }
582 for (i, name) in names.iter().enumerate() {
583 if names[..i].contains(name) { return Err(format!("duplicate name {name}")); }
584 }
585 self.game = Some(Game::new(seed, names, self.rules, true));
586 println!("game seed {} players {}\nturn 1 {}", seed, names.join(" "), names[0]);
587 }
588 "BOT" => {
589 valid_name(words[1])?;
590 let strategy = Strategy::parse(words[2])?;
591 let game = self.game_mut()?;
592 let player = game.players.iter_mut().find(|p| p.name == words[1]).ok_or_else(|| format!("no player {}", words[1]))?;
593 player.strategy = strategy;
594 println!("bot {} {}", words[1], words[2]);
595 }
596 "LOAD" | "KEEP" => {
597 let values = words[1..].iter().map(|word| positive(word, 6).map(|n| n as usize)).collect::<CommandResult<Vec<_>>>()?;
598 let game = self.game_mut()?;
599 if words[0] == "KEEP" { game.keep(&values)?; }
600 else { game.loaded.extend(values); println!("loaded {}", game.loaded.len()); }
601 }
602 "ROLL" => self.game_mut()?.roll()?,
603 "HOLD" | "ODDS" => {
604 let mask = parse_pattern(words[1])?;
605 if words[0] == "HOLD" { self.game_mut()?.hold(mask)?; }
606 else { self.game()?.odds(mask, outcomes)?; }
607 }
608 "SCORE" | "WHATIF" => {
609 let category = CATEGORIES.iter().position(|&c| c == words[1]).ok_or_else(|| format!("bad category {}", words[1]))?;
610 self.game_mut()?.score(category, words[0] == "WHATIF")?;
611 }
612 "UNDO" => self.game_mut()?.undo()?,
613 "REDO" => self.game_mut()?.redo()?,
614 "SUGGEST" => {
615 let game = self.game()?;
616 let turn = game.rolled()?;
617 let candidates = evaluate(&game.players[turn.seat], &face_counts(&turn.faces), game.rules.joker);
618 print!("suggest");
619 for (category, points) in CATEGORIES.iter().zip(candidates.points) {
620 if let Some(points) = points { print!(" {category} {points}"); }
621 }
622 println!(" best {} {}", CATEGORIES[candidates.best], candidates.points[candidates.best].unwrap());
623 }
624 "ADVISE" => self.game()?.advise(outcomes)?,
625 "AUTO" => {
626 let count = positive(words[1], 100)?;
627 let game = self.game_mut()?;
628 game.check_turn()?;
629 game.redo.clear();
630 for _ in 0..count { if game.turn.is_none() { break; } game.bot_turn(); }
631 }
632 "STATUS" | "CARD" | "BONUS" | "STATS" | "LOG" | "FACES" | "STANDINGS" => self.game()?.report(words[0]),
633 "END" => {
634 let game = self.game.take().ok_or_else(|| "no game".to_string())?;
635 println!("end {}", if game.turn.is_none() { "final" } else { "abandoned" });
636 for (rank, i) in game.ranking().into_iter().enumerate() {
637 let player = &game.players[i];
638 let total = player.total();
639 println!("rank {} {} {}", rank + 1, player.name, total);
640 if game.turn.is_none() {
641 let record = self.records.entry(player.name.clone()).or_default();
642 record.games += 1;
643 record.wins += usize::from(rank == 0);
644 record.sum += i64::from(total);
645 record.best = record.best.max(total);
646 }
647 }
648 }
649 "RECORDS" => {
650 let mut order: Vec<_> = self.records.iter().collect();
651 order.sort_by(|(an, a), (bn, b)| b.wins.cmp(&a.wins).then(b.best.cmp(&a.best)).then(an.cmp(bn)));
652 for (name, record) in order {
653 println!("record {} games {} wins {} best {} average {}", name, record.games, record.wins, record.best, average(record.sum, record.games));
654 }
655 if self.records.is_empty() { println!("no records"); }
656 }
657 "SIM" => {
658 let seed = positive(words[1], 2147483646)?;
659 let count = positive(words[2], 50)? as usize;
660 let bots = words[3..].iter().map(|word| Strategy::parse(word)).collect::<CommandResult<Vec<_>>>()?;
661 if bots.len() > 6 { return Err("too many players".into()); }
662 self.simulate(seed, count, &bots);
663 }
664 word => return Err(format!("unknown command {word}")),
665 }
666 Ok(())
667 }
668}

670fn main() {
671 let mut input = String::new();
672 io::stdin().read_to_string(&mut input).unwrap();
673 let mut lines = input.lines();
674 let count: usize = lines.next().unwrap().parse().unwrap();
675 let outcomes = make_outcomes();
676 let mut table = Table { game: None, rules: Rules { rolls: 3, joker: JokerRule::Forced }, records: BTreeMap::new() };
677 for line in lines.take(count) {
678 let words: Vec<_> = line.split_whitespace().collect();
679 if let Err(error) = table.execute(&words, &outcomes) { println!("error: {error}"); }
680 }
