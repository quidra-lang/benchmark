1CATEGORIES = %w[
2 ONES TWOS THREES FOURS FIVES SIXES THREEKIND FOURKIND FULLHOUSE
3 SMALL LARGE YAHTZEE CHANCE

13PATTERNS = (0...32).map { |i| i.to_s(2).rjust(5, '0').chars.map { |c| c == '1' } }.freeze

22def integer_word(text, minimum, maximum)
23 require_value(/\A-?(?:0|[1-9][0-9]*)\z/.match?(text) && text != '-0', "bad value #{text}")
24 value = text.to_i
25 require_value(value.between?(minimum, maximum), "bad value #{text}")
26 value
27end

49def face_text(faces)

53def pattern_text(holds)

57def held_faces(faces, holds)
58 faces.each_with_index.filter_map { |face, i| face if holds[i] }

61def fraction_text(numerator, denominator)
62 divisor = numerator.gcd(denominator)
63 numerator /= divisor
64 denominator /= divisor
65 denominator == 1 ? numerator.to_s : "#{numerator}/#{denominator}"
66end

68def average_text(total, count)
69 cents = count.zero? ? 0 : 100 * total / count
70 "#{cents / 100}.#{(cents % 100).to_s.rjust(2, '0')}"
71end

73def yahtzee?(faces)
74 faces.all? { |face| face == faces[0] }
75end

77def normal_score(category, faces)
78 upper = UPPER.index(category)
79 return (upper + 1) * faces.count(upper + 1) unless upper.nil?
80 counts = FACES.map { |face| faces.count(face) }
81 total = faces.sum
82 case category
83 when 'THREEKIND' then counts.max >= 3 ? total : 0
84 when 'FOURKIND' then counts.max >= 4 ? total : 0
85 when 'FULLHOUSE' then counts.include?(3) && counts.include?(2) ? 25 : 0
86 when 'SMALL' then (1..3).any? { |start| (start...start + 4).all? { |v| faces.include?(v) } } ? 30 : 0
87 when 'LARGE' then (1..2).any? { |start| (start...start + 5).all? { |v| faces.include?(v) } } ? 40 : 0
88 when 'YAHTZEE' then yahtzee?(faces) ? 50 : 0
89 else total
90 end
91end

93def best_candidate(candidates)
94 candidates.max_by { |_, score| score }
95end

97def most_frequent_face(faces)
98 FACES.max_by { |face| [faces.count(face), face] }
99end

101def bot_holds(strategy, faces)
102 return faces.map { |face| face >= 5 } if strategy == 'HIGH'
103 if strategy == 'STRAIGHT'
104 best_start = 0
105 best_length = 0
106 FACES.each do |start|
107 length = 0
108 length += 1 while faces.include?(start + length)
109 if length.positive? && length >= best_length
110 best_start, best_length = start, length
111 end
112 end
113 if best_length >= 3
114 holds = Array.new(5, false)
115 (best_start...best_start + best_length).each { |v| holds[faces.index(v)] = true }
116 return holds
117 end
118 end
119 target = most_frequent_face(faces)
120 faces.map { |face| face == target }
121end
122
123REROLLS = (0..5).map do |count|
124 FACES.repeated_combination(count).map do |rolled|
125 weight = FACTORIALS[count]
126 FACES.each { |face| weight /= FACTORIALS[rolled.count(face)] }
127 [rolled, weight]
128 end
129end.freeze

131class DiceRoller
132 attr_accessor :loaded
133 attr_reader :counts
134 def initialize(seed)
135 @state = seed
136 @loaded = []
137 @counts = Array.new(6, 0)
138 end
139
140 def roll_die
141 face = if @loaded.empty?
142 @state = MULTIPLIER * @state % MODULUS
143 @state % 6 + 1
144 else
145 @loaded.shift
146 end
147 @counts[face - 1] += 1
148 face
149 end
150end

201 attr_accessor :turn
202 attr_reader :players, :rules, :roller, :entries, :redo_list
203 def initialize(players, rules, roller)
204 @players, @rules, @roller = players, rules, roller
205 @turn = Turn.new(0)

207 end

217 def ranking

221 def roll
222 if @turn.rolls.zero?
223 @turn.faces = Array.new(5) { @roller.roll_die }
224 else

238 def joker_situation?(player, faces)
239 @rules[:joker] != 'OFF' && yahtzee?(faces) && !player.boxes['YAHTZEE'].nil?
240 end
241
242 def allowed_categories(player, faces, joker)
243 open = player.open_categories
244 return open unless joker && @rules[:joker] == 'FORCED'
245 upper = UPPER[faces[0] - 1]
246 return [upper] if player.boxes[upper].nil?
247 lower = open.select { |category| LOWER.include?(category) }
248 lower.empty? ? open : lower
249 end

251 def points_to_enter(faces, category, joker)
252 joker && JOKER_POINTS.key?(category) ? JOKER_POINTS.fetch(category) : normal_score(category, faces)
253 end
254
255 def candidates(player, faces)
256 joker = joker_situation?(player, faces)
257 allowed = allowed_categories(player, faces, joker)
258 player.open_categories.map do |category|
259 [category, allowed.include?(category) ? points_to_enter(faces, category, joker) : 0]
260 end
261 end
262
263 def best(player, faces)
264 best_candidate(candidates(player, faces))
265 end

267 def prepare_entry(category)
268 player, faces = current_player, @turn.faces
269 require_value(player.boxes[category].nil?, "#{category} already used")
270 joker = joker_situation?(player, faces)
271 unless allowed_categories(player, faces, joker).include?(category)
272 upper = UPPER[faces[0] - 1]
273 raise CommandError, player.boxes[upper].nil? ? "joker requires #{upper}" : 'joker requires lower box'
274 end
275 points = points_to_enter(faces, category, joker)
276 bonus = yahtzee?(faces) && player.boxes['YAHTZEE'] == 50 ? 100 : 0
277 ScoreEntry.new(player, category, points, bonus, joker, @turn.duplicate)
278 end

280 def score_line(entry, verb)
281 player = entry.player
282 parts = [player.name, verb, entry.points, 'in', entry.category]
283 parts << 'joker' if entry.joker
284 parts.concat(['ybonus', entry.bonus]) if entry.bonus.positive?
285 subtotal = player.upper_subtotal + (UPPER.include?(entry.category) ? entry.points : 0)
286 gain = player.upper_bonus.zero? && subtotal >= 63 ? 35 : 0
287 parts.concat(['ubonus', gain]) if gain.positive?
288 parts.concat(['total', player.total + entry.points + entry.bonus + gain])
289 parts.join(' ')
290 end
291
292 def enter(entry)
293 entry.player.boxes[entry.category] = entry.points
294 entry.player.yahtzee_bonus += entry.bonus
295 @entries << entry
296 end
297
298 def score(category)
299 entry = prepare_entry(category)
300 line = score_line(entry, 'scores')
301 enter(entry)
302 @redo_list.clear
303 if @players.all? { |player| player.used_count == 13 }
304 @turn = nil
305 return [line, 'game over']
306 end
307 @turn = Turn.new((@turn.seat + 1) % @players.length)
308 [line, "turn #{current_player.round} #{current_player.name}"]
309 end
310
311 def undo
312 entry = @entries.pop
313 entry.player.boxes[entry.category] = nil
314 entry.player.yahtzee_bonus -= entry.bonus
315 @redo_list << [@turn, entry]
316 @turn = entry.turn.duplicate
317 entry
318 end
319
320 def redo
321 turn, entry = @redo_list.pop
322 enter(entry)
323 @turn = turn
324 entry
325 end

327 def bot_category
328 player, faces = current_player, @turn.faces
329 if player.strategy == 'UPPER'
330 face = most_frequent_face(faces)
331 category = UPPER[face - 1]
332 allowed = allowed_categories(player, faces, joker_situation?(player, faces))
333 return category if faces.count(face) >= 3 && allowed.include?(category)
334 end
335 best(player, faces)[0]
336 end
337
338 def bot_turn
339 lines = []
340 lines << roll if @turn.rolls.zero?
341 while @turn.rolls < @rules[:rolls]
342 holds = bot_holds(current_player.strategy, @turn.faces)
343 break if holds.all?
344 lines << set_holds(holds)
345 lines << roll
346 end
347 lines.concat(score(bot_category))
348 end
349end

351class Advisor
352 def initialize(game, player)
353 @game, @player = game, player
354 @values, @expectations = {}, {}
355 end
356
357 def value(hand, level)
358 key = [level, hand.join]
359 return @values.fetch(key) if @values.key?(key)
360 score = if level.zero?
361 @game.best(@player, hand)[1]
362 else
363 PATTERNS.map { |holds| expectation(held_faces(hand, holds).sort, level) }.max
364 end
365 @values[key] = score
366 end
367
368 def expectation(held, level)
369 key = [level, held.join]
370 return @expectations.fetch(key) if @expectations.key?(key)
371 total = REROLLS[5 - held.length].sum do |rolled, weight|
372 weight * value((held + rolled).sort, level - 1)
373 end
374 @expectations[key] = total * 6**held.length
375 end
376
377 def best_hold(faces, levels)
378 best_holds, best_value = nil, -1
379 PATTERNS.each do |holds|
380 value = expectation(held_faces(faces, holds).sort, levels)
381 best_holds, best_value = holds, value if value > best_value
382 end
383 [best_holds, best_value]
384 end
385end

559 def odds(args)
560 holds = hold_pattern(args[0])
561 game = require_rolls_left
562 player = game.current_player
563 held = held_faces(game.turn.faces, holds)
564 open = player.open_categories
565 expected = open.to_h { |c| [c, 0] }
566 chances = open.to_h { |c| [c, 0] }
567 best_total = 0
568 REROLLS[5 - held.length].each do |rolled, weight|
569 candidates = game.candidates(player, held + rolled)
570 candidates.each do |category, score|
571 expected[category] += weight * score
572 chances[category] += weight if score.positive?
573 end
574 best_total += weight * best_candidate(candidates)[1]
575 end
576 outcomes = 6**(5 - held.length)
577 emit("odds #{pattern_text(holds)} outcomes #{outcomes}")
578 open.each do |category|
579 emit("#{category} expect #{fraction_text(expected[category], outcomes)} " \
580 "chance #{fraction_text(chances[category], outcomes)}")
581 end
582 emit("best expect #{fraction_text(best_total, outcomes)}")
583 end

699 def simulate(args)
700 seed = integer_word(args[0], 1, MODULUS - 1)
701 count = integer_word(args[1], 1, 50)
702 strategies = args.drop(2).map { |strategy| strategy_name(strategy) }
703 require_value(strategies.length <= 6, 'too many players')
704 roller = DiceRoller.new(seed)
705 summaries = strategies.map { |strategy| SeatSummary.new(strategy) }

735 def execute(line)
736 command, *args = line.split
737 handler = HANDLERS[command]
738 require_value(!handler.nil?, "unknown command #{command}")

741 else
742 send(handler)
743 end

746 end
