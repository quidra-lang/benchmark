29func validNumber(word string, lo, hi int) (int, string) {
30 if word == "-0" || !integerSyntax.MatchString(word) {
31 return 0, "bad value " + word
32 }
33 value, err := strconv.ParseInt(word, 10, 64)
34 if err != nil || value < int64(lo) || value > int64(hi) {
35 return 0, "bad value " + word
36 }
37 return int(value), ""
38}

97func gcd(a, b int64) int64 {

103func fractionText(a, b int64) string {
104 d := gcd(a, b)
105 a /= d
106 b /= d
107 if b == 1 {
108 return strconv.FormatInt(a, 10)
109 }
110 return strconv.FormatInt(a, 10) + "/" + strconv.FormatInt(b, 10)
111}

127func normalScore(category string, faces [5]int) int {
128 var counts [6]int
129 sum := 0
130 for _, face := range faces {
131 counts[face-1]++
132 sum += face
133 }
134 for f := 0; f < 6; f++ {
135 if categories[f] == category {
136 return (f + 1) * counts[f]
137 }
138 }
139 maxCount := 0
140 for _, count := range counts {
141 if count > maxCount {
142 maxCount = count
143 }
144 }
145 switch category {
146 case "THREEKIND":
147 if maxCount >= 3 {
148 return sum
149 }
150 case "FOURKIND":
151 if maxCount >= 4 {
152 return sum
153 }
154 case "FULLHOUSE":
155 three, two := false, false
156 for _, count := range counts {
157 if count == 3 {
158 three = true
159 }
160 if count == 2 {
161 two = true
162 }
163 }
164 if three && two {
165 return 25
166 }
167 case "SMALL":
168 for start := 0; start < 3; start++ {
169 if counts[start] > 0 && counts[start+1] > 0 && counts[start+2] > 0 && counts[start+3] > 0 {
170 return 30
171 }
172 }
173 case "LARGE":
174 for start := 0; start < 2; start++ {
175 if counts[start] > 0 && counts[start+1] > 0 && counts[start+2] > 0 && counts[start+3] > 0 && counts[start+4] > 0 {
176 return 40
177 }
178 }
179 case "YAHTZEE":
180 if maxCount == 5 {
181 return 50
182 }
183 case "CHANCE":
184 return sum
185 }
186 return 0
187}
188func mostFrequent(faces [5]int) int {
189 var counts [6]int
190 for _, face := range faces {
191 counts[face-1]++
192 }
193 best := 1
194 for face := 2; face <= 6; face++ {
195 if counts[face-1] >= counts[best-1] {
196 best = face
197 }
198 }
199 return best
200}
201func botHolds(strategy string, faces [5]int) [5]bool {
202 var holds [5]bool
203 if strategy == "HIGH" {
204 for i, face := range faces {
205 holds[i] = face >= 5
206 }
207 return holds
208 }
209 if strategy == "STRAIGHT" {
210 var present [7]bool
211 for _, face := range faces {
212 present[face] = true
213 }
214 bestStart, bestLength := 0, 0
215 for start := 1; start <= 6; start++ {
216 length := 0
217 for start+length <= 6 && present[start+length] {
218 length++
219 }
220 if length > 0 && length >= bestLength {
221 bestStart, bestLength = start, length
222 }
223 }
224 if bestLength >= 3 {
225 for face := bestStart; face < bestStart+bestLength; face++ {
226 for i, actual := range faces {
227 if actual == face {
228 holds[i] = true
229 break
230 }
231 }
232 }
233 return holds
234 }
235 }
236 most := mostFrequent(faces)
237 for i, face := range faces {
238 holds[i] = face == most
239 }
240 return holds
241}
242func allHeld(holds [5]bool) bool {
243 for _, hold := range holds {
244 if !hold {
245 return false
246 }
247 }
248 return true
249}

251type Outcome struct {
252 Faces []int
253 Weight int64
254}
255
256var rerolls [6][]Outcome
257
258func init() {
259 factorial := []int64{1, 1, 2, 6, 24, 120}
260 for count := 0; count <= 5; count++ {
261 var visit func(minimum, remain int, faces []int)
262 visit = func(minimum, remain int, faces []int) {
263 if remain == 0 {
264 weight := factorial[count]
265 for face := 1; face <= 6; face++ {
266 frequency := 0
267 for _, v := range faces {
268 if v == face {
269 frequency++
270 }
271 }
272 weight /= factorial[frequency]
273 }
274 rerolls[count] = append(rerolls[count], Outcome{append([]int(nil), faces...), weight})
275 return
276 }
277 for face := minimum; face <= 6; face++ {
278 visit(face, remain-1, append(faces, face))
279 }
280 }
281 visit(1, count, nil)
282 }
283}

285type Roller struct {
286 state int64
287 loaded []int
288 counts [6]int
289}

291func newRoller(seed int) *Roller { return &Roller{state: int64(seed)} }
292func (r *Roller) rollDie() int {
293 var face int
294 if len(r.loaded) > 0 {
295 face = r.loaded[0]
296 r.loaded = r.loaded[1:]
297 } else {
298 r.state = multiplier * r.state % modulus
299 face = int(r.state%6) + 1
300 }
301 r.counts[face-1]++
302 return face
303}

377func (g *Game) finished() bool { return g.Turn == nil }

379func (g *Game) ranking() []*Player {
380 result := append([]*Player(nil), g.Players...)
381 seat := make(map[*Player]int, len(g.Players))
382 for i, p := range g.Players {
383 seat[p] = i
384 }
385 sort.Slice(result, func(i, j int) bool {
386 a, b := result[i], result[j]
387 if a.total() != b.total() {
388 return a.total() > b.total()
389 }
390 return seat[a] < seat[b]
391 })
392 return result
393}
394func (g *Game) roll() string {
395 t := g.Turn
396 if t.Rolls == 0 {
397 for i := range t.Faces {
398 t.Faces[i] = g.Roller.rollDie()
399 }
400 } else {
401 for i := range t.Faces {
402 if !t.Holds[i] {
403 t.Faces[i] = g.Roller.rollDie()
404 }
405 }
406 }
407 t.Rolls++
408 g.Redos = nil
409 return fmt.Sprintf("%s roll %d: %s", g.current().Name, t.Rolls, faceText(t.Faces[:]))
410}

416func (g *Game) jokerSituation(p *Player, faces [5]int) bool {

420func (g *Game) allowed(p *Player, faces [5]int, joker bool) []string {
421 open := p.open()

423 return open

425 upper := categories[faces[0]-1]
426 if _, used := p.Boxes[upper]; !used {
427 return []string{upper}
428 }
429 lower := make([]string, 0, 7)
430 for _, category := range open {
431 if slices.Contains(categories[6:], category) {
432 lower = append(lower, category)
433 }
434 }
435 if len(lower) > 0 {
436 return lower
437 }
438 return open
439}

459func (g *Game) candidates(p *Player, faces [5]int) []Candidate {

462 result := make([]Candidate, 0, 13)

464 score := 0

469 }
470 return result

482func (g *Game) prepare(category string) (Entry, string) {
483 p := g.current()
484 faces := g.Turn.Faces
485 if _, used := p.Boxes[category]; used {
486 return Entry{}, category + " already used"
487 }
488 joker := g.jokerSituation(p, faces)
489 if !slices.Contains(g.allowed(p, faces, joker), category) {
490 upper := categories[faces[0]-1]
491 if _, used := p.Boxes[upper]; !used {
492 return Entry{}, "joker requires " + upper
493 }
494 return Entry{}, "joker requires lower box"
495 }
496 bonus := 0
497 if isYahtzee(faces) && p.Boxes["YAHTZEE"] == 50 {
498 bonus = 100
499 }
500 return Entry{p, category, g.points(faces, category, joker), bonus, joker, *g.Turn}, ""
501}
502func (g *Game) scoreLine(e Entry, verb string) string {
503 p := e.Player
504 parts := []string{p.Name, verb, intText(e.Points), "in", e.Category}
505 if e.Joker {
506 parts = append(parts, "joker")
507 }
508 if e.Bonus > 0 {
509 parts = append(parts, "ybonus", intText(e.Bonus))
510 }
511 subtotal := p.upper()
512 if slices.Contains(categories[:6], e.Category) {
513 subtotal += e.Points
514 }
515 gained := 0
516 if p.upperBonus() == 0 && subtotal >= 63 {
517 gained = 35
518 parts = append(parts, "ubonus", "35")
519 }
520 parts = append(parts, "total", intText(p.total()+e.Points+e.Bonus+gained))
521 return strings.Join(parts, " ")
522}
523func (g *Game) enter(e Entry) {
524 e.Player.Boxes[e.Category] = e.Points
525 e.Player.YBonus += e.Bonus
526 g.Entries = append(g.Entries, e)
527}
528func (g *Game) score(e Entry) []string {
529 line := g.scoreLine(e, "scores")
530 g.enter(e)
531 g.Redos = nil
532 completed := true
533 for _, player := range g.Players {
534 if len(player.Boxes) != 13 {
535 completed = false
536 break
537 }
538 }
539 if completed {
540 g.Turn = nil
541 return []string{line, "game over"}
542 }
543 g.Turn = &Turn{Seat: (g.Turn.Seat + 1) % len(g.Players)}
544 p := g.current()
545 return []string{line, fmt.Sprintf("turn %d %s", p.round(), p.Name)}
546}
547func (g *Game) undo() Entry {
548 i := len(g.Entries) - 1
549 e := g.Entries[i]
550 g.Entries = g.Entries[:i]
551 delete(e.Player.Boxes, e.Category)
552 e.Player.YBonus -= e.Bonus
553 g.Redos = append(g.Redos, Redo{g.Turn, e})
554 restored := e.Turn
555 g.Turn = &restored
556 return e
557}
558func (g *Game) redo() Entry {
559 i := len(g.Redos) - 1
560 row := g.Redos[i]
561 g.Redos = g.Redos[:i]
562 g.enter(row.Entry)
563 g.Turn = row.Turn
564 return row.Entry
565}
566func (g *Game) botCategory() string {
567 p, faces := g.current(), g.Turn.Faces
568 if p.Strategy == "UPPER" {
569 face := mostFrequent(faces)
570 category := categories[face-1]
571 count := 0
572 for _, v := range faces {
573 if v == face {
574 count++
575 }
576 }
577 if count >= 3 && slices.Contains(g.allowed(p, faces, g.jokerSituation(p, faces)), category) {
578 return category
579 }
580 }
581 return g.best(p, faces).Category
582}
583func (g *Game) botTurn() []string {
584 result := make([]string, 0, 9)
585 if g.Turn.Rolls == 0 {
586 result = append(result, g.roll())
587 }
588 for g.Turn.Rolls < g.Rules.Rolls {
589 holds := botHolds(g.current().Strategy, g.Turn.Faces)
590 if allHeld(holds) {
591 break
592 }
593 result = append(result, g.setHolds(holds), g.roll())
594 }
595 entry, msg := g.prepare(g.botCategory())
596 if msg != "" {
597 panic(msg)
598 }
599 return append(result, g.score(entry)...)
600}

617func encodeFaces(faces []int) string {
618 buf := make([]byte, len(faces))
619 for i, f := range faces {
620 buf[i] = byte(f) + '0'
621 }
622 return string(buf)

624func holdForPattern(faces []int, pattern int) []int {
625 held := make([]int, 0, 5)
626 for i, face := range faces {
627 if pattern&(1<<(4-i)) != 0 {
628 held = append(held, face)
629 }
630 }
631 return held
632}
633func (a *Advisor) value(hand []int, level int) int64 {
634 key := intText(level) + ":" + encodeFaces(hand)
635 if v, ok := a.values[key]; ok {
636 return v
637 }
638 var value int64
639 if level == 0 {
640 var faces [5]int
641 copy(faces[:], hand)
642 value = int64(a.game.best(a.player, faces).Score)
643 } else {
644 for pattern := 0; pattern < 32; pattern++ {
645 expected := a.expectation(holdForPattern(hand, pattern), level)
646 if pattern == 0 || expected > value {
647 value = expected
648 }
649 }
650 }
651 a.values[key] = value
652 return value
653}
654func (a *Advisor) expectation(held []int, level int) int64 {
655 key := intText(level) + ":" + encodeFaces(held)
656 if v, ok := a.expectations[key]; ok {
657 return v
658 }
659 var total int64
660 for _, outcome := range rerolls[5-len(held)] {
661 combined := append(append([]int(nil), held...), outcome.Faces...)
662 sort.Ints(combined)
663 total += outcome.Weight * a.value(combined, level-1)
664 }
665 for range len(held) {
666 total *= 6
667 }
668 a.expectations[key] = total
669 return total
670}
671func (a *Advisor) bestHold(faces [5]int, level int) ([5]bool, int64) {
672 var best [5]bool
673 var bestValue int64 = -1
674 for pattern := 0; pattern < 32; pattern++ {
675 held := holdForPattern(faces[:], pattern)
676 value := a.expectation(held, level)
677 if value > bestValue {
678 bestValue = value
679 for i := 0; i < 5; i++ {
680 best[i] = pattern&(1<<(4-i)) != 0
681 }
682 }
683 }
684 return best, bestValue
685}

767func (t *Table) command(command string, args []string) []string {
768 switch command {
769 case "RULE":
770 rule, value := args[0], args[1]
771 if rule == "ROLLS" {
772 rolls, msg := validNumber(value, 1, 4)
773 if msg != "" {
774 return errorLine(msg)
775 }
776 if t.game != nil {
777 return errorLine("game in progress")
778 }
779 t.rules.Rolls = rolls
780 } else if rule == "JOKER" {
781 if value != "FORCED" && value != "FREE" && value != "OFF" {
782 return errorLine("bad value " + value)
783 }
784 if t.game != nil {
785 return errorLine("game in progress")
786 }
787 t.rules.Joker = value
788 } else {
789 return errorLine("unknown rule " + rule)
790 }
791 return []string{"rule " + rule + " " + value}
792 case "GAME":
793 seed, msg := validNumber(args[0], 1, int(modulus-1))
794 if msg != "" {
795 return errorLine(msg)
796 }
797 names := args[1:]
798 for _, name := range names {
799 if msg := validName(name); msg != "" {
800 return errorLine(msg)
801 }
802 }
803 if len(names) > 6 {
804 return errorLine("too many players")
805 }
806 if t.game != nil {
807 return errorLine("game in progress")
808 }
809 seen := make(map[string]bool)
810 for _, name := range names {
811 if seen[name] {
812 return errorLine("duplicate name " + name)
813 }
814 seen[name] = true
815 }
816 players := make([]*Player, len(names))
817 for i, name := range names {
818 players[i] = newPlayer(name, "KIND")
819 }
820 t.game = newGame(players, &t.rules, newRoller(seed))
821 return []string{fmt.Sprintf("game seed %d players %s", seed, strings.Join(names, " ")), "turn 1 " + names[0]}
822 case "BOT":
823 name, strategy := args[0], args[1]
824 if msg := validName(name); msg != "" {
825 return errorLine(msg)
826 }
827 if msg := validStrategy(strategy); msg != "" {
828 return errorLine(msg)
