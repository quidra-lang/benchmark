37 static long positive(String word, long maximum) {
38 if (word.charAt(0) < '1' || word.charAt(0) > '9') throw error("bad value " + word);
39 for (int i = 0; i < word.length(); i++) {
40 if (word.charAt(i) < '0' || word.charAt(i) > '9') throw error("bad value " + word);
41 }
42 long value = Long.parseLong(word);
43 if (value > maximum) throw error("bad value " + word);
44 return value;
45 }

64 static int pattern(String word) {
65 if (word.length() != 5) throw error("bad pattern " + word);
66 int mask = 0;
67 for (int i = 0; i < 5; i++) {
68 char c = word.charAt(i);
69 if (c != '0' && c != '1') throw error("bad pattern " + word);
70 mask = 2 * mask + c - '0';
71 }
72 return mask;
73 }

75 static String fraction(long numerator, long denominator) {
76 long a = numerator, b = denominator;
77 while (b != 0) { long rest = a % b; a = b; b = rest; }
78 numerator /= a;
79 denominator /= a;
80 return denominator == 1 ? Long.toString(numerator) : numerator + "/" + denominator;
81 }

85 return String.format(Locale.ROOT, "%d.%02d", value / 100, value % 100);

126 static int normalScore(int category, int[] counts) {
127 if (category < 6) return (category + 1) * counts[category];
128 int sum = 0, maximum = 0, bits = 0;
129 boolean pair = false, triple = false;
130 for (int i = 0; i < 6; i++) {
131 sum += (i + 1) * counts[i];
132 maximum = Math.max(maximum, counts[i]);
133 if (counts[i] > 0) bits |= 1 << i;
134 pair |= counts[i] == 2;
135 triple |= counts[i] == 3;
136 }
137 return switch (category) {
138 case 6 -> maximum >= 3 ? sum : 0;
139 case 7 -> maximum >= 4 ? sum : 0;
140 case 8 -> pair && triple ? 25 : 0;
141 case 9 -> (bits & 15) == 15 || (bits & 30) == 30 || (bits & 60) == 60 ? 30 : 0;
142 case 10 -> bits == 31 || bits == 62 ? 40 : 0;
143 case 11 -> maximum == 5 ? 50 : 0;
144 default -> sum;
145 };
146 }

154 Player(String name) { this.name = name; Arrays.fill(scores, -1); }
155 int upper() { return Arrays.stream(scores, 0, 6).filter(n -> n >= 0).sum(); }
156 int upperBonus() { return upper() >= 63 ? 35 : 0; }
157 int total() { return Arrays.stream(scores).filter(n -> n >= 0).sum() + upperBonus() + yahtzeeBonus; }
158 int round() { return (int) Arrays.stream(scores).filter(n -> n >= 0).count() + 1; }

161 record Candidates(int[] points, int allowed, int best, int yahtzeeFace, boolean joker) {}
162
163 static Candidates evaluate(Player player, int[] counts, JokerRule rule) {
164 int face = -1, allowed = 0;
165 for (int i = 0; i < 6; i++) if (counts[i] == 5) face = i;
166 for (int c = 0; c < 13; c++) if (player.scores[c] < 0) allowed |= 1 << c;
167 boolean joker = rule != JokerRule.OFF && face >= 0 && player.scores[11] >= 0;
168 if (joker && rule == JokerRule.FORCED) {
169 int upper = 1 << face;
170 if ((allowed & upper) != 0) allowed = upper;
171 else if ((allowed & 0x1fc0) != 0) allowed &= 0x1fc0;
172 }
173 int[] points = new int[13];
174 Arrays.fill(points, -1);
175 int best = -1;
176 for (int c = 0; c < 13; c++) {
177 if (player.scores[c] >= 0) continue;
178 points[c] = (allowed & (1 << c)) == 0 ? 0
179 : joker && c >= 8 && c <= 10 ? c == 8 ? 25 : c == 9 ? 30 : 40 : normalScore(c, counts);
180 if (best < 0 || points[c] > points[best]) best = c;
181 }
182 return new Candidates(points, allowed, best, face, joker);
183 }

185 static void fillOutcomes(List<Outcome> outcomes, int size, int face, int left, int code, int divisor) {
186 if (face == 6) {
187 if (left == 0) outcomes.add(new Outcome(code, FACTORIAL[size] / divisor));
188 return;
189 }
190 for (int n = 0; n <= left; n++) {
191 fillOutcomes(outcomes, size, face + 1, left - n, code + n * POWERS[face], divisor * FACTORIAL[n]);
192 }
193 }

195 static List<List<Outcome>> makeOutcomes() {
196 List<List<Outcome>> result = new ArrayList<>();
197 for (int size = 0; size <= 5; size++) {
198 List<Outcome> outcomes = new ArrayList<>();
199 fillOutcomes(outcomes, size, 0, size, 0, 1);
200 result.add(List.copyOf(outcomes));
201 }
202 return List.copyOf(result);
203 }

246 void roll() {
247 Turn before = checkTurn();
248 if (before.rolls() == rules.rolls()) throw error("no rolls left");
249 if (before.hold() == 31) throw error("all dice held");
250 redo.clear();
251 int[] faces = before.faces().clone();
252 for (int i = 0; i < 5; i++) {
253 if (before.rolls() > 0 && held(before.hold(), i)) continue;
254 if (!loaded.isEmpty()) faces[i] = loaded.removeFirst();
255 else {
256 randomState = 48271 * randomState % 2147483647;
257 faces[i] = (int) (randomState % 6) + 1;
258 }
259 faceCounts[faces[i] - 1]++;
260 }
261 turn = new Turn(before.seat(), before.rolls() + 1, faces, before.hold());
262 if (verbose) line("%s roll %d: %s", players.get(turn.seat()).name, turn.rolls(), faceText(faces, 31));
263 }

286 void score(int category, boolean preview) {
287 Turn before = rolled();
288 Player player = players.get(before.seat());
289 if (player.scores[category] >= 0) throw error(CATEGORIES[category] + " already used");
290 Candidates candidates = evaluate(player, faceCounts(before.faces()), rules.joker());
291 if ((candidates.allowed() & (1 << category)) == 0) {
292 throw error("joker requires " + (player.scores[candidates.yahtzeeFace()] < 0
293 ? CATEGORIES[candidates.yahtzeeFace()] : "lower box"));
294 }
295 int points = candidates.points()[category];
296 int bonus = candidates.yahtzeeFace() >= 0 && player.scores[11] == 50 ? 100 : 0;
297 int upper = player.upper();
298 int gainedUpper = category < 6 && upper < 63 && upper + points >= 63 ? 35 : 0;
299 if (verbose) {
300 OUT.printf(Locale.ROOT, "%s %s %d in %s", player.name, preview ? "would score" : "scores", points, CATEGORIES[category]);
301 if (candidates.joker()) OUT.print(" joker");
302 if (bonus != 0) OUT.print(" ybonus 100");
303 if (gainedUpper != 0) OUT.print(" ubonus 35");
304 line(" total %d", player.total() + points + bonus + gainedUpper);
305 }
306 if (preview) return;
307 redo.clear();
308 entries.add(new ScoreEntry(before, category, points, bonus, candidates.joker()));
309 player.scores[category] = points;
310 player.yahtzeeBonus += bonus;
311 turn = entries.size() == 13 * players.size() ? null : Turn.start((before.seat() + 1) % players.size());
312 if (verbose) {
313 if (turn == null) line("game over");
314 else { Player next = players.get(turn.seat()); line("turn %d %s", next.round(), next.name); }
315 }
316 }

341 int botHold(Strategy strategy) {
342 int[] counts = faceCounts(turn.faces());
343 int favorite = frequent(counts) + 1, mask = 0;
344 if (strategy == Strategy.STRAIGHT) {
345 int bestStart = 0, bestLength = 0;
346 for (int start = 0; start < 6; start++) {
347 int end = start;
348 while (end < 6 && counts[end] > 0) end++;
349 if (end - start >= bestLength) { bestStart = start; bestLength = end - start; }
350 }
351 if (bestLength >= 3) {
352 for (int face = bestStart + 1; face <= bestStart + bestLength; face++) {
353 for (int i = 0; i < 5; i++) if (turn.faces()[i] == face) { mask |= 1 << (4 - i); break; }
354 }
355 return mask;
356 }
357 }
358 for (int i = 0; i < 5; i++) {
359 if (strategy == Strategy.HIGH ? turn.faces()[i] >= 5 : turn.faces()[i] == favorite) mask |= 1 << (4 - i);
360 }
361 return mask;
362 }

364 void botTurn() {
365 Player player = players.get(turn.seat());
366 Strategy strategy = player.strategy;
367 if (turn.rolls() == 0) roll();
368 while (turn.rolls() < rules.rolls()) {
369 int mask = botHold(strategy);
370 if (mask == 31) break;
371 hold(mask);
372 roll();
373 }
374 int[] counts = faceCounts(turn.faces());
375 Candidates candidates = evaluate(player, counts, rules.joker());
376 int favorite = frequent(counts);
377 int category = strategy == Strategy.UPPER && counts[favorite] >= 3 && (candidates.allowed() & (1 << favorite)) != 0
378 ? favorite : candidates.best();
379 score(category, false);
380 }

389 void odds(int mask) {
390 Turn current = rollsLeft();
391 Player player = players.get(current.seat());
392 int code = heldCode(current.faces(), mask), rerolled = 5 - Integer.bitCount(mask);
393 long[] sums = new long[13], chances = new long[13];
394 long best = 0, denominator = POWERS[rerolled];
395 for (Outcome outcome : OUTCOMES.get(rerolled)) {
396 Candidates candidates = evaluate(player, decode(code + outcome.code()), rules.joker());
397 for (int c = 0; c < 13; c++) if (candidates.points()[c] >= 0) {
398 sums[c] += (long) outcome.weight() * candidates.points()[c];
399 if (candidates.points()[c] > 0) chances[c] += outcome.weight();
400 }
401 best += (long) outcome.weight() * candidates.points()[candidates.best()];
402 }
403 line("odds %s outcomes %d", patternText(mask), denominator);
404 for (int c = 0; c < 13; c++) if (player.scores[c] < 0) {
405 line("%s expect %s chance %s", CATEGORIES[c], fraction(sums[c], denominator), fraction(chances[c], denominator));
406 }
407 line("best expect %s", fraction(best, denominator));
408 }

413 Adviser adviser = new Adviser(players.get(current.seat()), rules.joker(), depth);
414 long best = -1, denominator = 1;

421 }

506 static final class Adviser {
507 final Player player;
508 final JokerRule joker;
509 final long[][] values, expectations;
510
511 Adviser(Player player, JokerRule joker, int depth) {
512 this.player = player;
513 this.joker = joker;
514 values = new long[depth + 1][CODES];
515 expectations = new long[depth + 1][CODES];
516 for (long[] row : values) Arrays.fill(row, -1);
517 for (long[] row : expectations) Arrays.fill(row, -1);
518 }
519
520 long value(int code, int depth) {
521 if (values[depth][code] >= 0) return values[depth][code];
522 int[] counts = decode(code);
523 long value;
524 if (depth == 0) {
525 Candidates candidates = evaluate(player, counts, joker);
526 value = candidates.points()[candidates.best()];
527 } else value = searchHolds(counts, 0, 0, depth);
528 return values[depth][code] = value;
529 }
530
531 long searchHolds(int[] counts, int face, int code, int depth) {
532 if (face == 6) return expected(code, depth);
533 long best = -1;
534 for (int n = 0; n <= counts[face]; n++) best = Math.max(best, searchHolds(counts, face + 1, code + n * POWERS[face], depth));
535 return best;
536 }
537
538 long expected(int code, int depth) {
539 if (expectations[depth][code] >= 0) return expectations[depth][code];
540 int kept = Arrays.stream(decode(code)).sum();
541 long sum = 0;
542 for (Outcome outcome : OUTCOMES.get(5 - kept)) sum += outcome.weight() * value(code + outcome.code(), depth - 1);
543 return expectations[depth][code] = sum * POWERS[kept];
544 }
545 }

559 void simulate(long seed, int count, Strategy[] bots) {
560 int[] wins = new int[bots.length], best = new int[bots.length], bonus = new int[bots.length], yahtzee = new int[bots.length];
561 long[] sums = new long[bots.length];
562 long state = seed;
563 String[] names = Arrays.copyOf(new String[]{"A", "B", "C", "D", "E", "F"}, bots.length);
564 for (int n = 0; n < count; n++) {
565 Game trial = new Game(state, names, rules, false);
566 for (int i = 0; i < bots.length; i++) trial.players.get(i).strategy = bots[i];
567 while (trial.turn != null) trial.botTurn();
568 state = trial.randomState;
569 wins[trial.ranking().getFirst()]++;
570 for (int i = 0; i < bots.length; i++) {
571 Player player = trial.players.get(i);
572 int total = player.total();
573 sums[i] += total;
574 best[i] = Math.max(best[i], total);
575 if (player.upperBonus() == 35) bonus[i]++;
576 if (player.scores[11] == 50) yahtzee[i]++;
577 }
578 }
579 line("sim seed %d games %d players %d", seed, count, bots.length);
580 for (int i = 0; i < bots.length; i++) {
581 line("seat %d %s wins %d average %s best %d bonus %d yahtzee %d", i + 1, bots[i], wins[i], average(sums[i], count), best[i], bonus[i], yahtzee[i]);
582 }

585 void execute(String[] words) {
586 switch (words[0]) {
587 case "RULE" -> {
588 Rules next = switch (words[1]) {
589 case "ROLLS" -> new Rules((int) positive(words[2], 4), rules.joker());
590 case "JOKER" -> new Rules(rules.rolls(), switch (words[2]) {
591 case "FORCED" -> JokerRule.FORCED; case "FREE" -> JokerRule.FREE; case "OFF" -> JokerRule.OFF;
592 default -> throw error("bad value " + words[2]);
593 });
594 default -> throw error("unknown rule " + words[1]);
595 };
596 if (game != null) throw error("game in progress");
597 rules = next;
598 line("rule %s %s", words[1], words[2]);
599 }
600 case "GAME" -> {
601 long seed = positive(words[1], 2147483646);
602 String[] names = Arrays.copyOfRange(words, 2, words.length);
603 for (String name : names) validName(name);
604 if (names.length > 6) throw error("too many players");
605 if (game != null) throw error("game in progress");
606 for (int i = 0; i < names.length; i++) for (int j = 0; j < i; j++) {
607 if (names[i].equals(names[j])) throw error("duplicate name " + names[i]);
608 }
609 game = new Game(seed, names, rules, true);
610 line("game seed %d players %s", seed, String.join(" ", names));
611 line("turn 1 %s", names[0]);
612 }
613 case "BOT" -> {
614 validName(words[1]);
615 Strategy bot = strategy(words[2]);
616 Player player = requireGame().players.stream().filter(p -> p.name.equals(words[1])).findFirst()
617 .orElseThrow(() -> error("no player " + words[1]));
618 player.strategy = bot;
619 line("bot %s %s", words[1], words[2]);
620 }
621 case "LOAD", "KEEP" -> {
622 int[] values = new int[words.length - 1];
623 for (int i = 1; i < words.length; i++) values[i - 1] = (int) positive(words[i], 6);
624 Game current = requireGame();
625 if (words[0].equals("KEEP")) current.keep(values);
626 else { for (int value : values) current.loaded.addLast(value); line("loaded %d", current.loaded.size()); }
627 }
628 case "ROLL" -> requireGame().roll();
629 case "HOLD", "ODDS" -> {
630 int mask = pattern(words[1]);
631 if (words[0].equals("HOLD")) requireGame().hold(mask); else requireGame().odds(mask);
632 }
633 case "SCORE", "WHATIF" -> {
634 int category = Arrays.asList(CATEGORIES).indexOf(words[1]);
635 if (category < 0) throw error("bad category " + words[1]);
636 requireGame().score(category, words[0].equals("WHATIF"));
637 }
638 case "UNDO" -> requireGame().undo();
639 case "REDO" -> requireGame().redo();
640 case "SUGGEST" -> {
641 Game current = requireGame();
642 Turn turn = current.rolled();
643 Candidates candidates = evaluate(current.players.get(turn.seat()), faceCounts(turn.faces()), current.rules.joker());
644 OUT.print("suggest");
645 for (int c = 0; c < 13; c++) if (candidates.points()[c] >= 0) OUT.printf(Locale.ROOT, " %s %d", CATEGORIES[c], candidates.points()[c]);
646 line(" best %s %d", CATEGORIES[candidates.best()], candidates.points()[candidates.best()]);
647 }
648 case "ADVISE" -> requireGame().advise();
649 case "AUTO" -> {
650 int count = (int) positive(words[1], 100);
651 Game current = requireGame();
652 current.checkTurn();
653 current.redo.clear();
654 for (int i = 0; i < count && current.turn != null; i++) current.botTurn();
655 }
656 case "STATUS", "CARD", "BONUS", "STATS", "LOG", "FACES", "STANDINGS" -> requireGame().report(words[0]);
657 case "END" -> {
658 Game current = requireGame();
659 line("end %s", current.turn == null ? "final" : "abandoned");
660 List<Integer> order = current.ranking();
661 for (int rank = 0; rank < order.size(); rank++) {
662 Player player = current.players.get(order.get(rank));
663 int total = player.total();
664 line("rank %d %s %d", rank + 1, player.name, total);
665 if (current.turn == null) {
666 PlayerRecord record = records.computeIfAbsent(player.name, name -> new PlayerRecord());
667 record.games++;
668 if (rank == 0) record.wins++;
669 record.sum += total;
670 record.best = Math.max(record.best, total);
671 }
672 }
673 game = null;
674 }
675 case "RECORDS" -> {
676 List<Map.Entry<String, PlayerRecord>> order = new ArrayList<>(records.entrySet());
677 order.sort(Comparator.<Map.Entry<String, PlayerRecord>>comparingInt(e -> e.getValue().wins).reversed()
678 .thenComparing(Comparator.<Map.Entry<String, PlayerRecord>>comparingInt(e -> e.getValue().best).reversed())
679 .thenComparing(Map.Entry::getKey));
680 for (var item : order) {
681 PlayerRecord record = item.getValue();
682 line("record %s games %d wins %d best %d average %s", item.getKey(), record.games, record.wins, record.best, average(record.sum, record.games));
683 }
684 if (records.isEmpty()) line("no records");
685 }
686 case "SIM" -> {
687 long seed = positive(words[1], 2147483646);
688 int count = (int) positive(words[2], 50);
689 Strategy[] bots = new Strategy[words.length - 3];
690 for (int i = 3; i < words.length; i++) bots[i - 3] = strategy(words[i]);
691 if (bots.length > 6) throw error("too many players");
692 simulate(seed, count, bots);
693 }
694 default -> throw error("unknown command " + words[0]);
695 }
696 }
697 }
