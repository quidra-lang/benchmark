17integer_word <- function(text, minimum, maximum) {
18 if (!grepl("^-?(0|[1-9][0-9]*)$", text, perl = TRUE) || text == "-0") fail(paste("bad value", text))
19 value <- as.bigz(text)

21 as.integer(as.character(value))

47fraction <- function(numerator, denominator) {
48 numerator <- as.bigz(numerator)
49 denominator <- as.bigz(denominator)
50 divisor <- gcd.bigz(numerator, denominator)
51 numerator <- numerator %/% divisor
52 denominator <- denominator %/% divisor
53 if (denominator == 1L) as.character(numerator) else paste0(as.character(numerator), "/", as.character(denominator))
54}

66candidates <- function(card, counts, joker_rule) {
67 allowed <- card < 0L
68 joker <- joker_rule != "OFF" && any(counts == 5L) && card[12L] >= 0L
69 requirement <- ""
70 if (joker && joker_rule == "FORCED") {
71 face <- most_frequent(counts)
72 if (card[face] < 0L) {
73 allowed[] <- FALSE
74 allowed[face] <- TRUE
75 requirement <- categories[face]
76 } else if (any(allowed[7:13])) {
77 allowed[1:6] <- FALSE
78 requirement <- "lower box"
79 }
80 }
81 upper <- seq_len(6L) * counts
82 total <- sum(upper)
83 maximum <- max(counts)
84 small <- any(vapply(1:3, function(start) all(counts[start:(start + 3L)] > 0L), logical(1L)))
85 large <- any(vapply(1:2, function(start) all(counts[start:(start + 4L)] > 0L), logical(1L)))
86 points <- c(upper, if (maximum >= 3L) total else 0L, if (maximum >= 4L) total else 0L,
87 if (any(counts == 3L) && any(counts == 2L)) 25L else 0L,
88 if (small) 30L else 0L, if (large) 40L else 0L,
89 if (maximum == 5L) 50L else 0L, total)
90 if (joker) points[9:11] <- c(25L, 30L, 40L)
91 points[!allowed] <- 0L
92 open <- which(card < 0L)
93 list(allowed = allowed, points = points, best = open[which.max(points[open])],
94 joker = joker, requirement = requirement)
95}

131roll_dice <- function(game) {
132 if (game$turn$rolls == 0L) game$turn$dice <- integer(5L)
133 for (die in which(!game$turn$holds)) {
134 if (game$loaded_head <= length(game$loaded)) {
135 face <- game$loaded[game$loaded_head]
136 game$loaded_head <- game$loaded_head + 1L
137 } else {
138 game$state <- (game$state * 48271L) %% 2147483647L
139 face <- as.integer(as.character(game$state %% 6L)) + 1L
140 }
141 game$turn$dice[die] <- face
142 game$face_totals[face] <- game$face_totals[face] + 1L
143 }
144 game$turn$rolls <- game$turn$rolls + 1L
145 game$redo <- list()
146 game_emit(game, game$names[game$turn$player], "roll", paste0(game$turn$rolls, ":"), faces_text(game$turn$dice))
147 game
148}

175score_text <- function(game, entry, hypothetical = FALSE) {
176 player <- entry$turn$player
177 upper <- upper_subtotals(game)[player]
178 after <- upper + (if (entry$category <= 6L) entry$points else 0L)
179 added <- if (upper < 63L && after >= 63L) 35L else 0L
180 text <- paste(game$names[player], if (hypothetical) "would score" else "scores",
181 entry$points, "in", categories[entry$category])
182 if (entry$joker) text <- paste(text, "joker")
183 if (entry$ybonus) text <- paste(text, "ybonus 100")
184 if (added) text <- paste(text, "ubonus 35")
185 paste(text, "total", player_totals(game)[player] + entry$points + entry$ybonus + added)
186}

202bot_holds <- function(strategy, dice) {
203 if (strategy == "HIGH") return(dice >= 5L)
204 counts <- face_counts(dice)
205 if (strategy == "STRAIGHT") {
206 best_start <- best_length <- 0L
207 for (start in seq_len(6L)) {
208 run <- 0L
209 while (start + run <= 6L && counts[start + run] > 0L) run <- run + 1L
210 if (run >= best_length) {
211 best_start <- start
212 best_length <- run
213 }
214 }
215 if (best_length >= 3L) {
216 holds <- rep(FALSE, 5L)
217 holds[match(seq.int(best_start, length.out = best_length), dice)] <- TRUE
218 return(holds)
219 }
220 }
221 dice == most_frequent(counts)
222}

224bot_turn <- function(game) {
225 if (game$turn$rolls == 0L) game <- roll_dice(game)
226 strategy <- game$bots[game$turn$player]
227 while (game$turn$rolls < game$rules$rolls) {
228 holds <- bot_holds(strategy, game$turn$dice)
229 if (all(holds)) break
230 game <- roll_dice(set_holds(game, holds))
231 }
232 counts <- face_counts(game$turn$dice)
233 options <- candidates(game$cards[game$turn$player, ], counts, game$rules$joker)
234 face <- most_frequent(counts)
235 category <- if (strategy == "UPPER" && counts[face] >= 3L && options$allowed[face]) face else options$best
236 enter_score(game, category)
237}

239compositions <- function(total, faces = 6L) {
240 if (faces == 1L) return(matrix(total, 1L, 1L))
241 rows <- lapply(seq.int(0L, total), function(count) {
242 suffix <- compositions(total - count, faces - 1L)
243 cbind(count, suffix)
244 })
245 do.call(rbind, rows)

247count_key <- function(counts) sum(counts * c(7776L, 1296L, 216L, 36L, 6L, 1L)) + 1L

249make_tables <- function() {
250 outcomes <- lapply(0:5, function(total) {
251 counts <- compositions(total)
252 weights <- c_bigz(lapply(seq_len(nrow(counts)), function(row) {
253 factorialZ(total) %/% prod(factorialZ(counts[row, ]))
254 }))
255 list(counts = counts, weights = weights)
256 })
257 full <- outcomes[[6L]]$counts
258 held <- do.call(rbind, lapply(outcomes, `[[`, "counts"))
259 full_lookup <- held_lookup <- rep(NA_integer_, 46656L)
260 full_lookup[vapply(seq_len(nrow(full)), function(row) count_key(full[row, ]), integer(1L))] <- seq_len(nrow(full))
261 held_lookup[vapply(seq_len(nrow(held)), function(row) count_key(held[row, ]), integer(1L))] <- seq_len(nrow(held))
262 next_states <- weights <- vector("list", nrow(held))
263 for (row in seq_len(nrow(held))) {
264 kept <- sum(held[row, ])
265 rolled <- outcomes[[6L - kept]]
266 next_states[[row]] <- vapply(seq_len(nrow(rolled$counts)), function(index) {
267 full_lookup[count_key(held[row, ] + rolled$counts[index, ])]
268 }, integer(1L))
269 weights[[row]] <- rolled$weights * as.bigz(6L)^kept
270 }
271 subsets <- lapply(seq_len(nrow(full)), function(row) {
272 which(rowSums(sweep(held, 2L, full[row, ], FUN = `<=`)) == 6L)
273 })
274 patterns <- t(vapply(0:31, function(mask) {
275 bitwAnd(mask, c(16L, 8L, 4L, 2L, 1L)) != 0L
276 }, logical(5L)))
277 list(outcomes = outcomes, full = full, held = held, held_lookup = held_lookup,
278 next_states = next_states, weights = weights, subsets = subsets, patterns = patterns)
279}

281print_odds <- function(game, tables, holds) {
282 held <- face_counts(game$turn$dice[holds])
283 rolled <- 5L - sum(held)
284 distribution <- tables$outcomes[[rolled + 1L]]
285 denominator <- as.bigz(6L)^rolled
286 totals <- chances <- rep(zero, 13L)
287 best_total <- zero
288 card <- game$cards[game$turn$player, ]
289 for (row in seq_len(nrow(distribution$counts))) {
290 options <- candidates(card, held + distribution$counts[row, ], game$rules$joker)
291 weight <- distribution$weights[row]
292 totals <- totals + as.bigz(options$points) * weight
293 chances <- chances + as.bigz(as.integer(options$points > 0L)) * weight
294 best_total <- best_total + weight * options$points[options$best]
295 }
296 emit("odds", pattern_text(holds), "outcomes", as.character(denominator))
297 for (category in which(card < 0L)) {
298 emit(categories[category], "expect", fraction(totals[category], denominator),
299 "chance", fraction(chances[category], denominator))
300 }
301 emit("best expect", fraction(best_total, denominator))
302}

304print_advice <- function(game, tables) {
305 card <- game$cards[game$turn$player, ]
306 values <- as.bigz(vapply(seq_len(nrow(tables$full)), function(row) {
307 options <- candidates(card, tables$full[row, ], game$rules$joker)
308 as.integer(options$points[options$best])
309 }, integer(1L)))
310 remaining <- game$rules$rolls - game$turn$rolls
311 expected <- rep(zero, nrow(tables$held))
312 for (depth in seq_len(remaining)) {
313 for (index in seq_len(nrow(tables$held))) {
314 expected[index] <- sum(tables$weights[[index]] * values[tables$next_states[[index]]])
315 }
316 if (depth < remaining) {
317 for (row in seq_len(nrow(tables$full))) values[row] <- max(expected[tables$subsets[[row]]])
318 }
319 }
320 choices <- rep(zero, 32L)
321 for (index in seq_len(32L)) {
322 held <- face_counts(game$turn$dice[tables$patterns[index, ]])
323 choices[index] <- expected[tables$held_lookup[count_key(held)]]
324 }
325 index <- which.max(choices)
326 holds <- tables$patterns[index, ]
327 emit("advise", paste0(pattern_text(holds), ":"), faces_text(game$turn$dice[holds]),
328 "expect", fraction(choices[index], as.bigz(7776L)^remaining))
329}

391simulate <- function(seed, count, bots, rules) {
392 seats <- seq_along(bots)
393 totals <- matrix(0L, length(bots), 5L)
394 generator <- as.bigz(seed)
395 for (game_number in seq_len(count)) {
396 game <- new_game(generator, LETTERS[seats], rules, bots, TRUE)
397 while (game$turn$player != 0L) game <- bot_turn(game)
398 generator <- game$state
399 scores <- player_totals(game)
400 winner <- ranking(game)[1L]
401 totals[, 1L] <- totals[, 1L] + as.integer(seats == winner)
402 totals[, 2L] <- totals[, 2L] + scores
403 totals[, 3L] <- pmax(totals[, 3L], scores)
404 totals[, 4L] <- totals[, 4L] + as.integer(upper_subtotals(game) >= 63L)
405 totals[, 5L] <- totals[, 5L] + as.integer(game$cards[, 12L] == 50L)
406 }
407 emit("sim seed", seed, "games", count, "players", length(bots))
408 for (seat in seats) {
409 emit("seat", seat, bots[seat], "wins", totals[seat, 1L], "average", average(totals[seat, 2L], count),
410 "best", totals[seat, 3L], "bonus", totals[seat, 4L], "yahtzee", totals[seat, 5L])
411 }

414game_command <- function(app, tokens) {
415 op <- tokens[1L]
416 if (op == "BOT") {
417 name <- player_name(tokens[2L])
418 strategy <- strategy_word(tokens[3L])
419 } else if (op %in% c("LOAD", "KEEP")) {
420 faces <- face_words(tokens[-1L])
421 } else if (op %in% c("HOLD", "ODDS")) {
422 holds <- hold_pattern(tokens[2L])
423 } else if (op %in% c("SCORE", "WHATIF")) {
424 category <- category_word(tokens[2L])
425 } else if (op == "AUTO") {
426 count <- integer_word(tokens[2L], 1L, 100L)
427 }
428 game <- require_game(app)
429 if (op == "BOT") {
430 player <- match(name, game$names)
431 if (is.na(player)) fail(paste("no player", name))
432 game$bots[player] <- strategy
433 emit("bot", name, strategy)
434 } else if (op == "LOAD") {
435 game$loaded <- c(game$loaded, faces)
436 emit("loaded", length(game$loaded) - game$loaded_head + 1L)
437 } else if (op == "ROLL") {
438 turn_check(game)
439 if (game$turn$rolls == game$rules$rolls) fail("no rolls left")
440 if (all(game$turn$holds)) fail("all dice held")
441 game <- roll_dice(game)
442 } else if (op == "HOLD") {
443 rolls_left_check(game)
444 game <- set_holds(game, holds)
445 } else if (op == "KEEP") {
446 rolls_left_check(game)
447 holds <- rep(FALSE, 5L)
448 for (face in faces) {
449 available <- which(!holds & game$turn$dice == face)
450 if (!length(available)) fail(paste("not showing", face))
451 holds[available[1L]] <- TRUE
452 }
453 game <- set_holds(game, holds)
454 } else if (op %in% c("SCORE", "WHATIF")) {
455 score_check(game, category)
456 if (op == "WHATIF") {
457 emit(score_text(game, score_details(game, category), TRUE))
458 } else game <- enter_score(game, category)
459 } else if (op == "UNDO") {
460 if (!length(game$entries)) fail("nothing to undo")
461 entry <- game$entries[[length(game$entries)]]
462 game$redo[[length(game$redo) + 1L]] <- list(resume = game$turn, entry = entry)
463 game$entries <- game$entries[-length(game$entries)]
464 game$cards[entry$turn$player, entry$category] <- -1L
465 game$ybonus[entry$turn$player] <- game$ybonus[entry$turn$player] - entry$ybonus
466 game$turn <- entry$turn
467 emit("undo", game$names[entry$turn$player], categories[entry$category], entry$points)
468 } else if (op == "REDO") {
469 if (!length(game$redo)) fail("nothing to redo")
470 saved <- game$redo[[length(game$redo)]]
471 game$redo <- game$redo[-length(game$redo)]
472 entry <- saved$entry
473 game$cards[entry$turn$player, entry$category] <- entry$points
474 game$ybonus[entry$turn$player] <- game$ybonus[entry$turn$player] + entry$ybonus
475 game$entries[[length(game$entries) + 1L]] <- entry
476 game$turn <- saved$resume
477 emit("redo", game$names[entry$turn$player], categories[entry$category], entry$points)
478 } else if (op == "SUGGEST") {
479 rolled_check(game)
480 card <- game$cards[game$turn$player, ]
481 options <- candidates(card, face_counts(game$turn$dice), game$rules$joker)
482 open <- which(card < 0L)
483 emit("suggest", paste(categories[open], options$points[open], collapse = " "),
484 "best", categories[options$best], options$points[options$best])
485 } else if (op == "ODDS") {
486 rolls_left_check(game)
487 print_odds(game, app$tables, holds)
488 } else if (op == "ADVISE") {
489 rolls_left_check(game)
490 print_advice(game, app$tables)
491 } else if (op == "AUTO") {
492 turn_check(game)
493 game$redo <- list()
494 for (iteration in seq_len(count)) {
495 if (game$turn$player == 0L) break
496 game <- bot_turn(game)
497 }
498 } else if (op == "STATUS") {
499 player <- game$turn$player
500 if (player == 0L) {
501 emit("status game over")
502 } else {
503 emit("status round", player_round(game, player), "player", game$names[player],
504 "rolls", game$turn$rolls, "dice", faces_text(game$turn$dice), "holds", pattern_text(game$turn$holds))
505 }
506 } else if (op == "CARD") {
507 print_card(game)
508 } else if (op == "BONUS") {
509 upper <- upper_subtotals(game)
510 for (player in seq_along(game$names)) {
511 maximum <- upper[player] + sum(5L * (1:6)[game$cards[player, 1:6] < 0L])
512 status <- if (upper[player] >= 63L) "made" else if (maximum < 63L) "lost" else "open"
513 emit("bonus", game$names[player], "upper", upper[player], "need", max(63L - upper[player], 0L),
514 "max", maximum, status)
515 }
516 } else if (op == "STATS") {
517 print_stats(game)
518 } else if (op == "LOG") {
519 if (!length(game$entries)) emit("no scores")
520 for (index in seq_along(game$entries)) {
521 entry <- game$entries[[index]]
522 text <- paste("log", index, game$names[entry$turn$player], faces_text(entry$turn$dice),
523 categories[entry$category], entry$points)
524 if (entry$joker) text <- paste(text, "joker")
525 if (entry$ybonus) text <- paste(text, "ybonus 100")
526 emit(text)
527 }
528 } else if (op == "FACES") {
529 counts <- as.bigz(game$face_totals)
530 total <- sum(counts)
531 chi <- if (total == 0L) "0" else fraction(6L * sum(counts * counts) - total * total, total)
532 emit("faces", faces_text(game$face_totals), "total", as.character(total), "chi2", chi)
533 } else if (op %in% c("STANDINGS", "END")) {
534 order <- ranking(game)
535 totals <- player_totals(game)
536 finished <- game$turn$player == 0L
537 if (op == "END") emit("end", if (finished) "final" else "abandoned")
538 for (rank in seq_along(order)) {
539 player <- order[rank]
540 if (op == "STANDINGS") {
541 emit("standing", rank, game$names[player], totals[player], "behind", totals[order[1L]] - totals[player])
542 } else emit("rank", rank, game$names[player], totals[player])
543 }
544 if (op == "END") {
545 if (finished) app$records <- record_game(app$records, game)
546 app$game <- NULL
547 return(invisible(NULL))
548 }
549 }
550 app$game <- game
551 invisible(NULL)
552}

554execute <- function(app, tokens) {
555 op <- tokens[1L]
556 if (!op %in% commands) fail(paste("unknown command", op))
557 if (op == "RULE") {
558 rule <- tokens[2L]
559 value <- tokens[3L]
560 if (!rule %in% c("ROLLS", "JOKER")) fail(paste("unknown rule", rule))
561 changed <- app$rules
562 if (rule == "ROLLS") {
563 changed$rolls <- integer_word(value, 1L, 4L)
564 } else {
565 if (!value %in% c("FORCED", "FREE", "OFF")) fail(paste("bad value", value))
566 changed$joker <- value
567 }
568 if (!is.null(app$game)) fail("game in progress")
569 app$rules <- changed
570 emit("rule", rule, value)
571 } else if (op == "GAME") {
572 seed <- integer_word(tokens[2L], 1L, 2147483646L)
573 names <- tokens[-c(1L, 2L)]
574 for (name in names) player_name(name)
575 if (length(names) > 6L) fail("too many players")
576 if (!is.null(app$game)) fail("game in progress")
577 duplicates <- which(duplicated(names))
578 if (length(duplicates)) fail(paste("duplicate name", names[duplicates[1L]]))
579 app$game <- new_game(seed, names, app$rules)
580 emit("game seed", seed, "players", paste(names, collapse = " "))
581 emit("turn 1", names[1L])
582 } else if (op == "RECORDS") {
583 print_records(app$records)
584 } else if (op == "SIM") {
585 seed <- integer_word(tokens[2L], 1L, 2147483646L)
586 count <- integer_word(tokens[3L], 1L, 50L)
587 bots <- vapply(tokens[-c(1L, 2L, 3L)], strategy_word, character(1L))
588 if (length(bots) > 6L) fail("too many players")
589 simulate(seed, count, bots, app$rules)
590 } else game_command(app, tokens)
591 invisible(NULL)
592}

594input <- file("stdin", open = "r")
595command_count <- as.integer(readLines(input, n = 1L, warn = FALSE))
596app <- new.env(parent = emptyenv())
597app$rules <- list(rolls = 3L, joker = "FORCED")
598app$game <- NULL

600app$tables <- make_tables()
601for (command_number in seq_len(command_count)) {
602 tokens <- strsplit(readLines(input, n = 1L, warn = FALSE), " ", fixed = TRUE)[[1L]]
603 tryCatch(execute(app, tokens), command_error = function(problem) emit("error:", conditionMessage(problem)))
604}
