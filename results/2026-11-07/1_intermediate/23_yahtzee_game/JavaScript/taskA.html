3const FACES = [1, 2, 3, 4, 5, 6];
4const CATEGORIES = [
5 'ONES', 'TWOS', 'THREES', 'FOURS', 'FIVES', 'SIXES',
6 'THREEKIND', 'FOURKIND', 'FULLHOUSE', 'SMALL', 'LARGE', 'YAHTZEE', 'CHANCE'
7];
8const UPPER = CATEGORIES.slice(0, 6);
9const LOWER = CATEGORIES.slice(6);
10const STRATEGIES = new Set(['KIND', 'STRAIGHT', 'HIGH', 'UPPER']);
11const JOKER_MODES = new Set(['FORCED', 'FREE', 'OFF']);
12const JOKER_POINTS = { FULLHOUSE: 25, SMALL: 30, LARGE: 40 };
13const MODULUS = 2147483647;
14const MULTIPLIER = 48271;
15const PATTERNS = Array.from({ length: 32 }, (_, i) =>
16 i.toString(2).padStart(5, '0').split('').map(mark => mark === '1'));
17const FACTORIALS = [1, 1, 2, 6, 24, 120];

19class CommandError extends Error {}
20function check(condition, reason) {
21 if (!condition) throw new CommandError(reason);
22}
23function integerWord(text, low, high) {
24 check(/^-?(?:0|[1-9][0-9]*)$/.test(text) && text !== '-0', 'bad value ' + text);
25 const value = Number(text);
26 check(Number.isSafeInteger(value) && value >= low && value <= high, 'bad value ' + text);
27 return value;
28}
29function playerName(text) {
30 check(/^[A-Z]{1,8}$/.test(text), 'bad name ' + text);
31 return text;
32}
33function holdPattern(text) {
34 check(/^[01]{5}$/.test(text), 'bad pattern ' + text);
35 return [...text].map(mark => mark === '1');
36}
37function categoryName(text) {
38 check(CATEGORIES.includes(text), 'bad category ' + text);
39 return text;
40}
41function strategyName(text) {
42 check(STRATEGIES.has(text), 'bad strategy ' + text);
43 return text;
44}
45function faceText(faces) {
46 return faces.length ? faces.join(' ') : '-';
47}
48function patternText(holds) {
49 return holds.map(hold => hold ? '1' : '0').join('');
50}
51function heldFaces(faces, holds) {
52 return faces.filter((_, i) => holds[i]);
53}
54function gcd(a, b) {
55 while (b !== 0) [a, b] = [b, a % b];
56 return a;
57}
58function fractionText(numerator, denominator) {
59 const d = gcd(numerator, denominator);
60 return denominator / d === 1 ? String(numerator / d) : numerator / d + '/' + denominator / d;
61}
62function averageText(total, count) {
63 const amount = count ? Math.floor(100 * total / count) : 0;
64 return Math.floor(amount / 100) + '.' + String(amount % 100).padStart(2, '0');
65}
66function isYahtzee(faces) {
67 return faces.every(face => face === faces[0]);
68}
69function normalScore(category, faces) {
70 const upper = UPPER.indexOf(category);
71 const counts = FACES.map(face => faces.filter(v => v === face).length);
72 if (upper !== -1) return (upper + 1) * counts[upper];
73 const total = faces.reduce((a, b) => a + b, 0);
74 switch (category) {
75 case 'THREEKIND': return Math.max(...counts) >= 3 ? total : 0;
76 case 'FOURKIND': return Math.max(...counts) >= 4 ? total : 0;
77 case 'FULLHOUSE': return counts.includes(3) && counts.includes(2) ? 25 : 0;
78 case 'SMALL': return [1, 2, 3].some(start =>
79 [0, 1, 2, 3].every(offset => faces.includes(start + offset))) ? 30 : 0;
80 case 'LARGE': return [1, 2].some(start =>
81 [0, 1, 2, 3, 4].every(offset => faces.includes(start + offset))) ? 40 : 0;
82 case 'YAHTZEE': return isYahtzee(faces) ? 50 : 0;
83 default: return total;
84 }
85}
86function bestCandidate(candidates) {
87 return candidates.reduce((best, candidate) => candidate[1] > best[1] ? candidate : best);
88}
89function mostFrequentFace(faces) {
90 return FACES.reduce((best, f) => {
91 const count = faces.filter(face => face === f).length;
92 const prior = faces.filter(face => face === best).length;
93 return count > prior || (count === prior && f > best) ? f : best;
94 }, 1);
95}
96function botHolds(strategy, faces) {
97 if (strategy === 'HIGH') return faces.map(face => face >= 5);
98 if (strategy === 'STRAIGHT') {
99 let bestStart = 0, bestLength = 0;
100 for (const start of FACES) {
101 let length = 0;
102 while (faces.includes(start + length)) ++length;
103 if (length > 0 && length >= bestLength) [bestStart, bestLength] = [start, length];
104 }
105 if (bestLength >= 3) {
106 const holds = Array(5).fill(false);
107 for (let v = bestStart; v < bestStart + bestLength; ++v) holds[faces.indexOf(v)] = true;
108 return holds;
109 }
110 }
111 const face = mostFrequentFace(faces);
112 return faces.map(v => v === face);
113}
114function rerollOutcomes(size) {
115 const result = [];
116 function visit(minimum, remaining, values) {
117 if (!remaining) {
118 let weight = FACTORIALS[size];
119 for (const face of FACES) weight /= FACTORIALS[values.filter(v => v === face).length];
120 result.push([values, weight]);
121 return;
122 }
123 for (let face = minimum; face <= 6; ++face) visit(face, remaining - 1, [...values, face]);
124 }
125 visit(1, size, []);
126 return result;
127}
128const REROLLS = Array.from({ length: 6 }, (_, i) => rerollOutcomes(i));

130class DiceRoller {
131 constructor(seed) {
132 this.state = seed;
133 this.loaded = [];
134 this.counts = Array(6).fill(0);
135 }
136 rollDie() {
137 let face;
138 if (this.loaded.length) face = this.loaded.shift();
139 else {
140 this.state = MULTIPLIER * this.state % MODULUS;
141 face = this.state % 6 + 1;
142 }
143 ++this.counts[face - 1];
144 return face;
145 }
146}
147class Player {
148 constructor(name, strategy = 'KIND') {
149 this.name = name;
150 this.strategy = strategy;
151 this.boxes = Object.fromEntries(CATEGORIES.map(c => [c, null]));
152 this.yahtzeeBonus = 0;
153 }
154 openCategories() { return CATEGORIES.filter(c => this.boxes[c] === null); }
155 usedCount() { return CATEGORIES.length - this.openCategories().length; }
156 round() { return this.usedCount() + 1; }
157 upperSubtotal() { return UPPER.reduce((sum, c) => sum + (this.boxes[c] ?? 0), 0); }
158 upperBonus() { return this.upperSubtotal() >= 63 ? 35 : 0; }
159 total() {
160 return Object.values(this.boxes).reduce((sum, v) => sum + (v ?? 0), 0)
161 + this.upperBonus() + this.yahtzeeBonus;
162 }
163}
164class Turn {
165 constructor(seat, rolls = 0, faces = [], holds = Array(5).fill(false)) {
166 this.seat = seat;
167 this.rolls = rolls;
168 this.faces = [...faces];
169 this.holds = [...holds];
170 }
171 copy() { return new Turn(this.seat, this.rolls, this.faces, this.holds); }
172}
173class Game {
174 constructor(players, rules, roller) {
175 this.players = players;
176 this.rules = rules;
177 this.roller = roller;
178 this.turn = new Turn(0);
179 this.entries = [];
180 this.redoList = [];
181 }
182 finished() { return this.turn === null; }
183 currentPlayer() { return this.players[this.turn.seat]; }
184 ranking() { return [...this.players].sort((a, b) => b.total() - a.total()); }
185 roll() {
186 const turn = this.turn;
187 if (turn.rolls === 0) turn.faces = Array.from({ length: 5 }, () => this.roller.rollDie());
188 else for (let i = 0; i < 5; ++i) if (!turn.holds[i]) turn.faces[i] = this.roller.rollDie();
189 ++turn.rolls;
190 this.redoList.length = 0;
191 return `${this.currentPlayer().name} roll ${turn.rolls}: ${faceText(turn.faces)}`;
192 }
193 setHolds(holds) {
194 this.turn.holds = [...holds];
195 this.redoList.length = 0;
196 return `${this.currentPlayer().name} holds ${patternText(holds)}: ${faceText(heldFaces(this.turn.faces, holds))}`;
197 }
198 jokerSituation(player, faces) {
199 return this.rules.joker !== 'OFF' && isYahtzee(faces) && player.boxes.YAHTZEE !== null;
200 }
201 allowedCategories(player, faces, joker) {
202 const open = player.openCategories();
203 if (!joker || this.rules.joker === 'FREE') return open;
204 const upper = UPPER[faces[0] - 1];
205 if (player.boxes[upper] === null) return [upper];
206 const lower = open.filter(c => LOWER.includes(c));
207 return lower.length ? lower : open;
208 }
209 pointsToEnter(faces, category, joker) {
210 return joker && Object.hasOwn(JOKER_POINTS, category) ? JOKER_POINTS[category]
211 : normalScore(category, faces);
212 }
213 candidates(player, faces) {
214 const joker = this.jokerSituation(player, faces);
215 const allowed = new Set(this.allowedCategories(player, faces, joker));
216 return player.openCategories().map(category => [category,
217 allowed.has(category) ? this.pointsToEnter(faces, category, joker) : 0]);
218 }
219 best(player, faces) { return bestCandidate(this.candidates(player, faces)); }
220 prepareEntry(category) {
221 const player = this.currentPlayer(), faces = this.turn.faces;
222 check(player.boxes[category] === null, `${category} already used`);
223 const joker = this.jokerSituation(player, faces);
224 if (!this.allowedCategories(player, faces, joker).includes(category)) {
225 const upper = UPPER[faces[0] - 1];
226 check(player.boxes[upper] !== null, 'joker requires ' + upper);
227 throw new CommandError('joker requires lower box');
228 }
229 const points = this.pointsToEnter(faces, category, joker);
230 const bonus = isYahtzee(faces) && player.boxes.YAHTZEE === 50 ? 100 : 0;
231 return { player, category, points, bonus, joker, turn: this.turn.copy() };
232 }
233 scoreLine(entry, verb) {
234 const player = entry.player;
235 const parts = [player.name, verb, entry.points, 'in', entry.category];
236 if (entry.joker) parts.push('joker');
237 if (entry.bonus) parts.push('ybonus', entry.bonus);
238 const subtotal = player.upperSubtotal() + (UPPER.includes(entry.category) ? entry.points : 0);
239 const gained = player.upperBonus() === 0 && subtotal >= 63 ? 35 : 0;
240 if (gained) parts.push('ubonus', gained);
241 parts.push('total', player.total() + entry.points + entry.bonus + gained);
242 return parts.join(' ');
243 }
244 enter(entry) {
245 entry.player.boxes[entry.category] = entry.points;
246 entry.player.yahtzeeBonus += entry.bonus;
247 this.entries.push(entry);
248 }
249 score(category) {
250 const entry = this.prepareEntry(category), line = this.scoreLine(entry, 'scores');
251 this.enter(entry);
252 this.redoList.length = 0;
253 if (this.players.every(player => player.usedCount() === 13)) {
254 this.turn = null;
255 return [line, 'game over'];
256 }
257 this.turn = new Turn((this.turn.seat + 1) % this.players.length);
258 const player = this.currentPlayer();
259 return [line, `turn ${player.round()} ${player.name}`];
260 }
261 undo() {
262 const entry = this.entries.pop();
263 entry.player.boxes[entry.category] = null;
264 entry.player.yahtzeeBonus -= entry.bonus;
265 this.redoList.push([this.turn, entry]);
266 this.turn = entry.turn.copy();
267 return entry;
268 }
269 redo() {
270 const [turn, entry] = this.redoList.pop();
271 this.enter(entry);
272 this.turn = turn;
273 return entry;
274 }
275 botCategory() {
276 const player = this.currentPlayer(), faces = this.turn.faces;
277 if (player.strategy === 'UPPER') {
278 const face = mostFrequentFace(faces), category = UPPER[face - 1];
279 const allowed = this.allowedCategories(player, faces, this.jokerSituation(player, faces));
280 if (faces.filter(v => v === face).length >= 3 && allowed.includes(category)) return category;
281 }
282 return this.best(player, faces)[0];
283 }
284 botTurn() {
285 const lines = [];
286 if (!this.turn.rolls) lines.push(this.roll());
287 while (this.turn.rolls < this.rules.rolls) {
288 const holds = botHolds(this.currentPlayer().strategy, this.turn.faces);
289 if (holds.every(Boolean)) break;
290 lines.push(this.setHolds(holds));
291 lines.push(this.roll());
292 }
293 lines.push(...this.score(this.botCategory()));
294 return lines;
295 }
296}
297class Advisor {
298 constructor(game, player) {
299 this.game = game;
300 this.player = player;
301 this.values = new Map();
302 this.expectations = new Map();
303 }
304 value(hand, level) {
305 const key = level + ':' + hand.join('');
306 if (this.values.has(key)) return this.values.get(key);
307 const value = !level ? this.game.best(this.player, hand)[1]
308 : Math.max(...PATTERNS.map(holds =>
309 this.expectation(heldFaces(hand, holds).sort((a, b) => a - b), level)));
310 this.values.set(key, value);
311 return value;
312 }
313 expectation(held, level) {
314 const key = level + ':' + held.join('');
315 if (this.expectations.has(key)) return this.expectations.get(key);
316 let total = 0;
317 for (const [rolled, weight] of REROLLS[5 - held.length]) {
318 total += weight * this.value([...held, ...rolled].sort((a, b) => a - b), level - 1);
319 }
320 const value = total * 6 ** held.length;
321 this.expectations.set(key, value);
322 return value;
323 }
324 bestHold(faces, levels) {
325 let bestHolds = null, bestValue = -1;
326 for (const holds of PATTERNS) {
327 const value = this.expectation(heldFaces(faces, holds).sort((a, b) => a - b), levels);
328 if (value > bestValue) [bestHolds, bestValue] = [holds, value];
329 }
330 return [bestHolds, bestValue];
331 }
332}

360class YahtzeeTable {
361 constructor() {
362 this.rules = { rolls: 3, joker: 'FORCED' };
363 this.game = null;
364 this.records = new Map();
365 this.output = [];
366 this.handlers = {
367 RULE: this.setRule, GAME: this.startGame, BOT: this.setBot, LOAD: this.load,
368 ROLL: this.roll, HOLD: this.hold, KEEP: this.keep, SCORE: this.score,
369 WHATIF: this.whatIf, UNDO: this.undo, REDO: this.redo, SUGGEST: this.suggest,
370 ODDS: this.odds, ADVISE: this.advise, AUTO: this.auto, STATUS: this.status,
371 CARD: this.card, BONUS: this.bonus, STATS: this.stats, LOG: this.log,
372 FACES: this.faces, STANDINGS: this.standings, END: this.end,
373 RECORDS: this.showRecords, SIM: this.simulate
374 };
375 }
376 emit(line) { this.output.push(line); }
377 requireNoGame() { check(this.game === null, 'game in progress'); }
378 requireGame() { check(this.game !== null, 'no game'); return this.game; }
379 requireTurn() { const game = this.requireGame(); check(!game.finished(), 'game over'); return game; }
380 requireRolled() { const game = this.requireTurn(); check(game.turn.rolls !== 0, 'not rolled'); return game; }
381 requireRollsLeft() {
382 const game = this.requireRolled();
383 check(game.turn.rolls !== this.rules.rolls, 'no rolls left');
384 return game;
385 }
386 setRule(args) {
387 const [rule, value] = args;
388 if (rule === 'ROLLS') {
389 const rolls = integerWord(value, 1, 4);
390 this.requireNoGame();
391 this.rules.rolls = rolls;
392 } else if (rule === 'JOKER') {
393 check(JOKER_MODES.has(value), 'bad value ' + value);
394 this.requireNoGame();
395 this.rules.joker = value;
396 } else throw new CommandError('unknown rule ' + rule);
397 this.emit(`rule ${rule} ${value}`);
398 }
399 startGame(args) {
400 const seed = integerWord(args[0], 1, MODULUS - 1);
401 const names = args.slice(1).map(playerName);
402 check(names.length <= 6, 'too many players');
403 this.requireNoGame();
404 for (let i = 0; i < names.length; ++i)
405 check(!names.slice(0, i).includes(names[i]), 'duplicate name ' + names[i]);
406 this.game = new Game(names.map(name => new Player(name)), this.rules, new DiceRoller(seed));
407 this.emit(`game seed ${seed} players ${names.join(' ')}`);
408 this.emit(`turn 1 ${names[0]}`);
409 }
410 setBot(args) {
411 const name = playerName(args[0]), strategy = strategyName(args[1]);
412 const game = this.requireGame(), player = game.players.find(player => player.name === name);
413 check(player !== undefined, 'no player ' + name);
414 player.strategy = strategy;
415 this.emit(`bot ${name} ${strategy}`);
416 }
417 load(args) {
418 const faces = args.map(text => integerWord(text, 1, 6));
419 const game = this.requireGame();
420 game.roller.loaded.push(...faces);
421 this.emit(`loaded ${game.roller.loaded.length}`);
422 }
423 roll() {
424 const game = this.requireTurn();
425 check(game.turn.rolls !== this.rules.rolls, 'no rolls left');
426 check(!game.turn.holds.every(Boolean), 'all dice held');

428 }
429 hold(args) {
430 const holds = holdPattern(args[0]);
431 const game = this.requireRollsLeft();

434 keep(args) {
435 const faces = args.map(text => integerWord(text, 1, 6));
436 const game = this.requireRollsLeft(), holds = Array(5).fill(false);
437 for (const face of faces) {
438 const index = game.turn.faces.findIndex((v, i) => v === face && !holds[i]);
439 check(index >= 0, 'not showing ' + face);

441 }
442 this.emit(game.setHolds(holds));
443 }

446 this.output.push(...game.score(category));
447 }

453 const game = this.requireGame();
454 check(game.entries.length > 0, 'nothing to undo');
455 const entry = game.undo();
456 this.emit(`undo ${entry.player.name} ${entry.category} ${entry.points}`);

458 redo() {
459 const game = this.requireGame();
460 check(game.redoList.length > 0, 'nothing to redo');
461 const entry = game.redo();
462 this.emit(`redo ${entry.player.name} ${entry.category} ${entry.points}`);

464 suggest() {
465 const game = this.requireRolled();
466 const candidates = game.candidates(game.currentPlayer(), game.turn.faces);
467 const best = bestCandidate(candidates);
468 this.emit('suggest ' + candidates.flat().join(' ') + ' best ' + best.join(' '));

470 odds(args) {
471 const holds = holdPattern(args[0]), game = this.requireRollsLeft();
472 const player = game.currentPlayer();
473 const held = heldFaces(game.turn.faces, holds);
474 const open = player.openCategories(), expected = Object.fromEntries(open.map(c => [c, 0]));
475 const chances = Object.fromEntries(open.map(c => [c, 0]));
476 let bestTotal = 0;
477 for (const [rolled, weight] of REROLLS[5 - held.length]) {
478 const candidates = game.candidates(player, [...held, ...rolled]);
479 for (const [category, score] of candidates) {
480 expected[category] += weight * score;
481 if (score > 0) chances[category] += weight;
482 }
483 bestTotal += weight * bestCandidate(candidates)[1];
484 }
485 const outcomes = 6 ** (5 - held.length);
486 this.emit(`odds ${patternText(holds)} outcomes ${outcomes}`);
487 for (const category of open) this.emit(`${category} expect ${fractionText(expected[category], outcomes)}`
488 + ` chance ${fractionText(chances[category], outcomes)}`);
489 this.emit('best expect ' + fractionText(bestTotal, outcomes));
490 }
491 advise() {
492 const game = this.requireRollsLeft(), levels = this.rules.rolls - game.turn.rolls;
493 const advisor = new Advisor(game, game.currentPlayer());
494 const [holds, value] = advisor.bestHold(game.turn.faces, levels);
495 this.emit(`advise ${patternText(holds)}: ${faceText(heldFaces(game.turn.faces, holds))}`
496 + ` expect ${fractionText(value, 7776 ** levels)}`);
497 }
498 auto(args) {
499 const count = integerWord(args[0], 1, 100), game = this.requireTurn();
500 for (let i = 0; i < count; ++i) {
501 this.output.push(...game.botTurn());
502 if (game.finished()) break;
503 }
504 }
505 status() {
506 const game = this.requireGame();
507 if (game.finished()) { this.emit('status game over'); return; }
508 const turn = game.turn, player = game.currentPlayer();
509 this.emit(`status round ${player.round()} player ${player.name} rolls ${turn.rolls}`
510 + ` dice ${faceText(turn.faces)} holds ${patternText(turn.holds)}`);
511 }
512 card() {
513 const players = this.requireGame().players;
514 const widths = players.map(player => Math.max(4, player.name.length));
515 const rows = [['CATEGORY', players.map(player => player.name)]];
516 for (const category of CATEGORIES) rows.push([category, players.map(player =>
517 player.boxes[category] === null ? '-' : String(player.boxes[category]))]);
518 rows.push(['UPPER', players.map(p => String(p.upperSubtotal()))]);
519 rows.push(['BONUS', players.map(p => String(p.upperBonus()))]);
520 rows.push(['YBONUS', players.map(p => String(p.yahtzeeBonus))]);
521 rows.push(['TOTAL', players.map(p => String(p.total()))]);
522 for (const [heading, cells] of rows) {
523 this.emit(heading.padEnd(9) + cells.map((cell, i) => ' ' + cell.padStart(widths[i])).join(''));
524 }
525 }
526 bonus() {
527 const game = this.requireGame();
528 for (const player of game.players) {
529 const subtotal = player.upperSubtotal();
530 let maximum = subtotal;
531 for (let i = 0; i < 6; ++i) if (player.boxes[UPPER[i]] === null) maximum += 5 * (i + 1);
532 const state = subtotal >= 63 ? 'made' : maximum < 63 ? 'lost' : 'open';
533 this.emit(`bonus ${player.name} upper ${subtotal} need ${Math.max(0, 63 - subtotal)}`
534 + ` max ${maximum} ${state}`);
535 }
536 }
537 stats() {
538 const game = this.requireGame();
539 for (const player of game.players) {
540 const entries = game.entries.filter(entry => entry.player === player);
541 const rolls = entries.reduce((sum, entry) => sum + entry.turn.rolls, 0);
542 const zeros = entries.filter(entry => entry.points === 0).length;
543 const yahtzees = entries.filter(entry => isYahtzee(entry.turn.faces)).length;

546 + ` yahtzees ${yahtzees} average ${averageText(points, entries.length)}`);

561 faces() {
562 const counts = this.requireGame().roller.counts, n = counts.reduce((a, b) => a + b, 0);
563 const squares = counts.reduce((sum, count) => sum + count * count, 0);
564 const chi2 = n ? fractionText(6 * squares - n * n, n) : '0';
565 this.emit(`faces ${counts.join(' ')} total ${n} chi2 ${chi2}`);
566 }
567 standings() {

569 ranking.forEach((player, index) => this.emit(`standing ${index + 1} ${player.name}`
570 + ` ${player.total()} behind ${leader - player.total()}`));
571 }
572 end() {
573 const game = this.requireGame(), finished = game.finished();

575 game.ranking().forEach((player, index) => {

577 if (finished) {
578 if (!this.records.has(player.name)) this.records.set(player.name, new Record());

581 });
582 this.game = null;
583 }

591 simulate(args) {
592 const seed = integerWord(args[0], 1, MODULUS - 1);
593 const count = integerWord(args[1], 1, 50);
594 const strategies = args.slice(2).map(strategyName);
595 check(strategies.length <= 6, 'too many players');
596 const roller = new DiceRoller(seed), summaries = strategies.map(s => new SeatSummary(s));
597 for (let i = 0; i < count; ++i) {
598 const players = strategies.map((strategy, seat) => new Player(String(seat + 1), strategy));
599 const game = new Game(players, this.rules, roller);
600 while (!game.finished()) game.botTurn();
601 const winner = game.ranking()[0];
602 for (let seat = 0; seat < strategies.length; ++seat)
603 summaries[seat].add(players[seat], players[seat] === winner);
604 }
605 this.emit(`sim seed ${seed} games ${count} players ${strategies.length}`);
606 summaries.forEach((summary, seat) => this.emit(`seat ${seat + 1} ${summary.strategy}`
607 + ` wins ${summary.wins} average ${averageText(summary.total, count)}`
608 + ` best ${summary.best} bonus ${summary.bonus} yahtzee ${summary.yahtzee}`));
609 }
