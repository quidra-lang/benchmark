3const FACES = [1, 2, 3, 4, 5, 6];
4const CATEGORIES = [

24function check(condition: unknown, reason: string): asserts condition {

27function integerWord(text: string, low: number, high: number): number {
28 check(/^-?(?:0|[1-9][0-9]*)$/.test(text) && text !== '-0', 'bad value ' + text);
29 const value = Number(text);
30 check(Number.isSafeInteger(value) && value >= low && value <= high, 'bad value ' + text);
31 return value;
32}
33function playerName(text: string): string {
34 check(/^[A-Z]{1,8}$/.test(text), 'bad name ' + text);
35 return text;
36}
37function holdPattern(text: string): boolean[] {
38 check(/^[01]{5}$/.test(text), 'bad pattern ' + text);
39 return [...text].map(mark => mark === '1');
40}

49function faceText(faces: number[]): string {
50 return faces.length ? faces.join(' ') : '-';
51}
52function patternText(holds: boolean[]): string {
53 return holds.map(hold => hold ? '1' : '0').join('');
54}
55function heldFaces(faces: number[], holds: boolean[]): number[] {
56 return faces.filter((_, i) => holds[i]);
57}
58function gcd(a: number, b: number): number {
59 while (b !== 0) [a, b] = [b, a % b];
60 return a;
61}
62function fractionText(numerator: number, denominator: number): string {
63 const d = gcd(numerator, denominator);
64 return denominator / d === 1 ? String(numerator / d) : numerator / d + '/' + denominator / d;
65}
66function averageText(total: number, count: number): string {
67 const amount = count ? Math.floor(100 * total / count) : 0;
68 return Math.floor(amount / 100) + '.' + String(amount % 100).padStart(2, '0');
69}
70function isYahtzee(faces: number[]): boolean {
71 return faces.every(face => face === faces[0]);
72}
73function normalScore(category: string, faces: number[]): number {
74 const upper = UPPER.indexOf(category);
75 const counts = FACES.map(face => faces.filter(v => v === face).length);
76 if (upper !== -1) return (upper + 1) * counts[upper];
77 const total = faces.reduce((a, b) => a + b, 0);
78 switch (category) {
79 case 'THREEKIND': return Math.max(...counts) >= 3 ? total : 0;
80 case 'FOURKIND': return Math.max(...counts) >= 4 ? total : 0;
81 case 'FULLHOUSE': return counts.includes(3) && counts.includes(2) ? 25 : 0;
82 case 'SMALL': return [1, 2, 3].some(start =>
83 [0, 1, 2, 3].every(offset => faces.includes(start + offset))) ? 30 : 0;
84 case 'LARGE': return [1, 2].some(start =>
85 [0, 1, 2, 3, 4].every(offset => faces.includes(start + offset))) ? 40 : 0;
86 case 'YAHTZEE': return isYahtzee(faces) ? 50 : 0;
87 default: return total;
88 }
89}
90function bestCandidate(candidates: Candidate[]): Candidate {
91 return candidates.reduce((best, candidate) => candidate[1] > best[1] ? candidate : best);
92}
93function mostFrequentFace(faces: number[]): number {
94 return FACES.reduce((best, f) => {
95 const count = faces.filter(face => face === f).length;
96 const prior = faces.filter(face => face === best).length;
97 return count > prior || (count === prior && f > best) ? f : best;
98 }, 1);
99}
100function botHolds(strategy: string, faces: number[]): boolean[] {
101 if (strategy === 'HIGH') return faces.map(face => face >= 5);
102 if (strategy === 'STRAIGHT') {
103 let bestStart = 0, bestLength = 0;
104 for (const start of FACES) {
105 let length = 0;
106 while (faces.includes(start + length)) ++length;
107 if (length > 0 && length >= bestLength) [bestStart, bestLength] = [start, length];
108 }
109 if (bestLength >= 3) {
110 const holds = Array(5).fill(false);
111 for (let v = bestStart; v < bestStart + bestLength; ++v) holds[faces.indexOf(v)] = true;
112 return holds;
113 }
114 }
115 const face = mostFrequentFace(faces);
116 return faces.map(v => v === face);
117}
118function rerollOutcomes(size: number): Outcome[] {
119 const result: Outcome[] = [];
120 function visit(minimum: number, remaining: number, values: number[]): void {
121 if (!remaining) {
122 let weight = FACTORIALS[size];
123 for (const face of FACES) weight /= FACTORIALS[values.filter(v => v === face).length];
124 result.push([values, weight]);
125 return;
126 }
127 for (let face = minimum; face <= 6; ++face) visit(face, remaining - 1, [...values, face]);
128 }
129 visit(1, size, []);
130 return result;
131}
132const REROLLS = Array.from({ length: 6 }, (_, i) => rerollOutcomes(i));

134class DiceRoller {
135 state: number;
136 loaded: number[] = [];
137 counts: number[] = Array(6).fill(0);
138
139 constructor(seed: number) {
140 this.state = seed;
141 }
142 rollDie(): number {
143 let face: number;
144 if (this.loaded.length) face = this.loaded.shift()!;
145 else {
146 this.state = MULTIPLIER * this.state % MODULUS;
147 face = this.state % 6 + 1;
148 }
149 ++this.counts[face - 1];
150 return face;
151 }
152}
153class Player {
154 name: string;
155 strategy: string;
156 boxes: Record<string, number | null>;
157 yahtzeeBonus = 0;
158
159 constructor(name: string, strategy = 'KIND') {
160 this.name = name;
161 this.strategy = strategy;
162 this.boxes = Object.fromEntries(CATEGORIES.map(c => [c, null]));
163 }
164 openCategories(): string[] { return CATEGORIES.filter(c => this.boxes[c] === null); }
165 usedCount(): number { return CATEGORIES.length - this.openCategories().length; }
166 round(): number { return this.usedCount() + 1; }
167 upperSubtotal(): number { return UPPER.reduce((sum, c) => sum + (this.boxes[c] ?? 0), 0); }
168 upperBonus(): number { return this.upperSubtotal() >= 63 ? 35 : 0; }
169 total(): number {
170 return Object.values(this.boxes).reduce<number>((sum, v) => sum + (v ?? 0), 0)
171 + this.upperBonus() + this.yahtzeeBonus;
172 }
173}

190class Game {
191 players: Player[];
192 rules: Rules;
193 roller: DiceRoller;
194 turn: Turn | null = new Turn(0);
195 entries: Entry[] = [];
196 redoList: [Turn | null, Entry][] = [];
197
198 constructor(players: Player[], rules: Rules, roller: DiceRoller) {
199 this.players = players;
200 this.rules = rules;
201 this.roller = roller;
202 this.turn = new Turn(0);
203 }
204 finished(): boolean { return this.turn === null; }
205 currentPlayer(): Player { return this.players[this.turn!.seat]; }
206 ranking(): Player[] { return [...this.players].sort((a, b) => b.total() - a.total()); }
207 roll(): string {
208 const turn = this.turn!;
209 if (turn.rolls === 0) turn.faces = Array.from({ length: 5 }, () => this.roller.rollDie());
210 else for (let i = 0; i < 5; ++i) if (!turn.holds[i]) turn.faces[i] = this.roller.rollDie();
211 ++turn.rolls;
212 this.redoList.length = 0;
213 return `${this.currentPlayer().name} roll ${turn.rolls}: ${faceText(turn.faces)}`;
214 }
215 setHolds(holds: boolean[]): string {
216 this.turn!.holds = [...holds];
217 this.redoList.length = 0;
218 return `${this.currentPlayer().name} holds ${patternText(holds)}: ${faceText(heldFaces(this.turn!.faces, holds))}`;
219 }
220 jokerSituation(player: Player, faces: number[]): boolean {
221 return this.rules.joker !== 'OFF' && isYahtzee(faces) && player.boxes.YAHTZEE !== null;
222 }
223 allowedCategories(player: Player, faces: number[], joker: boolean): string[] {
224 const open = player.openCategories();
225 if (!joker || this.rules.joker === 'FREE') return open;
226 const upper = UPPER[faces[0] - 1];
227 if (player.boxes[upper] === null) return [upper];
228 const lower = open.filter(c => LOWER.includes(c));
229 return lower.length ? lower : open;
230 }
231 pointsToEnter(faces: number[], category: string, joker: boolean): number {
232 return joker && Object.hasOwn(JOKER_POINTS, category) ? JOKER_POINTS[category]
233 : normalScore(category, faces);
234 }
235 candidates(player: Player, faces: number[]): Candidate[] {
236 const joker = this.jokerSituation(player, faces);
237 const allowed = new Set(this.allowedCategories(player, faces, joker));
238 return player.openCategories().map((category): Candidate => [category,
239 allowed.has(category) ? this.pointsToEnter(faces, category, joker) : 0]);
240 }
241 best(player: Player, faces: number[]): Candidate { return bestCandidate(this.candidates(player, faces)); }
242 prepareEntry(category: string): Entry {
243 const player = this.currentPlayer(), faces = this.turn!.faces;
244 check(player.boxes[category] === null, `${category} already used`);
245 const joker = this.jokerSituation(player, faces);
246 if (!this.allowedCategories(player, faces, joker).includes(category)) {
247 const upper = UPPER[faces[0] - 1];
248 check(player.boxes[upper] !== null, 'joker requires ' + upper);
249 throw new CommandError('joker requires lower box');
250 }
251 const points = this.pointsToEnter(faces, category, joker);
252 const bonus = isYahtzee(faces) && player.boxes.YAHTZEE === 50 ? 100 : 0;
253 return { player, category, points, bonus, joker, turn: this.turn!.copy() };
254 }
255 scoreLine(entry: Entry, verb: string): string {
256 const player = entry.player;
257 const parts: (string | number)[] = [player.name, verb, entry.points, 'in', entry.category];
258 if (entry.joker) parts.push('joker');
259 if (entry.bonus) parts.push('ybonus', entry.bonus);
260 const subtotal = player.upperSubtotal() + (UPPER.includes(entry.category) ? entry.points : 0);
261 const gained = player.upperBonus() === 0 && subtotal >= 63 ? 35 : 0;
262 if (gained) parts.push('ubonus', gained);
263 parts.push('total', player.total() + entry.points + entry.bonus + gained);
264 return parts.join(' ');
265 }
266 enter(entry: Entry): void {
267 entry.player.boxes[entry.category] = entry.points;
268 entry.player.yahtzeeBonus += entry.bonus;
269 this.entries.push(entry);
270 }
271 score(category: string): string[] {
272 const entry = this.prepareEntry(category), line = this.scoreLine(entry, 'scores');
273 this.enter(entry);
274 this.redoList.length = 0;
275 if (this.players.every(player => player.usedCount() === 13)) {
276 this.turn = null;
277 return [line, 'game over'];
278 }
279 this.turn = new Turn((this.turn!.seat + 1) % this.players.length);
280 const player = this.currentPlayer();
281 return [line, `turn ${player.round()} ${player.name}`];
282 }
283 undo(): Entry {
284 const entry = this.entries.pop()!;
285 entry.player.boxes[entry.category] = null;
286 entry.player.yahtzeeBonus -= entry.bonus;
287 this.redoList.push([this.turn, entry]);
288 this.turn = entry.turn.copy();
289 return entry;
290 }
291 redo(): Entry {
292 const [turn, entry] = this.redoList.pop()!;
293 this.enter(entry);
294 this.turn = turn;
295 return entry;
296 }
297 botCategory(): string {
298 const player = this.currentPlayer(), faces = this.turn!.faces;
299 if (player.strategy === 'UPPER') {
300 const face = mostFrequentFace(faces), category = UPPER[face - 1];
301 const allowed = this.allowedCategories(player, faces, this.jokerSituation(player, faces));
302 if (faces.filter(v => v === face).length >= 3 && allowed.includes(category)) return category;
303 }
304 return this.best(player, faces)[0];
305 }
306 botTurn(): string[] {
307 const lines = [];
308 if (!this.turn!.rolls) lines.push(this.roll());
309 while (this.turn!.rolls < this.rules.rolls) {
310 const holds = botHolds(this.currentPlayer().strategy, this.turn!.faces);
311 if (holds.every(Boolean)) break;
312 lines.push(this.setHolds(holds));
313 lines.push(this.roll());
314 }
315 lines.push(...this.score(this.botCategory()));
316 return lines;
317 }
318}
319class Advisor {
320 game: Game;
321 player: Player;
322 values = new Map<string, number>();
323 expectations = new Map<string, number>();
324
325 constructor(game: Game, player: Player) {
326 this.game = game;
327 this.player = player;
328
329 }
330 value(hand: number[], level: number): number {
331 const key = level + ':' + hand.join('');
332 if (this.values.has(key)) return this.values.get(key)!;
333 const value = !level ? this.game.best(this.player, hand)[1]
334 : Math.max(...PATTERNS.map(holds =>
335 this.expectation(heldFaces(hand, holds).sort((a, b) => a - b), level)));
336 this.values.set(key, value);
337 return value;
338 }
339 expectation(held: number[], level: number): number {
340 const key = level + ':' + held.join('');
341 if (this.expectations.has(key)) return this.expectations.get(key)!;
342 let total = 0;
343 for (const [rolled, weight] of REROLLS[5 - held.length]) {
344 total += weight * this.value([...held, ...rolled].sort((a, b) => a - b), level - 1);
345 }
346 const value = total * 6 ** held.length;
347 this.expectations.set(key, value);
348 return value;
349 }
350 bestHold(faces: number[], levels: number): [boolean[], number] {
351 let bestHolds: boolean[] | null = null, bestValue = -1;
352 for (const holds of PATTERNS) {
353 const value = this.expectation(heldFaces(faces, holds).sort((a, b) => a - b), levels);
354 if (value > bestValue) [bestHolds, bestValue] = [holds, value];
355 }
356 return [bestHolds!, bestValue];
357 }
358}

398class YahtzeeTable {
399 rules: Rules = {rolls: 3, joker: 'FORCED'};
400 game: Game | null = null;
401 records = new Map<string, PlayerRecord>();
402 output: string[] = [];
403 handlers: Record<string, (args: string[]) => void>;
404
405 constructor() {
406
407 this.handlers = {
408 RULE: this.setRule, GAME: this.startGame, BOT: this.setBot, LOAD: this.load,
409 ROLL: this.roll, HOLD: this.hold, KEEP: this.keep, SCORE: this.score,
410 WHATIF: this.whatIf, UNDO: this.undo, REDO: this.redo, SUGGEST: this.suggest,
411 ODDS: this.odds, ADVISE: this.advise, AUTO: this.auto, STATUS: this.status,
412 CARD: this.card, BONUS: this.bonus, STATS: this.stats, LOG: this.log,
413 FACES: this.faces, STANDINGS: this.standings, END: this.end,
414 RECORDS: this.showRecords, SIM: this.simulate

416 }
417 emit(line: string): void { this.output.push(line); }
418 requireNoGame(): void { check(this.game === null, 'game in progress'); }
419 requireGame(): Game { check(this.game !== null, 'no game'); return this.game; }
420 requireTurn(): Game { const game = this.requireGame(); check(!game.finished(), 'game over'); return game; }
421 requireRolled(): Game { const game = this.requireTurn(); check(game.turn!.rolls !== 0, 'not rolled'); return game; }

427 setRule(args: string[]): void {
428 const [rule, value] = args;
429 if (rule === 'ROLLS') {
430 const rolls = integerWord(value, 1, 4);
431 this.requireNoGame();
432 this.rules.rolls = rolls;
433 } else if (rule === 'JOKER') {
434 check(JOKER_MODES.has(value), 'bad value ' + value);
435 this.requireNoGame();
436 this.rules.joker = value;
437 } else throw new CommandError('unknown rule ' + rule);
438 this.emit(`rule ${rule} ${value}`);
439 }
440 startGame(args: string[]): void {
441 const seed = integerWord(args[0], 1, MODULUS - 1);
442 const names = args.slice(1).map(playerName);
443 check(names.length <= 6, 'too many players');
444 this.requireNoGame();
445 for (let i = 0; i < names.length; ++i)
446 check(!names.slice(0, i).includes(names[i]), 'duplicate name ' + names[i]);
447 this.game = new Game(names.map(name => new Player(name)), this.rules, new DiceRoller(seed));
448 this.emit(`game seed ${seed} players ${names.join(' ')}`);
449 this.emit(`turn 1 ${names[0]}`);
450 }
451 setBot(args: string[]): void {
452 const name = playerName(args[0]), strategy = strategyName(args[1]);
453 const game = this.requireGame(), player = game.players.find(player => player.name === name);
454 check(player !== undefined, 'no player ' + name);
455 player.strategy = strategy;
456 this.emit(`bot ${name} ${strategy}`);
457 }

476 const faces = args.map(text => integerWord(text, 1, 6));
477 const game = this.requireRollsLeft(), holds = Array(5).fill(false);
478 for (const face of faces) {
479 const index = game.turn!.faces.findIndex((v, i) => v === face && !holds[i]);
480 check(index >= 0, 'not showing ' + face);

488 }
489 whatIf(args: string[]): void {

492 }
493 undo(): void {
494 const game = this.requireGame();
495 check(game.entries.length > 0, 'nothing to undo');
496 const entry = game.undo();
497 this.emit(`undo ${entry.player.name} ${entry.category} ${entry.points}`);
498 }
499 redo(): void {
500 const game = this.requireGame();
501 check(game.redoList.length > 0, 'nothing to redo');
502 const entry = game.redo();
503 this.emit(`redo ${entry.player.name} ${entry.category} ${entry.points}`);
504 }
505 suggest(): void {
506 const game = this.requireRolled();
507 const candidates = game.candidates(game.currentPlayer(), game.turn!.faces);
508 const best = bestCandidate(candidates);
509 this.emit('suggest ' + candidates.flat().join(' ') + ' best ' + best.join(' '));
510 }
511 odds(args: string[]): void {
512 const holds = holdPattern(args[0]), game = this.requireRollsLeft();
513 const player = game.currentPlayer();
514 const held = heldFaces(game.turn!.faces, holds);
515 const open = player.openCategories(), expected = Object.fromEntries(open.map(c => [c, 0]));
516 const chances = Object.fromEntries(open.map(c => [c, 0]));
517 let bestTotal = 0;
518 for (const [rolled, weight] of REROLLS[5 - held.length]) {
519 const candidates = game.candidates(player, [...held, ...rolled]);
520 for (const [category, score] of candidates) {
521 expected[category] += weight * score;
522 if (score > 0) chances[category] += weight;
523 }
524 bestTotal += weight * bestCandidate(candidates)[1];
525 }
526 const outcomes = 6 ** (5 - held.length);
527 this.emit(`odds ${patternText(holds)} outcomes ${outcomes}`);
528 for (const category of open) this.emit(`${category} expect ${fractionText(expected[category], outcomes)}`
529 + ` chance ${fractionText(chances[category], outcomes)}`);
530 this.emit('best expect ' + fractionText(bestTotal, outcomes));
531 }
532 advise(): void {
533 const game = this.requireRollsLeft(), levels = this.rules.rolls - game.turn!.rolls;
534 const advisor = new Advisor(game, game.currentPlayer());
535 const [holds, value] = advisor.bestHold(game.turn!.faces, levels);
536 this.emit(`advise ${patternText(holds)}: ${faceText(heldFaces(game.turn!.faces, holds))}`
537 + ` expect ${fractionText(value, 7776 ** levels)}`);
538 }
539 auto(args: string[]): void {
540 const count = integerWord(args[0], 1, 100), game = this.requireTurn();
541 for (let i = 0; i < count; ++i) {
542 this.output.push(...game.botTurn());
543 if (game.finished()) break;
544 }
545 }
546 status(): void {
547 const game = this.requireGame();
548 if (game.finished()) { this.emit('status game over'); return; }
549 const turn = game.turn!, player = game.currentPlayer();
550 this.emit(`status round ${player.round()} player ${player.name} rolls ${turn.rolls}`
551 + ` dice ${faceText(turn.faces)} holds ${patternText(turn.holds)}`);
552 }
553 card(): void {
554 const players = this.requireGame().players;
555 const widths = players.map(player => Math.max(4, player.name.length));
556 const rows: [string, string[]][] = [['CATEGORY', players.map(player => player.name)]];
557 for (const category of CATEGORIES) rows.push([category, players.map(player =>
558 player.boxes[category] === null ? '-' : String(player.boxes[category]))]);
559 rows.push(['UPPER', players.map(p => String(p.upperSubtotal()))]);
560 rows.push(['BONUS', players.map(p => String(p.upperBonus()))]);
561 rows.push(['YBONUS', players.map(p => String(p.yahtzeeBonus))]);
562 rows.push(['TOTAL', players.map(p => String(p.total()))]);
563 for (const [heading, cells] of rows) {

632 simulate(args: string[]): void {
633 const seed = integerWord(args[0], 1, MODULUS - 1);
634 const count = integerWord(args[1], 1, 50);
635 const strategies = args.slice(2).map(strategyName);
636 check(strategies.length <= 6, 'too many players');
637 const roller = new DiceRoller(seed), summaries = strategies.map(s => new SeatSummary(s));
638 for (let i = 0; i < count; ++i) {
639 const players = strategies.map((strategy, seat) => new Player(String(seat + 1), strategy));
640 const game = new Game(players, this.rules, roller);
641 while (!game.finished()) game.botTurn();
642 const winner = game.ranking()[0];
643 for (let seat = 0; seat < strategies.length; ++seat)
644 summaries[seat].add(players[seat], players[seat] === winner);
645 }
646 this.emit(`sim seed ${seed} games ${count} players ${strategies.length}`);
647 summaries.forEach((summary, seat) => this.emit(`seat ${seat + 1} ${summary.strategy}`
648 + ` wins ${summary.wins} average ${averageText(summary.total, count)}`
649 + ` best ${summary.best} bonus ${summary.bonus} yahtzee ${summary.yahtzee}`));
650 }
