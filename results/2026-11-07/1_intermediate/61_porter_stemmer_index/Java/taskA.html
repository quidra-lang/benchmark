111 static String letterPattern(String word) {
112 var kinds = new StringBuilder();
113 for (int index = 0; index < word.length(); index++) {
114 char letter = word.charAt(index);
115 boolean vowel = VOWELS.indexOf(letter) >= 0
116 || (letter == 'y' && index > 0 && kinds.charAt(index - 1) == 'C');
117 kinds.append(vowel ? 'V' : 'C');
118 }
119 return kinds.toString();
120 }

122 static int measure(String word) {
123 String pattern = letterPattern(word);
124 int count = 0;
125 for (int index = 1; index < pattern.length(); index++) {
126 if (pattern.charAt(index - 1) == 'V' && pattern.charAt(index) == 'C') {
127 count++;
128 }
129 }
130 return count;
131 }

196 static RuleOutcome applyRules(String word, List<Rule> rules) {
197 Optional<Rule> selected = rules.stream()
198 .filter(rule -> word.endsWith(rule.suffix()))
199 .max(Comparator.comparingInt(rule -> rule.suffix().length()));
200 if (selected.isEmpty()) {
201 return new RuleOutcome(word, null);
202 }
203 Rule rule = selected.get();
204 String remainder = word.substring(0, word.length() - rule.suffix().length());
205 if (!rule.condition().test(remainder)) {
206 return new RuleOutcome(word, null);
207 }
208 return new RuleOutcome(remainder + rule.replacement(), rule.suffix());
209 }

211 static String tidyStep1b(String word) {
212 if (word.endsWith("at") || word.endsWith("bl") || word.endsWith("iz")) {
213 return word + "e";
214 }
215 if (endsDouble(word) && "lsz".indexOf(word.charAt(word.length() - 1)) < 0) {
216 return word.substring(0, word.length() - 1);
217 }
218 if (measure(word) == 1 && endsCvc(word)) {
219 return word + "e";
220 }
221 return word;
222 }

224 static String applyStep(String name, String word) {
225 if (name.equals("5b")) {
226 if (measure(word) > 1 && endsDouble(word) && word.endsWith("l")) {
227 return word.substring(0, word.length() - 1);
228 }
229 return word;
230 }
231 RuleOutcome outcome = applyRules(word, RULE_TABLES.get(name));
232 String suffix = outcome.suffix();
233 if (name.equals("1b") && ("ed".equals(suffix) || "ing".equals(suffix))) {
234 return tidyStep1b(outcome.word());
235 }
236 return outcome.word();
237 }

239 static Stemming stemmingSteps(String word) {
240 var changes = new ArrayList<StepChange>();
241 String current = word;
242 if (word.length() > 2) {
243 for (String name : STEP_NAMES) {
244 String result = applyStep(name, current);
245 if (!result.equals(current)) {
246 changes.add(new StepChange(name, result));
247 }
248 current = result;
249 }
250 }
251 return new Stemming(current, changes);
252 }

304 static OptionalInt closestDistance(
305 List<Integer> positions, List<Integer> others, boolean sameStem) {
306 var gaps = new ArrayList<Integer>();
307 if (sameStem) {
308 for (int index = 1; index < positions.size(); index++) {
309 gaps.add(positions.get(index) - positions.get(index - 1));
310 }
311 } else {
312 record Marker(int position, int source) {}
313 var merged = new ArrayList<Marker>();
314 positions.forEach(position -> merged.add(new Marker(position, 0)));
315 others.forEach(other -> merged.add(new Marker(other, 1)));
316 merged.sort(Comparator.comparingInt(Marker::position));
317 for (int index = 1; index < merged.size(); index++) {
318 Marker earlier = merged.get(index - 1);
319 Marker later = merged.get(index);
320 if (earlier.source() != later.source()) {
321 gaps.add(later.position() - earlier.position());
322 }
323 }
324 }
325 return gaps.stream().mapToInt(Integer::intValue).min();
326 }

328 static QueryItem classify(String lexeme) {
329 Kind symbol = SYMBOLS.get(lexeme);
330 if (symbol != null) {
331 return new QueryItem(symbol, lexeme);
332 }
333 if (WORD_PATTERN.matcher(lexeme).matches()) {
334 return new QueryItem(Kind.TERM, lower(lexeme));
335 }
336 if (PREFIX_TERM_PATTERN.matcher(lexeme).matches()) {
337 return new QueryItem(Kind.PREFIX, lower(lexeme.substring(0, lexeme.length() - 1)));
338 }
339 throw new CommandError("BAD TERM " + lexeme);
340 }

374 static final class QueryParser {
375 private final List<QueryItem> items;
376 private final Map<Integer, Set<String>> stemSets;
377 private int position;
378
379 QueryParser(List<QueryItem> items, Map<Integer, Set<String>> stemSets) {
380 this.items = items;
381 this.stemSets = stemSets;
382 }
383
384 private Kind peek() {
385 return position < items.size() ? items.get(position).kind() : null;
386 }
387
388 private QueryItem take() {
389 if (position >= items.size()) {
390 throw new CommandError("SYNTAX");
391 }
392 return items.get(position++);
393 }
394
395 QueryResult parse() {
396 QueryResult result = expression();
397 if (position < items.size()) {
398 throw new CommandError("SYNTAX");
399 }
400 return result;
401 }
402
403 private QueryResult expression() {
404 QueryResult result = conjunction();
405 while (peek() == Kind.OR) {
406 position++;
407 QueryResult right = conjunction();
408 var documents = new TreeSet<>(result.documents());
409 documents.addAll(right.documents());
410 result = new QueryResult(
411 "(" + result.canonical() + " OR " + right.canonical() + ")", documents);
412 }
413 return result;
414 }
415
416 private QueryResult conjunction() {
417 QueryResult result = negation();
418 while (CONJUNCTION_CONTINUATIONS.contains(peek())) {
419 if (peek() == Kind.AND) {
420 position++;
421 }
422 QueryResult right = negation();
423 var documents = new TreeSet<>(result.documents());
424 documents.retainAll(right.documents());
425 result = new QueryResult(
426 "(" + result.canonical() + " AND " + right.canonical() + ")", documents);
427 }
428 return result;
429 }
430
431 private QueryResult negation() {
432 if (peek() == Kind.NOT) {
433 position++;
434 QueryResult inner = negation();
435 var documents = new TreeSet<>(stemSets.keySet());
436 documents.removeAll(inner.documents());
437 return new QueryResult("(NOT " + inner.canonical() + ")", documents);
438 }
439 return element();
440 }
441
442 private QueryResult element() {
443 QueryItem item = take();
444 return switch (item.kind()) {
445 case OPEN -> {
446 QueryResult result = expression();
447 if (take().kind() != Kind.CLOSE) {
448 throw new CommandError("SYNTAX");
449 }
450 yield result;
451 }
452 case TERM -> {
453 String stem = porterStem(item.spelling());
454 yield new QueryResult(stem, documentsWhere(stems -> stems.contains(stem)));
455 }
456 case PREFIX -> {
457 String prefix = item.spelling();
458 yield new QueryResult(prefix + "*", documentsWhere(
459 stems -> stems.stream().anyMatch(stem -> stem.startsWith(prefix))));
460 }
461 default -> throw new CommandError("SYNTAX");
462 };
463 }
464
465 private Set<Integer> documentsWhere(Predicate<Set<String>> condition) {
466 return stemSets.entrySet().stream()
467 .filter(entry -> condition.test(entry.getValue()))
468 .map(Map.Entry::getKey)
469 .collect(Collectors.toCollection(TreeSet::new));
470 }
471 }

660 private void search(List<String> arguments) {
661 List<QueryItem> items = LEXEME_PATTERN.matcher(arguments.get(0)).results()
662 .map(match -> classify(match.group()))
663 .toList();
664 var stemSets = new TreeMap<Integer, Set<String>>();
665 for (Document document : documents.values()) {
666 stemSets.put(document.number, stemSet(document));
667 }
668 QueryResult result = new QueryParser(items, stemSets).parse();
669 for (QueryItem item : items) {
670 if (item.kind() == Kind.TERM && isStopword(item.spelling())) {
671 throw new CommandError("STOPWORD " + item.spelling());
672 }
673 }
674 emit("SEARCH " + result.canonical() + " count=" + result.documents().size(),
675 result.documents().stream().sorted().map(String::valueOf).toList());
676 }

678 private void phrase(List<String> arguments) {
679 List<String> words = arguments.stream().map(Main::wordArgument).toList();
680 if (words.stream().allMatch(this::isStopword)) {
681 throw new CommandError("STOPWORDS ONLY");
682 }
683 List<String> targets = words.stream()
684 .map(word -> isStopword(word) ? "?" : porterStem(word))
685 .toList();
686 var matches = new ArrayList<String>();
687 for (Document document : documents.values()) {
688 List<Token> tokens = document.tokens;
689 for (int start = 0; start + targets.size() <= tokens.size(); start++) {
690 if (phraseMatches(tokens, start, targets)) {
691 matches.add(document.reference(tokens.get(start)));
692 }
693 }
694 }
695 emit("PHRASE \"" + String.join(" ", targets) + "\" matches=" + matches.size(), matches);
696 }

698 private void near(List<String> arguments) {
699 String first = termArgument(arguments.get(0));
700 String second = termArgument(arguments.get(1));
701 int limit = numberArgument(arguments.get(2), 1, MAX_COUNT);
702 var found = new ArrayList<String>();
703 for (Document document : documents.values()) {
704 OptionalInt distance = closestDistance(stemPositions(document, first),
705 stemPositions(document, second), first.equals(second));
706 if (distance.isPresent() && distance.getAsInt() <= limit) {
707 found.add(document.number + ":" + distance.getAsInt());
708 }
709 }
710 emit("NEAR %s %s %d count=%d".formatted(first, second, limit, found.size()), found);
711 }

732 private void kwic(List<String> arguments) {
733 String stem = termArgument(arguments.get(0));
734 List<Occurrence> found = occurrencesOf(stem);
735 emit("KWIC " + stem + " count=" + found.size());
736 for (Occurrence occurrence : found) {
737 Token token = occurrence.token();
738 String text = occurrence.document().lines.get(token.line() - 1);
739 String before =
740 text.substring(Math.max(0, token.start() - KWIC_WIDTH), token.start());
741 int end = Math.min(text.length(), token.end() + KWIC_WIDTH);
742 String after = text.substring(token.end(), end);
743 emit("|" + " ".repeat(KWIC_WIDTH - before.length()) + before
744 + "[" + token.text() + "]"
745 + after + " ".repeat(KWIC_WIDTH - after.length())
746 + "| " + occurrence.reference());
747 }
748 }

773 private void similar(List<String> arguments) {
774 Document document = documentArgument(arguments.get(0));
775 int limit = numberArgument(arguments.get(1), 1, MAX_COUNT);
776 Set<String> stems = stemSet(document);
777 var scores = new ArrayList<Similarity>();
778 for (Document other : documents.values()) {
779 if (other == document) {
780 continue;
781 }
782 Set<String> otherStems = stemSet(other);
783 var common = new HashSet<>(stems);
784 common.retainAll(otherStems);
785 var union = new HashSet<>(stems);
786 union.addAll(otherStems);
787 Fraction score = union.isEmpty()
788 ? new Fraction(0, 1)
789 : Fraction.of(common.size(), union.size());
790 scores.add(new Similarity(other.number, score));
791 }
792 scores.sort(Comparator.comparing(Similarity::score).reversed()
793 .thenComparingInt(Similarity::document));
794 emit("SIMILAR " + document.number, scores.stream()
795 .limit(limit)
796 .map(entry -> entry.document() + ":" + entry.score())
797 .toList());
798 }

800 private void rank(List<String> arguments) {
801 int limit = numberArgument(arguments.get(0), 1, MAX_COUNT);
802 var query = new HashSet<String>();
803 for (String argument : arguments.subList(1, arguments.size())) {
804 query.add(termArgument(argument));
805 }
806 var ranked = new ArrayList<Ranking>();
807 for (Document document : documents.values()) {
808 Map<String, Integer> counts = termFrequencies(document);
809 int matched = (int) query.stream().filter(counts::containsKey).count();
810 int total = query.stream().mapToInt(stem -> counts.getOrDefault(stem, 0)).sum();
811 if (matched > 0) {
812 ranked.add(new Ranking(document.number, matched, total));
813 }
814 }
815 ranked.sort(Comparator.comparingInt((Ranking entry) -> -entry.matched())
816 .thenComparingInt(entry -> -entry.total())
817 .thenComparingInt(Ranking::document));
818 emit("RANK count=" + ranked.size(), ranked.stream()
819 .limit(limit)
820 .map(entry -> entry.document() + ":" + entry.matched() + ":" + entry.total())
821 .toList());
822 }

894 void execute(String line) {
895 int space = line.indexOf(' ');
896 String name = space < 0 ? line : line.substring(0, space);
897 try {
898 Command command = handlers.get(name);
899 if (command == null) {
900 throw new CommandError("UNKNOWN COMMAND " + name);
901 }
902 List<String> arguments =
903 space < 0 ? List.of() : splitArguments(name, line.substring(space + 1));
904 if (arguments.size() < command.minimum() || arguments.size() > command.maximum()) {
905 throw new CommandError("BAD ARGUMENTS " + name);
906 }
907 command.handler().accept(arguments);
908 } catch (CommandError error) {
909 emit("ERROR " + error.getMessage());
910 }
911 }

914 public static void main(String[] args) {
915 var reader = new BufferedReader(new InputStreamReader(System.in));
916 List<String> lines = reader.lines().toList();
917 String[] header = lines.get(0).split(" ");
918 var stopwords = new HashSet<>(Arrays.asList(header).subList(1, header.length));
919 var engine = new SearchEngine(stopwords);
920 int index = 1;
921 while (lines.get(index).startsWith("DOC ")) {
922 Document document = engine.addDocument(Integer.parseInt(lines.get(index).substring(4)));
923 index++;
924 while (!lines.get(index).equals("ENDDOC")) {
925 document.addLine(lines.get(index));
926 index++;
927 }
928 index++;
929 }
930 for (String line : lines.subList(index, lines.size())) {
931 if (line.equals("END")) {
932 break;
933 }
934 engine.execute(line);
935 }
936 System.out.print(engine.output());
937 }
