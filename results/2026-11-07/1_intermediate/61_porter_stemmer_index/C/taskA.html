132static void *reserve(void *items, int *capacity, int needed, size_t item_size)
133{
134 if (needed <= *capacity) {

136 }
137 *capacity = needed > 2 * *capacity ? needed : 2 * *capacity;
138 return xrealloc(items, (size_t)*capacity * item_size);

186static void letter_pattern(const char *word, size_t length, char *kinds)
187{
188 for (size_t index = 0; index < length; index++) {
189 char letter = word[index];
190 bool vowel = strchr(VOWELS, letter) != NULL ||
191 (letter == 'y' && index > 0 && kinds[index - 1] == 'C');
192 kinds[index] = vowel ? 'V' : 'C';
193 }
194 kinds[length] = '\0';
195}

197static int measure(const char *word, size_t length)
198{
199 char kinds[WORD_CAPACITY];
200 letter_pattern(word, length, kinds);
201 int value = 0;
202 for (size_t index = 1; index < length; index++) {
203 if (kinds[index - 1] == 'V' && kinds[index] == 'C') {
204 value++;
205 }
206 }
207 return value;
208}

321static const Rule *apply_rules(char *word, const Rule *rules, size_t rule_count)
322{
323 size_t length = strlen(word);
324 const Rule *selected = NULL;
325 for (size_t index = 0; index < rule_count; index++) {
326 const Rule *rule = &rules[index];
327 if (ends_with(word, length, rule->suffix) &&
328 (selected == NULL || strlen(rule->suffix) > strlen(selected->suffix))) {
329 selected = rule;
330 }
331 }
332 if (selected == NULL) {
333 return NULL;
334 }
335 size_t remainder = length - strlen(selected->suffix);
336 if (!selected->condition(word, remainder)) {
337 return NULL;
338 }
339 strcpy(word + remainder, selected->replacement);
340 return selected;
341}

343static void tidy_step_1b(char *word)
344{
345 size_t length = strlen(word);
346 if (ends_with(word, length, "at") || ends_with(word, length, "bl") ||
347 ends_with(word, length, "iz")) {
348 strcat(word, "e");
349 } else if (ends_double(word, length) && strchr("lsz", word[length - 1]) == NULL) {
350 word[length - 1] = '\0';
351 } else if (measure(word, length) == 1 && ends_cvc(word, length)) {
352 strcat(word, "e");
353 }
354}

356static void apply_step(const Step *step, char *word)
357{
358 size_t length = strlen(word);
359 if (step->rules == NULL) {
360 if (measure(word, length) > 1 && ends_double(word, length) && word[length - 1] == 'l') {
361 word[length - 1] = '\0';
362 }
363 return;
364 }
365 const Rule *rule = apply_rules(word, step->rules, step->rule_count);
366 if (strcmp(step->name, "1b") == 0 && rule != NULL &&
367 (strcmp(rule->suffix, "ed") == 0 || strcmp(rule->suffix, "ing") == 0)) {
368 tidy_step_1b(word);
369 }
370}

372static int stemming_steps(const char *word, char *stem, Change *changes)
373{
374 strcpy(stem, word);
375 if (strlen(word) <= 2) {
376 return 0;
377 }
378 int change_count = 0;
379 for (size_t index = 0; index < LENGTH(STEPS); index++) {
380 char before[WORD_CAPACITY];
381 strcpy(before, stem);
382 apply_step(&STEPS[index], stem);
383 if (strcmp(before, stem) != 0) {
384 changes[change_count].step = STEPS[index].name;
385 strcpy(changes[change_count].word, stem);
386 change_count++;
387 }
388 }
389 return change_count;
390}

392static void porter_stem(const char *word, char *stem)
393{
394 Change changes[LENGTH(STEPS)];
395 stemming_steps(word, stem, changes);
396}

453static void add_token(Document *document, const char *text, int start, int length)
454{
455 document->tokens = reserve(document->tokens, &document->token_capacity,
456 document->token_count + 1, sizeof *document->tokens);
457 char *form = copy_string(text + start, (size_t)length);
458 lowercase(form);
459 char stem[WORD_CAPACITY];
460 porter_stem(form, stem);
461 document->tokens[document->token_count] = (Token){
462 .form = form,
463 .stem = copy_string(stem, strlen(stem)),

465 .position = document->token_count + 1,
466 .start = start,
467 .length = length,
468 };
469 document->token_count++;
470}

472static void add_line(Document *document, const char *text)
473{
474 document->lines = reserve(document->lines, &document->line_capacity, document->line_count + 1,
475 sizeof *document->lines);
476 document->lines[document->line_count++] = copy_string(text, strlen(text));
477 int start = 0;
478 while (text[start] != '\0') {
479 if (!isalpha((unsigned char)text[start])) {
480 start++;
481 continue;
482 }
483 int end = start;
484 while (isalpha((unsigned char)text[end])) {
485 end++;
486 }
487 add_token(document, text, start, end - start);
488 start = end;
489 }

672static int closest_distance(const Document *document, const char *first, const char *second)
673{
674 int best = -1;
675 int last_first = 0;
676 int last_second = 0;
677 for (int index = 0; index < document->token_count; index++) {
678 const Token *token = &document->tokens[index];
679 if (!is_indexed(token)) {
680 continue;
681 }
682 bool is_first = strcmp(token->stem, first) == 0;
683 bool is_second = strcmp(token->stem, second) == 0;
684 if (!is_first && !is_second) {
685 continue;
686 }
687 int previous = is_first ? last_second : last_first;
688 if (previous > 0 && (best < 0 || token->position - previous < best)) {
689 best = token->position - previous;
690 }
691 if (is_first) {
692 last_first = token->position;
693 }
694 if (is_second) {
695 last_second = token->position;
696 }
697 }
698 return best;
699}

757{

839static bool classify(const char *text, size_t length, Item *item)
840{
841 static const struct {
842 const char *text;
843 Kind kind;
844 } SYMBOLS[] = {
845 {"AND", KIND_AND}, {"OR", KIND_OR}, {"NOT", KIND_NOT}, {"(", KIND_OPEN}, {")", KIND_CLOSE},
846 };
847 for (size_t index = 0; index < LENGTH(SYMBOLS); index++) {
848 if (strlen(SYMBOLS[index].text) == length &&
849 strncmp(SYMBOLS[index].text, text, length) == 0) {
850 item->kind = SYMBOLS[index].kind;
851 return true;
852 }
853 }
854 size_t letters = length;
855 item->kind = KIND_TERM;
856 if (length > 1 && text[length - 1] == '*') {
857 letters = length - 1;
858 item->kind = KIND_PREFIX;
859 }
860 if (!is_word(text, letters)) {
861 return false;
862 }
863 memcpy(item->spelling, text, letters);
864 item->spelling[letters] = '\0';
865 lowercase(item->spelling);
866 return true;
867}

874static void combine(Evaluation *left, Evaluation *right, const char *operator_name)
875{
876 char *canonical = format_string("(%s %s %s)", left->canonical, operator_name, right->canonical);
877 free(left->canonical);
878 free(right->canonical);

881 for (int number = 1; number <= MAX_DOCUMENT_NUMBER; number++) {
882 left->documents[number] = is_or ? left->documents[number] || right->documents[number]
883 : left->documents[number] && right->documents[number];
884 }

895static bool parse_expression(Parser *parser, Evaluation *result);

897static bool parse_element(Parser *parser, Evaluation *result)
898{
899 if (parser->position >= parser->count) {
900 return false;
901 }
902 const Item *item = &parser->items[parser->position++];
903 switch (item->kind) {
904 case KIND_OPEN:
905 if (!parse_expression(parser, result)) {
906 return false;
907 }
908 if (peek(parser) != KIND_CLOSE) {
909 free(result->canonical);
910 return false;
911 }
912 parser->position++;
913 return true;
914 case KIND_TERM: {
915 char stem[WORD_CAPACITY];
916 porter_stem(item->spelling, stem);
917 result->canonical = copy_string(stem, strlen(stem));
918 select_documents(result, stem, false);
919 return true;
920 }
921 case KIND_PREFIX:
922 result->canonical = format_string("%s*", item->spelling);
923 select_documents(result, item->spelling, true);
924 return true;
925 default:
926 return false;
927 }
928}

930static bool parse_negation(Parser *parser, Evaluation *result)
931{
932 if (peek(parser) != KIND_NOT) {
933 return parse_element(parser, result);
934 }
935 parser->position++;
936 Evaluation operand;
937 if (!parse_negation(parser, &operand)) {
938 return false;
939 }
940 result->canonical = format_string("(NOT %s)", operand.canonical);
941 free(operand.canonical);
942 for (int number = 1; number <= MAX_DOCUMENT_NUMBER; number++) {
943 result->documents[number] = documents[number] != NULL && !operand.documents[number];
944 }
945 return true;
946}

948static bool continues_conjunction(Kind kind)
949{
950 return kind == KIND_AND || kind == KIND_NOT || kind == KIND_OPEN || kind == KIND_TERM ||
951 kind == KIND_PREFIX;
952}
953
954static bool parse_conjunction(Parser *parser, Evaluation *result)
955{
956 if (!parse_negation(parser, result)) {
957 return false;
958 }
959 while (continues_conjunction(peek(parser))) {
960 if (peek(parser) == KIND_AND) {
961 parser->position++;
962 }
963 Evaluation right;
964 if (!parse_negation(parser, &right)) {
965 free(result->canonical);
966 return false;
967 }
968 combine(result, &right, "AND");
969 }
970 return true;
971}
972
973static bool parse_expression(Parser *parser, Evaluation *result)
974{
975 if (!parse_conjunction(parser, result)) {
976 return false;
977 }
978 while (peek(parser) == KIND_OR) {
979 parser->position++;
980 Evaluation right;
981 if (!parse_conjunction(parser, &right)) {
982 free(result->canonical);
983 return false;
984 }
985 combine(result, &right, "OR");
986 }
987 return true;
988}

990static bool parse_query(Parser *parser, Evaluation *result)
991{
992 if (!parse_expression(parser, result)) {
993 return false;
994 }
995 if (parser->position == parser->count) {
996 return true;
997 }
998 free(result->canonical);
999 return false;
1000}

1484 Document *document = created ? create_document(number) : documents[number];
1485 int first = document->token_count;
1486 add_line(document, arguments[1]);
1487 int indexed = 0;
1488 for (int index = first; index < document->token_count; index++) {
1489 indexed += is_indexed(&document->tokens[index]);
1490 }
1491 printf("APPEND %d line=%d tokens=%d indexed=%d new=%s\n", number, document->line_count,
1492 document->token_count - first, indexed, yes_no(created));
1493}

1531static int split_arguments(const char *name, char *rest, char **arguments)
1532{
1533 if (strcmp(name, "SEARCH") == 0) {
1534 arguments[0] = rest;
1535 return 1;
1536 }
1537 bool is_append = strcmp(name, "APPEND") == 0;
1538 int count = 0;
1539 char *part = rest;
1540 while (true) {
1541 char *space = strchr(part, ' ');
1542 if (count < MAX_ARGUMENTS) {
1543 arguments[count] = part;
1544 }
1545 count++;
1546 if (space == NULL || (is_append && count == 2)) {
1547 return count;
1548 }
1549 *space = '\0';
1550 part = space + 1;
1551 }
1552}

1554static void execute(char *line)
1555{
1556 char *rest = strchr(line, ' ');
1557 if (rest != NULL) {
1558 *rest++ = '\0';
1559 }
1560 const Command *command = NULL;
1561 for (size_t index = 0; index < LENGTH(COMMANDS); index++) {
1562 if (strcmp(COMMANDS[index].name, line) == 0) {
1563 command = &COMMANDS[index];
1564 }
1565 }
1566 if (command == NULL) {
1567 printf("ERROR UNKNOWN COMMAND %s\n", line);
1568 return;
1569 }
1570 char *arguments[MAX_ARGUMENTS];
1571 int count = rest == NULL ? 0 : split_arguments(command->name, rest, arguments);
1572 if (count < command->minimum || count > command->maximum) {
1573 printf("ERROR BAD ARGUMENTS %s\n", command->name);
1574 return;
1575 }
1576 command->handler(arguments, count);
1577}

1579static bool read_line(char *line)
1580{

1582 return false;
1583 }

1585 return true;

1588int main(void)
1589{
1590 char line[LINE_CAPACITY];

1592 return EXIT_FAILURE;
1593 }
1594 strtok(line, " ");
1595 for (char *word = strtok(NULL, " "); word != NULL; word = strtok(NULL, " ")) {
1596 add_stopword(word);
1597 }
1598 bool more = read_line(line);
1599 while (more && strncmp(line, "DOC ", 4) == 0) {
1600 Document *document = create_document(atoi(line + 4));
1601 while (read_line(line) && strcmp(line, "ENDDOC") != 0) {
1602 add_line(document, line);
1603 }
1604 more = read_line(line);
1605 }
1606 while (more && strcmp(line, "END") != 0) {
1607 execute(line);
1608 more = read_line(line);
1609 }
1610 return EXIT_SUCCESS;
