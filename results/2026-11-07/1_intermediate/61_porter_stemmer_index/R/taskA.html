31letter_pattern <- function(word) {
32 indices <- seq_len(nchar(word))
33 word_letters <- substring(word, indices, indices)
34 kinds <- character(length(word_letters))
35 for (index in indices) {
36 letter <- word_letters[index]
37 vowel <- letter %in% VOWELS ||
38 (letter == "y" && index > 1 && kinds[index - 1] == "C")
39 kinds[index] <- if (vowel) "V" else "C"
40 }
41 paste(kinds, collapse = "")
42}

110apply_rules <- function(word, rules) {
111 matching <- Filter(function(candidate) endsWith(word, candidate$suffix), rules)
112 if (length(matching) == 0) {
113 return(list(word = word, suffix = NA_character_))
114 }
115 suffix_lengths <- vapply(matching, function(candidate) nchar(candidate$suffix), integer(1))
116 selected <- matching[[which.max(suffix_lengths)]]
117 remainder <- substr(word, 1, nchar(word) - nchar(selected$suffix))
118 if (!selected$condition(remainder)) {
119 return(list(word = word, suffix = NA_character_))
120 }
121 list(word = paste0(remainder, selected$replacement), suffix = selected$suffix)
122}

124tidy_step_1b <- function(word) {
125 if (endsWith(word, "at") || endsWith(word, "bl") || endsWith(word, "iz")) {
126 return(paste0(word, "e"))
127 }
128 if (ends_double(word) && !(last_letter(word) %in% c("l", "s", "z"))) {
129 return(drop_last(word))
130 }
131 if (measure(word) == 1 && ends_cvc(word)) {
132 return(paste0(word, "e"))
133 }
134 word
135}

137apply_step <- function(name, word) {
138 if (name == "5b") {
139 if (measure(word) > 1 && ends_double(word) && endsWith(word, "l")) {
140 return(drop_last(word))
141 }
142 return(word)
143 }
144 outcome <- apply_rules(word, RULE_TABLES[[name]])
145 if (name == "1b" && outcome$suffix %in% c("ed", "ing")) {
146 return(tidy_step_1b(outcome$word))
147 }
148 outcome$word
149}

151stemming_steps <- function(word) {
152 changes <- character(0)
153 if (nchar(word) > 2) {
154 for (name in STEP_NAMES) {
155 result <- apply_step(name, word)
156 if (result != word) {
157 changes[name] <- result
158 }
159 word <- result
160 }
161 }
162 list(stem = word, changes = changes)
163}

226closest_distance <- function(positions, others, same_stem) {
227 if (same_stem) {
228 gaps <- diff(positions)
229 } else {
230 merged <- c(positions, others)
231 sources <- rep(c(0, 1), c(length(positions), length(others)))
232 ordering <- order(merged)
233 merged <- merged[ordering]
234 sources <- sources[ordering]
235 adjacent <- which(diff(sources) != 0)
236 gaps <- merged[adjacent + 1] - merged[adjacent]
237 }
238 if (length(gaps) == 0) NULL else min(gaps)
239}

340parse_expression <- function(parser) {
341 result <- parse_conjunction(parser)
342 while (peek_kind(parser) == "OR") {
343 parser$position <- parser$position + 1
344 right <- parse_conjunction(parser)
345 result <- list(
346 canonical = sprintf("(%s OR %s)", result$canonical, right$canonical),
347 documents = union(result$documents, right$documents)
348 )
349 }
350 result
351}
352
353parse_conjunction <- function(parser) {
354 result <- parse_negation(parser)
355 while (peek_kind(parser) %in% CONJUNCTION_CONTINUATIONS) {
356 if (peek_kind(parser) == "AND") {
357 parser$position <- parser$position + 1
358 }
359 right <- parse_negation(parser)
360 result <- list(
361 canonical = sprintf("(%s AND %s)", result$canonical, right$canonical),
362 documents = intersect(result$documents, right$documents)
363 )
364 }
365 result
366}
367
368parse_negation <- function(parser) {
369 if (peek_kind(parser) != "NOT") {
370 return(parse_element(parser))
371 }
372 parser$position <- parser$position + 1
373 inner <- parse_negation(parser)
374 list(
375 canonical = sprintf("(NOT %s)", inner$canonical),
376 documents = setdiff(as.integer(names(parser$stem_sets)), inner$documents)
377 )
378}

380parse_element <- function(parser) {
381 item <- take_item(parser)
382 if (item$kind == "(") {
383 result <- parse_expression(parser)
384 if (take_item(parser)$kind != ")") {
385 command_error("SYNTAX")
386 }
387 return(result)
388 }
389 if (item$kind == "term") {
390 stem <- porter_stem(item$spelling)
391 return(list(canonical = stem, documents = documents_where(parser, \(stems) stem %in% stems)))
392 }
393 if (item$kind == "prefix") {
394 matches <- documents_where(parser, \(stems) any(startsWith(stems, item$spelling)))
395 return(list(canonical = paste0(item$spelling, "*"), documents = matches))
396 }
397 command_error("SYNTAX")
398}

470phrase_starts <- function(engine, tokens, targets) {
471 count <- nrow(tokens) - length(targets) + 1
472 if (count < 1) {
473 return(integer(0))
474 }
475 starts <- seq_len(count)
476 matched <- rep(TRUE, count)
477 indexed <- is_indexed(engine, tokens)
478 for (offset in seq_along(targets)) {
479 if (targets[offset] != "?") {
480 rows <- starts + offset - 1
481 matched <- matched & indexed[rows] & tokens$stem[rows] == targets[offset]
482 }
483 }
484 starts[matched]
485}

612 occurrences <- occurrence_table(engine)
613 found <- occurrences[occurrences$stem == stem, ]
614 emit("KWIC", stem, paste0("count=", nrow(found)))
615 for (row in seq_len(nrow(found))) {
616 occurrence <- found[row, ]
617 text <- find_document(engine, occurrence$document)$lines[occurrence$line]
618 after_start <- occurrence$start + nchar(occurrence$text)
619 before <- substr(text, max(1, occurrence$start - KWIC_WIDTH), occurrence$start - 1)
620 after <- substr(text, after_start, after_start + KWIC_WIDTH - 1)
621 context <- paste0(
622 "|", formatC(before, width = KWIC_WIDTH), "[", occurrence$text, "]",
623 formatC(after, width = KWIC_WIDTH, flag = "-"), "|"
624 )
625 emit(context, reference(occurrence))
626 }

648similar <- function(engine, arguments) {
649 document <- document_argument(engine, arguments[1])
650 limit <- number_argument(arguments[2], 1, MAX_COUNT)
651 stems <- stem_set(engine, document)
652 others <- Filter(\(other) other$number != document$number, ordered_documents(engine))
653 numbers <- vapply(others, \(other) other$number, integer(1))
654 scores <- vapply(others, function(other) {
655 other_stems <- stem_set(engine, other)
656 jaccard(length(intersect(stems, other_stems)), length(union(stems, other_stems)))
657 }, numeric(2))
658 shown <- head(order(-scores[1, ] / scores[2, ], numbers), limit)
659 emit(
660 "SIMILAR", document$number,
661 sprintf("%d:%d/%d", numbers[shown], scores[1, shown], scores[2, shown])
662 )
663}

677 shown <- head(ranked, limit)
678 emit(
679 paste0("RANK count=", nrow(ranked)),

681 )

698 if (line_number > length(document$lines)) {
699 command_error(sprintf("NO LINE %d:%d", document$number, line_number))
700 }
701 tokens <- document$tokens[document$tokens$line == line_number, ]
702 stems <- ifelse(is_indexed(engine, tokens), tokens$stem, "-")
703 emit(
704 sprintf("LINE %d:%d", document$number, line_number),
705 paste0("tokens=", nrow(tokens)), sprintf("%s/%s", tokens$form, stems)
706 )
707}

788execute <- function(engine, line) {
789 space <- regexpr(" ", line, fixed = TRUE)
790 name <- if (space > 0) substr(line, 1, space - 1) else line
791 tryCatch(
792 {
793 command <- COMMANDS[[name]]
794 if (is.null(command)) {
795 command_error(paste("UNKNOWN COMMAND", name))
796 }
797 arguments <- character(0)
798 if (space > 0) {
799 arguments <- split_arguments(name, substring(line, space + 1))
800 }
801 if (length(arguments) < command$minimum || length(arguments) > command$maximum) {
802 command_error(paste("BAD ARGUMENTS", name))
803 }
804 command$handler(engine, arguments)
805 },
806 command_error = function(error) emit("ERROR", conditionMessage(error))
807 )
808}

810main <- function() {
811 lines <- readLines(file("stdin"), warn = FALSE)
812 engine <- new_engine(strsplit(lines[1], " ", fixed = TRUE)[[1]][-1])
813 index <- 2
814 while (startsWith(lines[index], "DOC ")) {
815 document <- add_document(engine, as.integer(substring(lines[index], 5)))
816 index <- index + 1
817 while (lines[index] != "ENDDOC") {
818 add_line(document, lines[index])
819 index <- index + 1
820 }
821 index <- index + 1
822 }
823 for (line in lines[index:length(lines)]) {
824 if (line == "END") {
825 break
826 }
827 execute(engine, line)
828 }
829}
