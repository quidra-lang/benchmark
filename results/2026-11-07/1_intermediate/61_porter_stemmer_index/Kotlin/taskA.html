19fun letterPattern(word: String): String {
20 val kinds = StringBuilder()
21 for ((index, letter) in word.withIndex()) {
22 val vowel = letter in VOWELS || (letter == 'y' && index > 0 && kinds[index - 1] == 'C')
23 kinds.append(if (vowel) 'V' else 'C')
24 }
25 return kinds.toString()
26}
27
28fun measure(word: String): Int = letterPattern(word).windowed(2).count { it == "VC" }
29
30fun hasVowel(word: String): Boolean = 'V' in letterPattern(word)
31
32fun endsDouble(word: String): Boolean =
33 letterPattern(word).endsWith("CC") && word.last() == word[word.length - 2]
34
35fun endsCvc(word: String): Boolean =
36 letterPattern(word).endsWith("CVC") && word.last() !in "wxy"
37
38fun measureAboveZero(remainder: String): Boolean = measure(remainder) > 0
39
40fun measureAboveOne(remainder: String): Boolean = measure(remainder) > 1
41
42fun ionCondition(remainder: String): Boolean =
43 measure(remainder) > 1 && (remainder.endsWith("s") || remainder.endsWith("t"))
44
45fun finalECondition(remainder: String): Boolean {
46 val value = measure(remainder)
47 return value > 1 || (value == 1 && !endsCvc(remainder))
48}

84data class RuleOutcome(val word: String, val suffix: String?)
85
86fun applyRules(word: String, rules: List<Rule>): RuleOutcome {
87 val rule = rules.filter { word.endsWith(it.suffix) }.maxByOrNull { it.suffix.length }
88 ?: return RuleOutcome(word, null)
89 val remainder = word.dropLast(rule.suffix.length)
90 if (!rule.condition(remainder)) return RuleOutcome(word, null)
91 return RuleOutcome(remainder + rule.replacement, rule.suffix)
92}

94fun tidyStep1b(word: String): String = when {
95 listOf("at", "bl", "iz").any { word.endsWith(it) } -> word + "e"
96 endsDouble(word) && word.last() !in "lsz" -> word.dropLast(1)
97 measure(word) == 1 && endsCvc(word) -> word + "e"
98 else -> word
99}

101fun applyStep(name: String, word: String): String {
102 if (name == "5b") {
103 val removable = measure(word) > 1 && endsDouble(word) && word.endsWith("l")
104 return if (removable) word.dropLast(1) else word
105 }
106 val (result, suffix) = applyRules(word, RULE_TABLES.getValue(name))
107 return if (name == "1b" && (suffix == "ed" || suffix == "ing")) tidyStep1b(result) else result
108}

112data class Stemming(val stem: String, val changes: List<StepChange>)

114fun stemmingSteps(word: String): Stemming {
115 val changes = mutableListOf<StepChange>()
116 var current = word
117 if (word.length > 2) {
118 for (name in STEP_NAMES) {
119 val result = applyStep(name, current)
120 if (result != current) changes.add(StepChange(name, result))
121 current = result
122 }
123 }
124 return Stemming(current, changes)
125}

154data class Fraction(val numerator: Int, val denominator: Int) : Comparable<Fraction> {
155 override fun compareTo(other: Fraction): Int =
156 (numerator.toLong() * other.denominator).compareTo(other.numerator.toLong() * denominator)
157
158 override fun toString(): String = "$numerator/$denominator"
159
160 companion object {
161 fun of(numerator: Int, denominator: Int): Fraction {
162 val divisor = gcd(numerator, denominator)
163 return Fraction(numerator / divisor, denominator / divisor)
164 }
165 }

172fun closestDistance(positions: List<Int>, others: List<Int>, sameStem: Boolean): Int? {
173 val gaps = if (sameStem) {
174 positions.zipWithNext { earlier, later -> later - earlier }
175 } else {
176 val merged = (positions.map { it to 0 } + others.map { it to 1 }).sortedBy { it.first }
177 merged.zipWithNext()
178 .filter { (earlier, later) -> earlier.second != later.second }
179 .map { (earlier, later) -> later.first - earlier.first }
180 }
181 return gaps.minOrNull()
182}

190private val SYMBOLS = mapOf(
191 "AND" to Kind.AND, "OR" to Kind.OR, "NOT" to Kind.NOT, "(" to Kind.OPEN, ")" to Kind.CLOSE,
192)
193private val CONJUNCTION_CONTINUATIONS = setOf(Kind.AND, Kind.NOT, Kind.OPEN, Kind.TERM, Kind.PREFIX)

196 SYMBOLS[lexeme]?.let { return QueryItem(it, lexeme) }

198 WORD_PATTERN.matches(lexeme) -> QueryItem(Kind.TERM, lexeme.lowercase())

202 }

238class QueryParser(private val items: List<QueryItem>, private val stemSets: Map<Int, Set<String>>) {
239 private var position = 0
240
241 private fun peek(): Kind? = items.getOrNull(position)?.kind
242
243 private fun take(): QueryItem {
244 if (position >= items.size) throw CommandError("SYNTAX")
245 return items[position++]
246 }
247
248 fun parse(): QueryResult {
249 val result = expression()
250 if (position < items.size) throw CommandError("SYNTAX")
251 return result
252 }
253
254 private fun expression(): QueryResult {
255 var result = conjunction()
256 while (peek() == Kind.OR) {
257 position++
258 val right = conjunction()
259 result = QueryResult(
260 "(${result.canonical} OR ${right.canonical})",
261 result.documents union right.documents,
262 )
263 }
264 return result
265 }
266
267 private fun conjunction(): QueryResult {
268 var result = negation()
269 while (peek() in CONJUNCTION_CONTINUATIONS) {
270 if (peek() == Kind.AND) position++
271 val right = negation()
272 result = QueryResult(
273 "(${result.canonical} AND ${right.canonical})",
274 result.documents intersect right.documents,
275 )
276 }
277 return result
278 }
279
280 private fun negation(): QueryResult {
281 if (peek() == Kind.NOT) {
282 position++
283 val inner = negation()
284 return QueryResult("(NOT ${inner.canonical})", stemSets.keys - inner.documents)
285 }
286 return element()
287 }
288
289 private fun element(): QueryResult {
290 val item = take()
291 return when (item.kind) {
292 Kind.OPEN -> {
293 val result = expression()
294 if (take().kind != Kind.CLOSE) throw CommandError("SYNTAX")
295 result
296 }
297 Kind.TERM -> {
298 val stem = porterStem(item.spelling)
299 QueryResult(stem, documentsWhere { stem in it })
300 }
301 Kind.PREFIX -> QueryResult(
302 "${item.spelling}*",
303 documentsWhere { stems -> stems.any { it.startsWith(item.spelling) } },
304 )
305 else -> throw CommandError("SYNTAX")
306 }
307 }
308
309 private fun documentsWhere(condition: (Set<String>) -> Boolean): Set<Int> =
310 stemSets.filterValues(condition).keys
311}

427 private fun search(arguments: List<String>) {
428 val items = LEXEME_PATTERN.findAll(arguments[0]).map { classify(it.value) }.toList()
429 val stemSets = documents.values.associate { it.number to stemSet(it) }
430 val (canonical, matches) = QueryParser(items, stemSets).parse()
431 for (item in items) {
432 if (item.kind == Kind.TERM && isStopword(item.spelling)) {
433 throw CommandError("STOPWORD ${item.spelling}")
434 }
435 }
436 emit("SEARCH $canonical count=${matches.size}", matches.sorted().map { it.toString() })

439 private fun phrase(arguments: List<String>) {
440 val words = arguments.map(::wordArgument)
441 if (words.all(::isStopword)) throw CommandError("STOPWORDS ONLY")
442 val targets = words.map { if (isStopword(it)) "?" else porterStem(it) }
443 val matches = mutableListOf<String>()
444 for (document in documents.values) {
445 val tokens = document.tokens
446 for (start in 0..tokens.size - targets.size) {
447 if (targets.indices.all { tokenMatches(tokens[start + it], targets[it]) }) {
448 matches.add(document.reference(tokens[start]))
449 }
450 }
451 }
452 emit("PHRASE \"${targets.joinToString(" ")}\" matches=${matches.size}", matches)
453 }

484 val found = occurrencesOf(stem)
485 emit("KWIC $stem count=${found.size}")

514 private fun similar(arguments: List<String>) {
515 val document = documentArgument(arguments[0])
516 val limit = numberArgument(arguments[1], 1, MAX_COUNT)
517 val stems = stemSet(document)
518 val scores = documents.values.filter { it !== document }.map { other ->
519 val otherStems = stemSet(other)
520 val union = (stems union otherStems).size
521 val score = if (union > 0) {
522 Fraction.of((stems intersect otherStems).size, union)
523 } else {
524 Fraction(0, 1)
525 }
526 Similarity(other.number, score)
527 }
528 val ranked = scores.sortedWith(
529 compareByDescending<Similarity> { it.score }.thenBy { it.document },
530 )
531 emit("SIMILAR ${document.number}", ranked.take(limit).map { "${it.document}:${it.score}" })
532 }

534 private fun rank(arguments: List<String>) {
535 val limit = numberArgument(arguments[0], 1, MAX_COUNT)
536 val query = arguments.drop(1).map(::termArgument).toSet()
537 val ranked = documents.values.mapNotNull { document ->
538 val counts = termFrequencies(document)
539 val matched = query.count { it in counts }
540 val total = query.sumOf { counts[it] ?: 0 }
541 if (matched > 0) Ranking(document.number, matched, total) else null
542 }.sortedWith(
543 compareByDescending<Ranking> { it.matched }
544 .thenByDescending { it.total }
545 .thenBy { it.document },
546 )
547 emit(
548 "RANK count=${ranked.size}",
549 ranked.take(limit).map { "${it.document}:${it.matched}:${it.total}" },
550 )
551 }

623 fun execute(line: String) {
624 val name = line.substringBefore(" ")
625 try {
626 val command = commands[name] ?: throw CommandError("UNKNOWN COMMAND $name")
627 val arguments =
628 if (" " in line) splitArguments(name, line.substringAfter(" ")) else emptyList()
629 if (arguments.size !in command.arity) throw CommandError("BAD ARGUMENTS $name")
630 command.handler(arguments)
631 } catch (error: CommandError) {
632 emit("ERROR ${error.message}")
633 }
634 }
635}

637fun main() {
638 val lines = System.`in`.bufferedReader().readLines()
639 val engine = SearchEngine(lines[0].split(" ").drop(1).toMutableSet())
640 var index = 1
641 while (lines[index].startsWith("DOC ")) {
642 val document = engine.addDocument(lines[index].substring(4).toInt())
643 index++
644 while (lines[index] != "ENDDOC") {
645 document.addLine(lines[index])
646 index++
647 }
648 index++
649 }
650 for (line in lines.drop(index)) {
651 if (line == "END") break
652 engine.execute(line)
653 }
654 print(engine.output)
655}
