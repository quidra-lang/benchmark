65function letterPattern(word: string): string {
66 let kinds = '';
67 for (let index = 0; index < word.length; index++) {
68 const letter = word[index];
69 const vowel =
70 VOWELS.includes(letter) || (letter === 'y' && index > 0 && kinds[index - 1] === 'C');
71 kinds += vowel ? 'V' : 'C';
72 }
73 return kinds;
74}

76function measure(word: string): number {
77 return letterPattern(word).split('VC').length - 1;

80function hasVowel(word: string): boolean {
81 return letterPattern(word).includes('V');

84function endsDouble(word: string): boolean {
85 return letterPattern(word).endsWith('CC') && word.at(-1) === word.at(-2);

88function endsCvc(word: string): boolean {
89 return letterPattern(word).endsWith('CVC') && !'wxy'.includes(word.slice(-1));

113function rulesWith(condition: Condition, replacements: Record<string, string>): Rule[] {
114 return Object.entries(replacements).map(([suffix, replacement]) => ({

168function applyRules(word: string, rules: Rule[]): [word: string, suffix: string | null] {
169 const matching = rules.filter((rule) => word.endsWith(rule.suffix));
170 if (matching.length === 0) {
171 return [word, null];
172 }
173 const { suffix, replacement, condition } = matching.reduce((best, rule) =>
174 rule.suffix.length > best.suffix.length ? rule : best,
175 );
176 const remainder = word.slice(0, -suffix.length);
177 if (!condition(remainder)) {
178 return [word, null];
179 }
180 return [remainder + replacement, suffix];
181}

183function tidyStep1b(word: string): string {
184 if (['at', 'bl', 'iz'].some((suffix) => word.endsWith(suffix))) {
185 return word + 'e';
186 }
187 if (endsDouble(word) && !'lsz'.includes(word.slice(-1))) {
188 return word.slice(0, -1);
189 }
190 if (measure(word) === 1 && endsCvc(word)) {
191 return word + 'e';
192 }
193 return word;
194}

196function applyStep(name: string, word: string): string {
197 if (name === '5b') {
198 if (measure(word) > 1 && endsDouble(word) && word.endsWith('l')) {
199 return word.slice(0, -1);
200 }
201 return word;
202 }
203 const [result, suffix] = applyRules(word, RULE_TABLES[name]);
204 if (name === '1b' && (suffix === 'ed' || suffix === 'ing')) {
205 return tidyStep1b(result);
206 }
207 return result;
208}

210function stemmingSteps(word: string): [stem: string, changes: [name: string, result: string][]] {
211 const changes: [name: string, result: string][] = [];
212 if (word.length > 2) {
213 for (const name of STEP_NAMES) {
214 const result = applyStep(name, word);
215 if (result !== word) {
216 changes.push([name, result]);
217 }
218 word = result;
219 }
220 }
221 return [word, changes];
222}

291function closestDistance(positions: number[], others: number[], sameStem: boolean): number | null {
292 const gaps: number[] = [];
293 if (sameStem) {
294 for (let index = 1; index < positions.length; index++) {
295 gaps.push(positions[index] - positions[index - 1]);
296 }
297 } else {
298 const merged = [
299 ...positions.map((position): [number, number] => [position, 0]),
300 ...others.map((other): [number, number] => [other, 1]),
301 ].sort((left, right) => left[0] - right[0]);
302 for (let index = 1; index < merged.length; index++) {
303 if (merged[index - 1][1] !== merged[index][1]) {
304 gaps.push(merged[index][0] - merged[index - 1][0]);
305 }
306 }
307 }
308 return gaps.length > 0 ? Math.min(...gaps) : null;
309}

311function classify(lexeme: string): Item {
312 if (OPERATORS.has(lexeme) || PARENTHESES.has(lexeme)) {
313 return [lexeme as Kind, lexeme];
314 }
315 if (WORD_PATTERN.test(lexeme)) {
316 return ['term', lexeme.toLowerCase()];
317 }
318 if (PREFIX_TERM_PATTERN.test(lexeme)) {
319 return ['prefix', lexeme.slice(0, -1).toLowerCase()];
320 }
321 throw new CommandError(`BAD TERM ${lexeme}`);
322}

328function fraction(numerator: number, denominator: number): Fraction {
329 const divisor = gcd(numerator, denominator);

333function compareFractions(left: Fraction, right: Fraction): number {

386class QueryParser {
387 private readonly items: Item[];
388 private readonly stemSets: Map<number, Set<string>>;
389 private position = 0;
390
391 constructor(items: Item[], stemSets: Map<number, Set<string>>) {
392 this.items = items;
393 this.stemSets = stemSets;
394 }
395
396 private peek(): Kind | null {
397 return this.position < this.items.length ? this.items[this.position][0] : null;
398 }
399
400 private take(): Item {
401 if (this.position >= this.items.length) {
402 throw new CommandError('SYNTAX');
403 }
404 return this.items[this.position++];
405 }
406
407 parse(): Evaluation {
408 const result = this.expression();
409 if (this.position < this.items.length) {
410 throw new CommandError('SYNTAX');
411 }
412 return result;
413 }
414
415 private expression(): Evaluation {
416 let [canonical, documents] = this.conjunction();
417 while (this.peek() === 'OR') {
418 this.position += 1;
419 const [right, rightDocuments] = this.conjunction();
420 canonical = `(${canonical} OR ${right})`;
421 documents = documents.union(rightDocuments);
422 }
423 return [canonical, documents];
424 }
425
426 private conjunction(): Evaluation {
427 let [canonical, documents] = this.negation();
428 while (CONJUNCTION_CONTINUATIONS.has(this.peek())) {
429 if (this.peek() === 'AND') {
430 this.position += 1;
431 }
432 const [right, rightDocuments] = this.negation();
433 canonical = `(${canonical} AND ${right})`;
434 documents = documents.intersection(rightDocuments);
435 }
436 return [canonical, documents];
437 }
438
439 private negation(): Evaluation {
440 if (this.peek() === 'NOT') {
441 this.position += 1;
442 const [canonical, documents] = this.negation();
443 return [`(NOT ${canonical})`, new Set(this.stemSets.keys()).difference(documents)];
444 }
445 return this.element();
446 }
447
448 private element(): Evaluation {
449 const [kind, spelling] = this.take();
450 if (kind === '(') {
451 const result = this.expression();
452 if (this.take()[0] !== ')') {
453 throw new CommandError('SYNTAX');
454 }
455 return result;
456 }
457 if (kind === 'term') {
458 const stem = porterStem(spelling);
459 return [stem, this.documentsWhere((stems) => stems.has(stem))];
460 }
461 if (kind === 'prefix') {
462 const documents = this.documentsWhere((stems) =>
463 [...stems].some((stem) => stem.startsWith(spelling)),
464 );
465 return [`${spelling}*`, documents];
466 }
467 throw new CommandError('SYNTAX');
468 }
469
470 private documentsWhere(predicate: (stems: Set<string>) => boolean): Set<number> {
471 const numbers = [...this.stemSets].filter(([, stems]) => predicate(stems));
472 return new Set(numbers.map(([number]) => number));
473 }
474}

491 ['NEAR', [3, 3, this.near]],

548 private occurrenceMap(): Map<string, Occurrence[]> {

558 found.push([document, token]);

634 private showIndex(args: string[]): void {
635 const prefix = args.length > 0 ? wordArgument(args[0]) : '';
636 const occurrences = this.occurrenceMap();
637 const stems = [...occurrences.keys()].filter((stem) => stem.startsWith(prefix)).sort();
638 this.emit('INDEX', `stems=${stems.length}`);
639 for (const stem of stems) {
640 const found = occurrences.get(stem) ?? [];
641 this.emit(
642 stem,
643 `df=${documentCount(found)}`,
644 ...found.map(([document, token]) => document.reference(token)),
645 );
646 }

649 private search(args: string[]): void {
650 const items = (args[0].match(LEXEME_PATTERN) ?? []).map(classify);
651 const stemSets = new Map(
652 [...this.documents.values()].map((document) => [document.number, this.stemSet(document)]),
653 );
654 const [canonical, matches] = new QueryParser(items, stemSets).parse();
655 for (const [kind, spelling] of items) {
656 if (kind === 'term' && this.stopwords.has(spelling)) {
657 throw new CommandError(`STOPWORD ${spelling}`);
658 }
659 }
660 const numbers = [...matches].sort((left, right) => left - right);
661 this.emit('SEARCH', canonical, `count=${matches.size}`, ...numbers);
662 }

664 private phrase(args: string[]): void {
665 const words = args.map(wordArgument);
666 if (words.every((word) => this.stopwords.has(word))) {
667 throw new CommandError('STOPWORDS ONLY');
668 }
669 const targets = words.map((word) => (this.stopwords.has(word) ? '?' : porterStem(word)));
670 const matches: string[] = [];
671 for (const document of this.orderedDocuments()) {
672 const tokens = document.tokens;
673 for (let start = 0; start <= tokens.length - targets.length; start++) {
674 if (targets.every((target, offset) => this.tokenMatches(tokens[start + offset], target))) {
675 matches.push(document.reference(tokens[start]));
676 }
677 }
678 }
679 this.emit(`PHRASE "${targets.join(' ')}"`, `matches=${matches.length}`, ...matches);
680 }

682 private near(args: string[]): void {
683 const first = this.termArgument(args[0]);
684 const second = this.termArgument(args[1]);
685 const limit = numberArgument(args[2], 1, MAX_COUNT);
686 const found: string[] = [];
687 for (const document of this.orderedDocuments()) {
688 const distance = closestDistance(
689 this.stemPositions(document, first),
690 this.stemPositions(document, second),
691 first === second,
692 );
693 if (distance !== null && distance <= limit) {
694 found.push(`${document.number}:${distance}`);
695 }
696 }
697 this.emit('NEAR', first, second, limit, `count=${found.length}`, ...found);
698 }

700 private top(args: string[]): void {
701 const limit = numberArgument(args[0], 1, MAX_COUNT);
702 let counts: Map<string, number>;
703 let header: string;
704 if (args.length === 2) {
705 const document = this.documentArgument(args[1]);
706 counts = this.termFrequencies(document);
707 header = `TOP ${limit} doc=${document.number}`;
708 } else {
709 counts = this.totalFrequencies();
710 header = `TOP ${limit}`;
711 }
712 const ranked = [...counts].sort(byCount).slice(0, limit);
713 this.emit(header, ...ranked.map(([stem, count]) => `${stem}:${count}`));
714 }

716 private kwic(args: string[]): void {
717 const stem = this.termArgument(args[0]);
718 const found = this.occurrenceMap().get(stem) ?? [];
719 this.emit('KWIC', stem, `count=${found.length}`);
720 for (const [document, token] of found) {
721 const text = document.lines[token.line - 1];
722 const before = text.slice(Math.max(0, token.start - KWIC_WIDTH), token.start);
723 const after = text.slice(token.end, token.end + KWIC_WIDTH);
724 this.emit(
725 `|${before.padStart(KWIC_WIDTH)}[${token.text}]${after.padEnd(KWIC_WIDTH)}|`,
726 document.reference(token),
727 );
728 }
729 }

732 const document = this.documentArgument(args[0]);
733 const stem = this.termArgument(args[1]);
734 this.emit('TF', document.number, stem, this.termFrequencies(document).get(stem) ?? 0);

737 private documentFrequency(args: string[]): void {
738 const stem = this.termArgument(args[0]);
739 this.emit('DF', stem, documentCount(this.occurrenceMap().get(stem) ?? []));
740 }
741
742 private variants(args: string[]): void {
743 const stem = this.termArgument(args[0]);
744 const found = this.occurrenceMap().get(stem) ?? [];
745 const forms = [...countValues(found.map(([, token]) => token.form))].sort(([left], [right]) =>
746 compareStrings(left, right),
747 );
748 this.emit(
749 'VARIANTS',
750 stem,
751 `count=${forms.length}`,
752 ...forms.map(([form, count]) => `${form}:${count}`),
753 );
754 }
755
756 private similar(args: string[]): void {
757 const document = this.documentArgument(args[0]);
758 const limit = numberArgument(args[1], 1, MAX_COUNT);
759 const stems = this.stemSet(document);
760 const scores: { score: Fraction; number: number }[] = [];
761 for (const other of this.documents.values()) {
762 if (other === document) {
763 continue;
764 }
765 const otherStems = this.stemSet(other);
766 const union = stems.union(otherStems).size;
767 const score =
768 union > 0 ? fraction(stems.intersection(otherStems).size, union) : fraction(0, 1);
769 scores.push({ score, number: other.number });
770 }
771 scores.sort(
772 (left, right) => compareFractions(right.score, left.score) || left.number - right.number,
773 );
774 this.emit(
775 'SIMILAR',
776 document.number,
777 ...scores
778 .slice(0, limit)
779 .map(({ score, number }) => `${number}:${score.numerator}/${score.denominator}`),
780 );
781 }
782
783 private rank(args: string[]): void {
784 const limit = numberArgument(args[0], 1, MAX_COUNT);
785 const query = new Set(args.slice(1).map((argument) => this.termArgument(argument)));
786 const ranked: { matched: number; total: number; number: number }[] = [];
787 for (const document of this.documents.values()) {
788 const counts = this.termFrequencies(document);
789 const frequencies = [...query].map((stem) => counts.get(stem) ?? 0);
790 const matched = frequencies.filter((count) => count > 0).length;
791 const total = frequencies.reduce((sum, count) => sum + count, 0);
792 if (matched > 0) {
793 ranked.push({ matched, total, number: document.number });
794 }
795 }
796 ranked.sort(
797 (left, right) =>
798 right.matched - left.matched || right.total - left.total || left.number - right.number,
799 );
800 this.emit(
801 `RANK count=${ranked.length}`,
802 ...ranked
803 .slice(0, limit)
804 .map(({ matched, total, number }) => `${number}:${matched}:${total}`),
805 );
806 }

808 private info(args: string[]): void {
809 const document = this.documentArgument(args[0]);
810 const counts = this.termFrequencies(document);
811 let top = '-';
812 if (counts.size > 0) {
813 const [stem, count] = [...counts].sort(byCount)[0];
814 top = `${stem}:${count}`;
815 }
816 const indexed = [...counts.values()].reduce((sum, count) => sum + count, 0);
817 this.emit(
818 'INFO',
819 document.number,
820 `lines=${document.lines.length}`,
821 `tokens=${document.tokens.length}`,
822 `indexed=${indexed}`,
823 `stems=${counts.size}`,
824 `top=${top}`,
825 );
826 }

846 this.emit(

875 private append(args: string[]): void {

877 const existing = this.documents.get(number);
878 const document = existing ?? this.addDocument(number);
879 const tokens = document.addLine(args[1]);
880 const indexed = tokens.filter((token) => this.isIndexed(token)).length;
881 this.emit(
882 'APPEND',
883 number,
884 `line=${document.lines.length}`,
885 `tokens=${tokens.length}`,
886 `indexed=${indexed}`,
887 `new=${yesNo(existing === undefined)}`,
888 );
889 }

894 this.emit(
895 'DELETE',

898 `tokens=${document.tokens.length}`,

901
902 execute(line: string): void {
903 const separator = line.indexOf(' ');
904 const name = separator < 0 ? line : line.slice(0, separator);
905 try {
906 const command = this.handlers.get(name);
907 if (command === undefined) {
908 throw new CommandError(`UNKNOWN COMMAND ${name}`);
909 }
910 const [minimum, maximum, handler] = command;
911 const args = separator < 0 ? [] : splitArguments(name, line.slice(separator + 1));
912 if (args.length < minimum || args.length > maximum) {
913 throw new CommandError(`BAD ARGUMENTS ${name}`);
914 }
915 handler.call(this, args);
916 } catch (error) {
917 if (!(error instanceof CommandError)) {
918 throw error;
919 }
920 this.emit('ERROR', error.message);
921 }
922 }
923}

925function main(): void {
926 const lines = readFileSync(0, 'utf8').split(/\r?\n/);
927 const engine = new SearchEngine(new Set(lines[0].trim().split(/\s+/).slice(1)));
928 let index = 1;
929 while (lines[index].startsWith('DOC ')) {
930 const document = engine.addDocument(Number(lines[index].slice(4)));
931 index += 1;
932 while (lines[index] !== 'ENDDOC') {
933 document.addLine(lines[index]);
934 index += 1;
935 }
936 index += 1;
937 }
938 for (const line of lines.slice(index)) {
939 if (line === 'END') {
940 break;
941 }
942 engine.execute(line);
943 }
944 process.stdout.write(engine.output.join('\n') + '\n');
945}
