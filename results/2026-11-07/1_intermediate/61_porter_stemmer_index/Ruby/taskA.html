21Rule = Data.define(:suffix, :replacement, :condition)

23def letter_pattern(word)
24 kinds = []
25 word.each_char.with_index do |letter, index|
26 vowel = VOWELS.include?(letter) || (letter == 'y' && index.positive? && kinds[-1] == 'C')
27 kinds << (vowel ? 'V' : 'C')
28 end
29 kinds.join
30end
31
32def measure(word) = letter_pattern(word).scan('VC').size
33
34def vowel?(word) = letter_pattern(word).include?('V')
35
36def ends_double?(word) = letter_pattern(word).end_with?('CC') && word[-1] == word[-2]
37
38def ends_cvc?(word) = letter_pattern(word).end_with?('CVC') && !'wxy'.include?(word[-1])
39
40ALWAYS = ->(_remainder) { true }
41MEASURE_ABOVE_ZERO = ->(remainder) { measure(remainder).positive? }
42MEASURE_ABOVE_ONE = ->(remainder) { measure(remainder) > 1 }
43ION_CONDITION = ->(remainder) { measure(remainder) > 1 && remainder.end_with?('s', 't') }
44FINAL_E_CONDITION = lambda do |remainder|
45 value = measure(remainder)
46 value > 1 || (value == 1 && !ends_cvc?(remainder))
47end
48
49def rules_with(condition, replacements)
50 replacements.map { |suffix, replacement| Rule.new(suffix, replacement, condition) }
51end
52
53RULE_TABLES = {
54 '1a' => rules_with(ALWAYS, { 'sses' => 'ss', 'ies' => 'i', 'ss' => 'ss', 's' => '' }),
55 '1b' => [
56 Rule.new('eed', 'ee', MEASURE_ABOVE_ZERO),
57 Rule.new('ed', '', method(:vowel?)),
58 Rule.new('ing', '', method(:vowel?)),
59 ],
60 '1c' => [Rule.new('y', 'i', method(:vowel?))],
61 '2' => rules_with(MEASURE_ABOVE_ZERO, {
62 'ational' => 'ate', 'tional' => 'tion', 'enci' => 'ence', 'anci' => 'ance',
63 'izer' => 'ize', 'abli' => 'able', 'alli' => 'al', 'entli' => 'ent', 'eli' => 'e',
64 'ousli' => 'ous', 'ization' => 'ize', 'ation' => 'ate', 'ator' => 'ate',
65 'alism' => 'al', 'iveness' => 'ive', 'fulness' => 'ful', 'ousness' => 'ous',
66 'aliti' => 'al', 'iviti' => 'ive', 'biliti' => 'ble'
67 }),
68 '3' => rules_with(MEASURE_ABOVE_ZERO, {
69 'icate' => 'ic', 'ative' => '', 'alize' => 'al', 'iciti' => 'ic',
70 'ical' => 'ic', 'ful' => '', 'ness' => ''
71 }),
72 '4' => STEP_4_SUFFIXES.map do |suffix|
73 Rule.new(suffix, '', suffix == 'ion' ? ION_CONDITION : MEASURE_ABOVE_ONE)
74 end,
75 '5a' => [Rule.new('e', '', FINAL_E_CONDITION)],
76}.freeze

78def apply_rules(word, rules)
79 matching = rules.select { |candidate| word.end_with?(candidate.suffix) }
80 rule = matching.max_by { |candidate| candidate.suffix.size }
81 return [word, nil] if rule.nil?
82
83 remainder = word.delete_suffix(rule.suffix)
84 return [word, nil] unless rule.condition.call(remainder)
85
86 [remainder + rule.replacement, rule.suffix]
87end

89def tidy_step_1b(word)
90 return "#{word}e" if word.end_with?('at', 'bl', 'iz')
91 return word[0...-1] if ends_double?(word) && !'lsz'.include?(word[-1])
92 return "#{word}e" if measure(word) == 1 && ends_cvc?(word)
93
94 word
95end
96
97def apply_step(name, word)
98 if name == '5b'
99 return word[0...-1] if measure(word) > 1 && ends_double?(word) && word.end_with?('l')
100
101 return word
102 end
103 result, suffix = apply_rules(word, RULE_TABLES[name])
104 return tidy_step_1b(result) if name == '1b' && %w[ed ing].include?(suffix)
105
106 result
107end

109def stemming_steps(word)
110 changes = []
111 if word.size > 2
112 STEP_NAMES.each do |name|
113 result = apply_step(name, word)
114 changes << "#{name}:#{result}" if result != word
115 word = result
116 end
117 end
118 [word, changes]
119end

150def by_count(entry)
151 stem, count = entry
152 [-count, stem]

155def document_count(occurrences) = occurrences.map { |document, _| document.number }.uniq.size

157def closest_distance(positions, others, same_stem)
158 if same_stem
159 gaps = positions.each_cons(2).map { |earlier, later| later - earlier }
160 else
161 merged = (positions.map { |position| [position, 0] } + others.map { |other| [other, 1] }).sort
162 gaps = merged.each_cons(2).filter_map do |earlier, later|
163 later[0] - earlier[0] if earlier[1] != later[1]
164 end
165 end
166 gaps.min
167end

169def classify(lexeme)
170 return [lexeme, lexeme] if OPERATORS.include?(lexeme) || PARENTHESES.include?(lexeme)
171 return ['term', lexeme.downcase] if WORD_PATTERN.match?(lexeme)
172 return ['prefix', lexeme[0...-1].downcase] if PREFIX_TERM_PATTERN.match?(lexeme)
173
174 raise CommandError, "BAD TERM #{lexeme}"
175end

183class Document

186 def initialize(number)
187 @number = number

190 end

192 def add_line(text)
193 @lines << text
194 added = []
195 text.scan(/[A-Za-z]+/) do |word|
196 position = @tokens.size + added.size + 1
197 added << Token.new(word, @lines.size, position, Regexp.last_match.begin(0))

208class QueryParser
209 def initialize(items, stem_sets)
210 @items = items
211 @position = 0
212 @stem_sets = stem_sets
213 end
214
215 def parse
216 result = expression
217 raise CommandError, 'SYNTAX' if @position < @items.size
218
219 result
220 end
221
222 private
223
224 def peek = @items[@position]&.first
225
226 def take
227 raise CommandError, 'SYNTAX' if @position >= @items.size
228
229 item = @items[@position]
230 @position += 1
231 item
232 end
233
234 def expression
235 canonical, documents = conjunction
236 while peek == 'OR'
237 @position += 1
238 right, right_documents = conjunction
239 canonical = "(#{canonical} OR #{right})"
240 documents |= right_documents
241 end
242 [canonical, documents]
243 end
244
245 def conjunction
246 canonical, documents = negation
247 while CONJUNCTION_CONTINUATIONS.include?(peek)
248 @position += 1 if peek == 'AND'
249 right, right_documents = negation
250 canonical = "(#{canonical} AND #{right})"
251 documents &= right_documents
252 end
253 [canonical, documents]
254 end
255
256 def negation
257 return element unless peek == 'NOT'
258
259 @position += 1
260 canonical, documents = negation
261 ["(NOT #{canonical})", @stem_sets.keys.to_set - documents]
262 end
263
264 def element
265 kind, spelling = take
266 case kind
267 when '('
268 result = expression
269 raise CommandError, 'SYNTAX' unless take.first == ')'
270
271 result
272 when 'term'
273 stem = porter_stem(spelling)
274 [stem, @stem_sets.filter_map { |number, stems| number if stems.include?(stem) }.to_set]
275 when 'prefix'
276 documents = @stem_sets.filter_map do |number, stems|
277 number if stems.any? { |stem| stem.start_with?(spelling) }
278 end
279 ["#{spelling}*", documents.to_set]
280 else
281 raise CommandError, 'SYNTAX'
282 end
283 end
284end

286class SearchEngine
287 HANDLERS = {
288 'STEM' => [1, 10, :stem_words],
289 'STEPS' => [1, 1, :show_steps],
290 'MEASURE' => [1, 1, :show_measure],
291 'INDEX' => [0, 1, :show_index],
292 'SEARCH' => [1, 1, :search],
293 'PHRASE' => [2, 10, :phrase],
294 'NEAR' => [3, 3, :near],
295 'TOP' => [1, 2, :top],
296 'KWIC' => [1, 1, :kwic],
297 'TF' => [2, 2, :term_frequency],
298 'DF' => [1, 1, :document_frequency],
299 'VARIANTS' => [1, 1, :variants],
300 'SIMILAR' => [2, 2, :similar],
301 'RANK' => [2, 11, :rank],
302 'INFO' => [1, 1, :info],
303 'LINE' => [2, 2, :show_line],
304 'STATS' => [0, 0, :stats],
305 'STOP' => [1, 1, :stop],
306 'UNSTOP' => [1, 1, :unstop],
307 'APPEND' => [2, 2, :append],
308 'DELETE' => [1, 1, :delete],
309 }.freeze
310
311 attr_reader :output
312
313 def initialize(stopwords)
314 @stopwords = stopwords
315 @documents = {}
316 @output = []
317 end
318
319 def add_document(number)
320 @documents[number] = Document.new(number)
321 end
322
323 def execute(line)
324 name, separator, rest = line.partition(' ')
325 raise CommandError, "UNKNOWN COMMAND #{name}" unless HANDLERS.key?(name)
326
327 minimum, maximum, handler = HANDLERS[name]
328 arguments = separator.empty? ? [] : split_arguments(name, rest)
329 raise CommandError, "BAD ARGUMENTS #{name}" unless arguments.size.between?(minimum, maximum)
330
331 send(handler, arguments)
332 rescue CommandError => e
333 emit('ERROR', e.message)
334 end
335
336 private
337
338 def emit(*parts)
339 @output << parts.join(' ')
340 end

342 def ordered_documents = @documents.values.sort_by(&:number)
343
344 def indexed?(token) = !@stopwords.include?(token.form)
345
346 def indexed_tokens(document) = document.tokens.select { |token| indexed?(token) }
347
348 def term_frequencies(document)
349 indexed_tokens(document).map { |token| porter_stem(token.form) }.tally
350 end
351
352 def stem_set(document) = term_frequencies(document).keys.to_set
353
354 def total_frequencies
355 totals = Hash.new(0)
356 @documents.each_value do |document|
357 term_frequencies(document).each { |stem, count| totals[stem] += count }
358 end
359 totals
360 end

362 def occurrence_map
363 occurrences = {}
364 ordered_documents.each do |document|
365 indexed_tokens(document).each do |token|
366 (occurrences[porter_stem(token.form)] ||= []) << [document, token]
367 end
368 end
369 occurrences
370 end
371
372 def stem_positions(document, stem)
373 indexed_tokens(document).select { |token| porter_stem(token.form) == stem }.map(&:position)
374 end
375
376 def form_count(form)
377 @documents.values.sum { |document| document.tokens.count { |token| token.form == form } }
378 end
379
380 def token_matches?(token, target)
381 target == '?' || (indexed?(token) && porter_stem(token.form) == target)
382 end
383
384 def describe_token(token)
385 indexed?(token) ? "#{token.form}/#{porter_stem(token.form)}" : "#{token.form}/-"
386 end

388 def term_argument(text)

405 def show_steps(arguments)
406 word = word_argument(arguments[0])
407 _, changes = stemming_steps(word)

409 emit('STEPS', word, 'NONE')

415 def show_measure(arguments)
416 word = word_argument(arguments[0])
417 emit(
418 'MEASURE', word, letter_pattern(word), "m=#{measure(word)}",
419 "vowel=#{yes_no(vowel?(word))}", "double=#{yes_no(ends_double?(word))}",
420 "cvc=#{yes_no(ends_cvc?(word))}"
421 )
422 end
423
424 def show_index(arguments)
425 prefix = arguments.empty? ? '' : word_argument(arguments[0])
426 occurrences = occurrence_map
427 stems = occurrences.keys.select { |stem| stem.start_with?(prefix) }.sort
428 emit('INDEX', "stems=#{stems.size}")
429 stems.each do |stem|
430 found = occurrences[stem]
431 references = found.map { |document, token| document.reference(token) }
432 emit(stem, "df=#{document_count(found)}", *references)
433 end
434 end

436 def search(arguments)
437 items = arguments[0].scan(LEXEME_PATTERN).map { |lexeme| classify(lexeme) }
438 stem_sets = @documents.transform_values { |document| stem_set(document) }
439 canonical, matches = QueryParser.new(items, stem_sets).parse
440 items.each do |kind, spelling|
441 raise CommandError, "STOPWORD #{spelling}" if kind == 'term' && @stopwords.include?(spelling)
442 end
443 emit('SEARCH', canonical, "count=#{matches.size}", *matches.sort)
444 end
445
446 def phrase(arguments)
447 words = arguments.map { |argument| word_argument(argument) }
448 raise CommandError, 'STOPWORDS ONLY' if words.all? { |word| @stopwords.include?(word) }
449
450 targets = words.map { |word| @stopwords.include?(word) ? '?' : porter_stem(word) }
451 matches = []
452 ordered_documents.each do |document|
453 tokens = document.tokens
454 (0..tokens.size - targets.size).each do |start|
455 window = tokens[start, targets.size]
456 next unless window.zip(targets).all? { |token, target| token_matches?(token, target) }
457
458 matches << document.reference(tokens[start])
459 end
460 end
461 phrase_text = targets.join(' ')
462 emit("PHRASE \"#{phrase_text}\"", "matches=#{matches.size}", *matches)
463 end

465 def near(arguments)
466 first = term_argument(arguments[0])
467 second = term_argument(arguments[1])
468 limit = number_argument(arguments[2], 1, MAX_COUNT)
469 found = ordered_documents.filter_map do |document|
470 distance = closest_distance(
471 stem_positions(document, first), stem_positions(document, second), first == second
472 )
473 "#{document.number}:#{distance}" if distance && distance <= limit
474 end
475 emit('NEAR', first, second, limit, "count=#{found.size}", *found)
476 end
477
478 def top(arguments)
479 limit = number_argument(arguments[0], 1, MAX_COUNT)
480 if arguments.size == 2
481 document = document_argument(arguments[1])
482 counts = term_frequencies(document)
483 header = "TOP #{limit} doc=#{document.number}"
484 else
485 counts = total_frequencies
486 header = "TOP #{limit}"
487 end
488 ranked = counts.sort_by { |entry| by_count(entry) }.first(limit)
489 emit(header, *ranked.map { |stem, count| "#{stem}:#{count}" })
490 end
491
492 def kwic(arguments)
493 stem = term_argument(arguments[0])
494 found = occurrence_map.fetch(stem, [])
495 emit('KWIC', stem, "count=#{found.size}")
496 found.each do |document, token|
497 text = document.lines[token.line - 1]
498 before = text[[0, token.start - KWIC_WIDTH].max...token.start].rjust(KWIC_WIDTH)
499 after = text[token.finish, KWIC_WIDTH].ljust(KWIC_WIDTH)
500 emit("|#{before}[#{token.text}]#{after}|", document.reference(token))
501 end
502 end

504 def term_frequency(arguments)
505 document = document_argument(arguments[0])
506 stem = term_argument(arguments[1])
507 emit('TF', document.number, stem, term_frequencies(document).fetch(stem, 0))
508 end

510 def document_frequency(arguments)
511 stem = term_argument(arguments[0])
512 emit('DF', stem, document_count(occurrence_map.fetch(stem, [])))
513 end

515 def variants(arguments)
516 stem = term_argument(arguments[0])
517 forms = occurrence_map.fetch(stem, []).map { |_, token| token.form }.tally
518 shown = forms.sort.map { |form, count| "#{form}:#{count}" }
519 emit('VARIANTS', stem, "count=#{forms.size}", *shown)
520 end
521
522 def similar(arguments)
523 document = document_argument(arguments[0])
524 limit = number_argument(arguments[1], 1, MAX_COUNT)
525 stems = stem_set(document)
526 scores = @documents.values.reject { |other| other.equal?(document) }.map do |other|
527 other_stems = stem_set(other)
528 union = (stems | other_stems).size
529 score = union.zero? ? Rational(0) : Rational((stems & other_stems).size, union)
530 [score, other.number]
531 end
532 scores.sort_by! { |score, number| [-score, number] }
533 shown = scores.first(limit).map do |score, number|
534 "#{number}:#{score.numerator}/#{score.denominator}"
535 end
536 emit('SIMILAR', document.number, *shown)
537 end
538
539 def rank(arguments)
540 limit = number_argument(arguments[0], 1, MAX_COUNT)
541 query = arguments[1..].map { |argument| term_argument(argument) }.to_set
542 ranked = @documents.values.filter_map do |document|
543 counts = term_frequencies(document)
544 matched = query.count { |stem| counts.fetch(stem, 0).positive? }
545 total = query.sum { |stem| counts.fetch(stem, 0) }
546 [matched, total, document.number] if matched.positive?
547 end
548 ranked.sort_by! { |matched, total, number| [-matched, -total, number] }
549 shown = ranked.first(limit).map { |matched, total, number| "#{number}:#{matched}:#{total}" }
550 emit("RANK count=#{ranked.size}", *shown)
551 end

553 def info(arguments)
554 document = document_argument(arguments[0])
555 counts = term_frequencies(document)
556 top = counts.empty? ? '-' : counts.min_by { |entry| by_count(entry) }.join(':')
557 emit(
558 'INFO', document.number, "lines=#{document.lines.size}",
559 "tokens=#{document.tokens.size}", "indexed=#{counts.values.sum}",
560 "stems=#{counts.size}", "top=#{top}"
561 )
562 end

565 document = document_argument(arguments[0])
566 line_number = number_argument(arguments[1], 1, MAX_LINE_NUMBER)

580 emit(
581 'STATS', "docs=#{documents.size}",

590 word = word_argument(arguments[0])
591 raise CommandError, "ALREADY STOPWORD #{word}" if @stopwords.include?(word)

594 emit('STOP', word, "unindexed=#{form_count(word)}")

598 word = word_argument(arguments[0])
599 raise CommandError, "NOT STOPWORD #{word}" unless @stopwords.include?(word)

605 def append(arguments)
606 number = number_argument(arguments[0], 1, MAX_DOCUMENT_NUMBER)
607 created = !@documents.key?(number)
608 document = created ? add_document(number) : @documents[number]
609 tokens = document.add_line(arguments[1])
610 indexed = tokens.count { |token| indexed?(token) }
611 emit(
612 'APPEND', number, "line=#{document.lines.size}", "tokens=#{tokens.size}",

618 document = document_argument(arguments[0])
619 @documents.delete(document.number)
620 emit(
621 'DELETE', document.number, "lines=#{document.lines.size}", "tokens=#{document.tokens.size}"

626def main
627 lines = $stdin.read.lines(chomp: true)
628 engine = SearchEngine.new(lines[0].split[1..].to_set)
629 index = 1
630 while lines[index].start_with?('DOC ')
631 document = engine.add_document(lines[index][4..].to_i)
632 index += 1
633 until lines[index] == 'ENDDOC'
634 document.add_line(lines[index])
635 index += 1
636 end
637 index += 1
638 end
639 lines[index..].each do |line|
640 break if line == 'END'
641
642 engine.execute(line)
643 end
644 $stdout.write(engine.output.join("\n"), "\n")
645end
