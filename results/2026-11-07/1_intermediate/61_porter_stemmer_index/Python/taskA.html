33def letter_pattern(word):
34 kinds = []
35 for index, letter in enumerate(word):
36 if letter in VOWELS or (letter == "y" and index > 0 and kinds[-1] == "C"):
37 kinds.append("V")
38 else:
39 kinds.append("C")
40 return "".join(kinds)

80def rules_with(condition, replacements):
81 return [(suffix, replacement, condition) for suffix, replacement in replacements.items()]

84RULE_TABLES = {
85 "1a": rules_with(always, {"sses": "ss", "ies": "i", "ss": "ss", "s": ""}),
86 "1b": [("eed", "ee", measure_above_zero), ("ed", "", has_vowel), ("ing", "", has_vowel)],
87 "1c": [("y", "i", has_vowel)],
88 "2": rules_with(measure_above_zero, {
89 "ational": "ate", "tional": "tion", "enci": "ence", "anci": "ance",
90 "izer": "ize", "abli": "able", "alli": "al", "entli": "ent", "eli": "e",

92 "alism": "al", "iveness": "ive", "fulness": "ful", "ousness": "ous",
93 "aliti": "al", "iviti": "ive", "biliti": "ble",

97 "ical": "ic", "ful": "", "ness": "",
98 }),
99 "4": [

107def apply_rules(word, rules):
108 matching = [rule for rule in rules if word.endswith(rule[0])]
109 if not matching:
110 return word, None
111 suffix, replacement, condition = max(matching, key=lambda rule: len(rule[0]))
112 remainder = word[:-len(suffix)]
113 if not condition(remainder):
114 return word, None
115 return remainder + replacement, suffix

118def tidy_step_1b(word):
119 if word.endswith(("at", "bl", "iz")):
120 return word + "e"
121 if ends_double(word) and word[-1] not in "lsz":
122 return word[:-1]
123 if measure(word) == 1 and ends_cvc(word):
124 return word + "e"
125 return word

128def apply_step(name, word):
129 if name == "5b":
130 if measure(word) > 1 and ends_double(word) and word.endswith("l"):
131 return word[:-1]
132 return word
133 result, suffix = apply_rules(word, RULE_TABLES[name])
134 if name == "1b" and suffix in ("ed", "ing"):
135 return tidy_step_1b(result)
136 return result

139def stemming_steps(word):
140 changes = []
141 if len(word) > 2:
142 for name in STEP_NAMES:
143 result = apply_step(name, word)
144 if result != word:
145 changes.append((name, result))
146 word = result
147 return word, changes

191def closest_distance(positions, others, same_stem):
192 if same_stem:
193 gaps = [later - earlier for earlier, later in zip(positions, positions[1:])]
194 else:
195 merged = sorted(
196 [(position, 0) for position in positions] + [(other, 1) for other in others]
197 )
198 gaps = [
199 later[0] - earlier[0]
200 for earlier, later in zip(merged, merged[1:])
201 if earlier[1] != later[1]
202 ]
203 return min(gaps, default=None)

206def classify(lexeme):
207 if lexeme in OPERATORS or lexeme in PARENTHESES:
208 return lexeme, lexeme
209 if WORD_PATTERN.fullmatch(lexeme):
210 return "term", lexeme.lower()
211 if PREFIX_TERM_PATTERN.fullmatch(lexeme):
212 return "prefix", lexeme[:-1].lower()
213 raise CommandError(f"BAD TERM {lexeme}")

254class QueryParser:
255 def __init__(self, items, stem_sets):
256 self.items = items
257 self.position = 0
258 self.stem_sets = stem_sets
259
260 def peek(self):
261 if self.position < len(self.items):
262 return self.items[self.position][0]
263 return None
264
265 def take(self):
266 if self.position >= len(self.items):
267 raise CommandError("SYNTAX")
268 item = self.items[self.position]
269 self.position += 1
270 return item
271
272 def parse(self):
273 result = self.expression()
274 if self.position < len(self.items):
275 raise CommandError("SYNTAX")
276 return result
277
278 def expression(self):
279 canonical, documents = self.conjunction()
280 while self.peek() == "OR":
281 self.position += 1
282 right, right_documents = self.conjunction()
283 canonical = f"({canonical} OR {right})"
284 documents = documents | right_documents
285 return canonical, documents
286
287 def conjunction(self):
288 canonical, documents = self.negation()
289 while self.peek() in CONJUNCTION_CONTINUATIONS:
290 if self.peek() == "AND":
291 self.position += 1
292 right, right_documents = self.negation()
293 canonical = f"({canonical} AND {right})"
294 documents = documents & right_documents
295 return canonical, documents
296
297 def negation(self):
298 if self.peek() == "NOT":
299 self.position += 1
300 canonical, documents = self.negation()
301 return f"(NOT {canonical})", set(self.stem_sets) - documents
302 return self.element()
303
304 def element(self):
305 kind, spelling = self.take()
306 if kind == "(":
307 result = self.expression()
308 if self.take()[0] != ")":
309 raise CommandError("SYNTAX")
310 return result
311 if kind == "term":
312 stem = porter_stem(spelling)
313 documents = {number for number, stems in self.stem_sets.items() if stem in stems}
314 return stem, documents
315 if kind == "prefix":
316 documents = {
317 number for number, stems in self.stem_sets.items()
318 if any(stem.startswith(spelling) for stem in stems)
319 }
320 return f"{spelling}*", documents
321 raise CommandError("SYNTAX")

324class SearchEngine:
325 def __init__(self, stopwords):
326 self.stopwords = stopwords
327 self.documents = {}
328 self.output = []
329 self.handlers = {
330 "STEM": (1, 10, self.stem_words),
331 "STEPS": (1, 1, self.show_steps),
332 "MEASURE": (1, 1, self.show_measure),
333 "INDEX": (0, 1, self.show_index),
334 "SEARCH": (1, 1, self.search),
335 "PHRASE": (2, 10, self.phrase),
336 "NEAR": (3, 3, self.near),
337 "TOP": (1, 2, self.top),
338 "KWIC": (1, 1, self.kwic),
339 "TF": (2, 2, self.term_frequency),
340 "DF": (1, 1, self.document_frequency),
341 "VARIANTS": (1, 1, self.variants),
342 "SIMILAR": (2, 2, self.similar),
343 "RANK": (2, 11, self.rank),
344 "INFO": (1, 1, self.info),
345 "LINE": (2, 2, self.show_line),
346 "STATS": (0, 0, self.stats),
347 "STOP": (1, 1, self.stop),
348 "UNSTOP": (1, 1, self.unstop),
349 "APPEND": (2, 2, self.append),
350 "DELETE": (1, 1, self.delete),
351 }

353 def emit(self, *parts):
354 self.output.append(" ".join(str(part) for part in parts))

356 def add_document(self, number):
357 document = Document(number)
358 self.documents[number] = document
359 return document

361 def ordered_documents(self):
362 return [self.documents[number] for number in sorted(self.documents)]
363
364 def is_indexed(self, token):
365 return token.form not in self.stopwords
366
367 def indexed_tokens(self, document):
368 return [token for token in document.tokens if self.is_indexed(token)]
369
370 def term_frequencies(self, document):
371 return Counter(porter_stem(token.form) for token in self.indexed_tokens(document))
372
373 def stem_set(self, document):
374 return set(self.term_frequencies(document))
375
376 def total_frequencies(self):
377 totals = Counter()
378 for document in self.documents.values():
379 totals.update(self.term_frequencies(document))
380 return totals

382 def occurrence_map(self):
383 occurrences = {}
384 for document in self.ordered_documents():
385 for token in self.indexed_tokens(document):
386 occurrences.setdefault(porter_stem(token.form), []).append((document, token))
387 return occurrences

389 def stem_positions(self, document, stem):
390 return [
391 token.position for token in self.indexed_tokens(document)
392 if porter_stem(token.form) == stem
393 ]
394
395 def form_count(self, form):
396 return sum(
397 token.form == form
398 for document in self.documents.values()
399 for token in document.tokens

402 def token_matches(self, token, target):
403 return target == "?" or (self.is_indexed(token) and porter_stem(token.form) == target)
404
405 def describe_token(self, token):
406 if self.is_indexed(token):
407 return f"{token.form}/{porter_stem(token.form)}"
408 return f"{token.form}/-"

410 def term_argument(self, text):
411 word = word_argument(text)
412 if word in self.stopwords:

416 def document_argument(self, text):

422 def stem_words(self, arguments):
423 words = [word_argument(argument) for argument in arguments]
424 self.emit("STEM", *(f"{word}:{porter_stem(word)}" for word in words))

426 def show_steps(self, arguments):
427 word = word_argument(arguments[0])
428 _, changes = stemming_steps(word)
429 if not changes:
430 self.emit("STEPS", word, "NONE")

434 def show_measure(self, arguments):
435 word = word_argument(arguments[0])
436 self.emit(
437 "MEASURE", word, letter_pattern(word), f"m={measure(word)}",
438 f"vowel={yes_no(has_vowel(word))}", f"double={yes_no(ends_double(word))}",
439 f"cvc={yes_no(ends_cvc(word))}",
440 )

442 def show_index(self, arguments):
443 prefix = word_argument(arguments[0]) if arguments else ""
444 occurrences = self.occurrence_map()
445 stems = sorted(stem for stem in occurrences if stem.startswith(prefix))
446 self.emit("INDEX", f"stems={len(stems)}")
447 for stem in stems:
448 found = occurrences[stem]
449 self.emit(
450 stem, f"df={document_count(found)}",
451 *(document.reference(token) for document, token in found),

454 def search(self, arguments):
455 items = [classify(lexeme) for lexeme in LEXEME_PATTERN.findall(arguments[0])]
456 stem_sets = {
457 document.number: self.stem_set(document) for document in self.documents.values()
458 }
459 canonical, matches = QueryParser(items, stem_sets).parse()
460 for kind, spelling in items:
461 if kind == "term" and spelling in self.stopwords:
462 raise CommandError(f"STOPWORD {spelling}")
463 self.emit("SEARCH", canonical, f"count={len(matches)}", *sorted(matches))
464
465 def phrase(self, arguments):
466 words = [word_argument(argument) for argument in arguments]
467 if all(word in self.stopwords for word in words):
468 raise CommandError("STOPWORDS ONLY")
469 targets = ["?" if word in self.stopwords else porter_stem(word) for word in words]
470 matches = []
471 for document in self.ordered_documents():
472 tokens = document.tokens
473 for start in range(len(tokens) - len(targets) + 1):
474 if all(
475 self.token_matches(tokens[start + offset], target)
476 for offset, target in enumerate(targets)
477 ):
478 matches.append(document.reference(tokens[start]))
479 phrase_text = " ".join(targets)
480 self.emit(f'PHRASE "{phrase_text}"', f"matches={len(matches)}", *matches)
481
482 def near(self, arguments):
483 first = self.term_argument(arguments[0])
484 second = self.term_argument(arguments[1])
485 limit = number_argument(arguments[2], 1, MAX_COUNT)
486 found = []
487 for document in self.ordered_documents():
488 distance = closest_distance(
489 self.stem_positions(document, first),
490 self.stem_positions(document, second),
491 first == second,
492 )
493 if distance is not None and distance <= limit:
494 found.append(f"{document.number}:{distance}")
495 self.emit("NEAR", first, second, limit, f"count={len(found)}", *found)
496
497 def top(self, arguments):
498 limit = number_argument(arguments[0], 1, MAX_COUNT)
499 if len(arguments) == 2:
500 document = self.document_argument(arguments[1])
501 counts = self.term_frequencies(document)
502 header = f"TOP {limit} doc={document.number}"
503 else:
504 counts = self.total_frequencies()
505 header = f"TOP {limit}"
506 ranked = sorted(counts.items(), key=by_count)[:limit]
507 self.emit(header, *(f"{stem}:{count}" for stem, count in ranked))
508
509 def kwic(self, arguments):
510 stem = self.term_argument(arguments[0])
511 found = self.occurrence_map().get(stem, [])
512 self.emit("KWIC", stem, f"count={len(found)}")
513 for document, token in found:
514 text = document.lines[token.line - 1]
515 before = text[max(0, token.start - KWIC_WIDTH):token.start].rjust(KWIC_WIDTH)
516 after = text[token.end:token.end + KWIC_WIDTH].ljust(KWIC_WIDTH)
517 self.emit(f"|{before}[{token.text}]{after}|", document.reference(token))
518
519 def term_frequency(self, arguments):
520 document = self.document_argument(arguments[0])
521 stem = self.term_argument(arguments[1])
522 self.emit("TF", document.number, stem, self.term_frequencies(document)[stem])
523
524 def document_frequency(self, arguments):
525 stem = self.term_argument(arguments[0])
526 found = self.occurrence_map().get(stem, [])
527 self.emit("DF", stem, document_count(found))

529 def variants(self, arguments):
530 stem = self.term_argument(arguments[0])
531 found = self.occurrence_map().get(stem, [])
532 forms = Counter(token.form for _, token in found)
533 self.emit(
534 "VARIANTS", stem, f"count={len(forms)}",
535 *(f"{form}:{count}" for form, count in sorted(forms.items())),

538 def similar(self, arguments):
539 document = self.document_argument(arguments[0])
540 limit = number_argument(arguments[1], 1, MAX_COUNT)
541 stems = self.stem_set(document)
542 scores = []
543 for other in self.documents.values():
544 if other is document:
545 continue
546 other_stems = self.stem_set(other)
547 union = len(stems | other_stems)
548 score = Fraction(len(stems & other_stems), union) if union else Fraction(0)
549 scores.append((score, other.number))
550 scores.sort(key=lambda entry: (-entry[0], entry[1]))
551 self.emit(
552 "SIMILAR", document.number,
553 *(
554 f"{number}:{score.numerator}/{score.denominator}"
555 for score, number in scores[:limit]
556 ),
557 )
558
559 def rank(self, arguments):
560 limit = number_argument(arguments[0], 1, MAX_COUNT)
561 query = {self.term_argument(argument) for argument in arguments[1:]}
562 ranked = []
563 for document in self.documents.values():
564 counts = self.term_frequencies(document)
565 matched = sum(counts[stem] > 0 for stem in query)
566 total = sum(counts[stem] for stem in query)
567 if matched > 0:
568 ranked.append((matched, total, document.number))
569 ranked.sort(key=lambda entry: (-entry[0], -entry[1], entry[2]))
570 self.emit(
571 f"RANK count={len(ranked)}",
572 *(f"{number}:{matched}:{total}" for matched, total, number in ranked[:limit]),
573 )

575 def info(self, arguments):
576 document = self.document_argument(arguments[0])
577 counts = self.term_frequencies(document)
578 if counts:
579 stem, count = min(counts.items(), key=by_count)
580 top = f"{stem}:{count}"
581 else:
582 top = "-"
583 self.emit(
584 "INFO", document.number, f"lines={len(document.lines)}",
585 f"tokens={len(document.tokens)}", f"indexed={counts.total()}",
586 f"stems={len(counts)}", f"top={top}",
587 )

589 def show_line(self, arguments):
590 document = self.document_argument(arguments[0])
591 line_number = number_argument(arguments[1], 1, MAX_LINE_NUMBER)

600 def stats(self, arguments):
601 documents = self.documents.values()
602 self.emit(
603 "STATS", f"docs={len(documents)}",
604 f"lines={sum(len(document.lines) for document in documents)}",
605 f"tokens={sum(len(document.tokens) for document in documents)}",

610 def stop(self, arguments):
611 word = word_argument(arguments[0])
612 if word in self.stopwords:
613 raise CommandError(f"ALREADY STOPWORD {word}")
614 self.stopwords.add(word)
615 self.emit("STOP", word, f"unindexed={self.form_count(word)}")

617 def unstop(self, arguments):
618 word = word_argument(arguments[0])
619 if word not in self.stopwords:
620 raise CommandError(f"NOT STOPWORD {word}")
621 self.stopwords.remove(word)

624 def append(self, arguments):
625 number = number_argument(arguments[0], 1, MAX_DOCUMENT_NUMBER)
626 created = number not in self.documents
627 document = self.add_document(number) if created else self.documents[number]
628 tokens = document.add_line(arguments[1])
629 indexed = sum(self.is_indexed(token) for token in tokens)
630 self.emit(
631 "APPEND", number, f"line={len(document.lines)}", f"tokens={len(tokens)}",
632 f"indexed={indexed}", f"new={yes_no(created)}",
633 )
634
635 def delete(self, arguments):
636 document = self.document_argument(arguments[0])
637 del self.documents[document.number]
638 self.emit(
639 "DELETE", document.number, f"lines={len(document.lines)}",
640 f"tokens={len(document.tokens)}",
641 )

643 def execute(self, line):
644 name, separator, rest = line.partition(" ")
645 try:
646 if name not in self.handlers:
647 raise CommandError(f"UNKNOWN COMMAND {name}")
648 minimum, maximum, handler = self.handlers[name]
649 arguments = split_arguments(name, rest) if separator else []
650 if not minimum <= len(arguments) <= maximum:
651 raise CommandError(f"BAD ARGUMENTS {name}")
652 handler(arguments)
653 except CommandError as error:
654 self.emit("ERROR", error)

657def main():
658 lines = sys.stdin.read().splitlines()
659 engine = SearchEngine(set(lines[0].split()[1:]))
660 index = 1
661 while lines[index].startswith("DOC "):
662 document = engine.add_document(int(lines[index][4:]))
663 index += 1
664 while lines[index] != "ENDDOC":
665 document.add_line(lines[index])
666 index += 1
667 index += 1
668 for line in lines[index:]:
669 if line == "END":
670 break
671 engine.execute(line)
672 sys.stdout.write("\n".join(engine.output) + "\n")
