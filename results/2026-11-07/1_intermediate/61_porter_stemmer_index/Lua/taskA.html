85local function letter_pattern(word)
86 local kinds = {}
87 for index = 1, #word do
88 local letter = word:sub(index, index)
89 local consonant_before = index > 1 and kinds[index - 1] == "C"
90 if VOWELS:find(letter, 1, true) or (letter == "y" and consonant_before) then
91 kinds[index] = "V"
92 else
93 kinds[index] = "C"
94 end
95 end
96 return table.concat(kinds)
97end

155local RULE_TABLES = {
156 ["1a"] = rules_with(always, { sses = "ss", ies = "i", ss = "ss", s = "" }),
157 ["1b"] = {

176local function apply_rules(word, rules)
177 local selected
178 for _, candidate in ipairs(rules) do
179 local longer = selected == nil or #candidate.suffix > #selected.suffix
180 if longer and ends_with(word, candidate.suffix) then
181 selected = candidate
182 end
183 end
184 if selected == nil then
185 return word, nil
186 end
187 local remainder = word:sub(1, #word - #selected.suffix)
188 if not selected.condition(remainder) then
189 return word, nil
190 end
191 return remainder .. selected.replacement, selected.suffix
192end

194local function tidy_step_1b(word)
195 if ends_with(word, "at") or ends_with(word, "bl") or ends_with(word, "iz") then
196 return word .. "e"
197 end
198 if ends_double(word) and not ("lsz"):find(word:sub(-1), 1, true) then
199 return word:sub(1, -2)
200 end
201 if measure(word) == 1 and ends_cvc(word) then
202 return word .. "e"
203 end
204 return word
205end

207local function apply_step(name, word)
208 if name == "5b" then
209 if measure(word) > 1 and ends_double(word) and ends_with(word, "l") then
210 return word:sub(1, -2)
211 end
212 return word
213 end
214 local result, suffix = apply_rules(word, RULE_TABLES[name])
215 if name == "1b" and (suffix == "ed" or suffix == "ing") then
216 return tidy_step_1b(result)
217 end
218 return result
219end

221local function stemming_steps(word)
222 local changes = {}
223 if #word > 2 then
224 for _, name in ipairs(STEP_NAMES) do
225 local result = apply_step(name, word)
226 if result ~= word then
227 changes[#changes + 1] = name .. ":" .. result
228 end
229 word = result
230 end
231 end
232 return word, changes
233end

308local function closest_distance(positions, others, same_stem)
309 local closest
310 local function consider(gap)
311 if closest == nil or gap < closest then
312 closest = gap
313 end
314 end
315 if same_stem then
316 for index = 2, #positions do
317 consider(positions[index] - positions[index - 1])
318 end
319 else
320 local merged = {}
321 for _, position in ipairs(positions) do
322 merged[#merged + 1] = { position = position, side = 0 }
323 end
324 for _, other in ipairs(others) do
325 merged[#merged + 1] = { position = other, side = 1 }
326 end
327 table.sort(merged, function(first, second)
328 return first.position < second.position
329 end)
330 for index = 2, #merged do
331 if merged[index].side ~= merged[index - 1].side then
332 consider(merged[index].position - merged[index - 1].position)
333 end
334 end
335 end
336 return closest
337end

339local function reduced_fraction(numerator, denominator)
340 local a, b = numerator, denominator
341 while b ~= 0 do
342 a, b = b, a % b
343 end
344 return numerator // a, denominator // a
345end

347local function classify(lexeme)
348 if OPERATORS[lexeme] or PARENTHESES[lexeme] then
349 return { kind = lexeme, spelling = lexeme }
350 end
351 if is_word(lexeme) then
352 return { kind = "term", spelling = lexeme:lower() }
353 end
354 local letters = lexeme:sub(1, -2)
355 if ends_with(lexeme, "*") and is_word(letters) then
356 return { kind = "prefix", spelling = letters:lower() }
357 end
358 fail("BAD TERM " .. lexeme)
359end

402function QueryParser.new(items, stem_sets)
403 return setmetatable({ items = items, position = 1, stem_sets = stem_sets }, QueryParser)
404end
405
406function QueryParser:peek()
407 local item = self.items[self.position]
408 return item and item.kind
409end
410
411function QueryParser:take()
412 local item = self.items[self.position]
413 if item == nil then
414 fail("SYNTAX")
415 end
416 self.position = self.position + 1
417 return item
418end
419
420function QueryParser:parse()
421 local canonical, documents = self:expression()
422 if self.position <= #self.items then
423 fail("SYNTAX")
424 end
425 return canonical, documents
426end
427
428function QueryParser:expression()
429 local canonical, documents = self:conjunction()
430 while self:peek() == "OR" do
431 self.position = self.position + 1
432 local right, right_documents = self:conjunction()
433 canonical = "(" .. canonical .. " OR " .. right .. ")"
434 documents = union(documents, right_documents)
435 end
436 return canonical, documents
437end
438
439function QueryParser:conjunction()
440 local canonical, documents = self:negation()
441 while CONJUNCTION_CONTINUATIONS[self:peek()] do
442 if self:peek() == "AND" then
443 self.position = self.position + 1
444 end
445 local right, right_documents = self:negation()
446 canonical = "(" .. canonical .. " AND " .. right .. ")"
447 documents = intersection(documents, right_documents)
448 end
449 return canonical, documents
450end
451
452function QueryParser:negation()
453 if self:peek() == "NOT" then
454 self.position = self.position + 1
455 local canonical, documents = self:negation()
456 return "(NOT " .. canonical .. ")", difference(self.stem_sets, documents)
457 end
458 return self:element()
459end
460
461function QueryParser:matching(predicate)
462 local documents = {}
463 for number, stems in pairs(self.stem_sets) do
464 if predicate(stems) then
465 documents[number] = true
466 end
467 end
468 return documents
469end
470
471function QueryParser:element()
472 local item = self:take()
473 if item.kind == "(" then
474 local canonical, documents = self:expression()
475 if self:take().kind ~= ")" then
476 fail("SYNTAX")
477 end
478 return canonical, documents
479 end
480 if item.kind == "term" then
481 local stem = porter_stem(item.spelling)
482 return stem, self:matching(function(stems)
483 return stems[stem]
484 end)
485 end
486 if item.kind == "prefix" then
487 return item.spelling .. "*", self:matching(function(stems)
488 for stem in pairs(stems) do
489 if starts_with(stem, item.spelling) then
490 return true
491 end
492 end
493 return false
494 end)
495 end
496 fail("SYNTAX")
497end

679function SearchEngine:search(arguments)
680 local items = {}
681 for lexeme in arguments[1]:gsub("[()]", " %0 "):gmatch("[^ ]+") do

683 end

698function SearchEngine:phrase(arguments)
699 local words = {}
700 for index, argument in ipairs(arguments) do
701 words[index] = word_argument(argument)
702 end
703 local targets = {}
704 local only_stopwords = true
705 for index, word in ipairs(words) do
706 if self.stopwords[word] then
707 targets[index] = "?"
708 else
709 targets[index] = porter_stem(word)
710 only_stopwords = false
711 end
712 end
713 if only_stopwords then
714 fail("STOPWORDS ONLY")
715 end
716 local matches = {}
717 for _, document in ipairs(self:ordered_documents()) do
718 local tokens = document.tokens
719 for start = 1, #tokens - #targets + 1 do
720 local matched = true
721 for offset, target in ipairs(targets) do
722 if not self:token_matches(tokens[start + offset - 1], target) then
723 matched = false
724 break
725 end
726 end
727 if matched then
728 matches[#matches + 1] = document:reference(tokens[start])
729 end
730 end
731 end
732 local phrase_text = table.concat(targets, " ")
733 self:emit('PHRASE "' .. phrase_text .. '"', "matches=" .. #matches, table.unpack(matches))
734end

736function SearchEngine:near(arguments)
737 local first = self:term_argument(arguments[1])
738 local second = self:term_argument(arguments[2])
739 local limit = number_argument(arguments[3], 1, MAX_COUNT)
740 local found = {}
741 for _, document in ipairs(self:ordered_documents()) do
742 local distance = closest_distance(
743 self:stem_positions(document, first),
744 self:stem_positions(document, second),
745 first == second
746 )
747 if distance ~= nil and distance <= limit then
748 found[#found + 1] = document.number .. ":" .. distance
749 end
750 end
751 self:emit("NEAR", first, second, limit, "count=" .. #found, table.unpack(found))
752end

754function SearchEngine:top(arguments)

758 local document = self:document_argument(arguments[2])
759 counts = self:term_frequencies(document)
760 header = "TOP " .. limit .. " doc=" .. document.number

762 counts = self:total_frequencies()

773function SearchEngine:kwic(arguments)
774 local stem = self:term_argument(arguments[1])
775 local found = self:occurrence_map()[stem] or {}
776 self:emit("KWIC", stem, "count=" .. #found)
777 for _, occurrence in ipairs(found) do
778 local document, token = occurrence.document, occurrence.token
779 local text = document.lines[token.line]
780 local finish = token.start + #token.text
781 local before = text:sub(math.max(1, token.start - KWIC_WIDTH), token.start - 1)
782 local after = text:sub(finish, finish + KWIC_WIDTH - 1)
783 before = string.rep(" ", KWIC_WIDTH - #before) .. before
784 after = after .. string.rep(" ", KWIC_WIDTH - #after)
785 self:emit("|" .. before .. "[" .. token.text .. "]" .. after .. "|", document:reference(token))
786 end
787end

807 forms[form] = (forms[form] or 0) + 1

809 local items = {}
810 for index, form in ipairs(sorted_keys(forms)) do
811 items[index] = form .. ":" .. forms[form]
812 end
813 self:emit("VARIANTS", stem, "count=" .. #items, table.unpack(items))
814end

816function SearchEngine:similar(arguments)
817 local document = self:document_argument(arguments[1])
818 local limit = number_argument(arguments[2], 1, MAX_COUNT)
819 local stems = self:stem_set(document)
820 local scores = {}
821 for _, other in pairs(self.documents) do
822 if other ~= document then
823 local other_stems = self:stem_set(other)
824 local union_size = count_keys(union(stems, other_stems))
825 local numerator, denominator = 0, 1
826 if union_size > 0 then
827 numerator, denominator = reduced_fraction(
828 count_keys(intersection(stems, other_stems)), union_size
829 )
830 end
831 scores[#scores + 1] = {
832 numerator = numerator, denominator = denominator, number = other.number,
833 }
834 end
835 end
836 table.sort(scores, function(first, second)
837 local left = first.numerator * second.denominator
838 local right = second.numerator * first.denominator
839 if left ~= right then
840 return left > right
841 end
842 return first.number < second.number
843 end)
844 local items = {}
845 for index = 1, math.min(limit, #scores) do
846 local score = scores[index]
847 items[index] = score.number .. ":" .. score.numerator .. "/" .. score.denominator
848 end
849 self:emit("SIMILAR", document.number, table.unpack(items))
850end

852function SearchEngine:rank(arguments)
853 local limit = number_argument(arguments[1], 1, MAX_COUNT)
854 local query = {}
855 for index = 2, #arguments do
856 query[self:term_argument(arguments[index])] = true
857 end
858 local ranked = {}
859 for _, document in pairs(self.documents) do
860 local counts = self:term_frequencies(document)
861 local matched, total = 0, 0
862 for stem in pairs(query) do
863 local count = counts[stem] or 0
864 if count > 0 then
865 matched = matched + 1
866 end
867 total = total + count
868 end
869 if matched > 0 then
870 ranked[#ranked + 1] = { matched = matched, total = total, number = document.number }
871 end
872 end
873 table.sort(ranked, function(first, second)
874 if first.matched ~= second.matched then
875 return first.matched > second.matched
876 end
877 if first.total ~= second.total then
878 return first.total > second.total
879 end
880 return first.number < second.number
881 end)
882 local items = {}
883 for index = 1, math.min(limit, #ranked) do
884 local entry = ranked[index]
885 items[index] = entry.number .. ":" .. entry.matched .. ":" .. entry.total
886 end
887 self:emit("RANK count=" .. #ranked, table.unpack(items))
888end

1002function SearchEngine:execute(line)
1003 local name, rest = line:match("^(.-) (.*)$")
1004 if name == nil then
1005 name = line
1006 end
1007 local ok, problem = pcall(function()
1008 local handler = HANDLERS[name]
1009 if handler == nil then
1010 fail("UNKNOWN COMMAND " .. name)
1011 end
1012 local minimum, maximum, run = table.unpack(handler)
1013 local arguments = rest and split_arguments(name, rest) or {}
1014 if #arguments < minimum or #arguments > maximum then
1015 fail("BAD ARGUMENTS " .. name)
1016 end
1017 run(self, arguments)
1018 end)
1019 if not ok then
1020 if type(problem) ~= "table" then
1021 error(problem, 0)
1022 end
1023 self:emit("ERROR", problem.message)
1024 end
1025end

1027local function main()
1028 local lines = {}
1029 for line in io.lines() do
1030 lines[#lines + 1] = line
1031 end
1032 local stopwords = {}
1033 local header = split_words(lines[1])
1034 for index = 2, #header do
1035 stopwords[header[index]] = true
1036 end
1037 local engine = SearchEngine.new(stopwords)
1038 local index = 2
1039 while starts_with(lines[index], "DOC ") do
1040 local document = engine:add_document(tonumber(lines[index]:sub(5)))
1041 index = index + 1
1042 while lines[index] ~= "ENDDOC" do
1043 document:add_line(lines[index])
1044 index = index + 1
1045 end
1046 index = index + 1
1047 end
1048 while lines[index] ~= nil and lines[index] ~= "END" do
