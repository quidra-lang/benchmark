38func letterPattern(word string) string {
39 kinds := make([]byte, len(word))
40 for index := range len(word) {
41 letter := word[index]
42 vowel := strings.IndexByte(vowels, letter) >= 0 ||
43 (letter == 'y' && index > 0 && kinds[index-1] == 'C')
44 if vowel {
45 kinds[index] = 'V'
46 } else {
47 kinds[index] = 'C'
48 }
49 }
50 return string(kinds)
51}

53func measure(word string) int {
54 return strings.Count(letterPattern(word), "VC")
55}
56
57func hasVowel(word string) bool {
58 return strings.Contains(letterPattern(word), "V")
59}
60
61func endsDouble(word string) bool {
62 return strings.HasSuffix(letterPattern(word), "CC") && word[len(word)-1] == word[len(word)-2]
63}
64
65func endsCvc(word string) bool {
66 return strings.HasSuffix(letterPattern(word), "CVC") &&
67 strings.IndexByte("wxy", word[len(word)-1]) < 0
68}

82func ionCondition(remainder string) bool {
83 return measure(remainder) > 1 &&

98func rulesWith(condition func(string) bool, replacements [][2]string) []rule {

106func step4Rules() []rule {
107 rules := make([]rule, 0, len(step4Suffixes))
108 for _, suffix := range step4Suffixes {
109 condition := measureAboveOne
110 if suffix == "ion" {
111 condition = ionCondition
112 }
113 rules = append(rules, rule{suffix, "", condition})
114 }
115 return rules
116}

137func applyRules(word string, rules []rule) (string, string) {
138 var selected *rule
139 for index := range rules {
140 candidate := &rules[index]
141 longer := selected == nil || len(candidate.suffix) > len(selected.suffix)
142 if strings.HasSuffix(word, candidate.suffix) && longer {
143 selected = candidate
144 }
145 }
146 if selected == nil {
147 return word, ""
148 }
149 remainder := strings.TrimSuffix(word, selected.suffix)
150 if !selected.condition(remainder) {
151 return word, ""
152 }
153 return remainder + selected.replacement, selected.suffix
154}

156func tidyStep1b(word string) string {
157 if strings.HasSuffix(word, "at") || strings.HasSuffix(word, "bl") ||
158 strings.HasSuffix(word, "iz") {
159 return word + "e"
160 }
161 if endsDouble(word) && strings.IndexByte("lsz", word[len(word)-1]) < 0 {
162 return word[:len(word)-1]
163 }
164 if measure(word) == 1 && endsCvc(word) {
165 return word + "e"
166 }
167 return word
168}

170func applyStep(name, word string) string {
171 if name == "5b" {
172 if measure(word) > 1 && endsDouble(word) && strings.HasSuffix(word, "l") {
173 return word[:len(word)-1]
174 }
175 return word
176 }
177 result, suffix := applyRules(word, ruleTables[name])
178 if name == "1b" && (suffix == "ed" || suffix == "ing") {
179 return tidyStep1b(result)
180 }
181 return result
182}

189func stemmingSteps(word string) (string, []stepChange) {
190 var changes []stepChange
191 if len(word) > 2 {
192 for _, name := range stepNames {
193 result := applyStep(name, word)
194 if result != word {
195 changes = append(changes, stepChange{name, result})
196 }
197 word = result
198 }

200 return word, changes

249func rankedCounts(counts map[string]int) []stemCount {
250 ranked := make([]stemCount, 0, len(counts))
251 for stem, count := range counts {
252 ranked = append(ranked, stemCount{stem, count})
253 }
254 slices.SortFunc(ranked, func(left, right stemCount) int {

260func closestDistance(positions, others []int, sameStem bool) (int, bool) {
261 var gaps []int
262 if sameStem {
263 for index := 1; index < len(positions); index++ {
264 gaps = append(gaps, positions[index]-positions[index-1])
265 }
266 } else {
267 type marker struct {
268 position int
269 source int
270 }
271 var merged []marker
272 for _, position := range positions {
273 merged = append(merged, marker{position, 0})
274 }
275 for _, other := range others {
276 merged = append(merged, marker{other, 1})
277 }
278 slices.SortFunc(merged, func(left, right marker) int {
279 return cmp.Compare(left.position, right.position)
280 })
281 for index := 1; index < len(merged); index++ {
282 if merged[index-1].source != merged[index].source {
283 gaps = append(gaps, merged[index].position-merged[index-1].position)
284 }
285 }
286 }
287 if len(gaps) == 0 {
288 return 0, false
289 }
290 return slices.Min(gaps), true
291}

293func gcd(first, second int) int {
294 for second != 0 {
295 first, second = second, first%second
296 }
297 return first
298}

300func jaccard(common, union int) (int, int) {
301 if union == 0 {
302 return 0, 1
303 }
304 divisor := gcd(common, union)
305 return common / divisor, union / divisor
306}

371type lexemeKind int

374 kindEnd lexemeKind = iota
375 kindAnd
376 kindOr
377 kindNot

380 kindTerm

384var symbols = map[string]lexemeKind{
385 "AND": kindAnd, "OR": kindOr, "NOT": kindNot, "(": kindOpen, ")": kindClose,

388type queryItem struct {
389 kind lexemeKind
390 spelling string

393func classify(lexeme string) (queryItem, error) {
394 if kind, found := symbols[lexeme]; found {
395 return queryItem{kind, lexeme}, nil
396 }
397 if wordPattern.MatchString(lexeme) {
398 return queryItem{kindTerm, strings.ToLower(lexeme)}, nil
399 }
400 if prefixTermPattern.MatchString(lexeme) {
401 return queryItem{kindPrefix, strings.ToLower(strings.TrimSuffix(lexeme, "*"))}, nil
402 }
403 return queryItem{}, errors.New("BAD TERM " + lexeme)
404}

406type queryResult struct {
407 canonical string
408 documents map[int]bool
409}

411type queryParser struct {
412 items []queryItem
413 position int
414 stemSets map[int]map[string]bool
415}
416
417func (p *queryParser) peek() lexemeKind {
418 if p.position < len(p.items) {
419 return p.items[p.position].kind
420 }
421 return kindEnd
422}
423
424func (p *queryParser) take() (queryItem, error) {
425 if p.position >= len(p.items) {
426 return queryItem{}, errSyntax
427 }
428 item := p.items[p.position]
429 p.position++
430 return item, nil
431}
432
433func (p *queryParser) parse() (queryResult, error) {
434 result, err := p.expression()
435 if err != nil {
436 return queryResult{}, err
437 }
438 if p.position < len(p.items) {
439 return queryResult{}, errSyntax
440 }
441 return result, nil
442}
443
444func (p *queryParser) expression() (queryResult, error) {
445 result, err := p.conjunction()
446 if err != nil {
447 return queryResult{}, err
448 }
449 for p.peek() == kindOr {
450 p.position++
451 right, err := p.conjunction()
452 if err != nil {
453 return queryResult{}, err
454 }
455 documents := maps.Clone(result.documents)
456 maps.Copy(documents, right.documents)
457 result = queryResult{"(" + result.canonical + " OR " + right.canonical + ")", documents}
458 }
459 return result, nil
460}
461
462func continuesConjunction(kind lexemeKind) bool {
463 switch kind {
464 case kindAnd, kindNot, kindOpen, kindTerm, kindPrefix:
465 return true
466 }
467 return false
468}
469
470func (p *queryParser) conjunction() (queryResult, error) {
471 result, err := p.negation()
472 if err != nil {
473 return queryResult{}, err
474 }
475 for continuesConjunction(p.peek()) {
476 if p.peek() == kindAnd {
477 p.position++
478 }
479 right, err := p.negation()
480 if err != nil {
481 return queryResult{}, err
482 }
483 documents := make(map[int]bool)
484 for number := range result.documents {
485 if right.documents[number] {
486 documents[number] = true
487 }
488 }
489 result = queryResult{"(" + result.canonical + " AND " + right.canonical + ")", documents}
490 }
491 return result, nil
492}
493
494func (p *queryParser) negation() (queryResult, error) {
495 if p.peek() != kindNot {
496 return p.element()
497 }
498 p.position++
499 inner, err := p.negation()
500 if err != nil {
501 return queryResult{}, err
502 }
503 documents := make(map[int]bool)
504 for number := range p.stemSets {
505 if !inner.documents[number] {
506 documents[number] = true
507 }
508 }
509 return queryResult{"(NOT " + inner.canonical + ")", documents}, nil
510}
511
512func (p *queryParser) element() (queryResult, error) {
513 item, err := p.take()
514 if err != nil {
515 return queryResult{}, err
516 }
517 switch item.kind {
518 case kindOpen:
519 result, err := p.expression()
520 if err != nil {
521 return queryResult{}, err
522 }
523 closing, err := p.take()
524 if err != nil || closing.kind != kindClose {
525 return queryResult{}, errSyntax
526 }
527 return result, nil
528 case kindTerm:
529 stem := porterStem(item.spelling)
530 documents := p.documentsWhere(func(stems map[string]bool) bool {
531 return stems[stem]
532 })
533 return queryResult{stem, documents}, nil
534 case kindPrefix:
535 documents := p.documentsWhere(func(stems map[string]bool) bool {
536 for stem := range stems {
537 if strings.HasPrefix(stem, item.spelling) {
538 return true
539 }
540 }
541 return false
542 })
543 return queryResult{item.spelling + "*", documents}, nil
544 }
545 return queryResult{}, errSyntax
546}
547
548func (p *queryParser) documentsWhere(condition func(map[string]bool) bool) map[int]bool {
549 documents := make(map[int]bool)
550 for number, stems := range p.stemSets {
551 if condition(stems) {
552 documents[number] = true
553 }
554 }
555 return documents
556}

794func (e *searchEngine) search(arguments []string) error {
795 var items []queryItem
796 for _, lexeme := range lexemePattern.FindAllString(arguments[0], -1) {
797 item, err := classify(lexeme)
798 if err != nil {
799 return err
800 }
801 items = append(items, item)
802 }
803 stemSets := make(map[int]map[string]bool)
804 for number, doc := range e.documents {
805 stemSets[number] = e.stemSet(doc)
806 }
807 parser := queryParser{items: items, stemSets: stemSets}
808 result, err := parser.parse()
809 if err != nil {
810 return err
811 }
812 for _, item := range items {
813 if item.kind == kindTerm && e.stopwords[item.spelling] {
814 return errors.New("STOPWORD " + item.spelling)
815 }
816 }
817 var numbers []string
818 for _, number := range slices.Sorted(maps.Keys(result.documents)) {
819 numbers = append(numbers, strconv.Itoa(number))
820 }
821 e.emit(fmt.Sprintf("SEARCH %s count=%d", result.canonical, len(numbers)), numbers...)
822 return nil
823}

825func (e *searchEngine) phrase(arguments []string) error {
826 words := make([]string, 0, len(arguments))
827 for _, argument := range arguments {
828 word, err := wordArgument(argument)
829 if err != nil {
830 return err
831 }
832 words = append(words, word)
833 }
834 if !slices.ContainsFunc(words, func(word string) bool { return !e.stopwords[word] }) {
835 return errors.New("STOPWORDS ONLY")
836 }
837 targets := make([]string, len(words))
838 for index, word := range words {
839 if e.stopwords[word] {
840 targets[index] = "?"
841 } else {
842 targets[index] = porterStem(word)
843 }
844 }
845 var matches []string
846 for _, doc := range e.orderedDocuments() {
847 for start := 0; start+len(targets) <= len(doc.tokens); start++ {
848 matched := true
849 for offset, target := range targets {
850 if !e.tokenMatches(doc.tokens[start+offset], target) {
851 matched = false
852 break
853 }
854 }
855 if matched {
856 matches = append(matches, doc.reference(doc.tokens[start]))
857 }
858 }
859 }
860 header := fmt.Sprintf("PHRASE \"%s\" matches=%d", strings.Join(targets, " "), len(matches))
861 e.emit(header, matches...)
862 return nil
863}

972func (e *searchEngine) similar(arguments []string) error {
973 doc, err := e.documentArgument(arguments[0])
974 if err != nil {
975 return err
976 }
977 limit, err := numberArgument(arguments[1], 1, maxCount)
978 if err != nil {
979 return err
980 }
981 type similarity struct {
982 number int
983 numerator int
984 denominator int
985 }
986 stems := e.stemSet(doc)
987 var scores []similarity
988 for _, other := range e.documents {
989 if other == doc {
990 continue
991 }
992 otherStems := e.stemSet(other)
993 common := 0
994 for stem := range stems {
995 if otherStems[stem] {
996 common++
997 }
998 }
999 numerator, denominator := jaccard(common, len(stems)+len(otherStems)-common)
1000 scores = append(scores, similarity{other.number, numerator, denominator})
1001 }
1002 slices.SortFunc(scores, func(left, right similarity) int {
1003 return cmp.Or(
1004 cmp.Compare(right.numerator*left.denominator, left.numerator*right.denominator),
1005 cmp.Compare(left.number, right.number),
1006 )
1007 })
1008 var items []string
1009 for _, score := range scores[:min(limit, len(scores))] {
1010 items = append(items,
1011 fmt.Sprintf("%d:%d/%d", score.number, score.numerator, score.denominator))
1012 }
1013 e.emit(fmt.Sprintf("SIMILAR %d", doc.number), items...)
1014 return nil
1015}

1017func (e *searchEngine) rank(arguments []string) error {
1018 limit, err := numberArgument(arguments[0], 1, maxCount)
1019 if err != nil {
1020 return err
1021 }
1022 query := make(map[string]bool)
1023 for _, argument := range arguments[1:] {
1024 stem, err := e.termArgument(argument)
1025 if err != nil {
1026 return err
1027 }
1028 query[stem] = true
1029 }
1030 type ranking struct {
1031 number int
1032 matched int
1033 total int
1034 }
1035 var ranked []ranking
1036 for _, doc := range e.documents {
1037 counts := e.termFrequencies(doc)
1038 matched, total := 0, 0
1039 for stem := range query {
1040 if counts[stem] > 0 {
1041 matched++
1042 }
1043 total += counts[stem]
1044 }
1045 if matched > 0 {
1046 ranked = append(ranked, ranking{doc.number, matched, total})
1047 }
1048 }
1049 slices.SortFunc(ranked, func(left, right ranking) int {
1050 return cmp.Or(
1051 cmp.Compare(right.matched, left.matched),
1052 cmp.Compare(right.total, left.total),
1053 cmp.Compare(left.number, right.number),
1054 )
1055 })
1056 var items []string
1057 for _, entry := range ranked[:min(limit, len(ranked))] {
1058 items = append(items, fmt.Sprintf("%d:%d:%d", entry.number, entry.matched, entry.total))
1059 }
1060 e.emit(fmt.Sprintf("RANK count=%d", len(ranked)), items...)
1061 return nil
1062}
