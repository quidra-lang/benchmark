79std::string letterPattern(const std::string& word) {
80 std::string kinds;
81 for (std::size_t index = 0; index < word.size(); ++index) {
82 char letter = word[index];
83 bool vowel =
84 contains("aeiou", letter) || (letter == 'y' && index > 0 && kinds.back() == 'C');
85 kinds += vowel ? 'V' : 'C';
86 }
87 return kinds;
88}

90int measure(const std::string& word) {
91 std::string pattern = letterPattern(word);
92 int count = 0;
93 for (std::size_t index = 1; index < pattern.size(); ++index) {
94 if (pattern[index - 1] == 'V' && pattern[index] == 'C') {
95 ++count;
96 }
97 }
98 return count;
99}

177struct RuleOutcome {
178 std::string word;
179 std::optional<std::string> suffix;
180};
181
182RuleOutcome applyRules(const std::string& word, const std::vector<Rule>& rules) {
183 const Rule* selected = nullptr;
184 for (const Rule& rule : rules) {
185 bool longer = selected == nullptr || rule.suffix.size() > selected->suffix.size();
186 if (endsWith(word, rule.suffix) && longer) {
187 selected = &rule;
188 }
189 }
190 if (selected == nullptr) {
191 return {word, std::nullopt};
192 }
193 std::string remainder = word.substr(0, word.size() - selected->suffix.size());
194 if (!selected->condition(remainder)) {
195 return {word, std::nullopt};
196 }
197 return {remainder + selected->replacement, selected->suffix};
198}

200std::string tidyStep1b(const std::string& word) {
201 if (endsWith(word, "at") || endsWith(word, "bl") || endsWith(word, "iz")) {
202 return word + "e";
203 }
204 if (endsDouble(word) && !contains("lsz", word.back())) {
205 return word.substr(0, word.size() - 1);
206 }
207 if (measure(word) == 1 && endsCvc(word)) {
208 return word + "e";
209 }
210 return word;
211}

213std::string applyStep(const std::string& name, const std::string& word) {
214 if (name == "5b") {
215 if (measure(word) > 1 && endsDouble(word) && endsWith(word, "l")) {
216 return word.substr(0, word.size() - 1);
217 }
218 return word;
219 }
220 RuleOutcome outcome = applyRules(word, ruleTables.at(name));
221 if (name == "1b" && (outcome.suffix == "ed" || outcome.suffix == "ing")) {
222 return tidyStep1b(outcome.word);
223 }
224 return outcome.word;
225}

278std::vector<std::string> splitArguments(const std::string& name, const std::string& rest) {
279 if (name == "SEARCH") {
280 return {rest};
281 }
282 if (name == "APPEND") {
283 std::size_t space = rest.find(' ');
284 if (space == std::string::npos) {
285 return {rest};
286 }
287 return {rest.substr(0, space), rest.substr(space + 1)};
288 }
289 std::vector<std::string> parts;
290 std::size_t start = 0;
291 while (true) {
292 std::size_t space = rest.find(' ', start);
293 parts.push_back(rest.substr(start, space - start));
294 if (space == std::string::npos) {
295 return parts;
296 }
297 start = space + 1;
298 }
299}

311 });

320std::optional<int> closestDistance(const std::vector<int>& positions,
321 const std::vector<int>& others, bool sameStem) {
322 std::vector<int> gaps;
323 if (sameStem) {
324 for (std::size_t index = 1; index < positions.size(); ++index) {
325 gaps.push_back(positions[index] - positions[index - 1]);
326 }
327 } else {
328 std::vector<std::pair<int, int>> merged;
329 for (int position : positions) {
330 merged.emplace_back(position, 0);
331 }
332 for (int other : others) {
333 merged.emplace_back(other, 1);
334 }
335 std::sort(merged.begin(), merged.end());
336 for (std::size_t index = 1; index < merged.size(); ++index) {
337 if (merged[index - 1].second != merged[index].second) {
338 gaps.push_back(merged[index].first - merged[index - 1].first);
339 }
340 }
341 }
342 if (gaps.empty()) {
343 return std::nullopt;
344 }
345 return *std::min_element(gaps.begin(), gaps.end());
346}

433enum class ItemKind { And, Or, Not, Open, Close, Term, Prefix };
434
435struct QueryItem {
436 ItemKind kind;
437 std::string spelling;
438};
439
440const std::map<std::string, ItemKind> symbols = {
441 {"AND", ItemKind::And}, {"OR", ItemKind::Or}, {"NOT", ItemKind::Not},
442 {"(", ItemKind::Open}, {")", ItemKind::Close},
443};
444
445std::vector<std::string> lexemes(const std::string& expression) {
446 std::vector<std::string> result;
447 std::size_t index = 0;
448 while (index < expression.size()) {
449 char character = expression[index];
450 if (character == ' ') {
451 ++index;
452 } else if (character == '(' || character == ')') {
453 result.emplace_back(1, character);
454 ++index;
455 } else {
456 std::size_t start = index;
457 while (index < expression.size() && !contains(" ()", expression[index])) {
458 ++index;
459 }
460 result.push_back(expression.substr(start, index - start));
461 }
462 }
463 return result;
464}
465
466QueryItem classify(const std::string& lexeme) {
467 auto symbol = symbols.find(lexeme);
468 if (symbol != symbols.end()) {
469 return {symbol->second, lexeme};
470 }
471 if (isWord(lexeme)) {
472 return {ItemKind::Term, toLower(lexeme)};
473 }
474 std::string letters = lexeme.substr(0, lexeme.size() - 1);
475 if (lexeme.back() == '*' && isWord(letters)) {
476 return {ItemKind::Prefix, toLower(letters)};
477 }
478 throw CommandError("BAD TERM " + lexeme);
479}

481using StemSets = std::map<int, std::set<std::string>>;

483struct QueryResult {
484 std::string canonical;
485 std::set<int> documents;
486};
487
488class QueryParser {
489public:
490 QueryParser(const std::vector<QueryItem>& items, const StemSets& stemSets)
491 : items(items), stemSets(stemSets) {}
492
493 QueryResult parse() {
494 QueryResult result = expression();
495 if (position < items.size()) {
496 throw CommandError("SYNTAX");
497 }
498 return result;
499 }
500
501private:
502 const std::vector<QueryItem>& items;
503 const StemSets& stemSets;
504 std::size_t position = 0;
505
506 std::optional<ItemKind> peek() const {
507 if (position < items.size()) {
508 return items[position].kind;
509 }
510 return std::nullopt;
511 }
512
513 const QueryItem& take() {
514 if (position >= items.size()) {
515 throw CommandError("SYNTAX");
516 }
517 return items[position++];
518 }
519
520 bool continuesConjunction() const {
521 std::optional<ItemKind> kind = peek();
522 return kind == ItemKind::And || kind == ItemKind::Not || kind == ItemKind::Open ||
523 kind == ItemKind::Term || kind == ItemKind::Prefix;
524 }
525
526 QueryResult expression() {
527 QueryResult result = conjunction();
528 while (peek() == ItemKind::Or) {
529 ++position;
530 QueryResult right = conjunction();
531 result.canonical = "(" + result.canonical + " OR " + right.canonical + ")";
532 result.documents.insert(right.documents.begin(), right.documents.end());
533 }
534 return result;
535 }
536
537 QueryResult conjunction() {
538 QueryResult result = negation();
539 while (continuesConjunction()) {
540 if (peek() == ItemKind::And) {
541 ++position;
542 }
543 QueryResult right = negation();
544 std::set<int> common;
545 std::set_intersection(result.documents.begin(), result.documents.end(),
546 right.documents.begin(), right.documents.end(),
547 std::inserter(common, common.end()));
548 result.canonical = "(" + result.canonical + " AND " + right.canonical + ")";
549 result.documents = common;
550 }
551 return result;
552 }
553
554 QueryResult negation() {
555 if (peek() == ItemKind::Not) {
556 ++position;
557 QueryResult inner = negation();
558 std::set<int> documents;
559 for (const auto& [number, stems] : stemSets) {
560 if (inner.documents.count(number) == 0) {
561 documents.insert(number);
562 }
563 }
564 return {"(NOT " + inner.canonical + ")", documents};
565 }
566 return element();
567 }
568
569 QueryResult element() {
570 const QueryItem& item = take();
571 if (item.kind == ItemKind::Open) {
572 QueryResult result = expression();
573 if (take().kind != ItemKind::Close) {
574 throw CommandError("SYNTAX");
575 }
576 return result;
577 }
578 if (item.kind == ItemKind::Term) {
579 std::string stem = porterStem(item.spelling);
580 std::set<int> documents;
581 for (const auto& [number, stems] : stemSets) {
582 if (stems.count(stem) > 0) {
583 documents.insert(number);
584 }
585 }
586 return {stem, documents};
587 }
588 if (item.kind == ItemKind::Prefix) {
589 std::set<int> documents;
590 for (const auto& [number, stems] : stemSets) {
591 bool found =
592 std::any_of(stems.begin(), stems.end(), [&item](const std::string& stem) {
593 return startsWith(stem, item.spelling);
594 });
595 if (found) {
596 documents.insert(number);
597 }
598 }
599 return {item.spelling + "*", documents};
600 }
601 throw CommandError("SYNTAX");
602 }

605class SearchEngine {
606public:
607 SearchEngine(std::set<std::string> stopwords, std::ostream& out)
608 : stopwords(std::move(stopwords)), out(out) {}
609
610 Document& addDocument(int number) {
611 return documents.emplace(number, Document(number)).first->second;

613
614 void execute(const std::string& line) {
615 std::size_t space = line.find(' ');
616 std::string name = line.substr(0, space);
617 try {
618 auto found = commands().find(name);
619 if (found == commands().end()) {
620 throw CommandError("UNKNOWN COMMAND " + name);
621 }
622 const Command& command = found->second;
623 Arguments arguments;
624 if (space != std::string::npos) {
625 arguments = splitArguments(name, line.substr(space + 1));
626 }
627 if (arguments.size() < command.minimum || arguments.size() > command.maximum) {
628 throw CommandError("BAD ARGUMENTS " + name);
629 }
630 (this->*command.handler)(arguments);
631 } catch (const CommandError& error) {
632 emit("ERROR " + std::string(error.what()));
633 }
634 }

761 std::count_if(document.tokens.begin(), document.tokens.end(),
762 [&form](const Token& token) { return token.form == form; }));
763 }
764 return count;
765 }

865 void phrase(const Arguments& arguments) {
866 std::vector<std::string> words;
867 for (const std::string& argument : arguments) {
868 words.push_back(wordArgument(argument));
869 }
870 if (std::all_of(words.begin(), words.end(),
871 [this](const std::string& word) { return isStopword(word); })) {
872 throw CommandError("STOPWORDS ONLY");
873 }
874 std::vector<std::string> targets;
875 for (const std::string& word : words) {
876 targets.push_back(isStopword(word) ? "?" : porterStem(word));
877 }
878 std::vector<std::string> matches;
879 for (const auto& [number, document] : documents) {
880 const std::vector<Token>& tokens = document.tokens;
881 for (std::size_t start = 0; start + targets.size() <= tokens.size(); ++start) {
882 bool matched = true;
883 for (std::size_t offset = 0; offset < targets.size() && matched; ++offset) {
884 matched = tokenMatches(tokens[start + offset], targets[offset]);
885 }
886 if (matched) {
887 matches.push_back(document.reference(tokens[start]));
888 }
889 }
890 }
891 emit("PHRASE \"" + join(targets, " ") + "\" matches=" + std::to_string(matches.size()),
892 matches);
893 }

974 void similar(const Arguments& arguments) {
975 const Document& document = documentArgument(arguments[0]);
976 int limit = numberArgument(arguments[1], 1, maxCount);
977 std::set<std::string> stems = stemSet(document);
978 std::vector<Similarity> scores;
979 for (const auto& [number, other] : documents) {
980 if (number == document.number) {
981 continue;
982 }
983 std::set<std::string> otherStems = stemSet(other);
984 int common = static_cast<int>(
985 std::count_if(stems.begin(), stems.end(), [&otherStems](const std::string& stem) {
986 return otherStems.count(stem) > 0;
987 }));
988 int unionSize = static_cast<int>(stems.size() + otherStems.size()) - common;
989 Fraction score{0, 1};
990 if (unionSize > 0) {
991 int divisor = std::gcd(common, unionSize);
992 score = {common / divisor, unionSize / divisor};
993 }
994 scores.push_back({number, score});
995 }
996 auto byScore = [](const Similarity& left, const Similarity& right) {
997 long long leftScore =
998 static_cast<long long>(left.score.numerator) * right.score.denominator;
999 long long rightScore =
1000 static_cast<long long>(right.score.numerator) * left.score.denominator;
1001 if (leftScore != rightScore) {
1002 return leftScore > rightScore;
1003 }
1004 return left.document < right.document;
1005 };
1006 std::sort(scores.begin(), scores.end(), byScore);
1007 scores.resize(std::min(scores.size(), static_cast<std::size_t>(limit)));
1008 std::vector<std::string> items;
1009 for (const Similarity& entry : scores) {
1010 items.push_back(std::to_string(entry.document) + ":" +
1011 std::to_string(entry.score.numerator) + "/" +
1012 std::to_string(entry.score.denominator));
1013 }
1014 emit("SIMILAR " + std::to_string(document.number), items);
1015 }

1017 void rank(const Arguments& arguments) {
1018 int limit = numberArgument(arguments[0], 1, maxCount);
1019 std::set<std::string> query;
1020 for (std::size_t index = 1; index < arguments.size(); ++index) {
1021 query.insert(termArgument(arguments[index]));
1022 }
1023 std::vector<RankEntry> ranked;
1024 for (const auto& [number, document] : documents) {
1025 Counts counts = termFrequencies(document);
1026 int matched = 0;
1027 int total = 0;
1028 for (const std::string& stem : query) {
1029 int count = countOf(counts, stem);
1030 if (count > 0) {
1031 ++matched;
1032 }
1033 total += count;
1034 }
1035 if (matched > 0) {
1036 ranked.push_back({number, matched, total});
1037 }
1038 }
1039 std::sort(ranked.begin(), ranked.end(), [](const RankEntry& left, const RankEntry& right) {
1040 if (left.matched != right.matched) {
1041 return left.matched > right.matched;
1042 }
1043 if (left.total != right.total) {
1044 return left.total > right.total;
1045 }
1046 return left.document < right.document;
1047 });
1048 std::vector<std::string> items;
1049 std::size_t shown = std::min(ranked.size(), static_cast<std::size_t>(limit));
1050 for (std::size_t index = 0; index < shown; ++index) {
1051 const RankEntry& entry = ranked[index];
1052 items.push_back(std::to_string(entry.document) + ":" + std::to_string(entry.matched) +
1053 ":" + std::to_string(entry.total));
1054 }
1055 emit("RANK count=" + std::to_string(ranked.size()), items);
1056 }
