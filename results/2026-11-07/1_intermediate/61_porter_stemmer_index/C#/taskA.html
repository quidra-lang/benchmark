12var index = 1;

45 static readonly string[] Step4Suffixes =
46 [
47 "al", "ance", "ence", "er", "ic", "able", "ible", "ant", "ement", "ment",

52 {

54 ["1b"] =

57 ],
58 ["1c"] = [new("y", "i", HasVowel)],
59 ["2"] = RulesWith(

62 ("izer", "ize"), ("abli", "able"), ("alli", "al"), ("entli", "ent"), ("eli", "e"),

80 sealed record Rule(string Suffix, string Replacement, Func<string, bool> Condition);

82 static Rule[] RulesWith(
83 Func<string, bool> condition, params (string Suffix, string Replacement)[] rules) =>
84 [.. rules.Select(rule => new Rule(rule.Suffix, rule.Replacement, condition))];

86 static bool EndsWith(string word, string suffix) =>
87 word.EndsWith(suffix, StringComparison.Ordinal);
88
89 public static string LetterPattern(string word)
90 {
91 var kinds = new StringBuilder();
92 for (var index = 0; index < word.Length; index++)
93 {
94 var letter = word[index];
95 var vowel = Vowels.Contains(letter)
96 || (letter == 'y' && index > 0 && kinds[index - 1] == 'C');
97 kinds.Append(vowel ? 'V' : 'C');
98 }
99 return kinds.ToString();
100 }
101
102 public static int Measure(string word) => LetterPattern(word).Split("VC").Length - 1;
103
104 public static bool HasVowel(string word) => LetterPattern(word).Contains('V');
105
106 public static bool EndsDouble(string word) =>
107 EndsWith(LetterPattern(word), "CC") && word[^1] == word[^2];
108
109 public static bool EndsCvc(string word) =>
110 EndsWith(LetterPattern(word), "CVC") && !"wxy".Contains(word[^1]);
111
112 static bool MeasureAboveZero(string remainder) => Measure(remainder) > 0;
113
114 static bool MeasureAboveOne(string remainder) => Measure(remainder) > 1;
115
116 static bool IonCondition(string remainder) =>
117 Measure(remainder) > 1 && (remainder.EndsWith('s') || remainder.EndsWith('t'));
118
119 static bool FinalECondition(string remainder)
120 {
121 var value = Measure(remainder);
122 return value > 1 || (value == 1 && !EndsCvc(remainder));
123 }
124
125 static (string Word, string? Suffix) ApplyRules(string word, Rule[] rules)
126 {
127 var rule = rules
128 .Where(candidate => EndsWith(word, candidate.Suffix))
129 .MaxBy(candidate => candidate.Suffix.Length);
130 if (rule is null)
131 {
132 return (word, null);
133 }
134 var remainder = word[..^rule.Suffix.Length];
135 if (!rule.Condition(remainder))
136 {
137 return (word, null);
138 }
139 return (remainder + rule.Replacement, rule.Suffix);
140 }
141
142 static string TidyStep1b(string word)
143 {
144 if (EndsWith(word, "at") || EndsWith(word, "bl") || EndsWith(word, "iz"))
145 {
146 return word + "e";
147 }
148 if (EndsDouble(word) && !"lsz".Contains(word[^1]))
149 {
150 return word[..^1];
151 }
152 if (Measure(word) == 1 && EndsCvc(word))
153 {
154 return word + "e";
155 }
156 return word;
157 }
158
159 static string ApplyStep(string name, string word)
160 {
161 if (name == "5b")
162 {
163 return Measure(word) > 1 && EndsDouble(word) && word.EndsWith('l') ? word[..^1] : word;
164 }
165 var (result, suffix) = ApplyRules(word, RuleTables[name]);
166 return name == "1b" && suffix is "ed" or "ing" ? TidyStep1b(result) : result;
167 }
168
169 public static (string Stem, List<(string Name, string Result)> Changes) StemmingSteps(
170 string word)
171 {
172 var changes = new List<(string Name, string Result)>();
173 if (word.Length > 2)
174 {
175 foreach (var name in StepNames)
176 {
177 var result = ApplyStep(name, word);
178 if (result != word)
179 {
180 changes.Add((name, result));
181 }
182 word = result;
183 }
184 }
185 return (word, changes);
186 }
187
188 public static string Stem(string word)
189 {
190 if (!Cache.TryGetValue(word, out var stem))
191 {
192 stem = StemmingSteps(word).Stem;
193 Cache[word] = stem;
194 }
195 return stem;
196 }
197}

242record Lexeme(LexemeKind Kind, string Spelling);

244class QueryParser(List<Lexeme> items, Dictionary<int, HashSet<string>> stemSets)
245{
246 int position;
247
248 LexemeKind? Peek() => position < items.Count ? items[position].Kind : null;
249
250 Lexeme Take() => position < items.Count
251 ? items[position++]
252 : throw new CommandException("SYNTAX");
253
254 public (string Canonical, HashSet<int> Documents) Parse()
255 {
256 var result = Expression();
257 if (position < items.Count)
258 {
259 throw new CommandException("SYNTAX");
260 }
261 return result;
262 }
263
264 (string Canonical, HashSet<int> Documents) Expression()
265 {
266 var (canonical, documents) = Conjunction();
267 while (Peek() == LexemeKind.Or)
268 {
269 position++;
270 var (right, rightDocuments) = Conjunction();
271 canonical = $"({canonical} OR {right})";
272 documents.UnionWith(rightDocuments);
273 }
274 return (canonical, documents);
275 }
276
277 (string Canonical, HashSet<int> Documents) Conjunction()
278 {
279 var (canonical, documents) = Negation();
280 while (Peek() is LexemeKind.And or LexemeKind.Not or LexemeKind.Open or LexemeKind.Term
281 or LexemeKind.Prefix)
282 {
283 if (Peek() == LexemeKind.And)
284 {
285 position++;
286 }
287 var (right, rightDocuments) = Negation();
288 canonical = $"({canonical} AND {right})";
289 documents.IntersectWith(rightDocuments);
290 }
291 return (canonical, documents);
292 }
293
294 (string Canonical, HashSet<int> Documents) Negation()
295 {
296 if (Peek() == LexemeKind.Not)
297 {
298 position++;
299 var (canonical, documents) = Negation();
300 var complement = stemSets.Keys.Where(number => !documents.Contains(number)).ToHashSet();
301 return ($"(NOT {canonical})", complement);
302 }
303 return Element();
304 }
305
306 (string Canonical, HashSet<int> Documents) Element()
307 {
308 var item = Take();
309 switch (item.Kind)
310 {
311 case LexemeKind.Open:
312 {
313 var result = Expression();
314 if (Take().Kind != LexemeKind.Close)
315 {
316 throw new CommandException("SYNTAX");
317 }
318 return result;
319 }
320 case LexemeKind.Term:
321 {
322 var stem = Porter.Stem(item.Spelling);
323 return (stem, Matching(stems => stems.Contains(stem)));
324 }
325 case LexemeKind.Prefix:
326 {
327 var documents = Matching(stems => stems.Any(candidate =>
328 candidate.StartsWith(item.Spelling, StringComparison.Ordinal)));
329 return ($"{item.Spelling}*", documents);
330 }
331 default:
332 throw new CommandException("SYNTAX");
333 }
334 }
335
336 HashSet<int> Matching(Func<HashSet<string>, bool> predicate) =>
337 stemSets.Where(pair => predicate(pair.Value)).Select(pair => pair.Key).ToHashSet();
338}

340readonly record struct Fraction(int Numerator, int Denominator) : IComparable<Fraction>
341{
342 public static Fraction Create(int numerator, int denominator)
343 {
344 var divisor = Gcd(numerator, denominator);
345 return new Fraction(numerator / divisor, denominator / divisor);
346 }
347
348 static int Gcd(int a, int b) => b == 0 ? a : Gcd(b, a % b);
349
350 public int CompareTo(Fraction other) =>
351 (Numerator * other.Denominator).CompareTo(other.Numerator * Denominator);
352
353 public override string ToString() => $"{Numerator}/{Denominator}";
354}

442 static int? ClosestDistance(List<int> positions, List<int> others, bool sameStem)
443 {
444 IEnumerable<int> gaps;
445 if (sameStem)
446 {
447 gaps = positions.Zip(positions.Skip(1), (earlier, later) => later - earlier);
448 }
449 else
450 {
451 var merged = positions
452 .Select(position => (Position: position, Kind: 0))
453 .Concat(others.Select(other => (Position: other, Kind: 1)))
454 .OrderBy(entry => entry.Position)
455 .ToList();
456 gaps = merged
457 .Zip(merged.Skip(1))
458 .Where(pair => pair.First.Kind != pair.Second.Kind)
459 .Select(pair => pair.Second.Position - pair.First.Position);
460 }
461 return gaps.Select(gap => (int?)gap).Min();
462 }

464 static Lexeme Classify(string text) => text switch
465 {
466 "AND" => new(LexemeKind.And, text),
467 "OR" => new(LexemeKind.Or, text),
468 "NOT" => new(LexemeKind.Not, text),
469 "(" => new(LexemeKind.Open, text),
470 ")" => new(LexemeKind.Close, text),
471 _ when Patterns.Word.IsMatch(text) => new(LexemeKind.Term, text.ToLowerInvariant()),
472 _ when Patterns.PrefixTerm.IsMatch(text) =>
473 new(LexemeKind.Prefix, text[..^1].ToLowerInvariant()),
474 _ => throw new CommandException($"BAD TERM {text}"),
475 };

477 bool IsIndexed(Token token) => !stopwords.Contains(token.Form);

479 IEnumerable<Token> IndexedTokens(Document document) => document.Tokens.Where(IsIndexed);

481 Dictionary<string, int> TermFrequencies(Document document) =>
482 IndexedTokens(document).CountBy(token => Porter.Stem(token.Form)).ToDictionary();

484 HashSet<string> StemSet(Document document) =>

493 .SelectMany(document => IndexedTokens(document).Select(token => (document, token)))

495 .ToDictionary(group => group.Key, group => group.ToList());

582 void Search(string[] arguments)
583 {
584 var items = Patterns.Lexeme.Matches(arguments[0]).Select(match => Classify(match.Value))
585 .ToList();
586 var stemSets = documents.Values.ToDictionary(document => document.Number, StemSet);
587 var (canonical, matches) = new QueryParser(items, stemSets).Parse();
588 var stopword = items.FirstOrDefault(item =>
589 item.Kind == LexemeKind.Term && stopwords.Contains(item.Spelling));
590 if (stopword is not null)
591 {
592 throw new CommandException($"STOPWORD {stopword.Spelling}");
593 }
594 Emit(["SEARCH", canonical, $"count={matches.Count}", .. matches.Order()]);
595 }
596
597 void Phrase(string[] arguments)
598 {
599 var words = arguments.Select(WordArgument).ToList();
600 if (words.All(stopwords.Contains))
601 {
602 throw new CommandException("STOPWORDS ONLY");
603 }
604 var targets = words.Select(word => stopwords.Contains(word) ? "?" : Porter.Stem(word))
605 .ToList();
606 var matches = new List<string>();
607 foreach (var document in documents.Values)
608 {
609 var tokens = document.Tokens;
610 for (var start = 0; start <= tokens.Count - targets.Count; start++)
611 {
612 if (Enumerable.Range(0, targets.Count)
613 .All(offset => TokenMatches(tokens[start + offset], targets[offset])))
614 {
615 matches.Add(document.Reference(tokens[start]));
616 }
617 }
618 }
619 Emit([$"PHRASE \"{string.Join(" ", targets)}\"", $"matches={matches.Count}", .. matches]);
620 }

622 void Near(string[] arguments)
623 {
624 var first = TermArgument(arguments[0]);
625 var second = TermArgument(arguments[1]);
626 var limit = NumberArgument(arguments[2], 1, MaxCount);
627 var found = new List<string>();
628 foreach (var document in documents.Values)
629 {
630 var distance = ClosestDistance(
631 StemPositions(document, first), StemPositions(document, second), first == second);
632 if (distance is int value && value <= limit)
633 {
634 found.Add($"{document.Number}:{value}");
635 }
636 }
637 Emit(["NEAR", first, second, limit, $"count={found.Count}", .. found]);
638 }

641 {
642 var limit = NumberArgument(arguments[0], 1, MaxCount);
643 Dictionary<string, int> counts;

645 if (arguments.Length == 2)
646 {
647 var document = DocumentArgument(arguments[1]);

660 void Kwic(string[] arguments)
661 {
662 var stem = TermArgument(arguments[0]);
663 var found = Occurrences(stem);
664 Emit("KWIC", stem, $"count={found.Count}");
665 foreach (var (document, token) in found)
666 {
667 var text = document.Lines[token.Line - 1];
668 var before = text[Math.Max(0, token.Start - KwicWidth)..token.Start].PadLeft(KwicWidth);
669 var after = text[token.End..Math.Min(text.Length, token.End + KwicWidth)]
670 .PadRight(KwicWidth);
671 Emit($"|{before}[{token.Text}]{after}|", document.Reference(token));
672 }
673 }

702 void Similar(string[] arguments)
703 {
704 var document = DocumentArgument(arguments[0]);
705 var limit = NumberArgument(arguments[1], 1, MaxCount);
706 var stems = StemSet(document);
707 var scores = new List<(Fraction Score, int Number)>();
708 foreach (var other in documents.Values)
709 {
710 if (other == document)
711 {
712 continue;
713 }
714 var otherStems = StemSet(other);
715 var union = stems.Union(otherStems).Count();
716 var score = union > 0
717 ? Fraction.Create(stems.Intersect(otherStems).Count(), union)
718 : new Fraction(0, 1);
719 scores.Add((score, other.Number));
720 }
721 var ranked = scores
722 .OrderByDescending(entry => entry.Score)
723 .ThenBy(entry => entry.Number)
724 .Take(limit);
725 Emit(
726 [
727 "SIMILAR", document.Number,
728 .. ranked.Select(entry => $"{entry.Number}:{entry.Score}"),
729 ]);
730 }

732 void Rank(string[] arguments)
733 {
734 var limit = NumberArgument(arguments[0], 1, MaxCount);
735 var query = arguments.Skip(1).Select(TermArgument).ToHashSet();
736 var ranked = new List<(int Matched, int Total, int Number)>();
737 foreach (var document in documents.Values)
738 {
739 var counts = TermFrequencies(document);
740 var matched = query.Count(counts.ContainsKey);
741 var total = query.Sum(stem => counts.GetValueOrDefault(stem));
742 if (matched > 0)
743 {
744 ranked.Add((matched, total, document.Number));
745 }
746 }
747 var selected = ranked
748 .OrderByDescending(entry => entry.Matched)
749 .ThenByDescending(entry => entry.Total)
750 .ThenBy(entry => entry.Number)
751 .Take(limit);
752 Emit(
753 [
754 $"RANK count={ranked.Count}",
755 .. selected.Select(entry => $"{entry.Number}:{entry.Matched}:{entry.Total}"),
756 ]);
757 }

841 public void Execute(string line)
842 {
843 var separator = line.IndexOf(' ');
844 var name = separator < 0 ? line : line[..separator];
845 try
846 {
847 if (!handlers.TryGetValue(name, out var command))
848 {
849 throw new CommandException($"UNKNOWN COMMAND {name}");
850 }
851 string[] arguments = separator < 0 ? [] : SplitArguments(name, line[(separator + 1)..]);
852 if (arguments.Length < command.Minimum || arguments.Length > command.Maximum)
853 {
854 throw new CommandException($"BAD ARGUMENTS {name}");
855 }
856 command.Handler(arguments);
857 }
858 catch (CommandException error)
859 {
860 Emit("ERROR", error.Message);
861 }
862 }
