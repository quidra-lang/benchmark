51fn isVowel(word: []const u8, index: usize) bool {
52 return switch (word[index]) {
53 'a', 'e', 'i', 'o', 'u' => true,
54 'y' => index > 0 and !isVowel(word, index - 1),
55 else => false,
56 };

60 const kinds = try gpa.alloc(u8, word.len);

65fn measure(word: []const u8) usize {
66 var count: usize = 0;
67 var index: usize = 1;
68 while (index < word.len) : (index += 1) {
69 if (isVowel(word, index - 1) and !isVowel(word, index)) count += 1;
70 }
71 return count;
72}

74fn hasVowel(word: []const u8) bool {
75 for (0..word.len) |index| {
76 if (isVowel(word, index)) return true;
77 }
78 return false;
79}

81fn endsDouble(word: []const u8) bool {
82 const size = word.len;
83 return size >= 2 and word[size - 1] == word[size - 2] and
84 !isVowel(word, size - 1) and !isVowel(word, size - 2);
85}

87fn endsCvc(word: []const u8) bool {
88 const size = word.len;
89 return size >= 3 and !isVowel(word, size - 3) and isVowel(word, size - 2) and
90 !isVowel(word, size - 1) and std.mem.findScalar(u8, "wxy", word[size - 1]) == null;
91}

117fn rulesWith(
118 comptime condition: Condition,
119 comptime replacements: []const [2][]const u8,
120) [replacements.len]Rule {
121 var rules: [replacements.len]Rule = undefined;
122 for (replacements, &rules) |pair, *rule| {
123 rule.* = .{ .suffix = pair[0], .replacement = pair[1], .condition = condition };
124 }

128fn step4Rules() [step_4_suffixes.len]Rule {
129 var rules: [step_4_suffixes.len]Rule = undefined;
130 for (step_4_suffixes, &rules) |suffix, *rule| {
131 const condition = if (equal(suffix, "ion")) &ionCondition else &measureAboveOne;
132 rule.* = .{ .suffix = suffix, .replacement = "", .condition = condition };
133 }
134 return rules;
135}

164const RuleOutcome = struct { word: []const u8, suffix: ?[]const u8 };
165
166fn applyRules(gpa: Allocator, word: []const u8, rules: []const Rule) Allocator.Error!RuleOutcome {
167 var selected: ?Rule = null;
168 for (rules) |rule| {
169 const longer = selected == null or rule.suffix.len > selected.?.suffix.len;
170 if (std.mem.endsWith(u8, word, rule.suffix) and longer) selected = rule;
171 }
172 const rule = selected orelse return .{ .word = word, .suffix = null };
173 const remainder = word[0 .. word.len - rule.suffix.len];
174 if (!rule.condition(remainder)) return .{ .word = word, .suffix = null };
175 return .{
176 .word = try std.mem.concat(gpa, u8, &.{ remainder, rule.replacement }),
177 .suffix = rule.suffix,
178 };
179}

181fn tidyStep1b(gpa: Allocator, word: []const u8) Allocator.Error![]const u8 {
182 for ([_][]const u8{ "at", "bl", "iz" }) |suffix| {
183 if (std.mem.endsWith(u8, word, suffix)) return std.mem.concat(gpa, u8, &.{ word, "e" });
184 }
185 if (endsDouble(word) and std.mem.findScalar(u8, "lsz", word[word.len - 1]) == null) {
186 return word[0 .. word.len - 1];
187 }
188 if (measure(word) == 1 and endsCvc(word)) return std.mem.concat(gpa, u8, &.{ word, "e" });
189 return word;
190}

192fn applyStep(gpa: Allocator, name: []const u8, word: []const u8) Allocator.Error![]const u8 {
193 if (equal(name, "5b")) {
194 if (measure(word) > 1 and endsDouble(word) and std.mem.endsWith(u8, word, "l")) {
195 return word[0 .. word.len - 1];
196 }
197 return word;
198 }
199 const outcome = try applyRules(gpa, word, rule_tables.get(name).?);
200 if (equal(name, "1b")) {
201 if (outcome.suffix) |suffix| {
202 if (equal(suffix, "ed") or equal(suffix, "ing")) return tidyStep1b(gpa, outcome.word);
203 }
204 }
205 return outcome.word;
206}

208const StepChange = struct { step: []const u8, word: []const u8 };
209const Stemming = struct { stem: []const u8, changes: []const StepChange };

211fn stemmingSteps(gpa: Allocator, word: []const u8) Allocator.Error!Stemming {
212 var changes: std.ArrayList(StepChange) = .empty;
213 var current = word;
214 if (word.len > 2) {
215 for (step_names) |name| {
216 const result = try applyStep(gpa, name, current);
217 if (!equal(result, current)) try changes.append(gpa, .{ .step = name, .word = result });
218 current = result;
219 }
220 }
221 return .{ .stem = current, .changes = changes.items };
222}

262fn closestDistance(
263 gpa: Allocator,
264 positions: []const usize,
265 others: []const usize,
266 same_stem: bool,
267) Allocator.Error!?usize {
268 var gaps: std.ArrayList(usize) = .empty;
269 if (same_stem) {
270 var index: usize = 1;
271 while (index < positions.len) : (index += 1) {
272 try gaps.append(gpa, positions[index] - positions[index - 1]);
273 }
274 } else {
275 const Marker = struct {
276 position: usize,
277 source: u1,
278
279 fn before(_: void, left: @This(), right: @This()) bool {
280 return left.position < right.position;
281 }
282 };
283 var merged: std.ArrayList(Marker) = .empty;
284 for (positions) |position| try merged.append(gpa, .{ .position = position, .source = 0 });
285 for (others) |other| try merged.append(gpa, .{ .position = other, .source = 1 });
286 std.mem.sort(Marker, merged.items, {}, Marker.before);
287 var index: usize = 1;
288 while (index < merged.items.len) : (index += 1) {
289 const earlier = merged.items[index - 1];
290 const later = merged.items[index];
291 if (earlier.source != later.source) try gaps.append(gpa, later.position - earlier.position);
292 }
293 }
294 if (gaps.items.len == 0) return null;
295 return std.mem.min(usize, gaps.items);
296}

298const Similarity = struct {
299 number: usize,
300 numerator: usize,
301 denominator: usize,
302
303 fn jaccard(number: usize, common: usize, total: usize) Similarity {
304 if (total == 0) return .{ .number = number, .numerator = 0, .denominator = 1 };
305 const divisor = std.math.gcd(common, total);
306 return .{ .number = number, .numerator = common / divisor, .denominator = total / divisor };
307 }
308
309 fn before(_: void, left: Similarity, right: Similarity) bool {
310 const left_score = left.numerator * right.denominator;
311 const right_score = right.numerator * left.denominator;
312 if (left_score != right_score) return left_score > right_score;
313 return left.number < right.number;
314 }
315};

348 try document.lines.append(gpa, text);
349 const first = document.tokens.items.len;
350 var index: usize = 0;
351 while (index < text.len) {
352 if (!std.ascii.isAlphabetic(text[index])) {
353 index += 1;
354 continue;
355 }
356 const start = index;
357 while (index < text.len and std.ascii.isAlphabetic(text[index])) index += 1;
358 const form = try std.ascii.allocLowerString(gpa, text[start..index]);
359 try document.tokens.append(gpa, .{
360 .text = text[start..index],
361 .form = form,
362 .stem = try porterStem(gpa, form),
363 .line = document.lines.items.len,
364 .position = document.tokens.items.len + 1,
365 .start = start,
366 });
367 }
368 return document.tokens.items[first..];
369 }

412fn lexemes(gpa: Allocator, expression: []const u8) Allocator.Error![]const []const u8 {
413 var result: std.ArrayList([]const u8) = .empty;
414 var index: usize = 0;
415 while (index < expression.len) {

417 ' ' => index += 1,
418 '(', ')' => {

426 try result.append(gpa, expression[start..index]);
427 },
428 }
429 }

441const QueryParser = struct {
442 engine: *Engine,
443 items: []const QueryItem,
444 stem_sets: []const WordSet,
445 position: usize = 0,
446
447 fn peek(parser: *const QueryParser) ?Kind {
448 if (parser.position < parser.items.len) return parser.items[parser.position].kind;
449 return null;
450 }
451
452 fn take(parser: *QueryParser) Error!QueryItem {
453 if (parser.position >= parser.items.len) return parser.engine.fail("SYNTAX", .{});
454 parser.position += 1;
455 return parser.items[parser.position - 1];
456 }
457
458 fn parse(parser: *QueryParser) Error!QueryResult {
459 const result = try parser.expression();
460 if (parser.position < parser.items.len) return parser.engine.fail("SYNTAX", .{});
461 return result;
462 }
463
464 fn expression(parser: *QueryParser) Error!QueryResult {
465 var result = try parser.conjunction();
466 while (parser.peek() == Kind.or_operator) {
467 parser.position += 1;
468 const right = try parser.conjunction();
469 for (result.documents, right.documents) |*selected, other| selected.* = selected.* or other;
470 result.canonical = try std.fmt.allocPrint(
471 parser.engine.scratch,
472 "({s} OR {s})",
473 .{ result.canonical, right.canonical },
474 );
475 }
476 return result;
477 }
478
479 fn conjunction(parser: *QueryParser) Error!QueryResult {
480 var result = try parser.negation();
481 while (parser.peek()) |kind| {
482 switch (kind) {
483 .and_operator, .not_operator, .open, .term, .prefix => {},
484 .or_operator, .close => break,
485 }
486 if (kind == .and_operator) parser.position += 1;
487 const right = try parser.negation();
488 for (result.documents, right.documents) |*selected, other| selected.* = selected.* and other;
489 result.canonical = try std.fmt.allocPrint(
490 parser.engine.scratch,
491 "({s} AND {s})",
492 .{ result.canonical, right.canonical },
493 );
494 }
495 return result;
496 }
497
498 fn negation(parser: *QueryParser) Error!QueryResult {
499 if (parser.peek() != Kind.not_operator) return parser.element();
500 parser.position += 1;
501 const inner = try parser.negation();
502 for (inner.documents) |*selected| selected.* = !selected.*;
503 return .{
504 .canonical = try std.fmt.allocPrint(parser.engine.scratch, "(NOT {s})", .{inner.canonical}),
505 .documents = inner.documents,
506 };
507 }
508
509 fn element(parser: *QueryParser) Error!QueryResult {
510 const item = try parser.take();
511 const scratch = parser.engine.scratch;
512 switch (item.kind) {
513 .open => {
514 const result = try parser.expression();
515 if ((try parser.take()).kind != .close) return parser.engine.fail("SYNTAX", .{});
516 return result;
517 },
518 .term => {
519 const stem = try porterStem(scratch, item.spelling);
520 const documents = try scratch.alloc(bool, parser.stem_sets.len);
521 for (parser.stem_sets, documents) |stems, *selected| selected.* = stems.contains(stem);
522 return .{ .canonical = stem, .documents = documents };
523 },
524 .prefix => {
525 const documents = try scratch.alloc(bool, parser.stem_sets.len);
526 for (parser.stem_sets, documents) |stems, *selected| {
527 selected.* = hasPrefix(stems, item.spelling);
528 }
529 return .{
530 .canonical = try std.fmt.allocPrint(scratch, "{s}*", .{item.spelling}),
531 .documents = documents,
532 };
533 },
534 .and_operator, .or_operator, .not_operator, .close => {
535 return parser.engine.fail("SYNTAX", .{});
536 },
537 }
538 }
539};

579fn splitArguments(gpa: Allocator, command: Command, rest: []const u8) Allocator.Error![]const []const u8 {
580 switch (command) {
581 .SEARCH => return gpa.dupe([]const u8, &.{rest}),
582 .APPEND => {

587 var parts: std.ArrayList([]const u8) = .empty;
588 var pieces = std.mem.splitScalar(u8, rest, ' ');

636 for (document.tokens.items) |token| {
637 if (engine.isIndexed(token)) try increment(engine.scratch, &counts, token.stem, 1);
638 }
639 return counts;
640 }

642 fn stemSet(engine: *const Engine, document: *const Document) Allocator.Error!WordSet {
643 var stems: WordSet = .empty;
644 for (document.tokens.items) |token| {
645 if (engine.isIndexed(token)) try stems.put(engine.scratch, token.stem, {});
646 }
647 return stems;
648 }
649
650 fn totalFrequencies(engine: *const Engine) Allocator.Error!Counts {
651 var totals: Counts = .empty;
652 for (engine.documents.items) |document| {
653 const counts = try engine.termFrequencies(document);
654 var entries = counts.iterator();
655 while (entries.next()) |entry| {
656 try increment(engine.scratch, &totals, entry.key_ptr.*, entry.value_ptr.*);
657 }
658 }
659 return totals;
660 }

662 fn occurrenceMap(engine: *const Engine) Allocator.Error!OccurrenceMap {
663 var occurrences: OccurrenceMap = .empty;
664 for (engine.documents.items) |document| {
665 for (document.tokens.items) |token| {
666 if (!engine.isIndexed(token)) continue;
667 const entry = try occurrences.getOrPut(engine.scratch, token.stem);
668 if (!entry.found_existing) entry.value_ptr.* = .empty;
669 try entry.value_ptr.append(engine.scratch, .{ .document = document, .token = token });
670 }
671 }
672 return occurrences;
673 }

676 const occurrences = try engine.occurrenceMap();

678 }

790 fn search(engine: *Engine, arguments: []const []const u8) Error!void {
791 var items: std.ArrayList(QueryItem) = .empty;
792 for (try lexemes(engine.scratch, arguments[0])) |lexeme| {
793 if (symbols.get(lexeme)) |kind| {
794 try items.append(engine.scratch, .{ .kind = kind, .spelling = lexeme });
795 } else if (isWord(lexeme)) {
796 const spelling = try std.ascii.allocLowerString(engine.scratch, lexeme);
797 try items.append(engine.scratch, .{ .kind = .term, .spelling = spelling });
798 } else if (std.mem.endsWith(u8, lexeme, "*") and isWord(lexeme[0 .. lexeme.len - 1])) {
799 const spelling = try std.ascii.allocLowerString(engine.scratch, lexeme[0 .. lexeme.len - 1]);
800 try items.append(engine.scratch, .{ .kind = .prefix, .spelling = spelling });
801 } else {
802 return engine.fail("BAD TERM {s}", .{lexeme});
803 }
804 }
805 const stem_sets = try engine.scratch.alloc(WordSet, engine.documents.items.len);
806 for (engine.documents.items, stem_sets) |document, *stems| stems.* = try engine.stemSet(document);
807 var parser: QueryParser = .{ .engine = engine, .items = items.items, .stem_sets = stem_sets };
808 const result = try parser.parse();
809 for (items.items) |item| {
810 if (item.kind == .term and engine.isStopword(item.spelling)) {
811 return engine.fail("STOPWORD {s}", .{item.spelling});
812 }
813 }
814 const count = std.mem.count(bool, result.documents, &.{true});
815 try engine.out.print("SEARCH {s} count={d}", .{ result.canonical, count });
816 for (engine.documents.items, result.documents) |document, selected| {
817 if (selected) try engine.out.print(" {d}", .{document.number});
818 }
819 try engine.out.writeByte('\n');
820 }

822 fn phrase(engine: *Engine, arguments: []const []const u8) Error!void {
823 const words = try engine.scratch.alloc([]const u8, arguments.len);
824 for (arguments, words) |argument, *word| word.* = try engine.wordArgument(argument);
825 var only_stopwords = true;
826 for (words) |word| {
827 if (!engine.isStopword(word)) only_stopwords = false;
828 }
829 if (only_stopwords) return engine.fail("STOPWORDS ONLY", .{});
830 const targets = try engine.scratch.alloc([]const u8, words.len);
831 for (words, targets) |word, *target| {
832 target.* = if (engine.isStopword(word)) "?" else try porterStem(engine.scratch, word);
833 }
834 var matches: std.ArrayList(Occurrence) = .empty;
835 for (engine.documents.items) |document| {
836 const tokens = document.tokens.items;
837 if (tokens.len < targets.len) continue;
838 for (0..tokens.len - targets.len + 1) |start| {
839 if (engine.phraseMatches(tokens[start..][0..targets.len], targets)) {
840 try matches.append(engine.scratch, .{ .document = document, .token = tokens[start] });
841 }
842 }
843 }
844 const phrase_text = try std.mem.join(engine.scratch, " ", targets);
845 try engine.out.print("PHRASE \"{s}\" matches={d}", .{ phrase_text, matches.items.len });
846 for (matches.items) |occurrence| try engine.out.print(" {f}", .{occurrence});
847 try engine.out.writeByte('\n');
848 }

850 fn near(engine: *Engine, arguments: []const []const u8) Error!void {
851 const first = try engine.termArgument(arguments[0]);
852 const second = try engine.termArgument(arguments[1]);
853 const limit = try engine.numberArgument(arguments[2], 1, max_count);
854 const Found = struct { number: usize, distance: usize };
855 var found: std.ArrayList(Found) = .empty;
856 for (engine.documents.items) |document| {
857 const positions = try engine.stemPositions(document, first);
858 const others = try engine.stemPositions(document, second);
859 const distance = try closestDistance(engine.scratch, positions, others, equal(first, second));
860 if (distance != null and distance.? <= limit) {
861 try found.append(engine.scratch, .{ .number = document.number, .distance = distance.? });
862 }
863 }
864 try engine.out.print("NEAR {s} {s} {d} count={d}", .{ first, second, limit, found.items.len });
865 for (found.items) |entry| try engine.out.print(" {d}:{d}", .{ entry.number, entry.distance });
866 try engine.out.writeByte('\n');
867 }

870 const limit = try engine.numberArgument(arguments[0], 1, max_count);
871 const document = if (arguments.len == 2) try engine.documentArgument(arguments[1]) else null;
872 const counts = if (document) |chosen|
873 try engine.termFrequencies(chosen)

875 try engine.totalFrequencies();

879 for (ranked[0..@min(limit, ranked.len)]) |entry| {
880 try engine.out.print(" {s}:{d}", .{ entry.stem, entry.count });
881 }

886 const stem = try engine.termArgument(arguments[0]);
887 const found = try engine.occurrencesOf(stem);

889 for (found) |occurrence| {
890 const token = occurrence.token;
891 const text = occurrence.document.lines.items[token.line - 1];
892 const before = text[token.start -| kwic_width..token.start];

894 try engine.out.writeByte('|');
895 try engine.out.splatByteAll(' ', kwic_width - before.len);
896 try engine.out.print("{s}[{s}]{s}", .{ before, token.text, after });
897 try engine.out.splatByteAll(' ', kwic_width - after.len);
898 try printLine(engine.out, "| {f}", .{occurrence});
899 }

928 fn similar(engine: *Engine, arguments: []const []const u8) Error!void {
929 const document = try engine.documentArgument(arguments[0]);
930 const limit = try engine.numberArgument(arguments[1], 1, max_count);
931 const stems = try engine.stemSet(document);
932 var scores: std.ArrayList(Similarity) = .empty;
933 for (engine.documents.items) |other| {
934 if (other == document) continue;
935 const other_stems = try engine.stemSet(other);
936 var common: usize = 0;
937 var keys = stems.keyIterator();
938 while (keys.next()) |stem| {
939 if (other_stems.contains(stem.*)) common += 1;
940 }
941 const union_size = stems.count() + other_stems.count() - common;
942 try scores.append(engine.scratch, Similarity.jaccard(other.number, common, union_size));
943 }
944 std.mem.sort(Similarity, scores.items, {}, Similarity.before);
945 try engine.out.print("SIMILAR {d}", .{document.number});
946 for (scores.items[0..@min(limit, scores.items.len)]) |score| {
947 try engine.out.print(" {d}:{d}/{d}", .{ score.number, score.numerator, score.denominator });
948 }
949 try engine.out.writeByte('\n');
950 }

952 fn rank(engine: *Engine, arguments: []const []const u8) Error!void {
953 const limit = try engine.numberArgument(arguments[0], 1, max_count);
954 var query: WordSet = .empty;
955 for (arguments[1..]) |argument| try query.put(engine.scratch, try engine.termArgument(argument), {});
956 var ranked: std.ArrayList(Ranking) = .empty;
957 for (engine.documents.items) |document| {
958 const counts = try engine.termFrequencies(document);
959 var matched: usize = 0;
960 var total: usize = 0;
961 var stems = query.keyIterator();
962 while (stems.next()) |stem| {
963 const count = counts.get(stem.*) orelse 0;
964 if (count > 0) matched += 1;
965 total += count;
966 }
967 if (matched > 0) {
968 try ranked.append(engine.scratch, .{
969 .number = document.number,
970 .matched = matched,
971 .total = total,
972 });
973 }
974 }
975 std.mem.sort(Ranking, ranked.items, {}, Ranking.before);
976 try engine.out.print("RANK count={d}", .{ranked.items.len});
977 for (ranked.items[0..@min(limit, ranked.items.len)]) |entry| {
978 try engine.out.print(" {d}:{d}:{d}", .{ entry.number, entry.matched, entry.total });
979 }
980 try engine.out.writeByte('\n');
981 }

1067 fn execute(engine: *Engine, line: []const u8) Error!void {
1068 const space = std.mem.findScalar(u8, line, ' ');
1069 const name = line[0 .. space orelse line.len];
1070 const command = std.meta.stringToEnum(Command, name) orelse
1071 return engine.fail("UNKNOWN COMMAND {s}", .{name});
1072 const arguments: []const []const u8 = if (space) |index|
1073 try splitArguments(engine.scratch, command, line[index + 1 ..])
1074 else
1075 &.{};
1076 const minimum, const maximum = command.arity();
1077 if (arguments.len < minimum or arguments.len > maximum) {
1078 return engine.fail("BAD ARGUMENTS {s}", .{name});
1079 }
1080 switch (command) {
1081 .STEM => try engine.stemWords(arguments),
1082 .STEPS => try engine.showSteps(arguments),
1083 .MEASURE => try engine.showMeasure(arguments),
1084 .INDEX => try engine.showIndex(arguments),
1085 .SEARCH => try engine.search(arguments),
1086 .PHRASE => try engine.phrase(arguments),
1087 .NEAR => try engine.near(arguments),
1088 .TOP => try engine.top(arguments),
1089 .KWIC => try engine.kwic(arguments),
1090 .TF => try engine.termFrequency(arguments),
1091 .DF => try engine.documentFrequency(arguments),
1092 .VARIANTS => try engine.variants(arguments),
1093 .SIMILAR => try engine.similar(arguments),
1094 .RANK => try engine.rank(arguments),
1095 .INFO => try engine.info(arguments),
1096 .LINE => try engine.showLine(arguments),
1097 .STATS => try engine.stats(arguments),
1098 .STOP => try engine.stop(arguments),
1099 .UNSTOP => try engine.unstop(arguments),
1100 .APPEND => try engine.append(arguments),
1101 .DELETE => try engine.delete(arguments),
1102 }
1103 }

1106pub fn main(init: std.process.Init) !void {
1107 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1108 defer storage.deinit();
1109 const gpa = storage.allocator();
1110 var input_buffer: [4096]u8 = undefined;
1111 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
1112 var output_buffer: [4096]u8 = undefined;
1113 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
1114 const text = try input.interface.allocRemaining(gpa, .unlimited);
1115 var lines: std.ArrayList([]const u8) = .empty;
1116 var pieces = std.mem.splitScalar(u8, text, '\n');
1117 while (pieces.next()) |piece| try lines.append(gpa, piece);
1118
1119 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
1120 defer scratch.deinit();
1121 var engine: Engine = .{ .gpa = gpa, .scratch = scratch.allocator(), .out = &output.interface };
1122 var stopwords = std.mem.tokenizeScalar(u8, lines.items[0], ' ');
1123 _ = stopwords.next();
1124 while (stopwords.next()) |word| try engine.stopwords.put(gpa, word, {});
1125 var index: usize = 1;
1126 while (std.mem.startsWith(u8, lines.items[index], "DOC ")) : (index += 1) {
1127 const document = try engine.addDocument(try std.fmt.parseInt(usize, lines.items[index][4..], 10));
1128 index += 1;
1129 while (!equal(lines.items[index], "ENDDOC")) : (index += 1) {
1130 _ = try document.addLine(gpa, lines.items[index]);
1131 }
1132 }
1133 for (lines.items[index..]) |line| {
1134 if (equal(line, "END")) break;
1135 _ = scratch.reset(.retain_capacity);
1136 engine.execute(line) catch |err| {
1137 if (err != error.Invalid) return err;
1138 try printLine(&output.interface, "ERROR {s}", .{engine.problem});
1139 };
1140 }
1141 try output.interface.flush();
1142}
