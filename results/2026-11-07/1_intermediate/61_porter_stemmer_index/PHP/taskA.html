23function letterPattern(string $word): string
24{
25 $kinds = '';
26 for ($index = 0; $index < strlen($word); $index++) {
27 $letter = $word[$index];
28 $consonantBefore = $index > 0 && $kinds[$index - 1] === 'C';
29 $isVowel = str_contains(VOWELS, $letter) || ($letter === 'y' && $consonantBefore);
30 $kinds .= $isVowel ? 'V' : 'C';
31 }
32 return $kinds;
33}

91function ruleTables(): array
92{
93 static $tables = null;
94 $tables ??= [
95 '1a' => rulesWith(always(...), ['sses' => 'ss', 'ies' => 'i', 'ss' => 'ss', 's' => '']),
96 '1b' => [
97 ['eed', 'ee', measureAboveZero(...)],
98 ['ed', '', hasVowel(...)],
99 ['ing', '', hasVowel(...)],
100 ],
101 '1c' => [['y', 'i', hasVowel(...)]],
102 '2' => rulesWith(measureAboveZero(...), [
103 'ational' => 'ate', 'tional' => 'tion', 'enci' => 'ence', 'anci' => 'ance',
104 'izer' => 'ize', 'abli' => 'able', 'alli' => 'al', 'entli' => 'ent', 'eli' => 'e',
105 'ousli' => 'ous', 'ization' => 'ize', 'ation' => 'ate', 'ator' => 'ate',
106 'alism' => 'al', 'iveness' => 'ive', 'fulness' => 'ful', 'ousness' => 'ous',
107 'aliti' => 'al', 'iviti' => 'ive', 'biliti' => 'ble',
108 ]),
109 '3' => rulesWith(measureAboveZero(...), [
110 'icate' => 'ic', 'ative' => '', 'alize' => 'al', 'iciti' => 'ic',
111 'ical' => 'ic', 'ful' => '', 'ness' => '',
112 ]),
113 '4' => array_map(
114 fn(string $suffix): array => [
115 $suffix, '', $suffix === 'ion' ? ionCondition(...) : measureAboveOne(...),
116 ],
117 STEP_4_SUFFIXES,
118 ),
119 '5a' => [['e', '', finalECondition(...)]],
120 ];
121 return $tables;
122}

124function applyRules(string $word, array $rules): array
125{
126 $matching = array_filter($rules, fn(array $rule): bool => str_ends_with($word, $rule[0]));
127 if ($matching === []) {
128 return [$word, null];
129 }
130 usort(
131 $matching,
132 fn(array $first, array $second): int => strlen($second[0]) <=> strlen($first[0]),
133 );
134 [$suffix, $replacement, $condition] = $matching[0];
135 $remainder = substr($word, 0, -strlen($suffix));
136 if (!$condition($remainder)) {
137 return [$word, null];
138 }
139 return [$remainder . $replacement, $suffix];
140}

142function tidyStep1b(string $word): string
143{
144 if (str_ends_with($word, 'at') || str_ends_with($word, 'bl') || str_ends_with($word, 'iz')) {
145 return $word . 'e';
146 }
147 if (endsDouble($word) && !str_contains('lsz', $word[-1])) {
148 return substr($word, 0, -1);
149 }
150 if (measure($word) === 1 && endsCvc($word)) {
151 return $word . 'e';
152 }
153 return $word;
154}

156function applyStep(string $name, string $word): string
157{
158 if ($name === '5b') {
159 if (measure($word) > 1 && endsDouble($word) && str_ends_with($word, 'l')) {
160 return substr($word, 0, -1);
161 }
162 return $word;
163 }
164 [$result, $suffix] = applyRules($word, ruleTables()[$name]);
165 if ($name === '1b' && ($suffix === 'ed' || $suffix === 'ing')) {
166 return tidyStep1b($result);
167 }
168 return $result;
169}

171function stemmingSteps(string $word): array
172{
173 $changes = [];
174 if (strlen($word) > 2) {
175 foreach (STEP_NAMES as $name) {
176 $result = applyStep($name, $word);
177 if ($result !== $word) {
178 $changes[] = "$name:$result";
179 }
180 $word = $result;
181 }
182 }
183 return [$word, $changes];
184}

247function closestDistance(array $positions, array $others, bool $sameStem): ?int
248{
249 $gaps = [];
250 if ($sameStem) {
251 for ($index = 1; $index < count($positions); $index++) {
252 $gaps[] = $positions[$index] - $positions[$index - 1];
253 }
254 } else {
255 $merged = array_merge(
256 array_map(fn(int $position): array => [$position, 0], $positions),
257 array_map(fn(int $other): array => [$other, 1], $others),
258 );
259 sort($merged);
260 for ($index = 1; $index < count($merged); $index++) {
261 if ($merged[$index][1] !== $merged[$index - 1][1]) {
262 $gaps[] = $merged[$index][0] - $merged[$index - 1][0];
263 }
264 }
265 }
266 return $gaps === [] ? null : min($gaps);
267}

269function reducedFraction(int $numerator, int $denominator): array
270{
271 [$a, $b] = [$numerator, $denominator];
272 while ($b !== 0) {
273 [$a, $b] = [$b, $a % $b];
274 }
275 return [intdiv($numerator, $a), intdiv($denominator, $a)];
276}

278function classify(string $lexeme): array
279{
280 if (in_array($lexeme, OPERATORS, true) || in_array($lexeme, PARENTHESES, true)) {
281 return [$lexeme, $lexeme];
282 }
283 if (isWord($lexeme)) {
284 return ['term', strtolower($lexeme)];
285 }
286 $letters = substr($lexeme, 0, -1);
287 if (str_ends_with($lexeme, '*') && isWord($letters)) {
288 return ['prefix', strtolower($letters)];
289 }
290 throw new CommandError("BAD TERM $lexeme");
291}

347final class QueryParser
348{
349 private int $position = 0;
350
351 public function __construct(private readonly array $items, private readonly array $stemSets)
352 {
353 }
354
355 private function peek(): ?string
356 {
357 return $this->items[$this->position][0] ?? null;
358 }
359
360 private function take(): array
361 {
362 if ($this->position >= count($this->items)) {
363 throw new CommandError('SYNTAX');
364 }
365 return $this->items[$this->position++];
366 }
367
368 public function parse(): array
369 {
370 $result = $this->expression();
371 if ($this->position < count($this->items)) {
372 throw new CommandError('SYNTAX');
373 }
374 return $result;
375 }
376
377 private function expression(): array
378 {
379 [$canonical, $documents] = $this->conjunction();
380 while ($this->peek() === 'OR') {
381 $this->position++;
382 [$right, $rightDocuments] = $this->conjunction();
383 $canonical = "($canonical OR $right)";
384 $documents += $rightDocuments;
385 }
386 return [$canonical, $documents];
387 }
388
389 private function conjunction(): array
390 {
391 [$canonical, $documents] = $this->negation();
392 while (in_array($this->peek(), CONJUNCTION_CONTINUATIONS, true)) {
393 if ($this->peek() === 'AND') {
394 $this->position++;
395 }
396 [$right, $rightDocuments] = $this->negation();
397 $canonical = "($canonical AND $right)";
398 $documents = array_intersect_key($documents, $rightDocuments);
399 }
400 return [$canonical, $documents];
401 }
402
403 private function negation(): array
404 {
405 if ($this->peek() === 'NOT') {
406 $this->position++;
407 [$canonical, $documents] = $this->negation();
408 return ["(NOT $canonical)", array_diff_key($this->stemSets, $documents)];
409 }
410 return $this->element();
411 }
412
413 private function element(): array
414 {
415 [$kind, $spelling] = $this->take();
416 if ($kind === '(') {
417 $result = $this->expression();
418 if ($this->take()[0] !== ')') {
419 throw new CommandError('SYNTAX');
420 }
421 return $result;
422 }
423 if ($kind === 'term') {
424 $stem = porterStem($spelling);
425 $documents = array_filter(
426 $this->stemSets,
427 fn(array $stems): bool => isset($stems[$stem]),
428 );
429 return [$stem, $documents];
430 }
431 if ($kind === 'prefix') {
432 $documents = array_filter(
433 $this->stemSets,
434 function (array $stems) use ($spelling): bool {
435 foreach (array_keys($stems) as $stem) {
436 if (str_starts_with($stem, $spelling)) {
437 return true;
438 }
439 }
440 return false;
441 },
442 );
443 return ["$spelling*", $documents];
444 }
445 throw new CommandError('SYNTAX');
446 }
447}

638 private function showIndex(array $arguments): void
639 {
640 $prefix = $arguments === [] ? '' : wordArgument($arguments[0]);
641 $occurrences = $this->occurrenceMap();
642 $stems = array_filter(
643 array_keys($occurrences),
644 fn(string $stem): bool => str_starts_with($stem, $prefix),
645 );
646 sort($stems, SORT_STRING);
647 $this->emit('INDEX', 'stems=' . count($stems));
648 foreach ($stems as $stem) {
649 $found = $occurrences[$stem];
650 $references = array_map(
651 fn(array $occurrence): string => $occurrence[0]->reference($occurrence[1]),
652 $found,
653 );
654 $this->emit($stem, 'df=' . documentCount($found), ...$references);

658 private function search(array $arguments): void
659 {
660 preg_match_all('/[()]|[^ ()]+/', $arguments[0], $matches);
661 $items = array_map(classify(...), $matches[0]);
662 $stemSets = array_map($this->stemSet(...), $this->documents);
663 [$canonical, $found] = (new QueryParser($items, $stemSets))->parse();
664 foreach ($items as [$kind, $spelling]) {
665 if ($kind === 'term' && isset($this->stopwords[$spelling])) {
666 throw new CommandError("STOPWORD $spelling");
667 }
668 }
669 $numbers = array_keys($found);
670 sort($numbers);
671 $this->emit('SEARCH', $canonical, 'count=' . count($numbers), ...$numbers);
672 }

674 private function phrase(array $arguments): void
675 {
676 $words = array_map(wordArgument(...), $arguments);
677 $stopwords = array_filter($words, fn(string $word): bool => isset($this->stopwords[$word]));
678 if (count($stopwords) === count($words)) {
679 throw new CommandError('STOPWORDS ONLY');
680 }
681 $targets = array_map(
682 fn(string $word): string => isset($this->stopwords[$word]) ? '?' : porterStem($word),
683 $words,
684 );
685 $matches = [];
686 foreach ($this->orderedDocuments() as $document) {
687 $tokens = $document->tokens;
688 for ($start = 0; $start + count($targets) <= count($tokens); $start++) {
689 $matched = true;
690 foreach ($targets as $offset => $target) {
691 if (!$this->tokenMatches($tokens[$start + $offset], $target)) {
692 $matched = false;
693 break;
694 }
695 }
696 if ($matched) {
697 $matches[] = $document->reference($tokens[$start]);
698 }
699 }
700 }
701 $phraseText = implode(' ', $targets);
702 $this->emit("PHRASE \"$phraseText\"", 'matches=' . count($matches), ...$matches);
703 }

705 private function near(array $arguments): void
706 {
707 $first = $this->termArgument($arguments[0]);
708 $second = $this->termArgument($arguments[1]);
709 $limit = numberArgument($arguments[2], 1, MAX_COUNT);
710 $found = [];
711 foreach ($this->orderedDocuments() as $document) {
712 $distance = closestDistance(
713 $this->stemPositions($document, $first),
714 $this->stemPositions($document, $second),
715 $first === $second,
716 );
717 if ($distance !== null && $distance <= $limit) {
718 $found[] = "{$document->number}:$distance";
719 }
720 }
721 $this->emit('NEAR', $first, $second, $limit, 'count=' . count($found), ...$found);
722 }

724 private function top(array $arguments): void
725 {
726 $limit = numberArgument($arguments[0], 1, MAX_COUNT);
727 if (count($arguments) === 2) {
728 $document = $this->documentArgument($arguments[1]);
729 $counts = $this->termFrequencies($document);
730 $header = "TOP $limit doc={$document->number}";
731 } else {
732 $counts = $this->totalFrequencies();
733 $header = "TOP $limit";
734 }
735 $ranked = array_slice(sortedByCount($counts), 0, $limit, true);
736 $items = array_map(
737 fn(string $stem, int $count): string => "$stem:$count",
738 array_keys($ranked),
739 $ranked,
740 );
741 $this->emit($header, ...$items);
742 }

744 private function kwic(array $arguments): void
745 {
746 $stem = $this->termArgument($arguments[0]);
747 $found = $this->occurrenceMap()[$stem] ?? [];
748 $this->emit('KWIC', $stem, 'count=' . count($found));
749 foreach ($found as [$document, $token]) {
750 $text = $document->lines[$token->line - 1];
751 $from = max(0, $token->start - KWIC_WIDTH);
752 $before = substr($text, $from, $token->start - $from);
753 $before = str_pad($before, KWIC_WIDTH, ' ', STR_PAD_LEFT);
754 $after = str_pad(substr($text, $token->end(), KWIC_WIDTH), KWIC_WIDTH);
755 $this->emit("|{$before}[{$token->text}]{$after}|", $document->reference($token));
756 }
757 }

761 $document = $this->documentArgument($arguments[0]);
762 $stem = $this->termArgument($arguments[1]);
763 $this->emit('TF', $document->number, $stem, $this->termFrequencies($document)[$stem] ?? 0);

788 private function similar(array $arguments): void
789 {
790 $document = $this->documentArgument($arguments[0]);
791 $limit = numberArgument($arguments[1], 1, MAX_COUNT);
792 $stems = $this->stemSet($document);
793 $scores = [];
794 foreach ($this->documents as $other) {
795 if ($other === $document) {
796 continue;
797 }
798 $otherStems = $this->stemSet($other);
799 $union = count($stems + $otherStems);
800 $common = count(array_intersect_key($stems, $otherStems));
801 [$numerator, $denominator] = $union > 0 ? reducedFraction($common, $union) : [0, 1];
802 $scores[] = [$numerator, $denominator, $other->number];
803 }
804 usort(
805 $scores,
806 fn(array $first, array $second): int =>
807 ($second[0] * $first[1] <=> $first[0] * $second[1]) ?: $first[2] <=> $second[2],
808 );
809 $items = [];
810 foreach (array_slice($scores, 0, $limit) as [$numerator, $denominator, $number]) {
811 $items[] = "$number:$numerator/$denominator";
812 }
813 $this->emit('SIMILAR', $document->number, ...$items);
814 }

816 private function rank(array $arguments): void
817 {
818 $limit = numberArgument($arguments[0], 1, MAX_COUNT);
819 $query = [];
820 foreach (array_slice($arguments, 1) as $argument) {
821 $query[$this->termArgument($argument)] = true;
822 }
823 $ranked = [];
824 foreach ($this->documents as $document) {
825 $counts = $this->termFrequencies($document);
826 $matched = 0;
827 $total = 0;
828 foreach (array_keys($query) as $stem) {
829 $count = $counts[$stem] ?? 0;
830 if ($count > 0) {
831 $matched++;
832 }
833 $total += $count;
834 }
835 if ($matched > 0) {
836 $ranked[] = [$matched, $total, $document->number];
837 }
838 }
839 usort(
840 $ranked,
841 fn(array $first, array $second): int =>
842 [$second[0], $second[1], $first[2]] <=> [$first[0], $first[1], $second[2]],
843 );
844 $items = [];
845 foreach (array_slice($ranked, 0, $limit) as [$matched, $total, $number]) {
846 $items[] = "$number:$matched:$total";
847 }
848 $this->emit('RANK count=' . count($ranked), ...$items);
849 }

956 public function execute(string $line): void
957 {
958 $parts = explode(' ', $line, 2);
959 $name = $parts[0];
960 try {
961 if (!isset($this->handlers[$name])) {
962 throw new CommandError("UNKNOWN COMMAND $name");
963 }
964 [$minimum, $maximum, $handler] = $this->handlers[$name];
965 $arguments = count($parts) === 2 ? splitArguments($name, $parts[1]) : [];
966 if (count($arguments) < $minimum || count($arguments) > $maximum) {
967 throw new CommandError("BAD ARGUMENTS $name");
968 }
969 $handler($arguments);
970 } catch (CommandError $error) {
971 $this->emit('ERROR', $error->getMessage());
972 }
973 }

976function main(): void
977{
978 $lines = file('php://stdin', FILE_IGNORE_NEW_LINES);
979 $engine = new SearchEngine(array_fill_keys(array_slice(explode(' ', $lines[0]), 1), true));
980 $index = 1;
981 while (str_starts_with($lines[$index], 'DOC ')) {
982 $document = $engine->addDocument((int) substr($lines[$index], 4));
983 $index++;
984 while ($lines[$index] !== 'ENDDOC') {
985 $document->addLine($lines[$index]);
986 $index++;
987 }
988 $index++;
989 }
990 foreach (array_slice($lines, $index) as $line) {
991 if ($line === 'END') {
992 break;
993 }
994 $engine->execute($line);
995 }
996 echo implode("\n", $engine->output()), "\n";
997}
