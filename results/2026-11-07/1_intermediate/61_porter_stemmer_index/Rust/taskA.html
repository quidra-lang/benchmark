54fn letter_pattern(word: &str) -> String {
55 let mut kinds = String::with_capacity(word.len());
56 for (index, letter) in word.chars().enumerate() {
57 let vowel = VOWELS.contains(letter) || (letter == 'y' && index > 0 && kinds.ends_with('C'));
58 kinds.push(if vowel { 'V' } else { 'C' });
59 }
60 kinds
61}
62
63fn measure(word: &str) -> usize {
64 letter_pattern(word).matches("VC").count()
65}
66
67fn has_vowel(word: &str) -> bool {
68 letter_pattern(word).contains('V')
69}
70
71fn ends_double(word: &str) -> bool {
72 let bytes = word.as_bytes();
73 letter_pattern(word).ends_with("CC") && bytes[bytes.len() - 1] == bytes[bytes.len() - 2]
74}
75
76fn ends_cvc(word: &str) -> bool {
77 letter_pattern(word).ends_with("CVC") && !word.ends_with(['w', 'x', 'y'])
78}
79
80fn final_e_condition(remainder: &str) -> bool {
81 let value = measure(remainder);
82 value > 1 || (value == 1 && !ends_cvc(remainder))
83}

85fn apply_rules(
86 word: &str,
87 rules: impl IntoIterator<Item = (&'static str, &'static str)>,
88 condition: impl Fn(&str, &str) -> bool,
89) -> (String, Option<&'static str>) {
90 let longest = rules
91 .into_iter()
92 .filter(|(suffix, _)| word.ends_with(suffix))
93 .max_by_key(|(suffix, _)| suffix.len());
94 let Some((suffix, replacement)) = longest else {
95 return (word.to_string(), None);
96 };
97 let remainder = &word[..word.len() - suffix.len()];
98 if !condition(suffix, remainder) {
99 return (word.to_string(), None);
100 }
101 (format!("{remainder}{replacement}"), Some(suffix))
102}
103
104fn tidy_step_1b(word: String) -> String {
105 if ["at", "bl", "iz"]
106 .iter()
107 .any(|suffix| word.ends_with(suffix))
108 {
109 return word + "e";
110 }
111 if ends_double(&word) && !word.ends_with(['l', 's', 'z']) {
112 return word[..word.len() - 1].to_string();
113 }
114 if measure(&word) == 1 && ends_cvc(&word) {
115 return word + "e";
116 }
117 word
118}
119
120fn apply_step(name: &str, word: &str) -> String {
121 match name {
122 "1a" => apply_rules(word, STEP_1A.iter().copied(), |_, _| true).0,
123 "1b" => {
124 let (result, suffix) =
125 apply_rules(word, STEP_1B.iter().copied(), |suffix, remainder| {
126 if suffix == "eed" {
127 measure(remainder) > 0
128 } else {
129 has_vowel(remainder)
130 }
131 });
132 if matches!(suffix, Some("ed" | "ing")) {
133 tidy_step_1b(result)
134 } else {
135 result
136 }
137 }
138 "1c" => apply_rules(word, [("y", "i")], |_, remainder| has_vowel(remainder)).0,
139 "2" => {
140 apply_rules(word, STEP_2.iter().copied(), |_, remainder| {
141 measure(remainder) > 0
142 })
143 .0
144 }
145 "3" => {
146 apply_rules(word, STEP_3.iter().copied(), |_, remainder| {
147 measure(remainder) > 0
148 })
149 .0
150 }
151 "4" => {
152 let rules = STEP_4_SUFFIXES.iter().map(|&suffix| (suffix, ""));
153 apply_rules(word, rules, |suffix, remainder| {
154 measure(remainder) > 1 && (suffix != "ion" || remainder.ends_with(['s', 't']))
155 })
156 .0
157 }
158 "5a" => {
159 apply_rules(word, [("e", "")], |_, remainder| {
160 final_e_condition(remainder)
161 })
162 .0
163 }
164 _ => {
165 if measure(word) > 1 && ends_double(word) && word.ends_with('l') {
166 word[..word.len() - 1].to_string()
167 } else {
168 word.to_string()
169 }
170 }
171 }
172}

174fn stemming_steps(word: &str) -> (String, Vec<(&'static str, String)>) {
175 let mut current = word.to_string();
176 let mut changes = Vec::new();
177 if word.len() > 2 {
178 for name in STEP_NAMES {
179 let result = apply_step(name, &current);
180 if result != current {
181 changes.push((name, result.clone()));
182 }
183 current = result;
184 }
185 }
186 (current, changes)
187}
188
189fn porter_stem(word: &str) -> String {
190 stemming_steps(word).0
191}

210fn number_argument(text: &str, minimum: usize, maximum: usize) -> Result<usize, String> {
211 let digits = (1..=7).contains(&text.len()) && text.bytes().all(|byte| byte.is_ascii_digit());
212 match text.parse() {
213 Ok(value) if digits && (minimum..=maximum).contains(&value) => Ok(value),
214 _ => Err(format!("BAD NUMBER {text}")),
215 }

219 let range = match name {

266fn closest_distance(positions: &[usize], others: &[usize], same_stem: bool) -> Option<usize> {
267 if same_stem {
268 return positions.windows(2).map(|pair| pair[1] - pair[0]).min();
269 }
270 let mut merged: Vec<(usize, bool)> = positions
271 .iter()
272 .map(|&position| (position, false))
273 .chain(others.iter().map(|&other| (other, true)))
274 .collect();
275 merged.sort_unstable();
276 merged
277 .windows(2)
278 .filter(|pair| pair[0].1 != pair[1].1)
279 .map(|pair| pair[1].0 - pair[0].0)
280 .min()
281}

287#[derive(Clone, Copy, PartialEq, Eq)]
288struct Fraction {
289 numerator: usize,
290 denominator: usize,
291}
292
293impl Fraction {
294 fn new(numerator: usize, denominator: usize) -> Self {
295 if denominator == 0 {
296 return Fraction {
297 numerator: 0,
298 denominator: 1,
299 };
300 }
301 let divisor = gcd(numerator, denominator);
302 Fraction {
303 numerator: numerator / divisor,
304 denominator: denominator / divisor,
305 }
306 }
307}
308
309impl Ord for Fraction {
310 fn cmp(&self, other: &Self) -> Ordering {
311 (self.numerator * other.denominator).cmp(&(other.numerator * self.denominator))
312 }
313}
314
315impl PartialOrd for Fraction {
316 fn partial_cmp(&self, other: &Self) -> Option<Ordering> {
317 Some(self.cmp(other))
318 }
319}
320
321impl Display for Fraction {
322 fn fmt(&self, formatter: &mut fmt::Formatter) -> fmt::Result {
323 write!(formatter, "{}/{}", self.numerator, self.denominator)
324 }

327#[derive(Clone, PartialEq)]
328enum Lexeme {
329 And,
330 Or,
331 Not,
332 Open,
333 Close,
334 Term(String),
335 Prefix(String),
336}
337
338fn lexemes(expression: &str) -> Vec<&str> {
339 let mut found = Vec::new();
340 let mut start = None;
341 for (index, character) in expression.char_indices() {
342 if matches!(character, ' ' | '(' | ')') {
343 if let Some(begin) = start.take() {
344 found.push(&expression[begin..index]);
345 }
346 if character != ' ' {
347 found.push(&expression[index..index + 1]);
348 }
349 } else if start.is_none() {
350 start = Some(index);
351 }
352 }
353 if let Some(begin) = start {
354 found.push(&expression[begin..]);
355 }
356 found
357}
358
359fn classify(lexeme: &str) -> Result<Lexeme, String> {
360 match lexeme {
361 "AND" => Ok(Lexeme::And),
362 "OR" => Ok(Lexeme::Or),
363 "NOT" => Ok(Lexeme::Not),
364 "(" => Ok(Lexeme::Open),
365 ")" => Ok(Lexeme::Close),
366 _ if is_word(lexeme) => Ok(Lexeme::Term(lexeme.to_ascii_lowercase())),
367 _ => match lexeme.strip_suffix('*') {
368 Some(spelling) if is_word(spelling) => {
369 Ok(Lexeme::Prefix(spelling.to_ascii_lowercase()))
370 }
371 _ => Err(format!("BAD TERM {lexeme}")),
372 },
373 }
374}

406 fn add_line(&mut self, text: &str) -> &[Token] {
407 self.lines.push(text.to_string());
408 let first = self.tokens.len();
409 let bytes = text.as_bytes();
410 let mut start = 0;
411 while start < bytes.len() {
412 if !bytes[start].is_ascii_alphabetic() {
413 start += 1;
414 continue;
415 }
416 let end = bytes[start..]
417 .iter()
418 .position(|byte| !byte.is_ascii_alphabetic())
419 .map_or(bytes.len(), |length| start + length);
420 let word = &text[start..end];
421 let form = word.to_ascii_lowercase();
422 self.tokens.push(Token {
423 text: word.to_string(),
424 stem: porter_stem(&form),
425 form,
426 line: self.lines.len(),
427 position: self.tokens.len() + 1,
428 start,
429 });
430 start = end;
431 }
432 &self.tokens[first..]
433 }

446type Evaluation = (String, BTreeSet<usize>);
447
448struct QueryParser<'a> {
449 items: &'a [Lexeme],
450 position: usize,
451 stem_sets: &'a BTreeMap<usize, HashSet<&'a str>>,
452}
453
454impl<'a> QueryParser<'a> {
455 fn peek(&self) -> Option<&'a Lexeme> {
456 self.items.get(self.position)
457 }
458
459 fn take(&mut self) -> Result<&'a Lexeme, String> {
460 let item = self.peek().ok_or("SYNTAX")?;
461 self.position += 1;
462 Ok(item)
463 }
464
465 fn parse(mut self) -> Result<Evaluation, String> {
466 let result = self.expression()?;
467 if self.position < self.items.len() {
468 return Err("SYNTAX".to_string());
469 }
470 Ok(result)
471 }
472
473 fn expression(&mut self) -> Result<Evaluation, String> {
474 let (mut canonical, mut documents) = self.conjunction()?;
475 while self.peek() == Some(&Lexeme::Or) {
476 self.position += 1;
477 let (right, right_documents) = self.conjunction()?;
478 canonical = format!("({canonical} OR {right})");
479 documents.extend(right_documents);
480 }
481 Ok((canonical, documents))
482 }
483
484 fn conjunction(&mut self) -> Result<Evaluation, String> {
485 let (mut canonical, mut documents) = self.negation()?;
486 while matches!(
487 self.peek(),
488 Some(Lexeme::And | Lexeme::Not | Lexeme::Open | Lexeme::Term(_) | Lexeme::Prefix(_))
489 ) {
490 if self.peek() == Some(&Lexeme::And) {
491 self.position += 1;
492 }
493 let (right, right_documents) = self.negation()?;
494 canonical = format!("({canonical} AND {right})");
495 documents.retain(|number| right_documents.contains(number));
496 }
497 Ok((canonical, documents))
498 }
499
500 fn negation(&mut self) -> Result<Evaluation, String> {
501 if self.peek() == Some(&Lexeme::Not) {
502 self.position += 1;
503 let (canonical, documents) = self.negation()?;
504 let complement = self
505 .stem_sets
506 .keys()
507 .filter(|number| !documents.contains(number))
508 .copied()
509 .collect();
510 return Ok((format!("(NOT {canonical})"), complement));
511 }
512 self.element()
513 }
514
515 fn element(&mut self) -> Result<Evaluation, String> {
516 match self.take()? {
517 Lexeme::Open => {
518 let result = self.expression()?;
519 if self.take()? != &Lexeme::Close {
520 return Err("SYNTAX".to_string());
521 }
522 Ok(result)
523 }
524 Lexeme::Term(spelling) => {
525 let stem = porter_stem(spelling);
526 let documents = self.documents_where(|stems| stems.contains(stem.as_str()));
527 Ok((stem, documents))
528 }
529 Lexeme::Prefix(spelling) => {
530 let documents = self.documents_where(|stems| {
531 stems.iter().any(|stem| stem.starts_with(spelling.as_str()))
532 });
533 Ok((format!("{spelling}*"), documents))
534 }
535 _ => Err("SYNTAX".to_string()),
536 }
537 }
538
539 fn documents_where(&self, predicate: impl Fn(&HashSet<&str>) -> bool) -> BTreeSet<usize> {
540 self.stem_sets
541 .iter()
542 .filter(|(_, stems)| predicate(stems))
543 .map(|(&number, _)| number)
544 .collect()
545 }
546}

548struct SearchEngine {
549 stopwords: HashSet<String>,
550 documents: BTreeMap<usize, Document>,
551}
552
553impl SearchEngine {
554 fn is_indexed(&self, token: &Token) -> bool {
555 !self.stopwords.contains(&token.form)
556 }
557
558 fn indexed_tokens<'a>(&'a self, document: &'a Document) -> impl Iterator<Item = &'a Token> {
559 document
560 .tokens
561 .iter()
562 .filter(move |token| self.is_indexed(token))
563 }
564
565 fn term_frequencies<'a>(&'a self, document: &'a Document) -> HashMap<&'a str, usize> {
566 let mut counts: HashMap<&str, usize> = HashMap::new();
567 for token in self.indexed_tokens(document) {
568 *counts.entry(token.stem.as_str()).or_default() += 1;
569 }
570 counts
571 }
572
573 fn stem_set<'a>(&'a self, document: &'a Document) -> HashSet<&'a str> {
574 self.indexed_tokens(document)
575 .map(|token| token.stem.as_str())
576 .collect()
577 }
578
579 fn total_frequencies(&self) -> HashMap<&str, usize> {
580 let mut totals: HashMap<&str, usize> = HashMap::new();
581 for document in self.documents.values() {
582 for (stem, count) in self.term_frequencies(document) {
583 *totals.entry(stem).or_default() += count;
584 }
585 }
586 totals
587 }
588
589 fn occurrence_map(&self) -> BTreeMap<&str, Vec<(&Document, &Token)>> {
590 let mut occurrences: BTreeMap<&str, Vec<(&Document, &Token)>> = BTreeMap::new();
591 for document in self.documents.values() {
592 for token in self.indexed_tokens(document) {
593 occurrences
594 .entry(token.stem.as_str())
595 .or_default()
596 .push((document, token));
597 }
598 }
599 occurrences
600 }
601
602 fn occurrences(&self, stem: &str) -> Vec<(&Document, &Token)> {
603 self.occurrence_map().remove(stem).unwrap_or_default()
604 }
605
606 fn stem_positions(&self, document: &Document, stem: &str) -> Vec<usize> {
607 self.indexed_tokens(document)
608 .filter(|token| token.stem == stem)
609 .map(|token| token.position)
610 .collect()
611 }

613 fn form_count(&self, form: &str) -> usize {
614 self.documents
615 .values()
616 .flat_map(|document| &document.tokens)
617 .filter(|token| token.form == form)
618 .count()
619 }

674 Ok(format!(
675 "MEASURE {word} {} m={} vowel={} double={} cvc={}",

677 measure(&word),
678 yes_no(has_vowel(&word)),
679 yes_no(ends_double(&word)),
680 yes_no(ends_cvc(&word)),
681 ))
682 }

684 fn show_index(&self, arguments: &[&str]) -> CommandResult {
685 let prefix = match arguments.first() {
686 Some(argument) => word_argument(argument)?,
687 None => String::new(),
688 };
689 let occurrences = self.occurrence_map();
690 let stems: Vec<_> = occurrences
691 .iter()
692 .filter(|(stem, _)| stem.starts_with(prefix.as_str()))
693 .collect();
694 let mut lines = vec![format!("INDEX stems={}", stems.len())];
695 for (stem, found) in stems {
696 let references = found
697 .iter()
698 .map(|(document, token)| document.reference(token));
699 lines.push(with_items(
700 format!("{stem} df={}", document_count(found)),
701 references,
702 ));
703 }
704 Ok(lines.join("\n"))
705 }

707 fn search(&self, arguments: &[&str]) -> CommandResult {
708 let items = lexemes(arguments[0])
709 .into_iter()
710 .map(classify)
711 .collect::<Result<Vec<_>, _>>()?;
712 let stem_sets: BTreeMap<usize, HashSet<&str>> = self
713 .documents
714 .values()
715 .map(|document| (document.number, self.stem_set(document)))
716 .collect();
717 let parser = QueryParser {
718 items: &items,
719 position: 0,
720 stem_sets: &stem_sets,
721 };
722 let (canonical, matches) = parser.parse()?;
723 for item in &items {
724 if let Lexeme::Term(spelling) = item
725 && self.stopwords.contains(spelling)
726 {
727 return Err(format!("STOPWORD {spelling}"));
728 }
729 }
730 Ok(with_items(
731 format!("SEARCH {canonical} count={}", matches.len()),
732 matches,
733 ))
734 }

736 fn phrase(&self, arguments: &[&str]) -> CommandResult {
737 let words = arguments
738 .iter()
739 .copied()
740 .map(word_argument)
741 .collect::<Result<Vec<_>, _>>()?;
742 if words.iter().all(|word| self.stopwords.contains(word)) {
743 return Err("STOPWORDS ONLY".to_string());
744 }
745 let targets: Vec<String> = words
746 .iter()
747 .map(|word| {
748 if self.stopwords.contains(word) {
749 "?".to_string()
750 } else {
751 porter_stem(word)
752 }
753 })
754 .collect();
755 let mut matches = Vec::new();
756 for document in self.documents.values() {
757 for window in document.tokens.windows(targets.len()) {
758 if window
759 .iter()
760 .zip(&targets)
761 .all(|(token, target)| self.token_matches(token, target))
762 {
763 matches.push(document.reference(&window[0]));
764 }
765 }
766 }
767 let header = format!("PHRASE \"{}\" matches={}", targets.join(" "), matches.len());
768 Ok(with_items(header, matches))
769 }

771 fn near(&self, arguments: &[&str]) -> CommandResult {
772 let first = self.term_argument(arguments[0])?;
773 let second = self.term_argument(arguments[1])?;
774 let limit = number_argument(arguments[2], 1, MAX_COUNT)?;
775 let mut found = Vec::new();
776 for document in self.documents.values() {
777 let distance = closest_distance(
778 &self.stem_positions(document, &first),
779 &self.stem_positions(document, &second),
780 first == second,
781 );
782 if let Some(distance) = distance
783 && distance <= limit
784 {
785 found.push(format!("{}:{distance}", document.number));
786 }
787 }
788 let header = format!("NEAR {first} {second} {limit} count={}", found.len());
789 Ok(with_items(header, found))
790 }

792 fn top(&self, arguments: &[&str]) -> CommandResult {
793 let limit = number_argument(arguments[0], 1, MAX_COUNT)?;
794 let (header, counts) = match arguments.get(1) {
795 Some(argument) => {
796 let document = self.document_argument(argument)?;
797 (
798 format!("TOP {limit} doc={}", document.number),
799 self.term_frequencies(document),
800 )
801 }
802 None => (format!("TOP {limit}"), self.total_frequencies()),
803 };
804 let ranked = ranked_by_count(&counts);
805 let items = ranked
806 .iter()
807 .take(limit)
808 .map(|(stem, count)| format!("{stem}:{count}"));
809 Ok(with_items(header, items))

812 fn kwic(&self, arguments: &[&str]) -> CommandResult {
813 let stem = self.term_argument(arguments[0])?;
814 let found = self.occurrences(&stem);
815 let mut lines = vec![format!("KWIC {stem} count={}", found.len())];
816 for (document, token) in found {
817 let text = &document.lines[token.line - 1];
818 let before = &text[token.start.saturating_sub(KWIC_WIDTH)..token.start];
819 let after = &text[token.end()..(token.end() + KWIC_WIDTH).min(text.len())];
820 lines.push(format!(
821 "|{before:>width$}[{}]{after:<width$}| {}",
822 token.text,
823 document.reference(token),
824 width = KWIC_WIDTH,
825 ));
826 }
827 Ok(lines.join("\n"))
828 }

860 fn similar(&self, arguments: &[&str]) -> CommandResult {
861 let document = self.document_argument(arguments[0])?;
862 let limit = number_argument(arguments[1], 1, MAX_COUNT)?;
863 let stems = self.stem_set(document);
864 let mut scores = Vec::new();
865 for other in self.documents.values() {
866 if other.number == document.number {
867 continue;
868 }
869 let other_stems = self.stem_set(other);
870 let shared = stems.intersection(&other_stems).count();
871 let union = stems.union(&other_stems).count();
872 scores.push((Fraction::new(shared, union), other.number));
873 }
874 scores.sort_by_key(|&(score, number)| (Reverse(score), number));
875 let items = scores
876 .iter()
877 .take(limit)
878 .map(|(score, number)| format!("{number}:{score}"));
879 Ok(with_items(format!("SIMILAR {}", document.number), items))
880 }
881
882 fn rank(&self, arguments: &[&str]) -> CommandResult {
883 let limit = number_argument(arguments[0], 1, MAX_COUNT)?;
884 let query = arguments[1..]
885 .iter()
886 .map(|argument| self.term_argument(argument))
887 .collect::<Result<HashSet<_>, _>>()?;
888 let mut ranked = Vec::new();
889 for document in self.documents.values() {
890 let counts = self.term_frequencies(document);
891 let frequencies: Vec<usize> = query
892 .iter()
893 .map(|stem| counts.get(stem.as_str()).copied().unwrap_or(0))
894 .collect();
895 let matched = frequencies.iter().filter(|&&count| count > 0).count();
896 let total: usize = frequencies.iter().sum();
897 if matched > 0 {
898 ranked.push((matched, total, document.number));
899 }
900 }
901 ranked.sort_by_key(|&(matched, total, number)| (Reverse(matched), Reverse(total), number));
902 let items = ranked
903 .iter()
904 .take(limit)
905 .map(|(matched, total, number)| format!("{number}:{matched}:{total}"));
906 Ok(with_items(format!("RANK count={}", ranked.len()), items))
907 }

1011 fn run(&mut self, name: &str, rest: Option<&str>) -> CommandResult {
1012 let range = argument_range(name).ok_or_else(|| format!("UNKNOWN COMMAND {name}"))?;
1013 let arguments = rest.map_or_else(Vec::new, |rest| split_arguments(name, rest));
1014 if !range.contains(&arguments.len()) {
1015 return Err(format!("BAD ARGUMENTS {name}"));
1016 }
1017 match name {
1018 "STEM" => self.stem_words(&arguments),
1019 "STEPS" => self.show_steps(&arguments),
1020 "MEASURE" => self.show_measure(&arguments),
1021 "INDEX" => self.show_index(&arguments),
1022 "SEARCH" => self.search(&arguments),
1023 "PHRASE" => self.phrase(&arguments),
1024 "NEAR" => self.near(&arguments),
1025 "TOP" => self.top(&arguments),
1026 "KWIC" => self.kwic(&arguments),
1027 "TF" => self.term_frequency(&arguments),
1028 "DF" => self.document_frequency(&arguments),
1029 "VARIANTS" => self.variants(&arguments),
1030 "SIMILAR" => self.similar(&arguments),
1031 "RANK" => self.rank(&arguments),
1032 "INFO" => self.info(&arguments),
1033 "LINE" => self.show_line(&arguments),
1034 "STATS" => self.stats(),
1035 "STOP" => self.stop(&arguments),
1036 "UNSTOP" => self.unstop(&arguments),
1037 "APPEND" => self.append(&arguments),
1038 "DELETE" => self.delete(&arguments),
1039 _ => unreachable!("command names are checked by argument_range"),
1040 }
1041 }
1042
1043 fn execute(&mut self, line: &str) -> String {
1044 let (name, rest) = match line.split_once(' ') {
1045 Some((name, rest)) => (name, Some(rest)),
1046 None => (line, None),
1047 };
1048 self.run(name, rest)
1049 .unwrap_or_else(|message| format!("ERROR {message}"))
1050 }
1051}

1053fn main() {
1054 let mut input = String::new();
1055 io::stdin()
1056 .read_to_string(&mut input)
1057 .expect("failed to read standard input");
1058 let mut lines = input.lines().peekable();
1059 let stopwords = lines
1060 .next()
1061 .unwrap_or_default()
1062 .split_whitespace()
1063 .skip(1)
1064 .map(String::from)
1065 .collect();
1066 let mut engine = SearchEngine {
1067 stopwords,
1068 documents: BTreeMap::new(),
1069 };
1070 while let Some(header) = lines.next_if(|line| line.starts_with("DOC ")) {
1071 let number = header[4..].parse().expect("invalid document number");
1072 let mut document = Document::new(number);
1073 for line in lines.by_ref() {
1074 if line == "ENDDOC" {
1075 break;
1076 }
1077 document.add_line(line);
1078 }
1079 engine.documents.insert(number, document);
1080 }
1081 let mut output = BufWriter::new(io::stdout().lock());
1082 for line in lines.take_while(|&line| line != "END") {
1083 writeln!(output, "{}", engine.execute(line)).expect("failed to write output");
1084 }
1085}
