23function letter_pattern(word)
24 kinds = Char[]
25 for (index, letter) in enumerate(word)
26 vowel = letter in VOWELS || (letter == 'y' && index > 1 && kinds[end] == 'C')
27 push!(kinds, vowel ? 'V' : 'C')
28 end
29 return String(kinds)
30end
31
32measure(word) = count("VC", letter_pattern(word))
33
34has_vowel(word) = 'V' in letter_pattern(word)
35
36ends_double(word) = endswith(letter_pattern(word), "CC") && word[end] == word[end-1]
37
38ends_cvc(word) = endswith(letter_pattern(word), "CVC") && word[end] ∉ "wxy"
39
40always(remainder) = true
41
42measure_above_zero(remainder) = measure(remainder) > 0
43
44measure_above_one(remainder) = measure(remainder) > 1
45
46function ion_condition(remainder)
47 return measure(remainder) > 1 && (endswith(remainder, "s") || endswith(remainder, "t"))
48end
49
50function final_e_condition(remainder)
51 value = measure(remainder)
52 return value > 1 || (value == 1 && !ends_cvc(remainder))
53end

55struct Rule
56 suffix::String
57 replacement::String
58 condition::Function
59end
60
61function rules_with(condition, replacements)
62 return [Rule(suffix, replacement, condition) for (suffix, replacement) in replacements]
63end

65const RULE_TABLES = Dict(
66 "1a" => rules_with(always, ["sses" => "ss", "ies" => "i", "ss" => "ss", "s" => ""]),
67 "1b" => [
68 Rule("eed", "ee", measure_above_zero),
69 Rule("ed", "", has_vowel),
70 Rule("ing", "", has_vowel),
71 ],
72 "1c" => [Rule("y", "i", has_vowel)],
73 "2" => rules_with(measure_above_zero, [
74 "ational" => "ate", "tional" => "tion", "enci" => "ence", "anci" => "ance",
75 "izer" => "ize", "abli" => "able", "alli" => "al", "entli" => "ent", "eli" => "e",
76 "ousli" => "ous", "ization" => "ize", "ation" => "ate", "ator" => "ate",
77 "alism" => "al", "iveness" => "ive", "fulness" => "ful", "ousness" => "ous",
78 "aliti" => "al", "iviti" => "ive", "biliti" => "ble",
79 ]),
80 "3" => rules_with(measure_above_zero, [
81 "icate" => "ic", "ative" => "", "alize" => "al", "iciti" => "ic",
82 "ical" => "ic", "ful" => "", "ness" => "",
83 ]),
84 "4" => [
85 Rule(suffix, "", suffix == "ion" ? ion_condition : measure_above_one)
86 for suffix in STEP_4_SUFFIXES
87 ],
88 "5a" => [Rule("e", "", final_e_condition)],
89)

91function apply_rules(word, rules)
92 matching = filter(rule -> endswith(word, rule.suffix), rules)
93 isempty(matching) && return (word, nothing)
94 selected = argmax(rule -> length(rule.suffix), matching)
95 remainder = word[1:end-length(selected.suffix)]
96 selected.condition(remainder) || return (word, nothing)
97 return (remainder * selected.replacement, selected.suffix)
98end

100function tidy_step_1b(word)
101 if any(suffix -> endswith(word, suffix), ("at", "bl", "iz"))
102 return word * "e"
103 elseif ends_double(word) && word[end] ∉ "lsz"
104 return word[1:end-1]
105 elseif measure(word) == 1 && ends_cvc(word)
106 return word * "e"
107 end
108 return word
109end
110
111function apply_step(name, word)
112 if name == "5b"
113 if measure(word) > 1 && ends_double(word) && endswith(word, "l")
114 return word[1:end-1]
115 end
116 return word
117 end
118 result, suffix = apply_rules(word, RULE_TABLES[name])
119 if name == "1b" && suffix in ("ed", "ing")
120 return tidy_step_1b(result)
121 end
122 return result
123end
124
125function stemming_steps(word)
126 changes = Pair{String,String}[]
127 if length(word) > 2
128 for name in STEP_NAMES
129 result = apply_step(name, word)
130 result != word && push!(changes, name => result)
131 word = result
132 end
133 end
134 return word, changes
135end

165function closest_distance(positions, others, same_stem)
166 if same_stem
167 gaps = diff(positions)
168 else
169 marked = [[(position, 0) for position in positions]; [(other, 1) for other in others]]
170 merged = sort!(marked)
171 gaps = [
172 later[1] - earlier[1]
173 for (earlier, later) in zip(merged, merged[2:end]) if earlier[2] != later[2]
174 ]
175 end
176 return isempty(gaps) ? nothing : minimum(gaps)
177end

233function next_kind(parser)
234 parser.position <= length(parser.items) || return nothing
235 return parser.items[parser.position].kind
236end
237
238function take_item!(parser)
239 parser.position > length(parser.items) && throw(CommandError("SYNTAX"))
240 item = parser.items[parser.position]
241 parser.position += 1
242 return item
243end
244
245function parse_query!(parser)
246 result = parse_expression!(parser)
247 parser.position <= length(parser.items) && throw(CommandError("SYNTAX"))
248 return result
249end
250
251function parse_expression!(parser)
252 canonical, documents = parse_conjunction!(parser)
253 while next_kind(parser) == :or
254 parser.position += 1
255 right, right_documents = parse_conjunction!(parser)
256 canonical = "($canonical OR $right)"
257 documents = union(documents, right_documents)
258 end
259 return canonical, documents
260end
261
262function parse_conjunction!(parser)
263 canonical, documents = parse_negation!(parser)
264 while next_kind(parser) in CONJUNCTION_CONTINUATIONS
265 next_kind(parser) == :and && (parser.position += 1)
266 right, right_documents = parse_negation!(parser)
267 canonical = "($canonical AND $right)"
268 documents = intersect(documents, right_documents)
269 end
270 return canonical, documents
271end
272
273function parse_negation!(parser)
274 next_kind(parser) == :not || return parse_element!(parser)
275 parser.position += 1
276 canonical, documents = parse_negation!(parser)
277 return "(NOT $canonical)", setdiff(Set(keys(parser.stem_sets)), documents)
278end
279
280function parse_element!(parser)
281 item = take_item!(parser)
282 if item.kind == :open
283 result = parse_expression!(parser)
284 take_item!(parser).kind == :close || throw(CommandError("SYNTAX"))
285 return result
286 elseif item.kind == :term
287 stem = porter_stem(item.spelling)
288 return stem, documents_where(stems -> stem in stems, parser)
289 elseif item.kind == :prefix
290 prefix_documents = documents_where(stems -> any(startswith(item.spelling), stems), parser)
291 return "$(item.spelling)*", prefix_documents
292 end
293 throw(CommandError("SYNTAX"))
294end
295
296function documents_where(condition, parser)
297 return Set{Int}(number for (number, stems) in parser.stem_sets if condition(stems))
298end

416function search(engine, arguments)
417 items = [classify(lexeme.match) for lexeme in eachmatch(LEXEME_PATTERN, arguments[1])]
418 stem_sets = Dict{Int,Set{String}}(
419 document.number => stem_set(engine, document) for document in values(engine.documents)

421 canonical, matches = parse_query!(QueryParser(items, stem_sets))
422 for item in items

430function phrase(engine, arguments)
431 words = map(word_argument, arguments)
432 all(word -> word in engine.stopwords, words) && throw(CommandError("STOPWORDS ONLY"))
433 targets = [word in engine.stopwords ? "?" : porter_stem(word) for word in words]
434 matches = String[]
435 for document in ordered_documents(engine)
436 tokens = document.tokens
437 for start in 1:length(tokens)-length(targets)+1
438 if all(token_matches(engine, tokens[start+offset-1], target)
439 for (offset, target) in enumerate(targets))
440 push!(matches, reference(document, tokens[start]))
441 end
442 end
443 end
444 emit(engine, "PHRASE \"$(join(targets, " "))\"", "matches=$(length(matches))", matches...)
445end

447function near(engine, arguments)
448 first_stem = term_argument(engine, arguments[1])
449 second_stem = term_argument(engine, arguments[2])
450 limit = number_argument(arguments[3], 1, MAX_COUNT)
451 found = String[]
452 for document in ordered_documents(engine)
453 distance = closest_distance(
454 stem_positions(engine, document, first_stem),
455 stem_positions(engine, document, second_stem),
456 first_stem == second_stem,
457 )
458 if !isnothing(distance) && distance <= limit
459 push!(found, "$(document.number):$distance")
460 end
461 end
462 emit(engine, "NEAR", first_stem, second_stem, limit, "count=$(length(found))", found...)
463end

465function top(engine, arguments)
466 limit = number_argument(arguments[1], 1, MAX_COUNT)

479function kwic(engine, arguments)
480 stem = term_argument(engine, arguments[1])
481 found = occurrences_of(engine, stem)
482 emit(engine, "KWIC", stem, "count=$(length(found))")
483 for (document, token) in found
484 text = document.lines[token.line]
485 after_start = token.start + length(token.text)
486 before = text[max(1, token.start - KWIC_WIDTH):token.start-1]
487 after = text[after_start:min(lastindex(text), after_start + KWIC_WIDTH - 1)]
488 context = "|$(lpad(before, KWIC_WIDTH))[$(token.text)]$(rpad(after, KWIC_WIDTH))|"
489 emit(engine, context, reference(document, token))
490 end
491end

514function similar_documents(engine, arguments)
515 document = document_argument(engine, arguments[1])
516 limit = number_argument(arguments[2], 1, MAX_COUNT)
517 stems = stem_set(engine, document)
518 scores = Tuple{Rational{Int},Int}[]
519 for other in values(engine.documents)
520 other === document && continue
521 other_stems = stem_set(engine, other)
522 union_size = length(union(stems, other_stems))
523 score = union_size == 0 ? 0 // 1 : length(intersect(stems, other_stems)) // union_size
524 push!(scores, (score, other.number))
525 end
526 sort!(scores; by=((score, number),) -> (-score, number))
527 items = (
528 "$number:$(numerator(score))/$(denominator(score))"
529 for (score, number) in first(scores, limit)
530 )
531 emit(engine, "SIMILAR", document.number, items...)
532end

534function rank_documents(engine, arguments)
535 limit = number_argument(arguments[1], 1, MAX_COUNT)
536 query = Set(term_argument(engine, argument) for argument in arguments[2:end])
537 ranked = Tuple{Int,Int,Int}[]
538 for document in values(engine.documents)
539 counts = term_frequencies(engine, document)
540 matched = count(stem -> haskey(counts, stem), query)
541 total = sum(stem -> get(counts, stem, 0), query)
542 matched > 0 && push!(ranked, (matched, total, document.number))
543 end
544 sort!(ranked; by=((matched, total, number),) -> (-matched, -total, number))
545 items = ("$number:$matched:$total" for (matched, total, number) in first(ranked, limit))
546 emit(engine, "RANK", "count=$(length(ranked))", items...)
547end

653function execute!(engine, line)
654 space = findfirst(' ', line)
655 name = isnothing(space) ? line : line[1:space-1]
656 try
657 haskey(COMMANDS, name) || throw(CommandError("UNKNOWN COMMAND $name"))
658 command = COMMANDS[name]
659 arguments = isnothing(space) ? String[] : split_arguments(name, line[space+1:end])
660 length(arguments) in command.arity || throw(CommandError("BAD ARGUMENTS $name"))
661 command.handler(engine, arguments)
662 catch exception
663 exception isa CommandError || rethrow()
664 emit(engine, "ERROR", exception.message)
665 end
666end
