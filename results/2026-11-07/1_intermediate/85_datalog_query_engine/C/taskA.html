18static void *take(Allocation **arena, size_t size) {
19 Allocation *a = malloc(sizeof *a);
20 void *p = calloc(1, size ? size : 1);
21 if (!a || !p) exit(EXIT_FAILURE);
22 *a = (Allocation){p, *arena};
23 *arena = a;
24 return p;
25}

57static void *grow(void *old, int count, int *capacity, size_t size, bool lasting) {
58 int next = *capacity ? *capacity * 2 : 8;
59 void *p = lasting ? persist((size_t)next * size) : allocate((size_t)next * size);
60 if (count) memcpy(p, old, (size_t)count * size);
61 *capacity = next;
62 return p;
63}

81static bool listed(const char *name, const char *list) {
82 size_t length = strlen(name);
83 for (const char *p = list; *p;) {
84 const char *end = strchr(p, ' ');
85 size_t n = end ? (size_t)(end - p) : strlen(p);
86 if (length == n && !memcmp(p, name, n)) return true;
87 if (!end) break;
88 p = end + 1;
89 }
90 return false;
91}

100static const char *intern_part(const char *s, size_t length) {
101 unsigned long hash = 5381;
102 for (size_t i = 0; i < length; i++) hash = hash * 33 + (unsigned char)s[i];
103 int bucket = (int)(hash % 4093);
104 for (Name *n = names[bucket]; n; n = n->next) if (n->length == length && !memcmp(n->text, s, length)) return n->text;
105 Name *n = persist(sizeof *n + length + 1);
106 n->length = length;
107 memcpy(n->text, s, length);
108 n->next = names[bucket];
109 names[bucket] = n;
110 return n->text;
111}

114 return intern_part(s, strlen(s));

142 if (!!a.symbol != !!b.symbol) return a.symbol ? 1 : -1;
143 return a.symbol ? strcmp(a.symbol, b.symbol) : (a.number > b.number) - (a.number < b.number);

146static void value_text(Value value) {
147 if (!value.symbol) {
148 printf("%lld", value.number);
149 return;
150 }
151 bool plain = lower(*value.symbol) && !listed(value.symbol, RESERVED);
152 for (const char *p = value.symbol; *p; p++) plain = plain && word_char(*p);
153 if (!plain) putchar('"');
154 fputs(value.symbol, stdout);
155 if (!plain) putchar('"');
156}

219static Parser parser(const char *line) {
220 Parser p = {0};
221 p.variables = persist(MAX_TERMS * sizeof(char *));
222 size_t length = strlen(line);
223 for (size_t i = 0; i < length;) {
224 char c = line[i];
225 if (c == ' ') {
226 i++;
227 continue;
228 }
229 if (c == '%') break;
230 size_t first = i;
231 const char *kind;
232 if (c == '"') {
233 const char *end = strchr(line + i + 1, '"');
234 if (!end) fail("unterminated string");
235 i = (size_t)(end - line) + 1;
236 kind = "string";
237 } else if (i + 1 < length && listed(intern_part(line + i, 2), ":- ?- != <= >=")) {
238 i += 2;
239 kind = intern_part(line + first, 2);
240 } else if (strchr("(),.:=<>+-*/", c)) {
241 i++;
242 kind = intern_part(line + first, 1);
243 } else if (word_char(c)) {
244 while (i < length && word_char(line[i])) i++;
245 const char *s = intern_part(line + first, i - first);
246 if (lower(c)) kind = listed(s, RESERVED) ? s : "name";
247 else if (c >= 'A' && c <= 'Z') kind = "variable";
248 else if (equal(s, "_")) kind = "anonymous";
249 else {
250 bool digits = true;
251 for (const char *q = s; *q; q++) digits = digits && digit(*q);
252 if (!digits) fail("bad token '%s'", s);
253 if (strtoll(s, NULL, 10) > 1000000000LL) fail("integer out of range '%s'", s);
254 kind = "integer";
255 }
256 } else {
257 fail("bad character '%c'", c);
258 kind = "";
259 }
260 p.tokens[p.count++] = (Token){kind, intern_part(line + first, i - first)};
261 }
262 p.tokens[p.count++] = (Token){"end", ""};
263 return p;
264}

286static Term read_term(Parser *p) {
287 Token t = p->tokens[p->at];
288 Term out = {.variable = -1};
289 if (is(p, "name") || is(p, "string")) {
290 p->at++;
291 out.value = symbol(t.text);
292 if (equal(t.kind, "string")) out.value = symbol(intern_part(t.text + 1, strlen(t.text) - 2));
293 } else if (is(p, "integer") || is(p, "-")) {
294 bool minus = is(p, "-");
295 if (minus) p->at++;
296 long long n = strtoll(expect(p, "integer").text, NULL, 10);
297 out.value = integer(minus ? -n : n);
298 } else if (is(p, "variable")) {
299 p->at++;
300 out.variable = variable_index(p, t.text);
301 } else if (is(p, "anonymous")) {
302 p->at++;
303 out.anonymous = true;
304 } else syntax(p);
305 return out;
306}

308static Atom *read_atom(Parser *p) {
309 const char *name = expect(p, "name").text;
310 Term terms[MAX_TERMS];
311 int count = 0;
312 if (is(p, "(")) {
313 p->at++;
314 terms[count++] = read_term(p);
315 while (is(p, ",")) {
316 p->at++;
317 terms[count++] = read_term(p);
318 }
319 expect(p, ")");
320 }
321 Atom *a = persist(sizeof *a);
322 a->name = name;
323 a->count = count;
324 a->terms = persist((size_t)count * sizeof(Term));
325 if (count) memcpy(a->terms, terms, (size_t)count * sizeof(Term));
326 return a;
327}

338static Expr *factor(Parser *p) {
339 if (is(p, "(")) {
340 p->at++;
341 Expr *e = expression(p);
342 expect(p, ")");
343 return node("()", e, NULL);
344 }
345 Expr *e = persist(sizeof *e);
346 e->leaf = true;
347 e->term = read_term(p);
348 return e;
349}

351static Expr *product(Parser *p) {
352 Expr *out = factor(p);
353 while (is(p, "*") || is(p, "/") || is(p, "mod")) {
354 const char *op = p->tokens[p->at++].text;
355 out = node(op, out, factor(p));
356 }
357 return out;
358}

360static Expr *expression(Parser *p) {
361 Expr *out = product(p);
362 while (is(p, "+") || is(p, "-")) {
363 const char *op = p->tokens[p->at++].text;
364 out = node(op, out, product(p));
365 }
366 return out;
367}

369static Literal literal(Parser *p) {
370 Literal out = {0};
371 if (is(p, "not")) {
372 p->at++;
373 out.negative = true;
374 out.atom = read_atom(p);
375 } else if (is(p, "name") && !listed(p->tokens[p->at + 1].kind, "= != < <= > >= + - * / mod")) out.atom = read_atom(p);
376 else {
377 out.left = expression(p);
378 if (!listed(p->tokens[p->at].kind, "= != < <= > >=")) syntax(p);
379 out.op = p->tokens[p->at++].text;
380 out.right = expression(p);
381 }
382 return out;
383}

385static Body read_body(Parser *p) {
386 Literal items[MAX_TERMS];
387 int count = 0;
388 items[count++] = literal(p);
389 while (is(p, ",")) {
390 p->at++;
391 items[count++] = literal(p);
392 }
393 Body body = {persist((size_t)count * sizeof(Literal)), count};
394 memcpy(body.items, items, (size_t)count * sizeof(Literal));
395 return body;
396}

398static Statement statement(Parser *p) {
399 Statement s = {.kind = p->tokens[p->at].kind, .aggregate = "", .aggregate_variable = -1};
400 Rule *r = persist(sizeof *r);
401 s.rule = r;
402 if (is(p, "name") || is(p, ":-")) {
403 if (is(p, "name")) r->head = read_atom(p);
404 if (is(p, ":-")) {
405 p->at++;
406 r->body = read_body(p);
407 s.kind = "rule";
408 } else {
409 if (!r->head) syntax(p);
410 s.kind = "fact";
411 }
412 } else if (is(p, "?-")) {
413 p->at++;
414 s.kind = "query";
415 if (is(p, "count")) {
416 s.aggregate = "count";
417 p->at++;
418 } else if (is(p, "sum") || is(p, "min") || is(p, "max")) {
419 s.aggregate = p->tokens[p->at++].text;
420 s.aggregate_variable = variable_index(p, expect(p, "variable").text);
421 expect(p, ":");
422 }
423 r->body = read_body(p);
424 } else {
425 if (!listed(s.kind, "retract why dump forget listing strata stats check predicates undo halt")) syntax(p);
426 p->at++;
427 if (listed(s.kind, "retract why")) r->head = read_atom(p);
428 else if (equal(s.kind, "dump")) s.name = expect(p, "name").text;
429 else if (equal(s.kind, "forget")) s.number = strtoll(expect(p, "integer").text, NULL, 10);
430 else if (equal(s.kind, "listing") && is(p, "name")) s.name = p->tokens[p->at++].text;
431 }
432 expect(p, ".");
433 if (!is(p, "end")) syntax(p);
434 r->variables = p->variables;
435 r->variable_count = p->variable_count;
436 return s;
437}

491static void term_variables(Term t, Variables *vars, bool unique) {
492 if (t.variable < 0) return;
493 if (unique) for (int i = 0; i < vars->count; i++) if (vars->ids[i] == t.variable) return;
494 vars->ids[vars->count++] = t.variable;

497static void expr_variables(Expr *e, Variables *vars, bool unique) {
498 if (e->leaf) term_variables(e->term, vars, unique);
499 else {
500 expr_variables(e->left, vars, unique);
501 if (e->right) expr_variables(e->right, vars, unique);
502 }
503}

505static void literal_variables(Literal l, Variables *vars, bool unique) {
506 if (l.atom) for (int i = 0; i < l.atom->count; i++) term_variables(l.atom->terms[i], vars, unique);
507 else {
508 expr_variables(l.left, vars, unique);
509 expr_variables(l.right, vars, unique);
510 }
511}

513static Variables rule_variables(const Rule *r, bool include_head, bool unique) {
514 Variables out = {0};
515 if (include_head && r->head) for (int i = 0; i < r->head->count; i++) term_variables(r->head->terms[i], &out, unique);
516 for (int i = 0; i < r->body.count; i++) literal_variables(r->body.items[i], &out, unique);
517 return out;
518}

524static void safety(const Rule *r, int aggregate) {
525 if (r->head) for (int i = 0; i < r->head->count; i++) if (r->head->terms[i].anonymous) fail("anonymous variable in head");
526 for (int i = 0; i < r->body.count; i++) {
527 Literal l = r->body.items[i];
528 if (!l.atom && (expr_anonymous(l.left) || expr_anonymous(l.right))) fail("anonymous variable in comparison");
529 }
530 bool bound[MAX_TERMS] = {false};
531 for (int i = 0; i < r->body.count; i++) {
532 Literal l = r->body.items[i];
533 if (l.atom && !l.negative) for (int j = 0; j < l.atom->count; j++) if (l.atom->terms[j].variable >= 0) bound[l.atom->terms[j].variable] = true;
534 }
535 Variables needed = {0};
536 if (r->head) for (int i = 0; i < r->head->count; i++) term_variables(r->head->terms[i], &needed, true);
537 if (aggregate >= 0) needed.ids[needed.count++] = aggregate;
538 for (int i = 0; i < r->body.count; i++) {
539 Literal l = r->body.items[i];
540 if (!l.atom || l.negative) literal_variables(l, &needed, true);
541 }
542 for (int i = 0; i < needed.count; i++) if (!bound[needed.ids[i]]) fail("unsafe variable %s", r->variables[needed.ids[i]]);
543}
544typedef struct {
545 const char *names[252];
546 int arities[252], count;
547} ArityTable;

549static void check_atom(ArityTable *table, Atom *a, bool clause) {
550 int found = -1;
551 for (int i = 0; i < table->count; i++) if (equal(table->names[i], a->name)) {
552 found = i;
553 break;
554 }
555 if (found < 0) {
556 if (!clause) fail("unknown predicate %s", a->name);
557 table->names[table->count] = a->name;
558 table->arities[table->count++] = a->count;
559 } else if (table->arities[found] != a->count) fail("arity mismatch for %s: expected %d, got %d", a->name, table->arities[found], a->count);
560}
561
562static ArityTable predicate_check(const Rule *r, bool clause, bool head) {
563 ArityTable table = {0};
564 for (int i = 0; i < state.predicate_count; i++) {
565 table.names[table.count] = state.predicates[i].name;
566 table.arities[table.count++] = state.predicates[i].arity;
567 }
568 if (head && r->head) check_atom(&table, r->head, clause);
569 for (int i = 0; i < r->body.count; i++) if (r->body.items[i].atom) check_atom(&table, r->body.items[i].atom, clause);
570 return table;
571}

573static int table_index(const ArityTable *table, const char *name) {
574 for (int i = 0; i < table->count; i++) if (equal(table->names[i], name)) return i;
575 return -1;
576}

578static bool strata(const ArityTable *table, Rule *extra, int *levels) {
579 for (int i = 0; i < table->count; i++) levels[i] = 0;
580 for (int pass = 0; pass < table->count; pass++) {
581 bool changed = false;
582 for (int ri = 0; ri < state.rule_count + !!extra; ri++) {
583 Rule *r = ri < state.rule_count ? state.rules[ri] : extra;
584 if (!r->head) continue;
585 int h = table_index(table, r->head->name);
586 for (int j = 0; j < r->body.count; j++) {
587 Literal l = r->body.items[j];
588 if (l.atom) {
589 int value = levels[table_index(table, l.atom->name)] + l.negative;
590 if (levels[h] < value) {
591 levels[h] = value;
592 changed = true;
593 }
594 }
595 }
596 }
597 if (!changed) return true;
598 if (pass + 1 == table->count) return false;
599 }
600 return true;
601}

620static bool match(Atom *atom, const Tuple *tuple, Environment source, Environment result, int variable_count) {
621 if (variable_count) {
622 memcpy(result.values, source.values, (size_t)variable_count * sizeof(Value));
623 memcpy(result.bound, source.bound, (size_t)variable_count * sizeof(bool));
624 }
625 for (int i = 0; i < atom->count; i++) {
626 Term t = atom->terms[i];
627 Value value = tuple->values[i];
628 if (t.anonymous) continue;
629 if (t.variable < 0) {
630 if (value_compare(t.value, value)) return false;
631 } else if (result.bound[t.variable]) {
632 if (value_compare(result.values[t.variable], value)) return false;
633 } else {
634 result.bound[t.variable] = true;
635 result.values[t.variable] = value;
636 }
637 }
638 return true;
639}

647static long long floor_divide(long long a, long long b) {
648 long long q = a / b, r = a % b;
649 return r && ((r < 0) != (b < 0)) ? q - 1 : q;
650}
651
652static bool evaluate(Expr *e, Environment env, Value *out) {
653 if (e->leaf) {
654 *out = e->term.variable >= 0 ? env.values[e->term.variable] : e->term.value;
655 return true;
656 }
657 Value a, b;
658 if (!evaluate(e->left, env, &a)) return false;
659 if (equal(e->op, "()")) {
660 *out = a;
661 return true;
662 }
663 if (!evaluate(e->right, env, &b) || a.symbol || b.symbol || ((equal(e->op, "/") || equal(e->op, "mod")) && !b.number)) return false;
664 long long n;
665 if (equal(e->op, "+")) n = a.number + b.number;
666 else if (equal(e->op, "-")) n = a.number - b.number;
667 else if (equal(e->op, "*")) n = a.number * b.number;
668 else if (equal(e->op, "/")) n = floor_divide(a.number, b.number);
669 else n = a.number - b.number * floor_divide(a.number, b.number);
670 *out = integer(n);
671 return true;
672}

674static bool comparison(Literal l, Environment env) {
675 Value a, b;
676 if (!evaluate(l.left, env, &a) || !evaluate(l.right, env, &b)) return false;
677 int c = value_compare(a, b);
678 if (equal(l.op, "=")) return !c;
679 if (equal(l.op, "!=")) return c != 0;
680 if (a.symbol || b.symbol) return false;
681 return equal(l.op, "<") ? c < 0 : equal(l.op, "<=") ? c <= 0 : equal(l.op, ">") ? c > 0 : c >= 0;
682}
683typedef struct {
684 char *text;
685 size_t length, capacity;
686} Buffer;
687
688static void append(Buffer *b, const char *s) {
689 size_t n = strlen(s);
690 if (b->length + n + 1 > b->capacity) {
691 size_t capacity = b->capacity ? b->capacity : 64;
692 while (capacity < b->length + n + 1) capacity *= 2;
693 char *text = realloc(b->text, capacity);
694 if (!text) exit(EXIT_FAILURE);
695 b->text = text;
696 b->capacity = capacity;
697 }
698 memcpy(b->text + b->length, s, n + 1);
699 b->length += n;
700}
701
702static void append_value(Buffer *b, Value value) {
703 if (!value.symbol) {
704 char text[32];
705 snprintf(text, sizeof text, "%lld", value.number);
706 append(b, text);
707 return;
708 }
709 bool plain = lower(*value.symbol) && !listed(value.symbol, RESERVED);
710 for (const char *p = value.symbol; *p; p++) plain = plain && word_char(*p);
711 if (!plain) append(b, "\"");
712 append(b, value.symbol);
713 if (!plain) append(b, "\"");
714}

716static void append_term(Buffer *b, Term t, const Rule *r, const Environment *env) {
717 if (t.anonymous) append(b, "_");
718 else if (t.variable < 0) append_value(b, t.value);
719 else if (env) append_value(b, env->values[t.variable]);
720 else append(b, r->variables[t.variable]);
721}
722
723static void append_expr(Buffer *b, Expr *e, const Rule *r, const Environment *env) {
724 if (e->leaf) {
725 append_term(b, e->term, r, env);
726 return;
727 }
728 if (equal(e->op, "()")) {
729 append(b, "(");
730 append_expr(b, e->left, r, env);
731 append(b, ")");
732 return;
733 }
734 append_expr(b, e->left, r, env);
735 append(b, " ");
736 append(b, e->op);
737 append(b, " ");
738 append_expr(b, e->right, r, env);
739}

741static void append_atom(Buffer *b, Atom *a, const Rule *r, const Environment *env) {
742 append(b, a->name);
743 if (a->count) {
744 append(b, "(");
745 for (int i = 0; i < a->count; i++) {
746 if (i) append(b, ", ");
747 append_term(b, a->terms[i], r, env);
748 }
749 append(b, ")");
750 }
751}
752
753static void append_literal(Buffer *b, Literal l, const Rule *r, const Environment *env) {
754 if (l.atom) {
755 if (l.negative) append(b, "not ");
756 append_atom(b, l.atom, r, env);
757 } else {
758 append_expr(b, l.left, r, env);
759 append(b, " ");
760 append(b, l.op);
761 append(b, " ");
762 append_expr(b, l.right, r, env);
763 }
764}

766static const char *rule_text(const Rule *r) {
767 Buffer b = {0};
768 if (r->head) {
769 append_atom(&b, r->head, r, NULL);
770 append(&b, " :- ");
771 } else append(&b, ":- ");
772 for (int i = 0; i < r->body.count; i++) {
773 if (i) append(&b, ", ");
774 append_literal(&b, r->body.items[i], r, NULL);
775 }
776 append(&b, ".");
777 const char *out = intern(b.text);
778 free(b.text);
779 return out;
780}

806static int row_index(const Rows *rows, const Tuple *tuple) {
807 int lo = 0, hi = rows->count;
808 while (lo < hi) {
809 int mid = lo + (hi - lo) / 2;
810 if (tuple_compare(&rows->items[mid].tuple, tuple) < 0) lo = mid + 1;
811 else hi = mid;
812 }
813 return lo;
814}

816static bool has_row(const Rows *rows, const Tuple *tuple) {
817 int at = row_index(rows, tuple);
818 return at < rows->count && !tuple_compare(&rows->items[at].tuple, tuple);
819}

821static void add_row(Rows *rows, Tuple tuple, int round, int rule) {
822 int at = row_index(rows, &tuple);
823 if (at < rows->count && !tuple_compare(&rows->items[at].tuple, &tuple)) {
824 if (rule < rows->items[at].rule) rows->items[at].rule = rule;
825 return;
826 }
827 if (rows->count == rows->capacity) rows->items = grow(rows->items, rows->count, &rows->capacity, sizeof(Row), false);
828 memmove(rows->items + at + 1, rows->items + at, (size_t)(rows->count - at) * sizeof(Row));
829 rows->items[at] = (Row){tuple, round, rule};
830 rows->count++;
831}

833static bool filters(const Rule *r, const Model *model, Environment env, bool complete) {
834 Value values[MAX_TERMS];
835 bool bound[MAX_TERMS];
836 Environment scratch = {values, bound};
837 for (int i = 0; i < r->body.count; i++) {
838 Literal l = r->body.items[i];
839 if (l.atom && !l.negative) continue;
840 if (!complete) {
841 Variables needed = {0};
842 literal_variables(l, &needed, true);
843 bool ready = true;
844 for (int j = 0; j < needed.count; j++) if (!env.bound[needed.ids[j]]) ready = false;
845 if (!ready) continue;
846 }
847 if (!l.atom) {
848 if (!comparison(l, env)) return false;
849 } else {
850 const Rows *rows = &model->predicates[predicate_index(l.atom->name)];
851 for (int j = 0; j < rows->count; j++) if (match(l.atom, &rows->items[j].tuple, env, scratch, r->variable_count)) return false;
852 }
853 }
854 return true;
855}

861static void add_environment(Environments *all, Environment env) {
862 if (all->count == all->capacity) {

864 Environment *items = work_allocate((size_t)next * sizeof(Environment));
865 if (all->count) memcpy(items, all->items, (size_t)all->count * sizeof(Environment));
866 all->items = items;
867 all->capacity = next;
868 }
869 all->items[all->count++] = env;

871typedef struct EnvEntry {
872 Environment env;
873 struct EnvEntry *next;
874} EnvEntry;

876static unsigned long environment_hash(Environment env, int count) {
877 unsigned long hash = 5381;
878 for (int i = 0; i < count; i++) {
879 hash = hash * 33 + env.bound[i];
880 if (!env.bound[i]) continue;
881 Value v = env.values[i];
882 hash = hash * 33 + !!v.symbol;
883 if (v.symbol) for (const char *p = v.symbol; *p; p++) hash = hash * 33 + (unsigned char)*p;
884 else hash = hash * 33 + (unsigned long)v.number;
885 }
886 return hash;
887}
888
889static bool insert_environment(EnvEntry **buckets, Environment env, int count, Environments *out) {
890 size_t bucket = environment_hash(env, count) % 4093;
891 for (EnvEntry *e = buckets[bucket]; e; e = e->next) {
892 bool same = true;
893 for (int i = 0; i < count; i++) if (env.bound[i] != e->env.bound[i] || (env.bound[i] && value_compare(env.values[i], e->env.values[i]))) {
894 same = false;
895 break;
896 }
897 if (same) return false;
898 }
899 EnvEntry *e = work_allocate(sizeof *e);
900 e->env = clone_environment(env, count);
901 e->next = buckets[bucket];
902 buckets[bucket] = e;
903 add_environment(out, e->env);
904 return true;
905}

908static int environment_compare(const void *a, const void *b) {
909 const Environment *x = a, *y = b;
910 for (int i = 0; i < sort_variables.count; i++) {
911 int v = sort_variables.ids[i], c = value_compare(x->values[v], y->values[v]);
912 if (c) return c;
913 }
914 return 0;
915}

933static void visit_solutions(const Rule *r, const Model *model, int before,
934 const int *positives, int count, int at, Environment env,
935 EnvEntry **answers, Environments *out) {
936 if (!filters(r, model, env, at == count)) return;
937 if (at == count) {
938 insert_environment(answers, env, r->variable_count, out);
939 return;
940 }
941 Atom *atom = r->body.items[positives[at]].atom;
942 const Rows *rows = &model->predicates[predicate_index(atom->name)];
943 size_t bucket_count = (size_t)rows->count * 2 + 1;
944 Allocation *frame = NULL;
945 BindingEntry **seen = take(&frame, bucket_count * sizeof(BindingEntry *));
946 Value values[MAX_TERMS];
947 bool bound[MAX_TERMS];
948 Environment candidate = {values, bound};
949 for (int i = 0; i < rows->count; i++) {
950 if (before >= 0 && rows->items[i].round >= before) continue;
951 if (!match(atom, &rows->items[i].tuple, env, candidate, r->variable_count)) continue;
952 Tuple key = {0};
953 for (int j = 0; j < atom->count; j++) {
954 int variable = atom->terms[j].variable;
955 if (variable >= 0) key.values[key.count++] = candidate.values[variable];
956 }
957 size_t bucket = binding_hash(&key) % bucket_count;
958 bool duplicate = false;
959 for (BindingEntry *entry = seen[bucket]; entry; entry = entry->next) {
960 if (!tuple_compare(&entry->key, &key)) {
961 duplicate = true;
962 break;
963 }
964 }
965 if (duplicate) continue;
966 BindingEntry *entry = take(&frame, sizeof *entry);
967 *entry = (BindingEntry){key, seen[bucket]};
968 seen[bucket] = entry;
969 visit_solutions(r, model, before, positives, count, at + 1, candidate, answers, out);
970 }
971 clear(&frame);
972}

974static Environments solutions(const Rule *r, const Model *model, int before) {
975 int positives[MAX_TERMS], count = 0;
976 for (int i = 0; i < r->body.count; i++) if (r->body.items[i].atom && !r->body.items[i].negative) positives[count++] = i;
977 for (int i = 1; i < count; i++) {
978 int v = positives[i], j = i;
979 while (j && model->predicates[predicate_index(r->body.items[positives[j - 1]].atom->name)].count > model->predicates[predicate_index(r->body.items[v].atom->name)].count) {
980 positives[j] = positives[j - 1];
981 j--;
982 }
983 positives[j] = v;
984 }
985 Environments out = {0};
986 EnvEntry **answers = work_allocate(4093 * sizeof(EnvEntry *));
987 visit_solutions(r, model, before, positives, count, 0,
988 empty_environment(r->variable_count), answers, &out);
989 sort_variables = rule_variables(r, false, true);
990 if (out.count > 1) qsort(out.items, (size_t)out.count, sizeof(Environment), environment_compare);
991 return out;
992}

994static Model *build_model(void) {
995 Model *model = allocate(sizeof *model);
996 Rule empty = {0};
997 ArityTable table = predicate_check(&empty, true, false);
998 strata(&table, NULL, model->levels);
999 model->maximum = -1;
1000 for (int p = 0; p < state.predicate_count; p++) {
1001 if (model->levels[p] > model->maximum) model->maximum = model->levels[p];
1002 Facts facts = state.predicates[p].facts;
1003 for (int i = 0; i < facts.count; i++) add_row(&model->predicates[p], *facts.items[i], 0, 0);
1004 }
1005 int serial = 0;
1006 for (int level = 0; level <= model->maximum; level++) while (true) {
1007 Rows staged[MAX_PREDICATES] = {0};
1008 int added = 0;
1009 for (int i = 0; i < state.rule_count; i++) {
1010 Rule *r = state.rules[i];
1011 if (!r->head) continue;
1012 int p = predicate_index(r->head->name);
1013 if (model->levels[p] != level) continue;
1014 Allocation *checkpoint = workspace;
1015 Environments answers = solutions(r, model, -1);
1016 for (int j = 0; j < answers.count; j++) {
1017 Tuple tuple = atom_tuple(r->head, answers.items[j]);
1018 if (!has_row(&model->predicates[p], &tuple)) add_row(&staged[p], tuple, 0, r->number);
1019 }
1020 restore_workspace(checkpoint);
1021 }
1022 for (int p = 0; p < state.predicate_count; p++) added += staged[p].count;
1023 if (!added) break;
1024 serial++;
1025 model->rounds[level]++;
1026 model->added[level] += added;
1027 for (int p = 0; p < state.predicate_count; p++) for (int i = 0; i < staged[p].count; i++) add_row(&model->predicates[p], staged[p].items[i].tuple, serial, staged[p].items[i].rule);
1028 }
1029 return model;
1030}

1050static void proof(int predicate, const Tuple *tuple, const Model *model, int depth) {
1051 const Rows *rows = &model->predicates[predicate];
1052 Row row = rows->items[row_index(rows, tuple)];
1053 if (!row.round) {
1054 indent(depth);
1055 tuple_text(state.predicates[predicate].name, tuple);
1056 puts(" [fact]");
1057 return;
1058 }
1059 for (int ri = 0; ri < state.rule_count; ri++) {
1060 Rule *r = state.rules[ri];
1061 if (!r->head || !equal(r->head->name, state.predicates[predicate].name)) continue;
1062 Allocation *checkpoint = workspace;
1063 Environments choices = solutions(r, model, row.round);
1064 sort_variables = rule_variables(r, true, true);
1065 if (choices.count > 1) qsort(choices.items, (size_t)choices.count, sizeof(Environment), environment_compare);
1066 for (int i = 0; i < choices.count; i++) {
1067 Environment env = choices.items[i];
1068 Tuple head = atom_tuple(r->head, env);
1069 if (tuple_compare(&head, tuple)) continue;
1070 indent(depth);
1071 tuple_text(state.predicates[predicate].name, tuple);
1072 printf(" [rule %d]\n", r->number);
1073 for (int li = 0; li < r->body.count; li++) {
1074 Literal l = r->body.items[li];
1075 if (l.atom && !l.negative) {
1076 int p = predicate_index(l.atom->name);
1077 const Rows *premises = &model->predicates[p];
1078 Value values[MAX_TERMS];
1079 bool bound[MAX_TERMS];
1080 Environment scratch = {values, bound};
1081 for (int j = 0; j < premises->count; j++) if (premises->items[j].round < row.round && match(l.atom, &premises->items[j].tuple, env, scratch, r->variable_count)) {
1082 proof(p, &premises->items[j].tuple, model, depth + 1);
1083 break;
1084 }
1085 } else {
1086 Buffer b = {0};
1087 append_literal(&b, l, r, &env);
1088 indent(depth + 1);
1089 puts(b.text);
1090 free(b.text);
1091 }
1092 }
1093 restore_workspace(checkpoint);
1094 return;
1095 }
1096 restore_workspace(checkpoint);
1097 }

1110static void process(Statement s) {
1111 Rule *r = s.rule;
1112 if (equal(s.kind, "fact") || equal(s.kind, "rule")) {
1113 ArityTable arities = predicate_check(r, true, true);
1114 safety(r, -1);
1115 if (equal(s.kind, "rule")) {
1116 const char *notation = rule_text(r);
1117 for (int i = 0; i < state.rule_count; i++) if (equal(notation, rule_text(state.rules[i]))) fail("duplicate of rule %d", state.rules[i]->number);
1118 int levels[252];
1119 if (r->head && !strata(&arities, r, levels)) fail("negative cycle through %s", r->head->name);
1120 } else {
1121 int p = predicate_index(r->head->name);
1122 Tuple tuple = atom_tuple(r->head, (Environment){0});
1123 if (p >= 0) for (int i = 0; i < state.predicates[p].facts.count; i++) if (!tuple_compare(state.predicates[p].facts.items[i], &tuple)) return;
1124 }
1125 snapshot();
1126 for (int i = state.predicate_count; i < arities.count; i++) state.predicates[state.predicate_count++] = (Predicate){.name = arities.names[i], .arity = arities.arities[i]};
1127 if (equal(s.kind, "fact")) add_fact(&state.predicates[predicate_index(r->head->name)].facts, atom_tuple(r->head, (Environment){0}));
1128 else {
1129 r->number = next_rule++;
1130 state.rules[state.rule_count++] = r;
1131 Variables occurrences = rule_variables(r, true, false), ordered = rule_variables(r, true, true);
1132 for (int i = 0; i < ordered.count; i++) {
1133 int v = ordered.ids[i], count = 0;
1134 for (int j = 0; j < occurrences.count; j++) count += occurrences.ids[j] == v;
1135 if (count == 1) printf("WARNING line %d: singleton variable %s\n", line_number, r->variables[v]);
1136 }
1137 }
1138 return;
1139 }
1140 if (equal(s.kind, "undo")) {
1141 if (!history) fail("nothing to undo");
1142 State previous = history->state;
1143 int line = history->line;
1144 history = history->previous;
1145 state = previous;
1146 printf("undid line %d\n", line);
1147 return;
1148 }
1149 if (equal(s.kind, "forget")) {
1150 int at = -1;
1151 for (int i = 0; i < state.rule_count; i++) if (state.rules[i]->number == s.number) at = i;
1152 if (at < 0) fail("no rule %lld", s.number);
1153 snapshot();
1154 memmove(state.rules + at, state.rules + at + 1, (size_t)(state.rule_count - at - 1) * sizeof(Rule *));
1155 state.rule_count--;
1156 return;
1157 }
1158 if (listed(s.kind, "retract why")) predicate_check(r, false, true);
1159 if (equal(s.kind, "retract")) {
1160 int p = predicate_index(r->head->name);
1161 Facts *facts = &state.predicates[p].facts;
1162 int removed = 0;
1163 bool *remove = allocate((size_t)facts->count * sizeof(bool));
1164 Environment empty = empty_environment(r->variable_count);
1165 Value values[MAX_TERMS];
1166 bool bound[MAX_TERMS];
1167 Environment scratch = {values, bound};
1168 for (int i = 0; i < facts->count; i++) {
1169 remove[i] = match(r->head, facts->items[i], empty, scratch, r->variable_count);
1170 removed += remove[i];
1171 }
1172 if (!removed) fail("no matching facts");
1173 snapshot();
1174 int count = 0;
1175 for (int i = 0; i < facts->count; i++) if (!remove[i]) facts->items[count++] = facts->items[i];
1176 facts->count = count;
1177 printf("%d %s retracted\n", removed, removed == 1 ? "fact" : "facts");
1178 return;
1179 }
1180 if (listed(s.kind, "listing dump")) {
1181 if (s.name && predicate_index(s.name) < 0) fail("unknown predicate %s", s.name);
1182 if (equal(s.kind, "listing")) {
1183 bool any = false;
1184 for (int i = 0; i < state.rule_count; i++) if (!s.name || (state.rules[i]->head && equal(state.rules[i]->head->name, s.name))) {
1185 printf("%d: %s\n", state.rules[i]->number, rule_text(state.rules[i]));
1186 any = true;
1187 }
1188 if (!any) puts("no rules");
1189 return;
1190 }
1191 }
1192 if (equal(s.kind, "query")) {
1193 predicate_check(r, false, false);
1194 safety(r, s.aggregate_variable);
1195 }
1196 if (equal(s.kind, "why")) for (int i = 0; i < r->head->count; i++) if (r->head->terms[i].variable >= 0 || r->head->terms[i].anonymous) fail("why needs a ground atom");
1197 if (equal(s.kind, "strata")) {
1198 if (!state.predicate_count) {
1199 puts("no predicates");
1200 return;
1201 }
1202 Rule empty = {0};
1203 ArityTable table = predicate_check(&empty, true, false);
1204 int levels[MAX_PREDICATES];
1205 strata(&table, NULL, levels);
1206 bool reach[MAX_PREDICATES][MAX_PREDICATES] = {
1207 {false}
1208 };
1209 for (int i = 0; i < state.rule_count; i++) if (state.rules[i]->head) for (int j = 0; j < state.rules[i]->body.count; j++) {
1210 Literal l = state.rules[i]->body.items[j];
1211 if (l.atom) reach[predicate_index(state.rules[i]->head->name)][predicate_index(l.atom->name)] = true;
1212 }
1213 for (int k = 0; k < state.predicate_count; k++) for (int i = 0; i < state.predicate_count; i++) for (int j = 0; j < state.predicate_count; j++) reach[i][j] |= reach[i][k] && reach[k][j];
1214 int maximum = 0;
1215 for (int i = 0; i < state.predicate_count; i++) if (levels[i] > maximum) maximum = levels[i];
1216 order_predicates();
1217 for (int k = 0; k <= maximum; k++) {
1218 printf("stratum %d: ", k);
1219 bool first = true;
1220 for (int i = 0; i < state.predicate_count; i++) {
1221 int p = predicate_order[i];
1222 if (levels[p] == k) {
1223 if (!first) fputs(", ", stdout);
1224 printf("%s%s", state.predicates[p].name, reach[p][p] ? "*" : "");
1225 first = false;
1226 }
1227 }
1228 putchar('\n');
1229 }
1230 return;
1231 }
1232 Model *model = build_model();
1233 if (equal(s.kind, "query")) {
1234 Environments answers = solutions(r, model, -1);
