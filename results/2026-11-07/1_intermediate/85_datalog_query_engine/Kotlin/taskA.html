7private data class Value(val integer: Long? = null, val symbol: String = "") : Comparable<Value> {
8 override fun compareTo(other: Value): Int = when {
9 integer != null && other.integer != null -> integer.compareTo(other.integer)
10 integer != null -> -1
11 other.integer != null -> 1
12 else -> symbol.compareTo(other.symbol)
13 }
14 fun text(): String = integer?.toString() ?: if (symbol.matches(Regex("[a-z][A-Za-z0-9_]*")) && symbol !in reserved) symbol else "\"$symbol\""
15}

20private val tupleOrder = Comparator<Tuple> {
21 a, b ->
22 var result = 0
23 for (i in a.indices) {
24 result = a[i].compareTo(b[i])
25 if (result != 0) break
26 }
27 result
28}
29private data class Term(val kind: String, val value: Value = Value(), val name: String = "") {
30 fun text(env: Environment? = null): String = when (kind) {
31 "constant" -> value.text()
32 "variable" -> env?.get(name)?.text() ?: name
33 else -> "_"
34 }
35 fun value(env: Environment): Value? = when (kind) {
36 "constant" -> value
37 "variable" -> env[name]
38 else -> null
39 }
40 fun variables() = if (kind == "variable") listOf(name) else emptyList()
41}
42private data class Expression(val kind: String, val term: Term? = null, val left: Expression? = null, val right: Expression? = null) {
43 fun text(env: Environment? = null): String = when (kind) {
44 "term" -> term!!.text(env)
45 "parentheses" -> "(${left!!.text(env)})"
46 else -> "${left!!.text(env)} $kind ${right!!.text(env)}"
47 }
48 fun variables(): List<String> = if (kind == "term") term!!.variables() else left!!.variables() + (right?.variables() ?: emptyList())
49
50 fun anonymous(): Boolean = if (kind == "term") term!!.kind == "anonymous" else left!!.anonymous() || right?.anonymous() == true
51
52 fun value(env: Environment): Value? {
53 if (kind == "term") return term!!.value(env)
54 if (kind == "parentheses") return left!!.value(env)
55 val a = left!!.value(env)?.integer ?: return null
56 val b = right!!.value(env)?.integer ?: return null
57 val n = when (kind) {
58 "+" -> a + b
59 "-" -> a - b
60 "*" -> a * b
61 else -> {
62 if (b == 0L) return null
63 val q = a / b - if (a % b != 0L && (a < 0) != (b < 0)) 1 else 0
64 if (kind == "/") q else a - b * q
65 }
66 }
67 return Value(n)
68 }
69}
70private data class Atom(val name: String, val terms: List<Term>) {
71 fun text(env: Environment? = null) = name + if (terms.isEmpty()) "" else terms.joinToString(", ", "(", ")") {
72 it.text(env)
73 }
74 fun variables() = terms.flatMap {
75 it.variables()
76 }
77 fun tuple(env: Environment) = terms.map {
78 it.value(env)!!
79 }
80 fun matching(tuple: Tuple, env: Environment): Environment? {
81 val result = env.toMutableMap()
82 for ((term, value) in terms.zip(tuple)) {
83 if (term.kind == "constant" && term.value != value) return null
84 if (term.kind == "variable") {
85 if (term.name in result && result[term.name] != value) return null
86 result[term.name] = value
87 }
88 }
89 return result
90 }
91}
92private fun tupleText(name: String, tuple: Tuple) = Atom(name, tuple.map {
93 Term("constant", it)
94}).text()

96private data class Literal(val atom: Atom? = null, val negative: Boolean = false, val op: String = "", val left: Expression? = null, val right: Expression? = null) {
97 val positive get() = atom != null && !negative
98
99 fun variables() = atom?.variables() ?: (left!!.variables() + right!!.variables())
100
101 fun text(env: Environment? = null) = if (atom != null) (if (negative) "not " else "") + atom.text(env) else "${left!!.text(env)} $op ${right!!.text(env)}"
102
103 fun holds(env: Environment, data: Database): Boolean {
104 if (atom != null) return !negative || data.getValue(atom.name).none {
105 atom.matching(it, env) != null
106 }
107 val a = left!!.value(env) ?: return false
108 val b = right!!.value(env) ?: return false
109 if (op == "=") return a == b
110 if (op == "!=") return a != b
111 val x = a.integer ?: return false
112 val y = b.integer ?: return false
113 return when (op) {
114 "<" -> x < y
115 "<=" -> x <= y
116 ">" -> x > y
117 else -> x >= y
118 }
119 }
120}
121private data class Rule(val number: Int = 0, val head: Atom? = null, val body: List<Literal>) {
122 fun text() = (head?.let {
123 it.text() + " "
124 } ?: "") + ":- " + body.joinToString(", ") {
125 it.text()
126 } + "."
127
128 fun variables() = (head?.variables() ?: emptyList()) + body.flatMap {
129 it.variables()
130 }
131 fun atoms() = listOfNotNull(head) + body.mapNotNull {
132 it.atom
133 }
134}
135private data class Token(val kind: String, val text: String)

137private fun wordCharacter(c: Char) = c in 'a'..'z' || c in 'A'..'Z' || c in '0'..'9' || c == '_'
138
139private fun lex(line: String): List<Token> {
140 val result = mutableListOf<Token>()
141 var at = 0
142 while (at < line.length) {
143 if (line[at] == ' ') {
144 at++
145 continue
146 }
147 if (line[at] == '%') break
148 val start = at
149 val kind: String
150 if (line[at] == '"') {
151 at++
152 while (at < line.length && line[at] != '"') at++
153 if (at == line.length) fail("unterminated string")
154 at++
155 kind = "string"
156 } else if (at + 1 < line.length && line.substring(at, at + 2) in setOf(":-", "?-", "!=", "<=", ">=")) {
157 at += 2
158 kind = line.substring(start, at)
159 } else if (line[at] in "(),.:=<>+-*/") {
160 kind = line[at++].toString()
161 } else if (wordCharacter(line[at])) {
162 at++
163 while (at < line.length && wordCharacter(line[at])) at++
164 val word = line.substring(start, at)
165 kind = when {
166 word[0] in 'a'..'z' -> if (word in reserved) word else "name"
167 word[0] in 'A'..'Z' -> "variable"
168 word == "_" -> "anonymous"
169 word.all {
170 it in '0'..'9'
171 } -> {
172 if (word.toLong() > 1000000000L) fail("integer out of range '$word'")
173 "integer"
174 } else -> fail("bad token '$word'")
175 }
176 } else fail("bad character '${line[at]}'")
177 result.add(Token(kind, line.substring(start, at)))
178 }
179 return result
180}
181private data class Statement(val kind: String, val rule: Rule? = null, val atom: Atom? = null, val aggregate: String = "", val argument: String = "")

183private class Parser(val tokens: List<Token>) {
184 var at = 0
185 val kind get() = tokens.getOrNull(at)?.kind ?: "end"
186
187 fun error(): Nothing = fail(tokens.getOrNull(at)?.let {
188 "syntax error at '${it.text}'"
189 } ?: "syntax error at end of line")
190
191 fun take(expected: String): Token {
192 if (kind != expected) error()
193 return tokens[at++]
194 }
195 fun term(): Term = when (kind) {
196 "name" -> Term("constant", Value(symbol = take("name").text))
197 "string" -> Term("constant", Value(symbol = take("string").text.drop(1).dropLast(1)))
198 "integer" -> Term("constant", Value(take("integer").text.toLong()))
199 "-" -> {
200 take("-")
201 Term("constant", Value(-take("integer").text.toLong()))
202 }
203 "variable" -> Term("variable", name = take("variable").text)
204 "anonymous" -> {
205 take("anonymous")
206 Term("anonymous")
207 } else -> error()
208 }
209 fun atom(): Atom {
210 val name = take("name").text
211 val terms = mutableListOf<Term>()
212 if (kind == "(") {
213 take("(")
214 terms.add(term())
215 while (kind == ",") {
216 take(",")
217 terms.add(term())
218 }
219 take(")")
220 }
221 return Atom(name, terms)
222 }
223 fun factor(): Expression = if (kind == "(") {
224 take("(")
225 val e = expression()
226 take(")")
227 Expression("parentheses", left = e)
228 } else Expression("term", term())
229
230 fun product(): Expression {
231 var e = factor()
232 while (kind in setOf("*", "/", "mod")) {
233 val op = tokens[at++].text
234 e = Expression(op, left = e, right = factor())
235 }
236 return e
237 }
238 fun expression(): Expression {
239 var e = product()
240 while (kind in setOf("+", "-")) {
241 val op = tokens[at++].text
242 e = Expression(op, left = e, right = product())
243 }
244 return e
245 }
246 fun literal(): Literal {
247 if (kind == "not") {
248 take("not")
249 return Literal(atom(), true)
250 }
251 val following = tokens.getOrNull(at + 1)?.kind ?: "end"
252 if (kind == "name" && following !in setOf("=", "!=", "<", "<=", ">", ">=", "+", "-", "*", "/", "mod")) return Literal(atom())
253 val left = expression()
254 if (kind !in setOf("=", "!=", "<", "<=", ">", ">=")) error()
255 val op = tokens[at++].text
256 return Literal(op = op, left = left, right = expression())
257 }
258 fun body(): List<Literal> {
259 val result = mutableListOf(literal())
260 while (kind == ",") {
261 take(",")
262 result.add(literal())
263 }
264 return result
265 }
266 fun statement(): Statement {
267 val result = when (kind) {
268 "name" -> {
269 val head = atom()
270 val body = if (kind == ":-") {
271 take(":-")
272 body()
273 } else emptyList()
274 Statement("clause", Rule(head = head, body = body))
275 }
276 ":-" -> {
277 take(":-")
278 Statement("clause", Rule(body = body()))
279 }
280 "?-" -> {
281 take("?-")
282 var aggregate = ""
283 var variable = ""
284 if (kind in setOf("count", "sum", "min", "max")) {
285 aggregate = tokens[at++].text
286 if (aggregate != "count") {
287 variable = take("variable").text
288 take(":")
289 }
290 }
291 Statement("query", Rule(body = body()), aggregate = aggregate, argument = variable)
292 }
293 "retract", "why" -> {
294 val command = tokens[at++].text
295 Statement(command, atom = atom())
296 }
297 "dump", "forget", "listing", "strata", "stats", "check", "predicates", "undo", "halt" -> {
298 val command = tokens[at++].text
299 val argument = when {
300 command == "dump" -> take("name").text
301 command == "forget" -> take("integer").text
302 command == "listing" && kind == "name" -> take("name").text
303 else -> ""
304 }
305 Statement(command, argument = argument)
306 } else -> error()
307 }
308 take(".")
309 if (kind != "end") error()
310 return result
311 }
312}
313private data class State(val predicates: Map<String, Int> = emptyMap(), val facts: Database = emptyMap(), val rules: List<Rule> = emptyList())

315private fun safety(rule: Rule, aggregate: String = "") {
316 if (rule.head?.terms?.any {
317 it.kind == "anonymous"
318 } == true) fail("anonymous variable in head")
319 if (rule.body.any {
320 it.atom == null && (it.left!!.anonymous() || it.right!!.anonymous())
321 })
322 fail("anonymous variable in comparison")
323 val bound = rule.body.filter {
324 it.positive
325 }.flatMap {
326 it.variables()
327 }.toSet()
328 val required = (rule.head?.variables() ?: if (aggregate.isEmpty()) emptyList() else listOf(aggregate)) + rule.body.filter {
329 !it.positive
330 }.flatMap {
331 it.variables()
332 }
333 for (name in required) if (name !in bound) fail("unsafe variable $name")
334}
335private fun strata(state: State): Map<String, Int>? {
336 val result = state.predicates.keys.associateWith {
337 0
338 }.toMutableMap()
339 if (result.isEmpty()) return result
340 repeat(result.size) {
341 var changed = false
342 for (r in state.rules) if (r.head != null) for (l in r.body) if (l.atom != null) {
343 val value = result.getValue(l.atom.name) + if (l.negative) 1 else 0
344 if (value > result.getValue(r.head.name)) {
345 result[r.head.name] = value
346 changed = true
347 }
348 }
349 if (!changed) return result
350 }
351 return null
352}
353private fun solutions(body: List<Literal>, data: Database, derivations: Derivations, before: Int? = null): Set<Environment> {
354 val output = mutableSetOf<Environment>()
355
356 fun visit(remaining: List<Int>, env: Environment) {
357 for (l in body) if (!l.positive && l.variables().all {
358 it in env
359 } && !l.holds(env, data)) return
360 if (remaining.isEmpty()) {
361 output.add(env)
362 return
363 }
364 var best = -1
365 var candidates = emptySet<Environment>()
366 for (i in remaining) {
367 val a = body[i].atom!!
368 val possible = data.getValue(a.name).asSequence().filter {
369 before == null || derivations.getValue(a.name).getValue(it).first < before
370 }.mapNotNull {
371 a.matching(it, env)
372 }.toSet()
373 if (best < 0 || possible.size < candidates.size) {
374 best = i
375 candidates = possible
376 }
377 }
378 val rest = remaining.filter {
379 it != best
380 }
381 for (next in candidates) visit(rest, next)
382 }
383 visit(body.indices.filter {
384 body[it].positive
385 },
386 emptyMap())
387 return output
388}
389private class Model(val state: State) {
390 val levels = strata(state)!!
391 val data = state.facts.mapValues {
392 it.value.toMutableSet()
393 }.toMutableMap()
394 val derivations = state.facts.mapValues {
395 (_, ts) -> ts.associateWith {
396 0 to 0
397 }.toMutableMap()
398 }.toMutableMap()
399 val rounds = IntArray((levels.values.maxOrNull() ?: 0) + 1)
400 val additions = IntArray(rounds.size)
401 init {
402 var serial = 0
403 for (k in rounds.indices) while (true) {
404 val added = mutableMapOf<Pair<String, Tuple>, Int>()
405 for (r in state.rules) {
406 val h = r.head ?: continue
407 if (levels[h.name] != k) continue
408 for (env in solutions(r.body, data, derivations)) {
409 val tuple = h.tuple(env)
410 if (tuple !in data.getValue(h.name)) {
411 val key = h.name to tuple
412 added[key] = minOf(added[key] ?: r.number, r.number)
413 }
414 }
415 }
416 if (added.isEmpty()) break
417 serial++
418 rounds[k]++
419 additions[k] += added.size
420 for ((key, rule) in added) {
421 data.getValue(key.first).add(key.second)
422 derivations.getValue(key.first)[key.second] = serial to rule
423 }
424 }
425 }
426 fun answers(body: List<Literal>): Pair<List<String>, List<Tuple>> {
427 val names = body.flatMap {
428 it.variables()
429 }.distinct()
430 val answers = solutions(body, data, derivations).map {
431 env -> names.map {
432 env.getValue(it)
433 }
434 }.distinct().sortedWith(tupleOrder)
435 return names to answers
436 }
437 fun proof(name: String, tuple: Tuple, depth: Int, output: MutableList<String>) {
438 val round = derivations.getValue(name).getValue(tuple).first
439 val prefix = " ".repeat(depth) + tupleText(name, tuple)
440 if (round == 0) {
441 output.add("$prefix [fact]")
442 return
443 }
444 for (r in state.rules.filter {
445 it.head?.name == name
446 }) {
447 val order = r.variables().distinct()
448 val candidates = solutions(r.body, data, derivations, round).filter {
449 r.head!!.tuple(it) == tuple
450 }.sortedWith {
451 a, b -> tupleOrder.compare(order.map {
452 a.getValue(it)
453 },
454 order.map {
455 b.getValue(it)
456 })
457 }
458 val env = candidates.firstOrNull() ?: continue
459 output.add("$prefix [rule ${r.number}]")
460 for (l in r.body) {
461 if (l.positive) {
462 val a = l.atom!!
463 val child = data.getValue(a.name).filter {
464 derivations.getValue(a.name).getValue(it).first < round && a.matching(it, env) != null
465 }.minWith(tupleOrder)
466 proof(a.name, child, depth + 1, output)
467 } else output.add(" ".repeat(depth + 1) + l.text(env))
468 }
469 return
470 }
471 }
472}
473private fun answerText(names: List<String>, tuple: Tuple) = names.zip(tuple).joinToString(", ") {
474 (n, v) -> "$n = ${v.text()}"
475}
476private class Engine {
477 var state = State()
478 var nextRule = 1
479 val history = mutableListOf<Pair<State, Int>>()
480
481 fun predicateCheck(atoms: List<Atom>, register: Boolean): Map<String, Int> {
482 val predicates = state.predicates.toMutableMap()
483 for (a in atoms) {
484 val arity = predicates[a.name]
485 if (arity != null && arity != a.terms.size) fail("arity mismatch for ${a.name}: expected $arity, got ${a.terms.size}")
486 if (arity == null) {
487 if (!register) fail("unknown predicate ${a.name}")
488 predicates[a.name] = a.terms.size
489 }
490 }
491 return predicates
492 }
493 fun save(line: Int) {
494 history.add(state to line)
495 }
496 fun clause(rule: Rule, line: Int): List<String> {
497 val predicates = predicateCheck(rule.atoms(), true)
498 safety(rule)
499 if (rule.body.isEmpty()) {
500 val h = rule.head!!
501 val tuple = h.tuple(emptyMap())
502 if (tuple !in (state.facts[h.name] ?: emptySet())) {
503 save(line)
504 state = state.copy(predicates = predicates, facts = state.facts + (h.name to ((state.facts[h.name] ?: emptySet()) + setOf(tuple))))
505 }
506 return emptyList()
507 }
508 state.rules.firstOrNull {
509 it.text() == rule.text()
510 }?.let {
511 fail("duplicate of rule ${it.number}")
512 }
513 val numbered = rule.copy(number = nextRule)
514 val proposed = State(predicates, predicates.keys.associateWith {
515 state.facts[it] ?: emptySet()
516 },
517 state.rules + numbered)
518 if (strata(proposed) == null) fail("negative cycle through ${rule.head!!.name}")
519 save(line)
520 state = proposed
521 nextRule++
522 val all = rule.variables()
523 return all.distinct().filter {
524 n -> all.count {
525 it == n
526 } == 1
527 }.map {
528 "WARNING line $line: singleton variable $it"
529 }
530 }
531 fun execute(statement: Statement, line: Int): List<String> {
532 val kind = statement.kind
533 if (kind == "clause") return clause(statement.rule!!, line)
534 if (kind == "query") {
535 val rule = statement.rule!!
536 predicateCheck(rule.atoms(), false)
537 safety(rule, statement.argument)
538 val (names, answers) = Model(state).answers(rule.body)
539 if (statement.aggregate == "count") return listOf(answers.size.toString())
540 if (statement.aggregate in setOf("sum", "min", "max")) {
541 val i = names.indexOf(statement.argument)
542 if (statement.aggregate == "sum") {
543 var total = 0L
544 for (row in answers) total += row[i].integer ?: fail("non-integer value in sum")
545 return listOf(total.toString())
546 }
547 val values = answers.map {
548 it[i]
549 }.sorted()
550 return listOf(if (values.isEmpty()) "no answers" else (if (statement.aggregate == "min") values.first() else values.last()).text())
551 }
552 if (names.isEmpty()) return listOf(if (answers.isEmpty()) "false." else "true.")
553 return answers.map {
554 answerText(names, it)
555 } + "${answers.size} ${if (answers.size == 1) "answer" else "answers"}"
556 }
557 if (kind == "retract" || kind == "why") {
558 val a = statement.atom!!
559 predicateCheck(listOf(a), false)
560 if (kind == "retract") {
561 val removed = state.facts.getValue(a.name).filter {
562 a.matching(it, emptyMap()) != null
563 }.toSet()
564 if (removed.isEmpty()) fail("no matching facts")
565 save(line)
566 state = state.copy(facts = state.facts + (a.name to (state.facts.getValue(a.name) - removed)))
567 return listOf("${removed.size} ${if (removed.size == 1) "fact" else "facts"} retracted")
568 }
569 if (a.terms.any {
570 it.kind != "constant"
571 })
572 fail("why needs a ground atom")
573 val model = Model(state)
574 val tuple = a.tuple(emptyMap())
575 if (tuple !in model.data.getValue(a.name)) return listOf("false.")
576 val output = mutableListOf<String>()
577 model.proof(a.name, tuple, 0, output)
578 return output
579 }
580 return command(kind, statement.argument, line)
581 }
582 fun command(command: String, argument: String, line: Int): List<String> {
583 if (command == "undo") {
584 if (history.isEmpty()) fail("nothing to undo")
585 val (previous, number) = history.removeAt(history.lastIndex)
586 state = previous
587 return listOf("undid line $number")
588 }
589 if (command == "forget") {
590 val n = argument.toInt()
591 if (state.rules.none {
592 it.number == n
593 })
594 fail("no rule $n")
595 save(line)
596 state = state.copy(rules = state.rules.filter {
597 it.number != n
598 })
599 return emptyList()
600 }
601 if (command in setOf("dump", "listing") && argument.isNotEmpty() && argument !in state.predicates) fail("unknown predicate $argument")
602 if (command == "listing") return state.rules.filter {
603 argument.isEmpty() || it.head?.name == argument
604 }.map {
605 "${it.number}: ${it.text()}"
606 }.ifEmpty {
607 listOf("no rules")
608 }
609 val model = Model(state)
610 val output = mutableListOf<String>()
611 when (command) {
612 "dump" -> {
613 val tuples = model.data.getValue(argument).sortedWith(tupleOrder)
614 output.addAll(tuples.map {
615 tupleText(argument, it) + "."
616 })
617 output.add("${tuples.size} ${if (tuples.size == 1) "tuple" else "tuples"}")
618 }
619 "strata" -> {
620 if (state.predicates.isEmpty()) return listOf("no predicates")
621 val edges = state.predicates.keys.associateWith {
622 p -> state.rules.filter {
623 it.head?.name == p
624 }.flatMap {
625 it.body.mapNotNull {
626 l -> l.atom?.name
627 }
628 }
629 }
630 fun recursive(p: String): Boolean {
631 val visited = mutableSetOf<String>()
632
633 fun visit(n: String): Boolean = visited.add(n) && edges.getValue(n).any {
634 it == p || visit(it)
635 }
636 return visit(p)
637 }
638 for (k in model.rounds.indices) output.add("stratum $k: " + model.levels.keys.filter {
639 model.levels[it] == k
640 }.sorted().joinToString(", ") {
641 it + if (recursive(it)) "*" else ""
642 })
643 }
644 "stats" -> {
645 val constraints = state.rules.count {
646 it.head == null
647 }
648 output.add("facts ${state.facts.values.sumOf { it.size }}, rules ${state.rules.size - constraints}, constraints $constraints, tuples ${model.data.values.sumOf { it.size }}")
649 if (state.predicates.isNotEmpty()) for (k in model.rounds.indices) output.add("stratum $k: rounds ${model.rounds[k]}, new ${model.additions[k]}")
650 for (r in state.rules.filter {
651 it.head != null
652 })
653 output.add("rule ${r.number}: new ${model.derivations.values.sumOf { d -> d.values.count { it.second == r.number } }}")
654 }
655 "predicates" -> {
656 if (state.predicates.isEmpty()) return listOf("no predicates")
657 for (p in state.predicates.keys.sorted()) output.add("$p/${state.predicates.getValue(p)}: facts ${state.facts.getValue(p).size}, rules ${state.rules.count { it.head?.name == p }}, tuples ${model.data.getValue(p).size}")
658 }
659 "check" -> {
660 for (r in state.rules.filter {
661 it.head == null
662 }) {
663 val (names, answers) = model.answers(r.body)
664 output.add(if (answers.isEmpty()) "constraint ${r.number}: ok" else "constraint ${r.number}: ${answers.size} ${if (answers.size == 1) "violation" else "violations"}")
665 if (names.isNotEmpty()) output.addAll(answers.map {
666 " " + answerText(names, it)
667 })
668 }
669 if (output.isEmpty()) output.add("no constraints")
670 }
671 }
672 return output
673 }
674}
675fun main() {
676 val engine = Engine()
677 var lineNumber = 0
678 while (true) {
679 val line = readLine() ?: break
680 lineNumber++
681 try {
682 val tokens = lex(line)
683 if (tokens.isEmpty()) continue
684 val statement = Parser(tokens).statement()
685 if (statement.kind == "halt") break
686 engine.execute(statement, lineNumber).forEach(::println)
687 } catch (e: Problem) {
688 println("ERROR line $lineNumber: ${e.reason}")
689 }
690 }
