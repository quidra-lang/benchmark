31 n <- length(items)
32 if (n <= 1L) return(items)

81new_parser <- function(line) {
82 tokens <- list()
83 i <- 1L
84 length <- nchar(line)
85 add_token <- function(kind, text, value = NULL) tokens[[length(tokens) + 1L]] <<- list(kind = kind, text = text, value = value)
86 character_at <- function(i) substr(line, i, i)
87 while (i <= length) {
88 c <- character_at(i)
89 if (c == " ") {
90 i <- i + 1L
91 } else if (c == "%") {
92 break
93 } else if (c == '"') {
94 finish <- regexpr('"', substring(line, i + 1L), fixed = TRUE)[[1]]
95 if (finish < 0L) datalog_error("unterminated string")
96 finish <- i + finish
97 add_token("string", substr(line, i, finish), substr(line, i + 1L, finish - 1L))
98 i <- finish + 1L
99 } else if (substr(line, i, i + 1L) %in% c(":-", "?-", "!=", "<=", ">=")) {
100 text <- substr(line, i, i + 1L)
101 add_token(text, text)
102 i <- i + 2L
103 } else if (c %in% c("(", ")", ",", ".", ":", "=", "<", ">", "+", "-", "*", "/")) {
104 add_token(c, c)
105 i <- i + 1L
106 } else if (grepl("^[A-Za-z0-9_]$", c)) {
107 start <- i
108 i <- i + 1L
109 while (i <= length && grepl("^[A-Za-z0-9_]$", character_at(i))) i <- i + 1L
110 text <- substr(line, start, i - 1L)
111 first <- substr(text, 1L, 1L)
112 if (grepl("^[a-z]$", first)) {
113 add_token(if (text %in% reserved) text else "name", text, text)
114 } else if (grepl("^[A-Z]$", first)) {
115 add_token("variable", text, text)
116 } else if (text == "_") {
117 add_token("anonymous", text)
118 } else if (grepl("^[0-9]+$", text)) {
119 value <- as.numeric(text)
120 if (value > 1000000000) datalog_error(paste0("integer out of range '", text, "'"))
121 add_token("integer", text, value)
122 } else {
123 datalog_error(paste0("bad token '", text, "'"))
124 }
125 } else {
126 datalog_error(paste0("bad character '", c, "'"))
127 }
128 }

351 divmod <- function(a, b) {
352 q <- floor(abs(a) / abs(b))
353 r <- abs(a) - q * abs(b)
354 if (r < 0) {
355 q <- q - 1
356 r <- r + abs(b)
357 } else if (r >= abs(b)) {
358 q <- q + 1
359 r <- r - abs(b)
360 }
361 if ((a < 0) != (b < 0)) {
362 q <- -q
363 if (r != 0) {
364 q <- q - 1
365 r <- abs(b) - r
366 }
367 }
368 c(q, if (b < 0) -r else r)
369 }
370 evaluate <- function(expression, environment) {
371 if (expression$kind == "term") {
372 term <- expression$term
373 return(if (term$kind == "constant") term$value else environment[[term$value]])
374 }
375 if (expression$kind == "group") return(evaluate(expression$child, environment))
376 a <- evaluate(expression$left, environment)
377 b <- evaluate(expression$right, environment)
378 if (!is.numeric(a) || !is.numeric(b) || ((expression$op %in% c("/", "mod")) && b == 0)) return(NULL)
379 switch(expression$op, "+" = a + b, "-" = a - b, "*" = a * b, "/" = divmod(a, b)[[1]], mod = divmod(a, b)[[2]])
380 }
381 comparison <- function(literal, environment) {
382 a <- evaluate(literal$left, environment)
383 b <- evaluate(literal$right, environment)
384 if (is.null(a) || is.null(b)) return(FALSE)
385 if (literal$op == "=") return(value_equal(a, b))
386 if (literal$op == "!=") return(!value_equal(a, b))
387 if (!is.numeric(a) || !is.numeric(b)) return(FALSE)
388 switch(literal$op, "<" = a < b, "<=" = a <= b, ">" = a > b, ">=" = a >= b)
389 }
390 solutions <- function(body, data, metadata = NULL, before = NULL) {
391 positives <- Filter(function(literal) literal$kind == "positive", body)
392 bound_at <- list()

396 }
397 }
398 filters <- rep(list(list()), length(positives) + 1L)

401 stage <- 0L

413 visit <- function(stage, environment) {

418 return(invisible(NULL))

524 execute <- function(statement, line) {
525 kind <- statement$kind
526 if (kind %in% c("fact", "rule", "constraint")) {
527 prospective <- predicate_check(atoms(statement), TRUE)
528 safety(statement)
529 if (kind == "fact") {
530 atom <- statement$head
531 tuple <- head_tuple(atom, list())
532 key <- tuple_key(tuple)
533 if (!is.null(facts_of(atom$name)[[key]])) return(character())
534 snapshot(line)
535 state$arities <- prospective
536 state$facts[[atom$name]][[key]] <- tuple
537 return(character())
538 }
539 notation <- rule_text(statement)
540 for (rule in state$rules) if (rule$notation == notation) datalog_error(paste0("duplicate of rule ", rule$number))
541 if (kind == "rule" && is.null(strata(prospective, c(state$rules, list(statement))))) datalog_error(paste0("negative cycle through ", statement$head$name))
542 snapshot(line)
543 statement$number <- state$next_rule
544 statement$notation <- notation
545 state$next_rule <- state$next_rule + 1L
546 state$rules[[length(state$rules) + 1L]] <- statement
547 state$arities <- prospective
548 occurrences <- variable_occurrences(statement$head, statement$body)
549 names <- unique(occurrences)
550 names <- names[vapply(names, function(name) sum(occurrences == name) == 1L, logical(1))]
551 return(paste0("WARNING line ", line, ": singleton variable ", names)[seq_along(names)])
552 }
553 if (kind == "query") {
554 predicate_check(atoms(statement))
555 safety(statement)
556 result <- answers(statement$body, model()$data)
557 variables <- result$variables
558 tuples <- result$tuples
559 aggregate <- statement$aggregate
560 if (!is.null(aggregate) && aggregate == "count") return(as.character(length(tuples)))
561 if (!is.null(aggregate) && aggregate %in% c("sum", "min", "max")) {
562 position <- match(statement$variable, variables)
563 values <- lapply(tuples, function(tuple) tuple[[position]])
564 if (aggregate == "sum") {
565 if (any(!vapply(values, is.numeric, logical(1)))) datalog_error("non-integer value in sum")
566 return(value_text(sum(unlist(values))))
567 }
568 if (length(values) == 0L) return("no answers")
569 values <- sort_items(values, value_compare)
570 return(value_text(if (aggregate == "min") values[[1]] else tail(values, 1L)[[1]]))
571 }
572 if (length(variables) == 0L) return(if (length(tuples) == 0L) "false." else "true.")
573 return(c(vapply(tuples, function(tuple) answer_text(variables, tuple), character(1)), paste0(length(tuples), if (length(tuples) == 1L) " answer" else " answers")))
574 }
575 if (kind %in% c("retract", "why")) {
576 atom <- statement$atom
577 predicate_check(list(atom))
578 if (kind == "retract") {
579 facts <- facts_of(atom$name)
580 removed <- names(Filter(function(tuple) !is.null(match_atom(atom, tuple, list())), facts))
581 if (length(removed) == 0L) datalog_error("no matching facts")
582 snapshot(line)
583 state$facts[[atom$name]][removed] <- NULL
584 return(paste0(length(removed), if (length(removed) == 1L) " fact retracted" else " facts retracted"))
585 }
586 if (any(vapply(atom$terms, function(term) term$kind != "constant", logical(1)))) datalog_error("why needs a ground atom")
587 tuple <- head_tuple(atom, list())
588 result <- model()
589 return(if (is.null(result$data[[atom$name]][[tuple_key(tuple)]])) "false." else proof(atom$name, tuple, result$data, result$metadata))
590 }
591 if (kind == "dump") {
592 name <- statement$name
593 if (is.null(state$arities[[name]])) datalog_error(paste0("unknown predicate ", name))
594 tuples <- sort_items(unname(model()$data[[name]]))
595 return(c(vapply(tuples, function(tuple) paste0(tuple_text(name, tuple), "."), character(1)), paste0(length(tuples), if (length(tuples) == 1L) " tuple" else " tuples")))
596 }
597 if (kind == "forget") {
598 indices <- which(vapply(state$rules, function(rule) rule$number == statement$number, logical(1)))
599 if (length(indices) == 0L) datalog_error(paste0("no rule ", statement$number))
600 snapshot(line)
601 state$rules <- state$rules[-indices[[1]]]
602 return(character())
603 }
604 if (kind == "undo") {
605 if (length(state$history) == 0L) datalog_error("nothing to undo")
606 saved <- tail(state$history, 1L)[[1]]
607 state$history <- head(state$history, -1L)
608 state$arities <- saved$arities
609 state$facts <- saved$facts
610 state$rules <- saved$rules
611 state$cache <- NULL
612 return(paste0("undid line ", saved$line))
613 }
614 if (kind == "listing") {
615 name <- statement$name
616 if (!is.null(name) && is.null(state$arities[[name]])) datalog_error(paste0("unknown predicate ", name))
617 rules <- Filter(function(rule) is.null(name) || (!is.null(rule$head) && rule$head$name == name), state$rules)
618 if (length(rules) == 0L) return("no rules")
619 return(vapply(rules, function(rule) paste0(rule$number, ": ", rule$notation), character(1)))
620 }
621 if (kind == "strata") {
622 if (length(state$arities) == 0L) return("no predicates")
623 levels <- strata()
624 dependencies <- setNames(rep(list(character()), length(state$arities)), names(state$arities))
625 for (rule in state$rules) if (!is.null(rule$head)) dependencies[[rule$head$name]] <- c(dependencies[[rule$head$name]], vapply(atoms(rule)[-1L], function(atom) atom$name, character(1)))
626 recursive <- setNames(rep(FALSE, length(dependencies)), names(dependencies))
627 for (name in names(dependencies)) {
628 seen <- character()
629 search <- function(current) {
630 for (target in dependencies[[current]]) {
631 if (target == name) return(TRUE)
632 if (!(target %in% seen)) {
633 seen <<- c(seen, target)
634 if (search(target)) return(TRUE)
635 }
636 }
637 FALSE
638 }
639 recursive[[name]] <- search(name)
640 }
641 return(vapply(seq_len(max(unlist(levels)) + 1L) - 1L, function(level) {
642 names <- sort(names(levels)[unlist(levels) == level], method = "radix")
643 displayed <- vapply(names, function(name) paste0(name, if (recursive[[name]]) "*" else ""), character(1))
644 paste0("stratum ", level, ": ", paste(displayed, collapse = ", "))
645 }, character(1)))
646 }
647 if (kind == "stats") {
648 result <- model()
649 rules <- Filter(function(rule) !is.null(rule$head), state$rules)
650 lines <- paste0("facts ", sum(vapply(state$facts, length, integer(1))), ", rules ", length(rules), ", constraints ", length(state$rules) - length(rules), ", tuples ", sum(vapply(result$data, length, integer(1))))
651 for (level in seq_along(result$rounds) - 1L) lines <- c(lines, paste0("stratum ", level, ": rounds ", result$rounds[[level + 1L]], ", new ", result$added[[level + 1L]]))
652 for (rule in rules) {
653 count <- sum(vapply(result$metadata, function(entries) sum(vapply(entries, function(entry) !is.na(entry[[2]]) && entry[[2]] == rule$number, logical(1))), integer(1)))
654 lines <- c(lines, paste0("rule ", rule$number, ": new ", count))
655 }
656 return(lines)
657 }
658 if (kind == "predicates") {
659 if (length(state$arities) == 0L) return("no predicates")
660 data <- model()$data
661 return(vapply(sort(names(state$arities), method = "radix"), function(name) {
662 count <- sum(vapply(state$rules, function(rule) !is.null(rule$head) && rule$head$name == name, logical(1)))
663 paste0(name, "/", state$arities[[name]], ": facts ", length(facts_of(name)), ", rules ", count, ", tuples ", length(data[[name]]))
664 }, character(1)))
665 }
666 constraints <- Filter(function(rule) is.null(rule$head), state$rules)
667 if (length(constraints) == 0L) return("no constraints")
668 data <- model()$data
669 lines <- character()
670 for (rule in constraints) {
671 result <- answers(rule$body, data)
672 count <- length(result$tuples)
673 if (count == 0L) {
674 lines <- c(lines, paste0("constraint ", rule$number, ": ok"))
675 } else {
676 lines <- c(lines, paste0("constraint ", rule$number, ": ", count, if (count == 1L) " violation" else " violations"))
677 if (length(result$variables) != 0L) for (tuple in result$tuples) lines <- c(lines, paste0(" ", answer_text(result$variables, tuple)))
678 }
679 }
680 lines
681 }
682 input <- file("stdin", open = "r")
683 on.exit(close(input))
684 number <- 0L
685 repeat {
686 lines <- readLines(input, n = 1L, warn = FALSE)
687 if (length(lines) == 0L) break
688 number <- number + 1L
689 statement <- tryCatch(new_parser(lines[[1]])$statement(), datalog_error = function(error) {
690 cat("ERROR line ", number, ": ", error$message, "\n", sep = "")
691 NULL
692 })
693 if (is.null(statement)) next
694 if (statement$kind == "halt") break
695 output <- tryCatch(execute(statement, number), datalog_error = function(error) paste0("ERROR line ", number, ": ", error$message))
696 for (text in output) cat(text, "\n", sep = "")
697 }
