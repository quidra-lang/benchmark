57function d_lex(line)
58 tokens = DToken[]
59 i = 1
60 while i <= length(line)
61 c = line[i]
62 if c == ' '
63 i += 1
64 elseif c == '%'
65 break
66 elseif c == '"'
67 finish = findnext('"', line, i + 1)
68 finish === nothing && d_fail("unterminated string")
69 push!(tokens, DToken("STRING", line[i:finish], line[i+1:finish-1]))
70 i = finish + 1
71 elseif i < length(line) && line[i:i+1] in (":-", "?-", "!=", "<=", ">=")
72 text = line[i:i+1]
73 push!(tokens, DToken(text, text, nothing))
74 i += 2
75 elseif c in "(),.:=<>+-*/"
76 push!(tokens, DToken(string(c), string(c), nothing))
77 i += 1
78 elseif d_word(c)
79 start = i
80 while i <= length(line) && d_word(line[i])
81 i += 1
82 end
83 text = line[start:i-1]
84 if 'a' <= c <= 'z'
85 kind = text in D_RESERVED ? text : "NAME"
86 push!(tokens, DToken(kind, text, text))
87 elseif 'A' <= c <= 'Z'
88 push!(tokens, DToken("VARIABLE", text, text))
89 elseif text == "_"
90 push!(tokens, DToken("ANONYMOUS", text, nothing))
91 elseif all(ch -> '0' <= ch <= '9', text)
92 value = parse(Int, text)
93 value > 1_000_000_000 && d_fail("integer out of range '$text'")
94 push!(tokens, DToken("INTEGER", text, value))
95 else
96 d_fail("bad token '$text'")
97 end
98 else
99 d_fail("bad character '$c'")
100 end
101 end
102 push!(tokens, DToken("EOF", "", nothing))
103 tokens
104end

106mutable struct DParser
107 tokens::Vector{DToken}
108 position::Int
109end
110d_peek(p) = p.tokens[p.position]
111function d_advance!(p)
112 token = d_peek(p)
113 p.position += 1
114 token
115end
116function d_syntax(p)
117 token = d_peek(p)
118 d_fail(token.kind == "EOF" ? "syntax error at end of line" : "syntax error at '$(token.text)'")
119end
120function d_expect!(p, kind)
121 d_peek(p).kind == kind || d_syntax(p)
122 d_advance!(p)
123end
124function d_term!(p)
125 token = d_peek(p)
126 if token.kind in ("NAME", "STRING", "INTEGER")
127 d_advance!(p)
128 DTerm(:constant, token.value)
129 elseif token.kind == "VARIABLE"
130 d_advance!(p)
131 DTerm(:variable, token.text)
132 elseif token.kind == "ANONYMOUS"
133 d_advance!(p)
134 DTerm(:anonymous, nothing)
135 elseif token.kind == "-"
136 d_advance!(p)
137 DTerm(:constant, -d_expect!(p, "INTEGER").value)
138 else
139 d_syntax(p)
140 end
141end
142function d_atom!(p)
143 name = d_expect!(p, "NAME").text
144 terms = DTerm[]
145 if d_peek(p).kind == "("
146 d_advance!(p)
147 push!(terms, d_term!(p))
148 while d_peek(p).kind == ","
149 d_advance!(p)
150 push!(terms, d_term!(p))
151 end
152 d_expect!(p, ")")
153 end
154 DAtom(name, terms)
155end
156function d_factor!(p)
157 if d_peek(p).kind == "("
158 d_advance!(p)
159 result = DExpression(:group, nothing, [d_expression!(p)])
160 d_expect!(p, ")")
161 result
162 else
163 DExpression(:term, d_term!(p), DExpression[])
164 end
165end
166function d_product!(p)
167 result = d_factor!(p)
168 while d_peek(p).kind in ("*", "/", "mod")
169 operation = d_advance!(p).kind
170 result = DExpression(:binary, operation, [result, d_factor!(p)])
171 end
172 result
173end
174function d_expression!(p)
175 result = d_product!(p)
176 while d_peek(p).kind in ("+", "-")
177 operation = d_advance!(p).kind
178 result = DExpression(:binary, operation, [result, d_product!(p)])
179 end
180 result
181end
182function d_literal!(p)
183 token = d_peek(p)
184 if token.kind == "not"
185 d_advance!(p)
186 DLiteral(:negative, d_atom!(p), "", DExpression[])
187 elseif token.kind == "NAME" && !(p.tokens[p.position+1].kind in union(D_COMPARISONS, D_OPERATORS))
188 DLiteral(:positive, d_atom!(p), "", DExpression[])
189 else
190 left = d_expression!(p)
191 d_peek(p).kind in D_COMPARISONS || d_syntax(p)
192 operation = d_advance!(p).kind
193 DLiteral(:comparison, nothing, operation, [left, d_expression!(p)])
194 end
195end
196function d_body!(p)
197 body = [d_literal!(p)]
198 while d_peek(p).kind == ","
199 d_advance!(p)
200 push!(body, d_literal!(p))
201 end
202 body
203end
204function d_parse(line)
205 p = DParser(d_lex(line), 1)
206 kind = d_peek(p).kind
207 kind == "EOF" && return nothing
208 if kind == "NAME"
209 head = d_atom!(p)
210 if d_peek(p).kind == ":-"
211 d_advance!(p)
212 statement = DStatement(kind=:rule, head=head, body=d_body!(p))
213 else
214 statement = DStatement(kind=:fact, head=head)
215 end
216 elseif kind == ":-"
217 d_advance!(p)
218 statement = DStatement(kind=:constraint, body=d_body!(p))
219 elseif kind == "?-"
220 d_advance!(p)
221 aggregate = ""
222 variable = ""
223 if d_peek(p).kind in ("count", "sum", "min", "max")
224 aggregate = d_advance!(p).kind
225 if aggregate != "count"
226 variable = d_expect!(p, "VARIABLE").text
227 d_expect!(p, ":")
228 end
229 end
230 statement = DStatement(kind=:query, body=d_body!(p), aggregate=aggregate, variable=variable)
231 elseif kind in ("retract", "why")
232 d_advance!(p)
233 statement = DStatement(kind=Symbol(kind), head=d_atom!(p))
234 elseif kind == "dump"
235 d_advance!(p)
236 statement = DStatement(kind=:dump, name=d_expect!(p, "NAME").text)
237 elseif kind == "forget"
238 d_advance!(p)
239 statement = DStatement(kind=:forget, number=d_expect!(p, "INTEGER").value)
240 elseif kind == "listing"
241 d_advance!(p)
242 name = d_peek(p).kind == "NAME" ? d_advance!(p).text : ""
243 statement = DStatement(kind=:listing, name=name)
244 elseif kind in ("strata", "stats", "check", "predicates", "undo", "halt")
245 d_advance!(p)
246 statement = DStatement(kind=Symbol(kind))
247 else
248 d_syntax(p)
249 end
250 d_expect!(p, ".")
251 d_expect!(p, "EOF")
252 statement
253end

255function d_value_text(value)
256 value isa Int && return string(value)
257 occursin(r"^[a-z][A-Za-z0-9_]*$", value) && !(value in D_RESERVED) ? value : "\"" * value * "\""
258end
259function d_term_text(term, environment=nothing)
260 term.kind == :anonymous && return "_"
261 term.kind == :constant && return d_value_text(term.value)
262 environment === nothing ? term.value : d_value_text(environment[term.value])
263end
264function d_atom_text(atom, environment=nothing)
265 isempty(atom.terms) && return atom.name
266 atom.name * "(" * join((d_term_text(t, environment) for t in atom.terms), ", ") * ")"
267end
268d_tuple_text(name, tuple) = d_atom_text(DAtom(name, [DTerm(:constant, v) for v in tuple]))
269function d_expression_text(expression, environment=nothing)
270 expression.kind == :term && return d_term_text(expression.value, environment)
271 expression.kind == :group && return "(" * d_expression_text(expression.operands[1], environment) * ")"
272 d_expression_text(expression.operands[1], environment) * " " * expression.value * " " * d_expression_text(expression.operands[2], environment)
273end
274function d_literal_text(literal, environment=nothing)
275 literal.kind == :positive && return d_atom_text(literal.atom, environment)
276 literal.kind == :negative && return "not " * d_atom_text(literal.atom, environment)
277 d_expression_text(literal.expressions[1], environment) * " " * literal.operation * " " * d_expression_text(literal.expressions[2], environment)
278end
279d_rule_text(rule) = (rule.head === nothing ? ":- " : d_atom_text(rule.head) * " :- ") * join(d_literal_text.(rule.body), ", ") * "."
280function d_expression_terms(expression)
281 expression.kind == :term && return [expression.value]
282 reduce(vcat, d_expression_terms.(expression.operands); init=DTerm[])
283end
284d_literal_terms(literal) = literal.kind == :comparison ? reduce(vcat, d_expression_terms.(literal.expressions); init=DTerm[]) : literal.atom.terms
285function d_occurrences(head, body)
286 terms = head === nothing ? DTerm[] : copy(head.terms)
287 for literal in body
288 append!(terms, d_literal_terms(literal))
289 end
290 String[t.value for t in terms if t.kind == :variable]
291end
292d_variables(head, body) = unique(d_occurrences(head, body))
293d_value_less(a, b) = a isa Int ? (b isa String || a < b) : (b isa String && isless(a, b))
294function d_tuple_less(a, b)
295 for (x, y) in zip(a, b)
296 x == y || return d_value_less(x, y)
297 end
298 false
299end
300d_sorted(tuples) = sort!(collect(tuples); lt=d_tuple_less)

320function d_atoms(head, body)
321 result = head === nothing ? DAtom[] : [head]
322 append!(result, (literal.atom for literal in body if literal.kind != :comparison))
323 result

325function d_predicate_check(database, atoms; clause=false)
326 prospective = copy(database.arities)
327 for atom in atoms
328 if haskey(prospective, atom.name)
329 expected = prospective[atom.name]
330 expected == length(atom.terms) || d_fail("arity mismatch for $(atom.name): expected $expected, got $(length(atom.terms))")
331 elseif clause
332 prospective[atom.name] = length(atom.terms)
333 else
334 d_fail("unknown predicate $(atom.name)")
335 end
336 end
337 prospective
338end
339function d_safety(statement)
340 head, body = statement.head, statement.body
341 head !== nothing && any(t -> t.kind == :anonymous, head.terms) && d_fail("anonymous variable in head")
342 for literal in body
343 literal.kind == :comparison && any(t -> t.kind == :anonymous, d_literal_terms(literal)) && d_fail("anonymous variable in comparison")
344 end
345 bound = Set{String}()
346 for literal in body
347 if literal.kind == :positive
348 union!(bound, (t.value for t in literal.atom.terms if t.kind == :variable))
349 end
350 end
351 required = head === nothing ? DTerm[] : copy(head.terms)
352 !isempty(statement.variable) && push!(required, DTerm(:variable, statement.variable))
353 for literal in body
354 literal.kind != :positive && append!(required, d_literal_terms(literal))
355 end
356 for term in required
357 term.kind == :variable && !(term.value in bound) && d_fail("unsafe variable $(term.value)")
358 end
359end
360function d_levels(arities, rules)
361 levels = Dict(name => 0 for name in keys(arities))
362 for iteration in 1:length(arities)
363 changed = false
364 for rule in rules
365 rule.head === nothing && continue
366 for literal in rule.body
367 literal.kind == :comparison && continue
368 level = levels[literal.atom.name] + Int(literal.kind == :negative)
369 if level > levels[rule.head.name]
370 levels[rule.head.name] = level
371 changed = true
372 end
373 end
374 end
375 !changed && return levels
376 end
377 isempty(arities) ? levels : nothing
378end
379function d_snapshot!(database, line)
380 facts = Dict(name => copy(tuples) for (name, tuples) in database.facts)
381 push!(database.history, (copy(database.arities), facts, copy(database.rules), line))
382 database.cache = nothing
383end
384function d_match(atom, tuple, environment)
385 result = copy(environment)
386 for (term, value) in zip(atom.terms, tuple)
387 if term.kind == :constant
388 term.value == value || return nothing
389 elseif term.kind == :variable
390 haskey(result, term.value) && result[term.value] != value && return nothing
391 result[term.value] = value
392 end
393 end
394 result
395end
396function d_evaluate(expression, environment)
397 if expression.kind == :term
398 term = expression.value
399 return term.kind == :constant ? term.value : environment[term.value]
400 elseif expression.kind == :group
401 return d_evaluate(expression.operands[1], environment)
402 end
403 a = d_evaluate(expression.operands[1], environment)
404 b = d_evaluate(expression.operands[2], environment)
405 a isa Int && b isa Int || return nothing
406 operation = expression.value
407 operation == "+" && return a + b
408 operation == "-" && return a - b
409 operation == "*" && return a * b
410 b == 0 && return nothing
411 quotient = fld(a, b)
412 operation == "/" ? quotient : a - b * quotient
413end
414function d_comparison(literal, environment)
415 a = d_evaluate(literal.expressions[1], environment)
416 b = d_evaluate(literal.expressions[2], environment)
417 (a === nothing || b === nothing) && return false
418 literal.operation == "=" && return a == b
419 literal.operation == "!=" && return a != b
420 a isa Int && b isa Int || return false
421 literal.operation == "<" ? a < b : literal.operation == "<=" ? a <= b : literal.operation == ">" ? a > b : a >= b
422end
423function d_solutions(body, tuples; derivations=nothing, before=nothing)
424 variables = d_variables(nothing, body)
425 positive = [literal.atom for literal in body if literal.kind == :positive]
426 filters = [(literal, unique(String[term.value for term in d_literal_terms(literal) if term.kind == :variable]))
427 for literal in body if literal.kind != :positive]
428 environments = Dict{String,DValue}[]
429 final_seen = Set{Tuple}()
430 used = falses(length(positive))
431 function visit(environment, depth)
432 for (literal, required) in filters
433 all(name -> haskey(environment, name), required) || continue
434 if literal.kind == :comparison
435 d_comparison(literal, environment) || return
436 elseif any(tuple -> d_match(literal.atom, tuple, environment) !== nothing, tuples[literal.atom.name])
437 return
438 end
439 end
440 if depth == length(positive)
441 key = Tuple(environment[name] for name in variables)
442 if !(key in final_seen)
443 push!(final_seen, key)
444 push!(environments, environment)
445 end
446 return
447 end
448 choice, best_bound, best_size = 0, -1, typemax(Int)
449 for (i, atom) in enumerate(positive)
450 used[i] && continue
451 bound = count(term -> term.kind == :constant || (term.kind == :variable && haskey(environment, term.value)), atom.terms)
452 size = length(tuples[atom.name])
453 if bound > best_bound || (bound == best_bound && size < best_size)
454 choice, best_bound, best_size = i, bound, size
455 end
456 end
457 atom = positive[choice]
458 used[choice] = true
459 seen = Set{Tuple}()
460 for tuple in tuples[atom.name]
461 before !== nothing && derivations[atom.name][tuple][1] >= before && continue
462 matched = d_match(atom, tuple, environment)
463 matched === nothing && continue
464 key = Tuple(matched[name] for name in variables if haskey(matched, name))
465 if !(key in seen)
466 push!(seen, key)
467 visit(matched, depth+1)
468 end
469 end
470 used[choice] = false
471 end
472 visit(Dict{String,DValue}(), 0)
473 sort!(environments; lt=(a,b) -> d_tuple_less(Tuple(a[v] for v in variables), Tuple(b[v] for v in variables)))
474end
475d_head_tuple(atom, environment) = Tuple(t.kind == :constant ? t.value : environment[t.value] for t in atom.terms)
476function d_model(database)
477 database.cache !== nothing && return database.cache
478 levels = d_levels(database.arities, database.rules)
479 tuples = Dict(name => copy(get(database.facts, name, Set{Tuple}())) for name in keys(database.arities))
480 derivations = Dict(name => Dict{Tuple,Tuple{Int,Int}}(tuple => (0,0) for tuple in rows) for (name,rows) in tuples)
481 rounds, added = Dict{Int,Int}(), Dict{Int,Int}()
482 serial = 0
483 for stratum in 0:maximum(values(levels); init=-1)
484 rounds[stratum] = added[stratum] = 0
485 while true
486 new = Dict(name => Dict{Tuple,Int}() for name in keys(database.arities))
487 for rule in database.rules
488 (rule.head === nothing || levels[rule.head.name] != stratum) && continue
489 for environment in d_solutions(rule.body, tuples)
490 tuple = d_head_tuple(rule.head, environment)
491 if !(tuple in tuples[rule.head.name])
492 target = new[rule.head.name]
493 target[tuple] = min(get(target, tuple, rule.number), rule.number)
494 end
495 end
496 end
497 amount = sum(length(rows) for rows in values(new); init=0)
498 amount == 0 && break
499 serial += 1
500 rounds[stratum] += 1
501 added[stratum] += amount
502 for (name, rows) in new, (tuple, number) in rows
503 push!(tuples[name], tuple)
504 derivations[name][tuple] = (serial, number)
505 end
506 end
507 end
508 database.cache = DModel(tuples, derivations, levels, rounds, added)
509end
510function d_answer_lines(variables, environments; indent="")
511 [indent * join((name * " = " * d_value_text(environment[name]) for name in variables), ", ") for environment in environments]
512end
513d_plural(n, singular, plural=singular * "s") = string(n) * " " * (n == 1 ? singular : plural)
514function d_query(database, statement)
515 d_predicate_check(database, d_atoms(nothing, statement.body))
516 d_safety(statement)
517 model = d_model(database)
518 variables = d_variables(nothing, statement.body)
519 answers = d_solutions(statement.body, model.tuples)
520 aggregate = statement.aggregate
521 if aggregate == "count"
522 return [string(length(answers))]
523 elseif aggregate == "sum"
524 values = [environment[statement.variable] for environment in answers]
525 all(v -> v isa Int, values) || d_fail("non-integer value in sum")
526 return [string(sum(values; init=0))]
527 elseif aggregate in ("min", "max")
528 isempty(answers) && return ["no answers"]
529 values = sort!([environment[statement.variable] for environment in answers]; lt=d_value_less)
530 return [d_value_text(aggregate == "min" ? first(values) : last(values))]
531 elseif isempty(variables)
532 return [isempty(answers) ? "false." : "true."]
533 end
534 vcat(d_answer_lines(variables, answers), [d_plural(length(answers), "answer")])
535end
536function d_proof!(lines, database, model, name, tuple, depth)
537 indent = repeat(" ", 2depth)
538 serial = model.derivations[name][tuple][1]
539 if serial == 0
540 push!(lines, indent * d_tuple_text(name, tuple) * " [fact]")
541 return
542 end
543 for rule in database.rules
544 (rule.head === nothing || rule.head.name != name) && continue
545 solutions = d_solutions(rule.body, model.tuples; derivations=model.derivations, before=serial)
546 filter!(environment -> d_head_tuple(rule.head, environment) == tuple, solutions)
547 isempty(solutions) && continue
548 variables = d_variables(rule.head, rule.body)
549 sort!(solutions; lt=(a,b) -> d_tuple_less(Tuple(a[v] for v in variables), Tuple(b[v] for v in variables)))
550 environment = first(solutions)
551 push!(lines, indent * d_tuple_text(name, tuple) * " [rule $(rule.number)]")
552 for literal in rule.body
553 if literal.kind == :positive
554 atom = literal.atom
555 matches = [row for row in model.tuples[atom.name] if model.derivations[atom.name][row][1] < serial && d_match(atom, row, environment) !== nothing]
556 chosen = first(d_sorted(matches))
557 d_proof!(lines, database, model, atom.name, chosen, depth + 1)
558 else
559 push!(lines, indent * " " * d_literal_text(literal, environment))
560 end
561 end
562 return
563 end
564end
565function d_recursive(database, name)
566 pending = String[]
567 for rule in database.rules
568 rule.head !== nothing && rule.head.name == name || continue
569 append!(pending, (literal.atom.name for literal in rule.body if literal.kind != :comparison))
570 end
571 seen = Set{String}()
572 while !isempty(pending)
573 current = pop!(pending)
574 current == name && return true
575 current in seen && continue
576 push!(seen, current)
577 for rule in database.rules
578 rule.head !== nothing && rule.head.name == current || continue
579 append!(pending, (literal.atom.name for literal in rule.body if literal.kind != :comparison))
580 end
581 end
582 false
583end
584function d_register!(database, statement, line)
585 prospective = d_predicate_check(database, d_atoms(statement.head, statement.body); clause=true)
586 d_safety(statement)
587 if statement.kind == :fact
588 tuple = d_head_tuple(statement.head, Dict{String,DValue}())
589 tuple in get(database.facts, statement.head.name, Set{Tuple}()) && return String[]
590 d_snapshot!(database, line)
591 database.arities = prospective
592 push!(get!(database.facts, statement.head.name, Set{Tuple}()), tuple)
593 return String[]
594 end
595 rule = DRule(database.next_rule, statement.head, statement.body)
596 notation = d_rule_text(rule)
597 for previous in database.rules
598 d_rule_text(previous) == notation && d_fail("duplicate of rule $(previous.number)")
599 end
600 statement.kind == :rule && d_levels(prospective, vcat(database.rules, [rule])) === nothing && d_fail("negative cycle through $(statement.head.name)")
601 d_snapshot!(database, line)
602 database.arities = prospective
603 push!(database.rules, rule)
604 database.next_rule += 1
605 occurrences = d_occurrences(statement.head, statement.body)
606 ["WARNING line $line: singleton variable $name" for name in unique(occurrences) if count(==(name), occurrences) == 1]
607end
608function d_process!(database, statement, line)
609 kind = statement.kind
610 kind in (:fact, :rule, :constraint) && return d_register!(database, statement, line)
611 kind == :query && return d_query(database, statement)
612 if kind == :undo
613 isempty(database.history) && d_fail("nothing to undo")
614 arities, facts, rules, original = pop!(database.history)
615 database.arities, database.facts, database.rules = arities, facts, rules
616 database.cache = nothing
617 return ["undid line $original"]
618 elseif kind == :forget
619 at = findfirst(rule -> rule.number == statement.number, database.rules)
620 at === nothing && d_fail("no rule $(statement.number)")
621 d_snapshot!(database, line)
622 deleteat!(database.rules, at)
623 return String[]
624 elseif kind in (:retract, :why)
625 atom = statement.head
626 d_predicate_check(database, [atom])
627 if kind == :retract
628 rows = get(database.facts, atom.name, Set{Tuple}())
629 targets = [tuple for tuple in rows if d_match(atom, tuple, Dict{String,DValue}()) !== nothing]
630 isempty(targets) && d_fail("no matching facts")
631 d_snapshot!(database, line)
632 for tuple in targets
633 delete!(rows, tuple)
634 end
635 return [d_plural(length(targets), "fact") * " retracted"]
636 end
637 any(t -> t.kind != :constant, atom.terms) && d_fail("why needs a ground atom")
638 tuple = d_head_tuple(atom, Dict{String,DValue}())
639 model = d_model(database)
640 tuple in model.tuples[atom.name] || return ["false."]
641 lines = String[]
642 d_proof!(lines, database, model, atom.name, tuple, 0)
643 return lines
644 elseif kind in (:dump, :listing)
645 !isempty(statement.name) && !haskey(database.arities, statement.name) && d_fail("unknown predicate $(statement.name)")
646 if kind == :listing
647 rules = [rule for rule in database.rules if isempty(statement.name) || (rule.head !== nothing && rule.head.name == statement.name)]
648 return isempty(rules) ? ["no rules"] : ["$(rule.number): $(d_rule_text(rule))" for rule in rules]
649 end
650 rows = d_sorted(d_model(database).tuples[statement.name])
651 return vcat([d_tuple_text(statement.name, tuple) * "." for tuple in rows], [d_plural(length(rows), "tuple")])
652 end
653 model = d_model(database)
654 names = sort!(collect(keys(database.arities)))
655 if kind == :strata
656 isempty(names) && return ["no predicates"]
657 return ["stratum $k: " * join((name * (d_recursive(database, name) ? "*" : "") for name in names if model.levels[name] == k), ", ") for k in 0:maximum(values(model.levels))]
658 elseif kind == :stats
659 fact_count = sum(length(rows) for rows in values(database.facts); init=0)
660 rule_count = count(rule -> rule.head !== nothing, database.rules)
661 tuple_count = sum(length(rows) for rows in values(model.tuples); init=0)
662 lines = ["facts $fact_count, rules $rule_count, constraints $(length(database.rules)-rule_count), tuples $tuple_count"]
663 for k in 0:maximum(values(model.levels); init=-1)
664 push!(lines, "stratum $k: rounds $(model.rounds[k]), new $(model.added[k])")
665 end
666 for rule in database.rules
667 rule.head === nothing && continue
668 amount = count(info -> info[2] == rule.number, values(model.derivations[rule.head.name]))
669 push!(lines, "rule $(rule.number): new $amount")
670 end
671 return lines
672 elseif kind == :predicates
673 isempty(names) && return ["no predicates"]
674 return ["$name/$(database.arities[name]): facts $(length(get(database.facts,name,Set{Tuple}()))), rules $(count(rule -> rule.head !== nothing && rule.head.name == name, database.rules)), tuples $(length(model.tuples[name]))" for name in names]
675 elseif kind == :check
676 constraints = [rule for rule in database.rules if rule.head === nothing]
677 isempty(constraints) && return ["no constraints"]
678 lines = String[]
679 for rule in constraints
680 answers = d_solutions(rule.body, model.tuples)
681 if isempty(answers)
682 push!(lines, "constraint $(rule.number): ok")
683 else
684 push!(lines, "constraint $(rule.number): " * d_plural(length(answers), "violation"))
685 variables = d_variables(nothing, rule.body)
686 !isempty(variables) && append!(lines, d_answer_lines(variables, answers; indent=" "))
687 end
688 end
689 return lines
690 end
691 String[]
692end

695 for (line_number, line) in enumerate(eachline())

697 statement = d_parse(line)
698 statement === nothing && continue
699 statement.kind == :halt && break

701 println(output)
702 end
703 catch error
704 error isa DatalogError || rethrow()
705 println("ERROR line $line_number: $(error.reason)")
706 end
707 end
708end
