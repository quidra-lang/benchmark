18const set<string> reserved = {"not", "mod", "count", "sum", "min", "max",
19 "retract", "dump", "why", "forget", "listing", "strata", "stats", "check",
20 "predicates", "undo", "halt"};
21bool lower(char c) {
22 return c >= 'a' && c <= 'z';
23}
24bool upper(char c) {
25 return c >= 'A' && c <= 'Z';
26}
27bool digit(char c) {
28 return c >= '0' && c <= '9';
29}
30bool word(char c) {
31 return lower(c) || upper(c) || digit(c) || c == '_';
32}

34struct Value {
35 bool symbol = false;
36 Integer number = 0;
37 string text;
38 static Value integer(Integer n) {
39 return {false, n, ""};
40 }
41 static Value atom(string s) {
42 return {true, 0, move(s)};
43 }
44 bool operator<(const Value& b) const {
45 if (symbol != b.symbol) return !symbol;
46 return symbol ? text < b.text : number < b.number;
47 }
48 bool operator==(const Value& b) const {
49 return symbol == b.symbol && (symbol ? text == b.text : number == b.number);
50 }
51 string notation() const {
52 if (!symbol) return to_string(number);
53 bool bare = !text.empty() && lower(text[0]) && !reserved.count(text);
54 for (char c : text) bare = bare && word(c);
55 return bare ? text : "\"" + text + "\"";
56 }
57};
58using Tuple = vector<Value>;
59using Binding = map<string, Value>;
60struct Term {
61 char kind = 'c';
62 Value value;
63 string variable;
64 string notation(const Binding* binding = nullptr) const {
65 if (kind == 'c') return value.notation();
66 if (kind == 'a') return "_";
67 return binding ? binding->at(variable).notation() : variable;
68 }
69 Value evaluate(const Binding& binding) const {
70 return kind == 'c' ? value : binding.at(variable);
71 }
72};
73struct Expression {
74 string op;
75 Term term;
76 shared_ptr<Expression> left, right;
77 string notation(const Binding* binding = nullptr) const {
78 if (op.empty()) return term.notation(binding);
79 if (op == "()") return "(" + left->notation(binding) + ")";
80 return left->notation(binding) + " " + op + " " + right->notation(binding);
81 }
82 optional<Value> evaluate(const Binding& binding) const {
83 if (op.empty()) return term.evaluate(binding);
84 auto a = left->evaluate(binding);
85 if (op == "()") return a;
86 auto b = right->evaluate(binding);
87 if (!a || !b || a->symbol || b->symbol) return nullopt;
88 Integer x = a->number, y = b->number, n;
89 if (op == "+") n = x + y;
90 else if (op == "-") n = x - y;
91 else if (op == "*") n = x * y;
92 else {
93 if (!y) return nullopt;
94 Integer quotient = x / y;
95 if (x % y && ((x < 0) != (y < 0))) --quotient;
96 n = op == "/" ? quotient : x - y * quotient;
97 }
98 return Value::integer(n);
99 }
100 void terms(vector<Term>& result) const {
101 if (op.empty()) result.push_back(term);
102 else {
103 left->terms(result);
104 if (right) right->terms(result);
105 }
106 }
107};
108struct Atom {
109 string name;
110 vector<Term> terms;
111 string notation(const Binding* binding = nullptr) const {
112 string result = name;
113 if (!terms.empty()) {
114 result += '(';
115 for (size_t i = 0; i < terms.size(); ++i) {
116 if (i) result += ", ";
117 result += terms[i].notation(binding);
118 }
119 result += ')';
120 }
121 return result;
122 }
123 Tuple tuple(const Binding& binding) const {
124 Tuple result;
125 for (const Term& t : terms) result.push_back(t.evaluate(binding));
126 return result;
127 }
128 bool match(const Tuple& tuple, Binding& binding) const {
129 for (size_t i = 0; i < terms.size(); ++i) {
130 const Term& t = terms[i];
131 if (t.kind == 'a') continue;
132 if (t.kind == 'c') {
133 if (!(t.value == tuple[i])) return false;
134 } else {
135 auto found = binding.find(t.variable);
136 if (found == binding.end()) binding[t.variable] = tuple[i];
137 else if (!(found->second == tuple[i])) return false;
138 }
139 }
140 return true;
141 }
142};
143struct Literal {
144 char kind = 'p';
145 Atom atom;
146 string comparison;
147 shared_ptr<Expression> left, right;
148 vector<Term> terms() const {
149 if (kind != 'c') return atom.terms;
150 vector<Term> result;
151 left->terms(result);
152 right->terms(result);
153 return result;
154 }
155 string notation(const Binding* binding = nullptr) const {
156 if (kind != 'c') return (kind == 'n' ? "not " : "") + atom.notation(binding);
157 return left->notation(binding) + " " + comparison + " " + right->notation(binding);
158 }
159 bool holds(const Binding& binding) const {
160 auto a = left->evaluate(binding), b = right->evaluate(binding);
161 if (!a || !b) return false;
162 if (comparison == "=") return *a == *b;
163 if (comparison == "!=") return !(*a == *b);
164 if (a->symbol || b->symbol) return false;
165 if (comparison == "<") return a->number < b->number;
166 if (comparison == "<=") return a->number <= b->number;
167 if (comparison == ">") return a->number > b->number;
168 return a->number >= b->number;
169 }
170};
171struct Rule {
172 int number = 0;
173 optional<Atom> head;
174 vector<Literal> body;
175 string notation() const {
176 string result = head ? head->notation() + " :- " : ":- ";
177 for (size_t i = 0; i < body.size(); ++i) {
178 if (i) result += ", ";
179 result += body[i].notation();
180 }
181 return result + '.';
182 }

184struct Token {
185 string text, kind;

187vector<Token> lex(const string& line) {
188 vector<Token> tokens;
189 for (size_t i = 0; i < line.size();) {
190 char c = line[i];
191 if (c == ' ') {
192 ++i;
193 continue;
194 }
195 if (c == '%') break;
196 if (c == '"') {
197 size_t end = line.find('"', i + 1);
198 if (end == string::npos) throw runtime_error("unterminated string");
199 tokens.push_back({line.substr(i, end - i + 1), "string"});
200 i = end + 1;
201 continue;
202 }
203 string pair = line.substr(i, 2);
204 if (pair == ":-" || pair == "?-" || pair == "!=" || pair == "<=" || pair == ">=") {
205 tokens.push_back({pair, pair});
206 i += 2;
207 } else if (string("(),.:=<>+-*/").find(c) != string::npos) {
208 tokens.push_back({string(1, c), string(1, c)});
209 ++i;
210 } else if (word(c)) {
211 size_t start = i++;
212 while (i < line.size() && word(line[i])) ++i;
213 string text = line.substr(start, i - start), kind;
214 if (lower(c)) kind = reserved.count(text) ? text : "name";
215 else if (upper(c)) kind = "variable";
216 else if (text == "_") kind = "anonymous";
217 else if (all_of(text.begin(), text.end(), digit)) {
218 if (stoll(text) > 1000000000) throw runtime_error("integer out of range '" + text + "'");
219 kind = "integer";
220 } else throw runtime_error("bad token '" + text + "'");
221 tokens.push_back({text, kind});
222 } else throw runtime_error("bad character '" + string(1, c) + "'");
223 }
224 return tokens;
225}

232class Parser {
233 const vector<Token>& tokens;
234 size_t index = 0;
235 string kind(size_t ahead = 0) const {
236 return index + ahead < tokens.size() ? tokens[index + ahead].kind : "";
237 }
238 bool take(const string& s) {
239 if (kind() != s) return false;
240 ++index;
241 return true;
242 }
243 [[noreturn]] void error() const {
244 if (index == tokens.size()) throw runtime_error("syntax error at end of line");
245 throw runtime_error("syntax error at '" + tokens[index].text + "'");
246 }
247 Token need(const string& s) {
248 if (kind() != s) error();
249 return tokens[index++];
250 }
251 bool compare(const string& s) const {
252 return s == "=" || s == "!=" || s == "<" || s == "<=" || s == ">" || s == ">=";
253 }
254 Term term() {
255 Term result;
256 string k = kind();
257 if (k == "name") result.value = Value::atom(tokens[index++].text);
258 else if (k == "string") {
259 string s = tokens[index++].text;
260 result.value = Value::atom(s.substr(1, s.size() - 2));
261 } else if (k == "integer" || k == "-") {
262 bool negative = take("-");
263 result.value = Value::integer(stoll(need("integer").text) * (negative ? -1 : 1));
264 } else if (k == "variable") {
265 result.kind = 'v';
266 result.variable = tokens[index++].text;
267 } else if (take("anonymous")) result.kind = 'a';
268 else error();
269 return result;
270 }
271 Atom atom() {
272 Atom result;
273 result.name = need("name").text;
274 if (take("(")) {
275 result.terms.push_back(term());
276 while (take(",")) result.terms.push_back(term());
277 need(")");
278 }
279 return result;
280 }
281 shared_ptr<Expression> factor() {
282 auto result = make_shared<Expression>();
283 if (take("(")) {
284 result->op = "()";
285 result->left = expression();
286 need(")");
287 } else result->term = term();
288 return result;
289 }
290 shared_ptr<Expression> product() {
291 auto result = factor();
292 while (kind() == "*" || kind() == "/" || kind() == "mod") {
293 string op = tokens[index++].text;
294 auto right = factor();
295 result = make_shared<Expression>(Expression{op, {}, result, right});
296 }
297 return result;
298 }
299 shared_ptr<Expression> expression() {
300 auto result = product();
301 while (kind() == "+" || kind() == "-") {
302 string op = tokens[index++].text;
303 auto right = product();
304 result = make_shared<Expression>(Expression{op, {}, result, right});
305 }
306 return result;
307 }
308 Literal literal() {
309 Literal result;
310 if (take("not")) {
311 result.kind = 'n';
312 result.atom = atom();
313 } else if (kind() == "name" && !compare(kind(1)) && kind(1) != "+" && kind(1) != "-"
314 && kind(1) != "*" && kind(1) != "/" && kind(1) != "mod") result.atom = atom();
315 else {
316 result.kind = 'c';
317 result.left = expression();
318 if (!compare(kind())) error();
319 result.comparison = tokens[index++].text;
320 result.right = expression();
321 }
322 return result;
323 }
324 vector<Literal> body() {
325 vector<Literal> result{literal()};
326 while (take(",")) result.push_back(literal());
327 return result;
328 }
329public:
330 explicit Parser(const vector<Token>& ts) : tokens(ts) {
331 }
332 Statement parse() {
333 Statement result;
334 if (take("?-")) {
335 result.kind = "query";
336 if (take("count")) result.aggregate = "count";
337 else if (kind() == "sum" || kind() == "min" || kind() == "max") {
338 result.aggregate = tokens[index++].text;
339 result.aggregateVariable = need("variable").text;
340 need(":");
341 }
342 result.rule.body = body();
343 } else if (take(":-")) {
344 result.kind = "rule";
345 result.rule.body = body();
346 } else if (kind() == "name") {
347 result.atom = atom();
348 if (take(":-")) {
349 result.kind = "rule";
350 result.rule.head = result.atom;
351 result.rule.body = body();
352 } else result.kind = "fact";
353 } else {
354 result.kind = kind();
355 const set<string> commands = {"retract", "why", "dump", "forget", "listing", "strata",
356 "stats", "check", "predicates", "undo", "halt"};
357 if (!commands.count(result.kind)) error();
358 ++index;
359 if (result.kind == "retract" || result.kind == "why") result.atom = atom();
360 else if (result.kind == "dump") result.name = need("name").text;
361 else if (result.kind == "forget") result.number = stoll(need("integer").text);
362 else if (result.kind == "listing" && kind() == "name") result.name = tokens[index++].text;
363 }
364 need(".");
365 if (index != tokens.size()) error();
366 return result;
367 }
368};
369vector<Term> allTerms(const Rule& rule) {
370 vector<Term> result;
371 if (rule.head) result = rule.head->terms;
372 for (const Literal& literal : rule.body) {
373 auto ts = literal.terms();
374 result.insert(result.end(), ts.begin(), ts.end());
375 }
376 return result;
377}
378vector<string> variables(const vector<Term>& terms) {
379 vector<string> result;
380 set<string> seen;
381 for (const Term& t : terms) {
382 if (t.kind == 'v' && seen.insert(t.variable).second) result.push_back(t.variable);
383 }
384 return result;
385}
386Tuple bindingTuple(const Binding& binding, const vector<string>& vars) {
387 Tuple result;
388 for (const string& v : vars) result.push_back(binding.at(v));
389 return result;
390}
391string answerText(const Tuple& tuple, const vector<string>& vars) {
392 string result;
393 for (size_t i = 0; i < vars.size(); ++i) {
394 if (i) result += ", ";
395 result += vars[i] + " = " + tuple[i].notation();
396 }
397 return result;
398}

412vector<Atom> atoms(const Rule& rule) {
413 vector<Atom> result;
414 if (rule.head) result.push_back(*rule.head);
415 for (const Literal& l : rule.body) if (l.kind != 'c') result.push_back(l.atom);
416 return result;
417}
418void predicates(const State& state, const Rule& rule, bool registering) {
419 auto known = state.arities;
420 for (const Atom& a : atoms(rule)) {
421 auto found = known.find(a.name);
422 if (found == known.end()) {
423 if (!registering) throw runtime_error("unknown predicate " + a.name);
424 known[a.name] = static_cast<int>(a.terms.size());
425 } else if (found->second != static_cast<int>(a.terms.size())) {
426 throw runtime_error("arity mismatch for " + a.name + ": expected " + to_string(found->second)
427 + ", got " + to_string(a.terms.size()));
428 }
429 }
430}
431void safety(const Rule& rule, const string& aggregate = "") {
432 if (rule.head) for (const Term& t : rule.head->terms) {
433 if (t.kind == 'a') throw runtime_error("anonymous variable in head");
434 }
435 for (const Literal& l : rule.body) if (l.kind == 'c') for (const Term& t : l.terms()) {
436 if (t.kind == 'a') throw runtime_error("anonymous variable in comparison");
437 }
438 set<string> bound;
439 for (const Literal& l : rule.body) if (l.kind == 'p') for (const Term& t : l.atom.terms) {
440 if (t.kind == 'v') bound.insert(t.variable);
441 }
442 vector<Term> checked;
443 if (rule.head) checked = rule.head->terms;
444 if (!aggregate.empty()) checked.push_back({'v', {}, aggregate});
445 for (const Literal& l : rule.body) if (l.kind != 'p') {
446 auto ts = l.terms();
447 checked.insert(checked.end(), ts.begin(), ts.end());
448 }
449 for (const Term& t : checked) if (t.kind == 'v' && !bound.count(t.variable)) {
450 throw runtime_error("unsafe variable " + t.variable);
451 }
452}
453void registerPredicates(State& state, const Rule& rule) {
454 for (const Atom& a : atoms(rule)) {
455 state.arities[a.name] = static_cast<int>(a.terms.size());
456 state.facts.try_emplace(a.name);
457 }
458}
459map<string, int> strata(const State& state, const string& newHead = "") {
460 map<string, int> result;
461 for (const auto& p : state.arities) result[p.first] = 0;
462 for (size_t pass = 0; pass < result.size(); ++pass) {
463 bool changed = false;
464 for (const Rule& r : state.rules) if (r.head) for (const Literal& l : r.body) if (l.kind != 'c') {
465 int required = result.at(l.atom.name) + (l.kind == 'n');
466 int& s = result.at(r.head->name);
467 if (s < required) {
468 s = required;
469 changed = true;
470 }
471 }
472 if (!changed) break;
473 if (pass + 1 == result.size()) throw runtime_error("negative cycle through " + newHead);
474 }
475 return result;
476}
477set<Binding> solve(const vector<Literal>& body, const Database& db, int limit = -1) {
478 vector<const Atom*> positive;
479 vector<pair<const Literal*, vector<Term>>> filters;
480 for (const Literal& literal : body) {
481 if (literal.kind == 'p') positive.push_back(&literal.atom);
482 else filters.emplace_back(&literal, literal.terms());
483 }
484 set<Binding> solutions;
485 vector<bool> used(positive.size(), false);
486 function<void(const Binding&, size_t)> visit = [&](const Binding& binding, size_t depth) {
487 for (const auto& filter : filters) {
488 bool bound = true;
489 for (const Term& term : filter.second) {
490 if (term.kind == 'v' && !binding.count(term.variable)) bound = false;
491 }
492 if (!bound) continue;
493 const Literal& literal = *filter.first;
494 if (literal.kind == 'c') {
495 if (!literal.holds(binding)) return;
496 } else {
497 for (const auto& entry : db.at(literal.atom.name)) {
498 Binding candidate = binding;
499 if (literal.atom.match(entry.first, candidate)) return;
500 }
501 }
502 }
503 if (depth == positive.size()) {
504 solutions.insert(binding);
505 return;
506 }
507 size_t choice = 0, bestSize = 0;
508 int bestBound = -1;
509 for (size_t i = 0; i < positive.size(); ++i) if (!used[i]) {
510 int bound = 0;
511 for (const Term& term : positive[i]->terms) {
512 if (term.kind == 'c' || (term.kind == 'v' && binding.count(term.variable))) ++bound;
513 }
514 size_t size = db.at(positive[i]->name).size();
515 if (bound > bestBound || (bound == bestBound && size < bestSize)) {
516 choice = i;
517 bestBound = bound;
518 bestSize = size;
519 }
520 }
521 const Atom& atom = *positive[choice];
522 used[choice] = true;
523 set<Binding> seen;
524 for (const auto& entry : db.at(atom.name)) {
525 if (limit >= 0 && entry.second.round >= limit) continue;
526 Binding candidate = binding;
527 if (atom.match(entry.first, candidate) && seen.insert(candidate).second) visit(candidate, depth + 1);
528 }
529 used[choice] = false;
530 };
531 visit(Binding{}, 0);
532 return solutions;
533}
534Model model(const State& state) {
535 Model result;
536 auto levels = strata(state);
537 int maximum = -1;
538 for (const auto& p : levels) {
539 maximum = max(maximum, p.second);
540 auto& rows = result.database[p.first];
541 for (const Tuple& tuple : state.facts.at(p.first)) rows[tuple] = {};
542 }
543 result.rounds.resize(maximum + 1);
544 result.added.resize(maximum + 1);
545 int serial = 0;
546 for (int level = 0; level <= maximum; ++level) for (;;) {
547 map<pair<string, Tuple>, int> added;
548 for (const Rule& r : state.rules) if (r.head && levels.at(r.head->name) == level) {
549 for (const Binding& binding : solve(r.body, result.database)) {
550 Tuple tuple = r.head->tuple(binding);
551 if (result.database.at(r.head->name).count(tuple)) continue;
552 auto key = make_pair(r.head->name, move(tuple));
553 auto found = added.find(key);
554 if (found == added.end() || r.number < found->second) added[key] = r.number;
555 }
556 }
557 if (added.empty()) break;
558 ++serial;
559 ++result.rounds[level];
560 result.added[level] += static_cast<int>(added.size());
561 for (const auto& entry : added) result.database[entry.first.first][entry.first.second] = {serial, entry.second};
562 }
563 return result;
564}
565set<Tuple> answers(const Rule& rule, const Database& db, const vector<string>& vars) {
566 set<Tuple> result;
567 for (const Binding& binding : solve(rule.body, db)) result.insert(bindingTuple(binding, vars));
568 return result;
569}
570string groundNotation(const string& name, const Tuple& tuple) {
571 Atom atom;
572 atom.name = name;
573 for (const Value& value : tuple) atom.terms.push_back({'c', value, ""});
574 return atom.notation();
575}
576void proof(ostream& out, const State& state, const Database& db, const string& name, const Tuple& tuple, int depth) {
577 const Derivation& origin = db.at(name).at(tuple);
578 string indent(depth * 2, ' '), text = groundNotation(name, tuple);
579 if (!origin.round) {
580 out << indent << text << " [fact]\n";
581 return;
582 }
583 for (const Rule& r : state.rules) if (r.head && r.head->name == name) {
584 optional<Binding> chosen;
585 Tuple best;
586 auto vars = variables(allTerms(r));
587 for (const Binding& binding : solve(r.body, db, origin.round)) {
588 if (r.head->tuple(binding) != tuple) continue;
589 Tuple order = bindingTuple(binding, vars);
590 if (!chosen || order < best) {
591 chosen = binding;
592 best = move(order);
593 }
594 }
595 if (!chosen) continue;
596 out << indent << text << " [rule " << r.number << "]\n";
597 for (const Literal& l : r.body) {
598 if (l.kind != 'p') out << indent << " " << l.notation(&*chosen) << '\n';
599 else for (const auto& entry : db.at(l.atom.name)) {
600 if (entry.second.round >= origin.round) continue;
601 Binding binding = *chosen;
602 if (l.atom.match(entry.first, binding)) {
603 proof(out, state, db, l.atom.name, entry.first, depth + 1);
604 break;
605 }
606 }
607 }
608 return;
609 }
610}
611class Engine {
612 State state;
613 int nextRule = 1;
614 vector<pair<int, State>> history;
615 void save(int line) {
616 history.emplace_back(line, state);
617 }
618 void requirePredicate(const string& name) const {
619 if (!state.arities.count(name)) throw runtime_error("unknown predicate " + name);
620 }
621public:
622 bool execute(const Statement& statement, int line, ostream& out) {
623 const string& command = statement.kind;
624 if (command == "halt") return false;
625 if (command == "fact" || command == "rule") {
626 Rule r = statement.rule;
627 if (command == "fact") r.head = statement.atom;
628 predicates(state, r, true);
629 safety(r);
630 if (command == "fact") {
631 Tuple tuple = r.head->tuple({});
632 auto found = state.facts.find(r.head->name);
633 if (found != state.facts.end() && found->second.count(tuple)) return true;
634 save(line);
635 registerPredicates(state, r);
636 state.facts[r.head->name].insert(move(tuple));
637 } else {
638 for (const Rule& old : state.rules) if (old.notation() == r.notation()) {
639 throw runtime_error("duplicate of rule " + to_string(old.number));
640 }
641 State proposed = state;
642 registerPredicates(proposed, r);
643 proposed.rules.push_back(r);
644 if (r.head) strata(proposed, r.head->name);
645 save(line);
646 r.number = nextRule++;
647 registerPredicates(state, r);
648 state.rules.push_back(r);
649 auto ts = allTerms(r);
650 map<string, int> count;
651 for (const Term& t : ts) if (t.kind == 'v') ++count[t.variable];
652 for (const string& v : variables(ts)) if (count[v] == 1) {
653 out << "WARNING line " << line << ": singleton variable " << v << '\n';
654 }
655 }
656 } else if (command == "query") {
657 const Rule& r = statement.rule;
658 predicates(state, r, false);
659 safety(r, statement.aggregateVariable);
660 auto computed = model(state);
661 auto vars = variables(allTerms(r));
662 auto rows = answers(r, computed.database, vars);
663 const string& aggregate = statement.aggregate;
664 if (aggregate == "count") out << rows.size() << '\n';
665 else if (!aggregate.empty()) {
666 size_t position = find(vars.begin(), vars.end(), statement.aggregateVariable) - vars.begin();
667 if (aggregate == "sum") {
668 Integer total = 0;
669 for (const Tuple& row : rows) {
670 if (row[position].symbol) throw runtime_error("non-integer value in sum");
671 total += row[position].number;
672 }
673 out << total << '\n';
674 } else if (rows.empty()) out << "no answers\n";
675 else {
676 optional<Value> best;
677 for (const Tuple& row : rows) {
678 const Value& value = row[position];
679 if (!best || (aggregate == "min" ? value < *best : *best < value)) best = value;
680 }
681 out << best->notation() << '\n';
682 }
683 } else if (vars.empty()) out << (rows.empty() ? "false.\n" : "true.\n");
684 else {
685 for (const Tuple& row : rows) out << answerText(row, vars) << '\n';
686 out << rows.size() << (rows.size() == 1 ? " answer\n" : " answers\n");
687 }
688 } else if (command == "retract" || command == "why") {
689 Rule check;
690 check.head = statement.atom;
691 predicates(state, check, false);
692 const Atom& a = statement.atom;
693 if (command == "retract") {
694 vector<Tuple> removed;
695 for (const Tuple& tuple : state.facts.at(a.name)) {
696 Binding binding;
697 if (a.match(tuple, binding)) removed.push_back(tuple);
698 }
699 if (removed.empty()) throw runtime_error("no matching facts");
700 save(line);
701 for (const Tuple& tuple : removed) state.facts[a.name].erase(tuple);
702 out << removed.size() << (removed.size() == 1 ? " fact retracted\n" : " facts retracted\n");
703 } else {
704 for (const Term& t : a.terms) if (t.kind != 'c') throw runtime_error("why needs a ground atom");
705 auto computed = model(state);
706 Tuple tuple = a.tuple({});
707 if (!computed.database.at(a.name).count(tuple)) out << "false.\n";
708 else proof(out, state, computed.database, a.name, tuple, 0);
709 }
710 } else if (command == "forget") {
711 auto found = find_if(state.rules.begin(), state.rules.end(), [&](const Rule& r) {
712 return r.number == statement.number;
713 });
714 if (found == state.rules.end()) throw runtime_error("no rule " + to_string(statement.number));
715 size_t index = found - state.rules.begin();
716 save(line);
717 state.rules.erase(state.rules.begin() + index);
718 } else if (command == "undo") {
719 if (history.empty()) throw runtime_error("nothing to undo");
720 out << "undid line " << history.back().first << '\n';
721 state = move(history.back().second);
722 history.pop_back();
723 } else if (command == "listing") {
724 if (!statement.name.empty()) requirePredicate(statement.name);
725 bool any = false;
726 for (const Rule& r : state.rules) if (statement.name.empty() || (r.head && r.head->name == statement.name)) {
727 out << r.number << ": " << r.notation() << '\n';
728 any = true;
729 }
730 if (!any) out << "no rules\n";
731 } else if (command == "strata") {
732 auto levels = strata(state);
733 if (levels.empty()) out << "no predicates\n";
734 else {
735 map<string, set<string>> reach;
736 for (const Rule& r : state.rules) if (r.head) for (const Literal& l : r.body) {
737 if (l.kind != 'c') reach[r.head->name].insert(l.atom.name);
738 }
739 for (const auto& middle : state.arities) for (const auto& start : state.arities) {
740 if (reach[start.first].count(middle.first)) {
741 const auto destinations = reach[middle.first];
742 reach[start.first].insert(destinations.begin(), destinations.end());
743 }
744 }
745 int maximum = 0;
746 for (const auto& p : levels) maximum = max(maximum, p.second);
747 for (int k = 0; k <= maximum; ++k) {
748 out << "stratum " << k << ": ";
749 bool first = true;
750 for (const auto& p : levels) if (p.second == k) {
751 if (!first) out << ", ";
752 out << p.first << (reach[p.first].count(p.first) ? "*" : "");
753 first = false;
754 }
755 out << '\n';
756 }
757 }
758 } else if (command == "dump") {
759 requirePredicate(statement.name);
760 auto computed = model(state);
761 const auto& tuples = computed.database.at(statement.name);
762 for (const auto& t : tuples) out << groundNotation(statement.name, t.first) << ".\n";
763 out << tuples.size() << (tuples.size() == 1 ? " tuple\n" : " tuples\n");
764 } else if (command == "stats" || command == "predicates") {
765 auto computed = model(state);
766 size_t facts = 0, tuples = 0, rules = 0;
767 map<int, int> deriving;
768 for (const auto& p : state.arities) {
769 facts += state.facts.at(p.first).size();
770 tuples += computed.database.at(p.first).size();
771 for (const auto& row : computed.database.at(p.first)) if (row.second.rule) ++deriving[row.second.rule];
772 }
773 for (const Rule& r : state.rules) if (r.head) ++rules;
774 if (command == "stats") {
775 out << "facts " << facts << ", rules " << rules << ", constraints " << state.rules.size() - rules
776 << ", tuples " << tuples << '\n';
777 for (size_t k = 0; k < computed.rounds.size(); ++k) {
778 out << "stratum " << k << ": rounds " << computed.rounds[k] << ", new " << computed.added[k] << '\n';
779 }
780 for (const Rule& r : state.rules) if (r.head) out << "rule " << r.number << ": new " << deriving[r.number] << '\n';
781 } else {
782 if (state.arities.empty()) out << "no predicates\n";
783 for (const auto& p : state.arities) {
784 int count = 0;
785 for (const Rule& r : state.rules) if (r.head && r.head->name == p.first) ++count;
786 out << p.first << '/' << p.second << ": facts " << state.facts.at(p.first).size()
787 << ", rules " << count << ", tuples " << computed.database.at(p.first).size() << '\n';
788 }
789 }
790 } else if (command == "check") {
791 auto computed = model(state);
792 bool any = false;
793 for (const Rule& r : state.rules) if (!r.head) {
794 any = true;
795 auto vars = variables(allTerms(r));
796 auto rows = answers(r, computed.database, vars);
797 out << "constraint " << r.number << ": ";
798 if (rows.empty()) out << "ok\n";
799 else {
800 out << rows.size() << (rows.size() == 1 ? " violation\n" : " violations\n");
801 if (!vars.empty()) for (const Tuple& row : rows) out << " " << answerText(row, vars) << '\n';
802 }
803 }
804 if (!any) out << "no constraints\n";
805 }
806 return true;
807 }
808};
809int main() {
810 ios::sync_with_stdio(false);
811 cin.tie(nullptr);
812 Engine engine;
813 string line;
814 int number = 0;
815 while (getline(cin, line)) {
816 ++number;
817 try {
818 auto tokens = lex(line);
819 if (tokens.empty()) continue;
820 auto statement = Parser(tokens).parse();
821 ostringstream output;
822 if (!engine.execute(statement, number, output)) break;
823 cout << output.str();
824 } catch (const runtime_error& error) {
825 cout << "ERROR line " << number << ": " << error.what() << '\n';
826 }
827 }
