198Token[] | error tokenize(string line)
199 Token[] result := []
200 nat cursor := 0
201 while cursor < #line
202 c := line[cursor]
203 pair := if cursor + 1 < #line then line[cursor:<cursor + 2] else ""
204 if c == " "
205 cursor += 1
206 elif c == "%"
207 break
208 elif c == DQ
209 finish := cursor + 1
210 while finish < #line and line[finish] != DQ
211 finish += 1
212 if finish == #line
213 return error("unterminated string")
214 result = result.append(Token("STRING", line[cursor + 1:<finish]))
215 cursor = finish + 1
216 elif contains_text([":-", "?-", "!=", "<=", ">="], pair)
217 result = result.append(Token(pair, pair))
218 cursor += 2
219 elif "(),.:=<>+-*/".contains(c)
220 result = result.append(Token(c, c))
221 cursor += 1
222 elif word_character(c)
223 finish := cursor + 1
224 while finish < #line and word_character(line[finish])
225 finish += 1
226 word := line[cursor:<finish]
227 if "abcdefghijklmnopqrstuvwxyz".contains(c)
228 result = result.append(Token(if contains_text(reserved_words(), word) then word else "NAME", word))
229 elif "ABCDEFGHIJKLMNOPQRSTUVWXYZ".contains(c)
230 result = result.append(Token("VARIABLE", word))
231 elif word == "_"
232 result = result.append(Token("ANONYMOUS", word))
233 elif decimal_digits(word)
234 value := int(word)
235 if value > 1000000000
236 return error("integer out of range '" + word + "'")
237 result = result.append(Token("INTEGER", word, value))
238 else
239 return error("bad token '" + word + "'")
240 cursor = finish
241 else
242 return error("bad character '" + c + "'")
