function valueText(value) {
    return typeof value === "bigint"
        ? String(value)
        : /^[a-z][a-zA-Z0-9_]*$/.test(value) && !reserved.has(value)
          ? value
          : '"' + value + '"';
}
function valueCompare(a, b) {
    if (typeof a === "bigint") {
        if (typeof b !== "bigint") return -1;
        return a < b ? -1 : a > b ? 1 : 0;
    }
    if (typeof b === "bigint") return 1;
    return a < b ? -1 : a > b ? 1 : 0;
}
function tupleCompare(a, b) {
    for (let i = 0; i < a.length; i++) {
        const order = valueCompare(a[i], b[i]);
        if (order !== 0) return order;
    }
    return 0;
}
function tupleKey(values) {
    return JSON.stringify(
        values.map((value) =>
            typeof value === "bigint" ? ["integer", String(value)] : ["symbol", value],
        ),
    );
}
function bindingKey(binding) {
    const names = [...binding.keys()].sort();
    return JSON.stringify(names.map((name) => [name, tupleKey([binding.get(name)])]));
}

    return [...new Set(terms.filter((term) => term.kind === "variable").map((term) => term.name))];
