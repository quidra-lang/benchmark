14function valueText(value) {
15 return typeof value === "bigint"
16 ? String(value)
17 : /^[a-z][a-zA-Z0-9_]*$/.test(value) && !reserved.has(value)
18 ? value
19 : '"' + value + '"';

21function valueCompare(a, b) {
22 if (typeof a === "bigint") {
23 if (typeof b !== "bigint") return -1;
24 return a < b ? -1 : a > b ? 1 : 0;
25 }
26 if (typeof b === "bigint") return 1;
27 return a < b ? -1 : a > b ? 1 : 0;
28}
29function tupleCompare(a, b) {
30 for (let i = 0; i < a.length; i++) {
31 const order = valueCompare(a[i], b[i]);
32 if (order !== 0) return order;
33 }
34 return 0;

47function lex(line) {
48 const tokens = [];
49 for (let i = 0; i < line.length;) {
50 const c = line[i];
51 if (c === " ") {
52 i++;
53 continue;
54 }
55 if (c === "%") break;
56 if (c === '"') {
57 const end = line.indexOf('"', i + 1);
58 if (end < 0) fail("unterminated string");
59 tokens.push({
60 kind: "string",
61 text: line.slice(i, end + 1),
62 value: line.slice(i + 1, end),
63 });
64 i = end + 1;
65 continue;
66 }
67 const pair = line.slice(i, i + 2);
68 if ([":-", "?-", "!=", "<=", ">="].includes(pair)) {
69 tokens.push({ kind: pair, text: pair });
70 i += 2;
71 continue;
72 }
73 if ("(),.:=<>+-*/".includes(c)) {
74 tokens.push({ kind: c, text: c });
75 i++;
76 continue;
77 }
78 if (/[a-zA-Z0-9_]/.test(c)) {
79 const start = i;
80 while (i < line.length && /[a-zA-Z0-9_]/.test(line[i])) i++;
81 const text = line.slice(start, i);
82 if (/^[a-z]/.test(text))
83 tokens.push({ kind: reserved.has(text) ? text : "name", text, value: text });
84 else if (/^[A-Z]/.test(text)) tokens.push({ kind: "variable", text });
85 else if (text === "_") tokens.push({ kind: "anonymous", text });
86 else if (/^[0-9]+$/.test(text)) {
87 const value = BigInt(text);
88 if (value > 1000000000n) fail("integer out of range '" + text + "'");
89 tokens.push({ kind: "integer", text, value });
90 } else fail("bad token '" + text + "'");
91 continue;
92 }
93 fail("bad character '" + c + "'");
94 }
95 tokens.push({ kind: "end", text: "" });
96 return tokens;
97}
98class Parser {
99 constructor(tokens) {
100 this.tokens = tokens;
101 this.at = 0;
102 }
103 peek(offset = 0) {
104 return this.tokens[this.at + offset];
105 }
106 syntax() {
107 const token = this.peek();
108 fail(
109 token.kind === "end"
110 ? "syntax error at end of line"
111 : "syntax error at '" + token.text + "'",
112 );
113 }
114 take(kind) {
115 if (this.peek().kind !== kind) this.syntax();
116 return this.tokens[this.at++];
117 }
118 term() {
119 const token = this.peek();
120 if (token.kind === "name" || token.kind === "string" || token.kind === "integer") {
121 this.at++;
122 const value =
123 token.kind === "integer"
124 ? BigInt(token.text)
125 : token.kind === "string"
126 ? token.text.slice(1, -1)
127 : token.text;
128 return { kind: "constant", value };
129 }
130 if (token.kind === "-") {
131 this.at++;
132 return { kind: "constant", value: -BigInt(this.take("integer").text) };
133 }
134 if (token.kind === "variable") {
135 this.at++;
136 return { kind: "variable", name: token.text };
137 }
138 if (token.kind === "anonymous") {
139 this.at++;
140 return { kind: "anonymous" };
141 }
142 this.syntax();
143 }
144 atom() {
145 const name = this.take("name").text,
146 terms = [];
147 if (this.peek().kind === "(") {
148 this.at++;
149 terms.push(this.term());
150 while (this.peek().kind === ",") {
151 this.at++;
152 terms.push(this.term());
153 }
154 this.take(")");
155 }
156 return { name, terms };
157 }
158 factor() {
159 if (this.peek().kind === "(") {
160 this.at++;
161 const expression = this.expression();
162 this.take(")");
163 return { kind: "parentheses", expression };
164 }
165 return { kind: "term", term: this.term() };
166 }
167 product() {
168 let left = this.factor();
169 while (["*", "/", "mod"].includes(this.peek().kind)) {
170 const operator = this.tokens[this.at++].kind;
171 left = { kind: "operation", operator, left, right: this.factor() };
172 }
173 return left;
174 }
175 expression() {
176 let left = this.product();
177 while (this.peek().kind === "+" || this.peek().kind === "-") {
178 const operator = this.tokens[this.at++].kind;
179 left = { kind: "operation", operator, left, right: this.product() };
180 }
181 return left;
182 }
183 literal() {
184 if (this.peek().kind === "not") {
185 this.at++;
186 return { kind: "atom", atom: this.atom(), negative: true };
187 }
188 if (
189 this.peek().kind === "name" &&
190 !comparisons.has(this.peek(1).kind) &&
191 !arithmetic.has(this.peek(1).kind)
192 )
193 return { kind: "atom", atom: this.atom(), negative: false };
194 const left = this.expression(),
195 operator = this.peek().kind;
196 if (!comparisons.has(operator)) this.syntax();
197 this.at++;
198 return { kind: "comparison", left, operator, right: this.expression() };
199 }
200 body() {
201 const literals = [this.literal()];
202 while (this.peek().kind === ",") {
203 this.at++;
204 literals.push(this.literal());
205 }
206 return literals;
207 }
208 statement() {
209 let result;
210 const kind = this.peek().kind;
211 if (kind === "name" || kind === ":-") {
212 const head = kind === "name" ? this.atom() : null;
213 let body = [];
214 if (this.peek().kind === ":-") {
215 this.at++;
216 body = this.body();
217 }
218 result = { kind: "clause", head, body };
219 } else if (kind === "?-") {
220 this.at++;
221 let aggregate = "",
222 variable = "";
223 if (["count", "sum", "min", "max"].includes(this.peek().kind)) {
224 aggregate = this.tokens[this.at++].kind;
225 if (aggregate !== "count") {
226 variable = this.take("variable").text;
227 this.take(":");
228 }
229 }
230 result = { kind: "query", aggregate, variable, body: this.body() };
231 } else if (kind === "retract" || kind === "why") {
232 this.at++;
233 result = { kind: "atom-command", command: kind, atom: this.atom() };
234 } else if (kind === "dump" || kind === "listing") {
235 this.at++;
236 const name =
237 kind === "dump" || this.peek().kind === "name" ? this.take("name").text : null;
238 result = { kind: "named-command", command: kind, name };
239 } else if (kind === "forget") {
240 this.at++;
241 result = { kind: "forget", number: Number(this.take("integer").text) };
242 } else if (["strata", "stats", "check", "predicates", "undo", "halt"].includes(kind)) {
243 this.at++;
244 result = { kind: "simple", command: kind };
245 } else this.syntax();
246 this.take(".");
247 this.take("end");
248 return result;
249 }
250}
251function expressionTerms(expression) {
252 return expression.kind === "term"
253 ? [expression.term]

284 (atom.terms.length

292function expressionText(expression, binding = null) {
293 if (expression.kind === "term") return termText(expression.term, binding);
294 if (expression.kind === "parentheses")
295 return "(" + expressionText(expression.expression, binding) + ")";

304function literalText(literal, binding = null) {
305 return literal.kind === "atom"
306 ? (literal.negative ? "not " : "") + atomText(literal.atom, binding)
307 : expressionText(literal.left, binding) +
308 " " +
309 literal.operator +
310 " " +
311 expressionText(literal.right, binding);
312}
313function ruleText(head, body) {
314 return (
315 (head === null ? ":- " : atomText(head) + " :- ") +
316 body.map((literal) => literalText(literal)).join(", ") +
317 "."
318 );

320function expressionValue(expression, binding) {
321 if (expression.kind === "term") return termValue(expression.term, binding);
322 if (expression.kind === "parentheses") return expressionValue(expression.expression, binding);
323 const a = expressionValue(expression.left, binding),
324 b = expressionValue(expression.right, binding);
325 if (typeof a !== "bigint" || typeof b !== "bigint") return null;
326 switch (expression.operator) {
327 case "+":
328 return a + b;
329 case "-":
330 return a - b;
331 case "*":
332 return a * b;
333 default: {
334 if (b === 0n) return null;
335 const quotient = a / b,
336 q = a % b !== 0n && a < 0n !== b < 0n ? quotient - 1n : quotient;
337 return expression.operator === "/" ? q : a - b * q;
338 }
339 }
340}
341function comparisonHolds(literal, binding) {
342 const a = expressionValue(literal.left, binding),
343 b = expressionValue(literal.right, binding);
344 if (a === null || b === null) return false;
345 if (literal.operator === "=") return a === b;
346 if (literal.operator === "!=") return a !== b;
347 if (typeof a !== "bigint" || typeof b !== "bigint") return false;
348 return literal.operator === "<"
349 ? a < b
350 : literal.operator === "<="
351 ? a <= b
352 : literal.operator === ">"
353 ? a > b
354 : a >= b;
355}
356function match(atom, values, binding) {
357 const next = new Map(binding);
358 for (let i = 0; i < atom.terms.length; i++) {
359 const term = atom.terms[i],
360 value = values[i];
361 if (term.kind === "constant") {
362 if (term.value !== value) return null;
363 } else if (term.kind === "variable") {
364 if (next.has(term.name) && next.get(term.name) !== value) return null;
365 next.set(term.name, value);
366 }
367 }
368 return next;
369}
370function solutions(body, data, before = Infinity) {
371 const positives = body
372 .filter((literal) => literal.kind === "atom" && !literal.negative)
373 .map((literal) => literal.atom);
374 const filters = body.filter((literal) => literal.kind === "comparison" || literal.negative);
375 const bound = new Set(),
376 ordered = [];
377 while (positives.length > 0) {
378 positives.sort((a, b) => {
379 const score = (atom) =>
380 atom.terms.filter(
381 (term) =>
382 term.kind === "constant" ||
383 (term.kind === "variable" && bound.has(term.name)),
384 ).length;
385 return score(b) - score(a) || data.get(a.name).size - data.get(b.name).size;
386 });
387 const atom = positives.shift();
388 ordered.push({
389 atom,
390 tuples: [...data.get(atom.name).values()].filter((tuple) => tuple.round < before),
391 });
392 for (const name of variables(atom.terms)) bound.add(name);
393 }
394 function accepts(binding) {
395 for (const literal of filters)
396 if (variables(literalTerms(literal)).every((name) => binding.has(name))) {
397 if (literal.kind === "comparison") {
398 if (!comparisonHolds(literal, binding)) return false;
399 } else if (
400 [...data.get(literal.atom.name).values()].some(
401 (tuple) => match(literal.atom, tuple.values, binding) !== null,
402 )
403 )
404 return false;
405 }
406 return true;
407 }
408 const answers = new Map();
409 function visit(index, binding) {
410 if (index === ordered.length) {
411 if (accepts(binding)) answers.set(bindingKey(binding), binding);
412 return;
413 }
414 const { atom, tuples } = ordered[index],
415 seen = new Set();
416 for (const tuple of tuples) {
417 const candidate = match(atom, tuple.values, binding);
418 if (candidate === null || !accepts(candidate)) continue;
419 const key = bindingKey(candidate);
420 if (seen.has(key)) continue;
421 seen.add(key);
422 visit(index + 1, candidate);
423 }
424 }
425 visit(0, new Map());
426 return [...answers.values()];
427}
428function assignment(predicates, rules) {
429 const levels = new Map([...predicates.keys()].map((name) => [name, 0]));
430 for (let iteration = 0; iteration < predicates.size; iteration++) {
431 let changed = false;
432 for (const rule of rules)
433 if (rule.head !== null)
434 for (const literal of rule.body)
435 if (literal.kind === "atom") {
436 const level = levels.get(literal.atom.name) + (literal.negative ? 1 : 0);
437 if (level > levels.get(rule.head.name)) {
438 levels.set(rule.head.name, level);
439 changed = true;
440 }
441 }
442 if (!changed) return levels;
443 if (iteration === predicates.size - 1) return null;
444 }
445 return levels;
446}

448class Engine {
449 constructor() {
450 this.predicates = new Map();
451 this.facts = new Map();
452 this.rules = [];
453 this.history = [];
454 this.nextRule = 1;
455 this.cached = null;
456 }
457 save(line) {
458 this.history.push({
459 line,
460 predicates: new Map(this.predicates),
461 facts: new Map([...this.facts].map(([name, tuples]) => [name, new Map(tuples)])),
462 rules: [...this.rules],
463 });
464 }
465 predicateCheck(atoms, clauses) {
466 const predicates = new Map(this.predicates);
467 for (const atom of atoms) {
468 if (!predicates.has(atom.name)) {
469 if (!clauses) fail("unknown predicate " + atom.name);
470 predicates.set(atom.name, atom.terms.length);
471 } else if (predicates.get(atom.name) !== atom.terms.length)
472 fail(
473 `arity mismatch for ${atom.name}: expected ${predicates.get(atom.name)}, got ${atom.terms.length}`,
474 );
475 }
476 return predicates;
477 }
478 safety(head, body, aggregate = "") {
479 if (head !== null && head.terms.some((term) => term.kind === "anonymous"))
480 fail("anonymous variable in head");
481 for (const literal of body)
482 if (
483 literal.kind === "comparison" &&
484 literalTerms(literal).some((term) => term.kind === "anonymous")
485 )
486 fail("anonymous variable in comparison");
487 const bound = new Set(
488 body
489 .filter((literal) => literal.kind === "atom" && !literal.negative)
490 .flatMap((literal) => variables(literal.atom.terms)),
491 );
492 const required = head === null ? (aggregate ? [aggregate] : []) : variables(head.terms);
493 for (const literal of body)
494 if (literal.kind === "comparison" || literal.negative)
495 required.push(...variables(literalTerms(literal)));
496 for (const variable of required)
497 if (!bound.has(variable)) fail("unsafe variable " + variable);
498 }
499 register(statement, line) {
500 const { head, body } = statement,
501 atoms = [
502 ...(head === null ? [] : [head]),
503 ...body.filter((literal) => literal.kind === "atom").map((literal) => literal.atom),
504 ];
505 const predicates = this.predicateCheck(atoms, true);
506 this.safety(head, body);
507 if (head !== null && body.length === 0) {
508 const values = head.terms.map((term) => term.value),
509 key = tupleKey(values);
510 if (this.facts.get(head.name)?.has(key)) return [];
511 this.save(line);
512 this.predicates = predicates;
513 for (const name of predicates.keys())
514 if (!this.facts.has(name)) this.facts.set(name, new Map());
515 this.facts.get(head.name).set(key, values);
516 this.cached = null;
517 return [];
518 }
519 const text = ruleText(head, body),
520 duplicate = this.rules.find((rule) => ruleText(rule.head, rule.body) === text);
521 if (duplicate !== undefined) fail("duplicate of rule " + duplicate.number);
522 const rule = { head, body, number: this.nextRule };
523 if (head !== null && assignment(predicates, [...this.rules, rule]) === null)
524 fail("negative cycle through " + head.name);
525 this.save(line);
526 this.predicates = predicates;
527 for (const name of predicates.keys())
528 if (!this.facts.has(name)) this.facts.set(name, new Map());
529 this.rules.push(rule);
530 this.nextRule++;
531 this.cached = null;
532 const counts = new Map();
533 for (const term of allTerms(head, body))
534 if (term.kind === "variable") counts.set(term.name, (counts.get(term.name) ?? 0) + 1);
535 return [...counts]
536 .filter(([, count]) => count === 1)
537 .map(([name]) => `WARNING line ${line}: singleton variable ${name}`);
538 }
539 model() {
540 if (this.cached !== null) return this.cached;
541 const levels = assignment(this.predicates, this.rules),
542 maximum = Math.max(0, ...levels.values());
543 const data = new Map(
544 [...this.predicates.keys()].map((name) => [
545 name,
546 new Map(
547 [...this.facts.get(name)].map(([key, values]) => [
548 key,
549 { values, round: 0, rule: 0 },
550 ]),
551 ),
552 ]),
553 );
554 const rounds = Array(maximum + 1).fill(0),
555 added = Array(maximum + 1).fill(0);
556 let serial = 0;
557 for (let stratum = 0; stratum <= maximum; stratum++) {
558 const rules = this.rules
559 .filter((rule) => rule.head !== null && levels.get(rule.head.name) === stratum)
560 .sort((a, b) => a.number - b.number);
561 while (true) {
562 const pending = new Map();
563 for (const rule of rules)
564 for (const binding of solutions(rule.body, data)) {
565 const values = rule.head.terms.map((term) => termValue(term, binding)),
566 key = tupleKey(values),
567 name = rule.head.name;
568 if (data.get(name).has(key)) continue;
569 if (!pending.has(name)) pending.set(name, new Map());
570 if (!pending.get(name).has(key))
571 pending
572 .get(name)
573 .set(key, { values, round: serial + 1, rule: rule.number });
574 }
575 let count = 0;
576 for (const [name, tuples] of pending)
577 for (const [key, tuple] of tuples) {
578 data.get(name).set(key, tuple);
579 count++;
580 }
581 if (count === 0) break;
582 serial++;
583 rounds[stratum]++;
584 added[stratum] += count;
585 }
586 }
587 this.cached = { data, levels, maximum, rounds, added };
588 return this.cached;
589 }
590 answers(body, model) {
591 const names = variables(body.flatMap(literalTerms));
592 const rows = solutions(body, model.data)
593 .map((binding) => names.map((name) => binding.get(name)))
594 .sort(tupleCompare);
595 return { names, rows };
596 }
597 answerText(names, row) {
598 return names.map((name, i) => name + " = " + valueText(row[i])).join(", ");
599 }
600 proof(name, tuple, depth, model, output) {
601 const indent = " ".repeat(depth),
602 text = tupleText(name, tuple.values);
603 if (tuple.round === 0) {
604 output.push(indent + text + " [fact]");
605 return;
606 }
607 for (const rule of this.rules
608 .filter((rule) => rule.head !== null && rule.head.name === name)
609 .sort((a, b) => a.number - b.number)) {
610 const names = variables(allTerms(rule.head, rule.body));
611 const candidates = solutions(rule.body, model.data, tuple.round).filter(
612 (binding) =>
613 tupleKey(rule.head.terms.map((term) => termValue(term, binding))) ===
614 tupleKey(tuple.values),
615 );
616 if (candidates.length === 0) continue;
617 candidates.sort((a, b) =>
618 tupleCompare(
619 names.map((name) => a.get(name)),
620 names.map((name) => b.get(name)),
621 ),
622 );
623 const binding = candidates[0];
624 output.push(indent + text + ` [rule ${rule.number}]`);
625 for (const literal of rule.body) {
626 if (literal.kind === "atom" && !literal.negative) {
627 const tuples = [...model.data.get(literal.atom.name).values()]
628 .filter(
629 (other) =>
630 other.round < tuple.round &&
631 match(literal.atom, other.values, binding) !== null,
632 )
633 .sort((a, b) => tupleCompare(a.values, b.values));
634 this.proof(literal.atom.name, tuples[0], depth + 1, model, output);
635 } else output.push(indent + " " + literalText(literal, binding));
636 }
637 return;
638 }
639 }
640 recursive(name) {
641 const visited = new Set();
642 const visit = (current) => {
643 for (const rule of this.rules)
644 if (rule.head !== null && rule.head.name === current)
645 for (const literal of rule.body)
646 if (literal.kind === "atom") {
647 const next = literal.atom.name;
648 if (next === name) return true;
649 if (!visited.has(next)) {
650 visited.add(next);
651 if (visit(next)) return true;
652 }
653 }
654 return false;
655 };
656 return visit(name);
657 }
658 process(statement, line) {
659 if (statement.kind === "clause") return this.register(statement, line);
660 if (statement.kind === "query") {
661 this.predicateCheck(
662 statement.body
663 .filter((literal) => literal.kind === "atom")
664 .map((literal) => literal.atom),
665 false,
666 );
667 this.safety(null, statement.body, statement.variable);
668 const { names, rows } = this.answers(statement.body, this.model());
669 if (statement.aggregate === "count") return [String(rows.length)];
670 if (statement.aggregate === "sum") {
671 const index = names.indexOf(statement.variable);
672 let total = 0n;
673 for (const row of rows) {
674 if (typeof row[index] !== "bigint") fail("non-integer value in sum");
675 total += row[index];
676 }
677 return [String(total)];
678 }
679 if (statement.aggregate === "min" || statement.aggregate === "max") {
680 if (rows.length === 0) return ["no answers"];
681 const index = names.indexOf(statement.variable),
682 values = rows.map((row) => row[index]).sort(valueCompare);
683 return [
684 valueText(
685 statement.aggregate === "min" ? values[0] : values[values.length - 1],
686 ),
687 ];
688 }
689 if (names.length === 0) return [rows.length > 0 ? "true." : "false."];
690 return [
691 ...rows.map((row) => this.answerText(names, row)),
692 `${rows.length} ${rows.length === 1 ? "answer" : "answers"}`,
693 ];
694 }
695 if (statement.kind === "atom-command") {
696 const atom = statement.atom;
697 this.predicateCheck([atom], false);
698 if (statement.command === "retract") {
699 const facts = this.facts.get(atom.name),
700 keys = [...facts]

702 .map(([key]) => key);
703 if (keys.length === 0) fail("no matching facts");

705 for (const key of keys) facts.delete(key);
706 this.cached = null;
707 return [`${keys.length} ${keys.length === 1 ? "fact" : "facts"} retracted`];
708 }

714 .get(tupleKey(atom.terms.map((term) => term.value)));

720 if (statement.kind === "forget") {

727 }

729 if (statement.name !== null && !this.predicates.has(statement.name))

734 (rule) =>
735 statement.name === null ||

737 )

744 tupleCompare(a.values, b.values),

749 ];
750 }

754 this.predicates = snapshot.predicates;

757 this.cached = null;
758 return ["undid line " + snapshot.line];

760 const model = this.model(),
761 output = [];

764 for (let level = 0; level <= model.maximum; level++)
765 output.push(
766 `stratum ${level}: ` +

768 .filter((name) => model.levels.get(name) === level)
769 .sort()
770 .map((name) => name + (this.recursive(name) ? "*" : ""))
771 .join(", "),
772 );
773 } else if (statement.command === "stats") {

775 tuples = 0;

778 const rules = this.rules
779 .filter((rule) => rule.head !== null)
780 .sort((a, b) => a.number - b.number);
781 output.push(
782 `facts ${facts}, rules ${rules.length}, constraints ${this.rules.length - rules.length}, tuples ${tuples}`,
783 );
784 if (this.predicates.size > 0)
785 for (let level = 0; level <= model.maximum; level++)
786 output.push(
787 `stratum ${level}: rounds ${model.rounds[level]}, new ${model.added[level]}`,
788 );
789 for (const rule of rules) {
790 let count = 0;
791 for (const predicate of model.data.values())
792 for (const tuple of predicate.values()) if (tuple.rule === rule.number) count++;
793 output.push(`rule ${rule.number}: new ${count}`);
794 }
795 } else if (statement.command === "predicates") {
796 if (this.predicates.size === 0) return ["no predicates"];
797 for (const name of [...this.predicates.keys()].sort())
798 output.push(
799 `${name}/${this.predicates.get(name)}: facts ${this.facts.get(name).size}, rules ${this.rules.filter((rule) => rule.head !== null && rule.head.name === name).length}, tuples ${model.data.get(name).size}`,
800 );
801 } else if (statement.command === "check") {
802 const constraints = this.rules
803 .filter((rule) => rule.head === null)
804 .sort((a, b) => a.number - b.number);
805 if (constraints.length === 0) return ["no constraints"];
806 for (const constraint of constraints) {
807 const { names, rows } = this.answers(constraint.body, model);
808 output.push(
809 `constraint ${constraint.number}: ` +
810 (rows.length
811 ? `${rows.length} ${rows.length === 1 ? "violation" : "violations"}`
812 : "ok"),
813 );
814 if (names.length > 0)
815 for (const row of rows) output.push(" " + this.answerText(names, row));
816 }
817 }
818 return output;
