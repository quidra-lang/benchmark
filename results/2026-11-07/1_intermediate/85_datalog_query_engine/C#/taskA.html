22 record Value(bool Integer, long Number, string Symbol) : IComparable<Value>
23 {
24 public static Value Int(long n) => new(true, n, "");
25 public static Value Sym(string s) => new(false, 0, s);
26 public int CompareTo(Value? other)
27 {
28 if (other == null) return 1;
29 if (Integer != other.Integer) return Integer ? -1 : 1;
30 return Integer ? Number.CompareTo(other.Number) : StringComparer.Ordinal.Compare(Symbol, other.Symbol);
31 }
32 public override string ToString() => Integer ? Number.ToString(CultureInfo.InvariantCulture) : Symbol.Length > 0 && Lower(Symbol[0]) && !Reserved.Contains(Symbol) && Symbol.All(WordChar) ? Symbol : "\"" + Symbol + "\"";
33 }

50 record Term(string? Variable, Value? Value, bool Anonymous)
51 {
52 public string Text(Dictionary<string, Value>? env) => Anonymous ? "_" : Value != null ? Value.ToString() : env != null && env.TryGetValue(Variable!, out Value? value) ? value.ToString() : Variable!;
53 public Value Get(Dictionary<string, Value> env) => Value ?? env[Variable!];
54 }

56 record Expr(string Op, Term? Term, Expr? Left, Expr? Right)
57 {
58 public string Text(Dictionary<string, Value>? env) => Term != null ? Term.Text(env) : Op == "()" ? "(" + Left!.Text(env) + ")" : Left!.Text(env) + " " + Op + " " + Right!.Text(env);
59 public void Terms(List<Term> output)
60 {
61 if (Term != null) output.Add(Term);
62 else
63 {
64 Left!.Terms(output);
65 Right?.Terms(output);
66 }
67 }
68 public Value? Evaluate(Dictionary<string, Value> env)
69 {
70 if (Term != null) return Term.Get(env);
71 Value? a = Left!.Evaluate(env);
72 if (Op == "()") return a;
73 Value? b = Right!.Evaluate(env);
74 if (a == null || b == null || !a.Integer || !b.Integer || (Op is "/" or "mod") && b.Number == 0) return null;
75 long x = a.Number, y = b.Number;
76 return Value.Int(Op switch
77 {
78 "+" => x + y,
79 "-" => x - y,
80 "*" => x * y,
81 "/" => FloorDivide(x, y),
82 _ => x - y * FloorDivide(x, y)
83 });
84 }
85 }
86
87 static long FloorDivide(long a, long b)
88 {
89 long q = a / b, r = a % b;
90 return r != 0 && (r < 0) != (b < 0) ? q - 1 : q;
91 }

99 record Literal(Atom? Atom, bool Negative, Expr? Left, string? Op, Expr? Right)
100 {
101 public List<Term> Terms()
102 {
103 if (Atom != null) return Atom.Terms;
104 List<Term> output = new();
105 Left!.Terms(output);
106 Right!.Terms(output);
107 return output;
108 }
109 public string Text(Dictionary<string, Value>? env) => Atom != null ? (Negative ? "not " : "") + Atom.Text(env) : Left!.Text(env) + " " + Op + " " + Right!.Text(env);
110 public bool Comparison(Dictionary<string, Value> env)
111 {
112 Value? a = Left!.Evaluate(env), b = Right!.Evaluate(env);
113 if (a == null || b == null) return false;
114 if (Op == "=") return a == b;
115 if (Op == "!=") return a != b;
116 if (!a.Integer || !b.Integer) return false;
117 int c = a.Number.CompareTo(b.Number);
118 return Op switch
119 {
120 "<" => c < 0,
121 "<=" => c <= 0,
122 ">" => c > 0,
123 _ => c >= 0
124 };
125 }
126 }

138 class Parser
139 {
140 public readonly List<Token> Tokens = new();
141 int at;
142 public Parser(string line)
143 {
144 for (int i = 0; i < line.Length;)
145 {
146 char c = line[i];
147 if (c == ' ')
148 {
149 i++;
150 continue;
151 }
152 if (c == '%') break;
153 int first = i;
154 string kind;
155 if (c == '"')
156 {
157 int end = line.IndexOf('"', i + 1);
158 if (end < 0) Fail("unterminated string");
159 i = end + 1;
160 kind = "string";
161 }
162 else if (i + 1 < line.Length && new[] { ":-", "?-", "!=", "<=", ">=" }.Contains(line.Substring(i, 2)))
163 {
164 i += 2;
165 kind = line[first..i];
166 }
167 else if ("(),.:=<>+-*/".Contains(c))
168 {
169 i++;
170 kind = c.ToString();
171 }
172 else if (WordChar(c))
173 {
174 while (i < line.Length && WordChar(line[i])) i++;
175 string s = line[first..i];
176 if (Lower(c)) kind = Reserved.Contains(s) ? s : "name";
177 else if (c is >= 'A' and <= 'Z') kind = "variable";
178 else if (s == "_") kind = "anonymous";
179 else if (s.All(Digit))
180 {
181 if (long.Parse(s) > 1000000000) Fail("integer out of range '" + s + "'");
182 kind = "integer";
183 }
184 else
185 {
186 Fail("bad token '" + s + "'");
187 return;
188 }
189 }
190 else
191 {
192 Fail("bad character '" + c + "'");
193 return;
194 }
195 Tokens.Add(new(kind, line[first..i]));
196 }
197 Tokens.Add(new("end", ""));
198 }
199 Token Peek() => Tokens[at];
200 public bool Is(string kind) => Peek().Kind == kind;
201 void Syntax() => Fail(Is("end") ? "syntax error at end of line" : "syntax error at '" + Peek().Text + "'");
202 Token Expect(string kind)
203 {
204 if (!Is(kind)) Syntax();
205 return Tokens[at++];
206 }
207 Term ReadTerm()
208 {
209 Token t = Peek();
210 if (Is("name"))
211 {
212 at++;
213 return new(null, Value.Sym(t.Text), false);
214 }
215 if (Is("string"))
216 {
217 at++;
218 return new(null, Value.Sym(t.Text[1..^1]), false);
219 }
220 if (Is("integer") || Is("-"))
221 {
222 bool minus = Is("-");
223 if (minus) at++;
224 long n = long.Parse(Expect("integer").Text);
225 return new(null, Value.Int(minus ? -n : n), false);
226 }
227 if (Is("variable"))
228 {
229 at++;
230 return new(t.Text, null, false);
231 }
232 if (Is("anonymous"))
233 {
234 at++;
235 return new(null, null, true);
236 }
237 Syntax();
238 return null!;
239 }
240 Atom ReadAtom()
241 {
242 string name = Expect("name").Text;
243 List<Term> args = new();
244 if (Is("("))
245 {
246 at++;
247 args.Add(ReadTerm());
248 while (Is(","))
249 {
250 at++;
251 args.Add(ReadTerm());
252 }
253 Expect(")");
254 }
255 return new(name, args);
256 }
257 Expr Factor()
258 {
259 if (Is("("))
260 {
261 at++;
262 Expr inner = Expression();
263 Expect(")");
264 return new("()", null, inner, null);
265 }
266 return new("", ReadTerm(), null, null);
267 }
268 Expr Product()
269 {
270 Expr output = Factor();
271 while (Is("*") || Is("/") || Is("mod"))
272 {
273 string op = Tokens[at++].Text;
274 output = new(op, null, output, Factor());
275 }
276 return output;
277 }
278 Expr Expression()
279 {
280 Expr output = Product();
281 while (Is("+") || Is("-"))
282 {
283 string op = Tokens[at++].Text;
284 output = new(op, null, output, Product());
285 }
286 return output;
287 }
288 Literal ReadLiteral()
289 {
290 if (Is("not"))
291 {
292 at++;
293 return new(ReadAtom(), true, null, null, null);
294 }
295 if (Is("name") && !Comparisons.Contains(Tokens[at + 1].Kind) && !Operations.Contains(Tokens[at + 1].Kind)) return new(ReadAtom(), false, null, null, null);
296 Expr left = Expression();
297 if (!Comparisons.Contains(Peek().Kind)) Syntax();
298 string op = Tokens[at++].Text;
299 return new(null, false, left, op, Expression());
300 }
301 List<Literal> ReadBody()
302 {
303 List<Literal> output = new() { ReadLiteral() };
304 while (Is(","))
305 {
306 at++;
307 output.Add(ReadLiteral());
308 }
309 return output;
310 }
311 public Statement ReadStatement()
312 {
313 string kind = Peek().Kind, aggregate = "";
314 string? name = null, variable = null;
315 Atom? atom = null;
316 List<Literal> body = new();
317 long number = 0;
318 if (Is("name") || Is(":-"))
319 {
320 if (Is("name")) atom = ReadAtom();
321 if (Is(":-"))
322 {
323 at++;
324 body = ReadBody();
325 kind = "rule";
326 }
327 else
328 {
329 if (atom == null) Syntax();
330 kind = "fact";
331 }
332 }
333 else if (Is("?-"))
334 {
335 at++;
336 kind = "query";
337 if (Is("count"))
338 {
339 aggregate = "count";
340 at++;
341 }
342 else if (Is("sum") || Is("min") || Is("max"))
343 {
344 aggregate = Tokens[at++].Text;
345 variable = Expect("variable").Text;
346 Expect(":");
347 }
348 body = ReadBody();
349 }
350 else
351 {
352 if (!new[] { "retract", "why", "dump", "forget", "listing", "strata", "stats", "check", "predicates", "undo", "halt" }.Contains(kind)) Syntax();
353 at++;
354 if (kind is "retract" or "why") atom = ReadAtom();
355 else if (kind == "dump") name = Expect("name").Text;
356 else if (kind == "forget") number = long.Parse(Expect("integer").Text);
357 else if (kind == "listing" && Is("name")) name = Tokens[at++].Text;
358 }
359 Expect(".");
360 if (!Is("end")) Syntax();
361 return new(kind, atom, body, name, number, aggregate, variable);
362 }
363 }

398 static List<string> Variables(IEnumerable<Term> terms) => terms.Where(t => t.Variable != null).Select(t => t.Variable!).Distinct(StringComparer.Ordinal).ToList();
399
400 static List<Term> BodyTerms(List<Literal> body) => body.SelectMany(l => l.Terms()).ToList();

402 static List<Atom> Atoms(Atom? head, List<Literal> body) => (head == null ? Enumerable.Empty<Atom>() : new[] { head }).Concat(body.Where(l => l.Atom != null).Select(l => l.Atom!)).ToList();
403 SortedDictionary<string, int> PredicateCheck(Atom? head, List<Literal> body, bool clause)
404 {
405 SortedDictionary<string, int> arities = new(StringComparer.Ordinal);
406 foreach (var e in state.Predicates) arities[e.Key] = e.Value.Arity;
407 foreach (Atom a in Atoms(head, body))
408 {
409 bool exists = arities.TryGetValue(a.Name, out int expected);
410 if (!exists && !clause) Fail("unknown predicate " + a.Name);
411 if (exists && expected != a.Terms.Count) Fail($"arity mismatch for {a.Name}: expected {expected}, got {a.Terms.Count}");
412 arities[a.Name] = a.Terms.Count;
413 }
414 return arities;
415 }

417 static void Safety(Atom? head, List<Literal> body, string? aggregateVariable)
418 {
419 if (head != null && head.Terms.Any(t => t.Anonymous)) Fail("anonymous variable in head");
420 if (body.Any(l => l.Atom == null && l.Terms().Any(t => t.Anonymous))) Fail("anonymous variable in comparison");
421 HashSet<string> bound = new(body.Where(l => l.Atom != null && !l.Negative).SelectMany(l => Variables(l.Atom!.Terms)), StringComparer.Ordinal);
422 List<string> needed = head == null ? new() : Variables(head.Terms);
423 if (aggregateVariable != null) needed.Add(aggregateVariable);
424 foreach (Literal l in body) if (l.Atom == null || l.Negative) needed.AddRange(Variables(l.Terms()));
425 foreach (string name in needed) if (!bound.Contains(name)) Fail("unsafe variable " + name);
426 }

428 static Dictionary<string, int>? Strata(IEnumerable<string> predicates, List<Rule> rules)
429 {
430 Dictionary<string, int> result = predicates.ToDictionary(p => p, _ => 0, StringComparer.Ordinal);
431 for (int pass = 0; pass < result.Count; pass++)
432 {
433 bool changed = false;
434 foreach (Rule r in rules) if (r.Head != null) foreach (Literal l in r.Body) if (l.Atom != null)
435 {
436 int value = result[l.Atom.Name] + (l.Negative ? 1 : 0);
437 if (result[r.Head.Name] < value)
438 {
439 result[r.Head.Name] = value;
440 changed = true;
441 }
442 }
443 if (!changed) return result;
444 if (pass + 1 == result.Count) return null;
445 }
446 return result;
447 }

451 static Dictionary<string, Value>? Match(Atom atom, Tuple tuple, Dictionary<string, Value> original)
452 {
453 Dictionary<string, Value> env = new(original, StringComparer.Ordinal);
454 for (int i = 0; i < atom.Terms.Count; i++)
455 {
456 Term t = atom.Terms[i];
457 Value value = tuple.Values[i];
458 if (t.Anonymous) continue;
459 if (t.Value != null)
460 {
461 if (t.Value != value) return null;
462 }
463 else if (env.TryGetValue(t.Variable!, out Value? old))
464 {
465 if (old != value) return null;
466 }
467 else env[t.Variable!] = value;
468 }
469 return env;
470 }

472 static Tuple Key(List<string> vars, Dictionary<string, Value> env) => new(vars.Select(v => env[v]));

474 static bool Filters(List<Literal> body, Model model, Dictionary<string, Value> env, bool complete)
475 {
476 foreach (Literal l in body) if (l.Atom == null || l.Negative)
477 {
478 if (!complete && !Variables(l.Terms()).All(env.ContainsKey)) continue;
479 if (l.Atom == null)
480 {
481 if (!l.Comparison(env)) return false;
482 }
483 else if (model.Tuples[l.Atom.Name].Keys.Any(t => Match(l.Atom, t, env) != null)) return false;
484 }
485 return true;
486 }

488 static List<Dictionary<string, Value>> Solutions(List<Literal> body, Model model, int before = -1)
489 {
490 var positive = body.Where(l => l.Atom != null && !l.Negative).OrderBy(l => model.Tuples[l.Atom!.Name].Count).ToList();
491 List<string> variables = Variables(BodyTerms(body));
492 SortedDictionary<Tuple, Dictionary<string, Value>> output = new();
493
494 void Visit(int at, Dictionary<string, Value> env)
495 {
496 if (!Filters(body, model, env, at == positive.Count)) return;
497 if (at == positive.Count)
498 {
499 output.TryAdd(Key(variables, env), env);
500 return;
501 }
502 Atom atom = positive[at].Atom!;
503 List<string> names = Variables(atom.Terms);
504 SortedSet<Tuple> seen = new();
505 foreach (var entry in model.Tuples[atom.Name])
506 {
507 if (before >= 0 && entry.Value.Round >= before) continue;
508 var candidate = Match(atom, entry.Key, env);
509 if (candidate != null && seen.Add(Key(names, candidate))) Visit(at + 1, candidate);
510 }
511 }
512
513 Visit(0, new(StringComparer.Ordinal));
514 return output.Values.ToList();
515 }

517 Model BuildModel()
518 {
519 var levels = Strata(state.Predicates.Keys, state.Rules)!;
520 int maximum = levels.Values.DefaultIfEmpty(-1).Max();
521 Model model = new() { Rounds = new int[maximum + 1], Added = new int[maximum + 1] };
522 foreach (var e in state.Predicates)
523 {
524 SortedDictionary<Tuple, Derivation> tuples = new();
525 foreach (Tuple t in e.Value.Facts) tuples[t] = new(0, 0);
526 model.Tuples[e.Key] = tuples;
527 }
528 int serial = 0;
529 for (int level = 0; level <= maximum; level++) while (true)
530 {
531 SortedDictionary<string, SortedDictionary<Tuple, int>> staged = new(StringComparer.Ordinal);
532 foreach (Rule r in state.Rules) if (r.Head != null && levels[r.Head.Name] == level)
533 {
534 foreach (var env in Solutions(r.Body, model))
535 {
536 Tuple t = r.Head.Tuple(env);
537 if (model.Tuples[r.Head.Name].ContainsKey(t)) continue;
538 if (!staged.TryGetValue(r.Head.Name, out var tuples)) staged[r.Head.Name] = tuples = new();
539 if (!tuples.TryGetValue(t, out int old) || old > r.Number) tuples[t] = r.Number;
540 }
541 }
542 if (staged.Count == 0) break;
543 serial++;
544 model.Rounds[level]++;
545 foreach (var p in staged) foreach (var t in p.Value)
546 {
547 model.Tuples[p.Key][t.Key] = new(serial, t.Value);
548 model.Added[level]++;
549 }
550 }
551 return model;
552 }

554 static string TupleText(string predicate, Tuple tuple) => predicate + (tuple.Values.Length == 0 ? "" : "(" + string.Join(", ", tuple.Values.Select(v => v.ToString())) + ")");
555
556 static IEnumerable<string> AnswerLines(List<string> vars, List<Dictionary<string, Value>> answers, string prefix) => answers.Select(env => prefix + string.Join(", ", vars.Select(v => v + " = " + env[v])));

558 void Proof(string predicate, Tuple tuple, Model model, int depth, List<string> output)
559 {
560 string prefix = new string(' ', 2 * depth) + TupleText(predicate, tuple);
561 Derivation derivation = model.Tuples[predicate][tuple];
562 if (derivation.Round == 0)
563 {
564 output.Add(prefix + " [fact]");
565 return;
566 }
567 foreach (Rule r in state.Rules) if (r.Head != null && r.Head.Name == predicate)
568 {
569 List<string> vars = Variables(r.Terms());
570 SortedDictionary<Tuple, Dictionary<string, Value>> choices = new();
571 foreach (var env in Solutions(r.Body, model, derivation.Round)) if (r.Head.Tuple(env).CompareTo(tuple) == 0) choices[Key(vars, env)] = env;
572 if (choices.Count == 0) continue;
573 var chosen = choices.First().Value;
574 output.Add(prefix + " [rule " + r.Number + "]");
575 foreach (Literal l in r.Body)
576 {
577 if (l.Atom != null && !l.Negative)
578 {
579 foreach (var t in model.Tuples[l.Atom.Name]) if (t.Value.Round < derivation.Round && Match(l.Atom, t.Key, chosen) != null)
580 {
581 Proof(l.Atom.Name, t.Key, model, depth + 1, output);
582 break;
583 }
584 }
585 else output.Add(new string(' ', 2 * (depth + 1)) + l.Text(chosen));
586 }
587 return;
588 }
589 }

591 List<string> Process(Statement s, int line)
592 {
593 List<string> output = new();
594 if (s.Kind is "fact" or "rule")
595 {
596 var arities = PredicateCheck(s.Atom, s.Body, true);
597 Safety(s.Atom, s.Body, null);
598 Rule rule = new(nextRule, s.Atom, s.Body);
599 if (s.Kind == "rule")
600 {
601 foreach (Rule old in state.Rules) if (old.Text() == rule.Text()) Fail("duplicate of rule " + old.Number);
602 var trial = state.Rules.Append(rule).ToList();
603 if (s.Atom != null && Strata(arities.Keys, trial) == null) Fail("negative cycle through " + s.Atom.Name);
604 }
605 else if (state.Predicates.TryGetValue(s.Atom!.Name, out var predicate) && predicate.Facts.Contains(s.Atom.Tuple(new()))) return output;
606 Snapshot(line);
607 foreach (var e in arities) if (!state.Predicates.ContainsKey(e.Key)) state.Predicates[e.Key] = new(e.Value);
608 if (s.Kind == "fact") state.Predicates[s.Atom!.Name].Facts.Add(s.Atom.Tuple(new()));
609 else
610 {
611 state.Rules.Add(rule);
612 nextRule++;
613 foreach (var group in rule.Terms().Where(t => t.Variable != null).GroupBy(t => t.Variable, StringComparer.Ordinal)) if (group.Count() == 1) output.Add($"WARNING line {line}: singleton variable {group.Key}");
614 }
615 return output;
616 }
617 if (s.Kind == "undo")
618 {
619 if (history.Count == 0) Fail("nothing to undo");
620 History h = history[^1];
621 history.RemoveAt(history.Count - 1);
622 state = h.State;
623 return new() { "undid line " + h.Line };
624 }
625 if (s.Kind == "forget")
626 {
627 Rule? rule = state.Rules.FirstOrDefault(r => r.Number == s.Number);
628 if (rule == null) Fail("no rule " + s.Number);
629 Snapshot(line);
630 state.Rules.Remove(rule!);
631 return output;
632 }
633 if (s.Kind is "retract" or "why") PredicateCheck(s.Atom, new(), false);
634 if (s.Kind == "retract")
635 {
636 var removed = state.Predicates[s.Atom!.Name].Facts.Where(t => Match(s.Atom, t, new()) != null).ToList();
637 if (removed.Count == 0) Fail("no matching facts");
638 Snapshot(line);
639 state.Predicates[s.Atom.Name].Facts.ExceptWith(removed);
640 return new() { removed.Count + (removed.Count == 1 ? " fact retracted" : " facts retracted") };
641 }
642 if (s.Kind is "listing" or "dump")
643 {
644 if (s.Name != null && !state.Predicates.ContainsKey(s.Name)) Fail("unknown predicate " + s.Name);
645 if (s.Kind == "listing")
646 {
647 foreach (Rule r in state.Rules) if (s.Name == null || r.Head != null && r.Head.Name == s.Name) output.Add(r.Number + ": " + r.Text());
648 return output.Count == 0 ? new() { "no rules" } : output;
649 }
650 }
651 if (s.Kind == "query")
652 {
653 PredicateCheck(null, s.Body, false);
654 Safety(null, s.Body, s.Variable);
655 }
656 if (s.Kind == "why" && s.Atom!.Terms.Any(t => t.Value == null)) Fail("why needs a ground atom");
657 var levels = Strata(state.Predicates.Keys, state.Rules)!;
658 if (s.Kind == "strata")
659 {
660 if (levels.Count == 0) return new() { "no predicates" };
661 List<string> names = state.Predicates.Keys.ToList();
662 int n = names.Count;
663 bool[,] reach = new bool[n, n];
664 foreach (Rule r in state.Rules) if (r.Head != null) foreach (Literal l in r.Body) if (l.Atom != null) reach[names.IndexOf(r.Head.Name), names.IndexOf(l.Atom.Name)] = true;
665 for (int k = 0; k < n; k++) for (int i = 0; i < n; i++) for (int j = 0; j < n; j++) reach[i, j] |= reach[i, k] && reach[k, j];
666 for (int k = 0; k <= levels.Values.Max(); k++) output.Add("stratum " + k + ": " + string.Join(", ", names.Select((name, i) => levels[name] == k ? name + (reach[i, i] ? "*" : "") : null).Where(name => name != null)));
667 return output;
668 }
669 Model model = BuildModel();
670 switch (s.Kind)
671 {
672 case "query":
673 {
674 var answers = Solutions(s.Body, model);
675 var vars = Variables(BodyTerms(s.Body));
676 if (s.Aggregate == "count") output.Add(answers.Count.ToString());
677 else if (s.Aggregate == "sum")
678 {
679 long total = 0;
680 foreach (var env in answers)
681 {
682 Value v = env[s.Variable!];
683 if (!v.Integer) Fail("non-integer value in sum");
684 total += v.Number;
685 }
686 output.Add(total.ToString());
687 }
688 else if (s.Aggregate is "min" or "max")
689 {
690 Value? best = null;
691 foreach (var env in answers)
692 {
693 Value v = env[s.Variable!];
694 if (best == null || (s.Aggregate == "min" ? v.CompareTo(best) < 0 : v.CompareTo(best) > 0)) best = v;
695 }
696 output.Add(best?.ToString() ?? "no answers");
697 }
698 else if (vars.Count == 0) output.Add(answers.Count == 0 ? "false." : "true.");
699 else
700 {
701 output.AddRange(AnswerLines(vars, answers, ""));
702 output.Add(answers.Count + (answers.Count == 1 ? " answer" : " answers"));
703 }
704 break;
705 }
706 case "dump":
707 foreach (Tuple t in model.Tuples[s.Name!].Keys) output.Add(TupleText(s.Name!, t) + ".");
708 int count = model.Tuples[s.Name!].Count;
709 output.Add(count + (count == 1 ? " tuple" : " tuples"));
710 break;
711 case "why":
712 Tuple tuple = s.Atom!.Tuple(new());
713 if (!model.Tuples[s.Atom.Name].ContainsKey(tuple)) output.Add("false.");
714 else Proof(s.Atom.Name, tuple, model, 0, output);
715 break;
716 case "check":
717 foreach (Rule r in state.Rules) if (r.Head == null)
718 {
719 var answers = Solutions(r.Body, model);
720 var vars = Variables(BodyTerms(r.Body));
721 output.Add("constraint " + r.Number + ": " + (answers.Count == 0 ? "ok" : answers.Count + (answers.Count == 1 ? " violation" : " violations")));
722 if (vars.Count > 0) output.AddRange(AnswerLines(vars, answers, " "));
723 }
724 if (output.Count == 0) output.Add("no constraints");
725 break;
726 case "predicates":
727 foreach (var e in state.Predicates)
728 {
729 int rules = state.Rules.Count(r => r.Head != null && r.Head.Name == e.Key);
730 output.Add($"{e.Key}/{e.Value.Arity}: facts {e.Value.Facts.Count}, rules {rules}, tuples {model.Tuples[e.Key].Count}");
731 }
732 if (output.Count == 0) output.Add("no predicates");
733 break;
734 case "stats":
735 int facts = state.Predicates.Values.Sum(p => p.Facts.Count), tuples = model.Tuples.Values.Sum(p => p.Count), ruleCount = state.Rules.Count(r => r.Head != null);
736 output.Add($"facts {facts}, rules {ruleCount}, constraints {state.Rules.Count - ruleCount}, tuples {tuples}");
737 for (int k = 0; k < model.Rounds.Length; k++) output.Add($"stratum {k}: rounds {model.Rounds[k]}, new {model.Added[k]}");
738 foreach (Rule r in state.Rules) if (r.Head != null) output.Add($"rule {r.Number}: new {model.Tuples[r.Head.Name].Values.Count(d => d.Rule == r.Number)}");
739 break;
740 }
741 return output;
742 }
