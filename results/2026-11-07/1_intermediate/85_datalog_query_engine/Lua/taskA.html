33local function lower(c)
34 return c ~= "" and c >= "a" and c <= "z"
35end
36local function upper(c)
37 return c ~= "" and c >= "A" and c <= "Z"
38end
39local function digit(c)
40 return c ~= "" and c >= "0" and c <= "9"
41end
42local function word_character(c)
43 return lower(c) or upper(c) or digit(c) or c == "_"
44end
45local function lexical(line)
46 local result, i = {}, 1
47 while i <= #line do
48 local c, start = line:sub(i, i), i
49 if c == " " then
50 i = i + 1
51 elseif c == "%" then
52 break
53 elseif c == '"' then
54 local closing = line:find('"', i + 1, true)
55 if not closing then
56 fail("unterminated string")
57 end
58 result[#result + 1] = {kind = "string", text = line:sub(i, closing)}
59 i = closing + 1
60 elseif contains({":-", "?-", "!=", "<=", ">="}, line:sub(i, i + 1)) then
61 local text = line:sub(i, i + 1)
62 result[#result + 1] = {kind = text, text = text}
63 i = i + 2
64 elseif ("(),.:=<>+-*/"):find(c, 1, true) then
65 result[#result + 1] = {kind = c, text = c}
66 i = i + 1
67 else
68 if not word_character(c) then
69 fail("bad character '%s'", c)
70 end
71 repeat
72 i = i + 1
73 until not word_character(line:sub(i, i))
74 local text = line:sub(start, i - 1)
75 local item = {text = text}
76 if lower(c) then
77 item.kind = reserved[text] and text or "name"
78 elseif upper(c) then
79 item.kind = "variable"
80 elseif text == "_" then
81 item.kind = "anonymous"
82 else
83 if not text:match("^%d+$") then
84 fail("bad token '%s'", text)
85 end
86 item.kind, item.number = "integer", tonumber(text)
87 if item.number > 1000000000 then
88 fail("integer out of range '%s'", text)
89 end
90 end
91 result[#result + 1] = item
92 end
93 end
94 return result
95end
96local Parser = {}
97Parser.__index = Parser
98function Parser:peek()
99 return self.tokens[self.position] and self.tokens[self.position].kind or "end"
100end
101function Parser:invalid()
102 if self:peek() == "end" then
103 fail("syntax error at end of line")
104 end
105 fail("syntax error at '%s'", self.tokens[self.position].text)
106end
107function Parser:take(kind)
108 if self:peek() ~= kind then
109 self:invalid()
110 end
111 local item = self.tokens[self.position]
112 self.position = self.position + 1
113 return item
114end
115function Parser:term()
116 local kind = self:peek()
117 if kind == "name" then
118 return {kind = "constant", value = self:take(kind).text}
119 elseif kind == "string" then
120 return {kind = "constant", value = self:take(kind).text:sub(2, -2)}
121 elseif kind == "integer" then
122 return {kind = "constant", value = self:take(kind).number}
123 elseif kind == "-" then
124 self:take("-")
125 return {kind = "constant", value = -self:take("integer").number}
126 elseif kind == "variable" then
127 return {kind = kind, name = self:take(kind).text}
128 elseif kind == "anonymous" then
129 self:take(kind)
130 return {kind = kind}
131 end
132 self:invalid()
133end
134function Parser:atom()
135 local result = {name = self:take("name").text, terms = {}}
136 if self:peek() == "(" then
137 self:take("(")
138 result.terms[1] = self:term()
139 while self:peek() == "," do
140 self:take(",")
141 result.terms[#result.terms + 1] = self:term()
142 end
143 self:take(")")
144 end
145 return result
146end
147function Parser:factor()
148 if self:peek() == "(" then
149 self:take("(")
150 local inside = self:expression()
151 self:take(")")
152 return {kind = "group", left = inside}
153 end
154 return {kind = "term", term = self:term()}
155end
156function Parser:product()
157 local result = self:factor()
158 while contains({"*", "/", "mod"}, self:peek()) do
159 local operator = self:take(self:peek()).kind
160 result = {kind = "operation", operator = operator, left = result, right = self:factor()}
161 end
162 return result
163end
164function Parser:expression()
165 local result = self:product()
166 while self:peek() == "+" or self:peek() == "-" do
167 local operator = self:take(self:peek()).kind
168 result = {kind = "operation", operator = operator, left = result, right = self:product()}
169 end
170 return result
171end
172function Parser:literal()
173 if self:peek() == "not" then
174 self:take("not")
175 return {kind = "negative", atom = self:atom()}
176 end
177 if self:peek() == "name" then
178 local next_kind =
179 self.tokens[self.position + 1] and self.tokens[self.position + 1].kind or "end"
180 if not contains({"=", "!=", "<", "<=", ">", ">=", "+", "-", "*", "/", "mod"}, next_kind) then
181 return {kind = "positive", atom = self:atom()}
182 end
183 end
184 local left = self:expression()
185 if not contains({"=", "!=", "<", "<=", ">", ">="}, self:peek()) then
186 self:invalid()
187 end
188 local operator = self:take(self:peek()).kind
189 return {kind = "comparison", operator = operator, left = left, right = self:expression()}
190end
191function Parser:body()
192 local result = {self:literal()}
193 while self:peek() == "," do
194 self:take(",")
195 result[#result + 1] = self:literal()
196 end
197 return result
198end
199function Parser:statement()
200 local kind = self:peek()
201 local result = {body = {}}
202 if kind == "name" then
203 result.head, result.kind = self:atom(), "fact"
204 if self:peek() == ":-" then
205 self:take(":-")
206 result.kind, result.body = "rule", self:body()
207 end
208 elseif kind == ":-" then
209 self:take(kind)
210 result.kind, result.body = "constraint", self:body()
211 elseif kind == "?-" then
212 self:take(kind)
213 result.kind = "query"
214 if contains({"count", "sum", "min", "max"}, self:peek()) then
215 result.aggregate = self:take(self:peek()).kind
216 if result.aggregate ~= "count" then
217 result.variable = self:take("variable").text
218 self:take(":")
219 end
220 end
221 result.body = self:body()
222 elseif kind == "retract" or kind == "why" then
223 self:take(kind)
224 result.kind, result.head = kind, self:atom()
225 elseif kind == "dump" then
226 self:take(kind)
227 result.kind, result.name = kind, self:take("name").text
228 elseif kind == "forget" then
229 self:take(kind)
230 result.kind, result.number = kind, self:take("integer").number
231 elseif kind == "listing" then
232 self:take(kind)
233 result.kind = kind
234 if self:peek() == "name" then
235 result.name = self:take("name").text
236 end
237 elseif contains({"strata", "stats", "check", "predicates", "undo", "halt"}, kind) then
238 self:take(kind)
239 result.kind = kind
240 else
241 self:invalid()
242 end
243 self:take(".")
244 if self:peek() ~= "end" then
245 self:invalid()
246 end
247 return result
248end
249local function compare_value(a, b)
250 if type(a) ~= type(b) then
251 return type(a) == "number" and -1 or 1
252 end
253 return a < b and -1 or a > b and 1 or 0
254end
255local function compare_tuple(a, b)
256 for i = 1, #a do
257 local order = compare_value(a[i], b[i])
258 if order ~= 0 then
259 return order
260 end
261 end
262 return 0
263end
264local function notation(value)
265 if type(value) == "number" then
266 return tostring(value)
267 end
268 if value:match("^[a-z][a-zA-Z0-9_]*$") and not reserved[value] then
269 return value
270 end
271 return '"' .. value .. '"'
272end
273local function term_text(term, assignment)
274 if term.kind == "constant" then
275 return notation(term.value)
276 end
277 if term.kind == "anonymous" then
278 return "_"
279 end
280 return assignment and notation(assignment[term.name]) or term.name
281end
282local function atom_text(atom, assignment)
283 if #atom.terms == 0 then
284 return atom.name
285 end
286 local parts = {}
287 for _, term in ipairs(atom.terms) do
288 parts[#parts + 1] = term_text(term, assignment)
289 end
290 return atom.name .. "(" .. table.concat(parts, ", ") .. ")"
291end
292local function expression_text(expression, assignment)
293 if expression.kind == "term" then
294 return term_text(expression.term, assignment)
295 end
296 if expression.kind == "group" then
297 return "(" .. expression_text(expression.left, assignment) .. ")"
298 end
299 return expression_text(expression.left, assignment) ..
300 " " .. expression.operator .. " " .. expression_text(expression.right, assignment)
301end
302local function literal_text(literal, assignment)
303 if literal.kind == "comparison" then
304 return expression_text(literal.left, assignment) ..
305 " " .. literal.operator .. " " .. expression_text(literal.right, assignment)
306 end
307 return (literal.kind == "negative" and "not " or "") .. atom_text(literal.atom, assignment)
308end
309local function rule_text(statement)
310 local parts = {}
311 for _, literal in ipairs(statement.body) do
312 parts[#parts + 1] = literal_text(literal)
313 end
314 return (statement.head and atom_text(statement.head) .. " :- " or ":- ") ..
315 table.concat(parts, ", ") .. "."
316end
317local function expression_terms(expression, result)
318 if expression.kind == "term" then
319 result[#result + 1] = expression.term
320 else
321 expression_terms(expression.left, result)
322 if expression.kind == "operation" then
323 expression_terms(expression.right, result)
324 end
325 end
326end
327local function literal_terms(literal)
328 if literal.kind ~= "comparison" then
329 return literal.atom.terms
330 end
331 local result = {}
332 expression_terms(literal.left, result)
333 expression_terms(literal.right, result)
334 return result
335end

338 if statement.head then
339 for _, term in ipairs(statement.head.terms) do
340 result[#result + 1] = term
341 end
342 end
343 for _, literal in ipairs(statement.body) do
344 for _, term in ipairs(literal_terms(literal)) do
345 result[#result + 1] = term
346 end
347 end
348 return result

362 for _, value in ipairs(tuple) do

367end

372 end

408function Database:check_predicates(statement, clause)
409 local arities = {}
410 for name, predicate in pairs(self.predicates) do
411 arities[name] = predicate.arity
412 end
413 for _, atom in ipairs(atoms(statement)) do
414 local expected = arities[atom.name]
415 if expected == nil and not clause then
416 fail("unknown predicate %s", atom.name)
417 end
418 if expected ~= nil and expected ~= #atom.terms then
419 fail("arity mismatch for %s: expected %d, got %d", atom.name, expected, #atom.terms)
420 end
421 arities[atom.name] = #atom.terms
422 end
423 return arities
424end
425local function safety(statement)
426 if statement.head then
427 for _, term in ipairs(statement.head.terms) do
428 if term.kind == "anonymous" then
429 fail("anonymous variable in head")
430 end
431 end
432 end
433 local bound = {}
434 for _, literal in ipairs(statement.body) do
435 if literal.kind == "comparison" then
436 for _, term in ipairs(literal_terms(literal)) do
437 if term.kind == "anonymous" then
438 fail("anonymous variable in comparison")
439 end
440 end
441 end
442 if literal.kind == "positive" then
443 for _, name in ipairs(variables(literal.atom.terms)) do
444 bound[name] = true
445 end
446 end
447 end
448 local checked = {}
449 if statement.head then
450 for _, term in ipairs(statement.head.terms) do
451 checked[#checked + 1] = term
452 end
453 end
454 if statement.aggregate and statement.aggregate ~= "count" then
455 checked[#checked + 1] = {kind = "variable", name = statement.variable}
456 end
457 for _, literal in ipairs(statement.body) do
458 if literal.kind ~= "positive" then
459 for _, term in ipairs(literal_terms(literal)) do
460 checked[#checked + 1] = term
461 end
462 end
463 end
464 for _, term in ipairs(checked) do
465 if term.kind == "variable" and not bound[term.name] then
466 fail("unsafe variable %s", term.name)
467 end
468 end
469end
470local function strata(arities, rules)
471 local names = keys(arities)
472 local reach, levels, recursive = {}, {}, {}
473 for _, name in ipairs(names) do
474 reach[name] = {}
475 levels[name] = 0
476 end
477 for _, rule in ipairs(rules) do
478 if rule.statement.head then
479 for _, literal in ipairs(rule.statement.body) do
480 if literal.kind ~= "comparison" then
481 reach[rule.statement.head.name][literal.atom.name] = true
482 end
483 end
484 end
485 end
486 for _, middle in ipairs(names) do
487 for _, a in ipairs(names) do
488 for _, b in ipairs(names) do
489 if reach[a][middle] and reach[middle][b] then
490 reach[a][b] = true
491 end
492 end
493 end
494 end
495 for _, name in ipairs(names) do
496 recursive[name] = reach[name][name] or false
497 end
498 for _, rule in ipairs(rules) do
499 if rule.statement.head then
500 for _, literal in ipairs(rule.statement.body) do
501 if
502 literal.kind == "negative" and
503 (literal.atom.name == rule.statement.head.name or
504 reach[literal.atom.name][rule.statement.head.name])
505 then
506 return nil
507 end
508 end
509 end
510 end
511 local changed = true
512 while changed do
513 changed = false
514 for _, rule in ipairs(rules) do
515 if rule.statement.head then
516 for _, literal in ipairs(rule.statement.body) do
517 if literal.kind ~= "comparison" then
518 local required =
519 levels[literal.atom.name] + (literal.kind == "negative" and 1 or 0)
520 if levels[rule.statement.head.name] < required then
521 levels[rule.statement.head.name] = required
522 changed = true
523 end
524 end
525 end
526 end
527 end
528 end
529 return levels, recursive
530end
531function Database:levels()
532 local arities = {}
533 for name, predicate in pairs(self.predicates) do
534 arities[name] = predicate.arity
535 end
536 return strata(arities, self.rules)
537end
538local function match(atom, values, assignment)
539 local result = copy(assignment or {})
540 for i, term in ipairs(atom.terms) do
541 if term.kind == "constant" and term.value ~= values[i] then
542 return nil
543 end
544 if term.kind == "variable" then
545 if result[term.name] ~= nil and result[term.name] ~= values[i] then
546 return nil
547 end
548 result[term.name] = values[i]
549 end
550 end
551 return result
552end
553local function evaluate(expression, assignment)
554 if expression.kind == "term" then
555 return expression.term.kind == "constant" and expression.term.value or
556 assignment[expression.term.name]
557 end
558 if expression.kind == "group" then
559 return evaluate(expression.left, assignment)
560 end
561 local a, b = evaluate(expression.left, assignment), evaluate(expression.right, assignment)
562 if type(a) ~= "number" or type(b) ~= "number" then
563 return nil
564 end
565 local operator = expression.operator
566 if operator == "+" then
567 return a + b
568 elseif operator == "-" then
569 return a - b
570 elseif operator == "*" then
571 return a * b
572 end
573 if b == 0 then
574 return nil
575 end
576 local quotient = a // b
577 return operator == "mod" and a - b * quotient or quotient
578end
579local function comparison(literal, assignment)
580 local a, b = evaluate(literal.left, assignment), evaluate(literal.right, assignment)
581 if a == nil or b == nil then
582 return false
583 end
584 local operator = literal.operator
585 if operator == "=" then
586 return a == b
587 elseif operator == "!=" then
588 return a ~= b
589 end
590 if type(a) ~= "number" or type(b) ~= "number" then
591 return false
592 end
593 if operator == "<" then
594 return a < b
595 elseif operator == "<=" then
596 return a <= b
597 elseif operator == ">" then
598 return a > b
599 else
600 return a >= b
601 end
602end
603local function solutions(body, model, before)
604 local positives = {}
605 for _, literal in ipairs(body) do
606 if literal.kind == "positive" then
607 positives[#positives + 1] = literal
608 end
609 end
610 local function accepts_binding(binding)
611 for _, literal in ipairs(body) do
612 if literal.kind ~= "positive" then
613 local ready = true
614 for _, term in ipairs(literal_terms(literal)) do
615 if term.kind == "variable" and binding[term.name] == nil then
616 ready = false
617 break
618 end
619 end
620 if ready then
621 if literal.kind == "comparison" and not comparison(literal, binding) then
622 return false
623 end
624 if literal.kind == "negative" then
625 for _, tuple in pairs(model[literal.atom.name]) do
626 if match(literal.atom, tuple.values, binding) then
627 return false
628 end
629 end
630 end
631 end
632 end
633 end
634 return true
635 end
636 local result, finished = {}, {}
637 local function binding_key(binding)
638 local values = {}
639 for _, name in ipairs(keys(binding)) do
640 values[#values + 1] = binding[name]
641 end
642 return tuple_key(values)
643 end
644 local function visit(binding, remaining)
645 if not accepts_binding(binding) then
646 return
647 end
648 if #remaining == 0 then
649 local key = binding_key(binding)
650 if not finished[key] then
651 finished[key] = true
652 result[#result + 1] = binding
653 end
654 return
655 end
656 local selected, candidates
657 for i, literal in ipairs(remaining) do
658 local matches, seen = {}, {}
659 for _, tuple in pairs(model[literal.atom.name]) do
660 if not before or tuple.round < before then
661 local matched = match(literal.atom, tuple.values, binding)
662 if matched and accepts_binding(matched) then
663 local key = binding_key(matched)
664 if not seen[key] then
665 seen[key] = true
666 matches[#matches + 1] = matched
667 end
668 end
669 end
670 end
671 if #matches == 0 then
672 return
673 end
674 if not candidates or #matches < #candidates then
675 selected, candidates = i, matches
676 end
677 end
678 local next_remaining = {}
679 for i, literal in ipairs(remaining) do
680 if i ~= selected then
681 next_remaining[#next_remaining + 1] = literal
682 end
683 end
684 for _, matched in ipairs(candidates) do
685 visit(matched, next_remaining)
686 end
687 end
688 visit({}, positives)
689 return result
690end
691local function atom_values(atom, assignment)
692 local result = {}
693 for _, term in ipairs(atom.terms) do
694 result[#result + 1] = term.kind == "constant" and term.value or assignment[term.name]
695 end
696 return result
697end
698function Database:model()
699 local levels = self:levels()
700 local result = {tuples = {}, levels = levels, rounds = {}, added = {}}
701 local maximum = -1
702 for name, predicate in pairs(self.predicates) do
703 result.tuples[name] = {}
704 for key, values in pairs(predicate.facts) do
705 result.tuples[name][key] = {values = values, round = 0, rule = 0}
706 end
707 maximum = math.max(maximum, levels[name])
708 end
709 local serial = 0
710 for level = 0, maximum do
711 result.rounds[level], result.added[level] = 0, 0
712 while true do
713 local pending, count = {}, 0
714 for _, rule in ipairs(self.rules) do
715 local head = rule.statement.head
716 if head and levels[head.name] == level then
717 pending[head.name] = pending[head.name] or {}
718 for _, assignment in ipairs(solutions(rule.statement.body, result.tuples)) do
719 local values = atom_values(head, assignment)
720 local key = tuple_key(values)
721 if not result.tuples[head.name][key] and not pending[head.name][key] then
722 pending[head.name][key] = {values = values, rule = rule.number}
723 count = count + 1
724 end
725 end
726 end
727 end
728 if count == 0 then
729 break
730 end
731 serial = serial + 1
732 result.rounds[level] = result.rounds[level] + 1
733 result.added[level] = result.added[level] + count
734 for name, set in pairs(pending) do
735 for key, tuple in pairs(set) do
736 tuple.round = serial
737 result.tuples[name][key] = tuple
738 end
739 end
740 end
741 end
742 result.maximum = maximum
743 return result
744end
745local function answer_rows(statement, model)
746 local names = variables(all_terms({body = statement.body}))
747 local rows = {}
748 for _, assignment in ipairs(solutions(statement.body, model.tuples)) do
749 local row = {}
750 for _, name in ipairs(names) do
751 row[#row + 1] = assignment[name]
752 end
753 rows[#rows + 1] = row
754 end
755 table.sort(
756 rows,
757 function(a, b)
758 return compare_tuple(a, b) < 0
759 end
760 )
761 return names, rows
762end

770local function tuple_atom(name, values)
771 local atom = {name = name, terms = {}}
772 for _, value in ipairs(values) do

777function Database:proof(name, target, depth, model)
778 local prefix = string.rep(" ", depth * 2) .. atom_text(tuple_atom(name, target.values))
779 if target.round == 0 then
780 return {prefix .. " [fact]"}
781 end
782 for _, rule in ipairs(self.rules) do
783 local statement = rule.statement
784 if statement.head and statement.head.name == name then
785 local candidates = {}
786 for _, binding in ipairs(solutions(statement.body, model.tuples, target.round)) do
787 if compare_tuple(atom_values(statement.head, binding), target.values) == 0 then
788 candidates[#candidates + 1] = binding
789 end
790 end
791 if #candidates > 0 then
792 local names = variables(all_terms(statement))
793 table.sort(
794 candidates,
795 function(a, b)
796 for _, variable in ipairs(names) do
797 local order = compare_value(a[variable], b[variable])
798 if order ~= 0 then
799 return order < 0
800 end
801 end
802 return false
803 end
804 )
805 local binding = candidates[1]
806 local lines = {string.format("%s [rule %d]", prefix, rule.number)}
807 for _, literal in ipairs(statement.body) do
808 if literal.kind == "positive" then
809 for _, tuple in ipairs(ordered_tuples(model.tuples[literal.atom.name])) do
810 if
811 tuple.round < target.round and
812 match(literal.atom, tuple.values, binding)
813 then
814 for _, text in ipairs(
815 self:proof(literal.atom.name, tuple, depth + 1, model)
816 ) do
817 lines[#lines + 1] = text
818 end
819 break
820 end
821 end
822 else
823 lines[#lines + 1] =
824 string.rep(" ", (depth + 1) * 2) .. literal_text(literal, binding)
825 end
826 end
827 return lines
828 end
829 end
830 end
831end
832local function plural(count, singular)
833 return count == 1 and singular or singular .. "s"
834end
835function Database:execute(statement, line)
836 local out = {}
837 local function emit(format, ...)
838 out[#out + 1] = string.format(format, ...)
839 end
840 local kind = statement.kind
841 if kind == "fact" or kind == "rule" or kind == "constraint" then
842 local arities = self:check_predicates(statement, true)
843 safety(statement)
844 if kind == "fact" then
845 local head = statement.head
846 local values = atom_values(head)
847 local key = tuple_key(values)
848 if self.predicates[head.name] and self.predicates[head.name].facts[key] then
849 return out
850 end
851 self:save(line)
852 self.predicates[head.name] = self.predicates[head.name] or {arity = #values, facts = {}}
853 self.predicates[head.name].facts[key] = values
854 return out
855 end
856 local text = rule_text(statement)
857 for _, rule in ipairs(self.rules) do
858 if rule_text(rule.statement) == text then
859 fail("duplicate of rule %d", rule.number)
860 end
861 end
862 local candidate = {}
863 for i, rule in ipairs(self.rules) do
864 candidate[i] = rule
865 end
866 candidate[#candidate + 1] = {number = self.next_number, statement = statement}
867 if kind == "rule" and not strata(arities, candidate) then
868 fail("negative cycle through %s", statement.head.name)
869 end
870 self:save(line)
871 for name, arity in pairs(arities) do
872 self.predicates[name] = self.predicates[name] or {arity = arity, facts = {}}
873 end
874 self.rules = candidate
875 self.next_number = self.next_number + 1
876 local terms = all_terms(statement)
877 for _, name in ipairs(variables(terms)) do
878 local count = 0
879 for _, term in ipairs(terms) do
880 if term.kind == "variable" and term.name == name then
881 count = count + 1
882 end
883 end
884 if count == 1 then
885 emit("WARNING line %d: singleton variable %s", line, name)
886 end
887 end
888 return out
889 end
890 if kind == "undo" then
891 if #self.history == 0 then
892 fail("nothing to undo")
893 end
894 local previous = table.remove(self.history)
895 self.predicates, self.rules = previous.predicates, previous.rules
896 emit("undid line %d", previous.line)
897 return out
898 elseif kind == "forget" then
899 local index
900 for i, rule in ipairs(self.rules) do
901 if rule.number == statement.number then
902 index = i
903 break
904 end
905 end
906 if not index then
907 fail("no rule %d", statement.number)
908 end
909 self:save(line)
910 table.remove(self.rules, index)
911 return out
912 elseif kind == "listing" then
913 if statement.name and not self.predicates[statement.name] then
914 fail("unknown predicate %s", statement.name)
915 end
916 for _, rule in ipairs(self.rules) do
917 if
918 not statement.name or
919 rule.statement.head and rule.statement.head.name == statement.name
920 then
921 emit("%d: %s", rule.number, rule_text(rule.statement))
922 end
923 end
924 if #out == 0 then
925 emit("no rules")
926 end
927 return out
928 elseif kind == "strata" then
929 local levels, recursive = self:levels()
930 local maximum = -1
931 for _, level in pairs(levels) do
932 maximum = math.max(maximum, level)
933 end
934 if maximum < 0 then
935 emit("no predicates")
936 end
937 for level = 0, maximum do
938 local names = {}
939 for _, name in ipairs(keys(self.predicates)) do
940 if levels[name] == level then
941 names[#names + 1] = name .. (recursive[name] and "*" or "")
942 end
943 end
944 emit("stratum %d: %s", level, table.concat(names, ", "))
945 end
946 return out
947 end
948 if kind == "query" or kind == "retract" or kind == "why" then
949 self:check_predicates(statement, false)
950 if kind == "query" then
951 safety(statement)
952 end
953 end
954 if kind == "retract" then
955 local predicate = self.predicates[statement.head.name]
956 local removed = {}
957 for key, values in pairs(predicate.facts) do
958 if match(statement.head, values) then
959 removed[#removed + 1] = key
960 end
961 end
962 if #removed == 0 then
963 fail("no matching facts")
964 end
965 self:save(line)
966 for _, key in ipairs(removed) do
967 predicate.facts[key] = nil
968 end
969 emit("%d %s retracted", #removed, plural(#removed, "fact"))
970 return out
971 end
972 if kind == "why" then
973 for _, term in ipairs(statement.head.terms) do
974 if term.kind ~= "constant" then
975 fail("why needs a ground atom")
976 end
977 end
978 end
979 if kind == "dump" and not self.predicates[statement.name] then
980 fail("unknown predicate %s", statement.name)
981 end
982 local model = self:model()
983 if kind == "query" then
984 local names, answers = answer_rows(statement, model)
985 if statement.aggregate == "count" then
986 emit("%d", #answers)
987 elseif statement.aggregate == "sum" then
988 local column
989 for i, name in ipairs(names) do
990 if name == statement.variable then
991 column = i
992 end
993 end
994 local total = 0
995 for _, row in ipairs(answers) do
996 if type(row[column]) ~= "number" then
997 fail("non-integer value in sum")
998 end
999 total = total + row[column]
1000 end
1001 emit("%d", total)
1002 elseif statement.aggregate == "min" or statement.aggregate == "max" then
1003 if #answers == 0 then
1004 emit("no answers")
1005 else
1006 local column
1007 for i, name in ipairs(names) do
1008 if name == statement.variable then
1009 column = i
1010 end
1011 end
1012 local selected = answers[1][column]
1013 for _, row in ipairs(answers) do
1014 local order = compare_value(row[column], selected)
1015 if
1016 statement.aggregate == "min" and order < 0 or
1017 statement.aggregate == "max" and order > 0
1018 then
1019 selected = row[column]
1020 end
1021 end
1022 emit("%s", notation(selected))
1023 end
1024 elseif #names == 0 then
1025 emit(#answers > 0 and "true." or "false.")
1026 else
1027 for _, row in ipairs(answers) do
1028 emit("%s", answer_text(names, row))
1029 end
1030 emit("%d %s", #answers, plural(#answers, "answer"))
1031 end
1032 elseif kind == "dump" then
1033 local rows = ordered_tuples(model.tuples[statement.name])
1034 for _, tuple in ipairs(rows) do
1035 emit("%s.", atom_text(tuple_atom(statement.name, tuple.values)))
1036 end
1037 emit("%d %s", #rows, plural(#rows, "tuple"))
1038 elseif kind == "why" then
1039 local tuple = model.tuples[statement.head.name][tuple_key(atom_values(statement.head))]
1040 if not tuple then
1041 emit("false.")
1042 else
1043 out = self:proof(statement.head.name, tuple, 0, model)
1044 end
1045 elseif kind == "check" then
1046 local count = 0
1047 for _, rule in ipairs(self.rules) do
1048 if not rule.statement.head then
1049 count = count + 1
1050 local names, answers = answer_rows(rule.statement, model)
1051 if #answers == 0 then
1052 emit("constraint %d: ok", rule.number)
1053 else
1054 emit(
1055 "constraint %d: %d %s",
1056 rule.number,
1057 #answers,
1058 plural(#answers, "violation")
1059 )
1060 if #names > 0 then
1061 for _, row in ipairs(answers) do
1062 emit(" %s", answer_text(names, row))
1063 end
1064 end
1065 end
1066 end
1067 end
1068 if count == 0 then
1069 emit("no constraints")
1070 end
1071 elseif kind == "predicates" then
1072 if next(self.predicates) == nil then
1073 emit("no predicates")
1074 end
1075 for _, name in ipairs(keys(self.predicates)) do
1076 local count = 0
1077 for _, rule in ipairs(self.rules) do
1078 if rule.statement.head and rule.statement.head.name == name then
1079 count = count + 1
1080 end
1081 end
1082 local predicate = self.predicates[name]
1083 local facts, tuples = 0, 0
1084 for _ in pairs(predicate.facts) do
1085 facts = facts + 1
1086 end
1087 for _ in pairs(model.tuples[name]) do
1088 tuples = tuples + 1
1089 end
1090 emit(
1091 "%s/%d: facts %d, rules %d, tuples %d",
1092 name,
1093 predicate.arity,
1094 facts,
1095 count,
1096 tuples
1097 )
1098 end
1099 elseif kind == "stats" then
1100 local facts, rules, constraints, tuples = 0, 0, 0, 0
1101 for name, predicate in pairs(self.predicates) do
1102 for _ in pairs(predicate.facts) do
1103 facts = facts + 1
1104 end
1105 for _ in pairs(model.tuples[name]) do
1106 tuples = tuples + 1
1107 end
1108 end
1109 for _, rule in ipairs(self.rules) do
1110 if rule.statement.head then
1111 rules = rules + 1
1112 else
1113 constraints = constraints + 1
1114 end
1115 end
1116 emit("facts %d, rules %d, constraints %d, tuples %d", facts, rules, constraints, tuples)
1117 for level = 0, model.maximum do
1118 emit("stratum %d: rounds %d, new %d", level, model.rounds[level], model.added[level])
1119 end
1120 for _, rule in ipairs(self.rules) do
1121 if rule.statement.head then
1122 local count = 0
1123 for _, set in pairs(model.tuples) do
1124 for _, tuple in pairs(set) do
1125 if tuple.round > 0 and tuple.rule == rule.number then
1126 count = count + 1
1127 end
1128 end
1129 end
1130 emit("rule %d: new %d", rule.number, count)
1131 end
1132 end
1133 end
1134 return out
1135end
1136local database = Database.new()
1137local line_number = 0
1138for line in io.lines() do
1139 line_number = line_number + 1
1140 local halt = false
1141 local ok, result =
1142 pcall(
1143 function()
1144 local tokens = lexical(line)
1145 if #tokens == 0 then
1146 return {}
1147 end
1148 local statement = setmetatable({tokens = tokens, position = 1}, Parser):statement()
1149 if statement.kind == "halt" then
1150 halt = true
1151 return {}
1152 end
1153 return database:execute(statement, line_number)
1154 end
1155 )
1156 if not ok then
1157 if type(result) ~= "table" or not result.reason then
1158 error(result, 0)
1159 end
1160 io.write(string.format("ERROR line %d: %s\n", line_number, result.reason))
1161 else
1162 for _, text in ipairs(result) do
1163 io.write(text, "\n")
1164 end
1165 end
1166 if halt then
1167 break
1168 end
1169end
