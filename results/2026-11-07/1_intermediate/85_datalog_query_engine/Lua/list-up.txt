local function compare_value(a, b)
    if type(a) ~= type(b) then
        return type(a) == "number" and -1 or 1
    end
    return a < b and -1 or a > b and 1 or 0
end
local function compare_tuple(a, b)
    for i = 1, #a do
        local order = compare_value(a[i], b[i])
        if order ~= 0 then
            return order
        end
    end
    return 0
end
local function notation(value)
    if type(value) == "number" then
        return tostring(value)
    end
    if value:match("^[a-z][a-zA-Z0-9_]*$") and not reserved[value] then
        return value
    end
    return '"' .. value .. '"'
end
local function variables(terms)
    local result, seen = {}, {}
    for _, term in ipairs(terms) do
        if term.kind == "variable" and not seen[term.name] then
            seen[term.name] = true
            result[#result + 1] = term.name
        end
    end
    return result
end
local function tuple_key(tuple)
    local result = {}
    for _, value in ipairs(tuple) do
        result[#result + 1] =
            type(value) == "number" and "i" .. value .. ";" or "s" .. #value .. ":" .. value
    end
    return table.concat(result)
end
local function solutions(body, model, before)
    local result, finished = {}, {}
    local function binding_key(binding)
        local values = {}
        for _, name in ipairs(keys(binding)) do
            values[#values + 1] = binding[name]
        end
        return tuple_key(values)
    end
    local function visit(binding, remaining)
        if not accepts_binding(binding) then
            return
        end
        if #remaining == 0 then
            local key = binding_key(binding)
            if not finished[key] then
                finished[key] = true
                result[#result + 1] = binding
            end

    return result
end
local function answer_rows(statement, model)
    local names = variables(all_terms({body = statement.body}))
    local rows = {}
    for _, assignment in ipairs(solutions(statement.body, model.tuples)) do
        local row = {}
        for _, name in ipairs(names) do
            row[#row + 1] = assignment[name]
        end
        rows[#rows + 1] = row
    end
    table.sort(
        rows,
        function(a, b)
            return compare_tuple(a, b) < 0
        end
    )
    return names, rows
end
