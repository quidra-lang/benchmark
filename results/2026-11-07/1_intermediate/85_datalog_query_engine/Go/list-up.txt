func compareValue(a, b value) int {
	if a.integer != b.integer {
		if a.integer {
			return -1
		}
		return 1
	}
	if a.integer {
		if a.number < b.number {
			return -1
		}
		if a.number > b.number {
			return 1
		}
		return 0
	}
	return strings.Compare(a.symbol, b.symbol)
}
func compareTuple(a, b []value) int {
	for i := range a {
		if c := compareValue(a[i], b[i]); c != 0 {
			return c
		}
	}
	return 0
}
func notation(v value) string {
	if v.integer {
		return strconv.FormatInt(v.number, 10)
	}
	simple := len(v.symbol) > 0 && lower(v.symbol[0]) && !contains(reserved, v.symbol)
	for i := range v.symbol {
		if !wordCharacter(v.symbol[i]) {
			simple = false
		}
	}
	if simple {
		return v.symbol
	}
	return "\"" + v.symbol + "\""
}
func variables(terms []term) []string {
	result := []string{}
	for _, t := range terms {
		if t.kind == "variable" && !contains(result, t.name) {
			result = append(result, t.name)
		}
	}
	return result
}

func answerRows(s statement, m model) ([]string, [][]value) {
	names := variables(allTerms(statement{body: s.body}))
	result := [][]value{}
	for _, binding := range solutions(s.body, m.tuples, 0) {
		row := []value{}
		for _, name := range names {
			row = append(row, binding[name])
		}
		result = append(result, row)
	}
	sort.Slice(result, func(i, j int) bool { return compareTuple(result[i], result[j]) < 0 })
	return names, result
}
