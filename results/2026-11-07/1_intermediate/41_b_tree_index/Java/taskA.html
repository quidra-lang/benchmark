106 Search search(int key) {
107 int page = root;
108 var path = new ArrayList<String>();
109 while (true) {
110 var node = pages.get(page);
111 path.add(node.notation());
112 int position = node.position(key);
113 if (position < node.entries.size() && node.entries.get(position).key == key) return new Search(page, position, path);
114 if (node.children.isEmpty()) return new Search(null, position, path);
115 page = node.children.get(position);
116 }
117 }

119 String check(int page, String location, int depth, Integer low, Integer high, int[] reference) {
120 var node = pages.get(page);
121 int count = node.entries.size(), maximum = 2 * degree - 1;
122 int minimum = page == root ? (node.children.isEmpty() ? 0 : 1) : degree - 1;
123 String details = "at=" + location + " node=" + node.notation();
124 if (count < minimum || count > maximum) return format("COUNT %s keys=%d allowed=%d..%d", details, count, minimum, maximum);
125 for (int j = 0; j + 1 < count; j++) {
126 if (node.entries.get(j).key >= node.entries.get(j + 1).key)
127 return format("ORDER %s %d>=%d", details, node.entries.get(j).key, node.entries.get(j + 1).key);
128 }
129 for (var entry : node.entries) {
130 if (low != null && entry.key <= low || high != null && entry.key >= high)
131 return format("RANGE %s key=%d low=%s high=%s", details, entry.key, low == null ? "-inf" : low, high == null ? "+inf" : high);
132 }
133 if (node.children.isEmpty()) {
134 if (reference[0] < 0) reference[0] = depth;
135 else if (reference[0] != depth) return format("DEPTH %s depth=%d expected=%d", details, depth, reference[0]);
136 }
137 for (int j = 0; j < node.children.size(); j++) {
138 Integer lower = low, upper = high;
139 if (j > 0) lower = node.entries.get(j - 1).key;
140 if (j < count) upper = node.entries.get(j).key;
141 String problem = check(node.children.get(j), location + "." + j, depth + 1, lower, upper, reference);
142 if (problem != null) return problem;
143 }
144 return null;
145 }

147 String violation() { return check(root, "root", 0, null, null, new int[]{-1}); }
148 String status() { return violation() == null ? "OK" : "DAMAGED"; }
149 int height() {
150 int page = root, depth = 0;
151 while (!pages.get(page).children.isEmpty()) { page = pages.get(page).children.getFirst(); depth++; }
152 return depth;
153 }

162 Changes changes(Map<Integer, Node> before) {
163 int writes = 0, frees = 0;
164 for (var item : pages.entrySet()) {
165 Node old = before.get(item.getKey());
166 if (old == null || !item.getValue().sameContent(old)) writes++;
167 }
168 for (int page : before.keySet()) if (!pages.containsKey(page)) frees++;
169 return new Changes(writes, frees);
170 }

172 void split(int parentPage, int position, Trace trace) {
173 var parent = pages.get(parentPage);
174 int childPage = parent.children.get(position);
175 var child = pages.get(childPage);
176 String old = child.notation();
177 Entry middle = child.entries.get(degree - 1);
178 var right = new Node(child.entries.subList(degree, child.entries.size()),
179 child.children.isEmpty() ? List.of() : child.children.subList(degree, child.children.size()));
180 int rightPage = allocate(right);
181 child.entries.subList(degree - 1, child.entries.size()).clear();
182 if (!child.children.isEmpty()) child.children.subList(degree, child.children.size()).clear();
183 parent.entries.add(position, middle);
184 parent.children.add(position + 1, rightPage);
185 splits++;
186 trace.emit("STEP split page=%d keys=%s up=%d new=%d", childPage, old, middle.key, rightPage);
187 }

189 String upsert(Entry entry, Trace trace) {
190 var found = search(entry.key);
191 if (found.page != null) return pages.get(found.page).entries.set(found.position, entry).value;
192 if (pages.get(root).entries.size() == 2 * degree - 1) {
193 root = allocate(new Node(List.of(), List.of(root)));
194 trace.emit("STEP grow page=%d", root);
195 split(root, 0, trace);
196 }
197 int page = root;
198 while (!pages.get(page).children.isEmpty()) {
199 var node = pages.get(page);
200 int position = node.position(entry.key);
201 if (pages.get(node.children.get(position)).entries.size() == 2 * degree - 1) {
202 split(page, position, trace);
203 if (entry.key > node.entries.get(position).key) position++;
204 }
205 page = node.children.get(position);
206 }
207 var node = pages.get(page);
208 node.entries.add(node.position(entry.key), entry);
209 trace.emit("STEP place page=%d keys=%s", page, node.notation());
210 return null;
211 }

213 Entry extreme(int page, boolean last) {
214 while (true) {
215 var node = pages.get(page);
216 if (node.children.isEmpty()) return last ? node.entries.getLast() : node.entries.getFirst();
217 page = last ? node.children.getLast() : node.children.getFirst();
218 }
219 }

221 int merge(int parentPage, int position, String caseName, Trace trace) {
222 var parent = pages.get(parentPage);
223 int leftPage = parent.children.get(position), rightPage = parent.children.get(position + 1);
224 var left = pages.get(leftPage);
225 var right = pages.remove(rightPage);
226 left.entries.add(parent.entries.remove(position));
227 left.entries.addAll(right.entries);
228 left.children.addAll(right.children);
229 parent.children.remove(position + 1);
230 merges++;
231 trace.emit("STEP %s page=%d merge=%d+%d keys=%s", caseName, parentPage, leftPage, rightPage, left.notation());
232 if (parentPage == root && parent.entries.isEmpty()) {
233 root = leftPage;
234 pages.remove(parentPage);
235 trace.emit("STEP shrink page=%d", root);
236 }
237 return leftPage;
238 }

240 void removeAt(int page, int key, Trace trace) {
241 var node = pages.get(page);
242 int position = node.position(key);
243 if (position < node.entries.size() && node.entries.get(position).key == key) {
244 if (node.children.isEmpty()) {
245 trace.cases.add("1");
246 node.entries.remove(position);
247 trace.emit("STEP 1 page=%d keys=%s", page, node.notation());
248 return;
249 }
250 int left = node.children.get(position), right = node.children.get(position + 1);
251 if (pages.get(left).entries.size() >= degree) {
252 Entry predecessor = extreme(left, true);
253 trace.cases.add("2a");
254 trace.emit("STEP 2a page=%d pred=%d", page, predecessor.key);
255 removeAt(left, predecessor.key, trace);
256 node.entries.set(position, predecessor);
257 } else if (pages.get(right).entries.size() >= degree) {
258 Entry successor = extreme(right, false);
259 trace.cases.add("2b");
260 trace.emit("STEP 2b page=%d succ=%d", page, successor.key);
261 removeAt(right, successor.key, trace);
262 node.entries.set(position, successor);
263 } else {
264 trace.cases.add("2c");
265 int child = merge(page, position, "2c", trace);
266 removeAt(child, key, trace);
267 }
268 return;
269 }
270 int childPage = node.children.get(position);
271 var child = pages.get(childPage);
272 if (child.entries.size() == degree - 1) {
273 Integer left = position == 0 ? null : node.children.get(position - 1);
274 Integer right = position == node.entries.size() ? null : node.children.get(position + 1);
275 if (left != null && pages.get(left).entries.size() >= degree) {
276 var sibling = pages.get(left);
277 trace.cases.add("3a-left");
278 child.entries.addFirst(node.entries.get(position - 1));
279 node.entries.set(position - 1, sibling.entries.removeLast());
280 if (!sibling.children.isEmpty()) child.children.addFirst(sibling.children.removeLast());
281 borrows++;
282 trace.emit("STEP 3a-left page=%d child=%d from=%d keys=%s", page, childPage, left, child.notation());
283 } else if (right != null && pages.get(right).entries.size() >= degree) {
284 var sibling = pages.get(right);
285 trace.cases.add("3a-right");
286 child.entries.add(node.entries.get(position));
287 node.entries.set(position, sibling.entries.removeFirst());
288 if (!sibling.children.isEmpty()) child.children.add(sibling.children.removeFirst());
289 borrows++;
290 trace.emit("STEP 3a-right page=%d child=%d from=%d keys=%s", page, childPage, right, child.notation());
291 } else if (left != null) {
292 trace.cases.add("3b-left");
293 childPage = merge(page, position - 1, "3b-left", trace);
294 } else {
295 trace.cases.add("3b-right");
296 childPage = merge(page, position, "3b-right", trace);
297 }
298 }
299 removeAt(childPage, key, trace);
300 }

304 int scanNode(int page, int low, int high, List<Entry> result) {
305 var node = pages.get(page);
306 int visits = 1, count = node.entries.size();
307 for (int j = 0; j <= count; j++) {
308 if (!node.children.isEmpty() && (j == 0 || node.entries.get(j - 1).key < high) && (j == count || node.entries.get(j).key > low))
309 visits += scanNode(node.children.get(j), low, high, result);
310 if (j < count && low <= node.entries.get(j).key && node.entries.get(j).key <= high) result.add(node.entries.get(j));
311 }
312 return visits;
313 }

345 void compact() {
346 List<Entry> source = entries();
347 List<Integer> children = List.of();
348 var built = new ArrayList<Node>();
349 int oldRoot;
350 while (true) {
351 int groups = (source.size() + 2 * degree) / (2 * degree);
352 int keys = source.size() - (groups - 1), base = keys / groups, extra = keys % groups;
353 int position = 0, childPosition = 0;
354 var separators = new ArrayList<Entry>();
355 var level = new ArrayList<Integer>();
356 for (int j = 0; j < groups; j++) {
357 int count = base + (j < extra ? 1 : 0);
358 var node = new Node(source.subList(position, position + count),
359 children.isEmpty() ? List.of() : children.subList(childPosition, childPosition + count + 1));
360 position += count;
361 childPosition += count + 1;
362 level.add(built.size());
363 built.add(node);
364 if (j + 1 < groups) separators.add(source.get(position++));
365 }
366 if (groups == 1) { oldRoot = level.getFirst(); break; }
367 source = separators;
368 children = level;
369 }
370 pages.clear();
371 root = copyGroup(oldRoot, built);
372 splits = merges = borrows = 0;
373 }
374 }

376 static final class Parser {
377 final List<String> tokens;
378 final Tree tree;
379 int position;
380 Parser(int degree, List<String> tokens) { this.tokens = tokens; tree = new Tree(degree); tree.pages.clear(); }
381 String peek() { return position == tokens.size() ? "" : tokens.get(position); }
382 Entry entry() {
383 String token = peek();
384 int split = token.indexOf('=');
385 if (split < 0) throw new CommandError("SYNTAX " + (position + 1));
386 position++;
387 return new Entry(Integer.parseInt(token.substring(0, split)), token.substring(split + 1));
388 }
389 int node() {
390 if (!peek().equals("(")) throw new CommandError("SYNTAX " + (position + 1));
391 position++;
392 var node = new Node();
393 int page = tree.allocate(node);
394 boolean internal = peek().equals("(");
395 if (internal) node.children.add(node());
396 while (!peek().equals(")")) {
397 node.entries.add(entry());
398 if (internal) node.children.add(node());
399 }
400 position++;
401 return page;
402 }
403 }

417 static final class Workbench {
418 Tree index, saved;
419 Cursor cursor = new Cursor(CursorKind.UNSET);
420 boolean trace;
421
422 List<String> process(String line) {
423 String[] words = line.split(" ");
424 String command = words[0];
425 if (command.equals("TRACE")) {
426 trace = words[1].equals("ON");
427 return List.of("TRACE " + words[1]);
428 }
429 int first = words.length > 1 ? Integer.parseInt(words[1]) : 0;
430 boolean ranged = command.equals("DELRANGE") || command.equals("RANGE") || command.equals("COUNT");
431 int second = ranged ? Integer.parseInt(words[2]) : 0;
432 if (command.equals("CREATE") || command.equals("LOAD")) {
433 if (first < 2 || first > 5) throw new CommandError("BAD DEGREE " + first);
434 Tree replacement = command.equals("CREATE") ? new Tree(first) : Tree.load(first, Arrays.asList(words).subList(2, words.length));
435 String result = command.equals("CREATE") ? "CREATE t=" + first
436 : format("LOAD t=%d nodes=%d keys=%d %s", first, replacement.pages.size(), replacement.entries().size(), replacement.status());
437 index = replacement;
438 cursor = new Cursor(CursorKind.UNSET);
439 return List.of(result);
440 }
441 Tree tree = index;
442 if (tree == null) throw new CommandError("NO INDEX");
443 boolean needsValid = switch (command) {
444 case "INSERT", "BULKLOAD", "DELETE", "DELRANGE", "RANGE", "COUNT", "MIN", "MAX", "FLOOR", "CEIL",
445 "RANK", "SELECT", "SEEK", "NEXT", "PREV", "HEIGHT", "COMPACT" -> true;
446 default -> false;
447 };
448 if (needsValid && tree.violation() != null) throw new CommandError("DAMAGED");
449 var entries = tree.entries();
450 if (ranged && first > second) throw new CommandError(format("BAD RANGE %d %d", first, second));
451 if (command.equals("SELECT") && (first < 1 || first > entries.size())) throw new CommandError("BAD POSITION " + first);
452 if ((command.equals("NEXT") || command.equals("PREV")) && cursor.kind == CursorKind.UNSET) throw new CommandError("NO CURSOR");
453 if (command.equals("RESTORE") && saved == null) throw new CommandError("NO SAVE");
454 var output = new ArrayList<String>();
455 var steps = new Trace(trace && (command.equals("INSERT") || command.equals("DELETE")));
456 switch (command) {
457 case "INSERT" -> {
458 var before = tree.copyPages();
459 int splits = tree.splits;
460 String old = tree.upsert(new Entry(first, words[2]), steps);
461 int writes = tree.changes(before).writes;
462 output.addAll(steps.lines);
463 output.add(old != null ? format("INSERT %d UPDATED old=%s writes=%d", first, old, writes)
464 : format("INSERT %d NEW splits=%d height=%d writes=%d", first, tree.splits - splits, tree.height(), writes));
465 }
466 case "BULKLOAD" -> {
467 var before = tree.copyPages();
468 int splits = tree.splits, added = 0, updated = 0;
469 for (int i = 1; i < words.length; i += 2) {
470 if (tree.upsert(new Entry(Integer.parseInt(words[i]), words[i + 1]), steps) == null) added++;
471 else updated++;
472 }
473 output.add(format("BULKLOAD pairs=%d new=%d updated=%d splits=%d height=%d writes=%d",
474 (words.length - 1) / 2, added, updated, tree.splits - splits, tree.height(), tree.changes(before).writes));
475 }
476 case "DELETE" -> {
477 var found = tree.search(first);
478 if (found.page == null) output.add("DELETE " + first + " NOT FOUND");
479 else {
480 String value = tree.pages.get(found.page).entries.get(found.position).value;
481 var before = tree.copyPages();
482 tree.remove(first, steps);
483 var changes = tree.changes(before);
484 output.addAll(steps.lines);
485 output.add(format("DELETE %d REMOVED value=%s cases=%s height=%d writes=%d frees=%d",
486 first, value, String.join(",", steps.cases), tree.height(), changes.writes, changes.frees));
487 }
488 }
489 case "DELRANGE" -> {
490 var before = tree.copyPages();
491 int merges = tree.merges, borrows = tree.borrows, removed = 0;
492 for (var entry : entries) if (first <= entry.key && entry.key <= second) {
493 tree.remove(entry.key, steps);
494 removed++;
495 }
496 var changes = tree.changes(before);
497 output.add(format("DELRANGE %d %d removed=%d merges=%d borrows=%d height=%d writes=%d frees=%d",
498 first, second, removed, tree.merges - merges, tree.borrows - borrows, tree.height(), changes.writes, changes.frees));
499 }
500 case "GET" -> {
501 var found = tree.search(first);
502 String result = found.page == null ? "NOT FOUND" : "FOUND value=" + tree.pages.get(found.page).entries.get(found.position).value;
503 output.add(format("GET %d %s path=%s", first, result, String.join(" -> ", found.path)));
504 }
505 case "RANGE", "COUNT" -> {
506 var result = tree.scan(first, second);
507 String suffix = command.equals("RANGE") && !result.entries.isEmpty() ? " " + joined(result.entries, " ") : "";
508 output.add(format("%s %d %d count=%d visited=%d%s", command, first, second, result.entries.size(), result.visited, suffix));
509 }
510 case "MIN", "MAX" -> {
511 var entry = findEntry(entries, command.equals("MAX"), e -> true);
512 output.add(command + " " + display(entry, "EMPTY"));
513 }
514 case "FLOOR", "CEIL" -> {
515 boolean floor = command.equals("FLOOR");
516 var entry = findEntry(entries, floor, e -> floor ? e.key <= first : e.key >= first);
517 output.add(format("%s %d %s", command, first, display(entry, "NONE")));
518 }
519 case "RANK" -> output.add(format("RANK %d %d", first, entries.stream().filter(e -> e.key < first).count()));
520 case "SELECT" -> output.add(format("SELECT %d %s", first, entries.get(first - 1)));
521 case "SEEK" -> {
522 var entry = findEntry(entries, false, e -> e.key >= first);
523 cursor = entry == null ? new Cursor(CursorKind.END) : new Cursor(CursorKind.KEY, entry.key);
524 output.add(format("SEEK %d %s", first, display(entry, "END")));
525 }
526 case "NEXT", "PREV" -> {
527 boolean next = command.equals("NEXT");
528 Entry entry = switch (cursor.kind) {
529 case KEY -> findEntry(entries, !next, e -> next ? e.key > cursor.key : e.key < cursor.key);
530 case BEGIN -> next ? findEntry(entries, false, e -> true) : null;
531 case END -> next ? null : findEntry(entries, true, e -> true);
532 case UNSET -> throw new IllegalStateException();
533 };
534 String boundary = next ? "END" : "BEGIN";
535 cursor = entry == null ? new Cursor(next ? CursorKind.END : CursorKind.BEGIN) : new Cursor(CursorKind.KEY, entry.key);
536 output.add(command + " " + display(entry, boundary));
537 }
538 case "PRINT" -> {
539 List<Integer> level = List.of(tree.root);
540 int depth = 0;
541 while (!level.isEmpty()) {
542 var result = new StringBuilder("LEVEL " + depth++);
543 var next = new ArrayList<Integer>();
544 for (int page : level) {
545 var node = tree.pages.get(page);
546 result.append(' ').append(node.notation());
547 next.addAll(node.children);
548 }
549 output.add(result.toString());
550 level = next;
551 }
552 }
553 case "PAGES" -> {
554 for (var item : tree.pages.entrySet()) {
555 var node = item.getValue();
556 output.add(format("PAGE %d %s kids=%s", item.getKey(), node.notation(), node.children.isEmpty() ? "-" : joined(node.children, ",")));
557 }
558 }
559 case "DUMP" -> output.add(format("DUMP %d %s", tree.degree, tree.dump()));
560 case "HEIGHT" -> output.add("HEIGHT " + tree.height());
561 case "STATS" -> {
562 int nodes = tree.pages.size(), fill = 100 * entries.size() / ((2 * tree.degree - 1) * nodes);
563 long leaves = tree.pages.values().stream().filter(node -> node.children.isEmpty()).count();
564 output.add(format("STATS t=%d nodes=%d leaves=%d keys=%d fill=%d%% splits=%d merges=%d borrows=%d",
565 tree.degree, nodes, leaves, entries.size(), fill, tree.splits, tree.merges, tree.borrows));
566 }
567 case "VERIFY" -> {
568 String problem = tree.violation();
569 output.add(problem == null ? "VERIFY OK" : "VERIFY FAIL " + problem);
570 }
571 case "REBUILD" -> {
572 tree = new Tree(tree.degree);
573 for (var entry : entries) tree.upsert(entry, steps);
574 output.add(format("REBUILD entries=%d keys=%d nodes=%d splits=%d height=%d",
575 entries.size(), tree.entries().size(), tree.pages.size(), tree.splits, tree.height()));
576 }
577 case "COMPACT" -> {
578 int nodes = tree.pages.size();
579 tree.compact();
580 output.add(format("COMPACT nodes=%d->%d height=%d", nodes, tree.pages.size(), tree.height()));
581 }
582 case "SAVE" -> {
583 saved = tree.copy();
584 output.add(format("SAVE nodes=%d keys=%d %s", tree.pages.size(), entries.size(), tree.status()));
585 }
586 case "RESTORE" -> {
587 tree = saved.copy();
588 output.add(format("RESTORE nodes=%d keys=%d %s", tree.pages.size(), tree.entries().size(), tree.status()));
589 }
590 default -> throw new IllegalStateException();
591 }
592 index = tree;
593 return output;
594 }
595 }

597 public static void main(String[] args) throws Exception {
598 var input = new BufferedReader(new InputStreamReader(System.in));
599 var output = new BufferedWriter(new OutputStreamWriter(System.out));
600 int count = Integer.parseInt(input.readLine());
601 var workbench = new Workbench();
602 for (int i = 0; i < count; i++) {

604 for (String line : workbench.process(input.readLine())) { output.write(line); output.write('\n'); }
605 } catch (CommandError error) {
606 output.write("ERROR " + error.getMessage());
607 output.write('\n');
608 }
609 }
