12final class Node
13{
14 public function __construct(public array $entries = [], public array $children = []) {}
15 public function leaf(): bool { return $this->children === []; }
16 public function position(int $key): int
17 {
18 foreach ($this->entries as $i => $entry) {
19 if ($key <= $entry->key) return $i;
20 }
21 return count($this->entries);
22 }
23 public function copy(): self { return new self($this->entries, $this->children); }
24 public function sameContent(self $other): bool
25 {
26 if ($this->children !== $other->children || count($this->entries) !== count($other->entries)) return false;
27 foreach ($this->entries as $i => $entry) {
28 $old = $other->entries[$i];
29 if ($entry->key !== $old->key || $entry->value !== $old->value) return false;
30 }
31 return true;
32 }
33 public function __toString(): string
34 {
35 return '[' . implode(' ', array_map(fn(Entry $entry): int => $entry->key, $this->entries)) . ']';
36 }

77 public static function load(int $degree, array $tokens): self
78 {
79 $tree = new self($degree, false);
80 $position = 0;
81 $bad = static function() use (&$position): never {
82 throw new CommandError('SYNTAX ' . ($position + 1));
83 };
84 $readEntry = static function() use ($tokens, &$position, $bad): Entry {
85 $token = $tokens[$position] ?? null;
86 if ($token === null || $token === '(' || $token === ')') $bad();
87 [$key, $value] = explode('=', $token);
88 $position++;
89 return new Entry((int)$key, $value);
90 };
91 $readNode = function() use (&$readNode, $readEntry, $bad, $tokens, &$position, $tree): int {
92 if (($tokens[$position] ?? null) !== '(') $bad();
93 $position++;
94 $node = new Node();
95 $page = $tree->newPage($node);
96 if (($tokens[$position] ?? null) === '(') {
97 $node->children[] = $readNode();
98 while (($tokens[$position] ?? null) !== ')') {
99 $node->entries[] = $readEntry();
100 $node->children[] = $readNode();
101 }
102 } else {
103 while (($tokens[$position] ?? null) !== ')') $node->entries[] = $readEntry();
104 }
105 $position++;
106 return $page;
107 };
108 $tree->root = $readNode();
109 if ($position !== count($tokens)) $bad();
110 return $tree;
111 }
112 public function entries(): array

114 $result = [];
115 $visit = function(int $page) use (&$visit, &$result): void {

136 public function search(int $key): Search
137 {
138 $page = $this->root;
139 $path = [];
140 while (true) {
141 $node = $this->pages[$page];
142 $path[] = (string)$node;
143 $position = $node->position($key);
144 if ($position < count($node->entries) && $node->entries[$position]->key === $key) {

151 public function violation(): ?string
152 {
153 $reference = null;
154 $visit = function(int $page, string $location, int $depth, ?int $low, ?int $high) use (&$visit, &$reference): ?string {
155 $node = $this->pages[$page];
156 $count = count($node->entries);
157 $minimum = $page === $this->root ? ($node->leaf() ? 0 : 1) : $this->degree - 1;
158 $maximum = 2 * $this->degree - 1;
159 if ($count < $minimum || $count > $maximum) {
160 return "COUNT at=$location node=$node keys=$count allowed=$minimum..$maximum";
161 }
162 for ($i = 0; $i + 1 < $count; $i++) {
163 $left = $node->entries[$i]->key;
164 $right = $node->entries[$i + 1]->key;
165 if ($left >= $right) return "ORDER at=$location node=$node $left>=$right";
166 }
167 foreach ($node->entries as $entry) {
168 if (($low !== null && $entry->key <= $low) || ($high !== null && $entry->key >= $high)) {
169 $lo = $low ?? '-inf';
170 $hi = $high ?? '+inf';
171 return "RANGE at=$location node=$node key=$entry->key low=$lo high=$hi";
172 }
173 }
174 if ($node->leaf()) {
175 if ($reference === null) $reference = $depth;
176 elseif ($depth !== $reference) return "DEPTH at=$location node=$node depth=$depth expected=$reference";
177 }
178 foreach ($node->children as $i => $child) {
179 $problem = $visit($child, "$location.$i", $depth + 1,
180 $i === 0 ? $low : $node->entries[$i - 1]->key,
181 $i === $count ? $high : $node->entries[$i]->key);
182 if ($problem !== null) return $problem;
183 }
184 return null;
185 };
186 return $visit($this->root, 'root', 0, null, null);
187 }
188 public function changes(self $before): array
189 {
190 $writes = $frees = 0;
191 foreach ($this->pages as $page => $node) {
192 if (!isset($before->pages[$page]) || !$node->sameContent($before->pages[$page])) $writes++;
193 }
194 foreach ($before->pages as $page => $_) if (!isset($this->pages[$page])) $frees++;
195 return ['writes' => $writes, 'frees' => $frees];
196 }
197 private function splitChild(int $parentPage, int $position, Trace $trace): void
198 {
199 $parent = $this->pages[$parentPage];
200 $childPage = $parent->children[$position];
201 $child = $this->pages[$childPage];
202 $middle = $child->entries[$this->degree - 1];
203 $right = new Node(array_slice($child->entries, $this->degree),
204 $child->leaf() ? [] : array_slice($child->children, $this->degree));
205 $rightPage = $this->newPage($right);
206 $trace->emit("STEP split page=$childPage keys=$child up=$middle->key new=$rightPage");
207 $child->entries = array_slice($child->entries, 0, $this->degree - 1);
208 if (!$child->leaf()) $child->children = array_slice($child->children, 0, $this->degree);
209 array_splice($parent->entries, $position, 0, [$middle]);
210 array_splice($parent->children, $position + 1, 0, [$rightPage]);
211 $this->splits++;
212 }
213 public function upsert(int $key, string $value, Trace $trace): ?string
214 {
215 $found = $this->search($key);
216 if ($found->page !== null) {
217 $node = $this->pages[$found->page];
218 $old = $node->entries[$found->position]->value;
219 $node->entries[$found->position] = new Entry($key, $value);
220 return $old;
221 }
222 if (count($this->pages[$this->root]->entries) === 2 * $this->degree - 1) {
223 $previous = $this->root;
224 $this->root = $this->newPage(new Node([], [$previous]));
225 $trace->emit("STEP grow page=$this->root");
226 $this->splitChild($this->root, 0, $trace);
227 }
228 $page = $this->root;
229 while (!$this->pages[$page]->leaf()) {
230 $node = $this->pages[$page];
231 $position = $node->position($key);
232 if (count($this->pages[$node->children[$position]]->entries) === 2 * $this->degree - 1) {
233 $this->splitChild($page, $position, $trace);
234 if ($key > $node->entries[$position]->key) $position++;
235 }
236 $page = $node->children[$position];
237 }
238 $node = $this->pages[$page];
239 array_splice($node->entries, $node->position($key), 0, [new Entry($key, $value)]);
240 $trace->emit("STEP place page=$page keys=$node");
241 return null;
242 }
243 private function merge(int $parentPage, int $position, string $case, Trace $trace): int
244 {
245 $parent = $this->pages[$parentPage];
246 $leftPage = $parent->children[$position];
247 $rightPage = $parent->children[$position + 1];
248 $left = $this->pages[$leftPage];
249 $right = $this->pages[$rightPage];
250 $left->entries = array_merge($left->entries, [$parent->entries[$position]], $right->entries);
251 $left->children = array_merge($left->children, $right->children);
252 array_splice($parent->entries, $position, 1);
253 array_splice($parent->children, $position + 1, 1);
254 unset($this->pages[$rightPage]);
255 $this->merges++;
256 $trace->emit("STEP $case page=$parentPage merge=$leftPage+$rightPage keys=$left");
257 if ($parentPage === $this->root && $parent->entries === []) {
258 $this->root = $leftPage;
259 unset($this->pages[$parentPage]);
260 $trace->emit("STEP shrink page=$leftPage");
261 }
262 return $leftPage;
263 }
264 private function removeAt(int $page, int $key, Trace $trace): void
265 {
266 $node = $this->pages[$page];
267 $position = $node->position($key);
268 if ($position < count($node->entries) && $node->entries[$position]->key === $key) {
269 if ($node->leaf()) {
270 $trace->record('1');
271 array_splice($node->entries, $position, 1);
272 $trace->emit("STEP 1 page=$page keys=$node");
273 return;
274 }
275 $left = $node->children[$position];
276 $right = $node->children[$position + 1];
277 if (count($this->pages[$left]->entries) >= $this->degree) {
278 $last = $left;
279 while (!$this->pages[$last]->leaf()) $last = $this->pages[$last]->children[array_key_last($this->pages[$last]->children)];
280 $predecessor = $this->pages[$last]->entries[array_key_last($this->pages[$last]->entries)];
281 $trace->record('2a');
282 $trace->emit("STEP 2a page=$page pred=$predecessor->key");
283 $this->removeAt($left, $predecessor->key, $trace);
284 $node->entries[$position] = $predecessor;
285 } elseif (count($this->pages[$right]->entries) >= $this->degree) {
286 $first = $right;
287 while (!$this->pages[$first]->leaf()) $first = $this->pages[$first]->children[0];
288 $successor = $this->pages[$first]->entries[0];
289 $trace->record('2b');
290 $trace->emit("STEP 2b page=$page succ=$successor->key");
291 $this->removeAt($right, $successor->key, $trace);
292 $node->entries[$position] = $successor;
293 } else {
294 $trace->record('2c');
295 $child = $this->merge($page, $position, '2c', $trace);
296 $this->removeAt($child, $key, $trace);
297 }
298 return;
299 }
300 $childPage = $node->children[$position];
301 $child = $this->pages[$childPage];
302 if (count($child->entries) === $this->degree - 1) {
303 $left = $position > 0 ? $node->children[$position - 1] : null;
304 $right = $position < count($node->entries) ? $node->children[$position + 1] : null;
305 if ($left !== null && count($this->pages[$left]->entries) >= $this->degree) {
306 $sibling = $this->pages[$left];
307 $trace->record('3a-left');
308 array_unshift($child->entries, $node->entries[$position - 1]);
309 $node->entries[$position - 1] = array_pop($sibling->entries);
310 if (!$sibling->leaf()) array_unshift($child->children, array_pop($sibling->children));
311 $this->borrows++;
312 $trace->emit("STEP 3a-left page=$page child=$childPage from=$left keys=$child");
313 } elseif ($right !== null && count($this->pages[$right]->entries) >= $this->degree) {
314 $sibling = $this->pages[$right];
315 $trace->record('3a-right');
316 $child->entries[] = $node->entries[$position];
317 $node->entries[$position] = array_shift($sibling->entries);
318 if (!$sibling->leaf()) $child->children[] = array_shift($sibling->children);
319 $this->borrows++;
320 $trace->emit("STEP 3a-right page=$page child=$childPage from=$right keys=$child");
321 } elseif ($left !== null) {
322 $trace->record('3b-left');
323 $childPage = $this->merge($page, $position - 1, '3b-left', $trace);
324 } else {
325 $trace->record('3b-right');
326 $childPage = $this->merge($page, $position, '3b-right', $trace);
327 }
328 }
329 $this->removeAt($childPage, $key, $trace);
330 }

337 public function scan(int $low, int $high): array
338 {
339 $result = [];
340 $visited = 0;
341 $visit = function(int $page) use (&$visit, &$result, &$visited, $low, $high): void {
342 $visited++;
343 $node = $this->pages[$page];
344 $count = count($node->entries);
345 for ($i = 0; $i <= $count; $i++) {
346 if (!$node->leaf() && ($i === 0 || $node->entries[$i - 1]->key < $high)
347 && ($i === $count || $node->entries[$i]->key > $low)) $visit($node->children[$i]);
348 if ($i < $count && $low <= $node->entries[$i]->key && $node->entries[$i]->key <= $high) {
349 $result[] = $node->entries[$i];
350 }
351 }
352 };
353 $visit($this->root);
354 return ['entries' => $result, 'visited' => $visited];
355 }

372 public function compact(): void
373 {
374 $source = $this->entries();
375 $children = $built = [];
376 while (true) {
377 $groups = intdiv(count($source) + 2 * $this->degree, 2 * $this->degree);
378 $keys = count($source) - ($groups - 1);
379 $size = intdiv($keys, $groups);
380 $extra = $keys % $groups;
381 $entryIndex = $childIndex = 0;
382 $nodes = $separators = [];
383 for ($group = 0; $group < $groups; $group++) {
384 $count = $size + ($group < $extra ? 1 : 0);
385 $nodes[] = count($built);
386 $built[] = new Node(array_slice($source, $entryIndex, $count),
387 $children === [] ? [] : array_slice($children, $childIndex, $count + 1));
388 $entryIndex += $count;
389 $childIndex += $count + 1;
390 if ($group + 1 < $groups) $separators[] = $source[$entryIndex++];
391 }
392 if ($groups === 1) {
393 $oldRoot = $nodes[0];
394 break;
395 }
396 $source = $separators;
397 $children = $nodes;
398 }
399 $replacement = [];
400 $nextPage = 0;
401 $renumber = function(int $oldPage) use (&$renumber, $built, &$replacement, &$nextPage): int {
402 $page = $nextPage++;
403 $old = $built[$oldPage];
404 $children = [];
405 foreach ($old->children as $child) $children[] = $renumber($child);
406 $replacement[$page] = new Node($old->entries, $children);
407 return $page;
408 };
409 $this->root = $renumber($oldRoot);
410 $this->pages = $replacement;
411 $this->splits = $this->merges = $this->borrows = 0;
412 }

426 public function __construct() { $this->cursor = new Cursor(CursorKind::Fresh); }
427 private const NEEDS_VALID = ['INSERT', 'BULKLOAD', 'DELETE', 'DELRANGE', 'RANGE', 'COUNT',
428 'MIN', 'MAX', 'FLOOR', 'CEIL', 'RANK', 'SELECT', 'SEEK', 'NEXT', 'PREV', 'HEIGHT', 'COMPACT'];

430 private static function find(array $entries, Closure $predicate, bool $last = false): ?Entry
431 {
432 $result = null;
433 foreach ($entries as $entry) {
434 if (!$predicate($entry)) continue;
435 $result = $entry;
436 if (!$last) break;
437 }
438 return $result;

440 private static function summary(string $prefix, Tree $tree): string
441 {
442 $nodes = count($tree->pages);
443 $keys = count($tree->entries());
444 $status = $tree->violation() === null ? 'OK' : 'DAMAGED';
445 return "$prefix nodes=$nodes keys=$keys $status";
446 }
447 public function process(string $line): array
448 {
449 $words = explode(' ', $line);
450 $command = $words[0];
451 if ($command === 'TRACE') {
452 $this->trace = $words[1] === 'ON';
453 return ['TRACE ' . $words[1]];
454 }
455 $first = isset($words[1]) ? (int)$words[1] : 0;
456 $ranged = in_array($command, ['DELRANGE', 'RANGE', 'COUNT'], true);
457 $second = $ranged ? (int)$words[2] : 0;
458 if ($command === 'CREATE' || $command === 'LOAD') {
459 if ($first < 2 || $first > 5) throw new CommandError("BAD DEGREE $first");
460 $replacement = $command === 'CREATE' ? new Tree($first) : Tree::load($first, array_slice($words, 2));
461 $this->index = $replacement;
462 $this->cursor = new Cursor(CursorKind::Fresh);
463 return [$command === 'CREATE' ? "CREATE t=$first" : self::summary("LOAD t=$first", $replacement)];
464 }
465 $tree = $this->index;
466 if ($tree === null) throw new CommandError('NO INDEX');
467 if (in_array($command, self::NEEDS_VALID, true) && $tree->violation() !== null) throw new CommandError('DAMAGED');
468 $entries = $tree->entries();
469 if ($ranged && $first > $second) throw new CommandError("BAD RANGE $first $second");
470 if ($command === 'SELECT' && ($first < 1 || $first > count($entries))) throw new CommandError("BAD POSITION $first");
471 if (($command === 'NEXT' || $command === 'PREV') && $this->cursor->kind === CursorKind::Fresh) throw new CommandError('NO CURSOR');
472 if ($command === 'RESTORE' && $this->saved === null) throw new CommandError('NO SAVE');
473 $output = [];
474 $steps = new Trace($this->trace && ($command === 'INSERT' || $command === 'DELETE'));
475 switch ($command) {
476 case 'INSERT':
477 $before = $tree->copy();
478 $splits = $tree->splits;
479 $old = $tree->upsert($first, $words[2], $steps);
480 $writes = $tree->changes($before)['writes'];
481 $output = $steps->lines;
482 $output[] = $old === null
483 ? "INSERT $first NEW splits=" . ($tree->splits - $splits) . " height={$tree->height()} writes=$writes"
484 : "INSERT $first UPDATED old=$old writes=$writes";
485 break;
486 case 'BULKLOAD':
487 $before = $tree->copy();
488 $splits = $tree->splits;
489 $inserted = $updated = 0;
490 for ($i = 1; $i < count($words); $i += 2) {
491 if ($tree->upsert((int)$words[$i], $words[$i + 1], $steps) === null) $inserted++;
492 else $updated++;
493 }
494 $pairs = intdiv(count($words) - 1, 2);
495 $writes = $tree->changes($before)['writes'];
496 $output[] = "BULKLOAD pairs=$pairs new=$inserted updated=$updated splits=" . ($tree->splits - $splits)
497 . " height={$tree->height()} writes=$writes";
498 break;
499 case 'DELETE':
500 $found = $tree->search($first);
501 if ($found->page === null) {
502 $output[] = "DELETE $first NOT FOUND";
503 } else {
504 $value = $tree->pages[$found->page]->entries[$found->position]->value;
505 $before = $tree->copy();
506 $cases = implode(',', $tree->remove($first, $steps));
507 $changes = $tree->changes($before);
508 $output = $steps->lines;
509 $output[] = "DELETE $first REMOVED value=$value cases=$cases height={$tree->height()} writes={$changes['writes']} frees={$changes['frees']}";
510 }
511 break;
512 case 'DELRANGE':
513 $before = $tree->copy();
514 $merges = $tree->merges;
515 $borrows = $tree->borrows;
516 $removed = 0;
517 foreach ($entries as $entry) {
518 if ($first <= $entry->key && $entry->key <= $second) {
519 $tree->remove($entry->key, $steps);
520 $removed++;
521 }
522 }
523 $changes = $tree->changes($before);
524 $output[] = "DELRANGE $first $second removed=$removed merges=" . ($tree->merges - $merges)
525 . ' borrows=' . ($tree->borrows - $borrows)
526 . " height={$tree->height()} writes={$changes['writes']} frees={$changes['frees']}";
527 break;
528 case 'GET':
529 $found = $tree->search($first);
530 $result = $found->page === null ? 'NOT FOUND' : 'FOUND value=' . $tree->pages[$found->page]->entries[$found->position]->value;
531 $output[] = "GET $first $result path=" . implode(' -> ', $found->path);
532 break;
533 case 'RANGE':
534 case 'COUNT':
535 $result = $tree->scan($first, $second);
536 $output[] = "$command $first $second count=" . count($result['entries']) . " visited={$result['visited']}"
537 . ($command === 'RANGE' ? implode('', array_map(fn(Entry $entry): string => " $entry", $result['entries'])) : '');
538 break;
539 case 'MIN':
540 case 'MAX':
541 $output[] = "$command " . ($entries === [] ? 'EMPTY' : (string)$entries[$command === 'MIN' ? 0 : count($entries) - 1]);
542 break;
543 case 'FLOOR':
544 case 'CEIL':
545 $entry = $command === 'FLOOR'
546 ? self::find($entries, fn(Entry $value): bool => $value->key <= $first, true)
547 : self::find($entries, fn(Entry $value): bool => $value->key >= $first);
548 $output[] = "$command $first " . ($entry === null ? 'NONE' : (string)$entry);
549 break;
550 case 'RANK':
551 $rank = count(array_filter($entries, fn(Entry $entry): bool => $entry->key < $first));
552 $output[] = "RANK $first $rank";
553 break;
554 case 'SELECT':
555 $output[] = "SELECT $first " . $entries[$first - 1];
556 break;
557 case 'SEEK':
558 $entry = self::find($entries, fn(Entry $value): bool => $value->key >= $first);
559 $this->cursor = $entry === null ? new Cursor(CursorKind::End) : new Cursor(CursorKind::Key, $entry->key);
560 $output[] = "SEEK $first " . ($entry === null ? 'END' : (string)$entry);
561 break;
562 case 'NEXT':
563 case 'PREV':
564 $entry = null;
565 if ($command === 'NEXT') {
566 if ($this->cursor->kind === CursorKind::Begin) $entry = self::find($entries, fn(Entry $_): bool => true);
567 elseif ($this->cursor->kind === CursorKind::Key) {
568 $entry = self::find($entries, fn(Entry $value): bool => $value->key > $this->cursor->key);
569 }
570 } else {
571 if ($this->cursor->kind === CursorKind::End) $entry = self::find($entries, fn(Entry $_): bool => true, true);
572 elseif ($this->cursor->kind === CursorKind::Key) {
573 $entry = self::find($entries, fn(Entry $value): bool => $value->key < $this->cursor->key, true);
574 }
575 }
576 $marker = $command === 'NEXT' ? 'END' : 'BEGIN';
577 $kind = $command === 'NEXT' ? CursorKind::End : CursorKind::Begin;
578 $this->cursor = $entry === null ? new Cursor($kind) : new Cursor(CursorKind::Key, $entry->key);
579 $output[] = "$command " . ($entry === null ? $marker : (string)$entry);
580 break;
581 case 'PRINT':
582 $level = [$tree->root];
583 $depth = 0;
