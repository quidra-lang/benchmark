18struct Entry: Equatable {
19 let key: Int
20 var value: String
21 var display: String {
22 "\(key)=\(value)"
23 }
24}
25
26struct Node: Equatable {
27 var entries: [Entry] = []
28 var children: [Int] = []
29 var leaf: Bool {
30 children.isEmpty
31 }
32 var notation: String {
33 "[" + entries.map { String($0.key) }.joined(separator: " ") + "]"
34 }
35 func position(_ key: Int) -> Int {
36 entries.firstIndex { key <= $0.key } ?? entries.count
37 }
38}

40struct SearchResult {
41 let page: Int?
42 let index: Int
43 let path: [String]
44}

46enum UpsertResult { case inserted, updated(String) }

48struct Tree {
49 let degree: Int
50 var root = 0
51 var pages: [Int: Node] = [0: Node()]
52 var splits = 0
53 var merges = 0
54 var borrows = 0
55
56 func entries() -> [Entry] {
57 var result: [Entry] = []
58 func visit(_ page: Int) {
59 let node = pages[page]!
60 if !node.leaf {
61 visit(node.children[0])
62 }
63 for index in node.entries.indices {
64 result.append(node.entries[index])
65 if !node.leaf {
66 visit(node.children[index + 1])
67 }
68 }
69 }
70 visit(root)
71 return result
72 }
73 var height: Int {
74 var height = 0, page = root
75 while !pages[page]!.leaf {
76 height += 1
77 page = pages[page]!.children[0]
78 }
79 return height
80 }
81 func changes(from before: [Int: Node]) -> (writes: Int, frees: Int) {
82 let writes = pages.filter { before[$0.key] != $0.value }.count
83 let frees = before.keys.filter { pages[$0] == nil }.count
84 return (writes, frees)
85 }
86 mutating func allocate(_ node: Node) -> Int {
87 var page = 0
88 while pages[page] != nil {
89 page += 1
90 }
91 pages[page] = node
92 return page
93 }
94 func search(_ key: Int) -> SearchResult {
95 var page = root, path: [String] = []
96 while true {
97 let node = pages[page]!, index = node.position(key)
98 path.append(node.notation)
99 if index < node.entries.count && node.entries[index].key == key {
100 return SearchResult(page: page, index: index, path: path)
101 }
102 if node.leaf {
103 return SearchResult(page: nil, index: -1, path: path)
104 }
105 page = node.children[index]
106 }
107 }
108 func violation() -> String? {
109 var referenceDepth: Int?
110 func visit(_ page: Int, _ location: String, _ depth: Int, _ low: Int?, _ high: Int?) -> String? {
111 let node = pages[page]!, count = node.entries.count
112 let minimum = page == root ? (node.leaf ? 0 : 1) : degree - 1, maximum = 2 * degree - 1
113 let at = "at=\(location) node=\(node.notation)"
114 if count < minimum || count > maximum {
115 return "COUNT \(at) keys=\(count) allowed=\(minimum)..\(maximum)"
116 }
117 for index in 0..<max(0, count - 1) {
118 if node.entries[index].key >= node.entries[index + 1].key {
119 return "ORDER \(at) \(node.entries[index].key)>=\(node.entries[index + 1].key)"
120 }
121 }
122 for entry in node.entries {
123 if (low.map { entry.key <= $0 } ?? false) || (high.map { entry.key >= $0 } ?? false) {
124 return "RANGE \(at) key=\(entry.key) low=\(low.map { String($0) } ?? "-inf") high=\(high.map { String($0) } ?? "+inf")"
125 }
126 }
127 if node.leaf {
128 if let expected = referenceDepth {
129 if depth != expected {
130 return "DEPTH \(at) depth=\(depth) expected=\(expected)"
131 }
132 } else {
133 referenceDepth = depth
134 }
135 }
136 for index in node.children.indices {
137 if let problem = visit(node.children[index], "\(location).\(index)", depth + 1,
138 index == 0 ? low : node.entries[index - 1].key,
139 index == count ? high : node.entries[index].key)
140 {
141 return problem
142 }
143 }
144 return nil
145 }
146 return visit(root, "root", 0, nil, nil)
147 }
148 static func load(degree: Int, tokens: [String]) throws -> Tree {
149 var position = 0, nextPage = 0, pages: [Int: Node] = [:]
150 func syntax() -> CommandError {
151 CommandError("SYNTAX \(position + 1)")
152 }
153 func entry() throws -> Entry {
154 guard position < tokens.count, tokens[position] != "(", tokens[position] != ")" else {
155 throw syntax()
156 }
157 let words = tokens[position].split(separator: "=")
158 position += 1
159 return Entry(key: Int(words[0])!, value: String(words[1]))
160 }
161 func parse() throws -> Int {
162 guard position < tokens.count, tokens[position] == "(" else {
163 throw syntax()
164 }
165 position += 1
166 let page = nextPage
167 nextPage += 1
168 var node = Node()
169 if position < tokens.count && tokens[position] == "(" {
170 node.children.append(try parse())
171 while position == tokens.count || tokens[position] != ")" {
172 node.entries.append(try entry())
173 node.children.append(try parse())
174 }
175 } else {
176 while position == tokens.count || tokens[position] != ")" {
177 node.entries.append(try entry())
178 }
179 }
180 position += 1
181 pages[page] = node
182 return page
183 }
184 let root = try parse()
185 guard position == tokens.count else {
186 throw syntax()
187 }
188 var tree = Tree(degree: degree)
189 tree.root = root
190 tree.pages = pages
191 return tree
192 }
193 mutating func split(_ parentPage: Int, _ index: Int, trace: inout Trace) {
194 var parent = pages[parentPage]!
195 let childPage = parent.children[index]
196 var child = pages[childPage]!
197 let before = child.notation, middle = child.entries[degree - 1]
198 let right = Node(
199 entries: Array(child.entries.dropFirst(degree)),
200 children: Array(child.children.dropFirst(degree))
201 )
202 let newPage = allocate(right)
203 child.entries = Array(child.entries.prefix(degree - 1))
204 child.children = Array(child.children.prefix(degree))
205 parent.entries.insert(middle, at: index)
206 parent.children.insert(newPage, at: index + 1)
207 pages[childPage] = child
208 pages[parentPage] = parent
209 splits += 1
210 trace.emit("STEP split page=\(childPage) keys=\(before) up=\(middle.key) new=\(newPage)")
211 }
212 mutating func upsert(_ key: Int, _ value: String, trace: inout Trace) -> UpsertResult {
213 let found = search(key)
214 if let page = found.page {
215 let old = pages[page]!.entries[found.index].value
216 pages[page]!.entries[found.index].value = value
217 return .updated(old)
218 }
219 if pages[root]!.entries.count == 2 * degree - 1 {
220 let newRoot = allocate(Node(children: [root]))
221 root = newRoot
222 trace.emit("STEP grow page=\(root)")
223 split(root, 0, trace: &trace)
224 }
225 var page = root
226 while !pages[page]!.leaf {
227 var index = pages[page]!.position(key)
228 if pages[pages[page]!.children[index]]!.entries.count == 2 * degree - 1 {
229 split(page, index, trace: &trace)
230 if key > pages[page]!.entries[index].key {
231 index += 1
232 }
233 }
234 page = pages[page]!.children[index]
235 }
236 var node = pages[page]!
237 let position = node.position(key)
238 node.entries.insert(Entry(key: key, value: value), at: position)
239 pages[page] = node
240 trace.emit("STEP place page=\(page) keys=\(node.notation)")
241 return .inserted
242 }
243 mutating func merge(_ parentPage: Int, _ index: Int, _ caseName: String, trace: inout Trace) -> Int {
244 var parent = pages[parentPage]!
245 let leftPage = parent.children[index], rightPage = parent.children[index + 1], right = pages[rightPage]!
246 var left = pages[leftPage]!
247 left.entries.append(parent.entries.remove(at: index))
248 left.entries.append(contentsOf: right.entries)
249 left.children.append(contentsOf: right.children)
250 parent.children.remove(at: index + 1)
251 pages[leftPage] = left
252 pages[parentPage] = parent
253 pages[rightPage] = nil
254 merges += 1
255 trace.emit("STEP \(caseName) page=\(parentPage) merge=\(leftPage)+\(rightPage) keys=\(left.notation)")
256 if parentPage == root && parent.entries.isEmpty {
257 root = leftPage
258 pages[parentPage] = nil
259 trace.emit("STEP shrink page=\(root)")
260 }
261 return leftPage
262 }
263 mutating func removeAt(_ page: Int, _ key: Int, cases: inout [String], trace: inout Trace) {
264 var node = pages[page]!
265 let index = node.position(key)
266 if index < node.entries.count && node.entries[index].key == key {
267 if node.leaf {
268 cases.append("1")
269 node.entries.remove(at: index)
270 pages[page] = node
271 trace.emit("STEP 1 page=\(page) keys=\(node.notation)")
272 return
273 }
274 let left = node.children[index], right = node.children[index + 1]
275 if pages[left]!.entries.count >= degree {
276 var last = left
277 while !pages[last]!.leaf {
278 last = pages[last]!.children.last!
279 }
280 let predecessor = pages[last]!.entries.last!
281 cases.append("2a")
282 trace.emit("STEP 2a page=\(page) pred=\(predecessor.key)")
283 removeAt(left, predecessor.key, cases: &cases, trace: &trace)
284 pages[page]!.entries[index] = predecessor
285 } else if pages[right]!.entries.count >= degree {
286 var first = right
287 while !pages[first]!.leaf {
288 first = pages[first]!.children[0]
289 }
290 let successor = pages[first]!.entries[0]
291 cases.append("2b")
292 trace.emit("STEP 2b page=\(page) succ=\(successor.key)")
293 removeAt(right, successor.key, cases: &cases, trace: &trace)
294 pages[page]!.entries[index] = successor
295 } else {
296 cases.append("2c")
297 let child = merge(page, index, "2c", trace: &trace)
298 removeAt(child, key, cases: &cases, trace: &trace)
299 }
300 return
301 }
302 var childPage = node.children[index], child = pages[node.children[index]]!
303 if child.entries.count == degree - 1 {
304 let leftPage: Int? = index > 0 ? node.children[index - 1] : nil
305 let rightPage: Int? = index < node.entries.count ? node.children[index + 1] : nil
306 if let leftPage, pages[leftPage]!.entries.count >= degree {
307 var left = pages[leftPage]!
308 cases.append("3a-left")
309 child.entries.insert(node.entries[index - 1], at: 0)
310 node.entries[index - 1] = left.entries.removeLast()
311 if !left.leaf {
312 child.children.insert(left.children.removeLast(), at: 0)
313 }
314 pages[page] = node
315 pages[childPage] = child
316 pages[leftPage] = left
317 borrows += 1
318 trace.emit("STEP 3a-left page=\(page) child=\(childPage) from=\(leftPage) keys=\(child.notation)")
319 } else if let rightPage, pages[rightPage]!.entries.count >= degree {
320 var right = pages[rightPage]!
321 cases.append("3a-right")
322 child.entries.append(node.entries[index])
323 node.entries[index] = right.entries.removeFirst()
324 if !right.leaf {
325 child.children.append(right.children.removeFirst())
326 }
327 pages[page] = node
328 pages[childPage] = child
329 pages[rightPage] = right
330 borrows += 1
331 trace.emit("STEP 3a-right page=\(page) child=\(childPage) from=\(rightPage) keys=\(child.notation)")
332 } else if leftPage != nil {
333 cases.append("3b-left")
334 childPage = merge(page, index - 1, "3b-left", trace: &trace)
335 } else {
336 cases.append("3b-right")
337 childPage = merge(page, index, "3b-right", trace: &trace)
338 }
339 }
340 removeAt(childPage, key, cases: &cases, trace: &trace)
341 }
342 mutating func remove(_ key: Int, trace: inout Trace) -> [String] {
343 var cases: [String] = []
344 removeAt(root, key, cases: &cases, trace: &trace)
345 return cases
346 }
347 func scan(_ low: Int, _ high: Int) -> (entries: [Entry], visited: Int) {
348 var entries: [Entry] = [], visited = 0
349 func visit(_ page: Int) {
350 visited += 1
351 let node = pages[page]!, count = node.entries.count
352 for index in 0...count {
353 if !node.leaf && (index == 0 || node.entries[index - 1].key < high)
354 && (index == count || node.entries[index].key > low)
355 {
356 visit(node.children[index])
357 }
358 if index < count && low <= node.entries[index].key && node.entries[index].key <= high {
359 entries.append(node.entries[index])
360 }
361 }
362 }
363 visit(root)
364 return (entries, visited)
365 }
366 func dump() -> String {
367 var tokens: [String] = []
368 func visit(_ page: Int) {
369 let node = pages[page]!
370 tokens.append("(")
371 if !node.leaf {
372 visit(node.children[0])
373 }
374 for index in node.entries.indices {
375 tokens.append(node.entries[index].display)
376 if !node.leaf {
377 visit(node.children[index + 1])
378 }
379 }
380 tokens.append(")")
381 }
382 visit(root)
383 return tokens.joined(separator: " ")
384 }
385 mutating func compact() {
386 var source = entries(), children: [Int] = [], built: [Node] = []
387 var oldRoot = 0
388 while true {
389 let groups = (source.count + 2 * degree) / (2 * degree), keys = source.count - groups + 1
390 let size = keys / groups, extra = keys % groups
391 var nodes: [Int] = [], separators: [Entry] = [], entryIndex = 0, childIndex = 0
392 for group in 0..<groups {
393 let count = size + (group < extra ? 1 : 0)
394 let node = Node(entries: Array(source[entryIndex..<(entryIndex + count)]),
395 children: children
396 .isEmpty ? [] : Array(children[childIndex..<(childIndex + count + 1)]))
397 nodes.append(built.count)
398 built.append(node)
399 entryIndex += count
400 childIndex += count + 1
401 if group + 1 < groups {
402 separators.append(source[entryIndex])
403 entryIndex += 1
404 }
405 }
406 if groups == 1 {
407 oldRoot = nodes[0]
408 break
409 }
410 source = separators
411 children = nodes
412 }
413 var replacement: [Int: Node] = [:], nextPage = 0
414 func renumber(_ oldPage: Int) -> Int {
415 let page = nextPage
416 nextPage += 1
417 var node = built[oldPage]
418 node.children = node.children.map { renumber($0) }
419 replacement[page] = node
420 return page
421 }
422 root = renumber(oldRoot)
423 pages = replacement
424 splits = 0
425 merges = 0
426 borrows = 0
427 }
428}

430enum Verb: String {
431 case create = "CREATE", load = "LOAD", insert = "INSERT", bulkLoad = "BULKLOAD", delete = "DELETE"
432 case delRange = "DELRANGE", get = "GET", range = "RANGE", count = "COUNT", min = "MIN", max = "MAX"
433 case floor = "FLOOR", ceil = "CEIL", rank = "RANK", select = "SELECT", seek = "SEEK", next = "NEXT", prev = "PREV"
434 case print = "PRINT", pages = "PAGES", dump = "DUMP", height = "HEIGHT", stats = "STATS", verify = "VERIFY"
435 case rebuild = "REBUILD", compact = "COMPACT", save = "SAVE", restore = "RESTORE", trace = "TRACE"
436 var requiresValidTree: Bool {
437 switch self {
438 case .insert, .bulkLoad, .delete, .delRange, .range, .count, .min, .max, .floor, .ceil, .rank,
439 .select, .seek, .next, .prev, .height, .compact:
440 return true
441 default:
442 return false
443 }
444 }
445}

447enum Cursor: Equatable { case unset, key(Int), begin, end }
448
449final class Workbench {
450 private var index: Tree?
451 private var saved: Tree?
452 private var cursor = Cursor.unset
453 private var trace = false
454
455 private func summary(_ prefix: String, _ tree: Tree) -> String {
456 "\(prefix) nodes=\(tree.pages.count) keys=\(tree.entries().count) \(tree.violation() == nil ? "OK" : "DAMAGED")"
457 }
458 func command(_ line: String) throws -> [String] {
459 let words = line.split(separator: " ").map(String.init), verb = Verb(rawValue: words[0])!
460 let args = Array(words.dropFirst())
461 if verb == .trace {
462 trace = args[0] == "ON"
463 return ["TRACE \(args[0])"]
464 }
465 let first = args.isEmpty ? 0 : Int(args[0])!
466 let ranged = [Verb.delRange, .range, .count].contains(verb), second = ranged ? Int(args[1])! : 0
467 if verb == .create || verb == .load {
468 guard (2...5).contains(first) else {
469 throw CommandError("BAD DEGREE \(first)")
470 }
471 let replacement: Tree
472 if verb == .create {
473 replacement = Tree(degree: first)
474 } else {
475 replacement = try Tree.load(degree: first, tokens: Array(args.dropFirst()))
476 }
477 index = replacement
478 cursor = .unset
479 return [verb == .create ? "CREATE t=\(first)" : summary("LOAD t=\(first)", replacement)]
480 }
481 guard var tree = index else {
482 throw CommandError("NO INDEX")
483 }
484 if verb.requiresValidTree && tree.violation() != nil {
485 throw CommandError("DAMAGED")
486 }
487 let entries = tree.entries()
488 if ranged && first > second {
489 throw CommandError("BAD RANGE \(first) \(second)")
490 }
491 if verb == .select && (first < 1 || first > entries.count) {
492 throw CommandError("BAD POSITION \(first)")
493 }
494 if (verb == .next || verb == .prev) && cursor == .unset {
495 throw CommandError("NO CURSOR")
496 }
497 if verb == .restore && saved == nil {
498 throw CommandError("NO SAVE")
499 }
500 var output: [String] = [], steps = Trace(enabled: trace && (verb == .insert || verb == .delete))
501 switch verb {
502 case .insert:
503 let before = tree.pages, splits = tree.splits
504 let result = tree.upsert(first, args[1], trace: &steps), writes = tree.changes(from: before).writes
505 output += steps.lines
506 switch result {
507 case .inserted:
508 output
509 .append("INSERT \(first) NEW splits=\(tree.splits - splits) height=\(tree.height) writes=\(writes)")
510 case .updated(let old):
511 output.append("INSERT \(first) UPDATED old=\(old) writes=\(writes)")
512 }
513 case .bulkLoad:
514 let before = tree.pages, splits = tree.splits
515 var inserted = 0, updated = 0
516 for position in stride(from: 0, to: args.count, by: 2) {
517 if case .inserted = tree
518 .upsert(Int(args[position])!, args[position + 1], trace: &steps)
519 {
520 inserted += 1
521 } else {
522 updated += 1
523 }
524 }
525 output
526 .append(
527 "BULKLOAD pairs=\(args.count / 2) new=\(inserted) updated=\(updated) splits=\(tree.splits - splits) height=\(tree.height) writes=\(tree.changes(from: before).writes)"
528 )
529 case .delete:
530 let found = tree.search(first)
531 if let page = found.page {
532 let value = tree.pages[page]!.entries[found.index].value, before = tree.pages
533 let cases = tree.remove(first, trace: &steps), changes = tree.changes(from: before)
534 output += steps.lines
535 output
536 .append(
537 "DELETE \(first) REMOVED value=\(value) cases=\(cases.joined(separator: ",")) height=\(tree.height) writes=\(changes.writes) frees=\(changes.frees)"
538 )
539 } else {
540 output.append("DELETE \(first) NOT FOUND")
541 }
542 case .delRange:
543 let before = tree.pages, merges = tree.merges, borrows = tree.borrows
544 var removed = 0
545 for entry in entries where first <= entry.key && entry.key <= second {
546 _ = tree.remove(entry.key, trace: &steps)
547 removed += 1
548 }
549 let changes = tree.changes(from: before)
550 output
551 .append(
552 "DELRANGE \(first) \(second) removed=\(removed) merges=\(tree.merges - merges) borrows=\(tree.borrows - borrows) height=\(tree.height) writes=\(changes.writes) frees=\(changes.frees)"
553 )
554 case .get:
555 let found = tree.search(first)
556 let result = found.page.map { "FOUND value=\(tree.pages[$0]!.entries[found.index].value)" } ?? "NOT FOUND"
557 output.append("GET \(first) \(result) path=\(found.path.joined(separator: " -> "))")
558 case .range, .count:
559 let result = tree.scan(first, second)
560 let suffix = verb == .range ? result.entries.map { " " + $0.display }.joined() : ""
561 output
562 .append(
563 "\(verb.rawValue) \(first) \(second) count=\(result.entries.count) visited=\(result.visited)\(suffix)"
564 )
565 case .min, .max:
566 let entry = verb == .min ? entries.first : entries.last
567 output.append("\(verb.rawValue) \(entry?.display ?? "EMPTY")")
568 case .floor, .ceil:
569 let entry = verb == .floor ? entries.last(where: { $0.key <= first }) : entries
570 .first(where: { $0.key >= first })
571 output.append("\(verb.rawValue) \(first) \(entry?.display ?? "NONE")")
572 case .rank:
573 output.append("RANK \(first) \(entries.filter { $0.key < first }.count)")
574 case .select:
575 output.append("SELECT \(first) \(entries[first - 1].display)")
576 case .seek:
577 let entry = entries.first { $0.key >= first }
578 cursor = entry.map { .key($0.key) } ?? .end
579 output.append("SEEK \(first) \(entry?.display ?? "END")")
580 case .next, .prev:
581 var entry: Entry?
582 if verb == .next {
583 switch cursor {
584 case .begin:
585 entry = entries.first
586 case .key(let key):
587 entry = entries.first { $0.key > key }
588 case .end, .unset:
589 break
590 }
591 cursor = entry.map { .key($0.key) } ?? .end
592 } else {
593 switch cursor {
594 case .end:
595 entry = entries.last
596 case .key(let key):
597 entry = entries.last { $0.key < key }
598 case .begin, .unset:
599 break
600 }
601 cursor = entry.map { .key($0.key) } ?? .begin
602 }
603 output.append("\(verb.rawValue) \(entry?.display ?? (verb == .next ? "END" : "BEGIN"))")
604 case .print:
605 var level = [tree.root], depth = 0
606 while !level.isEmpty {
607 output.append("LEVEL \(depth) " + level.map { tree.pages[$0]!.notation }.joined(separator: " "))
608 level = level.flatMap { tree.pages[$0]!.children }
609 depth += 1
610 }
611 case .pages:
612 for page in tree.pages.keys.sorted() {
613 let node = tree.pages[page]!
614 let children = node.leaf ? "-" : node.children.map { String($0) }.joined(separator: ",")
615 output.append("PAGE \(page) \(node.notation) kids=\(children)")
616 }
617 case .dump:
618 output.append("DUMP \(tree.degree) \(tree.dump())")
619 case .height:
620 output.append("HEIGHT \(tree.height)")
621 case .stats:
622 let leaves = tree.pages.values.filter { $0.leaf }.count
623 let fill = 100 * entries.count / ((2 * tree.degree - 1) * tree.pages.count)
624 output
625 .append(
626 "STATS t=\(tree.degree) nodes=\(tree.pages.count) leaves=\(leaves) keys=\(entries.count) fill=\(fill)% splits=\(tree.splits) merges=\(tree.merges) borrows=\(tree.borrows)"
627 )
628 case .verify:
629 output.append(tree.violation().map { "VERIFY FAIL " + $0 } ?? "VERIFY OK")
630 case .rebuild:
631 tree = Tree(degree: tree.degree)
632 for entry in entries {
633 _ = tree.upsert(entry.key, entry.value, trace: &steps)
634 }
635 output
636 .append(
637 "REBUILD entries=\(entries.count) keys=\(tree.entries().count) nodes=\(tree.pages.count) splits=\(tree.splits) height=\(tree.height)"
638 )
639 case .compact:
640 let nodes = tree.pages.count
641 tree.compact()
642 output.append("COMPACT nodes=\(nodes)->\(tree.pages.count) height=\(tree.height)")
643 case .save:
644 saved = tree
645 output.append(summary("SAVE", tree))
646 case .restore:
647 tree = saved!
648 output.append(summary("RESTORE", tree))
649 case .create, .load, .trace:
650 break
651 }
652 index = tree
653 return output
654 }
655}
