61def count_changes(before, after):
62 writes = sum(1 for page, content in after.items() if before.get(page) != content)
63 frees = sum(1 for page in before if page not in after)
64 return writes, frees

93 def tree_notation(self):
94 tokens = ["("]
95 if self.is_leaf():
96 tokens.extend(entry_text(entry) for entry in self.entries)
97 else:
98 tokens.append(self.children[0].tree_notation())

100 tokens.append(entry_text(entry))

199 def search(self, key):
200 path = []
201 node = self.root
202 while True:
203 path.append(node)
204 j = position(node, key)
205 if j < len(node.entries) and node.key(j) == key:
206 return path, j
207 if node.is_leaf():
208 return path, None
209 node = node.children[j]

211 def split_child(self, node, j):
212 t = self.degree
213 child = node.children[j]
214 sibling = Node(self.free_page())
215 middle = child.entries[t - 1]
216 self.step(
217 f"split page={child.page} keys={child.notation()} "
218 f"up={middle[0]} new={sibling.page}"
219 )
220 sibling.entries = child.entries[t:]
221 child.entries = child.entries[:t - 1]
222 if not child.is_leaf():
223 sibling.children = child.children[t:]
224 child.children = child.children[:t]
225 node.entries.insert(j, middle)
226 node.children.insert(j + 1, sibling)
227 self.splits += 1

229 def insert(self, key, value):
230 if self.is_full(self.root):
231 self.root = Node(self.free_page(), [], [self.root])
232 self.step(f"grow page={self.root.page}")
233 self.split_child(self.root, 0)
234 node = self.root
235 while not node.is_leaf():
236 j = position(node, key)
237 if self.is_full(node.children[j]):
238 self.split_child(node, j)
239 if key > node.key(j):
240 j += 1
241 node = node.children[j]
242 node.entries.insert(position(node, key), (key, value))
243 self.step(f"place page={node.page} keys={node.notation()}")
244
245 def put(self, key, value):
246 self.steps = []
247 path, index = self.search(key)
248 if index is None:
249 self.insert(key, value)
250 return None
251 node = path[-1]
252 old = node.entries[index][1]
253 node.entries[index] = (key, value)
254 return old

256 def merge_child(self, node, j, case):
257 left = node.children[j]
258 right = node.children.pop(j + 1)
259 left.entries.append(node.entries.pop(j))
260 left.entries.extend(right.entries)
261 left.children.extend(right.children)
262 self.merges += 1
263 self.record(
264 case, f"page={node.page} merge={left.page}+{right.page} keys={left.notation()}"
265 )
266 if node is self.root and not node.entries:
267 self.root = left
268 self.step(f"shrink page={left.page}")
269 return left
270
271 def fill_child(self, node, j):
272 t = self.degree
273 child = node.children[j]
274 left = node.children[j - 1] if j > 0 else None
275 right = node.children[j + 1] if j < len(node.entries) else None
276 if left is not None and len(left.entries) >= t:
277 child.entries.insert(0, node.entries[j - 1])
278 node.entries[j - 1] = left.entries.pop()
279 if not left.is_leaf():
280 child.children.insert(0, left.children.pop())
281 self.borrows += 1
282 self.record(
283 "3a-left",
284 f"page={node.page} child={child.page} from={left.page} keys={child.notation()}",
285 )
286 return child
287 if right is not None and len(right.entries) >= t:
288 child.entries.append(node.entries[j])
289 node.entries[j] = right.entries.pop(0)
290 if not right.is_leaf():
291 child.children.append(right.children.pop(0))
292 self.borrows += 1
293 self.record(
294 "3a-right",
295 f"page={node.page} child={child.page} from={right.page} keys={child.notation()}",
296 )
297 return child
298 if left is not None:
299 return self.merge_child(node, j - 1, "3b-left")
300 return self.merge_child(node, j, "3b-right")

302 def remove(self, node, key):
303 t = self.degree
304 j = position(node, key)
305 if j < len(node.entries) and node.key(j) == key:
306 if node.is_leaf():
307 del node.entries[j]
308 self.record("1", f"page={node.page} keys={node.notation()}")
309 return
310 left = node.children[j]
311 right = node.children[j + 1]
312 if len(left.entries) >= t:
313 predecessor = left.last_entry()
314 self.record("2a", f"page={node.page} pred={predecessor[0]}")
315 self.remove(left, predecessor[0])
316 node.entries[j] = predecessor
317 elif len(right.entries) >= t:
318 successor = right.first_entry()
319 self.record("2b", f"page={node.page} succ={successor[0]}")
320 self.remove(right, successor[0])
321 node.entries[j] = successor
322 else:
323 merged = self.merge_child(node, j, "2c")
324 self.remove(merged, key)
325 return
326 child = node.children[j]
327 if len(child.entries) == t - 1:
328 child = self.fill_child(node, j)
329 self.remove(child, key)

331 def delete(self, key):
332 self.steps = []
333 self.cases = []

336 def scan(self, node, low, high, result):
337 visits = 1
338 m = len(node.entries)
339 for j in range(m + 1):
340 left_open = j == 0 or node.key(j - 1) < high
341 right_open = j == m or node.key(j) > low
342 if not node.is_leaf() and left_open and right_open:
343 visits += self.scan(node.children[j], low, high, result)
344 if j < m and low <= node.key(j) <= high:
345 result.append(node.entries[j])
346 return visits

348 def violation(self):
349 t = self.degree
350 reference_depth = None
351 stack = [(self.root, "root", 0, None, None)]
352 while stack:
353 node, location, depth, low, high = stack.pop()
354 keys = node.keys()
355 m = len(keys)
356 place = f"at={location} node={node.notation()}"
357 if node is not self.root:
358 smallest = t - 1
359 elif node.is_leaf():
360 smallest = 0
361 else:
362 smallest = 1
363 if not smallest <= m <= 2 * t - 1:
364 return f"COUNT {place} keys={m} allowed={smallest}..{2 * t - 1}"
365 for j in range(m - 1):
366 if keys[j] >= keys[j + 1]:
367 return f"ORDER {place} {keys[j]}>={keys[j + 1]}"
368 for key in keys:
369 if (low is not None and key <= low) or (high is not None and key >= high):
370 return (
371 f"RANGE {place} key={key} low={bound_text(low, '-inf')} "
372 f"high={bound_text(high, '+inf')}"
373 )
374 if node.is_leaf():
375 if reference_depth is None:
376 reference_depth = depth
377 elif depth != reference_depth:
378 return f"DEPTH {place} depth={depth} expected={reference_depth}"
379 for j in reversed(range(len(node.children))):
380 child_low = keys[j - 1] if j > 0 else low
381 child_high = keys[j] if j < m else high
382 stack.append(
383 (node.children[j], f"{location}.{j}", depth + 1, child_low, child_high)
384 )
385 return None

394def compact_root(entries, degree):
395 level_entries = entries
396 children = []
397 while True:
398 total = len(level_entries)
399 groups = (total + 2 * degree) // (2 * degree)
400 size, extra = divmod(total - (groups - 1), groups)
401 nodes = []
402 separators = []
403 start = 0
404 child_start = 0
405 for number in range(groups):
406 count = size + 1 if number < extra else size
407 node = Node(0, level_entries[start:start + count])
408 start += count
409 if children:
410 node.children = children[child_start:child_start + count + 1]
411 child_start += count + 1
412 nodes.append(node)
413 if number < groups - 1:
414 separators.append(level_entries[start])
415 start += 1
416 if groups == 1:
417 return nodes[0]
418 level_entries = separators
419 children = nodes

422class NotationParser:
423 def __init__(self, tokens):
424 self.tokens = tokens
425 self.position = 0
426
427 def current(self):
428 if self.position < len(self.tokens):
429 return self.tokens[self.position]
430 return None
431
432 def fail(self):
433 raise CommandError(f"SYNTAX {self.position + 1}")
434
435 def expect(self, token):
436 if self.current() != token:
437 self.fail()
438 self.position += 1
439
440 def at_entry(self):
441 return self.current() not in (None, "(", ")")
442
443 def entry(self):
444 key, value = self.current().split("=")
445 self.position += 1
446 return int(key), value
447
448 def node(self):
449 self.expect("(")
450 node = Node(0)
451 if self.current() == "(":
452 node.children.append(self.node())
453 while self.at_entry():
454 node.entries.append(self.entry())
455 node.children.append(self.node())
456 else:
457 while self.at_entry():
458 node.entries.append(self.entry())
459 self.expect(")")
460 return node
461
462 def tree(self):
463 root = self.node()
464 if self.position < len(self.tokens):
465 self.fail()
466 return root

562 def delete_range(self, arguments):
