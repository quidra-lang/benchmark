388Tree compact_tree(const Tree source)
389 entries := tree_entries(source, source.root)
390 int[] children := []
391 Node[] built := []
392 root := 0
393 while true
394 groups := (int(#entries) + 2 * source.degree) / (2 * source.degree)
395 keys := int(#entries) - (groups - 1)
396 base := keys / groups
397 extra := keys % groups
398 position := 0
399 child_position := 0
400 Entry[] separators := []
401 int[] level := []
402 for: group < groups
403 count := base + (if group < extra then 1 else 0)
404 int[] item_children := if #children == 0 then [] else children[child_position:<child_position + count + 1]
405 item := Node(entries[position:<position + count], item_children)
406 level = level.append(int(#built))
407 built = built.append(item)
408 position += count
409 child_position += count + 1
410 if group + 1 < groups
411 separators = separators.append(entries[position])
412 position += 1
413 if groups == 1
414 root = level[0]
415 break
416 entries = separators
417 children = level
418 result := Tree(source.degree)
419 .result.pages.remove(0)
420 result.root = copy_group(&result, built, root)
421 return result
