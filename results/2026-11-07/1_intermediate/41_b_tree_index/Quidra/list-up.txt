string[] dump_node(const Tree source, int page)
    item := get_node(source, page)
    tokens := ["("]
    for: 0 <= position <= int(#item.entries)
        if not item.leaf()
            tokens = tokens.concat(dump_node(source, item.children[position]))
        if position < int(#item.entries)
            tokens = tokens.append(item.entries[position].display())
    return tokens.append(")")
