12bool sameItems<T>(List<T> a, List<T> b) {
13 if (a.length != b.length) return false;
14 for (var i = 0; i < a.length; i++) {
15 if (a[i] != b[i]) return false;
16 }
17 return true;
18}

20final class Node {
21 final List<Entry> entries;
22 final List<int> children;
23 Node([List<Entry>? entries, List<int>? children])
24 : entries = entries ?? [], children = children ?? [];
25 bool get leaf => children.isEmpty;
26 String get notation => '[${entries.map((entry) => entry.key).join(' ')}]';
27 int position(int key) {
28 var index = 0;
29 while (index < entries.length && key > entries[index].key) {
30 index++;
31 }
32 return index;
33 }
34 Node copy() => Node(List.of(entries), List.of(children));
35 bool sameContent(Node other) =>
36 sameItems(entries, other.entries) && sameItems(children, other.children);
37}

61 ({int writes, int frees}) changes(Map<int, Node> before) {
62 var writes = 0, frees = 0;
63 for (final entry in pages.entries) {
64 final old = before[entry.key];
65 if (old == null || !entry.value.sameContent(old)) writes++;
66 }
67 for (final page in before.keys) {
68 if (!pages.containsKey(page)) frees++;

101 SearchResult search(int key) {
102 var page = root;
103 final path = <String>[];
104 while (true) {
105 final node = pages[page]!, index = node.position(key);
106 path.add(node.notation);
107 if (index < node.entries.length && node.entries[index].key == key) {
108 return (page: page, position: index, path: path);
109 }
110 if (node.leaf) return (page: null, position: -1, path: path);
111 page = node.children[index];
112 }

114 String? violation() {
115 int? referenceDepth;
116 String? visit(int page, String location, int depth, int? low, int? high) {
117 final node = pages[page]!, count = node.entries.length;
118 final minimum = page == root ? (node.leaf ? 0 : 1) : degree - 1;
119 final maximum = 2 * degree - 1, at = 'at=$location node=${node.notation}';
120 if (count < minimum || count > maximum) {
121 return 'COUNT $at keys=$count allowed=$minimum..$maximum';
122 }
123 for (var i = 0; i + 1 < count; i++) {
124 if (node.entries[i].key >= node.entries[i + 1].key) {
125 return 'ORDER $at ${node.entries[i].key}>=${node.entries[i + 1].key}';
126 }
127 }
128 for (final entry in node.entries) {
129 if ((low != null && entry.key <= low) || (high != null && entry.key >= high)) {
130 return 'RANGE $at key=${entry.key} low=${low ?? '-inf'} high=${high ?? '+inf'}';
131 }
132 }
133 if (node.leaf) {
134 final expected = referenceDepth;
135 if (expected != null && depth != expected) {
136 return 'DEPTH $at depth=$depth expected=$expected';
137 }
138 referenceDepth ??= depth;
139 }
140 for (var i = 0; i < node.children.length; i++) {
141 final problem = visit(node.children[i], '$location.$i', depth + 1,
142 i == 0 ? low : node.entries[i - 1].key,
143 i == count ? high : node.entries[i].key);
144 if (problem != null) return problem;
145 }
146 return null;
147 }
148 return visit(root, 'root', 0, null, null);
149 }
150 static Tree load(int degree, List<String> tokens) {
151 final tree = Tree(degree);
152 tree.pages.clear();
153 var position = 0, nextPage = 0;
154 CommandError syntax() => CommandError('SYNTAX ${position + 1}');
155 Entry readEntry() {
156 if (position == tokens.length || tokens[position] == '(' || tokens[position] == ')') {
157 throw syntax();
158 }
159 final pair = tokens[position++].split('=');
160 return (key: int.parse(pair[0]), value: pair[1]);
161 }
162 int readNode() {
163 if (position == tokens.length || tokens[position] != '(') throw syntax();
164 position++;
165 final page = nextPage++, node = Node();
166 tree.pages[page] = node;
167 if (position < tokens.length && tokens[position] == '(') {
168 node.children.add(readNode());
169 while (position == tokens.length || tokens[position] != ')') {
170 node.entries.add(readEntry());
171 node.children.add(readNode());
172 }
173 } else {
174 while (position == tokens.length || tokens[position] != ')') {
175 node.entries.add(readEntry());
176 }
177 }
178 position++;
179 return page;
180 }
181 tree.root = readNode();
182 if (position != tokens.length) throw syntax();
183 return tree;
184 }
185 void split(int parentPage, int index, Trace trace) {
186 final parent = pages[parentPage]!, childPage = parent.children[index];
187 final child = pages[childPage]!, before = child.notation, middle = child.entries[degree - 1];
188 final right = Node(child.entries.sublist(degree), child.children.skip(degree).toList());
189 final newPage = allocate(right);
190 child.entries.removeRange(degree - 1, child.entries.length);
191 if (!child.leaf) child.children.removeRange(degree, child.children.length);
192 parent.entries.insert(index, middle);
193 parent.children.insert(index + 1, newPage);
194 splits++;
195 trace.emit('STEP split page=$childPage keys=$before up=${middle.key} new=$newPage');
196 }
197 String? upsert(int key, String value, Trace trace) {
198 final found = search(key), existing = found.page;
199 if (existing != null) {
200 final old = pages[existing]!.entries[found.position];
201 pages[existing]!.entries[found.position] = (key: key, value: value);
202 return old.value;
203 }
204 if (pages[root]!.entries.length == 2 * degree - 1) {
205 root = allocate(Node([], [root]));
206 trace.emit('STEP grow page=$root');
207 split(root, 0, trace);
208 }
209 var page = root;
210 while (!pages[page]!.leaf) {
211 final node = pages[page]!;
212 var index = node.position(key);
213 if (pages[node.children[index]]!.entries.length == 2 * degree - 1) {
214 split(page, index, trace);
215 if (key > node.entries[index].key) index++;
216 }
217 page = node.children[index];
218 }
219 final leaf = pages[page]!;
220 leaf.entries.insert(leaf.position(key), (key: key, value: value));
221 trace.emit('STEP place page=$page keys=${leaf.notation}');
222 return null;
223 }
224 int merge(int parentPage, int index, String caseName, Trace trace) {
225 final parent = pages[parentPage]!;
226 final leftPage = parent.children[index], rightPage = parent.children[index + 1];
227 final left = pages[leftPage]!, right = pages[rightPage]!;
228 left.entries.add(parent.entries.removeAt(index));
229 left.entries.addAll(right.entries);
230 left.children.addAll(right.children);
231 parent.children.removeAt(index + 1);
232 pages.remove(rightPage);
233 merges++;
234 trace.emit('STEP $caseName page=$parentPage merge=$leftPage+$rightPage keys=${left.notation}');
235 if (parentPage == root && parent.entries.isEmpty) {
236 root = leftPage;
237 pages.remove(parentPage);
238 trace.emit('STEP shrink page=$root');
239 }
240 return leftPage;
241 }
242 void removeAt(int page, int key, List<String> cases, Trace trace) {
243 final node = pages[page]!, index = node.position(key);
244 if (index < node.entries.length && node.entries[index].key == key) {
245 if (node.leaf) {
246 cases.add('1');
247 node.entries.removeAt(index);
248 trace.emit('STEP 1 page=$page keys=${node.notation}');
249 return;
250 }
251 final left = node.children[index], right = node.children[index + 1];
252 if (pages[left]!.entries.length >= degree) {
253 var last = left;
254 while (!pages[last]!.leaf) {
255 last = pages[last]!.children.last;
256 }
257 final predecessor = pages[last]!.entries.last;
258 cases.add('2a');
259 trace.emit('STEP 2a page=$page pred=${predecessor.key}');
260 removeAt(left, predecessor.key, cases, trace);
261 node.entries[index] = predecessor;
262 } else if (pages[right]!.entries.length >= degree) {
263 var first = right;
264 while (!pages[first]!.leaf) {
265 first = pages[first]!.children.first;
266 }
267 final successor = pages[first]!.entries.first;
268 cases.add('2b');
269 trace.emit('STEP 2b page=$page succ=${successor.key}');
270 removeAt(right, successor.key, cases, trace);
271 node.entries[index] = successor;
272 } else {
273 cases.add('2c');
274 final child = merge(page, index, '2c', trace);
275 removeAt(child, key, cases, trace);
276 }
277 return;
278 }
279 var childPage = node.children[index];
280 final child = pages[childPage]!;
281 if (child.entries.length == degree - 1) {
282 final leftPage = index > 0 ? node.children[index - 1] : null;
283 final rightPage = index < node.entries.length ? node.children[index + 1] : null;
284 if (leftPage != null && pages[leftPage]!.entries.length >= degree) {
285 final sibling = pages[leftPage]!;
286 cases.add('3a-left');
287 child.entries.insert(0, node.entries[index - 1]);
288 node.entries[index - 1] = sibling.entries.removeLast();
289 if (!sibling.leaf) child.children.insert(0, sibling.children.removeLast());
290 borrows++;
291 trace.emit('STEP 3a-left page=$page child=$childPage from=$leftPage keys=${child.notation}');
292 } else if (rightPage != null && pages[rightPage]!.entries.length >= degree) {
293 final sibling = pages[rightPage]!;
294 cases.add('3a-right');
295 child.entries.add(node.entries[index]);
296 node.entries[index] = sibling.entries.removeAt(0);
297 if (!sibling.leaf) child.children.add(sibling.children.removeAt(0));
298 borrows++;
299 trace.emit('STEP 3a-right page=$page child=$childPage from=$rightPage keys=${child.notation}');
300 } else if (leftPage != null) {
301 cases.add('3b-left');
302 childPage = merge(page, index - 1, '3b-left', trace);
303 } else {
304 cases.add('3b-right');
305 childPage = merge(page, index, '3b-right', trace);
306 }
307 }
308 removeAt(childPage, key, cases, trace);
309 }
310 List<String> remove(int key, Trace trace) {
311 final cases = <String>[];
312 removeAt(root, key, cases, trace);
313 return cases;
314 }
315 ({List<Entry> entries, int visited}) scan(int low, int high) {
316 final result = <Entry>[];
317 var visited = 0;
318 void visit(int page) {
319 visited++;
320 final node = pages[page]!, count = node.entries.length;
321 for (var i = 0; i <= count; i++) {
322 if (!node.leaf && (i == 0 || node.entries[i - 1].key < high) &&
323 (i == count || node.entries[i].key > low)) {
324 visit(node.children[i]);
325 }
326 if (i < count && low <= node.entries[i].key && node.entries[i].key <= high) {
327 result.add(node.entries[i]);
328 }
329 }
330 }
331 visit(root);
332 return (entries: result, visited: visited);
333 }
334 String dump() {
335 final tokens = <String>[];
336 void visit(int page) {
337 final node = pages[page]!;
338 tokens.add('(');
339 if (!node.leaf) visit(node.children.first);
340 for (var i = 0; i < node.entries.length; i++) {
341 tokens.add(display(node.entries[i]));
342 if (!node.leaf) visit(node.children[i + 1]);
343 }
344 tokens.add(')');
345 }
346 visit(root);
347 return tokens.join(' ');

349 void compact() {
350 var source = entries(), children = <int>[];
351 final built = <Node>[];
352 int oldRoot;
353 while (true) {
354 final groups = (source.length + 2 * degree) ~/ (2 * degree);
355 final keys = source.length - groups + 1, size = keys ~/ groups, extra = keys % groups;
356 var entryIndex = 0, childIndex = 0;
357 final nodes = <int>[], separators = <Entry>[];
358 for (var group = 0; group < groups; group++) {
359 final count = size + (group < extra ? 1 : 0);
360 final node = Node(source.sublist(entryIndex, entryIndex + count),
361 children.isEmpty ? [] : children.sublist(childIndex, childIndex + count + 1));
362 nodes.add(built.length);
363 built.add(node);
364 entryIndex += count;
365 childIndex += count + 1;
366 if (group + 1 < groups) separators.add(source[entryIndex++]);
367 }
368 if (groups == 1) {
369 oldRoot = nodes.first;
370 break;
371 }
372 source = separators;
373 children = nodes;
374 }
375 final replacement = <int, Node>{};
376 var nextPage = 0;
377 int renumber(int oldPage) {
378 final page = nextPage++, old = built[oldPage];
379 replacement[page] = Node(List.of(old.entries), old.children.map(renumber).toList());
380 return page;
381 }
382 root = renumber(oldRoot);
383 pages = replacement;
384 splits = merges = borrows = 0;
385 }

395final class Workbench {
396 Tree? index, saved;
397 Cursor cursor = const Cursor(CursorKind.unset);
398 bool trace = false;
399 static const needsValid = {'INSERT', 'BULKLOAD', 'DELETE', 'DELRANGE', 'RANGE', 'COUNT',
400 'MIN', 'MAX', 'FLOOR', 'CEIL', 'RANK', 'SELECT', 'SEEK', 'NEXT', 'PREV', 'HEIGHT', 'COMPACT'};

402 static Entry? find(List<Entry> entries, bool Function(Entry) predicate, {bool last = false}) {
403 Entry? result;
404 for (final entry in entries) {
405 if (!predicate(entry)) continue;
406 result = entry;
407 if (!last) break;
408 }
409 return result;
410 }
411 static String summary(String prefix, Tree tree) =>
412 '$prefix nodes=${tree.pages.length} keys=${tree.entries().length} ${tree.violation() == null ? 'OK' : 'DAMAGED'}';
413 List<String> process(String line) {
414 final words = line.split(' '), command = words[0];
415 if (command == 'TRACE') {
416 trace = words[1] == 'ON';
417 return ['TRACE ${words[1]}'];
418 }
419 final first = words.length > 1 ? int.parse(words[1]) : 0;
420 final ranged = const {'DELRANGE', 'RANGE', 'COUNT'}.contains(command);
421 final second = ranged ? int.parse(words[2]) : 0;
422 if (command == 'CREATE' || command == 'LOAD') {
423 if (first < 2 || first > 5) throw CommandError('BAD DEGREE $first');
424 final replacement = command == 'CREATE' ? Tree(first) : Tree.load(first, words.sublist(2));
425 index = replacement;
426 cursor = const Cursor(CursorKind.unset);
427 return [command == 'CREATE' ? 'CREATE t=$first' : summary('LOAD t=$first', replacement)];
428 }
429 var tree = index;
430 if (tree == null) throw const CommandError('NO INDEX');
431 if (needsValid.contains(command) && tree.violation() != null) {
432 throw const CommandError('DAMAGED');
433 }
434 final entries = tree.entries();
435 if (ranged && first > second) throw CommandError('BAD RANGE $first $second');
436 if (command == 'SELECT' && (first < 1 || first > entries.length)) {
437 throw CommandError('BAD POSITION $first');
438 }
439 if ((command == 'NEXT' || command == 'PREV') && cursor.kind == CursorKind.unset) {
440 throw const CommandError('NO CURSOR');
441 }
442 if (command == 'RESTORE' && saved == null) throw const CommandError('NO SAVE');
443 final output = <String>[], steps = Trace(trace && (command == 'INSERT' || command == 'DELETE'));
444 switch (command) {
445 case 'INSERT':
446 final before = tree.copyPages(), splits = tree.splits;
447 final old = tree.upsert(first, words[2], steps), writes = tree.changes(before).writes;
448 output.addAll(steps.lines);
449 output.add(old == null
450 ? 'INSERT $first NEW splits=${tree.splits - splits} height=${tree.height} writes=$writes'
451 : 'INSERT $first UPDATED old=$old writes=$writes');
452 case 'BULKLOAD':
453 final before = tree.copyPages(), splits = tree.splits;
454 var inserted = 0, updated = 0;
455 for (var i = 1; i < words.length; i += 2) {
456 if (tree.upsert(int.parse(words[i]), words[i + 1], steps) == null) {
457 inserted++;
458 } else {
459 updated++;
460 }
461 }
462 output.add('BULKLOAD pairs=${(words.length - 1) ~/ 2} new=$inserted updated=$updated splits=${tree.splits - splits} height=${tree.height} writes=${tree.changes(before).writes}');
463 case 'DELETE':
464 final found = tree.search(first), page = found.page;
465 if (page == null) {
466 output.add('DELETE $first NOT FOUND');
467 } else {
468 final value = tree.pages[page]!.entries[found.position].value, before = tree.copyPages();
469 final cases = tree.remove(first, steps), changes = tree.changes(before);
470 output.addAll(steps.lines);
471 output.add('DELETE $first REMOVED value=$value cases=${cases.join(',')} height=${tree.height} writes=${changes.writes} frees=${changes.frees}');
472 }
473 case 'DELRANGE':
474 final before = tree.copyPages(), merges = tree.merges, borrows = tree.borrows;
475 var removed = 0;
476 for (final entry in entries) {
477 if (first <= entry.key && entry.key <= second) {
478 tree.remove(entry.key, steps);
479 removed++;
480 }
481 }
482 final changes = tree.changes(before);
483 output.add('DELRANGE $first $second removed=$removed merges=${tree.merges - merges} borrows=${tree.borrows - borrows} height=${tree.height} writes=${changes.writes} frees=${changes.frees}');
484 case 'GET':
485 final found = tree.search(first), page = found.page;
486 final result = page == null ? 'NOT FOUND' : 'FOUND value=${tree.pages[page]!.entries[found.position].value}';
487 output.add('GET $first $result path=${found.path.join(' -> ')}');
488 case 'RANGE':
489 case 'COUNT':
490 final result = tree.scan(first, second);
491 final suffix = command == 'RANGE' ? result.entries.map((entry) => ' ${display(entry)}').join() : '';
492 output.add('$command $first $second count=${result.entries.length} visited=${result.visited}$suffix');
493 case 'MIN':
494 case 'MAX':
495 final result = entries.isEmpty ? 'EMPTY' : display(command == 'MIN' ? entries.first : entries.last);
496 output.add('$command $result');
497 case 'FLOOR':
498 case 'CEIL':
499 final entry = command == 'FLOOR'
500 ? find(entries, (value) => value.key <= first, last: true)
501 : find(entries, (value) => value.key >= first);
502 output.add('$command $first ${entry == null ? 'NONE' : display(entry)}');
503 case 'RANK':
504 output.add('RANK $first ${entries.where((entry) => entry.key < first).length}');
505 case 'SELECT':
506 output.add('SELECT $first ${display(entries[first - 1])}');
507 case 'SEEK':
508 final entry = find(entries, (value) => value.key >= first);
509 cursor = entry == null ? const Cursor(CursorKind.end) : Cursor(CursorKind.key, entry.key);
510 output.add('SEEK $first ${entry == null ? 'END' : display(entry)}');
511 case 'NEXT':
512 case 'PREV':
513 Entry? entry;
514 if (command == 'NEXT') {
515 if (cursor.kind == CursorKind.begin) {
516 entry = find(entries, (_) => true);
517 } else if (cursor.kind == CursorKind.key) {
518 entry = find(entries, (value) => value.key > cursor.key);
519 }
520 } else {
521 if (cursor.kind == CursorKind.end) {
522 entry = find(entries, (_) => true, last: true);
523 } else if (cursor.kind == CursorKind.key) {
524 entry = find(entries, (value) => value.key < cursor.key, last: true);
525 }
526 }
527 cursor = entry == null
528 ? Cursor(command == 'NEXT' ? CursorKind.end : CursorKind.begin)
529 : Cursor(CursorKind.key, entry.key);
530 output.add('$command ${entry == null ? (command == 'NEXT' ? 'END' : 'BEGIN') : display(entry)}');
531 case 'PRINT':
532 var level = [tree.root], depth = 0;
533 while (level.isNotEmpty) {
534 output.add('LEVEL $depth ${level.map((page) => tree!.pages[page]!.notation).join(' ')}');
535 level = level.expand((page) => tree!.pages[page]!.children).toList();
536 depth++;
537 }
538 case 'PAGES':
539 for (final page in tree.pages.keys.toList()..sort()) {
540 final node = tree.pages[page]!;
541 output.add('PAGE $page ${node.notation} kids=${node.leaf ? '-' : node.children.join(',')}');
542 }
543 case 'DUMP':
544 output.add('DUMP ${tree.degree} ${tree.dump()}');
545 case 'HEIGHT':
546 output.add('HEIGHT ${tree.height}');
547 case 'STATS':
548 final leaves = tree.pages.values.where((node) => node.leaf).length;
549 final fill = 100 * entries.length ~/ ((2 * tree.degree - 1) * tree.pages.length);
550 output.add('STATS t=${tree.degree} nodes=${tree.pages.length} leaves=$leaves keys=${entries.length} fill=$fill% splits=${tree.splits} merges=${tree.merges} borrows=${tree.borrows}');
551 case 'VERIFY':
552 final problem = tree.violation();
553 output.add(problem == null ? 'VERIFY OK' : 'VERIFY FAIL $problem');
554 case 'REBUILD':
555 tree = Tree(tree.degree);
556 for (final entry in entries) {
557 tree.upsert(entry.key, entry.value, steps);
558 }
559 output.add('REBUILD entries=${entries.length} keys=${tree.entries().length} nodes=${tree.pages.length} splits=${tree.splits} height=${tree.height}');
560 case 'COMPACT':
561 final nodes = tree.pages.length;
562 tree.compact();
563 output.add('COMPACT nodes=$nodes->${tree.pages.length} height=${tree.height}');
564 case 'SAVE':
565 saved = tree.copy();
566 output.add(summary('SAVE', tree));
567 case 'RESTORE':
568 tree = saved!.copy();
569 output.add(summary('RESTORE', tree));
570 }
571 index = tree;
572 return output;
573 }
574}
