3struct CommandError <: Exception
4 message::String
5end
6fail(message) = throw(CommandError(message))

8struct Entry
9 key::Int
10 value::String
11end
12Base.:(==)(a::Entry, b::Entry) = a.key == b.key && a.value == b.value
13entry_display(entry::Entry) = "$(entry.key)=$(entry.value)"
14struct Node
15 entries::Vector{Entry}
16 kids::Vector{Int}
17end
18Node() = Node(Entry[], Int[])
19Base.copy(node::Node) = Node(copy(node.entries), copy(node.kids))
20Base.:(==)(a::Node, b::Node) = a.entries == b.entries && a.kids == b.kids
21notation(node::Node) = "[" * join((entry.key for entry in node.entries), " ") * "]"
22mutable struct Tree
23 degree::Int
24 root::Int
25 pages::Dict{Int,Node}
26 splits::Int
27 merges::Int
28 borrows::Int
29end
30Tree(degree::Int) = Tree(degree, 0, Dict(0 => Node()), 0, 0, 0)
31Base.copy(tree::Tree) = Tree(tree.degree, tree.root, Dict(page => copy(node) for (page, node) in tree.pages), tree.splits, tree.merges, tree.borrows)
32function allocate!(tree::Tree, node::Node = Node())
33 page = 0
34 while haskey(tree.pages, page)
35 page += 1
36 end
37 tree.pages[page] = node
38 page
39end
40function node_position(node::Node, key::Int)
41 found = findfirst(entry -> entry.key >= key, node.entries)
42 isnothing(found) ? length(node.entries) + 1 : found
43end
44struct Search
45 page::Int
46 at::Int
47 found::Bool
48 path::Vector{Int}
49end
50function search_tree(tree::Tree, key::Int)
51 page = tree.root
52 path = Int[]
53 while true
54 push!(path, page)
55 node = tree.pages[page]
56 at = node_position(node, key)
57 found = at <= length(node.entries) && node.entries[at].key == key
58 (found || isempty(node.kids)) && return Search(page, at, found, path)
59 page = node.kids[at]
60 end
61end
62function inorder(tree::Tree)
63 result = Entry[]
64 function visit(page)
65 node = tree.pages[page]
66 for j in eachindex(node.entries)
67 isempty(node.kids) || visit(node.kids[j])
68 push!(result, node.entries[j])
69 end
70 isempty(node.kids) || visit(last(node.kids))
71 end
72 visit(tree.root)
73 result
74end
75key_count(tree::Tree) = sum(length(node.entries) for node in values(tree.pages))
76function height(tree::Tree)
77 result = 0
78 node = tree.pages[tree.root]
79 while !isempty(node.kids)
80 result += 1
81 node = tree.pages[first(node.kids)]
82 end
83 result
84end
85function check_tree(tree::Tree)
86 reference = Ref{Union{Nothing,Int}}(nothing)
87 bound_text(value, lower) = isnothing(value) ? (lower ? "-inf" : "+inf") : string(value)
88 function visit(page, location, depth, lower, upper)
89 node = tree.pages[page]
90 n = length(node.entries)
91 minimum = page == tree.root ? (isempty(node.kids) ? 0 : 1) : tree.degree - 1
92 maximum = 2 * tree.degree - 1
93 detail = " at=$location node=$(notation(node))"
94 if n < minimum || n > maximum
95 return "FAIL COUNT$detail keys=$n allowed=$(minimum)..$(maximum)"
96 end
97 for j in 1:(n - 1)
98 if node.entries[j].key >= node.entries[j + 1].key
99 return "FAIL ORDER$detail $(node.entries[j].key)>=$(node.entries[j + 1].key)"
100 end
101 end
102 for entry in node.entries
103 if (!isnothing(lower) && entry.key <= lower) || (!isnothing(upper) && entry.key >= upper)
104 return "FAIL RANGE$detail key=$(entry.key) low=$(bound_text(lower, true)) high=$(bound_text(upper, false))"
105 end
106 end
107 if isempty(node.kids)
108 if isnothing(reference[])
109 reference[] = depth
110 elseif depth != reference[]
111 return "FAIL DEPTH$detail depth=$depth expected=$(reference[])"
112 end
113 else
114 for j in eachindex(node.kids)
115 problem = visit(node.kids[j], "$(location).$(j - 1)", depth + 1,
116 j == 1 ? lower : node.entries[j - 1].key, j > n ? upper : node.entries[j].key)
117 isnothing(problem) || return problem
118 end
119 end
120 nothing
121 end
122 visit(tree.root, "root", 0, nothing, nothing)
123end
124tree_status(tree::Tree) = isnothing(check_tree(tree)) ? "OK" : "DAMAGED"
125function changes(tree::Tree, before::Dict{Int,Node})
126 writes = sum(!haskey(before, page) || node != before[page] for (page, node) in tree.pages)
127 frees = count(page -> !haskey(tree.pages, page), keys(before))
128 (writes = writes, frees = frees)
129end
130mutable struct Trace
131 enabled::Bool
132 lines::Vector{String}
133 cases::Vector{String}
134end
135Trace(enabled) = Trace(enabled, String[], String[])
136function step!(trace::Union{Nothing,Trace}, text::String)
137 isnothing(trace) || !trace.enabled || push!(trace.lines, text)
138 nothing
139end
140function add_case!(trace::Union{Nothing,Trace}, name::String)
141 isnothing(trace) || push!(trace.cases, name)
142 nothing
143end
144function split_child!(tree::Tree, page::Int, at::Int, trace)
145 parent = tree.pages[page]
146 left_page = parent.kids[at]
147 left = tree.pages[left_page]
148 degree = tree.degree
149 middle = left.entries[degree]
150 right = Node(left.entries[(degree + 1):end], isempty(left.kids) ? Int[] : left.kids[(degree + 1):end])
151 right_page = allocate!(tree, right)
152 step!(trace, "STEP split page=$left_page keys=$(notation(left)) up=$(middle.key) new=$right_page")
153 resize!(left.entries, degree - 1)
154 isempty(left.kids) || resize!(left.kids, degree)
155 insert!(parent.entries, at, middle)
156 insert!(parent.kids, at + 1, right_page)
157 tree.splits += 1
158end
159function put_entry!(tree::Tree, entry::Entry, trace = nothing)
160 found = search_tree(tree, entry.key)
161 if found.found
162 node = tree.pages[found.page]
163 old = node.entries[found.at].value
164 node.entries[found.at] = entry
165 return (added = false, old = old)
166 end
167 if length(tree.pages[tree.root].entries) == 2 * tree.degree - 1
168 tree.root = allocate!(tree, Node(Entry[], [tree.root]))
169 step!(trace, "STEP grow page=$(tree.root)")
170 split_child!(tree, tree.root, 1, trace)
171 end
172 page = tree.root
173 while true
174 node = tree.pages[page]
175 at = node_position(node, entry.key)
176 if isempty(node.kids)
177 insert!(node.entries, at, entry)
178 step!(trace, "STEP place page=$page keys=$(notation(node))")
179 return (added = true, old = nothing)
180 end
181 if length(tree.pages[node.kids[at]].entries) == 2 * tree.degree - 1
182 split_child!(tree, page, at, trace)
183 entry.key > node.entries[at].key && (at += 1)
184 end
185 page = node.kids[at]
186 end
187end
188function merge_children!(tree::Tree, page::Int, at::Int, name::String, trace)
189 parent = tree.pages[page]
190 left_page, right_page = parent.kids[at], parent.kids[at + 1]
191 left, right = tree.pages[left_page], tree.pages[right_page]
192 push!(left.entries, parent.entries[at])
193 append!(left.entries, right.entries)
194 append!(left.kids, right.kids)
195 deleteat!(parent.entries, at)
196 deleteat!(parent.kids, at + 1)
197 delete!(tree.pages, right_page)
198 tree.merges += 1
199 step!(trace, "STEP $name page=$page merge=$left_page+$right_page keys=$(notation(left))")
200 if page == tree.root && isempty(parent.entries)
201 tree.root = left_page
202 delete!(tree.pages, page)
203 step!(trace, "STEP shrink page=$left_page")
204 end
205 left_page
206end
207function borrow_child!(tree::Tree, page::Int, at::Int, from_left::Bool, trace)
208 parent = tree.pages[page]
209 child_page = parent.kids[at]
210 sibling_page = parent.kids[at + (from_left ? -1 : 1)]
211 child, sibling = tree.pages[child_page], tree.pages[sibling_page]
212 separator = at - Int(from_left)
213 if from_left
214 pushfirst!(child.entries, parent.entries[separator])
215 parent.entries[separator] = pop!(sibling.entries)
216 isempty(sibling.kids) || pushfirst!(child.kids, pop!(sibling.kids))
217 else
218 push!(child.entries, parent.entries[separator])
219 parent.entries[separator] = popfirst!(sibling.entries)
220 isempty(sibling.kids) || push!(child.kids, popfirst!(sibling.kids))
221 end
222 tree.borrows += 1
223 side = from_left ? "left" : "right"
224 step!(trace, "STEP 3a-$side page=$page child=$child_page from=$sibling_page keys=$(notation(child))")
225end
226function erase_entry!(tree::Tree, page::Int, key::Int, trace = nothing)
227 node = tree.pages[page]
228 at = node_position(node, key)
229 if at <= length(node.entries) && node.entries[at].key == key
230 if isempty(node.kids)
231 add_case!(trace, "1")
232 deleteat!(node.entries, at)
233 step!(trace, "STEP 1 page=$page keys=$(notation(node))")
234 else
235 left, right = node.kids[at], node.kids[at + 1]
236 if length(tree.pages[left].entries) >= tree.degree || length(tree.pages[right].entries) >= tree.degree
237 predecessor = length(tree.pages[left].entries) >= tree.degree
238 child = predecessor ? left : right
239 extreme = tree.pages[child]
240 while !isempty(extreme.kids)
241 extreme = tree.pages[predecessor ? last(extreme.kids) : first(extreme.kids)]
242 end
243 replacement = predecessor ? last(extreme.entries) : first(extreme.entries)
244 name, detail = predecessor ? ("2a", "pred") : ("2b", "succ")
245 add_case!(trace, name)
246 step!(trace, "STEP $name page=$page $detail=$(replacement.key)")
247 erase_entry!(tree, child, replacement.key, trace)
248 node.entries[at] = replacement
249 else
250 add_case!(trace, "2c")
251 child = merge_children!(tree, page, at, "2c", trace)
252 erase_entry!(tree, child, key, trace)
253 end
254 end
255 else
256 child = node.kids[at]
257 if length(tree.pages[child].entries) == tree.degree - 1
258 if at > 1 && length(tree.pages[node.kids[at - 1]].entries) >= tree.degree
259 add_case!(trace, "3a-left")
260 borrow_child!(tree, page, at, true, trace)
261 elseif at < length(node.kids) && length(tree.pages[node.kids[at + 1]].entries) >= tree.degree
262 add_case!(trace, "3a-right")
263 borrow_child!(tree, page, at, false, trace)
264 else
265 name = at > 1 ? "3b-left" : "3b-right"
266 add_case!(trace, name)
267 child = merge_children!(tree, page, at > 1 ? at - 1 : at, name, trace)
268 end
269 end
270 erase_entry!(tree, child, key, trace)
271 end
272 nothing
273end
274function scan_tree(tree::Tree, lower::Int, upper::Int)
275 visited = 0
276 result = Entry[]
277 function visit(page)
278 visited += 1
279 node = tree.pages[page]
280 n = length(node.entries)
281 for j in 1:(n + 1)
282 if !isempty(node.kids) && (j == 1 || node.entries[j - 1].key < upper) && (j > n || node.entries[j].key > lower)
283 visit(node.kids[j])
284 end
285 if j <= n && lower <= node.entries[j].key <= upper
286 push!(result, node.entries[j])
287 end
288 end
289 end
290 visit(tree.root)
291 (entries = result, visited = visited)
292end
293function dump_tree(tree::Tree, page::Int = tree.root)
294 node = tree.pages[page]
295 result = ["("]
296 for j in eachindex(node.entries)
297 isempty(node.kids) || append!(result, dump_tree(tree, node.kids[j]))
298 push!(result, entry_display(node.entries[j]))
299 end
300 isempty(node.kids) || append!(result, dump_tree(tree, last(node.kids)))
301 push!(result, ")")
302end
303function load_tree(degree::Int, tokens::Vector{String})
304 tree = Tree(degree)
305 empty!(tree.pages)
306 at = 1
307 peek() = at <= length(tokens) ? tokens[at] : ""
308 syntax() = fail("ERROR SYNTAX $at")
309 function expect(text)
310 peek() == text || syntax()
311 at += 1
312 end
313 function read_entry()
314 text = peek()
315 text in ("", "(", ")") && syntax()
316 fields = split(text, '='; limit = 2)
317 at += 1
318 Entry(parse(Int, fields[1]), String(fields[2]))
319 end
320 function read_node()
321 expect("(")
322 page = allocate!(tree)
323 node = tree.pages[page]
324 internal = peek() == "("
325 internal && push!(node.kids, read_node())
326 while peek() != ")"
327 push!(node.entries, read_entry())
328 internal && push!(node.kids, read_node())
329 end
330 expect(")")
331 page
332 end
333 tree.root = read_node()
334 at <= length(tokens) && syntax()
335 tree
336end
337struct Group
338 entries::Vector{Entry}
339 kids::Vector{Group}
340end
341function compact_tree(source::Tree)
342 entries = inorder(source)
343 candidates = Group[]
344 degree = source.degree
345 local level
346 while true
347 n = length(entries)
348 g = div(n + 2 * degree, 2 * degree)
349 d, r = divrem(n - g + 1, g)
350 level = Group[]
351 separators = Entry[]
352 at, child_at = 1, 1
353 for i in 1:g
354 amount = d + Int(i <= r)
355 kids = isempty(candidates) ? Group[] : candidates[child_at:(child_at + amount)]
356 push!(level, Group(entries[at:(at + amount - 1)], kids))
357 at += amount
358 child_at += amount + 1
359 if i < g
360 push!(separators, entries[at])
361 at += 1
362 end
363 end
364 g == 1 && break
365 entries, candidates = separators, level
366 end
367 tree = Tree(degree)
368 empty!(tree.pages)
369 function assign_pages(group::Group)
370 page = allocate!(tree)
371 kids = Int[assign_pages(child) for child in group.kids]
372 tree.pages[page] = Node(copy(group.entries), kids)
373 page
374 end
375 tree.root = assign_pages(only(level))
376 tree
377end

379@enum CursorKind UNSET AT_KEY BEGINNING ENDING

383end
384mutable struct Laboratory
385 index::Union{Nothing,Tree}
386 saved::Union{Nothing,Tree}
387 cursor::Cursor
388 trace::Bool
389end
390Laboratory() = Laboratory(nothing, nothing, Cursor(UNSET, 0), false)
391
392function execute!(lab::Laboratory, words::Vector{String})
393 command, args = words[1], words[2:end]
394 if command == "TRACE"
395 lab.trace = args[1] == "ON"
396 return ["TRACE $(args[1])"]
397 elseif command in ("CREATE", "LOAD")
398 degree = parse(Int, args[1])
399 2 <= degree <= 5 || fail("ERROR BAD DEGREE $degree")
400 tree = command == "CREATE" ? Tree(degree) : load_tree(degree, args[2:end])
401 lab.index = tree
402 lab.cursor = Cursor(UNSET, 0)
403 return [command == "CREATE" ? "CREATE t=$degree" : "LOAD t=$degree nodes=$(length(tree.pages)) keys=$(key_count(tree)) $(tree_status(tree))"]
404 end
405 tree = lab.index
406 isnothing(tree) && fail("ERROR NO INDEX")
407 needs_valid = command in ("INSERT", "BULKLOAD", "DELETE", "DELRANGE", "RANGE", "COUNT", "MIN", "MAX", "FLOOR", "CEIL", "RANK", "SELECT", "SEEK", "NEXT", "PREV", "HEIGHT", "COMPACT")
408 needs_valid && !isnothing(check_tree(tree)) && fail("ERROR DAMAGED")
409 command in ("DELRANGE", "RANGE", "COUNT") && parse(Int, args[1]) > parse(Int, args[2]) && fail("ERROR BAD RANGE $(args[1]) $(args[2])")
410 command == "SELECT" && !(1 <= parse(Int, args[1]) <= key_count(tree)) && fail("ERROR BAD POSITION $(args[1])")
411 command in ("NEXT", "PREV") && lab.cursor.kind == UNSET && fail("ERROR NO CURSOR")
412 command == "RESTORE" && isnothing(lab.saved) && fail("ERROR NO SAVE")
413 if command == "INSERT"
414 key = parse(Int, args[1])
415 before = Dict(page => copy(node) for (page, node) in tree.pages)
416 splits = tree.splits
417 trace = Trace(lab.trace)
418 result = put_entry!(tree, Entry(key, args[2]), trace)
419 delta = changes(tree, before)
420 text = result.added ? "INSERT $key NEW splits=$(tree.splits - splits) height=$(height(tree)) writes=$(delta.writes)" :
421 "INSERT $key UPDATED old=$(result.old) writes=$(delta.writes)"
422 return vcat(trace.lines, [text])
423 elseif command == "BULKLOAD"
424 before = Dict(page => copy(node) for (page, node) in tree.pages)
425 splits, added = tree.splits, 0
426 for i in 1:2:length(args)
427 added += put_entry!(tree, Entry(parse(Int, args[i]), args[i + 1])).added
428 end
429 pairs = div(length(args), 2)
430 return ["BULKLOAD pairs=$pairs new=$added updated=$(pairs - added) splits=$(tree.splits - splits) height=$(height(tree)) writes=$(changes(tree, before).writes)"]
431 elseif command == "DELETE"
432 key = parse(Int, args[1])
433 found = search_tree(tree, key)
434 found.found || return ["DELETE $key NOT FOUND"]
435 old = tree.pages[found.page].entries[found.at].value
436 before = Dict(page => copy(node) for (page, node) in tree.pages)
437 trace = Trace(lab.trace)
438 erase_entry!(tree, tree.root, key, trace)
439 delta = changes(tree, before)
440 return vcat(trace.lines, ["DELETE $key REMOVED value=$old cases=$(join(trace.cases, ',')) height=$(height(tree)) writes=$(delta.writes) frees=$(delta.frees)"])
441 elseif command == "DELRANGE"
442 lower, upper = parse.(Int, args)
443 before = Dict(page => copy(node) for (page, node) in tree.pages)
444 merges, borrows = tree.merges, tree.borrows
445 selected = [entry.key for entry in inorder(tree) if lower <= entry.key <= upper]
446 for key in selected
447 erase_entry!(tree, tree.root, key)
448 end
449 delta = changes(tree, before)
450 return ["DELRANGE $lower $upper removed=$(length(selected)) merges=$(tree.merges - merges) borrows=$(tree.borrows - borrows) height=$(height(tree)) writes=$(delta.writes) frees=$(delta.frees)"]
451 elseif command == "GET"
452 key = parse(Int, args[1])
453 result = search_tree(tree, key)
454 outcome = result.found ? "FOUND value=$(tree.pages[result.page].entries[result.at].value)" : "NOT FOUND"
455 path = join((notation(tree.pages[page]) for page in result.path), " -> ")
456 return ["GET $key $outcome path=$path"]
457 elseif command in ("RANGE", "COUNT")
458 lower, upper = parse.(Int, args)
459 result = scan_tree(tree, lower, upper)
460 text = "$command $lower $upper count=$(length(result.entries)) visited=$(result.visited)"
461 command == "RANGE" && !isempty(result.entries) && (text *= " " * join(entry_display.(result.entries), " "))
462 return [text]
463 elseif command in ("MIN", "MAX", "FLOOR", "CEIL", "RANK", "SELECT", "SEEK", "NEXT", "PREV")
464 entries = inorder(tree)
465 if command in ("MIN", "MAX")
466 isempty(entries) && return ["$command EMPTY"]
467 return ["$command $(entry_display(command == "MIN" ? first(entries) : last(entries)))"]
468 elseif command == "SELECT"
469 return ["SELECT $(args[1]) $(entry_display(entries[parse(Int, args[1])]))"]
470 elseif command == "RANK"
471 return ["RANK $(args[1]) $(count(entry -> entry.key < parse(Int, args[1]), entries))"]
472 elseif command in ("FLOOR", "CEIL", "SEEK")
473 key = parse(Int, args[1])
474 chosen = command == "FLOOR" ? findlast(entry -> entry.key <= key, entries) : findfirst(entry -> entry.key >= key, entries)
475 if command == "SEEK"
476 lab.cursor = isnothing(chosen) ? Cursor(ENDING, 0) : Cursor(AT_KEY, entries[chosen].key)
477 end
478 text = isnothing(chosen) ? (command == "SEEK" ? "END" : "NONE") : entry_display(entries[chosen])
479 return ["$command $key $text"]
480 end
481 cursor = lab.cursor
482 if command == "NEXT"
483 chosen = cursor.kind == ENDING ? nothing : findfirst(entry -> cursor.kind == BEGINNING || entry.key > cursor.key, entries)
484 else
485 chosen = cursor.kind == BEGINNING ? nothing : findlast(entry -> cursor.kind == ENDING || entry.key < cursor.key, entries)
486 end
487 if isnothing(chosen)
488 lab.cursor = Cursor(command == "NEXT" ? ENDING : BEGINNING, 0)
489 return ["$command $(command == "NEXT" ? "END" : "BEGIN")"]
490 end
491 lab.cursor = Cursor(AT_KEY, entries[chosen].key)
492 return ["$command $(entry_display(entries[chosen]))"]
493 elseif command == "PRINT"
494 level, depth = [tree.root], 0
495 result = String[]
496 while !isempty(level)
497 push!(result, "LEVEL $depth " * join((notation(tree.pages[page]) for page in level), " "))
498 level = Int[child for page in level for child in tree.pages[page].kids]
499 depth += 1
500 end
501 return result
502 elseif command == "PAGES"
503 return ["PAGE $page $(notation(tree.pages[page])) kids=$(isempty(tree.pages[page].kids) ? "-" : join(tree.pages[page].kids, ','))" for page in sort!(collect(keys(tree.pages)))]
504 elseif command == "DUMP"
505 return ["DUMP $(tree.degree) $(join(dump_tree(tree), ' '))"]
506 elseif command == "HEIGHT"
507 return ["HEIGHT $(height(tree))"]
508 elseif command == "STATS"
509 n = length(tree.pages)
510 leaves = count(node -> isempty(node.kids), values(tree.pages))
511 nkeys = key_count(tree)
512 fill = div(100 * nkeys, (2 * tree.degree - 1) * n)
513 return ["STATS t=$(tree.degree) nodes=$n leaves=$leaves keys=$nkeys fill=$fill% splits=$(tree.splits) merges=$(tree.merges) borrows=$(tree.borrows)"]
514 elseif command == "VERIFY"
515 problem = check_tree(tree)
516 return ["VERIFY $(isnothing(problem) ? "OK" : problem)"]
517 elseif command == "REBUILD"
518 entries = inorder(tree)
519 rebuilt = Tree(tree.degree)
520 for entry in entries
521 put_entry!(rebuilt, entry)
522 end
523 lab.index = rebuilt
524 return ["REBUILD entries=$(length(entries)) keys=$(key_count(rebuilt)) nodes=$(length(rebuilt.pages)) splits=$(rebuilt.splits) height=$(height(rebuilt))"]
525 elseif command == "COMPACT"
526 compacted = compact_tree(tree)
527 lab.index = compacted
528 return ["COMPACT nodes=$(length(tree.pages))->$(length(compacted.pages)) height=$(height(compacted))"]
529 elseif command == "SAVE"
530 lab.saved = copy(tree)
531 return ["SAVE nodes=$(length(tree.pages)) keys=$(key_count(tree)) $(tree_status(tree))"]
532 else
533 restored = copy(lab.saved::Tree)
534 lab.index = restored
535 return ["RESTORE nodes=$(length(restored.pages)) keys=$(key_count(restored)) $(tree_status(restored))"]
536 end
537end

539function main()
540 eof(stdin) && return
541 commands = parse(Int, readline())
542 lab = Laboratory()
543 for _ in 1:commands
544 words = String.(split(readline(), ' '))
545 try
546 foreach(println, execute!(lab, words))
547 catch error
548 error isa CommandError || rethrow()
549 println(error.message)
550 end
551 end
552end
553main()
