20 std::string result;
21 for (const auto& word : words) {
22 if (!result.empty()) result += separator;

33struct Entry {
34 int key;
35 std::string value;
36 std::string display() const { return text(key, '=', value); }
37 bool operator==(const Entry& other) const { return key == other.key && value == other.value; }
38};
39
40struct Node {
41 std::vector<Entry> entries;
42 std::vector<int> children;
43 bool leaf() const { return children.empty(); }
44 int count() const { return static_cast<int>(entries.size()); }
45 int position(int key) const {
46 int index = 0;
47 while (index < count() && entries[index].key < key) ++index;
48 return index;
49 }
50 std::string notation() const {
51 std::vector<std::string> keys;
52 for (const auto& entry : entries) keys.push_back(std::to_string(entry.key));
53 return '[' + join(keys, " ") + ']';
54 }
55 bool operator==(const Node& other) const { return entries == other.entries && children == other.children; }
56};

58struct SearchResult {
59 int page = -1, index = -1;
60 std::vector<std::string> path;
61 bool found() const { return page >= 0; }
62};

64class Tree {
65public:
66 using Snapshot = std::map<int, Node>;
67 struct Changes { int writes = 0, frees = 0; };
68 struct Scan { std::vector<Entry> entries; int visited = 0; };
69 struct Upsert { bool inserted; std::string old; };
70 int degree, root = 0, splits = 0, merges = 0, borrows = 0;
71 Snapshot pages{{0, Node{}}};
72
73 explicit Tree(int degree) : degree(degree) {}
74 std::vector<Entry> entries() const {
75 std::vector<Entry> result;
76 std::function<void(int)> visit = [&](int page) {
77 const auto& node = pages.at(page);
78 if (!node.leaf()) visit(node.children[0]);
79 for (int index = 0; index < node.count(); ++index) {
80 result.push_back(node.entries[index]);
81 if (!node.leaf()) visit(node.children[index + 1]);
82 }
83 };
84 visit(root);
85 return result;
86 }
87 int height() const {
88 int height = 0, page = root;
89 while (!pages.at(page).leaf()) { ++height; page = pages.at(page).children[0]; }
90 return height;
91 }
92 Changes changes(const Snapshot& before) const {
93 Changes result;
94 for (const auto& [page, node] : pages) {
95 auto previous = before.find(page);
96 if (previous == before.end() || !(previous->second == node)) ++result.writes;
97 }
98 for (const auto& [page, node] : before) if (!pages.count(page)) ++result.frees;
99 return result;
100 }
101 int allocate(Node node) {
102 int page = 0;
103 while (pages.count(page)) ++page;
104 pages.emplace(page, std::move(node));
105 return page;
106 }
107 SearchResult search(int key) const {
108 SearchResult result;
109 int page = root;
110 while (true) {
111 const auto& node = pages.at(page);
112 result.path.push_back(node.notation());
113 int index = node.position(key);
114 if (index < node.count() && node.entries[index].key == key) {
115 result.page = page; result.index = index; return result;
116 }
117 if (node.leaf()) return result;
118 page = node.children[index];
119 }
120 }
121 std::optional<std::string> violation() const {
122 int referenceDepth = -1;
123 std::function<std::optional<std::string>(int, const std::string&, int, std::optional<int>, std::optional<int>)> visit;
124 visit = [&](int page, const std::string& location, int depth, std::optional<int> low, std::optional<int> high)
125 -> std::optional<std::string> {
126 const auto& node = pages.at(page);
127 int minimum = page == root ? (node.leaf() ? 0 : 1) : degree - 1, maximum = 2 * degree - 1;
128 auto at = text("at=", location, " node=", node.notation());
129 if (node.count() < minimum || node.count() > maximum)
130 return text("COUNT ", at, " keys=", node.count(), " allowed=", minimum, "..", maximum);
131 for (int index = 0; index + 1 < node.count(); ++index) {
132 if (node.entries[index].key >= node.entries[index + 1].key)
133 return text("ORDER ", at, ' ', node.entries[index].key, ">=", node.entries[index + 1].key);
134 }
135 for (const auto& entry : node.entries) {
136 if ((low && entry.key <= *low) || (high && entry.key >= *high))
137 return text("RANGE ", at, " key=", entry.key, " low=", low ? std::to_string(*low) : "-inf",
138 " high=", high ? std::to_string(*high) : "+inf");
139 }
140 if (node.leaf()) {
141 if (referenceDepth < 0) referenceDepth = depth;
142 else if (depth != referenceDepth) return text("DEPTH ", at, " depth=", depth, " expected=", referenceDepth);
143 }
144 for (int index = 0; index < static_cast<int>(node.children.size()); ++index) {
145 auto problem = visit(node.children[index], text(location, '.', index), depth + 1,
146 index == 0 ? low : std::optional<int>(node.entries[index - 1].key),
147 index == node.count() ? high : std::optional<int>(node.entries[index].key));
148 if (problem) return problem;
149 }
150 return std::nullopt;
151 };
152 return visit(root, "root", 0, std::nullopt, std::nullopt);
153 }
154 static Tree load(int degree, const std::vector<std::string>& tokens) {
155 Tree tree(degree);
156 tree.pages.clear();
157 std::size_t position = 0;
158 int nextPage = 0;
159 auto unexpected = [&] { fail(text("SYNTAX ", position + 1)); };
160 auto readEntry = [&]() -> Entry {
161 if (position == tokens.size() || tokens[position] == "(" || tokens[position] == ")") unexpected();
162 const auto& token = tokens[position++];
163 auto equals = token.find('=');
164 return {std::stoi(token.substr(0, equals)), token.substr(equals + 1)};
165 };
166 std::function<int()> parse = [&]() {
167 if (position == tokens.size() || tokens[position] != "(") unexpected();
168 ++position;
169 int page = nextPage++;
170 Node node;
171 if (position < tokens.size() && tokens[position] == "(") {
172 node.children.push_back(parse());
173 while (position == tokens.size() || tokens[position] != ")") {
174 node.entries.push_back(readEntry()); node.children.push_back(parse());
175 }
176 } else {
177 while (position == tokens.size() || tokens[position] != ")") node.entries.push_back(readEntry());
178 }
179 ++position;
180 tree.pages.emplace(page, std::move(node));
181 return page;
182 };
183 tree.root = parse();
184 if (position != tokens.size()) unexpected();
185 return tree;
186 }
187 void split(int parentPage, int index, Trace* trace) {
188 auto& parent = pages.at(parentPage);
189 int childPage = parent.children[index];
190 auto& child = pages.at(childPage);
191 auto before = child.notation();
192 Entry middle = child.entries[degree - 1];
193 Node right;
194 right.entries.assign(child.entries.begin() + degree, child.entries.end());
195 if (!child.leaf()) right.children.assign(child.children.begin() + degree, child.children.end());
196 int newPage = allocate(std::move(right));
197 child.entries.resize(degree - 1);
198 if (!child.leaf()) child.children.resize(degree);
199 parent.entries.insert(parent.entries.begin() + index, middle);
200 parent.children.insert(parent.children.begin() + index + 1, newPage);
201 ++splits;
202 emit(trace, text("STEP split page=", childPage, " keys=", before, " up=", middle.key, " new=", newPage));
203 }
204 Upsert upsert(int key, const std::string& value, Trace* trace = nullptr) {
205 auto found = search(key);
206 if (found.found()) {
207 auto& entry = pages.at(found.page).entries[found.index];
208 auto old = entry.value;
209 entry.value = value;
210 return {false, old};
211 }
212 if (pages.at(root).count() == 2 * degree - 1) {
213 root = allocate(Node{{}, {root}});
214 emit(trace, text("STEP grow page=", root));
215 split(root, 0, trace);
216 }
217 int page = root;
218 while (!pages.at(page).leaf()) {
219 auto& node = pages.at(page);
220 int index = node.position(key);
221 if (pages.at(node.children[index]).count() == 2 * degree - 1) {
222 split(page, index, trace);
223 if (key > node.entries[index].key) ++index;
224 }
225 page = node.children[index];
226 }
227 auto& node = pages.at(page);
228 node.entries.insert(node.entries.begin() + node.position(key), Entry{key, value});
229 emit(trace, text("STEP place page=", page, " keys=", node.notation()));
230 return {true, {}};
231 }
232 int merge(int parentPage, int index, const std::string& caseName, Trace* trace) {
233 auto& parent = pages.at(parentPage);
234 int leftPage = parent.children[index], rightPage = parent.children[index + 1];
235 auto& left = pages.at(leftPage);
236 const auto& right = pages.at(rightPage);
237 left.entries.push_back(parent.entries[index]);
238 left.entries.insert(left.entries.end(), right.entries.begin(), right.entries.end());
239 left.children.insert(left.children.end(), right.children.begin(), right.children.end());
240 parent.entries.erase(parent.entries.begin() + index);
241 parent.children.erase(parent.children.begin() + index + 1);
242 pages.erase(rightPage); ++merges;
243 emit(trace, text("STEP ", caseName, " page=", parentPage, " merge=", leftPage, '+', rightPage, " keys=", left.notation()));
244 if (parentPage == root && parent.entries.empty()) {
245 root = leftPage; pages.erase(parentPage);
246 emit(trace, text("STEP shrink page=", root));
247 }
248 return leftPage;
249 }
250 void removeAt(int page, int key, std::vector<std::string>& cases, Trace* trace) {
251 auto& node = pages.at(page);
252 int index = node.position(key);
253 if (index < node.count() && node.entries[index].key == key) {
254 if (node.leaf()) {
255 cases.push_back("1"); node.entries.erase(node.entries.begin() + index);
256 emit(trace, text("STEP 1 page=", page, " keys=", node.notation()));
257 return;
258 }
259 int left = node.children[index], right = node.children[index + 1];
260 if (pages.at(left).count() >= degree) {
261 int last = left;
262 while (!pages.at(last).leaf()) last = pages.at(last).children.back();
263 Entry predecessor = pages.at(last).entries.back();
264 cases.push_back("2a"); emit(trace, text("STEP 2a page=", page, " pred=", predecessor.key));
265 removeAt(left, predecessor.key, cases, trace);
266 node.entries[index] = predecessor;
267 } else if (pages.at(right).count() >= degree) {
268 int first = right;
269 while (!pages.at(first).leaf()) first = pages.at(first).children[0];
270 Entry successor = pages.at(first).entries[0];
271 cases.push_back("2b"); emit(trace, text("STEP 2b page=", page, " succ=", successor.key));
272 removeAt(right, successor.key, cases, trace);
273 node.entries[index] = successor;
274 } else {
275 cases.push_back("2c");
276 int child = merge(page, index, "2c", trace);
277 removeAt(child, key, cases, trace);
278 }
279 return;
280 }
281 int childPage = node.children[index];
282 if (pages.at(childPage).count() == degree - 1) {
283 int leftPage = index > 0 ? node.children[index - 1] : -1;
284 int rightPage = index < node.count() ? node.children[index + 1] : -1;
285 auto& child = pages.at(childPage);
286 if (leftPage >= 0 && pages.at(leftPage).count() >= degree) {
287 auto& left = pages.at(leftPage);
288 cases.push_back("3a-left");
289 child.entries.insert(child.entries.begin(), node.entries[index - 1]);
290 node.entries[index - 1] = left.entries.back(); left.entries.pop_back();
291 if (!left.leaf()) {
292 child.children.insert(child.children.begin(), left.children.back()); left.children.pop_back();
293 }
294 ++borrows;
295 emit(trace, text("STEP 3a-left page=", page, " child=", childPage, " from=", leftPage, " keys=", child.notation()));
296 } else if (rightPage >= 0 && pages.at(rightPage).count() >= degree) {
297 auto& right = pages.at(rightPage);
298 cases.push_back("3a-right");
299 child.entries.push_back(node.entries[index]);
300 node.entries[index] = right.entries[0]; right.entries.erase(right.entries.begin());
301 if (!right.leaf()) {
302 child.children.push_back(right.children[0]); right.children.erase(right.children.begin());
303 }
304 ++borrows;
305 emit(trace, text("STEP 3a-right page=", page, " child=", childPage, " from=", rightPage, " keys=", child.notation()));
306 } else if (leftPage >= 0) {
307 cases.push_back("3b-left"); childPage = merge(page, index - 1, "3b-left", trace);
308 } else {
309 cases.push_back("3b-right"); childPage = merge(page, index, "3b-right", trace);
310 }
311 }
312 removeAt(childPage, key, cases, trace);
313 }
314 void remove(int key) {
315 std::vector<std::string> cases;
316 removeAt(root, key, cases, nullptr);
317 }
318 Scan scan(int low, int high) const {
319 Scan result;
320 std::function<void(int)> visit = [&](int page) {
321 ++result.visited;
322 const auto& node = pages.at(page);
323 for (int index = 0; index <= node.count(); ++index) {
324 if (!node.leaf() && (index == 0 || node.entries[index - 1].key < high)
325 && (index == node.count() || node.entries[index].key > low)) visit(node.children[index]);
326 if (index < node.count() && low <= node.entries[index].key && node.entries[index].key <= high)
327 result.entries.push_back(node.entries[index]);
328 }
329 };
330 visit(root);
331 return result;
332 }
333 std::string dump() const {
334 std::vector<std::string> tokens;
335 std::function<void(int)> visit = [&](int page) {
336 const auto& node = pages.at(page);
337 tokens.push_back("(");
338 if (!node.leaf()) visit(node.children[0]);
339 for (int index = 0; index < node.count(); ++index) {
340 tokens.push_back(node.entries[index].display());
341 if (!node.leaf()) visit(node.children[index + 1]);
342 }
343 tokens.push_back(")");
344 };
345 visit(root);
346 return join(tokens, " ");
347 }
348 void compact() {
349 auto source = entries();
350 std::vector<int> children;
351 std::vector<Node> built;
352 int oldRoot;
353 while (true) {
354 int groups = (static_cast<int>(source.size()) + 2 * degree) / (2 * degree);
355 int keys = static_cast<int>(source.size()) - groups + 1, size = keys / groups, extra = keys % groups;
356 std::vector<int> nodes;
357 std::vector<Entry> separators;
358 int entryIndex = 0, childIndex = 0;
359 for (int group = 0; group < groups; ++group) {
360 int count = size + (group < extra ? 1 : 0);
361 Node node;
362 node.entries.assign(source.begin() + entryIndex, source.begin() + entryIndex + count);
363 if (!children.empty()) node.children.assign(children.begin() + childIndex, children.begin() + childIndex + count + 1);
364 nodes.push_back(static_cast<int>(built.size())); built.push_back(std::move(node));
365 entryIndex += count; childIndex += count + 1;
366 if (group + 1 < groups) separators.push_back(source[entryIndex++]);
367 }
368 if (groups == 1) { oldRoot = nodes[0]; break; }
369 source = std::move(separators); children = std::move(nodes);
370 }
371 Snapshot replacement;
372 int nextPage = 0;
373 std::function<int(int)> renumber = [&](int oldPage) {
374 int page = nextPage++;
375 Node node = built[oldPage];
376 for (int& child : node.children) child = renumber(child);
377 replacement.emplace(page, std::move(node));
378 return page;
379 };
380 root = renumber(oldRoot); pages = std::move(replacement);
381 splits = merges = borrows = 0;
382 }
383};

385enum class Verb {
386 Create, Load, Insert, BulkLoad, Delete, DelRange, Get, Range, Count, Min, Max,
387 Floor, Ceil, Rank, Select, Seek, Next, Prev, Print, Pages, Dump, Height, Stats,
388 Verify, Rebuild, Compact, Save, Restore, Trace
389};
390
391const std::map<std::string, Verb> verbs{
392 {"CREATE", Verb::Create}, {"LOAD", Verb::Load}, {"INSERT", Verb::Insert}, {"BULKLOAD", Verb::BulkLoad},
393 {"DELETE", Verb::Delete}, {"DELRANGE", Verb::DelRange}, {"GET", Verb::Get}, {"RANGE", Verb::Range},
394 {"COUNT", Verb::Count}, {"MIN", Verb::Min}, {"MAX", Verb::Max}, {"FLOOR", Verb::Floor}, {"CEIL", Verb::Ceil},
395 {"RANK", Verb::Rank}, {"SELECT", Verb::Select}, {"SEEK", Verb::Seek}, {"NEXT", Verb::Next}, {"PREV", Verb::Prev},
396 {"PRINT", Verb::Print}, {"PAGES", Verb::Pages}, {"DUMP", Verb::Dump}, {"HEIGHT", Verb::Height}, {"STATS", Verb::Stats},
397 {"VERIFY", Verb::Verify}, {"REBUILD", Verb::Rebuild}, {"COMPACT", Verb::Compact}, {"SAVE", Verb::Save},
398 {"RESTORE", Verb::Restore}, {"TRACE", Verb::Trace}
399};

401bool requiresValidTree(Verb verb) {
402 switch (verb) {
403 case Verb::Insert: case Verb::BulkLoad: case Verb::Delete: case Verb::DelRange: case Verb::Range: case Verb::Count:
404 case Verb::Min: case Verb::Max: case Verb::Floor: case Verb::Ceil: case Verb::Rank: case Verb::Select:
405 case Verb::Seek: case Verb::Next: case Verb::Prev: case Verb::Height: case Verb::Compact:
406 return true;
407 default: return false;
408 }
409}

411enum class CursorKind { Unset, Key, Begin, End };
412struct Cursor { CursorKind kind = CursorKind::Unset; int key = 0; };

414class Workbench {
415 std::optional<Tree> index, saved;
416 Cursor cursor;
417 bool trace = false;
418 std::string summary(const std::string& prefix) const {
419 return text(prefix, " nodes=", index->pages.size(), " keys=", index->entries().size(),
420 index->violation() ? " DAMAGED" : " OK");
421 }
422public:
423 std::vector<std::string> command(const std::string& line) {
424 std::istringstream input(line);
425 std::string name, word;
426 input >> name;
427 std::vector<std::string> args;
428 while (input >> word) args.push_back(word);
429 Verb verb = verbs.at(name);
430 if (verb == Verb::Trace) {
431 trace = args[0] == "ON";
432 return {text("TRACE ", args[0])};
433 }
434 int first = args.empty() ? 0 : std::stoi(args[0]);
435 bool range = verb == Verb::DelRange || verb == Verb::Range || verb == Verb::Count;
436 int second = range ? std::stoi(args[1]) : 0;
437 if (verb == Verb::Create || verb == Verb::Load) {
438 if (first < 2 || first > 5) fail(text("BAD DEGREE ", first));
439 Tree replacement = verb == Verb::Create ? Tree(first)
440 : Tree::load(first, std::vector<std::string>(args.begin() + 1, args.end()));
441 index = std::move(replacement); cursor = {};
442 return {verb == Verb::Create ? text("CREATE t=", first) : summary(text("LOAD t=", first))};
443 }
444 if (!index) fail("NO INDEX");
445 Tree& tree = *index;
446 if (requiresValidTree(verb) && tree.violation()) fail("DAMAGED");
447 auto entries = tree.entries();
448 if (range && first > second) fail(text("BAD RANGE ", first, ' ', second));
449 if (verb == Verb::Select && (first < 1 || first > static_cast<int>(entries.size()))) fail(text("BAD POSITION ", first));
450 if ((verb == Verb::Next || verb == Verb::Prev) && cursor.kind == CursorKind::Unset) fail("NO CURSOR");
451 if (verb == Verb::Restore && !saved) fail("NO SAVE");
452 std::vector<std::string> output;
453 switch (verb) {
454 case Verb::Insert: {
455 auto before = tree.pages;
456 int splits = tree.splits;
457 auto result = tree.upsert(first, args[1], trace ? &output : nullptr);
458 int writes = tree.changes(before).writes;
459 output.push_back(result.inserted
460 ? text("INSERT ", first, " NEW splits=", tree.splits - splits, " height=", tree.height(), " writes=", writes)
461 : text("INSERT ", first, " UPDATED old=", result.old, " writes=", writes));
462 break;
463 }
464 case Verb::BulkLoad: {
465 auto before = tree.pages;
466 int splits = tree.splits, inserted = 0, updated = 0;
467 for (std::size_t position = 0; position < args.size(); position += 2) {
468 if (tree.upsert(std::stoi(args[position]), args[position + 1]).inserted) ++inserted;
469 else ++updated;
470 }
471 output.push_back(text("BULKLOAD pairs=", args.size() / 2, " new=", inserted, " updated=", updated,
472 " splits=", tree.splits - splits, " height=", tree.height(), " writes=", tree.changes(before).writes));
473 break;
474 }
475 case Verb::Delete: {
476 auto found = tree.search(first);
477 if (!found.found()) { output.push_back(text("DELETE ", first, " NOT FOUND")); break; }
478 auto value = tree.pages.at(found.page).entries[found.index].value;
479 auto before = tree.pages;
480 std::vector<std::string> cases;
481 tree.removeAt(tree.root, first, cases, trace ? &output : nullptr);
482 auto changes = tree.changes(before);
483 output.push_back(text("DELETE ", first, " REMOVED value=", value, " cases=", join(cases, ","),
484 " height=", tree.height(), " writes=", changes.writes, " frees=", changes.frees));
485 break;
486 }
487 case Verb::DelRange: {
488 auto before = tree.pages;
489 int merges = tree.merges, borrows = tree.borrows, removed = 0;
490 for (const auto& entry : entries) if (first <= entry.key && entry.key <= second) { tree.remove(entry.key); ++removed; }
491 auto changes = tree.changes(before);
492 output.push_back(text("DELRANGE ", first, ' ', second, " removed=", removed, " merges=", tree.merges - merges,
493 " borrows=", tree.borrows - borrows, " height=", tree.height(), " writes=", changes.writes, " frees=", changes.frees));
494 break;
495 }
496 case Verb::Get: {
497 auto found = tree.search(first);
498 auto result = found.found() ? text("FOUND value=", tree.pages.at(found.page).entries[found.index].value) : "NOT FOUND";
499 output.push_back(text("GET ", first, ' ', result, " path=", join(found.path, " -> ")));
500 break;
501 }
502 case Verb::Range: case Verb::Count: {
503 auto result = tree.scan(first, second);
504 auto line = text(name, ' ', first, ' ', second, " count=", result.entries.size(), " visited=", result.visited);
505 if (verb == Verb::Range) for (const auto& entry : result.entries) line += ' ' + entry.display();
506 output.push_back(line);
507 break;
508 }
509 case Verb::Min: case Verb::Max:
510 output.push_back(text(name, ' ', entries.empty() ? "EMPTY" : (verb == Verb::Min ? entries.front() : entries.back()).display()));
511 break;
512 case Verb::Floor: case Verb::Ceil: {
513 std::optional<Entry> selected;
514 for (const auto& entry : entries) {
515 if (verb == Verb::Floor && entry.key <= first) selected = entry;
516 if (verb == Verb::Ceil && entry.key >= first) { selected = entry; break; }
517 }
518 output.push_back(text(name, ' ', first, ' ', selected ? selected->display() : "NONE"));
519 break;
520 }
521 case Verb::Rank:
522 output.push_back(text("RANK ", first, ' ', std::count_if(entries.begin(), entries.end(), [&](const Entry& entry) { return entry.key < first; })));
523 break;
524 case Verb::Select:
525 output.push_back(text("SELECT ", first, ' ', entries[first - 1].display()));
526 break;
527 case Verb::Seek: {
528 auto selected = std::find_if(entries.begin(), entries.end(), [&](const Entry& entry) { return entry.key >= first; });
529 cursor = selected == entries.end() ? Cursor{CursorKind::End} : Cursor{CursorKind::Key, selected->key};
530 output.push_back(text("SEEK ", first, ' ', selected == entries.end() ? "END" : selected->display()));
531 break;
532 }
533 case Verb::Next: case Verb::Prev: {
534 std::optional<Entry> selected;
535 if (verb == Verb::Next) {
536 for (const auto& entry : entries) {
537 if (cursor.kind == CursorKind::Begin || (cursor.kind == CursorKind::Key && entry.key > cursor.key)) { selected = entry; break; }
538 }
539 cursor = selected ? Cursor{CursorKind::Key, selected->key} : Cursor{CursorKind::End};
540 } else {
541 for (auto it = entries.rbegin(); it != entries.rend(); ++it) {
542 if (cursor.kind == CursorKind::End || (cursor.kind == CursorKind::Key && it->key < cursor.key)) { selected = *it; break; }
543 }
544 cursor = selected ? Cursor{CursorKind::Key, selected->key} : Cursor{CursorKind::Begin};
545 }
546 output.push_back(text(name, ' ', selected ? selected->display() : (verb == Verb::Next ? "END" : "BEGIN")));
547 break;
548 }
549 case Verb::Print: {
550 std::vector<int> level{tree.root};
551 int depth = 0;
552 while (!level.empty()) {
553 auto line = text("LEVEL ", depth++);
554 std::vector<int> next;
555 for (int page : level) {
556 const auto& node = tree.pages.at(page);
557 line += ' ' + node.notation();
558 next.insert(next.end(), node.children.begin(), node.children.end());
559 }
560 output.push_back(line); level = std::move(next);
561 }
562 break;
563 }
564 case Verb::Pages:
565 for (const auto& [page, node] : tree.pages) {
566 std::vector<std::string> children;
567 for (int child : node.children) children.push_back(std::to_string(child));
568 output.push_back(text("PAGE ", page, ' ', node.notation(), " kids=", node.leaf() ? "-" : join(children, ",")));
569 }
570 break;
571 case Verb::Dump: output.push_back(text("DUMP ", tree.degree, ' ', tree.dump())); break;
572 case Verb::Height: output.push_back(text("HEIGHT ", tree.height())); break;
573 case Verb::Stats: {
574 int leaves = 0;
575 for (const auto& [page, node] : tree.pages) if (node.leaf()) ++leaves;
576 auto fill = 100 * entries.size() / ((2 * tree.degree - 1) * tree.pages.size());
577 output.push_back(text("STATS t=", tree.degree, " nodes=", tree.pages.size(), " leaves=", leaves, " keys=", entries.size(),
578 " fill=", fill, "% splits=", tree.splits, " merges=", tree.merges, " borrows=", tree.borrows));
579 break;
580 }
581 case Verb::Verify: {
582 auto problem = tree.violation();
583 output.push_back(problem ? "VERIFY FAIL " + *problem : "VERIFY OK");
584 break;
585 }
586 case Verb::Rebuild: {
587 Tree replacement(tree.degree);
588 for (const auto& entry : entries) replacement.upsert(entry.key, entry.value);
589 index = std::move(replacement);
590 output.push_back(text("REBUILD entries=", entries.size(), " keys=", index->entries().size(), " nodes=", index->pages.size(),
591 " splits=", index->splits, " height=", index->height()));
592 break;
593 }
594 case Verb::Compact: {
595 auto nodes = tree.pages.size();
596 tree.compact();
597 output.push_back(text("COMPACT nodes=", nodes, "->", tree.pages.size(), " height=", tree.height()));
598 break;
599 }
600 case Verb::Save: saved = tree; output.push_back(summary("SAVE")); break;
601 case Verb::Restore: index = *saved; output.push_back(summary("RESTORE")); break;
602 case Verb::Create: case Verb::Load: case Verb::Trace: break;
603 }
604 return output;
605 }
606};
