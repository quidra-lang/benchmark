15class Node
16 attr_reader :entries, :children
17
18 def initialize(entries = [], children = [])
19 @entries, @children = entries, children
20 end
21
22 def initialize_copy(other)
23 super
24 @entries = other.entries.dup
25 @children = other.children.dup
26 end
27
28 def leaf? = @children.empty?
29 def position(key) = @entries.index { |entry| key <= entry.key } || @entries.length
30 def to_s = "[#{@entries.map(&:key).join(' ')}]"
31 def same_content?(other) = @entries == other.entries && @children == other.children
32end

47class Tree
48 attr_reader :degree, :pages
49 attr_accessor :root, :splits, :merges, :borrows
50
51 def initialize(degree)
52 @degree = degree
53 @root = @splits = @merges = @borrows = 0
54 @pages = { 0 => Node.new }
55 end
56
57 def initialize_copy(other)
58 super
59 @pages = other.pages.transform_values(&:dup)
60 end
61
62 def allocate(node)
63 page = 0
64 page += 1 while @pages.key?(page)
65 @pages[page] = node
66 page
67 end
68
69 def entries
70 result = []
71 collect(@root, result)
72 result
73 end
74
75 def collect(page, result)
76 node = @pages.fetch(page)
77 (0..node.entries.length).each do |j|
78 collect(node.children[j], result) unless node.leaf?
79 result << node.entries[j] if j < node.entries.length
80 end
81 end
82
83 def search(key)
84 page, path = @root, []
85 loop do
86 node = @pages.fetch(page)
87 path << node.to_s
88 position = node.position(key)
89 return Search.new(page, position, path) if node.entries[position]&.key == key
90 return Search.new(nil, position, path) if node.leaf?
91 page = node.children[position]
92 end
93 end
94
95 def check_node(page, location, depth, low, high, reference)
96 node = @pages.fetch(page)
97 count = node.entries.length
98 minimum = page == @root ? (node.leaf? ? 0 : 1) : @degree - 1
99 maximum = 2 * @degree - 1
100 details = "at=#{location} node=#{node}"
101 return "COUNT #{details} keys=#{count} allowed=#{minimum}..#{maximum}" unless (minimum..maximum).cover?(count)
102 node.entries.each_cons(2) do |left, right|
103 return "ORDER #{details} #{left.key}>=#{right.key}" if left.key >= right.key
104 end
105 node.entries.each do |entry|
106 if (low && entry.key <= low) || (high && entry.key >= high)
107 return "RANGE #{details} key=#{entry.key} low=#{low || '-inf'} high=#{high || '+inf'}"
108 end
109 end
110 if node.leaf?
111 if reference[0].nil?
112 reference[0] = depth
113 elsif depth != reference[0]
114 return "DEPTH #{details} depth=#{depth} expected=#{reference[0]}"
115 end
116 end
117 node.children.each_with_index do |child, j|
118 lower = j.zero? ? low : node.entries[j - 1].key
119 upper = j == count ? high : node.entries[j].key
120 problem = check_node(child, "#{location}.#{j}", depth + 1, lower, upper, reference)
121 return problem if problem
122 end
123 nil
124 end
125
126 def violation = check_node(@root, 'root', 0, nil, nil, [nil])
127 def status = violation ? 'DAMAGED' : 'OK'
128
129 def height
130 page, depth = @root, 0
131 until @pages.fetch(page).leaf?
132 page = @pages.fetch(page).children.first
133 depth += 1
134 end
135 depth
136 end
137
138 def self.load(degree, tokens) = Parser.new(degree, tokens).load
139
140 def changes(before)
141 writes = @pages.count { |page, node| !before.pages.key?(page) || !node.same_content?(before.pages[page]) }
142 frees = before.pages.keys.count { |page| !@pages.key?(page) }
143 Changes.new(writes, frees)
144 end
145
146 def split_child(parent_page, position, trace)
147 parent = @pages.fetch(parent_page)
148 child_page = parent.children[position]
149 child = @pages.fetch(child_page)
150 old = child.to_s
151 middle = child.entries[@degree - 1]
152 right = Node.new(child.entries.drop(@degree), child.children.drop(@degree))
153 right_page = allocate(right)
154 child.entries.slice!(@degree - 1, child.entries.length)
155 child.children.slice!(@degree, child.children.length) unless child.leaf?
156 parent.entries.insert(position, middle)
157 parent.children.insert(position + 1, right_page)
158 @splits += 1
159 trace.emit('STEP split page=%d keys=%s up=%d new=%d', child_page, old, middle.key, right_page)
160 end
161
162 def upsert(entry, trace)
163 found = search(entry.key)
164 unless found.page.nil?
165 node = @pages.fetch(found.page)
166 old = node.entries[found.position].value
167 node.entries[found.position] = entry
168 return old
169 end
170 if @pages.fetch(@root).entries.length == 2 * @degree - 1
171 @root = allocate(Node.new([], [@root]))
172 trace.emit('STEP grow page=%d', @root)
173 split_child(@root, 0, trace)
174 end
175 page = @root
176 until @pages.fetch(page).leaf?
177 node = @pages.fetch(page)
178 position = node.position(entry.key)
179 if @pages.fetch(node.children[position]).entries.length == 2 * @degree - 1
180 split_child(page, position, trace)
181 position += 1 if entry.key > node.entries[position].key
182 end
183 page = node.children[position]
184 end
185 node = @pages.fetch(page)
186 node.entries.insert(node.position(entry.key), entry)
187 trace.emit('STEP place page=%d keys=%s', page, node)
188 nil
189 end
190
191 def extreme(page, last)
192 loop do
193 node = @pages.fetch(page)
194 return last ? node.entries.last : node.entries.first if node.leaf?
195 page = last ? node.children.last : node.children.first
196 end
197 end
198
199 def merge(parent_page, position, case_name, trace)
200 parent = @pages.fetch(parent_page)
201 left_page, right_page = parent.children[position, 2]
202 left = @pages.fetch(left_page)
203 right = @pages.delete(right_page)
204 left.entries << parent.entries.delete_at(position)
205 left.entries.concat(right.entries)
206 left.children.concat(right.children)
207 parent.children.delete_at(position + 1)
208 @merges += 1
209 trace.emit('STEP %s page=%d merge=%d+%d keys=%s', case_name, parent_page, left_page, right_page, left)
210 if parent_page == @root && parent.entries.empty?
211 @root = left_page
212 @pages.delete(parent_page)
213 trace.emit('STEP shrink page=%d', @root)
214 end
215 left_page
216 end
217
218 def remove_at(page, key, trace)
219 node = @pages.fetch(page)
220 position = node.position(key)
221 if node.entries[position]&.key == key
222 if node.leaf?
223 trace.cases << '1'
224 node.entries.delete_at(position)
225 trace.emit('STEP 1 page=%d keys=%s', page, node)
226 return
227 end
228 left, right = node.children[position, 2]
229 if @pages.fetch(left).entries.length >= @degree
230 predecessor = extreme(left, true)
231 trace.cases << '2a'
232 trace.emit('STEP 2a page=%d pred=%d', page, predecessor.key)
233 remove_at(left, predecessor.key, trace)
234 node.entries[position] = predecessor
235 elsif @pages.fetch(right).entries.length >= @degree
236 successor = extreme(right, false)
237 trace.cases << '2b'
238 trace.emit('STEP 2b page=%d succ=%d', page, successor.key)
239 remove_at(right, successor.key, trace)
240 node.entries[position] = successor
241 else
242 trace.cases << '2c'
243 child = merge(page, position, '2c', trace)
244 remove_at(child, key, trace)
245 end
246 return
247 end
248 child_page = node.children[position]
249 child = @pages.fetch(child_page)
250 if child.entries.length == @degree - 1
251 left = position.positive? ? node.children[position - 1] : nil
252 right = node.children[position + 1]
253 if left && @pages.fetch(left).entries.length >= @degree
254 sibling = @pages.fetch(left)
255 trace.cases << '3a-left'
256 child.entries.unshift(node.entries[position - 1])
257 node.entries[position - 1] = sibling.entries.pop
258 child.children.unshift(sibling.children.pop) unless sibling.leaf?
259 @borrows += 1
260 trace.emit('STEP 3a-left page=%d child=%d from=%d keys=%s', page, child_page, left, child)
261 elsif right && @pages.fetch(right).entries.length >= @degree
262 sibling = @pages.fetch(right)
263 trace.cases << '3a-right'
264 child.entries << node.entries[position]
265 node.entries[position] = sibling.entries.shift
266 child.children << sibling.children.shift unless sibling.leaf?
267 @borrows += 1
268 trace.emit('STEP 3a-right page=%d child=%d from=%d keys=%s', page, child_page, right, child)
269 elsif left
270 trace.cases << '3b-left'
271 child_page = merge(page, position - 1, '3b-left', trace)
272 else
273 trace.cases << '3b-right'
274 child_page = merge(page, position, '3b-right', trace)
275 end
276 end
277 remove_at(child_page, key, trace)
278 end
279
280 def remove(key, trace)
281 trace.cases.clear
282 remove_at(@root, key, trace)
283 end

285 def scan_node(page, low, high, entries)
286 node = @pages.fetch(page)
287 visited = 1
288 count = node.entries.length
289 (0..count).each do |j|
290 if !node.leaf? && (j.zero? || node.entries[j - 1].key < high) && (j == count || node.entries[j].key > low)
291 visited += scan_node(node.children[j], low, high, entries)
292 end
293 entries << node.entries[j] if j < count && node.entries[j].key.between?(low, high)
294 end
295 visited
296 end
297
298 def scan(low, high)
299 entries = []
300 visited = scan_node(@root, low, high, entries)
301 Scan.new(entries, visited)
302 end

304 def dump_node(page, tokens)
305 node = @pages.fetch(page)
306 tokens << '('
307 (0..node.entries.length).each do |j|
308 dump_node(node.children[j], tokens) unless node.leaf?
309 tokens << node.entries[j].to_s if j < node.entries.length
310 end
311 tokens << ')'
312 end
313
314 def dump
315 tokens = []
316 dump_node(@root, tokens)
317 tokens.join(' ')
318 end

320 def copy_group(old, built)
321 source = built.fetch(old)
322 node = Node.new(source.entries.dup)
323 page = allocate(node)
324 source.children.each { |child| node.children << copy_group(child, built) }
325 page
326 end

328 def compact
329 source, children, built = entries, [], []
330 old_root = nil
331 loop do
332 groups = (source.length + 2 * @degree).div(2 * @degree)
333 keys = source.length - (groups - 1)
334 base, extra = keys.divmod(groups)
335 position = child_position = 0
336 separators, level = [], []
337 groups.times do |j|
338 count = base + (j < extra ? 1 : 0)
339 node = Node.new(source[position, count], children.empty? ? [] : children[child_position, count + 1])
340 position += count
341 child_position += count + 1
342 level << built.length
343 built << node
344 if j + 1 < groups
345 separators << source[position]
346 position += 1
347 end
348 end
349 if groups == 1
350 old_root = level.first
351 break
352 end
353 source, children = separators, level
354 end
355 @pages.clear
356 @root = copy_group(old_root, built)
357 @splits = @merges = @borrows = 0
358 end
359end

361class Parser
362 def initialize(degree, tokens)
363 @tree = Tree.new(degree)
364 @tree.pages.clear
365 @tokens, @position = tokens, 0
366 end
367
368 def entry
369 token = @tokens[@position]
370 split = token&.index('=')
371 raise CommandError, "SYNTAX #{@position + 1}" unless split
372 @position += 1
373 Entry.new(Integer(token[0...split], 10), token[(split + 1)..])
374 end
375
376 def read_node
377 raise CommandError, "SYNTAX #{@position + 1}" unless @tokens[@position] == '('
378 @position += 1
379 node = Node.new
380 page = @tree.allocate(node)
381 internal = @tokens[@position] == '('
382 node.children << read_node if internal
383 until @tokens[@position] == ')'
384 node.entries << entry
385 node.children << read_node if internal
386 end
387 @position += 1
388 page
389 end
390
391 def load
392 @tree.root = read_node
393 raise CommandError, "SYNTAX #{@position + 1}" unless @position == @tokens.length
394 @tree
395 end
396end

398class Workbench
399 NEEDS_VALID = %w[INSERT BULKLOAD DELETE DELRANGE RANGE COUNT MIN MAX FLOOR CEIL RANK SELECT SEEK NEXT PREV HEIGHT COMPACT].freeze
400
401 def initialize
402 @index = @saved = nil
403 @cursor = :unset
404 @trace = false
405 end
406
407 def find_entry(entries, last: false, &predicate)
408 (last ? entries.reverse_each : entries.each).find(&predicate)
409 end
410
411 def process(line)
412 words = line.split(' ')
413 command = words[0]
414 if command == 'TRACE'
415 @trace = words[1] == 'ON'
416 return ["TRACE #{words[1]}"]
417 end
418 first = words[1] ? Integer(words[1], 10) : 0
419 ranged = %w[DELRANGE RANGE COUNT].include?(command)
420 second = ranged ? Integer(words[2], 10) : 0
421 if command == 'CREATE' || command == 'LOAD'
422 raise CommandError, "BAD DEGREE #{first}" unless (2..5).cover?(first)
423 tree = command == 'CREATE' ? Tree.new(first) : Tree.load(first, words.drop(2))
424 result = command == 'CREATE' ? "CREATE t=#{first}" : "LOAD t=#{first} nodes=#{tree.pages.length} keys=#{tree.entries.length} #{tree.status}"
425 @index, @cursor = tree, :unset
426 return [result]
427 end
428 tree = @index
429 raise CommandError, 'NO INDEX' unless tree
430 raise CommandError, 'DAMAGED' if NEEDS_VALID.include?(command) && tree.violation
431 entries = tree.entries
432 raise CommandError, "BAD RANGE #{first} #{second}" if ranged && first > second
433 raise CommandError, "BAD POSITION #{first}" if command == 'SELECT' && !first.between?(1, entries.length)
434 raise CommandError, 'NO CURSOR' if %w[NEXT PREV].include?(command) && @cursor == :unset
435 raise CommandError, 'NO SAVE' if command == 'RESTORE' && @saved.nil?
436 output = []
437 steps = Trace.new(@trace && %w[INSERT DELETE].include?(command))
438 case command
439 when 'INSERT'
440 before, splits = tree.dup, tree.splits
441 old = tree.upsert(Entry.new(first, words[2]), steps)
442 writes = tree.changes(before).writes
443 output.concat(steps.lines)
444 output << if old
445 "INSERT #{first} UPDATED old=#{old} writes=#{writes}"
446 else
447 "INSERT #{first} NEW splits=#{tree.splits - splits} height=#{tree.height} writes=#{writes}"
448 end
449 when 'BULKLOAD'
450 before, splits = tree.dup, tree.splits
451 added = updated = 0
452 words.drop(1).each_slice(2) do |key, value|
453 if tree.upsert(Entry.new(Integer(key, 10), value), steps)
454 updated += 1
455 else
456 added += 1
457 end
458 end
459 output << "BULKLOAD pairs=#{(words.length - 1).div(2)} new=#{added} updated=#{updated} splits=#{tree.splits - splits} height=#{tree.height} writes=#{tree.changes(before).writes}"
460 when 'DELETE'
461 found = tree.search(first)
462 if found.page.nil?
463 output << "DELETE #{first} NOT FOUND"
464 else
465 value = tree.pages.fetch(found.page).entries[found.position].value
466 before = tree.dup
467 tree.remove(first, steps)
468 changes = tree.changes(before)
469 output.concat(steps.lines)
470 output << "DELETE #{first} REMOVED value=#{value} cases=#{steps.cases.join(',')} height=#{tree.height} writes=#{changes.writes} frees=#{changes.frees}"
471 end
472 when 'DELRANGE'
473 before, merges, borrows = tree.dup, tree.merges, tree.borrows
474 removed = 0
475 entries.each do |entry|
476 next unless entry.key.between?(first, second)
477 tree.remove(entry.key, steps)
478 removed += 1
479 end
480 changes = tree.changes(before)
481 output << "DELRANGE #{first} #{second} removed=#{removed} merges=#{tree.merges - merges} borrows=#{tree.borrows - borrows} height=#{tree.height} writes=#{changes.writes} frees=#{changes.frees}"
482 when 'GET'
483 found = tree.search(first)
484 result = found.page.nil? ? 'NOT FOUND' : "FOUND value=#{tree.pages.fetch(found.page).entries[found.position].value}"
485 output << "GET #{first} #{result} path=#{found.path.join(' -> ')}"
486 when 'RANGE', 'COUNT'
487 result = tree.scan(first, second)
488 suffix = command == 'RANGE' ? result.entries.map { |entry| " #{entry}" }.join : ''
489 output << "#{command} #{first} #{second} count=#{result.entries.length} visited=#{result.visited}#{suffix}"
490 when 'MIN', 'MAX'
491 entry = command == 'MIN' ? entries.first : entries.last
492 output << "#{command} #{entry || 'EMPTY'}"
493 when 'FLOOR', 'CEIL'
494 floor = command == 'FLOOR'
495 entry = find_entry(entries, last: floor) { |value| floor ? value.key <= first : value.key >= first }
496 output << "#{command} #{first} #{entry || 'NONE'}"
497 when 'RANK'
498 output << "RANK #{first} #{entries.count { |entry| entry.key < first }}"
499 when 'SELECT'
500 output << "SELECT #{first} #{entries[first - 1]}"
501 when 'SEEK'
502 entry = find_entry(entries) { |value| value.key >= first }
503 @cursor = entry&.key || :end
504 output << "SEEK #{first} #{entry || 'END'}"
505 when 'NEXT', 'PREV'
506 forward = command == 'NEXT'
507 entry = case @cursor
508 when Integer
509 find_entry(entries, last: !forward) { |value| forward ? value.key > @cursor : value.key < @cursor }
510 when :begin then forward ? entries.first : nil
511 when :end then forward ? nil : entries.last
512 end
513 boundary = forward ? :end : :begin
514 @cursor = entry&.key || boundary
515 output << "#{command} #{entry || boundary.to_s.upcase}"
516 when 'PRINT'
517 level, depth = [tree.root], 0
518 until level.empty?
519 output << "LEVEL #{depth} #{level.map { |page| tree.pages.fetch(page).to_s }.join(' ')}"
520 level = level.flat_map { |page| tree.pages.fetch(page).children }
521 depth += 1
522 end
523 when 'PAGES'
524 tree.pages.keys.sort.each do |page|
525 node = tree.pages.fetch(page)
526 kids = node.leaf? ? '-' : node.children.join(',')
527 output << "PAGE #{page} #{node} kids=#{kids}"
528 end
529 when 'DUMP'
530 output << "DUMP #{tree.degree} #{tree.dump}"
531 when 'HEIGHT'
532 output << "HEIGHT #{tree.height}"
533 when 'STATS'
534 nodes = tree.pages.length
535 leaves = tree.pages.values.count(&:leaf?)
536 fill = (100 * entries.length).div((2 * tree.degree - 1) * nodes)
537 output << "STATS t=#{tree.degree} nodes=#{nodes} leaves=#{leaves} keys=#{entries.length} fill=#{fill}% splits=#{tree.splits} merges=#{tree.merges} borrows=#{tree.borrows}"
538 when 'VERIFY'
539 problem = tree.violation
540 output << (problem ? "VERIFY FAIL #{problem}" : 'VERIFY OK')
541 when 'REBUILD'
542 tree = Tree.new(tree.degree)
543 entries.each { |entry| tree.upsert(entry, steps) }
544 output << "REBUILD entries=#{entries.length} keys=#{tree.entries.length} nodes=#{tree.pages.length} splits=#{tree.splits} height=#{tree.height}"
545 when 'COMPACT'
546 nodes = tree.pages.length
547 tree.compact
548 output << "COMPACT nodes=#{nodes}->#{tree.pages.length} height=#{tree.height}"
549 when 'SAVE'
550 @saved = tree.dup
551 output << "SAVE nodes=#{tree.pages.length} keys=#{entries.length} #{tree.status}"
552 when 'RESTORE'
553 tree = @saved.dup
554 output << "RESTORE nodes=#{tree.pages.length} keys=#{tree.entries.length} #{tree.status}"
555 end
556 @index = tree
557 output
558 end
559end
