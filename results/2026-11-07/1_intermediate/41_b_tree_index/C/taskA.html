9typedef struct { int key; char value[13]; } Entry;

61 Entry result = {.key = key};
62 strcpy(result.value, value);
63 return result;

66static void insert_entry(Node *node, int at, Entry value) {
67 if (node->count == node->capacity) {
68 node->capacity = node->capacity ? node->capacity * 2 : 8;
69 node->entries = resize(node->entries, (size_t)node->capacity * sizeof(Entry));
70 }
71 memmove(node->entries + at + 1, node->entries + at, (size_t)(node->count - at) * sizeof(Entry));
72 node->entries[at] = value;
73 ++node->count;
74}
75
76static Entry remove_entry(Node *node, int at) {
77 Entry result = node->entries[at];
78 --node->count;
79 memmove(node->entries + at, node->entries + at + 1, (size_t)(node->count - at) * sizeof(Entry));
80 return result;
81}

83static void insert_child(Node *node, int at, Node *child) {
84 if (node->child_count == node->child_capacity) {
85 node->child_capacity = node->child_capacity ? node->child_capacity * 2 : 8;
86 node->children = resize(node->children, (size_t)node->child_capacity * sizeof(Node *));
87 }
88 memmove(node->children + at + 1, node->children + at, (size_t)(node->child_count - at) * sizeof(Node *));
89 node->children[at] = child;
90 ++node->child_count;
91}

93static Node *remove_child(Node *node, int at) {
94 Node *result = node->children[at];
95 --node->child_count;
96 memmove(node->children + at, node->children + at + 1, (size_t)(node->child_count - at) * sizeof(Node *));
97 return result;
98}

101 int page = 0;
102 while (tree->pages[page]) ++page;
103 Node *node = allocate(sizeof(Node));
104 node->page = page;
105 tree->pages[page] = node;

109static void release_node(Tree *tree, Node *node) {

111 free(node->entries);
112 free(node->children);
113 free(node);

123static void release_tree(Tree *tree) {
124 if (!tree) return;

126 if (tree->pages[page]) release_node(tree, tree->pages[page]);

130static Tree *copy_tree(const Tree *source) {
131 Tree *result = allocate(sizeof(Tree));
132 result->degree = source->degree;
133 result->splits = source->splits;
134 result->merges = source->merges;
135 result->borrows = source->borrows;
136 for (int page = 0; page < MAX_PAGES; ++page) {
137 const Node *old = source->pages[page];
138 if (!old) continue;
139 Node *node = allocate(sizeof(Node));
140 node->page = page;
141 for (int i = 0; i < old->count; ++i) insert_entry(node, node->count, old->entries[i]);
142 result->pages[page] = node;
143 }
144 for (int page = 0; page < MAX_PAGES; ++page) {
145 const Node *old = source->pages[page];
146 if (!old) continue;
147 Node *node = result->pages[page];
148 for (int i = 0; i < old->child_count; ++i)
149 insert_child(node, node->child_count, result->pages[old->children[i]->page]);
150 }
151 result->root = result->pages[source->root->page];
152 return result;
153}

155static bool same_content(const Node *a, const Node *b) {
156 if (a->count != b->count || a->child_count != b->child_count) return false;
157 for (int i = 0; i < a->count; ++i)
158 if (a->entries[i].key != b->entries[i].key || strcmp(a->entries[i].value, b->entries[i].value)) return false;
159 for (int i = 0; i < a->child_count; ++i)
160 if (a->children[i]->page != b->children[i]->page) return false;
161 return true;
162}

164static Changes changes(const Tree *before, const Tree *after) {
165 Changes result = {0};
166 for (int page = 0; page < MAX_PAGES; ++page) {
167 const Node *old = before->pages[page], *now = after->pages[page];
168 if (now && (!old || !same_content(old, now))) ++result.writes;
169 if (old && !now) ++result.frees;
170 }
171 return result;
172}

174static Statistics statistics(const Tree *tree) {
175 Statistics result = {0};
176 for (int page = 0; page < MAX_PAGES; ++page) {
177 const Node *node = tree->pages[page];
178 if (!node) continue;
179 ++result.nodes;
180 result.keys += node->count;
181 if (!node->child_count) ++result.leaves;
182 }
183 return result;

199static bool inspect_node(const Tree *tree, const Node *node, const char *location,
200 int depth, int low, int high, int *reference, Violation *result) {
201 int minimum = node == tree->root ? (node->child_count ? 1 : 0) : tree->degree - 1;
202 result->kind = VALID;
203 result->minimum = minimum;
204 if (node->count < minimum || node->count > 2 * tree->degree - 1) result->kind = BAD_COUNT;
205 for (int i = 0; result->kind == VALID && i + 1 < node->count; ++i) {
206 if (node->entries[i].key >= node->entries[i + 1].key) { result->kind = BAD_ORDER; result->index = i; }
207 }
208 for (int i = 0; result->kind == VALID && i < node->count; ++i) {
209 if (node->entries[i].key <= low || node->entries[i].key >= high) { result->kind = BAD_RANGE; result->index = i; }
210 }
211 if (result->kind == VALID && !node->child_count) {
212 if (*reference < 0) *reference = depth;
213 else if (depth != *reference) result->kind = BAD_DEPTH;
214 }
215 if (result->kind != VALID) {
216 result->node = node;
217 strcpy(result->location, location);
218 result->depth = depth; result->expected = *reference;
219 result->low = low; result->high = high;
220 return false;
221 }
222 for (int j = 0; j < node->child_count; ++j) {
223 char child_location[64];
224 snprintf(child_location, sizeof(child_location), "%s.%d", location, j);
225 int child_low = j ? node->entries[j - 1].key : low;
226 int child_high = j < node->count ? node->entries[j].key : high;
227 if (!inspect_node(tree, node->children[j], child_location, depth + 1, child_low, child_high, reference, result)) return false;
228 }
229 return true;
230}

232static Violation inspect(const Tree *tree) {
233 Violation result = {0};
234 int reference = -1;
235 inspect_node(tree, tree->root, "root", 0, INT_MIN, INT_MAX, &reference, &result);
236 return result;
237}

247static Search search(const Tree *tree, int key) {
248 Search result = {0};
249 Node *node = tree->root;
250 while (true) {
251 result.path[result.depth++] = node;
252 int at = position(node, key);
253 if (at < node->count && node->entries[at].key == key) {
254 result.node = node; result.position = at;
255 return result;
256 }
257 if (!node->child_count) return result;
258 node = node->children[at];
259 }
260}

262static void inorder(const Node *node, Entries *result) {
263 for (int i = 0; i <= node->count; ++i) {
264 if (node->child_count) inorder(node->children[i], result);
265 if (i < node->count) result->entries[result->count++] = node->entries[i];
266 }
267}

269static void scan(const Node *node, int low, int high, Entries *result, int *visited) {
270 ++*visited;
271 for (int i = 0; i <= node->count; ++i) {
272 if (node->child_count && (!i || node->entries[i - 1].key < high) &&
273 (i == node->count || node->entries[i].key > low)) scan(node->children[i], low, high, result, visited);
274 if (i < node->count && node->entries[i].key >= low && node->entries[i].key <= high)
275 result->entries[result->count++] = node->entries[i];
276 }
277}

279static void split_child(Tree *tree, Node *parent, int at, bool trace) {
280 int t = tree->degree;
281 Node *left = parent->children[at], *right = new_node(tree);
282 Entry middle = left->entries[t - 1];
283 if (trace) {
284 printf("STEP split page=%d keys=", left->page); notation(left);
285 printf(" up=%d new=%d\n", middle.key, right->page);
286 }
287 for (int i = t; i < left->count; ++i) insert_entry(right, right->count, left->entries[i]);
288 for (int i = t; i < left->child_count; ++i) insert_child(right, right->child_count, left->children[i]);
289 left->count = t - 1;
290 if (left->child_count) left->child_count = t;
291 insert_entry(parent, at, middle);
292 insert_child(parent, at + 1, right);
293 ++tree->splits;
294}

296static bool put_entry(Tree *tree, Entry value, bool trace) {
297 Search found = search(tree, value.key);
298 if (found.node) { found.node->entries[found.position] = value; return false; }
299 if (tree->root->count == 2 * tree->degree - 1) {
300 Node *old = tree->root, *root = new_node(tree);
301 insert_child(root, 0, old);
302 tree->root = root;
303 if (trace) printf("STEP grow page=%d\n", root->page);
304 split_child(tree, root, 0, trace);
305 }
306 Node *node = tree->root;
307 while (node->child_count) {
308 int at = position(node, value.key);
309 if (node->children[at]->count == 2 * tree->degree - 1) {
310 split_child(tree, node, at, trace);
311 if (value.key > node->entries[at].key) ++at;
312 }
313 node = node->children[at];
314 }
315 insert_entry(node, position(node, value.key), value);
316 if (trace) { printf("STEP place page=%d keys=", node->page); notation(node); putchar('\n'); }
317 return true;
318}

321 if (!operation) return;

323 strcpy(operation->cases + operation->length, name);
324 operation->length += strlen(name);

327static Node *merge_children(Tree *tree, Node *parent, int at, const char *name, Deletion *operation) {
328 Node *left = parent->children[at], *right = parent->children[at + 1];
329 int parent_page = parent->page, right_page = right->page;
330 bool shrink = parent == tree->root && parent->count == 1;
331 insert_entry(left, left->count, parent->entries[at]);
332 for (int i = 0; i < right->count; ++i) insert_entry(left, left->count, right->entries[i]);
333 for (int i = 0; i < right->child_count; ++i) insert_child(left, left->child_count, right->children[i]);
334 remove_entry(parent, at); remove_child(parent, at + 1);
335 release_node(tree, right);
336 ++tree->merges;
337 if (shrink) { tree->root = left; release_node(tree, parent); }
338 if (operation && operation->trace) {
339 printf("STEP %s page=%d merge=%d+%d keys=", name, parent_page, left->page, right_page);
340 notation(left); putchar('\n');
341 if (shrink) printf("STEP shrink page=%d\n", left->page);
342 }
343 return left;
344}

351static void delete_key(Tree *tree, Node *node, int key, Deletion *operation) {
352 int at = position(node, key), t = tree->degree;
353 bool trace = operation && operation->trace;
354 if (at < node->count && node->entries[at].key == key) {
355 if (!node->child_count) {
356 add_case(operation, "1"); remove_entry(node, at);
357 if (trace) { printf("STEP 1 page=%d keys=", node->page); notation(node); putchar('\n'); }
358 return;
359 }
360 Node *left = node->children[at], *right = node->children[at + 1];
361 if (left->count >= t) {
362 Entry predecessor = extreme(left, true);
363 add_case(operation, "2a");
364 if (trace) printf("STEP 2a page=%d pred=%d\n", node->page, predecessor.key);
365 delete_key(tree, left, predecessor.key, operation);
366 node->entries[at] = predecessor;
367 } else if (right->count >= t) {
368 Entry successor = extreme(right, false);
369 add_case(operation, "2b");
370 if (trace) printf("STEP 2b page=%d succ=%d\n", node->page, successor.key);
371 delete_key(tree, right, successor.key, operation);
372 node->entries[at] = successor;
373 } else {
374 add_case(operation, "2c");
375 Node *merged = merge_children(tree, node, at, "2c", operation);
376 delete_key(tree, merged, key, operation);
377 }
378 return;
379 }
380 Node *child = node->children[at];
381 if (child->count == t - 1) {
382 Node *left = at ? node->children[at - 1] : NULL;
383 Node *right = at < node->count ? node->children[at + 1] : NULL;
384 if (left && left->count >= t) {
385 add_case(operation, "3a-left");
386 insert_entry(child, 0, node->entries[at - 1]);
387 node->entries[at - 1] = remove_entry(left, left->count - 1);
388 if (left->child_count) insert_child(child, 0, remove_child(left, left->child_count - 1));
389 ++tree->borrows;
390 if (trace) {
391 printf("STEP 3a-left page=%d child=%d from=%d keys=", node->page, child->page, left->page);
392 notation(child); putchar('\n');
393 }
394 } else if (right && right->count >= t) {
395 add_case(operation, "3a-right");
396 insert_entry(child, child->count, node->entries[at]);
397 node->entries[at] = remove_entry(right, 0);
398 if (right->child_count) insert_child(child, child->child_count, remove_child(right, 0));
399 ++tree->borrows;
400 if (trace) {
401 printf("STEP 3a-right page=%d child=%d from=%d keys=", node->page, child->page, right->page);
402 notation(child); putchar('\n');
403 }
404 } else if (left) {
405 add_case(operation, "3b-left");
406 child = merge_children(tree, node, at - 1, "3b-left", operation);
407 } else {
408 add_case(operation, "3b-right");
409 child = merge_children(tree, node, at, "3b-right", operation);
410 }
411 }
412 delete_key(tree, child, key, operation);
413}

415static Node *parse_node(Parser *parser) {
416 if (parser->position == parser->count || strcmp(parser->tokens[parser->position], "(")) {
417 parser->error = parser->position + 1;
418 return NULL;
419 }
420 ++parser->position;
421 Node *node = new_node(parser->tree);
422 if (parser->position == parser->count) { parser->error = parser->position + 1; return NULL; }
423 bool internal = !strcmp(parser->tokens[parser->position], "(");
424 if (internal) {
425 Node *child = parse_node(parser);
426 if (!child) return NULL;
427 insert_child(node, node->child_count, child);
428 }
429 while (parser->position < parser->count && strcmp(parser->tokens[parser->position], ")")) {
430 const char *token = parser->tokens[parser->position];
431 if (!strcmp(token, "(")) { parser->error = parser->position + 1; return NULL; }
432 insert_entry(node, node->count, entry(atoi(token), strchr(token, '=') + 1));
433 ++parser->position;
434 if (internal) {
435 Node *child = parse_node(parser);
436 if (!child) return NULL;
437 insert_child(node, node->child_count, child);
438 }
439 }
440 if (parser->position == parser->count) { parser->error = parser->position + 1; return NULL; }
441 ++parser->position;
442 return node;
443}

451static Tree *compact(const Tree *source) {
452 Entries entries = {0};
453 inorder(source->root, &entries);
454 Tree *tree = allocate(sizeof(Tree));
455 tree->degree = source->degree;
456 Node *candidates[MAX_PAGES];
457 int child_count = 0;
458 while (true) {
459 int groups = (entries.count + 2 * tree->degree) / (2 * tree->degree);
460 int keys = entries.count - groups + 1, size = keys / groups, remainder = keys % groups;
461 Entries separators = {0};
462 Node *level[MAX_PAGES];
463 int at = 0, child = 0;
464 for (int i = 0; i < groups; ++i) {
465 Node *node = new_node(tree);
466 int count = size + (i < remainder);
467 for (int j = 0; j < count; ++j) insert_entry(node, node->count, entries.entries[at++]);
468 if (child_count) for (int j = 0; j <= count; ++j) insert_child(node, node->child_count, candidates[child++]);
469 level[i] = node;
470 if (i + 1 < groups) separators.entries[separators.count++] = entries.entries[at++];
471 }
472 if (groups == 1) { tree->root = level[0]; break; }
473 entries = separators;
474 memcpy(candidates, level, (size_t)groups * sizeof(Node *));
475 child_count = groups;
476 }
477 memset(tree->pages, 0, sizeof(tree->pages));
478 int next = 0;
479 number_preorder(tree, tree->root, &next);
480 return tree;
481}

493 Node *queue[MAX_PAGES];
494 int begin = 0, end = 1, depth = 0;
495 queue[0] = tree->root;

508static void print_violation(const Tree *tree, Violation violation) {
509 static const char *const names[] = {"", "COUNT", "ORDER", "RANGE", "DEPTH"};
510 if (violation.kind == VALID) { puts("VERIFY OK"); return; }
511 const Node *node = violation.node;
512 printf("VERIFY FAIL %s at=%s node=", names[violation.kind], violation.location); notation(node);
513 switch (violation.kind) {
514 case BAD_COUNT: printf(" keys=%d allowed=%d..%d", node->count, violation.minimum, 2 * tree->degree - 1); break;
515 case BAD_ORDER: printf(" %d>=%d", node->entries[violation.index].key, node->entries[violation.index + 1].key); break;
516 case BAD_RANGE:
517 printf(" key=%d low=", node->entries[violation.index].key);
518 if (violation.low == INT_MIN) printf("-inf"); else printf("%d", violation.low);
519 printf(" high=");
520 if (violation.high == INT_MAX) printf("+inf"); else printf("%d", violation.high);
521 break;
522 case BAD_DEPTH: printf(" depth=%d expected=%d", violation.depth, violation.expected); break;
523 case VALID: break;
524 }
525 putchar('\n');
526}

528static bool needs_valid(Command command) {
529 switch (command) {
530 case INSERT: case BULKLOAD: case DELETE: case DELRANGE: case RANGE: case COUNT:
531 case MIN: case MAX: case FLOOR: case CEIL: case RANK: case SELECT: case SEEK:
532 case NEXT: case PREV: case HEIGHT: case COMPACT: return true;
533 default: return false;
534 }
535}

537static void process(Laboratory *lab, Command command, char **args, int count) {
538 if (command == TRACE) {
539 lab->trace = !strcmp(args[0], "ON");
540 printf("TRACE %s\n", args[0]);
541 return;
542 }
543 if (command == CREATE || command == LOAD) {
544 int degree = atoi(args[0]);
545 if (degree < 2 || degree > 5) { printf("ERROR BAD DEGREE %d\n", degree); return; }
546 Tree *replacement;
547 if (command == CREATE) replacement = new_tree(degree);
548 else {
549 replacement = allocate(sizeof(Tree));
550 replacement->degree = degree;
551 Parser parser = {.tree = replacement, .tokens = args + 1, .count = count - 1};
552 replacement->root = parse_node(&parser);
553 if (!parser.error && parser.position != parser.count) parser.error = parser.position + 1;
554 if (parser.error) {
555 printf("ERROR SYNTAX %d\n", parser.error); release_tree(replacement); return;
556 }
557 }
558 release_tree(lab->tree); lab->tree = replacement;
559 lab->cursor = (Cursor){.state = UNSET};
560 if (command == CREATE) printf("CREATE t=%d\n", degree);
561 else {
562 Statistics stats = statistics(replacement);
563 printf("LOAD t=%d nodes=%d keys=%d %s\n", degree, stats.nodes, stats.keys, status(replacement));
564 }
565 return;
566 }
567 Tree *tree = lab->tree;
568 if (!tree) { puts("ERROR NO INDEX"); return; }
569 Violation violation = inspect(tree);
570 if (needs_valid(command) && violation.kind != VALID) { puts("ERROR DAMAGED"); return; }
571 int first = count ? atoi(args[0]) : 0;
572 int second = command == DELRANGE || command == RANGE || command == COUNT ? atoi(args[1]) : 0;
573 if ((command == DELRANGE || command == RANGE || command == COUNT) && first > second) {
574 printf("ERROR BAD RANGE %d %d\n", first, second); return;
575 }
576 Statistics stats = statistics(tree);
577 if (command == SELECT && (first < 1 || first > stats.keys)) { printf("ERROR BAD POSITION %d\n", first); return; }
578 if ((command == NEXT || command == PREV) && lab->cursor.state == UNSET) { puts("ERROR NO CURSOR"); return; }
579 if (command == RESTORE && !lab->saved) { puts("ERROR NO SAVE"); return; }
580
581 if (command == INSERT || command == BULKLOAD || command == DELETE || command == DELRANGE) {
582 Search found = {0};
583 if (command == INSERT || command == DELETE) found = search(tree, first);
584 if (command == DELETE && !found.node) { printf("DELETE %d NOT FOUND\n", first); return; }
585 Tree *before = copy_tree(tree);
586 int old_splits = tree->splits, old_merges = tree->merges, old_borrows = tree->borrows;
587 if (command == INSERT) {
588 Entry previous = {0};
589 if (found.node) previous = found.node->entries[found.position];
590 bool added = put_entry(tree, entry(first, args[1]), lab->trace);
591 Changes delta = changes(before, tree);
592 if (added) printf("INSERT %d NEW splits=%d height=%d writes=%d\n", first, tree->splits - old_splits, height(tree), delta.writes);
593 else printf("INSERT %d UPDATED old=%s writes=%d\n", first, previous.value, delta.writes);
594 } else if (command == BULKLOAD) {
595 int added = 0;
596 for (int i = 0; i < count; i += 2) added += put_entry(tree, entry(atoi(args[i]), args[i + 1]), false);
597 Changes delta = changes(before, tree);
598 printf("BULKLOAD pairs=%d new=%d updated=%d splits=%d height=%d writes=%d\n",
599 count / 2, added, count / 2 - added, tree->splits - old_splits, height(tree), delta.writes);
600 } else if (command == DELETE) {
601 Entry previous = found.node->entries[found.position];
602 Deletion operation = {.trace = lab->trace};
603 delete_key(tree, tree->root, first, &operation);
604 Changes delta = changes(before, tree);
605 printf("DELETE %d REMOVED value=%s cases=%s height=%d writes=%d frees=%d\n",
606 first, previous.value, operation.cases, height(tree), delta.writes, delta.frees);
607 } else {
608 Entries selected = {0};
609 int visited = 0;
610 scan(tree->root, first, second, &selected, &visited);
611 for (int i = 0; i < selected.count; ++i) delete_key(tree, tree->root, selected.entries[i].key, NULL);
612 Changes delta = changes(before, tree);
613 printf("DELRANGE %d %d removed=%d merges=%d borrows=%d height=%d writes=%d frees=%d\n",
614 first, second, selected.count, tree->merges - old_merges, tree->borrows - old_borrows,
615 height(tree), delta.writes, delta.frees);
616 }
617 release_tree(before);
618 return;
619 }
620 if (command == GET) {
621 Search found = search(tree, first);
622 printf("GET %d ", first);
623 if (found.node) printf("FOUND value=%s path=", found.node->entries[found.position].value);
624 else printf("NOT FOUND path=");
625 for (int i = 0; i < found.depth; ++i) { if (i) printf(" -> "); notation(found.path[i]); }
626 putchar('\n'); return;
627 }
628 if (command == RANGE || command == COUNT) {
629 Entries result = {0};
630 int visited = 0;
631 scan(tree->root, first, second, &result, &visited);
632 printf("%s %d %d count=%d visited=%d", COMMANDS[command], first, second, result.count, visited);
633 if (command == RANGE) for (int i = 0; i < result.count; ++i) printf(" %d=%s", result.entries[i].key, result.entries[i].value);
634 putchar('\n'); return;
635 }
636 if (command >= MIN && command <= PREV) {
637 Entries all = {0}; inorder(tree->root, &all);
638 const char *name = COMMANDS[command];
639 int selected = -1;
640 if (command == MIN || command == MAX) {
641 if (!all.count) { printf("%s EMPTY\n", name); return; }
642 selected = command == MIN ? 0 : all.count - 1;
643 } else if (command == FLOOR || command == CEIL || command == SEEK) {
644 for (int i = 0; i < all.count; ++i) {
645 if (command == FLOOR) { if (all.entries[i].key <= first) selected = i; }
646 else if (all.entries[i].key >= first) { selected = i; break; }
647 }
648 } else if (command == RANK) {
649 int rank = 0; while (rank < all.count && all.entries[rank].key < first) ++rank;
650 printf("RANK %d %d\n", first, rank); return;
651 } else if (command == SELECT) selected = first - 1;
652 else if (command == NEXT) {
653 for (int i = 0; i < all.count; ++i)
654 if (lab->cursor.state == BEGIN || (lab->cursor.state == AT_KEY && all.entries[i].key > lab->cursor.key)) { selected = i; break; }
655 } else {
656 for (int i = all.count - 1; i >= 0; --i)
657 if (lab->cursor.state == END || (lab->cursor.state == AT_KEY && all.entries[i].key < lab->cursor.key)) { selected = i; break; }
658 }
659 if (command == SEEK || command == NEXT || command == PREV) {
660 lab->cursor.state = selected >= 0 ? AT_KEY : command == PREV ? BEGIN : END;
661 if (selected >= 0) lab->cursor.key = all.entries[selected].key;
662 }
663 printf("%s", name);
664 if (command == FLOOR || command == CEIL || command == SELECT || command == SEEK) printf(" %d", first);
665 if (selected >= 0) printf(" %d=%s\n", all.entries[selected].key, all.entries[selected].value);
666 else printf(" %s\n", command == PREV ? "BEGIN" : command == NEXT || command == SEEK ? "END" : "NONE");
667 return;
668 }
669 switch (command) {
670 case PRINT: print_levels(tree); break;
671 case PAGES:
672 for (int page = 0; page < MAX_PAGES; ++page) {
673 Node *node = tree->pages[page];
674 if (!node) continue;
675 printf("PAGE %d ", page); notation(node); printf(" kids=");
676 if (!node->child_count) putchar('-');
677 for (int i = 0; i < node->child_count; ++i) printf("%s%d", i ? "," : "", node->children[i]->page);
678 putchar('\n');
679 }
680 break;
681 case DUMP: printf("DUMP %d ", tree->degree); dump_node(tree->root); putchar('\n'); break;
682 case HEIGHT: printf("HEIGHT %d\n", height(tree)); break;
683 case STATS:
684 printf("STATS t=%d nodes=%d leaves=%d keys=%d fill=%d%% splits=%d merges=%d borrows=%d\n",
685 tree->degree, stats.nodes, stats.leaves, stats.keys, 100 * stats.keys / ((2 * tree->degree - 1) * stats.nodes),
686 tree->splits, tree->merges, tree->borrows);
687 break;
688 case VERIFY: print_violation(tree, violation); break;
689 case REBUILD: {
690 Entries all = {0}; inorder(tree->root, &all);
691 Tree *replacement = new_tree(tree->degree);
692 for (int i = 0; i < all.count; ++i) put_entry(replacement, all.entries[i], false);
693 release_tree(tree); lab->tree = replacement;
694 Statistics now = statistics(replacement);
695 printf("REBUILD entries=%d keys=%d nodes=%d splits=%d height=%d\n", all.count, now.keys, now.nodes, replacement->splits, height(replacement));
696 break;
697 }
698 case COMPACT: {
699 Tree *replacement = compact(tree);
700 release_tree(tree); lab->tree = replacement;
701 printf("COMPACT nodes=%d->%d height=%d\n", stats.nodes, statistics(replacement).nodes, height(replacement));
702 break;
703 }
704 case SAVE:
705 release_tree(lab->saved); lab->saved = copy_tree(tree);
706 printf("SAVE nodes=%d keys=%d %s\n", stats.nodes, stats.keys, status(tree));
707 break;
708 case RESTORE:
709 release_tree(tree); lab->tree = copy_tree(lab->saved);
710 stats = statistics(lab->tree);
711 printf("RESTORE nodes=%d keys=%d %s\n", stats.nodes, stats.keys, status(lab->tree));
712 break;
713 default: break;
714 }
715}

717int main(void) {
718 char line[4002];
719 if (!fgets(line, sizeof(line), stdin)) return 0;
720 int commands = atoi(line);
721 Laboratory laboratory = {0};
722 for (int i = 0; i < commands; ++i) {
723 if (!fgets(line, sizeof(line), stdin)) break;
724 line[strcspn(line, "\r\n")] = '\0';
725 char *words[1503];
726 int count = 0;
727 for (char *word = strtok(line, " "); word; word = strtok(NULL, " ")) words[count++] = word;
728 int command = 0;
729 while (strcmp(words[0], COMMANDS[command])) ++command;
730 process(&laboratory, (Command)command, words + 1, count - 1);
731 }
732 release_tree(laboratory.tree);
733 release_tree(laboratory.saved);
734 return 0;
735}
