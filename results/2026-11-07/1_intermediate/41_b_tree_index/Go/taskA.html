27 for i, e := range n.entries {

43 result := make(map[int]*node, len(pages))
44 for id, n := range pages {

86func (t *tree) entries() []entry {
87 var result []entry
88 var visit func(int)
89 visit = func(page int) {
90 n := t.pages[page]
91 for j := 0; j <= len(n.entries); j++ {
92 if len(n.children) != 0 {
93 visit(n.children[j])
94 }
95 if j < len(n.entries) {
96 result = append(result, n.entries[j])
97 }
98 }
99 }
100 visit(t.root)
101 return result

104type searchResult struct {
105 page, position int
106 found bool
107 path []string
108}

110func (t *tree) search(key int) searchResult {
111 result := searchResult{page: t.root}
112 for {
113 n := t.pages[result.page]
114 result.path = append(result.path, n.notation())
115 result.position = n.position(key)
116 if result.position < len(n.entries) && n.entries[result.position].key == key {
117 result.found = true
118 return result
119 }
120 if len(n.children) == 0 {
121 return result
122 }
123 result.page = n.children[result.position]
124 }
125}

127func (t *tree) violation() string {
128 referenceDepth := -1
129 var visit func(int, string, int, *int, *int) string
130 visit = func(page int, location string, depth int, low, high *int) string {
131 n := t.pages[page]
132 count, minimum, maximum := len(n.entries), t.degree-1, 2*t.degree-1
133 if page == t.root {
134 minimum = 0
135 if len(n.children) != 0 {
136 minimum = 1
137 }
138 }
139 details := fmt.Sprintf("at=%s node=%s", location, n.notation())
140 if count < minimum || count > maximum {
141 return fmt.Sprintf("COUNT %s keys=%d allowed=%d..%d", details, count, minimum, maximum)
142 }
143 for j := 0; j+1 < count; j++ {
144 if n.entries[j].key >= n.entries[j+1].key {
145 return fmt.Sprintf("ORDER %s %d>=%d", details, n.entries[j].key, n.entries[j+1].key)
146 }
147 }
148 for _, e := range n.entries {
149 if low != nil && e.key <= *low || high != nil && e.key >= *high {
150 lower, upper := "-inf", "+inf"
151 if low != nil {
152 lower = strconv.Itoa(*low)
153 }
154 if high != nil {
155 upper = strconv.Itoa(*high)
156 }
157 return fmt.Sprintf("RANGE %s key=%d low=%s high=%s", details, e.key, lower, upper)
158 }
159 }
160 if len(n.children) == 0 {
161 if referenceDepth < 0 {
162 referenceDepth = depth
163 } else if referenceDepth != depth {
164 return fmt.Sprintf("DEPTH %s depth=%d expected=%d", details, depth, referenceDepth)
165 }
166 }
167 for j, child := range n.children {
168 lower, upper := low, high
169 if j > 0 {
170 lower = &n.entries[j-1].key
171 }
172 if j < count {
173 upper = &n.entries[j].key
174 }
175 if problem := visit(child, location+"."+strconv.Itoa(j), depth+1, lower, upper); problem != "" {
176 return problem
177 }
178 }
179 return ""
180 }
181 return visit(t.root, "root", 0, nil, nil)
182}

205type parser struct {
206 tokens []string
207 position int
208 index *tree
209}
210
211func (p *parser) peek() string {
212 if p.position == len(p.tokens) {
213 return ""
214 }
215 return p.tokens[p.position]
216}
217
218func (p *parser) entry() (entry, error) {
219 key, value, ok := strings.Cut(p.peek(), "=")
220 if !ok {
221 return entry{}, fmt.Errorf("SYNTAX %d", p.position+1)
222 }
223 p.position++
224 return entry{number(key), value}, nil
225}
226
227func (p *parser) node() (int, error) {
228 if p.peek() != "(" {
229 return 0, fmt.Errorf("SYNTAX %d", p.position+1)
230 }
231 p.position++
232 n := &node{}
233 page := p.index.allocate(n)
234 internal := p.peek() == "("
235 if internal {
236 child, err := p.node()
237 if err != nil {
238 return 0, err
239 }
240 n.children = append(n.children, child)
241 }
242 for p.peek() != ")" {
243 e, err := p.entry()
244 if err != nil {
245 return 0, err
246 }
247 n.entries = append(n.entries, e)
248 if internal {
249 child, err := p.node()
250 if err != nil {
251 return 0, err
252 }
253 n.children = append(n.children, child)
254 }
255 }
256 p.position++
257 return page, nil
258}

260func loadTree(degree int, tokens []string) (*tree, error) {
261 p := parser{tokens: tokens, index: &tree{degree: degree, pages: make(map[int]*node)}}
262 root, err := p.node()
263 if err != nil {
264 return nil, err
265 }
266 if p.position != len(tokens) {
267 return nil, fmt.Errorf("SYNTAX %d", p.position+1)
268 }
269 p.index.root = root
270 return p.index, nil
271}

273func (t *tree) changes(before map[int]*node) (writes, frees int) {
274 for page, after := range t.pages {
275 old := before[page]
276 if old == nil || !slices.Equal(old.entries, after.entries) || !slices.Equal(old.children, after.children) {
277 writes++
278 }
279 }
280 for page := range before {
281 if t.pages[page] == nil {
282 frees++
283 }
284 }
285 return
286}

288func (t *tree) split(parent, position int, tr *trace) {
289 x := t.pages[parent]
290 child := x.children[position]
291 y := t.pages[child]
292 old, middle := y.notation(), y.entries[t.degree-1]
293 z := &node{entries: slices.Clone(y.entries[t.degree:])}
294 if len(y.children) != 0 {
295 z.children = slices.Clone(y.children[t.degree:])
296 y.children = y.children[:t.degree]
297 }
298 y.entries = y.entries[:t.degree-1]
299 newPage := t.allocate(z)
300 x.entries = slices.Insert(x.entries, position, middle)
301 x.children = slices.Insert(x.children, position+1, newPage)
302 t.splits++
303 tr.emit("STEP split page=%d keys=%s up=%d new=%d", child, old, middle.key, newPage)
304}

306func (t *tree) upsert(e entry, tr *trace) (string, bool) {
307 found := t.search(e.key)
308 if found.found {
309 n := t.pages[found.page]
310 old := n.entries[found.position].value
311 n.entries[found.position].value = e.value
312 return old, true
313 }
314 if len(t.pages[t.root].entries) == 2*t.degree-1 {
315 t.root = t.allocate(&node{children: []int{t.root}})
316 tr.emit("STEP grow page=%d", t.root)
317 t.split(t.root, 0, tr)
318 }
319 page := t.root
320 for len(t.pages[page].children) != 0 {
321 n := t.pages[page]
322 position := n.position(e.key)
323 if len(t.pages[n.children[position]].entries) == 2*t.degree-1 {
324 t.split(page, position, tr)
325 if e.key > n.entries[position].key {
326 position++
327 }
328 }
329 page = n.children[position]
330 }
331 n := t.pages[page]
332 n.entries = slices.Insert(n.entries, n.position(e.key), e)
333 tr.emit("STEP place page=%d keys=%s", page, n.notation())
334 return "", false
335}

337func (t *tree) extreme(page int, last bool) entry {
338 for {
339 n := t.pages[page]
340 if len(n.children) == 0 {
341 if last {
342 return n.entries[len(n.entries)-1]
343 }
344 return n.entries[0]
345 }
346 page = n.children[0]
347 if last {
348 page = n.children[len(n.children)-1]
349 }
350 }
351}

353func (t *tree) merge(parent, position int, caseName string, tr *trace) int {
354 x := t.pages[parent]
355 left, right := x.children[position], x.children[position+1]
356 y, z := t.pages[left], t.pages[right]
357 y.entries = append(y.entries, x.entries[position])
358 y.entries = append(y.entries, z.entries...)
359 y.children = append(y.children, z.children...)
360 x.entries = slices.Delete(x.entries, position, position+1)
361 x.children = slices.Delete(x.children, position+1, position+2)
362 delete(t.pages, right)
363 t.merges++
364 tr.emit("STEP %s page=%d merge=%d+%d keys=%s", caseName, parent, left, right, y.notation())
365 if parent == t.root && len(x.entries) == 0 {
366 t.root = left
367 delete(t.pages, parent)
368 tr.emit("STEP shrink page=%d", left)
369 }
370 return left
371}

373func (t *tree) removeAt(page, key int, cases *[]string, tr *trace) {
374 x := t.pages[page]
375 position := x.position(key)
376 if position < len(x.entries) && x.entries[position].key == key {
377 if len(x.children) == 0 {
378 *cases = append(*cases, "1")
379 x.entries = slices.Delete(x.entries, position, position+1)
380 tr.emit("STEP 1 page=%d keys=%s", page, x.notation())
381 return
382 }
383 left, right := x.children[position], x.children[position+1]
384 if len(t.pages[left].entries) >= t.degree {
385 *cases = append(*cases, "2a")
386 predecessor := t.extreme(left, true)
387 tr.emit("STEP 2a page=%d pred=%d", page, predecessor.key)
388 t.removeAt(left, predecessor.key, cases, tr)
389 x.entries[position] = predecessor
390 } else if len(t.pages[right].entries) >= t.degree {
391 *cases = append(*cases, "2b")
392 successor := t.extreme(right, false)
393 tr.emit("STEP 2b page=%d succ=%d", page, successor.key)
394 t.removeAt(right, successor.key, cases, tr)
395 x.entries[position] = successor
396 } else {
397 *cases = append(*cases, "2c")
398 child := t.merge(page, position, "2c", tr)
399 t.removeAt(child, key, cases, tr)
400 }
401 return
402 }
403 child := x.children[position]
404 c := t.pages[child]
405 if len(c.entries) == t.degree-1 {
406 left, right := -1, -1
407 if position > 0 {
408 left = x.children[position-1]
409 }
410 if position+1 < len(x.children) {
411 right = x.children[position+1]
412 }
413 switch {
414 case left >= 0 && len(t.pages[left].entries) >= t.degree:
415 *cases = append(*cases, "3a-left")
416 sibling := t.pages[left]
417 last := len(sibling.entries) - 1
418 c.entries = slices.Insert(c.entries, 0, x.entries[position-1])
419 x.entries[position-1] = sibling.entries[last]
420 sibling.entries = sibling.entries[:last]
421 if len(sibling.children) != 0 {
422 last := len(sibling.children) - 1
423 c.children = slices.Insert(c.children, 0, sibling.children[last])
424 sibling.children = sibling.children[:last]
425 }
426 t.borrows++
427 tr.emit("STEP 3a-left page=%d child=%d from=%d keys=%s", page, child, left, c.notation())
428 case right >= 0 && len(t.pages[right].entries) >= t.degree:
429 *cases = append(*cases, "3a-right")
430 sibling := t.pages[right]
431 c.entries = append(c.entries, x.entries[position])
432 x.entries[position] = sibling.entries[0]
433 sibling.entries = slices.Delete(sibling.entries, 0, 1)
434 if len(sibling.children) != 0 {
435 c.children = append(c.children, sibling.children[0])
436 sibling.children = slices.Delete(sibling.children, 0, 1)
437 }
438 t.borrows++
439 tr.emit("STEP 3a-right page=%d child=%d from=%d keys=%s", page, child, right, c.notation())
440 case left >= 0:
441 *cases = append(*cases, "3b-left")
442 child = t.merge(page, position-1, "3b-left", tr)
443 default:
444 *cases = append(*cases, "3b-right")
445 child = t.merge(page, position, "3b-right", tr)
446 }
447 }
448 t.removeAt(child, key, cases, tr)
449}

451func (t *tree) scan(low, high int) ([]entry, int) {
452 var entries []entry
453 visits := 0
454 var visit func(int)
455 visit = func(page int) {
456 visits++
457 n := t.pages[page]
458 for j := 0; j <= len(n.entries); j++ {
459 if len(n.children) != 0 && (j == 0 || n.entries[j-1].key < high) && (j == len(n.entries) || n.entries[j].key > low) {
460 visit(n.children[j])
461 }
462 if j < len(n.entries) && low <= n.entries[j].key && n.entries[j].key <= high {
463 entries = append(entries, n.entries[j])
464 }
465 }
466 }
467 visit(t.root)
468 return entries, visits
469}

471func (t *tree) dump() string {
472 var tokens []string
473 var visit func(int)
474 visit = func(page int) {
475 n := t.pages[page]
476 tokens = append(tokens, "(")
477 for j := 0; j <= len(n.entries); j++ {
478 if len(n.children) != 0 {
479 visit(n.children[j])
480 }
481 if j < len(n.entries) {
482 tokens = append(tokens, n.entries[j].String())
483 }
484 }
485 tokens = append(tokens, ")")
486 }
487 visit(t.root)
488 return strings.Join(tokens, " ")
489}

491func (t *tree) compact() {
492 source := t.entries()
493 var children []int
494 var built []*node
495 oldRoot := 0
496 for {
497 groups := (len(source) + 2*t.degree) / (2 * t.degree)
498 keys := len(source) - (groups - 1)
499 base, extra := keys/groups, keys%groups
500 position, childPosition := 0, 0
501 var separators []entry
502 var level []int
503 for j := range groups {
504 count := base
505 if j < extra {
506 count++
507 }
508 n := &node{entries: slices.Clone(source[position : position+count])}
509 position += count
510 if len(children) != 0 {
511 n.children = slices.Clone(children[childPosition : childPosition+count+1])
512 childPosition += count + 1
513 }
514 level = append(level, len(built))
515 built = append(built, n)
516 if j+1 < groups {
517 separators = append(separators, source[position])
518 position++
519 }
520 }
521 if groups == 1 {
522 oldRoot = level[0]
523 break
524 }
525 source, children = separators, level
526 }
527 t.pages = make(map[int]*node)
528 var renumber func(int) int
529 renumber = func(old int) int {
530 page := len(t.pages)
531 n := &node{entries: built[old].entries}
532 t.pages[page] = n
533 for _, child := range built[old].children {
534 n.children = append(n.children, renumber(child))
535 }
536 return page
537 }
538 t.root = renumber(oldRoot)
539 t.splits, t.merges, t.borrows = 0, 0, 0
540}

582func (w *workbench) process(line string) ([]string, error) {
583 words := strings.Split(line, " ")
584 command := words[0]
585 if command == "TRACE" {
586 w.trace = words[1] == "ON"
587 return []string{"TRACE " + words[1]}, nil
588 }
589 first, second := 0, 0
590 if len(words) > 1 {
591 first = number(words[1])
592 }
593 ranged := command == "DELRANGE" || command == "RANGE" || command == "COUNT"
594 if ranged {
595 second = number(words[2])
596 }
597 if command == "CREATE" || command == "LOAD" {
598 if first < 2 || first > 5 {
599 return nil, fmt.Errorf("BAD DEGREE %d", first)
600 }
601 t := newTree(first)
602 result := fmt.Sprintf("CREATE t=%d", first)
603 if command == "LOAD" {
604 loaded, err := loadTree(first, words[2:])
605 if err != nil {
606 return nil, err
607 }
608 t = loaded
609 result = fmt.Sprintf("LOAD t=%d nodes=%d keys=%d %s", first, len(t.pages), len(t.entries()), t.status())
610 }
611 w.index, w.cursor = t, cursor{}
612 return []string{result}, nil
613 }
614 t := w.index
615 if t == nil {
616 return nil, errors.New("NO INDEX")
617 }
618 switch command {
619 case "INSERT", "BULKLOAD", "DELETE", "DELRANGE", "RANGE", "COUNT", "MIN", "MAX", "FLOOR", "CEIL", "RANK", "SELECT", "SEEK", "NEXT", "PREV", "HEIGHT", "COMPACT":
620 if t.violation() != "" {
621 return nil, errors.New("DAMAGED")
622 }
623 }
624 entries := t.entries()
625 if ranged && first > second {
626 return nil, fmt.Errorf("BAD RANGE %d %d", first, second)
627 }
628 if command == "SELECT" && (first < 1 || first > len(entries)) {
629 return nil, fmt.Errorf("BAD POSITION %d", first)
630 }
631 if (command == "NEXT" || command == "PREV") && w.cursor.kind == unset {
632 return nil, errors.New("NO CURSOR")
633 }
634 if command == "RESTORE" && w.saved == nil {
635 return nil, errors.New("NO SAVE")
636 }
637 var output []string
638 tr := &trace{enabled: w.trace && (command == "INSERT" || command == "DELETE")}
639 switch command {
640 case "INSERT":
641 before, splits := copyPages(t.pages), t.splits
642 old, updated := t.upsert(entry{first, words[2]}, tr)
643 writes, _ := t.changes(before)
644 output = append(output, tr.lines...)
645 if updated {
646 output = append(output, fmt.Sprintf("INSERT %d UPDATED old=%s writes=%d", first, old, writes))
647 } else {
648 output = append(output, fmt.Sprintf("INSERT %d NEW splits=%d height=%d writes=%d", first, t.splits-splits, t.height(), writes))
649 }
650 case "BULKLOAD":
651 before, splits := copyPages(t.pages), t.splits
652 added, updated := 0, 0
653 for i := 1; i < len(words); i += 2 {
654 if _, replaced := t.upsert(entry{number(words[i]), words[i+1]}, tr); replaced {
655 updated++
656 } else {
657 added++
658 }
659 }
660 writes, _ := t.changes(before)
661 output = append(output, fmt.Sprintf("BULKLOAD pairs=%d new=%d updated=%d splits=%d height=%d writes=%d", (len(words)-1)/2, added, updated, t.splits-splits, t.height(), writes))
662 case "DELETE":
663 found := t.search(first)
664 if !found.found {
665 output = append(output, fmt.Sprintf("DELETE %d NOT FOUND", first))
666 break
667 }
668 value := t.pages[found.page].entries[found.position].value
669 before := copyPages(t.pages)
670 var cases []string
671 t.removeAt(t.root, first, &cases, tr)
672 writes, frees := t.changes(before)
673 output = append(output, tr.lines...)
674 output = append(output, fmt.Sprintf("DELETE %d REMOVED value=%s cases=%s height=%d writes=%d frees=%d", first, value, strings.Join(cases, ","), t.height(), writes, frees))
675 case "DELRANGE":
676 before := copyPages(t.pages)
677 merges, borrows, removed := t.merges, t.borrows, 0
678 for _, e := range entries {
679 if first <= e.key && e.key <= second {
680 var cases []string
681 t.removeAt(t.root, e.key, &cases, tr)
682 removed++
683 }
684 }
685 writes, frees := t.changes(before)
686 output = append(output, fmt.Sprintf("DELRANGE %d %d removed=%d merges=%d borrows=%d height=%d writes=%d frees=%d", first, second, removed, t.merges-merges, t.borrows-borrows, t.height(), writes, frees))
687 case "GET":
688 found := t.search(first)
689 result := "NOT FOUND"
690 if found.found {
691 result = "FOUND value=" + t.pages[found.page].entries[found.position].value
692 }
693 output = append(output, fmt.Sprintf("GET %d %s path=%s", first, result, strings.Join(found.path, " -> ")))
694 case "RANGE", "COUNT":
695 found, visits := t.scan(first, second)
696 result := fmt.Sprintf("%s %d %d count=%d visited=%d", command, first, second, len(found), visits)
697 if command == "RANGE" {
698 for _, e := range found {
699 result += " " + e.String()
700 }
701 }
702 output = append(output, result)
703 case "MIN", "MAX":
704 e := findEntry(entries, command == "MAX", func(entry) bool { return true })
705 output = append(output, command+" "+display(e, "EMPTY"))
706 case "FLOOR", "CEIL":
707 e := findEntry(entries, command == "FLOOR", func(e entry) bool {
708 if command == "FLOOR" {
709 return e.key <= first
710 }
711 return e.key >= first
712 })
713 output = append(output, fmt.Sprintf("%s %d %s", command, first, display(e, "NONE")))
714 case "RANK":
715 rank := 0
716 for _, e := range entries {
717 if e.key < first {
718 rank++
719 }
720 }
721 output = append(output, fmt.Sprintf("RANK %d %d", first, rank))
722 case "SELECT":
723 output = append(output, fmt.Sprintf("SELECT %d %s", first, entries[first-1]))
724 case "SEEK":
725 e := findEntry(entries, false, func(e entry) bool { return e.key >= first })
726 w.cursor = cursor{kind: end}
727 if e != nil {
728 w.cursor = cursor{atKey, e.key}
729 }
730 output = append(output, fmt.Sprintf("SEEK %d %s", first, display(e, "END")))
731 case "NEXT", "PREV":
732 var e *entry
733 if command == "NEXT" {
734 if w.cursor.kind == begin || w.cursor.kind == atKey {
735 e = findEntry(entries, false, func(e entry) bool { return w.cursor.kind == begin || e.key > w.cursor.key })
736 }
737 w.cursor = cursor{kind: end}
738 output = append(output, "NEXT "+display(e, "END"))
739 } else {
740 if w.cursor.kind == end || w.cursor.kind == atKey {
741 e = findEntry(entries, true, func(e entry) bool { return w.cursor.kind == end || e.key < w.cursor.key })
742 }
743 w.cursor = cursor{kind: begin}
744 output = append(output, "PREV "+display(e, "BEGIN"))
745 }
746 if e != nil {
747 w.cursor = cursor{atKey, e.key}
748 }
749 case "PRINT":
750 level := []int{t.root}
751 for depth := 0; len(level) != 0; depth++ {
752 result := fmt.Sprintf("LEVEL %d", depth)
753 var next []int
754 for _, page := range level {
755 n := t.pages[page]
756 result += " " + n.notation()
757 next = append(next, n.children...)
758 }
759 output = append(output, result)
760 level = next
761 }
762 case "PAGES":
763 pages := make([]int, 0, len(t.pages))
764 for page := range t.pages {
765 pages = append(pages, page)
766 }
767 slices.Sort(pages)
768 for _, page := range pages {
769 n := t.pages[page]
770 children := make([]string, len(n.children))
771 for j, child := range n.children {
772 children[j] = strconv.Itoa(child)
773 }
774 kids := "-"
775 if len(children) != 0 {
776 kids = strings.Join(children, ",")
777 }
778 output = append(output, fmt.Sprintf("PAGE %d %s kids=%s", page, n.notation(), kids))
779 }
780 case "DUMP":
781 output = append(output, fmt.Sprintf("DUMP %d %s", t.degree, t.dump()))
782 case "HEIGHT":
783 output = append(output, fmt.Sprintf("HEIGHT %d", t.height()))
784 case "STATS":
785 leaves := 0
786 for _, n := range t.pages {
787 if len(n.children) == 0 {
788 leaves++
789 }
790 }
791 fill := 100 * len(entries) / ((2*t.degree - 1) * len(t.pages))
792 output = append(output, fmt.Sprintf("STATS t=%d nodes=%d leaves=%d keys=%d fill=%d%% splits=%d merges=%d borrows=%d", t.degree, len(t.pages), leaves, len(entries), fill, t.splits, t.merges, t.borrows))
793 case "VERIFY":
794 result := "VERIFY OK"
795 if problem := t.violation(); problem != "" {
796 result = "VERIFY FAIL " + problem
797 }
798 output = append(output, result)
799 case "REBUILD":
800 t = newTree(t.degree)
801 for _, e := range entries {
802 t.upsert(e, tr)
803 }
804 output = append(output, fmt.Sprintf("REBUILD entries=%d keys=%d nodes=%d splits=%d height=%d", len(entries), len(t.entries()), len(t.pages), t.splits, t.height()))
805 case "COMPACT":
806 nodes := len(t.pages)
807 t.compact()
808 output = append(output, fmt.Sprintf("COMPACT nodes=%d->%d height=%d", nodes, len(t.pages), t.height()))
809 case "SAVE":
810 w.saved = t.clone()
811 output = append(output, fmt.Sprintf("SAVE nodes=%d keys=%d %s", len(t.pages), len(entries), t.status()))
812 case "RESTORE":
813 t = w.saved.clone()
814 output = append(output, fmt.Sprintf("RESTORE nodes=%d keys=%d %s", len(t.pages), len(t.entries()), t.status()))
815 }
816 w.index = t
817 return output, nil
818}
