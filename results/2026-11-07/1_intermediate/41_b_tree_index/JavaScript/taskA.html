3class CommandError extends Error {}
4function fail(reason) { throw new CommandError(reason); }
5const entryText = entry => `${entry.key}=${entry.value}`;
6const emit = (trace, line) => { if (trace !== null) trace.push(line); };

8class Node {
9 constructor(page, entries = [], children = []) {
10 this.page = page; this.entries = entries; this.children = children;
11 }
12 get leaf() { return this.children.length === 0; }
13 notation() { return `[${this.entries.map(entry => entry.key).join(' ')}]`; }
14 position(key) {
15 const found = this.entries.findIndex(entry => key <= entry.key);
16 return found < 0 ? this.entries.length : found;
17 }
18 clone() { return new Node(this.page, this.entries.map(entry => ({ ...entry })), this.children.map(child => child.clone())); }
19 content() { return JSON.stringify([this.entries.map(entry => [entry.key, entry.value]), this.children.map(child => child.page)]); }
20 dump() {
21 const tokens = ['('];
22 if (!this.leaf) tokens.push(...this.children[0].dump());
23 for (let index = 0; index < this.entries.length; ++index) {
24 tokens.push(entryText(this.entries[index]));
25 if (!this.leaf) tokens.push(...this.children[index + 1].dump());
26 }
27 tokens.push(')');
28 return tokens;
29 }
30}

32class Tree {
33 constructor(degree) {
34 this.degree = degree; this.root = new Node(0);
35 this.splits = 0; this.merges = 0; this.borrows = 0;
36 }
37 clone() {
38 const copy = new Tree(this.degree);
39 copy.root = this.root.clone();
40 copy.splits = this.splits; copy.merges = this.merges; copy.borrows = this.borrows;
41 return copy;
42 }
43 nodes() {
44 const result = [];
45 const visit = node => { result.push(node); node.children.forEach(visit); };
46 visit(this.root);
47 return result;
48 }
49 entries() {
50 const result = [];
51 const visit = node => {
52 if (!node.leaf) visit(node.children[0]);
53 for (let index = 0; index < node.entries.length; ++index) {
54 result.push(node.entries[index]);
55 if (!node.leaf) visit(node.children[index + 1]);
56 }
57 };
58 visit(this.root);
59 return result;
60 }
61 height() {
62 let height = 0, node = this.root;
63 while (!node.leaf) { ++height; node = node.children[0]; }
64 return height;
65 }
66 snapshot() { return new Map(this.nodes().map(node => [node.page, node.content()])); }
67 changes(before) {
68 const after = this.snapshot();
69 return {
70 writes: [...after].filter(([page, content]) => before.get(page) !== content).length,
71 frees: [...before.keys()].filter(page => !after.has(page)).length,
72 };
73 }
74 allocate(entries = [], children = []) {
75 const used = new Set(this.nodes().map(node => node.page));
76 let page = 0;
77 while (used.has(page)) ++page;
78 return new Node(page, entries, children);
79 }
80 search(key) {
81 const path = [];
82 let node = this.root;
83 while (true) {
84 path.push(node.notation());
85 const index = node.position(key);
86 if (index < node.entries.length && node.entries[index].key === key) return { node, index, path };
87 if (node.leaf) return { node: null, index: -1, path };
88 node = node.children[index];
89 }
90 }
91 violation() {
92 let referenceDepth = null;
93 const visit = (node, location, depth, low, high) => {
94 const count = node.entries.length, minimum = node === this.root ? (node.leaf ? 0 : 1) : this.degree - 1;
95 const maximum = 2 * this.degree - 1, at = `at=${location} node=${node.notation()}`;
96 if (count < minimum || count > maximum) return `COUNT ${at} keys=${count} allowed=${minimum}..${maximum}`;
97 for (let index = 0; index + 1 < count; ++index) if (node.entries[index].key >= node.entries[index + 1].key) {
98 return `ORDER ${at} ${node.entries[index].key}>=${node.entries[index + 1].key}`;
99 }
100 for (const entry of node.entries) if (entry.key <= low || entry.key >= high) {
101 return `RANGE ${at} key=${entry.key} low=${low === -Infinity ? '-inf' : low} high=${high === Infinity ? '+inf' : high}`;
102 }
103 if (node.leaf) {
104 if (referenceDepth === null) referenceDepth = depth;
105 else if (depth !== referenceDepth) return `DEPTH ${at} depth=${depth} expected=${referenceDepth}`;
106 }
107 for (let index = 0; index < node.children.length; ++index) {
108 const problem = visit(node.children[index], `${location}.${index}`, depth + 1,
109 index === 0 ? low : node.entries[index - 1].key, index === count ? high : node.entries[index].key);
110 if (problem !== null) return problem;
111 }
112 return null;
113 };
114 return visit(this.root, 'root', 0, -Infinity, Infinity);
115 }
116 static load(degree, tokens) {
117 let position = 0, page = 0;
118 const unexpected = () => fail(`SYNTAX ${position + 1}`);
119 const entry = () => {
120 if (position === tokens.length || tokens[position] === '(' || tokens[position] === ')') unexpected();
121 const [key, value] = tokens[position++].split('=');
122 return { key: Number(key), value };
123 };
124 const parse = () => {
125 if (position === tokens.length || tokens[position] !== '(') unexpected();
126 ++position;
127 const node = new Node(page++);
128 if (tokens[position] === '(') {
129 node.children.push(parse());
130 while (tokens[position] !== ')') { node.entries.push(entry()); node.children.push(parse()); }
131 } else {
132 while (tokens[position] !== ')') node.entries.push(entry());
133 }
134 ++position;
135 return node;
136 };
137 const tree = new Tree(degree);
138 tree.root = parse();
139 if (position !== tokens.length) unexpected();
140 return tree;
141 }
142 split(parent, index, trace) {
143 const child = parent.children[index], before = child.notation(), middle = child.entries[this.degree - 1];
144 const right = this.allocate(child.entries.slice(this.degree), child.children.slice(this.degree));
145 child.entries = child.entries.slice(0, this.degree - 1);
146 child.children = child.children.slice(0, this.degree);
147 parent.entries.splice(index, 0, middle); parent.children.splice(index + 1, 0, right);
148 ++this.splits;
149 emit(trace, `STEP split page=${child.page} keys=${before} up=${middle.key} new=${right.page}`);
150 }
151 upsert(key, value, trace = null) {
152 const found = this.search(key);
153 if (found.node !== null) {
154 const entry = found.node.entries[found.index], old = entry.value;
155 entry.value = value;
156 return { inserted: false, old };
157 }
158 if (this.root.entries.length === 2 * this.degree - 1) {
159 this.root = this.allocate([], [this.root]);
160 emit(trace, `STEP grow page=${this.root.page}`);
161 this.split(this.root, 0, trace);
162 }
163 let node = this.root;
164 while (!node.leaf) {
165 let index = node.position(key);
166 if (node.children[index].entries.length === 2 * this.degree - 1) {
167 this.split(node, index, trace);
168 if (key > node.entries[index].key) ++index;
169 }
170 node = node.children[index];
171 }
172 node.entries.splice(node.position(key), 0, { key, value });
173 emit(trace, `STEP place page=${node.page} keys=${node.notation()}`);
174 return { inserted: true };
175 }
176 merge(parent, index, caseName, trace) {
177 const left = parent.children[index], right = parent.children[index + 1];
178 left.entries.push(parent.entries[index], ...right.entries);
179 left.children.push(...right.children);
180 parent.entries.splice(index, 1); parent.children.splice(index + 1, 1);
181 ++this.merges;
182 emit(trace, `STEP ${caseName} page=${parent.page} merge=${left.page}+${right.page} keys=${left.notation()}`);
183 if (parent === this.root && parent.entries.length === 0) {
184 this.root = left;
185 emit(trace, `STEP shrink page=${left.page}`);
186 }
187 return left;
188 }
189 remove(key, cases = [], trace = null, node = this.root) {
190 const index = node.position(key);
191 if (index < node.entries.length && node.entries[index].key === key) {
192 if (node.leaf) {
193 cases.push('1'); node.entries.splice(index, 1);
194 emit(trace, `STEP 1 page=${node.page} keys=${node.notation()}`);
195 return;
196 }
197 const left = node.children[index], right = node.children[index + 1];
198 if (left.entries.length >= this.degree) {
199 let last = left;
200 while (!last.leaf) last = last.children.at(-1);
201 const predecessor = { ...last.entries.at(-1) };
202 cases.push('2a'); emit(trace, `STEP 2a page=${node.page} pred=${predecessor.key}`);
203 this.remove(predecessor.key, cases, trace, left);
204 node.entries[index] = predecessor;
205 } else if (right.entries.length >= this.degree) {
206 let first = right;
207 while (!first.leaf) first = first.children[0];
208 const successor = { ...first.entries[0] };
209 cases.push('2b'); emit(trace, `STEP 2b page=${node.page} succ=${successor.key}`);
210 this.remove(successor.key, cases, trace, right);
211 node.entries[index] = successor;
212 } else {
213 cases.push('2c');
214 const child = this.merge(node, index, '2c', trace);
215 this.remove(key, cases, trace, child);
216 }
217 return;
218 }
219 let child = node.children[index];
220 if (child.entries.length === this.degree - 1) {
221 const left = node.children[index - 1], right = node.children[index + 1];
222 if (left !== undefined && left.entries.length >= this.degree) {
223 cases.push('3a-left');
224 child.entries.unshift(node.entries[index - 1]);
225 node.entries[index - 1] = left.entries.pop();
226 if (!left.leaf) child.children.unshift(left.children.pop());
227 ++this.borrows;
228 emit(trace, `STEP 3a-left page=${node.page} child=${child.page} from=${left.page} keys=${child.notation()}`);
229 } else if (right !== undefined && right.entries.length >= this.degree) {
230 cases.push('3a-right');
231 child.entries.push(node.entries[index]);
232 node.entries[index] = right.entries.shift();
233 if (!right.leaf) child.children.push(right.children.shift());
234 ++this.borrows;
235 emit(trace, `STEP 3a-right page=${node.page} child=${child.page} from=${right.page} keys=${child.notation()}`);
236 } else if (left !== undefined) {
237 cases.push('3b-left'); child = this.merge(node, index - 1, '3b-left', trace);
238 } else {
239 cases.push('3b-right'); child = this.merge(node, index, '3b-right', trace);
240 }
241 }
242 this.remove(key, cases, trace, child);
243 }
244 scan(low, high) {
245 let visited = 0;
246 const entries = [];
247 const visit = node => {
248 ++visited;
249 for (let index = 0; index <= node.entries.length; ++index) {
250 if (!node.leaf && (index === 0 || node.entries[index - 1].key < high)
251 && (index === node.entries.length || node.entries[index].key > low)) visit(node.children[index]);
252 if (index < node.entries.length && low <= node.entries[index].key && node.entries[index].key <= high) {
253 entries.push(node.entries[index]);
254 }
255 }
256 };
257 visit(this.root);
258 return { entries, visited };
259 }
260 compact() {
261 let entries = this.entries(), children = [];
262 while (true) {
263 const groups = Math.floor((entries.length + 2 * this.degree) / (2 * this.degree));
264 const keys = entries.length - groups + 1, size = Math.floor(keys / groups), extra = keys % groups;
265 const nodes = [], separators = [];
266 let entryIndex = 0, childIndex = 0;
267 for (let group = 0; group < groups; ++group) {
268 const count = size + (group < extra ? 1 : 0);
269 nodes.push(new Node(-1, entries.slice(entryIndex, entryIndex + count),
270 children.length === 0 ? [] : children.slice(childIndex, childIndex + count + 1)));
271 entryIndex += count; childIndex += count + 1;
272 if (group + 1 < groups) separators.push(entries[entryIndex++]);
273 }
274 if (groups === 1) { this.root = nodes[0]; break; }
275 entries = separators; children = nodes;
276 }
277 this.nodes().forEach((node, page) => { node.page = page; });
278 this.splits = 0; this.merges = 0; this.borrows = 0;
279 }

282const requiresValidTree = new Set([
283 'INSERT', 'BULKLOAD', 'DELETE', 'DELRANGE', 'RANGE', 'COUNT', 'MIN', 'MAX',
284 'FLOOR', 'CEIL', 'RANK', 'SELECT', 'SEEK', 'NEXT', 'PREV', 'HEIGHT', 'COMPACT',
285]);

287class Workbench {
288 constructor() {
289 this.tree = null; this.saved = null; this.cursor = null; this.trace = false;
290 }
291 summary(prefix) {
292 return `${prefix} nodes=${this.tree.nodes().length} keys=${this.tree.entries().length} ${this.tree.violation() === null ? 'OK' : 'DAMAGED'}`;
293 }
294 command(line) {
295 const [verb, ...args] = line.split(' '), first = Number(args[0]), second = Number(args[1]);
296 if (verb === 'TRACE') {
297 this.trace = args[0] === 'ON';
298 return [`TRACE ${args[0]}`];
299 }
300 if (verb === 'CREATE' || verb === 'LOAD') {
301 if (first < 2 || first > 5) fail(`BAD DEGREE ${first}`);
302 const replacement = verb === 'CREATE' ? new Tree(first) : Tree.load(first, args.slice(1));
303 this.tree = replacement; this.cursor = null;
304 return [verb === 'CREATE' ? `CREATE t=${first}` : this.summary(`LOAD t=${first}`)];
305 }
306 if (this.tree === null) fail('NO INDEX');
307 const tree = this.tree;
308 if (requiresValidTree.has(verb) && tree.violation() !== null) fail('DAMAGED');
309 const entries = tree.entries();
310 if (['DELRANGE', 'RANGE', 'COUNT'].includes(verb) && first > second) fail(`BAD RANGE ${first} ${second}`);
311 if (verb === 'SELECT' && (first < 1 || first > entries.length)) fail(`BAD POSITION ${first}`);
312 if ((verb === 'NEXT' || verb === 'PREV') && this.cursor === null) fail('NO CURSOR');
313 if (verb === 'RESTORE' && this.saved === null) fail('NO SAVE');
314 const output = [];
315 switch (verb) {
316 case 'INSERT': {
317 const before = tree.snapshot(), splits = tree.splits;
318 const result = tree.upsert(first, args[1], this.trace ? output : null);
319 const { writes } = tree.changes(before);
320 output.push(result.inserted
321 ? `INSERT ${first} NEW splits=${tree.splits - splits} height=${tree.height()} writes=${writes}`
322 : `INSERT ${first} UPDATED old=${result.old} writes=${writes}`);
323 break;
324 }
325 case 'BULKLOAD': {
326 const before = tree.snapshot(), splits = tree.splits;
327 let inserted = 0, updated = 0;
328 for (let index = 0; index < args.length; index += 2) {
329 if (tree.upsert(Number(args[index]), args[index + 1]).inserted) ++inserted;
330 else ++updated;
331 }
332 output.push(`BULKLOAD pairs=${args.length / 2} new=${inserted} updated=${updated} splits=${tree.splits - splits} height=${tree.height()} writes=${tree.changes(before).writes}`);
333 break;
334 }
335 case 'DELETE': {
336 const found = tree.search(first);
337 if (found.node === null) { output.push(`DELETE ${first} NOT FOUND`); break; }
338 const value = found.node.entries[found.index].value, before = tree.snapshot(), cases = [];
339 tree.remove(first, cases, this.trace ? output : null);
340 const { writes, frees } = tree.changes(before);
341 output.push(`DELETE ${first} REMOVED value=${value} cases=${cases.join(',')} height=${tree.height()} writes=${writes} frees=${frees}`);
342 break;
343 }
344 case 'DELRANGE': {
345 const keys = entries.filter(entry => first <= entry.key && entry.key <= second).map(entry => entry.key);
346 const before = tree.snapshot(), merges = tree.merges, borrows = tree.borrows;
347 for (const key of keys) tree.remove(key);
348 const { writes, frees } = tree.changes(before);
349 output.push(`DELRANGE ${first} ${second} removed=${keys.length} merges=${tree.merges - merges} borrows=${tree.borrows - borrows} height=${tree.height()} writes=${writes} frees=${frees}`);
350 break;
351 }
352 case 'GET': {
353 const { node, index, path } = tree.search(first);
354 const result = node === null ? 'NOT FOUND' : `FOUND value=${node.entries[index].value}`;
355 output.push(`GET ${first} ${result} path=${path.join(' -> ')}`);
356 break;
357 }
358 case 'RANGE': case 'COUNT': {
359 const result = tree.scan(first, second);
360 const suffix = verb === 'RANGE' ? result.entries.map(entry => ` ${entryText(entry)}`).join('') : '';
361 output.push(`${verb} ${first} ${second} count=${result.entries.length} visited=${result.visited}${suffix}`);
362 break;
363 }
364 case 'MIN': case 'MAX': {
365 const entry = verb === 'MIN' ? entries[0] : entries.at(-1);
366 output.push(`${verb} ${entry === undefined ? 'EMPTY' : entryText(entry)}`);
367 break;
368 }
369 case 'FLOOR': case 'CEIL': {
370 const entry = verb === 'FLOOR' ? entries.findLast(entry => entry.key <= first) : entries.find(entry => entry.key >= first);
371 output.push(`${verb} ${first} ${entry === undefined ? 'NONE' : entryText(entry)}`);
372 break;
373 }
374 case 'RANK':
375 output.push(`RANK ${first} ${entries.filter(entry => entry.key < first).length}`);
376 break;
377 case 'SELECT':
378 output.push(`SELECT ${first} ${entryText(entries[first - 1])}`);
379 break;
380 case 'SEEK': {
381 const entry = entries.find(entry => entry.key >= first);
382 this.cursor = entry === undefined ? 'END' : entry.key;
383 output.push(`SEEK ${first} ${entry === undefined ? 'END' : entryText(entry)}`);
384 break;
385 }
386 case 'NEXT': case 'PREV': {
387 let entry;
388 if (verb === 'NEXT') {
389 if (this.cursor === 'BEGIN') entry = entries[0];
390 else if (typeof this.cursor === 'number') entry = entries.find(entry => entry.key > this.cursor);
391 this.cursor = entry === undefined ? 'END' : entry.key;
392 } else {
393 if (this.cursor === 'END') entry = entries.at(-1);
394 else if (typeof this.cursor === 'number') entry = entries.findLast(entry => entry.key < this.cursor);
395 this.cursor = entry === undefined ? 'BEGIN' : entry.key;
396 }
397 output.push(`${verb} ${entry === undefined ? this.cursor : entryText(entry)}`);
398 break;
399 }
400 case 'PRINT': {
401 let level = [tree.root], depth = 0;
402 while (level.length > 0) {
403 output.push(`LEVEL ${depth++} ${level.map(node => node.notation()).join(' ')}`);
404 level = level.flatMap(node => node.children);
405 }
406 break;
407 }
408 case 'PAGES':
409 for (const node of tree.nodes().sort((left, right) => left.page - right.page)) {
410 output.push(`PAGE ${node.page} ${node.notation()} kids=${node.leaf ? '-' : node.children.map(child => child.page).join(',')}`);
411 }
412 break;
413 case 'DUMP':
414 output.push(`DUMP ${tree.degree} ${tree.root.dump().join(' ')}`);
415 break;
416 case 'HEIGHT':
417 output.push(`HEIGHT ${tree.height()}`);
418 break;
419 case 'STATS': {
420 const nodes = tree.nodes(), leaves = nodes.filter(node => node.leaf).length;
421 const fill = Math.floor(100 * entries.length / ((2 * tree.degree - 1) * nodes.length));
422 output.push(`STATS t=${tree.degree} nodes=${nodes.length} leaves=${leaves} keys=${entries.length} fill=${fill}% splits=${tree.splits} merges=${tree.merges} borrows=${tree.borrows}`);
423 break;
424 }
425 case 'VERIFY': {
426 const violation = tree.violation();
427 output.push(violation === null ? 'VERIFY OK' : `VERIFY FAIL ${violation}`);
428 break;
429 }
430 case 'REBUILD': {
431 this.tree = new Tree(tree.degree);
432 for (const entry of entries) this.tree.upsert(entry.key, entry.value);
433 output.push(`REBUILD entries=${entries.length} keys=${this.tree.entries().length} nodes=${this.tree.nodes().length} splits=${this.tree.splits} height=${this.tree.height()}`);
434 break;
435 }
436 case 'COMPACT': {
437 const nodes = tree.nodes().length;
438 tree.compact();
439 output.push(`COMPACT nodes=${nodes}->${tree.nodes().length} height=${tree.height()}`);
440 break;
441 }
442 case 'SAVE':
443 this.saved = tree.clone(); output.push(this.summary('SAVE'));
444 break;
445 case 'RESTORE':
446 this.tree = this.saved.clone(); output.push(this.summary('RESTORE'));
447 break;
448 }
449 return output;
450 }
451}

453const lines = readFileSync(0, 'utf8').trimEnd().split(/\r?\n/);
454const workbench = new Workbench(), output = [];
455for (let index = 1; index <= Number(lines[0]); ++index) {
456 try { output.push(...workbench.command(lines[index])); }
457 catch (error) {
458 if (!(error instanceof CommandError)) throw error;
459 output.push(`ERROR ${error.message}`);
460 }
461}
462process.stdout.write(output.join('\n') + '\n');
