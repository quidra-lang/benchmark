3interface Entry { readonly key: number; value: string }
4interface SearchResult { node: Node | null; index: number; path: string[] }
5interface Changes { writes: number; frees: number }
6interface ScanResult { entries: Entry[]; visited: number }
7type Insertion = { inserted: true } | { inserted: false; old: string };
8type MergeCase = '2c' | '3b-left' | '3b-right';
9type DeleteCase = '1' | '2a' | '2b' | '3a-left' | '3a-right' | MergeCase;
10type Trace = string[] | null;
11type Cursor = number | 'BEGIN' | 'END' | null;

13class CommandError extends Error {}
14function fail(reason: string): never { throw new CommandError(reason); }
15const entryText = (entry: Entry): string => `${entry.key}=${entry.value}`;
16const emit = (trace: Trace, line: string): void => { if (trace !== null) trace.push(line); };

18class Node {
19 page: number;
20 entries: Entry[];
21 children: Node[];
22 constructor(page: number, entries: Entry[] = [], children: Node[] = []) {
23 this.page = page; this.entries = entries; this.children = children;
24 }
25 get leaf(): boolean { return this.children.length === 0; }
26 notation(): string { return `[${this.entries.map(entry => entry.key).join(' ')}]`; }
27 position(key: number): number {
28 const found = this.entries.findIndex(entry => key <= entry.key);
29 return found < 0 ? this.entries.length : found;
30 }
31 clone(): Node { return new Node(this.page, this.entries.map(entry => ({ ...entry })), this.children.map(child => child.clone())); }
32 content(): string { return JSON.stringify([this.entries.map(entry => [entry.key, entry.value]), this.children.map(child => child.page)]); }
33 dump(): string[] {
34 const tokens = ['('];
35 if (!this.leaf) tokens.push(...this.children[0].dump());
36 for (let index = 0; index < this.entries.length; ++index) {
37 tokens.push(entryText(this.entries[index]));
38 if (!this.leaf) tokens.push(...this.children[index + 1].dump());
39 }
40 tokens.push(')');
41 return tokens;
42 }

45class Tree {
46 readonly degree: number;
47 root = new Node(0);
48 splits = 0;
49 merges = 0;
50 borrows = 0;
51 constructor(degree: number) { this.degree = degree; }
52 clone(): Tree {
53 const copy = new Tree(this.degree);
54 copy.root = this.root.clone();
55 copy.splits = this.splits; copy.merges = this.merges; copy.borrows = this.borrows;
56 return copy;
57 }
58 nodes(): Node[] {
59 const result: Node[] = [];
60 const visit = (node: Node): void => { result.push(node); node.children.forEach(visit); };
61 visit(this.root);
62 return result;
63 }
64 entries(): Entry[] {
65 const result: Entry[] = [];
66 const visit = (node: Node): void => {
67 if (!node.leaf) visit(node.children[0]);
68 for (let index = 0; index < node.entries.length; ++index) {
69 result.push(node.entries[index]);
70 if (!node.leaf) visit(node.children[index + 1]);
71 }
72 };
73 visit(this.root);
74 return result;
75 }
76 height(): number {
77 let height = 0, node = this.root;
78 while (!node.leaf) { ++height; node = node.children[0]; }
79 return height;
80 }
81 snapshot(): Map<number, string> { return new Map<number, string>(this.nodes().map(node => [node.page, node.content()])); }
82 changes(before: ReadonlyMap<number, string>): Changes {
83 const after = this.snapshot();
84 return {
85 writes: [...after].filter(([page, content]) => before.get(page) !== content).length,
86 frees: [...before.keys()].filter(page => !after.has(page)).length,
87 };
88 }
89 allocate(entries: Entry[] = [], children: Node[] = []): Node {
90 const used = new Set(this.nodes().map(node => node.page));
91 let page = 0;
92 while (used.has(page)) ++page;
93 return new Node(page, entries, children);
94 }
95 search(key: number): SearchResult {
96 const path: string[] = [];
97 let node = this.root;
98 while (true) {
99 path.push(node.notation());
100 const index = node.position(key);
101 if (index < node.entries.length && node.entries[index].key === key) return { node, index, path };
102 if (node.leaf) return { node: null, index: -1, path };
103 node = node.children[index];
104 }
105 }
106 violation(): string | null {
107 let referenceDepth: number | null = null;
108 const visit = (node: Node, location: string, depth: number, low: number, high: number): string | null => {
109 const count = node.entries.length, minimum = node === this.root ? (node.leaf ? 0 : 1) : this.degree - 1;
110 const maximum = 2 * this.degree - 1, at = `at=${location} node=${node.notation()}`;
111 if (count < minimum || count > maximum) return `COUNT ${at} keys=${count} allowed=${minimum}..${maximum}`;
112 for (let index = 0; index + 1 < count; ++index) if (node.entries[index].key >= node.entries[index + 1].key) {
113 return `ORDER ${at} ${node.entries[index].key}>=${node.entries[index + 1].key}`;
114 }
115 for (const entry of node.entries) if (entry.key <= low || entry.key >= high) {
116 return `RANGE ${at} key=${entry.key} low=${low === -Infinity ? '-inf' : low} high=${high === Infinity ? '+inf' : high}`;
117 }
118 if (node.leaf) {
119 if (referenceDepth === null) referenceDepth = depth;
120 else if (depth !== referenceDepth) return `DEPTH ${at} depth=${depth} expected=${referenceDepth}`;
121 }
122 for (let index = 0; index < node.children.length; ++index) {
123 const problem = visit(node.children[index], `${location}.${index}`, depth + 1,
124 index === 0 ? low : node.entries[index - 1].key, index === count ? high : node.entries[index].key);
125 if (problem !== null) return problem;
126 }
127 return null;
128 };
129 return visit(this.root, 'root', 0, -Infinity, Infinity);
130 }
131 static load(degree: number, tokens: readonly string[]): Tree {
132 let position = 0, page = 0;
133 const unexpected = (): never => fail(`SYNTAX ${position + 1}`);
134 const entry = (): Entry => {
135 if (position === tokens.length || tokens[position] === '(' || tokens[position] === ')') unexpected();
136 const [key, value] = tokens[position++].split('=');
137 return { key: Number(key), value };
138 };
139 const parse = (): Node => {
140 if (position === tokens.length || tokens[position] !== '(') unexpected();
141 ++position;
142 const node = new Node(page++);
143 if (tokens[position] === '(') {
144 node.children.push(parse());
145 while (tokens[position] !== ')') { node.entries.push(entry()); node.children.push(parse()); }
146 } else {
147 while (tokens[position] !== ')') node.entries.push(entry());
148 }
149 ++position;
150 return node;
151 };
152 const tree = new Tree(degree);
153 tree.root = parse();
154 if (position !== tokens.length) unexpected();
155 return tree;
156 }
157 split(parent: Node, index: number, trace: Trace): void {
158 const child = parent.children[index], before = child.notation(), middle = child.entries[this.degree - 1];
159 const right = this.allocate(child.entries.slice(this.degree), child.children.slice(this.degree));
160 child.entries = child.entries.slice(0, this.degree - 1);
161 child.children = child.children.slice(0, this.degree);
162 parent.entries.splice(index, 0, middle); parent.children.splice(index + 1, 0, right);
163 ++this.splits;
164 emit(trace, `STEP split page=${child.page} keys=${before} up=${middle.key} new=${right.page}`);
165 }
166 upsert(key: number, value: string, trace: Trace = null): Insertion {
167 const found = this.search(key);
168 if (found.node !== null) {
169 const entry = found.node.entries[found.index], old = entry.value;
170 entry.value = value;
171 return { inserted: false, old };
172 }
173 if (this.root.entries.length === 2 * this.degree - 1) {
174 this.root = this.allocate([], [this.root]);
175 emit(trace, `STEP grow page=${this.root.page}`);
176 this.split(this.root, 0, trace);
177 }
178 let node = this.root;
179 while (!node.leaf) {
180 let index = node.position(key);
181 if (node.children[index].entries.length === 2 * this.degree - 1) {
182 this.split(node, index, trace);
183 if (key > node.entries[index].key) ++index;
184 }
185 node = node.children[index];
186 }
187 node.entries.splice(node.position(key), 0, { key, value });
188 emit(trace, `STEP place page=${node.page} keys=${node.notation()}`);
189 return { inserted: true };
190 }
191 merge(parent: Node, index: number, caseName: MergeCase, trace: Trace): Node {
192 const left = parent.children[index], right = parent.children[index + 1];
193 left.entries.push(parent.entries[index], ...right.entries);
194 left.children.push(...right.children);
195 parent.entries.splice(index, 1); parent.children.splice(index + 1, 1);
196 ++this.merges;
197 emit(trace, `STEP ${caseName} page=${parent.page} merge=${left.page}+${right.page} keys=${left.notation()}`);
198 if (parent === this.root && parent.entries.length === 0) {
199 this.root = left;
200 emit(trace, `STEP shrink page=${left.page}`);
201 }
202 return left;
203 }
204 remove(key: number, cases: DeleteCase[] = [], trace: Trace = null, node: Node = this.root): void {
205 const index = node.position(key);
206 if (index < node.entries.length && node.entries[index].key === key) {
207 if (node.leaf) {
208 cases.push('1'); node.entries.splice(index, 1);
209 emit(trace, `STEP 1 page=${node.page} keys=${node.notation()}`);
210 return;
211 }
212 const left = node.children[index], right = node.children[index + 1];
213 if (left.entries.length >= this.degree) {
214 let last = left;
215 while (!last.leaf) last = last.children.at(-1)!;
216 const predecessor: Entry = { ...last.entries.at(-1)! };
217 cases.push('2a'); emit(trace, `STEP 2a page=${node.page} pred=${predecessor.key}`);
218 this.remove(predecessor.key, cases, trace, left);
219 node.entries[index] = predecessor;
220 } else if (right.entries.length >= this.degree) {
221 let first = right;
222 while (!first.leaf) first = first.children[0];
223 const successor = { ...first.entries[0] };
224 cases.push('2b'); emit(trace, `STEP 2b page=${node.page} succ=${successor.key}`);
225 this.remove(successor.key, cases, trace, right);
226 node.entries[index] = successor;
227 } else {
228 cases.push('2c');
229 const child = this.merge(node, index, '2c', trace);
230 this.remove(key, cases, trace, child);
231 }
232 return;
233 }
234 let child = node.children[index];
235 if (child.entries.length === this.degree - 1) {
236 const left = node.children[index - 1], right = node.children[index + 1];
237 if (left !== undefined && left.entries.length >= this.degree) {
238 cases.push('3a-left');
239 child.entries.unshift(node.entries[index - 1]);
240 node.entries[index - 1] = left.entries.pop()!;
241 if (!left.leaf) child.children.unshift(left.children.pop()!);
242 ++this.borrows;
243 emit(trace, `STEP 3a-left page=${node.page} child=${child.page} from=${left.page} keys=${child.notation()}`);
244 } else if (right !== undefined && right.entries.length >= this.degree) {
245 cases.push('3a-right');
246 child.entries.push(node.entries[index]);
247 node.entries[index] = right.entries.shift()!;
248 if (!right.leaf) child.children.push(right.children.shift()!);
249 ++this.borrows;
250 emit(trace, `STEP 3a-right page=${node.page} child=${child.page} from=${right.page} keys=${child.notation()}`);
251 } else if (left !== undefined) {
252 cases.push('3b-left'); child = this.merge(node, index - 1, '3b-left', trace);
253 } else {
254 cases.push('3b-right'); child = this.merge(node, index, '3b-right', trace);
255 }
256 }
257 this.remove(key, cases, trace, child);
258 }
259 scan(low: number, high: number): ScanResult {
260 let visited = 0;
261 const entries: Entry[] = [];
262 const visit = (node: Node): void => {
263 ++visited;
264 for (let index = 0; index <= node.entries.length; ++index) {
265 if (!node.leaf && (index === 0 || node.entries[index - 1].key < high)
266 && (index === node.entries.length || node.entries[index].key > low)) visit(node.children[index]);
267 if (index < node.entries.length && low <= node.entries[index].key && node.entries[index].key <= high) {
268 entries.push(node.entries[index]);
269 }
270 }
271 };
272 visit(this.root);
273 return { entries, visited };
274 }
275 compact(): void {
276 let entries = this.entries();
277 let children: Node[] = [];
278 while (true) {
279 const groups = Math.floor((entries.length + 2 * this.degree) / (2 * this.degree));
280 const keys = entries.length - groups + 1, size = Math.floor(keys / groups), extra = keys % groups;
281 const nodes: Node[] = [], separators: Entry[] = [];
282 let entryIndex = 0, childIndex = 0;
283 for (let group = 0; group < groups; ++group) {
284 const count = size + (group < extra ? 1 : 0);
285 nodes.push(new Node(-1, entries.slice(entryIndex, entryIndex + count),
286 children.length === 0 ? [] : children.slice(childIndex, childIndex + count + 1)));
287 entryIndex += count; childIndex += count + 1;
288 if (group + 1 < groups) separators.push(entries[entryIndex++]);
289 }
290 if (groups === 1) { this.root = nodes[0]; break; }
291 entries = separators; children = nodes;
292 }
293 this.nodes().forEach((node, page) => { node.page = page; });
294 this.splits = 0; this.merges = 0; this.borrows = 0;
295 }

298const requiresValidTree = new Set([
299 'INSERT', 'BULKLOAD', 'DELETE', 'DELRANGE', 'RANGE', 'COUNT', 'MIN', 'MAX',
300 'FLOOR', 'CEIL', 'RANK', 'SELECT', 'SEEK', 'NEXT', 'PREV', 'HEIGHT', 'COMPACT',
301]);

303class Workbench {
304 private tree: Tree | null = null;
305 private saved: Tree | null = null;
306 private cursor: Cursor = null;
307 private trace = false;
308 private summary(prefix: string, tree: Tree): string {
309 return `${prefix} nodes=${tree.nodes().length} keys=${tree.entries().length} ${tree.violation() === null ? 'OK' : 'DAMAGED'}`;
310 }
311 command(line: string): string[] {
312 const [verb, ...args] = line.split(' '), first = Number(args[0]), second = Number(args[1]);
313 if (verb === 'TRACE') {
314 this.trace = args[0] === 'ON';
315 return [`TRACE ${args[0]}`];
316 }
317 if (verb === 'CREATE' || verb === 'LOAD') {
318 if (first < 2 || first > 5) fail(`BAD DEGREE ${first}`);
319 const replacement = verb === 'CREATE' ? new Tree(first) : Tree.load(first, args.slice(1));
320 this.tree = replacement; this.cursor = null;
321 return [verb === 'CREATE' ? `CREATE t=${first}` : this.summary(`LOAD t=${first}`, replacement)];
322 }
323 if (this.tree === null) fail('NO INDEX');
324 const tree = this.tree;
325 if (requiresValidTree.has(verb) && tree.violation() !== null) fail('DAMAGED');
326 const entries = tree.entries();
327 if (['DELRANGE', 'RANGE', 'COUNT'].includes(verb) && first > second) fail(`BAD RANGE ${first} ${second}`);
328 if (verb === 'SELECT' && (first < 1 || first > entries.length)) fail(`BAD POSITION ${first}`);
329 if ((verb === 'NEXT' || verb === 'PREV') && this.cursor === null) fail('NO CURSOR');
330 const output: string[] = [];
331 switch (verb) {
332 case 'INSERT': {
333 const before = tree.snapshot(), splits = tree.splits;
334 const result = tree.upsert(first, args[1], this.trace ? output : null);
335 const { writes } = tree.changes(before);
336 output.push(result.inserted
337 ? `INSERT ${first} NEW splits=${tree.splits - splits} height=${tree.height()} writes=${writes}`
338 : `INSERT ${first} UPDATED old=${result.old} writes=${writes}`);
339 break;
340 }
341 case 'BULKLOAD': {
342 const before = tree.snapshot(), splits = tree.splits;
343 let inserted = 0, updated = 0;
344 for (let index = 0; index < args.length; index += 2) {
345 if (tree.upsert(Number(args[index]), args[index + 1]).inserted) ++inserted;
346 else ++updated;
347 }
348 output.push(`BULKLOAD pairs=${args.length / 2} new=${inserted} updated=${updated} splits=${tree.splits - splits} height=${tree.height()} writes=${tree.changes(before).writes}`);
349 break;
350 }
351 case 'DELETE': {
352 const found = tree.search(first);
353 if (found.node === null) { output.push(`DELETE ${first} NOT FOUND`); break; }
354 const value = found.node.entries[found.index].value, before = tree.snapshot();
355 const cases: DeleteCase[] = [];
356 tree.remove(first, cases, this.trace ? output : null);
357 const { writes, frees } = tree.changes(before);
358 output.push(`DELETE ${first} REMOVED value=${value} cases=${cases.join(',')} height=${tree.height()} writes=${writes} frees=${frees}`);
359 break;
360 }
361 case 'DELRANGE': {
362 const keys = entries.filter(entry => first <= entry.key && entry.key <= second).map(entry => entry.key);
363 const before = tree.snapshot(), merges = tree.merges, borrows = tree.borrows;
364 for (const key of keys) tree.remove(key);
365 const { writes, frees } = tree.changes(before);
366 output.push(`DELRANGE ${first} ${second} removed=${keys.length} merges=${tree.merges - merges} borrows=${tree.borrows - borrows} height=${tree.height()} writes=${writes} frees=${frees}`);
367 break;
368 }
369 case 'GET': {
370 const { node, index, path } = tree.search(first);
371 const result = node === null ? 'NOT FOUND' : `FOUND value=${node.entries[index].value}`;
372 output.push(`GET ${first} ${result} path=${path.join(' -> ')}`);
373 break;
374 }
375 case 'RANGE': case 'COUNT': {
376 const result = tree.scan(first, second);
377 const suffix = verb === 'RANGE' ? result.entries.map(entry => ` ${entryText(entry)}`).join('') : '';
378 output.push(`${verb} ${first} ${second} count=${result.entries.length} visited=${result.visited}${suffix}`);
379 break;
380 }
381 case 'MIN': case 'MAX': {
382 const entry = verb === 'MIN' ? entries[0] : entries.at(-1);
383 output.push(`${verb} ${entry === undefined ? 'EMPTY' : entryText(entry)}`);
384 break;
385 }
386 case 'FLOOR': case 'CEIL': {
387 const entry = verb === 'FLOOR' ? entries.findLast(entry => entry.key <= first) : entries.find(entry => entry.key >= first);
388 output.push(`${verb} ${first} ${entry === undefined ? 'NONE' : entryText(entry)}`);
389 break;
390 }
391 case 'RANK':
392 output.push(`RANK ${first} ${entries.filter(entry => entry.key < first).length}`);
393 break;
394 case 'SELECT':
395 output.push(`SELECT ${first} ${entryText(entries[first - 1])}`);
396 break;
397 case 'SEEK': {
398 const entry = entries.find(entry => entry.key >= first);
399 this.cursor = entry === undefined ? 'END' : entry.key;
400 output.push(`SEEK ${first} ${entry === undefined ? 'END' : entryText(entry)}`);
401 break;
402 }
403 case 'NEXT': case 'PREV': {
404 let entry: Entry | undefined;
405 const cursor = this.cursor;
406 if (verb === 'NEXT') {
407 if (cursor === 'BEGIN') entry = entries[0];
408 else if (typeof cursor === 'number') entry = entries.find(entry => entry.key > cursor);
409 this.cursor = entry === undefined ? 'END' : entry.key;
410 } else {
411 if (cursor === 'END') entry = entries.at(-1);
412 else if (typeof cursor === 'number') entry = entries.findLast(entry => entry.key < cursor);
413 this.cursor = entry === undefined ? 'BEGIN' : entry.key;
414 }
415 output.push(`${verb} ${entry === undefined ? this.cursor : entryText(entry)}`);
416 break;
417 }
418 case 'PRINT': {
419 let level = [tree.root], depth = 0;
420 while (level.length > 0) {
421 output.push(`LEVEL ${depth++} ${level.map(node => node.notation()).join(' ')}`);
422 level = level.flatMap(node => node.children);
423 }
424 break;
425 }
426 case 'PAGES':
427 for (const node of tree.nodes().sort((left, right) => left.page - right.page)) {
428 output.push(`PAGE ${node.page} ${node.notation()} kids=${node.leaf ? '-' : node.children.map(child => child.page).join(',')}`);
429 }
430 break;
431 case 'DUMP':
432 output.push(`DUMP ${tree.degree} ${tree.root.dump().join(' ')}`);
433 break;
434 case 'HEIGHT':
435 output.push(`HEIGHT ${tree.height()}`);
436 break;
437 case 'STATS': {
438 const nodes = tree.nodes(), leaves = nodes.filter(node => node.leaf).length;
439 const fill = Math.floor(100 * entries.length / ((2 * tree.degree - 1) * nodes.length));
440 output.push(`STATS t=${tree.degree} nodes=${nodes.length} leaves=${leaves} keys=${entries.length} fill=${fill}% splits=${tree.splits} merges=${tree.merges} borrows=${tree.borrows}`);
441 break;
442 }
443 case 'VERIFY': {
444 const violation = tree.violation();
445 output.push(violation === null ? 'VERIFY OK' : `VERIFY FAIL ${violation}`);
446 break;
447 }
448 case 'REBUILD': {
449 this.tree = new Tree(tree.degree);
450 for (const entry of entries) this.tree.upsert(entry.key, entry.value);
451 output.push(`REBUILD entries=${entries.length} keys=${this.tree.entries().length} nodes=${this.tree.nodes().length} splits=${this.tree.splits} height=${this.tree.height()}`);
452 break;
453 }
454 case 'COMPACT': {
455 const nodes = tree.nodes().length;
456 tree.compact();
457 output.push(`COMPACT nodes=${nodes}->${tree.nodes().length} height=${tree.height()}`);
458 break;
459 }
460 case 'SAVE':
461 this.saved = tree.clone(); output.push(this.summary('SAVE', tree));
462 break;
463 case 'RESTORE': {
464 if (this.saved === null) fail('NO SAVE');
465 this.tree = this.saved.clone(); output.push(this.summary('RESTORE', this.tree));
466 break;
467 }
468 }
469 return output;
470 }
471}

474const workbench = new Workbench();
475const output: string[] = [];
476for (let index = 1; index <= Number(lines[0]); ++index) {
477 try { output.push(...workbench.command(lines[index])); }
478 catch (error) {
479 if (!(error instanceof CommandError)) throw error;
480 output.push(`ERROR ${error.message}`);
