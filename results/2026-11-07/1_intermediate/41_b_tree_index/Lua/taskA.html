30function Node:same_content(other)
31 if #self.entries ~= #other.entries or not same_values(self.children, other.children) then return false end
32 for i, entry in ipairs(self.entries) do
33 if entry.key ~= other.entries[i].key or entry.value ~= other.entries[i].value then return false end
34 end
35 return true
36end

75function Tree.load(degree, tokens)
76 local tree, position = Tree.new(degree, false), 1
77 local function read_entry()
78 local token = tokens[position]
79 if token == nil or token == "(" or token == ")" then fail("SYNTAX %d", position) end
80 local key, value = token:match("^(%-?%d+)=(.+)$")
81 position = position + 1
82 return { key = tonumber(key), value = value }
83 end
84 local function read_node()
85 if tokens[position] ~= "(" then fail("SYNTAX %d", position) end
86 position = position + 1
87 local node = Node.new()
88 local page = tree:new_page(node)
89 if tokens[position] == "(" then
90 node.children[#node.children + 1] = read_node()
91 while tokens[position] ~= ")" do
92 node.entries[#node.entries + 1] = read_entry()
93 node.children[#node.children + 1] = read_node()
94 end
95 else
96 while tokens[position] ~= ")" do node.entries[#node.entries + 1] = read_entry() end
97 end
98 position = position + 1
99 return page
100 end
101 tree.root = read_node()
102 if position ~= #tokens + 1 then fail("SYNTAX %d", position) end
103 return tree
104end
105function Tree:entries()
106 local result = {}
107 local function visit(page)
108 local node = self.pages[page]
109 if not node:leaf() then visit(node.children[1]) end
110 for i, entry in ipairs(node.entries) do
111 result[#result + 1] = entry
112 if not node:leaf() then visit(node.children[i + 1]) end
113 end
114 end
115 visit(self.root)
116 return result
117end
118function Tree:height()
119 local page, depth = self.root, 0
120 while not self.pages[page]:leaf() do
121 page, depth = self.pages[page].children[1], depth + 1
122 end
123 return depth
124end
125function Tree:search(key)
126 local page, path = self.root, {}
127 while true do
128 local node = self.pages[page]
129 path[#path + 1] = tostring(node)
130 local position = node:position(key)
131 if position <= #node.entries and node.entries[position].key == key then
132 return { page = page, position = position, path = path }
133 end
134 if node:leaf() then return { path = path } end
135 page = node.children[position]
136 end
137end
138function Tree:violation()
139 local reference
140 local function visit(page, location, depth, low, high)
141 local node = self.pages[page]
142 local count = #node.entries
143 local minimum = page == self.root and (node:leaf() and 0 or 1) or self.degree - 1
144 local maximum = 2 * self.degree - 1
145 if count < minimum or count > maximum then
146 return string.format("COUNT at=%s node=%s keys=%d allowed=%d..%d", location, node, count, minimum, maximum)
147 end
148 for i = 1, count - 1 do
149 local left, right = node.entries[i].key, node.entries[i + 1].key
150 if left >= right then return string.format("ORDER at=%s node=%s %d>=%d", location, node, left, right) end
151 end
152 for _, entry in ipairs(node.entries) do
153 if (low ~= nil and entry.key <= low) or (high ~= nil and entry.key >= high) then
154 return string.format("RANGE at=%s node=%s key=%d low=%s high=%s", location, node, entry.key,
155 low == nil and "-inf" or low, high == nil and "+inf" or high)
156 end
157 end
158 if node:leaf() then
159 if reference == nil then reference = depth
160 elseif depth ~= reference then
161 return string.format("DEPTH at=%s node=%s depth=%d expected=%d", location, node, depth, reference)
162 end
163 end
164 for i, child in ipairs(node.children) do
165 local child_low, child_high = low, high
166 if i > 1 then child_low = node.entries[i - 1].key end
167 if i <= count then child_high = node.entries[i].key end
168 local problem = visit(child, location .. "." .. (i - 1), depth + 1, child_low, child_high)
169 if problem ~= nil then return problem end
170 end
171 return nil
172 end
173 return visit(self.root, "root", 0, nil, nil)
174end
175function Tree:changes(before)
176 local writes, frees = 0, 0
177 for page, node in pairs(self.pages) do
178 if before.pages[page] == nil or not node:same_content(before.pages[page]) then writes = writes + 1 end
179 end
180 for page in pairs(before.pages) do if self.pages[page] == nil then frees = frees + 1 end end
181 return writes, frees
182end
183function Tree:split_child(parent_page, position, trace)
184 local parent = self.pages[parent_page]
185 local child_page = parent.children[position]
186 local child = self.pages[child_page]
187 local middle = child.entries[self.degree]
188 local right = Node.new(slice(child.entries, self.degree + 1),
189 child:leaf() and {} or slice(child.children, self.degree + 1))
190 local right_page = self:new_page(right)
191 trace:emit("STEP split page=%d keys=%s up=%d new=%d", child_page, child, middle.key, right_page)
192 child.entries = slice(child.entries, 1, self.degree - 1)
193 if not child:leaf() then child.children = slice(child.children, 1, self.degree) end
194 table.insert(parent.entries, position, middle)
195 table.insert(parent.children, position + 1, right_page)
196 self.splits = self.splits + 1
197end
198function Tree:upsert(key, value, trace)
199 local found = self:search(key)
200 if found.page ~= nil then
201 local node = self.pages[found.page]
202 local old = node.entries[found.position].value
203 node.entries[found.position] = { key = key, value = value }
204 return old
205 end
206 if #self.pages[self.root].entries == 2 * self.degree - 1 then
207 local previous = self.root
208 self.root = self:new_page(Node.new({}, { previous }))
209 trace:emit("STEP grow page=%d", self.root)
210 self:split_child(self.root, 1, trace)
211 end
212 local page = self.root
213 while not self.pages[page]:leaf() do
214 local node = self.pages[page]
215 local position = node:position(key)
216 if #self.pages[node.children[position]].entries == 2 * self.degree - 1 then
217 self:split_child(page, position, trace)
218 if key > node.entries[position].key then position = position + 1 end
219 end
220 page = node.children[position]
221 end
222 local node = self.pages[page]
223 table.insert(node.entries, node:position(key), { key = key, value = value })
224 trace:emit("STEP place page=%d keys=%s", page, node)
225 return nil
226end
227function Tree:merge(parent_page, position, case, trace)
228 local parent = self.pages[parent_page]
229 local left_page, right_page = parent.children[position], parent.children[position + 1]
230 local left, right = self.pages[left_page], self.pages[right_page]
231 left.entries[#left.entries + 1] = parent.entries[position]
232 append(left.entries, right.entries)
233 append(left.children, right.children)
234 table.remove(parent.entries, position)
235 table.remove(parent.children, position + 1)
236 self.pages[right_page] = nil
237 self.merges = self.merges + 1
238 trace:emit("STEP %s page=%d merge=%d+%d keys=%s", case, parent_page, left_page, right_page, left)
239 if parent_page == self.root and #parent.entries == 0 then
240 self.root = left_page
241 self.pages[parent_page] = nil
242 trace:emit("STEP shrink page=%d", left_page)
243 end
244 return left_page
245end
246function Tree:remove_at(page, key, trace)
247 local node = self.pages[page]
248 local position = node:position(key)
249 if position <= #node.entries and node.entries[position].key == key then
250 if node:leaf() then
251 trace:record("1")
252 table.remove(node.entries, position)
253 trace:emit("STEP 1 page=%d keys=%s", page, node)
254 return
255 end
256 local left, right = node.children[position], node.children[position + 1]
257 if #self.pages[left].entries >= self.degree then
258 local last = left
259 while not self.pages[last]:leaf() do
260 local children = self.pages[last].children
261 last = children[#children]
262 end
263 local rows = self.pages[last].entries
264 local predecessor = rows[#rows]
265 trace:record("2a")
266 trace:emit("STEP 2a page=%d pred=%d", page, predecessor.key)
267 self:remove_at(left, predecessor.key, trace)
268 node.entries[position] = predecessor
269 elseif #self.pages[right].entries >= self.degree then
270 local first = right
271 while not self.pages[first]:leaf() do first = self.pages[first].children[1] end
272 local successor = self.pages[first].entries[1]
273 trace:record("2b")
274 trace:emit("STEP 2b page=%d succ=%d", page, successor.key)
275 self:remove_at(right, successor.key, trace)
276 node.entries[position] = successor
277 else
278 trace:record("2c")
279 local child = self:merge(page, position, "2c", trace)
280 self:remove_at(child, key, trace)
281 end
282 return
283 end
284 local child_page = node.children[position]
285 local child = self.pages[child_page]
286 if #child.entries == self.degree - 1 then
287 local left = position > 1 and node.children[position - 1] or nil
288 local right = position <= #node.entries and node.children[position + 1] or nil
289 if left ~= nil and #self.pages[left].entries >= self.degree then
290 local sibling = self.pages[left]
291 trace:record("3a-left")
292 table.insert(child.entries, 1, node.entries[position - 1])
293 node.entries[position - 1] = table.remove(sibling.entries)
294 if not sibling:leaf() then table.insert(child.children, 1, table.remove(sibling.children)) end
295 self.borrows = self.borrows + 1
296 trace:emit("STEP 3a-left page=%d child=%d from=%d keys=%s", page, child_page, left, child)
297 elseif right ~= nil and #self.pages[right].entries >= self.degree then
298 local sibling = self.pages[right]
299 trace:record("3a-right")
300 child.entries[#child.entries + 1] = node.entries[position]
301 node.entries[position] = table.remove(sibling.entries, 1)
302 if not sibling:leaf() then child.children[#child.children + 1] = table.remove(sibling.children, 1) end
303 self.borrows = self.borrows + 1
304 trace:emit("STEP 3a-right page=%d child=%d from=%d keys=%s", page, child_page, right, child)
305 elseif left ~= nil then
306 trace:record("3b-left")
307 child_page = self:merge(page, position - 1, "3b-left", trace)
308 else
309 trace:record("3b-right")
310 child_page = self:merge(page, position, "3b-right", trace)
311 end
312 end
313 self:remove_at(child_page, key, trace)
314end
315function Tree:remove(key, trace)
316 trace.cases = {}
317 self:remove_at(self.root, key, trace)
318 return trace.cases
319end
320function Tree:scan(low, high)
321 local result, visited = {}, 0
322 local function visit(page)
323 visited = visited + 1
324 local node = self.pages[page]
325 for i = 1, #node.entries + 1 do
326 if not node:leaf() and (i == 1 or node.entries[i - 1].key < high)
327 and (i == #node.entries + 1 or node.entries[i].key > low) then visit(node.children[i]) end
328 if i <= #node.entries and low <= node.entries[i].key and node.entries[i].key <= high then
329 result[#result + 1] = node.entries[i]
330 end
331 end
332 end
333 visit(self.root)
334 return result, visited
335end
336function Tree:dump()
337 local tokens = {}
338 local function visit(page)
339 local node = self.pages[page]
340 tokens[#tokens + 1] = "("
341 if not node:leaf() then visit(node.children[1]) end
342 for i, entry in ipairs(node.entries) do
343 tokens[#tokens + 1] = display(entry)
344 if not node:leaf() then visit(node.children[i + 1]) end
345 end
346 tokens[#tokens + 1] = ")"
347 end
348 visit(self.root)
349 return table.concat(tokens, " ")
350end
351function Tree:compact()
352 local source, children, built = self:entries(), {}, {}
353 local old_root
354 while true do
355 local groups = (#source + 2 * self.degree) // (2 * self.degree)
356 local keys = #source - (groups - 1)
357 local size, extra = keys // groups, keys % groups
358 local entry_index, child_index = 1, 1
359 local nodes, separators = {}, {}
360 for group = 1, groups do
361 local count = size + (group <= extra and 1 or 0)
362 nodes[#nodes + 1] = #built + 1
363 built[#built + 1] = Node.new(slice(source, entry_index, entry_index + count - 1),
364 #children == 0 and {} or slice(children, child_index, child_index + count))
365 entry_index, child_index = entry_index + count, child_index + count + 1
366 if group < groups then
367 separators[#separators + 1] = source[entry_index]
368 entry_index = entry_index + 1
369 end
370 end
371 if groups == 1 then old_root = nodes[1]; break end
372 source, children = separators, nodes
373 end
374 local replacement, next_page = {}, 0
375 local function renumber(old_page)
376 local page = next_page
377 next_page = next_page + 1
378 local old, kids = built[old_page], {}
379 for _, child in ipairs(old.children) do kids[#kids + 1] = renumber(child) end
380 replacement[page] = Node.new(slice(old.entries), kids)
381 return page
382 end
383 self.root = renumber(old_root)
384 self.pages = replacement
385 self.splits, self.merges, self.borrows = 0, 0, 0
386end

408local Workbench = {}
409Workbench.__index = Workbench
410function Workbench.new() return setmetatable({ cursor = "UNSET", trace = false }, Workbench) end
411function Workbench:process(source)
412 local words = {}
413 for word in source:gmatch("%S+") do words[#words + 1] = word end
414 local command = words[1]
415 if command == "TRACE" then
416 self.trace = words[2] == "ON"
417 return { "TRACE " .. words[2] }
418 end
419 local first = words[2] and tonumber(words[2]) or 0
420 local ranged = command == "DELRANGE" or command == "RANGE" or command == "COUNT"
421 local second = ranged and tonumber(words[3]) or 0
422 if command == "CREATE" or command == "LOAD" then
423 if first < 2 or first > 5 then fail("BAD DEGREE %d", first) end
424 local tree = command == "CREATE" and Tree.new(first) or Tree.load(first, slice(words, 3))
425 self.index, self.cursor = tree, "UNSET"
426 return { command == "CREATE" and ("CREATE t=" .. first) or summary("LOAD t=" .. first, tree) }
427 end
428 local tree = self.index
429 if tree == nil then fail("NO INDEX") end
430 if needs_valid[command] and tree:violation() ~= nil then fail("DAMAGED") end
431 local entries = tree:entries()
432 if ranged and first > second then fail("BAD RANGE %d %d", first, second) end
433 if command == "SELECT" and (first < 1 or first > #entries) then fail("BAD POSITION %d", first) end
434 if (command == "NEXT" or command == "PREV") and self.cursor == "UNSET" then fail("NO CURSOR") end
435 if command == "RESTORE" and self.saved == nil then fail("NO SAVE") end
436 local output = {}
437 local steps = Trace.new(self.trace and (command == "INSERT" or command == "DELETE"))
438 if command == "INSERT" then
439 local before, splits = tree:copy(), tree.splits
440 local old = tree:upsert(first, words[3], steps)
441 local writes = tree:changes(before)
442 output = steps.lines
443 output[#output + 1] = old ~= nil
444 and string.format("INSERT %d UPDATED old=%s writes=%d", first, old, writes)
445 or string.format("INSERT %d NEW splits=%d height=%d writes=%d", first, tree.splits - splits, tree:height(), writes)
446 elseif command == "BULKLOAD" then
447 local before, splits = tree:copy(), tree.splits
448 local inserted, updated = 0, 0
449 for i = 2, #words, 2 do
450 if tree:upsert(tonumber(words[i]), words[i + 1], steps) == nil then inserted = inserted + 1
451 else updated = updated + 1 end
452 end
453 local writes = tree:changes(before)
454 output[1] = string.format("BULKLOAD pairs=%d new=%d updated=%d splits=%d height=%d writes=%d",
455 (#words - 1) // 2, inserted, updated, tree.splits - splits, tree:height(), writes)
456 elseif command == "DELETE" then
457 local found = tree:search(first)
458 if found.page == nil then output[1] = string.format("DELETE %d NOT FOUND", first)
459 else
460 local value, before = tree.pages[found.page].entries[found.position].value, tree:copy()
461 local cases = tree:remove(first, steps)
462 local writes, frees = tree:changes(before)
463 output = steps.lines
464 output[#output + 1] = string.format("DELETE %d REMOVED value=%s cases=%s height=%d writes=%d frees=%d",
465 first, value, table.concat(cases, ","), tree:height(), writes, frees)
466 end
467 elseif command == "DELRANGE" then
468 local before, merges, borrows = tree:copy(), tree.merges, tree.borrows
469 local removed = 0
470 for _, entry in ipairs(entries) do
471 if first <= entry.key and entry.key <= second then
472 tree:remove(entry.key, steps)
473 removed = removed + 1
474 end
475 end
476 local writes, frees = tree:changes(before)
477 output[1] = string.format("DELRANGE %d %d removed=%d merges=%d borrows=%d height=%d writes=%d frees=%d",
478 first, second, removed, tree.merges - merges, tree.borrows - borrows, tree:height(), writes, frees)
479 elseif command == "GET" then
480 local found = tree:search(first)
481 local result = found.page ~= nil and ("FOUND value=" .. tree.pages[found.page].entries[found.position].value) or "NOT FOUND"
482 output[1] = string.format("GET %d %s path=%s", first, result, table.concat(found.path, " -> "))
483 elseif command == "RANGE" or command == "COUNT" then
484 local rows, visited = tree:scan(first, second)
485 local text = string.format("%s %d %d count=%d visited=%d", command, first, second, #rows, visited)
486 if command == "RANGE" then for _, entry in ipairs(rows) do text = text .. " " .. display(entry) end end
487 output[1] = text
488 elseif command == "MIN" or command == "MAX" then
489 output[1] = command .. " " .. (#entries == 0 and "EMPTY" or display(entries[command == "MIN" and 1 or #entries]))
490 elseif command == "FLOOR" or command == "CEIL" then
491 local entry
492 if command == "FLOOR" then entry = find(entries, function(value) return value.key <= first end, true)
493 else entry = find(entries, function(value) return value.key >= first end) end
494 output[1] = string.format("%s %d %s", command, first, entry and display(entry) or "NONE")
495 elseif command == "RANK" then
496 local rank = 0
497 for _, entry in ipairs(entries) do if entry.key < first then rank = rank + 1 end end
498 output[1] = string.format("RANK %d %d", first, rank)
499 elseif command == "SELECT" then
500 output[1] = string.format("SELECT %d %s", first, display(entries[first]))
501 elseif command == "SEEK" then
502 local entry = find(entries, function(value) return value.key >= first end)
503 self.cursor = entry and entry.key or "END"
504 output[1] = string.format("SEEK %d %s", first, entry and display(entry) or "END")
505 elseif command == "NEXT" or command == "PREV" then
506 local entry
507 if command == "NEXT" then
508 if self.cursor == "BEGIN" then entry = entries[1]
509 elseif type(self.cursor) == "number" then entry = find(entries, function(value) return value.key > self.cursor end) end
510 else
511 if self.cursor == "END" then entry = entries[#entries]
512 elseif type(self.cursor) == "number" then entry = find(entries, function(value) return value.key < self.cursor end, true) end
513 end
514 local marker = command == "NEXT" and "END" or "BEGIN"
515 self.cursor = entry and entry.key or marker
516 output[1] = command .. " " .. (entry and display(entry) or marker)
517 elseif command == "PRINT" then
518 local level, depth = { tree.root }, 0
519 while #level > 0 do
520 local notations, next_level = {}, {}
521 for _, page in ipairs(level) do
522 notations[#notations + 1] = tostring(tree.pages[page])
523 append(next_level, tree.pages[page].children)
524 end
525 output[#output + 1] = string.format("LEVEL %d %s", depth, table.concat(notations, " "))
526 level, depth = next_level, depth + 1
527 end
528 elseif command == "PAGES" then
529 local pages = {}
530 for page in pairs(tree.pages) do pages[#pages + 1] = page end
531 table.sort(pages)
532 for _, page in ipairs(pages) do
533 local node = tree.pages[page]
534 output[#output + 1] = string.format("PAGE %d %s kids=%s", page, node,
535 node:leaf() and "-" or table.concat(node.children, ","))
536 end
537 elseif command == "DUMP" then output[1] = string.format("DUMP %d %s", tree.degree, tree:dump())
538 elseif command == "HEIGHT" then output[1] = "HEIGHT " .. tree:height()
539 elseif command == "STATS" then
540 local leaves, nodes = 0, tree:node_count()
541 for _, node in pairs(tree.pages) do if node:leaf() then leaves = leaves + 1 end end
542 local fill = 100 * #entries // ((2 * tree.degree - 1) * nodes)
543 output[1] = string.format("STATS t=%d nodes=%d leaves=%d keys=%d fill=%d%% splits=%d merges=%d borrows=%d",
544 tree.degree, nodes, leaves, #entries, fill, tree.splits, tree.merges, tree.borrows)
545 elseif command == "VERIFY" then
546 local problem = tree:violation()
547 output[1] = problem and ("VERIFY FAIL " .. problem) or "VERIFY OK"
548 elseif command == "REBUILD" then
549 tree = Tree.new(tree.degree)
550 for _, entry in ipairs(entries) do tree:upsert(entry.key, entry.value, steps) end
551 output[1] = string.format("REBUILD entries=%d keys=%d nodes=%d splits=%d height=%d",
552 #entries, #tree:entries(), tree:node_count(), tree.splits, tree:height())
553 elseif command == "COMPACT" then
554 local nodes = tree:node_count()
555 tree:compact()
