7readonly record struct Entry(int Key, string Value)

9 public override string ToString() => $"{Key}={Value}";

13sealed class Node
14{
15 public List<Entry> Entries { get; } = [];
16 public List<int> Children { get; } = [];
17 public bool Leaf => Children.Count == 0;
18 public string Notation => "[" + string.Join(" ", Entries.Select(entry => entry.Key)) + "]";
19 public int Position(int key)
20 {
21 int position = 0;
22 while (position < Entries.Count && key > Entries[position].Key) position++;
23 return position;
24 }
25 public Node Copy()
26 {
27 Node node = new();
28 node.Entries.AddRange(Entries); node.Children.AddRange(Children);
29 return node;
30 }
31 public bool SameContent(Node other) => Entries.SequenceEqual(other.Entries) && Children.SequenceEqual(other.Children);
32}

40sealed class Tree(int degree)
41{
42 public int Degree { get; } = degree;
43 public int Root { get; private set; }
44 public Dictionary<int, Node> Pages { get; private set; } = new() { [0] = new Node() };
45 public int Splits { get; private set; }
46 public int Merges { get; private set; }
47 public int Borrows { get; private set; }
48
49 public Dictionary<int, Node> CopyPages() => Pages.ToDictionary(pair => pair.Key, pair => pair.Value.Copy());
50 public Tree Copy() => new(Degree) { Root = Root, Pages = CopyPages(), Splits = Splits, Merges = Merges, Borrows = Borrows };
51 public (int Writes, int Frees) Changes(Dictionary<int, Node> before)
52 {
53 int writes = Pages.Count(pair => !before.TryGetValue(pair.Key, out Node? old) || !pair.Value.SameContent(old));
54 int frees = before.Keys.Count(page => !Pages.ContainsKey(page));
55 return (writes, frees);
56 }
57 private int Allocate(Node node)
58 {
59 int page = 0;
60 while (Pages.ContainsKey(page)) page++;
61 Pages.Add(page, node);
62 return page;
63 }
64 public int Height
65 {
66 get
67 {
68 int height = 0, page = Root;
69 while (!Pages[page].Leaf) { page = Pages[page].Children[0]; height++; }
70 return height;
71 }
72 }
73 public List<Entry> Entries()
74 {
75 List<Entry> result = [];
76 void Visit(int page)
77 {
78 Node node = Pages[page];
79 if (!node.Leaf) Visit(node.Children[0]);
80 for (int i = 0; i < node.Entries.Count; i++)
81 {
82 result.Add(node.Entries[i]);
83 if (!node.Leaf) Visit(node.Children[i + 1]);
84 }
85 }
86 Visit(Root);
87 return result;
88 }
89 public SearchResult Search(int key)
90 {
91 int page = Root;
92 List<string> path = [];
93 while (true)
94 {
95 Node node = Pages[page];
96 int position = node.Position(key);
97 path.Add(node.Notation);
98 if (position < node.Entries.Count && node.Entries[position].Key == key) return new(page, position, path);
99 if (node.Leaf) return new(null, -1, path);
100 page = node.Children[position];
101 }
102 }
103 public string? Violation()
104 {
105 int? referenceDepth = null;
106 string? Visit(int page, string location, int depth, int? low, int? high)
107 {
108 Node node = Pages[page];
109 int count = node.Entries.Count, minimum = page == Root ? (node.Leaf ? 0 : 1) : Degree - 1, maximum = 2 * Degree - 1;
110 string at = $"at={location} node={node.Notation}";
111 if (count < minimum || count > maximum) return $"COUNT {at} keys={count} allowed={minimum}..{maximum}";
112 for (int i = 0; i + 1 < count; i++)
113 if (node.Entries[i].Key >= node.Entries[i + 1].Key) return $"ORDER {at} {node.Entries[i].Key}>={node.Entries[i + 1].Key}";
114 foreach (Entry entry in node.Entries)
115 if ((low.HasValue && entry.Key <= low.Value) || (high.HasValue && entry.Key >= high.Value))
116 return $"RANGE {at} key={entry.Key} low={low?.ToString() ?? "-inf"} high={high?.ToString() ?? "+inf"}";
117 if (node.Leaf)
118 {
119 if (referenceDepth is int expected && depth != expected) return $"DEPTH {at} depth={depth} expected={expected}";
120 referenceDepth ??= depth;
121 }
122 for (int i = 0; i < node.Children.Count; i++)
123 {
124 string? problem = Visit(node.Children[i], $"{location}.{i}", depth + 1,
125 i == 0 ? low : node.Entries[i - 1].Key, i == count ? high : node.Entries[i].Key);
126 if (problem is not null) return problem;
127 }
128 return null;
129 }
130 return Visit(Root, "root", 0, null, null);
131 }
132 public static Tree Load(int degree, string[] tokens)
133 {
134 Tree tree = new(degree);
135 tree.Pages.Clear();
136 int position = 0, nextPage = 0;
137 CommandError Syntax() => new($"SYNTAX {position + 1}");
138 Entry ReadEntry()
139 {
140 if (position == tokens.Length || tokens[position] is "(" or ")") throw Syntax();
141 string[] pair = tokens[position++].Split('=');
142 return new(int.Parse(pair[0]), pair[1]);
143 }
144 int ReadNode()
145 {
146 if (position == tokens.Length || tokens[position] != "(") throw Syntax();
147 position++;
148 int page = nextPage++;
149 Node node = new();
150 tree.Pages.Add(page, node);
151 if (position < tokens.Length && tokens[position] == "(")
152 {
153 node.Children.Add(ReadNode());
154 while (position == tokens.Length || tokens[position] != ")")
155 {
156 node.Entries.Add(ReadEntry()); node.Children.Add(ReadNode());
157 }
158 }
159 else while (position == tokens.Length || tokens[position] != ")") node.Entries.Add(ReadEntry());
160 position++;
161 return page;
162 }
163 tree.Root = ReadNode();
164 if (position != tokens.Length) throw Syntax();
165 return tree;
166 }
167 private void Split(int parentPage, int position, Trace trace)
168 {
169 Node parent = Pages[parentPage];
170 int childPage = parent.Children[position];
171 Node child = Pages[childPage], right = new();
172 string before = child.Notation;
173 Entry middle = child.Entries[Degree - 1];
174 right.Entries.AddRange(child.Entries.Skip(Degree));
175 right.Children.AddRange(child.Children.Skip(Degree));
176 int newPage = Allocate(right);
177 child.Entries.RemoveRange(Degree - 1, Degree);
178 if (!child.Leaf) child.Children.RemoveRange(Degree, Degree);
179 parent.Entries.Insert(position, middle); parent.Children.Insert(position + 1, newPage);
180 Splits++;
181 trace.Emit($"STEP split page={childPage} keys={before} up={middle.Key} new={newPage}");
182 }
183 public string? Upsert(int key, string value, Trace trace)
184 {
185 SearchResult found = Search(key);
186 if (found.Page is int existing)
187 {
188 Entry old = Pages[existing].Entries[found.Position];
189 Pages[existing].Entries[found.Position] = new(key, value);
190 return old.Value;
191 }
192 if (Pages[Root].Entries.Count == 2 * Degree - 1)
193 {
194 Node root = new(); root.Children.Add(Root);
195 Root = Allocate(root);
196 trace.Emit($"STEP grow page={Root}");
197 Split(Root, 0, trace);
198 }
199 int page = Root;
200 while (!Pages[page].Leaf)
201 {
202 Node node = Pages[page];
203 int position = node.Position(key);
204 if (Pages[node.Children[position]].Entries.Count == 2 * Degree - 1)
205 {
206 Split(page, position, trace);
207 if (key > node.Entries[position].Key) position++;
208 }
209 page = node.Children[position];
210 }
211 Node leaf = Pages[page];
212 leaf.Entries.Insert(leaf.Position(key), new(key, value));
213 trace.Emit($"STEP place page={page} keys={leaf.Notation}");
214 return null;
215 }
216 private int Merge(int parentPage, int position, string caseName, Trace trace)
217 {
218 Node parent = Pages[parentPage];
219 int leftPage = parent.Children[position], rightPage = parent.Children[position + 1];
220 Node left = Pages[leftPage], right = Pages[rightPage];
221 left.Entries.Add(parent.Entries[position]); left.Entries.AddRange(right.Entries); left.Children.AddRange(right.Children);
222 parent.Entries.RemoveAt(position); parent.Children.RemoveAt(position + 1);
223 Pages.Remove(rightPage); Merges++;
224 trace.Emit($"STEP {caseName} page={parentPage} merge={leftPage}+{rightPage} keys={left.Notation}");
225 if (parentPage == Root && parent.Entries.Count == 0)
226 {
227 Root = leftPage; Pages.Remove(parentPage);
228 trace.Emit($"STEP shrink page={Root}");
229 }
230 return leftPage;
231 }
232 private void RemoveAt(int page, int key, List<string> cases, Trace trace)
233 {
234 Node node = Pages[page];
235 int position = node.Position(key);
236 if (position < node.Entries.Count && node.Entries[position].Key == key)
237 {
238 if (node.Leaf)
239 {
240 cases.Add("1"); node.Entries.RemoveAt(position);
241 trace.Emit($"STEP 1 page={page} keys={node.Notation}");
242 return;
243 }
244 int left = node.Children[position], right = node.Children[position + 1];
245 if (Pages[left].Entries.Count >= Degree)
246 {
247 int last = left;
248 while (!Pages[last].Leaf) last = Pages[last].Children[^1];
249 Entry predecessor = Pages[last].Entries[^1];
250 cases.Add("2a"); trace.Emit($"STEP 2a page={page} pred={predecessor.Key}");
251 RemoveAt(left, predecessor.Key, cases, trace);
252 node.Entries[position] = predecessor;
253 }
254 else if (Pages[right].Entries.Count >= Degree)
255 {
256 int first = right;
257 while (!Pages[first].Leaf) first = Pages[first].Children[0];
258 Entry successor = Pages[first].Entries[0];
259 cases.Add("2b"); trace.Emit($"STEP 2b page={page} succ={successor.Key}");
260 RemoveAt(right, successor.Key, cases, trace);
261 node.Entries[position] = successor;
262 }
263 else
264 {
265 cases.Add("2c");
266 int child = Merge(page, position, "2c", trace);
267 RemoveAt(child, key, cases, trace);
268 }
269 return;
270 }
271 int childPage = node.Children[position];
272 Node target = Pages[childPage];
273 if (target.Entries.Count == Degree - 1)
274 {
275 int? leftPage = position > 0 ? node.Children[position - 1] : null;
276 int? rightPage = position < node.Entries.Count ? node.Children[position + 1] : null;
277 if (leftPage is int left && Pages[left].Entries.Count >= Degree)
278 {
279 Node sibling = Pages[left];
280 cases.Add("3a-left"); target.Entries.Insert(0, node.Entries[position - 1]);
281 node.Entries[position - 1] = sibling.Entries[^1]; sibling.Entries.RemoveAt(sibling.Entries.Count - 1);
282 if (!sibling.Leaf) { target.Children.Insert(0, sibling.Children[^1]); sibling.Children.RemoveAt(sibling.Children.Count - 1); }
283 Borrows++;
284 trace.Emit($"STEP 3a-left page={page} child={childPage} from={left} keys={target.Notation}");
285 }
286 else if (rightPage is int right && Pages[right].Entries.Count >= Degree)
287 {
288 Node sibling = Pages[right];
289 cases.Add("3a-right"); target.Entries.Add(node.Entries[position]);
290 node.Entries[position] = sibling.Entries[0]; sibling.Entries.RemoveAt(0);
291 if (!sibling.Leaf) { target.Children.Add(sibling.Children[0]); sibling.Children.RemoveAt(0); }
292 Borrows++;
293 trace.Emit($"STEP 3a-right page={page} child={childPage} from={right} keys={target.Notation}");
294 }
295 else if (leftPage.HasValue) { cases.Add("3b-left"); childPage = Merge(page, position - 1, "3b-left", trace); }
296 else { cases.Add("3b-right"); childPage = Merge(page, position, "3b-right", trace); }
297 }
298 RemoveAt(childPage, key, cases, trace);
299 }
300 public List<string> Remove(int key, Trace trace)
301 {
302 List<string> cases = [];
303 RemoveAt(Root, key, cases, trace);
304 return cases;
305 }
306 public (List<Entry> Entries, int Visited) Scan(int low, int high)
307 {
308 List<Entry> result = [];
309 int visited = 0;
310 void Visit(int page)
311 {
312 visited++;
313 Node node = Pages[page];
314 int count = node.Entries.Count;
315 for (int i = 0; i <= count; i++)
316 {
317 if (!node.Leaf && (i == 0 || node.Entries[i - 1].Key < high) && (i == count || node.Entries[i].Key > low)) Visit(node.Children[i]);
318 if (i < count && low <= node.Entries[i].Key && node.Entries[i].Key <= high) result.Add(node.Entries[i]);
319 }
320 }
321 Visit(Root);
322 return (result, visited);
323 }
324 public string Dump()
325 {
326 List<string> tokens = [];
327 void Visit(int page)
328 {
329 Node node = Pages[page];
330 tokens.Add("(");
331 if (!node.Leaf) Visit(node.Children[0]);
332 for (int i = 0; i < node.Entries.Count; i++)
333 {
334 tokens.Add(node.Entries[i].ToString());
335 if (!node.Leaf) Visit(node.Children[i + 1]);
336 }
337 tokens.Add(")");
338 }
339 Visit(Root);
340 return string.Join(" ", tokens);
341 }
342 public void Compact()
343 {
344 List<Entry> source = Entries();
345 List<int> children = [];
346 List<Node> built = [];
347 int oldRoot;
348 while (true)
349 {
350 int groups = (source.Count + 2 * Degree) / (2 * Degree), keys = source.Count - groups + 1;
351 int size = keys / groups, extra = keys % groups, entryIndex = 0, childIndex = 0;
352 List<int> nodes = [];
353 List<Entry> separators = [];
354 for (int group = 0; group < groups; group++)
355 {
356 int count = size + (group < extra ? 1 : 0);
357 Node node = new();
358 node.Entries.AddRange(source.GetRange(entryIndex, count));
359 if (children.Count > 0) node.Children.AddRange(children.GetRange(childIndex, count + 1));
360 nodes.Add(built.Count); built.Add(node);
361 entryIndex += count; childIndex += count + 1;
362 if (group + 1 < groups) separators.Add(source[entryIndex++]);
363 }
364 if (groups == 1) { oldRoot = nodes[0]; break; }
365 source = separators; children = nodes;
366 }
367 Dictionary<int, Node> replacement = [];
368 int nextPage = 0;
369 int Renumber(int oldPage)
370 {
371 int page = nextPage++;
372 Node old = built[oldPage], node = new();
373 node.Entries.AddRange(old.Entries);
374 foreach (int child in old.Children) node.Children.Add(Renumber(child));
375 replacement.Add(page, node);
376 return page;
377 }
378 Root = Renumber(oldRoot); Pages = replacement;
379 Splits = Merges = Borrows = 0;
380 }
381}

383enum CursorKind { Unset, Key, Begin, End }
384readonly record struct Cursor(CursorKind Kind, int Key = 0);
385
386sealed class Workbench
387{
388 private Tree? index;
389 private Tree? saved;
390 private Cursor cursor = new(CursorKind.Unset);
391 private bool trace;
392 private static readonly HashSet<string> NeedsValid = ["INSERT", "BULKLOAD", "DELETE", "DELRANGE", "RANGE", "COUNT", "MIN", "MAX",
393 "FLOOR", "CEIL", "RANK", "SELECT", "SEEK", "NEXT", "PREV", "HEIGHT", "COMPACT"];
394
395 private static Entry? Find(List<Entry> entries, Func<Entry, bool> predicate, bool last = false)
396 {
397 Entry? result = null;
398 foreach (Entry entry in entries)
399 {
400 if (!predicate(entry)) continue;
401 result = entry;
402 if (!last) break;
403 }
404 return result;
405 }
406 private static string Summary(string prefix, Tree tree) => $"{prefix} nodes={tree.Pages.Count} keys={tree.Entries().Count} {(tree.Violation() is null ? "OK" : "DAMAGED")}";
407 public List<string> Process(string line)
408 {
409 string[] words = line.Split(' ');
410 string command = words[0];
411 if (command == "TRACE") { trace = words[1] == "ON"; return [$"TRACE {words[1]}"]; }
412 int first = words.Length > 1 ? int.Parse(words[1]) : 0;
413 bool ranged = command is "DELRANGE" or "RANGE" or "COUNT";
414 int second = ranged ? int.Parse(words[2]) : 0;
415 if (command is "CREATE" or "LOAD")
416 {
417 if (first < 2 || first > 5) throw new CommandError($"BAD DEGREE {first}");
418 Tree replacement = command == "CREATE" ? new(first) : Tree.Load(first, words[2..]);
419 index = replacement; cursor = new(CursorKind.Unset);
420 return [command == "CREATE" ? $"CREATE t={first}" : Summary($"LOAD t={first}", replacement)];
421 }
422 Tree tree = index ?? throw new CommandError("NO INDEX");
423 if (NeedsValid.Contains(command) && tree.Violation() is not null) throw new CommandError("DAMAGED");
424 List<Entry> entries = tree.Entries();
425 if (ranged && first > second) throw new CommandError($"BAD RANGE {first} {second}");
426 if (command == "SELECT" && (first < 1 || first > entries.Count)) throw new CommandError($"BAD POSITION {first}");
427 if ((command is "NEXT" or "PREV") && cursor.Kind == CursorKind.Unset) throw new CommandError("NO CURSOR");
428 if (command == "RESTORE" && saved is null) throw new CommandError("NO SAVE");
429 List<string> output = [];
430 Trace steps = new(trace && (command is "INSERT" or "DELETE"));
431 switch (command)
432 {
433 case "INSERT":
434 {
435 Dictionary<int, Node> before = tree.CopyPages();
436 int splits = tree.Splits;
437 string? old = tree.Upsert(first, words[2], steps);
438 int writes = tree.Changes(before).Writes;
439 output.AddRange(steps.Lines);
440 output.Add(old is null ? $"INSERT {first} NEW splits={tree.Splits - splits} height={tree.Height} writes={writes}"
441 : $"INSERT {first} UPDATED old={old} writes={writes}");
442 break;
443 }
444 case "BULKLOAD":
445 {
446 Dictionary<int, Node> before = tree.CopyPages();
447 int splits = tree.Splits, inserted = 0, updated = 0;
448 for (int i = 1; i < words.Length; i += 2)
449 if (tree.Upsert(int.Parse(words[i]), words[i + 1], steps) is null) inserted++; else updated++;
450 output.Add($"BULKLOAD pairs={(words.Length - 1) / 2} new={inserted} updated={updated} splits={tree.Splits - splits} height={tree.Height} writes={tree.Changes(before).Writes}");
451 break;
452 }
453 case "DELETE":
454 {
455 SearchResult found = tree.Search(first);
456 if (found.Page is int page)
457 {
458 string value = tree.Pages[page].Entries[found.Position].Value;
459 Dictionary<int, Node> before = tree.CopyPages();
460 List<string> cases = tree.Remove(first, steps);
461 (int writes, int frees) = tree.Changes(before);
462 output.AddRange(steps.Lines);
463 output.Add($"DELETE {first} REMOVED value={value} cases={string.Join(",", cases)} height={tree.Height} writes={writes} frees={frees}");
464 }
465 else output.Add($"DELETE {first} NOT FOUND");
466 break;
467 }
468 case "DELRANGE":
469 {
470 Dictionary<int, Node> before = tree.CopyPages();
471 int merges = tree.Merges, borrows = tree.Borrows, removed = 0;
472 foreach (Entry entry in entries)
473 if (first <= entry.Key && entry.Key <= second) { tree.Remove(entry.Key, steps); removed++; }
474 (int writes, int frees) = tree.Changes(before);
475 output.Add($"DELRANGE {first} {second} removed={removed} merges={tree.Merges - merges} borrows={tree.Borrows - borrows} height={tree.Height} writes={writes} frees={frees}");
476 break;
477 }
478 case "GET":
479 {
480 SearchResult found = tree.Search(first);
481 string result = found.Page is int page ? "FOUND value=" + tree.Pages[page].Entries[found.Position].Value : "NOT FOUND";
482 output.Add($"GET {first} {result} path={string.Join(" -> ", found.Path)}");
483 break;
484 }
485 case "RANGE": case "COUNT":
486 {
487 (List<Entry> selected, int visited) = tree.Scan(first, second);
488 string suffix = command == "RANGE" ? string.Concat(selected.Select(entry => " " + entry)) : "";
489 output.Add($"{command} {first} {second} count={selected.Count} visited={visited}{suffix}");
490 break;
491 }
492 case "MIN": case "MAX":
493 output.Add($"{command} {(entries.Count == 0 ? "EMPTY" : (command == "MIN" ? entries[0] : entries[^1]).ToString())}");
494 break;
495 case "FLOOR": case "CEIL":
496 {
497 Entry? entry = command == "FLOOR" ? Find(entries, value => value.Key <= first, true) : Find(entries, value => value.Key >= first);
498 output.Add($"{command} {first} {entry?.ToString() ?? "NONE"}");
499 break;
500 }
501 case "RANK": output.Add($"RANK {first} {entries.Count(entry => entry.Key < first)}"); break;
502 case "SELECT": output.Add($"SELECT {first} {entries[first - 1]}"); break;
503 case "SEEK":
504 {
505 Entry? entry = Find(entries, value => value.Key >= first);
506 cursor = entry is Entry found ? new(CursorKind.Key, found.Key) : new(CursorKind.End);
507 output.Add($"SEEK {first} {entry?.ToString() ?? "END"}");
508 break;
509 }
510 case "NEXT": case "PREV":
511 {
512 Entry? entry = null;
513 if (command == "NEXT")
514 {
515 if (cursor.Kind == CursorKind.Begin) entry = Find(entries, _ => true);
516 else if (cursor.Kind == CursorKind.Key) entry = Find(entries, value => value.Key > cursor.Key);
517 }
518 else
519 {
520 if (cursor.Kind == CursorKind.End) entry = Find(entries, _ => true, true);
521 else if (cursor.Kind == CursorKind.Key) entry = Find(entries, value => value.Key < cursor.Key, true);
522 }
523 cursor = entry is Entry found ? new(CursorKind.Key, found.Key) : new(command == "NEXT" ? CursorKind.End : CursorKind.Begin);
524 output.Add($"{command} {entry?.ToString() ?? (command == "NEXT" ? "END" : "BEGIN")}");
525 break;
526 }
527 case "PRINT":
528 {
529 List<int> level = [tree.Root];
530 int depth = 0;
531 while (level.Count > 0)
532 {
533 output.Add($"LEVEL {depth} " + string.Join(" ", level.Select(page => tree.Pages[page].Notation)));
534 level = level.SelectMany(page => tree.Pages[page].Children).ToList(); depth++;
535 }
536 break;
537 }
538 case "PAGES":
539 foreach (int page in tree.Pages.Keys.Order())
540 {
541 Node node = tree.Pages[page];
542 output.Add($"PAGE {page} {node.Notation} kids={(node.Leaf ? "-" : string.Join(",", node.Children))}");
543 }
544 break;
545 case "DUMP": output.Add($"DUMP {tree.Degree} {tree.Dump()}"); break;
546 case "HEIGHT": output.Add($"HEIGHT {tree.Height}"); break;
547 case "STATS":
548 {
549 int leaves = tree.Pages.Values.Count(node => node.Leaf);
550 int fill = 100 * entries.Count / ((2 * tree.Degree - 1) * tree.Pages.Count);
551 output.Add($"STATS t={tree.Degree} nodes={tree.Pages.Count} leaves={leaves} keys={entries.Count} fill={fill}% splits={tree.Splits} merges={tree.Merges} borrows={tree.Borrows}");
552 break;
553 }
554 case "VERIFY": output.Add(tree.Violation() is string problem ? "VERIFY FAIL " + problem : "VERIFY OK"); break;
555 case "REBUILD":
556 tree = new(tree.Degree);
557 foreach (Entry entry in entries) tree.Upsert(entry.Key, entry.Value, steps);
558 output.Add($"REBUILD entries={entries.Count} keys={tree.Entries().Count} nodes={tree.Pages.Count} splits={tree.Splits} height={tree.Height}");
559 break;
560 case "COMPACT":
561 {
562 int nodes = tree.Pages.Count;
563 tree.Compact();
564 output.Add($"COMPACT nodes={nodes}->{tree.Pages.Count} height={tree.Height}");
565 break;
566 }
567 case "SAVE": saved = tree.Copy(); output.Add(Summary("SAVE", tree)); break;
568 case "RESTORE": tree = saved!.Copy(); output.Add(Summary("RESTORE", tree)); break;
569 }
570 index = tree;
571 return output;
572 }
573}
