42const Node = struct {
43 entries: std.ArrayList(Entry) = .empty,
44 children: std.ArrayList(usize) = .empty,
45 fn leaf(node: *const Node) bool {
46 return node.children.items.len == 0;
47 }
48 fn position(node: *const Node, key: i64) usize {
49 for (node.entries.items, 0..) |entry, i| if (key <= entry.key) return i;
50 return node.entries.items.len;
51 }
52 fn destroy(node: *Node, gpa: Allocator) void {
53 node.entries.deinit(gpa);
54 node.children.deinit(gpa);
55 gpa.destroy(node);
56 }
57 fn clone(node: *const Node, gpa: Allocator) Allocator.Error!*Node {
58 const result = try gpa.create(Node);
59 result.* = .{};
60 errdefer result.destroy(gpa);
61 result.entries = try node.entries.clone(gpa);
62 result.children = try node.children.clone(gpa);
63 return result;
64 }
65 fn sameContent(a: *const Node, b: *const Node) bool {
66 if (a.entries.items.len != b.entries.items.len or !std.mem.eql(usize, a.children.items, b.children.items)) return false;
67 for (a.entries.items, b.entries.items) |left, right| if (!left.eql(right)) return false;
68 return true;
69 }
70 pub fn format(node: Node, out: *Writer) Writer.Error!void {
71 try out.writeByte('[');
72 for (node.entries.items, 0..) |entry, i| {
73 if (i > 0) try out.writeByte(' ');
74 try out.print("{d}", .{entry.key});
75 }
76 try out.writeByte(']');
77 }

109 fn init(gpa: Allocator, degree: usize) Allocator.Error!Tree {
110 var tree: Tree = .{ .gpa = gpa, .degree = degree };
111 errdefer tree.deinit();
112 tree.root = try tree.newPage();
113 return tree;
114 }
115 fn deinit(tree: *Tree) void {
116 for (tree.pages.items) |slot| if (slot) |item| item.destroy(tree.gpa);
117 tree.pages.deinit(tree.gpa);
118 }
119 fn node(tree: *const Tree, page: usize) *Node {
120 return tree.pages.items[page].?;
121 }
122 fn newPage(tree: *Tree) Allocator.Error!usize {
123 const result = try tree.gpa.create(Node);
124 result.* = .{};
125 errdefer result.destroy(tree.gpa);
126 for (tree.pages.items, 0..) |slot, page| if (slot == null) {
127 tree.pages.items[page] = result;
128 return page;
129 };
130 try tree.pages.append(tree.gpa, result);
131 return tree.pages.items.len - 1;
132 }
133 fn freePage(tree: *Tree, page: usize) void {
134 tree.node(page).destroy(tree.gpa);
135 tree.pages.items[page] = null;
136 }
137 fn clone(tree: *const Tree) Allocator.Error!Tree {
138 var result: Tree = .{ .gpa = tree.gpa, .degree = tree.degree, .root = tree.root, .splits = tree.splits, .merges = tree.merges, .borrows = tree.borrows };
139 errdefer result.deinit();
140 try result.pages.ensureTotalCapacity(tree.gpa, tree.pages.items.len);
141 for (tree.pages.items) |slot| result.pages.appendAssumeCapacity(if (slot) |item| try item.clone(tree.gpa) else null);
142 return result;
143 }
144 fn countNodes(tree: *const Tree) usize {
145 var count: usize = 0;
146 for (tree.pages.items) |slot| if (slot != null) {
147 count += 1;
148 };
149 return count;
150 }
151 fn countEntries(tree: *const Tree) usize {
152 var count: usize = 0;
153 for (tree.pages.items) |slot| if (slot) |item| {
154 count += item.entries.items.len;
155 };
156 return count;

158 fn changes(tree: *const Tree, before: *const Tree) struct { writes: usize, frees: usize } {
159 var writes: usize = 0;
160 var frees: usize = 0;
161 for (tree.pages.items, 0..) |slot, page| if (slot) |item| {
162 const old = if (page < before.pages.items.len) before.pages.items[page] else null;
163 if (old == null or !item.sameContent(old.?)) writes += 1;
164 };
165 for (before.pages.items, 0..) |slot, page| if (slot != null and (page >= tree.pages.items.len or tree.pages.items[page] == null)) {
166 frees += 1;
167 };
168 return .{ .writes = writes, .frees = frees };
169 }
170 fn height(tree: *const Tree) usize {
171 var page = tree.root;
172 var depth: usize = 0;
173 while (!tree.node(page).leaf()) : (depth += 1) page = tree.node(page).children.items[0];
174 return depth;
175 }
176 fn collect(tree: *const Tree, page: usize, gpa: Allocator, result: *std.ArrayList(Entry)) Allocator.Error!void {
177 const item = tree.node(page);
178 if (!item.leaf()) try tree.collect(item.children.items[0], gpa, result);
179 for (item.entries.items, 0..) |entry, i| {
180 try result.append(gpa, entry);
181 if (!item.leaf()) try tree.collect(item.children.items[i + 1], gpa, result);
182 }
183 }
184 fn entries(tree: *const Tree, gpa: Allocator) Allocator.Error![]const Entry {
185 var result: std.ArrayList(Entry) = .empty;
186 try tree.collect(tree.root, gpa, &result);
187 return result.toOwnedSlice(gpa);
188 }
189 fn search(tree: *const Tree, key: i64) Search {
190 var result: Search = .{};
191 var page = tree.root;
192 while (true) {
193 const item = tree.node(page);
194 const position = item.position(key);
195 result.path[result.len] = page;
196 result.len += 1;
197 if (position < item.entries.items.len and item.entries.items[position].key == key) {
198 result.page = page;
199 result.position = position;
200 return result;
201 }
202 if (item.leaf()) return result;
203 page = item.children.items[position];
204 }
205 }
206 fn checkNode(tree: *const Tree, page: usize, location: []const u8, depth: usize, low: ?i64, high: ?i64, expected: *?usize, gpa: Allocator) Allocator.Error!?[]const u8 {
207 const item = tree.node(page);
208 const count = item.entries.items.len;
209 const minimum = if (page == tree.root) @as(usize, if (item.leaf()) 0 else 1) else tree.degree - 1;
210 const maximum = 2 * tree.degree - 1;
211 if (count < minimum or count > maximum) return try std.fmt.allocPrint(gpa, "COUNT at={s} node={f} keys={d} allowed={d}..{d}", .{ location, item.*, count, minimum, maximum });
212 for (item.entries.items, 0..) |entry, i| {
213 if (i + 1 < count and entry.key >= item.entries.items[i + 1].key) return try std.fmt.allocPrint(gpa, "ORDER at={s} node={f} {d}>={d}", .{ location, item.*, entry.key, item.entries.items[i + 1].key });
214 }
215 for (item.entries.items) |entry| {
216 if ((low != null and entry.key <= low.?) or (high != null and entry.key >= high.?)) return try std.fmt.allocPrint(gpa, "RANGE at={s} node={f} key={d} low={f} high={f}", .{ location, item.*, entry.key, Bound{ .value = low, .lower = true }, Bound{ .value = high, .lower = false } });
217 }
218 if (item.leaf()) {
219 if (expected.*) |reference| {
220 if (depth != reference) return try std.fmt.allocPrint(gpa, "DEPTH at={s} node={f} depth={d} expected={d}", .{ location, item.*, depth, reference });
221 } else expected.* = depth;
222 }
223 for (item.children.items, 0..) |child, i| {
224 const path = try std.fmt.allocPrint(gpa, "{s}.{d}", .{ location, i });
225 if (try tree.checkNode(child, path, depth + 1, if (i == 0) low else item.entries.items[i - 1].key, if (i == count) high else item.entries.items[i].key, expected, gpa)) |problem| return problem;
226 }
227 return null;
228 }
229 fn violation(tree: *const Tree, gpa: Allocator) Allocator.Error!?[]const u8 {
230 var expected: ?usize = null;
231 return tree.checkNode(tree.root, "root", 0, null, null, &expected, gpa);
232 }
233 fn splitChild(tree: *Tree, parent_page: usize, position: usize, trace: *Trace) Error!void {
234 const parent = tree.node(parent_page);
235 const child_page = parent.children.items[position];
236 const child = tree.node(child_page);
237 const middle = child.entries.items[tree.degree - 1];
238 const right_page = try tree.newPage();
239 const right = tree.node(right_page);
240 try trace.emit("STEP split page={d} keys={f} up={d} new={d}", .{ child_page, child.*, middle.key, right_page });
241 try right.entries.appendSlice(tree.gpa, child.entries.items[tree.degree..]);
242 if (!child.leaf()) try right.children.appendSlice(tree.gpa, child.children.items[tree.degree..]);
243 child.entries.shrinkRetainingCapacity(tree.degree - 1);
244 if (!child.leaf()) child.children.shrinkRetainingCapacity(tree.degree);
245 try parent.entries.insert(tree.gpa, position, middle);
246 try parent.children.insert(tree.gpa, position + 1, right_page);
247 tree.splits += 1;
248 }
249 fn upsert(tree: *Tree, key: i64, value: Value, trace: *Trace) Error!?Value {
250 const found = tree.search(key);
251 if (found.page) |page| {
252 const item = tree.node(page);
253 const old = item.entries.items[found.position].value;
254 item.entries.items[found.position] = .{ .key = key, .value = value };
255 return old;
256 }
257 if (tree.node(tree.root).entries.items.len == 2 * tree.degree - 1) {
258 const previous = tree.root;
259 tree.root = try tree.newPage();
260 try tree.node(tree.root).children.append(tree.gpa, previous);
261 try trace.emit("STEP grow page={d}", .{tree.root});
262 try tree.splitChild(tree.root, 0, trace);
263 }
264 var page = tree.root;
265 while (!tree.node(page).leaf()) {
266 const item = tree.node(page);
267 var position = item.position(key);
268 if (tree.node(item.children.items[position]).entries.items.len == 2 * tree.degree - 1) {
269 try tree.splitChild(page, position, trace);
270 if (key > item.entries.items[position].key) position += 1;
271 }
272 page = item.children.items[position];
273 }
274 const leaf = tree.node(page);
275 try leaf.entries.insert(tree.gpa, leaf.position(key), .{ .key = key, .value = value });
276 try trace.emit("STEP place page={d} keys={f}", .{ page, leaf.* });
277 return null;
278 }
279 fn merge(tree: *Tree, parent_page: usize, position: usize, case_name: []const u8, trace: *Trace) Error!usize {
280 const parent = tree.node(parent_page);
281 const left_page = parent.children.items[position];
282 const right_page = parent.children.items[position + 1];
283 const left = tree.node(left_page);
284 const right = tree.node(right_page);
285 try left.entries.append(tree.gpa, parent.entries.orderedRemove(position));
286 try left.entries.appendSlice(tree.gpa, right.entries.items);
287 try left.children.appendSlice(tree.gpa, right.children.items);
288 _ = parent.children.orderedRemove(position + 1);
289 tree.freePage(right_page);
290 tree.merges += 1;
291 try trace.emit("STEP {s} page={d} merge={d}+{d} keys={f}", .{ case_name, parent_page, left_page, right_page, left.* });
292 if (parent_page == tree.root and parent.entries.items.len == 0) {
293 tree.root = left_page;
294 tree.freePage(parent_page);
295 try trace.emit("STEP shrink page={d}", .{tree.root});
296 }
297 return left_page;
298 }
299 fn removeAt(tree: *Tree, page: usize, key: i64, trace: *Trace) Error!void {
300 const item = tree.node(page);
301 const position = item.position(key);
302 if (position < item.entries.items.len and item.entries.items[position].key == key) {
303 if (item.leaf()) {
304 try trace.record("1");
305 _ = item.entries.orderedRemove(position);
306 try trace.emit("STEP 1 page={d} keys={f}", .{ page, item.* });
307 return;
308 }
309 const left = item.children.items[position];
310 const right = item.children.items[position + 1];
311 if (tree.node(left).entries.items.len >= tree.degree) {
312 var last = left;
313 while (!tree.node(last).leaf()) {
314 const children = tree.node(last).children.items;
315 last = children[children.len - 1];
316 }
317 const rows = tree.node(last).entries.items;
318 const predecessor = rows[rows.len - 1];
319 try trace.record("2a");
320 try trace.emit("STEP 2a page={d} pred={d}", .{ page, predecessor.key });
321 try tree.removeAt(left, predecessor.key, trace);
322 item.entries.items[position] = predecessor;
323 } else if (tree.node(right).entries.items.len >= tree.degree) {
324 var first = right;
325 while (!tree.node(first).leaf()) first = tree.node(first).children.items[0];
326 const successor = tree.node(first).entries.items[0];
327 try trace.record("2b");
328 try trace.emit("STEP 2b page={d} succ={d}", .{ page, successor.key });
329 try tree.removeAt(right, successor.key, trace);
330 item.entries.items[position] = successor;
331 } else {
332 try trace.record("2c");
333 const child = try tree.merge(page, position, "2c", trace);
334 try tree.removeAt(child, key, trace);
335 }
336 return;
337 }
338 var child_page = item.children.items[position];
339 const child = tree.node(child_page);
340 if (child.entries.items.len == tree.degree - 1) {
341 const left_page: ?usize = if (position > 0) item.children.items[position - 1] else null;
342 const right_page: ?usize = if (position < item.entries.items.len) item.children.items[position + 1] else null;
343 if (left_page != null and tree.node(left_page.?).entries.items.len >= tree.degree) {
344 const sibling = tree.node(left_page.?);
345 try trace.record("3a-left");
346 try child.entries.insert(tree.gpa, 0, item.entries.items[position - 1]);
347 item.entries.items[position - 1] = sibling.entries.pop().?;
348 if (!sibling.leaf()) try child.children.insert(tree.gpa, 0, sibling.children.pop().?);
349 tree.borrows += 1;
350 try trace.emit("STEP 3a-left page={d} child={d} from={d} keys={f}", .{ page, child_page, left_page.?, child.* });
351 } else if (right_page != null and tree.node(right_page.?).entries.items.len >= tree.degree) {
352 const sibling = tree.node(right_page.?);
353 try trace.record("3a-right");
354 try child.entries.append(tree.gpa, item.entries.items[position]);
355 item.entries.items[position] = sibling.entries.orderedRemove(0);
356 if (!sibling.leaf()) try child.children.append(tree.gpa, sibling.children.orderedRemove(0));
357 tree.borrows += 1;
358 try trace.emit("STEP 3a-right page={d} child={d} from={d} keys={f}", .{ page, child_page, right_page.?, child.* });
359 } else if (left_page != null) {
360 try trace.record("3b-left");
361 child_page = try tree.merge(page, position - 1, "3b-left", trace);
362 } else {
363 try trace.record("3b-right");
364 child_page = try tree.merge(page, position, "3b-right", trace);
365 }
366 }
367 try tree.removeAt(child_page, key, trace);
368 }
369 fn remove(tree: *Tree, key: i64, trace: *Trace) Error!void {
370 trace.cases.clearRetainingCapacity();
371 try tree.removeAt(tree.root, key, trace);
372 }
373 fn scanNode(tree: *const Tree, page: usize, low: i64, high: i64, gpa: Allocator, result: *std.ArrayList(Entry), visited: *usize) Allocator.Error!void {
374 visited.* += 1;
375 const item = tree.node(page);
376 const count = item.entries.items.len;
377 for (0..count + 1) |i| {
378 if (!item.leaf() and (i == 0 or item.entries.items[i - 1].key < high) and (i == count or item.entries.items[i].key > low)) {
379 try tree.scanNode(item.children.items[i], low, high, gpa, result, visited);
380 }
381 if (i < count and low <= item.entries.items[i].key and item.entries.items[i].key <= high) try result.append(gpa, item.entries.items[i]);
382 }
383 }
384 fn dumpNode(tree: *const Tree, page: usize, out: *Writer) Writer.Error!void {
385 const item = tree.node(page);
386 try out.writeByte('(');
387 if (!item.leaf()) {
388 try out.writeByte(' ');
389 try tree.dumpNode(item.children.items[0], out);
390 }
391 for (item.entries.items, 0..) |entry, i| {
392 try out.print(" {f}", .{entry});
393 if (!item.leaf()) {
394 try out.writeByte(' ');
395 try tree.dumpNode(item.children.items[i + 1], out);
396 }
397 }
398 try out.writeAll(" )");

400 const Group = struct { rows: []const Entry, children: []const usize };
401 fn copyGroup(tree: *Tree, groups: []const Group, old_page: usize) Allocator.Error!usize {
402 const page = try tree.newPage();
403 const item = tree.node(page);
404 const old = groups[old_page];
405 try item.entries.appendSlice(tree.gpa, old.rows);
406 for (old.children) |child| {
407 const new_child = try tree.copyGroup(groups, child);
408 try item.children.append(tree.gpa, new_child);
409 }
410 return page;
411 }
412 fn compact(tree: *Tree, scratch: Allocator) Allocator.Error!void {
413 var source = try tree.entries(scratch);
414 var children: []const usize = &.{};
415 var built: std.ArrayList(Group) = .empty;
416 var old_root: usize = undefined;
417 while (true) {
418 const groups = (source.len + 2 * tree.degree) / (2 * tree.degree);
419 const count = source.len - (groups - 1);
420 const size = count / groups;
421 const extra = count % groups;
422 var entry_index: usize = 0;
423 var child_index: usize = 0;
424 var nodes: std.ArrayList(usize) = .empty;
425 var separators: std.ArrayList(Entry) = .empty;
426 for (0..groups) |group| {
427 const length = size + @intFromBool(group < extra);
428 try nodes.append(scratch, built.items.len);
429 try built.append(scratch, .{ .rows = source[entry_index..][0..length], .children = if (children.len == 0) &.{} else children[child_index..][0 .. length + 1] });
430 entry_index += length;
431 child_index += length + 1;
432 if (group + 1 < groups) {
433 try separators.append(scratch, source[entry_index]);
434 entry_index += 1;
435 }
436 }
437 if (groups == 1) {
438 old_root = nodes.items[0];
439 break;
440 }
441 source = separators.items;
442 children = nodes.items;
443 }
444 var replacement: Tree = .{ .gpa = tree.gpa, .degree = tree.degree };
445 errdefer replacement.deinit();
446 replacement.root = try replacement.copyGroup(built.items, old_root);
447 tree.deinit();
448 tree.* = replacement;
449 }

452const Loader = struct {
453 tree: *Tree,
454 tokens: []const []const u8,
455 position: usize = 0,
456 fn entry(loader: *Loader) error{Invalid}!Entry {
457 if (loader.position == loader.tokens.len) return error.Invalid;
458 const token = loader.tokens[loader.position];
459 if (equal(token, "(") or equal(token, ")")) return error.Invalid;
460 const split = std.mem.findScalar(u8, token, '=').?;
461 loader.position += 1;
462 return .{ .key = integer(token[0..split]), .value = Value.init(token[split + 1 ..]) };
463 }
464 fn parseNode(loader: *Loader) (Allocator.Error || error{Invalid})!usize {
465 if (loader.position == loader.tokens.len or !equal(loader.tokens[loader.position], "(")) return error.Invalid;
466 loader.position += 1;
467 const page = try loader.tree.newPage();
468 const item = loader.tree.node(page);
469 if (loader.position < loader.tokens.len and equal(loader.tokens[loader.position], "(")) {
470 try item.children.append(loader.tree.gpa, try loader.parseNode());
471 while (loader.position == loader.tokens.len or !equal(loader.tokens[loader.position], ")")) {
472 try item.entries.append(loader.tree.gpa, try loader.entry());
473 try item.children.append(loader.tree.gpa, try loader.parseNode());
474 }
475 } else {
476 while (loader.position == loader.tokens.len or !equal(loader.tokens[loader.position], ")")) {
477 try item.entries.append(loader.tree.gpa, try loader.entry());
478 }
479 }
480 loader.position += 1;
481 return page;
482 }

500const Relation = enum { all, less, greater, at_most, at_least };
501fn candidate(rows: []const Entry, key: i64, relation: Relation, last: bool) ?Entry {
502 var result: ?Entry = null;
503 for (rows) |entry| {
504 const matches = switch (relation) {
505 .all => true,
506 .less => entry.key < key,
507 .greater => entry.key > key,
508 .at_most => entry.key <= key,
509 .at_least => entry.key >= key,
510 };
511 if (!matches) continue;
512 result = entry;
513 if (!last) break;
514 }
515 return result;
516}

525 }
526 fn summary(session: *Session, prefix: []const u8, tree: *const Tree) Error!void {
527 try line(session.out, "{s} nodes={d} keys={d} {s}", .{ prefix, tree.countNodes(), tree.countEntries(), if (try tree.violation(session.gpa) == null) "OK" else "DAMAGED" });
528 }
529 fn execute(session: *Session, words: []const []const u8) Error!void {
530 const command = std.meta.stringToEnum(Command, words[0]).?;
531 const out = session.out;
532 const lab = session.lab;
533 if (command == .TRACE) {
534 lab.trace = equal(words[1], "ON");
535 return line(out, "TRACE {s}", .{words[1]});
536 }
537 const first = if (words.len > 1) integer(words[1]) else 0;
538 const ranged = command == .DELRANGE or command == .RANGE or command == .COUNT;
539 const second = if (ranged) integer(words[2]) else 0;
540 if (command == .CREATE or command == .LOAD) {
541 if (first < 2 or first > 5) return session.fail("BAD DEGREE {d}", .{first});
542 const degree: usize = @intCast(first);
543 var replacement: Tree = .{ .gpa = lab.gpa, .degree = degree };
544 errdefer replacement.deinit();
545 if (command == .CREATE) {
546 replacement.root = try replacement.newPage();
547 } else {
548 var loader: Loader = .{ .tree = &replacement, .tokens = words[2..] };
549 replacement.root = loader.parseNode() catch |err| switch (err) {
550 error.Invalid => return session.fail("SYNTAX {d}", .{loader.position + 1}),
551 else => return err,
552 };
553 if (loader.position != loader.tokens.len) return session.fail("SYNTAX {d}", .{loader.position + 1});
554 }
555 if (command == .CREATE) {
556 try line(out, "CREATE t={d}", .{first});
557 } else {
558 try session.summary(try std.fmt.allocPrint(session.gpa, "LOAD t={d}", .{first}), &replacement);
559 }
560 lab.replace(replacement);
561 lab.cursor = .unset;
562 return;
563 }
564 const tree = if (lab.index) |*value| value else return session.fail("NO INDEX", .{});
565 const needs_valid = switch (command) {
566 .INSERT, .BULKLOAD, .DELETE, .DELRANGE, .RANGE, .COUNT, .MIN, .MAX, .FLOOR, .CEIL, .RANK, .SELECT, .SEEK, .NEXT, .PREV, .HEIGHT, .COMPACT => true,
567 else => false,
568 };
569 if (needs_valid and try tree.violation(session.gpa) != null) return session.fail("DAMAGED", .{});
570 const rows = try tree.entries(session.gpa);
571 if (ranged and first > second) return session.fail("BAD RANGE {d} {d}", .{ first, second });
572 if (command == .SELECT and (first < 1 or first > @as(i64, @intCast(rows.len)))) return session.fail("BAD POSITION {d}", .{first});
573 if (command == .NEXT or command == .PREV) switch (lab.cursor) {
574 .unset => return session.fail("NO CURSOR", .{}),
575 else => {},
576 };
577 if (command == .RESTORE and lab.saved == null) return session.fail("NO SAVE", .{});
578 var trace: Trace = .{ .gpa = session.gpa, .out = out, .enabled = lab.trace and (command == .INSERT or command == .DELETE) };
579 switch (command) {
580 .INSERT => {
581 var before = try tree.clone();
582 defer before.deinit();
583 const splits = tree.splits;
584 const old = try tree.upsert(first, Value.init(words[2]), &trace);
585 const writes = tree.changes(&before).writes;
586 if (old) |value| {
587 try line(out, "INSERT {d} UPDATED old={f} writes={d}", .{ first, value, writes });
588 } else try line(out, "INSERT {d} NEW splits={d} height={d} writes={d}", .{ first, tree.splits - splits, tree.height(), writes });
589 },
590 .BULKLOAD => {
591 var before = try tree.clone();
592 defer before.deinit();
593 const splits = tree.splits;
594 var inserted: usize = 0;
595 var updated: usize = 0;
596 var i: usize = 1;
597 while (i < words.len) : (i += 2) {
598 if (try tree.upsert(integer(words[i]), Value.init(words[i + 1]), &trace) == null) inserted += 1 else updated += 1;
599 }
600 try line(out, "BULKLOAD pairs={d} new={d} updated={d} splits={d} height={d} writes={d}", .{ (words.len - 1) / 2, inserted, updated, tree.splits - splits, tree.height(), tree.changes(&before).writes });
601 },
602 .DELETE => {
603 const found = tree.search(first);
604 if (found.page) |page| {
605 const value = tree.node(page).entries.items[found.position].value;
606 var before = try tree.clone();
607 defer before.deinit();
608 try tree.remove(first, &trace);
609 const changes = tree.changes(&before);
610 try out.print("DELETE {d} REMOVED value={f} cases=", .{ first, value });
611 for (trace.cases.items, 0..) |name, i| {
612 if (i > 0) try out.writeByte(',');
613 try out.writeAll(name);
614 }
615 try line(out, " height={d} writes={d} frees={d}", .{ tree.height(), changes.writes, changes.frees });
616 } else try line(out, "DELETE {d} NOT FOUND", .{first});
617 },
618 .DELRANGE => {
619 var before = try tree.clone();
620 defer before.deinit();
621 const merges = tree.merges;
622 const borrows = tree.borrows;
623 var removed: usize = 0;
624 for (rows) |entry| if (first <= entry.key and entry.key <= second) {
625 try tree.remove(entry.key, &trace);
626 removed += 1;
627 };
628 const changes = tree.changes(&before);
629 try line(out, "DELRANGE {d} {d} removed={d} merges={d} borrows={d} height={d} writes={d} frees={d}", .{ first, second, removed, tree.merges - merges, tree.borrows - borrows, tree.height(), changes.writes, changes.frees });
630 },
631 .GET => {
632 const found = tree.search(first);
633 try out.print("GET {d} ", .{first});
634 if (found.page) |page| {
635 try out.print("FOUND value={f}", .{tree.node(page).entries.items[found.position].value});
636 } else try out.writeAll("NOT FOUND");
637 try out.writeAll(" path=");
638 for (found.path[0..found.len], 0..) |page, i| {
639 if (i > 0) try out.writeAll(" -> ");
640 try out.print("{f}", .{tree.node(page).*});
641 }
642 try out.writeByte('\n');
643 },
644 .RANGE, .COUNT => {
645 var result: std.ArrayList(Entry) = .empty;
646 var visited: usize = 0;
647 try tree.scanNode(tree.root, first, second, session.gpa, &result, &visited);
648 try out.print("{s} {d} {d} count={d} visited={d}", .{ words[0], first, second, result.items.len, visited });
649 if (command == .RANGE) for (result.items) |entry| {
650 try out.print(" {f}", .{entry});
651 };
652 try out.writeByte('\n');
653 },
654 .MIN, .MAX => {
655 try out.print("{s} ", .{words[0]});
656 if (rows.len == 0) try line(out, "EMPTY", .{}) else try line(out, "{f}", .{rows[if (command == .MIN) 0 else rows.len - 1]});
657 },
658 .FLOOR, .CEIL => {
659 const entry = candidate(rows, first, if (command == .FLOOR) .at_most else .at_least, command == .FLOOR);
660 try out.print("{s} {d} ", .{ words[0], first });
661 if (entry) |value| try line(out, "{f}", .{value}) else try line(out, "NONE", .{});
662 },
663 .RANK => {
664 var rank: usize = 0;
665 for (rows) |entry| if (entry.key < first) {
666 rank += 1;
667 };
668 try line(out, "RANK {d} {d}", .{ first, rank });
669 },
670 .SELECT => try line(out, "SELECT {d} {f}", .{ first, rows[@intCast(first - 1)] }),
671 .SEEK => {
672 const entry = candidate(rows, first, .at_least, false);
673 lab.cursor = if (entry) |value| .{ .key = value.key } else .end;
674 try out.print("SEEK {d} ", .{first});
675 if (entry) |value| try line(out, "{f}", .{value}) else try line(out, "END", .{});
676 },
677 .NEXT, .PREV => {
678 const entry = if (command == .NEXT) switch (lab.cursor) {
679 .begin => candidate(rows, 0, .all, false),
680 .key => |key| candidate(rows, key, .greater, false),
681 .end => null,
682 .unset => unreachable,
683 } else switch (lab.cursor) {
684 .end => candidate(rows, 0, .all, true),
685 .key => |key| candidate(rows, key, .less, true),
686 .begin => null,
687 .unset => unreachable,
688 };
689 lab.cursor = if (entry) |value| .{ .key = value.key } else if (command == .NEXT) .end else .begin;
690 try out.print("{s} ", .{words[0]});
691 if (entry) |value| try line(out, "{f}", .{value}) else try line(out, "{s}", .{if (command == .NEXT) "END" else "BEGIN"});
692 },
693 .PRINT => {
694 var level: std.ArrayList(usize) = .empty;
695 try level.append(session.gpa, tree.root);
696 var depth: usize = 0;
697 while (level.items.len > 0) : (depth += 1) {
698 var next: std.ArrayList(usize) = .empty;
699 try out.print("LEVEL {d}", .{depth});
700 for (level.items) |page| {
701 const item = tree.node(page);
702 try out.print(" {f}", .{item.*});
703 try next.appendSlice(session.gpa, item.children.items);
704 }
705 try out.writeByte('\n');
706 level = next;
707 }
708 },
709 .PAGES => {
710 for (tree.pages.items, 0..) |slot, page| if (slot) |item| {
711 try out.print("PAGE {d} {f} kids=", .{ page, item.* });
712 if (item.leaf()) try out.writeByte('-') else for (item.children.items, 0..) |child, i| {
713 if (i > 0) try out.writeByte(',');
714 try out.print("{d}", .{child});
715 }
716 try out.writeByte('\n');
717 };
718 },
719 .DUMP => {
720 try out.print("DUMP {d} ", .{tree.degree});
721 try tree.dumpNode(tree.root, out);
722 try out.writeByte('\n');
723 },
724 .HEIGHT => try line(out, "HEIGHT {d}", .{tree.height()}),
725 .STATS => {
726 var leaves: usize = 0;
727 for (tree.pages.items) |slot| if (slot) |item| {
728 if (item.leaf()) leaves += 1;
729 };
730 const nodes = tree.countNodes();
731 const fill = 100 * rows.len / ((2 * tree.degree - 1) * nodes);
732 try line(out, "STATS t={d} nodes={d} leaves={d} keys={d} fill={d}% splits={d} merges={d} borrows={d}", .{ tree.degree, nodes, leaves, rows.len, fill, tree.splits, tree.merges, tree.borrows });
733 },
734 .VERIFY => {
735 if (try tree.violation(session.gpa)) |problem| try line(out, "VERIFY FAIL {s}", .{problem}) else try line(out, "VERIFY OK", .{});
736 },
737 .REBUILD => {
738 var replacement = try Tree.init(lab.gpa, tree.degree);
739 errdefer replacement.deinit();
740 for (rows) |entry| _ = try replacement.upsert(entry.key, entry.value, &trace);
741 try line(out, "REBUILD entries={d} keys={d} nodes={d} splits={d} height={d}", .{ rows.len, replacement.countEntries(), replacement.countNodes(), replacement.splits, replacement.height() });
742 lab.replace(replacement);
743 },
744 .COMPACT => {
745 const nodes = tree.countNodes();
746 try tree.compact(session.gpa);
747 try line(out, "COMPACT nodes={d}->{d} height={d}", .{ nodes, tree.countNodes(), tree.height() });
748 },
749 .SAVE => {
750 const replacement = try tree.clone();
751 if (lab.saved) |*old| old.deinit();
752 lab.saved = replacement;
753 try session.summary("SAVE", tree);
754 },
755 .RESTORE => {
756 const replacement = try lab.saved.?.clone();
757 lab.replace(replacement);
758 try session.summary("RESTORE", &lab.index.?);
759 },
760 .CREATE, .LOAD, .TRACE => unreachable,
761 }
762 }
763};

765fn readLine(reader: *std.Io.Reader, gpa: Allocator) !?[]const u8 {
766 var buffer: Writer.Allocating = .init(gpa);
767 defer buffer.deinit();
768 _ = reader.streamDelimiterEnding(&buffer.writer, '\n') catch |err| switch (err) {
769 error.EndOfStream => {
770 if (buffer.written().len == 0) return null;
771 return try buffer.toOwnedSlice();
772 },
773 else => return err,
774 };
775 reader.toss(1);
776 return try buffer.toOwnedSlice();
777}
778fn readWords(gpa: Allocator, source: []const u8) Allocator.Error![]const []const u8 {
779 var words: std.ArrayList([]const u8) = .empty;
780 var tokens = std.mem.tokenizeAny(u8, source, " \r");
781 while (tokens.next()) |word| try words.append(gpa, word);
782 return words.toOwnedSlice(gpa);
783}
784pub fn main(init: std.process.Init) !void {
785 var input_buffer: [4096]u8 = undefined;
786 var output_buffer: [4096]u8 = undefined;
787 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
788 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
789 var first_arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
790 defer first_arena.deinit();
791 const count: usize = @intCast(integer(std.mem.trim(u8, (try readLine(&input.interface, first_arena.allocator())).?, "\r")));
792 var lab: Workbench = .{ .gpa = std.heap.smp_allocator };
793 defer lab.deinit();
794 for (0..count) |_| {
795 var arena: std.heap.ArenaAllocator = .init(std.heap.page_allocator);
796 defer arena.deinit();
797 const gpa = arena.allocator();
798 const words = try readWords(gpa, (try readLine(&input.interface, gpa)).?);
799 var response: Writer.Allocating = .init(gpa);
800 defer response.deinit();
801 var session: Session = .{ .gpa = gpa, .out = &response.writer, .lab = &lab };
802 session.execute(words) catch |err| switch (err) {
803 error.Invalid => {
804 try line(&output.interface, "ERROR {s}", .{session.problem});
805 try output.interface.flush();
806 continue;
807 },
808 else => return err,
809 };
810 try output.interface.writeAll(response.written());
811 try output.interface.flush();
812 }
