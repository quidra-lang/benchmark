75check_tree <- function(tree) {
76 reference <- NULL
77 bound_text <- function(value, lower) if (is.null(value)) if (lower) "-inf" else "+inf" else as.character(value)
78 visit <- function(page, location, depth, lower, upper) {
79 value <- get_node(tree, page)
80 count <- length(value$keys)
81 minimum <- if (page == tree$root) as.integer(length(value$kids) > 0L) else tree$degree - 1L
82 maximum <- 2L * tree$degree - 1L
83 where <- paste0(" at=", location, " node=", notation(value))
84 if (count < minimum || count > maximum)
85 return(paste0("FAIL COUNT", where, " keys=", count, " allowed=", minimum, "..", maximum))
86 if (count > 1L) {
87 bad <- which(value$keys[-count] >= value$keys[-1L])
88 if (length(bad)) return(paste0("FAIL ORDER", where, " ", value$keys[bad[1L]], ">=", value$keys[bad[1L] + 1L]))
89 }
90 for (key in value$keys) {
91 if ((!is.null(lower) && key <= lower) || (!is.null(upper) && key >= upper))
92 return(paste0("FAIL RANGE", where, " key=", key, " low=", bound_text(lower, TRUE), " high=", bound_text(upper, FALSE)))
93 }
94 if (!length(value$kids)) {
95 if (is.null(reference)) reference <<- depth else if (depth != reference)
96 return(paste0("FAIL DEPTH", where, " depth=", depth, " expected=", reference))
97 } else {
98 for (j in seq_along(value$kids)) {
99 problem <- visit(value$kids[j], paste0(location, ".", j - 1L), depth + 1L,
100 if (j == 1L) lower else value$keys[j - 1L],
101 if (j > count) upper else value$keys[j])
102 if (!is.null(problem)) return(problem)
103 }
104 }
105 NULL
106 }
107 visit(tree$root, "root", 0L, NULL, NULL)

129split_child <- function(tree, page, at, context) {
130 parent <- get_node(tree, page)
131 left_page <- parent$kids[at]
132 left <- get_node(tree, left_page)
133 degree <- tree$degree
134 middle_key <- left$keys[degree]
135 middle_value <- left$values[degree]
136 right_indices <- seq.int(degree + 1L, length(left$keys))
137 right <- node(left$keys[right_indices], left$values[right_indices],
138 if (length(left$kids)) left$kids[seq.int(degree + 1L, 2L * degree)] else integer())
139 right_page <- allocate_node(tree, right)
140 step(context, paste0("STEP split page=", left_page, " keys=", notation(left), " up=", middle_key, " new=", right_page))
141 left$keys <- left$keys[seq_len(degree - 1L)]
142 left$values <- left$values[seq_len(degree - 1L)]
143 if (length(left$kids)) left$kids <- left$kids[seq_len(degree)]
144 parent$keys <- append(parent$keys, middle_key, after = at - 1L)
145 parent$values <- append(parent$values, middle_value, after = at - 1L)
146 parent$kids <- append(parent$kids, right_page, after = at)
147 set_node(tree, left_page, left)
148 set_node(tree, page, parent)
149 tree$counters["splits"] <- tree$counters["splits"] + 1L

151insert_entry <- function(tree, key, text, context = NULL) {
152 found <- search_tree(tree, key)
153 if (found$found) {
154 value <- get_node(tree, found$page)
155 old <- value$values[found$at]
156 value$values[found$at] <- text
157 set_node(tree, found$page, value)
158 return(list(added = FALSE, old = old))
159 }
160 if (length(get_node(tree, tree$root)$keys) == 2L * tree$degree - 1L) {
161 tree$root <- allocate_node(tree, node(kids = tree$root))
162 step(context, paste0("STEP grow page=", tree$root))
163 split_child(tree, tree$root, 1L, context)
164 }
165 page <- tree$root
166 repeat {
167 value <- get_node(tree, page)
168 at <- position(value, key)
169 if (!length(value$kids)) break
170 if (length(get_node(tree, value$kids[at])$keys) == 2L * tree$degree - 1L) {
171 split_child(tree, page, at, context)
172 value <- get_node(tree, page)
173 if (key > value$keys[at]) at <- at + 1L
174 }
175 page <- value$kids[at]
176 }
177 value$keys <- append(value$keys, key, after = at - 1L)
178 value$values <- append(value$values, text, after = at - 1L)
179 set_node(tree, page, value)
180 step(context, paste0("STEP place page=", page, " keys=", notation(value)))
181 list(added = TRUE, old = NULL)

183merge_children <- function(tree, page, at, name, context) {
184 parent <- get_node(tree, page)
185 left_page <- parent$kids[at]
186 right_page <- parent$kids[at + 1L]
187 left <- get_node(tree, left_page)
188 right <- get_node(tree, right_page)
189 left$keys <- c(left$keys, parent$keys[at], right$keys)
190 left$values <- c(left$values, parent$values[at], right$values)
191 left$kids <- c(left$kids, right$kids)
192 parent$keys <- parent$keys[-at]
193 parent$values <- parent$values[-at]
194 parent$kids <- parent$kids[-(at + 1L)]
195 set_node(tree, left_page, left)
196 set_node(tree, page, parent)
197 release_node(tree, right_page)
198 tree$counters["merges"] <- tree$counters["merges"] + 1L
199 step(context, paste0("STEP ", name, " page=", page, " merge=", left_page, "+", right_page, " keys=", notation(left)))
200 if (page == tree$root && !length(parent$keys)) {
201 tree$root <- left_page
202 release_node(tree, page)
203 step(context, paste0("STEP shrink page=", left_page))
204 }
205 left_page

207borrow_child <- function(tree, page, at, from_left, context) {
208 parent <- get_node(tree, page)
209 child_page <- parent$kids[at]
210 sibling_page <- parent$kids[at + if (from_left) -1L else 1L]
211 child <- get_node(tree, child_page)
212 sibling <- get_node(tree, sibling_page)
213 separator <- at - as.integer(from_left)
214 take <- if (from_left) length(sibling$keys) else 1L
215 child$keys <- append(child$keys, parent$keys[separator], after = if (from_left) 0L else length(child$keys))
216 child$values <- append(child$values, parent$values[separator], after = if (from_left) 0L else length(child$values))
217 parent$keys[separator] <- sibling$keys[take]
218 parent$values[separator] <- sibling$values[take]
219 sibling$keys <- sibling$keys[-take]
220 sibling$values <- sibling$values[-take]
221 if (length(sibling$kids)) {
222 take_child <- if (from_left) length(sibling$kids) else 1L
223 child$kids <- append(child$kids, sibling$kids[take_child], after = if (from_left) 0L else length(child$kids))
224 sibling$kids <- sibling$kids[-take_child]
225 }
226 set_node(tree, child_page, child)
227 set_node(tree, sibling_page, sibling)
228 set_node(tree, page, parent)
229 tree$counters["borrows"] <- tree$counters["borrows"] + 1L
230 step(context, paste0("STEP 3a-", if (from_left) "left" else "right", " page=", page, " child=", child_page,
231 " from=", sibling_page, " keys=", notation(child)))
232}
233delete_entry <- function(tree, page, key, context = NULL) {
234 value <- get_node(tree, page)
235 at <- position(value, key)
236 if (at <= length(value$keys) && value$keys[at] == key) {
237 if (!length(value$kids)) {
238 add_case(context, "1")
239 value$keys <- value$keys[-at]
240 value$values <- value$values[-at]
241 set_node(tree, page, value)
242 step(context, paste0("STEP 1 page=", page, " keys=", notation(value)))
243 } else {
244 left <- value$kids[at]
245 right <- value$kids[at + 1L]
246 if (length(get_node(tree, left)$keys) >= tree$degree || length(get_node(tree, right)$keys) >= tree$degree) {
247 predecessor <- length(get_node(tree, left)$keys) >= tree$degree
248 child <- if (predecessor) left else right
249 extreme <- get_node(tree, child)
250 while (length(extreme$kids)) extreme <- get_node(tree, extreme$kids[if (predecessor) length(extreme$kids) else 1L])
251 pick <- if (predecessor) length(extreme$keys) else 1L
252 replacement_key <- extreme$keys[pick]
253 replacement_value <- extreme$values[pick]
254 name <- if (predecessor) "2a" else "2b"
255 add_case(context, name)
256 step(context, paste0("STEP ", name, " page=", page, if (predecessor) " pred=" else " succ=", replacement_key))
257 delete_entry(tree, child, replacement_key, context)
258 value <- get_node(tree, page)
259 value$keys[at] <- replacement_key
260 value$values[at] <- replacement_value
261 set_node(tree, page, value)
262 } else {
263 add_case(context, "2c")
264 child <- merge_children(tree, page, at, "2c", context)
265 delete_entry(tree, child, key, context)
266 }
267 }
268 } else {
269 child <- value$kids[at]
270 if (length(get_node(tree, child)$keys) == tree$degree - 1L) {
271 if (at > 1L && length(get_node(tree, value$kids[at - 1L])$keys) >= tree$degree) {
272 add_case(context, "3a-left")
273 borrow_child(tree, page, at, TRUE, context)
274 } else if (at < length(value$kids) && length(get_node(tree, value$kids[at + 1L])$keys) >= tree$degree) {
275 add_case(context, "3a-right")
276 borrow_child(tree, page, at, FALSE, context)
277 } else {
278 name <- if (at > 1L) "3b-left" else "3b-right"
279 add_case(context, name)
280 child <- merge_children(tree, page, if (at > 1L) at - 1L else at, name, context)
281 }
282 }
283 delete_entry(tree, child, key, context)
284 }
285 invisible(NULL)
286}
287scan_tree <- function(tree, lower, upper) {
288 visited <- 0L
289 result <- list(keys = integer(), values = character())
290 visit <- function(page) {
291 visited <<- visited + 1L
292 value <- get_node(tree, page)
293 count <- length(value$keys)
294 for (j in seq_len(count + 1L)) {
295 if (length(value$kids) && (j == 1L || value$keys[j - 1L] < upper) && (j > count || value$keys[j] > lower)) visit(value$kids[j])
296 if (j <= count && value$keys[j] >= lower && value$keys[j] <= upper) {
297 result$keys <<- c(result$keys, value$keys[j])
298 result$values <<- c(result$values, value$values[j])
299 }
300 }
301 }
302 visit(tree$root)
303 result$visited <- visited
304 result

315load_tree <- function(degree, tokens) {
316 tree <- new_tree(degree)
317 tree$pages <- list()
318 at <- 1L
319 peek <- function() if (at <= length(tokens)) tokens[at] else ""
320 syntax <- function() fail(paste("ERROR SYNTAX", at))
321 expect <- function(text) {
322 if (peek() != text) syntax()
323 at <<- at + 1L
324 }
325 read_entry <- function() {
326 text <- peek()
327 if (text %in% c("", "(", ")")) syntax()
328 fields <- strsplit(text, "=", fixed = TRUE)[[1L]]
329 at <<- at + 1L
330 list(key = as.integer(fields[1L]), value = fields[2L])
331 }
332 read_node <- function() {
333 expect("(")
334 page <- allocate_node(tree)
335 value <- node()
336 internal <- peek() == "("
337 if (internal) value$kids <- read_node()
338 while (peek() != ")") {
339 entry <- read_entry()
340 value$keys <- c(value$keys, entry$key)
341 value$values <- c(value$values, entry$value)
342 if (internal) value$kids <- c(value$kids, read_node())
343 }
344 expect(")")
345 set_node(tree, page, value)
346 page
347 }
348 tree$root <- read_node()
349 if (at <= length(tokens)) syntax()
350 tree
351}
352compact_tree <- function(source) {
353 entries <- inorder(source)
354 candidates <- list()
355 degree <- source$degree
356 repeat {
357 count <- length(entries$keys)
358 groups <- (count + 2L * degree) %/% (2L * degree)
359 kept <- count - groups + 1L
360 sizes <- rep.int(kept %/% groups, groups) + as.integer(seq_len(groups) <= kept %% groups)
361 level <- vector("list", groups)
362 separators <- list(keys = integer(), values = character())
363 at <- 1L
364 child_at <- 1L
365 for (i in seq_len(groups)) {
366 chosen <- at - 1L + seq_len(sizes[i])
367 kids <- if (length(candidates)) candidates[child_at - 1L + seq_len(sizes[i] + 1L)] else list()
368 level[[i]] <- list(keys = entries$keys[chosen], values = entries$values[chosen], kids = kids)
369 at <- at + sizes[i]
370 child_at <- child_at + sizes[i] + 1L
371 if (i < groups) {
372 separators$keys <- c(separators$keys, entries$keys[at])
373 separators$values <- c(separators$values, entries$values[at])
374 at <- at + 1L
375 }
376 }
377 if (groups == 1L) break
378 entries <- separators
379 candidates <- level
380 }
381 tree <- new_tree(degree)
382 tree$pages <- list()
383 assign_pages <- function(value) {
384 page <- allocate_node(tree)
385 kids <- vapply(value$kids, assign_pages, integer(1L))
386 set_node(tree, page, node(value$keys, value$values, kids))
387 page
388 }
389 tree$root <- assign_pages(level[[1L]])
390 tree
391}

393execute <- function(state, words) {
394 command <- words[1L]
395 args <- words[-1L]
396 if (command == "TRACE") {
397 state$trace <- args[1L] == "ON"
398 return(paste("TRACE", args[1L]))
399 }
400 if (command %in% c("CREATE", "LOAD")) {
401 degree <- as.integer(args[1L])
402 if (degree < 2L || degree > 5L) fail(paste("ERROR BAD DEGREE", degree))
403 tree <- if (command == "CREATE") new_tree(degree) else load_tree(degree, args[-1L])
404 state$index <- tree
405 state$cursor <- list(kind = "UNSET", key = 0L)
406 if (command == "CREATE") return(paste0("CREATE t=", degree))
407 return(paste0("LOAD t=", degree, " nodes=", length(live_pages(tree$pages)), " keys=", key_count(tree), " ", tree_status(tree)))
408 }
409 tree <- state$index
410 if (is.null(tree)) fail("ERROR NO INDEX")
411 needs_valid <- c("INSERT", "BULKLOAD", "DELETE", "DELRANGE", "RANGE", "COUNT", "MIN", "MAX", "FLOOR", "CEIL", "RANK", "SELECT", "SEEK", "NEXT", "PREV", "HEIGHT", "COMPACT")
412 if (command %in% needs_valid && !is.null(check_tree(tree))) fail("ERROR DAMAGED")
413 if (command %in% c("DELRANGE", "RANGE", "COUNT") && as.integer(args[1L]) > as.integer(args[2L]))
414 fail(paste("ERROR BAD RANGE", args[1L], args[2L]))
415 if (command == "SELECT" && (as.integer(args[1L]) < 1L || as.integer(args[1L]) > key_count(tree)))
416 fail(paste("ERROR BAD POSITION", args[1L]))
417 if (command %in% c("NEXT", "PREV") && state$cursor$kind == "UNSET") fail("ERROR NO CURSOR")
418 if (command == "RESTORE" && is.null(state$saved)) fail("ERROR NO SAVE")
419 switch(command,
420 INSERT = {
421 key <- as.integer(args[1L])
422 before <- tree$pages
423 splits <- tree$counters["splits"]
424 context <- trace_context(state$trace)
425 result <- insert_entry(tree, key, args[2L], context)
426 delta <- changes(tree, before)
427 text <- if (result$added) paste0("INSERT ", key, " NEW splits=", tree$counters["splits"] - splits, " height=", height(tree), " writes=", delta["writes"]) else
428 paste0("INSERT ", key, " UPDATED old=", result$old, " writes=", delta["writes"])
429 c(context$lines, text)
430 },
431 BULKLOAD = {
432 before <- tree$pages
433 splits <- tree$counters["splits"]
434 added <- 0L
435 for (i in seq.int(1L, length(args), by = 2L)) added <- added + as.integer(insert_entry(tree, as.integer(args[i]), args[i + 1L])$added)
436 pairs <- length(args) %/% 2L
437 paste0("BULKLOAD pairs=", pairs, " new=", added, " updated=", pairs - added, " splits=", tree$counters["splits"] - splits,
438 " height=", height(tree), " writes=", changes(tree, before)["writes"])
439 },
440 DELETE = {
441 key <- as.integer(args[1L])
442 found <- search_tree(tree, key)
443 if (!found$found) return(paste("DELETE", key, "NOT FOUND"))
444 old <- get_node(tree, found$page)$values[found$at]
445 before <- tree$pages
446 context <- trace_context(state$trace)
447 delete_entry(tree, tree$root, key, context)
448 delta <- changes(tree, before)
449 c(context$lines, paste0("DELETE ", key, " REMOVED value=", old, " cases=", paste(context$cases, collapse = ","),
450 " height=", height(tree), " writes=", delta["writes"], " frees=", delta["frees"]))
451 },
452 DELRANGE = {
453 lower <- as.integer(args[1L])
454 upper <- as.integer(args[2L])
455 before <- tree$pages
456 counts <- tree$counters
457 keys <- inorder(tree)$keys
458 keys <- keys[keys >= lower & keys <= upper]
459 for (key in keys) delete_entry(tree, tree$root, key)
460 delta <- changes(tree, before)
461 paste0("DELRANGE ", lower, " ", upper, " removed=", length(keys), " merges=", tree$counters["merges"] - counts["merges"],
462 " borrows=", tree$counters["borrows"] - counts["borrows"], " height=", height(tree), " writes=", delta["writes"], " frees=", delta["frees"])
463 },
464 GET = {
465 key <- as.integer(args[1L])
466 result <- search_tree(tree, key)
467 outcome <- if (result$found) paste0("FOUND value=", get_node(tree, result$page)$values[result$at]) else "NOT FOUND"
468 path <- vapply(result$path, function(page) notation(get_node(tree, page)), character(1L))
469 paste0("GET ", key, " ", outcome, " path=", paste(path, collapse = " -> "))
470 },
471 RANGE =, COUNT = {
472 result <- scan_tree(tree, as.integer(args[1L]), as.integer(args[2L]))
473 text <- paste0(command, " ", args[1L], " ", args[2L], " count=", length(result$keys), " visited=", result$visited)
474 if (command == "RANGE" && length(result$keys)) paste(text, paste(entry_text(result$keys, result$values), collapse = " ")) else text
475 },
476 MIN =, MAX =, FLOOR =, CEIL =, RANK =, SELECT =, SEEK =, NEXT =, PREV = {
477 entries <- inorder(tree)
478 keys <- entries$keys
479 displays <- if (length(keys)) entry_text(keys, entries$values) else character()
480 if (command %in% c("MIN", "MAX")) {
481 if (!length(keys)) return(paste(command, "EMPTY"))
482 return(paste(command, displays[if (command == "MIN") 1L else length(keys)]))
483 }
484 if (command == "SELECT") return(paste("SELECT", args[1L], displays[as.integer(args[1L])]))
485 if (command == "RANK") return(paste("RANK", args[1L], sum(keys < as.integer(args[1L]))))
486 if (command %in% c("FLOOR", "CEIL", "SEEK")) {
487 key <- as.integer(args[1L])
488 indices <- if (command == "FLOOR") which(keys <= key) else which(keys >= key)
489 chosen <- if (length(indices)) indices[if (command == "FLOOR") length(indices) else 1L] else NA_integer_
490 if (command == "SEEK") state$cursor <- if (is.na(chosen)) list(kind = "END", key = 0L) else list(kind = "KEY", key = keys[chosen])
491 return(paste(command, key, if (!is.na(chosen)) displays[chosen] else if (command == "SEEK") "END" else "NONE"))
492 }
493 cursor <- state$cursor
494 if (command == "NEXT") {
495 indices <- if (cursor$kind == "END") integer() else if (cursor$kind == "BEGIN") seq_along(keys) else which(keys > cursor$key)
496 chosen <- if (length(indices)) indices[1L] else NA_integer_
497 } else {
498 indices <- if (cursor$kind == "BEGIN") integer() else if (cursor$kind == "END") seq_along(keys) else which(keys < cursor$key)
499 chosen <- if (length(indices)) indices[length(indices)] else NA_integer_
500 }
501 if (is.na(chosen)) {
502 state$cursor <- list(kind = if (command == "NEXT") "END" else "BEGIN", key = 0L)
503 paste(command, state$cursor$kind)
504 } else {
505 state$cursor <- list(kind = "KEY", key = keys[chosen])
506 paste(command, displays[chosen])
507 }
508 },
509 PRINT = {
510 level <- tree$root
511 depth <- 0L
512 result <- character()
