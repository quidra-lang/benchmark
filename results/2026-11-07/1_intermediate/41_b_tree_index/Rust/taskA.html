17#[derive(Clone, Default, PartialEq, Eq)]
18struct Node {
19 entries: Vec<Entry>,
20 children: Vec<usize>,
21}

23impl Node {
24 fn notation(&self) -> String {
25 format!("[{}]", self.entries.iter().map(|e| e.key.to_string()).collect::<Vec<_>>().join(" "))
26 }
27
28 fn position(&self, key: i32) -> usize {
29 self.entries.iter().position(|entry| key <= entry.key).unwrap_or(self.entries.len())
30 }

47#[derive(Clone)]
48struct Tree {
49 degree: usize,
50 root: usize,
51 pages: BTreeMap<usize, Node>,
52 splits: usize,
53 merges: usize,
54 borrows: usize,
55}
56
57impl Tree {
58 fn new(degree: usize) -> Self {
59 Self { degree, root: 0, pages: BTreeMap::from([(0, Node::default())]), splits: 0, merges: 0, borrows: 0 }
60 }
61
62 fn node(&self, page: usize) -> &Node {
63 &self.pages[&page]
64 }
65
66 fn node_mut(&mut self, page: usize) -> &mut Node {
67 self.pages.get_mut(&page).unwrap()
68 }
69
70 fn allocate(&mut self, node: Node) -> usize {
71 let mut page = 0;
72 while self.pages.contains_key(&page) {
73 page += 1;
74 }
75 self.pages.insert(page, node);
76 page
77 }
78
79 fn entries(&self) -> Vec<Entry> {
80 fn visit(tree: &Tree, page: usize, result: &mut Vec<Entry>) {
81 let node = tree.node(page);
82 for j in 0..=node.entries.len() {
83 if !node.children.is_empty() {
84 visit(tree, node.children[j], result);
85 }
86 if let Some(entry) = node.entries.get(j) {
87 result.push(entry.clone());
88 }
89 }
90 }
91 let mut result = Vec::new();
92 visit(self, self.root, &mut result);
93 result
94 }
95
96 fn search(&self, key: i32) -> (Option<(usize, usize)>, Vec<String>) {
97 let mut page = self.root;
98 let mut path = Vec::new();
99 loop {
100 let node = self.node(page);
101 path.push(node.notation());
102 let position = node.position(key);
103 if node.entries.get(position).is_some_and(|entry| entry.key == key) {
104 return (Some((page, position)), path);
105 }
106 if node.children.is_empty() {
107 return (None, path);
108 }
109 page = node.children[position];
110 }
111 }
112
113 fn violation(&self) -> Option<String> {
114 fn visit(tree: &Tree, page: usize, location: String, depth: usize, low: Option<i32>, high: Option<i32>, reference: &mut Option<usize>) -> Option<String> {
115 let node = tree.node(page);
116 let count = node.entries.len();
117 let minimum = if page == tree.root { usize::from(!node.children.is_empty()) } else { tree.degree - 1 };
118 let maximum = 2 * tree.degree - 1;
119 let details = format!("at={location} node={}", node.notation());
120 if count < minimum || count > maximum {
121 return Some(format!("COUNT {details} keys={count} allowed={minimum}..{maximum}"));
122 }
123 for pair in node.entries.windows(2) {
124 if pair[0].key >= pair[1].key {
125 return Some(format!("ORDER {details} {}>={}", pair[0].key, pair[1].key));
126 }
127 }
128 for entry in &node.entries {
129 if low.is_some_and(|bound| entry.key <= bound) || high.is_some_and(|bound| entry.key >= bound) {
130 return Some(format!("RANGE {details} key={} low={} high={}", entry.key,
131 low.map_or_else(|| "-inf".into(), |n| n.to_string()), high.map_or_else(|| "+inf".into(), |n| n.to_string())));
132 }
133 }
134 if node.children.is_empty() {
135 if let Some(expected) = *reference {
136 if depth != expected {
137 return Some(format!("DEPTH {details} depth={depth} expected={expected}"));
138 }
139 } else {
140 *reference = Some(depth);
141 }
142 }
143 for (j, &child) in node.children.iter().enumerate() {
144 let lower = if j == 0 { low } else { Some(node.entries[j - 1].key) };
145 let upper = if j == count { high } else { Some(node.entries[j].key) };
146 if let Some(error) = visit(tree, child, format!("{location}.{j}"), depth + 1, lower, upper, reference) {
147 return Some(error);
148 }
149 }
150 None
151 }
152 visit(self, self.root, "root".into(), 0, None, None, &mut None)
153 }
154
155 fn status(&self) -> &'static str {
156 if self.violation().is_none() { "OK" } else { "DAMAGED" }
157 }
158
159 fn height(&self) -> usize {
160 let mut page = self.root;
161 let mut depth = 0;
162 while let Some(&child) = self.node(page).children.first() {
163 page = child;
164 depth += 1;
165 }
166 depth
167 }

169 fn load(degree: usize, tokens: &[&str]) -> Result<Self, usize> {
170 struct Parser<'a> {
171 tokens: &'a [&'a str],
172 position: usize,
173 tree: Tree,
174 }
175 impl Parser<'_> {
176 fn entry(&mut self) -> Result<Entry, usize> {
177 let token = self.tokens.get(self.position).ok_or(self.position + 1)?;
178 let (key, value) = token.split_once('=').ok_or(self.position + 1)?;
179 self.position += 1;
180 Ok(Entry { key: key.parse().unwrap(), value: value.into() })
181 }
182
183 fn node(&mut self) -> Result<usize, usize> {
184 if self.tokens.get(self.position) != Some(&"(") {
185 return Err(self.position + 1);
186 }
187 self.position += 1;
188 let page = self.tree.allocate(Node::default());
189 let mut node = Node::default();
190 if self.tokens.get(self.position) == Some(&"(") {
191 node.children.push(self.node()?);
192 while self.tokens.get(self.position) != Some(&")") {
193 node.entries.push(self.entry()?);
194 node.children.push(self.node()?);
195 }
196 } else {
197 while self.tokens.get(self.position) != Some(&")") {
198 node.entries.push(self.entry()?);
199 }
200 }
201 self.position += 1;
202 self.tree.pages.insert(page, node);
203 Ok(page)
204 }
205 }
206 let mut parser = Parser { tokens, position: 0, tree: Self::new(degree) };
207 parser.tree.pages.clear();
208 parser.tree.root = parser.node()?;
209 if parser.position != tokens.len() {
210 return Err(parser.position + 1);
211 }
212 Ok(parser.tree)
213 }
214
215 fn changes(&self, before: &BTreeMap<usize, Node>) -> (usize, usize) {
216 let writes = self.pages.iter().filter(|&(id, node)| before.get(id) != Some(node)).count();
217 let frees = before.keys().filter(|id| !self.pages.contains_key(*id)).count();
218 (writes, frees)
219 }
220
221 fn split(&mut self, parent: usize, position: usize, trace: &mut Trace) {
222 let child = self.node(parent).children[position];
223 let old_notation = self.node(child).notation();
224 let degree = self.degree;
225 let (middle, right) = {
226 let node = self.node_mut(child);
227 let entries = node.entries.split_off(degree);
228 let children = if node.children.is_empty() { Vec::new() } else { node.children.split_off(degree) };
229 (node.entries.pop().unwrap(), Node { entries, children })
230 };
231 let new_page = self.allocate(right);
232 let middle_key = middle.key;
233 let node = self.node_mut(parent);
234 node.entries.insert(position, middle);
235 node.children.insert(position + 1, new_page);
236 self.splits += 1;
237 trace.emit(format!("STEP split page={child} keys={old_notation} up={middle_key} new={new_page}"));
238 }
239
240 fn upsert(&mut self, entry: Entry, trace: &mut Trace) -> Option<String> {
241 if let Some((page, position)) = self.search(entry.key).0 {
242 return Some(std::mem::replace(&mut self.node_mut(page).entries[position].value, entry.value));
243 }
244 if self.node(self.root).entries.len() == 2 * self.degree - 1 {
245 self.root = self.allocate(Node { entries: Vec::new(), children: vec![self.root] });
246 trace.emit(format!("STEP grow page={}", self.root));
247 self.split(self.root, 0, trace);
248 }
249 let mut page = self.root;
250 while !self.node(page).children.is_empty() {
251 let mut position = self.node(page).position(entry.key);
252 let child = self.node(page).children[position];
253 if self.node(child).entries.len() == 2 * self.degree - 1 {
254 self.split(page, position, trace);
255 if entry.key > self.node(page).entries[position].key {
256 position += 1;
257 }
258 }
259 page = self.node(page).children[position];
260 }
261 let position = self.node(page).position(entry.key);
262 self.node_mut(page).entries.insert(position, entry);
263 trace.emit(format!("STEP place page={page} keys={}", self.node(page).notation()));
264 None
265 }
266
267 fn extreme(&self, mut page: usize, last: bool) -> Entry {
268 loop {
269 let node = self.node(page);
270 if node.children.is_empty() {
271 return if last { node.entries.last().unwrap() } else { &node.entries[0] }.clone();
272 }
273 page = if last { *node.children.last().unwrap() } else { node.children[0] };
274 }
275 }
276
277 fn merge(&mut self, parent: usize, position: usize, case: &str, trace: &mut Trace) -> usize {
278 let left = self.node(parent).children[position];
279 let right = self.node(parent).children[position + 1];
280 let removed = self.pages.remove(&right).unwrap();
281 let node = self.node_mut(parent);
282 let middle = node.entries.remove(position);
283 node.children.remove(position + 1);
284 let node = self.node_mut(left);
285 node.entries.push(middle);
286 node.entries.extend(removed.entries);
287 node.children.extend(removed.children);
288 self.merges += 1;
289 trace.emit(format!("STEP {case} page={parent} merge={left}+{right} keys={}", self.node(left).notation()));
290 if parent == self.root && self.node(parent).entries.is_empty() {
291 self.root = left;
292 self.pages.remove(&parent);
293 trace.emit(format!("STEP shrink page={left}"));
294 }
295 left
296 }

298 fn remove_at(&mut self, page: usize, key: i32, cases: &mut Vec<&'static str>, trace: &mut Trace) {
299 let position = self.node(page).position(key);
300 if self.node(page).entries.get(position).is_some_and(|entry| entry.key == key) {
301 if self.node(page).children.is_empty() {
302 cases.push("1");
303 self.node_mut(page).entries.remove(position);
304 trace.emit(format!("STEP 1 page={page} keys={}", self.node(page).notation()));
305 return;
306 }
307 let left = self.node(page).children[position];
308 let right = self.node(page).children[position + 1];
309 if self.node(left).entries.len() >= self.degree {
310 cases.push("2a");
311 let predecessor = self.extreme(left, true);
312 trace.emit(format!("STEP 2a page={page} pred={}", predecessor.key));
313 self.remove_at(left, predecessor.key, cases, trace);
314 self.node_mut(page).entries[position] = predecessor;
315 } else if self.node(right).entries.len() >= self.degree {
316 cases.push("2b");
317 let successor = self.extreme(right, false);
318 trace.emit(format!("STEP 2b page={page} succ={}", successor.key));
319 self.remove_at(right, successor.key, cases, trace);
320 self.node_mut(page).entries[position] = successor;
321 } else {
322 cases.push("2c");
323 let child = self.merge(page, position, "2c", trace);
324 self.remove_at(child, key, cases, trace);
325 }
326 return;
327 }
328 let mut child = self.node(page).children[position];
329 if self.node(child).entries.len() == self.degree - 1 {
330 let left = position.checked_sub(1).map(|j| self.node(page).children[j]);
331 let right = self.node(page).children.get(position + 1).copied();
332 if let Some(sibling) = left.filter(|&id| self.node(id).entries.len() >= self.degree) {
333 cases.push("3a-left");
334 let middle = self.node(page).entries[position - 1].clone();
335 let replacement = self.node_mut(sibling).entries.pop().unwrap();
336 let moved_child = self.node_mut(sibling).children.pop();
337 self.node_mut(page).entries[position - 1] = replacement;
338 self.node_mut(child).entries.insert(0, middle);
339 if let Some(id) = moved_child {
340 self.node_mut(child).children.insert(0, id);
341 }
342 self.borrows += 1;
343 trace.emit(format!("STEP 3a-left page={page} child={child} from={sibling} keys={}", self.node(child).notation()));
344 } else if let Some(sibling) = right.filter(|&id| self.node(id).entries.len() >= self.degree) {
345 cases.push("3a-right");
346 let middle = self.node(page).entries[position].clone();
347 let replacement = self.node_mut(sibling).entries.remove(0);
348 let moved_child = if self.node(sibling).children.is_empty() { None } else { Some(self.node_mut(sibling).children.remove(0)) };
349 self.node_mut(page).entries[position] = replacement;
350 self.node_mut(child).entries.push(middle);
351 if let Some(id) = moved_child {
352 self.node_mut(child).children.push(id);
353 }
354 self.borrows += 1;
355 trace.emit(format!("STEP 3a-right page={page} child={child} from={sibling} keys={}", self.node(child).notation()));
356 } else if left.is_some() {
357 cases.push("3b-left");
358 child = self.merge(page, position - 1, "3b-left", trace);
359 } else {
360 cases.push("3b-right");
361 child = self.merge(page, position, "3b-right", trace);
362 }
363 }
364 self.remove_at(child, key, cases, trace);
365 }

367 fn scan(&self, low: i32, high: i32) -> (Vec<Entry>, usize) {
368 fn visit(tree: &Tree, page: usize, low: i32, high: i32, entries: &mut Vec<Entry>, visits: &mut usize) {
369 *visits += 1;
370 let node = tree.node(page);
371 for j in 0..=node.entries.len() {
372 if !node.children.is_empty() && (j == 0 || node.entries[j - 1].key < high)
373 && (j == node.entries.len() || node.entries[j].key > low) {
374 visit(tree, node.children[j], low, high, entries, visits);
375 }
376 if let Some(entry) = node.entries.get(j).filter(|e| low <= e.key && e.key <= high) {
377 entries.push(entry.clone());
378 }
379 }
380 }
381 let mut entries = Vec::new();
382 let mut visits = 0;
383 visit(self, self.root, low, high, &mut entries, &mut visits);
384 (entries, visits)
385 }

387 fn dump(&self) -> String {
388 fn visit(tree: &Tree, page: usize, tokens: &mut Vec<String>) {
389 let node = tree.node(page);
390 tokens.push("(".into());
391 for j in 0..=node.entries.len() {
392 if !node.children.is_empty() {
393 visit(tree, node.children[j], tokens);
394 }
395 if let Some(entry) = node.entries.get(j) {
396 tokens.push(entry.to_string());
397 }
398 }
399 tokens.push(")".into());
400 }
401 let mut tokens = Vec::new();
402 visit(self, self.root, &mut tokens);
403 tokens.join(" ")
404 }

406 fn compact(&mut self) {
407 let mut source = self.entries();
408 let mut children = Vec::new();
409 let mut built = Vec::new();
410 let old_root;
411 loop {
412 let groups = (source.len() + 2 * self.degree) / (2 * self.degree);
413 let keys = source.len() - (groups - 1);
414 let (base, extra) = (keys / groups, keys % groups);
415 let mut position = 0;
416 let mut child_position = 0;
417 let mut separators = Vec::new();
418 let mut level = Vec::new();
419 for j in 0..groups {
420 let count = base + usize::from(j < extra);
421 let mut node = Node { entries: source[position..position + count].to_vec(), children: Vec::new() };
422 position += count;
423 if !children.is_empty() {
424 node.children = children[child_position..child_position + count + 1].to_vec();
425 child_position += count + 1;
426 }
427 level.push(built.len());
428 built.push(node);
429 if j + 1 < groups {
430 separators.push(source[position].clone());
431 position += 1;
432 }
433 }
434 if groups == 1 {
435 old_root = level[0];
436 break;
437 }
438 source = separators;
439 children = level;
440 }
441 fn renumber(old: usize, built: &[Node], pages: &mut BTreeMap<usize, Node>) -> usize {
442 let page = pages.len();
443 pages.insert(page, Node::default());
444 let children = built[old].children.iter().map(|&child| renumber(child, built, pages)).collect();
445 pages.insert(page, Node { entries: built[old].entries.clone(), children });
446 page
447 }
448 self.pages.clear();
449 self.root = renumber(old_root, &built, &mut self.pages);
450 self.splits = 0;
451 self.merges = 0;
452 self.borrows = 0;
453 }
454}

456#[derive(Clone, Copy, Default)]
457enum Cursor {
458 #[default]
459 Unset,
460 Key(i32),
461 Begin,
462 End,
463}
464
465#[derive(Default)]
466struct Workbench {
467 index: Option<Tree>,
468 saved: Option<Tree>,
469 cursor: Cursor,
470 trace: bool,
471}
472
473impl Workbench {
474 fn process(&mut self, line: &str) -> Result<Vec<String>, String> {
475 let words: Vec<_> = line.split(' ').collect();
476 let command = words[0];
477 if command == "TRACE" {
478 self.trace = words[1] == "ON";
479 return Ok(vec![format!("TRACE {}", words[1])]);
480 }
481 let first: i32 = words.get(1).map_or(0, |word| word.parse().unwrap());
482 let ranged = matches!(command, "DELRANGE" | "RANGE" | "COUNT");
483 let second: i32 = if ranged { words[2].parse().unwrap() } else { 0 };
484 if matches!(command, "CREATE" | "LOAD") {
485 if !(2..=5).contains(&first) {
486 return Err(format!("BAD DEGREE {first}"));
487 }
488 let tree = if command == "CREATE" { Tree::new(first as usize) }
489 else { Tree::load(first as usize, &words[2..]).map_err(|q| format!("SYNTAX {q}"))? };
490 let result = if command == "CREATE" { format!("CREATE t={first}") }
491 else { format!("LOAD t={first} nodes={} keys={} {}", tree.pages.len(), tree.entries().len(), tree.status()) };
492 self.index = Some(tree);
493 self.cursor = Cursor::Unset;
494 return Ok(vec![result]);
495 }
496 let tree = self.index.as_mut().ok_or("NO INDEX")?;
497 let needs_valid = matches!(command, "INSERT" | "BULKLOAD" | "DELETE" | "DELRANGE" | "RANGE" | "COUNT"
498 | "MIN" | "MAX" | "FLOOR" | "CEIL" | "RANK" | "SELECT" | "SEEK" | "NEXT" | "PREV" | "HEIGHT" | "COMPACT");
499 if needs_valid && tree.violation().is_some() {
500 return Err("DAMAGED".into());
501 }
502 let entries = tree.entries();
503 if ranged && first > second {
504 return Err(format!("BAD RANGE {first} {second}"));
505 }
506 if command == "SELECT" && (first < 1 || first as usize > entries.len()) {
507 return Err(format!("BAD POSITION {first}"));
508 }
509 if matches!(command, "NEXT" | "PREV") && matches!(self.cursor, Cursor::Unset) {
510 return Err("NO CURSOR".into());
511 }
512 if command == "RESTORE" && self.saved.is_none() {
513 return Err("NO SAVE".into());
514 }
515 let mut output = Vec::new();
516 let mut trace = Trace { enabled: self.trace && matches!(command, "INSERT" | "DELETE"), lines: Vec::new() };
517 match command {
518 "INSERT" => {
519 let before = tree.pages.clone();
520 let splits = tree.splits;
521 let old = tree.upsert(Entry { key: first, value: words[2].into() }, &mut trace);
522 let writes = tree.changes(&before).0;
523 output.extend(trace.lines);
524 output.push(if let Some(value) = old { format!("INSERT {first} UPDATED old={value} writes={writes}") }
525 else { format!("INSERT {first} NEW splits={} height={} writes={writes}", tree.splits - splits, tree.height()) });
526 }
527 "BULKLOAD" => {
528 let before = tree.pages.clone();
529 let splits = tree.splits;
530 let mut new = 0;
531 let mut updated = 0;
532 for pair in words[1..].chunks_exact(2) {
533 if tree.upsert(Entry { key: pair[0].parse().unwrap(), value: pair[1].into() }, &mut trace).is_some() {
534 updated += 1;
535 } else {
536 new += 1;
537 }
538 }
539 output.push(format!("BULKLOAD pairs={} new={new} updated={updated} splits={} height={} writes={}",
540 (words.len() - 1) / 2, tree.splits - splits, tree.height(), tree.changes(&before).0));
541 }
542 "DELETE" => {
543 if let Some((page, position)) = tree.search(first).0 {
544 let value = tree.node(page).entries[position].value.clone();
545 let before = tree.pages.clone();
546 let mut cases = Vec::new();
547 tree.remove_at(tree.root, first, &mut cases, &mut trace);
548 let (writes, frees) = tree.changes(&before);
549 output.extend(trace.lines);
550 output.push(format!("DELETE {first} REMOVED value={value} cases={} height={} writes={writes} frees={frees}",
551 cases.join(","), tree.height()));
552 } else {
553 output.push(format!("DELETE {first} NOT FOUND"));
554 }
555 }
556 "DELRANGE" => {
557 let before = tree.pages.clone();
558 let (merges, borrows) = (tree.merges, tree.borrows);
559 let mut removed = 0;
560 for entry in entries.iter().filter(|entry| first <= entry.key && entry.key <= second) {
561 tree.remove_at(tree.root, entry.key, &mut Vec::new(), &mut trace);
562 removed += 1;
563 }
564 let (writes, frees) = tree.changes(&before);
565 output.push(format!("DELRANGE {first} {second} removed={removed} merges={} borrows={} height={} writes={writes} frees={frees}",
566 tree.merges - merges, tree.borrows - borrows, tree.height()));
567 }
568 "GET" => {
569 let (found, path) = tree.search(first);
570 let result = found.map_or_else(|| "NOT FOUND".into(), |(page, position)| format!("FOUND value={}", tree.node(page).entries[position].value));
571 output.push(format!("GET {first} {result} path={}", path.join(" -> ")));
572 }
573 "RANGE" | "COUNT" => {
574 let (found, visits) = tree.scan(first, second);
575 let mut result = format!("{command} {first} {second} count={} visited={visits}", found.len());
576 if command == "RANGE" {
577 for entry in found {
578 result.push(' ');
579 result.push_str(&entry.to_string());
580 }
581 }
582 output.push(result);
583 }
584 "MIN" | "MAX" => {
585 let entry = if command == "MIN" { entries.first() } else { entries.last() };
586 output.push(format!("{command} {}", entry.map_or_else(|| "EMPTY".into(), ToString::to_string)));
587 }
588 "FLOOR" | "CEIL" => {
589 let entry = if command == "FLOOR" { entries.iter().rev().find(|e| e.key <= first) }
590 else { entries.iter().find(|e| e.key >= first) };
591 output.push(format!("{command} {first} {}", entry.map_or_else(|| "NONE".into(), ToString::to_string)));
592 }
593 "RANK" => output.push(format!("RANK {first} {}", entries.iter().filter(|e| e.key < first).count())),
594 "SELECT" => output.push(format!("SELECT {first} {}", entries[first as usize - 1])),
595 "SEEK" => {
596 let entry = entries.iter().find(|e| e.key >= first);
597 self.cursor = entry.map_or(Cursor::End, |e| Cursor::Key(e.key));
598 output.push(format!("SEEK {first} {}", entry.map_or_else(|| "END".into(), ToString::to_string)));
599 }
600 "NEXT" | "PREV" => {
601 let entry = match (command, self.cursor) {
602 ("NEXT", Cursor::Key(key)) => entries.iter().find(|e| e.key > key),
603 ("NEXT", Cursor::Begin) => entries.first(),
604 ("PREV", Cursor::Key(key)) => entries.iter().rev().find(|e| e.key < key),
605 ("PREV", Cursor::End) => entries.last(),
606 _ => None,
607 };
608 let boundary = if command == "NEXT" { Cursor::End } else { Cursor::Begin };
609 self.cursor = entry.map_or(boundary, |e| Cursor::Key(e.key));
610 output.push(format!("{command} {}", entry.map_or_else(|| if command == "NEXT" { "END".into() } else { "BEGIN".into() }, ToString::to_string)));
611 }
612 "PRINT" => {
613 let mut level = vec![tree.root];
614 let mut depth = 0;
615 while !level.is_empty() {
616 output.push(format!("LEVEL {depth} {}", level.iter().map(|&page| tree.node(page).notation()).collect::<Vec<_>>().join(" ")));
617 level = level.iter().flat_map(|&page| tree.node(page).children.iter().copied()).collect();
618 depth += 1;
619 }
620 }
621 "PAGES" => {
622 for (page, node) in &tree.pages {
623 let children = if node.children.is_empty() { "-".into() }
624 else { node.children.iter().map(ToString::to_string).collect::<Vec<_>>().join(",") };
625 output.push(format!("PAGE {page} {} kids={children}", node.notation()));
626 }
627 }
628 "DUMP" => output.push(format!("DUMP {} {}", tree.degree, tree.dump())),
629 "HEIGHT" => output.push(format!("HEIGHT {}", tree.height())),
630 "STATS" => {
631 let nodes = tree.pages.len();
632 let leaves = tree.pages.values().filter(|node| node.children.is_empty()).count();
633 output.push(format!("STATS t={} nodes={nodes} leaves={leaves} keys={} fill={}% splits={} merges={} borrows={}",
634 tree.degree, entries.len(), 100 * entries.len() / ((2 * tree.degree - 1) * nodes), tree.splits, tree.merges, tree.borrows));
635 }
636 "VERIFY" => output.push(tree.violation().map_or_else(|| "VERIFY OK".into(), |error| format!("VERIFY FAIL {error}"))),
637 "REBUILD" => {
638 *tree = Tree::new(tree.degree);
639 for entry in &entries {
640 tree.upsert(entry.clone(), &mut trace);
641 }
642 output.push(format!("REBUILD entries={} keys={} nodes={} splits={} height={}",
643 entries.len(), tree.entries().len(), tree.pages.len(), tree.splits, tree.height()));
644 }
645 "COMPACT" => {
646 let nodes = tree.pages.len();
647 tree.compact();
648 output.push(format!("COMPACT nodes={nodes}->{} height={}", tree.pages.len(), tree.height()));
649 }
650 "SAVE" => {
651 self.saved = Some(tree.clone());
652 output.push(format!("SAVE nodes={} keys={} {}", tree.pages.len(), entries.len(), tree.status()));
653 }
654 "RESTORE" => {
655 *tree = self.saved.as_ref().unwrap().clone();
656 output.push(format!("RESTORE nodes={} keys={} {}", tree.pages.len(), tree.entries().len(), tree.status()));
657 }
658 _ => unreachable!(),
659 }
660 Ok(output)
661 }
662}

664fn main() {
665 let stdin = io::stdin();
666 let mut lines = stdin.lock().lines();
667 let count: usize = lines.next().unwrap().unwrap().parse().unwrap();
668 let mut workbench = Workbench::default();
669 let mut output = io::BufWriter::new(io::stdout().lock());
670 for line in lines.take(count) {
671 match workbench.process(&line.unwrap()) {
672 Ok(lines) => {
673 for line in lines {
674 writeln!(output, "{line}").unwrap();
675 }
676 }
677 Err(error) => writeln!(output, "ERROR {error}").unwrap(),
678 }
679 }
