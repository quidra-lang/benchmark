31 if (kind.equals("in")) return chart.states.containsKey(text) ? null : "unknown state " + text;

36 long evaluate(Chart chart) {
37 if (kind.equals("number")) return Long.parseLong(text);
38 if (kind.equals("variable")) return text.equals("_data") ? chart.data : chart.variables.get(text);
39 if (kind.equals("in")) return chart.states.get(text).active ? 1 : 0;
40 long a = left.evaluate(chart);
41 if (kind.equals("unary")) return text.equals("-") ? -a : a == 0 ? 1 : 0;
42 if (text.equals("&&") && a == 0) return 0;
43 if (text.equals("||") && a != 0) return 1;
44 long b = right.evaluate(chart);
45 if ((text.equals("/") || text.equals("%")) && b == 0) throw new EvaluationError();
46 return switch (text) {
47 case "+" -> a + b;
48 case "-" -> a - b;
49 case "*" -> a * b;
50 case "/" -> a / b;
51 case "%" -> a % b;
52 case "==" -> a == b ? 1 : 0;
53 case "!=" -> a != b ? 1 : 0;
54 case "<" -> a < b ? 1 : 0;
55 case "<=" -> a <= b ? 1 : 0;
56 case ">" -> a > b ? 1 : 0;
57 case ">=" -> a >= b ? 1 : 0;
58 case "&&", "||" -> b != 0 ? 1 : 0;
59 default -> throw new IllegalStateException();
60 };
61 }

64 static class ExpressionParser {
65 static final List<List<String>> OPERATORS = List.of(List.of("||"), List.of("&&"), List.of("==", "!=", "<", "<=", ">", ">="), List.of("+", "-"), List.of("*", "/", "%"));
66 List<String> tokens = new ArrayList<>();
67 int position;
68
69 ExpressionParser(String text) {
70 for (int i = 0; i < text.length();) {
71 char c = text.charAt(i);
72 if (c == ' ') {
73 i++;
74 continue;
75 }
76 int start = i;
77 if (Character.isLetterOrDigit(c) || c == '_') {
78 while (i < text.length() && (Character.isLetterOrDigit(text.charAt(i)) || text.charAt(i) == '_')) i++;
79 } else if (i + 1 < text.length() && List.of("||", "&&", "==", "!=", "<=", ">=").contains(text.substring(i, i + 2))) i += 2;
80 else i++;
81 tokens.add(text.substring(start, i));
82 }
83 }
84
85 Node parse() {
86 return level(0);
87 }
88
89 Node level(int precedence) {
90 if (precedence == OPERATORS.size()) return unary();
91 Node result = level(precedence + 1);
92 while (position < tokens.size() && OPERATORS.get(precedence).contains(tokens.get(position))) {
93 String operator = tokens.get(position++);
94 result = new Node("binary", operator, result, level(precedence + 1));
95 }
96 return result;
97 }
98
99 Node unary() {
100 String token = tokens.get(position++);
101 if (token.equals("!") || token.equals("-")) return new Node("unary", token, unary(), null);
102 if (token.equals("(")) {
103 Node result = level(0);
104 position++;
105 return result;
106 }
107 if (token.equals("in")) {
108 position++;
109 String state = tokens.get(position++);
110 position++;
111 return new Node("in", state, null, null);
112 }
113 return new Node(token.charAt(0) >= '0' && token.charAt(0) <= '9' ? "number" : "variable", token, null, null);
114 }
115 }

132 static List<Action> actions(String text) {
133 List<Action> result = new ArrayList<>();
134 for (String raw : text.split("; ", -1)) {
135 Action action = new Action();
136 String[] parts = raw.split(" ", 2);
137 String first = parts[0], rest = parts.length > 1 ? parts[1] : "";
138 switch (first) {
139 case "raise", "send" -> {
140 action.kind = first;
141 String[] args = rest.split(" ", first.equals("raise") ? 2 : 3);
142 action.name = args[0];
143 if (first.equals("send")) {
144 action.delay = Integer.parseInt(args[1]);
145 if (args.length == 3) action.expression = expression(args[2]);
146 } else if (args.length == 2) action.expression = expression(args[1]);
147 }
148 case "cancel" -> {
149 action.kind = "cancel";
150 action.name = rest;
151 }
152 case "log" -> {
153 action.kind = "log";
154 action.literal = "";
155 if (rest.startsWith("\"")) {
156 int end = rest.indexOf('"', 1);
157 action.literal = rest.substring(1, end);
158 if (end + 1 < rest.length()) action.expression = expression(rest.substring(end + 2));
159 } else action.expression = expression(rest);
160 }
161 default -> {
162 action.kind = "assign";
163 action.name = first;
164 action.expression = expression(rest.substring(2));
165 }
166 }
167 result.add(action);
168 }
169 return result;
170 }

276 Transition parseTransition(String text) {
277 int end = text.indexOf(' ', 3);
278 Transition transition = new Transition();
279 transition.source = states.get(text.substring(3, end));
280 String rest = text.substring(end + 1);
281 if (rest.startsWith("after ")) {
282 int space = rest.indexOf(' ', 6);
283 transition.trigger = "timer";
284 transition.delay = Integer.parseInt(space < 0 ? rest.substring(6) : rest.substring(6, space));
285 rest = space < 0 ? "" : rest.substring(space + 1);
286 } else {
287 int space = rest.indexOf(' ');
288 String trigger = space < 0 ? rest : rest.substring(0, space);
289 transition.trigger = trigger.equals("always") ? "always" : "event";
290 transition.descriptors = trigger.split("\\|");
291 rest = space < 0 ? "" : rest.substring(space + 1);
292 }
293 if (rest.startsWith("[")) {
294 int close = rest.indexOf(']');
295 transition.guard = expression(rest.substring(1, close));
296 rest = rest.substring(close + 1).stripLeading();
297 }
298 if (rest.startsWith("-> ") || rest.startsWith("~> ")) {
299 transition.mode = rest.startsWith("~>") ? "local" : "external";
300 rest = rest.substring(3);
301 int space = rest.indexOf(' ');
302 transition.targetName = space < 0 ? rest : rest.substring(0, space);
303 rest = space < 0 ? "" : rest.substring(space + 1);
304 }
305 if (rest.startsWith("/ ")) transition.actions = actions(rest.substring(2));
306 return transition;
307 }

309 boolean define(List<Line> lines) {
310 List<Definition> definitions = new ArrayList<>();
311 for (Line line : lines) {
312 Definition definition = new Definition(line);
313 definitions.add(definition);
314 String[] a = line.text.split(" ");
315 String kind = a[0];
316 if (List.of("state", "parallel", "final", "history").contains(kind)) {
317 String parentName = a.length > 2 && !a[2].equals("initial") ? a[2] : null;
318 if (states.containsKey(a[1])) definition.error = "duplicate state " + a[1];
319 else if (parentName != null && !states.containsKey(parentName)) definition.error = "unknown state " + parentName;
320 else if (parentName != null && List.of("final", "history").contains(states.get(parentName).kind)) definition.error = parentName + " cannot have children";
321 else {
322 State parent = parentName == null ? root : states.get(parentName);
323 State state = new State(a[1], kind, parent);
324 definition.state = state;
325 states.put(a[1], state);
326 parent.children.add(state);
327 if (kind.equals("state")) for (int i = 2; i < a.length; i++) if (a[i].equals("initial")) state.initialName = a[i + 1];
328 if (kind.equals("history")) {
329 state.deep = a[3].equals("deep");
330 if (a.length == 5) state.defaultName = a[4];
331 }
332 }
333 } else if (kind.equals("var")) {
334 if (variables.containsKey(a[1])) definition.error = "duplicate variable " + a[1];
335 else variables.put(a[1], Long.parseLong(a[2]));
336 }
337 }
338 for (Definition definition : definitions) {
339 if (definition.error != null) continue;
340 String text = definition.line.text;
341 String[] a = text.split(" ");
342 String kind = a[0];
343 State state = definition.state;
344 if (kind.equals("state") && state.initialName != null) {
345 State child = states.get(state.initialName);
346 if (child == null) definition.error = "unknown state " + state.initialName;
347 else if (child.parent != state || child.kind.equals("history")) definition.error = child.name + " is not a child of " + state.name;
348 } else if (kind.equals("final") || kind.equals("history")) {
349 if (!state.parent.kind.equals("state") || state.parent.childStates().isEmpty()) definition.error = state.name + " needs a compound parent";
350 else if (kind.equals("history") && state.defaultName != null) {
351 State target = states.get(state.defaultName);
352 if (target == null) definition.error = "unknown state " + state.defaultName;
353 else if (!target.descendantOf(state.parent) || target.kind.equals("history")) definition.error = target.name + " is not a descendant of " + state.parent.name;
354 }
355 } else if (kind.equals("entry") || kind.equals("exit")) {
356 State source = states.get(a[1]);
357 if (source == null) definition.error = "unknown state " + a[1];
358 else if (source.kind.equals("history")) definition.error = source.name + " cannot have actions";
359 else {
360 List<Action> list = actions(text.substring(text.indexOf(" / ") + 3));
361 definition.error = firstReference(list, this);
362 if (definition.error == null) (kind.equals("entry") ? source.entries : source.exits).add(list);
363 }
364 } else if (kind.equals("on")) {
365 State source = states.get(a[1]);
366 if (source == null) definition.error = "unknown state " + a[1];
367 else if (List.of("final", "history").contains(source.kind)) definition.error = source.name + " cannot have transitions";
368 else {
369 Transition transition = parseTransition(text);
370 String error = transition.guard == null ? null : transition.guard.reference(this);
371 if (error == null && transition.targetName != null && !states.containsKey(transition.targetName)) error = "unknown state " + transition.targetName;
372 if (error == null) error = firstReference(transition.actions, this);
373 definition.error = error;
374 if (error == null) {
375 transition.target = transition.targetName == null ? null : states.get(transition.targetName);
376 source.transitions.add(transition);
377 }
378 }
379 }
380 }
381 boolean valid = true;
382 for (Definition definition : definitions) if (definition.error != null) {
383 System.out.println("error line " + definition.line.number + ": " + definition.error);
384 valid = false;
385 }
386 if (!valid) return false;
387 for (State child : root.children) visit(child);
388 return true;
389 }

414 Choice choice(Transition transition) {
415 if (transition.target == null) return new Choice(transition, null, Set.of());
416 List<State> targets = effective(transition);
417 State scope = null;
418 if (transition.mode.equals("local") && transition.source.compound() && targets.stream().allMatch(s -> s.descendantOf(transition.source))) scope = transition.source;
419 else for (State ancestor = transition.source.parent; ancestor != null; ancestor = ancestor.parent) {
420 State candidate = ancestor;
421 if (candidate.compound() && targets.stream().allMatch(s -> s.descendantOf(candidate))) {
422 scope = candidate;
423 break;
424 }
425 }
426 Set<State> exits = new HashSet<>();
427 for (State state : order) if (state.active && state.descendantOf(scope)) exits.add(state);
428 return new Choice(transition, scope, exits);
429 }

431 boolean matches(Transition transition, Trigger trigger) {
432 if (trigger.kind.equals("always")) return transition.trigger.equals("always");
433 if (trigger.kind.equals("timer")) return transition == trigger.timer;
434 if (!transition.trigger.equals("event")) return false;
435 for (String descriptor : transition.descriptors) if (descriptor.equals("*") || descriptor.equals(trigger.event.name) || trigger.event.name.startsWith(descriptor + ".")) return true;
436 return false;
437 }

439 List<Choice> select(Trigger trigger) {
440 List<Transition> candidates = new ArrayList<>();
441 Set<Transition> seen = new HashSet<>();
442 for (State atomic : order) if (atomic.active && atomic.atomic()) {
443 Transition selected = null;
444 search: for (State state = atomic; state != root; state = state.parent) for (Transition transition : state.transitions) if (matches(transition, trigger)) {
445 boolean guard = true;
446 try {
447 guard = transition.guard == null || transition.guard.evaluate(this) != 0;
448 } catch (EvaluationError error) {
449 queue.add(new Event("error.execution", 0));
450 guard = false;
451 }
452 if (guard) {
453 selected = transition;
454 break search;
455 }
456 }
457 if (selected != null && seen.add(selected)) candidates.add(selected);
458 }
459 List<Choice> result = new ArrayList<>();
460 Set<State> exited = new HashSet<>();
461 for (Transition transition : candidates) {
462 Choice candidate = choice(transition);
463 if (candidate.exits.stream().noneMatch(exited::contains)) {
464 result.add(candidate);
465 exited.addAll(candidate.exits);
466 }
467 }
468 return result;
469 }

544 boolean micro(List<Choice> selected, Trigger trigger) {
545 if (inMacro == 1000) {
546 System.out.println("LIVELOCK t=" + time);
547 halted = true;
548 return false;
549 }
550 inMacro++;
551 microsteps++;
552 Set<State> exits = new HashSet<>();
553 for (Choice choice : selected) exits.addAll(choice.exits);
554 for (State history : order) if (history.kind.equals("history") && exits.contains(history.parent)) history.history = order.stream().filter(s -> s.active && (history.deep ? s.atomic() && s.descendantOf(history.parent) : s.parent == history.parent && !s.kind.equals("history"))).toList();
555 Set<State> entry = new HashSet<>();
556 for (Choice choice : selected) if (choice.transition.target != null) for (State target : effective(choice.transition)) for (State state = target; state != choice.scope; state = state.parent) entry.add(state);
557 expand(entry);
558 List<State> exitOrder = order.stream().filter(exits::contains).sorted(Comparator.comparingInt((State s) -> s.order).reversed()).toList();
559 if (trace) System.out.println("[t=" + time + "] " + display(trigger) + " (" + list(selected.stream().map(c -> c.transition.source).toList()) + "): exit " + list(exitOrder) + "; enter " + list(order.stream().filter(entry::contains).toList()));
560 for (State state : exitOrder) {
561 for (List<Action> list : state.exits) perform(list);
562 state.active = false;
563 state.dwell += time - state.entryTime;
564 reservations.removeIf(r -> r.timer != null && r.timer.source == state);
565 }
566 for (Choice choice : selected) {
567 perform(choice.transition.actions);
568 choice.transition.source.taken++;
569 }
570 enter(entry);
571 return true;
572 }

574 void macro(Trigger trigger) {
575 macrosteps++;
576 inMacro = 0;
577 if (trigger.kind.equals("start")) {
578 data = 0;
579 Set<State> entry = new HashSet<>();
580 entry.add(initial(root));
581 expand(entry);
582 enter(entry);
583 } else {
584 data = trigger.kind.equals("timer") ? 0 : trigger.event.value;
585 List<Choice> selected = select(trigger);
586 if (!selected.isEmpty() && !micro(selected, trigger)) return;
587 }
588 while (!halted) {
589 data = 0;
590 Trigger always = new Trigger("always", null, null);
591 List<Choice> selected = select(always);
592 if (!selected.isEmpty()) {
593 if (!micro(selected, always)) return;
594 continue;
595 }
596 if (queue.isEmpty()) return;
597 Event event = queue.remove();
598 data = event.value;
599 Trigger internal = new Trigger("event", event, null);
600 selected = select(internal);
601 if (!selected.isEmpty() && !micro(selected, internal)) return;
602 }

604 static final Comparator<Reservation> RESERVATION_ORDER = Comparator.comparingLong((Reservation r) -> r.due).thenComparingLong(r -> r.number);

606 void script(String text) {
607 if (halted) return;
608 String[] a = text.split(" ", 3);
609 switch (a[0]) {
610 case "event" -> macro(new Trigger("event", new Event(a[1], a.length == 3 ? Long.parseLong(a[2]) : 0), null));
611 case "tick" -> {
612 long finish = time + Long.parseLong(a[1]);
613 while (!halted) {
614 Reservation next = reservations.stream().filter(r -> r.due <= finish).min(RESERVATION_ORDER).orElse(null);
615 if (next == null) break;
616 reservations.remove(next);
617 time = next.due;
618 macro(next.timer == null ? new Trigger("event", next.event, null) : new Trigger("timer", null, next.timer));
619 }
620 if (!halted) time = finish;
621 }
622 case "set" -> {
623 if (!variables.containsKey(a[1])) System.out.println("error: unknown variable " + a[1]);
624 else variables.put(a[1], Long.parseLong(a[2]));
625 }
626 case "trace" -> trace = a[1].equals("on");
627 case "eval" -> {
628 String raw = text.substring(5);
629 Node node = expression(raw);
630 String error = node.reference(this);
631 if (error != null) System.out.println("error: " + error);
632 else {
633 data = 0;
634 try {
635 System.out.println(raw + " -> " + node.evaluate(this));
636 } catch (EvaluationError failure) {
637 System.out.println("error: division by zero");
638 }
639 }
640 }
641 case "dump" -> {
642 System.out.println("dump t=" + time);
643 for (State state : order) if (state.active) System.out.println(" ".repeat(state.depth) + state.name);
644 for (Map.Entry<String, Long> variable : variables.entrySet()) System.out.println(" var " + variable.getKey() + " = " + variable.getValue());
645 for (State state : order) if (state.kind.equals("history") && state.history != null) System.out.println(" history " + state.name + " = " + list(state.history));
646 for (Reservation reservation : reservations.stream().sorted(RESERVATION_ORDER).toList()) {
647 if (reservation.timer != null) System.out.println(" timer " + reservation.due + " after(" + reservation.timer.delay + ")@" + reservation.timer.source.name);
648 else System.out.println(" send " + reservation.due + " " + reservation.event.name + " " + reservation.event.value);
649 }
650 }
651 default -> throw new IllegalStateException();
652 }

671 number++;
672 System.out.println("chart " + header.substring(6));
673 Chart chart = new Chart();
674 if (!chart.define(definitions)) continue;
675 chart.warnings();
676 chart.macro(new Trigger("start", null, null));
