15static void *allocate(size_t size) {
16 void *pointer = calloc(1, size ? size : 1);
17 Allocation *allocation = malloc(sizeof(*allocation));
18 if (!pointer || !allocation) exit(EXIT_FAILURE);
19 *allocation = (Allocation){pointer, allocations};
20 allocations = allocation;
21 return pointer;

34static char *format(const char *pattern, ...) {
35 va_list args, saved;
36 va_start(args, pattern);
37 va_copy(saved, args);
38 int length = vsnprintf(NULL, 0, pattern, saved);
39 va_end(saved);
40 char *result = allocate((size_t)length + 1);
41 vsnprintf(result, (size_t)length + 1, pattern, args);
42 va_end(args);
43 return result;
44}

55static char *read_line(void) {
56 size_t length = 0, capacity = 128;
57 char *text = malloc(capacity);
58 if (!text) exit(EXIT_FAILURE);
59 int c;
60 while ((c = getchar()) != EOF && c != '\n') {
61 if (length + 1 == capacity) {
62 capacity *= 2;
63 char *grown = realloc(text, capacity);
64 if (!grown) exit(EXIT_FAILURE);
65 text = grown;
66 }
67 text[length++] = (char)c;
68 }
69 if (length && text[length - 1] == '\r') length--;
70 text[length] = '\0';
71 return text;
72}

82static char *word(const char **text) {
83 while (**text == ' ') (*text)++;
84 const char *start = *text;
85 while (**text && **text != ' ') (*text)++;
86 char *result = substring(start, (size_t)(*text - start));
87 while (**text == ' ') (*text)++;
88 return result;

290static bool two_character_operator(const char *text) {
291 return starts(text, "||") || starts(text, "&&") || starts(text, "==") || starts(text, "!=") || starts(text, "<=") || starts(text, ">=");
292}
293static Node *parse_level(ExpressionParser *parser, int precedence);
294
295static Node *parse_unary(ExpressionParser *parser) {
296 char *token = parser->tokens.items[parser->position++];
297 if (equal(token, "!") || equal(token, "-")) return node(UNARY, token, parse_unary(parser), NULL);
298 if (equal(token, "(")) {
299 Node *result = parse_level(parser, 0);
300 parser->position++;
301 return result;
302 }
303 if (equal(token, "in")) {
304 parser->position++;
305 char *state = parser->tokens.items[parser->position++];
306 parser->position++;
307 return node(IN_STATE, state, NULL, NULL);
308 }
309 return node(token[0] >= '0' && token[0] <= '9' ? NUMBER : VARIABLE, token, NULL, NULL);
310}

312static bool operator_at(const char *text, int precedence) {
313 switch (precedence) {
314 case 0: return equal(text, "||");
315 case 1: return equal(text, "&&");
316 case 2: return equal(text, "==") || equal(text, "!=") || equal(text, "<") || equal(text, "<=") || equal(text, ">") || equal(text, ">=");
317 case 3: return equal(text, "+") || equal(text, "-");
318 default:
319 return equal(text, "*") || equal(text, "/") || equal(text, "%");
320 }
321}
322
323static Node *parse_level(ExpressionParser *parser, int precedence) {
324 if (precedence == 5) return parse_unary(parser);
325 Node *result = parse_level(parser, precedence + 1);
326 while (parser->position < parser->tokens.count && operator_at(parser->tokens.items[parser->position], precedence)) {
327 char *operation = parser->tokens.items[parser->position++];
328 Node *right = parse_level(parser, precedence + 1);
329 result = node(BINARY, operation, result, right);
330 }
331 return result;
332}

334static Node *expression(const char *text) {
335 ExpressionParser parser = {
336 {allocate((strlen(text) + 1) * sizeof(char *)), 0}, 0
337 };
338 for (size_t i = 0; text[i];) {
339 if (text[i] == ' ') {
340 i++;
341 continue;
342 }
343 size_t start = i;
344 if (name_char(text[i])) while (name_char(text[i])) i++;
345 else if (two_character_operator(text + i)) i += 2;
346 else i++;
347 parser.tokens.items[parser.tokens.count++] = substring(text + start, i - start);
348 }
349 return parse_level(&parser, 0);
350}

352static char *reference(const Chart *chart, const Node *n) {
353 if (!n) return NULL;
354 if (n->kind == VARIABLE) return equal(n->text, "_data") || find_variable(chart, n->text) >= 0 ? NULL : format("unknown variable %s", n->text);
355 if (n->kind == IN_STATE) return find_state(chart, n->text) >= 0 ? NULL : format("unknown state %s", n->text);
356 char *error = reference(chart, n->left);
357 return error ? error : reference(chart, n->right);
358}

360static long long evaluate(Chart *chart, const Node *n, bool *success) {
361 if (n->kind == NUMBER) return strtoll(n->text, NULL, 10);
362 if (n->kind == VARIABLE) return equal(n->text, "_data") ? chart->data : chart->variables[find_variable(chart, n->text)].value;
363 if (n->kind == IN_STATE) return chart->states[find_state(chart, n->text)].active ? 1 : 0;
364 long long a = evaluate(chart, n->left, success);
365 if (!*success) return 0;
366 if (n->kind == UNARY) return equal(n->text, "-") ? -a : a == 0;
367 if (equal(n->text, "&&") && a == 0) return 0;
368 if (equal(n->text, "||") && a != 0) return 1;
369 long long b = evaluate(chart, n->right, success);
370 if (!*success) return 0;
371 if ((equal(n->text, "/") || equal(n->text, "%")) && b == 0) {
372 *success = false;
373 return 0;
374 }
375 if (equal(n->text, "+")) return a + b;
376 if (equal(n->text, "-")) return a - b;
377 if (equal(n->text, "*")) return a * b;
378 if (equal(n->text, "/")) return a / b;
379 if (equal(n->text, "%")) return a % b;
380 if (equal(n->text, "==")) return a == b;
381 if (equal(n->text, "!=")) return a != b;
382 if (equal(n->text, "<")) return a < b;
383 if (equal(n->text, "<=")) return a <= b;
384 if (equal(n->text, ">")) return a > b;
385 if (equal(n->text, ">=")) return a >= b;
386 return b != 0;
387}

389static Actions actions(const char *text) {
390 Actions result = {allocate(10 * sizeof(Action)), 0};
391 while (*text) {
392 const char *separator = strstr(text, "; ");
393 const char *rest = separator ? substring(text, (size_t)(separator - text)) : text;
394 char *first = word(&rest);
395 Action *a = &result.items[result.count++];
396 a->literal = "";
397 if (equal(first, "raise") || equal(first, "send")) {
398 a->kind = equal(first, "raise") ? RAISE : SEND;
399 a->name = word(&rest);
400 if (a->kind == SEND) a->delay = atoi(word(&rest));
401 if (*rest) a->expression = expression(rest);
402 } else if (equal(first, "cancel")) {
403 a->kind = CANCEL;
404 a->name = copy_string(rest);
405 } else if (equal(first, "log")) {
406 a->kind = LOG;
407 if (*rest == '"') {
408 const char *end = strchr(rest + 1, '"');
409 a->literal = substring(rest + 1, (size_t)(end - rest - 1));
410 if (end[1]) a->expression = expression(end + 2);
411 } else a->expression = expression(rest);
412 } else {
413 a->kind = ASSIGN;
414 a->name = first;
415 a->expression = expression(rest + 2);
416 }
417 if (!separator) break;
418 text = separator + 2;
419 }
420 return result;
421}

423static char *action_references(const Chart *chart, Actions list) {
424 for (int i = 0; i < list.count; i++) {
425 Action *a = &list.items[i];
426 if (a->kind == ASSIGN && find_variable(chart, a->name) < 0) return format("unknown variable %s", a->name);
427 char *error = reference(chart, a->expression);
428 if (error) return error;
429 }
430 return NULL;
431}

433static Transition *parse_transition(const Chart *chart, const char *text) {
434 Transition *t = allocate(sizeof(*t));
435 const char *rest = text + 3;
436 t->source = find_state(chart, word(&rest));
437 t->target = -1;
438 char *trigger = word(&rest);
439 if (equal(trigger, "after")) {
440 t->trigger = TIMER_TRIGGER;
441 t->delay = atoi(word(&rest));
442 } else if (equal(trigger, "always")) t->trigger = ALWAYS_TRIGGER;
443 else {
444 t->trigger = EVENT_TRIGGER;
445 t->descriptors = allocate(5 * sizeof(char *));
446 char *start = trigger;
447 for (char *p = trigger;; p++) if (*p == '|' || *p == '\0') {
448 t->descriptors[t->descriptor_count++] = substring(start, (size_t)(p - start));
449 if (!*p) break;
450 start = p + 1;
451 }
452 }
453 if (*rest == '[') {
454 const char *close = strchr(rest, ']');
455 t->guard = expression(substring(rest + 1, (size_t)(close - rest - 1)));
456 rest = close + 1;
457 while (*rest == ' ') rest++;
458 }
459 if (starts(rest, "-> ") || starts(rest, "~> ")) {
460 t->mode = rest[0] == '~' ? LOCAL : EXTERNAL;
461 rest += 3;
462 t->target_name = word(&rest);
463 }
464 if (starts(rest, "/ ")) t->actions = actions(rest + 2);
465 return t;
466}

468static bool define(Chart *chart, Definition *definitions, int count) {
469 for (int i = 0; i < count; i++) {
470 Definition *d = &definitions[i];
471 Words a = words(d->text);
472 const char *kind = a.items[0];
473 d->state = -1;
474 if (equal(kind, "state") || equal(kind, "parallel") || equal(kind, "final") || equal(kind, "history")) {
475 char *parent_name = a.count > 2 && !equal(a.items[2], "initial") ? a.items[2] : NULL;
476 int parent = parent_name ? find_state(chart, parent_name) : 0;
477 if (find_state(chart, a.items[1]) >= 0) d->error = format("duplicate state %s", a.items[1]);
478 else if (parent < 0) d->error = format("unknown state %s", parent_name);
479 else if (chart->states[parent].kind == FINAL || chart->states[parent].kind == HISTORY) d->error = format("%s cannot have children", parent_name);
480 else {
481 int index = chart->state_count++;
482 State *state = &chart->states[index];
483 d->state = index;
484 state->name = a.items[1];
485 state->kind = equal(kind, "state") ? STATE : equal(kind, "parallel") ? PARALLEL : equal(kind, "final") ? FINAL : HISTORY;
486 state->parent = parent;
487 state->depth = chart->states[parent].depth + 1;
488 chart->states[parent].children[chart->states[parent].child_count++] = index;
489 if (state->kind == STATE) for (int j = 2; j < a.count; j++) if (equal(a.items[j], "initial")) state->initial_name = a.items[j + 1];
490 if (state->kind == HISTORY) {
491 state->deep = equal(a.items[3], "deep");
492 if (a.count == 5) state->default_name = a.items[4];
493 }
494 }
495 } else if (equal(kind, "var")) {
496 if (find_variable(chart, a.items[1]) >= 0) d->error = format("duplicate variable %s", a.items[1]);
497 else chart->variables[chart->variable_count++] = (Variable){a.items[1], strtoll(a.items[2], NULL, 10)};
498 }
499 }
500 for (int i = 0; i < count; i++) {
501 Definition *d = &definitions[i];
502 if (d->error) continue;
503 Words a = words(d->text);
504 const char *kind = a.items[0];
505 State *state = d->state >= 0 ? &chart->states[d->state] : NULL;
506 if (equal(kind, "state") && state->initial_name) {
507 int child = find_state(chart, state->initial_name);
508 if (child < 0) d->error = format("unknown state %s", state->initial_name);
509 else if (chart->states[child].parent != d->state || chart->states[child].kind == HISTORY) d->error = format("%s is not a child of %s", state->initial_name, state->name);
510 } else if (equal(kind, "final") || equal(kind, "history")) {
511 if (chart->states[state->parent].kind != STATE || child_state_count(chart, state->parent) == 0) d->error = format("%s needs a compound parent", state->name);
512 else if (equal(kind, "history") && state->default_name) {
513 int target = find_state(chart, state->default_name);
514 if (target < 0) d->error = format("unknown state %s", state->default_name);
515 else if (!descendant(chart, target, state->parent) || chart->states[target].kind == HISTORY) d->error = format("%s is not a descendant of %s", state->default_name, chart->states[state->parent].name);
516 }
517 } else if (equal(kind, "entry") || equal(kind, "exit")) {
518 int source = find_state(chart, a.items[1]);
519 if (source < 0) d->error = format("unknown state %s", a.items[1]);
520 else if (chart->states[source].kind == HISTORY) d->error = format("%s cannot have actions", a.items[1]);
521 else {
522 Actions list = actions(strstr(d->text, " / ") + 3);
523 d->error = action_references(chart, list);
524 if (!d->error) {
525 State *s = &chart->states[source];
526 if (equal(kind, "entry")) s->entries[s->entry_count++] = list;
527 else s->exits[s->exit_count++] = list;
528 }
529 }
530 } else if (equal(kind, "on")) {
531 int source = find_state(chart, a.items[1]);
532 if (source < 0) d->error = format("unknown state %s", a.items[1]);
533 else if (chart->states[source].kind == FINAL || chart->states[source].kind == HISTORY) d->error = format("%s cannot have transitions", a.items[1]);
534 else {
535 Transition *t = parse_transition(chart, d->text);
536 d->error = reference(chart, t->guard);
537 if (!d->error && t->target_name && find_state(chart, t->target_name) < 0) d->error = format("unknown state %s", t->target_name);
538 if (!d->error) d->error = action_references(chart, t->actions);
539 if (!d->error) {
540 if (t->target_name) t->target = find_state(chart, t->target_name);
541 chart->states[source].transitions[chart->states[source].transition_count++] = t;
542 }
543 }
544 }
545 }
546 bool valid = true;
547 for (int i = 0; i < count; i++) if (definitions[i].error) {
548 printf("error line %d: %s\n", definitions[i].number, definitions[i].error);
549 valid = false;
550 }
551 if (!valid) return false;
552 for (int i = 0; i < chart->states[0].child_count; i++) visit(chart, chart->states[0].children[i]);
553 return true;
554}

556static void warnings(Chart *chart) {
557 uint64_t reachable = bit(initial(chart, 0)), before;
558 do {
559 before = reachable;
560 for (int i = 0; i < chart->order_count; i++) {
561 int index = chart->order[i];
562 if (!contains(before, index)) continue;
563 State *s = &chart->states[index];
564 if (s->parent != 0) reachable |= bit(s->parent);
565 if (compound(chart, index)) reachable |= bit(initial(chart, index));
566 if (parallel(chart, index)) for (int j = 0; j < s->child_count; j++) if (chart->states[s->children[j]].kind != HISTORY) reachable |= bit(s->children[j]);
567 for (int j = 0; j < s->transition_count; j++) {
568 int target = s->transitions[j]->target;
569 if (target >= 0) reachable |= bit(chart->states[target].kind == HISTORY ? default_target(chart, target) : target);
570 }
571 }
572 } while (reachable != before);
573 for (int i = 0; i < chart->order_count; i++) {
574 int index = chart->order[i];
575 if (chart->states[index].kind != HISTORY && !contains(reachable, index)) printf("warning: unreachable state %s\n", chart->states[index].name);
576 }
577}

580 const State *target = &chart->states[t->target];

585static bool all_descendants(const Chart *chart, uint64_t targets, int scope) {
586 for (int i = 0; i < chart->order_count; i++) {
587 int state = chart->order[i];
588 if (contains(targets, state) && !descendant(chart, state, scope)) return false;
589 }
590 return true;

593static Choice make_choice(const Chart *chart, Transition *t) {
594 Choice choice = {t, -1, 0};
595 if (t->target < 0) return choice;
596 uint64_t targets = effective(chart, t);
597 if (t->mode == LOCAL && compound(chart, t->source) && all_descendants(chart, targets, t->source)) choice.scope = t->source;
598 else for (int ancestor = chart->states[t->source].parent; ancestor >= 0; ancestor = chart->states[ancestor].parent) {
599 if (compound(chart, ancestor) && all_descendants(chart, targets, ancestor)) {
600 choice.scope = ancestor;
601 break;
602 }
603 }
604 for (int i = 0; i < chart->order_count; i++) {
605 int state = chart->order[i];
606 if (chart->states[state].active && descendant(chart, state, choice.scope)) choice.exits |= bit(state);
607 }
608 return choice;
609}

611static bool matches(const Transition *t, Trigger trigger) {
612 if (trigger.kind == ALWAYS_TRIGGER) return t->trigger == ALWAYS_TRIGGER;
613 if (trigger.kind == TIMER_TRIGGER) return t == trigger.timer;
614 if (t->trigger != EVENT_TRIGGER) return false;
615 for (int i = 0; i < t->descriptor_count; i++) {
616 const char *d = t->descriptors[i];
617 size_t length = strlen(d);
618 if (equal(d, "*") || equal(d, trigger.event.name) || (starts(trigger.event.name, d) && trigger.event.name[length] == '.')) return true;
619 }
620 return false;
621}

627static Choices select_transitions(Chart *chart, Trigger trigger) {
628 Transition *candidates[40];
629 int count = 0;
630 for (int i = 0; i < chart->order_count; i++) {
631 int state = chart->order[i];
632 if (!chart->states[state].active || !atomic(chart, state)) continue;
633 Transition *selected = NULL;
634 for (int source = state; source != 0 && !selected; source = chart->states[source].parent) {
635 State *s = &chart->states[source];
636 for (int j = 0; j < s->transition_count; j++) if (matches(s->transitions[j], trigger)) {
637 Transition *t = s->transitions[j];
638 bool success = true;
639 long long guard = t->guard ? evaluate(chart, t->guard, &success) : 1;
640 if (!success) enqueue(chart, "error.execution", 0);
641 if (success && guard != 0) {
642 selected = t;
643 break;
644 }
645 }
646 }
647 if (selected) {
648 bool seen = false;
649 for (int j = 0; j < count; j++) if (candidates[j] == selected) seen = true;
650 if (!seen) candidates[count++] = selected;
651 }
652 }
653 Choices result = {0};
654 uint64_t exited = 0;
655 for (int i = 0; i < count; i++) {
656 Choice choice = make_choice(chart, candidates[i]);
657 if (!(choice.exits & exited)) {
658 result.items[result.count++] = choice;
659 exited |= choice.exits;
660 }
661 }
662 return result;
663}

674static void perform(Chart *chart, Actions list) {
675 for (int i = 0; i < list.count; i++) {
676 Action *a = &list.items[i];
677 bool success = true;
678 long long value = a->expression ? evaluate(chart, a->expression, &success) : 0;
679 if (!success) {
680 enqueue(chart, "error.execution", 0);
681 return;
682 }
683 switch (a->kind) {
684 case ASSIGN:
685 chart->variables[find_variable(chart, a->name)].value = value;
686 break;
687 case RAISE:
688 enqueue(chart, a->name, value);
689 break;
690 case SEND:
691 reserve(chart, NULL, (Event){a->name, value}, a->delay);
692 break;
693 case CANCEL:
694 for (int j = 0; j < chart->reservation_count;) {
695 Reservation *r = &chart->reservations[j];
696 if (!r->timer && equal(r->event.name, a->name)) remove_reservation(chart, j);
697 else j++;
698 }
699 break;
700 case LOG:
701 fputs(a->literal, stdout);
702 if (a->expression) printf("%lld", value);
703 putchar('\n');
704 break;
705 }
706 }
707}

709static uint64_t expand(const Chart *chart, uint64_t entry) {
710 uint64_t before;
711 do {
712 before = entry;
713 for (int i = 0; i < chart->order_count; i++) {
714 int state = chart->order[i];
715 if (!contains(entry, state)) continue;
716 const State *s = &chart->states[state];
717 bool has_child = false;
718 for (int j = 0; j < s->child_count; j++) {
719 int child = s->children[j];
720 if (chart->states[child].kind == HISTORY) continue;
721 if (parallel(chart, state)) entry |= bit(child);
722 if (contains(entry, child)) has_child = true;
723 }
724 if (compound(chart, state) && !has_child) entry |= bit(initial(chart, state));
725 }
726 } while (entry != before);
727 return entry;
728}

730static bool complete(const Chart *chart, int state) {
731 const State *s = &chart->states[state];
732 if (compound(chart, state)) {
733 for (int i = 0; i < s->child_count; i++) {
734 const State *child = &chart->states[s->children[i]];
735 if (child->kind == FINAL && child->active) return true;
736 }
737 return false;
738 }
739 if (!parallel(chart, state)) return false;
740 for (int i = 0; i < s->child_count; i++) if (chart->states[s->children[i]].kind != HISTORY && !complete(chart, s->children[i])) return false;
741 return true;
742}

744static void enter(Chart *chart, uint64_t entry) {
745 for (int i = 0; i < chart->order_count; i++) {
746 int state = chart->order[i];
747 if (!contains(entry, state)) continue;
748 State *s = &chart->states[state];
749 s->active = true;
750 s->entered++;
751 s->entry_time = chart->time;
752 for (int j = 0; j < s->transition_count; j++) if (s->transitions[j]->trigger == TIMER_TRIGGER) reserve(chart, s->transitions[j], (Event){0}, s->transitions[j]->delay);
753 for (int j = 0; j < s->entry_count; j++) perform(chart, s->entries[j]);
754 if (s->kind == FINAL) {
755 int parent = s->parent;
756 enqueue(chart, format("done.state.%s", chart->states[parent].name), 0);
757 int grandparent = chart->states[parent].parent;
758 if (grandparent >= 0 && parallel(chart, grandparent) && complete(chart, grandparent)) enqueue(chart, format("done.state.%s", chart->states[grandparent].name), 0);
759 }
760 }
761}

776static void print_trigger(const Chart *chart, Trigger trigger) {
777 if (trigger.kind == ALWAYS_TRIGGER) fputs("always", stdout);
778 else if (trigger.kind == TIMER_TRIGGER) printf("after(%d)@%s", trigger.timer->delay, chart->states[trigger.timer->source].name);
779 else {
780 fputs(trigger.event.name, stdout);
781 if (trigger.event.value != 0) printf("=%lld", trigger.event.value);

785static bool micro(Chart *chart, Choices selected, Trigger trigger) {
786 if (chart->in_macro == 1000) {
787 printf("LIVELOCK t=%lld\n", chart->time);
788 chart->halted = true;
789 return false;
790 }
791 chart->in_macro++;
792 chart->microsteps++;
793 uint64_t exits = 0;
794 for (int i = 0; i < selected.count; i++) exits |= selected.items[i].exits;
795 for (int i = 0; i < chart->order_count; i++) {
796 State *history = &chart->states[chart->order[i]];
797 if (history->kind != HISTORY || !contains(exits, history->parent)) continue;
798 history->history_recorded = true;
799 history->history = 0;
800 for (int j = 0; j < chart->order_count; j++) {
801 int state = chart->order[j];
802 if (chart->states[state].active && (history->deep ? atomic(chart, state) && descendant(chart, state, history->parent) : chart->states[state].parent == history->parent && chart->states[state].kind != HISTORY)) history->history |= bit(state);
803 }
804 }
805 uint64_t entry = 0;
806 for (int i = 0; i < selected.count; i++) {
807 Choice choice = selected.items[i];
808 if (choice.transition->target < 0) continue;
809 uint64_t targets = effective(chart, choice.transition);
810 for (int j = 0; j < chart->order_count; j++) {
811 int target = chart->order[j];
812 if (contains(targets, target)) for (int state = target; state != choice.scope; state = chart->states[state].parent) entry |= bit(state);
813 }
814 }
815 entry = expand(chart, entry);
816 if (chart->trace) {
817 printf("[t=%lld] ", chart->time);
818 print_trigger(chart, trigger);
819 fputs(" (", stdout);
820 for (int i = 0; i < selected.count; i++) {
821 if (i) putchar(',');
822 fputs(chart->states[selected.items[i].transition->source].name, stdout);
823 }
824 fputs("): exit ", stdout);
825 print_states(chart, exits, true);
826 fputs("; enter ", stdout);
827 print_states(chart, entry, false);
828 putchar('\n');
829 }
830 for (int i = chart->order_count - 1; i >= 0; i--) {
831 int state = chart->order[i];
832 if (!contains(exits, state)) continue;
833 State *s = &chart->states[state];
834 for (int j = 0; j < s->exit_count; j++) perform(chart, s->exits[j]);
835 s->active = false;
836 s->dwell += chart->time - s->entry_time;
837 for (int j = 0; j < chart->reservation_count;) {
838 if (chart->reservations[j].timer && chart->reservations[j].timer->source == state) remove_reservation(chart, j);
839 else j++;
840 }
841 }
842 for (int i = 0; i < selected.count; i++) {
843 Transition *t = selected.items[i].transition;
844 perform(chart, t->actions);
845 chart->states[t->source].taken++;
846 }
847 enter(chart, entry);
848 return true;
849}

851static void macro(Chart *chart, Trigger trigger) {
852 chart->macrosteps++;
853 chart->in_macro = 0;
854 if (trigger.kind == START_TRIGGER) {
855 chart->data = 0;
856 enter(chart, expand(chart, bit(initial(chart, 0))));
857 } else {
858 chart->data = trigger.kind == TIMER_TRIGGER ? 0 : trigger.event.value;
859 Choices selected = select_transitions(chart, trigger);
860 if (selected.count && !micro(chart, selected, trigger)) return;
861 }
862 while (!chart->halted) {
863 chart->data = 0;
864 Trigger always = {
865 ALWAYS_TRIGGER, {0}, NULL
866 };
867 Choices selected = select_transitions(chart, always);
868 if (selected.count) {
869 if (!micro(chart, selected, always)) return;
870 continue;
871 }
872 if (chart->queue_front == chart->queue_count) return;
873 Event event = chart->queue[chart->queue_front++];
874 chart->data = event.value;
875 Trigger internal = {EVENT_TRIGGER, event, NULL};
876 selected = select_transitions(chart, internal);
877 if (selected.count && !micro(chart, selected, internal)) return;
878 }
879}

881static int compare_reservations(const void *a, const void *b) {
882 const Reservation *x = a, *y = b;
883 if (x->due != y->due) return x->due < y->due ? -1 : 1;
884 return (x->number > y->number) - (x->number < y->number);
885}

887static void dump(Chart *chart) {
888 printf("dump t=%lld\n", chart->time);
889 for (int i = 0; i < chart->order_count; i++) {
890 State *s = &chart->states[chart->order[i]];
891 if (s->active) printf("%*s%s\n", 2 * s->depth, "", s->name);
892 }
893 int variables[20];
894 for (int i = 0; i < chart->variable_count; i++) {
895 int j = i;
896 while (j > 0 && strcmp(chart->variables[variables[j - 1]].name, chart->variables[i].name) > 0) {
897 variables[j] = variables[j - 1];
898 j--;
899 }
900 variables[j] = i;
901 }
902 for (int i = 0; i < chart->variable_count; i++) printf(" var %s = %lld\n", chart->variables[variables[i]].name, chart->variables[variables[i]].value);
903 for (int i = 0; i < chart->order_count; i++) {
904 State *s = &chart->states[chart->order[i]];
905 if (s->kind == HISTORY && s->history_recorded) {
906 printf(" history %s = ", s->name);
907 print_states(chart, s->history, false);
908 putchar('\n');
909 }
910 }
911 qsort(chart->reservations, (size_t)chart->reservation_count, sizeof(Reservation), compare_reservations);
912 for (int i = 0; i < chart->reservation_count; i++) {
913 Reservation *r = &chart->reservations[i];
914 if (r->timer) printf(" timer %lld after(%d)@%s\n", r->due, r->timer->delay, chart->states[r->timer->source].name);
915 else printf(" send %lld %s %lld\n", r->due, r->event.name, r->event.value);
916 }
917}

919static void script(Chart *chart, const char *text) {
920 if (chart->halted) return;
921 const char *rest = text;
922 char *command = word(&rest);
923 if (equal(command, "event")) {
924 char *name = word(&rest);
925 macro(chart, (Trigger) {
926 EVENT_TRIGGER, {name, *rest ? strtoll(rest, NULL, 10) : 0}, NULL
927 });
928 } else if (equal(command, "tick")) {
929 long long finish = chart->time + strtoll(rest, NULL, 10);
930 while (!chart->halted) {
931 int next = -1;
932 for (int i = 0; i < chart->reservation_count; i++) {
933 Reservation *r = &chart->reservations[i];
934 if (r->due <= finish && (next < 0 || compare_reservations(r, &chart->reservations[next]) < 0)) next = i;
935 }
936 if (next < 0) break;
937 Reservation r = chart->reservations[next];
938 remove_reservation(chart, next);
939 chart->time = r.due;
940 macro(chart, (Trigger){r.timer ? TIMER_TRIGGER : EVENT_TRIGGER, r.event, r.timer});
941 }
942 if (!chart->halted) chart->time = finish;
943 } else if (equal(command, "set")) {
944 char *name = word(&rest);
945 int index = find_variable(chart, name);
946 if (index < 0) printf("error: unknown variable %s\n", name);
947 else chart->variables[index].value = strtoll(rest, NULL, 10);
948 } else if (equal(command, "trace")) chart->trace = equal(rest, "on");
949 else if (equal(command, "eval")) {
950 Node *n = expression(rest);
951 char *error = reference(chart, n);
952 if (error) printf("error: %s\n", error);
953 else {
954 chart->data = 0;
955 bool success = true;
956 long long value = evaluate(chart, n, &success);
957 if (!success) puts("error: division by zero");
958 else printf("%s -> %lld\n", rest, value);
959 }
960 } else if (equal(command, "dump")) dump(chart);
961}
