9const LEVELS=(("||",), ("&&",), ("==", "!=", "<", "<=", ">", ">="), ("+", "-"), ("*", "/", "%"))
10mutable struct ExpressionReader
11 tokens::Vector{String}
12 position::Int
13end
14function expression(text)
15 tokens=[
16 String(m.match) for
17 m in eachmatch(r"[0-9]+|[A-Za-z_][A-Za-z0-9_]*|\|\||&&|[=!<>]=|[-+*/%!()<>]", text)
18 ]
19 readexpression!(ExpressionReader(tokens, 1), 1)
20end
21exprpeek(r) = r.position<=length(r.tokens) ? r.tokens[r.position] : ""
22function readtoken!(r::ExpressionReader)
23 token=r.tokens[r.position]
24 r.position+=1
25 token
26end
27function readexpression!(r, level)
28 level==6 && return readunary!(r)
29 node=readexpression!(r, level+1)
30 while exprpeek(r) in LEVELS[level]
31 op=readtoken!(r)
32 right=readexpression!(r, level+1)
33 node=ExpressionNode(:binary, op, 0, [node, right])
34 end
35 node
36end
37function readunary!(r)
38 token=readtoken!(r)
39 token in ("!", "-") && return ExpressionNode(:unary, token, 0, [readunary!(r)])
40 if token=="("
41 node=readexpression!(r, 1)
42 readtoken!(r)
43 return node
44 end
45 isdigit(token[1]) && return ExpressionNode(:number, "", parse(Int, token))
46 if token=="in"
47 readtoken!(r)
48 name=readtoken!(r)
49 readtoken!(r)
50 return ExpressionNode(:state, name)
51 end
52 ExpressionNode(token=="_data" ? :data : :variable, token)
53end
54struct Action
55 kind::Symbol
56 name::String
57 delay::Int
58 text::String

61function actions(text)
62 result=Action[]
63 isempty(text) && return result
64 for item in split(text, "; ")
65 parts=split(item, ' '; limit = 2)
66 word=parts[1]
67 rest=parts[2]
68 if word=="raise"
69 fields=split(rest, ' '; limit = 2)
70 push!(
71 result,
72 Action(
73 :raise,
74 fields[1],
75 0,
76 "",
77 length(fields)==2 ? expression(fields[2]) : nothing,
78 ),
79 )
80 elseif word=="send"
81 fields=split(rest, ' '; limit = 3)
82 push!(
83 result,
84 Action(
85 :send,
86 fields[1],
87 parse(Int, fields[2]),
88 "",
89 length(fields)==3 ? expression(fields[3]) : nothing,
90 ),
91 )
92 elseif word=="cancel"
93 push!(result, Action(:cancel, rest, 0, "", nothing))
94 elseif word=="log"
95 if startswith(rest, '"')
96 closing=findnext('"', rest, 2)
97 tail=closing<length(rest) ? rest[(closing+2):end] : ""
98 push!(
99 result,
100 Action(
101 :log,
102 "",
103 0,
104 rest[2:(closing-1)],
105 isempty(tail) ? nothing : expression(tail),
106 ),
107 )
108 else
109 push!(result, Action(:log, "", 0, "", expression(rest)))
110 end
111 else
112 push!(result, Action(:assign, word, 0, "", expression(rest[3:end])))
113 end
114 end
115 result
116end

177function checkexpression(c, node)
178 node===nothing && return ""
179 node.kind==:variable &&
180 !haskey(c.variables, node.text) &&
181 return "unknown variable $(node.text)"
182 node.kind==:state && !haskey(c.names, node.text) && return "unknown state $(node.text)"
183 for child in node.children
184 error=checkexpression(c, child)
185 !isempty(error) && return error
186 end
187 ""
188end

411struct EvaluationError <: Exception end
412function evaluate(e, node)
413 node.kind==:number && return node.value
414 node.kind==:variable && return e.variables[node.text]
415 node.kind==:data && return e.data
416 node.kind==:state && return Int(e.chart.names[node.text] in e.config)
417 if node.kind==:unary
418 value=evaluate(e, node.children[1])
419 return node.text=="!" ? Int(value==0) : -value
420 end
421 a=evaluate(e, node.children[1])
422 op=node.text
423 op=="&&" && a==0 && return 0
424 op=="||" && a!=0 && return 1
425 b=evaluate(e, node.children[2])
426 op in ("&&", "||") && return Int(b!=0)
427 if op in ("/", "%")
428 b==0 && throw(EvaluationError())
429 q=div(a, b)
430 return op=="/" ? q : a-q*b
431 end
432 op=="+" && return a+b
433 op=="-" && return a-b
434 op=="*" && return a*b
435 Int(op=="==" ? a==b : op=="!=" ? a!=b : op=="<" ? a<b : op=="<=" ? a<=b : op==">" ? a>b : a>=b)
436end
437function reserve!(e, delay, transition, event)
438 e.serial+=1
439 push!(e.reservations, Reservation(e.clock+delay, e.serial, transition, event))
440end
441function performactions!(e, list)
442 try
443 for action in list
444 value=action.expr===nothing ? 0 : evaluate(e, action.expr)
445 if action.kind==:assign
446 e.variables[action.name]=value
447 elseif action.kind==:raise
448 push!(e.queue, ChartEvent(action.name, value))
449 elseif action.kind==:send
450 reserve!(e, action.delay, 0, ChartEvent(action.name, value))
451 elseif action.kind==:cancel
452 filter!(r->r.transition!=0 || r.event.name!=action.name, e.reservations)
453 elseif action.kind==:log
454 println(action.text, action.expr===nothing ? "" : string(value))
455 end
456 end
457 catch error
458 error isa EvaluationError || rethrow()
459 push!(e.queue, ChartEvent("error.execution", 0))
460 end
461end

527 end

532function matches(tr, kind, event, timer, index)
533 kind=="always" && return tr.kind=="always"
534 kind=="timer" && return index==timer
535 tr.kind=="event" &&
536 any(d->d=="*" || d==event.name || startswith(event.name, d*"."), tr.descriptors)
537end
538function selecttransitions!(e, kind, event, timer = 0)
539 c=e.chart
540 candidates=Int[]
541 for a in c.order
542 a in e.config && atomic(c, a) || continue
543 s=a
544 found=0
545 while s!=1 && found==0
546 for t in c.states[s].transitions
547 tr=c.transitions[t]
548 matches(tr, kind, event, timer, t) || continue
549 accepted=true
550 try
551 accepted=tr.guard===nothing || evaluate(e, tr.guard)!=0
552 catch error
553 error isa EvaluationError || rethrow()
554 push!(e.queue, ChartEvent("error.execution", 0))
555 accepted=false
556 end
557 if accepted
558 found=t
559 break
560 end
561 end
562 s=c.states[s].parent
563 end
564 found!=0 && !(found in candidates) && push!(candidates, found)
565 end
566 kept=Int[]
567 exiting=Set{Int}()
568 for t in candidates
569 exits=exitset(e, c.transitions[t])
570 if isempty(intersect(exiting, exits))
571 push!(kept, t)
572 union!(exiting, exits)
573 end
574 end
575 kept
576end
577statenames(c, ids) = isempty(ids) ? "-" : join((c.states[s].name for s in ids), ',')
578timername(c, t) = "after($(c.transitions[t].delay))@$(c.states[c.transitions[t].source].name)"
579eventname(event) = event.value==0 ? event.name : "$(event.name)=$(event.value)"
580function microstep!(e, selected, display)
581 if e.within==1000
582 println("LIVELOCK t=$(e.clock)")
583 e.halted=true
584 return
585 end
586 e.within+=1
587 e.micros+=1
588 c=e.chart
589 scopes=[scope(e, c.transitions[t]) for t in selected]
590 exiting=Set{Int}()
591 for bound in scopes
592 bound!=0 && union!(exiting, (s for s in e.config if descendant(c, s, bound)))
593 end
594 for s in c.order
595 s in exiting || continue
596 for h in c.states[s].children
597 c.states[h].kind=="history" || continue
598 e.histories[h]=[
599 x for x in c.order if x in e.config &&
600 (c.states[h].deep ? atomic(c, x) && descendant(c, x, s) : c.states[x].parent==s)
601 ]
602 end
603 end
604 seeds=Int[]
605 for (t, bound) in zip(selected, scopes)
606 tr=c.transitions[t]
607 tr.target==0 && continue
608 for target in effective(e, tr)
609 push!(seeds, target)
610 s=c.states[target].parent
611 while s!=bound
612 push!(seeds, s)
613 s=c.states[s].parent
614 end
615 end
616 end
617 entering=closedentries(c, seeds)
618 e.trace && println(
619 "[t=$(e.clock)] $display (",
620 statenames(c, [c.transitions[t].source for t in selected]),
621 "): exit ",
622 statenames(c, [s for s in Iterators.reverse(c.order) if s in exiting]),
623 "; enter ",
624 statenames(c, [s for s in c.order if s in entering]),
625 )
626 for s in Iterators.reverse(c.order)
627 s in exiting || continue
628 for list in c.states[s].exits
629 performactions!(e, list)
630 end
631 delete!(e.config, s)
632 e.dwell[s]+=e.clock-e.entrytime[s]
633 filter!(r->r.transition==0 || c.transitions[r.transition].source!=s, e.reservations)
634 end
635 for t in selected
636 tr=c.transitions[t]
637 performactions!(e, tr.actions)
638 e.taken[tr.source]+=1
639 end
640 enterstates!(e, entering)
641end
642function macrostep!(e, kind, event = ChartEvent("", 0), timer = 0)
643 e.macros+=1
644 e.within=0
645 e.data=event.value
646 if kind=="start"
647 enterstates!(e, closedentries(e.chart, [initial(e.chart, 1)]))
648 else
649 selected=selecttransitions!(e, kind, event, timer)
650 !isempty(selected) &&
651 microstep!(e, selected, kind=="timer" ? timername(e.chart, timer) : eventname(event))
652 end
653 while !e.halted
654 e.data=0
655 selected=selecttransitions!(e, "always", ChartEvent("", 0))
656 if !isempty(selected)
657 microstep!(e, selected, "always")
658 continue
659 end
660 isempty(e.queue) && break
661 queued=popfirst!(e.queue)
662 e.data=queued.value
663 selected=selecttransitions!(e, "event", queued)
664 !isempty(selected) && microstep!(e, selected, eventname(queued))
665 end
666end

675 p!=1 && push!(reached, p)
676 compound(c, s) && push!(reached, initial(c, s))

678 for t in c.states[s].transitions
679 target=c.transitions[t].target
680 target!=0 && push!(
681 reached,
682 c.states[target].kind=="history" ? defaulttarget(c, target) : target,
683 )
684 end
685 end
686 changed=before!=length(reached)
687 end
688 for s in c.order
689 c.states[s].kind!="history" &&
690 !(s in reached) &&

692 end
693end

714function script!(e, text)
715 fields=split(text, ' ')
716 command=fields[1]
717 if command=="event"
718 macrostep!(e, "event", ChartEvent(fields[2], length(fields)==3 ? parse(Int, fields[3]) : 0))
719 elseif command=="tick"
720 finish=e.clock+parse(Int, fields[2])
721 while !e.halted && !isempty(e.reservations)
722 index=argmin([(r.due, r.number) for r in e.reservations])
723 r=e.reservations[index]
724 r.due>finish && break
725 deleteat!(e.reservations, index)
726 e.clock=r.due
727 macrostep!(e, r.transition==0 ? "event" : "timer", r.event, r.transition)
728 end
729 !e.halted && (e.clock=finish)
730 elseif command=="set"
731 name=fields[2]
732 if haskey(e.variables, name)
733 e.variables[name]=parse(Int, fields[3])
734 else
735 println("error: unknown variable $name")
736 end
737 elseif command=="trace"
738 e.trace=fields[2]=="on"
739 elseif command=="dump"
740 dumpstate(e)
741 elseif command=="eval"
742 text=text[6:end]
743 node=expression(text)
744 error=checkexpression(e.chart, node)
745 if !isempty(error)
746 println("error: $error")
747 else
748 e.data=0
749 try
750 value=evaluate(e, node)
751 println("$text -> $value")
752 catch error
753 error isa EvaluationError || rethrow()
754 println("error: division by zero")
755 end
756 end
757 end
758end
759function runchart(name, definitions, scripts)
760 println("chart $name")
761 c, errors=buildchart(definitions)
762 if !isempty(errors)
763 for line in sort(collect(keys(errors)))
764 println("error line $line: $(errors[line])")
765 end
766 return
767 end
768 warnings(c)
769 e=Engine(c)
770 macrostep!(e, "start")
771 for text in scripts
772 e.halted && break
773 script!(e, text)
774 end
775 println("end t=$(e.clock) macrosteps=$(e.macros) microsteps=$(e.micros)")
776 for s in c.order
777 c.states[s].kind=="history" && continue
778 total=e.dwell[s]+(s in e.config ? e.clock-e.entrytime[s] : 0)
779 println(" $(c.states[s].name) entered=$(e.entered[s]) taken=$(e.taken[s]) time=$total")
780 end
781end
782function main()
783 lines=readlines()
784 position=2
785 for _ in 1:parse(Int, lines[1])
786 name=lines[position][7:end]
787 position+=1
788 definitions=Tuple{Int,String}[]
789 while lines[position]!="RUN"
790 push!(definitions, (position, lines[position]))
791 position+=1
792 end
793 position+=1
794 scripts=String[]
795 while lines[position]!="END"
796 push!(scripts, lines[position])
797 position+=1
798 end
799 position+=1
800 runchart(name, definitions, scripts)
801 end
802end
