local output = {}
local function emit(format, ...)
    output[#output + 1] = string.format(format, ...)
local function words(text)
    local result = {}
    for word in text:gmatch("%S+") do
        result[#result + 1] = word
    end
    return result
local function split(text, separator)
    local result, start = {}, 1
    while true do
        local first, last = text:find(separator, start, true)
        if not first then
            result[#result + 1] = text:sub(start)
            return result
        end
        result[#result + 1] = text:sub(start, first - 1)
        start = last + 1
    end
local function parse_actions(text)
    local actions = {}
    if not text or text == "" then
        return actions
    end
    for _, item in ipairs(split(text, "; ")) do
        local action = {}
        if item:sub(1, 6) == "raise " then
        elseif item:sub(1, 4) == "log " then
            action.kind, action.text = "log", ""
            local tail = item:sub(5)
            if tail:sub(1, 1) == '"' then
                local last = tail:find('"', 2, true)
                action.text = tail:sub(2, last - 1)
                if last < #tail then
                    action.value = parse_expression(tail:sub(last + 2))
                end
            end
        end
        actions[#actions + 1] = action
    end
    return actions
local Chart = {}
Chart.__index = Chart
function Chart.new()
    local root = {
        name = "",
        kind = "root",
        children = {},
        child_states = {},
        depth = 0,
        transitions = {}
    }
    return setmetatable(
        {
            root = root,
            states = {},
            names = {},
            variables = {},
            variable_names = {},
            document = {},
            active = {},
            history = {},
            queue = {},
            queue_head = 1,
            reservations = {},
            time = 0,
            creation = 0,
            data = 0,
            macrosteps = 0,
            microsteps = 0,
            trace = false,
            halted = false
        },
        Chart
function Chart:compound(state)
    return state == self.root or state.kind == "state" and #state.child_states > 0
end
function Chart:parallel(state)
    return state.kind == "parallel" and #state.child_states > 0
end

function Chart:reference(expression)
    if not expression then
        return
    end
function Chart:action_references(actions)
    for _, action in ipairs(actions) do
        if action.kind == "assign" and self.variables[action.name] == nil then
            return "unknown variable " .. action.name
        end
        local problem = self:reference(action.value)
        if problem then
            return problem
        end
    end
function Chart:parse_transition(text)
    local source, tail = text:match("^on (%S+) (.+)$")
    local transition = {source = self.names[source], mode = "internal", actions = {}}
    local bracket = 0
    for i = 1, #tail - 2 do
        local c = tail:sub(i, i)
        if c == "[" then
            bracket = bracket + 1
        elseif c == "]" then
            bracket = bracket - 1
        end
        if bracket == 0 and tail:sub(i, i + 2) == " / " then
            transition.actions = parse_actions(tail:sub(i + 3))
            tail = tail:sub(1, i - 1)
            break
        end
    end
    local opening = tail:find(" [", 1, true)
    if opening then
        local closing = tail:find("]", opening + 2, true)
        transition.guard = parse_expression(tail:sub(opening + 2, closing - 1))
    end
    for _, arrow in ipairs({" -> ", " ~> "}) do
        local position = tail:find(arrow, 1, true)
        if position then
        end
    end
    if tail == "always" then
        transition.kind = "always"
    elseif tail:sub(1, 6) == "after " then
        transition.kind, transition.delay = "timer", tonumber(tail:sub(7))
    else
        transition.kind, transition.descriptors = "event", split(tail, "|")
    end
    return transition
function Chart:define(definitions)
    for _, definition in ipairs(definitions) do
        local fields = words(definition.text)
        local kind, name = fields[1], fields[2]
        definition.kind = kind
        if kind == "state" or kind == "parallel" or kind == "final" or kind == "history" then
            local parent_name = fields[3] ~= "initial" and fields[3] or nil
            local parent = parent_name and self.names[parent_name] or self.root
            if self.names[name] then
                definition.error = "duplicate state " .. name
            elseif parent_name and not self.names[parent_name] then
                definition.error = "unknown state " .. parent_name
            elseif parent.kind == "final" or parent.kind == "history" then
                definition.error = parent.name .. " cannot have children"
            else
                local state = {
                    name = name,
                    kind = kind,
                    parent = parent,
                    depth = parent.depth + 1,
                    children = {},
                    child_states = {},
                    entries = {},
                    exits = {},
                    transitions = {},
                    entered = 0,
                    taken = 0,
                    dwell = 0,
                    entry_time = 0
                }
                if kind == "state" then
                    for i = 3, #fields do
                        if fields[i] == "initial" then
                            state.initial_name = fields[i + 1]
                            break
                        end
                    end
                end
                definition.state = state
                self.names[name], self.states[#self.states + 1] = state, state
                parent.children[#parent.children + 1] = state
                if kind ~= "history" then
                    parent.child_states[#parent.child_states + 1] = state
                end
            end
        end
    end
    for _, definition in ipairs(definitions) do
        if not definition.error then
            local fields, kind, state = words(definition.text), definition.kind, definition.state
            if kind == "state" and state.initial_name then
            elseif kind == "on" then
                state = self.names[fields[2]]
                if not state then
                    definition.error = "unknown state " .. fields[2]
                elseif state.kind == "final" or state.kind == "history" then
                    definition.error = state.name .. " cannot have transitions"
                else
                    local transition = self:parse_transition(definition.text)
                    definition.error = self:reference(transition.guard)
                    if not definition.error and transition.target_name and not transition.target then
                        definition.error = "unknown state " .. transition.target_name
                    end
                    if not definition.error then
                        definition.error = self:action_references(transition.actions)
                    end
                    if not definition.error then
                        state.transitions[#state.transitions + 1] = transition
                    end
                end
            end
        end
    end
    local valid = true
    for _, definition in ipairs(definitions) do
        if definition.error then
            emit("error line %d: %s", definition.line, definition.error)
            valid = false
        end
    end
    if not valid then
        return false
    end
    self.root.initial = self.root.child_states[1]
    for _, state in ipairs(self.states) do
        state.initial =
            state.initial_name and self.names[state.initial_name] or state.child_states[1]
    end
    local function visit(state)
        self.document[#self.document + 1] = state
        for _, child in ipairs(state.children) do
            visit(child)
        end
    end
    for _, child in ipairs(self.root.children) do
        visit(child)
    end
    table.sort(self.variable_names)
    return true
function Chart:enqueue(name, value)
    self.queue[#self.queue + 1] = {name = name, value = value}
function Chart:perform(actions)
    for _, action in ipairs(actions) do
        local value = 0
        if action.value then
            value = self:evaluate(action.value)
        end
        if value == nil then
            self:enqueue("error.execution", 0)
            return
        end
        if action.kind == "assign" then
            self.variables[action.name] = value
        elseif action.kind == "raise" then
            self:enqueue(action.name, value)
        elseif action.kind == "log" then
            emit("%s", action.text .. (action.value and tostring(value) or ""))
        end
    end
function Chart:transition_set(transition)
    local selected = {transition = transition, exits = {}}
    if transition.mode == "internal" then
        return selected
    end
local function matches(transition, trigger)
    if transition.kind ~= trigger.kind then
        return false
    end
    if trigger.kind == "always" then
        return true
    end
    if trigger.kind == "timer" then
        return transition == trigger.timer
    end
    for _, descriptor in ipairs(transition.descriptors) do
        local name = trigger.event.name
        if
            descriptor == "*" or descriptor == name or
                name:sub(1, #descriptor + 1) == descriptor .. "."
         then
            return true
        end
    end
    return false
end
function Chart:select_transitions(trigger)
    local result, seen, occupied = {}, {}, {}
    for _, atomic in ipairs(self.document) do
        if self.active[atomic] and #atomic.child_states == 0 then
            local state, found = atomic, nil
            while state ~= self.root and not found do
                for _, transition in ipairs(state.transitions) do
                    if matches(transition, trigger) then
                        local value = 1
                        if transition.guard then
                            value = self:evaluate(transition.guard)
                        end
                        if value == nil then
                            self:enqueue("error.execution", 0)
                        elseif value ~= 0 then
                            found = transition
                            break
                        end
                    end
                end
                state = state.parent
            end
            if found and not seen[found] then
                seen[found] = true
                local selected, conflict = self:transition_set(found), false
                for state in pairs(selected.exits) do
                    if occupied[state] then
                        conflict = true
                        break
                    end
                end
                if not conflict then
                    result[#result + 1] = selected
                    for state in pairs(selected.exits) do
                        occupied[state] = true
                    end
                end
            end
        end
    end
    return result
function Chart:expand(entries)
    local changed = true
    while changed do
        changed = false
        local function add(state)
            if not entries[state] then
                entries[state], changed = true, true
            end
        end
        for _, state in ipairs(self.document) do
            if entries[state] then
                if self:parallel(state) then
                    for _, child in ipairs(state.child_states) do
                        add(child)
                    end
                elseif self:compound(state) then
                end
            end
        end
    end
function Chart:enter(entries)
    for _, state in ipairs(self.document) do
        if entries[state] then
            self.active[state] = true
            state.entered, state.entry_time = state.entered + 1, self.time
            for _, transition in ipairs(state.transitions) do
                if transition.kind == "timer" then
                    self:reserve({due = self.time + transition.delay, timer = transition})
                end
            end
            for _, actions in ipairs(state.entries) do
                self:perform(actions)
            end
            if state.kind == "final" then
                local parent = state.parent
                self:enqueue("done.state." .. parent.name, 0)
            end
        end
    end
function Chart:microstep(trigger, transitions)
    local exits, entries = {}, {}
    for _, selected in ipairs(transitions) do
        for state in pairs(selected.exits) do
            exits[state] = true
        end
    end
    for _, history in ipairs(self.document) do
        if history.kind == "history" and exits[history.parent] then
        end
    end
    for _, selected in ipairs(transitions) do
        if selected.transition.mode ~= "internal" then
        end
    end
    self:expand(entries)
    for i = #self.document, 1, -1 do
        local state = self.document[i]
        if exits[state] then
            for _, actions in ipairs(state.exits) do
                self:perform(actions)
            end
            self.active[state] = nil
        end
    end
    for _, selected in ipairs(transitions) do
        self:perform(selected.transition.actions)
        selected.transition.source.taken = selected.transition.source.taken + 1
    end
    self:enter(entries)
function Chart:macrostep(start, trigger)
    self.macrosteps = self.macrosteps + 1
    local count = 0
    local function perform(selected_trigger, transitions)
        if #transitions == 0 then
            return true
        end
        if count == 1000 then
            emit("LIVELOCK t=%d", self.time)
            self.halted = true
            return false
        end
        count, self.microsteps = count + 1, self.microsteps + 1
        self:microstep(selected_trigger, transitions)
        return true
    end
    self.data = 0
    if start then
        local entries = {[self.root.initial] = true}
        self:expand(entries)
        self:enter(entries)
    else
        if trigger.kind == "event" then
            self.data = trigger.event.value
        end
        if not perform(trigger, self:select_transitions(trigger)) then
            return
        end
    end
    while true do
        self.data = 0
        local always = {kind = "always"}
        local transitions = self:select_transitions(always)
        if #transitions > 0 then
            if not perform(always, transitions) then
                return
            end
        else
            if self.queue_head > #self.queue then
                return
            end
            local event = self.queue[self.queue_head]
            self.queue_head = self.queue_head + 1
            self.data = event.value
            local next_trigger = {kind = "event", event = event}
            if not perform(next_trigger, self:select_transitions(next_trigger)) then
                return
            end
        end
    end
function Chart:dump()
    emit("dump t=%d", self.time)
    for _, state in ipairs(self.document) do
        if self.active[state] then
            emit("%s%s", string.rep(" ", 2 * state.depth), state.name)
        end
    end
function Chart:script(text)
    if self.halted then
        return
    end
    local fields = words(text)
    if fields[1] == "event" then
        self:macrostep(
            false,
            {kind = "event", event = {name = fields[2], value = tonumber(fields[3]) or 0}}
        )
    elseif fields[1] == "tick" then

            self.time = reservation.due
            self:macrostep(
                false,
                reservation.timer and {kind = "timer", timer = reservation.timer} or
                    {kind = "event", event = reservation.event}
            )
    elseif fields[1] == "dump" then
        self:dump()
    end
local input = {}
for line in io.lines() do
    input[#input + 1] = line:gsub("\r$", "")
end
local position = 2
for _ = 1, tonumber(input[1]) do
    emit("chart %s", input[position]:sub(7))
    position = position + 1
    local definitions = {}
    while input[position] ~= "RUN" do
        definitions[#definitions + 1] = {text = input[position], line = position}
        position = position + 1
    end
    position = position + 1
    local chart = Chart.new()
    local valid = chart:define(definitions)
    if valid then
        chart:macrostep(true)
    end
    while input[position] ~= "END" do
        if valid then
            chart:script(input[position])
        end
        position = position + 1
    end
end
io.write(table.concat(output, "\n"), "\n")
