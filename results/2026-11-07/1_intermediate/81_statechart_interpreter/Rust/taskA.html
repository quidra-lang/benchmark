4#[derive(Clone)]
5struct Expr {
6 op: String,
7 text: String,
8 args: Vec<Expr>,
9}
10
11struct Expression<'a> {
12 tokens: Vec<String>,
13 at: usize,
14 variables: &'a BTreeMap<String, i64>,
15 states: &'a BTreeMap<String, usize>,
16}
17
18impl<'a> Expression<'a> {
19 fn new(
20 text: &str,
21 variables: &'a BTreeMap<String, i64>,
22 states: &'a BTreeMap<String, usize>,
23 ) -> Self {
24 let b = text.as_bytes();
25 let mut i = 0;
26 let mut tokens = Vec::new();
27 while i < b.len() {
28 if b[i] == b' ' {
29 i += 1;
30 continue;
31 }
32 let start = i;
33 if b[i].is_ascii_digit() {
34 while i < b.len() && b[i].is_ascii_digit() {
35 i += 1;
36 }
37 } else if b[i].is_ascii_alphabetic() || b[i] == b'_' {
38 while i < b.len() && (b[i].is_ascii_alphanumeric() || b[i] == b'_') {
39 i += 1;
40 }
41 } else {
42 i += 1;
43 if i < b.len()
44 && ["||", "&&", "==", "!=", "<=", ">="].contains(&&text[start..i + 1])
45 {
46 i += 1;
47 }
48 }
49 tokens.push(text[start..i].into());
50 }
51 Self {
52 tokens,
53 at: 0,
54 variables,
55 states,
56 }
57 }
58
59 fn parse(&mut self, level: usize) -> Result<Expr, String> {
60 if level == 5 {
61 let t = self.tokens[self.at].clone();
62 self.at += 1;
63 if t == "!" || t == "-" {
64 return Ok(Expr {
65 op: format!("unary{}", t),
66 text: String::new(),
67 args: vec![self.parse(5)?],
68 });
69 }
70 if t == "(" {
71 let e = self.parse(0)?;
72 self.at += 1;
73 return Ok(e);
74 }
75 if t.as_bytes()[0].is_ascii_digit() {
76 return Ok(Expr {
77 op: "number".into(),
78 text: t,
79 args: Vec::new(),
80 });
81 }
82 if t == "in" {
83 self.at += 1;
84 let name = self.tokens[self.at].clone();
85 self.at += 2;
86 if !self.states.contains_key(&name) {
87 return Err(format!("unknown state {}", name));
88 }
89 return Ok(Expr {
90 op: "in".into(),
91 text: name,
92 args: Vec::new(),
93 });
94 }
95 if t != "_data" && !self.variables.contains_key(&t) {
96 return Err(format!("unknown variable {}", t));
97 }
98 return Ok(Expr {
99 op: "variable".into(),
100 text: t,
101 args: Vec::new(),
102 });
103 }
104 let operators: &[&str] = match level {
105 0 => &["||"],
106 1 => &["&&"],
107 2 => &["==", "!=", "<", "<=", ">", ">="],
108 3 => &["+", "-"],
109 _ => &["*", "/", "%"],
110 };
111 let mut left = self.parse(level + 1)?;
112 while self.at < self.tokens.len() && operators.contains(&self.tokens[self.at].as_str()) {
113 let op = self.tokens[self.at].clone();
114 self.at += 1;
115 left = Expr {
116 op,
117 text: String::new(),
118 args: vec![left, self.parse(level + 1)?],
119 };
120 }
121 Ok(left)
122 }
123}

328 fn actions(&self, text: &str) -> Result<Vec<Action>, String> {
329 let mut result = Vec::new();
330 for raw in text.split("; ") {
331 let w: Vec<&str> = raw.split(' ').collect();
332 let mut a = Action {
333 kind: w[0].into(),
334 name: String::new(),
335 expr: None,
336 text: None,
337 delay: 0,
338 };
339 match w[0] {
340 "raise" => {
341 a.name = w[1].into();
342 if w.len() > 2 {
343 a.expr = Some(self.expression(&w[2..].join(" "))?);
344 }
345 }
346 "send" => {
347 a.name = w[1].into();
348 a.delay = w[2].parse().unwrap();
349 if w.len() > 3 {
350 a.expr = Some(self.expression(&w[3..].join(" "))?);
351 }
352 }
353 "cancel" => a.name = w[1].into(),
354 "log" => {
355 let rest = &raw[4..];
356 if rest.starts_with('"') {
357 let end = rest[1..].find('"').unwrap() + 1;
358 a.text = Some(rest[1..end].into());
359 if end + 1 < rest.len() {
360 a.expr = Some(self.expression(&rest[end + 2..])?);
361 }
362 } else {
363 a.expr = Some(self.expression(rest)?);
364 }
365 }
366 _ => {
367 if !self.variables.contains_key(w[0]) {
368 return Err(format!("unknown variable {}", w[0]));
369 }
370 a.kind = "assign".into();
371 a.name = w[0].into();
372 a.expr = Some(self.expression(&w[2..].join(" "))?);
373 }
374 }
375 result.push(a);
376 }
377 Ok(result)
378 }

380 fn definition(&mut self, d: &Definition) -> Result<(), String> {
381 let w: Vec<&str> = d.text.split(' ').collect();
382 if w[0] == "state" {
383 if let Some(i) = d.node {
384 if let Some(name) = &self.states[i].initial_name {
385 let child = *self
386 .ids
387 .get(name)
388 .ok_or_else(|| format!("unknown state {}", name))?;
389 if self.states[child].parent != i || self.states[child].kind == "history" {
390 return Err(format!("{} is not a child of {}", name, self.states[i].id));
391 }
392 }
393 }
394 }
395 if w[0] == "final" || w[0] == "history" {
396 let s = &self.states[d.node.unwrap()];
397 let p = &self.states[s.parent];
398 if p.kind != "state" || self.children(s.parent).is_empty() {
399 return Err(format!("{} needs a compound parent", s.id));
400 }
401 if let Some(name) = &s.default_name {
402 let target = *self
403 .ids
404 .get(name)
405 .ok_or_else(|| format!("unknown state {}", name))?;
406 if !self.descendant(target, s.parent) || self.states[target].kind == "history" {
407 return Err(format!("{} is not a descendant of {}", name, p.id));
408 }
409 }
410 }
411 if w[0] == "entry" || w[0] == "exit" {
412 let i = *self
413 .ids
414 .get(w[1])
415 .ok_or_else(|| format!("unknown state {}", w[1]))?;
416 if self.states[i].kind == "history" {
417 return Err(format!("{} cannot have actions", w[1]));
418 }
419 let a = self.actions(d.text.split_once(" / ").unwrap().1)?;
420 if w[0] == "entry" {
421 self.states[i].entries.push(a);
422 } else {
423 self.states[i].exits.push(a);
424 }
425 }
426 if w[0] == "on" {
427 let source = *self
428 .ids
429 .get(w[1])
430 .ok_or_else(|| format!("unknown state {}", w[1]))?;
431 if ["final", "history"].contains(&self.states[source].kind.as_str()) {
432 return Err(format!("{} cannot have transitions", w[1]));
433 }
434 let mut rest = &d.text[4 + w[1].len()..];
435 let trigger;
436 let mut delay = 0;
437 let mut descriptors = Vec::new();
438 if rest.starts_with("after ") {
439 rest = &rest[6..];
440 let word = rest.split(' ').next().unwrap();
441 delay = word.parse().unwrap();
442 rest = rest[word.len()..].trim_start_matches(' ');
443 trigger = "timer";
444 } else {
445 let word = rest.split(' ').next().unwrap();
446 rest = rest[word.len()..].trim_start_matches(' ');
447 trigger = if word == "always" { "always" } else { "event" };
448 if trigger == "event" {
449 descriptors = word.split('|').map(str::to_string).collect();
450 }
451 }
452 let mut guard = None;
453 if rest.starts_with('[') {
454 let end = rest.find(']').unwrap();
455 guard = Some(self.expression(&rest[1..end])?);
456 rest = rest[end + 1..].trim_start_matches(' ');
457 }
458 let mut target = None;
459 let mut mode = "internal";
460 if rest.starts_with("-> ") || rest.starts_with("~> ") {
461 mode = if rest.starts_with('-') {
462 "external"
463 } else {
464 "local"
465 };
466 rest = &rest[3..];
467 let name = rest.split(' ').next().unwrap();
468 target = Some(
469 *self
470 .ids
471 .get(name)
472 .ok_or_else(|| format!("unknown state {}", name))?,
473 );
474 rest = rest[name.len()..].trim_start_matches(' ');
475 }
476 let actions = if rest.starts_with("/ ") {
477 self.actions(&rest[2..])?
478 } else {
479 Vec::new()
480 };
481 self.states[source].transitions.push(self.transitions.len());
482 self.transitions.push(Transition {
483 source,
484 trigger: trigger.into(),
485 descriptors,
486 delay,
487 guard,
488 target,
489 mode: mode.into(),
490 actions,
491 });
492 }
493 Ok(())
494 }

496 fn define(&mut self, lines: &[(usize, &str)]) -> bool {
497 let mut errors = BTreeMap::new();
498 let mut definitions = Vec::new();
499 for &(line, text) in lines {
500 let w: Vec<&str> = text.split(' ').collect();
501 let mut node = None;
502 if ["state", "parallel", "final", "history"].contains(&w[0]) {
503 let parent_name = if w.len() > 2 && w[2] != "initial" {
504 Some(w[2])
505 } else {
506 None
507 };
508 let reason = if self.ids.contains_key(w[1]) {
509 Some(format!("duplicate state {}", w[1]))
510 } else if parent_name.is_some_and(|p| !self.ids.contains_key(p)) {
511 Some(format!("unknown state {}", parent_name.unwrap()))
512 } else if parent_name.is_some_and(|p| {
513 ["final", "history"].contains(&self.states[self.ids[p]].kind.as_str())
514 }) {
515 Some(format!("{} cannot have children", parent_name.unwrap()))
516 } else {
517 None
518 };
519 if let Some(reason) = reason {
520 errors.insert(line, reason);
521 } else {
522 let parent = parent_name.map_or(0, |p| self.ids[p]);
523 let mut s = State::new(w[1].into(), w[0].into(), parent);
524 if let Some(at) = w.iter().position(|s| *s == "initial") {
525 s.initial_name = Some(w[at + 1].into());
526 }
527 s.deep = w[0] == "history" && w[3] == "deep";
528 if w[0] == "history" && w.len() == 5 {
529 s.default_name = Some(w[4].into());
530 }
531 s.depth = self.states[parent].depth + 1;
532 let i = self.states.len();
533 node = Some(i);
534 self.ids.insert(s.id.clone(), i);
535 self.states.push(s);
536 self.states[parent].children.push(i);
537 }
538 } else if w[0] == "var" {
539 if self.variables.contains_key(w[1]) {
540 errors.insert(line, format!("duplicate variable {}", w[1]));
541 } else {
542 self.variables.insert(w[1].into(), w[2].parse().unwrap());
543 }
544 }
545 definitions.push(Definition {
546 line,
547 text: text.into(),
548 node,
549 });
550 }
551 for d in &definitions {
552 if !errors.contains_key(&d.line) {
553 if let Err(reason) = self.definition(d) {
554 errors.insert(d.line, reason);
555 }
556 }
557 }
558 if !errors.is_empty() {
559 for (line, reason) in errors {
560 self.output.push(format!("error line {}: {}", line, reason));
561 }
562 return false;
563 }
564 for i in 0..self.states.len() {
565 if self.compound(i) {
566 self.states[i].initial = self.states[i]
567 .initial_name
568 .as_ref()
569 .map_or_else(|| self.children(i)[0], |n| self.ids[n]);
570 }
571 if self.states[i].kind == "history" {
572 self.states[i].default = self.states[i]
573 .default_name
574 .as_ref()
575 .map_or(self.states[self.states[i].parent].initial, |n| self.ids[n]);
576 }
577 }

591 fn warnings(&mut self) {
592 let mut reached = BTreeSet::from([self.states[0].initial]);
593 loop {
594 let before = reached.len();
595 for i in reached.clone() {
596 if self.states[i].parent != 0 {
597 reached.insert(self.states[i].parent);
598 }
599 if self.compound(i) {
600 reached.insert(self.states[i].initial);
601 }
602 if self.parallel(i) {
603 reached.extend(self.children(i));
604 }
605 for &index in &self.states[i].transitions {
606 if let Some(t) = self.transitions[index].target {
607 reached.insert(if self.states[t].kind == "history" {
608 self.states[t].default
609 } else {
610 t
611 });
612 }
613 }
614 }
615 if before == reached.len() {
616 break;
617 }
618 }
619 for &i in &self.document {
620 if self.states[i].kind != "history" && !reached.contains(&i) {
621 self.output
622 .push(format!("warning: unreachable state {}", self.states[i].id));
623 }
624 }
625 }

627 fn evaluate(&self, e: &Expr) -> Result<i64, ()> {
628 if e.op == "number" {
629 return Ok(e.text.parse().unwrap());
630 }
631 if e.op == "variable" {
632 return Ok(if e.text == "_data" {
633 self.data
634 } else {
635 self.variables[&e.text]
636 });
637 }
638 if e.op == "in" {
639 return Ok(i64::from(self.active.contains(&self.ids[&e.text])));
640 }
641 let a = self.evaluate(&e.args[0])?;
642 if e.op == "unary!" {
643 return Ok(i64::from(a == 0));
644 }
645 if e.op == "unary-" {
646 return Ok(-a);
647 }
648 if e.op == "&&" && a == 0 {
649 return Ok(0);
650 }
651 if e.op == "||" && a != 0 {
652 return Ok(1);
653 }
654 let b = self.evaluate(&e.args[1])?;
655 if (e.op == "/" || e.op == "%") && b == 0 {
656 return Err(());
657 }
658 Ok(match e.op.as_str() {
659 "+" => a + b,
660 "-" => a - b,
661 "*" => a * b,
662 "/" => a / b,
663 "%" => a % b,
664 "==" => i64::from(a == b),
665 "!=" => i64::from(a != b),
666 "<" => i64::from(a < b),
667 "<=" => i64::from(a <= b),
668 ">" => i64::from(a > b),
669 ">=" => i64::from(a >= b),
670 _ => i64::from(b != 0),
671 })
672 }

684 fn run_actions(&mut self, actions: &[Action]) {
685 for a in actions {
686 let v = match a.expr.as_ref().map_or(Ok(0), |e| self.evaluate(e)) {
687 Ok(v) => v,
688 Err(()) => {
689 self.queue.push_back(Event::new("error.execution", 0));
690 break;
691 }
692 };
693 match a.kind.as_str() {
694 "assign" => {
695 self.variables.insert(a.name.clone(), v);
696 }
697 "raise" => self.queue.push_back(Event::new(a.name.clone(), v)),
698 "send" => self.reserve(None, Some(Event::new(a.name.clone(), v)), a.delay),
699 "cancel" => self
700 .reservations
701 .retain(|r| r.event.as_ref().is_none_or(|e| e.name != a.name)),
702 "log" => self.output.push(
703 a.text.clone().unwrap_or_default()
704 + &if a.expr.is_some() {
705 v.to_string()
706 } else {
707 String::new()
708 },
709 ),
710 _ => unreachable!(),
711 }
712 }

717 if self.states[i].kind == "history" {
718 self.states[i]

721 .unwrap_or_else(|| vec![self.states[i].default])
722 } else {
723 vec![i]
724 }

728 fn scope(&self, t: &Transition) -> Option<usize> {
729 t.target?;
730 let targets = self.effective(t);
731 if t.mode == "local"
732 && self.compound(t.source)
733 && targets.iter().all(|i| self.descendant(*i, t.source))
734 {
735 return Some(t.source);
736 }
737 let mut p = self.states[t.source].parent;
738 while !(self.compound(p) && targets.iter().all(|i| self.descendant(*i, p))) {
739 p = self.states[p].parent;
740 }
741 Some(p)
742 }

744 fn matches(&self, t: &Transition, trigger: &Trigger, index: usize) -> bool {
745 match trigger.kind {
746 "always" => t.trigger == "always",
747 "timer" => trigger.transition == Some(index),
748 _ => {
749 let name = &trigger.event.as_ref().unwrap().name;
750 t.trigger == "event"
751 && t.descriptors
752 .iter()
753 .any(|d| d == "*" || d == name || name.starts_with(&(d.clone() + ".")))
754 }
755 }

758 fn select(&mut self, trigger: &Trigger) -> Vec<Selected> {
759 let mut candidates = Vec::new();
760 let mut seen = BTreeSet::new();
761 for leaf in self.document.clone() {
762 if !self.active.contains(&leaf) || !self.atomic(leaf) {
763 continue;
764 }
765 let mut source = leaf;
766 let mut chosen = None;
767 while source != 0 && chosen.is_none() {
768 for index in self.states[source].transitions.clone() {
769 let t = &self.transitions[index];
770 if !self.matches(t, trigger, index) {
771 continue;
772 }
773 let enabled = match t
774 .guard
775 .as_ref()
776 .map_or(Ok(true), |e| self.evaluate(e).map(|v| v != 0))
777 {
778 Ok(v) => v,
779 Err(()) => {
780 self.queue.push_back(Event::new("error.execution", 0));
781 false
782 }
783 };
784 if enabled {
785 chosen = Some(index);
786 break;
787 }
788 }
789 source = self.states[source].parent;
790 }
791 if let Some(i) = chosen {
792 if seen.insert(i) {
793 candidates.push(i);
794 }
795 }
796 }
797 let mut occupied = BTreeSet::new();
798 let mut result = Vec::new();
799 for i in candidates {
800 let scope = self.scope(&self.transitions[i]);
801 let exits = scope.map_or(BTreeSet::new(), |p| {
802 self.active
803 .iter()
804 .copied()
805 .filter(|i| self.descendant(*i, p))
806 .collect()
807 });
808 if exits.is_disjoint(&occupied) {
809 occupied.extend(&exits);
810 result.push(Selected {
811 transition: i,
812 scope,
813 exits,
814 });
815 }
816 }
817 result
818 }

820 fn expand(&self, entries: &mut BTreeSet<usize>) {
821 loop {
822 let before = entries.len();
823 for i in entries.clone() {
824 if self.parallel(i) {
825 entries.extend(self.children(i));
826 }
827 if self.compound(i) && self.children(i).iter().all(|c| !entries.contains(c)) {
828 entries.insert(self.states[i].initial);
829 }
830 }
831 if before == entries.len() {
832 return;
833 }
834 }
835 }

837 fn complete(&self, i: usize) -> bool {
838 if self.compound(i) {
839 self.children(i)
840 .iter()
841 .any(|c| self.states[*c].kind == "final" && self.active.contains(c))
842 } else {
843 self.parallel(i) && self.children(i).iter().all(|c| self.complete(*c))
844 }
845 }
846
847 fn enter(&mut self, entries: &BTreeSet<usize>) {
848 for i in self.document.clone() {
849 if !entries.contains(&i) {
850 continue;
851 }
852 self.active.insert(i);
853 self.states[i].entered += 1;
854 self.states[i].entry_time = self.time;
855 for index in self.states[i].transitions.clone() {
856 if self.transitions[index].trigger == "timer" {
857 self.reserve(Some(index), None, self.transitions[index].delay);
858 }
859 }
860 for a in self.states[i].entries.clone() {
861 self.run_actions(&a);
862 }
863 if self.states[i].kind == "final" {
864 let p = self.states[i].parent;
865 self.queue
866 .push_back(Event::new(format!("done.state.{}", self.states[p].id), 0));
867 let g = self.states[p].parent;
868 if self.parallel(g) && self.children(g).iter().all(|c| self.complete(*c)) {
869 self.queue
870 .push_back(Event::new(format!("done.state.{}", self.states[g].id), 0));
871 }
872 }
873 }
874 }

878 "-".into()
879 } else {
880 values

882 .map(|i| self.states[*i].id.as_str())

884 .join(",")

888 fn display(&self, trigger: &Trigger) -> String {
889 match trigger.kind {
890 "always" => "always".into(),
891 "timer" => {
892 let t = &self.transitions[trigger.transition.unwrap()];
893 format!("after({})@{}", t.delay, self.states[t.source].id)
894 }
895 _ => {
896 let e = trigger.event.as_ref().unwrap();
897 e.name.clone()
898 + &if e.value == 0 {
899 String::new()
900 } else {
901 format!("={}", e.value)
902 }
903 }
904 }

907 fn micro(&mut self, selected: &[Selected], trigger: &Trigger) {
908 let exits: BTreeSet<usize> = selected
909 .iter()
910 .flat_map(|s| s.exits.iter().copied())
911 .collect();
912 for i in self.document.clone() {
913 if self.states[i].kind == "history" && exits.contains(&self.states[i].parent) {
914 let p = self.states[i].parent;
915 let history = self
916 .document
917 .iter()
918 .copied()
919 .filter(|c| {
920 self.active.contains(c)
921 && if self.states[i].deep {
922 self.atomic(*c) && self.descendant(*c, p)
923 } else {
924 self.states[*c].parent == p && self.states[*c].kind != "history"
925 }
926 })
927 .collect();
928 self.states[i].history = Some(history);
929 }
930 }
931 let mut entries = BTreeSet::new();
932 for choice in selected {
933 if let Some(scope) = choice.scope {
934 for target in self.effective(&self.transitions[choice.transition]) {
935 let mut i = target;
936 while i != scope {
937 entries.insert(i);
938 i = self.states[i].parent;
939 }
940 }
941 }
942 }
943 self.expand(&mut entries);
944 if self.trace {
945 let sources = selected
946 .iter()
947 .map(|s| self.transitions[s.transition].source)
948 .collect::<Vec<_>>();
949 let exit_list = self
950 .document
951 .iter()
952 .rev()
953 .copied()
954 .filter(|i| exits.contains(i))
955 .collect::<Vec<_>>();
956 let entry_list = self
957 .document
958 .iter()
959 .copied()
960 .filter(|i| entries.contains(i))
961 .collect::<Vec<_>>();
962 self.output.push(format!(
963 "[t={}] {} ({}): exit {}; enter {}",
964 self.time,
965 self.display(trigger),
966 self.list(&sources),
967 self.list(&exit_list),
968 self.list(&entry_list)
969 ));
970 }
971 for i in self.document.clone().into_iter().rev() {
972 if exits.contains(&i) {
973 for a in self.states[i].exits.clone() {
974 self.run_actions(&a);
975 }
976 self.active.remove(&i);
977 self.states[i].dwell += self.time - self.states[i].entry_time;
978 let transitions = &self.transitions;
979 self.reservations
980 .retain(|r| r.transition.is_none_or(|t| transitions[t].source != i));
981 }
982 }
983 for choice in selected {
984 let t = self.transitions[choice.transition].clone();
985 self.run_actions(&t.actions);
986 self.states[t.source].taken += 1;
987 }
988 self.enter(&entries);
989 }

991 fn perform(&mut self, selected: Vec<Selected>, trigger: &Trigger, count: &mut usize) -> bool {
992 if selected.is_empty() {
993 return true;
994 }
995 if *count == 1000 {
996 self.output.push(format!("LIVELOCK t={}", self.time));
997 self.halted = true;
998 return false;
999 }
1000 *count += 1;
1001 self.micros += 1;
1002 self.micro(&selected, trigger);
1003 true
1004 }
1005
1006 fn macrostep(&mut self, trigger: Trigger) {
1007 self.macros += 1;
1008 let mut count = 0;
1009 if trigger.kind == "start" {
1010 self.data = 0;
1011 let mut entries = BTreeSet::from([self.states[0].initial]);
1012 self.expand(&mut entries);
1013 self.enter(&entries);
1014 } else {
1015 self.data = trigger.event.as_ref().map_or(0, |e| e.value);
1016 let selected = self.select(&trigger);
1017 if !self.perform(selected, &trigger, &mut count) {
1018 return;
1019 }
1020 }
1021 loop {
1022 self.data = 0;
1023 let always = Trigger::kind("always");
1024 let selected = self.select(&always);
1025 if !selected.is_empty() {
1026 if !self.perform(selected, &always, &mut count) {
1027 return;
1028 }
1029 continue;
1030 }
1031 let Some(event) = self.queue.pop_front() else {
1032 return;
1033 };
1034 self.data = event.value;
1035 let next = Trigger::event(event);
1036 let selected = self.select(&next);
1037 if !self.perform(selected, &next, &mut count) {
1038 return;
1039 }
1040 }
1041 }

1043 fn script(&mut self, line: &str) {
1044 if self.halted {
1045 return;
1046 }
1047 let w: Vec<&str> = line.split(' ').collect();
1048 match w[0] {
1049 "event" => self.macrostep(Trigger::event(Event::new(
1050 w[1],
1051 w.get(2).map_or(0, |s| s.parse().unwrap()),
1052 ))),
1053 "tick" => {
1054 let finish = self.time + w[1].parse::<i64>().unwrap();
1055 while !self.halted {
1056 let next = self
1057 .reservations
1058 .iter()
1059 .enumerate()
1060 .filter(|(_, r)| r.due <= finish)
1061 .min_by_key(|(_, r)| (r.due, r.creation))
1062 .map(|(i, _)| i);
1063 let Some(index) = next else {
1064 break;
1065 };
1066 let r = self.reservations.remove(index);
1067 self.time = r.due;
1068 self.macrostep(if let Some(t) = r.transition {
1069 Trigger {
1070 kind: "timer",
1071 event: None,
1072 transition: Some(t),
1073 }
1074 } else {
1075 Trigger::event(r.event.unwrap())
1076 });
1077 }
1078 if !self.halted {
1079 self.time = finish;
1080 }
1081 }
1082 "set" => {
1083 if self.variables.contains_key(w[1]) {
1084 self.variables.insert(w[1].into(), w[2].parse().unwrap());
1085 } else {
1086 self.output
1087 .push(format!("error: unknown variable {}", w[1]));
1088 }
1089 }
1090 "trace" => self.trace = w[1] == "on",
1091 "eval" => {
1092 let text = &line[5..];
1093 match self.expression(text) {
1094 Err(reason) => self.output.push(format!("error: {}", reason)),
1095 Ok(e) => {
1096 self.data = 0;
1097 let result = self.evaluate(&e);
1098 self.output.push(match result {
1099 Ok(value) => format!("{} -> {}", text, value),
1100 Err(()) => "error: division by zero".into(),
1101 });
1102 }
1103 }
1104 }
1105 "dump" => {
1106 self.output.push(format!("dump t={}", self.time));
1107 for &i in &self.document {
1108 if self.active.contains(&i) {
1109 self.output
1110 .push(" ".repeat(self.states[i].depth) + &self.states[i].id);
1111 }
1112 }
1113 for (name, value) in &self.variables {
1114 self.output.push(format!(" var {} = {}", name, value));
1115 }
1116 for &i in &self.document {
1117 if let Some(h) = &self.states[i].history {
1118 self.output.push(format!(
1119 " history {} = {}",
1120 self.states[i].id,
1121 self.list(h)
1122 ));
1123 }
1124 }
1125 let mut reservations = self.reservations.clone();
1126 reservations.sort_by_key(|r| (r.due, r.creation));
1127 for r in reservations {
1128 if let Some(i) = r.transition {
1129 let t = &self.transitions[i];
1130 self.output.push(format!(
1131 " timer {} after({})@{}",
1132 r.due, t.delay, self.states[t.source].id
1133 ));
1134 } else {
1135 let e = r.event.unwrap();
1136 self.output
1137 .push(format!(" send {} {} {}", r.due, e.name, e.value));
1138 }
1139 }
1140 }
1141 _ => unreachable!(),
1142 }
1143 }

1145 fn statistics(&mut self) {
1146 self.output.push(format!(
1147 "end t={} macrosteps={} microsteps={}",
1148 self.time, self.macros, self.micros
1149 ));
1150 for &i in &self.document {
1151 let s = &self.states[i];
1152 if s.kind != "history" {
1153 let dwell = s.dwell
1154 + if self.active.contains(&i) {
1155 self.time - s.entry_time
1156 } else {
1157 0
1158 };
1159 self.output.push(format!(
1160 " {} entered={} taken={} time={}",
1161 s.id, s.entered, s.taken, dwell
1162 ));
1163 }
1164 }
