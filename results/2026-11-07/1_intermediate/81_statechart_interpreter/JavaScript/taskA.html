4function expression(text) {
5 const tokens = text.match(/[0-9]+|[A-Za-z_][A-Za-z0-9_]*|\|\||&&|==|!=|<=|>=|[<>+*/%!()-]/g);
6 const references = [];
7 let at = 0;
8 function primary() {
9 const token = tokens[at++];
10 if (token === "-" || token === "!")
11 return { kind: "unary", operator: token, value: primary() };
12 if (token === "(") {
13 const value = binary(0);
14 at++;
15 return value;
16 }
17 if (/^[0-9]/.test(token)) return { kind: "number", value: Number(token) };
18 if (token === "in") {
19 at++;
20 const name = tokens[at++];
21 at++;
22 references.push({ kind: "state", name });
23 return { kind: "state", name };
24 }
25 if (token !== "_data") references.push({ kind: "variable", name: token });
26 return { kind: "variable", name: token };
27 }
28 function binary(level) {
29 if (level === 5) return primary();
30 const operators = [
31 ["||"],
32 ["&&"],
33 ["==", "!=", "<", "<=", ">", ">="],
34 ["+", "-"],
35 ["*", "/", "%"],
36 ][level];
37 let left = binary(level + 1);
38 while (operators.includes(tokens[at])) {
39 const operator = tokens[at++],
40 right = binary(level + 1);
41 left = { kind: "binary", operator, left, right };
42 }
43 return left;
44 }
45 return { node: binary(0), references };
46}
47function actions(text) {
48 if (text === "") return [];
49 return text.split("; ").map((part) => {
50 const space = part.indexOf(" "),
51 first = part.slice(0, space),
52 rest = part.slice(space + 1);
53 if (first === "log") {
54 if (rest.startsWith('"')) {
55 const close = rest.indexOf('"', 1);
56 return {
57 kind: "log",
58 text: rest.slice(1, close),
59 expression:
60 close + 1 === rest.length ? null : expression(rest.slice(close + 2)),
61 };
62 }
63 return { kind: "log", text: "", expression: expression(rest) };
64 }
65 if (first === "cancel") return { kind: "cancel", event: rest, expression: null };
66 if (first === "raise" || first === "send") {
67 const words = rest.split(" "),
68 event = words.shift(),
69 delay = first === "send" ? Number(words.shift()) : 0;
70 return {
71 kind: first,
72 event,
73 delay,
74 expression: words.length ? expression(words.join(" ")) : null,
75 };
76 }
77 return { kind: "assign", name: first, expression: expression(rest.slice(2)) };
78 });
79}

83 ...(action.expression === null ? [] : action.expression.references),
84 ]);

86function transitionParts(text) {
87 const first = text.indexOf(" ");
88 let trigger, rest;
89 if (text.startsWith("after ")) {
90 const end = text.indexOf(" ", 6);
91 trigger = end < 0 ? text : text.slice(0, end);
92 rest = end < 0 ? "" : text.slice(end);
93 } else {
94 trigger = first < 0 ? text : text.slice(0, first);
95 rest = first < 0 ? "" : text.slice(first);
96 }
97 let guard = null,
98 target = null,
99 kind = "internal";
100 if (rest.startsWith(" [")) {
101 const end = rest.indexOf("]");
102 guard = expression(rest.slice(2, end));
103 rest = rest.slice(end + 1);
104 }
105 if (rest.startsWith(" -> ") || rest.startsWith(" ~> ")) {
106 kind = rest[1] === "-" ? "external" : "local";
107 rest = rest.slice(4);
108 const end = rest.indexOf(" ");
109 target = end < 0 ? rest : rest.slice(0, end);
110 rest = end < 0 ? "" : rest.slice(end);
111 }
112 return { trigger, guard, target, kind, actions: actions(rest === "" ? "" : rest.slice(3)) };
113}

167function build(lines) {
168 const root = stateNode("", "state", null, 0, null, "", null),
169 states = new Map(),
170 variables = new Map(),
171 errors = new Map(),
172 definitions = [];
173 for (const source of lines) {
174 const words = source.text.split(" "),
175 kind = words[0],
176 name = words[1];
177 const definition = { ...source, kind, name, state: null, actions: null, transition: null };
178 definitions.push(definition);
179 if (["state", "parallel", "final", "history"].includes(kind)) {
180 let parentName = null,
181 selected = null,
182 mode = "",
183 defaultTarget = null;
184 if (kind === "state") {
185 if (words[2] === "initial") selected = words[3];
186 else {
187 parentName = words[2] ?? null;
188 if (words[3] === "initial") selected = words[4];
189 }
190 } else {
191 parentName = words[2] ?? null;
192 if (kind === "history") {
193 mode = words[3];
194 defaultTarget = words[4] ?? null;
195 }
196 }
197 if (states.has(name)) errors.set(source.line, "duplicate state " + name);
198 else if (parentName !== null && !states.has(parentName))
199 errors.set(source.line, "unknown state " + parentName);
200 else if (
201 parentName !== null &&
202 ["final", "history"].includes(states.get(parentName).kind)
203 )
204 errors.set(source.line, parentName + " cannot have children");
205 else {
206 const parent = parentName === null ? root : states.get(parentName),
207 state = stateNode(
208 name,
209 kind,
210 parent,
211 source.line,
212 selected,
213 mode,
214 defaultTarget,
215 );
216 parent.children.push(state);
217 states.set(name, state);
218 definition.state = state;
219 }
220 } else if (kind === "var") {
221 if (variables.has(name)) errors.set(source.line, "duplicate variable " + name);
222 else variables.set(name, Number(words[2]));
223 }

225 function checkReferences(references) {
226 for (const reference of references)
227 if (!(reference.kind === "state" ? states : variables).has(reference.name))
228 return "unknown " + reference.kind + " " + reference.name;
229 return null;
230 }
231 for (const definition of definitions) {
232 if (errors.has(definition.line)) continue;
233 const { kind, name, state } = definition;
234 let error = null;
235 if (kind === "state" && state.initial !== null) {
236 const child = states.get(state.initial);
237 if (child === undefined) error = "unknown state " + state.initial;
238 else if (child.parent !== state || child.kind === "history")
239 error = child.name + " is not a child of " + name;
240 } else if (kind === "final" || kind === "history") {
241 if (state.parent.kind !== "state" || !compound(state.parent))
242 error = name + " needs a compound parent";
243 else if (kind === "history" && state.fallback !== null) {
244 const target = states.get(state.fallback);
245 if (target === undefined) error = "unknown state " + state.fallback;
246 else if (!descendant(target, state.parent) || target.kind === "history")
247 error = target.name + " is not a descendant of " + state.parent.name;
248 }
249 } else if (kind === "entry" || kind === "exit" || kind === "on") {
250 const source = states.get(name);
251 if (source === undefined) error = "unknown state " + name;
252 else if (source.kind === "history" && kind !== "on")
253 error = name + " cannot have actions";
254 else if (kind === "on" && ["final", "history"].includes(source.kind))
255 error = name + " cannot have transitions";
256 else if (kind === "on") {
257 const part = transitionParts(definition.text.slice(("on " + name + " ").length));
258 error = checkReferences([
259 ...(part.guard === null ? [] : part.guard.references),
260 ...(part.target === null ? [] : [{ kind: "state", name: part.target }]),
261 ...actionReferences(part.actions),
262 ]);
263 if (error === null)
264 source.transitions.push({
265 source,
266 kind: part.kind,
267 target: part.target === null ? null : states.get(part.target),
268 guard: part.guard,
269 type:
270 part.trigger === "always"
271 ? "always"
272 : part.trigger.startsWith("after ")
273 ? "timer"
274 : "event",
275 delay: part.trigger.startsWith("after ")
276 ? Number(part.trigger.slice(6))
277 : 0,
278 descriptors: part.trigger.split("|"),
279 actions: part.actions,
280 });
281 } else {
282 const list = actions(definition.text.slice(definition.text.indexOf(" / ") + 3));
283 error = checkReferences(actionReferences(list));
284 if (error === null) (kind === "entry" ? source.entries : source.exits).push(list);
285 }
286 }
287 if (error !== null) errors.set(definition.line, error);
288 }
289 return { root, states, variables, errors, order: document(root), checkReferences };
290}

294function warnings(chart, output) {
295 const reachable = new Set([initial(chart.root, chart.states)]);
296 let changed = true;
297 while (changed) {
298 changed = false;
299 function add(state) {
300 if (!reachable.has(state)) {
301 reachable.add(state);
302 changed = true;
303 }
304 }
305 for (const state of [...reachable]) {
306 if (state.parent !== chart.root) add(state.parent);
307 if (compound(state)) add(initial(state, chart.states));
308 if (parallel(state)) for (const child of childStates(state)) add(child);
309 for (const transition of state.transitions)
310 if (transition.target !== null)
311 add(
312 transition.target.kind === "history"
313 ? fallback(transition.target, chart.states)
314 : transition.target,
315 );
316 }
317 }
318 for (const state of chart.order)
319 if (state.kind !== "history" && !reachable.has(state))
320 output.push("warning: unreachable state " + state.name);
321}

338 evaluate(expression) {
339 return this.value(expression.node);
340 }
341 value(node) {
342 if (node.kind === "number") return node.value;
343 if (node.kind === "variable")
344 return node.name === "_data" ? this.data : this.variables.get(node.name);
345 if (node.kind === "state")
346 return Number(this.configuration.has(this.chart.states.get(node.name)));
347 if (node.kind === "unary") {
348 const value = this.value(node.value);
349 return node.operator === "-" ? -value : Number(value === 0);
350 }
351 const a = this.value(node.left);
352 if (node.operator === "&&" && a === 0) return 0;
353 if (node.operator === "||" && a !== 0) return 1;
354 const b = this.value(node.right);
355 if (["/", "%"].includes(node.operator) && b === 0) throw new EvaluationError();
356 switch (node.operator) {
357 case "+":
358 return a + b;
359 case "-":
360 return a - b;
361 case "*":
362 return a * b;
363 case "/":
364 return Math.trunc(a / b);
365 case "%":
366 return a % b;
367 case "==":
368 return Number(a === b);
369 case "!=":
370 return Number(a !== b);
371 case "<":
372 return Number(a < b);
373 case "<=":
374 return Number(a <= b);
375 case ">":
376 return Number(a > b);
377 case ">=":
378 return Number(a >= b);
379 default:
380 return Number(b !== 0);
381 }

383 enqueue(name, value = 0) {
384 this.queue.push({ name, value });
385 }
386 reserve(reservation) {
387 this.reservations.push({ ...reservation, serial: ++this.serial });
388 }
389 actionList(list) {
390 for (const action of list) {
391 try {
392 const value = action.expression === null ? 0 : this.evaluate(action.expression);
393 switch (action.kind) {
394 case "assign":
395 this.variables.set(action.name, value);
396 break;
397 case "raise":
398 this.enqueue(action.event, value);
399 break;
400 case "send":
401 this.reserve({
402 kind: "send",
403 due: this.time + action.delay,
404 event: { name: action.event, value },
405 });
406 break;
407 case "cancel":
408 this.reservations = this.reservations.filter(
409 (item) => item.kind !== "send" || item.event.name !== action.event,
410 );
411 break;
412 case "log":
413 this.output.push(
414 action.text + (action.expression === null ? "" : String(value)),
415 );
416 break;
417 }
418 } catch (error) {
419 if (!(error instanceof EvaluationError)) throw error;
420 this.enqueue("error.execution");
421 break;
422 }
423 }

425 effective(transition) {
426 const target = transition.target;
427 return target.kind === "history"
428 ? (target.history ?? [fallback(target, this.chart.states)])
429 : [target];
430 }
431 choice(transition) {
432 if (transition.kind === "internal") return { transition, scope: null, exits: new Set() };
433 const targets = this.effective(transition);
434 let scope =
435 transition.kind === "local" &&
436 compound(transition.source) &&
437 targets.every((target) => descendant(target, transition.source))
438 ? transition.source
439 : transition.source.parent;
440 while (!compound(scope) || !targets.every((target) => descendant(target, scope)))
441 scope = scope.parent;
442 return {
443 transition,
444 scope,
445 exits: new Set(
446 this.chart.order.filter(
447 (state) => this.configuration.has(state) && descendant(state, scope),
448 ),
449 ),
450 };
451 }
452 matches(transition, trigger) {
453 if (trigger.kind === "always") return transition.type === "always";
454 if (trigger.kind === "timer") return transition === trigger.transition;
455 return (
456 transition.type === "event" &&
457 transition.descriptors.some(
458 (descriptor) =>
459 descriptor === "*" ||
460 descriptor === trigger.event.name ||
461 trigger.event.name.startsWith(descriptor + "."),
462 )
463 );
464 }
465 select(trigger) {
466 const candidates = [],
467 seen = new Set();
468 for (const leaf of this.chart.order)
469 if (atomic(leaf) && this.configuration.has(leaf)) {
470 let found = null;
471 for (
472 let state = leaf;
473 state !== this.chart.root && found === null;
474 state = state.parent
475 ) {
476 for (const transition of state.transitions)
477 if (this.matches(transition, trigger)) {
478 let enabled = true;
479 try {
480 enabled =
481 transition.guard === null ||
482 this.evaluate(transition.guard) !== 0;
483 } catch (error) {
484 if (!(error instanceof EvaluationError)) throw error;
485 this.enqueue("error.execution");
486 enabled = false;
487 }
488 if (enabled) {
489 found = transition;
490 break;
491 }
492 }
493 }
494 if (found !== null && !seen.has(found)) {
495 seen.add(found);
496 candidates.push(this.choice(found));
497 }
498 }
499 const chosen = [],
500 exited = new Set();
501 for (const choice of candidates)
502 if (![...choice.exits].some((state) => exited.has(state))) {
503 chosen.push(choice);
504 for (const state of choice.exits) exited.add(state);
505 }
506 return chosen;
507 }
508 closeEntry(set) {
509 let changed = true;
510 while (changed) {
511 changed = false;
512 for (const state of [...set]) {
513 const children = parallel(state)
514 ? childStates(state)
515 : compound(state) && !childStates(state).some((child) => set.has(child))
516 ? [initial(state, this.chart.states)]
517 : [];
518 for (const child of children)
519 if (!set.has(child)) {
520 set.add(child);
521 changed = true;
522 }
523 }
524 }
525 return this.chart.order.filter((state) => set.has(state));
526 }
527 complete(state) {
528 if (atomic(state)) return false;
529 if (compound(state))
530 return childStates(state).some(
531 (child) => child.kind === "final" && this.configuration.has(child),
532 );
533 return parallel(state) && childStates(state).every((child) => this.complete(child));
534 }
535 enter(states) {
536 for (const state of states) {
537 this.configuration.add(state);
538 state.entered++;
539 state.enteredAt = this.time;
540 for (const transition of state.transitions)
541 if (transition.type === "timer")
542 this.reserve({ kind: "timer", due: this.time + transition.delay, transition });
543 for (const list of state.entries) this.actionList(list);
544 if (state.kind === "final") {
545 this.enqueue("done.state." + state.parent.name);
546 const grandparent = state.parent.parent;
547 if (
548 parallel(grandparent) &&
549 childStates(grandparent).every((child) => this.complete(child))
550 )
551 this.enqueue("done.state." + grandparent.name);
552 }
553 }
554 }
555 display(trigger) {
556 return trigger.kind === "always"
557 ? "always"
558 : trigger.kind === "timer"
559 ? `after(${trigger.transition.delay})@${trigger.transition.source.name}`
560 : trigger.event.name + (trigger.event.value === 0 ? "" : "=" + trigger.event.value);

562 microstep(chosen, trigger) {
563 const exits = new Set(chosen.flatMap((choice) => [...choice.exits]));
564 for (const history of this.chart.order)
565 if (history.kind === "history" && exits.has(history.parent))
566 history.history = this.chart.order.filter(
567 (state) =>
568 this.configuration.has(state) &&
569 (history.mode === "deep"
570 ? atomic(state) && descendant(state, history.parent)
571 : state.parent === history.parent && state.kind !== "history"),
572 );
573 const set = new Set();
574 for (const choice of chosen)
575 if (choice.transition.kind !== "internal")
576 for (const target of this.effective(choice.transition))
577 for (let state = target; state !== choice.scope; state = state.parent)
578 set.add(state);
579 const entries = this.closeEntry(set),
580 leaving = this.chart.order.filter((state) => exits.has(state)).reverse();
581 if (this.trace)
582 this.output.push(
583 `[t=${this.time}] ${this.display(trigger)} (${names(chosen.map((choice) => choice.transition.source))}): exit ${names(leaving)}; enter ${names(entries)}`,
584 );
585 for (const state of leaving) {
586 for (const list of state.exits) this.actionList(list);
587 this.configuration.delete(state);
588 state.dwell += this.time - state.enteredAt;
589 this.reservations = this.reservations.filter(
590 (item) => item.kind !== "timer" || item.transition.source !== state,
591 );
592 }
593 for (const choice of chosen) {
594 this.actionList(choice.transition.actions);
595 choice.transition.source.taken++;
596 }
597 this.enter(entries);
598 }
599 macrostep(trigger) {
600 this.macrosteps++;
601 let count = 0;
602 const perform = (chosen, cause) => {
603 if (count === 1000) {
604 this.output.push("LIVELOCK t=" + this.time);
605 this.halted = true;
606 return;
607 }
608 count++;
609 this.microsteps++;
610 this.microstep(chosen, cause);
611 };
612 if (trigger.kind === "start") {
613 this.data = 0;
614 this.enter(this.closeEntry(new Set([initial(this.chart.root, this.chart.states)])));
615 } else {
616 this.data = trigger.kind === "timer" ? 0 : trigger.event.value;
617 const chosen = this.select(trigger);
618 if (chosen.length) perform(chosen, trigger);
619 }
620 while (!this.halted) {
621 this.data = 0;
622 const always = { kind: "always" },
623 chosen = this.select(always);
624 if (chosen.length) {
625 perform(chosen, always);
626 continue;
627 }
628 if (!this.queue.length) break;
629 const event = this.queue.shift(),
630 cause = { kind: "event", event };
631 this.data = event.value;
632 const selected = this.select(cause);
633 if (selected.length) perform(selected, cause);
634 }
635 }
636 sortedReservations() {

638 }

641 for (const item of this.reservations) {

651 script(line) {
652 if (this.halted) return;
653 const words = line.split(" "),
654 command = words[0];
655 if (command === "event")
656 this.macrostep({
657 kind: "event",
658 event: { name: words[1], value: words.length === 3 ? Number(words[2]) : 0 },
659 });
660 else if (command === "tick") {
661 const finish = this.time + Number(words[1]);
662 while (!this.halted) {
663 const first = this.nextReservation();
664 if (first === undefined || first.due > finish) break;
665 this.reservations.splice(this.reservations.indexOf(first), 1);
666 this.time = first.due;
667 this.macrostep(
668 first.kind === "timer"
669 ? { kind: "timer", transition: first.transition }
670 : { kind: "event", event: first.event },
671 );
672 }
673 if (!this.halted) this.time = finish;
674 } else if (command === "set") {
675 if (!this.variables.has(words[1]))
676 this.output.push("error: unknown variable " + words[1]);
677 else this.variables.set(words[1], Number(words[2]));
678 } else if (command === "trace") this.trace = words[1] === "on";
679 else if (command === "eval") {
680 const text = line.slice(5),
681 parsed = expression(text),
682 error = this.chart.checkReferences(parsed.references);
683 if (error !== null) this.output.push("error: " + error);
684 else {
685 this.data = 0;
686 try {
687 this.output.push(text + " -> " + this.evaluate(parsed));
688 } catch (error) {
689 if (!(error instanceof EvaluationError)) throw error;
690 this.output.push("error: division by zero");
691 }
692 }
693 } else if (command === "dump") {
694 this.output.push("dump t=" + this.time);
695 for (const state of this.chart.order)
696 if (this.configuration.has(state))
697 this.output.push(" ".repeat(2 * state.depth) + state.name);
698 for (const name of [...this.variables.keys()].sort())
699 this.output.push(` var ${name} = ${this.variables.get(name)}`);
700 for (const state of this.chart.order)
701 if (state.kind === "history" && state.history !== null)
702 this.output.push(` history ${state.name} = ${names(state.history)}`);
703 for (const reservation of this.sortedReservations())
704 this.output.push(
705 reservation.kind === "timer"
706 ? ` timer ${reservation.due} after(${reservation.transition.delay})@${reservation.transition.source.name}`
707 : ` send ${reservation.due} ${reservation.event.name} ${reservation.event.value}`,
708 );
709 }
710 }
711 statistics() {
712 this.output.push(
713 `end t=${this.time} macrosteps=${this.macrosteps} microsteps=${this.microsteps}`,
714 );
715 for (const state of this.chart.order)
716 if (state.kind !== "history")
717 this.output.push(
718 ` ${state.name} entered=${state.entered} taken=${state.taken} time=${state.dwell + (this.configuration.has(state) ? this.time - state.enteredAt : 0)}`,
719 );
