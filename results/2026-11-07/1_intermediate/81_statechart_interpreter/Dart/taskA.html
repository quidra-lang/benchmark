33class ExpressionParser {
34 final List<String> tokens;
35 final references = <Reference>[];
36 int at = 0;
37 ExpressionParser(String text) : tokens = RegExp(r'[0-9]+|[A-Za-z_][A-Za-z0-9_]*|\|\||&&|==|!=|<=|>=|[<>+*/%!()-]')
38 .allMatches(text).map((match) => match[0]!).toList();
39 Node primary() {
40 final token = tokens[at++];
41 if (token == '-' || token == '!') return UnaryNode(token, primary());
42 if (token == '(') {
43 final value = binary(0);
44 at++;
45 return value;
46 }
47 if (token.codeUnitAt(0) >= 48 && token.codeUnitAt(0) <= 57) return NumberNode(int.parse(token));
48 if (token == 'in') {
49 at++;
50 final name = tokens[at++];
51 at++;
52 references.add(Reference('state', name));
53 return NameNode('state', name);
54 }
55 if (token != '_data') references.add(Reference('variable', token));
56 return NameNode('variable', token);
57 }
58 Node binary(int level) {
59 if (level == 5) return primary();
60 final operators = [['||'], ['&&'], ['==', '!=', '<', '<=', '>', '>='], ['+', '-'], ['*', '/', '%']][level];
61 Node left = binary(level + 1);
62 while (at < tokens.length && operators.contains(tokens[at])) {
63 final operator = tokens[at++], right = binary(level + 1);
64 left = BinaryNode(operator, left, right);
65 }
66 return left;
67 }
68 Expression parse() => Expression(binary(0), references);
69}
70Expression expression(String text) => ExpressionParser(text).parse();

72class Action {
73 final String kind, name, event, text;
74 final int delay;
75 final Expression? expression;

78List<Action> actions(String text) {
79 if (text.isEmpty) return [];
80 return text.split('; ').map((part) {
81 final space = part.indexOf(' '), first = part.substring(0, space), rest = part.substring(space + 1);
82 if (first == 'log') {
83 if (rest.startsWith('"')) {
84 final close = rest.indexOf('"', 1);
85 return Action('log', text: rest.substring(1, close),
86 expression: close + 1 == rest.length ? null : expression(rest.substring(close + 2)));
87 }
88 return Action('log', expression: expression(rest));
89 }
90 if (first == 'cancel') return Action('cancel', event: rest);
91 if (first == 'raise' || first == 'send') {
92 final words = rest.split(' '), event = words.removeAt(0);
93 final delay = first == 'send' ? int.parse(words.removeAt(0)) : 0;
94 return Action(first, event: event, delay: delay,
95 expression: words.isEmpty ? null : expression(words.join(' ')));
96 }
97 return Action('assign', name: first, expression: expression(rest.substring(2)));
98 }).toList();
99}
100List<Reference> actionReferences(List<Action> list) {
101 final references = <Reference>[];
102 for (final action in list) {
103 if (action.kind == 'assign') references.add(Reference('variable', action.name));
104 references.addAll(action.expression?.references ?? []);
105 }
106 return references;
107}
108class TransitionParts {
109 final String trigger, kind;
110 final Expression? guard;
111 final String? target;
112 final List<Action> actions;
113 TransitionParts(this.trigger, this.guard, this.target, this.kind, this.actions);
114}
115TransitionParts transitionParts(String text) {
116 final first = text.indexOf(' ');
117 String trigger, rest;
118 if (text.startsWith('after ')) {
119 final end = text.indexOf(' ', 6);
120 trigger = end < 0 ? text : text.substring(0, end);
121 rest = end < 0 ? '' : text.substring(end);
122 } else {
123 trigger = first < 0 ? text : text.substring(0, first);
124 rest = first < 0 ? '' : text.substring(first);
125 }
126 Expression? guard;
127 String? target;
128 String kind = 'internal';
129 if (rest.startsWith(' [')) {
130 final end = rest.indexOf(']');
131 guard = expression(rest.substring(2, end));
132 rest = rest.substring(end + 1);
133 }
134 if (rest.startsWith(' -> ') || rest.startsWith(' ~> ')) {
135 kind = rest[1] == '-' ? 'external' : 'local';
136 rest = rest.substring(4);
137 final end = rest.indexOf(' ');
138 target = end < 0 ? rest : rest.substring(0, end);
139 rest = end < 0 ? '' : rest.substring(end);
140 }
141 return TransitionParts(trigger, guard, target, kind, actions(rest.isEmpty ? '' : rest.substring(3)));
142}

173class Chart {
174 final root = State('', 'state', null, 0, null, '', null);
175 final states = <String, State>{}, variables = <String, int>{}, errors = <int, String>{};
176 List<State> order = [];
177 String? checkReferences(List<Reference> references) {
178 for (final reference in references) {
179 if (!(reference.kind == 'state' ? states.containsKey(reference.name) : variables.containsKey(reference.name))) {
180 return 'unknown ${reference.kind} ${reference.name}';
181 }
182 }
183 return null;
184 }
185}
186List<State> childStates(State state) => state.children.where((child) => child.kind != 'history').toList();
187bool compound(State state) => state.parent == null || state.kind == 'state' && childStates(state).isNotEmpty;
188bool parallel(State state) => state.kind == 'parallel' && childStates(state).isNotEmpty;
189bool atomic(State state) => state.kind != 'history' && childStates(state).isEmpty;
190bool descendant(State state, State ancestor) {
191 for (State? parent = state.parent; parent != null; parent = parent.parent) {
192 if (identical(parent, ancestor)) return true;
193 }
194 return false;
195}
196State initial(State state, Chart chart) => state.initial == null ? childStates(state).first : chart.states[state.initial]!;
197State fallback(State state, Chart chart) => state.fallback == null ? initial(state.parent!, chart) : chart.states[state.fallback]!;
198List<State> document(State root) {
199 final result = <State>[];
200 void visit(State state) {
201 result.add(state);
202 for (final child in state.children) { visit(child); }
203 }
204 for (final state in root.children) { visit(state); }
205 return result;
206}
207Chart build(List<(String, int)> lines) {
208 final chart = Chart(), definitions = <Definition>[];
209 for (final (text, line) in lines) {
210 final words = text.split(' '), kind = words[0], name = words[1];
211 final definition = Definition(text, line);
212 definitions.add(definition);
213 if (['state', 'parallel', 'final', 'history'].contains(kind)) {
214 String? parentName, selected, defaultTarget;
215 String mode = '';
216 if (kind == 'state') {
217 if (words.length > 2 && words[2] == 'initial') { selected = words[3]; }
218 else {
219 parentName = words.length > 2 ? words[2] : null;
220 if (words.length > 3 && words[3] == 'initial') selected = words[4];
221 }
222 } else {
223 parentName = words.length > 2 ? words[2] : null;
224 if (kind == 'history') { mode = words[3]; defaultTarget = words.length > 4 ? words[4] : null; }
225 }
226 if (chart.states.containsKey(name)) { chart.errors[line] = 'duplicate state $name'; }
227 else if (parentName != null && !chart.states.containsKey(parentName)) { chart.errors[line] = 'unknown state $parentName'; }
228 else if (parentName != null && ['final', 'history'].contains(chart.states[parentName]!.kind)) { chart.errors[line] = '$parentName cannot have children'; }
229 else {
230 final parent = parentName == null ? chart.root : chart.states[parentName]!;
231 final state = State(name, kind, parent, line, selected, mode, defaultTarget);
232 parent.children.add(state);
233 chart.states[name] = state;
234 definition.state = state;
235 }
236 } else if (kind == 'var') {
237 if (chart.variables.containsKey(name)) { chart.errors[line] = 'duplicate variable $name'; }
238 else { chart.variables[name] = int.parse(words[2]); }
239 }
240 }
241 for (final definition in definitions) {
242 if (chart.errors.containsKey(definition.line)) continue;
243 final kind = definition.kind, name = definition.name;
244 String? error;
245 if (kind == 'state') {
246 final state = definition.state!;
247 if (state.initial != null) {
248 final child = chart.states[state.initial];
249 if (child == null) { error = 'unknown state ${state.initial}'; }
250 else if (child.parent != state || child.kind == 'history') { error = '${child.name} is not a child of $name'; }
251 }
252 } else if (kind == 'final' || kind == 'history') {
253 final state = definition.state!;
254 if (state.parent!.kind != 'state' || !compound(state.parent!)) { error = '$name needs a compound parent'; }
255 else if (kind == 'history' && state.fallback != null) {
256 final target = chart.states[state.fallback];
257 if (target == null) { error = 'unknown state ${state.fallback}'; }
258 else if (!descendant(target, state.parent!) || target.kind == 'history') { error = '${target.name} is not a descendant of ${state.parent!.name}'; }
259 }
260 } else if (['entry', 'exit', 'on'].contains(kind)) {
261 final source = chart.states[name];
262 if (source == null) { error = 'unknown state $name'; }
263 else if (source.kind == 'history' && kind != 'on') { error = '$name cannot have actions'; }
264 else if (kind == 'on' && ['final', 'history'].contains(source.kind)) { error = '$name cannot have transitions'; }
265 else if (kind == 'on') {
266 final part = transitionParts(definition.text.substring('on $name '.length));
267 final references = <Reference>[...?part.guard?.references];
268 if (part.target != null) references.add(Reference('state', part.target!));
269 references.addAll(actionReferences(part.actions));
270 error = chart.checkReferences(references);
271 if (error == null) {
272 source.transitions.add(Transition(source, part.kind, part.target == null ? null : chart.states[part.target], part.guard,
273 part.trigger == 'always' ? 'always' : part.trigger.startsWith('after ') ? 'timer' : 'event',
274 part.trigger.startsWith('after ') ? int.parse(part.trigger.substring(6)) : 0,
275 part.trigger.split('|'), part.actions));
276 }
277 } else {
278 final list = actions(definition.text.substring(definition.text.indexOf(' / ') + 3));
279 error = chart.checkReferences(actionReferences(list));
280 if (error == null) (kind == 'entry' ? source.entries : source.exits).add(list);
281 }
282 }
283 if (error != null) chart.errors[definition.line] = error;
284 }
285 chart.order = document(chart.root);
286 return chart;
287}
288String names(Iterable<State> states) {
289 final result = states.map((state) => state.name).join(',');
290 return result.isEmpty ? '-' : result;
291}
292void warnings(Chart chart, List<String> output) {
293 final reachable = <State>{initial(chart.root, chart)};
294 bool changed = true;
295 while (changed) {
296 changed = false;
297 void add(State state) {
298 if (reachable.add(state)) changed = true;
299 }
300 for (final state in reachable.toList()) {
301 if (state.parent != chart.root) add(state.parent!);
302 if (compound(state)) add(initial(state, chart));
303 if (parallel(state)) { for (final child in childStates(state)) { add(child); } }
304 for (final transition in state.transitions) {
305 final target = transition.target;
306 if (target != null) add(target.kind == 'history' ? fallback(target, chart) : target);
307 }
308 }
309 }
310 for (final state in chart.order) {
311 if (state.kind != 'history' && !reachable.contains(state)) output.add('warning: unreachable state ${state.name}');
312 }

349class Machine {
350 final Chart chart;
351 final List<String> output;
352 final configuration = <State>{};
353 final Map<String, int> variables;
354 final queue = Queue<ChartEvent>();
355 List<Reservation> reservations = [];
356 int data = 0, serial = 0, time = 0, macrosteps = 0, microsteps = 0;
357 bool trace = false, halted = false;
358 Machine(this.chart, this.output) : variables = Map.of(chart.variables);
359 int evaluate(Expression expression) => value(expression.node);
360 int value(Node node) {
361 if (node is NumberNode) return node.value;
362 if (node is NameNode) {
363 if (node.kind == 'state') return configuration.contains(chart.states[node.name]) ? 1 : 0;
364 return node.name == '_data' ? data : variables[node.name]!;
365 }
366 if (node is UnaryNode) {
367 final result = value(node.value);
368 return node.operator == '-' ? -result : result == 0 ? 1 : 0;
369 }
370 final binary = node as BinaryNode;
371 final a = value(binary.left);
372 if (binary.operator == '&&' && a == 0) return 0;
373 if (binary.operator == '||' && a != 0) return 1;
374 final b = value(binary.right);
375 if (['/', '%'].contains(binary.operator) && b == 0) throw EvaluationError();
376 return switch (binary.operator) {
377 '+' => a + b, '-' => a - b, '*' => a * b, '/' => a ~/ b, '%' => a - (a ~/ b) * b,
378 '==' => a == b ? 1 : 0, '!=' => a != b ? 1 : 0, '<' => a < b ? 1 : 0,
379 '<=' => a <= b ? 1 : 0, '>' => a > b ? 1 : 0, '>=' => a >= b ? 1 : 0,
380 _ => b != 0 ? 1 : 0
381 };
382 }
383 void enqueue(String name, [int value = 0]) => queue.add(ChartEvent(name, value));
384 void actionList(List<Action> list) {
385 for (final action in list) {
386 try {
387 final result = action.expression == null ? 0 : evaluate(action.expression!);
388 switch (action.kind) {
389 case 'assign': variables[action.name] = result; break;
390 case 'raise': enqueue(action.event, result); break;
391 case 'send': reservations.add(SendReservation(time + action.delay, ++serial, ChartEvent(action.event, result))); break;
392 case 'cancel': reservations.removeWhere((item) => item is SendReservation && item.event.name == action.event); break;
393 case 'log': output.add('${action.text}${action.expression == null ? '' : result}'); break;
394 }
395 } on EvaluationError {
396 enqueue('error.execution');
397 break;
398 }
399 }
400 }
401 List<State> effective(Transition transition) {
402 final target = transition.target!;
403 return target.kind == 'history' ? target.history ?? [fallback(target, chart)] : [target];
404 }
405 Choice choice(Transition transition) {
406 if (transition.kind == 'internal') return Choice(transition, null, {});
407 final targets = effective(transition);
408 State? scope = transition.kind == 'local' && compound(transition.source)
409 && targets.every((target) => descendant(target, transition.source)) ? transition.source : transition.source.parent;
410 while (!compound(scope!) || !targets.every((target) => descendant(target, scope!))) { scope = scope!.parent; }
411 final result = scope;
412 return Choice(transition, result, chart.order.where((state) => configuration.contains(state) && descendant(state, result!)).toSet());
413 }
414 bool matches(Transition transition, Trigger trigger) {
415 if (trigger is AlwaysTrigger) return transition.type == 'always';
416 if (trigger is TimerTrigger) return identical(transition, trigger.transition);
417 final event = (trigger as EventTrigger).event;
418 return transition.type == 'event' && transition.descriptors.any((descriptor) =>
419 descriptor == '*' || descriptor == event.name || event.name.startsWith('$descriptor.'));
420 }
421 List<Choice> select(Trigger trigger) {
422 final candidates = <Choice>[], seen = <Transition>{};
423 for (final leaf in chart.order) {
424 if (!atomic(leaf) || !configuration.contains(leaf)) continue;
425 Transition? found;
426 for (State state = leaf; state != chart.root && found == null; state = state.parent!) {
427 for (final transition in state.transitions) {
428 if (!matches(transition, trigger)) continue;
429 bool enabled = true;
430 try { enabled = transition.guard == null || evaluate(transition.guard!) != 0; }
431 on EvaluationError { enqueue('error.execution'); enabled = false; }
432 if (enabled) { found = transition; break; }
433 }
434 }
435 if (found != null && seen.add(found)) candidates.add(choice(found));
436 }
437 final chosen = <Choice>[], exited = <State>{};
438 for (final choice in candidates) {
439 if (choice.exits.any(exited.contains)) continue;
440 chosen.add(choice);
441 exited.addAll(choice.exits);
442 }
443 return chosen;
444 }
445 List<State> closeEntry(Set<State> set) {
446 bool changed = true;
447 while (changed) {
448 changed = false;
449 for (final state in set.toList()) {
450 final children = parallel(state) ? childStates(state)
451 : compound(state) && !childStates(state).any(set.contains) ? [initial(state, chart)] : <State>[];
452 for (final child in children) { if (set.add(child)) changed = true; }
453 }
454 }
455 return chart.order.where(set.contains).toList();
456 }
457 bool complete(State state) {
458 if (atomic(state)) return false;
459 if (compound(state)) return childStates(state).any((child) => child.kind == 'final' && configuration.contains(child));
460 return parallel(state) && childStates(state).every(complete);
461 }
462 void enter(List<State> states) {
463 for (final state in states) {
464 configuration.add(state);
465 state.entered++;
466 state.enteredAt = time;
467 for (final transition in state.transitions) {
468 if (transition.type == 'timer') reservations.add(TimerReservation(time + transition.delay, ++serial, transition));
469 }
470 for (final list in state.entries) { actionList(list); }
471 if (state.kind == 'final') {
472 enqueue('done.state.${state.parent!.name}');
473 final grandparent = state.parent!.parent!;
474 if (parallel(grandparent) && childStates(grandparent).every(complete)) enqueue('done.state.${grandparent.name}');
475 }
476 }
477 }
478 String display(Trigger trigger) {
479 if (trigger is AlwaysTrigger) return 'always';
480 if (trigger is TimerTrigger) return 'after(${trigger.transition.delay})@${trigger.transition.source.name}';
481 final event = (trigger as EventTrigger).event;
482 return '${event.name}${event.value == 0 ? '' : '=${event.value}'}';
483 }
484 void microstep(List<Choice> chosen, Trigger trigger) {
485 final exits = <State>{for (final choice in chosen) ...choice.exits};
486 for (final history in chart.order) {
487 if (history.kind == 'history' && exits.contains(history.parent)) {
488 history.history = chart.order.where((state) => configuration.contains(state)
489 && (history.mode == 'deep' ? atomic(state) && descendant(state, history.parent!)
490 : state.parent == history.parent && state.kind != 'history')).toList();
491 }
492 }
493 final set = <State>{};
494 for (final choice in chosen) {
495 if (choice.transition.kind == 'internal') continue;
496 for (final target in effective(choice.transition)) {
497 for (State? state = target; state != choice.scope; state = state!.parent) { set.add(state!); }
498 }
499 }
500 final entries = closeEntry(set), leaving = chart.order.where(exits.contains).toList().reversed.toList();
501 if (trace) output.add('[t=$time] ${display(trigger)} (${names(chosen.map((choice) => choice.transition.source))}): exit ${names(leaving)}; enter ${names(entries)}');
502 for (final state in leaving) {
503 for (final list in state.exits) { actionList(list); }
504 configuration.remove(state);
505 state.dwell += time - state.enteredAt;
506 reservations.removeWhere((item) => item is TimerReservation && item.transition.source == state);
507 }
508 for (final choice in chosen) { actionList(choice.transition.actions); choice.transition.source.taken++; }
509 enter(entries);
510 }
511 void macrostep(Trigger trigger) {
512 macrosteps++;
513 int count = 0;
514 void perform(List<Choice> chosen, Trigger cause) {
515 if (count == 1000) { output.add('LIVELOCK t=$time'); halted = true; return; }
516 count++;
517 microsteps++;
518 microstep(chosen, cause);
519 }
520 if (trigger is StartTrigger) {
521 data = 0;
522 enter(closeEntry({initial(chart.root, chart)}));
523 } else {
524 data = trigger is TimerTrigger ? 0 : (trigger as EventTrigger).event.value;
525 final chosen = select(trigger);
526 if (chosen.isNotEmpty) perform(chosen, trigger);
527 }
528 while (!halted) {
529 data = 0;
530 final always = AlwaysTrigger();
531 final chosen = select(always);
532 if (chosen.isNotEmpty) { perform(chosen, always); continue; }
533 if (queue.isEmpty) break;
534 final event = queue.removeFirst(), cause = EventTrigger(event);
535 data = event.value;
536 final selected = select(cause);
537 if (selected.isNotEmpty) perform(selected, cause);
538 }
539 }
540 List<Reservation> sortedReservations() => List<Reservation>.of(reservations)
541 ..sort((a, b) => a.due != b.due ? a.due.compareTo(b.due) : a.serial.compareTo(b.serial));
542 Reservation? nextReservation() {
543 Reservation? result;
544 for (final item in reservations) {
545 if (result == null || item.due < result.due || item.due == result.due && item.serial < result.serial) result = item;
546 }
547 return result;

549 void script(String line) {
550 if (halted) return;
551 final words = line.split(' '), command = words[0];
552 if (command == 'event') {
553 macrostep(EventTrigger(ChartEvent(words[1], words.length == 3 ? int.parse(words[2]) : 0)));
554 } else if (command == 'tick') {
555 final finish = time + int.parse(words[1]);
556 while (!halted) {
557 final first = nextReservation();
558 if (first == null || first.due > finish) break;
559 reservations.remove(first);
560 time = first.due;
561 macrostep(first is TimerReservation ? TimerTrigger(first.transition) : EventTrigger((first as SendReservation).event));
562 }
563 if (!halted) time = finish;
564 } else if (command == 'set') {
565 if (!variables.containsKey(words[1])) { output.add('error: unknown variable ${words[1]}'); }
566 else { variables[words[1]] = int.parse(words[2]); }
567 } else if (command == 'trace') { trace = words[1] == 'on'; }
568 else if (command == 'eval') {
569 final text = line.substring(5), parsed = expression(line.substring(5));
570 final error = chart.checkReferences(parsed.references);
571 if (error != null) { output.add('error: $error'); }
572 else {
573 data = 0;
574 try { output.add('$text -> ${evaluate(parsed)}'); }
575 on EvaluationError { output.add('error: division by zero'); }
576 }
577 } else if (command == 'dump') {
578 output.add('dump t=$time');
579 for (final state in chart.order) { if (configuration.contains(state)) output.add('${' ' * (2 * state.depth)}${state.name}'); }
580 final variableNames = variables.keys.toList()..sort();
581 for (final name in variableNames) { output.add(' var $name = ${variables[name]}'); }
582 for (final state in chart.order) {
583 if (state.kind == 'history' && state.history != null) output.add(' history ${state.name} = ${names(state.history!)}');
584 }
585 for (final reservation in sortedReservations()) {
586 if (reservation is TimerReservation) {
587 output.add(' timer ${reservation.due} after(${reservation.transition.delay})@${reservation.transition.source.name}');
588 } else {
589 final event = (reservation as SendReservation).event;
590 output.add(' send ${reservation.due} ${event.name} ${event.value}');
591 }
592 }
593 }
594 }
595 void statistics() {
596 output.add('end t=$time macrosteps=$macrosteps microsteps=$microsteps');
597 for (final state in chart.order) {
598 if (state.kind != 'history') output.add(' ${state.name} entered=${state.entered} taken=${state.taken} time=${state.dwell + (configuration.contains(state) ? time - state.enteredAt : 0)}');
599 }
600 }

603void main() {
604 final output = <String>[];
605 final count = int.parse(stdin.readLineSync()!);
606 int line = 1;
607 for (int chartNumber = 0; chartNumber < count; chartNumber++) {
608 final name = stdin.readLineSync()!.substring(6);
609 line++;
610 final definitions = <(String, int)>[], script = <String>[];
611 while (true) {
612 final text = stdin.readLineSync()!;
613 line++;
614 if (text == 'RUN') break;
615 definitions.add((text, line));
616 }
617 while (true) {
618 final text = stdin.readLineSync()!;
619 line++;
620 if (text == 'END') break;
621 script.add(text);
622 }
623 output.add('chart $name');
624 final chart = build(definitions);
625 if (chart.errors.isNotEmpty) {
626 final errorLines = chart.errors.keys.toList()..sort();
627 for (final number in errorLines) { output.add('error line $number: ${chart.errors[number]}'); }
628 continue;
629 }
630 warnings(chart, output);
631 final machine = Machine(chart, output);
632 machine.macrostep(StartTrigger());
633 for (final text in script) { machine.script(text); }
634 machine.statistics();
635 }
636 stdout.write('${output.join('\n')}\n');
637}
