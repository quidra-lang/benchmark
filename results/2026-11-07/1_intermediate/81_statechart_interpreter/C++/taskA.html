45class ExpressionParser {
46 vector<string> tokens;
47 size_t position = 0;
48 vector<pair<string, string>> references;
49 static int precedence(const string& token) {
50 if (token == "||") return 0;
51 if (token == "&&") return 1;
52 if (token == "==" || token == "!=" || token == "<" || token == "<=" || token == ">" || token == ">=") return 2;
53 if (token == "+" || token == "-") return 3;
54 if (token == "*" || token == "/" || token == "%") return 4;
55 return -1;
56 }
57 Expr unary() {
58 string token = tokens[position++];
59 if (token == "!" || token == "-") return make_shared<Expression>(Expression{"unary", token, 0, unary(), nullptr});
60 if (token == "(") {
61 Expr result = binary(0);
62 ++position;
63 return result;
64 }
65 if (token == "in") {
66 ++position;
67 string name = tokens[position++];
68 ++position;
69 references.push_back({"state", name});
70 return make_shared<Expression>(Expression{"state", name, 0, nullptr, nullptr});
71 }
72 if (isdigit(static_cast<unsigned char>(token[0])))
73 return make_shared<Expression>(Expression{"number", "", stoll(token), nullptr, nullptr});
74 if (token != "_data") references.push_back({"variable", token});
75 return make_shared<Expression>(Expression{"variable", token, 0, nullptr, nullptr});
76 }
77 Expr binary(int minimum) {
78 Expr left = unary();
79 while (position < tokens.size() && precedence(tokens[position]) >= minimum) {
80 string operation = tokens[position++];
81 Expr right = binary(precedence(operation) + 1);
82 left = make_shared<Expression>(Expression{"binary", operation, 0, left, right});
83 }
84 return left;
85 }
86public:
87 explicit ExpressionParser(const string& text) {
88 for (size_t i = 0; i < text.size();) {
89 unsigned char ch = text[i];
90 if (ch == ' ') {
91 ++i;
92 continue;
93 }
94 size_t start = i++;
95 if (isalnum(ch) || ch == '_') {
96 while (i < text.size() && (isalnum(static_cast<unsigned char>(text[i])) || text[i] == '_')) ++i;
97 } else if (i < text.size()) {
98 string pair = text.substr(start, 2);
99 if (pair == "||" || pair == "&&" || pair == "==" || pair == "!=" || pair == "<=" || pair == ">=") ++i;
100 }
101 tokens.push_back(text.substr(start, i - start));
102 }
103 }
104 ParsedExpression parse() {
105 Expr tree = binary(0);
106 return {tree, references};
107 }
108};
109ParsedExpression expression(const string& text) {
110 return ExpressionParser(text).parse();
111}

119vector<Action> parse_actions(const string& text) {
120 vector<Action> result;
121 for (size_t start = 0; start < text.size();) {
122 size_t end = text.find("; ", start);
123 string part = text.substr(start, end == string::npos ? end : end - start);
124 Action action;
125 size_t assignment = part.find(" = ");
126 if (assignment != string::npos && part.rfind("log ", 0) != 0) {
127 action.kind = "assign";
128 action.name = part.substr(0, assignment);
129 action.argument = expression(part.substr(assignment + 3));
130 } else if (part.rfind("log ", 0) == 0) {
131 action.kind = "log";
132 string rest = part.substr(4);
133 if (rest[0] == '"') {
134 size_t quote = rest.find('"', 1);
135 action.has_string = true;
136 action.prefix = rest.substr(1, quote - 1);
137 if (quote + 1 < rest.size()) action.argument = expression(rest.substr(quote + 2));
138 } else action.argument = expression(rest);
139 } else {
140 size_t first = part.find(' ');
141 action.kind = part.substr(0, first);
142 size_t second = part.find(' ', first + 1);
143 action.name = part.substr(first + 1, second == string::npos ? second : second - first - 1);
144 if (action.kind == "raise") {
145 if (second != string::npos) action.argument = expression(part.substr(second + 1));
146 } else if (action.kind == "send") {
147 size_t third = part.find(' ', second + 1);
148 action.delay = stoi(part.substr(second + 1, third == string::npos ? third : third - second - 1));
149 if (third != string::npos) action.argument = expression(part.substr(third + 1));
150 }
151 }
152 result.push_back(move(action));
153 if (end == string::npos) break;
154 start = end + 2;
155 }
156 return result;
157}
158struct Transition {
159 string source_name, target_name, kind, trigger;
160 vector<string> descriptors;
161 int source = -1, target = -1, delay = 0;
162 ParsedExpression guard;
163 vector<Action> actions;
164};
165Transition parse_transition(const string& text) {
166 Transition result;
167 size_t position = 3, end = text.find(' ', position);
168 result.source_name = text.substr(position, end - position);
169 position = end + 1;
170 end = text.find(' ', position);
171 string trigger = text.substr(position, end == string::npos ? end : end - position);
172 position = end == string::npos ? text.size() : end;
173 if (trigger == "after") {
174 result.trigger = "timer";
175 ++position;
176 end = text.find(' ', position);
177 result.delay = stoi(text.substr(position, end == string::npos ? end : end - position));
178 position = end == string::npos ? text.size() : end;
179 } else if (trigger == "always") result.trigger = "always";
180 else {
181 result.trigger = "event";
182 for (size_t start = 0;;) {
183 size_t separator = trigger.find('|', start);
184 result.descriptors.push_back(trigger.substr(start, separator == string::npos ? separator : separator - start));
185 if (separator == string::npos) break;
186 start = separator + 1;
187 }
188 }
189 if (text.compare(position, 2, " [") == 0) {
190 end = text.find(']', position + 2);
191 result.guard = expression(text.substr(position + 2, end - position - 2));
192 position = end + 1;
193 }
194 result.kind = "internal";
195 if (text.compare(position, 4, " -> ") == 0 || text.compare(position, 4, " ~> ") == 0) {
196 result.kind = text[position + 1] == '-' ? "external" : "local";
197 position += 4;
198 end = text.find(' ', position);
199 result.target_name = text.substr(position, end == string::npos ? end : end - position);
200 position = end == string::npos ? text.size() : end;
201 }
202 if (position < text.size()) result.actions = parse_actions(text.substr(position + 3));
203 return result;
204}
205struct Definition {
206 int line = 0, state = -1;
207 string kind, name, parent, initial, history_kind, default_name;
208 Integer value = 0;
209 vector<Action> actions;
210 Transition transition;
211 string error;
212};
213Definition parse_definition(const string& text, int line) {
214 auto items = words(text);
215 Definition result;
216 result.line = line;
217 result.kind = items[0];
218 result.name = items[1];
219 if (result.kind == "var") result.value = stoll(items[2]);
220 else if (result.kind == "entry" || result.kind == "exit") result.actions = parse_actions(text.substr(text.find(" / ") + 3));
221 else if (result.kind == "on") result.transition = parse_transition(text);
222 else if (result.kind == "state") {
223 size_t position = 2;
224 if (position < items.size() && items[position] != "initial") result.parent = items[position++];
225 if (position < items.size()) result.initial = items[position + 1];
226 } else {
227 if (items.size() >= 3) result.parent = items[2];
228 if (result.kind == "history") {
229 result.history_kind = items[3];
230 if (items.size() == 5) result.default_name = items[4];
231 }
232 }
233 return result;
234}

292 string bad_reference(const vector<pair<string, string>>& references) const {
293 for (const auto& [kind, name] : references) {
294 if (kind == "state" && !identifiers.count(name)) return "unknown state " + name;
295 if (kind == "variable" && !variables.count(name)) return "unknown variable " + name;

299 string bad_actions(const vector<Action>& actions) const {
300 for (const auto& action : actions) {
301 if (action.kind == "assign" && !variables.count(action.name)) return "unknown variable " + action.name;
302 string error = bad_reference(action.argument.references);
303 if (!error.empty()) return error;
304 }
305 return "";

307 Integer evaluate(const Expr& expression) const {
308 if (expression->kind == "number") return expression->value;
309 if (expression->kind == "variable") return expression->text == "_data" ? current_value : variables.at(expression->text);
310 if (expression->kind == "state") return active[identifiers.at(expression->text)];
311 Integer a = evaluate(expression->left);
312 string op = expression->text;
313 if (expression->kind == "unary") return op == "!" ? a == 0 : -a;
314 if (op == "&&" && a == 0) return 0;
315 if (op == "||" && a != 0) return 1;
316 Integer b = evaluate(expression->right);
317 if ((op == "/" || op == "%") && b == 0) throw EvaluationError{};
318 if (op == "+") return a + b;
319 if (op == "-") return a - b;
320 if (op == "*") return a * b;
321 if (op == "/") return a / b;
322 if (op == "%") return a % b;
323 if (op == "==") return a == b;
324 if (op == "!=") return a != b;
325 if (op == "<") return a < b;
326 if (op == "<=") return a <= b;
327 if (op == ">") return a > b;
328 if (op == ">=") return a >= b;
329 return b != 0;
330 }
331 void execute(const vector<Action>& actions) {
332 for (const auto& action : actions) {
333 Integer value = 0;
334 try {
335 if (action.argument.tree) value = evaluate(action.argument.tree);
336 }
337 catch (const EvaluationError&) {
338 internal.push_back({"error.execution", 0});
339 break;
340 }
341 if (action.kind == "assign") variables[action.name] = value;
342 else if (action.kind == "raise") internal.push_back({action.name, value});
343 else if (action.kind == "send") reservations.push_back({clock + action.delay, ++creation, -1, {action.name, value}});
344 else if (action.kind == "cancel") reservations.erase(remove_if(reservations.begin(), reservations.end(), [&](const Reservation& r) {
345 return r.transition < 0 && r.event.name == action.name;
346 }), reservations.end());
347 else if (action.kind == "log") {
348 if (action.has_string) cout << action.prefix;
349 if (action.argument.tree) cout << value;
350 cout << '\n';
351 }
352 }
353 }
354 vector<int> effective(const Transition& transition) const {
355 if (states[transition.target].kind != "history") return {transition.target};
356 if (history[transition.target]) return *history[transition.target];
357 return {states[transition.target].default_target};
358 }
359 Selected selection(int id) const {
360 const auto& transition = transitions[id];
361 Selected result{id, -1, vector<bool>(states.size())};
362 if (transition.kind == "internal") return result;
363 auto targets = effective(transition);
364 auto contains_all = [&](int parent) {
365 return all_of(targets.begin(), targets.end(), [&](int target) {
366 return descendant(target, parent);
367 });
368 };
369 if (transition.kind == "local" && compound(transition.source) && contains_all(transition.source)) result.scope = transition.source;
370 else for (int p = states[transition.source].parent; p >= 0; p = states[p].parent)
371 if (compound(p) && contains_all(p)) {
372 result.scope = p;
373 break;
374 }
375 for (int state : document) result.exits[state] = active[state] && descendant(state, result.scope);
376 return result;
377 }
378 bool matches(const Transition& transition, int id, const Trigger& trigger) const {
379 if (trigger.kind == "always") return transition.trigger == "always";
380 if (trigger.kind == "timer") return transition.trigger == "timer" && id == trigger.transition;
381 if (transition.trigger != "event") return false;
382 for (const auto& descriptor : transition.descriptors)
383 if (descriptor == "*" || descriptor == trigger.event.name || trigger.event.name.rfind(descriptor + ".", 0) == 0) return true;
384 return false;
385 }
386 vector<Selected> select(const Trigger& trigger) {
387 vector<int> candidates;
388 set<int> seen;
389 for (int leaf : document) if (active[leaf] && atomic(leaf)) {
390 int chosen = -1;
391 for (int source = leaf; source > 0 && chosen < 0; source = states[source].parent) {
392 for (int id : states[source].transitions) {
393 const auto& transition = transitions[id];
394 if (!matches(transition, id, trigger)) continue;
395 bool enabled = true;
396 try {
397 if (transition.guard.tree) enabled = evaluate(transition.guard.tree) != 0;
398 }
399 catch (const EvaluationError&) {
400 internal.push_back({"error.execution", 0});
401 enabled = false;
402 }
403 if (enabled) {
404 chosen = id;
405 break;
406 }
407 }
408 }
409 if (chosen >= 0 && seen.insert(chosen).second) candidates.push_back(chosen);
410 }
411 vector<Selected> result;
412 vector<bool> exiting(states.size());
413 for (int candidate : candidates) {
414 Selected current = selection(candidate);
415 bool conflict = false;
416 for (int state : document) if (current.exits[state] && exiting[state]) {
417 conflict = true;
418 break;
419 }
420 if (!conflict) {
421 for (int state : document) exiting[state] = exiting[state] || current.exits[state];
422 result.push_back(move(current));
423 }
424 }
425 return result;
426 }
427 void expand(vector<bool>& entries) const {
428 bool changed = true;
429 while (changed) {
430 changed = false;
431 for (int state : document) if (entries[state]) {
432 if (parallel(state)) {
433 for (int child : states[state].child_states) if (!entries[child]) {
434 entries[child] = true;
435 changed = true;
436 }
437 } else if (compound(state)) {
438 bool has_child = false;
439 for (int child : states[state].child_states) has_child = has_child || entries[child];
440 if (!has_child) {
441 entries[states[state].initial] = true;
442 changed = true;
443 }
444 }
445 }
446 }
447 }
448 bool complete(int state) const {
449 if (compound(state)) {
450 for (int child : states[state].child_states) if (states[child].kind == "final" && active[child]) return true;
451 return false;
452 }
453 if (parallel(state)) {
454 for (int child : states[state].child_states) if (!complete(child)) return false;
455 return true;
456 }
457 return false;
458 }
459 void enter(const vector<bool>& entries) {
460 for (int id : document) if (entries[id]) {
461 State& state = states[id];
462 active[id] = true;
463 ++state.entered;
464 state.entry_time = clock;
465 for (int transition : state.transitions) if (transitions[transition].trigger == "timer")
466 reservations.push_back({clock + transitions[transition].delay, ++creation, transition, {}});
467 for (const auto& actions : state.entry) execute(actions);
468 if (state.kind == "final") {
469 int parent = state.parent;
470 internal.push_back({"done.state." + states[parent].name, 0});
471 int grandparent = states[parent].parent;
472 if (grandparent > 0 && parallel(grandparent) && complete(grandparent))
473 internal.push_back({"done.state." + states[grandparent].name, 0});
474 }
475 }

483 string trigger_name(const Trigger& trigger) const {
484 if (trigger.kind == "always") return "always";
485 if (trigger.kind == "timer") {
486 const auto& transition = transitions[trigger.transition];
487 return "after(" + to_string(transition.delay) + ")@" + states[transition.source].name;

491 void microstep(const vector<Selected>& selected, const Trigger& trigger) {
492 vector<bool> exits(states.size()), entries(states.size());
493 for (const auto& item : selected) for (int state : document) exits[state] = exits[state] || item.exits[state];
494 for (int state : document) if (exits[state]) {
495 for (int child : states[state].children) if (states[child].kind == "history") {
496 vector<int> saved;
497 for (int candidate : document) if (active[candidate]) {
498 bool record = states[child].history_kind == "deep" ? atomic(candidate) && descendant(candidate, state) :
499 states[candidate].parent == state;
500 if (record) saved.push_back(candidate);
501 }
502 history[child] = move(saved);
503 }
504 }
505 for (const auto& item : selected) {
506 const auto& transition = transitions[item.transition];
507 if (transition.kind == "internal") continue;
508 for (int target : effective(transition))
509 for (int state = target; state != item.scope && state > 0; state = states[state].parent) entries[state] = true;
510 }
511 expand(entries);
512 if (tracing) {
513 vector<string> sources;
514 for (const auto& item : selected) sources.push_back(states[transitions[item.transition].source].name);
515 cout << "[t=" << clock << "] " << trigger_name(trigger) << " (" << join(sources) << "): exit " << listed(exits, true)
516 << "; enter " << listed(entries) << '\n';
517 }
518 for (auto it = document.rbegin(); it != document.rend(); ++it) if (exits[*it]) {
519 int id = *it;
520 State& state = states[id];
521 for (const auto& actions : state.exit) execute(actions);
522 active[id] = false;
523 state.dwell += clock - state.entry_time;
524 reservations.erase(remove_if(reservations.begin(), reservations.end(), [&](const Reservation& reservation) {
525 return reservation.transition >= 0 && transitions[reservation.transition].source == id;
526 }), reservations.end());
527 }
528 for (const auto& item : selected) {
529 const auto& transition = transitions[item.transition];
530 execute(transition.actions);
531 ++states[transition.source].taken;
532 }
533 enter(entries);
534 }
535 void macrostep(const Trigger& initial) {
536 ++macrosteps;
537 int count = 0;
538 auto perform = [&](const vector<Selected>& selected, const Trigger& trigger) {
539 if (count == 1000) {
540 cout << "LIVELOCK t=" << clock << '\n';
541 halted = true;
542 return;
543 }
544 ++count;
545 ++microsteps;
546 microstep(selected, trigger);
547 };
548 if (initial.kind == "start") {
549 current_value = 0;
550 vector<bool> entries(states.size());
551 entries[states[0].initial] = true;
552 expand(entries);
553 enter(entries);
554 } else {
555 current_value = initial.kind == "timer" ? 0 : initial.event.value;
556 auto selected = select(initial);
557 if (!selected.empty()) perform(selected, initial);
558 }
559 while (!halted) {
560 current_value = 0;
561 Trigger eventless{"always", {}, -1};
562 auto selected = select(eventless);
563 if (!selected.empty()) {
564 perform(selected, eventless);
565 continue;
566 }
567 if (internal.empty()) break;
568 Event event = internal.front();
569 internal.pop_front();
570 current_value = event.value;
571 Trigger trigger{"event", event, -1};
572 selected = select(trigger);
573 if (!selected.empty()) perform(selected, trigger);
574 }
575 }
576 void warnings() const {
577 vector<bool> reachable(states.size());
578 reachable[states[0].initial] = true;
579 bool changed = true;
580 while (changed) {
581 changed = false;
582 auto add = [&](int state) {
583 if (state > 0 && !reachable[state]) {
584 reachable[state] = true;
585 changed = true;
586 }
587 };
588 for (int state : document) if (reachable[state]) {
589 add(states[state].parent);
590 if (compound(state)) add(states[state].initial);
591 if (parallel(state)) for (int child : states[state].child_states) add(child);
592 for (int id : states[state].transitions) if (transitions[id].target >= 0) {
593 int target = transitions[id].target;
594 add(states[target].kind == "history" ? states[target].default_target : target);
595 }
596 }
597 }
598 for (int state : document) if (states[state].kind != "history" && !reachable[state])
599 cout << "warning: unreachable state " << states[state].name << '\n';
600 }
601 void dump() const {
602 cout << "dump t=" << clock << '\n';
603 for (int state : document) if (active[state]) cout << string(2 * states[state].depth, ' ') << states[state].name << '\n';
604 for (const auto& [name, value] : variables) cout << " var " << name << " = " << value << '\n';
605 for (int state : document) if (states[state].kind == "history" && history[state]) {
606 vector<string> names;
607 for (int saved : *history[state]) names.push_back(states[saved].name);
608 cout << " history " << states[state].name << " = " << join(names) << '\n';
609 }
610 auto ordered = reservations;
611 sort(ordered.begin(), ordered.end(), [](const Reservation& a, const Reservation& b) {
612 return tie(a.due, a.sequence) < tie(b.due, b.sequence);
613 });
614 for (const auto& reservation : ordered) {
615 if (reservation.transition < 0) cout << " send " << reservation.due << ' ' << reservation.event.name << ' ' << reservation.event.value << '\n';
616 else {
617 const auto& transition = transitions[reservation.transition];
618 cout << " timer " << reservation.due << " after(" << transition.delay << ")@" << states[transition.source].name << '\n';
619 }
620 }

623 bool define(vector<Definition>& definitions) {
624 states[0].kind = "state";
625 for (auto& definition : definitions) {
626 const string& kind = definition.kind;
627 if (kind == "var") {
628 if (variables.count(definition.name)) definition.error = "duplicate variable " + definition.name;
629 else variables[definition.name] = definition.value;
630 } else if (kind == "state" || kind == "parallel" || kind == "final" || kind == "history") {
631 if (identifiers.count(definition.name)) definition.error = "duplicate state " + definition.name;
632 else if (!definition.parent.empty() && !identifiers.count(definition.parent)) definition.error = "unknown state " + definition.parent;
633 else {
634 int parent = definition.parent.empty() ? 0 : identifiers.at(definition.parent);
635 if (states[parent].kind == "final" || states[parent].kind == "history") definition.error = definition.parent + " cannot have children";
636 else {
637 int id = states.size();
638 State state;
639 state.name = definition.name;
640 state.kind = kind;
641 state.parent = parent;
642 state.depth = states[parent].depth + 1;
643 state.initial_name = definition.initial;
644 state.history_kind = definition.history_kind;
645 state.default_name = definition.default_name;
646 states.push_back(move(state));
647 states[parent].children.push_back(id);
648 if (kind != "history") states[parent].child_states.push_back(id);
649 identifiers[definition.name] = id;
650 definition.state = id;
651 }
652 }
653 }
654 }
655 for (size_t id = 0; id < states.size(); ++id) if (compound(id)) {
656 if (states[id].initial_name.empty()) {
657 if (!states[id].child_states.empty()) states[id].initial = states[id].child_states[0];
658 } else if (identifiers.count(states[id].initial_name)) states[id].initial = identifiers.at(states[id].initial_name);
659 }
660 for (auto& definition : definitions) if (definition.error.empty()) {
661 const auto& kind = definition.kind;
662 if (kind == "state" && !definition.initial.empty()) {
663 if (!identifiers.count(definition.initial)) definition.error = "unknown state " + definition.initial;
664 else {
665 int child = identifiers.at(definition.initial);
666 if (states[child].parent != definition.state || states[child].kind == "history")
667 definition.error = definition.initial + " is not a child of " + definition.name;
668 }
669 } else if (kind == "final" || kind == "history") {
670 int id = definition.state, parent = states[id].parent;
671 if (states[parent].kind != "state" || !compound(parent)) definition.error = definition.name + " needs a compound parent";
672 else if (kind == "history" && !definition.default_name.empty()) {
673 if (!identifiers.count(definition.default_name)) definition.error = "unknown state " + definition.default_name;
674 else {
675 int target = identifiers.at(definition.default_name);
676 if (!descendant(target, parent) || states[target].kind == "history")
677 definition.error = definition.default_name + " is not a descendant of " + definition.parent;
678 }
679 }
680 } else if (kind == "entry" || kind == "exit" || kind == "on") {
681 if (!identifiers.count(definition.name)) definition.error = "unknown state " + definition.name;
682 else {
683 int source = identifiers.at(definition.name);
684 if (kind != "on" && states[source].kind == "history") definition.error = definition.name + " cannot have actions";
685 else if (kind == "on" && (states[source].kind == "history" || states[source].kind == "final")) definition.error = definition.name + " cannot have transitions";
686 else {
687 if (kind == "on") {
688 auto& transition = definition.transition;
689 definition.error = bad_reference(transition.guard.references);
690 if (definition.error.empty() && !transition.target_name.empty() && !identifiers.count(transition.target_name))
691 definition.error = "unknown state " + transition.target_name;
692 if (definition.error.empty()) definition.error = bad_actions(transition.actions);
693 if (definition.error.empty()) {
694 transition.source = source;
695 if (!transition.target_name.empty()) transition.target = identifiers.at(transition.target_name);
696 states[source].transitions.push_back(transitions.size());
697 transitions.push_back(transition);
698 }
699 } else {
700 definition.error = bad_actions(definition.actions);
701 if (definition.error.empty()) (kind == "entry" ? states[source].entry : states[source].exit).push_back(definition.actions);
702 }
703 }
704 }
705 }
706 }
707 bool valid = true;
708 for (const auto& definition : definitions) if (!definition.error.empty()) {
709 valid = false;
710 cout << "error line " << definition.line << ": " << definition.error << '\n';
711 }
712 if (!valid) return false;
713 for (size_t id = 1; id < states.size(); ++id) if (states[id].kind == "history")
714 states[id].default_target = states[id].default_name.empty() ? states[states[id].parent].initial : identifiers.at(states[id].default_name);
715 for (int child : states[0].children) visit(child);
716 active.resize(states.size());
717 history.resize(states.size());
718 warnings();
719 macrostep({"start", {}, -1});
720 return true;
721 }
722 void script(const string& text) {
723 if (halted) return;
724 auto items = words(text);
725 if (items[0] == "event") macrostep({"event", {items[1], items.size() == 3 ? stoll(items[2]) : 0}, -1});
726 else if (items[0] == "tick") {
727 Integer finish = clock + stoll(items[1]);
728 while (!halted) {
729 auto first = min_element(reservations.begin(), reservations.end(), [](const Reservation& a, const Reservation& b) {
730 return tie(a.due, a.sequence) < tie(b.due, b.sequence);
731 });
732 if (first == reservations.end() || first->due > finish) break;
733 Reservation next = *first;
734 reservations.erase(first);
735 clock = next.due;
736 if (next.transition >= 0) macrostep({"timer", {}, next.transition});
737 else macrostep({"event", next.event, -1});
738 }
739 if (!halted) clock = finish;
740 } else if (items[0] == "set") {
741 if (!variables.count(items[1])) cout << "error: unknown variable " << items[1] << '\n';
742 else variables[items[1]] = stoll(items[2]);
743 } else if (items[0] == "trace") tracing = items[1] == "on";
744 else if (items[0] == "eval") {
745 string original = text.substr(5);
746 auto parsed = expression(original);
747 string error = bad_reference(parsed.references);
748 if (!error.empty()) cout << "error: " << error << '\n';
749 else {
750 current_value = 0;
751 try {
752 Integer value = evaluate(parsed.tree);
753 cout << original << " -> " << value << '\n';
754 }
755 catch (const EvaluationError&) {
756 cout << "error: division by zero\n";
757 }
758 }
759 } else if (items[0] == "dump") dump();
760 }
761 void finish() const {
762 cout << "end t=" << clock << " macrosteps=" << macrosteps << " microsteps=" << microsteps << '\n';
763 for (int id : document) if (states[id].kind != "history") {
764 const State& state = states[id];
765 cout << " " << state.name << " entered=" << state.entered << " taken=" << state.taken << " time="
766 << state.dwell + (active[id] ? clock - state.entry_time : 0) << '\n';
767 }
768 }
769};
