5expression <- function(text) {
6 if (is.null(text)) return(NULL)
7 tokens <- regmatches(text, gregexpr("\\|\\||&&|==|!=|<=|>=|[()+*/%!<>-]|[0-9]+|[A-Za-z_][A-Za-z0-9_]*", text, perl = TRUE))[[1]]
8 index <- 1L
9 references <- list()
10 precedence <- list("||" = 0L, "&&" = 1L, "==" = 2L, "!=" = 2L, "<" = 2L, "<=" = 2L, ">" = 2L, ">=" = 2L, "+" = 3L, "-" = 3L, "*" = 4L, "/" = 4L, "%" = 4L)
11 parse <- function(level) {
12 if (level == 5L) return(unary())
13 left <- parse(level + 1L)
14 repeat {
15 token <- if (index <= length(tokens)) tokens[index] else ""
16 priority <- precedence[[token]]
17 if (is.null(priority) || priority != level) break
18 index <<- index + 1L
19 right <- parse(level + 1L)
20 left <- list(operator = token, left = left, right = right)
21 }
22 left
23 }
24 unary <- function() {
25 token <- tokens[index]
26 index <<- index + 1L
27 if (token %in% c("!", "-")) return(list(operator = if (token == "!") "not" else "negative", left = unary()))
28 if (token == "(") {
29 result <- parse(0L)
30 index <<- index + 1L
31 return(result)
32 }
33 if (token == "in") {
34 index <<- index + 1L
35 state <- tokens[index]
36 index <<- index + 2L
37 references[[length(references) + 1L]] <<- list(kind = "state", name = state)
38 return(list(operator = "in", left = state))
39 }
40 if (grepl("\\A[0-9]+\\z", token, perl = TRUE)) return(list(operator = "number", left = as.numeric(token)))
41 if (token != "_data") references[[length(references) + 1L]] <<- list(kind = "variable", name = token)
42 list(operator = "variable", left = token)
43 }
44 tree <- parse(0L)
45 list(tree = tree, references = references)
46}
47evaluate_expression <- function(expression, variables, data, configuration) {
48 evaluate <- function(node) {
49 operator <- node$operator
50 if (operator == "number") return(node$left)
51 if (operator == "variable") return(if (node$left == "_data") data else variables[[node$left]])
52 if (operator == "in") return(as.numeric(!is.null(configuration[[node$left]])))
53 if (operator %in% c("not", "negative")) {
54 value <- evaluate(node$left)
55 return(if (operator == "not") as.numeric(value == 0) else -value)
56 }
57 a <- evaluate(node$left)
58 if (operator == "&&" && a == 0) return(0)
59 if (operator == "||" && a != 0) return(1)
60 b <- evaluate(node$right)
61 if (operator %in% c("/", "%") && b == 0) evaluation_error()
62 switch(operator,
63 "+" = a + b, "-" = a - b, "*" = a * b, "/" = trunc(a / b), "%" = a - trunc(a / b) * b,
64 "==" = as.numeric(a == b), "!=" = as.numeric(a != b), "<" = as.numeric(a < b),
65 "<=" = as.numeric(a <= b), ">" = as.numeric(a > b), ">=" = as.numeric(a >= b),
66 "&&" = as.numeric(b != 0), "||" = as.numeric(b != 0))
67 }
68 evaluate(expression$tree)
69}

115 actions <- function(text) {
116 references <- result <- list()
117 parts <- if (is.null(text)) character() else strsplit(text, "; ", fixed = TRUE)[[1]]
118 for (action in parts) {
119 space <- regexpr(" ", action, fixed = TRUE)[1]
120 kind <- substr(action, 1, space - 1L)
121 rest <- substring(action, space + 1L)
122 record <- list(kind = kind)
123 if (kind == "raise") {
124 space <- regexpr(" ", rest, fixed = TRUE)[1]
125 record$event <- if (space < 0) rest else substr(rest, 1, space - 1L)
126 record$expression <- expression(if (space < 0) NULL else substring(rest, space + 1L))
127 } else if (kind == "send") {
128 words <- strsplit(rest, " ", fixed = TRUE)[[1]]
129 record$event <- words[1]
130 record$delay <- as.numeric(words[2])
131 record$expression <- expression(if (length(words) <= 2) NULL else paste(words[-c(1, 2)], collapse = " "))
132 } else if (kind == "cancel") {
133 record$event <- rest
134 } else if (kind == "log") {
135 if (startsWith(rest, "\"")) {
136 finish <- regexpr("\"", substring(rest, 2), fixed = TRUE)[1] + 1L
137 record$text <- substr(rest, 2, finish - 1L)
138 if (finish < nchar(rest)) record$expression <- expression(substring(rest, finish + 2L))
139 } else {
140 record$expression <- expression(rest)
141 }
142 } else {
143 equals <- regexpr(" = ", action, fixed = TRUE)[1]
144 name <- substr(action, 1, equals - 1L)
145 record <- list(kind = "assign", name = name, expression = expression(substring(action, equals + 3L)))
146 references[[length(references) + 1L]] <- list(kind = "variable", name = name)
147 }
148 if (!is.null(record$expression)) references <- c(references, record$expression$references)
149 result[[length(result) + 1L]] <- record
150 }
151 list(actions = result, references = references)
152 }

256 transition$descriptors <- if (transition$kind == "event") strsplit(trigger, "|", fixed = TRUE)[[1]] else character()
257 transition$delay <- if (transition$kind == "timer") as.numeric(strsplit(trigger, " ", fixed = TRUE)[[1]][2]) else 0
258 transition$guard <- guard
259 transition$target <- if (is.null(target_name)) NULL else chart$states[[target_name]]
260 transition$mode <- if (is.null(arrow)) "internal" else if (arrow == "~>") "local" else "external"
261 transition$actions <- parsed$actions

263 }
264 }
265 }
266 if (!is.null(reason)) chart$errors[[number]] <- reason
267 }
268 if (length(chart$errors) > 0) {
269 for (number in sort(as.integer(names(chart$errors)))) emit("error line ", number, ": ", chart$errors[[as.character(number)]])
270 return(FALSE)

312 for (action in list) {
313 value <- if (is.null(action$expression)) 0 else evaluate(action$expression)

318 if (action$kind == "log") emit(if (is.null(action$expression)) action$text else paste0(if (is.null(action$text)) "" else action$text, value))
319 }
320 }, evaluation_error = function(error) {

330 choice <- function(transition) {
331 if (transition$mode == "internal") return(list(transition = transition, scope = NULL, exits = list()))
332 targets <- effective_targets(transition)
333 all_descendants <- function(ancestor) all(vapply(targets, function(target) descendant(target, ancestor), logical(1)))
334 source <- transition$source
335 if (transition$mode == "local" && compound(source) && all_descendants(source)) {
336 scope <- source
337 } else {
338 scope <- source$parent
339 while (!compound(scope) || !all_descendants(scope)) scope <- scope$parent
340 }
341 list(transition = transition, scope = scope, exits = Filter(function(state) descendant(state, scope), chart$configuration))
342 }
343 matches <- function(transition, trigger) {

345 if (trigger$kind == "timer") return(identical(transition, trigger$transition))
346 if (transition$kind != "event") return(FALSE)
347 any(vapply(transition$descriptors, function(descriptor) descriptor == "*" || descriptor == trigger$event || startsWith(trigger$event, paste0(descriptor, ".")), logical(1)))

349 select_transitions <- function(trigger) {
350 selected <- list()
351 for (active in chart$document) {
352 if (is.null(chart$configuration[[active$name]]) || !atomic(active)) next
353 state <- active
354 found <- NULL
355 while (!identical(state, chart$root) && is.null(found)) {
356 for (transition in state$transitions) {
357 if (!matches(transition, trigger)) next
358 accepted <- tryCatch(is.null(transition$guard) || evaluate(transition$guard) != 0, evaluation_error = function(error) {
359 chart$queue[[length(chart$queue) + 1L]] <- list(event = "error.execution", value = 0)
360 FALSE
361 })
362 if (accepted) {
363 found <- transition
364 break
365 }
366 }
367 state <- state$parent
368 }
369 if (!is.null(found) && !any(vapply(selected, function(t) identical(t, found), logical(1)))) selected[[length(selected) + 1L]] <- found
370 }
371 occupied <- result <- list()
372 for (transition in selected) {
373 candidate <- choice(transition)
374 if (any(names(candidate$exits) %in% names(occupied))) next
375 occupied <- merge_set(occupied, candidate$exits)
376 result[[length(result) + 1L]] <- candidate
377 }
378 result
379 }

421 microstep <- function(selected, trigger) {
422 exits <- list()
423 for (candidate in selected) exits <- merge_set(exits, candidate$exits)
424 for (state in chart$document) {
425 if (state$kind != "history" || is.null(exits[[state$parent$name]])) next

427 !is.null(chart$configuration[[active$name]]) && if (state$history_kind == "deep") atomic(active) && descendant(active, state$parent) else identical(active$parent, state$parent)
428 }, chart$document)
429 }
430 entries <- list()
431 for (candidate in selected) {
432 transition <- candidate$transition
433 if (transition$mode == "internal") next
434 for (target in effective_targets(transition)) {
435 state <- target
436 while (descendant(state, candidate$scope)) {
437 entries[[state$name]] <- state

439 }
440 }
441 }

462 }
463 macrostep <- function(trigger) {
464 chart$macrosteps <- chart$macrosteps + 1
465 count <- 0L
466 perform <- function(selected, current) {
467 if (count == 1000L) {
468 emit("LIVELOCK t=", chart$time)
469 chart$halted <- TRUE
470 } else {
471 count <<- count + 1L
472 chart$microsteps <- chart$microsteps + 1
473 microstep(selected, current)
474 }
475 }
476 chart$data <- if (trigger$kind == "event") trigger$value else 0
477 if (trigger$kind == "start") {
478 initial <- initial_child(chart$root)
479 enter(expand_entry(setNames(list(initial), initial$name)))
480 } else {
481 selected <- select_transitions(trigger)
482 if (length(selected) > 0) perform(selected, trigger)
483 }
484 while (!chart$halted) {
485 chart$data <- 0
486 current <- list(kind = "always")
487 selected <- select_transitions(current)
488 if (length(selected) > 0) {
489 perform(selected, current)
490 next
491 }
492 if (length(chart$queue) == 0) break
493 event <- chart$queue[[1]]
494 chart$queue <- chart$queue[-1]
495 chart$data <- event$value
496 current <- list(kind = "event", event = event$event, value = event$value)
497 selected <- select_transitions(current)
498 if (length(selected) > 0) perform(selected, current)
499 }

515 run <- function(script) {
516 if (!validate()) return(invisible(NULL))
517 warn_unreachable()
518 macrostep(list(kind = "start"))
519 for (line in script) {
520 if (chart$halted) break
521 space <- regexpr(" ", line, fixed = TRUE)[1]
522 command <- if (space < 0) line else substr(line, 1, space - 1L)
523 rest <- if (space < 0) "" else substring(line, space + 1L)
524 if (command == "event") {
525 parts <- strsplit(rest, " ", fixed = TRUE)[[1]]
526 macrostep(list(kind = "event", event = parts[1], value = if (length(parts) >= 2) as.numeric(parts[2]) else 0))
527 } else if (command == "tick") {
528 finish <- chart$time + as.numeric(rest)
529 while (!chart$halted) {
530 reservations <- ordered_reservations()
531 if (length(reservations) == 0 || reservations[[1]]$due > finish) break
532 next_reservation <- reservations[[1]]
533 chart$reservations <- Filter(function(r) r$number != next_reservation$number, chart$reservations)
534 chart$time <- next_reservation$due
535 trigger <- if (next_reservation$kind == "timer") next_reservation else list(kind = "event", event = next_reservation$event, value = next_reservation$value)
536 macrostep(trigger)
537 }
538 if (!chart$halted) chart$time <- finish
539 } else if (command == "set") {
540 parts <- strsplit(rest, " ", fixed = TRUE)[[1]]
541 name <- parts[1]
542 if (!is.null(chart$variables[[name]])) chart$variables[[name]] <- as.numeric(parts[2]) else emit("error: unknown variable ", name)
543 } else if (command == "trace") {
544 chart$trace <- rest == "on"
545 } else if (command == "eval") {
546 value <- expression(rest)
547 reason <- check_references(value$references)
548 if (!is.null(reason)) {
549 emit("error: ", reason)
550 } else {
551 chart$data <- 0
552 tryCatch({
553 result <- evaluate(value)
554 emit(rest, " -> ", result)
555 }, evaluation_error = function(error) emit("error: division by zero"))
556 }
557 } else if (command == "dump") {
558 dump()
559 }
560 }
561 emit("end t=", chart$time, " macrosteps=", chart$macrosteps, " microsteps=", chart$microsteps)
562 for (state in chart$document) {
563 if (state$kind == "history") next
564 time <- state$dwell + if (is.null(chart$configuration[[state$name]])) 0 else chart$time - state$entry_time
565 emit(" ", state$name, " entered=", state$entered, " taken=", state$taken, " time=", time)
566 }
567 invisible(NULL)
568 }
569 run
570}
