3class Expression
4 attr_reader :references
5
6 def initialize(text)
7 @tokens = text.scan(/\|\||&&|==|!=|<=|>=|[()+*\/%!<>-]|[0-9]+|[A-Za-z_][A-Za-z0-9_]*/)
8 @index = 0
9 @references = []
10 @tree = parse(0)
11 end
12
13 PRECEDENCE = { '||' => 0, '&&' => 1, '==' => 2, '!=' => 2, '<' => 2, '<=' => 2,
14 '>' => 2, '>=' => 2, '+' => 3, '-' => 3, '*' => 4, '/' => 4, '%' => 4 }.freeze
15
16 def parse(level)
17 return unary if level == 5
18 left = parse(level + 1)
19 while PRECEDENCE[@tokens[@index]] == level
20 operator = @tokens[@index]
21 @index += 1
22 left = [operator, left, parse(level + 1)]
23 end
24 left
25 end
26
27 def unary
28 token = @tokens[@index]
29 @index += 1
30 if %w[! -].include?(token)
31 [token == '!' ? :not : :negative, unary]
32 elsif token == '('
33 result = parse(0)
34 @index += 1
35 result
36 elsif token == 'in'
37 @index += 1
38 state = @tokens[@index]
39 @index += 2
40 @references << [:state, state]
41 [:in, state]
42 elsif /\A[0-9]+\z/.match?(token)
43 [:number, token.to_i]
44 else
45 @references << [:variable, token] unless token == '_data'
46 [:variable, token]
47 end
48 end
49
50 def evaluate(variables, data, configuration, states, node = @tree)
51 operator, left, right = node
52 return left if operator == :number
53 return left == '_data' ? data : variables[left] if operator == :variable
54 return configuration.key?(states[left]) ? 1 : 0 if operator == :in
55 if operator == :not || operator == :negative
56 value = evaluate(variables, data, configuration, states, left)
57 return operator == :not ? (value.zero? ? 1 : 0) : -value
58 end
59 a = evaluate(variables, data, configuration, states, left)
60 return 0 if operator == '&&' && a.zero?
61 return 1 if operator == '||' && !a.zero?
62 b = evaluate(variables, data, configuration, states, right)
63 raise EvaluationError if %w[/ %].include?(operator) && b.zero?
64 case operator
65 when '+' then a + b
66 when '-' then a - b
67 when '*' then a * b
68 when '/', '%'
69 quotient = a.abs / b.abs
70 quotient = -quotient if a.negative? != b.negative?
71 operator == '/' ? quotient : a - quotient * b
72 when '==' then a == b ? 1 : 0
73 when '!=' then a != b ? 1 : 0
74 when '<' then a < b ? 1 : 0
75 when '<=' then a <= b ? 1 : 0
76 when '>' then a > b ? 1 : 0
77 when '>=' then a >= b ? 1 : 0
78 when '&&', '||' then b.zero? ? 0 : 1
79 end
80 end
81end

139 def descendant?(state, ancestor)
140 parent = state.parent
141 while parent
142 return true if parent == ancestor
143 parent = parent.parent
144 end
145 false

148 def check_references(references)
149 references.each do |kind, name|
150 return "unknown variable #{name}" if kind == :variable && !@variables.key?(name)
151 return "unknown state #{name}" if kind == :state && !@states.key?(name)
152 end
153 nil
154 end

156 def expression(text)
157 text ? Expression.new(text) : nil
158 end
159
160 def actions(text)
161 references = []
162 result = (text ? text.split('; ') : []).map do |action|
163 kind, rest = action.split(' ', 2)
164 record = { kind: kind }
165 case kind
166 when 'raise'
167 event, value = rest.split(' ', 2)
168 record.merge!(event: event, expression: expression(value))
169 when 'send'
170 event, delay, value = rest.split(' ', 3)
171 record.merge!(event: event, delay: delay.to_i, expression: expression(value))
172 when 'cancel'
173 record[:event] = rest
174 when 'log'
175 if rest.start_with?('"')
176 finish = rest.index('"', 1)
177 record[:text] = rest[1...finish]
178 record[:expression] = expression(rest[(finish + 2)..]) if finish + 1 < rest.length
179 else
180 record[:expression] = expression(rest)
181 end
182 else
183 name, value = action.split(' = ', 2)
184 record.merge!(kind: 'assign', name: name, expression: expression(value))
185 references << [:variable, name]
186 end
187 references.concat(record[:expression].references) if record[:expression]
188 record
189 end
190 [result, references]
191 end

193 def validate
194 @definitions.each do |line, number|
195 parts = line.split(' ')
196 kind, name = parts.take(2)
197 if %w[state parallel final history].include?(kind)
198 parent_name = kind == 'state' && parts[2] == 'initial' ? nil : parts[2]
199 if @states.key?(name)
200 @errors[number] = "duplicate state #{name}"
201 elsif parent_name && !@states.key?(parent_name)
202 @errors[number] = "unknown state #{parent_name}"
203 elsif parent_name && %w[final history].include?(@states[parent_name].kind)
204 @errors[number] = "#{parent_name} cannot have children"
205 else
206 parent = parent_name ? @states[parent_name] : @root
207 state = State.new(name, kind, parent)
208 state.history_kind = parts[3] if kind == 'history'
209 state.default = parts[4] if kind == 'history'
210 if kind == 'state' && (at = parts.index('initial'))
211 state.initial = parts[at + 1]
212 end
213 @states[name] = state
214 parent.children << state
215 end
216 elsif kind == 'var'
217 if @variables.key?(name)
218 @errors[number] = "duplicate variable #{name}"
219 else
220 @variables[name] = parts[2].to_i
221 end
222 end
223 end
224 visit = lambda do |state|
225 @document << state
226 state.children.each { |child| visit.call(child) }
227 end
228 @root.children.each { |state| visit.call(state) }
229 @definitions.each do |line, number|
230 next if @errors.key?(number)
231 kind, name = line.split(' ').take(2)
232 state = @states[name]
233 reason = nil
234 if kind == 'state' && state.initial
235 child = @states[state.initial]
236 reason = child.nil? ? "unknown state #{state.initial}" : child.parent != state || child.kind == 'history' ? "#{child.name} is not a child of #{name}" : nil
237 elsif %w[final history].include?(kind)
238 parent = state.parent
239 if parent.kind != 'state' || parent.child_states.empty?
240 reason = "#{name} needs a compound parent"
241 elsif kind == 'history' && state.default
242 target = @states[state.default]
243 reason = target.nil? ? "unknown state #{state.default}" : !descendant?(target, parent) || target.kind == 'history' ? "#{target.name} is not a descendant of #{parent.name}" : nil
244 end
245 elsif %w[entry exit].include?(kind)
246 if state.nil?
247 reason = "unknown state #{name}"
248 elsif state.kind == 'history'
249 reason = "#{name} cannot have actions"
250 else
251 list, references = actions(line.split(' / ', 2)[1])
252 reason = check_references(references)
253 (kind == 'entry' ? state.entries : state.exits) << list unless reason
254 end
255 elsif kind == 'on'
256 if state.nil?
257 reason = "unknown state #{name}"
258 elsif %w[final history].include?(state.kind)
259 reason = "#{name} cannot have transitions"
260 else
261 match = /\Aon (\S+) (always|after [0-9]+|[^ ]+)(?: \[(.*?)\])?(?: (->|~>) (\S+))?(?: \/ (.*))?\z/.match(line)
262 trigger, guard_text, arrow, target_name, action_text = match.captures.drop(1)
263 guard = expression(guard_text)
264 list, references = actions(action_text)
265 ordered_references = guard ? guard.references.dup : []
266 ordered_references << [:state, target_name] if target_name
267 reason = check_references(ordered_references + references)
268 unless reason
269 transition = Transition.new
270 transition.source = state
271 transition.kind = trigger == 'always' ? :always : trigger.start_with?('after ') ? :timer : :event
272 transition.descriptors = trigger.split('|') if transition.kind == :event
273 transition.delay = trigger.split(' ')[1].to_i if transition.kind == :timer
274 transition.guard = guard
275 transition.target = target_name ? @states[target_name] : nil
276 transition.mode = arrow == '~>' ? :local : arrow ? :external : :internal
277 transition.actions = list
278 state.transitions << transition
279 end
280 end
281 end
282 @errors[number] = reason if reason
283 end
284 unless @errors.empty?
285 @errors.keys.sort.each { |number| puts "error line #{number}: #{@errors[number]}" }
286 return false
287 end
288 @document.each do |state|
289 state.initial = @states[state.initial] if state.initial
290 state.default = @states[state.default] if state.default
291 end
292 true
293 end

295 def default_target(history)
296 history.default || history.parent.initial_child
297 end

299 def warn_unreachable
300 reachable = { @root.initial_child => true }
301 loop do
302 before = reachable.length
303 reachable.keys.each do |state|
304 reachable[state.parent] = true if state.parent && state.parent != @root
305 reachable[state.initial_child] = true if state.compound?
306 state.child_states.each { |child| reachable[child] = true } if state.parallel?
307 state.transitions.each do |transition|
308 target = transition.target
309 reachable[target.kind == 'history' ? default_target(target) : target] = true if target
310 end
311 end
312 break if reachable.length == before
313 end
314 @document.each { |state| puts "warning: unreachable state #{state.name}" if state.kind != 'history' && !reachable[state] }
315 end

321 def reserve(record)
322 @creation += 1
323 @reservations << record.merge(due: @time + record[:delay], number: @creation)
324 end

326 def run_actions(list)
327 list.each do |action|
328 value = action[:expression] ? evaluate(action[:expression]) : 0
329 case action[:kind]
330 when 'assign' then @variables[action[:name]] = value
331 when 'raise' then @queue << [action[:event], value]
332 when 'send' then reserve(kind: :send, event: action[:event], value: value, delay: action[:delay])
333 when 'cancel' then @reservations.reject! { |r| r[:kind] == :send && r[:event] == action[:event] }
334 when 'log'
335 puts(action[:expression] ? (action[:text] || '') + value.to_s : action[:text])
336 end
337 end
338 rescue EvaluationError
339 @queue << ['error.execution', 0]
340 end

342 def effective_targets(transition)
343 target = transition.target
344 return [] unless target
345 target.kind == 'history' ? (@history.key?(target) ? @history[target] : [default_target(target)]) : [target]
346 end

348 def choice(transition)
349 return { transition: transition, scope: nil, exits: {} } if transition.mode == :internal
350 targets = effective_targets(transition)
351 source = transition.source
352 if transition.mode == :local && source.compound? && targets.all? { |target| descendant?(target, source) }
353 scope = source
354 else
355 scope = source.parent
356 scope = scope.parent until scope.compound? && targets.all? { |target| descendant?(target, scope) }
357 end
358 { transition: transition, scope: scope, exits: @configuration.select { |state, _| descendant?(state, scope) } }
359 end

361 def matches?(transition, trigger)
362 if trigger[:kind] == :always
363 transition.kind == :always
364 elsif trigger[:kind] == :timer
365 transition == trigger[:transition]
366 else
367 transition.kind == :event && transition.descriptors.any? { |descriptor| descriptor == '*' || descriptor == trigger[:event] || trigger[:event].start_with?(descriptor + '.') }
368 end
369 end

371 def select_transitions(trigger)
372 selected = []
373 @document.each do |atomic|
374 next unless @configuration[atomic] && atomic.atomic?
375 state = atomic
376 found = nil
377 while state != @root && !found
378 state.transitions.each do |transition|
379 next unless matches?(transition, trigger)
380 begin
381 accepted = transition.guard.nil? || !evaluate(transition.guard).zero?
382 rescue EvaluationError
383 @queue << ['error.execution', 0]
384 accepted = false
385 end
386 if accepted
387 found = transition
388 break
389 end
390 end
391 state = state.parent
392 end
393 selected << found if found && !selected.include?(found)
394 end
395 occupied = {}
396 selected.filter_map do |transition|
397 candidate = choice(transition)
398 next if candidate[:exits].keys.any? { |state| occupied[state] }
399 occupied.merge!(candidate[:exits])
400 candidate
401 end
402 end

404 def expand_entry(set)
405 loop do
406 before = set.length
407 set.keys.each do |state|
408 if state.parallel?
409 state.child_states.each { |child| set[child] = true }
410 elsif state.compound? && state.child_states.none? { |child| set[child] }
411 set[state.initial_child] = true
412 end
413 end
414 break if set.length == before
415 end
416 set
417 end

419 def complete?(state)
420 if state.compound?
421 state.child_states.any? { |child| child.kind == 'final' && @configuration[child] }
422 elsif state.parallel?
423 state.child_states.all? { |child| complete?(child) }
424 else
425 false
426 end
427 end

429 def enter(set)
430 @document.each do |state|
431 next unless set[state]
432 @configuration[state] = true
433 state.entered += 1
434 state.entry_time = @time
435 state.transitions.each { |transition| reserve(kind: :timer, transition: transition, delay: transition.delay) if transition.kind == :timer }
436 state.entries.each { |list| run_actions(list) }
437 next unless state.kind == 'final'
438 parent = state.parent
439 @queue << ["done.state.#{parent.name}", 0]
440 grandparent = parent.parent
441 @queue << ["done.state.#{grandparent.name}", 0] if grandparent.parallel? && grandparent.child_states.all? { |child| complete?(child) }
442 end
443 end

449 def trigger_name(trigger)
450 if trigger[:kind] == :always
451 'always'
452 elsif trigger[:kind] == :timer
453 transition = trigger[:transition]
454 "after(#{transition.delay})@#{transition.source.name}"
455 else
456 trigger[:event] + (trigger[:value].zero? ? '' : "=#{trigger[:value]}")
457 end

460 def microstep(selected, trigger)
461 exits = selected.each_with_object({}) { |candidate, set| set.merge!(candidate[:exits]) }
462 @document.each do |state|
463 next unless state.kind == 'history' && exits[state.parent]
464 @history[state] = @document.select do |active|
465 @configuration[active] && (state.history_kind == 'deep' ? active.atomic? && descendant?(active, state.parent) : active.parent == state.parent)
466 end
467 end
468 entries = {}
469 selected.each do |candidate|
470 transition = candidate[:transition]
471 next if transition.mode == :internal
472 effective_targets(transition).each do |target|
473 state = target
474 while descendant?(state, candidate[:scope])
475 entries[state] = true
476 state = state.parent
477 end
478 end
479 end
480 expand_entry(entries)
481 if @trace
482 sources = selected.map { |candidate| candidate[:transition].source }
483 puts "[t=#{@time}] #{trigger_name(trigger)} (#{list(sources)}): exit #{list(@document.reverse.select { |state| exits[state] })}; enter #{list(@document.select { |state| entries[state] })}"
484 end
485 @document.reverse_each do |state|
486 next unless exits[state]
487 state.exits.each { |actions| run_actions(actions) }
488 @configuration.delete(state)
489 state.dwell += @time - state.entry_time
490 @reservations.reject! { |r| r[:kind] == :timer && r[:transition].source == state }
491 end
492 selected.each do |candidate|
493 transition = candidate[:transition]
494 run_actions(transition.actions)
495 transition.source.taken += 1
496 end
497 enter(entries)
498 end

500 def macrostep(trigger)
501 @macrosteps += 1
502 count = 0
503 perform = lambda do |selected, current|
504 if count == 1000
505 puts "LIVELOCK t=#{@time}"
506 @halted = true
507 else
508 count += 1
509 @microsteps += 1
510 microstep(selected, current)
511 end
512 end
513 @data = trigger[:kind] == :event ? trigger[:value] : 0
514 if trigger[:kind] == :start
515 enter(expand_entry({ @root.initial_child => true }))
516 else
517 selected = select_transitions(trigger)
518 perform.call(selected, trigger) unless selected.empty?
519 end
520 until @halted
521 @data = 0
522 current = { kind: :always }
523 selected = select_transitions(current)
524 unless selected.empty?
525 perform.call(selected, current)
526 next
527 end
528 break if @queue.empty?
529 event, value = @queue.shift
530 @data = value
531 current = { kind: :event, event: event, value: value }
532 selected = select_transitions(current)
533 perform.call(selected, current) unless selected.empty?
534 end
535 end

537 def dump
538 puts "dump t=#{@time}"
539 @document.each { |state| puts ' ' * state.depth + state.name if @configuration[state] }
540 @variables.keys.sort.each { |name| puts " var #{name} = #{@variables[name]}" }
541 @document.each { |state| puts " history #{state.name} = #{list(@history[state])}" if @history.key?(state) }
542 @reservations.sort_by { |r| [r[:due], r[:number]] }.each do |reservation|
543 if reservation[:kind] == :timer
544 puts " timer #{reservation[:due]} #{trigger_name(reservation)}"
545 else
546 puts " send #{reservation[:due]} #{reservation[:event]} #{reservation[:value]}"
547 end
548 end
549 end

551 def run(script)
552 return unless validate
553 warn_unreachable
554 macrostep(kind: :start)
555 script.each do |line|
556 break if @halted
557 command, rest = line.split(' ', 2)
558 case command
559 when 'event'
560 name, value = rest.split(' ')
561 macrostep(kind: :event, event: name, value: value.to_i)
562 when 'tick'
563 finish = @time + rest.to_i
564 loop do
565 break if @halted
566 next_reservation = @reservations.select { |r| r[:due] <= finish }.min_by { |r| [r[:due], r[:number]] }
567 break unless next_reservation
568 @reservations.delete(next_reservation)
569 @time = next_reservation[:due]
570 trigger = next_reservation[:kind] == :timer ? next_reservation : { kind: :event, event: next_reservation[:event], value: next_reservation[:value] }
571 macrostep(trigger)
572 end
573 @time = finish unless @halted
574 when 'set'
575 name, value = rest.split(' ')
576 if @variables.key?(name)
577 @variables[name] = value.to_i
578 else
579 puts "error: unknown variable #{name}"
580 end
581 when 'trace'
582 @trace = rest == 'on'
583 when 'eval'
584 value = expression(rest)
585 reason = check_references(value.references)
586 if reason
587 puts "error: #{reason}"
588 else
589 @data = 0
590 begin
591 puts "#{rest} -> #{evaluate(value)}"
592 rescue EvaluationError
593 puts 'error: division by zero'
594 end
595 end
596 when 'dump'
597 dump
598 end
599 end
600 puts "end t=#{@time} macrosteps=#{@macrosteps} microsteps=#{@microsteps}"
601 @document.each do |state|
602 next if state.kind == 'history'
603 time = state.dwell + (@configuration[state] ? @time - state.entry_time : 0)
604 puts " #{state.name} entered=#{state.entered} taken=#{state.taken} time=#{time}"
605 end
606 end
607end
