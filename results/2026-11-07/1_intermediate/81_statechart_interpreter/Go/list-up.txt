func contains(items []string, value string) bool {
	for _, item := range items {
		if item == value {
			return true
		}
	}
	return false
}
type action struct {
	kind, name, text string
	delay            int64
	value            *expression
}

func parseActions(text string) []action {
	if text == "" {
		return nil
	}
	result := []action{}
	for _, item := range strings.Split(text, "; ") {
		var a action
		switch {
		case strings.HasPrefix(item, "log "):
			a.kind = "log"
			tail := item[4:]
			if tail[0] == '"' {
				end := strings.IndexByte(tail[1:], '"') + 1
				a.text = tail[1:end]
				if end+1 < len(tail) {
					a.value = parseExpression(tail[end+2:])
				}
		result = append(result, a)
	return result

func (c *chart) parseTransition(text string) *transition {
	parts := strings.SplitN(text, " ", 3)
	t := &transition{id: len(c.transitions), source: c.names[parts[1]], target: -1, mode: "internal"}
	tail := parts[2]
	bracket := 0
	for i := 0; i+2 < len(tail); i++ {
		if tail[i] == '[' {
			bracket++
		}
		if tail[i] == ']' {
			bracket--
		}
		if bracket == 0 && tail[i:i+3] == " / " {
			t.actions = parseActions(tail[i+3:])
			tail = tail[:i]
			break
		}
	}
	if start := strings.Index(tail, " ["); start >= 0 {
		end := strings.IndexByte(tail[start+2:], ']') + start + 2
		t.guard = parseExpression(tail[start+2 : end])
		tail = tail[:start] + tail[end+1:]
	}
	for _, arrow := range []string{" -> ", " ~> "} {
		if index := strings.Index(tail, arrow); index >= 0 {
	if tail == "always" {
		t.kind = "always"
	} else if strings.HasPrefix(tail, "after ") {
		t.kind = "timer"
		t.delay, _ = strconv.ParseInt(tail[6:], 10, 64)
	} else {
		t.kind = "event"
		if t.target != -2 {
			t.descriptors = strings.Split(tail, "|")
		}
	}
	return t

func matches(t *transition, tr trigger) bool {
	if t.kind != tr.kind {
		return false
	}
	if tr.kind == "always" {
		return true
	}
	if tr.kind == "timer" {
		return t == tr.timer
	}
	for _, descriptor := range t.descriptors {
		if descriptor == "*" || descriptor == tr.event.name || strings.HasPrefix(tr.event.name, descriptor+".") {
			return true
		}
	}
	return false
}

func (c *chart) selectTransitions(tr trigger) []selected {
	result := []selected{}
	seen := map[*transition]bool{}
	occupied := make([]bool, len(c.states))
	for _, atomic := range c.document {
		if !c.active[atomic] || len(c.states[atomic].childStates) != 0 {
			continue
		}
		var found *transition
		for id := atomic; id != 0 && found == nil; id = c.states[id].parent {
			for _, t := range c.states[id].transitions {
				if !matches(t, tr) {
					continue
				}
				value, ok := int64(1), true
				if t.guard != nil {
					value, ok = c.evaluate(t.guard)
				}
				if !ok {
					c.queue = append(c.queue, event{"error.execution", 0})
				} else if value != 0 {
					found = t
					break
				}
			}
		}
		if found == nil || seen[found] {
			continue
		}
		seen[found] = true
		item := c.transitionSet(found)
		conflict := false
		for _, id := range c.document {
			if item.exits[id] && occupied[id] {
				conflict = true
				break
			}
		}
		if !conflict {
			result = append(result, item)
			for _, id := range c.document {
				occupied[id] = occupied[id] || item.exits[id]
			}
		}
	}
	return result
