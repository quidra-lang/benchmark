12 class Node(string kind, string text, Node? left = null, Node? right = null)
13 {
14 public string? Reference(Chart chart)
15 {
16 if (kind == "variable") return text == "_data" || chart.Variables.ContainsKey(text) ? null : "unknown variable " + text;
17 if (kind == "in") return chart.States.ContainsKey(text) ? null : "unknown state " + text;
18 return left?.Reference(chart) ?? right?.Reference(chart);
19 }
20
21 public long Evaluate(Chart chart)
22 {
23 if (kind == "number") return long.Parse(text, CultureInfo.InvariantCulture);
24 if (kind == "variable") return text == "_data" ? chart.Data : chart.Variables[text];
25 if (kind == "in") return chart.States[text].Active ? 1 : 0;
26 long a = left!.Evaluate(chart);
27 if (kind == "unary") return text == "-" ? -a : a == 0 ? 1 : 0;
28 if (text == "&&" && a == 0) return 0;
29 if (text == "||" && a != 0) return 1;
30 long b = right!.Evaluate(chart);
31 if ((text == "/" || text == "%") && b == 0) throw new EvaluationError();
32 return text switch
33 {
34 "+" => a + b,
35 "-" => a - b,
36 "*" => a * b,
37 "/" => a / b,
38 "%" => a % b,
39 "==" => a == b ? 1 : 0,
40 "!=" => a != b ? 1 : 0,
41 "<" => a < b ? 1 : 0,
42 "<=" => a <= b ? 1 : 0,
43 ">" => a > b ? 1 : 0,
44 ">=" => a >= b ? 1 : 0,
45 "&&" or "||" => b != 0 ? 1 : 0,
46 _ => throw new InvalidOperationException()
47 };
48 }
49 }

51 class ExpressionParser
52 {
53 static readonly string[][] Operators = [["||"], ["&&"], ["==", "!=", "<", "<=", ">", ">="], ["+", "-"], ["*", "/", "%"]];
54 readonly List<string> tokens = [];
55 int position;
56
57 public ExpressionParser(string text)
58 {
59 for (int i = 0; i < text.Length;)
60 {
61 char c = text[i];
62 if (c == ' ')
63 {
64 i++;
65 continue;
66 }
67 int start = i;
68 if (char.IsAsciiLetterOrDigit(c) || c == '_')
69 {
70 while (i < text.Length && (char.IsAsciiLetterOrDigit(text[i]) || text[i] == '_')) i++;
71 }
72 else if (i + 1 < text.Length && new[] { "||", "&&", "==", "!=", "<=", ">=" }.Contains(text.Substring(i, 2))) i += 2;
73 else i++;
74 tokens.Add(text[start..i]);
75 }
76 }
77
78 public Node Parse() => Level(0);
79
80 Node Level(int precedence)
81 {
82 if (precedence == Operators.Length) return Unary();
83 Node result = Level(precedence + 1);
84 while (position < tokens.Count && Operators[precedence].Contains(tokens[position]))
85 {
86 string operation = tokens[position++];
87 result = new Node("binary", operation, result, Level(precedence + 1));
88 }
89 return result;
90 }
91
92 Node Unary()
93 {
94 string token = tokens[position++];
95 if (token is "!" or "-") return new Node("unary", token, Unary());
96 if (token == "(")
97 {
98 Node result = Level(0);
99 position++;
100 return result;
101 }
102 if (token == "in")
103 {
104 position++;
105 string state = tokens[position++];
106 position++;
107 return new Node("in", state);
108 }
109 return new Node(token[0] is >= '0' and <= '9' ? "number" : "variable", token);
110 }
111 }
112
113 static Node Expression(string text) => new ExpressionParser(text).Parse();

128 static List<Action> Actions(string text)
129 {
130 List<Action> result = [];
131 foreach (string raw in text.Split("; ", StringSplitOptions.None))
132 {
133 Action action = new();
134 string[] parts = raw.Split(' ', 2);
135 string first = parts[0], rest = parts.Length == 2 ? parts[1] : "";
136 switch (first)
137 {
138 case "raise":
139 case "send":
140 action.Kind = first;
141 string[] arguments = rest.Split(' ', first == "raise" ? 2 : 3);
142 action.Name = arguments[0];
143 if (first == "send")
144 {
145 action.Delay = int.Parse(arguments[1]);
146 if (arguments.Length == 3) action.Expression = Expression(arguments[2]);
147 }
148 else if (arguments.Length == 2) action.Expression = Expression(arguments[1]);
149 break;
150 case "cancel":
151 action.Kind = "cancel";
152 action.Name = rest;
153 break;
154 case "log":
155 action.Kind = "log";
156 if (rest.StartsWith('"'))
157 {
158 int end = rest.IndexOf('"', 1);
159 action.Literal = rest[1..end];
160 if (end + 1 < rest.Length) action.Expression = Expression(rest[(end + 2)..]);
161 }
162 else action.Expression = Expression(rest);
163 break;
164 default:
165 action.Kind = "assign";
166 action.Name = first;
167 action.Expression = Expression(rest[2..]);
168 break;
169 }
170 result.Add(action);
171 }
172 return result;
173 }

266 Transition ParseTransition(string text)
267 {
268 int end = text.IndexOf(' ', 3);
269 Transition transition = new() { Source = States[text[3..end]] };
270 string rest = text[(end + 1)..];
271 if (rest.StartsWith("after ", StringComparison.Ordinal))
272 {
273 int space = rest.IndexOf(' ', 6);
274 transition.Trigger = "timer";
275 transition.Delay = int.Parse(space < 0 ? rest[6..] : rest[6..space]);
276 rest = space < 0 ? "" : rest[(space + 1)..];
277 }
278 else
279 {
280 int space = rest.IndexOf(' ');
281 string trigger = space < 0 ? rest : rest[..space];
282 transition.Trigger = trigger == "always" ? "always" : "event";
283 transition.Descriptors = trigger.Split('|');
284 rest = space < 0 ? "" : rest[(space + 1)..];
285 }
286 if (rest.StartsWith('['))
287 {
288 int close = rest.IndexOf(']');
289 transition.Guard = Expression(rest[1..close]);
290 rest = rest[(close + 1)..].TrimStart(' ');
291 }
292 if (rest.StartsWith("-> ", StringComparison.Ordinal) || rest.StartsWith("~> ", StringComparison.Ordinal))
293 {
294 transition.Mode = rest[0] == '~' ? "local" : "external";
295 rest = rest[3..];
296 int space = rest.IndexOf(' ');
297 transition.TargetName = space < 0 ? rest : rest[..space];
298 rest = space < 0 ? "" : rest[(space + 1)..];
299 }
300 if (rest.StartsWith("/ ", StringComparison.Ordinal)) transition.Actions = Program.Actions(rest[2..]);
301 return transition;
302 }

304 public bool Define(List<Line> lines)
305 {
306 List<Definition> definitions = [];
307 foreach (Line line in lines)
308 {
309 Definition definition = new(line);
310 definitions.Add(definition);
311 string[] a = line.Text.Split(' ');
312 string kind = a[0];
313 if (kind is "state" or "parallel" or "final" or "history")
314 {
315 string? parentName = a.Length > 2 && a[2] != "initial" ? a[2] : null;
316 if (States.ContainsKey(a[1])) definition.Error = "duplicate state " + a[1];
317 else if (parentName != null && !States.ContainsKey(parentName)) definition.Error = "unknown state " + parentName;
318 else if (parentName != null && (States[parentName].Kind is "final" or "history")) definition.Error = parentName + " cannot have children";
319 else
320 {
321 State parent = parentName == null ? root : States[parentName];
322 State state = new(a[1], kind, parent);
323 definition.State = state;
324 States.Add(a[1], state);
325 parent.Children.Add(state);
326 if (kind == "state") for (int i = 2; i < a.Length; i++) if (a[i] == "initial") state.InitialName = a[i + 1];
327 if (kind == "history")
328 {
329 state.Deep = a[3] == "deep";
330 if (a.Length == 5) state.DefaultName = a[4];
331 }
332 }
333 }
334 else if (kind == "var")
335 {
336 if (Variables.ContainsKey(a[1])) definition.Error = "duplicate variable " + a[1];
337 else Variables.Add(a[1], long.Parse(a[2]));
338 }
339 }
340 foreach (Definition definition in definitions)
341 {
342 if (definition.Error != null) continue;
343 string text = definition.Line.Text;
344 string[] a = text.Split(' ');
345 string kind = a[0];
346 State? state = definition.State;
347 if (kind == "state" && state!.InitialName != null)
348 {
349 if (!States.TryGetValue(state.InitialName, out State? child)) definition.Error = "unknown state " + state.InitialName;
350 else if (child.Parent != state || child.Kind == "history") definition.Error = child.Name + " is not a child of " + state.Name;
351 }
352 else if (kind is "final" or "history")
353 {
354 if (state!.Parent!.Kind != "state" || !state.Parent.ChildStates.Any()) definition.Error = state.Name + " needs a compound parent";
355 else if (kind == "history" && state.DefaultName != null)
356 {
357 if (!States.TryGetValue(state.DefaultName, out State? target)) definition.Error = "unknown state " + state.DefaultName;
358 else if (!target.DescendantOf(state.Parent) || target.Kind == "history") definition.Error = target.Name + " is not a descendant of " + state.Parent.Name;
359 }
360 }
361 else if (kind is "entry" or "exit")
362 {
363 if (!States.TryGetValue(a[1], out State? source)) definition.Error = "unknown state " + a[1];
364 else if (source.Kind == "history") definition.Error = source.Name + " cannot have actions";
365 else
366 {
367 List<Action> list = Program.Actions(text[(text.IndexOf(" / ", StringComparison.Ordinal) + 3)..]);
368 definition.Error = FirstReference(list);
369 if (definition.Error == null) (kind == "entry" ? source.Entries : source.Exits).Add(list);
370 }
371 }
372 else if (kind == "on")
373 {
374 if (!States.TryGetValue(a[1], out State? source)) definition.Error = "unknown state " + a[1];
375 else if (source.Kind is "final" or "history") definition.Error = source.Name + " cannot have transitions";
376 else
377 {
378 Transition transition = ParseTransition(text);
379 string? error = transition.Guard?.Reference(this);
380 if (error == null && transition.TargetName != null && !States.ContainsKey(transition.TargetName)) error = "unknown state " + transition.TargetName;
381 error ??= FirstReference(transition.Actions);
382 definition.Error = error;
383 if (error == null)
384 {
385 transition.Target = transition.TargetName == null ? null : States[transition.TargetName];
386 source.Transitions.Add(transition);
387 }
388 }
389 }
390 }
391 bool valid = true;
392 foreach (Definition definition in definitions) if (definition.Error != null)
393 {
394 Console.WriteLine($"error line {definition.Line.Number}: {definition.Error}");
395 valid = false;
396 }
397 if (!valid) return false;
398 foreach (State child in root.Children) Visit(child);
399 return true;
400 }

402 public void Warnings()
403 {
404 HashSet<State> reachable = [Initial(root)];
405 bool changed;
406 do
407 {
408 int before = reachable.Count;
409 foreach (State state in reachable.ToArray())
410 {
411 if (state.Parent != root) reachable.Add(state.Parent!);
412 if (state.Compound) reachable.Add(Initial(state));
413 if (state.Parallel) reachable.UnionWith(state.ChildStates);
414 foreach (Transition transition in state.Transitions) if (transition.Target != null) reachable.Add(transition.Target.Kind == "history" ? DefaultTarget(transition.Target) : transition.Target);
415 }
416 changed = reachable.Count != before;
417 } while (changed);
418 foreach (State state in order) if (state.Kind != "history" && !reachable.Contains(state)) Console.WriteLine("warning: unreachable state " + state.Name);
419 }

421 IEnumerable<State> Effective(Transition transition)
422 {
423 State target = transition.Target!;
424 return target.Kind != "history" ? [target] : target.History ?? [DefaultTarget(target)];
425 }
426
427 Choice MakeChoice(Transition transition)
428 {
429 if (transition.Target == null) return new Choice(transition, null, []);
430 State[] targets = Effective(transition).ToArray();
431 State? scope = null;
432 if (transition.Mode == "local" && transition.Source.Compound && targets.All(s => s.DescendantOf(transition.Source))) scope = transition.Source;
433 else for (State? ancestor = transition.Source.Parent; ancestor != null; ancestor = ancestor.Parent) if (ancestor.Compound && targets.All(s => s.DescendantOf(ancestor)))
434 {
435 scope = ancestor;
436 break;
437 }
438 return new Choice(transition, scope, order.Where(s => s.Active && s.DescendantOf(scope)).ToHashSet());
439 }

441 static bool Matches(Transition transition, Trigger trigger)
442 {
443 if (trigger.Kind == "always") return transition.Trigger == "always";
444 if (trigger.Kind == "timer") return transition == trigger.Timer;
445 if (transition.Trigger != "event") return false;
446 string name = trigger.Event!.Name;
447 return transition.Descriptors.Any(d => d == "*" || d == name || name.StartsWith(d + ".", StringComparison.Ordinal));
448 }
449
450 List<Choice> Select(Trigger trigger)
451 {
452 List<Transition> candidates = [];
453 HashSet<Transition> seen = [];
454 foreach (State atomic in order.Where(s => s.Active && s.Atomic))
455 {
456 Transition? selected = null;
457 for (State? state = atomic; state != root && selected == null; state = state!.Parent)
458 {
459 foreach (Transition transition in state!.Transitions) if (Matches(transition, trigger))
460 {
461 bool guard;
462 try
463 {
464 guard = transition.Guard == null || transition.Guard.Evaluate(this) != 0;
465 }
466 catch (EvaluationError)
467 {
468 queue.Enqueue(new Event("error.execution", 0));
469 guard = false;
470 }
471 if (guard)
472 {
473 selected = transition;
474 break;
475 }
476 }
477 }
478 if (selected != null && seen.Add(selected)) candidates.Add(selected);
479 }
480 List<Choice> result = [];
481 HashSet<State> exited = [];
482 foreach (Transition transition in candidates)
483 {
484 Choice candidate = MakeChoice(transition);
485 if (!candidate.Exits.Overlaps(exited))
486 {
487 result.Add(candidate);
488 exited.UnionWith(candidate.Exits);
489 }
490 }
491 return result;
492 }

494 void Reserve(Transition? timer, Event? @event, long delay) => reservations.Add(new Reservation { Timer = timer, Event = @event, Due = time + delay, Number = ++creations });
495
496 void Perform(List<Action> list)
497 {
498 foreach (Action action in list)
499 {
500 long value;
501 try
502 {
503 value = action.Expression?.Evaluate(this) ?? 0;
504 }
505 catch (EvaluationError)
506 {
507 queue.Enqueue(new Event("error.execution", 0));
508 return;
509 }
510 switch (action.Kind)
511 {
512 case "assign":
513 Variables[action.Name] = value;
514 break;
515 case "raise":
516 queue.Enqueue(new Event(action.Name, value));
517 break;
518 case "send":
519 Reserve(null, new Event(action.Name, value), action.Delay);
520 break;
521 case "cancel":
522 reservations.RemoveAll(r => r.Event != null && r.Event.Name == action.Name);
523 break;
524 case "log":
525 Console.WriteLine(action.Literal + (action.Expression == null ? "" : value.ToString(CultureInfo.InvariantCulture)));
526 break;
527 }
528 }
529 }

531 void Expand(HashSet<State> entry)
532 {
533 bool changed;
534 do
535 {
536 int before = entry.Count;
537 foreach (State state in order) if (entry.Contains(state))
538 {
539 if (state.Parallel) entry.UnionWith(state.ChildStates);
540 if (state.Compound && !state.ChildStates.Any(entry.Contains)) entry.Add(Initial(state));
541 }
542 changed = entry.Count != before;
543 } while (changed);
544 }
545
546 bool Complete(State state)
547 {
548 if (state.Compound) return state.ChildStates.Any(s => s.Kind == "final" && s.Active);
549 return state.Parallel && state.ChildStates.All(Complete);
550 }
551
552 void Enter(HashSet<State> entry)
553 {
554 foreach (State state in order) if (entry.Contains(state))
555 {
556 state.Active = true;
557 state.Entered++;
558 state.EntryTime = time;
559 foreach (Transition transition in state.Transitions) if (transition.Trigger == "timer") Reserve(transition, null, transition.Delay);
560 foreach (List<Action> list in state.Entries) Perform(list);
561 if (state.Kind == "final")
562 {
563 State parent = state.Parent!;
564 queue.Enqueue(new Event("done.state." + parent.Name, 0));
565 State? grandparent = parent.Parent;
566 if (grandparent != null && grandparent.Parallel && grandparent.ChildStates.All(Complete)) queue.Enqueue(new Event("done.state." + grandparent.Name, 0));
567 }
568 }

584 bool Micro(List<Choice> selected, Trigger trigger)
585 {
586 if (inMacro == 1000)
587 {
588 Console.WriteLine($"LIVELOCK t={time}");
589 halted = true;
590 return false;
591 }
592 inMacro++;
593 microsteps++;
594 HashSet<State> exits = selected.SelectMany(c => c.Exits).ToHashSet();
595 foreach (State history in order) if (history.Kind == "history" && exits.Contains(history.Parent!)) history.History = order.Where(s => s.Active && (history.Deep ? s.Atomic && s.DescendantOf(history.Parent) : s.Parent == history.Parent && s.Kind != "history")).ToList();
596 HashSet<State> entry = [];
597 foreach (Choice choice in selected) if (choice.Transition.Target != null) foreach (State target in Effective(choice.Transition)) for (State? state = target; state != choice.Scope; state = state!.Parent) entry.Add(state!);
598 Expand(entry);
599 List<State> exitOrder = order.Where(exits.Contains).Reverse().ToList();
600 if (trace) Console.WriteLine($"[t={time}] {Display(trigger)} ({Names(selected.Select(c => c.Transition.Source))}): exit {Names(exitOrder)}; enter {Names(order.Where(entry.Contains))}");
601 foreach (State state in exitOrder)
602 {
603 foreach (List<Action> list in state.Exits) Perform(list);
604 state.Active = false;
605 state.Dwell += time - state.EntryTime;
606 reservations.RemoveAll(r => r.Timer != null && r.Timer.Source == state);
607 }
608 foreach (Choice choice in selected)
609 {
610 Perform(choice.Transition.Actions);
611 choice.Transition.Source.Taken++;
612 }
613 Enter(entry);
614 return true;
615 }

617 public void Macro(Trigger trigger)
618 {
619 macrosteps++;
620 inMacro = 0;
621 if (trigger.Kind == "start")
622 {
623 Data = 0;
624 HashSet<State> entry = [Initial(root)];
625 Expand(entry);
626 Enter(entry);
627 }
628 else
629 {
630 Data = trigger.Kind == "timer" ? 0 : trigger.Event!.Value;
631 List<Choice> selected = Select(trigger);
632 if (selected.Count > 0 && !Micro(selected, trigger)) return;
633 }
634 while (!halted)
635 {
636 Data = 0;
637 Trigger always = new("always");
638 List<Choice> selected = Select(always);
639 if (selected.Count > 0)
640 {
641 if (!Micro(selected, always)) return;
642 continue;
643 }
644 if (queue.Count == 0) return;
645 Event @event = queue.Dequeue();
646 Data = @event.Value;
647 Trigger internalTrigger = new("event", @event);
648 selected = Select(internalTrigger);
649 if (selected.Count > 0 && !Micro(selected, internalTrigger)) return;
650 }
651 }

653 IEnumerable<Reservation> OrderedReservations() => reservations.OrderBy(r => r.Due).ThenBy(r => r.Number);

655 public void Script(string text)
656 {
657 if (halted) return;
658 string[] a = text.Split(' ', 3);
659 switch (a[0])
660 {
661 case "event":
662 Macro(new Trigger("event", new Event(a[1], a.Length == 3 ? long.Parse(a[2]) : 0)));
663 break;
664 case "tick":
665 long finish = time + long.Parse(a[1]);
666 while (!halted)
667 {
668 Reservation? next = OrderedReservations().FirstOrDefault(r => r.Due <= finish);
669 if (next == null) break;
670 reservations.Remove(next);
671 time = next.Due;
672 Macro(next.Timer == null ? new Trigger("event", next.Event) : new Trigger("timer", Timer: next.Timer));
673 }
674 if (!halted) time = finish;
675 break;
676 case "set":
677 if (!Variables.ContainsKey(a[1])) Console.WriteLine("error: unknown variable " + a[1]);
678 else Variables[a[1]] = long.Parse(a[2]);
679 break;
680 case "trace":
681 trace = a[1] == "on";
682 break;
683 case "eval":
684 string raw = text[5..];
685 Node node = Expression(raw);
686 string? error = node.Reference(this);
687 if (error != null) Console.WriteLine("error: " + error);
688 else
689 {
690 Data = 0;
691 try
692 {
693 Console.WriteLine(raw + " -> " + node.Evaluate(this));
694 }
695 catch (EvaluationError)
696 {
697 Console.WriteLine("error: division by zero");
698 }
699 }
700 break;
701 case "dump":
702 Console.WriteLine($"dump t={time}");
703 foreach (State state in order) if (state.Active) Console.WriteLine(new string(' ', 2 * state.Depth) + state.Name);
704 foreach (var variable in Variables) Console.WriteLine($" var {variable.Key} = {variable.Value}");
705 foreach (State state in order) if (state.Kind == "history" && state.History != null) Console.WriteLine($" history {state.Name} = {Names(state.History)}");
706 foreach (Reservation reservation in OrderedReservations())
707 {
708 if (reservation.Timer != null) Console.WriteLine($" timer {reservation.Due} after({reservation.Timer.Delay})@{reservation.Timer.Source.Name}");
709 else Console.WriteLine($" send {reservation.Due} {reservation.Event!.Name} {reservation.Event.Value}");
710 }
711 break;
712 }
713 }
