10TOKEN = re.compile(r"\d+|\w+|\|\||&&|[=!<>]=|[-+*/%!()<>]")

53class ExpressionParser:
54 def __init__(self, text):
55 self.tokens = TOKEN.findall(text)
56 self.position = 0
57
58 def peek(self):
59 if self.position < len(self.tokens):
60 return self.tokens[self.position]
61 return None
62
63 def advance(self):
64 token = self.tokens[self.position]
65 self.position += 1
66 return token
67
68 def parse(self, level=0):
69 if level == len(PRECEDENCE):
70 return self.parse_unary()
71 node = self.parse(level + 1)
72 while self.peek() in PRECEDENCE[level]:
73 op = self.advance()
74 node = (op, node, self.parse(level + 1))
75 return node
76
77 def parse_unary(self):
78 token = self.advance()
79 if token == "!":
80 return ("!", self.parse_unary())
81 if token == "-":
82 return ("neg", self.parse_unary())
83 if token == "(":
84 node = self.parse()
85 self.advance()
86 return node
87 if token.isdigit():
88 return ("num", int(token))
89 if token == "_data":
90 return ("data",)
91 if token == "in":
92 self.advance()
93 name = self.advance()
94 self.advance()
95 return ("in", name)
96 return ("var", token)

99def parse_expression(text):
100 return ExpressionParser(text).parse()

103def parse_optional_expression(text):
104 return parse_expression(text) if text else None

107def expression_references(node):
108 match node:
109 case ("var", name):
110 yield "variable", name
111 case ("in", name):
112 yield "state", name
113 case ("!" | "neg", operand):
114 yield from expression_references(operand)
115 case (_, left, right):
116 yield from expression_references(left)
117 yield from expression_references(right)

119
120def apply_operator(op, left, right):
121 if op not in ("/", "%"):
122 return int(OPERATORS[op](left, right))
123 if right == 0:
124 raise EvaluationError
125 quotient = abs(left) // abs(right)
126 if (left < 0) != (right < 0):
127 quotient = -quotient
128 return quotient if op == "/" else left - quotient * right

131@dataclass
132class Action:
133 kind: str
134 name: str = ""
135 expression: tuple | None = None

140def parse_action(text):
141 word, _, rest = text.partition(" ")
142 match word:
143 case "raise":
144 event, _, expression = rest.partition(" ")
145 return Action("raise", event, parse_optional_expression(expression))
146 case "send":
147 event, _, rest = rest.partition(" ")
148 delay, _, expression = rest.partition(" ")
149 return Action("send", event, parse_optional_expression(expression), int(delay))
150 case "cancel":
151 return Action("cancel", rest)
152 case "log" if rest.startswith('"'):
153 end = rest.index('"', 1)
154 expression = parse_optional_expression(rest[end + 2 :])
155 return Action("log", expression=expression, text=rest[1:end])
156 case "log":
157 return Action("log", expression=parse_expression(rest))
158 case _:
159 return Action("assign", word, parse_expression(rest.removeprefix("= ")))
160
161
162def parse_actions(text):
163 return [parse_action(part) for part in text.split("; ")]
164
165
166def action_references(actions):
167 for action in actions:
168 if action.kind == "assign":
169 yield "variable", action.name
170 if action.expression is not None:
171 yield from expression_references(action.expression)

270def complete_entry_set(states):
271 entering = set(states)
272 pending = list(entering)
273 while pending:
274 state = pending.pop()
275 if state.is_parallel:
276 children = state.child_states
277 elif state.is_compound and entering.isdisjoint(state.child_states):
278 children = [state.initial_child]
279 else:
280 continue
281 for child in children:
282 if child not in entering:
283 entering.add(child)
284 pending.append(child)
285 return entering

288class ChartDefinition:
289 def __init__(self, lines):
290 self.root = State("", "state", None, 0)
291 self.states = {}
292 self.variables = {}
293 self.errors = {}
294 self.ordered = []
295 deferred = []
296 for number, text in lines:
297 kind = text.partition(" ")[0]
298 if kind in STATE_KINDS:
299 self.record(number, self.define_state(text.split(" "), number))
300 elif kind == "var":
301 self.record(number, self.declare_variable(text))
302 else:
303 deferred.append((number, kind, text))
304 self.number_states(self.root)
305 for state in self.states.values():
306 self.record(state.line, self.check_state(state))
307 for number, kind, text in deferred:
308 if kind == "on":
309 self.record(number, self.add_transition(text))
310 else:
311 self.record(number, self.add_actions(kind, text))
312
313 def record(self, number, error):
314 if error is not None:
315 self.errors[number] = error
316
317 def define_state(self, words, number):
318 kind, name, *rest = words
319 parent_name = rest.pop(0) if rest and rest[0] != "initial" else None
320 if name in self.states:
321 return f"duplicate state {name}"
322 parent = self.root
323 if parent_name is not None:
324 parent = self.states.get(parent_name)
325 if parent is None:
326 return f"unknown state {parent_name}"
327 if parent.kind in ("final", "history"):
328 return f"{parent_name} cannot have children"
329 state = State(name, kind, parent, number)
330 if kind == "history":
331 state.deep = rest[0] == "deep"
332 state.default_name = rest[1] if len(rest) > 1 else None
333 elif rest:
334 state.initial_name = rest[1]
335 parent.children.append(state)
336 self.states[name] = state
337 return None
338
339 def declare_variable(self, text):
340 _, name, value = text.split(" ")
341 if name in self.variables:
342 return f"duplicate variable {name}"
343 self.variables[name] = int(value)
344 return None
345
346 def number_states(self, parent):
347 for child in parent.children:
348 child.order = len(self.ordered)
349 child.depth = parent.depth + 1
350 self.ordered.append(child)
351 self.number_states(child)
352
353 def check_state(self, state):
354 if state.initial_name is not None:
355 child = self.states.get(state.initial_name)
356 if child is None:
357 return f"unknown state {state.initial_name}"
358 if child.parent is not state or child.kind == "history":
359 return f"{child.name} is not a child of {state.name}"
360 state.initial = child
361 if state.kind in ("final", "history") and not state.parent.is_compound:
362 return f"{state.name} needs a compound parent"
363 if state.default_name is not None:
364 default = self.states.get(state.default_name)
365 if default is None:
366 return f"unknown state {state.default_name}"
367 if default.kind == "history" or not default.is_descendant_of(state.parent):
368 return f"{default.name} is not a descendant of {state.parent.name}"
369 state.default = default
370 return None
371
372 def find_unknown(self, references):
373 for kind, name in references:
374 if kind == "variable" and name not in self.variables:
375 return f"unknown variable {name}"
376 if kind == "state" and name not in self.states:
377 return f"unknown state {name}"
378 return None
379
380 def add_actions(self, kind, text):
381 _, name, _, body = text.split(" ", 3)
382 state = self.states.get(name)
383 if state is None:
384 return f"unknown state {name}"
385 if state.kind == "history":
386 return f"{name} cannot have actions"
387 actions = parse_actions(body)
388 error = self.find_unknown(action_references(actions))
389 if error is None:
390 action_lists = state.entry_actions if kind == "entry" else state.exit_actions
391 action_lists.append(actions)
392 return error
393
394 def add_transition(self, text):
395 match = ON_LINE.fullmatch(text)
396 source_name, trigger, delay, guard_text, arrow, target_name, actions_text = match.groups()
397 source = self.states.get(source_name)
398 if source is None:
399 return f"unknown state {source_name}"
400 if source.kind in ("final", "history"):
401 return f"{source_name} cannot have transitions"
402 guard = parse_optional_expression(guard_text)
403 actions = parse_actions(actions_text) if actions_text else []
404 references = []
405 if guard is not None:
406 references.extend(expression_references(guard))
407 if target_name is not None:
408 references.append(("state", target_name))
409 references.extend(action_references(actions))
410 error = self.find_unknown(references)
411 if error is not None:
412 return error
413 if trigger == "always":
414 kind = "always"
415 elif delay is not None:
416 kind = "after"
417 else:
418 kind = "event"
419 transition = Transition(
420 source=source,
421 kind=kind,
422 descriptors=trigger.split("|") if kind == "event" else [],
423 delay=int(delay or 0),
424 guard=guard,
425 target=self.states[target_name] if target_name else None,
426 local=arrow == "~",
427 actions=actions,
428 )
429 source.transitions.append(transition)
430 return None
431
432 def unreachable_states(self):
433 reachable = set()
434 pending = [self.root.initial_child]
435 while pending:
436 state = pending.pop()
437 if state is self.root or state in reachable:
438 continue
439 reachable.add(state)
440 pending.append(state.parent)
441 if state.is_compound:
442 pending.append(state.initial_child)
443 elif state.is_parallel:
444 pending.extend(state.child_states)
445 for transition in state.transitions:
446 target = transition.target
447 if target is not None:
448 pending.append(target.default_target if target.kind == "history" else target)
449 return [s for s in self.ordered if s.kind != "history" and s not in reachable]

452class Machine:
453 def __init__(self, chart):
454 self.chart = chart
455 self.variables = dict(chart.variables)
456 self.configuration = set()
457 self.current_value = 0
458 self.queue = deque()
459 self.reservations = []
460 self.reservations_created = 0
461 self.history = {}
462 self.time = 0
463 self.trace = False
464 self.halted = False
465 self.macrosteps = 0
466 self.microsteps = 0
467 self.steps_in_macrostep = 0
468 self.entered = Counter()
469 self.taken = Counter()
470 self.dwell = Counter()
471 self.entry_time = {}

473 def evaluate(self, node):
474 match node:
475 case ("num", value):
476 return value
477 case ("var", name):
478 return self.variables[name]
479 case ("data",):
480 return self.current_value
481 case ("in", name):
482 return int(self.chart.states[name] in self.configuration)
483 case ("!", operand):
484 return int(self.evaluate(operand) == 0)
485 case ("neg", operand):
486 return -self.evaluate(operand)
487 case ("&&", left, right):
488 return int(self.evaluate(left) != 0 and self.evaluate(right) != 0)
489 case ("||", left, right):
490 return int(self.evaluate(left) != 0 or self.evaluate(right) != 0)
491 case (op, left, right):
492 return apply_operator(op, self.evaluate(left), self.evaluate(right))
493
494 def run_actions(self, actions):
495 for action in actions:
496 try:
497 self.run_action(action)
498 except EvaluationError:
499 self.queue.append(ERROR_EVENT)
500 return
501
502 def run_action(self, action):
503 value = 0 if action.expression is None else self.evaluate(action.expression)
504 match action.kind:
505 case "assign":
506 self.variables[action.name] = value
507 case "raise":
508 self.queue.append(Event(action.name, value))
509 case "send":
510 self.reserve(action.delay, event=Event(action.name, value))
511 case "cancel":
512 self.reservations = [
513 reservation
514 for reservation in self.reservations
515 if reservation.event is None or reservation.event.name != action.name
516 ]
517 case "log":
518 print(action.text if action.expression is None else f"{action.text}{value}")
519
520 def reserve(self, delay, transition=None, event=None):
521 self.reservations_created += 1
522 reservation = Reservation(self.time + delay, self.reservations_created, transition, event)
523 self.reservations.append(reservation)
524
525 def guard_holds(self, transition):
526 if transition.guard is None:
527 return True
528 try:
529 return self.evaluate(transition.guard) != 0
530 except EvaluationError:
531 self.queue.append(ERROR_EVENT)
532 return False
533
534 def find_transition(self, atomic, matches):
535 state = atomic
536 while state is not self.chart.root:
537 for transition in state.transitions:
538 if matches(transition) and self.guard_holds(transition):
539 return transition
540 state = state.parent
541 return None

543 def select(self, matches):
544 candidates = []
545 for atomic in document_order(s for s in self.configuration if s.is_atomic):
546 transition = self.find_transition(atomic, matches)
547 if transition is not None and transition not in candidates:
548 candidates.append(transition)
549 selected = []
550 exited = set()
551 for transition in candidates:
552 exit_set = self.exit_set(self.scope(transition))
553 if exited.isdisjoint(exit_set):
554 selected.append(transition)
555 exited |= exit_set
556 return selected
557
558 def effective_targets(self, transition):
559 target = transition.target
560 if target.kind != "history":
561 return [target]
562 return self.history.get(target, [target.default_target])
563
564 def scope(self, transition):
565 if transition.target is None:
566 return None
567 targets = self.effective_targets(transition)
568 source = transition.source
569 if transition.local and source.is_compound and source.encloses(targets):
570 return source
571 return next(
572 ancestor
573 for ancestor in source.ancestors()
574 if ancestor.is_compound and ancestor.encloses(targets)
575 )
576
577 def exit_set(self, scope):
578 if scope is None:
579 return set()
580 return {state for state in self.configuration if state.is_descendant_of(scope)}
581
582 def is_complete(self, state):
583 if state.is_compound:
584 return any(
585 child.kind == "final" and child in self.configuration for child in state.children
586 )
587 if state.is_parallel:
588 return all(self.is_complete(child) for child in state.child_states)
589 return False

591 def record_history(self, exiting):
592 for state in exiting:
593 for history in state.children:
594 if history.kind != "history":
595 continue
596 if history.deep:
597 recorded = [
598 s for s in self.configuration if s.is_atomic and s.is_descendant_of(state)
599 ]
600 else:
601 recorded = [s for s in state.child_states if s in self.configuration]
602 self.history[history] = document_order(recorded)

604 def microstep(self, display_name, transitions):
605 scopes = [self.scope(transition) for transition in transitions]
606 exiting = set()
607 for scope in scopes:
608 exiting |= self.exit_set(scope)
609 self.record_history(exiting)
610 entering = set()
611 for transition, scope in zip(transitions, scopes):
612 if scope is None:
613 continue
614 for target in self.effective_targets(transition):
615 entering.add(target)
616 entering.update(takewhile(lambda state: state is not scope, target.ancestors()))
617 exit_order = document_order(exiting)[::-1]
618 entry_order = document_order(complete_entry_set(entering))
619 if self.trace:
620 sources = join_names(transition.source for transition in transitions)
621 print(
622 f"[t={self.time}] {display_name} ({sources}): "
623 f"exit {join_names(exit_order)}; enter {join_names(entry_order)}"
624 )
625 for state in exit_order:
626 self.exit_state(state)
627 for transition in transitions:
628 self.run_actions(transition.actions)
629 self.taken[transition.source] += 1
630 self.enter(entry_order)
631
632 def exit_state(self, state):
633 for actions in state.exit_actions:
634 self.run_actions(actions)
635 self.configuration.remove(state)
636 self.dwell[state] += self.time - self.entry_time[state]
637 self.reservations = [r for r in self.reservations if not r.is_timer_of(state)]
638
639 def enter(self, states):
640 for state in states:
641 self.configuration.add(state)
642 self.entered[state] += 1
643 self.entry_time[state] = self.time
644 for transition in state.transitions:
645 if transition.kind == "after":
646 self.reserve(transition.delay, transition=transition)
647 for actions in state.entry_actions:
648 self.run_actions(actions)
649 if state.kind == "final":
650 parent = state.parent
651 self.queue.append(Event(f"done.state.{parent.name}"))
652 grandparent = parent.parent
653 if grandparent.is_parallel and self.is_complete(grandparent):
654 self.queue.append(Event(f"done.state.{grandparent.name}"))
655
656 def take_transitions(self, display_name, matches):
657 transitions = self.select(matches)
658 if not transitions:
659 return False
660 if self.steps_in_macrostep == LIVELOCK_LIMIT:
661 print(f"LIVELOCK t={self.time}")
662 self.halted = True
663 else:
664 self.steps_in_macrostep += 1
665 self.microsteps += 1
666 self.microstep(display_name, transitions)
667 return True

669 def process_event(self, event):
670 self.current_value = event.value
671 self.take_transitions(event.display_name, lambda t: t.handles(event.name))
672
673 def fire_timer(self, timer):
674 self.current_value = 0
675 self.take_transitions(timer.timer_name, lambda t: t is timer)
676
677 def enter_initial(self):
678 self.current_value = 0
679 self.enter(document_order(complete_entry_set([self.chart.root.initial_child])))
680
681 def macrostep(self, first_step):
682 self.macrosteps += 1
683 self.steps_in_macrostep = 0
684 first_step()
685 while not self.halted:
686 self.current_value = 0
687 if self.take_transitions("always", lambda t: t.kind == "always"):
688 continue
689 if not self.queue:
690 break
691 self.process_event(self.queue.popleft())
692
693 def tick(self, amount):
694 limit = self.time + amount
695 while not self.halted:
696 due = [r for r in self.reservations if r.due <= limit]
697 if not due:
698 break
699 reservation = min(due, key=RESERVATION_ORDER)
700 self.reservations.remove(reservation)
701 self.time = reservation.due
702 if reservation.event is None:
703 self.macrostep(partial(self.fire_timer, reservation.transition))
704 else:
705 self.macrostep(partial(self.process_event, reservation.event))
706 if not self.halted:
707 self.time = limit

709 def execute(self, command):
710 word, _, argument = command.partition(" ")
711 match word:
712 case "event":
713 name, _, value = argument.partition(" ")
714 self.macrostep(partial(self.process_event, Event(name, int(value or 0))))
715 case "tick":
716 self.tick(int(argument))
717 case "set":
718 name, value = argument.split(" ")
719 if name in self.variables:
720 self.variables[name] = int(value)
721 else:
722 print(f"error: unknown variable {name}")
723 case "trace":
724 self.trace = argument == "on"
725 case "eval":
726 self.evaluate_command(argument)
727 case "dump":
728 self.dump()

730 def evaluate_command(self, text):
731 expression = parse_expression(text)
732 error = self.chart.find_unknown(expression_references(expression))
733 if error is not None:
734 print(f"error: {error}")
735 return
736 self.current_value = 0
737 try:
738 value = self.evaluate(expression)
739 except EvaluationError:
740 print("error: division by zero")
741 else:
742 print(f"{text} -> {value}")

744 def dump(self):
745 print(f"dump t={self.time}")
746 for state in document_order(self.configuration):
747 print(" " * state.depth + state.name)
748 for name in sorted(self.variables):
749 print(f" var {name} = {self.variables[name]}")
750 for state in self.chart.ordered:
751 if state in self.history:
752 print(f" history {state.name} = {join_names(self.history[state])}")
753 for reservation in sorted(self.reservations, key=RESERVATION_ORDER):
754 if reservation.event is None:
755 print(f" timer {reservation.due} {reservation.transition.timer_name}")
756 else:
757 event = reservation.event
758 print(f" send {reservation.due} {event.name} {event.value}")
759
760 def print_statistics(self):
761 print(f"end t={self.time} macrosteps={self.macrosteps} microsteps={self.microsteps}")
762 for state in self.chart.ordered:
763 if state.kind == "history":
764 continue
765 dwell = self.dwell[state]
766 if state in self.configuration:
767 dwell += self.time - self.entry_time[state]
768 print(
769 f" {state.name} entered={self.entered[state]} "
770 f"taken={self.taken[state]} time={dwell}"
771 )
