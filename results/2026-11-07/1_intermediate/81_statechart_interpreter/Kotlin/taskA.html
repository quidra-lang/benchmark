7private class Expression(text: String, val variables: Map<String, Long>, val states: Map<String, Int>) {
8 val tokens = mutableListOf<String>()
9 var at = 0
10 init {
11 var i = 0
12 while (i < text.length) {
13 if (text[i] == ' ') {
14 i++
15 continue
16 }
17 val start = i
18 if (text[i] in '0'..'9') {
19 while (i < text.length && text[i] in '0'..'9') i++
20 } else if (text[i].isLetter() || text[i] == '_') {
21 while (i < text.length && (text[i].isLetterOrDigit() || text[i] == '_')) i++
22 } else {
23 i++
24 if (i < text.length && text.substring(start, i + 1) in setOf("||", "&&", "==", "!=", "<=", ">=")) i++
25 }
26 tokens.add(text.substring(start, i))
27 }
28 }
29
30 fun parse(level: Int = 0): Expr {
31 if (level == 5) {
32 val t = tokens[at++]
33 if (t == "!" || t == "-") return Expr("unary$t", args = listOf(parse(5)))
34 if (t == "(") {
35 val e = parse()
36 at++
37 return e
38 }
39 if (t[0] in '0'..'9') return Expr("number", t)
40 if (t == "in") {
41 at++
42 val name = tokens[at++]
43 if (name !in states) throw Reference("unknown state $name")
44 at++
45 return Expr("in", name)
46 }
47 if (t != "_data" && t !in variables) throw Reference("unknown variable $t")
48 return Expr("variable", t)
49 }
50 val operators = when (level) {
51 0 -> setOf("||")
52 1 -> setOf("&&")
53 2 -> setOf("==", "!=", "<", "<=", ">", ">=")
54 3 -> setOf("+", "-")
55 else -> setOf("*", "/", "%")
56 }
57 var left = parse(level + 1)
58 while (at < tokens.size && tokens[at] in operators) {
59 val op = tokens[at++]
60 left = Expr(op, args = listOf(left, parse(level + 1)))
61 }
62 return left
63 }
64}

133 fun actions(text: String): List<Action> = text.split("; ").map {
134 raw ->
135 val w = raw.split(' ')
136 when (w[0]) {
137 "raise" -> Action("raise", w[1], if (w.size > 2) expression(w.drop(2).joinToString(" ")) else null)
138 "send" -> Action("send", w[1], if (w.size > 3) expression(w.drop(3).joinToString(" ")) else null, delay = w[2].toInt())
139 "cancel" -> Action("cancel", w[1])
140 "log" -> {
141 val rest = raw.drop(4)
142 if (rest.startsWith('"')) {
143 val end = rest.indexOf('"', 1)
144 Action("log", expr = if (end + 1 < rest.length) expression(rest.drop(end + 2)) else null, text = rest.substring(1, end))
145 } else Action("log", expr = expression(rest))
146 } else -> {
147 if (w[0] !in variables) throw Reference("unknown variable ${w[0]}")
148 Action("assign", w[0], expression(w.drop(2).joinToString(" ")))
149 }
150 }

153 fun define(lines: List<Pair<Int, String>>): Boolean {
154 val errors = sortedMapOf<Int, String>()
155 val definitions = mutableListOf<Definition>()
156 for ((line, text) in lines) {
157 val w = text.split(' ')
158 var node: Int? = null
159 if (w[0] in setOf("state", "parallel", "final", "history")) {
160 val parentName = if (w.size > 2 && w[2] != "initial") w[2] else null
161 val reason = when {
162 w[1] in ids -> "duplicate state ${w[1]}"
163 parentName != null && parentName !in ids -> "unknown state $parentName"
164 parentName != null && states[ids.getValue(parentName)].kind in setOf("final", "history") -> "$parentName cannot have children"
165 else -> null
166 }
167 if (reason != null) errors[line] = reason else {
168 val parent = parentName?.let {
169 ids.getValue(it)
170 } ?: 0
171 val initial = w.indexOf("initial")
172 node = states.size
173 ids[w[1]] = node
174 val s = State(w[1], w[0], parent, if (initial >= 0) w[initial + 1] else null, w[0] == "history" && w[3] == "deep", if (w[0] == "history" && w.size == 5) w[4] else null)
175 s.depth = states[parent].depth + 1
176 states.add(s)
177 states[parent].children.add(node)
178 }
179 } else if (w[0] == "var") {
180 if (w[1] in variables) errors[line] = "duplicate variable ${w[1]}" else variables[w[1]] = w[2].toLong()
181 }
182 definitions.add(Definition(line, text, node))
183 }
184 for (d in definitions) {
185 if (d.line in errors) continue
186 val w = d.text.split(' ')
187 try {
188 if (w[0] == "state" && d.node != null && states[d.node].initialName != null) {
189 val s = states[d.node]
190 val child = ids[s.initialName] ?: throw Reference("unknown state ${s.initialName}")
191 if (states[child].parent != d.node || states[child].kind == "history") throw Reference("${s.initialName} is not a child of ${s.id}")
192 }
193 if (w[0] == "final" || w[0] == "history") {
194 val s = states[d.node!!]
195 val p = states[s.parent]
196 if (p.kind != "state" || children(s.parent).isEmpty()) throw Reference("${s.id} needs a compound parent")
197 if (s.defaultName != null) {
198 val target = ids[s.defaultName] ?: throw Reference("unknown state ${s.defaultName}")
199 if (!descendant(target, s.parent) || states[target].kind == "history") throw Reference("${s.defaultName} is not a descendant of ${p.id}")
200 }
201 }
202 if (w[0] == "entry" || w[0] == "exit") {
203 val i = ids[w[1]] ?: throw Reference("unknown state ${w[1]}")
204 val s = states[i]
205 if (s.kind == "history") throw Reference("${s.id} cannot have actions")
206 val a = actions(d.text.substringAfter(" / "))
207 if (w[0] == "entry") s.entries.add(a) else s.exits.add(a)
208 }
209 if (w[0] == "on") {
210 val source = ids[w[1]] ?: throw Reference("unknown state ${w[1]}")
211 if (states[source].kind in setOf("final", "history")) throw Reference("${w[1]} cannot have transitions")
212 var rest = d.text.drop(4 + w[1].length)
213 var trigger: String
214 var delay = 0
215 var descriptors = emptyList<String>()
216 if (rest.startsWith("after ")) {
217 rest = rest.drop(6)
218 val word = rest.substringBefore(' ')
219 delay = word.toInt()
220 rest = rest.drop(word.length).trimStart(' ')
221 trigger = "timer"
222 } else {
223 val word = rest.substringBefore(' ')
224 rest = rest.drop(word.length).trimStart(' ')
225 trigger = if (word == "always") "always" else "event"
226 if (trigger == "event") descriptors = word.split('|')
227 }
228 var guard: Expr? = null
229 if (rest.startsWith('[')) {
230 val end = rest.indexOf(']')
231 guard = expression(rest.substring(1, end))
232 rest = rest.drop(end + 1).trimStart(' ')
233 }
234 var target: Int? = null
235 var mode = "internal"
236 if (rest.startsWith("-> ") || rest.startsWith("~> ")) {
237 mode = if (rest[0] == '-') "external" else "local"
238 rest = rest.drop(3)
239 val name = rest.substringBefore(' ')
240 target = ids[name] ?: throw Reference("unknown state $name")
241 rest = rest.drop(name.length).trimStart(' ')
242 }
243 val a = if (rest.startsWith("/ ")) actions(rest.drop(2)) else emptyList()
244 states[source].transitions.add(transitions.size)
245 transitions.add(Transition(source, trigger, descriptors, delay, guard, target, mode, a))
246 }
247 } catch (e: Reference) {
248 errors[d.line] = e.reason
249 }
250 }
251 if (errors.isNotEmpty()) {
252 for ((line, reason) in errors) output.add("error line $line: $reason")
253 return false
254 }
255 for (i in states.indices) {
256 if (compound(i)) states[i].initial = states[i].initialName?.let {
257 ids.getValue(it)
258 } ?: children(i).first()
259 if (states[i].kind == "history") states[i].default = states[i].defaultName?.let {
260 ids.getValue(it)
261 } ?: states[states[i].parent].initial
262 }

272 fun warnings() {
273 val reached = mutableSetOf(states[0].initial)
274 while (true) {
275 val before = reached.size
276 for (i in reached.toList()) {
277 if (states[i].parent != 0) reached.add(states[i].parent)
278 if (compound(i)) reached.add(states[i].initial)
279 if (parallel(i)) reached.addAll(children(i))
280 for (t in states[i].transitions.map {
281 transitions[it]
282 }) if (t.target != null) reached.add(if (states[t.target].kind == "history") states[t.target].default else t.target)
283 }
284 if (before == reached.size) break
285 }
286 for (i in document) if (states[i].kind != "history" && i !in reached) output.add("warning: unreachable state ${states[i].id}")
287 }

289 fun evaluate(e: Expr): Long {
290 if (e.op == "number") return e.text.toLong()
291 if (e.op == "variable") return if (e.text == "_data") data else variables.getValue(e.text)
292 if (e.op == "in") return if (ids.getValue(e.text) in active) 1 else 0
293 val a = evaluate(e.args[0])
294 if (e.op == "unary!") return if (a == 0L) 1 else 0
295 if (e.op == "unary-") return -a
296 if (e.op == "&&" && a == 0L) return 0
297 if (e.op == "||" && a != 0L) return 1
298 val b = evaluate(e.args[1])
299 if ((e.op == "/" || e.op == "%") && b == 0L) throw Evaluation()
300 return when (e.op) {
301 "+" -> a + b
302 "-" -> a - b
303 "*" -> a * b
304 "/" -> a / b
305 "%" -> a % b
306 "==" -> if (a == b) 1 else 0
307 "!=" -> if (a != b) 1 else 0
308 "<" -> if (a < b) 1 else 0
309 "<=" -> if (a <= b) 1 else 0
310 ">" -> if (a > b) 1 else 0
311 ">=" -> if (a >= b) 1 else 0
312 else -> if (b != 0L) 1 else 0
313 }
314 }

321 fun actions(list: List<Action>) {
322 for (a in list) try {
323 val v = a.expr?.let(::evaluate) ?: 0L
324 when (a.kind) {
325 "assign" -> variables[a.name] = v
326 "raise" -> queue.add(Event(a.name, v))
327 "send" -> reserve(event = Event(a.name, v), delay = a.delay)
328 "cancel" -> reservations.removeAll {
329 it.event?.name == a.name
330 }
331 "log" -> output.add((a.text ?: "") + if (a.expr == null) "" else v.toString())
332 }
333 } catch (_: Evaluation) {
334 queue.add(Event("error.execution"))
335 break
336 }
337 }

339 fun effective(t: Transition): List<Int> {
340 val i = t.target ?: return emptyList()
341 return if (states[i].kind == "history") states[i].history ?: listOf(states[i].default) else listOf(i)

344 fun scope(t: Transition): Int? {
345 if (t.target == null) return null
346 val targets = effective(t)
347 if (t.mode == "local" && compound(t.source) && targets.all {
348 descendant(it, t.source)
349 }) return t.source
350 var p = states[t.source].parent
351 while (!(compound(p) && targets.all {
352 descendant(it, p)
353 })) p = states[p].parent
354 return p
355 }

357 fun matches(t: Transition, trigger: Trigger, index: Int): Boolean = when (trigger.kind) {
358 "always" -> t.trigger == "always"
359 "timer" -> index == trigger.transition
360 else -> {
361 val name = trigger.event!!.name
362 t.trigger == "event" && t.descriptors.any {
363 it == "*" || it == name || name.startsWith("$it.")
364 }

368 fun select(trigger: Trigger): List<Selected> {
369 val candidates = mutableListOf<Int>()
370 val seen = mutableSetOf<Int>()
371 for (leaf in document.filter {
372 it in active && atomic(it)
373 }) {
374 var source = leaf
375 var chosen: Int? = null
376 while (source != 0 && chosen == null) {
377 for (index in states[source].transitions) {
378 val t = transitions[index]
379 if (!matches(t, trigger, index)) continue
380 val enabled = try {
381 t.guard?.let {
382 evaluate(it) != 0L
383 } ?: true
384 } catch (_: Evaluation) {
385 queue.add(Event("error.execution"))
386 false
387 }
388 if (enabled) {
389 chosen = index
390 break
391 }
392 }
393 source = states[source].parent
394 }
395 if (chosen != null && seen.add(chosen)) candidates.add(chosen)
396 }
397 val occupied = mutableSetOf<Int>()
398 val result = mutableListOf<Selected>()
399 for (i in candidates) {
400 val scope = scope(transitions[i])
401 val exits = if (scope == null) emptySet() else active.filter {
402 descendant(it, scope)
403 }.toSet()
404 if (exits.none {
405 it in occupied
406 }) {
407 result.add(Selected(i, scope, exits))
408 occupied.addAll(exits)
409 }
410 }
411 return result
412 }

431 fun enter(entries: Set<Int>) {
432 for (i in document) if (i in entries) {
433 val s = states[i]
434 active.add(i)
435 s.entered++
436 s.entryTime = time
437 for (index in s.transitions) if (transitions[index].trigger == "timer") reserve(transition = index, delay = transitions[index].delay)
438 for (list in s.entries) actions(list)
439 if (s.kind == "final") {
440 val p = s.parent
441 queue.add(Event("done.state.${states[p].id}"))
442 val g = states[p].parent
443 if (g >= 0 && parallel(g) && children(g).all(::complete)) queue.add(Event("done.state.${states[g].id}"))
444 }
445 }

462 fun micro(selected: List<Selected>, trigger: Trigger) {
463 val exits = selected.flatMap {
464 it.exits
465 }.toSet()
466 for (i in document) if (states[i].kind == "history" && states[i].parent in exits) {
467 val parent = states[i].parent
468 states[i].history = document.filter {
469 it in active && if (states[i].deep) atomic(it) && descendant(it, parent) else states[it].parent == parent && states[it].kind != "history"
470 }
471 }
472 val entries = mutableSetOf<Int>()
473 for (choice in selected) if (choice.scope != null) for (target in effective(transitions[choice.transition])) {
474 var i = target
475 while (i != choice.scope) {
476 entries.add(i)
477 i = states[i].parent
478 }
479 }
480 expand(entries)
481 if (trace) output.add("[t=$time] ${display(trigger)} (${list(selected.map { transitions[it.transition].source })}): exit ${list(document.asReversed().filter { it in exits })}; enter ${list(document.filter { it in entries })}")
482 for (i in document.asReversed()) if (i in exits) {
483 val s = states[i]
484 for (a in s.exits) actions(a)
485 active.remove(i)
486 s.dwell += time - s.entryTime
487 reservations.removeAll {
488 it.transition?.let {
489 transitions[it].source == i
490 } ?: false
491 }
492 }
493 for (choice in selected) {
494 val t = transitions[choice.transition]
495 actions(t.actions)
496 states[t.source].taken++
497 }
498 enter(entries)
499 }

501 fun macro(trigger: Trigger) {
502 macros++
503 var count = 0
504
505 fun perform(selected: List<Selected>, trigger: Trigger): Boolean {
506 if (selected.isEmpty()) return true
507 if (count == 1000) {
508 output.add("LIVELOCK t=$time")
509 halted = true
510 return false
511 }
512 count++
513 micros++
514 micro(selected, trigger)
515 return true
516 }
517 if (trigger.kind == "start") {
518 data = 0
519 val entries = mutableSetOf(states[0].initial)
520 expand(entries)
521 enter(entries)
522 } else {
523 data = trigger.event?.value ?: 0
524 if (!perform(select(trigger), trigger)) return
525 }
526 while (true) {
527 data = 0
528 val always = Trigger("always")
529 val selected = select(always)
530 if (selected.isNotEmpty()) {
531 if (!perform(selected, always)) return
532 continue
533 }
534 if (queue.isEmpty()) return
535 val event = queue.removeFirst()
536 data = event.value
537 val next = Trigger("event", event)
538 if (!perform(select(next), next)) return
539 }
540 }

542 fun script(line: String) {
543 if (halted) return
544 val w = line.split(' ')
545 when (w[0]) {
546 "event" -> macro(Trigger("event", Event(w[1], w.getOrNull(2)?.toLong() ?: 0)))
547 "tick" -> {
548 val finish = time + w[1].toInt()
549 while (!halted) {
550 val next = reservations.filter {
551 it.due <= finish
552 }.minWithOrNull(compareBy<Reservation> {
553 it.due
554 }.thenBy {
555 it.creation
556 }) ?: break
557 reservations.remove(next)
558 time = next.due
559 macro(if (next.transition == null) Trigger("event", next.event) else Trigger("timer", transition = next.transition))
560 }
561 if (!halted) time = finish
562 }
563 "set" -> if (w[1] in variables) variables[w[1]] = w[2].toLong() else output.add("error: unknown variable ${w[1]}")
564 "trace" -> trace = w[1] == "on"
565 "eval" -> {
566 val text = line.drop(5)
567 try {
568 val e = expression(text)
569 data = 0
570 try {
571 output.add("$text -> ${evaluate(e)}")
572 } catch (_: Evaluation) {
573 output.add("error: division by zero")
574 }
575 } catch (e: Reference) {
576 output.add("error: ${e.reason}")
577 }
578 }
579 "dump" -> {
580 output.add("dump t=$time")
581 for (i in document) if (i in active) output.add(" ".repeat(states[i].depth) + states[i].id)
582 for ((name, value) in variables) output.add(" var $name = $value")
583 for (i in document) states[i].history?.let {
584 output.add(" history ${states[i].id} = ${list(it)}")
585 }
586 for (r in reservations.sortedWith(compareBy<Reservation> {
587 it.due
588 }.thenBy {
589 it.creation
590 })) {
591 if (r.transition != null) {
592 val t = transitions[r.transition]
593 output.add(" timer ${r.due} after(${t.delay})@${states[t.source].id}")
594 } else output.add(" send ${r.due} ${r.event!!.name} ${r.event.value}")
595 }
596 }
597 }
598 }

600 fun statistics() {
601 output.add("end t=$time macrosteps=$macros microsteps=$micros")
602 for (i in document) if (states[i].kind != "history") {
603 val s = states[i]
604 val dwell = s.dwell + if (i in active) time - s.entryTime else 0
605 output.add(" ${s.id} entered=${s.entered} taken=${s.taken} time=$dwell")
606 }
607 }

610fun main() {
611 val lines = generateSequence(::readLine).toList()
612 val output = mutableListOf<String>()
613 var at = 1
614 repeat(lines[0].toInt()) {
615 output.add("chart ${lines[at++].drop(6)}")
616 val definitions = mutableListOf<Pair<Int, String>>()
617 while (lines[at] != "RUN") {
618 definitions.add(at + 1 to lines[at])
619 at++
620 }
621 at++
622 val scripts = mutableListOf<String>()
623 while (lines[at] != "END") scripts.add(lines[at++])
624 at++
625 val chart = Chart(output)
626 if (chart.define(definitions)) {
627 chart.warnings()
628 chart.macro(Trigger("start"))
629 for (line in scripts) chart.script(line)
630 chart.statistics()
631 }
632 }
633 for (line in output) println(line)
634}
