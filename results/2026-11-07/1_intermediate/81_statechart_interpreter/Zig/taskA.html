16fn words(gpa: Allocator, text: Text) Error![]const Text {
17 var result: std.ArrayList(Text) = .empty;
18 var iterator = std.mem.tokenizeScalar(u8, text, ' ');
19 while (iterator.next()) |item| try result.append(gpa, item);
20 return result.items;
21}

38const Expression = struct {
39 op: Text,
40 name: Text = "",
41 value: Int = 0,
42 left: ?*Expression = null,
43 right: ?*Expression = null,
44};
45const operators = [_][]const Text{ &.{"||"}, &.{"&&"}, &.{ "==", "!=", "<", "<=", ">", ">=" }, &.{ "+", "-" }, &.{ "*", "/", "%" } };
46fn member(items: []const Text, value: Text) bool {
47 for (items) |item| if (equal(item, value)) return true;
48 return false;
49}
50const ExpressionParser = struct {
51 gpa: Allocator,
52 tokens: []const Text,
53 position: usize = 0,
54 fn node(self: *ExpressionParser, value: Expression) Error!*Expression {
55 const result = try self.gpa.create(Expression);
56 result.* = value;
57 return result;
58 }
59 fn binary(self: *ExpressionParser, level: usize) Error!*Expression {
60 if (level == operators.len) return self.unary();
61 var result = try self.binary(level + 1);
62 while (self.position < self.tokens.len and member(operators[level], self.tokens[self.position])) {
63 const op = self.tokens[self.position];
64 self.position += 1;
65 result = try self.node(.{ .op = op, .left = result, .right = try self.binary(level + 1) });
66 }
67 return result;
68 }
69 fn unary(self: *ExpressionParser) Error!*Expression {
70 const token = self.tokens[self.position];
71 self.position += 1;
72 if (equal(token, "!") or equal(token, "-")) return self.node(.{ .op = if (equal(token, "!")) "unary!" else "unary-", .left = try self.unary() });
73 if (equal(token, "(")) {
74 const result = try self.binary(0);
75 self.position += 1;
76 return result;
77 }
78 if (equal(token, "in")) {
79 self.position += 1;
80 const name = self.tokens[self.position];
81 self.position += 2;
82 return self.node(.{ .op = "in", .name = name });
83 }
84 if (std.ascii.isDigit(token[0])) return self.node(.{ .op = "number", .value = number(token) });
85 return self.node(.{ .op = "variable", .name = token });
86 }
87};
88fn parseExpression(gpa: Allocator, text: Text) Error!*Expression {
89 var tokens: std.ArrayList(Text) = .empty;
90 var i: usize = 0;
91 while (i < text.len) {
92 const c = text[i];
93 if (c == ' ') {
94 i += 1;
95 continue;
96 }
97 const start = i;
98 if (std.ascii.isDigit(c)) {
99 while (i < text.len and std.ascii.isDigit(text[i])) i += 1;
100 } else if (std.ascii.isAlphabetic(c) or c == '_') {
101 while (i < text.len and (std.ascii.isAlphanumeric(text[i]) or text[i] == '_')) i += 1;
102 } else {
103 i += 1;
104 if (i < text.len and member(&.{ "||", "&&", "==", "!=", "<=", ">=" }, text[start .. i + 1])) i += 1;
105 }
106 try tokens.append(gpa, text[start..i]);
107 }
108 var parser: ExpressionParser = .{ .gpa = gpa, .tokens = tokens.items };
109 return parser.binary(0);
110}
111const ActionKind = enum { assign, raise, send, cancel, log };
112const Action = struct {
113 kind: ActionKind,
114 name: Text = "",
115 text: Text = "",
116 delay: Int = 0,
117 value: ?*Expression = null,
118};
119fn parseActions(gpa: Allocator, text: Text) Error![]const Action {
120 var result: std.ArrayList(Action) = .empty;
121 if (text.len == 0) return result.items;
122 var iterator = std.mem.splitSequence(u8, text, "; ");
123 while (iterator.next()) |item| {
124 var action: Action = undefined;
125 if (std.mem.startsWith(u8, item, "raise ")) {
126 const tail = item[6..];
127 const end = std.mem.findScalar(u8, tail, ' ') orelse tail.len;
128 action = .{ .kind = .raise, .name = tail[0..end] };
129 if (end < tail.len) action.value = try parseExpression(gpa, tail[end + 1 ..]);
130 } else if (std.mem.startsWith(u8, item, "send ")) {
131 const tail = item[5..];
132 const end = std.mem.findScalar(u8, tail, ' ').?;
133 const remainder = tail[end + 1 ..];
134 const delayEnd = std.mem.findScalar(u8, remainder, ' ') orelse remainder.len;
135 action = .{ .kind = .send, .name = tail[0..end], .delay = number(remainder[0..delayEnd]) };
136 if (delayEnd < remainder.len) action.value = try parseExpression(gpa, remainder[delayEnd + 1 ..]);
137 } else if (std.mem.startsWith(u8, item, "cancel ")) {
138 action = .{ .kind = .cancel, .name = item[7..] };
139 } else if (std.mem.startsWith(u8, item, "log ")) {
140 action = .{ .kind = .log };
141 const tail = item[4..];
142 if (tail[0] == '"') {
143 const end = std.mem.findScalar(u8, tail[1..], '"').? + 1;
144 action.text = tail[1..end];
145 if (end + 1 < tail.len) action.value = try parseExpression(gpa, tail[end + 2 ..]);
146 } else action.value = try parseExpression(gpa, tail);
147 } else {
148 const end = std.mem.find(u8, item, " = ").?;
149 action = .{ .kind = .assign, .name = item[0..end], .value = try parseExpression(gpa, item[end + 3 ..]) };
150 }
151 try result.append(gpa, action);
152 }
153 return result.items;
154}

222 fn descendant(self: *Chart, child: usize, ancestor: usize) bool {
223 var id = child;
224 while (id != 0) {
225 id = self.states.items[id].parent;

227 }
228 return false;

231 return id == 0 or self.states.items[id].kind == .state and self.states.items[id].childStates.items.len > 0;
232 }

236 fn problem(self: *Chart, comptime format: Text, args: anytype) Error!?Text {

239 fn reference(self: *Chart, expression: ?*const Expression) Error!?Text {
240 const e = expression orelse return null;
241 if (equal(e.op, "variable") and !equal(e.name, "_data") and !self.variables.contains(e.name)) return self.problem("unknown variable {s}", .{e.name});
242 if (equal(e.op, "in") and !self.names.contains(e.name)) return self.problem("unknown state {s}", .{e.name});
243 if (try self.reference(e.left)) |message| return message;
244 return self.reference(e.right);
245 }
246 fn actionReferences(self: *Chart, actions: []const Action) Error!?Text {
247 for (actions) |action| {
248 if (action.kind == .assign and !self.variables.contains(action.name)) return self.problem("unknown variable {s}", .{action.name});
249 if (try self.reference(action.value)) |message| return message;
250 }
251 return null;
252 }
253 fn parseTransition(self: *Chart, text: Text) Error!*Transition {
254 const endSource = std.mem.findScalar(u8, text[3..], ' ').? + 3;
255 const transition = try self.gpa.create(Transition);
256 transition.* = .{ .source = self.names.get(text[3..endSource]).? };
257 var tail = text[endSource + 1 ..];
258 var bracket: usize = 0;
259 var i: usize = 0;
260 while (i + 2 < tail.len) : (i += 1) {
261 if (tail[i] == '[') bracket += 1;
262 if (tail[i] == ']') bracket -= 1;
263 if (bracket == 0 and equal(tail[i .. i + 3], " / ")) {
264 transition.actions = try parseActions(self.gpa, tail[i + 3 ..]);
265 tail = tail[0..i];
266 break;
267 }
268 }
269 if (std.mem.find(u8, tail, " [")) |start| {
270 const end = std.mem.findScalar(u8, tail[start + 2 ..], ']').? + start + 2;
271 transition.guard = try parseExpression(self.gpa, tail[start + 2 .. end]);
272 tail = try std.mem.concat(self.gpa, u8, &.{ tail[0..start], tail[end + 1 ..] });
273 }
274 for ([_]Text{ " -> ", " ~> " }) |arrow| {
275 if (std.mem.find(u8, tail, arrow)) |position| {
276 transition.mode = if (equal(arrow, " -> ")) .external else .local;
277 transition.targetName = tail[position + 4 ..];
278 transition.target = self.names.get(transition.targetName);
279 tail = tail[0..position];
280 break;
281 }
282 }
283 if (equal(tail, "always")) {
284 transition.kind = .always;
285 } else if (std.mem.startsWith(u8, tail, "after ")) {
286 transition.kind = .timer;
287 transition.delay = number(tail[6..]);
288 } else {
289 var descriptors: std.ArrayList(Text) = .empty;
290 var iterator = std.mem.splitScalar(u8, tail, '|');
291 while (iterator.next()) |descriptor| try descriptors.append(self.gpa, descriptor);
292 transition.descriptors = descriptors.items;
293 }
294 return transition;
295 }
296 fn visit(self: *Chart, id: usize) Error!void {
297 try self.document.append(self.gpa, id);
298 for (self.states.items[id].children.items) |child| try self.visit(child);
299 }
300 fn define(self: *Chart, definitions: []Definition) Error!bool {
301 const root = try self.gpa.create(State);
302 root.* = .{ .kind = .root };
303 try self.states.append(self.gpa, root);
304 for (definitions) |*definition| {
305 const fields = try words(self.gpa, definition.text);
306 definition.kind = fields[0];
307 if (member(&.{ "state", "parallel", "final", "history" }, definition.kind)) {
308 const name = fields[1];
309 if (self.names.contains(name)) {
310 definition.problem = try self.problem("duplicate state {s}", .{name});
311 continue;
312 }
313 const parentName = if (fields.len > 2 and !equal(fields[2], "initial")) fields[2] else "";
314 var parent: usize = 0;
315 if (parentName.len > 0) {
316 parent = self.names.get(parentName) orelse {
317 definition.problem = try self.problem("unknown state {s}", .{parentName});
318 continue;
319 };
320 if (self.states.items[parent].kind == .final or self.states.items[parent].kind == .history) {
321 definition.problem = try self.problem("{s} cannot have children", .{parentName});
322 continue;
323 }
324 }
325 const state = try self.gpa.create(State);
326 state.* = .{ .name = name, .kind = std.meta.stringToEnum(StateKind, definition.kind).?, .parent = parent, .depth = self.states.items[parent].depth + 1 };
327 if (state.kind == .state) {
328 for (fields[2..], 2..) |field, index| if (equal(field, "initial")) {
329 state.initialName = fields[index + 1];
330 break;
331 };
332 }
333 if (state.kind == .history) {
334 state.deep = equal(fields[3], "deep");
335 if (fields.len == 5) state.defaultName = fields[4];
336 }
337 definition.state = self.states.items.len;
338 try self.names.put(self.gpa, name, definition.state);
339 try self.states.append(self.gpa, state);
340 try self.states.items[parent].children.append(self.gpa, definition.state);
341 if (state.kind != .history) try self.states.items[parent].childStates.append(self.gpa, definition.state);
342 } else if (equal(definition.kind, "var")) {
343 if (self.variables.contains(fields[1])) {
344 definition.problem = try self.problem("duplicate variable {s}", .{fields[1]});
345 } else {
346 try self.variables.put(self.gpa, fields[1], number(fields[2]));
347 try self.variableNames.append(self.gpa, fields[1]);
348 }
349 }
350 }
351 for (definitions) |*definition| {
352 if (definition.problem != null) continue;
353 const fields = try words(self.gpa, definition.text);
354 if (equal(definition.kind, "state")) {
355 const state = self.states.items[definition.state];
356 if (state.initialName.len > 0) {
357 if (self.names.get(state.initialName)) |child| {
358 if (self.states.items[child].parent != definition.state or self.states.items[child].kind == .history) definition.problem = try self.problem("{s} is not a child of {s}", .{ state.initialName, state.name });
359 } else definition.problem = try self.problem("unknown state {s}", .{state.initialName});
360 }
361 } else if (member(&.{ "final", "history" }, definition.kind)) {
362 const state = self.states.items[definition.state];
363 const parent = self.states.items[state.parent];
364 if (parent.kind != .state or parent.childStates.items.len == 0) {
365 definition.problem = try self.problem("{s} needs a compound parent", .{state.name});
366 } else if (state.kind == .history and state.defaultName.len > 0) {
367 if (self.names.get(state.defaultName)) |target| {
368 if (!self.descendant(target, state.parent) or self.states.items[target].kind == .history) definition.problem = try self.problem("{s} is not a descendant of {s}", .{ state.defaultName, parent.name });
369 } else definition.problem = try self.problem("unknown state {s}", .{state.defaultName});
370 }
371 } else if (member(&.{ "entry", "exit" }, definition.kind)) {
372 if (self.names.get(fields[1])) |id| {
373 const state = self.states.items[id];
374 if (state.kind == .history) {
375 definition.problem = try self.problem("{s} cannot have actions", .{state.name});
376 } else {
377 const separator = std.mem.find(u8, definition.text, " / ").?;
378 const actions = try parseActions(self.gpa, definition.text[separator + 3 ..]);
379 definition.problem = try self.actionReferences(actions);
380 if (definition.problem == null) {
381 if (equal(definition.kind, "entry")) {
382 try state.entries.append(self.gpa, actions);
383 } else try state.exits.append(self.gpa, actions);
384 }
385 }
386 } else definition.problem = try self.problem("unknown state {s}", .{fields[1]});
387 } else if (equal(definition.kind, "on")) {
388 if (self.names.get(fields[1])) |id| {
389 const state = self.states.items[id];
390 if (state.kind == .final or state.kind == .history) {
391 definition.problem = try self.problem("{s} cannot have transitions", .{state.name});
392 } else {
393 const transition = try self.parseTransition(definition.text);
394 definition.problem = try self.reference(transition.guard);
395 if (definition.problem == null and transition.targetName.len > 0 and transition.target == null) definition.problem = try self.problem("unknown state {s}", .{transition.targetName});
396 if (definition.problem == null) definition.problem = try self.actionReferences(transition.actions);
397 if (definition.problem == null) try state.transitions.append(self.gpa, transition);
398 }
399 } else definition.problem = try self.problem("unknown state {s}", .{fields[1]});
400 }
401 }
402 var valid = true;
403 for (definitions) |definition| if (definition.problem) |message| {
404 try self.out.line("error line {d}: {s}", .{ definition.line, message });
405 valid = false;
406 };
407 if (!valid) return false;
408 for (self.states.items) |state| if (state.childStates.items.len > 0) {
409 state.initial = if (state.initialName.len > 0) self.names.get(state.initialName).? else state.childStates.items[0];
410 };
411 for (self.states.items) |state| if (state.kind == .history) {
412 state.defaultTarget = if (state.defaultName.len > 0) self.names.get(state.defaultName).? else self.states.items[state.parent].initial;
413 };
414 for (root.children.items) |child| try self.visit(child);
415 std.mem.sort(Text, self.variableNames.items, {}, struct {
416 fn less(_: void, a: Text, b: Text) bool {
417 return std.mem.order(u8, a, b) == .lt;
418 }
419 }.less);
420 self.active = try self.makeSet();
421 return true;
422 }
423 fn warn(self: *Chart) Error!void {
424 const reachable = try self.makeSet();
425 reachable[self.states.items[0].initial] = true;
426 var changed = true;
427 while (changed) {
428 changed = false;
429 for (self.document.items) |id| {
430 if (!reachable[id]) continue;
431 const state = self.states.items[id];
432 if (state.parent != 0 and !reachable[state.parent]) {
433 reachable[state.parent] = true;
434 changed = true;
435 }
436 if (self.compound(id) and !reachable[state.initial]) {
437 reachable[state.initial] = true;
438 changed = true;
439 }
440 if (self.parallel(id)) for (state.childStates.items) |child| {
441 if (!reachable[child]) {
442 reachable[child] = true;
443 changed = true;
444 }
445 };
446 for (state.transitions.items) |transition| if (transition.target) |rawTarget| {
447 const target = if (self.states.items[rawTarget].kind == .history) self.states.items[rawTarget].defaultTarget else rawTarget;
448 if (!reachable[target]) {
449 reachable[target] = true;
450 changed = true;
451 }
452 };
453 }
454 }
455 for (self.document.items) |id| if (self.states.items[id].kind != .history and !reachable[id]) try self.out.line("warning: unreachable state {s}", .{self.states.items[id].name});
456 }
457 fn evaluate(self: *Chart, e: *const Expression) ?Int {
458 if (equal(e.op, "number")) return e.value;
459 if (equal(e.op, "variable")) return if (equal(e.name, "_data")) self.data else self.variables.get(e.name).?;
460 if (equal(e.op, "in")) return truth(self.active[self.names.get(e.name).?]);
461 const a = self.evaluate(e.left.?) orelse return null;
462 if (equal(e.op, "unary!")) return truth(a == 0);
463 if (equal(e.op, "unary-")) return -a;
464 if (equal(e.op, "&&") and a == 0) return 0;
465 if (equal(e.op, "||") and a != 0) return 1;
466 const b = self.evaluate(e.right.?) orelse return null;
467 if (equal(e.op, "+")) return a + b;
468 if (equal(e.op, "-")) return a - b;
469 if (equal(e.op, "*")) return a * b;
470 if (equal(e.op, "/") or equal(e.op, "%")) {
471 if (b == 0) return null;
472 return if (equal(e.op, "/")) @divTrunc(a, b) else @rem(a, b);
473 }
474 if (equal(e.op, "==")) return truth(a == b);
475 if (equal(e.op, "!=")) return truth(a != b);
476 if (equal(e.op, "<")) return truth(a < b);
477 if (equal(e.op, "<=")) return truth(a <= b);
478 if (equal(e.op, ">")) return truth(a > b);
479 if (equal(e.op, ">=")) return truth(a >= b);
480 return truth(b != 0);
481 }
482 fn enqueue(self: *Chart, name: Text, value: Int) Error!void {
483 try self.queue.append(self.gpa, .{ .name = name, .value = value });
484 }
485 fn reserve(self: *Chart, raw: Reservation) Error!void {
486 self.creation += 1;
487 var reservation = raw;
488 reservation.order = self.creation;
489 try self.reservations.append(self.gpa, reservation);

491 fn perform(self: *Chart, actions: []const Action) Error!void {
492 for (actions) |action| {
493 var value: Int = 0;
494 if (action.value) |expression| value = self.evaluate(expression) orelse {
495 try self.enqueue("error.execution", 0);
496 return;
497 };
498 switch (action.kind) {
499 .assign => try self.variables.put(self.gpa, action.name, value),
500 .raise => try self.enqueue(action.name, value),
501 .send => try self.reserve(.{ .due = self.time + action.delay, .event = .{ .name = action.name, .value = value } }),
502 .cancel => {
503 var i: usize = 0;
504 while (i < self.reservations.items.len) {
505 const reservation = self.reservations.items[i];
506 if (reservation.timer == null and equal(reservation.event.name, action.name)) {
507 _ = self.reservations.orderedRemove(i);
508 } else i += 1;
509 }
510 },
511 .log => {
512 if (action.value != null) {
513 try self.out.line("{s}{d}", .{ action.text, value });
514 } else try self.out.line("{s}", .{action.text});
515 },
516 }
517 }
518 }
519 fn targets(self: *Chart, transition: *const Transition) Error![]const usize {
520 const target = transition.target.?;
521 if (self.states.items[target].kind == .history) {
522 if (self.states.items[target].history) |recorded| return recorded;
523 const result = try self.gpa.alloc(usize, 1);
524 result[0] = self.states.items[target].defaultTarget;
525 return result;
526 }
527 const result = try self.gpa.alloc(usize, 1);
528 result[0] = target;
529 return result;
530 }

535 fn transitionSet(self: *Chart, transition: *Transition) Error!Selected {
536 var result: Selected = .{ .transition = transition, .exits = try self.makeSet() };
537 if (transition.mode == .internal) return result;
538 const targetList = try self.targets(transition);
539 if (transition.mode == .local and self.compound(transition.source) and self.containsTargets(transition.source, targetList)) {
540 result.scope = transition.source;
541 } else {
542 var id = self.states.items[transition.source].parent;
543 while (!(self.compound(id) and self.containsTargets(id, targetList))) id = self.states.items[id].parent;
544 result.scope = id;
545 }
546 for (self.document.items) |id| result.exits[id] = self.active[id] and self.descendant(id, result.scope);
547 return result;
548 }
549 fn matches(transition: *const Transition, trigger: Trigger) bool {
550 if (transition.kind != trigger.kind) return false;
551 if (trigger.kind == .always) return true;
552 if (trigger.kind == .timer) return transition == trigger.timer.?;
553 for (transition.descriptors) |descriptor| {
554 const name = trigger.event.name;
555 if (equal(descriptor, "*") or equal(descriptor, name) or name.len > descriptor.len and std.mem.startsWith(u8, name, descriptor) and name[descriptor.len] == '.') return true;
556 }
557 return false;
558 }
559 fn selectTransitions(self: *Chart, trigger: Trigger) Error![]const Selected {
560 var result: std.ArrayList(Selected) = .empty;
561 var seen: std.ArrayList(*Transition) = .empty;
562 const occupied = try self.makeSet();
563 for (self.document.items) |atomic| {
564 if (!self.active[atomic] or self.states.items[atomic].childStates.items.len != 0) continue;
565 var found: ?*Transition = null;
566 var id = atomic;
567 while (id != 0 and found == null) : (id = self.states.items[id].parent) {
568 for (self.states.items[id].transitions.items) |transition| {
569 if (!matches(transition, trigger)) continue;
570 const value: ?Int = if (transition.guard) |guard| self.evaluate(guard) else 1;
571 if (value) |actual| {
572 if (actual != 0) {
573 found = transition;
574 break;
575 }
576 } else try self.enqueue("error.execution", 0);
577 }
578 }
579 const transition = found orelse continue;
580 if (std.mem.findScalar(*Transition, seen.items, transition) != null) continue;
581 try seen.append(self.gpa, transition);
582 const selected = try self.transitionSet(transition);
583 var conflict = false;
584 for (self.document.items) |candidate| if (selected.exits[candidate] and occupied[candidate]) {
585 conflict = true;
586 break;
587 };
588 if (!conflict) {
589 try result.append(self.gpa, selected);
590 for (self.document.items) |candidate| occupied[candidate] = occupied[candidate] or selected.exits[candidate];
591 }
592 }
593 return result.items;
594 }
595 fn expand(self: *Chart, entries: []bool) void {
596 var changed = true;
597 while (changed) {
598 changed = false;
599 for (self.document.items) |id| {
600 if (!entries[id]) continue;
601 const state = self.states.items[id];
602 if (self.parallel(id)) {
603 for (state.childStates.items) |child| if (!entries[child]) {
604 entries[child] = true;
605 changed = true;
606 };
607 } else if (self.compound(id)) {
608 var hasChild = false;
609 for (state.childStates.items) |child| if (entries[child]) {
610 hasChild = true;
611 break;
612 };
613 if (!hasChild) {
614 entries[state.initial] = true;
615 changed = true;
616 }
617 }
618 }
619 }
620 }
621 fn complete(self: *Chart, id: usize) bool {
622 const state = self.states.items[id];
623 if (self.compound(id)) {
624 for (state.childStates.items) |child| if (self.states.items[child].kind == .final and self.active[child]) return true;
625 } else if (self.parallel(id)) {
626 for (state.childStates.items) |child| if (!self.complete(child)) return false;
627 return true;
628 }
629 return false;
630 }
631 fn enter(self: *Chart, entries: []const bool) Error!void {
632 for (self.document.items) |id| {
633 if (!entries[id]) continue;
634 const state = self.states.items[id];
635 self.active[id] = true;
636 state.entered += 1;
637 state.entryTime = self.time;
638 for (state.transitions.items) |transition| if (transition.kind == .timer) try self.reserve(.{ .due = self.time + transition.delay, .timer = transition });
639 for (state.entries.items) |actions| try self.perform(actions);
640 if (state.kind == .final) {
641 const parent = state.parent;
642 try self.enqueue(try std.fmt.allocPrint(self.gpa, "done.state.{s}", .{self.states.items[parent].name}), 0);
643 const grandparent = self.states.items[parent].parent;
644 if (self.parallel(grandparent) and self.complete(grandparent)) try self.enqueue(try std.fmt.allocPrint(self.gpa, "done.state.{s}", .{self.states.items[grandparent].name}), 0);
645 }
646 }
647 }
648 fn stateList(self: *Chart, set: []const bool, reverse: bool) Error!Text {
649 var result: Builder = .{ .gpa = self.gpa };
650 var count: usize = 0;
651 var index: usize = 0;
652 while (index < self.document.items.len) : (index += 1) {
653 const id = self.document.items[if (reverse) self.document.items.len - 1 - index else index];
654 if (set[id]) {
655 if (count > 0) try result.add(",");
656 try result.add(self.states.items[id].name);
657 count += 1;
658 }
659 }
660 return if (count == 0) "-" else result.bytes.items;
661 }
662 fn microstep(self: *Chart, trigger: Trigger, transitions: []const Selected) Error!void {
663 const exits = try self.makeSet();
664 for (transitions) |selected| for (self.document.items) |id| {
665 exits[id] = exits[id] or selected.exits[id];
666 };
667 for (self.document.items) |id| {
668 const history = self.states.items[id];
669 if (history.kind != .history or !exits[history.parent]) continue;
670 var recorded: std.ArrayList(usize) = .empty;
671 for (self.document.items) |candidate| {
672 const state = self.states.items[candidate];
673 if (self.active[candidate] and (history.deep and state.childStates.items.len == 0 and self.descendant(candidate, history.parent) or !history.deep and state.parent == history.parent)) try recorded.append(self.gpa, candidate);
674 }
675 history.history = recorded.items;
676 }
677 const entries = try self.makeSet();
678 for (transitions) |selected| {
679 if (selected.transition.mode == .internal) continue;
680 for (try self.targets(selected.transition)) |target| {
681 entries[target] = true;
682 var parent = self.states.items[target].parent;
683 while (parent != 0 and self.descendant(parent, selected.scope)) : (parent = self.states.items[parent].parent) entries[parent] = true;
684 }
685 }
686 self.expand(entries);
687 if (self.trace) {
688 var sources: Builder = .{ .gpa = self.gpa };
689 for (transitions, 0..) |selected, i| {
690 if (i > 0) try sources.add(",");
691 try sources.add(self.states.items[selected.transition.source].name);
692 }
693 const display = switch (trigger.kind) {
694 .always => "always",
695 .timer => try std.fmt.allocPrint(self.gpa, "after({d})@{s}", .{ trigger.timer.?.delay, self.states.items[trigger.timer.?.source].name }),
696 .event => if (trigger.event.value == 0) trigger.event.name else try std.fmt.allocPrint(self.gpa, "{s}={d}", .{ trigger.event.name, trigger.event.value }),
697 };
698 try self.out.line("[t={d}] {s} ({s}): exit {s}; enter {s}", .{ self.time, display, sources.bytes.items, try self.stateList(exits, true), try self.stateList(entries, false) });
699 }
700 var index = self.document.items.len;
701 while (index > 0) {
702 index -= 1;
703 const id = self.document.items[index];
704 if (!exits[id]) continue;
705 const state = self.states.items[id];
706 for (state.exits.items) |actions| try self.perform(actions);
707 self.active[id] = false;
708 state.dwell += self.time - state.entryTime;
709 var i: usize = 0;
710 while (i < self.reservations.items.len) {
711 if (self.reservations.items[i].timer) |timer| {
712 if (timer.source == id) {
713 _ = self.reservations.orderedRemove(i);
714 continue;
715 }
716 }
717 i += 1;
718 }
719 }
720 for (transitions) |selected| {
721 try self.perform(selected.transition.actions);
722 self.states.items[selected.transition.source].taken += 1;
723 }
724 try self.enter(entries);
725 }
726 fn performMicrostep(self: *Chart, trigger: Trigger, transitions: []const Selected, count: *usize) Error!bool {
727 if (transitions.len == 0) return true;
728 if (count.* == 1000) {
729 try self.out.line("LIVELOCK t={d}", .{self.time});
730 self.halted = true;
731 return false;
732 }
733 count.* += 1;
734 self.microsteps += 1;
735 try self.microstep(trigger, transitions);
736 return true;
737 }
738 fn macrostep(self: *Chart, start: bool, trigger: Trigger) Error!void {
739 self.macrosteps += 1;
740 var count: usize = 0;
741 self.data = 0;
742 if (start) {
743 const entries = try self.makeSet();
744 entries[self.states.items[0].initial] = true;
745 self.expand(entries);
746 try self.enter(entries);
747 } else {
748 if (trigger.kind == .event) self.data = trigger.event.value;
749 if (!try self.performMicrostep(trigger, try self.selectTransitions(trigger), &count)) return;
750 }
751 while (true) {
752 self.data = 0;
753 const always: Trigger = .{ .kind = .always };
754 const transitions = try self.selectTransitions(always);
755 if (transitions.len > 0) {
756 if (!try self.performMicrostep(always, transitions, &count)) return;
757 continue;
758 }
759 if (self.queueHead == self.queue.items.len) return;
760 const event = self.queue.items[self.queueHead];
761 self.queueHead += 1;
762 self.data = event.value;
763 const next: Trigger = .{ .kind = .event, .event = event };
764 if (!try self.performMicrostep(next, try self.selectTransitions(next), &count)) return;
765 }
766 }
767 fn dump(self: *Chart) Error!void {
768 try self.out.line("dump t={d}", .{self.time});
769 for (self.document.items) |id| if (self.active[id]) {
770 for (0..self.states.items[id].depth * 2) |_| try self.out.add(" ");
771 try self.out.line("{s}", .{self.states.items[id].name});
772 };
773 for (self.variableNames.items) |name| try self.out.line(" var {s} = {d}", .{ name, self.variables.get(name).? });
774 for (self.document.items) |id| if (self.states.items[id].history) |recorded| {
775 var list: Builder = .{ .gpa = self.gpa };
776 for (recorded, 0..) |target, i| {
777 if (i > 0) try list.add(",");
778 try list.add(self.states.items[target].name);
779 }
780 try self.out.line(" history {s} = {s}", .{ self.states.items[id].name, if (recorded.len == 0) "-" else list.bytes.items });
781 };
782 const reservations = try self.gpa.dupe(Reservation, self.reservations.items);
783 std.mem.sort(Reservation, reservations, {}, struct {
784 fn less(_: void, a: Reservation, b: Reservation) bool {
785 return reservationLess(a, b);
786 }
787 }.less);
788 for (reservations) |reservation| {
789 if (reservation.timer) |timer| {
790 try self.out.line(" timer {d} after({d})@{s}", .{ reservation.due, timer.delay, self.states.items[timer.source].name });
791 } else try self.out.line(" send {d} {s} {d}", .{ reservation.due, reservation.event.name, reservation.event.value });
792 }
793 }
794 fn script(self: *Chart, text: Text) Error!void {
795 if (self.halted) return;
796 const fields = try words(self.gpa, text);
797 if (equal(fields[0], "event")) {
798 try self.macrostep(false, .{ .event = .{ .name = fields[1], .value = if (fields.len == 3) number(fields[2]) else 0 } });
799 } else if (equal(fields[0], "tick")) {
800 const finish = self.time + number(fields[1]);
801 while (!self.halted) {
802 var best: ?usize = null;
803 for (self.reservations.items, 0..) |reservation, i| if (reservation.due <= finish and (best == null or reservationLess(reservation, self.reservations.items[best.?]))) {
804 best = i;
805 };
806 const index = best orelse break;
807 const reservation = self.reservations.orderedRemove(index);
808 self.time = reservation.due;
809 try self.macrostep(false, if (reservation.timer) |timer| .{ .kind = .timer, .timer = timer } else .{ .event = reservation.event });
810 }
811 if (!self.halted) self.time = finish;
812 } else if (equal(fields[0], "set")) {
813 if (self.variables.contains(fields[1])) {
814 try self.variables.put(self.gpa, fields[1], number(fields[2]));
815 } else try self.out.line("error: unknown variable {s}", .{fields[1]});
816 } else if (equal(fields[0], "trace")) {
817 self.trace = equal(fields[1], "on");
818 } else if (equal(fields[0], "eval")) {
819 const raw = text[5..];
820 const expression = try parseExpression(self.gpa, raw);
821 if (try self.reference(expression)) |message| {
822 try self.out.line("error: {s}", .{message});
823 } else {
824 self.data = 0;
825 if (self.evaluate(expression)) |value| {
826 try self.out.line("{s} -> {d}", .{ raw, value });
827 } else try self.out.line("error: division by zero", .{});
828 }
829 } else if (equal(fields[0], "dump")) try self.dump();
830 }
831 fn statistics(self: *Chart) Error!void {
832 try self.out.line("end t={d} macrosteps={d} microsteps={d}", .{ self.time, self.macrosteps, self.microsteps });
833 for (self.document.items) |id| {
834 const state = self.states.items[id];
835 if (state.kind == .history) continue;
836 const dwell = state.dwell + if (self.active[id]) self.time - state.entryTime else 0;
837 try self.out.line(" {s} entered={d} taken={d} time={d}", .{ state.name, state.entered, state.taken, dwell });
838 }
839 }
840};
841fn reservationLess(a: Reservation, b: Reservation) bool {
842 return a.due < b.due or a.due == b.due and a.order < b.order;
843}

845pub fn main(init: std.process.Init) !void {
846 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
847 defer arena.deinit();
848 const gpa = arena.allocator();
849 var inputBuffer: [8192]u8 = undefined;
850 var input = std.Io.File.stdin().readerStreaming(init.io, &inputBuffer);
851 const text = try input.interface.allocRemaining(gpa, .unlimited);
852 var lines = std.mem.splitScalar(u8, text, '\n');
853 const count = number(std.mem.trimEnd(u8, lines.next().?, "\r"));
854 var lineNumber: usize = 1;
855 var out: Builder = .{ .gpa = gpa };
856 var chartIndex: Int = 0;
857 while (chartIndex < count) : (chartIndex += 1) {
858 const header = std.mem.trimEnd(u8, lines.next().?, "\r");
859 lineNumber += 1;
860 try out.line("chart {s}", .{header[6..]});
861 var chartArena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
862 defer chartArena.deinit();
863 const chartAllocator = chartArena.allocator();
864 var definitions: std.ArrayList(Definition) = .empty;
865 while (lines.next()) |raw| {
866 lineNumber += 1;
867 const line = std.mem.trimEnd(u8, raw, "\r");
868 if (equal(line, "RUN")) break;
869 try definitions.append(chartAllocator, .{ .text = line, .line = lineNumber });
870 }
871 var chart: Chart = .{ .gpa = chartAllocator, .out = &out };
872 const valid = try chart.define(definitions.items);
873 if (valid) {
874 try chart.warn();
875 try chart.macrostep(true, .{});
876 }
877 while (lines.next()) |raw| {
878 lineNumber += 1;
879 const line = std.mem.trimEnd(u8, raw, "\r");
880 if (equal(line, "END")) break;
881 if (valid) try chart.script(line);
882 }
883 if (valid) try chart.statistics();
884 }
885 var outputBuffer: [8192]u8 = undefined;
886 var output = std.Io.File.stdout().writerStreaming(init.io, &outputBuffer);
887 try output.interface.writeAll(out.bytes.items);
888 try output.interface.flush();
889}
