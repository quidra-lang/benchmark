    println(text)
bool includes_id(nat[] choices, nat value)
    for choice := choices[]
        if choice == value
            return true
    return false

bool[] flags(nat size)
    return [for i := (0:<nat(size))[] then false]
class TextPair(string head, string tail)
TextPair cut_once(string text, separator := " ")
    match text.find(separator)
        nat at
            return TextPair(text[:<at], text[at + #separator:])
        none
            return TextPair(text, "")

class ExprNode(string kind, text := "", value := 0, nat[] operands := [])
class Expression(nat root := 0, ExprNode[] nodes := [])
class Action(string kind, name := "", nat delay := 0, text := "", expression := Expression())
class ActionList(Action[] items := [])

ActionList read_actions(string text)
    Action[] result := []
    if text == ""
        return ActionList()
    for item := text.split("; ")[]
        first := cut_once(item)
        word := first.head
        rest := first.tail
        elif word == "log"
            if rest.starts_with(DQ)
                nat closing := 0
                match rest[1:].find(DQ)
                    nat found
                        closing = found + 1
                    none
                        void
                tail := if closing + 1 < #rest then rest[closing + 2:] else ""
                expr := if tail == "" then Expression() else read_expression(tail)
                result = result.append(Action("log", text = rest[1:<closing], expression = expr))
            else
                result = result.append(Action("log", expression = read_expression(rest)))
    return ActionList(result)
class OnParts(string source_name, string trigger_kind, string descriptors, nat delay, string guard_text, string target_name, bool local_target, string action_text)

OnParts read_transition(string text)
    first := cut_once(text[3:])
    trigger := cut_once(first.tail)
    rest := trigger.tail
    trigger_kind := "event"
    nat delay := 0
    if trigger.head == "always"
        trigger_kind = "always"
    elif trigger.head == "after"
        trigger_kind = "timer"
        number := cut_once(rest)
        delay = nat(number.head)
        rest = number.tail
    guard_text := ""
        guard_text = rest[1:<closing]
        rest = if closing + 1 < #rest then rest[closing + 2:] else ""
    target_name := ""
    local_target := false
    if rest.starts_with("-> ") or rest.starts_with("~> ")
        local_target = rest.starts_with("~>")
        target := cut_once(rest[3:])
        target_name = target.head
        rest = target.tail
    action_text := if rest.starts_with("/ ") then rest[2:] else ""
    return OnParts(first.head, trigger_kind, trigger.head, delay, guard_text, target_name, local_target, action_text)

class ChartModel()
    states := [State("", "root", -1, 0)]
    by_name := map<string, nat>()
    initial_variables := map<string, int>()
    Transition[] transitions := []
    nat[] document_order := []
int state_index(ChartModel chart, string name)
    match chart.by_name.get(name)
        nat index
            return int(index)
        none
            return -1

nat[] child_states(ChartModel chart, nat state_id)
    nat[] result := []
    for child := chart.states[state_id].children[]
        if chart.states[child].kind != "history"
            result = result.append(child)
    return result

bool is_compound(ChartModel chart, nat state_id)
    return state_id == 0 or (chart.states[state_id].kind == "state" and #child_states(chart, state_id) > 0)

bool is_parallel(ChartModel chart, nat state_id)
    return chart.states[state_id].kind == "parallel" and #child_states(chart, state_id) > 0

bool is_atomic(ChartModel chart, nat state_id)
    return chart.states[state_id].kind != "history" and #child_states(chart, state_id) == 0
nat initial_id(ChartModel chart, nat state_id)
    name := chart.states[state_id].initial_name
    return if name == "" then child_states(chart, state_id)[0] else nat(state_index(chart, name))
string check_expression(ChartModel chart, Expression expr)
    return if #expr.nodes == 0 then "" else check_expression_node(chart, expr, expr.root)

string check_actions(ChartModel chart, ActionList list)
    for action := list.items[]
        if action.kind == "assign"
            match chart.initial_variables.get(action.name)
                int value
                    void
                none
                    return "unknown variable " + action.name
        failure := check_expression(chart, action.expression)
        if failure != ""
            return failure
    return ""
class Builder(DefinitionLine[] lines)
    model := ChartModel()
    string[] failures := [for i := (nat(0):<#lines)[] then ""]
    nat[] state_for_line := [for i := (nat(0):<#lines)[] then 0]

    void .first_phase()
        for: row < #lines
            words := lines[row].text.split(" ")
            kind := words[0]
            if includes_text(["state", "parallel", "final", "history"], kind)
                name := words[1]
                parent_name := ""
                init := ""
                fallback := ""
                deep := false
                if kind == "state"
                    nat at := 2
                    if #words > 2 and words[2] != "initial"
                        parent_name = words[2]
                        at = 3
                    if at < #words
                        init = words[at + 1]
                parent_index := if parent_name == "" then 0 else state_index(model, parent_name)
                if state_index(model, name) >= 0
                    failures[row] = "duplicate state " + name
                elif parent_index < 0
                    failures[row] = "unknown state " + parent_name
                elif includes_text(["final", "history"], model.states[parent_index].kind)
                    failures[row] = parent_name + " cannot have children"
                else
                    index := #model.states
                    state_for_line[row] = index
                    .model.by_name.set(name, index)
                    model.states = model.states.append(State(name, kind, parent_index, model.states[parent_index].depth + 1, init, fallback, deep))
                    model.states[parent_index].children = model.states[parent_index].children.append(index)
    void .second_phase()
        for: row < #lines
            if failures[row] != ""
                continue
            text := lines[row].text
            words := text.split(" ")
            kind := words[0]
            failure := ""
            if includes_text(["state", "final", "history"], kind)
                index := state_for_line[row]
                State node := model.states[index]
                if kind == "state" and node.initial_name != ""
            elif kind == "on"
                parts := read_transition(text)
                index := state_index(model, parts.source_name)
                if index < 0
                    failure = "unknown state " + parts.source_name
                elif includes_text(["final", "history"], model.states[index].kind)
                    failure = parts.source_name + " cannot have transitions"
                else
                    guard := if parts.guard_text == "" then Expression() else read_expression(parts.guard_text)
                    failure = check_expression(model, guard)
                    target := if parts.target_name == "" then -1 else state_index(model, parts.target_name)
                    if failure == "" and parts.target_name != "" and target < 0
                        failure = "unknown state " + parts.target_name
                    list := read_actions(parts.action_text)
                    if failure == ""
                        failure = check_actions(model, list)
                    if failure == ""
                        transition_index := #model.transitions
                        model.transitions = model.transitions.append(Transition(nat(index), parts.trigger_kind, parts.descriptors.split("|"), parts.delay, guard, target, parts.local_target, list))
                        model.states[index].transition_ids = model.states[index].transition_ids.append(transition_index)
            failures[row] = failure
    void .visit(nat state_id)
        model.document_order = model.document_order.append(state_id)
        for child := model.states[state_id].children[]
            .visit(child)

    ChartModel .build()
        .first_phase()
        .second_phase()
        for child := model.states[0].children[]
            .visit(child)
        return model
class Event(name := "", value := 0)
class Reservation(int due, nat creation, transition_id := -1, held_event := Event())
class HistoryValue(recorded := false, nat[] ids := [])
string timer_name(ChartModel chart, nat transition_id)
    item := chart.transitions[transition_id]
    return "after({item.delay})@" + chart.states[item.source_id].name

string event_name(Event item)
    return item.name + (if item.value == 0 then "" else "={item.value}")
bool[] close_entries(ChartModel chart, bool[] seeds)
    result := seeds
    changed := true
    while changed
        changed = false
        for id := chart.document_order[]
            if not result[id]
                continue
            children := child_states(chart, id)
            nat[] extra := []
            if is_parallel(chart, id)
                extra = children
            elif is_compound(chart, id)
                has_child := false
                for child := children[]
                    if result[child]
                        has_child = true
                if not has_child
                    extra = [initial_id(chart, id)]
            for child := extra[]
                if not result[child]
                    result[child] = true
                    changed = true
    return result
class Engine(ChartModel chart)
    bool[] configuration := flags(#chart.states)
    map<string, int> values := chart.initial_variables
    data_value := 0
    Event[] queue := []
    Reservation[] reservations := []
    nat serial_counter := 0
    HistoryValue[] history_values := [for i := (nat(0):<#chart.states)[] then HistoryValue()]
    clock := 0
    trace_enabled := false
    stopped := false
    nat macro_count := 0
    nat micro_count := 0
    nat within_count := 0
    nat[] entry_counts := [for i := (nat(0):<#chart.states)[] then 0]
    nat[] transition_counts := [for i := (nat(0):<#chart.states)[] then 0]
    int[] dwell_values := [for i := (nat(0):<#chart.states)[] then 0]
    int[] entry_times := [for i := (nat(0):<#chart.states)[] then 0]
    void .perform_actions(ActionList list)
        for item := list.items[]
            value := 0
            if #item.expression.nodes > 0
                match evaluate_expression(item.expression)
            if item.kind == "assign"
                .values.set(item.name, value)
            elif item.kind == "raise"
                queue = queue.append(Event(item.name, value))
            elif item.kind == "send"
                .add_reservation(item.delay, held = Event(item.name, value))
            elif item.kind == "log"
                emit(item.text + (if #item.expression.nodes == 0 then "" else string(value)))
    void .enter_states(bool[] entering)
        for state_id := chart.document_order[]
            if not entering[state_id]
                continue
            item := chart.states[state_id]
            configuration[state_id] = true
            entry_counts[state_id] += 1
            entry_times[state_id] = clock
            for transition_id := item.transition_ids[]
                branch := chart.transitions[transition_id]
                if branch.trigger_kind == "timer"
                    .add_reservation(branch.delay, int(transition_id))
            for list := item.entries[]
                .perform_actions(list)
            if item.kind == "final"
                parent_id := nat(item.parent)
    int transition_scope(Transition item)
        if item.target_id < 0
            return -1
    bool[] exits_for(Transition item)
        result := flags(#chart.states)
        int bound := transition_scope(item)
        if bound >= 0
            for id := chart.document_order[]
                result[id] = configuration[id] and is_descendant(chart, id, nat(bound))
        return result
    bool matches_trigger(nat transition_id, string trigger_kind, Event item, int timer_id)
        branch := chart.transitions[transition_id]
        if trigger_kind == "always"
            return branch.trigger_kind == "always"
        if trigger_kind == "timer"
            return int(transition_id) == timer_id
        if branch.trigger_kind != "event"
            return false
        for descriptor := branch.descriptors[]
            if descriptor == "*" or descriptor == item.name or item.name.starts_with(descriptor + ".")
                return true
        return false
    bool .guard_accepts(Transition item)
        if #item.guard.nodes == 0
            return true
    nat[] .select_transitions(string trigger_kind, item := Event(), timer_id := -1)
        nat[] candidates := []
        for atomic_id := chart.document_order[]
            if not configuration[atomic_id] or not is_atomic(chart, atomic_id)
                continue
            current := atomic_id
            found := -1
            while current != 0 and found < 0
                for transition_id := chart.states[current].transition_ids[]
                    if matches_trigger(transition_id, trigger_kind, item, timer_id)
                        if .guard_accepts(chart.transitions[transition_id])
                            found = int(transition_id)
                            break
                current = nat(chart.states[current].parent)
            if found >= 0 and not includes_id(candidates, nat(found))
                candidates = candidates.append(nat(found))
        nat[] kept := []
        exiting := flags(#chart.states)
        for transition_id := candidates[]
            bool[] proposed := exits_for(chart.transitions[transition_id])
            conflict := false
            for id := chart.document_order[]
                if exiting[id] and proposed[id]
                    conflict = true
            if not conflict
                kept = kept.append(transition_id)
                for id := chart.document_order[]
                    exiting[id] = exiting[id] or proposed[id]
        return kept
    void .micro_step(nat[] selected, string display)
        if within_count == 1000
            emit("LIVELOCK t={clock}")
            stopped = true
            return
        within_count += 1
        micro_count += 1
        int[] scopes := [for transition_id := selected[] then transition_scope(chart.transitions[transition_id])]
        exiting := flags(#chart.states)
        for bound := scopes[]
        for state_id := chart.document_order[]
            if not exiting[state_id]
                continue
        seeds := flags(#chart.states)
        for: i < #selected
            branch := chart.transitions[selected[i]]
            if branch.target_id < 0
                continue
        entering := close_entries(chart, seeds)
        for state_id := chart.document_order[::-1][]
            if not exiting[state_id]
                continue
            for list := chart.states[state_id].exits[]
                .perform_actions(list)
            configuration[state_id] = false
        for transition_id := selected[]
            branch := chart.transitions[transition_id]
            .perform_actions(branch.actions)
            transition_counts[branch.source_id] += 1
        .enter_states(entering)

    void .macro_step(string trigger_kind, item := Event(), timer_id := -1)
        macro_count += 1
        within_count = 0
        data_value = item.value
        if trigger_kind == "start"
            seeds := flags(#chart.states)
            seeds[initial_id(chart, 0)] = true
            .enter_states(close_entries(chart, seeds))
        else
            nat[] selected := .select_transitions(trigger_kind, item, timer_id)
            if #selected > 0
                .micro_step(selected, if trigger_kind == "timer" then timer_name(chart, nat(timer_id)) else event_name(item))
        while not stopped
            data_value = 0
            nat[] selected := .select_transitions("always")
            if #selected > 0
                .micro_step(selected, "always")
                continue
            if #queue == 0
                break
            next := queue[0]
            queue = queue[1:]
            data_value = next.value
            selected = .select_transitions("event", next)
            if #selected > 0
                .micro_step(selected, event_name(next))
    void write_dump()
        emit("dump t={clock}")
        for id := chart.document_order[]
            if configuration[id]
                emit(" " * (chart.states[id].depth * 2) + chart.states[id].name)
    void .script_line(string text)
        words := text.split(" ")
        command := words[0]
        if command == "event"
            .macro_step("event", Event(words[1], if #words == 3 then int(words[2]) else 0))
                reservations = reservations[:<index].concat(reservations[index + 1:])
                clock = booking.due
                .macro_step(if booking.transition_id < 0 then "event" else "timer", booking.held_event, booking.transition_id)
        elif command == "dump"
            write_dump()
void run_chart(string name, DefinitionLine[] definitions, string[] scripts)
    emit("chart " + name)
    builder := Builder(definitions)
    chart := .builder.build()
    invalid := false
    for: row < #definitions
        if builder.failures[row] != ""
            invalid = true
            emit("error line {definitions[row].number}: " + builder.failures[row])
    if invalid
        return
    engine := Engine(chart)
    .engine.macro_step("start")
    for text := scripts[]
        if engine.stopped
            break
        .engine.script_line(text)
void read_charts()
    line := ""
    scan(&line)
    count := nat(line)
    nat number := 1
    for: i < count
        scan(&line)
        number += 1
        name := line[6:]
        DefinitionLine[] definitions := []
        while true
            scan(&line)
            number += 1
            if line == "RUN"
                break
            definitions = definitions.append(DefinitionLine(number, line))
        string[] scripts := []
        while true
            scan(&line)
            number += 1
            if line == "END"
                break
            scripts = scripts.append(line)
        run_chart(name, definitions, scripts)
