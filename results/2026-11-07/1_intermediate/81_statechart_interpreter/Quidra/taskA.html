773 void .script_line(string text)
774 words := text.split(" ")
775 command := words[0]
776 if command == "event"
777 .macro_step("event", Event(words[1], if #words == 3 then int(words[2]) else 0))
778 elif command == "tick"
779 int finish := clock + int(words[1])
780 while not stopped and #reservations > 0
781 nat index := 0
782 for: i < #reservations
783 if reservations[i].due < reservations[index].due or (reservations[i].due == reservations[index].due and reservations[i].creation < reservations[index].creation)
784 index = i
785 booking := reservations[index]
786 if booking.due > finish
787 break
788 reservations = reservations[:<index].concat(reservations[index + 1:])
789 clock = booking.due
790 .macro_step(if booking.transition_id < 0 then "event" else "timer", booking.held_event, booking.transition_id)
791 if not stopped
792 clock = finish
793 elif command == "set"
794 match values.get(words[1])
795 int old
796 .values.set(words[1], int(words[2]))
797 none
798 emit("error: unknown variable " + words[1])
799 elif command == "trace"
800 trace_enabled = words[1] == "on"
801 elif command == "dump"
802 write_dump()
803 elif command == "eval"
804 original := text[5:]
805 expr := read_expression(original)
806 failure := check_expression(chart, expr)
807 if failure != ""
808 emit("error: " + failure)
809 else
810 data_value = 0
811 match evaluate_expression(expr)
812 int value
813 emit(original + " -> {value}")
814 error evaluation_failure
815 emit("error: division by zero")
