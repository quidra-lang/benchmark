75String clock(int value) => '${(value ~/ 60).toString().padLeft(2, '0')}:${(value % 60).toString().padLeft(2, '0')}';

97List<Line> orderedLines() => lines.values.toList()..sort((a, b) => a.id.compareTo(b.id));
98List<Direction> directions(Line item) {
99 final total = item.offsets.last;
100 return [Direction(item, 0, item.stops, item.offsets),
101 Direction(item, 1, item.stops.reversed.toList(),
102 item.offsets.reversed.map((offset) => total - offset).toList())];
103}
104int trips(Line item) => (item.last - item.first) ~/ item.headway + 1;
105int? nextTrain(Direction direction, int at, int ready) {
106 final item = direction.line;
107 final q = math.max(0, ((ready - item.first - item.delay - direction.offsets[at]) / item.headway).ceil());
108 return q < trips(item) ? q : null;
109}
110int trainTime(Direction direction, int q, int at) =>
111 direction.line.first + direction.line.delay + q * direction.line.headway + direction.offsets[at];
112Ride ride(Direction direction, int boarding, int alighting, int q) =>
113 Ride(direction.line.id, direction.stops.last, direction.stops[boarding],
114 trainTime(direction, q, boarding), direction.stops[alighting],
115 trainTime(direction, q, alighting), alighting - boarding);
116bool update(Map<String, Label> labels, String id, Label candidate) {
117 final old = labels[id];
118 if (old == null || old.time > candidate.time) {
119 labels[id] = candidate;
120 return true;
121 }
122 return false;
123}
124Map<String, Label> earliest(String origin, int departure) {
125 final labels = <String, Label>{origin: Label(departure, [])};
126 final settled = <String>{}, service = orderedLines();
127 while (true) {
128 final choices = labels.keys.where((id) => !settled.contains(id)).toList();
129 if (choices.isEmpty) return labels;
130 choices.sort((a, b) {
131 final order = labels[a]!.time.compareTo(labels[b]!.time);
132 return order == 0 ? a.compareTo(b) : order;
133 });
134 final u = choices.first, label = labels[choices.first]!, s = stations[choices.first]!;
135 settled.add(u);
136 final ready = label.time + (u == origin ? 0 : s.transfer);
137 for (final item in service) {
138 if (item.suspended || !item.stops.contains(u)) continue;
139 for (final direction in directions(item)) {
140 final i = direction.stops.indexOf(u), last = direction.stops.length - 1;
141 if (i == last) continue;
142 final q = nextTrain(direction, i, ready);
143 if (q == null) continue;
144 for (var j = i + 1; j <= last; j++) {
145 final v = direction.stops[j];
146 if (closed.contains(pair(direction.stops[j - 1], v))) break;
147 if (stations[v]!.closed) continue;
148 final leg = ride(direction, i, j, q);
149 update(labels, v, Label(leg.arrival, [...label.journey, leg]));
150 }
151 }
152 }
153 for (final v in s.walks.keys.toList()..sort()) {
154 final arrival = label.time + s.walks[v]!;
155 if (!stations[v]!.closed && arrival <= 1440) {
156 final leg = Walk(u, label.time, v, arrival);
157 update(labels, v, Label(arrival, [...label.journey, leg]));
158 }
159 }
160 }
161}
162void walkStep(Map<String, Label> labels, Set<String> sources, Set<String> marked) {
163 final recorded = <String, Label>{for (final id in sources) id: labels[id]!};
164 for (final u in sources.toList()..sort()) {
165 final label = recorded[u]!, s = stations[u]!;
166 for (final v in s.walks.keys.toList()..sort()) {
167 final arrival = label.time + s.walks[v]!;
168 if (stations[v]!.closed || arrival > 1440) continue;
169 final leg = Walk(u, label.time, v, arrival);
170 if (update(labels, v, Label(arrival, [...label.journey, leg]))) marked.add(v);
171 }
172 }
173}
174List<Map<String, Label>> rounds(String origin, int departure) {
175 var labels = <String, Label>{origin: Label(departure, [])};
176 var marked = <String>{origin};
177 walkStep(labels, {origin}, marked);
178 final result = <Map<String, Label>>[labels], service = orderedLines();
179 while (marked.isNotEmpty) {
180 final previous = labels, previousMarked = marked;
181 labels = Map.of(previous);
182 marked = {};
183 final walking = <String>{};
184 for (final item in service) {
185 if (item.suspended) continue;
186 for (final direction in directions(item)) {
187 int? boarded;
188 var boarding = 0;
189 var journey = <Leg>[];
190 for (var j = 0; j < direction.stops.length; j++) {
191 final v = direction.stops[j];
192 if (j > 0 && closed.contains(pair(direction.stops[j - 1], v))) boarded = null;
193 if (stations[v]!.closed) continue;
194 if (boarded != null) {
195 final leg = ride(direction, boarding, j, boarded);
196 if (update(labels, v, Label(leg.arrival, [...journey, leg]))) {
197 walking.add(v);
198 marked.add(v);
199 }
200 }
201 if (j + 1 < direction.stops.length && previousMarked.contains(v)) {
202 final old = previous[v]!;
203 final ready = old.time + (v == origin ? 0 : stations[v]!.transfer);
204 final q = nextTrain(direction, j, ready);
205 if (q != null && (boarded == null || q < boarded)) {
206 boarded = q;
207 boarding = j;
208 journey = old.journey;
209 }
210 }
211 }
212 }
213 }
214 walkStep(labels, walking, marked);
215 result.add(labels);
216 }
217 return result;
218}
219void printJourney(String heading, int departure, Label label) {
220 final rides = label.journey.whereType<Ride>().length, walks = label.journey.whereType<Walk>().length;
221 output.add('$heading ${clock(departure)} -> ${clock(label.time)} total ${label.time - departure} rides $rides walks $walks');
222 var previous = departure;
223 for (final leg in label.journey) {
224 if (leg is Ride) {
225 output.add('RIDE ${leg.line} toward ${leg.terminal} ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)} wait ${leg.departure - previous} stops ${leg.stops}');
226 } else if (leg is Walk) {
227 output.add('WALK ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)}');
228 } else if (leg is Stay) {
229 output.add('STAY ${leg.station} ${clock(leg.departure)} -> ${clock(leg.arrival)}');
230 }
231 previous = leg.arrival;
232 }

234void printFare(String origin, String destination, String word, String fareClass, Label label) {
235 final tickets = <String>[];
236 var total = 0;
237 Ride? previous;
238 int half(int value) => ((value + 19) ~/ 20) * 10;
239 for (final leg in label.journey.whereType<Ride>()) {
240 final item = lines[leg.line]!;
241 final direction = directions(item).firstWhere((d) => d.stops.last == leg.terminal);
242 final start = direction.stops.indexOf(leg.from), end = direction.stops.indexOf(leg.to);
243 final zones = direction.stops.sublist(start, end + 1).map((id) => stations[id]!.zone).toSet().length;
244 final base = tariff!.prices[math.min(zones, tariff!.prices.length) - 1];
245 final price = fareClass == 'CHILD' ? half(base) : base;
246 final cap = fareClass == 'CHILD' ? half(tariff!.discount) : tariff!.discount;
247 final discount = previous != null && leg.departure - previous.arrival <= tariff!.window ? math.min(cap, price) : 0;
248 final payment = price - discount;
249 total += payment;
250 tickets.add('TICKET ${leg.line} ${leg.from} ${leg.to} zones $zones price $price discount $discount pay $payment');
251 previous = leg;
252 }
253 output.add('FARE $origin $destination $word $fareClass rides ${tickets.length} total $total');
254 output.addAll(tickets);
255}
256void execute(String commandLine) {
257 final words = commandLine.split(' '), command = words.first, args = words.skip(1).toList();
258 final a = args.isEmpty ? '' : args[0], b = args.length < 2 ? '' : args[1];
259 switch (command) {
260 case 'STATION':
261 if (stations.containsKey(a)) fail('station $a already exists');
262 stations[a] = Station(a, int.parse(b), args[2]);
263 output.add('STATION $a zone $b ${args[2]}');
264 case 'LINE':
265 if (lines.containsKey(a)) fail('line $a already exists');
266 final stops = <String>[], offsets = <int>[0];
267 for (var i = 4; i < args.length; i += 2) { stops.add(args[i]); }
268 for (final id in stops) { station(id); }
269 final seen = <String>{};
270 for (final id in stops) {
271 if (!seen.add(id)) fail('station $id appears twice on line $a');
272 }
273 for (var i = 5; i < args.length; i += 2) { offsets.add(offsets.last + int.parse(args[i])); }
274 final first = time(b), last = time(args[2]), headway = int.parse(args[3]), total = offsets.last;
275 if (first > last) fail('line $a first departure is after last departure');
276 if (last + total > 1440) fail('line $a runs past 24:00');
277 final item = Line(a, first, last, headway, stops, offsets);
278 lines[a] = item;
279 output.add('LINE $a stops ${stops.length} time $total trips ${trips(item)}');
280 case 'TRANSFER':
281 station(a).transfer = int.parse(b);
282 output.add('TRANSFER $a $b');
283 case 'WALK':
284 final x = station(a), y = station(b), duration = int.parse(args[2]);
285 if (a == b) fail('walk from $a to itself');
286 final old = x.walks[b];
287 x.walks[b] = duration;
288 y.walks[a] = duration;
289 output.add('WALK $a $b $duration${old == null ? '' : ' was $old'}');
290 case 'TARIFF':
291 final prices = args.skip(2).map(int.parse).toList();
292 for (var i = 1; i < prices.length; i++) {
293 if (prices[i] < prices[i - 1]) fail('tariff prices must not decrease');
294 }
295 tariff = Tariff(int.parse(a), int.parse(b), prices);
296 output.add('TARIFF zones ${prices.length} discount $a window $b');
297 case 'CLOSE':
298 case 'OPEN':
299 final s = station(a), closing = command == 'CLOSE';
300 if (args.length == 1) {
301 if (s.closed == closing) fail('station $a is ${closing ? 'already closed' : 'not closed'}');
302 s.closed = closing;
303 output.add('$command $a');
304 } else {
305 station(b);
306 final key = pair(a, b);
307 final affected = orderedLines().where((item) {
308 for (var i = 1; i < item.stops.length; i++) {
309 if (pair(item.stops[i - 1], item.stops[i]) == key) return true;
310 }
311 return false;
312 }).toList();
313 if (closing && affected.isEmpty) fail('no segment $a-$b');
314 if (closed.contains(key) == closing) fail('segment $a-$b is ${closing ? 'already closed' : 'not closed'}');
315 if (closing) { closed.add(key); } else { closed.remove(key); }
316 output.add('$command $a-$b${closing ? ' lines ${affected.map((item) => item.id).join(' ')}' : ''}');
317 }
318 case 'DELAY':
319 final item = line(a), delay = int.parse(b);
320 if (item.last + delay + item.offsets.last > 1440) fail('line $a runs past 24:00');
321 item.delay = delay;
322 output.add('DELAY $a $delay');
323 case 'SUSPEND':
324 case 'RESUME':
325 final item = line(a), suspended = command == 'SUSPEND';
326 if (item.suspended == suspended) fail('line $a is ${suspended ? 'already suspended' : 'not suspended'}');
327 item.suspended = suspended;
328 output.add('$command $a');
329 case 'ROUTE':
330 case 'FEWEST':
331 case 'PARETO':
332 case 'FARE':
333 case 'LASTTRAIN':
334 case 'ARRIVEBY':
335 case 'PROFILE':
336 common(a, b);
337 if (command == 'FARE') {
338 if (args[3] != 'ADULT' && args[3] != 'CHILD') fail('unknown fare class ${args[3]}');
339 if (tariff == null) fail('no tariff');
340 }
341 if (command == 'PROFILE' && time(args[2]) > time(args[3])) fail('empty interval ${args[2]} ${args[3]}');
342 if (command == 'LASTTRAIN' || command == 'ARRIVEBY') {
343 final deadline = command == 'LASTTRAIN' ? 1440 : time(args[2]);
344 final heading = '$command $a $b${command == 'ARRIVEBY' ? ' by ${args[2]}' : ''}';
345 var found = false;
346 for (var departure = deadline - 1; departure >= 0; departure--) {
347 final label = earliest(a, departure)[b];
348 if (label != null && label.time <= deadline) {
349 printJourney(heading, departure, label);
350 found = true;
351 break;
352 }
353 }
354 if (!found) output.add('$heading none');
355 } else if (command == 'PROFILE') {
356 final start = time(args[2]), end = time(args[3]), arrivals = <int?>[];
357 for (var departure = start; departure <= end; departure++) {
358 arrivals.add(earliest(a, departure)[b]?.time);
359 }
360 final selected = <int>[];
361 for (var i = 0; i < arrivals.length; i++) {
362 if (arrivals[i] != null && (i == arrivals.length - 1 || arrivals[i + 1] == null || arrivals[i]! < arrivals[i + 1]!)) selected.add(i);
363 }
364 output.add('PROFILE $a $b ${args[2]} ${args[3]} count ${selected.length}');
365 for (final i in selected) {
366 output.add('${clock(start + i)} -> ${clock(arrivals[i]!)} total ${arrivals[i]! - start - i}');
367 }
368 } else if (command == 'FEWEST' || command == 'PARETO') {
369 final departure = time(args[2]), all = rounds(a, departure);
370 if (command == 'FEWEST') {
371 final k = all.indexWhere((labels) => labels.containsKey(b));
372 if (k < 0) {
373 output.add('FEWEST $a $b ${args[2]} unreachable');
374 } else {
375 printJourney('FEWEST $a $b trips $k', departure, all[k][b]!);
376 }
377 } else {
378 final selected = <int>[];
379 for (var k = 0; k < all.length; k++) {
380 if (all[k].containsKey(b) && (k == 0 || !all[k - 1].containsKey(b) || all[k - 1][b]!.time > all[k][b]!.time)) selected.add(k);
381 }
382 output.add('PARETO $a $b ${args[2]} options ${selected.length}');
383 for (final k in selected) {
384 printJourney('OPTION trips $k', departure, all[k][b]!);
385 }
386 }
387 } else {
388 final departure = time(args[2]), label = earliest(a, departure)[b];
389 if (label == null) {
390 output.add('$command $a $b ${args[2]}${command == 'FARE' ? ' ${args[3]}' : ''} unreachable');
391 } else if (command == 'FARE') {
392 printFare(a, b, args[2], args[3], label);
393 } else {
394 printJourney('ROUTE $a $b', departure, label);
395 }
396 }
397 case 'VIA':
398 final via = b, destination = args[2];
399 station(a);
400 station(via);
401 station(destination);
402 if (a == destination) fail('origin and destination are the same');
403 if (via == a || via == destination) fail('via station $via must differ from origin and destination');
404 openStation(a);
405 openStation(via);
406 openStation(destination);
407 final departure = time(args[3]), first = earliest(a, departure)[via];
408 final ready = first == null ? 0 : first.time + stations[via]!.transfer;
409 final second = first == null ? null : earliest(via, ready)[destination];
410 if (first == null || second == null) {
411 output.add('VIA $a $via $destination ${args[3]} unreachable');
412 } else {
413 final stay = Stay(via, first.time, ready);
414 printJourney('VIA $a $via $destination', departure,
415 Label(second.time, [...first.journey, stay, ...second.journey]));
416 }
417 case 'REACH':
418 station(a);
419 openStation(a);
420 final departure = time(b), labels = earliest(a, departure), limit = departure + int.parse(args[2]);
421 final reached = labels.keys.where((id) => labels[id]!.time <= limit).toList();
422 reached.sort((x, y) {
423 final order = labels[x]!.time.compareTo(labels[y]!.time);
424 return order == 0 ? x.compareTo(y) : order;
425 });
426 output.add('REACH $a $b ${args[2]} count ${reached.length}');
427 for (final id in reached) {
428 final label = labels[id]!;
429 output.add('$id ${stations[id]!.name} ${clock(label.time)} ${label.time - departure} rides ${label.journey.whereType<Ride>().length}');
430 }
431 case 'TIMETABLE':
432 final s = station(a), item = line(b), toward = args[2];
433 if (!item.stops.contains(a)) fail('station $a is not on line $b');
434 if (toward != item.stops.first && toward != item.stops.last) fail('$toward is not a terminal of line $b');
435 if (a == toward) fail('no departures from $a toward $toward');
436 if (item.suspended) fail('line $b is suspended');
437 if (s.closed) fail('station $a is closed');
438 final direction = directions(item).firstWhere((d) => d.stops.last == toward);
439 final at = direction.stops.indexOf(a), q = nextTrain(direction, direction.stops.indexOf(a), time(args[3]));
440 final times = <String>[];
441 if (q != null) {
442 for (var i = q; i < trips(item) && times.length < int.parse(args[4]); i++) {
443 times.add(clock(trainTime(direction, i, at)));
444 }
445 }
446 output.add('TIMETABLE $a $b toward $toward ${args[3]} next ${times.length}${times.isEmpty ? '' : ' ${times.join(' ')}'}');
447 case 'DEPARTURES':
448 case 'ARRIVALS':
449 station(a);
450 openStation(a);
451 final events = <Event>[];
452 for (final item in orderedLines()) {
453 if (item.suspended || !item.stops.contains(a)) continue;
454 for (final direction in directions(item)) {
455 final at = direction.stops.indexOf(a), last = direction.stops.length - 1;
456 if (command == 'DEPARTURES' ? at == last : at == 0) continue;
457 final q = nextTrain(direction, at, time(b));
458 if (q != null) {
459 for (var i = q; i < trips(item); i++) {
460 events.add(Event(trainTime(direction, i, at), item.id,
461 direction.order, direction.stops.last, direction.stops.first));
462 }
463 }
464 }
465 }
466 events.sort((x, y) {
467 var order = x.time.compareTo(y.time);
468 if (order == 0) order = x.line.compareTo(y.line);
469 return order == 0 ? x.order.compareTo(y.order) : order;
470 });
471 final selected = events.take(int.parse(args[2])).toList();
472 output.add('$command $a $b count ${selected.length}');
473 for (final event in selected) {
474 output.add('${clock(event.time)} ${event.line} ${command == 'DEPARTURES' ? 'toward ${event.terminal}' : 'from ${event.start}'}');
475 }
476 case 'TRIP':
477 final item = line(a);
478 if (b != item.stops.first && b != item.stops.last) fail('$b is not a terminal of line $a');
479 if (item.suspended) fail('line $a is suspended');
480 final direction = directions(item).firstWhere((d) => d.stops.last == b);
481 final q = nextTrain(direction, 0, time(args[2]));
482 if (q == null) {
483 output.add('TRIP $a toward $b ${args[2]} none');
484 } else {
485 output.add('TRIP $a toward $b ${args[2]} trip ${q + 1} departs ${clock(trainTime(direction, q, 0))}');
486 for (var j = 0; j < direction.stops.length; j++) {
487 final id = direction.stops[j];
488 output.add('${clock(trainTime(direction, q, j))} $id${stations[id]!.closed ? ' closed' : ''}${j > 0 && closed.contains(pair(direction.stops[j - 1], id)) ? ' blocked' : ''}');
489 }
490 }
491 case 'SHOWLINE':
492 final item = line(a);
493 output.add('SHOWLINE $a ${item.suspended ? 'suspended' : 'running'} first ${clock(item.first)} last ${clock(item.last)} headway ${item.headway} delay ${item.delay} trips ${trips(item)}');
494 for (var i = 0; i < item.stops.length; i++) {
495 final id = item.stops[i];
496 output.add('${item.offsets[i]} $id zone ${stations[id]!.zone}${stations[id]!.closed ? ' closed' : ''}${i > 0 && closed.contains(pair(item.stops[i - 1], id)) ? ' blocked' : ''}');
497 }
498 case 'INFO':
499 final s = station(a), service = orderedLines().where((item) => item.stops.contains(a)).toList();
500 final neighbors = s.walks.keys.toList()..sort();
501 output.add('INFO $a ${s.name} zone ${s.zone} transfer ${s.transfer} ${s.closed ? 'closed' : 'open'}');
502 output.add('LINES ${service.isEmpty ? 'none' : service.map((item) => item.id).join(' ')}');
503 output.add('WALKS ${neighbors.isEmpty ? 'none' : neighbors.map((id) => '$id ${s.walks[id]}').join(' ')}');
504 case 'STATUS':
505 final walkCount = stations.values.fold<int>(0, (sum, s) => sum + s.walks.length) ~/ 2;
506 output.add('STATUS stations ${stations.length} lines ${lines.length} walks $walkCount');
507 for (final id in stations.keys.toList()..sort()) {
508 if (stations[id]!.closed) output.add('CLOSED $id');
509 }
510 final segments = closed.toList()..sort((x, y) {
511 final a = x.split('-'), b = y.split('-');
512 final order = a[0].compareTo(b[0]);
513 return order == 0 ? a[1].compareTo(b[1]) : order;
514 });
515 for (final segment in segments) { output.add('CLOSED $segment'); }
516 for (final item in orderedLines()) {
517 if (item.suspended) output.add('SUSPENDED ${item.id}');
518 }
519 for (final item in orderedLines()) {
520 if (item.delay != 0) output.add('DELAYED ${item.id} ${item.delay}');
521 }
522 }

524void main() {
525 final count = int.parse(stdin.readLineSync()!);
