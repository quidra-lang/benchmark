1private class Problem(val text: String) : RuntimeException(text)
2private fun fail(reason: String): Nothing = throw Problem(reason)
3private fun time(s: String): Int {
4 val p = s.split(':')
5 return p[0].toInt()*60+p[1].toInt()
6}
7private fun clock(t: Int) = "%02d:%02d".format(t/60,t%60)
8private fun pair(a: String, b: String) = if (a < b) a to b else b to a
9private data class Station(val zone: Int, val name: String, var transfer: Int = 0, var closed: Boolean = false)
10private data class Line(val first: Int, val last: Int, val headway: Int, val stops: List<String>, val offsets: List<Int>, var delay: Int = 0, var suspended: Boolean = false) {
11 val total get() = offsets.last()
12 val trips get() = (last-first)/headway+1
13 fun direction(reverse: Boolean) = if (reverse) stops.asReversed() to offsets.asReversed().map {
14 total-it
15 } else stops to offsets
16 fun next(offset: Int, ready: Int): Int? {
17 val delta = ready-first-delay-offset
18 val q = if (delta <= 0) 0 else (delta+headway-1)/headway
19 return q.takeIf {
20 it < trips
21 }
22 }
23 fun at(q: Int, offset: Int) = first+delay+q*headway+offset
24}
25private sealed interface Leg {
26 val end: Int
27}
28private data class Ride(val line: String, val terminal: String, val from: String, val departure: Int, val to: String, val arrival: Int, val stops: Int) : Leg {
29 override val end get() = arrival
30}
31private data class Walk(val from: String, val departure: Int, val to: String, val arrival: Int) : Leg {
32 override val end get() = arrival
33}
34private data class Stay(val station: String, val start: Int, override val end: Int) : Leg
35private data class Label(val arrival: Int, val journey: List<Leg>)
36private typealias Labels = Map<String,Label>
37private data class Tariff(val discount: Int, val window: Int, val prices: List<Int>)
38private fun rides(journey: List<Leg>) = journey.count {
39 it is Ride
40}
41private fun relax(labels: MutableMap<String,Label>, station: String, arrival: Int, journey: List<Leg>, leg: Leg): Boolean {
42 if (labels[station]?.let {
43 it.arrival <= arrival
44 } == true) return false
45 labels[station] = Label(arrival,journey+leg)
46 return true
47}
48private fun printJourney(heading: String, departure: Int, label: Label): List<String> {
49 val out = mutableListOf("$heading ${clock(departure)} -> ${clock(label.arrival)} total ${label.arrival-departure} rides ${rides(label.journey)} walks ${label.journey.count { it is Walk }}")
50 var previous = departure
51 for (leg in label.journey) {
52 out.add(when(leg) {
53 is Ride -> "RIDE ${leg.line} toward ${leg.terminal} ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)} wait ${leg.departure-previous} stops ${leg.stops}"
54 is Walk -> "WALK ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)}"
55 is Stay -> "STAY ${leg.station} ${clock(leg.start)} -> ${clock(leg.end)}"
56 })
57 previous = leg.end
58 }
59 return out
60}
61private class Planner {
62 val stations = sortedMapOf<String,Station>()
63 val lines = sortedMapOf<String,Line>()
64 val walks = mutableMapOf<Pair<String,String>,Int>()
65 val closed = mutableSetOf<Pair<String,String>>()
66 var tariff: Tariff? = null
67 fun station(s: String) = stations[s] ?: fail("unknown station $s")
68 fun line(s: String) = lines[s] ?: fail("unknown line $s")
69 fun openStation(s: String) {
70 if (station(s).closed) fail("station $s is closed")
71 }
72 fun common(a: String, b: String) {
73 station(a)
74 station(b)
75 if (a == b) fail("origin and destination are the same")
76 openStation(a)
77 openStation(b)
78 }
79 fun neighbors(s: String) = walks.mapNotNull {
80 (key,t) -> when(s) {
81 key.first -> key.second to t
82 key.second -> key.first to t
83 else -> null
84 }
85 }.sortedBy {
86 it.first
87 }
88 fun earliest(origin: String, departure: Int): Labels {
89 val labels = mutableMapOf(origin to Label(departure,emptyList()))
90 val settled = mutableSetOf<String>()
91 while (true) {
92 val entry = labels.entries.filter {
93 it.key !in settled
94 }.minWithOrNull(compareBy<Map.Entry<String,Label>> {
95 it.value.arrival
96 }.thenBy {
97 it.key
98 }) ?: break
99 val u = entry.key
100 val label = entry.value
101 settled.add(u)
102 val ready = label.arrival+(if (u == origin) 0 else station(u).transfer)
103 for ((id,line) in lines) {
104 if (line.suspended || u !in line.stops) continue
105 for (reverse in listOf(false,true)) {
106 val (stops,offsets) = line.direction(reverse)
107 val i = stops.indexOf(u)
108 if (i == stops.lastIndex) continue
109 val q = line.next(offsets[i],ready) ?: continue
110 val d = line.at(q,offsets[i])
111 for (j in i+1 until stops.size) {
112 if (pair(stops[j-1],stops[j]) in closed) break
113 if (station(stops[j]).closed) continue
114 val t = line.at(q,offsets[j])
115 relax(labels,stops[j],t,label.journey,Ride(id,stops.last(),u,d,stops[j],t,j-i))
116 }
117 }
118 }
119 for ((v,w) in neighbors(u)) {
120 val t = label.arrival+w
121 if (!station(v).closed && t <= 1440) relax(labels,v,t,label.journey,Walk(u,label.arrival,v,t))
122 }
123 }
124 return labels
125 }
126 fun walkStep(labels: MutableMap<String,Label>, sources: Set<String>, changed: MutableSet<String>) {
127 val recorded = sources.sorted().map {
128 it to labels.getValue(it)
129 }
130 for ((u,label) in recorded) for ((v,w) in neighbors(u)) {
131 val t = label.arrival+w
132 if (!station(v).closed && t <= 1440 && relax(labels,v,t,label.journey,Walk(u,label.arrival,v,t))) changed.add(v)
133 }
134 }
135 fun rounds(origin: String, departure: Int): List<Labels> {
136 val labels = mutableMapOf(origin to Label(departure,emptyList()))
137 var changed = mutableSetOf(origin)
138 walkStep(labels,setOf(origin),changed)
139 val result = mutableListOf(labels.toMap())
140 data class Boarded(val q: Int, val index: Int, val departure: Int, val journey: List<Leg>)
141 while (changed.isNotEmpty()) {
142 val previous = labels.toMap()
143 val marked = changed
144 changed = mutableSetOf()
145 val sources = mutableSetOf<String>()
146 for ((id,line) in lines) {
147 if (line.suspended) continue
148 for (reverse in listOf(false,true)) {
149 val (stops,offsets) = line.direction(reverse)
150 var boarded: Boarded? = null
151 for (j in stops.indices) {
152 if (j > 0 && pair(stops[j-1],stops[j]) in closed) boarded = null
153 if (station(stops[j]).closed) continue
154 boarded?.let {
155 b -> val t = line.at(b.q,offsets[j])
156 if (relax(labels,stops[j],t,b.journey,Ride(id,stops.last(),stops[b.index],b.departure,stops[j],t,j-b.index))) {
157 sources.add(stops[j])
158 changed.add(stops[j])
159 }
160 }
161 if (j < stops.lastIndex && stops[j] in marked) {
162 val old = previous.getValue(stops[j])
163 val ready = old.arrival+(if (stops[j] == origin) 0 else station(stops[j]).transfer)
164 val q = line.next(offsets[j],ready)
165 if (q != null && (boarded == null || q < boarded!!.q)) boarded = Boarded(q,j,line.at(q,offsets[j]),old.journey)
166 }
167 }
168 }
169 }
170 walkStep(labels,sources,changed)
171 result.add(labels.toMap())
172 }
173 return result
174 }
175 fun fare(journey: List<Leg>, child: Boolean): Pair<Int,List<String>> {
176 val tariff = tariff!!
177 fun half(n: Int) = (n+19)/20*10
178 val cap = if (child) half(tariff.discount) else tariff.discount
179 var previous: Int? = null
180 var total = 0
181 val tickets = mutableListOf<String>()
182 for (leg in journey) if (leg is Ride) {
183 val stops = line(leg.line).stops
184 val a = stops.indexOf(leg.from)
185 val b = stops.indexOf(leg.to)
186 val zones = stops.subList(minOf(a,b),maxOf(a,b)+1).map {
187 station(it).zone
188 }.toSet().size
189 var price = tariff.prices[minOf(zones,tariff.prices.size)-1]
190 if (child) price = half(price)
191 val reduction = if (previous?.let {
192 leg.departure-it <= tariff.window
193 } == true) minOf(cap,price) else 0
194 val pay = price-reduction
195 total += pay
196 previous = leg.arrival
197 tickets.add("TICKET ${leg.line} ${leg.from} ${leg.to} zones $zones price $price discount $reduction pay $pay")
198 }
199 return total to tickets
200 }
201 fun command(w: List<String>): List<String> {
202 val c = w[0]
203 val a = w.drop(1)
204 fun num(i: Int) = a[i].toInt()
205 val out = mutableListOf<String>()
206 when(c) {
207 "STATION" -> {
208 if (a[0] in stations) fail("station ${a[0]} already exists")
209 stations[a[0]] = Station(num(1),a[2])
210 out.add("STATION ${a[0]} zone ${num(1)} ${a[2]}")
211 }
212 "LINE" -> {
213 if (a[0] in lines) fail("line ${a[0]} already exists")
214 val stops = (4 until a.size step 2).map {
215 a[it]
216 }
217 for (s in stops) station(s)
218 val seen = mutableSetOf<String>()
219 for (s in stops) if (!seen.add(s)) fail("station $s appears twice on line ${a[0]}")
220 val first = time(a[1])
221 val last = time(a[2])
222 val offsets = mutableListOf(0)
223 for (i in 5 until a.size step 2) offsets.add(offsets.last()+num(i))
224 if (first > last) fail("line ${a[0]} first departure is after last departure")
225 if (last+offsets.last() > 1440) fail("line ${a[0]} runs past 24:00")
226 val line = Line(first,last,num(3),stops,offsets)
227 lines[a[0]] = line
228 out.add("LINE ${a[0]} stops ${stops.size} time ${line.total} trips ${line.trips}")
229 }
230 "TRANSFER" -> {
231 station(a[0]).transfer = num(1)
232 out.add("TRANSFER ${a[0]} ${num(1)}")
233 }
234 "WALK" -> {
235 station(a[0])
236 station(a[1])
237 if (a[0] == a[1]) fail("walk from ${a[0]} to itself")
238 val old = walks.put(pair(a[0],a[1]),num(2))
239 out.add("WALK ${a[0]} ${a[1]} ${num(2)}${old?.let { " was $it" } ?: ""}")
240 }
241 "TARIFF" -> {
242 val prices = (2 until a.size).map(::num)
243 if (prices.zipWithNext().any {
244 it.first > it.second
245 }) fail("tariff prices must not decrease")
246 tariff = Tariff(num(0),num(1),prices)
247 out.add("TARIFF zones ${prices.size} discount ${num(0)} window ${num(1)}")
248 }
249 "CLOSE","OPEN" -> {
250 station(a[0])
251 val close = c == "CLOSE"
252 if (a.size == 1) {
253 val station = station(a[0])
254 if (station.closed == close) fail("station ${a[0]} is ${if (close) "already" else "not"} closed")
255 station.closed = close
256 out.add("$c ${a[0]}")
257 }
258 else {
259 station(a[1])
260 val key = pair(a[0],a[1])
261 val ids = lines.filterValues {
262 it.stops.zipWithNext().any {
263 p -> pair(p.first,p.second) == key
264 }
265 }.keys.toList()
266 if (close && ids.isEmpty()) fail("no segment ${a[0]}-${a[1]}")
267 if ((key in closed) == close) fail("segment ${a[0]}-${a[1]} is ${if (close) "already" else "not"} closed")
268 if (close) closed.add(key) else closed.remove(key)
269 out.add("$c ${a[0]}-${a[1]}${if (close) " lines "+ids.joinToString(" ") else ""}")
270 }
271 }
272 "DELAY" -> {
273 val line = line(a[0])
274 if (line.last+num(1)+line.total > 1440) fail("line ${a[0]} runs past 24:00")
275 line.delay = num(1)
276 out.add("DELAY ${a[0]} ${num(1)}")
277 }
278 "SUSPEND","RESUME" -> {
279 val line = line(a[0])
280 val suspended = c == "SUSPEND"
281 if (line.suspended == suspended) fail("line ${a[0]} is ${if (suspended) "already" else "not"} suspended")
282 line.suspended = suspended
283 out.add("$c ${a[0]}")
284 }
285 "ROUTE","FEWEST","PARETO","FARE","LASTTRAIN","ARRIVEBY","PROFILE" -> {
286 common(a[0],a[1])
287 if (c == "FARE") {
288 if (a[3] !in setOf("ADULT","CHILD")) fail("unknown fare class ${a[3]}")
289 if (tariff == null) fail("no tariff")
290 }
291 if (c == "PROFILE" && time(a[2]) > time(a[3])) fail("empty interval ${a[2]} ${a[3]}")
292 val heading = "$c ${a[0]} ${a[1]}"
293 if (c == "LASTTRAIN" || c == "ARRIVEBY") {
294 val bound = if (c == "LASTTRAIN") 1440 else time(a[2])
295 val title = if (c == "ARRIVEBY") "$heading by ${a[2]}" else heading
296 for (t in bound-1 downTo 0) {
297 val label = earliest(a[0],t)[a[1]]
298 if (label != null && (c == "LASTTRAIN" || label.arrival <= bound)) return printJourney(title,t,label)
299 }
300 out.add("$title none")
301 } else if (c == "PROFILE") {
302 val start = time(a[2])
303 val end = time(a[3])
304 val arrivals = (start..end).map {
305 earliest(a[0],it)[a[1]]?.arrival
306 }
307 val selected = mutableListOf<String>()
308 for (i in arrivals.indices) {
309 val arrival = arrivals[i] ?: continue
310 if (i == arrivals.lastIndex || arrivals[i+1] == null || arrival < arrivals[i+1]!!) selected.add("${clock(start+i)} -> ${clock(arrival)} total ${arrival-start-i}")
311 }
312 out.add("$heading ${a[2]} ${a[3]} count ${selected.size}")
313 out.addAll(selected)
314 } else if (c == "FEWEST" || c == "PARETO") {
315 val start = time(a[2])
316 val rounds = rounds(a[0],start)
317 if (c == "FEWEST") {
318 for ((k,labels) in rounds.withIndex()) labels[a[1]]?.let {
319 return printJourney("$heading trips $k",start,it)
320 }
321 out.add("$heading ${a[2]} unreachable")
322 }
323 else {
324 val options = mutableListOf<String>()
325 var previous: Int? = null
326 var count = 0
327 for ((k,labels) in rounds.withIndex()) {
328 val label = labels[a[1]] ?: continue
329 if (previous == null || previous!! > label.arrival) {
330 count++
331 options.addAll(printJourney("OPTION trips $k",start,label))
332 }
333 previous = label.arrival
334 }
335 out.add("$heading ${a[2]} options $count")
336 out.addAll(options)
337 }
338 } else {
339 val start = time(a[2])
340 val label = earliest(a[0],start)[a[1]]
341 if (label == null) out.add("$heading ${a[2]}${if (c == "FARE") " "+a[3] else ""} unreachable")
342 else if (c == "FARE") {
343 val (total,tickets) = fare(label.journey,a[3] == "CHILD")
344 out.add("$heading ${a[2]} ${a[3]} rides ${rides(label.journey)} total $total")
345 out.addAll(tickets)
346 }
347 else return printJourney(heading,start,label)
348 }
349 }
350 "VIA" -> {
351 station(a[0])
352 station(a[1])
353 station(a[2])
354 if (a[0] == a[2]) fail("origin and destination are the same")
355 if (a[1] == a[0] || a[1] == a[2]) fail("via station ${a[1]} must differ from origin and destination")
356 openStation(a[0])
357 openStation(a[1])
358 openStation(a[2])
359 val start = time(a[3])
360 val heading = "VIA ${a[0]} ${a[1]} ${a[2]}"
361 val first = earliest(a[0],start)[a[1]]
362 if (first != null) {
363 val next = first.arrival+station(a[1]).transfer
364 val second = earliest(a[1],next)[a[2]]
365 if (second != null) return printJourney(heading,start,Label(second.arrival,first.journey+Stay(a[1],first.arrival,next)+second.journey))
366 }
367 out.add("$heading ${a[3]} unreachable")
368 }
369 "REACH" -> {
370 openStation(a[0])
371 val start = time(a[1])
372 val selected = earliest(a[0],start).entries.filter {
373 it.value.arrival <= start+num(2)
374 }.sortedWith(compareBy<Map.Entry<String,Label>> {
375 it.value.arrival
376 }.thenBy {
377 it.key
378 })
379 out.add("REACH ${a[0]} ${a[1]} ${num(2)} count ${selected.size}")
380 for ((s,label) in selected) out.add("$s ${station(s).name} ${clock(label.arrival)} ${label.arrival-start} rides ${rides(label.journey)}")
381 }
382 "TIMETABLE" -> {
383 station(a[0])
384 val line = line(a[1])
385 if (a[0] !in line.stops) fail("station ${a[0]} is not on line ${a[1]}")
386 if (a[2] != line.stops.first() && a[2] != line.stops.last()) fail("${a[2]} is not a terminal of line ${a[1]}")
387 if (a[0] == a[2]) fail("no departures from ${a[0]} toward ${a[2]}")
388 if (line.suspended) fail("line ${a[1]} is suspended")
389 openStation(a[0])
390 val (stops,offsets) = line.direction(a[2] == line.stops.first())
391 val i = stops.indexOf(a[0])
392 val selected = (0 until line.trips).map {
393 line.at(it,offsets[i])
394 }.filter {
395 it >= time(a[3])
396 }.take(num(4))
397 out.add("TIMETABLE ${a[0]} ${a[1]} toward ${a[2]} ${a[3]} next ${selected.size}"+selected.joinToString("") {
398 " ${clock(it)}"
399 })
400 }
401 "DEPARTURES","ARRIVALS" -> {
402 openStation(a[0])
403 data class Event(val time: Int, val line: String, val direction: Int, val endpoint: String)
404 val events = mutableListOf<Event>()
405 for ((id,line) in lines) if (!line.suspended && a[0] in line.stops) for (direction in 0..1) {
406 val (stops,offsets) = line.direction(direction == 1)
407 val i = stops.indexOf(a[0])
408 if ((c == "DEPARTURES" && i == stops.lastIndex) || (c == "ARRIVALS" && i == 0)) continue
409 for (q in 0 until line.trips) {
410 val t = line.at(q,offsets[i])
411 if (t >= time(a[1])) events.add(Event(t,id,direction,if (c == "DEPARTURES") stops.last() else stops.first()))
412 }
413 }
414 val selected = events.sortedWith(compareBy<Event> {
415 it.time
416 }.thenBy {
417 it.line
418 }.thenBy {
419 it.direction
420 }).take(num(2))
421 out.add("$c ${a[0]} ${a[1]} count ${selected.size}")
422 for (e in selected) out.add("${clock(e.time)} ${e.line} ${if (c == "DEPARTURES") "toward" else "from"} ${e.endpoint}")
423 }
424 "TRIP" -> {
425 val line = line(a[0])
426 if (a[1] != line.stops.first() && a[1] != line.stops.last()) fail("${a[1]} is not a terminal of line ${a[0]}")
427 if (line.suspended) fail("line ${a[0]} is suspended")
428 val (stops,offsets) = line.direction(a[1] == line.stops.first())
429 val heading = "TRIP ${a[0]} toward ${a[1]} ${a[2]}"
430 val q = line.next(0,time(a[2]))
431 if (q == null) out.add("$heading none") else {
432 out.add("$heading trip ${q+1} departs ${clock(line.at(q,0))}")
433 for (j in stops.indices) out.add("${clock(line.at(q,offsets[j]))} ${stops[j]}${if (station(stops[j]).closed) " closed" else ""}${if (j > 0 && pair(stops[j-1],stops[j]) in closed) " blocked" else ""}")
434 }
435 }
436 "SHOWLINE" -> {
437 val line = line(a[0])
438 out.add("SHOWLINE ${a[0]} ${if (line.suspended) "suspended" else "running"} first ${clock(line.first)} last ${clock(line.last)} headway ${line.headway} delay ${line.delay} trips ${line.trips}")
439 for (j in line.stops.indices) {
440 val s = line.stops[j]
441 out.add("${line.offsets[j]} $s zone ${station(s).zone}${if (station(s).closed) " closed" else ""}${if (j > 0 && pair(line.stops[j-1],s) in closed) " blocked" else ""}")
442 }
443 }
444 "INFO" -> {
445 val s = station(a[0])
446 out.add("INFO ${a[0]} ${s.name} zone ${s.zone} transfer ${s.transfer} ${if (s.closed) "closed" else "open"}")
447 val ids = lines.filterValues {
448 a[0] in it.stops
449 }.keys
450 out.add("LINES ${if (ids.isEmpty()) "none" else ids.joinToString(" ")}")
451 val links = neighbors(a[0])
452 out.add("WALKS ${if (links.isEmpty()) "none" else links.joinToString(" ") { "${it.first} ${it.second}" }}")
453 }
454 "STATUS" -> {
455 out.add("STATUS stations ${stations.size} lines ${lines.size} walks ${walks.size}")
456 for ((s,station) in stations) if (station.closed) out.add("CLOSED $s")
457 for ((a,b) in closed.sortedWith(compareBy<Pair<String,String>> {
458 it.first
459 }.thenBy {
460 it.second
461 })) out.add("CLOSED $a-$b")
462 for ((id,line) in lines) if (line.suspended) out.add("SUSPENDED $id")
463 for ((id,line) in lines) if (line.delay != 0) out.add("DELAYED $id ${line.delay}")
464 }
465 }
466 return out
467 }
468}
