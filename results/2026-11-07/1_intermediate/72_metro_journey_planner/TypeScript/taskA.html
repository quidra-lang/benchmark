3interface Station { id: string; zone: number; name: string; transfer: number; closed: boolean; walks: Map<string, number> }
4interface Line { id: string; first: number; last: number; headway: number; stops: string[]; offsets: number[]; delay: number; suspended: boolean }
5interface Direction { line: Line; order: number; stops: string[]; offsets: number[] }
6interface Ride { kind: 'ride'; line: string; terminal: string; from: string; departure: number; to: string; arrival: number; stops: number }
7interface Walk { kind: 'walk'; from: string; departure: number; to: string; arrival: number }
8interface Stay { kind: 'stay'; station: string; departure: number; arrival: number }
9type Leg = Ride | Walk | Stay;
10interface Label { time: number; journey: Leg[] }
11interface Tariff { discount: number; window: number; prices: number[] }
12interface Event { time: number; line: string; order: number; terminal: string; start: string }
13const stations = new Map<string, Station>();
14const lines = new Map<string, Line>();
15const closed = new Set<string>();
16const output: string[] = [];
17let tariff: Tariff | null = null;
18function fail(message: string): never { throw new Error(`ERROR ${message}`); }
19function compare(a: string, b: string): number { return a < b ? -1 : a > b ? 1 : 0; }
20function time(word: string): number { const [h, m] = word.split(':').map(Number); return h * 60 + m; }
21function clock(value: number): string { return `${Math.floor(value / 60).toString().padStart(2, '0')}:${(value % 60).toString().padStart(2, '0')}`; }
22function pair(a: string, b: string): string { return a < b ? `${a}-${b}` : `${b}-${a}`; }
23function station(id: string): Station {
24 const result = stations.get(id);
25 if (!result) fail(`unknown station ${id}`);
26 return result;
27}
28function line(id: string): Line {
29 const result = lines.get(id);
30 if (!result) fail(`unknown line ${id}`);
31 return result;
32}
33function openStation(id: string): void { if (stations.get(id)!.closed) fail(`station ${id} is closed`); }
34function common(origin: string, destination: string): void {
35 station(origin); station(destination);
36 if (origin === destination) fail('origin and destination are the same');
37 openStation(origin); openStation(destination);
38}
39function orderedLines(): Line[] { return [...lines.values()].sort((a, b) => compare(a.id, b.id)); }
40function directions(item: Line): Direction[] {
41 const total = item.offsets[item.offsets.length - 1];
42 return [{ line: item, order: 0, stops: item.stops, offsets: item.offsets },
43 { line: item, order: 1, stops: [...item.stops].reverse(), offsets: [...item.offsets].reverse().map(offset => total - offset) }];
44}
45function trips(item: Line): number { return Math.floor((item.last - item.first) / item.headway) + 1; }
46function nextTrain(direction: Direction, at: number, ready: number): number | null {
47 const item = direction.line;
48 const q = Math.max(0, Math.ceil((ready - item.first - item.delay - direction.offsets[at]) / item.headway));
49 return q < trips(item) ? q : null;
50}
51function trainTime(direction: Direction, q: number, at: number): number {
52 return direction.line.first + direction.line.delay + q * direction.line.headway + direction.offsets[at];
53}
54function ride(direction: Direction, boarding: number, alighting: number, q: number): Ride {
55 return { kind: 'ride', line: direction.line.id, terminal: direction.stops[direction.stops.length - 1],
56 from: direction.stops[boarding], departure: trainTime(direction, q, boarding),
57 to: direction.stops[alighting], arrival: trainTime(direction, q, alighting), stops: alighting - boarding };
58}
59function update(labels: Map<string, Label>, id: string, candidate: Label): boolean {
60 const old = labels.get(id);
61 if (!old || old.time > candidate.time) { labels.set(id, candidate); return true; }
62 return false;
63}
64function earliest(origin: string, departure: number): Map<string, Label> {
65 const labels = new Map<string, Label>([[origin, { time: departure, journey: [] }]]), settled = new Set<string>();
66 const service = orderedLines();
67 while (true) {
68 const choices = [...labels.keys()].filter(id => !settled.has(id));
69 if (!choices.length) return labels;
70 choices.sort((a, b) => labels.get(a)!.time - labels.get(b)!.time || compare(a, b));
71 const u = choices[0], label = labels.get(u)!, s = stations.get(u)!;
72 settled.add(u);
73 const ready = label.time + (u === origin ? 0 : s.transfer);
74 for (const item of service) if (!item.suspended && item.stops.includes(u)) {
75 for (const direction of directions(item)) {
76 const i = direction.stops.indexOf(u), last = direction.stops.length - 1;
77 if (i === last) continue;
78 const q = nextTrain(direction, i, ready);
79 if (q === null) continue;
80 for (let j = i + 1; j <= last; j++) {
81 const v = direction.stops[j];
82 if (closed.has(pair(direction.stops[j - 1], v))) break;
83 if (stations.get(v)!.closed) continue;
84 const leg = ride(direction, i, j, q);
85 update(labels, v, { time: leg.arrival, journey: [...label.journey, leg] });
86 }
87 }
88 }
89 for (const v of [...s.walks.keys()].sort()) {
90 const arrival = label.time + s.walks.get(v)!;
91 if (!stations.get(v)!.closed && arrival <= 1440) {
92 const leg: Walk = { kind: 'walk', from: u, departure: label.time, to: v, arrival };
93 update(labels, v, { time: arrival, journey: [...label.journey, leg] });
94 }
95 }
96 }
97}
98function walkStep(labels: Map<string, Label>, sources: Set<string>, marked: Set<string>): void {
99 const recorded = new Map<string, Label>([...sources].map((id): [string, Label] => [id, labels.get(id)!]));
100 for (const u of [...sources].sort()) {
101 const label = recorded.get(u)!, s = stations.get(u)!;
102 for (const v of [...s.walks.keys()].sort()) {
103 const arrival = label.time + s.walks.get(v)!;
104 if (stations.get(v)!.closed || arrival > 1440) continue;
105 const leg: Walk = { kind: 'walk', from: u, departure: label.time, to: v, arrival };
106 if (update(labels, v, { time: arrival, journey: [...label.journey, leg] })) marked.add(v);
107 }
108 }
109}
110function rounds(origin: string, departure: number): Map<string, Label>[] {
111 let labels = new Map<string, Label>([[origin, { time: departure, journey: [] }]]), marked = new Set([origin]);
112 walkStep(labels, new Set([origin]), marked);
113 const result = [labels], service = orderedLines();
114 while (marked.size) {
115 const previous = labels, previousMarked = marked;
116 labels = new Map(previous); marked = new Set();
117 const walking = new Set<string>();
118 for (const item of service) if (!item.suspended) {
119 for (const direction of directions(item)) {
120 let boarded: number | null = null, boarding = 0;
121 let journey: Leg[] = [];
122 for (let j = 0; j < direction.stops.length; j++) {
123 const v = direction.stops[j];
124 if (j && closed.has(pair(direction.stops[j - 1], v))) boarded = null;
125 if (stations.get(v)!.closed) continue;
126 if (boarded !== null) {
127 const leg = ride(direction, boarding, j, boarded);
128 if (update(labels, v, { time: leg.arrival, journey: [...journey, leg] })) {
129 walking.add(v); marked.add(v);
130 }
131 }
132 if (j + 1 < direction.stops.length && previousMarked.has(v)) {
133 const old = previous.get(v)!, ready = old.time + (v === origin ? 0 : stations.get(v)!.transfer);
134 const q = nextTrain(direction, j, ready);
135 if (q !== null && (boarded === null || q < boarded)) { boarded = q; boarding = j; journey = old.journey; }
136 }
137 }
138 }
139 }
140 walkStep(labels, walking, marked);
141 result.push(labels);
142 }
143 return result;
144}
145function printJourney(heading: string, departure: number, label: Label): void {
146 const rides = label.journey.filter(leg => leg.kind === 'ride').length, walks = label.journey.filter(leg => leg.kind === 'walk').length;
147 output.push(`${heading} ${clock(departure)} -> ${clock(label.time)} total ${label.time - departure} rides ${rides} walks ${walks}`);
148 let previous = departure;
149 for (const leg of label.journey) {
150 if (leg.kind === 'ride') output.push(`RIDE ${leg.line} toward ${leg.terminal} ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)} wait ${leg.departure - previous} stops ${leg.stops}`);
151 else if (leg.kind === 'walk') output.push(`WALK ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)}`);
152 else output.push(`STAY ${leg.station} ${clock(leg.departure)} -> ${clock(leg.arrival)}`);
153 previous = leg.arrival;
154 }

156function printFare(origin: string, destination: string, word: string, fareClass: string, label: Label): void {
157 const tickets: string[] = [];
158 let total = 0, previous: Ride | null = null;
159 const half = (value: number): number => Math.ceil(value / 20) * 10;
160 for (const leg of label.journey) if (leg.kind === 'ride') {
161 const item = lines.get(leg.line)!, direction = directions(item).find(d => d.stops[d.stops.length - 1] === leg.terminal)!;
162 const start = direction.stops.indexOf(leg.from), end = direction.stops.indexOf(leg.to);
163 const zones = new Set(direction.stops.slice(start, end + 1).map(id => stations.get(id)!.zone)).size;
164 const base = tariff!.prices[Math.min(zones, tariff!.prices.length) - 1], price = fareClass === 'CHILD' ? half(base) : base;
165 const cap = fareClass === 'CHILD' ? half(tariff!.discount) : tariff!.discount;
166 const discount = previous && leg.departure - previous.arrival <= tariff!.window ? Math.min(cap, price) : 0;
167 const payment = price - discount;
168 total += payment;
169 tickets.push(`TICKET ${leg.line} ${leg.from} ${leg.to} zones ${zones} price ${price} discount ${discount} pay ${payment}`);
170 previous = leg;
171 }
172 output.push(`FARE ${origin} ${destination} ${word} ${fareClass} rides ${tickets.length} total ${total}`, ...tickets);
173}
174function execute(commandLine: string): void {
175 const [command, ...args] = commandLine.split(' '), a = args[0], b = args[1];
176 switch (command) {
177 case 'STATION':
178 if (stations.has(a)) fail(`station ${a} already exists`);
179 stations.set(a, { id: a, zone: Number(b), name: args[2], transfer: 0, closed: false, walks: new Map() });
180 output.push(`STATION ${a} zone ${b} ${args[2]}`);
181 break;
182 case 'LINE': {
183 if (lines.has(a)) fail(`line ${a} already exists`);
184 const stops: string[] = [], offsets = [0];
185 for (let i = 4; i < args.length; i += 2) stops.push(args[i]);
186 for (const id of stops) station(id);
187 const seen = new Set<string>();
188 for (const id of stops) { if (seen.has(id)) fail(`station ${id} appears twice on line ${a}`); seen.add(id); }
189 for (let i = 5; i < args.length; i += 2) offsets.push(offsets[offsets.length - 1] + Number(args[i]));
190 const first = time(b), last = time(args[2]), headway = Number(args[3]), total = offsets[offsets.length - 1];
191 if (first > last) fail(`line ${a} first departure is after last departure`);
192 if (last + total > 1440) fail(`line ${a} runs past 24:00`);
193 const item: Line = { id: a, first, last, headway, stops, offsets, delay: 0, suspended: false };
194 lines.set(a, item);
195 output.push(`LINE ${a} stops ${stops.length} time ${total} trips ${trips(item)}`);
196 break;
197 }
198 case 'TRANSFER': station(a).transfer = Number(b); output.push(`TRANSFER ${a} ${b}`); break;
199 case 'WALK': {
200 const x = station(a), y = station(b), duration = Number(args[2]);
201 if (a === b) fail(`walk from ${a} to itself`);
202 const old = x.walks.get(b);
203 x.walks.set(b, duration); y.walks.set(a, duration);
204 output.push(`WALK ${a} ${b} ${duration}${old === undefined ? '' : ` was ${old}`}`);
205 break;
206 }
207 case 'TARIFF': {
208 const prices = args.slice(2).map(Number);
209 if (prices.some((price, i) => i > 0 && price < prices[i - 1])) fail('tariff prices must not decrease');
210 tariff = { discount: Number(a), window: Number(b), prices };
211 output.push(`TARIFF zones ${prices.length} discount ${a} window ${b}`);
212 break;
213 }
214 case 'CLOSE': case 'OPEN': {
215 const s = station(a), closing = command === 'CLOSE';
216 if (args.length === 1) {
217 if (s.closed === closing) fail(`station ${a} is ${closing ? 'already closed' : 'not closed'}`);
218 s.closed = closing;
219 output.push(`${command} ${a}`);
220 } else {
221 station(b);
222 const key = pair(a, b);
223 const affected = orderedLines().filter(item => item.stops.some((id, i) => i > 0 && pair(item.stops[i - 1], id) === key));
224 if (closing && !affected.length) fail(`no segment ${a}-${b}`);
225 if (closed.has(key) === closing) fail(`segment ${a}-${b} is ${closing ? 'already closed' : 'not closed'}`);
226 if (closing) closed.add(key); else closed.delete(key);
227 output.push(`${command} ${a}-${b}${closing ? ` lines ${affected.map(item => item.id).join(' ')}` : ''}`);
228 }
229 break;
230 }
231 case 'DELAY': {
232 const item = line(a), delay = Number(b);
233 if (item.last + delay + item.offsets[item.offsets.length - 1] > 1440) fail(`line ${a} runs past 24:00`);
234 item.delay = delay; output.push(`DELAY ${a} ${delay}`);
235 break;
236 }
237 case 'SUSPEND': case 'RESUME': {
238 const item = line(a), suspended = command === 'SUSPEND';
239 if (item.suspended === suspended) fail(`line ${a} is ${suspended ? 'already suspended' : 'not suspended'}`);
240 item.suspended = suspended; output.push(`${command} ${a}`);
241 break;
242 }
243 case 'ROUTE': case 'FEWEST': case 'PARETO': case 'FARE': case 'LASTTRAIN': case 'ARRIVEBY': case 'PROFILE': {
244 common(a, b);
245 if (command === 'FARE') {
246 if (args[3] !== 'ADULT' && args[3] !== 'CHILD') fail(`unknown fare class ${args[3]}`);
247 if (!tariff) fail('no tariff');
248 }
249 if (command === 'PROFILE' && time(args[2]) > time(args[3])) fail(`empty interval ${args[2]} ${args[3]}`);
250 if (command === 'LASTTRAIN' || command === 'ARRIVEBY') {
251 const deadline = command === 'LASTTRAIN' ? 1440 : time(args[2]);
252 const heading = `${command} ${a} ${b}${command === 'ARRIVEBY' ? ` by ${args[2]}` : ''}`;
253 let found = false;
254 for (let departure = deadline - 1; departure >= 0; departure--) {
255 const label = earliest(a, departure).get(b);
256 if (label && label.time <= deadline) { printJourney(heading, departure, label); found = true; break; }
257 }
258 if (!found) output.push(`${heading} none`);
259 } else if (command === 'PROFILE') {
260 const start = time(args[2]), end = time(args[3]), arrivals: (number | undefined)[] = [];
261 for (let departure = start; departure <= end; departure++) arrivals.push(earliest(a, departure).get(b)?.time);
262 const selected: number[] = [];
263 for (let i = 0; i < arrivals.length; i++) if (arrivals[i] !== undefined &&
264 (i === arrivals.length - 1 || arrivals[i + 1] === undefined || arrivals[i]! < arrivals[i + 1]!)) selected.push(i);
265 output.push(`PROFILE ${a} ${b} ${args[2]} ${args[3]} count ${selected.length}`);
266 for (const i of selected) output.push(`${clock(start + i)} -> ${clock(arrivals[i]!)} total ${arrivals[i]! - start - i}`);
267 } else if (command === 'FEWEST' || command === 'PARETO') {
268 const departure = time(args[2]), all = rounds(a, departure);
269 if (command === 'FEWEST') {
270 const k = all.findIndex(labels => labels.has(b));
271 if (k < 0) output.push(`FEWEST ${a} ${b} ${args[2]} unreachable`);
272 else printJourney(`FEWEST ${a} ${b} trips ${k}`, departure, all[k].get(b)!);
273 } else {
274 const selected = all.map((labels, k) => k).filter(k => all[k].has(b) && (!k || !all[k - 1].has(b) || all[k - 1].get(b)!.time > all[k].get(b)!.time));
275 output.push(`PARETO ${a} ${b} ${args[2]} options ${selected.length}`);
276 for (const k of selected) printJourney(`OPTION trips ${k}`, departure, all[k].get(b)!);
277 }
278 } else {
279 const departure = time(args[2]), label = earliest(a, departure).get(b);
280 if (!label) output.push(`${command} ${a} ${b} ${args[2]}${command === 'FARE' ? ` ${args[3]}` : ''} unreachable`);
281 else if (command === 'FARE') printFare(a, b, args[2], args[3], label);
282 else printJourney(`ROUTE ${a} ${b}`, departure, label);
283 }
284 break;
285 }
286 case 'VIA': {
287 const via = b, destination = args[2];
288 station(a); station(via); station(destination);
289 if (a === destination) fail('origin and destination are the same');
290 if (via === a || via === destination) fail(`via station ${via} must differ from origin and destination`);
291 openStation(a); openStation(via); openStation(destination);
292 const departure = time(args[3]), first = earliest(a, departure).get(via);
293 const ready = first ? first.time + stations.get(via)!.transfer : 0;
294 const second = first ? earliest(via, ready).get(destination) : undefined;
295 if (!first || !second) output.push(`VIA ${a} ${via} ${destination} ${args[3]} unreachable`);
296 else {
297 const stay: Stay = { kind: 'stay', station: via, departure: first.time, arrival: ready };
298 printJourney(`VIA ${a} ${via} ${destination}`, departure, { time: second.time, journey: [...first.journey, stay, ...second.journey] });
299 }
300 break;
301 }
302 case 'REACH': {
303 station(a); openStation(a);
304 const departure = time(b), labels = earliest(a, departure), limit = departure + Number(args[2]);
305 const reached = [...labels.keys()].filter(id => labels.get(id)!.time <= limit).sort((x, y) => labels.get(x)!.time - labels.get(y)!.time || compare(x, y));
306 output.push(`REACH ${a} ${b} ${args[2]} count ${reached.length}`);
307 for (const id of reached) {
308 const label = labels.get(id)!;
309 output.push(`${id} ${stations.get(id)!.name} ${clock(label.time)} ${label.time - departure} rides ${label.journey.filter(leg => leg.kind === 'ride').length}`);
310 }
311 break;
312 }
313 case 'TIMETABLE': {
314 const s = station(a), item = line(b), toward = args[2];
315 if (!item.stops.includes(a)) fail(`station ${a} is not on line ${b}`);
316 if (toward !== item.stops[0] && toward !== item.stops[item.stops.length - 1]) fail(`${toward} is not a terminal of line ${b}`);
317 if (a === toward) fail(`no departures from ${a} toward ${toward}`);
318 if (item.suspended) fail(`line ${b} is suspended`);
319 if (s.closed) fail(`station ${a} is closed`);
320 const direction = directions(item).find(d => d.stops[d.stops.length - 1] === toward)!, at = direction.stops.indexOf(a);
321 const q = nextTrain(direction, at, time(args[3])), times: string[] = [];
322 if (q !== null) for (let i = q; i < trips(item) && times.length < Number(args[4]); i++) times.push(clock(trainTime(direction, i, at)));
323 output.push(`TIMETABLE ${a} ${b} toward ${toward} ${args[3]} next ${times.length}${times.length ? ` ${times.join(' ')}` : ''}`);
324 break;
325 }
326 case 'DEPARTURES': case 'ARRIVALS': {
327 station(a); openStation(a);
328 const events: Event[] = [];
329 for (const item of orderedLines()) if (!item.suspended && item.stops.includes(a)) for (const direction of directions(item)) {
330 const at = direction.stops.indexOf(a), last = direction.stops.length - 1;
331 if (command === 'DEPARTURES' ? at === last : at === 0) continue;
332 const q = nextTrain(direction, at, time(b));
333 if (q !== null) for (let i = q; i < trips(item); i++) events.push({ time: trainTime(direction, i, at), line: item.id, order: direction.order, terminal: direction.stops[last], start: direction.stops[0] });
334 }
335 events.sort((x, y) => x.time - y.time || compare(x.line, y.line) || x.order - y.order);
336 const selected = events.slice(0, Number(args[2]));
337 output.push(`${command} ${a} ${b} count ${selected.length}`);
338 for (const event of selected) output.push(`${clock(event.time)} ${event.line} ${command === 'DEPARTURES' ? `toward ${event.terminal}` : `from ${event.start}`}`);
339 break;
340 }
341 case 'TRIP': {
342 const item = line(a);
343 if (b !== item.stops[0] && b !== item.stops[item.stops.length - 1]) fail(`${b} is not a terminal of line ${a}`);
344 if (item.suspended) fail(`line ${a} is suspended`);
345 const direction = directions(item).find(d => d.stops[d.stops.length - 1] === b)!, q = nextTrain(direction, 0, time(args[2]));
346 if (q === null) output.push(`TRIP ${a} toward ${b} ${args[2]} none`);
347 else {
348 output.push(`TRIP ${a} toward ${b} ${args[2]} trip ${q + 1} departs ${clock(trainTime(direction, q, 0))}`);
349 direction.stops.forEach((id, j) => output.push(`${clock(trainTime(direction, q, j))} ${id}${stations.get(id)!.closed ? ' closed' : ''}${j && closed.has(pair(direction.stops[j - 1], id)) ? ' blocked' : ''}`));
350 }
351 break;
352 }
353 case 'SHOWLINE': {
354 const item = line(a);
355 output.push(`SHOWLINE ${a} ${item.suspended ? 'suspended' : 'running'} first ${clock(item.first)} last ${clock(item.last)} headway ${item.headway} delay ${item.delay} trips ${trips(item)}`);
356 item.stops.forEach((id, i) => output.push(`${item.offsets[i]} ${id} zone ${stations.get(id)!.zone}${stations.get(id)!.closed ? ' closed' : ''}${i && closed.has(pair(item.stops[i - 1], id)) ? ' blocked' : ''}`));
357 break;
358 }
359 case 'INFO': {
360 const s = station(a), service = orderedLines().filter(item => item.stops.includes(a)), neighbors = [...s.walks.keys()].sort();
361 output.push(`INFO ${a} ${s.name} zone ${s.zone} transfer ${s.transfer} ${s.closed ? 'closed' : 'open'}`,
362 `LINES ${service.length ? service.map(item => item.id).join(' ') : 'none'}`,
363 `WALKS ${neighbors.length ? neighbors.map(id => `${id} ${s.walks.get(id)!}`).join(' ') : 'none'}`);
364 break;
365 }
366 case 'STATUS': {
367 const walkCount = [...stations.values()].reduce((sum, s) => sum + s.walks.size, 0) / 2;
368 output.push(`STATUS stations ${stations.size} lines ${lines.size} walks ${walkCount}`);
369 for (const id of [...stations.keys()].sort()) if (stations.get(id)!.closed) output.push(`CLOSED ${id}`);
370 const segments = [...closed].sort((x, y) => { const a = x.split('-'), b = y.split('-'); return compare(a[0], b[0]) || compare(a[1], b[1]); });
371 for (const segment of segments) output.push(`CLOSED ${segment}`);
372 for (const item of orderedLines()) if (item.suspended) output.push(`SUSPENDED ${item.id}`);
373 for (const item of orderedLines()) if (item.delay) output.push(`DELAYED ${item.id} ${item.delay}`);
374 break;
375 }
376 }
377}
378const input = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(input[0]);
379for (const command of input.slice(1, count + 1)) {
380 try { execute(command); }
381 catch (error) { output.push((error as Error).message); }
382}
383if (output.length) process.stdout.write(output.join('\n') + '\n');
