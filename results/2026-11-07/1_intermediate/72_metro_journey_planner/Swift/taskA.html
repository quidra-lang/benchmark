1import Foundation
2struct Problem: Error {
3 let text: String
4}
5func fail(_ reason: String) throws -> Never {
6 throw Problem(text: reason)
7}
8func time(_ s: String) -> Int {
9 let p = s.components(separatedBy: ":")
10 return Int(p[0])! * 60 + Int(p[1])!
11}
12func clock(_ t: Int) -> String {
13 String(format: "%02d:%02d",t / 60,t % 60)
14}
15struct Segment: Hashable {
16 let a: String, b: String
17 init(_ x: String, _ y: String) {
18 a = min(x,y)
19 b = max(x,y)
20 }
21}
22struct Station {
23 let zone: Int, name: String
24 var transfer = 0, closed = false
25}
26struct Line {
27 let first: Int, last: Int, headway: Int, stops: [String], offsets: [Int]
28 var delay = 0, suspended = false
29 var total: Int {
30 offsets.last!
31 }
32 var trips: Int {
33 (last - first) / headway + 1
34 }
35 func direction(_ reverse: Bool) -> ([String],[Int]) {
36 reverse ? (Array(stops.reversed()),offsets.reversed().map {
37 total - $0
38 }) : (stops,offsets)
39 }
40 func next(_ offset: Int, _ ready: Int) -> Int? {
41 let delta = ready - first - delay - offset, q = delta <= 0 ? 0 : (delta + headway - 1) / headway
42 return q < trips ? q : nil
43 }
44 func at(_ q: Int, _ offset: Int) -> Int {
45 first + delay + q * headway + offset
46 }
47}
48enum Leg {
49 case ride(line: String, terminal: String, from: String, departure: Int, to: String, arrival: Int, stops: Int)
50 case walk(from: String, departure: Int, to: String, arrival: Int)
51 case stay(station: String, start: Int, end: Int)
52 var end: Int {
53 switch self {
54 case .ride(_,_,_,_,_,let t,_), .walk(_,_,_,let t), .stay(_,_,let t): return t
55 }
56 }
57 var isRide: Bool {
58 if case .ride = self {
59 return true
60 }
61 return false
62 }
63 var isWalk: Bool {
64 if case .walk = self {
65 return true
66 }
67 return false
68 }
69}
70struct Label {
71 let arrival: Int, journey: [Leg]
72}
73typealias Labels = [String: Label]
74struct Tariff {
75 let discount: Int, window: Int, prices: [Int]
76}
77@discardableResult func relax(_ labels: inout Labels, _ station: String, _ arrival: Int, _ journey: [Leg], _ leg: Leg) -> Bool {
78 if let old = labels[station], old.arrival <= arrival {
79 return false
80 }
81 labels[station] = Label(arrival: arrival,journey: journey + [leg])
82 return true
83}
84func printJourney(_ heading: String, _ departure: Int, _ label: Label) -> [String] {
85 var out = ["\(heading) \(clock(departure)) -> \(clock(label.arrival)) total \(label.arrival - departure) rides \(label.journey.filter(\.isRide).count) walks \(label.journey.filter(\.isWalk).count)"], previous = departure
86 for leg in label.journey {
87 switch leg {
88 case .ride(let line,let terminal,let from,let d,let to,let a,let stops): out.append("RIDE \(line) toward \(terminal) \(from) \(clock(d)) -> \(to) \(clock(a)) wait \(d - previous) stops \(stops)")
89 case .walk(let from,let d,let to,let a): out.append("WALK \(from) \(clock(d)) -> \(to) \(clock(a))")
90 case .stay(let s,let start,let end): out.append("STAY \(s) \(clock(start)) -> \(clock(end))")
91 }
92 previous = leg.end
93 }
94 return out
95}
96final class Planner {
97 var stations: [String: Station] = [:], lines: [String: Line] = [:], walks: [Segment: Int] = [:], closed = Set<Segment>(), tariff: Tariff? = nil
98 func station(_ s: String) throws -> Station {
99 guard let station = stations[s] else {
100 try fail("unknown station " + s)
101 }
102 return station
103 }
104 func line(_ s: String) throws -> Line {
105 guard let line = lines[s] else {
106 try fail("unknown line " + s)
107 }
108 return line
109 }
110 func openStation(_ s: String) throws {
111 if try station(s).closed {
112 try fail("station \(s) is closed")
113 }
114 }
115 func common(_ a: String, _ b: String) throws {
116 _ = try station(a)
117 _ = try station(b)
118 if a == b {
119 try fail("origin and destination are the same")
120 }
121 try openStation(a)
122 try openStation(b)
123 }
124 func neighbors(_ s: String) -> [(String,Int)] {
125 walks.compactMap {
126 key,t -> (String,Int)? in key.a == s ? (key.b,t) : (key.b == s ? (key.a,t) : nil)
127 }.sorted {
128 $0.0 < $1.0
129 }
130 }
131 func earliest(_ origin: String, _ departure: Int) -> Labels {
132 var labels = [origin: Label(arrival: departure,journey: [])], settled = Set<String>()
133 while let entry = labels.filter({
134 !settled.contains($0.key)
135 }).min(by: {
136 a,b in a.value.arrival == b.value.arrival ? a.key < b.key : a.value.arrival < b.value.arrival
137 }) {
138 let u = entry.key, label = entry.value, ready = label.arrival + (u == origin ? 0 : stations[u]!.transfer)
139 settled.insert(u)
140 for id in lines.keys.sorted() {
141 let line = lines[id]!
142 if line.suspended || !line.stops.contains(u) {
143 continue
144 }
145 for reverse in [false,true] {
146 let (stops,offsets) = line.direction(reverse), i = stops.firstIndex(of: u)!
147 if i + 1 == stops.count {
148 continue
149 }
150 guard let q = line.next(offsets[i],ready) else {
151 continue
152 }
153 let d = line.at(q,offsets[i])
154 for j in (i + 1)..<stops.count {
155 if closed.contains(Segment(stops[j - 1],stops[j])) {
156 break
157 }
158 if stations[stops[j]]!.closed {
159 continue
160 }
161 let t = line.at(q,offsets[j])
162 relax(&labels,stops[j],t,label.journey,.ride(line: id,terminal: stops.last!,from: u,departure: d,to: stops[j],arrival: t,stops: j - i))
163 }
164 }
165 }
166 for (v,w) in neighbors(u) {
167 let t = label.arrival + w
168 if !stations[v]!.closed && t <= 1440 {
169 relax(&labels,v,t,label.journey,.walk(from: u,departure: label.arrival,to: v,arrival: t))
170 }
171 }
172 }
173 return labels
174 }
175 func walkStep(_ labels: inout Labels, _ sources: Set<String>, _ changed: inout Set<String>) {
176 let recorded = sources.sorted().map {
177 ($0,labels[$0]!)
178 }
179 for (u,label) in recorded {
180 for (v,w) in neighbors(u) {
181 let t = label.arrival + w
182 if !stations[v]!.closed && t <= 1440 && relax(&labels,v,t,label.journey,.walk(from: u,departure: label.arrival,to: v,arrival: t)) {
183 changed.insert(v)
184 }
185 }
186 }
187 }
188 func rounds(_ origin: String, _ departure: Int) -> [Labels] {
189 struct Boarded {
190 let q: Int, index: Int, departure: Int, journey: [Leg]
191 }
192 var labels = [origin: Label(arrival: departure,journey: [])], changed: Set<String> = [origin]
193 walkStep(&labels,[origin],&changed)
194 var result = [labels]
195 while !changed.isEmpty {
196 let previous = labels, marked = changed
197 changed = []
198 var sources = Set<String>()
199 for id in lines.keys.sorted() {
200 let line = lines[id]!
201 if line.suspended {
202 continue
203 }
204 for reverse in [false,true] {
205 let (stops,offsets) = line.direction(reverse)
206 var boarded: Boarded? = nil
207 for j in stops.indices {
208 if j > 0 && closed.contains(Segment(stops[j - 1],stops[j])) {
209 boarded = nil
210 }
211 if stations[stops[j]]!.closed {
212 continue
213 }
214 if let b = boarded {
215 let t = line.at(b.q,offsets[j])
216 if relax(&labels,stops[j],t,b.journey,.ride(line: id,terminal: stops.last!,from: stops[b.index],departure: b.departure,to: stops[j],arrival: t,stops: j - b.index)) {
217 sources.insert(stops[j])
218 changed.insert(stops[j])
219 }
220 }
221 if j + 1 < stops.count && marked.contains(stops[j]) {
222 let old = previous[stops[j]]!, ready = old.arrival + (stops[j] == origin ? 0 : stations[stops[j]]!.transfer)
223 if let q = line.next(offsets[j],ready), boarded == nil || q < boarded!.q {
224 boarded = Boarded(q: q,index: j,departure: line.at(q,offsets[j]),journey: old.journey)
225 }
226 }
227 }
228 }
229 }
230 walkStep(&labels,sources,&changed)
231 result.append(labels)
232 }
233 return result
234 }
235 func fare(_ journey: [Leg], _ child: Bool) -> (Int,[String]) {
236 let tariff = tariff!
237 func half(_ n: Int) -> Int {
238 (n + 19) / 20 * 10
239 }
240 let cap = child ? half(tariff.discount) : tariff.discount
241 var previous: Int? = nil, total = 0, tickets: [String] = []
242 for leg in journey {
243 if case .ride(let id,_,let from,let departure,let to,let arrival,_) = leg {
244 let stops = lines[id]!.stops, a = stops.firstIndex(of: from)!, b = stops.firstIndex(of: to)!, zones = Set(stops[min(a,b)...max(a,b)].map {
245 stations[$0]!.zone
246 }).count
247 var price = tariff.prices[min(zones,tariff.prices.count) - 1]
248 if child {
249 price = half(price)
250 }
251 let reduction = previous.map {
252 departure - $0 <= tariff.window
253 } == true ? min(cap,price) : 0, pay = price - reduction
254 total += pay
255 previous = arrival
256 tickets.append("TICKET \(id) \(from) \(to) zones \(zones) price \(price) discount \(reduction) pay \(pay)")
257 }
258 }
259 return (total,tickets)
260 }
261 func command(_ w: [String]) throws -> [String] {
262 let c = w[0], a = Array(w.dropFirst())
263 func num(_ i: Int) -> Int {
264 Int(a[i])!
265 }
266 var out: [String] = []
267 switch c {
268 case "STATION":
269 if stations[a[0]] != nil {
270 try fail("station \(a[0]) already exists")
271 }
272 stations[a[0]] = Station(zone: num(1),name: a[2])
273 out.append("STATION \(a[0]) zone \(num(1)) \(a[2])")
274 case "LINE":
275 if lines[a[0]] != nil {
276 try fail("line \(a[0]) already exists")
277 }
278 let stops = stride(from: 4,to: a.count,by: 2).map {
279 a[$0]
280 }
281 for s in stops {
282 _ = try station(s)
283 }
284 var seen = Set<String>()
285 for s in stops {
286 if !seen.insert(s).inserted {
287 try fail("station \(s) appears twice on line \(a[0])")
288 }
289 }
290 let first = time(a[1]), last = time(a[2])
291 var offsets = [0]
292 for i in stride(from: 5,to: a.count,by: 2) {
293 offsets.append(offsets.last! + num(i))
294 }
295 if first > last {
296 try fail("line \(a[0]) first departure is after last departure")
297 }
298 if last + offsets.last! > 1440 {
299 try fail("line \(a[0]) runs past 24:00")
300 }
301 let line = Line(first: first,last: last,headway: num(3),stops: stops,offsets: offsets)
302 lines[a[0]] = line
303 out.append("LINE \(a[0]) stops \(stops.count) time \(line.total) trips \(line.trips)")
304 case "TRANSFER": _ = try station(a[0])
305 stations[a[0]]!.transfer = num(1)
306 out.append("TRANSFER \(a[0]) \(num(1))")
307 case "WALK":
308 _ = try station(a[0])
309 _ = try station(a[1])
310 if a[0] == a[1] {
311 try fail("walk from \(a[0]) to itself")
312 }
313 let old = walks.updateValue(num(2),forKey: Segment(a[0],a[1]))
314 out.append("WALK \(a[0]) \(a[1]) \(num(2))" + (old.map {
315 " was \($0)"
316 } ?? ""))
317 case "TARIFF":
318 let prices = (2..<a.count).map(num)
319 if zip(prices,prices.dropFirst()).contains(where: {
320 $0.0 > $0.1
321 }) {
322 try fail("tariff prices must not decrease")
323 }
324 tariff = Tariff(discount: num(0),window: num(1),prices: prices)
325 out.append("TARIFF zones \(prices.count) discount \(num(0)) window \(num(1))")
326 case "CLOSE","OPEN":
327 _ = try station(a[0])
328 let close = c == "CLOSE"
329 if a.count == 1 {
330 if stations[a[0]]!.closed == close {
331 try fail("station \(a[0]) is \(close ? "already" : "not") closed")
332 }
333 stations[a[0]]!.closed = close
334 out.append(c + " " + a[0])
335 }
336 else {
337 _ = try station(a[1])
338 let key = Segment(a[0],a[1]), ids = lines.keys.sorted().filter {
339 id in let stops = lines[id]!.stops
340 return zip(stops,stops.dropFirst()).contains {
341 Segment($0.0,$0.1) == key
342 }
343 }
344 if close && ids.isEmpty {
345 try fail("no segment \(a[0])-\(a[1])")
346 }
347 if closed.contains(key) == close {
348 try fail("segment \(a[0])-\(a[1]) is \(close ? "already" : "not") closed")
349 }
350 if close {
351 closed.insert(key)
352 } else {
353 closed.remove(key)
354 }
355 out.append("\(c) \(a[0])-\(a[1])" + (close ? " lines " + ids.joined(separator: " ") : ""))
356 }
357 case "DELAY":
358 let line = try line(a[0])
359 if line.last + num(1) + line.total > 1440 {
360 try fail("line \(a[0]) runs past 24:00")
361 }
362 lines[a[0]]!.delay = num(1)
363 out.append("DELAY \(a[0]) \(num(1))")
364 case "SUSPEND","RESUME":
365 let line = try line(a[0]), suspended = c == "SUSPEND"
366 if line.suspended == suspended {
367 try fail("line \(a[0]) is \(suspended ? "already" : "not") suspended")
368 }
369 lines[a[0]]!.suspended = suspended
370 out.append(c + " " + a[0])
371 case "ROUTE","FEWEST","PARETO","FARE","LASTTRAIN","ARRIVEBY","PROFILE":
372 try common(a[0],a[1])
373 if c == "FARE" {
374 if !["ADULT","CHILD"].contains(a[3]) {
375 try fail("unknown fare class " + a[3])
376 }
377 if tariff == nil {
378 try fail("no tariff")
379 }
380 }
381 if c == "PROFILE" && time(a[2]) > time(a[3]) {
382 try fail("empty interval \(a[2]) \(a[3])")
383 }
384 let heading = "\(c) \(a[0]) \(a[1])"
385 if c == "LASTTRAIN" || c == "ARRIVEBY" {
386 let bound = c == "LASTTRAIN" ? 1440 : time(a[2]), title = c == "ARRIVEBY" ? heading + " by " + a[2] : heading
387 if bound > 0 {
388 for t in stride(from: bound - 1,through: 0,by: -1) {
389 if let label = earliest(a[0],t)[a[1]], c == "LASTTRAIN" || label.arrival <= bound {
390 return printJourney(title,t,label)
391 }
392 }
393 }
394 out.append(title + " none")
395 } else if c == "PROFILE" {
396 let start = time(a[2]), end = time(a[3]), arrivals = (start...end).map {
397 earliest(a[0],$0)[a[1]]?.arrival
398 }
399 var selected: [String] = []
400 for i in arrivals.indices {
401 if let arrival = arrivals[i], i + 1 == arrivals.count || arrivals[i + 1] == nil || arrival < arrivals[i + 1]! {
402 selected.append("\(clock(start + i)) -> \(clock(arrival)) total \(arrival - start - i)")
403 }
404 }
405 out.append("\(heading) \(a[2]) \(a[3]) count \(selected.count)")
406 out += selected
407 } else if c == "FEWEST" || c == "PARETO" {
408 let start = time(a[2]), rounds = rounds(a[0],start)
409 if c == "FEWEST" {
410 for (k,labels) in rounds.enumerated() {
411 if let label = labels[a[1]] {
412 return printJourney("\(heading) trips \(k)",start,label)
413 }
414 }
415 out.append("\(heading) \(a[2]) unreachable")
416 }
417 else {
418 var options: [String] = [], previous: Int? = nil, count = 0
419 for (k,labels) in rounds.enumerated() {
420 if let label = labels[a[1]] {
421 if previous == nil || previous! > label.arrival {
422 count += 1
423 options += printJourney("OPTION trips \(k)",start,label)
424 }
425 previous = label.arrival
426 }
427 }
428 out.append("\(heading) \(a[2]) options \(count)")
429 out += options
430 }
431 } else {
432 let start = time(a[2])
433 if let label = earliest(a[0],start)[a[1]] {
434 if c == "FARE" {
435 let (total,tickets) = fare(label.journey,a[3] == "CHILD")
436 out.append("\(heading) \(a[2]) \(a[3]) rides \(label.journey.filter(\.isRide).count) total \(total)")
437 out += tickets
438 }
439 else {
440 return printJourney(heading,start,label)
441 }
442 } else {
443 out.append("\(heading) \(a[2])" + (c == "FARE" ? " " + a[3] : "") + " unreachable")
444 }
445 }
446 case "VIA":
447 _ = try station(a[0])
448 _ = try station(a[1])
449 _ = try station(a[2])
450 if a[0] == a[2] {
451 try fail("origin and destination are the same")
452 }
453 if a[1] == a[0] || a[1] == a[2] {
454 try fail("via station \(a[1]) must differ from origin and destination")
455 }
456 try openStation(a[0])
457 try openStation(a[1])
458 try openStation(a[2])
459 let start = time(a[3]), heading = "VIA \(a[0]) \(a[1]) \(a[2])"
460 if let first = earliest(a[0],start)[a[1]] {
461 let next = first.arrival + stations[a[1]]!.transfer
462 if let second = earliest(a[1],next)[a[2]] {
463 return printJourney(heading,start,Label(arrival: second.arrival,journey: first.journey + [.stay(station: a[1],start: first.arrival,end: next)] + second.journey))
464 }
465 }
466 out.append("\(heading) \(a[3]) unreachable")
467 case "REACH":
468 try openStation(a[0])
469 let start = time(a[1]), selected = earliest(a[0],start).filter {
470 $0.value.arrival <= start + num(2)
471 }.sorted {
472 a,b in a.value.arrival == b.value.arrival ? a.key < b.key : a.value.arrival < b.value.arrival
473 }
474 out.append("REACH \(a[0]) \(a[1]) \(num(2)) count \(selected.count)")
475 for (s,label) in selected {
476 out.append("\(s) \(stations[s]!.name) \(clock(label.arrival)) \(label.arrival - start) rides \(label.journey.filter(\.isRide).count)")
477 }
478 case "TIMETABLE":
479 _ = try station(a[0])
480 let line = try line(a[1])
481 if !line.stops.contains(a[0]) {
482 try fail("station \(a[0]) is not on line \(a[1])")
483 }
484 if a[2] != line.stops.first! && a[2] != line.stops.last! {
485 try fail("\(a[2]) is not a terminal of line \(a[1])")
486 }
487 if a[0] == a[2] {
488 try fail("no departures from \(a[0]) toward \(a[2])")
489 }
490 if line.suspended {
491 try fail("line \(a[1]) is suspended")
492 }
493 try openStation(a[0])
494 let (stops,offsets) = line.direction(a[2] == line.stops.first!), i = stops.firstIndex(of: a[0])!, selected = (0..<line.trips).map {
495 line.at($0,offsets[i])
496 }.filter {
497 $0 >= time(a[3])
498 }.prefix(num(4))
499 out.append("TIMETABLE \(a[0]) \(a[1]) toward \(a[2]) \(a[3]) next \(selected.count)" + selected.map {
500 " " + clock($0)
501 }.joined())
502 case "DEPARTURES","ARRIVALS":
503 struct Event {
504 let time: Int, line: String, direction: Int, endpoint: String
505 }
506 try openStation(a[0])
507 var events: [Event] = []
508 for id in lines.keys.sorted() {
509 let line = lines[id]!
510 if line.suspended || !line.stops.contains(a[0]) {
511 continue
512 }
513 for direction in 0...1 {
514 let (stops,offsets) = line.direction(direction == 1), i = stops.firstIndex(of: a[0])!
515 if (c == "DEPARTURES" && i + 1 == stops.count) || (c == "ARRIVALS" && i == 0) {
516 continue
517 }
518 for q in 0..<line.trips {
519 let t = line.at(q,offsets[i])
520 if t >= time(a[1]) {
521 events.append(Event(time: t,line: id,direction: direction,endpoint: c == "DEPARTURES" ? stops.last! : stops.first!))
522 }
523 }
524 }
525 }
526 let selected = events.sorted {
527 a,b in a.time != b.time ? a.time < b.time : (a.line != b.line ? a.line < b.line : a.direction < b.direction)
528 }.prefix(num(2))
529 out.append("\(c) \(a[0]) \(a[1]) count \(selected.count)")
530 for e in selected {
531 out.append("\(clock(e.time)) \(e.line) \(c == "DEPARTURES" ? "toward" : "from") \(e.endpoint)")
532 }
533 case "TRIP":
534 let line = try line(a[0])
535 if a[1] != line.stops.first! && a[1] != line.stops.last! {
536 try fail("\(a[1]) is not a terminal of line \(a[0])")
537 }
538 if line.suspended {
539 try fail("line \(a[0]) is suspended")
540 }
541 let (stops,offsets) = line.direction(a[1] == line.stops.first!), heading = "TRIP \(a[0]) toward \(a[1]) \(a[2])"
542 if let q = line.next(0,time(a[2])) {
543 out.append("\(heading) trip \(q + 1) departs \(clock(line.at(q,0)))")
544 for j in stops.indices {
545 out.append("\(clock(line.at(q,offsets[j]))) \(stops[j])" + (stations[stops[j]]!.closed ? " closed" : "") + (j > 0 && closed.contains(Segment(stops[j - 1],stops[j])) ? " blocked" : ""))
546 }
547 }
548 else {
549 out.append(heading + " none")
550 }
551 case "SHOWLINE":
552 let line = try line(a[0])
553 out.append("SHOWLINE \(a[0]) \(line.suspended ? "suspended" : "running") first \(clock(line.first)) last \(clock(line.last)) headway \(line.headway) delay \(line.delay) trips \(line.trips)")
554 for j in line.stops.indices {
555 let s = line.stops[j]
556 out.append("\(line.offsets[j]) \(s) zone \(stations[s]!.zone)" + (stations[s]!.closed ? " closed" : "") + (j > 0 && closed.contains(Segment(line.stops[j - 1],s)) ? " blocked" : ""))
557 }
558 case "INFO":
559 let s = try station(a[0]), ids = lines.keys.sorted().filter {
560 lines[$0]!.stops.contains(a[0])
561 }, links = neighbors(a[0])
562 out = ["INFO \(a[0]) \(s.name) zone \(s.zone) transfer \(s.transfer) \(s.closed ? "closed" : "open")","LINES " + (ids.isEmpty ? "none" : ids.joined(separator: " ")),"WALKS " + (links.isEmpty ? "none" : links.map {
563 "\($0.0) \($0.1)"
564 }.joined(separator: " "))]
565 case "STATUS":
566 out.append("STATUS stations \(stations.count) lines \(lines.count) walks \(walks.count)")
567 for s in stations.keys.sorted() where stations[s]!.closed {
568 out.append("CLOSED " + s)
569 }
570 for segment in closed.sorted(by: {
571 $0.a == $1.a ? $0.b < $1.b : $0.a < $1.a
572 }) {
573 out.append("CLOSED \(segment.a)-\(segment.b)")
574 }
575 for id in lines.keys.sorted() where lines[id]!.suspended {
576 out.append("SUSPENDED " + id)
577 }
578 for id in lines.keys.sorted() where lines[id]!.delay != 0 {
579 out.append("DELAYED \(id) \(lines[id]!.delay)")
580 }
581 default: fatalError("unknown command")
582 }
583 return out
584 }
