67class Direction:
68 def __init__(self, line, stops, offsets):
69 self.line = line
70 self.stops = stops
71 self.offsets = offsets
72 self.start = stops[0]
73 self.terminal = stops[-1]
74 self.index = {station_id: j for j, station_id in enumerate(stops)}
75
76 def time(self, train, j):
77 line = self.line
78 return line.first + line.delay + train * line.headway + self.offsets[j]
79
80 def first_train(self, j, ready):
81 train = max(0, -((self.time(0, j) - ready) // self.line.headway))
82 return train if train < self.line.trips else None
83
84 def upcoming(self, j, time, count):
85 first = self.first_train(j, time)
86 if first is None:
87 return range(0)
88 return range(first, min(first + count, self.line.trips))
89
90 def ride(self, train, board, alight):
91 return Ride(
92 self.line.id,
93 self.terminal,
94 self.stops[board],
95 self.time(train, board),
96 self.stops[alight],
97 self.time(train, alight),
98 alight - board,
99 )

102class Line:
103 def __init__(self, line_id, first, last, headway, stops, offsets):
104 self.id = line_id
105 self.first = first
106 self.last = last
107 self.headway = headway
108 self.stops = stops
109 self.offsets = offsets
110 self.total = offsets[-1]
111 self.trips = (last - first) // headway + 1
112 self.delay = 0
113 self.suspended = False
114 self.segments = {segment(a, b) for a, b in zip(stops, stops[1:])}
115 self.directions = (
116 Direction(self, stops, offsets),
117 Direction(self, stops[::-1], [self.total - offset for offset in reversed(offsets)]),
118 )
119
120 def toward(self, terminal):
121 for direction in self.directions:
122 if direction.terminal == terminal:
123 return direction
124 return None

218 def ready_time(self, origin, station_id, arrival):
219 if station_id == origin:
220 return arrival

223 def walk_from(self, labels, station_id, arrival, journey):
224 reached = []
225 for neighbor, minutes in sorted(self.stations[station_id].walks.items()):
226 time = arrival + minutes
227 if (
228 not self.stations[neighbor].closed
229 and time <= END_OF_SERVICE
230 and improves(labels, neighbor, time)
231 ):
232 leg = Walk(station_id, arrival, neighbor, time)
233 labels[neighbor] = (time, journey + (leg,))
234 reached.append(neighbor)
235 return reached

237 def ride_from(self, labels, direction, station_id, ready, journey):
238 i = direction.index.get(station_id)
239 if i is None or i == len(direction.stops) - 1:
240 return
241 train = direction.first_train(i, ready)
242 if train is None:
243 return
244 for j in range(i + 1, len(direction.stops)):
245 stop = direction.stops[j]
246 if self.blocked(direction.stops[j - 1], stop):
247 return
248 if self.stations[stop].closed:
249 continue
250 time = direction.time(train, j)
251 if improves(labels, stop, time):
252 labels[stop] = (time, journey + (direction.ride(train, i, j),))

254 def earliest_arrival(self, origin, departure):
255 lines = self.active_lines()
256 labels = {origin: (departure, ())}
257 settled = set()
258 while True:
259 pending = [(labels[key][0], key) for key in labels if key not in settled]
260 if not pending:
261 return labels
262 arrival, current = min(pending)
263 settled.add(current)
264 journey = labels[current][1]
265 ready = self.ready_time(origin, current, arrival)
266 for line in lines:
267 for direction in line.directions:
268 self.ride_from(labels, direction, current, ready, journey)
269 self.walk_from(labels, current, arrival, journey)

271 def walk_step(self, labels, starts, marked):
272 recorded = {station_id: labels[station_id] for station_id in starts}
273 for station_id in sorted(starts):
274 arrival, journey = recorded[station_id]
275 marked.update(self.walk_from(labels, station_id, arrival, journey))

277 def ride_round(self, origin, direction, previous, previous_marked, labels, marked):
278 boarded = None
279 last = len(direction.stops) - 1
280 for j, stop in enumerate(direction.stops):
281 if j >= 1 and self.blocked(direction.stops[j - 1], stop):
282 boarded = None
283 if self.stations[stop].closed:
284 continue
285 if boarded is not None:
286 train, board, journey = boarded
287 time = direction.time(train, j)
288 if improves(labels, stop, time):
289 labels[stop] = (time, journey + (direction.ride(train, board, j),))
290 marked.add(stop)
291 if j < last and stop in previous_marked:
292 arrival, journey = previous[stop]
293 train = direction.first_train(j, self.ready_time(origin, stop, arrival))
294 if train is not None and (boarded is None or train < boarded[0]):
295 boarded = (train, j, journey)

297 def round_search(self, origin, departure):
298 labels = {origin: (departure, ())}
299 marked = {origin}
300 self.walk_step(labels, {origin}, marked)
301 rounds = [labels]
302 lines = self.active_lines()
303 while marked:
304 previous, previous_marked = labels, marked
305 labels = dict(previous)
306 ridden = set()
307 for line in lines:
308 for direction in line.directions:
309 self.ride_round(origin, direction, previous, previous_marked, labels, ridden)
310 marked = set(ridden)
311 self.walk_step(labels, ridden, marked)
312 rounds.append(labels)
313 return rounds

315 def emit_journey(self, heading, departure, arrival, journey):
316 self.emit(
317 f"{heading} {clock(departure)} -> {clock(arrival)} total {arrival - departure} "
318 f"rides {ride_count(journey)} walks {walk_count(journey)}"
319 )
320 previous_end = departure
321 for leg in journey:
322 self.emit(leg_text(leg, previous_end))
323 previous_end = leg.end

325 def tickets(self, journey, child):
326 cap, window, prices = self.tariff
327 if child:
328 cap = half(cap)
329 tickets = []
330 previous = None
331 for leg in journey:
332 if not isinstance(leg, Ride):
333 continue
334 direction = self.lines[leg.line].toward(leg.terminal)
335 start = direction.index[leg.board]
336 covered = direction.stops[start:start + leg.stops + 1]
337 zones = len({self.stations[stop].zone for stop in covered})
338 price = prices[min(zones, len(prices)) - 1]
339 if child:
340 price = half(price)
341 discount = 0
342 if previous is not None and leg.start - previous.end <= window:
343 discount = min(cap, price)
344 tickets.append((leg, zones, price, discount))
345 previous = leg
346 return tickets

348 def add_station(self, arguments):
349 station_id, zone, name = arguments
350 if station_id in self.stations:
351 raise CommandError(f"station {station_id} already exists")
352 self.stations[station_id] = Station(station_id, int(zone), name)
353 self.emit(f"STATION {station_id} zone {zone} {name}")

355 def add_line(self, arguments):
356 line_id, first_text, last_text, headway, *route = arguments
357 if line_id in self.lines:
358 raise CommandError(f"line {line_id} already exists")
359 stops = route[::2]
360 for stop in stops:
361 self.station(stop)
362 seen = set()
363 for stop in stops:
364 if stop in seen:
365 raise CommandError(f"station {stop} appears twice on line {line_id}")
366 seen.add(stop)
367 first = parse_time(first_text)
368 last = parse_time(last_text)
369 if first > last:
370 raise CommandError(f"line {line_id} first departure is after last departure")
371 offsets = [0, *accumulate(int(minutes) for minutes in route[1::2])]
372 if last + offsets[-1] > END_OF_SERVICE:
373 raise CommandError(f"line {line_id} runs past 24:00")
374 line = Line(line_id, first, last, int(headway), stops, offsets)
375 self.lines[line_id] = line
376 self.emit(f"LINE {line_id} stops {len(stops)} time {line.total} trips {line.trips}")

424 def close_segment(self, first, second):

426 self.station(second)

479 def route(self, arguments):
480 origin, destination, time_text = arguments
481 self.check_endpoints(origin, destination)
482 departure = parse_time(time_text)
483 labels = self.earliest_arrival(origin, departure)
484 if destination not in labels:
485 self.emit(f"ROUTE {origin} {destination} {time_text} unreachable")
486 return
487 arrival, journey = labels[destination]
488 self.emit_journey(f"ROUTE {origin} {destination}", departure, arrival, journey)

490 def via(self, arguments):
491 origin, via, destination, time_text = arguments
492 self.station(origin)
493 self.station(via)
494 self.station(destination)
495 if origin == destination:
496 raise CommandError("origin and destination are the same")
497 if via in (origin, destination):
498 raise CommandError(f"via station {via} must differ from origin and destination")
499 for station_id in (origin, via, destination):
500 self.require_open(station_id)
501 departure = parse_time(time_text)
502 heading = f"VIA {origin} {via} {destination}"
503 first_labels = self.earliest_arrival(origin, departure)
504 if via in first_labels:
505 reached, first_journey = first_labels[via]
506 resumed = reached + self.stations[via].transfer
507 second_labels = self.earliest_arrival(via, resumed)
508 if destination in second_labels:
509 arrival, second_journey = second_labels[destination]
510 journey = first_journey + (Stay(via, reached, resumed),) + second_journey
511 self.emit_journey(heading, departure, arrival, journey)
512 return
513 self.emit(f"{heading} {time_text} unreachable")
514
515 def fewest(self, arguments):
516 origin, destination, time_text = arguments
517 self.check_endpoints(origin, destination)
518 departure = parse_time(time_text)
519 for rides, labels in enumerate(self.round_search(origin, departure)):
520 if destination in labels:
521 arrival, journey = labels[destination]
522 heading = f"FEWEST {origin} {destination} trips {rides}"
523 self.emit_journey(heading, departure, arrival, journey)
524 return
525 self.emit(f"FEWEST {origin} {destination} {time_text} unreachable")

527 def pareto(self, arguments):
528 origin, destination, time_text = arguments
529 self.check_endpoints(origin, destination)
530 departure = parse_time(time_text)
531 options = []
532 previous = {}
533 for rides, labels in enumerate(self.round_search(origin, departure)):
534 if destination in labels and improves(previous, destination, labels[destination][0]):
535 options.append((rides, *labels[destination]))
536 previous = labels
537 self.emit(f"PARETO {origin} {destination} {time_text} options {len(options)}")
538 for rides, arrival, journey in options:
539 self.emit_journey(f"OPTION trips {rides}", departure, arrival, journey)

541 def fare(self, arguments):
542 origin, destination, time_text, fare_class = arguments
543 self.check_endpoints(origin, destination)
544 if fare_class not in FARE_CLASSES:
545 raise CommandError(f"unknown fare class {fare_class}")
546 if self.tariff is None:
547 raise CommandError("no tariff")
548 labels = self.earliest_arrival(origin, parse_time(time_text))
549 heading = f"FARE {origin} {destination} {time_text} {fare_class}"
550 if destination not in labels:
551 self.emit(f"{heading} unreachable")
552 return
553 tickets = self.tickets(labels[destination][1], fare_class == "CHILD")
554 total = sum(price - discount for _, _, price, discount in tickets)
555 self.emit(f"{heading} rides {len(tickets)} total {total}")
556 for leg, zones, price, discount in tickets:
557 self.emit(
558 f"TICKET {leg.line} {leg.board} {leg.alight} zones {zones} "
559 f"price {price} discount {discount} pay {price - discount}"
560 )

574 def arrive_by(self, arguments):
575 origin, destination, time_text = arguments
576 self.check_endpoints(origin, destination)
577 deadline = parse_time(time_text)
578 heading = f"ARRIVEBY {origin} {destination} by {time_text}"

587 def profile(self, arguments):
588 origin, destination, start_text, end_text = arguments
589 self.check_endpoints(origin, destination)
590 start = parse_time(start_text)
591 end = parse_time(end_text)
592 if start > end:
593 raise CommandError(f"empty interval {start_text} {end_text}")
594 arrivals = {}
595 for departure in range(start, end + 1):
596 labels = self.earliest_arrival(origin, departure)
597 if destination in labels:
598 arrivals[departure] = labels[destination][0]
599 selected = [
600 departure
601 for departure, arrival in arrivals.items()
602 if departure == end
603 or departure + 1 not in arrivals
604 or arrival < arrivals[departure + 1]
605 ]
606 self.emit(
607 f"PROFILE {origin} {destination} {start_text} {end_text} count {len(selected)}"
608 )
609 for departure in selected:
610 arrival = arrivals[departure]
611 self.emit(f"{clock(departure)} -> {clock(arrival)} total {arrival - departure}")

613 def reach(self, arguments):
614 station_id, time_text, minutes = arguments
615 self.station(station_id)
616 self.require_open(station_id)
617 departure = parse_time(time_text)
618 limit = departure + int(minutes)
619 labels = self.earliest_arrival(station_id, departure)
620 reached = sorted(
621 (arrival, reached_id)
622 for reached_id, (arrival, _) in labels.items()
623 if arrival <= limit
624 )
625 self.emit(f"REACH {station_id} {time_text} {minutes} count {len(reached)}")
626 for arrival, reached_id in reached:
627 name = self.stations[reached_id].name
628 rides = ride_count(labels[reached_id][1])
629 self.emit(
630 f"{reached_id} {name} {clock(arrival)} {arrival - departure} rides {rides}"
631 )

633 def timetable(self, arguments):
634 station_id, line_id, toward, time_text, count = arguments
635 self.station(station_id)
636 line = self.line(line_id)
637 if station_id not in line.stops:
638 raise CommandError(f"station {station_id} is not on line {line_id}")
639 direction = self.terminal_direction(line, toward)
640 if station_id == toward:
641 raise CommandError(f"no departures from {station_id} toward {toward}")
642 if line.suspended:
643 raise CommandError(f"line {line_id} is suspended")
644 self.require_open(station_id)
645 j = direction.index[station_id]
646 trains = direction.upcoming(j, parse_time(time_text), int(count))
647 times = [clock(direction.time(train, j)) for train in trains]
648 heading = f"TIMETABLE {station_id} {line_id} toward {toward} {time_text} next {len(times)}"
649 self.emit(" ".join([heading, *times]))

651 def passing_trains(self, station_id, time, count, arriving):
652 trains = []
653 for line in self.active_lines():
654 if station_id not in line.stops:
655 continue
656 for direction in line.directions:
657 j = direction.index[station_id]
658 if j == (0 if arriving else len(direction.stops) - 1):
659 continue
660 for train in direction.upcoming(j, time, count):
661 trains.append((direction.time(train, j), direction))
662 trains.sort(key=lambda entry: entry[0])
663 return trains[:count]

665 def departures(self, arguments):
666 station_id, time_text, count = arguments
667 self.station(station_id)
668 self.require_open(station_id)
669 trains = self.passing_trains(station_id, parse_time(time_text), int(count), False)
670 self.emit(f"DEPARTURES {station_id} {time_text} count {len(trains)}")
671 for time, direction in trains:
672 self.emit(f"{clock(time)} {direction.line.id} toward {direction.terminal}")
673
674 def arrivals(self, arguments):
675 station_id, time_text, count = arguments
676 self.station(station_id)
677 self.require_open(station_id)
678 trains = self.passing_trains(station_id, parse_time(time_text), int(count), True)
679 self.emit(f"ARRIVALS {station_id} {time_text} count {len(trains)}")
680 for time, direction in trains:
681 self.emit(f"{clock(time)} {direction.line.id} from {direction.start}")
682
683 def trip(self, arguments):
684 line_id, toward, time_text = arguments
685 line = self.line(line_id)
686 direction = self.terminal_direction(line, toward)
687 if line.suspended:
688 raise CommandError(f"line {line_id} is suspended")
689 train = direction.first_train(0, parse_time(time_text))
690 heading = f"TRIP {line_id} toward {toward} {time_text}"
691 if train is None:
692 self.emit(f"{heading} none")
693 return
694 self.emit(f"{heading} trip {train + 1} departs {clock(direction.time(train, 0))}")
695 for j, stop in enumerate(direction.stops):
696 flags = self.stop_flags(direction.stops, j)
697 self.emit(f"{clock(direction.time(train, j))} {stop}{flags}")
698
699 def show_line(self, arguments):
700 line = self.line(arguments[0])
701 state = "suspended" if line.suspended else "running"
702 self.emit(
703 f"SHOWLINE {line.id} {state} first {clock(line.first)} last {clock(line.last)} "
704 f"headway {line.headway} delay {line.delay} trips {line.trips}"
705 )
706 for i, stop in enumerate(line.stops):
707 zone = self.stations[stop].zone
708 flags = self.stop_flags(line.stops, i)
709 self.emit(f"{line.offsets[i]} {stop} zone {zone}{flags}")
710
711 def info(self, arguments):
712 station = self.station(arguments[0])
713 state = "closed" if station.closed else "open"
714 self.emit(
715 f"INFO {station.id} {station.name} zone {station.zone} "
716 f"transfer {station.transfer} {state}"
717 )
718 line_ids = [line.id for line in self.sorted_lines() if station.id in line.stops]
719 self.emit("LINES " + (" ".join(line_ids) or "none"))
720 walks = [f"{neighbor} {minutes}" for neighbor, minutes in sorted(station.walks.items())]
721 self.emit("WALKS " + (" ".join(walks) or "none"))
722
723 def status(self, arguments):
724 walks = sum(len(station.walks) for station in self.stations.values()) // 2
725 self.emit(f"STATUS stations {len(self.stations)} lines {len(self.lines)} walks {walks}")
726 for station_id in sorted(self.stations):
727 if self.stations[station_id].closed:
728 self.emit(f"CLOSED {station_id}")
729 for first, second in sorted(self.closed_segments):
730 self.emit(f"CLOSED {first}-{second}")
731 lines = self.sorted_lines()
732 for line in lines:
733 if line.suspended:
734 self.emit(f"SUSPENDED {line.id}")
735 for line in lines:
736 if line.delay:
737 self.emit(f"DELAYED {line.id} {line.delay}")
