3const stations = new Map();
4const lines = new Map();
5const closed = new Set();
6const output = [];
7let tariff = null;
8function fail(message) { throw new Error(`ERROR ${message}`); }
9function compare(a, b) { return a < b ? -1 : a > b ? 1 : 0; }
10function time(word) { const [h, m] = word.split(':').map(Number); return h * 60 + m; }
11function clock(value) { return `${Math.floor(value / 60).toString().padStart(2, '0')}:${(value % 60).toString().padStart(2, '0')}`; }
12function pair(a, b) { return a < b ? `${a}-${b}` : `${b}-${a}`; }
13function station(id) {
14 const result = stations.get(id);
15 if (!result) fail(`unknown station ${id}`);
16 return result;
17}
18function line(id) {
19 const result = lines.get(id);
20 if (!result) fail(`unknown line ${id}`);
21 return result;
22}
23function openStation(id) { if (stations.get(id).closed) fail(`station ${id} is closed`); }
24function common(origin, destination) {
25 station(origin); station(destination);
26 if (origin === destination) fail('origin and destination are the same');
27 openStation(origin); openStation(destination);
28}
29function orderedLines() { return [...lines.values()].sort((a, b) => compare(a.id, b.id)); }
30function directions(item) {
31 const total = item.offsets[item.offsets.length - 1];
32 return [{ line: item, order: 0, stops: item.stops, offsets: item.offsets },
33 { line: item, order: 1, stops: [...item.stops].reverse(), offsets: [...item.offsets].reverse().map(offset => total - offset) }];
34}
35function trips(item) { return Math.floor((item.last - item.first) / item.headway) + 1; }
36function nextTrain(direction, at, ready) {
37 const item = direction.line;
38 const q = Math.max(0, Math.ceil((ready - item.first - item.delay - direction.offsets[at]) / item.headway));
39 return q < trips(item) ? q : null;
40}
41function trainTime(direction, q, at) {
42 return direction.line.first + direction.line.delay + q * direction.line.headway + direction.offsets[at];
43}
44function ride(direction, boarding, alighting, q) {
45 return { kind: 'ride', line: direction.line.id, terminal: direction.stops[direction.stops.length - 1],
46 from: direction.stops[boarding], departure: trainTime(direction, q, boarding),
47 to: direction.stops[alighting], arrival: trainTime(direction, q, alighting), stops: alighting - boarding };
48}
49function update(labels, id, candidate) {
50 const old = labels.get(id);
51 if (!old || old.time > candidate.time) { labels.set(id, candidate); return true; }
52 return false;
53}
54function earliest(origin, departure) {
55 const labels = new Map([[origin, { time: departure, journey: [] }]]), settled = new Set();
56 const service = orderedLines();
57 while (true) {
58 const choices = [...labels.keys()].filter(id => !settled.has(id));
59 if (!choices.length) return labels;
60 choices.sort((a, b) => labels.get(a).time - labels.get(b).time || compare(a, b));
61 const u = choices[0], label = labels.get(u), s = stations.get(u);
62 settled.add(u);
63 const ready = label.time + (u === origin ? 0 : s.transfer);
64 for (const item of service) if (!item.suspended && item.stops.includes(u)) {
65 for (const direction of directions(item)) {
66 const i = direction.stops.indexOf(u), last = direction.stops.length - 1;
67 if (i === last) continue;
68 const q = nextTrain(direction, i, ready);
69 if (q === null) continue;
70 for (let j = i + 1; j <= last; j++) {
71 const v = direction.stops[j];
72 if (closed.has(pair(direction.stops[j - 1], v))) break;
73 if (stations.get(v).closed) continue;
74 const leg = ride(direction, i, j, q);
75 update(labels, v, { time: leg.arrival, journey: [...label.journey, leg] });
76 }
77 }
78 }
79 for (const v of [...s.walks.keys()].sort()) {
80 const arrival = label.time + s.walks.get(v);
81 if (!stations.get(v).closed && arrival <= 1440) {
82 const leg = { kind: 'walk', from: u, departure: label.time, to: v, arrival };
83 update(labels, v, { time: arrival, journey: [...label.journey, leg] });
84 }
85 }
86 }
87}
88function walkStep(labels, sources, marked) {
89 const recorded = new Map([...sources].map(id => [id, labels.get(id)]));
90 for (const u of [...sources].sort()) {
91 const label = recorded.get(u), s = stations.get(u);
92 for (const v of [...s.walks.keys()].sort()) {
93 const arrival = label.time + s.walks.get(v);
94 if (stations.get(v).closed || arrival > 1440) continue;
95 const leg = { kind: 'walk', from: u, departure: label.time, to: v, arrival };
96 if (update(labels, v, { time: arrival, journey: [...label.journey, leg] })) marked.add(v);
97 }
98 }
99}
100function rounds(origin, departure) {
101 let labels = new Map([[origin, { time: departure, journey: [] }]]), marked = new Set([origin]);
102 walkStep(labels, new Set([origin]), marked);
103 const result = [labels], service = orderedLines();
104 while (marked.size) {
105 const previous = labels, previousMarked = marked;
106 labels = new Map(previous); marked = new Set();
107 const walking = new Set();
108 for (const item of service) if (!item.suspended) {
109 for (const direction of directions(item)) {
110 let boarded = null, boarding = 0;
111 let journey = [];
112 for (let j = 0; j < direction.stops.length; j++) {
113 const v = direction.stops[j];
114 if (j && closed.has(pair(direction.stops[j - 1], v))) boarded = null;
115 if (stations.get(v).closed) continue;
116 if (boarded !== null) {
117 const leg = ride(direction, boarding, j, boarded);
118 if (update(labels, v, { time: leg.arrival, journey: [...journey, leg] })) {
119 walking.add(v); marked.add(v);
120 }
121 }
122 if (j + 1 < direction.stops.length && previousMarked.has(v)) {
123 const old = previous.get(v), ready = old.time + (v === origin ? 0 : stations.get(v).transfer);
124 const q = nextTrain(direction, j, ready);
125 if (q !== null && (boarded === null || q < boarded)) { boarded = q; boarding = j; journey = old.journey; }
126 }
127 }
128 }
129 }
130 walkStep(labels, walking, marked);
131 result.push(labels);
132 }
133 return result;
134}
135function printJourney(heading, departure, label) {
136 const rides = label.journey.filter(leg => leg.kind === 'ride').length, walks = label.journey.filter(leg => leg.kind === 'walk').length;
137 output.push(`${heading} ${clock(departure)} -> ${clock(label.time)} total ${label.time - departure} rides ${rides} walks ${walks}`);
138 let previous = departure;
139 for (const leg of label.journey) {
140 if (leg.kind === 'ride') output.push(`RIDE ${leg.line} toward ${leg.terminal} ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)} wait ${leg.departure - previous} stops ${leg.stops}`);
141 else if (leg.kind === 'walk') output.push(`WALK ${leg.from} ${clock(leg.departure)} -> ${leg.to} ${clock(leg.arrival)}`);
142 else output.push(`STAY ${leg.station} ${clock(leg.departure)} -> ${clock(leg.arrival)}`);
143 previous = leg.arrival;
144 }
145}
146function printFare(origin, destination, word, fareClass, label) {
147 const tickets = [];
148 let total = 0, previous = null;
149 const half = (value) => Math.ceil(value / 20) * 10;
150 for (const leg of label.journey) if (leg.kind === 'ride') {
151 const item = lines.get(leg.line), direction = directions(item).find(d => d.stops[d.stops.length - 1] === leg.terminal);
152 const start = direction.stops.indexOf(leg.from), end = direction.stops.indexOf(leg.to);
153 const zones = new Set(direction.stops.slice(start, end + 1).map(id => stations.get(id).zone)).size;
154 const base = tariff.prices[Math.min(zones, tariff.prices.length) - 1], price = fareClass === 'CHILD' ? half(base) : base;
155 const cap = fareClass === 'CHILD' ? half(tariff.discount) : tariff.discount;
156 const discount = previous && leg.departure - previous.arrival <= tariff.window ? Math.min(cap, price) : 0;
157 const payment = price - discount;
158 total += payment;
159 tickets.push(`TICKET ${leg.line} ${leg.from} ${leg.to} zones ${zones} price ${price} discount ${discount} pay ${payment}`);
160 previous = leg;
161 }
162 output.push(`FARE ${origin} ${destination} ${word} ${fareClass} rides ${tickets.length} total ${total}`, ...tickets);
163}
164function execute(commandLine) {
165 const [command, ...args] = commandLine.split(' '), a = args[0], b = args[1];
166 switch (command) {
167 case 'STATION':
168 if (stations.has(a)) fail(`station ${a} already exists`);
169 stations.set(a, { id: a, zone: Number(b), name: args[2], transfer: 0, closed: false, walks: new Map() });
170 output.push(`STATION ${a} zone ${b} ${args[2]}`);
171 break;
172 case 'LINE': {
173 if (lines.has(a)) fail(`line ${a} already exists`);
174 const stops = [], offsets = [0];
175 for (let i = 4; i < args.length; i += 2) stops.push(args[i]);
176 for (const id of stops) station(id);
177 const seen = new Set();
178 for (const id of stops) { if (seen.has(id)) fail(`station ${id} appears twice on line ${a}`); seen.add(id); }
179 for (let i = 5; i < args.length; i += 2) offsets.push(offsets[offsets.length - 1] + Number(args[i]));
180 const first = time(b), last = time(args[2]), headway = Number(args[3]), total = offsets[offsets.length - 1];
181 if (first > last) fail(`line ${a} first departure is after last departure`);
182 if (last + total > 1440) fail(`line ${a} runs past 24:00`);
183 const item = { id: a, first, last, headway, stops, offsets, delay: 0, suspended: false };
184 lines.set(a, item);
185 output.push(`LINE ${a} stops ${stops.length} time ${total} trips ${trips(item)}`);
186 break;
187 }
188 case 'TRANSFER': station(a).transfer = Number(b); output.push(`TRANSFER ${a} ${b}`); break;
189 case 'WALK': {
190 const x = station(a), y = station(b), duration = Number(args[2]);
191 if (a === b) fail(`walk from ${a} to itself`);
192 const old = x.walks.get(b);
193 x.walks.set(b, duration); y.walks.set(a, duration);
194 output.push(`WALK ${a} ${b} ${duration}${old === undefined ? '' : ` was ${old}`}`);
195 break;
196 }
197 case 'TARIFF': {
198 const prices = args.slice(2).map(Number);
199 if (prices.some((price, i) => i > 0 && price < prices[i - 1])) fail('tariff prices must not decrease');
200 tariff = { discount: Number(a), window: Number(b), prices };
201 output.push(`TARIFF zones ${prices.length} discount ${a} window ${b}`);
202 break;
203 }
204 case 'CLOSE': case 'OPEN': {
205 const s = station(a), closing = command === 'CLOSE';
206 if (args.length === 1) {
207 if (s.closed === closing) fail(`station ${a} is ${closing ? 'already closed' : 'not closed'}`);
208 s.closed = closing;
209 output.push(`${command} ${a}`);
210 } else {
211 station(b);
212 const key = pair(a, b);
213 const affected = orderedLines().filter(item => item.stops.some((id, i) => i > 0 && pair(item.stops[i - 1], id) === key));
214 if (closing && !affected.length) fail(`no segment ${a}-${b}`);
215 if (closed.has(key) === closing) fail(`segment ${a}-${b} is ${closing ? 'already closed' : 'not closed'}`);
216 if (closing) closed.add(key); else closed.delete(key);
217 output.push(`${command} ${a}-${b}${closing ? ` lines ${affected.map(item => item.id).join(' ')}` : ''}`);
218 }
219 break;
220 }
221 case 'DELAY': {
222 const item = line(a), delay = Number(b);
223 if (item.last + delay + item.offsets[item.offsets.length - 1] > 1440) fail(`line ${a} runs past 24:00`);
224 item.delay = delay; output.push(`DELAY ${a} ${delay}`);
225 break;
226 }
227 case 'SUSPEND': case 'RESUME': {
228 const item = line(a), suspended = command === 'SUSPEND';
229 if (item.suspended === suspended) fail(`line ${a} is ${suspended ? 'already suspended' : 'not suspended'}`);
230 item.suspended = suspended; output.push(`${command} ${a}`);
231 break;
232 }
233 case 'ROUTE': case 'FEWEST': case 'PARETO': case 'FARE': case 'LASTTRAIN': case 'ARRIVEBY': case 'PROFILE': {
234 common(a, b);
235 if (command === 'FARE') {
236 if (args[3] !== 'ADULT' && args[3] !== 'CHILD') fail(`unknown fare class ${args[3]}`);
237 if (!tariff) fail('no tariff');
238 }
239 if (command === 'PROFILE' && time(args[2]) > time(args[3])) fail(`empty interval ${args[2]} ${args[3]}`);
240 if (command === 'LASTTRAIN' || command === 'ARRIVEBY') {
241 const deadline = command === 'LASTTRAIN' ? 1440 : time(args[2]);
242 const heading = `${command} ${a} ${b}${command === 'ARRIVEBY' ? ` by ${args[2]}` : ''}`;
243 let found = false;
244 for (let departure = deadline - 1; departure >= 0; departure--) {
245 const label = earliest(a, departure).get(b);
246 if (label && label.time <= deadline) { printJourney(heading, departure, label); found = true; break; }
247 }
248 if (!found) output.push(`${heading} none`);
249 } else if (command === 'PROFILE') {
250 const start = time(args[2]), end = time(args[3]), arrivals = [];
251 for (let departure = start; departure <= end; departure++) arrivals.push(earliest(a, departure).get(b)?.time);
252 const selected = [];
253 for (let i = 0; i < arrivals.length; i++) if (arrivals[i] !== undefined &&
254 (i === arrivals.length - 1 || arrivals[i + 1] === undefined || arrivals[i] < arrivals[i + 1])) selected.push(i);
255 output.push(`PROFILE ${a} ${b} ${args[2]} ${args[3]} count ${selected.length}`);
256 for (const i of selected) output.push(`${clock(start + i)} -> ${clock(arrivals[i])} total ${arrivals[i] - start - i}`);
257 } else if (command === 'FEWEST' || command === 'PARETO') {
258 const departure = time(args[2]), all = rounds(a, departure);
259 if (command === 'FEWEST') {
260 const k = all.findIndex(labels => labels.has(b));
261 if (k < 0) output.push(`FEWEST ${a} ${b} ${args[2]} unreachable`);
262 else printJourney(`FEWEST ${a} ${b} trips ${k}`, departure, all[k].get(b));
263 } else {
264 const selected = all.map((labels, k) => k).filter(k => all[k].has(b) && (!k || !all[k - 1].has(b) || all[k - 1].get(b).time > all[k].get(b).time));
265 output.push(`PARETO ${a} ${b} ${args[2]} options ${selected.length}`);
266 for (const k of selected) printJourney(`OPTION trips ${k}`, departure, all[k].get(b));
267 }
268 } else {
269 const departure = time(args[2]), label = earliest(a, departure).get(b);
270 if (!label) output.push(`${command} ${a} ${b} ${args[2]}${command === 'FARE' ? ` ${args[3]}` : ''} unreachable`);
271 else if (command === 'FARE') printFare(a, b, args[2], args[3], label);
272 else printJourney(`ROUTE ${a} ${b}`, departure, label);
273 }
274 break;
275 }
276 case 'VIA': {
277 const via = b, destination = args[2];
278 station(a); station(via); station(destination);
279 if (a === destination) fail('origin and destination are the same');
280 if (via === a || via === destination) fail(`via station ${via} must differ from origin and destination`);
281 openStation(a); openStation(via); openStation(destination);
282 const departure = time(args[3]), first = earliest(a, departure).get(via);
283 const ready = first ? first.time + stations.get(via).transfer : 0;
284 const second = first ? earliest(via, ready).get(destination) : undefined;
285 if (!first || !second) output.push(`VIA ${a} ${via} ${destination} ${args[3]} unreachable`);
286 else {
287 const stay = { kind: 'stay', station: via, departure: first.time, arrival: ready };
288 printJourney(`VIA ${a} ${via} ${destination}`, departure, { time: second.time, journey: [...first.journey, stay, ...second.journey] });
289 }
290 break;
291 }
292 case 'REACH': {
293 station(a); openStation(a);
294 const departure = time(b), labels = earliest(a, departure), limit = departure + Number(args[2]);
295 const reached = [...labels.keys()].filter(id => labels.get(id).time <= limit).sort((x, y) => labels.get(x).time - labels.get(y).time || compare(x, y));
296 output.push(`REACH ${a} ${b} ${args[2]} count ${reached.length}`);
297 for (const id of reached) {
298 const label = labels.get(id);
299 output.push(`${id} ${stations.get(id).name} ${clock(label.time)} ${label.time - departure} rides ${label.journey.filter(leg => leg.kind === 'ride').length}`);
300 }
301 break;
302 }
303 case 'TIMETABLE': {
304 const s = station(a), item = line(b), toward = args[2];
305 if (!item.stops.includes(a)) fail(`station ${a} is not on line ${b}`);
306 if (toward !== item.stops[0] && toward !== item.stops[item.stops.length - 1]) fail(`${toward} is not a terminal of line ${b}`);
307 if (a === toward) fail(`no departures from ${a} toward ${toward}`);
308 if (item.suspended) fail(`line ${b} is suspended`);
309 if (s.closed) fail(`station ${a} is closed`);
310 const direction = directions(item).find(d => d.stops[d.stops.length - 1] === toward), at = direction.stops.indexOf(a);
311 const q = nextTrain(direction, at, time(args[3])), times = [];
312 if (q !== null) for (let i = q; i < trips(item) && times.length < Number(args[4]); i++) times.push(clock(trainTime(direction, i, at)));
313 output.push(`TIMETABLE ${a} ${b} toward ${toward} ${args[3]} next ${times.length}${times.length ? ` ${times.join(' ')}` : ''}`);
314 break;
315 }
316 case 'DEPARTURES': case 'ARRIVALS': {
317 station(a); openStation(a);
318 const events = [];
319 for (const item of orderedLines()) if (!item.suspended && item.stops.includes(a)) for (const direction of directions(item)) {
320 const at = direction.stops.indexOf(a), last = direction.stops.length - 1;
321 if (command === 'DEPARTURES' ? at === last : at === 0) continue;
322 const q = nextTrain(direction, at, time(b));
323 if (q !== null) for (let i = q; i < trips(item); i++) events.push({ time: trainTime(direction, i, at), line: item.id, order: direction.order, terminal: direction.stops[last], start: direction.stops[0] });
324 }
325 events.sort((x, y) => x.time - y.time || compare(x.line, y.line) || x.order - y.order);
326 const selected = events.slice(0, Number(args[2]));
327 output.push(`${command} ${a} ${b} count ${selected.length}`);
328 for (const event of selected) output.push(`${clock(event.time)} ${event.line} ${command === 'DEPARTURES' ? `toward ${event.terminal}` : `from ${event.start}`}`);
329 break;
330 }
331 case 'TRIP': {
332 const item = line(a);
333 if (b !== item.stops[0] && b !== item.stops[item.stops.length - 1]) fail(`${b} is not a terminal of line ${a}`);
334 if (item.suspended) fail(`line ${a} is suspended`);
335 const direction = directions(item).find(d => d.stops[d.stops.length - 1] === b), q = nextTrain(direction, 0, time(args[2]));
336 if (q === null) output.push(`TRIP ${a} toward ${b} ${args[2]} none`);
337 else {
338 output.push(`TRIP ${a} toward ${b} ${args[2]} trip ${q + 1} departs ${clock(trainTime(direction, q, 0))}`);
339 direction.stops.forEach((id, j) => output.push(`${clock(trainTime(direction, q, j))} ${id}${stations.get(id).closed ? ' closed' : ''}${j && closed.has(pair(direction.stops[j - 1], id)) ? ' blocked' : ''}`));
340 }
341 break;
342 }
343 case 'SHOWLINE': {
344 const item = line(a);
345 output.push(`SHOWLINE ${a} ${item.suspended ? 'suspended' : 'running'} first ${clock(item.first)} last ${clock(item.last)} headway ${item.headway} delay ${item.delay} trips ${trips(item)}`);
346 item.stops.forEach((id, i) => output.push(`${item.offsets[i]} ${id} zone ${stations.get(id).zone}${stations.get(id).closed ? ' closed' : ''}${i && closed.has(pair(item.stops[i - 1], id)) ? ' blocked' : ''}`));
347 break;
348 }
349 case 'INFO': {
350 const s = station(a), service = orderedLines().filter(item => item.stops.includes(a)), neighbors = [...s.walks.keys()].sort();
351 output.push(`INFO ${a} ${s.name} zone ${s.zone} transfer ${s.transfer} ${s.closed ? 'closed' : 'open'}`,
352 `LINES ${service.length ? service.map(item => item.id).join(' ') : 'none'}`,
353 `WALKS ${neighbors.length ? neighbors.map(id => `${id} ${s.walks.get(id)}`).join(' ') : 'none'}`);
354 break;
355 }
356 case 'STATUS': {
357 const walkCount = [...stations.values()].reduce((sum, s) => sum + s.walks.size, 0) / 2;
358 output.push(`STATUS stations ${stations.size} lines ${lines.size} walks ${walkCount}`);
359 for (const id of [...stations.keys()].sort()) if (stations.get(id).closed) output.push(`CLOSED ${id}`);
360 const segments = [...closed].sort((x, y) => { const a = x.split('-'), b = y.split('-'); return compare(a[0], b[0]) || compare(a[1], b[1]); });
361 for (const segment of segments) output.push(`CLOSED ${segment}`);
362 for (const item of orderedLines()) if (item.suspended) output.push(`SUSPENDED ${item.id}`);
363 for (const item of orderedLines()) if (item.delay) output.push(`DELAYED ${item.id} ${item.delay}`);
364 break;
365 }
366 }
367}
368const input = fs.readFileSync(0, 'utf8').split(/\r?\n/), count = Number(input[0]);
369for (const command of input.slice(1, count + 1)) {
370 try { execute(command); }
371 catch (error) { output.push(error.message); }
372}
373if (output.length) process.stdout.write(output.join('\n') + '\n');
