1require 'set'

3Station = Struct.new(:zone, :name, :transfer, :closed)
4MetroLine = Struct.new(:id, :first, :last, :headway, :stops, :offsets, :delay, :suspended) do
5 def trips
6 (last - first) / headway + 1
7 end
8
9 def duration
10 offsets.last
11 end
12
13 def directions
14 [[stops, offsets], [stops.reverse, offsets.reverse.map { |c| duration - c }]]
15 end
16
17 def next_train(offset, ready)
18 q = [0, (ready - first - delay - offset + headway - 1) / headway].max
19 q < trips ? q : nil
20 end
21
22 def time(q, offset)
23 first + delay + q * headway + offset
24 end
25end
26
27class MetroError < StandardError; end
28
29class Metro
30 def initialize
31 @stations = {}
32 @lines = {}
33 @walks = {}
34 @closed = Set.new
35 @tariff = nil
36 end
37
38 def minutes(text)
39 hour, minute = text.split(':').map(&:to_i)
40 hour * 60 + minute
41 end
42
43 def clock(value)
44 format('%02d:%02d', value / 60, value % 60)
45 end
46
47 def segment(a, b)
48 [a, b].sort
49 end
50
51 def blocked?(a, b)
52 @closed.include?(segment(a, b))
53 end
54
55 def station_check(id)
56 raise MetroError, "unknown station #{id}" unless @stations.key?(id)
57 end
58
59 def line_check(id)
60 raise MetroError, "unknown line #{id}" unless @lines.key?(id)
61 end
62
63 def open_check(id)
64 raise MetroError, "station #{id} is closed" if @stations[id].closed
65 end
66
67 def journey_check(origin, destination)
68 station_check(origin)
69 station_check(destination)
70 raise MetroError, 'origin and destination are the same' if origin == destination
71 open_check(origin)
72 open_check(destination)
73 end
74
75 def running_lines
76 @lines.keys.sort.map { |id| @lines[id] }.reject(&:suspended)
77 end
78
79 def update(labels, destination, time, journey)
80 return false if labels.key?(destination) && labels[destination][0] <= time
81 labels[destination] = [time, journey]
82 true
83 end
84
85 def earliest(origin, departure)
86 labels = { origin => [departure, []] }
87 settled = Set.new
88 lines = running_lines
89 loop do
90 u = (labels.keys - settled.to_a).min_by { |id| [labels[id][0], id] }
91 break unless u
92 settled << u
93 arrival, journey = labels[u]
94 ready = arrival + (u == origin ? 0 : @stations[u].transfer)
95 lines.each do |line|
96 next unless line.stops.include?(u)
97 line.directions.each do |stops, offsets|
98 i = stops.index(u)
99 next if i == stops.length - 1
100 q = line.next_train(offsets[i], ready)
101 next unless q
102 boarding = line.time(q, offsets[i])
103 ((i + 1)...stops.length).each do |j|
104 break if blocked?(stops[j - 1], stops[j])
105 next if @stations[stops[j]].closed
106 time = line.time(q, offsets[j])
107 leg = [:ride, line.id, stops.last, u, boarding, stops[j], time, j - i]
108 update(labels, stops[j], time, journey + [leg])
109 end
110 end
111 end
112 @walks[u].keys.sort.each do |v|
113 time = arrival + @walks[u][v]
114 next if @stations[v].closed || time > 1440
115 update(labels, v, time, journey + [[:walk, u, arrival, v, time]])
116 end
117 end
118 labels
119 end
120
121 def walk_step(labels, selected, marked)
122 snapshot = selected.to_a.sort.to_h { |u| [u, labels[u]] }
123 snapshot.each do |u, (arrival, journey)|
124 @walks[u].keys.sort.each do |v|
125 time = arrival + @walks[u][v]
126 next if @stations[v].closed || time > 1440
127 marked << v if update(labels, v, time, journey + [[:walk, u, arrival, v, time]])
128 end
129 end
130 end
131
132 def rounds(origin, departure)
133 labels = { origin => [departure, []] }
134 marked = Set[origin]
135 walk_step(labels, Set[origin], marked)
136 result = [labels]
137 lines = running_lines
138 until marked.empty?
139 previous = labels
140 previous_marked = marked
141 labels = previous.dup
142 marked = Set.new
143 selected = Set.new
144 lines.each do |line|
145 line.directions.each do |stops, offsets|
146 boarded = nil
147 b = nil
148 boarding = nil
149 journey = nil
150 stops.each_with_index do |u, j|
151 boarded = nil if j.positive? && blocked?(stops[j - 1], u)
152 next if @stations[u].closed
153 if boarded
154 time = line.time(boarded, offsets[j])
155 leg = [:ride, line.id, stops.last, stops[b], boarding, u, time, j - b]
156 if update(labels, u, time, journey + [leg])
157 selected << u
158 marked << u
159 end
160 end
161 next if j == stops.length - 1 || !previous_marked.include?(u)
162 arrival, candidate_journey = previous[u]
163 ready = arrival + (u == origin ? 0 : @stations[u].transfer)
164 candidate = line.next_train(offsets[j], ready)
165 if candidate && (boarded.nil? || candidate < boarded)
166 boarded = candidate
167 b = j
168 boarding = line.time(candidate, offsets[j])
169 journey = candidate_journey
170 end
171 end
172 end
173 end
174 walk_step(labels, selected, marked)
175 result << labels
176 end
177 result
178 end

180 def print_journey(heading, departure, label)
181 arrival, journey = label
182 rides = journey.count { |leg| leg[0] == :ride }
183 walks = journey.count { |leg| leg[0] == :walk }
184 puts "#{heading} #{clock(departure)} -> #{clock(arrival)} total #{arrival - departure} rides #{rides} walks #{walks}"
185 previous_end = departure
186 journey.each do |leg|
187 case leg[0]
188 when :ride
189 _, line, toward, from, start, to, finish, count = leg
190 puts "RIDE #{line} toward #{toward} #{from} #{clock(start)} -> #{to} #{clock(finish)} wait #{start - previous_end} stops #{count}"
191 previous_end = finish
192 when :walk
193 _, from, start, to, finish = leg
194 puts "WALK #{from} #{clock(start)} -> #{to} #{clock(finish)}"
195 previous_end = finish
196 when :stay
197 _, station, start, finish = leg
198 puts "STAY #{station} #{clock(start)} -> #{clock(finish)}"
199 previous_end = finish
200 end
201 end
202 end

204 def print_fare(origin, destination, time_text, fare_class, journey)
205 discount_cap, window, prices = @tariff
206 half = ->(x) { (x + 19) / 20 * 10 }
207 cap = fare_class == 'CHILD' ? half.call(discount_cap) : discount_cap
208 tickets = []
209 previous_end = nil
210 journey.each do |leg|
211 next unless leg[0] == :ride
212 _, line_id, toward, from, boarding, to, arrival, = leg
213 line = @lines[line_id]
214 stops = line.directions.find { |sequence, _| sequence.last == toward }[0]
215 zones = stops[stops.index(from)..stops.index(to)].map { |id| @stations[id].zone }.uniq.length
216 price = prices[[zones, prices.length].min - 1]
217 price = half.call(price) if fare_class == 'CHILD'
218 discount = previous_end && boarding - previous_end <= window ? [cap, price].min : 0
219 tickets << [line_id, from, to, zones, price, discount, price - discount]
220 previous_end = arrival
221 end
222 puts "FARE #{origin} #{destination} #{time_text} #{fare_class} rides #{tickets.length} total #{tickets.sum { |t| t[6] }}"
223 tickets.each do |line, from, to, zones, price, discount, pay|
224 puts "TICKET #{line} #{from} #{to} zones #{zones} price #{price} discount #{discount} pay #{pay}"
225 end
226 end

228 def search_command(command, args)
229 origin, destination = args
230 journey_check(origin, destination)
231 departure = minutes(args[2]) if args[2]
232 if command == 'FARE'
233 raise MetroError, "unknown fare class #{args[3]}" unless %w[ADULT CHILD].include?(args[3])
234 raise MetroError, 'no tariff' unless @tariff
235 elsif command == 'PROFILE' && departure > minutes(args[3])
236 raise MetroError, "empty interval #{args[2]} #{args[3]}"
237 end
238 case command
239 when 'ROUTE', 'FARE'
240 label = earliest(origin, departure)[destination]
241 if label
242 if command == 'ROUTE'
243 print_journey("ROUTE #{origin} #{destination}", departure, label)
244 else
245 print_fare(origin, destination, args[2], args[3], label[1])
246 end
247 else
248 puts "#{command} #{args.join(' ')} unreachable"
249 end
250 when 'FEWEST', 'PARETO'
251 all = rounds(origin, departure)
252 if command == 'FEWEST'
253 k = all.index { |labels| labels.key?(destination) }
254 if k
255 print_journey("FEWEST #{origin} #{destination} trips #{k}", departure, all[k][destination])
256 else
257 puts "FEWEST #{args.join(' ')} unreachable"
258 end
259 else
260 options = all.each_index.select do |k|
261 all[k].key?(destination) && (k.zero? || !all[k - 1].key?(destination) || all[k][destination][0] < all[k - 1][destination][0])
262 end
263 puts "PARETO #{args.join(' ')} options #{options.length}"
264 options.each { |k| print_journey("OPTION trips #{k}", departure, all[k][destination]) }
265 end
266 when 'LASTTRAIN', 'ARRIVEBY'
267 start = command == 'LASTTRAIN' ? 1439 : departure - 1
268 heading = "#{command} #{origin} #{destination}"
269 heading += " by #{args[2]}" if command == 'ARRIVEBY'
270 start.downto(0) do |time|
271 label = earliest(origin, time)[destination]
272 next unless label && (command == 'LASTTRAIN' || label[0] <= departure)
273 print_journey(heading, time, label)
274 return
275 end
276 puts "#{heading} none"
277 when 'PROFILE'
278 finish = minutes(args[3])
279 arrivals = (departure..finish).to_h { |t| [t, earliest(origin, t)[destination]&.first] }
280 selected = (departure..finish).select do |t|
281 arrivals[t] && (t == finish || !arrivals[t + 1] || arrivals[t] < arrivals[t + 1])
282 end
283 puts "PROFILE #{args.join(' ')} count #{selected.length}"
284 selected.each { |t| puts "#{clock(t)} -> #{clock(arrivals[t])} total #{arrivals[t] - t}" }
285 end
286 end

288 def execute(line)
289 command, *args = line.split(' ')
290 a, b = args
291 case command
292 when 'STATION'
293 raise MetroError, "station #{a} already exists" if @stations.key?(a)
294 @stations[a] = Station.new(b.to_i, args[2], 0, false)
295 @walks[a] = {}
296 puts "STATION #{a} zone #{b} #{args[2]}"
297 when 'LINE'
298 raise MetroError, "line #{a} already exists" if @lines.key?(a)
299 stops = args.drop(4).each_slice(2).map(&:first)
300 stops.each { |id| station_check(id) }
301 seen = Set.new
302 stops.each do |id|
303 raise MetroError, "station #{id} appears twice on line #{a}" if seen.include?(id)
304 seen << id
305 end
306 first = minutes(b)
307 last = minutes(args[2])
308 raise MetroError, "line #{a} first departure is after last departure" if first > last
309 offsets = [0]
310 args.drop(5).each_slice(2) { |duration, _| offsets << offsets.last + duration.to_i }
311 raise MetroError, "line #{a} runs past 24:00" if last + offsets.last > 1440
312 service = MetroLine.new(a, first, last, args[3].to_i, stops, offsets, 0, false)
313 @lines[a] = service
314 puts "LINE #{a} stops #{stops.length} time #{service.duration} trips #{service.trips}"
315 when 'TRANSFER'
316 station_check(a)
317 @stations[a].transfer = b.to_i
318 puts "TRANSFER #{a} #{b}"
319 when 'WALK'
320 station_check(a)
321 station_check(b)
322 raise MetroError, "walk from #{a} to itself" if a == b
323 old = @walks[a][b]
324 @walks[a][b] = @walks[b][a] = args[2].to_i
325 puts "WALK #{args.join(' ')}#{old ? " was #{old}" : ''}"
326 when 'TARIFF'
327 prices = args.drop(2).map(&:to_i)
328 raise MetroError, 'tariff prices must not decrease' if prices.each_cons(2).any? { |x, y| x > y }
329 @tariff = [a.to_i, b.to_i, prices]
330 puts "TARIFF zones #{prices.length} discount #{a} window #{b}"
331 when 'CLOSE', 'OPEN'
332 station_check(a)
333 if b
334 station_check(b)
335 key = segment(a, b)
336 if command == 'CLOSE'
337 affected = @lines.keys.sort.select { |id| @lines[id].stops.each_cons(2).any? { |x, y| segment(x, y) == key } }
338 raise MetroError, "no segment #{a}-#{b}" if affected.empty?
339 raise MetroError, "segment #{a}-#{b} is already closed" if @closed.include?(key)
340 @closed << key
341 puts "CLOSE #{a}-#{b} lines #{affected.join(' ')}"
342 else
343 raise MetroError, "segment #{a}-#{b} is not closed" unless @closed.include?(key)
344 @closed.delete(key)
345 puts "OPEN #{a}-#{b}"
346 end
347 else
348 current = @stations[a].closed
349 if command == 'CLOSE' && current
350 raise MetroError, "station #{a} is already closed"
351 elsif command == 'OPEN' && !current
352 raise MetroError, "station #{a} is not closed"
353 end
354 @stations[a].closed = command == 'CLOSE'
355 puts "#{command} #{a}"
356 end
357 when 'DELAY'
358 line_check(a)
359 raise MetroError, "line #{a} runs past 24:00" if @lines[a].last + b.to_i + @lines[a].duration > 1440
360 @lines[a].delay = b.to_i
361 puts "DELAY #{a} #{b}"
362 when 'SUSPEND', 'RESUME'
363 line_check(a)
364 if command == 'SUSPEND' && @lines[a].suspended
365 raise MetroError, "line #{a} is already suspended"
366 elsif command == 'RESUME' && !@lines[a].suspended
367 raise MetroError, "line #{a} is not suspended"
368 end
369 @lines[a].suspended = command == 'SUSPEND'
370 puts "#{command} #{a}"
371 when 'ROUTE', 'FEWEST', 'PARETO', 'FARE', 'LASTTRAIN', 'ARRIVEBY', 'PROFILE'
372 search_command(command, args)
373 when 'VIA'
374 origin, via, destination, time_text = args
375 [origin, via, destination].each { |id| station_check(id) }
376 raise MetroError, 'origin and destination are the same' if origin == destination
377 if via == origin || via == destination
378 raise MetroError, "via station #{via} must differ from origin and destination"
379 end
380 [origin, via, destination].each { |id| open_check(id) }
381 departure = minutes(time_text)
382 first = earliest(origin, departure)[via]
383 if first
384 ready = first[0] + @stations[via].transfer
385 second = earliest(via, ready)[destination]
386 end
387 if first && second
388 journey = first[1] + [[:stay, via, first[0], ready]] + second[1]
389 print_journey("VIA #{origin} #{via} #{destination}", departure, [second[0], journey])
390 else
391 puts "VIA #{args.join(' ')} unreachable"
392 end
393 when 'REACH'
394 station_check(a)
395 open_check(a)
396 departure = minutes(b)
397 labels = earliest(a, departure)
398 selected = labels.keys.select { |id| labels[id][0] <= departure + args[2].to_i }.sort_by { |id| [labels[id][0], id] }
399 puts "REACH #{args.join(' ')} count #{selected.length}"
400 selected.each do |id|
401 arrival, journey = labels[id]
402 puts "#{id} #{@stations[id].name} #{clock(arrival)} #{arrival - departure} rides #{journey.count { |leg| leg[0] == :ride }}"
403 end
404 when 'TIMETABLE'
405 station_check(a)
406 line_check(b)
407 service = @lines[b]
408 raise MetroError, "station #{a} is not on line #{b}" unless service.stops.include?(a)
409 toward = args[2]
410 direction = service.directions.find { |stops, _| stops.last == toward }
411 raise MetroError, "#{toward} is not a terminal of line #{b}" unless direction
412 raise MetroError, "no departures from #{a} toward #{toward}" if a == toward
413 raise MetroError, "line #{b} is suspended" if service.suspended
414 open_check(a)
415 stops, offsets = direction
416 offset = offsets[stops.index(a)]
417 times = (0...service.trips).map { |q| service.time(q, offset) }.select { |t| t >= minutes(args[3]) }.take(args[4].to_i)
418 puts "TIMETABLE #{a} #{b} toward #{toward} #{args[3]} next #{times.length}#{times.empty? ? '' : " #{times.map { |t| clock(t) }.join(' ')}"}"
419 when 'DEPARTURES', 'ARRIVALS'
420 station_check(a)
421 open_check(a)
422 entries = []
423 running_lines.each do |service|
424 next unless service.stops.include?(a)
425 service.directions.each_with_index do |(stops, offsets), direction|
426 i = stops.index(a)
427 next if command == 'DEPARTURES' ? i == stops.length - 1 : i.zero?
428 (0...service.trips).each do |q|
429 time = service.time(q, offsets[i])
430 entries << [time, service.id, direction, command == 'DEPARTURES' ? stops.last : stops.first] if time >= minutes(b)
431 end
432 end
433 end
434 entries = entries.sort.take(args[2].to_i)
435 puts "#{command} #{a} #{b} count #{entries.length}"
436 entries.each { |time, id, _, terminal| puts "#{clock(time)} #{id} #{command == 'DEPARTURES' ? 'toward' : 'from'} #{terminal}" }
437 when 'TRIP'
438 line_check(a)
439 service = @lines[a]
440 direction = service.directions.find { |stops, _| stops.last == b }
441 raise MetroError, "#{b} is not a terminal of line #{a}" unless direction
442 raise MetroError, "line #{a} is suspended" if service.suspended
443 q = service.next_train(0, minutes(args[2]))
444 if q
445 stops, offsets = direction
446 puts "TRIP #{a} toward #{b} #{args[2]} trip #{q + 1} departs #{clock(service.time(q, 0))}"
447 stops.each_with_index do |id, i|
448 suffix = @stations[id].closed ? ' closed' : ''
449 suffix += ' blocked' if i.positive? && blocked?(stops[i - 1], id)
450 puts "#{clock(service.time(q, offsets[i]))} #{id}#{suffix}"
451 end
452 else
453 puts "TRIP #{a} toward #{b} #{args[2]} none"
454 end
455 when 'SHOWLINE'
456 line_check(a)
457 service = @lines[a]
458 puts "SHOWLINE #{a} #{service.suspended ? 'suspended' : 'running'} first #{clock(service.first)} last #{clock(service.last)} headway #{service.headway} delay #{service.delay} trips #{service.trips}"
459 service.stops.each_with_index do |id, i|
460 suffix = @stations[id].closed ? ' closed' : ''
461 suffix += ' blocked' if i.positive? && blocked?(service.stops[i - 1], id)
462 puts "#{service.offsets[i]} #{id} zone #{@stations[id].zone}#{suffix}"
463 end
464 when 'INFO'
465 station_check(a)
466 station = @stations[a]
467 puts "INFO #{a} #{station.name} zone #{station.zone} transfer #{station.transfer} #{station.closed ? 'closed' : 'open'}"
468 lines = @lines.keys.sort.select { |id| @lines[id].stops.include?(a) }
469 puts "LINES #{lines.empty? ? 'none' : lines.join(' ')}"
470 neighbors = @walks[a].keys.sort
471 puts "WALKS #{neighbors.empty? ? 'none' : neighbors.map { |v| "#{v} #{@walks[a][v]}" }.join(' ')}"
472 when 'STATUS'
473 puts "STATUS stations #{@stations.length} lines #{@lines.length} walks #{@walks.values.sum(&:length) / 2}"
474 @stations.keys.sort.each { |id| puts "CLOSED #{id}" if @stations[id].closed }
475 @closed.to_a.sort.each { |x, y| puts "CLOSED #{x}-#{y}" }
476 @lines.keys.sort.each { |id| puts "SUSPENDED #{id}" if @lines[id].suspended }
477 @lines.keys.sort.each { |id| puts "DELAYED #{id} #{@lines[id].delay}" unless @lines[id].delay.zero? }
478 end
479 rescue MetroError => error
480 puts "ERROR #{error.message}"
481 end
482end
