1struct CommandError <: Exception
2 message::String
3end
4fail(s) = throw(CommandError(s))
5check(b, s) = b ? nothing : fail(s)
6parsetime(s) = 60*parse(Int, s[1:2])+parse(Int, s[4:5])
7clocktext(t) = lpad(string(div(t, 60)), 2, '0')*":"*lpad(string(t%60), 2, '0')
8halfprice(n) = cld(n, 20)*10
9segmentkey(a, b) = minmax(a, b)
10mutable struct Station
11 id::String
12 zone::Int
13 name::String
14 transfer::Int
15 closed::Bool
16 walks::Dict{String,Int}
17end
18Station(id, zone, name) = Station(id, zone, name, 0, false, Dict{String,Int}())
19mutable struct TransitLine
20 id::String
21 first::Int
22 last::Int
23 headway::Int
24 stops::Vector{String}
25 offsets::Vector{Int}
26 delay::Int
27 suspended::Bool
28end
29trips(line) = div(line.last-line.first, line.headway)+1
30struct Direction
31 line::TransitLine
32 stops::Vector{String}
33 offsets::Vector{Int}
34end
35directions(line) = (
36 Direction(line, line.stops, line.offsets),
37 Direction(line, reverse(line.stops), last(line.offsets) .- reverse(line.offsets)),
38)
39traintime(d, q, j) = d.line.first+d.line.delay+q*d.line.headway+d.offsets[j]
40function firsttrain(d, j, ready)
41 q=max(0, cld(ready-traintime(d, 0, j), d.line.headway))
42 q<trips(d.line) ? q : -1
43end
44function upcoming(d, j, time, count)
45 q=firsttrain(d, j, time)
46 q<0 ? Int[] : collect(q:(min(q+count, trips(d.line))-1))
47end
48abstract type Leg end
49struct Ride <: Leg
50 line::String
51 terminal::String
52 board::String
53 start::Int
54 alight::String
55 finish::Int
56 stops::Int
57end
58struct Walk <: Leg
59 origin::String
60 start::Int
61 destination::String
62 finish::Int
63end
64struct Stay <: Leg
65 station::String
66 start::Int
67 finish::Int
68end
69ride(d, q, i, j) = Ride(
70 d.line.id,
71 last(d.stops),
72 d.stops[i],
73 traintime(d, q, i),
74 d.stops[j],
75 traintime(d, q, j),
76 j-i,
77)

80 journey::Vector{Leg}

82const Labels=Dict{String,Label}

84 stations::Dict{String,Station}
85 lines::Dict{String,TransitLine}

87 tariff::Union{Nothing,Tuple{Int,Int,Vector{Int}}}
88end

95function station(p, name)
96 check(haskey(p.stations, name), "unknown station $name")
97 p.stations[name]
98end
99function transitline(p, name)
100 check(haskey(p.lines, name), "unknown line $name")
101 p.lines[name]
102end
103checkopen(p, name) = check(!p.stations[name].closed, "station $name is closed")
104function endpoints(p, origin, destination)
105 station(p, origin)
106 station(p, destination)
107 check(origin!=destination, "origin and destination are the same")
108 checkopen(p, origin)
109 checkopen(p, destination)
110end
111function toward(line, terminal)
112 check(
113 terminal in (first(line.stops), last(line.stops)),
114 "$terminal is not a terminal of line $(line.id)",
115 )
116 directions(line)[terminal==last(line.stops) ? 1 : 2]
117end
118orderedlines(p; active = false) =
119 [p.lines[id] for id in sort!(collect(keys(p.lines))) if !active || !p.lines[id].suspended]
120blocked(p, a, b) = segmentkey(a, b) in p.closed
121improves(labels, name, time) = !haskey(labels, name) || labels[name].time>time
122readytime(p, origin, name, time) = time+(origin==name ? 0 : p.stations[name].transfer)
123function walkfrom!(p, labels, name, time, journey)
124 changed=Set{String}()
125 for neighbor in sort!(collect(keys(p.stations[name].walks)))
126 arrival=time+p.stations[name].walks[neighbor]
127 if !p.stations[neighbor].closed && arrival<=1440 && improves(labels, neighbor, arrival)
128 labels[neighbor]=Label(arrival, [journey; Leg[Walk(name, time, neighbor, arrival)]])
129 push!(changed, neighbor)
130 end
131 end
132 changed
133end
134function ridefrom!(p, labels, d, name, ready, journey)
135 i=findfirst(==(name), d.stops)
136 (i===nothing || i==length(d.stops)) && return
137 q=firsttrain(d, i, ready)
138 q<0 && return
139 for j in (i+1):length(d.stops)
140 stop=d.stops[j]
141 blocked(p, d.stops[j-1], stop) && break
142 p.stations[stop].closed && continue
143 time=traintime(d, q, j)
144 improves(labels, stop, time) && (labels[stop]=Label(time, [journey; Leg[ride(d, q, i, j)]]))
145 end
146end
147function earliest(p, origin, departure)
148 lines=orderedlines(p; active = true)
149 labels=Labels(origin=>Label(departure, Leg[]))
150 settled=Set{String}()
151 while true
152 pending=[(value.time, name) for (name, value) in labels if !(name in settled)]
153 isempty(pending) && return labels
154 time, name=minimum(pending)
155 push!(settled, name)
156 journey=labels[name].journey
157 ready=readytime(p, origin, name, time)
158 for line in lines, d in directions(line)
159 ridefrom!(p, labels, d, name, ready, journey)
160 end
161 walkfrom!(p, labels, name, time, journey)
162 end
163end
164function walkstep!(p, labels, starts, marked)
165 recorded=Labels(name=>labels[name] for name in starts)
166 for name in sort!(collect(starts))
167 value=recorded[name]
168 union!(marked, walkfrom!(p, labels, name, value.time, value.journey))
169 end
170end
171function rideround!(p, origin, d, previous, previousmarked, labels, ridden)
172 boarded=-1
173 board=0
174 journey=Leg[]
175 for j in eachindex(d.stops)
176 name=d.stops[j]
177 j>1 && blocked(p, d.stops[j-1], name) && (boarded=-1)
178 p.stations[name].closed && continue
179 if boarded>=0
180 time=traintime(d, boarded, j)
181 if improves(labels, name, time)
182 labels[name]=Label(time, [journey; Leg[ride(d, boarded, board, j)]])
183 push!(ridden, name)
184 end
185 end
186 if j<length(d.stops) && name in previousmarked
187 value=previous[name]
188 q=firsttrain(d, j, readytime(p, origin, name, value.time))
189 if q>=0 && (boarded<0 || q<boarded)
190 boarded=q
191 board=j
192 journey=value.journey
193 end
194 end
195 end
196end
197function roundsearch(p, origin, departure)
198 labels=Labels(origin=>Label(departure, Leg[]))
199 marked=Set([origin])
200 walkstep!(p, labels, Set([origin]), marked)
201 rounds=[labels]
202 lines=orderedlines(p; active = true)
203 while !isempty(marked)
204 previous=labels
205 previousmarked=marked
206 labels=copy(previous)
207 ridden=Set{String}()
208 for line in lines, d in directions(line)
209 rideround!(p, origin, d, previous, previousmarked, labels, ridden)
210 end
211 marked=copy(ridden)
212 walkstep!(p, labels, ridden, marked)
213 push!(rounds, labels)
214 end
215 rounds
216end
217function journeyreport(heading, departure, value)
218 journey=value.journey
219 arrival=value.time
220 rides=count(x->x isa Ride, journey)
221 walks=count(x->x isa Walk, journey)
222 println(
223 "$heading $(clocktext(departure)) -> $(clocktext(arrival)) total $(arrival-departure) rides $rides walks $walks",
224 )
225 previous=departure
226 for leg in journey
227 if leg isa Ride
228 println(
229 "RIDE $(leg.line) toward $(leg.terminal) $(leg.board) $(clocktext(leg.start)) -> $(leg.alight) $(clocktext(leg.finish)) wait $(leg.start-previous) stops $(leg.stops)",
230 )
231 elseif leg isa Walk
232 println(
233 "WALK $(leg.origin) $(clocktext(leg.start)) -> $(leg.destination) $(clocktext(leg.finish))",
234 )
235 else
236 println("STAY $(leg.station) $(clocktext(leg.start)) -> $(clocktext(leg.finish))")
237 end
238 previous=leg.finish
239 end
240end
241function stopflags(p, stops, j)
242 (p.stations[stops[j]].closed ? " closed" : "")*(
243 j>1 && blocked(p, stops[j-1], stops[j]) ? " blocked" : ""
244 )
245end
246function fareitems(p, journey, child)
247 cap, window, prices=p.tariff
248 child && (cap=halfprice(cap))
249 previous=nothing
250 items=Tuple{Ride,Int,Int,Int}[]
251 for leg in journey
252 leg isa Ride || continue
253 d=toward(p.lines[leg.line], leg.terminal)
254 at=findfirst(==(leg.board), d.stops)
255 zones=length(Set(p.stations[name].zone for name in d.stops[at:(at+leg.stops)]))
256 price=prices[min(zones, length(prices))]
257 child && (price=halfprice(price))
258 discount=previous!==nothing && leg.start-previous.finish<=window ? min(cap, price) : 0
259 push!(items, (leg, zones, price, discount))
260 previous=leg
261 end
262 items
263end
264function passing(p, name, time, count, arriving)
265 result=Tuple{Int,String,Int,Direction}[]
266 for line in orderedlines(p; active = true)
267 at=findfirst(==(name), line.stops)
268 at===nothing && continue
269 for (order, d) in enumerate(directions(line))
270 j=findfirst(==(name), d.stops)
271 j==(arriving ? 1 : length(d.stops)) && continue
272 for q in upcoming(d, j, time, count)
273 push!(result, (traintime(d, q, j), line.id, order, d))
274 end
275 end
276 end
277 sort!(result; by = x->x[1:3])
278 result[1:min(count, length(result))]
279end
280function execute!(p, line)
281 words=String.(split(line, ' '))
282 command=words[1]
283 a=words[2:end]
284 if command=="STATION"
285 id=a[1]
286 check(!haskey(p.stations, id), "station $id already exists")
287 p.stations[id]=Station(id, parse(Int, a[2]), a[3])
288 println("STATION $id zone $(a[2]) $(a[3])")
289 elseif command=="LINE"
290 id=a[1]
291 check(!haskey(p.lines, id), "line $id already exists")
292 stops=a[5:2:end]
293 for name in stops
294 station(p, name)
295 end
296 seen=Set{String}()
297 for name in stops
298 check(!(name in seen), "station $name appears twice on line $id")
299 push!(seen, name)
300 end
301 firsttime=parsetime(a[2])
302 lasttime=parsetime(a[3])
303 check(firsttime<=lasttime, "line $id first departure is after last departure")
304 offsets=[0; cumsum(parse.(Int, a[6:2:end]))]
305 check(lasttime+last(offsets)<=1440, "line $id runs past 24:00")
306 value=TransitLine(id, firsttime, lasttime, parse(Int, a[4]), stops, offsets, 0, false)
307 p.lines[id]=value
308 println("LINE $id stops $(length(stops)) time $(last(offsets)) trips $(trips(value))")
309 elseif command=="TRANSFER"
310 value=station(p, a[1])
311 value.transfer=parse(Int, a[2])
312 println("TRANSFER $(a[1]) $(a[2])")
313 elseif command=="WALK"
314 firststation=station(p, a[1])
315 secondstation=station(p, a[2])
316 check(a[1]!=a[2], "walk from $(a[1]) to itself")
317 minutes=parse(Int, a[3])
318 old=get(firststation.walks, a[2], nothing)
319 firststation.walks[a[2]]=minutes
320 secondstation.walks[a[1]]=minutes
321 println("WALK $(a[1]) $(a[2]) $minutes", old===nothing ? "" : " was $old")
322 elseif command=="TARIFF"
323 tariffvalues=parse.(Int, a)
324 prices=tariffvalues[3:end]
325 check(
326 all(i->prices[i]<=prices[i+1], 1:(length(prices)-1)),
327 "tariff prices must not decrease",
328 )
329 p.tariff=(tariffvalues[1], tariffvalues[2], prices)
330 println(
331 "TARIFF zones $(length(prices)) discount $(tariffvalues[1]) window $(tariffvalues[2])",
332 )
333 elseif command in ("CLOSE", "OPEN")
334 firststation=station(p, a[1])
335 closing=command=="CLOSE"
336 if length(a)==1
337 check(
338 firststation.closed!=closing,
339 "station $(a[1]) is "*(closing ? "already closed" : "not closed"),
340 )
341 firststation.closed=closing
342 println(command, " ", a[1])
343 else
344 station(p, a[2])
345 key=segmentkey(a[1], a[2])
346 label="$(a[1])-$(a[2])"
347 if closing
348 lines=[
349 value.id for value in orderedlines(p) if any(
350 i->segmentkey(value.stops[i], value.stops[i+1])==key,
351 1:(length(value.stops)-1),
352 )
353 ]
354 check(!isempty(lines), "no segment $label")
355 check(!(key in p.closed), "segment $label is already closed")
356 push!(p.closed, key)
357 println("CLOSE $label lines ", join(lines, ' '))
358 else
359 check(key in p.closed, "segment $label is not closed")
360 delete!(p.closed, key)
361 println("OPEN $label")
362 end
363 end
364 elseif command in ("DELAY", "SUSPEND", "RESUME")
365 value=transitline(p, a[1])
366 if command=="DELAY"
367 minutes=parse(Int, a[2])
368 check(value.last+minutes+last(value.offsets)<=1440, "line $(a[1]) runs past 24:00")
369 value.delay=minutes
370 println("DELAY $(a[1]) $minutes")
371 else
372 suspend=command=="SUSPEND"
373 check(
374 value.suspended!=suspend,
375 "line $(a[1]) is "*(suspend ? "already suspended" : "not suspended"),
376 )
377 value.suspended=suspend
378 println(command, " ", a[1])
379 end
380 elseif command=="VIA"
381 origin, via, destination, time=a
382 station(p, origin)
383 station(p, via)
384 station(p, destination)
385 check(origin!=destination, "origin and destination are the same")
386 check(
387 !(via in (origin, destination)),
388 "via station $via must differ from origin and destination",
389 )
390 for name in (origin, via, destination)
391 checkopen(p, name)
392 end
393 departure=parsetime(time)
394 heading="VIA $origin $via $destination"
395 firstlabels=earliest(p, origin, departure)
396 if haskey(firstlabels, via)
397 firstvalue=firstlabels[via]
398 resume=firstvalue.time+p.stations[via].transfer
399 secondlabels=earliest(p, via, resume)
400 if haskey(secondlabels, destination)
401 value=secondlabels[destination]
402 journey=[firstvalue.journey; Leg[Stay(via, firstvalue.time, resume)]; value.journey]
403 journeyreport(heading, departure, Label(value.time, journey))
404 return
405 end
406 end
407 println("$heading $time unreachable")
408 elseif command in ("ROUTE", "FEWEST", "PARETO", "FARE", "LASTTRAIN", "ARRIVEBY", "PROFILE")
409 origin=a[1]
410 destination=a[2]
411 endpoints(p, origin, destination)
412 if command=="FARE"
413 check(a[4] in ("ADULT", "CHILD"), "unknown fare class $(a[4])")
414 check(p.tariff!==nothing, "no tariff")
415 end
416 if command in ("LASTTRAIN", "ARRIVEBY")
417 deadline=command=="LASTTRAIN" ? 1440 : parsetime(a[3])
418 heading=command=="LASTTRAIN" ? "LASTTRAIN $origin $destination" :
419 "ARRIVEBY $origin $destination by $(a[3])"
420 for departure in (deadline-1):-1:0
421 labels=earliest(p, origin, departure)
422 if haskey(labels, destination) && labels[destination].time<=deadline
423 journeyreport(heading, departure, labels[destination])
424 return
425 end
426 end
427 println("$heading none")
428 return
429 end
430 departure=parsetime(a[3])
431 heading="$command $origin $destination"
432 if command=="PROFILE"
433 finish=parsetime(a[4])
434 check(departure<=finish, "empty interval $(a[3]) $(a[4])")
435 arrivals=Dict{Int,Int}()
436 for time in departure:finish
437 labels=earliest(p, origin, time)
438 haskey(labels, destination) && (arrivals[time]=labels[destination].time)
439 end
440 selected=[
441 t for t in departure:finish if haskey(arrivals, t) &&
442 (t==finish || !haskey(arrivals, t+1) || arrivals[t]<arrivals[t+1])
443 ]
444 println("$heading $(a[3]) $(a[4]) count $(length(selected))")
445 for t in selected
446 println("$(clocktext(t)) -> $(clocktext(arrivals[t])) total $(arrivals[t]-t)")
447 end
448 elseif command in ("FEWEST", "PARETO")
449 rounds=roundsearch(p, origin, departure)
450 options=Tuple{Int,Label}[]
451 previous=nothing
452 for (i, labels) in enumerate(rounds)
453 if haskey(labels, destination)
454 value=labels[destination]
455 if previous===nothing || value.time<previous
456 push!(options, (i-1, value))
457 end
458 previous=value.time
459 end
460 end
461 if command=="PARETO"
462 println("$heading $(a[3]) options $(length(options))")
463 for (rides, value) in options
464 journeyreport("OPTION trips $rides", departure, value)
465 end
466 elseif isempty(options)
467 println("$heading $(a[3]) unreachable")
468 else
469 rides, value=first(options)
470 journeyreport("$heading trips $rides", departure, value)
471 end
472 else
473 labels=earliest(p, origin, departure)
474 if !haskey(labels, destination)
475 println("$heading $(a[3])", command=="FARE" ? " $(a[4])" : "", " unreachable")
476 return
477 end
478 value=labels[destination]
479 if command=="ROUTE"
480 journeyreport(heading, departure, value)
481 else
482 items=fareitems(p, value.journey, a[4]=="CHILD")
483 total=sum(x->x[3]-x[4], items; init = 0)
484 println("$heading $(a[3]) $(a[4]) rides $(length(items)) total $total")
485 for (leg, zones, price, discount) in items
486 println(
487 "TICKET $(leg.line) $(leg.board) $(leg.alight) zones $zones price $price discount $discount pay $(price-discount)",
488 )
489 end
490 end
491 end
492 elseif command=="REACH"
493 name=a[1]
494 station(p, name)
495 checkopen(p, name)
496 departure=parsetime(a[2])
497 labels=earliest(p, name, departure)
498 limit=departure+parse(Int, a[3])
499 selected=sort!([(value.time, id) for (id, value) in labels if value.time<=limit])
500 println("REACH $name $(a[2]) $(a[3]) count $(length(selected))")
501 for (arrival, id) in selected
502 println(
503 "$id $(p.stations[id].name) $(clocktext(arrival)) $(arrival-departure) rides $(count(x->x isa Ride,labels[id].journey))",
504 )
505 end
506 elseif command=="TIMETABLE"
507 name, id, terminal, time, countword=a
508 station(p, name)
509 value=transitline(p, id)
510 check(name in value.stops, "station $name is not on line $id")
511 d=toward(value, terminal)
512 check(name!=terminal, "no departures from $name toward $terminal")
513 check(!value.suspended, "line $id is suspended")
514 checkopen(p, name)
515 j=findfirst(==(name), d.stops)
516 times=[
517 clocktext(traintime(d, q, j)) for
518 q in upcoming(d, j, parsetime(time), parse(Int, countword))
519 ]
520 println(
521 "TIMETABLE $name $id toward $terminal $time next $(length(times))",
522 isempty(times) ? "" : " "*join(times, ' '),
523 )
524 elseif command in ("DEPARTURES", "ARRIVALS")
525 name=a[1]
526 station(p, name)
527 checkopen(p, name)
528 arriving=command=="ARRIVALS"
529 trains=passing(p, name, parsetime(a[2]), parse(Int, a[3]), arriving)
530 println("$command $name $(a[2]) count $(length(trains))")
531 for (time, id, _, d) in trains
532 println(
533 "$(clocktext(time)) $id ",
534 arriving ? "from $(first(d.stops))" : "toward $(last(d.stops))",
535 )
536 end
537 elseif command in ("TRIP", "SHOWLINE")
538 value=transitline(p, a[1])
539 if command=="SHOWLINE"
540 println(
541 "SHOWLINE $(value.id) $(value.suspended ? "suspended" : "running") first $(clocktext(value.first)) last $(clocktext(value.last)) headway $(value.headway) delay $(value.delay) trips $(trips(value))",
542 )
543 for (j, name) in enumerate(value.stops)
544 println(
545 "$(value.offsets[j]) $name zone $(p.stations[name].zone)",
546 stopflags(p, value.stops, j),
547 )
548 end
549 else
550 d=toward(value, a[2])
551 check(!value.suspended, "line $(value.id) is suspended")
552 q=firsttrain(d, 1, parsetime(a[3]))
553 heading="TRIP $(value.id) toward $(a[2]) $(a[3])"
554 if q<0
555 println("$heading none")
556 else
557 println("$heading trip $(q+1) departs $(clocktext(traintime(d,q,1)))")
558 for (j, name) in enumerate(d.stops)
559 println("$(clocktext(traintime(d,q,j))) $name", stopflags(p, d.stops, j))
560 end
561 end
562 end
563 elseif command=="INFO"
564 value=station(p, a[1])
565 println(
566 "INFO $(value.id) $(value.name) zone $(value.zone) transfer $(value.transfer) $(value.closed ? "closed" : "open")",
567 )
568 lines=[line.id for line in orderedlines(p) if value.id in line.stops]
569 println("LINES ", isempty(lines) ? "none" : join(lines, ' '))
570 walks=["$name $(value.walks[name])" for name in sort!(collect(keys(value.walks)))]
571 println("WALKS ", isempty(walks) ? "none" : join(walks, ' '))
572 elseif command=="STATUS"
573 walks=div(sum(x->length(x.walks), values(p.stations); init = 0), 2)
574 println("STATUS stations $(length(p.stations)) lines $(length(p.lines)) walks $walks")
575 for id in sort!(collect(keys(p.stations)))
576 p.stations[id].closed && println("CLOSED $id")
577 end
578 for (a, b) in sort!(collect(p.closed))
579 println("CLOSED $a-$b")
580 end
581 for line in orderedlines(p)
582 line.suspended && println("SUSPENDED $(line.id)")
583 end
584 for line in orderedlines(p)
585 line.delay!=0 && println("DELAYED $(line.id) $(line.delay)")
586 end
587 end
588end
589function main()
590 planner=Planner()
591 n=parse(Int, readline())
592 for _ in 1:n
593 line=readline()
594 try
595 execute!(planner, line)
596 catch e
597 e isa CommandError || rethrow()
598 println("ERROR $(e.message)")
599 end
600 end
601end
