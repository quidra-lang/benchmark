5const Text = []const u8;

7const Error = Allocator.Error || error{Invalid};
8var failure: Text = "";

10fn equal(a: Text, b: Text) bool {
11 return std.mem.eql(u8, a, b);
12}
13
14fn bad(gpa: Allocator, comptime pattern: Text, args: anytype) Error {
15 failure = std.fmt.allocPrint(gpa, pattern, args) catch return error.OutOfMemory;
16 return error.Invalid;
17}
18
19fn number(text: Text) usize {
20 return std.fmt.parseInt(usize, text, 10) catch unreachable;
21}
22
23fn timeValue(text: Text) usize {
24 return number(text[0..2]) * 60 + number(text[3..5]);
25}
26
27fn timeText(gpa: Allocator, value: usize) Allocator.Error!Text {
28 return std.fmt.allocPrint(gpa, "{d:0>2}:{d:0>2}", .{ value / 60, value % 60 });
29}

31const Builder = struct {
32 gpa: Allocator,
33 bytes: std.ArrayList(u8) = .empty,
34 fn add(self: *Builder, text: Text) Allocator.Error!void {
35 try self.bytes.appendSlice(self.gpa, text);
36 }
37 fn print(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
38 try self.add(try std.fmt.allocPrint(self.gpa, pattern, args));
39 }
40 fn line(self: *Builder, comptime pattern: Text, args: anytype) Allocator.Error!void {
41 try self.print(pattern ++ "\n", args);
42 }
43};
44
45fn textLess(_: void, a: Text, b: Text) bool {
46 return std.mem.order(u8, a, b) == .lt;
47}
48
49fn keys(gpa: Allocator, map: anytype) Allocator.Error![]Text {
50 var result: std.ArrayList(Text) = .empty;
51 var iterator = map.keyIterator();
52 while (iterator.next()) |key| try result.append(gpa, key.*);
53 std.mem.sort(Text, result.items, {}, textLess);
54 return result.items;
55}

57const Station = struct {
58 id: Text,
59 name: Text,
60 index: usize,
61 zone: usize,
62 transfer: usize = 0,
63 closed: bool = false,
64 walks: std.StringHashMapUnmanaged(usize) = .empty,
65};

67const Direction = struct { stops: []const *Station, offsets: []const usize };
68
69const Line = struct {
70 id: Text,
71 first: usize,
72 last: usize,
73 headway: usize,
74 total: usize,
75 stops: []const *Station,
76 cumulative: []const usize,
77 delay: usize = 0,
78 suspended: bool = false,
79 fn trains(self: *const Line) usize {
80 return (self.last - self.first) / self.headway + 1;
81 }
82 fn firstTrain(self: *const Line, offset: usize, ready: usize) ?usize {
83 const base = self.first + self.delay + offset;
84 const q = if (ready > base) (ready - base + self.headway - 1) / self.headway else 0;
85 return if (q < self.trains()) q else null;
86 }
87 fn time(self: *const Line, q: usize, offset: usize) usize {
88 return self.first + self.delay + q * self.headway + offset;
89 }
90 fn direction(self: *const Line, gpa: Allocator, which: usize) Allocator.Error!Direction {
91 if (which == 0) return .{ .stops = self.stops, .offsets = self.cumulative };
92 const stops = try gpa.alloc(*Station, self.stops.len);
93 const offsets = try gpa.alloc(usize, self.stops.len);
94 for (stops, 0..) |*stop, i| {
95 const j = stops.len - 1 - i;
96 stop.* = self.stops[j];
97 offsets[i] = self.total - self.cumulative[j];
98 }
99 return .{ .stops = stops, .offsets = offsets };
100 }
101};

124fn indexOf(stops: []const *Station, station: *Station) ?usize {
125 for (stops, 0..) |candidate, i| if (candidate == station) return i;
126 return null;
127}

129fn extend(gpa: Allocator, journey: []const Leg, leg: Leg) Allocator.Error![]const Leg {
130 const result = try gpa.alloc(Leg, journey.len + 1);
131 @memcpy(result[0..journey.len], journey);
132 result[journey.len] = leg;
133 return result;
134}

136fn improves(labels: Labels, target: *Station, arrival: usize) bool {
137 return labels[target.index] == null or labels[target.index].?.arrival > arrival;
138}

140fn ride(service: *Line, d: Direction, a: usize, b: usize, depart: usize, arrive: usize) Leg {
141 return .{ .kind = .ride, .service = service, .terminal = d.stops[d.stops.len - 1], .from = d.stops[a], .to = d.stops[b], .depart = depart, .arrive = arrive, .stops = b - a };
142}

144fn walk(a: *Station, b: *Station, depart: usize, arrive: usize) Leg {
145 return .{ .kind = .walk, .from = a, .to = b, .depart = depart, .arrive = arrive };

148fn counts(journey: []const Leg) struct { rides: usize, walks: usize } {
149 var rides: usize = 0;
150 var walks: usize = 0;
151 for (journey) |leg| {
152 if (leg.kind == .ride) rides += 1 else if (leg.kind == .walk) walks += 1;
153 }
154 return .{ .rides = rides, .walks = walks };
155}

159const Network = struct {
160 gpa: Allocator,
161 out: *Builder,
162 stations: std.StringHashMapUnmanaged(*Station) = .empty,
163 lines: std.StringHashMapUnmanaged(*Line) = .empty,
164 closed: std.AutoHashMapUnmanaged(usize, void) = .empty,
165 walk_count: usize = 0,
166 tariff: ?Tariff = null,
167 fn pair(a: *Station, b: *Station) usize {
168 return @min(a.index, b.index) * 30 + @max(a.index, b.index);
169 }
170 fn blocked(self: *const Network, a: *Station, b: *Station) bool {
171 return self.closed.contains(pair(a, b));
172 }
173 fn getStation(self: *Network, id: Text) Error!*Station {
174 return self.stations.get(id) orelse return bad(self.gpa, "unknown station {s}", .{id});
175 }
176 fn getLine(self: *Network, id: Text) Error!*Line {
177 return self.lines.get(id) orelse return bad(self.gpa, "unknown line {s}", .{id});
178 }
179 fn common(self: *Network, a: Text, b: Text) Error!struct { origin: *Station, destination: *Station } {
180 const origin = try self.getStation(a);
181 const destination = try self.getStation(b);
182 if (origin == destination) return bad(self.gpa, "origin and destination are the same", .{});
183 if (origin.closed) return bad(self.gpa, "station {s} is closed", .{a});
184 if (destination.closed) return bad(self.gpa, "station {s} is closed", .{b});
185 return .{ .origin = origin, .destination = destination };
186 }
187 fn earliest(self: *Network, gpa: Allocator, origin: *Station, start: usize) Allocator.Error!Labels {
188 const labels = try gpa.alloc(?Label, self.stations.count());
189 @memset(labels, null);
190 labels[origin.index] = .{ .arrival = start, .journey = &.{} };
191 const settled = try gpa.alloc(bool, labels.len);
192 @memset(settled, false);
193 const station_ids = try keys(gpa, self.stations);
194 const service_ids = try keys(gpa, self.lines);
195 while (true) {
196 var selected: ?*Station = null;
197 for (station_ids) |id| {
198 const station = self.stations.get(id).?;
199 if (!settled[station.index]) if (labels[station.index]) |value| {
200 if (selected == null or value.arrival < labels[selected.?.index].?.arrival) selected = station;
201 };
202 }
203 const source = selected orelse return labels;
204 settled[source.index] = true;
205 const current = labels[source.index].?;
206 const ready = current.arrival + if (source == origin) @as(usize, 0) else source.transfer;
207 for (service_ids) |id| {
208 const service = self.lines.get(id).?;
209 if (service.suspended) continue;
210 for (0..2) |which| {
211 const d = try service.direction(gpa, which);
212 const i = indexOf(d.stops, source) orelse continue;
213 if (i + 1 == d.stops.len) continue;
214 const q = service.firstTrain(d.offsets[i], ready) orelse continue;
215 const depart = service.time(q, d.offsets[i]);
216 for (i + 1..d.stops.len) |j| {
217 const target = d.stops[j];
218 if (self.blocked(d.stops[j - 1], target)) break;
219 if (target.closed) continue;
220 const arrival = service.time(q, d.offsets[j]);
221 if (improves(labels, target, arrival)) labels[target.index] = .{
222 .arrival = arrival,
223 .journey = try extend(gpa, current.journey, ride(service, d, i, j, depart, arrival)),
224 };
225 }
226 }
227 }
228 for (try keys(gpa, source.walks)) |id| {
229 const target = self.stations.get(id).?;
230 const arrival = current.arrival + source.walks.get(id).?;
231 if (!target.closed and arrival <= 1440 and improves(labels, target, arrival)) labels[target.index] = .{
232 .arrival = arrival,
233 .journey = try extend(gpa, current.journey, walk(source, target, current.arrival, arrival)),
234 };
235 }
236 }
237 }
238 fn search(self: *Network, origin: *Station, start: usize) Allocator.Error!Search {
239 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
240 errdefer arena.deinit();
241 const labels = try self.earliest(arena.allocator(), origin, start);
242 return .{ .arena = arena, .labels = labels };
243 }
244 fn walkStep(self: *Network, gpa: Allocator, labels: Labels, sources: []const bool, marked: []bool) Allocator.Error!void {
245 const recorded = try gpa.dupe(?Label, labels);
246 for (try keys(gpa, self.stations)) |id| {
247 const source = self.stations.get(id).?;
248 if (!sources[source.index]) continue;
249 const current = recorded[source.index].?;
250 for (try keys(gpa, source.walks)) |neighbor| {
251 const target = self.stations.get(neighbor).?;
252 const arrival = current.arrival + source.walks.get(neighbor).?;
253 if (!target.closed and arrival <= 1440 and improves(labels, target, arrival)) {
254 labels[target.index] = .{ .arrival = arrival, .journey = try extend(gpa, current.journey, walk(source, target, current.arrival, arrival)) };
255 marked[target.index] = true;
256 }
257 }
258 }
259 }
260 fn rounds(self: *Network, gpa: Allocator, origin: *Station, start: usize, stop: ?*Station) Allocator.Error![]Labels {
261 var current = try gpa.alloc(?Label, self.stations.count());
262 @memset(current, null);
263 current[origin.index] = .{ .arrival = start, .journey = &.{} };
264 var marked = try gpa.alloc(bool, current.len);
265 @memset(marked, false);
266 marked[origin.index] = true;
267 const sources = try gpa.dupe(bool, marked);
268 try self.walkStep(gpa, current, sources, marked);
269 var result: std.ArrayList(Labels) = .empty;
270 try result.append(gpa, current);
271 const service_ids = try keys(gpa, self.lines);
272 while (std.mem.findScalar(bool, marked, true) != null) {
273 if (stop) |target| {
274 if (current[target.index] != null) break;
275 }
276 const following = try gpa.dupe(?Label, current);
277 const updated = try gpa.alloc(bool, current.len);
278 const walking = try gpa.alloc(bool, current.len);
279 @memset(updated, false);
280 @memset(walking, false);
281 for (service_ids) |id| {
282 const service = self.lines.get(id).?;
283 if (service.suspended) continue;
284 for (0..2) |which| {
285 const d = try service.direction(gpa, which);
286 var boarded: ?usize = null;
287 var boarding: usize = 0;
288 var depart: usize = 0;
289 var prefix: []const Leg = &.{};
290 for (d.stops, 0..) |target, j| {
291 if (j > 0 and self.blocked(d.stops[j - 1], target)) boarded = null;
292 if (target.closed) continue;
293 if (boarded) |q| {
294 const arrival = service.time(q, d.offsets[j]);
295 if (improves(following, target, arrival)) {
296 following[target.index] = .{ .arrival = arrival, .journey = try extend(gpa, prefix, ride(service, d, boarding, j, depart, arrival)) };
297 walking[target.index] = true;
298 updated[target.index] = true;
299 }
300 }
301 if (j + 1 < d.stops.len and marked[target.index]) {
302 const previous = current[target.index].?;
303 const ready = previous.arrival + if (target == origin) @as(usize, 0) else target.transfer;
304 if (service.firstTrain(d.offsets[j], ready)) |candidate| {
305 if (boarded == null or candidate < boarded.?) {
306 boarded = candidate;
307 boarding = j;
308 depart = service.time(candidate, d.offsets[j]);
309 prefix = previous.journey;
310 }
311 }
312 }
313 }
314 }
315 }
316 try self.walkStep(gpa, following, walking, updated);
317 try result.append(gpa, following);
318 current = following;
319 marked = updated;
320 }
321 return result.items;
322 }
323 fn printJourney(self: *Network, scratch: Allocator, heading: Text, start: usize, result: Label) Allocator.Error!void {
324 const count = counts(result.journey);
325 try self.out.line("{s} {s} -> {s} total {d} rides {d} walks {d}", .{ heading, try timeText(scratch, start), try timeText(scratch, result.arrival), result.arrival - start, count.rides, count.walks });
326 var previous = start;
327 for (result.journey) |leg| {
328 switch (leg.kind) {
329 .ride => try self.out.line("RIDE {s} toward {s} {s} {s} -> {s} {s} wait {d} stops {d}", .{ leg.service.?.id, leg.terminal.?.id, leg.from.id, try timeText(scratch, leg.depart), leg.to.id, try timeText(scratch, leg.arrive), leg.depart - previous, leg.stops }),
330 .walk => try self.out.line("WALK {s} {s} -> {s} {s}", .{ leg.from.id, try timeText(scratch, leg.depart), leg.to.id, try timeText(scratch, leg.arrive) }),
331 .stay => try self.out.line("STAY {s} {s} -> {s}", .{ leg.from.id, try timeText(scratch, leg.depart), try timeText(scratch, leg.arrive) }),
332 }
333 previous = leg.arrive;
334 }
335 }
336 fn segmentLines(self: *Network, gpa: Allocator, a: *Station, b: *Station) Allocator.Error![]Text {
337 var result: std.ArrayList(Text) = .empty;
338 for (try keys(gpa, self.lines)) |id| {
339 const service = self.lines.get(id).?;
340 for (1..service.stops.len) |i| {
341 const x = service.stops[i - 1];
342 const y = service.stops[i];
343 if ((x == a and y == b) or (x == b and y == a)) {
344 try result.append(gpa, id);
345 break;
346 }
347 }
348 }
349 return result.items;
350 }
351 fn half(value: usize) usize {
352 return (value + 19) / 20 * 10;
353 }
354 fn printFare(self: *Network, scratch: Allocator, origin: Text, destination: Text, start: usize, class: Text, result: Label) Allocator.Error!void {
355 const Ticket = struct { leg: Leg, zones: usize, price: usize, discount: usize };
356 var tickets: std.ArrayList(Ticket) = .empty;
357 var total: usize = 0;
358 var previous: ?usize = null;
359 const tariff = self.tariff.?;
360 for (result.journey) |leg| {
361 if (leg.kind != .ride) continue;
362 const service = leg.service.?;
363 const a = indexOf(service.stops, leg.from).?;
364 const b = indexOf(service.stops, leg.to).?;
365 var zones: [10]bool = @splat(false);
366 var zone_count: usize = 0;
367 for (service.stops[@min(a, b) .. @max(a, b) + 1]) |station| {
368 if (!zones[station.zone]) {
369 zones[station.zone] = true;
370 zone_count += 1;
371 }
372 }
373 var price = tariff.prices[@min(zone_count, tariff.prices.len) - 1];
374 var cap = tariff.discount;
375 if (equal(class, "CHILD")) {
376 price = half(price);
377 cap = half(cap);
378 }
379 const discount = if (previous != null and leg.depart - previous.? <= tariff.window) @min(cap, price) else 0;
380 total += price - discount;
381 try tickets.append(scratch, .{ .leg = leg, .zones = zone_count, .price = price, .discount = discount });
382 previous = leg.arrive;
383 }
384 try self.out.line("FARE {s} {s} {s} {s} rides {d} total {d}", .{ origin, destination, try timeText(scratch, start), class, tickets.items.len, total });
385 for (tickets.items) |ticket| try self.out.line("TICKET {s} {s} {s} zones {d} price {d} discount {d} pay {d}", .{ ticket.leg.service.?.id, ticket.leg.from.id, ticket.leg.to.id, ticket.zones, ticket.price, ticket.discount, ticket.price - ticket.discount });
386 }
387 fn command(self: *Network, scratch: Allocator, w: []const Text) Error!void {
388 const cmd = w[0];
389 if (equal(cmd, "STATION")) {
390 if (self.stations.contains(w[1])) return bad(self.gpa, "station {s} already exists", .{w[1]});
391 const station = try self.gpa.create(Station);
392 station.* = .{ .id = w[1], .zone = number(w[2]), .name = w[3], .index = self.stations.count() };
393 try self.stations.put(self.gpa, station.id, station);
394 try self.out.line("STATION {s} zone {s} {s}", .{ w[1], w[2], w[3] });
395 } else if (equal(cmd, "LINE")) {
396 if (self.lines.contains(w[1])) return bad(self.gpa, "line {s} already exists", .{w[1]});
397 var stops: std.ArrayList(*Station) = .empty;
398 var at: usize = 5;
399 while (at < w.len) : (at += 2) try stops.append(self.gpa, try self.getStation(w[at]));
400 for (stops.items, 0..) |station, i| {
401 if (indexOf(stops.items[0..i], station) != null) return bad(self.gpa, "station {s} appears twice on line {s}", .{ station.id, w[1] });
402 }
403 const first = timeValue(w[2]);
404 const last = timeValue(w[3]);
405 if (first > last) return bad(self.gpa, "line {s} first departure is after last departure", .{w[1]});
406 const cumulative = try self.gpa.alloc(usize, stops.items.len);
407 cumulative[0] = 0;
408 for (1..cumulative.len) |i| cumulative[i] = cumulative[i - 1] + number(w[4 + 2 * i]);
409 const total = cumulative[cumulative.len - 1];
410 if (last + total > 1440) return bad(self.gpa, "line {s} runs past 24:00", .{w[1]});
411 const service = try self.gpa.create(Line);
412 service.* = .{ .id = w[1], .first = first, .last = last, .headway = number(w[4]), .total = total, .stops = stops.items, .cumulative = cumulative };
413 try self.lines.put(self.gpa, service.id, service);
414 try self.out.line("LINE {s} stops {d} time {d} trips {d}", .{ service.id, service.stops.len, total, service.trains() });
415 } else if (equal(cmd, "TRANSFER")) {
416 const station = try self.getStation(w[1]);
417 station.transfer = number(w[2]);
418 try self.out.line("TRANSFER {s} {d}", .{ station.id, station.transfer });
419 } else if (equal(cmd, "WALK")) {
420 const a = try self.getStation(w[1]);
421 const b = try self.getStation(w[2]);
422 if (a == b) return bad(self.gpa, "walk from {s} to itself", .{a.id});
423 const previous = a.walks.get(b.id);
424 const minutes = number(w[3]);
425 try a.walks.put(self.gpa, b.id, minutes);
426 try b.walks.put(self.gpa, a.id, minutes);
427 try self.out.print("WALK {s} {s} {d}", .{ a.id, b.id, minutes });
428 if (previous) |old| try self.out.print(" was {d}", .{old}) else self.walk_count += 1;
429 try self.out.add("\n");
430 } else if (equal(cmd, "TARIFF")) {
431 const prices = try self.gpa.alloc(usize, w.len - 3);
432 for (w[3..], 0..) |word, i| {
433 prices[i] = number(word);
434 if (i > 0 and prices[i] < prices[i - 1]) return bad(self.gpa, "tariff prices must not decrease", .{});
435 }
436 self.tariff = .{ .discount = number(w[1]), .window = number(w[2]), .prices = prices };
437 try self.out.line("TARIFF zones {d} discount {s} window {s}", .{ prices.len, w[1], w[2] });
438 } else if (equal(cmd, "CLOSE") or equal(cmd, "OPEN")) {
439 const a = try self.getStation(w[1]);
440 const closing = equal(cmd, "CLOSE");
441 if (w.len == 2) {
442 if (closing and a.closed) return bad(self.gpa, "station {s} is already closed", .{a.id});
443 if (!closing and !a.closed) return bad(self.gpa, "station {s} is not closed", .{a.id});
444 a.closed = closing;
445 try self.out.line("{s} {s}", .{ cmd, a.id });
446 } else {
447 const b = try self.getStation(w[2]);
448 const key = pair(a, b);
449 if (closing) {
450 const services = try self.segmentLines(scratch, a, b);
451 if (services.len == 0) return bad(self.gpa, "no segment {s}-{s}", .{ a.id, b.id });
452 if (self.closed.contains(key)) return bad(self.gpa, "segment {s}-{s} is already closed", .{ a.id, b.id });
453 try self.closed.put(self.gpa, key, {});
454 try self.out.print("CLOSE {s}-{s} lines", .{ a.id, b.id });
455 for (services) |id| try self.out.print(" {s}", .{id});
456 try self.out.add("\n");
457 } else {
458 if (!self.closed.contains(key)) return bad(self.gpa, "segment {s}-{s} is not closed", .{ a.id, b.id });
459 _ = self.closed.remove(key);
460 try self.out.line("OPEN {s}-{s}", .{ a.id, b.id });
461 }
462 }
463 } else if (equal(cmd, "DELAY") or equal(cmd, "SUSPEND") or equal(cmd, "RESUME")) {
464 const service = try self.getLine(w[1]);
465 if (equal(cmd, "DELAY")) {
466 const minutes = number(w[2]);
467 if (service.last + minutes + service.total > 1440) return bad(self.gpa, "line {s} runs past 24:00", .{service.id});
468 service.delay = minutes;
469 try self.out.line("DELAY {s} {d}", .{ service.id, minutes });
470 } else {
471 const suspending = equal(cmd, "SUSPEND");
472 if (suspending and service.suspended) return bad(self.gpa, "line {s} is already suspended", .{service.id});
473 if (!suspending and !service.suspended) return bad(self.gpa, "line {s} is not suspended", .{service.id});
474 service.suspended = suspending;
475 try self.out.line("{s} {s}", .{ cmd, service.id });
476 }
477 } else if (equal(cmd, "ROUTE") or equal(cmd, "FEWEST") or equal(cmd, "PARETO") or equal(cmd, "FARE") or
478 equal(cmd, "LASTTRAIN") or equal(cmd, "ARRIVEBY") or equal(cmd, "PROFILE"))
479 {
480 const endpoints = try self.common(w[1], w[2]);
481 const origin = endpoints.origin;
482 const destination = endpoints.destination;
483 if (equal(cmd, "FARE")) {
484 if (!equal(w[4], "ADULT") and !equal(w[4], "CHILD")) return bad(self.gpa, "unknown fare class {s}", .{w[4]});
485 if (self.tariff == null) return bad(self.gpa, "no tariff", .{});
486 }
487 const start = if (w.len > 3) timeValue(w[3]) else 0;
488 if (equal(cmd, "PROFILE")) {
489 const finish = timeValue(w[4]);
490 if (start > finish) return bad(self.gpa, "empty interval {s} {s}", .{ w[3], w[4] });
491 const arrivals = try scratch.alloc(?usize, finish - start + 1);
492 for (arrivals, 0..) |*arrival, i| {
493 var found = try self.search(origin, start + i);
494 defer found.arena.deinit();
495 arrival.* = if (found.labels[destination.index]) |label| label.arrival else null;
496 }
497 var selected: std.ArrayList(usize) = .empty;
498 for (arrivals, 0..) |arrival, i| {
499 if (arrival) |value| {
500 if (i + 1 == arrivals.len or arrivals[i + 1] == null or value < arrivals[i + 1].?) try selected.append(scratch, i);
501 }
502 }
503 try self.out.line("PROFILE {s} {s} {s} {s} count {d}", .{ origin.id, destination.id, w[3], w[4], selected.items.len });
504 for (selected.items) |i| try self.out.line("{s} -> {s} total {d}", .{ try timeText(scratch, start + i), try timeText(scratch, arrivals[i].?), arrivals[i].? - start - i });
505 } else if (equal(cmd, "LASTTRAIN") or equal(cmd, "ARRIVEBY")) {
506 const last_train = equal(cmd, "LASTTRAIN");
507 const heading = if (last_train) try std.fmt.allocPrint(scratch, "LASTTRAIN {s} {s}", .{ origin.id, destination.id }) else try std.fmt.allocPrint(scratch, "ARRIVEBY {s} {s} by {s}", .{ origin.id, destination.id, w[3] });
508 var remaining: usize = if (last_train) 1440 else start;
509 while (remaining > 0) {
510 remaining -= 1;
511 var found = try self.search(origin, remaining);
512 defer found.arena.deinit();
513 if (found.labels[destination.index]) |label| {
514 if (last_train or label.arrival <= start) {
515 try self.printJourney(scratch, heading, remaining, label);
516 return;
517 }
518 }
519 }
520 try self.out.line("{s} none", .{heading});
521 } else if (equal(cmd, "FEWEST") or equal(cmd, "PARETO")) {
522 const fewest = equal(cmd, "FEWEST");
523 const results = try self.rounds(scratch, origin, start, if (fewest) destination else null);
524 if (fewest) {
525 for (results, 0..) |labels, k| {
526 if (labels[destination.index]) |label| {
527 const heading = try std.fmt.allocPrint(scratch, "FEWEST {s} {s} trips {d}", .{ origin.id, destination.id, k });
528 try self.printJourney(scratch, heading, start, label);
529 return;
530 }
531 }
532 try self.out.line("FEWEST {s} {s} {s} unreachable", .{ origin.id, destination.id, w[3] });
533 } else {
534 var options: std.ArrayList(usize) = .empty;
535 var previous: ?usize = null;
536 for (results, 0..) |labels, k| {
537 if (labels[destination.index]) |label| {
538 if (previous == null or label.arrival < previous.?) try options.append(scratch, k);
539 previous = label.arrival;
540 }
541 }
542 try self.out.line("PARETO {s} {s} {s} options {d}", .{ origin.id, destination.id, w[3], options.items.len });
543 for (options.items) |k| {
544 const heading = try std.fmt.allocPrint(scratch, "OPTION trips {d}", .{k});
545 try self.printJourney(scratch, heading, start, results[k][destination.index].?);
546 }
547 }
548 } else {
549 var found = try self.search(origin, start);
550 defer found.arena.deinit();
551 if (found.labels[destination.index]) |label| {
552 if (equal(cmd, "FARE")) try self.printFare(scratch, origin.id, destination.id, start, w[4], label) else try self.printJourney(scratch, try std.fmt.allocPrint(scratch, "ROUTE {s} {s}", .{ origin.id, destination.id }), start, label);
553 } else if (equal(cmd, "FARE")) try self.out.line("FARE {s} {s} {s} {s} unreachable", .{ origin.id, destination.id, w[3], w[4] }) else try self.out.line("ROUTE {s} {s} {s} unreachable", .{ origin.id, destination.id, w[3] });
554 }
555 } else if (equal(cmd, "VIA")) {
556 const origin = try self.getStation(w[1]);
557 const via = try self.getStation(w[2]);
558 const destination = try self.getStation(w[3]);
559 if (origin == destination) return bad(self.gpa, "origin and destination are the same", .{});
560 if (via == origin or via == destination) return bad(self.gpa, "via station {s} must differ from origin and destination", .{via.id});
561 for ([_]*Station{ origin, via, destination }) |station| {
562 if (station.closed) return bad(self.gpa, "station {s} is closed", .{station.id});
563 }
564 const start = timeValue(w[4]);
565 var first = try self.search(origin, start);
566 defer first.arena.deinit();
567 if (first.labels[via.index]) |first_label| {
568 const ready = first_label.arrival + via.transfer;
569 var second = try self.search(via, ready);
570 defer second.arena.deinit();
571 if (second.labels[destination.index]) |second_label| {
572 var journey: std.ArrayList(Leg) = .empty;
573 try journey.appendSlice(scratch, first_label.journey);
574 try journey.append(scratch, .{ .kind = .stay, .from = via, .to = via, .depart = first_label.arrival, .arrive = ready });
575 try journey.appendSlice(scratch, second_label.journey);
576 const heading = try std.fmt.allocPrint(scratch, "VIA {s} {s} {s}", .{ origin.id, via.id, destination.id });
577 try self.printJourney(scratch, heading, start, .{ .arrival = second_label.arrival, .journey = journey.items });
578 return;
579 }
580 }
581 try self.out.line("VIA {s} {s} {s} {s} unreachable", .{ origin.id, via.id, destination.id, w[4] });
582 } else if (equal(cmd, "REACH")) {
583 const origin = try self.getStation(w[1]);
584 if (origin.closed) return bad(self.gpa, "station {s} is closed", .{origin.id});
585 const start = timeValue(w[2]);
586 const duration = number(w[3]);
587 var found = try self.search(origin, start);
588 defer found.arena.deinit();
589 var selected: std.ArrayList(*Station) = .empty;
590 for (try keys(scratch, self.stations)) |id| {
591 const station = self.stations.get(id).?;
592 if (found.labels[station.index]) |label| {
593 if (label.arrival <= start + duration) try selected.append(scratch, station);
594 }
595 }
596 const Less = struct {
597 fn less(labels: Labels, a: *Station, b: *Station) bool {
598 const x = labels[a.index].?.arrival;
599 const y = labels[b.index].?.arrival;
600 return if (x != y) x < y else std.mem.order(u8, a.id, b.id) == .lt;
601 }
602 };
603 std.mem.sort(*Station, selected.items, found.labels, Less.less);
604 try self.out.line("REACH {s} {s} {d} count {d}", .{ origin.id, w[2], duration, selected.items.len });
605 for (selected.items) |station| {
606 const label = found.labels[station.index].?;
607 try self.out.line("{s} {s} {s} {d} rides {d}", .{ station.id, station.name, try timeText(scratch, label.arrival), label.arrival - start, counts(label.journey).rides });
608 }
609 } else if (equal(cmd, "TIMETABLE")) {
610 const station = try self.getStation(w[1]);
611 const service = try self.getLine(w[2]);
612 if (indexOf(service.stops, station) == null) return bad(self.gpa, "station {s} is not on line {s}", .{ station.id, service.id });
613 const first = service.stops[0];
614 const last = service.stops[service.stops.len - 1];
615 if (!equal(w[3], first.id) and !equal(w[3], last.id)) return bad(self.gpa, "{s} is not a terminal of line {s}", .{ w[3], service.id });
616 if (equal(station.id, w[3])) return bad(self.gpa, "no departures from {s} toward {s}", .{ station.id, w[3] });
617 if (service.suspended) return bad(self.gpa, "line {s} is suspended", .{service.id});
618 if (station.closed) return bad(self.gpa, "station {s} is closed", .{station.id});
619 const d = try service.direction(scratch, if (equal(w[3], first.id)) 1 else 0);
620 const at = indexOf(d.stops, station).?;
621 const q = service.firstTrain(d.offsets[at], timeValue(w[4]));
622 const count = if (q) |train| @min(number(w[5]), service.trains() - train) else 0;
623 try self.out.print("TIMETABLE {s} {s} toward {s} {s} next {d}", .{ station.id, service.id, w[3], w[4], count });
624 for (0..count) |i| try self.out.print(" {s}", .{try timeText(scratch, service.time(q.? + i, d.offsets[at]))});
625 try self.out.add("\n");
626 } else if (equal(cmd, "DEPARTURES") or equal(cmd, "ARRIVALS")) {
627 const station = try self.getStation(w[1]);
628 if (station.closed) return bad(self.gpa, "station {s} is closed", .{station.id});
629 const departures = equal(cmd, "DEPARTURES");
630 var choices: std.ArrayList(Departure) = .empty;
631 for (try keys(scratch, self.lines)) |id| {
632 const service = self.lines.get(id).?;
633 if (service.suspended) continue;
634 for (0..2) |which| {
635 const d = try service.direction(scratch, which);
636 const at = indexOf(d.stops, station) orelse continue;
637 if ((departures and at + 1 == d.stops.len) or (!departures and at == 0)) continue;
638 const q = service.firstTrain(d.offsets[at], timeValue(w[2])) orelse continue;
639 for (q..service.trains()) |train| try choices.append(scratch, .{ .time = service.time(train, d.offsets[at]), .service = service, .which = which, .terminal = d.stops[d.stops.len - 1], .start = d.stops[0] });
640 }
641 }
642 const Less = struct {
643 fn less(_: void, a: Departure, b: Departure) bool {
644 if (a.time != b.time) return a.time < b.time;
645 if (!equal(a.service.id, b.service.id)) return std.mem.order(u8, a.service.id, b.service.id) == .lt;
646 return a.which < b.which;
647 }
648 };
649 std.mem.sort(Departure, choices.items, {}, Less.less);
650 const count = @min(choices.items.len, number(w[3]));
651 try self.out.line("{s} {s} {s} count {d}", .{ cmd, station.id, w[2], count });
652 for (choices.items[0..count]) |choice| try self.out.line("{s} {s} {s} {s}", .{ try timeText(scratch, choice.time), choice.service.id, if (departures) @as(Text, "toward") else "from", if (departures) choice.terminal.id else choice.start.id });
653 } else if (equal(cmd, "TRIP") or equal(cmd, "SHOWLINE")) {
654 const service = try self.getLine(w[1]);
655 if (equal(cmd, "SHOWLINE")) {
656 try self.out.line("SHOWLINE {s} {s} first {s} last {s} headway {d} delay {d} trips {d}", .{ service.id, if (service.suspended) @as(Text, "suspended") else "running", try timeText(scratch, service.first), try timeText(scratch, service.last), service.headway, service.delay, service.trains() });
657 for (service.stops, 0..) |station, i| {
658 try self.out.print("{d} {s} zone {d}", .{ service.cumulative[i], station.id, station.zone });
659 if (station.closed) try self.out.add(" closed");
660 if (i > 0 and self.blocked(service.stops[i - 1], station)) try self.out.add(" blocked");
661 try self.out.add("\n");
662 }
663 } else {
664 const first = service.stops[0];
665 const last = service.stops[service.stops.len - 1];
666 if (!equal(w[2], first.id) and !equal(w[2], last.id)) return bad(self.gpa, "{s} is not a terminal of line {s}", .{ w[2], service.id });
667 if (service.suspended) return bad(self.gpa, "line {s} is suspended", .{service.id});
668 const d = try service.direction(scratch, if (equal(w[2], first.id)) 1 else 0);
669 const q = service.firstTrain(0, timeValue(w[3])) orelse {
670 try self.out.line("TRIP {s} toward {s} {s} none", .{ service.id, w[2], w[3] });
671 return;
672 };
673 try self.out.line("TRIP {s} toward {s} {s} trip {d} departs {s}", .{ service.id, w[2], w[3], q + 1, try timeText(scratch, service.time(q, 0)) });
674 for (d.stops, 0..) |station, i| {
675 try self.out.print("{s} {s}", .{ try timeText(scratch, service.time(q, d.offsets[i])), station.id });
676 if (station.closed) try self.out.add(" closed");
677 if (i > 0 and self.blocked(d.stops[i - 1], station)) try self.out.add(" blocked");
678 try self.out.add("\n");
679 }
680 }
681 } else if (equal(cmd, "INFO")) {
682 const station = try self.getStation(w[1]);
683 try self.out.line("INFO {s} {s} zone {d} transfer {d} {s}", .{ station.id, station.name, station.zone, station.transfer, if (station.closed) @as(Text, "closed") else "open" });
684 try self.out.add("LINES");
685 var count: usize = 0;
686 for (try keys(scratch, self.lines)) |id| {
687 if (indexOf(self.lines.get(id).?.stops, station) != null) {
688 try self.out.print(" {s}", .{id});
689 count += 1;
690 }
691 }
692 if (count == 0) try self.out.add(" none");
693 try self.out.add("\nWALKS");
694 const walks = try keys(scratch, station.walks);
695 if (walks.len == 0) try self.out.add(" none");
696 for (walks) |id| try self.out.print(" {s} {d}", .{ id, station.walks.get(id).? });
697 try self.out.add("\n");
698 } else if (equal(cmd, "STATUS")) {
699 const station_ids = try keys(scratch, self.stations);
700 const service_ids = try keys(scratch, self.lines);
701 try self.out.line("STATUS stations {d} lines {d} walks {d}", .{ self.stations.count(), self.lines.count(), self.walk_count });
702 for (station_ids) |id| {
703 if (self.stations.get(id).?.closed) try self.out.line("CLOSED {s}", .{id});
704 }
705 for (station_ids, 0..) |a, i| {
706 for (station_ids[i + 1 ..]) |b| {
707 if (self.blocked(self.stations.get(a).?, self.stations.get(b).?)) try self.out.line("CLOSED {s}-{s}", .{ a, b });
708 }
709 }
710 for (service_ids) |id| {
711 if (self.lines.get(id).?.suspended) try self.out.line("SUSPENDED {s}", .{id});
712 }
713 for (service_ids) |id| {
714 const delay = self.lines.get(id).?.delay;
715 if (delay != 0) try self.out.line("DELAYED {s} {d}", .{ id, delay });
716 }
717 }
718 }

721pub fn main(init: std.process.Init) !void {
722 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
723 defer arena.deinit();
724 const gpa = arena.allocator();
725 var input_buffer: [8192]u8 = undefined;
726 var input_reader = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
727 const input = try input_reader.interface.allocRemaining(gpa, .unlimited);
728 var input_lines = std.mem.splitScalar(u8, input, '\n');
729 const count = number(input_lines.next().?);
730 var out: Builder = .{ .gpa = gpa };
731 var network: Network = .{ .gpa = gpa, .out = &out };
732 for (0..count) |_| {
733 var scratch_arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
734 defer scratch_arena.deinit();
735 const scratch = scratch_arena.allocator();
736 var words: std.ArrayList(Text) = .empty;
737 var iterator = std.mem.splitScalar(u8, input_lines.next().?, ' ');
738 while (iterator.next()) |word| try words.append(scratch, word);
739 network.command(scratch, words.items) catch |err| switch (err) {
740 error.Invalid => try out.line("ERROR {s}", .{failure}),
741 else => return err,
742 };
743 }
744 var output_buffer: [8192]u8 = undefined;
745 var output_writer = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
746 try output_writer.interface.writeAll(out.bytes.items);
747 try output_writer.interface.flush();
748}
