1use std::collections::{BTreeMap, BTreeSet};
2use std::io::{self, Read};
3fn time(s: &str) -> i32 {
4 let (h,m) = s.split_once(':').unwrap();
5 h.parse::<i32>().unwrap()*60+m.parse::<i32>().unwrap()
6}
7fn clock(t: i32) -> String {
8 format!("{:02}:{:02}",t/60,t%60)
9}
10fn pair(a: &str, b: &str) -> (String,String) {
11 if a < b {
12 (a.into(),b.into())
13 } else {
14 (b.into(),a.into())
15 }
16}
17#[derive(Clone)]
18struct Station {
19 zone: i32, name: String, transfer: i32, closed: bool
20}
21#[derive(Clone)]
22struct Line {
23 first: i32, last: i32, headway: i32, stops: Vec<String>, offsets: Vec<i32>, delay: i32, suspended: bool
24}
25impl Line {
26 fn total(&self) -> i32 {
27 *self.offsets.last().unwrap()
28 }
29 fn trips(&self) -> i32 {
30 (self.last-self.first)/self.headway+1
31 }
32 fn direction(&self, reverse: bool) -> (Vec<String>,Vec<i32>) {
33 if reverse {
34 (self.stops.iter().rev().cloned().collect(),self.offsets.iter().rev().map(|&t|self.total()-t).collect())
35 } else {
36 (self.stops.clone(),self.offsets.clone())
37 }
38 }
39 fn next(&self, offset: i32, ready: i32) -> Option<i32> {
40 let delta = ready-self.first-self.delay-offset;
41 let q = if delta <= 0 {
42 0
43 } else {
44 (delta+self.headway-1)/self.headway
45 };
46 (q < self.trips()).then_some(q)
47 }
48 fn at(&self, q: i32, offset: i32) -> i32 {
49 self.first+self.delay+q*self.headway+offset
50 }
51}
52#[derive(Clone)]
53enum Leg {
54 Ride {
55 line: String, terminal: String, from: String, departure: i32, to: String, arrival: i32, stops: usize
56 },
57 Walk {
58 from: String, departure: i32, to: String, arrival: i32
59 },
60 Stay {
61 station: String, start: i32, end: i32
62 },
63}
64impl Leg {
65 fn end(&self) -> i32 {
66 match self {
67 Self::Ride {
68 arrival,..
69 }|Self::Walk {
70 arrival,..
71 } => *arrival, Self::Stay {
72 end,..
73 } => *end
74 }
75 }
76}
77#[derive(Clone)]
78struct Label {
79 arrival: i32, journey: Vec<Leg>
80}
81type Labels = BTreeMap<String,Label>;
82fn rides(j: &[Leg]) -> usize {
83 j.iter().filter(|l|matches!(l,Leg::Ride {
84 ..
85 })).count()
86}
87fn relax(labels: &mut Labels, station: &str, arrival: i32, journey: &[Leg], leg: Leg) -> bool {
88 if labels.get(station).is_none_or(|old|old.arrival > arrival) {
89 let mut j = journey.to_vec();
90 j.push(leg);
91 labels.insert(station.into(),Label {
92 arrival,journey:j
93 });
94 true
95 } else {
96 false
97 }
98}
99fn print_journey(heading: &str, departure: i32, label: &Label) -> Vec<String> {
100 let mut out = vec![format!("{heading} {} -> {} total {} rides {} walks {}",clock(departure),clock(label.arrival),label.arrival-departure,rides(&label.journey),label.journey.iter().filter(|l|matches!(l,Leg::Walk {
101 ..
102 })).count())];
103 let mut previous = departure;
104 for leg in &label.journey {
105 out.push(match leg {
106 Leg::Ride {
107 line,terminal,from,departure,to,arrival,stops
108 } => format!("RIDE {line} toward {terminal} {from} {} -> {to} {} wait {} stops {stops}",clock(*departure),clock(*arrival),departure-previous),
109 Leg::Walk {
110 from,departure,to,arrival
111 } => format!("WALK {from} {} -> {to} {}",clock(*departure),clock(*arrival)),
112 Leg::Stay {
113 station,start,end
114 } => format!("STAY {station} {} -> {}",clock(*start),clock(*end)),
115 });
116 previous = leg.end();
117 }
118 out
119}
120struct Planner {
121 stations: BTreeMap<String,Station>, lines: BTreeMap<String,Line>, walks: BTreeMap<(String,String),i32>, closed: BTreeSet<(String,String)>, tariff: Option<(i32,i32,Vec<i32>)>
122}
123impl Planner {
124 fn station(&self, s: &str) -> Result<&Station,String> {
125 self.stations.get(s).ok_or_else(||format!("unknown station {s}"))
126 }
127 fn line(&self, s: &str) -> Result<&Line,String> {
128 self.lines.get(s).ok_or_else(||format!("unknown line {s}"))
129 }
130 fn open_station(&self, s: &str) -> Result<(),String> {
131 if self.station(s)?.closed {
132 Err(format!("station {s} is closed"))
133 } else {
134 Ok(())
135 }
136 }
137 fn common(&self, a: &str, b: &str) -> Result<(),String> {
138 self.station(a)?;
139 self.station(b)?;
140 if a == b {
141 return Err("origin and destination are the same".into());
142 }
143 self.open_station(a)?;
144 self.open_station(b)
145 }
146 fn neighbors(&self, s: &str) -> Vec<(String,i32)> {
147 let mut out: Vec<_> = self.walks.iter().filter_map(|((a,b),&t)|if a == s {
148 Some((b.clone(),t))
149 } else if b == s {
150 Some((a.clone(),t))
151 } else {
152 None
153 }).collect();
154 out.sort_by(|a,b|a.0.cmp(&b.0));
155 out
156 }
157 fn earliest(&self, origin: &str, departure: i32) -> Labels {
158 let mut labels = Labels::from([(origin.into(),Label {
159 arrival:departure,journey:Vec::new()
160 })]);
161 let mut settled = BTreeSet::new();
162 loop {
163 let Some((u,label)) = labels.iter().filter(|(s,_)|!settled.contains(*s)).min_by_key(|(s,l)|(l.arrival,*s)).map(|(s,l)|(s.clone(),l.clone())) else {
164 break;
165 };
166 settled.insert(u.clone());
167 let ready = label.arrival+if u == origin {
168 0
169 } else {
170 self.stations[&u].transfer
171 };
172 for (id,line) in &self.lines {
173 if line.suspended || !line.stops.contains(&u) {
174 continue;
175 }
176 for reverse in [false,true] {
177 let (stops,offsets) = line.direction(reverse);
178 let i = stops.iter().position(|s|s == &u).unwrap();
179 if i+1 == stops.len() {
180 continue;
181 }
182 let Some(q) = line.next(offsets[i],ready) else {
183 continue;
184 };
185 let d = line.at(q,offsets[i]);
186 for j in i+1..stops.len() {
187 if self.closed.contains(&pair(&stops[j-1],&stops[j])) {
188 break;
189 }
190 if self.stations[&stops[j]].closed {
191 continue;
192 }
193 let t = line.at(q,offsets[j]);
194 relax(&mut labels,&stops[j],t,&label.journey,Leg::Ride {
195 line:id.clone(),terminal:stops.last().unwrap().clone(),from:u.clone(),departure:d,to:stops[j].clone(),arrival:t,stops:j-i
196 });
197 }
198 }
199 }
200 for (v,w) in self.neighbors(&u) {
201 let t = label.arrival+w;
202 if !self.stations[&v].closed && t <= 1440 {
203 relax(&mut labels,&v,t,&label.journey,Leg::Walk {
204 from:u.clone(),departure:label.arrival,to:v.clone(),arrival:t
205 });
206 }
207 }
208 }
209 labels
210 }
211 fn walk_step(&self, labels: &mut Labels, sources: &BTreeSet<String>, changed: &mut BTreeSet<String>) {
212 let recorded: Vec<_> = sources.iter().map(|s|(s.clone(),labels[s].clone())).collect();
213 for (u,label) in recorded {
214 for (v,w) in self.neighbors(&u) {
215 let t = label.arrival+w;
216 if !self.stations[&v].closed && t <= 1440 && relax(labels,&v,t,&label.journey,Leg::Walk {
217 from:u.clone(),departure:label.arrival,to:v.clone(),arrival:t
218 }) {
219 changed.insert(v);
220 }
221 }
222 }
223 }
224 fn rounds(&self, origin: &str, departure: i32) -> Vec<Labels> {
225 let mut labels = Labels::from([(origin.into(),Label {
226 arrival:departure,journey:Vec::new()
227 })]);
228 let mut changed = BTreeSet::from([origin.to_string()]);
229 self.walk_step(&mut labels,&BTreeSet::from([origin.to_string()]),&mut changed);
230 let mut result = vec![labels.clone()];
231 while !changed.is_empty() {
232 let previous = labels.clone();
233 let marked = changed;
234 changed = BTreeSet::new();
235 let mut sources = BTreeSet::new();
236 for (id,line) in &self.lines {
237 if line.suspended {
238 continue;
239 }
240 for reverse in [false,true] {
241 let (stops,offsets) = line.direction(reverse);
242 let mut boarded: Option<(i32,usize,i32,Vec<Leg>)> = None;
243 for j in 0..stops.len() {
244 if j > 0 && self.closed.contains(&pair(&stops[j-1],&stops[j])) {
245 boarded = None;
246 }
247 if self.stations[&stops[j]].closed {
248 continue;
249 }
250 if let Some((q,b,d,journey)) = &boarded {
251 let t = line.at(*q,offsets[j]);
252 if relax(&mut labels,&stops[j],t,journey,Leg::Ride {
253 line:id.clone(),terminal:stops.last().unwrap().clone(),from:stops[*b].clone(),departure:*d,to:stops[j].clone(),arrival:t,stops:j-*b
254 }) {
255 sources.insert(stops[j].clone());
256 changed.insert(stops[j].clone());
257 }
258 }
259 if j+1 < stops.len() && marked.contains(&stops[j]) {
260 let old = &previous[&stops[j]];
261 let ready = old.arrival+if stops[j] == origin {
262 0
263 } else {
264 self.stations[&stops[j]].transfer
265 };
266 if let Some(q) = line.next(offsets[j],ready) {
267 if boarded.as_ref().is_none_or(|b|q < b.0) {
268 boarded = Some((q,j,line.at(q,offsets[j]),old.journey.clone()));
269 }
270 }
271 }
272 }
273 }
274 }
275 self.walk_step(&mut labels,&sources,&mut changed);
276 result.push(labels.clone());
277 }
278 result
279 }
280 fn fare(&self, journey: &[Leg], child: bool) -> (i32,Vec<String>) {
281 let (discount,window,prices) = self.tariff.as_ref().unwrap();
282 let half = |n:i32|(n+19)/20*10;
283 let cap = if child {
284 half(*discount)
285 } else {
286 *discount
287 };
288 let mut previous = None;
289 let mut total = 0;
290 let mut tickets = Vec::new();
291 for leg in journey {
292 if let Leg::Ride {
293 line,from,to,departure,arrival,..
294 } = leg {
295 let stops = &self.lines[line].stops;
296 let a = stops.iter().position(|s|s == from).unwrap();
297 let b = stops.iter().position(|s|s == to).unwrap();
298 let zones = stops[a.min(b)..=a.max(b)].iter().map(|s|self.stations[s].zone).collect::<BTreeSet<_>>().len();
299 let mut price = prices[zones.min(prices.len())-1];
300 if child {
301 price = half(price);
302 }
303 let reduction = if previous.is_some_and(|t|departure-t <= *window) {
304 cap.min(price)
305 } else {
306 0
307 };
308 let pay = price-reduction;
309 total += pay;
310 previous = Some(*arrival);
311 tickets.push(format!("TICKET {line} {from} {to} zones {zones} price {price} discount {reduction} pay {pay}"));
312 }
313 } (total,tickets)
314 }
315 fn command(&mut self, w: &[&str]) -> Result<Vec<String>,String> {
316 let c = w[0];
317 let a = &w[1..];
318 let num = |i:usize|a[i].parse::<i32>().unwrap();
319 let mut out = Vec::new();
320 match c {
321 "STATION" => {
322 if self.stations.contains_key(a[0]) {
323 return Err(format!("station {} already exists",a[0]));
324 }
325 self.stations.insert(a[0].into(),Station {
326 zone:num(1),name:a[2].into(),transfer:0,closed:false
327 });
328 out.push(format!("STATION {} zone {} {}",a[0],num(1),a[2]));
329 }
330 "LINE" => {
331 if self.lines.contains_key(a[0]) {
332 return Err(format!("line {} already exists",a[0]));
333 }
334 let stops: Vec<String> = a[4..].iter().step_by(2).map(|s|s.to_string()).collect();
335 for s in &stops {
336 self.station(s)?;
337 }
338 let mut seen = BTreeSet::new();
339 for s in &stops {
340 if !seen.insert(s.clone()) {
341 return Err(format!("station {s} appears twice on line {}",a[0]));
342 }
343 }
344 let first = time(a[1]);
345 let last = time(a[2]);
346 let mut offsets = vec![0];
347 for s in a[5..].iter().step_by(2) {
348 offsets.push(offsets.last().unwrap()+s.parse::<i32>().unwrap());
349 }
350 if first > last {
351 return Err(format!("line {} first departure is after last departure",a[0]));
352 }
353 if last+offsets.last().unwrap() > 1440 {
354 return Err(format!("line {} runs past 24:00",a[0]));
355 }
356 let line = Line {
357 first,last,headway:num(3),stops,offsets,delay:0,suspended:false
358 };
359 out.push(format!("LINE {} stops {} time {} trips {}",a[0],line.stops.len(),line.total(),line.trips()));
360 self.lines.insert(a[0].into(),line);
361 }
362 "TRANSFER" => {
363 self.station(a[0])?;
364 self.stations.get_mut(a[0]).unwrap().transfer = num(1);
365 out.push(format!("TRANSFER {} {}",a[0],num(1)));
366 }
367 "WALK" => {
368 self.station(a[0])?;
369 self.station(a[1])?;
370 if a[0] == a[1] {
371 return Err(format!("walk from {} to itself",a[0]));
372 }
373 let old = self.walks.insert(pair(a[0],a[1]),num(2));
374 out.push(format!("WALK {} {} {}{}",a[0],a[1],num(2),old.map_or(String::new(),|n|format!(" was {n}"))));
375 }
376 "TARIFF" => {
377 let prices: Vec<_> = (2..a.len()).map(num).collect();
378 if prices.windows(2).any(|p|p[0] > p[1]) {
379 return Err("tariff prices must not decrease".into());
380 }
381 out.push(format!("TARIFF zones {} discount {} window {}",prices.len(),num(0),num(1)));
382 self.tariff = Some((num(0),num(1),prices));
383 }
384 "CLOSE" | "OPEN" => {
385 self.station(a[0])?;
386 let close = c == "CLOSE";
387 if a.len() == 1 {
388 let station = self.stations.get_mut(a[0]).unwrap();
389 if station.closed == close {
390 return Err(format!("station {} is {}closed",a[0],if close {
391 "already "
392 } else {
393 "not "
394 }));
395 } station.closed = close;
396 out.push(format!("{c} {}",a[0]));
397 }
398 else {
399 self.station(a[1])?;
400 let key = pair(a[0],a[1]);
401 let lines: Vec<_> = self.lines.iter().filter(|(_,l)|l.stops.windows(2).any(|s|pair(&s[0],&s[1]) == key)).map(|(id,_)|id.clone()).collect();
402 if close && lines.is_empty() {
403 return Err(format!("no segment {}-{}",a[0],a[1]));
404 }
405 if self.closed.contains(&key) == close {
406 return Err(format!("segment {}-{} is {}closed",a[0],a[1],if close {
407 "already "
408 } else {
409 "not "
410 }));
411 }
412 if close {
413 self.closed.insert(key);
414 } else {
415 self.closed.remove(&key);
416 }
417 out.push(format!("{c} {}-{}{}",a[0],a[1],if close {
418 " lines ".to_string()+&lines.join(" ")
419 } else {
420 String::new()
421 }));
422 }
423 }
424 "DELAY" => {
425 let line = self.line(a[0])?;
426 if line.last+num(1)+line.total() > 1440 {
427 return Err(format!("line {} runs past 24:00",a[0]));
428 }
429 self.lines.get_mut(a[0]).unwrap().delay = num(1);
430 out.push(format!("DELAY {} {}",a[0],num(1)));
431 }
432 "SUSPEND" | "RESUME" => {
433 let suspended = c == "SUSPEND";
434 let line = self.line(a[0])?;
435 if line.suspended == suspended {
436 return Err(format!("line {} is {}suspended",a[0],if suspended {
437 "already "
438 } else {
439 "not "
440 }));
441 }
442 self.lines.get_mut(a[0]).unwrap().suspended = suspended;
443 out.push(format!("{c} {}",a[0]));
444 }
445 "ROUTE" | "FEWEST" | "PARETO" | "FARE" | "LASTTRAIN" | "ARRIVEBY" | "PROFILE" => {
446 self.common(a[0],a[1])?;
447 if c == "FARE" {
448 if !["ADULT","CHILD"].contains(&a[3]) {
449 return Err(format!("unknown fare class {}",a[3]));
450 }
451 if self.tariff.is_none() {
452 return Err("no tariff".into());
453 }
454 }
455 if c == "PROFILE" && time(a[2]) > time(a[3]) {
456 return Err(format!("empty interval {} {}",a[2],a[3]));
457 }
458 let heading = format!("{c} {} {}",a[0],a[1]);
459 if c == "LASTTRAIN" || c == "ARRIVEBY" {
460 let bound = if c == "LASTTRAIN" {
461 1440
462 } else {
463 time(a[2])
464 };
465 let title = if c == "ARRIVEBY" {
466 format!("{heading} by {}",a[2])
467 } else {
468 heading.clone()
469 };
470 for t in (0..bound).rev() {
471 if let Some(label) = self.earliest(a[0],t).get(a[1]) {
472 if c == "LASTTRAIN" || label.arrival <= bound {
473 return Ok(print_journey(&title,t,label));
474 }
475 }
476 }
477 out.push(format!("{title} none"));
478 }
479 else if c == "PROFILE" {
480 let start = time(a[2]);
481 let end = time(a[3]);
482 let arrivals: Vec<_> = (start..=end).map(|t|self.earliest(a[0],t).get(a[1]).map(|l|l.arrival)).collect();
483 let mut selected = Vec::new();
484 for (i,&arrival) in arrivals.iter().enumerate() {
485 if let Some(arrival) = arrival {
486 if i+1 == arrivals.len() || arrivals[i+1].is_none_or(|next|arrival < next) {
487 selected.push(format!("{} -> {} total {}",clock(start+i as i32),clock(arrival),arrival-start-i as i32));
488 }
489 }
490 }
491 out.push(format!("{heading} {} {} count {}",a[2],a[3],selected.len()));
492 out.extend(selected);
493 }
494 else if c == "FEWEST" || c == "PARETO" {
495 let start = time(a[2]);
496 let rounds = self.rounds(a[0],start);
497 if c == "FEWEST" {
498 for (k,labels) in rounds.iter().enumerate() {
499 if let Some(label) = labels.get(a[1]) {
500 return Ok(print_journey(&format!("{heading} trips {k}"),start,label));
501 }
502 }
503 out.push(format!("{heading} {} unreachable",a[2]));
504 } else {
505 let mut options = Vec::new();
506 let mut previous = None;
507 let mut count = 0;
508 for (k,labels) in rounds.iter().enumerate() {
509 if let Some(label) = labels.get(a[1]) {
510 if previous.is_none_or(|t|t > label.arrival) {
511 count += 1;
512 options.extend(print_journey(&format!("OPTION trips {k}"),start,label));
513 } previous = Some(label.arrival);
514 }
515 }
516 out.push(format!("{heading} {} options {count}",a[2]));
517 out.extend(options);
518 }
519 }
520 else {
521 let start = time(a[2]);
522 let labels = self.earliest(a[0],start);
523 if let Some(label) = labels.get(a[1]) {
524 if c == "FARE" {
525 let (total,tickets) = self.fare(&label.journey,a[3] == "CHILD");
526 out.push(format!("{heading} {} {} rides {} total {total}",a[2],a[3],rides(&label.journey)));
527 out.extend(tickets);
528 } else {
529 out = print_journey(&heading,start,label);
530 }
531 } else {
532 out.push(format!("{heading} {}{} unreachable",a[2],if c == "FARE" {
533 " ".to_string()+a[3]
534 } else {
535 String::new()
536 }));
537 }
538 }
539 }
540 "VIA" => {
541 self.station(a[0])?;
542 self.station(a[1])?;
543 self.station(a[2])?;
544 if a[0] == a[2] {
545 return Err("origin and destination are the same".into());
546 }
547 if a[1] == a[0] || a[1] == a[2] {
548 return Err(format!("via station {} must differ from origin and destination",a[1]));
549 }
550 self.open_station(a[0])?;
551 self.open_station(a[1])?;
552 self.open_station(a[2])?;
553 let start = time(a[3]);
554 let heading = format!("VIA {} {} {}",a[0],a[1],a[2]);
555 if let Some(first) = self.earliest(a[0],start).get(a[1]) {
556 let next = first.arrival+self.stations[a[1]].transfer;
557 if let Some(second) = self.earliest(a[1],next).get(a[2]) {
558 let mut journey = first.journey.clone();
559 journey.push(Leg::Stay {
560 station:a[1].into(),start:first.arrival,end:next
561 });
562 journey.extend(second.journey.clone());
563 return Ok(print_journey(&heading,start,&Label {
564 arrival:second.arrival,journey
565 }));
566 }
567 }
568 out.push(format!("{heading} {} unreachable",a[3]));
569 }
570 "REACH" => {
571 self.open_station(a[0])?;
572 let start = time(a[1]);
573 let mut selected: Vec<_> = self.earliest(a[0],start).into_iter().filter(|(_,l)|l.arrival <= start+num(2)).collect();
574 selected.sort_by(|a,b|a.1.arrival.cmp(&b.1.arrival).then(a.0.cmp(&b.0)));
575 out.push(format!("REACH {} {} {} count {}",a[0],a[1],num(2),selected.len()));
576 for (s,label) in selected {
577 out.push(format!("{s} {} {} {} rides {}",self.stations[&s].name,clock(label.arrival),label.arrival-start,rides(&label.journey)));
578 }
579 }
580 "TIMETABLE" => {
581 self.station(a[0])?;
582 let line = self.line(a[1])?;
583 if !line.stops.iter().any(|s|s == a[0]) {
584 return Err(format!("station {} is not on line {}",a[0],a[1]));
585 }
586 if a[2] != line.stops[0] && a[2] != *line.stops.last().unwrap() {
587 return Err(format!("{} is not a terminal of line {}",a[2],a[1]));
588 }
589 if a[0] == a[2] {
590 return Err(format!("no departures from {} toward {}",a[0],a[2]));
591 }
592 if line.suspended {
593 return Err(format!("line {} is suspended",a[1]));
594 }
595 self.open_station(a[0])?;
596 let (stops,offsets) = line.direction(a[2] == line.stops[0]);
597 let i = stops.iter().position(|s|s == a[0]).unwrap();
598 let selected: Vec<_> = (0..line.trips()).map(|q|line.at(q,offsets[i])).filter(|&t|t >= time(a[3])).take(num(4) as usize).map(clock).collect();
599 out.push(format!("TIMETABLE {} {} toward {} {} next {}{}",a[0],a[1],a[2],a[3],selected.len(),if selected.is_empty() {
600 String::new()
601 } else {
602 " ".to_string()+&selected.join(" ")
603 }));
604 }
605 "DEPARTURES" | "ARRIVALS" => {
606 self.open_station(a[0])?;
607 let mut events = Vec::new();
608 for (id,line) in &self.lines {
609 if line.suspended || !line.stops.iter().any(|s|s == a[0]) {
610 continue;
611 }
612 for reverse in [false,true] {
613 let (stops,offsets) = line.direction(reverse);
614 let i = stops.iter().position(|s|s == a[0]).unwrap();
615 if (c == "DEPARTURES" && i+1 == stops.len()) || (c == "ARRIVALS" && i == 0) {
616 continue;
617 }
618 for q in 0..line.trips() {
619 let t = line.at(q,offsets[i]);
620 if t >= time(a[1]) {
621 events.push((t,id.clone(),reverse,if c == "DEPARTURES" {
622 stops.last().unwrap().clone()
623 } else {
624 stops[0].clone()
625 }));
626 }
627 }
628 }
629 } events.sort();
630 events.truncate(num(2) as usize);
631 out.push(format!("{c} {} {} count {}",a[0],a[1],events.len()));
632 for (t,id,_,endpoint) in events {
633 out.push(format!("{} {id} {} {endpoint}",clock(t),if c == "DEPARTURES" {
634 "toward"
635 } else {
636 "from"
637 }));
638 }
639 }
640 "TRIP" => {
641 let line = self.line(a[0])?;
642 if a[1] != line.stops[0] && a[1] != *line.stops.last().unwrap() {
643 return Err(format!("{} is not a terminal of line {}",a[1],a[0]));
644 }
645 if line.suspended {
646 return Err(format!("line {} is suspended",a[0]));
647 }
648 let (stops,offsets) = line.direction(a[1] == line.stops[0]);
649 let heading = format!("TRIP {} toward {} {}",a[0],a[1],a[2]);
650 if let Some(q) = line.next(0,time(a[2])) {
651 out.push(format!("{heading} trip {} departs {}",q+1,clock(line.at(q,0))));
652 for j in 0..stops.len() {
653 out.push(format!("{} {}{}{}",clock(line.at(q,offsets[j])),stops[j],if self.stations[&stops[j]].closed {
654 " closed"
655 } else {
656 ""
657 },if j > 0 && self.closed.contains(&pair(&stops[j-1],&stops[j])) {
658 " blocked"
659 } else {
660 ""
661 }));
662 }
663 } else {
664 out.push(format!("{heading} none"));
665 }
666 }
667 "SHOWLINE" => {
668 let line = self.line(a[0])?;
669 out.push(format!("SHOWLINE {} {} first {} last {} headway {} delay {} trips {}",a[0],if line.suspended {
670 "suspended"
671 } else {
672 "running"
673 },clock(line.first),clock(line.last),line.headway,line.delay,line.trips()));
674 for j in 0..line.stops.len() {
675 let s = &line.stops[j];
676 out.push(format!("{} {s} zone {}{}{}",line.offsets[j],self.stations[s].zone,if self.stations[s].closed {
677 " closed"
678 } else {
679 ""
680 },if j > 0 && self.closed.contains(&pair(&line.stops[j-1],s)) {
681 " blocked"
682 } else {
683 ""
684 }));
685 }
686 }
687 "INFO" => {
688 let s = self.station(a[0])?;
689 out.push(format!("INFO {} {} zone {} transfer {} {}",a[0],s.name,s.zone,s.transfer,if s.closed {
690 "closed"
691 } else {
692 "open"
693 }));
694 let lines: Vec<_> = self.lines.iter().filter(|(_,l)|l.stops.iter().any(|s|s == a[0])).map(|(id,_)|id.clone()).collect();
695 out.push(format!("LINES {}",if lines.is_empty() {
696 "none".into()
697 } else {
698 lines.join(" ")
699 }));
700 let walks = self.neighbors(a[0]);
701 out.push(format!("WALKS {}",if walks.is_empty() {
702 "none".into()
703 } else {
704 walks.iter().map(|(s,t)|format!("{s} {t}")).collect::<Vec<_>>().join(" ")
705 }));
706 }
707 "STATUS" => {
708 out.push(format!("STATUS stations {} lines {} walks {}",self.stations.len(),self.lines.len(),self.walks.len()));
709 for (s,station) in &self.stations {
710 if station.closed {
711 out.push(format!("CLOSED {s}"));
712 }
713 }
714 for (a,b) in &self.closed {
715 out.push(format!("CLOSED {a}-{b}"));
716 }
717 for (id,line) in &self.lines {
718 if line.suspended {
719 out.push(format!("SUSPENDED {id}"));
720 }
721 }
722 for (id,line) in &self.lines {
723 if line.delay != 0 {
724 out.push(format!("DELAYED {id} {}",line.delay));
725 }
726 }
727 }
728 _ => unreachable!(),
729 } Ok(out)
730 }
731}
732fn main() {
733 let mut input = String::new();
734 io::stdin().read_to_string(&mut input).unwrap();
735 let mut lines = input.lines();
736 let n: usize = lines.next().unwrap().parse().unwrap();
737 let mut planner = Planner {
738 stations:BTreeMap::new(),lines:BTreeMap::new(),walks:BTreeMap::new(),closed:BTreeSet::new(),tariff:None
739 };
740 for line in lines.take(n) {
741 match planner.command(&line.split(' ').collect::<Vec<_>>()) {
742 Ok(out) => for line in out {
743 println!("{line}");
744 }, Err(error) => println!("ERROR {error}")
745 }
746 }
747}
