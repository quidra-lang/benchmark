17static std::string clockTime(int value) {
18 std::ostringstream result;
19 result << std::setfill('0') << std::setw(2) << value / 60 << ':' << std::setw(2) << value % 60;
20 return result.str();
21}

39struct Line {
40 int first;
41 int last;
42 int headway;
43 std::vector<std::string> stops;
44 std::vector<int> cumulative;
45 int delay = 0;
46 bool suspended = false;
47
48 int trips() const {
49 return (last - first) / headway + 1;
50 }
51
52 bool contains(const std::string& station) const {
53 return std::find(stops.begin(), stops.end(), station) != stops.end();
54 }
55
56 Direction direction(int reverse) const {
57 if (reverse == 0) return {stops, cumulative};
58 Direction result;
59 for (std::size_t j = stops.size(); j-- > 0;) {
60 result.stops.push_back(stops[j]);
61 result.offsets.push_back(cumulative.back() - cumulative[j]);
62 }
63 return result;
64 }
65
66 int nextTrain(int ready, int offset) const {
67 int start = first + delay + offset;
68 int train = ready <= start ? 0 : (ready - start + headway - 1) / headway;
69 return train < trips() ? train : -1;
70 }
71
72 int at(int train, int offset) const {
73 return first + delay + train * headway + offset;
74 }
75};

99static Segment segment(const std::string& a, const std::string& b) {
100 return a < b ? Segment{a, b} : Segment{b, a};
101}

103static bool improve(Labels& labels, const std::string& station, const Journey& journey, const Leg& leg) {
104 auto found = labels.find(station);
105 if (found != labels.end() && found->second.arrival <= leg.arrival) return false;
106 Journey next = journey;
107 next.push_back(leg);
108 labels[station] = {leg.arrival, std::move(next)};
109 return true;
110}

112static int legCount(const Journey& journey, char kind) {
113 return static_cast<int>(std::count_if(journey.begin(), journey.end(), [kind](const Leg& leg) {
114 return leg.kind == kind;
115 }));
116}

118static void printJourney(const std::string& heading, int departure, const Label& label) {
119 std::cout << heading << ' ' << clockTime(departure) << " -> " << clockTime(label.arrival)
120 << " total " << label.arrival - departure << " rides " << legCount(label.journey, 'R')
121 << " walks " << legCount(label.journey, 'W') << '\n';
122 int previous = departure;
123 for (const Leg& leg : label.journey) {
124 if (leg.kind == 'R') {
125 std::cout << "RIDE " << leg.line << " toward " << leg.terminal << ' ' << leg.from << ' '
126 << clockTime(leg.departure) << " -> " << leg.to << ' ' << clockTime(leg.arrival)
127 << " wait " << leg.departure - previous << " stops " << leg.stops << '\n';
128 } else if (leg.kind == 'W') {
129 std::cout << "WALK " << leg.from << ' ' << clockTime(leg.departure) << " -> " << leg.to << ' ' << clockTime(leg.arrival) << '\n';
130 } else {
131 std::cout << "STAY " << leg.from << ' ' << clockTime(leg.departure) << " -> " << clockTime(leg.arrival) << '\n';
132 }
133 previous = leg.arrival;
134 }

137class Network {
138 std::map<std::string, Station> stations;
139 std::map<std::string, Line> lines;
140 std::map<Segment, int> walks;
141 std::set<Segment> closedSegments;
142 int discount = 0;
143 int discountWindow = 0;
144 std::vector<int> prices;

146 const Station& requireStation(const std::string& name) const {
147 auto found = stations.find(name);
148 if (found == stations.end()) fail("unknown station " + name);
149 return found->second;
150 }
151
152 const Line& requireLine(const std::string& name) const {
153 auto found = lines.find(name);
154 if (found == lines.end()) fail("unknown line " + name);
155 return found->second;
156 }

158 void requireOpen(const std::string& name) const {
159 if (stations.at(name).closed) fail("station " + name + " is closed");
160 }
161
162 void checkSearch(const std::string& origin, const std::string& destination) const {
163 requireStation(origin);
164 requireStation(destination);
165 if (origin == destination) fail("origin and destination are the same");
166 requireOpen(origin);
167 requireOpen(destination);
168 }

170 bool blocked(const std::string& a, const std::string& b) const {
171 return closedSegments.count(segment(a, b)) != 0;
172 }
173
174 std::map<std::string, int> neighbors(const std::string& station) const {
175 std::map<std::string, int> result;
176 for (const auto& [edge, duration] : walks) {
177 if (edge.first == station) result[edge.second] = duration;
178 else if (edge.second == station) result[edge.first] = duration;
179 }
180 return result;
181 }
182
183 Labels earliest(const std::string& origin, int departure) const {
184 Labels labels = {{origin, {departure, {}}}};
185 Stations settled;
186 while (true) {
187 std::string current;
188 int best = 0;
189 for (const auto& [name, label] : labels) {
190 if (!settled.count(name) && (current.empty() || label.arrival < best)) {
191 current = name;
192 best = label.arrival;
193 }
194 }
195 if (current.empty()) break;
196 settled.insert(current);
197 Label label = labels.at(current);
198 int ready = label.arrival + (current == origin ? 0 : stations.at(current).transfer);
199 for (const auto& [name, line] : lines) {
200 if (line.suspended || !line.contains(current)) continue;
201 for (int reverse = 0; reverse < 2; ++reverse) {
202 Direction direction = line.direction(reverse);
203 int board = static_cast<int>(std::find(direction.stops.begin(), direction.stops.end(), current) - direction.stops.begin());
204 if (board + 1 == static_cast<int>(direction.stops.size())) continue;
205 int train = line.nextTrain(ready, direction.offsets[board]);
206 if (train < 0) continue;
207 int boardTime = line.at(train, direction.offsets[board]);
208 for (int j = board + 1; j < static_cast<int>(direction.stops.size()); ++j) {
209 if (blocked(direction.stops[j - 1], direction.stops[j])) break;
210 const std::string& target = direction.stops[j];
211 if (stations.at(target).closed) continue;
212 Leg ride = {'R', name, direction.stops.back(), current, boardTime,
213 target, line.at(train, direction.offsets[j]), j - board};
214 improve(labels, target, label.journey, ride);
215 }
216 }
217 }
218 for (const auto& [target, duration] : neighbors(current)) {
219 int arrival = label.arrival + duration;
220 if (!stations.at(target).closed && arrival <= 1440) {
221 improve(labels, target, label.journey, {'W', "", "", current, label.arrival, target, arrival});
222 }
223 }
224 }
225 return labels;
226 }
227
228 void walkStep(Labels& labels, const Stations& sources, Stations& marked) const {
229 Labels recorded;
230 for (const std::string& source : sources) recorded[source] = labels.at(source);
231 for (const auto& [source, label] : recorded) {
232 for (const auto& [target, duration] : neighbors(source)) {
233 int arrival = label.arrival + duration;
234 if (!stations.at(target).closed && arrival <= 1440 &&
235 improve(labels, target, label.journey, {'W', "", "", source, label.arrival, target, arrival})) {
236 marked.insert(target);
237 }
238 }
239 }
240 }
241
242 std::vector<Labels> rounds(const std::string& origin, int departure) const {
243 Labels labels = {{origin, {departure, {}}}};
244 Stations marked = {origin};
245 walkStep(labels, {origin}, marked);
246 std::vector<Labels> result = {labels};
247 while (!marked.empty()) {
248 Labels previous = labels;
249 Stations previousMarked = marked;
250 marked.clear();
251 Stations rideUpdates;
252 for (const auto& [name, line] : lines) {
253 if (line.suspended) continue;
254 for (int reverse = 0; reverse < 2; ++reverse) {
255 Direction direction = line.direction(reverse);
256 int train = -1;
257 int board = -1;
258 int boardTime = 0;
259 Journey boardedJourney;
260 for (int j = 0; j < static_cast<int>(direction.stops.size()); ++j) {
261 const std::string& station = direction.stops[j];
262 if (j > 0 && blocked(direction.stops[j - 1], station)) train = -1;
263 if (stations.at(station).closed) continue;
264 if (train >= 0) {
265 Leg ride = {'R', name, direction.stops.back(), direction.stops[board], boardTime,
266 station, line.at(train, direction.offsets[j]), j - board};
267 if (improve(labels, station, boardedJourney, ride)) {
268 rideUpdates.insert(station);
269 marked.insert(station);
270 }
271 }
272 if (j + 1 < static_cast<int>(direction.stops.size()) && previousMarked.count(station)) {
273 const Label& old = previous.at(station);
274 int ready = old.arrival + (station == origin ? 0 : stations.at(station).transfer);
275 int candidate = line.nextTrain(ready, direction.offsets[j]);
276 if (candidate >= 0 && (train < 0 || candidate < train)) {
277 train = candidate;
278 board = j;
279 boardTime = line.at(train, direction.offsets[j]);
280 boardedJourney = old.journey;
281 }
282 }
283 }
284 }
285 }
286 walkStep(labels, rideUpdates, marked);
287 result.push_back(labels);
288 }
289 return result;
290 }

292 void fare(const std::string& heading, const Journey& journey, bool child) const {
293 std::vector<std::string> tickets;
294 int total = 0;
295 int previousArrival = -1;
296 for (const Leg& leg : journey) {
297 if (leg.kind != 'R') continue;
298 const Line& line = lines.at(leg.line);
299 Direction direction = line.direction(leg.terminal == line.stops.back() ? 0 : 1);
300 auto first = std::find(direction.stops.begin(), direction.stops.end(), leg.from);
301 auto last = std::find(direction.stops.begin(), direction.stops.end(), leg.to);
302 std::set<int> zones;
303 for (auto at = first; at <= last; ++at) zones.insert(stations.at(*at).zone);
304 int price = prices[std::min(zones.size(), prices.size()) - 1];
305 if (child) price = (price + 19) / 20 * 10;
306 int cap = child ? (discount + 19) / 20 * 10 : discount;
307 int reduction = previousArrival >= 0 && leg.departure - previousArrival <= discountWindow ? std::min(cap, price) : 0;
308 int payment = price - reduction;
309 total += payment;
310 previousArrival = leg.arrival;
311 std::ostringstream ticket;
312 ticket << "TICKET " << leg.line << ' ' << leg.from << ' ' << leg.to << " zones " << zones.size()
313 << " price " << price << " discount " << reduction << " pay " << payment;
314 tickets.push_back(ticket.str());
315 }
316 std::cout << heading << " rides " << tickets.size() << " total " << total << '\n';
317 for (const std::string& ticket : tickets) std::cout << ticket << '\n';
318 }

320 int directionTo(const std::string& name, const Line& line, const std::string& terminal) const {
321 if (terminal != line.stops.front() && terminal != line.stops.back()) fail(terminal + " is not a terminal of line " + name);
322 return terminal == line.stops.back() ? 0 : 1;
323 }

325 void departures(const std::vector<std::string>& words, bool arrivals) const {
326 const std::string& station = words[1];
327 requireStation(station);
328 requireOpen(station);
329 int time = timeValue(words[2]);
330 int count = std::stoi(words[3]);
331 using Event = std::tuple<int, std::string, int, std::string>;
332 std::vector<Event> events;
333 for (const auto& [name, line] : lines) {
334 if (line.suspended || !line.contains(station)) continue;
335 for (int reverse = 0; reverse < 2; ++reverse) {
336 Direction direction = line.direction(reverse);
337 int position = static_cast<int>(std::find(direction.stops.begin(), direction.stops.end(), station) - direction.stops.begin());
338 if (arrivals ? position == 0 : position + 1 == static_cast<int>(direction.stops.size())) continue;
339 int first = line.nextTrain(time, direction.offsets[position]);
340 if (first < 0) continue;
341 for (int train = first; train < line.trips(); ++train) {
342 events.emplace_back(line.at(train, direction.offsets[position]), name, reverse,
343 arrivals ? direction.stops.front() : direction.stops.back());
344 }
345 }
346 }
347 std::sort(events.begin(), events.end());
348 if (events.size() > static_cast<std::size_t>(count)) events.resize(count);
349 std::cout << words[0] << ' ' << station << ' ' << words[2] << " count " << events.size() << '\n';
350 for (const auto& [at, name, reverse, terminal] : events) {
351 std::cout << clockTime(at) << ' ' << name << (arrivals ? " from " : " toward ") << terminal << '\n';
352 }
353 }

355 void searchCommand(const std::vector<std::string>& words) const {
356 const std::string& command = words[0];
357 const std::string& origin = words[1];
358 const std::string& destination = words[2];
359 checkSearch(origin, destination);
360 std::string heading = command + ' ' + origin + ' ' + destination;
361 if (command == "LASTTRAIN" || command == "ARRIVEBY") {
362 int deadline = command == "LASTTRAIN" ? 1440 : timeValue(words[3]);
363 if (command == "ARRIVEBY") heading += " by " + words[3];
364 for (int departure = deadline - 1; departure >= 0; --departure) {
365 Labels labels = earliest(origin, departure);
366 auto target = labels.find(destination);
367 if (target != labels.end() && target->second.arrival <= deadline) {
368 printJourney(heading, departure, target->second);
369 return;
370 }
371 }
372 std::cout << heading << " none\n";
373 return;
374 }
375 int departure = timeValue(words[3]);
376 if (command == "PROFILE") {
377 int end = timeValue(words[4]);
378 if (departure > end) fail("empty interval " + words[3] + ' ' + words[4]);
379 std::vector<int> arrivals(end - departure + 1, -1);
380 for (int time = departure; time <= end; ++time) {
381 Labels labels = earliest(origin, time);
382 if (labels.count(destination)) arrivals[time - departure] = labels.at(destination).arrival;
383 }
384 std::vector<std::pair<int, int>> selected;
385 for (int time = departure; time <= end; ++time) {
386 int arrival = arrivals[time - departure];
387 if (arrival >= 0 && (time == end || arrivals[time - departure + 1] < 0 || arrival < arrivals[time - departure + 1])) selected.emplace_back(time, arrival);
388 }
389 std::cout << heading << ' ' << words[3] << ' ' << words[4] << " count " << selected.size() << '\n';
390 for (const auto& [time, arrival] : selected) std::cout << clockTime(time) << " -> " << clockTime(arrival) << " total " << arrival - time << '\n';
391 return;
392 }
393 if (command == "FEWEST" || command == "PARETO") {
394 std::vector<Labels> labels = rounds(origin, departure);
395 std::vector<std::pair<int, Label>> options;
396 int previous = -1;
397 for (std::size_t round = 0; round < labels.size(); ++round) {
398 auto target = labels[round].find(destination);
399 if (target != labels[round].end() && (previous < 0 || target->second.arrival < previous)) {
400 options.emplace_back(static_cast<int>(round), target->second);
401 }
402 if (target != labels[round].end()) previous = target->second.arrival;
403 }
404 if (command == "PARETO") {
405 std::cout << heading << ' ' << words[3] << " options " << options.size() << '\n';
406 for (const auto& [rides, label] : options) printJourney("OPTION trips " + std::to_string(rides), departure, label);
407 } else if (options.empty()) std::cout << heading << ' ' << words[3] << " unreachable\n";
408 else printJourney(heading + " trips " + std::to_string(options[0].first), departure, options[0].second);
409 return;
410 }
411 if (command == "FARE") {
412 if (words[4] != "ADULT" && words[4] != "CHILD") fail("unknown fare class " + words[4]);
413 if (prices.empty()) fail("no tariff");
414 heading += ' ' + words[3] + ' ' + words[4];
415 }
416 Labels labels = earliest(origin, departure);
417 auto target = labels.find(destination);
418 if (target == labels.end()) std::cout << heading << (command == "FARE" ? "" : " " + words[3]) << " unreachable\n";
419 else if (command == "FARE") fare(heading, target->second.journey, words[4] == "CHILD");
420 else printJourney(heading, departure, target->second);
421 }

423public:
424 void execute(const std::string& text) {
425 std::istringstream input(text);
426 std::vector<std::string> words;
427 for (std::string word; input >> word;) words.push_back(word);
428 const std::string& command = words[0];
429 const std::string first = words.size() > 1 ? words[1] : "";
430 if (command == "STATION") {
431 if (stations.count(first)) fail("station " + first + " already exists");
432 stations.emplace(first, Station{std::stoi(words[2]), words[3]});
433 std::cout << "STATION " << first << " zone " << words[2] << ' ' << words[3] << '\n';
434 } else if (command == "LINE") {
435 if (lines.count(first)) fail("line " + first + " already exists");
436 Line line{timeValue(words[2]), timeValue(words[3]), std::stoi(words[4]), {}, {0}};
437 for (std::size_t i = 5; i < words.size(); i += 2) {
438 requireStation(words[i]);
439 line.stops.push_back(words[i]);
440 if (i > 5) line.cumulative.push_back(line.cumulative.back() + std::stoi(words[i - 1]));
441 }
442 Stations seen;
443 for (const std::string& stop : line.stops) {
444 if (!seen.insert(stop).second) fail("station " + stop + " appears twice on line " + first);
445 }
446 if (line.first > line.last) fail("line " + first + " first departure is after last departure");
447 if (line.last + line.cumulative.back() > 1440) fail("line " + first + " runs past 24:00");
448 lines.emplace(first, line);
449 std::cout << "LINE " << first << " stops " << line.stops.size() << " time " << line.cumulative.back() << " trips " << line.trips() << '\n';
450 } else if (command == "TRANSFER") {
451 requireStation(first);
452 stations.at(first).transfer = std::stoi(words[2]);
453 std::cout << "TRANSFER " << first << ' ' << words[2] << '\n';
454 } else if (command == "WALK") {
455 requireStation(first);
456 requireStation(words[2]);
457 if (first == words[2]) fail("walk from " + first + " to itself");
458 Segment edge = segment(first, words[2]);
459 auto found = walks.find(edge);
460 std::cout << "WALK " << first << ' ' << words[2] << ' ' << words[3];
461 if (found != walks.end()) std::cout << " was " << found->second;
462 std::cout << '\n';
463 walks[edge] = std::stoi(words[3]);
464 } else if (command == "TARIFF") {
465 std::vector<int> updated;
466 for (std::size_t i = 3; i < words.size(); ++i) updated.push_back(std::stoi(words[i]));
467 if (!std::is_sorted(updated.begin(), updated.end())) fail("tariff prices must not decrease");
468 discount = std::stoi(words[1]);
469 discountWindow = std::stoi(words[2]);
470 prices = updated;
471 std::cout << "TARIFF zones " << prices.size() << " discount " << discount << " window " << discountWindow << '\n';
472 } else if (command == "CLOSE" || command == "OPEN") {
473 requireStation(first);
474 bool closing = command == "CLOSE";
475 if (words.size() == 2) {
476 if (stations.at(first).closed == closing) fail("station " + first + (closing ? " is already closed" : " is not closed"));
477 stations.at(first).closed = closing;
478 std::cout << command << ' ' << first << '\n';
479 } else {
480 requireStation(words[2]);
481 Segment edge = segment(first, words[2]);
482 std::string label = first + '-' + words[2];
483 std::vector<std::string> containing;
484 if (closing) {
485 for (const auto& [name, line] : lines) {
486 for (std::size_t i = 1; i < line.stops.size(); ++i) {
487 if (segment(line.stops[i - 1], line.stops[i]) == edge) {
488 containing.push_back(name);
489 break;
490 }
491 }
492 }
493 if (containing.empty()) fail("no segment " + label);
494 if (closedSegments.count(edge)) fail("segment " + label + " is already closed");
495 closedSegments.insert(edge);
496 } else {
497 if (!closedSegments.count(edge)) fail("segment " + label + " is not closed");
498 closedSegments.erase(edge);
499 }
500 std::cout << command << ' ' << label;
501 if (closing) {
502 std::cout << " lines";
503 for (const std::string& name : containing) std::cout << ' ' << name;
504 }
505 std::cout << '\n';
506 }
507 } else if (command == "DELAY" || command == "SUSPEND" || command == "RESUME") {
508 requireLine(first);
509 Line& line = lines.at(first);
510 if (command == "DELAY") {
511 int delay = std::stoi(words[2]);
512 if (line.last + delay + line.cumulative.back() > 1440) fail("line " + first + " runs past 24:00");
513 line.delay = delay;
514 std::cout << "DELAY " << first << ' ' << delay << '\n';
515 } else {
516 bool suspending = command == "SUSPEND";
517 if (line.suspended == suspending) fail("line " + first + (suspending ? " is already suspended" : " is not suspended"));
518 line.suspended = suspending;
519 std::cout << command << ' ' << first << '\n';
520 }
521 } else if (command == "VIA") {
522 requireStation(first);
523 requireStation(words[2]);
524 requireStation(words[3]);
525 if (first == words[3]) fail("origin and destination are the same");
526 if (words[2] == first || words[2] == words[3]) fail("via station " + words[2] + " must differ from origin and destination");
527 requireOpen(first);
528 requireOpen(words[2]);
529 requireOpen(words[3]);
530 int departure = timeValue(words[4]);
531 std::string heading = "VIA " + first + ' ' + words[2] + ' ' + words[3];
532 Labels before = earliest(first, departure);
533 if (before.count(words[2])) {
534 Label via = before.at(words[2]);
535 int ready = via.arrival + stations.at(words[2]).transfer;
536 Labels after = earliest(words[2], ready);
537 if (after.count(words[3])) {
538 via.journey.push_back({'S', "", "", words[2], via.arrival, words[2], ready});
539 const Label& final = after.at(words[3]);
540 via.journey.insert(via.journey.end(), final.journey.begin(), final.journey.end());
541 via.arrival = final.arrival;
542 printJourney(heading, departure, via);
543 return;
544 }
545 }
546 std::cout << heading << ' ' << words[4] << " unreachable\n";
547 } else if (command == "ROUTE" || command == "FEWEST" || command == "PARETO" || command == "FARE" ||
548 command == "LASTTRAIN" || command == "ARRIVEBY" || command == "PROFILE") {
549 searchCommand(words);
550 } else if (command == "REACH") {
551 requireStation(first);
552 requireOpen(first);
553 int departure = timeValue(words[2]);
554 int limit = departure + std::stoi(words[3]);
555 Labels labels = earliest(first, departure);
556 std::vector<std::pair<int, std::string>> selected;
557 for (const auto& [name, label] : labels) {
558 if (label.arrival <= limit) selected.emplace_back(label.arrival, name);
559 }
560 std::sort(selected.begin(), selected.end());
561 std::cout << "REACH " << first << ' ' << words[2] << ' ' << words[3] << " count " << selected.size() << '\n';
562 for (const auto& [arrival, name] : selected) {
563 std::cout << name << ' ' << stations.at(name).name << ' ' << clockTime(arrival) << ' ' << arrival - departure
564 << " rides " << legCount(labels.at(name).journey, 'R') << '\n';
565 }
566 } else if (command == "TIMETABLE") {
567 requireStation(first);
568 const Line& line = requireLine(words[2]);
569 if (!line.contains(first)) fail("station " + first + " is not on line " + words[2]);
570 Direction direction = line.direction(directionTo(words[2], line, words[3]));
571 if (first == words[3]) fail("no departures from " + first + " toward " + words[3]);
572 if (line.suspended) fail("line " + words[2] + " is suspended");
573 requireOpen(first);
574 int position = static_cast<int>(std::find(direction.stops.begin(), direction.stops.end(), first) - direction.stops.begin());
575 int train = line.nextTrain(timeValue(words[4]), direction.offsets[position]);
576 std::vector<int> times;
577 if (train >= 0) {
578 for (; train < line.trips() && times.size() < static_cast<std::size_t>(std::stoi(words[5])); ++train) times.push_back(line.at(train, direction.offsets[position]));
579 }
580 std::cout << "TIMETABLE " << first << ' ' << words[2] << " toward " << words[3] << ' ' << words[4] << " next " << times.size();
581 for (int time : times) std::cout << ' ' << clockTime(time);
582 std::cout << '\n';
583 } else if (command == "DEPARTURES" || command == "ARRIVALS") {
584 departures(words, command == "ARRIVALS");
585 } else if (command == "TRIP") {
586 const Line& line = requireLine(first);
587 Direction direction = line.direction(directionTo(first, line, words[2]));
588 if (line.suspended) fail("line " + first + " is suspended");
589 int train = line.nextTrain(timeValue(words[3]), 0);
590 std::cout << "TRIP " << first << " toward " << words[2] << ' ' << words[3];
591 if (train < 0) std::cout << " none\n";
592 else {
593 std::cout << " trip " << train + 1 << " departs " << clockTime(line.at(train, 0)) << '\n';
594 for (std::size_t j = 0; j < direction.stops.size(); ++j) {
595 const std::string& station = direction.stops[j];
596 std::cout << clockTime(line.at(train, direction.offsets[j])) << ' ' << station;
597 if (stations.at(station).closed) std::cout << " closed";
598 if (j > 0 && blocked(direction.stops[j - 1], station)) std::cout << " blocked";
599 std::cout << '\n';
600 }
601 }
602 } else if (command == "SHOWLINE") {
603 const Line& line = requireLine(first);
604 std::cout << "SHOWLINE " << first << (line.suspended ? " suspended" : " running") << " first " << clockTime(line.first)
605 << " last " << clockTime(line.last) << " headway " << line.headway << " delay " << line.delay << " trips " << line.trips() << '\n';
606 for (std::size_t j = 0; j < line.stops.size(); ++j) {
607 const std::string& station = line.stops[j];
608 std::cout << line.cumulative[j] << ' ' << station << " zone " << stations.at(station).zone;
609 if (stations.at(station).closed) std::cout << " closed";
610 if (j > 0 && blocked(line.stops[j - 1], station)) std::cout << " blocked";
611 std::cout << '\n';
612 }
613 } else if (command == "INFO") {
614 const Station& station = requireStation(first);
615 std::cout << "INFO " << first << ' ' << station.name << " zone " << station.zone << " transfer " << station.transfer << (station.closed ? " closed\n" : " open\n");
616 bool any = false;
617 std::cout << "LINES";
618 for (const auto& [name, line] : lines) {
619 if (line.contains(first)) {
620 std::cout << ' ' << name;
621 any = true;
622 }
623 }
624 if (!any) std::cout << " none";
625 std::cout << "\nWALKS";
626 std::map<std::string, int> linked = neighbors(first);
627 if (linked.empty()) std::cout << " none";
628 for (const auto& [name, duration] : linked) std::cout << ' ' << name << ' ' << duration;
629 std::cout << '\n';
630 } else if (command == "STATUS") {
631 std::cout << "STATUS stations " << stations.size() << " lines " << lines.size() << " walks " << walks.size() << '\n';
632 for (const auto& [name, station] : stations) {
633 if (station.closed) std::cout << "CLOSED " << name << '\n';
634 }
635 for (const auto& [a, b] : closedSegments) std::cout << "CLOSED " << a << '-' << b << '\n';
636 for (const auto& [name, line] : lines) {
637 if (line.suspended) std::cout << "SUSPENDED " << name << '\n';
638 }
639 for (const auto& [name, line] : lines) {
640 if (line.delay != 0) std::cout << "DELAYED " << name << ' ' << line.delay << '\n';
641 }
642 }
643 }
644};
