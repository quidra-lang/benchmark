9class MetroLine
10{
11 public int $delay = 0;
12 public bool $suspended = false;
13 public function __construct(public string $id, public int $first, public int $last,
14 public int $headway, public array $stops, public array $offsets) {}
15
16 public function trips(): int
17 {
18 return intdiv($this->last - $this->first, $this->headway) + 1;
19 }
20
21 public function duration(): int
22 {
23 return $this->offsets[count($this->offsets) - 1];
24 }
25
26 public function directions(): array
27 {
28 return [[$this->stops, $this->offsets], [array_reverse($this->stops),
29 array_map(fn($c) => $this->duration() - $c, array_reverse($this->offsets))]];
30 }
31
32 public function nextTrain(int $offset, int $ready): ?int
33 {
34 $difference = max(0, $ready - $this->first - $this->delay - $offset);
35 $q = intdiv($difference + $this->headway - 1, $this->headway);
36 return $q < $this->trips() ? $q : null;
37 }
38
39 public function time(int $q, int $offset): int
40 {
41 return $this->first + $this->delay + $q * $this->headway + $offset;
42 }
43}

127 {

135 private function earliest(string $origin, int $departure): array
136 {
137 $labels = [$origin => [$departure, []]];
138 $settled = [];
139 $lines = $this->runningLines();
140 while (true) {
141 $u = null;
142 foreach ($this->sortedKeys($labels) as $id) {
143 if (!isset($settled[$id]) && ($u === null || $labels[$id][0] < $labels[$u][0])) {
144 $u = $id;
145 }
146 }
147 if ($u === null) {
148 break;
149 }
150 $settled[$u] = true;
151 [$arrival, $journey] = $labels[$u];
152 $ready = $arrival + ($u === $origin ? 0 : $this->stations[$u]->transfer);
153 foreach ($lines as $line) {
154 if (!in_array($u, $line->stops, true)) {
155 continue;
156 }
157 foreach ($line->directions() as [$stops, $offsets]) {
158 $i = array_search($u, $stops, true);
159 if ($i === count($stops) - 1) {
160 continue;
161 }
162 $q = $line->nextTrain($offsets[$i], $ready);
163 if ($q === null) {
164 continue;
165 }
166 $boarding = $line->time($q, $offsets[$i]);
167 for ($j = $i + 1; $j < count($stops); $j++) {
168 if ($this->blocked($stops[$j - 1], $stops[$j])) {
169 break;
170 }
171 if ($this->stations[$stops[$j]]->closed) {
172 continue;
173 }
174 $time = $line->time($q, $offsets[$j]);
175 $leg = ['ride', $line->id, $stops[count($stops) - 1], $u, $boarding, $stops[$j], $time, $j - $i];
176 $this->update($labels, $stops[$j], $time, [...$journey, $leg]);
177 }
178 }
179 }
180 foreach ($this->sortedKeys($this->walks[$u]) as $v) {
181 $time = $arrival + $this->walks[$u][$v];
182 if (!$this->stations[$v]->closed && $time <= 1440) {
183 $this->update($labels, $v, $time, [...$journey, ['walk', $u, $arrival, $v, $time]]);
184 }
185 }
186 }
187 return $labels;
188 }
189
190 private function walkStep(array &$labels, array $selected, array &$marked): void
191 {
192 $snapshot = [];
193 foreach ($this->sortedKeys($selected) as $u) {
194 $snapshot[$u] = $labels[$u];
195 }
196 foreach ($snapshot as $u => [$arrival, $journey]) {
197 foreach ($this->sortedKeys($this->walks[$u]) as $v) {
198 $time = $arrival + $this->walks[$u][$v];
199 if (!$this->stations[$v]->closed && $time <= 1440 &&
200 $this->update($labels, $v, $time, [...$journey, ['walk', $u, $arrival, $v, $time]])) {
201 $marked[$v] = true;
202 }
203 }
204 }
205 }

207 private function rounds(string $origin, int $departure): array
208 {
209 $labels = [$origin => [$departure, []]];
210 $marked = [$origin => true];
211 $this->walkStep($labels, [$origin => true], $marked);
212 $result = [$labels];
213 $lines = $this->runningLines();
214 while ($marked) {
215 $previous = $labels;
216 $previousMarked = $marked;
217 $labels = $previous;
218 $marked = [];
219 $selected = [];
220 foreach ($lines as $line) {
221 foreach ($line->directions() as [$stops, $offsets]) {
222 $boarded = null;
223 $b = null;
224 $boarding = null;
225 $journey = [];
226 foreach ($stops as $j => $u) {
227 if ($j > 0 && $this->blocked($stops[$j - 1], $u)) {
228 $boarded = null;
229 }
230 if ($this->stations[$u]->closed) {
231 continue;
232 }
233 if ($boarded !== null) {
234 $time = $line->time($boarded, $offsets[$j]);
235 $leg = ['ride', $line->id, $stops[count($stops) - 1], $stops[$b], $boarding, $u, $time, $j - $b];
236 if ($this->update($labels, $u, $time, [...$journey, $leg])) {
237 $selected[$u] = true;
238 $marked[$u] = true;
239 }
240 }
241 if ($j === count($stops) - 1 || !isset($previousMarked[$u])) {
242 continue;
243 }
244 [$arrival, $candidateJourney] = $previous[$u];
245 $ready = $arrival + ($u === $origin ? 0 : $this->stations[$u]->transfer);
246 $candidate = $line->nextTrain($offsets[$j], $ready);
247 if ($candidate !== null && ($boarded === null || $candidate < $boarded)) {
248 $boarded = $candidate;
249 $b = $j;
250 $boarding = $line->time($candidate, $offsets[$j]);
251 $journey = $candidateJourney;
252 }
253 }
254 }
255 }
256 $this->walkStep($labels, $selected, $marked);
257 $result[] = $labels;
258 }
259 return $result;
260 }

267 private function printJourney(string $heading, int $departure, array $label): void
268 {
269 [$arrival, $journey] = $label;
270 $rides = $this->rideCount($journey);
271 $walks = count(array_filter($journey, fn($leg) => $leg[0] === 'walk'));
272 echo "$heading ", $this->clock($departure), ' -> ', $this->clock($arrival),
273 ' total ', $arrival - $departure, " rides $rides walks $walks\n";
274 $previousEnd = $departure;
275 foreach ($journey as $leg) {
276 if ($leg[0] === 'ride') {
277 [, $line, $toward, $from, $start, $to, $finish, $count] = $leg;
278 echo "RIDE $line toward $toward $from ", $this->clock($start), " -> $to ", $this->clock($finish),
279 ' wait ', $start - $previousEnd, " stops $count\n";
280 $previousEnd = $finish;
281 } elseif ($leg[0] === 'walk') {
282 [, $from, $start, $to, $finish] = $leg;
283 echo "WALK $from ", $this->clock($start), " -> $to ", $this->clock($finish), "\n";
284 $previousEnd = $finish;
285 } else {
286 [, $station, $start, $finish] = $leg;
287 echo "STAY $station ", $this->clock($start), ' -> ', $this->clock($finish), "\n";
288 $previousEnd = $finish;
289 }
290 }

293 private function printFare(string $origin, string $destination, string $timeText, string $class, array $journey): void
294 {
295 [$discountCap, $window, $prices] = $this->tariff;
296 $half = fn($x) => intdiv($x + 19, 20) * 10;
297 $cap = $class === 'CHILD' ? $half($discountCap) : $discountCap;
298 $tickets = [];
299 $previousEnd = null;
300 foreach ($journey as $leg) {
301 if ($leg[0] !== 'ride') {
302 continue;
303 }
304 [, $lineId, $toward, $from, $boarding, $to, $arrival] = $leg;
305 $line = $this->lines[$lineId];
306 foreach ($line->directions() as [$stops, $_]) {
307 if ($stops[count($stops) - 1] === $toward) {
308 break;
309 }
310 }
311 $start = array_search($from, $stops, true);
312 $finish = array_search($to, $stops, true);
313 $zones = [];
314 for ($i = $start; $i <= $finish; $i++) {
315 $zones[$this->stations[$stops[$i]]->zone] = true;
316 }
317 $zoneCount = count($zones);
318 $price = $prices[min($zoneCount, count($prices)) - 1];
319 if ($class === 'CHILD') {
320 $price = $half($price);
321 }
322 $discount = $previousEnd !== null && $boarding - $previousEnd <= $window ? min($cap, $price) : 0;
323 $tickets[] = [$lineId, $from, $to, $zoneCount, $price, $discount, $price - $discount];
324 $previousEnd = $arrival;
325 }
326 echo "FARE $origin $destination $timeText $class rides ", count($tickets),
327 ' total ', array_sum(array_column($tickets, 6)), "\n";
328 foreach ($tickets as [$line, $from, $to, $zones, $price, $discount, $pay]) {
329 echo "TICKET $line $from $to zones $zones price $price discount $discount pay $pay\n";
330 }

333 private function searchCommand(string $command, array $args): void
334 {
335 [$origin, $destination] = $args;
336 $this->journeyCheck($origin, $destination);
337 $departure = isset($args[2]) ? $this->minutes($args[2]) : null;
338 if ($command === 'FARE') {
339 if (!in_array($args[3], ['ADULT', 'CHILD'], true)) {
340 throw new MetroError("unknown fare class {$args[3]}");
341 }
342 if ($this->tariff === null) {
343 throw new MetroError('no tariff');
344 }
345 } elseif ($command === 'PROFILE' && $departure > $this->minutes($args[3])) {
346 throw new MetroError("empty interval {$args[2]} {$args[3]}");
347 }
348 if ($command === 'ROUTE' || $command === 'FARE') {
349 $label = $this->earliest($origin, $departure)[$destination] ?? null;
350 if ($label === null) {
351 echo "$command ", implode(' ', $args), " unreachable\n";
352 } elseif ($command === 'ROUTE') {
353 $this->printJourney("ROUTE $origin $destination", $departure, $label);
354 } else {
355 $this->printFare($origin, $destination, $args[2], $args[3], $label[1]);
356 }
357 } elseif ($command === 'FEWEST' || $command === 'PARETO') {
358 $all = $this->rounds($origin, $departure);
359 $options = [];
360 foreach ($all as $k => $labels) {
361 if (isset($labels[$destination]) && ($k === 0 || !isset($all[$k - 1][$destination]) ||
362 $labels[$destination][0] < $all[$k - 1][$destination][0])) {
363 $options[] = $k;
364 }
365 }
366 if ($command === 'FEWEST') {
367 if ($options) {
368 $k = $options[0];
369 $this->printJourney("FEWEST $origin $destination trips $k", $departure, $all[$k][$destination]);
370 } else {
371 echo 'FEWEST ', implode(' ', $args), " unreachable\n";
372 }
373 } else {
374 echo 'PARETO ', implode(' ', $args), ' options ', count($options), "\n";
375 foreach ($options as $k) {
376 $this->printJourney("OPTION trips $k", $departure, $all[$k][$destination]);
377 }
378 }
379 } elseif ($command === 'LASTTRAIN' || $command === 'ARRIVEBY') {
380 $start = $command === 'LASTTRAIN' ? 1439 : $departure - 1;
381 $heading = "$command $origin $destination" . ($command === 'ARRIVEBY' ? " by {$args[2]}" : '');
382 for ($time = $start; $time >= 0; $time--) {
383 $label = $this->earliest($origin, $time)[$destination] ?? null;
384 if ($label !== null && ($command === 'LASTTRAIN' || $label[0] <= $departure)) {
385 $this->printJourney($heading, $time, $label);
386 return;
387 }
388 }
389 echo "$heading none\n";
390 } elseif ($command === 'PROFILE') {
391 $finish = $this->minutes($args[3]);
392 $arrivals = [];
393 for ($time = $departure; $time <= $finish; $time++) {
394 $arrivals[$time] = $this->earliest($origin, $time)[$destination][0] ?? null;
395 }
396 $selected = [];
397 for ($time = $departure; $time <= $finish; $time++) {
398 if ($arrivals[$time] !== null && ($time === $finish || $arrivals[$time + 1] === null ||
399 $arrivals[$time] < $arrivals[$time + 1])) {
400 $selected[] = $time;
401 }
402 }
403 echo 'PROFILE ', implode(' ', $args), ' count ', count($selected), "\n";
404 foreach ($selected as $time) {
405 echo $this->clock($time), ' -> ', $this->clock($arrivals[$time]), ' total ', $arrivals[$time] - $time, "\n";
406 }
407 }
408 }

410 public function execute(string $text): void
411 {
412 $args = explode(' ', $text);
413 $command = array_shift($args);
414 $a = $args[0] ?? null;
415 $b = $args[1] ?? null;
416 try {
417 switch ($command) {
418 case 'STATION':
419 if (isset($this->stations[$a])) {
420 throw new MetroError("station $a already exists");
421 }
422 $this->stations[$a] = new Station((int)$b, $args[2]);
423 $this->walks[$a] = [];
424 echo "STATION $a zone $b {$args[2]}\n";
425 break;
426 case 'LINE':
427 if (isset($this->lines[$a])) {
428 throw new MetroError("line $a already exists");
429 }
430 $stops = [];
431 $offsets = [0];
432 for ($i = 4; $i < count($args); $i += 2) {
433 $stops[] = $args[$i];
434 $this->stationCheck($args[$i]);
435 }
436 $seen = [];
437 foreach ($stops as $id) {
438 if (isset($seen[$id])) {
439 throw new MetroError("station $id appears twice on line $a");
440 }
441 $seen[$id] = true;
442 }
443 $first = $this->minutes($b);
444 $last = $this->minutes($args[2]);
445 if ($first > $last) {
446 throw new MetroError("line $a first departure is after last departure");
447 }
448 for ($i = 5; $i < count($args); $i += 2) {
449 $offsets[] = $offsets[count($offsets) - 1] + (int)$args[$i];
450 }
451 if ($last + $offsets[count($offsets) - 1] > 1440) {
452 throw new MetroError("line $a runs past 24:00");
453 }
454 $line = new MetroLine($a, $first, $last, (int)$args[3], $stops, $offsets);
455 $this->lines[$a] = $line;
456 echo "LINE $a stops ", count($stops), ' time ', $line->duration(), ' trips ', $line->trips(), "\n";
457 break;
458 case 'TRANSFER':
459 $this->stationCheck($a);
460 $this->stations[$a]->transfer = (int)$b;
461 echo "TRANSFER $a $b\n";
462 break;
463 case 'WALK':
464 $this->stationCheck($a);
465 $this->stationCheck($b);
466 if ($a === $b) {
467 throw new MetroError("walk from $a to itself");
468 }
469 $old = $this->walks[$a][$b] ?? null;
470 $this->walks[$a][$b] = (int)$args[2];
471 $this->walks[$b][$a] = (int)$args[2];
472 echo 'WALK ', implode(' ', $args), $old !== null ? " was $old" : '', "\n";
473 break;
474 case 'TARIFF':
475 $prices = array_map('intval', array_slice($args, 2));
476 for ($i = 1; $i < count($prices); $i++) {
477 if ($prices[$i] < $prices[$i - 1]) {
478 throw new MetroError('tariff prices must not decrease');
479 }
480 }
481 $this->tariff = [(int)$a, (int)$b, $prices];
482 echo 'TARIFF zones ', count($prices), " discount $a window $b\n";
483 break;
484 case 'CLOSE':
485 case 'OPEN':
486 $this->stationCheck($a);
487 if ($b !== null) {
488 $this->stationCheck($b);
489 $key = $this->segment($a, $b);
490 if ($command === 'CLOSE') {
491 $affected = [];
492 foreach ($this->sortedKeys($this->lines) as $id) {
493 $stops = $this->lines[$id]->stops;
494 for ($i = 1; $i < count($stops); $i++) {
495 if ($this->segment($stops[$i - 1], $stops[$i]) === $key) {
496 $affected[] = $id;
497 break;
498 }
499 }
500 }
501 if (!$affected) {
502 throw new MetroError("no segment $a-$b");
503 }
504 if (isset($this->closed[$key])) {
505 throw new MetroError("segment $a-$b is already closed");
506 }
507 $this->closed[$key] = [$a, $b];
508 echo "CLOSE $a-$b lines ", implode(' ', $affected), "\n";
509 } else {
510 if (!isset($this->closed[$key])) {
511 throw new MetroError("segment $a-$b is not closed");
512 }
513 unset($this->closed[$key]);
514 echo "OPEN $a-$b\n";
515 }
516 } else {
517 $current = $this->stations[$a]->closed;
518 if ($command === 'CLOSE' && $current) {
519 throw new MetroError("station $a is already closed");
520 }
521 if ($command === 'OPEN' && !$current) {
522 throw new MetroError("station $a is not closed");
523 }
524 $this->stations[$a]->closed = $command === 'CLOSE';
525 echo "$command $a\n";
526 }
527 break;
528 case 'DELAY':
529 $this->lineCheck($a);
530 if ($this->lines[$a]->last + (int)$b + $this->lines[$a]->duration() > 1440) {
531 throw new MetroError("line $a runs past 24:00");
532 }
533 $this->lines[$a]->delay = (int)$b;
534 echo "DELAY $a $b\n";
535 break;
536 case 'SUSPEND':
537 case 'RESUME':
538 $this->lineCheck($a);
539 if ($command === 'SUSPEND' && $this->lines[$a]->suspended) {
540 throw new MetroError("line $a is already suspended");
541 }
542 if ($command === 'RESUME' && !$this->lines[$a]->suspended) {
543 throw new MetroError("line $a is not suspended");
544 }
545 $this->lines[$a]->suspended = $command === 'SUSPEND';
546 echo "$command $a\n";
547 break;
548 case 'ROUTE':
549 case 'FEWEST':
550 case 'PARETO':
551 case 'FARE':
552 case 'LASTTRAIN':
553 case 'ARRIVEBY':
554 case 'PROFILE':
555 $this->searchCommand($command, $args);
556 break;
557 case 'VIA':
558 [$origin, $via, $destination, $timeText] = $args;
559 foreach ([$origin, $via, $destination] as $id) {
560 $this->stationCheck($id);
561 }
562 if ($origin === $destination) {
563 throw new MetroError('origin and destination are the same');
564 }
565 if ($via === $origin || $via === $destination) {
566 throw new MetroError("via station $via must differ from origin and destination");
567 }
568 foreach ([$origin, $via, $destination] as $id) {
569 $this->openCheck($id);
570 }
571 $departure = $this->minutes($timeText);
572 $first = $this->earliest($origin, $departure)[$via] ?? null;
573 $second = null;
574 if ($first !== null) {
575 $ready = $first[0] + $this->stations[$via]->transfer;
576 $second = $this->earliest($via, $ready)[$destination] ?? null;
577 }
578 if ($second !== null) {
579 $journey = [...$first[1], ['stay', $via, $first[0], $ready], ...$second[1]];
580 $this->printJourney("VIA $origin $via $destination", $departure, [$second[0], $journey]);
581 } else {
582 echo 'VIA ', implode(' ', $args), " unreachable\n";
583 }
584 break;
585 case 'REACH':
586 $this->stationCheck($a);
587 $this->openCheck($a);
588 $departure = $this->minutes($b);
589 $labels = $this->earliest($a, $departure);
590 $selected = array_keys(array_filter($labels, fn($label) => $label[0] <= $departure + (int)$args[2]));
591 usort($selected, fn($x, $y) => ($labels[$x][0] <=> $labels[$y][0]) ?: strcmp($x, $y));
592 echo 'REACH ', implode(' ', $args), ' count ', count($selected), "\n";
593 foreach ($selected as $id) {
594 [$arrival, $journey] = $labels[$id];
595 echo "$id {$this->stations[$id]->name} ", $this->clock($arrival), ' ', $arrival - $departure,
596 ' rides ', $this->rideCount($journey), "\n";
597 }
598 break;
599 case 'TIMETABLE':
600 $this->stationCheck($a);
601 $this->lineCheck($b);
602 $line = $this->lines[$b];
603 if (!in_array($a, $line->stops, true)) {
604 throw new MetroError("station $a is not on line $b");
605 }
606 $toward = $args[2];
607 $direction = $this->direction($line, $toward);
608 if ($a === $toward) {
609 throw new MetroError("no departures from $a toward $toward");
610 }
611 if ($line->suspended) {
612 throw new MetroError("line $b is suspended");
613 }
614 $this->openCheck($a);
615 [$stops, $offsets] = $direction;
616 $offset = $offsets[array_search($a, $stops, true)];
617 $times = [];
618 for ($q = 0; $q < $line->trips() && count($times) < (int)$args[4]; $q++) {
619 $time = $line->time($q, $offset);
620 if ($time >= $this->minutes($args[3])) {
621 $times[] = $this->clock($time);
622 }
623 }
624 echo "TIMETABLE $a $b toward $toward {$args[3]} next ", count($times),
625 $times ? ' ' . implode(' ', $times) : '', "\n";
626 break;
627 case 'DEPARTURES':
628 case 'ARRIVALS':
629 $this->stationCheck($a);
630 $this->openCheck($a);
631 $entries = [];
632 foreach ($this->runningLines() as $line) {
633 if (!in_array($a, $line->stops, true)) {
634 continue;
635 }
636 foreach ($line->directions() as $direction => [$stops, $offsets]) {
637 $i = array_search($a, $stops, true);
638 if ($command === 'DEPARTURES' ? $i === count($stops) - 1 : $i === 0) {
639 continue;
640 }
641 for ($q = 0; $q < $line->trips(); $q++) {
642 $time = $line->time($q, $offsets[$i]);
643 if ($time >= $this->minutes($b)) {
644 $entries[] = [$time, $line->id, $direction,
645 $command === 'DEPARTURES' ? $stops[count($stops) - 1] : $stops[0]];
646 }
647 }
648 }
649 }
650 usort($entries, fn($x, $y) => ($x[0] <=> $y[0]) ?: (strcmp($x[1], $y[1]) ?: ($x[2] <=> $y[2])));
651 $entries = array_slice($entries, 0, (int)$args[2]);
652 echo "$command $a $b count ", count($entries), "\n";
653 foreach ($entries as [$time, $id, $_, $terminal]) {
654 echo $this->clock($time), " $id ", $command === 'DEPARTURES' ? 'toward' : 'from', " $terminal\n";
655 }
656 break;
657 case 'TRIP':
658 $this->lineCheck($a);
659 $line = $this->lines[$a];
660 [$stops, $offsets] = $this->direction($line, $b);
661 if ($line->suspended) {
662 throw new MetroError("line $a is suspended");
663 }
664 $q = $line->nextTrain(0, $this->minutes($args[2]));
665 if ($q === null) {
666 echo "TRIP $a toward $b {$args[2]} none\n";
667 } else {
668 echo "TRIP $a toward $b {$args[2]} trip ", $q + 1, ' departs ', $this->clock($line->time($q, 0)), "\n";
669 foreach ($stops as $i => $id) {
670 $suffix = $this->stations[$id]->closed ? ' closed' : '';
671 if ($i > 0 && $this->blocked($stops[$i - 1], $id)) {
672 $suffix .= ' blocked';
673 }
674 echo $this->clock($line->time($q, $offsets[$i])), " $id$suffix\n";
675 }
676 }
677 break;
678 case 'SHOWLINE':
679 $this->lineCheck($a);
680 $line = $this->lines[$a];
681 echo "SHOWLINE $a ", $line->suspended ? 'suspended' : 'running', ' first ', $this->clock($line->first),
682 ' last ', $this->clock($line->last), " headway {$line->headway} delay {$line->delay} trips ", $line->trips(), "\n";
683 foreach ($line->stops as $i => $id) {
684 $suffix = $this->stations[$id]->closed ? ' closed' : '';
685 if ($i > 0 && $this->blocked($line->stops[$i - 1], $id)) {
686 $suffix .= ' blocked';
687 }
688 echo "{$line->offsets[$i]} $id zone {$this->stations[$id]->zone}$suffix\n";
689 }
690 break;
691 case 'INFO':
692 $this->stationCheck($a);
693 $station = $this->stations[$a];
694 echo "INFO $a {$station->name} zone {$station->zone} transfer {$station->transfer} ",
695 $station->closed ? 'closed' : 'open', "\n";
696 $lines = [];
697 foreach ($this->sortedKeys($this->lines) as $id) {
698 if (in_array($a, $this->lines[$id]->stops, true)) {
699 $lines[] = $id;
700 }
701 }
702 echo 'LINES ', $lines ? implode(' ', $lines) : 'none', "\n";
703 $walks = [];
704 foreach ($this->sortedKeys($this->walks[$a]) as $v) {
705 $walks[] = "$v {$this->walks[$a][$v]}";
706 }
707 echo 'WALKS ', $walks ? implode(' ', $walks) : 'none', "\n";
708 break;
709 case 'STATUS':
710 echo 'STATUS stations ', count($this->stations), ' lines ', count($this->lines),
711 ' walks ', intdiv(array_sum(array_map('count', $this->walks)), 2), "\n";
712 foreach ($this->sortedKeys($this->stations) as $id) {
713 if ($this->stations[$id]->closed) {
714 echo "CLOSED $id\n";
715 }
716 }
717 $segments = [];
718 foreach ($this->closed as $key => $_) {
719 $segments[] = explode('-', $key);
720 }
721 usort($segments, fn($x, $y) => strcmp($x[0], $y[0]) ?: strcmp($x[1], $y[1]));
722 foreach ($segments as [$x, $y]) {
723 echo "CLOSED $x-$y\n";
724 }
725 foreach ($this->sortedKeys($this->lines) as $id) {
726 if ($this->lines[$id]->suspended) {
727 echo "SUSPENDED $id\n";
728 }
729 }
730 foreach ($this->sortedKeys($this->lines) as $id) {
731 if ($this->lines[$id]->delay !== 0) {
732 echo "DELAYED $id {$this->lines[$id]->delay}\n";
733 }
734 }
735 break;
736 }
737 } catch (MetroError $error) {
738 echo 'ERROR ', $error->getMessage(), "\n";
739 }
