24 static class Line {
25 final String id;
26 final int first, last, headway;
27 final List<Station> stops;
28 final int[] cumulative;
29 int delay;
30 boolean suspended;
31 Line(String id, int first, int last, int headway, List<Station> stops, int[] cumulative) {
32 this.id = id;
33 this.first = first;
34 this.last = last;
35 this.headway = headway;
36 this.stops = List.copyOf(stops);
37 this.cumulative = cumulative.clone();
38 }
39 int total() {
40 return cumulative[cumulative.length - 1];
41 }
42 int trips() {
43 return (last - first) / headway + 1;
44 }
45 Station stop(int direction, int index) {
46 return stops.get(direction == 0 ? index : stops.size() - 1 - index);
47 }
48 int offset(int direction, int index) {
49 return direction == 0 ? cumulative[index] : total() - cumulative[stops.size() - 1 - index];
50 }
51 int position(int direction, Station station) {
52 int index = stops.indexOf(station);
53 return index < 0 ? -1 : direction == 0 ? index : stops.size() - 1 - index;
54 }
55 int time(int direction, int index, int train) {
56 return first + delay + train * headway + offset(direction, index);
57 }
58 int next(int direction, int index, int ready) {
59 int start = time(direction, index, 0);
60 int train = ready <= start ? 0 : (ready - start + headway - 1) / headway;
61 return train < trips() ? train : -1;
62 }
63 }

123 static Map<Station, Label> earliest(Station origin, int departure) {
124 Map<Station, Label> labels = new HashMap<>();
125 Set<Station> settled = new HashSet<>();
126 labels.put(origin, new Label(departure, List.of()));
127 while (true) {
128 Station selected = null;
129 for (Station station : stations.values()) if (!settled.contains(station) && labels.containsKey(station) && (selected == null || labels.get(station).time() < labels.get(selected).time())) selected = station;
130 if (selected == null) return labels;
131 Station u = selected;
132 settled.add(u);
133 Label label = labels.get(u);
134 int ready = label.time() + (u == origin ? 0 : u.transfer);
135 for (Line line : lines.values()) if (!line.suspended && line.stops.contains(u)) for (int direction = 0; direction < 2; direction++) {
136 int i = line.position(direction, u), last = line.stops.size() - 1;
137 if (i == last) continue;
138 int train = line.next(direction, i, ready);
139 if (train < 0) continue;
140 int start = line.time(direction, i, train);
141 for (int j = i + 1; j <= last; j++) {
142 Station v = line.stop(direction, j);
143 if (blocked(line.stop(direction, j - 1), v)) break;
144 if (v.closed) continue;
145 int arrival = line.time(direction, j, train);
146 relax(labels, v, arrival, label.journey(), new Leg('R', line, line.stop(direction, last), u, start, v, arrival, j - i));
147 }
148 }
149 for (var walk : u.walks.entrySet()) {
150 Station v = stations.get(walk.getKey());
151 int arrival = label.time() + walk.getValue();
152 if (!v.closed && arrival <= 1440) relax(labels, v, arrival, label.journey(), new Leg('W', null, null, u, label.time(), v, arrival, 0));
153 }
154 }
155 }

157 static void walkStep(Map<Station, Label> labels, Set<Station> origins, Set<Station> marked) {
158 Map<Station, Label> recorded = new HashMap<>();
159 for (Station origin : origins) recorded.put(origin, labels.get(origin));
160 for (Station u : stations.values()) if (origins.contains(u)) for (var walk : u.walks.entrySet()) {
161 Label label = recorded.get(u);
162 Station v = stations.get(walk.getKey());
163 int arrival = label.time() + walk.getValue();
164 if (!v.closed && arrival <= 1440 && (!labels.containsKey(v) || labels.get(v).time() > arrival)) {
165 labels.put(v, new Label(arrival, append(label.journey(), new Leg('W', null, null, u, label.time(), v, arrival, 0))));
166 marked.add(v);
167 }
168 }

171 static List<Map<Station, Label>> rounds(Station origin, int departure) {
172 List<Map<Station, Label>> result = new ArrayList<>();
173 Map<Station, Label> previous = new HashMap<>();
174 previous.put(origin, new Label(departure, List.of()));
175 Set<Station> marked = new HashSet<>();
176 marked.add(origin);
177 walkStep(previous, Set.of(origin), marked);
178 result.add(previous);
179 while (!marked.isEmpty()) {
180 Map<Station, Label> current = new HashMap<>(previous);
181 Set<Station> updated = new HashSet<>(), walks = new HashSet<>();
182 for (Line line : lines.values()) if (!line.suspended) for (int direction = 0; direction < 2; direction++) {
183 int boarded = -1, boardIndex = -1, boardTime = 0, last = line.stops.size() - 1;
184 List<Leg> boardJourney = null;
185 for (int j = 0; j <= last; j++) {
186 Station u = line.stop(direction, j);
187 if (j > 0 && blocked(line.stop(direction, j - 1), u)) boarded = -1;
188 if (u.closed) continue;
189 if (boarded >= 0) {
190 int arrival = line.time(direction, j, boarded);
191 if (!current.containsKey(u) || current.get(u).time() > arrival) {
192 current.put(u, new Label(arrival, append(boardJourney, new Leg('R', line, line.stop(direction, last), line.stop(direction, boardIndex), boardTime, u, arrival, j - boardIndex))));
193 walks.add(u);
194 updated.add(u);
195 }
196 }
197 if (j < last && marked.contains(u)) {
198 Label label = previous.get(u);
199 int ready = label.time() + (u == origin ? 0 : u.transfer), train = line.next(direction, j, ready);
200 if (train >= 0 && (boarded < 0 || train < boarded)) {
201 boarded = train;
202 boardIndex = j;
203 boardTime = line.time(direction, j, train);
204 boardJourney = label.journey();
205 }
206 }
207 }
208 }
209 walkStep(current, walks, updated);
210 result.add(current);
211 previous = current;
212 marked = updated;
213 }
214 return result;
215 }

248 static void fare(String heading, Station origin, Station destination, int departure, String fareClass) {
249 Label label = earliest(origin, departure).get(destination);
250 if (label == null) {
251 System.out.println(heading + " unreachable");
252 return;
253 }
254 List<String> tickets = new ArrayList<>();
255 int total = 0, lastRide = -1;
256 boolean child = fareClass.equals("CHILD");
257 for (Leg leg : label.journey()) if (leg.kind() == 'R') {
258 Line line = leg.line();
259 int a = line.stops.indexOf(leg.from()), b = line.stops.indexOf(leg.to());
260 Set<Integer> zones = new HashSet<>();
261 for (int i = Math.min(a, b); i <= Math.max(a, b); i++) zones.add(line.stops.get(i).zone);
262 int price = prices[Math.min(zones.size(), prices.length) - 1];
263 if (child) price = half(price);
264 int reduction = lastRide >= 0 && leg.departure() - lastRide <= window ? Math.min(price, child ? half(discount) : discount) : 0;
265 int pay = price - reduction;
266 total += pay;
267 tickets.add("TICKET " + line.id + " " + leg.from().id + " " + leg.to().id + " zones " + zones.size() + " price " + price + " discount " + reduction + " pay " + pay);
268 lastRide = leg.arrival();
269 }
270 System.out.println(heading + " rides " + rides(label.journey()) + " total " + total);
271 tickets.forEach(System.out::println);
272 }

379 int departure = time(w[4]);

510 Map<Integer, Integer> arrivals = new TreeMap<>();
511 for (int t = departure; t <= last; t++) {
512 Label label = earliest(origin, t).get(destination);
513 if (label != null) arrivals.put(t, label.time());
514 }
515 List<Integer> selected = new ArrayList<>();
516 for (int t : arrivals.keySet()) if (t == last || !arrivals.containsKey(t + 1) || arrivals.get(t) < arrivals.get(t + 1)) selected.add(t);
517 System.out.println(heading + " " + w[3] + " " + w[4] + " count " + selected.size());
518 for (int t : selected) System.out.println(clock(t) + " -> " + clock(arrivals.get(t)) + " total " + (arrivals.get(t) - t));
519 return;
520 }
521 if (command.equals("FEWEST") || command.equals("PARETO")) {
522 var rounds = rounds(origin, departure);
523 List<Integer> options = new ArrayList<>();
524 for (int k = 0; k < rounds.size(); k++) {
525 Label label = rounds.get(k).get(destination), previous = k == 0 ? null : rounds.get(k - 1).get(destination);
526 if (label != null && (previous == null || previous.time() > label.time())) options.add(k);
527 }
528 if (command.equals("FEWEST")) {
529 if (options.isEmpty()) System.out.println(heading + " " + w[3] + " unreachable");
530 else {
531 int k = options.getFirst();
532 printJourney(heading + " trips " + k, departure, rounds.get(k).get(destination));
533 }
534 } else {
535 System.out.println(heading + " " + w[3] + " options " + options.size());
536 for (int k : options) printJourney("OPTION trips " + k, departure, rounds.get(k).get(destination));
537 }
538 return;
539 }
540 Label label = earliest(origin, departure).get(destination);
541 if (label == null) System.out.println(heading + " " + w[3] + " unreachable");
542 else printJourney(heading, departure, label);
543 }
