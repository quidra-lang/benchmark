101static char *format(const char *pattern, ...) {
102 va_list args, saved;
103 va_start(args, pattern);
104 va_copy(saved, args);
105 int n = vsnprintf(NULL, 0, pattern, args);
106 va_end(args);
107 char *out = allocate((size_t)n + 1);
108 vsnprintf(out, (size_t)n + 1, pattern, saved);
109 va_end(saved);
110 return out;
111}

113static void error(const char *pattern, ...) {
114 va_list args, saved;
115 va_start(args, pattern);
116 va_copy(saved, args);
117 int n = vsnprintf(NULL, 0, pattern, args);
118 va_end(args);
119 error_reason = allocate((size_t)n + 1);
120 vsnprintf(error_reason, (size_t)n + 1, pattern, saved);
121 va_end(saved);
122 longjmp(failure, 1);
123}

125static void append(Buffer *b, const char *s) {
126 size_t n = strlen(s);
127 if (b->length + n + 1 > b->capacity) {
128 b->capacity = (b->length + n + 1) * 2;
129 b->data = resize(b->data, b->capacity);
130 }
131 memcpy(b->data + b->length, s, n);
132 b->length += n;
133 b->data[b->length] = 0;
134}

190static bool blocked(Station *a, Station *b) {

210static Station *stop(Line *line, int direction, int index) {
211 return line->stops[direction ? line->count - 1 - (size_t)index : (size_t)index];
212}

214static int offset(Line *line, int direction, int index) {
215 return direction ? total(line) - line->cumulative[line->count - 1 - (size_t)index] : line->cumulative[index];
216}

218static int position(Line *line, int direction, Station *station) {
219 for (size_t i = 0; i < line->count; i++) if (line->stops[i] == station) return direction ? (int)(line->count - 1 - i) : (int)i;
220 return -1;
221}

223static int train_time(Line *line, int direction, int index, int train) {
224 return line->first + line->delay + train * line->headway + offset(line, direction, index);
225}

227static int next_train(Line *line, int direction, int index, int ready) {
228 int start = train_time(line, direction, index, 0), train = ready <= start ? 0 : (ready - start + line->headway - 1) / line->headway;
229 return train < trips(line) ? train : -1;
230}

234 return !strcmp(line->stops[line->count - 1]->id, terminal) ? 0 : 1;
235}

271static Search earliest(Station *origin, int departure) {
272 Search result = {.labels = allocate(station_count * sizeof(*result.labels))};
273 bool *settled = allocate(station_count * sizeof(*settled));
274 result.labels[origin->index] = (Label){true, departure, NULL};
275 while (true) {
276 Station *selected = NULL;
277 for (size_t i = 0; i < station_count; i++) {
278 Station *s = stations[i];
279 if (!settled[s->index] && result.labels[s->index].present && (!selected || result.labels[s->index].time < result.labels[selected->index].time)) selected = s;
280 }
281 if (!selected) break;
282 Station *u = selected;
283 settled[u->index] = true;
284 Label label = result.labels[u->index];
285 int ready = label.time + (u == origin ? 0 : u->transfer);
286 for (size_t l = 0; l < line_count; l++) {
287 Line *current = lines[l];
288 if (current->suspended || position(current, 0, u) < 0) continue;
289 for (int d = 0; d < 2; d++) {
290 int i = position(current, d, u), last = (int)current->count - 1;
291 if (i == last) continue;
292 int train = next_train(current, d, i, ready);
293 if (train < 0) continue;
294 int start = train_time(current, d, i, train);
295 for (int j = i + 1; j <= last; j++) {
296 Station *v = stop(current, d, j);
297 if (blocked(stop(current, d, j - 1), v)) break;
298 if (v->closed) continue;
299 int arrival = train_time(current, d, j, train);
300 relax(result.labels, &result.allocated, v, arrival, label.journey, (Leg){'R', current, stop(current, d, last), u, v, start, arrival, j - i});
301 }
302 }
303 }
304 for (size_t i = 0; i < u->walk_count; i++) {
305 Walk walk = u->walks[i];
306 int arrival = label.time + walk.minutes;
307 if (!walk.neighbor->closed && arrival <= 1440) relax(result.labels, &result.allocated, walk.neighbor, arrival, label.journey, (Leg){'W', NULL, NULL, u, walk.neighbor, label.time, arrival, 0});
308 }
309 }
310 free(settled);
311 return result;
312}

314static void walk_step(Label *labels, bool *origins, bool *marked, Node **allocated) {
315 Label *recorded = copy_labels(labels);
316 for (size_t i = 0; i < station_count; i++) {
317 Station *u = stations[i];
318 if (!origins[u->index]) continue;
319 Label label = recorded[u->index];
320 for (size_t j = 0; j < u->walk_count; j++) {
321 Walk walk = u->walks[j];
322 int arrival = label.time + walk.minutes;
323 if (!walk.neighbor->closed && arrival <= 1440 && better(labels, walk.neighbor, arrival)) {
324 relax(labels, allocated, walk.neighbor, arrival, label.journey, (Leg){'W', NULL, NULL, u, walk.neighbor, label.time, arrival, 0});
325 marked[walk.neighbor->index] = true;
326 }
327 }
328 }
329 free(recorded);
330}

337static Rounds rounds(Station *origin, int departure) {
338 Rounds result = {0};
339 Label *previous = allocate(station_count * sizeof(*previous));
340 previous[origin->index] = (Label){true, departure, NULL};
341 bool *marked = allocate(station_count * sizeof(*marked)), *initial = allocate(station_count * sizeof(*initial));
342 marked[origin->index] = initial[origin->index] = true;
343 walk_step(previous, initial, marked, &result.allocated);
344 free(initial);
345 APPEND(result.labels, result.count, previous);
346 while (any_marked(marked)) {
347 Label *current = copy_labels(previous);
348 bool *updated = allocate(station_count * sizeof(*updated)), *walks = allocate(station_count * sizeof(*walks));
349 for (size_t l = 0; l < line_count; l++) {
350 Line *line = lines[l];
351 if (line->suspended) continue;
352 for (int d = 0; d < 2; d++) {
353 int boarded = -1, board_index = -1, board_time = 0, last = (int)line->count - 1;
354 Node *board_journey = NULL;
355 for (int j = 0; j <= last; j++) {
356 Station *u = stop(line, d, j);
357 if (j && blocked(stop(line, d, j - 1), u)) boarded = -1;
358 if (u->closed) continue;
359 if (boarded >= 0) {
360 int arrival = train_time(line, d, j, boarded);
361 if (better(current, u, arrival)) {
362 relax(current, &result.allocated, u, arrival, board_journey, (Leg){'R', line, stop(line, d, last), stop(line, d, board_index), u, board_time, arrival, j - board_index});
363 walks[u->index] = updated[u->index] = true;
364 }
365 }
366 if (j < last && marked[u->index]) {
367 Label label = previous[u->index];
368 int ready = label.time + (u == origin ? 0 : u->transfer), train = next_train(line, d, j, ready);
369 if (train >= 0 && (boarded < 0 || train < boarded)) {
370 boarded = train;
371 board_index = j;
372 board_time = train_time(line, d, j, train);
373 board_journey = label.journey;
374 }
375 }
376 }
377 }
378 }
379 walk_step(current, walks, updated, &result.allocated);
380 free(walks);
381 free(marked);
382 marked = updated;
383 APPEND(result.labels, result.count, current);
384 previous = current;
385 }
386 free(marked);
387 return result;
388}

392 free(result.labels);
393 free_nodes(result.allocated);

397 int count = 0;
398 for (; journey; journey = journey->previous) if (journey->leg.kind == 'R') count++;
399 return count;

402static Leg *journey_legs(Node *journey, size_t *count) {
403 *count = 0;
404 for (Node *node = journey; node; node = node->previous) (*count)++;
405 Leg *result = allocate(*count * sizeof(*result));
406 size_t at = *count;
407 for (; journey; journey = journey->previous) result[--at] = journey->leg;
408 return result;
409}

412 size_t count;
413 Leg *legs = journey_legs(label.journey, &count);
414 int walks = 0;
415 for (size_t i = 0; i < count; i++) if (legs[i].kind == 'W') walks++;
416 printf("%s %s -> %s total %d rides %d walks %d\n", heading, clock_text(departure), clock_text(label.time), label.time - departure, rides(label.journey), walks);
417 int previous = departure;
418 for (size_t i = 0; i < count; i++) {
419 Leg leg = legs[i];
420 if (leg.kind == 'R') printf("RIDE %s toward %s %s %s -> %s %s wait %d stops %d\n", leg.line->id, leg.terminal->id, leg.from->id, clock_text(leg.departure), leg.to->id, clock_text(leg.arrival), leg.departure - previous, leg.stops);

438static void fare(const char *heading, Station *origin, Station *destination, int departure, bool child) {
439 Search result = earliest(origin, departure);
440 Label label = result.labels[destination->index];
441 if (!label.present) {
442 printf("%s unreachable\n", heading);
443 free_search(result);
444 return;
445 }
446 Strings tickets = {0};
447 size_t count;
448 Leg *legs = journey_legs(label.journey, &count);
449 int total_payment = 0, last_ride = -1;
450 for (size_t i = 0; i < count; i++) {
451 Leg leg = legs[i];
452 if (leg.kind != 'R') continue;
453 int a = position(leg.line, 0, leg.from), b = position(leg.line, 0, leg.to), first = a < b ? a : b, last = a > b ? a : b;
454 bool seen[10] = {0};
455 int zones = 0;
456 for (int j = first; j <= last; j++) {
457 int zone = leg.line->stops[j]->zone;
458 if (!seen[zone]) zones++;
459 seen[zone] = true;
460 }
461 int price = prices[(size_t)zones <= price_count ? (size_t)zones - 1 : price_count - 1];
462 if (child) price = half(price);
463 int reduction = 0, cap = child ? half(discount) : discount;
464 if (last_ride >= 0 && leg.departure - last_ride <= window) reduction = cap < price ? cap : price;
465 int pay = price - reduction;
466 total_payment += pay;
467 APPEND(tickets.items, tickets.count, format("TICKET %s %s %s zones %d price %d discount %d pay %d", leg.line->id, leg.from->id, leg.to->id, zones, price, reduction, pay));
468 last_ride = leg.arrival;
469 }
470 printf("%s rides %d total %d\n", heading, rides(label.journey), total_payment);
471 for (size_t i = 0; i < tickets.count; i++) puts(tickets.items[i]);
472 free(legs);
473 free_search(result);
474}

476static void set_walk(Station *a, Station *b, int minutes) {
477 for (size_t i = 0; i < a->walk_count; i++) if (a->walks[i].neighbor == b) {
478 a->walks[i].minutes = minutes;
479 return;
480 }
481 APPEND(a->walks, a->walk_count, ((Walk){b, minutes}));
482 qsort(a->walks, a->walk_count, sizeof(*a->walks), walk_compare);
483}

493 Strings result = {0};
494 size_t start = 0;
495 for (size_t i = 0;; i++) if (!text[i] || text[i] == ' ') {
496 APPEND(result.items, result.count, slice(text + start, i - start));
497 start = i + 1;

499 }

503static void command(char **w, size_t argc) {
504 char *cmd = w[0];
505 if (!strcmp(cmd, "STATION")) {
506 if (find_station(w[1])) error("station %s already exists", w[1]);
507 Station *added = allocate(sizeof(*added));
508 added->id = w[1];
509 added->name = w[3];
510 added->zone = atoi(w[2]);
511 added->index = station_count;
512 APPEND(stations, station_count, added);
513 qsort(stations, station_count, sizeof(*stations), station_compare);
514 printf("STATION %s zone %s %s\n", w[1], w[2], w[3]);
515 return;
516 }
517 if (!strcmp(cmd, "LINE")) {
518 if (find_line(w[1])) error("line %s already exists", w[1]);
519 Station **stops = NULL;
520 size_t count = 0;
521 for (size_t i = 5; i < argc; i += 2) APPEND(stops, count, station(w[i]));
522 for (size_t i = 0; i < count; i++) for (size_t j = 0; j < i; j++) if (stops[i] == stops[j]) error("station %s appears twice on line %s", stops[i]->id, w[1]);
523 int first = time_value(w[2]), last = time_value(w[3]);
524 if (first > last) error("line %s first departure is after last departure", w[1]);
525 int *cumulative = allocate(count * sizeof(*cumulative));
526 for (size_t i = 1; i < count; i++) cumulative[i] = cumulative[i - 1] + atoi(w[4 + 2 * i]);
527 if (last + cumulative[count - 1] > 1440) error("line %s runs past 24:00", w[1]);
528 Line *added = allocate(sizeof(*added));
529 *added = (Line){.id = w[1], .first = first, .last = last, .headway = atoi(w[4]), .stops = stops, .cumulative = cumulative, .count = count};
530 APPEND(lines, line_count, added);
531 qsort(lines, line_count, sizeof(*lines), line_compare);
532 printf("LINE %s stops %zu time %d trips %d\n", added->id, count, total(added), trips(added));
533 return;
534 }
535 if (!strcmp(cmd, "TRANSFER")) {
536 Station *s = station(w[1]);
537 s->transfer = atoi(w[2]);
538 printf("TRANSFER %s %s\n", w[1], w[2]);
539 return;
540 }
541 if (!strcmp(cmd, "WALK")) {
542 Station *a = station(w[1]), *b = station(w[2]);
543 if (a == b) error("walk from %s to itself", a->id);
544 int before = -1;
545 for (size_t i = 0; i < a->walk_count; i++) if (a->walks[i].neighbor == b) before = a->walks[i].minutes;
546 int minutes = atoi(w[3]);
547 set_walk(a, b, minutes);
548 set_walk(b, a, minutes);
549 printf("WALK %s %s %d", a->id, b->id, minutes);
550 if (before >= 0) printf(" was %d", before);
551 putchar('\n');
552 return;
553 }
554 if (!strcmp(cmd, "TARIFF")) {
555 size_t count = argc - 3;
556 int *added = allocate(count * sizeof(*added));
557 for (size_t i = 0; i < count; i++) {
558 added[i] = atoi(w[i + 3]);
559 if (i && added[i] < added[i - 1]) error("tariff prices must not decrease");
560 }
561 prices = added;
562 price_count = count;
563 discount = atoi(w[1]);
564 window = atoi(w[2]);
565 printf("TARIFF zones %zu discount %d window %d\n", count, discount, window);
566 return;
567 }
568 if (!strcmp(cmd, "CLOSE") || !strcmp(cmd, "OPEN")) {
569 Station *a = station(w[1]);
570 bool close = !strcmp(cmd, "CLOSE");
571 if (argc == 2) {
572 if (close == a->closed) error("station %s is %sclosed", a->id, close ? "already " : "not ");
573 a->closed = close;
574 printf("%s %s\n", cmd, a->id);
575 } else {
576 Station *b = station(w[2]);
577 Segment key = segment(a, b);
578 Strings containing = {0};
579 if (close) {
580 for (size_t l = 0; l < line_count; l++) for (size_t i = 1; i < lines[l]->count; i++) if (same_segment(segment(lines[l]->stops[i - 1], lines[l]->stops[i]), key)) {
581 APPEND(containing.items, containing.count, lines[l]->id);
582 break;
583 }
584 if (!containing.count) error("no segment %s-%s", a->id, b->id);
585 }
586 if (close == blocked(a, b)) error("segment %s-%s is %sclosed", a->id, b->id, close ? "already " : "not ");
587 if (close) APPEND(closed_segments, closed_count, key);
588 else {
589 size_t next = 0;
590 for (size_t i = 0; i < closed_count; i++) if (!same_segment(closed_segments[i], key)) closed_segments[next++] = closed_segments[i];
591 closed_count = next;
592 }
593 printf("%s %s-%s", cmd, a->id, b->id);
594 if (close) {
595 printf(" lines");
596 for (size_t i = 0; i < containing.count; i++) printf(" %s", containing.items[i]);
597 }
598 putchar('\n');
599 }
600 return;
601 }
602 if (!strcmp(cmd, "DELAY")) {
603 Line *current = line(w[1]);

611 Line *current = line(w[1]);
612 bool suspend = !strcmp(cmd, "SUSPEND");
613 if (suspend == current->suspended) error("line %s is %ssuspended", current->id, suspend ? "already " : "not ");
614 current->suspended = suspend;
615 printf("%s %s\n", cmd, current->id);
616 return;

618 if (!strcmp(cmd, "VIA")) {
619 Station *origin = station(w[1]), *via = station(w[2]), *destination = station(w[3]);
620 if (origin == destination) error("origin and destination are the same");
621 if (via == origin || via == destination) error("via station %s must differ from origin and destination", via->id);
622 if (origin->closed) error("station %s is closed", origin->id);
623 if (via->closed) error("station %s is closed", via->id);
624 if (destination->closed) error("station %s is closed", destination->id);
625 int departure = time_value(w[4]);
626 char *heading = format("VIA %s %s %s", origin->id, via->id, destination->id);
627 Search first = earliest(origin, departure);
628 Label first_label = first.labels[via->index];
629 bool found = false;
630 if (first_label.present) {
631 int ready = first_label.time + via->transfer;
632 Search second = earliest(via, ready);
633 Label second_label = second.labels[destination->index];
634 if (second_label.present) {
635 Node *allocated = NULL, *journey = add_leg(&allocated, first_label.journey, (Leg){'S', NULL, NULL, via, via, first_label.time, ready, 0});
636 size_t count;
637 Leg *legs = journey_legs(second_label.journey, &count);
638 for (size_t i = 0; i < count; i++) journey = add_leg(&allocated, journey, legs[i]);
639 print_journey(heading, departure, (Label){true, second_label.time, journey});
640 free(legs);
641 free_nodes(allocated);
642 found = true;
643 }
644 free_search(second);
645 }
646 if (!found) printf("%s %s unreachable\n", heading, clock_text(departure));
647 free_search(first);
648 return;
649 }
650 if (!strcmp(cmd, "REACH")) {
651 Station *origin = station(w[1]);
652 if (origin->closed) error("station %s is closed", origin->id);
653 int departure = time_value(w[2]), limit = departure + atoi(w[3]);
654 Search result = earliest(origin, departure);
655 Station **selected = NULL;
656 size_t count = 0;
657 for (size_t i = 0; i < station_count; i++) {
658 Station *s = stations[i];
659 if (result.labels[s->index].present && result.labels[s->index].time <= limit) {
660 APPEND(selected, count, s);
661 size_t j = count - 1;
662 while (j && result.labels[selected[j - 1]->index].time > result.labels[s->index].time) {
663 selected[j] = selected[j - 1];
664 j--;
665 }
666 selected[j] = s;
667 }
668 }
669 printf("REACH %s %s %s count %zu\n", origin->id, w[2], w[3], count);
670 for (size_t i = 0; i < count; i++) {
671 Station *s = selected[i];
672 Label label = result.labels[s->index];
673 printf("%s %s %s %d rides %d\n", s->id, s->name, clock_text(label.time), label.time - departure, rides(label.journey));
674 }
675 free(selected);
676 free_search(result);
677 return;
678 }
679 if (!strcmp(cmd, "TIMETABLE")) {
680 Station *s = station(w[1]);
681 Line *current = line(w[2]);
682 int pos = position(current, 0, s);
683 if (pos < 0) error("station %s is not on line %s", s->id, current->id);
684 int d = direction(current, w[3]);
685 if (!strcmp(s->id, w[3])) error("no departures from %s toward %s", s->id, w[3]);
686 if (current->suspended) error("line %s is suspended", current->id);
687 if (s->closed) error("station %s is closed", s->id);
688 pos = position(current, d, s);
689 int start = next_train(current, d, pos, time_value(w[4])), count = start < 0 ? 0 : trips(current) - start;
690 if (count > atoi(w[5])) count = atoi(w[5]);
691 printf("TIMETABLE %s %s toward %s %s next %d", s->id, current->id, w[3], w[4], count);
692 for (int q = start; q < start + count; q++) printf(" %s", clock_text(train_time(current, d, pos, q)));
693 putchar('\n');
694 return;
695 }
696 if (!strcmp(cmd, "DEPARTURES") || !strcmp(cmd, "ARRIVALS")) {
697 Station *s = station(w[1]);
698 if (s->closed) error("station %s is closed", s->id);
699 bool arrivals = !strcmp(cmd, "ARRIVALS");
700 Departure *selected = NULL;
701 size_t count = 0;
702 for (size_t l = 0; l < line_count; l++) {

704 if (current->suspended || position(current, 0, s) < 0) continue;
705 for (int d = 0; d < 2; d++) {
706 int pos = position(current, d, s);
707 if (arrivals ? pos == 0 : pos == (int)current->count - 1) continue;
708 int start = next_train(current, d, pos, time_value(w[2]));
709 if (start >= 0) for (int q = start; q < trips(current); q++) APPEND(selected, count, ((Departure){train_time(current, d, pos, q), d, current}));
710 }
711 }
712 if (count) qsort(selected, count, sizeof(*selected), departure_compare);
713 if (count > (size_t)atoi(w[3])) count = (size_t)atoi(w[3]);
714 printf("%s %s %s count %zu\n", cmd, s->id, w[2], count);
715 for (size_t i = 0; i < count; i++) {
716 Departure entry = selected[i];
717 printf("%s %s %s %s\n", clock_text(entry.time), entry.line->id, arrivals ? "from" : "toward", stop(entry.line, entry.direction, arrivals ? 0 : (int)entry.line->count - 1)->id);
718 }
719 free(selected);
720 return;
721 }
722 if (!strcmp(cmd, "TRIP")) {
723 Line *current = line(w[1]);
724 int d = direction(current, w[2]);
725 if (current->suspended) error("line %s is suspended", current->id);
726 int train = next_train(current, d, 0, time_value(w[3]));
727 char *heading = format("TRIP %s toward %s %s", current->id, w[2], w[3]);
728 if (train < 0) printf("%s none\n", heading);

730 printf("%s trip %d departs %s\n", heading, train + 1, clock_text(train_time(current, d, 0, train)));
731 for (int j = 0; j < (int)current->count; j++) {
732 Station *s = stop(current, d, j);
733 printf("%s %s%s%s\n", clock_text(train_time(current, d, j, train)), s->id, s->closed ? " closed" : "", j && blocked(stop(current, d, j - 1), s) ? " blocked" : "");
734 }
735 }

737 }
738 if (!strcmp(cmd, "SHOWLINE")) {
739 Line *current = line(w[1]);
740 printf("SHOWLINE %s %s first %s last %s headway %d delay %d trips %d\n", current->id, current->suspended ? "suspended" : "running", clock_text(current->first), clock_text(current->last), current->headway, current->delay, trips(current));
741 for (size_t j = 0; j < current->count; j++) {
742 Station *s = current->stops[j];
743 printf("%d %s zone %d%s%s\n", current->cumulative[j], s->id, s->zone, s->closed ? " closed" : "", j && blocked(current->stops[j - 1], s) ? " blocked" : "");
744 }

747 if (!strcmp(cmd, "INFO")) {
748 Station *s = station(w[1]);
749 printf("INFO %s %s zone %d transfer %d %s\nLINES", s->id, s->name, s->zone, s->transfer, s->closed ? "closed" : "open");
750 bool any = false;

763 size_t walks = 0;

765 printf("STATUS stations %zu lines %zu walks %zu\n", station_count, line_count, walks / 2);
766 for (size_t i = 0; i < station_count; i++) if (stations[i]->closed) printf("CLOSED %s\n", stations[i]->id);
767 if (closed_count) qsort(closed_segments, closed_count, sizeof(*closed_segments), segment_compare);
768 for (size_t i = 0; i < closed_count; i++) printf("CLOSED %s-%s\n", closed_segments[i].a->id, closed_segments[i].b->id);
769 for (size_t i = 0; i < line_count; i++) if (lines[i]->suspended) printf("SUSPENDED %s\n", lines[i]->id);
770 for (size_t i = 0; i < line_count; i++) if (lines[i]->delay) printf("DELAYED %s %d\n", lines[i]->id, lines[i]->delay);

773 Station *origin = station(w[1]), *destination = station(w[2]);
774 search_check(origin, destination);

778 if (!prices) error("no tariff");

782 if (!strcmp(cmd, "LASTTRAIN") || !strcmp(cmd, "ARRIVEBY")) {
783 int deadline = !strcmp(cmd, "LASTTRAIN") ? 1440 : time_value(w[3]);
784 if (!strcmp(cmd, "ARRIVEBY")) heading = format("%s by %s", heading, w[3]);
785 for (int t = deadline - 1; t >= 0; t--) {

795 printf("%s none\n", heading);

798 int departure = time_value(w[3]);
799 if (!strcmp(cmd, "PROFILE")) {
800 int last = time_value(w[4]);
801 if (departure > last) error("empty interval %s %s", w[3], w[4]);
802 size_t length = (size_t)(last - departure + 1);
803 int *arrivals = allocate(length * sizeof(*arrivals));
804 for (int t = departure; t <= last; t++) {
805 Search result = earliest(origin, t);
806 Label label = result.labels[destination->index];
807 arrivals[t - departure] = label.present ? label.time : -1;
808 free_search(result);
809 }
810 size_t count = 0;
811 for (size_t i = 0; i < length; i++) if (arrivals[i] >= 0 && (i + 1 == length || arrivals[i + 1] < 0 || arrivals[i] < arrivals[i + 1])) count++;
812 printf("%s %s %s count %zu\n", heading, w[3], w[4], count);
813 for (size_t i = 0; i < length; i++) if (arrivals[i] >= 0 && (i + 1 == length || arrivals[i + 1] < 0 || arrivals[i] < arrivals[i + 1])) printf("%s -> %s total %d\n", clock_text(departure + (int)i), clock_text(arrivals[i]), arrivals[i] - departure - (int)i);
814 free(arrivals);
815 return;
816 }
817 if (!strcmp(cmd, "FEWEST") || !strcmp(cmd, "PARETO")) {
818 Rounds result = rounds(origin, departure);
819 size_t *options = NULL, count = 0;
820 for (size_t k = 0; k < result.count; k++) {
821 Label label = result.labels[k][destination->index], previous = k ? result.labels[k - 1][destination->index] : (Label){0};
822 if (label.present && (!previous.present || previous.time > label.time)) APPEND(options, count, k);
823 }
824 if (!strcmp(cmd, "FEWEST")) {
825 if (!count) printf("%s %s unreachable\n", heading, w[3]);
826 else {
827 size_t k = options[0];
828 print_journey(format("%s trips %zu", heading, k), departure, result.labels[k][destination->index]);
829 }
830 } else {
831 printf("%s %s options %zu\n", heading, w[3], count);
832 for (size_t i = 0; i < count; i++) {
833 size_t k = options[i];
834 print_journey(format("OPTION trips %zu", k), departure, result.labels[k][destination->index]);
835 }
836 }
837 free(options);
838 free_rounds(result);
839 return;
840 }
841 Search result = earliest(origin, departure);
842 Label label = result.labels[destination->index];
843 if (label.present) print_journey(heading, departure, label);
844 else printf("%s %s unreachable\n", heading, w[3]);
845 free_search(result);
846}

848static char *read_line(void) {
849 Buffer b = {0};
850 int c;
851 while ((c = getchar()) != EOF && c != '\n') append(&b, (char[]){(char)c, 0});
852 if (c == EOF && !b.length) return NULL;
853 return finish(&b);
854}
855int main(void) {
856 int count = atoi(read_line());
857 for (int i = 0; i < count; i++) {
858 Strings words = split(read_line());
859 if (setjmp(failure)) printf("ERROR %s\n", error_reason);
860 else command(words.items, words.count);
861 }
862 return 0;
