37trips <- function(line) (line$last - line$first) %/% line$headway + 1L
38duration <- function(line) tail(line$offsets, 1L)
39directions <- function(line) {
40 list(list(stops = line$stops, offsets = line$offsets),
41 list(stops = rev(line$stops), offsets = duration(line) - rev(line$offsets)))

43next_train <- function(line, offset, ready) {
44 difference <- max(0L, ready - line$first - line$delay - offset)
45 q <- (difference + line$headway - 1L) %/% line$headway
46 if (q < trips(line)) q else NULL
47}
48train_time <- function(line, q, offset) line$first + line$delay + q * line$headway + offset

61earliest <- function(origin, departure) {
62 labels <- new.env(parent = emptyenv())
63 labels[[origin]] <- list(time = departure, journey = list())
64 settled <- character()
65 lines <- running_lines()
66 repeat {
67 candidates <- setdiff(ordered(ls(labels)), settled)
68 if (!length(candidates)) break
69 times <- vapply(candidates, function(id) labels[[id]]$time, numeric(1))
70 u <- candidates[[which.min(times)]]
71 settled <- c(settled, u)
72 arrival <- labels[[u]]$time
73 journey <- labels[[u]]$journey
74 ready <- arrival + if (u == origin) 0L else state$stations[[u]]$transfer
75 for (line in lines) {
76 if (!(u %in% line$stops)) next
77 for (direction in directions(line)) {
78 stops <- direction$stops
79 offsets <- direction$offsets
80 i <- match(u, stops)
81 if (i == length(stops)) next
82 q <- next_train(line, offsets[[i]], ready)
83 if (is.null(q)) next
84 boarding <- train_time(line, q, offsets[[i]])
85 for (j in (i + 1L):length(stops)) {
86 if (blocked(stops[[j - 1L]], stops[[j]])) break
87 if (state$stations[[stops[[j]]]]$closed) next
88 time <- train_time(line, q, offsets[[j]])
89 leg <- ride_leg(line$id, tail(stops, 1L), u, boarding, stops[[j]], time, j - i)
90 update_label(labels, stops[[j]], time, c(journey, list(leg)))
91 }
92 }
93 }
94 for (v in keys(state$walks[[u]])) {
95 time <- arrival + state$walks[[u]][[v]]
96 if (state$stations[[v]]$closed || time > 1440L) next
97 update_label(labels, v, time, c(journey, list(walk_leg(u, arrival, v, time))))
98 }
99 }
100 as.list(labels)
101}
102walk_step <- function(labels, selected, marked) {
103 selected <- ordered(selected)
104 snapshot <- lapply(selected, function(u) labels[[u]])
105 names(snapshot) <- selected
106 for (u in selected) {
107 arrival <- snapshot[[u]]$time
108 journey <- snapshot[[u]]$journey
109 for (v in keys(state$walks[[u]])) {
110 time <- arrival + state$walks[[u]][[v]]
111 if (state$stations[[v]]$closed || time > 1440L) next
112 if (update_label(labels, v, time, c(journey, list(walk_leg(u, arrival, v, time))))) marked <- union(marked, v)
113 }
114 }
115 marked

117round_search <- function(origin, departure) {
118 labels <- new.env(parent = emptyenv())
119 labels[[origin]] <- list(time = departure, journey = list())
120 marked <- walk_step(labels, origin, origin)
121 result <- list(as.list(labels))
122 lines <- running_lines()
123 while (length(marked)) {
124 previous <- as.list(labels)
125 previous_marked <- marked
126 labels <- list2env(previous, parent = emptyenv())
127 marked <- character()
128 selected <- character()
129 for (line in lines) {
130 for (direction in directions(line)) {
131 stops <- direction$stops
132 offsets <- direction$offsets
133 boarded <- NULL
134 b <- NULL
135 boarding <- NULL
136 journey <- list()
137 for (j in seq_along(stops)) {
138 u <- stops[[j]]
139 if (j > 1L && blocked(stops[[j - 1L]], u)) boarded <- NULL
140 if (state$stations[[u]]$closed) next
141 if (!is.null(boarded)) {
142 time <- train_time(line, boarded, offsets[[j]])
143 leg <- ride_leg(line$id, tail(stops, 1L), stops[[b]], boarding, u, time, j - b)
144 if (update_label(labels, u, time, c(journey, list(leg)))) {
145 selected <- union(selected, u)
146 marked <- union(marked, u)
147 }
148 }
149 if (j == length(stops) || !(u %in% previous_marked)) next
150 arrival <- previous[[u]]$time
151 ready <- arrival + if (u == origin) 0L else state$stations[[u]]$transfer
152 candidate <- next_train(line, offsets[[j]], ready)
153 if (!is.null(candidate) && (is.null(boarded) || candidate < boarded)) {
154 boarded <- candidate
155 b <- j
156 boarding <- train_time(line, candidate, offsets[[j]])
157 journey <- previous[[u]]$journey
158 }
159 }
160 }
161 }
162 marked <- walk_step(labels, selected, marked)
163 result[[length(result) + 1L]] <- as.list(labels)
164 }
165 result
166}
167ride_count <- function(journey) sum(vapply(journey, function(leg) leg$kind == "ride", logical(1)))
168print_journey <- function(heading, departure, label) {
169 arrival <- label$time
170 journey <- label$journey
171 walks <- sum(vapply(journey, function(leg) leg$kind == "walk", logical(1)))
172 say(heading, " ", clock(departure), " -> ", clock(arrival), " total ", arrival - departure,
173 " rides ", ride_count(journey), " walks ", walks)
174 previous_end <- departure
175 for (leg in journey) {
176 if (leg$kind == "ride") {
177 say("RIDE ", leg$line, " toward ", leg$terminal, " ", leg$from, " ", clock(leg$start),
178 " -> ", leg$to, " ", clock(leg$finish), " wait ", leg$start - previous_end, " stops ", leg$stops)
179 } else if (leg$kind == "walk") {
180 say("WALK ", leg$from, " ", clock(leg$start), " -> ", leg$to, " ", clock(leg$finish))
181 } else {
182 say("STAY ", leg$station, " ", clock(leg$start), " -> ", clock(leg$finish))
183 }
184 previous_end <- leg$finish
185 }

187print_fare <- function(origin, destination, time_text, fare_class, journey) {
188 tariff <- state$tariff
189 half <- function(x) ((x + 19L) %/% 20L) * 10L
190 cap <- if (fare_class == "CHILD") half(tariff$discount) else tariff$discount
191 tickets <- list()
192 previous_end <- NULL
193 for (leg in journey) {
194 if (leg$kind != "ride") next
195 line <- state$lines[[leg$line]]
196 for (direction in directions(line)) if (tail(direction$stops, 1L) == leg$terminal) break
197 stops <- direction$stops
198 selected <- stops[match(leg$from, stops):match(leg$to, stops)]
199 zone_count <- length(unique(vapply(selected, function(id) state$stations[[id]]$zone, integer(1))))
200 price <- tariff$prices[[min(zone_count, length(tariff$prices))]]
201 if (fare_class == "CHILD") price <- half(price)
202 discount <- if (!is.null(previous_end) && leg$start - previous_end <= tariff$window) min(cap, price) else 0L
203 tickets[[length(tickets) + 1L]] <- list(line = leg$line, from = leg$from, to = leg$to,
204 zones = zone_count, price = price, discount = discount, pay = price - discount)
205 previous_end <- leg$finish
206 }
207 total <- sum(vapply(tickets, function(ticket) ticket$pay, numeric(1)))
208 say("FARE ", origin, " ", destination, " ", time_text, " ", fare_class, " rides ", length(tickets), " total ", total)
209 for (ticket in tickets) say("TICKET ", ticket$line, " ", ticket$from, " ", ticket$to, " zones ", ticket$zones,
210 " price ", ticket$price, " discount ", ticket$discount, " pay ", ticket$pay)
211}
212search_command <- function(command, args) {
213 origin <- args[[1]]
214 destination <- args[[2]]
215 journey_check(origin, destination)
216 departure <- if (length(args) >= 3L) minutes(args[[3]]) else NULL
217 if (command == "FARE") {
218 if (!(args[[4]] %in% c("ADULT", "CHILD"))) fail(paste("unknown fare class", args[[4]]))
219 if (is.null(state$tariff)) fail("no tariff")
220 } else if (command == "PROFILE" && departure > minutes(args[[4]])) {
221 fail(paste("empty interval", args[[3]], args[[4]]))
222 }
223 if (command %in% c("ROUTE", "FARE")) {
224 label <- earliest(origin, departure)[[destination]]
225 if (is.null(label)) {
226 say(command, " ", paste(args, collapse = " "), " unreachable")
227 } else if (command == "ROUTE") {
228 print_journey(paste("ROUTE", origin, destination), departure, label)
229 } else {
230 print_fare(origin, destination, args[[3]], args[[4]], label$journey)
231 }
232 } else if (command %in% c("FEWEST", "PARETO")) {
233 all <- round_search(origin, departure)
234 options <- integer()
235 for (i in seq_along(all)) {
236 label <- all[[i]][[destination]]
237 if (!is.null(label) && (i == 1L || is.null(all[[i - 1L]][[destination]]) || label$time < all[[i - 1L]][[destination]]$time)) options <- c(options, i)
238 }
239 if (command == "FEWEST") {
240 if (length(options)) {
241 i <- options[[1]]
242 print_journey(paste("FEWEST", origin, destination, "trips", i - 1L), departure, all[[i]][[destination]])
243 } else {
244 say("FEWEST ", paste(args, collapse = " "), " unreachable")
245 }
246 } else {
247 say("PARETO ", paste(args, collapse = " "), " options ", length(options))
248 for (i in options) print_journey(paste("OPTION trips", i - 1L), departure, all[[i]][[destination]])
249 }
250 } else if (command %in% c("LASTTRAIN", "ARRIVEBY")) {
251 start <- if (command == "LASTTRAIN") 1439L else departure - 1L
252 heading <- paste(command, origin, destination)
253 if (command == "ARRIVEBY") heading <- paste(heading, "by", args[[3]])
254 if (start >= 0L) {
255 for (time in seq.int(start, 0L)) {
256 label <- earliest(origin, time)[[destination]]
257 if (!is.null(label) && (command == "LASTTRAIN" || label$time <= departure)) {
258 print_journey(heading, time, label)
259 return(invisible(NULL))
260 }
261 }
262 }
263 say(heading, " none")
264 } else if (command == "PROFILE") {
265 finish <- minutes(args[[4]])
266 times <- seq.int(departure, finish)
267 arrivals <- vapply(times, function(time) {
268 label <- earliest(origin, time)[[destination]]
269 if (is.null(label)) Inf else label$time
270 }, numeric(1))
271 selected <- which(is.finite(arrivals) & c(arrivals[-length(arrivals)] < arrivals[-1L], TRUE))
272 say("PROFILE ", paste(args, collapse = " "), " count ", length(selected))
273 for (i in selected) say(clock(times[[i]]), " -> ", clock(arrivals[[i]]), " total ", arrivals[[i]] - times[[i]])
274 }
275}

280execute <- function(text) {
281 words <- strsplit(text, " ", fixed = TRUE)[[1]]
282 command <- words[[1]]
283 args <- words[-1]
284 a <- if (length(args)) args[[1]] else NULL
285 b <- if (length(args) >= 2L) args[[2]] else NULL
286 switch(command,
287 STATION = {
288 if (!is.null(state$stations[[a]])) fail(paste("station", a, "already exists"))
289 state$stations[[a]] <- list(zone = as.integer(b), name = args[[3]], transfer = 0L, closed = FALSE)
290 state$walks[[a]] <- list()
291 say("STATION ", a, " zone ", b, " ", args[[3]])
292 },
293 LINE = {
294 if (!is.null(state$lines[[a]])) fail(paste("line", a, "already exists"))
295 stops <- args[seq.int(5L, length(args), by = 2L)]
296 for (id in stops) station_check(id)
297 seen <- character()
298 for (id in stops) {
299 if (id %in% seen) fail(paste("station", id, "appears twice on line", a))
300 seen <- c(seen, id)
301 }
302 first <- minutes(b)
303 last <- minutes(args[[3]])
304 if (first > last) fail(paste("line", a, "first departure is after last departure"))
305 offsets <- c(0L, cumsum(as.integer(args[seq.int(6L, length(args) - 1L, by = 2L)])))
306 if (last + tail(offsets, 1L) > 1440L) fail(paste("line", a, "runs past 24:00"))
307 line <- list(id = a, first = first, last = last, headway = as.integer(args[[4]]),
308 stops = stops, offsets = offsets, delay = 0L, suspended = FALSE)
309 state$lines[[a]] <- line
310 say("LINE ", a, " stops ", length(stops), " time ", duration(line), " trips ", trips(line))
311 },
312 TRANSFER = {
313 station_check(a)
314 state$stations[[a]]$transfer <- as.integer(b)
315 say("TRANSFER ", a, " ", b)
316 },
317 WALK = {
318 station_check(a)
319 station_check(b)
320 if (a == b) fail(paste("walk from", a, "to itself"))
321 old <- state$walks[[a]][[b]]
322 state$walks[[a]][[b]] <- as.integer(args[[3]])
323 state$walks[[b]][[a]] <- as.integer(args[[3]])
324 say("WALK ", paste(args, collapse = " "), if (is.null(old)) "" else paste0(" was ", old))
325 },
326 TARIFF = {
327 prices <- as.integer(args[-c(1L, 2L)])
328 if (any(diff(prices) < 0L)) fail("tariff prices must not decrease")
329 state$tariff <- list(discount = as.integer(a), window = as.integer(b), prices = prices)
330 say("TARIFF zones ", length(prices), " discount ", a, " window ", b)
331 },
332 CLOSE = closure_command(args, TRUE),
333 OPEN = closure_command(args, FALSE),
334 DELAY = {
335 line_check(a)
336 if (state$lines[[a]]$last + as.integer(b) + duration(state$lines[[a]]) > 1440L) fail(paste("line", a, "runs past 24:00"))
337 state$lines[[a]]$delay <- as.integer(b)
338 say("DELAY ", a, " ", b)
339 },
340 SUSPEND = suspend_command(a, TRUE),
341 RESUME = suspend_command(a, FALSE),
342 ROUTE = search_command(command, args),
343 FEWEST = search_command(command, args),
344 PARETO = search_command(command, args),
345 FARE = search_command(command, args),
346 LASTTRAIN = search_command(command, args),
347 ARRIVEBY = search_command(command, args),
348 PROFILE = search_command(command, args),
349 VIA = {
350 origin <- a
351 via <- b
352 destination <- args[[3]]
353 time_text <- args[[4]]
354 for (id in c(origin, via, destination)) station_check(id)
355 if (origin == destination) fail("origin and destination are the same")
356 if (via == origin || via == destination) fail(paste("via station", via, "must differ from origin and destination"))
357 for (id in c(origin, via, destination)) open_check(id)
358 departure <- minutes(time_text)
359 first <- earliest(origin, departure)[[via]]
360 second <- NULL
361 if (!is.null(first)) {
362 ready <- first$time + state$stations[[via]]$transfer
363 second <- earliest(via, ready)[[destination]]
364 }
365 if (!is.null(second)) {
366 stay <- list(kind = "stay", station = via, start = first$time, finish = ready)
367 journey <- c(first$journey, list(stay), second$journey)
368 print_journey(paste("VIA", origin, via, destination), departure, list(time = second$time, journey = journey))
369 } else {
370 say("VIA ", paste(args, collapse = " "), " unreachable")
371 }
372 },
373 REACH = {
374 station_check(a)
375 open_check(a)
376 departure <- minutes(b)
377 labels <- earliest(a, departure)
378 selected <- keys(labels)
379 times <- vapply(selected, function(id) labels[[id]]$time, numeric(1))
380 selected <- selected[order(times, selected, method = "radix")]
381 selected <- selected[vapply(selected, function(id) labels[[id]]$time <= departure + as.integer(args[[3]]), logical(1))]
382 say("REACH ", paste(args, collapse = " "), " count ", length(selected))
383 for (id in selected) {
384 label <- labels[[id]]
385 say(id, " ", state$stations[[id]]$name, " ", clock(label$time), " ", label$time - departure,
386 " rides ", ride_count(label$journey))
387 }
388 },
389 TIMETABLE = {
390 station_check(a)
391 line_check(b)
392 line <- state$lines[[b]]
393 if (!(a %in% line$stops)) fail(paste("station", a, "is not on line", b))
394 toward <- args[[3]]
395 direction <- get_direction(line, toward)
396 if (a == toward) fail(paste("no departures from", a, "toward", toward))
397 if (line$suspended) fail(paste("line", b, "is suspended"))
398 open_check(a)
399 offset <- direction$offsets[[match(a, direction$stops)]]
400 times <- train_time(line, seq_len(trips(line)) - 1L, offset)
401 times <- head(times[times >= minutes(args[[4]])], as.integer(args[[5]]))
402 say("TIMETABLE ", a, " ", b, " toward ", toward, " ", args[[4]], " next ", length(times),
403 if (length(times)) paste0(" ", paste(vapply(times, clock, character(1)), collapse = " ")) else "")
404 },
405 DEPARTURES = station_trains(command, args),
406 ARRIVALS = station_trains(command, args),
407 TRIP = {
408 line_check(a)
409 line <- state$lines[[a]]
410 direction <- get_direction(line, b)
411 if (line$suspended) fail(paste("line", a, "is suspended"))
412 q <- next_train(line, 0L, minutes(args[[3]]))
413 if (is.null(q)) {
414 say("TRIP ", a, " toward ", b, " ", args[[3]], " none")
415 } else {
416 say("TRIP ", a, " toward ", b, " ", args[[3]], " trip ", q + 1L, " departs ", clock(train_time(line, q, 0L)))
417 for (i in seq_along(direction$stops)) {
418 id <- direction$stops[[i]]
419 suffix <- if (state$stations[[id]]$closed) " closed" else ""
420 if (i > 1L && blocked(direction$stops[[i - 1L]], id)) suffix <- paste0(suffix, " blocked")
421 say(clock(train_time(line, q, direction$offsets[[i]])), " ", id, suffix)
422 }
423 }
424 },
425 SHOWLINE = {
426 line_check(a)
427 line <- state$lines[[a]]
428 say("SHOWLINE ", a, " ", if (line$suspended) "suspended" else "running", " first ", clock(line$first),
429 " last ", clock(line$last), " headway ", line$headway, " delay ", line$delay, " trips ", trips(line))
430 for (i in seq_along(line$stops)) {
431 id <- line$stops[[i]]
432 suffix <- if (state$stations[[id]]$closed) " closed" else ""
433 if (i > 1L && blocked(line$stops[[i - 1L]], id)) suffix <- paste0(suffix, " blocked")
434 say(line$offsets[[i]], " ", id, " zone ", state$stations[[id]]$zone, suffix)
435 }
436 },
437 INFO = {
438 station_check(a)
439 station <- state$stations[[a]]
440 say("INFO ", a, " ", station$name, " zone ", station$zone, " transfer ", station$transfer,
441 " ", if (station$closed) "closed" else "open")
442 lines <- keys(state$lines)
443 lines <- lines[vapply(lines, function(id) a %in% state$lines[[id]]$stops, logical(1))]
444 say("LINES ", if (length(lines)) paste(lines, collapse = " ") else "none")
445 neighbors <- keys(state$walks[[a]])
446 walks <- vapply(neighbors, function(v) paste(v, state$walks[[a]][[v]]), character(1))
447 say("WALKS ", if (length(walks)) paste(walks, collapse = " ") else "none")
448 },
449 STATUS = {
450 say("STATUS stations ", length(state$stations), " lines ", length(state$lines), " walks ", sum(lengths(state$walks)) %/% 2L)
451 for (id in keys(state$stations)) if (state$stations[[id]]$closed) say("CLOSED ", id)
452 segments <- unname(state$closed)
453 if (length(segments)) {
454 first <- vapply(segments, function(pair) pair[[1]], character(1))
455 second <- vapply(segments, function(pair) pair[[2]], character(1))
456 for (i in order(first, second, method = "radix")) say("CLOSED ", first[[i]], "-", second[[i]])
457 }
458 for (id in keys(state$lines)) if (state$lines[[id]]$suspended) say("SUSPENDED ", id)
459 for (id in keys(state$lines)) if (state$lines[[id]]$delay != 0L) say("DELAYED ", id, " ", state$lines[[id]]$delay)
460 }
461 )

463closure_command <- function(args, close) {
464 a <- args[[1]]
465 station_check(a)
466 command <- if (close) "CLOSE" else "OPEN"
467 if (length(args) == 2L) {
468 b <- args[[2]]
469 station_check(b)
470 key <- segment(a, b)
471 if (close) {
472 affected <- character()
473 for (id in keys(state$lines)) {
474 stops <- state$lines[[id]]$stops
475 for (i in seq_len(length(stops) - 1L)) {
476 if (segment(stops[[i]], stops[[i + 1L]]) == key) {
477 affected <- c(affected, id)
478 break
479 }
480 }
481 }
482 if (!length(affected)) fail(paste("no segment", paste0(a, "-", b)))
483 if (!is.null(state$closed[[key]])) fail(paste("segment", paste0(a, "-", b), "is already closed"))
484 state$closed[[key]] <- ordered(c(a, b))
485 say("CLOSE ", a, "-", b, " lines ", paste(affected, collapse = " "))
486 } else {
487 if (is.null(state$closed[[key]])) fail(paste("segment", paste0(a, "-", b), "is not closed"))
488 state$closed[[key]] <- NULL
489 say("OPEN ", a, "-", b)
490 }
491 } else {
492 current <- state$stations[[a]]$closed
493 if (close && current) fail(paste("station", a, "is already closed"))
494 if (!close && !current) fail(paste("station", a, "is not closed"))
495 state$stations[[a]]$closed <- close
496 say(command, " ", a)
497 }

506station_trains <- function(command, args) {
507 station <- args[[1]]
508 station_check(station)
509 open_check(station)
510 entries <- list()
511 for (line in running_lines()) {
512 if (!(station %in% line$stops)) next
513 dirs <- directions(line)
514 for (d in seq_along(dirs)) {
515 direction <- dirs[[d]]
516 stops <- direction$stops
517 i <- match(station, stops)
518 if (if (command == "DEPARTURES") i == length(stops) else i == 1L) next
519 for (q in seq_len(trips(line)) - 1L) {
