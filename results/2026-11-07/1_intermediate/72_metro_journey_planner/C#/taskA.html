19 class MetroLine(string id, int first, int last, int headway, List<Station> stops, int[] cumulative)
20 {
21 public readonly string Id = id;
22 public readonly int First = first, Last = last, Headway = headway;
23 public readonly List<Station> Stops = [.. stops];
24 public readonly int[] Cumulative = (int[])cumulative.Clone();
25 public int Delay;
26 public bool Suspended;
27 public int Total => Cumulative[^1];
28 public int Trips => (Last - First) / Headway + 1;
29 public Station Stop(int direction, int index) => Stops[direction == 0 ? index : Stops.Count - 1 - index];
30 public int Offset(int direction, int index) => direction == 0 ? Cumulative[index] : Total - Cumulative[Stops.Count - 1 - index];
31 public int Position(int direction, Station station)
32 {
33 int index = Stops.IndexOf(station);
34 return index < 0 ? -1 : direction == 0 ? index : Stops.Count - 1 - index;
35 }
36 public int Time(int direction, int index, int train) => First + Delay + train * Headway + Offset(direction, index);
37 public int Next(int direction, int index, int ready)
38 {
39 int start = Time(direction, index, 0), train = ready <= start ? 0 : (ready - start + Headway - 1) / Headway;
40 return train < Trips ? train : -1;
41 }
42 }

91 static Dictionary<Station, Label> Earliest(Station origin, int departure)
92 {
93 Dictionary<Station, Label> labels = new() { [origin] = new(departure, []) };
94 HashSet<Station> settled = [];
95 while (true)
96 {
97 Station? selected = null;
98 foreach (Station station in Stations.Values) if (!settled.Contains(station) && labels.ContainsKey(station) && (selected is null || labels[station].Time < labels[selected].Time)) selected = station;
99 if (selected is null) return labels;
100 Station u = selected;
101 settled.Add(u);
102 Label label = labels[u];
103 int ready = label.Time + (u == origin ? 0 : u.Transfer);
104 foreach (MetroLine line in Lines.Values) if (!line.Suspended && line.Stops.Contains(u)) for (int direction = 0; direction < 2; direction++)
105 {
106 int i = line.Position(direction, u), last = line.Stops.Count - 1;
107 if (i == last) continue;
108 int train = line.Next(direction, i, ready);
109 if (train < 0) continue;
110 int start = line.Time(direction, i, train);
111 for (int j = i + 1; j <= last; j++)
112 {
113 Station v = line.Stop(direction, j);
114 if (Blocked(line.Stop(direction, j - 1), v)) break;
115 if (v.Closed) continue;
116 int arrival = line.Time(direction, j, train);
117 Relax(labels, v, arrival, label.Journey, new('R', line, line.Stop(direction, last), u, start, v, arrival, j - i));
118 }
119 }
120 foreach (var walk in u.Walks)
121 {
122 Station v = Stations[walk.Key];
123 int arrival = label.Time + walk.Value;
124 if (!v.Closed && arrival <= 1440) Relax(labels, v, arrival, label.Journey, new('W', null, null, u, label.Time, v, arrival, 0));
125 }
126 }
127 }

129 static void WalkStep(Dictionary<Station, Label> labels, HashSet<Station> origins, HashSet<Station> marked)
130 {
131 var recorded = origins.ToDictionary(origin => origin, origin => labels[origin]);
132 foreach (Station u in Stations.Values) if (origins.Contains(u)) foreach (var walk in u.Walks)
133 {
134 Label label = recorded[u];
135 Station v = Stations[walk.Key];
136 int arrival = label.Time + walk.Value;
137 if (!v.Closed && arrival <= 1440 && (!labels.ContainsKey(v) || labels[v].Time > arrival))
138 {
139 labels[v] = new(arrival, Append(label.Journey, new('W', null, null, u, label.Time, v, arrival, 0)));
140 marked.Add(v);
141 }
142 }
143 }

145 static List<Dictionary<Station, Label>> Rounds(Station origin, int departure)
146 {
147 List<Dictionary<Station, Label>> result = [];
148 Dictionary<Station, Label> previous = new() { [origin] = new(departure, []) };
149 HashSet<Station> marked = [origin];
150 WalkStep(previous, [origin], marked);
151 result.Add(previous);
152 while (marked.Count > 0)
153 {
154 Dictionary<Station, Label> current = new(previous);
155 HashSet<Station> updated = [], walks = [];
156 foreach (MetroLine line in Lines.Values) if (!line.Suspended) for (int direction = 0; direction < 2; direction++)
157 {
158 int boarded = -1, boardIndex = -1, boardTime = 0, last = line.Stops.Count - 1;
159 List<Leg>? boardJourney = null;
160 for (int j = 0; j <= last; j++)
161 {
162 Station u = line.Stop(direction, j);
163 if (j > 0 && Blocked(line.Stop(direction, j - 1), u)) boarded = -1;
164 if (u.Closed) continue;
165 if (boarded >= 0)
166 {
167 int arrival = line.Time(direction, j, boarded);
168 if (!current.ContainsKey(u) || current[u].Time > arrival)
169 {
170 current[u] = new(arrival, Append(boardJourney!, new('R', line, line.Stop(direction, last), line.Stop(direction, boardIndex), boardTime, u, arrival, j - boardIndex)));
171 walks.Add(u);
172 updated.Add(u);
173 }
174 }
175 if (j < last && marked.Contains(u))
176 {
177 Label label = previous[u];
178 int ready = label.Time + (u == origin ? 0 : u.Transfer), train = line.Next(direction, j, ready);
179 if (train >= 0 && (boarded < 0 || train < boarded))
180 {
181 boarded = train;
182 boardIndex = j;
183 boardTime = line.Time(direction, j, train);
184 boardJourney = label.Journey;
185 }
186 }
187 }
188 }
189 WalkStep(current, walks, updated);
190 result.Add(current);
191 previous = current;
192 marked = updated;
193 }
194 return result;
195 }

199 static void PrintJourney(string heading, int departure, Label label)
200 {
201 Console.WriteLine($"{heading} {Clock(departure)} -> {Clock(label.Time)} total {label.Time - departure} rides {Rides(label.Journey)} walks {label.Journey.Count(leg => leg.Kind == 'W')}");
202 int previous = departure;
203 foreach (Leg leg in label.Journey)
204 {
205 if (leg.Kind == 'R') Console.WriteLine($"RIDE {leg.Line!.Id} toward {leg.Terminal!.Id} {leg.From.Id} {Clock(leg.Departure)} -> {leg.To.Id} {Clock(leg.Arrival)} wait {leg.Departure - previous} stops {leg.Stops}");
206 else if (leg.Kind == 'W') Console.WriteLine($"WALK {leg.From.Id} {Clock(leg.Departure)} -> {leg.To.Id} {Clock(leg.Arrival)}");
207 else Console.WriteLine($"STAY {leg.From.Id} {Clock(leg.Departure)} -> {Clock(leg.Arrival)}");
208 previous = leg.Arrival;
209 }

220 {
221 Require(line.Stops[0].Id == terminal || line.Stops[^1].Id == terminal, terminal + " is not a terminal of line " + line.Id);

225 static int Half(int price) => (price + 19) / 20 * 10;
226
227 static void Fare(string heading, Station origin, Station destination, int departure, string fareClass)
228 {
229 Label? label = Earliest(origin, departure).GetValueOrDefault(destination);
230 if (label is null)
231 {
232 Console.WriteLine(heading + " unreachable");
233 return;
234 }
235 List<string> tickets = [];
236 int total = 0, lastRide = -1;
237 bool child = fareClass == "CHILD";
238 foreach (Leg leg in label.Journey) if (leg.Kind == 'R')
239 {
240 MetroLine line = leg.Line!;
241 int a = line.Stops.IndexOf(leg.From), b = line.Stops.IndexOf(leg.To);
242 HashSet<int> zones = [];
243 for (int i = Math.Min(a, b); i <= Math.Max(a, b); i++) zones.Add(line.Stops[i].Zone);
244 int price = prices![Math.Min(zones.Count, prices.Length) - 1];
245 if (child) price = Half(price);
246 int reduction = lastRide >= 0 && leg.Departure - lastRide <= window ? Math.Min(price, child ? Half(discount) : discount) : 0;
247 int pay = price - reduction;
248 total += pay;
249 tickets.Add($"TICKET {line.Id} {leg.From.Id} {leg.To.Id} zones {zones.Count} price {price} discount {reduction} pay {pay}");
250 lastRide = leg.Arrival;
251 }
252 Console.WriteLine($"{heading} rides {Rides(label.Journey)} total {total}");
253 tickets.ForEach(Console.WriteLine);
254 }

256 static void Command(string[] w)
257 {
258 string command = w[0];
259 switch (command)
260 {
261 case "STATION":
262 Require(!Stations.ContainsKey(w[1]), "station " + w[1] + " already exists");
263 Stations[w[1]] = new(w[1], int.Parse(w[2]), w[3]);
264 Console.WriteLine($"STATION {w[1]} zone {w[2]} {w[3]}");
265 return;
266 case "LINE":
267 {
268 Require(!Lines.ContainsKey(w[1]), "line " + w[1] + " already exists");
269 List<Station> stops = [];
270 for (int i = 5; i < w.Length; i += 2) stops.Add(GetStation(w[i]));
271 HashSet<Station> seen = [];
272 foreach (Station stop in stops) Require(seen.Add(stop), "station " + stop.Id + " appears twice on line " + w[1]);
273 int first = Time(w[2]), last = Time(w[3]);
274 Require(first <= last, "line " + w[1] + " first departure is after last departure");
275 int[] cumulative = new int[stops.Count];
276 for (int i = 1; i < stops.Count; i++) cumulative[i] = cumulative[i - 1] + int.Parse(w[4 + 2 * i]);
277 Require(last + cumulative[^1] <= 1440, "line " + w[1] + " runs past 24:00");
278 MetroLine line = new(w[1], first, last, int.Parse(w[4]), stops, cumulative);
279 Lines[w[1]] = line;
280 Console.WriteLine($"LINE {line.Id} stops {stops.Count} time {line.Total} trips {line.Trips}");
281 return;
282 }
283 case "TRANSFER":
284 {
285 Station station = GetStation(w[1]);
286 station.Transfer = int.Parse(w[2]);
287 Console.WriteLine($"TRANSFER {w[1]} {w[2]}");
288 return;
289 }
290 case "WALK":
291 {
292 Station a = GetStation(w[1]), b = GetStation(w[2]);
293 Require(a != b, "walk from " + a.Id + " to itself");
294 int? before = a.Walks.TryGetValue(b.Id, out int old) ? old : null;
295 int minutes = int.Parse(w[3]);
296 a.Walks[b.Id] = minutes;
297 b.Walks[a.Id] = minutes;
298 Console.WriteLine($"WALK {a.Id} {b.Id} {minutes}" + (before is null ? "" : " was " + before));
299 return;
300 }
301 case "TARIFF":
302 {
303 int[] added = w.Skip(3).Select(int.Parse).ToArray();
304 for (int i = 1; i < added.Length; i++) Require(added[i] >= added[i - 1], "tariff prices must not decrease");
305 prices = added;
306 discount = int.Parse(w[1]);
307 window = int.Parse(w[2]);
308 Console.WriteLine($"TARIFF zones {prices.Length} discount {discount} window {window}");
309 return;
310 }
311 case "CLOSE":
312 case "OPEN":
313 {
314 Station a = GetStation(w[1]);
315 bool close = command == "CLOSE";
316 if (w.Length == 2)
317 {
318 Require(close != a.Closed, "station " + a.Id + (close ? " is already closed" : " is not closed"));
319 a.Closed = close;
320 Console.WriteLine(command + " " + a.Id);
321 }
322 else
323 {
324 Station b = GetStation(w[2]);
325 Segment key = Pair(a, b);
326 List<string> containing = [];
327 if (close)
328 {
329 foreach (MetroLine line in Lines.Values) for (int i = 1; i < line.Stops.Count; i++) if (Pair(line.Stops[i - 1], line.Stops[i]) == key)
330 {
331 containing.Add(line.Id);
332 break;
333 }
334 Require(containing.Count > 0, "no segment " + a.Id + "-" + b.Id);
335 }
336 Require(close != ClosedSegments.Contains(key), "segment " + a.Id + "-" + b.Id + (close ? " is already closed" : " is not closed"));
337 if (close) ClosedSegments.Add(key);
338 else ClosedSegments.Remove(key);
339 Console.WriteLine(command + " " + a.Id + "-" + b.Id + (close ? " lines " + string.Join(" ", containing) : ""));
340 }
341 return;
342 }
343 case "DELAY":
344 {
345 MetroLine line = GetLine(w[1]);
346 int delay = int.Parse(w[2]);
347 Require(line.Last + delay + line.Total <= 1440, "line " + line.Id + " runs past 24:00");
348 line.Delay = delay;
349 Console.WriteLine($"DELAY {line.Id} {delay}");
350 return;
351 }
352 case "SUSPEND":
353 case "RESUME":
354 {
355 MetroLine line = GetLine(w[1]);
356 bool suspend = command == "SUSPEND";
357 Require(line.Suspended != suspend, "line " + line.Id + (suspend ? " is already suspended" : " is not suspended"));
358 line.Suspended = suspend;
359 Console.WriteLine(command + " " + line.Id);
360 return;
361 }
362 case "VIA":
363 {
364 Station origin = GetStation(w[1]), via = GetStation(w[2]), destination = GetStation(w[3]);
365 Require(origin != destination, "origin and destination are the same");
366 Require(via != origin && via != destination, "via station " + via.Id + " must differ from origin and destination");
367 foreach (Station station in new[] {origin, via, destination}) Require(!station.Closed, "station " + station.Id + " is closed");
368 int departure = Time(w[4]);
369 Label? first = Earliest(origin, departure).GetValueOrDefault(via), result = null;
370 if (first is not null)
371 {
372 int ready = first.Time + via.Transfer;
373 Label? second = Earliest(via, ready).GetValueOrDefault(destination);
374 if (second is not null)
375 {
376 List<Leg> journey = Append(first.Journey, new('S', null, null, via, first.Time, via, ready, 0));
377 journey.AddRange(second.Journey);
378 result = new(second.Time, journey);
379 }
380 }
381 string heading = $"VIA {origin.Id} {via.Id} {destination.Id}";
382 if (result is null) Console.WriteLine($"{heading} {Clock(departure)} unreachable");
383 else PrintJourney(heading, departure, result);
384 return;
385 }
386 case "REACH":
387 {
388 Station origin = GetStation(w[1]);
389 Require(!origin.Closed, "station " + origin.Id + " is closed");
390 int departure = Time(w[2]), limit = departure + int.Parse(w[3]);
391 var labels = Earliest(origin, departure);
392 var reached = Stations.Values.Where(station => labels.ContainsKey(station) && labels[station].Time <= limit).OrderBy(station => labels[station].Time).ThenBy(station => station.Id, StringComparer.Ordinal).ToList();
393 Console.WriteLine($"REACH {origin.Id} {w[2]} {w[3]} count {reached.Count}");
394 foreach (Station station in reached) Console.WriteLine($"{station.Id} {station.Name} {Clock(labels[station].Time)} {labels[station].Time - departure} rides {Rides(labels[station].Journey)}");
395 return;
396 }
397 case "TIMETABLE":
398 {
399 Station station = GetStation(w[1]);
400 MetroLine line = GetLine(w[2]);
401 Require(line.Stops.Contains(station), "station " + station.Id + " is not on line " + line.Id);
402 int direction = Direction(line, w[3]);
403 Require(station.Id != w[3], "no departures from " + station.Id + " toward " + w[3]);
404 Require(!line.Suspended, "line " + line.Id + " is suspended");
405 Require(!station.Closed, "station " + station.Id + " is closed");
406 int position = line.Position(direction, station), start = line.Next(direction, position, Time(w[4])), count = int.Parse(w[5]);
407 List<string> times = [];
408 if (start >= 0) for (int q = start; q < line.Trips && times.Count < count; q++) times.Add(Clock(line.Time(direction, position, q)));
409 Console.WriteLine($"TIMETABLE {station.Id} {line.Id} toward {w[3]} {w[4]} next {times.Count}" + (times.Count == 0 ? "" : " " + string.Join(" ", times)));
410 return;
411 }
412 case "DEPARTURES":
413 case "ARRIVALS":
414 {
415 Station station = GetStation(w[1]);
416 Require(!station.Closed, "station " + station.Id + " is closed");
417 bool arrivals = command == "ARRIVALS";
418 List<Departure> departures = [];
419 foreach (MetroLine line in Lines.Values) if (!line.Suspended && line.Stops.Contains(station)) for (int direction = 0; direction < 2; direction++)
420 {
421 int position = line.Position(direction, station);
422 if (arrivals ? position == 0 : position == line.Stops.Count - 1) continue;
423 int start = line.Next(direction, position, Time(w[2]));
424 if (start >= 0) for (int q = start; q < line.Trips; q++) departures.Add(new(line.Time(direction, position, q), line, direction));
425 }
426 var selected = departures.OrderBy(entry => entry.Time).ThenBy(entry => entry.Line.Id, StringComparer.Ordinal).ThenBy(entry => entry.Direction).Take(int.Parse(w[3])).ToList();
427 Console.WriteLine($"{command} {station.Id} {w[2]} count {selected.Count}");
428 foreach (var entry in selected) Console.WriteLine(Clock(entry.Time) + " " + entry.Line.Id + (arrivals ? " from " + entry.Line.Stop(entry.Direction, 0).Id : " toward " + entry.Line.Stop(entry.Direction, entry.Line.Stops.Count - 1).Id));
429 return;
430 }
431 case "TRIP":
432 {
433 MetroLine line = GetLine(w[1]);
434 int direction = Direction(line, w[2]);
435 Require(!line.Suspended, "line " + line.Id + " is suspended");
436 int train = line.Next(direction, 0, Time(w[3]));
437 string heading = $"TRIP {line.Id} toward {w[2]} {w[3]}";
438 if (train < 0) Console.WriteLine(heading + " none");
439 else
440 {
441 Console.WriteLine($"{heading} trip {train + 1} departs {Clock(line.Time(direction, 0, train))}");
442 for (int j = 0; j < line.Stops.Count; j++)
443 {
444 Station station = line.Stop(direction, j);
445 Console.WriteLine(Clock(line.Time(direction, j, train)) + " " + station.Id + (station.Closed ? " closed" : "") + (j > 0 && Blocked(line.Stop(direction, j - 1), station) ? " blocked" : ""));
446 }
447 }
448 return;
449 }
450 case "SHOWLINE":
451 {
452 MetroLine line = GetLine(w[1]);
453 Console.WriteLine($"SHOWLINE {line.Id} {(line.Suspended ? "suspended" : "running")} first {Clock(line.First)} last {Clock(line.Last)} headway {line.Headway} delay {line.Delay} trips {line.Trips}");
454 for (int j = 0; j < line.Stops.Count; j++)
455 {
456 Station station = line.Stops[j];
457 Console.WriteLine($"{line.Cumulative[j]} {station.Id} zone {station.Zone}" + (station.Closed ? " closed" : "") + (j > 0 && Blocked(line.Stops[j - 1], station) ? " blocked" : ""));
458 }
459 return;
460 }
461 case "INFO":
462 {
463 Station station = GetStation(w[1]);
464 Console.WriteLine($"INFO {station.Id} {station.Name} zone {station.Zone} transfer {station.Transfer} {(station.Closed ? "closed" : "open")}");
465 var containing = Lines.Values.Where(line => line.Stops.Contains(station)).Select(line => line.Id).ToList();
466 Console.WriteLine("LINES " + (containing.Count == 0 ? "none" : string.Join(" ", containing)));
467 Console.WriteLine("WALKS " + (station.Walks.Count == 0 ? "none" : string.Join(" ", station.Walks.Select(entry => entry.Key + " " + entry.Value))));
468 return;
469 }
470 case "STATUS": Console.WriteLine($"STATUS stations {Stations.Count} lines {Lines.Count} walks {Stations.Values.Sum(station => station.Walks.Count) / 2}");
471 foreach (Station station in Stations.Values) if (station.Closed) Console.WriteLine("CLOSED " + station.Id);
472 foreach (var segment in ClosedSegments.OrderBy(entry => entry.Low, StringComparer.Ordinal).ThenBy(entry => entry.High, StringComparer.Ordinal)) Console.WriteLine("CLOSED " + segment.Low + "-" + segment.High);
473 foreach (MetroLine line in Lines.Values) if (line.Suspended) Console.WriteLine("SUSPENDED " + line.Id);
474 foreach (MetroLine line in Lines.Values) if (line.Delay != 0) Console.WriteLine($"DELAYED {line.Id} {line.Delay}");
475 return;
476 }
477 Station originStation = GetStation(w[1]), destinationStation = GetStation(w[2]);
478 SearchCheck(originStation, destinationStation);
479 string title = $"{command} {originStation.Id} {destinationStation.Id}";
480 if (command == "FARE")
481 {
482 Require(w[4] is "ADULT" or "CHILD", "unknown fare class " + w[4]);
483 Require(prices is not null, "no tariff");
484 Fare(title + " " + w[3] + " " + w[4], originStation, destinationStation, Time(w[3]), w[4]);
485 return;
486 }
487 if (command is "LASTTRAIN" or "ARRIVEBY")
488 {
489 int deadline = command == "LASTTRAIN" ? 1440 : Time(w[3]);
490 if (command == "ARRIVEBY") title += " by " + w[3];
491 for (int t = deadline - 1; t >= 0; t--)
492 {
493 Label? label = Earliest(originStation, t).GetValueOrDefault(destinationStation);
494 if (label is not null && label.Time <= deadline)
495 {
496 PrintJourney(title, t, label);
497 return;
498 }
499 }
500 Console.WriteLine(title + " none");
501 return;
502 }
503 int startTime = Time(w[3]);
504 if (command == "PROFILE")
505 {
506 int last = Time(w[4]);
507 Require(startTime <= last, "empty interval " + w[3] + " " + w[4]);
508 SortedDictionary<int, int> arrivals = [];
509 for (int t = startTime; t <= last; t++)
510 {
511 Label? label = Earliest(originStation, t).GetValueOrDefault(destinationStation);
512 if (label is not null) arrivals[t] = label.Time;
513 }
514 var selected = arrivals.Keys.Where(t => t == last || !arrivals.ContainsKey(t + 1) || arrivals[t] < arrivals[t + 1]).ToList();
515 Console.WriteLine($"{title} {w[3]} {w[4]} count {selected.Count}");
516 foreach (int t in selected) Console.WriteLine($"{Clock(t)} -> {Clock(arrivals[t])} total {arrivals[t] - t}");
517 return;
518 }
519 if (command is "FEWEST" or "PARETO")
520 {
521 var rounds = Rounds(originStation, startTime);
522 List<int> options = [];
523 for (int k = 0; k < rounds.Count; k++)
524 {
525 Label? label = rounds[k].GetValueOrDefault(destinationStation), previous = k == 0 ? null : rounds[k - 1].GetValueOrDefault(destinationStation);
526 if (label is not null && (previous is null || previous.Time > label.Time)) options.Add(k);
527 }
