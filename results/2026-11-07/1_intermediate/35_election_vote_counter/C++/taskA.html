70const std::array<string, 18> method_names{"PLURALITY", "RUNOFF", "IRV", "COOMBS", "BUCKLIN", "BORDA", "BALDWIN", "NANSON", "BLACK", "VETO", "CONDORCET", "COPELAND", "MINIMAX", "SCHULZE", "RANKEDPAIRS", "KEMENY", "APPROVAL", "STV"};
71Method parse_method(const Lines& words, std::size_t start = 0) {
72 auto found = std::find(method_names.begin(), method_names.end(), words[start]);
73 Method method{static_cast<Kind>(found - method_names.begin())};
74 if (method.kind == Kind::Approval || method.kind == Kind::Stv) method.parameter = std::stoi(words[start + 1]);
75 if (method.kind == Kind::Minimax) method.score = words[start + 1] == "WINNING" ? Minimax::Winning : words[start + 1] == "MARGIN" ? Minimax::Margin : Minimax::Opposition;
76 return method;
77}
78string method_text(Method method) {
79 string name = method_names[static_cast<std::size_t>(method.kind)];
80 if (method.kind == Kind::Minimax) name += string(" ") + (method.score == Minimax::Winning ? "WINNING" : method.score == Minimax::Margin ? "MARGIN" : "OPPOSITION");
81 if (method.kind == Kind::Approval || method.kind == Kind::Stv) name += " " + to_string(method.parameter);
82 return name;
83}

96 Election(Lines names, int number) : names(std::move(names)), number(number), withdrawn(this->names.size(), false), tie_rank(this->names.size(), 0) {
97 Field order(this->names.size());
98 std::iota(order.begin(), order.end(), 0);
99 std::sort(order.begin(), order.end(), [&](int a, int b) { return this->names[a] < this->names[b]; });
100 set_priority(order);
101 }
102 Values zeros() const { return Values(names.size(), 0); }
103 Matrix matrix() const { return Matrix(names.size(), zeros()); }
104 Relation relation() const { return Relation(names.size(), vector<bool>(names.size(), false)); }
105 Field active() const {
106 Field field;
107 for (int candidate = 0; candidate < static_cast<int>(names.size()); ++candidate) if (!withdrawn[candidate]) field.push_back(candidate);
108 return field;

110 int candidate(const string& name) const {
111 auto found = std::find(names.begin(), names.end(), name);

138 Field ordered(Field field, const Values& values, bool ascending = false) const {
139 std::sort(field.begin(), field.end(), [&](int a, int b) { return values[a] == values[b] ? tie_rank[a] < tie_rank[b] : ascending ? values[a] < values[b] : values[a] > values[b]; });
140 return field;
141 }
142 Field preferred(Field field) const {
143 std::sort(field.begin(), field.end(), [&](int a, int b) { return tie_rank[a] < tie_rank[b]; });
144 return field;

146 const Aggregate& aggregate(const Field& field) {
147 int mask = field_mask(field);
148 auto found = cache.find(mask);
149 if (found != cache.end()) return found->second;
150 std::map<Field, int> combined;
151 for (const auto& entry : entries) for (const Ballot& group : entry) {
152 Field ranking = select(group.ranking, [=](int candidate) { return mask & (1 << candidate); });
153 combined[ranking] += group.votes;
154 }
155 Aggregate result{{}, zeros(), zeros(), zeros(), Matrix(names.size(), Values(field.size(), 0)), matrix()};
156 for (const auto& [ranking, votes] : combined) {
157 result.groups.push_back({votes, ranking});
158 result.total += votes;
159 if (ranking.empty()) result.blank += votes;
160 else result.first[ranking[0]] += votes;
161 Values ranks(names.size(), static_cast<int>(field.size()));
162 for (int place = 0; place < static_cast<int>(ranking.size()); ++place) {
163 int candidate = ranking[place];
164 ranks[candidate] = place;
165 result.positions[candidate][place] += votes;
166 result.borda[candidate] += votes * (static_cast<int>(field.size()) - place - 1);
167 }
168 for (int candidate : field) {
169 if (ranking.size() < field.size() ? ranks[candidate] == static_cast<int>(field.size()) : candidate == ranking.back()) result.last[candidate] += votes;
170 for (int other : field) if (ranks[candidate] < ranks[other]) result.pairwise[candidate][other] += votes;
171 }
172 }
173 return cache.emplace(mask, std::move(result)).first->second;
174 }

177Choice eliminate(const Election& election, const Field& field, const Values& values, const vector<Values>& history) {
178 int minimum = values[election.ordered(field, values, true)[0]];
179 Field choices = select(field, [&](int candidate) { return values[candidate] == minimum; });
180 bool tie = choices.size() > 1;
181 for (auto round = history.rbegin(); round != history.rend() && choices.size() > 1; ++round) {
182 minimum = (*round)[election.ordered(choices, *round, true)[0]];
183 choices = select(choices, [&](int candidate) { return (*round)[candidate] == minimum; });
184 }
185 return {election.preferred(choices).back(), tie};
186}
187Result finish(const Election& election, Field standings, bool tie, Lines rows) {
188 rows.push_back("winner: " + election.names[standings[0]] + tie_text(tie));
189 return {std::move(standings), tie, std::move(rows), {}};
190}
191Result scored(const Election& election, const Field& field, const Values& values, const string& label, bool ascending = false) {
192 Field standings = election.ordered(field, values, ascending);
193 bool tie = values[standings[0]] == values[standings[1]];
194 return finish(election, standings, tie, {label + ": " + election.values(field, values)});
195}
196Lines pair_lines(const Election& election, const Field& field, const Matrix& matrix) {
197 Lines rows;
198 for (std::size_t i = 0; i < field.size(); ++i) for (std::size_t j = i + 1; j < field.size(); ++j) {
199 int first = field[i], second = field[j], a = matrix[first][second], b = matrix[second][first];
200 string result = a > b ? election.names[first] : b > a ? election.names[second] : "tie";
201 rows.push_back(election.names[first] + " vs " + election.names[second] + ": " + to_string(a) + "-" + to_string(b) + " (" + result + ")");
202 }
203 return rows;
204}
205Result runoff(Election& election, const Field& field) {
206 const Values& first = election.aggregate(field).first;
207 Field order = election.ordered(field, first);
208 Lines rows{"round 1: " + election.values(field, first)};
209 if (2 * first[order[0]] > sum_field(field, first)) return finish(election, order, false, rows);
210 Field finalists(order.begin(), order.begin() + 2);
211 std::sort(finalists.begin(), finalists.end());
212 const Values& second = election.aggregate(finalists).first;
213 bool advance = order.size() >= 3 && first[order[1]] == first[order[2]];
214 rows.push_back("finalists: " + election.list(finalists) + tie_text(advance));
215 rows.push_back("round 2: " + election.values(finalists, second));
216 bool tie = second[order[0]] == second[order[1]];
217 if (second[order[1]] > second[order[0]]) std::swap(order[0], order[1]);
218 return finish(election, order, tie, rows);
219}
220Result elimination_rounds(Election& election, const Field& field, bool coombs) {
221 Field remaining = field, removed;
222 vector<Values> history;
223 Lines rows, exhausted;
224 for (int round = 1; ; ++round) {
225 const Aggregate& aggregate = election.aggregate(remaining);
226 const Values& values = aggregate.first;
227 Field ordered = election.ordered(remaining, values);
228 int active = sum_field(remaining, values);
229 string row = "round " + to_string(round) + ": " + election.values(remaining, values);
230 exhausted.push_back("round " + to_string(round) + ": active " + to_string(active) + ", exhausted " + to_string(aggregate.total - active));
231 if (remaining.size() == 1 || 2 * values[ordered[0]] > active) {
232 ordered.insert(ordered.end(), removed.rbegin(), removed.rend());
233 rows.push_back(row);
234 Result result = finish(election, ordered, false, rows);
235 result.exhausted = std::move(exhausted);
236 return result;
237 }
238 Choice chosen;
239 if (coombs) {
240 const Values& last = aggregate.last;
241 int maximum = last[election.ordered(remaining, last)[0]];
242 Field tied = select(remaining, [&](int candidate) { return last[candidate] == maximum; });
243 chosen = {election.ordered(tied, values).back(), tied.size() > 1};
244 row += "; last " + election.values(remaining, last);
245 } else chosen = eliminate(election, remaining, values, history);
246 rows.push_back(row + "; eliminated " + election.names[chosen.candidate] + tie_text(chosen.tie));
247 history.push_back(values);
248 removed.push_back(chosen.candidate);
249 remaining = without(remaining, chosen.candidate);
250 }

252Result bucklin(Election& election, const Field& field) {
253 const Aggregate& aggregate = election.aggregate(field);
254 int majority = (aggregate.total - aggregate.blank) / 2 + 1;
255 vector<Values> history;
256 Lines rows{"majority: " + to_string(majority)};
257 for (std::size_t depth = 1; depth <= field.size(); ++depth) {
258 Values values = election.zeros();
259 for (int candidate : field) values[candidate] = std::accumulate(aggregate.positions[candidate].begin(), aggregate.positions[candidate].begin() + depth, 0);
260 history.push_back(values);
261 rows.push_back("round " + to_string(depth) + ": " + election.values(field, values));
262 if (values[election.ordered(field, values)[0]] >= majority) break;
263 }
264 Field standings = field;
265 std::sort(standings.begin(), standings.end(), [&](int a, int b) {
266 for (auto round = history.rbegin(); round != history.rend(); ++round) if ((*round)[a] != (*round)[b]) return (*round)[a] > (*round)[b];
267 return election.tie_rank[a] < election.tie_rank[b];
268 });
269 bool tie = history.back()[standings[0]] == history.back()[standings[1]];
270 return finish(election, standings, tie, rows);
271}
272Result borda_rounds(Election& election, const Field& field, bool nanson) {
273 Field remaining = field, tail;
274 vector<Values> history;
275 Lines rows;
276 for (int round = 1; remaining.size() >= 2; ++round) {
277 const Values& values = election.aggregate(remaining).borda;
278 string row = "round " + to_string(round) + ": " + election.values(remaining, values);
279 Field removed;
280 if (nanson) {
281 int total = sum_field(remaining, values);
282 removed = select(remaining, [&](int candidate) { return values[candidate] * static_cast<int>(remaining.size()) < total; });
283 if (removed.empty()) { rows.push_back(row); break; }
284 row += "; eliminated " + election.list(removed);
285 removed = election.ordered(removed, values);
286 } else {
287 Choice chosen = eliminate(election, remaining, values, history);
288 removed = {chosen.candidate};
289 row += "; eliminated " + election.names[chosen.candidate] + tie_text(chosen.tie);
290 }
291 tail.insert(tail.begin(), removed.begin(), removed.end());
292 remaining = select(remaining, [&](int candidate) { return !contains(removed, candidate); });
293 history.push_back(values);
294 rows.push_back(row);
295 }
296 Field order = election.preferred(remaining);
297 order.insert(order.end(), tail.begin(), tail.end());
298 return finish(election, order, remaining.size() > 1, rows);
299}
300int condorcet_winner(const Field& field, const Matrix& pairwise, bool loser = false) {
301 for (int candidate : field) {
302 bool all = true;
303 for (int other : field) if (other != candidate) {
304 if (loser ? pairwise[candidate][other] >= pairwise[other][candidate] : pairwise[candidate][other] <= pairwise[other][candidate]) all = false;
305 }
306 if (all) return candidate;
307 }
308 return -1;
309}
310Relation closure(const Field& field, Relation edges) {
311 for (int middle : field) for (int first : field) for (int second : field) {
312 if (edges[first][middle] && edges[middle][second]) edges[first][second] = true;
313 }
314 return edges;
315}
316Result condorcet(const Election& election, const Field& field, const Matrix& pairwise) {
317 int winner = condorcet_winner(field, pairwise), loser = condorcet_winner(field, pairwise, true);
318 Relation weak = election.relation(), strong = election.relation();
319 for (int first : field) for (int second : field) {
320 weak[first][second] = first == second || pairwise[first][second] >= pairwise[second][first];
321 strong[first][second] = first != second && pairwise[first][second] > pairwise[second][first];
322 }
323 weak = closure(field, weak);
324 strong = closure(field, strong);
325 Field smith, schwartz;
326 for (int candidate : field) {
327 bool top = true, unbeaten = true;
328 for (int other : field) {
329 if (!weak[candidate][other]) top = false;
330 if (strong[other][candidate] && !strong[candidate][other]) unbeaten = false;
331 }
332 if (top) smith.push_back(candidate);
333 if (unbeaten) schwartz.push_back(candidate);
334 }
335 return {{}, false, {"winner: " + (winner < 0 ? string("NONE") : election.names[winner]), "loser: " + (loser < 0 ? string("NONE") : election.names[loser]), "smith: " + election.list(smith), "schwartz: " + election.list(schwartz)}, {}};

337std::pair<Field, bool> topological(const Election& election, const Field& field, const Relation& beats) {
338 Field remaining = field, standings;
339 bool tie = false;
340 while (!remaining.empty()) {
341 Field eligible = select(remaining, [&](int candidate) {
342 return std::none_of(remaining.begin(), remaining.end(), [&](int other) { return candidate != other && beats[other][candidate]; });
343 });
344 if (standings.empty()) tie = eligible.size() > 1;
345 int candidate = election.preferred(eligible)[0];
346 standings.push_back(candidate);
347 remaining = without(remaining, candidate);
348 }
349 return {standings, tie};
350}
351Result schulze(const Election& election, const Field& field, const Matrix& pairwise) {
352 Matrix strengths = election.matrix();
353 for (int first : field) for (int second : field) {
354 if (first != second && pairwise[first][second] > pairwise[second][first]) strengths[first][second] = pairwise[first][second];
355 }
356 for (int middle : field) for (int first : field) if (first != middle) {
357 for (int second : field) if (second != middle && second != first) strengths[first][second] = std::max(strengths[first][second], std::min(strengths[first][middle], strengths[middle][second]));
358 }
359 Relation beats = election.relation();
360 for (int first : field) for (int second : field) beats[first][second] = strengths[first][second] > strengths[second][first];
361 auto [standings, tie] = topological(election, field, beats);
362 Lines rows = pair_lines(election, field, strengths);
363 rows.push_back("ranking: " + election.list(standings, " > "));
364 return finish(election, standings, tie, rows);
365}
366Result ranked_pairs(const Election& election, const Field& field, const Matrix& pairwise) {
367 vector<std::pair<int, int>> pairs;
368 Relation edges = election.relation();
369 Lines rows;
370 for (int first : field) for (int second : field) if (pairwise[first][second] > pairwise[second][first]) pairs.emplace_back(first, second);
371 std::sort(pairs.begin(), pairs.end(), [&](auto first, auto second) {
372 auto [a, b] = first;
373 auto [c, d] = second;
374 if (pairwise[a][b] != pairwise[c][d]) return pairwise[a][b] > pairwise[c][d];
375 if (pairwise[b][a] != pairwise[d][c]) return pairwise[b][a] < pairwise[d][c];
376 if (a != c) return election.tie_rank[a] < election.tie_rank[c];
377 return election.tie_rank[b] < election.tie_rank[d];
378 });
379 for (auto [first, second] : pairs) {
380 bool lock = !closure(field, edges)[second][first];
381 if (lock) edges[first][second] = true;
382 rows.push_back(string(lock ? "lock " : "skip ") + election.names[first] + ">" + election.names[second] + " " + to_string(pairwise[first][second]) + "-" + to_string(pairwise[second][first]));
383 }
384 auto [standings, tie] = topological(election, field, edges);
385 return finish(election, standings, tie, rows);
386}
387Result kemeny(const Election& election, const Field& field, const Matrix& pairwise) {
388 struct Optimum { int score = 0; int count = 0; Field first; };
389 int mask = field_mask(field);
390 vector<Optimum> optimum(1 << election.names.size());
391 optimum[0].count = 1;
392 for (int subset = 1; subset <= mask; ++subset) {
393 if ((subset & mask) != subset) continue;
394 Optimum& current = optimum[subset];
395 current.score = -1;
396 for (int candidate : field) if (subset & (1 << candidate)) {
397 int rest = subset ^ (1 << candidate), score = optimum[rest].score;
398 for (int other : field) if (rest & (1 << other)) score += pairwise[candidate][other];
399 if (score > current.score) { current.score = score; current.count = 0; current.first.clear(); }
400 if (score == current.score) { current.count += optimum[rest].count; current.first.push_back(candidate); }
401 }
402 }
403 Field standings;
404 for (int remaining = mask; remaining != 0;) {
405 int candidate = election.preferred(optimum[remaining].first)[0];
406 standings.push_back(candidate);
407 remaining ^= 1 << candidate;
408 }
409 const Optimum& best = optimum[mask];
410 return finish(election, standings, best.first.size() > 1, {"score: " + to_string(best.score), "optimal: " + to_string(best.count), "ranking: " + election.list(standings, " > ")});
411}
412Result stv(Election& election, const Field& field, int seats) {
413 if (seats >= static_cast<int>(field.size())) fail("too many seats");
414 const Aggregate& aggregate = election.aggregate(field);
415 vector<Ballot> groups;
416 for (const Ballot& group : aggregate.groups) if (!group.ranking.empty()) groups.push_back(group);
417 int quota = 100 * (aggregate.total - aggregate.blank) / (seats + 1) + 1;
418 Values per_vote(groups.size(), 100);
419 Field continuing = field, elected;
420 vector<Values> history;
421 Lines rows{"quota: " + money(quota)};
422 for (int round = 1; ; ++round) {
423 Values values = election.zeros();
424 Field current(groups.size(), -1);
425 int exhausted = 0;
426 for (std::size_t i = 0; i < groups.size(); ++i) {
427 for (int candidate : groups[i].ranking) if (contains(continuing, candidate)) { current[i] = candidate; break; }
428 int value = groups[i].votes * per_vote[i];
429 if (current[i] < 0) exhausted += value;
430 else values[current[i]] += value;
431 }
432 string row = "round " + to_string(round) + ": " + election.values(continuing, values, true) + "; exhausted " + money(exhausted);
433 Field ordered = election.ordered(continuing, values);
434 if (static_cast<int>(continuing.size()) == seats - static_cast<int>(elected.size())) {
435 elected.insert(elected.end(), ordered.begin(), ordered.end());
436 rows.push_back(row + "; elected " + election.list(ordered));
437 break;
438 }
439 int removed;
440 if (values[ordered[0]] >= quota) {
441 removed = ordered[0];
442 elected.push_back(removed);
443 int total = values[removed], surplus = total - quota;
444 for (std::size_t i = 0; i < current.size(); ++i) if (current[i] == removed) per_vote[i] = per_vote[i] * surplus / total;
445 row += "; elected " + election.names[removed] + " (surplus " + money(surplus) + ")";
446 } else {
447 Choice chosen = eliminate(election, continuing, values, history);
448 removed = chosen.candidate;
449 row += "; excluded " + election.names[removed] + tie_text(chosen.tie);
450 }
451 rows.push_back(row);
452 if (static_cast<int>(elected.size()) == seats) break;
453 history.push_back(values);
454 continuing = without(continuing, removed);
455 }
456 rows.push_back("elected: " + election.list(elected));
457 return {{}, false, std::move(rows), {}};
458}
459Result tally(Election& election, Method method, const Field& field) {
460 const Aggregate& aggregate = election.aggregate(field);
461 const Matrix& pairwise = aggregate.pairwise;
462 switch (method.kind) {
463 case Kind::Plurality: return scored(election, field, aggregate.first, "votes");
464 case Kind::Runoff: return runoff(election, field);
465 case Kind::Irv: return elimination_rounds(election, field, false);
466 case Kind::Coombs: return elimination_rounds(election, field, true);
467 case Kind::Bucklin: return bucklin(election, field);
468 case Kind::Borda: return scored(election, field, aggregate.borda, "scores");
469 case Kind::Baldwin: return borda_rounds(election, field, false);
470 case Kind::Nanson: return borda_rounds(election, field, true);
471 case Kind::Black: {
472 int winner = condorcet_winner(field, pairwise);
473 Field ordered = election.ordered(field, aggregate.borda);
474 bool tie = winner < 0 && aggregate.borda[ordered[0]] == aggregate.borda[ordered[1]];
475 if (winner >= 0) { ordered = without(ordered, winner); ordered.insert(ordered.begin(), winner); }
476 return finish(election, ordered, tie, {"condorcet: " + (winner < 0 ? string("NONE") : election.names[winner]), "scores: " + election.values(field, aggregate.borda)});
477 }
478 case Kind::Veto: return scored(election, field, aggregate.last, "vetoes", true);
479 case Kind::Condorcet: return condorcet(election, field, pairwise);
480 case Kind::Copeland: {
481 Values scores = election.zeros();
482 Lines parts;
483 for (int candidate : field) {
484 int wins = 0, draws = 0, losses = 0;
485 for (int other : field) if (other != candidate) {
486 if (pairwise[candidate][other] > pairwise[other][candidate]) ++wins;
487 else if (pairwise[candidate][other] == pairwise[other][candidate]) ++draws;
488 else ++losses;
489 }
490 int score = 2 * wins + draws;
491 scores[candidate] = score;
492 parts.push_back(election.names[candidate] + " " + to_string(score / 2) + "." + (score % 2 ? "5" : "0") + " (" + to_string(wins) + "-" + to_string(draws) + "-" + to_string(losses) + ")");
493 }
494 Field standings = election.ordered(field, scores);
495 return finish(election, standings, scores[standings[0]] == scores[standings[1]], {"scores: " + join(parts)});
496 }
497 case Kind::Minimax: {
498 Values scores = election.zeros();
499 for (int candidate : field) for (int other : field) if (candidate != other) {
500 int opposition = pairwise[other][candidate], support = pairwise[candidate][other];
501 if (method.score == Minimax::Opposition) scores[candidate] = std::max(scores[candidate], opposition);
502 else if (opposition > support) scores[candidate] = std::max(scores[candidate], method.score == Minimax::Margin ? opposition - support : opposition);
503 }
504 return scored(election, field, scores, "scores", true);
505 }
506 case Kind::Schulze: return schulze(election, field, pairwise);
507 case Kind::RankedPairs: return ranked_pairs(election, field, pairwise);
508 case Kind::Kemeny: return kemeny(election, field, pairwise);
509 case Kind::Approval: {
510 Values approvals = election.zeros();
511 int depth = std::min(method.parameter, static_cast<int>(field.size()));
512 for (int candidate : field) approvals[candidate] = std::accumulate(aggregate.positions[candidate].begin(), aggregate.positions[candidate].begin() + depth, 0);
513 return scored(election, field, approvals, "approvals");
514 }
515 case Kind::Stv: return stv(election, field, method.parameter);
516 }
517 throw std::logic_error("Unknown method");
518}
519Lines reports(Election& election, const string& command, const Lines& args, const Field& field) {
520 const Aggregate& aggregate = election.aggregate(field);
521 if (command == "STATUS") {
522 Field withdrawn;
523 for (int candidate = 0; candidate < static_cast<int>(election.names.size()); ++candidate) if (election.withdrawn[candidate]) withdrawn.push_back(candidate);
524 return {"election " + to_string(election.number), "active: " + election.list(field), "withdrawn: " + (withdrawn.empty() ? "none" : election.list(withdrawn)), "tiebreak: " + election.list(election.priority), "ballots: " + to_string(aggregate.total) + " in " + to_string(election.entries.size()) + " entries"};
525 }
526 if (command == "BALLOTS") {
527 vector<std::pair<int, string>> groups;
528 for (const Ballot& group : aggregate.groups) groups.emplace_back(group.votes, group.ranking.empty() ? "-" : election.list(group.ranking, ">"));
529 std::sort(groups.begin(), groups.end(), [](const auto& a, const auto& b) { return a.first == b.first ? a.second < b.second : a.first > b.first; });
530 Lines rows{"total " + to_string(aggregate.total) + ", blank " + to_string(aggregate.blank) + ", distinct " + to_string(groups.size())};
531 for (const auto& [votes, notation] : groups) rows.push_back(to_string(votes) + " " + notation);
532 return rows;
533 }
534 if (command == "PROFILE") {
535 Lines rows;
536 for (int candidate : field) {
537 Lines counts;
538 for (int count : aggregate.positions[candidate]) counts.push_back(to_string(count));
539 rows.push_back(election.names[candidate] + ": " + join(counts, " ") + " unranked " + to_string(aggregate.total - sum(aggregate.positions[candidate])));
540 }
541 return rows;
542 }
543 if (command == "PAIRWISE") return pair_lines(election, field, aggregate.pairwise);
544 if (command == "EXHAUSTED") return elimination_rounds(election, field, false).exhausted;
545 if (command == "RANKING" || command == "SPOILERS") {
546 Method method = parse_method(args);
547 if (method.kind == Kind::Condorcet || method.kind == Kind::Stv) fail("not rankable");
548 if (command == "SPOILERS" && field.size() < 3) fail("too few candidates");
549 Field standings = tally(election, method, field).standings;
550 if (command == "RANKING") return {election.list(standings, " > ")};
551 int winner = standings[0];
552 Lines rows{"winner: " + election.names[winner]};
553 for (int excluded : field) if (excluded != winner) {
554 int other = tally(election, method, without(field, excluded)).standings[0];
555 rows.push_back("without " + election.names[excluded] + ": " + election.names[other] + (other != winner ? " (changed)" : ""));
556 }
557 return rows;
558 }
559 if (command == "SUMMARY") {
560 Values wins = election.zeros();
561 Lines rows;
562 for (Method method : summary_methods) {
563 int winner = tally(election, method, field).standings[0];
564 ++wins[winner];
565 rows.push_back(method_text(method) + ": " + election.names[winner]);
566 }
567 rows.push_back("wins: " + election.values(field, wins));
568 return rows;
569 }
570 if (command == "COMPARE") {
571 Field candidates;
572 for (const string& name : args) {
573 int candidate = election.candidate(name);
574 if (election.withdrawn[candidate]) fail("withdrawn candidate " + name);
575 candidates.push_back(candidate);
576 }
577 int first = candidates[0], second = candidates[1];
578 if (first == second) fail("same candidate");
579 int a = aggregate.pairwise[first][second], b = aggregate.pairwise[second][first];
580 std::array<Lines, 2> ahead;
581 for (Method method : summary_methods) {
582 Field standings = tally(election, method, field).standings;
583 int side = std::find(standings.begin(), standings.end(), first) < std::find(standings.begin(), standings.end(), second) ? 0 : 1;
584 ahead[side].push_back(method_text(method));
585 }
586 return {args[0] + " " + to_string(a) + ", " + args[1] + " " + to_string(b) + ", neither " + to_string(aggregate.total - a - b), args[0] + " ahead: " + (ahead[0].empty() ? "none" : join(ahead[0])), args[1] + " ahead: " + (ahead[1].empty() ? "none" : join(ahead[1]))};
587 }
588 Lines words{command};
589 words.insert(words.end(), args.begin(), args.end());
590 return tally(election, parse_method(words), field).rows;
591}

593class VoteCounter {
594 std::optional<Election> election_;
595 int number_ = 0;
596public:
597 Lines execute(const string& line) {
598 Lines words = split(line), args(words.begin() + 1, words.end());
599 const string& command = words[0];
600 if (command == "CANDIDATES") {
601 if (args.size() < 2) fail("too few candidates");
602 std::set<string> seen;
603 for (const string& name : args) if (!seen.insert(name).second) fail("duplicate candidate " + name);
604 election_.emplace(args, ++number_);
605 return {"election " + to_string(number_) + ": " + to_string(args.size()) + " candidates"};
606 }
607 if (!election_) fail("no election");
608 Election& election = *election_;
609 Field field = election.active();
610 if (command == "BALLOT") {
611 Field ranking = args[1] == "-" ? Field{} : election.checked_names(split(args[1], '>'));
612 election.entries.push_back({{std::stoi(args[0]), ranking}});
613 election.cache.clear();
614 return {};
615 }
616 if (command == "RANDOM") {
617 std::int64_t state = std::stoll(args[1]);
618 auto draw = [&]() { state = 48271 * state % 2147483647; return state; };
619 vector<Ballot> groups;
620 for (int group = 0; group < std::stoi(args[0]); ++group) {
621 Field ranking = field;
622 for (int index = static_cast<int>(ranking.size()) - 1; index >= 1; --index) {
623 int other = static_cast<int>(draw() % (index + 1));
624 std::swap(ranking[index], ranking[other]);
625 }
626 ranking.resize(static_cast<std::size_t>(draw() % static_cast<int>(ranking.size()) + 1));
627 groups.push_back({1, std::move(ranking)});
628 }
629 election.entries.push_back(std::move(groups));
630 election.cache.clear();
631 return {};
632 }
633 if (command == "UNDO") {
634 if (election.entries.empty()) fail("nothing to undo");
635 election.entries.pop_back();
636 election.cache.clear();
637 return {};
638 }
639 if (command == "WITHDRAW" || command == "RESTORE") {
640 int candidate = election.candidate(args[0]);
641 if (command == "WITHDRAW") {
642 if (election.withdrawn[candidate]) fail("already withdrawn " + args[0]);
643 if (field.size() <= 2) fail("too few candidates");
644 election.withdrawn[candidate] = true;
645 } else {
646 if (!election.withdrawn[candidate]) fail("not withdrawn " + args[0]);
647 election.withdrawn[candidate] = false;
648 }
649 return {};
650 }
651 if (command == "TIEBREAK") {
652 Field order = election.checked_names(args);
653 if (order.size() != election.names.size()) fail("incomplete tiebreak");
654 election.set_priority(order);
655 return {};
656 }
657 if (command != "STATUS" && election.entries.empty()) fail("no ballots");
658 Lines rows = reports(election, command, args, field);
659 rows.insert(rows.begin(), "[" + line + "]");
660 return rows;
661 }
662};

666 std::cin.tie(nullptr);
667 int count;
668 std::cin >> count;

678 }
