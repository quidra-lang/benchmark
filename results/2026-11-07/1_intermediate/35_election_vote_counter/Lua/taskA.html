35local function ascii_less(a, b)
36 for i = 1, math.min(#a, #b) do
37 local x, y = a:byte(i), b:byte(i)
38 if x ~= y then return x < y end
39 end
40 return #a < #b
41end

50 start = start or 1

66Election.__index = Election
67function Election.new(number, names)
68 local self = setmetatable({number = number, names = names, withdrawn = {}, tie_order = {},
69 priority = {}, entries = {}, cache = {}}, Election)
70 for i = 1, #names do self.withdrawn[i], self.tie_order[i] = false, i end
71 table.sort(self.tie_order, function(a, b) return ascii_less(names[a], names[b]) end)
72 for position, candidate in ipairs(self.tie_order) do self.priority[candidate] = position end
73 return self
74end
75function Election:active()
76 local field = {}
77 for candidate = 1, #self.names do if not self.withdrawn[candidate] then field[#field + 1] = candidate end end
78 return field
79end
80function Election:candidate(name)
81 for i, candidate in ipairs(self.names) do if name == candidate then return i end end
82 fail("unknown candidate " .. name)
83end
84function Election:checked_names(names)
85 local result = {}
86 for _, name in ipairs(names) do
87 local candidate = self:candidate(name)
88 if contains(result, candidate) then fail("duplicate candidate " .. name) end
89 result[#result + 1] = candidate
90 end
91 return result
92end
93function Election:name_list(field, separator)
94 local names = {}
95 for _, candidate in ipairs(field) do names[#names + 1] = self.names[candidate] end
96 return table.concat(names, separator or ", ")
97end
98function Election:ordered(field, values, ascending)
99 local result = copy(field)
100 table.sort(result, function(a, b)
101 if values[a] == values[b] then return self.priority[a] < self.priority[b] end
102 if ascending then return values[a] < values[b] end
103 return values[a] > values[b]
104 end)
105 return result
106end
107function Election:preferred(field)
108 local result = copy(field)
109 table.sort(result, function(a, b) return self.priority[a] < self.priority[b] end)
110 return result
111end
112function Election:value_list(field, values, cents)
113 local parts = {}
114 for _, candidate in ipairs(field) do
115 parts[#parts + 1] = self.names[candidate] .. " " .. (cents and money(values[candidate]) or values[candidate])
116 end
117 return table.concat(parts, ", ")
118end
119function Election:aggregate(field)
120 local mask = 0
121 for _, candidate in ipairs(field) do mask = mask | (1 << (candidate - 1)) end
122 if self.cache[mask] then return self.cache[mask] end
123 local groups, by_ranking = {}, {}
124 for _, entry in ipairs(self.entries) do
125 for _, group in ipairs(entry) do
126 local ranking = filtered(group.ranking, function(c) return mask & (1 << (c - 1)) ~= 0 end)
127 local code = 1
128 for _, candidate in ipairs(ranking) do code = 8 * code + candidate end
129 if by_ranking[code] then by_ranking[code].votes = by_ranking[code].votes + group.votes
130 else
131 local merged = {votes = group.votes, ranking = ranking}
132 by_ranking[code], groups[#groups + 1] = merged, merged
133 end
134 end
135 end
136 local n = #self.names
137 local data = {groups = groups, first = zeros(n), last = zeros(n), borda = zeros(n),
138 positions = matrix(n, #field, 0), pairwise = matrix(n, n, 0), total = 0, blank = 0}
139 for _, group in ipairs(groups) do
140 local ranking, votes = group.ranking, group.votes
141 data.total = data.total + votes
142 if #ranking == 0 then data.blank = data.blank + votes
143 else data.first[ranking[1]] = data.first[ranking[1]] + votes end
144 local ranks = {}
145 for i = 1, n do ranks[i] = #field + 1 end
146 for place, candidate in ipairs(ranking) do
147 ranks[candidate] = place
148 data.positions[candidate][place] = data.positions[candidate][place] + votes
149 data.borda[candidate] = data.borda[candidate] + votes * (#field - place)
150 end
151 for _, candidate in ipairs(field) do
152 if (#ranking < #field and ranks[candidate] == #field + 1)
153 or (#ranking == #field and candidate == ranking[#ranking]) then
154 data.last[candidate] = data.last[candidate] + votes
155 end
156 for _, other in ipairs(field) do
157 if ranks[candidate] < ranks[other] then data.pairwise[candidate][other] = data.pairwise[candidate][other] + votes end
158 end
159 end
160 end
161 self.cache[mask] = data
162 return data
163end

167function Tally.new(election, field)
168 field = field or election:active()
169 return setmetatable({election = election, field = field, data = election:aggregate(field)}, Tally)
170end
171function Tally:finish(order, tie, lines, exhaustion)
172 lines[#lines + 1] = "winner: " .. self.election.names[order[1]] .. tie_text(tie)
173 return {order = order, lines = lines, exhaustion = exhaustion}
174end
175function Tally:scored(values, label, ascending)
176 local order = self.election:ordered(self.field, values, ascending)
177 return self:finish(order, values[order[1]] == values[order[2]], {label .. ": " .. self.election:value_list(self.field, values)})
178end
179function Tally:eliminate(field, values, history)
180 local minimum = values[self.election:ordered(field, values, true)[1]]
181 local pool = filtered(field, function(c) return values[c] == minimum end)
182 local tie = #pool > 1
183 for round = #history, 1, -1 do
184 local old = history[round]
185 minimum = old[self.election:ordered(pool, old, true)[1]]
186 pool = filtered(pool, function(c) return old[c] == minimum end)
187 end
188 pool = self.election:preferred(pool)
189 return pool[#pool], tie
190end
191function Tally:pair_lines(values)
192 local lines, names = {}, self.election.names
193 values = values or self.data.pairwise
194 for i, x in ipairs(self.field) do
195 for j = i + 1, #self.field do
196 local y = self.field[j]
197 local a, b = values[x][y], values[y][x]
198 local winner = a == b and "tie" or names[a > b and x or y]
199 lines[#lines + 1] = names[x] .. " vs " .. names[y] .. ": " .. a .. "-" .. b .. " (" .. winner .. ")"
200 end
201 end
202 return lines
203end
204function Tally:runoff()
205 local e, first = self.election, self.data.first
206 local order = e:ordered(self.field, first)
207 local lines = {"round 1: " .. e:value_list(self.field, first)}
208 if 2 * first[order[1]] > sum(first) then return self:finish(order, false, lines) end
209 local finalists = {order[1], order[2]}
210 table.sort(finalists)
211 local advance = #order > 2 and first[order[2]] == first[order[3]]
212 local second = e:aggregate(finalists).first
213 local tie = second[order[1]] == second[order[2]]
214 if second[order[2]] > second[order[1]] then order[1], order[2] = order[2], order[1] end
215 lines[#lines + 1] = "finalists: " .. e:name_list(finalists) .. tie_text(advance)
216 lines[#lines + 1] = "round 2: " .. e:value_list(finalists, second)
217 return self:finish(order, tie, lines)
218end
219function Tally:instant_runoff(coombs)
220 local e, remaining = self.election, copy(self.field)
221 local removed, history, lines, exhaustion = {}, {}, {}, {}
222 for round = 1, #self.field do
223 local data = e:aggregate(remaining)
224 local first, order = data.first, e:ordered(remaining, data.first)
225 local active = sum(first)
226 local line = "round " .. round .. ": " .. e:value_list(remaining, first)
227 exhaustion[#exhaustion + 1] = "round " .. round .. ": active " .. active .. ", exhausted " .. (data.total - active)
228 if #remaining == 1 or 2 * first[order[1]] > active then
229 for i = #removed, 1, -1 do order[#order + 1] = removed[i] end
230 lines[#lines + 1] = line
231 return self:finish(order, false, lines, exhaustion)
232 end
233 local loser, tie
234 if coombs then
235 local high = data.last[e:ordered(remaining, data.last)[1]]
236 local pool = filtered(remaining, function(c) return data.last[c] == high end)
237 pool = e:ordered(pool, first)
238 loser, tie = pool[#pool], #pool > 1
239 line = line .. "; last " .. e:value_list(remaining, data.last)
240 else loser, tie = self:eliminate(remaining, first, history) end
241 lines[#lines + 1] = line .. "; eliminated " .. e.names[loser] .. tie_text(tie)
242 history[#history + 1], removed[#removed + 1] = first, loser
243 remaining = without(remaining, loser)
244 end
245end
246function Tally:approval(depth)
247 local values = zeros(#self.election.names)
248 for _, candidate in ipairs(self.field) do
249 for position = 1, math.min(depth, #self.field) do values[candidate] = values[candidate] + self.data.positions[candidate][position] end
250 end
251 return values
252end
253function Tally:bucklin()
254 local e, history = self.election, {}
255 local majority = (self.data.total - self.data.blank) // 2 + 1
256 local lines = {"majority: " .. majority}
257 for depth = 1, #self.field do
258 local values = self:approval(depth)
259 history[#history + 1] = values
260 lines[#lines + 1] = "round " .. depth .. ": " .. e:value_list(self.field, values)
261 if values[e:ordered(self.field, values)[1]] >= majority then break end
262 end
263 local order = copy(self.field)
264 table.sort(order, function(a, b)
265 for round = #history, 1, -1 do
266 if history[round][a] ~= history[round][b] then return history[round][a] > history[round][b] end
267 end
268 return e.priority[a] < e.priority[b]
269 end)
270 return self:finish(order, history[#history][order[1]] == history[#history][order[2]], lines)
271end
272function Tally:borda_rounds(nanson)
273 local e, remaining = self.election, copy(self.field)
274 local tail, history, lines = {}, {}, {}
275 while #remaining > 1 do
276 local values = e:aggregate(remaining).borda
277 local line = "round " .. (#lines + 1) .. ": " .. e:value_list(remaining, values)
278 local losers
279 if nanson then
280 local total = sum(values)
281 losers = filtered(remaining, function(c) return values[c] * #remaining < total end)
282 if #losers == 0 then lines[#lines + 1] = line; break end
283 line = line .. "; eliminated " .. e:name_list(losers)
284 else
285 local loser, tie = self:eliminate(remaining, values, history)
286 losers = {loser}
287 line = line .. "; eliminated " .. e.names[loser] .. tie_text(tie)
288 end
289 lines[#lines + 1], history[#history + 1] = line, values
290 tail = append(e:ordered(losers, values), tail)
291 remaining = filtered(remaining, function(c) return not contains(losers, c) end)
292 end
293 return self:finish(append(e:preferred(remaining), tail), nanson and #remaining > 1, lines)
294end
295function Tally:condorcet(loser)
296 for _, x in ipairs(self.field) do
297 local all = true
298 for _, y in ipairs(self.field) do
299 if x ~= y then
300 local a, b = self.data.pairwise[x][y], self.data.pairwise[y][x]
301 if (loser and a >= b) or (not loser and a <= b) then all = false end
302 end
303 end
304 if all then return x end
305 end
306end
307function Tally:black()
308 local e, values = self.election, self.data.borda
309 local winner, order = self:condorcet(), e:ordered(self.field, values)
310 local tie = not winner and values[order[1]] == values[order[2]]
311 if winner then order = append({winner}, without(order, winner)) end
312 return self:finish(order, tie, {"condorcet: " .. (winner and e.names[winner] or "NONE"), "scores: " .. e:value_list(self.field, values)})
313end
314function Tally:closure(edges)
315 for _, k in ipairs(self.field) do
316 for _, x in ipairs(self.field) do
317 for _, y in ipairs(self.field) do edges[x][y] = edges[x][y] or (edges[x][k] and edges[k][y]) end
318 end
319 end
320 return edges
321end
322function Tally:condorcet_report()
323 local n, e = #self.election.names, self.election
324 local weak, strong = matrix(n, n, false), matrix(n, n, false)
325 for _, x in ipairs(self.field) do
326 for _, y in ipairs(self.field) do
327 weak[x][y] = self.data.pairwise[x][y] >= self.data.pairwise[y][x]
328 strong[x][y] = self.data.pairwise[x][y] > self.data.pairwise[y][x]
329 end
330 end
331 self:closure(weak); self:closure(strong)
332 local smith, schwartz = {}, {}
333 for _, x in ipairs(self.field) do
334 local top, unbeaten = true, true
335 for _, y in ipairs(self.field) do
336 if not weak[x][y] then top = false end
337 if strong[y][x] and not strong[x][y] then unbeaten = false end
338 end
339 if top then smith[#smith + 1] = x end
340 if unbeaten then schwartz[#schwartz + 1] = x end
341 end
342 local winner, loser = self:condorcet(), self:condorcet(true)
343 return {order = {}, lines = {"winner: " .. (winner and e.names[winner] or "NONE"),
344 "loser: " .. (loser and e.names[loser] or "NONE"), "smith: " .. e:name_list(smith), "schwartz: " .. e:name_list(schwartz)}}
345end
346function Tally:copeland()
347 local e, values, parts = self.election, zeros(#self.election.names), {}
348 for _, x in ipairs(self.field) do
349 local wins, draws, losses = 0, 0, 0
350 for _, y in ipairs(self.field) do
351 if x ~= y then
352 local a, b = self.data.pairwise[x][y], self.data.pairwise[y][x]
353 if a > b then wins = wins + 1 elseif a == b then draws = draws + 1 else losses = losses + 1 end
354 end
355 end
356 values[x] = 2 * wins + draws
357 parts[#parts + 1] = string.format("%s %d.%d (%d-%d-%d)", e.names[x], values[x] // 2, 5 * (values[x] % 2), wins, draws, losses)
358 end
359 local order = e:ordered(self.field, values)
360 return self:finish(order, values[order[1]] == values[order[2]], {"scores: " .. table.concat(parts, ", ")})
361end
362function Tally:minimax(kind)
363 local values = zeros(#self.election.names)
364 for _, x in ipairs(self.field) do
365 for _, y in ipairs(self.field) do
366 if x ~= y then
367 local against, support = self.data.pairwise[y][x], self.data.pairwise[x][y]
368 if kind == "OPPOSITION" or against > support then
369 values[x] = math.max(values[x], kind == "MARGIN" and against - support or against)
370 end
371 end
372 end
373 end
374 return self:scored(values, "scores", true)
375end
376function Tally:dominance_order(dominates)
377 local remaining, order, tie = copy(self.field), {}, false
378 while #remaining > 0 do
379 local eligible = filtered(remaining, function(x)
380 for _, y in ipairs(remaining) do if x ~= y and dominates(y, x) then return false end end
381 return true
382 end)
383 if #order == 0 then tie = #eligible > 1 end
384 local chosen = self.election:preferred(eligible)[1]
385 order[#order + 1] = chosen
386 remaining = without(remaining, chosen)
387 end
388 return order, tie
389end
390function Tally:schulze()
391 local paths, d = matrix(#self.election.names, #self.election.names, 0), self.data.pairwise
392 for _, x in ipairs(self.field) do
393 for _, y in ipairs(self.field) do if d[x][y] > d[y][x] then paths[x][y] = d[x][y] end end
394 end
395 for _, i in ipairs(self.field) do
396 for _, j in ipairs(self.field) do
397 for _, k in ipairs(self.field) do
398 if i ~= j and i ~= k and j ~= k then paths[j][k] = math.max(paths[j][k], math.min(paths[j][i], paths[i][k])) end
399 end
400 end
401 end
402 local order, tie = self:dominance_order(function(x, y) return paths[x][y] > paths[y][x] end)
403 local lines = self:pair_lines(paths)
404 lines[#lines + 1] = "ranking: " .. self.election:name_list(order, " > ")
405 return self:finish(order, tie, lines)
406end
407function Tally:ranked_pairs()
408 local e, d, pairs = self.election, self.data.pairwise, {}
409 for _, x in ipairs(self.field) do
410 for _, y in ipairs(self.field) do if d[x][y] > d[y][x] then pairs[#pairs + 1] = {x, y} end end
411 end
412 table.sort(pairs, function(a, b)
413 if d[a[1]][a[2]] ~= d[b[1]][b[2]] then return d[a[1]][a[2]] > d[b[1]][b[2]] end
414 if d[a[2]][a[1]] ~= d[b[2]][b[1]] then return d[a[2]][a[1]] < d[b[2]][b[1]] end
415 if a[1] ~= b[1] then return e.priority[a[1]] < e.priority[b[1]] end
416 return e.priority[a[2]] < e.priority[b[2]]
417 end)
418 local locked, reached, lines = matrix(#e.names, #e.names, false), matrix(#e.names, #e.names, false), {}
419 for _, pair in ipairs(pairs) do
420 local x, y = pair[1], pair[2]
421 local lock = not reached[y][x]
422 if lock then locked[x][y], reached[x][y] = true, true; self:closure(reached) end
423 lines[#lines + 1] = (lock and "lock " or "skip ") .. e.names[x] .. ">" .. e.names[y] .. " " .. d[x][y] .. "-" .. d[y][x]
424 end
425 local order, tie = self:dominance_order(function(x, y) return locked[x][y] end)
426 return self:finish(order, tie, lines)
427end
428function Tally:kemeny()
429 local mask = 0
430 for _, c in ipairs(self.field) do mask = mask | (1 << (c - 1)) end
431 local best, ways, heads = {[0] = 0}, {[0] = 1}, {[0] = {}}
432 for subset = 1, mask do
433 if subset & mask == subset then
434 best[subset], ways[subset], heads[subset] = -1, 0, {}
435 for _, candidate in ipairs(self.field) do
436 local bit = 1 << (candidate - 1)
437 if subset & bit ~= 0 then
438 local rest = subset ~ bit
439 local score = best[rest]
440 for _, other in ipairs(self.field) do
441 if rest & (1 << (other - 1)) ~= 0 then score = score + self.data.pairwise[candidate][other] end
442 end
443 if score > best[subset] then best[subset], ways[subset], heads[subset] = score, 0, {} end
444 if score == best[subset] then
445 ways[subset] = ways[subset] + ways[rest]
446 heads[subset][#heads[subset] + 1] = candidate
447 end
448 end
449 end
450 end
451 end
452 local remaining, order = mask, {}
453 while remaining ~= 0 do
454 local candidate = self.election:preferred(heads[remaining])[1]
455 order[#order + 1] = candidate
456 remaining = remaining ~ (1 << (candidate - 1))
457 end
458 return self:finish(order, #heads[mask] > 1, {"score: " .. best[mask], "optimal: " .. ways[mask],
459 "ranking: " .. self.election:name_list(order, " > ")})
460end
461function Tally:stv(seats)
462 local e = self.election
463 local groups = filtered(self.data.groups, function(group) return #group.ranking > 0 end)
464 local quota = 100 * (self.data.total - self.data.blank) // (seats + 1) + 1
465 local per_vote = {}
466 for i = 1, #groups do per_vote[i] = 100 end
467 local continuing, elected, history = copy(self.field), {}, {}
468 local lines = {"quota: " .. money(quota)}
469 for round = 1, #self.field do
470 local values, owners, exhausted = zeros(#e.names), {}, 0
471 for i, group in ipairs(groups) do
472 owners[i] = 0
473 for _, candidate in ipairs(group.ranking) do
474 if contains(continuing, candidate) then owners[i] = candidate; break end
475 end
476 local amount = group.votes * per_vote[i]
477 if owners[i] == 0 then exhausted = exhausted + amount else values[owners[i]] = values[owners[i]] + amount end
478 end
479 local line = "round " .. round .. ": " .. e:value_list(continuing, values, true) .. "; exhausted " .. money(exhausted)
480 local order = e:ordered(continuing, values)
481 if #continuing == seats - #elected then
482 append(elected, order)
483 lines[#lines + 1] = line .. "; elected " .. e:name_list(order)
484 break
485 end
486 local chosen = order[1]
487 if values[chosen] >= quota then
488 local surplus = values[chosen] - quota
489 elected[#elected + 1] = chosen
490 for i, owner in ipairs(owners) do if owner == chosen then per_vote[i] = per_vote[i] * surplus // values[chosen] end end
491 line = line .. "; elected " .. e.names[chosen] .. " (surplus " .. money(surplus) .. ")"
492 else
493 local tie
494 chosen, tie = self:eliminate(continuing, values, history)
495 line = line .. "; excluded " .. e.names[chosen] .. tie_text(tie)
496 end
497 lines[#lines + 1], history[#history + 1] = line, values
498 continuing = without(continuing, chosen)
499 if #elected == seats then break end
500 end
501 lines[#lines + 1] = "elected: " .. e:name_list(elected)
502 return {order = {}, lines = lines}
503end
504function Tally:run(method)
505 local kind = method.kind
506 if kind == "PLURALITY" then return self:scored(self.data.first, "votes")
507 elseif kind == "RUNOFF" then return self:runoff()
508 elseif kind == "IRV" then return self:instant_runoff()
509 elseif kind == "COOMBS" then return self:instant_runoff(true)
510 elseif kind == "BUCKLIN" then return self:bucklin()
511 elseif kind == "BORDA" then return self:scored(self.data.borda, "scores")
512 elseif kind == "BALDWIN" then return self:borda_rounds(false)
513 elseif kind == "NANSON" then return self:borda_rounds(true)
514 elseif kind == "BLACK" then return self:black()
515 elseif kind == "VETO" then return self:scored(self.data.last, "vetoes", true)
516 elseif kind == "CONDORCET" then return self:condorcet_report()
517 elseif kind == "COPELAND" then return self:copeland()
518 elseif kind == "MINIMAX" then return self:minimax(method.minimax)
519 elseif kind == "SCHULZE" then return self:schulze()
520 elseif kind == "RANKEDPAIRS" then return self:ranked_pairs()
521 elseif kind == "KEMENY" then return self:kemeny()
522 elseif kind == "APPROVAL" then return self:scored(self:approval(method.parameter), "approvals")
523 elseif kind == "STV" then return self:stv(method.parameter) end
524end
525function Tally:ballots_report()
526 local e, groups = self.election, {}
527 for _, group in ipairs(self.data.groups) do
528 groups[#groups + 1] = {votes = group.votes, notation = #group.ranking == 0 and "-" or e:name_list(group.ranking, ">")}
529 end
530 table.sort(groups, function(a, b)
531 if a.votes ~= b.votes then return a.votes > b.votes end
532 return ascii_less(a.notation, b.notation)
533 end)
534 local lines = {"total " .. self.data.total .. ", blank " .. self.data.blank .. ", distinct " .. #groups}
535 for _, group in ipairs(groups) do lines[#lines + 1] = group.votes .. " " .. group.notation end
536 return lines
537end
538function Tally:profile()
539 local lines = {}
540 for _, candidate in ipairs(self.field) do
541 local counts = self.data.positions[candidate]
542 lines[#lines + 1] = self.election.names[candidate] .. ": " .. table.concat(counts, " ") .. " unranked " .. (self.data.total - sum(counts))
543 end
544 return lines
545end

547local current, election_count
548election_count = 0
549local function execute(line)
550 local words = split(line)
551 local command = words[1]
552 if command == "CANDIDATES" then
553 local names = {table.unpack(words, 2)}
554 if #names < 2 then fail("too few candidates") end
555 local seen = {}
556 for _, name in ipairs(names) do
557 if seen[name] then fail("duplicate candidate " .. name) end
558 seen[name] = true
559 end
560 election_count = election_count + 1
561 current = Election.new(election_count, names)
562 return {"election " .. election_count .. ": " .. #names .. " candidates"}
563 end
564 if not current then fail("no election") end
565 local e = current
566 if command == "BALLOT" then
567 local ranking = words[3] == "-" and {} or e:checked_names(split(words[3], "[^>]+"))
568 e.entries[#e.entries + 1] = {{votes = tonumber(words[2]), ranking = ranking}}
569 e.cache = {}
570 return {}
571 elseif command == "RANDOM" then
572 local entry, state = {}, tonumber(words[3])
573 for _ = 1, tonumber(words[2]) do
574 local ranking = e:active()
575 for i = #ranking, 2, -1 do
576 state = (48271 * state) % 2147483647
577 local j = state % i + 1
578 ranking[i], ranking[j] = ranking[j], ranking[i]
579 end
580 state = (48271 * state) % 2147483647
581 entry[#entry + 1] = {votes = 1, ranking = {table.unpack(ranking, 1, state % #ranking + 1)}}
582 end
583 e.entries[#e.entries + 1], e.cache = entry, {}
584 return {}
585 elseif command == "UNDO" then
586 if #e.entries == 0 then fail("nothing to undo") end
587 table.remove(e.entries)
588 e.cache = {}
589 return {}
590 elseif command == "WITHDRAW" or command == "RESTORE" then
591 local candidate = e:candidate(words[2])
592 if command == "WITHDRAW" then
593 if e.withdrawn[candidate] then fail("already withdrawn " .. words[2]) end
594 if #e:active() <= 2 then fail("too few candidates") end
595 e.withdrawn[candidate] = true
596 else
597 if not e.withdrawn[candidate] then fail("not withdrawn " .. words[2]) end
598 e.withdrawn[candidate] = false
599 end
600 return {}
601 elseif command == "TIEBREAK" then
602 local order = e:checked_names({table.unpack(words, 2)})
603 if #order ~= #e.names then fail("incomplete tiebreak") end
604 e.tie_order = order
605 for position, candidate in ipairs(order) do e.priority[candidate] = position end
606 return {}
607 end
608 local field = e:active()
609 if command ~= "STATUS" and #e.entries == 0 then fail("no ballots") end
610 local tally, lines = Tally.new(e, field), {}
611 if command == "STATUS" then
612 local withdrawn = {}
613 for candidate = 1, #e.names do if e.withdrawn[candidate] then withdrawn[#withdrawn + 1] = candidate end end
614 lines = {"election " .. e.number, "active: " .. e:name_list(field),
615 "withdrawn: " .. (#withdrawn == 0 and "none" or e:name_list(withdrawn)),
616 "tiebreak: " .. e:name_list(e.tie_order), "ballots: " .. tally.data.total .. " in " .. #e.entries .. " entries"}
617 elseif command == "BALLOTS" then lines = tally:ballots_report()
618 elseif command == "PROFILE" then lines = tally:profile()
619 elseif command == "PAIRWISE" then lines = tally:pair_lines()
620 elseif command == "EXHAUSTED" then lines = tally:instant_runoff().exhaustion
621 elseif command == "RANKING" or command == "SPOILERS" then
622 local method = parse_method(words, 2)
623 if method.kind == "CONDORCET" or method.kind == "STV" then fail("not rankable") end
624 if command == "SPOILERS" and #field < 3 then fail("too few candidates") end
625 local order = tally:run(method).order
626 if command == "RANKING" then lines = {e:name_list(order, " > ")}
627 else
628 local winner = order[1]
629 lines = {"winner: " .. e.names[winner]}
630 for _, candidate in ipairs(field) do
631 if candidate ~= winner then
632 local other = Tally.new(e, without(field, candidate)):run(method).order[1]
633 lines[#lines + 1] = "without " .. e.names[candidate] .. ": " .. e.names[other] .. (other ~= winner and " (changed)" or "")
634 end
635 end
636 end
637 elseif command == "SUMMARY" then
638 local wins = zeros(#e.names)
639 for _, text in ipairs(summary_methods) do
640 local method = parse_method(split(text))
641 local winner = tally:run(method).order[1]
642 wins[winner] = wins[winner] + 1
643 lines[#lines + 1] = method_text(method) .. ": " .. e.names[winner]
644 end
645 lines[#lines + 1] = "wins: " .. e:value_list(field, wins)
646 elseif command == "COMPARE" then
647 local x = e:candidate(words[2])
648 if e.withdrawn[x] then fail("withdrawn candidate " .. words[2]) end
649 local y = e:candidate(words[3])
650 if e.withdrawn[y] then fail("withdrawn candidate " .. words[3]) end
651 if x == y then fail("same candidate") end
652 local ahead = {{}, {}}
653 for _, text in ipairs(summary_methods) do
654 local order = tally:run(parse_method(split(text))).order
655 for _, candidate in ipairs(order) do
656 if candidate == x or candidate == y then
657 local side = candidate == x and 1 or 2
658 ahead[side][#ahead[side] + 1] = text
659 break
660 end
661 end
662 end
663 local a, b = tally.data.pairwise[x][y], tally.data.pairwise[y][x]
664 lines = {e.names[x] .. " " .. a .. ", " .. e.names[y] .. " " .. b .. ", neither " .. (tally.data.total - a - b),
665 e.names[x] .. " ahead: " .. (#ahead[1] == 0 and "none" or table.concat(ahead[1], ", ")),
666 e.names[y] .. " ahead: " .. (#ahead[2] == 0 and "none" or table.concat(ahead[2], ", "))}
667 else
668 local method = parse_method(words)
669 if method.kind == "STV" and method.parameter >= #field then fail("too many seats") end
670 lines = tally:run(method).lines
671 end
672 table.insert(lines, 1, "[" .. line .. "]")
673 return lines
674end

676local count = tonumber(io.read("l"))
677for _ = 1, count do
678 local ok, result = pcall(execute, io.read("l"))
679 if ok then for _, line in ipairs(result) do io.write(line, "\n") end
680 else io.write(result, "\n") end
681end
