87func parseMethod(words []string) Method {
88 method := Method{kind: MethodKind(slices.Index(methodNames[:], words[0]))}
89 switch method.kind {
90 case Minimax:
91 method.minimax = map[string]MinimaxKind{"WINNING": Winning, "MARGIN": Margin, "OPPOSITION": Opposition}[words[1]]
92 case Approval, STV:
93 method.parameter = number(words[1])
94 }
95 return method
96}

98func (method Method) String() string {
99 name := methodNames[method.kind]
100 switch method.kind {
101 case Minimax:
102 return name + " " + [...]string{"WINNING", "MARGIN", "OPPOSITION"}[method.minimax]
103 case Approval, STV:
104 return fmt.Sprintf("%s %d", name, method.parameter)
105 }
106 return name
107}

179func majority(field Field, scores Row) bool {
180 total := sum(field, scores)
181 for _, a := range field.members() {
182 if 2*scores[a] > total {
183 return true
184 }
185 }
186 return false

189func closure(field Field, relation *Relation) {
190 list := field.members()
191 for _, via := range list {
192 for _, a := range list {
193 for _, b := range list {
194 relation[a][b] = relation[a][b] || relation[a][via] && relation[via][b]
195 }
196 }
197 }
198}

292func (e *Election) tallies(field Field) Tallies {
293 if result := e.cache[field]; result.ready {
294 return result
295 }
296 result := Tallies{ready: true}
297 m := field.count()
298 for _, group := range e.groups {
299 ranking := effective(group.ranking, field)
300 ranked := mask(ranking)
301 if len(ranking) > 0 {
302 result.first[ranking[0]] += group.votes
303 result.nonblank += group.votes
304 }
305 if len(ranking) == m {
306 result.last[ranking[m-1]] += group.votes
307 } else {
308 for _, a := range (field &^ ranked).members() {
309 result.last[a] += group.votes
310 }
311 }
312 for j, a := range ranking {
313 result.borda[a] += group.votes * (m - j - 1)
314 for depth := j; depth < m; depth++ {
315 result.approval[depth][a] += group.votes
316 }
317 }
318 }
319 e.cache[field] = result
320 return result
321}

323func (e *Election) pairwise() Matrix {
324 if e.pairsReady {
325 return e.pairs
326 }
327 var matrix Matrix
328 for _, group := range e.groups {
329 positions := Row{}
330 for a := range e.names {
331 positions[a] = limit
332 }
333 for i, a := range group.ranking {
334 positions[a] = i
335 }
336 for a := range e.names {
337 for b := range e.names {
338 if positions[a] < positions[b] {
339 matrix[a][b] += group.votes
340 }
341 }
342 }
343 }
344 e.pairs, e.pairsReady = matrix, true
345 return matrix
346}

348func (e *Election) condorcet(field Field, loser bool) int {
349 matrix := e.pairwise()
350 list := field.members()
351 for _, a := range list {
352 all := true
353 for _, b := range list {
354 if a == b {
355 continue
356 }
357 if loser && matrix[a][b] >= matrix[b][a] || !loser && matrix[a][b] <= matrix[b][a] {
358 all = false
359 }
360 }
361 if all {
362 return a
363 }
364 }
365 return -1
366}

368func (e *Election) lookBack(field Field, history []Row) (int, bool) {
369 choices := field.members()
370 tie := false
371 for r := len(history) - 1; r >= 0; r-- {
372 minimum := history[r][choices[0]]
373 for _, a := range choices {
374 minimum = min(minimum, history[r][a])
375 }
376 choices = slices.DeleteFunc(choices, func(a int) bool { return history[r][a] != minimum })
377 if r == len(history)-1 {
378 tie = len(choices) > 1
379 }
380 }
381 eliminated := choices[0]
382 for _, a := range choices {
383 if e.priority[a] > e.priority[eliminated] {
384 eliminated = a
385 }
386 }
387 return eliminated, tie
388}

390func (e *Election) elimination(field Field, coombs, exhausted bool) Outcome {
391 remaining := field
392 var removed []int
393 var history []Row
394 var lines []string
395 for {
396 scores := e.tallies(remaining)
397 round := len(lines) + 1
398 line := fmt.Sprintf("round %d: %s", round, e.values(remaining, scores.first, false))
399 if exhausted {
400 active := sum(remaining, scores.first)
401 line = fmt.Sprintf("round %d: active %d, exhausted %d", round, active, e.total-active)
402 }
403 if remaining.count() == 1 || majority(remaining, scores.first) {
404 lines = append(lines, line)
405 order := e.ordered(remaining, scores.first, true)
406 slices.Reverse(removed)
407 order = append(order, removed...)
408 if exhausted {
409 return Outcome{order: order, lines: lines}
410 }
411 return e.finish(order, false, lines)
412 }
413 var eliminated int
414 var tie bool
415 if coombs {
416 list := e.ordered(remaining, scores.last, true)
417 maximum := scores.last[list[0]]
418 choices := slices.DeleteFunc(list, func(a int) bool { return scores.last[a] != maximum })
419 tie = len(choices) > 1
420 least := e.ordered(mask(choices), scores.first, true)
421 eliminated = least[len(least)-1]
422 line += "; last " + e.values(remaining, scores.last, false)
423 } else {
424 history = append(history, scores.first)
425 eliminated, tie = e.lookBack(remaining, history)
426 }
427 if !exhausted {
428 line += "; eliminated " + e.names[eliminated] + tieMark(tie)
429 }
430 lines = append(lines, line)
431 removed = append(removed, eliminated)
432 remaining &^= bit(eliminated)
433 }
434}

436func (e *Election) runoff(field Field) Outcome {
437 first := e.tallies(field)
438 order := e.ordered(field, first.first, true)
439 lines := []string{"round 1: " + e.values(field, first.first, false)}
440 tie := false
441 if !majority(field, first.first) {
442 a, b := order[0], order[1]
443 finalists := bit(a) | bit(b)
444 advancementTie := len(order) >= 3 && first.first[b] == first.first[order[2]]
445 second := e.tallies(finalists)
446 tie = second.first[a] == second.first[b]
447 if second.first[b] > second.first[a] {
448 order[0], order[1] = b, a
449 }
450 lines = append(lines, "finalists: "+e.namesText(finalists.members(), ", ")+tieMark(advancementTie),
451 "round 2: "+e.values(finalists, second.first, false))
452 }
453 return e.finish(order, tie, lines)
454}

456func (e *Election) bucklin(field Field) Outcome {
457 scores := e.tallies(field)
458 majority := scores.nonblank/2 + 1
459 lines := []string{fmt.Sprintf("majority: %d", majority)}
460 order := field.members()
461 final := 0
462 for depth := range len(order) {
463 final = depth
464 lines = append(lines, fmt.Sprintf("round %d: %s", depth+1, e.values(field, scores.approval[depth], false)))
465 done := false
466 for _, a := range order {
467 if scores.approval[depth][a] >= majority {
468 done = true
469 }
470 }
471 if done {
472 break
473 }
474 }
475 slices.SortFunc(order, func(a, b int) int {
476 for depth := final; depth >= 0; depth-- {
477 if comparison := cmp.Compare(scores.approval[depth][b], scores.approval[depth][a]); comparison != 0 {
478 return comparison
479 }
480 }
481 return cmp.Compare(e.priority[a], e.priority[b])
482 })
483 tie := scores.approval[final][order[0]] == scores.approval[final][order[1]]
484 return e.finish(order, tie, lines)
485}

487func (e *Election) bordaElimination(field Field, nanson bool) Outcome {
488 remaining := field
489 var tail []int
490 var lines []string
491 var history []Row
492 for remaining.count() >= 2 {
493 scores := e.tallies(remaining).borda
494 line := fmt.Sprintf("round %d: %s", len(lines)+1, e.values(remaining, scores, false))
495 var removed []int
496 if nanson {
497 total := sum(remaining, scores)
498 var below Field
499 for _, a := range remaining.members() {
500 if scores[a]*remaining.count() < total {
501 below |= bit(a)
502 }
503 }
504 if below == 0 {
505 lines = append(lines, line)
506 break
507 }
508 line += "; eliminated " + e.namesText(below.members(), ", ")
509 removed = e.ordered(below, scores, true)
510 } else {
511 history = append(history, scores)
512 a, tie := e.lookBack(remaining, history)
513 line += "; eliminated " + e.names[a] + tieMark(tie)
514 removed = []int{a}
515 }
516 lines = append(lines, line)
517 remaining &^= mask(removed)
518 tail = append(removed, tail...)
519 }
520 order := e.ordered(remaining, Row{}, true)
521 tie := nanson && len(order) > 1
522 return e.finish(append(order, tail...), tie, lines)
523}

542func (e *Election) condorcetReport(field Field) Outcome {
543 win, lose := e.condorcet(field, false), e.condorcet(field, true)
544 matrix := e.pairwise()
545 var weak, strong Relation
546 list := field.members()
547 for _, a := range list {
548 for _, b := range list {
549 weak[a][b] = matrix[a][b] >= matrix[b][a]
550 strong[a][b] = matrix[a][b] > matrix[b][a]
551 }
552 }
553 closure(field, &weak)
554 closure(field, &strong)
555 var smith, schwartz []int
556 for _, a := range list {
557 top, unbeaten := true, true
558 for _, b := range list {
559 top = top && weak[a][b]
560 unbeaten = unbeaten && (!strong[b][a] || strong[a][b])
561 }
562 if top {
563 smith = append(smith, a)
564 }
565 if unbeaten {
566 schwartz = append(schwartz, a)
567 }
568 }
569 winner, loser := "NONE", "NONE"
570 if win >= 0 {
571 winner = e.names[win]
572 }
573 if lose >= 0 {
574 loser = e.names[lose]
575 }
576 return Outcome{lines: []string{"winner: " + winner, "loser: " + loser,
577 "smith: " + e.namesText(smith, ", "), "schwartz: " + e.namesText(schwartz, ", ")}}
578}

580func (e *Election) topological(field Field, relation Relation) ([]int, bool) {
581 remaining := field
582 var order []int
583 tie := false
584 for remaining != 0 {
585 var choices Field
586 list := remaining.members()
587 for _, a := range list {
588 incoming := false
589 for _, b := range list {
590 if a != b && relation[b][a] {
591 incoming = true
592 }
593 }
594 if !incoming {
595 choices |= bit(a)
596 }
597 }
598 if len(order) == 0 {
599 tie = choices.count() > 1
600 }
601 for _, a := range e.tiebreak {
602 if choices&bit(a) != 0 {
603 order = append(order, a)
604 remaining &^= bit(a)
605 break
606 }
607 }
608 }
609 return order, tie
610}

612func (e *Election) schulze(field Field) Outcome {
613 matrix := e.pairwise()
614 list := field.members()
615 var paths Matrix
616 for _, a := range list {
617 for _, b := range list {
618 if matrix[a][b] > matrix[b][a] {
619 paths[a][b] = matrix[a][b]
620 }
621 }
622 }
623 for _, via := range list {
624 for _, a := range list {
625 for _, b := range list {
626 if a != via && b != via && a != b {
627 paths[a][b] = max(paths[a][b], min(paths[a][via], paths[via][b]))
628 }
629 }
630 }
631 }
632 var relation Relation
633 for _, a := range list {
634 for _, b := range list {
635 relation[a][b] = paths[a][b] > paths[b][a]
636 }
637 }
638 order, tie := e.topological(field, relation)
639 lines := append(e.pairLines(field, paths), "ranking: "+e.namesText(order, " > "))
640 return e.finish(order, tie, lines)
641}

643type Edge struct{ from, to int }

645func (e *Election) rankedPairs(field Field) Outcome {
646 matrix := e.pairwise()
647 list := field.members()
648 var pairs []Edge
649 for _, a := range list {
650 for _, b := range list {
651 if matrix[a][b] > matrix[b][a] {
652 pairs = append(pairs, Edge{a, b})
653 }
654 }
655 }
656 slices.SortFunc(pairs, func(a, b Edge) int {
657 if comparison := cmp.Compare(matrix[b.from][b.to], matrix[a.from][a.to]); comparison != 0 {
658 return comparison
659 }
660 if comparison := cmp.Compare(matrix[a.to][a.from], matrix[b.to][b.from]); comparison != 0 {
661 return comparison
662 }
663 if comparison := cmp.Compare(e.priority[a.from], e.priority[b.from]); comparison != 0 {
664 return comparison
665 }
666 return cmp.Compare(e.priority[a.to], e.priority[b.to])
667 })
668 var locked, reached Relation
669 var lines []string
670 for _, edge := range pairs {
671 a, b := edge.from, edge.to
672 action := "skip"
673 if !reached[b][a] {
674 action = "lock"
675 locked[a][b], reached[a][b] = true, true
676 closure(field, &reached)
677 }
678 lines = append(lines, fmt.Sprintf("%s %s>%s %d-%d", action, e.names[a], e.names[b], matrix[a][b], matrix[b][a]))
679 }
680 order, tie := e.topological(field, locked)
681 return e.finish(order, tie, lines)
682}

684func (e *Election) kemeny(field Field) Outcome {
685 matrix := e.pairwise()
686 var best, ways [1 << limit]int
687 var heads [1 << limit]Field
688 ways[0] = 1
689 for subset := Field(1); subset <= field; subset++ {
690 if subset &^ field != 0 {
691 continue
692 }
693 best[subset] = -1
694 list := subset.members()
695 for _, a := range list {
696 rest := subset &^ bit(a)
697 score := best[rest]
698 for _, b := range list {
699 if a != b {
700 score += matrix[a][b]
701 }
702 }
703 if score > best[subset] {
704 best[subset], ways[subset], heads[subset] = score, 0, 0
705 }
706 if score == best[subset] {
707 ways[subset] += ways[rest]
708 heads[subset] |= bit(a)
709 }
710 }
711 }
712 var order []int
713 for remaining := field; remaining != 0; {
714 for _, a := range e.tiebreak {
715 if heads[remaining]&bit(a) != 0 {
716 order = append(order, a)
717 remaining &^= bit(a)
718 break
719 }
720 }
721 }
722 return e.finish(order, heads[field].count() > 1, []string{
723 fmt.Sprintf("score: %d", best[field]), fmt.Sprintf("optimal: %d", ways[field]), "ranking: " + e.namesText(order, " > "),
724 })
725}

731func (e *Election) stv(field Field, seats int) Outcome {
732 parcels := make([]Parcel, len(e.groups))
733 for i, group := range e.groups {
734 if len(effective(group.ranking, field)) > 0 {
735 parcels[i].value = 100
736 }
737 }
738 quota := e.tallies(field).nonblank*100/(seats+1) + 1
739 lines := []string{"quota: " + money(quota)}
740 continuing := field
741 var elected []int
742 var history []Row
743 for {
744 var scores Row
745 exhausted := 0
746 for i, group := range e.groups {
747 parcel := &parcels[i]
748 parcel.owner = -1
749 if parcel.value == 0 {
750 continue
751 }
752 for _, a := range group.ranking {
753 if continuing&bit(a) != 0 {
754 parcel.owner = a
755 break
756 }
757 }
758 value := group.votes * parcel.value
759 if parcel.owner >= 0 {
760 scores[parcel.owner] += value
761 } else {
762 exhausted += value
763 }
764 }
765 history = append(history, scores)
766 line := fmt.Sprintf("round %d: %s; exhausted %s", len(history), e.values(continuing, scores, true), money(exhausted))
767 order := e.ordered(continuing, scores, true)
768 if len(order) == seats-len(elected) {
769 line += "; elected " + e.namesText(order, ", ")
770 elected = append(elected, order...)
771 lines = append(lines, line)
772 break
773 }
774 a := order[0]
775 if scores[a] >= quota {
776 surplus := scores[a] - quota
777 elected = append(elected, a)
778 continuing &^= bit(a)
779 for i := range parcels {
780 if parcels[i].owner == a {
781 parcels[i].value = parcels[i].value * surplus / scores[a]
782 }
783 }
784 line += "; elected " + e.names[a] + " (surplus " + money(surplus) + ")"
785 lines = append(lines, line)
786 if len(elected) == seats {
787 break
788 }
789 } else {
790 excluded, tie := e.lookBack(continuing, history)
791 continuing &^= bit(excluded)
792 line += "; excluded " + e.names[excluded] + tieMark(tie)
793 lines = append(lines, line)
794 }
795 }
796 lines = append(lines, "elected: "+e.namesText(elected, ", "))
797 return Outcome{lines: lines}
798}

800func (e *Election) tally(field Field, method Method) Outcome {
801 switch method.kind {
802 case Runoff:
803 return e.runoff(field)
804 case IRV, Coombs:
805 return e.elimination(field, method.kind == Coombs, false)
806 case Bucklin:
807 return e.bucklin(field)
808 case Baldwin, Nanson:
809 return e.bordaElimination(field, method.kind == Nanson)
810 case Condorcet:
811 return e.condorcetReport(field)
812 case Schulze:
813 return e.schulze(field)
814 case RankedPairs:
815 return e.rankedPairs(field)
816 case Kemeny:
817 return e.kemeny(field)
818 case STV:
819 return e.stv(field, method.parameter)
820 }
821 base := e.tallies(field)
822 var scores Row
823 var lines []string
824 label, descending := "scores", true
825 condorcetWinner := -1
826 switch method.kind {
827 case Plurality:
828 scores, label = base.first, "votes"
829 case Borda, Black:
830 scores = base.borda
831 if method.kind == Black {
832 condorcetWinner = e.condorcet(field, false)
833 name := "NONE"
834 if condorcetWinner >= 0 {
835 name = e.names[condorcetWinner]
836 }
837 lines = append(lines, "condorcet: "+name)
838 }
839 case Veto:
840 scores, label, descending = base.last, "vetoes", false
841 case Approval:
842 scores, label = base.approval[min(method.parameter, field.count())-1], "approvals"
843 case Copeland:
844 matrix := e.pairwise()
845 var descriptions []string
846 list := field.members()
847 for _, a := range list {
848 wins, draws, losses := 0, 0, 0
849 for _, b := range list {
850 if a == b {
851 continue
852 }
853 switch cmp.Compare(matrix[a][b], matrix[b][a]) {
854 case 1:
855 wins++
856 case 0:
857 draws++
858 case -1:
859 losses++
860 }
861 }
862 scores[a] = 2*wins + draws
863 descriptions = append(descriptions, fmt.Sprintf("%s %d.%d (%d-%d-%d)", e.names[a], scores[a]/2, scores[a]%2*5, wins, draws, losses))
864 }
865 lines = append(lines, "scores: "+strings.Join(descriptions, ", "))
866 case Minimax:
867 matrix := e.pairwise()
868 descending = false
869 list := field.members()
870 for _, a := range list {
871 for _, b := range list {
872 if a == b {
873 continue
874 }
875 opposition, support, value := matrix[b][a], matrix[a][b], 0
876 if method.minimax == Opposition || opposition > support {
877 value = opposition
878 if method.minimax == Margin {
879 value -= support
880 }
881 }
882 scores[a] = max(scores[a], value)
883 }
884 }
885 default:
886 panic("unreachable method")
887 }
888 order := e.ordered(field, scores, descending)
889 tie := scores[order[0]] == scores[order[1]]
890 if condorcetWinner >= 0 {
891 index := slices.Index(order, condorcetWinner)
892 copy(order[1:index+1], order[:index])
893 order[0] = condorcetWinner
894 tie = false
895 }
896 if method.kind != Copeland {
897 lines = append(lines, label+": "+e.values(field, scores, false))
898 }
899 return e.finish(order, tie, lines)
900}

902func (e *Election) ballotsReport() []string {
903 counts := make(map[string]int)
904 for _, group := range e.groups {
905 ranking := effective(group.ranking, e.active)
906 notation := "-"
907 if len(ranking) > 0 {
908 notation = e.namesText(ranking, ">")
909 }
910 counts[notation] += group.votes
911 }
912 type distribution struct {
913 notation string
914 votes int
915 }
916 groups := make([]distribution, 0, len(counts))
917 for notation, votes := range counts {
918 groups = append(groups, distribution{notation, votes})
919 }
920 slices.SortFunc(groups, func(a, b distribution) int {
921 if comparison := cmp.Compare(b.votes, a.votes); comparison != 0 {
922 return comparison
923 }
924 return strings.Compare(a.notation, b.notation)
925 })
926 lines := []string{fmt.Sprintf("total %d, blank %d, distinct %d", e.total, counts["-"], len(groups))}
927 for _, group := range groups {
928 lines = append(lines, fmt.Sprintf("%d %s", group.votes, group.notation))
929 }
930 return lines
931}

933type Lab struct {
934 election *Election
935 number int

938func (lab *Lab) execute(line string) ([]string, error) {
939 words := strings.Split(line, " ")
940 command := words[0]
941 if command == "CANDIDATES" {
942 names := words[1:]
943 if len(names) == 1 {
944 return nil, errors.New("too few candidates")
945 }
946 seen := make(map[string]bool)
947 for _, name := range names {
948 if seen[name] {
949 return nil, fmt.Errorf("duplicate candidate %s", name)
950 }
951 seen[name] = true
952 }
953 lab.number++
954 lab.election = newElection(lab.number, names)
955 return []string{fmt.Sprintf("election %d: %d candidates", lab.number, len(names))}, nil
956 }
957 e := lab.election
958 if e == nil {
959 return nil, errors.New("no election")
960 }
961 switch command {
962 case "BALLOT":
963 var ranking []int
964 if words[2] != "-" {
965 var err error
966 ranking, err = e.checkNames(strings.Split(words[2], ">"))
967 if err != nil {
968 return nil, err
969 }
970 }
971 votes := number(words[1])
972 e.groups = append(e.groups, Ballot{votes, ranking})
973 e.total += votes
974 e.finishEntry()
975 return nil, nil
976 case "RANDOM":
977 count := number(words[1])
978 state, _ := strconv.ParseInt(words[2], 10, 64)
979 for range count {
980 ranking := e.active.members()
981 for i := len(ranking) - 1; i > 0; i-- {
982 state = 48271 * state % 2147483647
983 j := int(state % int64(i+1))
984 ranking[i], ranking[j] = ranking[j], ranking[i]
985 }
986 state = 48271 * state % 2147483647
987 ranking = ranking[:int(state%int64(len(ranking)))+1]
988 e.groups = append(e.groups, Ballot{1, ranking})
989 }
990 e.total += count
991 e.finishEntry()
992 return nil, nil
993 case "UNDO":
994 if len(e.entryEnds) == 0 {
995 return nil, errors.New("nothing to undo")
996 }
997 e.entryEnds = e.entryEnds[:len(e.entryEnds)-1]
998 end := 0
999 if len(e.entryEnds) > 0 {
1000 end = e.entryEnds[len(e.entryEnds)-1]
1001 }
1002 for _, group := range e.groups[end:] {
1003 e.total -= group.votes
1004 }
1005 clear(e.groups[end:])
1006 e.groups = e.groups[:end]
1007 e.invalidate()
1008 return nil, nil
1009 case "WITHDRAW", "RESTORE":
1010 a, err := e.candidate(words[1])
1011 if err != nil {
1012 return nil, err
1013 }
1014 active := e.active&bit(a) != 0
1015 if command == "WITHDRAW" {
1016 if !active {
1017 return nil, fmt.Errorf("already withdrawn %s", words[1])
1018 }
1019 if e.active.count() <= 2 {
1020 return nil, errors.New("too few candidates")
1021 }
1022 e.active &^= bit(a)
1023 } else {
1024 if active {
1025 return nil, fmt.Errorf("not withdrawn %s", words[1])
1026 }
1027 e.active |= bit(a)
1028 }
1029 return nil, nil
1030 case "TIEBREAK":
1031 order, err := e.checkNames(words[1:])
1032 if err != nil {
1033 return nil, err
1034 }
1035 if len(order) != len(e.names) {
1036 return nil, errors.New("incomplete tiebreak")
1037 }
1038 e.setTiebreak(order)
1039 return nil, nil
1040 case "STATUS":
1041 withdrawn := ((bit(len(e.names)) - 1) &^ e.active).members()
1042 names := "none"
1043 if len(withdrawn) > 0 {
1044 names = e.namesText(withdrawn, ", ")
1045 }
1046 return []string{"[" + line + "]", fmt.Sprintf("election %d", e.number),
1047 "active: " + e.namesText(e.active.members(), ", "), "withdrawn: " + names,
1048 "tiebreak: " + e.namesText(e.tiebreak, ", "), fmt.Sprintf("ballots: %d in %d entries", e.total, len(e.entryEnds))}, nil
1049 }
1050 if len(e.entryEnds) == 0 {
1051 return nil, errors.New("no ballots")
1052 }
1053 var requested Method
1054 if command == "RANKING" || command == "SPOILERS" {
1055 requested = parseMethod(words[1:])
1056 if !requested.rankable() {
1057 return nil, errors.New("not rankable")
1058 }
1059 if command == "SPOILERS" && e.active.count() < 3 {
1060 return nil, errors.New("too few candidates")
1061 }
1062 }
1063 var compared [2]int
1064 if command == "COMPARE" {
1065 for i := range 2 {
1066 a, err := e.candidate(words[i+1])
1067 if err != nil {
1068 return nil, err
1069 }
1070 if e.active&bit(a) == 0 {
1071 return nil, fmt.Errorf("withdrawn candidate %s", words[i+1])
1072 }
1073 compared[i] = a
1074 }
1075 if compared[0] == compared[1] {
1076 return nil, errors.New("same candidate")
1077 }
1078 }
1079 if command == "STV" && number(words[1]) >= e.active.count() {
1080 return nil, errors.New("too many seats")
1081 }
1082 lines := []string{"[" + line + "]"}
1083 switch command {
1084 case "RANKING", "SPOILERS":
1085 result := e.tally(e.active, requested)
1086 if command == "RANKING" {
1087 lines = append(lines, e.namesText(result.order, " > "))
1088 } else {
1089 win := result.order[0]
1090 lines = append(lines, "winner: "+e.names[win])
1091 for _, a := range e.active.members() {
1092 if a == win {
1093 continue
1094 }
1095 other := e.tally(e.active &^ bit(a), requested).order[0]
1096 mark := ""
1097 if other != win {
1098 mark = " (changed)"
1099 }
1100 lines = append(lines, "without "+e.names[a]+": "+e.names[other]+mark)
1101 }
1102 }
1103 case "BALLOTS":
1104 lines = append(lines, e.ballotsReport()...)
1105 case "PROFILE":
1106 scores := e.tallies(e.active)
1107 size := e.active.count()
1108 for _, a := range e.active.members() {
1109 counts := make([]string, size)
1110 previous := 0
1111 for j := range size {
1112 counts[j] = strconv.Itoa(scores.approval[j][a] - previous)
1113 previous = scores.approval[j][a]
1114 }
1115 lines = append(lines, fmt.Sprintf("%s: %s unranked %d", e.names[a], strings.Join(counts, " "), e.total-previous))
1116 }
1117 case "PAIRWISE":
1118 lines = append(lines, e.pairLines(e.active, e.pairwise())...)
1119 case "EXHAUSTED":
1120 lines = append(lines, e.elimination(e.active, false, true).lines...)
1121 case "SUMMARY", "COMPARE":
1122 var wins Row
1123 var ahead [2][]string
1124 for _, method := range summaryMethods {
1125 result := e.tally(e.active, method)
1126 if command == "SUMMARY" {
1127 win := result.order[0]
1128 wins[win]++
1129 lines = append(lines, fmt.Sprintf("%s: %s", method, e.names[win]))
1130 } else {
1131 side := 0
1132 if slices.Index(result.order, compared[0]) > slices.Index(result.order, compared[1]) {
1133 side = 1
1134 }
1135 ahead[side] = append(ahead[side], method.String())
1136 }
1137 }
1138 if command == "SUMMARY" {
1139 lines = append(lines, "wins: "+e.values(e.active, wins, false))
1140 } else {
1141 matrix := e.pairwise()
1142 a, b := compared[0], compared[1]
1143 lines = append(lines, fmt.Sprintf("%s %d, %s %d, neither %d", e.names[a], matrix[a][b], e.names[b], matrix[b][a], e.total-matrix[a][b]-matrix[b][a]))
1144 for i := range 2 {
1145 methods := "none"
1146 if len(ahead[i]) > 0 {
1147 methods = strings.Join(ahead[i], ", ")
1148 }
1149 lines = append(lines, e.names[compared[i]]+" ahead: "+methods)
1150 }
1151 }
1152 default:
1153 lines = append(lines, e.tally(e.active, parseMethod(words)).lines...)
1154 }
1155 return lines, nil
1156}

1162 }
1163 count := number(input.Text())

1177 }
1178 }
1179 }
