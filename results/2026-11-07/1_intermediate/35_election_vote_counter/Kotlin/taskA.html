8private enum class Kind {
9 PLURALITY, RUNOFF, IRV, COOMBS, BUCKLIN, BORDA, BALDWIN, NANSON, BLACK,
10 VETO, CONDORCET, COPELAND, MINIMAX, SCHULZE, RANKEDPAIRS, KEMENY, APPROVAL, STV
11}
12private enum class MinimaxKind { WINNING, MARGIN, OPPOSITION }
13
14private data class Method(val kind: Kind, val parameter: Int = 0, val minimax: MinimaxKind = MinimaxKind.WINNING) {
15 val rankable: Boolean get() = kind != Kind.CONDORCET && kind != Kind.STV
16 override fun toString(): String = when (kind) {
17 Kind.MINIMAX -> "$kind $minimax"
18 Kind.APPROVAL, Kind.STV -> "$kind $parameter"
19 else -> kind.toString()
20 }
21 companion object {
22 fun parse(words: List<String>, start: Int = 0): Method {
23 val kind = Kind.valueOf(words[start])
24 return when (kind) {
25 Kind.MINIMAX -> Method(kind, minimax = MinimaxKind.valueOf(words[start + 1]))
26 Kind.APPROVAL, Kind.STV -> Method(kind, words[start + 1].toInt())
27 else -> Method(kind)
28 }
29 }
30 }
31}

45private class Tallies {
46 val first = IntArray(LIMIT)
47 val last = IntArray(LIMIT)
48 val borda = IntArray(LIMIT)
49 val approval = Array(LIMIT) { IntArray(LIMIT) }
50 var nonblank = 0

53private class CommandError(reason: String) : RuntimeException(reason)
54private fun bit(a: Int): Int = 1 shl a
55private fun members(field: Int): List<Int> = (0 until LIMIT).filter { (field and bit(it)) != 0 }
56private fun mask(list: List<Int>): Int = list.fold(0) { field, a -> field or bit(a) }
57private fun effective(ranking: List<Int>, field: Int): List<Int> = ranking.filter { (field and bit(it)) != 0 }
58private fun sum(field: Int, scores: Row): Int = members(field).sumOf { scores[it] }
59private fun majority(field: Int, scores: Row): Boolean {
60 val total = sum(field, scores)
61 return members(field).any { 2 * scores[it] > total }
62}
63private fun tieMark(tie: Boolean): String = if (tie) " (tie)" else ""
64private fun money(value: Int): String = "%d.%02d".format(Locale.ROOT, value / 100, value % 100)
65private fun closure(field: Int, relation: Relation) {
66 val list = members(field)
67 for (via in list) for (a in list) for (b in list) {
68 relation[a][b] = relation[a][b] || relation[a][via] && relation[via][b]
69 }
70}

72private class Election(val number: Int, val names: List<String>) {
73 var active = bit(names.size) - 1
74 var tiebreak = members(active).sortedBy { names[it] }
75 private set
76 private val priority = IntArray(LIMIT)
77 val groups = mutableListOf<Ballot>()
78 val entryEnds = mutableListOf<Int>()
79 var total = 0
80 private val cache = arrayOfNulls<Tallies>(1 shl LIMIT)
81 private var pairs: Matrix? = null
82
83 init { updateTiebreak(tiebreak) }
84
85 fun updateTiebreak(order: List<Int>) {
86 order.forEachIndexed { i, a -> priority[a] = i }
87 tiebreak = order
88 }
89 fun candidate(name: String): Int {
90 val a = names.indexOf(name)
91 if (a < 0) throw CommandError("unknown candidate $name")
92 return a
93 }
94 fun checkedNames(words: List<String>): List<Int> {
95 val result = mutableListOf<Int>()
96 for (name in words) {
97 val a = candidate(name)
98 if (a in result) throw CommandError("duplicate candidate $name")
99 result.add(a)
100 }
101 return result
102 }
103 fun namesText(list: List<Int>, separator: String = ", "): String = list.joinToString(separator) { names[it] }
104 fun values(field: Int, scores: Row, cents: Boolean = false): String = members(field).joinToString(", ") {
105 "${names[it]} ${if (cents) money(scores[it]) else scores[it].toString()}"
106 }
107 private fun ordered(field: Int, scores: Row, descending: Boolean = true): MutableList<Int> =
108 members(field).sortedWith { a, b ->
109 val comparison = if (descending) scores[b].compareTo(scores[a]) else scores[a].compareTo(scores[b])
110 if (comparison != 0) comparison else priority[a].compareTo(priority[b])
111 }.toMutableList()
112
113 private fun finish(order: List<Int>, tie: Boolean, lines: MutableList<String>): Outcome {
114 lines.add("winner: ${names[order.first()]}${tieMark(tie)}")
115 return Outcome(order, lines)
116 }
117 fun invalidate() { cache.fill(null); pairs = null }
118 fun finishEntry() { entryEnds.add(groups.size); invalidate() }

120 fun tallies(field: Int): Tallies {
121 cache[field]?.let { return it }
122 val scores = Tallies()
123 val candidates = members(field)
124 val size = candidates.size
125 for (group in groups) {
126 val ranking = effective(group.ranking, field)
127 val ranked = mask(ranking)
128 if (ranking.isNotEmpty()) {
129 scores.first[ranking.first()] += group.votes
130 scores.nonblank += group.votes
131 }
132 if (ranking.size == size) scores.last[ranking.last()] += group.votes
133 else for (a in candidates) if ((ranked and bit(a)) == 0) scores.last[a] += group.votes
134 ranking.forEachIndexed { j, a ->
135 scores.borda[a] += group.votes * (size - j - 1)
136 for (depth in j until size) scores.approval[depth][a] += group.votes
137 }
138 }
139 cache[field] = scores
140 return scores
141 }
142 fun pairwise(): Matrix {
143 pairs?.let { return it }
144 val matrix = Array(LIMIT) { IntArray(LIMIT) }
145 for (group in groups) {
146 val positions = IntArray(LIMIT) { LIMIT }
147 group.ranking.forEachIndexed { i, a -> positions[a] = i }
148 for (a in names.indices) for (b in names.indices) {
149 if (positions[a] < positions[b]) matrix[a][b] += group.votes
150 }
151 }
152 pairs = matrix
153 return matrix
154 }
155 private fun condorcet(field: Int, loser: Boolean = false): Int? {
156 val matrix = pairwise()
157 val list = members(field)
158 return list.firstOrNull { a -> list.all { b ->
159 a == b || if (loser) matrix[a][b] < matrix[b][a] else matrix[a][b] > matrix[b][a]
160 } }
161 }
162 private fun lookBack(field: Int, history: List<Row>): Removal {
163 val current = history.last()
164 val minimum = members(field).minOf { current[it] }
165 var choices = members(field).filter { current[it] == minimum }
166 val tie = choices.size > 1
167 for (i in history.lastIndex - 1 downTo 0) {
168 val scores = history[i]
169 val low = choices.minOf { scores[it] }
170 choices = choices.filter { scores[it] == low }
171 }
172 return Removal(choices.maxBy { priority[it] }, tie)
173 }

175 fun elimination(field: Int, coombs: Boolean = false, exhausted: Boolean = false): Outcome {
176 var remaining = field
177 val removed = mutableListOf<Int>()
178 val history = mutableListOf<Row>()
179 val lines = mutableListOf<String>()
180 while (true) {
181 val scores = tallies(remaining)
182 val activeVotes = sum(remaining, scores.first)
183 var line = "round ${lines.size + 1}: " + if (exhausted)
184 "active $activeVotes, exhausted ${total - activeVotes}" else values(remaining, scores.first)
185 if (remaining.countOneBits() == 1 || majority(remaining, scores.first)) {
186 lines.add(line)
187 val order = ordered(remaining, scores.first) + removed.asReversed()
188 return if (exhausted) Outcome(order, lines) else finish(order, false, lines)
189 }
190 val choice = if (coombs) {
191 val maximum = members(remaining).maxOf { scores.last[it] }
192 val choices = members(remaining).filter { scores.last[it] == maximum }
193 line += "; last ${values(remaining, scores.last)}"
194 Removal(ordered(mask(choices), scores.first).last(), choices.size > 1)
195 } else {
196 history.add(scores.first)
197 lookBack(remaining, history)
198 }
199 if (!exhausted) line += "; eliminated ${names[choice.candidate]}${tieMark(choice.tie)}"
200 lines.add(line)
201 removed.add(choice.candidate)
202 remaining = remaining and bit(choice.candidate).inv()
203 }
204 }
205 private fun runoff(field: Int): Outcome {
206 val first = tallies(field).first
207 val order = ordered(field, first)
208 val lines = mutableListOf("round 1: ${values(field, first)}")
209 var tie = false
210 if (!majority(field, first)) {
211 val a = order[0]
212 val b = order[1]
213 val finalists = bit(a) or bit(b)
214 val advanceTie = order.size >= 3 && first[b] == first[order[2]]
215 val second = tallies(finalists).first
216 tie = second[a] == second[b]
217 if (second[b] > second[a]) { order[0] = b; order[1] = a }
218 lines.add("finalists: ${namesText(members(finalists))}${tieMark(advanceTie)}")
219 lines.add("round 2: ${values(finalists, second)}")
220 }
221 return finish(order, tie, lines)
222 }
223 private fun bucklin(field: Int): Outcome {
224 val scores = tallies(field)
225 val majority = scores.nonblank / 2 + 1
226 val lines = mutableListOf("majority: $majority")
227 var finalDepth = 0
228 val list = members(field)
229 for (depth in list.indices) {
230 finalDepth = depth
231 lines.add("round ${depth + 1}: ${values(field, scores.approval[depth])}")
232 if (list.any { scores.approval[depth][it] >= majority }) break
233 }
234 val order = list.sortedWith { a, b ->
235 var comparison = 0
236 for (depth in finalDepth downTo 0) {
237 comparison = scores.approval[depth][b].compareTo(scores.approval[depth][a])
238 if (comparison != 0) break
239 }
240 if (comparison != 0) comparison else priority[a].compareTo(priority[b])
241 }
242 return finish(order, scores.approval[finalDepth][order[0]] == scores.approval[finalDepth][order[1]], lines)
243 }
244 private fun bordaElimination(field: Int, nanson: Boolean): Outcome {
245 var remaining = field
246 var tail = emptyList<Int>()
247 val lines = mutableListOf<String>()
248 val history = mutableListOf<Row>()
249 while (remaining.countOneBits() >= 2) {
250 val scores = tallies(remaining).borda
251 var line = "round ${lines.size + 1}: ${values(remaining, scores)}"
252 val removed = if (nanson) {
253 val total = sum(remaining, scores)
254 val below = members(remaining).filter { scores[it] * remaining.countOneBits() < total }
255 if (below.isEmpty()) { lines.add(line); break }
256 line += "; eliminated ${namesText(below)}"
257 ordered(mask(below), scores)
258 } else {
259 history.add(scores)
260 val choice = lookBack(remaining, history)
261 line += "; eliminated ${names[choice.candidate]}${tieMark(choice.tie)}"
262 listOf(choice.candidate)
263 }
264 lines.add(line)
265 remaining = remaining and mask(removed).inv()
266 tail = removed + tail
267 }
268 val finalOrder = ordered(remaining, IntArray(LIMIT))
269 return finish(finalOrder + tail, nanson && finalOrder.size > 1, lines)
270 }

272 fun pairLines(field: Int, matrix: Matrix): List<String> {
273 val list = members(field)
274 val lines = mutableListOf<String>()
275 for (i in list.indices) for (j in i + 1 until list.size) {
276 val a = list[i]
277 val b = list[j]
278 val result = if (matrix[a][b] == matrix[b][a]) "tie" else names[if (matrix[a][b] > matrix[b][a]) a else b]
279 lines.add("${names[a]} vs ${names[b]}: ${matrix[a][b]}-${matrix[b][a]} ($result)")
280 }
281 return lines
282 }
283 private fun condorcetReport(field: Int): Outcome {
284 val winner = condorcet(field)
285 val loser = condorcet(field, true)
286 val matrix = pairwise()
287 val weak = Array(LIMIT) { BooleanArray(LIMIT) }
288 val strong = Array(LIMIT) { BooleanArray(LIMIT) }
289 val list = members(field)
290 for (a in list) for (b in list) {
291 weak[a][b] = matrix[a][b] >= matrix[b][a]
292 strong[a][b] = matrix[a][b] > matrix[b][a]
293 }
294 closure(field, weak)
295 closure(field, strong)
296 val smith = list.filter { a -> list.all { b -> weak[a][b] } }
297 val schwartz = list.filter { a -> list.all { b -> !strong[b][a] || strong[a][b] } }
298 return Outcome(emptyList(), listOf(
299 "winner: ${winner?.let { names[it] } ?: "NONE"}", "loser: ${loser?.let { names[it] } ?: "NONE"}",
300 "smith: ${namesText(smith)}", "schwartz: ${namesText(schwartz)}"
301 ))
302 }
303 private fun topological(field: Int, relation: Relation): Standing {
304 var remaining = field
305 val order = mutableListOf<Int>()
306 var tie = false
307 while (remaining != 0) {
308 val list = members(remaining)
309 val choices = list.filter { a -> list.all { b -> a == b || !relation[b][a] } }
310 if (order.isEmpty()) tie = choices.size > 1
311 val a = choices.minBy { priority[it] }
312 order.add(a)
313 remaining = remaining and bit(a).inv()
314 }
315 return Standing(order, tie)
316 }
317 private fun schulze(field: Int): Outcome {
318 val matrix = pairwise()
319 val paths = Array(LIMIT) { IntArray(LIMIT) }
320 val list = members(field)
321 for (a in list) for (b in list) if (matrix[a][b] > matrix[b][a]) paths[a][b] = matrix[a][b]
322 for (via in list) for (a in list) for (b in list) {
323 if (a != via && b != via && a != b) paths[a][b] = maxOf(paths[a][b], minOf(paths[a][via], paths[via][b]))
324 }
325 val relation = Array(LIMIT) { BooleanArray(LIMIT) }
326 for (a in list) for (b in list) relation[a][b] = paths[a][b] > paths[b][a]
327 val result = topological(field, relation)
328 val lines = pairLines(field, paths).toMutableList()
329 lines.add("ranking: ${namesText(result.order, " > ")}")
330 return finish(result.order, result.tie, lines)
331 }
332 private fun rankedPairs(field: Int): Outcome {
333 val matrix = pairwise()
334 val list = members(field)
335 val edges = buildList { for (a in list) for (b in list) if (matrix[a][b] > matrix[b][a]) add(Edge(a, b)) }
336 .sortedWith(compareBy<Edge>({ -matrix[it.from][it.to] }, { matrix[it.to][it.from] }, { priority[it.from] }, { priority[it.to] }))
337 val locked = Array(LIMIT) { BooleanArray(LIMIT) }
338 val reached = Array(LIMIT) { BooleanArray(LIMIT) }
339 val lines = mutableListOf<String>()
340 for ((a, b) in edges) {
341 val skip = reached[b][a]
342 lines.add("${if (skip) "skip" else "lock"} ${names[a]}>${names[b]} ${matrix[a][b]}-${matrix[b][a]}")
343 if (!skip) {
344 locked[a][b] = true
345 reached[a][b] = true
346 closure(field, reached)
347 }
348 }
349 val result = topological(field, locked)
350 return finish(result.order, result.tie, lines)
351 }
352 private fun kemeny(field: Int): Outcome {
353 val matrix = pairwise()
354 val best = IntArray(1 shl LIMIT)
355 val ways = IntArray(1 shl LIMIT)
356 val heads = IntArray(1 shl LIMIT)
357 ways[0] = 1
358 for (subset in 1..field) {
359 if ((subset and field.inv()) != 0) continue
360 best[subset] = -1
361 val list = members(subset)
362 for (a in list) {
363 val rest = subset and bit(a).inv()
364 val score = best[rest] + list.filter { it != a }.sumOf { matrix[a][it] }
365 if (score > best[subset]) { best[subset] = score; ways[subset] = 0; heads[subset] = 0 }
366 if (score == best[subset]) { ways[subset] += ways[rest]; heads[subset] = heads[subset] or bit(a) }
367 }
368 }
369 val order = mutableListOf<Int>()
370 var remaining = field
371 while (remaining != 0) {
372 val a = tiebreak.first { (heads[remaining] and bit(it)) != 0 }
373 order.add(a)
374 remaining = remaining and bit(a).inv()
375 }
376 val lines = mutableListOf("score: ${best[field]}", "optimal: ${ways[field]}", "ranking: ${namesText(order, " > ")}")
377 return finish(order, heads[field].countOneBits() > 1, lines)
378 }
379 private fun stv(field: Int, seats: Int): Outcome {
380 val value = IntArray(groups.size) { if (effective(groups[it].ranking, field).isEmpty()) 0 else 100 }
381 val owner = IntArray(groups.size)
382 val quota = tallies(field).nonblank * 100 / (seats + 1) + 1
383 val lines = mutableListOf("quota: ${money(quota)}")
384 var continuing = field
385 val elected = mutableListOf<Int>()
386 val history = mutableListOf<Row>()
387 while (true) {
388 val scores = IntArray(LIMIT)
389 var exhausted = 0
390 owner.fill(-1)
391 for (i in groups.indices) {
392 if (value[i] == 0) continue
393 val group = groups[i]
394 owner[i] = group.ranking.firstOrNull { (continuing and bit(it)) != 0 } ?: -1
395 val amount = group.votes * value[i]
396 if (owner[i] < 0) exhausted += amount else scores[owner[i]] += amount
397 }
398 history.add(scores)
399 val line = "round ${history.size}: ${values(continuing, scores, true)}; exhausted ${money(exhausted)}"
400 val order = ordered(continuing, scores)
401 if (order.size == seats - elected.size) {
402 lines.add("$line; elected ${namesText(order)}")
403 elected.addAll(order)
404 break
405 }
406 val a = order.first()
407 if (scores[a] >= quota) {
408 val surplus = scores[a] - quota
409 elected.add(a)
410 continuing = continuing and bit(a).inv()
411 for (i in groups.indices) if (owner[i] == a) value[i] = value[i] * surplus / scores[a]
412 lines.add("$line; elected ${names[a]} (surplus ${money(surplus)})")
413 if (elected.size == seats) break
414 } else {
415 val choice = lookBack(continuing, history)
416 continuing = continuing and bit(choice.candidate).inv()
417 lines.add("$line; excluded ${names[choice.candidate]}${tieMark(choice.tie)}")
418 }
419 }
420 lines.add("elected: ${namesText(elected)}")
421 return Outcome(emptyList(), lines)
422 }

424 fun tally(field: Int, method: Method): Outcome {
425 when (method.kind) {
426 Kind.RUNOFF -> return runoff(field)
427 Kind.IRV, Kind.COOMBS -> return elimination(field, method.kind == Kind.COOMBS)
428 Kind.BUCKLIN -> return bucklin(field)
429 Kind.BALDWIN, Kind.NANSON -> return bordaElimination(field, method.kind == Kind.NANSON)
430 Kind.CONDORCET -> return condorcetReport(field)
431 Kind.SCHULZE -> return schulze(field)
432 Kind.RANKEDPAIRS -> return rankedPairs(field)
433 Kind.KEMENY -> return kemeny(field)
434 Kind.STV -> return stv(field, method.parameter)
435 else -> Unit
436 }
437 val base = tallies(field)
438 val lines = mutableListOf<String>()
439 var label: String? = "scores"
440 var descending = true
441 var condorcetWinner: Int? = null
442 val scores = when (method.kind) {
443 Kind.PLURALITY -> { label = "votes"; base.first }
444 Kind.BORDA, Kind.BLACK -> {
445 if (method.kind == Kind.BLACK) {
446 condorcetWinner = condorcet(field)
447 lines.add("condorcet: ${condorcetWinner?.let { names[it] } ?: "NONE"}")
448 }
449 base.borda
450 }
451 Kind.VETO -> { label = "vetoes"; descending = false; base.last }
452 Kind.APPROVAL -> { label = "approvals"; base.approval[minOf(method.parameter, field.countOneBits()) - 1] }
453 Kind.COPELAND -> {
454 val matrix = pairwise()
455 val scores = IntArray(LIMIT)
456 val list = members(field)
457 val descriptions = list.map { a ->
458 var wins = 0
459 var draws = 0
460 var losses = 0
461 for (b in list) if (a != b) {
462 when {
463 matrix[a][b] > matrix[b][a] -> wins++
464 matrix[a][b] == matrix[b][a] -> draws++
465 else -> losses++
466 }
467 }
468 scores[a] = 2 * wins + draws
469 "${names[a]} ${scores[a] / 2}.${scores[a] % 2 * 5} ($wins-$draws-$losses)"
470 }
471 lines.add("scores: ${descriptions.joinToString(", ")}")
472 label = null
473 scores
474 }
475 Kind.MINIMAX -> {
476 val matrix = pairwise()
477 val scores = IntArray(LIMIT)
478 val list = members(field)
479 descending = false
480 for (a in list) for (b in list) if (a != b) {
481 val opposition = matrix[b][a]
482 val support = matrix[a][b]
483 val value = when {
484 method.minimax == MinimaxKind.OPPOSITION -> opposition
485 opposition <= support -> 0
486 method.minimax == MinimaxKind.MARGIN -> opposition - support
487 else -> opposition
488 }
489 scores[a] = maxOf(scores[a], value)
490 }
491 scores
492 }
493 else -> error("unreachable method")
494 }
495 val order = ordered(field, scores, descending)
496 var tie = scores[order[0]] == scores[order[1]]
497 condorcetWinner?.let { a ->
498 order.remove(a)
499 order.add(0, a)
500 tie = false
501 }
502 if (label != null) lines.add("$label: ${values(field, scores)}")
503 return finish(order, tie, lines)
504 }
505 fun ballotsReport(): List<String> {
506 val counts = mutableMapOf<String, Int>()
507 for (group in groups) {
508 val ranking = effective(group.ranking, active)
509 val notation = if (ranking.isEmpty()) "-" else namesText(ranking, ">")
510 counts[notation] = counts.getOrDefault(notation, 0) + group.votes
511 }
512 val ranked = counts.keys.sortedWith(compareByDescending<String> { counts.getValue(it) }.thenBy { it })
513 return listOf("total $total, blank ${counts.getOrDefault("-", 0)}, distinct ${counts.size}") + ranked.map { "${counts[it]} $it" }
514 }
515}

517private class Lab {
518 private var election: Election? = null
519 private var number = 0
520
521 fun execute(line: String): List<String> {
522 val words = line.split(' ')
523 val command = words[0]
524 if (command == "CANDIDATES") {
525 val names = words.drop(1)
526 if (names.size < 2) throw CommandError("too few candidates")
527 val seen = mutableSetOf<String>()
528 for (name in names) if (!seen.add(name)) throw CommandError("duplicate candidate $name")
529 election = Election(++number, names)
530 return listOf("election $number: ${names.size} candidates")
531 }
532 val e = election ?: throw CommandError("no election")
533 when (command) {
534 "BALLOT" -> {
535 val ranking = if (words[2] == "-") emptyList() else e.checkedNames(words[2].split('>'))
536 val votes = words[1].toInt()
537 e.groups.add(Ballot(votes, ranking))
538 e.total += votes
539 e.finishEntry()
540 return emptyList()
541 }
542 "RANDOM" -> {
543 val count = words[1].toInt()
544 var state = words[2].toLong()
545 repeat(count) {
546 val ranking = members(e.active).toMutableList()
547 for (i in ranking.lastIndex downTo 1) {
548 state = 48271 * state % 2147483647
549 val j = (state % (i + 1)).toInt()
550 val saved = ranking[i]
551 ranking[i] = ranking[j]
552 ranking[j] = saved
553 }
554 state = 48271 * state % 2147483647
555 val length = (state % ranking.size).toInt() + 1
556 e.groups.add(Ballot(1, ranking.take(length)))
557 }
558 e.total += count
559 e.finishEntry()
560 return emptyList()
561 }
562 "UNDO" -> {
563 if (e.entryEnds.isEmpty()) throw CommandError("nothing to undo")
564 e.entryEnds.removeAt(e.entryEnds.lastIndex)
565 val end = e.entryEnds.lastOrNull() ?: 0
566 val removed = e.groups.subList(end, e.groups.size)
567 e.total -= removed.sumOf { it.votes }
568 removed.clear()
569 e.invalidate()
570 return emptyList()
571 }
572 "WITHDRAW", "RESTORE" -> {
573 val a = e.candidate(words[1])
574 val active = (e.active and bit(a)) != 0
575 if (command == "WITHDRAW") {
576 if (!active) throw CommandError("already withdrawn ${words[1]}")
577 if (e.active.countOneBits() <= 2) throw CommandError("too few candidates")
578 e.active = e.active and bit(a).inv()
579 } else {
580 if (active) throw CommandError("not withdrawn ${words[1]}")
581 e.active = e.active or bit(a)
582 }
583 return emptyList()
584 }
585 "TIEBREAK" -> {
586 val order = e.checkedNames(words.drop(1))
587 if (order.size != e.names.size) throw CommandError("incomplete tiebreak")
588 e.updateTiebreak(order)
589 return emptyList()
590 }
591 "STATUS" -> {
592 val withdrawn = members((bit(e.names.size) - 1) and e.active.inv())
593 return listOf("[$line]", "election ${e.number}", "active: ${e.namesText(members(e.active))}",
594 "withdrawn: ${if (withdrawn.isEmpty()) "none" else e.namesText(withdrawn)}",
595 "tiebreak: ${e.namesText(e.tiebreak)}", "ballots: ${e.total} in ${e.entryEnds.size} entries")
596 }
597 }
598 if (e.entryEnds.isEmpty()) throw CommandError("no ballots")
599 val requested = if (command == "RANKING" || command == "SPOILERS") {
600 val method = Method.parse(words, 1)
601 if (!method.rankable) throw CommandError("not rankable")
602 if (command == "SPOILERS" && e.active.countOneBits() < 3) throw CommandError("too few candidates")
603 method
604 } else null
605 val compared = IntArray(2)
606 if (command == "COMPARE") {
607 for (i in 0..1) {
608 compared[i] = e.candidate(words[i + 1])
609 if ((e.active and bit(compared[i])) == 0) throw CommandError("withdrawn candidate ${words[i + 1]}")
610 }
611 if (compared[0] == compared[1]) throw CommandError("same candidate")
612 }
613 if (command == "STV" && words[1].toInt() >= e.active.countOneBits()) throw CommandError("too many seats")
614 val lines = mutableListOf("[$line]")
615 when (command) {
616 "RANKING", "SPOILERS" -> {
617 val method = requested!!
618 val result = e.tally(e.active, method)
619 if (command == "RANKING") lines.add(e.namesText(result.order, " > "))
620 else {
621 val winner = result.order.first()
622 lines.add("winner: ${e.names[winner]}")
623 for (a in members(e.active)) if (a != winner) {
624 val other = e.tally(e.active and bit(a).inv(), method).order.first()
625 lines.add("without ${e.names[a]}: ${e.names[other]}${if (other != winner) " (changed)" else ""}")
626 }
627 }
628 }
629 "BALLOTS" -> lines.addAll(e.ballotsReport())
630 "PROFILE" -> {
631 val scores = e.tallies(e.active)
632 val size = e.active.countOneBits()
633 for (a in members(e.active)) {
634 val counts = (0 until size).joinToString(" ") { j ->
635 (scores.approval[j][a] - if (j == 0) 0 else scores.approval[j - 1][a]).toString()
636 }
637 lines.add("${e.names[a]}: $counts unranked ${e.total - scores.approval[size - 1][a]}")
638 }
639 }
640 "PAIRWISE" -> lines.addAll(e.pairLines(e.active, e.pairwise()))
641 "EXHAUSTED" -> lines.addAll(e.elimination(e.active, exhausted = true).lines)
642 "SUMMARY", "COMPARE" -> {
643 val wins = IntArray(LIMIT)
644 val ahead = List(2) { mutableListOf<String>() }
645 for (method in summaryMethods) {
646 val result = e.tally(e.active, method)
647 if (command == "SUMMARY") {
648 val winner = result.order.first()
649 wins[winner]++
650 lines.add("$method: ${e.names[winner]}")
651 } else {
652 val side = if (result.order.indexOf(compared[0]) < result.order.indexOf(compared[1])) 0 else 1
653 ahead[side].add(method.toString())
654 }
655 }
656 if (command == "SUMMARY") lines.add("wins: ${e.values(e.active, wins)}")
657 else {
658 val matrix = e.pairwise()
659 val (a, b) = compared
660 lines.add("${e.names[a]} ${matrix[a][b]}, ${e.names[b]} ${matrix[b][a]}, neither ${e.total - matrix[a][b] - matrix[b][a]}")
661 for (i in 0..1) lines.add("${e.names[compared[i]]} ahead: ${if (ahead[i].isEmpty()) "none" else ahead[i].joinToString(", ")}")
662 }
663 }
664 else -> lines.addAll(e.tally(e.active, Method.parse(words)).lines)
665 }
666 return lines
667 }
668}
