50tallies <- function(election, field) {
51 key <- as.character(field_id(field))
52 if (exists(key, election$cache, inherits = FALSE)) return(get(key, election$cache, inherits = FALSE))
53 count <- length(election$names)
54 size <- length(field)
55 first <- last <- borda <- integer(count)
56 approval <- matrix(0L, size, count)
57 nonblank <- 0L
58 for (index in seq_along(election$votes)) {
59 votes <- election$votes[index]
60 ranking <- effective(election$rankings[[index]], field)
61 positions <- match(seq_len(count), ranking, nomatch = count + 1L)
62 if (length(ranking)) nonblank <- nonblank + votes
63 first <- first + votes * (positions == 1L)
64 if (length(ranking) == size) {
65 last <- last + votes * (positions == size)
66 } else {
67 last <- last + votes * ((seq_len(count) %in% field) & positions == count + 1L)
68 }
69 borda <- borda + votes * pmax.int(0L, size - positions)
70 approval <- approval + votes * outer(seq_len(size), positions, ">=")
71 }
72 result <- list(first = first, last = last, borda = borda, approval = approval, nonblank = nonblank)
73 assign(key, result, election$cache)
74 result
75}
76pairwise <- function(election) {
77 if (exists("pairs", election$cache, inherits = FALSE)) return(election$cache$pairs)
78 count <- length(election$names)
79 result <- matrix(0L, count, count)
80 for (index in seq_along(election$votes)) {
81 ranking <- election$rankings[[index]]
82 positions <- match(seq_len(count), ranking, nomatch = count + 1L)
83 result <- result + election$votes[index] * outer(positions, positions, "<")
84 }
85 election$cache$pairs <- result
86 result

88majority <- function(field, scores) any(2L * scores[field] > sum(scores[field]))
89look_back <- function(election, field, history) {
90 choices <- field
91 tie <- FALSE
92 for (round in rev(seq_along(history))) {
93 scores <- history[[round]]
94 choices <- choices[scores[choices] == min(scores[choices])]
95 if (round == length(history)) tie <- length(choices) > 1L
96 }
97 list(candidate = choices[which.max(election$priority[choices])], tie = tie)
98}
99condorcet <- function(election, field, loser = FALSE) {
100 matrix <- pairwise(election)
101 for (candidate in field) {
102 others <- setdiff(field, candidate)
103 all <- if (loser) all(matrix[candidate, others] < matrix[others, candidate]) else
104 all(matrix[candidate, others] > matrix[others, candidate])
105 if (all) return(candidate)
106 }
107 0L
108}
109elimination <- function(election, field, coombs = FALSE, exhausted = FALSE) {
110 remaining <- field
111 removed <- integer()
112 history <- list()
113 rows <- character()
114 repeat {
115 scores <- tallies(election, remaining)
116 round <- length(rows) + 1L
117 if (exhausted) {
118 active <- sum(scores$first[remaining])
119 row <- sprintf("round %d: active %d, exhausted %d", round, active, election$total - active)
120 } else {
121 row <- sprintf("round %d: %s", round, value_list(election, remaining, scores$first))
122 }
123 if (length(remaining) == 1L || majority(remaining, scores$first)) {
124 order <- c(ordered(election, remaining, scores$first), rev(removed))
125 rows <- c(rows, row)
126 if (exhausted) return(list(order = order, rows = rows))
127 return(finish(election, order, FALSE, rows))
128 }
129 if (coombs) {
130 choices <- remaining[scores$last[remaining] == max(scores$last[remaining])]
131 order <- ordered(election, choices, scores$first)
132 choice <- list(candidate = tail(order, 1L), tie = length(choices) > 1L)
133 row <- paste0(row, "; last ", value_list(election, remaining, scores$last))
134 } else {
135 history <- c(history, list(scores$first))
136 choice <- look_back(election, remaining, history)
137 }
138 if (!exhausted) row <- paste0(row, "; eliminated ", election$names[choice$candidate], tie_text(choice$tie))
139 rows <- c(rows, row)
140 removed <- c(removed, choice$candidate)
141 remaining <- setdiff(remaining, choice$candidate)
142 }

144runoff <- function(election, field) {
145 first <- tallies(election, field)$first
146 order <- ordered(election, field, first)
147 rows <- paste0("round 1: ", value_list(election, field, first))
148 tie <- FALSE
149 if (!majority(field, first)) {
150 finalists <- sort(order[1:2])
151 advance <- length(order) >= 3L && first[order[2L]] == first[order[3L]]
152 second <- tallies(election, finalists)$first
153 tie <- second[order[1L]] == second[order[2L]]
154 if (second[order[2L]] > second[order[1L]]) order[1:2] <- order[2:1]
155 rows <- c(rows, paste0("finalists: ", name_list(election, finalists), tie_text(advance)),
156 paste0("round 2: ", value_list(election, finalists, second)))
157 }
158 finish(election, order, tie, rows)
159}
160bucklin <- function(election, field) {
161 scores <- tallies(election, field)
162 line <- scores$nonblank %/% 2L + 1L
163 rows <- paste("majority:", line)
164 order <- effective(election$tiebreak, field)
165 for (depth in seq_along(field)) {
166 final <- scores$approval[depth, ]
167 order <- order[order(-final[order], method = "radix")]
168 rows <- c(rows, sprintf("round %d: %s", depth, value_list(election, field, final)))
169 if (any(final[field] >= line)) break
170 }
171 finish(election, order, final[order[1L]] == final[order[2L]], rows)
172}
173borda_elimination <- function(election, field, nanson = FALSE) {
174 remaining <- field
175 tail <- integer()
176 history <- list()
177 rows <- character()
178 while (length(remaining) >= 2L) {
179 scores <- tallies(election, remaining)$borda
180 row <- sprintf("round %d: %s", length(rows) + 1L, value_list(election, remaining, scores))
181 if (nanson) {
182 below <- remaining[scores[remaining] * length(remaining) < sum(scores[remaining])]
183 if (!length(below)) {
184 rows <- c(rows, row)
185 break
186 }
187 row <- paste0(row, "; eliminated ", name_list(election, below))
188 removed <- ordered(election, below, scores)
189 } else {
190 history <- c(history, list(scores))
191 choice <- look_back(election, remaining, history)
192 removed <- choice$candidate
193 row <- paste0(row, "; eliminated ", election$names[choice$candidate], tie_text(choice$tie))
194 }
195 rows <- c(rows, row)
196 remaining <- setdiff(remaining, removed)
197 tail <- c(removed, tail)
198 }
199 order <- c(effective(election$tiebreak, remaining), tail)
200 finish(election, order, nanson && length(remaining) > 1L, rows)

219condorcet_report <- function(election, field) {
220 matrix <- pairwise(election)
221 winner <- condorcet(election, field)
222 loser <- condorcet(election, field, TRUE)
223 weak <- closure(field, matrix >= t(matrix))
224 strong <- closure(field, matrix > t(matrix))
225 smith <- field[vapply(field, function(a) all(weak[a, field]), logical(1))]
226 schwartz <- field[vapply(field, function(a) all(!strong[field, a] | strong[a, field]), logical(1))]
227 list(order = integer(), rows = c(
228 paste("winner:", if (winner == 0L) "NONE" else election$names[winner]),
229 paste("loser:", if (loser == 0L) "NONE" else election$names[loser]),
230 paste0("smith: ", name_list(election, smith)),
231 paste0("schwartz: ", name_list(election, schwartz))))

233topological <- function(election, field, relation) {
234 remaining <- field
235 order <- integer()
236 tie <- FALSE
237 while (length(remaining)) {
238 choices <- remaining[vapply(remaining, function(a) !any(relation[setdiff(remaining, a), a]), logical(1))]
239 if (!length(order)) tie <- length(choices) > 1L
240 chosen <- choices[which.min(election$priority[choices])]
241 order <- c(order, chosen)
242 remaining <- setdiff(remaining, chosen)
243 }
244 list(order = order, tie = tie)

246schulze <- function(election, field) {
247 matrix <- pairwise(election)
248 paths <- matrix * (matrix > t(matrix))
249 for (via in field) {
250 for (first in setdiff(field, via)) {
251 for (second in setdiff(field, c(via, first))) {
252 paths[first, second] <- max(paths[first, second], min(paths[first, via], paths[via, second]))
253 }
254 }
255 }
256 result <- topological(election, field, paths > t(paths))
257 rows <- c(pair_lines(election, field, paths), paste0("ranking: ", name_list(election, result$order, " > ")))
258 finish(election, result$order, result$tie, rows)

260ranked_pairs <- function(election, field) {
261 matrix <- pairwise(election)
262 pairs <- matrix(integer(), nrow = 0L, ncol = 2L)
263 for (first in field) {
264 for (second in field) if (matrix[first, second] > matrix[second, first]) pairs <- rbind(pairs, c(first, second))
265 }
266 if (nrow(pairs)) {
267 reverse <- pairs[, 2:1, drop = FALSE]
268 order <- order(-matrix[pairs], matrix[reverse], election$priority[pairs[, 1L]],
269 election$priority[pairs[, 2L]], method = "radix")
270 pairs <- pairs[order, , drop = FALSE]
271 }
272 locked <- reached <- matrix(FALSE, length(election$names), length(election$names))
273 rows <- character()
274 for (index in seq_len(nrow(pairs))) {
275 first <- pairs[index, 1L]
276 second <- pairs[index, 2L]
277 accepted <- !reached[second, first]
278 label <- if (accepted) "lock" else "skip"
279 rows <- c(rows, sprintf("%s %s>%s %d-%d", label, election$names[first], election$names[second],
280 matrix[first, second], matrix[second, first]))
281 if (accepted) {
282 locked[first, second] <- reached[first, second] <- TRUE
283 reached <- closure(field, reached)
284 }
285 }
286 result <- topological(election, field, locked)
287 finish(election, result$order, result$tie, rows)
288}
289kemeny <- function(election, field) {
290 matrix <- pairwise(election)
291 limit <- bitwShiftL(1L, length(election$names))
292 best <- ways <- integer(limit)
293 heads <- rep(list(integer()), limit)
294 ways[1L] <- 1L
295 for (subset in seq_len(limit - 1L)) {
296 candidates <- members(subset, length(election$names))
297 if (!all(candidates %in% field)) next
298 index <- subset + 1L
299 best[index] <- -1L
300 for (candidate in candidates) {
301 rest <- subset - bitwShiftL(1L, candidate - 1L)
302 score <- best[rest + 1L] + sum(matrix[candidate, setdiff(candidates, candidate)])
303 if (score > best[index]) {
304 best[index] <- score
305 ways[index] <- 0L
306 heads[[index]] <- integer()
307 }
308 if (score == best[index]) {
309 ways[index] <- ways[index] + ways[rest + 1L]
310 heads[[index]] <- c(heads[[index]], candidate)
311 }
312 }
313 }
314 key <- field_id(field)
315 remaining <- key
316 order <- integer()
317 while (remaining > 0L) {
318 chosen <- effective(election$tiebreak, heads[[remaining + 1L]])[1L]
319 order <- c(order, chosen)
320 remaining <- remaining - bitwShiftL(1L, chosen - 1L)
321 }
322 rows <- c(paste("score:", best[key + 1L]), paste("optimal:", ways[key + 1L]),
323 paste0("ranking: ", name_list(election, order, " > ")))
324 finish(election, order, length(heads[[key + 1L]]) > 1L, rows)
325}
326stv <- function(election, field, seats) {
327 values <- vapply(election$rankings, function(ranking) if (any(ranking %in% field)) 100L else 0L, integer(1))
328 quota <- (100L * tallies(election, field)$nonblank) %/% (seats + 1L) + 1L
329 continuing <- field
330 elected <- integer()
331 history <- list()
332 rows <- paste0("quota: ", money(quota))
333 repeat {
334 scores <- integer(length(election$names))
335 owners <- integer(length(election$votes))
336 exhausted <- 0L
337 for (index in seq_along(election$votes)) {
338 if (values[index] == 0L) next
339 ranking <- effective(election$rankings[[index]], continuing)
340 value <- election$votes[index] * values[index]
341 if (length(ranking)) {
342 owners[index] <- ranking[1L]
343 scores[ranking[1L]] <- scores[ranking[1L]] + value
344 } else {
345 exhausted <- exhausted + value
346 }
347 }
348 history <- c(history, list(scores))
349 row <- sprintf("round %d: %s; exhausted %s", length(history), value_list(election, continuing, scores, TRUE), money(exhausted))
350 order <- ordered(election, continuing, scores)
351 if (length(continuing) == seats - length(elected)) {
352 elected <- c(elected, order)
353 rows <- c(rows, paste0(row, "; elected ", name_list(election, order)))
354 break
355 }
356 chosen <- order[1L]
357 if (scores[chosen] >= quota) {
358 surplus <- scores[chosen] - quota
359 elected <- c(elected, chosen)
360 continuing <- setdiff(continuing, chosen)
361 indices <- which(owners == chosen)
362 values[indices] <- (values[indices] * surplus) %/% scores[chosen]
363 rows <- c(rows, paste0(row, "; elected ", election$names[chosen], " (surplus ", money(surplus), ")"))
364 if (length(elected) == seats) break
365 } else {
366 choice <- look_back(election, continuing, history)
367 continuing <- setdiff(continuing, choice$candidate)
368 rows <- c(rows, paste0(row, "; excluded ", election$names[choice$candidate], tie_text(choice$tie)))
369 }
370 }
371 list(order = integer(), rows = c(rows, paste0("elected: ", name_list(election, elected))))
372}
373tally <- function(election, field, method) {
374 words <- strsplit(method, " ", fixed = TRUE)[[1L]]
375 kind <- words[1L]
376 if (kind == "RUNOFF") return(runoff(election, field))
377 if (kind %in% c("IRV", "COOMBS")) return(elimination(election, field, kind == "COOMBS"))
378 if (kind == "BUCKLIN") return(bucklin(election, field))
379 if (kind %in% c("BALDWIN", "NANSON")) return(borda_elimination(election, field, kind == "NANSON"))
380 if (kind == "CONDORCET") return(condorcet_report(election, field))
381 if (kind == "SCHULZE") return(schulze(election, field))
382 if (kind == "RANKEDPAIRS") return(ranked_pairs(election, field))
383 if (kind == "KEMENY") return(kemeny(election, field))
384 if (kind == "STV") return(stv(election, field, as.integer(words[2L])))
385 base <- tallies(election, field)
386 scores <- integer(length(election$names))
387 label <- "scores"
388 descending <- TRUE
389 rows <- character()
390 winner <- 0L
391 if (kind == "PLURALITY") {
392 scores <- base$first
393 label <- "votes"
394 } else if (kind %in% c("BORDA", "BLACK")) {
395 scores <- base$borda
396 if (kind == "BLACK") {
397 winner <- condorcet(election, field)
398 rows <- paste("condorcet:", if (winner == 0L) "NONE" else election$names[winner])
399 }
400 } else if (kind == "VETO") {
401 scores <- base$last
402 label <- "vetoes"
403 descending <- FALSE
404 } else if (kind == "APPROVAL") {
405 scores <- base$approval[min(as.integer(words[2L]), length(field)), ]
406 label <- "approvals"
407 } else if (kind == "COPELAND") {
408 matrix <- pairwise(election)
409 parts <- character()
410 for (candidate in field) {
411 others <- setdiff(field, candidate)
412 forward <- matrix[candidate, others]
413 backward <- matrix[others, candidate]
414 wins <- sum(forward > backward)
415 draws <- sum(forward == backward)
416 losses <- sum(forward < backward)
417 scores[candidate] <- 2L * wins + draws
418 score <- scores[candidate]
419 parts <- c(parts, sprintf("%s %d.%d (%d-%d-%d)", election$names[candidate],
420 score %/% 2L, (score %% 2L) * 5L, wins, draws, losses))
421 }
422 rows <- paste0("scores: ", paste(parts, collapse = ", "))
423 } else if (kind == "MINIMAX") {
424 matrix <- pairwise(election)
425 descending <- FALSE
426 for (candidate in field) {
427 others <- setdiff(field, candidate)
428 opposition <- matrix[others, candidate]
429 support <- matrix[candidate, others]
430 values <- if (words[2L] == "OPPOSITION") opposition else
431 if (words[2L] == "MARGIN") pmax.int(0L, opposition - support) else opposition * (opposition > support)
432 scores[candidate] <- max(values)
433 }
434 }
435 order <- ordered(election, field, scores, descending)
436 tie <- scores[order[1L]] == scores[order[2L]]
437 if (winner != 0L) {
438 order <- c(winner, setdiff(order, winner))
439 tie <- FALSE
440 }
441 if (kind != "COPELAND") rows <- c(rows, paste0(label, ": ", value_list(election, field, scores)))
442 finish(election, order, tie, rows)
443}
444ballots_report <- function(election) {

453 c(rows, paste(counts[order], distinct[order]))
454}

462execute <- function(lab, line) {
463 words <- strsplit(line, " ", fixed = TRUE)[[1L]]
464 command <- words[1L]
465 if (command == "CANDIDATES") {
466 names <- words[-1L]
467 if (length(names) == 1L) fail("too few candidates")
468 duplicate <- which(duplicated(names))
469 if (length(duplicate)) fail(paste("duplicate candidate", names[duplicate[1L]]))
470 election <- new_election(lab$number + 1L, names)
471 lab$number <- lab$number + 1L
472 lab$current <- election
473 return(sprintf("election %d: %d candidates", lab$number, length(names)))
474 }
475 election <- lab$current
476 if (is.null(election)) fail("no election")
477 if (command == "BALLOT") {
478 ranking <- if (words[3L] == "-") integer() else checked_names(election, strsplit(words[3L], ">", fixed = TRUE)[[1L]])
479 votes <- as.integer(words[2L])
480 election$votes <- c(election$votes, votes)
481 election$rankings <- c(election$rankings, list(ranking))
482 election$total <- election$total + votes
483 election$ends <- c(election$ends, length(election$votes))
484 election$cache <- new.env(parent = emptyenv())
485 lab$current <- election
486 return(character())
487 } else if (command == "RANDOM") {
488 count <- as.integer(words[2L])
489 state <- as.bigz(words[3L])
490 for (group in seq_len(count)) {
491 ranking <- which(election$active)
492 for (size in rev(seq_len(length(ranking) - 1L) + 1L)) {
493 state <- (48271L * state) %% 2147483647L
494 index <- as.integer(state %% size) + 1L
495 ranking[c(size, index)] <- ranking[c(index, size)]
496 }
497 state <- (48271L * state) %% 2147483647L
498 ranking <- ranking[seq_len(as.integer(state %% length(ranking)) + 1L)]
499 election$votes <- c(election$votes, 1L)
500 election$rankings <- c(election$rankings, list(ranking))
501 }
502 election$total <- election$total + count
503 election$ends <- c(election$ends, length(election$votes))
504 election$cache <- new.env(parent = emptyenv())
505 lab$current <- election
506 return(character())
507 } else if (command == "UNDO") {
508 if (!length(election$ends)) fail("nothing to undo")
509 election$ends <- head(election$ends, -1L)
510 end <- tail(c(0L, election$ends), 1L)
511 election$votes <- election$votes[seq_len(end)]
512 election$rankings <- election$rankings[seq_len(end)]
513 election$total <- sum(election$votes)
514 election$cache <- new.env(parent = emptyenv())
515 lab$current <- election
516 return(character())
517 } else if (command %in% c("WITHDRAW", "RESTORE")) {
518 candidate <- candidate_index(election, words[2L])
519 if (command == "WITHDRAW") {
520 if (!election$active[candidate]) fail(paste("already withdrawn", words[2L]))
521 if (sum(election$active) <= 2L) fail("too few candidates")
522 election$active[candidate] <- FALSE
523 } else {
524 if (election$active[candidate]) fail(paste("not withdrawn", words[2L]))
525 election$active[candidate] <- TRUE
526 }
527 lab$current <- election
528 return(character())
529 } else if (command == "TIEBREAK") {
530 order <- checked_names(election, words[-1L])
531 if (length(order) != length(election$names)) fail("incomplete tiebreak")
532 election$tiebreak <- order
533 election$priority <- match(seq_along(election$names), order)
534 lab$current <- election
535 return(character())
536 }
537 heading <- paste0("[", line, "]")
538 if (command == "STATUS") return(c(heading, status(election)))
539 if (!length(election$ends)) fail("no ballots")
540 field <- which(election$active)
541 if (command %in% c("RANKING", "SPOILERS")) {
542 method <- paste(words[-1L], collapse = " ")
543 if (words[2L] %in% c("CONDORCET", "STV")) fail("not rankable")
544 if (command == "SPOILERS" && length(field) < 3L) fail("too few candidates")
545 result <- tally(election, field, method)
546 if (command == "RANKING") return(c(heading, name_list(election, result$order, " > ")))
547 winner <- result$order[1L]
548 rows <- paste0("winner: ", election$names[winner])
549 for (candidate in setdiff(field, winner)) {
550 other <- tally(election, setdiff(field, candidate), method)$order[1L]
551 rows <- c(rows, paste0("without ", election$names[candidate], ": ", election$names[other],
552 if (other != winner) " (changed)" else ""))
553 }
554 return(c(heading, rows))
555 }
556 compared <- integer()
557 if (command == "COMPARE") {
558 for (name in words[-1L]) {
559 candidate <- candidate_index(election, name)
560 if (!election$active[candidate]) fail(paste("withdrawn candidate", name))
561 compared <- c(compared, candidate)
562 }
563 if (compared[1L] == compared[2L]) fail("same candidate")
564 }
565 if (command == "STV" && as.integer(words[2L]) >= length(field)) fail("too many seats")
566 rows <- switch(command,
567 BALLOTS = ballots_report(election),
568 PROFILE = {
569 scores <- tallies(election, field)
570 vapply(field, function(candidate) {
571 cumulative <- scores$approval[, candidate]
572 counts <- cumulative - c(0L, head(cumulative, -1L))
573 sprintf("%s: %s unranked %d", election$names[candidate], paste(counts, collapse = " "), election$total - tail(cumulative, 1L))
574 }, character(1))
575 },
576 PAIRWISE = pair_lines(election, field, pairwise(election)),
577 EXHAUSTED = elimination(election, field, exhausted = TRUE)$rows,
578 SUMMARY = NULL,
579 COMPARE = NULL,
580 tally(election, field, line)$rows
581 )
582 if (command %in% c("SUMMARY", "COMPARE")) {
583 rows <- character()
584 wins <- integer(length(election$names))
585 ahead <- list(character(), character())
586 for (method in summary_methods) {
587 result <- tally(election, field, method)
588 if (command == "SUMMARY") {
589 winner <- result$order[1L]
590 wins[winner] <- wins[winner] + 1L
591 rows <- c(rows, paste0(method, ": ", election$names[winner]))
592 } else {
593 side <- if (match(compared[1L], result$order) < match(compared[2L], result$order)) 1L else 2L
594 ahead[[side]] <- c(ahead[[side]], method)
595 }
596 }
597 if (command == "SUMMARY") {
598 rows <- c(rows, paste0("wins: ", value_list(election, field, wins)))
599 } else {
600 matrix <- pairwise(election)
601 first <- compared[1L]
602 second <- compared[2L]
603 forward <- matrix[first, second]
604 backward <- matrix[second, first]
605 rows <- sprintf("%s %d, %s %d, neither %d", election$names[first], forward,
606 election$names[second], backward, election$total - forward - backward)
607 for (side in 1:2) {
608 rows <- c(rows, paste0(election$names[compared[side]], " ahead: ",
609 if (length(ahead[[side]])) paste(ahead[[side]], collapse = ", ") else "none"))
610 }
611 }
612 }
613 c(heading, rows)
614}
