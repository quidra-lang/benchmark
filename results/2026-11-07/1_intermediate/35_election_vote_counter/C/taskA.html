71static Field bit(int candidate) { return 1U << candidate; }
72static List members(Field field) {
73 List result = {0};
74 for (int a = 0; a < MAX_CANDIDATES; ++a)
75 if (field & bit(a)) result.items[result.count++] = a;
76 return result;
77}
78static Field mask(List list) {
79 Field result = 0;
80 for (int i = 0; i < list.count; ++i) result |= bit(list.items[i]);
81 return result;
82}
83static int position(List list, int candidate) {
84 for (int i = 0; i < list.count; ++i) if (list.items[i] == candidate) return i;
85 return -1;
86}
87static List effective(List ranking, Field field) {
88 List result = {0};
89 for (int i = 0; i < ranking.count; ++i)
90 if (field & bit(ranking.items[i])) result.items[result.count++] = ranking.items[i];
91 return result;
92}
93static Text names(const Election *e, List list, const char *separator) {
94 Text result = {0};
95 for (int i = 0; i < list.count; ++i)
96 append(&result, "%s%s", i ? separator : "", e->names[list.items[i]]);
97 return result;
98}
99static Text values(const Election *e, Field field, const int scores[], bool cents) {
100 Text result = {0};
101 List list = members(field);
102 for (int i = 0; i < list.count; ++i) {
103 int a = list.items[i];
104 append(&result, "%s%s ", i ? ", " : "", e->names[a]);
105 if (cents) append(&result, "%d.%02d", scores[a] / 100, scores[a] % 100);
106 else append(&result, "%d", scores[a]);
107 }
108 return result;
109}
110static const char *tie_mark(bool tie) { return tie ? " (tie)" : ""; }
111static void winner(FILE *output, const Election *e, Result result) {
112 emit(output, "winner: %s%s\n", e->names[result.order.items[0]], tie_mark(result.tie));
113}
114static List ordered(const Election *e, Field field, const int scores[], bool descending) {
115 List result = members(field);
116 for (int i = 1; i < result.count; ++i) {
117 int a = result.items[i], j = i;
118 while (j > 0) {
119 int b = result.items[j - 1];
120 bool before = scores[a] == scores[b] ? e->priority[a] < e->priority[b]
121 : descending ? scores[a] > scores[b] : scores[a] < scores[b];
122 if (!before) break;
123 result.items[j] = b;
124 --j;
125 }
126 result.items[j] = a;
127 }
128 return result;
129}
130static Result scored(const Election *e, Field field, const int scores[], bool descending) {
131 Result result = {.order = ordered(e, field, scores, descending)};
132 result.tie = result.order.count > 1 && scores[result.order.items[0]] == scores[result.order.items[1]];
133 return result;
134}
135static int earliest(const Election *e, Field field) {
136 for (int i = 0; i < e->count; ++i) if (field & bit(e->tiebreak.items[i])) return e->tiebreak.items[i];
137 return -1;
138}
139static void invalidate(Election *e) {
140 for (int i = 0; i < SUBSETS; ++i) e->cache[i].ready = false;
141 e->pairs_ready = false;
142}
143static Tallies *tallies(Election *e, Field field) {
144 Tallies *result = &e->cache[field];
145 if (result->ready) return result;
146 *result = (Tallies){0};
147 List candidates = members(field);
148 int m = candidates.count;
149 for (int i = 0; i < e->group_count; ++i) {
150 Ballot group = e->groups[i];
151 List rank = effective(group.ranking, field);
152 Field ranked = mask(rank);
153 if (rank.count) {
154 result->first[rank.items[0]] += group.votes;
155 result->nonblank += group.votes;
156 }
157 if (rank.count == m) result->last[rank.items[m - 1]] += group.votes;
158 else for (int j = 0; j < m; ++j)
159 if (!(ranked & bit(candidates.items[j]))) result->last[candidates.items[j]] += group.votes;
160 for (int j = 0; j < rank.count; ++j) {
161 int a = rank.items[j];
162 result->borda[a] += group.votes * (m - j - 1);
163 for (int depth = j; depth < m; ++depth) result->approval[depth][a] += group.votes;
164 }
165 }
166 result->ready = true;
167 return result;
168}
169static void pairwise(Election *e) {
170 if (e->pairs_ready) return;
171 memset(e->pairwise, 0, sizeof e->pairwise);
172 for (int i = 0; i < e->group_count; ++i) {
173 Ballot group = e->groups[i];
174 int positions[MAX_CANDIDATES];
175 for (int a = 0; a < e->count; ++a) positions[a] = MAX_CANDIDATES;
176 for (int j = 0; j < group.ranking.count; ++j) positions[group.ranking.items[j]] = j;
177 for (int a = 0; a < e->count; ++a) for (int b = 0; b < e->count; ++b)
178 if (positions[a] < positions[b]) e->pairwise[a][b] += group.votes;
179 }
180 e->pairs_ready = true;
181}
182static int condorcet(Election *e, Field field, bool loser) {
183 pairwise(e);
184 List list = members(field);
185 for (int i = 0; i < list.count; ++i) {
186 int a = list.items[i];
187 bool all = true;
188 for (int j = 0; j < list.count; ++j) if (i != j) {
189 int b = list.items[j];
190 if (loser ? e->pairwise[a][b] >= e->pairwise[b][a] : e->pairwise[a][b] <= e->pairwise[b][a]) all = false;
191 }
192 if (all) return a;
193 }
194 return -1;
195}
196static int sum(Field field, const int scores[]) {
197 int result = 0;
198 for (int a = 0; a < MAX_CANDIDATES; ++a) if (field & bit(a)) result += scores[a];
199 return result;
200}
201static bool majority(Field field, const int scores[]) {
202 int total = sum(field, scores);
203 for (int a = 0; a < MAX_CANDIDATES; ++a) if ((field & bit(a)) && 2 * scores[a] > total) return true;
204 return false;
205}
206static int look_back(const Election *e, Field field, int history[][MAX_CANDIDATES], int rounds, bool *tie) {
207 Field choices = field;
208 for (int r = rounds - 1; r >= 0; --r) {
209 int least = 2147483647;
210 List list = members(choices);
211 for (int i = 0; i < list.count; ++i) if (history[r][list.items[i]] < least) least = history[r][list.items[i]];
212 for (int i = 0; i < list.count; ++i) if (history[r][list.items[i]] != least) choices &= ~bit(list.items[i]);
213 if (r == rounds - 1) *tie = members(choices).count > 1;
214 }
215 for (int i = e->count - 1; i >= 0; --i) if (choices & bit(e->tiebreak.items[i])) return e->tiebreak.items[i];
216 return -1;
217}
218static Result elimination(Election *e, Field field, MethodKind kind, FILE *output, bool exhausted) {
219 Field remaining = field;
220 List removed = {0};
221 int history[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}}, round = 0;
222 for (;;) {
223 Tallies *current = tallies(e, remaining);
224 ++round;
225 bool done = members(remaining).count == 1 || majority(remaining, current->first);
226 if (exhausted) {
227 int active = sum(remaining, current->first);
228 emit(output, "round %d: active %d, exhausted %d\n", round, active, e->total - active);
229 } else emit(output, "round %d: %s", round, values(e, remaining, current->first, false).data);
230 if (done) {
231 if (!exhausted) emit(output, "\n");
232 Result result = {.order = ordered(e, remaining, current->first, true)};
233 for (int i = removed.count - 1; i >= 0; --i) result.order.items[result.order.count++] = removed.items[i];
234 if (!exhausted) winner(output, e, result);
235 return result;
236 }
237 bool tie = false;
238 int eliminated;
239 if (kind == COOMBS) {
240 List worst = ordered(e, remaining, current->last, true);
241 int maximum = current->last[worst.items[0]];
242 Field choices = 0;
243 for (int i = 0; i < worst.count; ++i) if (current->last[worst.items[i]] == maximum) choices |= bit(worst.items[i]);
244 tie = members(choices).count > 1;
245 List least = ordered(e, choices, current->first, true);
246 eliminated = least.items[least.count - 1];
247 emit(output, "; last %s", values(e, remaining, current->last, false).data);
248 } else {
249 memcpy(history[round - 1], current->first, sizeof current->first);
250 eliminated = look_back(e, remaining, history, round, &tie);
251 }
252 if (!exhausted) emit(output, "; eliminated %s%s\n", e->names[eliminated], tie_mark(tie));
253 removed.items[removed.count++] = eliminated;
254 remaining &= ~bit(eliminated);
255 }
256}
257static Result runoff(Election *e, Field field, FILE *output) {
258 Tallies *first = tallies(e, field);
259 Result result = {.order = ordered(e, field, first->first, true)};
260 emit(output, "round 1: %s\n", values(e, field, first->first, false).data);
261 if (!majority(field, first->first)) {
262 int a = result.order.items[0], b = result.order.items[1];
263 Field finalists = bit(a) | bit(b);
264 bool advance_tie = result.order.count >= 3 && first->first[b] == first->first[result.order.items[2]];
265 Tallies *second = tallies(e, finalists);
266 result.tie = second->first[a] == second->first[b];
267 if (second->first[b] > second->first[a]) {
268 result.order.items[0] = b;
269 result.order.items[1] = a;
270 }
271 emit(output, "finalists: %s%s\n", names(e, members(finalists), ", ").data, tie_mark(advance_tie));
272 emit(output, "round 2: %s\n", values(e, finalists, second->first, false).data);
273 }
274 winner(output, e, result);
275 return result;
276}
277static Result bucklin(Election *e, Field field, FILE *output) {
278 Tallies *scores = tallies(e, field);
279 int line = scores->nonblank / 2 + 1, final = 0;
280 List list = members(field);
281 emit(output, "majority: %d\n", line);
282 for (int depth = 0; depth < list.count; ++depth) {
283 final = depth;
284 emit(output, "round %d: %s\n", depth + 1, values(e, field, scores->approval[depth], false).data);
285 bool done = false;
286 for (int i = 0; i < list.count; ++i) if (scores->approval[depth][list.items[i]] >= line) done = true;
287 if (done) break;
288 }
289 Result result = {.order = list};
290 for (int i = 1; i < list.count; ++i) {
291 int a = result.order.items[i], j = i;
292 while (j > 0) {
293 int b = result.order.items[j - 1], depth = final;
294 while (depth >= 0 && scores->approval[depth][a] == scores->approval[depth][b]) --depth;
295 bool before = depth < 0 ? e->priority[a] < e->priority[b] : scores->approval[depth][a] > scores->approval[depth][b];
296 if (!before) break;
297 result.order.items[j--] = b;
298 }
299 result.order.items[j] = a;
300 }
301 result.tie = list.count > 1 && scores->approval[final][result.order.items[0]] == scores->approval[final][result.order.items[1]];
302 winner(output, e, result);
303 return result;
304}
305static Result borda_elimination(Election *e, Field field, bool nanson, FILE *output) {
306 Field remaining = field;
307 List tail = {0};
308 int history[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}}, round = 0;
309 while (members(remaining).count >= 2) {
310 Tallies *scores = tallies(e, remaining);
311 ++round;
312 emit(output, "round %d: %s", round, values(e, remaining, scores->borda, false).data);
313 List removed;
314 if (nanson) {
315 int total = sum(remaining, scores->borda), size = members(remaining).count;
316 Field below = 0;
317 for (int a = 0; a < e->count; ++a) if ((remaining & bit(a)) && scores->borda[a] * size < total) below |= bit(a);
318 if (!below) { emit(output, "\n"); break; }
319 emit(output, "; eliminated %s\n", names(e, members(below), ", ").data);
320 removed = ordered(e, below, scores->borda, true);
321 } else {
322 memcpy(history[round - 1], scores->borda, sizeof scores->borda);
323 bool tie = false;
324 int a = look_back(e, remaining, history, round, &tie);
325 removed = members(bit(a));
326 emit(output, "; eliminated %s%s\n", e->names[a], tie_mark(tie));
327 }
328 remaining &= ~mask(removed);
329 for (int i = 0; i < tail.count; ++i) removed.items[removed.count++] = tail.items[i];
330 tail = removed;
331 }
332 int zero[MAX_CANDIDATES] = {0};
333 Result result = {.order = ordered(e, remaining, zero, true), .tie = nanson && members(remaining).count > 1};
334 for (int i = 0; i < tail.count; ++i) result.order.items[result.order.count++] = tail.items[i];
335 winner(output, e, result);
336 return result;
337}
338static void pair_lines(FILE *output, const Election *e, Field field, int matrix[][MAX_CANDIDATES]) {
339 List list = members(field);
340 for (int i = 0; i < list.count; ++i) for (int j = i + 1; j < list.count; ++j) {
341 int a = list.items[i], b = list.items[j];
342 const char *win = matrix[a][b] == matrix[b][a] ? "tie" : e->names[matrix[a][b] > matrix[b][a] ? a : b];
343 emit(output, "%s vs %s: %d-%d (%s)\n", e->names[a], e->names[b], matrix[a][b], matrix[b][a], win);
344 }
345}
346static void closure(Field field, bool relation[][MAX_CANDIDATES]) {
347 List list = members(field);
348 for (int k = 0; k < list.count; ++k) for (int i = 0; i < list.count; ++i) for (int j = 0; j < list.count; ++j) {
349 int a = list.items[i], b = list.items[j], via = list.items[k];
350 relation[a][b] = relation[a][b] || (relation[a][via] && relation[via][b]);
351 }
352}
353static void condorcet_report(Election *e, Field field, FILE *output) {
354 int win = condorcet(e, field, false), lose = condorcet(e, field, true);
355 bool weak[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}}, strong[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}};
356 List list = members(field), smith = {0}, schwartz = {0};
357 for (int i = 0; i < list.count; ++i) for (int j = 0; j < list.count; ++j) {
358 int a = list.items[i], b = list.items[j];
359 weak[a][b] = e->pairwise[a][b] >= e->pairwise[b][a];
360 strong[a][b] = e->pairwise[a][b] > e->pairwise[b][a];
361 }
362 closure(field, weak);
363 closure(field, strong);
364 for (int i = 0; i < list.count; ++i) {
365 int a = list.items[i];
366 bool top = true, unbeaten = true;
367 for (int j = 0; j < list.count; ++j) {
368 int b = list.items[j];
369 top = top && weak[a][b];
370 unbeaten = unbeaten && (!strong[b][a] || strong[a][b]);
371 }
372 if (top) smith.items[smith.count++] = a;
373 if (unbeaten) schwartz.items[schwartz.count++] = a;
374 }
375 emit(output, "winner: %s\nloser: %s\nsmith: %s\nschwartz: %s\n", win < 0 ? "NONE" : e->names[win],
376 lose < 0 ? "NONE" : e->names[lose], names(e, smith, ", ").data, names(e, schwartz, ", ").data);
377}
378static Result topological(const Election *e, Field field, bool relation[][MAX_CANDIDATES]) {
379 Result result = {0};
380 Field remaining = field;
381 while (remaining) {
382 Field choices = 0;
383 for (int a = 0; a < e->count; ++a) if (remaining & bit(a)) {
384 bool incoming = false;
385 for (int b = 0; b < e->count; ++b) if (b != a && (remaining & bit(b)) && relation[b][a]) incoming = true;
386 if (!incoming) choices |= bit(a);
387 }
388 if (result.order.count == 0) result.tie = members(choices).count > 1;
389 int a = earliest(e, choices);
390 result.order.items[result.order.count++] = a;
391 remaining &= ~bit(a);
392 }
393 return result;
394}
395static Result schulze(Election *e, Field field, FILE *output) {
396 pairwise(e);
397 int paths[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}};
398 bool relation[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}};
399 List list = members(field);
400 for (int i = 0; i < list.count; ++i) for (int j = 0; j < list.count; ++j) {
401 int a = list.items[i], b = list.items[j];
402 if (e->pairwise[a][b] > e->pairwise[b][a]) paths[a][b] = e->pairwise[a][b];
403 }
404 for (int i = 0; i < list.count; ++i) for (int j = 0; j < list.count; ++j) for (int k = 0; k < list.count; ++k) {
405 if (i == j || i == k || j == k) continue;
406 int via = list.items[i], a = list.items[j], b = list.items[k];
407 int strength = paths[a][via] < paths[via][b] ? paths[a][via] : paths[via][b];
408 if (strength > paths[a][b]) paths[a][b] = strength;
409 }
410 for (int a = 0; a < e->count; ++a) for (int b = 0; b < e->count; ++b) relation[a][b] = paths[a][b] > paths[b][a];
411 Result result = topological(e, field, relation);
412 pair_lines(output, e, field, paths);
413 emit(output, "ranking: %s\n", names(e, result.order, " > ").data);
414 winner(output, e, result);
415 return result;
416}
417typedef struct { int from, to; } Edge;
418static bool edge_before(const Election *e, Edge a, Edge b) {
419 int aw = e->pairwise[a.from][a.to], bw = e->pairwise[b.from][b.to];
420 if (aw != bw) return aw > bw;
421 int al = e->pairwise[a.to][a.from], bl = e->pairwise[b.to][b.from];
422 if (al != bl) return al < bl;
423 if (a.from != b.from) return e->priority[a.from] < e->priority[b.from];
424 return e->priority[a.to] < e->priority[b.to];
425}
426static Result ranked_pairs(Election *e, Field field, FILE *output) {
427 pairwise(e);
428 Edge pairs[MAX_CANDIDATES * MAX_CANDIDATES];
429 int count = 0;
430 List list = members(field);
431 for (int i = 0; i < list.count; ++i) for (int j = 0; j < list.count; ++j) {
432 int a = list.items[i], b = list.items[j];
433 if (e->pairwise[a][b] > e->pairwise[b][a]) pairs[count++] = (Edge){a, b};
434 }
435 for (int i = 1; i < count; ++i) {
436 Edge edge = pairs[i];
437 int j = i;
438 while (j > 0 && edge_before(e, edge, pairs[j - 1])) { pairs[j] = pairs[j - 1]; --j; }
439 pairs[j] = edge;
440 }
441 bool locked[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}}, reached[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}};
442 for (int i = 0; i < count; ++i) {
443 int a = pairs[i].from, b = pairs[i].to;
444 bool lock = !reached[b][a];
445 emit(output, "%s %s>%s %d-%d\n", lock ? "lock" : "skip", e->names[a], e->names[b], e->pairwise[a][b], e->pairwise[b][a]);
446 if (lock) {
447 locked[a][b] = reached[a][b] = true;
448 closure(field, reached);
449 }
450 }
451 Result result = topological(e, field, locked);
452 winner(output, e, result);
453 return result;
454}
455static Result kemeny(Election *e, Field field, FILE *output) {
456 pairwise(e);
457 int best[SUBSETS] = {0}, ways[SUBSETS] = {0};
458 Field heads[SUBSETS] = {0};
459 ways[0] = 1;
460 for (Field subset = 1; subset <= field; ++subset) {
461 if (subset & ~field) continue;
462 best[subset] = -1;
463 List list = members(subset);
464 for (int i = 0; i < list.count; ++i) {
465 int a = list.items[i];
466 Field rest = subset & ~bit(a);
467 int score = best[rest];
468 for (int j = 0; j < list.count; ++j) if (i != j) score += e->pairwise[a][list.items[j]];
469 if (score > best[subset]) { best[subset] = score; ways[subset] = 0; heads[subset] = 0; }
470 if (score == best[subset]) { ways[subset] += ways[rest]; heads[subset] |= bit(a); }
471 }
472 }
473 Result result = {.tie = members(heads[field]).count > 1};
474 for (Field remaining = field; remaining;) {
475 int a = earliest(e, heads[remaining]);
476 result.order.items[result.order.count++] = a;
477 remaining &= ~bit(a);
478 }
479 emit(output, "score: %d\noptimal: %d\nranking: %s\n", best[field], ways[field], names(e, result.order, " > ").data);
480 winner(output, e, result);
481 return result;
482}

484typedef struct { int value, owner; } Parcel;
485static void stv(Election *e, Field field, int seats, FILE *output) {
486 Parcel *parcels = calloc((size_t)e->group_count, sizeof *parcels);
487 if (!parcels) exit(EXIT_FAILURE);
488 for (int i = 0; i < e->group_count; ++i)
489 if (effective(e->groups[i].ranking, field).count) parcels[i].value = 100;
490 int quota = tallies(e, field)->nonblank * 100 / (seats + 1) + 1;
491 emit(output, "quota: %d.%02d\n", quota / 100, quota % 100);
492 Field continuing = field;
493 List elected = {0};
494 int history[MAX_CANDIDATES][MAX_CANDIDATES] = {{0}}, round = 0;
495 for (;;) {
496 int scores[MAX_CANDIDATES] = {0}, exhausted = 0;
497 for (int i = 0; i < e->group_count; ++i) {
498 parcels[i].owner = -1;
499 if (!parcels[i].value) continue;
500 List rank = effective(e->groups[i].ranking, continuing);
501 int value = e->groups[i].votes * parcels[i].value;
502 if (rank.count) {
503 parcels[i].owner = rank.items[0];
504 scores[rank.items[0]] += value;
505 } else exhausted += value;
506 }
507 memcpy(history[round], scores, sizeof scores);
508 ++round;
509 emit(output, "round %d: %s; exhausted %d.%02d", round, values(e, continuing, scores, true).data, exhausted / 100, exhausted % 100);
510 List order = ordered(e, continuing, scores, true);
511 if (order.count == seats - elected.count) {
512 for (int i = 0; i < order.count; ++i) elected.items[elected.count++] = order.items[i];
513 emit(output, "; elected %s\n", names(e, order, ", ").data);
514 break;
515 }
516 int a = order.items[0];
517 if (scores[a] >= quota) {
518 int surplus = scores[a] - quota;
519 elected.items[elected.count++] = a;
520 continuing &= ~bit(a);
521 for (int i = 0; i < e->group_count; ++i)
522 if (parcels[i].owner == a) parcels[i].value = parcels[i].value * surplus / scores[a];
523 emit(output, "; elected %s (surplus %d.%02d)\n", e->names[a], surplus / 100, surplus % 100);
524 if (elected.count == seats) break;
525 } else {
526 bool tie = false;
527 a = look_back(e, continuing, history, round, &tie);
528 continuing &= ~bit(a);
529 emit(output, "; excluded %s%s\n", e->names[a], tie_mark(tie));
530 }
531 }
532 emit(output, "elected: %s\n", names(e, elected, ", ").data);
533 free(parcels);
534}

536static Result tally(Election *e, Field field, Method method, FILE *output) {
537 switch (method.kind) {
538 case RUNOFF: return runoff(e, field, output);
539 case IRV: case COOMBS: return elimination(e, field, method.kind, output, false);
540 case BUCKLIN: return bucklin(e, field, output);
541 case BALDWIN: case NANSON: return borda_elimination(e, field, method.kind == NANSON, output);
542 case SCHULZE: return schulze(e, field, output);
543 case RANKEDPAIRS: return ranked_pairs(e, field, output);
544 case KEMENY: return kemeny(e, field, output);
545 case CONDORCET: condorcet_report(e, field, output); return (Result){0};
546 case STV: stv(e, field, method.parameter, output); return (Result){0};
547 default: break;
548 }
549 Tallies *base = tallies(e, field);
550 int scores[MAX_CANDIDATES] = {0};
551 const int *selected = scores;
552 const char *label = "scores";
553 bool descending = true;
554 int condorcet_winner = -1;
555 switch (method.kind) {
556 case PLURALITY: selected = base->first; label = "votes"; break;
557 case BORDA: case BLACK:
558 selected = base->borda;
559 if (method.kind == BLACK) {
560 condorcet_winner = condorcet(e, field, false);
561 emit(output, "condorcet: %s\n", condorcet_winner < 0 ? "NONE" : e->names[condorcet_winner]);
562 }
563 break;
564 case VETO: selected = base->last; label = "vetoes"; descending = false; break;
565 case APPROVAL: {
566 int depth = method.parameter < members(field).count ? method.parameter : members(field).count;
567 selected = base->approval[depth - 1];
568 label = "approvals";
569 break;
570 }
571 case COPELAND: {
572 pairwise(e);
573 Text text = {0};
574 List list = members(field);
575 for (int i = 0; i < list.count; ++i) {
576 int a = list.items[i], wins = 0, draws = 0, losses = 0;
577 for (int j = 0; j < list.count; ++j) if (i != j) {
578 int b = list.items[j];
579 if (e->pairwise[a][b] > e->pairwise[b][a]) ++wins;
580 else if (e->pairwise[a][b] == e->pairwise[b][a]) ++draws;
581 else ++losses;
582 }
583 scores[a] = 2 * wins + draws;
584 append(&text, "%s%s %d.%d (%d-%d-%d)", i ? ", " : "", e->names[a], scores[a] / 2, (scores[a] % 2) * 5, wins, draws, losses);
585 }
586 emit(output, "scores: %s\n", text.data);
587 break;
588 }
589 case MINIMAX:
590 pairwise(e);
591 descending = false;
592 for (int a = 0; a < e->count; ++a) if (field & bit(a))
593 for (int b = 0; b < e->count; ++b) if (a != b && (field & bit(b))) {
594 int opposition = e->pairwise[b][a], support = e->pairwise[a][b], value = 0;
595 if (method.parameter == 2 || opposition > support) value = method.parameter == 1 ? opposition - support : opposition;
596 if (value > scores[a]) scores[a] = value;
597 }
598 break;
599 default: abort();
600 }
601 Result result = scored(e, field, selected, descending);
602 if (condorcet_winner >= 0) {
603 int index = position(result.order, condorcet_winner);
604 for (int i = index; i > 0; --i) result.order.items[i] = result.order.items[i - 1];
605 result.order.items[0] = condorcet_winner;
606 result.tie = false;
607 }
608 if (method.kind != COPELAND) emit(output, "%s: %s\n", label, values(e, field, selected, false).data);
609 winner(output, e, result);
610 return result;
611}

613static int split(char *text, char separator, char *parts[]) {
614 int count = 0;
615 parts[count++] = text;
616 for (char *p = text; *p; ++p) if (*p == separator) { *p = '\0'; parts[count++] = p + 1; }
617 return count;
618}
619static int candidate_index(const Election *e, const char *name) {
620 for (int i = 0; i < e->count; ++i) if (strcmp(e->names[i], name) == 0) return i;
621 return -1;
622}
623static bool checked_names(const Election *e, char *words[], int count, List *result) {
624 *result = (List){0};
625 for (int i = 0; i < count; ++i) {
626 int a = candidate_index(e, words[i]);
627 if (a < 0) { error("unknown candidate %s", words[i]); return false; }
628 if (position(*result, a) >= 0) { error("duplicate candidate %s", words[i]); return false; }
629 result->items[result->count++] = a;
630 }
631 return true;
632}
633static Method parse_method(char *words[]) {
634 Method result = {0};
635 for (int i = 0; i < 18; ++i) if (strcmp(words[0], method_names[i]) == 0) { result.kind = (MethodKind)i; break; }
636 if (result.kind == APPROVAL || result.kind == STV) result.parameter = atoi(words[1]);
637 else if (result.kind == MINIMAX) result.parameter = strcmp(words[1], "WINNING") == 0 ? 0 : strcmp(words[1], "MARGIN") == 0 ? 1 : 2;
638 return result;
639}
640static const char *summary_name(int index) {
641 return summary_methods[index].kind == MINIMAX ? "MINIMAX WINNING" : method_names[summary_methods[index].kind];
642}

644typedef struct { int votes; char notation[91]; } Distribution;
645static int notation_order(const void *left, const void *right) {
646 const Distribution *a = left, *b = right;
647 return strcmp(a->notation, b->notation);
648}
649static int distribution_order(const void *left, const void *right) {
650 const Distribution *a = left, *b = right;
651 return a->votes != b->votes ? (a->votes > b->votes ? -1 : 1) : strcmp(a->notation, b->notation);
652}
653static void ballots_report(Election *e) {
654 Distribution *groups = malloc((size_t)e->group_count * sizeof *groups);
655 if (!groups) exit(EXIT_FAILURE);
656 int blank = 0;
657 for (int i = 0; i < e->group_count; ++i) {
658 List rank = effective(e->groups[i].ranking, e->active);
659 groups[i].votes = e->groups[i].votes;
660 strcpy(groups[i].notation, rank.count ? names(e, rank, ">").data : "-");
661 if (!rank.count) blank += groups[i].votes;
662 }
663 qsort(groups, (size_t)e->group_count, sizeof *groups, notation_order);
664 int distinct = 0;
665 for (int i = 0; i < e->group_count; ++i) {
666 if (distinct > 0 && strcmp(groups[i].notation, groups[distinct - 1].notation) == 0) groups[distinct - 1].votes += groups[i].votes;
667 else groups[distinct++] = groups[i];
668 }
669 qsort(groups, (size_t)distinct, sizeof *groups, distribution_order);
670 printf("total %d, blank %d, distinct %d\n", e->total, blank, distinct);
671 for (int i = 0; i < distinct; ++i) printf("%d %s\n", groups[i].votes, groups[i].notation);
672 free(groups);
673}
674static void set_tiebreak(Election *e, List order) {
675 e->tiebreak = order;
676 for (int i = 0; i < order.count; ++i) e->priority[order.items[i]] = i;
677}
678static void finish_entry(Election *e) {
679 e->entry_ends[e->entry_count++] = e->group_count;
680 invalidate(e);
681}
682static void execute(Election *e, int *election_number, char *line) {
683 char original[201];
684 strcpy(original, line);
685 char *words[12];
686 int count = split(line, ' ', words);
687 const char *command = words[0];
688 if (strcmp(command, "CANDIDATES") == 0) {
689 if (count == 2) { error("too few candidates"); return; }
690 for (int i = 1; i < count; ++i) for (int j = 1; j < i; ++j)
691 if (strcmp(words[i], words[j]) == 0) { error("duplicate candidate %s", words[i]); return; }
692 memset(e, 0, sizeof *e);
693 e->number = ++*election_number;
694 e->count = count - 1;
695 e->active = bit(e->count) - 1;
696 for (int i = 0; i < e->count; ++i) strcpy(e->names[i], words[i + 1]);
697 List order = members(e->active);
698 for (int i = 1; i < order.count; ++i) {
699 int a = order.items[i], j = i;
700 while (j > 0 && strcmp(e->names[a], e->names[order.items[j - 1]]) < 0) { order.items[j] = order.items[j - 1]; --j; }
701 order.items[j] = a;
702 }
703 set_tiebreak(e, order);
704 printf("election %d: %d candidates\n", e->number, e->count);
705 return;
706 }
707 if (!e->number) { error("no election"); return; }
708 if (strcmp(command, "BALLOT") == 0) {
709 List rank = {0};
710 if (strcmp(words[2], "-") != 0) {
711 char *choices[10];
712 int length = split(words[2], '>', choices);
713 if (!checked_names(e, choices, length, &rank)) return;
714 }
715 int votes = atoi(words[1]);
716 e->groups[e->group_count++] = (Ballot){votes, rank};
717 e->total += votes;
718 finish_entry(e);
719 return;
720 }
721 if (strcmp(command, "RANDOM") == 0) {
722 int groups = atoi(words[1]);
723 int64_t state = strtoll(words[2], NULL, 10);
724 for (int i = 0; i < groups; ++i) {
725 List rank = members(e->active);
726 for (int j = rank.count - 1; j > 0; --j) {
727 state = 48271 * state % 2147483647;
728 int other = (int)(state % (j + 1)), temporary = rank.items[j];
729 rank.items[j] = rank.items[other];
730 rank.items[other] = temporary;
731 }
732 state = 48271 * state % 2147483647;
733 rank.count = (int)(state % rank.count) + 1;
734 e->groups[e->group_count++] = (Ballot){1, rank};
735 }
736 e->total += groups;
737 finish_entry(e);
738 return;
739 }
740 if (strcmp(command, "UNDO") == 0) {
741 if (!e->entry_count) { error("nothing to undo"); return; }
742 --e->entry_count;
743 int end = e->entry_count ? e->entry_ends[e->entry_count - 1] : 0;
744 for (int i = end; i < e->group_count; ++i) e->total -= e->groups[i].votes;
745 e->group_count = end;
746 invalidate(e);
747 return;
748 }
749 if (strcmp(command, "WITHDRAW") == 0 || strcmp(command, "RESTORE") == 0) {
750 int a = candidate_index(e, words[1]);
751 if (a < 0) { error("unknown candidate %s", words[1]); return; }
752 bool active = (e->active & bit(a)) != 0;
753 if (strcmp(command, "WITHDRAW") == 0) {
754 if (!active) { error("already withdrawn %s", words[1]); return; }
755 if (members(e->active).count <= 2) { error("too few candidates"); return; }
756 e->active &= ~bit(a);
757 } else {
758 if (active) { error("not withdrawn %s", words[1]); return; }
759 e->active |= bit(a);
760 }
761 return;
762 }
763 if (strcmp(command, "TIEBREAK") == 0) {
764 List order;
765 if (!checked_names(e, words + 1, count - 1, &order)) return;
766 if (order.count != e->count) { error("incomplete tiebreak"); return; }
767 set_tiebreak(e, order);
768 return;
769 }
770 if (strcmp(command, "STATUS") == 0) {
771 List withdrawn = members((bit(e->count) - 1) & ~e->active);
772 printf("[%s]\nelection %d\nactive: %s\nwithdrawn: %s\ntiebreak: %s\nballots: %d in %d entries\n",
773 original, e->number, names(e, members(e->active), ", ").data,
774 withdrawn.count ? names(e, withdrawn, ", ").data : "none", names(e, e->tiebreak, ", ").data, e->total, e->entry_count);
775 return;
776 }
777 if (!e->entry_count) { error("no ballots"); return; }
778 bool ranking = strcmp(command, "RANKING") == 0, spoilers = strcmp(command, "SPOILERS") == 0;
779 Method method = {0};
780 if (ranking || spoilers) {
781 method = parse_method(words + 1);
782 if (method.kind == CONDORCET || method.kind == STV) { error("not rankable"); return; }
783 if (spoilers && members(e->active).count < 3) { error("too few candidates"); return; }
784 }
785 int compared[2] = {0};
786 if (strcmp(command, "COMPARE") == 0) {
787 for (int i = 0; i < 2; ++i) {
788 compared[i] = candidate_index(e, words[i + 1]);
789 if (compared[i] < 0) { error("unknown candidate %s", words[i + 1]); return; }
790 if (!(e->active & bit(compared[i]))) { error("withdrawn candidate %s", words[i + 1]); return; }
791 }
792 if (compared[0] == compared[1]) { error("same candidate"); return; }
793 }
794 if (strcmp(command, "STV") == 0 && atoi(words[1]) >= members(e->active).count) { error("too many seats"); return; }
795 printf("[%s]\n", original);
796 if (ranking || spoilers) {
797 Result result = tally(e, e->active, method, NULL);
798 if (ranking) puts(names(e, result.order, " > ").data);
799 else {
800 int win = result.order.items[0];
801 printf("winner: %s\n", e->names[win]);
802 for (int a = 0; a < e->count; ++a) if (a != win && (e->active & bit(a))) {
803 int other = tally(e, e->active & ~bit(a), method, NULL).order.items[0];
804 printf("without %s: %s%s\n", e->names[a], e->names[other], other != win ? " (changed)" : "");
805 }
806 }
807 } else if (strcmp(command, "BALLOTS") == 0) ballots_report(e);
808 else if (strcmp(command, "PROFILE") == 0) {
809 Tallies *scores = tallies(e, e->active);
810 int size = members(e->active).count;
811 for (int a = 0; a < e->count; ++a) if (e->active & bit(a)) {
812 printf("%s:", e->names[a]);
813 for (int j = 0; j < size; ++j) printf(" %d", scores->approval[j][a] - (j ? scores->approval[j - 1][a] : 0));
814 printf(" unranked %d\n", e->total - scores->approval[size - 1][a]);
815 }
816 } else if (strcmp(command, "PAIRWISE") == 0) {
817 pairwise(e);
818 pair_lines(stdout, e, e->active, e->pairwise);
819 } else if (strcmp(command, "EXHAUSTED") == 0) elimination(e, e->active, IRV, stdout, true);
820 else if (strcmp(command, "SUMMARY") == 0 || strcmp(command, "COMPARE") == 0) {
821 int wins[MAX_CANDIDATES] = {0};
822 Text ahead[2] = {{0}, {0}};
823 bool compare = strcmp(command, "COMPARE") == 0;
824 for (int i = 0; i < 15; ++i) {
825 Result result = tally(e, e->active, summary_methods[i], NULL);
826 if (compare) {
827 int side = position(result.order, compared[0]) < position(result.order, compared[1]) ? 0 : 1;
828 append(&ahead[side], "%s%s", ahead[side].length ? ", " : "", summary_name(i));
829 } else {
830 int win = result.order.items[0];
831 ++wins[win];
832 printf("%s: %s\n", summary_name(i), e->names[win]);
833 }
834 }
835 if (compare) {
836 pairwise(e);
837 int a = compared[0], b = compared[1];
838 printf("%s %d, %s %d, neither %d\n%s ahead: %s\n%s ahead: %s\n", e->names[a], e->pairwise[a][b], e->names[b], e->pairwise[b][a],
839 e->total - e->pairwise[a][b] - e->pairwise[b][a], e->names[a], ahead[0].length ? ahead[0].data : "none",
840 e->names[b], ahead[1].length ? ahead[1].data : "none");
841 } else printf("wins: %s\n", values(e, e->active, wins, false).data);
842 } else tally(e, e->active, parse_method(words), stdout);
843}

846 static Election election;
847 int election_number = 0;
848 char line[256];
849 if (!fgets(line, sizeof line, stdin)) return 0;

852 if (!fgets(line, sizeof line, stdin)) break;
853 line[strcspn(line, "\r\n")] = '\0';
854 execute(&election, &election_number, line);
855 }
856 return 0;
