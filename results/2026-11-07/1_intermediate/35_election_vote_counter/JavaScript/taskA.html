4class CommandError extends Error {}
5const fail = reason => { throw new CommandError(reason); };
6const sum = values => values.reduce((total, value) => total + value, 0);
7const ascii = (first, second) => first < second ? -1 : first > second ? 1 : 0;
8const money = value => `${Math.floor(value / 100)}.${String(value % 100).padStart(2, '0')}`;
9const summaryMethods = ['PLURALITY', 'RUNOFF', 'IRV', 'COOMBS', 'BUCKLIN', 'BORDA', 'BALDWIN', 'NANSON', 'BLACK', 'VETO', 'COPELAND', 'MINIMAX WINNING', 'SCHULZE', 'RANKEDPAIRS', 'KEMENY'];

11class Election {
12 constructor(names, number) {
13 this.names = names;
14 this.number = number;
15 this.withdrawn = new Set();
16 this.entries = [];
17 this.cache = new Map();
18 this.setPriority(names.map((_, index) => index).sort((a, b) => ascii(names[a], names[b])));
19 }
20 zeros() { return this.names.map(() => 0); }
21 matrix() { return this.names.map(() => this.zeros()); }
22 active() { return this.names.map((_, index) => index).filter(index => !this.withdrawn.has(index)); }
23 candidate(name) {
24 const index = this.names.indexOf(name);
25 if (index < 0) fail(`unknown candidate ${name}`);
26 return index;
27 }
28 nameList(names) {
29 const found = [];
30 for (const name of names) {
31 const candidate = this.candidate(name);
32 if (found.includes(candidate)) fail(`duplicate candidate ${name}`);
33 found.push(candidate);
34 }
35 return found;
36 }
37 setPriority(order) {
38 this.priority = order;
39 this.tieRank = this.zeros();
40 order.forEach((candidate, index) => { this.tieRank[candidate] = index; });
41 }
42 list(field, separator = ', ') { return field.map(candidate => this.names[candidate]).join(separator); }
43 values(field, values, format = String) { return field.map(candidate => `${this.names[candidate]} ${format(values[candidate])}`).join(', '); }
44 ordered(field, values, ascending = false) {
45 const direction = ascending ? 1 : -1;
46 return field.slice().sort((a, b) => direction * (values[a] - values[b]) || this.tieRank[a] - this.tieRank[b]);
47 }
48 preferred(field) { return field.slice().sort((a, b) => this.tieRank[a] - this.tieRank[b]); }
49 aggregate(field) {
50 const mask = field.reduce((bits, candidate) => bits | (1 << candidate), 0);
51 if (this.cache.has(mask)) return this.cache.get(mask);
52 const combined = new Map();
53 for (const entry of this.entries) for (const group of entry) {
54 const ranking = group.ranking.filter(candidate => mask & (1 << candidate));
55 const key = ranking.join(',');
56 if (combined.has(key)) combined.get(key).votes += group.votes;
57 else combined.set(key, { votes: group.votes, ranking });
58 }
59 const groups = [...combined.values()], first = this.zeros(), last = this.zeros(), borda = this.zeros();
60 const positions = this.names.map(() => field.map(() => 0)), pairwise = this.matrix();
61 let total = 0, blank = 0;
62 for (const { votes, ranking } of groups) {
63 total += votes;
64 if (ranking.length === 0) blank += votes;
65 else first[ranking[0]] += votes;
66 const ranks = this.names.map(() => field.length);
67 ranking.forEach((candidate, index) => {
68 ranks[candidate] = index;
69 positions[candidate][index] += votes;
70 borda[candidate] += votes * (field.length - index - 1);
71 });
72 for (const candidate of field) {
73 if (ranking.length < field.length ? ranks[candidate] === field.length : candidate === ranking.at(-1)) last[candidate] += votes;
74 for (const other of field) if (ranks[candidate] < ranks[other]) pairwise[candidate][other] += votes;
75 }
76 }
77 const aggregate = { groups, first, last, borda, positions, pairwise, total, blank };
78 this.cache.set(mask, aggregate);
79 return aggregate;
80 }
81}

83function eliminate(election, field, values, history) {
84 const minimum = Math.min(...field.map(candidate => values[candidate]));
85 let candidates = field.filter(candidate => values[candidate] === minimum);
86 const tie = candidates.length > 1;
87 for (let round = history.length - 1; round >= 0 && candidates.length > 1; --round) {
88 const oldMinimum = Math.min(...candidates.map(candidate => history[round][candidate]));
89 candidates = candidates.filter(candidate => history[round][candidate] === oldMinimum);
90 }
91 return { candidate: election.preferred(candidates).at(-1), tie };
92}
93function winnerLine(election, standings, tie) { return `winner: ${election.names[standings[0]]}${tie ? ' (tie)' : ''}`; }
94function rankedResult(election, standings, tie, lines) { return { standings, tie, lines: [...lines, winnerLine(election, standings, tie)] }; }
95function scoredResult(election, field, values, label, ascending = false) {
96 const standings = election.ordered(field, values, ascending);
97 return rankedResult(election, standings, values[standings[0]] === values[standings[1]], [`${label}: ${election.values(field, values)}`]);
98}
99function pairLines(election, field, matrix) {
100 const lines = [];
101 for (let i = 0; i < field.length; ++i) for (let j = i + 1; j < field.length; ++j) {
102 const first = field[i], second = field[j], a = matrix[first][second], b = matrix[second][first];
103 const result = a > b ? election.names[first] : b > a ? election.names[second] : 'tie';
104 lines.push(`${election.names[first]} vs ${election.names[second]}: ${a}-${b} (${result})`);
105 }
106 return lines;
107}
108function runoff(election, field) {
109 const first = election.aggregate(field).first, order = election.ordered(field, first);
110 const lines = [`round 1: ${election.values(field, first)}`];
111 if (2 * first[order[0]] > sum(field.map(candidate => first[candidate]))) return rankedResult(election, order, false, lines);
112 const finalists = order.slice(0, 2).sort((a, b) => a - b), second = election.aggregate(finalists).first;
113 const advancementTie = order.length >= 3 && first[order[1]] === first[order[2]];
114 lines.push(`finalists: ${election.list(finalists)}${advancementTie ? ' (tie)' : ''}`, `round 2: ${election.values(finalists, second)}`);
115 const winners = finalists.slice().sort((a, b) => second[b] - second[a] || first[b] - first[a] || election.tieRank[a] - election.tieRank[b]);
116 return rankedResult(election, [...winners, ...order.slice(2)], second[winners[0]] === second[winners[1]], lines);
117}
118function eliminationRounds(election, field, coombs) {
119 let remaining = field.slice();
120 const eliminated = [], history = [], lines = [], exhausted = [];
121 for (let round = 1; ; ++round) {
122 const aggregate = election.aggregate(remaining), values = aggregate.first;
123 const ordered = election.ordered(remaining, values), active = sum(remaining.map(candidate => values[candidate]));
124 let line = `round ${round}: ${election.values(remaining, values)}`;
125 exhausted.push(`round ${round}: active ${active}, exhausted ${aggregate.total - active}`);
126 if (remaining.length === 1 || 2 * values[ordered[0]] > active) {
127 const result = rankedResult(election, [...ordered, ...eliminated.reverse()], false, [...lines, line]);
128 return { ...result, exhausted };
129 }
130 let chosen;
131 if (coombs) {
132 const last = aggregate.last, maximum = Math.max(...remaining.map(candidate => last[candidate]));
133 const tied = remaining.filter(candidate => last[candidate] === maximum);
134 const candidate = tied.slice().sort((a, b) => values[a] - values[b] || election.tieRank[b] - election.tieRank[a])[0];
135 chosen = { candidate, tie: tied.length > 1 };
136 line += `; last ${election.values(remaining, last)}`;
137 } else chosen = eliminate(election, remaining, values, history);
138 lines.push(`${line}; eliminated ${election.names[chosen.candidate]}${chosen.tie ? ' (tie)' : ''}`);
139 history.push(values);
140 eliminated.push(chosen.candidate);
141 remaining = remaining.filter(candidate => candidate !== chosen.candidate);
142 }
143}
144function bucklin(election, field) {
145 const aggregate = election.aggregate(field), majority = Math.floor((aggregate.total - aggregate.blank) / 2) + 1;
146 const history = [], lines = [`majority: ${majority}`];
147 for (let depth = 1; depth <= field.length; ++depth) {
148 const values = aggregate.positions.map(positions => sum(positions.slice(0, depth)));
149 history.push(values);
150 lines.push(`round ${depth}: ${election.values(field, values)}`);
151 if (field.some(candidate => values[candidate] >= majority) || depth === field.length) break;
152 }
153 const standings = field.slice().sort((a, b) => {
154 for (let round = history.length - 1; round >= 0; --round) if (history[round][a] !== history[round][b]) return history[round][b] - history[round][a];
155 return election.tieRank[a] - election.tieRank[b];
156 });
157 const final = history.at(-1);
158 return rankedResult(election, standings, final[standings[0]] === final[standings[1]], lines);
159}
160function bordaRounds(election, field, nanson) {
161 let remaining = field.slice();
162 const eliminated = [], history = [], lines = [];
163 for (let round = 1; remaining.length >= 2; ++round) {
164 const values = election.aggregate(remaining).borda;
165 let line = `round ${round}: ${election.values(remaining, values)}`;
166 if (nanson) {
167 const total = sum(remaining.map(candidate => values[candidate]));
168 const removed = remaining.filter(candidate => values[candidate] * remaining.length < total);
169 if (removed.length === 0) { lines.push(line); break; }
170 line += `; eliminated ${election.list(removed)}`;
171 eliminated.unshift(election.ordered(removed, values));
172 remaining = remaining.filter(candidate => !removed.includes(candidate));
173 } else {
174 const chosen = eliminate(election, remaining, values, history);
175 line += `; eliminated ${election.names[chosen.candidate]}${chosen.tie ? ' (tie)' : ''}`;
176 eliminated.unshift([chosen.candidate]);
177 remaining = remaining.filter(candidate => candidate !== chosen.candidate);
178 }
179 history.push(values);
180 lines.push(line);
181 }
182 return rankedResult(election, [...election.preferred(remaining), ...eliminated.flat()], remaining.length > 1, lines);
183}
184function condorcetWinner(field, pairwise, loser = false) {
185 return field.find(candidate => field.every(other => candidate === other || (loser ? pairwise[candidate][other] < pairwise[other][candidate] : pairwise[candidate][other] > pairwise[other][candidate])));
186}
187function closure(field, edges) {
188 const reach = edges.map(row => row.slice());
189 for (const middle of field) for (const first of field) for (const second of field) {
190 if (reach[first][middle] && reach[middle][second]) reach[first][second] = true;
191 }
192 return reach;
193}
194function condorcet(election, field, pairwise) {
195 const winner = condorcetWinner(field, pairwise), loser = condorcetWinner(field, pairwise, true);
196 const weak = election.names.map(() => election.names.map(() => false)), strong = weak.map(row => row.slice());
197 for (const first of field) for (const second of field) {
198 weak[first][second] = first === second || pairwise[first][second] >= pairwise[second][first];
199 strong[first][second] = first !== second && pairwise[first][second] > pairwise[second][first];
200 }
201 const weakReach = closure(field, weak), strongReach = closure(field, strong);
202 const smith = field.filter(candidate => field.every(other => weakReach[candidate][other]));
203 const schwartz = field.filter(candidate => field.every(other => !strongReach[other][candidate] || strongReach[candidate][other]));
204 return { standings: [], tie: false, lines: [`winner: ${winner === undefined ? 'NONE' : election.names[winner]}`, `loser: ${loser === undefined ? 'NONE' : election.names[loser]}`, `smith: ${election.list(smith)}`, `schwartz: ${election.list(schwartz)}`] };
205}
206function topological(election, field, beats) {
207 let remaining = field.slice(), tie = false;
208 const standings = [];
209 while (remaining.length > 0) {
210 const eligible = remaining.filter(candidate => remaining.every(other => candidate === other || !beats(other, candidate)));
211 if (standings.length === 0) tie = eligible.length > 1;
212 const next = election.preferred(eligible)[0];
213 standings.push(next);
214 remaining = remaining.filter(candidate => candidate !== next);
215 }
216 return { standings, tie };
217}
218function schulze(election, field, pairwise) {
219 const strengths = election.matrix();
220 for (const first of field) for (const second of field) {
221 if (first !== second && pairwise[first][second] > pairwise[second][first]) strengths[first][second] = pairwise[first][second];
222 }
223 for (const middle of field) for (const first of field) if (first !== middle) {
224 for (const second of field) if (second !== middle && second !== first) {
225 strengths[first][second] = Math.max(strengths[first][second], Math.min(strengths[first][middle], strengths[middle][second]));
226 }
227 }
228 const { standings, tie } = topological(election, field, (first, second) => strengths[first][second] > strengths[second][first]);
229 return rankedResult(election, standings, tie, [...pairLines(election, field, strengths), `ranking: ${election.list(standings, ' > ')}`]);
230}
231function rankedPairs(election, field, pairwise) {
232 const pairs = [], edges = election.names.map(() => election.names.map(() => false)), lines = [];
233 for (const first of field) for (const second of field) if (pairwise[first][second] > pairwise[second][first]) pairs.push([first, second]);
234 pairs.sort(([a, b], [c, d]) => pairwise[c][d] - pairwise[a][b] || pairwise[b][a] - pairwise[d][c] || election.tieRank[a] - election.tieRank[c] || election.tieRank[b] - election.tieRank[d]);
235 for (const [first, second] of pairs) {
236 const lock = !closure(field, edges)[second][first];
237 if (lock) edges[first][second] = true;
238 lines.push(`${lock ? 'lock' : 'skip'} ${election.names[first]}>${election.names[second]} ${pairwise[first][second]}-${pairwise[second][first]}`);
239 }
240 const { standings, tie } = topological(election, field, (first, second) => edges[first][second]);
241 return rankedResult(election, standings, tie, lines);
242}
243function kemeny(election, field, pairwise) {
244 const memo = new Map([[0, { score: 0, count: 1, first: [] }]]);
245 function solve(mask) {
246 if (memo.has(mask)) return memo.get(mask);
247 let score = -1, count = 0, first = [];
248 for (const candidate of field) if (mask & (1 << candidate)) {
249 const rest = mask ^ (1 << candidate), suffix = solve(rest);
250 const value = suffix.score + sum(field.filter(other => rest & (1 << other)).map(other => pairwise[candidate][other]));
251 if (value > score) { score = value; count = suffix.count; first = [candidate]; }
252 else if (value === score) { count += suffix.count; first.push(candidate); }
253 }
254 const result = { score, count, first };
255 memo.set(mask, result);
256 return result;
257 }
258 let mask = field.reduce((bits, candidate) => bits | (1 << candidate), 0);
259 const optimal = solve(mask), standings = [];
260 while (mask !== 0) {
261 const candidate = election.preferred(solve(mask).first)[0];
262 standings.push(candidate);
263 mask ^= 1 << candidate;
264 }
265 return rankedResult(election, standings, optimal.first.length > 1, [`score: ${optimal.score}`, `optimal: ${optimal.count}`, `ranking: ${election.list(standings, ' > ')}`]);
266}
267function stv(election, field, seats) {
268 if (seats >= field.length) fail('too many seats');
269 const aggregate = election.aggregate(field), groups = aggregate.groups.filter(group => group.ranking.length > 0);
270 const quota = Math.floor(100 * (aggregate.total - aggregate.blank) / (seats + 1)) + 1;
271 const perVote = groups.map(() => 100), elected = [], history = [], lines = [`quota: ${money(quota)}`];
272 let continuing = field.slice();
273 for (let round = 1; ; ++round) {
274 const values = election.zeros(), current = [];
275 let exhausted = 0;
276 groups.forEach((group, index) => {
277 const candidate = group.ranking.find(candidate => continuing.includes(candidate));
278 current.push(candidate);
279 if (candidate === undefined) exhausted += group.votes * perVote[index];
280 else values[candidate] += group.votes * perVote[index];
281 });
282 let line = `round ${round}: ${election.values(continuing, values, money)}; exhausted ${money(exhausted)}`;
283 const ordered = election.ordered(continuing, values);
284 if (continuing.length === seats - elected.length) {
285 elected.push(...ordered);
286 lines.push(`${line}; elected ${election.list(ordered)}`);
287 break;
288 }
289 let removed;
290 if (values[ordered[0]] >= quota) {
291 removed = ordered[0];
292 elected.push(removed);
293 const total = values[removed], surplus = total - quota;
294 current.forEach((candidate, index) => { if (candidate === removed) perVote[index] = Math.floor(perVote[index] * surplus / total); });
295 line += `; elected ${election.names[removed]} (surplus ${money(surplus)})`;
296 } else {
297 const chosen = eliminate(election, continuing, values, history);
298 removed = chosen.candidate;
299 line += `; excluded ${election.names[removed]}${chosen.tie ? ' (tie)' : ''}`;
300 }
301 lines.push(line);
302 if (elected.length === seats) break;
303 history.push(values);
304 continuing = continuing.filter(candidate => candidate !== removed);
305 }
306 return { standings: [], tie: false, lines: [...lines, `elected: ${election.list(elected)}`] };
307}
308function tally(election, method, field) {
309 const [kind, argument] = method.split(' '), aggregate = election.aggregate(field), pairwise = aggregate.pairwise;
310 switch (kind) {
311 case 'PLURALITY': return scoredResult(election, field, aggregate.first, 'votes');
312 case 'RUNOFF': return runoff(election, field);
313 case 'IRV': return eliminationRounds(election, field, false);
314 case 'COOMBS': return eliminationRounds(election, field, true);
315 case 'BUCKLIN': return bucklin(election, field);
316 case 'BORDA': return scoredResult(election, field, aggregate.borda, 'scores');
317 case 'BALDWIN': return bordaRounds(election, field, false);
318 case 'NANSON': return bordaRounds(election, field, true);
319 case 'BLACK': {
320 const winner = condorcetWinner(field, pairwise), ordered = election.ordered(field, aggregate.borda);
321 const standings = winner === undefined ? ordered : [winner, ...ordered.filter(candidate => candidate !== winner)];
322 const tie = winner === undefined && aggregate.borda[ordered[0]] === aggregate.borda[ordered[1]];
323 return rankedResult(election, standings, tie, [`condorcet: ${winner === undefined ? 'NONE' : election.names[winner]}`, `scores: ${election.values(field, aggregate.borda)}`]);
324 }
325 case 'VETO': return scoredResult(election, field, aggregate.last, 'vetoes', true);
326 case 'CONDORCET': return condorcet(election, field, pairwise);
327 case 'COPELAND': {
328 const scores = election.zeros(), parts = [];
329 for (const candidate of field) {
330 let wins = 0, draws = 0, losses = 0;
331 for (const other of field) if (other !== candidate) {
332 if (pairwise[candidate][other] > pairwise[other][candidate]) ++wins;
333 else if (pairwise[candidate][other] === pairwise[other][candidate]) ++draws;
334 else ++losses;
335 }
336 scores[candidate] = 2 * wins + draws;
337 parts.push(`${election.names[candidate]} ${Math.floor(scores[candidate] / 2)}.${scores[candidate] % 2 ? '5' : '0'} (${wins}-${draws}-${losses})`);
338 }
339 const standings = election.ordered(field, scores);
340 return rankedResult(election, standings, scores[standings[0]] === scores[standings[1]], [`scores: ${parts.join(', ')}`]);
341 }
342 case 'MINIMAX': {
343 const scores = election.zeros();
344 for (const candidate of field) for (const other of field) if (candidate !== other) {
345 const opposition = pairwise[other][candidate], support = pairwise[candidate][other];
346 if (argument === 'OPPOSITION') scores[candidate] = Math.max(scores[candidate], opposition);
347 else if (opposition > support) scores[candidate] = Math.max(scores[candidate], argument === 'MARGIN' ? opposition - support : opposition);
348 }
349 return scoredResult(election, field, scores, 'scores', true);
350 }
351 case 'SCHULZE': return schulze(election, field, pairwise);
352 case 'RANKEDPAIRS': return rankedPairs(election, field, pairwise);
353 case 'KEMENY': return kemeny(election, field, pairwise);
354 case 'APPROVAL': return scoredResult(election, field, aggregate.positions.map(positions => sum(positions.slice(0, Number(argument)))), 'approvals');
355 case 'STV': return stv(election, field, Number(argument));
356 default: throw new Error(`Unknown method ${method}`);
357 }

359function reports(election, command, args, field) {
360 const aggregate = election.aggregate(field);
361 switch (command) {
362 case 'STATUS': {
363 const withdrawn = election.names.map((_, index) => index).filter(candidate => election.withdrawn.has(candidate));
364 return [`election ${election.number}`, `active: ${election.list(field)}`, `withdrawn: ${withdrawn.length ? election.list(withdrawn) : 'none'}`, `tiebreak: ${election.list(election.priority)}`, `ballots: ${aggregate.total} in ${election.entries.length} entries`];
365 }
366 case 'BALLOTS': {
367 const groups = aggregate.groups.map(group => ({ votes: group.votes, notation: group.ranking.length ? election.list(group.ranking, '>') : '-' }));
368 groups.sort((a, b) => b.votes - a.votes || ascii(a.notation, b.notation));
369 return [`total ${aggregate.total}, blank ${aggregate.blank}, distinct ${groups.length}`, ...groups.map(group => `${group.votes} ${group.notation}`)];
370 }
371 case 'PROFILE': return field.map(candidate => `${election.names[candidate]}: ${aggregate.positions[candidate].join(' ')} unranked ${aggregate.total - sum(aggregate.positions[candidate])}`);
372 case 'PAIRWISE': return pairLines(election, field, aggregate.pairwise);
373 case 'EXHAUSTED': return eliminationRounds(election, field, false).exhausted;
374 case 'RANKING':
375 case 'SPOILERS': {
376 if (args[0] === 'CONDORCET' || args[0] === 'STV') fail('not rankable');
377 if (command === 'SPOILERS' && field.length < 3) fail('too few candidates');
378 const method = args.join(' '), standings = tally(election, method, field).standings;
379 if (command === 'RANKING') return [election.list(standings, ' > ')];
380 const winner = standings[0], lines = [`winner: ${election.names[winner]}`];
381 for (const excluded of field) if (excluded !== winner) {
382 const other = tally(election, method, field.filter(candidate => candidate !== excluded)).standings[0];
383 lines.push(`without ${election.names[excluded]}: ${election.names[other]}${other !== winner ? ' (changed)' : ''}`);
384 }
385 return lines;
386 }
387 case 'SUMMARY': {
388 const wins = election.zeros(), lines = [];
389 for (const method of summaryMethods) {
390 const winner = tally(election, method, field).standings[0];
391 ++wins[winner];
392 lines.push(`${method}: ${election.names[winner]}`);
393 }
394 return [...lines, `wins: ${election.values(field, wins)}`];
395 }
396 case 'COMPARE': {
397 const candidates = [];
398 for (const name of args) {
399 const candidate = election.candidate(name);
400 if (election.withdrawn.has(candidate)) fail(`withdrawn candidate ${name}`);
401 candidates.push(candidate);
402 }
403 const [first, second] = candidates;
404 if (first === second) fail('same candidate');
405 const a = aggregate.pairwise[first][second], b = aggregate.pairwise[second][first], ahead = [[], []];
406 for (const method of summaryMethods) {
407 const standings = tally(election, method, field).standings;
408 ahead[standings.indexOf(first) < standings.indexOf(second) ? 0 : 1].push(method);
409 }
410 return [`${args[0]} ${a}, ${args[1]} ${b}, neither ${aggregate.total - a - b}`, ...args.map((name, index) => `${name} ahead: ${ahead[index].length ? ahead[index].join(', ') : 'none'}`)];
411 }
412 default: return tally(election, [command, ...args].join(' '), field).lines;
413 }

415class VoteCounter {
416 constructor() { this.election = null; this.count = 0; }
417 execute(line) {
418 const [command, ...args] = line.split(' ');
419 if (command === 'CANDIDATES') {
420 if (args.length < 2) fail('too few candidates');
421 const seen = new Set();
422 for (const name of args) {
423 if (seen.has(name)) fail(`duplicate candidate ${name}`);
424 seen.add(name);
425 }
426 this.election = new Election(args, ++this.count);
427 return [`election ${this.count}: ${args.length} candidates`];
428 }
429 const election = this.election;
430 if (election === null) fail('no election');
431 const field = election.active();
432 switch (command) {
433 case 'BALLOT': {
434 const ranking = args[1] === '-' ? [] : election.nameList(args[1].split('>'));
435 election.entries.push([{ votes: Number(args[0]), ranking }]);
436 election.cache.clear();
437 return [];
438 }
439 case 'RANDOM': {
440 let state = Number(args[1]);
441 const draw = () => { state = 48271 * state % 2147483647; return state; };
442 const groups = [];
443 for (let group = 0; group < Number(args[0]); ++group) {
444 const ranking = field.slice();
445 for (let index = ranking.length - 1; index >= 1; --index) {
446 const other = draw() % (index + 1);
447 [ranking[index], ranking[other]] = [ranking[other], ranking[index]];
448 }
449 groups.push({ votes: 1, ranking: ranking.slice(0, draw() % ranking.length + 1) });
450 }
451 election.entries.push(groups);
452 election.cache.clear();
453 return [];
454 }
455 case 'UNDO':
456 if (election.entries.length === 0) fail('nothing to undo');
457 election.entries.pop();
458 election.cache.clear();
459 return [];
460 case 'WITHDRAW':
461 case 'RESTORE': {
462 const candidate = election.candidate(args[0]);
463 if (command === 'WITHDRAW') {
464 if (election.withdrawn.has(candidate)) fail(`already withdrawn ${args[0]}`);
465 if (field.length <= 2) fail('too few candidates');
466 election.withdrawn.add(candidate);
467 } else {
468 if (!election.withdrawn.has(candidate)) fail(`not withdrawn ${args[0]}`);
469 election.withdrawn.delete(candidate);
470 }
471 return [];
472 }
473 case 'TIEBREAK': {
474 const order = election.nameList(args);
475 if (order.length !== election.names.length) fail('incomplete tiebreak');
476 election.setPriority(order);
477 return [];
478 }
479 default:
480 if (command !== 'STATUS' && election.entries.length === 0) fail('no ballots');
481 return [`[${line}]`, ...reports(election, command, args, field)];
482 }
483 }
484}
