5const LIMIT: usize = 7;
6type Field = u8;
7type Row = [i32; LIMIT];
8type Matrix = [Row; LIMIT];
9type Relation = [[bool; LIMIT]; LIMIT];
10
11struct Ballot {
12 votes: i32,
13 ranking: Vec<usize>,
14}
15
16#[derive(Clone, Copy, Default)]
17struct Tallies {
18 first: Row,
19 last: Row,
20 borda: Row,
21 approval: Matrix,
22 nonblank: i32,
23}

25#[derive(Clone, Copy, PartialEq, Eq)]
26enum MinimaxKind { Winning, Margin, Opposition }
27
28#[derive(Clone, Copy, PartialEq, Eq)]
29enum Method {
30 Plurality, Runoff, Irv, Coombs, Bucklin, Borda, Baldwin, Nanson, Black, Veto,
31 Condorcet, Copeland, Minimax(MinimaxKind), Schulze, RankedPairs, Kemeny,
32 Approval(usize), Stv(usize),
33}
34
35impl Method {
36 fn parse(words: &[&str]) -> Self {
37 match words[0] {
38 "PLURALITY" => Self::Plurality,
39 "RUNOFF" => Self::Runoff,
40 "IRV" => Self::Irv,
41 "COOMBS" => Self::Coombs,
42 "BUCKLIN" => Self::Bucklin,
43 "BORDA" => Self::Borda,
44 "BALDWIN" => Self::Baldwin,
45 "NANSON" => Self::Nanson,
46 "BLACK" => Self::Black,
47 "VETO" => Self::Veto,
48 "CONDORCET" => Self::Condorcet,
49 "COPELAND" => Self::Copeland,
50 "MINIMAX" => Self::Minimax(match words[1] {
51 "WINNING" => MinimaxKind::Winning,
52 "MARGIN" => MinimaxKind::Margin,
53 "OPPOSITION" => MinimaxKind::Opposition,
54 _ => unreachable!(),
55 }),
56 "SCHULZE" => Self::Schulze,
57 "RANKEDPAIRS" => Self::RankedPairs,
58 "KEMENY" => Self::Kemeny,
59 "APPROVAL" => Self::Approval(words[1].parse().unwrap()),
60 "STV" => Self::Stv(words[1].parse().unwrap()),
61 _ => unreachable!(),
62 }
63 }
64
65 fn rankable(self) -> bool { !matches!(self, Self::Condorcet | Self::Stv(_)) }
66}

103struct Election {
104 number: usize,
105 names: Vec<String>,
106 active: Field,
107 tiebreak: Vec<usize>,
108 priority: [usize; LIMIT],

116fn bit(a: usize) -> Field { 1 << a }
117fn members(field: Field) -> Vec<usize> { (0..LIMIT).filter(|&a| field & bit(a) != 0).collect() }
118fn mask(list: &[usize]) -> Field { list.iter().fold(0, |field, &a| field | bit(a)) }
119fn effective(list: &[usize], field: Field) -> Vec<usize> {
120 list.iter().copied().filter(|&a| field & bit(a) != 0).collect()
121}
122fn tie_mark(tie: bool) -> &'static str { if tie { " (tie)" } else { "" } }
123fn money(value: i32) -> String { format!("{}.{:02}", value / 100, value % 100) }
124fn sum(field: Field, scores: &Row) -> i32 { members(field).iter().map(|&a| scores[a]).sum() }
125fn majority(field: Field, scores: &Row) -> bool {
126 let total = sum(field, scores);
127 members(field).iter().any(|&a| 2 * scores[a] > total)
128}
129fn closure(field: Field, relation: &mut Relation) {
130 let list = members(field);
131 for &k in &list { for &a in &list { for &b in &list {
132 let reachable = relation[a][b] || (relation[a][k] && relation[k][b]);
133 relation[a][b] = reachable;
134 } } }
135}

137impl Election {
138 fn new(number: usize, names: Vec<String>) -> Self {
139 let active = bit(names.len()) - 1;
140 let mut tiebreak = members(active);
141 tiebreak.sort_by(|&a, &b| names[a].cmp(&names[b]));
142 let mut priority = [0; LIMIT];
143 for (i, &a) in tiebreak.iter().enumerate() { priority[a] = i; }
144 Self {
145 number, names, active, tiebreak, priority, groups: Vec::new(), entry_ends: Vec::new(),
146 total: 0, cache: [None; 1 << LIMIT], pairs: None,
147 }
148 }
149
150 fn candidate(&self, name: &str) -> Result<usize, String> {
151 self.names.iter().position(|item| item == name).ok_or_else(|| format!("unknown candidate {name}"))
152 }
153
154 fn checked_names(&self, names: &[&str]) -> Result<Vec<usize>, String> {
155 let mut result = Vec::new();
156 for &name in names {
157 let a = self.candidate(name)?;
158 if result.contains(&a) { return Err(format!("duplicate candidate {name}")); }
159 result.push(a);
160 }
161 Ok(result)
162 }
163
164 fn names_text(&self, list: &[usize], separator: &str) -> String {
165 list.iter().map(|&a| self.names[a].as_str()).collect::<Vec<_>>().join(separator)
166 }
167
168 fn values(&self, field: Field, scores: &Row, cents: bool) -> String {
169 members(field).iter().map(|&a| {
170 format!("{} {}", self.names[a], if cents { money(scores[a]) } else { scores[a].to_string() })
171 }).collect::<Vec<_>>().join(", ")
172 }
173
174 fn ordered(&self, field: Field, scores: &Row, descending: bool) -> Vec<usize> {
175 let mut order = members(field);
176 order.sort_by(|&a, &b| {
177 let comparison = if descending { scores[b].cmp(&scores[a]) } else { scores[a].cmp(&scores[b]) };
178 comparison.then_with(|| self.priority[a].cmp(&self.priority[b]))
179 });
180 order
181 }
182
183 fn finish(&self, order: Vec<usize>, tie: bool, mut lines: Vec<String>) -> Outcome {
184 lines.push(format!("winner: {}{}", self.names[order[0]], tie_mark(tie)));
185 Outcome { order, lines }
186 }

188 fn invalidate(&mut self) {
189 self.cache.fill(None);
190 self.pairs = None;
191 }

193 fn finish_entry(&mut self) {
194 self.entry_ends.push(self.groups.len());
195 self.invalidate();
196 }

198 fn tallies(&mut self, field: Field) -> Tallies {
199 if let Some(result) = self.cache[usize::from(field)] { return result; }
200 let mut result = Tallies::default();
201 let candidates = members(field);
202 let m = candidates.len();
203 for group in &self.groups {
204 let ranking = effective(&group.ranking, field);
205 let ranked = mask(&ranking);
206 if let Some(&first) = ranking.first() {
207 result.first[first] += group.votes;
208 result.nonblank += group.votes;
209 }
210 if ranking.len() == m { result.last[ranking[m - 1]] += group.votes; }
211 else {
212 for &a in &candidates { if ranked & bit(a) == 0 { result.last[a] += group.votes; } }
213 }
214 for (j, &a) in ranking.iter().enumerate() {
215 result.borda[a] += group.votes * (m - j - 1) as i32;
216 for depth in j..m { result.approval[depth][a] += group.votes; }
217 }
218 }
219 self.cache[usize::from(field)] = Some(result);
220 result
221 }
222
223 fn pairwise(&mut self) -> Matrix {
224 if let Some(matrix) = self.pairs { return matrix; }
225 let mut matrix = [[0; LIMIT]; LIMIT];
226 for group in &self.groups {
227 let mut positions = [LIMIT; LIMIT];
228 for (i, &a) in group.ranking.iter().enumerate() { positions[a] = i; }
229 for a in 0..self.names.len() { for b in 0..self.names.len() {
230 if positions[a] < positions[b] { matrix[a][b] += group.votes; }
231 } }
232 }
233 self.pairs = Some(matrix);
234 matrix
235 }

237 fn condorcet(&mut self, field: Field, loser: bool) -> Option<usize> {
238 let matrix = self.pairwise();
239 let list = members(field);
240 list.iter().copied().find(|&a| list.iter().all(|&b| {
241 a == b || if loser { matrix[a][b] < matrix[b][a] } else { matrix[a][b] > matrix[b][a] }
242 }))
243 }

245 fn look_back(&self, field: Field, history: &[Row]) -> (usize, bool) {
246 let current = history.last().unwrap();
247 let mut choices = members(field);
248 let minimum = choices.iter().map(|&a| current[a]).min().unwrap();
249 choices.retain(|&a| current[a] == minimum);
250 let tie = choices.len() > 1;
251 for scores in history.iter().rev().skip(1) {
252 let minimum = choices.iter().map(|&a| scores[a]).min().unwrap();
253 choices.retain(|&a| scores[a] == minimum);
254 }
255 (*choices.iter().max_by_key(|&&a| self.priority[a]).unwrap(), tie)
256 }

258 fn elimination(&mut self, field: Field, coombs: bool, exhausted: bool) -> Outcome {
259 let mut remaining = field;
260 let mut removed = Vec::new();
261 let mut history = Vec::new();
262 let mut lines = Vec::new();
263 loop {
264 let scores = self.tallies(remaining);
265 let round = lines.len() + 1;
266 let mut line = if exhausted {
267 let active = sum(remaining, &scores.first);
268 format!("round {round}: active {active}, exhausted {}", self.total - active)
269 } else { format!("round {round}: {}", self.values(remaining, &scores.first, false)) };
270 if remaining.count_ones() == 1 || majority(remaining, &scores.first) {
271 lines.push(line);
272 let mut order = self.ordered(remaining, &scores.first, true);
273 order.extend(removed.into_iter().rev());
274 return if exhausted { Outcome { order, lines } } else { self.finish(order, false, lines) };
275 }
276 let (eliminated, tie) = if coombs {
277 let list = members(remaining);
278 let maximum = list.iter().map(|&a| scores.last[a]).max().unwrap();
279 let choices: Vec<_> = list.into_iter().filter(|&a| scores.last[a] == maximum).collect();
280 let least = self.ordered(mask(&choices), &scores.first, true);
281 line.push_str(&format!("; last {}", self.values(remaining, &scores.last, false)));
282 (*least.last().unwrap(), choices.len() > 1)
283 } else {
284 history.push(scores.first);
285 self.look_back(remaining, &history)
286 };
287 if !exhausted { line.push_str(&format!("; eliminated {}{}", self.names[eliminated], tie_mark(tie))); }
288 lines.push(line);
289 removed.push(eliminated);
290 remaining &= !bit(eliminated);
291 }
292 }

294 fn runoff(&mut self, field: Field) -> Outcome {
295 let first = self.tallies(field);
296 let mut order = self.ordered(field, &first.first, true);
297 let mut lines = vec![format!("round 1: {}", self.values(field, &first.first, false))];
298 let mut tie = false;
299 if !majority(field, &first.first) {
300 let (a, b) = (order[0], order[1]);
301 let finalists = bit(a) | bit(b);
302 let advancement_tie = order.len() >= 3 && first.first[b] == first.first[order[2]];
303 let second = self.tallies(finalists);
304 tie = second.first[a] == second.first[b];
305 if second.first[b] > second.first[a] { order.swap(0, 1); }
306 lines.push(format!("finalists: {}{}", self.names_text(&members(finalists), ", "), tie_mark(advancement_tie)));
307 lines.push(format!("round 2: {}", self.values(finalists, &second.first, false)));
308 }
309 self.finish(order, tie, lines)
310 }

312 fn bucklin(&mut self, field: Field) -> Outcome {
313 let scores = self.tallies(field);
314 let majority = scores.nonblank / 2 + 1;
315 let mut lines = vec![format!("majority: {majority}")];
316 let mut order = members(field);
317 let mut final_depth = 0;
318 for depth in 0..order.len() {
319 final_depth = depth;
320 lines.push(format!("round {}: {}", depth + 1, self.values(field, &scores.approval[depth], false)));
321 if order.iter().any(|&a| scores.approval[depth][a] >= majority) { break; }
322 }
323 order.sort_by(|&a, &b| {
324 for depth in (0..=final_depth).rev() {
325 let comparison = scores.approval[depth][b].cmp(&scores.approval[depth][a]);
326 if !comparison.is_eq() { return comparison; }
327 }
328 self.priority[a].cmp(&self.priority[b])
329 });
330 let tie = scores.approval[final_depth][order[0]] == scores.approval[final_depth][order[1]];
331 self.finish(order, tie, lines)
332 }

334 fn borda_elimination(&mut self, field: Field, nanson: bool) -> Outcome {
335 let mut remaining = field;
336 let mut tail = Vec::new();
337 let mut lines = Vec::new();
338 let mut history = Vec::new();
339 while remaining.count_ones() >= 2 {
340 let scores = self.tallies(remaining).borda;
341 let mut line = format!("round {}: {}", lines.len() + 1, self.values(remaining, &scores, false));
342 let mut removed = if nanson {
343 let total = sum(remaining, &scores);
344 let below: Vec<_> = members(remaining).into_iter().filter(|&a| scores[a] * (remaining.count_ones() as i32) < total).collect();
345 if below.is_empty() { lines.push(line); break; }
346 line.push_str(&format!("; eliminated {}", self.names_text(&below, ", ")));
347 self.ordered(mask(&below), &scores, true)
348 } else {
349 history.push(scores);
350 let (a, tie) = self.look_back(remaining, &history);
351 line.push_str(&format!("; eliminated {}{}", self.names[a], tie_mark(tie)));
352 vec![a]
353 };
354 lines.push(line);
355 remaining &= !mask(&removed);
356 removed.extend(tail);
357 tail = removed;
358 }
359 let mut order = self.ordered(remaining, &[0; LIMIT], true);
360 let tie = nanson && order.len() > 1;
361 order.extend(tail);
362 self.finish(order, tie, lines)
363 }

365 fn pair_lines(&self, field: Field, matrix: &Matrix) -> Vec<String> {
366 let list = members(field);
367 let mut lines = Vec::new();
368 for (i, &a) in list.iter().enumerate() { for &b in &list[i + 1..] {
369 let result = if matrix[a][b] == matrix[b][a] { "tie" }
370 else { &self.names[if matrix[a][b] > matrix[b][a] { a } else { b }] };
371 lines.push(format!("{} vs {}: {}-{} ({result})", self.names[a], self.names[b], matrix[a][b], matrix[b][a]));
372 } }
373 lines
374 }

376 fn condorcet_report(&mut self, field: Field) -> Outcome {
377 let win = self.condorcet(field, false);
378 let lose = self.condorcet(field, true);
379 let matrix = self.pairwise();
380 let mut weak = [[false; LIMIT]; LIMIT];
381 let mut strong = weak;
382 let list = members(field);
383 for &a in &list { for &b in &list {
384 weak[a][b] = matrix[a][b] >= matrix[b][a];
385 strong[a][b] = matrix[a][b] > matrix[b][a];
386 } }
387 closure(field, &mut weak);
388 closure(field, &mut strong);
389 let smith: Vec<_> = list.iter().copied().filter(|&a| list.iter().all(|&b| weak[a][b])).collect();
390 let schwartz: Vec<_> = list.iter().copied().filter(|&a| list.iter().all(|&b| !strong[b][a] || strong[a][b])).collect();
391 Outcome { order: Vec::new(), lines: vec![
392 format!("winner: {}", win.map(|a| self.names[a].as_str()).unwrap_or("NONE")),
393 format!("loser: {}", lose.map(|a| self.names[a].as_str()).unwrap_or("NONE")),
394 format!("smith: {}", self.names_text(&smith, ", ")),
395 format!("schwartz: {}", self.names_text(&schwartz, ", ")),
396 ] }
397 }

399 fn topological(&self, field: Field, relation: &Relation) -> (Vec<usize>, bool) {
400 let mut remaining = field;
401 let mut order = Vec::new();
402 let mut tie = false;
403 while remaining != 0 {
404 let list = members(remaining);
405 let choices: Vec<_> = list.iter().copied().filter(|&a| list.iter().all(|&b| a == b || !relation[b][a])).collect();
406 if order.is_empty() { tie = choices.len() > 1; }
407 let a = *choices.iter().min_by_key(|&&a| self.priority[a]).unwrap();
408 order.push(a);
409 remaining &= !bit(a);
410 }
411 (order, tie)
412 }

414 fn schulze(&mut self, field: Field) -> Outcome {
415 let matrix = self.pairwise();
416 let list = members(field);
417 let mut paths = [[0; LIMIT]; LIMIT];
418 for &a in &list { for &b in &list {
419 if matrix[a][b] > matrix[b][a] { paths[a][b] = matrix[a][b]; }
420 } }
421 for &via in &list { for &a in &list { for &b in &list {
422 if a != via && b != via && a != b { paths[a][b] = paths[a][b].max(paths[a][via].min(paths[via][b])); }
423 } } }
424 let mut relation = [[false; LIMIT]; LIMIT];
425 for &a in &list { for &b in &list { relation[a][b] = paths[a][b] > paths[b][a]; } }
426 let (order, tie) = self.topological(field, &relation);
427 let mut lines = self.pair_lines(field, &paths);
428 lines.push(format!("ranking: {}", self.names_text(&order, " > ")));
429 self.finish(order, tie, lines)
430 }

432 fn ranked_pairs(&mut self, field: Field) -> Outcome {
433 let matrix = self.pairwise();
434 let list = members(field);
435 let mut pairs = Vec::new();
436 for &a in &list { for &b in &list { if matrix[a][b] > matrix[b][a] { pairs.push((a, b)); } } }
437 pairs.sort_by_key(|&(a, b)| (-matrix[a][b], matrix[b][a], self.priority[a], self.priority[b]));
438 let mut locked = [[false; LIMIT]; LIMIT];
439 let mut reached = locked;
440 let mut lines = Vec::new();
441 for (a, b) in pairs {
442 let action = if reached[b][a] { "skip" } else { "lock" };
443 lines.push(format!("{action} {}>{} {}-{}", self.names[a], self.names[b], matrix[a][b], matrix[b][a]));
444 if !reached[b][a] {
445 locked[a][b] = true;
446 reached[a][b] = true;
447 closure(field, &mut reached);
448 }
449 }
450 let (order, tie) = self.topological(field, &locked);
451 self.finish(order, tie, lines)
452 }

454 fn kemeny(&mut self, field: Field) -> Outcome {
455 let matrix = self.pairwise();
456 let mut best = [0; 1 << LIMIT];
457 let mut ways = [0; 1 << LIMIT];
458 let mut heads = [0; 1 << LIMIT];
459 ways[0] = 1;
460 for subset in 1..=usize::from(field) {
461 let current = subset as Field;
462 if current & !field != 0 { continue; }
463 best[subset] = -1;
464 let list = members(current);
465 for &a in &list {
466 let rest = usize::from(current & !bit(a));
467 let score = best[rest] + list.iter().filter(|&&b| b != a).map(|&b| matrix[a][b]).sum::<i32>();
468 if score > best[subset] { best[subset] = score; ways[subset] = 0; heads[subset] = 0; }
469 if score == best[subset] {
470 let count = ways[rest];
471 ways[subset] += count;
472 heads[subset] |= bit(a);
473 }
474 }
475 }
476 let mut order = Vec::new();
477 let mut remaining = field;
478 while remaining != 0 {
479 let choices = heads[usize::from(remaining)];
480 let a = *self.tiebreak.iter().find(|&&a| choices & bit(a) != 0).unwrap();
481 order.push(a);
482 remaining &= !bit(a);
483 }
484 let index = usize::from(field);
485 let lines = vec![format!("score: {}", best[index]), format!("optimal: {}", ways[index]), format!("ranking: {}", self.names_text(&order, " > "))];
486 self.finish(order, heads[index].count_ones() > 1, lines)
487 }

489 fn stv(&mut self, field: Field, seats: usize) -> Outcome {
490 let mut parcels: Vec<_> = self.groups.iter().map(|group| Parcel {
491 value: if group.ranking.iter().any(|&a| field & bit(a) != 0) { 100 } else { 0 },
492 owner: None,
493 }).collect();
494 let quota = self.tallies(field).nonblank * 100 / (seats as i32 + 1) + 1;
495 let mut lines = vec![format!("quota: {}", money(quota))];
496 let mut continuing = field;
497 let mut elected = Vec::new();
498 let mut history = Vec::new();
499 loop {
500 let mut scores = [0; LIMIT];
501 let mut exhausted = 0;
502 for (parcel, group) in parcels.iter_mut().zip(&self.groups) {
503 parcel.owner = None;
504 if parcel.value == 0 { continue; }
505 parcel.owner = group.ranking.iter().copied().find(|&a| continuing & bit(a) != 0);
506 let value = group.votes * parcel.value;
507 if let Some(a) = parcel.owner { scores[a] += value; }
508 else { exhausted += value; }
509 }
510 history.push(scores);
511 let mut line = format!("round {}: {}; exhausted {}", history.len(), self.values(continuing, &scores, true), money(exhausted));
512 let order = self.ordered(continuing, &scores, true);
513 if order.len() == seats - elected.len() {
514 line.push_str(&format!("; elected {}", self.names_text(&order, ", ")));
515 elected.extend(order);
516 lines.push(line);
517 break;
518 }
519 let a = order[0];
520 if scores[a] >= quota {
521 let surplus = scores[a] - quota;
522 elected.push(a);
523 continuing &= !bit(a);
524 for parcel in &mut parcels {
525 if parcel.owner == Some(a) { parcel.value = parcel.value * surplus / scores[a]; }
526 }
527 line.push_str(&format!("; elected {} (surplus {})", self.names[a], money(surplus)));
528 lines.push(line);
529 if elected.len() == seats { break; }
530 } else {
531 let (excluded, tie) = self.look_back(continuing, &history);
532 continuing &= !bit(excluded);
533 line.push_str(&format!("; excluded {}{}", self.names[excluded], tie_mark(tie)));
534 lines.push(line);
535 }
536 }
537 lines.push(format!("elected: {}", self.names_text(&elected, ", ")));
538 Outcome { order: Vec::new(), lines }
539 }

541 fn tally(&mut self, field: Field, method: Method) -> Outcome {
542 match method {
543 Method::Runoff => return self.runoff(field),
544 Method::Irv | Method::Coombs => return self.elimination(field, method == Method::Coombs, false),
545 Method::Bucklin => return self.bucklin(field),
546 Method::Baldwin | Method::Nanson => return self.borda_elimination(field, method == Method::Nanson),
547 Method::Schulze => return self.schulze(field),
548 Method::RankedPairs => return self.ranked_pairs(field),
549 Method::Kemeny => return self.kemeny(field),
550 Method::Condorcet => return self.condorcet_report(field),
551 Method::Stv(seats) => return self.stv(field, seats),
552 _ => {}
553 }
554 let base = self.tallies(field);
555 let mut lines = Vec::new();
556 let mut condorcet_winner = None;
557 let (scores, label, descending) = match method {
558 Method::Plurality => (base.first, Some("votes"), true),
559 Method::Borda | Method::Black => {
560 if method == Method::Black {
561 condorcet_winner = self.condorcet(field, false);
562 lines.push(format!("condorcet: {}", condorcet_winner.map(|a| self.names[a].as_str()).unwrap_or("NONE")));
563 }
564 (base.borda, Some("scores"), true)
565 }
566 Method::Veto => (base.last, Some("vetoes"), false),
567 Method::Approval(k) => (base.approval[k.min(field.count_ones() as usize) - 1], Some("approvals"), true),
568 Method::Copeland => {
569 let matrix = self.pairwise();
570 let list = members(field);
571 let mut scores = [0; LIMIT];
572 let mut descriptions = Vec::new();
573 for &a in &list {
574 let (mut wins, mut draws, mut losses) = (0, 0, 0);
575 for &b in &list { if a != b {
576 match matrix[a][b].cmp(&matrix[b][a]) {
577 std::cmp::Ordering::Greater => wins += 1,
578 std::cmp::Ordering::Equal => draws += 1,
579 std::cmp::Ordering::Less => losses += 1,
580 }
581 } }
582 scores[a] = 2 * wins + draws;
583 descriptions.push(format!("{} {}.{} ({wins}-{draws}-{losses})", self.names[a], scores[a] / 2, scores[a] % 2 * 5));
584 }
585 lines.push(format!("scores: {}", descriptions.join(", ")));
586 (scores, None, true)
587 }
588 Method::Minimax(kind) => {
589 let matrix = self.pairwise();
590 let list = members(field);
591 let mut scores = [0; LIMIT];
592 for &a in &list { for &b in &list { if a != b {
593 let (opposition, support) = (matrix[b][a], matrix[a][b]);
594 let value = if kind == MinimaxKind::Opposition { opposition }
595 else if opposition > support { if kind == MinimaxKind::Margin { opposition - support } else { opposition } }
596 else { 0 };
597 scores[a] = scores[a].max(value);
598 } } }
599 (scores, Some("scores"), false)
600 }
601 _ => unreachable!(),
602 };
603 let mut order = self.ordered(field, &scores, descending);
604 let mut tie = scores[order[0]] == scores[order[1]];
605 if let Some(a) = condorcet_winner {
606 order.remove(order.iter().position(|&b| b == a).unwrap());
607 order.insert(0, a);
608 tie = false;
609 }
610 if let Some(label) = label { lines.push(format!("{label}: {}", self.values(field, &scores, false))); }
611 self.finish(order, tie, lines)
612 }
613
614 fn ballots_report(&self) -> Vec<String> {
615 let mut counts = BTreeMap::<String, i32>::new();
616 for group in &self.groups {
617 let ranking = effective(&group.ranking, self.active);
618 let notation = if ranking.is_empty() { "-".to_owned() } else { self.names_text(&ranking, ">") };
619 *counts.entry(notation).or_default() += group.votes;
620 }
621 let blank = counts.get("-").copied().unwrap_or(0);
622 let mut groups: Vec<_> = counts.into_iter().collect();
623 groups.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| a.0.cmp(&b.0)));
624 let mut lines = vec![format!("total {}, blank {blank}, distinct {}", self.total, groups.len())];
625 lines.extend(groups.into_iter().map(|(notation, votes)| format!("{votes} {notation}")));
626 lines
627 }
628}

630#[derive(Default)]
631struct Lab {
632 election: Option<Election>,
633 number: usize,
634}
635
636impl Lab {
637 fn execute(&mut self, line: &str) -> Result<Vec<String>, String> {
638 let words: Vec<_> = line.split(' ').collect();
639 let command = words[0];
640 if command == "CANDIDATES" {
641 let names = &words[1..];
642 if names.len() == 1 { return Err("too few candidates".into()); }
643 for (i, &name) in names.iter().enumerate() {
644 if names[..i].contains(&name) { return Err(format!("duplicate candidate {name}")); }
645 }
646 self.number += 1;
647 self.election = Some(Election::new(self.number, names.iter().map(|name| (*name).to_owned()).collect()));
648 return Ok(vec![format!("election {}: {} candidates", self.number, names.len())]);
649 }
650 let e = self.election.as_mut().ok_or_else(|| "no election".to_owned())?;
651 match command {
652 "BALLOT" => {
653 let ranking = if words[2] == "-" { Vec::new() }
654 else { e.checked_names(&words[2].split('>').collect::<Vec<_>>())? };
655 let votes = words[1].parse().unwrap();
656 e.groups.push(Ballot { votes, ranking });
657 e.total += votes;
658 e.finish_entry();
659 return Ok(Vec::new());
660 }
661 "RANDOM" => {
662 let count: usize = words[1].parse().unwrap();
663 let mut state: i64 = words[2].parse().unwrap();
664 for _ in 0..count {
665 let mut ranking = members(e.active);
666 for i in (1..ranking.len()).rev() {
667 state = 48271 * state % 2147483647;
668 ranking.swap(i, (state % (i as i64 + 1)) as usize);
669 }
670 state = 48271 * state % 2147483647;
671 ranking.truncate((state % ranking.len() as i64) as usize + 1);
672 e.groups.push(Ballot { votes: 1, ranking });
673 }
674 e.total += count as i32;
675 e.finish_entry();
676 return Ok(Vec::new());
677 }
678 "UNDO" => {
679 e.entry_ends.pop().ok_or_else(|| "nothing to undo".to_owned())?;
680 let end = e.entry_ends.last().copied().unwrap_or(0);
681 e.total -= e.groups[end..].iter().map(|group| group.votes).sum::<i32>();
682 e.groups.truncate(end);
683 e.invalidate();
684 return Ok(Vec::new());
685 }
686 "WITHDRAW" | "RESTORE" => {
687 let a = e.candidate(words[1])?;
688 let active = e.active & bit(a) != 0;
689 if command == "WITHDRAW" {
690 if !active { return Err(format!("already withdrawn {}", words[1])); }
691 if e.active.count_ones() <= 2 { return Err("too few candidates".into()); }
692 e.active &= !bit(a);
693 } else {
694 if active { return Err(format!("not withdrawn {}", words[1])); }
695 e.active |= bit(a);
696 }
697 return Ok(Vec::new());
698 }
699 "TIEBREAK" => {
700 let order = e.checked_names(&words[1..])?;
701 if order.len() != e.names.len() { return Err("incomplete tiebreak".into()); }
702 for (i, &a) in order.iter().enumerate() { e.priority[a] = i; }
703 e.tiebreak = order;
704 return Ok(Vec::new());
705 }
706 "STATUS" => {
707 let withdrawn = members((bit(e.names.len()) - 1) & !e.active);
708 return Ok(vec![format!("[{line}]"), format!("election {}", e.number),
709 format!("active: {}", e.names_text(&members(e.active), ", ")),
710 format!("withdrawn: {}", if withdrawn.is_empty() { "none".into() } else { e.names_text(&withdrawn, ", ") }),
711 format!("tiebreak: {}", e.names_text(&e.tiebreak, ", ")),
712 format!("ballots: {} in {} entries", e.total, e.entry_ends.len()),
713 ]);
714 }
715 _ => {}
716 }
717 if e.entry_ends.is_empty() { return Err("no ballots".into()); }
718 let requested = if matches!(command, "RANKING" | "SPOILERS") {
719 let method = Method::parse(&words[1..]);
720 if !method.rankable() { return Err("not rankable".into()); }
721 if command == "SPOILERS" && e.active.count_ones() < 3 { return Err("too few candidates".into()); }
722 Some(method)
723 } else { None };
724 let mut compared = [0; 2];
725 if command == "COMPARE" {
726 for i in 0..2 {
727 compared[i] = e.candidate(words[i + 1])?;
728 if e.active & bit(compared[i]) == 0 { return Err(format!("withdrawn candidate {}", words[i + 1])); }
729 }
730 if compared[0] == compared[1] { return Err("same candidate".into()); }
731 }
732 if command == "STV" && words[1].parse::<u32>().unwrap() >= e.active.count_ones() { return Err("too many seats".into()); }
733 let mut lines = vec![format!("[{line}]")];
734 match command {
735 "RANKING" | "SPOILERS" => {
736 let method = requested.unwrap();
737 let result = e.tally(e.active, method);
738 if command == "RANKING" { lines.push(e.names_text(&result.order, " > ")); }
739 else {
740 let win = result.order[0];
741 lines.push(format!("winner: {}", e.names[win]));
742 for a in members(e.active) { if a != win {
743 let other = e.tally(e.active & !bit(a), method).order[0];
744 lines.push(format!("without {}: {}{}", e.names[a], e.names[other], if other != win { " (changed)" } else { "" }));
745 } }
746 }
747 }
748 "BALLOTS" => lines.extend(e.ballots_report()),
749 "PROFILE" => {
750 let scores = e.tallies(e.active);
751 let size = e.active.count_ones() as usize;
752 for a in members(e.active) {
753 let counts: Vec<_> = (0..size).map(|j| {
754 (scores.approval[j][a] - if j == 0 { 0 } else { scores.approval[j - 1][a] }).to_string()
755 }).collect();
756 lines.push(format!("{}: {} unranked {}", e.names[a], counts.join(" "), e.total - scores.approval[size - 1][a]));
757 }
758 }
759 "PAIRWISE" => {
760 let matrix = e.pairwise();
761 lines.extend(e.pair_lines(e.active, &matrix));
762 }
763 "EXHAUSTED" => lines.extend(e.elimination(e.active, false, true).lines),
764 "SUMMARY" | "COMPARE" => {
765 let mut wins = [0; LIMIT];
766 let mut ahead = [Vec::new(), Vec::new()];
767 for method in SUMMARY {
768 let result = e.tally(e.active, method);
769 if command == "SUMMARY" {
770 let win = result.order[0];
771 wins[win] += 1;
772 lines.push(format!("{method}: {}", e.names[win]));
773 } else {
774 let first = result.order.iter().position(|&a| a == compared[0]).unwrap();
775 let second = result.order.iter().position(|&a| a == compared[1]).unwrap();
776 let side = if first < second { 0 } else { 1 };
777 ahead[side].push(method.to_string());
778 }
779 }
780 if command == "SUMMARY" { lines.push(format!("wins: {}", e.values(e.active, &wins, false))); }
781 else {
782 let matrix = e.pairwise();
783 let [a, b] = compared;
784 lines.push(format!("{} {}, {} {}, neither {}", e.names[a], matrix[a][b], e.names[b], matrix[b][a], e.total - matrix[a][b] - matrix[b][a]));
785 for i in 0..2 {
786 lines.push(format!("{} ahead: {}", e.names[compared[i]], if ahead[i].is_empty() { "none".into() } else { ahead[i].join(", ") }));
787 }
788 }
789 }
790 _ => lines.extend(e.tally(e.active, Method::parse(&words)).lines),
791 }
792 Ok(lines)
793 }
794}
795
796fn main() {
797 let stdin = io::stdin();
798 let mut input = stdin.lock().lines();
799 let count: usize = input.next().unwrap().unwrap().parse().unwrap();
800 let stdout = io::stdout();
801 let mut output = io::BufWriter::new(stdout.lock());
802 let mut lab = Lab::default();
803 for _ in 0..count {
804 let line = input.next().unwrap().unwrap();
805 match lab.execute(&line) {
806 Ok(lines) => for row in lines { writeln!(output, "{row}").unwrap(); },
807 Err(reason) => writeln!(output, "ERROR {reason}").unwrap(),
808 }
809 }
810}
