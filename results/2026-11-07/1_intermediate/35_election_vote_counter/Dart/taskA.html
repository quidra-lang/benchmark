10enum Method {
11 plurality, runoff, irv, coombs, bucklin, borda, baldwin, nanson,
12 black, veto, condorcet, copeland, minimax, schulze, rankedPairs,
13 kemeny, approval, stv,
14}
15enum MinimaxKind { winning, margin, opposition }
16enum ScoreKind { first, last, borda }

18final class MethodSpec {
19 final Method kind;
20 final int parameter;
21 final MinimaxKind minimax;
22 const MethodSpec(this.kind, [this.parameter = 0, this.minimax = MinimaxKind.winning]);
23 bool get rankable => kind != Method.condorcet && kind != Method.stv;
24 @override
25 String toString() => switch (kind) {
26 Method.minimax => 'MINIMAX ${minimax.name.toUpperCase()}',
27 Method.approval || Method.stv => '${kind.name.toUpperCase()} $parameter',
28 _ => kind.name.toUpperCase(),
29 };
30 factory MethodSpec.parse(List<String> words, [int start = 0]) {
31 final kind = Method.values.firstWhere((m) => m.name.toUpperCase() == words[start]);
32 return switch (kind) {
33 Method.minimax => MethodSpec(kind, 0,
34 MinimaxKind.values.firstWhere((m) => m.name.toUpperCase() == words[start + 1])),
35 Method.approval || Method.stv => MethodSpec(kind, int.parse(words[start + 1])),
36 _ => MethodSpec(kind),
37 };
38 }
39}

55int sum(Iterable<int> values) => values.fold(0, (a, b) => a + b);
56int minimum(Iterable<int> values) => values.reduce((a, b) => a < b ? a : b);

59final class Election {
60 final int number;
61 final List<String> names;
62 final List<bool> withdrawn;
63 List<int> tieOrder;
64 final List<List<Ballot>> entries = [];
65 Election(this.number, this.names)
66 : withdrawn = List.filled(names.length, false),
67 tieOrder = List.generate(names.length, (i) => i)
68 ..sort((a, b) => names[a].compareTo(names[b]));
69 List<int> get active => [for (var i = 0; i < names.length; i++) if (!withdrawn[i]) i];
70 Iterable<Ballot> get groups => entries.expand((entry) => entry);

89final class Tally {
90 final Election election;
91 final List<int> field;
92 late final List<int> priority;
93 late final List<Ballot> groups;
94 late final List<List<int>> pairwise;
95 late final int total;
96 int get candidateCount => election.names.length;
97 Tally(this.election, [List<int>? candidates]) : field = candidates ?? election.active {
98 priority = List.filled(candidateCount, 0);
99 for (var i = 0; i < election.tieOrder.length; i++) {
100 priority[election.tieOrder[i]] = i;
101 }
102 final included = field.toSet();
103 groups = [for (final group in election.groups)
104 Ballot(group.votes, group.ranking.where(included.contains).toList())];
105 total = sum(groups.map((group) => group.votes));
106 pairwise = List.generate(candidateCount, (_) => List.filled(candidateCount, 0));
107 for (final group in groups) {
108 final ranks = List.filled(candidateCount, field.length);
109 for (var i = 0; i < group.ranking.length; i++) {
110 ranks[group.ranking[i]] = i;
111 }
112 for (final x in field) {
113 for (final y in field) {
114 if (ranks[x] < ranks[y]) pairwise[x][y] += group.votes;
115 }
116 }
117 }
118 }

120 String name(int c) => election.names[c];
121 String names(Iterable<int> candidates) => election.nameList(candidates);
122 String ranking(Iterable<int> order) => order.map(name).join(' > ');
123 static String amount(int value) => '${value ~/ 100}.${(value % 100).toString().padLeft(2, '0')}';
124 String values(Iterable<int> candidates, List<int> scores, [bool amounts = false]) =>
125 candidates.map((c) => '${name(c)} ${amounts ? amount(scores[c]) : scores[c]}').join(', ');

127 List<int> scores(List<int> candidates, ScoreKind kind) {
128 final result = List.filled(candidateCount, 0);
129 final included = candidates.toSet();
130 for (final group in groups) {
131 final ranked = group.ranking.where(included.contains).toList();
132 switch (kind) {
133 case ScoreKind.first:
134 if (ranked.isNotEmpty) result[ranked.first] += group.votes;
135 case ScoreKind.last:
136 if (ranked.length < candidates.length) {
137 for (final candidate in candidates) {
138 if (!ranked.contains(candidate)) result[candidate] += group.votes;
139 }
140 } else {
141 result[ranked.last] += group.votes;
142 }
143 case ScoreKind.borda:
144 for (var i = 0; i < ranked.length; i++) {
145 result[ranked[i]] += group.votes * (candidates.length - i - 1);
146 }
147 }
148 }
149 return result;
150 }
151 List<int> approval(int depth) {
152 final result = List.filled(candidateCount, 0);
153 for (final group in groups) {
154 for (final candidate in group.ranking.take(depth)) {
155 result[candidate] += group.votes;
156 }
157 }
158 return result;
159 }
160 List<int> order(Iterable<int> candidates, List<int> scores, [bool descending = true]) =>
161 candidates.toList()..sort((a, b) {
162 final difference = descending ? scores[b].compareTo(scores[a]) : scores[a].compareTo(scores[b]);
163 return difference != 0 ? difference : priority[a].compareTo(priority[b]);
164 });
165 List<int> priorityOrder(Iterable<int> candidates) => candidates.toList()
166 ..sort((a, b) => priority[a].compareTo(priority[b]));
167 bool topTie(List<int> order, List<int> scores) => order.length > 1 && scores[order[0]] == scores[order[1]];
168 (int, bool) eliminate(List<int> candidates, List<int> values, List<List<int>> history) {
169 final low = minimum(candidates.map((c) => values[c]));
170 var pool = candidates.where((c) => values[c] == low).toList();
171 final tie = pool.length > 1;
172 for (final previous in history.reversed) {
173 final previousLow = minimum(pool.map((c) => previous[c]));
174 pool = pool.where((c) => previous[c] == previousLow).toList();
175 }
176 return (priorityOrder(pool).last, tie);
177 }
178 TallyResult finish(List<int> order, bool tie, List<String> lines, [List<String> exhaustion = const []]) {
179 lines.add('winner: ${name(order.first)}${tie ? ' (tie)' : ''}');
180 return TallyResult(order, tie, lines, exhaustion);
181 }
182 TallyResult scored(List<int> scores, String label, [bool descending = true]) {
183 final standings = order(field, scores, descending);
184 return finish(standings, topTie(standings, scores), ['$label: ${values(field, scores)}']);
185 }
186 int condorcet([bool loser = false]) {
187 for (final x in field) {
188 if (field.every((y) => x == y || (loser ? pairwise[x][y] < pairwise[y][x] : pairwise[x][y] > pairwise[y][x]))) {
189 return x;
190 }
191 }
192 return -1;
193 }
194 List<String> pairLines([List<List<int>>? table]) {
195 final counts = table ?? pairwise;
196 final lines = <String>[];
197 for (var i = 0; i < field.length; i++) {
198 for (var j = i + 1; j < field.length; j++) {
199 final x = field[i], y = field[j];
200 final winner = counts[x][y] > counts[y][x] ? name(x) : counts[x][y] < counts[y][x] ? name(y) : 'tie';
201 lines.add('${name(x)} vs ${name(y)}: ${counts[x][y]}-${counts[y][x]} ($winner)');
202 }
203 }
204 return lines;
205 }
206 TallyResult runoff() {
207 final first = scores(field, ScoreKind.first);
208 final standings = order(field, first);
209 final lines = ['round 1: ${values(field, first)}'];
210 if (2 * first[standings.first] > sum(first)) return finish(standings, false, lines);
211 final finalists = field.where(standings.take(2).contains).toList();
212 final advanceTie = field.length > 2 && first[standings[1]] == first[standings[2]];
213 final second = scores(finalists, ScoreKind.first);
214 final finalOrder = finalists.toList()..sort((a, b) {
215 var difference = second[b].compareTo(second[a]);
216 if (difference == 0) difference = first[b].compareTo(first[a]);
217 return difference != 0 ? difference : priority[a].compareTo(priority[b]);
218 });
219 lines.add('finalists: ${names(finalists)}${advanceTie ? ' (tie)' : ''}');
220 lines.add('round 2: ${values(finalists, second)}');
221 return finish([...finalOrder, ...standings.skip(2)], second[finalists[0]] == second[finalists[1]], lines);
222 }
223 TallyResult instantRunoff([bool coombs = false]) {
224 final remaining = field.toList();
225 final removed = <int>[], history = <List<int>>[];
226 final lines = <String>[], exhaustion = <String>[];
227 var round = 0;
228 while (true) {
229 final first = scores(remaining, ScoreKind.first);
230 final active = sum(first);
231 var line = 'round ${++round}: ${values(remaining, first)}';
232 exhaustion.add('round $round: active $active, exhausted ${total - active}');
233 if (remaining.length == 1 || remaining.any((c) => 2 * first[c] > active)) {
234 lines.add(line);
235 return finish([...order(remaining, first), ...removed.reversed], false, lines, exhaustion);
236 }
237 int loser;
238 bool tie;
239 if (coombs) {
240 final last = scores(remaining, ScoreKind.last);
241 final high = maximum(remaining.map((c) => last[c]));
242 var pool = remaining.where((c) => last[c] == high).toList();
243 tie = pool.length > 1;
244 final low = minimum(pool.map((c) => first[c]));
245 pool = pool.where((c) => first[c] == low).toList();
246 loser = priorityOrder(pool).last;
247 line += '; last ${values(remaining, last)}';
248 } else {
249 (loser, tie) = eliminate(remaining, first, history);
250 }
251 lines.add('$line; eliminated ${name(loser)}${tie ? ' (tie)' : ''}');
252 removed.add(loser);
253 history.add(first);
254 remaining.remove(loser);
255 }
256 }
257 TallyResult bucklin() {
258 final majority = sum(groups.where((g) => g.ranking.isNotEmpty).map((g) => g.votes)) ~/ 2 + 1;
259 final rounds = <List<int>>[], lines = ['majority: $majority'];
260 for (var round = 1; round <= field.length; round++) {
261 final current = approval(round);
262 rounds.add(current);
263 lines.add('round $round: ${values(field, current)}');
264 if (field.any((c) => current[c] >= majority)) break;
265 }
266 final standings = field.toList()..sort((a, b) {
267 for (final round in rounds.reversed) {
268 final difference = round[b].compareTo(round[a]);
269 if (difference != 0) return difference;
270 }
271 return priority[a].compareTo(priority[b]);
272 });
273 return finish(standings, topTie(standings, rounds.last), lines);
274 }
275 TallyResult bordaElimination(bool nanson) {
276 var remaining = field.toList();
277 final removed = <List<int>>[], history = <List<int>>[];
278 final lines = <String>[];
279 var round = 0;
280 while (remaining.length > 1) {
281 final current = scores(remaining, ScoreKind.borda);
282 var line = 'round ${++round}: ${values(remaining, current)}';
283 List<int> losers;
284 if (nanson) {
285 final total = sum(current);
286 losers = remaining.where((c) => current[c] * remaining.length < total).toList();
287 if (losers.isEmpty) { lines.add(line); break; }
288 line += '; eliminated ${names(losers)}';
289 } else {
290 final (loser, tie) = eliminate(remaining, current, history);
291 losers = [loser];
292 line += '; eliminated ${name(loser)}${tie ? ' (tie)' : ''}';
293 }
294 lines.add(line);
295 history.add(current);
296 removed.add(order(losers, current));
297 remaining = remaining.where((c) => !losers.contains(c)).toList();
298 }
299 return finish([...priorityOrder(remaining), ...removed.reversed.expand((group) => group)], nanson && remaining.length > 1, lines);
300 }
301 TallyResult black() {
302 final winner = condorcet(), current = scores(field, ScoreKind.borda);
303 var standings = order(field, current);
304 final tie = winner < 0 && topTie(standings, current);
305 if (winner >= 0) standings = [winner, ...standings.where((c) => c != winner)];
306 return finish(standings, tie, ['condorcet: ${winner < 0 ? 'NONE' : name(winner)}', 'scores: ${values(field, current)}']);
307 }
308 TallyResult condorcetReport() {
309 final weak = List.generate(candidateCount, (_) => List.filled(candidateCount, false));
310 final strong = List.generate(candidateCount, (_) => List.filled(candidateCount, false));
311 for (final x in field) {
312 for (final y in field) {
313 weak[x][y] = x == y || pairwise[x][y] >= pairwise[y][x];
314 strong[x][y] = x != y && pairwise[x][y] > pairwise[y][x];
315 }
316 }
317 for (final via in field) {
318 for (final x in field) {
319 for (final y in field) {
320 weak[x][y] = weak[x][y] || weak[x][via] && weak[via][y];
321 strong[x][y] = strong[x][y] || strong[x][via] && strong[via][y];
322 }
323 }
324 }
325 final winner = condorcet(), loser = condorcet(true);
326 return TallyResult([], false, [
327 'winner: ${winner < 0 ? 'NONE' : name(winner)}',
328 'loser: ${loser < 0 ? 'NONE' : name(loser)}',
329 'smith: ${names(field.where((x) => field.every((y) => weak[x][y])))}',
330 'schwartz: ${names(field.where((x) => field.every((y) => !strong[y][x] || strong[x][y])))}',
331 ]);
332 }
333 TallyResult copeland() {
334 final current = List.filled(candidateCount, 0), items = <String>[];
335 for (final x in field) {
336 var wins = 0, draws = 0, losses = 0;
337 for (final y in field) {
338 if (x == y) continue;
339 if (pairwise[x][y] > pairwise[y][x]) { wins++; }
340 else if (pairwise[x][y] == pairwise[y][x]) { draws++; }
341 else { losses++; }
342 }
343 current[x] = 2 * wins + draws;
344 items.add('${name(x)} ${current[x] ~/ 2}.${current[x].isEven ? 0 : 5} ($wins-$draws-$losses)');
345 }
346 final standings = order(field, current);
347 return finish(standings, topTie(standings, current), ['scores: ${items.join(', ')}']);
348 }
349 TallyResult minimax(MinimaxKind kind) {
350 final current = List.filled(candidateCount, 0);
351 for (final x in field) {
352 for (final y in field) {
353 if (x == y || kind != MinimaxKind.opposition && pairwise[y][x] <= pairwise[x][y]) continue;
354 final value = kind == MinimaxKind.margin ? pairwise[y][x] - pairwise[x][y] : pairwise[y][x];
355 if (value > current[x]) current[x] = value;
356 }
357 }
358 return scored(current, 'scores', false);
359 }
360 (List<int>, bool) dominanceOrder(bool Function(int, int) dominates) {
361 final remaining = field.toList(), standings = <int>[];
362 var tie = false;
363 while (remaining.isNotEmpty) {
364 final eligible = remaining.where((x) => remaining.every((y) => x == y || !dominates(y, x))).toList();
365 if (standings.isEmpty) tie = eligible.length > 1;
366 final next = priorityOrder(eligible).first;
367 standings.add(next);
368 remaining.remove(next);
369 }
370 return (standings, tie);
371 }
372 TallyResult schulze() {
373 final paths = List.generate(candidateCount, (_) => List.filled(candidateCount, 0));
374 for (final x in field) {
375 for (final y in field) {
376 if (x != y && pairwise[x][y] > pairwise[y][x]) paths[x][y] = pairwise[x][y];
377 }
378 }
379 for (final via in field) {
380 for (final x in field) {
381 if (x == via) continue;
382 for (final y in field) {
383 if (y == via || y == x) continue;
384 final strength = paths[x][via] < paths[via][y] ? paths[x][via] : paths[via][y];
385 if (strength > paths[x][y]) paths[x][y] = strength;
386 }
387 }
388 }
389 final (standings, tie) = dominanceOrder((y, x) => paths[y][x] > paths[x][y]);
390 return finish(standings, tie, [...pairLines(paths), 'ranking: ${ranking(standings)}']);
391 }
392 TallyResult rankedPairs() {
393 final pairs = <(int, int)>[
394 for (final x in field) for (final y in field) if (pairwise[x][y] > pairwise[y][x]) (x, y),
395 ]..sort((a, b) {
396 var difference = pairwise[b.$1][b.$2].compareTo(pairwise[a.$1][a.$2]);
397 if (difference == 0) difference = pairwise[a.$2][a.$1].compareTo(pairwise[b.$2][b.$1]);
398 if (difference == 0) difference = priority[a.$1].compareTo(priority[b.$1]);
399 return difference != 0 ? difference : priority[a.$2].compareTo(priority[b.$2]);
400 });
401 final locked = List.generate(candidateCount, (_) => List.filled(candidateCount, false));
402 bool reachable(int from, int target) {
403 final seen = <int>{}, pending = [from];
404 while (pending.isNotEmpty) {
405 final point = pending.removeLast();
406 if (point == target) return true;
407 if (!seen.add(point)) continue;
408 for (final next in field) {
409 if (locked[point][next]) pending.add(next);
410 }
411 }
412 return false;
413 }
414 final lines = <String>[];
415 for (final (x, y) in pairs) {
416 final skip = reachable(y, x);
417 if (!skip) locked[x][y] = true;
418 lines.add('${skip ? 'skip' : 'lock'} ${name(x)}>${name(y)} ${pairwise[x][y]}-${pairwise[y][x]}');
419 }
420 final (standings, tie) = dominanceOrder((y, x) => locked[y][x]);
421 return finish(standings, tie, lines);
422 }
423 TallyResult kemeny() {
424 final candidates = priorityOrder(field), path = <int>[], firsts = <int>{};
425 final used = List.filled(candidateCount, false);
426 var best = -1, optimal = 0;
427 var standings = <int>[];
428 void visit(int score) {
429 if (path.length == field.length) {
430 if (score > best) {
431 best = score;
432 optimal = 0;
433 firsts.clear();
434 standings = path.toList();
435 }
436 if (score == best) { optimal++; firsts.add(path.first); }
437 return;
438 }
439 for (final next in candidates) {
440 if (used[next]) continue;
441 final additional = sum(path.map((before) => pairwise[before][next]));
442 path.add(next);
443 used[next] = true;
444 visit(score + additional);
445 used[next] = false;
446 path.removeLast();
447 }
448 }
449 visit(0);
450 return finish(standings, firsts.length > 1, ['score: $best', 'optimal: $optimal', 'ranking: ${ranking(standings)}']);
451 }
452 TallyResult stv(int seats) {
453 final ballots = groups.where((group) => group.ranking.isNotEmpty).toList();
454 final perVote = List.filled(ballots.length, 100);
455 final quota = sum(ballots.map((group) => group.votes)) * 100 ~/ (seats + 1) + 1;
456 final remaining = field.toList(), elected = <int>[];
457 final history = <List<int>>[], lines = ['quota: ${amount(quota)}'];
458 var round = 0;
459 while (true) {
460 final current = List.filled(candidateCount, 0);
461 final assignments = List.filled(ballots.length, -1);
462 var exhausted = 0;
463 for (var i = 0; i < ballots.length; i++) {
464 for (final candidate in ballots[i].ranking) {
465 if (remaining.contains(candidate)) { assignments[i] = candidate; break; }
466 }
467 final value = ballots[i].votes * perVote[i];
468 if (assignments[i] < 0) { exhausted += value; }
469 else { current[assignments[i]] += value; }
470 }
471 var line = 'round ${++round}: ${values(remaining, current, true)}; exhausted ${amount(exhausted)}';
472 if (remaining.length == seats - elected.length) {
473 final added = order(remaining, current);
474 elected.addAll(added);
475 lines.add('$line; elected ${names(added)}');
476 break;
477 }
478 final eligible = order(remaining.where((c) => current[c] >= quota), current);
479 if (eligible.isNotEmpty) {
480 final winner = eligible.first, surplus = current[winner] - quota;
481 elected.add(winner);
482 remaining.remove(winner);
483 for (var i = 0; i < ballots.length; i++) {
484 if (assignments[i] == winner) perVote[i] = perVote[i] * surplus ~/ current[winner];
485 }
486 line += '; elected ${name(winner)} (surplus ${amount(surplus)})';
487 } else {
488 final (loser, tie) = eliminate(remaining, current, history);
489 remaining.remove(loser);
490 line += '; excluded ${name(loser)}${tie ? ' (tie)' : ''}';
491 }
492 lines.add(line);
493 history.add(current);
494 if (elected.length == seats) break;
495 }
496 lines.add('elected: ${names(elected)}');
497 return TallyResult([], false, lines);
498 }
499 TallyResult run(MethodSpec method) => switch (method.kind) {
500 Method.plurality => scored(scores(field, ScoreKind.first), 'votes'),
501 Method.runoff => runoff(),
502 Method.irv => instantRunoff(),
503 Method.coombs => instantRunoff(true),
504 Method.bucklin => bucklin(),
505 Method.borda => scored(scores(field, ScoreKind.borda), 'scores'),
506 Method.baldwin => bordaElimination(false),
507 Method.nanson => bordaElimination(true),
508 Method.black => black(),
509 Method.veto => scored(scores(field, ScoreKind.last), 'vetoes', false),
510 Method.condorcet => condorcetReport(),
511 Method.copeland => copeland(),
512 Method.minimax => minimax(method.minimax),
513 Method.schulze => schulze(),
514 Method.rankedPairs => rankedPairs(),
515 Method.kemeny => kemeny(),
516 Method.approval => scored(approval(method.parameter), 'approvals'),
517 Method.stv => stv(method.parameter),
518 };
519 List<String> ballots() {
520 final counts = <String, int>{};
521 var blank = 0;
522 for (final group in groups) {
523 final text = group.ranking.isEmpty ? '-' : group.ranking.map(name).join('>');
524 counts[text] = (counts[text] ?? 0) + group.votes;
525 if (group.ranking.isEmpty) blank += group.votes;
526 }
527 final ranked = counts.keys.toList()..sort((a, b) {
528 final difference = counts[b]!.compareTo(counts[a]!);
529 return difference != 0 ? difference : a.compareTo(b);
530 });
531 return ['total $total, blank $blank, distinct ${counts.length}', for (final text in ranked) '${counts[text]} $text'];
532 }
533 List<String> profile() => [for (final candidate in field) _profileRow(candidate)];
534 String _profileRow(int candidate) {
535 final positions = List.filled(field.length, 0);
536 var unranked = 0;
537 for (final group in groups) {
538 final index = group.ranking.indexOf(candidate);
539 if (index < 0) { unranked += group.votes; }
540 else { positions[index] += group.votes; }
541 }
542 return '${name(candidate)}: ${positions.join(' ')} unranked $unranked';
543 }

554final class Counter {
555 Election? current;
556 int electionCount = 0;
557 List<String> execute(String line) {
558 final words = line.split(' '), command = words.first;
559 if (command == 'CANDIDATES') {
560 final names = words.skip(1).toList();
561 if (names.length < 2) throw const CommandError('too few candidates');
562 final seen = <String>{};
563 for (final name in names) {
564 if (!seen.add(name)) throw CommandError('duplicate candidate $name');
565 }
566 final election = Election(electionCount + 1, names);
567 current = election;
568 electionCount++;
569 return ['election $electionCount: ${names.length} candidates'];
570 }
571 final election = current;
572 if (election == null) throw const CommandError('no election');
573 switch (command) {
574 case 'BALLOT':
575 final ranking = words[2] == '-' ? <int>[] : election.checkNames(words[2].split('>'));
576 election.entries.add([Ballot(int.parse(words[1]), ranking)]);
577 return [];
578 case 'RANDOM':
579 var state = int.parse(words[2]);
580 int draw() => state = 48271 * state % 2147483647;
581 final groups = <Ballot>[];
582 for (var count = int.parse(words[1]); count > 0; count--) {
583 final ranking = election.active;
584 for (var i = ranking.length - 1; i > 0; i--) {
585 final j = draw() % (i + 1), saved = ranking[i];
586 ranking[i] = ranking[j];
587 ranking[j] = saved;
588 }
589 final length = draw() % ranking.length + 1;
590 groups.add(Ballot(1, ranking.sublist(0, length)));
591 }
592 election.entries.add(groups);
593 return [];
594 case 'UNDO':
595 if (election.entries.isEmpty) throw const CommandError('nothing to undo');
596 election.entries.removeLast();
597 return [];
598 case 'WITHDRAW':
599 case 'RESTORE':
600 final candidate = election.candidate(words[1]);
601 if (command == 'WITHDRAW') {
602 if (election.withdrawn[candidate]) throw CommandError('already withdrawn ${words[1]}');
603 if (election.active.length <= 2) throw const CommandError('too few candidates');
604 election.withdrawn[candidate] = true;
605 } else {
606 if (!election.withdrawn[candidate]) throw CommandError('not withdrawn ${words[1]}');
607 election.withdrawn[candidate] = false;
608 }
609 return [];
610 case 'TIEBREAK':
611 final order = election.checkNames(words.skip(1));
612 if (order.length != election.names.length) throw const CommandError('incomplete tiebreak');
613 election.tieOrder = order;
614 return [];
615 case 'STATUS':
616 final withdrawn = [for (var i = 0; i < election.names.length; i++) if (election.withdrawn[i]) i];
617 return ['[$line]', 'election ${election.number}', 'active: ${election.nameList(election.active)}',
618 'withdrawn: ${withdrawn.isEmpty ? 'none' : election.nameList(withdrawn)}',
619 'tiebreak: ${election.nameList(election.tieOrder)}', 'ballots: ${election.total} in ${election.entries.length} entries'];
620 }
621 if (election.entries.isEmpty) throw const CommandError('no ballots');
622 final tally = Tally(election);
623 List<String> lines;
624 if (command == 'RANKING' || command == 'SPOILERS') {
625 final method = MethodSpec.parse(words, 1);
626 if (!method.rankable) throw const CommandError('not rankable');
627 if (command == 'SPOILERS' && tally.field.length < 3) throw const CommandError('too few candidates');
628 final standings = tally.run(method).standings;
629 if (command == 'RANKING') {
630 lines = [tally.ranking(standings)];
631 } else {
632 final winner = standings.first;
633 lines = ['winner: ${tally.name(winner)}'];
634 for (final excluded in tally.field) {
635 if (excluded == winner) continue;
636 final trial = Tally(election, tally.field.where((c) => c != excluded).toList());
637 final result = trial.run(method).standings.first;
638 lines.add('without ${tally.name(excluded)}: ${tally.name(result)}${result != winner ? ' (changed)' : ''}');
639 }
640 }
641 } else if (command == 'COMPARE') {
642 final first = election.candidate(words[1]);
643 if (election.withdrawn[first]) throw CommandError('withdrawn candidate ${words[1]}');
644 final second = election.candidate(words[2]);
645 if (election.withdrawn[second]) throw CommandError('withdrawn candidate ${words[2]}');
646 if (first == second) throw const CommandError('same candidate');
647 final ahead = <String>[], behind = <String>[];
648 for (final method in summaryMethods) {
649 final standings = tally.run(method).standings;
650 (standings.indexOf(first) < standings.indexOf(second) ? ahead : behind).add(method.toString());
651 }
652 final firstVotes = tally.pairwise[first][second], secondVotes = tally.pairwise[second][first];
653 lines = ['${tally.name(first)} $firstVotes, ${tally.name(second)} $secondVotes, neither ${tally.total - firstVotes - secondVotes}',
654 '${tally.name(first)} ahead: ${ahead.isEmpty ? 'none' : ahead.join(', ')}',
655 '${tally.name(second)} ahead: ${behind.isEmpty ? 'none' : behind.join(', ')}'];
656 } else if (command == 'SUMMARY') {
657 final wins = List.filled(election.names.length, 0);
658 lines = [];
659 for (final method in summaryMethods) {
660 final winner = tally.run(method).standings.first;
661 wins[winner]++;
662 lines.add('$method: ${tally.name(winner)}');
663 }
664 lines.add('wins: ${tally.values(tally.field, wins)}');
665 } else {
666 lines = switch (command) {
667 'BALLOTS' => tally.ballots(),
668 'PROFILE' => tally.profile(),
669 'PAIRWISE' => tally.pairLines(),
670 'EXHAUSTED' => tally.instantRunoff().exhaustion,
671 _ => _methodLines(tally, MethodSpec.parse(words)),
672 };
673 }
674 return ['[$line]', ...lines];
675 }
676 List<String> _methodLines(Tally tally, MethodSpec method) {
677 if (method.kind == Method.stv && method.parameter >= tally.field.length) throw const CommandError('too many seats');
678 return tally.run(method).lines;
679 }
