6struct Method
7 kind::MethodKind
8 parameter::Int
9 minimax::MinimaxKind
10end
11Method(kind::MethodKind) = Method(kind, 0, WINNING)
12const METHOD_KINDS = Dict(string(kind) => kind for kind in instances(MethodKind))
13const MINIMAX_KINDS = Dict(string(kind) => kind for kind in instances(MinimaxKind))
14const SUMMARY = Method.([PLURALITY, RUNOFF, IRV, COOMBS, BUCKLIN, BORDA, BALDWIN,
15 NANSON, BLACK, VETO, COPELAND, MINIMAX, SCHULZE, RANKEDPAIRS, KEMENY])

17function parse_method(words::AbstractVector{<:AbstractString})
18 kind = METHOD_KINDS[words[1]]
19 kind == MINIMAX && return Method(kind, 0, MINIMAX_KINDS[words[2]])
20 kind in (APPROVAL, STV) && return Method(kind, parse(Int, words[2]), WINNING)
21 Method(kind)
22end
23function method_text(method::Method)
24 method.kind == MINIMAX && return "$(method.kind) $(method.minimax)"
25 method.kind in (APPROVAL, STV) && return "$(method.kind) $(method.parameter)"
26 string(method.kind)
27end

76bit(candidate::Int) = 1 << (candidate - 1)
77field_key(field::Vector{Int}) = sum(bit, field)
78members(mask::Int, count::Int) = [a for a in 1:count if (mask & bit(a)) != 0]
79effective(ranking::Vector{Int}, field::Vector{Int}) = filter(in(field), ranking)
80name_list(election::Election, candidates::Vector{Int}, separator::String = ", ") =
81 join(election.names[candidates], separator)
82tie_mark(tie::Bool) = tie ? " (tie)" : ""
83money(value::Int) = @sprintf("%d.%02d", div(value, 100), mod(value, 100))
84function value_list(election::Election, field::Vector{Int}, scores::Vector{Int}; cents::Bool = false)
85 join(("$(election.names[a]) $(cents ? money(scores[a]) : string(scores[a]))" for a in field), ", ")

87ordered(election::Election, field::Vector{Int}, scores::Vector{Int}; descending::Bool = true) =
88 sort(field; by = a -> (descending ? -scores[a] : scores[a], election.priority[a]))
89preferred(election::Election, field::Vector{Int}) = sort(field; by = a -> election.priority[a])

114function tallies(election::Election, field::Vector{Int})
115 key = field_key(field)
116 haskey(election.cache, key) && return election.cache[key]
117 count, size = length(election.names), length(field)
118 first_votes, last_votes, borda = zeros(Int, count), zeros(Int, count), zeros(Int, count)
119 approval = zeros(Int, size, count)
120 nonblank = 0
121 for group in election.groups
122 ranking = effective(group.ranking, field)
123 if !isempty(ranking)
124 first_votes[ranking[1]] += group.votes
125 nonblank += group.votes
126 end
127 if length(ranking) == size
128 last_votes[ranking[end]] += group.votes
129 else
130 for candidate in setdiff(field, ranking)
131 last_votes[candidate] += group.votes
132 end
133 end
134 for (position, candidate) in enumerate(ranking)
135 borda[candidate] += group.votes * (size - position)
136 approval[position:size, candidate] .+= group.votes
137 end
138 end
139 result = Tallies(first_votes, last_votes, borda, approval, nonblank)
140 election.cache[key] = result
141 result
142end
143function pairwise(election::Election)
144 !isnothing(election.pairs) && return election.pairs
145 count = length(election.names)
146 result = zeros(Int, count, count)
147 for group in election.groups
148 positions = fill(count + 1, count)
149 for (position, candidate) in enumerate(group.ranking)
150 positions[candidate] = position
151 end
152 result .+= group.votes .* (positions .< transpose(positions))
153 end
154 election.pairs = result
155 result
156end
157function majority(field::Vector{Int}, scores::Vector{Int})
158 total = sum(scores[field])
159 any(a -> 2 * scores[a] > total, field)
160end
161function look_back(election::Election, field::Vector{Int}, history::Vector{Vector{Int}})
162 choices = copy(field)
163 tie = false
164 for round in length(history):-1:1
165 scores = history[round]
166 minimum_score = minimum(scores[choices])
167 filter!(a -> scores[a] == minimum_score, choices)
168 round == length(history) && (tie = length(choices) > 1)
169 end
170 Removal(choices[argmax(election.priority[choices])], tie)
171end
172function condorcet(election::Election, field::Vector{Int}; loser::Bool = false)
173 scores = pairwise(election)
174 for candidate in field
175 others = filter(!=(candidate), field)
176 found = loser ? all(b -> scores[candidate, b] < scores[b, candidate], others) :
177 all(b -> scores[candidate, b] > scores[b, candidate], others)
178 found && return candidate
179 end
180 0
181end

183function elimination(election::Election, field::Vector{Int}; coombs::Bool = false, exhausted::Bool = false)
184 remaining, removed = copy(field), Int[]
185 history, lines = Vector{Int}[], String[]
186 while true
187 scores = tallies(election, remaining)
188 round = length(lines) + 1
189 active = sum(scores.first[remaining])
190 line = exhausted ? "round $round: active $active, exhausted $(election.total - active)" :
191 "round $round: $(value_list(election, remaining, scores.first))"
192 if length(remaining) == 1 || majority(remaining, scores.first)
193 order = vcat(ordered(election, remaining, scores.first), reverse(removed))
194 push!(lines, line)
195 return exhausted ? Outcome(order, lines) : finish(election, order, false, lines)
196 end
197 if coombs
198 maximum_last = maximum(scores.last[remaining])
199 choices = filter(a -> scores.last[a] == maximum_last, remaining)
200 sort!(choices; by = a -> (scores.first[a], -election.priority[a]))
201 choice = Removal(choices[1], length(choices) > 1)
202 line *= "; last $(value_list(election, remaining, scores.last))"
203 else
204 push!(history, scores.first)
205 choice = look_back(election, remaining, history)
206 end
207 !exhausted && (line *= "; eliminated $(election.names[choice.candidate])$(tie_mark(choice.tie))")
208 push!(lines, line)
209 push!(removed, choice.candidate)
210 filter!(!=(choice.candidate), remaining)
211 end
212end
213function runoff(election::Election, field::Vector{Int})
214 first_votes = tallies(election, field).first
215 order = ordered(election, field, first_votes)
216 lines = ["round 1: $(value_list(election, field, first_votes))"]
217 tie = false
218 if !majority(field, first_votes)
219 finalists = sort(order[1:2])
220 advance_tie = length(order) >= 3 && first_votes[order[2]] == first_votes[order[3]]
221 second = tallies(election, finalists).first
222 tie = second[order[1]] == second[order[2]]
223 if second[order[2]] > second[order[1]]
224 order[1], order[2] = order[2], order[1]
225 end
226 push!(lines, "finalists: $(name_list(election, finalists))$(tie_mark(advance_tie))")
227 push!(lines, "round 2: $(value_list(election, finalists, second))")
228 end
229 finish(election, order, tie, lines)
230end
231function bucklin(election::Election, field::Vector{Int})
232 scores = tallies(election, field)
233 line = div(scores.nonblank, 2) + 1
234 lines = ["majority: $line"]
235 final_depth = length(field)
236 for depth in eachindex(field)
237 push!(lines, "round $depth: $(value_list(election, field, scores.approval[depth, :]))")
238 if any(a -> scores.approval[depth, a] >= line, field)
239 final_depth = depth
240 break
241 end
242 end
243 function before(a, b)
244 for depth in final_depth:-1:1
245 scores.approval[depth, a] != scores.approval[depth, b] &&
246 return scores.approval[depth, a] > scores.approval[depth, b]
247 end
248 election.priority[a] < election.priority[b]
249 end
250 order = sort(field; lt = before)
251 finish(election, order, scores.approval[final_depth, order[1]] == scores.approval[final_depth, order[2]], lines)
252end
253function borda_elimination(election::Election, field::Vector{Int}; nanson::Bool = false)
254 remaining, tail = copy(field), Int[]
255 history, lines = Vector{Int}[], String[]
256 while length(remaining) >= 2
257 scores = tallies(election, remaining).borda
258 line = "round $(length(lines) + 1): $(value_list(election, remaining, scores))"
259 if nanson
260 total = sum(scores[remaining])
261 below = filter(a -> scores[a] * length(remaining) < total, remaining)
262 if isempty(below)
263 push!(lines, line)
264 break
265 end
266 line *= "; eliminated $(name_list(election, below))"
267 removed = ordered(election, below, scores)
268 else
269 push!(history, scores)
270 choice = look_back(election, remaining, history)
271 removed = [choice.candidate]
272 line *= "; eliminated $(election.names[choice.candidate])$(tie_mark(choice.tie))"
273 end
274 push!(lines, line)
275 remaining = setdiff(remaining, removed)
276 tail = vcat(removed, tail)
277 end
278 order = vcat(preferred(election, remaining), tail)
279 finish(election, order, nanson && length(remaining) > 1, lines)
280end

282function pair_lines(election::Election, field::Vector{Int}, scores::Matrix{Int})
283 lines = String[]
284 for (index, first_candidate) in enumerate(field), second_candidate in field[(index + 1):end]
285 forward, backward = scores[first_candidate, second_candidate], scores[second_candidate, first_candidate]
286 result = forward == backward ? "tie" : election.names[forward > backward ? first_candidate : second_candidate]
287 push!(lines, "$(election.names[first_candidate]) vs $(election.names[second_candidate]): $forward-$backward ($result)")
288 end
289 lines
290end
291function closure!(field::Vector{Int}, relation::AbstractMatrix{Bool})
292 for via in field, first_candidate in field, second_candidate in field
293 relation[first_candidate, second_candidate] |= relation[first_candidate, via] && relation[via, second_candidate]
294 end
295 relation
296end
297function condorcet_report(election::Election, field::Vector{Int})
298 scores = pairwise(election)
299 winner, loser = condorcet(election, field), condorcet(election, field; loser = true)
300 weak = closure!(field, scores .>= transpose(scores))
301 strong = closure!(field, scores .> transpose(scores))
302 smith = filter(a -> all(b -> weak[a, b], field), field)
303 schwartz = filter(a -> all(b -> !strong[b, a] || strong[a, b], field), field)
304 Outcome(Int[], ["winner: $(winner == 0 ? "NONE" : election.names[winner])",
305 "loser: $(loser == 0 ? "NONE" : election.names[loser])",
306 "smith: $(name_list(election, smith))", "schwartz: $(name_list(election, schwartz))"])
307end
308function topological(election::Election, field::Vector{Int}, relation::AbstractMatrix{Bool})
309 remaining, order = copy(field), Int[]
310 tie = false
311 while !isempty(remaining)
312 choices = filter(a -> all(b -> a == b || !relation[b, a], remaining), remaining)
313 isempty(order) && (tie = length(choices) > 1)
314 chosen = choices[argmin(election.priority[choices])]
315 push!(order, chosen)
316 filter!(!=(chosen), remaining)
317 end
318 order, tie
319end
320function schulze(election::Election, field::Vector{Int})
321 scores = pairwise(election)
322 paths = zeros(Int, length(election.names), length(election.names))
323 for a in field, b in field
324 scores[a, b] > scores[b, a] && (paths[a, b] = scores[a, b])
325 end
326 for via in field, a in field, b in field
327 if a != via && b != via && a != b
328 paths[a, b] = max(paths[a, b], min(paths[a, via], paths[via, b]))
329 end
330 end
331 order, tie = topological(election, field, paths .> transpose(paths))
332 lines = pair_lines(election, field, paths)
333 push!(lines, "ranking: $(name_list(election, order, " > "))")
334 finish(election, order, tie, lines)
335end
336function ranked_pairs(election::Election, field::Vector{Int})
337 scores = pairwise(election)
338 pairs = [(a, b) for a in field for b in field if scores[a, b] > scores[b, a]]
339 sort!(pairs; by = p -> (-scores[p[1], p[2]], scores[p[2], p[1]], election.priority[p[1]], election.priority[p[2]]))
340 locked, reached = falses(size(scores)), falses(size(scores))
341 lines = String[]
342 for (a, b) in pairs
343 accepted = !reached[b, a]
344 push!(lines, "$(accepted ? "lock" : "skip") $(election.names[a])>$(election.names[b]) $(scores[a, b])-$(scores[b, a])")
345 if accepted
346 locked[a, b] = reached[a, b] = true
347 closure!(field, reached)
348 end
349 end
350 order, tie = topological(election, field, locked)
351 finish(election, order, tie, lines)
352end
353function kemeny(election::Election, field::Vector{Int})
354 scores = pairwise(election)
355 count = length(election.names)
356 limit = 1 << count
357 best, ways, heads = zeros(Int, limit), zeros(Int, limit), zeros(Int, limit)
358 ways[1] = 1
359 mask = field_key(field)
360 for subset in 1:(limit - 1)
361 (subset & ~mask) != 0 && continue
362 candidates = members(subset, count)
363 best[subset + 1] = -1
364 for candidate in candidates
365 rest = subset - bit(candidate)
366 score = best[rest + 1] + sum((scores[candidate, b] for b in candidates if b != candidate); init = 0)
367 if score > best[subset + 1]
368 best[subset + 1], ways[subset + 1], heads[subset + 1] = score, 0, 0
369 end
370 if score == best[subset + 1]
371 ways[subset + 1] += ways[rest + 1]
372 heads[subset + 1] |= bit(candidate)
373 end
374 end
375 end
376 remaining, order = mask, Int[]
377 while remaining != 0
378 chosen = first(a for a in election.tiebreak if (heads[remaining + 1] & bit(a)) != 0)
379 push!(order, chosen)
380 remaining -= bit(chosen)
381 end
382 lines = ["score: $(best[mask + 1])", "optimal: $(ways[mask + 1])", "ranking: $(name_list(election, order, " > "))"]
383 finish(election, order, count_ones(heads[mask + 1]) > 1, lines)
384end
385function stv(election::Election, field::Vector{Int}, seats::Int)
386 values = [isempty(effective(group.ranking, field)) ? 0 : 100 for group in election.groups]
387 quota = div(100 * tallies(election, field).nonblank, seats + 1) + 1
388 continuing, elected = copy(field), Int[]
389 history, lines = Vector{Int}[], ["quota: $(money(quota))"]
390 while true
391 scores, owners = zeros(Int, length(election.names)), zeros(Int, length(election.groups))
392 exhausted = 0
393 for (index, group) in enumerate(election.groups)
394 values[index] == 0 && continue
395 ranking = effective(group.ranking, continuing)
396 amount = group.votes * values[index]
397 if isempty(ranking)
398 exhausted += amount
399 else
400 owners[index] = ranking[1]
401 scores[ranking[1]] += amount
402 end
403 end
404 push!(history, scores)
405 line = "round $(length(history)): $(value_list(election, continuing, scores; cents = true)); exhausted $(money(exhausted))"
406 order = ordered(election, continuing, scores)
407 if length(continuing) == seats - length(elected)
408 append!(elected, order)
409 push!(lines, line * "; elected $(name_list(election, order))")
410 break
411 end
412 chosen = order[1]
413 if scores[chosen] >= quota
414 surplus = scores[chosen] - quota
415 push!(elected, chosen)
416 filter!(!=(chosen), continuing)
417 for index in eachindex(values)
418 owners[index] == chosen && (values[index] = div(values[index] * surplus, scores[chosen]))
419 end
420 push!(lines, line * "; elected $(election.names[chosen]) (surplus $(money(surplus)))")
421 length(elected) == seats && break
422 else
423 choice = look_back(election, continuing, history)
424 filter!(!=(choice.candidate), continuing)
425 push!(lines, line * "; excluded $(election.names[choice.candidate])$(tie_mark(choice.tie))")
426 end
427 end
428 push!(lines, "elected: $(name_list(election, elected))")
429 Outcome(Int[], lines)
430end

432function tally(election::Election, field::Vector{Int}, method::Method)
433 kind = method.kind
434 kind == RUNOFF && return runoff(election, field)
435 kind in (IRV, COOMBS) && return elimination(election, field; coombs = kind == COOMBS)
436 kind == BUCKLIN && return bucklin(election, field)
437 kind in (BALDWIN, NANSON) && return borda_elimination(election, field; nanson = kind == NANSON)
438 kind == CONDORCET && return condorcet_report(election, field)
439 kind == SCHULZE && return schulze(election, field)
440 kind == RANKEDPAIRS && return ranked_pairs(election, field)
441 kind == KEMENY && return kemeny(election, field)
442 kind == STV && return stv(election, field, method.parameter)
443 base = tallies(election, field)
444 scores, lines = zeros(Int, length(election.names)), String[]
445 label, descending, winner = "scores", true, 0
446 if kind == PLURALITY
447 scores, label = base.first, "votes"
448 elseif kind in (BORDA, BLACK)
449 scores = base.borda
450 if kind == BLACK
451 winner = condorcet(election, field)
452 push!(lines, "condorcet: $(winner == 0 ? "NONE" : election.names[winner])")
453 end
454 elseif kind == VETO
455 scores, label, descending = base.last, "vetoes", false
456 elseif kind == APPROVAL
457 scores, label = base.approval[min(method.parameter, length(field)), :], "approvals"
458 elseif kind == COPELAND
459 matrix = pairwise(election)
460 parts = String[]
461 for a in field
462 others = filter(!=(a), field)
463 wins = count(b -> matrix[a, b] > matrix[b, a], others)
464 draws = count(b -> matrix[a, b] == matrix[b, a], others)
465 losses = count(b -> matrix[a, b] < matrix[b, a], others)
466 scores[a] = 2 * wins + draws
467 push!(parts, "$(election.names[a]) $(div(scores[a], 2)).$(mod(scores[a], 2) * 5) ($wins-$draws-$losses)")
468 end
469 push!(lines, "scores: $(join(parts, ", "))")
470 elseif kind == MINIMAX
471 matrix = pairwise(election)
472 descending = false
473 for a in field, b in field
474 a == b && continue
475 opposition, support = matrix[b, a], matrix[a, b]
476 value = method.minimax == OPPOSITION ? opposition :
477 opposition <= support ? 0 : method.minimax == MARGIN ? opposition - support : opposition
478 scores[a] = max(scores[a], value)
479 end
480 end
481 order = ordered(election, field, scores; descending)
482 tie = scores[order[1]] == scores[order[2]]
483 if winner != 0
484 filter!(!=(winner), order)
485 pushfirst!(order, winner)
486 tie = false
487 end
488 kind != COPELAND && push!(lines, "$label: $(value_list(election, field, scores))")
489 finish(election, order, tie, lines)
490end
491function ballots_report(election::Election)
492 counts = Dict{String,Int}()
493 field = findall(election.active)
494 for group in election.groups
495 ranking = effective(group.ranking, field)
496 notation = isempty(ranking) ? "-" : name_list(election, ranking, ">")
497 counts[notation] = get(counts, notation, 0) + group.votes
498 end
499 order = sort!(collect(keys(counts)); by = text -> (-counts[text], text))
500 lines = ["total $(election.total), blank $(get(counts, "-", 0)), distinct $(length(counts))"]
501 append!(lines, ["$(counts[text]) $text" for text in order])
502end

504function execute!(lab::Laboratory, line::String)
505 words = split(line)
506 command = words[1]
507 if command == "CANDIDATES"
508 names = String.(words[2:end])
509 length(names) == 1 && fail("too few candidates")
510 seen = Set{String}()
511 for name in names
512 name in seen && fail("duplicate candidate $name")
513 push!(seen, name)
514 end
515 election = Election(lab.number + 1, names)
516 lab.number += 1
517 lab.election = election
518 return ["election $(lab.number): $(length(names)) candidates"]
519 end
520 election = lab.election
521 isnothing(election) && fail("no election")
522 if command == "BALLOT"
523 ranking = words[3] == "-" ? Int[] : checked_names(election, split(words[3], '>'))
524 votes = parse(Int, words[2])
525 push!(election.groups, Ballot(votes, ranking))
526 election.total += votes
527 finish_entry!(election)
528 return String[]
529 elseif command == "RANDOM"
530 random_count = parse(Int, words[2])
531 state = parse(Int64, words[3])
532 for _ in 1:random_count
533 ranking = findall(election.active)
534 for size in length(ranking):-1:2
535 state = mod(48271 * state, 2147483647)
536 index = Int(mod(state, size)) + 1
537 ranking[size], ranking[index] = ranking[index], ranking[size]
538 end
539 state = mod(48271 * state, 2147483647)
540 ranking = ranking[1:(Int(mod(state, length(ranking))) + 1)]
541 push!(election.groups, Ballot(1, ranking))
542 end
543 election.total += random_count
544 finish_entry!(election)
545 return String[]
546 elseif command == "UNDO"
547 isempty(election.entry_ends) && fail("nothing to undo")
548 pop!(election.entry_ends)
549 stop = isempty(election.entry_ends) ? 0 : election.entry_ends[end]
550 for group in @view election.groups[(stop + 1):end]
551 election.total -= group.votes
552 end
553 resize!(election.groups, stop)
554 empty!(election.cache)
555 election.pairs = nothing
556 return String[]
557 elseif command in ("WITHDRAW", "RESTORE")
558 candidate = candidate_index(election, words[2])
559 if command == "WITHDRAW"
560 !election.active[candidate] && fail("already withdrawn $(words[2])")
561 count(election.active) <= 2 && fail("too few candidates")
562 election.active[candidate] = false
563 else
564 election.active[candidate] && fail("not withdrawn $(words[2])")
565 election.active[candidate] = true
566 end
567 return String[]
568 elseif command == "TIEBREAK"
569 order = checked_names(election, words[2:end])
570 length(order) != length(election.names) && fail("incomplete tiebreak")
571 election.tiebreak = order
572 election.priority = invperm(order)
573 return String[]
574 end
575 heading = "[$line]"
576 field = findall(election.active)
577 if command == "STATUS"
578 withdrawn = findall(!, election.active)
579 return [heading, "election $(election.number)", "active: $(name_list(election, field))",
580 "withdrawn: $(isempty(withdrawn) ? "none" : name_list(election, withdrawn))",
581 "tiebreak: $(name_list(election, election.tiebreak))",
582 "ballots: $(election.total) in $(length(election.entry_ends)) entries"]
583 end
584 isempty(election.entry_ends) && fail("no ballots")
585 if command in ("RANKING", "SPOILERS")
586 method = parse_method(words[2:end])
587 method.kind in (CONDORCET, STV) && fail("not rankable")
588 command == "SPOILERS" && length(field) < 3 && fail("too few candidates")
589 result = tally(election, field, method)
590 command == "RANKING" && return [heading, name_list(election, result.order, " > ")]
591 winner = result.order[1]
592 lines = [heading, "winner: $(election.names[winner])"]
593 for candidate in field
594 candidate == winner && continue
595 other = tally(election, filter(!=(candidate), field), method).order[1]
596 push!(lines, "without $(election.names[candidate]): $(election.names[other])$(other == winner ? "" : " (changed)")")
597 end
598 return lines
599 end
600 compared = Int[]
601 if command == "COMPARE"
602 for name in words[2:end]
603 candidate = candidate_index(election, name)
604 !election.active[candidate] && fail("withdrawn candidate $name")
605 push!(compared, candidate)
606 end
607 compared[1] == compared[2] && fail("same candidate")
608 end
609 command == "STV" && parse(Int, words[2]) >= length(field) && fail("too many seats")
610 lines = [heading]
611 if command == "BALLOTS"
612 append!(lines, ballots_report(election))
613 elseif command == "PROFILE"
614 scores = tallies(election, field)
615 for candidate in field
616 cumulative = scores.approval[:, candidate]
617 counts = diff(vcat(0, cumulative))
618 push!(lines, "$(election.names[candidate]): $(join(counts, " ")) unranked $(election.total - cumulative[end])")
619 end
620 elseif command == "PAIRWISE"
621 append!(lines, pair_lines(election, field, pairwise(election)))
622 elseif command == "EXHAUSTED"
623 append!(lines, elimination(election, field; exhausted = true).lines)
624 elseif command in ("SUMMARY", "COMPARE")
625 wins = zeros(Int, length(election.names))
626 ahead = [String[], String[]]
627 for method in SUMMARY
628 result = tally(election, field, method)
629 if command == "SUMMARY"
630 winner = result.order[1]
631 wins[winner] += 1
632 push!(lines, "$(method_text(method)): $(election.names[winner])")
633 else
634 first_position = findfirst(==(compared[1]), result.order)
635 second_position = findfirst(==(compared[2]), result.order)
636 push!(ahead[first_position < second_position ? 1 : 2], method_text(method))
637 end
638 end
639 if command == "SUMMARY"
640 push!(lines, "wins: $(value_list(election, field, wins))")
641 else
642 matrix = pairwise(election)
643 a, b = compared
644 push!(lines, "$(election.names[a]) $(matrix[a, b]), $(election.names[b]) $(matrix[b, a]), neither $(election.total - matrix[a, b] - matrix[b, a])")
645 for side in 1:2
646 text = isempty(ahead[side]) ? "none" : join(ahead[side], ", ")
647 push!(lines, "$(election.names[compared[side]]) ahead: $text")
648 end
649 end
650 else
651 append!(lines, tally(election, field, parse_method(words)).lines)
652 end
653 lines
654end
