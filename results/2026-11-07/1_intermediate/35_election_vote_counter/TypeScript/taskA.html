52function parseMethod(words: readonly string[]): Method {
53 switch (words[0]) {
54 case 'MINIMAX': return { kind: 'MINIMAX', metric: words[1] as MinimaxKind };
55 case 'APPROVAL': return { kind: 'APPROVAL', count: Number(words[1]) };
56 case 'STV': return { kind: 'STV', seats: Number(words[1]) };
57 case 'CONDORCET': return { kind: 'CONDORCET' };
58 default: return { kind: words[0] as PlainMethod };
59 }

61function methodText(method: Method): string {
62 switch (method.kind) {
63 case 'MINIMAX': return 'MINIMAX ' + method.metric;
64 case 'APPROVAL': return 'APPROVAL ' + method.count;
65 case 'STV': return 'STV ' + method.seats;
66 default: return method.kind;
67 }

69function rankable(method: Method): method is RankedMethod {
70 return method.kind !== 'CONDORCET' && method.kind !== 'STV';

72function parseCommand(line: string): Command {
73 const [kind, ...args] = line.split(' ');
74 switch (kind) {
75 case 'CANDIDATES': case 'TIEBREAK': return { kind, names: args };
76 case 'BALLOT': return { kind, votes: Number(args[0]), names: args[1] === '-' ? [] : args[1].split('>') };
77 case 'RANDOM': return { kind, count: Number(args[0]), seed: Number(args[1]) };
78 case 'UNDO': return { kind };
79 case 'WITHDRAW': case 'RESTORE': return { kind, name: args[0] };
80 case 'STATUS': case 'BALLOTS': case 'PROFILE': case 'PAIRWISE': case 'EXHAUSTED': case 'SUMMARY': return { kind };
81 case 'COMPARE': return { kind, names: [args[0], args[1]] };
82 case 'RANKING': case 'SPOILERS': return { kind, method: parseMethod(args) };
83 default: return { kind: 'TALLY', method: parseMethod([kind, ...args]) };
84 }

87class Election {
88 readonly names: readonly string[];
89 readonly number: number;
90 readonly withdrawn = new Set<number>();
91 readonly entries: (readonly Ballot[])[] = [];
92 private readonly cache = new Map<number, Aggregate>();
93 priority: Field;

96 constructor(names: readonly string[], number: number) {
97 this.names = names;
98 this.number = number;
99 this.priority = names.map((_, index) => index).sort((a, b) => ascii(names[a], names[b]));
100 this.tieRank = names.map((_, candidate) => this.priority.indexOf(candidate));
101 }
102 zeros(): number[] { return this.names.map(() => 0); }
103 matrix(): number[][] { return this.names.map(() => this.zeros()); }
104 active(): number[] { return this.names.map((_, index) => index).filter(index => !this.withdrawn.has(index)); }
105 candidate(name: string): number {
106 const candidate = this.names.indexOf(name);
107 if (candidate < 0) fail('unknown candidate ' + name);
108 return candidate;
109 }
110 checkedNames(names: readonly string[]): number[] {
111 const result: number[] = [];
112 for (const name of names) {
113 const candidate = this.candidate(name);
114 if (result.includes(candidate)) fail('duplicate candidate ' + name);
115 result.push(candidate);
116 }
117 return result;
118 }
119 setPriority(order: Field): void {
120 this.priority = order;
121 this.tieRank = this.names.map((_, candidate) => order.indexOf(candidate));
122 }
123 addEntry(groups: readonly Ballot[]): void { this.entries.push(groups); this.cache.clear(); }
124 undo(): void {
125 if (this.entries.length === 0) fail('nothing to undo');
126 this.entries.pop();
127 this.cache.clear();
128 }
129 list(field: Field, separator = ', '): string { return field.map(candidate => this.names[candidate]).join(separator); }
130 values(field: Field, values: Scores, format: (value: number) => string = String): string {
131 return field.map(candidate => this.names[candidate] + ' ' + format(values[candidate])).join(', ');
132 }
133 ordered(field: Field, scores: Scores, ascending = false): number[] {
134 const direction = ascending ? 1 : -1;
135 return field.slice().sort((a, b) => direction * (scores[a] - scores[b]) || this.tieRank[a] - this.tieRank[b]);
136 }
137 preferred(field: Field): number[] { return field.slice().sort((a, b) => this.tieRank[a] - this.tieRank[b]); }

139 aggregate(field: Field): Aggregate {
140 const mask = field.reduce((bits, candidate) => bits | (1 << candidate), 0);
141 const cached = this.cache.get(mask);
142 if (cached !== undefined) return cached;
143 const combined = new Map<string, { votes: number; ranking: number[] }>();
144 for (const entry of this.entries) for (const group of entry) {
145 const ranking = group.ranking.filter(candidate => (mask & (1 << candidate)) !== 0);
146 const key = ranking.join(',');
147 const previous = combined.get(key);
148 if (previous === undefined) combined.set(key, { votes: group.votes, ranking });
149 else previous.votes += group.votes;
150 }
151 const groups = [...combined.values()];
152 const first = this.zeros(), last = this.zeros(), borda = this.zeros();
153 const positions = this.names.map(() => field.map(() => 0)), pairwise = this.matrix();
154 let total = 0, blank = 0;
155 for (const group of groups) {
156 const { votes, ranking } = group;
157 total += votes;
158 if (ranking.length === 0) blank += votes;
159 else first[ranking[0]] += votes;
160 const ranks = this.names.map(() => field.length);
161 ranking.forEach((candidate, index) => {
162 ranks[candidate] = index;
163 positions[candidate][index] += votes;
164 borda[candidate] += votes * (field.length - index - 1);
165 });
166 for (const candidate of field) {
167 if (ranking.length < field.length ? ranks[candidate] === field.length : candidate === ranking.at(-1)) last[candidate] += votes;
168 for (const other of field) if (ranks[candidate] < ranks[other]) pairwise[candidate][other] += votes;
169 }
170 }
171 const result: Aggregate = { groups, first, last, borda, positions, pairwise, total, blank };
172 this.cache.set(mask, result);
173 return result;
174 }
175}

177function eliminate(election: Election, field: Field, scores: Scores, history: readonly Scores[]): Removal {
178 const minimum = Math.min(...field.map(candidate => scores[candidate]));
179 let choices = field.filter(candidate => scores[candidate] === minimum);
180 const tie = choices.length > 1;
181 for (let round = history.length - 1; round >= 0 && choices.length > 1; --round) {
182 const previous = history[round];
183 const low = Math.min(...choices.map(candidate => previous[candidate]));
184 choices = choices.filter(candidate => previous[candidate] === low);
185 }
186 return { candidate: election.preferred(choices).at(-1)!, tie };
187}
188function finish(election: Election, standings: Field, tie: boolean, lines: readonly string[]): RankedResult {
189 return { standings, lines: [...lines, 'winner: ' + election.names[standings[0]] + tieMark(tie)] };
190}
191function scoredResult(election: Election, field: Field, scores: Scores, label: string, ascending = false): RankedResult {
192 const order = election.ordered(field, scores, ascending);
193 return finish(election, order, scores[order[0]] === scores[order[1]], [label + ': ' + election.values(field, scores)]);
194}
195function pairLines(election: Election, field: Field, matrix: Matrix): string[] {
196 const lines: string[] = [];
197 for (let i = 0; i < field.length; ++i) for (let j = i + 1; j < field.length; ++j) {
198 const a = field[i], b = field[j], forward = matrix[a][b], backward = matrix[b][a];
199 const result = forward === backward ? 'tie' : election.names[forward > backward ? a : b];
200 lines.push(election.names[a] + ' vs ' + election.names[b] + ': ' + forward + '-' + backward + ' (' + result + ')');
201 }
202 return lines;
203}
204function runoff(election: Election, field: Field): RankedResult {
205 const first = election.aggregate(field).first, order = election.ordered(field, first);
206 const lines = ['round 1: ' + election.values(field, first)];
207 if (2 * first[order[0]] > sum(field.map(a => first[a]))) return finish(election, order, false, lines);
208 const finalists = order.slice(0, 2).sort((a, b) => a - b), second = election.aggregate(finalists).first;
209 const advanceTie = order.length >= 3 && first[order[1]] === first[order[2]];
210 lines.push('finalists: ' + election.list(finalists) + tieMark(advanceTie), 'round 2: ' + election.values(finalists, second));
211 const winners = finalists.slice().sort((a, b) => second[b] - second[a] || first[b] - first[a] || election.tieRank[a] - election.tieRank[b]);
212 return finish(election, [...winners, ...order.slice(2)], second[winners[0]] === second[winners[1]], lines);
213}
214function eliminationRounds(election: Election, field: Field, coombs = false): RankedResult & { readonly exhausted: readonly string[] } {
215 let remaining = field.slice();
216 const eliminated: number[] = [], history: Scores[] = [], lines: string[] = [], exhausted: string[] = [];
217 for (let round = 1; ; ++round) {
218 const aggregate = election.aggregate(remaining), scores = aggregate.first;
219 const order = election.ordered(remaining, scores), active = sum(remaining.map(a => scores[a]));
220 let line = 'round ' + round + ': ' + election.values(remaining, scores);
221 exhausted.push('round ' + round + ': active ' + active + ', exhausted ' + (aggregate.total - active));
222 if (remaining.length === 1 || 2 * scores[order[0]] > active) {
223 return { ...finish(election, [...order, ...eliminated.reverse()], false, [...lines, line]), exhausted };
224 }
225 let choice: Removal;
226 if (coombs) {
227 const maximum = Math.max(...remaining.map(a => aggregate.last[a]));
228 const tied = remaining.filter(a => aggregate.last[a] === maximum);
229 tied.sort((a, b) => scores[a] - scores[b] || election.tieRank[b] - election.tieRank[a]);
230 choice = { candidate: tied[0], tie: tied.length > 1 };
231 line += '; last ' + election.values(remaining, aggregate.last);
232 } else choice = eliminate(election, remaining, scores, history);
233 lines.push(line + '; eliminated ' + election.names[choice.candidate] + tieMark(choice.tie));
234 history.push(scores);
235 eliminated.push(choice.candidate);
236 remaining = remaining.filter(a => a !== choice.candidate);
237 }

239function bucklin(election: Election, field: Field): RankedResult {
240 const aggregate = election.aggregate(field), majority = Math.floor((aggregate.total - aggregate.blank) / 2) + 1;
241 const history: Scores[] = [], lines = ['majority: ' + majority];
242 for (let depth = 1; depth <= field.length; ++depth) {
243 const scores = aggregate.positions.map(row => sum(row.slice(0, depth)));
244 history.push(scores);
245 lines.push('round ' + depth + ': ' + election.values(field, scores));
246 if (field.some(a => scores[a] >= majority)) break;
247 }
248 const order = field.slice().sort((a, b) => {
249 for (let round = history.length - 1; round >= 0; --round) {
250 const scores = history[round];
251 if (scores[a] !== scores[b]) return scores[b] - scores[a];
252 }
253 return election.tieRank[a] - election.tieRank[b];
254 });
255 const final = history.at(-1)!;
256 return finish(election, order, final[order[0]] === final[order[1]], lines);
257}
258function bordaRounds(election: Election, field: Field, nanson: boolean): RankedResult {
259 let remaining = field.slice();
260 const eliminated: number[][] = [], history: Scores[] = [], lines: string[] = [];
261 for (let round = 1; remaining.length >= 2; ++round) {
262 const scores = election.aggregate(remaining).borda;
263 let line = 'round ' + round + ': ' + election.values(remaining, scores);
264 if (nanson) {
265 const total = sum(remaining.map(a => scores[a]));
266 const removed = remaining.filter(a => scores[a] * remaining.length < total);
267 if (removed.length === 0) { lines.push(line); break; }
268 line += '; eliminated ' + election.list(removed);
269 eliminated.unshift(election.ordered(removed, scores));
270 remaining = remaining.filter(a => !removed.includes(a));
271 } else {
272 const choice = eliminate(election, remaining, scores, history);
273 line += '; eliminated ' + election.names[choice.candidate] + tieMark(choice.tie);
274 eliminated.unshift([choice.candidate]);
275 remaining = remaining.filter(a => a !== choice.candidate);
276 }
277 history.push(scores);
278 lines.push(line);
279 }
280 return finish(election, [...election.preferred(remaining), ...eliminated.flat()], nanson && remaining.length > 1, lines);
281}
282function condorcetWinner(field: Field, pairwise: Matrix, loser = false): number | undefined {
283 return field.find(a => field.every(b => a === b || (loser ? pairwise[a][b] < pairwise[b][a] : pairwise[a][b] > pairwise[b][a])));
284}
285function closure(field: Field, edges: readonly (readonly boolean[])[]): boolean[][] {
286 const reach = edges.map(row => row.slice());
287 for (const via of field) for (const a of field) for (const b of field) reach[a][b] ||= reach[a][via] && reach[via][b];
288 return reach;
289}
290function condorcet(election: Election, field: Field, pairwise: Matrix): Result {
291 const winner = condorcetWinner(field, pairwise), loser = condorcetWinner(field, pairwise, true);
292 const weak = election.names.map(() => election.names.map(() => false)), strong = weak.map(row => row.slice());
293 for (const a of field) for (const b of field) {
294 weak[a][b] = pairwise[a][b] >= pairwise[b][a];
295 strong[a][b] = pairwise[a][b] > pairwise[b][a];
296 }
297 const weakReach = closure(field, weak), strongReach = closure(field, strong);
298 const smith = field.filter(a => field.every(b => weakReach[a][b]));
299 const schwartz = field.filter(a => field.every(b => !strongReach[b][a] || strongReach[a][b]));
300 return { lines: ['winner: ' + (winner === undefined ? 'NONE' : election.names[winner]),
301 'loser: ' + (loser === undefined ? 'NONE' : election.names[loser]),
302 'smith: ' + election.list(smith), 'schwartz: ' + election.list(schwartz)] };
303}
304function topological(election: Election, field: Field, beats: (a: number, b: number) => boolean): { standings: number[]; tie: boolean } {
305 let remaining = field.slice(), tie = false;
306 const standings: number[] = [];
307 while (remaining.length > 0) {
308 const choices = remaining.filter(a => remaining.every(b => a === b || !beats(b, a)));
309 if (standings.length === 0) tie = choices.length > 1;
310 const chosen = election.preferred(choices)[0];
311 standings.push(chosen);
312 remaining = remaining.filter(a => a !== chosen);
313 }
314 return { standings, tie };
315}
316function schulze(election: Election, field: Field, pairwise: Matrix): RankedResult {
317 const paths = election.matrix();
318 for (const a of field) for (const b of field) if (pairwise[a][b] > pairwise[b][a]) paths[a][b] = pairwise[a][b];
319 for (const via of field) for (const a of field) if (a !== via) {
320 for (const b of field) if (b !== via && b !== a) paths[a][b] = Math.max(paths[a][b], Math.min(paths[a][via], paths[via][b]));
321 }
322 const { standings, tie } = topological(election, field, (a, b) => paths[a][b] > paths[b][a]);
323 return finish(election, standings, tie, [...pairLines(election, field, paths), 'ranking: ' + election.list(standings, ' > ')]);
324}
325function rankedPairs(election: Election, field: Field, pairwise: Matrix): RankedResult {
326 const pairs: [number, number][] = [], edges = election.names.map(() => election.names.map(() => false)), lines: string[] = [];
327 for (const a of field) for (const b of field) if (pairwise[a][b] > pairwise[b][a]) pairs.push([a, b]);
328 pairs.sort(([a, b], [c, d]) => pairwise[c][d] - pairwise[a][b] || pairwise[b][a] - pairwise[d][c] || election.tieRank[a] - election.tieRank[c] || election.tieRank[b] - election.tieRank[d]);
329 for (const [a, b] of pairs) {
330 const lock = !closure(field, edges)[b][a];
331 if (lock) edges[a][b] = true;
332 lines.push((lock ? 'lock ' : 'skip ') + election.names[a] + '>' + election.names[b] + ' ' + pairwise[a][b] + '-' + pairwise[b][a]);
333 }
334 const { standings, tie } = topological(election, field, (a, b) => edges[a][b]);
335 return finish(election, standings, tie, lines);
336}
337function kemeny(election: Election, field: Field, pairwise: Matrix): RankedResult {
338 interface Optimum { readonly score: number; readonly count: number; readonly first: Field }
339 const memo = new Map<number, Optimum>([[0, { score: 0, count: 1, first: [] }]]);
340 function solve(mask: number): Optimum {
341 const cached = memo.get(mask);
342 if (cached !== undefined) return cached;
343 let score = -1, count = 0;
344 let first: number[] = [];
345 for (const candidate of field) if ((mask & (1 << candidate)) !== 0) {
346 const rest = mask ^ (1 << candidate), suffix = solve(rest);
347 const value = suffix.score + sum(field.filter(b => (rest & (1 << b)) !== 0).map(b => pairwise[candidate][b]));
348 if (value > score) { score = value; count = suffix.count; first = [candidate]; }
349 else if (value === score) { count += suffix.count; first.push(candidate); }
350 }
351 const result: Optimum = { score, count, first };
352 memo.set(mask, result);
353 return result;
354 }
355 let mask = field.reduce((bits, candidate) => bits | (1 << candidate), 0);
356 const optimum = solve(mask), standings: number[] = [];
357 while (mask !== 0) {
358 const candidate = election.preferred(solve(mask).first)[0];
359 standings.push(candidate);
360 mask ^= 1 << candidate;
361 }
362 return finish(election, standings, optimum.first.length > 1, ['score: ' + optimum.score,
363 'optimal: ' + optimum.count, 'ranking: ' + election.list(standings, ' > ')]);
364}
365function stv(election: Election, field: Field, seats: number): Result {
366 if (seats >= field.length) fail('too many seats');
367 const aggregate = election.aggregate(field), groups = aggregate.groups.filter(group => group.ranking.length > 0);
368 const quota = Math.floor(100 * (aggregate.total - aggregate.blank) / (seats + 1)) + 1;
369 const perVote = groups.map(() => 100), elected: number[] = [], history: Scores[] = [], lines = ['quota: ' + money(quota)];
370 let continuing = field.slice();
371 for (let round = 1; ; ++round) {
372 const scores = election.zeros(), owners: (number | undefined)[] = [];
373 let exhausted = 0;
374 groups.forEach((group, index) => {
375 const owner = group.ranking.find(a => continuing.includes(a));
376 owners.push(owner);
377 if (owner === undefined) exhausted += group.votes * perVote[index];
378 else scores[owner] += group.votes * perVote[index];
379 });
380 let line = 'round ' + round + ': ' + election.values(continuing, scores, money) + '; exhausted ' + money(exhausted);
381 const order = election.ordered(continuing, scores);
382 if (continuing.length === seats - elected.length) {
383 elected.push(...order);
384 lines.push(line + '; elected ' + election.list(order));
385 break;
386 }
387 let removed: number;
388 if (scores[order[0]] >= quota) {
389 removed = order[0];
390 elected.push(removed);
391 const total = scores[removed], surplus = total - quota;
392 owners.forEach((owner, index) => {
393 if (owner === removed) perVote[index] = Math.floor(perVote[index] * surplus / total);
394 });
395 line += '; elected ' + election.names[removed] + ' (surplus ' + money(surplus) + ')';
396 } else {
397 const choice = eliminate(election, continuing, scores, history);
398 removed = choice.candidate;
399 line += '; excluded ' + election.names[removed] + tieMark(choice.tie);
400 }
401 lines.push(line);
402 if (elected.length === seats) break;
403 history.push(scores);
404 continuing = continuing.filter(a => a !== removed);
405 }
406 return { lines: [...lines, 'elected: ' + election.list(elected)] };
407}

411function tally(election: Election, method: Method, field: Field): Result {
412 const aggregate = election.aggregate(field), pairwise = aggregate.pairwise;
413 switch (method.kind) {
414 case 'PLURALITY': return scoredResult(election, field, aggregate.first, 'votes');
415 case 'RUNOFF': return runoff(election, field);
416 case 'IRV': return eliminationRounds(election, field);
417 case 'COOMBS': return eliminationRounds(election, field, true);
418 case 'BUCKLIN': return bucklin(election, field);
419 case 'BORDA': return scoredResult(election, field, aggregate.borda, 'scores');
420 case 'BALDWIN': return bordaRounds(election, field, false);
421 case 'NANSON': return bordaRounds(election, field, true);
422 case 'BLACK': {
423 const winner = condorcetWinner(field, pairwise), order = election.ordered(field, aggregate.borda);
424 const standings = winner === undefined ? order : [winner, ...order.filter(a => a !== winner)];
425 const tie = winner === undefined && aggregate.borda[order[0]] === aggregate.borda[order[1]];
426 return finish(election, standings, tie, ['condorcet: ' + (winner === undefined ? 'NONE' : election.names[winner]),
427 'scores: ' + election.values(field, aggregate.borda)]);
428 }
429 case 'VETO': return scoredResult(election, field, aggregate.last, 'vetoes', true);
430 case 'CONDORCET': return condorcet(election, field, pairwise);
431 case 'COPELAND': {
432 const scores = election.zeros(), parts: string[] = [];
433 for (const candidate of field) {
434 let wins = 0, draws = 0, losses = 0;
435 for (const other of field) if (candidate !== other) {
436 if (pairwise[candidate][other] > pairwise[other][candidate]) ++wins;
437 else if (pairwise[candidate][other] === pairwise[other][candidate]) ++draws;
438 else ++losses;
439 }
440 scores[candidate] = 2 * wins + draws;
441 parts.push(election.names[candidate] + ' ' + Math.floor(scores[candidate] / 2) + '.'
442 + (scores[candidate] % 2 ? '5' : '0') + ' (' + wins + '-' + draws + '-' + losses + ')');
443 }
444 const order = election.ordered(field, scores);
445 return finish(election, order, scores[order[0]] === scores[order[1]], ['scores: ' + parts.join(', ')]);
446 }
447 case 'MINIMAX': {
448 const scores = election.zeros();
449 for (const candidate of field) for (const other of field) if (candidate !== other) {
450 const opposition = pairwise[other][candidate], support = pairwise[candidate][other];
451 if (method.metric === 'OPPOSITION') scores[candidate] = Math.max(scores[candidate], opposition);
452 else if (opposition > support) scores[candidate] = Math.max(scores[candidate], method.metric === 'MARGIN' ? opposition - support : opposition);
453 }
454 return scoredResult(election, field, scores, 'scores', true);
455 }
456 case 'SCHULZE': return schulze(election, field, pairwise);
457 case 'RANKEDPAIRS': return rankedPairs(election, field, pairwise);
458 case 'KEMENY': return kemeny(election, field, pairwise);
459 case 'APPROVAL': return scoredResult(election, field, aggregate.positions.map(row => sum(row.slice(0, method.count))), 'approvals');
460 case 'STV': return stv(election, field, method.seats);
461 }
462}
463type QueryCommand = Extract<Command, { readonly kind: ReportKind | 'COMPARE' | 'RANKING' | 'SPOILERS' | 'TALLY' }>;
464function reports(election: Election, command: QueryCommand, field: Field): readonly string[] {
465 const aggregate = election.aggregate(field);
466 switch (command.kind) {
467 case 'STATUS': {
468 const withdrawn = election.names.map((_, index) => index).filter(a => election.withdrawn.has(a));
469 return ['election ' + election.number, 'active: ' + election.list(field),
470 'withdrawn: ' + (withdrawn.length ? election.list(withdrawn) : 'none'),
471 'tiebreak: ' + election.list(election.priority),
472 'ballots: ' + aggregate.total + ' in ' + election.entries.length + ' entries'];
473 }
474 case 'BALLOTS': {
475 const groups = aggregate.groups.map(group => ({
476 votes: group.votes, notation: group.ranking.length ? election.list(group.ranking, '>') : '-'
477 }));
478 groups.sort((a, b) => b.votes - a.votes || ascii(a.notation, b.notation));
479 return ['total ' + aggregate.total + ', blank ' + aggregate.blank + ', distinct ' + groups.length,
480 ...groups.map(group => group.votes + ' ' + group.notation)];
481 }
482 case 'PROFILE': return field.map(a => election.names[a] + ': ' + aggregate.positions[a].join(' ')
483 + ' unranked ' + (aggregate.total - sum(aggregate.positions[a])));
484 case 'PAIRWISE': return pairLines(election, field, aggregate.pairwise);
485 case 'EXHAUSTED': return eliminationRounds(election, field).exhausted;
486 case 'RANKING': case 'SPOILERS': {
487 const method = command.method;
488 if (!rankable(method)) fail('not rankable');
489 if (command.kind === 'SPOILERS' && field.length < 3) fail('too few candidates');
490 const standings = tally(election, method, field).standings;
491 if (command.kind === 'RANKING') return [election.list(standings, ' > ')];
492 const winner = standings[0], lines = ['winner: ' + election.names[winner]];
493 for (const excluded of field) if (excluded !== winner) {
494 const other = tally(election, method, field.filter(a => a !== excluded)).standings[0];
495 lines.push('without ' + election.names[excluded] + ': ' + election.names[other] + (other === winner ? '' : ' (changed)'));
496 }
497 return lines;
498 }
499 case 'SUMMARY': {
500 const wins = election.zeros(), lines: string[] = [];
501 for (const method of summaryMethods) {
502 const winner = tally(election, method, field).standings[0];
503 ++wins[winner];
504 lines.push(methodText(method) + ': ' + election.names[winner]);
505 }
506 return [...lines, 'wins: ' + election.values(field, wins)];
507 }
508 case 'COMPARE': {
509 const compared: number[] = [];
510 for (const name of command.names) {
511 const candidate = election.candidate(name);
512 if (election.withdrawn.has(candidate)) fail('withdrawn candidate ' + name);
513 compared.push(candidate);
514 }
515 const [first, second] = compared;
516 if (first === second) fail('same candidate');
517 const forward = aggregate.pairwise[first][second], backward = aggregate.pairwise[second][first];
518 const ahead: [string[], string[]] = [[], []];
519 for (const method of summaryMethods) {
520 const standings = tally(election, method, field).standings;
521 ahead[standings.indexOf(first) < standings.indexOf(second) ? 0 : 1].push(methodText(method));
522 }
523 return [command.names[0] + ' ' + forward + ', ' + command.names[1] + ' ' + backward
524 + ', neither ' + (aggregate.total - forward - backward),
525 ...command.names.map((name, side) => name + ' ahead: ' + (ahead[side].length ? ahead[side].join(', ') : 'none'))];
526 }
527 case 'TALLY': return tally(election, command.method, field).lines;
528 }
529}

531class VoteCounter {
532 private election: Election | null = null;
533 private count = 0;
534
535 execute(line: string): readonly string[] {
536 const command = parseCommand(line);
537 if (command.kind === 'CANDIDATES') {
538 if (command.names.length < 2) fail('too few candidates');
539 const seen = new Set<string>();
540 for (const name of command.names) {
541 if (seen.has(name)) fail('duplicate candidate ' + name);
542 seen.add(name);
543 }
544 this.election = new Election(command.names, ++this.count);
545 return ['election ' + this.count + ': ' + command.names.length + ' candidates'];
546 }
547 const election = this.election;
548 if (election === null) fail('no election');
549 const field = election.active();
550 switch (command.kind) {
551 case 'BALLOT': {
552 const ranking = election.checkedNames(command.names);
553 election.addEntry([{ votes: command.votes, ranking }]);
554 return [];
555 }
556 case 'RANDOM': {
557 let state = command.seed;
558 const draw = (): number => { state = 48271 * state % 2147483647; return state; };
559 const groups: Ballot[] = [];
560 for (let group = 0; group < command.count; ++group) {
561 const ranking = field.slice();
562 for (let index = ranking.length - 1; index >= 1; --index) {
563 const other = draw() % (index + 1);
564 [ranking[index], ranking[other]] = [ranking[other], ranking[index]];
565 }
566 groups.push({ votes: 1, ranking: ranking.slice(0, draw() % ranking.length + 1) });
567 }
568 election.addEntry(groups);
569 return [];
570 }
571 case 'UNDO': election.undo(); return [];
572 case 'WITHDRAW': case 'RESTORE': {
573 const candidate = election.candidate(command.name);
574 if (command.kind === 'WITHDRAW') {
575 if (election.withdrawn.has(candidate)) fail('already withdrawn ' + command.name);
576 if (field.length <= 2) fail('too few candidates');
577 election.withdrawn.add(candidate);
578 } else {
579 if (!election.withdrawn.has(candidate)) fail('not withdrawn ' + command.name);
580 election.withdrawn.delete(candidate);
581 }
582 return [];
583 }
584 case 'TIEBREAK': {
585 const order = election.checkedNames(command.names);
586 if (order.length !== election.names.length) fail('incomplete tiebreak');
587 election.setPriority(order);
588 return [];
589 }
590 default:
591 if (command.kind !== 'STATUS' && election.entries.length === 0) fail('no ballots');
592 return ['[' + line + ']', ...reports(election, command, field)];
593 }
594 }
595}

598const counter = new VoteCounter(), output: string[] = [];
599for (let index = 1; index <= Number(input[0]); ++index) {
