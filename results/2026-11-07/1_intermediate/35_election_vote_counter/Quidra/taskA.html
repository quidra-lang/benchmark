278 rows := ["round 1: " + value_list(election, field, first.first)]

282 second_candidate := order[1]

285 second := tallies(&election, finalists)
286 tie = second.first[first_candidate] == second.first[second_candidate]

288 order[0], order[1] = order[1], order[0]
289 rows = rows.append("finalists: " + name_list(election, finalists) + tie_text(advance))
290 rows = rows.append("round 2: " + value_list(election, finalists, second.first))

356Outcome condorcet_report(Election &election, const int[] field)
357 win := condorcet(&election, field, false)
358 lose := condorcet(&election, field, true)
359 matrix := pairwise(&election)
360 bool[][] weak := array(#election.names, fill = array(#election.names, fill = false))
361 strong := weak
362 for first := field[]
363 for second := field[]
364 weak[first][second] = matrix[first][second] >= matrix[second][first]
365 strong[first][second] = matrix[first][second] > matrix[second][first]
366 weak = closure(field, weak)
367 strong = closure(field, strong)
368 int[] smith := []
369 int[] schwartz := []
370 for candidate := field[]
371 top := true
372 unbeaten := true
373 for other := field[]
374 top = top and weak[candidate][other]
375 unbeaten = unbeaten and (not strong[other][candidate] or strong[candidate][other])
376 if top
377 smith = smith.append(candidate)
378 if unbeaten
379 schwartz = schwartz.append(candidate)
380 winner := if win < 0 then "NONE" else election.names[win]
381 loser := if lose < 0 then "NONE" else election.names[lose]
382 return Outcome([], ["winner: " + winner, "loser: " + loser, "smith: " + name_list(election, smith), "schwartz: " + name_list(election, schwartz)])

384(int[], bool) topological(const Election election, const int[] field, const bool[][] relation)
385 remaining := field
386 int[] order := []
387 tie := false
388 while #remaining != 0
389 int[] choices := []
390 for candidate := remaining[]
391 incoming := false
392 for other := without(remaining, [candidate])[]
393 incoming = incoming or relation[other][candidate]
394 if not incoming
395 choices = choices.append(candidate)
396 if #order == 0
397 tie = #choices > 1
398 chosen := effective(election.tiebreak, choices)[0]
399 order = order.append(chosen)
400 remaining = without(remaining, [chosen])
401 return (order, tie)

403Outcome schulze(Election &election, const int[] field)
404 matrix := pairwise(&election)
405 int[][] paths := array(#election.names, fill = array(#election.names, fill = 0))
406 for first := field[]
407 for second := field[]
408 if matrix[first][second] > matrix[second][first]
409 paths[first][second] = matrix[first][second]
410 for via := field[]
411 for first := without(field, [via])[]
412 for second := without(field, [via, first])[]
413 paths[first][second] = math.max(paths[first][second], math.min(paths[first][via], paths[via][second]))
414 bool[][] relation := array(#election.names, fill = array(#election.names, fill = false))
415 for first := field[]
416 for second := field[]
417 relation[first][second] = paths[first][second] > paths[second][first]
418 order, tie := topological(election, field, relation)
419 rows := pair_lines(election, field, paths).append("ranking: " + name_list(election, order, " > "))
420 return finish(election, order, tie, rows)

422Outcome ranked_pairs(Election &election, const int[] field)
423 matrix := pairwise(&election)
424 Edge[] edges := []
425 int[] wins := []
426 int[] losses := []
427 int[] sources := []
428 int[] destinations := []
429 for first := field[]
430 for second := field[]
431 if matrix[first][second] > matrix[second][first]
432 edges = edges.append(Edge(first, second))
433 wins = wins.append(matrix[first][second])
434 losses = losses.append(matrix[second][first])
435 sources = sources.append(election.priority[first])
436 destinations = destinations.append(election.priority[second])
437 order := stable_order(sequence(int(#edges)), destinations, false)
438 order = stable_order(order, sources, false)
439 order = stable_order(order, losses, false)
440 order = stable_order(order, wins, true)
441 bool[][] locked := array(#election.names, fill = array(#election.names, fill = false))
442 reached := locked
443 string[] rows := []
444 for index := order[]
445 edge := edges[index]
446 accepted := not reached[edge.to][edge.from]
447 label := if accepted then "lock" else "skip"
448 rows = rows.append(label + " " + election.names[edge.from] + ">" + election.names[edge.to] + " " + string(matrix[edge.from][edge.to]) + "-" + string(matrix[edge.to][edge.from]))
449 if accepted
450 locked[edge.from][edge.to] = true
451 reached[edge.from][edge.to] = true
452 reached = closure(field, reached)
453 standings, tie := topological(election, field, locked)
454 return finish(election, standings, tie, rows)

456Outcome kemeny(Election &election, const int[] field)
457 matrix := pairwise(&election)
458 limit := 2 ^ int(#election.names)
459 best := array(nat(limit), fill = 0)
460 ways := array(nat(limit), fill = 0)
461 int[][] heads := array(nat(limit), fill = [])
462 ways[0] = 1
463 for: 1 <= subset < limit
464 candidates := members(subset, int(#election.names))
465 if #without(candidates, field) > 0
466 continue
467 best[subset] = -1
468 for candidate := candidates[]
469 rest := subset - 2 ^ candidate
470 score := best[rest] + selected(matrix[candidate], without(candidates, [candidate])).sum()
471 if score > best[subset]
472 best[subset] = score
473 ways[subset] = 0
474 heads[subset] = []
475 if score == best[subset]
476 ways[subset] += ways[rest]
477 heads[subset] = heads[subset].append(candidate)
478 key := field_id(field)
479 remaining := key
480 int[] order := []
481 while remaining > 0
482 candidate := effective(election.tiebreak, heads[remaining])[0]
483 order = order.append(candidate)
484 remaining -= 2 ^ candidate
485 rows := ["score: " + string(best[key]), "optimal: " + string(ways[key]), "ranking: " + name_list(election, order, " > ")]
486 return finish(election, order, #heads[key] > 1, rows)

488Outcome stv(Election &election, const int[] field, int seats)
489 values := array(#election.groups, fill = 0)
490 for index := election.groups.indices()[]
491 if #effective(election.groups[index].ranking, field) > 0
492 values[index] = 100
493 quota := tallies(&election, field).nonblank * 100 / (seats + 1) + 1
494 continuing := field
495 int[] elected := []
496 int[][] history := []
497 rows := ["quota: " + money(quota)]
498 while true
499 scores := array(#election.names, fill = 0)
500 owners := array(#election.groups, fill = -1)
501 exhausted := 0
502 for index := election.groups.indices()[]
503 if values[index] == 0
504 continue
505 group := election.groups[index]
506 ranking := effective(group.ranking, continuing)
507 value := group.votes * values[index]
508 if #ranking > 0
509 owners[index] = ranking[0]
510 scores[ranking[0]] += value
511 else
512 exhausted += value
513 history = history.append(scores)
514 row := "round {#history}: " + value_list(election, continuing, scores, true) + "; exhausted " + money(exhausted)
515 order := ordered(election, continuing, scores, true)
516 if int(#continuing) == seats - int(#elected)
517 elected = elected.concat(order)
518 rows = rows.append(row + "; elected " + name_list(election, order))
519 break
520 chosen := order[0]
521 if scores[chosen] >= quota
522 surplus := scores[chosen] - quota
523 elected = elected.append(chosen)
524 continuing = without(continuing, [chosen])
525 for index := owners.indices()[]
526 if owners[index] == chosen
527 values[index] = values[index] * surplus / scores[chosen]
528 rows = rows.append(row + "; elected " + election.names[chosen] + " (surplus " + money(surplus) + ")")
529 if int(#elected) == seats
530 break
531 else
532 choice := look_back(election, continuing, history)
533 continuing = without(continuing, [choice.candidate])
534 rows = rows.append(row + "; excluded " + election.names[choice.candidate] + tie_text(choice.tie))
535 return Outcome([], rows.append("elected: " + name_list(election, elected)))

564Outcome tally(Election &election, const int[] field, const Method method)
565 kind := method.kind
566 if kind == MethodKind.Runoff
567 return runoff(&election, field)
568 elif kind == MethodKind.Irv or kind == MethodKind.Coombs
569 return elimination(&election, field, kind == MethodKind.Coombs)
570 elif kind == MethodKind.Bucklin
571 return bucklin(&election, field)
572 elif kind == MethodKind.Baldwin or kind == MethodKind.Nanson
573 return borda_elimination(&election, field, kind == MethodKind.Nanson)
574 elif kind == MethodKind.Condorcet
575 return condorcet_report(&election, field)
576 elif kind == MethodKind.Schulze
577 return schulze(&election, field)
578 elif kind == MethodKind.RankedPairs
579 return ranked_pairs(&election, field)
580 elif kind == MethodKind.Kemeny
581 return kemeny(&election, field)
582 elif kind == MethodKind.Stv
583 return stv(&election, field, method.parameter)
584 base := tallies(&election, field)
585 scores := array(#election.names, fill = 0)
586 label := "scores"
587 descending := true
588 string[] rows := []
589 winner := -1
590 if kind == MethodKind.Plurality
591 scores = base.first
592 label = "votes"
593 elif kind == MethodKind.Borda or kind == MethodKind.Black
594 scores = base.borda
595 if kind == MethodKind.Black
596 winner = condorcet(&election, field, false)
597 rows = rows.append("condorcet: " + (if winner < 0 then "NONE" else election.names[winner]))
598 elif kind == MethodKind.Veto
599 scores = base.last
600 label = "vetoes"
601 descending = false
602 elif kind == MethodKind.Approval
603 scores = base.approval[math.min(method.parameter, int(#field)) - 1]
604 label = "approvals"
605 elif kind == MethodKind.Copeland
606 matrix := pairwise(&election)
607 string[] parts := []
608 for candidate := field[]
609 wins := 0
610 draws := 0
611 losses := 0
612 for other := without(field, [candidate])[]
613 if matrix[candidate][other] > matrix[other][candidate]
614 wins += 1
615 elif matrix[candidate][other] == matrix[other][candidate]
616 draws += 1
617 else
618 losses += 1
619 scores[candidate] = 2 * wins + draws
620 score := scores[candidate]
621 parts = parts.append(election.names[candidate] + " {score / 2}.{score % 2 * 5} ({wins}-{draws}-{losses})")
622 rows = rows.append("scores: " + parts.join(", "))
623 elif kind == MethodKind.Minimax
624 matrix := pairwise(&election)
625 descending = false
626 for candidate := field[]
627 int[] possibilities := []
628 for other := without(field, [candidate])[]
629 opposition := matrix[other][candidate]
630 support := matrix[candidate][other]
631 value := 0
632 if method.score == MinimaxKind.Opposition
633 value = opposition
634 elif opposition > support
635 value = if method.score == MinimaxKind.Margin then opposition - support else opposition
636 possibilities = possibilities.append(value)
637 scores[candidate] = possibilities.max()
638 order := ordered(election, field, scores, descending)
639 tie := scores[order[0]] == scores[order[1]]
640 if winner >= 0
641 order = [winner].concat(without(order, [winner]))
642 tie = false
