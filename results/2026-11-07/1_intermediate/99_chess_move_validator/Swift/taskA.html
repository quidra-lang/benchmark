11func square(_ s: String) -> Int? {
12 let a = Array(s.utf8)
13 guard a.count == 2, (97...104).contains(a[0]), (49...56).contains(a[1]) else {
14 return nil
15 }
16 return Int(a[1] - 49) * 8 + Int(a[0] - 97)

42struct Position {
43 var board = Array(repeating: ".", count: 64)
44 var white = true
45 var rights = ""
46 var ep: Int? = nil
47 var half = 0
48 var full = 1
49 func attacks(_ s: Int, _ t: Int) -> Bool {
50 if s == t || board[s] == "." {
51 return false
52 }
53 let dx = t % 8 - s % 8
54 let dy = t / 8 - s / 8
55 let p = board[s].uppercased()
56 if p == "P" {
57 return abs(dx) == 1 && dy == (color(board[s]) ? 1 : -1)
58 }
59 if p == "N" {
60 return abs(dx) * abs(dy) == 2
61 }
62 if p == "K" {
63 return max(abs(dx), abs(dy)) == 1
64 }
65 let straight = dx == 0 || dy == 0
66 let diagonal = abs(dx) == abs(dy)
67 if !(p == "Q" && (straight || diagonal) || p == "R" && straight || p == "B" && diagonal) {
68 return false
69 }
70 let sx = dx.signum()
71 let sy = dy.signum()
72 var x = s % 8 + sx
73 var y = s / 8 + sy
74 while x != t % 8 || y != t / 8 {
75 if board[y * 8 + x] != "." {
76 return false
77 }
78 x += sx
79 y += sy
80 }
81 return true
82 }
83 func attacked(_ t: Int, _ byWhite: Bool) -> Bool {
84 (0..<64).contains {
85 board[$0] != "." && color(board[$0]) == byWhite && attacks($0, t)
86 }
87 }
88 func check(_ side: Bool) -> Bool {
89 guard let king = board.firstIndex(of: side ? "K" : "k") else {
90 return true
91 }
92 return attacked(king, !side)
93 }
94 func castle(_ m: Move) -> Bool {
95 board[m.s].uppercased() == "K" && abs(m.s - m.t) == 2
96 }
97 func enPassant(_ m: Move) -> Bool {
98 board[m.s].uppercased() == "P" && m.s % 8 != m.t % 8 && board[m.t] == "."
99 }
100 func capture(_ m: Move) -> Bool {
101 board[m.t] != "." || enPassant(m)
102 }
103 func after(_ m: Move) -> Position {
104 var q = self
105 let p = board[m.s]
106 let pawn = p.uppercased() == "P"
107 if enPassant(m) {
108 q.board[m.t + (white ? -8 : 8)] = "."
109 }
110 if castle(m) {
111 let r = m.s / 8 * 8
112 let from = r + (m.t > m.s ? 7 : 0)
113 let to = r + (m.t > m.s ? 5 : 3)
114 q.board[to] = q.board[from]
115 q.board[from] = "."
116 }
117 q.board[m.t] = m.promotion.isEmpty ? p : (white ? m.promotion : m.promotion.lowercased())
118 q.board[m.s] = "."
119 q.half = pawn || capture(m) ? 0 : half + 1
120 if p.uppercased() == "K" {
121 q.rights.removeAll {
122 (white ? "KQ" : "kq").contains($0)
123 }
124 }
125 for (s, r) in [(0, Character("Q")), (7, Character("K")), (56, Character("q")), (63, Character("k"))] where m.s == s || m.t == s {
126 q.rights.removeAll {
127 $0 == r
128 }
129 }
130 q.ep = pawn && abs(m.t - m.s) == 16 ? (m.s + m.t) / 2 : nil
131 q.full += white ? 0 : 1
132 q.white.toggle()
133 return q
134 }
135 func pseudo() -> [Move] {
136 var result: [Move] = []
137 for s in 0..<64 where board[s] != "." && color(board[s]) == white {
138 let pawn = board[s].uppercased() == "P"
139 let step = white ? 8 : -8
140 for t in 0..<64 {
141 if board[t] != "." && color(board[t]) == white {
142 continue
143 }
144 var valid = attacks(s, t)
145 if pawn {
146 valid = (t == s + step && board[t] == ".") || (t == s + step * 2 && s / 8 == (white ? 1 : 6) && board[s + step] == "." && board[t] == ".") || (attacks(s, t) && (board[t] != "." || ep == t))
147 }
148 if valid {
149 if pawn && t / 8 == (white ? 7 : 0) {
150 for p in ["Q", "R", "B", "N"] {
151 result.append(Move(s: s, t: t, promotion: p))
152 }
153 }
154 else {
155 result.append(Move(s: s, t: t))
156 }
157 }
158 }
159 }
160 return result
161 }
162 func castleReason(_ kingSide: Bool) -> String? {
163 let right = white ? (kingSide ? "K" : "Q") : (kingSide ? "k" : "q")
164 let base = white ? 0 : 56
165 if !rights.contains(right) {
166 return "no castling right"
167 }
168 if (kingSide ? [5, 6] : [1, 2, 3]).contains(where: {
169 board[base + $0] != "."
170 }) {
171 return "path blocked"
172 }
173 if check(white) {
174 return "in check"
175 }
176 if attacked(base + (kingSide ? 5 : 3), !white) {
177 return "through check"
178 }
179 if after(Move(s: base + 4, t: base + (kingSide ? 6 : 2))).check(white) {
180 return "into check"
181 }
182 return nil
183 }
184 func legal() -> [Move] {
185 var moves = pseudo().filter {
186 !after($0).check(white)
187 }
188 let base = white ? 0 : 56
189 for ks in [true, false] where castleReason(ks) == nil {
190 moves.append(Move(s: base + 4, t: base + (ks ? 6 : 2)))
191 }
192 return moves
193 }
194 func disambiguation(_ m: Move, _ moves: [Move]) -> String {
195 let others = moves.filter {
196 $0.s != m.s && $0.t == m.t && board[$0.s] == board[m.s] && !castle($0)
197 }
198 if others.isEmpty {
199 return ""
200 }
201 if others.allSatisfy({
202 $0.s % 8 != m.s % 8
203 }) {
204 return String(name(m.s).prefix(1))
205 }
206 if others.allSatisfy({
207 $0.s / 8 != m.s / 8
208 }) {
209 return String(m.s / 8 + 1)
210 }
211 return name(m.s)
212 }
213 func mark(_ m: Move) -> String {
214 let q = after(m)
215 return q.check(q.white) ? (q.legal().isEmpty ? "#" : "+") : ""
216 }
217 func san(_ m: Move, _ moves: [Move]) -> String {
218 var s = ""
219 if castle(m) {
220 s = m.t > m.s ? "O-O" : "O-O-O"
221 }
222 else if board[m.s].uppercased() == "P" {
223 s = (capture(m) ? String(name(m.s).prefix(1)) + "x" : "") + name(m.t)
224 if !m.promotion.isEmpty {
225 s += "=" + m.promotion
226 }
227 } else {
228 s = board[m.s].uppercased() + disambiguation(m, moves) + (capture(m) ? "x" : "") + name(m.t)
229 }
230 return s + mark(m)
231 }
232 func readMove(_ w: String, _ uci: Bool) throws -> Move {

234 var promo = ""
235 var suffix = ""
236 var kind = "P"
237 var file = ""
238 var rank = ""
239 var cap = false
240 var ks: Bool? = nil
241 var from: Int? = nil
242 var target = 0
243 if uci {
244 let a = Array(w)
245 guard a.count == 4 || a.count == 5, let s = square(String(a[0..<2])), let t = square(String(a[2..<4])), a.count == 4 || "qrbn".contains(a[4]) else {
246 try fail("syntax")
247 }
248 from = s
249 target = t
250 if a.count == 5 {
251 promo = String(a[4]).uppercased()
252 }
253 } else {
254 if u.last == "+" || u.last == "#" {
255 suffix = String(u.removeLast())
256 }
257 if u == "O-O" || u == "O-O-O" {
258 ks = u == "O-O"
259 }
260 else {
261 if u.count >= 2 && u.dropLast().last == "=" {
262 promo = String(u.removeLast())
263 u.removeLast()
264 if !["Q", "R", "B", "N"].contains(promo) {
265 try fail("syntax")
266 }
267 }
268 guard u.count >= 2, let t = square(String(u.suffix(2))) else {
269 try fail("syntax")
270 }
271 target = t
272 u.removeLast(2)
273 if u.last == "x" {
274 cap = true
275 u.removeLast()
276 }
277 if let c = u.first, "KQRBN".contains(c) {
278 kind = String(u.removeFirst())
279 if let c = u.first, "abcdefgh".contains(c) {
280 file = String(u.removeFirst())
281 }
282 if let c = u.first, "12345678".contains(c) {
283 rank = String(u.removeFirst())
284 }
285 if !u.isEmpty || !promo.isEmpty {
286 try fail("syntax")
287 }
288 } else if cap {
289 if u.count != 1 || !"abcdefgh".contains(u) {
290 try fail("syntax")
291 }
292 file = u
293 } else if !u.isEmpty {
294 try fail("syntax")
295 }
296 }
297 }
298 let moves = legal()
299 if moves.isEmpty {
300 try fail("game over")
301 }
302 if let ks = ks {
303 if let r = castleReason(ks) {
304 try fail(r)
305 }
306 let base = white ? 0 : 56
307 let m = Move(s: base + 4, t: base + (ks ? 6 : 2))
308 if mark(m) != suffix {
309 try fail("check mark")
310 }
311 return m
312 }
313 if let s = from, board[s] == "." || color(board[s]) != white {
314 try fail("no piece")
315 }
316 func matches(_ m: Move) -> Bool {
317 if m.t != target {
318 return false
319 }
320 if let s = from {
321 return m.s == s
322 }
323 return !castle(m) && board[m.s].uppercased() == kind && (file.isEmpty || name(m.s).hasPrefix(file)) && (rank.isEmpty || name(m.s).hasSuffix(rank)) && (kind != "P" || (m.s % 8 != m.t % 8) == cap)
324 }
325 let matching = moves.filter(matches)
326 if matching.isEmpty {
327 try fail(pseudo().contains(where: matches) ? "leaves king in check" : "no such move")
328 }
329 if Set(matching.map {
330 $0.s
331 }).count > 1 {
332 try fail("ambiguous")
333 }
334 let m = matching[0]
335 if !uci && capture(m) != cap {
336 try fail("capture mark")
337 }
338 if !m.promotion.isEmpty && promo.isEmpty {
339 try fail("missing promotion")
340 }
341 if m.promotion.isEmpty && !promo.isEmpty {
342 try fail("invalid promotion")
343 }
344 if !uci && kind != "P" && file + rank != disambiguation(m, moves) {
345 try fail("disambiguation")
346 }
347 let chosen = matching.first {
348 $0.promotion == promo
349 }!
350 if !uci && mark(chosen) != suffix {
351 try fail("check mark")
352 }
353 return chosen
354 }
355 func fen() -> String {
356 var ranks: [String] = []
357 for r in stride(from: 7, through: 0, by: -1) {
358 var row = ""
359 var empty = 0
360 for f in 0..<8 {
361 let p = board[r * 8 + f]
362 if p == "." {
363 empty += 1
364 } else {
365 if empty > 0 {
366 row += String(empty)
367 empty = 0
368 }
369 row += p
370 }
371 }
372 if empty > 0 {
373 row += String(empty)
374 }
375 ranks.append(row)
376 }
377 return ranks.joined(separator: "/") + " \(white ? "w" : "b") \(rights.isEmpty ? "-" : rights) \(ep.map(name) ?? "-") \(half) \(full)"
378 }
379 func key() -> String {
380 let e = legal().contains {
381 enPassant($0)
382 } ? ep.map(name) ?? "-" : "-"
383 return board.joined() + (white ? "w" : "b") + rights + e
384 }
385}

387func load(_ fields: [String]) throws -> Position {
388 if fields.count != 6 {
389 try fail("fields")
390 }
391 let ranks = fields[0].components(separatedBy: "/")
392 if ranks.count != 8 {
393 try fail("placement")
394 }
395 var p = Position()
396 for (i, rank) in ranks.enumerated() {
397 var row: [String] = []
398 var digit = false
399 for c in rank {
400 if "12345678".contains(c) {
401 if digit {
402 try fail("placement")
403 }
404 row += Array(repeating: ".", count: Int(String(c))!)
405 digit = true
406 } else {
407 if !"KQRBNPkqrbnp".contains(c) {
408 try fail("placement")
409 }
410 row.append(String(c))
411 digit = false
412 }
413 }
414 if row.count != 8 {
415 try fail("placement")
416 }
417 for f in 0..<8 {
418 p.board[(7 - i) * 8 + f] = row[f]
419 }
420 }
421 if p.board.filter({
422 $0 == "K"
423 }).count != 1 || p.board.filter({
424 $0 == "k"
425 }).count != 1 {
426 try fail("kings")
427 }
428 if (Array(0..<8) + Array(56..<64)).contains(where: {
429 p.board[$0].uppercased() == "P"
430 }) {
431 try fail("pawns")
432 }
433 if fields[1] != "w" && fields[1] != "b" {
434 try fail("side")
435 }
436 p.white = fields[1] == "w"
437 let rights = fields[2] == "-" ? "" : fields[2]
438 if fields[2] != "-" && (rights.isEmpty || String("KQkq".filter {
439 rights.contains($0)
440 }) != rights) {
441 try fail("castling")
442 }
443 for (r, k, s, piece) in [("K",4,7,"R"),("Q",4,0,"R"),("k",60,63,"r"),("q",60,56,"r")] where rights.contains(r) {
444 if p.board[k] != (k == 4 ? "K" : "k") || p.board[s] != piece {
445 try fail("castling")
446 }
447 }
448 p.rights = rights
449 if fields[3] != "-" {
450 guard let e = square(fields[3]) else {
451 try fail("en passant")
452 }
453 if e / 8 != (p.white ? 5 : 2) || p.board[e] != "." || p.board[e + (p.white ? 8 : -8)] != "." || p.board[e + (p.white ? -8 : 8)] != (p.white ? "p" : "P") {
454 try fail("en passant")
455 }
456 p.ep = e
457 }
458 guard let h = number(fields[4],0,999), let f = number(fields[5],1,999), p.ep == nil || h == 0 else {
459 try fail("clocks")
460 }
461 p.half = h
462 p.full = f
463 if p.check(!p.white) {
464 try fail("check")
465 }
466 return p
467}

469func perft(_ p: Position, _ d: Int) -> [Int] {
470 if d == 0 {
471 return [1,0,0,0,0,0,0]
472 }
473 var counts = Array(repeating: 0, count: 7)
474 for m in p.legal() {
475 let q = p.after(m)
476 if d == 1 {
477 counts[0] += 1
478 counts[1] += p.capture(m) ? 1 : 0
479 counts[2] += p.enPassant(m) ? 1 : 0
480 counts[3] += p.castle(m) ? 1 : 0
481 counts[4] += m.promotion.isEmpty ? 0 : 1
482 if q.check(q.white) {
483 counts[5] += 1
484 if q.legal().isEmpty {
485 counts[6] += 1
486 }
487 }
488 } else {
489 let c = perft(q,d-1)
490 for i in 0..<7 {
491 counts[i] += c[i]
492 }
493 }
494 }
495 return counts
496}
