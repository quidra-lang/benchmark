34local function square(s)
35 if not s:match("^[a-h][1-8]$") then
36 return -1
37 end
38 return s:byte(1) - 97 + 8 * (s:byte(2) - 49)
39end
40local function name(s)
41 return files:sub(s % 8 + 1, s % 8 + 1) .. ranks:sub(s // 8 + 1, s // 8 + 1)
42end
43local function number(s, lo, hi)
44 if not s:match("^%d+$") or (#s > 1 and s:sub(1, 1) == "0") then
45 return -1
46 end
47 local n = tonumber(s)
48 return n >= lo and n <= hi and n or -1

50local function shift(s, dx, dy)
51 local x, y = s % 8 + dx, s // 8 + dy
52 if x < 0 or x > 7 or y < 0 or y > 7 then
53 return -1
54 end
55 return 8 * y + x
56end
57local function targets(b, s)
58 local p = b[s]
59 local kind = p:upper()
60 local result = {}
61 local dirs
62 if contains("PNK", kind) then
63 dirs =
64 kind == "P" and {{-1, color(p) == "w" and 1 or -1}, {1, color(p) == "w" and 1 or -1}} or
65 kind == "N" and knight or
66 king
67 for _, d in ipairs(dirs) do
68 local t = shift(s, d[1], d[2])
69 if t >= 0 then
70 result[#result + 1] = t
71 end
72 end
73 else
74 dirs = kind == "Q" and king or kind == "R" and rook or bishop
75 for _, d in ipairs(dirs) do
76 local t = shift(s, d[1], d[2])
77 while t >= 0 do
78 result[#result + 1] = t
79 if b[t] ~= "." then
80 break
81 end
82 t = shift(t, d[1], d[2])
83 end
84 end
85 end
86 return result
87end

92 for _, v in ipairs(targets(b, s)) do
93 if v == t then

96 end
97 end
98 end
99 end
100 return a

110function Position:check(c)
111 c = c or self.side
112 local k
113 for s = 0, 63 do
114 if self.board[s] == colored("K", c) then
115 k = s
116 break
117 end
118 end
119 return #attackers(self.board, k, opposite(c)) > 0
120end
121function Position:castle(m)
122 return self.board[m.s]:upper() == "K" and math.abs(m.t - m.s) == 2
123end
124function Position:enpassant(m)
125 return self.board[m.s]:upper() == "P" and m.s % 8 ~= m.t % 8 and self.board[m.t] == "."
126end
127function Position:capture(m)
128 return self.board[m.t] ~= "." or self:enpassant(m)
129end
130function Position:captured(m)
131 return self.board[m.t] ~= "." and self.board[m.t]:lower() or self:enpassant(m) and "p" or "-"
132end
133function Position:play(m)
134 local b = {}
135 for s = 0, 63 do
136 b[s] = self.board[s]
137 end
138 local piece = b[m.s]
139 local kind = piece:upper()
140 if self:enpassant(m) then
141 b[m.t - step(self.side)] = "."
142 end
143 if self:castle(m) then
144 local r = rules[colored(m.t > m.s and "K" or "Q", self.side)]
145 b[r[4]] = b[r[2]]
146 b[r[2]] = "."
147 end
148 b[m.t] = m.p ~= "" and colored(m.p, self.side) or piece
149 b[m.s] = "."
150 local rights = ""
151 for r in self.rights:gmatch(".") do
152 if (kind ~= "K" or color(r) ~= self.side) and rules[r][2] ~= m.s and rules[r][2] ~= m.t then
153 rights = rights .. r
154 end
155 end
156 return Position.new(
157 b,
158 opposite(self.side),
159 rights,
160 kind == "P" and math.abs(m.t - m.s) == 16 and (m.s + m.t) // 2 or -1,
161 (kind == "P" or self:capture(m)) and 0 or self.half + 1,
162 self.full + (self.side == "b" and 1 or 0)
163 )
164end
165function Position:pseudo()
166 local result = {}
167 for s = 0, 63 do
168 local p = self.board[s]
169 if p ~= "." and color(p) == self.side then
170 local pawn = p:upper() == "P"
171 local a = {}
172 if pawn then
173 local d = step(self.side)
174 if self.board[s + d] == "." then
175 a[#a + 1] = s + d
176 if s // 8 == (self.side == "w" and 1 or 6) and self.board[s + 2 * d] == "." then
177 a[#a + 1] = s + 2 * d
178 end
179 end
180 for _, t in ipairs(targets(self.board, s)) do
181 if (self.board[t] ~= "." and color(self.board[t]) ~= self.side) or t == self.ep then
182 a[#a + 1] = t
183 end
184 end
185 else
186 for _, t in ipairs(targets(self.board, s)) do
187 if self.board[t] == "." or color(self.board[t]) ~= self.side then
188 a[#a + 1] = t
189 end
190 end
191 end
192 for _, t in ipairs(a) do
193 if pawn and t // 8 == (self.side == "w" and 7 or 0) then
194 for p in ("QRBN"):gmatch(".") do
195 result[#result + 1] = {s = s, t = t, p = p}
196 end
197 else
198 result[#result + 1] = {s = s, t = t, p = ""}
199 end
200 end
201 end
202 end
203 return result
204end
205function Position:castlingMove(k)
206 local r = rules[colored(k and "K" or "Q", self.side)]
207 return {s = r[1], t = r[3], p = ""}
208end
209function Position:castlingProblem(k)
210 local right = colored(k and "K" or "Q", self.side)
211 local r = rules[right]
212 if not contains(self.rights, right) then
213 return "no castling right"
214 end
215 for _, s in ipairs(r[5]) do
216 if self.board[s] ~= "." then
217 return "path blocked"
218 end
219 end
220 if self:check() then
221 return "in check"
222 end
223 if #attackers(self.board, r[4], opposite(self.side)) > 0 then
224 return "through check"
225 end
226 if self:play(self:castlingMove(k)):check(self.side) then
227 return "into check"
228 end
229 return nil
230end
231function Position:legal()
232 if self.cached then
233 return self.cached
234 end
235 local a = {}
236 for _, m in ipairs(self:pseudo()) do
237 if not self:play(m):check(self.side) then
238 a[#a + 1] = m
239 end
240 end
241 for _, k in ipairs({true, false}) do
242 if not self:castlingProblem(k) then
243 a[#a + 1] = self:castlingMove(k)
244 end
245 end
246 self.cached = a
247 return a
248end
249function Position:mark(m)

252end
253function Position:disambiguation(m)
254 local any, file, rank = false, true, true
255 for _, x in ipairs(self:legal()) do
256 if not self:castle(x) and x.t == m.t and x.s ~= m.s and self.board[x.s] == self.board[m.s] then
257 any = true
258 file = file and x.s % 8 ~= m.s % 8
259 rank = rank and x.s // 8 ~= m.s // 8
260 end
261 end
262 if not any then
263 return ""
264 end
265 if file then
266 return files:sub(m.s % 8 + 1, m.s % 8 + 1)
267 end
268 if rank then
269 return ranks:sub(m.s // 8 + 1, m.s // 8 + 1)
270 end
271 return name(m.s)
272end
273function Position:san(m)
274 local kind = self.board[m.s]:upper()
275 local text
276 if self:castle(m) then
277 text = m.t > m.s and "O-O" or "O-O-O"
278 elseif kind == "P" then
279 text =
280 (self:capture(m) and files:sub(m.s % 8 + 1, m.s % 8 + 1) .. "x" or "") ..
281 name(m.t) .. (m.p ~= "" and "=" .. m.p or "")
282 else
283 text = kind .. self:disambiguation(m) .. (self:capture(m) and "x" or "") .. name(m.t)
284 end
285 return text .. self:mark(m)
286end

293 break
294 end
295 end
296 end
297 local b = ""

329function Position:pins()
330 local k
331 for s = 0, 63 do
332 if self.board[s] == colored("K", self.side) then
333 k = s
334 end
335 end
336 local found = {}
337 for _, d in ipairs(king) do
338 local first = -1
339 local t = shift(k, d[1], d[2])
340 while t >= 0 do
341 local p = self.board[t]
342 if p ~= "." then
343 if first < 0 then
344 if color(p) ~= self.side then
345 break
346 end
347 first = t
348 else
349 if
350 color(p) ~= self.side and
351 contains("Q" .. ((d[1] == 0 or d[2] == 0) and "R" or "B"), p:upper())
352 then
353 found[#found + 1] = {first, t}
354 end
355 break
356 end
357 end
358 t = shift(t, d[1], d[2])
359 end
360 end
361 table.sort(
362 found,
363 function(a, b)
364 return a[1] < b[1]
365 end
366 )
367 return found
368end
369function Position:fen()
370 local rows = {}
371 for r = 7, 0, -1 do
372 local row = ""
373 for f = 0, 7 do
374 row = row .. self.board[8 * r + f]
375 end
376 rows[#rows + 1] =
377 row:gsub(
378 "%.+",
379 function(run)
380 return tostring(#run)
381 end
382 )
383 end
384 return table.concat(
385 {
386 table.concat(rows, "/"),
387 self.side,
388 self.rights ~= "" and self.rights or "-",
389 self.ep >= 0 and name(self.ep) or "-",
390 self.half,
391 self.full
392 },
393 " "
394 )
395end
396local function split(s)
397 local a = {}
398 for x in s:gmatch("%S+") do
399 a[#a + 1] = x
400 end
401 return a
402end
403local function parseFen(a)
404 if #a ~= 6 then
405 fail("fields")
406 end
407 local rows = {}
408 for row in (a[1] .. "/"):gmatch("(.-)/") do
409 rows[#rows + 1] = row
410 end
411 if #rows ~= 8 then
412 fail("placement")
413 end
414 local b = {}
415 for s = 0, 63 do
416 b[s] = "."
417 end
418 for i, row in ipairs(rows) do
419 local f, digit = 0, false
420 for c in row:gmatch(".") do
421 if contains("12345678", c) then
422 if digit then
423 fail("placement")
424 end
425 f = f + tonumber(c)
426 digit = true
427 else
428 if not contains("KQRBNPkqrbnp", c) or f >= 8 then
429 fail("placement")
430 end
431 b[(8 - i) * 8 + f] = c
432 f = f + 1
433 digit = false
434 end
435 end
436 if f ~= 8 then
437 fail("placement")
438 end
439 end
440 local white, black = 0, 0
441 for s = 0, 63 do
442 if b[s] == "K" then
443 white = white + 1
444 elseif b[s] == "k" then
445 black = black + 1
446 end
447 end
448 if white ~= 1 or black ~= 1 then
449 fail("kings")
450 end
451 for s = 0, 7 do
452 if b[s]:upper() == "P" or b[56 + s]:upper() == "P" then
453 fail("pawns")
454 end
455 end
456 local side = a[2]
457 if side ~= "w" and side ~= "b" then
458 fail("side")
459 end
460 local rights = a[3] == "-" and "" or a[3]
461 local previous = 0
462 if a[3] ~= "-" then
463 if #rights == 0 then
464 fail("castling")
465 end
466 for r in rights:gmatch(".") do
467 local i = ("KQkq"):find(r, 1, true)
468 if not i or i <= previous then
469 fail("castling")
470 end
471 previous = i
472 end
473 end
474 for r in rights:gmatch(".") do
475 local rule = rules[r]
476 if b[rule[1]] ~= colored("K", color(r)) or b[rule[2]] ~= colored("R", color(r)) then
477 fail("castling")
478 end
479 end
480 local ep = -1
481 if a[4] ~= "-" then
482 ep = square(a[4])
483 local d = step(side)
484 if
485 ep < 0 or ep // 8 ~= (side == "w" and 5 or 2) or b[ep] ~= "." or b[ep + d] ~= "." or
486 b[ep - d] ~= colored("P", opposite(side))
487 then
488 fail("en passant")
489 end
490 end
491 local half, full = number(a[5], 0, 999), number(a[6], 1, 999)
492 if half < 0 or full < 0 or (ep >= 0 and half ~= 0) then
493 fail("clocks")
494 end
495 local p = Position.new(b, side, rights, ep, half, full)
496 if p:check(opposite(side)) then
497 fail("check")
498 end
499 return p
500end
501local function readSan(pos, text)
502 local last = text:sub(-1)
503 local mark = (last == "+" or last == "#") and last or ""
504 local body = mark ~= "" and text:sub(1, -2) or text
505 local castling = body == "O-O" or body == "O-O-O"
506 local promotion, target, capture, kind, file, rank = "", -1, false, "P", "", ""
507 if not castling then
508 if #body >= 2 and body:sub(-2, -2) == "=" then
509 promotion = body:sub(-1)
510 if not contains("QRBN", promotion) then
511 fail("syntax")
512 end
513 body = body:sub(1, -3)
514 end
515 target = square(body:sub(-2))
516 if target < 0 then
517 fail("syntax")
518 end
519 body = body:sub(1, -3)
520 capture = body:sub(-1) == "x"
521 if capture then
522 body = body:sub(1, -2)
523 end
524 if #body > 0 and contains("KQRBN", body:sub(1, 1)) then
525 kind = body:sub(1, 1)
526 local f, r = body:sub(2):match("^([a-h]?)([1-8]?)$")
527 if not f or promotion ~= "" then
528 fail("syntax")
529 end
530 file, rank = f, r
531 else
532 if not body:match(capture and "^[a-h]$" or "^$") then
533 fail("syntax")
534 end
535 file = body
536 end
537 end
538 local legal = pos:legal()
539 if #legal == 0 then
540 fail("game over")
541 end
542 local move
543 if castling then
544 local k = body == "O-O"
545 local problem = pos:castlingProblem(k)
546 if problem then
547 fail(problem)
548 end
549 move = pos:castlingMove(k)
550 else
551 local function matches(m)
552 return pos.board[m.s]:upper() == kind and m.t == target and not pos:castle(m) and
553 (file == "" or files:sub(m.s % 8 + 1, m.s % 8 + 1) == file) and
554 (rank == "" or ranks:sub(m.s // 8 + 1, m.s // 8 + 1) == rank) and
555 (kind ~= "P" or (m.s % 8 ~= m.t % 8) == capture)
556 end
557 local candidates = {}
558 for _, m in ipairs(legal) do
559 if matches(m) then
560 candidates[#candidates + 1] = m
561 end
562 end
563 if #candidates == 0 then
564 for _, m in ipairs(pos:pseudo()) do
565 if matches(m) then
566 fail("leaves king in check")
567 end
568 end
569 fail("no such move")
570 end
571 move = candidates[1]
572 for _, m in ipairs(candidates) do
573 if m.s ~= move.s then
574 fail("ambiguous")
575 end
576 end
577 if pos:capture(move) ~= capture then
578 fail("capture mark")
579 end
580 if move.p ~= "" and promotion == "" then
581 fail("missing promotion")
582 end
583 if promotion ~= "" and move.p == "" then
584 fail("invalid promotion")
585 end
586 if kind ~= "P" and file .. rank ~= pos:disambiguation(move) then
587 fail("disambiguation")
588 end
589 move = {s = move.s, t = move.t, p = promotion}
590 end
591 if pos:mark(move) ~= mark then
592 fail("check mark")
593 end
594 return move
595end
596local function readUci(pos, text)
597 if
598 (#text ~= 4 and #text ~= 5) or square(text:sub(1, 2)) < 0 or square(text:sub(3, 4)) < 0 or
599 (#text == 5 and not contains("qrbn", text:sub(5)))
600 then
601 fail("syntax")
602 end
603 local s, t, p = square(text:sub(1, 2)), square(text:sub(3, 4)), text:sub(5):upper()
604 local legal = pos:legal()
605 if #legal == 0 then
606 fail("game over")
607 end
608 if pos.board[s] == "." or color(pos.board[s]) ~= pos.side then
609 fail("no piece")
610 end
611 for _, m in ipairs(legal) do
612 if m.s == s and m.t == t then
613 if m.p ~= "" and p == "" then
614 fail("missing promotion")
615 end
616 if p ~= "" and m.p == "" then
617 fail("invalid promotion")
618 end
619 return {s = s, t = t, p = p}
620 end
621 end
622 for _, m in ipairs(pos:pseudo()) do
623 if m.s == s and m.t == t then
624 fail("leaves king in check")
625 end
626 end
627 fail("no such move")
628end
629local function perft(p, d, stats)
630 if d == 0 then
631 return 1
632 end
633 local total = 0
634 for _, m in ipairs(p:legal()) do
635 local q = p:play(m)
636 total = total + perft(q, d - 1, stats)
637 if d == 1 and stats then
638 stats[1] = stats[1] + 1
639 if p:capture(m) then
640 stats[2] = stats[2] + 1
641 end
642 if p:enpassant(m) then
643 stats[3] = stats[3] + 1
644 end
645 if p:castle(m) then
646 stats[4] = stats[4] + 1
647 end
648 if m.p ~= "" then
649 stats[5] = stats[5] + 1
650 end
651 if q:check() then
652 stats[6] = stats[6] + 1
653 if #q:legal() == 0 then
654 stats[7] = stats[7] + 1
655 end
656 end
657 end
658 end
659 return total
660end

704function Game:result()
705 local p = self:current()
706 local state = p:state()
707 if state == "checkmate" then
708 return p.side == "b" and "1-0" or "0-1"
709 end

712local game = Game.new()
713local function execute(line)
714 local a = split(line)
715 local cmd = table.remove(a, 1)
716 if
717 not contains(
718 " NEW LOAD FEN BOARD MOVE UCI LEGAL STATUS UNDO PERFT DIVIDE ATTACKS PINS HISTORY PGN ",
719 " " .. cmd .. " "
720 )
721 then
722 emit("ERROR unknown command")
723 return
724 end
725 local bad =
726 contains(" NEW FEN BOARD STATUS PINS HISTORY PGN ", " " .. cmd .. " ") and #a ~= 0 or
727 (cmd == "MOVE" or cmd == "UCI") and #a == 0
728 if cmd == "LEGAL" then
729 bad = #a > 1 or #a == 1 and square(a[1]) < 0
730 end
731 if cmd == "ATTACKS" then
732 bad = #a ~= 1 or #a == 1 and square(a[1]) < 0
733 end
734 local n = -1
735 if cmd == "UNDO" or cmd == "PERFT" or cmd == "DIVIDE" then
736 n = #a == 1 and number(a[1], 1, cmd == "UNDO" and 9999 or 3) or -1
737 bad = n < 0
738 end
739 if bad then
740 emit("ERROR bad arguments")
741 return
742 end
743 local p = game:current()
744 if cmd == "NEW" then
745 game = Game.new()
746 emit("OK")
747 elseif cmd == "LOAD" then
748 local ok, q = pcall(parseFen, a)
749 if ok then
750 game = Game.new(q, true)
751 emit("OK")
752 else
753 emit("ERROR bad fen (" .. q .. ")")
754 end
755 elseif cmd == "FEN" then
756 emit(p:fen())
757 elseif cmd == "BOARD" then
758 for r = 7, 0, -1 do
759 local row = {r + 1}
760 for f = 0, 7 do
761 row[#row + 1] = p.board[8 * r + f]
762 end
763 emit(table.concat(row, " "))
764 end
765 emit(" a b c d e f g h")
766 elseif cmd == "MOVE" or cmd == "UCI" then
767 local played = {}
768 local failure
769 for i, text in ipairs(a) do
770 local ok, m = pcall(cmd == "MOVE" and readSan or readUci, game:current(), text)
771 if not ok then
772 failure = "ILLEGAL move " .. i .. ": " .. text .. " (" .. m .. ")"
773 break
774 end
775 game:play(m)
776 played[#played + 1] = game.records[#game.records].san
777 end
778 if cmd == "UCI" and #played > 0 then
779 emit("san " .. table.concat(played, " "))
780 end
781 if failure then
782 emit(failure)
783 elseif cmd == "MOVE" then
784 emit("OK")
785 end
786 elseif cmd == "LEGAL" then
787 local from = #a == 1 and square(a[1]) or -1
788 local list = {}
789 for _, m in ipairs(p:legal()) do
790 if from < 0 or m.s == from then
791 list[#list + 1] = p:san(m)
792 end
793 end
794 table.sort(list)
795 local row = {"legal", #list}
796 for _, s in ipairs(list) do
797 row[#row + 1] = s
798 end
799 emit(table.concat(row, " "))
800 elseif cmd == "STATUS" then
801 local row = {
802 "status",
803 p.side == "w" and "white" or "black",
804 p:state(),
805 "rep",
806 game:repetition()
807 }
808 for _, s in ipairs(game:draws()) do
809 row[#row + 1] = s
810 end
811 emit(table.concat(row, " "))
812 elseif cmd == "UNDO" then
813 if n > #game.records then
814 emit("ERROR cannot undo")
815 else
816 for _ = 1, n do
817 table.remove(game.records)
818 table.remove(game.positions)
819 end
820 emit("OK")
821 end
822 elseif cmd == "PERFT" then
823 local stats = {0, 0, 0, 0, 0, 0, 0}
824 perft(p, n, stats)
825 local row = {"perft", n}
826 for i, f in ipairs({"nodes", "captures", "ep", "castles", "promotions", "checks", "mates"}) do
827 row[#row + 1] = f
828 row[#row + 1] = stats[i]
829 end
830 emit(table.concat(row, " "))
831 elseif cmd == "DIVIDE" then
832 local rows = {}
833 local total = 0
834 for _, m in ipairs(p:legal()) do
835 local v = perft(p:play(m), n - 1)
836 rows[#rows + 1] = {p:san(m), v}
837 total = total + v
838 end
839 table.sort(
840 rows,
841 function(x, y)
842 return x[1] < y[1]
843 end
844 )
845 for _, r in ipairs(rows) do
846 emit(r[1] .. " " .. r[2])
847 end
848 emit("total " .. total)
849 elseif cmd == "ATTACKS" then
850 local row = {"attacks", a[1]}
851 for _, c in ipairs({"w", "b"}) do
852 local found = attackers(p.board, square(a[1]), c)
853 row[#row + 1] = c == "w" and "white" or "black"
854 row[#row + 1] = #found
855 for _, s in ipairs(found) do
856 row[#row + 1] = name(s)
857 end
858 end
859 emit(table.concat(row, " "))
860 elseif cmd == "PINS" then
861 local pins = p:pins()
862 local row = {"pins", #pins}
863 for _, pair in ipairs(pins) do
864 row[#row + 1] = name(pair[1]) .. "-" .. name(pair[2])
865 end
866 emit(table.concat(row, " "))
867 elseif cmd == "HISTORY" then
868 emit("history " .. #game.records)
869 for i, r in ipairs(game.records) do
870 emit(i .. " " .. r.uci .. " " .. r.san .. " " .. r.captured)
871 end
872 elseif cmd == "PGN" then
873 local result = game:result()
874 emit('[Result "' .. result .. '"]')
875 if game.loaded then
876 emit('[SetUp "1"]')
877 emit('[FEN "' .. game.positions[1]:fen() .. '"]')
878 end
879 local row = {}
880 for i, r in ipairs(game.records) do
881 local q = game.positions[i]
882 if q.side == "w" then
883 row[#row + 1] = q.full .. "."
884 elseif i == 1 then
885 row[#row + 1] = q.full .. "..."
886 end
887 row[#row + 1] = r.san
888 end
889 row[#row + 1] = result
890 emit(table.concat(row, " "))
891 end
892end
