53function targets(board, s) {
54 const piece = board[s],
55 kind = piece.toUpperCase();
56 if ("PNK".includes(kind)) {
57 const moves =
58 kind === "P"
59 ? [
60 [-1, color(piece) === "w" ? 1 : -1],
61 [1, color(piece) === "w" ? 1 : -1],
62 ]
63 : kind === "N"
64 ? KNIGHT
65 : KING;
66 return moves.map(([dx, dy]) => shift(s, dx, dy)).filter((t) => t >= 0);
67 }
68 const directions = kind === "Q" ? KING : kind === "R" ? ROOK : BISHOP,
69 result = [];
70 for (const [dx, dy] of directions)
71 for (let t = shift(s, dx, dy); t >= 0; t = shift(t, dx, dy)) {
72 result.push(t);
73 if (board[t] !== ".") break;
74 }
75 return result;

84class Position {
85 constructor(board, side, rights, ep, half, full) {
86 this.board = board;
87 this.side = side;

94 check(c = this.side) {
95 return attackers(this.board, this.board.indexOf(colored("K", c)), opposite(c)).length > 0;
96 }
97 castle(m) {
98 return this.board[m.s].toUpperCase() === "K" && Math.abs(m.t - m.s) === 2;
99 }
100 enpassant(m) {
101 return (
102 this.board[m.s].toUpperCase() === "P" && m.s % 8 !== m.t % 8 && this.board[m.t] === "."

104 }

106 return this.board[m.t] !== "." || this.enpassant(m);

108 captured(m) {

111 : this.enpassant(m)

115 play(m) {
116 const b = [...this.board],
117 piece = b[m.s],
118 kind = piece.toUpperCase();
119 if (this.enpassant(m)) b[m.t - step(this.side)] = ".";
120 if (this.castle(m)) {
121 const r = RULES[colored(m.t > m.s ? "K" : "Q", this.side)];
122 b[r[3]] = b[r[1]];
123 b[r[1]] = ".";
124 }
125 b[m.t] = m.p ? colored(m.p, this.side) : piece;
126 b[m.s] = ".";
127 const rights = [...this.rights]
128 .filter(
129 (r) =>
130 (kind !== "K" || color(r) !== this.side) && ![m.s, m.t].includes(RULES[r][1]),
131 )
132 .join("");
133 return new Position(
134 b,
135 opposite(this.side),
136 rights,
137 kind === "P" && Math.abs(m.t - m.s) === 16 ? (m.s + m.t) / 2 : -1,
138 kind === "P" || this.capture(m) ? 0 : this.half + 1,
139 this.full + (this.side === "b" ? 1 : 0),
140 );
141 }
142 pseudo() {
143 const result = [];
144 for (let s = 0; s < 64; s++) {
145 const p = this.board[s];
146 if (p === "." || color(p) !== this.side) continue;
147 const pawn = p.toUpperCase() === "P";
148 let a = [];
149 if (pawn) {
150 const d = step(this.side),
151 rank = Math.floor(s / 8);
152 if (this.board[s + d] === ".") {
153 a.push(s + d);
154 if (rank === (this.side === "w" ? 1 : 6) && this.board[s + 2 * d] === ".")
155 a.push(s + 2 * d);
156 }
157 a.push(
158 ...targets(this.board, s).filter(
159 (t) =>
160 (this.board[t] !== "." && color(this.board[t]) !== this.side) ||
161 t === this.ep,
162 ),
163 );
164 } else
165 a = targets(this.board, s).filter(
166 (t) => this.board[t] === "." || color(this.board[t]) !== this.side,
167 );
168 for (const t of a)
169 for (const promotion of pawn && Math.floor(t / 8) === (this.side === "w" ? 7 : 0)
170 ? "QRBN"
171 : [""])
172 result.push({ s, t, p: promotion });
173 }
174 return result;

176 castlingMove(kingside) {
177 const r = RULES[colored(kingside ? "K" : "Q", this.side)];
178 return { s: r[0], t: r[2], p: "" };
179 }
180 castlingProblem(kingside) {
181 const right = colored(kingside ? "K" : "Q", this.side),
182 r = RULES[right];
183 if (!this.rights.includes(right)) return "no castling right";
184 if (r[4].some((s) => this.board[s] !== ".")) return "path blocked";
185 if (this.check()) return "in check";
186 if (attackers(this.board, r[3], opposite(this.side)).length) return "through check";
187 if (this.play(this.castlingMove(kingside)).check(this.side)) return "into check";
188 return "";
189 }
190 legal() {
191 if (this.cached !== null) return this.cached;
192 const a = this.pseudo().filter((m) => !this.play(m).check(this.side));
193 for (const k of [true, false]) if (!this.castlingProblem(k)) a.push(this.castlingMove(k));
194 this.cached = a;
195 return a;
196 }
197 mark(m) {
198 const q = this.play(m);
199 return q.check() ? (q.legal().length ? "+" : "#") : "";
200 }
201 disambiguation(m) {
202 const others = this.legal().filter(
203 (x) =>
204 !this.castle(x) &&
205 x.t === m.t &&
206 x.s !== m.s &&
207 this.board[x.s] === this.board[m.s],
208 );
209 if (!others.length) return "";
210 if (others.every((x) => x.s % 8 !== m.s % 8)) return FILES[m.s % 8];
211 if (others.every((x) => Math.floor(x.s / 8) !== Math.floor(m.s / 8)))
212 return RANKS[Math.floor(m.s / 8)];
213 return name(m.s);
214 }
215 san(m) {
216 const kind = this.board[m.s].toUpperCase();
217 let text;
218 if (this.castle(m)) text = m.t > m.s ? "O-O" : "O-O-O";
219 else if (kind === "P")
220 text =
221 (this.capture(m) ? FILES[m.s % 8] + "x" : "") + name(m.t) + (m.p ? "=" + m.p : "");
222 else text = kind + this.disambiguation(m) + (this.capture(m) ? "x" : "") + name(m.t);
223 return text + this.mark(m);
224 }

272 found.push([first, t]);
273 break;
274 }

280 const rows = [];

297 rows.join("/"),

299 this.rights || "-",
300 this.ep < 0 ? "-" : name(this.ep),

302 this.full,
303 ].join(" ");
304 }

306function parseFen(a) {
307 if (a.length !== 6) fail("fields");
308 const ranks = a[0].split("/"),
309 board = Array(64).fill(".");
310 if (ranks.length !== 8) fail("placement");
311 for (let i = 0; i < 8; i++) {
312 let f = 0,
313 digit = false;
314 for (const c of ranks[i]) {
315 if ("12345678".includes(c)) {
316 if (digit) fail("placement");
317 f += +c;
318 digit = true;
319 } else {
320 if (!"KQRBNPkqrbnp".includes(c) || f >= 8) fail("placement");
321 board[(7 - i) * 8 + f++] = c;
322 digit = false;
323 }
324 }
325 if (f !== 8) fail("placement");
326 }
327 if (board.filter((p) => p === "K").length !== 1 || board.filter((p) => p === "k").length !== 1)
328 fail("kings");
329 if ([...board.slice(0, 8), ...board.slice(56)].some((p) => p === "P" || p === "p"))
330 fail("pawns");
331 const side = a[1];
332 if (!["w", "b"].includes(side)) fail("side");
333 if (a[2] !== "-" && !/^(?=.+)K?Q?k?q?$/.test(a[2])) fail("castling");
334 const rights = a[2] === "-" ? "" : a[2];
335 for (const r of rights)
336 if (
337 board[RULES[r][0]] !== colored("K", color(r)) ||
338 board[RULES[r][1]] !== colored("R", color(r))
339 )
340 fail("castling");
341 let ep = -1;
342 if (a[3] !== "-") {
343 ep = square(a[3]);
344 const d = step(side);
345 if (
346 ep < 0 ||
347 Math.floor(ep / 8) !== (side === "w" ? 5 : 2) ||
348 board[ep] !== "." ||
349 board[ep + d] !== "." ||
350 board[ep - d] !== colored("P", opposite(side))
351 )
352 fail("en passant");
353 }
354 const half = number(a[4], 0, 999),
355 full = number(a[5], 1, 999);
356 if (half < 0 || full < 0 || (ep >= 0 && half !== 0)) fail("clocks");
357 const p = new Position(board, side, rights, ep, half, full);
358 if (p.check(opposite(side))) fail("check");
359 return p;
360}
361function readSan(pos, text) {
362 const mark = /[+#]$/.test(text) ? text.at(-1) : "";
363 let body = mark ? text.slice(0, -1) : text;
364 const castling = body === "O-O" || body === "O-O-O";
365 let p = "",
366 target = -1,
367 capture = false,
368 kind = "P",
369 file = "",
370 rank = "";
371 if (!castling) {
372 if (body.length >= 2 && body.at(-2) === "=") {
373 p = body.at(-1);
374 if (!"QRBN".includes(p)) fail("syntax");
375 body = body.slice(0, -2);
376 }
377 target = square(body.slice(-2));
378 if (target < 0) fail("syntax");
379 body = body.slice(0, -2);
380 capture = body.endsWith("x");
381 if (capture) body = body.slice(0, -1);
382 if (body && "KQRBN".includes(body[0])) {
383 kind = body[0];
384 const hints = /^([a-h]?)([1-8]?)$/.exec(body.slice(1));
385 if (!hints || p) fail("syntax");
386 file = hints[1];
387 rank = hints[2];
388 } else {
389 if (!(capture ? /^[a-h]$/ : /^$/).test(body)) fail("syntax");
390 file = body;
391 }
392 }
393 const legal = pos.legal();
394 if (!legal.length) fail("game over");
395 let move;
396 if (castling) {
397 const kingside = body === "O-O",
398 problem = pos.castlingProblem(kingside);
399 if (problem) fail(problem);
400 move = pos.castlingMove(kingside);
401 } else {
402 const matches = (m) =>
403 pos.board[m.s].toUpperCase() === kind &&
404 m.t === target &&
405 !pos.castle(m) &&
406 (!file || FILES[m.s % 8] === file) &&
407 (!rank || RANKS[Math.floor(m.s / 8)] === rank) &&
408 (kind !== "P" || (m.s % 8 !== m.t % 8) === capture);
409 const candidates = legal.filter(matches);
410 if (!candidates.length)
411 fail(pos.pseudo().some(matches) ? "leaves king in check" : "no such move");
412 if (new Set(candidates.map((m) => m.s)).size > 1) fail("ambiguous");
413 move = candidates[0];
414 if (pos.capture(move) !== capture) fail("capture mark");
415 if (move.p && !p) fail("missing promotion");
416 if (p && !move.p) fail("invalid promotion");
417 if (kind !== "P" && file + rank !== pos.disambiguation(move)) fail("disambiguation");
418 move = { ...move, p };
419 }
420 if (pos.mark(move) !== mark) fail("check mark");
421 return move;
422}
423function readUci(pos, text) {
424 if (!/^[a-h][1-8][a-h][1-8][qrbn]?$/.test(text)) fail("syntax");
425 const s = square(text.slice(0, 2)),
426 t = square(text.slice(2, 4)),
427 p = text.slice(4).toUpperCase(),
428 legal = pos.legal();
429 if (!legal.length) fail("game over");
430 if (pos.board[s] === "." || color(pos.board[s]) !== pos.side) fail("no piece");
431 const found = legal.find((m) => m.s === s && m.t === t);
432 if (!found)
433 fail(
434 pos.pseudo().some((m) => m.s === s && m.t === t)
435 ? "leaves king in check"
436 : "no such move",
437 );
438 if (found.p && !p) fail("missing promotion");
439 if (p && !found.p) fail("invalid promotion");
440 return { s, t, p };
441}
442function perft(p, d, stats = null) {
443 if (!d) return 1;
444 let total = 0;
445 for (const m of p.legal()) {
446 const q = p.play(m);
447 total += perft(q, d - 1, stats);
448 if (d === 1 && stats) {
449 stats[0]++;
450 stats[1] += +p.capture(m);
451 stats[2] += +p.enpassant(m);
452 stats[3] += +p.castle(m);
453 stats[4] += +(m.p !== "");
454 if (q.check()) {
455 stats[5]++;
456 if (!q.legal().length) stats[6]++;
457 }
458 }
459 }
460 return total;
461}
462class Game {
463 constructor(start = parseFen(INITIAL.split(" ")), loaded = false) {
464 this.positions = [start];

491 }

494 state = p.state();
495 return state === "checkmate"
496 ? p.side === "b"

505function execute(line) {
506 const [cmd, ...a] = line.split(" "),
507 commands =
508 "NEW LOAD FEN BOARD MOVE UCI LEGAL STATUS UNDO PERFT DIVIDE ATTACKS PINS HISTORY PGN".split(
509 " ",
510 );
511 if (!commands.includes(cmd)) {
512 emit("ERROR unknown command");
513 return;
514 }
515 let n = -1;
516 const noargs = ["NEW", "FEN", "BOARD", "STATUS", "PINS", "HISTORY", "PGN"];
517 let bad =
518 (noargs.includes(cmd) && a.length !== 0) ||
519 (["MOVE", "UCI"].includes(cmd) && a.length === 0);
520 if (cmd === "LEGAL") bad = a.length > 1 || (a.length === 1 && square(a[0]) < 0);
521 if (cmd === "ATTACKS") bad = a.length !== 1 || square(a[0]) < 0;
522 if (["UNDO", "PERFT", "DIVIDE"].includes(cmd)) {
523 n = a.length === 1 ? number(a[0], 1, cmd === "UNDO" ? 9999 : 3) : -1;
524 bad = n < 0;
525 }
526 if (bad) {
527 emit("ERROR bad arguments");
528 return;
529 }
530 const p = game.current();
531 if (cmd === "NEW") {
532 game = new Game();
533 emit("OK");
534 } else if (cmd === "LOAD") {
535 try {
536 const q = parseFen(a);
537 game = new Game(q, true);
538 emit("OK");
539 } catch (e) {
540 emit(`ERROR bad fen (${e.message})`);
541 }
542 } else if (cmd === "FEN") emit(p.fen());
543 else if (cmd === "BOARD") {
544 for (let r = 7; r >= 0; r--) emit([r + 1, ...p.board.slice(8 * r, 8 * r + 8)].join(" "));
545 emit(" a b c d e f g h");
546 } else if (cmd === "MOVE" || cmd === "UCI") {
547 const played = [];
548 let error = "";
549 for (let i = 0; i < a.length; i++) {
550 try {
551 const m =
552 cmd === "MOVE" ? readSan(game.current(), a[i]) : readUci(game.current(), a[i]);
553 game.play(m);
554 played.push(game.records.at(-1).san);
555 } catch (e) {
556 error = `ILLEGAL move ${i + 1}: ${a[i]} (${e.message})`;
557 break;
558 }
559 }
560 if (cmd === "UCI" && played.length) emit("san " + played.join(" "));
561 if (error) emit(error);
562 else if (cmd === "MOVE") emit("OK");
563 } else if (cmd === "LEGAL") {
564 const from = a.length ? square(a[0]) : -1,
565 list = p
566 .legal()
567 .filter((m) => from < 0 || m.s === from)
568 .map((m) => p.san(m))
569 .sort();
570 emit(["legal", list.length, ...list].join(" "));
571 } else if (cmd === "STATUS")
572 emit(
573 [
574 "status",
575 p.side === "w" ? "white" : "black",
576 p.state(),
577 "rep",
578 game.repetition(),
579 ...game.draws(),
580 ].join(" "),
581 );
582 else if (cmd === "UNDO") {
583 if (n > game.records.length) emit("ERROR cannot undo");
584 else {
585 game.records.splice(-n);
586 game.positions.splice(-n);
587 emit("OK");
588 }
589 } else if (cmd === "PERFT") {
590 const stats = Array(7).fill(0);
591 perft(p, n, stats);
592 emit(
593 [
594 "perft",
595 n,
596 ...["nodes", "captures", "ep", "castles", "promotions", "checks", "mates"].flatMap(
597 (f, i) => [f, stats[i]],
598 ),
599 ].join(" "),
600 );
601 } else if (cmd === "DIVIDE") {
602 const rows = p
603 .legal()
604 .map((m) => [p.san(m), perft(p.play(m), n - 1)])
605 .sort((x, y) => (x[0] < y[0] ? -1 : x[0] > y[0] ? 1 : 0));
606 for (const row of rows) emit(row.join(" "));
607 emit("total " + rows.reduce((s, r) => s + r[1], 0));
608 } else if (cmd === "ATTACKS") {
609 const t = square(a[0]),
610 w = attackers(p.board, t, "w"),
611 b = attackers(p.board, t, "b");
612 emit(
613 [
614 "attacks",
615 a[0],
616 "white",
617 w.length,
618 ...w.map(name),
619 "black",
620 b.length,
621 ...b.map(name),
622 ].join(" "),
623 );
624 } else if (cmd === "PINS") {
625 const pins = p.pins();
626 emit(["pins", pins.length, ...pins.map(([s, t]) => name(s) + "-" + name(t))].join(" "));
627 } else if (cmd === "HISTORY") {
628 emit("history " + game.records.length);
629 game.records.forEach((r, i) => emit(`${i + 1} ${r.uci} ${r.san} ${r.captured}`));
630 } else if (cmd === "PGN") {
631 const result = game.result();
632 emit(`[Result "${result}"]`);
633 if (game.loaded) {
634 emit('[SetUp "1"]');
635 emit(`[FEN "${game.positions[0].fen()}"]`);
636 }
637 const words = [];
638 game.records.forEach((r, i) => {
639 const q = game.positions[i];
640 if (q.side === "w") words.push(q.full + ".");
641 else if (!i) words.push(q.full + "...");
642 words.push(r.san);
643 });
644 words.push(result);
645 emit(words.join(" "));
646 }
