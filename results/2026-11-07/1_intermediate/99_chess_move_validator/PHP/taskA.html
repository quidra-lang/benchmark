53function shiftSquare(int $s, int $dx, int $dy): int
54{
55 $x = ($s % 8) + $dx;
56 $y = intdiv($s, 8) + $dy;
57 return $x >= 0 && $x < 8 && $y >= 0 && $y < 8 ? 8 * $y + $x : -1;

59function targets(array $board, int $s): array
60{
61 $piece = $board[$s];
62 $kind = strtoupper($piece);
63 $result = [];
64 if (in_array($kind, ["P", "N", "K"], true)) {
65 $dirs =
66 $kind === "P"
67 ? [[-1, color($piece) === "w" ? 1 : -1], [1, color($piece) === "w" ? 1 : -1]]
68 : ($kind === "N"
69 ? KNIGHT
70 : KING);
71 foreach ($dirs as [$dx, $dy]) {
72 $t = shiftSquare($s, $dx, $dy);
73 if ($t >= 0) {
74 $result[] = $t;
75 }
76 }
77 return $result;
78 }
79 $dirs = $kind === "Q" ? KING : ($kind === "R" ? ROOK : BISHOP);
80 foreach ($dirs as [$dx, $dy]) {
81 for ($t = shiftSquare($s, $dx, $dy); $t >= 0; $t = shiftSquare($t, $dx, $dy)) {
82 $result[] = $t;
83 if ($board[$t] !== ".") {
84 break;
85 }
86 };
87 }
88 return $result;
89}

99 $a[] = $s;

101 }
102 return $a;

134 public function castle(Move $m): bool

156 public function play(Move $m): Position
157 {
158 $b = $this->board;
159 $piece = $b[$m->source];
160 $kind = strtoupper($piece);
161 if ($this->enpassant($m)) {
162 $b[$m->target - step($this->side)] = ".";
163 }
164 if ($this->castle($m)) {
165 $r = RULES[colored($m->target > $m->source ? "K" : "Q", $this->side)];
166 $b[$r[3]] = $b[$r[1]];
167 $b[$r[1]] = ".";
168 }
169 $b[$m->target] = $m->promotion !== "" ? colored($m->promotion, $this->side) : $piece;
170 $b[$m->source] = ".";
171 $rights = "";
172 foreach (str_split($this->rights) as $r) {
173 if (
174 ($kind !== "K" || color($r) !== $this->side) &&
175 !in_array(RULES[$r][1], [$m->source, $m->target], true)
176 ) {
177 $rights .= $r;
178 }
179 }
180 return new Position(
181 $b,
182 opposite($this->side),
183 $rights,
184 $kind === "P" && abs($m->target - $m->source) === 16
185 ? intdiv($m->source + $m->target, 2)
186 : -1,
187 $kind === "P" || $this->capture($m) ? 0 : $this->half + 1,
188 $this->full + ($this->side === "b" ? 1 : 0),
189 );
190 }
191 public function pseudo(): array
192 {
193 $result = [];
194 foreach ($this->board as $s => $piece) {
195 if ($piece === "." || color($piece) !== $this->side) {
196 continue;
197 }
198 $pawn = strtoupper($piece) === "P";
199 $a = [];
200 if ($pawn) {
201 $d = step($this->side);
202 if ($this->board[$s + $d] === ".") {
203 $a[] = $s + $d;
204 if (
205 intdiv($s, 8) === ($this->side === "w" ? 1 : 6) &&
206 $this->board[$s + 2 * $d] === "."
207 ) {
208 $a[] = $s + 2 * $d;
209 }
210 }
211 foreach (targets($this->board, $s) as $t) {
212 if (
213 ($this->board[$t] !== "." && color($this->board[$t]) !== $this->side) ||
214 $t === $this->ep
215 ) {
216 $a[] = $t;
217 }
218 }
219 } else {
220 foreach (targets($this->board, $s) as $t) {
221 if ($this->board[$t] === "." || color($this->board[$t]) !== $this->side) {
222 $a[] = $t;
223 }
224 };
225 }
226 foreach ($a as $t) {
227 foreach (
228 $pawn && intdiv($t, 8) === ($this->side === "w" ? 7 : 0)
229 ? ["Q", "R", "B", "N"]
230 : [""]
231 as $p
232 ) {
233 $result[] = new Move($s, $t, $p);
234 };
235 }
236 }
237 return $result;

239 public function castlingMove(bool $kingside): Move

244 public function castlingProblem(bool $kingside): string
245 {
246 $right = colored($kingside ? "K" : "Q", $this->side);
247 $r = RULES[$right];
248 if (!str_contains($this->rights, $right)) {
249 return "no castling right";
250 }
251 foreach ($r[4] as $s) {
252 if ($this->board[$s] !== ".") {
253 return "path blocked";
254 }
255 }
256 if ($this->check()) {
257 return "in check";
258 }
259 if (attackers($this->board, $r[3], opposite($this->side))) {
260 return "through check";
261 }
262 if ($this->play($this->castlingMove($kingside))->check($this->side)) {
263 return "into check";
264 }
265 return "";

267 public function legal(): array
268 {
269 if ($this->cached !== null) {
270 return $this->cached;
271 }
272 $a = [];
273 foreach ($this->pseudo() as $m) {
274 if (!$this->play($m)->check($this->side)) {
275 $a[] = $m;
276 }
277 }
278 foreach ([true, false] as $k) {
279 if ($this->castlingProblem($k) === "") {
280 $a[] = $this->castlingMove($k);
281 }
282 }
283 return $this->cached = $a;

292 $a = [];
293 foreach ($this->legal() as $x) {
294 if (
295 !$this->castle($x) &&
296 $x->target === $m->target &&

304 return "";
305 }

307 $rank = true;

309 $file = $file && $s % 8 !== $m->source % 8;
310 $rank = $rank && intdiv($s, 8) !== intdiv($m->source, 8);

318 public function san(Move $m): string
319 {
320 $kind = strtoupper($this->board[$m->source]);
321 if ($this->castle($m)) {
322 $text = $m->target > $m->source ? "O-O" : "O-O-O";
323 } elseif ($kind === "P") {
324 $text =
325 ($this->capture($m) ? FILES[$m->source % 8] . "x" : "") .
326 squareName($m->target) .
327 ($m->promotion !== "" ? "=" . $m->promotion : "");
328 } else {
329 $text =
330 $kind .
331 $this->disambiguation($m) .
332 ($this->capture($m) ? "x" : "") .
333 squareName($m->target);
334 }

360 public function insufficient(): bool
361 {
362 $a = [];
363 foreach ($this->board as $s => $p) {
364 if ($p !== "." && strtoupper($p) !== "K") {
365 if (str_contains("PRQ", strtoupper($p))) {
366 return false;
367 }
368 $a[] = $s;
369 }
370 }

372 return true;
373 }

375 foreach ($a as $s) {
376 if (strtoupper($this->board[$s]) !== "B" || (($s % 8) + intdiv($s, 8)) % 2 !== $shade) {
377 return false;
378 }

382 public function pins(): array
383 {
384 $king = array_search(colored("K", $this->side), $this->board, true);
385 $found = [];
386 foreach (KING as [$dx, $dy]) {
387 $first = -1;
388 for ($t = shiftSquare($king, $dx, $dy); $t >= 0; $t = shiftSquare($t, $dx, $dy)) {
389 $p = $this->board[$t];
390 if ($p === ".") {
391 continue;
392 }
393 if ($first < 0) {
394 if (color($p) !== $this->side) {
395 break;
396 }
397 $first = $t;
398 } else {
399 if (
400 color($p) !== $this->side &&
401 str_contains("Q" . ($dx === 0 || $dy === 0 ? "R" : "B"), strtoupper($p))
402 ) {
403 $found[] = [$first, $t];
404 }
405 break;
406 }
407 }
408 }
409 usort($found, fn($a, $b) => $a[0] <=> $b[0]);
410 return $found;
411 }

415 for ($r = 7; $r >= 0; $r--) {
416 $rows[] = preg_replace_callback(

419 implode("", array_slice($this->board, 8 * $r, 8)),

421 }
422 return implode(" ", [
423 implode("/", $rows),
424 $this->side,
425 $this->rights !== "" ? $this->rights : "-",
426 $this->ep < 0 ? "-" : squareName($this->ep),
427 $this->half,
428 $this->full,
429 ]);
430 }

432function parseFen(array $a): Position
433{
434 if (count($a) !== 6) {
435 fail("fields");
436 }
437 $ranks = explode("/", $a[0]);
438 if (count($ranks) !== 8) {
439 fail("placement");
440 }
441 $board = array_fill(0, 64, ".");
442 foreach ($ranks as $i => $row) {
443 $f = 0;
444 $digit = false;
445 foreach (str_split($row) as $c) {
446 if (str_contains("12345678", $c)) {
447 if ($digit) {
448 fail("placement");
449 }
450 $f += (int) $c;
451 $digit = true;
452 } else {
453 if (!str_contains("KQRBNPkqrbnp", $c) || $f >= 8) {
454 fail("placement");
455 }
456 $board[(7 - $i) * 8 + $f++] = $c;
457 $digit = false;
458 }
459 }
460 if ($f !== 8) {
461 fail("placement");
462 }
463 }
464 if (count(array_keys($board, "K", true)) !== 1 || count(array_keys($board, "k", true)) !== 1) {
465 fail("kings");
466 }
467 foreach (array_merge(array_slice($board, 0, 8), array_slice($board, 56)) as $p) {
468 if ($p === "P" || $p === "p") {
469 fail("pawns");
470 }
471 }
472 $side = $a[1];
473 if ($side !== "w" && $side !== "b") {
474 fail("side");
475 }
476 if ($a[2] !== "-" && !preg_match("/\A(?=.)K?Q?k?q?\z/", $a[2])) {
477 fail("castling");
478 }
479 $rights = $a[2] === "-" ? "" : $a[2];
480 foreach (str_split($rights) as $r) {
481 if (
482 $board[RULES[$r][0]] !== colored("K", color($r)) ||
483 $board[RULES[$r][1]] !== colored("R", color($r))
484 ) {
485 fail("castling");
486 }
487 }
488 $ep = -1;
489 if ($a[3] !== "-") {
490 $ep = square($a[3]);
491 $d = step($side);
492 if (
493 $ep < 0 ||
494 intdiv($ep, 8) !== ($side === "w" ? 5 : 2) ||
495 $board[$ep] !== "." ||
496 $board[$ep + $d] !== "." ||
497 $board[$ep - $d] !== colored("P", opposite($side))
498 ) {
499 fail("en passant");
500 }
501 }
502 $half = number($a[4], 0, 999);
503 $full = number($a[5], 1, 999);
504 if ($half < 0 || $full < 0 || ($ep >= 0 && $half !== 0)) {
505 fail("clocks");
506 }
507 $p = new Position($board, $side, $rights, $ep, $half, $full);
508 if ($p->check(opposite($side))) {
509 fail("check");
510 }
511 return $p;
512}
513function readSan(Position $pos, string $text): Move
514{
515 $mark = str_ends_with($text, "+") || str_ends_with($text, "#") ? substr($text, -1) : "";
516 $body = $mark !== "" ? substr($text, 0, -1) : $text;
517 $castling = $body === "O-O" || $body === "O-O-O";
518 $promotion = "";
519 $target = -1;
520 $capture = false;
521 $kind = "P";
522 $file = "";
523 $rank = "";
524 if (!$castling) {
525 if (strlen($body) >= 2 && $body[strlen($body) - 2] === "=") {
526 $promotion = substr($body, -1);
527 if (!str_contains("QRBN", $promotion)) {
528 fail("syntax");
529 }
530 $body = substr($body, 0, -2);
531 }
532 $target = square(substr($body, -2));
533 if ($target < 0) {
534 fail("syntax");
535 }
536 $body = substr($body, 0, -2);
537 $capture = str_ends_with($body, "x");
538 if ($capture) {
539 $body = substr($body, 0, -1);
540 }
541 if ($body !== "" && str_contains("KQRBN", $body[0])) {
542 $kind = $body[0];
543 if ($promotion !== "" || !preg_match("/\A([a-h]?)([1-8]?)\z/", substr($body, 1), $h)) {
544 fail("syntax");
545 }
546 $file = $h[1];
547 $rank = $h[2];
548 } else {
549 if (!preg_match($capture ? "/\A[a-h]\z/" : "/\A\z/", $body)) {
550 fail("syntax");
551 }
552 $file = $body;
553 }
554 }
555 $legal = $pos->legal();
556 if (!$legal) {
557 fail("game over");
558 }
559 if ($castling) {
560 $k = $body === "O-O";
561 $problem = $pos->castlingProblem($k);
562 if ($problem !== "") {
563 fail($problem);
564 }
565 $move = $pos->castlingMove($k);
566 } else {
567 $matches = fn($m) => strtoupper($pos->board[$m->source]) === $kind &&
568 $m->target === $target &&
569 !$pos->castle($m) &&
570 ($file === "" || FILES[$m->source % 8] === $file) &&
571 ($rank === "" || RANKS[intdiv($m->source, 8)] === $rank) &&
572 ($kind !== "P" || ($m->source % 8 !== $m->target % 8) === $capture);
573 $candidates = array_values(array_filter($legal, $matches));
574 if (!$candidates) {
575 fail(array_filter($pos->pseudo(), $matches) ? "leaves king in check" : "no such move");
576 }
577 if (count(array_unique(array_map(fn($m) => $m->source, $candidates))) > 1) {
578 fail("ambiguous");
579 }
580 $move = $candidates[0];
581 if ($pos->capture($move) !== $capture) {
582 fail("capture mark");
583 }
584 if ($move->promotion !== "" && $promotion === "") {
585 fail("missing promotion");
586 }
587 if ($promotion !== "" && $move->promotion === "") {
588 fail("invalid promotion");
589 }
590 if ($kind !== "P" && $file . $rank !== $pos->disambiguation($move)) {
591 fail("disambiguation");
592 }
593 $move = new Move($move->source, $move->target, $promotion);
594 }
595 if ($pos->mark($move) !== $mark) {
596 fail("check mark");
597 }
598 return $move;
599}
600function readUci(Position $pos, string $text): Move
601{
602 if (!preg_match("/\A[a-h][1-8][a-h][1-8][qrbn]?\z/", $text)) {
603 fail("syntax");
604 }
605 $s = square(substr($text, 0, 2));
606 $t = square(substr($text, 2, 2));
607 $p = strtoupper(substr($text, 4));
608 $legal = $pos->legal();
609 if (!$legal) {
610 fail("game over");
611 }
612 if ($pos->board[$s] === "." || color($pos->board[$s]) !== $pos->side) {
613 fail("no piece");
614 }
615 foreach ($legal as $m) {
616 if ($m->source === $s && $m->target === $t) {
617 if ($m->promotion !== "" && $p === "") {
618 fail("missing promotion");
619 }
620 if ($p !== "" && $m->promotion === "") {
621 fail("invalid promotion");
622 }
623 return new Move($s, $t, $p);
624 }
625 }
626 foreach ($pos->pseudo() as $m) {
627 if ($m->source === $s && $m->target === $t) {
628 fail("leaves king in check");
629 }
630 }
631 fail("no such move");
632}
633function perft(Position $p, int $depth, ?array &$stats = null): int
634{
635 if ($depth === 0) {
636 return 1;
637 }
638 $total = 0;
639 foreach ($p->legal() as $m) {
640 $q = $p->play($m);
641 $total += perft($q, $depth - 1, $stats);
642 if ($depth === 1 && $stats !== null) {
643 $stats[0]++;
644 $stats[1] += (int) $p->capture($m);
645 $stats[2] += (int) $p->enpassant($m);
646 $stats[3] += (int) $p->castle($m);
647 $stats[4] += (int) ($m->promotion !== "");
648 if ($q->check()) {
649 $stats[5]++;
650 $stats[6] += (int) !$q->legal();
651 }
652 }
653 }
654 return $total;
655}
656class Game
657{
658 public array $positions,
659 $records = [];
660 public function __construct(?Position $start = null, public bool $loaded = false)
661 {
662 $this->positions = [$start ?? parseFen(explode(" ", INITIAL))];
663 }
664 public function current(): Position
665 {
666 return $this->positions[count($this->positions) - 1];
667 }
668 public function play(Move $m): void
669 {
670 $p = $this->current();
671 $this->records[] = [
672 "san" => $p->san($m),
673 "uci" => squareName($m->source) . squareName($m->target) . strtolower($m->promotion),
674 "captured" => $p->captured($m),
675 ];
676 $this->positions[] = $p->play($m);
677 }
678 public function repetition(): int
679 {
680 $key = $this->current()->key();
681 return count(array_filter($this->positions, fn($p) => $p->key() === $key));
682 }
683 public function draws(): array
684 {
685 $p = $this->current();
686 $a = [];
687 if ($p->half >= 100) {
688 $a[] = "fifty";
689 }
690 if ($this->repetition() >= 3) {
691 $a[] = "threefold";
692 }
693 if ($p->insufficient()) {
694 $a[] = "insufficient";
695 }
696 return $a;
697 }
698 public function result(): string
699 {
700 $p = $this->current();
701 $state = $p->state();
702 return $state === "checkmate"
703 ? ($p->side === "b"
704 ? "1-0"
705 : "0-1")
706 : ($state === "stalemate" || $this->draws()
707 ? "1/2-1/2"
708 : "*");
709 }
710}
711function execute(Game &$game, string $line): void
712{
713 $w = explode(" ", $line);
714 $cmd = array_shift($w);
715 $a = $w;
716 $commands = explode(
717 " ",
718 "NEW LOAD FEN BOARD MOVE UCI LEGAL STATUS UNDO PERFT DIVIDE ATTACKS PINS HISTORY PGN",
719 );
720 if (!in_array($cmd, $commands, true)) {
721 emit("ERROR unknown command");
722 return;
723 }
724 $bad =
725 in_array($cmd, ["NEW", "FEN", "BOARD", "STATUS", "PINS", "HISTORY", "PGN"], true) &&
726 count($a) !== 0;
727 if ($cmd === "MOVE" || $cmd === "UCI") {
