57 static List<Integer> targets(char[] b, int s) {
58 char p = b[s], k = Character.toUpperCase(p);
59 List<Integer> a = new ArrayList<>();
60 if (k == 'P' || k == 'N' || k == 'K') {
61 int[][] dirs = k == 'P' ? new int[][] {{-1, color(p) == 'w' ? 1 : -1},
62 {1, color(p) == 'w' ? 1 : -1}}
63 : k == 'N' ? KNIGHT
64 : KING;
65 for (int[] d : dirs) {
66 int t = shift(s, d[0], d[1]);
67 if (t >= 0)
68 a.add(t);
69 }
70 return a;
71 }
72 int[][] dirs = k == 'Q' ? KING : k == 'R' ? ROOK : BISHOP;
73 for (int[] d : dirs)
74 for (int t = shift(s, d[0], d[1]); t >= 0; t = shift(t, d[0], d[1])) {
75 a.add(t);
76 if (b[t] != '.')
77 break;
78 }
79 return a;

130 Position play(Move m) {
131 char[] b = board.clone();
132 char piece = b[m.source], kind = Character.toUpperCase(piece);
133 if (enpassant(m))
134 b[m.target - step(side)] = '.';
135 if (castle(m)) {
136 Rule r = RULES.get(colored(m.target > m.source ? 'K' : 'Q', side));
137 b[r.rookTarget] = b[r.rook];
138 b[r.rook] = '.';
139 }
140 b[m.target] = m.promotion != 0 ? colored(m.promotion, side) : piece;
141 b[m.source] = '.';
142 StringBuilder rights = new StringBuilder();
143 for (char r : this.rights.toCharArray())
144 if ((kind != 'K' || color(r) != side) && RULES.get(r).rook != m.source &&
145 RULES.get(r).rook != m.target)
146 rights.append(r);
147 return new Position(
148 b, opposite(side), rights.toString(),
149 kind == 'P' && Math.abs(m.target - m.source) == 16 ? (m.source + m.target) / 2 : -1,
150 kind == 'P' || capture(m) ? 0 : half + 1, full + (side == 'b' ? 1 : 0));
151 }
152 List<Move> pseudo() {
153 List<Move> result = new ArrayList<>();
154 for (int s = 0; s < 64; s++) {
155 char p = board[s];
156 if (p == '.' || color(p) != side)
157 continue;
158 boolean pawn = Character.toUpperCase(p) == 'P';
159 List<Integer> a = new ArrayList<>();
160 if (pawn) {
161 int d = step(side);
162 if (board[s + d] == '.') {
163 a.add(s + d);
164 if (s / 8 == (side == 'w' ? 1 : 6) && board[s + 2 * d] == '.')
165 a.add(s + 2 * d);
166 }
167 for (int t : targets(board, s))
168 if ((board[t] != '.' && color(board[t]) != side) || t == ep)
169 a.add(t);
170 } else
171 for (int t : targets(board, s))
172 if (board[t] == '.' || color(board[t]) != side)
173 a.add(t);
174 for (int t : a)
175 if (pawn && t / 8 == (side == 'w' ? 7 : 0)) {
176 for (char promotion : "QRBN".toCharArray())
177 result.add(new Move(s, t, promotion));
178 } else
179 result.add(new Move(s, t, (char)0));
180 }
181 return result;
182 }

189 Rule r = RULES.get(right);

285 if (color(p) != side)

293 break;
294 }
295 }
296 }

309 else {

311 row.append(empty);

314 }
315 }

317 row.append(empty);
318 rows.add(row.toString());
319 }

321 " " + (ep < 0 ? "-" : name(ep)) + " " + half + " " + full;

324 static Position parseFen(String[] a) {
325 if (a.length != 6)
326 fail("fields");
327 String[] ranks = a[0].split("/", -1);
328 if (ranks.length != 8)
329 fail("placement");
330 char[] b = new char[64];
331 Arrays.fill(b, '.');
332 for (int i = 0; i < 8; i++) {
333 int f = 0;
334 boolean digit = false;
335 for (char c : ranks[i].toCharArray())
336 if (c >= '1' && c <= '8') {
337 if (digit)
338 fail("placement");
339 f += c - '0';
340 digit = true;
341 } else {
342 if ("KQRBNPkqrbnp".indexOf(c) < 0 || f >= 8)
343 fail("placement");
344 b[(7 - i) * 8 + f++] = c;
345 digit = false;
346 }
347 if (f != 8)
348 fail("placement");
349 }
350 int white = 0, black = 0;
351 for (char c : b) {
352 if (c == 'K')
353 white++;
354 if (c == 'k')
355 black++;
356 }
357 if (white != 1 || black != 1)
358 fail("kings");
359 for (int i = 0; i < 8; i++)
360 if (Character.toUpperCase(b[i]) == 'P' || Character.toUpperCase(b[56 + i]) == 'P')
361 fail("pawns");
362 if (!a[1].equals("w") && !a[1].equals("b"))
363 fail("side");
364 char side = a[1].charAt(0);
365 if (!a[2].equals("-") && (a[2].isEmpty() || !a[2].matches("K?Q?k?q?")))
366 fail("castling");
367 String rights = a[2].equals("-") ? "" : a[2];
368 for (char r : rights.toCharArray()) {
369 Rule rule = RULES.get(r);
370 if (b[rule.king] != colored('K', color(r)) || b[rule.rook] != colored('R', color(r)))
371 fail("castling");
372 }
373 int ep = -1;
374 if (!a[3].equals("-")) {
375 ep = square(a[3]);
376 int d = step(side);
377 if (ep < 0 || ep / 8 != (side == 'w' ? 5 : 2) || b[ep] != '.' || b[ep + d] != '.' ||
378 b[ep - d] != colored('P', opposite(side)))
379 fail("en passant");
380 }
381 int half = number(a[4], 0, 999), full = number(a[5], 1, 999);
382 if (half < 0 || full < 0 || (ep >= 0 && half != 0))
383 fail("clocks");
384 Position p = new Position(b, side, rights, ep, half, full);
385 if (p.check(opposite(side)))
386 fail("check");
387 return p;
388 }
389 static Move readSan(Position pos, String text) {
390 String mark = text.endsWith("+") || text.endsWith("#") ? text.substring(text.length() - 1)
391 : "",
392 body = mark.isEmpty() ? text : text.substring(0, text.length() - 1);
393 boolean castling = body.equals("O-O") || body.equals("O-O-O"), capture = false;
394 char promotion = 0, kind = 'P';
395 String file = "", rank = "";
396 int target = -1;
397 if (!castling) {
398 if (body.length() >= 2 && body.charAt(body.length() - 2) == '=') {
399 promotion = body.charAt(body.length() - 1);
400 if ("QRBN".indexOf(promotion) < 0)
401 fail("syntax");
402 body = body.substring(0, body.length() - 2);
403 }
404 if (body.length() < 2)
405 fail("syntax");
406 target = square(body.substring(body.length() - 2));
407 if (target < 0)
408 fail("syntax");
409 body = body.substring(0, body.length() - 2);
410 capture = body.endsWith("x");
411 if (capture)
412 body = body.substring(0, body.length() - 1);
413 if (!body.isEmpty() && "KQRBN".indexOf(body.charAt(0)) >= 0) {
414 kind = body.charAt(0);
415 Matcher h = Pattern.compile("([a-h]?)([1-8]?)").matcher(body.substring(1));
416 if (promotion != 0 || !h.matches())
417 fail("syntax");
418 file = h.group(1);
419 rank = h.group(2);
420 } else {
421 if (!body.matches(capture ? "[a-h]" : ""))
422 fail("syntax");
423 file = body;
424 }
425 }
426 List<Move> legal = pos.legal();
427 if (legal.isEmpty())
428 fail("game over");
429 Move move;
430 if (castling) {
431 boolean k = body.equals("O-O");
432 String problem = pos.castlingProblem(k);
433 if (!problem.isEmpty())
434 fail(problem);
435 move = pos.castlingMove(k);
436 } else {
437 List<Move> candidates = new ArrayList<>();
438 for (Move m : legal)
439 if (matches(pos, m, kind, target, file, rank, capture))
440 candidates.add(m);
441 if (candidates.isEmpty()) {
442 for (Move m : pos.pseudo())
443 if (matches(pos, m, kind, target, file, rank, capture))
444 fail("leaves king in check");
445 fail("no such move");
446 }
447 move = candidates.get(0);
448 for (Move m : candidates)
449 if (m.source != move.source)
450 fail("ambiguous");
451 if (pos.capture(move) != capture)
452 fail("capture mark");
453 if (move.promotion != 0 && promotion == 0)
454 fail("missing promotion");
455 if (promotion != 0 && move.promotion == 0)
456 fail("invalid promotion");
457 if (kind != 'P' && !(file + rank).equals(pos.disambiguation(move)))
458 fail("disambiguation");
459 move = new Move(move.source, move.target, promotion);
460 }
461 if (!pos.mark(move).equals(mark))
462 fail("check mark");
463 return move;
464 }
465 static boolean matches(Position p, Move m, char kind, int target, String file, String rank,
466 boolean capture) {
467 return Character.toUpperCase(p.board[m.source]) == kind && m.target == target &&
468 !p.castle(m) && (file.isEmpty() || FILES.charAt(m.source % 8) == file.charAt(0)) &&
469 (rank.isEmpty() || RANKS.charAt(m.source / 8) == rank.charAt(0)) &&
470 (kind != 'P' || (m.source % 8 != m.target % 8) == capture);
471 }
472 static Move readUci(Position p, String text) {
473 if (!text.matches("[a-h][1-8][a-h][1-8][qrbn]?"))
474 fail("syntax");
475 int s = square(text.substring(0, 2)), t = square(text.substring(2, 4));
476 char promotion = text.length() == 5 ? Character.toUpperCase(text.charAt(4)) : 0;
477 List<Move> legal = p.legal();
478 if (legal.isEmpty())
479 fail("game over");
480 if (p.board[s] == '.' || color(p.board[s]) != p.side)
481 fail("no piece");
482 for (Move m : legal)
483 if (m.source == s && m.target == t) {
484 if (m.promotion != 0 && promotion == 0)
485 fail("missing promotion");
486 if (promotion != 0 && m.promotion == 0)
487 fail("invalid promotion");
488 return new Move(s, t, promotion);
489 }
490 for (Move m : p.pseudo())
491 if (m.source == s && m.target == t)
492 fail("leaves king in check");
493 fail("no such move");
494 return null;
495 }
496 static long perft(Position p, int d, long[] stats) {
497 if (d == 0)
498 return 1;
499 long total = 0;
500 for (Move m : p.legal()) {
501 Position q = p.play(m);
502 total += perft(q, d - 1, stats);
503 if (d == 1 && stats != null) {
504 stats[0]++;
505 if (p.capture(m))
506 stats[1]++;
507 if (p.enpassant(m))
508 stats[2]++;
509 if (p.castle(m))
510 stats[3]++;
511 if (m.promotion != 0)
512 stats[4]++;
513 if (q.check()) {
514 stats[5]++;
515 if (q.legal().isEmpty())
516 stats[6]++;
517 }
518 }
519 }
520 return total;
521 }

571 static void execute(String line) {
572 String[] w = line.split(" "), a = Arrays.copyOfRange(w, 1, w.length);
573 String cmd = w[0];
574 if (!Set.of("NEW", "LOAD", "FEN", "BOARD", "MOVE", "UCI", "LEGAL", "STATUS", "UNDO",
575 "PERFT", "DIVIDE", "ATTACKS", "PINS", "HISTORY", "PGN")
576 .contains(cmd)) {
577 emit("ERROR unknown command");
578 return;
579 }
580 boolean bad =
581 Set.of("NEW", "FEN", "BOARD", "STATUS", "PINS", "HISTORY", "PGN").contains(cmd) &&
582 a.length != 0;
583 if (cmd.equals("MOVE") || cmd.equals("UCI"))
584 bad = a.length == 0;
585 if (cmd.equals("LEGAL"))
586 bad = a.length > 1 || (a.length == 1 && square(a[0]) < 0);
587 if (cmd.equals("ATTACKS"))
588 bad = a.length != 1 || (a.length == 1 && square(a[0]) < 0);
589 int n = -1;
590 if (Set.of("UNDO", "PERFT", "DIVIDE").contains(cmd)) {
591 n = a.length == 1 ? number(a[0], 1, cmd.equals("UNDO") ? 9999 : 3) : -1;
592 bad = n < 0;
593 }
594 if (bad) {
595 emit("ERROR bad arguments");
596 return;
597 }
598 Position p = game.current();
599 switch (cmd) {
600 case "NEW" -> {
601 game = new Game();
602 emit("OK");
603 }
604 case "LOAD" -> {
605 try {
606 Position q = parseFen(a);
607 game = new Game(q, true);
608 emit("OK");
609 } catch (IllegalArgumentException e) {
610 emit("ERROR bad fen (" + e.getMessage() + ")");
611 }
612 }
613 case "FEN" -> emit(p.fen());
614 case "BOARD" -> {
615 for (int r = 7; r >= 0; r--) {
616 StringBuilder row = new StringBuilder("" + (r + 1));
617 for (int f = 0; f < 8; f++)
618 row.append(' ').append(p.board[r * 8 + f]);
619 emit(row.toString());
620 }
621 emit(" a b c d e f g h");
622 }
623 case "MOVE", "UCI" -> {
624 List<String> played = new ArrayList<>();
625 String failure = "";
626 for (int i = 0; i < a.length; i++) {
627 try {
628 Move m = cmd.equals("MOVE") ? readSan(game.current(), a[i])
629 : readUci(game.current(), a[i]);
630 game.play(m);
631 played.add(game.records.get(game.records.size() - 1).san);
632 } catch (IllegalArgumentException e) {
633 failure = "ILLEGAL move " + (i + 1) + ": " + a[i] + " (" + e.getMessage() + ")";
634 break;
635 }
636 }
637 if (cmd.equals("UCI") && !played.isEmpty())
638 emit("san " + String.join(" ", played));
639 if (!failure.isEmpty())
640 emit(failure);
641 else if (cmd.equals("MOVE"))
642 emit("OK");
643 }
644 case "LEGAL" -> {
645 List<String> list = new ArrayList<>();
646 for (Move m : p.legal())
647 if (a.length == 0 || m.source == square(a[0]))
648 list.add(p.san(m));
649 Collections.sort(list);
650 emit("legal " + list.size() + (list.isEmpty() ? "" : " " + String.join(" ", list)));
651 }
652 case "STATUS" -> {
653 List<String> marks = game.draws();
654 emit("status " + (p.side == 'w' ? "white" : "black") + " " + p.state() + " rep " +
655 game.repetition() + (marks.isEmpty() ? "" : " " + String.join(" ", marks)));
656 }
657 case "UNDO" -> {
658 if (n > game.records.size())
659 emit("ERROR cannot undo");
660 else {
661 game.records.subList(game.records.size() - n, game.records.size()).clear();
662 game.positions.subList(game.positions.size() - n, game.positions.size()).clear();
663 emit("OK");
664 }
665 }
666 case "PERFT" -> {
667 long[] stats = new long[7];
668 perft(p, n, stats);
669 StringBuilder row = new StringBuilder("perft " + n);
670 String[] fields = {"nodes", "captures", "ep", "castles",
671 "promotions", "checks", "mates"};
672 for (int i = 0; i < 7; i++)
673 row.append(' ').append(fields[i]).append(' ').append(stats[i]);
674 emit(row.toString());
675 }
676 case "DIVIDE" -> {
677 TreeMap<String, Long> rows = new TreeMap<>();
678 long total = 0;
679 for (Move m : p.legal()) {
680 long v = perft(p.play(m), n - 1, null);
681 rows.put(p.san(m), v);
682 total += v;
683 }
684 for (Map.Entry<String, Long> r : rows.entrySet())
685 emit(r.getKey() + " " + r.getValue());
686 emit("total " + total);
687 }
688 case "ATTACKS" -> {
689 StringBuilder row = new StringBuilder("attacks " + a[0]);
690 for (char c : new char[] {'w', 'b'}) {
691 List<Integer> found = attackers(p.board, square(a[0]), c);
692 row.append(' ')
693 .append(c == 'w' ? "white" : "black")
694 .append(' ')
695 .append(found.size());
696 for (int s : found)
697 row.append(' ').append(name(s));
698 }
699 emit(row.toString());
700 }
701 case "PINS" -> {
702 List<int[]> pins = p.pins();
703 StringBuilder row = new StringBuilder("pins " + pins.size());
704 for (int[] pair : pins)
705 row.append(' ').append(name(pair[0])).append('-').append(name(pair[1]));
706 emit(row.toString());
707 }
708 case "HISTORY" -> {
709 emit("history " + game.records.size());
710 for (int i = 0; i < game.records.size(); i++) {
711 Record r = game.records.get(i);
712 emit((i + 1) + " " + r.uci + " " + r.san + " " + r.captured);
713 }
714 }
715 case "PGN" -> {
716 String result = game.result();
717 emit("[Result \"" + result + "\"]");
718 if (game.loaded) {
719 emit("[SetUp \"1\"]");
720 emit("[FEN \"" + game.positions.get(0).fen() + "\"]");
721 }
722 List<String> text = new ArrayList<>();
723 for (int i = 0; i < game.records.size(); i++) {
724 Position q = game.positions.get(i);
725 if (q.side == 'w')
726 text.add(q.full + ".");
727 else if (i == 0)
728 text.add(q.full + "...");
729 text.add(game.records.get(i).san);
730 }
731 text.add(result);
732 emit(String.join(" ", text));
733 }
734 }
735 }
736 public static void main(String[] args) throws IOException {
737 BufferedReader in = new BufferedReader(new InputStreamReader(System.in));
738 int n = Integer.parseInt(in.readLine());
739 for (int i = 0; i < n; i++)
740 execute(in.readLine());
741 out.flush();
742 }
