60int square(const string &s) {
61 return s.size() == 2 && s[0] >= 'a' && s[0] <= 'h' && s[1] >= '1' && s[1] <= '8'
62 ? s[0] - 'a' + 8 * (s[1] - '1')
63 : -1;
64}
65string name(int s) {
66 return string(1, FILES[s % 8]) + RANKS[s / 8];
67}
68int number(const string &s, int lo, int hi) {
69 if (s.empty() || (s.size() > 1 && s[0] == '0') || s.size() > 9)
70 return -1;
71 int v = 0;
72 for (char c : s) {
73 if (c < '0' || c > '9')
74 return -1;
75 v = v * 10 + c - '0';
76 }
77 return v >= lo && v <= hi ? v : -1;
78}
79int shift(int s, int dx, int dy) {
80 int x = s % 8 + dx, y = s / 8 + dy;
81 return x >= 0 && x < 8 && y >= 0 && y < 8 ? 8 * y + x : -1;
82}

100vector<int> targets(const array<char, 64> &board, int s) {
101 char piece = board[s], kind = upper(piece);
102 vector<int> result;
103 if (kind == 'P' || kind == 'N' || kind == 'K') {
104 vector<pair<int, int>> dirs =
105 kind == 'P' ? vector<pair<int, int>>{{-1, color(piece) == 'w' ? 1 : -1},
106 {1, color(piece) == 'w' ? 1 : -1}}
107 : kind == 'N' ? KNIGHT
108 : KING;
109 for (auto [dx, dy] : dirs) {
110 int t = shift(s, dx, dy);
111 if (t >= 0)
112 result.push_back(t);
113 }
114 return result;
115 }
116 const vector<pair<int, int>> &dirs = kind == 'Q' ? KING : kind == 'R' ? ROOK : BISHOP;
117 for (auto [dx, dy] : dirs)
118 for (int t = shift(s, dx, dy); t >= 0; t = shift(t, dx, dy)) {
119 result.push_back(t);
120 if (board[t] != '.')
121 break;
122 }
123 return result;
124}
125vector<int> attackers(const array<char, 64> &board, int t, char c) {
126 vector<int> a;
127 for (int s = 0; s < 64; s++)
128 if (board[s] != '.' && color(board[s]) == c) {
129 vector<int> b = targets(board, s);
130 if (find(b.begin(), b.end(), t) != b.end())
131 a.push_back(s);
132 }
133 return a;
134}
135struct Position {
136 array<char, 64> board;
137 char side;
138 string rights;
139 int ep, half, full;
140 bool check(char c) const {
141 int k = int(find(board.begin(), board.end(), colored('K', c)) - board.begin());
142 return !attackers(board, k, opposite(c)).empty();
143 }
144 bool check() const {
145 return check(side);
146 }
147 bool castle(Move m) const {
148 return upper(board[m.source]) == 'K' && abs(m.target - m.source) == 2;
149 }
150 bool enpassant(Move m) const {
151 return upper(board[m.source]) == 'P' && m.source % 8 != m.target % 8 &&
152 board[m.target] == '.';
153 }
154 bool capture(Move m) const {
155 return board[m.target] != '.' || enpassant(m);
156 }
157 char captured(Move m) const {
158 return board[m.target] != '.' ? lower(board[m.target]) : enpassant(m) ? 'p' : '-';
159 }
160 Position play(Move m) const {
161 Position q = *this;
162 char piece = board[m.source], kind = upper(piece);
163 if (enpassant(m))
164 q.board[m.target - step(side)] = '.';
165 if (castle(m)) {
166 const CastleRule &r = RULES.at(colored(m.target > m.source ? 'K' : 'Q', side));
167 q.board[r.rookTarget] = q.board[r.rook];
168 q.board[r.rook] = '.';
169 }
170 q.board[m.target] = m.promotion ? colored(m.promotion, side) : piece;
171 q.board[m.source] = '.';
172 q.rights.clear();
173 for (char r : rights)
174 if ((kind != 'K' || color(r) != side) && RULES.at(r).rook != m.source &&
175 RULES.at(r).rook != m.target)
176 q.rights += r;
177 q.ep = kind == 'P' && abs(m.target - m.source) == 16 ? (m.source + m.target) / 2 : -1;
178 q.half = kind == 'P' || capture(m) ? 0 : half + 1;
179 q.full = full + (side == 'b');
180 q.side = opposite(side);
181 return q;
182 }
183 vector<Move> pseudo() const {
184 vector<Move> result;
185 for (int s = 0; s < 64; s++) {
186 char p = board[s];
187 if (p == '.' || color(p) != side)
188 continue;
189 bool pawn = upper(p) == 'P';
190 vector<int> a;
191 if (pawn) {
192 int d = step(side);
193 if (board[s + d] == '.') {
194 a.push_back(s + d);
195 if (s / 8 == (side == 'w' ? 1 : 6) && board[s + 2 * d] == '.')
196 a.push_back(s + 2 * d);
197 }
198 for (int t : targets(board, s))
199 if ((board[t] != '.' && color(board[t]) != side) || t == ep)
200 a.push_back(t);
201 } else
202 for (int t : targets(board, s))
203 if (board[t] == '.' || color(board[t]) != side)
204 a.push_back(t);
205 for (int t : a)
206 if (pawn && t / 8 == (side == 'w' ? 7 : 0)) {
207 for (char p : string("QRBN"))
208 result.push_back({s, t, p});
209 } else
210 result.push_back({s, t, 0});
211 }
212 return result;
213 }
214 Move castlingMove(bool kingside) const {
215 const CastleRule &r = RULES.at(colored(kingside ? 'K' : 'Q', side));
216 return {r.king, r.target, 0};
217 }
218 string castlingProblem(bool kingside) const {
219 char right = colored(kingside ? 'K' : 'Q', side);
220 const CastleRule &r = RULES.at(right);
221 if (rights.find(right) == string::npos)
222 return "no castling right";
223 for (int s : r.path)
224 if (board[s] != '.')
225 return "path blocked";
226 if (check())
227 return "in check";
228 if (!attackers(board, r.rookTarget, opposite(side)).empty())
229 return "through check";
230 if (play(castlingMove(kingside)).check(side))
231 return "into check";
232 return "";
233 }
234 vector<Move> legal() const {
235 vector<Move> a;
236 for (Move m : pseudo())
237 if (!play(m).check(side))
238 a.push_back(m);
239 for (bool k : {true, false})
240 if (castlingProblem(k).empty())
241 a.push_back(castlingMove(k));
242 return a;
243 }
244 string mark(Move m) const {
245 Position q = play(m);
246 return q.check() ? (q.legal().empty() ? "#" : "+") : "";
247 }
248 string disambiguation(Move m) const {
249 vector<int> a;
250 for (Move x : legal())
251 if (!castle(x) && x.target == m.target && x.source != m.source &&
252 board[x.source] == board[m.source])
253 a.push_back(x.source);
254 if (a.empty())
255 return "";
256 bool differentFile = true, differentRank = true;
257 for (int s : a) {
258 differentFile &= s % 8 != m.source % 8;
259 differentRank &= s / 8 != m.source / 8;
260 }
261 if (differentFile)
262 return string(1, FILES[m.source % 8]);
263 if (differentRank)
264 return string(1, RANKS[m.source / 8]);
265 return name(m.source);
266 }
267 string san(Move m) const {
268 string text;
269 char kind = upper(board[m.source]);
270 if (castle(m))
271 text = m.target > m.source ? "O-O" : "O-O-O";
272 else if (kind == 'P')
273 text = (capture(m) ? string(1, FILES[m.source % 8]) + "x" : "") + name(m.target) +
274 (m.promotion ? "=" + string(1, m.promotion) : "");
275 else
276 text = string(1, kind) + disambiguation(m) + (capture(m) ? "x" : "") + name(m.target);
277 return text + mark(m);
278 }
279 string key() const {
280 int target = -1;
281 if (ep >= 0)
282 for (Move m : legal())
283 if (enpassant(m)) {
284 target = ep;
285 break;
286 }
287 return string(board.begin(), board.end()) + side + rights + " " + to_string(target);
288 }
289 string state() const {
290 bool any = !legal().empty();
291 return check() ? (any ? "check" : "checkmate") : (any ? "normal" : "stalemate");
292 }
293 bool insufficient() const {
294 vector<int> a;
295 for (int s = 0; s < 64; s++)
296 if (board[s] != '.' && upper(board[s]) != 'K') {
297 if (string("PRQ").find(upper(board[s])) != string::npos)
298 return false;
299 a.push_back(s);
300 }
301 if (a.size() <= 1)
302 return true;
303 int shade = (a[0] % 8 + a[0] / 8) % 2;
304 for (int s : a)
305 if (upper(board[s]) != 'B' || (s % 8 + s / 8) % 2 != shade)
306 return false;
307 return true;
308 }
309 vector<pair<int, int>> pins() const {
310 int king = int(find(board.begin(), board.end(), colored('K', side)) - board.begin());
311 vector<pair<int, int>> result;
312 for (auto [dx, dy] : KING) {
313 int first = -1;
314 for (int t = shift(king, dx, dy); t >= 0; t = shift(t, dx, dy)) {
315 char p = board[t];
316 if (p == '.')
317 continue;
318 if (first < 0) {
319 if (color(p) != side)
320 break;
321 first = t;
322 } else {
323 if (color(p) != side &&
324 (upper(p) == 'Q' || upper(p) == (dx == 0 || dy == 0 ? 'R' : 'B')))
325 result.push_back({first, t});
326 break;
327 }
328 }
329 }
330 sort(result.begin(), result.end());
331 return result;
332 }
333 string fen() const {
334 vector<string> rows;
335 for (int r = 7; r >= 0; r--) {
336 string row;
337 int empty = 0;
338 for (int f = 0; f < 8; f++) {
339 char p = board[r * 8 + f];
340 if (p == '.')
341 empty++;
342 else {
343 if (empty)
344 row += to_string(empty);
345 empty = 0;
346 row += p;
347 }
348 }
349 if (empty)
350 row += to_string(empty);
351 rows.push_back(row);
352 }
353 return join(rows, "/") + " " + side + " " + (rights.empty() ? "-" : rights) + " " +
354 (ep < 0 ? "-" : name(ep)) + " " + to_string(half) + " " + to_string(full);
355 }

357Position parseFen(const vector<string> &a) {
358 if (a.size() != 6)
359 fail("fields");
360 vector<string> ranks;
361 size_t start = 0;
362 for (size_t i = 0; i <= a[0].size(); i++)
363 if (i == a[0].size() || a[0][i] == '/') {
364 ranks.push_back(a[0].substr(start, i - start));
365 start = i + 1;
366 }
367 if (ranks.size() != 8)
368 fail("placement");
369 Position p{};
370 p.board.fill('.');
371 for (int i = 0; i < 8; i++) {
372 int f = 0;
373 bool digit = false;
374 for (char c : ranks[i])
375 if (c >= '1' && c <= '8') {
376 if (digit)
377 fail("placement");
378 f += c - '0';
379 digit = true;
380 } else {
381 if (string("KQRBNPkqrbnp").find(c) == string::npos || f >= 8)
382 fail("placement");
383 p.board[(7 - i) * 8 + f++] = c;
384 digit = false;
385 }
386 if (f != 8)
387 fail("placement");
388 }
389 if (count(p.board.begin(), p.board.end(), 'K') != 1 ||
390 count(p.board.begin(), p.board.end(), 'k') != 1)
391 fail("kings");
392 for (int i = 0; i < 8; i++)
393 if (upper(p.board[i]) == 'P' || upper(p.board[56 + i]) == 'P')
394 fail("pawns");
395 if (a[1] != "w" && a[1] != "b")
396 fail("side");
397 p.side = a[1][0];
398 if (a[2] != "-" && (!regex_match(a[2], regex("K?Q?k?q?")) || a[2].empty()))
399 fail("castling");
400 p.rights = a[2] == "-" ? "" : a[2];
401 for (char r : p.rights) {
402 const CastleRule &rule = RULES.at(r);
403 if (p.board[rule.king] != colored('K', color(r)) ||
404 p.board[rule.rook] != colored('R', color(r)))
405 fail("castling");
406 }
407 p.ep = -1;
408 if (a[3] != "-") {
409 p.ep = square(a[3]);
410 int d = step(p.side);
411 if (p.ep < 0 || p.ep / 8 != (p.side == 'w' ? 5 : 2) || p.board[p.ep] != '.' ||
412 p.board[p.ep + d] != '.' || p.board[p.ep - d] != colored('P', opposite(p.side)))
413 fail("en passant");
414 }
415 p.half = number(a[4], 0, 999);
416 p.full = number(a[5], 1, 999);
417 if (p.half < 0 || p.full < 0 || (p.ep >= 0 && p.half != 0))
418 fail("clocks");
419 if (p.check(opposite(p.side)))
420 fail("check");
421 return p;
422}
423Move readSan(const Position &p, const string &text) {
424 string body = text, mark;
425 if (!body.empty() && (body.back() == '+' || body.back() == '#')) {
426 mark = body.back();
427 body.pop_back();
428 }
429 bool castling = body == "O-O" || body == "O-O-O", capture = false;
430 char promotion = 0, kind = 'P';
431 string file, rank;
432 int target = -1;
433 if (!castling) {
434 if (body.size() >= 2 && body[body.size() - 2] == '=') {
435 promotion = body.back();
436 if (string("QRBN").find(promotion) == string::npos)
437 fail("syntax");
438 body.resize(body.size() - 2);
439 }
440 if (body.size() < 2)
441 fail("syntax");
442 target = square(body.substr(body.size() - 2));
443 if (target < 0)
444 fail("syntax");
445 body.resize(body.size() - 2);
446 if (!body.empty() && body.back() == 'x') {
447 capture = true;
448 body.pop_back();
449 }
450 if (!body.empty() && string("KQRBN").find(body[0]) != string::npos) {
451 kind = body[0];
452 body.erase(0, 1);
453 smatch m;
454 if (promotion || !regex_match(body, m, regex("([a-h]?)([1-8]?)")))
455 fail("syntax");
456 file = m[1];
457 rank = m[2];
458 } else {
459 if (!regex_match(body, regex(capture ? "[a-h]" : "")))
460 fail("syntax");
461 file = body;
462 }
463 }
464 vector<Move> legal = p.legal();
465 if (legal.empty())
466 fail("game over");
467 Move move{};
468 if (castling) {
469 bool k = body == "O-O";
470 string problem = p.castlingProblem(k);
471 if (!problem.empty())
472 fail(problem);
473 move = p.castlingMove(k);
474 } else {
475 auto matches = [&](Move m) {
476 return upper(p.board[m.source]) == kind && m.target == target && !p.castle(m) &&
477 (file.empty() || FILES[m.source % 8] == file[0]) &&
478 (rank.empty() || RANKS[m.source / 8] == rank[0]) &&
479 (kind != 'P' || (m.source % 8 != m.target % 8) == capture);
480 };
481 vector<Move> a;
482 for (Move m : legal)
483 if (matches(m))
484 a.push_back(m);
485 if (a.empty()) {
486 for (Move m : p.pseudo())
487 if (matches(m))
488 fail("leaves king in check");
489 fail("no such move");
490 }
491 for (Move m : a)
492 if (m.source != a[0].source)
493 fail("ambiguous");
494 move = a[0];
495 if (p.capture(move) != capture)
496 fail("capture mark");
497 if (move.promotion && !promotion)
498 fail("missing promotion");
499 if (promotion && !move.promotion)
500 fail("invalid promotion");
501 if (kind != 'P' && file + rank != p.disambiguation(move))
502 fail("disambiguation");
503 move.promotion = promotion;
504 }
505 if (p.mark(move) != mark)
506 fail("check mark");
507 return move;
508}
509Move readUci(const Position &p, const string &text) {
510 if (!regex_match(text, regex("[a-h][1-8][a-h][1-8][qrbn]?")))
511 fail("syntax");
512 int s = square(text.substr(0, 2)), t = square(text.substr(2, 2));
513 char promotion = text.size() == 5 ? upper(text[4]) : 0;
514 vector<Move> legal = p.legal();
515 if (legal.empty())
516 fail("game over");
517 if (p.board[s] == '.' || color(p.board[s]) != p.side)
518 fail("no piece");
519 for (Move m : legal)
520 if (m.source == s && m.target == t) {
521 if (m.promotion && !promotion)
522 fail("missing promotion");
523 if (promotion && !m.promotion)
524 fail("invalid promotion");
525 return {s, t, promotion};
526 }
527 for (Move m : p.pseudo())
528 if (m.source == s && m.target == t)
529 fail("leaves king in check");
530 fail("no such move");
531 return {};
532}
533long long perft(const Position &p, int d, array<long long, 7> *stats = nullptr) {
534 if (!d)
535 return 1;
536 long long total = 0;
537 for (Move m : p.legal()) {
538 Position q = p.play(m);
539 total += perft(q, d - 1, stats);
540 if (d == 1 && stats) {
541 (*stats)[0]++;
542 (*stats)[1] += p.capture(m);
543 (*stats)[2] += p.enpassant(m);
544 (*stats)[3] += p.castle(m);
545 (*stats)[4] += m.promotion != 0;
546 if (q.check()) {
547 (*stats)[5]++;
548 (*stats)[6] += q.legal().empty();
549 }
550 }
551 }
552 return total;
553}
554struct Game {
555 vector<Position> positions;
556 vector<Record> records;
557 bool loaded;
558 Game(Position p = parseFen(words(INITIAL)), bool load = false) : positions{p}, loaded(load) {
559 }
560 const Position &current() const {
561 return positions.back();
562 }
563 void play(Move m) {
564 const Position &p = current();
565 records.push_back(
566 {p.san(m),
567 name(m.source) + name(m.target) + (m.promotion ? string(1, lower(m.promotion)) : ""),
568 p.captured(m)});
569 Position q = p.play(m);
570 positions.push_back(q);
571 }
572 int repetition() const {
573 string key = current().key();
574 int n = 0;
575 for (const Position &p : positions)
576 n += p.key() == key;
577 return n;
578 }
579 vector<string> draws() const {
580 vector<string> a;
581 if (current().half >= 100)
582 a.push_back("fifty");
583 if (repetition() >= 3)
584 a.push_back("threefold");
585 if (current().insufficient())
586 a.push_back("insufficient");
587 return a;
588 }
589 string result() const {
590 string s = current().state();
591 if (s == "checkmate")
592 return current().side == 'b' ? "1-0" : "0-1";
593 return s == "stalemate" || !draws().empty() ? "1/2-1/2" : "*";
594 }
595};
596void execute(Game &g, const string &line) {
597 vector<string> w = words(line);
598 string cmd = w[0];
599 vector<string> a(w.begin() + 1, w.end());
600 const set<string> commands = {"NEW", "LOAD", "FEN", "BOARD", "MOVE",
601 "UCI", "LEGAL", "STATUS", "UNDO", "PERFT",
602 "DIVIDE", "ATTACKS", "PINS", "HISTORY", "PGN"};
603 if (!commands.count(cmd)) {
604 cout << "ERROR unknown command\n";
605 return;
606 }
607 const set<string> noargs = {"NEW", "FEN", "BOARD", "STATUS", "PINS", "HISTORY", "PGN"};
608 bool bad = noargs.count(cmd) && !a.empty();
609 if (cmd == "MOVE" || cmd == "UCI")
610 bad = a.empty();
611 if (cmd == "LEGAL")
612 bad = a.size() > 1 || (a.size() == 1 && square(a[0]) < 0);
613 if (cmd == "ATTACKS")
614 bad = a.size() != 1 || (a.size() == 1 && square(a[0]) < 0);
615 int n = -1;
616 if (cmd == "UNDO" || cmd == "PERFT" || cmd == "DIVIDE") {
617 n = a.size() == 1 ? number(a[0], 1, cmd == "UNDO" ? 9999 : 3) : -1;
618 bad = n < 0;
619 }
620 if (bad) {
621 cout << "ERROR bad arguments\n";
622 return;
623 }
624 const Position p = g.current();
625 if (cmd == "NEW") {
626 g = Game();
627 cout << "OK\n";
628 } else if (cmd == "LOAD") {
629 try {
630 Position q = parseFen(a);
631 g = Game(q, true);
632 cout << "OK\n";
633 } catch (const runtime_error &e) {
634 cout << "ERROR bad fen (" << e.what() << ")\n";
635 }
636 } else if (cmd == "FEN")
637 cout << p.fen() << '\n';
638 else if (cmd == "BOARD") {
639 for (int r = 7; r >= 0; r--) {
640 cout << r + 1;
641 for (int f = 0; f < 8; f++)
642 cout << ' ' << p.board[r * 8 + f];
643 cout << '\n';
644 }
645 cout << " a b c d e f g h\n";
646 } else if (cmd == "MOVE" || cmd == "UCI") {
647 vector<string> played;
648 string failure;
649 for (size_t i = 0; i < a.size(); i++) {
650 try {
651 Move m = cmd == "MOVE" ? readSan(g.current(), a[i]) : readUci(g.current(), a[i]);
652 g.play(m);
653 played.push_back(g.records.back().san);
654 } catch (const runtime_error &e) {
655 failure = "ILLEGAL move " + to_string(i + 1) + ": " + a[i] + " (" + e.what() + ")";
656 break;
657 }
658 }
659 if (cmd == "UCI" && !played.empty())
660 cout << "san " << join(played) << '\n';
661 if (!failure.empty())
662 cout << failure << '\n';
663 else if (cmd == "MOVE")
664 cout << "OK\n";
665 } else if (cmd == "LEGAL") {
666 vector<string> list;
667 for (Move m : p.legal())
668 if (a.empty() || m.source == square(a[0]))
669 list.push_back(p.san(m));
670 sort(list.begin(), list.end());
671 cout << "legal " << list.size();
672 for (const string &s : list)
673 cout << ' ' << s;
674 cout << '\n';
675 } else if (cmd == "STATUS") {
676 cout << "status " << (p.side == 'w' ? "white" : "black") << ' ' << p.state() << " rep "
677 << g.repetition();
678 for (const string &s : g.draws())
679 cout << ' ' << s;
680 cout << '\n';
681 } else if (cmd == "UNDO") {
682 if (size_t(n) > g.records.size())
683 cout << "ERROR cannot undo\n";
684 else {
685 g.records.resize(g.records.size() - n);
686 g.positions.resize(g.positions.size() - n);
687 cout << "OK\n";
688 }
689 } else if (cmd == "PERFT") {
690 array<long long, 7> stats{};
691 perft(p, n, &stats);
692 cout << "perft " << n;
693 vector<string> fields = {"nodes", "captures", "ep", "castles",
694 "promotions", "checks", "mates"};
695 for (int i = 0; i < 7; i++)
696 cout << ' ' << fields[i] << ' ' << stats[i];
697 cout << '\n';
698 } else if (cmd == "DIVIDE") {
699 vector<pair<string, long long>> rows;
700 long long total = 0;
701 for (Move m : p.legal()) {
702 long long v = perft(p.play(m), n - 1);
703 rows.push_back({p.san(m), v});
704 total += v;
705 }
706 sort(rows.begin(), rows.end());
707 for (const auto &row : rows)
708 cout << row.first << ' ' << row.second << '\n';
709 cout << "total " << total << '\n';
710 } else if (cmd == "ATTACKS") {
711 cout << "attacks " << a[0];
712 for (char c : {'w', 'b'}) {
713 vector<int> found = attackers(p.board, square(a[0]), c);
714 cout << ' ' << (c == 'w' ? "white" : "black") << ' ' << found.size();
715 for (int s : found)
716 cout << ' ' << name(s);
717 }
718 cout << '\n';
719 } else if (cmd == "PINS") {
720 vector<pair<int, int>> pins = p.pins();
721 cout << "pins " << pins.size();
722 for (auto [s, t] : pins)
723 cout << ' ' << name(s) << '-' << name(t);
724 cout << '\n';
725 } else if (cmd == "HISTORY") {
726 cout << "history " << g.records.size() << '\n';
727 for (size_t i = 0; i < g.records.size(); i++) {
728 const Record &r = g.records[i];
729 cout << i + 1 << ' ' << r.uci << ' ' << r.san << ' ' << r.captured << '\n';
730 }
731 } else if (cmd == "PGN") {
732 string result = g.result();
733 cout << "[Result \"" << result << "\"]\n";
734 if (g.loaded)
735 cout << "[SetUp \"1\"]\n[FEN \"" << g.positions[0].fen() << "\"]\n";
736 vector<string> text;
737 for (size_t i = 0; i < g.records.size(); i++) {
738 const Position &q = g.positions[i];
739 if (q.side == 'w')
740 text.push_back(to_string(q.full) + ".");
741 else if (i == 0)
742 text.push_back(to_string(q.full) + "...");
743 text.push_back(g.records[i].san);
744 }
745 text.push_back(result);
746 cout << join(text) << '\n';
747 }
748}
