36color(p) = isuppercase(p) ? 'w' : 'b'

42function number(s, lo, hi)

47function shift(s, dx, dy)
48 x, y=s%8+dx, s÷8+dy
49 0<=x<8&&0<=y<8 ? 8*y+x : -1
50end
51function targets(b, s)
52 piece=b[s+1]
53 kind=uppercase(piece)
54 a=Int[]
55 if kind in "PNK"
56 dirs=kind=='P' ? [(-1, color(piece)=='w' ? 1 : -1), (1, color(piece)=='w' ? 1 : -1)] :
57 kind=='N' ? KNIGHT : KING
58 for (dx, dy) in dirs
59 t=shift(s, dx, dy)
60 t>=0&&push!(a, t)
61 end
62 else
63 dirs=kind=='Q' ? KING : kind=='R' ? ROOK : BISHOP
64 for (dx, dy) in dirs
65 t=shift(s, dx, dy)
66 while t>=0
67 push!(a, t)
68 b[t+1]!='.'&&break
69 t=shift(t, dx, dy)
70 end
71 end
72 end
73 a
74end
75attackers(b, t, c) = [s for s in 0:63 if b[s+1]!='.'&&color(b[s+1])==c&&t in targets(b, s)]

85Position(b, s, r, e, h, f) = Position(b, s, r, e, h, f, nothing)
86incheck(p::Position, c = p.side) =
87 !isempty(attackers(p.board, findfirst(==(colored('K', c)), p.board)-1, opposite(c)))
88iscastle(p, m) = uppercase(p.board[m.source+1])=='K'&&abs(m.target-m.source)==2
89isenpassant(p, m) =
90 uppercase(p.board[m.source+1])=='P'&&m.source%8!=m.target%8&&p.board[m.target+1]=='.'
91iscapture(p, m) = p.board[m.target+1]!='.'||isenpassant(p, m)
92captured(p, m) =
93 p.board[m.target+1]!='.' ? lowercase(p.board[m.target+1]) : isenpassant(p, m) ? 'p' : '-'
94function play(p::Position, m::Move)
95 b=copy(p.board)
96 piece=b[m.source+1]
97 kind=uppercase(piece)
98 isenpassant(p, m)&&(b[m.target-pawnstep(p.side)+1]='.')
99 if iscastle(p, m)
100 r=RULES[colored(m.target>m.source ? 'K' : 'Q', p.side)]
101 b[r.rooktarget+1]=b[r.rook+1]
102 b[r.rook+1]='.'
103 end
104 b[m.target+1]=m.promotion=='\0' ? piece : colored(m.promotion, p.side)
105 b[m.source+1]='.'
106 rights=join(
107 r for r in p.rights if
108 (kind!='K'||color(r)!=p.side)&&RULES[r].rook!=m.source&&RULES[r].rook!=m.target
109 )
110 Position(
111 b,
112 opposite(p.side),
113 rights,
114 kind=='P'&&abs(m.target-m.source)==16 ? (m.source+m.target)÷2 : -1,
115 kind=='P'||iscapture(p, m) ? 0 : p.half+1,
116 p.full+(p.side=='b'),
117 )
118end
119function pseudo(p::Position)
120 result=Move[]
121 for s in 0:63
122 piece=p.board[s+1]
123 (piece=='.'||color(piece)!=p.side)&&continue
124 pawn=uppercase(piece)=='P'
125 a=Int[]
126 if pawn
127 d=pawnstep(p.side)
128 if p.board[s+d+1]=='.'
129 push!(a, s+d)
130 s÷8==(p.side=='w' ? 1 : 6)&&p.board[s+2*d+1]=='.'&&push!(a, s+2*d)
131 end
132 append!(
133 a,
134 [
135 t for t in targets(p.board, s) if
136 (p.board[t+1]!='.'&&color(p.board[t+1])!=p.side)||t==p.ep
137 ],
138 )
139 else
140 append!(
141 a,
142 [t for t in targets(p.board, s) if p.board[t+1]=='.'||color(p.board[t+1])!=p.side],
143 )
144 end
145 for t in a
146 if pawn&&t÷8==(p.side=='w' ? 7 : 0)
147 for promotion in "QRBN"
148 push!(result, Move(s, t, promotion))
149 end
150 else
151 push!(result, Move(s, t))
152 end
153 end
154 end
155 result
156end
157function castlingmove(p, k)
158 r=RULES[colored(k ? 'K' : 'Q', p.side)]
159 Move(r.king, r.target)
160end
161function castlingproblem(p, k)
162 right=colored(k ? 'K' : 'Q', p.side)
163 r=RULES[right]
164 right in p.rights||return "no castling right"
165 any(p.board[s+1]!='.' for s in r.path)&&return "path blocked"
166 incheck(p)&&return "in check"
167 !isempty(attackers(p.board, r.rooktarget, opposite(p.side)))&&return "through check"
168 incheck(play(p, castlingmove(p, k)), p.side)&&return "into check"
169 ""
170end
171function legal(p::Position)
172 p.cached!==nothing&&return p.cached
173 a=[m for m in pseudo(p) if !incheck(play(p, m), p.side)]
174 for k in (true, false)
175 isempty(castlingproblem(p, k))&&push!(a, castlingmove(p, k))
176 end
177 p.cached=a
178 a
179end
180function checkmark(p, m)
181 q=play(p, m)
182 incheck(q) ? (isempty(legal(q)) ? "#" : "+") : ""
183end
184function disambiguation(p, m)
185 a=[
186 x for x in legal(p) if
187 !iscastle(p, x)&&x.target==m.target&&x.source!=m.source&&p.board[x.source+1]==p.board[m.source+1]
188 ]
189 isempty(a)&&return ""
190 all(x.source%8!=m.source%8 for x in a)&&return string(FILES[m.source%8+1])
191 all(x.source÷8!=m.source÷8 for x in a)&&return string(RANKS[m.source÷8+1])
192 sqname(m.source)
193end
194function san(p, m)
195 kind=uppercase(p.board[m.source+1])
196 text=if iscastle(p, m)
197 m.target>m.source ? "O-O" : "O-O-O"
198 elseif kind=='P'
199 (iscapture(p, m) ? string(FILES[m.source%8+1], "x") : "")*sqname(m.target)*(
200 m.promotion=='\0' ? "" : "="*string(m.promotion)
201 )
202 else
203 string(kind)*disambiguation(p, m)*(iscapture(p, m) ? "x" : "")*sqname(m.target)
204 end
205 text*checkmark(p, m)
206end
207key(p) =
208 join(p.board)*" "*string(p.side)*" "*p.rights*" "*string(
209 p.ep>=0&&any(isenpassant(p, m) for m in legal(p)) ? p.ep : -1,
210 )
211state(p) =
212 incheck(p) ? (isempty(legal(p)) ? "checkmate" : "check") :
213 (isempty(legal(p)) ? "stalemate" : "normal")
214function insufficient(p)
215 a=[s for s in 0:63 if p.board[s+1]!='.'&&uppercase(p.board[s+1])!='K']
216 any(uppercase(p.board[s+1]) in "PRQ" for s in a)&&return false
217 length(a)<=1||(
218 all(uppercase(p.board[s+1])=='B' for s in a)&&length(unique([(s%8+s÷8)%2 for s in a]))==1
219 )
220end
221function pins(p)
222 k=findfirst(==(colored('K', p.side)), p.board)-1
223 a=Tuple{Int,Int}[]
224 for (dx, dy) in KING
225 first=-1
226 t=shift(k, dx, dy)
227 while t>=0
228 piece=p.board[t+1]
229 if piece!='.'
230 if first<0
231 color(piece)!=p.side&&break
232 first=t
233 else
234 color(piece)!=p.side&&(
235 uppercase(piece)=='Q'||uppercase(piece)==(dx==0||dy==0 ? 'R' : 'B')
236 )&&push!(a, (first, t))
237 break
238 end
239 end
240 t=shift(t, dx, dy)
241 end
242 end
243 sort(a)
244end
245function fen(p)
246 rows=String[]
247 for r in 7:-1:0
248 row=IOBuffer()
249 empty=0
250 for f in 0:7
251 piece=p.board[8*r+f+1]
252 if piece=='.'
253 empty+=1
254 else
255 empty>0&&print(row, empty)
256 empty=0
257 print(row, piece)
258 end
259 end
260 empty>0&&print(row, empty)
261 push!(rows, String(take!(row)))
262 end
263 join(
264 (
265 join(rows, "/"),
266 p.side,
267 isempty(p.rights) ? "-" : p.rights,
268 p.ep<0 ? "-" : sqname(p.ep),
269 p.half,
270 p.full,
271 ),
272 " ",
273 )
274end
275function parsefen(a)
276 length(a)==6||fail("fields")
277 rows=split(a[1], '/'; keepempty = true)
278 length(rows)==8||fail("placement")
279 b=fill('.', 64)
280 for (i, row) in enumerate(rows)
281 f=0
282 digit=false
283 for c in row
284 if c in '1':'8'
285 digit&&fail("placement")
286 f+=Int(c-'0')
287 digit=true
288 else
289 (c in "KQRBNPkqrbnp"&&f<8)||fail("placement")
290 b[(8-i)*8+f+1]=c
291 f+=1
292 digit=false
293 end
294 end
295 f==8||fail("placement")
296 end
297 (count(==('K'), b)==1&&count(==('k'), b)==1)||fail("kings")
298 any(c in "Pp" for c in vcat(b[1:8], b[57:64]))&&fail("pawns")
299 a[2] in ("w", "b")||fail("side")
300 side=a[2][1]
301 (a[3]=="-"||(!isempty(a[3])&&occursin(r"^K?Q?k?q?$", a[3])))||fail("castling")
302 rights=a[3]=="-" ? "" : String(a[3])
303 for r in rights
304 rule=RULES[r]
305 (b[rule.king+1]==colored('K', color(r))&&b[rule.rook+1]==colored('R', color(r)))||fail(
306 "castling",
307 )
308 end
309 ep=-1
310 if a[4]!="-"
311 ep=square(a[4])
312 d=pawnstep(side)
313 (
314 ep>=0&&ep÷8==(side=='w' ? 5 : 2)&&b[ep+1]=='.'&&b[ep+d+1]=='.'&&b[ep-d+1]==colored(
315 'P',
316 opposite(side),
317 )
318 )||fail("en passant")
319 end
320 half=number(a[5], 0, 999)
321 full=number(a[6], 1, 999)
322 (half>=0&&full>=0&&(ep<0||half==0))||fail("clocks")
323 p=Position(b, side, rights, ep, half, full)
324 incheck(p, opposite(side))&&fail("check")
325 p
326end
327function readsan(p, text)
328 mark=last(text) in "+#" ? string(last(text)) : ""
329 body=isempty(mark) ? String(text) : chop(text)
330 castling=body in ("O-O", "O-O-O")
331 promotion='\0'
332 target=-1
333 capture=false
334 kind='P'
335 file=""
336 rank=""
337 if !castling
338 if length(body)>=2&&body[end-1]=='='
339 promotion=body[end]
340 promotion in "QRBN"||fail("syntax")
341 body=chop(body; tail = 2)
342 end
343 length(body)>=2||fail("syntax")
344 target=square(body[(end-1):end])
345 target>=0||fail("syntax")
346 body=chop(body; tail = 2)
347 capture=endswith(body, "x")
348 capture&&(body=chop(body))
349 if !isempty(body)&&body[1] in "KQRBN"
350 kind=body[1]
351 h=match(r"^([a-h]?)([1-8]?)$", body[2:end])
352 (h!==nothing&&promotion=='\0')||fail("syntax")
353 file=String(h[1])
354 rank=String(h[2])
355 else
356 occursin(capture ? r"^[a-h]$" : r"^$", body)||fail("syntax")
357 file=body
358 end
359 end
360 moves=legal(p)
361 isempty(moves)&&fail("game over")
362 if castling
363 k=body=="O-O"
364 problem=castlingproblem(p, k)
365 isempty(problem)||fail(problem)
366 m=castlingmove(p, k)
367 else
368 matches(m) =
369 uppercase(p.board[m.source+1])==kind&&m.target==target&&!iscastle(p, m)&&(
370 isempty(file)||FILES[m.source%8+1]==file[1]
371 )&&(isempty(rank)||RANKS[m.source÷8+1]==rank[1])&&(
372 kind!='P'||(m.source%8!=m.target%8)==capture
373 )
374 a=filter(matches, moves)
375 isempty(a)&&fail(any(matches, pseudo(p)) ? "leaves king in check" : "no such move")
376 length(unique(m.source for m in a))>1&&fail("ambiguous")
377 m=a[1]
378 iscapture(p, m)==capture||fail("capture mark")
379 m.promotion!='\0'&&promotion=='\0'&&fail("missing promotion")
380 promotion!='\0'&&m.promotion=='\0'&&fail("invalid promotion")
381 kind!='P'&&file*rank!=disambiguation(p, m)&&fail("disambiguation")
382 m=Move(m.source, m.target, promotion)
383 end
384 checkmark(p, m)==mark||fail("check mark")
385 m
386end
387function readuci(p, text)
388 occursin(r"^[a-h][1-8][a-h][1-8][qrbn]?$", text)||fail("syntax")
389 s=square(text[1:2])
390 t=square(text[3:4])
391 promotion=length(text)==5 ? uppercase(text[5]) : '\0'
392 moves=legal(p)
393 isempty(moves)&&fail("game over")
394 (p.board[s+1]!='.'&&color(p.board[s+1])==p.side)||fail("no piece")
395 a=[m for m in moves if m.source==s&&m.target==t]
396 isempty(a)&&fail(
397 any(m.source==s&&m.target==t for m in pseudo(p)) ? "leaves king in check" : "no such move",
398 )
399 m=a[1]
400 m.promotion!='\0'&&promotion=='\0'&&fail("missing promotion")
401 promotion!='\0'&&m.promotion=='\0'&&fail("invalid promotion")
402 Move(s, t, promotion)
403end
404function perft(p, d, stats = nothing)
405 d==0&&return 1
406 total=0
407 for m in legal(p)
408 q=play(p, m)
409 total+=perft(q, d-1, stats)
410 if d==1&&stats!==nothing
411 stats[1]+=1
412 stats[2]+=iscapture(p, m)
413 stats[3]+=isenpassant(p, m)
414 stats[4]+=iscastle(p, m)
415 stats[5]+=m.promotion!='\0'
416 if incheck(q)
417 stats[6]+=1
418 stats[7]+=isempty(legal(q))
419 end
420 end
421 end
422 total
423end

434 g.records,

446function draws(g)
447 a=String[]
448 current(g).half>=100&&push!(a, "fifty")
449 repetition(g)>=3&&push!(a, "threefold")
450 insufficient(current(g))&&push!(a, "insufficient")

453function result(g)
454 s=state(current(g))
455 s=="checkmate" ? (current(g).side=='b' ? "1-0" : "0-1") :
456 s=="stalemate"||!isempty(draws(g)) ? "1/2-1/2" : "*"
457end
458function execute(g, line)
459 w=split(line)
460 cmd=w[1]
461 a=w[2:end]
462 if !(
463 cmd in split(
464 "NEW LOAD FEN BOARD MOVE UCI LEGAL STATUS UNDO PERFT DIVIDE ATTACKS PINS HISTORY PGN",
465 )
466 )
467 println("ERROR unknown command")
468 return g
469 end
470 bad=cmd in split("NEW FEN BOARD STATUS PINS HISTORY PGN")&&!isempty(a)
471 cmd in ("MOVE", "UCI")&&(bad=isempty(a))
472 cmd=="LEGAL"&&(bad=length(a)>1||(length(a)==1&&square(a[1])<0))
473 cmd=="ATTACKS"&&(bad=length(a)!=1||(length(a)==1&&square(a[1])<0))
474 n=-1
475 if cmd in ("UNDO", "PERFT", "DIVIDE")
476 n=length(a)==1 ? number(a[1], 1, cmd=="UNDO" ? 9999 : 3) : -1
477 bad=n<0
478 end
479 if bad
480 println("ERROR bad arguments")
481 return g
482 end
483 p=current(g)
484 if cmd=="NEW"
485 g=Game()
486 println("OK")
487 elseif cmd=="LOAD"
488 try
489 q=parsefen(a)
490 g=Game(q, true)
491 println("OK")
492 catch e
493 e isa InputError||rethrow()
494 println("ERROR bad fen ($(e.reason))")
495 end
496 elseif cmd=="FEN"
497 println(fen(p))
498 elseif cmd=="BOARD"
499 for r in 7:-1:0
500 println(join(vcat(string(r+1), string.(p.board[(8*r+1):(8*r+8)])), " "))
501 end
502 println(" a b c d e f g h")
503 elseif cmd in ("MOVE", "UCI")
504 played=String[]
505 failure=""
506 for (i, text) in enumerate(a)
507 try
508 m=cmd=="MOVE" ? readsan(current(g), text) : readuci(current(g), text)
509 play!(g, m)
510 push!(played, g.records[end].san)
511 catch e
512 e isa InputError||rethrow()
513 failure="ILLEGAL move $i: $text ($(e.reason))"
514 break
515 end
516 end
517 cmd=="UCI"&&!isempty(played)&&println("san ", join(played, " "))
518 if !isempty(failure)
519 println(failure)
520 elseif cmd=="MOVE"
521 println("OK")
522 end
523 elseif cmd=="LEGAL"
524 list=sort([san(p, m) for m in legal(p) if isempty(a)||m.source==square(a[1])])
525 println(join(vcat(["legal", string(length(list))], list), " "))
526 elseif cmd=="STATUS"
527 println(
528 join(
529 vcat(
530 [
531 "status",
532 p.side=='w' ? "white" : "black",
533 state(p),
534 "rep",
535 string(repetition(g)),
536 ],
537 draws(g),
538 ),
539 " ",
540 ),
541 )
542 elseif cmd=="UNDO"
543 if n>length(g.records)
544 println("ERROR cannot undo")
545 else
546 resize!(g.records, length(g.records)-n)
547 resize!(g.positions, length(g.positions)-n)
548 println("OK")
549 end
550 elseif cmd=="PERFT"
551 stats=zeros(Int, 7)
552 perft(p, n, stats)
553 row=["perft", string(n)]
554 for (i, f) in
555 enumerate(("nodes", "captures", "ep", "castles", "promotions", "checks", "mates"))
556 append!(row, [f, string(stats[i])])
557 end
558 println(join(row, " "))
559 elseif cmd=="DIVIDE"
560 rows=sort([(san(p, m), perft(play(p, m), n-1)) for m in legal(p)]; by = first)
561 for (s, v) in rows
562 println(s, " ", v)
563 end
564 println("total ", sum((last(r) for r in rows); init = 0))
565 elseif cmd=="ATTACKS"
566 row=["attacks", String(a[1])]
567 for c in ('w', 'b')
568 found=attackers(p.board, square(a[1]), c)
569 append!(row, [c=='w' ? "white" : "black", string(length(found))])
570 append!(row, sqname.(found))
571 end
572 println(join(row, " "))
573 elseif cmd=="PINS"
574 found=pins(p)
575 println(
576 join(
577 vcat(
578 ["pins", string(length(found))],
579 [sqname(s)*"-"*sqname(t) for (s, t) in found],
580 ),
581 " ",
582 ),
583 )
584 elseif cmd=="HISTORY"
585 println("history ", length(g.records))
586 for (i, r) in enumerate(g.records)
587 println(i, " ", r.uci, " ", r.san, " ", r.captured)
588 end
589 elseif cmd=="PGN"
590 r=result(g)
591 println("[Result \"$r\"]")
592 if g.loaded
593 println("[SetUp \"1\"]")
594 println("[FEN \"$(fen(g.positions[1]))\"]")
595 end
596 row=String[]
597 for (i, record) in enumerate(g.records)
598 q=g.positions[i]
599 if q.side=='w'
600 push!(row, "$(q.full).")
601 elseif i==1
602 push!(row, "$(q.full)...")
603 end
604 push!(row, record.san)
605 end
606 push!(row, r)
607 println(join(row, " "))
608 end
609 g
610end

616 end
