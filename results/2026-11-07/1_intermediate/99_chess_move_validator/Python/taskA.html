110def shift(square, file_step, rank_step):
111 file = square % 8 + file_step
112 rank = square // 8 + rank_step
113 if 0 <= file < 8 and 0 <= rank < 8:
114 return rank * 8 + file
115 return None

123def ray(square, file_step, rank_step):
124 squares = []
125 current = shift(square, file_step, rank_step)
126 while current is not None:
127 squares.append(current)
128 current = shift(current, file_step, rank_step)

132KNIGHT_TARGETS = [step_targets(square, KNIGHT_STEPS) for square in range(64)]
133KING_TARGETS = [step_targets(square, KING_STEPS) for square in range(64)]

135 'w': [step_targets(square, ((-1, 1), (1, 1))) for square in range(64)],

142def piece_targets(board, square):
143 kind = board[square].upper()
144 if kind == 'N':
145 return KNIGHT_TARGETS[square]
146 if kind == 'K':
147 return KING_TARGETS[square]
148 lines = []
149 if kind in 'RQ':
150 lines += ROOK_RAYS[square]
151 if kind in 'BQ':
152 lines += BISHOP_RAYS[square]
153 targets = []
154 for line in lines:
155 for target in line:
156 targets.append(target)
157 if board[target] is not None:
158 break
159 return targets

162def attackers(board, target, color):
163 pawn, knight, bishop, rook, queen, king = (colored(kind, color) for kind in 'PNBRQK')
164 found = [square for square in PAWN_TARGETS[opponent(color)][target] if board[square] == pawn]
165 found += [square for square in KNIGHT_TARGETS[target] if board[square] == knight]
166 found += [square for square in KING_TARGETS[target] if board[square] == king]
167 for rays, slider in ((ROOK_RAYS, rook), (BISHOP_RAYS, bishop)):
168 for line in rays[target]:
169 for square in line:
170 piece = board[square]
171 if piece is not None:
172 if piece in (slider, queen):
173 found.append(square)
174 break
175 return found

213 def play(self, move):
214 board = self.board.copy()
215 piece = board[move.source]
216 kind = piece.upper()
217 if self.is_en_passant(move):
218 board[move.target - pawn_step(self.side)] = None
219 if self.is_castling(move):
220 rule = CASTLING_RULES[castling_right(self.side, move.target > move.source)]
221 board[rule.rook_target] = board[rule.rook_square]
222 board[rule.rook_square] = None
223 board[move.target] = colored(move.promotion, self.side) if move.promotion else piece
224 board[move.source] = None
225 castling = self.castling
226 if kind == 'K':
227 castling = ''.join(right for right in castling if color_of(right) != self.side)
228 castling = ''.join(
229 right for right in castling
230 if CASTLING_RULES[right].rook_square not in (move.source, move.target)
231 )
232 double_push = kind == 'P' and abs(move.target - move.source) == 16
233 return Position(
234 board=board,
235 side=opponent(self.side),
236 castling=castling,
237 ep_square=(move.source + move.target) // 2 if double_push else None,
238 halfmove=0 if kind == 'P' or self.is_capture(move) else self.halfmove + 1,
239 fullmove=self.fullmove + 1 if self.side == 'b' else self.fullmove,
240 )

242 def pseudo_legal_moves(self):
243 moves = []
244 for square, piece in enumerate(self.board):
245 if piece is None or color_of(piece) != self.side:
246 continue
247 if piece.upper() == 'P':
248 moves.extend(self.pawn_moves(square))
249 continue
250 for target in piece_targets(self.board, square):
251 occupant = self.board[target]
252 if occupant is None or color_of(occupant) != self.side:
253 moves.append(Move(square, target))
254 return moves

256 def pawn_moves(self, square):
257 step = pawn_step(self.side)
258 start_rank, last_rank = (1, 7) if self.side == 'w' else (6, 0)
259 targets = []
260 if self.board[square + step] is None:
261 targets.append(square + step)
262 if square // 8 == start_rank and self.board[square + 2 * step] is None:
263 targets.append(square + 2 * step)
264 for target in PAWN_TARGETS[self.side][square]:
265 occupant = self.board[target]
266 enemy = occupant is not None and color_of(occupant) != self.side
267 if enemy or target == self.ep_square:
268 targets.append(target)
269 moves = []
270 for target in targets:
271 if target // 8 == last_rank:
272 moves.extend(Move(square, target, kind) for kind in PROMOTION_TYPES)
273 else:
274 moves.append(Move(square, target))
275 return moves

281 def castling_problem(self, kingside):
282 right = castling_right(self.side, kingside)
283 rule = CASTLING_RULES[right]
284 if right not in self.castling:
285 return 'no castling right'
286 if any(self.board[square] is not None for square in rule.path):
287 return 'path blocked'
288 if self.in_check():
289 return 'in check'
290 if attackers(self.board, rule.rook_target, opponent(self.side)):
291 return 'through check'
292 if self.play(self.castling_move(kingside)).in_check(self.side):
293 return 'into check'
294 return None

296 def successors(self):
297 for move in self.pseudo_legal_moves():
298 after = self.play(move)
299 if not after.in_check(self.side):
300 yield move, after
301 for kingside in (True, False):
302 if self.castling_problem(kingside) is None:
303 move = self.castling_move(kingside)
304 yield move, self.play(move)

332 def san(self, move):
333 piece = self.board[move.source]
334 target = square_name(move.target)
335 if self.is_castling(move):
336 text = 'O-O' if move.target > move.source else 'O-O-O'
337 elif piece.upper() == 'P':
338 prefix = FILES[move.source % 8] + 'x' if self.is_capture(move) else ''
339 promotion = '=' + move.promotion if move.promotion else ''
340 text = prefix + target + promotion
341 else:
342 capture = 'x' if self.is_capture(move) else ''
343 text = piece.upper() + self.disambiguation(move) + capture + target
344 return text + self.check_mark(move)

385 def fen(self):
386 rows = []
387 for rank in range(7, -1, -1):
388 cells = ''.join(piece or '1' for piece in self.board[rank * 8:rank * 8 + 8])
389 rows.append(re.sub(r'1+', lambda run: str(len(run.group())), cells))
390 ep = '-' if self.ep_square is None else square_name(self.ep_square)
391 fields = [
392 '/'.join(rows),
393 self.side,
394 self.castling or '-',
395 ep,
396 str(self.halfmove),
397 str(self.fullmove),
398 ]
399 return ' '.join(fields)

430def parse_fen(fields):
431 if len(fields) != 6:
432 raise FenError('fields')
433 placement, side, castling, ep_field, halfmove_field, fullmove_field = fields
434 rows = [parse_rank(row) for row in placement.split('/')]
435 if len(rows) != 8 or None in rows:
436 raise FenError('placement')
437 board = [piece for row in reversed(rows) for piece in row]
438 if board.count('K') != 1 or board.count('k') != 1:
439 raise FenError('kings')
440 if any(piece in ('P', 'p') for piece in board[:8] + board[56:]):
441 raise FenError('pawns')
442 if side not in ('w', 'b'):
443 raise FenError('side')
444 if castling != '-' and not re.fullmatch(r'K?Q?k?q?', castling):
445 raise FenError('castling')
446 rights = '' if castling == '-' else castling
447 for right in rights:
448 rule = CASTLING_RULES[right]
449 color = color_of(right)
450 if board[rule.king_square] != colored('K', color):
451 raise FenError('castling')
452 if board[rule.rook_square] != colored('R', color):
453 raise FenError('castling')
454 ep_square = None
455 if ep_field != '-':
456 ep_square = parse_square(ep_field)
457 if ep_square is None or not valid_en_passant(board, side, ep_square):
458 raise FenError('en passant')
459 halfmove = parse_number(halfmove_field, 0, 999)
460 fullmove = parse_number(fullmove_field, 1, 999)
461 if halfmove is None or fullmove is None or (ep_square is not None and halfmove != 0):
462 raise FenError('clocks')
463 position = Position(board, side, rights, ep_square, halfmove, fullmove)
464 if position.in_check(opponent(side)):
465 raise FenError('check')
466 return position

469def parse_san_body(body):
470 promotion = None
471 if len(body) >= 2 and body[-2] == '=':
472 promotion = body[-1]
473 if promotion not in PROMOTION_TYPES:
474 raise IllegalMoveError('syntax')
475 body = body[:-2]
476 target = parse_square(body[-2:])
477 if target is None:
478 raise IllegalMoveError('syntax')
479 body = body[:-2]
480 capture = body.endswith('x')
481 if capture:
482 body = body[:-1]
483 if body and body[0] in 'KQRBN':
484 hints = re.fullmatch(r'([a-h]?)([1-8]?)', body[1:])
485 if hints is None or promotion is not None:
486 raise IllegalMoveError('syntax')
487 return SanRequest(body[0], target, capture, hints[1], hints[2], promotion)
488 if not re.fullmatch('[a-h]' if capture else '', body):
489 raise IllegalMoveError('syntax')
490 return SanRequest('P', target, capture, body, '', promotion)

493def find_san_move(position, request):
494 def matches(move):
495 return (
496 position.board[move.source].upper() == request.kind
497 and move.target == request.target
498 and not position.is_castling(move)
499 and (not request.file_hint or FILES[move.source % 8] == request.file_hint)
500 and (not request.rank_hint or RANKS[move.source // 8] == request.rank_hint)
501 and (request.kind != 'P' or (move.source % 8 != move.target % 8) == request.capture)
502 )
503
504 candidates = [move for move in position.legal_moves if matches(move)]
505 if not candidates:
506 if any(matches(move) for move in position.pseudo_legal_moves()):
507 raise IllegalMoveError('leaves king in check')
508 raise IllegalMoveError('no such move')
509 if len({move.source for move in candidates}) > 1:
510 raise IllegalMoveError('ambiguous')
511 found = candidates[0]
512 if position.is_capture(found) != request.capture:
513 raise IllegalMoveError('capture mark')
514 if found.promotion and not request.promotion:
515 raise IllegalMoveError('missing promotion')
516 if request.promotion and not found.promotion:
517 raise IllegalMoveError('invalid promotion')
518 hint = request.file_hint + request.rank_hint
519 if request.kind != 'P' and hint != position.disambiguation(found):
520 raise IllegalMoveError('disambiguation')
521 return Move(found.source, found.target, request.promotion)

524def read_san(position, text):
525 mark = text[-1] if text[-1] in '+#' else ''
526 body = text[:-1] if mark else text
527 request = None if body in ('O-O', 'O-O-O') else parse_san_body(body)
528 if not position.legal_moves:
529 raise IllegalMoveError('game over')
530 if request is None:
531 kingside = body == 'O-O'
532 problem = position.castling_problem(kingside)
533 if problem is not None:
534 raise IllegalMoveError(problem)
535 move = position.castling_move(kingside)
536 else:
537 move = find_san_move(position, request)
538 if position.check_mark(move) != mark:
539 raise IllegalMoveError('check mark')
540 return move

543def read_uci(position, text):
544 source = parse_square(text[:2])
545 target = parse_square(text[2:4])
546 valid = (
547 len(text) in (4, 5)
548 and source is not None
549 and target is not None
550 and text[4:] in ('', 'q', 'r', 'b', 'n')
551 )
552 if not valid:
553 raise IllegalMoveError('syntax')
554 promotion = text[4:].upper() or None
555 if not position.legal_moves:
556 raise IllegalMoveError('game over')
557 piece = position.board[source]
558 if piece is None or color_of(piece) != position.side:
559 raise IllegalMoveError('no piece')
560 candidates = [
561 move for move in position.legal_moves if move.source == source and move.target == target
562 ]
563 if not candidates:
564 pseudo_legal = position.pseudo_legal_moves()
565 if any(move.source == source and move.target == target for move in pseudo_legal):
566 raise IllegalMoveError('leaves king in check')
567 raise IllegalMoveError('no such move')
568 if candidates[0].promotion and not promotion:
569 raise IllegalMoveError('missing promotion')
570 if promotion and not candidates[0].promotion:
571 raise IllegalMoveError('invalid promotion')
572 return Move(source, target, promotion)

581def perft_statistics(position, depth):
582 stats = dict.fromkeys(PERFT_FIELDS, 0)
583
584 def visit(node, remaining):
585 for move, child in node.successors():
586 if remaining > 1:
587 visit(child, remaining - 1)
588 continue
589 stats['nodes'] += 1
590 stats['captures'] += node.is_capture(move)
591 stats['ep'] += node.is_en_passant(move)
592 stats['castles'] += node.is_castling(move)
593 stats['promotions'] += move.promotion is not None
594 if child.in_check():
595 stats['checks'] += 1
596 stats['mates'] += not child.legal_moves
597
598 visit(position, depth)
599 return stats

627 position = self.current
628 marks = []
