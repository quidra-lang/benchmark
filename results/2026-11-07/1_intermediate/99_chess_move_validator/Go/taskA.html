54func upper(p byte) byte {
55 if p >= 'a' && p <= 'z' {
56 return p - 32
57 }
58 return p
59}
60func lower(p byte) byte {
61 if p >= 'A' && p <= 'Z' {
62 return p + 32
63 }
64 return p
65}
66func color(p byte) byte {
67 if p == upper(p) {
68 return 'w'
69 }
70 return 'b'
71}
72func opposite(c byte) byte {
73 if c == 'w' {
74 return 'b'
75 }
76 return 'w'
77}
78func colored(p, c byte) byte {
79 if c == 'w' {
80 return p
81 }
82 return lower(p)
83}
84func step(c byte) int {
85 if c == 'w' {
86 return 8
87 }
88 return -8
89}
90func square(s string) int {
91 if len(s) != 2 || s[0] < 'a' || s[0] > 'h' || s[1] < '1' || s[1] > '8' {
92 return -1
93 }
94 return int(s[0]-'a') + 8*int(s[1]-'1')
95}
96func name(s int) string { return string([]byte{files[s%8], ranks[s/8]}) }
97func number(s string, lo, hi int) int {
98 if s == "" || len(s) > 9 || (len(s) > 1 && s[0] == '0') {
99 return -1
100 }
101 for _, c := range s {
102 if c < '0' || c > '9' {
103 return -1
104 }
105 }
106 n, _ := strconv.Atoi(s)
107 if n < lo || n > hi {
108 return -1
109 }
110 return n
111}
112func shift(s, dx, dy int) int {
113 x, y := s%8+dx, s/8+dy
114 if x < 0 || x > 7 || y < 0 || y > 7 {
115 return -1
116 }
117 return 8*y + x
118}
119func targets(b [64]byte, s int) []int {
120 p := b[s]
121 kind := upper(p)
122 a := []int{}
123 if strings.ContainsRune("PNK", rune(kind)) {
124 dirs := king
125 if kind == 'N' {
126 dirs = knight
127 } else if kind == 'P' {
128 dy := 1
129 if color(p) == 'b' {
130 dy = -1
131 }
132 dirs = [][2]int{{-1, dy}, {1, dy}}
133 }
134 for _, d := range dirs {
135 t := shift(s, d[0], d[1])
136 if t >= 0 {
137 a = append(a, t)
138 }
139 }
140 return a
141 }
142 dirs := bishop
143 if kind == 'R' {
144 dirs = rook
145 } else if kind == 'Q' {
146 dirs = king
147 }
148 for _, d := range dirs {
149 for t := shift(s, d[0], d[1]); t >= 0; t = shift(t, d[0], d[1]) {
150 a = append(a, t)
151 if b[t] != '.' {
152 break
153 }
154 }
155 }
156 return a
157}
158func attackers(b [64]byte, t int, c byte) []int {
159 a := []int{}
160 for s, p := range b {
161 if p == '.' || color(p) != c {
162 continue
163 }
164 for _, x := range targets(b, s) {
165 if x == t {
166 a = append(a, s)
167 break
168 }
169 }
170 }
171 return a
172}

174type Position struct {

181func (p Position) check(c byte) bool {
182 for s, x := range p.board {
183 if x == colored('K', c) {
184 return len(attackers(p.board, s, opposite(c))) > 0
185 }
186 }
187 return false
188}
189func (p Position) castle(m Move) bool {
190 d := m.target - m.source
191 return upper(p.board[m.source]) == 'K' && (d == 2 || d == -2)
192}
193func (p Position) enpassant(m Move) bool {
194 return upper(p.board[m.source]) == 'P' && m.source%8 != m.target%8 && p.board[m.target] == '.'
195}
196func (p Position) capture(m Move) bool { return p.board[m.target] != '.' || p.enpassant(m) }
197func (p Position) captured(m Move) byte {
198 if p.board[m.target] != '.' {
199 return lower(p.board[m.target])
200 }
201 if p.enpassant(m) {
202 return 'p'
203 }
204 return '-'
205}
206func (p Position) play(m Move) Position {
207 q := p
208 piece := p.board[m.source]
209 kind := upper(piece)
210 if p.enpassant(m) {
211 q.board[m.target-step(p.side)] = '.'
212 }
213 if p.castle(m) {
214 right := byte('Q')
215 if m.target > m.source {
216 right = 'K'
217 }
218 r := rules[colored(right, p.side)]
219 q.board[r.rookTarget] = q.board[r.rook]
220 q.board[r.rook] = '.'
221 }
222 q.board[m.target] = piece
223 if m.promotion != 0 {
224 q.board[m.target] = colored(m.promotion, p.side)
225 }
226 q.board[m.source] = '.'
227 q.rights = ""
228 for i := range p.rights {
229 r := p.rights[i]
230 if (kind != 'K' || color(r) != p.side) && rules[r].rook != m.source && rules[r].rook != m.target {
231 q.rights += string(r)
232 }
233 }
234 q.ep = -1
235 d := m.target - m.source
236 if kind == 'P' && (d == 16 || d == -16) {
237 q.ep = (m.source + m.target) / 2
238 }
239 q.half = p.half + 1
240 if kind == 'P' || p.capture(m) {
241 q.half = 0
242 }
243 if p.side == 'b' {
244 q.full++
245 }
246 q.side = opposite(p.side)
247 return q
248}
249func (p Position) pseudo() []Move {
250 result := []Move{}
251 for s, piece := range p.board {
252 if piece == '.' || color(piece) != p.side {
253 continue
254 }
255 pawn := upper(piece) == 'P'
256 a := []int{}
257 if pawn {
258 d := step(p.side)
259 start := 1
260 if p.side == 'b' {
261 start = 6
262 }
263 if p.board[s+d] == '.' {
264 a = append(a, s+d)
265 if s/8 == start && p.board[s+2*d] == '.' {
266 a = append(a, s+2*d)
267 }
268 }
269 for _, t := range targets(p.board, s) {
270 if (p.board[t] != '.' && color(p.board[t]) != p.side) || t == p.ep {
271 a = append(a, t)
272 }
273 }
274 } else {
275 for _, t := range targets(p.board, s) {
276 if p.board[t] == '.' || color(p.board[t]) != p.side {
277 a = append(a, t)
278 }
279 }
280 }
281 last := 7
282 if p.side == 'b' {
283 last = 0
284 }
285 for _, t := range a {
286 if pawn && t/8 == last {
287 for _, promotion := range []byte("QRBN") {
288 result = append(result, Move{s, t, promotion})
289 }
290 } else {
291 result = append(result, Move{s, t, 0})
292 }
293 }
294 }
295 return result
296}
297func (p Position) castlingMove(kingside bool) Move {
298 right := byte('Q')
299 if kingside {
300 right = 'K'
301 }
302 r := rules[colored(right, p.side)]
303 return Move{r.king, r.target, 0}
304}
305func (p Position) castlingProblem(kingside bool) string {
306 right := byte('Q')
307 if kingside {
308 right = 'K'
309 }
310 right = colored(right, p.side)
311 r := rules[right]
312 if !strings.ContainsRune(p.rights, rune(right)) {
313 return "no castling right"
314 }
315 for _, s := range r.path {
316 if p.board[s] != '.' {
317 return "path blocked"
318 }
319 }
320 if p.check(p.side) {
321 return "in check"
322 }
323 if len(attackers(p.board, r.rookTarget, opposite(p.side))) > 0 {
324 return "through check"
325 }
326 if p.play(p.castlingMove(kingside)).check(p.side) {
327 return "into check"
328 }
329 return ""
330}
331func (p Position) legal() []Move {
332 a := []Move{}
333 for _, m := range p.pseudo() {
334 if !p.play(m).check(p.side) {
335 a = append(a, m)
336 }
337 }
338 for _, k := range []bool{true, false} {
339 if p.castlingProblem(k) == "" {
340 a = append(a, p.castlingMove(k))
341 }
342 }
343 return a
344}
345func (p Position) mark(m Move) string {
346 q := p.play(m)
347 if !q.check(q.side) {
348 return ""
349 }
350 if len(q.legal()) == 0 {
351 return "#"
352 }
353 return "+"
354}
355func (p Position) disambiguation(m Move) string {
356 any, file, rank := false, true, true
357 for _, x := range p.legal() {
358 if !p.castle(x) && x.target == m.target && x.source != m.source && p.board[x.source] == p.board[m.source] {
359 any = true
360 file = file && x.source%8 != m.source%8
361 rank = rank && x.source/8 != m.source/8
362 }
363 }
364 if !any {
365 return ""
366 }
367 if file {
368 return string(files[m.source%8])
369 }
370 if rank {
371 return string(ranks[m.source/8])
372 }
373 return name(m.source)
374}
375func (p Position) san(m Move) string {
376 kind := upper(p.board[m.source])
377 text := ""
378 if p.castle(m) {
379 text = "O-O-O"
380 if m.target > m.source {
381 text = "O-O"
382 }
383 } else if kind == 'P' {
384 if p.capture(m) {
385 text = string(files[m.source%8]) + "x"
386 }
387 text += name(m.target)
388 if m.promotion != 0 {
389 text += "=" + string(m.promotion)
390 }
391 } else {
392 text = string(kind) + p.disambiguation(m)
393 if p.capture(m) {
394 text += "x"
395 }
396 text += name(m.target)
397 }
398 return text + p.mark(m)
399}
400func (p Position) key() string {
401 ep := -1
402 if p.ep >= 0 {
403 for _, m := range p.legal() {
404 if p.enpassant(m) {
405 ep = p.ep
406 break
407 }
408 }
409 }
410 return string(p.board[:]) + " " + string(p.side) + " " + p.rights + " " + strconv.Itoa(ep)

412func (p Position) state() string {
413 any := len(p.legal()) > 0
414 if p.check(p.side) {
415 if any {
416 return "check"
417 }
418 return "checkmate"
419 }
420 if any {
421 return "normal"
422 }
423 return "stalemate"

425func (p Position) insufficient() bool {
426 a := []int{}
427 for s, piece := range p.board {
428 if piece != '.' && upper(piece) != 'K' {
429 if strings.ContainsRune("PRQ", rune(upper(piece))) {
430 return false
431 }
432 a = append(a, s)
433 }
434 }
435 if len(a) <= 1 {
436 return true
437 }
438 shade := (a[0]%8 + a[0]/8) % 2
439 for _, s := range a {
440 if upper(p.board[s]) != 'B' || (s%8+s/8)%2 != shade {
441 return false
442 }
443 }
444 return true
445}
446func (p Position) pins() [][2]int {
447 k := 0
448 for s, piece := range p.board {
449 if piece == colored('K', p.side) {
450 k = s
451 break
452 }
453 }
454 a := [][2]int{}
455 for _, d := range king {
456 first := -1
457 for t := shift(k, d[0], d[1]); t >= 0; t = shift(t, d[0], d[1]) {
458 piece := p.board[t]
459 if piece == '.' {
460 continue
461 }
462 if first < 0 {
463 if color(piece) != p.side {
464 break
465 }
466 first = t
467 } else {
468 slider := byte('B')
469 if d[0] == 0 || d[1] == 0 {
470 slider = 'R'
471 }
472 if color(piece) != p.side && (upper(piece) == 'Q' || upper(piece) == slider) {
473 a = append(a, [2]int{first, t})
474 }
475 break
476 }
477 }
478 }
479 sort.Slice(a, func(i, j int) bool { return a[i][0] < a[j][0] })
480 return a
481}
482func (p Position) fen() string {
483 rows := []string{}
484 for r := 7; r >= 0; r-- {
485 row := ""
486 empty := 0
487 for f := 0; f < 8; f++ {
488 piece := p.board[r*8+f]
489 if piece == '.' {
490 empty++
491 } else {
492 if empty > 0 {
493 row += strconv.Itoa(empty)
494 }
495 empty = 0
496 row += string(piece)
497 }
498 }
499 if empty > 0 {
500 row += strconv.Itoa(empty)
501 }
502 rows = append(rows, row)
503 }
504 rights, ep := p.rights, "-"
505 if rights == "" {
506 rights = "-"
507 }
508 if p.ep >= 0 {
509 ep = name(p.ep)
510 }
511 return strings.Join(rows, "/") + " " + string(p.side) + " " + rights + " " + ep + " " + strconv.Itoa(p.half) + " " + strconv.Itoa(p.full)
512}
513func parseFen(a []string) (p Position, err error) {
514 defer recoverInput(&err)
515 if len(a) != 6 {
516 fail("fields")
517 }
518 rows := strings.Split(a[0], "/")
519 if len(rows) != 8 {
520 fail("placement")
521 }
522 for s := range p.board {
523 p.board[s] = '.'
524 }
525 for i, row := range rows {
526 f := 0
527 digit := false
528 for j := range row {
529 c := row[j]
530 if c >= '1' && c <= '8' {
531 if digit {
532 fail("placement")
533 }
534 f += int(c - '0')
535 digit = true
536 } else {
537 if !strings.ContainsRune("KQRBNPkqrbnp", rune(c)) || f >= 8 {
538 fail("placement")
539 }
540 p.board[(7-i)*8+f] = c
541 f++
542 digit = false
543 }
544 }
545 if f != 8 {
546 fail("placement")
547 }
548 }
549 white, black := 0, 0
550 for _, c := range p.board {
551 if c == 'K' {
552 white++
553 }
554 if c == 'k' {
555 black++
556 }
557 }
558 if white != 1 || black != 1 {
559 fail("kings")
560 }
561 for i := 0; i < 8; i++ {
562 if upper(p.board[i]) == 'P' || upper(p.board[56+i]) == 'P' {
563 fail("pawns")
564 }
565 }
566 if a[1] != "w" && a[1] != "b" {
567 fail("side")
568 }
569 p.side = a[1][0]
570 if a[2] != "-" {
571 previous := -1
572 for i := range a[2] {
573 j := strings.IndexByte("KQkq", a[2][i])
574 if j <= previous {
575 fail("castling")
576 }
577 previous = j
578 }
579 p.rights = a[2]
580 }
581 for i := range p.rights {
582 r := p.rights[i]
583 rule := rules[r]
584 if p.board[rule.king] != colored('K', color(r)) || p.board[rule.rook] != colored('R', color(r)) {
585 fail("castling")
586 }
587 }
588 p.ep = -1
589 if a[3] != "-" {
590 p.ep = square(a[3])
591 expected := 5
592 if p.side == 'b' {
593 expected = 2
594 }
595 d := step(p.side)
596 if p.ep < 0 || p.ep/8 != expected || p.board[p.ep] != '.' || p.board[p.ep+d] != '.' || p.board[p.ep-d] != colored('P', opposite(p.side)) {
597 fail("en passant")
598 }
599 }
600 p.half = number(a[4], 0, 999)
601 p.full = number(a[5], 1, 999)
602 if p.half < 0 || p.full < 0 || (p.ep >= 0 && p.half != 0) {
603 fail("clocks")
604 }
605 if p.check(opposite(p.side)) {
606 fail("check")
607 }
608 return
609}
610func readSan(p Position, text string) (move Move, err error) {
611 defer recoverInput(&err)
612 mark, body := "", text
613 if strings.HasSuffix(body, "+") || strings.HasSuffix(body, "#") {
614 mark = body[len(body)-1:]
615 body = body[:len(body)-1]
616 }
617 castling := body == "O-O" || body == "O-O-O"
618 promotion := byte(0)
619 target := -1
620 capture := false
621 kind := byte('P')
622 file, rank := "", ""
623 if !castling {
624 if len(body) >= 2 && body[len(body)-2] == '=' {
625 promotion = body[len(body)-1]
626 if !strings.ContainsRune("QRBN", rune(promotion)) {
627 fail("syntax")
628 }
629 body = body[:len(body)-2]
630 }
631 if len(body) < 2 {
632 fail("syntax")
633 }
634 target = square(body[len(body)-2:])
635 if target < 0 {
636 fail("syntax")
637 }
638 body = body[:len(body)-2]
639 capture = strings.HasSuffix(body, "x")
640 if capture {
641 body = body[:len(body)-1]
642 }
643 if len(body) > 0 && strings.ContainsRune("KQRBN", rune(body[0])) {
644 kind = body[0]
645 h := regexp.MustCompile(`^([a-h]?)([1-8]?)$`).FindStringSubmatch(body[1:])
646 if h == nil || promotion != 0 {
647 fail("syntax")
648 }
649 file, rank = h[1], h[2]
650 } else {
651 if (capture && (len(body) != 1 || body[0] < 'a' || body[0] > 'h')) || (!capture && body != "") {
652 fail("syntax")
653 }
654 file = body
655 }
656 }
657 legal := p.legal()
658 if len(legal) == 0 {
659 fail("game over")
660 }
661 if castling {
662 k := body == "O-O"
663 if problem := p.castlingProblem(k); problem != "" {
664 fail(problem)
665 }
666 move = p.castlingMove(k)
667 } else {
668 matches := func(m Move) bool {
669 return upper(p.board[m.source]) == kind && m.target == target && !p.castle(m) && (file == "" || files[m.source%8] == file[0]) && (rank == "" || ranks[m.source/8] == rank[0]) && (kind != 'P' || (m.source%8 != m.target%8) == capture)
670 }
671 candidates := []Move{}
672 for _, m := range legal {
673 if matches(m) {
674 candidates = append(candidates, m)
675 }
676 }
677 if len(candidates) == 0 {
678 for _, m := range p.pseudo() {
679 if matches(m) {
680 fail("leaves king in check")
681 }
682 }
683 fail("no such move")
684 }
685 move = candidates[0]
686 for _, m := range candidates {
687 if m.source != move.source {
688 fail("ambiguous")
689 }
690 }
691 if p.capture(move) != capture {
692 fail("capture mark")
693 }
694 if move.promotion != 0 && promotion == 0 {
695 fail("missing promotion")
696 }
697 if promotion != 0 && move.promotion == 0 {
698 fail("invalid promotion")
699 }
700 if kind != 'P' && file+rank != p.disambiguation(move) {
701 fail("disambiguation")
702 }
703 move.promotion = promotion
704 }
705 if p.mark(move) != mark {
706 fail("check mark")
707 }
708 return
709}
710func readUci(p Position, text string) (move Move, err error) {
711 defer recoverInput(&err)
712 if !regexp.MustCompile(`^[a-h][1-8][a-h][1-8][qrbn]?$`).MatchString(text) {
713 fail("syntax")
714 }
715 s, t := square(text[:2]), square(text[2:4])
716 promotion := byte(0)
717 if len(text) == 5 {
718 promotion = upper(text[4])
719 }
720 legal := p.legal()
721 if len(legal) == 0 {
722 fail("game over")
723 }
724 if p.board[s] == '.' || color(p.board[s]) != p.side {
725 fail("no piece")
726 }
727 for _, m := range legal {
728 if m.source == s && m.target == t {
729 if m.promotion != 0 && promotion == 0 {
730 fail("missing promotion")
731 }
732 if promotion != 0 && m.promotion == 0 {
733 fail("invalid promotion")
734 }
735 return Move{s, t, promotion}, nil
736 }
737 }
738 for _, m := range p.pseudo() {
739 if m.source == s && m.target == t {
740 fail("leaves king in check")
741 }
742 }
743 fail("no such move")
744 return
745}
746func perft(p Position, d int, stats *[7]int64) int64 {
747 if d == 0 {
748 return 1
749 }
750 var total int64
751 for _, m := range p.legal() {
752 q := p.play(m)
753 total += perft(q, d-1, stats)
754 if d == 1 && stats != nil {
755 stats[0]++
756 if p.capture(m) {
757 stats[1]++
758 }
759 if p.enpassant(m) {
760 stats[2]++
761 }
762 if p.castle(m) {
763 stats[3]++
764 }
765 if m.promotion != 0 {
766 stats[4]++
767 }
768 if q.check(q.side) {
769 stats[5]++
770 if len(q.legal()) == 0 {
771 stats[6]++
772 }
773 }
774 }
775 }
776 return total
777}

789func (g *Game) current() Position { return g.positions[len(g.positions)-1] }
790func (g *Game) play(m Move) {
791 p := g.current()
792 uci := name(m.source) + name(m.target)
793 if m.promotion != 0 {
794 uci += string(lower(m.promotion))
795 }
796 g.records = append(g.records, Record{p.san(m), uci, p.captured(m)})
797 g.positions = append(g.positions, p.play(m))

799func (g *Game) repetition() int {
800 key := g.current().key()
801 n := 0
802 for _, p := range g.positions {
803 if p.key() == key {
804 n++
805 }
806 }
807 return n
808}
809func (g *Game) draws() []string {
810 a := []string{}
811 p := g.current()
812 if p.half >= 100 {
813 a = append(a, "fifty")
814 }
815 if g.repetition() >= 3 {
816 a = append(a, "threefold")
817 }
818 if p.insufficient() {
819 a = append(a, "insufficient")
820 }
821 return a
822}
823func (g *Game) result() string {
824 p := g.current()
825 state := p.state()
826 if state == "checkmate" {
827 if p.side == 'b' {
828 return "1-0"
829 }
830 return "0-1"
831 }
832 if state == "stalemate" || len(g.draws()) > 0 {
833 return "1/2-1/2"
834 }
835 return "*"
836}
837func has(a []string, s string) bool {
838 for _, x := range a {
839 if x == s {
840 return true
841 }
842 }
843 return false
844}
845func execute(game **Game, line string) {
846 w := strings.Split(line, " ")
847 cmd, a := w[0], w[1:]
848 if !has(strings.Fields("NEW LOAD FEN BOARD MOVE UCI LEGAL STATUS UNDO PERFT DIVIDE ATTACKS PINS HISTORY PGN"), cmd) {
849 emit("ERROR unknown command")
850 return
851 }
852 bad := has(strings.Fields("NEW FEN BOARD STATUS PINS HISTORY PGN"), cmd) && len(a) != 0
853 if cmd == "MOVE" || cmd == "UCI" {
854 bad = len(a) == 0
855 }
856 if cmd == "LEGAL" {
857 bad = len(a) > 1 || (len(a) == 1 && square(a[0]) < 0)
858 }
859 if cmd == "ATTACKS" {
860 bad = len(a) != 1 || (len(a) == 1 && square(a[0]) < 0)
861 }
862 n := -1
863 if has([]string{"UNDO", "PERFT", "DIVIDE"}, cmd) {
864 hi := 3
865 if cmd == "UNDO" {
866 hi = 9999
867 }
868 if len(a) == 1 {
869 n = number(a[0], 1, hi)
870 }
871 bad = n < 0
872 }
873 if bad {
874 emit("ERROR bad arguments")
875 return
876 }
877 g := *game
878 p := g.current()
879 switch cmd {
880 case "NEW":
881 *game = newGame()
882 emit("OK")
883 case "LOAD":
884 q, err := parseFen(a)
885 if err != nil {
886 emit("ERROR bad fen (" + err.Error() + ")")
887 } else {
888 *game = &Game{positions: []Position{q}, loaded: true}
889 emit("OK")
890 }
891 case "FEN":
892 emit(p.fen())
893 case "BOARD":
894 for r := 7; r >= 0; r-- {
895 row := []string{strconv.Itoa(r + 1)}
896 for f := 0; f < 8; f++ {
897 row = append(row, string(p.board[8*r+f]))
898 }
899 emit(strings.Join(row, " "))
900 }
901 emit(" a b c d e f g h")
902 case "MOVE", "UCI":
903 played := []string{}
904 failure := ""
905 for i, text := range a {
906 var m Move
907 var err error
908 if cmd == "MOVE" {
909 m, err = readSan(g.current(), text)
910 } else {
911 m, err = readUci(g.current(), text)
912 }
913 if err != nil {
914 failure = fmt.Sprintf("ILLEGAL move %d: %s (%s)", i+1, text, err)
915 break
916 }
917 g.play(m)
918 played = append(played, g.records[len(g.records)-1].san)
919 }
920 if cmd == "UCI" && len(played) > 0 {
921 emit("san " + strings.Join(played, " "))
922 }
923 if failure != "" {
924 emit(failure)
925 } else if cmd == "MOVE" {
926 emit("OK")
927 }
928 case "LEGAL":
929 list := []string{}
930 for _, m := range p.legal() {
931 if len(a) == 0 || m.source == square(a[0]) {
932 list = append(list, p.san(m))
933 }
934 }
935 sort.Strings(list)
936 row := append([]string{"legal", strconv.Itoa(len(list))}, list...)
937 emit(strings.Join(row, " "))
938 case "STATUS":
939 side := "white"
940 if p.side == 'b' {
941 side = "black"
942 }
943 row := append([]string{"status", side, p.state(), "rep", strconv.Itoa(g.repetition())}, g.draws()...)
944 emit(strings.Join(row, " "))
945 case "UNDO":
946 if n > len(g.records) {
947 emit("ERROR cannot undo")
948 } else {
949 g.records = g.records[:len(g.records)-n]
950 g.positions = g.positions[:len(g.positions)-n]
951 emit("OK")
952 }
953 case "PERFT":
954 stats := [7]int64{}
955 perft(p, n, &stats)
956 row := []string{"perft", strconv.Itoa(n)}
957 for i, f := range []string{"nodes", "captures", "ep", "castles", "promotions", "checks", "mates"} {
958 row = append(row, f, strconv.FormatInt(stats[i], 10))
959 }
960 emit(strings.Join(row, " "))
961 case "DIVIDE":
962 type Row struct {
963 san string
964 count int64
965 }
966 rows := []Row{}
967 var total int64
968 for _, m := range p.legal() {
969 v := perft(p.play(m), n-1, nil)
970 rows = append(rows, Row{p.san(m), v})
971 total += v
972 }
973 sort.Slice(rows, func(i, j int) bool { return rows[i].san < rows[j].san })
974 for _, r := range rows {
975 emit(fmt.Sprintf("%s %d", r.san, r.count))
976 }
977 emit(fmt.Sprintf("total %d", total))
978 case "ATTACKS":
979 row := []string{"attacks", a[0]}
980 for _, c := range []byte{'w', 'b'} {
981 found := attackers(p.board, square(a[0]), c)
982 label := "white"
983 if c == 'b' {
984 label = "black"
985 }
986 row = append(row, label, strconv.Itoa(len(found)))
987 for _, s := range found {
988 row = append(row, name(s))
989 }
990 }
991 emit(strings.Join(row, " "))
992 case "PINS":
993 pins := p.pins()
994 row := []string{"pins", strconv.Itoa(len(pins))}
995 for _, pair := range pins {
996 row = append(row, name(pair[0])+"-"+name(pair[1]))
997 }
998 emit(strings.Join(row, " "))
999 case "HISTORY":
1000 emit(fmt.Sprintf("history %d", len(g.records)))
1001 for i, r := range g.records {
1002 emit(fmt.Sprintf("%d %s %s %c", i+1, r.uci, r.san, r.captured))
1003 }
1004 case "PGN":
1005 result := g.result()
1006 emit(`[Result "` + result + `"]`)
1007 if g.loaded {
1008 emit(`[SetUp "1"]`)
1009 emit(`[FEN "` + g.positions[0].fen() + `"]`)
1010 }
1011 row := []string{}
1012 for i, r := range g.records {
1013 q := g.positions[i]
1014 if q.side == 'w' {
1015 row = append(row, strconv.Itoa(q.full)+".")
1016 } else if i == 0 {
1017 row = append(row, strconv.Itoa(q.full)+"...")
1018 }
1019 row = append(row, r.san)
1020 }
1021 row = append(row, result)
1022 emit(strings.Join(row, " "))
1023 }
1024}
1025func main() {
1026 defer out.Flush()
1027 scanner := bufio.NewScanner(os.Stdin)
1028 scanner.Scan()
1029 n, _ := strconv.Atoi(scanner.Text())
1030 game := newGame()
1031 for i := 0; i < n && scanner.Scan(); i++ {
1032 execute(&game, scanner.Text())
1033 }
1034}
