32new_game <- function(definitions) {
33 game <- new.env(parent = emptyenv())
34 kinds <- vapply(definitions, `[[`, character(1L), 1L)
35 room_ids <- vapply(definitions[kinds == "ROOM"], `[[`, character(1L), 2L)
36 item_ids <- sort(vapply(definitions[kinds == "ITEM"], `[[`, character(1L), 2L), method = "radix")
37 room_index <- setNames(seq_along(room_ids), room_ids)
38 item_index <- setNames(seq_along(item_ids), item_ids)
39 game$room_count <- length(room_ids)
40 game$rooms <- lapply(room_ids, function(id) list(name = "", dark = FALSE, description = character()))
41 game$items <- lapply(item_ids, function(id) list(id = id, name = "", words = character(), nouns = id,
42 weight = 0L, treasure = 0L, light = FALSE, capacity = 0L, key = 0L, fixed = FALSE, details = character()))
43 game$state <- list(room = 1L, places = integer(length(item_ids)), containers = rep("open", length(item_ids)),
44 destinations = matrix(0L, game$room_count, 6L), keys = matrix(0L, game$room_count, 6L),
45 locked = matrix(FALSE, game$room_count, 6L), lit = rep(FALSE, length(item_ids)), fuel = integer(length(item_ids)),
46 flags = character(), bonus = 0L, moves = 0L, visited = rep(FALSE, game$room_count), darkness = 0L, ended = "none")
47 game$rules <- list()
48 game$pronoun <- 0L
49 game$previous <- NULL
50 game$verbose <- FALSE
51 game$saved <- NULL
52 for (tokens in definitions) switch(tokens[1L],
53 ROOM = {
54 room <- room_index[[tokens[2L]]]
55 dark <- tokens[3L] == "DARK"
56 game$rooms[[room]]$name <- joined(after(tokens, if (dark) 3L else 2L))
57 game$rooms[[room]]$dark <- dark
58 },
59 DESC = {
60 room <- room_index[[tokens[2L]]]
61 game$rooms[[room]]$description <- c(game$rooms[[room]]$description, joined(after(tokens, 2L)))
62 },
63 EXIT = {
64 room <- room_index[[tokens[2L]]]
65 d <- direction(tokens[3L])
66 game$state$destinations[room, d] <- room_index[[tokens[4L]]]
67 if (length(tokens) > 4L) {
68 game$state$keys[room, d] <- item_index[[tokens[6L]]]
69 game$state$locked[room, d] <- TRUE
70 }
71 },
72 ITEM = {
73 item <- item_index[[tokens[2L]]]
74 game$state$places[item] <- if (tokens[3L] == "PLAYER") player else if (tokens[3L] == "NOWHERE") nowhere else room_index[[tokens[3L]]]
75 value <- game$items[[item]]
76 value$weight <- as.integer(tokens[4L])
77 position <- 5L
78 if (tokens[position] == "TREASURE") { value$treasure <- as.integer(tokens[position + 1L]); position <- position + 2L }
79 if (tokens[position] == "LIGHT") {
80 value$light <- TRUE
81 game$state$fuel[item] <- as.integer(tokens[position + 1L])
82 position <- position + 2L
83 }
84 if (tokens[position] == "CONTAINER") {
85 value$capacity <- as.integer(tokens[position + 1L])
86 position <- position + 2L
87 if (tokens[position] == "CLOSED") { game$state$containers[item] <- "closed"; position <- position + 1L }
88 if (tokens[position] == "LOCKED") {
89 value$key <- item_index[[tokens[position + 1L]]]
90 game$state$containers[item] <- "locked"
91 position <- position + 2L
92 }
93 }
94 if (tokens[position] == "FIXED") { value$fixed <- TRUE; position <- position + 1L }
95 value$words <- after(tokens, position - 1L)
96 value$name <- joined(value$words)
97 game$items[[item]] <- value
98 },
99 INSIDE = game$state$places[item_index[[tokens[2L]]]] <- inside(game, item_index[[tokens[3L]]]),
100 ALIAS = {
101 item <- item_index[[tokens[2L]]]
102 game$items[[item]]$nouns <- union(game$items[[item]]$nouns, tokens[3L])
103 },
104 DETAIL = {
105 item <- item_index[[tokens[2L]]]
106 game$items[[item]]$details <- c(game$items[[item]]$details, joined(after(tokens, 2L)))
107 },
108 RULE = , ENTER = {
109 use <- tokens[1L] == "RULE"
110 rule <- list(kind = if (use) "use" else "enter", a = if (use) item_index[[tokens[2L]]] else room_index[[tokens[2L]]],
111 b = if (use) item_index[[tokens[3L]]] else 0L, condition = NULL)
112 position <- if (use) 4L else 3L
113 if (tokens[position] %in% c("IF", "UNLESS")) {
114 rule$condition <- list(flag = tokens[position + 1L], on = tokens[position] == "IF")
115 position <- position + 2L
116 }
117 args <- after(tokens, position)
118 rule$effect <- switch(tokens[position],
119 MESSAGE = list(kind = "message", text = joined(args)),
120 OPEN = list(kind = "open", room = room_index[[args[1L]]], direction = direction(args[2L]), destination = room_index[[args[3L]]]),
121 REMOVE = list(kind = "move", item = item_index[[args[1L]]], place = nowhere),
122 ADD = list(kind = "move", item = item_index[[args[1L]]], place = room_index[[args[2L]]]),
123 SET = list(kind = "flag", flag = args[1L], on = TRUE),
124 CLEAR = list(kind = "flag", flag = args[1L], on = FALSE),
125 POINTS = list(kind = "points", value = as.integer(args[1L])))
126 game$rules[[length(game$rules) + 1L]] <- rule
127 },
128 TROPHY = game$trophy <- room_index[[tokens[2L]]],
129 START = game$state$room <- room_index[[tokens[2L]]],
130 MAXWEIGHT = game$max_weight <- as.integer(tokens[2L]))
131 game$state$visited[game$state$room] <- TRUE
132 game$vocabulary <- unique(c(character(), unlist(lapply(game$items, function(item) c(item$nouns, item$words)), use.names = FALSE)))
133 game
134}

200resolve <- function(game, phrase) {
201 seen <- visible(game)
202 if (identical(phrase, "it")) {
203 if (game$pronoun == 0L) fail('I don\'t know what "it" refers to.')
204 if (!seen[game$pronoun]) fail("You can't see any such thing.")
205 return(game$pronoun)
206 }
207 for (word in phrase) if (!word %in% game$vocabulary) fail(paste0('I don\'t know the word "', word, '".'))
208 noun <- tail(phrase, 1L)
209 adjectives <- head(phrase, -1L)
210 matches <- which(seen & vapply(game$items, function(item) noun %in% item$nouns && all(adjectives %in% item$words), logical(1L)))
211 if (!length(matches)) fail("You can't see any such thing.")
212 if (length(matches) > 1L) fail(paste0("Which do you mean: ", item_names(game, matches), "?"))
213 game$pronoun <- matches[1L]
214 game$pronoun
215}

217run_rules <- function(game, kind, a, b = 0L) {
218 executed <- 0L
219 message <- FALSE
220 for (rule in game$rules) {
221 if (rule$kind != kind || rule$a != a || rule$b != b) next
222 condition <- rule$condition
223 if (!is.null(condition) && (condition$flag %in% game$state$flags) != condition$on) next
224 executed <- executed + 1L
225 effect <- rule$effect
226 switch(effect$kind,
227 message = { say(effect$text); message <- TRUE },
228 open = {
229 game$state$destinations[effect$room, effect$direction] <- effect$destination
230 game$state$keys[effect$room, effect$direction] <- 0L
231 game$state$locked[effect$room, effect$direction] <- FALSE
232 },
233 move = game$state$places[effect$item] <- effect$place,
234 flag = game$state$flags <- if (effect$on) union(game$state$flags, effect$flag) else setdiff(game$state$flags, effect$flag),
235 points = game$state$bonus <- game$state$bonus + effect$value)
236 }
237 list(executed = executed, message = message)
238}
239pass_turn <- function(game, baseline) {
240 game$state$moves <- game$state$moves + 1L
241 close_by <- nearby(game)
242 for (item in which(game$state$lit)) {
243 game$state$fuel[item] <- game$state$fuel[item] - 1L
244 if (game$state$fuel[item] == 0L) {
245 game$state$lit[item] <- FALSE
246 if (close_by[item]) say(paste0("The ", game$items[[item]]$name, " goes out."))
247 }
248 }
249 lit <- is_lit(game)
250 if (baseline && !lit) say("It is now pitch dark.") else if (!baseline && lit) describe(game, TRUE)
251 game$state$darkness <- if (lit) 0L else game$state$darkness + 1L
252 if (game$state$darkness == 2L) say("You hear something moving in the dark.")
253 if (game$state$darkness == 3L) game$state$ended <- "dead"
254 treasure <- treasures(game)
255 if (game$state$ended == "none" && treasure$total > 0L && treasure$stored == treasure$total) game$state$ended <- "won"
256 if (game$state$ended != "none") {
257 say(if (game$state$ended == "dead") "*** You have died ***" else "*** You have won ***")
258 say(sprintf("You scored %d points in %d moves.", treasure$points + game$state$bonus, game$state$moves))
259 }

265put_item <- function(game, item, container) {
266 name <- game$items[[item]]$name
267 target <- game$items[[container]]$name
268 if (within(game, container, item)) return(paste0("The ", target, " is inside the ", name, "."))
269 if (game$state$places[item] == inside(game, container)) return(paste0("The ", name, " is already in the ", target, "."))
270 weight <- place_weight(game, inside(game, container)) + if (within(game, item, container)) 0L else total_weight(game, item)
271 if (weight > game$items[[container]]$capacity) return(paste0("The ", name, " doesn't fit in the ", target, "."))
272 game$state$places[item] <- inside(game, container)
273 ""

275take_items <- function(game, first, second) {
276 if (!is_lit(game)) fail("It is pitch dark.")
277 container <- 0L
278 source <- game$state$room
279 if (length(second)) { container <- resolve(game, second); require_container(game, container); source <- inside(game, container) }
280 if (all_items(first)) {
281 candidates <- direct(game, source)
282 candidates <- candidates[!vapply(game$items[candidates], `[[`, logical(1L), "fixed")]
283 if (!length(candidates)) fail(if (container == 0L) "There is nothing here to take." else paste0("There is nothing in the ", game$items[[container]]$name, " to take."))
284 taken <- FALSE
285 for (item in candidates) {
286 if (!carried(game, item) && place_weight(game, player) + total_weight(game, item) > game$max_weight) {
287 say(paste0(game$items[[item]]$name, ": You can't carry any more."))
288 } else {
289 game$state$places[item] <- player
290 say(paste0(game$items[[item]]$name, ": Taken."))
291 taken <- TRUE
292 }
293 }
294 return(taken)
295 }
296 item <- resolve(game, first)
297 name <- game$items[[item]]$name
298 if (container != 0L && game$state$places[item] != source) fail(paste0("The ", name, " is not in the ", game$items[[container]]$name, "."))
299 if (game$state$places[item] == player) fail(paste0("You already have the ", name, "."))
300 if (game$items[[item]]$fixed) fail(paste0("The ", name, " is fixed in place."))
301 if (!carried(game, item) && place_weight(game, player) + total_weight(game, item) > game$max_weight) fail("You can't carry any more.")
302 game$state$places[item] <- player
303 say("Taken.")
304 TRUE
305}
306drop_items <- function(game, phrase) {

314 game$state$places[item] <- game$state$room

316 }

318put_items <- function(game, first, second) {
319 item <- if (all_items(first)) 0L else resolve(game, first)
320 container <- resolve(game, second)
321 if (item != 0L) {
322 if (!carried(game, item)) fail(paste0("You aren't carrying the ", game$items[[item]]$name, "."))
323 if (item == container) fail(paste0("You can't put the ", game$items[[item]]$name, " inside itself."))
324 }
325 require_container(game, container)
326 if (item != 0L) {
327 reason <- put_item(game, item, container)
328 if (reason != "") fail(reason)
329 say(paste0("You put the ", game$items[[item]]$name, " in the ", game$items[[container]]$name, "."))
330 return(TRUE)
331 }
332 candidates <- setdiff(direct(game, player), container)
333 if (!length(candidates)) fail(paste0("You have nothing to put in the ", game$items[[container]]$name, "."))
334 moved <- FALSE
335 for (candidate in candidates) {
336 reason <- put_item(game, candidate, container)
337 if (reason == "") moved <- TRUE
338 say(paste0(game$items[[candidate]]$name, ": ", if (reason == "") "Done." else reason))
339 }
340 moved

352change_lock <- function(game, first, second, locking) {
353 d <- if (length(first) == 1L) direction(first) else 0L
354 status <- if (locking) "already locked" else "not locked"
355 if (d != 0L) {
356 room <- game$state$room
357 destination <- game$state$destinations[room, d]
358 if (destination == 0L) fail(paste0("There is no exit ", directions[d], "."))
359 key <- game$state$keys[room, d]
360 if (key == 0L) fail(paste0("The way ", directions[d], " has no lock."))
361 if (game$state$locked[room, d] == locking) fail(paste0("The way ", directions[d], " is ", status, "."))
362 key_check(game, key, second)
363 game$state$locked[room, d] <- locking
364 opposite <- bitwXor(d - 1L, 1L) + 1L
365 if (game$state$destinations[destination, opposite] == room && game$state$keys[destination, opposite] == key) {
366 game$state$locked[destination, opposite] <- locking
367 }
368 } else {
369 item <- resolve(game, first)
370 name <- game$items[[item]]$name
371 key <- game$items[[item]]$key
372 if (game$items[[item]]$capacity == 0L || key == 0L) fail(paste0("The ", name, " has no lock."))
373 if ((game$state$containers[item] == "locked") == locking) fail(paste0("The ", name, " is ", status, "."))
374 if (locking && game$state$containers[item] == "open") fail(paste0("You must close the ", name, " first."))
375 key_check(game, key, second)
376 game$state$containers[item] <- if (locking) "locked" else "closed"
377 }
378 say(if (locking) "Locked." else "Unlocked.")
379}

394execute <- function(game, verb, first, second, d) {
395 baseline <- is_lit(game)
396 success <- TRUE
397 switch(verb,
398 go = {
399 destination <- game$state$destinations[game$state$room, d]
400 if (destination == 0L) fail("You can't go that way.")
401 if (game$state$locked[game$state$room, d]) fail(paste0("The way ", directions[d], " is locked."))
402 game$state$room <- destination
403 baseline <- is_lit(game)
404 describe(game, game$verbose || !game$state$visited[destination])
405 game$state$visited[destination] <- TRUE
406 run_rules(game, "enter", destination)
407 },
408 look = describe(game, TRUE),
409 examine = examine(game, resolve(game, first)),
410 take = success <- take_items(game, first, second),
411 drop = drop_items(game, first),
412 put = success <- put_items(game, first, second),
413 inventory = {
414 if (!length(direct(game, player))) say("You are empty-handed.") else {
415 say("You are carrying:"); item_tree(game, player, 1L)
416 say(sprintf("Total weight: %d/%d", place_weight(game, player), game$max_weight))
417 }
418 },
419 open = {
420 item <- resolve(game, first)
421 name <- game$items[[item]]$name
422 if (game$items[[item]]$capacity == 0L) fail(paste0("You can't open the ", name, "."))
423 if (game$state$containers[item] == "open") fail(paste0("The ", name, " is already open."))
424 if (game$state$containers[item] == "locked") fail(paste0("The ", name, " is locked."))
425 game$state$containers[item] <- "open"
426 children <- direct(game, inside(game, item))
427 say(if (length(children)) paste0("Opening the ", name, " reveals ", item_names(game, children), ".") else "Opened.")
428 },
429 close = {
430 item <- resolve(game, first)
431 name <- game$items[[item]]$name
432 if (game$items[[item]]$capacity == 0L) fail(paste0("You can't close the ", name, "."))
433 if (game$state$containers[item] != "open") fail(paste0("The ", name, " is already closed."))
434 game$state$containers[item] <- "closed"
435 say("Closed.")
436 },
437 lock = , unlock = change_lock(game, first, second, verb == "lock"),
438 use = {
439 first_item <- resolve(game, first)
440 second_item <- resolve(game, second)
441 if (!carried(game, first_item)) fail(paste0("You aren't carrying the ", game$items[[first_item]]$name, "."))
442 result <- run_rules(game, "use", first_item, second_item)
443 if (result$executed == 0L) fail("Nothing happens.")
444 if (!result$message) say("Done.")
445 },
446 light = , extinguish = {
447 item <- resolve(game, first)
448 name <- game$items[[item]]$name
449 lighting <- verb == "light"
450 if (!game$items[[item]]$light) fail(paste0("You can't ", verb, " the ", name, "."))
451 if (game$state$lit[item] == lighting) fail(paste0("The ", name, " is ", if (lighting) "already lit" else "not lit", "."))
452 if (lighting && game$state$fuel[item] == 0L) fail(paste0("The ", name, " has no fuel."))
453 game$state$lit[item] <- lighting
454 say(paste0("The ", name, " is now ", if (lighting) "lit" else "out", "."))
455 },
456 wait = say("Time passes."),

474interpret <- function(game, tokens) {
475 if (!length(tokens)) fail("I beg your pardon?")
476 if (direction(tokens[1L]) != 0L) tokens <- c("go", tokens)
477 verb <- unname(verbs[tokens[1L]])
478 if (is.na(verb)) fail(paste0('I don\'t know the verb "', tokens[1L], '".'))
479 if (game$state$ended != "none" && verb != "restore") fail("The game is over.")
480 first <- after(tokens, 1L)
481 second <- character()
482 d <- 0L
483 if (verb == "look" && length(first)) {
484 if (first[1L] != "at") fail("I don't understand that.")
485 verb <- "examine"
486 first <- after(first, 1L)
487 }
488 if (verb %in% no_arguments) {
489 if (length(first)) fail("I don't understand that.")
490 } else {
491 if (!length(first)) fail(if (verb == "go") "Which way do you want to go?" else paste0("What do you want to ", verb, "?"))
492 if (verb == "go") {
493 if (length(first) == 1L) d <- direction(first)
494 if (d == 0L) fail("I don't understand that.")
495 } else {
496 prepositions <- switch(verb, put = c("in", "into"), use = c("on", "with"), lock = "with", unlock = "with", take = "from", character())
497 split <- which(first %in% prepositions)
498 if (length(split)) {
499 index <- split[1L]
500 if (index == 1L || index == length(first)) fail("I don't understand that.")
501 second <- after(first, index)
502 first <- head(first, index - 1L)
503 } else if (verb %in% c("put", "use")) fail("I don't understand that.")
504 }
505 }
506 execute(game, verb, first, second, d)
507}
