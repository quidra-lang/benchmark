10local verbs = {}
11for verb, names in pairs(aliases) do for word in names:gmatch("[^ ]+") do verbs[word] = verb end end
12local noObjects = {look = true, inventory = true, wait = true, score = true, save = true, restore = true, verbose = true, brief = true}
13local actions = {go = true, take = true, drop = true, put = true, open = true, close = true, unlock = true, lock = true,

15local function fail(reason)
16 error({command_error = true, lines = type(reason) == "table" and reason or {reason}}, 0)
17end
18local function words(line)
19 local result = {}
20 for word in line:gmatch("[^ ]+") do result[#result + 1] = word end
21 return result
22end
23local function normalized(line)
24 line = line:gsub("[A-Z]", function(letter) return string.char(letter:byte() + 32) end)
25 local result = {}
26 for word in line:gmatch("[^ ]+") do
27 if word ~= "the" and word ~= "a" and word ~= "an" then result[#result + 1] = word end
28 end
29 return result
30end
31local function slice(values, first, last)
32 local result = {}
33 for i = first, last or #values do result[#result + 1] = values[i] end
34 return result
35end
36local function append(result, values) for _, value in ipairs(values) do result[#result + 1] = value end end
37local function copy(values)
38 local result = {}
39 for key, value in pairs(values) do result[key] = value end
40 return result
41end
42local function asciiLess(a, b)
43 for i = 1, math.min(#a, #b) do if a:byte(i) ~= b:byte(i) then return a:byte(i) < b:byte(i) end end
44 return #a < #b
45end
46local function only(phrase, word) return #phrase == 1 and phrase[1] == word end
47local function place(kind, id) return {kind = kind, id = id, key = kind .. ":" .. (id or "")} end
48local function newState()
49 return {room = "", places = {}, containers = {}, fuel = {}, lit = {}, exits = {}, flags = {}, visited = {},
50 bonus = 0, moves = 0, darkness = 0, ended = "none"}
51end
52local function copyState(state)
53 local result = copy(state)
54 for _, field in ipairs({"places", "containers", "fuel", "lit", "flags", "visited"}) do result[field] = copy(state[field]) end
55 result.exits = {}
56 for room, exits in pairs(state.exits) do result.exits[room] = copy(exits) end
57 return result
58end

62function Adventure.new(definitions)
63 local self = setmetatable({rooms = {}, items = {}, ids = {}, rules = {}, enterRules = {}, vocabulary = {},
64 state = newState(), verbose = false}, Adventure)
65 for _, line in ipairs(definitions) do self:define(line) end
66 for id, item in pairs(self.items) do
67 self.ids[#self.ids + 1] = id
68 for word in pairs(item.nouns) do self.vocabulary[word] = true end
69 for word in pairs(item.nameWords) do self.vocabulary[word] = true end
70 end
71 table.sort(self.ids, asciiLess)
72 self.state.visited[self.state.room] = true
73 return self
74end
75function Adventure:setExit(room, direction, destination, key)
76 if not self.state.exits[room] then self.state.exits[room] = {} end
77 self.state.exits[room][direction] = {destination = destination, key = key, locked = key ~= nil}
78end
79function Adventure:define(line)
80 local w = words(line)
81 local kind, id = w[1], w[2]
82 local function text(first) return table.concat(w, " ", first) end
83 if kind == "ROOM" then
84 local dark = w[3] == "DARK"
85 self.rooms[id] = {name = text(dark and 4 or 3), dark = dark, description = {}}
86 elseif kind == "DESC" then
87 local description = self.rooms[id].description
88 description[#description + 1] = text(3)
89 elseif kind == "EXIT" then
90 self:setExit(id, directions[w[3]], w[4], #w > 4 and w[6] or nil)
91 elseif kind == "ITEM" then
92 local item = {name = "", weight = tonumber(w[4], 10), points = 0, light = false, fixed = false,
93 nouns = {[id] = true}, nameWords = {}, details = {}}
94 if w[3] == "PLAYER" then self.state.places[id] = place("player")
95 elseif w[3] == "NOWHERE" then self.state.places[id] = place("nowhere")
96 else self.state.places[id] = place("room", w[3]) end
97 self.state.containers[id], self.state.fuel[id], self.state.lit[id] = "open", 0, false
98 local index = 5
99 if w[index] == "TREASURE" then item.points = tonumber(w[index + 1], 10); index = index + 2 end
100 if w[index] == "LIGHT" then item.light = true; self.state.fuel[id] = tonumber(w[index + 1], 10); index = index + 2 end
101 if w[index] == "CONTAINER" then
102 item.capacity = tonumber(w[index + 1], 10); index = index + 2
103 if w[index] == "CLOSED" then self.state.containers[id] = "closed"; index = index + 1 end
104 if w[index] == "LOCKED" then item.key = w[index + 1]; self.state.containers[id] = "locked"; index = index + 2 end
105 end
106 if w[index] == "FIXED" then item.fixed = true; index = index + 1 end
107 item.name = text(index)
108 for i = index, #w do item.nameWords[w[i]] = true end
109 self.items[id] = item
110 elseif kind == "INSIDE" then self.state.places[id] = place("inside", w[3])
111 elseif kind == "ALIAS" then self.items[id].nouns[w[3]] = true
112 elseif kind == "DETAIL" then
113 local details = self.items[id].details
114 details[#details + 1] = text(3)
115 elseif kind == "RULE" or kind == "ENTER" then
116 local index = kind == "RULE" and 4 or 3
117 local flag, whenOn = nil, true
118 if w[index] == "IF" or w[index] == "UNLESS" then
119 whenOn, flag = w[index] == "IF", w[index + 1]; index = index + 2
120 end
121 local effect = {kind = w[index]}
122 index = index + 1
123 if effect.kind == "MESSAGE" then effect.text = text(index)
124 elseif effect.kind == "OPEN" then effect.room, effect.direction, effect.destination = w[index], directions[w[index + 1]], w[index + 2]
125 elseif effect.kind == "REMOVE" then effect.item, effect.place = w[index], place("nowhere")
126 elseif effect.kind == "ADD" then effect.item, effect.place = w[index], place("room", w[index + 1])
127 elseif effect.kind == "SET" or effect.kind == "CLEAR" then effect.flag = w[index]
128 else effect.amount = tonumber(w[index], 10) end
129 local rule = {first = id, second = kind == "RULE" and w[3] or nil, flag = flag, whenOn = whenOn, effect = effect}
130 local target = kind == "RULE" and self.rules or self.enterRules
131 target[#target + 1] = rule
132 elseif kind == "TROPHY" then self.trophy = id
133 elseif kind == "START" then self.state.room = id
134 elseif kind == "MAXWEIGHT" then self.maxWeight = tonumber(id, 10) end
135end
136function Adventure:contents(target)
137 local result = {}
138 for _, id in ipairs(self.ids) do if self.state.places[id].key == target.key then result[#result + 1] = id end end
139 return result
140end
141function Adventure:children()
142 local result = {}
143 for _, id in ipairs(self.ids) do
144 local key = self.state.places[id].key
145 if not result[key] then result[key] = {} end
146 result[key][#result[key] + 1] = id
147 end
148 return result
149end
150function Adventure:isOpen(id) return self.items[id].capacity ~= nil and self.state.containers[id] == "open" end
151function Adventure:reachable(root)
152 local children, result = self:children(), {}
153 local function visit(target)
154 for _, id in ipairs(children[target.key] or {}) do
155 result[id] = true
156 if self:isOpen(id) then visit(place("inside", id)) end
157 end
158 end
159 visit(root)
160 return result
161end
162function Adventure:carried() return self:reachable(place("player")) end
163function Adventure:isCarried(id) return self:carried()[id] == true end
164function Adventure:roomItems() return self:reachable(place("room", self.state.room)) end
165function Adventure:nearby()
166 local result = self:carried()
167 for id in pairs(self:roomItems()) do result[id] = true end
168 return result
169end
170function Adventure:isLit()
171 if not self.rooms[self.state.room].dark then return true end
172 for id in pairs(self:nearby()) do if self.state.lit[id] then return true end end
173 return false
174end
175function Adventure:visible() return self:isLit() and self:nearby() or self:carried() end
176function Adventure:weights()
177 local children, result = self:children(), {}
178 local function total(id)
179 if result[id] ~= nil then return result[id] end
180 local weight = self.items[id].weight
181 for _, child in ipairs(children[place("inside", id).key] or {}) do weight = weight + total(child) end
182 result[id] = weight
183 return weight
184 end
185 for _, id in ipairs(self.ids) do total(id) end
186 return result
187end
188function Adventure:totalWeight(id) return self:weights()[id] end
189function Adventure:contentWeight(id) return self:totalWeight(id) - self.items[id].weight end
190function Adventure:carriedWeight()
191 local weights, result = self:weights(), 0
192 for _, id in ipairs(self:contents(place("player"))) do result = result + weights[id] end
193 return result
194end
195function Adventure:nestedIn(id, ancestor)
196 local target = self.state.places[id]
197 while target.kind == "inside" do
198 if target.id == ancestor then return true end
199 target = self.state.places[target.id]
200 end
201 return false
202end
203function Adventure:score()
204 local stored, treasures, points = 0, 0, 0
205 for _, id in ipairs(self.ids) do
206 local item = self.items[id]
207 if item.points > 0 then
208 treasures = treasures + 1
209 local target = self.state.places[id]
210 while target.kind == "inside" do target = self.state.places[target.id] end
211 if target.kind == "room" and target.id == self.trophy then stored = stored + 1; points = points + item.points end
212 end
213 end
214 return {stored = stored, treasures = treasures, points = points, total = points + self.state.bonus}
215end
216function Adventure:display(id)
217 local item, annotations = self.items[id], {}
218 if item.capacity then annotations[#annotations + 1] = self.state.containers[id] end
219 if item.light and self.state.lit[id] then annotations[#annotations + 1] = "lit" end
220 return item.name .. (#annotations > 0 and " (" .. table.concat(annotations, ", ") .. ")" or "")
221end
222function Adventure:tree(root)
223 local children, lines = self:children(), {}
224 local function visit(target, depth)
225 for _, id in ipairs(children[target.key] or {}) do
226 lines[#lines + 1] = string.rep(" ", depth) .. self:display(id)
227 if self:isOpen(id) then visit(place("inside", id), depth + 1) end
228 end
229 end
230 visit(root, 1)
231 return lines
232end
233function Adventure:describe(full)
234 if not self:isLit() then return {"It is pitch dark."} end
235 if full == nil then full = true end
236 local room, lines = self.rooms[self.state.room], {}
237 lines[#lines + 1] = room.name
238 if full then append(lines, room.description) end
239 local exits = {}
240 for direction, name in ipairs(directionNames) do
241 local exit = (self.state.exits[self.state.room] or {})[direction]
242 if exit then exits[#exits + 1] = name .. (exit.locked and " (locked)" or "") end
243 end
244 lines[#lines + 1] = "Exits: " .. (#exits > 0 and table.concat(exits, ", ") or "none")
245 local floor = place("room", self.state.room)
246 if #self:contents(floor) > 0 then lines[#lines + 1] = "You see:"; append(lines, self:tree(floor)) end
247 return lines
248end
249function Adventure:resolve(phrase)
250 local visible = self:visible()
251 if only(phrase, "it") then
252 if not self.pronoun then fail('I don\'t know what "it" refers to.') end
253 if not visible[self.pronoun] then fail("You can't see any such thing.") end
254 return self.pronoun
255 end
256 for _, word in ipairs(phrase) do if not self.vocabulary[word] then fail('I don\'t know the word "' .. word .. '".') end end
257 local matches = {}
258 for _, id in ipairs(self.ids) do
259 local item, matchesWords = self.items[id], true
260 if visible[id] and item.nouns[phrase[#phrase]] then
261 for i = 1, #phrase - 1 do if not item.nameWords[phrase[i]] then matchesWords = false; break end end
262 if matchesWords then matches[#matches + 1] = id end
263 end
264 end
265 if #matches == 0 then fail("You can't see any such thing.") end
266 if #matches > 1 then
267 local names = {}
268 for _, id in ipairs(matches) do names[#names + 1] = self.items[id].name end
269 fail("Which do you mean: " .. table.concat(names, ", ") .. "?")
270 end
271 self.pronoun = matches[1]
272 return matches[1]
273end
274function Adventure:runRules(rules, first, second)
275 local lines, executed = {}, 0
276 for _, rule in ipairs(rules) do
277 if rule.first == first and rule.second == second and (rule.flag == nil or (self.state.flags[rule.flag] == true) == rule.whenOn) then
278 executed = executed + 1
279 local effect = rule.effect
280 if effect.kind == "MESSAGE" then lines[#lines + 1] = effect.text
281 elseif effect.kind == "OPEN" then self:setExit(effect.room, effect.direction, effect.destination)
282 elseif effect.kind == "REMOVE" or effect.kind == "ADD" then self.state.places[effect.item] = effect.place
283 elseif effect.kind == "SET" or effect.kind == "CLEAR" then self.state.flags[effect.flag] = effect.kind == "SET" or nil
284 else self.state.bonus = self.state.bonus + effect.amount end
285 end
286 end
287 return lines, executed
288end
289function Adventure:requireContainer(id)
290 local item = self.items[id]
291 if not item.capacity then fail("The " .. item.name .. " is not a container.") end
292 if not self:isOpen(id) then fail("The " .. item.name .. " is closed.") end
293end
294function Adventure:take(first, second)
295 if not self:isLit() then fail("It is pitch dark.") end
296 local container, source = nil, place("room", self.state.room)
297 if second then container = self:resolve(second); self:requireContainer(container); source = place("inside", container) end
298 if only(first, "all") then
299 local candidates = {}
300 for _, id in ipairs(self:contents(source)) do if not self.items[id].fixed then candidates[#candidates + 1] = id end end
301 if #candidates == 0 then
302 fail(container and "There is nothing in the " .. self.items[container].name .. " to take." or "There is nothing here to take.")
303 end
304 local lines, taken = {}, false
305 for _, id in ipairs(candidates) do
306 local outcome = "You can't carry any more."
307 if self:isCarried(id) or self:carriedWeight() + self:totalWeight(id) <= self.maxWeight then
308 self.state.places[id], outcome, taken = place("player"), "Taken.", true
309 end
310 lines[#lines + 1] = self.items[id].name .. ": " .. outcome
311 end
312 if not taken then fail(lines) end
313 return lines
314 end
315 local id = self:resolve(first)
316 local item = self.items[id]
317 if container and self.state.places[id].key ~= source.key then fail("The " .. item.name .. " is not in the " .. self.items[container].name .. ".") end
318 if self.state.places[id].kind == "player" then fail("You already have the " .. item.name .. ".") end
319 if item.fixed then fail("The " .. item.name .. " is fixed in place.") end
320 if not self:isCarried(id) and self:carriedWeight() + self:totalWeight(id) > self.maxWeight then fail("You can't carry any more.") end
321 self.state.places[id] = place("player")
322 return {"Taken."}
323end
324function Adventure:drop(phrase)
325 if only(phrase, "all") then
326 local candidates, lines = self:contents(place("player")), {}
327 if #candidates == 0 then fail("You are empty-handed.") end
328 for _, id in ipairs(candidates) do
329 self.state.places[id] = place("room", self.state.room)
330 lines[#lines + 1] = self.items[id].name .. ": Dropped."
331 end
332 return lines
333 end
334 local id = self:resolve(phrase)
335 if not self:isCarried(id) then fail("You aren't carrying the " .. self.items[id].name .. ".") end
336 self.state.places[id] = place("room", self.state.room)
337 return {"Dropped."}
338end
339function Adventure:putItem(id, container)
340 local item, target = self.items[id], self.items[container]
341 if self:nestedIn(container, id) then return "The " .. target.name .. " is inside the " .. item.name .. "." end
342 if self.state.places[id].key == place("inside", container).key then return "The " .. item.name .. " is already in the " .. target.name .. "." end
343 local weight = self:contentWeight(container) + (self:nestedIn(id, container) and 0 or self:totalWeight(id))
344 if weight > target.capacity then return "The " .. item.name .. " doesn't fit in the " .. target.name .. "." end
345 self.state.places[id] = place("inside", container)
346 return nil
347end
348function Adventure:put(first, second)
349 if only(first, "all") then
350 local target = self:resolve(second)
351 self:requireContainer(target)
352 local candidates = {}
353 for _, id in ipairs(self:contents(place("player"))) do if id ~= target then candidates[#candidates + 1] = id end end
354 if #candidates == 0 then fail("You have nothing to put in the " .. self.items[target].name .. ".") end
355 local lines, succeeded = {}, false
356 for _, id in ipairs(candidates) do
357 local reason = self:putItem(id, target)
358 if not reason then succeeded = true end
359 lines[#lines + 1] = self.items[id].name .. ": " .. (reason or "Done.")
360 end
361 if not succeeded then fail(lines) end
362 return lines
363 end
364 local id = self:resolve(first)
365 local target = self:resolve(second)
366 local name = self.items[id].name
367 if not self:isCarried(id) then fail("You aren't carrying the " .. name .. ".") end
368 if id == target then fail("You can't put the " .. name .. " inside itself.") end
369 self:requireContainer(target)
370 local reason = self:putItem(id, target)
371 if reason then fail(reason) end
372 return {"You put the " .. name .. " in the " .. self.items[target].name .. "."}
373end
374function Adventure:keyCheck(key, phrase)
375 if phrase then
376 local id = self:resolve(phrase)
377 local name = self.items[id].name
378 if not self:isCarried(id) then fail("You aren't carrying the " .. name .. ".") end
379 if id ~= key then fail("The " .. name .. " doesn't fit.") end
380 return {}
381 end
382 if not self:isCarried(key) then fail("You don't have the right key.") end
383 return {"(with the " .. self.items[key].name .. ")"}
384end
385function Adventure:lockObject(first, second, locking)
386 local direction = #first == 1 and directions[first[1]] or nil
387 local lines
388 if direction then
389 local room, name = self.state.room, directionNames[direction]
390 local exit = (self.state.exits[room] or {})[direction]
391 if not exit then fail("There is no exit " .. name .. ".") end
392 if not exit.key then fail("The way " .. name .. " has no lock.") end
393 if exit.locked == locking then fail("The way " .. name .. " is " .. (locking and "already locked." or "not locked.")) end
394 lines = self:keyCheck(exit.key, second)
395 self.state.exits[room][direction] = {destination = exit.destination, key = exit.key, locked = locking}
396 local opposite = ((direction - 1) ~ 1) + 1
397 local back = (self.state.exits[exit.destination] or {})[opposite]
398 if back and back.destination == room and back.key == exit.key then
399 self.state.exits[exit.destination][opposite] = {destination = back.destination, key = back.key, locked = locking}
400 end
401 else
402 local id = self:resolve(first)
403 local item, value = self.items[id], self.state.containers[id]
404 if not item.capacity or not item.key then fail("The " .. item.name .. " has no lock.") end
405 if (value == "locked") == locking then fail("The " .. item.name .. " is " .. (locking and "already locked." or "not locked.")) end
406 if locking and value == "open" then fail("You must close the " .. item.name .. " first.") end
407 lines = self:keyCheck(item.key, second)
408 self.state.containers[id] = locking and "locked" or "closed"
409 end
410 lines[#lines + 1] = locking and "Locked." or "Unlocked."
411 return lines
412end
413function Adventure:execute(verb, first, second)
414 if verb == "look" then return self:describe()
415 elseif verb == "inventory" then
416 local player, lines = place("player"), {"You are carrying:"}
417 if #self:contents(player) == 0 then return {"You are empty-handed."} end
418 append(lines, self:tree(player)); lines[#lines + 1] = "Total weight: " .. self:carriedWeight() .. "/" .. self.maxWeight
419 return lines
420 elseif verb == "wait" then return {"Time passes."}
421 elseif verb == "score" then
422 local score = self:score()
423 return {"You have scored " .. score.total .. " points (treasure " .. score.points .. ", bonus " .. self.state.bonus .. ") in " .. self.state.moves .. " moves.",
424 "Treasures stored: " .. score.stored .. " of " .. score.treasures .. "."}
425 elseif verb == "save" then self.saved = copyState(self.state); return {"Game saved."}
426 elseif verb == "restore" then
427 if not self.saved then fail("There is no saved game.") end
428 self.state = copyState(self.saved)
429 local lines = {"Game restored."}; append(lines, self:describe()); return lines
430 elseif verb == "verbose" then self.verbose = true; return {"Verbose mode."}
431 elseif verb == "brief" then self.verbose = false; return {"Brief mode."}
432 elseif verb == "take" then return self:take(first, second)
433 elseif verb == "drop" then return self:drop(first)
434 elseif verb == "put" then return self:put(first, second)
435 elseif verb == "unlock" or verb == "lock" then return self:lockObject(first, second, verb == "lock")
436 elseif verb == "use" then
437 local id = self:resolve(first)
438 local target = self:resolve(second)
439 if not self:isCarried(id) then fail("You aren't carrying the " .. self.items[id].name .. ".") end
440 local lines, executed = self:runRules(self.rules, id, target)
441 if executed == 0 then fail("Nothing happens.") end
442 return #lines > 0 and lines or {"Done."}
443 end
444 local id = self:resolve(first)
445 local item, value = self.items[id], self.state.containers[id]
446 if verb == "examine" then
447 local lines = #item.details > 0 and copy(item.details) or {"You see nothing special about the " .. item.name .. "."}
448 if item.light then lines[#lines + 1] = "It is " .. (self.state.lit[id] and "lit" or "unlit") .. " (fuel " .. self.state.fuel[id] .. ")." end
449 if item.capacity then
450 if value ~= "open" then lines[#lines + 1] = "It is " .. value .. "."
451 elseif #self:contents(place("inside", id)) == 0 then lines[#lines + 1] = "It is empty."
452 else lines[#lines + 1] = "It contains:"; append(lines, self:tree(place("inside", id))) end
453 end
454 if item.points > 0 then lines[#lines + 1] = "It is a treasure worth " .. item.points .. " points." end
455 lines[#lines + 1] = "It weighs " .. self:totalWeight(id) .. "."
456 return lines
457 elseif verb == "open" then
458 if not item.capacity then fail("You can't open the " .. item.name .. ".") end
459 if value == "open" then fail("The " .. item.name .. " is already open.") end
460 if value == "locked" then fail("The " .. item.name .. " is locked.") end
461 self.state.containers[id] = "open"
462 local names = {}
463 for _, child in ipairs(self:contents(place("inside", id))) do names[#names + 1] = self.items[child].name end
464 return #names > 0 and {"Opening the " .. item.name .. " reveals " .. table.concat(names, ", ") .. "."} or {"Opened."}
465 elseif verb == "close" then
466 if not item.capacity then fail("You can't close the " .. item.name .. ".") end
467 if value ~= "open" then fail("The " .. item.name .. " is already closed.") end
468 self.state.containers[id] = "closed"
469 return {"Closed."}
470 elseif verb == "light" then
471 if not item.light then fail("You can't light the " .. item.name .. ".") end
472 if self.state.lit[id] then fail("The " .. item.name .. " is already lit.") end
473 if self.state.fuel[id] == 0 then fail("The " .. item.name .. " has no fuel.") end
474 self.state.lit[id] = true
475 return {"The " .. item.name .. " is now lit."}
476 elseif verb == "extinguish" then
477 if not item.light then fail("You can't extinguish the " .. item.name .. ".") end
478 if not self.state.lit[id] then fail("The " .. item.name .. " is not lit.") end
479 self.state.lit[id] = false
480 return {"The " .. item.name .. " is now out."}
481 end
482end
483function Adventure:endTurn(baseline)
484 self.state.moves = self.state.moves + 1
485 local nearby, lines = self:nearby(), {}
486 for _, id in ipairs(self.ids) do
487 if self.state.lit[id] then
488 self.state.fuel[id] = self.state.fuel[id] - 1
489 if self.state.fuel[id] == 0 then
490 self.state.lit[id] = false
491 if nearby[id] then lines[#lines + 1] = "The " .. self.items[id].name .. " goes out." end
492 end
493 end
494 end
495 local lit = self:isLit()
496 if baseline and not lit then lines[#lines + 1] = "It is now pitch dark."
497 elseif not baseline and lit then append(lines, self:describe()) end
498 self.state.darkness = lit and 0 or self.state.darkness + 1
499 if self.state.darkness == 2 then lines[#lines + 1] = "You hear something moving in the dark." end
500 if self.state.darkness == 3 then self.state.ended = "dead" end
501 local score = self:score()
502 if self.state.ended == "none" and score.treasures > 0 and score.stored == score.treasures then self.state.ended = "won" end
503 if self.state.ended ~= "none" then
504 lines[#lines + 1] = self.state.ended == "dead" and "*** You have died ***" or "*** You have won ***"
505 lines[#lines + 1] = "You scored " .. score.total .. " points in " .. self.state.moves .. " moves."
506 end
507 return lines
508end
509function Adventure:interpret(line)
510 local w = normalized(line)
511 if #w == 0 then fail("I beg your pardon?") end
512 if directions[w[1]] then table.insert(w, 1, "go") end
513 local verb = verbs[w[1]]
514 if not verb then fail('I don\'t know the verb "' .. w[1] .. '".') end
515 if self.state.ended ~= "none" and verb ~= "restore" then fail("The game is over.") end
516 local rest = slice(w, 2)
517 if verb == "look" and #rest > 0 then
518 if rest[1] ~= "at" then fail("I don't understand that.") end
519 verb, rest = "examine", slice(rest, 2)
520 end
521 local first, second = rest, nil
522 if noObjects[verb] then
523 if #rest > 0 then fail("I don't understand that.") end
524 else
525 if #rest == 0 then fail(verb == "go" and "Which way do you want to go?" or "What do you want to " .. verb .. "?") end
526 if verb == "go" then
527 if #rest ~= 1 or not directions[rest[1]] then fail("I don't understand that.") end
528 else
529 local prepositions
530 if verb == "put" then prepositions = {["in"] = true, into = true}
531 elseif verb == "use" then prepositions = {on = true, with = true}
532 elseif verb == "unlock" or verb == "lock" then prepositions = {with = true}
533 elseif verb == "take" then prepositions = {from = true} end
534 if prepositions then
535 local index
536 for i, word in ipairs(rest) do if prepositions[word] then index = i; break end end
537 if index then
538 if index == 1 or index == #rest then fail("I don't understand that.") end
539 first, second = slice(rest, 1, index - 1), slice(rest, index + 1)
540 elseif verb == "put" or verb == "use" then fail("I don't understand that.") end
541 end
542 end
543 end
544 local baseline, lines = self:isLit(), nil
545 if verb == "go" then
546 local direction = directions[rest[1]]
547 local exit = (self.state.exits[self.state.room] or {})[direction]
548 if not exit then fail("You can't go that way.") end
549 if exit.locked then fail("The way " .. directionNames[direction] .. " is locked.") end
550 self.state.room = exit.destination
551 baseline = self:isLit()
552 lines = self:describe(self.verbose or not self.state.visited[self.state.room])
553 self.state.visited[self.state.room] = true
554 local extra = self:runRules(self.enterRules, self.state.room)
555 append(lines, extra)
556 else lines = self:execute(verb, first, second) end
557 if actions[verb] then append(lines, self:endTurn(baseline)) end
558 return lines
559end
560function Adventure:command(line)
561 local ok, result = pcall(function()
562 local w = normalized(line)
563 if w[1] == "again" or w[1] == "g" then
564 if #w ~= 1 then fail("I don't understand that.") end
565 if not self.previous then fail("There is nothing to repeat.") end
566 return self:interpret(self.previous)
567 end
568 self.previous = line
569 return self:interpret(line)
570 end)
571 if ok then return result end
572 if type(result) == "table" and result.command_error then return result.lines end
573 error(result, 0)
574end
575function Adventure:finalScore() return "Final score: " .. self:score().total .. " points in " .. self.state.moves .. " moves." end

577local function readLine()
578 local line = io.read("l")
579 if line then return (line:gsub("\r$", "")) end
580end
581local function writeLines(lines) for _, line in ipairs(lines) do io.write(line, "\n") end end
582local definitions = {}
583while true do
584 local line = readLine()
585 if not line or line == "BEGIN" then break end
586 definitions[#definitions + 1] = line
587end
588local adventure = Adventure.new(definitions)
589writeLines(adventure:describe())
590while true do
591 local line = readLine()
592 if not line then break end
593 if line == "QUIT" then io.write(adventure:finalScore(), "\n"); break end
594 io.write("> ", line, "\n")
595 writeLines(adventure:command(line))
596end
