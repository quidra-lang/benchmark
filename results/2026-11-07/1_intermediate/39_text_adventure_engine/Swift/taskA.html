14enum Direction: String, CaseIterable, Hashable {

16 static func parse(_ word: String) -> Direction? {
17 allCases
18 .first { $0.rawValue == word || String($0.rawValue.prefix(1)) == word }
19 }
20 var opposite: Direction {
21 Self.allCases[Self.allCases.firstIndex(of: self)! ^ 1]
22 }
23}

137 private func define(_ line: String) {
138 let parts = words(line), kind = parts[0], id = parts[1]
139 func text(_ index: Int) -> String {
140 parts[index...].joined(separator: " ")
141 }
142 switch kind {
143 case "ROOM":
144 let dark = parts[2] == "DARK"
145 rooms[id] = Room(name: text(dark ? 3 : 2), dark: dark)
146 case "DESC":
147 rooms[id]!.description.append(text(2))
148 case "EXIT":
149 setExit(id, Direction.parse(parts[2])!, parts[3], key: parts.count > 4 ? parts[5] : nil)
150 case "ITEM":
151 var item = Item(id: id, weight: Int(parts[3])!, nouns: [id])
152 var value =
153 ItemState(place: parts[2] == "PLAYER" ? .player : parts[2] == "NOWHERE" ? .nowhere : .room(parts[2]))
154 var index = 4
155 if parts[index] == "TREASURE" {
156 item.points = Int(parts[index + 1])!
157 index += 2
158 }
159 if parts[index] == "LIGHT" {
160 item.light = true
161 value.fuel = Int(parts[index + 1])!
162 index += 2
163 }
164 if parts[index] == "CONTAINER" {
165 item.capacity = Int(parts[index + 1])!
166 index += 2
167 if parts[index] == "CLOSED" {
168 value.container = .closed
169 index += 1
170 }
171 if parts[index] == "LOCKED" {
172 item.key = parts[index + 1]
173 value.container = .locked
174 index += 2
175 }
176 }
177 if parts[index] == "FIXED" {
178 item.fixed = true
179 index += 1
180 }
181 item.name = text(index)
182 item.adjectives = Set(parts[index...])
183 items[id] = item
184 state.items[id] = value
185 case "INSIDE":
186 state.items[id]!.place = .inside(parts[2])
187 case "ALIAS":
188 items[id]!.nouns.insert(parts[2])
189 case "DETAIL":
190 items[id]!.details.append(text(2))
191 case "RULE", "ENTER":
192 var index = kind == "RULE" ? 3 : 2
193 var condition: Condition?
194 if parts[index] == "IF" || parts[index] == "UNLESS" {
195 condition = Condition(flag: parts[index + 1], on: parts[index] == "IF")
196 index += 2
197 }
198 let name = parts[index]
199 index += 1
200 let effect: Effect
201 switch name {
202 case "MESSAGE":
203 effect = .message(text(index))
204 case "OPEN":
205 effect = .openExit(parts[index], Direction.parse(parts[index + 1])!, parts[index + 2])
206 case "REMOVE":
207 effect = .removeItem(parts[index])
208 case "ADD":
209 effect = .addItem(parts[index], parts[index + 1])
210 case "SET", "CLEAR":
211 effect = .setFlag(parts[index], name == "SET")
212 default:
213 effect = .addPoints(Int(parts[index])!)
214 }
215 let rule = Rule(first: id, second: kind == "RULE" ? parts[2] : "", condition: condition, effect: effect)
216 if kind == "RULE" {
217 rules.append(rule)
218 } else {
219 enterRules.append(rule)
220 }
221 case "TROPHY":
222 trophy = id
223 case "START":
224 state.room = id
225 case "MAXWEIGHT":
226 maxWeight = Int(id)!
227 default:
228 break
229 }
230 }

241 private func isOpen(_ id: String) -> Bool {
242 items[id]!.capacity != nil && state.items[id]!.container == .open

244 private func reachable(_ place: Place) -> Set<String> {
245 let grouped = children()
246 var result = Set<String>()
247 func visit(_ place: Place) {
248 for id in grouped[place, default: []] {
249 result.insert(id)
250 if isOpen(id) {
251 visit(.inside(id))
252 }
253 }
254 }
255 visit(place)
256 return result
257 }
258 private func carried() -> Set<String> {
259 reachable(.player)
260 }
261 private func roomItems() -> Set<String> {
262 reachable(.room(state.room))

264 private func isLit() -> Bool {
265 !rooms[state.room]!.dark || carried().union(roomItems())
266 .contains { state.items[$0]!.lit }
267 }
268 private func visible() -> Set<String> {
269 isLit() ? carried().union(roomItems()) : carried()

271 private func weights() -> [String: Int] {
272 let grouped = children()
273 var result = [String: Int]()
274 func total(_ id: String) -> Int {
275 if let weight = result[id] {
276 return weight
277 }
278 let weight = items[id]!.weight + grouped[.inside(id), default: []].reduce(0) { $0 + total($1) }
279 result[id] = weight
280 return weight
281 }
282 for id in ids {
283 _ = total(id)
284 }
285 return result
286 }
287 private func totalWeight(_ id: String) -> Int {
288 weights()[id]!
289 }
290 private func contentWeight(_ id: String) -> Int {
291 totalWeight(id) - items[id]!.weight
292 }
293 private func carriedWeight() -> Int {
294 let totals = weights()
295 return contents(.player).reduce(0) { $0 + totals[$1]! }
296 }
297 private func nestedIn(_ id: String, _ ancestor: String) -> Bool {
298 var place = state.items[id]!.place
299 while case .inside(let parent) = place {
300 if parent == ancestor {
301 return true
302 }
303 place = state.items[parent]!.place
304 }
305 return false
306 }
307 private func score() -> Score {
308 var result = Score()
309 for id in ids where items[id]!.points > 0 {
310 result.treasures += 1
311 var place = state.items[id]!.place
312 while case .inside(let parent) = place {
313 place = state.items[parent]!.place
314 }
315 if place == .room(trophy) {
316 result.stored += 1
317 result.points += items[id]!.points
318 }
319 }
320 result.total = result.points + state.bonus
321 return result
322 }
323 private func display(_ id: String) -> String {
324 let item = items[id]!, value = state.items[id]!
325 var annotations = [String]()
326 if item.capacity != nil {
327 annotations.append(value.container.rawValue)
328 }
329 if item.light && value.lit {
330 annotations.append("lit")

334 private func tree(_ place: Place) -> [String] {
335 let grouped = children()
336 var rows = [String]()
337 func visit(_ place: Place, depth: Int) {
338 for id in grouped[place, default: []] {
339 rows.append(String(repeating: " ", count: depth) + display(id))
340 if isOpen(id) {
341 visit(.inside(id), depth: depth + 1)
342 }
343 }
344 }
345 visit(place, depth: 1)
346 return rows
347 }
348 func describe(full: Bool = true) -> [String] {
349 guard isLit() else {
350 return ["It is pitch dark."]
351 }
352 let room = rooms[state.room]!, exits = state.exits[state.room, default: [:]]
353 let names = Direction.allCases.compactMap { direction -> String? in
354 guard let exit = exits[direction] else {
355 return nil
356 }
357 return direction.rawValue + (exit.locked ? " (locked)" : "")
358 }
359 var rows = [room.name] + (full ? room.description : [])
360 rows.append("Exits: " + (names.isEmpty ? "none" : names.joined(separator: ", ")))
361 if !contents(.room(state.room)).isEmpty {
362 rows += ["You see:"] + tree(.room(state.room))
363 }
364 return rows
365 }
366 private func resolve(_ phrase: [String]) throws -> String {
367 let available = visible()
368 if phrase == ["it"] {
369 guard let pronoun else {
370 try fail("I don't know what \"it\" refers to.")
371 }
372 guard available.contains(pronoun) else {
373 try fail("You can't see any such thing.")
374 }
375 return pronoun
376 }
377 for word in phrase where !vocabulary.contains(word) {
378 try fail("I don't know the word \"\(word)\".")
379 }
380 let matches = ids.filter { id in
381 let item = items[id]!
382 return available.contains(id) && item.nouns.contains(phrase.last!) && phrase.dropLast()
383 .allSatisfy { item.adjectives.contains($0) }
384 }
385 guard !matches.isEmpty else {
386 try fail("You can't see any such thing.")
387 }
388 guard matches.count == 1
389 else {
390 try fail("Which do you mean: " + matches.map { items[$0]!.name }.joined(separator: ", ") + "?")
391 }
392 pronoun = matches[0]
393 return matches[0]
394 }
395 private func runRules(_ rules: [Rule], first: String, second: String = "") -> (rows: [String], executed: Int) {
396 var rows = [String](), executed = 0
397 for rule in rules where rule.first == first && rule.second == second {
398 if let condition = rule.condition, state.flags.contains(condition.flag) != condition.on {
399 continue
400 }
401 executed += 1
402 switch rule.effect {
403 case .message(let text):
404 rows.append(text)
405 case .openExit(let room, let direction, let destination):
406 setExit(room, direction, destination)
407 case .removeItem(let id):
408 state.items[id]!.place = .nowhere
409 case .addItem(let id, let room):
410 state.items[id]!.place = .room(room)
411 case .setFlag(let id, let on):
412 if on { state.flags.insert(id) } else { state.flags.remove(id) }
413 case .addPoints(let amount):
414 state.bonus += amount
415 }
416 }
417 return (rows, executed)
418 }
419 private func requireContainer(_ id: String) throws {
420 guard items[id]!.capacity != nil else {
421 try fail("The \(items[id]!.name) is not a container.")
422 }
423 guard isOpen(id) else {
424 try fail("The \(items[id]!.name) is closed.")
425 }

427 private func take(_ first: [String], _ second: [String]?) throws -> [String] {
428 guard isLit() else {
429 try fail("It is pitch dark.")
430 }
431 var container: String?, source = Place.room(state.room)
432 if let second {
433 let target = try resolve(second)
434 try requireContainer(target)
435 container = target
436 source = .inside(target)
437 }
438 if first == ["all"] {
439 let candidates = contents(source).filter { !items[$0]!.fixed }
440 guard !candidates.isEmpty else {
441 try fail(container == nil ? "There is nothing here to take." :
442 "There is nothing in the \(items[container!]!.name) to take.")
443 }
444 var rows = [String](), taken = false
445 for id in candidates {
446 var outcome = "You can't carry any more."
447 if carried().contains(id) || carriedWeight() + totalWeight(id) <= maxWeight {
448 state.items[id]!.place = .player
449 outcome = "Taken."
450 taken = true
451 }
452 rows.append("\(items[id]!.name): \(outcome)")
453 }
454 guard taken else {
455 try fail(rows)
456 }
457 return rows
458 }
459 let id = try resolve(first), item = items[id]!, value = state.items[id]!
460 if let container,
461 value.place != source
462 {
463 try fail("The \(item.name) is not in the \(items[container]!.name).")
464 }
465 guard value.place != .player else {
466 try fail("You already have the \(item.name).")
467 }
468 guard !item.fixed else {
469 try fail("The \(item.name) is fixed in place.")
470 }
471 guard carried().contains(id) || carriedWeight() + totalWeight(id) <= maxWeight
472 else {
473 try fail("You can't carry any more.")
474 }
475 state.items[id]!.place = .player
476 return ["Taken."]
477 }
478 private func drop(_ phrase: [String]) throws -> [String] {
479 if phrase == ["all"] {
480 let candidates = contents(.player)
481 guard !candidates.isEmpty else {
482 try fail("You are empty-handed.")
483 }
484 return candidates.map { id in
485 state.items[id]!.place = .room(state.room)
486 return "\(items[id]!.name): Dropped."
487 }
488 }
489 let id = try resolve(phrase)
490 guard carried().contains(id) else {
491 try fail("You aren't carrying the \(items[id]!.name).")
492 }
493 state.items[id]!.place = .room(state.room)
494 return ["Dropped."]
495 }
496 private func putItem(_ id: String, _ container: String) -> String? {
497 let item = items[id]!, target = items[container]!
498 if nestedIn(container, id) {
499 return "The \(target.name) is inside the \(item.name)."
500 }
501 if state.items[id]!.place == .inside(container) {
502 return "The \(item.name) is already in the \(target.name)."
503 }
504 let weight = contentWeight(container) + (nestedIn(id, container) ? 0 : totalWeight(id))
505 if weight > target.capacity! {
506 return "The \(item.name) doesn't fit in the \(target.name)."
507 }
508 state.items[id]!.place = .inside(container)
509 return nil
510 }
511 private func put(_ first: [String], _ second: [String]) throws -> [String] {
512 if first == ["all"] {
513 let target = try resolve(second)
514 try requireContainer(target)
515 let candidates = contents(.player).filter { $0 != target }
516 guard !candidates.isEmpty else {
517 try fail("You have nothing to put in the \(items[target]!.name).")
518 }
519 var rows = [String](), succeeded = false
520 for id in candidates {
521 let reason = putItem(id, target)
522 if reason == nil {
523 succeeded = true
524 }
525 rows.append("\(items[id]!.name): \(reason ?? "Done.")")
526 }
527 guard succeeded else {
528 try fail(rows)
529 }
530 return rows
531 }
532 let id = try resolve(first)
533 let target = try resolve(second), item = items[id]!
534 guard carried().contains(id) else {
535 try fail("You aren't carrying the \(item.name).")
536 }
537 guard id != target else {
538 try fail("You can't put the \(item.name) inside itself.")
539 }
540 try requireContainer(target)
541 if let reason = putItem(id, target) {
542 try fail(reason)
543 }
544 return ["You put the \(item.name) in the \(items[target]!.name)."]
545 }
546 private func keyCheck(_ key: String, _ phrase: [String]?) throws -> [String] {
547 if let phrase {
548 let id = try resolve(phrase), name = items[id]!.name
549 guard carried().contains(id) else {
550 try fail("You aren't carrying the \(name).")
551 }
552 guard id == key else {
553 try fail("The \(name) doesn't fit.")
554 }
555 return []
556 }
557 guard carried().contains(key) else {
558 try fail("You don't have the right key.")
559 }
560 return ["(with the \(items[key]!.name))"]
561 }
562 private func lock(_ first: [String], _ second: [String]?, locking: Bool) throws -> [String] {
563 if first.count == 1, let direction = Direction.parse(first[0]) {
564 let room = state.room, name = direction.rawValue
565 guard var exit = state.exits[room]?[direction] else {
566 try fail("There is no exit \(name).")
567 }
568 guard let key = exit.key else {
569 try fail("The way \(name) has no lock.")
570 }
571 guard exit.locked != locking
572 else {
573 try fail("The way \(name) is \(locking ? "already locked" : "not locked").")
574 }
575 let rows = try keyCheck(key, second)
576 exit.locked = locking
577 state.exits[room]![direction] = exit
578 if var back = state.exits[exit.destination]?[direction.opposite], back.destination == room,
579 back.key == key
580 {
581 back.locked = locking
582 state.exits[exit.destination]![direction.opposite] = back
583 }
584 return rows + [locking ? "Locked." : "Unlocked."]
585 }
586 let id = try resolve(first), item = items[id]!, value = state.items[id]!
587 guard item.capacity != nil, let key = item.key else {
588 try fail("The \(item.name) has no lock.")
589 }
590 guard (value.container == .locked) != locking
591 else {
592 try fail("The \(item.name) is \(locking ? "already locked" : "not locked").")
593 }
594 guard !locking || value.container != .open else {
595 try fail("You must close the \(item.name) first.")
596 }
597 let rows = try keyCheck(key, second)
598 state.items[id]!.container = locking ? .locked : .closed
599 return rows + [locking ? "Locked." : "Unlocked."]
600 }
601 private func execute(_ verb: Verb, _ first: [String], _ second: [String]?) throws -> [String] {
602 switch verb {
603 case .look:
604 return describe()
605 case .inventory:
606 if contents(.player).isEmpty {
607 return ["You are empty-handed."]
608 }
609 return ["You are carrying:"] + tree(.player) + ["Total weight: \(carriedWeight())/\(maxWeight)"]
610 case .wait:
611 return ["Time passes."]
612 case .score:
613 let value = score()
614 return [
615 "You have scored \(value.total) points (treasure \(value.points), bonus \(state.bonus)) in \(state.moves) moves.",
616 "Treasures stored: \(value.stored) of \(value.treasures)."
617 ]
618 case .save:
619 saved = state
620 return ["Game saved."]
621 case .restore:
622 guard let saved else {
623 try fail("There is no saved game.")
624 }
625 state = saved
626 return ["Game restored."] + describe()
627 case .verbose:
628 verbose = true
629 return ["Verbose mode."]
630 case .brief:
631 verbose = false
632 return ["Brief mode."]
633 case .take:
634 return try take(first, second)
635 case .drop:
636 return try drop(first)
637 case .put:
638 return try put(first, second!)
639 case .unlock, .lock:
640 return try lock(first, second, locking: verb == .lock)
641 case .use:
642 let id = try resolve(first)
643 let target = try resolve(second!)
644 guard carried().contains(id) else {
645 try fail("You aren't carrying the \(items[id]!.name).")
646 }
647 let result = runRules(rules, first: id, second: target)
648 guard result.executed > 0 else {
649 try fail("Nothing happens.")
650 }
651 return result.rows.isEmpty ? ["Done."] : result.rows
652 default:
653 break
654 }
655 let id = try resolve(first), item = items[id]!, value = state.items[id]!
656 switch verb {
657 case .examine:
658 var rows = item.details.isEmpty ? ["You see nothing special about the \(item.name)."] : item.details
659 if item.light {
660 rows.append("It is \(value.lit ? "lit" : "unlit") (fuel \(value.fuel)).")
661 }
662 if item.capacity != nil {
663 if value.container != .open {
664 rows.append("It is \(value.container.rawValue).")
665 } else if !contents(.inside(id)).isEmpty {
666 rows += ["It contains:"] + tree(.inside(id))
667 } else {
668 rows.append("It is empty.")
669 }
670 }
671 if item.points > 0 {
672 rows.append("It is a treasure worth \(item.points) points.")
673 }
674 rows.append("It weighs \(totalWeight(id)).")
675 return rows
676 case .openContainer:
677 guard item.capacity != nil else {
678 try fail("You can't open the \(item.name).")
679 }
680 guard value.container != .open else {
681 try fail("The \(item.name) is already open.")
682 }
683 guard value.container != .locked else {
684 try fail("The \(item.name) is locked.")
685 }
686 state.items[id]!.container = .open
687 let names = contents(.inside(id)).map { items[$0]!.name }
688 return names
689 .isEmpty ? ["Opened."] : ["Opening the \(item.name) reveals " + names.joined(separator: ", ") + "."]
690 case .closeContainer:
691 guard item.capacity != nil else {
692 try fail("You can't close the \(item.name).")
693 }
694 guard value.container == .open else {
695 try fail("The \(item.name) is already closed.")
696 }
697 state.items[id]!.container = .closed
698 return ["Closed."]
699 case .light:
700 guard item.light else {
701 try fail("You can't light the \(item.name).")
702 }
703 guard !value.lit else {
704 try fail("The \(item.name) is already lit.")
705 }
706 guard value.fuel > 0 else {
707 try fail("The \(item.name) has no fuel.")
708 }
709 state.items[id]!.lit = true
710 return ["The \(item.name) is now lit."]
711 case .extinguish:
712 guard item.light else {
713 try fail("You can't extinguish the \(item.name).")
714 }
715 guard value.lit else {
716 try fail("The \(item.name) is not lit.")
717 }
718 state.items[id]!.lit = false
719 return ["The \(item.name) is now out."]
720 default:
721 return []
722 }
723 }
724 private func endTurn(_ baseline: Bool) -> [String] {
725 state.moves += 1
726 let nearby = carried().union(roomItems())
727 var rows = [String]()
728 for id in ids where state.items[id]!.lit {
729 state.items[id]!.fuel -= 1
730 if state.items[id]!.fuel == 0 {
731 state.items[id]!.lit = false
732 if nearby.contains(id) {
733 rows.append("The \(items[id]!.name) goes out.")
734 }
735 }
736 }
737 let lit = isLit()
738 if baseline && !lit {
739 rows.append("It is now pitch dark.")
740 } else if !baseline && lit {
741 rows += describe()
742 }
743 state.darkness = lit ? 0 : state.darkness + 1
744 if state.darkness == 2 {
745 rows.append("You hear something moving in the dark.")
746 }
747 if state.darkness == 3 {
748 state.ended = .dead
749 }
750 let value = score()
751 if state.ended == .none && value.treasures > 0 && value.stored == value.treasures {
752 state.ended = .won
753 }
754 if state.ended != .none {
755 rows.append(state.ended == .dead ? "*** You have died ***" : "*** You have won ***")
756 rows.append("You scored \(value.total) points in \(state.moves) moves.")
757 }
758 return rows
759 }
760 private func interpret(_ line: String) throws -> [String] {
761 var parts = normalized(line)
762 guard !parts.isEmpty else {
763 try fail("I beg your pardon?")
764 }
765 if Direction.parse(parts[0]) != nil {
766 parts.insert("go", at: 0)
767 }
768 guard var verb = verbWords[parts[0]] else {
769 try fail("I don't know the verb \"\(parts[0])\".")
770 }
771 var rest = Array(parts.dropFirst())
772 guard state.ended == .none || verb == .restore else {
773 try fail("The game is over.")
774 }
775 if verb == .look && !rest.isEmpty {
776 guard rest[0] == "at" else {
777 try fail("I don't understand that.")
778 }
779 verb = .examine
780 rest.removeFirst()
781 }
782 var first = rest
783 var second: [String]?
784 if noObjects.contains(verb) {
785 guard rest.isEmpty else {
786 try fail("I don't understand that.")
787 }
788 } else {
789 guard !rest.isEmpty
790 else {
791 try fail(verb == .go ? "Which way do you want to go?" : "What do you want to \(verb.rawValue)?")
792 }
793 if verb == .go {
794 guard rest.count == 1, Direction.parse(rest[0]) != nil else {
795 try fail("I don't understand that.")
796 }
797 } else {
798 var prepositions = Set<String>()
799 if verb == .put {
800 prepositions = ["in", "into"]
801 } else if verb == .use {
802 prepositions = ["on", "with"]
803 } else if (verb == .unlock || verb == .lock) && rest.contains("with") {
804 prepositions = ["with"]
805 } else if verb == .take && rest.contains("from") {
806 prepositions = ["from"]
807 }
808 if !prepositions.isEmpty {
809 guard let index = rest.firstIndex(where: { prepositions.contains($0) }), index > 0,
810 index + 1 < rest.count
811 else {
812 try fail("I don't understand that.")
813 }
814 first = Array(rest[..<index])
815 second = Array(rest[(index + 1)...])
816 }
817 }
818 }
819 var baseline = isLit()
820 var rows: [String]
821 if verb == .go {
822 let direction = Direction.parse(rest[0])!
823 guard let exit = state.exits[state.room]?[direction] else {
824 try fail("You can't go that way.")
825 }
826 guard !exit.locked else {
827 try fail("The way \(direction.rawValue) is locked.")
828 }
829 state.room = exit.destination
830 baseline = isLit()
831 rows = describe(full: verbose || !state.visited.contains(state.room))
832 state.visited.insert(state.room)
833 rows += runRules(enterRules, first: state.room).rows
834 } else {
835 rows = try execute(verb, first, second)
836 }
837 if actions.contains(verb) {
838 rows += endTurn(baseline)
839 }
840 return rows
841 }
842 func command(_ line: String) throws -> [String] {
843 do {
844 let parts = normalized(line)
845 if let first = parts.first, first == "again" || first == "g" {
846 guard parts.count == 1 else {
847 try fail("I don't understand that.")
848 }
849 guard let previous else {
850 try fail("There is nothing to repeat.")
851 }
852 return try interpret(previous)
853 }
854 previous = line
855 return try interpret(line)
856 } catch let error as Failure { return error.rows }
857 }

859 "Final score: \(score().total) points in \(state.moves) moves."
860 }
