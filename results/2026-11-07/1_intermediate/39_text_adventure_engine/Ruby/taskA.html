137 def read_rule(words, at, trigger)
138 condition = nil
139 if %w[IF UNLESS].include?(words[at])
140 condition = Condition.new(words[at + 1], words[at] == "IF")
141 at += 2
142 end
143
144 effect = case words[at]
145 when "MESSAGE"
146 Message.new(tail(words, at + 1))
147 when "OPEN"
148 OpenExit.new(words[at + 1], direction(words[at + 2]), words[at + 3])
149 when "REMOVE"
150 MoveItem.new(words[at + 1], NOWHERE)
151 when "ADD"
152 MoveItem.new(words[at + 1], Place.new(:floor, words[at + 2]))
153 when "SET", "CLEAR"
154 SetFlag.new(words[at + 1], words[at] == "SET")
155 when "POINTS"
156 Points.new(words[at + 1].to_i(10))
157 end
158
159 Rule.new(trigger, condition, effect)
160 end

162 def define(w)
163 id = w[1]
164 case w[0]
165 when "ROOM"
166 dark = w[2] == "DARK"
167 @rooms[id] = Room.new(tail(w, dark ? 3 : 2), dark, [])
168 when "DESC"
169 @rooms[id].description << tail(w, 2)
170 when "EXIT"
171 set_exit(id, direction(w[2]), w[3], w.size > 4 ? w[5] : nil)
172 when "ITEM"
173 @state.places[id] = case w[2]
174 when "PLAYER"
175 PLAYER
176 when "NOWHERE"
177 NOWHERE
178 else
179 Place.new(:floor, w[2])
180 end
181
182 @state.containers[id] = :open
183 @state.fuel[id] = 0
184 @state.lit[id] = false
185 item = Item.new(
186 name: "",
187 nouns: Set[id],
188 name_words: Set.new,
189 details: [],
190 weight: w[3].to_i(10),
191 treasure: nil,
192 light: false,
193 capacity: nil,
194 key: nil,
195 fixed: false
196 )
197 at = 4
198 if w[at] == "TREASURE"
199 item.treasure = w[at + 1].to_i(10)
200 at += 2
201 end
202
203 if w[at] == "LIGHT"
204 item.light = true
205 @state.fuel[id] = w[at + 1].to_i(10)
206 at += 2
207 end
208
209 if w[at] == "CONTAINER"
210 item.capacity = w[at + 1].to_i(10)
211 at += 2
212 if w[at] == "CLOSED"
213 @state.containers[id] = :closed
214 at += 1
215 end
216
217 if w[at] == "LOCKED"
218 item.key = w[at + 1]
219 @state.containers[id] = :locked
220 at += 2
221 end
222 end
223
224 if w[at] == "FIXED"
225 item.fixed = true
226 at += 1
227 end
228
229 item.name = tail(w, at)
230 item.name_words.merge(w[at..])
231 @items[id] = item
232 when "INSIDE"
233 @state.places[id] = inside(w[2])
234 when "ALIAS"
235 @items[id].nouns.add(w[2])
236 when "DETAIL"
237 @items[id].details << tail(w, 2)
238 when "RULE"
239 @rules << read_rule(w, 3, Trigger.new(:use, id, w[2]))
240 when "ENTER"
241 @rules << read_rule(w, 2, Trigger.new(:enter, id, nil))
242 when "TROPHY"
243 @trophy = id
244 when "START"
245 @state.room = id
246 when "MAXWEIGHT"
247 @max_weight = id.to_i(10)
248 end
249 end

251 def direct(place) = @ids.select { |id| @state.places[id] == place }
252
253 def reachable?(id, root)
254 place = @state.places[id]
255 while place.kind == :inside
256 return false unless @state.containers[place.id] == :open
257 place = @state.places[place.id]
258 end
259
260 place == root
261 end
262
263 def carried?(id) = reachable?(id, PLAYER)
264 def room_item?(id) = reachable?(id, floor)
265 def lit? = !@rooms[@state.room].dark || @ids.any? { |id| @state.lit[id] && (carried?(id) || room_item?(id)) }
266 def total_weight(id) = @items[id].weight + place_weight(inside(id))
267 def place_weight(place) = direct(place).sum { |id| total_weight(id) }
268
269 def within?(id, ancestor)
270 place = @state.places[id]
271 while place.kind == :inside
272 return true if place.id == ancestor
273 place = @state.places[place.id]
274 end
275
276 false
277 end

279 def stored?(id)
280 place = @state.places[id]
281 place = @state.places[place.id] while place.kind == :inside
282 place == Place.new(:floor, @trophy)
283 end

337 def resolve(phrase)
338 lit = lit?
339 if phrase == ["it"]
340 failure("I don't know what \"it\" refers to.") unless @pronoun
341 failure("You can't see any such thing.") unless carried?(@pronoun) || lit && room_item?(@pronoun)
342 return @pronoun
343 end
344
345 phrase.each { |word| failure("I don't know the word \"#{word}\".") unless @vocabulary.include?(word) }
346 matches = @ids.select do |id|
347 (carried?(id) || lit && room_item?(id)) &&
348 @items[id].nouns.include?(phrase.last) &&
349 phrase[0...-1].all? { |word| @items[id].name_words.include?(word) }
350 end
351
352 failure("You can't see any such thing.") if matches.empty?
353 failure("Which do you mean: #{names(matches)}?") if matches.size > 1
354 @pronoun = matches.first
355 end

357 def run_rules(trigger)
358 executed = 0
359 message = false
360 @rules.each do |rule|
361 condition = rule.condition
362 next unless rule.trigger == trigger
363 next if condition && @state.flags.include?(condition.flag) != condition.on
364 executed += 1
365 effect = rule.effect
366 case effect
367 when Message
368 say(effect.text)
369 message = true
370 when OpenExit
371 set_exit(effect.room, effect.direction, effect.destination)
372 when MoveItem
373 @state.places[effect.item] = effect.place
374 when SetFlag
375 effect.on ? @state.flags.add(effect.flag) : @state.flags.delete(effect.flag)
376 when Points
377 @state.bonus += effect.amount
378 end
379 end
380
381 RuleResult.new(executed, message)
382 end

384 def turn(baseline)
385 @state.moves += 1
386 nearby = @ids.to_h { |id| [id, carried?(id) || room_item?(id)] }
387 @ids.each do |id|
388 next unless @state.lit[id]
389 @state.fuel[id] -= 1
390 next unless @state.fuel[id].zero?
391 @state.lit[id] = false
392 say("The #{@items[id].name} goes out.") if nearby[id]
393 end
394
395 lit = lit?
396 if baseline && !lit
397 say("It is now pitch dark.")
398 elsif !baseline && lit
399 describe(true)
400 end
401
402 @state.darkness = lit ? 0 : @state.darkness + 1
403 say("You hear something moving in the dark.") if @state.darkness == 2
404 @state.ended = :dead if @state.darkness == 3
405 treasure = treasures
406 @state.ended = :won if !@state.ended && treasure.total.positive? && treasure.stored == treasure.total
407 return unless @state.ended
408 say(@state.ended == :dead ? "*** You have died ***" : "*** You have won ***")
409 say("You scored #{treasure.points + @state.bonus} points in #{@state.moves} moves.")
410 end

413 failure("The #{@items[id].name} is not a container.") unless @items[id].capacity
414 failure("The #{@items[id].name} is closed.") unless @state.containers[id] == :open

417 def put_item(id, container)
418 failure("The #{@items[container].name} is inside the #{@items[id].name}.") if within?(container, id)
419 if @state.places[id] == inside(container)
420 failure("The #{@items[id].name} is already in the #{@items[container].name}.")
421 end
422
423 weight = place_weight(inside(container)) + (within?(id, container) ? 0 : total_weight(id))
424 if weight > @items[container].capacity
425 failure("The #{@items[id].name} doesn't fit in the #{@items[container].name}.")
426 end
427
428 @state.places[id] = inside(container)
429 end

431 def take(target, from)
432 failure("It is pitch dark.") unless lit?
433 container = nil
434 source = floor
435 unless from.empty?
436 container = resolve(from)
437 require_container(container)
438 source = inside(container)
439 end
440
441 if all?(target)
442 candidates = direct(source).reject { |id| @items[id].fixed }
443 if candidates.empty?
444 failure(
445 container ? "There is nothing in the #{@items[container].name} to take." : "There is nothing here to take."
446 )
447 end
448
449 taken = false
450 candidates.each do |id|
451 if !carried?(id) && place_weight(PLAYER) + total_weight(id) > @max_weight
452 say("#{@items[id].name}: You can't carry any more.")
453 else
454 @state.places[id] = PLAYER
455 say("#{@items[id].name}: Taken.")
456 taken = true
457 end
458 end
459
460 return taken
461 end
462
463 id = resolve(target)
464 if container && @state.places[id] != source
465 failure("The #{@items[id].name} is not in the #{@items[container].name}.")
466 end
467
468 failure("You already have the #{@items[id].name}.") if @state.places[id] == PLAYER
469 failure("The #{@items[id].name} is fixed in place.") if @items[id].fixed
470 failure("You can't carry any more.") if !carried?(id) && place_weight(PLAYER) + total_weight(id) > @max_weight
471 @state.places[id] = PLAYER
472 say("Taken.")
473 true
474 end

476 def drop(target)
477 if all?(target)
478 candidates = direct(PLAYER)
479 failure("You are empty-handed.") if candidates.empty?
480 candidates.each { |id|
481 @state.places[id] = floor
482 say("#{@items[id].name}: Dropped.")
483 }
484 else
485 id = resolve(target)
486 failure("You aren't carrying the #{@items[id].name}.") unless carried?(id)
487 @state.places[id] = floor
488 say("Dropped.")
489 end
490 end

492 def put(first, second)
493 id = all?(first) ? nil : resolve(first)
494 container = resolve(second)
495 if id
496 failure("You aren't carrying the #{@items[id].name}.") unless carried?(id)
497 failure("You can't put the #{@items[id].name} inside itself.") if id == container
498 end
499
500 require_container(container)
501 if id
502 put_item(id, container)
503 say("You put the #{@items[id].name} in the #{@items[container].name}.")
504 return true
505 end
506
507 candidates = direct(PLAYER).reject { |item| item == container }
508 failure("You have nothing to put in the #{@items[container].name}.") if candidates.empty?
509 moved = false
510 candidates.each do |item|
511 message = begin
512 put_item(item, container)
513 moved = true
514 "Done."
515 rescue CommandFailure => error
516 error.message
517 end
518
519 say("#{@items[item].name}: #{message}")
520 end
521
522 moved
523 end

536 def change_lock(target, key_phrase, locking)
537 d = target.size == 1 ? direction(target.first) : nil
538 status = locking ? "already locked" : "not locked"
539 if d
540 exit = @state.exits.fetch(@state.room, {})[d]
541 failure("There is no exit #{DIRECTIONS[d]}.") unless exit
542 failure("The way #{DIRECTIONS[d]} has no lock.") unless exit.key
543 failure("The way #{DIRECTIONS[d]} is #{status}.") if exit.locked == locking
544 key_check(exit.key, key_phrase)
545 @state.exits[@state.room][d] = exit.with(locked: locking)
546 opposite = @state.exits.fetch(exit.destination, {})[d ^ 1]
547 if opposite && opposite.destination == @state.room && opposite.key == exit.key
548 @state.exits[exit.destination][d ^ 1] = opposite.with(locked: locking)
549 end
550 else
551 id = resolve(target)
552 failure("The #{@items[id].name} has no lock.") unless @items[id].capacity && @items[id].key
553 failure("The #{@items[id].name} is #{status}.") if (@state.containers[id] == :locked) == locking
554 failure("You must close the #{@items[id].name} first.") if locking && @state.containers[id] == :open
555 key_check(@items[id].key, key_phrase)
556 @state.containers[id] = locking ? :locked : :closed
557 end
558
559 say(locking ? "Locked." : "Unlocked.")
560 end

590 def execute(verb, first, second, d)
591 baseline = lit?
592 success = true
593 case verb
594 when :go
595 exit = @state.exits.fetch(@state.room, {})[d]
596 failure("You can't go that way.") unless exit
597 failure("The way #{DIRECTIONS[d]} is locked.") if exit.locked
598 @state.room = exit.destination
599 baseline = lit?
600 describe(@verbose || !@state.visited.include?(@state.room))
601 @state.visited.add(@state.room)
602 run_rules(Trigger.new(:enter, @state.room, nil))
603 when :look
604 describe(true)
605 when :examine
606 examine(resolve(first))
607 when :take
608 success = take(first, second)
609 when :drop
610 drop(first)
611 when :put
612 success = put(first, second)
613 when :inventory
614 if direct(PLAYER).empty?
615 say("You are empty-handed.")
616 else
617 say("You are carrying:")
618 item_tree(PLAYER, 1)
619 say("Total weight: #{place_weight(PLAYER)}/#{@max_weight}")
620 end
621
622 when :open
623 id = resolve(first)
624 failure("You can't open the #{@items[id].name}.") unless @items[id].capacity
625 failure("The #{@items[id].name} is already open.") if @state.containers[id] == :open
626 failure("The #{@items[id].name} is locked.") if @state.containers[id] == :locked
627 @state.containers[id] = :open
628 children = direct(inside(id))
629 say(children.empty? ? "Opened." : "Opening the #{@items[id].name} reveals #{names(children)}.")
630 when :close
631 id = resolve(first)
632 failure("You can't close the #{@items[id].name}.") unless @items[id].capacity
633 failure("The #{@items[id].name} is already closed.") unless @state.containers[id] == :open
634 @state.containers[id] = :closed
635 say("Closed.")
636 when :lock, :unlock
637 change_lock(first, second, verb == :lock)
638 when :use
639 a = resolve(first)
640 b = resolve(second)
641 failure("You aren't carrying the #{@items[a].name}.") unless carried?(a)
642 result = run_rules(Trigger.new(:use, a, b))
643 failure("Nothing happens.") if result.executed.zero?
644 say("Done.") unless result.message
645 when :light, :extinguish
646 id = resolve(first)
647 lighting = verb == :light
648 failure("You can't #{verb} the #{@items[id].name}.") unless @items[id].light
649 failure("The #{@items[id].name} is #{lighting ? "already lit" : "not lit"}.") if @state.lit[id] == lighting
650 failure("The #{@items[id].name} has no fuel.") if lighting && @state.fuel[id].zero?
651 @state.lit[id] = lighting
652 say("The #{@items[id].name} is now #{lighting ? "lit" : "out"}.")
653 when :wait
654 say("Time passes.")
655 when :score
656 treasure = treasures
657 say(
658 "You have scored #{treasure.points + @state.bonus} points (treasure #{treasure.points}, bonus #{@state.bonus}) in #{@state.moves} moves."
659 )
660 say("Treasures stored: #{treasure.stored} of #{treasure.total}.")
661 when :save
662 @saved = @state.dup
663 say("Game saved.")
664 when :restore
665 failure("There is no saved game.") unless @saved
666 @state = @saved.dup
667 say("Game restored.")
668 describe(true)
669 when :verbose
670 @verbose = true
671 say("Verbose mode.")
672 when :brief
673 @verbose = false
674 say("Brief mode.")
675 end
676
677 turn(baseline) if success && ACTIONS.include?(verb)
678 end

680 def interpret(input)
681 failure("I beg your pardon?") if input.empty?
682 direction_word = !direction(input.first).nil?
683 verb = direction_word ? :go : VERBS[input.first]
684 failure("I don't know the verb \"#{input.first}\".") unless verb
685 failure("The game is over.") if @state.ended && verb != :restore
686 first = input.drop(direction_word ? 0 : 1)
687 second = []
688 d = nil
689 if verb == :look && !first.empty?
690 failure("I don't understand that.") unless first.first == "at"
691 verb = :examine
692 first = first.drop(1)
693 end
694
695 if NO_ARGUMENTS.include?(verb)
696 failure("I don't understand that.") unless first.empty?
697 else
698 failure(verb == :go ? "Which way do you want to go?" : "What do you want to #{verb}?") if first.empty?
699 if verb == :go
700 d = direction(first.first) if first.size == 1
701 failure("I don't understand that.") unless d
702 else
703 split = first.index do |word|
704 case verb
705 when :put
706 %w[in into].include?(word)
707 when :use
708 %w[on with].include?(word)
709 when :lock, :unlock
710 word == "with"
711 when :take
712 word == "from"
713 else
714 false
715 end
716 end
717
718 if split
719 failure("I don't understand that.") if split.zero? || split + 1 == first.size
720 second = first.drop(split + 1)
721 first = first.take(split)
722 elsif verb == :put || verb == :use
723 failure("I don't understand that.")
724 end
725 end
726 end
