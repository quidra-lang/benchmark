90Rule read_rule(const string[] words, int start, Trigger trigger, const string[] room_ids, const string[] item_ids)
91 at := start
92 Condition | none condition := none
93 if words[at] == "IF" or words[at] == "UNLESS"
94 condition = Condition(words[at + 1], words[at] == "IF")
95 at += 2
96 MessageEffect | OpenEffect | MoveEffect | FlagEffect | PointsEffect effect
97 if words[at] == "MESSAGE"
98 effect = MessageEffect(words[at + 1:].join(" "))
99 elif words[at] == "OPEN"
100 effect = OpenEffect(position_of(room_ids, words[at + 1]), direction_index(words[at + 2]), position_of(room_ids, words[at + 3]))
101 elif words[at] == "REMOVE"
102 effect = MoveEffect(position_of(item_ids, words[at + 1]), Place(PlaceKind.Nowhere))
103 elif words[at] == "ADD"
104 effect = MoveEffect(position_of(item_ids, words[at + 1]), Place(PlaceKind.Floor, position_of(room_ids, words[at + 2])))
105 elif words[at] == "SET" or words[at] == "CLEAR"
106 effect = FlagEffect(words[at + 1], words[at] == "SET")
107 else
108 effect = PointsEffect(int(words[at + 1]))
109 return Rule(trigger, condition, effect)

111Adventure read_world(const string[][] definitions)
112 room_ids := [for words := definitions[] if words[0] == "ROOM" then words[1]]
113 item_ids := [for words := definitions[] if words[0] == "ITEM" then words[1]].sorted()
114 rooms := array(#room_ids, fill = Room())
115 items := array(#item_ids, fill = Item())
116 for index := item_ids.indices()[]

118 state := State(0, array(#item_ids, fill = Place(PlaceKind.Nowhere)), array(#item_ids, fill = ContainerState.Open), array(#item_ids, fill = false), array(#item_ids, fill = 0), array(#room_ids, fill = array(6, fill = Exit())), array(#room_ids, fill = false))
119 Rule[] rules := []
120 trophy := 0
121 max_weight := 0
122 for words := definitions[]
123 command := words[0]
124 if command == "ROOM"
125 index := position_of(room_ids, words[1])
126 dark := words[2] == "DARK"
127 start := if dark then 3 else 2
128 rooms[index].dark = dark
129 rooms[index].name = words[start:].join(" ")
130 elif command == "DESC"
131 index := position_of(room_ids, words[1])
132 rooms[index].description = rooms[index].description.append(words[2:].join(" "))
133 elif command == "EXIT"
134 room := position_of(room_ids, words[1])
135 direction := direction_index(words[2])
136 locked := #words > 4
137 key := if locked then position_of(item_ids, words[5]) else -1
138 state.exits[room][direction] = Exit(position_of(room_ids, words[3]), key, locked)
139 elif command == "ITEM"
140 index := position_of(item_ids, words[1])
141 if words[2] == "PLAYER"
142 state.places[index] = Place(PlaceKind.Player)
143 elif words[2] == "NOWHERE"
144 state.places[index] = Place(PlaceKind.Nowhere)
145 else
146 state.places[index] = Place(PlaceKind.Floor, position_of(room_ids, words[2]))
147 item := items[index]
148 item.weight = int(words[3])
149 at := 4
150 if words[at] == "TREASURE"
151 item.treasure = int(words[at + 1])
152 at += 2
153 if words[at] == "LIGHT"
154 item.light = true
155 state.fuel[index] = int(words[at + 1])
156 at += 2
157 if words[at] == "CONTAINER"
158 item.capacity = int(words[at + 1])
159 at += 2
160 if words[at] == "CLOSED"
161 state.containers[index] = ContainerState.Closed
162 at += 1
163 if words[at] == "LOCKED"
164 item.key = position_of(item_ids, words[at + 1])
165 state.containers[index] = ContainerState.Locked
166 at += 2
167 if words[at] == "FIXED"
168 item.fixed = true
169 at += 1
170 item.name = words[at:].join(" ")
171 item.name_words = words[at:]
172 items[index] = item
173 elif command == "INSIDE"
174 state.places[position_of(item_ids, words[1])] = Place(PlaceKind.Inside, position_of(item_ids, words[2]))
175 elif command == "ALIAS"
176 index := position_of(item_ids, words[1])
177 items[index].nouns = items[index].nouns.append(words[2])
178 elif command == "DETAIL"
179 index := position_of(item_ids, words[1])
180 items[index].details = items[index].details.append(words[2:].join(" "))
181 elif command == "RULE"
182 rules = rules.append(read_rule(words, 3, Trigger(TriggerKind.Use, position_of(item_ids, words[1]), position_of(item_ids, words[2])), room_ids, item_ids))
183 elif command == "ENTER"
184 rules = rules.append(read_rule(words, 2, Trigger(TriggerKind.Enter, position_of(room_ids, words[1])), room_ids, item_ids))
185 elif command == "TROPHY"
186 trophy = position_of(room_ids, words[1])
187 elif command == "START"
188 state.room = position_of(room_ids, words[1])
189 elif command == "MAXWEIGHT"
190 max_weight = int(words[1])
191 state.visited[state.room] = true
192 distinct<string> vocabulary
193 for item := items[]
194 for word := item.nouns.concat(item.name_words)[]
195 .vocabulary.add(word)
196 return Adventure(World(rooms, items, rules, vocabulary, trophy, max_weight), state)
