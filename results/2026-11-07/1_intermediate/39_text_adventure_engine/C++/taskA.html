129 void define(const string& line) {
130 const auto parts = words(line);
131 const auto& kind = parts[0]; const auto& id = parts[1];
132 if (kind == "ROOM") {
133 const bool dark = parts[2] == "DARK";
134 rooms[id] = {join(tail(parts, dark ? 3 : 2)), dark, {}};
135 } else if (kind == "DESC") rooms.at(id).description.push_back(join(tail(parts, 2)));
136 else if (kind == "EXIT") setExit(id, *directionWord(parts[2]), parts[3], parts.size() > 4 ? optional<string>(parts[5]) : nullopt);
137 else if (kind == "ITEM") {
138 Item item; item.id = id; item.weight = stoi(parts[3]); item.nouns.insert(id);
139 ItemState value;
140 value.place = parts[2] == "PLAYER" ? player() : parts[2] == "NOWHERE" ? Place{} : floorOf(parts[2]);
141 size_t index = 4;
142 if (parts[index] == "TREASURE") { item.points = stoi(parts[index + 1]); index += 2; }
143 if (parts[index] == "LIGHT") { item.light = true; value.fuel = stoi(parts[index + 1]); index += 2; }
144 if (parts[index] == "CONTAINER") {
145 item.capacity = stoi(parts[index + 1]); index += 2;
146 if (parts[index] == "CLOSED") { value.container = ContainerState::Closed; ++index; }
147 if (parts[index] == "LOCKED") { item.key = parts[index + 1]; value.container = ContainerState::Locked; index += 2; }
148 }
149 if (parts[index] == "FIXED") { item.fixed = true; ++index; }
150 item.name = join(tail(parts, index));
151 item.adjectives.insert(parts.begin() + index, parts.end());
152 items.emplace(id, move(item)); state.items.emplace(id, move(value));
153 } else if (kind == "INSIDE") state.items.at(id).place = inside(parts[2]);
154 else if (kind == "ALIAS") items.at(id).nouns.insert(parts[2]);
155 else if (kind == "DETAIL") items.at(id).details.push_back(join(tail(parts, 2)));
156 else if (kind == "RULE" || kind == "ENTER") {
157 Rule rule; rule.first = id;
158 size_t index = kind == "RULE" ? 3 : 2;
159 if (kind == "RULE") rule.second = parts[2];
160 if (parts[index] == "IF" || parts[index] == "UNLESS") {
161 rule.condition = Condition{parts[index + 1], parts[index] == "IF"}; index += 2;
162 }
163 const string& effect = parts[index++];
164 if (effect == "MESSAGE") rule.effect = Message{join(tail(parts, index))};
165 else if (effect == "OPEN") rule.effect = OpenExit{parts[index], *directionWord(parts[index + 1]), parts[index + 2]};
166 else if (effect == "REMOVE") rule.effect = RemoveItem{parts[index]};
167 else if (effect == "ADD") rule.effect = AddItem{parts[index], parts[index + 1]};
168 else if (effect == "SET" || effect == "CLEAR") rule.effect = SetFlag{parts[index], effect == "SET"};
169 else rule.effect = AddPoints{stoi(parts[index])};
170 (kind == "RULE" ? rules : enterRules).push_back(move(rule));
171 } else if (kind == "TROPHY") trophy = id;
172 else if (kind == "START") state.room = id;
173 else if (kind == "MAXWEIGHT") maxWeight = stoi(id);
174 }

186 set<string> reachable(const Place& place) const {
187 const auto grouped = children();
188 set<string> result;
189 function<void(const Place&)> visit = [&](const Place& location) {
190 const auto found = grouped.find(location);
191 if (found == grouped.end()) return;
192 for (const auto& id : found->second) {
193 result.insert(id);
194 if (isOpen(id)) visit(inside(id));
195 }
196 };
197 visit(place);
198 return result;

214 map<string, int> weights() const {
215 const auto grouped = children();
216 map<string, int> result;
217 function<int(const string&)> total = [&](const string& id) {
218 if (const auto found = result.find(id); found != result.end()) return found->second;
219 int weight = items.at(id).weight;
220 if (const auto found = grouped.find(inside(id)); found != grouped.end())
221 for (const auto& child : found->second) weight += total(child);
222 result[id] = weight;
223 return weight;
224 };
225 for (const auto& [id, item] : items) total(id);
226 return result;
227 }

262 Lines tree(const Place& place) const {
263 const auto grouped = children();
264 Lines rows;
265 function<void(const Place&, int)> visit = [&](const Place& location, int depth) {
266 const auto found = grouped.find(location);
267 if (found == grouped.end()) return;
268 for (const auto& id : found->second) {
269 rows.push_back(string(2 * depth, ' ') + display(id));
270 if (isOpen(id)) visit(inside(id), depth + 1);
271 }
272 };
273 visit(place, 1);
274 return rows;
275 }
276 Lines describe(bool full = true) const {
277 if (!isLit()) return {"It is pitch dark."};
278 const auto& room = rooms.at(state.room);
279 Lines rows = {room.name}, exits;
280 if (full) append(rows, room.description);
281 if (const auto found = state.exits.find(state.room); found != state.exits.end())
282 for (const auto& [direction, exit] : found->second) exits.push_back(directionName(direction) + (exit.locked ? " (locked)" : ""));
283 rows.push_back("Exits: " + (exits.empty() ? "none" : join(exits, ", ")));
284 if (!contents(floorOf(state.room)).empty()) { rows.push_back("You see:"); append(rows, tree(floorOf(state.room))); }
285 return rows;

287 string resolve(const Lines& phrase) {
288 const auto available = visible();
289 if (phrase.size() == 1 && phrase[0] == "it") {
290 if (!pronoun) fail("I don't know what \"it\" refers to.");
291 if (!available.count(*pronoun)) fail("You can't see any such thing.");
292 return *pronoun;
293 }
294 for (const auto& word : phrase) if (!vocabulary.count(word)) fail("I don't know the word \"" + word + "\".");
295 Lines matches, names;
296 for (const auto& [id, item] : items) if (available.count(id) && item.nouns.count(phrase.back()) &&
297 all_of(phrase.begin(), phrase.end() - 1, [&](const string& word) { return item.adjectives.count(word); })) {
298 matches.push_back(id); names.push_back(item.name);
299 }
300 if (matches.empty()) fail("You can't see any such thing.");
301 if (matches.size() > 1) fail("Which do you mean: " + join(names, ", ") + "?");
302 pronoun = matches[0];
303 return matches[0];
304 }
305 RuleResult runRules(const vector<Rule>& source, const string& first, const string& second = "") {
306 RuleResult result;
307 for (const auto& rule : source) {
308 if (rule.first != first || rule.second != second) continue;
309 if (rule.condition && static_cast<bool>(state.flags.count(rule.condition->flag)) != rule.condition->on) continue;
310 ++result.executed;
311 visit(Overloaded{
312 [&](const Message& effect) { result.rows.push_back(effect.text); },
313 [&](const OpenExit& effect) { setExit(effect.room, effect.direction, effect.destination); },
314 [&](const RemoveItem& effect) { state.items.at(effect.id).place = Place{}; },
315 [&](const AddItem& effect) { state.items.at(effect.id).place = floorOf(effect.room); },
316 [&](const SetFlag& effect) { if (effect.on) state.flags.insert(effect.id); else state.flags.erase(effect.id); },
317 [&](const AddPoints& effect) { state.bonus += effect.amount; }
318 }, rule.effect);
319 }
320 return result;
321 }

327 Lines take(const Lines& first, const optional<Lines>& second) {
328 if (!isLit()) fail("It is pitch dark.");
329 optional<string> container;
330 Place source = floorOf(state.room);
331 if (second) { container = resolve(*second); requireContainer(*container); source = inside(*container); }
332 if (first.size() == 1 && first[0] == "all") {
333 Lines candidates, rows;
334 for (const auto& id : contents(source)) if (!items.at(id).fixed) candidates.push_back(id);
335 if (candidates.empty()) fail(container ? "There is nothing in the " + items.at(*container).name + " to take." : "There is nothing here to take.");
336 bool taken = false;
337 for (const auto& id : candidates) {
338 string outcome = "You can't carry any more.";
339 if (carried().count(id) || carriedWeight() + totalWeight(id) <= maxWeight) {
340 state.items.at(id).place = player(); outcome = "Taken."; taken = true;
341 }
342 rows.push_back(items.at(id).name + ": " + outcome);
343 }
344 if (!taken) fail(move(rows));
345 return rows;
346 }
347 const string id = resolve(first);
348 const auto& item = items.at(id); auto& value = state.items.at(id);
349 if (container && !(value.place == source)) fail("The " + item.name + " is not in the " + items.at(*container).name + ".");
350 if (value.place.kind == PlaceKind::Player) fail("You already have the " + item.name + ".");
351 if (item.fixed) fail("The " + item.name + " is fixed in place.");
352 if (!carried().count(id) && carriedWeight() + totalWeight(id) > maxWeight) fail("You can't carry any more.");
353 value.place = player();
354 return {"Taken."};
355 }

372 optional<string> putItem(const string& id, const string& container) {
373 const auto& item = items.at(id); const auto& target = items.at(container); auto& value = state.items.at(id);
374 if (nestedIn(container, id)) return "The " + target.name + " is inside the " + item.name + ".";
375 if (value.place == inside(container)) return "The " + item.name + " is already in the " + target.name + ".";
376 const int weight = contentWeight(container) + (nestedIn(id, container) ? 0 : totalWeight(id));
377 if (weight > *target.capacity) return "The " + item.name + " doesn't fit in the " + target.name + ".";
378 value.place = inside(container);
379 return nullopt;
380 }
381 Lines put(const Lines& first, const Lines& second) {
382 if (first.size() == 1 && first[0] == "all") {
383 const string target = resolve(second);
384 requireContainer(target);
385 auto candidates = contents(player());
386 candidates.erase(remove(candidates.begin(), candidates.end(), target), candidates.end());
387 if (candidates.empty()) fail("You have nothing to put in the " + items.at(target).name + ".");
388 Lines rows;
389 bool succeeded = false;
390 for (const auto& id : candidates) {
391 const auto reason = putItem(id, target);
392 if (!reason) succeeded = true;
393 rows.push_back(items.at(id).name + ": " + reason.value_or("Done."));
394 }
395 if (!succeeded) fail(move(rows));
396 return rows;
397 }
398 const string id = resolve(first);
399 const string target = resolve(second);
400 const auto& item = items.at(id);
401 if (!carried().count(id)) fail("You aren't carrying the " + item.name + ".");
402 if (id == target) fail("You can't put the " + item.name + " inside itself.");
403 requireContainer(target);
404 if (const auto reason = putItem(id, target)) fail(*reason);
405 return {"You put the " + item.name + " in the " + items.at(target).name + "."};
406 }

418 Lines lock(const Lines& first, const optional<Lines>& second, bool locking) {
419 const auto direction = first.size() == 1 ? directionWord(first[0]) : nullopt;
420 if (direction) {
421 Exit* exit = exitAt(state.room, *direction);
422 const string name = directionName(*direction);
423 if (!exit) fail("There is no exit " + name + ".");
424 if (!exit->key) fail("The way " + name + " has no lock.");
425 if (exit->locked == locking) fail("The way " + name + (locking ? " is already locked." : " is not locked."));
426 auto rows = keyCheck(*exit->key, second);
427 exit->locked = locking;
428 Exit* back = exitAt(exit->destination, opposite(*direction));
429 if (back && back->destination == state.room && back->key == exit->key) back->locked = locking;
430 rows.push_back(locking ? "Locked." : "Unlocked.");
431 return rows;
432 }
433 const string id = resolve(first);
434 const auto& item = items.at(id); auto& value = state.items.at(id);
435 if (!item.capacity || !item.key) fail("The " + item.name + " has no lock.");
436 if ((value.container == ContainerState::Locked) == locking) fail("The " + item.name + (locking ? " is already locked." : " is not locked."));
437 if (locking && value.container == ContainerState::Open) fail("You must close the " + item.name + " first.");
438 auto rows = keyCheck(*item.key, second);
439 value.container = locking ? ContainerState::Locked : ContainerState::Closed;
440 rows.push_back(locking ? "Locked." : "Unlocked.");
441 return rows;
442 }
443 Lines execute(Verb verb, const Lines& first, const optional<Lines>& second) {
444 switch (verb) {
445 case Verb::Look: return describe();
446 case Verb::Inventory: {
447 if (contents(player()).empty()) return {"You are empty-handed."};
448 Lines rows = {"You are carrying:"}; append(rows, tree(player()));
449 rows.push_back("Total weight: " + to_string(carriedWeight()) + "/" + to_string(maxWeight));
450 return rows;
451 }
452 case Verb::Wait: return {"Time passes."};
453 case Verb::Score: {
454 const auto value = score();
455 return {"You have scored " + to_string(value.total) + " points (treasure " + to_string(value.points) + ", bonus " +
456 to_string(state.bonus) + ") in " + to_string(state.moves) + " moves.",
457 "Treasures stored: " + to_string(value.stored) + " of " + to_string(value.treasures) + "."};
458 }
459 case Verb::Save: saved = state; return {"Game saved."};
460 case Verb::Restore: {
461 if (!saved) fail("There is no saved game.");
462 state = *saved;
463 Lines rows = {"Game restored."}; append(rows, describe());
464 return rows;
465 }
466 case Verb::Verbose: verbose = true; return {"Verbose mode."};
467 case Verb::Brief: verbose = false; return {"Brief mode."};
468 case Verb::Take: return take(first, second);
469 case Verb::Drop: return drop(first);
470 case Verb::Put: return put(first, *second);
471 case Verb::Unlock: case Verb::Lock: return lock(first, second, verb == Verb::Lock);
472 case Verb::Use: {
473 const string id = resolve(first);
474 const string target = resolve(*second);
475 if (!carried().count(id)) fail("You aren't carrying the " + items.at(id).name + ".");
476 auto result = runRules(rules, id, target);
477 if (!result.executed) fail("Nothing happens.");
478 return result.rows.empty() ? Lines{"Done."} : result.rows;
479 }
480 default: break;
481 }
482 const string id = resolve(first);
483 const auto& item = items.at(id); auto& value = state.items.at(id);
484 switch (verb) {
485 case Verb::Examine: {
486 Lines rows = item.details.empty() ? Lines{"You see nothing special about the " + item.name + "."} : item.details;
487 if (item.light) rows.push_back(string("It is ") + (value.lit ? "lit" : "unlit") + " (fuel " + to_string(value.fuel) + ").");
488 if (item.capacity) {
489 if (value.container != ContainerState::Open) rows.push_back("It is " + containerText(value.container) + ".");
490 else if (!contents(inside(id)).empty()) { rows.push_back("It contains:"); append(rows, tree(inside(id))); }
491 else rows.push_back("It is empty.");
492 }
493 if (item.points > 0) rows.push_back("It is a treasure worth " + to_string(item.points) + " points.");
494 rows.push_back("It weighs " + to_string(totalWeight(id)) + ".");
495 return rows;
496 }
497 case Verb::Open: {
498 if (!item.capacity) fail("You can't open the " + item.name + ".");
499 if (value.container == ContainerState::Open) fail("The " + item.name + " is already open.");
500 if (value.container == ContainerState::Locked) fail("The " + item.name + " is locked.");
501 value.container = ContainerState::Open;
502 Lines names;
503 for (const auto& child : contents(inside(id))) names.push_back(items.at(child).name);
504 return names.empty() ? Lines{"Opened."} : Lines{"Opening the " + item.name + " reveals " + join(names, ", ") + "."};
505 }
506 case Verb::Close:
507 if (!item.capacity) fail("You can't close the " + item.name + ".");
508 if (value.container != ContainerState::Open) fail("The " + item.name + " is already closed.");
509 value.container = ContainerState::Closed;
510 return {"Closed."};
511 case Verb::Light:
512 if (!item.light) fail("You can't light the " + item.name + ".");
513 if (value.lit) fail("The " + item.name + " is already lit.");
514 if (value.fuel == 0) fail("The " + item.name + " has no fuel.");
515 value.lit = true;
516 return {"The " + item.name + " is now lit."};
517 case Verb::Extinguish:
518 if (!item.light) fail("You can't extinguish the " + item.name + ".");
519 if (!value.lit) fail("The " + item.name + " is not lit.");
520 value.lit = false;
521 return {"The " + item.name + " is now out."};
522 default: return {};
523 }
524 }
525 Lines endTurn(bool baseline) {
526 ++state.moves;
527 auto nearby = carried();
528 const auto ground = roomItems();
529 nearby.insert(ground.begin(), ground.end());
530 Lines rows;
531 for (auto& [id, value] : state.items) if (value.lit && --value.fuel == 0) {
532 value.lit = false;
533 if (nearby.count(id)) rows.push_back("The " + items.at(id).name + " goes out.");
534 }
535 const bool lit = isLit();
536 if (baseline && !lit) rows.push_back("It is now pitch dark.");
537 else if (!baseline && lit) append(rows, describe());
538 state.darkness = lit ? 0 : state.darkness + 1;
539 if (state.darkness == 2) rows.push_back("You hear something moving in the dark.");
540 if (state.darkness == 3) state.ended = EndState::Dead;
541 const auto value = score();
542 if (state.ended == EndState::None && value.treasures > 0 && value.stored == value.treasures) state.ended = EndState::Won;
543 if (state.ended != EndState::None) {
544 rows.push_back(state.ended == EndState::Dead ? "*** You have died ***" : "*** You have won ***");
545 rows.push_back("You scored " + to_string(value.total) + " points in " + to_string(state.moves) + " moves.");
546 }
547 return rows;
548 }
549 Lines interpret(const string& line) {
550 auto parts = normalized(line);
551 if (parts.empty()) fail("I beg your pardon?");
552 if (directionWord(parts[0])) parts.insert(parts.begin(), "go");
553 const auto found = verbWords.find(parts[0]);
554 if (found == verbWords.end()) fail("I don't know the verb \"" + parts[0] + "\".");
555 Verb verb = found->second;
556 auto rest = tail(parts, 1);
557 if (state.ended != EndState::None && verb != Verb::Restore) fail("The game is over.");
558 if (verb == Verb::Look && !rest.empty()) {
559 if (rest[0] != "at") fail("I don't understand that.");
560 verb = Verb::Examine; rest = tail(rest, 1);
561 }
562 Lines first = rest;
563 optional<Lines> second;
564 if (noObjects.count(verb)) {
565 if (!rest.empty()) fail("I don't understand that.");
566 } else {
567 if (rest.empty()) fail(verb == Verb::Go ? "Which way do you want to go?" : "What do you want to " + verbNames[static_cast<size_t>(verb)] + "?");
568 if (verb == Verb::Go) {
569 if (rest.size() != 1 || !directionWord(rest[0])) fail("I don't understand that.");
570 } else {
571 set<string> prepositions;
572 if (verb == Verb::Put) prepositions = {"in", "into"};
573 else if (verb == Verb::Use) prepositions = {"on", "with"};
574 else if ((verb == Verb::Unlock || verb == Verb::Lock) && find(rest.begin(), rest.end(), "with") != rest.end()) prepositions = {"with"};
575 else if (verb == Verb::Take && find(rest.begin(), rest.end(), "from") != rest.end()) prepositions = {"from"};
576 if (!prepositions.empty()) {
577 const auto split = find_if(rest.begin(), rest.end(), [&](const string& word) { return prepositions.count(word); });
578 if (split == rest.end() || split == rest.begin() || split + 1 == rest.end()) fail("I don't understand that.");
579 first = Lines(rest.begin(), split); second = Lines(split + 1, rest.end());
580 }
581 }
582 }
583 bool baseline = isLit();
584 Lines rows;
585 if (verb == Verb::Go) {
586 const Direction direction = *directionWord(rest[0]);
587 Exit* exit = exitAt(state.room, direction);
588 if (!exit) fail("You can't go that way.");
589 if (exit->locked) fail("The way " + directionName(direction) + " is locked.");
590 state.room = exit->destination;
591 baseline = isLit();
592 rows = describe(verbose || !state.visited.count(state.room));
593 state.visited.insert(state.room);
594 append(rows, runRules(enterRules, state.room).rows);
595 } else rows = execute(verb, first, second);
596 if (actions.count(verb)) append(rows, endTurn(baseline));
597 return rows;
598 }

600public:
601 explicit Adventure(const Lines& definitions) {
602 for (const auto& line : definitions) define(line);
603 state.visited.insert(state.room);
604 for (const auto& [id, item] : items) {
605 vocabulary.insert(item.nouns.begin(), item.nouns.end());
606 vocabulary.insert(item.adjectives.begin(), item.adjectives.end());
607 }
608 }
609 Lines initialDescription() const { return describe(); }
610 string finalScore() const { return "Final score: " + to_string(score().total) + " points in " + to_string(state.moves) + " moves."; }
611 Lines command(const string& line) {
612 try {
613 const auto parts = normalized(line);
614 if (!parts.empty() && (parts[0] == "again" || parts[0] == "g")) {
615 if (parts.size() != 1) fail("I don't understand that.");
616 if (!previous) fail("There is nothing to repeat.");
617 return interpret(*previous);
618 }
619 previous = line;
620 return interpret(line);
621 } catch (const Failure& error) { return error.rows; }
622 }
623};

631 Adventure adventure(definitions);
632 for (const auto& row : adventure.initialDescription()) cout << row << '\n';
633 while (getline(cin, line)) {
634 if (line == "QUIT") { cout << adventure.finalScore() << '\n'; break; }
635 cout << "> " << line << '\n';
636 for (const auto& row : adventure.command(line)) cout << row << '\n';
