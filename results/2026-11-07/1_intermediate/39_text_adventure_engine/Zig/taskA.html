82fn equal(a: []const u8, b: []const u8) bool {
83 return std.mem.eql(u8, a, b);
84}
85fn contains(words: []const []const u8, word: []const u8) bool {

89fn integer(text: []const u8) i32 {
90 return std.fmt.parseInt(i32, text, 10) catch unreachable;
91}
92fn directionOf(word: []const u8) ?Direction {
93 for (directions, 0..) |name, index| if (equal(word, name) or equal(word, name[0..1])) return @fromBackingInt(@intCast(index));
94 return null;
95}
96fn verbOf(word: []const u8) ?Verb {
97 for (synonyms, 0..) |group, index| {
98 var tokens = std.mem.tokenizeScalar(u8, group, ' ');
99 while (tokens.next()) |alias| if (equal(alias, word)) return @fromBackingInt(@intCast(index));
100 }
101 return null;
102}
103fn all(phrase: []const []const u8) bool {
104 return phrase.len == 1 and equal(phrase[0], "all");
105}
106fn line(out: *Writer, comptime format: []const u8, args: anytype) Writer.Error!void {
107 try out.print(format ++ "\n", args);
108}
109fn splitWords(gpa: Allocator, text: []const u8) Allocator.Error![]const []const u8 {
110 var result: std.ArrayList([]const u8) = .empty;
111 var tokens = std.mem.tokenizeScalar(u8, text, ' ');
112 while (tokens.next()) |word| try result.append(gpa, word);
113 return result.toOwnedSlice(gpa);
114}
115fn normalized(gpa: Allocator, text: []const u8) Allocator.Error![]const []const u8 {
116 const lower = try gpa.dupe(u8, text);
117 for (lower) |*character| if (character.* >= 'A' and character.* <= 'Z') {
118 character.* += 'a' - 'A';
119 };
120 var result: std.ArrayList([]const u8) = .empty;
121 var tokens = std.mem.tokenizeScalar(u8, lower, ' ');
122 while (tokens.next()) |word| {
123 if (!equal(word, "the") and !equal(word, "a") and !equal(word, "an")) try result.append(gpa, word);
124 }
125 return result.toOwnedSlice(gpa);
126}
127fn joined(gpa: Allocator, words: []const []const u8) Error![]const u8 {
128 var buffer: Writer.Allocating = .init(gpa);
129 defer buffer.deinit();
130 for (words, 0..) |word, index| {
131 if (index > 0) try buffer.writer.writeByte(' ');
132 try buffer.writer.writeAll(word);
133 }
134 return buffer.toOwnedSlice();
135}

152 for (game.rooms.items, 0..) |room, index| if (equal(room.id, id)) return index;

159 fn flagId(game: *Adventure, id: []const u8) Allocator.Error!usize {
160 for (game.flags.items, 0..) |flag, index| if (equal(flag, id)) return index;
161 const index = game.flags.items.len;
162 try game.flags.append(game.gpa, id);
163 return index;

165 fn parseRule(game: *Adventure, words: []const []const u8, start: usize) Error!Rule {
166 var index = start;
167 var flag: ?usize = null;
168 var when_on = true;
169 if (equal(words[index], "IF") or equal(words[index], "UNLESS")) {
170 when_on = equal(words[index], "IF");
171 flag = try game.flagId(words[index + 1]);
172 index += 2;
173 }
174 const effect: Effect = if (equal(words[index], "MESSAGE"))
175 .{ .message = try joined(game.gpa, words[index + 1 ..]) }
176 else if (equal(words[index], "OPEN"))
177 .{ .open = .{ .room = game.roomId(words[index + 1]), .direction = directionOf(words[index + 2]).?, .destination = game.roomId(words[index + 3]) } }
178 else if (equal(words[index], "REMOVE"))
179 .{ .move = .{ .item = game.itemId(words[index + 1]), .place = .{} } }
180 else if (equal(words[index], "ADD"))
181 .{ .move = .{ .item = game.itemId(words[index + 1]), .place = .{ .kind = .room, .index = game.roomId(words[index + 2]) } } }
182 else if (equal(words[index], "SET") or equal(words[index], "CLEAR"))
183 .{ .flag = .{ .index = try game.flagId(words[index + 1]), .on = equal(words[index], "SET") } }
184 else
185 .{ .points = integer(words[index + 1]) };
186 return .{ .flag = flag, .when_on = when_on, .effect = effect };
187 }
188 fn lessItem(_: void, a: Item, b: Item) bool {
189 return std.mem.order(u8, a.id, b.id) == .lt;
190 }
191 fn create(gpa: Allocator, definitions: []const []const u8) Error!*Adventure {
192 const game = try gpa.create(Adventure);
193 game.* = .{ .gpa = gpa };
194 for (definitions) |text| {
195 const words = try splitWords(gpa, text);
196 if (equal(words[0], "ROOM")) try game.rooms.append(gpa, .{ .id = words[1] });
197 if (equal(words[0], "ITEM")) try game.items.append(gpa, .{ .id = words[1] });
198 }
199 std.sort.heap(Item, game.items.items, {}, lessItem);
200 for (definitions) |text| {
201 const words = try splitWords(gpa, text);
202 const kind = words[0];
203 if (equal(kind, "ROOM")) {
204 const room = &game.rooms.items[game.roomId(words[1])];
205 room.dark = equal(words[2], "DARK");
206 const start: usize = if (room.dark) 3 else 2;
207 room.name = try joined(gpa, words[start..]);
208 } else if (equal(kind, "DESC")) {
209 try game.rooms.items[game.roomId(words[1])].descriptions.append(gpa, try joined(gpa, words[2..]));
210 } else if (equal(kind, "EXIT")) {
211 game.state.exits[game.roomId(words[1])][@backingInt(directionOf(words[2]).?)] = .{
212 .destination = game.roomId(words[3]),
213 .key = if (words.len > 4) game.itemId(words[5]) else null,
214 .locked = words.len > 4,
215 };
216 } else if (equal(kind, "ITEM")) {
217 const x = game.itemId(words[1]);
218 const item = &game.items.items[x];
219 game.state.places[x] = if (equal(words[2], "PLAYER")) player else if (equal(words[2], "NOWHERE")) .{} else .{ .kind = .room, .index = game.roomId(words[2]) };
220 item.weight = integer(words[3]);
221 try item.nouns.append(gpa, words[1]);
222 var index: usize = 4;
223 if (equal(words[index], "TREASURE")) {
224 item.treasure = integer(words[index + 1]);
225 index += 2;
226 }
227 if (equal(words[index], "LIGHT")) {
228 item.light = true;
229 game.state.fuel[x] = integer(words[index + 1]);
230 index += 2;
231 }
232 if (equal(words[index], "CONTAINER")) {
233 item.capacity = integer(words[index + 1]);
234 index += 2;
235 if (equal(words[index], "CLOSED")) {
236 game.state.containers[x] = .closed;
237 index += 1;
238 }
239 if (equal(words[index], "LOCKED")) {
240 item.key = game.itemId(words[index + 1]);
241 game.state.containers[x] = .locked;
242 index += 2;
243 }
244 }
245 if (equal(words[index], "FIXED")) {
246 item.fixed = true;
247 index += 1;
248 }
249 item.name = try joined(gpa, words[index..]);
250 item.name_words = words[index..];
251 } else if (equal(kind, "INSIDE")) {
252 game.state.places[game.itemId(words[1])] = .{ .kind = .inside, .index = game.itemId(words[2]) };
253 } else if (equal(kind, "ALIAS")) {
254 try game.items.items[game.itemId(words[1])].nouns.append(gpa, words[2]);
255 } else if (equal(kind, "DETAIL")) {
256 try game.items.items[game.itemId(words[1])].details.append(gpa, try joined(gpa, words[2..]));
257 } else if (equal(kind, "RULE")) {
258 try game.use_rules.append(gpa, .{ .first = game.itemId(words[1]), .second = game.itemId(words[2]), .rule = try game.parseRule(words, 3) });
259 } else if (equal(kind, "ENTER")) {
260 try game.rooms.items[game.roomId(words[1])].enter.append(gpa, try game.parseRule(words, 2));
261 } else if (equal(kind, "TROPHY")) game.trophy = game.roomId(words[1]) else if (equal(kind, "START")) game.state.room = game.roomId(words[1]) else if (equal(kind, "MAXWEIGHT")) game.max_weight = integer(words[1]);
262 }
263 game.state.visited[game.state.room] = true;
264 return game;
265 }
266 fn floor(game: *const Adventure) Place {
267 return .{ .kind = .room, .index = game.state.room };
268 }
269 fn direct(game: *const Adventure, place: Place) Selection {
270 var result: Selection = .{};
271 for (0..game.items.items.len) |x| if (game.state.places[x].equal(place)) {
272 result.add(x);
273 };
274 return result;
275 }
276 fn reachable(game: *const Adventure, x: usize, root: Place) bool {
277 var place = game.state.places[x];
278 while (true) {
279 if (place.equal(root)) return true;
280 if (place.kind != .inside or game.state.containers[place.index] != .open) return false;
281 place = game.state.places[place.index];
282 }
283 }
284 fn carried(game: *const Adventure, x: usize) bool {
285 return game.reachable(x, player);
286 }
287 fn isLit(game: *const Adventure) bool {
288 if (!game.rooms.items[game.state.room].dark) return true;
289 for (0..game.items.items.len) |x| if (game.state.lit[x] and (game.carried(x) or game.reachable(x, game.floor()))) return true;
290 return false;
291 }
292 fn visible(game: *const Adventure, x: usize, lit: bool) bool {
293 return game.carried(x) or (lit and game.reachable(x, game.floor()));
294 }
295 fn contentWeight(game: *const Adventure, container: usize) i32 {
296 var result: i32 = 0;
297 for (0..game.items.items.len) |x| if (game.state.places[x].equal(.{ .kind = .inside, .index = container })) {
298 result += game.totalWeight(x);
299 };
300 return result;
301 }
302 fn totalWeight(game: *const Adventure, x: usize) i32 {
303 return game.items.items[x].weight + game.contentWeight(x);
304 }
305 fn carriedWeight(game: *const Adventure) i32 {
306 var result: i32 = 0;
307 for (0..game.items.items.len) |x| if (game.state.places[x].equal(player)) {
308 result += game.totalWeight(x);
309 };
310 return result;
311 }
312 fn within(game: *const Adventure, item: usize, container: usize) bool {
313 var place = game.state.places[item];
314 while (place.kind == .inside) {
315 if (place.index == container) return true;
316 place = game.state.places[place.index];
317 }
318 return false;
319 }
320 fn treasures(game: *const Adventure) TreasureScore {
321 var result: TreasureScore = .{};
322 for (game.items.items, 0..) |item, x| if (item.treasure) |points| {
323 result.total += 1;
324 var place = game.state.places[x];
325 while (place.kind == .inside) place = game.state.places[place.index];
326 if (place.equal(.{ .kind = .room, .index = game.trophy })) {
327 result.stored += 1;
328 result.points += points;
329 }
330 };
331 return result;
332 }
333 fn knownWord(game: *const Adventure, word: []const u8) bool {
334 for (game.items.items) |item| if (contains(item.nouns.items, word) or contains(item.name_words, word)) return true;
335 return false;
336 }

367const Session = struct {
368 game: *Adventure,
369 gpa: Allocator,
370 out: *Writer,
371 problem: []const u8 = "",
372
373 fn fail(session: *Session, comptime format: []const u8, args: anytype) Error {
374 session.problem = try std.fmt.allocPrint(session.gpa, format, args);
375 return error.Invalid;
376 }
377 fn say(session: *Session, comptime format: []const u8, args: anytype) Error!void {
378 try line(session.out, format, args);
379 }
380 fn itemTree(session: *Session, place: Place, depth: usize) Error!void {
381 const game = session.game;
382 const children = game.direct(place);
383 for (children.slice()) |x| {
384 try session.out.splatByteAll(' ', depth * 2);
385 try session.say("{f}", .{DisplayName{ .game = game, .index = x }});
386 if (game.items.items[x].capacity != null and game.state.containers[x] == .open)
387 try session.itemTree(.{ .kind = .inside, .index = x }, depth + 1);
388 }
389 }
390 fn describe(session: *Session, full: bool) Error!void {
391 const game = session.game;
392 if (!game.isLit()) return session.say("It is pitch dark.", .{});
393 const room = &game.rooms.items[game.state.room];
394 try session.say("{s}", .{room.name});
395 if (full) for (room.descriptions.items) |text| {
396 try session.say("{s}", .{text});
397 };
398 try session.out.writeAll("Exits: ");
399 var first = true;
400 for (game.state.exits[game.state.room], 0..) |possible, d| if (possible) |exit| {
401 if (!first) try session.out.writeAll(", ");
402 first = false;
403 try session.out.writeAll(directions[d]);
404 if (exit.locked) try session.out.writeAll(" (locked)");
405 };
406 if (first) try session.out.writeAll("none");
407 try session.out.writeByte('\n');
408 if (game.direct(game.floor()).len > 0) {
409 try session.say("You see:", .{});
410 try session.itemTree(game.floor(), 1);
411 }
412 }
413 fn resolve(session: *Session, phrase: []const []const u8) Error!usize {
414 const game = session.game;
415 const lit = game.isLit();
416 if (phrase.len == 1 and equal(phrase[0], "it")) {
417 const x = game.pronoun orelse return session.fail("I don't know what \"it\" refers to.", .{});
418 if (!game.visible(x, lit)) return session.fail("You can't see any such thing.", .{});
419 return x;
420 }
421 for (phrase) |word| if (!game.knownWord(word)) return session.fail("I don't know the word \"{s}\".", .{word});
422 var matches: Selection = .{};
423 for (game.items.items, 0..) |item, x| {
424 if (!game.visible(x, lit) or !contains(item.nouns.items, phrase[phrase.len - 1])) continue;
425 var match = true;
426 for (phrase[0 .. phrase.len - 1]) |word| if (!contains(item.name_words, word)) {
427 match = false;
428 break;
429 };
430 if (match) matches.add(x);
431 }
432 if (matches.len == 0) return session.fail("You can't see any such thing.", .{});
433 if (matches.len > 1) return session.fail("Which do you mean: {f}?", .{Names{ .game = game, .selected = matches }});
434 game.pronoun = matches.data[0];
435 return matches.data[0];
436 }
437 fn examine(session: *Session, x: usize) Error!void {
438 const game = session.game;
439 const item = &game.items.items[x];
440 if (item.details.items.len == 0) try session.say("You see nothing special about the {s}.", .{item.name}) else for (item.details.items) |text| {
441 try session.say("{s}", .{text});
442 }
443 if (item.light) try session.say("It is {s} (fuel {d}).", .{ if (game.state.lit[x]) "lit" else "unlit", game.state.fuel[x] });
444 if (item.capacity != null) switch (game.state.containers[x]) {
445 .locked => try session.say("It is locked.", .{}),
446 .closed => try session.say("It is closed.", .{}),
447 .open => {
448 const place: Place = .{ .kind = .inside, .index = x };
449 if (game.direct(place).len == 0) try session.say("It is empty.", .{}) else {
450 try session.say("It contains:", .{});
451 try session.itemTree(place, 1);
452 }
453 },
454 };
455 if (item.treasure) |points| try session.say("It is a treasure worth {d} points.", .{points});
456 try session.say("It weighs {d}.", .{game.totalWeight(x)});
457 }
458 fn requireContainer(session: *Session, x: usize) Error!void {
459 if (session.game.items.items[x].capacity == null) return session.fail("The {s} is not a container.", .{session.game.items.items[x].name});
460 if (session.game.state.containers[x] != .open) return session.fail("The {s} is closed.", .{session.game.items.items[x].name});
461 }
462 fn take(session: *Session, target: []const []const u8, from: ?[]const []const u8) Error!bool {
463 const game = session.game;
464 if (!game.isLit()) return session.fail("It is pitch dark.", .{});
465 var container: ?usize = null;
466 var source = game.floor();
467 if (from) |phrase| {
468 const c = try session.resolve(phrase);
469 try session.requireContainer(c);
470 container = c;
471 source = .{ .kind = .inside, .index = c };
472 }
473 if (all(target)) {
474 var candidates: Selection = .{};
475 const direct = game.direct(source);
476 for (direct.slice()) |x| if (!game.items.items[x].fixed) {
477 candidates.add(x);
478 };
479 if (candidates.len == 0) {
480 if (container) |c| return session.fail("There is nothing in the {s} to take.", .{game.items.items[c].name});
481 return session.fail("There is nothing here to take.", .{});
482 }
483 var taken = false;
484 for (candidates.slice()) |x| {
485 if (!game.carried(x) and game.carriedWeight() + game.totalWeight(x) > game.max_weight)
486 try session.say("{s}: You can't carry any more.", .{game.items.items[x].name})
487 else {
488 game.state.places[x] = player;
489 taken = true;
490 try session.say("{s}: Taken.", .{game.items.items[x].name});
491 }
492 }
493 return taken;
494 }
495 const x = try session.resolve(target);
496 if (container) |c| if (!game.state.places[x].equal(source)) {
497 return session.fail("The {s} is not in the {s}.", .{ game.items.items[x].name, game.items.items[c].name });
498 };
499 if (game.state.places[x].equal(player)) return session.fail("You already have the {s}.", .{game.items.items[x].name});
500 if (game.items.items[x].fixed) return session.fail("The {s} is fixed in place.", .{game.items.items[x].name});
501 if (!game.carried(x) and game.carriedWeight() + game.totalWeight(x) > game.max_weight) return session.fail("You can't carry any more.", .{});
502 game.state.places[x] = player;
503 try session.say("Taken.", .{});
504 return true;
505 }
506 fn drop(session: *Session, target: []const []const u8) Error!void {
507 const game = session.game;
508 if (all(target)) {
509 const direct = game.direct(player);
510 if (direct.len == 0) return session.fail("You are empty-handed.", .{});
511 for (direct.slice()) |x| {
512 game.state.places[x] = game.floor();
513 try session.say("{s}: Dropped.", .{game.items.items[x].name});
514 }
515 return;
516 }
517 const x = try session.resolve(target);
518 if (!game.carried(x)) return session.fail("You aren't carrying the {s}.", .{game.items.items[x].name});
519 game.state.places[x] = game.floor();
520 try session.say("Dropped.", .{});
521 }
522 fn putItem(session: *Session, x: usize, y: usize) Error!?[]const u8 {
523 const game = session.game;
524 const xn = game.items.items[x].name;
525 const yn = game.items.items[y].name;
526 if (game.within(y, x)) return try std.fmt.allocPrint(session.gpa, "The {s} is inside the {s}.", .{ yn, xn });
527 if (game.state.places[x].equal(.{ .kind = .inside, .index = y })) return try std.fmt.allocPrint(session.gpa, "The {s} is already in the {s}.", .{ xn, yn });
528 const weight = game.contentWeight(y) + (if (game.within(x, y)) @as(i32, 0) else game.totalWeight(x));
529 if (weight > game.items.items[y].capacity.?) return try std.fmt.allocPrint(session.gpa, "The {s} doesn't fit in the {s}.", .{ xn, yn });
530 game.state.places[x] = .{ .kind = .inside, .index = y };
531 return null;
532 }
533 fn put(session: *Session, first: []const []const u8, second: []const []const u8) Error!bool {
534 const game = session.game;
535 if (all(first)) {
536 const y = try session.resolve(second);
537 try session.requireContainer(y);
538 const direct = game.direct(player);
539 var candidates: Selection = .{};
540 for (direct.slice()) |x| if (x != y) {
541 candidates.add(x);
542 };
543 if (candidates.len == 0) return session.fail("You have nothing to put in the {s}.", .{game.items.items[y].name});
544 var moved = false;
545 for (candidates.slice()) |x| {
546 const reason = try session.putItem(x, y);
547 if (reason == null) moved = true;
548 try session.say("{s}: {s}", .{ game.items.items[x].name, reason orelse "Done." });
549 }
550 return moved;
551 }
552 const x = try session.resolve(first);
553 const y = try session.resolve(second);
554 if (!game.carried(x)) return session.fail("You aren't carrying the {s}.", .{game.items.items[x].name});
555 if (x == y) return session.fail("You can't put the {s} inside itself.", .{game.items.items[x].name});
556 try session.requireContainer(y);
557 if (try session.putItem(x, y)) |reason| return session.fail("{s}", .{reason});
558 try session.say("You put the {s} in the {s}.", .{ game.items.items[x].name, game.items.items[y].name });
559 return true;
560 }
561 fn open(session: *Session, x: usize) Error!void {
562 const game = session.game;
563 const name = game.items.items[x].name;
564 if (game.items.items[x].capacity == null) return session.fail("You can't open the {s}.", .{name});
565 if (game.state.containers[x] == .open) return session.fail("The {s} is already open.", .{name});
566 if (game.state.containers[x] == .locked) return session.fail("The {s} is locked.", .{name});
567 game.state.containers[x] = .open;
568 const children = game.direct(.{ .kind = .inside, .index = x });
569 if (children.len == 0) try session.say("Opened.", .{}) else try session.say("Opening the {s} reveals {f}.", .{ name, Names{ .game = game, .selected = children } });
570 }
571 fn close(session: *Session, x: usize) Error!void {
572 const game = session.game;
573 if (game.items.items[x].capacity == null) return session.fail("You can't close the {s}.", .{game.items.items[x].name});
574 if (game.state.containers[x] != .open) return session.fail("The {s} is already closed.", .{game.items.items[x].name});
575 game.state.containers[x] = .closed;
576 try session.say("Closed.", .{});
577 }
578 fn checkKey(session: *Session, key: usize, phrase: ?[]const []const u8) Error!void {
579 const game = session.game;
580 if (phrase) |words| {
581 const x = try session.resolve(words);
582 if (!game.carried(x)) return session.fail("You aren't carrying the {s}.", .{game.items.items[x].name});
583 if (x != key) return session.fail("The {s} doesn't fit.", .{game.items.items[x].name});
584 } else {
585 if (!game.carried(key)) return session.fail("You don't have the right key.", .{});
586 try session.say("(with the {s})", .{game.items.items[key].name});
587 }
588 }
589 fn lock(session: *Session, target: []const []const u8, key_phrase: ?[]const []const u8, locking: bool) Error!void {
590 const game = session.game;
591 const direction: ?Direction = if (target.len == 1) directionOf(target[0]) else null;
592 if (direction) |d| {
593 const index = @backingInt(d);
594 const exit = game.state.exits[game.state.room][index] orelse return session.fail("There is no exit {s}.", .{@tagName(d)});
595 const key = exit.key orelse return session.fail("The way {s} has no lock.", .{@tagName(d)});
596 if (exit.locked == locking) return session.fail("The way {s} is {s}.", .{ @tagName(d), if (locking) "already locked" else "not locked" });
597 try session.checkKey(key, key_phrase);
598 game.state.exits[game.state.room][index].?.locked = locking;
599 if (game.state.exits[exit.destination][index ^ 1]) |back| {
600 if (back.destination == game.state.room and back.key == exit.key) game.state.exits[exit.destination][index ^ 1].?.locked = locking;
601 }
602 } else {
603 const x = try session.resolve(target);
604 const item = &game.items.items[x];
605 if (item.capacity == null or item.key == null) return session.fail("The {s} has no lock.", .{item.name});
606 if ((game.state.containers[x] == .locked) == locking) return session.fail("The {s} is {s}.", .{ item.name, if (locking) "already locked" else "not locked" });
607 if (locking and game.state.containers[x] == .open) return session.fail("You must close the {s} first.", .{item.name});
608 try session.checkKey(item.key.?, key_phrase);
609 game.state.containers[x] = if (locking) .locked else .closed;
610 }
611 try session.say("{s}", .{if (locking) "Locked." else "Unlocked."});
612 }
613 fn runRule(session: *Session, rule: Rule) Error!?bool {
614 const game = session.game;
615 if (rule.flag) |flag| if (game.state.flags[flag] != rule.when_on) return null;
616 switch (rule.effect) {
617 .message => |text| {
618 try session.say("{s}", .{text});
619 return true;
620 },
621 .open => |effect| game.state.exits[effect.room][@backingInt(effect.direction)] = .{ .destination = effect.destination },
622 .move => |effect| game.state.places[effect.item] = effect.place,
623 .flag => |effect| game.state.flags[effect.index] = effect.on,
624 .points => |amount| game.state.bonus += amount,
625 }
626 return false;
627 }
628 fn use(session: *Session, first: []const []const u8, second: []const []const u8) Error!void {
629 const game = session.game;
630 const x = try session.resolve(first);
631 const y = try session.resolve(second);
632 if (!game.carried(x)) return session.fail("You aren't carrying the {s}.", .{game.items.items[x].name});
633 var executed: usize = 0;
634 var message = false;
635 for (game.use_rules.items) |entry| if (entry.first == x and entry.second == y) {
636 if (try session.runRule(entry.rule)) |printed| {
637 executed += 1;
638 message = message or printed;
639 }
640 };
641 if (executed == 0) return session.fail("Nothing happens.", .{});
642 if (!message) try session.say("Done.", .{});
643 }
644 fn light(session: *Session, x: usize, lighting: bool) Error!void {
645 const game = session.game;
646 const name = game.items.items[x].name;
647 if (!game.items.items[x].light) return session.fail("You can't {s} the {s}.", .{ if (lighting) "light" else "extinguish", name });
648 if (game.state.lit[x] == lighting) return session.fail("The {s} is {s}.", .{ name, if (lighting) "already lit" else "not lit" });
649 if (lighting and game.state.fuel[x] == 0) return session.fail("The {s} has no fuel.", .{name});
650 game.state.lit[x] = lighting;
651 try session.say("The {s} is now {s}.", .{ name, if (lighting) "lit" else "out" });
652 }
653 fn move(session: *Session, direction: Direction) Error!bool {
654 const game = session.game;
655 const exit = game.state.exits[game.state.room][@backingInt(direction)] orelse return session.fail("You can't go that way.", .{});
656 if (exit.locked) return session.fail("The way {s} is locked.", .{@tagName(direction)});
657 game.state.room = exit.destination;
658 const baseline = game.isLit();
659 try session.describe(game.verbose or !game.state.visited[game.state.room]);
660 game.state.visited[game.state.room] = true;
661 for (game.rooms.items[game.state.room].enter.items) |rule| _ = try session.runRule(rule);
662 return baseline;
663 }
664 fn turn(session: *Session, baseline: bool) Error!void {
665 const game = session.game;
666 game.state.moves += 1;
667 var reachable: [200]bool = @splat(false);
668 for (0..game.items.items.len) |x| reachable[x] = game.carried(x) or game.reachable(x, game.floor());
669 for (0..game.items.items.len) |x| if (game.state.lit[x]) {
670 game.state.fuel[x] -= 1;
671 if (game.state.fuel[x] == 0) {
672 game.state.lit[x] = false;
673 if (reachable[x]) try session.say("The {s} goes out.", .{game.items.items[x].name});
674 }
675 };
676 const lit = game.isLit();
677 if (baseline and !lit) try session.say("It is now pitch dark.", .{}) else if (!baseline and lit) try session.describe(true);
678 game.state.darkness = if (lit) 0 else game.state.darkness + 1;
679 if (game.state.darkness == 2) try session.say("You hear something moving in the dark.", .{});
680 if (game.state.darkness == 3) game.state.end = .dead;
681 const treasure = game.treasures();
682 if (game.state.end == .none and treasure.total > 0 and treasure.stored == treasure.total) game.state.end = .won;
683 if (game.state.end != .none) {
684 try session.say("{s}", .{if (game.state.end == .dead) "*** You have died ***" else "*** You have won ***"});
685 try session.say("You scored {d} points in {d} moves.", .{ treasure.points + game.state.bonus, game.state.moves });
686 }
687 }
688 fn run(session: *Session, verb: Verb, first: []const []const u8, second: ?[]const []const u8) Error!void {
689 const game = session.game;
690 var baseline = game.isLit();
691 var success = true;
692 switch (verb) {
693 .go => baseline = try session.move(directionOf(first[0]).?),
694 .look => try session.describe(true),
695 .examine => try session.examine(try session.resolve(first)),
696 .take => success = try session.take(first, second),
697 .drop => try session.drop(first),
698 .put => success = try session.put(first, second.?),
699 .open => try session.open(try session.resolve(first)),
700 .close => try session.close(try session.resolve(first)),
701 .unlock => try session.lock(first, second, false),
702 .lock => try session.lock(first, second, true),
703 .use => try session.use(first, second.?),
704 .light => try session.light(try session.resolve(first), true),
705 .extinguish => try session.light(try session.resolve(first), false),
706 .wait => try session.say("Time passes.", .{}),
707 .inventory => {
708 if (game.direct(player).len == 0) try session.say("You are empty-handed.", .{}) else {
709 try session.say("You are carrying:", .{});
710 try session.itemTree(player, 1);
711 try session.say("Total weight: {d}/{d}", .{ game.carriedWeight(), game.max_weight });
712 }
713 },
714 .score => {
715 const treasure = game.treasures();
716 try session.say("You have scored {d} points (treasure {d}, bonus {d}) in {d} moves.", .{
717 treasure.points + game.state.bonus, treasure.points, game.state.bonus, game.state.moves,
718 });
719 try session.say("Treasures stored: {d} of {d}.", .{ treasure.stored, treasure.total });
720 },
721 .save => {
722 game.saved = game.state;
723 try session.say("Game saved.", .{});
724 },
725 .restore => {
726 game.state = game.saved orelse return session.fail("There is no saved game.", .{});
727 try session.say("Game restored.", .{});
728 try session.describe(true);
729 },
730 .verbose => {
731 game.verbose = true;
732 try session.say("Verbose mode.", .{});
733 },
734 .brief => {
735 game.verbose = false;
736 try session.say("Brief mode.", .{});
737 },
738 }
739 const action = switch (verb) {
740 .go, .take, .drop, .put, .open, .close, .unlock, .lock, .use, .light, .extinguish, .wait => true,
741 else => false,
742 };
743 if (success and action) try session.turn(baseline);
744 }
745 fn interpret(session: *Session, words: []const []const u8) Error!void {
746 if (words.len == 0) return session.fail("I beg your pardon?", .{});
747 var rest = words[1..];
748 var verb: Verb = undefined;
749 if (directionOf(words[0]) != null) {
750 verb = .go;
751 rest = words;
752 } else verb = verbOf(words[0]) orelse return session.fail("I don't know the verb \"{s}\".", .{words[0]});
753 if (session.game.state.end != .none and verb != .restore) return session.fail("The game is over.", .{});
754 if (verb == .look and rest.len > 0) {
755 if (!equal(rest[0], "at")) return session.fail("I don't understand that.", .{});
756 verb = .examine;
757 rest = rest[1..];
758 }
759 const no_objects = switch (verb) {
760 .look, .inventory, .wait, .score, .save, .restore, .verbose, .brief => true,
761 else => false,
762 };
763 if (no_objects) {
764 if (rest.len != 0) return session.fail("I don't understand that.", .{});
765 return session.run(verb, rest, null);
766 }
767 if (rest.len == 0) {
768 if (verb == .go) return session.fail("Which way do you want to go?", .{});
769 return session.fail("What do you want to {s}?", .{@tagName(verb)});
770 }
771 if (verb == .go) {
772 if (rest.len != 1 or directionOf(rest[0]) == null) return session.fail("I don't understand that.", .{});
773 return session.run(verb, rest, null);
774 }
775 const prepositions: []const []const u8 = switch (verb) {
776 .put => &.{ "in", "into" },
777 .use => &.{ "on", "with" },
778 .lock, .unlock => if (contains(rest, "with")) &.{"with"} else &.{},
779 .take => if (contains(rest, "from")) &.{"from"} else &.{},
780 else => &.{},
781 };
782 if (prepositions.len == 0) return session.run(verb, rest, null);
783 var at: usize = 0;
784 while (at < rest.len and !contains(prepositions, rest[at])) : (at += 1) {}
785 if (at == 0 or at >= rest.len - 1) return session.fail("I don't understand that.", .{});
786 try session.run(verb, rest[0..at], rest[at + 1 ..]);
787 }
788 fn process(session: *Session, text: []const u8) Error!void {
789 try session.say("> {s}", .{text});
790 var words = try normalized(session.gpa, text);
791 if (words.len > 0 and (equal(words[0], "again") or equal(words[0], "g"))) {
792 if (words.len != 1) return session.fail("I don't understand that.", .{});
793 const previous = session.game.previous orelse return session.fail("There is nothing to repeat.", .{});
794 words = try normalized(session.gpa, previous);
795 } else session.game.previous = try session.game.gpa.dupe(u8, text);
796 try session.interpret(words);
797 }
798};

800fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
801 var buffer: Writer.Allocating = .init(gpa);
802 defer buffer.deinit();
803 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
804 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
805 const text = try buffer.toOwnedSlice();
806 return if (text.len > 0 and text[text.len - 1] == '\r') text[0 .. text.len - 1] else text;
807}
808pub fn main(init: std.process.Init) !void {
809 var storage = std.heap.ArenaAllocator.init(std.heap.page_allocator);
810 defer storage.deinit();
811 const gpa = storage.allocator();
812 var input_buffer: [1024]u8 = undefined;
813 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
814 var output_buffer: [4096]u8 = undefined;
815 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
816 var definitions: std.ArrayList([]const u8) = .empty;
817 while (true) {
818 const text = try readLine(gpa, &input.interface);
819 if (equal(text, "BEGIN")) break;
820 try definitions.append(gpa, text);
821 }
822 const game = try Adventure.create(gpa, definitions.items);
823 var initial: Session = .{ .game = game, .gpa = gpa, .out = &output.interface };
824 try initial.describe(true);
825 try output.interface.flush();
826 while (true) {
827 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
828 defer scratch.deinit();
829 const temporary = scratch.allocator();
830 const text = try readLine(temporary, &input.interface);
831 if (equal(text, "QUIT")) break;
832 var response: Writer.Allocating = .init(temporary);
833 defer response.deinit();
834 var session: Session = .{ .game = game, .gpa = temporary, .out = &response.writer };
835 session.process(text) catch |err| {
836 if (err != error.Invalid) return err;
837 try line(&response.writer, "{s}", .{session.problem});
838 };
839 try output.interface.writeAll(response.written());
840 try output.interface.flush();
841 }
842 const treasure = game.treasures();
843 try line(&output.interface, "Final score: {d} points in {d} moves.", .{ treasure.points + game.state.bonus, game.state.moves });
844 try output.interface.flush();
845}
