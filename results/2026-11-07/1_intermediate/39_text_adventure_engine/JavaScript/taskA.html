27class Adventure {
28 constructor(lines) {
29 this.rooms = new Map(); this.items = new Map(); this.rules = []; this.enterRules = [];
30 this.state = { room: null, items: new Map(), exits: new Map(), flags: new Set(), bonus: 0, moves: 0,
31 visited: new Set(), darkness: 0, ended: 'none' };
32 this.it = null; this.previous = null; this.verbose = false; this.saved = null;
33 for (const line of lines) this.define(line);
34 this.state.visited.add(this.state.room);
35 this.ids = [...this.items.keys()].sort();
36 this.vocabulary = new Set([...this.items.values()].flatMap(item => [...item.nouns, ...item.words]));

38 define(line) {
39 const words = line.split(' '), [kind, id] = words;
40 switch (kind) {
41 case 'ROOM': {
42 const dark = words[2] === 'DARK';
43 this.rooms.set(id, { dark, name: words.slice(dark ? 3 : 2).join(' '), description: [] });
44 break;
45 }
46 case 'DESC': this.rooms.get(id).description.push(words.slice(2).join(' ')); break;
47 case 'EXIT': this.setExit(id, words[2], words[3], words[4] === 'LOCKED' ? words[5] : null); break;
48 case 'ITEM': {
49 const item = { id, weight: Number(words[3]), points: 0, light: false, capacity: null, key: null,
50 fixed: false, nouns: new Set([id]), details: [] };
51 const state = { place: words[2] === 'PLAYER' ? player() : words[2] === 'NOWHERE' ? nowhere() : floor(words[2]),
52 container: 'open', lit: false, fuel: 0 };
53 let index = 4;
54 if (words[index] === 'TREASURE') { item.points = Number(words[index + 1]); index += 2; }
55 if (words[index] === 'LIGHT') { item.light = true; state.fuel = Number(words[index + 1]); index += 2; }
56 if (words[index] === 'CONTAINER') {
57 item.capacity = Number(words[index + 1]); index += 2;
58 if (words[index] === 'CLOSED') { state.container = 'closed'; ++index; }
59 if (words[index] === 'LOCKED') { item.key = words[index + 1]; state.container = 'locked'; index += 2; }
60 }
61 if (words[index] === 'FIXED') { item.fixed = true; ++index; }
62 item.name = words.slice(index).join(' '); item.words = new Set(words.slice(index));
63 this.items.set(id, item); this.state.items.set(id, state);
64 break;
65 }
66 case 'INSIDE': this.state.items.get(id).place = inside(words[2]); break;
67 case 'ALIAS': this.items.get(id).nouns.add(words[2]); break;
68 case 'DETAIL': this.items.get(id).details.push(words.slice(2).join(' ')); break;
69 case 'RULE': case 'ENTER': {
70 let index = kind === 'RULE' ? 3 : 2, condition = null;
71 if (words[index] === 'IF' || words[index] === 'UNLESS') {
72 condition = { flag: words[index + 1], on: words[index] === 'IF' }; index += 2;
73 }
74 const rule = { first: id, second: kind === 'RULE' ? words[2] : null, condition, effect: words.slice(index) };
75 (kind === 'RULE' ? this.rules : this.enterRules).push(rule);
76 break;
77 }
78 case 'TROPHY': this.trophy = id; break;
79 case 'START': this.state.room = id; break;
80 case 'MAXWEIGHT': this.maxWeight = Number(id); break;
81 }
82 }
83 setExit(room, direction, destination, key = null) {
84 if (!this.state.exits.has(room)) this.state.exits.set(room, new Map());
85 this.state.exits.get(room).set(direction, { destination, key, locked: key !== null });
86 }
87 contents(place) { return this.ids.filter(id => samePlace(this.state.items.get(id).place, place)); }
88 children() {
89 const result = new Map();
90 for (const id of this.ids) {
91 const key = placeKey(this.state.items.get(id).place);
92 if (!result.has(key)) result.set(key, []);
93 result.get(key).push(id);
94 }
95 return result;

97 isOpen(id) { return this.items.get(id).capacity !== null && this.state.items.get(id).container === 'open'; }
98 reachable(place) {
99 const result = new Set(), children = this.children();
100 const visit = place => {
101 for (const id of children.get(placeKey(place)) ?? []) {
102 result.add(id);
103 if (this.isOpen(id)) visit(inside(id));
104 }
105 };
106 visit(place);
107 return result;
108 }
109 carried() { return this.reachable(player()); }
110 roomItems() { return this.reachable(floor(this.state.room)); }
111 isLit() {
112 return !this.rooms.get(this.state.room).dark || [...this.carried(), ...this.roomItems()].some(id => this.state.items.get(id).lit);
113 }
114 visible() { return this.isLit() ? new Set([...this.carried(), ...this.roomItems()]) : this.carried(); }
115 weights() {
116 const result = new Map(), children = this.children();
117 const total = id => {
118 if (!result.has(id)) result.set(id, this.items.get(id).weight +
119 (children.get(placeKey(inside(id))) ?? []).reduce((weight, child) => weight + total(child), 0));
120 return result.get(id);
121 };
122 for (const id of this.ids) total(id);
123 return result;
124 }
125 totalWeight(id) { return this.weights().get(id); }
126 contentWeight(id) { return this.totalWeight(id) - this.items.get(id).weight; }
127 carriedWeight() { const weights = this.weights(); return this.contents(player()).reduce((weight, id) => weight + weights.get(id), 0); }
128 nestedIn(id, ancestor) {
129 let place = this.state.items.get(id).place;
130 while (place.kind === 'inside') {
131 if (place.id === ancestor) return true;
132 place = this.state.items.get(place.id).place;
133 }
134 return false;
135 }
136 score() {
137 let stored = 0, treasures = 0, points = 0;
138 for (const id of this.ids) if (this.items.get(id).points > 0) {
139 ++treasures;
140 let place = this.state.items.get(id).place;
141 while (place.kind === 'inside') place = this.state.items.get(place.id).place;
142 if (samePlace(place, floor(this.trophy))) { ++stored; points += this.items.get(id).points; }
143 }
144 return { stored, treasures, points, total: points + this.state.bonus };
145 }
146 display(id) {
147 const item = this.items.get(id), state = this.state.items.get(id), annotations = [];
148 if (item.capacity !== null) annotations.push(state.container);
149 if (item.light && state.lit) annotations.push('lit');
150 return item.name + (annotations.length ? ` (${annotations.join(', ')})` : '');
151 }
152 tree(place, depth = 1, children = this.children()) {
153 const rows = [];
154 for (const id of children.get(placeKey(place)) ?? []) {
155 rows.push(' '.repeat(depth) + this.display(id));
156 if (this.isOpen(id)) rows.push(...this.tree(inside(id), depth + 1, children));
157 }
158 return rows;
159 }
160 describe(full = true) {
161 if (!this.isLit()) return ['It is pitch dark.'];
162 const room = this.rooms.get(this.state.room), exits = this.state.exits.get(this.state.room), names = [];
163 for (const direction of directions) if (exits?.has(direction)) names.push(direction + (exits.get(direction).locked ? ' (locked)' : ''));
164 const rows = [room.name, ...(full ? room.description : []), 'Exits: ' + (names.length ? names.join(', ') : 'none')];
165 if (this.contents(floor(this.state.room)).length) rows.push('You see:', ...this.tree(floor(this.state.room)));
166 return rows;

168 resolve(phrase) {
169 const visible = this.visible();
170 if (phrase.length === 1 && phrase[0] === 'it') {
171 if (this.it === null) fail('I don\'t know what "it" refers to.');
172 if (!visible.has(this.it)) fail("You can't see any such thing.");
173 return this.it;
174 }
175 for (const word of phrase) if (!this.vocabulary.has(word)) fail(`I don't know the word "${word}".`);
176 const matches = this.ids.filter(id => visible.has(id) && this.items.get(id).nouns.has(phrase.at(-1)) &&
177 phrase.slice(0, -1).every(word => this.items.get(id).words.has(word)));
178 if (!matches.length) fail("You can't see any such thing.");
179 if (matches.length > 1) fail(`Which do you mean: ${matches.map(id => this.items.get(id).name).join(', ')}?`);
180 this.it = matches[0];
181 return matches[0];
182 }
183 runRules(rules) {
184 const rows = [];
185 let executed = 0;
186 for (const rule of rules) {
187 if (rule.condition && this.state.flags.has(rule.condition.flag) !== rule.condition.on) continue;
188 ++executed;
189 const [effect, first, second, third] = rule.effect;
190 switch (effect) {
191 case 'MESSAGE': rows.push(rule.effect.slice(1).join(' ')); break;
192 case 'OPEN': this.setExit(first, second, third); break;
193 case 'REMOVE': this.state.items.get(first).place = nowhere(); break;
194 case 'ADD': this.state.items.get(first).place = floor(second); break;
195 case 'SET': this.state.flags.add(first); break;
196 case 'CLEAR': this.state.flags.delete(first); break;
197 case 'POINTS': this.state.bonus += Number(first); break;
198 }
199 }
200 return { rows, executed };
201 }
202 requireContainer(id) {
203 const name = this.items.get(id).name;
204 if (this.items.get(id).capacity === null) fail(`The ${name} is not a container.`);
205 if (!this.isOpen(id)) fail(`The ${name} is closed.`);
206 }
207 take(first, second) {
208 if (!this.isLit()) fail('It is pitch dark.');
209 let container = null, source = floor(this.state.room);
210 if (second !== null) { container = this.resolve(second); this.requireContainer(container); source = inside(container); }
211 if (first.length === 1 && first[0] === 'all') {
212 const candidates = this.contents(source).filter(id => !this.items.get(id).fixed), rows = [];
213 if (!candidates.length) fail(container === null ? 'There is nothing here to take.' : `There is nothing in the ${this.items.get(container).name} to take.`);
214 let taken = false;
215 for (const id of candidates) {
216 let outcome = "You can't carry any more.";
217 if (this.carried().has(id) || this.carriedWeight() + this.totalWeight(id) <= this.maxWeight) {
218 this.state.items.get(id).place = player(); outcome = 'Taken.'; taken = true;
219 }
220 rows.push(`${this.items.get(id).name}: ${outcome}`);
221 }
222 if (!taken) fail(rows);
223 return rows;
224 }
225 const id = this.resolve(first), item = this.items.get(id), state = this.state.items.get(id);
226 if (container !== null && !samePlace(state.place, source)) fail(`The ${item.name} is not in the ${this.items.get(container).name}.`);
227 if (state.place.kind === 'player') fail(`You already have the ${item.name}.`);
228 if (item.fixed) fail(`The ${item.name} is fixed in place.`);
229 if (!this.carried().has(id) && this.carriedWeight() + this.totalWeight(id) > this.maxWeight) fail("You can't carry any more.");
230 state.place = player();
231 return ['Taken.'];
232 }
233 drop(first) {
234 if (first.length === 1 && first[0] === 'all') {
235 const candidates = this.contents(player());
236 if (!candidates.length) fail('You are empty-handed.');
237 return candidates.map(id => { this.state.items.get(id).place = floor(this.state.room); return `${this.items.get(id).name}: Dropped.`; });
238 }
239 const id = this.resolve(first);
240 if (!this.carried().has(id)) fail(`You aren't carrying the ${this.items.get(id).name}.`);
241 this.state.items.get(id).place = floor(this.state.room);
242 return ['Dropped.'];
243 }
244 putItem(id, container) {
245 const item = this.items.get(id), target = this.items.get(container), state = this.state.items.get(id);
246 if (this.nestedIn(container, id)) return `The ${target.name} is inside the ${item.name}.`;
247 if (samePlace(state.place, inside(container))) return `The ${item.name} is already in the ${target.name}.`;
248 const weight = this.contentWeight(container) + (this.nestedIn(id, container) ? 0 : this.totalWeight(id));
249 if (weight > target.capacity) return `The ${item.name} doesn't fit in the ${target.name}.`;
250 state.place = inside(container);
251 return null;
252 }
253 put(first, second) {
254 if (first.length === 1 && first[0] === 'all') {
255 const target = this.resolve(second); this.requireContainer(target);
256 const candidates = this.contents(player()).filter(id => id !== target), rows = [];
257 if (!candidates.length) fail(`You have nothing to put in the ${this.items.get(target).name}.`);
258 let succeeded = false;
259 for (const id of candidates) {
260 const reason = this.putItem(id, target);
261 if (reason === null) succeeded = true;
262 rows.push(`${this.items.get(id).name}: ${reason ?? 'Done.'}`);
263 }
264 if (!succeeded) fail(rows);
265 return rows;
266 }
267 const id = this.resolve(first), target = this.resolve(second), item = this.items.get(id);
268 if (!this.carried().has(id)) fail(`You aren't carrying the ${item.name}.`);
269 if (id === target) fail(`You can't put the ${item.name} inside itself.`);
270 this.requireContainer(target);
271 const reason = this.putItem(id, target);
272 if (reason !== null) fail(reason);
273 return [`You put the ${item.name} in the ${this.items.get(target).name}.`];
274 }
275 keyCheck(key, phrase) {
276 if (phrase !== null) {
277 const id = this.resolve(phrase), name = this.items.get(id).name;
278 if (!this.carried().has(id)) fail(`You aren't carrying the ${name}.`);
279 if (id !== key) fail(`The ${name} doesn't fit.`);
280 return [];
281 }
282 if (!this.carried().has(key)) fail("You don't have the right key.");
283 return [`(with the ${this.items.get(key).name})`];
284 }
285 lock(first, second, locking) {
286 const direction = first.length === 1 ? directionWords.get(first[0]) : undefined;
287 if (direction !== undefined) {
288 const exit = this.state.exits.get(this.state.room)?.get(direction);
289 if (exit === undefined) fail(`There is no exit ${direction}.`);
290 if (exit.key === null) fail(`The way ${direction} has no lock.`);
291 if (exit.locked === locking) fail(`The way ${direction} is ${locking ? 'already locked' : 'not locked'}.`);
292 const rows = this.keyCheck(exit.key, second);
293 exit.locked = locking;
294 const opposite = directions[directions.indexOf(direction) ^ 1], back = this.state.exits.get(exit.destination)?.get(opposite);
295 if (back?.destination === this.state.room && back.key === exit.key) back.locked = locking;
296 return [...rows, locking ? 'Locked.' : 'Unlocked.'];
297 }
298 const id = this.resolve(first), item = this.items.get(id), state = this.state.items.get(id);
299 if (item.capacity === null || item.key === null) fail(`The ${item.name} has no lock.`);
300 if ((state.container === 'locked') === locking) fail(`The ${item.name} is ${locking ? 'already locked' : 'not locked'}.`);
301 if (locking && state.container === 'open') fail(`You must close the ${item.name} first.`);
302 const rows = this.keyCheck(item.key, second);
303 state.container = locking ? 'locked' : 'closed';
304 return [...rows, locking ? 'Locked.' : 'Unlocked.'];
305 }
306 execute(verb, first, second) {
307 switch (verb) {
308 case 'look': return this.describe();
309 case 'inventory':
310 return this.contents(player()).length ? ['You are carrying:', ...this.tree(player()), `Total weight: ${this.carriedWeight()}/${this.maxWeight}`] : ['You are empty-handed.'];
311 case 'wait': return ['Time passes.'];
312 case 'score': {
313 const score = this.score();
314 return [`You have scored ${score.total} points (treasure ${score.points}, bonus ${this.state.bonus}) in ${this.state.moves} moves.`,
315 `Treasures stored: ${score.stored} of ${score.treasures}.`];
316 }
317 case 'save': this.saved = structuredClone(this.state); return ['Game saved.'];
318 case 'restore':
319 if (this.saved === null) fail('There is no saved game.');
320 this.state = structuredClone(this.saved);
321 return ['Game restored.', ...this.describe()];
322 case 'verbose': this.verbose = true; return ['Verbose mode.'];
323 case 'brief': this.verbose = false; return ['Brief mode.'];
324 case 'take': return this.take(first, second);
325 case 'drop': return this.drop(first);
326 case 'put': return this.put(first, second);
327 case 'unlock': case 'lock': return this.lock(first, second, verb === 'lock');
328 case 'use': {
329 const id = this.resolve(first), target = this.resolve(second);
330 if (!this.carried().has(id)) fail(`You aren't carrying the ${this.items.get(id).name}.`);
331 const result = this.runRules(this.rules.filter(rule => rule.first === id && rule.second === target));
332 if (!result.executed) fail('Nothing happens.');
333 return result.rows.length ? result.rows : ['Done.'];
334 }
335 }
336 const id = this.resolve(first), item = this.items.get(id), state = this.state.items.get(id);
337 switch (verb) {
338 case 'examine': {
339 const rows = item.details.length ? item.details.slice() : [`You see nothing special about the ${item.name}.`];
340 if (item.light) rows.push(`It is ${state.lit ? 'lit' : 'unlit'} (fuel ${state.fuel}).`);
341 if (item.capacity !== null) {
342 if (state.container !== 'open') rows.push(`It is ${state.container}.`);
343 else if (this.contents(inside(id)).length) rows.push('It contains:', ...this.tree(inside(id)));
344 else rows.push('It is empty.');
345 }
346 if (item.points > 0) rows.push(`It is a treasure worth ${item.points} points.`);
347 rows.push(`It weighs ${this.totalWeight(id)}.`);
348 return rows;
349 }
350 case 'open': {
351 if (item.capacity === null) fail(`You can't open the ${item.name}.`);
352 if (state.container === 'open') fail(`The ${item.name} is already open.`);
353 if (state.container === 'locked') fail(`The ${item.name} is locked.`);
354 state.container = 'open';
355 const contents = this.contents(inside(id));
356 return contents.length ? [`Opening the ${item.name} reveals ${contents.map(child => this.items.get(child).name).join(', ')}.`] : ['Opened.'];
357 }
358 case 'close':
359 if (item.capacity === null) fail(`You can't close the ${item.name}.`);
360 if (state.container !== 'open') fail(`The ${item.name} is already closed.`);
361 state.container = 'closed';
362 return ['Closed.'];
363 case 'light':
364 if (!item.light) fail(`You can't light the ${item.name}.`);
365 if (state.lit) fail(`The ${item.name} is already lit.`);
366 if (state.fuel === 0) fail(`The ${item.name} has no fuel.`);
367 state.lit = true;
368 return [`The ${item.name} is now lit.`];
369 case 'extinguish':
370 if (!item.light) fail(`You can't extinguish the ${item.name}.`);
371 if (!state.lit) fail(`The ${item.name} is not lit.`);
372 state.lit = false;
373 return [`The ${item.name} is now out.`];
374 }
375 }
376 endTurn(baseline) {
377 const rows = [], nearby = new Set([...this.carried(), ...this.roomItems()]);
378 ++this.state.moves;
379 for (const id of this.ids) {
380 const state = this.state.items.get(id);
381 if (state.lit && --state.fuel === 0) {
382 state.lit = false;
383 if (nearby.has(id)) rows.push(`The ${this.items.get(id).name} goes out.`);
384 }
385 }
386 const lit = this.isLit();
387 if (baseline && !lit) rows.push('It is now pitch dark.');
388 else if (!baseline && lit) rows.push(...this.describe());
389 this.state.darkness = lit ? 0 : this.state.darkness + 1;
390 if (this.state.darkness === 2) rows.push('You hear something moving in the dark.');
391 if (this.state.darkness === 3) this.state.ended = 'dead';
392 const score = this.score();
393 if (this.state.ended === 'none' && score.treasures > 0 && score.stored === score.treasures) this.state.ended = 'won';
394 if (this.state.ended !== 'none') {
395 rows.push(this.state.ended === 'dead' ? '*** You have died ***' : '*** You have won ***');
396 rows.push(`You scored ${score.total} points in ${this.state.moves} moves.`);
397 }
398 return rows;
399 }
400 interpret(line) {
401 let words = normalize(line);
402 if (!words.length) fail('I beg your pardon?');
403 if (directionWords.has(words[0])) words = ['go', ...words];
404 let verb = verbs.get(words[0]), rest = words.slice(1);
405 if (verb === undefined) fail(`I don't know the verb "${words[0]}".`);
406 if (this.state.ended !== 'none' && verb !== 'restore') fail('The game is over.');
407 if (verb === 'look' && rest.length) {
408 if (rest[0] !== 'at') fail("I don't understand that.");
409 verb = 'examine'; rest = rest.slice(1);
410 }
411 let first = rest, second = null;
412 if (noObjects.has(verb)) {
413 if (rest.length) fail("I don't understand that.");
414 } else {
415 if (!rest.length) fail(verb === 'go' ? 'Which way do you want to go?' : `What do you want to ${verb}?`);
416 if (verb === 'go') {
417 if (rest.length !== 1 || !directionWords.has(rest[0])) fail("I don't understand that.");
418 } else {
419 let prepositions = [];
420 if (verb === 'put') prepositions = ['in', 'into'];
421 else if (verb === 'use') prepositions = ['on', 'with'];
422 else if ((verb === 'unlock' || verb === 'lock') && rest.includes('with')) prepositions = ['with'];
423 else if (verb === 'take' && rest.includes('from')) prepositions = ['from'];
424 if (prepositions.length) {
425 const index = rest.findIndex(word => prepositions.includes(word));
426 if (index <= 0 || index === rest.length - 1) fail("I don't understand that.");
427 first = rest.slice(0, index); second = rest.slice(index + 1);
428 }
429 }
430 }
431 let baseline = this.isLit(), rows;
432 if (verb === 'go') {
433 const direction = directionWords.get(rest[0]), exit = this.state.exits.get(this.state.room)?.get(direction);
434 if (exit === undefined) fail("You can't go that way.");
435 if (exit.locked) fail(`The way ${direction} is locked.`);
436 this.state.room = exit.destination;
437 baseline = this.isLit();
438 rows = this.describe(this.verbose || !this.state.visited.has(this.state.room));
439 this.state.visited.add(this.state.room);
440 rows.push(...this.runRules(this.enterRules.filter(rule => rule.first === this.state.room)).rows);
441 } else rows = this.execute(verb, first, second);
442 if (actions.has(verb)) rows.push(...this.endTurn(baseline));
443 return rows;
444 }
445 command(line) {
446 try {
447 const words = normalize(line);
448 if (words[0] === 'again' || words[0] === 'g') {
449 if (words.length !== 1) fail("I don't understand that.");
450 if (this.previous === null) fail('There is nothing to repeat.');
451 return this.interpret(this.previous);
452 }
453 this.previous = line;
454 return this.interpret(line);
455 } catch (error) {
456 if (!(error instanceof Failure)) throw error;
457 return error.lines;
458 }
459 }
