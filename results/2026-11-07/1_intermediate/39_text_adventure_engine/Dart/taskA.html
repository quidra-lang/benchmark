194 Adventure(List<List<String>> definitions) {
195 rooms = [
196 for (final row in definitions)
197 if (row[0] == 'ROOM') Room(row[1]),
198 ];
199 items = [
200 for (final row in definitions)
201 if (row[0] == 'ITEM') Item(row[1]),
202 ]..sort((a, b) => a.id.compareTo(b.id));
203 final roomIndex = {for (final (i, room) in rooms.indexed) room.id: i};
204 final itemIndex = {for (final (i, item) in items.indexed) item.id: i};
205 state = GameState(rooms.length, items.length);
206 Place initialPlace(String text) => switch (text) {
207 'PLAYER' => player,
208 'NOWHERE' => (PlaceKind.nowhere, 0),
209 _ => (PlaceKind.room, roomIndex[text]!),
210 };
211 Rule parseRule(List<String> row, int index) {
212 String? flag;
213 var whenOn = true;
214 if (row[index] == 'IF' || row[index] == 'UNLESS') {
215 whenOn = row[index] == 'IF';
216 flag = row[index + 1];
217 index += 2;
218 }
219 final Effect effect = switch (row[index]) {
220 'MESSAGE' => MessageEffect(row.skip(index + 1).join(' ')),
221 'OPEN' => OpenEffect(
222 roomIndex[row[index + 1]]!,
223 directionOf(row[index + 2])!,
224 roomIndex[row[index + 3]]!,
225 ),
226 'REMOVE' => MoveEffect(itemIndex[row[index + 1]]!, (
227 PlaceKind.nowhere,
228 0,
229 )),
230 'ADD' => MoveEffect(itemIndex[row[index + 1]]!, (
231 PlaceKind.room,
232 roomIndex[row[index + 2]]!,
233 )),
234 'SET' => FlagEffect(row[index + 1], true),
235 'CLEAR' => FlagEffect(row[index + 1], false),
236 'POINTS' => PointsEffect(int.parse(row[index + 1])),
237 _ => throw StateError('Invalid definition'),
238 };
239 return Rule(flag, whenOn, effect);
240 }
241
242 for (final row in definitions) {
243 switch (row[0]) {
244 case 'ROOM':
245 final room = rooms[roomIndex[row[1]]!];
246 room.dark = row[2] == 'DARK';
247 room.name = row.skip(room.dark ? 3 : 2).join(' ');
248 case 'DESC':
249 rooms[roomIndex[row[1]]!].descriptions.add(row.skip(2).join(' '));
250 case 'EXIT':
251 state.exits[roomIndex[row[1]]!][directionOf(row[2])!] = (
252 destination: roomIndex[row[3]]!,
253 key: row.length > 4 ? itemIndex[row[5]]! : null,
254 locked: row.length > 4,
255 );
256 case 'ITEM':
257 final x = itemIndex[row[1]]!, item = items[itemIndex[row[1]]!];
258 var index = 4;
259 state.places[x] = initialPlace(row[2]);
260 item.weight = int.parse(row[3]);
261 if (row[index] == 'TREASURE') {
262 item.treasure = int.parse(row[index + 1]);
263 index += 2;
264 }
265 if (row[index] == 'LIGHT') {
266 item.light = true;
267 state.fuel[x] = int.parse(row[index + 1]);
268 index += 2;
269 }
270 if (row[index] == 'CONTAINER') {
271 item.capacity = int.parse(row[index + 1]);
272 index += 2;
273 if (row[index] == 'CLOSED') {
274 state.containers[x] = ContainerState.closed;
275 index++;
276 }
277 if (row[index] == 'LOCKED') {
278 item.key = itemIndex[row[index + 1]]!;
279 state.containers[x] = ContainerState.locked;
280 index += 2;
281 }
282 }
283 if (row[index] == 'FIXED') {
284 item.fixed = true;
285 index++;
286 }
287 item.name = row.skip(index).join(' ');
288 item.nameWords.addAll(row.skip(index));
289 case 'INSIDE':
290 state.places[itemIndex[row[1]]!] = (
291 PlaceKind.inside,
292 itemIndex[row[2]]!,
293 );
294 case 'ALIAS':
295 items[itemIndex[row[1]]!].nouns.add(row[2]);
296 case 'DETAIL':
297 items[itemIndex[row[1]]!].details.add(row.skip(2).join(' '));
298 case 'RULE':
299 useRules.add((
300 first: itemIndex[row[1]]!,
301 second: itemIndex[row[2]]!,
302 rule: parseRule(row, 3),
303 ));
304 case 'ENTER':
305 rooms[roomIndex[row[1]]!].enter.add(parseRule(row, 2));
306 case 'TROPHY':
307 trophy = roomIndex[row[1]]!;
308 case 'START':
309 state.room = roomIndex[row[1]]!;
310 case 'MAXWEIGHT':
311 maxWeight = int.parse(row[1]);
312 }
313 }
314 state.visited[state.room] = true;
315 for (final item in items) {
316 vocabulary.addAll(item.nouns);
317 vocabulary.addAll(item.nameWords);
318 }
319 }

321 void say(String text) => output.add(text);
322 Place get floor => (PlaceKind.room, state.room);
323 List<int> direct(Place place) => [
324 for (var x = 0; x < items.length; x++)
325 if (state.places[x] == place) x,
326 ];
327 Set<int> reachable(Place place) {
328 final result = <int>{};
329 void visit(Place source) {
330 for (final x in direct(source)) {
331 result.add(x);
332 if (items[x].capacity != null &&
333 state.containers[x] == ContainerState.open)
334 visit((PlaceKind.inside, x));

336 }

342 Set<int> carried() => reachable(player);
343 bool isLit() =>
344 !rooms[state.room].dark ||
345 {...carried(), ...reachable(floor)}.any((x) => state.lit[x]);
346 Set<int> visible() => carried()..addAll(isLit() ? reachable(floor) : <int>{});
347 int totalWeight(int x) => items[x].weight + contentWeight(x);
348 int contentWeight(int x) =>
349 direct((PlaceKind.inside, x))
350 .fold(0, (sum, item) => sum + totalWeight(item));
351 int carriedWeight() =>
352 direct(player).fold(0, (sum, item) => sum + totalWeight(item));
353 bool within(int x, int container) {
354 while (state.places[x].$1 == PlaceKind.inside) {
355 x = state.places[x].$2;
356 if (x == container) return true;
357 }
358 return false;
359 }

361 bool stored(int x) {
362 while (state.places[x].$1 == PlaceKind.inside) {
363 x = state.places[x].$2;

365 return state.places[x] == (PlaceKind.room, trophy);

422 int resolve(List<String> phrase) {
423 final seen = visible();
424 if (phrase.length == 1 && phrase[0] == 'it') {
425 final item = pronoun;
426 if (item == null) fail('I don\'t know what "it" refers to.');
427 if (!seen.contains(item)) fail('You can\'t see any such thing.');
428 return item;
429 }
430 for (final word in phrase) {
431 if (!vocabulary.contains(word)) fail('I don\'t know the word "$word".');
432 }
433 final matches = [
434 for (var x = 0; x < items.length; x++)
435 if (seen.contains(x) &&
436 items[x].nouns.contains(phrase.last) &&
437 phrase.take(phrase.length - 1).every(items[x].nameWords.contains))
438 x,
439 ];
440 if (matches.isEmpty) fail('You can\'t see any such thing.');
441 if (matches.length > 1)
442 fail(
443 'Which do you mean: ${matches.map((x) => items[x].name).join(', ')}?',
444 );
445 pronoun = matches.single;
446 return matches.single;
447 }

450 final item = items[x];
451 if (item.details.isEmpty) {
452 say('You see nothing special about the ${item.name}.');
453 } else {
454 output.addAll(item.details);
455 }

458 if (item.capacity != null) {
459 if (state.containers[x] == ContainerState.locked) {

461 } else if (state.containers[x] == ContainerState.closed) {
462 say('It is closed.');
463 } else if (direct((PlaceKind.inside, x)).isEmpty) {
464 say('It is empty.');
465 } else {
466 say('It contains:');
467 itemTree((PlaceKind.inside, x), 1);
468 }
469 }
470 if (item.treasure != null)
471 say('It is a treasure worth ${item.treasure} points.');
472 say('It weighs ${totalWeight(x)}.');
473 }

482 bool take(List<String> target, List<String>? from) {
483 if (!isLit()) fail('It is pitch dark.');
484 int? container;
485 var source = floor;
486 if (from != null) {
487 container = resolve(from);
488 requireContainer(container);
489 source = (PlaceKind.inside, container);
490 }
491 if (all(target)) {
492 final candidates = direct(source).where((x) => !items[x].fixed).toList();
493 if (candidates.isEmpty)
494 fail(
495 container == null
496 ? 'There is nothing here to take.'
497 : 'There is nothing in the ${items[container].name} to take.',
498 );
499 var taken = false;
500 for (final x in candidates) {
501 if (!carried().contains(x) &&
502 carriedWeight() + totalWeight(x) > maxWeight) {
503 say('${items[x].name}: You can\'t carry any more.');
504 } else {
505 state.places[x] = player;
506 taken = true;
507 say('${items[x].name}: Taken.');
508 }
509 }
510 return taken;
511 }
512 final x = resolve(target);
513 if (container != null && state.places[x] != source)
514 fail('The ${items[x].name} is not in the ${items[container].name}.');
515 if (state.places[x] == player)
516 fail('You already have the ${items[x].name}.');
517 if (items[x].fixed) fail('The ${items[x].name} is fixed in place.');
518 if (!carried().contains(x) && carriedWeight() + totalWeight(x) > maxWeight)
519 fail('You can\'t carry any more.');
520 state.places[x] = player;
521 say('Taken.');
522 return true;
523 }

542 String? putItem(int x, int y) {
543 if (within(y, x))
544 return 'The ${items[y].name} is inside the ${items[x].name}.';
545 if (state.places[x] == (PlaceKind.inside, y))
546 return 'The ${items[x].name} is already in the ${items[y].name}.';
547 final weight = contentWeight(y) + (within(x, y) ? 0 : totalWeight(x));
548 if (weight > items[y].capacity!)
549 return 'The ${items[x].name} doesn\'t fit in the ${items[y].name}.';
550 state.places[x] = (PlaceKind.inside, y);
551 return null;
552 }

554 bool put(List<String> first, List<String> second) {
555 if (all(first)) {
556 final y = resolve(second);
557 requireContainer(y);
558 final candidates = direct(player).where((x) => x != y).toList();
559 if (candidates.isEmpty)
560 fail('You have nothing to put in the ${items[y].name}.');
561 var moved = false;
562 for (final x in candidates) {
563 final reason = putItem(x, y);
564 if (reason == null) moved = true;
565 say('${items[x].name}: ${reason ?? 'Done.'}');
566 }
567 return moved;
568 }
569 final x = resolve(first), y = resolve(second);
570 if (!carried().contains(x))
571 fail('You aren\'t carrying the ${items[x].name}.');
572 if (x == y) fail('You can\'t put the ${items[x].name} inside itself.');
573 requireContainer(y);
574 final reason = putItem(x, y);
575 if (reason != null) fail(reason);
576 say('You put the ${items[x].name} in the ${items[y].name}.');
577 return true;
578 }

617 void lock(List<String> target, List<String>? keyPhrase, bool locking) {
618 final d = target.length == 1 ? directionOf(target[0]) : null;
619 if (d != null) {
620 final exit = state.exits[state.room][d];
621 if (exit == null) fail('There is no exit ${directions[d]}.');
622 if (exit.key == null) fail('The way ${directions[d]} has no lock.');
623 if (exit.locked == locking)
624 fail(
625 'The way ${directions[d]} is ${locking ? 'already locked' : 'not locked'}.',
626 );
627 checkKey(exit.key!, keyPhrase);
628 state.exits[state.room][d] = (
629 destination: exit.destination,
630 key: exit.key,
631 locked: locking,
632 );
633 final opposite = state.exits[exit.destination][d ^ 1];
634 if (opposite != null &&
635 opposite.destination == state.room &&
636 opposite.key == exit.key) {
637 state.exits[exit.destination][d ^ 1] = (
638 destination: opposite.destination,
639 key: opposite.key,
640 locked: locking,
641 );
642 }
643 } else {
644 final x = resolve(target), item = items[x];
645 if (item.capacity == null || item.key == null)
646 fail('The ${item.name} has no lock.');
647 if ((state.containers[x] == ContainerState.locked) == locking)
648 fail(
649 'The ${item.name} is ${locking ? 'already locked' : 'not locked'}.',
650 );
651 if (locking && state.containers[x] == ContainerState.open)
652 fail('You must close the ${item.name} first.');
653 checkKey(item.key!, keyPhrase);
654 state.containers[x] = locking
655 ? ContainerState.locked
656 : ContainerState.closed;
657 }
658 say(locking ? 'Locked.' : 'Unlocked.');
659 }

661 ({int executed, bool message}) runRules(Iterable<Rule> rules) {
662 var executed = 0, message = false;
663 for (final rule in rules) {
664 if (rule.flag != null && state.flags.contains(rule.flag) != rule.whenOn)
665 continue;
666 executed++;
667 switch (rule.effect) {
668 case MessageEffect(:final text):
669 say(text);
670 message = true;
671 case OpenEffect(:final room, :final direction, :final destination):
672 state.exits[room][direction] = (
673 destination: destination,
674 key: null,
675 locked: false,
676 );
677 case MoveEffect(:final item, :final place):
678 state.places[item] = place;
679 case FlagEffect(:final flag, :final on):
680 if (on) {
681 state.flags.add(flag);
682 } else {
683 state.flags.remove(flag);
684 }
685 case PointsEffect(:final amount):
686 state.bonus += amount;
687 }
688 }
689 return (executed: executed, message: message);
690 }

692 void use(List<String> first, List<String> second) {
693 final x = resolve(first), y = resolve(second);
694 if (!carried().contains(x))
695 fail('You aren\'t carrying the ${items[x].name}.');
696 final result = runRules(
697 useRules
698 .where((rule) => rule.first == x && rule.second == y)
699 .map((rule) => rule.rule),
700 );
701 if (result.executed == 0) fail('Nothing happens.');
702 if (!result.message) say('Done.');
703 }

727 void turn(bool baseline) {
728 state.moves++;
729 final reachable = carried()..addAll(this.reachable(floor));
730 for (var x = 0; x < items.length; x++) {
731 if (!state.lit[x]) continue;
732 state.fuel[x]--;
733 if (state.fuel[x] == 0) {
734 state.lit[x] = false;
735 if (reachable.contains(x)) say('The ${items[x].name} goes out.');
736 }
737 }
738 final lit = isLit();
739 if (baseline && !lit) {
740 say('It is now pitch dark.');
741 } else if (!baseline && lit) {
742 describe(true);
743 }
744 state.darkness = lit ? 0 : state.darkness + 1;
745 if (state.darkness == 2) say('You hear something moving in the dark.');
746 if (state.darkness == 3) state.end = EndStatus.dead;
747 final treasure = treasures();
748 if (state.end == EndStatus.none &&
749 treasure.total > 0 &&
750 treasure.stored == treasure.total)
751 state.end = EndStatus.won;
752 if (state.end != EndStatus.none) {
753 say(
754 state.end == EndStatus.dead
755 ? '*** You have died ***'
756 : '*** You have won ***',
757 );
758 say(
759 'You scored ${treasure.points + state.bonus} points in ${state.moves} moves.',
760 );
761 }
762 }

764 void run(
765 Verb verb,
766 List<String> first, {
767 List<String>? second,
768 int direction = 0,
769 }) {
770 var baseline = isLit(), success = true;
771 switch (verb) {
772 case Verb.go:
773 baseline = move(direction);
774 case Verb.look:
775 describe(true);
776 case Verb.examine:
777 examine(resolve(first));
778 case Verb.take:
779 success = take(first, second);
780 case Verb.drop:
781 drop(first);
782 case Verb.put:
783 success = put(first, second!);
784 case Verb.open:
785 open(resolve(first));
786 case Verb.close:
787 close(resolve(first));
788 case Verb.unlock:
789 lock(first, second, false);
790 case Verb.lock:
791 lock(first, second, true);
792 case Verb.use:
793 use(first, second!);
794 case Verb.light:
795 light(resolve(first), true);
796 case Verb.extinguish:
797 light(resolve(first), false);
798 case Verb.wait:
799 say('Time passes.');
800 case Verb.inventory:
801 if (direct(player).isEmpty) {
802 say('You are empty-handed.');
803 } else {
804 say('You are carrying:');
805 itemTree(player, 1);
806 say('Total weight: ${carriedWeight()}/$maxWeight');
807 }
808 case Verb.score:
809 final treasure = treasures();
810 say(
811 'You have scored ${treasure.points + state.bonus} points (treasure ${treasure.points}, bonus ${state.bonus}) in ${state.moves} moves.',
812 );
813 say('Treasures stored: ${treasure.stored} of ${treasure.total}.');
814 case Verb.save:
815 saved = state.copy();
816 say('Game saved.');
817 case Verb.restore:
818 final snapshot = saved;
819 if (snapshot == null) fail('There is no saved game.');
820 state = snapshot.copy();
821 say('Game restored.');
822 describe(true);
823 case Verb.verbose:
824 verbose = true;
825 say('Verbose mode.');
826 case Verb.brief:
827 verbose = false;
828 say('Brief mode.');
829 }
830 if (success &&
831 const {
832 Verb.go,
833 Verb.take,
834 Verb.drop,
835 Verb.put,
836 Verb.open,
837 Verb.close,
838 Verb.unlock,
839 Verb.lock,
840 Verb.use,
841 Verb.light,
842 Verb.extinguish,
843 Verb.wait,
844 }.contains(verb))
845 turn(baseline);
846 }

848 void interpret(List<String> input) {
849 if (input.isEmpty) fail('I beg your pardon?');
850 final tokens = directionOf(input[0]) != null ? ['go', ...input] : input;
851 var verb = verbs[tokens[0]];
852 if (verb == null) fail('I don\'t know the verb "${tokens[0]}".');
853 var rest = tokens.sublist(1);
854 if (state.end != EndStatus.none && verb != Verb.restore)
855 fail('The game is over.');
856 if (verb == Verb.look && rest.isNotEmpty) {
857 if (rest[0] != 'at') fail('I don\'t understand that.');
858 verb = Verb.examine;
859 rest = rest.sublist(1);
860 }
861 if (const {
862 Verb.look,
863 Verb.inventory,
864 Verb.wait,
865 Verb.score,
866 Verb.save,
867 Verb.restore,
868 Verb.verbose,
869 Verb.brief,
870 }.contains(verb)) {
871 if (rest.isNotEmpty) fail('I don\'t understand that.');
872 run(verb, rest);
873 return;
874 }
875 if (rest.isEmpty)
876 fail(
877 verb == Verb.go
878 ? 'Which way do you want to go?'
879 : 'What do you want to ${verb.name}?',
880 );
881 if (verb == Verb.go) {
882 final direction = rest.length == 1 ? directionOf(rest[0]) : null;
883 if (direction == null) fail('I don\'t understand that.');
884 run(verb, rest, direction: direction);
885 return;
886 }
887 final prepositions = switch (verb) {
888 Verb.put => const {'in', 'into'},
889 Verb.use => const {'on', 'with'},
890 Verb.lock || Verb.unlock when rest.contains('with') => const {'with'},
891 Verb.take when rest.contains('from') => const {'from'},
892 _ => const <String>{},
893 };
894 if (prepositions.isEmpty) {
895 run(verb, rest);
896 return;
897 }
898 final split = rest.indexWhere(prepositions.contains);
899 if (split <= 0 || split == rest.length - 1)
900 fail('I don\'t understand that.');
901 run(verb, rest.sublist(0, split), second: rest.sublist(split + 1));
902 }

904 List<String> start() {
905 output = [];
906 describe(true);
907 return output;

910 List<String> process(String line) {
911 output = ['> $line'];
912 try {
913 var tokens = words(line);
914 if (tokens.isNotEmpty && (tokens[0] == 'again' || tokens[0] == 'g')) {
915 if (tokens.length != 1) fail('I don\'t understand that.');
916 final old = previous;
917 if (old == null) fail('There is nothing to repeat.');
918 tokens = words(old);
919 } else {
920 previous = line;
921 }
922 interpret(tokens);
923 } on CommandFailure catch (failure) {
924 say(failure.message);
925 }
926 return output;
927 }
