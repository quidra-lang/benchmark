150func readRule(input []string, at int, trigger Trigger, rooms, items map[string]int) Rule {
151 rule := Rule{trigger: trigger}
152 if input[at] == "IF" || input[at] == "UNLESS" {
153 rule.flag, rule.expected = input[at+1], input[at] == "IF"
154 at += 2
155 }
156 switch input[at] {
157 case "MESSAGE":
158 rule.effect = Effect{kind: messageEffect, text: strings.Join(input[at+1:], " ")}
159 case "OPEN":
160 rule.effect = Effect{kind: openEffect, room: rooms[input[at+1]], direction: direction(input[at+2]), destination: rooms[input[at+3]]}
161 case "REMOVE":
162 rule.effect = Effect{kind: moveEffect, item: items[input[at+1]], place: Place{kind: nowhere}}
163 case "ADD":
164 rule.effect = Effect{kind: moveEffect, item: items[input[at+1]], place: Place{floor, rooms[input[at+2]]}}
165 case "SET", "CLEAR":
166 rule.effect = Effect{kind: flagEffect, text: input[at+1], on: input[at] == "SET"}
167 case "POINTS":
168 rule.effect = Effect{kind: pointsEffect, points: number(input[at+1])}
169 }
170 return rule
171}

258func newAdventure(definitions [][]string) *Adventure {
259 roomIDs, itemIDs := make(map[string]int), make(map[string]int)
260 var itemNames []string
261 for _, w := range definitions {
262 if w[0] == "ROOM" { roomIDs[w[1]] = len(roomIDs) }
263 if w[0] == "ITEM" { itemNames = append(itemNames, w[1]) }
264 }
265 slices.Sort(itemNames)
266 game := &Adventure{rooms: make([]Room, len(roomIDs)), items: make([]Item, len(itemNames)),
267 vocabulary: make(map[string]bool), pronoun: -1, state: newState(len(roomIDs), len(itemNames))}
268 for i, id := range itemNames {
269 itemIDs[id] = i
270 game.items[i] = Item{nouns: map[string]bool{id: true}, nameWords: make(map[string]bool), key: -1}
271 }
272 for _, w := range definitions {
273 switch w[0] {
274 case "ROOM":
275 room := &game.rooms[roomIDs[w[1]]]
276 room.dark = w[2] == "DARK"
277 at := 2
278 if room.dark { at++ }
279 room.name = strings.Join(w[at:], " ")
280 case "DESC":
281 room := &game.rooms[roomIDs[w[1]]]
282 room.description = append(room.description, strings.Join(w[2:], " "))
283 case "EXIT":
284 exit := Exit{exists: true, destination: roomIDs[w[3]], key: -1, locked: len(w) > 4}
285 if exit.locked { exit.key = itemIDs[w[5]] }
286 game.state.exits[roomIDs[w[1]]][direction(w[2])] = exit
287 case "ITEM":
288 x := itemIDs[w[1]]
289 switch w[2] {
290 case "PLAYER": game.state.places[x] = held
291 case "NOWHERE": game.state.places[x] = Place{kind: nowhere}
292 default: game.state.places[x] = Place{floor, roomIDs[w[2]]}
293 }
294 item, at := &game.items[x], 4
295 item.weight = number(w[3])
296 if w[at] == "TREASURE" { item.treasure = number(w[at+1]); at += 2 }
297 if w[at] == "LIGHT" { item.light = true; game.state.fuel[x] = number(w[at+1]); at += 2 }
298 if w[at] == "CONTAINER" {
299 item.capacity = number(w[at+1]); at += 2
300 if w[at] == "CLOSED" { game.state.containers[x] = closed; at++ }
301 if w[at] == "LOCKED" { item.key = itemIDs[w[at+1]]; game.state.containers[x] = locked; at += 2 }
302 }
303 if w[at] == "FIXED" { item.fixed = true; at++ }
304 item.name = strings.Join(w[at:], " ")
305 for _, word := range w[at:] { item.nameWords[word] = true }
306 case "INSIDE": game.state.places[itemIDs[w[1]]] = Place{inside, itemIDs[w[2]]}
307 case "ALIAS": game.items[itemIDs[w[1]]].nouns[w[2]] = true
308 case "DETAIL":
309 item := &game.items[itemIDs[w[1]]]
310 item.details = append(item.details, strings.Join(w[2:], " "))
311 case "RULE": game.rules = append(game.rules, readRule(w, 3, Trigger{first: itemIDs[w[1]], second: itemIDs[w[2]]}, roomIDs, itemIDs))
312 case "ENTER": game.rules = append(game.rules, readRule(w, 2, Trigger{enter: true, first: roomIDs[w[1]]}, roomIDs, itemIDs))
313 case "TROPHY": game.trophy = roomIDs[w[1]]
314 case "START": game.state.room = roomIDs[w[1]]
315 case "MAXWEIGHT": game.maxWeight = number(w[1])
316 }
317 }
318 game.state.visited[game.state.room] = true
319 for _, item := range game.items {
320 maps.Copy(game.vocabulary, item.nouns)
321 maps.Copy(game.vocabulary, item.nameWords)
322 }
323 return game
324}

338func (game *Adventure) reachable(item int, root Place) bool {
339 place := game.state.places[item]
340 for place.kind == inside {
341 if game.state.containers[place.index] != open { return false }
342 place = game.state.places[place.index]
343 }
344 return place == root
345}

351 if !game.rooms[game.state.room].dark { return true }

406func (game *Adventure) describe(full bool) {
407 if !game.isLit() { game.say("It is pitch dark."); return }
408 room := game.rooms[game.state.room]
409 game.say(room.name)
410 if full { game.output = append(game.output, room.description...) }
411 var exits []string
412 for d, exit := range game.state.exits[game.state.room] {
413 if exit.exists {
414 name := directions[d]
415 if exit.locked { name += " (locked)" }
416 exits = append(exits, name)
417 }
418 }
419 if len(exits) == 0 { exits = append(exits, "none") }
420 game.say("Exits: " + strings.Join(exits, ", "))
421 if len(game.direct(game.floor())) > 0 { game.say("You see:"); game.itemTree(game.floor(), 1) }
422}

424func (game *Adventure) resolve(phrase []string) (int, error) {
425 lit := game.isLit()
426 if len(phrase) == 1 && phrase[0] == "it" {
427 if game.pronoun < 0 { return 0, errors.New("I don't know what \"it\" refers to.") }
428 if !game.carried(game.pronoun) && !(lit && game.roomItem(game.pronoun)) { return 0, errors.New("You can't see any such thing.") }
429 return game.pronoun, nil
430 }
431 for _, word := range phrase {
432 if !game.vocabulary[word] { return 0, fmt.Errorf("I don't know the word \"%s\".", word) }
433 }
434 var matches []int
435 for i, item := range game.items {
436 if !(game.carried(i) || lit && game.roomItem(i)) || !item.nouns[phrase[len(phrase)-1]] { continue }
437 if slices.ContainsFunc(phrase[:len(phrase)-1], func(word string) bool { return !item.nameWords[word] }) { continue }
438 matches = append(matches, i)
439 }
440 if len(matches) == 0 { return 0, errors.New("You can't see any such thing.") }
441 if len(matches) > 1 {
442 var names []string
443 for _, i := range matches { names = append(names, game.items[i].name) }
444 return 0, fmt.Errorf("Which do you mean: %s?", strings.Join(names, ", "))
445 }
446 game.pronoun = matches[0]
447 return matches[0], nil
448}

450func (game *Adventure) runRules(trigger Trigger) (executed int, message bool) {
451 for _, rule := range game.rules {
452 if rule.trigger != trigger || rule.flag != "" && game.state.flags[rule.flag] != rule.expected { continue }
453 executed++
454 effect := rule.effect
455 switch effect.kind {
456 case messageEffect: game.say(effect.text); message = true
457 case openEffect: game.state.exits[effect.room][effect.direction] = Exit{exists: true, destination: effect.destination, key: -1}
458 case moveEffect: game.state.places[effect.item] = effect.place
459 case flagEffect:
460 if effect.on { game.state.flags[effect.text] = true } else { delete(game.state.flags, effect.text) }
461 case pointsEffect: game.state.bonus += effect.points
462 }
463 }
464 return
465}

467func (game *Adventure) turn(baseline bool) {
468 game.state.moves++
469 nearby := make([]bool, len(game.items))
470 for i := range nearby { nearby[i] = game.carried(i) || game.roomItem(i) }
471 for i, present := range nearby {
472 if game.state.lit[i] {
473 game.state.fuel[i]--
474 if game.state.fuel[i] == 0 {
475 game.state.lit[i] = false
476 if present { game.sayf("The %s goes out.", game.items[i].name) }
477 }
478 }
479 }
480 lit := game.isLit()
481 if baseline && !lit { game.say("It is now pitch dark.") } else if !baseline && lit { game.describe(true) }
482 if lit { game.state.darkness = 0 } else { game.state.darkness++ }
483 if game.state.darkness == 2 { game.say("You hear something moving in the dark.") }
484 if game.state.darkness == 3 { game.state.ended = dead }
485 points, stored, total := game.treasures()
486 if game.state.ended == playing && total > 0 && stored == total { game.state.ended = won }
487 if game.state.ended != playing {
488 if game.state.ended == dead { game.say("*** You have died ***") } else { game.say("*** You have won ***") }
489 game.sayf("You scored %d points in %d moves.", points+game.state.bonus, game.state.moves)
490 }
491}

493func (game *Adventure) requireContainer(item int) error {
494 if game.items[item].capacity == 0 { return fmt.Errorf("The %s is not a container.", game.items[item].name) }
495 if game.state.containers[item] != open { return fmt.Errorf("The %s is closed.", game.items[item].name) }
496 return nil

499func (game *Adventure) putItem(item, container int) error {
500 if game.within(container, item) { return fmt.Errorf("The %s is inside the %s.", game.items[container].name, game.items[item].name) }
501 if game.state.places[item] == (Place{inside, container}) { return fmt.Errorf("The %s is already in the %s.", game.items[item].name, game.items[container].name) }
502 weight := game.placeWeight(Place{inside, container})
503 if !game.within(item, container) { weight += game.totalWeight(item) }
504 if weight > game.items[container].capacity { return fmt.Errorf("The %s doesn't fit in the %s.", game.items[item].name, game.items[container].name) }
505 game.state.places[item] = Place{inside, container}
506 return nil
507}

509func (game *Adventure) take(target, from []string) (bool, error) {
510 if !game.isLit() { return false, errors.New("It is pitch dark.") }
511 container, source := -1, game.floor()
512 if len(from) > 0 {
513 var err error
514 container, err = game.resolve(from)
515 if err != nil { return false, err }
516 if err = game.requireContainer(container); err != nil { return false, err }
517 source = Place{inside, container}
518 }
519 if all(target) {
520 candidates := slices.DeleteFunc(game.direct(source), func(i int) bool { return game.items[i].fixed })
521 if len(candidates) == 0 {
522 if container < 0 { return false, errors.New("There is nothing here to take.") }
523 return false, fmt.Errorf("There is nothing in the %s to take.", game.items[container].name)
524 }
525 taken := false
526 for _, item := range candidates {
527 if !game.carried(item) && game.placeWeight(held)+game.totalWeight(item) > game.maxWeight {
528 game.sayf("%s: You can't carry any more.", game.items[item].name)
529 } else {
530 game.state.places[item] = held
531 game.sayf("%s: Taken.", game.items[item].name)
532 taken = true
533 }
534 }
535 return taken, nil
536 }
537 item, err := game.resolve(target)
538 if err != nil { return false, err }
539 if container >= 0 && game.state.places[item] != source { return false, fmt.Errorf("The %s is not in the %s.", game.items[item].name, game.items[container].name) }
540 if game.state.places[item] == held { return false, fmt.Errorf("You already have the %s.", game.items[item].name) }
541 if game.items[item].fixed { return false, fmt.Errorf("The %s is fixed in place.", game.items[item].name) }
542 if !game.carried(item) && game.placeWeight(held)+game.totalWeight(item) > game.maxWeight { return false, errors.New("You can't carry any more.") }
543 game.state.places[item] = held
544 game.say("Taken.")
545 return true, nil
546}

548func (game *Adventure) drop(target []string) error {
549 if all(target) {
550 candidates := game.direct(held)
551 if len(candidates) == 0 { return errors.New("You are empty-handed.") }
552 for _, item := range candidates { game.state.places[item] = game.floor(); game.sayf("%s: Dropped.", game.items[item].name) }
553 } else {
554 item, err := game.resolve(target)
555 if err != nil { return err }
556 if !game.carried(item) { return fmt.Errorf("You aren't carrying the %s.", game.items[item].name) }
557 game.state.places[item] = game.floor()
558 game.say("Dropped.")
559 }
560 return nil
561}

563func (game *Adventure) put(first, second []string) (bool, error) {
564 item := -1
565 var err error
566 if !all(first) { item, err = game.resolve(first); if err != nil { return false, err } }
567 container, err := game.resolve(second)
568 if err != nil { return false, err }
569 if item >= 0 {
570 if !game.carried(item) { return false, fmt.Errorf("You aren't carrying the %s.", game.items[item].name) }
571 if item == container { return false, fmt.Errorf("You can't put the %s inside itself.", game.items[item].name) }
572 }
573 if err = game.requireContainer(container); err != nil { return false, err }
574 if item >= 0 {
575 if err = game.putItem(item, container); err != nil { return false, err }
576 game.sayf("You put the %s in the %s.", game.items[item].name, game.items[container].name)
577 return true, nil
578 }
579 candidates := slices.DeleteFunc(game.direct(held), func(i int) bool { return i == container })
580 if len(candidates) == 0 { return false, fmt.Errorf("You have nothing to put in the %s.", game.items[container].name) }
581 moved := false
582 for _, item := range candidates {
583 message := "Done."
584 if err := game.putItem(item, container); err != nil { message = err.Error() } else { moved = true }
585 game.sayf("%s: %s", game.items[item].name, message)
586 }
587 return moved, nil
588}

590func (game *Adventure) keyCheck(key int, phrase []string) error {
591 if len(phrase) == 0 {
592 if !game.carried(key) { return errors.New("You don't have the right key.") }
593 game.sayf("(with the %s)", game.items[key].name)
594 } else {
595 item, err := game.resolve(phrase)
596 if err != nil { return err }
597 if !game.carried(item) { return fmt.Errorf("You aren't carrying the %s.", game.items[item].name) }
598 if item != key { return fmt.Errorf("The %s doesn't fit.", game.items[item].name) }
599 }
600 return nil
601}

603func (game *Adventure) changeLock(target, keyPhrase []string, locking bool) error {
604 d, status := -1, "not locked"
605 if len(target) == 1 { d = direction(target[0]) }
606 if locking { status = "already locked" }
607 if d >= 0 {
608 exit := game.state.exits[game.state.room][d]
609 if !exit.exists { return fmt.Errorf("There is no exit %s.", directions[d]) }
610 if exit.key < 0 { return fmt.Errorf("The way %s has no lock.", directions[d]) }
611 if exit.locked == locking { return fmt.Errorf("The way %s is %s.", directions[d], status) }
612 if err := game.keyCheck(exit.key, keyPhrase); err != nil { return err }
613 exit.locked = locking
614 game.state.exits[game.state.room][d] = exit
615 opposite := game.state.exits[exit.destination][d^1]
616 if opposite.exists && opposite.destination == game.state.room && opposite.key == exit.key {
617 opposite.locked = locking
618 game.state.exits[exit.destination][d^1] = opposite
619 }
620 } else {
621 item, err := game.resolve(target)
622 if err != nil { return err }
623 if game.items[item].capacity == 0 || game.items[item].key < 0 { return fmt.Errorf("The %s has no lock.", game.items[item].name) }
624 if (game.state.containers[item] == locked) == locking { return fmt.Errorf("The %s is %s.", game.items[item].name, status) }
625 if locking && game.state.containers[item] == open { return fmt.Errorf("You must close the %s first.", game.items[item].name) }
626 if err := game.keyCheck(game.items[item].key, keyPhrase); err != nil { return err }
627 game.state.containers[item] = closed
628 if locking { game.state.containers[item] = locked }
629 }
630 if locking { game.say("Locked.") } else { game.say("Unlocked.") }
631 return nil
632}

634func (game *Adventure) examine(item int) {
635 definition := game.items[item]
636 if len(definition.details) == 0 { game.sayf("You see nothing special about the %s.", definition.name) } else { game.output = append(game.output, definition.details...) }
637 if definition.light {
638 status := "unlit"
639 if game.state.lit[item] { status = "lit" }
640 game.sayf("It is %s (fuel %d).", status, game.state.fuel[item])
641 }
642 if definition.capacity > 0 {
643 switch game.state.containers[item] {
644 case locked: game.say("It is locked.")
645 case closed: game.say("It is closed.")
646 case open:
647 if len(game.direct(Place{inside, item})) == 0 { game.say("It is empty.") } else { game.say("It contains:"); game.itemTree(Place{inside, item}, 1) }
648 }
649 }
650 if definition.treasure > 0 { game.sayf("It is a treasure worth %d points.", definition.treasure) }
651 game.sayf("It weighs %d.", game.totalWeight(item))
652}

654func (game *Adventure) execute(verb Verb, first, second []string, d int, baseline *bool) (bool, error) {
655 switch verb {
656 case goVerb:
657 exit := game.state.exits[game.state.room][d]
658 if !exit.exists { return false, errors.New("You can't go that way.") }
659 if exit.locked { return false, fmt.Errorf("The way %s is locked.", directions[d]) }
660 game.state.room = exit.destination
661 *baseline = game.isLit()
662 game.describe(game.verbose || !game.state.visited[game.state.room])
663 game.state.visited[game.state.room] = true
664 game.runRules(Trigger{enter: true, first: game.state.room})
665 case look: game.describe(true)
666 case examine:
667 item, err := game.resolve(first)
668 if err != nil { return false, err }
669 game.examine(item)
670 case take: return game.take(first, second)
671 case drop: if err := game.drop(first); err != nil { return false, err }
672 case put: return game.put(first, second)
673 case inventory:
674 if len(game.direct(held)) == 0 { game.say("You are empty-handed.") } else {
675 game.say("You are carrying:"); game.itemTree(held, 1)
676 game.sayf("Total weight: %d/%d", game.placeWeight(held), game.maxWeight)
677 }
678 case openVerb:
679 item, err := game.resolve(first)
680 if err != nil { return false, err }
681 if game.items[item].capacity == 0 { return false, fmt.Errorf("You can't open the %s.", game.items[item].name) }
682 if game.state.containers[item] == open { return false, fmt.Errorf("The %s is already open.", game.items[item].name) }
683 if game.state.containers[item] == locked { return false, fmt.Errorf("The %s is locked.", game.items[item].name) }
684 game.state.containers[item] = open
685 var names []string
686 for _, child := range game.direct(Place{inside, item}) { names = append(names, game.items[child].name) }
687 if len(names) == 0 { game.say("Opened.") } else { game.sayf("Opening the %s reveals %s.", game.items[item].name, strings.Join(names, ", ")) }
688 case closeVerb:
689 item, err := game.resolve(first)
690 if err != nil { return false, err }
691 if game.items[item].capacity == 0 { return false, fmt.Errorf("You can't close the %s.", game.items[item].name) }
692 if game.state.containers[item] != open { return false, fmt.Errorf("The %s is already closed.", game.items[item].name) }
693 game.state.containers[item] = closed
694 game.say("Closed.")
695 case unlock, lock: if err := game.changeLock(first, second, verb == lock); err != nil { return false, err }
696 case use:
697 a, err := game.resolve(first)
698 if err != nil { return false, err }
699 b, err := game.resolve(second)
700 if err != nil { return false, err }
701 if !game.carried(a) { return false, fmt.Errorf("You aren't carrying the %s.", game.items[a].name) }
702 executed, message := game.runRules(Trigger{first: a, second: b})
703 if executed == 0 { return false, errors.New("Nothing happens.") }
704 if !message { game.say("Done.") }
705 case light, extinguish:
706 item, err := game.resolve(first)
707 if err != nil { return false, err }
708 lighting := verb == light
709 if !game.items[item].light { return false, fmt.Errorf("You can't %s the %s.", verb, game.items[item].name) }
710 if game.state.lit[item] == lighting {
711 status := "not lit"
712 if lighting { status = "already lit" }
713 return false, fmt.Errorf("The %s is %s.", game.items[item].name, status)
714 }
715 if lighting && game.state.fuel[item] == 0 { return false, fmt.Errorf("The %s has no fuel.", game.items[item].name) }
716 game.state.lit[item] = lighting
717 status := "out"
718 if lighting { status = "lit" }
719 game.sayf("The %s is now %s.", game.items[item].name, status)
720 case wait: game.say("Time passes.")
721 case score:
722 points, stored, total := game.treasures()
723 game.sayf("You have scored %d points (treasure %d, bonus %d) in %d moves.", points+game.state.bonus, points, game.state.bonus, game.state.moves)
724 game.sayf("Treasures stored: %d of %d.", stored, total)
725 case save:
726 copy := game.state.clone()
727 game.saved = &copy
728 game.say("Game saved.")
729 case restore:
730 if game.saved == nil { return false, errors.New("There is no saved game.") }
731 game.state = game.saved.clone()
732 game.say("Game restored.")
733 game.describe(true)
734 case verbose: game.verbose = true; game.say("Verbose mode.")
735 case brief: game.verbose = false; game.say("Brief mode.")
736 }
737 return true, nil
738}

740func (game *Adventure) interpret(input []string) error {
741 if len(input) == 0 { return errors.New("I beg your pardon?") }
742 if direction(input[0]) >= 0 { input = append([]string{"go"}, input...) }
743 verb := parseVerb(input[0])
744 if verb == unknown { return fmt.Errorf("I don't know the verb \"%s\".", input[0]) }
745 if game.state.ended != playing && verb != restore { return errors.New("The game is over.") }
746 first, second, d := input[1:], []string(nil), -1
747 if verb == look && len(first) > 0 {
748 if first[0] != "at" { return errors.New("I don't understand that.") }
749 verb, first = examine, first[1:]
750 }
751 if verb.noArguments() {
752 if len(first) > 0 { return errors.New("I don't understand that.") }
753 } else {
754 if len(first) == 0 {
755 if verb == goVerb { return errors.New("Which way do you want to go?") }
756 return fmt.Errorf("What do you want to %s?", verb)
757 }
758 if verb == goVerb {
759 if len(first) == 1 { d = direction(first[0]) }
760 if d < 0 { return errors.New("I don't understand that.") }
761 } else {
762 at := slices.IndexFunc(first, func(word string) bool {
763 switch verb {
764 case put: return word == "in" || word == "into"
765 case use: return word == "on" || word == "with"
766 case unlock, lock: return word == "with"
767 case take: return word == "from"
768 }
769 return false
770 })
771 if at >= 0 {
772 if at == 0 || at+1 == len(first) { return errors.New("I don't understand that.") }
773 first, second = first[:at], first[at+1:]
774 } else if verb == put || verb == use { return errors.New("I don't understand that.") }
775 }
776 }
777 baseline := game.isLit()
778 succeeded, err := game.execute(verb, first, second, d, &baseline)
779 if err != nil { return err }
780 if succeeded && verb.action() { game.turn(baseline) }
781 return nil
782}

784func (game *Adventure) dispatch(line string) error {
785 input := words(line)
786 if len(input) > 0 && (input[0] == "again" || input[0] == "g") {
787 if len(input) != 1 { return errors.New("I don't understand that.") }
788 if game.previous == "" { return errors.New("There is nothing to repeat.") }
789 input = words(game.previous)
790 } else { game.previous = line }
791 return game.interpret(input)
792}

794func main() {
795 input := bufio.NewScanner(os.Stdin)
796 var definitions [][]string
797 for input.Scan() {
798 if input.Text() == "BEGIN" { break }
799 definitions = append(definitions, strings.Fields(input.Text()))
800 }
801 game := newAdventure(definitions)
802 output := bufio.NewWriter(os.Stdout)
803 defer output.Flush()
804 flush := func() {
805 for _, line := range game.output { fmt.Fprintln(output, line) }
806 game.output = nil
807 }
808 game.describe(true)
809 flush()
810 for input.Scan() {
811 line := input.Text()
812 if line == "QUIT" { break }
813 game.say("> " + line)
814 if err := game.dispatch(line); err != nil { game.say(err.Error()) }
815 flush()
816 }
817 fmt.Fprintf(output, "Final score: %d points in %d moves.\n", game.score(), game.state.moves)
818}
