128function read_rule(tokens::Vector{String}, position::Int, trigger::Trigger, rooms::Dict{String,Int}, items::Dict{String,Int})
129 condition = nothing
130 if tokens[position] in ("IF", "UNLESS")
131 condition = Condition(tokens[position + 1], tokens[position] == "IF")
132 position += 2
133 end
134 kind = tokens[position]
135 args = tokens[position+1:end]
136 effect = if kind == "MESSAGE"
137 MessageEffect(join(args, " "))
138 elseif kind == "OPEN"
139 OpenEffect(rooms[args[1]], direction(args[2]), rooms[args[3]])
140 elseif kind == "REMOVE"
141 MoveEffect(items[args[1]], REMOVED)
142 elseif kind == "ADD"
143 MoveEffect(items[args[1]], floor_place(rooms[args[2]]))
144 elseif kind in ("SET", "CLEAR")
145 FlagEffect(args[1], kind == "SET")
146 else
147 PointsEffect(parse(Int, args[1]))
148 end
149 Rule(trigger, condition, effect)
150end
151function Adventure(definitions::Vector{Vector{String}})
152 room_ids = [tokens[2] for tokens in definitions if tokens[1] == "ROOM"]
153 item_ids = sort([tokens[2] for tokens in definitions if tokens[1] == "ITEM"])
154 room_index = Dict{String,Int}(id => index for (index, id) in enumerate(room_ids))
155 item_index = Dict{String,Int}(id => index for (index, id) in enumerate(item_ids))
156 rooms = [Room("", false, String[]) for _ in room_ids]
157 items = Item.(item_ids)
158 exits = fill!(Matrix{Union{Nothing,ExitLink}}(undef, length(rooms), 6), nothing)
159 state = GameState(1, fill(PLAYER, length(items)), fill(OPEN, length(items)), exits, falses(length(items)),
160 zeros(Int, length(items)), Set{String}(), 0, 0, falses(length(rooms)), 0, PLAYING)
161 rules = Rule[]
162 trophy = max_weight = 0
163 for tokens in definitions
164 kind = tokens[1]
165 if kind == "ROOM"
166 dark = tokens[3] == "DARK"
167 rooms[room_index[tokens[2]]] = Room(join(tokens[(dark ? 4 : 3):end], " "), dark, String[])
168 elseif kind == "DESC"
169 push!(rooms[room_index[tokens[2]]].description, join(tokens[3:end], " "))
170 elseif kind == "EXIT"
171 key = length(tokens) > 4 ? item_index[tokens[6]] : nothing
172 state.exits[room_index[tokens[2]], direction(tokens[3])] = ExitLink(room_index[tokens[4]], key, key !== nothing)
173 elseif kind == "ITEM"
174 index = item_index[tokens[2]]
175 item = items[index]
176 state.places[index] = tokens[3] == "PLAYER" ? PLAYER : tokens[3] == "NOWHERE" ? REMOVED : floor_place(room_index[tokens[3]])
177 item.weight = parse(Int, tokens[4])
178 position = 5
179 if tokens[position] == "TREASURE"
180 item.treasure = parse(Int, tokens[position + 1]); position += 2
181 end
182 if tokens[position] == "LIGHT"
183 item.light = true; state.fuel[index] = parse(Int, tokens[position + 1]); position += 2
184 end
185 if tokens[position] == "CONTAINER"
186 item.capacity = parse(Int, tokens[position + 1]); position += 2
187 if tokens[position] == "CLOSED"
188 state.containers[index] = CLOSED; position += 1
189 end
190 if tokens[position] == "LOCKED"
191 item.key = item_index[tokens[position + 1]]; state.containers[index] = LOCKED; position += 2
192 end
193 end
194 if tokens[position] == "FIXED"
195 item.fixed = true; position += 1
196 end
197 item.name = join(tokens[position:end], " ")
198 union!(item.words, tokens[position:end])
199 elseif kind == "INSIDE"
200 state.places[item_index[tokens[2]]] = inside(item_index[tokens[3]])
201 elseif kind == "ALIAS"
202 push!(items[item_index[tokens[2]]].nouns, tokens[3])
203 elseif kind == "DETAIL"
204 push!(items[item_index[tokens[2]]].details, join(tokens[3:end], " "))
205 elseif kind == "RULE"
206 push!(rules, read_rule(tokens, 4, Trigger(USE_ITEMS, item_index[tokens[2]], item_index[tokens[3]]), room_index, item_index))
207 elseif kind == "ENTER"
208 push!(rules, read_rule(tokens, 3, Trigger(ENTER_ROOM, room_index[tokens[2]], 0), room_index, item_index))
209 elseif kind == "TROPHY"
210 trophy = room_index[tokens[2]]
211 elseif kind == "START"
212 state.room = room_index[tokens[2]]
213 elseif kind == "MAXWEIGHT"
214 max_weight = parse(Int, tokens[2])
215 end
216 end
217 state.visited[state.room] = true
218 vocabulary = Set{String}()
219 for item in items
220 union!(vocabulary, item.nouns, item.words)
221 end
222 Adventure(World(rooms, items, rules, vocabulary, trophy, max_weight), state, nothing, nothing, nothing, false)
223end

225current_floor(game::Adventure) = floor_place(game.state.room)
226direct(game::Adventure, place::Place) = findall(==(place), game.state.places)
227function reachable(game::Adventure, item::Int, root::Place)
228 place = game.state.places[item]
229 while place.kind == INSIDE
230 game.state.containers[place.index] == OPEN || return false
231 place = game.state.places[place.index]
232 end
233 place == root
234end
235carried(game::Adventure, item::Int) = reachable(game, item, PLAYER)
236room_item(game::Adventure, item::Int) = reachable(game, item, current_floor(game))
237is_lit(game::Adventure) = !game.world.rooms[game.state.room].dark ||
238 any(i -> game.state.lit[i] && (carried(game, i) || room_item(game, i)), eachindex(game.world.items))
239function visible(game::Adventure)
240 lit = is_lit(game)
241 [carried(game, i) || (lit && room_item(game, i)) for i in eachindex(game.world.items)]
242end
243function within(game::Adventure, item::Int, ancestor::Int)
244 place = game.state.places[item]
245 while place.kind == INSIDE
246 place.index == ancestor && return true
247 place = game.state.places[place.index]
248 end
249 false
250end
251total_weight(game::Adventure, item::Int) = game.world.items[item].weight + place_weight(game, inside(item))
252place_weight(game::Adventure, place::Place) = sum(i -> total_weight(game, i), direct(game, place); init = 0)
253function stored(game::Adventure, item::Int)
254 place = game.state.places[item]
255 while place.kind == INSIDE
256 place = game.state.places[place.index]
257 end
258 place == floor_place(game.world.trophy)
259end
260function treasures(game::Adventure)
261 indices = findall(item -> item.treasure > 0, game.world.items)
262 collected = filter(i -> stored(game, i), indices)
263 (points = sum(i -> game.world.items[i].treasure, collected; init = 0), stored = length(collected), total = length(indices))
264end
265score(game::Adventure) = treasures(game).points + game.state.bonus
266item_names(game::Adventure, items::Vector{Int}) = join((game.world.items[i].name for i in items), ", ")
267function display_name(game::Adventure, item::Int)
268 annotations = String[]

270 game.state.lit[item] && push!(annotations, "lit")
271 game.world.items[item].name * (isempty(annotations) ? "" : " ($(join(annotations, ", ")))")
272end
273function item_tree(game::Adventure, place::Place, depth::Int)
274 for item in direct(game, place)
275 println(" " ^ depth, display_name(game, item))
276 if game.world.items[item].capacity > 0 && game.state.containers[item] == OPEN
277 item_tree(game, inside(item), depth + 1)
278 end
279 end

281function describe(game::Adventure, full::Bool)
282 if !is_lit(game)
283 println("It is pitch dark.")
284 return
285 end
286 room = game.state.room
287 println(game.world.rooms[room].name)
288 full && foreach(println, game.world.rooms[room].description)
289 exits = String[]
290 for d in 1:6
291 exit = game.state.exits[room, d]
292 exit === nothing || push!(exits, DIRECTIONS[d] * (exit.locked ? " (locked)" : ""))
293 end
294 println("Exits: ", isempty(exits) ? "none" : join(exits, ", "))
295 if !isempty(direct(game, current_floor(game)))
296 println("You see:"); item_tree(game, current_floor(game), 1)
297 end

299function resolve!(game::Adventure, phrase::Vector{String})
300 seen = visible(game)
301 if phrase == ["it"]
302 game.pronoun === nothing && fail("I don't know what \"it\" refers to.")
303 seen[game.pronoun] || fail("You can't see any such thing.")
304 return game.pronoun::Int
305 end
306 for word in phrase
307 word in game.world.vocabulary || fail("I don't know the word \"$word\".")
308 end
309 matches = [i for i in eachindex(game.world.items) if seen[i] && last(phrase) in game.world.items[i].nouns &&
310 all(word -> word in game.world.items[i].words, phrase[1:end-1])]
311 isempty(matches) && fail("You can't see any such thing.")
312 length(matches) > 1 && fail("Which do you mean: $(item_names(game, matches))?")
313 game.pronoun = only(matches)
314 game.pronoun::Int
315end
316function run_rules!(game::Adventure, trigger::Trigger)
317 executed = 0
318 message = false
319 for rule in game.world.rules
320 rule.trigger == trigger || continue
321 condition = rule.condition
322 condition !== nothing && (condition.flag in game.state.flags) != condition.on && continue
323 executed += 1
324 effect = rule.effect
325 if effect isa MessageEffect
326 println(effect.text); message = true
327 elseif effect isa OpenEffect
328 game.state.exits[effect.room, effect.direction] = ExitLink(effect.destination, nothing, false)
329 elseif effect isa MoveEffect
330 game.state.places[effect.item] = effect.place
331 elseif effect isa FlagEffect
332 effect.on ? push!(game.state.flags, effect.flag) : delete!(game.state.flags, effect.flag)
333 elseif effect isa PointsEffect
334 game.state.bonus += effect.amount
335 end
336 end
337 (; executed, message)
338end
339function pass_turn!(game::Adventure, baseline::Bool)
340 game.state.moves += 1
341 nearby = [carried(game, i) || room_item(game, i) for i in eachindex(game.world.items)]
342 for item in eachindex(game.world.items)
343 game.state.lit[item] || continue
344 game.state.fuel[item] -= 1
345 if game.state.fuel[item] == 0
346 game.state.lit[item] = false
347 nearby[item] && println("The $(game.world.items[item].name) goes out.")
348 end
349 end
350 lit = is_lit(game)
351 if baseline && !lit
352 println("It is now pitch dark.")
353 elseif !baseline && lit
354 describe(game, true)
355 end
356 game.state.darkness = lit ? 0 : game.state.darkness + 1
357 game.state.darkness == 2 && println("You hear something moving in the dark.")
358 game.state.darkness == 3 && (game.state.ended = DEAD)
359 treasure = treasures(game)
360 if game.state.ended == PLAYING && treasure.total > 0 && treasure.stored == treasure.total
361 game.state.ended = WON
362 end
363 if game.state.ended != PLAYING
364 println(game.state.ended == DEAD ? "*** You have died ***" : "*** You have won ***")
365 println("You scored $(treasure.points + game.state.bonus) points in $(game.state.moves) moves.")
366 end
367end

369function require_container(game::Adventure, item::Int)
370 game.world.items[item].capacity > 0 || fail("The $(game.world.items[item].name) is not a container.")
371 game.state.containers[item] == OPEN || fail("The $(game.world.items[item].name) is closed.")

373function put_item!(game::Adventure, item::Int, container::Int)
374 name, target = game.world.items[item].name, game.world.items[container].name
375 within(game, container, item) && return "The $target is inside the $name."
376 game.state.places[item] == inside(container) && return "The $name is already in the $target."
377 weight = place_weight(game, inside(container)) + (within(game, item, container) ? 0 : total_weight(game, item))
378 weight > game.world.items[container].capacity && return "The $name doesn't fit in the $target."
379 game.state.places[item] = inside(container)
380 ""
381end
382function take_items!(game::Adventure, first::Vector{String}, second::Vector{String})
383 is_lit(game) || fail("It is pitch dark.")
384 container = nothing
385 source = current_floor(game)
386 if !isempty(second)
387 container = resolve!(game, second); require_container(game, container); source = inside(container)
388 end
389 if all_items(first)
390 candidates = filter(i -> !game.world.items[i].fixed, direct(game, source))
391 isempty(candidates) && fail(container === nothing ? "There is nothing here to take." : "There is nothing in the $(game.world.items[container].name) to take.")
392 taken = false
393 for item in candidates
394 if !carried(game, item) && place_weight(game, PLAYER) + total_weight(game, item) > game.world.max_weight
395 println("$(game.world.items[item].name): You can't carry any more.")
396 else
397 game.state.places[item] = PLAYER; println("$(game.world.items[item].name): Taken."); taken = true
398 end
399 end
400 return taken
401 end
402 item = resolve!(game, first)
403 name = game.world.items[item].name
404 container !== nothing && game.state.places[item] != source && fail("The $name is not in the $(game.world.items[container].name).")
405 game.state.places[item] == PLAYER && fail("You already have the $name.")
406 game.world.items[item].fixed && fail("The $name is fixed in place.")
407 !carried(game, item) && place_weight(game, PLAYER) + total_weight(game, item) > game.world.max_weight && fail("You can't carry any more.")
408 game.state.places[item] = PLAYER
409 println("Taken.")
410 true
411end
412function drop_items!(game::Adventure, target::Vector{String})
413 if all_items(target)
414 candidates = direct(game, PLAYER)
415 isempty(candidates) && fail("You are empty-handed.")
416 for item in candidates
417 game.state.places[item] = current_floor(game)
418 println("$(game.world.items[item].name): Dropped.")
419 end
420 else
421 item = resolve!(game, target)
422 carried(game, item) || fail("You aren't carrying the $(game.world.items[item].name).")
423 game.state.places[item] = current_floor(game)
424 println("Dropped.")
425 end
426end
427function put_items!(game::Adventure, first::Vector{String}, second::Vector{String})
428 item = all_items(first) ? nothing : resolve!(game, first)
429 container = resolve!(game, second)
430 if item !== nothing
431 carried(game, item) || fail("You aren't carrying the $(game.world.items[item].name).")
432 item == container && fail("You can't put the $(game.world.items[item].name) inside itself.")
433 end
434 require_container(game, container)
435 if item !== nothing
436 reason = put_item!(game, item, container)
437 isempty(reason) || fail(reason)
438 println("You put the $(game.world.items[item].name) in the $(game.world.items[container].name).")
439 return true
440 end
441 candidates = filter(!=(container), direct(game, PLAYER))
442 isempty(candidates) && fail("You have nothing to put in the $(game.world.items[container].name).")
443 moved = false
444 for candidate in candidates
445 reason = put_item!(game, candidate, container)
446 isempty(reason) && (moved = true)
447 println("$(game.world.items[candidate].name): ", isempty(reason) ? "Done." : reason)
448 end
449 moved
450end

461function change_lock!(game::Adventure, target::Vector{String}, key_phrase::Vector{String}, locking::Bool)
462 d = length(target) == 1 ? direction(only(target)) : 0
463 status = locking ? "already locked" : "not locked"
464 if d != 0
465 room = game.state.room
466 exit = game.state.exits[room, d]
467 exit === nothing && fail("There is no exit $(DIRECTIONS[d]).")
468 key = exit.key
469 key === nothing && fail("The way $(DIRECTIONS[d]) has no lock.")
470 exit.locked == locking && fail("The way $(DIRECTIONS[d]) is $status.")
471 key_check!(game, key, key_phrase)
472 game.state.exits[room, d] = ExitLink(exit.destination, key, locking)
473 reverse = xor(d - 1, 1) + 1
474 opposite = game.state.exits[exit.destination, reverse]
475 if opposite !== nothing && opposite.destination == room && opposite.key == key
476 game.state.exits[exit.destination, reverse] = ExitLink(opposite.destination, key, locking)
477 end
478 else
479 item = resolve!(game, target)
480 value = game.world.items[item]
481 key = value.key
482 (value.capacity == 0 || key === nothing) && fail("The $(value.name) has no lock.")
483 (game.state.containers[item] == LOCKED) == locking && fail("The $(value.name) is $status.")
484 locking && game.state.containers[item] == OPEN && fail("You must close the $(value.name) first.")
485 key_check!(game, key, key_phrase)
486 game.state.containers[item] = locking ? LOCKED : CLOSED
487 end
488 println(locking ? "Locked." : "Unlocked.")
489end
490function examine(game::Adventure, item::Int)
491 value = game.world.items[item]
492 isempty(value.details) ? println("You see nothing special about the $(value.name).") : foreach(println, value.details)
493 value.light && println("It is $(game.state.lit[item] ? "lit" : "unlit") (fuel $(game.state.fuel[item])).")
494 if value.capacity > 0
495 state = game.state.containers[item]
496 if state != OPEN
497 println("It is $(lowercase(string(state))).")
498 elseif isempty(direct(game, inside(item)))
499 println("It is empty.")
500 else
501 println("It contains:"); item_tree(game, inside(item), 1)
502 end
503 end
504 value.treasure > 0 && println("It is a treasure worth $(value.treasure) points.")
505 println("It weighs $(total_weight(game, item)).")
506end

508function execute!(game::Adventure, verb::Verb, first::Vector{String}, second::Vector{String}, d::Int)
509 baseline = is_lit(game)
510 success = true
511 if verb == GO
512 exit = game.state.exits[game.state.room, d]
513 exit === nothing && fail("You can't go that way.")
514 exit.locked && fail("The way $(DIRECTIONS[d]) is locked.")
515 game.state.room = exit.destination
516 baseline = is_lit(game)
517 describe(game, game.verbose || !game.state.visited[game.state.room])
518 game.state.visited[game.state.room] = true
519 run_rules!(game, Trigger(ENTER_ROOM, game.state.room, 0))
520 elseif verb == LOOK
521 describe(game, true)
522 elseif verb == EXAMINE
523 examine(game, resolve!(game, first))
524 elseif verb == TAKE
525 success = take_items!(game, first, second)
526 elseif verb == DROP
527 drop_items!(game, first)
528 elseif verb == PUT
529 success = put_items!(game, first, second)
530 elseif verb == INVENTORY
531 if isempty(direct(game, PLAYER))
532 println("You are empty-handed.")
533 else
534 println("You are carrying:"); item_tree(game, PLAYER, 1)
535 println("Total weight: $(place_weight(game, PLAYER))/$(game.world.max_weight)")
536 end
537 elseif verb == OPEN_ITEM
538 item = resolve!(game, first)
539 name = game.world.items[item].name
540 game.world.items[item].capacity > 0 || fail("You can't open the $name.")
541 game.state.containers[item] == OPEN && fail("The $name is already open.")
542 game.state.containers[item] == LOCKED && fail("The $name is locked.")
543 game.state.containers[item] = OPEN
544 children = direct(game, inside(item))
545 println(isempty(children) ? "Opened." : "Opening the $name reveals $(item_names(game, children)).")
546 elseif verb == CLOSE_ITEM
547 item = resolve!(game, first)
548 name = game.world.items[item].name
549 game.world.items[item].capacity > 0 || fail("You can't close the $name.")
550 game.state.containers[item] == OPEN || fail("The $name is already closed.")
551 game.state.containers[item] = CLOSED
552 println("Closed.")
553 elseif verb in (UNLOCK, LOCK)
554 change_lock!(game, first, second, verb == LOCK)
555 elseif verb == USE
556 a = resolve!(game, first)
557 b = resolve!(game, second)
558 carried(game, a) || fail("You aren't carrying the $(game.world.items[a].name).")
559 result = run_rules!(game, Trigger(USE_ITEMS, a, b))
560 result.executed > 0 || fail("Nothing happens.")
561 result.message || println("Done.")
562 elseif verb in (LIGHT, EXTINGUISH)
563 item = resolve!(game, first)
564 value = game.world.items[item]
565 lighting = verb == LIGHT
566 value.light || fail("You can't $(canonical(verb)) the $(value.name).")
567 game.state.lit[item] == lighting && fail("The $(value.name) is $(lighting ? "already lit" : "not lit").")
568 lighting && game.state.fuel[item] == 0 && fail("The $(value.name) has no fuel.")
569 game.state.lit[item] = lighting
570 println("The $(value.name) is now $(lighting ? "lit" : "out").")
571 elseif verb == WAIT
572 println("Time passes.")
573 elseif verb == SCORE
574 treasure = treasures(game)
575 println("You have scored $(treasure.points + game.state.bonus) points (treasure $(treasure.points), bonus $(game.state.bonus)) in $(game.state.moves) moves.")
576 println("Treasures stored: $(treasure.stored) of $(treasure.total).")
577 elseif verb == SAVE
578 game.saved = copy(game.state)
579 println("Game saved.")
580 elseif verb == RESTORE
581 game.saved === nothing && fail("There is no saved game.")
582 game.state = copy(game.saved)
583 println("Game restored."); describe(game, true)
584 elseif verb == VERBOSE
585 game.verbose = true; println("Verbose mode.")
586 elseif verb == BRIEF
587 game.verbose = false; println("Brief mode.")
588 end
589 success && verb in ACTIONS && pass_turn!(game, baseline)
590 nothing
591end
592function interpret!(game::Adventure, tokens::Vector{String})
593 isempty(tokens) && fail("I beg your pardon?")
594 direction(tokens[1]) != 0 && pushfirst!(tokens, "go")
595 verb = get(VERBS, tokens[1], nothing)
596 verb === nothing && fail("I don't know the verb \"$(tokens[1])\".")
597 game.state.ended != PLAYING && verb != RESTORE && fail("The game is over.")
598 first, second = tokens[2:end], String[]
599 d = 0
600 if verb == LOOK && !isempty(first)
601 first[1] == "at" || fail("I don't understand that.")
602 verb = EXAMINE; first = first[2:end]
603 end
604 if verb in NO_ARGUMENTS
605 isempty(first) || fail("I don't understand that.")
606 else
607 isempty(first) && fail(verb == GO ? "Which way do you want to go?" : "What do you want to $(canonical(verb))?")
608 if verb == GO
609 length(first) == 1 && (d = direction(first[1]))
610 d == 0 && fail("I don't understand that.")
611 else
612 prepositions = verb == PUT ? ("in", "into") : verb == USE ? ("on", "with") :
613 verb in (UNLOCK, LOCK) ? ("with",) : verb == TAKE ? ("from",) : ()
614 split = findfirst(word -> word in prepositions, first)
615 if split !== nothing
616 (split == 1 || split == length(first)) && fail("I don't understand that.")
617 second, first = first[split+1:end], first[1:split-1]
618 elseif verb in (PUT, USE)
619 fail("I don't understand that.")
620 end
621 end
622 end
623 execute!(game, verb, first, second, d)
624end
625function process_line!(game::Adventure, line::String)
626 println("> ", line)
627 try
628 tokens = command_words(line)
629 if !isempty(tokens) && tokens[1] in ("again", "g")
630 length(tokens) == 1 || fail("I don't understand that.")
631 game.previous === nothing && fail("There is nothing to repeat.")
632 tokens = command_words(game.previous)
633 else
634 game.previous = line
635 end
636 interpret!(game, tokens)
637 catch error
638 error isa CommandError || rethrow()
639 println(error.message)
640 end
641end

644 definitions = Vector{String}[]
645 for line in eachline(stdin)
646 line == "BEGIN" && break
