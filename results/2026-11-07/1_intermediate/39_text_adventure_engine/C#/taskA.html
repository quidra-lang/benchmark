79 static Dictionary<string, Verb> MakeVerbs()
80 {
81 string[] groups = ["go walk run", "look l", "examine x inspect read", "take get grab", "drop discard",
82 "put insert place", "inventory inv i", "open", "close shut", "unlock", "lock", "use apply",
83 "light ignite", "extinguish douse snuff", "wait z", "score", "save", "restore load", "verbose", "brief"];
84 Dictionary<string, Verb> result = new(StringComparer.Ordinal);
85 for (int i = 0; i < groups.Length; i++)
86 foreach (string word in groups[i].Split(' ')) result[word] = (Verb)i;
87 return result;

97 public Adventure(List<string[]> definitions)
98 {
99 rooms = definitions.Where(x => x[0] == "ROOM").Select(x => new Room(x[1])).ToArray();
100 items = definitions.Where(x => x[0] == "ITEM").Select(x => new Item(x[1]))
101 .OrderBy(x => x.Id, StringComparer.Ordinal).ToArray();
102 var roomIndex = rooms.Select((x, i) => (x.Id, i)).ToDictionary(x => x.Id, x => x.i, StringComparer.Ordinal);
103 var itemIndex = items.Select((x, i) => (x.Id, i)).ToDictionary(x => x.Id, x => x.i, StringComparer.Ordinal);
104 state = new(rooms.Length, items.Length);
105 Place InitialPlace(string text) => text switch
106 {
107 "PLAYER" => Player, "NOWHERE" => new(PlaceKind.Nowhere), _ => new(PlaceKind.Room, roomIndex[text])
108 };
109 Rule ParseRule(string[] words, int index)
110 {
111 string? flag = null;
112 bool required = true;
113 if (words[index] is "IF" or "UNLESS")
114 {
115 required = words[index] == "IF"; flag = words[index + 1]; index += 2;
116 }
117 Effect effect = words[index] switch
118 {
119 "MESSAGE" => new Message(string.Join(' ', words.Skip(index + 1))),
120 "OPEN" => new OpenExit(roomIndex[words[index + 1]], Direction(words[index + 2])!.Value, roomIndex[words[index + 3]]),
121 "REMOVE" => new MoveItem(itemIndex[words[index + 1]], new(PlaceKind.Nowhere)),
122 "ADD" => new MoveItem(itemIndex[words[index + 1]], new(PlaceKind.Room, roomIndex[words[index + 2]])),
123 "SET" => new FlagChange(words[index + 1], true),
124 "CLEAR" => new FlagChange(words[index + 1], false),
125 "POINTS" => new Points(int.Parse(words[index + 1])),
126 _ => throw new InvalidOperationException()
127 };
128 return new(flag, required, effect);
129 }
130 foreach (string[] words in definitions)
131 {
132 switch (words[0])
133 {
134 case "ROOM":
135 {
136 Room room = rooms[roomIndex[words[1]]];
137 room.Dark = words[2] == "DARK";
138 room.Name = string.Join(' ', words.Skip(room.Dark ? 3 : 2));
139 break;
140 }
141 case "DESC": rooms[roomIndex[words[1]]].Descriptions.Add(string.Join(' ', words.Skip(2))); break;
142 case "EXIT":
143 state.Exits[roomIndex[words[1]], Direction(words[2])!.Value] =
144 new Exit(roomIndex[words[3]], words.Length > 4 ? itemIndex[words[5]] : null, words.Length > 4);
145 break;
146 case "ITEM":
147 {
148 int x = itemIndex[words[1]], index = 4;
149 Item item = items[x];
150 state.Places[x] = InitialPlace(words[2]); item.Weight = int.Parse(words[3]);
151 if (words[index] == "TREASURE") { item.Treasure = int.Parse(words[index + 1]); index += 2; }
152 if (words[index] == "LIGHT") { item.Light = true; state.Fuel[x] = int.Parse(words[index + 1]); index += 2; }
153 if (words[index] == "CONTAINER")
154 {
155 item.Capacity = int.Parse(words[index + 1]); index += 2;
156 if (words[index] == "CLOSED") { state.Containers[x] = ContainerState.Closed; index++; }
157 if (words[index] == "LOCKED") { item.Key = itemIndex[words[index + 1]]; state.Containers[x] = ContainerState.Locked; index += 2; }
158 }
159 if (words[index] == "FIXED") { item.Fixed = true; index++; }
160 item.Name = string.Join(' ', words.Skip(index)); item.NameWords.UnionWith(words.Skip(index));
161 break;
162 }
163 case "INSIDE": state.Places[itemIndex[words[1]]] = new(PlaceKind.Inside, itemIndex[words[2]]); break;
164 case "ALIAS": items[itemIndex[words[1]]].Nouns.Add(words[2]); break;
165 case "DETAIL": items[itemIndex[words[1]]].Details.Add(string.Join(' ', words.Skip(2))); break;
166 case "RULE": useRules.Add(new(itemIndex[words[1]], itemIndex[words[2]], ParseRule(words, 3))); break;
167 case "ENTER": rooms[roomIndex[words[1]]].Enter.Add(ParseRule(words, 2)); break;
168 case "TROPHY": trophy = roomIndex[words[1]]; break;
169 case "START": state.Room = roomIndex[words[1]]; break;
170 case "MAXWEIGHT": maxWeight = int.Parse(words[1]); break;
171 }
172 }
173 state.Visited[state.Room] = true;
174 foreach (Item item in items) { vocabulary.UnionWith(item.Nouns); vocabulary.UnionWith(item.NameWords); }
175 }

177 static void Fail(string message) => throw new CommandFailure(message);
178 void Say(string text) => output.Add(text);
179 Place Floor => new(PlaceKind.Room, state.Room);
180 int[] Direct(Place place) => Enumerable.Range(0, items.Length).Where(x => state.Places[x] == place).ToArray();
181 HashSet<int> Reachable(Place place)
182 {
183 HashSet<int> result = [];
184 void Visit(Place source)
185 {
186 foreach (int x in Direct(source))
187 {
188 result.Add(x);
189 if (items[x].Capacity.HasValue && state.Containers[x] == ContainerState.Open) Visit(new(PlaceKind.Inside, x));
190 }
191 }
192 Visit(place);
193 return result;
194 }
195 HashSet<int> Carried() => Reachable(Player);
196 bool IsLit() => !rooms[state.Room].Dark || Reachable(Player).Concat(Reachable(Floor)).Any(x => state.Lit[x]);
197 HashSet<int> Visible()
198 {
199 HashSet<int> result = Carried();
200 if (IsLit()) result.UnionWith(Reachable(Floor));
201 return result;
202 }
203 int TotalWeight(int x) => items[x].Weight + ContentWeight(x);
204 int ContentWeight(int x) => Direct(new(PlaceKind.Inside, x)).Sum(TotalWeight);
205 int CarriedWeight() => Direct(Player).Sum(TotalWeight);
206 bool Within(int x, int container)
207 {
208 while (state.Places[x].Kind == PlaceKind.Inside)
209 {
210 x = state.Places[x].Index;
211 if (x == container) return true;
212 }
213 return false;
214 }
215 bool Stored(int x)
216 {
217 while (state.Places[x].Kind == PlaceKind.Inside) x = state.Places[x].Index;
218 return state.Places[x] == new Place(PlaceKind.Room, trophy);
219 }
220 (int Points, int Stored, int Total) Treasures()
221 {
222 int points = 0, stored = 0, total = 0;
223 for (int x = 0; x < items.Length; x++) if (items[x].Treasure is int value)
224 {
225 total++;
226 if (Stored(x)) { points += value; stored++; }
227 }
228 return (points, stored, total);
229 }
230 int Score => Treasures().Points + state.Bonus;
231 string DisplayName(int x)
232 {
233 List<string> annotations = [];
234 if (items[x].Capacity.HasValue) annotations.Add(state.Containers[x].ToString().ToLowerInvariant());
235 if (state.Lit[x]) annotations.Add("lit");
236 return items[x].Name + (annotations.Count == 0 ? "" : " (" + string.Join(", ", annotations) + ")");
237 }
238 void ItemTree(Place place, int depth)
239 {
240 foreach (int x in Direct(place))
241 {
242 Say(new string(' ', depth * 2) + DisplayName(x));
243 if (items[x].Capacity.HasValue && state.Containers[x] == ContainerState.Open) ItemTree(new(PlaceKind.Inside, x), depth + 1);
244 }

246 void Describe(bool full)
247 {
248 if (!IsLit()) { Say("It is pitch dark."); return; }
249 Room room = rooms[state.Room];
250 Say(room.Name);
251 if (full) output.AddRange(room.Descriptions);
252 List<string> exits = [];
253 for (int d = 0; d < 6; d++) if (state.Exits[state.Room, d] is Exit exit)
254 exits.Add(Directions[d] + (exit.Locked ? " (locked)" : ""));
255 Say("Exits: " + (exits.Count == 0 ? "none" : string.Join(", ", exits)));
256 if (Direct(Floor).Length > 0) { Say("You see:"); ItemTree(Floor, 1); }
257 }
258 int Resolve(string[] phrase)
259 {
260 HashSet<int> visible = Visible();
261 if (phrase.Length == 1 && phrase[0] == "it")
262 {
263 if (!pronoun.HasValue) Fail("I don't know what \"it\" refers to.");
264 if (!visible.Contains(pronoun!.Value)) Fail("You can't see any such thing.");
265 return pronoun.Value;
266 }
267 foreach (string word in phrase) if (!vocabulary.Contains(word)) Fail($"I don't know the word \"{word}\".");
268 int[] matches = Enumerable.Range(0, items.Length).Where(x => visible.Contains(x) && items[x].Nouns.Contains(phrase[^1]) &&
269 phrase.Take(phrase.Length - 1).All(items[x].NameWords.Contains)).ToArray();
270 if (matches.Length == 0) Fail("You can't see any such thing.");
271 if (matches.Length > 1) Fail("Which do you mean: " + string.Join(", ", matches.Select(x => items[x].Name)) + "?");
272 pronoun = matches[0];
273 return matches[0];
274 }
275 void Examine(int x)
276 {
277 Item item = items[x];
278 if (item.Details.Count == 0) Say($"You see nothing special about the {item.Name}."); else output.AddRange(item.Details);
279 if (item.Light) Say($"It is {(state.Lit[x] ? "lit" : "unlit")} (fuel {state.Fuel[x]}).");
280 if (item.Capacity.HasValue)
281 {
282 if (state.Containers[x] == ContainerState.Locked) Say("It is locked.");
283 else if (state.Containers[x] == ContainerState.Closed) Say("It is closed.");
284 else if (Direct(new(PlaceKind.Inside, x)).Length == 0) Say("It is empty.");
285 else { Say("It contains:"); ItemTree(new(PlaceKind.Inside, x), 1); }
286 }
287 if (item.Treasure is int points) Say($"It is a treasure worth {points} points.");
288 Say($"It weighs {TotalWeight(x)}.");
289 }

291 static bool All(string[] phrase) => phrase.Length == 1 && phrase[0] == "all";
292 void RequireContainer(int x)
293 {
294 if (!items[x].Capacity.HasValue) Fail($"The {items[x].Name} is not a container.");
295 if (state.Containers[x] != ContainerState.Open) Fail($"The {items[x].Name} is closed.");
296 }
297 bool Take(string[] target, string[]? from)
298 {
299 if (!IsLit()) Fail("It is pitch dark.");
300 int? container = null;
301 Place source = Floor;
302 if (from is not null)
303 {
304 container = Resolve(from); RequireContainer(container.Value);
305 source = new(PlaceKind.Inside, container.Value);
306 }
307 if (All(target))
308 {
309 int[] candidates = Direct(source).Where(x => !items[x].Fixed).ToArray();
310 if (candidates.Length == 0) Fail(container.HasValue ? $"There is nothing in the {items[container.Value].Name} to take." : "There is nothing here to take.");
311 bool taken = false;
312 foreach (int x in candidates)
313 {
314 if (!Carried().Contains(x) && CarriedWeight() + TotalWeight(x) > maxWeight)
315 Say($"{items[x].Name}: You can't carry any more.");
316 else { state.Places[x] = Player; taken = true; Say($"{items[x].Name}: Taken."); }
317 }
318 return taken;
319 }
320 int item = Resolve(target);
321 if (container.HasValue && state.Places[item] != source) Fail($"The {items[item].Name} is not in the {items[container.Value].Name}.");
322 if (state.Places[item] == Player) Fail($"You already have the {items[item].Name}.");
323 if (items[item].Fixed) Fail($"The {items[item].Name} is fixed in place.");
324 if (!Carried().Contains(item) && CarriedWeight() + TotalWeight(item) > maxWeight) Fail("You can't carry any more.");
325 state.Places[item] = Player; Say("Taken.");
326 return true;
327 }
328 void Drop(string[] target)
329 {
330 if (All(target))
331 {
332 int[] direct = Direct(Player);
333 if (direct.Length == 0) Fail("You are empty-handed.");
334 foreach (int x in direct) { state.Places[x] = Floor; Say($"{items[x].Name}: Dropped."); }
335 return;
336 }
337 int item = Resolve(target);
338 if (!Carried().Contains(item)) Fail($"You aren't carrying the {items[item].Name}.");
339 state.Places[item] = Floor; Say("Dropped.");
340 }
341 string? PutItem(int x, int y)
342 {
343 if (Within(y, x)) return $"The {items[y].Name} is inside the {items[x].Name}.";
344 if (state.Places[x] == new Place(PlaceKind.Inside, y)) return $"The {items[x].Name} is already in the {items[y].Name}.";
345 int weight = ContentWeight(y) + (Within(x, y) ? 0 : TotalWeight(x));
346 if (weight > items[y].Capacity!.Value) return $"The {items[x].Name} doesn't fit in the {items[y].Name}.";
347 state.Places[x] = new(PlaceKind.Inside, y);
348 return null;
349 }
350 bool Put(string[] first, string[] second)
351 {
352 if (All(first))
353 {
354 int y = Resolve(second); RequireContainer(y);
355 int[] candidates = Direct(Player).Where(x => x != y).ToArray();
356 if (candidates.Length == 0) Fail($"You have nothing to put in the {items[y].Name}.");
357 bool moved = false;
358 foreach (int x in candidates)
359 {
360 string? reason = PutItem(x, y);
361 if (reason is null) moved = true;
362 Say(items[x].Name + ": " + (reason ?? "Done."));
363 }
364 return moved;
365 }
366 int item = Resolve(first), container = Resolve(second);
367 if (!Carried().Contains(item)) Fail($"You aren't carrying the {items[item].Name}.");
368 if (item == container) Fail($"You can't put the {items[item].Name} inside itself.");
369 RequireContainer(container);
370 string? failure = PutItem(item, container);
371 if (failure is not null) Fail(failure);
372 Say($"You put the {items[item].Name} in the {items[container].Name}.");
373 return true;
374 }
375 void Open(int x)
376 {
377 string name = items[x].Name;
378 if (!items[x].Capacity.HasValue) Fail($"You can't open the {name}.");
379 if (state.Containers[x] == ContainerState.Open) Fail($"The {name} is already open.");
380 if (state.Containers[x] == ContainerState.Locked) Fail($"The {name} is locked.");
381 state.Containers[x] = ContainerState.Open;
382 int[] children = Direct(new(PlaceKind.Inside, x));
383 Say(children.Length == 0 ? "Opened." : $"Opening the {name} reveals " + string.Join(", ", children.Select(y => items[y].Name)) + ".");
384 }

386 {
387 if (!items[x].Capacity.HasValue) Fail($"You can't close the {items[x].Name}.");
388 if (state.Containers[x] != ContainerState.Open) Fail($"The {items[x].Name} is already closed.");
389 state.Containers[x] = ContainerState.Closed; Say("Closed.");
390 }
391 void CheckKey(int key, string[]? phrase)
392 {
393 if (phrase is not null)
394 {
395 int x = Resolve(phrase);
396 if (!Carried().Contains(x)) Fail($"You aren't carrying the {items[x].Name}.");
397 if (x != key) Fail($"The {items[x].Name} doesn't fit.");
398 }

405 void Lock(string[] target, string[]? keyPhrase, bool locking)
406 {
407 int? direction = target.Length == 1 ? Direction(target[0]) : null;
408 if (direction is int d)
409 {
410 Exit? possible = state.Exits[state.Room, d];
411 if (!possible.HasValue) Fail($"There is no exit {Directions[d]}.");
412 Exit exit = possible!.Value;
413 if (!exit.Key.HasValue) Fail($"The way {Directions[d]} has no lock.");
414 if (exit.Locked == locking) Fail($"The way {Directions[d]} is {(locking ? "already locked" : "not locked")}.");
415 CheckKey(exit.Key!.Value, keyPhrase);
416 state.Exits[state.Room, d] = exit with { Locked = locking };
417 if (state.Exits[exit.Destination, d ^ 1] is Exit opposite && opposite.Destination == state.Room && opposite.Key == exit.Key)
418 state.Exits[exit.Destination, d ^ 1] = opposite with { Locked = locking };
419 }
420 else
421 {
422 int x = Resolve(target);
423 if (!items[x].Capacity.HasValue || !items[x].Key.HasValue) Fail($"The {items[x].Name} has no lock.");
424 if ((state.Containers[x] == ContainerState.Locked) == locking)
425 Fail($"The {items[x].Name} is {(locking ? "already locked" : "not locked")}.");
426 if (locking && state.Containers[x] == ContainerState.Open) Fail($"You must close the {items[x].Name} first.");
427 CheckKey(items[x].Key!.Value, keyPhrase);
428 state.Containers[x] = locking ? ContainerState.Locked : ContainerState.Closed;
429 }
430 Say(locking ? "Locked." : "Unlocked.");
431 }
432 (int Executed, bool Message) RunRules(IEnumerable<Rule> rules)
433 {
434 int executed = 0;
435 bool message = false;
436 foreach (Rule rule in rules)
437 {
438 if (rule.Flag is not null && state.Flags.Contains(rule.Flag) != rule.Required) continue;
439 executed++;
440 switch (rule.Effect)
441 {
442 case Message effect: Say(effect.Text); message = true; break;
443 case OpenExit effect: state.Exits[effect.Room, effect.Direction] = new Exit(effect.Destination, null, false); break;
444 case MoveItem effect: state.Places[effect.Item] = effect.Place; break;
445 case FlagChange effect:
446 if (effect.On) state.Flags.Add(effect.Flag); else state.Flags.Remove(effect.Flag);
447 break;
448 case Points effect: state.Bonus += effect.Amount; break;
449 }
450 }
451 return (executed, message);
452 }
453 void Use(string[] first, string[] second)
454 {
455 int x = Resolve(first), y = Resolve(second);
456 if (!Carried().Contains(x)) Fail($"You aren't carrying the {items[x].Name}.");
457 var result = RunRules(useRules.Where(rule => rule.First == x && rule.Second == y).Select(rule => rule.Rule));
458 if (result.Executed == 0) Fail("Nothing happens.");
459 if (!result.Message) Say("Done.");
460 }
461 void Light(int x, bool lighting)
462 {
463 string verb = lighting ? "light" : "extinguish", name = items[x].Name;
464 if (!items[x].Light) Fail($"You can't {verb} the {name}.");
465 if (state.Lit[x] == lighting) Fail($"The {name} is {(lighting ? "already lit" : "not lit")}.");
466 if (lighting && state.Fuel[x] == 0) Fail($"The {name} has no fuel.");
467 state.Lit[x] = lighting;
468 Say($"The {name} is now {(lighting ? "lit" : "out")}.");
469 }
470 bool Move(int direction)
471 {
472 Exit? possible = state.Exits[state.Room, direction];
473 if (!possible.HasValue) Fail("You can't go that way.");
474 Exit exit = possible!.Value;
475 if (exit.Locked) Fail($"The way {Directions[direction]} is locked.");
476 state.Room = exit.Destination;
477 bool baseline = IsLit();
478 Describe(verbose || !state.Visited[state.Room]);
479 state.Visited[state.Room] = true;
480 RunRules(rooms[state.Room].Enter);
481 return baseline;
482 }
483 void Turn(bool baseline)
484 {
485 state.Moves++;
486 HashSet<int> reachable = Carried(); reachable.UnionWith(Reachable(Floor));
487 for (int x = 0; x < items.Length; x++) if (state.Lit[x])
488 {
489 state.Fuel[x]--;
490 if (state.Fuel[x] == 0)
491 {
492 state.Lit[x] = false;
493 if (reachable.Contains(x)) Say($"The {items[x].Name} goes out.");
494 }
495 }
496 bool lit = IsLit();
497 if (baseline && !lit) Say("It is now pitch dark.");
498 else if (!baseline && lit) Describe(true);
499 state.Darkness = lit ? 0 : state.Darkness + 1;
500 if (state.Darkness == 2) Say("You hear something moving in the dark.");
501 if (state.Darkness == 3) state.End = EndStatus.Dead;
502 var treasure = Treasures();
503 if (state.End == EndStatus.None && treasure.Total > 0 && treasure.Stored == treasure.Total) state.End = EndStatus.Won;
504 if (state.End != EndStatus.None)
505 {
506 Say(state.End == EndStatus.Dead ? "*** You have died ***" : "*** You have won ***");
507 Say($"You scored {treasure.Points + state.Bonus} points in {state.Moves} moves.");
508 }
509 }
510 void Run(Verb verb, string[] first, string[]? second = null, int direction = 0)
511 {
512 bool baseline = IsLit(), success = true;
513 switch (verb)
514 {
515 case Verb.Go: baseline = Move(direction); break;
516 case Verb.Look: Describe(true); break;
517 case Verb.Examine: Examine(Resolve(first)); break;
518 case Verb.Take: success = Take(first, second); break;
519 case Verb.Drop: Drop(first); break;
520 case Verb.Put: success = Put(first, second!); break;
521 case Verb.Open: Open(Resolve(first)); break;
522 case Verb.Close: Close(Resolve(first)); break;
523 case Verb.Unlock: Lock(first, second, false); break;
524 case Verb.Lock: Lock(first, second, true); break;
525 case Verb.Use: Use(first, second!); break;
526 case Verb.Light: Light(Resolve(first), true); break;
527 case Verb.Extinguish: Light(Resolve(first), false); break;
528 case Verb.Wait: Say("Time passes."); break;
529 case Verb.Inventory:
530 if (Direct(Player).Length == 0) Say("You are empty-handed.");
531 else { Say("You are carrying:"); ItemTree(Player, 1); Say($"Total weight: {CarriedWeight()}/{maxWeight}"); }
532 break;
533 case Verb.Score:
534 var treasure = Treasures();
535 Say($"You have scored {treasure.Points + state.Bonus} points (treasure {treasure.Points}, bonus {state.Bonus}) in {state.Moves} moves.");
536 Say($"Treasures stored: {treasure.Stored} of {treasure.Total}.");
537 break;
538 case Verb.Save: saved = state.Copy(); Say("Game saved."); break;
539 case Verb.Restore:
540 if (saved is null) Fail("There is no saved game.");
541 state = saved!.Copy(); Say("Game restored."); Describe(true);
542 break;
543 case Verb.Verbose: verbose = true; Say("Verbose mode."); break;
544 case Verb.Brief: verbose = false; Say("Brief mode."); break;
545 }
546 if (success && verb is Verb.Go or Verb.Take or Verb.Drop or Verb.Put or Verb.Open or Verb.Close or Verb.Unlock or Verb.Lock or Verb.Use or Verb.Light or Verb.Extinguish or Verb.Wait)
547 Turn(baseline);
548 }
549 static string[] Words(string line) => line.ToLowerInvariant().Split(' ', StringSplitOptions.RemoveEmptyEntries)
550 .Where(word => word is not ("the" or "a" or "an")).ToArray();
551 void Interpret(string[] words)
552 {
553 if (words.Length == 0) Fail("I beg your pardon?");
554 if (Direction(words[0]).HasValue) words = ["go", .. words];
555 if (!Verbs.TryGetValue(words[0], out Verb verb)) Fail($"I don't know the verb \"{words[0]}\".");
556 string[] rest = words[1..];
557 if (state.End != EndStatus.None && verb != Verb.Restore) Fail("The game is over.");
558 if (verb == Verb.Look && rest.Length > 0)
559 {
560 if (rest[0] != "at") Fail("I don't understand that.");
561 verb = Verb.Examine; rest = rest[1..];
562 }
563 if (verb is Verb.Look or Verb.Inventory or Verb.Wait or Verb.Score or Verb.Save or Verb.Restore or Verb.Verbose or Verb.Brief)
564 {
565 if (rest.Length != 0) Fail("I don't understand that.");
566 Run(verb, rest); return;
567 }
568 if (rest.Length == 0) Fail(verb == Verb.Go ? "Which way do you want to go?" : $"What do you want to {verb.ToString().ToLowerInvariant()}?");
569 if (verb == Verb.Go)
570 {
571 int? direction = rest.Length == 1 ? Direction(rest[0]) : null;
572 if (!direction.HasValue) Fail("I don't understand that.");
573 Run(verb, rest, direction: direction!.Value); return;
574 }
575 string[] prepositions = verb switch
576 {
577 Verb.Put => ["in", "into"], Verb.Use => ["on", "with"],
578 Verb.Lock or Verb.Unlock when rest.Contains("with") => ["with"],
579 Verb.Take when rest.Contains("from") => ["from"], _ => []
580 };
581 if (prepositions.Length == 0) { Run(verb, rest); return; }
582 int split = Array.FindIndex(rest, prepositions.Contains);
583 if (split <= 0 || split == rest.Length - 1) Fail("I don't understand that.");
584 Run(verb, rest[..split], rest[(split + 1)..]);
585 }
586 public List<string> Start()
587 {
588 output = []; Describe(true); return output;
589 }
590 public List<string> Process(string line)
591 {
592 output = []; Say("> " + line);
593 try
594 {
595 string[] words = Words(line);
596 if (words.Length > 0 && words[0] is "again" or "g")
597 {
598 if (words.Length != 1) Fail("I don't understand that.");
599 if (previous is null) Fail("There is nothing to repeat.");
600 words = Words(previous!);
601 }
602 else previous = line;
603 Interpret(words);
604 }
605 catch (CommandFailure failure) { Say(failure.Message); }
606 return output;
607 }
608 public string FinalScore() => $"Final score: {Score} points in {state.Moves} moves.";
609}
