32type Place = Readonly<{ kind: 'room' | 'inside'; id: string } | { kind: 'player' | 'nowhere' }>;
33const floor = (id: string): Place => ({ kind: 'room', id });
34const inside = (id: string): Place => ({ kind: 'inside', id });
35const player: Place = { kind: 'player' };
36const nowhere: Place = { kind: 'nowhere' };
37const samePlace = (a: Place, b: Place): boolean => a.kind === b.kind && (!('id' in a) || ('id' in b && a.id === b.id));
38const placeKey = (place: Place): string => `${place.kind}:${'id' in place ? place.id : ''}`;

70function readItem(words: readonly string[]): { item: Item; state: ItemState } {
71 const id = words[1], weight = Number(words[3]);
72 let points = 0, light = false, capacity: number | null = null, key: string | null = null, fixed = false, index = 4;
73 const state: ItemState = {
74 place: words[2] === 'PLAYER' ? player : words[2] === 'NOWHERE' ? nowhere : floor(words[2]),
75 container: 'open', lit: false, fuel: 0,
76 };
77 if (words[index] === 'TREASURE') { points = Number(words[index + 1]); index += 2; }
78 if (words[index] === 'LIGHT') { light = true; state.fuel = Number(words[index + 1]); index += 2; }
79 if (words[index] === 'CONTAINER') {
80 capacity = Number(words[index + 1]); index += 2;
81 if (words[index] === 'CLOSED') { state.container = 'closed'; ++index; }
82 if (words[index] === 'LOCKED') { key = words[index + 1]; state.container = 'locked'; index += 2; }
83 }
84 if (words[index] === 'FIXED') { fixed = true; ++index; }
85 return { item: { id, weight, points, light, capacity, key, fixed, name: words.slice(index).join(' '),
86 nouns: new Set([id]), words: new Set(words.slice(index)), details: [] }, state };
87}
88function readRule(words: readonly string[], index: number, trigger: Trigger): Rule {
89 let condition: Condition | null = null;
90 if (words[index] === 'IF' || words[index] === 'UNLESS') {
91 condition = { flag: words[index + 1], on: words[index] === 'IF' }; index += 2;
92 }
93 const args = words.slice(index + 1);
94 let effect: Effect;
95 switch (words[index]) {
96 case 'MESSAGE': effect = { kind: 'message', text: args.join(' ') }; break;
97 case 'OPEN': effect = { kind: 'open', room: args[0], direction: directionWords.get(args[1])!, destination: args[2] }; break;
98 case 'REMOVE': effect = { kind: 'move', item: args[0], place: nowhere }; break;
99 case 'ADD': effect = { kind: 'move', item: args[0], place: floor(args[1]) }; break;
100 case 'SET': case 'CLEAR': effect = { kind: 'flag', flag: args[0], on: words[index] === 'SET' }; break;
101 default: effect = { kind: 'points', amount: Number(args[0]) }; break;
102 }
103 return { trigger, condition, effect };
104}

121 constructor(lines: readonly string[]) {
122 for (const line of lines) this.define(line.split(' '));
123 this.state.visited.add(this.state.room);
124 this.ids = [...this.items.keys()].sort();
125 this.vocabulary = new Set([...this.items.values()].flatMap(item => [...item.nouns, ...item.words]));

127 private item(id: string): Item { return this.items.get(id)!; }
128 private itemState(id: string): ItemState { return this.state.items.get(id)!; }
129 private define(words: readonly string[]): void {
130 const [kind, id] = words;
131 switch (kind) {
132 case 'ROOM': {
133 const dark = words[2] === 'DARK';
134 this.rooms.set(id, { dark, name: words.slice(dark ? 3 : 2).join(' '), description: [] }); break;
135 }
136 case 'DESC': this.rooms.get(id)!.description.push(words.slice(2).join(' ')); break;
137 case 'EXIT': this.setExit(id, directionWords.get(words[2])!, words[3], words.length > 4 ? words[5] : null); break;
138 case 'ITEM': {
139 const { item, state } = readItem(words);
140 this.items.set(id, item); this.state.items.set(id, state); break;
141 }
142 case 'INSIDE': this.itemState(id).place = inside(words[2]); break;
143 case 'ALIAS': this.item(id).nouns.add(words[2]); break;
144 case 'DETAIL': this.item(id).details.push(words.slice(2).join(' ')); break;
145 case 'RULE': this.rules.push(readRule(words, 3, { kind: 'use', first: id, second: words[2] })); break;
146 case 'ENTER': this.rules.push(readRule(words, 2, { kind: 'enter', room: id })); break;
147 case 'TROPHY': this.trophy = id; break;
148 case 'START': this.state.room = id; break;
149 case 'MAXWEIGHT': this.maxWeight = Number(id); break;
150 }
151 }
152 private setExit(room: string, direction: Direction, destination: string, key: string | null = null): void {
153 let exits = this.state.exits.get(room);
154 if (exits === undefined) { exits = new Map(); this.state.exits.set(room, exits); }
155 exits.set(direction, { destination, key, locked: key !== null });
156 }
157 private contents(place: Place): string[] { return this.ids.filter(id => samePlace(this.itemState(id).place, place)); }
158 private children(): Map<string, string[]> {
159 const result = new Map<string, string[]>();
160 for (const id of this.ids) {
161 const key = placeKey(this.itemState(id).place);
162 let children = result.get(key);
163 if (children === undefined) { children = []; result.set(key, children); }
164 children.push(id);
165 }
166 return result;
167 }
168 private isOpen(id: string): boolean { return this.item(id).capacity !== null && this.itemState(id).container === 'open'; }
169 private reachable(place: Place): Set<string> {
170 const result = new Set<string>(), children = this.children();
171 const visit = (place: Place): void => {
172 for (const id of children.get(placeKey(place)) ?? []) {
173 result.add(id);
174 if (this.isOpen(id)) visit(inside(id));
175 }
176 };
177 visit(place);
178 return result;
179 }
180 private carried(): Set<string> { return this.reachable(player); }
181 private roomItems(): Set<string> { return this.reachable(floor(this.state.room)); }
182 private isLit(): boolean {
183 return !this.rooms.get(this.state.room)!.dark || [...this.carried(), ...this.roomItems()].some(id => this.itemState(id).lit);
184 }
185 private visible(): Set<string> { return this.isLit() ? new Set([...this.carried(), ...this.roomItems()]) : this.carried(); }
186 private weights(): Map<string, number> {
187 const result = new Map<string, number>(), children = this.children();
188 const total = (id: string): number => {
189 const cached = result.get(id);
190 if (cached !== undefined) return cached;
191 const weight = this.item(id).weight + (children.get(placeKey(inside(id))) ?? []).reduce((sum, child) => sum + total(child), 0);
192 result.set(id, weight);
193 return weight;
194 };
195 for (const id of this.ids) total(id);
196 return result;
197 }
198 private totalWeight(id: string): number { return this.weights().get(id)!; }
199 private contentWeight(id: string): number { return this.totalWeight(id) - this.item(id).weight; }
200 private carriedWeight(): number {
201 const weights = this.weights();
202 return this.contents(player).reduce((sum, id) => sum + weights.get(id)!, 0);
203 }
204 private nestedIn(id: string, ancestor: string): boolean {
205 let place = this.itemState(id).place;
206 while (place.kind === 'inside') {
207 if (place.id === ancestor) return true;
208 place = this.itemState(place.id).place;
209 }
210 return false;
211 }
212 score(): Score {
213 let stored = 0, treasures = 0, points = 0;
214 for (const id of this.ids) if (this.item(id).points > 0) {
215 ++treasures;
216 let place = this.itemState(id).place;
217 while (place.kind === 'inside') place = this.itemState(place.id).place;
218 if (samePlace(place, floor(this.trophy))) { ++stored; points += this.item(id).points; }
219 }
220 return { stored, treasures, points, total: points + this.state.bonus };
221 }
222 private display(id: string): string {
223 const item = this.item(id), state = this.itemState(id), annotations: string[] = [];
224 if (item.capacity !== null) annotations.push(state.container);
225 if (item.light && state.lit) annotations.push('lit');
226 return item.name + (annotations.length ? ` (${annotations.join(', ')})` : '');
227 }
228 private tree(place: Place, depth = 1, children: ReadonlyMap<string, readonly string[]> = this.children()): string[] {
229 const rows: string[] = [];
230 for (const id of children.get(placeKey(place)) ?? []) {
231 rows.push(' '.repeat(depth) + this.display(id));
232 if (this.isOpen(id)) rows.push(...this.tree(inside(id), depth + 1, children));
233 }
234 return rows;
235 }
236 describe(full = true): string[] {
237 if (!this.isLit()) return ['It is pitch dark.'];
238 const room = this.rooms.get(this.state.room)!, exits = this.state.exits.get(this.state.room), names: string[] = [];
239 for (const direction of directions) {
240 const exit = exits?.get(direction);
241 if (exit !== undefined) names.push(direction + (exit.locked ? ' (locked)' : ''));
242 }
243 const rows = [room.name, ...(full ? room.description : []), 'Exits: ' + (names.length ? names.join(', ') : 'none')];
244 if (this.contents(floor(this.state.room)).length) rows.push('You see:', ...this.tree(floor(this.state.room)));
245 return rows;
246 }
247 private resolve(phrase: Phrase): string {
248 const visible = this.visible();
249 if (phrase.length === 1 && phrase[0] === 'it') {
250 if (this.it === null) fail('I don\'t know what "it" refers to.');
251 if (!visible.has(this.it)) fail("You can't see any such thing.");
252 return this.it;
253 }
254 for (const word of phrase) if (!this.vocabulary.has(word)) fail(`I don't know the word "${word}".`);
255 const matches = this.ids.filter(id => visible.has(id) && this.item(id).nouns.has(phrase[phrase.length - 1]) &&
256 phrase.slice(0, -1).every(word => this.item(id).words.has(word)));
257 if (!matches.length) fail("You can't see any such thing.");
258 if (matches.length > 1) fail(`Which do you mean: ${matches.map(id => this.item(id).name).join(', ')}?`);
259 this.it = matches[0];
260 return matches[0];
261 }
262 private runRules(trigger: Trigger): RuleResult {
263 const rows: string[] = [];
264 let executed = 0;
265 for (const rule of this.rules) {
266 const target = rule.trigger;
267 if (trigger.kind === 'enter') {
268 if (target.kind !== 'enter' || target.room !== trigger.room) continue;
269 } else if (target.kind !== 'use' || target.first !== trigger.first || target.second !== trigger.second) continue;
270 if (rule.condition !== null && this.state.flags.has(rule.condition.flag) !== rule.condition.on) continue;
271 ++executed;
272 const effect = rule.effect;
273 switch (effect.kind) {
274 case 'message': rows.push(effect.text); break;
275 case 'open': this.setExit(effect.room, effect.direction, effect.destination); break;
276 case 'move': this.itemState(effect.item).place = effect.place; break;
277 case 'flag': if (effect.on) this.state.flags.add(effect.flag); else this.state.flags.delete(effect.flag); break;
278 case 'points': this.state.bonus += effect.amount; break;
279 }
280 }
281 return { rows, executed };
282 }
283 private requireContainer(id: string): number {
284 const item = this.item(id);
285 if (item.capacity === null) fail(`The ${item.name} is not a container.`);
286 if (!this.isOpen(id)) fail(`The ${item.name} is closed.`);
287 return item.capacity;
288 }
289 private take(first: Phrase, second: Phrase | null): string[] {
290 if (!this.isLit()) fail('It is pitch dark.');
291 let container: string | null = null, source = floor(this.state.room);
292 if (second !== null) { container = this.resolve(second); this.requireContainer(container); source = inside(container); }
293 if (allItems(first)) {
294 const candidates = this.contents(source).filter(id => !this.item(id).fixed), rows: string[] = [];
295 if (!candidates.length) fail(container === null ? 'There is nothing here to take.' : `There is nothing in the ${this.item(container).name} to take.`);
296 let taken = false;
297 for (const id of candidates) {
298 let outcome = "You can't carry any more.";
299 if (this.carried().has(id) || this.carriedWeight() + this.totalWeight(id) <= this.maxWeight) {
300 this.itemState(id).place = player; outcome = 'Taken.'; taken = true;
301 }
302 rows.push(`${this.item(id).name}: ${outcome}`);
303 }
304 if (!taken) fail(rows);
305 return rows;
306 }
307 const id = this.resolve(first), item = this.item(id), state = this.itemState(id);
308 if (container !== null && !samePlace(state.place, source)) fail(`The ${item.name} is not in the ${this.item(container).name}.`);
309 if (state.place.kind === 'player') fail(`You already have the ${item.name}.`);
310 if (item.fixed) fail(`The ${item.name} is fixed in place.`);
311 if (!this.carried().has(id) && this.carriedWeight() + this.totalWeight(id) > this.maxWeight) fail("You can't carry any more.");
312 state.place = player;
313 return ['Taken.'];
314 }
315 private drop(target: Phrase): string[] {
316 if (allItems(target)) {
317 const candidates = this.contents(player);
318 if (!candidates.length) fail('You are empty-handed.');
319 return candidates.map(id => { this.itemState(id).place = floor(this.state.room); return `${this.item(id).name}: Dropped.`; });
320 }
321 const id = this.resolve(target);
322 if (!this.carried().has(id)) fail(`You aren't carrying the ${this.item(id).name}.`);
323 this.itemState(id).place = floor(this.state.room);
324 return ['Dropped.'];
325 }
326 private putItem(id: string, container: string, capacity: number): string | null {
327 const item = this.item(id), target = this.item(container), state = this.itemState(id);
328 if (this.nestedIn(container, id)) return `The ${target.name} is inside the ${item.name}.`;
329 if (samePlace(state.place, inside(container))) return `The ${item.name} is already in the ${target.name}.`;
330 const weight = this.contentWeight(container) + (this.nestedIn(id, container) ? 0 : this.totalWeight(id));
331 if (weight > capacity) return `The ${item.name} doesn't fit in the ${target.name}.`;
332 state.place = inside(container);
333 return null;
334 }
335 private put(first: Phrase, second: Phrase): string[] {
336 if (allItems(first)) {
337 const target = this.resolve(second), capacity = this.requireContainer(target);
338 const candidates = this.contents(player).filter(id => id !== target), rows: string[] = [];
339 if (!candidates.length) fail(`You have nothing to put in the ${this.item(target).name}.`);
340 let succeeded = false;
341 for (const id of candidates) {
342 const reason = this.putItem(id, target, capacity);
343 if (reason === null) succeeded = true;
344 rows.push(`${this.item(id).name}: ${reason ?? 'Done.'}`);
345 }
346 if (!succeeded) fail(rows);
347 return rows;
348 }
349 const id = this.resolve(first), target = this.resolve(second), item = this.item(id);
350 if (!this.carried().has(id)) fail(`You aren't carrying the ${item.name}.`);
351 if (id === target) fail(`You can't put the ${item.name} inside itself.`);
352 const capacity = this.requireContainer(target), reason = this.putItem(id, target, capacity);
353 if (reason !== null) fail(reason);
354 return [`You put the ${item.name} in the ${this.item(target).name}.`];
355 }
356 private keyCheck(key: string, phrase: Phrase | null): string[] {
357 if (phrase !== null) {
358 const id = this.resolve(phrase), name = this.item(id).name;
359 if (!this.carried().has(id)) fail(`You aren't carrying the ${name}.`);
360 if (id !== key) fail(`The ${name} doesn't fit.`);
361 return [];
362 }
363 if (!this.carried().has(key)) fail("You don't have the right key.");
364 return [`(with the ${this.item(key).name})`];
365 }
366 private changeLock(first: Phrase, second: Phrase | null, locking: boolean): string[] {
367 const direction = first.length === 1 ? directionWords.get(first[0]) : undefined;
368 if (direction !== undefined) {
369 const exit = this.state.exits.get(this.state.room)?.get(direction);
370 if (exit === undefined) fail(`There is no exit ${direction}.`);
371 if (exit.key === null) fail(`The way ${direction} has no lock.`);
372 if (exit.locked === locking) fail(`The way ${direction} is ${locking ? 'already locked' : 'not locked'}.`);
373 const rows = this.keyCheck(exit.key, second);
374 exit.locked = locking;
375 const opposite = directions[directions.indexOf(direction) ^ 1], back = this.state.exits.get(exit.destination)?.get(opposite);
376 if (back !== undefined && back.destination === this.state.room && back.key === exit.key) back.locked = locking;
377 return [...rows, locking ? 'Locked.' : 'Unlocked.'];
378 }
379 const id = this.resolve(first), item = this.item(id), state = this.itemState(id);
380 if (item.capacity === null || item.key === null) fail(`The ${item.name} has no lock.`);
381 if ((state.container === 'locked') === locking) fail(`The ${item.name} is ${locking ? 'already locked' : 'not locked'}.`);
382 if (locking && state.container === 'open') fail(`You must close the ${item.name} first.`);
383 const rows = this.keyCheck(item.key, second);
384 state.container = locking ? 'locked' : 'closed';
385 return [...rows, locking ? 'Locked.' : 'Unlocked.'];
386 }
387 private itemAction(verb: ItemVerb, target: Phrase): string[] {
388 const id = this.resolve(target), item = this.item(id), state = this.itemState(id);
389 switch (verb) {
390 case 'examine': {
391 const rows = item.details.length ? item.details.slice() : [`You see nothing special about the ${item.name}.`];
392 if (item.light) rows.push(`It is ${state.lit ? 'lit' : 'unlit'} (fuel ${state.fuel}).`);
393 if (item.capacity !== null) {
394 if (state.container !== 'open') rows.push(`It is ${state.container}.`);
395 else if (this.contents(inside(id)).length) rows.push('It contains:', ...this.tree(inside(id)));
396 else rows.push('It is empty.');
397 }
398 if (item.points > 0) rows.push(`It is a treasure worth ${item.points} points.`);
399 rows.push(`It weighs ${this.totalWeight(id)}.`);
400 return rows;
401 }
402 case 'open': {
403 if (item.capacity === null) fail(`You can't open the ${item.name}.`);
404 if (state.container === 'open') fail(`The ${item.name} is already open.`);
405 if (state.container === 'locked') fail(`The ${item.name} is locked.`);
406 state.container = 'open';
407 const contents = this.contents(inside(id));
408 return contents.length ? [`Opening the ${item.name} reveals ${contents.map(child => this.item(child).name).join(', ')}.`] : ['Opened.'];
409 }
410 case 'close':
411 if (item.capacity === null) fail(`You can't close the ${item.name}.`);
412 if (state.container !== 'open') fail(`The ${item.name} is already closed.`);
413 state.container = 'closed'; return ['Closed.'];
414 case 'light':
415 if (!item.light) fail(`You can't light the ${item.name}.`);
416 if (state.lit) fail(`The ${item.name} is already lit.`);
417 if (state.fuel === 0) fail(`The ${item.name} has no fuel.`);
418 state.lit = true; return [`The ${item.name} is now lit.`];
419 case 'extinguish':
420 if (!item.light) fail(`You can't extinguish the ${item.name}.`);
421 if (!state.lit) fail(`The ${item.name} is not lit.`);
422 state.lit = false; return [`The ${item.name} is now out.`];
423 }
424 }
425 private execute(command: Exclude<Command, { verb: 'go' }>): string[] {
426 switch (command.verb) {
427 case 'look': return this.describe();
428 case 'inventory':
429 return this.contents(player).length ? ['You are carrying:', ...this.tree(player), `Total weight: ${this.carriedWeight()}/${this.maxWeight}`] : ['You are empty-handed.'];
430 case 'wait': return ['Time passes.'];
431 case 'score': {
432 const score = this.score();
433 return [`You have scored ${score.total} points (treasure ${score.points}, bonus ${this.state.bonus}) in ${this.state.moves} moves.`,
434 `Treasures stored: ${score.stored} of ${score.treasures}.`];
435 }
436 case 'save': this.saved = structuredClone(this.state); return ['Game saved.'];
437 case 'restore':
438 if (this.saved === null) fail('There is no saved game.');
439 this.state = structuredClone(this.saved); return ['Game restored.', ...this.describe()];
440 case 'verbose': this.verbose = true; return ['Verbose mode.'];
441 case 'brief': this.verbose = false; return ['Brief mode.'];
442 case 'take': return this.take(command.target, command.from);
443 case 'drop': return this.drop(command.target);
444 case 'put': return this.put(command.first, command.second);
445 case 'unlock': case 'lock': return this.changeLock(command.target, command.key, command.verb === 'lock');
446 case 'use': {
447 const first = this.resolve(command.first), second = this.resolve(command.second);
448 if (!this.carried().has(first)) fail(`You aren't carrying the ${this.item(first).name}.`);
449 const result = this.runRules({ kind: 'use', first, second });
450 if (result.executed === 0) fail('Nothing happens.');
451 return result.rows.length ? result.rows : ['Done.'];
452 }
453 case 'examine': case 'open': case 'close': case 'light': case 'extinguish': return this.itemAction(command.verb, command.target);
454 }
455 }
456 private endTurn(baseline: boolean): string[] {
457 const rows: string[] = [];
458 ++this.state.moves;
459 const nearby = new Set([...this.carried(), ...this.roomItems()]);
460 for (const id of this.ids) {
461 const state = this.itemState(id);
462 if (state.lit && --state.fuel === 0) {
463 state.lit = false;
464 if (nearby.has(id)) rows.push(`The ${this.item(id).name} goes out.`);
465 }
466 }
467 const lit = this.isLit();
468 if (baseline && !lit) rows.push('It is now pitch dark.');
469 else if (!baseline && lit) rows.push(...this.describe());
470 this.state.darkness = lit ? 0 : this.state.darkness + 1;
471 if (this.state.darkness === 2) rows.push('You hear something moving in the dark.');
472 if (this.state.darkness === 3) this.state.ended = 'dead';
473 const score = this.score();
474 if (this.state.ended === 'none' && score.treasures > 0 && score.stored === score.treasures) this.state.ended = 'won';
475 if (this.state.ended !== 'none') {
476 rows.push(this.state.ended === 'dead' ? '*** You have died ***' : '*** You have won ***');
477 rows.push(`You scored ${score.total} points in ${this.state.moves} moves.`);
478 }
479 return rows;
480 }
481 private parse(line: string): Command {
482 let words = normalize(line);
483 if (!words.length) fail('I beg your pardon?');
484 if (directionWords.has(words[0])) words = ['go', ...words];
485 let verb = verbs.get(words[0]), rest = words.slice(1);
486 if (verb === undefined) fail(`I don't know the verb "${words[0]}".`);
487 if (this.state.ended !== 'none' && verb !== 'restore') fail('The game is over.');
488 if (verb === 'look' && rest.length) {
489 if (rest[0] !== 'at') fail("I don't understand that.");
490 verb = 'examine'; rest = rest.slice(1);
491 }
492 if (hasNoObjects(verb)) {
493 if (rest.length) fail("I don't understand that.");
494 return { verb };
495 }
496 if (!rest.length) fail(verb === 'go' ? 'Which way do you want to go?' : `What do you want to ${verb}?`);
497 if (verb === 'go') {
498 const direction = rest.length === 1 ? directionWords.get(rest[0]) : undefined;
499 if (direction === undefined) fail("I don't understand that.");
500 return { verb, direction };
501 }
502 let first = rest, second: string[] | null = null;
503 const prepositions = verb === 'put' ? ['in', 'into'] : verb === 'use' ? ['on', 'with'] :
504 verb === 'lock' || verb === 'unlock' ? ['with'] : verb === 'take' ? ['from'] : [];
505 const index = rest.findIndex(word => prepositions.includes(word));
506 if (index >= 0) {
507 if (index === 0 || index === rest.length - 1) fail("I don't understand that.");
508 first = rest.slice(0, index); second = rest.slice(index + 1);
509 }
510 if (verb === 'put' || verb === 'use') {
511 if (second === null) fail("I don't understand that.");
512 return { verb, first, second };
513 }
514 if (verb === 'take') return { verb, target: first, from: second };
515 if (verb === 'lock' || verb === 'unlock') return { verb, target: first, key: second };
516 return { verb, target: first };
517 }
518 private interpret(line: string): string[] {
519 const command = this.parse(line);
520 let baseline = this.isLit(), rows: string[];
521 if (command.verb === 'go') {
522 const exit = this.state.exits.get(this.state.room)?.get(command.direction);
523 if (exit === undefined) fail("You can't go that way.");
524 if (exit.locked) fail(`The way ${command.direction} is locked.`);
525 this.state.room = exit.destination;
526 baseline = this.isLit();
527 rows = this.describe(this.verbose || !this.state.visited.has(this.state.room));
528 this.state.visited.add(this.state.room);
529 rows.push(...this.runRules({ kind: 'enter', room: this.state.room }).rows);
530 } else rows = this.execute(command);
531 if (actions.has(command.verb)) rows.push(...this.endTurn(baseline));
532 return rows;
533 }
534 command(line: string): readonly string[] {
535 try {
536 const words = normalize(line);
537 if (words[0] === 'again' || words[0] === 'g') {
538 if (words.length !== 1) fail("I don't understand that.");
539 if (this.previous === null) fail('There is nothing to repeat.');
540 return this.interpret(this.previous);
541 }
542 this.previous = line;
543 return this.interpret(line);
544 } catch (error) {
545 if (!(error instanceof Failure)) throw error;
546 return error.lines;
547 }
548 }
