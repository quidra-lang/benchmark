63static void split_words(const char *line, Words *result) {
64 strcpy(result->storage, line);
65 result->count = 0;
66 for (char *word = strtok(result->storage, " "); word; word = strtok(NULL, " "))
67 result->words[result->count++] = word;
68}

70static void normalize(const char *line, Words *result) {
71 split_words(line, result);
72 int count = 0;
73 for (int i = 0; i < result->count; ++i) {
74 char *word = result->words[i];
75 for (char *p = word; *p; ++p) if (*p >= 'A' && *p <= 'Z') *p += 'a' - 'A';
76 if (strcmp(word, "the") && strcmp(word, "a") && strcmp(word, "an")) result->words[count++] = word;
77 }
78 result->count = count;
79}

107static Effect read_effect(const char *line, const Words *words, int at) {
108 char *const *w = words->words;
109 Effect effect = {0};
110 if (!strcmp(w[at], "MESSAGE")) { effect.kind = MESSAGE; effect.text = remaining(line, words, at + 1); }
111 else if (!strcmp(w[at], "OPEN")) {
112 effect.kind = OPEN_EXIT; effect.a = room_id(w[at + 1]); effect.b = direction(w[at + 2]); effect.c = room_id(w[at + 3]);
113 } else if (!strcmp(w[at], "REMOVE")) { effect.kind = REMOVE_ITEM; effect.a = item_id(w[at + 1]); }
114 else if (!strcmp(w[at], "ADD")) { effect.kind = ADD_ITEM; effect.a = item_id(w[at + 1]); effect.b = room_id(w[at + 2]); }
115 else if (!strcmp(w[at], "SET")) { effect.kind = SET_FLAG; effect.a = flag_id(w[at + 1]); }
116 else if (!strcmp(w[at], "CLEAR")) { effect.kind = CLEAR_FLAG; effect.a = flag_id(w[at + 1]); }
117 else { effect.kind = POINTS; effect.a = atoi(w[at + 1]); }
118 return effect;
119}

125static void build_world(void) {
126 for (int i = 0; i < definition_count; ++i) {
127 Words words; split_words(definitions[i], &words);
128 if (!strcmp(words.words[0], "ROOM")) strcpy(rooms[room_count++].id, words.words[1]);
129 else if (!strcmp(words.words[0], "ITEM")) strcpy(items[item_count++].id, words.words[1]);
130 }
131 for (int r = 0; r < room_count; ++r) for (int d = 0; d < 6; ++d) state.exits[r][d] = (Exit){-1, -1, false};
132 for (int i = 0; i < item_count; ++i) { state.places[i] = (Place){NOWHERE, 0}; items[i].key = -1; ordered[i] = i; }
133 for (int i = 0; i < definition_count; ++i) {
134 const char *line = definitions[i];
135 Words words; split_words(line, &words); char **w = words.words;
136 if (!strcmp(w[0], "ROOM")) {
137 Room *room = &rooms[room_id(w[1])]; room->dark = !strcmp(w[2], "DARK");
138 room->name = remaining(line, &words, room->dark ? 3 : 2);
139 } else if (!strcmp(w[0], "DESC")) {
140 TextLines *text = &rooms[room_id(w[1])].description;
141 text->lines[text->count++] = remaining(line, &words, 2);
142 } else if (!strcmp(w[0], "EXIT")) {
143 bool locked = words.count > 4;
144 state.exits[room_id(w[1])][direction(w[2])] = (Exit){room_id(w[3]), locked ? item_id(w[5]) : -1, locked};
145 } else if (!strcmp(w[0], "ITEM")) {
146 int id = item_id(w[1]), at = 4; Item *item = &items[id];
147 state.places[id] = !strcmp(w[2], "PLAYER") ? (Place){PLAYER, 0} : !strcmp(w[2], "NOWHERE") ? (Place){NOWHERE, 0} : (Place){FLOOR, room_id(w[2])};
148 item->weight = atoi(w[3]);
149 if (!strcmp(w[at], "TREASURE")) { item->points = atoi(w[at + 1]); at += 2; }
150 if (!strcmp(w[at], "LIGHT")) { item->light = true; state.fuel[id] = atoi(w[at + 1]); at += 2; }
151 if (!strcmp(w[at], "CONTAINER")) {
152 item->container = true; item->capacity = atoi(w[at + 1]); at += 2;
153 if (!strcmp(w[at], "CLOSED")) { state.containers[id] = CLOSED; ++at; }
154 if (!strcmp(w[at], "LOCKED")) { item->key = item_id(w[at + 1]); state.containers[id] = LOCKED; at += 2; }
155 }
156 if (!strcmp(w[at], "FIXED")) { item->fixed = true; ++at; }
157 item->name = remaining(line, &words, at);
158 } else if (!strcmp(w[0], "INSIDE")) state.places[item_id(w[1])] = (Place){INSIDE, item_id(w[2])};
159 else if (!strcmp(w[0], "ALIAS")) {
160 aliases[alias_count].item = item_id(w[1]); strcpy(aliases[alias_count++].word, w[2]);
161 } else if (!strcmp(w[0], "DETAIL")) {
162 TextLines *text = &items[item_id(w[1])].details;
163 text->lines[text->count++] = remaining(line, &words, 2);
164 } else if (!strcmp(w[0], "RULE") || !strcmp(w[0], "ENTER")) {
165 Rule rule = {0}; rule.entry = !strcmp(w[0], "ENTER");
166 rule.first = rule.entry ? room_id(w[1]) : item_id(w[1]);
167 int at = rule.entry ? 2 : 3;
168 if (!rule.entry) rule.second = item_id(w[2]);
169 if (!strcmp(w[at], "IF") || !strcmp(w[at], "UNLESS")) {
170 rule.condition = !strcmp(w[at], "IF") ? IF_FLAG : UNLESS_FLAG;
171 rule.flag = flag_id(w[at + 1]); at += 2;
172 }
173 rule.effect = read_effect(line, &words, at); rules[rule_count++] = rule;
174 } else if (!strcmp(w[0], "TROPHY")) trophy = room_id(w[1]);
175 else if (!strcmp(w[0], "START")) state.room = room_id(w[1]);
176 else if (!strcmp(w[0], "MAXWEIGHT")) max_weight = atoi(w[1]);
177 }
178 state.visited[state.room] = true;
179 qsort(ordered, item_count, sizeof(*ordered), compare_items);
180}

184static bool reachable(int item, Place root) {
185 Place place = state.places[item];
186 while (place.kind == INSIDE) {
187 if (state.containers[place.index] != OPEN) return false;
188 place = state.places[place.index];
189 }
190 return same_place(place, root);
191}

281static bool name_word(const char *name, const char *word) {
282 size_t length = strlen(word);
283 for (const char *p = name; *p;) {
284 const char *end = strchr(p, ' ');
285 size_t n = end ? (size_t)(end - p) : strlen(p);
286 if (length == n && !strncmp(p, word, n)) return true;
287 if (!end) break;
288 p = end + 1;
289 }
290 return false;
291}

304static int resolve(Phrase phrase) {
305 bool lit = room_lit();
306 if (phrase.count == 1 && !strcmp(phrase.words[0], "it")) {
307 if (pronoun < 0) fail("I don't know what \"it\" refers to.");
308 if (!carried(pronoun) && !(lit && room_item(pronoun))) fail("You can't see any such thing.");
309 return pronoun;
310 }
311 for (int i = 0; i < phrase.count; ++i) if (!known_word(phrase.words[i])) fail("I don't know the word \"%s\".", phrase.words[i]);
312 int candidates[MAX_DEFINITIONS], count = 0;
313 for (int i = 0; i < item_count; ++i) {
314 int item = ordered[i];
315 if ((!carried(item) && !(lit && room_item(item))) || !noun(item, phrase.words[phrase.count - 1])) continue;
316 bool matches = true;
317 for (int j = 0; j + 1 < phrase.count; ++j) if (!name_word(items[item].name, phrase.words[j])) matches = false;
318 if (matches) candidates[count++] = item;
319 }
320 if (!count) fail("You can't see any such thing.");
321 if (count > 1) {
322 fputs("Which do you mean: ", stdout);
323 for (int i = 0; i < count; ++i) { if (i) fputs(", ", stdout); fputs(items[candidates[i]].name, stdout); }
324 puts("?"); longjmp(command_error, 1);
325 }
326 return pronoun = candidates[0];
327}

329static int execute_rules(bool entry, int first, int second, bool *message) {
330 int executed = 0;
331 for (int i = 0; i < rule_count; ++i) {
332 const Rule *rule = &rules[i];
333 if (rule->entry != entry || rule->first != first || (!entry && rule->second != second)) continue;
334 if ((rule->condition == IF_FLAG && !state.flags[rule->flag]) || (rule->condition == UNLESS_FLAG && state.flags[rule->flag])) continue;
335 ++executed;
336 Effect effect = rule->effect;
337 switch (effect.kind) {
338 case MESSAGE: puts(effect.text); *message = true; break;
339 case OPEN_EXIT: state.exits[effect.a][effect.b] = (Exit){effect.c, -1, false}; break;
340 case REMOVE_ITEM: state.places[effect.a] = (Place){NOWHERE, 0}; break;
341 case ADD_ITEM: state.places[effect.a] = (Place){FLOOR, effect.b}; break;
342 case SET_FLAG: state.flags[effect.a] = true; break;
343 case CLEAR_FLAG: state.flags[effect.a] = false; break;
344 case POINTS: state.bonus += effect.a; break;
345 }
346 }
347 return executed;
348}

350static void pass_turn(bool baseline) {
351 ++state.moves;
352 bool nearby[MAX_DEFINITIONS];
353 for (int i = 0; i < item_count; ++i) nearby[i] = carried(i) || room_item(i);
354 for (int i = 0; i < item_count; ++i) {
355 int item = ordered[i];
356 if (state.lit[item] && --state.fuel[item] == 0) {
357 state.lit[item] = false;
358 if (nearby[item]) printf("The %s goes out.\n", items[item].name);
359 }
360 }
361 bool lit = room_lit();
362 if (baseline && !lit) puts("It is now pitch dark.");
363 else if (!baseline && lit) describe(true);
364 state.darkness = lit ? 0 : state.darkness + 1;
365 if (state.darkness == 2) puts("You hear something moving in the dark.");
366 if (state.darkness == 3) state.ended = DEAD;
367 if (state.ended == NONE) {
368 int treasures = 0, count = 0;
369 for (int i = 0; i < item_count; ++i) if (items[i].points) { ++treasures; if (stored(i)) ++count; }
370 if (treasures > 0 && count == treasures) state.ended = WON;
371 }
372 if (state.ended != NONE) {
373 puts(state.ended == DEAD ? "*** You have died ***" : "*** You have won ***");
374 printf("You scored %d points in %d moves.\n", score(), state.moves);
375 }
376}

380static bool put_item(int item, int container, char *reason) {
381 if (nested_in(container, item)) {
382 snprintf(reason, 256, "The %s is inside the %s.", items[container].name, items[item].name);
383 return false;
384 }
385 if (same_place(state.places[item], (Place){INSIDE, container})) {
386 snprintf(reason, 256, "The %s is already in the %s.", items[item].name, items[container].name);
387 return false;
388 }
389 int weight = place_weight((Place){INSIDE, container});
390 if (!nested_in(item, container)) weight += total_weight(item);
391 if (weight > items[container].capacity) {
392 snprintf(reason, 256, "The %s doesn't fit in the %s.", items[item].name, items[container].name);
393 return false;
394 }
395 state.places[item] = (Place){INSIDE, container};
396 return true;
397}

399static bool take_items(Phrase target, Phrase from) {
400 if (!room_lit()) fail("It is pitch dark.");
401 Place source = {FLOOR, state.room};
402 int container = -1;
403 if (from.count) {
404 container = resolve(from);
405 if (!items[container].container) fail("The %s is not a container.", items[container].name);
406 if (state.containers[container] != OPEN) fail("The %s is closed.", items[container].name);
407 source = (Place){INSIDE, container};
408 }
409 if (all_form(target)) {
410 int candidates[MAX_DEFINITIONS], length = children(source, candidates), count = 0;
411 for (int i = 0; i < length; ++i) if (!items[candidates[i]].fixed) candidates[count++] = candidates[i];
412 if (!count) {
413 if (container < 0) fail("There is nothing here to take.");
414 fail("There is nothing in the %s to take.", items[container].name);
415 }
416 bool success = false;
417 for (int i = 0; i < count; ++i) {
418 int item = candidates[i];
419 if (!carried(item) && place_weight((Place){PLAYER, 0}) + total_weight(item) > max_weight)
420 printf("%s: You can't carry any more.\n", items[item].name);
421 else { state.places[item] = (Place){PLAYER, 0}; printf("%s: Taken.\n", items[item].name); success = true; }
422 }
423 return success;
424 }
425 int item = resolve(target);
426 if (container >= 0 && !same_place(state.places[item], source)) fail("The %s is not in the %s.", items[item].name, items[container].name);
427 if (same_place(state.places[item], (Place){PLAYER, 0})) fail("You already have the %s.", items[item].name);
428 if (items[item].fixed) fail("The %s is fixed in place.", items[item].name);
429 if (!carried(item) && place_weight((Place){PLAYER, 0}) + total_weight(item) > max_weight) fail("You can't carry any more.");
430 state.places[item] = (Place){PLAYER, 0}; puts("Taken.");
431 return true;
432}

449static bool put_items(Phrase first, Phrase second) {
450 int item = all_form(first) ? -1 : resolve(first);
451 int container = resolve(second);
452 if (item >= 0) {
453 if (!carried(item)) fail("You aren't carrying the %s.", items[item].name);
454 if (item == container) fail("You can't put the %s inside itself.", items[item].name);
455 }
456 if (!items[container].container) fail("The %s is not a container.", items[container].name);
457 if (state.containers[container] != OPEN) fail("The %s is closed.", items[container].name);
458 char reason[256];
459 if (item >= 0) {
460 if (!put_item(item, container, reason)) fail("%s", reason);
461 printf("You put the %s in the %s.\n", items[item].name, items[container].name);
462 return true;
463 }
464 int candidates[MAX_DEFINITIONS], length = children((Place){PLAYER, 0}, candidates), count = 0;
465 for (int i = 0; i < length; ++i) if (candidates[i] != container) candidates[count++] = candidates[i];
466 if (!count) fail("You have nothing to put in the %s.", items[container].name);
467 bool success = false;
468 for (int i = 0; i < count; ++i) {
469 int candidate = candidates[i];
470 bool done = put_item(candidate, container, reason);
471 printf("%s: %s\n", items[candidate].name, done ? "Done." : reason);
472 success = success || done;
473 }
474 return success;
475}

488static void change_lock(Phrase target, Phrase explicit_key, bool locking) {
489 int d = target.count == 1 ? direction(target.words[0]) : -1;
490 if (d >= 0) {
491 Exit *exit = &state.exits[state.room][d];
492 if (exit->destination < 0) fail("There is no exit %s.", directions[d]);
493 if (exit->key < 0) fail("The way %s has no lock.", directions[d]);
494 if (exit->locked == locking) fail(locking ? "The way %s is already locked." : "The way %s is not locked.", directions[d]);
495 key_check(exit->key, explicit_key);
496 exit->locked = locking;
497 Exit *opposite = &state.exits[exit->destination][d ^ 1];
498 if (opposite->destination == state.room && opposite->key == exit->key) opposite->locked = locking;
499 } else {
500 int item = resolve(target);
501 if (!items[item].container || items[item].key < 0) fail("The %s has no lock.", items[item].name);
502 if ((state.containers[item] == LOCKED) == locking) fail(locking ? "The %s is already locked." : "The %s is not locked.", items[item].name);
503 if (locking && state.containers[item] == OPEN) fail("You must close the %s first.", items[item].name);
504 key_check(items[item].key, explicit_key);
505 state.containers[item] = locking ? LOCKED : CLOSED;
506 }
507 puts(locking ? "Locked." : "Unlocked.");
508}

510static void examine(int item) {
511 Item *definition = &items[item];
512 if (!definition->details.count) printf("You see nothing special about the %s.\n", definition->name);
513 else for (int i = 0; i < definition->details.count; ++i) puts(definition->details.lines[i]);
514 if (definition->light) printf("It is %s (fuel %d).\n", state.lit[item] ? "lit" : "unlit", state.fuel[item]);
515 if (definition->container) {
516 if (state.containers[item] == LOCKED) puts("It is locked.");
517 else if (state.containers[item] == CLOSED) puts("It is closed.");
518 else {
519 int direct[MAX_DEFINITIONS];
520 if (children((Place){INSIDE, item}, direct)) { puts("It contains:"); item_tree((Place){INSIDE, item}, 1); }
521 else puts("It is empty.");
522 }
523 }
524 if (definition->points) printf("It is a treasure worth %d points.\n", definition->points);
525 printf("It weighs %d.\n", total_weight(item));
526}

528static bool execute(Verb verb, Phrase first, Phrase second, int d, bool *baseline) {
529 switch (verb) {
530 case GO: {
531 Exit exit = state.exits[state.room][d];
532 if (exit.destination < 0) fail("You can't go that way.");
533 if (exit.locked) fail("The way %s is locked.", directions[d]);
534 state.room = exit.destination; *baseline = room_lit();
535 describe(verbose || !state.visited[state.room]); state.visited[state.room] = true;
536 bool message = false; execute_rules(true, state.room, 0, &message);
537 break;
538 }
539 case LOOK: describe(true); break;
540 case EXAMINE: examine(resolve(first)); break;
541 case TAKE: return take_items(first, second);
542 case DROP: drop_items(first); break;
543 case PUT: return put_items(first, second);
544 case INVENTORY: {
545 int direct[MAX_DEFINITIONS];
546 if (!children((Place){PLAYER, 0}, direct)) puts("You are empty-handed.");
547 else { puts("You are carrying:"); item_tree((Place){PLAYER, 0}, 1); printf("Total weight: %d/%d\n", place_weight((Place){PLAYER, 0}), max_weight); }
548 break;
549 }
550 case OPEN_CONTAINER: {
551 int item = resolve(first);
552 if (!items[item].container) fail("You can't open the %s.", items[item].name);
553 if (state.containers[item] == OPEN) fail("The %s is already open.", items[item].name);
554 if (state.containers[item] == LOCKED) fail("The %s is locked.", items[item].name);
555 state.containers[item] = OPEN;
556 int direct[MAX_DEFINITIONS], count = children((Place){INSIDE, item}, direct);
557 if (!count) puts("Opened.");
558 else {
559 printf("Opening the %s reveals ", items[item].name);
560 for (int i = 0; i < count; ++i) { if (i) fputs(", ", stdout); fputs(items[direct[i]].name, stdout); }
561 puts(".");
562 }
563 break;
564 }
565 case CLOSE_CONTAINER: {
566 int item = resolve(first);
567 if (!items[item].container) fail("You can't close the %s.", items[item].name);
568 if (state.containers[item] != OPEN) fail("The %s is already closed.", items[item].name);
569 state.containers[item] = CLOSED; puts("Closed."); break;
570 }
571 case UNLOCK: case LOCK: change_lock(first, second, verb == LOCK); break;
572 case USE: {
573 int a = resolve(first);
574 int b = resolve(second);
575 if (!carried(a)) fail("You aren't carrying the %s.", items[a].name);
576 bool message = false;
577 if (!execute_rules(false, a, b, &message)) fail("Nothing happens.");
578 if (!message) puts("Done.");
579 break;
580 }
581 case LIGHT: {
582 int item = resolve(first);
583 if (!items[item].light) fail("You can't light the %s.", items[item].name);
584 if (state.lit[item]) fail("The %s is already lit.", items[item].name);
585 if (!state.fuel[item]) fail("The %s has no fuel.", items[item].name);
586 state.lit[item] = true; printf("The %s is now lit.\n", items[item].name); break;
587 }
588 case EXTINGUISH: {
589 int item = resolve(first);
590 if (!items[item].light) fail("You can't extinguish the %s.", items[item].name);
591 if (!state.lit[item]) fail("The %s is not lit.", items[item].name);
592 state.lit[item] = false; printf("The %s is now out.\n", items[item].name); break;
593 }
594 case WAIT: puts("Time passes."); break;
595 case SCORE: {
596 int points = treasure_points(), treasures = 0, count = 0;
597 for (int i = 0; i < item_count; ++i) if (items[i].points) { ++treasures; if (stored(i)) ++count; }
598 printf("You have scored %d points (treasure %d, bonus %d) in %d moves.\n", points + state.bonus, points, state.bonus, state.moves);
599 printf("Treasures stored: %d of %d.\n", count, treasures); break;
600 }
601 case SAVE: saved = state; has_save = true; puts("Game saved."); break;
602 case RESTORE:
603 if (!has_save) fail("There is no saved game.");
604 state = saved; puts("Game restored."); describe(true); break;
605 case VERBOSE: verbose = true; puts("Verbose mode."); break;
606 case BRIEF: verbose = false; puts("Brief mode."); break;
607 }
608 return true;
609}

621static bool no_arguments(Verb verb) {
622 return verb == LOOK || verb == INVENTORY || verb == WAIT || verb == SCORE || verb == SAVE || verb == RESTORE || verb == VERBOSE || verb == BRIEF;
623}

625static bool action(Verb verb) {
626 return verb == GO || verb == TAKE || verb == DROP || verb == PUT || verb == OPEN_CONTAINER || verb == CLOSE_CONTAINER ||
627 verb == UNLOCK || verb == LOCK || verb == USE || verb == LIGHT || verb == EXTINGUISH || verb == WAIT;
628}

630static void interpret(Words *words) {
631 if (!words->count) fail("I beg your pardon?");
632 if (direction(words->words[0]) >= 0) {
633 memmove(words->words + 1, words->words, words->count * sizeof(*words->words));
634 words->words[0] = "go"; ++words->count;
635 }
636 int parsed = -1;
637 for (int i = 0; i < 20; ++i) if (name_word(verb_aliases[i], words->words[0])) { parsed = i; break; }
638 if (parsed < 0) fail("I don't know the verb \"%s\".", words->words[0]);
639 Verb verb = (Verb)parsed;
640 if (state.ended != NONE && verb != RESTORE) fail("The game is over.");
641 Phrase first = {words->words + 1, words->count - 1}, second = {NULL, 0};
642 if (verb == LOOK && first.count) {
643 if (strcmp(first.words[0], "at")) fail("I don't understand that.");
644 verb = EXAMINE; ++first.words; --first.count;
645 }
646 int d = -1;
647 if (no_arguments(verb)) {
648 if (first.count) fail("I don't understand that.");
649 } else {
650 if (!first.count) {
651 if (verb == GO) fail("Which way do you want to go?");
652 fail("What do you want to %s?", verb_names[verb]);
653 }
654 if (verb == GO) {
655 if (first.count != 1 || (d = direction(first.words[0])) < 0) fail("I don't understand that.");
656 } else {
657 int split = -1;
658 for (int i = 0; i < first.count; ++i) {
659 const char *word = first.words[i];
660 bool preposition = (verb == PUT && (!strcmp(word, "in") || !strcmp(word, "into"))) ||
661 (verb == USE && (!strcmp(word, "on") || !strcmp(word, "with"))) ||
662 ((verb == LOCK || verb == UNLOCK) && !strcmp(word, "with")) || (verb == TAKE && !strcmp(word, "from"));
663 if (preposition) { split = i; break; }
664 }
665 if (split >= 0) {
666 if (!split || split + 1 == first.count) fail("I don't understand that.");
667 second = (Phrase){first.words + split + 1, first.count - split - 1}; first.count = split;
668 } else if (verb == PUT || verb == USE) fail("I don't understand that.");
669 }
670 }
671 bool baseline = room_lit();
672 if (execute(verb, first, second, d, &baseline) && action(verb)) pass_turn(baseline);
673}

675static void process_line(const char *line) {
676 if (setjmp(command_error)) return;
677 Words words; normalize(line, &words);
678 if (words.count && (!strcmp(words.words[0], "again") || !strcmp(words.words[0], "g"))) {
679 if (words.count != 1) fail("I don't understand that.");
680 if (!has_previous) fail("There is nothing to repeat.");
681 normalize(previous, &words);
682 } else { strcpy(previous, line); has_previous = true; }
683 interpret(&words);
684}
