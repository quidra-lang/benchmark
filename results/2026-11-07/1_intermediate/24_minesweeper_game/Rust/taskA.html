65 fn new(width: usize, height: usize, mines: i32, seed: i64) -> Self {
66 let cells = width * height;
67 let neighbors = (0..cells).map(|cell| {
68 let (x, y) = ((cell % width) as isize, (cell / width) as isize);
69 let mut list = Vec::new();
70 for dy in -1..=1 {
71 for dx in -1..=1 {
72 if (dx != 0 || dy != 0) && (0..width as isize).contains(&(x + dx)) && (0..height as isize).contains(&(y + dy)) {
73 list.push((y + dy) as usize * width + (x + dx) as usize);
74 }
75 }
76 }
77 list
78 }).collect();
79 Self {
80 width, height, mine_count: mines, seed, determined: false,
81 mine: vec![false; cells], numbers: vec![0; cells], neighbors,
82 revealed: vec![false; cells], marks: vec![Mark::None; cells],
83 state: State::Ready, exploded: None, moves: Vec::new(), undo: Vec::new(),
84 }

87 fn cells(&self) -> usize { self.width * self.height }

116 fn place_mines(&mut self, first: usize, safety: Safety) {
117 let (sx, sy) = (first % self.width, first / self.width);
118 let mut candidates: Vec<_> = (0..self.cells()).filter(|&cell| match safety {
119 Safety::None => true,
120 Safety::Cell => cell != first,
121 Safety::Block => (cell % self.width).abs_diff(sx) >= 2 || (cell / self.width).abs_diff(sy) >= 2,
122 }).collect();
123 let mut state = self.seed;
124 for _ in 0..self.mine_count {
125 state = 48271 * state % 2147483647;
126 let selected = candidates.remove((state % candidates.len() as i64) as usize);
127 self.mine[selected] = true;
128 }
129 self.number_board();
130 self.determined = true;
131 self.state = State::Playing;
132 }

134 fn reveal(&mut self, cell: usize, safety: Safety) -> usize {
135 if !self.determined { self.place_mines(cell, safety); }
136 if self.mine[cell] { self.state = State::Lost; self.exploded = Some(cell); return 0; }
137 let mut queue = vec![cell];
138 self.revealed[cell] = true;
139 self.marks[cell] = Mark::None;
140 let mut head = 0;
141 while head < queue.len() {
142 let current = queue[head];
143 head += 1;
144 if self.numbers[current] != 0 { continue; }
145 for &next in &self.neighbors[current] {
146 if !self.unknown(next) { continue; }
147 self.revealed[next] = true;
148 self.marks[next] = Mark::None;
149 queue.push(next);
150 }
151 }
152 queue.len()
153 }

155 fn chord(&mut self, cell: usize, safety: Safety) -> usize {
156 let mut opened = 0;
157 for i in 0..self.neighbors[cell].len() {
158 let next = self.neighbors[cell][i];
159 if self.unknown(next) { opened += self.reveal(next, safety); }
160 if self.state == State::Lost { break; }
161 }
162 opened
163 }

165 fn check_win(&mut self) {
166 if self.state != State::Playing || self.opened() != self.safe_cells() { return; }
167 self.state = State::Won;
168 for cell in 0..self.cells() { if self.mine[cell] { self.marks[cell] = Mark::Flag; } }
169 }

171 fn show(&self) {
172 for y in 0..self.height {
173 let row: String = (y * self.width..(y + 1) * self.width).map(|cell| {
174 if self.state == State::Lost && self.mine[cell] {
175 if self.exploded == Some(cell) { '!' } else if self.marks[cell] == Mark::Flag { 'F' } else { '*' }
176 } else if self.state == State::Lost && self.marks[cell] == Mark::Flag { 'X' }
177 else if self.revealed[cell] { if self.numbers[cell] == 0 { '.' } else { (b'0' + self.numbers[cell] as u8) as char } }
178 else { match self.marks[cell] { Mark::None => '#', Mark::Flag => 'F', Mark::Question => '?' } }
179 }).collect();
180 println!("{row}");
181 }
182 }

185 match self.state {
186 State::Lost => { println!("boom {}", self.coord(self.exploded.unwrap())); self.show(); }
187 State::Won => println!("win moves {}", self.moves.len()),
188 _ => {}

192 fn constraints(&self) -> Vec<Constraint> {
193 let mut result = Vec::new();
194 for cell in 0..self.cells() {
195 if !self.revealed[cell] || self.numbers[cell] == 0 { continue; }
196 let flags = self.neighbors[cell].iter().filter(|&&next| self.marks[next] == Mark::Flag).count();
197 let unknown: Vec<_> = self.neighbors[cell].iter().copied().filter(|&next| self.unknown(next)).collect();
198 if !unknown.is_empty() { result.push(Constraint { cell, remaining: self.numbers[cell] as i32 - flags as i32, unknown }); }
199 }
200 result
201 }

203 fn deduce(&self) -> Deduction {
204 let constraints = self.constraints();
205 let mut result = Deduction::default();
206 for safe in [true, false] {
207 for c in &constraints {
208 let target = if safe { 0 } else { c.unknown.len() as i32 };
209 if c.remaining != target { continue; }
210 for &cell in &c.unknown { result.add(safe, cell, Reason::Single(c.cell)); }
211 }
212 if !result.empty() { return result; }
213 }
214 let mut at = vec![None; self.cells()];
215 for (i, c) in constraints.iter().enumerate() { at[c.cell] = Some(i); }
216 for a in &constraints {
217 let (ax, ay) = (a.cell % self.width, a.cell / self.width);
218 for y in ay.saturating_sub(2)..=(ay + 2).min(self.height - 1) {
219 for x in ax.saturating_sub(2)..=(ax + 2).min(self.width - 1) {
220 let Some(index) = at[y * self.width + x] else { continue; };
221 let b = &constraints[index];
222 if a.unknown.len() >= b.unknown.len() || !a.unknown.iter().all(|cell| b.unknown.contains(cell)) { continue; }
223 let difference = b.remaining - a.remaining;
224 if difference != 0 && difference != (b.unknown.len() - a.unknown.len()) as i32 { continue; }
225 for &cell in &b.unknown {
226 if !a.unknown.contains(&cell) { result.add(difference == 0, cell, Reason::Subset(a.cell, b.cell)); }
227 }
228 }
229 }
230 }
231 if !result.empty() { return result; }
232 if self.state == State::Playing && self.remaining() == 0 {
233 for cell in 0..self.cells() { if self.unknown(cell) { result.add(true, cell, Reason::Count); } }
234 }
235 result
236 }

238 fn hint(&self) {
239 let result = self.deduce();
240 if let Some((&cell, reason)) = result.safe.iter().next() {
241 println!("hint safe {} by {}", self.coord(cell), reason.text(self));
242 } else if let Some((&cell, reason)) = result.mines.iter().next() {
243 println!("hint mine {} by {}", self.coord(cell), reason.text(self));
244 } else { println!("hint none"); }

247 fn auto_play(&mut self, safety: Safety) -> AutoResult {
248 let mut total = AutoResult::default();
249 while self.state == State::Playing {
250 let result = self.deduce();
251 if result.empty() { break; }
252 total.rounds += 1;
253 for &cell in result.safe.keys() {
254 if self.revealed[cell] { continue; }
255 self.moves.push(Move { kind: MoveKind::AutoReveal, cell });
256 total.opened += self.reveal(cell, safety);
257 self.check_win();
258 if self.state.finished() { return total; }
259 }
260 for &cell in result.mines.keys() {
261 if !self.unknown(cell) { continue; }
262 self.marks[cell] = Mark::Flag;
263 total.flagged += 1;
264 self.moves.push(Move { kind: MoveKind::AutoFlag, cell });
265 }
266 }
267 total
268 }

270 fn solve(&mut self, safety: Safety) -> AutoResult {
271 let mut total = AutoResult::default();
272 while !self.state.finished() {
273 let step = self.auto_play(safety);
274 total.rounds += step.rounds;
275 total.flagged += step.flagged;
276 total.opened += step.opened;
277 if self.state.finished() { break; }
278 let mut frontier = Frontier::new(self);
279 for id in 0..frontier.components.len() { frontier.count(id); }
280 let unknown = (0..self.cells()).filter(|&cell| self.unknown(cell)).count() as i64;
281 let remaining = i64::from(self.remaining().max(0));
282 let mut choice = None;
283 let (mut best_numerator, mut best_denominator) = (0, 1);
284 for cell in 0..self.cells() {
285 if !self.unknown(cell) { continue; }
286 let (numerator, denominator) = if let Some(id) = frontier.component_of[cell] {
287 (frontier.ways[cell], frontier.components[id].configs)
288 } else { (remaining, unknown) };
289 if denominator == 0 { continue; }
290 if choice.is_none() || numerator * best_denominator < best_numerator * denominator {
291 choice = Some(cell);
292 best_numerator = numerator;
293 best_denominator = denominator;
294 }
295 }
296 let Some(cell) = choice else { break; };
297 total.guesses += 1;
298 self.moves.push(Move { kind: MoveKind::Guess, cell });
299 total.opened += self.reveal(cell, safety);
300 self.check_win();
301 }
302 total
303 }

305 fn probability(&self, cell: usize) {
306 let mut frontier = Frontier::new(self);
307 let Some(id) = frontier.component_of[cell] else { println!("prob {} unconstrained", self.coord(cell)); return; };
308 frontier.count(id);
309 let component = &frontier.components[id];
310 if component.configs == 0 { println!("prob {} inconsistent", self.coord(cell)); }
311 else {
312 println!("prob {} {} size {} configs {}", self.coord(cell), fraction(frontier.ways[cell], component.configs), component.cells.len(), component.configs);
313 }

329 fn replay(&self, count: usize, safety: Safety) {
330 let mut copy = self.copy(false);
331 copy.restart();
332 for movement in &self.moves[..count] {
333 let cell = movement.cell;
334 match movement.kind {
335 MoveKind::Reveal | MoveKind::AutoReveal | MoveKind::Guess => { copy.reveal(cell, safety); }
336 MoveKind::Chord => { copy.chord(cell, safety); }
337 MoveKind::Flag | MoveKind::AutoFlag => copy.marks[cell] = Mark::Flag,
338 MoveKind::Unflag => copy.marks[cell] = Mark::None,
339 MoveKind::Question => copy.marks[cell] = if copy.marks[cell] == Mark::Question { Mark::None } else { Mark::Question },
340 }
341 copy.check_win();
342 }
343 println!("replay {count}/{}", self.moves.len());
344 copy.show();
345 }

347 fn board_info(&self) {
348 let mut visited = vec![false; self.cells()];
349 let (mut openings, mut largest, mut isolated) = (0, 0_usize, 0);
350 for cell in 0..self.cells() {
351 if self.mine[cell] { continue; }
352 if self.numbers[cell] != 0 {
353 if !self.neighbors[cell].iter().any(|&next| !self.mine[next] && self.numbers[next] == 0) { isolated += 1; }
354 } else if !visited[cell] {
355 openings += 1;
356 let mut covered = vec![false; self.cells()];
357 let mut queue = vec![cell];
358 visited[cell] = true;
359 covered[cell] = true;
360 let mut head = 0;
361 while head < queue.len() {
362 let current = queue[head];
363 head += 1;
364 for &next in &self.neighbors[current] {
365 covered[next] = true;
366 if self.numbers[next] == 0 && !visited[next] { visited[next] = true; queue.push(next); }
367 }
368 }
369 largest = largest.max(covered.iter().filter(|&&cell| cell).count());
370 }
371 }
372 println!("3bv {} openings {} largest {} mines {} safe {}", openings + isolated, openings, largest, self.mine_count, self.safe_cells());
373 }

375 fn process(&mut self, command: &str, cell: usize, value: i64, safety: Safety) -> CommandResult {
376 if matches!(command, "REVEAL" | "FLAG" | "UNFLAG" | "QUESTION" | "PROB") {
377 if self.revealed[cell] { return Err("already revealed".into()); }
378 if matches!(command, "REVEAL" | "PROB") && self.marks[cell] == Mark::Flag { return Err("flagged".into()); }
379 if command == "FLAG" && self.marks[cell] == Mark::Flag { return Err("already flagged".into()); }
380 if command == "UNFLAG" && self.marks[cell] != Mark::Flag { return Err("not flagged".into()); }
381 }
382 match command {
383 "RESTART" => { self.restart(); println!("restart {}x{} mines {}", self.width, self.height, self.mine_count); }
384 "REVEAL" => {
385 self.moves.push(Move { kind: MoveKind::Reveal, cell });
386 let opened = self.reveal(cell, safety);
387 if self.state != State::Lost { println!("revealed {opened}"); self.check_win(); }
388 self.ending_output();
389 }
390 "FLAG" => {
391 self.marks[cell] = Mark::Flag;
392 self.moves.push(Move { kind: MoveKind::Flag, cell });
393 println!("flagged {} left {}", self.coord(cell), self.remaining());
394 }
395 "UNFLAG" => {
396 self.marks[cell] = Mark::None;
397 self.moves.push(Move { kind: MoveKind::Unflag, cell });
398 println!("unflagged {} left {}", self.coord(cell), self.remaining());
399 }
400 "QUESTION" => {
401 self.moves.push(Move { kind: MoveKind::Question, cell });
402 self.marks[cell] = if self.marks[cell] == Mark::Question { Mark::None } else { Mark::Question };
403 println!("{} {} left {}", if self.marks[cell] == Mark::None { "unmarked" } else { "questioned" }, self.coord(cell), self.remaining());
404 }
405 "CHORD" => {
406 if !self.revealed[cell] { return Err("not revealed".into()); }
407 if self.numbers[cell] == 0 { return Err("no number".into()); }
408 let flags = self.neighbors[cell].iter().filter(|&&next| self.marks[next] == Mark::Flag).count();
409 if flags != self.numbers[cell] { return Err(format!("flags {flags} of {}", self.numbers[cell])); }
410 if !self.neighbors[cell].iter().any(|&next| self.unknown(next)) { return Err("nothing to reveal".into()); }
411 self.moves.push(Move { kind: MoveKind::Chord, cell });
412 let opened = self.chord(cell, safety);
413 if self.state != State::Lost { println!("chorded {opened}"); self.check_win(); }
414 self.ending_output();
415 }
416 "UNDO" => {
417 let snapshot = self.undo.pop().ok_or_else(|| "nothing to undo".to_string())?;
418 self.revealed = snapshot.revealed;
419 self.marks = snapshot.marks;
420 self.moves.truncate(snapshot.moves);
421 println!("undo {} moves {}", snapshot.name, self.moves.len());
422 }
423 "SHOW" => self.show(),
424 "STATUS" => println!("status {} left {} moves {} revealed {}/{}", self.state.name(), self.remaining(), self.moves.len(), self.opened(), self.safe_cells()),
425 "CELL" => {
426 if self.revealed[cell] {
427 let flags = self.neighbors[cell].iter().filter(|&&next| self.marks[next] == Mark::Flag).count();
428 let unknown = self.neighbors[cell].iter().filter(|&&next| self.unknown(next)).count();
429 println!("cell {} number {} flags {} unknown {}", self.coord(cell), self.numbers[cell], flags, unknown);
430 } else {
431 let opened = self.neighbors[cell].iter().filter(|&&next| self.revealed[next]).count();
432 let mark = match self.marks[cell] { Mark::None => "none", Mark::Flag => "flag", Mark::Question => "question" };
433 println!("cell {} hidden mark {} open {}", self.coord(cell), mark, opened);
434 }
435 }
436 "HISTORY" => {
437 println!("history {}", self.moves.len());
438 let count = value.min(self.moves.len() as i64) as usize;
439 for i in self.moves.len() - count..self.moves.len() {
440 let movement = self.moves[i];
441 println!("{} {} {}", i + 1, movement.kind.name(), self.coord(movement.cell));
442 }
443 }
444 "REPLAY" => {
445 if value > self.moves.len() as i64 { return Err(format!("no move {value}")); }
446 self.replay(value as usize, safety);
447 }
448 "HINT" => self.hint(),
449 "AUTO" | "SOLVE" => {
450 let result = if command == "AUTO" { self.auto_play(safety) } else { self.solve(safety) };
451 if command == "SOLVE" { println!("solve guesses {} rounds {} flagged {} revealed {}", result.guesses, result.rounds, result.flagged, result.opened); }
452 else { println!("auto rounds {} flagged {} revealed {}", result.rounds, result.flagged, result.opened); }
453 self.ending_output();
454 }
455 "ANALYZE" => {
456 let mut copy = self.copy(true);
457 let result = copy.auto_play(safety);
458 match copy.state {
459 State::Lost => println!("analyze boom {} rounds {}", copy.coord(copy.exploded.unwrap()), result.rounds),
460 State::Won => println!("analyze clear rounds {}", result.rounds),
461 _ => println!("analyze stuck rounds {} revealed {}/{}", result.rounds, copy.opened(), copy.safe_cells()),
462 }
463 }
464 "PROB" => self.probability(cell),
465 "FRONTIER" => self.frontier_report(),
466 "BOARDINFO" => self.board_info(),
467 _ => unreachable!(),
468 }
469 Ok(())
470 }
471}

475#[derive(Clone, Copy)]
476enum Reason { Single(usize), Subset(usize, usize), Count }

478impl Reason {
479 fn text(self, game: &Game) -> String {
480 match self {
481 Self::Single(cell) => format!("single {}", game.coord(cell)),
482 Self::Subset(a, b) => format!("subset {} {}", game.coord(a), game.coord(b)),
483 Self::Count => "count".into(),
484 }
485 }

488#[derive(Default)]
489struct Deduction { safe: BTreeMap<usize, Reason>, mines: BTreeMap<usize, Reason> }
490
491impl Deduction {
492 fn empty(&self) -> bool { self.safe.is_empty() && self.mines.is_empty() }
493 fn add(&mut self, safe: bool, cell: usize, reason: Reason) {
494 let side = if safe { &mut self.safe } else { &mut self.mines };
495 side.entry(cell).or_insert(reason);
496 }
497}

511
512fn root(parent: &mut [usize], mut cell: usize) -> usize {
513 while parent[cell] != cell { parent[cell] = parent[parent[cell]]; cell = parent[cell]; }
514 cell
515}
516
517impl Frontier {
518 fn new(game: &Game) -> Self {
519 let constraints = game.constraints();
520 let mut parent: Vec<_> = (0..game.cells()).collect();
521 let mut present = vec![false; game.cells()];
522 for c in &constraints {
523 for &cell in &c.unknown {
524 present[cell] = true;
525 let left = root(&mut parent, cell);
526 let right = root(&mut parent, c.unknown[0]);
527 parent[left] = right;
528 }
529 }
530 let mut ids = vec![None; game.cells()];
531 let mut component_of = vec![None; game.cells()];
532 let mut components = Vec::<Component>::new();
533 for cell in 0..game.cells() {
534 if !present[cell] { continue; }
535 let representative = root(&mut parent, cell);
536 let id = if let Some(id) = ids[representative] { id } else {
537 let id = components.len();
538 ids[representative] = Some(id);
539 components.push(Component::default());
540 id
541 };
542 component_of[cell] = Some(id);
543 components[id].cells.push(cell);
544 }
545 Self { constraints, component_of, components, ways: vec![0; game.cells()] }
546 }
547
548 fn count(&mut self, id: usize) {
549 let cells = &self.components[id].cells;
550 let mut local = vec![0; self.component_of.len()];
551 for (i, &cell) in cells.iter().enumerate() { local[cell] = i; }
552 let constraints = self.constraints.iter().filter(|c| self.component_of[c.unknown[0]] == Some(id))
553 .map(|c| (c.unknown.iter().fold(0, |mask, &cell| mask | (1_u32 << local[cell])), c.remaining)).collect();
554 let mut search = Search { size: cells.len(), constraints, configs: 0, minimum: cells.len() + 1, maximum: 0, ways: vec![0; cells.len()] };
555 search.visit(0, 0);
556 for (i, &cell) in cells.iter().enumerate() { self.ways[cell] = search.ways[i]; }
557 let component = &mut self.components[id];
558 component.configs = search.configs;
559 component.minimum = search.minimum;
560 component.maximum = search.maximum;
561 }
562}
563
564struct Search { size: usize, constraints: Vec<(u32, i32)>, configs: i64, minimum: usize, maximum: usize, ways: Vec<i64> }
565
566impl Search {
567 fn visit(&mut self, depth: usize, mines: u32) {
568 let assigned = (1_u32 << depth) - 1;
569 for &(mask, target) in &self.constraints {
570 let used = (mask & mines).count_ones() as i32;
571 let left = (mask & !assigned).count_ones() as i32;
572 if used > target || used + left < target { return; }
573 }
574 if depth == self.size {
575 let count = mines.count_ones() as usize;
576 self.configs += 1;
577 self.minimum = self.minimum.min(count);
578 self.maximum = self.maximum.max(count);
579 for i in 0..self.size { if mines & (1 << i) != 0 { self.ways[i] += 1; } }
580 } else {
581 self.visit(depth + 1, mines);
582 self.visit(depth + 1, mines | (1 << depth));
583 }
584 }
585}

587fn fraction(numerator: i64, denominator: i64) -> String {
588 let (mut a, mut b) = (numerator, denominator);
589 while b != 0 { (a, b) = (b, a % b); }
590 format!("{}/{}", numerator / a, denominator / a)
591}

593fn number_word(word: &str) -> CommandResult<i64> {
594 if word.len() > 10 || (word.len() > 1 && word.starts_with('0')) || !word.bytes().all(|b| b.is_ascii_digit()) {
595 Err("bad arguments".into())
596 } else { Ok(word.parse().unwrap()) }
597}

599#[derive(Default)]
600struct Statistics { played: usize, won: usize, lost: usize, streak: usize, best: usize, fastest: Option<usize> }
601
602impl Statistics {
603 fn finish(&mut self, game: &Game) {
604 self.played += 1;
605 if game.state == State::Won {
606 self.won += 1;
607 self.streak += 1;
608 self.best = self.best.max(self.streak);
609 self.fastest = Some(self.fastest.map_or(game.moves.len(), |old| old.min(game.moves.len())));
610 } else { self.lost += 1; self.streak = 0; }
611 }
612
613 fn show(&self) {
614 let rate = if self.played == 0 { "-".into() } else { format!("{}%", 100 * self.won / self.played) };
615 let fastest = self.fastest.map_or_else(|| "-".into(), |n| n.to_string());
616 println!("stats played {} won {} lost {} rate {} streak {} best {} fastest {}", self.played, self.won, self.lost, rate, self.streak, self.best, fastest);
617 }
618}
619
620struct Table { game: Option<Game>, safety: Safety, stats: Statistics }
621
622impl Table {
623 fn layout(&mut self, width: usize, height: usize, rows: &[&str]) -> CommandResult {
624 for (i, row) in rows.iter().enumerate() {
625 if row.len() != width || !row.bytes().all(|b| b == b'*' || b == b'.') { return Err(format!("bad row {}", i + 1)); }
626 }
627 let mine: Vec<_> = rows.iter().flat_map(|row| row.bytes().map(|b| b == b'*')).collect();
628 let count = mine.iter().filter(|&&cell| cell).count();
629 if count == 0 || count == width * height { return Err(format!("bad mine count {count}")); }
630 let mut game = Game::new(width, height, count as i32, 0);
631 game.mine = mine;
632 game.determined = true;
633 game.state = State::Playing;
634 game.number_board();
635 self.game = Some(game);
636 println!("layout {width}x{height} mines {count}");
637 Ok(())
638 }
639
640 fn execute(&mut self, words: &[&str]) -> CommandResult {
641 let command = words[0];
642 let &(_, arguments) = COMMANDS.iter().find(|&&(name, _)| name == command).ok_or_else(|| format!("unknown command {command}"))?;
643 if words.len() != arguments + 1 { return Err("bad arguments".into()); }
644 let mut values = Vec::new();
645 let mut safety = self.safety;
646 if command == "PRESET" {
647 let (width, height, mines) = match words[1] {
648 "beginner" => (9, 9, 10), "intermediate" => (16, 16, 40), "expert" => (30, 16, 99),
649 _ => return Err("bad arguments".into()),
650 };
651 values = vec![width, height, mines, number_word(words[2])?];
652 } else if command == "SAFETY" {
653 safety = match words[1] {
654 "block" => Safety::Block, "cell" => Safety::Cell, "none" => Safety::None,
655 _ => return Err("bad arguments".into()),
656 };
657 } else { for word in &words[1..] { values.push(number_word(word)?); } }
658 if matches!(command, "NEW" | "PRESET") {
659 let (width, height, mines, seed) = (values[0], values[1], values[2], values[3]);
660 if !(1..=30).contains(&width) || !(1..=30).contains(&height) || mines < 1 || mines > width * height - 9 || !(1..=2147483646).contains(&seed) {
661 return Err("bad value".into());
662 }
663 self.game = Some(Game::new(width as usize, height as usize, mines as i32, seed));
664 println!("new {width}x{height} mines {mines}");
665 return Ok(());
666 }
667 if command == "HISTORY" && values[0] == 0 { return Err("bad value".into()); }
668 if command == "SAFETY" { self.safety = safety; println!("safety {}", words[1]); return Ok(()); }
669 if command == "STATS" { self.stats.show(); return Ok(()); }
670 let game = self.game.as_mut().ok_or_else(|| "no game".to_string())?;
671 let has_cell = matches!(command, "REVEAL" | "FLAG" | "UNFLAG" | "QUESTION" | "CHORD" | "CELL" | "PROB");
672 if has_cell && (values[0] >= game.width as i64 || values[1] >= game.height as i64) { return Err("out of range".into()); }
673 let needs_unfinished = matches!(command, "REVEAL" | "FLAG" | "UNFLAG" | "QUESTION" | "CHORD" | "UNDO" | "HINT" | "AUTO" | "ANALYZE" | "PROB" | "FRONTIER" | "SOLVE");
674 if needs_unfinished && game.state.finished() { return Err("game over".into()); }
675 if matches!(command, "ANALYZE" | "BOARDINFO") && !game.determined { return Err("not started".into()); }
676 let records_undo = matches!(command, "REVEAL" | "FLAG" | "UNFLAG" | "QUESTION" | "CHORD" | "AUTO" | "SOLVE");
677 let snapshot = records_undo.then(|| Snapshot {
678 name: command.to_ascii_lowercase(), revealed: game.revealed.clone(), marks: game.marks.clone(), moves: game.moves.len(),
679 });
680 let was_finished = game.state.finished();
681 let cell = if has_cell { values[1] as usize * game.width + values[0] as usize } else { 0 };
682 game.process(command, cell, values.first().copied().unwrap_or(0), safety)?;
683 if let Some(snapshot) = snapshot { if game.moves.len() > snapshot.moves { game.undo.push(snapshot); } }
684 if !was_finished && game.state.finished() { self.stats.finish(game); }
685 Ok(())
686 }
687}

689fn main() {
690 let mut input = String::new();
691 io::stdin().read_to_string(&mut input).unwrap();
692 let mut lines = input.lines();
693 let count: usize = lines.next().unwrap().parse().unwrap();
694 let mut table = Table { game: None, safety: Safety::Block, stats: Statistics::default() };
695 for _ in 0..count {
696 let words: Vec<_> = lines.next().unwrap().split(' ').collect();
697 let result = if words[0] == "LAYOUT" {
698 let width = words[1].parse().unwrap();
699 let height = words[2].parse().unwrap();
700 let rows: Vec<_> = (0..height).map(|_| lines.next().unwrap()).collect();
701 table.layout(width, height, &rows)
702 } else { table.execute(&words) };
703 if let Err(error) = result { println!("error: {error}"); }
704 }
705}
