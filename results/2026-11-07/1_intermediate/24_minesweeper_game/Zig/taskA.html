150const Component = struct {
151 cells: std.ArrayList(usize) = .empty,
152 constraints: std.ArrayList(Constraint) = .empty,
153 configurations: usize = 0,
154 minimum: usize = 0,
155 maximum: usize = 0,
156 mine_counts: [max_component]usize = @splat(0),
157 counted: bool = false,
158 fn count(component: *Component) void {
159 if (component.counted) return;
160 component.counted = true;
161 component.minimum = component.cells.items.len;
162 for (component.constraints.items) |c| if (c.remaining < 0 or c.remaining > @as(i32, @intCast(c.unknown.len))) return;
163 var search: Counter = .{ .component = component };
164 var index: [max_cells]usize = undefined;
165 for (component.cells.items, 0..) |cell, i| {
166 index[cell] = i;
167 search.order[i] = i;
168 }
169 for (component.constraints.items, 0..) |c, i| {
170 search.remaining[i] = @intCast(c.unknown.len);
171 for (c.unknown.cells[0..c.unknown.len]) |cell| search.links[index[cell]].append(i);
172 }
173 const size = component.cells.items.len;
174 std.mem.sort(usize, search.order[0..size], @as([]const Nearby, search.links[0..size]), Counter.before);
175 search.visit(0, 0);
176 }
177};

179const Counter = struct {
180 component: *Component,
181 links: [max_component]Nearby = @splat(.{}),
182 order: [max_component]usize = undefined,
183 assignment: [max_component]usize = @splat(0),
184 remaining: [max_cells]i32 = @splat(0),
185 sums: [max_cells]i32 = @splat(0),
186 fn before(links: []const Nearby, a: usize, b: usize) bool {
187 return if (links[a].len != links[b].len) links[a].len > links[b].len else a < b;
188 }
189 fn visit(search: *Counter, depth: usize, mines: usize) void {
190 const component = search.component;
191 if (depth == component.cells.items.len) {
192 component.configurations += 1;
193 component.minimum = @min(component.minimum, mines);
194 component.maximum = @max(component.maximum, mines);
195 for (0..component.cells.items.len) |i| component.mine_counts[i] += search.assignment[i];
196 return;
197 }
198 const variable = search.order[depth];
199 const links = &search.links[variable];
200 for (0..2) |value| {
201 search.assignment[variable] = value;
202 var possible = true;
203 for (links.cells[0..links.len]) |c| {
204 search.remaining[c] -= 1;
205 search.sums[c] += @intCast(value);
206 const required = component.constraints.items[c].remaining;
207 if (search.sums[c] > required or search.sums[c] + search.remaining[c] < required) possible = false;
208 }
209 if (possible) search.visit(depth + 1, mines + value);
210 for (links.cells[0..links.len]) |c| {
211 search.sums[c] -= @intCast(value);
212 search.remaining[c] += 1;
213 }
214 }
215 }
216};

220fn rootOf(parents: *[max_cells]usize, first: usize) usize {
221 var cell = first;
222 while (parents[cell] != cell) {
223 parents[cell] = parents[parents[cell]];
224 cell = parents[cell];
225 }
226 return cell;
227}

246 fn initialize(game: *Game, gpa: Allocator, width: usize, height: usize, mines: usize, seed: u64, placement: ?[max_cells]bool) void {
247 game.* = .{ .gpa = gpa, .width = width, .height = height, .mine_count = mines, .seed = seed };
248 for (0..game.cellCount()) |cell| {
249 const x: i32 = @intCast(cell % width);
250 const y: i32 = @intCast(cell / width);
251 var dy: i32 = -1;
252 while (dy <= 1) : (dy += 1) {
253 var dx: i32 = -1;
254 while (dx <= 1) : (dx += 1) {
255 const nx = x + dx;
256 const ny = y + dy;
257 if ((dx != 0 or dy != 0) and nx >= 0 and ny >= 0 and nx < @as(i32, @intCast(width)) and ny < @as(i32, @intCast(height)))
258 game.neighbors[cell].append(@as(usize, @intCast(ny)) * width + @as(usize, @intCast(nx)));
259 }
260 }
261 }
262 if (placement) |fixed| game.setPlacement(fixed);
263 }

306 fn setPlacement(game: *Game, placement: [max_cells]bool) void {
307 game.mines = placement;
308 game.placed = true;
309 for (game.neighbors[0..game.cellCount()], 0..) |nearby, cell| {
310 game.numbers[cell] = 0;
311 for (nearby.cells[0..nearby.len]) |neighbor| if (game.mines[neighbor]) {
312 game.numbers[cell] += 1;
313 };
314 }
315 game.state = .playing;

317 fn generatePlacement(game: *Game, first: usize, safety: Safety) void {
318 var candidates: [max_cells]usize = undefined;
319 var length: usize = 0;
320 const sx: i32 = @intCast(first % game.width);
321 const sy: i32 = @intCast(first / game.width);
322 for (0..game.cellCount()) |cell| {
323 const eligible = switch (safety) {
324 .none => true,
325 .cell => cell != first,
326 .block => @abs(@as(i32, @intCast(cell % game.width)) - sx) >= 2 or @abs(@as(i32, @intCast(cell / game.width)) - sy) >= 2,
327 };
328 if (eligible) {
329 candidates[length] = cell;
330 length += 1;
331 }
332 }
333 var placement: [max_cells]bool = @splat(false);
334 var random = game.seed;
335 for (0..game.mine_count) |_| {
336 random = 48271 * random % 2147483647;
337 const index: usize = @intCast(random % @as(u64, @intCast(length)));
338 placement[candidates[index]] = true;
339 std.mem.copyForwards(usize, candidates[index .. length - 1], candidates[index + 1 .. length]);
340 length -= 1;
341 }
342 game.setPlacement(placement);
343 }

372 fn reveal(game: *Game, first: usize, safety: Safety) usize {
373 if (!game.placed) game.generatePlacement(first, safety);
374 if (game.mines[first]) {
375 game.state = .lost;
376 game.exploded = first;
377 return 0;
378 }
379 var queue: [max_cells]usize = undefined;
380 queue[0] = first;
381 var head: usize = 0;
382 var length: usize = 1;
383 game.revealed[first] = true;
384 game.marks[first] = .none;
385 while (head < length) : (head += 1) {
386 const cell = queue[head];
387 if (game.numbers[cell] != 0) continue;
388 const nearby = &game.neighbors[cell];
389 for (nearby.cells[0..nearby.len]) |neighbor| if (game.isUnknown(neighbor)) {
390 game.revealed[neighbor] = true;
391 game.marks[neighbor] = .none;
392 queue[length] = neighbor;
393 length += 1;
394 };
395 }
396 return length;
397 }
398 fn chord(game: *Game, cell: usize, safety: Safety) usize {
399 var opened: usize = 0;
400 const nearby = &game.neighbors[cell];
401 for (nearby.cells[0..nearby.len]) |neighbor| if (game.isUnknown(neighbor)) {
402 opened += game.reveal(neighbor, safety);
403 if (game.state == .lost) break;
404 };
405 return opened;
406 }
407 fn checkWin(game: *Game) void {
408 if (game.state != .playing or game.revealedCount() != game.cellCount() - game.mine_count) return;
409 game.state = .won;
410 for (game.mines[0..game.cellCount()], 0..) |mine, cell| if (mine) {
411 game.marks[cell] = .flag;
412 };
413 }
414 fn show(game: *const Game, out: *std.Io.Writer) std.Io.Writer.Error!void {
415 for (0..game.height) |y| {
416 var row: [30]u8 = undefined;
417 for (0..game.width) |x| {
418 const cell = y * game.width + x;
419 row[x] = if (game.state == .lost and game.mines[cell])
420 (if (cell == game.exploded) '!' else if (game.marks[cell] == .flag) 'F' else '*')
421 else if (game.state == .lost and game.marks[cell] == .flag) 'X' else if (game.revealed[cell]) (if (game.numbers[cell] == 0) '.' else '0' + game.numbers[cell]) else switch (game.marks[cell]) {
422 .none => '#',
423 .flag => 'F',
424 .question => '?',
425 };
426 }
427 try out.writeAll(row[0..game.width]);
428 try out.writeByte('\n');
429 }
430 }
431 fn makeConstraints(game: *const Game) Constraints {
432 var result: Constraints = .{};
433 for (0..game.cellCount()) |cell| if (game.revealed[cell] and game.numbers[cell] > 0) {
434 const unknown = game.unknownNeighbors(cell);
435 if (unknown.len > 0) {
436 result.items[result.len] = .{ .cell = cell, .unknown = unknown, .remaining = @as(i32, game.numbers[cell]) - @as(i32, @intCast(game.flagNeighbors(cell))) };
437 result.len += 1;
438 }
439 };
440 return result;
441 }
442 fn deduce(game: *const Game) Deduction {
443 const constraints = game.makeConstraints();
444 var result: Deduction = .{};
445 for (constraints.items[0..constraints.len]) |c| if (c.remaining == 0) {
446 for (c.unknown.cells[0..c.unknown.len]) |cell| if (result.safe[cell] == null) {
447 result.safe[cell] = .{ .kind = .single, .first = c.cell };
448 };
449 };
450 if (result.any(game.cellCount())) return result;
451 for (constraints.items[0..constraints.len]) |c| if (c.remaining == @as(i32, @intCast(c.unknown.len))) {
452 for (c.unknown.cells[0..c.unknown.len]) |cell| if (result.mines[cell] == null) {
453 result.mines[cell] = .{ .kind = .single, .first = c.cell };
454 };
455 };
456 if (result.any(game.cellCount())) return result;
457 for (constraints.items[0..constraints.len]) |a| {
458 for (constraints.items[0..constraints.len]) |b| {
459 if (a.unknown.len >= b.unknown.len) continue;
460 var subset = true;
461 for (a.unknown.cells[0..a.unknown.len]) |cell| if (!b.unknown.contains(cell)) {
462 subset = false;
463 break;
464 };
465 if (!subset) continue;
466 const difference: i32 = @intCast(b.unknown.len - a.unknown.len);
467 const left = b.remaining - a.remaining;
468 if (left != 0 and left != difference) continue;
469 const side = if (left == 0) &result.safe else &result.mines;
470 for (b.unknown.cells[0..b.unknown.len]) |cell| if (!a.unknown.contains(cell) and side[cell] == null) {
471 side[cell] = .{ .kind = .subset, .first = a.cell, .second = b.cell };
472 };
473 }
474 }
475 if (result.any(game.cellCount())) return result;
476 if (game.state == .playing and game.remainingMines() == 0)
477 for (0..game.cellCount()) |cell| if (game.isUnknown(cell)) {
478 result.safe[cell] = .{ .kind = .count };
479 };
480 return result;
481 }
482 fn buildFrontier(game: *const Game, gpa: Allocator) Allocator.Error!Frontier {
483 const constraints = game.makeConstraints();
484 var parents: [max_cells]usize = undefined;
485 var frontier: [max_cells]bool = @splat(false);
486 for (0..game.cellCount()) |cell| parents[cell] = cell;
487 for (constraints.items[0..constraints.len]) |c| {
488 for (c.unknown.cells[0..c.unknown.len]) |cell| {
489 frontier[cell] = true;
490 parents[rootOf(&parents, cell)] = rootOf(&parents, c.unknown.cells[0]);
491 }
492 }
493 var result: Frontier = .{};
494 var groups: [max_cells]?usize = @splat(null);
495 for (0..game.cellCount()) |cell| if (frontier[cell]) {
496 const root = rootOf(&parents, cell);
497 if (groups[root] == null) {
498 groups[root] = result.components.items.len;
499 try result.components.append(gpa, .{});
500 }
501 const group = groups[root].?;
502 result.at[cell] = group;
503 try result.components.items[group].cells.append(gpa, cell);
504 };
505 for (constraints.items[0..constraints.len]) |c| {
506 const group = result.at[c.unknown.cells[0]].?;
507 try result.components.items[group].constraints.append(gpa, c);
508 }
509 return result;
510 }
511 fn playAuto(game: *Game, safety: Safety) Allocator.Error!Totals {
512 var totals: Totals = .{};
513 while (game.state == .playing) {
514 const deduction = game.deduce();
515 if (!deduction.any(game.cellCount())) break;
516 totals.rounds += 1;
517 for (0..game.cellCount()) |cell| if (deduction.safe[cell] != null and !game.revealed[cell]) {
518 try game.appendMove(.auto_reveal, cell);
519 totals.opened += game.reveal(cell, safety);
520 game.checkWin();
521 if (game.isFinished()) return totals;
522 };
523 for (0..game.cellCount()) |cell| if (deduction.mines[cell] != null and game.isUnknown(cell)) {
524 game.marks[cell] = .flag;
525 totals.flagged += 1;
526 try game.appendMove(.auto_flag, cell);
527 };
528 }
529 return totals;
530 }
531 fn playSolve(game: *Game, safety: Safety) Allocator.Error!Totals {
532 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
533 defer arena.deinit();
534 var totals: Totals = .{};
535 while (!game.isFinished()) {
536 totals.add(try game.playAuto(safety));
537 if (game.isFinished()) break;
538 _ = arena.reset(.retain_capacity);
539 const frontier = try game.buildFrontier(arena.allocator());
540 var numerators: [max_cells]usize = @splat(0);
541 var denominators: [max_cells]usize = @splat(0);
542 for (frontier.components.items) |*component| {
543 component.count();
544 for (component.cells.items, 0..) |cell, i| {
545 numerators[cell] = component.mine_counts[i];
546 denominators[cell] = component.configurations;
547 }
548 }
549 var unknown_count: usize = 0;
550 for (0..game.cellCount()) |cell| if (game.isUnknown(cell)) {
551 unknown_count += 1;
552 };
553 const background: usize = @intCast(@max(game.remainingMines(), 0));
554 var chosen: ?usize = null;
555 var best_numerator: usize = 0;
556 var best_denominator: usize = 1;
557 for (0..game.cellCount()) |cell| if (game.isUnknown(cell)) {
558 const numerator = if (frontier.at[cell] == null) background else numerators[cell];
559 const denominator = if (frontier.at[cell] == null) unknown_count else denominators[cell];
560 if (denominator != 0 and (chosen == null or numerator * best_denominator < best_numerator * denominator)) {
561 chosen = cell;
562 best_numerator = numerator;
563 best_denominator = denominator;
564 }
565 };
566 const chosen_cell = chosen orelse break;
567 totals.guesses += 1;
568 try game.appendMove(.guess, chosen_cell);
569 totals.opened += game.reveal(chosen_cell, safety);
570 game.checkWin();
571 }
572 return totals;
573 }
574 fn replay(game: *const Game, gpa: Allocator, count: usize, safety: Safety) Allocator.Error!*Game {
575 const copy = try game.clone(gpa, true);
576 for (game.moves.items[0..count]) |move| {
577 switch (move.kind) {
578 .reveal, .auto_reveal, .guess => {
579 _ = copy.reveal(move.cell, safety);
580 },
581 .chord => {
582 _ = copy.chord(move.cell, safety);
583 },
584 .flag, .auto_flag => copy.marks[move.cell] = .flag,
585 .unflag => copy.marks[move.cell] = .none,
586 .question => copy.marks[move.cell] = if (copy.marks[move.cell] == .question) .none else .question,
587 }
588 try copy.moves.append(gpa, move);
589 copy.checkWin();
590 }
591 return copy;
592 }
593 fn isBlank(game: *const Game, cell: usize) bool {
594 return !game.mines[cell] and game.numbers[cell] == 0;
595 }
596 fn boardInfo(game: *const Game) struct { bv: usize, openings: usize, largest: usize } {
597 var visited: [max_cells]bool = @splat(false);
598 var openings: usize = 0;
599 var largest: usize = 0;
600 var isolated: usize = 0;
601 for (0..game.cellCount()) |first| {
602 if (game.isBlank(first) and !visited[first]) {
603 openings += 1;
604 var area: [max_cells]bool = @splat(false);
605 var queue: [max_cells]usize = undefined;
606 queue[0] = first;
607 area[first] = true;
608 visited[first] = true;
609 var head: usize = 0;
610 var length: usize = 1;
611 var area_size: usize = 1;
612 while (head < length) : (head += 1) {
613 const nearby = &game.neighbors[queue[head]];
614 for (nearby.cells[0..nearby.len]) |neighbor| {
615 if (!area[neighbor]) {
616 area[neighbor] = true;
617 area_size += 1;
618 }
619 if (game.isBlank(neighbor) and !visited[neighbor]) {
620 visited[neighbor] = true;
621 queue[length] = neighbor;
622 length += 1;
623 }
624 }
625 }
626 largest = @max(largest, area_size);
627 }
628 if (!game.mines[first] and game.numbers[first] > 0) {
629 const nearby = &game.neighbors[first];
630 var has_blank = false;
631 for (nearby.cells[0..nearby.len]) |neighbor| if (game.isBlank(neighbor)) {
632 has_blank = true;
633 break;
634 };
635 if (!has_blank) isolated += 1;
636 }
637 }
638 return .{ .bv = openings + isolated, .openings = openings, .largest = largest };
639 }
640};

642const Statistics = struct {
643 played: usize = 0,
644 wins: usize = 0,
645 losses: usize = 0,
646 streak: usize = 0,
647 best: usize = 0,
648 fastest: ?usize = null,
649 fn record(stats: *Statistics, game: *const Game) void {
650 stats.played += 1;
651 if (game.state == .won) {
652 stats.wins += 1;
653 stats.streak += 1;
654 stats.best = @max(stats.best, stats.streak);
655 stats.fastest = @min(stats.fastest orelse game.moves.items.len, game.moves.items.len);
656 } else {
657 stats.losses += 1;
658 stats.streak = 0;
659 }
660 }
661};

663const Program = struct {
664 arena: std.heap.ArenaAllocator,
665 scratch: Allocator,
666 input: *std.Io.Reader,
667 out: *std.Io.Writer,
668 game: ?*Game = null,
669 safety: Safety = .block,
670 stats: Statistics = .{},
671 error_message: []const u8 = "",
672 fn fail(program: *Program, comptime fmt: []const u8, args: anytype) Error {
673 program.error_message = try std.fmt.allocPrint(program.scratch, fmt, args);
674 return error.BadCommand;
675 }
676 fn emit(program: *Program, comptime fmt: []const u8, args: anytype) Error!void {
677 try program.out.print(fmt ++ "\n", args);
678 }
679 fn parseNumber(program: *Program, word: []const u8) Error!i64 {
680 if (word.len > 10 or (word.len > 1 and word[0] == '0')) return program.fail("bad arguments", .{});
681 for (word) |c| if (c < '0' or c > '9') return program.fail("bad arguments", .{});
682 return std.fmt.parseInt(i64, word, 10) catch unreachable;
683 }
684 fn newGame(program: *Program, width: usize, height: usize, mines: usize, seed: u64, placement: ?[max_cells]bool) Allocator.Error!void {
685 _ = program.arena.reset(.retain_capacity);
686 const gpa = program.arena.allocator();
687 const game = try gpa.create(Game);
688 game.initialize(gpa, width, height, mines, seed, placement);
689 program.game = game;
690 }
691 fn layout(program: *Program, words: []const []const u8) !void {
692 const width = std.fmt.parseInt(usize, words[1], 10) catch unreachable;
693 const height = std.fmt.parseInt(usize, words[2], 10) catch unreachable;
694 var rows: [30][]const u8 = undefined;
695 for (rows[0..height]) |*row| row.* = try readLine(program.scratch, program.input);
696 for (rows[0..height], 1..) |row, i| {
697 if (row.len != width) return program.fail("bad row {d}", .{i});
698 for (row) |c| if (c != '*' and c != '.') return program.fail("bad row {d}", .{i});
699 }
700 var placement: [max_cells]bool = @splat(false);
701 var mines: usize = 0;
702 for (rows[0..height], 0..) |row, y| for (row, 0..) |c, x| {
703 placement[y * width + x] = c == '*';
704 if (c == '*') mines += 1;
705 };
706 if (mines == 0 or mines == width * height) return program.fail("bad mine count {d}", .{mines});
707 try program.newGame(width, height, mines, 0, placement);
708 try program.emit("layout {d}x{d} mines {d}", .{ width, height, mines });
709 }
710 fn finishOutput(program: *Program, game: *const Game) Error!void {
711 if (game.state == .lost) {
712 try program.emit("boom {f}", .{game.position(game.exploded)});
713 try game.show(program.out);
714 } else if (game.state == .won) try program.emit("win moves {d}", .{game.moves.items.len});
715 }
716 fn execute(program: *Program, words: []const []const u8) !void {
717 if (std.mem.eql(u8, words[0], "LAYOUT")) return program.layout(words);
718 const command = std.meta.stringToEnum(Command, words[0]) orelse return program.fail("unknown command {s}", .{words[0]});
719 if (words.len != command.arity() + 1) return program.fail("bad arguments", .{});
720 var values: [4]i64 = @splat(0);
721 var preset: Preset = .beginner;
722 var safety = program.safety;
723 for (words[1..], 0..) |word, i| {
724 if (command == .PRESET and i == 0) preset = std.meta.stringToEnum(Preset, word) orelse return program.fail("bad arguments", .{}) else if (command == .SAFETY) safety = std.meta.stringToEnum(Safety, word) orelse return program.fail("bad arguments", .{}) else values[i] = try program.parseNumber(word);
725 }
726 if (command == .NEW or command == .PRESET) {
727 const dimensions: [3]i64 = if (command == .NEW) values[0..3].* else switch (preset) {
728 .beginner => .{ 9, 9, 10 },
729 .intermediate => .{ 16, 16, 40 },
730 .expert => .{ 30, 16, 99 },
731 };
732 const width = dimensions[0];
733 const height = dimensions[1];
734 const mines = dimensions[2];
735 const seed = if (command == .NEW) values[3] else values[1];
736 if (width < 1 or width > 30 or height < 1 or height > 30 or mines < 1 or mines > width * height - 9 or seed < 1 or seed > 2147483646)
737 return program.fail("bad value", .{});
738 try program.newGame(@intCast(width), @intCast(height), @intCast(mines), @intCast(seed), null);
739 try program.emit("new {d}x{d} mines {d}", .{ width, height, mines });
740 return;
741 }
742 if (command == .HISTORY and values[0] == 0) return program.fail("bad value", .{});
743 if (command == .SAFETY) {
744 program.safety = safety;
745 try program.emit("safety {s}", .{@tagName(safety)});
746 return;
747 }
748 if (command == .STATS) {
749 const stats = program.stats;
750 try program.out.print("stats played {d} won {d} lost {d} rate ", .{ stats.played, stats.wins, stats.losses });
751 if (stats.played == 0) try program.out.writeAll("-") else try program.out.print("{d}%", .{stats.wins * 100 / stats.played});
752 try program.emit(" streak {d} best {d} fastest {f}", .{ stats.streak, stats.best, OptionalNumber{ .value = stats.fastest } });
753 return;
754 }
755 const game = program.game orelse return program.fail("no game", .{});
756 var cell: usize = 0;
757 if (command.hasCell()) {
758 if (values[0] >= @as(i64, @intCast(game.width)) or values[1] >= @as(i64, @intCast(game.height))) return program.fail("out of range", .{});
759 cell = @as(usize, @intCast(values[1])) * game.width + @as(usize, @intCast(values[0]));
760 }
761 if (game.isFinished() and command.activeOnly()) return program.fail("game over", .{});
762 if (!game.placed and (command == .ANALYZE or command == .BOARDINFO)) return program.fail("not started", .{});
763 const was_finished = game.isFinished();
764 const snapshot: ?Snapshot = if (command.undoable()) game.saveSnapshot(command) else null;
765 switch (command) {
766 .RESTART => {
767 game.restart();
768 try program.emit("restart {d}x{d} mines {d}", .{ game.width, game.height, game.mine_count });
769 },
770 .REVEAL => {
771 if (game.revealed[cell]) return program.fail("already revealed", .{});
772 if (game.marks[cell] == .flag) return program.fail("flagged", .{});
773 try game.appendMove(.reveal, cell);
774 const opened = game.reveal(cell, program.safety);
775 if (game.state != .lost) try program.emit("revealed {d}", .{opened});
776 game.checkWin();
777 try program.finishOutput(game);
778 },
779 .FLAG, .UNFLAG, .QUESTION => {
780 if (game.revealed[cell]) return program.fail("already revealed", .{});
781 if (command == .FLAG and game.marks[cell] == .flag) return program.fail("already flagged", .{});
782 if (command == .UNFLAG and game.marks[cell] != .flag) return program.fail("not flagged", .{});
783 const action: []const u8 = if (command == .FLAG) "flagged" else if (command == .UNFLAG) "unflagged" else if (game.marks[cell] == .question) "unmarked" else "questioned";
784 game.marks[cell] = if (command == .FLAG) .flag else if (command == .UNFLAG or game.marks[cell] == .question) .none else .question;
785 try game.appendMove(if (command == .FLAG) .flag else if (command == .UNFLAG) .unflag else .question, cell);
786 try program.emit("{s} {f} left {d}", .{ action, game.position(cell), game.remainingMines() });
787 },
788 .CHORD => {
789 if (!game.revealed[cell]) return program.fail("not revealed", .{});
790 if (game.numbers[cell] == 0) return program.fail("no number", .{});
791 const flags = game.flagNeighbors(cell);
792 if (flags != game.numbers[cell]) return program.fail("flags {d} of {d}", .{ flags, game.numbers[cell] });
793 if (game.unknownNeighbors(cell).len == 0) return program.fail("nothing to reveal", .{});
794 try game.appendMove(.chord, cell);
795 const opened = game.chord(cell, program.safety);
796 if (game.state != .lost) try program.emit("chorded {d}", .{opened});
797 game.checkWin();
798 try program.finishOutput(game);
799 },
800 .UNDO => {
801 const saved = game.undo.pop() orelse return program.fail("nothing to undo", .{});
802 game.restoreSnapshot(saved);
803 try program.emit("undo {s} moves {d}", .{ saved.command.undoName(), game.moves.items.len });
804 },
805 .SHOW => try game.show(program.out),
806 .STATUS => try program.emit("status {s} left {d} moves {d} revealed {d}/{d}", .{ @tagName(game.state), game.remainingMines(), game.moves.items.len, game.revealedCount(), game.cellCount() - game.mine_count }),
807 .CELL => {
808 if (game.revealed[cell]) {
809 try program.emit("cell {f} number {d} flags {d} unknown {d}", .{ game.position(cell), game.numbers[cell], game.flagNeighbors(cell), game.unknownNeighbors(cell).len });
810 } else {
811 var opened: usize = 0;
812 const nearby = &game.neighbors[cell];
813 for (nearby.cells[0..nearby.len]) |neighbor| if (game.revealed[neighbor]) {
814 opened += 1;
815 };
816 try program.emit("cell {f} hidden mark {s} open {d}", .{ game.position(cell), @tagName(game.marks[cell]), opened });
817 }
818 },
819 .HISTORY => {
820 const length = game.moves.items.len;
821 const count: usize = @intCast(@min(values[0], @as(i64, @intCast(length))));
822 try program.emit("history {d}", .{length});
823 for (game.moves.items[length - count ..], length - count + 1..) |move, i|
824 try program.emit("{d} {s} {f}", .{ i, move.name(), game.position(move.cell) });
825 },
826 .REPLAY => {
827 if (values[0] > @as(i64, @intCast(game.moves.items.len))) return program.fail("no move {d}", .{values[0]});
828 try program.emit("replay {d}/{d}", .{ values[0], game.moves.items.len });
829 const copy = try game.replay(program.scratch, @intCast(values[0]), program.safety);
830 try copy.show(program.out);
831 },
832 .HINT => {
833 const deduction = game.deduce();
834 var found = false;
835 for (deduction.safe[0..game.cellCount()], 0..) |maybe_reason, target| if (maybe_reason) |reason| {
836 try program.emit("hint safe {f} by {f}", .{ game.position(target), Explanation{ .reason = reason, .width = game.width } });
837 found = true;
838 break;
839 };
840 if (!found) for (deduction.mines[0..game.cellCount()], 0..) |maybe_reason, target| if (maybe_reason) |reason| {
841 try program.emit("hint mine {f} by {f}", .{ game.position(target), Explanation{ .reason = reason, .width = game.width } });
842 found = true;
843 break;
844 };
845 if (!found) try program.emit("hint none", .{});
846 },
847 .AUTO, .SOLVE => {
848 const totals = if (command == .AUTO) try game.playAuto(program.safety) else try game.playSolve(program.safety);
849 if (command == .AUTO) try program.emit("auto rounds {d} flagged {d} revealed {d}", .{ totals.rounds, totals.flagged, totals.opened }) else try program.emit("solve guesses {d} rounds {d} flagged {d} revealed {d}", .{ totals.guesses, totals.rounds, totals.flagged, totals.opened });
850 try program.finishOutput(game);
851 },
852 .ANALYZE => {
853 const copy = try game.clone(program.scratch, false);
854 const totals = try copy.playAuto(program.safety);
855 if (copy.state == .lost) try program.emit("analyze boom {f} rounds {d}", .{ copy.position(copy.exploded), totals.rounds }) else if (copy.state == .won) try program.emit("analyze clear rounds {d}", .{totals.rounds}) else try program.emit("analyze stuck rounds {d} revealed {d}/{d}", .{ totals.rounds, copy.revealedCount(), copy.cellCount() - copy.mine_count });
856 },
857 .PROB => {
858 if (game.revealed[cell]) return program.fail("already revealed", .{});
859 if (game.marks[cell] == .flag) return program.fail("flagged", .{});
860 const frontier = try game.buildFrontier(program.scratch);
861 if (frontier.at[cell]) |group| {
862 const component = &frontier.components.items[group];
863 component.count();
864 if (component.configurations == 0) try program.emit("prob {f} inconsistent", .{game.position(cell)}) else {
865 var numerator: usize = 0;
866 for (component.cells.items, 0..) |candidate, i| if (candidate == cell) {
867 numerator = component.mine_counts[i];
868 break;
869 };
870 const denominator = component.configurations;
871 var a = numerator;
872 var b = denominator;
873 while (b != 0) {
874 const remainder = a % b;
875 a = b;
876 b = remainder;
877 }
878 try program.emit("prob {f} {d}/{d} size {d} configs {d}", .{ game.position(cell), numerator / a, denominator / a, component.cells.items.len, denominator });
879 }
880 } else try program.emit("prob {f} unconstrained", .{game.position(cell)});
881 },
882 .FRONTIER => {
883 const frontier = try game.buildFrontier(program.scratch);
884 var count: usize = 0;
885 for (frontier.components.items) |component| count += component.cells.items.len;
886 try program.emit("frontier {d} components {d}", .{ count, frontier.components.items.len });
887 for (frontier.components.items) |*component| {
888 component.count();
889 if (component.configurations == 0) try program.emit("component {f} size {d} inconsistent", .{ game.position(component.cells.items[0]), component.cells.items.len }) else try program.emit("component {f} size {d} configs {d} min {d} max {d}", .{ game.position(component.cells.items[0]), component.cells.items.len, component.configurations, component.minimum, component.maximum });
890 }
891 },
892 .BOARDINFO => {
893 const info = game.boardInfo();
894 try program.emit("3bv {d} openings {d} largest {d} mines {d} safe {d}", .{ info.bv, info.openings, info.largest, game.mine_count, game.cellCount() - game.mine_count });
895 },
896 .NEW, .PRESET, .LAYOUT, .SAFETY, .STATS => unreachable,
897 }
898 if (snapshot) |saved| if (game.moves.items.len > saved.move_count) {
899 try game.undo.append(game.gpa, saved);
900 };
901 if (!was_finished and game.isFinished()) program.stats.record(game);
902 }
903};

905fn readLine(gpa: Allocator, input: *std.Io.Reader) ![]const u8 {
906 var buffer: std.Io.Writer.Allocating = .init(gpa);
907 defer buffer.deinit();
908 _ = try input.streamDelimiterEnding(&buffer.writer, '\n');
909 if (input.seek < input.end) input.toss(1) else if (buffer.written().len == 0) return error.EndOfStream;
910 const line = try buffer.toOwnedSlice();
911 return if (line.len > 0 and line[line.len - 1] == '\r') line[0 .. line.len - 1] else line;
912}
913fn readWords(gpa: Allocator, input: *std.Io.Reader) ![]const []const u8 {
914 const line = try readLine(gpa, input);
915 var words: std.ArrayList([]const u8) = .empty;
916 var parts = std.mem.tokenizeScalar(u8, line, ' ');
917 while (parts.next()) |word| try words.append(gpa, word);
918 return try words.toOwnedSlice(gpa);
919}

921pub fn main(init: std.process.Init) !void {
922 var scratch = std.heap.ArenaAllocator.init(std.heap.page_allocator);
923 defer scratch.deinit();
924 var input_buffer: [4096]u8 = undefined;
925 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
926 var output_buffer: [4096]u8 = undefined;
927 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
928 var program: Program = .{ .arena = .init(std.heap.page_allocator), .scratch = scratch.allocator(), .input = &input.interface, .out = &output.interface };
929 defer program.arena.deinit();
930 const count = std.fmt.parseInt(usize, (try readWords(scratch.allocator(), &input.interface))[0], 10) catch unreachable;
931 for (0..count) |_| {
932 _ = scratch.reset(.retain_capacity);
933 const words = try readWords(scratch.allocator(), &input.interface);
934 var response: std.Io.Writer.Allocating = .init(scratch.allocator());
935 defer response.deinit();
936 program.out = &response.writer;
937 program.execute(words) catch |err| {
938 if (err != error.BadCommand) return err;
939 try output.interface.print("error: {s}\n", .{program.error_message});
940 continue;
941 };
942 try output.interface.writeAll(response.written());
943 }
944 try output.interface.flush();
945}
