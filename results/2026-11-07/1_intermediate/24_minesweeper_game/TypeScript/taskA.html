53function fraction(numerator: number, denominator: number): string {
54 let a = numerator;
55 let b = denominator;
56 while (b !== 0) [a, b] = [b, a % b];
57 return `${numerator / a}/${denominator / a}`;
58}

62 readonly neighbors: readonly (readonly number[])[];

64 numbers: readonly number[];
65 visible: boolean[] = [];

67 moves: Move[] = [];

72 constructor(readonly width: number, readonly height: number,
73 readonly mineCount: number, readonly seed: number) {
74 this.size = width * height;
75 this.numbers = Array<number>(this.size).fill(0);
76 this.neighbors = Array.from({ length: this.size }, (_, cell) => {
77 const result: number[] = [];
78 const x = cell % width;
79 const y = Math.floor(cell / width);
80 for (let dy = -1; dy <= 1; dy++) {
81 for (let dx = -1; dx <= 1; dx++) {
82 if ((dx === 0 && dy === 0) || x + dx < 0 || x + dx >= width ||
83 y + dy < 0 || y + dy >= height) continue;
84 result.push((y + dy) * width + x + dx);
85 }
86 }
87 return result;
88 });
89 this.restart();
90 }

132 place(first: number, safety: Safety): void {
133 const x = first % this.width;
134 const y = Math.floor(first / this.width);
135 const candidates: number[] = [];
136 for (let cell = 0; cell < this.size; cell++) {
137 if (safety === 'none' || (safety === 'cell' && cell !== first) ||
138 (safety === 'block' && (Math.abs(cell % this.width - x) >= 2 ||
139 Math.abs(Math.floor(cell / this.width) - y) >= 2))) candidates.push(cell);
140 }
141 const mines = Array<boolean>(this.size).fill(false);
142 let random = this.seed;
143 for (let i = 0; i < this.mineCount; i++) {
144 random = 48271 * random % 2147483647;
145 const index = random % candidates.length;
146 mines[candidates[index]] = true;
147 candidates.splice(index, 1);
148 }
149 this.determine(mines);
150 }

152 reveal(cell: number, safety: Safety): number {
153 if (this.mines === null) this.place(cell, safety);
154 if (this.mines![cell]) {
155 this.state = 'lost';
156 this.exploded = cell;
157 return 0;
158 }
159 const queue = [cell];
160 this.visible[cell] = true;
161 this.marks[cell] = 'none';
162 for (let head = 0; head < queue.length; head++) {
163 const current = queue[head];
164 if (this.numbers[current] !== 0) continue;
165 for (const next of this.neighbors[current]) {
166 if (!this.unknown(next)) continue;
167 this.visible[next] = true;
168 this.marks[next] = 'none';
169 queue.push(next);
170 }
171 }
172 return queue.length;
173 }

175 chord(cell: number, safety: Safety): number {
176 let opened = 0;
177 for (const next of this.neighbors[cell]) {
178 if (this.unknown(next)) opened += this.reveal(next, safety);
179 if (this.state === 'lost') break;
180 }
181 return opened;
182 }

184 checkWin(): void {
185 if (this.state !== 'playing' || this.revealedCount !== this.safeCount) return;
186 this.state = 'won';
187 for (let cell = 0; cell < this.size; cell++) {
188 if (this.mines![cell]) this.marks[cell] = 'flag';
189 }
190 }

192 display(): string[] {
193 const cells = this.visible.map((visible, cell) => {
194 const mark = this.marks[cell];
195 if (this.state === 'lost') {
196 if (this.mines![cell]) return cell === this.exploded ? '!' : mark === 'flag' ? 'F' : '*';
197 if (mark === 'flag') return 'X';
198 }
199 if (visible) return this.numbers[cell] === 0 ? '.' : String(this.numbers[cell]);
200 return mark === 'flag' ? 'F' : mark === 'question' ? '?' : '#';
201 });
202 return Array.from({ length: this.height }, (_, y) =>
203 cells.slice(y * this.width, (y + 1) * this.width).join(''));
204 }
205
206 finishOutput(): string[] {
207 if (this.state === 'lost') return [`boom ${this.coordinate(this.exploded!)}`, ...this.display()];
208 if (this.state === 'won') return [`win moves ${this.moves.length}`];
209 return [];
210 }

212 constraints(): Constraint[] {
213 const result: Constraint[] = [];
214 for (let cell = 0; cell < this.size; cell++) {
215 if (!this.visible[cell] || this.numbers[cell] === 0) continue;
216 const cells = this.neighbors[cell].filter(next => this.unknown(next));
217 if (cells.length === 0) continue;
218 const flags = this.neighbors[cell].filter(next => this.marks[next] === 'flag').length;
219 result.push({ cell, cells, remaining: this.numbers[cell] - flags });
220 }
221 return result;
222 }

224 deduce(): Deduction {
225 const constraints = this.constraints();
226 const safe = new Map<number, string>();
227 const mines = new Map<number, string>();
228 const include = (target: Map<number, string>, cells: readonly number[], reason: string): void => {
229 for (const cell of cells) if (!target.has(cell)) target.set(cell, reason);
230 };
231 for (const c of constraints) {
232 if (c.remaining === 0) include(safe, c.cells, `single ${this.coordinate(c.cell)}`);
233 }
234 if (safe.size !== 0) return { safe, mines };
235 for (const c of constraints) {
236 if (c.remaining === c.cells.length) include(mines, c.cells, `single ${this.coordinate(c.cell)}`);
237 }
238 if (mines.size !== 0) return { safe, mines };
239 for (const a of constraints) {
240 for (const b of constraints) {
241 if (a.cells.length >= b.cells.length || !a.cells.every(cell => b.cells.includes(cell))) continue;
242 const difference = b.cells.filter(cell => !a.cells.includes(cell));
243 const remaining = b.remaining - a.remaining;
244 const reason = `subset ${this.coordinate(a.cell)} ${this.coordinate(b.cell)}`;
245 if (remaining === 0) include(safe, difference, reason);
246 if (remaining === difference.length) include(mines, difference, reason);
247 }
248 }
249 if (safe.size === 0 && mines.size === 0 && this.state === 'playing' && this.remaining === 0) {
250 for (let cell = 0; cell < this.size; cell++) if (this.unknown(cell)) safe.set(cell, 'count');
251 }
252 return { safe, mines };
253 }

255 frontier(): Component[] {
256 const constraints = this.constraints();
257 const parent = Array.from({ length: this.size }, (_, cell) => cell);
258 const present = Array<boolean>(this.size).fill(false);
259 const root = (cell: number): number => {
260 while (parent[cell] !== cell) {
261 parent[cell] = parent[parent[cell]];
262 cell = parent[cell];
263 }
264 return cell;
265 };
266 for (const c of constraints) {
267 for (const cell of c.cells) {
268 present[cell] = true;
269 parent[root(cell)] = root(c.cells[0]);
270 }
271 }
272 const groups = new Map<number, Component>();
273 for (let cell = 0; cell < this.size; cell++) {
274 if (!present[cell]) continue;
275 const representative = root(cell);
276 if (!groups.has(representative)) groups.set(representative, { cells: [], constraints: [] });
277 groups.get(representative)!.cells.push(cell);
278 }
279 for (const c of constraints) groups.get(root(c.cells[0]))!.constraints.push(c);
280 return [...groups.values()];
281 }

283 auto(safety: Safety): AutoTotals {
284 const totals: AutoTotals = { rounds: 0, flagged: 0, opened: 0 };
285 while (this.state === 'playing') {
286 const { safe, mines } = this.deduce();
287 if (safe.size === 0 && mines.size === 0) break;
288 totals.rounds++;
289 for (const cell of [...safe.keys()].sort((a, b) => a - b)) {
290 if (this.visible[cell]) continue;
291 this.moves.push({ kind: 'auto-reveal', cell });
292 totals.opened += this.reveal(cell, safety);
293 this.checkWin();
294 if (this.finished) return totals;
295 }
296 for (const cell of [...mines.keys()].sort((a, b) => a - b)) {
297 if (!this.unknown(cell)) continue;
298 this.marks[cell] = 'flag';
299 totals.flagged++;
300 this.moves.push({ kind: 'auto-flag', cell });
301 }
302 }
303 return totals;
304 }

306 solve(safety: Safety): SolveTotals {
307 const totals: SolveTotals = { guesses: 0, rounds: 0, flagged: 0, opened: 0 };
308 while (!this.finished) {
309 const automatic = this.auto(safety);
310 totals.rounds += automatic.rounds;
311 totals.flagged += automatic.flagged;
312 totals.opened += automatic.opened;
313 if (this.finished) break;
314 const probabilities = Array<Probability | null>(this.size).fill(null);
315 const onFrontier = Array<boolean>(this.size).fill(false);
316 for (const component of this.frontier()) {
317 const counts = countConfigurations(component);
318 component.cells.forEach((cell, index) => {
319 onFrontier[cell] = true;
320 if (counts.configs !== 0) probabilities[cell] = [counts.frequencies[index], counts.configs];
321 });
322 }
323 let unknownCount = 0;
324 for (let cell = 0; cell < this.size; cell++) if (this.unknown(cell)) unknownCount++;
325 const unconstrainedNumerator = Math.max(this.remaining, 0);
326 let chosen = -1;
327 let bestNumerator = 0;
328 let bestDenominator = 1;
329 for (let cell = 0; cell < this.size; cell++) {
330 if (!this.unknown(cell)) continue;
331 const probability: Probability | null = onFrontier[cell] ? probabilities[cell] : [unconstrainedNumerator, unknownCount];
332 if (probability === null) continue;
333 const [numerator, denominator] = probability;
334 if (chosen === -1 || numerator * bestDenominator < bestNumerator * denominator) {
335 chosen = cell;
336 bestNumerator = numerator;
337 bestDenominator = denominator;
338 }
339 }
340 if (chosen === -1) break;
341 totals.guesses++;
342 this.moves.push({ kind: 'guess', cell: chosen });
343 totals.opened += this.reveal(chosen, safety);
344 this.checkWin();
345 }
346 return totals;
347 }

349 replay(count: number, safety: Safety): string[] {
350 const board = this.copy();
351 board.restart();
352 for (const move of this.moves.slice(0, count)) {
353 switch (move.kind) {
354 case 'reveal': case 'auto-reveal': case 'guess': board.reveal(move.cell, safety); break;
355 case 'chord': board.chord(move.cell, safety); break;
356 case 'flag': case 'auto-flag': board.marks[move.cell] = 'flag'; break;
357 case 'unflag': board.marks[move.cell] = 'none'; break;
358 case 'question': board.marks[move.cell] = board.marks[move.cell] === 'question' ? 'none' : 'question'; break;
359 }
360 board.checkWin();
361 }
362 return board.display();
363 }

365 boardInfo(): string {
366 const mines = this.mines!;
367 const visited = Array<boolean>(this.size).fill(false);
368 let openings = 0;
369 let largest = 0;
370 let isolatedNumbers = 0;
371 for (let cell = 0; cell < this.size; cell++) {
372 if (mines[cell]) continue;
373 if (this.numbers[cell] !== 0) {
374 if (!this.neighbors[cell].some(next => !mines[next] && this.numbers[next] === 0)) isolatedNumbers++;
375 continue;
376 }
377 if (visited[cell]) continue;
378 openings++;
379 const queue = [cell];
380 const opening = new Set([cell]);
381 visited[cell] = true;
382 for (let head = 0; head < queue.length; head++) {
383 for (const next of this.neighbors[queue[head]]) {
384 opening.add(next);
385 if (!mines[next] && this.numbers[next] === 0 && !visited[next]) {
386 visited[next] = true;
387 queue.push(next);
388 }
389 }
390 }
391 largest = Math.max(largest, opening.size);
392 }
393 return `3bv ${openings + isolatedNumbers} openings ${openings} largest ${largest} mines ${this.mineCount} safe ${this.safeCount}`;
394 }

397function countConfigurations(component: Component): ConfigurationCounts {
398 const positions = new Map<number, number>(component.cells.map((cell, index): [number, number] => [cell, index]));
399 const constraints = component.constraints.map(c => ({
400 cells: c.cells.map(cell => positions.get(cell)!), remaining: c.remaining,
401 }));
402 const incidence = Array.from({ length: component.cells.length }, (): number[] => []);
403 constraints.forEach((c, index) => {
404 for (const cell of c.cells) incidence[cell].push(index);
405 });
406 const assigned = Array<number>(constraints.length).fill(0);
407 const unassigned = constraints.map(c => c.cells.length);
408 const values = Array<number>(component.cells.length).fill(0);
409 const result: ConfigurationCounts = { configs: 0, minimum: component.cells.length, maximum: 0, frequencies: values.slice() };
410 if (constraints.some(c => c.remaining < 0 || c.remaining > c.cells.length)) return result;
411 const order = component.cells.map((_, index) => index).sort((a, b) => incidence[b].length - incidence[a].length || a - b);
412 const search = (depth: number, mineCount: number): void => {
413 if (depth === order.length) {
414 result.configs++;
415 result.minimum = Math.min(result.minimum, mineCount);
416 result.maximum = Math.max(result.maximum, mineCount);
417 values.forEach((value, index) => { result.frequencies[index] += value; });
418 return;
419 }
420 const cell = order[depth];
421 for (let value = 0; value <= 1; value++) {
422 values[cell] = value;
423 let possible = true;
424 for (const index of incidence[cell]) {
425 assigned[index] += value;
426 unassigned[index]--;
427 const target = constraints[index].remaining;
428 if (assigned[index] > target || assigned[index] + unassigned[index] < target) possible = false;
429 }
430 if (possible) search(depth + 1, mineCount + value);
431 for (const index of incidence[cell]) {
432 assigned[index] -= value;
433 unassigned[index]++;
434 }
435 }
436 values[cell] = 0;
437 };
438 search(0, 0);
439 return result;
440}

442const formats = {
443 NEW: 'nnnn', PRESET: 'pn', SAFETY: 's', RESTART: '',
444 REVEAL: 'nn', FLAG: 'nn', UNFLAG: 'nn', QUESTION: 'nn', CHORD: 'nn',
445 UNDO: '', SHOW: '', STATUS: '', CELL: 'nn', HISTORY: 'n', REPLAY: 'n',

473 execute(words: readonly string[], rows: readonly string[]): string[] {
474 const [command, ...args] = words;
475 if (command === 'LAYOUT') {
476 const [width, height] = args.map(Number);
477 const badRow = rows.findIndex(row => row.length !== width || /[^*.]/.test(row));
478 if (badRow !== -1) fail(`bad row ${badRow + 1}`);
479 const mines = rows.join('').split('').map(cell => cell === '*');
480 const count = mines.filter(Boolean).length;
481 if (count === 0 || count === width * height) fail(`bad mine count ${count}`);
482 this.board = new Board(width, height, count, 1);
483 this.board.determine(mines);
484 return [`layout ${width}x${height} mines ${count}`];
485 }
486 if (!isCommand(command)) fail(`unknown command ${command}`);
487 const format = formats[command];
488 if (args.length !== format.length) fail('bad arguments');
489 for (let i = 0; i < args.length; i++) {
490 const valid = format[i] === 'n' ? /^(?:0|[1-9][0-9]{0,9})$/.test(args[i]) :
491 format[i] === 'p' ? presets.has(args[i]) : isSafety(args[i]);
492 if (!valid) fail('bad arguments');
493 }
494 const numbers = args.map(Number);
495 if (command === 'NEW' || command === 'PRESET') {
496 const [width, height, count, seed] = command === 'NEW' ? numbers : [...presets.get(args[0])!, numbers[1]];
497 if (width < 1 || width > 30 || height < 1 || height > 30 || count < 1 ||
498 count > width * height - 9 || seed < 1 || seed > 2147483646) fail('bad value');
499 this.board = new Board(width, height, count, seed);
500 return [`new ${width}x${height} mines ${count}`];
501 }
502 if (command === 'HISTORY' && numbers[0] === 0) fail('bad value');
503 if (command === 'SAFETY') {
504 this.safety = args[0] as Safety;
505 return [`safety ${this.safety}`];
506 }
507 if (command === 'STATS') {
508 const s = this.stats;
509 const rate = s.played === 0 ? '-' : `${Math.floor(s.won * 100 / s.played)}%`;
510 return [`stats played ${s.played} won ${s.won} lost ${s.lost} rate ${rate} streak ${s.streak} best ${s.best} fastest ${s.fastest ?? '-'}`];
511 }
512 const board = this.board;
513 if (board === null) fail('no game');
514 if (cellCommands.has(command) && (numbers[0] >= board.width || numbers[1] >= board.height)) fail('out of range');
515 if (liveCommands.has(command) && board.finished) fail('game over');
516 if ((command === 'ANALYZE' || command === 'BOARDINFO') && board.state === 'ready') fail('not started');
517 const cell = numbers[1] * board.width + numbers[0];
518 const undoKind = undoKinds.get(command);
519 const snapshot: UndoSnapshot | null = undoKind === undefined ? null : {
520 command: undoKind, visible: [...board.visible], marks: [...board.marks], moveCount: board.moves.length,
521 };
522 const response = this.boardCommand(command, numbers, cell, board);
523 if (snapshot !== null && board.moves.length > snapshot.moveCount) board.undo.push(snapshot);
524 if (['REVEAL', 'CHORD', 'AUTO', 'SOLVE'].includes(command) && board.finished) {
525 const s = this.stats;
526 s.played++;
527 if (board.state === 'won') {
528 s.won++;
529 s.streak++;
530 s.best = Math.max(s.best, s.streak);
531 s.fastest = Math.min(s.fastest ?? board.moves.length, board.moves.length);
532 } else {
533 s.lost++;
534 s.streak = 0;
535 }
536 }
537 return response;
538 }

540 boardCommand(command: BoardCommand, numbers: readonly number[], cell: number, board: Board): string[] {
541 const coordinate = () => board.coordinate(cell);
542 switch (command) {
543 case 'RESTART':
544 board.restart();
545 return [`restart ${board.width}x${board.height} mines ${board.mineCount}`];
546 case 'REVEAL': case 'FLAG': case 'UNFLAG': case 'QUESTION': {
547 if (board.visible[cell]) fail('already revealed');
548 const mark = board.marks[cell];
549 if (command === 'REVEAL' && mark === 'flag') fail('flagged');
550 if (command === 'FLAG' && mark === 'flag') fail('already flagged');
551 if (command === 'UNFLAG' && mark !== 'flag') fail('not flagged');
552 board.moves.push({ kind: manualMoveKinds[command], cell });
553 if (command === 'REVEAL') {
554 const opened = board.reveal(cell, this.safety);
555 if (board.state === 'lost') return board.finishOutput();
556 board.checkWin();
557 return [`revealed ${opened}`, ...board.finishOutput()];
558 }
559 board.marks[cell] = command === 'FLAG' ? 'flag' : command === 'UNFLAG' || mark === 'question' ? 'none' : 'question';
560 const action = command === 'FLAG' ? 'flagged' : command === 'UNFLAG' ? 'unflagged' : mark === 'question' ? 'unmarked' : 'questioned';
561 return [`${action} ${coordinate()} left ${board.remaining}`];
562 }
563 case 'CHORD': {
564 if (!board.visible[cell]) fail('not revealed');
565 const number = board.numbers[cell];
566 if (number === 0) fail('no number');
567 const flags = board.neighbors[cell].filter(next => board.marks[next] === 'flag').length;
568 if (flags !== number) fail(`flags ${flags} of ${number}`);
569 if (!board.neighbors[cell].some(next => board.unknown(next))) fail('nothing to reveal');
570 board.moves.push({ kind: 'chord', cell });
571 const opened = board.chord(cell, this.safety);
572 if (board.state === 'lost') return board.finishOutput();
573 board.checkWin();
574 return [`chorded ${opened}`, ...board.finishOutput()];
575 }
576 case 'UNDO': {
577 if (board.undo.length === 0) fail('nothing to undo');
578 const previous = board.undo.pop()!;
579 board.visible = [...previous.visible];
580 board.marks = [...previous.marks];
581 board.moves.length = previous.moveCount;
582 return [`undo ${previous.command} moves ${board.moves.length}`];
583 }
584 case 'SHOW': return board.display();
585 case 'STATUS': return [`status ${board.state} left ${board.remaining} moves ${board.moves.length} revealed ${board.revealedCount}/${board.safeCount}`];
586 case 'CELL': {
587 const neighbors = board.neighbors[cell];
588 if (!board.visible[cell]) return [`cell ${coordinate()} hidden mark ${board.marks[cell]} open ${neighbors.filter(next => board.visible[next]).length}`];
589 const flags = neighbors.filter(next => board.marks[next] === 'flag').length;
590 return [`cell ${coordinate()} number ${board.numbers[cell]} flags ${flags} unknown ${neighbors.filter(next => board.unknown(next)).length}`];
591 }
592 case 'HISTORY': {
593 const lines = [`history ${board.moves.length}`];
594 for (let i = Math.max(0, board.moves.length - numbers[0]); i < board.moves.length; i++) {
595 lines.push(`${i + 1} ${board.moves[i].kind} ${board.coordinate(board.moves[i].cell)}`);
596 }
597 return lines;
598 }
599 case 'REPLAY':
600 if (numbers[0] > board.moves.length) fail(`no move ${numbers[0]}`);
601 return [`replay ${numbers[0]}/${board.moves.length}`, ...board.replay(numbers[0], this.safety)];
602 case 'HINT': {
603 const { safe, mines } = board.deduce();
604 const choices = safe.size !== 0 ? safe : mines;
605 if (choices.size === 0) return ['hint none'];
606 const chosen = Math.min(...choices.keys());
607 return [`hint ${safe.size !== 0 ? 'safe' : 'mine'} ${board.coordinate(chosen)} by ${choices.get(chosen)}`];
608 }
609 case 'AUTO': {
610 const result = board.auto(this.safety);
611 return [`auto rounds ${result.rounds} flagged ${result.flagged} revealed ${result.opened}`, ...board.finishOutput()];
612 }
613 case 'ANALYZE': {
614 const copy = board.copy();
615 const result = copy.auto(this.safety);
616 if (copy.state === 'lost') return [`analyze boom ${copy.coordinate(copy.exploded!)} rounds ${result.rounds}`];
617 if (copy.state === 'won') return [`analyze clear rounds ${result.rounds}`];
618 return [`analyze stuck rounds ${result.rounds} revealed ${copy.revealedCount}/${copy.safeCount}`];
619 }
620 case 'PROB': {
621 if (board.visible[cell]) fail('already revealed');
622 if (board.marks[cell] === 'flag') fail('flagged');
623 const component = board.frontier().find(group => group.cells.includes(cell));
624 if (component === undefined) return [`prob ${coordinate()} unconstrained`];
625 const result = countConfigurations(component);
626 if (result.configs === 0) return [`prob ${coordinate()} inconsistent`];
627 return [`prob ${coordinate()} ${fraction(result.frequencies[component.cells.indexOf(cell)], result.configs)} size ${component.cells.length} configs ${result.configs}`];
628 }
629 case 'FRONTIER': {
630 const components = board.frontier();
631 const lines = [`frontier ${components.reduce((sum, c) => sum + c.cells.length, 0)} components ${components.length}`];
632 for (const component of components) {
633 const result = countConfigurations(component);
634 const prefix = `component ${board.coordinate(component.cells[0])} size ${component.cells.length}`;
635 lines.push(result.configs === 0 ? `${prefix} inconsistent` : `${prefix} configs ${result.configs} min ${result.minimum} max ${result.maximum}`);
636 }
637 return lines;
638 }
639 case 'SOLVE': {
640 const result = board.solve(this.safety);
641 return [`solve guesses ${result.guesses} rounds ${result.rounds} flagged ${result.flagged} revealed ${result.opened}`, ...board.finishOutput()];
642 }
643 case 'BOARDINFO': return [board.boardInfo()];
644 }
645 throw new Error('Unreachable command');
646 }
647}

649const lines = readFileSync(0, 'utf8').split(/\r?\n/);
650const commandCount = Number(lines[0]);
651const program = new Program();
652let cursor = 1;
653for (let i = 0; i < commandCount; i++) {
654 const words = lines[cursor++].split(' ');
655 const rowCount = words[0] === 'LAYOUT' ? Number(words[2]) : 0;
656 const rows = lines.slice(cursor, cursor + rowCount);
657 cursor += rowCount;
658 try {
659 process.stdout.write(program.execute(words, rows).join('\n') + '\n');
660 } catch (error) {
661 if (!(error instanceof CommandError)) throw error;
662 process.stdout.write(`error: ${error.message}\n`);
663 }
