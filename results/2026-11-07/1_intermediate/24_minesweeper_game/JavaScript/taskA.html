9function fraction(numerator, denominator) {
10 let a = numerator;
11 let b = denominator;
12 while (b !== 0) [a, b] = [b, a % b];
13 return `${numerator / a}/${denominator / a}`;
14}

16class Board {
17 constructor(width, height, mineCount, seed) {
18 this.width = width;
19 this.height = height;
20 this.mineCount = mineCount;
21 this.seed = seed;
22 this.size = width * height;
23 this.mines = null;
24 this.numbers = Array(this.size).fill(0);
25 this.neighbors = Array.from({ length: this.size }, (_, cell) => {
26 const result = [];
27 const x = cell % width;
28 const y = Math.floor(cell / width);
29 for (let dy = -1; dy <= 1; dy++) {
30 for (let dx = -1; dx <= 1; dx++) {
31 if ((dx === 0 && dy === 0) || x + dx < 0 || x + dx >= width ||
32 y + dy < 0 || y + dy >= height) continue;
33 result.push((y + dy) * width + x + dx);
34 }
35 }
36 return result;
37 });
38 this.restart();
39 }

41 restart() {
42 this.visible = Array(this.size).fill(false);
43 this.marks = Array(this.size).fill('none');
44 this.moves = [];
45 this.undo = [];
46 this.exploded = -1;
47 this.state = this.mines === null ? 'ready' : 'playing';
48 }

50 copy() {
51 const board = Object.assign(Object.create(Board.prototype), this);
52 board.visible = [...this.visible];
53 board.marks = [...this.marks];
54 board.moves = [...this.moves];
55 board.undo = [];
56 return board;

59 coordinate(cell) {
60 return `${cell % this.width},${Math.floor(cell / this.width)}`;

62
63 get finished() { return this.state === 'won' || this.state === 'lost'; }
64 get safeCount() { return this.size - this.mineCount; }
65 get revealedCount() { return this.visible.filter(Boolean).length; }
66 get remaining() { return this.mineCount - this.marks.filter(mark => mark === 'flag').length; }

68 unknown(cell) {
69 return !this.visible[cell] && this.marks[cell] !== 'flag';
70 }
71
72 determine(mines) {
73 this.mines = mines;
74 this.numbers = this.neighbors.map(cells => cells.filter(cell => mines[cell]).length);
75 this.state = 'playing';

78 place(first, safety) {
79 const x = first % this.width;
80 const y = Math.floor(first / this.width);
81 const candidates = [];
82 for (let cell = 0; cell < this.size; cell++) {
83 if (safety === 'none' || (safety === 'cell' && cell !== first) ||
84 (safety === 'block' && (Math.abs(cell % this.width - x) >= 2 ||
85 Math.abs(Math.floor(cell / this.width) - y) >= 2))) candidates.push(cell);
86 }
87 const mines = Array(this.size).fill(false);
88 let random = this.seed;
89 for (let i = 0; i < this.mineCount; i++) {
90 random = 48271 * random % 2147483647;
91 const index = random % candidates.length;
92 mines[candidates[index]] = true;
93 candidates.splice(index, 1);
94 }
95 this.determine(mines);
96 }

98 reveal(cell, safety) {
99 if (this.mines === null) this.place(cell, safety);
100 if (this.mines[cell]) {
101 this.state = 'lost';
102 this.exploded = cell;
103 return 0;
104 }
105 const queue = [cell];
106 this.visible[cell] = true;
107 this.marks[cell] = 'none';
108 for (let head = 0; head < queue.length; head++) {
109 const current = queue[head];
110 if (this.numbers[current] !== 0) continue;
111 for (const next of this.neighbors[current]) {
112 if (!this.unknown(next)) continue;
113 this.visible[next] = true;
114 this.marks[next] = 'none';
115 queue.push(next);
116 }
117 }
118 return queue.length;
119 }

121 chord(cell, safety) {
122 let opened = 0;
123 for (const next of this.neighbors[cell]) {
124 if (this.unknown(next)) opened += this.reveal(next, safety);
125 if (this.state === 'lost') break;
126 }
127 return opened;
128 }
129
130 checkWin() {
131 if (this.state !== 'playing' || this.revealedCount !== this.safeCount) return;
132 this.state = 'won';
133 for (let cell = 0; cell < this.size; cell++) {
134 if (this.mines[cell]) this.marks[cell] = 'flag';
135 }
136 }
137
138 display() {
139 const cells = this.visible.map((visible, cell) => {
140 const mark = this.marks[cell];
141 if (this.state === 'lost') {
142 if (this.mines[cell]) return cell === this.exploded ? '!' : mark === 'flag' ? 'F' : '*';
143 if (mark === 'flag') return 'X';
144 }
145 if (visible) return this.numbers[cell] === 0 ? '.' : String(this.numbers[cell]);
146 return mark === 'flag' ? 'F' : mark === 'question' ? '?' : '#';
147 });
148 return Array.from({ length: this.height }, (_, y) =>
149 cells.slice(y * this.width, (y + 1) * this.width).join(''));
150 }
151
152 finishOutput() {
153 if (this.state === 'lost') return [`boom ${this.coordinate(this.exploded)}`, ...this.display()];
154 if (this.state === 'won') return [`win moves ${this.moves.length}`];
155 return [];
156 }
157
158 constraints() {
159 const result = [];
160 for (let cell = 0; cell < this.size; cell++) {
161 if (!this.visible[cell] || this.numbers[cell] === 0) continue;
162 const cells = this.neighbors[cell].filter(next => this.unknown(next));
163 if (cells.length === 0) continue;
164 const flags = this.neighbors[cell].filter(next => this.marks[next] === 'flag').length;
165 result.push({ cell, cells, remaining: this.numbers[cell] - flags });
166 }
167 return result;
168 }

170 deduce() {
171 const constraints = this.constraints();
172 const safe = new Map();
173 const mines = new Map();
174 const include = (target, cells, reason) => {
175 for (const cell of cells) if (!target.has(cell)) target.set(cell, reason);
176 };
177 for (const c of constraints) {
178 if (c.remaining === 0) include(safe, c.cells, `single ${this.coordinate(c.cell)}`);
179 }
180 if (safe.size !== 0) return { safe, mines };
181 for (const c of constraints) {
182 if (c.remaining === c.cells.length) include(mines, c.cells, `single ${this.coordinate(c.cell)}`);
183 }
184 if (mines.size !== 0) return { safe, mines };
185 for (const a of constraints) {
186 for (const b of constraints) {
187 if (a.cells.length >= b.cells.length || !a.cells.every(cell => b.cells.includes(cell))) continue;
188 const difference = b.cells.filter(cell => !a.cells.includes(cell));
189 const remaining = b.remaining - a.remaining;
190 const reason = `subset ${this.coordinate(a.cell)} ${this.coordinate(b.cell)}`;
191 if (remaining === 0) include(safe, difference, reason);
192 if (remaining === difference.length) include(mines, difference, reason);
193 }
194 }
195 if (safe.size === 0 && mines.size === 0 && this.state === 'playing' && this.remaining === 0) {
196 for (let cell = 0; cell < this.size; cell++) if (this.unknown(cell)) safe.set(cell, 'count');
197 }
198 return { safe, mines };
199 }

201 frontier() {
202 const constraints = this.constraints();
203 const parent = Array.from({ length: this.size }, (_, cell) => cell);
204 const present = Array(this.size).fill(false);
205 const root = cell => {
206 while (parent[cell] !== cell) {
207 parent[cell] = parent[parent[cell]];
208 cell = parent[cell];
209 }
210 return cell;
211 };
212 for (const c of constraints) {
213 for (const cell of c.cells) {
214 present[cell] = true;
215 parent[root(cell)] = root(c.cells[0]);
216 }
217 }
218 const groups = new Map();
219 for (let cell = 0; cell < this.size; cell++) {
220 if (!present[cell]) continue;
221 const representative = root(cell);
222 if (!groups.has(representative)) groups.set(representative, { cells: [], constraints: [] });
223 groups.get(representative).cells.push(cell);
224 }
225 for (const c of constraints) groups.get(root(c.cells[0])).constraints.push(c);
226 return [...groups.values()];
227 }

229 auto(safety) {
230 const totals = { rounds: 0, flagged: 0, opened: 0 };
231 while (this.state === 'playing') {
232 const { safe, mines } = this.deduce();
233 if (safe.size === 0 && mines.size === 0) break;
234 totals.rounds++;
235 for (const cell of [...safe.keys()].sort((a, b) => a - b)) {
236 if (this.visible[cell]) continue;
237 this.moves.push({ kind: 'auto-reveal', cell });
238 totals.opened += this.reveal(cell, safety);
239 this.checkWin();
240 if (this.finished) return totals;
241 }
242 for (const cell of [...mines.keys()].sort((a, b) => a - b)) {
243 if (!this.unknown(cell)) continue;
244 this.marks[cell] = 'flag';
245 totals.flagged++;
246 this.moves.push({ kind: 'auto-flag', cell });
247 }
248 }
249 return totals;
250 }
251
252 solve(safety) {
253 const totals = { guesses: 0, rounds: 0, flagged: 0, opened: 0 };
254 while (!this.finished) {
255 const automatic = this.auto(safety);
256 totals.rounds += automatic.rounds;
257 totals.flagged += automatic.flagged;
258 totals.opened += automatic.opened;
259 if (this.finished) break;
260 const probabilities = Array(this.size).fill(null);
261 const onFrontier = Array(this.size).fill(false);
262 for (const component of this.frontier()) {
263 const counts = countConfigurations(component);
264 component.cells.forEach((cell, index) => {
265 onFrontier[cell] = true;
266 if (counts.configs !== 0) probabilities[cell] = [counts.frequencies[index], counts.configs];
267 });
268 }
269 let unknownCount = 0;
270 for (let cell = 0; cell < this.size; cell++) if (this.unknown(cell)) unknownCount++;
271 const unconstrainedNumerator = Math.max(this.remaining, 0);
272 let chosen = -1;
273 let bestNumerator = 0;
274 let bestDenominator = 1;
275 for (let cell = 0; cell < this.size; cell++) {
276 if (!this.unknown(cell)) continue;
277 const probability = onFrontier[cell] ? probabilities[cell] : [unconstrainedNumerator, unknownCount];
278 if (probability === null) continue;
279 const [numerator, denominator] = probability;
280 if (chosen === -1 || numerator * bestDenominator < bestNumerator * denominator) {
281 chosen = cell;
282 bestNumerator = numerator;
283 bestDenominator = denominator;
284 }
285 }
286 if (chosen === -1) break;
287 totals.guesses++;
288 this.moves.push({ kind: 'guess', cell: chosen });
289 totals.opened += this.reveal(chosen, safety);
290 this.checkWin();
291 }
292 return totals;
293 }
294
295 replay(count, safety) {
296 const board = this.copy();
297 board.restart();
298 for (const move of this.moves.slice(0, count)) {
299 switch (move.kind) {
300 case 'reveal': case 'auto-reveal': case 'guess': board.reveal(move.cell, safety); break;
301 case 'chord': board.chord(move.cell, safety); break;
302 case 'flag': case 'auto-flag': board.marks[move.cell] = 'flag'; break;
303 case 'unflag': board.marks[move.cell] = 'none'; break;
304 case 'question': board.marks[move.cell] = board.marks[move.cell] === 'question' ? 'none' : 'question'; break;
305 }
306 board.checkWin();
307 }
308 return board.display();
309 }

311 boardInfo() {
312 const visited = Array(this.size).fill(false);
313 let openings = 0;
314 let largest = 0;
315 let isolatedNumbers = 0;
316 for (let cell = 0; cell < this.size; cell++) {
317 if (this.mines[cell]) continue;
318 if (this.numbers[cell] !== 0) {
319 if (!this.neighbors[cell].some(next => !this.mines[next] && this.numbers[next] === 0)) isolatedNumbers++;
320 continue;
321 }
322 if (visited[cell]) continue;
323 openings++;
324 const queue = [cell];
325 const opening = new Set([cell]);
326 visited[cell] = true;
327 for (let head = 0; head < queue.length; head++) {
328 for (const next of this.neighbors[queue[head]]) {
329 opening.add(next);
330 if (!this.mines[next] && this.numbers[next] === 0 && !visited[next]) {
331 visited[next] = true;
332 queue.push(next);
333 }
334 }
335 }
336 largest = Math.max(largest, opening.size);
337 }
338 return `3bv ${openings + isolatedNumbers} openings ${openings} largest ${largest} mines ${this.mineCount} safe ${this.safeCount}`;
339 }

342function countConfigurations(component) {
343 const positions = new Map(component.cells.map((cell, index) => [cell, index]));
344 const constraints = component.constraints.map(c => ({
345 cells: c.cells.map(cell => positions.get(cell)), remaining: c.remaining,
346 }));
347 const incidence = Array.from({ length: component.cells.length }, () => []);
348 constraints.forEach((c, index) => {
349 for (const cell of c.cells) incidence[cell].push(index);
350 });
351 const assigned = Array(constraints.length).fill(0);
352 const unassigned = constraints.map(c => c.cells.length);
353 const values = Array(component.cells.length).fill(0);
354 const result = { configs: 0, minimum: component.cells.length, maximum: 0, frequencies: values.slice() };
355 if (constraints.some(c => c.remaining < 0 || c.remaining > c.cells.length)) return result;
356 const order = component.cells.map((_, index) => index).sort((a, b) => incidence[b].length - incidence[a].length || a - b);
357 const search = (depth, mineCount) => {
358 if (depth === order.length) {
359 result.configs++;
360 result.minimum = Math.min(result.minimum, mineCount);
361 result.maximum = Math.max(result.maximum, mineCount);
362 values.forEach((value, index) => { result.frequencies[index] += value; });
363 return;
364 }
365 const cell = order[depth];
366 for (let value = 0; value <= 1; value++) {
367 values[cell] = value;
368 let possible = true;
369 for (const index of incidence[cell]) {
370 assigned[index] += value;
371 unassigned[index]--;
372 const target = constraints[index].remaining;
373 if (assigned[index] > target || assigned[index] + unassigned[index] < target) possible = false;
374 }
375 if (possible) search(depth + 1, mineCount + value);
376 for (const index of incidence[cell]) {
377 assigned[index] -= value;
378 unassigned[index]++;
379 }
380 }
381 values[cell] = 0;
382 };
383 search(0, 0);
384 return result;
385}

387const formats = new Map([
388 ['NEW', 'nnnn'], ['PRESET', 'pn'], ['SAFETY', 's'], ['RESTART', ''],
389 ['REVEAL', 'nn'], ['FLAG', 'nn'], ['UNFLAG', 'nn'], ['QUESTION', 'nn'], ['CHORD', 'nn'],
390 ['UNDO', ''], ['SHOW', ''], ['STATUS', ''], ['CELL', 'nn'], ['HISTORY', 'n'], ['REPLAY', 'n'],
391 ['HINT', ''], ['AUTO', ''], ['ANALYZE', ''], ['PROB', 'nn'], ['FRONTIER', ''],
392 ['SOLVE', ''], ['BOARDINFO', ''], ['STATS', ''],
393]);
394const presets = new Map([['beginner', [9, 9, 10]], ['intermediate', [16, 16, 40]], ['expert', [30, 16, 99]]]);
395const cellCommands = new Set(['REVEAL', 'FLAG', 'UNFLAG', 'QUESTION', 'CHORD', 'CELL', 'PROB']);
396const liveCommands = new Set(['REVEAL', 'FLAG', 'UNFLAG', 'QUESTION', 'CHORD', 'UNDO', 'HINT', 'AUTO', 'ANALYZE', 'PROB', 'FRONTIER', 'SOLVE']);
397const undoCommands = new Set(['REVEAL', 'FLAG', 'UNFLAG', 'QUESTION', 'CHORD', 'AUTO', 'SOLVE']);

399class Program {
400 constructor() {
401 this.board = null;
402 this.safety = 'block';
403 this.stats = { played: 0, won: 0, lost: 0, streak: 0, best: 0, fastest: null };
404 }
405
406 execute(words, rows) {
407 const [command, ...args] = words;
408 if (command === 'LAYOUT') {
409 const [width, height] = args.map(Number);
410 const badRow = rows.findIndex(row => row.length !== width || /[^*.]/.test(row));
411 if (badRow !== -1) fail(`bad row ${badRow + 1}`);
412 const mines = rows.join('').split('').map(cell => cell === '*');
413 const count = mines.filter(Boolean).length;
414 if (count === 0 || count === width * height) fail(`bad mine count ${count}`);
415 this.board = new Board(width, height, count, 1);
416 this.board.determine(mines);
417 return [`layout ${width}x${height} mines ${count}`];
418 }
419 const format = formats.get(command);
420 if (format === undefined) fail(`unknown command ${command}`);
421 if (args.length !== format.length) fail('bad arguments');
422 for (let i = 0; i < args.length; i++) {
423 const valid = format[i] === 'n' ? /^(?:0|[1-9][0-9]{0,9})$/.test(args[i]) :
424 format[i] === 'p' ? presets.has(args[i]) : ['block', 'cell', 'none'].includes(args[i]);
425 if (!valid) fail('bad arguments');
426 }
427 const numbers = args.map(Number);
428 if (command === 'NEW' || command === 'PRESET') {
429 const [width, height, count, seed] = command === 'NEW' ? numbers : [...presets.get(args[0]), numbers[1]];
430 if (width < 1 || width > 30 || height < 1 || height > 30 || count < 1 ||
431 count > width * height - 9 || seed < 1 || seed > 2147483646) fail('bad value');
432 this.board = new Board(width, height, count, seed);
433 return [`new ${width}x${height} mines ${count}`];
434 }
435 if (command === 'HISTORY' && numbers[0] === 0) fail('bad value');
436 if (command === 'SAFETY') {
437 this.safety = args[0];
438 return [`safety ${this.safety}`];
439 }
440 if (command === 'STATS') {
441 const s = this.stats;
442 const rate = s.played === 0 ? '-' : `${Math.floor(s.won * 100 / s.played)}%`;
443 return [`stats played ${s.played} won ${s.won} lost ${s.lost} rate ${rate} streak ${s.streak} best ${s.best} fastest ${s.fastest ?? '-'}`];
444 }
445 const board = this.board;
446 if (board === null) fail('no game');
447 if (cellCommands.has(command) && (numbers[0] >= board.width || numbers[1] >= board.height)) fail('out of range');
448 if (liveCommands.has(command) && board.finished) fail('game over');
449 if ((command === 'ANALYZE' || command === 'BOARDINFO') && board.state === 'ready') fail('not started');
450 const cell = numbers[1] * board.width + numbers[0];
451 const snapshot = undoCommands.has(command) ? {
452 command: command.toLowerCase(), visible: [...board.visible], marks: [...board.marks], moveCount: board.moves.length,
453 } : null;
454 const response = this.boardCommand(command, numbers, cell, board);
455 if (snapshot !== null && board.moves.length > snapshot.moveCount) board.undo.push(snapshot);
456 if (['REVEAL', 'CHORD', 'AUTO', 'SOLVE'].includes(command) && board.finished) {
457 const s = this.stats;
458 s.played++;
459 if (board.state === 'won') {
460 s.won++;
461 s.streak++;
462 s.best = Math.max(s.best, s.streak);
463 s.fastest = Math.min(s.fastest ?? board.moves.length, board.moves.length);
464 } else {
465 s.lost++;
466 s.streak = 0;
467 }
468 }
469 return response;
470 }

472 boardCommand(command, numbers, cell, board) {
473 const coordinate = () => board.coordinate(cell);
474 switch (command) {
475 case 'RESTART':
476 board.restart();
477 return [`restart ${board.width}x${board.height} mines ${board.mineCount}`];
478 case 'REVEAL': case 'FLAG': case 'UNFLAG': case 'QUESTION': {
479 if (board.visible[cell]) fail('already revealed');
480 const mark = board.marks[cell];
481 if (command === 'REVEAL' && mark === 'flag') fail('flagged');
482 if (command === 'FLAG' && mark === 'flag') fail('already flagged');
483 if (command === 'UNFLAG' && mark !== 'flag') fail('not flagged');
484 board.moves.push({ kind: command.toLowerCase(), cell });
485 if (command === 'REVEAL') {
486 const opened = board.reveal(cell, this.safety);
487 if (board.state === 'lost') return board.finishOutput();
488 board.checkWin();
489 return [`revealed ${opened}`, ...board.finishOutput()];
490 }
491 board.marks[cell] = command === 'FLAG' ? 'flag' : command === 'UNFLAG' || mark === 'question' ? 'none' : 'question';
492 const action = command === 'FLAG' ? 'flagged' : command === 'UNFLAG' ? 'unflagged' : mark === 'question' ? 'unmarked' : 'questioned';
493 return [`${action} ${coordinate()} left ${board.remaining}`];
494 }
495 case 'CHORD': {
496 if (!board.visible[cell]) fail('not revealed');
497 const number = board.numbers[cell];
498 if (number === 0) fail('no number');
499 const flags = board.neighbors[cell].filter(next => board.marks[next] === 'flag').length;
500 if (flags !== number) fail(`flags ${flags} of ${number}`);
501 if (!board.neighbors[cell].some(next => board.unknown(next))) fail('nothing to reveal');
502 board.moves.push({ kind: 'chord', cell });
503 const opened = board.chord(cell, this.safety);
504 if (board.state === 'lost') return board.finishOutput();
505 board.checkWin();
506 return [`chorded ${opened}`, ...board.finishOutput()];
507 }
508 case 'UNDO': {
509 if (board.undo.length === 0) fail('nothing to undo');
510 const previous = board.undo.pop();
511 board.visible = previous.visible;
512 board.marks = previous.marks;
513 board.moves.length = previous.moveCount;
514 return [`undo ${previous.command} moves ${board.moves.length}`];
515 }
516 case 'SHOW': return board.display();
517 case 'STATUS': return [`status ${board.state} left ${board.remaining} moves ${board.moves.length} revealed ${board.revealedCount}/${board.safeCount}`];
518 case 'CELL': {
519 const neighbors = board.neighbors[cell];
520 if (!board.visible[cell]) return [`cell ${coordinate()} hidden mark ${board.marks[cell]} open ${neighbors.filter(next => board.visible[next]).length}`];
521 const flags = neighbors.filter(next => board.marks[next] === 'flag').length;
522 return [`cell ${coordinate()} number ${board.numbers[cell]} flags ${flags} unknown ${neighbors.filter(next => board.unknown(next)).length}`];
523 }
524 case 'HISTORY': {
525 const lines = [`history ${board.moves.length}`];
526 for (let i = Math.max(0, board.moves.length - numbers[0]); i < board.moves.length; i++) {
527 lines.push(`${i + 1} ${board.moves[i].kind} ${board.coordinate(board.moves[i].cell)}`);
528 }
529 return lines;
530 }
531 case 'REPLAY':
532 if (numbers[0] > board.moves.length) fail(`no move ${numbers[0]}`);
533 return [`replay ${numbers[0]}/${board.moves.length}`, ...board.replay(numbers[0], this.safety)];
534 case 'HINT': {
535 const { safe, mines } = board.deduce();
536 const choices = safe.size !== 0 ? safe : mines;
537 if (choices.size === 0) return ['hint none'];
538 const chosen = Math.min(...choices.keys());
539 return [`hint ${safe.size !== 0 ? 'safe' : 'mine'} ${board.coordinate(chosen)} by ${choices.get(chosen)}`];
540 }
541 case 'AUTO': {
542 const result = board.auto(this.safety);
543 return [`auto rounds ${result.rounds} flagged ${result.flagged} revealed ${result.opened}`, ...board.finishOutput()];
544 }
545 case 'ANALYZE': {
546 const copy = board.copy();
547 const result = copy.auto(this.safety);
548 if (copy.state === 'lost') return [`analyze boom ${copy.coordinate(copy.exploded)} rounds ${result.rounds}`];
549 if (copy.state === 'won') return [`analyze clear rounds ${result.rounds}`];
550 return [`analyze stuck rounds ${result.rounds} revealed ${copy.revealedCount}/${copy.safeCount}`];
551 }
552 case 'PROB': {
553 if (board.visible[cell]) fail('already revealed');
554 if (board.marks[cell] === 'flag') fail('flagged');
555 const component = board.frontier().find(group => group.cells.includes(cell));
556 if (component === undefined) return [`prob ${coordinate()} unconstrained`];
557 const result = countConfigurations(component);
558 if (result.configs === 0) return [`prob ${coordinate()} inconsistent`];
559 return [`prob ${coordinate()} ${fraction(result.frequencies[component.cells.indexOf(cell)], result.configs)} size ${component.cells.length} configs ${result.configs}`];
560 }
561 case 'FRONTIER': {
562 const components = board.frontier();
563 const lines = [`frontier ${components.reduce((sum, c) => sum + c.cells.length, 0)} components ${components.length}`];
564 for (const component of components) {
565 const result = countConfigurations(component);
566 const prefix = `component ${board.coordinate(component.cells[0])} size ${component.cells.length}`;
567 lines.push(result.configs === 0 ? `${prefix} inconsistent` : `${prefix} configs ${result.configs} min ${result.minimum} max ${result.maximum}`);
568 }
569 return lines;
570 }
571 case 'SOLVE': {
572 const result = board.solve(this.safety);
573 return [`solve guesses ${result.guesses} rounds ${result.rounds} flagged ${result.flagged} revealed ${result.opened}`, ...board.finishOutput()];
574 }
575 case 'BOARDINFO': return [board.boardInfo()];
576 }
577 throw new Error('Unreachable command');
578 }
579}
