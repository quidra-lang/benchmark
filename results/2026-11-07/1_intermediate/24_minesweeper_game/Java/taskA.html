57 if (word.length() > 10 || word.length() > 1 && word.charAt(0) == '0') throw error("bad arguments");
58 for (int i = 0; i < word.length(); i++) if (word.charAt(i) < '0' || word.charAt(i) > '9') throw error("bad arguments");

62 static String fraction(long numerator, long denominator) {
63 long a = numerator, b = denominator;
64 while (b != 0) { long rest = a % b; a = b; b = rest; }
65 return numerator / a + "/" + denominator / a;
66 }

115 Game(int width, int height, int mines, long seed) {
116 this.width = width; this.height = height; this.mineCount = mines; this.seed = seed;
117 cells = width * height;
118 mine = new boolean[cells]; numbers = new int[cells]; neighbors = new int[cells][];
119 revealed = new boolean[cells]; marks = new Mark[cells]; Arrays.fill(marks, Mark.NONE);
120 moves = new ArrayList<>();
121 for (int cell = 0; cell < cells; cell++) {
122 int x = cell % width, y = cell / width, count = 0;
123 int[] list = new int[8];
124 for (int dy = -1; dy <= 1; dy++) for (int dx = -1; dx <= 1; dx++) {
125 if ((dx != 0 || dy != 0) && x + dx >= 0 && x + dx < width && y + dy >= 0 && y + dy < height) {
126 list[count++] = (y + dy) * width + x + dx;
127 }
128 }
129 neighbors[cell] = Arrays.copyOf(list, count);
130 }
131 }

138 moves = keepMoves ? new ArrayList<>(source.moves) : new ArrayList<>();
139 }

157 void numberBoard() {
158 Arrays.fill(numbers, 0);
159 for (int cell = 0; cell < cells; cell++) for (int next : neighbors[cell]) if (mine[next]) numbers[cell]++;

162 void placeMines(int first, Safety safety) {
163 int sx = first % width, sy = first / width;
164 List<Integer> candidates = new ArrayList<>();
165 for (int cell = 0; cell < cells; cell++) {
166 int x = cell % width, y = cell / width;
167 if (safety == Safety.NONE || safety == Safety.CELL && cell != first
168 || safety == Safety.BLOCK && (Math.abs(x - sx) >= 2 || Math.abs(y - sy) >= 2)) candidates.add(cell);
169 }
170 long random = seed;
171 for (int i = 0; i < mineCount; i++) {
172 random = 48271 * random % 2147483647;
173 mine[candidates.remove((int) (random % candidates.size()))] = true;
174 }
175 numberBoard(); determined = true; state = State.PLAYING;
176 }

178 int reveal(int cell, Safety safety) {
179 if (!determined) placeMines(cell, safety);
180 if (mine[cell]) { state = State.LOST; exploded = cell; return 0; }
181 int[] queue = new int[cells];
182 int head = 0, tail = 0;
183 revealed[cell] = true; marks[cell] = Mark.NONE; queue[tail++] = cell;
184 while (head < tail) {
185 int current = queue[head++];
186 if (numbers[current] != 0) continue;
187 for (int next : neighbors[current]) if (unknown(next)) {
188 revealed[next] = true; marks[next] = Mark.NONE; queue[tail++] = next;
189 }
190 }
191 return tail;
192 }

194 int chord(int cell, Safety safety) {
195 int opened = 0;
196 for (int next : neighbors[cell]) {
197 if (unknown(next)) opened += reveal(next, safety);
198 if (state == State.LOST) break;
199 }
200 return opened;
201 }

203 void checkWin() {
204 if (state != State.PLAYING || opened() != safeCells()) return;
205 state = State.WON;
206 for (int cell = 0; cell < cells; cell++) if (mine[cell]) marks[cell] = Mark.FLAG;
207 }

209 void show() {
210 for (int y = 0; y < height; y++) {
211 StringBuilder row = new StringBuilder(width);
212 for (int x = 0; x < width; x++) {
213 int cell = y * width + x;
214 char symbol;
215 if (state == State.LOST && mine[cell]) symbol = cell == exploded ? '!' : marks[cell] == Mark.FLAG ? 'F' : '*';
216 else if (state == State.LOST && marks[cell] == Mark.FLAG) symbol = 'X';
217 else if (revealed[cell]) symbol = numbers[cell] == 0 ? '.' : (char) ('0' + numbers[cell]);
218 else symbol = switch (marks[cell]) { case NONE -> '#'; case FLAG -> 'F'; case QUESTION -> '?'; };
219 row.append(symbol);
220 }
221 line("%s", row);
222 }

226 if (state == State.LOST) { line("boom %s", coord(exploded)); show(); }
227 else if (state == State.WON) line("win moves %d", moves.size());

230 List<Constraint> constraints() {
231 List<Constraint> result = new ArrayList<>();
232 for (int cell = 0; cell < cells; cell++) {
233 if (!revealed[cell] || numbers[cell] == 0) continue;
234 int remaining = numbers[cell], count = 0;
235 int[] unknown = new int[8];
236 for (int next : neighbors[cell]) {
237 if (marks[next] == Mark.FLAG) remaining--;
238 if (unknown(next)) unknown[count++] = next;
239 }
240 if (count > 0) result.add(new Constraint(cell, remaining, Arrays.copyOf(unknown, count)));
241 }
242 return result;
243 }
244
245 Deduction deduce() {
246 List<Constraint> constraints = constraints();
247 Deduction result = new Deduction();
248 for (int rule = 1; rule <= 2; rule++) {
249 for (Constraint c : constraints) {
250 if (c.remaining() != (rule == 1 ? 0 : c.unknown().length)) continue;
251 for (int cell : c.unknown()) result.add(rule == 1, cell, new Reason(ReasonKind.SINGLE, c.cell(), 0));
252 }
253 if (!result.empty()) return result;
254 }
255 Constraint[] at = new Constraint[cells];
256 for (Constraint c : constraints) at[c.cell()] = c;
257 for (Constraint a : constraints) {
258 int ax = a.cell() % width, ay = a.cell() / width;
259 for (int y = Math.max(0, ay - 2); y <= Math.min(height - 1, ay + 2); y++) {
260 for (int x = Math.max(0, ax - 2); x <= Math.min(width - 1, ax + 2); x++) {
261 Constraint b = at[y * width + x];
262 if (b == null || a.unknown().length >= b.unknown().length) continue;
263 boolean subset = true;
264 for (int cell : a.unknown()) if (!contains(b.unknown(), cell)) { subset = false; break; }
265 if (!subset) continue;
266 int difference = b.remaining() - a.remaining();
267 if (difference != 0 && difference != b.unknown().length - a.unknown().length) continue;
268 for (int cell : b.unknown()) if (!contains(a.unknown(), cell)) {
269 result.add(difference == 0, cell, new Reason(ReasonKind.SUBSET, a.cell(), b.cell()));
270 }
271 }
272 }
273 }
274 if (!result.empty()) return result;
275 if (state == State.PLAYING && remaining() == 0) {
276 for (int cell = 0; cell < cells; cell++) if (unknown(cell)) result.add(true, cell, new Reason(ReasonKind.COUNT, 0, 0));
277 }
278 return result;
279 }

292 AutoResult autoPlay(Safety safety) {
293 AutoResult total = new AutoResult();
294 while (state == State.PLAYING) {
295 Deduction result = deduce();
296 if (result.empty()) break;
297 total.rounds++;
298 for (int cell : result.safe.keySet()) {
299 if (revealed[cell]) continue;
300 moves.add(new Move(MoveKind.AUTO_REVEAL, cell));
301 total.opened += reveal(cell, safety);
302 checkWin();
303 if (state.finished()) return total;
304 }
305 for (int cell : result.mines.keySet()) if (unknown(cell)) {
306 marks[cell] = Mark.FLAG; total.flagged++;
307 moves.add(new Move(MoveKind.AUTO_FLAG, cell));
308 }
309 }
310 return total;
311 }
312
313 AutoResult solve(Safety safety) {
314 AutoResult total = new AutoResult();
315 while (!state.finished()) {
316 AutoResult step = autoPlay(safety);
317 total.rounds += step.rounds; total.flagged += step.flagged; total.opened += step.opened;
318 if (state.finished()) break;
319 Frontier frontier = new Frontier(this);
320 for (int id = 0; id < frontier.components.size(); id++) frontier.count(id);
321 long unknown = 0, remaining = Math.max(remaining(), 0);
322 for (int cell = 0; cell < cells; cell++) if (unknown(cell)) unknown++;
323 int choice = -1;
324 long bestNumerator = 0, bestDenominator = 1;
325 for (int cell = 0; cell < cells; cell++) {
326 if (!unknown(cell)) continue;
327 int id = frontier.componentOf[cell];
328 long numerator = id < 0 ? remaining : frontier.ways[cell];
329 long denominator = id < 0 ? unknown : frontier.components.get(id).configs;
330 if (denominator == 0) continue;
331 if (choice < 0 || numerator * bestDenominator < bestNumerator * denominator) {
332 choice = cell; bestNumerator = numerator; bestDenominator = denominator;
333 }
334 }
335 if (choice < 0) break;
336 total.guesses++;
337 moves.add(new Move(MoveKind.GUESS, choice));
338 total.opened += reveal(choice, safety);
339 checkWin();
340 }
341 return total;
342 }

344 void probability(int cell) {
345 Frontier frontier = new Frontier(this);
346 int id = frontier.componentOf[cell];
347 if (id < 0) { line("prob %s unconstrained", coord(cell)); return; }
348 frontier.count(id);
349 Component component = frontier.components.get(id);
350 if (component.configs == 0) line("prob %s inconsistent", coord(cell));
351 else line("prob %s %s size %d configs %d", coord(cell), fraction(frontier.ways[cell], component.configs), component.cells.size(), component.configs);
352 }

354 void frontierReport() {
355 Frontier frontier = new Frontier(this);
356 int total = frontier.components.stream().mapToInt(c -> c.cells.size()).sum();
357 line("frontier %d components %d", total, frontier.components.size());
358 for (int id = 0; id < frontier.components.size(); id++) {
359 frontier.count(id);
360 Component c = frontier.components.get(id);
361 if (c.configs == 0) line("component %s size %d inconsistent", coord(c.cells.getFirst()), c.cells.size());
362 else line("component %s size %d configs %d min %d max %d", coord(c.cells.getFirst()), c.cells.size(), c.configs, c.minimum, c.maximum);
363 }

366 void replay(int count, Safety safety) {
367 Game copy = new Game(this, false);
368 copy.restart();
369 for (Move move : moves.subList(0, count)) {
370 int cell = move.cell();
371 switch (move.kind()) {
372 case REVEAL, AUTO_REVEAL, GUESS -> copy.reveal(cell, safety);
373 case CHORD -> copy.chord(cell, safety);
374 case FLAG, AUTO_FLAG -> copy.marks[cell] = Mark.FLAG;
375 case UNFLAG -> copy.marks[cell] = Mark.NONE;
376 case QUESTION -> copy.marks[cell] = copy.marks[cell] == Mark.QUESTION ? Mark.NONE : Mark.QUESTION;
377 }
378 copy.checkWin();
379 }
380 line("replay %d/%d", count, moves.size());
381 copy.show();
382 }

384 void boardInfo() {
385 boolean[] visited = new boolean[cells];
386 int openings = 0, largest = 0, isolated = 0;
387 for (int cell = 0; cell < cells; cell++) {
388 if (mine[cell]) continue;
389 if (numbers[cell] != 0) {
390 boolean adjacentBlank = false;
391 for (int next : neighbors[cell]) adjacentBlank |= !mine[next] && numbers[next] == 0;
392 if (!adjacentBlank) isolated++;
393 } else if (!visited[cell]) {
394 openings++;
395 boolean[] covered = new boolean[cells];
396 int[] queue = new int[cells];
397 int head = 0, tail = 0;
398 queue[tail++] = cell; visited[cell] = true; covered[cell] = true;
399 while (head < tail) {
400 for (int next : neighbors[queue[head++]]) {
401 covered[next] = true;
402 if (numbers[next] == 0 && !visited[next]) { visited[next] = true; queue[tail++] = next; }
403 }
404 }
405 largest = Math.max(largest, countTrue(covered));
406 }
407 }
408 line("3bv %d openings %d largest %d mines %d safe %d", openings + isolated, openings, largest, mineCount, safeCells());
409 }

411 void process(Command command, int cell, long value, Safety safety) {
412 switch (command) {
413 case REVEAL, FLAG, UNFLAG, QUESTION, PROB -> {
414 if (revealed[cell]) throw error("already revealed");
415 if ((command == Command.REVEAL || command == Command.PROB) && marks[cell] == Mark.FLAG) throw error("flagged");
416 if (command == Command.FLAG && marks[cell] == Mark.FLAG) throw error("already flagged");
417 if (command == Command.UNFLAG && marks[cell] != Mark.FLAG) throw error("not flagged");
418 }
419 default -> {}
420 }
421 switch (command) {
422 case RESTART -> { restart(); line("restart %dx%d mines %d", width, height, mineCount); }
423 case REVEAL -> {
424 moves.add(new Move(MoveKind.REVEAL, cell));
425 int opened = reveal(cell, safety);
426 if (state != State.LOST) { line("revealed %d", opened); checkWin(); }
427 endingOutput();
428 }
429 case FLAG -> {
430 marks[cell] = Mark.FLAG; moves.add(new Move(MoveKind.FLAG, cell));
431 line("flagged %s left %d", coord(cell), remaining());
432 }
433 case UNFLAG -> {
434 marks[cell] = Mark.NONE; moves.add(new Move(MoveKind.UNFLAG, cell));
435 line("unflagged %s left %d", coord(cell), remaining());
436 }
437 case QUESTION -> {
438 moves.add(new Move(MoveKind.QUESTION, cell));
439 marks[cell] = marks[cell] == Mark.QUESTION ? Mark.NONE : Mark.QUESTION;
440 line("%s %s left %d", marks[cell] == Mark.NONE ? "unmarked" : "questioned", coord(cell), remaining());
441 }
442 case CHORD -> {
443 if (!revealed[cell]) throw error("not revealed");
444 if (numbers[cell] == 0) throw error("no number");
445 int flags = 0, unknown = 0;
446 for (int next : neighbors[cell]) { if (marks[next] == Mark.FLAG) flags++; if (unknown(next)) unknown++; }
447 if (flags != numbers[cell]) throw error("flags " + flags + " of " + numbers[cell]);
448 if (unknown == 0) throw error("nothing to reveal");
449 moves.add(new Move(MoveKind.CHORD, cell));
450 int opened = chord(cell, safety);
451 if (state != State.LOST) { line("chorded %d", opened); checkWin(); }
452 endingOutput();
453 }
454 case UNDO -> {
455 if (undo.isEmpty()) throw error("nothing to undo");
456 Snapshot snapshot = undo.removeLast();
457 revealed = snapshot.revealed(); marks = snapshot.marks();
458 moves.subList(snapshot.moves(), moves.size()).clear();
459 line("undo %s moves %d", snapshot.command().name().toLowerCase(Locale.ROOT), moves.size());
460 }
461 case SHOW -> show();
462 case STATUS -> line("status %s left %d moves %d revealed %d/%d", state.name().toLowerCase(Locale.ROOT), remaining(), moves.size(), opened(), safeCells());
463 case CELL -> {
464 int flags = 0, unknown = 0, opened = 0;
465 for (int next : neighbors[cell]) {
466 if (marks[next] == Mark.FLAG) flags++; if (unknown(next)) unknown++; if (revealed[next]) opened++;
467 }
468 if (revealed[cell]) line("cell %s number %d flags %d unknown %d", coord(cell), numbers[cell], flags, unknown);
469 else line("cell %s hidden mark %s open %d", coord(cell), marks[cell].name().toLowerCase(Locale.ROOT), opened);
470 }
471 case HISTORY -> {
472 line("history %d", moves.size());
473 int count = (int) Math.min(value, moves.size());
474 for (int i = moves.size() - count; i < moves.size(); i++) {
475 Move move = moves.get(i); line("%d %s %s", i + 1, move.kind().text, coord(move.cell()));
476 }
477 }
478 case REPLAY -> {
479 if (value > moves.size()) throw error("no move " + value);
480 replay((int) value, safety);
481 }
482 case HINT -> hint();
483 case AUTO, SOLVE -> {
484 AutoResult result = command == Command.AUTO ? autoPlay(safety) : solve(safety);
485 if (command == Command.SOLVE) line("solve guesses %d rounds %d flagged %d revealed %d", result.guesses, result.rounds, result.flagged, result.opened);
486 else line("auto rounds %d flagged %d revealed %d", result.rounds, result.flagged, result.opened);
487 endingOutput();
488 }
489 case ANALYZE -> {
490 Game copy = new Game(this, true);
491 AutoResult result = copy.autoPlay(safety);
492 if (copy.state == State.LOST) line("analyze boom %s rounds %d", copy.coord(copy.exploded), result.rounds);
493 else if (copy.state == State.WON) line("analyze clear rounds %d", result.rounds);
494 else line("analyze stuck rounds %d revealed %d/%d", result.rounds, copy.opened(), copy.safeCells());
495 }
496 case PROB -> probability(cell);
497 case FRONTIER -> frontierReport();
498 case BOARDINFO -> boardInfo();
499 default -> throw new IllegalStateException(command.name());
500 }
501 }
502 }

504 static final class Component {
505 final List<Integer> cells = new ArrayList<>();
506 long configs;
507 int minimum, maximum;
508 }
509 record Restriction(int mask, int target) {}

511 static int root(int[] parent, int cell) {
512 while (parent[cell] != cell) { parent[cell] = parent[parent[cell]]; cell = parent[cell]; }
513 return cell;
514 }

516 static final class Frontier {
517 final List<Constraint> constraints;
518 final int[] componentOf;
519 final List<Component> components = new ArrayList<>();
520 final long[] ways;
521
522 Frontier(Game game) {
523 constraints = game.constraints(); componentOf = new int[game.cells]; ways = new long[game.cells];
524 Arrays.fill(componentOf, -1);
525 int[] parent = new int[game.cells], ids = new int[game.cells];
526 boolean[] present = new boolean[game.cells];
527 Arrays.fill(ids, -1);
528 for (int i = 0; i < game.cells; i++) parent[i] = i;
529 for (Constraint c : constraints) for (int cell : c.unknown()) {
530 present[cell] = true;
531 int left = root(parent, cell), right = root(parent, c.unknown()[0]);
532 parent[left] = right;
533 }
534 for (int cell = 0; cell < game.cells; cell++) if (present[cell]) {
535 int representative = root(parent, cell);
536 if (ids[representative] < 0) { ids[representative] = components.size(); components.add(new Component()); }
537 int id = ids[representative];
538 componentOf[cell] = id;
539 components.get(id).cells.add(cell);
540 }
541 }
542
543 void count(int id) {
544 Component component = components.get(id);
545 int[] local = new int[componentOf.length];
546 for (int i = 0; i < component.cells.size(); i++) local[component.cells.get(i)] = i;
547 List<Restriction> restrictions = new ArrayList<>();
548 for (Constraint c : constraints) if (componentOf[c.unknown()[0]] == id) {
549 int mask = 0;
550 for (int cell : c.unknown()) mask |= 1 << local[cell];
551 restrictions.add(new Restriction(mask, c.remaining()));
552 }
553 Search search = new Search(component.cells.size(), restrictions);
554 search.visit(0, 0);
555 component.configs = search.configs; component.minimum = search.minimum; component.maximum = search.maximum;
556 for (int i = 0; i < component.cells.size(); i++) ways[component.cells.get(i)] = search.ways[i];
557 }
558 }

560 static final class Search {
561 final int size;
562 final List<Restriction> restrictions;
563 final long[] ways;
564 long configs;
565 int minimum, maximum;
566
567 Search(int size, List<Restriction> restrictions) {
568 this.size = size; this.restrictions = restrictions; ways = new long[size]; minimum = size + 1;
569 }
570
571 void visit(int depth, int mines) {
572 int assigned = (1 << depth) - 1;
573 for (Restriction c : restrictions) {
574 int used = Integer.bitCount(c.mask() & mines), left = Integer.bitCount(c.mask() & ~assigned);
575 if (used > c.target() || used + left < c.target()) return;
576 }
577 if (depth == size) {
578 int count = Integer.bitCount(mines);
579 configs++; minimum = Math.min(minimum, count); maximum = Math.max(maximum, count);
580 for (int i = 0; i < size; i++) if ((mines & (1 << i)) != 0) ways[i]++;
581 } else {
582 visit(depth + 1, mines);
583 visit(depth + 1, mines | (1 << depth));
584 }
585 }
586 }

588 static final class Statistics {
589 int played, won, lost, streak, best;
590 Integer fastest;
591 void finish(Game game) {
592 played++;
593 if (game.state == State.WON) {
594 won++; streak++; best = Math.max(best, streak);
595 fastest = fastest == null ? game.moves.size() : Math.min(fastest, game.moves.size());
596 } else { lost++; streak = 0; }
597 }
598 void show() {
599 String rate = played == 0 ? "-" : won * 100 / played + "%";
600 line("stats played %d won %d lost %d rate %s streak %d best %d fastest %s", played, won, lost, rate, streak, best, fastest == null ? "-" : fastest.toString());
601 }
602 }

604 static final class Table {
605 Game game;
606 Safety safety = Safety.BLOCK;
607 final Statistics stats = new Statistics();
608
609 void layout(int width, int height, String[] rows) {
610 for (int i = 0; i < height; i++) {
611 if (rows[i].length() != width) throw error("bad row " + (i + 1));
612 for (int x = 0; x < width; x++) if (rows[i].charAt(x) != '*' && rows[i].charAt(x) != '.') throw error("bad row " + (i + 1));
613 }
614 int count = 0;
615 for (String row : rows) for (int x = 0; x < width; x++) if (row.charAt(x) == '*') count++;
616 if (count == 0 || count == width * height) throw error("bad mine count " + count);
617 Game next = new Game(width, height, count, 0);
618 for (int y = 0; y < height; y++) for (int x = 0; x < width; x++) next.mine[y * width + x] = rows[y].charAt(x) == '*';
619 next.determined = true; next.state = State.PLAYING; next.numberBoard();
620 game = next;
621 line("layout %dx%d mines %d", width, height, count);
622 }
623
624 void execute(String[] words) {
625 Command command = Command.parse(words[0]);
626 if (words.length != command.arguments + 1) throw error("bad arguments");
627 long[] values = new long[4];
628 Safety nextSafety = safety;
629 if (command == Command.PRESET) {
630 int[] preset = switch (words[1]) {
631 case "beginner" -> new int[]{9, 9, 10}; case "intermediate" -> new int[]{16, 16, 40}; case "expert" -> new int[]{30, 16, 99};
632 default -> throw error("bad arguments");
633 };
634 for (int i = 0; i < 3; i++) values[i] = preset[i];
635 values[3] = numberWord(words[2]);
636 } else if (command == Command.SAFETY) {
637 nextSafety = switch (words[1]) {
638 case "block" -> Safety.BLOCK; case "cell" -> Safety.CELL; case "none" -> Safety.NONE;
639 default -> throw error("bad arguments");
640 };
641 } else for (int i = 1; i < words.length; i++) values[i - 1] = numberWord(words[i]);
642 if (command == Command.NEW || command == Command.PRESET) {
643 long width = values[0], height = values[1], mines = values[2], seed = values[3];
644 if (width < 1 || width > 30 || height < 1 || height > 30 || mines < 1 || mines > width * height - 9 || seed < 1 || seed > 2147483646) throw error("bad value");
645 game = new Game((int) width, (int) height, (int) mines, seed);
646 line("new %dx%d mines %d", width, height, mines);
647 return;
648 }
649 if (command == Command.HISTORY && values[0] == 0) throw error("bad value");
650 if (command == Command.SAFETY) { safety = nextSafety; line("safety %s", words[1]); return; }
651 if (command == Command.STATS) { stats.show(); return; }
652 if (game == null) throw error("no game");
653 if (command.hasCell() && (values[0] >= game.width || values[1] >= game.height)) throw error("out of range");
654 if (command.needsUnfinished() && game.state.finished()) throw error("game over");
655 if ((command == Command.ANALYZE || command == Command.BOARDINFO) && !game.determined) throw error("not started");
656 Snapshot snapshot = command.recordsUndo() ? new Snapshot(command, game.revealed.clone(), game.marks.clone(), game.moves.size()) : null;
657 boolean wasFinished = game.state.finished();
658 int cell = command.hasCell() ? (int) values[1] * game.width + (int) values[0] : 0;
659 game.process(command, cell, values[0], safety);
660 if (snapshot != null && game.moves.size() > snapshot.moves()) game.undo.addLast(snapshot);
661 if (!wasFinished && game.state.finished()) stats.finish(game);
662 }
663 }

665 public static void main(String[] args) throws Exception {
666 BufferedReader input = new BufferedReader(new InputStreamReader(System.in));
667 int count = Integer.parseInt(input.readLine());
668 Table table = new Table();
669 for (int i = 0; i < count; i++) {
670 String[] words = input.readLine().split(" ");
671 try {
672 if (words[0].equals("LAYOUT")) {
673 int width = Integer.parseInt(words[1]), height = Integer.parseInt(words[2]);
674 String[] rows = new String[height];
675 for (int y = 0; y < height; y++) rows[y] = input.readLine();
676 table.layout(width, height, rows);
677 } else table.execute(words);
678 } catch (CommandError error) { line("error: %s", error.getMessage()); }
679 }
680 OUT.flush();
681 }
