44static void *grow(void *memory, size_t *capacity, size_t item_size) {
45 *capacity = *capacity ? *capacity * 2 : 16;
46 void *next = realloc(memory, *capacity * item_size);
47 if (!next) exit(EXIT_FAILURE);
48 return next;
49}

51static char *read_line(void) {
52 size_t length = 0, capacity = 128;
53 char *line = allocate(capacity);
54 int c;
55 while ((c = getchar()) != EOF && c != '\n') {
56 if (length + 1 == capacity) line = grow(line, &capacity, 1);
57 line[length++] = (char)c;
58 }
59 if (c == EOF && !length) { free(line); return NULL; }
60 if (length && line[length - 1] == '\r') length--;
61 line[length] = '\0';
62 return line;
63}

107static Game *new_game(int width, int height, int mines, int64_t seed) {
108 Game *game = calloc(1, sizeof *game);
109 if (!game) exit(EXIT_FAILURE);
110 game->width = width;
111 game->height = height;
112 game->cells = width * height;
113 game->mines = mines;
114 game->seed = seed;
115 game->exploded = -1;
116 for (int cell = 0; cell < game->cells; cell++) {
117 int x = cell % width, y = cell / width;
118 for (int dy = -1; dy <= 1; dy++) for (int dx = -1; dx <= 1; dx++) {
119 if ((!dx && !dy) || x + dx < 0 || x + dx >= width || y + dy < 0 || y + dy >= height) continue;
120 game->neighbors[cell][game->degree[cell]++] = (y + dy) * width + x + dx;
121 }
122 }
123 return game;
124}

168static void place_mines(Game *game, int first, Safety safety) {
169 int candidates[MAX_CELLS], count = 0;
170 int sx = first % game->width, sy = first / game->width;
171 for (int cell = 0; cell < game->cells; cell++) {
172 int x = cell % game->width, y = cell / game->width;
173 if (safety == NONE || (safety == CELL && cell != first) || (safety == BLOCK && (abs(x - sx) >= 2 || abs(y - sy) >= 2))) {
174 candidates[count++] = cell;
175 }
176 }
177 int64_t state = game->seed;
178 for (int i = 0; i < game->mines; i++) {
179 state = 48271 * state % 2147483647;
180 int chosen = (int)(state % count);
181 game->mine[candidates[chosen]] = true;
182 memmove(candidates + chosen, candidates + chosen + 1, (size_t)(count - chosen - 1) * sizeof *candidates);
183 count--;
184 }
185 number_board(game);
186 game->determined = true;
187 game->state = PLAYING;
188}

190static int reveal(Game *game, int cell, Safety safety) {
191 if (!game->determined) place_mines(game, cell, safety);
192 if (game->mine[cell]) { game->state = LOST; game->exploded = cell; return 0; }
193 int queue[MAX_CELLS], head = 0, tail = 0;
194 game->revealed[cell] = true;
195 game->marks[cell] = UNMARKED;
196 queue[tail++] = cell;
197 while (head < tail) {
198 int current = queue[head++];
199 if (game->number[current]) continue;
200 for (int i = 0; i < game->degree[current]; i++) {
201 int next = game->neighbors[current][i];
202 if (!unknown(game, next)) continue;
203 game->revealed[next] = true;
204 game->marks[next] = UNMARKED;
205 queue[tail++] = next;
206 }
207 }
208 return tail;
209}

214 int next = game->neighbors[cell][i];

244typedef struct { int cell, rem, count, unknown[8]; } Constraint;

246static int constraints(const Game *game, Constraint result[MAX_CELLS]) {
247 int count = 0;
248 for (int cell = 0; cell < game->cells; cell++) {
249 if (!game->revealed[cell] || !game->number[cell]) continue;
250 Constraint constraint = {.cell = cell, .rem = game->number[cell]};
251 for (int i = 0; i < game->degree[cell]; i++) {
252 int next = game->neighbors[cell][i];
253 constraint.rem -= game->marks[next] == FLAGGED;
254 if (unknown(game, next)) constraint.unknown[constraint.count++] = next;
255 }
256 if (constraint.count) result[count++] = constraint;
257 }
258 return count;
259}

265static void add_reason(Deduction *result, bool safe, int cell, Reason reason) {

276static void deduce(const Game *game, Deduction *result) {
277 memset(result, 0, sizeof *result);
278 Constraint list[MAX_CELLS];
279 int count = constraints(game, list);
280 for (int rule = 1; rule <= 2; rule++) {
281 for (int i = 0; i < count; i++) {
282 const Constraint *c = &list[i];
283 if (c->rem != (rule == 1 ? 0 : c->count)) continue;
284 for (int j = 0; j < c->count; j++) add_reason(result, rule == 1, c->unknown[j], (Reason){SINGLE, c->cell, 0});
285 }
286 if (result->any) return;
287 }
288 int at[MAX_CELLS];
289 for (int cell = 0; cell < game->cells; cell++) at[cell] = -1;
290 for (int i = 0; i < count; i++) at[list[i].cell] = i;
291 for (int i = 0; i < count; i++) {
292 const Constraint *a = &list[i];
293 int ax = a->cell % game->width, ay = a->cell / game->width;
294 for (int y = ay - 2; y <= ay + 2; y++) for (int x = ax - 2; x <= ax + 2; x++) {
295 if (x < 0 || x >= game->width || y < 0 || y >= game->height) continue;
296 int index = at[y * game->width + x];
297 if (index < 0 || index == i) continue;
298 const Constraint *b = &list[index];
299 if (a->count >= b->count) continue;
300 bool subset = true;
301 for (int j = 0; j < a->count; j++) if (!contains(b, a->unknown[j])) { subset = false; break; }
302 if (!subset) continue;
303 int difference = b->rem - a->rem;
304 if (difference != 0 && difference != b->count - a->count) continue;
305 for (int j = 0; j < b->count; j++) if (!contains(a, b->unknown[j])) {
306 add_reason(result, difference == 0, b->unknown[j], (Reason){SUBSET, a->cell, b->cell});
307 }
308 }
309 }
310 if (result->any) return;
311 if (game->state == PLAYING && remaining(game) == 0) {
312 for (int cell = 0; cell < game->cells; cell++) if (unknown(game, cell)) add_reason(result, true, cell, (Reason){COUNT, 0, 0});
313 }
314}

316static void hint(const Game *game) {
317 Deduction result;
318 deduce(game, &result);
319 for (int side = 0; side < 2; side++) for (int cell = 0; cell < game->cells; cell++) {
320 Reason reason = side == 0 ? result.safe[cell] : result.mines[cell];
321 if (reason.kind == NO_REASON) continue;
322 printf("hint %s %d,%d by ", side == 0 ? "safe" : "mine", cell % game->width, cell / game->width);
323 if (reason.kind == COUNT) puts("count");
324 else {
325 printf("%s %d,%d", reason.kind == SINGLE ? "single" : "subset", reason.a % game->width, reason.a / game->width);
326 if (reason.kind == SUBSET) printf(" %d,%d", reason.b % game->width, reason.b / game->width);
327 putchar('\n');
328 }
329 return;
330 }
331 puts("hint none");
332}

343static int root(int parent[MAX_CELLS], int cell) {
344 while (parent[cell] != cell) { parent[cell] = parent[parent[cell]]; cell = parent[cell]; }
345 return cell;
346}

348static Frontier *make_frontier(const Game *game) {
349 Frontier *frontier = calloc(1, sizeof *frontier);
350 if (!frontier) exit(EXIT_FAILURE);
351 frontier->constraint_count = constraints(game, frontier->constraints);
352 int parent[MAX_CELLS], ids[MAX_CELLS];
353 bool present[MAX_CELLS] = {false};
354 for (int i = 0; i < game->cells; i++) { parent[i] = i; ids[i] = -1; frontier->component_of[i] = -1; }
355 for (int i = 0; i < frontier->constraint_count; i++) {
356 Constraint *c = &frontier->constraints[i];
357 for (int j = 0; j < c->count; j++) {
358 present[c->unknown[j]] = true;
359 int left = root(parent, c->unknown[j]);
360 int right = root(parent, c->unknown[0]);
361 parent[left] = right;
362 }
363 }
364 for (int cell = 0; cell < game->cells; cell++) if (present[cell]) {
365 int representative = root(parent, cell);
366 if (ids[representative] < 0) ids[representative] = frontier->component_count++;
367 int id = ids[representative];
368 frontier->component_of[cell] = id;
369 frontier->components[id].size++;
370 frontier->total++;
371 }
372 int position[MAX_CELLS] = {0}, offset = 0;
373 for (int i = 0; i < frontier->component_count; i++) {
374 frontier->components[i].start = offset;
375 offset += frontier->components[i].size;
376 }
377 for (int cell = 0; cell < game->cells; cell++) {
378 int id = frontier->component_of[cell];
379 if (id >= 0) frontier->cells[frontier->components[id].start + position[id]++] = cell;
380 }
381 return frontier;
382}

384static int bit_count(uint32_t bits) {
385 int count = 0;
386 while (bits) { bits &= bits - 1; count++; }
387 return count;
388}

390typedef struct {
391 uint32_t masks[MAX_CELLS];
392 int targets[MAX_CELLS], constraint_count, size;
393 int configs, minimum, maximum, ways[MAX_COMPONENT];
394} Search;
395
396static void search_configurations(Search *search, int depth, uint32_t mines) {
397 uint32_t assigned = (UINT32_C(1) << depth) - 1;
398 for (int i = 0; i < search->constraint_count; i++) {
399 int used = bit_count(search->masks[i] & mines);
400 int left = bit_count(search->masks[i] & ~assigned);
401 if (used > search->targets[i] || used + left < search->targets[i]) return;
402 }
403 if (depth == search->size) {
404 int total = bit_count(mines);
405 search->configs++;
406 if (total < search->minimum) search->minimum = total;
407 if (total > search->maximum) search->maximum = total;
408 for (int i = 0; i < search->size; i++) search->ways[i] += (mines & (UINT32_C(1) << i)) != 0;
409 return;
410 }
411 search_configurations(search, depth + 1, mines);
412 search_configurations(search, depth + 1, mines | (UINT32_C(1) << depth));
413}

415static void count_component(Frontier *frontier, int id) {
416 Component *component = &frontier->components[id];
417 Search search = {.size = component->size, .minimum = component->size + 1, .maximum = -1};
418 int local[MAX_CELLS];
419 for (int i = 0; i < component->size; i++) local[frontier->cells[component->start + i]] = i;
420 for (int i = 0; i < frontier->constraint_count; i++) {
421 const Constraint *c = &frontier->constraints[i];
422 if (frontier->component_of[c->unknown[0]] != id) continue;
423 uint32_t mask = 0;
424 for (int j = 0; j < c->count; j++) mask |= UINT32_C(1) << local[c->unknown[j]];
425 search.masks[search.constraint_count] = mask;
426 search.targets[search.constraint_count++] = c->rem;
427 }
428 search_configurations(&search, 0, 0);
429 component->configs = search.configs;
430 component->minimum = search.minimum;
431 component->maximum = search.maximum;
432 for (int i = 0; i < component->size; i++) frontier->ways[frontier->cells[component->start + i]] = search.ways[i];
433}

435static int gcd(int a, int b) {
436 while (b) { int rest = a % b; a = b; b = rest; }
437 return a;
438}

471typedef struct { int rounds, flagged, opened, guesses; } AutoResult;
472
473static AutoResult auto_play(Game *game, Safety safety) {
474 AutoResult total = {0};
475 while (game->state == PLAYING) {
476 Deduction result;
477 deduce(game, &result);
478 if (!result.any) break;
479 total.rounds++;
480 for (int cell = 0; cell < game->cells; cell++) if (result.safe[cell].kind != NO_REASON && !game->revealed[cell]) {
481 add_move(game, AUTO_REVEAL, cell);
482 total.opened += reveal(game, cell, safety);
483 check_win(game);
484 if (finished(game)) return total;
485 }
486 for (int cell = 0; cell < game->cells; cell++) if (result.mines[cell].kind != NO_REASON && unknown(game, cell)) {
487 game->marks[cell] = FLAGGED;
488 total.flagged++;
489 add_move(game, AUTO_FLAG, cell);
490 }
491 }
492 return total;
493}

495static AutoResult solve(Game *game, Safety safety) {
496 AutoResult total = {0};
497 while (!finished(game)) {
498 AutoResult step = auto_play(game, safety);
499 total.rounds += step.rounds;
500 total.flagged += step.flagged;
501 total.opened += step.opened;
502 if (finished(game)) break;
503 Frontier *frontier = make_frontier(game);
504 for (int i = 0; i < frontier->component_count; i++) count_component(frontier, i);
505 int unknown_count = 0;
506 for (int cell = 0; cell < game->cells; cell++) unknown_count += unknown(game, cell);
507 int outside_mines = remaining(game);
508 if (outside_mines < 0) outside_mines = 0;
509 int guess = -1, best_numerator = 0, best_denominator = 1;
510 for (int cell = 0; cell < game->cells; cell++) if (unknown(game, cell)) {
511 int id = frontier->component_of[cell], numerator, denominator;
512 if (id < 0) { numerator = outside_mines; denominator = unknown_count; }
513 else {
514 numerator = frontier->ways[cell];
515 denominator = frontier->components[id].configs;
516 if (!denominator) continue;
517 }
518 if (guess < 0 || (int64_t)numerator * best_denominator < (int64_t)best_numerator * denominator) {
519 guess = cell;
520 best_numerator = numerator;
521 best_denominator = denominator;
522 }
523 }
524 free(frontier);
525 if (guess < 0) break;
526 total.guesses++;
527 add_move(game, GUESS, guess);
528 total.opened += reveal(game, guess, safety);
529 check_win(game);
530 }
531 return total;
532}

534static void replay(const Game *game, size_t count, Safety safety) {
535 Game *copy = copy_game(game, false);
536 restart(copy);
537 for (size_t i = 0; i < count; i++) {
538 Move move = game->moves[i];
539 switch (move.kind) {
540 case REVEAL: case AUTO_REVEAL: case GUESS: reveal(copy, move.cell, safety); break;
541 case CHORD: chord(copy, move.cell, safety); break;
542 case FLAG: case AUTO_FLAG: copy->marks[move.cell] = FLAGGED; break;
543 case UNFLAG: copy->marks[move.cell] = UNMARKED; break;
544 case QUESTION: copy->marks[move.cell] = copy->marks[move.cell] == QUESTIONED ? UNMARKED : QUESTIONED; break;
545 }
546 check_win(copy);
547 }
548 printf("replay %zu/%zu\n", count, game->move_count);
549 show(copy);
550 free_game(copy);
551}

553static void board_info(const Game *game) {
554 bool visited[MAX_CELLS] = {false};
555 int openings = 0, largest = 0, isolated = 0;
556 for (int cell = 0; cell < game->cells; cell++) {
557 if (game->mine[cell]) continue;
558 if (game->number[cell]) {
559 bool adjacent_blank = false;
560 for (int i = 0; i < game->degree[cell]; i++) {
561 int next = game->neighbors[cell][i];
562 adjacent_blank |= !game->mine[next] && game->number[next] == 0;
563 }
564 isolated += !adjacent_blank;
565 } else if (!visited[cell]) {
566 openings++;
567 bool covered[MAX_CELLS] = {false};
568 int queue[MAX_CELLS], head = 0, tail = 0;
569 queue[tail++] = cell;
570 visited[cell] = true;
571 covered[cell] = true;
572 while (head < tail) {
573 int current = queue[head++];
574 for (int i = 0; i < game->degree[current]; i++) {
575 int next = game->neighbors[current][i];
576 covered[next] = true;
577 if (!game->number[next] && !visited[next]) { visited[next] = true; queue[tail++] = next; }
578 }
579 }
580 int size = 0;
581 for (int i = 0; i < game->cells; i++) size += covered[i];
582 if (size > largest) largest = size;
583 }
584 }
585 printf("3bv %d openings %d largest %d mines %d safe %d\n", openings + isolated, openings, largest, game->mines, game->cells - game->mines);
586}

588static void neighbor_counts(const Game *game, int cell, int *flags, int *unknowns, int *opened) {
589 *flags = *unknowns = *opened = 0;
590 for (int i = 0; i < game->degree[cell]; i++) {
591 int next = game->neighbors[cell][i];
592 *flags += game->marks[next] == FLAGGED;
593 *unknowns += unknown(game, next);
594 *opened += game->revealed[next];
595 }
596}

598static bool process(Game *game, Command command, int cell, int64_t value, Safety safety) {
599 int x = cell % game->width, y = cell / game->width;
600 if (command == CMD_REVEAL || command == CMD_FLAG || command == CMD_UNFLAG || command == CMD_QUESTION || command == PROB) {
601 if (game->revealed[cell]) return fail("already revealed");
602 if ((command == CMD_REVEAL || command == PROB) && game->marks[cell] == FLAGGED) return fail("flagged");
603 if (command == CMD_FLAG && game->marks[cell] == FLAGGED) return fail("already flagged");
604 if (command == CMD_UNFLAG && game->marks[cell] != FLAGGED) return fail("not flagged");
605 }
606 switch (command) {
607 case RESTART:
608 restart(game);
609 printf("restart %dx%d mines %d\n", game->width, game->height, game->mines);
610 break;
611 case CMD_REVEAL: {
612 add_move(game, REVEAL, cell);
613 int opened = reveal(game, cell, safety);
614 if (game->state != LOST) { printf("revealed %d\n", opened); check_win(game); }
615 ending_output(game);
616 break;
617 }
618 case CMD_FLAG:
619 game->marks[cell] = FLAGGED;
620 add_move(game, FLAG, cell);
621 printf("flagged %d,%d left %d\n", x, y, remaining(game));
622 break;
623 case CMD_UNFLAG:
624 game->marks[cell] = UNMARKED;
625 add_move(game, UNFLAG, cell);
626 printf("unflagged %d,%d left %d\n", x, y, remaining(game));
627 break;
628 case CMD_QUESTION:
629 add_move(game, QUESTION, cell);
630 game->marks[cell] = game->marks[cell] == QUESTIONED ? UNMARKED : QUESTIONED;
631 printf("%s %d,%d left %d\n", game->marks[cell] == UNMARKED ? "unmarked" : "questioned", x, y, remaining(game));
632 break;
633 case CMD_CHORD: {
634 if (!game->revealed[cell]) return fail("not revealed");
635 if (!game->number[cell]) return fail("no number");
636 int flags, unknowns, opened;
637 neighbor_counts(game, cell, &flags, &unknowns, &opened);
638 if (flags != game->number[cell]) return fail("flags %d of %d", flags, game->number[cell]);
639 if (!unknowns) return fail("nothing to reveal");
640 add_move(game, CHORD, cell);
641 opened = chord(game, cell, safety);
642 if (game->state != LOST) { printf("chorded %d\n", opened); check_win(game); }
643 ending_output(game);
644 break;
645 }
646 case UNDO: {
647 if (!game->undo_count) return fail("nothing to undo");
648 Snapshot *snapshot = &game->undo[--game->undo_count];
649 memcpy(game->revealed, snapshot->revealed, sizeof game->revealed);
650 memcpy(game->marks, snapshot->marks, sizeof game->marks);
651 game->move_count = snapshot->move_count;
652 const char *name = snapshot->command == AUTO ? "auto" : snapshot->command == SOLVE ? "solve" : move_names[snapshot->command - CMD_REVEAL];
653 printf("undo %s moves %zu\n", name, game->move_count);
654 break;
655 }
656 case SHOW: show(game); break;
657 case STATUS:
658 printf("status %s left %d moves %zu revealed %d/%d\n", state_names[game->state], remaining(game), game->move_count, opened_count(game), game->cells - game->mines);
659 break;
660 case CMD_CELL: {
661 int flags, unknowns, opened;
662 neighbor_counts(game, cell, &flags, &unknowns, &opened);
663 if (game->revealed[cell]) printf("cell %d,%d number %d flags %d unknown %d\n", x, y, game->number[cell], flags, unknowns);
664 else printf("cell %d,%d hidden mark %s open %d\n", x, y, game->marks[cell] == FLAGGED ? "flag" : game->marks[cell] == QUESTIONED ? "question" : "none", opened);
665 break;
666 }
667 case HISTORY: {
668 printf("history %zu\n", game->move_count);
669 size_t count = (uint64_t)value < game->move_count ? (size_t)value : game->move_count;
670 for (size_t i = game->move_count - count; i < game->move_count; i++) {
671 Move move = game->moves[i];
672 printf("%zu %s %d,%d\n", i + 1, move_names[move.kind], move.cell % game->width, move.cell / game->width);
673 }
674 break;
675 }
676 case REPLAY:
677 if ((uint64_t)value > game->move_count) return fail("no move %" PRId64, value);
678 replay(game, (size_t)value, safety);
679 break;
680 case HINT: hint(game); break;
681 case AUTO: case SOLVE: {
682 AutoResult result = command == AUTO ? auto_play(game, safety) : solve(game, safety);
683 if (command == SOLVE) printf("solve guesses %d rounds %d flagged %d revealed %d\n", result.guesses, result.rounds, result.flagged, result.opened);
684 else printf("auto rounds %d flagged %d revealed %d\n", result.rounds, result.flagged, result.opened);
685 ending_output(game);
686 break;
687 }
688 case ANALYZE: {
689 Game *copy = copy_game(game, true);
690 AutoResult result = auto_play(copy, safety);
691 if (copy->state == LOST) printf("analyze boom %d,%d rounds %d\n", copy->exploded % copy->width, copy->exploded / copy->width, result.rounds);
692 else if (copy->state == WON) printf("analyze clear rounds %d\n", result.rounds);
693 else printf("analyze stuck rounds %d revealed %d/%d\n", result.rounds, opened_count(copy), copy->cells - copy->mines);
694 free_game(copy);
695 break;
696 }
697 case PROB: probability(game, cell); break;
698 case FRONTIER: frontier_report(game); break;
699 case BOARDINFO: board_info(game); break;
700 default: break;
701 }
702 return true;
703}

705static bool number_word(const char *word, int64_t *value) {
706 size_t length = strlen(word);
707 if (length > 10 || (length > 1 && word[0] == '0')) return false;
708 for (size_t i = 0; i < length; i++) if (word[i] < '0' || word[i] > '9') return false;
709 *value = strtoll(word, NULL, 10);
710 return true;
711}

726static bool layout(Table *table, int width, int height) {
727 bool mines[MAX_CELLS] = {false};
728 int count = 0, bad_row = 0;
729 for (int y = 0; y < height; y++) {
730 char *row = read_line();
731 bool good = strlen(row) == (size_t)width;
732 for (size_t x = 0; row[x]; x++) if (row[x] != '*' && row[x] != '.') good = false;
733 if (!good) { if (!bad_row) bad_row = y + 1; }
734 else for (int x = 0; x < width; x++) { mines[y * width + x] = row[x] == '*'; count += row[x] == '*'; }
735 free(row);
736 }
737 if (bad_row) return fail("bad row %d", bad_row);
738 if (count == 0 || count == width * height) return fail("bad mine count %d", count);
739 Game *game = new_game(width, height, count, 0);
740 memcpy(game->mine, mines, sizeof mines);
741 game->determined = true;
742 game->state = PLAYING;
743 number_board(game);
744 free_game(table->game);
745 table->game = game;
746 printf("layout %dx%d mines %d\n", width, height, count);
747 return true;
748}

750static bool execute(Table *table, char **words, int count) {
751 if (strcmp(words[0], "LAYOUT") == 0) return layout(table, atoi(words[1]), atoi(words[2]));
752 int index = -1;
753 for (int i = 0; i < 24; i++) if (strcmp(words[0], command_names[i]) == 0) { index = i; break; }
754 if (index < 0) return fail("unknown command %s", words[0]);
755 Command command = (Command)index;
756 if (count != arguments[command] + 1) return fail("bad arguments");
757 int64_t values[4] = {0};
758 int preset = -1;
759 Safety safety = table->safety;
760 if (command == PRESET) {
761 const char *presets[] = {"beginner", "intermediate", "expert"};
762 for (int i = 0; i < 3; i++) if (strcmp(words[1], presets[i]) == 0) preset = i;
763 if (preset < 0 || !number_word(words[2], &values[0])) return fail("bad arguments");
764 } else if (command == SAFETY) {
765 if (strcmp(words[1], "block") == 0) safety = BLOCK;
766 else if (strcmp(words[1], "cell") == 0) safety = CELL;
767 else if (strcmp(words[1], "none") == 0) safety = NONE;
768 else return fail("bad arguments");
769 } else for (int i = 1; i < count; i++) if (!number_word(words[i], &values[i - 1])) return fail("bad arguments");
770 if (command == NEW || command == PRESET) {
771 if (command == PRESET) {
772 const int widths[] = {9, 16, 30}, heights[] = {9, 16, 16}, mines[] = {10, 40, 99};
773 values[3] = values[0]; values[0] = widths[preset]; values[1] = heights[preset]; values[2] = mines[preset];
774 }
775 if (values[0] < 1 || values[0] > 30 || values[1] < 1 || values[1] > 30 || values[2] < 1
776 || values[2] > values[0] * values[1] - 9 || values[3] < 1 || values[3] > 2147483646) return fail("bad value");
777 Game *game = new_game((int)values[0], (int)values[1], (int)values[2], values[3]);
778 free_game(table->game);
779 table->game = game;
780 printf("new %dx%d mines %d\n", game->width, game->height, game->mines);
781 return true;
782 }
783 if (command == HISTORY && values[0] == 0) return fail("bad value");
784 if (command == SAFETY) { table->safety = safety; printf("safety %s\n", words[1]); return true; }
785 if (command == STATS) {
786 Statistics *stats = &table->stats;
787 printf("stats played %d won %d lost %d rate ", stats->played, stats->won, stats->lost);
788 if (stats->played) printf("%d%%", 100 * stats->won / stats->played); else putchar('-');
789 printf(" streak %d best %d fastest ", stats->streak, stats->best);
790 if (stats->has_fastest) printf("%zu\n", stats->fastest); else puts("-");
791 return true;
792 }
793 Game *game = table->game;
794 if (!game) return fail("no game");
795 if (has_cell(command) && (values[0] >= game->width || values[1] >= game->height)) return fail("out of range");
796 if (needs_unfinished(command) && finished(game)) return fail("game over");
797 if ((command == ANALYZE || command == BOARDINFO) && !game->determined) return fail("not started");
798 Snapshot snapshot = {.command = command, .move_count = game->move_count};
799 if (records_undo(command)) {
800 memcpy(snapshot.revealed, game->revealed, sizeof snapshot.revealed);
801 memcpy(snapshot.marks, game->marks, sizeof snapshot.marks);
802 }
803 State before = game->state;
804 int cell = has_cell(command) ? (int)values[1] * game->width + (int)values[0] : 0;
805 if (!process(game, command, cell, values[0], safety)) return false;
806 if (records_undo(command) && game->move_count > snapshot.move_count) {
807 if (game->undo_count == game->undo_capacity) game->undo = grow(game->undo, &game->undo_capacity, sizeof *game->undo);
808 game->undo[game->undo_count++] = snapshot;
809 }
810 if (before != WON && before != LOST && finished(game)) {
811 Statistics *stats = &table->stats;
812 stats->played++;
813 if (game->state == WON) {
814 stats->won++; stats->streak++;
815 if (stats->streak > stats->best) stats->best = stats->streak;
816 if (!stats->has_fastest || game->move_count < stats->fastest) { stats->fastest = game->move_count; stats->has_fastest = true; }
817 } else { stats->lost++; stats->streak = 0; }
818 }
819 return true;
820}

824 if (!line) return 0;

835 }
836 free_game(table.game);
837 return 0;
