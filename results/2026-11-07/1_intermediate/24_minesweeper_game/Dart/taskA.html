51class Component {
52 final List<int> cells;
53 final List<Constraint> constraints;
54 final int boardSize;
55 final List<int> mineCounts;
56 int configurations = 0, maximum = 0, minimum;
57 bool _counted = false;
58 Component(this.cells, this.constraints, this.boardSize)
59 : minimum = cells.length, mineCounts = List.filled(cells.length, 0);
60
61 void count() {
62 if (_counted) return;
63 _counted = true;
64 if (constraints.any((c) => c.remaining < 0 || c.remaining > c.unknown.length)) return;
65 final index = List.filled(boardSize, 0);
66 for (var i = 0; i < cells.length; i++) index[cells[i]] = i;
67 final links = List.generate(cells.length, (_) => <int>[]);
68 for (var c = 0; c < constraints.length; c++) {
69 for (final cell in constraints[c].unknown) links[index[cell]].add(c);
70 }
71 final order = List.generate(cells.length, (i) => i);
72 order.sort((a, b) {
73 final degree = links[b].length.compareTo(links[a].length);
74 return degree != 0 ? degree : a.compareTo(b);
75 });
76 final remaining = [for (final c in constraints) c.unknown.length];
77 final sums = List.filled(constraints.length, 0);
78 final assignment = List.filled(cells.length, 0);
79 void search(int depth, int mines) {
80 if (depth == cells.length) {
81 configurations++;
82 minimum = min(minimum, mines);
83 maximum = max(maximum, mines);
84 for (var i = 0; i < cells.length; i++) mineCounts[i] += assignment[i];
85 return;
86 }
87 final variable = order[depth];
88 for (var value = 0; value <= 1; value++) {
89 assignment[variable] = value;
90 var possible = true;
91 for (final c in links[variable]) {
92 remaining[c]--;
93 sums[c] += value;
94 final required = constraints[c].remaining;
95 if (sums[c] > required || sums[c] + remaining[c] < required) possible = false;
96 }
97 if (possible) search(depth + 1, mines + value);
98 for (final c in links[variable]) {
99 sums[c] -= value;
100 remaining[c]++;
101 }
102 }
103 }
104 search(0, 0);
105 }
106}

112 for (var dx = -1; dx <= 1; dx++)
113 if ((dx != 0 || dy != 0) && x + dx >= 0 && x + dx < width && y + dy >= 0 && y + dy < height)

137 Game.copy(Game source, {bool reset = false})

139 neighbors = source.neighbors, mines = source.mines, numbers = source.numbers,

141 marks = reset ? List.filled(source.size, Mark.none) : List.of(source.marks),

145 int get size => width * height;
146 bool get finished => state == GameState.won || state == GameState.lost;
147 int get remaining => mineCount - marks.where((mark) => mark == Mark.flag).length;
148 int get openCount => revealed.where((value) => value).length;

150 bool unknown(int cell) => !revealed[cell] && marks[cell] != Mark.flag;

152 void _setPlacement(List<bool> placement) {
153 mines = List.unmodifiable(placement);
154 numbers = List.unmodifiable(neighbors.map((nearby) => nearby.where((cell) => placement[cell]).length));
155 state = GameState.playing;

158 void _place(int first, String safety) {
159 final sx = first % width, sy = first ~/ width;
160 final candidates = <int>[];
161 for (var cell = 0; cell < size; cell++) {
162 if (safety == 'none' || safety == 'cell' && cell != first ||
163 safety == 'block' && ((cell % width - sx).abs() >= 2 || (cell ~/ width - sy).abs() >= 2)) {
164 candidates.add(cell);
165 }
166 }
167 final placement = List.filled(size, false);
168 var random = seed;
169 for (var i = 0; i < mineCount; i++) {
170 random = 48271 * random % 2147483647;
171 placement[candidates.removeAt(random % candidates.length)] = true;
172 }
173 _setPlacement(placement);
174 }

178 marks = List.filled(size, Mark.none);

192 int reveal(int first, String safety) {
193 if (mines == null) _place(first, safety);
194 if (mines![first]) {
195 state = GameState.lost;
196 exploded = first;
197 return 0;
198 }
199 final queue = ListQueue<int>()..add(first);
200 revealed[first] = true;
201 marks[first] = Mark.none;
202 var opened = 1;
203 while (queue.isNotEmpty) {
204 final cell = queue.removeFirst();
205 if (numbers[cell] == 0) {
206 for (final neighbor in neighbors[cell]) {
207 if (unknown(neighbor)) {
208 revealed[neighbor] = true;
209 marks[neighbor] = Mark.none;
210 queue.add(neighbor);
211 opened++;
212 }
213 }
214 }
215 }
216 return opened;
217 }

219 int chord(int cell, String safety) {
220 var opened = 0;
221 for (final neighbor in neighbors[cell]) {
222 if (unknown(neighbor)) {
223 opened += reveal(neighbor, safety);
224 if (state == GameState.lost) break;
225 }
226 }
227 return opened;

230 bool checkWin() {
231 if (state != GameState.playing || openCount != size - mineCount) return false;
232 state = GameState.won;
233 for (var cell = 0; cell < size; cell++) {
234 if (mines![cell]) marks[cell] = Mark.flag;
235 }
236 return true;
237 }

254 List<Constraint> _constraints() {
255 final result = <Constraint>[];
256 for (var cell = 0; cell < size; cell++) {
257 if (revealed[cell] && numbers[cell] > 0) {
258 final cells = neighbors[cell].where(unknown).toList();
259 if (cells.isNotEmpty) {

261 }
262 }
263 }

267 Deduction deduce() {
268 final constraints = _constraints();
269 final result = Deduction(size);
270 for (final c in constraints) {
271 if (c.remaining == 0) {
272 for (final cell in c.unknown) result.safe[cell] ??= 'single ${cellText(c.cell)}';
273 }
274 }
275 if (result.any) return result;
276 for (final c in constraints) {
277 if (c.remaining == c.unknown.length) {
278 for (final cell in c.unknown) result.mines[cell] ??= 'single ${cellText(c.cell)}';
279 }
280 }
281 if (result.any) return result;
282 for (final a in constraints) {
283 for (final b in constraints) {
284 if (a.unknown.length >= b.unknown.length || !a.unknown.every(b.unknown.contains)) continue;
285 final difference = b.unknown.length - a.unknown.length;
286 final remaining = b.remaining - a.remaining;
287 if (remaining != 0 && remaining != difference) continue;
288 final side = remaining == 0 ? result.safe : result.mines;
289 for (final cell in b.unknown) {
290 if (!a.unknown.contains(cell)) side[cell] ??= 'subset ${cellText(a.cell)} ${cellText(b.cell)}';
291 }
292 }
293 }
294 if (result.any) return result;
295 if (state == GameState.playing && remaining == 0) {
296 for (var cell = 0; cell < size; cell++) {
297 if (unknown(cell)) result.safe[cell] = 'count';
298 }
299 }
300 return result;
301 }

303 Frontier buildFrontier() {
304 final constraints = _constraints();
305 final parents = List.generate(size, (i) => i);
306 final frontier = List.filled(size, false);
307 int root(int cell) {
308 while (parents[cell] != cell) {
309 parents[cell] = parents[parents[cell]];
310 cell = parents[cell];
311 }
312 return cell;
313 }
314 for (final c in constraints) {
315 for (final cell in c.unknown) {
316 frontier[cell] = true;
317 parents[root(cell)] = root(c.unknown.first);
318 }
319 }
320 final groupForRoot = List.filled(size, -1);
321 final groups = <List<int>>[];
322 for (var cell = 0; cell < size; cell++) {
323 if (!frontier[cell]) continue;
324 final representative = root(cell);
325 if (groupForRoot[representative] == -1) {
326 groupForRoot[representative] = groups.length;
327 groups.add([]);
328 }
329 groups[groupForRoot[representative]].add(cell);
330 }
331 final byGroup = List.generate(groups.length, (_) => <Constraint>[]);
332 for (final c in constraints) byGroup[groupForRoot[root(c.unknown.first)]].add(c);
333 final components = <Component>[];
334 final at = List<Component?>.filled(size, null);
335 for (var i = 0; i < groups.length; i++) {
336 final component = Component(groups[i], byGroup[i], size);
337 components.add(component);
338 for (final cell in component.cells) at[cell] = component;
339 }
340 return (components: components, at: at);
341 }

343 Totals auto(String safety) {
344 final totals = Totals();
345 while (state == GameState.playing) {
346 final deduction = deduce();
347 if (!deduction.any) break;
348 totals.rounds++;
349 for (var cell = 0; cell < size; cell++) {
350 if (deduction.safe[cell] != null && !revealed[cell]) {
351 moves.add((kind: MoveKind.autoReveal, cell: cell));
352 totals.opened += reveal(cell, safety);
353 checkWin();
354 if (finished) return totals;
355 }
356 }
357 for (var cell = 0; cell < size; cell++) {
358 if (deduction.mines[cell] != null && unknown(cell)) {
359 marks[cell] = Mark.flag;
360 totals.flagged++;
361 moves.add((kind: MoveKind.autoFlag, cell: cell));
362 }
363 }
364 }
365 return totals;
366 }

368 Totals solve(String safety) {
369 final totals = Totals();
370 while (!finished) {
371 totals.add(auto(safety));
372 if (finished) break;
373 final frontier = buildFrontier();
374 final numerators = List.filled(size, 0), denominators = List.filled(size, 0);
375 for (final component in frontier.components) {
376 component.count();
377 for (var i = 0; i < component.cells.length; i++) {
378 numerators[component.cells[i]] = component.mineCounts[i];
379 denominators[component.cells[i]] = component.configurations;
380 }
381 }
382 var unknownCount = 0;
383 for (var cell = 0; cell < size; cell++) {
384 if (unknown(cell)) unknownCount++;
385 }
386 final backgroundMines = max(remaining, 0);
387 var chosen = -1, bestNumerator = 0, bestDenominator = 1;
388 for (var cell = 0; cell < size; cell++) {
389 if (!unknown(cell)) continue;
390 final numerator = frontier.at[cell] == null ? backgroundMines : numerators[cell];
391 final denominator = frontier.at[cell] == null ? unknownCount : denominators[cell];
392 if (denominator != 0 && (chosen == -1 || numerator * bestDenominator < bestNumerator * denominator)) {
393 chosen = cell;
394 bestNumerator = numerator;
395 bestDenominator = denominator;
396 }
397 }
398 if (chosen == -1) break;
399 totals.guesses++;
400 moves.add((kind: MoveKind.guess, cell: chosen));
401 totals.opened += reveal(chosen, safety);
402 checkWin();
403 }
404 return totals;
405 }

407 Game replay(int count, String safety) {
408 final copy = Game.copy(this, reset: true);
409 for (final move in moves.take(count)) {
410 switch (move.kind) {
411 case MoveKind.reveal: case MoveKind.autoReveal: case MoveKind.guess:
412 copy.reveal(move.cell, safety);
413 break;
414 case MoveKind.chord: copy.chord(move.cell, safety); break;
415 case MoveKind.flag: case MoveKind.autoFlag: copy.marks[move.cell] = Mark.flag; break;
416 case MoveKind.unflag: copy.marks[move.cell] = Mark.none; break;
417 case MoveKind.question:
418 copy.marks[move.cell] = copy.marks[move.cell] == Mark.question ? Mark.none : Mark.question;
419 break;
420 }
421 copy.moves.add(move);
422 copy.checkWin();
423 }
424 return copy;
425 }

427 ({int bv, int openings, int largest}) boardInfo() {
428 final visited = List.filled(size, false);
429 var openings = 0, largest = 0, isolated = 0;
430 bool blank(int cell) => !mines![cell] && numbers[cell] == 0;
431 for (var first = 0; first < size; first++) {
432 if (blank(first) && !visited[first]) {
433 openings++;
434 final area = <int>{first};
435 final queue = ListQueue<int>()..add(first);
436 visited[first] = true;
437 while (queue.isNotEmpty) {
438 final cell = queue.removeFirst();
439 for (final neighbor in neighbors[cell]) {
440 area.add(neighbor);
441 if (blank(neighbor) && !visited[neighbor]) {
442 visited[neighbor] = true;
443 queue.add(neighbor);
444 }
445 }
446 }
447 largest = max(largest, area.length);
448 }
449 if (!mines![first] && numbers[first] > 0 && !neighbors[first].any(blank)) isolated++;
450 }
451 return (bv: openings + isolated, openings: openings, largest: largest);
452 }

484class Program {
485 Game? game;
486 String safety = 'block';
487 int played = 0, wins = 0, losses = 0, streak = 0, best = 0;
488 int? fastest;
489
490 void recordFinish(Game game) {
491 played++;
492 if (game.state == GameState.won) {
493 wins++;
494 streak++;
495 best = max(best, streak);
496 fastest = fastest == null ? game.moves.length : min(fastest!, game.moves.length);
497 } else {
498 losses++;
499 streak = 0;

503 void layout(List<String> words) {
504 final width = int.parse(words[1]), height = int.parse(words[2]);
505 final rows = [for (var y = 0; y < height; y++) stdin.readLineSync()!];
506 for (var y = 0; y < height; y++) {
507 if (rows[y].length != width || rows[y].codeUnits.any((c) => c != 42 && c != 46)) fail('bad row ${y + 1}');
508 }
509 final mines = [for (final row in rows) for (final c in row.codeUnits) c == 42];
510 final count = mines.where((value) => value).length;
511 if (count == 0 || count == width * height) fail('bad mine count $count');
512 game = Game(width, height, count, 0, placement: mines);
513 emit('layout ${width}x$height mines $count');
514 }
515
516 void execute(List<String> words) {
517 final command = words[0];
518 if (command == 'LAYOUT') { layout(words); return; }
519 final arity = arities[command];
520 if (arity == null) fail('unknown command $command');
521 if (words.length != arity + 1) fail('bad arguments');
522 final values = List.filled(arity, 0);
523 for (var i = 0; i < arity; i++) {
524 final word = words[i + 1];
525 if (command == 'PRESET' && i == 0) {
526 if (!const {'beginner', 'intermediate', 'expert'}.contains(word)) fail('bad arguments');
527 } else if (command == 'SAFETY') {
528 if (!const {'block', 'cell', 'none'}.contains(word)) fail('bad arguments');
529 } else {
530 values[i] = number(word);
531 }
532 }
533 if (command == 'NEW' || command == 'PRESET') {
534 int width, height, mines, seed;
535 if (command == 'NEW') {
536 width = values[0]; height = values[1]; mines = values[2]; seed = values[3];
537 } else {
538 (width, height, mines) = switch (words[1]) {
539 'beginner' => (9, 9, 10), 'intermediate' => (16, 16, 40), _ => (30, 16, 99),
540 };
541 seed = values[1];
542 }
543 if (width < 1 || width > 30 || height < 1 || height > 30 || mines < 1 || mines > width * height - 9 || seed < 1 || seed > 2147483646) fail('bad value');
544 game = Game(width, height, mines, seed);
545 emit('new ${width}x$height mines $mines');
546 return;
547 }
548 if (command == 'HISTORY' && values[0] == 0) fail('bad value');
549 if (command == 'SAFETY') { safety = words[1]; emit('safety $safety'); return; }
550 if (command == 'STATS') {
551 final rate = played == 0 ? '-' : '${wins * 100 ~/ played}%';
552 emit('stats played $played won $wins lost $losses rate $rate streak $streak best $best fastest ${fastest ?? '-'}');
553 return;
554 }
555 final board = game;
556 if (board == null) fail('no game');
557 var cell = 0;
558 if (cellCommands.contains(command)) {
559 if (values[0] >= board.width || values[1] >= board.height) fail('out of range');
560 cell = values[1] * board.width + values[0];
561 }
562 if (board.finished && activeCommands.contains(command)) fail('game over');
563 if (board.mines == null && (command == 'ANALYZE' || command == 'BOARDINFO')) fail('not started');
564 final wasFinished = board.finished;
565 final snapshot = undoCommands.contains(command) ? board.snapshot(command.toLowerCase()) : null;
566 switch (command) {
567 case 'RESTART':
568 board.restart();
569 emit('restart ${board.width}x${board.height} mines ${board.mineCount}');
570 break;
571 case 'REVEAL':
572 if (board.revealed[cell]) fail('already revealed');
573 if (board.marks[cell] == Mark.flag) fail('flagged');
574 board.moves.add((kind: MoveKind.reveal, cell: cell));
575 final opened = board.reveal(cell, safety);
576 if (board.state != GameState.lost) emit('revealed $opened');
577 board.checkWin();
578 finishOutput(board);
579 break;
580 case 'FLAG': case 'UNFLAG': case 'QUESTION':
581 if (board.revealed[cell]) fail('already revealed');
582 if (command == 'FLAG' && board.marks[cell] == Mark.flag) fail('already flagged');
583 if (command == 'UNFLAG' && board.marks[cell] != Mark.flag) fail('not flagged');
584 String action;
585 if (command == 'FLAG') { board.marks[cell] = Mark.flag; action = 'flagged'; }
586 else if (command == 'UNFLAG') { board.marks[cell] = Mark.none; action = 'unflagged'; }
587 else if (board.marks[cell] == Mark.question) { board.marks[cell] = Mark.none; action = 'unmarked'; }
588 else { board.marks[cell] = Mark.question; action = 'questioned'; }
589 board.moves.add((kind: command == 'FLAG' ? MoveKind.flag : command == 'UNFLAG' ? MoveKind.unflag : MoveKind.question, cell: cell));
590 emit('$action ${board.cellText(cell)} left ${board.remaining}');
591 break;
592 case 'CHORD':
593 if (!board.revealed[cell]) fail('not revealed');
594 if (board.numbers[cell] == 0) fail('no number');
595 final flags = board.neighbors[cell].where((n) => board.marks[n] == Mark.flag).length;
596 if (flags != board.numbers[cell]) fail('flags $flags of ${board.numbers[cell]}');
597 if (!board.neighbors[cell].any(board.unknown)) fail('nothing to reveal');
598 board.moves.add((kind: MoveKind.chord, cell: cell));
599 final chorded = board.chord(cell, safety);
600 if (board.state != GameState.lost) emit('chorded $chorded');
601 board.checkWin();
602 finishOutput(board);
603 break;
604 case 'UNDO':
605 if (board.undo.isEmpty) fail('nothing to undo');
606 final saved = board.undo.removeLast();
607 board.restore(saved);
608 emit('undo ${saved.name} moves ${board.moves.length}');
609 break;
610 case 'SHOW': show(board); break;
611 case 'STATUS':
612 emit('status ${board.state.name} left ${board.remaining} moves ${board.moves.length} revealed ${board.openCount}/${board.size - board.mineCount}');
613 break;
614 case 'CELL':
615 if (board.revealed[cell]) {
616 emit('cell ${board.cellText(cell)} number ${board.numbers[cell]} flags ${board.neighbors[cell].where((n) => board.marks[n] == Mark.flag).length} unknown ${board.neighbors[cell].where(board.unknown).length}');
617 } else {
618 emit('cell ${board.cellText(cell)} hidden mark ${board.marks[cell].name} open ${board.neighbors[cell].where((n) => board.revealed[n]).length}');
619 }
620 break;
621 case 'HISTORY':
622 emit('history ${board.moves.length}');
623 for (var i = board.moves.length - min(values[0], board.moves.length); i < board.moves.length; i++) {
624 emit('${i + 1} ${board.moves[i].kind.label} ${board.cellText(board.moves[i].cell)}');
625 }
626 break;
627 case 'REPLAY':
628 if (values[0] > board.moves.length) fail('no move ${values[0]}');
629 emit('replay ${values[0]}/${board.moves.length}');
630 show(board.replay(values[0], safety));
631 break;
632 case 'HINT':
633 final deduction = board.deduce();
634 var target = deduction.safe.indexWhere((reason) => reason != null);
635 if (target != -1) {
636 emit('hint safe ${board.cellText(target)} by ${deduction.safe[target]}');
637 } else {
638 target = deduction.mines.indexWhere((reason) => reason != null);
639 emit(target == -1 ? 'hint none' : 'hint mine ${board.cellText(target)} by ${deduction.mines[target]}');
640 }
641 break;
642 case 'AUTO': case 'SOLVE':
643 final totals = command == 'AUTO' ? board.auto(safety) : board.solve(safety);
644 emit(command == 'AUTO' ? 'auto rounds ${totals.rounds} flagged ${totals.flagged} revealed ${totals.opened}'
645 : 'solve guesses ${totals.guesses} rounds ${totals.rounds} flagged ${totals.flagged} revealed ${totals.opened}');
646 finishOutput(board);
647 break;
648 case 'ANALYZE':
649 final copy = Game.copy(board);
650 final analysis = copy.auto(safety);
651 emit(copy.state == GameState.lost ? 'analyze boom ${copy.cellText(copy.exploded)} rounds ${analysis.rounds}'
652 : copy.state == GameState.won ? 'analyze clear rounds ${analysis.rounds}'
653 : 'analyze stuck rounds ${analysis.rounds} revealed ${copy.openCount}/${copy.size - copy.mineCount}');
654 break;
655 case 'PROB':
656 if (board.revealed[cell]) fail('already revealed');
657 if (board.marks[cell] == Mark.flag) fail('flagged');
658 final component = board.buildFrontier().at[cell];
659 if (component == null) { emit('prob ${board.cellText(cell)} unconstrained'); }
660 else {
661 component.count();
662 if (component.configurations == 0) { emit('prob ${board.cellText(cell)} inconsistent'); }
663 else {
664 final numerator = component.mineCounts[component.cells.indexOf(cell)], denominator = component.configurations;
665 final divisor = numerator.gcd(denominator);
666 emit('prob ${board.cellText(cell)} ${numerator ~/ divisor}/${denominator ~/ divisor} size ${component.cells.length} configs ${component.configurations}');
667 }
668 }
669 break;
670 case 'FRONTIER':
671 final frontier = board.buildFrontier();
672 final count = frontier.components.fold(0, (sum, part) => sum + part.cells.length);
673 emit('frontier $count components ${frontier.components.length}');
674 for (final part in frontier.components) {
675 part.count();
676 final prefix = 'component ${board.cellText(part.cells.first)} size ${part.cells.length}';
677 emit(part.configurations == 0 ? '$prefix inconsistent' : '$prefix configs ${part.configurations} min ${part.minimum} max ${part.maximum}');
678 }
679 break;
680 case 'BOARDINFO':
681 final info = board.boardInfo();
682 emit('3bv ${info.bv} openings ${info.openings} largest ${info.largest} mines ${board.mineCount} safe ${board.size - board.mineCount}');
683 break;
684 }
685 if (snapshot != null && board.moves.length > snapshot.moveCount) board.undo.add(snapshot);
686 if (!wasFinished && board.finished) recordFinish(board);
687 }
688}

692 final program = Program();
693 for (var i = 0; i < count; i++) {
694 final words = stdin.readLineSync()!.split(' ');
695 try { program.execute(words); }
696 on CommandError catch (error) { emit('error: ${error.message}'); }
697 }
