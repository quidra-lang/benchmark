86func newGame(width, height, mines int, seed int64) *Game {
87 cells := width * height
88 game := &Game{
89 width: width, height: height, mineCount: mines, seed: seed,
90 mine: make([]bool, cells), revealed: make([]bool, cells), numbers: make([]int, cells),
91 marks: make([]Mark, cells), neighbors: make([][]int, cells), state: ready, exploded: -1,
92 }
93 for cell := 0; cell < cells; cell++ {
94 x, y := cell%width, cell/width
95 for dy := -1; dy <= 1; dy++ {
96 for dx := -1; dx <= 1; dx++ {
97 if (dx != 0 || dy != 0) && x+dx >= 0 && x+dx < width && y+dy >= 0 && y+dy < height {
98 game.neighbors[cell] = append(game.neighbors[cell], (y+dy)*width+x+dx)
99 }
100 }
101 }
102 }
103 return game
104}

163func (g *Game) placeMines(first int, safety Safety) {
164 sx, sy := first%g.width, first/g.width
165 var candidates []int
166 for cell := 0; cell < g.cells(); cell++ {
167 x, y := cell%g.width, cell/g.width
168 if safety == noSafety || safety == cellSafety && cell != first || safety == blockSafety && (abs(x-sx) >= 2 || abs(y-sy) >= 2) {
169 candidates = append(candidates, cell)
170 }
171 }
172 state := g.seed
173 for i := 0; i < g.mineCount; i++ {
174 state = 48271 * state % 2147483647
175 index := int(state % int64(len(candidates)))
176 g.mine[candidates[index]] = true
177 candidates = slices.Delete(candidates, index, index+1)
178 }
179 g.numberBoard()
180 g.determined = true
181 g.state = playing
182}

184func (g *Game) reveal(cell int, safety Safety) int {
185 if !g.determined { g.placeMines(cell, safety) }
186 if g.mine[cell] {
187 g.state, g.exploded = lost, cell
188 return 0
189 }
190 queue := []int{cell}
191 g.revealed[cell], g.marks[cell] = true, unmarked
192 for head := 0; head < len(queue); head++ {
193 current := queue[head]
194 if g.numbers[current] != 0 { continue }
195 for _, next := range g.neighbors[current] {
196 if !g.unknown(next) { continue }
197 g.revealed[next], g.marks[next] = true, unmarked
198 queue = append(queue, next)
199 }
200 }
201 return len(queue)
202}

204func (g *Game) chord(cell int, safety Safety) int {
205 opened := 0
206 for _, next := range g.neighbors[cell] {
207 if g.unknown(next) { opened += g.reveal(next, safety) }
208 if g.state == lost { break }
209 }
210 return opened

221func (g *Game) show() {
222 for y := 0; y < g.height; y++ {
223 row := make([]byte, g.width)
224 for x := 0; x < g.width; x++ {
225 cell := y*g.width+x
226 switch {
227 case g.state == lost && g.mine[cell]:
228 row[x] = '*'
229 if g.marks[cell] == flagged { row[x] = 'F' }
230 if cell == g.exploded { row[x] = '!' }
231 case g.state == lost && g.marks[cell] == flagged:
232 row[x] = 'X'
233 case g.revealed[cell]:
234 row[x] = '.'
235 if g.numbers[cell] > 0 { row[x] = byte('0' + g.numbers[cell]) }
236 case g.marks[cell] == flagged:
237 row[x] = 'F'
238 case g.marks[cell] == questioned:
239 row[x] = '?'
240 default:
241 row[x] = '#'
242 }
243 }
244 line("%s", row)
245 }
246}

257type Constraint struct {
258 cell, remaining int
259 unknown []int
260}
261
262func (g *Game) constraints() []Constraint {
263 var result []Constraint
264 for cell := 0; cell < g.cells(); cell++ {
265 if !g.revealed[cell] || g.numbers[cell] == 0 { continue }
266 c := Constraint{cell: cell, remaining: g.numbers[cell]}
267 for _, next := range g.neighbors[cell] {
268 if g.marks[next] == flagged { c.remaining-- }
269 if g.unknown(next) { c.unknown = append(c.unknown, next) }
270 }
271 if len(c.unknown) > 0 { result = append(result, c) }
272 }
273 return result
274}

281func (r Reason) text(g *Game) string {
282 switch r.kind {
283 case "single":

297func (d *Deduction) add(safe bool, cell int, reason Reason) {
298 side := d.mines
299 if safe { side = d.safe }
300 if side[cell].kind == "" { side[cell] = reason }
301 d.any = true

304func (g *Game) deduce() Deduction {
305 list := g.constraints()
306 result := Deduction{safe: make([]Reason, g.cells()), mines: make([]Reason, g.cells())}
307 for rule := 1; rule <= 2; rule++ {
308 for _, c := range list {
309 target := 0
310 if rule == 2 { target = len(c.unknown) }
311 if c.remaining != target { continue }
312 for _, cell := range c.unknown { result.add(rule == 1, cell, Reason{kind: "single", a: c.cell}) }
313 }
314 if result.any { return result }
315 }
316 at := make([]int, g.cells())
317 for i := range at { at[i] = -1 }
318 for i, c := range list { at[c.cell] = i }
319 for _, a := range list {
320 ax, ay := a.cell%g.width, a.cell/g.width
321 for y := max(0, ay-2); y <= min(g.height-1, ay+2); y++ {
322 for x := max(0, ax-2); x <= min(g.width-1, ax+2); x++ {
323 index := at[y*g.width+x]
324 if index < 0 { continue }
325 b := list[index]
326 if len(a.unknown) >= len(b.unknown) { continue }
327 subset := true
328 for _, cell := range a.unknown {
329 if !slices.Contains(b.unknown, cell) { subset = false; break }
330 }
331 if !subset { continue }
332 difference := b.remaining - a.remaining
333 if difference != 0 && difference != len(b.unknown)-len(a.unknown) { continue }
334 for _, cell := range b.unknown {
335 if !slices.Contains(a.unknown, cell) { result.add(difference == 0, cell, Reason{kind: "subset", a: a.cell, b: b.cell}) }
336 }
337 }
338 }
339 }
340 if result.any { return result }
341 if g.state == playing && g.remaining() == 0 {
342 for cell := 0; cell < g.cells(); cell++ {
343 if g.unknown(cell) { result.add(true, cell, Reason{kind: "count"}) }
344 }
345 }
346 return result
347}

375func root(parent []int, cell int) int {
376 for parent[cell] != cell {
377 parent[cell] = parent[parent[cell]]
378 cell = parent[cell]
379 }
380 return cell
381}
382
383func makeFrontier(g *Game) *Frontier {
384 f := &Frontier{constraints: g.constraints(), componentOf: make([]int, g.cells()), ways: make([]int64, g.cells())}
385 parent, ids := make([]int, g.cells()), make([]int, g.cells())
386 present := make([]bool, g.cells())
387 for i := range parent { parent[i], ids[i], f.componentOf[i] = i, -1, -1 }
388 for _, c := range f.constraints {
389 for _, cell := range c.unknown {
390 present[cell] = true
391 left, right := root(parent, cell), root(parent, c.unknown[0])
392 parent[left] = right
393 }
394 }
395 for cell, exists := range present {
396 if !exists { continue }
397 representative := root(parent, cell)
398 if ids[representative] < 0 {
399 ids[representative] = len(f.components)
400 f.components = append(f.components, Component{})
401 }
402 id := ids[representative]
403 f.componentOf[cell] = id
404 f.components[id].cells = append(f.components[id].cells, cell)
405 }
406 return f
407}

409type Restriction struct {
410 mask uint32
411 target int
412}
413
414type Search struct {
415 size int
416 constraints []Restriction
417 configs int64
418 minimum, maximum int
419 ways []int64
420}
421
422func (s *Search) visit(depth int, mines uint32) {
423 assigned := (uint32(1) << depth) - 1
424 for _, c := range s.constraints {
425 used := bits.OnesCount32(c.mask & mines)
426 left := bits.OnesCount32(c.mask & ^assigned)
427 if used > c.target || used+left < c.target { return }
428 }
429 if depth == s.size {
430 count := bits.OnesCount32(mines)
431 s.configs++
432 s.minimum, s.maximum = min(s.minimum, count), max(s.maximum, count)
433 for i := range s.ways {
434 if mines&(uint32(1)<<i) != 0 { s.ways[i]++ }
435 }
436 return
437 }
438 s.visit(depth+1, mines)
439 s.visit(depth+1, mines|(uint32(1)<<depth))
440}

442func (f *Frontier) count(id int) {
443 component := &f.components[id]
444 local := make([]int, len(f.componentOf))
445 for i, cell := range component.cells { local[cell] = i }
446 search := Search{size: len(component.cells), minimum: len(component.cells)+1, ways: make([]int64, len(component.cells))}
447 for _, c := range f.constraints {
448 if f.componentOf[c.unknown[0]] != id { continue }
449 var mask uint32
450 for _, cell := range c.unknown { mask |= uint32(1) << local[cell] }
451 search.constraints = append(search.constraints, Restriction{mask: mask, target: c.remaining})
452 }
453 search.visit(0, 0)
454 component.configs, component.minimum, component.maximum = search.configs, search.minimum, search.maximum
455 for i, cell := range component.cells { f.ways[cell] = search.ways[i] }
456}

458func fraction(numerator, denominator int64) string {
459 a, b := numerator, denominator
460 for b != 0 { a, b = b, a%b }
461 return fmt.Sprintf("%d/%d", numerator/a, denominator/a)
462}

464func (g *Game) probability(cell int) {
465 f := makeFrontier(g)
466 id := f.componentOf[cell]
467 if id < 0 {
468 line("prob %s unconstrained", g.coord(cell))
469 return
470 }
471 f.count(id)
472 component := f.components[id]
473 if component.configs == 0 {
474 line("prob %s inconsistent", g.coord(cell))
475 } else {
476 line("prob %s %s size %d configs %d", g.coord(cell), fraction(f.ways[cell], component.configs), len(component.cells), component.configs)
477 }
478}

482 total := 0
483 for _, c := range f.components { total += len(c.cells) }
484 line("frontier %d components %d", total, len(f.components))
485 for id := range f.components {

496type AutoResult struct { rounds, flagged, opened, guesses int }
497
498func (g *Game) autoPlay(safety Safety) AutoResult {
499 var total AutoResult
500 for g.state == playing {
501 result := g.deduce()
502 if !result.any { break }
503 total.rounds++
504 for cell, reason := range result.safe {
505 if reason.kind == "" || g.revealed[cell] { continue }
506 g.moves = append(g.moves, Move{autoReveal, cell})
507 total.opened += g.reveal(cell, safety)
508 g.checkWin()
509 if g.state.finished() { return total }
510 }
511 for cell, reason := range result.mines {
512 if reason.kind == "" || !g.unknown(cell) { continue }
513 g.marks[cell] = flagged
514 total.flagged++
515 g.moves = append(g.moves, Move{autoFlag, cell})
516 }
517 }
518 return total
519}

521func (g *Game) solve(safety Safety) AutoResult {
522 var total AutoResult
523 for !g.state.finished() {
524 step := g.autoPlay(safety)
525 total.rounds += step.rounds
526 total.flagged += step.flagged
527 total.opened += step.opened
528 if g.state.finished() { break }
529 f := makeFrontier(g)
530 for id := range f.components { f.count(id) }
531 unknown := int64(0)
532 for cell := 0; cell < g.cells(); cell++ {
533 if g.unknown(cell) { unknown++ }
534 }
535 remaining := int64(max(g.remaining(), 0))
536 choice := -1
537 bestNumerator, bestDenominator := int64(0), int64(1)
538 for cell := 0; cell < g.cells(); cell++ {
539 if !g.unknown(cell) { continue }
540 numerator, denominator := remaining, unknown
541 if id := f.componentOf[cell]; id >= 0 { numerator, denominator = f.ways[cell], f.components[id].configs }
542 if denominator == 0 { continue }
543 if choice < 0 || numerator*bestDenominator < bestNumerator*denominator {
544 choice = cell
545 bestNumerator, bestDenominator = numerator, denominator
546 }
547 }
548 if choice < 0 { break }
549 total.guesses++
550 g.moves = append(g.moves, Move{guessMove, choice})
551 total.opened += g.reveal(choice, safety)
552 g.checkWin()
553 }
554 return total
555}

557func (g *Game) replay(count int, safety Safety) {
558 copy := g.clone(false)
559 copy.restart()
560 for _, move := range g.moves[:count] {
561 switch move.kind {
562 case revealMove, autoReveal, guessMove:
563 copy.reveal(move.cell, safety)
564 case chordMove:
565 copy.chord(move.cell, safety)
566 case flagMove, autoFlag:
567 copy.marks[move.cell] = flagged
568 case unflagMove:
569 copy.marks[move.cell] = unmarked
570 case questionMove:
571 if copy.marks[move.cell] == questioned { copy.marks[move.cell] = unmarked } else { copy.marks[move.cell] = questioned }
572 }
573 copy.checkWin()
574 }
575 line("replay %d/%d", count, len(g.moves))
576 copy.show()
577}

579func (g *Game) boardInfo() {
580 visited := make([]bool, g.cells())
581 openings, largest, isolated := 0, 0, 0
582 for cell := 0; cell < g.cells(); cell++ {
583 if g.mine[cell] { continue }
584 if g.numbers[cell] != 0 {
585 adjacentBlank := false
586 for _, next := range g.neighbors[cell] { adjacentBlank = adjacentBlank || !g.mine[next] && g.numbers[next] == 0 }
587 if !adjacentBlank { isolated++ }
588 } else if !visited[cell] {
589 openings++
590 covered := make([]bool, g.cells())
591 queue := []int{cell}
592 visited[cell], covered[cell] = true, true
593 for head := 0; head < len(queue); head++ {
594 for _, next := range g.neighbors[queue[head]] {
595 covered[next] = true
596 if g.numbers[next] == 0 && !visited[next] { visited[next] = true; queue = append(queue, next) }
597 }
598 }
599 size := 0
600 for _, cell := range covered { if cell { size++ } }
601 largest = max(largest, size)
602 }
603 }
604 line("3bv %d openings %d largest %d mines %d safe %d", openings+isolated, openings, largest, g.mineCount, g.safeCells())
605}

607func (g *Game) neighborCounts(cell int) (flags, unknown, opened int) {
608 for _, next := range g.neighbors[cell] {
609 if g.marks[next] == flagged { flags++ }
610 if g.unknown(next) { unknown++ }
611 if g.revealed[next] { opened++ }
612 }
613 return
614}

616func (g *Game) process(command string, cell int, value int64, safety Safety) error {
617 switch command {
618 case "REVEAL", "FLAG", "UNFLAG", "QUESTION", "PROB":
619 if g.revealed[cell] { return fmt.Errorf("already revealed") }
620 if (command == "REVEAL" || command == "PROB") && g.marks[cell] == flagged { return fmt.Errorf("flagged") }
621 if command == "FLAG" && g.marks[cell] == flagged { return fmt.Errorf("already flagged") }
622 if command == "UNFLAG" && g.marks[cell] != flagged { return fmt.Errorf("not flagged") }
623 }
624 switch command {
625 case "RESTART":
626 g.restart()
627 line("restart %dx%d mines %d", g.width, g.height, g.mineCount)
628 case "REVEAL":
629 g.moves = append(g.moves, Move{revealMove, cell})
630 opened := g.reveal(cell, safety)
631 if g.state != lost { line("revealed %d", opened); g.checkWin() }
632 g.endingOutput()
633 case "FLAG":
634 g.marks[cell] = flagged
635 g.moves = append(g.moves, Move{flagMove, cell})
636 line("flagged %s left %d", g.coord(cell), g.remaining())
637 case "UNFLAG":
638 g.marks[cell] = unmarked
639 g.moves = append(g.moves, Move{unflagMove, cell})
640 line("unflagged %s left %d", g.coord(cell), g.remaining())
641 case "QUESTION":
642 g.moves = append(g.moves, Move{questionMove, cell})
643 name := "questioned"
644 if g.marks[cell] == questioned { g.marks[cell], name = unmarked, "unmarked" } else { g.marks[cell] = questioned }
645 line("%s %s left %d", name, g.coord(cell), g.remaining())
646 case "CHORD":
647 if !g.revealed[cell] { return fmt.Errorf("not revealed") }
648 if g.numbers[cell] == 0 { return fmt.Errorf("no number") }
649 flags, unknown, _ := g.neighborCounts(cell)
650 if flags != g.numbers[cell] { return fmt.Errorf("flags %d of %d", flags, g.numbers[cell]) }
651 if unknown == 0 { return fmt.Errorf("nothing to reveal") }
652 g.moves = append(g.moves, Move{chordMove, cell})
653 opened := g.chord(cell, safety)
654 if g.state != lost { line("chorded %d", opened); g.checkWin() }
655 g.endingOutput()
656 case "UNDO":
657 if len(g.undo) == 0 { return fmt.Errorf("nothing to undo") }
658 snapshot := g.undo[len(g.undo)-1]
659 g.undo = g.undo[:len(g.undo)-1]
660 g.revealed, g.marks = snapshot.revealed, snapshot.marks
661 g.moves = g.moves[:snapshot.moves]
662 line("undo %s moves %d", snapshot.name, len(g.moves))
663 case "SHOW":
664 g.show()
665 case "STATUS":
666 line("status %s left %d moves %d revealed %d/%d", g.state, g.remaining(), len(g.moves), g.opened(), g.safeCells())
667 case "CELL":
668 flags, unknown, opened := g.neighborCounts(cell)
669 if g.revealed[cell] {
670 line("cell %s number %d flags %d unknown %d", g.coord(cell), g.numbers[cell], flags, unknown)
671 } else {
672 mark := []string{"none", "flag", "question"}[g.marks[cell]]
673 line("cell %s hidden mark %s open %d", g.coord(cell), mark, opened)
674 }
675 case "HISTORY":
676 line("history %d", len(g.moves))
677 count := int(min(value, int64(len(g.moves))))
678 for i := len(g.moves)-count; i < len(g.moves); i++ {
679 move := g.moves[i]
680 line("%d %s %s", i+1, move.kind, g.coord(move.cell))
681 }
682 case "REPLAY":
683 if value > int64(len(g.moves)) { return fmt.Errorf("no move %d", value) }
684 g.replay(int(value), safety)
685 case "HINT":
686 g.hint()
687 case "AUTO", "SOLVE":
688 var result AutoResult
689 if command == "AUTO" { result = g.autoPlay(safety) } else { result = g.solve(safety) }
690 if command == "SOLVE" {
691 line("solve guesses %d rounds %d flagged %d revealed %d", result.guesses, result.rounds, result.flagged, result.opened)
692 } else {
693 line("auto rounds %d flagged %d revealed %d", result.rounds, result.flagged, result.opened)
694 }
695 g.endingOutput()
696 case "ANALYZE":
697 copy := g.clone(true)
698 result := copy.autoPlay(safety)
699 switch copy.state {
700 case lost:
701 line("analyze boom %s rounds %d", copy.coord(copy.exploded), result.rounds)
702 case won:
703 line("analyze clear rounds %d", result.rounds)
704 default:
705 line("analyze stuck rounds %d revealed %d/%d", result.rounds, copy.opened(), copy.safeCells())
706 }
707 case "PROB":
708 g.probability(cell)
709 case "FRONTIER":
710 g.frontierReport()
711 case "BOARDINFO":
712 g.boardInfo()
713 }
714 return nil
715}

717func numberWord(word string) (int64, error) {
718 if len(word) == 0 || len(word) > 10 || len(word) > 1 && word[0] == '0' { return 0, fmt.Errorf("bad arguments") }
719 for i := 0; i < len(word); i++ {
720 if word[i] < '0' || word[i] > '9' { return 0, fmt.Errorf("bad arguments") }
721 }
722 return strconv.ParseInt(word, 10, 64)
723}

740type Statistics struct {
741 played, won, lost, streak, best int
742 fastest int
743 hasFastest bool
744}
745
746func (s *Statistics) finish(g *Game) {
747 s.played++
748 if g.state == won {
749 s.won++
750 s.streak++
751 s.best = max(s.best, s.streak)
752 if !s.hasFastest || len(g.moves) < s.fastest { s.fastest, s.hasFastest = len(g.moves), true }
753 } else {
754 s.lost++
755 s.streak = 0
756 }
757}

759func (s *Statistics) show() {
760 rate, fastest := "-", "-"
761 if s.played > 0 { rate = strconv.Itoa(s.won*100/s.played) + "%" }
762 if s.hasFastest { fastest = strconv.Itoa(s.fastest) }
763 line("stats played %d won %d lost %d rate %s streak %d best %d fastest %s", s.played, s.won, s.lost, rate, s.streak, s.best, fastest)
764}

772func (t *Table) layout(width, height int, rows []string) error {
773 for i, row := range rows {
774 if len(row) != width { return fmt.Errorf("bad row %d", i+1) }
775 for _, c := range row {
776 if c != '*' && c != '.' { return fmt.Errorf("bad row %d", i+1) }
777 }
778 }
779 mines := make([]bool, width*height)
780 count := 0
781 for y, row := range rows {
782 for x, c := range row {
783 if c == '*' { mines[y*width+x] = true; count++ }
784 }
785 }
786 if count == 0 || count == width*height { return fmt.Errorf("bad mine count %d", count) }
787 game := newGame(width, height, count, 0)
788 game.mine, game.determined, game.state = mines, true, playing
789 game.numberBoard()
790 t.game = game
791 line("layout %dx%d mines %d", width, height, count)
792 return nil
793}

795func (t *Table) execute(words []string) error {
796 command := words[0]
797 arguments, exists := commandArgs[command]
798 if !exists { return fmt.Errorf("unknown command %s", command) }
799 if len(words) != arguments+1 { return fmt.Errorf("bad arguments") }
800 values := make([]int64, arguments)
801 safety := t.safety
802 if command == "PRESET" {
803 var dimensions []int64
804 switch words[1] {
805 case "beginner": dimensions = []int64{9, 9, 10}
806 case "intermediate": dimensions = []int64{16, 16, 40}
807 case "expert": dimensions = []int64{30, 16, 99}
808 default: return fmt.Errorf("bad arguments")
809 }
810 seed, err := numberWord(words[2])
811 if err != nil { return err }
812 values = append(dimensions, seed)
813 } else if command == "SAFETY" {
814 switch words[1] {
815 case "block": safety = blockSafety
816 case "cell": safety = cellSafety
817 case "none": safety = noSafety
818 default: return fmt.Errorf("bad arguments")
819 }
820 } else {
821 for i, word := range words[1:] {
822 value, err := numberWord(word)
823 if err != nil { return err }
824 values[i] = value
825 }
826 }
827 if command == "NEW" || command == "PRESET" {
828 width, height, mines, seed := values[0], values[1], values[2], values[3]
829 if width < 1 || width > 30 || height < 1 || height > 30 || mines < 1 || mines > width*height-9 || seed < 1 || seed > 2147483646 {
830 return fmt.Errorf("bad value")
831 }
832 t.game = newGame(int(width), int(height), int(mines), seed)
833 line("new %dx%d mines %d", width, height, mines)
834 return nil
835 }
836 if command == "HISTORY" && values[0] == 0 { return fmt.Errorf("bad value") }
837 if command == "SAFETY" { t.safety = safety; line("safety %s", safety); return nil }
838 if command == "STATS" { t.stats.show(); return nil }
839 game := t.game
840 if game == nil { return fmt.Errorf("no game") }
841 if hasCell(command) && (values[0] >= int64(game.width) || values[1] >= int64(game.height)) { return fmt.Errorf("out of range") }
842 if needsUnfinished(command) && game.state.finished() { return fmt.Errorf("game over") }
843 if (command == "ANALYZE" || command == "BOARDINFO") && !game.determined { return fmt.Errorf("not started") }
844 snapshot := Snapshot{name: strings.ToLower(command), moves: len(game.moves)}
845 if recordsUndo(command) { snapshot.revealed, snapshot.marks = slices.Clone(game.revealed), slices.Clone(game.marks) }
846 wasFinished := game.state.finished()
847 cell, value := 0, int64(0)
848 if hasCell(command) { cell = int(values[1])*game.width + int(values[0]) }
849 if len(values) > 0 { value = values[0] }
850 if err := game.process(command, cell, value, safety); err != nil { return err }
851 if recordsUndo(command) && len(game.moves) > snapshot.moves { game.undo = append(game.undo, snapshot) }
852 if !wasFinished && game.state.finished() { t.stats.finish(game) }
853 return nil
854}

856func readLine(input *bufio.Reader) string {
857 text, _ := input.ReadString('\n')
858 return strings.TrimSuffix(strings.TrimSuffix(text, "\n"), "\r")
859}

861func main() {
862 input := bufio.NewReader(os.Stdin)
863 count, _ := strconv.Atoi(readLine(input))
864 table := Table{safety: blockSafety}
865 for i := 0; i < count; i++ {
866 words := strings.Split(readLine(input), " ")
867 var err error
868 if words[0] == "LAYOUT" {
869 width, _ := strconv.Atoi(words[1])
870 height, _ := strconv.Atoi(words[2])
871 rows := make([]string, height)
872 for y := range rows { rows[y] = readLine(input) }
873 err = table.layout(width, height, rows)
874 } else { err = table.execute(words) }
875 if err != nil { line("error: %s", err) }
876 }
877 output.Flush()
878}
