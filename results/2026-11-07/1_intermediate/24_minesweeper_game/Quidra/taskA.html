207bool has_deduction(const Deduction result)
208 return result.safe.size() > 0 or result.mines.size() > 0

210Deduction deduction(const Board board)
211 known := constraints(board)
212 result := Deduction()
213 for clue := known[]
214 if clue.remaining == 0
215 add_reasons(&result.safe, clue.unknown, "single {cell_text(board, clue.cell)}")
216 if has_deduction(result)
217 return result
218 for clue := known[]
219 if clue.remaining == int(#clue.unknown)
220 add_reasons(&result.mines, clue.unknown, "single {cell_text(board, clue.cell)}")
221 if has_deduction(result)
222 return result
223 for left := known[]
224 for right := known[]
225 if #left.unknown >= #right.unknown
226 continue
227 contained := true
228 for cell := left.unknown[]
229 if not includes(right.unknown, cell)
230 contained = false
231 break
232 if not contained
233 continue
234 difference := [for cell := right.unknown[] if not includes(left.unknown, cell) then cell]
235 needed := right.remaining - left.remaining
236 reason := "subset {cell_text(board, left.cell)} {cell_text(board, right.cell)}"
237 if needed == 0
238 add_reasons(&result.safe, difference, reason)
239 elif needed == int(#difference)
240 add_reasons(&result.mines, difference, reason)
241 if not has_deduction(result) and board.state == "playing" and remaining_mines(board) == 0
242 add_reasons(&result.safe, unknown_cells(board), "count")
243 return result

276class Search(int[] required, int[] left, int[][] touches, int[] order)

284void search_configs(Search &search, int depth)
285 if depth == int(#search.order)
286 search.configurations += 1
287 search.mine_counts += search.values
288 int total := search.values.sum()
289 if search.configurations == 1
290 search.minimum = total
291 search.maximum = total
292 else
293 search.minimum = math.min(search.minimum, total)
294 search.maximum = math.max(search.maximum, total)
295 return
296 cell := search.order[depth]
297 affected := search.touches[cell]
298 for index := affected[]
299 search.left[index] -= 1
300 for: 0 <= value <= 1
301 valid := true
302 for index := affected[]
303 search.sums[index] += value
304 int needed := search.required[index] - search.sums[index]
305 if needed < 0 or needed > search.left[index]
306 valid = false
307 if valid
308 search.values[cell] = value
309 search_configs(&search, depth + 1)
310 for index := affected[]
311 search.sums[index] -= value
312 for index := affected[]
313 search.left[index] += 1

315Component count_component(const int[] cells, const Constraint[] known)
316 int[][] touches := array(#cells, fill = [])
317 for index := known.indices()[]
318 for cell := known[index].unknown[]
319 local_index := position_of(cells, cell)
320 touches[local_index] = touches[local_index].append(int(index))
321 degrees := [for adjacent := touches[] then int(#adjacent)]
322 width := int(#cells) + 1
323 largest := degrees.max()
324 keys := [for index := degrees.indices()[] then (largest - degrees[index]) * width + int(index)]
325 order := keys.sorted() % width
326 required := [for clue := known[] then clue.remaining]
327 left := [for clue := known[] then int(#clue.unknown)]
328 search := Search(required, left, touches, order)
329 possible := true
330 for index := known.indices()[]
331 if required[index] < 0 or required[index] > left[index]
332 possible = false
333 if possible
334 search_configs(&search, 0)
335 return Component(cells, search.mine_counts, search.configurations, search.minimum, search.maximum)

337Frontier frontier(const Board board, target := -1)
338 known := constraints(board)
339 int[][] members := array(#board.revealed, fill = [])
340 cells := array(#board.revealed, fill = false)
341 for index := known.indices()[]
342 for cell := known[index].unknown[]
343 cells[cell] = true
344 members[cell] = members[cell].append(int(index))
345 seen := array(#cells, fill = false)
346 Component[] components := []
347 for start := cells.indices()[]
348 if not cells[start] or seen[start]
349 continue
350 queue := [int(start)]
351 seen[start] = true
352 used := array(#known, fill = false)
353 nat head := 0
354 while head < #queue
355 cell := queue[head]
356 head += 1
357 for index := members[cell][]
358 if used[index]
359 continue
360 used[index] = true
361 for neighbor := known[index].unknown[]
362 if not seen[neighbor]
363 seen[neighbor] = true
364 queue = queue.append(neighbor)
365 if target < 0 or includes(queue, target)
366 selected := [for index := known.indices()[] if used[index] then known[index]]
367 components = components.append(count_component(queue.sorted(), selected))
368 return Frontier(cells, components)

370int guess_cell(const Board board)
371 boundary := frontier(board)
372 unknown := unknown_cells(board)
373 numerators := array(#board.revealed, fill = -1)
374 denominators := array(#board.revealed, fill = 1)
375 for component := boundary.components[]
376 if component.configurations > 0
377 for index := component.cells.indices()[]
378 cell := component.cells[index]
379 numerators[cell] = component.mine_counts[index]
380 denominators[cell] = component.configurations
381 remainder := math.max(remaining_mines(board), 0)
382 for cell := unknown[]
383 if not boundary.cells[cell]
384 numerators[cell] = remainder
385 denominators[cell] = int(#unknown)
386 chosen := -1
387 best_numerator := 0
388 best_denominator := 1
389 for cell := unknown[]
390 if numerators[cell] >= 0 and (chosen < 0 or numerators[cell] * best_denominator < best_numerator * denominators[cell])
391 chosen = cell
392 best_numerator = numerators[cell]
393 best_denominator = denominators[cell]
394 return chosen

586 return error("unknown command " + op)
587 if int(#tokens) != arities[operation] + 1
588 return error("bad arguments")
589 int[] values := []
590 mode := ""
591 if op == "PRESET"
592 preset := index_of(["beginner", "intermediate", "expert"], tokens[1])
593 if preset < 0
594 return error("bad arguments")

598 elif op == "SAFETY"

601 return error("bad arguments")
602 else

604 value := try number_word(token)
605 values = values.append(value)
606 if op == "NEW" or op == "PRESET"

608 height := values[1]

617 if op == "HISTORY" and values[0] == 0
618 return error("bad value")
619 if op == "SAFETY"
620 program.safety = mode
621 emit("safety " + mode)
622 return void
623 if op == "STATS"
624 print_stats(program.stats)
625 return void
626 board := try require_board(program)
627 cell := -1
628 if index_of(["REVEAL", "FLAG", "UNFLAG", "QUESTION", "CHORD", "CELL", "PROB"], op) >= 0
629 if values[0] >= board.width or values[1] >= board.height
630 return error("out of range")
631 cell = values[0] + board.width * values[1]
632 if index_of(["REVEAL", "FLAG", "UNFLAG", "QUESTION", "CHORD", "UNDO", "HINT", "AUTO", "ANALYZE", "PROB", "FRONTIER", "SOLVE"], op) >= 0 and finished(board)
633 return error("game over")
634 if (op == "ANALYZE" or op == "BOARDINFO") and not board.placed
635 return error("not started")
636 undo_index := index_of(["REVEAL", "FLAG", "UNFLAG", "QUESTION", "CHORD", "AUTO", "SOLVE"], op)
637 undo_name := ""
638 if undo_index >= 0
639 names := ["reveal", "flag", "unflag", "question", "chord", "auto", "solve"]
640 undo_name = names[undo_index]
641 before := UndoRecord(undo_name, board.revealed, board.marks, board.moves)
642 previous_moves := int(#board.moves)
643 if op == "RESTART"
644 reset_board(&board)
645 emit("restart {size_text(board)} mines {board.mine_count}")
646 elif index_of(["REVEAL", "FLAG", "UNFLAG", "QUESTION", "PROB"], op) >= 0
647 if board.revealed[cell]
648 return error("already revealed")
649 if (op == "REVEAL" or op == "PROB") and board.marks[cell] == 1
650 return error("flagged")
651 if op == "FLAG" and board.marks[cell] == 1
652 return error("already flagged")
653 if op == "UNFLAG" and board.marks[cell] != 1
654 return error("not flagged")
655 if op == "REVEAL"

658 if board.state != "lost"
659 emit("revealed {opened}")
660 win_check(&board)
661 finish_output(board)
662 elif op == "FLAG"
663 board.marks[cell] = 1
664 board.moves = board.moves.append(Move("flag", cell))
665 emit("flagged {cell_text(board, cell)} left {remaining_mines(board)}")
666 elif op == "UNFLAG"
667 board.marks[cell] = 0
668 board.moves = board.moves.append(Move("unflag", cell))
