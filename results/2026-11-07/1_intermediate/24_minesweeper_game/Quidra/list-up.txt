void place_mines(Board &board, int first, string safety)
    x := first % board.width
    y := first / board.width
    int[] choices := []
    for: 0 <= cell < board.width * board.height
        allowed := safety == "none" or (safety == "cell" and cell != first)
        if safety == "block"
            allowed = math.abs(cell % board.width - x) >= 2 or math.abs(cell / board.width - y) >= 2
        if allowed
            choices = choices.append(cell)
    generator := board.seed
    for: 0 <= placed < board.mine_count
        generator = 48271 * generator % 2147483647
        index := generator % int(#choices)
        board.mines[choices[index]] = true
        choices = choices[0:<index].concat(choices[index + 1:])
    board.placed = true
void reset_board(Board &board)
    board.revealed = array(nat(board.width * board.height), fill = false)
    board.marks = array(nat(board.width * board.height), fill = 0)
    board.moves = []
    board.undos = []
    board.exploded = -1
    board.state = if board.placed then "playing" else "ready"
