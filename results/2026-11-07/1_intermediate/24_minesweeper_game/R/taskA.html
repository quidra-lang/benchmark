24fraction <- function(numerator, denominator) {
25 a <- numerator
26 b <- denominator
27 while (b != 0L) {
28 rest <- a %% b
29 a <- b
30 b <- rest
31 }
32 paste0(numerator %/% a, "/", denominator %/% a)
33}

35new_board <- function(width, height, mine_count, seed) {
36 size <- width * height
37 neighbors <- lapply(seq_len(size), function(cell) {
38 column <- (cell - 1L) %% width
39 row <- (cell - 1L) %/% width
40 result <- integer()
41 for (dy in -1:1) for (dx in -1:1) {
42 x <- column + dx
43 y <- row + dy
44 if ((dx != 0L || dy != 0L) && x >= 0L && x < width && y >= 0L && y < height) {
45 result <- c(result, x + width * y + 1L)
46 }
47 }
48 result
49 })
50 list2env(list(width = width, height = height, mine_count = mine_count, seed = seed,
51 placed = FALSE, mines = rep(FALSE, size), numbers = integer(size),
52 revealed = rep(FALSE, size), marks = integer(size), neighbors = neighbors,
53 state = "ready", exploded = 0L, moves = list(), undos = list()), parent = emptyenv())
54}

70place_mines <- function(board, first, safety) {
71 choices <- seq_along(board$mines)
72 if (safety == "cell") choices <- choices[choices != first]
73 if (safety == "block") {
74 x <- (first - 1L) %% board$width
75 y <- (first - 1L) %/% board$width
76 choices <- choices[abs((choices - 1L) %% board$width - x) >= 2L |
77 abs((choices - 1L) %/% board$width - y) >= 2L]
78 }
79 generator <- as.bigz(board$seed)
80 for (i in seq_len(board$mine_count)) {
81 generator <- (48271L * generator) %% 2147483647L
82 index <- as.integer(as.character(generator %% length(choices))) + 1L
83 board$mines[choices[index]] <- TRUE
84 choices <- choices[-index]
85 }
86 compute_numbers(board)
87 board$placed <- TRUE
88 board$state <- "playing"
89}

100reveal_cell <- function(board, cell, safety) {
101 if (!board$placed) place_mines(board, cell, safety)
102 if (board$mines[cell]) {
103 board$state <- "lost"
104 board$exploded <- cell
105 return(0L)
106 }
107 board$revealed[cell] <- TRUE
108 board$marks[cell] <- 0L
109 queue <- integer(length(board$mines))
110 queue[1L] <- cell
111 head <- 1L
112 tail <- 1L
113 while (head <= tail) {
114 current <- queue[head]
115 head <- head + 1L
116 if (board$numbers[current] == 0L) for (neighbor in board$neighbors[[current]]) {
117 if (is_unknown(board, neighbor)) {
118 board$revealed[neighbor] <- TRUE
119 board$marks[neighbor] <- 0L
120 tail <- tail + 1L
121 queue[tail] <- neighbor
122 }
123 }
124 }
125 tail
126}

169constraints <- function(board) {
170 result <- list()
171 for (cell in which(board$revealed & board$numbers > 0L)) {
172 neighbors <- board$neighbors[[cell]]
173 unknown <- neighbors[is_unknown(board, neighbors)]
174 if (length(unknown)) {
175 result[[length(result) + 1L]] <- list(cell = cell, unknown = unknown,
176 remaining = board$numbers[cell] - sum(board$marks[neighbors] == 1L))
177 }
178 }
179 result

182deduction <- function(board) {
183 known <- constraints(board)
184 safe <- mines <- rep(NA_character_, length(board$mines))
185 add <- function(reasons, cells, reason) {
186 reasons[cells[is.na(reasons[cells])]] <- reason
187 reasons
188 }
189 for (clue in known) if (clue$remaining == 0L) {
190 safe <- add(safe, clue$unknown, paste("single", cell_text(board, clue$cell)))
191 }
192 if (any(!is.na(safe))) return(list(safe = safe, mines = mines))
193 for (clue in known) if (clue$remaining == length(clue$unknown)) {
194 mines <- add(mines, clue$unknown, paste("single", cell_text(board, clue$cell)))
195 }
196 if (any(!is.na(mines))) return(list(safe = safe, mines = mines))
197 for (left in known) for (right in known) {
198 if (length(left$unknown) >= length(right$unknown) || !all(left$unknown %in% right$unknown)) next
199 difference <- right$unknown[!right$unknown %in% left$unknown]
200 needed <- right$remaining - left$remaining
201 reason <- paste("subset", cell_text(board, left$cell), cell_text(board, right$cell))
202 if (needed == 0L) safe <- add(safe, difference, reason)
203 else if (needed == length(difference)) mines <- add(mines, difference, reason)
204 }
205 if (!any(!is.na(safe)) && !any(!is.na(mines)) && board$state == "playing" && remaining_mines(board) == 0L) {
206 safe <- add(safe, unknown_cells(board), "count")
207 }
208 list(safe = safe, mines = mines)
209}

211auto_play <- function(board, safety) {
212 totals <- c(rounds = 0L, flagged = 0L, opened = 0L)
213 while (board$state == "playing") {
214 found <- deduction(board)
215 safe <- which(!is.na(found$safe))
216 mines <- which(!is.na(found$mines))
217 if (!length(safe) && !length(mines)) break
218 totals["rounds"] <- totals["rounds"] + 1L
219 for (cell in safe) if (!board$revealed[cell]) {
220 record_move(board, "auto-reveal", cell)
221 totals["opened"] <- totals["opened"] + reveal_cell(board, cell, safety)
222 win_check(board)
223 if (finished(board)) return(totals)
224 }
225 for (cell in mines) if (is_unknown(board, cell)) {
226 board$marks[cell] <- 1L
227 totals["flagged"] <- totals["flagged"] + 1L
228 record_move(board, "auto-flag", cell)
229 }
230 }
231 totals
232}

234count_component <- function(cells, clues) {
235 local_sets <- lapply(clues, function(clue) match(clue$unknown, cells))
236 required <- vapply(clues, function(clue) clue$remaining, integer(1L))
237 touched <- lapply(seq_along(cells), function(cell) which(vapply(local_sets, function(set) cell %in% set, logical(1L))))
238 order <- order(-lengths(touched), seq_along(cells), method = "radix")
239 left <- lengths(local_sets)
240 assigned <- integer(length(clues))
241 choice <- hits <- integer(length(cells))
242 configs <- 0L
243 minimum <- length(cells)
244 maximum <- 0L
245 search <- function(depth, total) {
246 if (depth > length(cells)) {
247 configs <<- configs + 1L
248 hits <<- hits + choice
249 minimum <<- min(minimum, total)
250 maximum <<- max(maximum, total)
251 return(invisible(NULL))
252 }
253 position <- order[depth]
254 involved <- touched[[position]]
255 left[involved] <<- left[involved] - 1L
256 for (value in 0:1) {
257 choice[position] <<- value
258 assigned[involved] <<- assigned[involved] + value
259 if (all(assigned[involved] <= required[involved] & assigned[involved] + left[involved] >= required[involved])) {
260 search(depth + 1L, total + value)
261 }
262 assigned[involved] <<- assigned[involved] - value
263 }
264 left[involved] <<- left[involved] + 1L
265 }
266 if (all(required >= 0L & required <= left)) search(1L, 0L)
267 list(cells = cells, configs = configs, hits = hits, minimum = minimum, maximum = maximum)
268}

270frontier <- function(board, target = 0L) {
271 known <- constraints(board)
272 incident <- vector("list", length(board$mines))
273 for (index in seq_along(known)) for (cell in known[[index]]$unknown) {
274 incident[[cell]] <- c(incident[[cell]], index)
275 }
276 mask <- lengths(incident) > 0L
277 result <- list(mask = mask, components = list())
278 if (target != 0L && !mask[target]) return(result)
279 visited <- rep(FALSE, length(mask))
280 expanded <- rep(FALSE, length(known))
281 for (root in which(mask)) {
282 if (visited[root]) next
283 queue <- root
284 visited[root] <- TRUE
285 clue_ids <- integer()
286 head <- 1L
287 while (head <= length(queue)) {
288 cell <- queue[head]
289 head <- head + 1L
290 for (index in incident[[cell]]) {
291 if (expanded[index]) next
292 expanded[index] <- TRUE
293 clue_ids <- c(clue_ids, index)
294 added <- known[[index]]$unknown[!visited[known[[index]]$unknown]]
295 visited[added] <- TRUE
296 queue <- c(queue, added)
297 }
298 }
299 cells <- sort(queue, method = "radix")
300 if (target == 0L || target %in% cells) {
301 result$components[[length(result$components) + 1L]] <- count_component(cells, known[clue_ids])
302 if (target != 0L) break
303 }
304 }
305 result
306}

308guess_cell <- function(board) {
309 boundary <- frontier(board)
310 numerator <- integer(length(board$mines))
311 denominator <- integer(length(board$mines))
312 unknown <- unknown_cells(board)
313 outside <- unknown[!boundary$mask[unknown]]
314 numerator[outside] <- max(remaining_mines(board), 0L)
315 denominator[outside] <- length(unknown)
316 for (component in boundary$components) if (component$configs > 0L) {
317 numerator[component$cells] <- component$hits
318 denominator[component$cells] <- component$configs
319 }
320 chosen <- 0L
321 for (cell in which(denominator > 0L)) {
322 if (chosen == 0L || numerator[cell] * denominator[chosen] < numerator[chosen] * denominator[cell]) chosen <- cell
323 }
324 chosen
325}

327solve_board <- function(board, safety) {
328 totals <- c(guesses = 0L, rounds = 0L, flagged = 0L, opened = 0L)
329 repeat {
330 totals[c("rounds", "flagged", "opened")] <- totals[c("rounds", "flagged", "opened")] + auto_play(board, safety)
331 if (finished(board)) break
332 cell <- guess_cell(board)
333 if (cell == 0L) break
334 totals["guesses"] <- totals["guesses"] + 1L
335 record_move(board, "guess", cell)
336 totals["opened"] <- totals["opened"] + reveal_cell(board, cell, safety)
337 win_check(board)
338 if (finished(board)) break
339 }
340 totals
341}

343replay_board <- function(board, count, safety) {
344 replay <- copy_board(board)
345 reset_board(replay)
346 for (i in seq_len(count)) {
347 move <- board$moves[[i]]
348 cell <- move$cell
349 if (move$kind %in% c("reveal", "auto-reveal", "guess")) reveal_cell(replay, cell, safety)
350 else if (move$kind == "chord") chord_cells(replay, cell, safety)
351 else if (move$kind %in% c("flag", "auto-flag")) replay$marks[cell] <- 1L
352 else if (move$kind == "unflag") replay$marks[cell] <- 0L
353 else replay$marks[cell] <- if (replay$marks[cell] == 2L) 0L else 2L
354 win_check(replay)
355 }
356 replay
357}

359board_info <- function(board) {
360 blank <- !board$mines & board$numbers == 0L
361 visited <- covered <- rep(FALSE, length(blank))
362 openings <- largest <- 0L
363 for (root in which(blank)) {
364 if (visited[root]) next
365 openings <- openings + 1L
366 queue <- root
367 visited[root] <- TRUE
368 coverage <- rep(FALSE, length(blank))
369 coverage[root] <- TRUE
370 head <- 1L
371 while (head <= length(queue)) {
372 cell <- queue[head]
373 head <- head + 1L
374 neighbors <- board$neighbors[[cell]]
375 coverage[neighbors] <- TRUE
376 added <- neighbors[blank[neighbors] & !visited[neighbors]]
377 visited[added] <- TRUE
378 queue <- c(queue, added)
379 }
380 covered <- covered | coverage
381 largest <- max(largest, sum(coverage))
382 }
383 emit("3bv", openings + sum(!board$mines & !covered), "openings", openings,
384 "largest", largest, "mines", board$mine_count, "safe", safe_count(board))
385}

418execute <- function(app, words) {
419 command <- words[1L]
420 arguments <- words[-1L]
421 if (!command %in% names(arity)) fail(paste("unknown command", command))
422 if (length(arguments) != arity[[command]]) fail("bad arguments")
423 values <- numeric()
424 if (command == "PRESET") {
425 if (!arguments[1L] %in% c("beginner", "intermediate", "expert")) fail("bad arguments")
426 values <- number_word(arguments[2L])
427 } else if (command == "SAFETY") {
428 if (!arguments[1L] %in% c("block", "cell", "none")) fail("bad arguments")
429 } else if (length(arguments)) values <- vapply(arguments, number_word, numeric(1L))
430 if (command %in% c("NEW", "PRESET")) {
431 if (command == "NEW") {
432 width <- values[1L]
433 height <- values[2L]
434 mines <- values[3L]
435 seed <- values[4L]
436 } else {
437 preset <- switch(arguments[1L], beginner = c(9L, 9L, 10L),
438 intermediate = c(16L, 16L, 40L), expert = c(30L, 16L, 99L))
439 width <- preset[1L]
440 height <- preset[2L]
441 mines <- preset[3L]
442 seed <- values[1L]
443 }
444 if (width < 1L || width > 30L || height < 1L || height > 30L ||
445 mines < 1L || mines > width * height - 9L || seed < 1L || seed > 2147483646L) fail("bad value")
446 app$board <- new_board(as.integer(width), as.integer(height), as.integer(mines), as.integer(seed))
447 emit("new", size_text(app$board), "mines", mines)
448 return(invisible(NULL))
449 }
450 if (command == "HISTORY" && values[1L] == 0) fail("bad value")
451 if (command == "SAFETY") {
452 app$safety <- arguments[1L]
453 emit("safety", app$safety)
454 return(invisible(NULL))
455 }
456 if (command == "STATS") {
457 stats <- app$stats
458 rate <- if (stats$played == 0L) "-" else paste0((100L * stats$won) %/% stats$played, "%")
459 fastest <- if (is.null(stats$fastest)) "-" else stats$fastest
460 emit("stats played", stats$played, "won", stats$won, "lost", stats$lost, "rate", rate,
461 "streak", stats$streak, "best", stats$best, "fastest", fastest)
462 return(invisible(NULL))
463 }
464 board <- app$board
465 if (is.null(board)) fail("no game")
466 cell <- 0L
467 if (command %in% cell_commands) {
468 if (values[1L] >= board$width || values[2L] >= board$height) fail("out of range")
469 cell <- as.integer(values[1L] + board$width * values[2L]) + 1L
470 }
471 if (command %in% active_commands && finished(board)) fail("game over")
472 if (command %in% c("ANALYZE", "BOARDINFO") && !board$placed) fail("not started")
473 before <- if (command %in% undo_commands) {
474 list(kind = tolower(command), revealed = board$revealed, marks = board$marks, moves = board$moves)
475 } else NULL
476 switch(command,
477 RESTART = {
478 reset_board(board)
479 emit("restart", size_text(board), "mines", board$mine_count)
480 },
481 REVEAL = {
482 if (board$revealed[cell]) fail("already revealed")
483 if (board$marks[cell] == 1L) fail("flagged")
484 record_move(board, "reveal", cell)
485 opened <- reveal_cell(board, cell, app$safety)
486 if (board$state != "lost") {
487 emit("revealed", opened)
488 win_check(board)
489 }
490 finish_output(board)
491 },
492 FLAG = {
493 if (board$revealed[cell]) fail("already revealed")
494 if (board$marks[cell] == 1L) fail("already flagged")
495 board$marks[cell] <- 1L
496 record_move(board, "flag", cell)
497 emit("flagged", cell_text(board, cell), "left", remaining_mines(board))
498 },
499 UNFLAG = {
500 if (board$revealed[cell]) fail("already revealed")
501 if (board$marks[cell] != 1L) fail("not flagged")
502 board$marks[cell] <- 0L
503 record_move(board, "unflag", cell)
504 emit("unflagged", cell_text(board, cell), "left", remaining_mines(board))
505 },
506 QUESTION = {
507 if (board$revealed[cell]) fail("already revealed")
508 record_move(board, "question", cell)
509 board$marks[cell] <- if (board$marks[cell] == 2L) 0L else 2L
510 emit(if (board$marks[cell] == 0L) "unmarked" else "questioned", cell_text(board, cell), "left", remaining_mines(board))
511 },
512 CHORD = {
513 if (!board$revealed[cell]) fail("not revealed")
514 if (board$numbers[cell] == 0L) fail("no number")
515 neighbors <- board$neighbors[[cell]]
516 flags <- sum(board$marks[neighbors] == 1L)
517 if (flags != board$numbers[cell]) fail(paste("flags", flags, "of", board$numbers[cell]))
518 if (!any(is_unknown(board, neighbors))) fail("nothing to reveal")
519 record_move(board, "chord", cell)
520 opened <- chord_cells(board, cell, app$safety)
521 if (board$state != "lost") {
522 emit("chorded", opened)
523 win_check(board)
524 }
525 finish_output(board)
526 },
527 UNDO = {
528 count <- length(board$undos)
529 if (count == 0L) fail("nothing to undo")
530 saved <- board$undos[[count]]
531 board$undos <- board$undos[-count]
532 board$revealed <- saved$revealed
533 board$marks <- saved$marks
534 board$moves <- saved$moves
535 emit("undo", saved$kind, "moves", length(board$moves))
536 },
537 SHOW = show_board(board),
538 STATUS = emit("status", board$state, "left", remaining_mines(board), "moves", length(board$moves),
539 "revealed", paste0(revealed_count(board), "/", safe_count(board))),
540 CELL = {
541 neighbors <- board$neighbors[[cell]]
542 if (board$revealed[cell]) {
543 emit("cell", cell_text(board, cell), "number", board$numbers[cell],
544 "flags", sum(board$marks[neighbors] == 1L), "unknown", sum(is_unknown(board, neighbors)))
545 } else {
546 emit("cell", cell_text(board, cell), "hidden mark", c("none", "flag", "question")[board$marks[cell] + 1L],
547 "open", sum(board$revealed[neighbors]))
548 }
549 },
550 HISTORY = {
551 count <- length(board$moves)
552 emit("history", count)
553 if (count > 0L) for (i in seq.int(max(1L, count - values[1L] + 1L), count)) {
554 move <- board$moves[[i]]
555 emit(i, move$kind, cell_text(board, move$cell))
556 }
557 },
558 REPLAY = {
559 if (values[1L] > length(board$moves)) fail(paste("no move", arguments[1L]))
560 emit("replay", paste0(arguments[1L], "/", length(board$moves)))
561 show_board(replay_board(board, as.integer(values[1L]), app$safety))
562 },
563 HINT = {
564 found <- deduction(board)
565 safe <- which(!is.na(found$safe))
566 mines <- which(!is.na(found$mines))
567 if (length(safe)) emit("hint safe", cell_text(board, safe[1L]), "by", found$safe[safe[1L]])
568 else if (length(mines)) emit("hint mine", cell_text(board, mines[1L]), "by", found$mines[mines[1L]])
569 else emit("hint none")
570 },
571 AUTO = {
572 totals <- auto_play(board, app$safety)
573 emit("auto rounds", totals["rounds"], "flagged", totals["flagged"], "revealed", totals["opened"])
574 finish_output(board)
575 },
576 ANALYZE = {
577 trial <- copy_board(board)
578 totals <- auto_play(trial, app$safety)
579 if (trial$state == "lost") emit("analyze boom", cell_text(trial, trial$exploded), "rounds", totals["rounds"])
580 else if (trial$state == "won") emit("analyze clear rounds", totals["rounds"])
