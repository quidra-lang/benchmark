33sealed class Component(int[] cells, List<Constraint> constraints, int boardSize)
34{
35 public int[] Cells { get; } = cells;
36 public int Configurations { get; private set; }
37 public int Minimum { get; private set; } = cells.Length;
38 public int Maximum { get; private set; }
39 public int[] MineCounts { get; } = new int[cells.Length];
40 private bool counted;
41
42 public void Count()
43 {
44 if (counted) return;
45 counted = true;
46 if (constraints.Any(c => c.Remaining < 0 || c.Remaining > c.Unknown.Length)) return;
47 var index = new int[boardSize];
48 for (int i = 0; i < Cells.Length; i++) index[Cells[i]] = i;
49 var links = Enumerable.Range(0, Cells.Length).Select(_ => new List<int>()).ToArray();
50 for (int c = 0; c < constraints.Count; c++)
51 foreach (int cell in constraints[c].Unknown) links[index[cell]].Add(c);
52 int[] order = Enumerable.Range(0, Cells.Length).OrderByDescending(i => links[i].Count).ThenBy(i => i).ToArray();
53 int[] remaining = constraints.Select(c => c.Unknown.Length).ToArray();
54 var sums = new int[constraints.Count];
55 var assignment = new int[Cells.Length];
56
57 void Search(int depth, int mines)
58 {
59 if (depth == Cells.Length)
60 {
61 Configurations++;
62 Minimum = Math.Min(Minimum, mines);
63 Maximum = Math.Max(Maximum, mines);
64 for (int i = 0; i < Cells.Length; i++) MineCounts[i] += assignment[i];
65 return;
66 }
67 int variable = order[depth];
68 for (int value = 0; value <= 1; value++)
69 {
70 assignment[variable] = value;
71 bool possible = true;
72 foreach (int c in links[variable])
73 {
74 remaining[c]--;
75 sums[c] += value;
76 int required = constraints[c].Remaining;
77 if (sums[c] > required || sums[c] + remaining[c] < required) possible = false;
78 }
79 if (possible) Search(depth + 1, mines + value);
80 foreach (int c in links[variable]) { sums[c] -= value; remaining[c]++; }
81 }
82 }
83 Search(0, 0);
84 }
85}

89sealed class Game
90{
91 public int Width { get; }
92 public int Height { get; }
93 public int MineCount { get; }
94 public long Seed { get; }
95 public int Size => Width * Height;
96 public bool[]? Mines { get; private set; }
97 public int[] Numbers { get; private set; }
98 public int[][] Neighbors { get; }
99 public bool[] Revealed { get; private set; }
100 public Mark[] Marks { get; private set; }
101 public GameState State { get; private set; }
102 public int Exploded { get; private set; } = -1;
103 public List<Move> Moves { get; private set; } = [];
104 public List<UndoRecord> Undo { get; } = [];
105 public bool Finished => State is GameState.Won or GameState.Lost;
106 public int FlagCount => Marks.Count(mark => mark == Mark.Flag);
107 public int Remaining => MineCount - FlagCount;
108 public int OpenCount => Revealed.Count(value => value);
109 public string Cell(int cell) => $"{cell % Width},{cell / Width}";
110 public bool Unknown(int cell) => !Revealed[cell] && Marks[cell] != Mark.Flag;

112 public Game(int width, int height, int mines, long seed, bool[]? placement = null)
113 {
114 Width = width; Height = height; MineCount = mines; Seed = seed;
115 Revealed = new bool[Size];
116 Marks = new Mark[Size];
117 Numbers = new int[Size];
118 Neighbors = new int[Size][];
119 for (int cell = 0; cell < Size; cell++)
120 {
121 var nearby = new List<int>();
122 int x = cell % Width, y = cell / Width;
123 for (int dy = -1; dy <= 1; dy++)
124 for (int dx = -1; dx <= 1; dx++)
125 if ((dx != 0 || dy != 0) && x + dx >= 0 && x + dx < Width && y + dy >= 0 && y + dy < Height)
126 nearby.Add((y + dy) * Width + x + dx);
127 Neighbors[cell] = nearby.ToArray();
128 }
129 if (placement is not null) SetPlacement(placement);
130 }

132 private Game(Game source, bool reset)
133 {

136 Revealed = reset ? new bool[Size] : (bool[])source.Revealed.Clone();
137 Marks = reset ? new Mark[Size] : (Mark[])source.Marks.Clone();

139 Exploded = reset ? -1 : source.Exploded;

152 private void Place(int first, string safety)
153 {
154 int sx = first % Width, sy = first / Width;
155 var candidates = new List<int>();
156 for (int cell = 0; cell < Size; cell++)
157 if (safety == "none" || safety == "cell" && cell != first ||
158 safety == "block" && (Math.Abs(cell % Width - sx) >= 2 || Math.Abs(cell / Width - sy) >= 2))
159 candidates.Add(cell);
160 var mines = new bool[Size];
161 long random = Seed;
162 for (int i = 0; i < MineCount; i++)
163 {
164 random = 48271 * random % 2147483647;
165 int index = (int)(random % candidates.Count);
166 mines[candidates[index]] = true;
167 candidates.RemoveAt(index);
168 }
169 SetPlacement(mines);
170 }

186 public int Reveal(int first, string safety)
187 {
188 if (Mines is null) Place(first, safety);
189 if (Mines![first]) { State = GameState.Lost; Exploded = first; return 0; }
190 var queue = new Queue<int>();
191 Revealed[first] = true; Marks[first] = Mark.None; queue.Enqueue(first);
192 int opened = 1;
193 while (queue.TryDequeue(out int cell))
194 if (Numbers[cell] == 0)
195 foreach (int neighbor in Neighbors[cell])
196 if (Unknown(neighbor))
197 {
198 Revealed[neighbor] = true; Marks[neighbor] = Mark.None;
199 queue.Enqueue(neighbor); opened++;
200 }
201 return opened;
202 }

204 public int Chord(int cell, string safety)
205 {
206 int opened = 0;
207 foreach (int neighbor in Neighbors[cell])
208 if (Unknown(neighbor))
209 {
210 opened += Reveal(neighbor, safety);
211 if (State == GameState.Lost) break;
212 }
213 return opened;
214 }

216 public bool CheckWin()
217 {
218 if (State != GameState.Playing || OpenCount != Size - MineCount) return false;
219 State = GameState.Won;
220 for (int cell = 0; cell < Size; cell++) if (Mines![cell]) Marks[cell] = Mark.Flag;
221 return true;
222 }

224 public IEnumerable<string> Display()
225 {

228 var row = new char[Width];

236 : Marks[cell] == Mark.Flag ? 'F' : Marks[cell] == Mark.Question ? '?' : '#';

242 private List<Constraint> Constraints()
243 {
244 var result = new List<Constraint>();
245 for (int cell = 0; cell < Size; cell++)
246 if (Revealed[cell] && Numbers[cell] > 0)
247 {
248 int[] unknown = Neighbors[cell].Where(Unknown).ToArray();
249 if (unknown.Length > 0)
250 result.Add(new(cell, unknown, Numbers[cell] - Neighbors[cell].Count(n => Marks[n] == Mark.Flag)));
251 }
252 return result;
253 }

255 public Deduction Deduce()
256 {
257 List<Constraint> constraints = Constraints();
258 var result = new Deduction(Size);
259 foreach (Constraint c in constraints)
260 if (c.Remaining == 0)
261 foreach (int cell in c.Unknown) result.Safe[cell] ??= $"single {Cell(c.Cell)}";
262 if (result.Any) return result;
263 foreach (Constraint c in constraints)
264 if (c.Remaining == c.Unknown.Length)
265 foreach (int cell in c.Unknown) result.Mines[cell] ??= $"single {Cell(c.Cell)}";
266 if (result.Any) return result;
267 foreach (Constraint a in constraints)
268 foreach (Constraint b in constraints)
269 if (a.Unknown.Length < b.Unknown.Length && a.Unknown.All(b.Unknown.Contains))
270 {
271 int difference = b.Unknown.Length - a.Unknown.Length;
272 int remaining = b.Remaining - a.Remaining;
273 if (remaining != 0 && remaining != difference) continue;
274 string?[] side = remaining == 0 ? result.Safe : result.Mines;
275 foreach (int cell in b.Unknown)
276 if (!a.Unknown.Contains(cell)) side[cell] ??= $"subset {Cell(a.Cell)} {Cell(b.Cell)}";
277 }
278 if (result.Any) return result;
279 if (State == GameState.Playing && Remaining == 0)
280 for (int cell = 0; cell < Size; cell++) if (Unknown(cell)) result.Safe[cell] = "count";
281 return result;
282 }

284 public Frontier BuildFrontier()
285 {
286 List<Constraint> constraints = Constraints();
287 int[] parents = Enumerable.Range(0, Size).ToArray();
288 var frontier = new bool[Size];
289 int Root(int cell)
290 {
291 while (parents[cell] != cell) { parents[cell] = parents[parents[cell]]; cell = parents[cell]; }
292 return cell;
293 }
294 foreach (Constraint c in constraints)
295 foreach (int cell in c.Unknown)
296 {
297 frontier[cell] = true;
298 parents[Root(cell)] = Root(c.Unknown[0]);
299 }
300 int[] groupForRoot = Enumerable.Repeat(-1, Size).ToArray();
301 var groups = new List<List<int>>();
302 for (int cell = 0; cell < Size; cell++)
303 if (frontier[cell])
304 {
305 int root = Root(cell);
306 if (groupForRoot[root] == -1) { groupForRoot[root] = groups.Count; groups.Add([]); }
307 groups[groupForRoot[root]].Add(cell);
308 }
309 var byGroup = groups.Select(_ => new List<Constraint>()).ToArray();
310 foreach (Constraint c in constraints) byGroup[groupForRoot[Root(c.Unknown[0])]].Add(c);
311 var components = new List<Component>();
312 var at = new Component?[Size];
313 for (int i = 0; i < groups.Count; i++)
314 {
315 var component = new Component(groups[i].ToArray(), byGroup[i], Size);
316 components.Add(component);
317 foreach (int cell in component.Cells) at[cell] = component;
318 }
319 return new(components, at);
320 }

322 public Totals Auto(string safety)
323 {
324 var totals = new Totals();
325 while (State == GameState.Playing)
326 {
327 Deduction deduction = Deduce();
328 if (!deduction.Any) break;
329 totals.Rounds++;
330 for (int cell = 0; cell < Size; cell++)
331 if (deduction.Safe[cell] is not null && !Revealed[cell])
332 {
333 Moves.Add(new(MoveKind.AutoReveal, cell));
334 totals.Opened += Reveal(cell, safety);
335 CheckWin();
336 if (Finished) return totals;
337 }
338 for (int cell = 0; cell < Size; cell++)
339 if (deduction.Mines[cell] is not null && Unknown(cell))
340 {
341 Marks[cell] = Mark.Flag; totals.Flagged++;
342 Moves.Add(new(MoveKind.AutoFlag, cell));
343 }
344 }
345 return totals;
346 }

348 public Totals Solve(string safety)
349 {
350 var totals = new Totals();
351 while (!Finished)
352 {
353 totals.Add(Auto(safety));
354 if (Finished) break;
355 Frontier frontier = BuildFrontier();
356 var numerators = new int[Size];
357 var denominators = new int[Size];
358 foreach (Component component in frontier.Components)
359 {
360 component.Count();
361 for (int i = 0; i < component.Cells.Length; i++)
362 {
363 numerators[component.Cells[i]] = component.MineCounts[i];
364 denominators[component.Cells[i]] = component.Configurations;
365 }
366 }
367 int unknown = Enumerable.Range(0, Size).Count(Unknown);
368 int backgroundMines = Math.Max(Remaining, 0);
369 int chosen = -1, bestNumerator = 0, bestDenominator = 1;
370 for (int cell = 0; cell < Size; cell++)
371 if (Unknown(cell))
372 {
373 int numerator = frontier.At[cell] is null ? backgroundMines : numerators[cell];
374 int denominator = frontier.At[cell] is null ? unknown : denominators[cell];
375 if (denominator != 0 && (chosen == -1 || (long)numerator * bestDenominator < (long)bestNumerator * denominator))
376 { chosen = cell; bestNumerator = numerator; bestDenominator = denominator; }
377 }
378 if (chosen == -1) break;
379 totals.Guesses++;
380 Moves.Add(new(MoveKind.Guess, chosen));
381 totals.Opened += Reveal(chosen, safety);
382 CheckWin();
383 }
384 return totals;
385 }

387 public Game Replay(int count, string safety)
388 {
389 Game copy = Copy(true);
390 foreach (Move move in Moves.Take(count))
391 {
392 switch (move.Kind)
393 {
394 case MoveKind.Reveal: case MoveKind.AutoReveal: case MoveKind.Guess: copy.Reveal(move.Cell, safety); break;
395 case MoveKind.Chord: copy.Chord(move.Cell, safety); break;
396 case MoveKind.Flag: case MoveKind.AutoFlag: copy.Marks[move.Cell] = Mark.Flag; break;
397 case MoveKind.Unflag: copy.Marks[move.Cell] = Mark.None; break;
398 case MoveKind.Question: copy.Marks[move.Cell] = copy.Marks[move.Cell] == Mark.Question ? Mark.None : Mark.Question; break;
399 }
400 copy.Moves.Add(move);
401 copy.CheckWin();
402 }
403 return copy;
404 }

406 public (int Bv, int Openings, int Largest) BoardInfo()
407 {
408 var visited = new bool[Size];
409 int openings = 0, largest = 0, isolated = 0;
410 bool Blank(int cell) => !Mines![cell] && Numbers[cell] == 0;
411 for (int first = 0; first < Size; first++)
412 {
413 if (Blank(first) && !visited[first])
414 {
415 openings++;
416 var area = new HashSet<int> { first };
417 var queue = new Queue<int>();
418 visited[first] = true; queue.Enqueue(first);
419 while (queue.TryDequeue(out int cell))
420 foreach (int neighbor in Neighbors[cell])
421 {
422 area.Add(neighbor);
423 if (Blank(neighbor) && !visited[neighbor]) { visited[neighbor] = true; queue.Enqueue(neighbor); }
424 }
425 largest = Math.Max(largest, area.Count);
426 }
427 if (!Mines![first] && Numbers[first] > 0 && !Neighbors[first].Any(Blank)) isolated++;
428 }
429 return (openings + isolated, openings, largest);
430 }

433sealed class Program
434{
435 private static readonly Dictionary<string, int> Arity = new(StringComparer.Ordinal)
436 {
437 ["NEW"] = 4, ["PRESET"] = 2, ["LAYOUT"] = 2, ["SAFETY"] = 1, ["RESTART"] = 0,
438 ["REVEAL"] = 2, ["FLAG"] = 2, ["UNFLAG"] = 2, ["QUESTION"] = 2, ["CHORD"] = 2,
439 ["UNDO"] = 0, ["SHOW"] = 0, ["STATUS"] = 0, ["CELL"] = 2, ["HISTORY"] = 1,
440 ["REPLAY"] = 1, ["HINT"] = 0, ["AUTO"] = 0, ["ANALYZE"] = 0, ["PROB"] = 2,
441 ["FRONTIER"] = 0, ["SOLVE"] = 0, ["BOARDINFO"] = 0, ["STATS"] = 0
442 };
443 private Game? game;
444 private string safety = "block";
445 private int played, wins, losses, streak, best;
446 private int? fastest;
447 private static void Emit(string line) => Console.WriteLine(line);
448 private static void Fail(string message) => throw new CommandError(message);
449 private static long Number(string word)
450 {
451 if (word.Length > 10 || word.Length > 1 && word[0] == '0' || word.Any(c => c < '0' || c > '9')) Fail("bad arguments");
452 return long.Parse(word, CultureInfo.InvariantCulture);
453 }
454 private static string StateName(GameState state) => state.ToString().ToLowerInvariant();
455 private static string MoveName(MoveKind kind) => kind switch
456 {
457 MoveKind.AutoReveal => "auto-reveal", MoveKind.AutoFlag => "auto-flag", _ => kind.ToString().ToLowerInvariant()
458 };
459 private static void Show(Game board) { foreach (string row in board.Display()) Emit(row); }
460 private static void FinishOutput(Game board)
461 {
462 if (board.State == GameState.Lost) { Emit($"boom {board.Cell(board.Exploded)}"); Show(board); }
463 else if (board.State == GameState.Won) Emit($"win moves {board.Moves.Count}");
464 }
465 private void RecordFinish(Game board)
466 {
467 played++;
468 if (board.State == GameState.Won)
469 {
470 wins++; streak++; best = Math.Max(best, streak);
471 fastest = fastest is null ? board.Moves.Count : Math.Min(fastest.Value, board.Moves.Count);
472 }
473 else { losses++; streak = 0; }
474 }

476 private void Layout(string[] words)
477 {
478 int width = int.Parse(words[1]), height = int.Parse(words[2]);
479 string[] rows = Enumerable.Range(0, height).Select(_ => Console.ReadLine()!).ToArray();
480 for (int y = 0; y < height; y++)
481 if (rows[y].Length != width || rows[y].Any(c => c != '*' && c != '.')) Fail($"bad row {y + 1}");
482 bool[] mines = string.Concat(rows).Select(c => c == '*').ToArray();
483 int count = mines.Count(mine => mine);
484 if (count == 0 || count == width * height) Fail($"bad mine count {count}");
485 game = new Game(width, height, count, 0, mines);
486 Emit($"layout {width}x{height} mines {count}");
487 }

489 private void Execute(string[] words)
490 {
491 string command = words[0];
492 if (command == "LAYOUT") { Layout(words); return; }
493 if (!Arity.TryGetValue(command, out int arity)) Fail($"unknown command {command}");
494 if (words.Length != arity + 1) Fail("bad arguments");
495 var values = new long[arity];
496 for (int i = 0; i < arity; i++)
497 {
498 string word = words[i + 1];
499 if (command == "PRESET" && i == 0)
500 { if (word is not ("beginner" or "intermediate" or "expert")) Fail("bad arguments"); }
501 else if (command == "SAFETY")
502 { if (word is not ("block" or "cell" or "none")) Fail("bad arguments"); }
503 else values[i] = Number(word);
504 }
505 if (command is "NEW" or "PRESET")
506 {
507 long width, height, mines, seed;
508 if (command == "NEW") { width = values[0]; height = values[1]; mines = values[2]; seed = values[3]; }
509 else
510 {
511 (width, height, mines) = words[1] switch { "beginner" => (9L, 9L, 10L), "intermediate" => (16L, 16L, 40L), _ => (30L, 16L, 99L) };
512 seed = values[1];
513 }
514 if (width < 1 || width > 30 || height < 1 || height > 30 || mines < 1 || mines > width * height - 9 || seed < 1 || seed > 2147483646) Fail("bad value");
515 game = new Game((int)width, (int)height, (int)mines, seed);
516 Emit($"new {width}x{height} mines {mines}"); return;
517 }
518 if (command == "HISTORY" && values[0] == 0) Fail("bad value");
519 if (command == "SAFETY") { safety = words[1]; Emit($"safety {safety}"); return; }
520 if (command == "STATS")
521 {
522 string rate = played == 0 ? "-" : $"{wins * 100 / played}%";
523 Emit($"stats played {played} won {wins} lost {losses} rate {rate} streak {streak} best {best} fastest {fastest?.ToString() ?? "-"}"); return;
524 }
525 Game board = game ?? throw new CommandError("no game");
526 int cell = 0;
527 if (command is "REVEAL" or "FLAG" or "UNFLAG" or "QUESTION" or "CHORD" or "CELL" or "PROB")
528 {
529 if (values[0] >= board.Width || values[1] >= board.Height) Fail("out of range");
530 cell = (int)values[1] * board.Width + (int)values[0];
531 }
532 if (board.Finished && command is ("REVEAL" or "FLAG" or "UNFLAG" or "QUESTION" or "CHORD" or "UNDO" or "HINT" or "AUTO" or "ANALYZE" or "PROB" or "FRONTIER" or "SOLVE")) Fail("game over");
533 if (board.Mines is null && command is ("ANALYZE" or "BOARDINFO")) Fail("not started");
534 bool wasFinished = board.Finished;
535 UndoRecord? snapshot = command is "REVEAL" or "FLAG" or "UNFLAG" or "QUESTION" or "CHORD" or "AUTO" or "SOLVE"
536 ? board.Snapshot(command.ToLowerInvariant()) : null;
537 switch (command)
538 {
539 case "RESTART": board.Restart(); Emit($"restart {board.Width}x{board.Height} mines {board.MineCount}"); break;
540 case "REVEAL":
541 if (board.Revealed[cell]) Fail("already revealed");
542 if (board.Marks[cell] == Mark.Flag) Fail("flagged");
543 board.Moves.Add(new(MoveKind.Reveal, cell));
544 int opened = board.Reveal(cell, safety);
545 if (board.State != GameState.Lost) Emit($"revealed {opened}");
546 board.CheckWin(); FinishOutput(board); break;
547 case "FLAG": case "UNFLAG": case "QUESTION":
548 if (board.Revealed[cell]) Fail("already revealed");
549 if (command == "FLAG" && board.Marks[cell] == Mark.Flag) Fail("already flagged");
550 if (command == "UNFLAG" && board.Marks[cell] != Mark.Flag) Fail("not flagged");
551 string action;
552 if (command == "FLAG") { board.Marks[cell] = Mark.Flag; action = "flagged"; }
553 else if (command == "UNFLAG") { board.Marks[cell] = Mark.None; action = "unflagged"; }
554 else if (board.Marks[cell] == Mark.Question) { board.Marks[cell] = Mark.None; action = "unmarked"; }
555 else { board.Marks[cell] = Mark.Question; action = "questioned"; }
556 board.Moves.Add(new(command == "FLAG" ? MoveKind.Flag : command == "UNFLAG" ? MoveKind.Unflag : MoveKind.Question, cell));
557 Emit($"{action} {board.Cell(cell)} left {board.Remaining}"); break;
558 case "CHORD":
559 if (!board.Revealed[cell]) Fail("not revealed");
560 if (board.Numbers[cell] == 0) Fail("no number");
561 int flags = board.Neighbors[cell].Count(n => board.Marks[n] == Mark.Flag);
562 if (flags != board.Numbers[cell]) Fail($"flags {flags} of {board.Numbers[cell]}");
563 if (!board.Neighbors[cell].Any(board.Unknown)) Fail("nothing to reveal");
564 board.Moves.Add(new(MoveKind.Chord, cell));
565 int chorded = board.Chord(cell, safety);
566 if (board.State != GameState.Lost) Emit($"chorded {chorded}");
567 board.CheckWin(); FinishOutput(board); break;
568 case "UNDO":
569 if (board.Undo.Count == 0) Fail("nothing to undo");
570 UndoRecord saved = board.Undo[^1]; board.Undo.RemoveAt(board.Undo.Count - 1); board.Restore(saved);
571 Emit($"undo {saved.Name} moves {board.Moves.Count}"); break;
572 case "SHOW": Show(board); break;
573 case "STATUS": Emit($"status {StateName(board.State)} left {board.Remaining} moves {board.Moves.Count} revealed {board.OpenCount}/{board.Size - board.MineCount}"); break;
574 case "CELL":
575 if (board.Revealed[cell])
576 Emit($"cell {board.Cell(cell)} number {board.Numbers[cell]} flags {board.Neighbors[cell].Count(n => board.Marks[n] == Mark.Flag)} unknown {board.Neighbors[cell].Count(board.Unknown)}");
577 else Emit($"cell {board.Cell(cell)} hidden mark {board.Marks[cell].ToString().ToLowerInvariant()} open {board.Neighbors[cell].Count(n => board.Revealed[n])}");
578 break;
579 case "HISTORY":
580 Emit($"history {board.Moves.Count}");
581 for (int i = board.Moves.Count - (int)Math.Min(values[0], board.Moves.Count); i < board.Moves.Count; i++)
582 Emit($"{i + 1} {MoveName(board.Moves[i].Kind)} {board.Cell(board.Moves[i].Cell)}");
583 break;
584 case "REPLAY":
585 if (values[0] > board.Moves.Count) Fail($"no move {values[0]}");
586 Emit($"replay {values[0]}/{board.Moves.Count}"); Show(board.Replay((int)values[0], safety)); break;
587 case "HINT":
588 Deduction deduction = board.Deduce();
589 int target = Array.FindIndex(deduction.Safe, reason => reason is not null);
590 if (target != -1) Emit($"hint safe {board.Cell(target)} by {deduction.Safe[target]}");
591 else
592 {
593 target = Array.FindIndex(deduction.Mines, reason => reason is not null);
594 Emit(target == -1 ? "hint none" : $"hint mine {board.Cell(target)} by {deduction.Mines[target]}");
595 }
596 break;
597 case "AUTO": case "SOLVE":
598 Totals totals = command == "AUTO" ? board.Auto(safety) : board.Solve(safety);
599 Emit(command == "AUTO" ? $"auto rounds {totals.Rounds} flagged {totals.Flagged} revealed {totals.Opened}"
600 : $"solve guesses {totals.Guesses} rounds {totals.Rounds} flagged {totals.Flagged} revealed {totals.Opened}");
601 FinishOutput(board); break;
602 case "ANALYZE":
603 Game copy = board.Copy(); Totals analysis = copy.Auto(safety);
604 Emit(copy.State == GameState.Lost ? $"analyze boom {copy.Cell(copy.Exploded)} rounds {analysis.Rounds}"
605 : copy.State == GameState.Won ? $"analyze clear rounds {analysis.Rounds}"
606 : $"analyze stuck rounds {analysis.Rounds} revealed {copy.OpenCount}/{copy.Size - copy.MineCount}");
607 break;
608 case "PROB":
609 if (board.Revealed[cell]) Fail("already revealed");
610 if (board.Marks[cell] == Mark.Flag) Fail("flagged");
611 Component? component = board.BuildFrontier().At[cell];
612 if (component is null) Emit($"prob {board.Cell(cell)} unconstrained");
613 else
614 {
615 component.Count();
616 if (component.Configurations == 0) Emit($"prob {board.Cell(cell)} inconsistent");
617 else
618 {
619 int numerator = component.MineCounts[Array.IndexOf(component.Cells, cell)], denominator = component.Configurations;
620 int a = numerator, b = denominator;
621 while (b != 0) (a, b) = (b, a % b);
622 Emit($"prob {board.Cell(cell)} {numerator / a}/{denominator / a} size {component.Cells.Length} configs {component.Configurations}");
623 }
624 }
625 break;
626 case "FRONTIER":
627 Frontier frontier = board.BuildFrontier();
628 Emit($"frontier {frontier.Components.Sum(c => c.Cells.Length)} components {frontier.Components.Count}");
629 foreach (Component part in frontier.Components)
630 {
631 part.Count();
632 string prefix = $"component {board.Cell(part.Cells[0])} size {part.Cells.Length}";
633 Emit(part.Configurations == 0 ? $"{prefix} inconsistent" : $"{prefix} configs {part.Configurations} min {part.Minimum} max {part.Maximum}");
634 }
635 break;
636 case "BOARDINFO":
637 var info = board.BoardInfo();
638 Emit($"3bv {info.Bv} openings {info.Openings} largest {info.Largest} mines {board.MineCount} safe {board.Size - board.MineCount}"); break;
639 }
640 if (snapshot is not null && board.Moves.Count > snapshot.MoveCount) board.Undo.Add(snapshot);
641 if (!wasFinished && board.Finished) RecordFinish(board);
642 }
