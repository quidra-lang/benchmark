63function Board(width::Int, height::Int, mine_count::Int, seed::Int64)
64 cell_count = width * height
65 neighbors = [Int[] for _ in 1:cell_count]
66 for cell in 1:cell_count
67 column, row = (cell - 1) % width, div(cell - 1, width)
68 for dy in -1:1, dx in -1:1
69 x, y = column + dx, row + dy
70 if (dx != 0 || dy != 0) && 0 <= x < width && 0 <= y < height
71 push!(neighbors[cell], x + width * y + 1)
72 end
73 end
74 end
75 Board(width, height, mine_count, seed, false, falses(cell_count), zeros(Int, cell_count),
76 falses(cell_count), fill(UNMARKED, cell_count), neighbors, READY, nothing, Move[], UndoRecord[])
77end

79cell_text(board::Board, cell::Int) = string((cell - 1) % board.width, ",", div(cell - 1, board.width))
80size_text(board::Board) = "$(board.width)x$(board.height)"
81remaining_mines(board::Board)::Int = board.mine_count - count(==(FLAGGED), board.marks)
82revealed_count(board::Board)::Int = count(board.revealed)
83safe_count(board::Board)::Int = board.width * board.height - board.mine_count
84is_unknown(board::Board, cell::Int)::Bool = !board.revealed[cell] && board.marks[cell] != FLAGGED
85unknown_cells(board::Board)::Vector{Int} = [cell for cell in eachindex(board.mines) if is_unknown(board, cell)]
86finished(board::Board)::Bool = board.state == WON || board.state == LOST
87record_move!(board::Board, kind::MoveKind, cell::Int) = push!(board.moves, Move(kind, cell))

89function compute_numbers!(board::Board)
90 for cell in eachindex(board.numbers)
91 board.numbers[cell] = count(neighbor -> board.mines[neighbor], board.neighbors[cell])
92 end
93end

95function place_mines!(board::Board, first_cell::Int, safety::Safety)
96 x, y = (first_cell - 1) % board.width, div(first_cell - 1, board.width)
97 choices = collect(eachindex(board.mines))
98 if safety == CELL_SAFETY
99 filter!(cell -> cell != first_cell, choices)
100 elseif safety == BLOCK
101 filter!(cell -> abs((cell - 1) % board.width - x) >= 2 ||
102 abs(div(cell - 1, board.width) - y) >= 2, choices)
103 end
104 generator = board.seed
105 for _ in 1:board.mine_count
106 generator = 48271 * generator % 2147483647
107 index = Int(generator % length(choices)) + 1
108 board.mines[choices[index]] = true
109 deleteat!(choices, index)
110 end
111 compute_numbers!(board)
112 board.placed = true
113 board.state = PLAYING
114end

116function reset_board!(board::Board)
117 fill!(board.revealed, false)
118 fill!(board.marks, UNMARKED)
119 empty!(board.moves)
120 empty!(board.undos)
121 board.exploded = nothing
122 board.state = board.placed ? PLAYING : READY

125function reveal_cell!(board::Board, cell::Int, safety::Safety)::Int
126 board.placed || place_mines!(board, cell, safety)
127 if board.mines[cell]
128 board.state = LOST
129 board.exploded = cell
130 return 0
131 end
132 board.revealed[cell] = true
133 board.marks[cell] = UNMARKED
134 queue = Int[cell]
135 head = 1
136 while head <= length(queue)
137 current = queue[head]
138 head += 1
139 board.numbers[current] == 0 || continue
140 for neighbor in board.neighbors[current]
141 if is_unknown(board, neighbor)
142 board.revealed[neighbor] = true
143 board.marks[neighbor] = UNMARKED
144 push!(queue, neighbor)
145 end
146 end
147 end
148 length(queue)
149end

151function chord_cells!(board::Board, cell::Int, safety::Safety)::Int
152 opened = 0
153 for neighbor in board.neighbors[cell]
154 is_unknown(board, neighbor) || continue
155 opened += reveal_cell!(board, neighbor, safety)
156 board.state == LOST && break
157 end
158 opened
159end

161function win_check!(board::Board)
162 if board.state == PLAYING && revealed_count(board) == safe_count(board)
163 board.state = WON
164 board.marks[board.mines] .= FLAGGED
165 end
166end

168function show_board(board::Board)
169 symbols = Vector{Char}(undef, length(board.mines))
170 for cell in eachindex(symbols)
171 symbols[cell] = if board.state == LOST && board.mines[cell]
172 cell == board.exploded ? '!' : board.marks[cell] == FLAGGED ? 'F' : '*'
173 elseif board.state == LOST && board.marks[cell] == FLAGGED
174 'X'
175 elseif board.revealed[cell]
176 board.numbers[cell] == 0 ? '.' : Char(Int('0') + board.numbers[cell])
177 else
178 board.marks[cell] == FLAGGED ? 'F' : board.marks[cell] == QUESTIONED ? '?' : '#'
179 end
180 end
181 for row in 0:(board.height - 1)
182 emit(join(view(symbols, (row * board.width + 1):((row + 1) * board.width))))
183 end
184end

186function finish_output(board::Board)
187 if board.state == LOST
188 emit("boom", cell_text(board, something(board.exploded)))
189 show_board(board)
190 elseif board.state == WON
191 emit("win moves", length(board.moves))
192 end

195struct Constraint
196 cell::Int
197 unknown::Vector{Int}
198 remaining::Int
199end
200
201function constraints(board::Board)::Vector{Constraint}
202 result = Constraint[]
203 for cell in eachindex(board.mines)
204 board.revealed[cell] && board.numbers[cell] > 0 || continue
205 neighbors = board.neighbors[cell]
206 unknown = [neighbor for neighbor in neighbors if is_unknown(board, neighbor)]
207 isempty(unknown) && continue
208 flags = count(neighbor -> board.marks[neighbor] == FLAGGED, neighbors)
209 push!(result, Constraint(cell, unknown, board.numbers[cell] - flags))
210 end
211 result
212end

214struct Deduction
215 safe::Dict{Int,String}
216 mines::Dict{Int,String}
217end
218Deduction() = Deduction(Dict{Int,String}(), Dict{Int,String}())
219has_deduction(found::Deduction) = !isempty(found.safe) || !isempty(found.mines)
220
221function add_reasons!(reasons::Dict{Int,String}, cells::Vector{Int}, reason::String)
222 for cell in cells
223 get!(reasons, cell, reason)
224 end
225end
226
227function deduction(board::Board)::Deduction
228 known = constraints(board)
229 result = Deduction()
230 for clue in known
231 clue.remaining == 0 && add_reasons!(result.safe, clue.unknown, "single $(cell_text(board, clue.cell))")
232 end
233 has_deduction(result) && return result
234 for clue in known
235 clue.remaining == length(clue.unknown) && add_reasons!(result.mines, clue.unknown, "single $(cell_text(board, clue.cell))")
236 end
237 has_deduction(result) && return result
238 for left in known, right in known
239 length(left.unknown) < length(right.unknown) || continue
240 all(cell -> cell in right.unknown, left.unknown) || continue
241 difference = [cell for cell in right.unknown if !(cell in left.unknown)]
242 needed = right.remaining - left.remaining
243 reason = "subset $(cell_text(board, left.cell)) $(cell_text(board, right.cell))"
244 if needed == 0
245 add_reasons!(result.safe, difference, reason)
246 elseif needed == length(difference)
247 add_reasons!(result.mines, difference, reason)
248 end
249 end
250 if !has_deduction(result) && board.state == PLAYING && remaining_mines(board) == 0
251 add_reasons!(result.safe, unknown_cells(board), "count")
252 end
253 result
254end

257 rounds::Int
258 flagged::Int
259 opened::Int
260end
261AutoTotals() = AutoTotals(0, 0, 0)

263function auto_play!(board::Board, safety::Safety)::AutoTotals
264 totals = AutoTotals()
265 while board.state == PLAYING
266 found = deduction(board)
267 has_deduction(found) || break
268 totals.rounds += 1
269 for cell in sort!(collect(keys(found.safe)))
270 board.revealed[cell] && continue
271 record_move!(board, AUTO_REVEAL, cell)
272 totals.opened += reveal_cell!(board, cell, safety)
273 win_check!(board)
274 finished(board) && return totals
275 end
276 for cell in sort!(collect(keys(found.mines)))
277 is_unknown(board, cell) || continue
278 board.marks[cell] = FLAGGED
279 totals.flagged += 1
280 record_move!(board, AUTO_FLAG, cell)
281 end
282 end
283 totals
284end

294mutable struct ConfigurationSearch
295 order::Vector{Int}
296 touched::Vector{Vector{Int}}
297 required::Vector{Int}
298 left::Vector{Int}
299 assigned::Vector{Int}
300 choice::BitVector
301 configs::Int
302 hits::Vector{Int}
303 minimum::Int
304 maximum::Int
305end
306
307function search_configurations!(search::ConfigurationSearch, depth::Int, total::Int)
308 if depth > length(search.order)
309 search.configs += 1
310 search.hits .+= search.choice
311 search.minimum = min(search.minimum, total)
312 search.maximum = max(search.maximum, total)
313 return
314 end
315 position = search.order[depth]
316 involved = search.touched[position]
317 search.left[involved] .-= 1
318 for value in 0:1
319 search.choice[position] = value == 1
320 search.assigned[involved] .+= value
321 if all(index -> search.assigned[index] <= search.required[index] <= search.assigned[index] + search.left[index], involved)
322 search_configurations!(search, depth + 1, total + value)
323 end
324 search.assigned[involved] .-= value
325 end
326 search.left[involved] .+= 1
327end

329function count_component(cells::Vector{Int}, clues::Vector{Constraint})::Component
330 positions = Dict(cell => index for (index, cell) in enumerate(cells))
331 local_sets = [[positions[cell] for cell in clue.unknown] for clue in clues]
332 touched = [Int[] for _ in cells]
333 for (index, local_set) in enumerate(local_sets), cell in local_set
334 push!(touched[cell], index)
335 end
336 required = [clue.remaining for clue in clues]
337 left = length.(local_sets)
338 order = sortperm(eachindex(cells); by = cell -> (-length(touched[cell]), cell))
339 search = ConfigurationSearch(order, touched, required, left, zeros(Int, length(clues)),
340 falses(length(cells)), 0, zeros(Int, length(cells)), length(cells), 0)
341 if all(index -> 0 <= required[index] <= left[index], eachindex(required))
342 search_configurations!(search, 1, 0)
343 end
344 Component(cells, search.configs, search.hits, search.minimum, search.maximum)
345end

347struct Frontier
348 mask::BitVector
349 components::Vector{Component}
350end
351
352function frontier(board::Board, target::Union{Nothing,Int}=nothing)::Frontier
353 known = constraints(board)
354 incident = [Int[] for _ in board.mines]
355 for (index, clue) in enumerate(known), cell in clue.unknown
356 push!(incident[cell], index)
357 end
358 mask = BitVector(length.(incident) .> 0)
359 result = Frontier(mask, Component[])
360 target !== nothing && !mask[target] && return result
361 visited = falses(length(mask))
362 expanded = falses(length(known))
363 for root in findall(mask)
364 visited[root] && continue
365 queue = Int[root]
366 visited[root] = true
367 clue_ids = Int[]
368 head = 1
369 while head <= length(queue)
370 cell = queue[head]
371 head += 1
372 for index in incident[cell]
373 expanded[index] && continue
374 expanded[index] = true
375 push!(clue_ids, index)
376 for neighbor in known[index].unknown
377 visited[neighbor] && continue
378 visited[neighbor] = true
379 push!(queue, neighbor)
380 end
381 end
382 end
383 cells = sort!(queue)
384 if target === nothing || target in cells
385 push!(result.components, count_component(cells, known[clue_ids]))
386 target !== nothing && break
387 end
388 end
389 result
390end

392function guess_cell(board::Board)::Union{Nothing,Int}
393 boundary = frontier(board)
394 numerators = zeros(Int, length(board.mines))
395 denominators = zeros(Int, length(board.mines))
396 unknown = unknown_cells(board)
397 for cell in unknown
398 boundary.mask[cell] && continue
399 numerators[cell] = max(remaining_mines(board), 0)
400 denominators[cell] = length(unknown)
401 end
402 for component in boundary.components
403 component.configs == 0 && continue
404 numerators[component.cells] .= component.hits
405 denominators[component.cells] .= component.configs
406 end
407 chosen = nothing
408 for cell in eachindex(denominators)
409 denominators[cell] == 0 && continue
410 if chosen === nothing || numerators[cell] * denominators[chosen] < numerators[chosen] * denominators[cell]
411 chosen = cell
412 end
413 end
414 chosen
415end

417function solve_board!(board::Board, safety::Safety)
418 totals = AutoTotals()
419 guesses = 0
420 while true
421 part = auto_play!(board, safety)
422 totals.rounds += part.rounds
423 totals.flagged += part.flagged
424 totals.opened += part.opened
425 finished(board) && break
426 cell = guess_cell(board)
427 cell === nothing && break
428 guesses += 1
429 record_move!(board, GUESS, cell)
430 totals.opened += reveal_cell!(board, cell, safety)
431 win_check!(board)
432 finished(board) && break
433 end
434 (; guesses, rounds = totals.rounds, flagged = totals.flagged, opened = totals.opened)
435end

437function replay_board(board::Board, move_count::Int, safety::Safety)::Board
438 replay = deepcopy(board)
439 reset_board!(replay)
440 for index in 1:move_count
441 move = board.moves[index]
442 if move.kind in (REVEAL_MOVE, AUTO_REVEAL, GUESS)
443 reveal_cell!(replay, move.cell, safety)
444 elseif move.kind == CHORD_MOVE
445 chord_cells!(replay, move.cell, safety)
446 elseif move.kind in (FLAG_MOVE, AUTO_FLAG)
447 replay.marks[move.cell] = FLAGGED
448 elseif move.kind == UNFLAG_MOVE
449 replay.marks[move.cell] = UNMARKED
450 else
451 replay.marks[move.cell] = replay.marks[move.cell] == QUESTIONED ? UNMARKED : QUESTIONED
452 end
453 win_check!(replay)
454 end
455 replay
456end

458function board_info(board::Board)
459 blank = .!board.mines .& (board.numbers .== 0)
460 visited, covered = falses(length(blank)), falses(length(blank))
461 openings = 0
462 largest = 0
463 for root in findall(blank)
464 visited[root] && continue
465 openings += 1
466 queue = Int[root]
467 visited[root] = true
468 coverage = falses(length(blank))
469 coverage[root] = true
470 head = 1
471 while head <= length(queue)
472 cell = queue[head]
473 head += 1
474 for neighbor in board.neighbors[cell]
475 coverage[neighbor] = true
476 if blank[neighbor] && !visited[neighbor]
477 visited[neighbor] = true
478 push!(queue, neighbor)
479 end
480 end
481 end
482 covered .|= coverage
483 largest = max(largest, count(coverage))
484 end
485 bv = openings + count(cell -> !board.mines[cell] && !covered[cell], eachindex(board.mines))
486 emit("3bv", bv, "openings", openings, "largest", largest, "mines", board.mine_count, "safe", safe_count(board))
487end

519function layout_command!(app::Program, width::Int, height::Int, rows::Vector{String})
520 for (index, row) in enumerate(rows)
521 ncodeunits(row) == width && all(character -> character == '*' || character == '.', row) || fail("bad row $index")
522 end
523 mines = BitVector([character == '*' for row in rows for character in row])
524 mine_count = count(mines)
525 0 < mine_count < width * height || fail("bad mine count $mine_count")
526 board = Board(width, height, mine_count, Int64(0))
527 board.mines = mines
528 compute_numbers!(board)
529 board.placed = true
530 board.state = PLAYING
531 app.board = board
532 emit("layout", size_text(board), "mines", mine_count)

535function execute!(app::Program, words::Vector{String})
536 command = words[1]
537 arguments = words[2:end]
538 haskey(ARITY, command) || fail("unknown command $command")
539 length(arguments) == ARITY[command] || fail("bad arguments")
540 values = Int64[]
541 if command == "PRESET"
542 haskey(PRESETS, arguments[1]) || fail("bad arguments")
543 push!(values, number_word(arguments[2]))
544 elseif command == "SAFETY"
545 haskey(SAFETY_MODES, arguments[1]) || fail("bad arguments")
546 else
547 values = number_word.(arguments)
548 end
549 if command == "NEW" || command == "PRESET"
550 if command == "NEW"
551 width, height, mines, seed = values
552 else
553 width, height, mines = PRESETS[arguments[1]]
554 seed = values[1]
555 end
556 1 <= width <= 30 && 1 <= height <= 30 && 1 <= mines <= width * height - 9 &&
557 1 <= seed <= 2147483646 || fail("bad value")
558 app.board = Board(Int(width), Int(height), Int(mines), seed)
559 emit("new", size_text(app.board), "mines", mines)
560 return
561 end
562 command == "HISTORY" && values[1] == 0 && fail("bad value")
563 if command == "SAFETY"
564 app.safety = SAFETY_MODES[arguments[1]]
565 emit("safety", arguments[1])
566 return
567 elseif command == "STATS"
568 stats = app.stats
569 rate = stats.played == 0 ? "-" : string(div(100 * stats.won, stats.played), "%")
570 emit("stats played", stats.played, "won", stats.won, "lost", stats.lost, "rate", rate,
571 "streak", stats.streak, "best", stats.best, "fastest", something(stats.fastest, "-"))
572 return
573 end
574 board = app.board
575 board === nothing && fail("no game")
576 cell = 0
577 if command in CELL_COMMANDS
578 values[1] < board.width && values[2] < board.height || fail("out of range")
579 cell = Int(values[1] + board.width * values[2]) + 1
580 end
581 command in ACTIVE_COMMANDS && finished(board) && fail("game over")
582 command in ("ANALYZE", "BOARDINFO") && !board.placed && fail("not started")
583 before = command in UNDO_COMMANDS ? UndoRecord(lowercase(command), copy(board.revealed), copy(board.marks), copy(board.moves)) : nothing
584 if command == "RESTART"
585 reset_board!(board)
586 emit("restart", size_text(board), "mines", board.mine_count)
587 elseif command == "REVEAL"
588 board.revealed[cell] && fail("already revealed")
589 board.marks[cell] == FLAGGED && fail("flagged")
590 record_move!(board, REVEAL_MOVE, cell)
591 opened = reveal_cell!(board, cell, app.safety)
592 if board.state != LOST
593 emit("revealed", opened)
594 win_check!(board)
595 end
596 finish_output(board)
597 elseif command == "FLAG"
598 board.revealed[cell] && fail("already revealed")
599 board.marks[cell] == FLAGGED && fail("already flagged")
600 board.marks[cell] = FLAGGED
601 record_move!(board, FLAG_MOVE, cell)
602 emit("flagged", cell_text(board, cell), "left", remaining_mines(board))
603 elseif command == "UNFLAG"
604 board.revealed[cell] && fail("already revealed")
605 board.marks[cell] == FLAGGED || fail("not flagged")
606 board.marks[cell] = UNMARKED
607 record_move!(board, UNFLAG_MOVE, cell)
608 emit("unflagged", cell_text(board, cell), "left", remaining_mines(board))
609 elseif command == "QUESTION"
610 board.revealed[cell] && fail("already revealed")
611 record_move!(board, QUESTION_MOVE, cell)
612 board.marks[cell] = board.marks[cell] == QUESTIONED ? UNMARKED : QUESTIONED
613 emit(board.marks[cell] == UNMARKED ? "unmarked" : "questioned", cell_text(board, cell), "left", remaining_mines(board))
614 elseif command == "CHORD"
615 board.revealed[cell] || fail("not revealed")
616 board.numbers[cell] > 0 || fail("no number")
617 neighbors = board.neighbors[cell]
618 flags = count(neighbor -> board.marks[neighbor] == FLAGGED, neighbors)
619 flags == board.numbers[cell] || fail("flags $flags of $(board.numbers[cell])")
620 any(neighbor -> is_unknown(board, neighbor), neighbors) || fail("nothing to reveal")
621 record_move!(board, CHORD_MOVE, cell)
622 opened = chord_cells!(board, cell, app.safety)
623 if board.state != LOST
624 emit("chorded", opened)
625 win_check!(board)
626 end
627 finish_output(board)
628 elseif command == "UNDO"
629 isempty(board.undos) && fail("nothing to undo")
630 saved = pop!(board.undos)
631 board.revealed = copy(saved.revealed)
632 board.marks = copy(saved.marks)
633 board.moves = copy(saved.moves)
634 emit("undo", saved.command, "moves", length(board.moves))
635 elseif command == "SHOW"
636 show_board(board)
637 elseif command == "STATUS"
638 emit("status", lowercase(string(board.state)), "left", remaining_mines(board), "moves", length(board.moves),
639 "revealed", "$(revealed_count(board))/$(safe_count(board))")
640 elseif command == "CELL"
641 neighbors = board.neighbors[cell]
642 if board.revealed[cell]
643 emit("cell", cell_text(board, cell), "number", board.numbers[cell],
644 "flags", count(neighbor -> board.marks[neighbor] == FLAGGED, neighbors),
645 "unknown", count(neighbor -> is_unknown(board, neighbor), neighbors))
646 else
647 mark = board.marks[cell] == FLAGGED ? "flag" : board.marks[cell] == QUESTIONED ? "question" : "none"
648 emit("cell", cell_text(board, cell), "hidden mark", mark,
649 "open", count(neighbor -> board.revealed[neighbor], neighbors))
650 end
651 elseif command == "HISTORY"
652 move_count = length(board.moves)
653 emit("history", move_count)
654 for index in max(1, move_count - values[1] + 1):move_count
655 move = board.moves[index]
656 emit(index, MOVE_NAMES[Int(move.kind)], cell_text(board, move.cell))
657 end
658 elseif command == "REPLAY"
659 values[1] <= length(board.moves) || fail("no move $(values[1])")
660 emit("replay", "$(values[1])/$(length(board.moves))")
661 show_board(replay_board(board, Int(values[1]), app.safety))
662 elseif command == "HINT"
663 found = deduction(board)
664 if !isempty(found.safe)
665 target = minimum(keys(found.safe))
666 emit("hint safe", cell_text(board, target), "by", found.safe[target])
667 elseif !isempty(found.mines)
668 target = minimum(keys(found.mines))
669 emit("hint mine", cell_text(board, target), "by", found.mines[target])
670 else
671 emit("hint none")
672 end
673 elseif command == "AUTO"
674 totals = auto_play!(board, app.safety)
675 emit("auto rounds", totals.rounds, "flagged", totals.flagged, "revealed", totals.opened)
676 finish_output(board)
677 elseif command == "ANALYZE"
678 trial = deepcopy(board)
679 totals = auto_play!(trial, app.safety)
680 if trial.state == LOST
681 emit("analyze boom", cell_text(trial, something(trial.exploded)), "rounds", totals.rounds)
682 elseif trial.state == WON
683 emit("analyze clear rounds", totals.rounds)
684 else
685 emit("analyze stuck rounds", totals.rounds, "revealed", "$(revealed_count(trial))/$(safe_count(trial))")
686 end
687 elseif command == "PROB"
688 board.revealed[cell] && fail("already revealed")
689 board.marks[cell] == FLAGGED && fail("flagged")
690 boundary = frontier(board, cell)
691 if !boundary.mask[cell]
692 emit("prob", cell_text(board, cell), "unconstrained")
693 else
694 component = only(boundary.components)
695 if component.configs == 0
696 emit("prob", cell_text(board, cell), "inconsistent")
697 else
698 index = findfirst(==(cell), component.cells)
699 emit("prob", cell_text(board, cell), fraction(component.hits[index], component.configs),
700 "size", length(component.cells), "configs", component.configs)
701 end
702 end
703 elseif command == "FRONTIER"
704 boundary = frontier(board)
705 emit("frontier", count(boundary.mask), "components", length(boundary.components))
706 for component in boundary.components
707 prefix = "component $(cell_text(board, first(component.cells))) size $(length(component.cells))"
708 if component.configs == 0
709 emit(prefix, "inconsistent")
710 else
711 emit(prefix, "configs", component.configs, "min", component.minimum, "max", component.maximum)
712 end
713 end
714 elseif command == "SOLVE"
715 result = solve_board!(board, app.safety)
716 emit("solve guesses", result.guesses, "rounds", result.rounds, "flagged", result.flagged, "revealed", result.opened)
717 finish_output(board)
718 elseif command == "BOARDINFO"
719 board_info(board)
720 end
721 if before !== nothing && length(board.moves) > length(before.moves)
722 push!(board.undos, before)
723 finished(board) && update_statistics!(app.stats, board)
724 end
725end

727function main()
728 command_count = parse(Int, readline())
729 app = Program()
730 for _ in 1:command_count
731 words = String.(split(readline(), ' '))
732 rows = words[1] == "LAYOUT" ? [readline() for _ in 1:parse(Int, words[3])] : String[]
733 try
734 if words[1] == "LAYOUT"
735 layout_command!(app, parse(Int, words[2]), parse(Int, words[3]), rows)
736 else
737 execute!(app, words)
738 end
739 catch error
740 error isa CommandError || rethrow()
741 emit("error:", error.reason)
742 end
743 end
744end
