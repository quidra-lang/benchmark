73 def initialize(width, height, mine_count, seed, layout: nil)
74 @width, @height, @mine_count, @seed = width, height, mine_count, seed
75 @cells = width * height
76 @safe_cells = cells - mine_count

78 x, y = cell % width, cell / width

81 (-1..1).each do |dx|

221 def constraints
222 (0...cells).filter_map do |cell|
223 next unless revealed[cell] && numbers[cell].positive?
224
225 flags = neighbors[cell].count { |other| marks[other] == :flag }
226 unknown_cells = neighbors[cell].select { |other| unknown?(other) }
227 Constraint.new(cell, numbers[cell] - flags, unknown_cells) unless unknown_cells.empty?
228 end

231 def deduce
232 constraints = self.constraints
233 result = Deduction.new
234 (1..2).each do |rule|
235 constraints.each do |constraint|
236 target = rule == 1 ? 0 : constraint.unknown.length
237 next unless constraint.remaining == target
238
239 reason = Reason.new(:single, constraint.cell, nil)
240 constraint.unknown.each { |cell| result.add(rule == 1, cell, reason) }
241 end
242 return result unless result.empty?
243 end
244 at = Array.new(cells)
245 constraints.each { |constraint| at[constraint.cell] = constraint }
246 constraints.each do |a|
247 ax, ay = a.cell % width, a.cell / width
248 ([0, ay - 2].max..[height - 1, ay + 2].min).each do |y|
249 ([0, ax - 2].max..[width - 1, ax + 2].min).each do |x|
250 b = at[y * width + x]
251 next unless b && a.unknown.length < b.unknown.length && a.unknown.all? { |cell| b.unknown.include?(cell) }
252
253 difference = b.unknown - a.unknown
254 remaining = b.remaining - a.remaining
255 next unless remaining.zero? || remaining == difference.length
256
257 reason = Reason.new(:subset, a.cell, b.cell)
258 difference.each { |cell| result.add(remaining.zero?, cell, reason) }
259 end
260 end
261 end
262 return result unless result.empty?
263
264 if state == :playing && self.remaining.zero?
265 reason = Reason.new(:count, nil, nil)
266 cells.times { |cell| result.add(true, cell, reason) if unknown?(cell) }
267 end
268 result
269 end

285 total = AutoResult.new
286 while state == :playing
287 result = deduce

292 next if revealed[cell]

296 check_win

302 marks[cell] = :flag

310 def solve(safety)
311 total = AutoResult.new
312 until finished?
313 step = auto_play(safety)
314 total.rounds += step.rounds
315 total.flagged += step.flagged
316 total.opened += step.opened
317 break if finished?
318
319 frontier = Frontier.new(self)
320 frontier.components.each_index { |id| frontier.count(id) }
321 unknown_count = (0...cells).count { |cell| unknown?(cell) }
322 remaining_count = [remaining, 0].max
323 choice = nil
324 best_numerator, best_denominator = 0, 1
325 cells.times do |cell|
326 next unless unknown?(cell)
327
328 id = frontier.component_of[cell]
329 numerator = id.negative? ? remaining_count : frontier.ways[cell]
330 denominator = id.negative? ? unknown_count : frontier.components[id].configs
331 next if denominator.zero?
332
333 if choice.nil? || numerator * best_denominator < best_numerator * denominator
334 choice = cell
335 best_numerator, best_denominator = numerator, denominator
336 end
337 end
338 break if choice.nil?
339
340 total.guesses += 1
341 moves << Move.new(:guess, choice)
342 total.opened += reveal(choice, safety)
343 check_win
344 end
345 total
346 end

348 def probability(cell)
349 frontier = Frontier.new(self)
350 id = frontier.component_of[cell]
351 if id.negative?
352 puts "prob #{coord(cell)} unconstrained"
353 return
354 end
355 frontier.count(id)
356 component = frontier.components[id]
357 if component.configs.zero?
358 puts "prob #{coord(cell)} inconsistent"
359 else
360 puts "prob #{coord(cell)} #{fraction(frontier.ways[cell], component.configs)} size #{component.cells.length} configs #{component.configs}"
361 end
362 end

367 frontier.components.each_with_index do |component, id|

370 if component.configs.zero?
371 puts "#{start} inconsistent"

378 def replay(count, safety)
379 copy = dup
380 copy.restart
381 moves.first(count).each do |move|
382 cell = move.cell
383 case move.kind
384 when :reveal, :auto_reveal, :guess then copy.reveal(cell, safety)
385 when :chord then copy.chord(cell, safety)
386 when :flag, :auto_flag then copy.marks[cell] = :flag
387 when :unflag then copy.marks[cell] = :none
388 when :question then copy.marks[cell] = copy.marks[cell] == :question ? :none : :question
389 end
390 copy.check_win
391 end
392 puts "replay #{count}/#{moves.length}"
393 copy.show

396 def board_info
397 visited = Array.new(cells, false)
398 openings = largest = isolated = 0
399 cells.times do |cell|
400 next if mine[cell]
401
402 if numbers[cell].positive?
403 isolated += 1 if neighbors[cell].none? { |other| !mine[other] && numbers[other].zero? }
404 elsif !visited[cell]
405 openings += 1
406 covered = Array.new(cells, false)
407 queue = [cell]
408 head = 0
409 visited[cell] = covered[cell] = true
410 while head < queue.length
411 current = queue[head]
412 head += 1
413 neighbors[current].each do |other|
414 covered[other] = true
415 next unless !mine[other] && numbers[other].zero? && !visited[other]
416
417 visited[other] = true
418 queue << other
419 end
420 end
421 largest = [largest, covered.count(true)].max
422 end
423 end
424 puts "3bv #{openings + isolated} openings #{openings} largest #{largest} mines #{mine_count} safe #{safe_cells}"

427 def process(command, cell, value, safety)
428 if %w[REVEAL FLAG UNFLAG QUESTION PROB].include?(command)
429 raise CommandError, 'already revealed' if revealed[cell]
430 raise CommandError, 'flagged' if %w[REVEAL PROB].include?(command) && marks[cell] == :flag
431 raise CommandError, 'already flagged' if command == 'FLAG' && marks[cell] == :flag
432 raise CommandError, 'not flagged' if command == 'UNFLAG' && marks[cell] != :flag
433 end
434 case command
435 when 'RESTART'
436 restart
437 puts "restart #{width}x#{height} mines #{mine_count}"
438 when 'REVEAL'
439 moves << Move.new(:reveal, cell)
440 count = reveal(cell, safety)
441 unless state == :lost
442 puts "revealed #{count}"
443 check_win
444 end
445 ending_output
446 when 'FLAG'
447 marks[cell] = :flag
448 moves << Move.new(:flag, cell)
449 puts "flagged #{coord(cell)} left #{remaining}"
450 when 'UNFLAG'
451 marks[cell] = :none
452 moves << Move.new(:unflag, cell)
453 puts "unflagged #{coord(cell)} left #{remaining}"
454 when 'QUESTION'
455 moves << Move.new(:question, cell)
456 marks[cell] = marks[cell] == :question ? :none : :question
457 puts "#{marks[cell] == :none ? 'unmarked' : 'questioned'} #{coord(cell)} left #{remaining}"
458 when 'CHORD'
459 raise CommandError, 'not revealed' unless revealed[cell]
460 raise CommandError, 'no number' if numbers[cell].zero?
461
462 flags = neighbors[cell].count { |other| marks[other] == :flag }
463 raise CommandError, "flags #{flags} of #{numbers[cell]}" if flags != numbers[cell]
464 raise CommandError, 'nothing to reveal' if neighbors[cell].none? { |other| unknown?(other) }
465
466 moves << Move.new(:chord, cell)
467 count = chord(cell, safety)
468 unless state == :lost
469 puts "chorded #{count}"
470 check_win
471 end
472 ending_output
473 when 'UNDO'
474 snapshot = undo.pop
475 raise CommandError, 'nothing to undo' unless snapshot
476
477 @revealed, @marks = snapshot.revealed, snapshot.marks
478 moves.slice!(snapshot.moves, moves.length - snapshot.moves)
479 puts "undo #{snapshot.command.downcase} moves #{moves.length}"
480 when 'SHOW' then show
481 when 'STATUS'
482 puts "status #{state} left #{remaining} moves #{moves.length} revealed #{opened}/#{safe_cells}"
483 when 'CELL'
484 if revealed[cell]
485 flags = neighbors[cell].count { |other| marks[other] == :flag }
486 unknown_count = neighbors[cell].count { |other| unknown?(other) }
487 puts "cell #{coord(cell)} number #{numbers[cell]} flags #{flags} unknown #{unknown_count}"
488 else
489 open_count = neighbors[cell].count { |other| revealed[other] }
490 puts "cell #{coord(cell)} hidden mark #{marks[cell]} open #{open_count}"
491 end
492 when 'HISTORY'
493 puts "history #{moves.length}"
494 count = [value, moves.length].min
495 (moves.length - count...moves.length).each do |i|
496 move = moves[i]
497 puts "#{i + 1} #{move.kind.to_s.tr('_', '-')} #{coord(move.cell)}"
498 end
499 when 'REPLAY'
500 raise CommandError, "no move #{value}" if value > moves.length
501
502 replay(value, safety)
503 when 'HINT' then hint
504 when 'AUTO', 'SOLVE'
505 result = command == 'AUTO' ? auto_play(safety) : solve(safety)
506 prefix = command == 'AUTO' ? 'auto' : "solve guesses #{result.guesses}"
507 puts "#{prefix} rounds #{result.rounds} flagged #{result.flagged} revealed #{result.opened}"
508 ending_output
509 when 'ANALYZE'
510 copy = dup
511 result = copy.auto_play(safety)
512 case copy.state
513 when :lost then puts "analyze boom #{copy.coord(copy.exploded)} rounds #{result.rounds}"
514 when :won then puts "analyze clear rounds #{result.rounds}"
515 else puts "analyze stuck rounds #{result.rounds} revealed #{copy.opened}/#{copy.safe_cells}"
516 end
517 when 'PROB' then probability(cell)
518 when 'FRONTIER' then frontier_report
519 when 'BOARDINFO' then board_info
520 end
521 end
522end

534class Frontier
535 attr_reader :component_of, :components, :ways
536
537 def initialize(game)
538 @constraints = game.constraints
539 @component_of = Array.new(game.cells, -1)
540 @components = []
541 @ways = Array.new(game.cells, 0)
542 parent = (0...game.cells).to_a
543 ids = Array.new(game.cells, -1)
544 present = Array.new(game.cells, false)
545 @constraints.each do |constraint|
546 constraint.unknown.each do |cell|
547 present[cell] = true
548 left = root(parent, cell)
549 right = root(parent, constraint.unknown.first)
550 parent[left] = right
551 end
552 end
553 game.cells.times do |cell|
554 next unless present[cell]
555
556 representative = root(parent, cell)
557 if ids[representative].negative?
558 ids[representative] = components.length
559 components << Component.new
560 end
561 id = ids[representative]
562 component_of[cell] = id
563 components[id].cells << cell
564 end
565 end
566
567 def count(id)
568 component = components[id]
569 local = Array.new(component_of.length)
570 component.cells.each_with_index { |cell, i| local[cell] = i }
571 restrictions = @constraints.filter_map do |constraint|
572 next unless component_of[constraint.unknown.first] == id
573
574 mask = constraint.unknown.reduce(0) { |bits, cell| bits | (1 << local[cell]) }
575 Restriction.new(mask, constraint.remaining)
576 end
577 search = ConfigurationSearch.new(component.cells.length, restrictions)
578 search.visit(0, 0)
579 component.configs = search.configs
580 component.minimum = search.minimum
581 component.maximum = search.maximum
582 component.cells.each_with_index { |cell, i| ways[cell] = search.ways[i] }
583 end
584
585 private
586
587 def root(parent, cell)
588 while parent[cell] != cell
589 parent[cell] = parent[parent[cell]]
590 cell = parent[cell]
591 end
592 cell
593 end
594end

596class ConfigurationSearch
597 attr_reader :configs, :minimum, :maximum, :ways
598
599 def initialize(size, restrictions)
600 @size, @restrictions = size, restrictions
601 @ways = Array.new(size, 0)
602 @configs = @maximum = 0
603 @minimum = size + 1
604 end
605
606 def visit(depth, mines)
607 assigned = (1 << depth) - 1
608 @restrictions.each do |constraint|
609 used = bit_count(constraint.mask & mines)
610 left = bit_count(constraint.mask & ~assigned)
611 return if used > constraint.target || used + left < constraint.target
612 end
613 if depth == @size
614 count = bit_count(mines)
615 @configs += 1
616 @minimum = [minimum, count].min
617 @maximum = [maximum, count].max
618 @size.times { |i| ways[i] += 1 unless (mines & (1 << i)).zero? }
619 else
620 visit(depth + 1, mines)
621 visit(depth + 1, mines | (1 << depth))
622 end
623 end
624end

658 def layout(width, height, rows)
659 rows.each_with_index do |row, i|
660 raise CommandError, "bad row #{i + 1}" unless row.length == width && /\A[*.]+\z/.match?(row)
661 end

663 raise CommandError, "bad mine count #{count}" if count.zero? || count == width * height

665 @game = Game.new(width, height, count, 0, layout: rows)

669 def execute(words)
670 command = words.first
671 raise CommandError, "unknown command #{command}" unless COMMANDS.key?(command)
672 raise CommandError, 'bad arguments' unless words.length == COMMANDS[command] + 1
673
674 values = []
675 if command == 'PRESET'
676 preset = PRESETS[words[1]]
677 raise CommandError, 'bad arguments' unless preset
678
679 values = [*preset, number_word(words[2])]
680 elsif command == 'SAFETY'
681 raise CommandError, 'bad arguments' unless %w[block cell none].include?(words[1])
682 else
683 values = words.drop(1).map { |word| number_word(word) }
684 end
685 if command == 'NEW' || command == 'PRESET'
686 width, height, mines, seed = values
687 unless (1..30).cover?(width) && (1..30).cover?(height) && mines >= 1 && mines <= width * height - 9 && (1..2_147_483_646).cover?(seed)
688 raise CommandError, 'bad value'
689 end
690 @game = Game.new(width, height, mines, seed)
691 puts "new #{width}x#{height} mines #{mines}"
692 return
693 end
694 raise CommandError, 'bad value' if command == 'HISTORY' && values[0].zero?
695
696 if command == 'SAFETY'
697 @safety = words[1]
698 puts "safety #{@safety}"
699 return
700 end
701 if command == 'STATS'
702 @stats.show
703 return
704 end
705 raise CommandError, 'no game' unless @game
706
707 has_cell = CELL_COMMANDS.include?(command)
708 raise CommandError, 'out of range' if has_cell && (values[0] >= @game.width || values[1] >= @game.height)
709 raise CommandError, 'game over' if UNFINISHED_COMMANDS.include?(command) && @game.finished?
710 raise CommandError, 'not started' if %w[ANALYZE BOARDINFO].include?(command) && !@game.determined?
711
712 snapshot = Snapshot.new(command, @game.revealed.dup, @game.marks.dup, @game.moves.length) if UNDO_COMMANDS.include?(command)
713 was_finished = @game.finished?
714 cell = has_cell ? values[1] * @game.width + values[0] : 0
715 @game.process(command, cell, values[0], @safety)
716 @game.undo << snapshot if snapshot && @game.moves.length > snapshot.moves
717 @stats.finish(@game) if !was_finished && @game.finished?
718 end
719end
