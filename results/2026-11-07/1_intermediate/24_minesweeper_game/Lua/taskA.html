80function Component:search(depth, mines)
81 if depth > #self.cells then
82 self.configurations = self.configurations + 1
83 self.minimum, self.maximum = math.min(self.minimum, mines), math.max(self.maximum, mines)
84 for i, value in ipairs(self.assignment) do
85 self.mine_counts[i] = self.mine_counts[i] + value
86 end
87 return
88 end
89 local variable = self.order[depth]
90 for value = 0, 1 do
91 self.assignment[variable] = value
92 local possible = true
93 for _, c in ipairs(self.links[variable]) do
94 self.remaining[c] = self.remaining[c] - 1
95 self.sums[c] = self.sums[c] + value
96 local required = self.constraints[c].remaining
97 if self.sums[c] > required or self.sums[c] + self.remaining[c] < required then
98 possible = false
99 end
100 end
101 if possible then
102 self:search(depth + 1, mines + value)
103 end
104 for _, c in ipairs(self.links[variable]) do
105 self.sums[c], self.remaining[c] = self.sums[c] - value, self.remaining[c] + 1
106 end
107 end
108end
109function Component:count()
110 if self.counted then
111 return
112 end
113 self.counted = true
114 for _, c in ipairs(self.constraints) do
115 if c.remaining < 0 or c.remaining > #c.unknown then
116 return
117 end
118 end
119 self.index, self.links, self.order, self.remaining = {}, {}, {}, {}
120 for i, cell in ipairs(self.cells) do
121 self.index[cell], self.links[i], self.order[i] = i, {}, i
122 end
123 for c, constraint in ipairs(self.constraints) do
124 self.remaining[c] = #constraint.unknown
125 for _, cell in ipairs(constraint.unknown) do
126 local links = self.links[self.index[cell]]
127 links[#links + 1] = c
128 end
129 end
130 table.sort(
131 self.order,
132 function(a, b)
133 if #self.links[a] ~= #self.links[b] then
134 return #self.links[a] > #self.links[b]
135 end
136 return a < b
137 end
138 )
139 self.sums, self.assignment = fill(#self.constraints, 0), fill(#self.cells, 0)
140 self:search(1, 0)
141end
142function Component:mine_count(cell)
143 return self.mine_counts[self.index[cell]]
144end

148function Game.new(width, height, mine_count, seed, placement)
149 local size = width * height
150 local self =
151 setmetatable(
152 {
153 width = width,
154 height = height,
155 size = size,
156 mine_count = mine_count,
157 seed = seed,
158 numbers = fill(size, 0),
159 neighbors = {},
160 revealed = fill(size, false),
161 marks = fill(size, "none"),
162 state = "ready",
163 moves = {},
164 undo = {}
165 },
166 Game
167 )
168 for cell = 1, size do
169 local x, y = (cell - 1) % width, (cell - 1) // width
170 local nearby = {}
171 for dy = -1, 1 do
172 for dx = -1, 1 do
173 if
174 (dx ~= 0 or dy ~= 0) and x + dx >= 0 and x + dx < width and y + dy >= 0 and
175 y + dy < height
176 then
177 nearby[#nearby + 1] = (y + dy) * width + x + dx + 1
178 end
179 end
180 end
181 self.neighbors[cell] = nearby
182 end
183 if placement then
184 self:set_placement(placement)
185 end
186 return self
187end

242function Game:set_placement(mines)
243 local numbers = {}
244 for cell, nearby in ipairs(self.neighbors) do
245 local number = 0
246 for _, other in ipairs(nearby) do
247 if mines[other] then
248 number = number + 1
249 end
250 end
251 numbers[cell] = number
252 end
253 self.mines, self.numbers, self.state = mines, numbers, "playing"
254end
255function Game:place(first, safety)
256 local sx, sy = (first - 1) % self.width, (first - 1) // self.width
257 local candidates = {}
258 for cell = 1, self.size do
259 if
260 safety == "none" or safety == "cell" and cell ~= first or
261 safety == "block" and
262 (math.abs((cell - 1) % self.width - sx) >= 2 or
263 math.abs((cell - 1) // self.width - sy) >= 2)
264 then
265 candidates[#candidates + 1] = cell
266 end
267 end
268 local mines, random = fill(self.size, false), self.seed
269 for _ = 1, self.mine_count do
270 random = 48271 * random % 2147483647
271 local index = random % #candidates + 1
272 mines[candidates[index]] = true
273 table.remove(candidates, index)
274 end
275 self:set_placement(mines)
276end

279 self.moves, self.undo, self.exploded = {}, {}, nil

281end

299function Game:reveal(first, safety)
300 if not self.mines then
301 self:place(first, safety)
302 end
303 if self.mines[first] then
304 self.state, self.exploded = "lost", first
305 return 0
306 end
307 self.revealed[first], self.marks[first] = true, "none"
308 local queue, head = {first}, 1
309 while head <= #queue do
310 local cell = queue[head]
311 head = head + 1
312 if self.numbers[cell] == 0 then
313 for _, other in ipairs(self.neighbors[cell]) do
314 if self:unknown(other) then
315 self.revealed[other], self.marks[other] = true, "none"
316 queue[#queue + 1] = other
317 end
318 end
319 end
320 end
321 return #queue
322end
323function Game:chord(cell, safety)
324 local opened = 0
325 for _, other in ipairs(self.neighbors[cell]) do
326 if self:unknown(other) then
327 opened = opened + self:reveal(other, safety)
328 if self.state == "lost" then
329 break
330 end
331 end
332 end
333 return opened
334end
335function Game:check_win()
336 if self.state ~= "playing" or self:open_count() ~= self.size - self.mine_count then
337 return false
338 end
339 self.state = "won"
340 for cell, mine in ipairs(self.mines) do
341 if mine then
342 self.marks[cell] = "flag"
343 end
344 end
345 return true
346end
347function Game:display()
348 for y = 0, self.height - 1 do
349 local row = {}
350 for x = 0, self.width - 1 do
351 local cell = y * self.width + x + 1
352 local character
353 if self.state == "lost" and self.mines[cell] then
354 character =
355 cell == self.exploded and "!" or (self.marks[cell] == "flag" and "F" or "*")
356 elseif self.state == "lost" and self.marks[cell] == "flag" then
357 character = "X"
358 elseif self.revealed[cell] then
359 character = self.numbers[cell] == 0 and "." or tostring(self.numbers[cell])
360 else
361 character =
362 self.marks[cell] == "flag" and "F" or
363 (self.marks[cell] == "question" and "?" or "#")
364 end
365 row[#row + 1] = character
366 end
367 emit(table.concat(row))
368 end
369end

378function Game:constraints()
379 local constraints = {}
380 for cell = 1, self.size do
381 if self.revealed[cell] and self.numbers[cell] > 0 then
382 local unknown = self:unknown_neighbors(cell)
383 if #unknown > 0 then
384 local members = {}
385 for _, other in ipairs(unknown) do
386 members[other] = true
387 end
388 constraints[#constraints + 1] = {
389 cell = cell,
390 unknown = unknown,
391 members = members,
392 remaining = self.numbers[cell] - self:flags(cell)
393 }
394 end
395 end
396 end
397 return constraints
398end
399function Game:deduce()
400 local result, constraints = {safe = {}, mines = {}}, self:constraints()
401 for _, c in ipairs(constraints) do
402 if c.remaining == 0 then
403 for _, cell in ipairs(c.unknown) do
404 result.safe[cell] = result.safe[cell] or "single " .. self:cell(c.cell)
405 end
406 end
407 end
408 if any_deduction(result) then
409 return result
410 end
411 for _, c in ipairs(constraints) do
412 if c.remaining == #c.unknown then
413 for _, cell in ipairs(c.unknown) do
414 result.mines[cell] = result.mines[cell] or "single " .. self:cell(c.cell)
415 end
416 end
417 end
418 if any_deduction(result) then
419 return result
420 end
421 for _, a in ipairs(constraints) do
422 for _, b in ipairs(constraints) do
423 if #a.unknown < #b.unknown then
424 local subset = true
425 for _, cell in ipairs(a.unknown) do
426 if not b.members[cell] then
427 subset = false
428 break
429 end
430 end
431 if subset then
432 local difference = {}
433 for _, cell in ipairs(b.unknown) do
434 if not a.members[cell] then
435 difference[#difference + 1] = cell
436 end
437 end
438 local remaining = b.remaining - a.remaining
439 local reason = "subset " .. self:cell(a.cell) .. " " .. self:cell(b.cell)
440 if remaining == 0 then
441 for _, cell in ipairs(difference) do
442 result.safe[cell] = result.safe[cell] or reason
443 end
444 end
445 if remaining == #difference then
446 for _, cell in ipairs(difference) do
447 result.mines[cell] = result.mines[cell] or reason
448 end
449 end
450 end
451 end
452 end
453 end
454 if any_deduction(result) then
455 return result
456 end
457 if self.state == "playing" and self:remaining() == 0 then
458 for cell = 1, self.size do
459 if self:unknown(cell) then
460 result.safe[cell] = "count"
461 end
462 end
463 end
464 return result
465end
466function Game:frontier()
467 local constraints, links = self:constraints(), {}
468 for cell = 1, self.size do
469 links[cell] = {}
470 end
471 for c, constraint in ipairs(constraints) do
472 for _, cell in ipairs(constraint.unknown) do
473 links[cell][#links[cell] + 1] = c
474 end
475 end
476 local seen, components, at = {}, {}, {}
477 for start = 1, self.size do
478 if not seen[start] and #links[start] > 0 then
479 local cells, selected, picked, head = {start}, {}, {}, 1
480 seen[start] = true
481 while head <= #cells do
482 for _, c in ipairs(links[cells[head]]) do
483 if not picked[c] then
484 picked[c] = true
485 selected[#selected + 1] = constraints[c]
486 for _, cell in ipairs(constraints[c].unknown) do
487 if not seen[cell] then
488 seen[cell] = true
489 cells[#cells + 1] = cell
490 end
491 end
492 end
493 end
494 head = head + 1
495 end
496 table.sort(cells)
497 local component = Component.new(cells, selected)
498 components[#components + 1] = component
499 for _, cell in ipairs(cells) do
500 at[cell] = component
501 end
502 end
503 end
504 return {components = components, at = at}
505end
506function Game:auto(safety)
507 local total = totals()
508 while self.state == "playing" do
509 local deduction = self:deduce()
510 if not any_deduction(deduction) then
511 break
512 end
513 total.rounds = total.rounds + 1
514 for cell = 1, self.size do
515 if deduction.safe[cell] and not self.revealed[cell] then
516 self:record("auto-reveal", cell)
517 total.opened = total.opened + self:reveal(cell, safety)
518 self:check_win()
519 if self:finished() then
520 return total
521 end
522 end
523 end
524 for cell = 1, self.size do
525 if deduction.mines[cell] and self:unknown(cell) then
526 self.marks[cell] = "flag"
527 total.flagged = total.flagged + 1
528 self:record("auto-flag", cell)
529 end
530 end
531 end
532 return total
533end
534function Game:solve(safety)
535 local total = totals()
536 while not self:finished() do
537 add_totals(total, self:auto(safety))
538 if self:finished() then
539 break
540 end
541 local frontier = self:frontier()
542 for _, component in ipairs(frontier.components) do
543 component:count()
544 end
545 local unknown_count = 0
546 for cell = 1, self.size do
547 if self:unknown(cell) then
548 unknown_count = unknown_count + 1
549 end
550 end
551 local background = math.max(self:remaining(), 0)
552 local best, best_numerator, best_denominator = nil, 0, 1
553 for cell = 1, self.size do
554 if self:unknown(cell) then
555 local component = frontier.at[cell]
556 local numerator, denominator
557 if not component then
558 numerator, denominator = background, unknown_count
559 elseif component.configurations > 0 then
560 numerator, denominator = component:mine_count(cell), component.configurations
561 end
562 if
563 numerator and
564 (not best or numerator * best_denominator < best_numerator * denominator)
565 then
566 best, best_numerator, best_denominator = cell, numerator, denominator
567 end
568 end
569 end
570 if not best then
571 break
572 end
573 total.guesses = total.guesses + 1
574 self:record("guess", best)
575 total.opened = total.opened + self:reveal(best, safety)
576 self:check_win()
577 end
578 return total
579end
580function Game:replay(count, safety)
581 local copy = self:copy(true)
582 for i = 1, count do
583 local move = self.moves[i]
584 local kind, cell = move.kind, move.cell
585 if kind == "reveal" or kind == "auto-reveal" or kind == "guess" then
586 copy:reveal(cell, safety)
587 elseif kind == "chord" then
588 copy:chord(cell, safety)
589 elseif kind == "flag" or kind == "auto-flag" then
590 copy.marks[cell] = "flag"
591 elseif kind == "unflag" then
592 copy.marks[cell] = "none"
593 else
594 copy.marks[cell] = copy.marks[cell] == "question" and "none" or "question"
595 end
596 copy:check_win()
597 end
598 return copy
599end
600function Game:board_info()
601 local blank, seen = {}, {}
602 for cell = 1, self.size do
603 blank[cell] = not self.mines[cell] and self.numbers[cell] == 0
604 end
605 local openings, largest, isolated = 0, 0, 0
606 for start = 1, self.size do
607 if blank[start] and not seen[start] then
608 openings = openings + 1
609 local queue, head, area = {start}, 1, {[start] = true}
610 seen[start] = true
611 while head <= #queue do
612 for _, cell in ipairs(self.neighbors[queue[head]]) do
613 area[cell] = true
614 if blank[cell] and not seen[cell] then
615 seen[cell] = true
616 queue[#queue + 1] = cell
617 end
618 end
619 head = head + 1
620 end
621 largest = math.max(largest, count_true(area))
622 end
623 if not self.mines[start] and not blank[start] then
624 local adjacent_blank = false
625 for _, cell in ipairs(self.neighbors[start]) do
626 if blank[cell] then
627 adjacent_blank = true
628 break
629 end
630 end
631 if not adjacent_blank then
632 isolated = isolated + 1
633 end
634 end
635 end
636 emit(
637 "3bv",
638 openings + isolated,
639 "openings",
640 openings,
641 "largest",
642 largest,
643 "mines",
644 self.mine_count,
645 "safe",
646 self.size - self.mine_count
647 )
648end

724function Program:layout(words)

777function Program:execute(words)
778 local op = words[1]
779 if op == "LAYOUT" then
780 self:layout(words)
781 return
782 end
783 if ARITY[op] == nil then
784 fail("unknown command " .. op)
785 end
786 if #words ~= ARITY[op] + 1 then
787 fail("bad arguments")
788 end
789 local values = {}
790 if op == "NEW" or CELL_COMMANDS[op] or op == "HISTORY" or op == "REPLAY" then
791 for i = 2, #words do
792 values[#values + 1] = number_word(words[i])
793 end
794 elseif op == "PRESET" then
795 if not PRESETS[words[2]] then
796 fail("bad arguments")
797 end
798 values[1] = number_word(words[3])
799 elseif op == "SAFETY" and not SAFETY[words[2]] then
800 fail("bad arguments")
801 end
802
803 if op == "NEW" or op == "PRESET" then
804 local width, height, mines, seed
805 if op == "NEW" then
806 width, height, mines, seed = table.unpack(values)
807 else
808 width, height, mines = table.unpack(PRESETS[words[2]])
809 seed = values[1]
810 end
811 if
812 width < 1 or width > 30 or height < 1 or height > 30 or mines < 1 or
813 mines > width * height - 9 or
814 seed < 1 or
815 seed > 2147483646
816 then
817 fail("bad value")
818 end
819 self.game = Game.new(width, height, mines, seed)
820 emit("new", width .. "x" .. height, "mines", mines)
821 return
822 end
823 if op == "HISTORY" and values[1] == 0 then
824 fail("bad value")
825 end
826 if op == "SAFETY" then
827 self.safety = words[2]
828 emit("safety", self.safety)
829 return
830 end
831 if op == "STATS" then
832 self:stats()
833 return
834 end
835 local game = self.game
836 if not game then
837 fail("no game")
838 end
839 if CELL_COMMANDS[op] and (values[1] >= game.width or values[2] >= game.height) then
840 fail("out of range")
841 end
842 local cell = CELL_COMMANDS[op] and (values[2] * game.width + values[1] + 1) or 1
843 if game:finished() and ACTIVE_COMMANDS[op] then
844 fail("game over")
845 end
846 if (op == "ANALYZE" or op == "BOARDINFO") and not game.mines then
847 fail("not started")
848 end
849 local was_finished = game:finished()
850 local saved = UNDOABLE[op] and game:snapshot(op:lower()) or nil
851
852 if op == "RESTART" then
853 game:restart()
854 emit("restart", game.width .. "x" .. game.height, "mines", game.mine_count)
855 elseif op == "REVEAL" then
856 if game.revealed[cell] then
857 fail("already revealed")
858 end
859 if game.marks[cell] == "flag" then
860 fail("flagged")
861 end
862 game:record("reveal", cell)
863 local opened = game:reveal(cell, self.safety)
864 if game.state ~= "lost" then
865 emit("revealed", opened)
866 game:check_win()
867 end
868 game:ending()
869 elseif op == "FLAG" or op == "UNFLAG" or op == "QUESTION" then
870 if game.revealed[cell] then
871 fail("already revealed")
872 end
873 if op == "FLAG" and game.marks[cell] == "flag" then
874 fail("already flagged")
875 end
876 if op == "UNFLAG" and game.marks[cell] ~= "flag" then
877 fail("not flagged")
878 end
879 local mark, verb
880 if op == "FLAG" then
881 mark, verb = "flag", "flagged"
882 elseif op == "UNFLAG" then
883 mark, verb = "none", "unflagged"
884 else
885 mark = game.marks[cell] == "question" and "none" or "question"
886 verb = mark == "none" and "unmarked" or "questioned"
887 end
888 game.marks[cell] = mark
889 game:record(op:lower(), cell)
890 emit(verb, game:cell(cell), "left", game:remaining())
891 elseif op == "CHORD" then
892 if not game.revealed[cell] then
893 fail("not revealed")
894 end
895 if game.numbers[cell] == 0 then
896 fail("no number")
897 end
898 local flags = game:flags(cell)
899 if flags ~= game.numbers[cell] then
900 fail("flags " .. flags .. " of " .. game.numbers[cell])
901 end
902 if #game:unknown_neighbors(cell) == 0 then
903 fail("nothing to reveal")
904 end
905 game:record("chord", cell)
906 local opened = game:chord(cell, self.safety)
907 if game.state ~= "lost" then
908 emit("chorded", opened)
909 game:check_win()
910 end
911 game:ending()
912 elseif op == "UNDO" then
913 if #game.undo == 0 then
914 fail("nothing to undo")
915 end
916 local record = table.remove(game.undo)
917 game:restore(record)
918 emit("undo", record.name, "moves", #game.moves)
919 elseif op == "SHOW" then
920 game:display()
921 elseif op == "STATUS" then
922 emit(
923 "status",
924 game.state,
925 "left",
926 game:remaining(),
927 "moves",
928 #game.moves,
929 "revealed",
930 game:open_count() .. "/" .. (game.size - game.mine_count)
931 )
932 elseif op == "CELL" then
933 if game.revealed[cell] then
934 emit(
935 "cell",
936 game:cell(cell),
937 "number",
938 game.numbers[cell],
939 "flags",
940 game:flags(cell),
941 "unknown",
942 #game:unknown_neighbors(cell)
943 )
944 else
945 local opened = 0
946 for _, other in ipairs(game.neighbors[cell]) do
947 if game.revealed[other] then
948 opened = opened + 1
949 end
950 end
951 emit("cell", game:cell(cell), "hidden mark", game.marks[cell], "open", opened)
952 end
953 elseif op == "HISTORY" then
954 emit("history", #game.moves)
955 for i = math.max(1, #game.moves - values[1] + 1), #game.moves do
956 local move = game.moves[i]
957 emit(i, move.kind, game:cell(move.cell))
958 end
959 elseif op == "REPLAY" then
960 if values[1] > #game.moves then
961 fail("no move " .. values[1])
962 end
963 emit("replay", values[1] .. "/" .. #game.moves)
964 game:replay(values[1], self.safety):display()
965 elseif op == "HINT" then
966 local deduction, found = game:deduce(), false
967 for _, kind in ipairs({"safe", "mine"}) do
968 local reasons = kind == "safe" and deduction.safe or deduction.mines
969 for at = 1, game.size do
970 if reasons[at] then
971 emit("hint", kind, game:cell(at), "by", reasons[at])
972 found = true
973 break
974 end
975 end
976 if found then
977 break
978 end
979 end
980 if not found then
981 emit("hint none")
982 end
983 elseif op == "AUTO" or op == "SOLVE" then
984 local total = op == "AUTO" and game:auto(self.safety) or game:solve(self.safety)
985 if op == "AUTO" then
986 emit("auto rounds", total.rounds, "flagged", total.flagged, "revealed", total.opened)
987 else
988 emit(
989 "solve guesses",
990 total.guesses,
991 "rounds",
992 total.rounds,
993 "flagged",
994 total.flagged,
995 "revealed",
996 total.opened
997 )
998 end
999 game:ending()
1000 elseif op == "ANALYZE" then
1001 local copy = game:copy()
1002 local total = copy:auto(self.safety)
1003 if copy.state == "lost" then
1004 emit("analyze boom", copy:cell(copy.exploded), "rounds", total.rounds)
1005 elseif copy.state == "won" then
1006 emit("analyze clear rounds", total.rounds)
1007 else
1008 emit(
1009 "analyze stuck rounds",
1010 total.rounds,
1011 "revealed",
1012 copy:open_count() .. "/" .. (copy.size - copy.mine_count)
1013 )
1014 end
1015 elseif op == "PROB" then
1016 if game.revealed[cell] then
1017 fail("already revealed")
1018 end
1019 if game.marks[cell] == "flag" then
1020 fail("flagged")
1021 end
1022 local component = game:frontier().at[cell]
1023 if not component then
1024 emit("prob", game:cell(cell), "unconstrained")
1025 else
1026 component:count()
1027 if component.configurations == 0 then
1028 emit("prob", game:cell(cell), "inconsistent")
1029 else
1030 emit(
1031 "prob",
1032 game:cell(cell),
1033 fraction(component:mine_count(cell), component.configurations),
1034 "size",
1035 #component.cells,
1036 "configs",
1037 component.configurations
1038 )
1039 end
1040 end
1041 elseif op == "FRONTIER" then
1042 local frontier = game:frontier()
1043 emit("frontier", count_true(frontier.at), "components", #frontier.components)
1044 for _, component in ipairs(frontier.components) do
1045 component:count()
1046 if component.configurations == 0 then
1047 emit(
1048 "component",
1049 game:cell(component.cells[1]),
1050 "size",
1051 #component.cells,
1052 "inconsistent"
1053 )
1054 else
1055 emit(
1056 "component",
1057 game:cell(component.cells[1]),
1058 "size",
1059 #component.cells,
1060 "configs",
1061 component.configurations,
1062 "min",
1063 component.minimum,
1064 "max",
1065 component.maximum
1066 )
1067 end
1068 end
1069 elseif op == "BOARDINFO" then
1070 game:board_info()
1071 end
1072 if saved and #game.moves > saved.move_count then
1073 game.undo[#game.undo + 1] = saved
1074 end
1075 if not was_finished and game:finished() then
1076 self:record_finish(game)
1077 end
1078end

1084 if not ok then
1085 if type(problem) == "table" and problem.command_error then
1086 emit("error:", problem.message)
1087 else
1088 error(problem, 0)
1089 end
1090 end
