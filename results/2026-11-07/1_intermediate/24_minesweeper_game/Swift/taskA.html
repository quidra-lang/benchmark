49struct ConfigurationSearch {
50 let incidence: [[Int]]
51 let targets: [Int]
52 let order: [Int]
53 var assigned: [Int]
54 var unassigned: [Int]
55 var values: [Int]
56 var result: Configurations
57
58 mutating func search(depth: Int, mines: Int) {
59 if depth == order.count {
60 result.count += 1
61 result.minimum = min(result.minimum, mines)
62 result.maximum = max(result.maximum, mines)
63 for index in values.indices {
64 result.frequencies[index] += values[index]
65 }
66 return
67 }
68 let cell = order[depth]
69 let affected = incidence[cell]
70 for value in 0...1 {
71 values[cell] = value
72 var possible = true
73 for index in affected {
74 assigned[index] += value
75 unassigned[index] -= 1
76 if assigned[index] > targets[index] || assigned[index] + unassigned[index] <
77 targets[index]
78 {
79 possible = false
80 }
81 }
82 if possible {
83 search(depth: depth + 1, mines: mines + value)
84 }
85 for index in affected {
86 assigned[index] -= value
87 unassigned[index] += 1
88 }
89 }
90 values[cell] = 0
91 }
92}

94func countConfigurations(_ component: Component) -> Configurations {
95 let size = component.cells.count
96 let empty = Configurations(minimum: size, frequencies: Array(repeating: 0, count: size))
97 if component.constraints.contains(where: { $0.remaining < 0 || $0.remaining > $0.cells.count }) {
98 return empty
99 }
100 let positions = Dictionary(uniqueKeysWithValues: component.cells.enumerated().map { ($0.element, $0.offset) })
101 var incidence = Array(repeating: [Int](), count: size)
102 for (index, constraint) in component.constraints.enumerated() {
103 for cell in constraint.cells {
104 incidence[positions[cell]!].append(index)
105 }
106 }
107 let order = Array(0..<size).sorted {
108 incidence[$0].count == incidence[$1].count ? $0 < $1 : incidence[$0].count > incidence[$1].count
109 }
110 var search = ConfigurationSearch(
111 incidence: incidence, targets: component.constraints.map(\.remaining), order: order,
112 assigned: Array(repeating: 0, count: component.constraints.count),
113 unassigned: component.constraints.map { $0.cells.count }, values: Array(repeating: 0, count: size),
114 result: empty
115 )
116 search.search(depth: 0, mines: 0)
117 return search.result
118}

120struct Board {
121 let width: Int
122 let height: Int
123 let mineCount: Int
124 let seed: Int64
125 let neighbors: [[Int]]
126 var mines: [Bool]?
127 var numbers: [Int]
128 var visible: [Bool]
129 var marks: [Mark]
130 var moves: [Move] = []
131 var undo: [Snapshot] = []
132 var state: GameState = .ready
133 var exploded = -1
134
135 init(width: Int, height: Int, mines: Int, seed: Int64) {
136 self.width = width
137 self.height = height
138 mineCount = mines
139 self.seed = seed
140 let size = width * height
141 numbers = Array(repeating: 0, count: size)
142 visible = Array(repeating: false, count: size)
143 marks = Array(repeating: .none, count: size)
144 neighbors = (0..<size).map { cell in
145 let x = cell % width, y = cell / width
146 var result: [Int] = []
147 for dy in -1...1 {
148 for dx in -1...1 {
149 if (dx == 0 && dy == 0) || x + dx < 0 || x + dx >= width || y + dy < 0 || y + dy >=
150 height
151 {
152 continue
153 }
154 result.append((y + dy) * width + x + dx)
155 }
156 }
157 return result
158 }
159 }

161 var size: Int {
162 width * height
163 }
164 var safeCount: Int {
165 size - mineCount
166 }
167 var revealedCount: Int {
168 visible.filter { $0 }.count
169 }
170 var remaining: Int {
171 mineCount - marks.filter { $0 == .flag }.count
172 }
173 var finished: Bool {
174 state == .won || state == .lost
175 }

184 mutating func restart() {
185 visible = Array(repeating: false, count: size)
186 marks = Array(repeating: .none, count: size)
187 moves = []
188 undo = []
189 exploded = -1
190 state = mines == nil ? .ready : .playing
191 }

193 mutating func determine(_ placement: [Bool]) {
194 mines = placement
195 numbers = neighbors.map { $0.filter { placement[$0] }.count }
196 state = .playing
197 }

199 mutating func place(first: Int, safety: String) {
200 let x = first % width, y = first / width
201 var candidates = (0..<size).filter { cell in
202 safety == "none" || (safety == "cell" && cell != first) ||
203 (safety == "block" && (abs(cell % width - x) >= 2 || abs(cell / width - y) >= 2))
204 }
205 var placement = Array(repeating: false, count: size)
206 var random = seed
207 for _ in 0..<mineCount {
208 random = 48271 * random % 2147483647
209 let index = Int(random % Int64(candidates.count))
210 placement[candidates.remove(at: index)] = true
211 }
212 determine(placement)
213 }

215 mutating func reveal(_ cell: Int, safety: String) -> Int {
216 if mines == nil {
217 place(first: cell, safety: safety)
218 }
219 if mines![cell] {
220 state = .lost
221 exploded = cell
222 return 0
223 }
224 var queue = [cell]
225 visible[cell] = true
226 marks[cell] = .none
227 var head = 0
228 while head < queue.count {
229 let current = queue[head]
230 head += 1
231 if numbers[current] != 0 {
232 continue
233 }
234 for next in neighbors[current] where unknown(next) {
235 visible[next] = true
236 marks[next] = .none
237 queue.append(next)
238 }
239 }
240 return queue.count
241 }

243 mutating func chord(_ cell: Int, safety: String) -> Int {
244 var opened = 0
245 for next in neighbors[cell] {
246 if unknown(next) {
247 opened += reveal(next, safety: safety)
248 }
249 if state == .lost {
250 break
251 }
252 }
253 return opened
254 }

256 mutating func checkWin() {
257 if state != .playing || revealedCount != safeCount {
258 return
259 }
260 state = .won
261 for cell in 0..<size where mines![cell] {
262 marks[cell] = .flag
263 }
264 }

266 func display() -> [String] {
267 var rows: [String] = []
268 for y in 0..<height {
269 var row = ""
270 for x in 0..<width {
271 let cell = y * width + x
272 var symbol = marks[cell] == .flag ? "F" : marks[cell] == .question ? "?" : "#"
273 if visible[cell] {
274 symbol = numbers[cell] == 0 ? "." : String(numbers[cell])
275 }
276 if state == .lost {
277 if mines![cell] {
278 symbol = cell == exploded ? "!" : marks[cell] == .flag ? "F" : "*"
279 } else if marks[cell] == .flag {
280 symbol = "X"
281 }
282 }
283 row += symbol
284 }
285 rows.append(row)
286 }
287 return rows
288 }

290 func finishOutput() -> [String] {
291 if state == .lost {
292 return ["boom \(coordinate(exploded))"] + display()
293 }
294 if state == .won {
295 return ["win moves \(moves.count)"]
296 }
297 return []
298 }

300 func constraints() -> [Constraint] {
301 var result: [Constraint] = []
302 for cell in 0..<size where visible[cell] && numbers[cell] != 0 {
303 let cells = neighbors[cell].filter { unknown($0) }
304 if cells.isEmpty {
305 continue
306 }
307 let flags = neighbors[cell].filter { marks[$0] == .flag }.count
308 result.append(Constraint(source: cell, cells: cells, remaining: numbers[cell] - flags))
309 }
310 return result
311 }

313 func deduce() -> Deduction {
314 let rules = constraints()
315 var result = Deduction()
316 for clue in rules where clue.remaining == 0 {
317 for cell in clue.cells
318 where result.safe[cell] == nil
319 {
320 result.safe[cell] = "single \(coordinate(clue.source))"
321 }
322 }
323 if !result.safe.isEmpty {
324 return result
325 }
326 for clue in rules where clue.remaining == clue.cells.count {
327 for cell in clue.cells
328 where result.mines[cell] == nil
329 {
330 result.mines[cell] = "single \(coordinate(clue.source))"
331 }
332 }
333 if !result.mines.isEmpty {
334 return result
335 }
336 for a in rules {
337 for b in rules {
338 if a.cells.count >= b.cells.count || !a.cells.allSatisfy({ b.cells.contains($0) }) {
339 continue
340 }
341 let difference = b.cells.filter { !a.cells.contains($0) }
342 let remaining = b.remaining - a.remaining
343 let reason = "subset \(coordinate(a.source)) \(coordinate(b.source))"
344 for cell in difference {
345 if remaining == 0 && result.safe[cell] == nil {
346 result.safe[cell] = reason
347 }
348 if remaining == difference.count && result.mines[cell] == nil {
349 result.mines[cell] = reason
350 }
351 }
352 }
353 }
354 if result.safe.isEmpty && result.mines.isEmpty && state == .playing && remaining == 0 {
355 for cell in 0..<size where unknown(cell) {
356 result.safe[cell] = "count"
357 }
358 }
359 return result
360 }

362 func frontier() -> [Component] {
363 let rules = constraints()
364 var parent = Array(0..<size)
365 var present = Array(repeating: false, count: size)
366 func root(_ original: Int) -> Int {
367 var cell = original
368 while parent[cell] != cell {
369 parent[cell] = parent[parent[cell]]
370 cell = parent[cell]
371 }
372 return cell
373 }
374 for clue in rules {
375 for cell in clue.cells {
376 present[cell] = true
377 let first = root(clue.cells[0])
378 let second = root(cell)
379 parent[second] = first
380 }
381 }
382 var groups: [Int: Int] = [:]
383 var result: [Component] = []
384 for cell in 0..<size where present[cell] {
385 let representative = root(cell)
386 if groups[representative] == nil {
387 groups[representative] = result.count
388 result.append(Component())
389 }
390 result[groups[representative]!].cells.append(cell)
391 }
392 for clue in rules {
393 result[groups[root(clue.cells[0])]!].constraints.append(clue)
394 }
395 return result
396 }

398 mutating func automatic(safety: String) -> Totals {
399 var totals = Totals()
400 while state == .playing {
401 let deduction = deduce()
402 if deduction.safe.isEmpty && deduction.mines.isEmpty {
403 break
404 }
405 totals.rounds += 1
406 for cell in deduction.safe.keys.sorted() where !visible[cell] {
407 moves.append(Move(kind: .autoReveal, cell: cell))
408 totals.opened += reveal(cell, safety: safety)
409 checkWin()
410 if finished {
411 return totals
412 }
413 }
414 for cell in deduction.mines.keys.sorted() where unknown(cell) {
415 marks[cell] = .flag
416 totals.flagged += 1
417 moves.append(Move(kind: .autoFlag, cell: cell))
418 }
419 }
420 return totals
421 }

423 mutating func solve(safety: String) -> Totals {
424 var totals = Totals()
425 while !finished {
426 let step = automatic(safety: safety)
427 totals.rounds += step.rounds
428 totals.flagged += step.flagged
429 totals.opened += step.opened
430 if finished {
431 break
432 }
433 var onFrontier = Array(repeating: false, count: size)
434 var numerators = Array(repeating: 0, count: size)
435 var denominators = Array(repeating: 0, count: size)
436 for component in frontier() {
437 let configurations = countConfigurations(component)
438 for (index, cell) in component.cells.enumerated() {
439 onFrontier[cell] = true
440 numerators[cell] = configurations.frequencies[index]
441 denominators[cell] = configurations.count
442 }
443 }
444 let unknownCount = (0..<size).filter { unknown($0) }.count
445 let unconstrainedNumerator = max(remaining, 0)
446 var chosen: Int?
447 var bestNumerator = 0, bestDenominator = 1
448 for cell in 0..<size where unknown(cell) {
449 let numerator = onFrontier[cell] ? numerators[cell] : unconstrainedNumerator
450 let denominator = onFrontier[cell] ? denominators[cell] : unknownCount
451 if denominator == 0 {
452 continue
453 }
454 if chosen == nil || numerator * bestDenominator < bestNumerator * denominator {
455 chosen = cell
456 bestNumerator = numerator
457 bestDenominator = denominator
458 }
459 }
460 guard let cell = chosen else {
461 break
462 }
463 totals.guesses += 1
464 moves.append(Move(kind: .guess, cell: cell))
465 totals.opened += reveal(cell, safety: safety)
466 checkWin()
467 }
468 return totals
469 }

471 func replay(_ count: Int, safety: String) -> [String] {
472 var copy = self
473 copy.restart()
474 for move in moves.prefix(count) {
475 switch move.kind {
476 case .reveal, .autoReveal, .guess:
477 _ = copy.reveal(move.cell, safety: safety)
478 case .chord:
479 _ = copy.chord(move.cell, safety: safety)
480 case .flag, .autoFlag:
481 copy.marks[move.cell] = .flag
482 case .unflag:
483 copy.marks[move.cell] = .none
484 case .question:
485 copy.marks[move.cell] = copy.marks[move.cell] == .question ? .none : .question
486 }
487 copy.checkWin()
488 }
489 return copy.display()
490 }

492 func boardInfo() -> String {
493 let placement = mines!
494 var visited = Array(repeating: false, count: size)
495 var openings = 0, largest = 0, isolated = 0
496 for cell in 0..<size where !placement[cell] {
497 if numbers[cell] != 0 {
498 if !neighbors[cell].contains(where: { !placement[$0] && numbers[$0] == 0 }) {
499 isolated += 1
500 }
501 continue
502 }
503 if visited[cell] {
504 continue
505 }
506 openings += 1
507 var queue = [cell]
508 var area: Set<Int> = [cell]
509 var head = 0
510 visited[cell] = true
511 while head < queue.count {
512 let current = queue[head]
513 head += 1
514 for next in neighbors[current] {
515 area.insert(next)
516 if !placement[next] && numbers[next] == 0 && !visited[next] {
517 visited[next] = true
518 queue.append(next)
519 }
520 }
521 }
522 largest = max(largest, area.count)
523 }
524 return "3bv \(openings + isolated) openings \(openings) largest \(largest) mines \(mineCount) safe \(safeCount)"
525 }
526}

562func numberWord(_ text: String) throws -> Int64 {
563 let bytes = Array(text.utf8)
564 if bytes.isEmpty || bytes.count > 10 || (bytes.count > 1 && bytes[0] == 48) || !bytes
565 .allSatisfy({ 48...57 ~= $0 })
566 {
567 try fail("bad arguments")
568 }
569 return Int64(text)!
570}

572func fraction(_ numerator: Int, _ denominator: Int) -> String {
573 var a = numerator, b = denominator
574 while b != 0 {
575 (a, b) = (b, a % b)
576 }
577 return "\(numerator / a)/\(denominator / a)"
578}

580struct Statistics {
581 var played = 0, won = 0, lost = 0, streak = 0, best = 0
582 var fastest: Int?
583
584 mutating func finish(_ board: Board) {
585 played += 1
586 if board.state == .won {
587 won += 1
588 streak += 1
589 best = max(best, streak)
590 fastest = min(fastest ?? board.moves.count, board.moves.count)
591 } else {
592 lost += 1
593 streak = 0
594 }
595 }
596
597 func display() -> String {
598 let rate = played == 0 ? "-" : "\(won * 100 / played)%"
599 let minimum = fastest.map { String($0) } ?? "-"
600 return "stats played \(played) won \(won) lost \(lost) rate \(rate) streak \(streak) best \(best) fastest \(minimum)"
601 }
602}

604struct Program {
605 var game: Board?
606 var safety = "block"
607 var stats = Statistics()

609 mutating func execute(_ words: [String], rows: [String]) throws -> [String] {
610 guard let command = Command(rawValue: words[0]) else {
611 try fail("unknown command \(words[0])")
612 }
613 if command == .layout {
614 let width = Int(words[1])!, height = Int(words[2])!
615 for (index, row) in rows.enumerated() {
616 if row.utf8.count != width || !row.utf8
617 .allSatisfy({ $0 == 42 || $0 == 46 })
618 {
619 try fail("bad row \(index + 1)")
620 }
621 }
622 let mines = rows.joined().utf8.map { $0 == 42 }
623 let count = mines.filter { $0 }.count
624 if count == 0 || count == width * height {
625 try fail("bad mine count \(count)")
626 }
627 var board = Board(width: width, height: height, mines: count, seed: 1)
628 board.determine(mines)
629 game = board
630 return ["layout \(width)x\(height) mines \(count)"]
631 }
632 let format = Array(formats[command]!)
633 if words.count - 1 != format.count {
634 try fail("bad arguments")
635 }
636 var values = Array(repeating: Int64(0), count: format.count)
637 for (index, kind) in format.enumerated() {
638 let word = words[index + 1]
639 switch kind {
640 case "n":
641 values[index] = try numberWord(word)
642 case "p":
643 if presets[word] == nil {
644 try fail("bad arguments")
645 }
646 default:
647 if !["block", "cell", "none"].contains(word) {
648 try fail("bad arguments")
649 }
650 }
651 }
652 if command == .new || command == .preset {
653 let width: Int64, height: Int64, count: Int64, seed: Int64
654 if command == .new {
655 width = values[0]
656 height = values[1]
657 count = values[2]
658 seed = values[3]
659 } else {
660 let preset = presets[words[1]]!
661 width = Int64(preset.width)
662 height = Int64(preset.height)
663 count = Int64(preset.mines)
664 seed = values[1]
665 }
666 if width < 1 || width > 30 || height < 1 || height > 30 || count < 1 || count > width * height - 9 || seed <
667 1 || seed > 2147483646
668 {
669 try fail("bad value")
670 }
671 game = Board(width: Int(width), height: Int(height), mines: Int(count), seed: seed)
672 return ["new \(width)x\(height) mines \(count)"]
673 }
674 if command == .history && values[0] == 0 {
675 try fail("bad value")
676 }
677 if command == .safety {
678 safety = words[1]
679 return ["safety \(safety)"]
680 }
681 if command == .stats {
682 return [stats.display()]
683 }
684 guard var board = game else {
685 try fail("no game")
686 }
687 var cell = -1
688 if cellCommands.contains(command) {
689 if values[0] >= Int64(board.width) || values[1] >= Int64(board.height) {
690 try fail("out of range")
691 }
692 cell = Int(values[1]) * board.width + Int(values[0])
693 }
694 if liveCommands.contains(command) && board.finished {
695 try fail("game over")
696 }
697 if (command == .analyze || command == .boardInfo) && board.state == .ready {
698 try fail("not started")
699 }
700 let snapshot = undoCommands.contains(command) ? Snapshot(
701 command: command.rawValue.lowercased(), visible: board.visible, marks: board.marks,
702 moveCount: board.moves.count
703 ) : nil
704 let output = try boardCommand(command, values: values, cell: cell, board: &board)
705 if let snapshot, board.moves.count > snapshot.moveCount {
706 board.undo.append(snapshot)
707 }
708 if [Command.reveal, .chord, .auto, .solve].contains(command) && board.finished {
709 stats.finish(board)
710 }
711 game = board
712 return output
713 }

715 func boardCommand(_ command: Command, values: [Int64], cell: Int, board: inout Board) throws -> [String] {
716 switch command {
717 case .restart:
718 board.restart()
719 return ["restart \(board.width)x\(board.height) mines \(board.mineCount)"]
720 case .reveal, .flag, .unflag, .question:
721 if board.visible[cell] {
722 try fail("already revealed")
723 }
724 let mark = board.marks[cell]
725 if command == .reveal && mark == .flag {
726 try fail("flagged")
727 }
728 if command == .flag && mark == .flag {
729 try fail("already flagged")
730 }
731 if command == .unflag && mark != .flag {
732 try fail("not flagged")
733 }
734 board.moves.append(Move(kind: MoveKind(rawValue: command.rawValue.lowercased())!, cell: cell))
735 if command == .reveal {
736 let opened = board.reveal(cell, safety: safety)
737 if board.state == .lost {
738 return board.finishOutput()
739 }
740 board.checkWin()
741 return ["revealed \(opened)"] + board.finishOutput()
742 }
743 board.marks[cell] = command == .flag ? .flag : command == .unflag || mark == .question ? .none : .question
744 let action = command == .flag ? "flagged" : command == .unflag ? "unflagged" : mark == .question ?
745 "unmarked" : "questioned"
746 return ["\(action) \(board.coordinate(cell)) left \(board.remaining)"]
747 case .chord:
748 if !board.visible[cell] {
749 try fail("not revealed")
750 }
751 let number = board.numbers[cell]
752 if number == 0 {
753 try fail("no number")
754 }
755 let flags = board.neighbors[cell].filter { board.marks[$0] == .flag }.count
756 if flags != number {
757 try fail("flags \(flags) of \(number)")
758 }
759 if !board.neighbors[cell].contains(where: { board.unknown($0) }) {
760 try fail("nothing to reveal")
761 }
762 board.moves.append(Move(kind: .chord, cell: cell))
763 let opened = board.chord(cell, safety: safety)
764 if board.state == .lost {
765 return board.finishOutput()
766 }
767 board.checkWin()
768 return ["chorded \(opened)"] + board.finishOutput()
769 case .undo:
770 guard let previous = board.undo.popLast() else {
771 try fail("nothing to undo")
772 }
773 board.visible = previous.visible
774 board.marks = previous.marks
775 board.moves.removeSubrange(previous.moveCount...)
776 return ["undo \(previous.command) moves \(board.moves.count)"]
777 case .show:
778 return board.display()
779 case .status:
780 return [
781 "status \(board.state.rawValue) left \(board.remaining) moves \(board.moves.count) revealed \(board.revealedCount)/\(board.safeCount)"
782 ]
783 case .cell:
784 let neighbors = board.neighbors[cell]
785 if !board.visible[cell] {
786 let opened = neighbors.filter { board.visible[$0] }.count
787 return ["cell \(board.coordinate(cell)) hidden mark \(board.marks[cell].rawValue) open \(opened)"]
788 }
789 let flags = neighbors.filter { board.marks[$0] == .flag }.count
790 let unknown = neighbors.filter { board.unknown($0) }.count
791 return ["cell \(board.coordinate(cell)) number \(board.numbers[cell]) flags \(flags) unknown \(unknown)"]
792 case .history:
793 var output = ["history \(board.moves.count)"]
794 let count = Int(min(values[0], Int64(board.moves.count)))
795 for index in (board.moves.count - count)..<board.moves.count {
796 let move = board.moves[index]
797 output.append("\(index + 1) \(move.kind.rawValue) \(board.coordinate(move.cell))")
798 }
799 return output
800 case .replay:
801 if values[0] > Int64(board.moves.count) {
802 try fail("no move \(values[0])")
803 }
804 return ["replay \(values[0])/\(board.moves.count)"] + board.replay(Int(values[0]), safety: safety)
805 case .hint:
806 let deduction = board.deduce()
807 let choices = deduction.safe.isEmpty ? deduction.mines : deduction.safe
808 guard let chosen = choices.keys.min() else {
809 return ["hint none"]
810 }
811 let kind = deduction.safe.isEmpty ? "mine" : "safe"
812 return ["hint \(kind) \(board.coordinate(chosen)) by \(choices[chosen]!)"]
813 case .auto, .solve:
814 let result = command == .auto ? board.automatic(safety: safety) : board.solve(safety: safety)
815 let prefix = command == .auto ? "auto" : "solve guesses \(result.guesses)"
816 return ["\(prefix) rounds \(result.rounds) flagged \(result.flagged) revealed \(result.opened)"] + board
817 .finishOutput()
818 case .analyze:
819 var copy = board
820 copy.undo = []
821 let result = copy.automatic(safety: safety)
822 if copy.state == .lost {
823 return ["analyze boom \(copy.coordinate(copy.exploded)) rounds \(result.rounds)"]
824 }
825 if copy.state == .won {
826 return ["analyze clear rounds \(result.rounds)"]
827 }
828 return ["analyze stuck rounds \(result.rounds) revealed \(copy.revealedCount)/\(copy.safeCount)"]
829 case .prob:
830 if board.visible[cell] {
831 try fail("already revealed")
832 }
833 if board.marks[cell] == .flag {
834 try fail("flagged")
835 }
836 guard let component = board.frontier().first(where: { $0.cells.contains(cell) }) else {
837 return ["prob \(board.coordinate(cell)) unconstrained"]
838 }
839 let configurations = countConfigurations(component)
840 if configurations.count == 0 {
841 return ["prob \(board.coordinate(cell)) inconsistent"]
842 }
843 let numerator = configurations.frequencies[component.cells.firstIndex(of: cell)!]
844 return [
845 "prob \(board.coordinate(cell)) \(fraction(numerator, configurations.count)) size \(component.cells.count) configs \(configurations.count)"
846 ]
847 case .frontier:
848 let components = board.frontier()
849 let count = components.reduce(0) { $0 + $1.cells.count }
850 var output = ["frontier \(count) components \(components.count)"]
851 for component in components {
852 let configurations = countConfigurations(component)
853 let prefix = "component \(board.coordinate(component.cells[0])) size \(component.cells.count)"
854 output.append(configurations.count == 0 ? "\(prefix) inconsistent" :
855 "\(prefix) configs \(configurations.count) min \(configurations.minimum) max \(configurations.maximum)")
856 }
857 return output
858 case .boardInfo:
859 return [board.boardInfo()]
860 default:
861 preconditionFailure("Global command reached board dispatch")
862 }
863 }
864}
