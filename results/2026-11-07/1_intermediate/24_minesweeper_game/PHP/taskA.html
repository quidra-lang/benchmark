112final class Component
113{
114 public int $configurations = 0,
115 $maximum = 0,
116 $minimum;
117 public array $mineCounts;
118 private bool $counted = false;
119 private array $index = [],
120 $links = [],
121 $order = [],
122 $remaining = [],
123 $sums = [],
124 $assignment = [];
125
126 public function __construct(public readonly array $cells, private readonly array $constraints)
127 {
128 $this->minimum = count($cells);
129 $this->mineCounts = array_fill(0, count($cells), 0);
130 }
131 public function count(): void
132 {
133 if ($this->counted) {
134 return;
135 }
136 $this->counted = true;
137 foreach ($this->constraints as $constraint) {
138 if (
139 $constraint->remaining < 0 ||
140 $constraint->remaining > count($constraint->unknown)
141 ) {
142 return;
143 }
144 }
145 $this->index = array_flip($this->cells);
146 $this->links = array_fill(0, count($this->cells), []);
147 foreach ($this->constraints as $c => $constraint) {
148 foreach ($constraint->unknown as $cell) {
149 $this->links[$this->index[$cell]][] = $c;
150 };
151 }
152 $this->order = array_keys($this->cells);
153 usort(
154 $this->order,
155 fn(int $a, int $b): int => count($this->links[$b]) <=> count($this->links[$a]) ?:
156 $a <=> $b,
157 );
158 $this->remaining = array_map(
159 fn(Constraint $c): int => count($c->unknown),
160 $this->constraints,
161 );
162 $this->sums = array_fill(0, count($this->constraints), 0);
163 $this->assignment = array_fill(0, count($this->cells), 0);
164 $this->search(0, 0);
165 }
166 private function search(int $depth, int $mines): void
167 {
168 if ($depth === count($this->cells)) {
169 $this->configurations++;
170 $this->minimum = min($this->minimum, $mines);
171 $this->maximum = max($this->maximum, $mines);
172 foreach ($this->assignment as $i => $value) {
173 $this->mineCounts[$i] += $value;
174 }
175 return;
176 }
177 $variable = $this->order[$depth];
178 for ($value = 0; $value <= 1; $value++) {
179 $this->assignment[$variable] = $value;
180 $possible = true;
181 foreach ($this->links[$variable] as $c) {
182 $this->remaining[$c]--;
183 $this->sums[$c] += $value;
184 $required = $this->constraints[$c]->remaining;
185 if (
186 $this->sums[$c] > $required ||
187 $this->sums[$c] + $this->remaining[$c] < $required
188 ) {
189 $possible = false;
190 }
191 }
192 if ($possible) {
193 $this->search($depth + 1, $mines + $value);
194 }
195 foreach ($this->links[$variable] as $c) {
196 $this->sums[$c] -= $value;
197 $this->remaining[$c]++;
198 }
199 }
200 }
201 public function mineCount(int $cell): int
202 {
203 return $this->mineCounts[$this->index[$cell]];
204 }
205}

314 private function place(int $first, Safety $safety): void
315 {
316 $sx = $first % $this->width;
317 $sy = intdiv($first, $this->width);
318 $candidates = [];
319 for ($cell = 0; $cell < $this->size; $cell++) {
320 if (
321 $safety === Safety::None ||
322 ($safety === Safety::Cell && $cell !== $first) ||
323 ($safety === Safety::Block &&
324 (abs(($cell % $this->width) - $sx) >= 2 ||
325 abs(intdiv($cell, $this->width) - $sy) >= 2))

327 $candidates[] = $cell;
328 }
329 }
330 $mines = array_fill(0, $this->size, false);
331 $random = $this->seed;
332 for ($i = 0; $i < $this->mineCount; $i++) {
333 $random = (48271 * $random) % 2147483647;
334 $index = $random % count($candidates);
335 $mines[$candidates[$index]] = true;
336 array_splice($candidates, $index, 1);

471 public function deduce(): Deduction
472 {
473 $result = new Deduction($this->size);
474 $constraints = $this->constraints();
475 foreach ($constraints as $c) {
476 if ($c->remaining === 0) {
477 foreach ($c->unknown as $cell) {
478 $result->safe[$cell] ??= "single " . $this->cell($c->cell);
479 };
480 }
481 }
482 if ($result->any()) {
483 return $result;
484 }
485 foreach ($constraints as $c) {
486 if ($c->remaining === count($c->unknown)) {
487 foreach ($c->unknown as $cell) {
488 $result->mines[$cell] ??= "single " . $this->cell($c->cell);
489 };
490 }
491 }
492 if ($result->any()) {
493 return $result;
494 }
495 foreach ($constraints as $a) {
496 foreach ($constraints as $b) {
497 if (
498 count($a->unknown) >= count($b->unknown) ||
499 array_diff($a->unknown, $b->unknown) !== []
500 ) {
501 continue;
502 }
503 $difference = array_diff($b->unknown, $a->unknown);
504 $remaining = $b->remaining - $a->remaining;
505 $reason = "subset " . $this->cell($a->cell) . " " . $this->cell($b->cell);
506 if ($remaining === 0) {
507 foreach ($difference as $cell) {
508 $result->safe[$cell] ??= $reason;
509 };
510 }
511 if ($remaining === count($difference)) {
512 foreach ($difference as $cell) {
513 $result->mines[$cell] ??= $reason;
514 };
515 }
516 };
517 }
518 if ($result->any()) {
519 return $result;
520 }
521 if ($this->state === GameState::Playing && $this->remaining() === 0) {
522 for ($cell = 0; $cell < $this->size; $cell++) {
523 if ($this->unknown($cell)) {
524 $result->safe[$cell] = "count";
525 }
526 };
527 }
528 return $result;
529 }
530 public function frontier(): Frontier
531 {
532 $constraints = $this->constraints();
533 $links = array_fill(0, $this->size, []);
534 foreach ($constraints as $i => $c) {
535 foreach ($c->unknown as $cell) {
536 $links[$cell][] = $i;
537 };
538 }
539 $seen = array_fill(0, $this->size, false);
540 $at = array_fill(0, $this->size, null);
541 $components = [];
542 for ($start = 0; $start < $this->size; $start++) {
543 if ($seen[$start] || $links[$start] === []) {
544 continue;
545 }
546 $cells = [$start];
547 $selected = [];
548 $picked = [];
549 $seen[$start] = true;
550 for ($head = 0; $head < count($cells); $head++) {
551 foreach ($links[$cells[$head]] as $c) {
552 if (isset($picked[$c])) {
553 continue;
554 }
555 $picked[$c] = true;
556 $selected[] = $constraints[$c];
557 foreach ($constraints[$c]->unknown as $cell) {
558 if (!$seen[$cell]) {
559 $seen[$cell] = true;
560 $cells[] = $cell;
561 }
562 }
563 };
564 }
565 sort($cells, SORT_NUMERIC);
566 $component = new Component($cells, $selected);
567 $components[] = $component;
568 foreach ($cells as $cell) {
569 $at[$cell] = $component;
570 }
571 }
572 return new Frontier($components, $at);
573 }
574 public function auto(Safety $safety): Totals
575 {
576 $total = new Totals();
577 while ($this->state === GameState::Playing) {
578 $deduction = $this->deduce();
579 if (!$deduction->any()) {
580 break;
581 }
582 $total->rounds++;
583 foreach ($deduction->safe as $cell => $reason) {
584 if ($reason !== null && !$this->revealed[$cell]) {
585 $this->moves[] = new Move(MoveKind::AutoReveal, $cell);
586 $total->opened += $this->reveal($cell, $safety);
587 $this->checkWin();
588 if ($this->finished()) {
589 return $total;
590 }
591 }
592 }
593 foreach ($deduction->mines as $cell => $reason) {
594 if ($reason !== null && $this->unknown($cell)) {
595 $this->marks[$cell] = Mark::Flag;
596 $total->flagged++;
597 $this->moves[] = new Move(MoveKind::AutoFlag, $cell);
598 }
599 }
600 }
601 return $total;
602 }
603 public function solve(Safety $safety): Totals
604 {
605 $total = new Totals();
606 while (!$this->finished()) {
607 $total->add($this->auto($safety));
608 if ($this->finished()) {
609 break;
610 }
611 $frontier = $this->frontier();
612 foreach ($frontier->components as $component) {
613 $component->count();
614 }
615 $unknownCount = 0;
616 for ($cell = 0; $cell < $this->size; $cell++) {
617 if ($this->unknown($cell)) {
618 $unknownCount++;
619 }
620 }
621 $background = max($this->remaining(), 0);
622 $best = null;
623 $bestNumerator = 0;
624 $bestDenominator = 1;
625 for ($cell = 0; $cell < $this->size; $cell++) {
626 if (!$this->unknown($cell)) {
627 continue;
628 }
629 $component = $frontier->at[$cell];
630 if ($component !== null) {
631 if ($component->configurations === 0) {
632 continue;
633 }
634 $numerator = $component->mineCount($cell);
635 $denominator = $component->configurations;
636 } else {
637 $numerator = $background;
638 $denominator = $unknownCount;
639 }
640 if (
641 $best === null ||
642 $numerator * $bestDenominator < $bestNumerator * $denominator
643 ) {
644 $best = $cell;
645 $bestNumerator = $numerator;
646 $bestDenominator = $denominator;
647 }
648 }
649 if ($best === null) {
650 break;
651 }
652 $total->guesses++;
653 $this->moves[] = new Move(MoveKind::Guess, $best);
654 $total->opened += $this->reveal($best, $safety);
655 $this->checkWin();
656 }
657 return $total;
658 }
659 public function replay(int $count, Safety $safety): self
660 {
661 $copy = $this->copy(true);
662 for ($i = 0; $i < $count; $i++) {
663 $move = $this->moves[$i];
664 $cell = $move->cell;
665 switch ($move->kind) {
666 case MoveKind::Reveal:
667 case MoveKind::AutoReveal:
668 case MoveKind::Guess:
669 $copy->reveal($cell, $safety);
670 break;
671 case MoveKind::Chord:
672 $copy->chord($cell, $safety);
673 break;
674 case MoveKind::Flag:
675 case MoveKind::AutoFlag:
676 $copy->marks[$cell] = Mark::Flag;
677 break;
678 case MoveKind::Unflag:
679 $copy->marks[$cell] = Mark::None;
680 break;
681 case MoveKind::Question:
682 $copy->marks[$cell] =
683 $copy->marks[$cell] === Mark::Question ? Mark::None : Mark::Question;
684 break;
685 }
686 $copy->checkWin();
687 }
688 return $copy;
689 }
690 public function boardInfo(): void
691 {
692 $blank = [];
693 for ($cell = 0; $cell < $this->size; $cell++) {
694 $blank[] = !$this->mines[$cell] && $this->numbers[$cell] === 0;
695 }
696 $seen = array_fill(0, $this->size, false);
697 $openings = 0;
698 $largest = 0;
699 $isolated = 0;
700 for ($start = 0; $start < $this->size; $start++) {
701 if ($blank[$start] && !$seen[$start]) {
702 $openings++;
703 $queue = [$start];
704 $seen[$start] = true;
705 $area = [$start => true];
706 for ($head = 0; $head < count($queue); $head++) {
707 foreach ($this->neighbors[$queue[$head]] as $cell) {
708 $area[$cell] = true;
709 if ($blank[$cell] && !$seen[$cell]) {
710 $seen[$cell] = true;
711 $queue[] = $cell;
712 }
713 };
714 }
715 $largest = max($largest, count($area));
716 }
717 if (
718 !$this->mines[$start] &&
719 !$blank[$start] &&
720 !array_filter($this->neighbors[$start], fn(int $cell): bool => $blank[$cell])
721 ) {
722 $isolated++;
723 }
724 }
725 emit(
726 "3bv",
727 $openings + $isolated,
728 "openings",
729 $openings,
730 "largest",
731 $largest,
732 "mines",
733 $this->mineCount,
734 "safe",
735 $this->size - $this->mineCount,
736 );
737 }

812 $this->played,
813 "won",
814 $this->won,
815 "lost",
816 $this->lost,
817 "rate",
818 $rate,
819 "streak",
820 $this->streak,
821 "best",
822 $this->bestStreak,
823 "fastest",
824 $this->fastest ?? "-",
825 );
826 }
827 private function recordFinish(Game $game): void
828 {
829 $this->played++;
830 if ($game->state === GameState::Won) {
831 $this->won++;
832 $this->streak++;
833 $this->bestStreak = max($this->bestStreak, $this->streak);
834 $this->fastest = min($this->fastest ?? count($game->moves), count($game->moves));
835 } else {
836 $this->lost++;
837 $this->streak = 0;
838 }
839 }
840 public function execute(array $words): void
841 {
842 $op = $words[0];
843 if ($op === "LAYOUT") {
844 $this->layout($words);
845 return;
846 }
847 if (!isset(self::ARITY[$op])) {
848 fail("unknown command $op");
849 }
850 if (count($words) !== self::ARITY[$op] + 1) {
851 fail("bad arguments");
852 }
853 $hasCell = in_array(
854 $op,
855 ["REVEAL", "FLAG", "UNFLAG", "QUESTION", "CHORD", "CELL", "PROB"],
856 true,
857 );
858 $values = [];
859 if ($op === "NEW" || $hasCell || $op === "HISTORY" || $op === "REPLAY") {
860 $values = array_map($this->number(...), array_slice($words, 1));
861 } elseif ($op === "PRESET") {
862 if (!in_array($words[1], ["beginner", "intermediate", "expert"], true)) {
863 fail("bad arguments");
864 }
865 $values = [$this->number($words[2])];
866 } elseif ($op === "SAFETY" && Safety::tryFrom($words[1]) === null) {
867 fail("bad arguments");
868 }
869
870 if ($op === "NEW" || $op === "PRESET") {
871 if ($op === "NEW") {
872 [$width, $height, $mines, $seed] = $values;
873 } else {
874 [$width, $height, $mines] = match ($words[1]) {
875 "beginner" => [9, 9, 10],
876 "intermediate" => [16, 16, 40],
877 "expert" => [30, 16, 99],
878 };
879 $seed = $values[0];
880 }
881 if (
882 $width < 1 ||
883 $width > 30 ||
884 $height < 1 ||
885 $height > 30 ||
886 $mines < 1 ||
887 $mines > $width * $height - 9 ||
888 $seed < 1 ||
889 $seed > 2147483646
890 ) {
891 fail("bad value");
892 }
893 $this->game = new Game($width, $height, $mines, $seed);
894 emit("new", "{$width}x{$height}", "mines", $mines);
895 return;
896 }
897 if ($op === "HISTORY" && $values[0] === 0) {
898 fail("bad value");
899 }
900 if ($op === "SAFETY") {
901 $this->safety = Safety::from($words[1]);
902 emit("safety", $words[1]);
903 return;
904 }
905 if ($op === "STATS") {
906 $this->stats();
907 return;
908 }
909 $game = $this->game ?? fail("no game");
910 if ($hasCell && ($values[0] >= $game->width || $values[1] >= $game->height)) {
911 fail("out of range");
912 }
913 $cell = $hasCell ? $values[1] * $game->width + $values[0] : 0;
914 if (
915 $game->finished() &&
916 in_array(
917 $op,
918 [
919 "REVEAL",
920 "FLAG",
921 "UNFLAG",
922 "QUESTION",
923 "CHORD",
924 "UNDO",
925 "HINT",
926 "AUTO",
927 "ANALYZE",
928 "PROB",
929 "FRONTIER",
930 "SOLVE",
931 ],
932 true,
933 )
934 ) {
935 fail("game over");
936 }
937 if (($op === "ANALYZE" || $op === "BOARDINFO") && $game->mines === null) {
938 fail("not started");
939 }
940 $wasFinished = $game->finished();
941 $saved = in_array(
942 $op,
943 ["REVEAL", "FLAG", "UNFLAG", "QUESTION", "CHORD", "AUTO", "SOLVE"],
944 true,
945 )
946 ? $game->snapshot(strtolower($op))
947 : null;
948
949 switch ($op) {
950 case "RESTART":
951 $game->restart();
952 emit("restart", "{$game->width}x{$game->height}", "mines", $game->mineCount);
953 break;
954 case "REVEAL":
955 if ($game->revealed[$cell]) {
956 fail("already revealed");
957 }
958 if ($game->marks[$cell] === Mark::Flag) {
959 fail("flagged");
960 }
961 $game->moves[] = new Move(MoveKind::Reveal, $cell);
962 $opened = $game->reveal($cell, $this->safety);
963 if ($game->state !== GameState::Lost) {
964 emit("revealed", $opened);
965 $game->checkWin();
966 }
967 $game->ending();
968 break;
969 case "FLAG":
970 case "UNFLAG":
971 case "QUESTION":
972 if ($game->revealed[$cell]) {
973 fail("already revealed");
974 }
975 if ($op === "FLAG" && $game->marks[$cell] === Mark::Flag) {
976 fail("already flagged");
977 }
978 if ($op === "UNFLAG" && $game->marks[$cell] !== Mark::Flag) {
979 fail("not flagged");
980 }
981 if ($op === "FLAG") {
982 $mark = Mark::Flag;
983 $kind = MoveKind::Flag;
984 $verb = "flagged";
985 } elseif ($op === "UNFLAG") {
986 $mark = Mark::None;
987 $kind = MoveKind::Unflag;
988 $verb = "unflagged";
989 } else {
990 $mark = $game->marks[$cell] === Mark::Question ? Mark::None : Mark::Question;
991 $kind = MoveKind::Question;
992 $verb = $mark === Mark::None ? "unmarked" : "questioned";
993 }
994 $game->marks[$cell] = $mark;
995 $game->moves[] = new Move($kind, $cell);
996 emit($verb, $game->cell($cell), "left", $game->remaining());
997 break;
998 case "CHORD":
999 if (!$game->revealed[$cell]) {
1000 fail("not revealed");
1001 }
1002 if ($game->numbers[$cell] === 0) {
1003 fail("no number");
1004 }
1005 $flags = $game->flags($cell);
1006 if ($flags !== $game->numbers[$cell]) {
1007 fail("flags $flags of {$game->numbers[$cell]}");
1008 }
1009 if ($game->unknownNeighbors($cell) === []) {
1010 fail("nothing to reveal");
1011 }
1012 $game->moves[] = new Move(MoveKind::Chord, $cell);
1013 $opened = $game->chord($cell, $this->safety);
1014 if ($game->state !== GameState::Lost) {
1015 emit("chorded", $opened);
1016 $game->checkWin();
1017 }
1018 $game->ending();
1019 break;
1020 case "UNDO":
1021 if ($game->undo === []) {
1022 fail("nothing to undo");
1023 }
1024 $record = array_pop($game->undo);
1025 $game->restore($record);
1026 emit("undo", $record->name, "moves", count($game->moves));
1027 break;
1028 case "SHOW":
1029 $game->display();
1030 break;
1031 case "STATUS":
1032 emit(
1033 "status",
1034 $game->state->value,
1035 "left",
1036 $game->remaining(),
1037 "moves",
1038 count($game->moves),
1039 "revealed",
1040 $game->openCount() . "/" . ($game->size - $game->mineCount),
1041 );
1042 break;
1043 case "CELL":
1044 if ($game->revealed[$cell]) {
1045 emit(
1046 "cell",
1047 $game->cell($cell),
1048 "number",
1049 $game->numbers[$cell],
1050 "flags",
1051 $game->flags($cell),
1052 "unknown",
1053 count($game->unknownNeighbors($cell)),
1054 );
1055 } else {
1056 emit(
1057 "cell",
1058 $game->cell($cell),
1059 "hidden mark",
1060 $game->marks[$cell]->value,
1061 "open",
1062 count(
1063 array_filter(
1064 $game->neighbors[$cell],
1065 fn(int $other): bool => $game->revealed[$other],
1066 ),
1067 ),
1068 );
1069 }
1070 break;
1071 case "HISTORY":
1072 $count = count($game->moves);
1073 emit("history", $count);
1074 for ($i = max(0, $count - $values[0]); $i < $count; $i++) {
1075 $move = $game->moves[$i];
1076 emit($i + 1, $move->kind->value, $game->cell($move->cell));
1077 }
1078 break;
1079 case "REPLAY":
1080 if ($values[0] > count($game->moves)) {
1081 fail("no move {$values[0]}");
1082 }
1083 emit("replay", $values[0] . "/" . count($game->moves));
1084 $game->replay($values[0], $this->safety)->display();
1085 break;
1086 case "HINT":
1087 $deduction = $game->deduce();
1088 $found = false;
1089 foreach (
1090 ["safe" => $deduction->safe, "mine" => $deduction->mines]
1091 as $kind => $reasons
1092 ) {
1093 foreach ($reasons as $at => $reason) {
1094 if ($reason !== null) {
1095 emit("hint", $kind, $game->cell($at), "by", $reason);
1096 $found = true;
1097 break 2;
1098 }
1099 }
1100 }
1101 if (!$found) {
1102 emit("hint none");
1103 }
1104 break;
1105 case "AUTO":
1106 case "SOLVE":
1107 $total = $op === "AUTO" ? $game->auto($this->safety) : $game->solve($this->safety);
1108 if ($op === "AUTO") {
1109 emit(
1110 "auto rounds",
1111 $total->rounds,
1112 "flagged",
1113 $total->flagged,
1114 "revealed",
1115 $total->opened,
1116 );
1117 } else {
1118 emit(
1119 "solve guesses",
1120 $total->guesses,
1121 "rounds",
1122 $total->rounds,
1123 "flagged",
1124 $total->flagged,
1125 "revealed",
1126 $total->opened,
1127 );
1128 }
1129 $game->ending();
1130 break;
1131 case "ANALYZE":
1132 $copy = $game->copy();
1133 $total = $copy->auto($this->safety);
1134 if ($copy->state === GameState::Lost) {
1135 emit("analyze boom", $copy->cell($copy->exploded), "rounds", $total->rounds);
1136 } elseif ($copy->state === GameState::Won) {
1137 emit("analyze clear rounds", $total->rounds);
1138 } else {
1139 emit(
1140 "analyze stuck rounds",
1141 $total->rounds,
1142 "revealed",
1143 $copy->openCount() . "/" . ($copy->size - $copy->mineCount),
1144 );
1145 }
1146 break;
1147 case "PROB":
1148 if ($game->revealed[$cell]) {
1149 fail("already revealed");
1150 }
1151 if ($game->marks[$cell] === Mark::Flag) {
1152 fail("flagged");
1153 }
1154 $component = $game->frontier()->at[$cell];
1155 if ($component === null) {
1156 emit("prob", $game->cell($cell), "unconstrained");
1157 } else {
1158 $component->count();
1159 if ($component->configurations === 0) {
1160 emit("prob", $game->cell($cell), "inconsistent");
1161 } else {
1162 emit(
1163 "prob",
1164 $game->cell($cell),
1165 fraction($component->mineCount($cell), $component->configurations),
1166 "size",
1167 count($component->cells),
1168 "configs",
1169 $component->configurations,
1170 );
1171 }
1172 }
1173 break;
1174 case "FRONTIER":
1175 $frontier = $game->frontier();
1176 emit(
1177 "frontier",
1178 count(array_filter($frontier->at)),
1179 "components",
1180 count($frontier->components),
1181 );
1182 foreach ($frontier->components as $component) {
1183 $component->count();
1184 $at = $game->cell($component->cells[0]);
1185 $size = count($component->cells);
1186 if ($component->configurations === 0) {
1187 emit("component", $at, "size", $size, "inconsistent");
1188 } else {
1189 emit(
1190 "component",
1191 $at,
1192 "size",
1193 $size,
1194 "configs",
1195 $component->configurations,
1196 "min",
1197 $component->minimum,
1198 "max",
1199 $component->maximum,
1200 );
1201 }
1202 }
1203 break;
1204 case "BOARDINFO":
1205 $game->boardInfo();
1206 break;
1207 }
1208 if ($saved !== null && count($game->moves) > $saved->moveCount) {
1209 $game->undo[] = $saved;
1210 }
1211 if (!$wasFinished && $game->finished()) {
1212 $this->recordFinish($game);
1213 }
1214 }
1215}
