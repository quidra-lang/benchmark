74function flips(b: string, s: Color, p: number): number[] {
75 if (b[p] !== ".") return [];
76 const out: number[] = [],
77 r = Math.floor(p / 8),
78 c = p % 8;
79 for (let dr = -1; dr <= 1; dr++)
80 for (let dc = -1; dc <= 1; dc++) {
81 if (!dr && !dc) continue;
82 let y = r + dr,
83 x = c + dc;
84 const line: number[] = [];
85 while (inside(y, x) && b[8 * y + x] === other(s)) {
86 line.push(8 * y + x);
87 y += dr;
88 x += dc;
89 }
90 if (line.length && inside(y, x) && b[8 * y + x] === s) out.push(...line);
91 }
92 return out.sort((a, b) => a - b);
93}

109function gameOver(b: string) {
110 const x = count(b, "B"),
111 y = count(b, "W");

114function evaluate(b: string, s: Color) {
115 let w = 0;
116 for (let i = 0; i < 64; i++) w += b[i] === s ? WEIGHTS[i] : b[i] === other(s) ? -WEIGHTS[i] : 0;
117 const own = legal(b, s).length,
118 enemy = legal(b, other(s)).length;
119 return [w + 10 * (own - enemy), w, own, enemy];
120}
121const searchCache = new Map<string, number>(),

123function search(b: string, s: Color, d: number): number {
124 const key = b + s + ":" + d;
125 if (searchCache.has(key)) return searchCache.get(key)!;
126 const moves = legal(b, s),
127 o = other(s);
128 let v: number;
129 if (!moves.length && !legal(b, o).length) v = (d < 0 ? 1 : 1000) * (count(b, s) - count(b, o));
130 else if (d === 0) v = evaluate(b, s)[0];
131 else if (!moves.length) v = -search(b, o, d < 0 ? -1 : d - 1);
132 else {
133 v = -1000000;
134 for (const p of moves) v = Math.max(v, -search(place(b, s, p), o, d < 0 ? -1 : d - 1));
135 }
136 if (searchCache.size > 300000) searchCache.clear();
137 searchCache.set(key, v);
138 return v;
139}
140const ranked = (b: string, s: Color, d: number): Ranking[] =>

142 .map((p): Ranking => [-search(place(b, s, p), other(s), d < 0 ? -1 : d - 1), p])
143 .sort((a, b) => b[0] - a[0] || a[1] - b[1]);
144function variation(b: string, s: Color, d: number) {
145 const steps: string[] = [];
146 while (!terminal(b) && d !== 0) {
147 const ms = ranked(b, s, d);
148 if (ms.length) {
149 steps.push(name(ms[0][1]));
150 b = place(b, s, ms[0][1]);
151 } else steps.push("pass");
152 s = other(s);
153 if (d > 0) d--;
154 }
155 return {
156 steps,
157 board: b,
158 };
159}
160function perft(b: string, s: Color, d: number): number {
161 if (!d) return 1;
162 const key = b + s + d;
163 if (perftCache.has(key)) return perftCache.get(key)!;
164 const ms = legal(b, s),
165 o = other(s);
166 const n = ms.length
167 ? ms.reduce((t, p) => t + perft(place(b, s, p), o, d - 1), 0)
168 : legal(b, o).length
169 ? perft(b, o, d - 1)
170 : 1;
171 if (perftCache.size > 200000) perftCache.clear();
172 perftCache.set(key, n);
173 return n;
174}
175function stable(b: string) {
176 const set = new Set<number>(),
177 axes = [
178 [0, 1],
179 [1, 0],
180 [1, 1],
181 [1, -1],
182 ];
183 while (true) {
184 const found: number[] = [];
185 for (let p = 0; p < 64; p++) {
186 if (b[p] === "." || set.has(p)) continue;
187 const row = Math.floor(p / 8),
188 col = p % 8;
189 let ok = true;
190 for (const [dr, dc] of axes) {
191 let full = true,
192 anchor = false;
193 for (const sign of [-1, 1]) {
194 let r = row + dr * sign,
195 c = col + dc * sign;
196 if (!inside(r, c) || (set.has(8 * r + c) && b[8 * r + c] === b[p]))
197 anchor = true;
198 while (inside(r, c)) {
199 if (b[8 * r + c] === ".") full = false;
200 r += dr * sign;
201 c += dc * sign;
202 }
203 }
204 if (!full && !anchor) {
205 ok = false;
206 break;
207 }
208 }
209 if (ok) found.push(p);
210 }
211 if (!found.length) break;
212 for (const p of found) set.add(p);
213 }
214 return ["B", "W"].map((s) => [...set].filter((p) => b[p] === s).length);
215}
216function transform(p: number, op: string) {
217 const r = Math.floor(p / 8),
218 c = p % 8;
219 switch (op) {
220 case "rot90":
221 return 8 * c + 7 - r;
222 case "rot180":
223 return 63 - p;
224 case "rot270":
225 return 8 * (7 - c) + r;
226 case "mirror":
227 return 8 * r + 7 - c;
228 case "flip":
229 return 8 * (7 - r) + c;
230 case "diag":
231 return 8 * c + r;
232 default:
233 return 8 * (7 - c) + 7 - r;
234 }
235}
236function transformBoard(b: string, op: string) {
237 const a = Array(64).fill(".");
238 for (let p = 0; p < 64; p++) a[transform(p, op)] = b[p];
239 return a.join("");
240}

257 }

432 monteCarlo(k: number, seed: number) {
433 const rng = new Random(seed);
434 let best = -1,
435 bp = -1;
436 for (const p of legal(this.board, this.side)) {
437 let w = 0,
438 d = 0,
439 l = 0;
440 const start = place(this.board, this.side, p);
441 for (let i = 0; i < k; i++) {
442 const end = randomGame(start, other(this.side), rng),
443 diff = count(end, this.side) - count(end, other(this.side));
444 if (diff > 0) w++;
445 else if (diff < 0) l++;
446 else d++;
447 }
448 this.emit(name(p), w, d, l);
449 if (2 * w + d > best) {
450 best = 2 * w + d;
451 bp = p;
452 }
453 }
454 this.emit("best", name(bp), best);
455 }
456 execute(text: string) {
457 const [cmd, ...a] = text.split(" ");
458 let error = "";
459 if (!COMMANDS.includes(cmd)) error = "unknown command";
460 else if (cmd === "LOAD") {
461 if (a.length !== 2 || !/^[.BW]{64}$/.test(a[0]) || !["B", "W"].includes(a[1]))
462 error = "bad board";
463 } else if (["MOVE", "FLIPS"].includes(cmd)) {
464 if (a.length !== 1) error = "bad arguments";
465 else if (!/^[a-h][1-8]$/.test(a[0])) error = "bad square";
466 } else if (["PLAY", "RECORD"].includes(cmd)) {
467 if (a.length !== 1) error = "bad arguments";
468 else if (a[0].length > 120 || !/^([a-h][1-8])+$/.test(a[0])) error = "bad record";
469 } else if (ZERO.includes(cmd)) {
470 if (a.length) error = "bad arguments";
471 } else if (cmd === "TRANSFORM") {
472 if (a.length !== 1 || !TRANSFORMS.includes(a[0])) error = "bad arguments";
473 } else {
474 const n = a.map(Number),
475 expected = ["AUTOPLAY", "MC"].includes(cmd) ? 2 : 1;
476 if (a.length !== expected || a.some((x) => !/^[0-9]+$/.test(x)))
477 error = "bad arguments";
478 else {
479 let ok;
480 if (cmd === "AUTOPLAY") ok = n.every((v) => v >= 1 && v <= 4);
481 else if (["HINT", "PV", "AI", "ANALYZE"].includes(cmd)) ok = n[0] >= 1 && n[0] <= 4;
482 else if (["PERFT", "DIVIDE"].includes(cmd)) ok = n[0] >= 1 && n[0] <= 8;
483 else if (cmd === "MC")
484 ok = n[0] >= 1 && n[0] <= 100 && n[1] >= 1 && n[1] <= 2147483646;
485 else if (cmd === "RANDOM") ok = n[0] >= 1 && n[0] <= 2147483646;
486 else ok = n[0] >= 1;
487 if (!ok) error = "bad arguments";
488 }
489 }
490 if (error) {
491 this.emit("ERROR", error);
492 return;
493 }
494 if (ACTIVE.includes(cmd) && this.over) {
495 this.emit("ERROR game over");
496 return;
497 }
498 const n = Number(a[0]);
499 switch (cmd) {
500 case "NEW":
501 this.reset();
502 this.emit("OK");
503 break;
504 case "LOAD":
505 this.board = a[0];
506 this.side = a[1] as Color;
507 this.over = false;
508 this.history = [];
509 this.redo = [];
510 this.emit("OK");
511 this.adjust();
512 break;
513 case "DUMP":
514 this.emit("board", this.board, this.side);
515 break;
516 case "SHOW": {
517 this.emit(" a b c d e f g h");
518 const ms = this.over ? [] : legal(this.board, this.side);
519 for (let r = 0; r < 8; r++)
520 this.emit(
521 r + 1,
522 ...Array.from(
523 {
524 length: 8,
525 },
526 (_, c) => (ms.includes(8 * r + c) ? "*" : this.board[8 * r + c]),
527 ),
528 );
529 this.emit(this.status());
530 break;
531 }
532 case "SCORE":
533 this.emit(
534 "score B",
535 count(this.board, "B"),
536 "W",
537 count(this.board, "W"),
538 "empty",
539 count(this.board, "."),
540 );
541 break;
542 case "LEGAL": {
543 const ms = this.over ? [] : legal(this.board, this.side);
544 this.emit("legal", ms.length, ...ms.map(name));
545 break;
546 }
547 case "MOVE":
548 case "FLIPS": {
549 const p = square(a[0]),
550 why = this.reason(p);
551 if (why) this.emit("ILLEGAL", why);
552 else if (cmd === "MOVE") this.move(p);
553 else {
554 const f = flips(this.board, this.side, p);
555 this.emit("flips", a[0], f.length, ...f.map(name));
556 }
557 break;
558 }
559 case "PLAY": {
560 for (let i = 0; i < a[0].length; i += 2) {
561 const p = square(a[0].slice(i, i + 2)),
562 why = this.reason(p);
563 if (why) {
564 this.emit("ILLEGAL move", i / 2 + 1, name(p), why);
565 break;
566 }
567 this.move(p);
568 }
569 break;
570 }
571 case "RECORD":
572 this.record(a[0]);
573 break;
574 case "UNDO":
575 if (n > this.history.length) this.emit("ERROR cannot undo");
576 else {
577 const removed = this.history.splice(this.history.length - n);
578 this.restore(removed[0].before);
579 this.redo = [...removed, ...this.redo];
580 this.emit(this.status());
581 }
582 break;
583 case "REDO":
584 if (n > this.redo.length) this.emit("ERROR cannot redo");
585 else {
586 const added = this.redo.splice(0, n);
587 this.history.push(...added);
588 this.restore(added[added.length - 1].after);
589 this.emit(this.status());
590 }
591 break;
592 case "HISTORY":
593 this.emit("history", this.history.length);
594 this.history.forEach((r, i) =>
595 this.emit(
596 i + 1,
597 r.side,
598 name(r.square),
599 r.count,
600 ...(r.passed ? ["pass"] : []),
601 ),
602 );
603 break;
604 case "SHEET":
605 this.sheet();
606 break;
607 case "STATS":
608 this.stats();
609 break;
610 case "TRANSFORM": {
611 const op = a[0];
612 this.board = transformBoard(this.board, op);
613 for (const r of [...this.history, ...this.redo]) {
614 r.square = transform(r.square, op);
615 r.before = {
616 ...r.before,
617 board: transformBoard(r.before.board, op),
618 };
619 r.after = {
620 ...r.after,
621 board: transformBoard(r.after.board, op),
622 };
623 }
624 this.emit("OK");
625 break;
626 }
627 case "EVAL": {
628 const [v, w, x, y] = evaluate(this.board, this.side);
629 this.emit("eval", v, "weights", w, "mobility", x, y);
630 break;
631 }
632 case "STABLE": {
633 const [b, w] = stable(this.board);
634 this.emit("stable B", b, "W", w);
635 break;
636 }
637 case "HINT":
638 for (const [v, p] of ranked(this.board, this.side, n)) this.emit(name(p), v);
639 break;
640 case "PV": {
641 const v = search(this.board, this.side, n),
642 pv = variation(this.board, this.side, n);
643 this.emit("pv", v, ...pv.steps);
644 break;
645 }
646 case "AI": {
647 const [v, p] = ranked(this.board, this.side, n)[0];
648 this.move(p, false, v);
649 break;
650 }
651 case "ANALYZE": {
652 const loss = {
653 B: 0,
654 W: 0,
655 };
656 this.history.forEach((r, i) => {
657 const rankedMoves = ranked(r.before.board, r.side, n),
658 [best, bp] = rankedMoves[0],
659 v = rankedMoves.find((x) => x[1] === r.square)![0];
660 loss[r.side] += best - v;
661 this.emit(i + 1, r.side, name(r.square), v, "best", name(bp), best);
662 });
663 this.emit("loss B", loss.B, "W", loss.W);
664 break;
665 }
666 case "SOLVE":
667 if (count(this.board, ".") >= 9) this.emit("ERROR too many empties");
668 else {
669 for (const [v, p] of ranked(this.board, this.side, -1)) this.emit(name(p), v);
670 const pv = variation(this.board, this.side, -1);
671 this.emit("line", ...pv.steps);
672 this.emit("final B", count(pv.board, "B"), "W", count(pv.board, "W"));
673 }
674 break;
675 case "AUTOPLAY":
676 this.auto(n, Number(a[1]));
677 break;
678 case "RANDOM":
679 this.auto(0, 0, n);
680 break;
681 case "MC":
682 this.monteCarlo(n, Number(a[1]));
683 break;
684 case "PERFT":
685 this.emit("perft", n, perft(this.board, this.side, n));
686 break;
687 case "DIVIDE": {
688 let total = 0;
689 for (const p of legal(this.board, this.side)) {
690 const v = perft(place(this.board, this.side, p), other(this.side), n - 1);
