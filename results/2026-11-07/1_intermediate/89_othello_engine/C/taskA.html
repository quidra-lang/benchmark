78static int flips(const char *board, char side, int p, int result[64]) {
79 if (board[p] != '.')
80 return 0;
81 int n = 0, row = p / 8, col = p % 8;
82 for (int dr = -1; dr <= 1; ++dr) {
83 for (int dc = -1; dc <= 1; ++dc) {
84 if (dr == 0 && dc == 0)
85 continue;
86 int r = row + dr, c = col + dc, path[8], count = 0;
87 while (inside(r, c) && board[8 * r + c] == other(side)) {
88 path[count++] = 8 * r + c;
89 r += dr;
90 c += dc;
91 }
92 if (count != 0 && inside(r, c) && board[8 * r + c] == side) {
93 for (int i = 0; i < count; ++i)
94 result[n++] = path[i];
95 }
96 }
97 }
98 qsort(result, (size_t)n, sizeof(int), compare_int);
99 return n;
100}

106 result[count++] = p;

144static int evaluate(const char *board, char side, int values[4]) {
145 int weight = 0, moves[64];
146 for (int p = 0; p < 64; ++p) {
147 if (board[p] == side)
148 weight += weights[p];
149 else if (board[p] == other(side))
150 weight -= weights[p];
151 }
152 int own = legal(board, side, moves);
153 int enemy = legal(board, other(side), moves);
154 values[0] = weight + 10 * (own - enemy);
155 values[1] = weight;
156 values[2] = own;
157 values[3] = enemy;
158 return values[0];
159}

161static int search(const char *board, char side, int depth) {
162 int moves[64], enemy[64], count = legal(board, side, moves);
163 char opposite = other(side);
164 if (!count && !legal(board, opposite, enemy)) {
165 return (depth < 0 ? 1 : 1000) * (disc_count(board, side) - disc_count(board, opposite));
166 }
167 if (!depth) {
168 int values[4];
169 return evaluate(board, side, values);
170 }
171 if (!count)
172 return -search(board, opposite, depth < 0 ? -1 : depth - 1);
173 int best = -10000000;
174 for (int i = 0; i < count; ++i) {
175 char next[65];
176 place(board, side, moves[i], next);
177 int value = -search(next, opposite, depth < 0 ? -1 : depth - 1);
178 if (value > best)
179 best = value;
180 }
181 return best;
182}

184static int compare_ranking(const void *a, const void *b) {
185 const Ranking *x = a, *y = b;
186 if (x->value != y->value)
187 return (x->value < y->value) - (x->value > y->value);
188 return (x->p > y->p) - (x->p < y->p);
189}

191static int ranked(const char *board, char side, int depth, Ranking result[64]) {
192 int moves[64], n = legal(board, side, moves);
193 for (int i = 0; i < n; ++i) {
194 char next[65];
195 place(board, side, moves[i], next);
196 result[i] = (Ranking){moves[i], -search(next, other(side), depth < 0 ? -1 : depth - 1)};
197 }
198 qsort(result, (size_t)n, sizeof(Ranking), compare_ranking);
199 return n;
200}

202static void variation(const char *board, char side, int depth, char path[256], char final[65]) {
203 memcpy(final, board, 65);
204 path[0] = '\0';
205 while (depth != 0 && !terminal(final)) {
206 Ranking choices[64];
207 int n = ranked(final, side, depth, choices);
208 char word[8];
209 if (n) {
210 square_name(choices[0].p, word);
211 char next[65];
212 place(final, side, choices[0].p, next);
213 memcpy(final, next, 65);
214 } else {
215 strcpy(word, "pass");
216 }
217 if (path[0] != '\0')
218 strcat(path, " ");
219 strcat(path, word);
220 side = other(side);
221 if (depth > 0)
222 --depth;
223 }
224}

226static uint64_t perft(const char *board, char side, int depth) {
227 if (depth == 0)
228 return 1;
229 int moves[64], other_moves[64], n = legal(board, side, moves);
230 char opponent = other(side);
231 if (!n) {
232 return legal(board, opponent, other_moves) == 0 ? 1 : perft(board, opponent, depth - 1);
233 }
234 uint64_t count = 0;
235 for (int i = 0; i < n; ++i) {
236 char next[65];
237 place(board, side, moves[i], next);
238 count += perft(next, opponent, depth - 1);
239 }
240 return count;
241}

243static void stable(const char *board, int result[2]) {
244 bool set[64] = {false};
245 const int dr[4] = {0, 1, 1, 1};
246 const int dc[4] = {1, 0, 1, -1};
247 while (true) {
248 int found[64], n = 0;
249 for (int p = 0; p < 64; ++p) {
250 if (board[p] == '.' || set[p])
251 continue;
252 bool valid = true;
253 for (int axis = 0; axis < 4; ++axis) {
254 bool full = true, anchor = false;
255 for (int sign = -1; sign <= 1; sign += 2) {
256 int r = p / 8 + dr[axis] * sign;
257 int c = p % 8 + dc[axis] * sign;
258 if (!inside(r, c) || (set[8 * r + c] && board[8 * r + c] == board[p]))
259 anchor = true;
260 while (inside(r, c)) {
261 if (board[8 * r + c] == '.')
262 full = false;
263 r += dr[axis] * sign;
264 c += dc[axis] * sign;
265 }
266 }
267 if (!full && !anchor) {
268 valid = false;
269 break;
270 }
271 }
272 if (valid)
273 found[n++] = p;
274 }
275 if (!n)
276 break;
277 for (int i = 0; i < n; ++i)
278 set[found[i]] = true;
279 }
280 result[0] = result[1] = 0;
281 for (int p = 0; p < 64; ++p) {
282 if (set[p] && board[p] == 'B')
283 ++result[0];
284 if (set[p] && board[p] == 'W')
285 ++result[1];
286 }
287}

289static int transform_square(int p, const char *op) {
290 int r = p / 8, c = p % 8;
291 if (!strcmp(op, "rot90"))
292 return 8 * c + 7 - r;
293 if (!strcmp(op, "rot180"))
294 return 63 - p;
295 if (!strcmp(op, "rot270"))
296 return 8 * (7 - c) + r;
297 if (!strcmp(op, "mirror"))
298 return 8 * r + 7 - c;
299 if (!strcmp(op, "flip"))
300 return 8 * (7 - r) + c;
301 if (!strcmp(op, "diag"))
302 return 8 * c + r;
303 return 8 * (7 - c) + 7 - r;
304}

319static void random_game(const char *board, char side, Random *random, char result[65]) {
320 memcpy(result, board, 65);
321 while (true) {
322 int moves[64], enemy[64], n = legal(result, side, moves);
323 if (n) {
324 char next[65];
325 place(result, side, moves[draw(random) % (uint64_t)n], next);
326 memcpy(result, next, 65);
327 } else if (!legal(result, other(side), enemy)) {
328 return;
329 }
330 side = other(side);
331 }

335 int moves[64];
336 if (legal(g->pos.board, g->pos.side, moves))

353static bool play(Game *g, int p, bool quiet, bool have_value, int value) {
354 Position before = g->pos;
355 char side = g->pos.side;
356 int flipped[64], n = flips(before.board, side, p, flipped);
357 char board[65];
358 place(before.board, side, p, board);
359 memcpy(g->pos.board, board, 65);
360 if (!quiet) {
361 char sq[3];
362 square_name(p, sq);
363 if (have_value)
364 printf("%c %s flips %d value %d\n", side, sq, n, value);
365 else
366 printf("%c %s flips %d\n", side, sq, n);
367 }
368 g->pos.side = other(side);
369 bool passed = adjust(g, quiet);
370 g->history[g->nh++] = (Move){side, p, n, passed, before, g->pos};
371 g->nr = 0;
372 return passed;
373}

386static void record(Game *g, const char *text) {
387 Game *trial = malloc(sizeof(Game));
388 if (!trial)
389 exit(EXIT_FAILURE);
390 initialize(trial);
391 int positions[60], total = (int)strlen(text) / 2, passes = 0;
392 for (int i = 0; i < total; ++i) {
393 int p = 8 * (text[2 * i + 1] - '1') + text[2 * i] - 'a';
394 const char *why = reason(trial, p);
395 if (why) {
396 char sq[3];
397 square_name(p, sq);
398 printf("ILLEGAL move %d %s %s\n", i + 1, sq, why);
399 free(trial);
400 return;
401 }
402 positions[i] = p;
403 if (play(trial, p, true, false, 0))
404 ++passes;
405 }
406 *g = *trial;
407 free(trial);
408 char first[3];
409 square_name(positions[0], first);
410 const char *op = !strcmp(first, "e6") ? "diag"
411 : !strcmp(first, "d3") ? "anti"
412 : !strcmp(first, "c4") ? "rot180"
413 : "";
414 char normalized[121];
415 for (int i = 0; i < total; ++i) {
416 int p = *op ? transform_square(positions[i], op) : positions[i];
417 square_name(p, normalized + 2 * i);
418 }
419 normalized[2 * total] = '\0';
420 printf("record %d moves %d passes\nnormal %s\n", total, passes, normalized);
421 char line[100];
422 status(g, line);
423 puts(line);
424}

453static void stats(const Game *g) {
454 int total = 0, passes = 0;
455 for (int i = 0; i < g->nh; ++i) {
456 total += g->history[i].flipped;
457 passes += g->history[i].passed;
458 }
459 if (!g->nh)
460 puts("moves 0");
461 else {
462 char mean[32];
463 average(total, g->nh, mean);
464 printf("moves %d flips %d avg %s passes %d\n", g->nh, total, mean, passes);
465 }
466 for (char side = 'B'; side <= 'W'; side += 'W' - 'B') {
467 int moves = 0, sum = 0, maximum = -1, max_square = -1, corners = 0;
468 for (int i = 0; i < g->nh; ++i) {
469 const Move *m = &g->history[i];
470 if (m->side != side)
471 continue;
472 ++moves;
473 sum += m->flipped;
474 if (m->flipped > maximum) {
475 maximum = m->flipped;
476 max_square = m->square;
477 }
478 if (m->square == 0 || m->square == 7 || m->square == 56 || m->square == 63)
479 ++corners;
480 }
481 if (!moves)
482 printf("%c moves 0\n", side);
483 else {
484 char mean[32], sq[3];
485 average(sum, moves, mean);
486 square_name(max_square, sq);
487 printf("%c moves %d flips %d avg %s max %d %s corners %d\n", side, moves, sum, mean,
488 maximum, sq, corners);
489 }
490 }

494 Random rng = {seed};
495 int moves = 0, passes = 0;
496 char text[121] = "";
497 while (!g->pos.over) {
498 int p;
499 if (random) {
500 int choices[64], n = legal(g->pos.board, g->pos.side, choices);

515 text);
516 char line[100];
517 game_over(g->pos.board, line);

521static void monte_carlo(const Game *g, int games, uint64_t seed) {
522 Random rng = {seed};
523 int positions[64], n = legal(g->pos.board, g->pos.side, positions);
524 int best = -1, best_pos = -1;
525 for (int i = 0; i < n; ++i) {
526 char start[65];
527 place(g->pos.board, g->pos.side, positions[i], start);
528 int w = 0, d = 0, l = 0;
529 for (int j = 0; j < games; ++j) {
530 char end[65];
531 random_game(start, other(g->pos.side), &rng, end);
532 int delta = disc_count(end, g->pos.side) - disc_count(end, other(g->pos.side));
533 if (delta > 0)
534 ++w;
535 else if (delta == 0)
536 ++d;
537 else
538 ++l;
539 }
540 char sq[3];
541 square_name(positions[i], sq);
542 printf("%s %d %d %d\n", sq, w, d, l);
543 if (2 * w + d > best) {
544 best = 2 * w + d;
545 best_pos = positions[i];
546 }
547 }
548 char sq[3];
549 square_name(best_pos, sq);
550 printf("best %s %d\n", sq, best);
551}

586static void execute(Game *g, char *line) {
587 static const char *zero[] = {"NEW", "DUMP", "SHOW", "SCORE", "LEGAL", "HISTORY",
588 "SHEET", "STATS", "EVAL", "STABLE", "SOLVE"};
589 static const char *numeric[] = {"UNDO", "REDO", "HINT", "PV", "AI",
590 "ANALYZE", "RANDOM", "PERFT", "DIVIDE"};
591 static const char *active[] = {"FLIPS", "MOVE", "PLAY", "HINT", "PV", "AI",
592 "SOLVE", "AUTOPLAY", "RANDOM", "MC", "DIVIDE"};
593 static const char *additional[] = {"LOAD", "TRANSFORM", "AUTOPLAY", "MC",
594 "FLIPS", "MOVE", "PLAY", "RECORD"};
595 static const char *ops[] = {"rot90", "rot180", "rot270", "mirror", "flip", "diag", "anti"};
596 char *parts[210];
597 int argc = 0;
598 for (char *p = strtok(line, " "); p != NULL; p = strtok(NULL, " ")) {
599 parts[argc++] = p;
600 }
601 if (!argc)
602 return;
603 const char *cmd = parts[0];
604 char **args = parts + 1;
605 int count = argc - 1;
606 if (!among(cmd, zero, sizeof(zero) / sizeof(zero[0])) &&
607 !among(cmd, numeric, sizeof(numeric) / sizeof(numeric[0])) &&
608 !among(cmd, additional, sizeof(additional) / sizeof(additional[0]))) {
609 puts("ERROR unknown command");
610 return;
611 }
612 const char *error = NULL;
613 uint64_t a = 0, b = 0;
614 if (!strcmp(cmd, "LOAD")) {
615 if (count != 2 || strlen(args[0]) != 64 || (strcmp(args[1], "B") && strcmp(args[1], "W")))
616 error = "bad board";
617 if (!error)
618 for (int i = 0; i < 64; ++i) {
619 if (args[0][i] != 'B' && args[0][i] != 'W' && args[0][i] != '.')
620 error = "bad board";
621 }
622 } else if (!strcmp(cmd, "MOVE") || !strcmp(cmd, "FLIPS")) {
623 if (count != 1)
624 error = "bad arguments";
625 else if (!valid_square(args[0]))
626 error = "bad square";
627 } else if (!strcmp(cmd, "PLAY") || !strcmp(cmd, "RECORD")) {
628 if (count != 1)
629 error = "bad arguments";
630 else if (!valid_record(args[0]))
631 error = "bad record";
632 } else if (among(cmd, zero, sizeof(zero) / sizeof(zero[0]))) {
633 if (count != 0)
634 error = "bad arguments";
635 } else if (!strcmp(cmd, "TRANSFORM")) {
636 if (count != 1 || !among(args[0], ops, sizeof(ops) / sizeof(ops[0])))
637 error = "bad arguments";
638 } else {
639 int expected = !strcmp(cmd, "MC") || !strcmp(cmd, "AUTOPLAY") ? 2 : 1;
640 if (count != expected || !unsigned_number(args[0], &a) ||
641 (expected == 2 && !unsigned_number(args[1], &b)))
642 error = "bad arguments";
643 if (!error) {
644 if (!strcmp(cmd, "AUTOPLAY")) {
645 if (a < 1 || a > 4 || b < 1 || b > 4)
646 error = "bad arguments";
647 } else if (!strcmp(cmd, "HINT") || !strcmp(cmd, "PV") || !strcmp(cmd, "AI") ||
648 !strcmp(cmd, "ANALYZE")) {
649 if (a < 1 || a > 4)
650 error = "bad arguments";
651 } else if (!strcmp(cmd, "PERFT") || !strcmp(cmd, "DIVIDE")) {
652 if (a < 1 || a > 8)
653 error = "bad arguments";
654 } else if (!strcmp(cmd, "MC")) {
655 if (a < 1 || a > 100 || b < 1 || b > 2147483646ULL)
656 error = "bad arguments";
657 } else if (!strcmp(cmd, "RANDOM")) {
658 if (a < 1 || a > 2147483646ULL)
659 error = "bad arguments";
660 } else if (a < 1)
661 error = "bad arguments";
662 }
663 }
664 if (error) {
665 printf("ERROR %s\n", error);
666 return;
667 }
668 if (g->pos.over && among(cmd, active, sizeof(active) / sizeof(active[0]))) {
669 puts("ERROR game over");
670 return;
671 }
672 int value = (int)a;
673 if (!strcmp(cmd, "NEW")) {
674 initialize(g);
675 puts("OK");
676 } else if (!strcmp(cmd, "LOAD")) {
677 memcpy(g->pos.board, args[0], 65);
678 g->pos.side = args[1][0];
679 g->pos.over = false;
680 g->nh = g->nr = 0;
681 puts("OK");
682 adjust(g, false);
683 } else if (!strcmp(cmd, "DUMP")) {
684 printf("board %s %c\n", g->pos.board, g->pos.side);
685 } else if (!strcmp(cmd, "SHOW")) {
686 int moves[64], n = g->pos.over ? 0 : legal(g->pos.board, g->pos.side, moves);
687 bool possible[64] = {false};
688 for (int i = 0; i < n; ++i)
689 possible[moves[i]] = true;
690 puts(" a b c d e f g h");
691 for (int r = 0; r < 8; ++r) {
692 printf("%d", r + 1);
693 for (int c = 0; c < 8; ++c) {
694 int p = 8 * r + c;
695 printf(" %c", possible[p] ? '*' : g->pos.board[p]);
696 }
697 putchar('\n');
698 }
699 char line[100];
700 status(g, line);
701 puts(line);
702 } else if (!strcmp(cmd, "SCORE")) {
703 printf("score B %d W %d empty %d\n", disc_count(g->pos.board, 'B'),
704 disc_count(g->pos.board, 'W'), disc_count(g->pos.board, '.'));
705 } else if (!strcmp(cmd, "LEGAL")) {
706 int moves[64], n = g->pos.over ? 0 : legal(g->pos.board, g->pos.side, moves);
707 printf("legal %d", n);
708 for (int i = 0; i < n; ++i) {
709 char sq[3];
710 square_name(moves[i], sq);
711 printf(" %s", sq);
712 }
713 putchar('\n');
714 } else if (!strcmp(cmd, "MOVE") || !strcmp(cmd, "FLIPS")) {
715 int p = square_number(args[0]);
716 const char *why = reason(g, p);
717 if (why)
718 printf("ILLEGAL %s\n", why);
719 else if (!strcmp(cmd, "MOVE"))
720 play(g, p, false, false, 0);
721 else {
722 int results[64], n = flips(g->pos.board, g->pos.side, p, results);
723 printf("flips %s %d", args[0], n);
724 for (int i = 0; i < n; ++i) {
725 char sq[3];
726 square_name(results[i], sq);
727 printf(" %s", sq);
728 }
729 putchar('\n');
730 }
731 } else if (!strcmp(cmd, "PLAY")) {
732 size_t len = strlen(args[0]);
733 for (size_t i = 0; i < len; i += 2) {
734 int p = 8 * (args[0][i + 1] - '1') + args[0][i] - 'a';
735 const char *why = reason(g, p);
736 if (why) {
737 char sq[3];
738 square_name(p, sq);
739 printf("ILLEGAL move %zu %s %s\n", i / 2 + 1, sq, why);
740 break;
741 }
742 play(g, p, false, false, 0);
743 }
744 } else if (!strcmp(cmd, "RECORD")) {
745 record(g, args[0]);
746 } else if (!strcmp(cmd, "UNDO")) {
747 if ((uint64_t)g->nh < a)
748 puts("ERROR cannot undo");
749 else {
750 int start = g->nh - value;
751 g->pos = g->history[start].before;
752 memmove(g->redo + value, g->redo, (size_t)g->nr * sizeof(Move));
753 memcpy(g->redo, g->history + start, (size_t)value * sizeof(Move));
754 g->nh = start;
755 g->nr += value;
756 char line[100];
757 status(g, line);
758 puts(line);
759 }
760 } else if (!strcmp(cmd, "REDO")) {
761 if ((uint64_t)g->nr < a)
762 puts("ERROR cannot redo");
763 else {
764 memcpy(g->history + g->nh, g->redo, (size_t)value * sizeof(Move));
765 g->pos = g->redo[value - 1].after;
766 g->nh += value;
767 g->nr -= value;
768 memmove(g->redo, g->redo + value, (size_t)g->nr * sizeof(Move));
769 char line[100];
770 status(g, line);
771 puts(line);
772 }
773 } else if (!strcmp(cmd, "HISTORY")) {
774 printf("history %d\n", g->nh);
775 for (int i = 0; i < g->nh; ++i) {
776 const Move *m = &g->history[i];
777 char sq[3];
778 square_name(m->square, sq);
779 printf("%d %c %s %d%s\n", i + 1, m->side, sq, m->flipped, m->passed ? " pass" : "");
780 }
781 } else if (!strcmp(cmd, "SHEET")) {
782 show_sheet(g);
783 } else if (!strcmp(cmd, "STATS")) {
784 stats(g);
785 } else if (!strcmp(cmd, "TRANSFORM")) {
786 char board[65];
787 transform_board(g->pos.board, args[0], board);
788 memcpy(g->pos.board, board, 65);
789 for (int which = 0; which < 2; ++which) {
790 Move *array = which == 0 ? g->history : g->redo;
791 int n = which == 0 ? g->nh : g->nr;
792 for (int i = 0; i < n; ++i) {
793 Move *m = array + i;
794 m->square = transform_square(m->square, args[0]);
795 transform_board(m->before.board, args[0], board);
796 memcpy(m->before.board, board, 65);
797 transform_board(m->after.board, args[0], board);
798 memcpy(m->after.board, board, 65);
799 }
800 }
801 puts("OK");
802 } else if (!strcmp(cmd, "EVAL")) {
803 int v[4];
804 evaluate(g->pos.board, g->pos.side, v);
805 printf("eval %d weights %d mobility %d %d\n", v[0], v[1], v[2], v[3]);
806 } else if (!strcmp(cmd, "STABLE")) {
807 int count[2];
808 stable(g->pos.board, count);
809 printf("stable B %d W %d\n", count[0], count[1]);
810 } else if (!strcmp(cmd, "HINT")) {
811 Ranking moves[64];
812 int n = ranked(g->pos.board, g->pos.side, value, moves);
813 for (int i = 0; i < n; ++i) {
814 char sq[3];
815 square_name(moves[i].p, sq);
816 printf("%s %d\n", sq, moves[i].value);
817 }
818 } else if (!strcmp(cmd, "PV")) {
819 char path[256], final[65];
820 variation(g->pos.board, g->pos.side, value, path, final);
821 printf("pv %d %s\n", search(g->pos.board, g->pos.side, value), path);
822 } else if (!strcmp(cmd, "AI")) {
823 Ranking choices[64];
824 ranked(g->pos.board, g->pos.side, value, choices);
825 play(g, choices[0].p, false, true, choices[0].value);
826 } else if (!strcmp(cmd, "ANALYZE")) {
827 int losses[2] = {0, 0};
828 for (int i = 0; i < g->nh; ++i) {
829 Move *m = &g->history[i];
830 Ranking choices[64];
831 int n = ranked(m->before.board, m->side, value, choices);
832 int played = 0;
833 for (int j = 0; j < n; ++j) {
834 if (choices[j].p == m->square)
835 played = choices[j].value;
836 }
837 losses[m->side == 'B' ? 0 : 1] += choices[0].value - played;
838 char sq[3], best[3];
839 square_name(m->square, sq);
840 square_name(choices[0].p, best);
841 printf("%d %c %s %d best %s %d\n", i + 1, m->side, sq, played, best, choices[0].value);
842 }
843 printf("loss B %d W %d\n", losses[0], losses[1]);
844 } else if (!strcmp(cmd, "SOLVE")) {
845 if (disc_count(g->pos.board, '.') >= 9)
846 puts("ERROR too many empties");
847 else {
848 Ranking choices[64];
849 int n = ranked(g->pos.board, g->pos.side, -1, choices);
850 for (int i = 0; i < n; ++i) {
851 char sq[3];
852 square_name(choices[i].p, sq);
853 printf("%s %d\n", sq, choices[i].value);
854 }
855 char path[256], final[65];
856 variation(g->pos.board, g->pos.side, -1, path, final);
857 printf("line %s\nfinal B %d W %d\n", path, disc_count(final, 'B'),
858 disc_count(final, 'W'));
859 }
860 } else if (!strcmp(cmd, "AUTOPLAY")) {
861 automatic(g, value, (int)b, false, 0);
862 } else if (!strcmp(cmd, "RANDOM")) {
863 automatic(g, 0, 0, true, a);
864 } else if (!strcmp(cmd, "MC")) {
865 monte_carlo(g, value, b);
866 } else if (!strcmp(cmd, "PERFT")) {
867 printf("perft %d %llu\n", value,
868 (unsigned long long)perft(g->pos.board, g->pos.side, value));
869 } else if (!strcmp(cmd, "DIVIDE")) {
870 int moves[64], n = legal(g->pos.board, g->pos.side, moves);
871 uint64_t total = 0;
872 for (int i = 0; i < n; ++i) {
873 char next[65], sq[3];
874 place(g->pos.board, g->pos.side, moves[i], next);
875 square_name(moves[i], sq);
876 uint64_t count = perft(next, other(g->pos.side), value - 1);
877 printf("%s %llu\n", sq, (unsigned long long)count);
878 total += count;
879 }
880 printf("total %llu\n", (unsigned long long)total);
881 }
882}
