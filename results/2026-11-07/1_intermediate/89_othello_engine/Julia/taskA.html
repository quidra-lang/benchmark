67const TRANSFORMS=["rot90", "rot180", "rot270", "mirror", "flip", "diag", "anti"]
68const ZERO=split("NEW DUMP SHOW SCORE LEGAL HISTORY SHEET STATS EVAL STABLE SOLVE")
69const NUMERIC=split("UNDO REDO HINT PV AI ANALYZE RANDOM PERFT DIVIDE")
70const ACTIVE=split("FLIPS MOVE PLAY HINT PV AI SOLVE AUTOPLAY RANDOM MC DIVIDE")
71const COMMANDS=vcat(ZERO, NUMERIC, split("LOAD TRANSFORM AUTOPLAY MC FLIPS MOVE PLAY RECORD"))
72other(s) = s=='B' ? 'W' : 'B'
73square_name(p) = string(Char(Int('a')+mod(p, 8)), div(p, 8)+1)
74square_index(s) = 8*(Int(s[2])-Int('1'))+Int(s[1])-Int('a')
75inside(r, c) = 0<=r<8&&0<=c<8
76function initial()
77 b=fill('.', 64)
78 b[28]=b[37]='W'
79 b[29]=b[36]='B'
80 String(b)
81end
82function flips(b, s, p)
83 b[p+1]!='.'&&return Int[]
84 row, col=divrem(p, 8)
85 out=Int[]
86 for dr in -1:1, dc in -1:1
87 dr==0&&dc==0&&continue
88 r=row+dr
89 c=col+dc
90 line=Int[]
91 while inside(r, c)&&b[8*r+c+1]==other(s)
92 push!(line, 8*r+c)
93 r+=dr
94 c+=dc
95 end
96 !isempty(line)&&inside(r, c)&&b[8*r+c+1]==s&&append!(out, line)
97 end
98 sort!(out)
99end
100legal(b, s) = [p for p in 0:63 if b[p+1]=='.'&&!isempty(flips(b, s, p))]
101function place(b, s, p)
102 a=collect(b)
103 for q in flips(b, s, p)
104 a[q+1]=s
105 end
106 a[p+1]=s
107 String(a)
108end
109terminal(b) = isempty(legal(b, 'B'))&&isempty(legal(b, 'W'))
110function game_over(b)
111 x=count(==('B'), b)
112 y=count(==('W'), b)
113 "game over B $x W $y "*(x>y ? "winner B" : x<y ? "winner W" : "draw")
114end
115function evaluate(b, s)
116 w=sum(b[i]==s ? WEIGHTS[i] : b[i]==other(s) ? -WEIGHTS[i] : 0 for i in 1:64)
117 own=length(legal(b, s))
118 enemy=length(legal(b, other(s)))
119 (w+10*(own-enemy), w, own, enemy)
120end
121const SEARCH_CACHE=Dict{Tuple{String,Char,Int},Int}();
122const PERFT_CACHE=Dict{Tuple{String,Char,Int},Int}()
123function search(b, s, d)
124 key=(b, s, d)
125 haskey(SEARCH_CACHE, key)&&return SEARCH_CACHE[key]
126 ms=legal(b, s)
127 o=other(s)
128 v=if isempty(ms)&&isempty(legal(b, o))
129 (d<0 ? 1 : 1000)*(count(==(s), b)-count(==(o), b))
130 elseif d==0
131 evaluate(b, s)[1]
132 elseif isempty(ms)
133 -search(b, o, d<0 ? -1 : d-1)
134 else
135 maximum(-search(place(b, s, p), o, d<0 ? -1 : d-1) for p in ms)
136 end
137 length(SEARCH_CACHE)>300000&&empty!(SEARCH_CACHE)
138 SEARCH_CACHE[key]=v
139 v
140end
141ranked(b, s, d) = sort(
142 [(-search(place(b, s, p), other(s), d<0 ? -1 : d-1), p) for p in legal(b, s)];
143 by = x->(-x[1], x[2]),
144)
145function variation(b, s, d)
146 steps=String[]
147 while d!=0&&!terminal(b)
148 ms=ranked(b, s, d)
149 if isempty(ms)
150 push!(steps, "pass")
151 else
152 p=ms[1][2]
153 push!(steps, square_name(p))
154 b=place(b, s, p)
155 end
156 s=other(s)
157 d>0&&(d-=1)
158 end
159 steps, b
160end
161function perft(b, s, d)
162 d==0&&return 1
163 key=(b, s, d)
164 haskey(PERFT_CACHE, key)&&return PERFT_CACHE[key]
165 ms=legal(b, s)
166 o=other(s)
167 n=isempty(ms) ? (isempty(legal(b, o)) ? 1 : perft(b, o, d-1)) :
168 sum(perft(place(b, s, p), o, d-1) for p in ms)
169 length(PERFT_CACHE)>200000&&empty!(PERFT_CACHE)
170 PERFT_CACHE[key]=n
171 n
172end
173function stable(b)
174 stable=falses(64)
175 axes=[(0, 1), (1, 0), (1, 1), (1, -1)]
176 while true
177 found=Int[]
178 for p in 0:63
179 (b[p+1]=='.'||stable[p+1])&&continue
180 row, col=divrem(p, 8)
181 ok=true
182 for (dr, dc) in axes
183 full=true
184 anchor=false
185 for sign in [-1, 1]
186 r=row+dr*sign
187 c=col+dc*sign
188 (!inside(r, c)||(stable[8*r+c+1]&&b[8*r+c+1]==b[p+1]))&&(anchor=true)
189 while inside(r, c)
190 b[8*r+c+1]=='.'&&(full=false)
191 r+=dr*sign
192 c+=dc*sign
193 end
194 end
195 if !full&&!anchor
196 ok=false
197 break
198 end
199 end
200 ok&&push!(found, p)
201 end
202 isempty(found)&&break
203 for p in found
204 stable[p+1]=true
205 end
206 end
207 [count(i->stable[i]&&b[i]==s, 1:64) for s in ['B', 'W']]
208end
209function transform(p, op)
210 r, c=divrem(p, 8)
211 op=="rot90"&&return 8*c+7-r
212 op=="rot180"&&return 63-p
213 op=="rot270"&&return 8*(7-c)+r
214 op=="mirror"&&return 8*r+7-c
215 op=="flip"&&return 8*(7-r)+c
216 op=="diag"&&return 8*c+r
217 8*(7-c)+7-r
218end
219function transform_board(b, op)
220 a=fill('.', 64)
221 for p in 0:63
222 a[transform(p, op)+1]=b[p+1]
223 end
224 String(a)
225end

229draw!(rng) = (rng.value=mod(48271*rng.value, 2147483647))
230function random_game(b, s, rng)
231 while true
232 ms=legal(b, s)
233 if !isempty(ms)
234 b=place(b, s, ms[mod(draw!(rng), length(ms))+1])
235 elseif isempty(legal(b, other(s)))
236 return b
237 end
238 s=other(s)
239 end
240end

242 let v=div(100*f, n)
243 string(div(v, 100), ".", lpad(string(mod(v, 100)), 2, '0'))
244 end

274function reset!(e)
275 restore!(e, Position(initial(), 'B', false))
276 empty!(e.history)
277 empty!(e.redo)

279function adjust!(e, quiet = false)
280 !isempty(legal(e.board, e.side))&&return false
281 if !isempty(legal(e.board, other(e.side)))
282 !quiet&&emit(e.side, "passes")
283 e.side=other(e.side)
284 return true
285 end
286 e.over=true
287 !quiet&&emit(game_over(e.board))
288 false
289end
290function move!(e, p, quiet = false, value = nothing)
291 before=snapshot(e)
292 s=e.side
293 k=length(flips(e.board, s, p))
294 e.board=place(e.board, s, p)
295 if !quiet
296 value===nothing ? emit(s, square_name(p), "flips", k) :
297 emit(s, square_name(p), "flips", k, "value", value)
298 end
299 e.side=other(s)
300 passed=adjust!(e, quiet)
301 push!(e.history, Move(s, p, k, passed, before, snapshot(e)))
302 empty!(e.redo)
303 passed
304end
305reason(e, p) =
306 e.over ? "game over" :
307 e.board[p+1]!='.' ? "occupied" : isempty(flips(e.board, e.side, p)) ? "no flips" : ""
308function record!(e, text)
309 temp=Othello()
310 ps=[square_index(text[i:(i+1)]) for i in 1:2:length(text)]
311 passes=0
312 for (i, p) in enumerate(ps)
313 why=reason(temp, p)
314 if !isempty(why)
315 emit("ILLEGAL move", i, square_name(p), why)
316 return
317 end
318 passes+=move!(temp, p, true)
319 end
320 restore!(e, snapshot(temp))
321 e.history=temp.history
322 empty!(e.redo)
323 op=get(Dict("f5"=>"", "e6"=>"diag", "d3"=>"anti", "c4"=>"rot180"), square_name(ps[1]), "")
324 emit("record", length(ps), "moves", passes, "passes")
325 emit("normal", join(square_name(isempty(op) ? p : transform(p, op)) for p in ps))
326 emit(status(e))
327end
328function sheet(e)
329 fields=string.(collect(isempty(e.history) ? e.board : e.history[1].before.board))
330 for (i, r) in enumerate(e.history)
331 fields[r.square+1]=string(i)
332 end
333 emit(" "*join(lpad(string(c), 3) for c in 'a':'h'))
334 for r in 0:7
335 emit(string(r+1)*join(lpad(x, 3) for x in fields[(8*r+1):(8*r+8)]))
336 end
337end
338function stats(e)
339 rows=e.history
340 n=length(rows)
341 total=sum(r.count for r in rows; init = 0)
342 n==0 ? emit("moves 0") :
343 emit("moves", n, "flips", total, "avg", average(total, n), "passes", count(r->r.passed, rows))
344 for s in ['B', 'W']
345 rs=filter(r->r.side==s, rows)
346 if isempty(rs)
347 emit(s, "moves 0")
348 continue
349 end
350 total=sum(r.count for r in rs)
351 max=maximum(r.count for r in rs)
352 first=rs[findfirst(r->r.count==max, rs)]
353 emit(
354 s,
355 "moves",
356 length(rs),
357 "flips",
358 total,
359 "avg",
360 average(total, length(rs)),
361 "max",
362 max,
363 square_name(first.square),
364 "corners",
365 count(r->r.square in [0, 7, 56, 63], rs),
366 )
367 end
368end
369function autoplay!(e, black, white, seed = 0)
370 rng=Minstd(seed)
371 ps=String[]
372 passes=0
373 while !e.over
374 ms=legal(e.board, e.side)
375 p=seed!=0 ? ms[mod(draw!(rng), length(ms))+1] :
376 ranked(e.board, e.side, e.side=='B' ? black : white)[1][2]
377 push!(ps, square_name(p))
378 passes+=move!(e, p, true)
379 end
380 emit(seed!=0 ? "random" : "autoplay", length(ps), "moves", passes, "passes")
381 emit("moves", join(ps))
382 emit(game_over(e.board))
383end
384function monte_carlo(e, k, seed)
385 rng=Minstd(seed)
386 best=-1
387 bp=-1
388 for p in legal(e.board, e.side)
389 w=d=l=0
390 start=place(e.board, e.side, p)
391 for _ in 1:k
392 b=random_game(start, other(e.side), rng)
393 diff=count(==(e.side), b)-count(==(other(e.side)), b)
394 if diff>0
395 w+=1
396 elseif diff<0
397 l+=1
398 else
399 d+=1
400 end
401 end
402 emit(square_name(p), w, d, l)
403 if 2*w+d>best
404 best=2*w+d
405 bp=p
406 end
407 end
408 emit("best", square_name(bp), best)
409end
410function execute!(e, text)
411 a=String.(split(text))
412 cmd=popfirst!(a)
413 error=""
414 if !(cmd in COMMANDS)
415 error="unknown command"
416 elseif cmd=="LOAD"
417 !(length(a)==2&&occursin(r"^[.BW]{64}$", a[1])&&a[2] in ["B", "W"])&&(error="bad board")
418 elseif cmd in ["MOVE", "FLIPS"]
419 error=length(a)!=1 ? "bad arguments" : occursin(r"^[a-h][1-8]$", a[1]) ? "" : "bad square"
420 elseif cmd in ["PLAY", "RECORD"]
421 error=length(a)!=1 ? "bad arguments" :
422 length(a[1])<=120&&occursin(r"^([a-h][1-8])+$", a[1]) ? "" : "bad record"
423 elseif cmd in ZERO
424 !isempty(a)&&(error="bad arguments")
425 elseif cmd=="TRANSFORM"
426 !(length(a)==1&&a[1] in TRANSFORMS)&&(error="bad arguments")
427 else
428 expected=cmd in ["AUTOPLAY", "MC"] ? 2 : 1
429 valid=length(a)==expected&&all(s->occursin(r"^[0-9]+$", s), a)
430 if valid
431 n=parse.(Int, a)
432 valid=if cmd=="AUTOPLAY"
433 all(x->1<=x<=4, n)
434 elseif cmd in ["HINT", "PV", "AI", "ANALYZE"]
435 1<=n[1]<=4
436 elseif cmd in ["PERFT", "DIVIDE"]
437 1<=n[1]<=8
438 elseif cmd=="MC"
439 1<=n[1]<=100&&1<=n[2]<=2147483646
440 elseif cmd=="RANDOM"
441 1<=n[1]<=2147483646
442 else
443 n[1]>=1
444 end
445 end
446 !valid&&(error="bad arguments")
447 end
448 if !isempty(error)
449 emit("ERROR", error)
450 return
451 end
452 if e.over&&cmd in ACTIVE
453 emit("ERROR game over")
454 return
455 end
456 n=!isempty(a)&&occursin(r"^[0-9]+$", a[1]) ? parse(Int, a[1]) : 0
457 if cmd=="NEW"
458 reset!(e)
459 emit("OK")
460 elseif cmd=="LOAD"
461 restore!(e, Position(a[1], a[2][1], false))
462 empty!(e.history)
463 empty!(e.redo)
464 emit("OK")
465 adjust!(e)
466 elseif cmd=="DUMP"
467 emit("board", e.board, e.side)
468 elseif cmd=="SHOW"
469 emit(" a b c d e f g h")
470 ms=e.over ? Int[] : legal(e.board, e.side)
471 for r in 0:7
472 emit(r+1, [8*r+c in ms ? '*' : e.board[8*r+c+1] for c in 0:7]...)
473 end
474 emit(status(e))
475 elseif cmd=="SCORE"
476 emit(
477 "score B",
478 count(==('B'), e.board),
479 "W",
480 count(==('W'), e.board),
481 "empty",
482 count(==('.'), e.board),
483 )
484 elseif cmd=="LEGAL"
485 ms=e.over ? Int[] : legal(e.board, e.side)
486 emit("legal", length(ms), square_name.(ms)...)
487 elseif cmd in ["MOVE", "FLIPS"]
488 p=square_index(a[1])
489 why=reason(e, p)
490 if !isempty(why)
491 emit("ILLEGAL", why)
492 elseif cmd=="MOVE"
493 move!(e, p)
494 else
495 f=flips(e.board, e.side, p)
496 emit("flips", a[1], length(f), square_name.(f)...)
497 end
498 elseif cmd=="PLAY"
499 for (i, j) in enumerate(1:2:length(a[1]))
500 p=square_index(a[1][j:(j+1)])
501 why=reason(e, p)
502 if !isempty(why)
503 emit("ILLEGAL move", i, square_name(p), why)
504 break
505 end
506 move!(e, p)
507 end
508 elseif cmd=="RECORD"
509 record!(e, a[1])
510 elseif cmd=="UNDO"
511 if n>length(e.history)
512 emit("ERROR cannot undo")
513 else
514 start=length(e.history)-n+1
515 removed=e.history[start:end]
516 restore!(e, removed[1].before)
517 resize!(e.history, start-1)
518 e.redo=vcat(removed, e.redo)
519 emit(status(e))
520 end
521 elseif cmd=="REDO"
522 if n>length(e.redo)
523 emit("ERROR cannot redo")
524 else
525 added=e.redo[1:n]
526 e.redo=e.redo[(n+1):end]
527 append!(e.history, added)
528 restore!(e, added[end].after)
529 emit(status(e))
530 end
531 elseif cmd=="HISTORY"
532 emit("history", length(e.history))
533 for (i, r) in enumerate(e.history)
534 r.passed ? emit(i, r.side, square_name(r.square), r.count, "pass") :
535 emit(i, r.side, square_name(r.square), r.count)
536 end
537 elseif cmd=="SHEET"
538 sheet(e)
539 elseif cmd=="STATS"
540 stats(e)
541 elseif cmd=="TRANSFORM"
542 op=a[1]
543 e.board=transform_board(e.board, op)
544 tr(r) = Move(
545 r.side,
546 transform(r.square, op),
547 r.count,
548 r.passed,
549 Position(transform_board(r.before.board, op), r.before.side, r.before.over),
550 Position(transform_board(r.after.board, op), r.after.side, r.after.over),
551 )
552 e.history=tr.(e.history)
553 e.redo=tr.(e.redo)
554 emit("OK")
555 elseif cmd=="EVAL"
556 v, w, x, y=evaluate(e.board, e.side)
557 emit("eval", v, "weights", w, "mobility", x, y)
558 elseif cmd=="STABLE"
559 b, w=stable(e.board)
560 emit("stable B", b, "W", w)
561 elseif cmd=="HINT"
562 for (v, p) in ranked(e.board, e.side, n)
563 emit(square_name(p), v)
564 end
565 elseif cmd=="PV"
566 steps, _=variation(e.board, e.side, n)
567 emit("pv", search(e.board, e.side, n), steps...)
568 elseif cmd=="AI"
569 v, p=ranked(e.board, e.side, n)[1]
570 move!(e, p, false, v)
571 elseif cmd=="ANALYZE"
572 loss=Dict('B'=>0, 'W'=>0)
573 for (i, r) in enumerate(e.history)
574 rows=ranked(r.before.board, r.side, n)
575 best, bp=rows[1]
576 v=rows[findfirst(x->x[2]==r.square, rows)][1]
577 loss[r.side]+=best-v
578 emit(i, r.side, square_name(r.square), v, "best", square_name(bp), best)
579 end
580 emit("loss B", loss['B'], "W", loss['W'])
581 elseif cmd=="SOLVE"
582 if count(==('.'), e.board)>=9
583 emit("ERROR too many empties")
584 else
585 for (v, p) in ranked(e.board, e.side, -1)
586 emit(square_name(p), v)
587 end
588 steps, b=variation(e.board, e.side, -1)
589 emit("line", steps...)
590 emit("final B", count(==('B'), b), "W", count(==('W'), b))
591 end
592 elseif cmd=="AUTOPLAY"
593 autoplay!(e, n, parse(Int, a[2]))
594 elseif cmd=="RANDOM"
595 autoplay!(e, 0, 0, n)
596 elseif cmd=="MC"
597 monte_carlo(e, n, parse(Int, a[2]))
598 elseif cmd=="PERFT"
599 emit("perft", n, perft(e.board, e.side, n))
600 elseif cmd=="DIVIDE"
601 total=0
602 for p in legal(e.board, e.side)
603 v=perft(place(e.board, e.side, p), other(e.side), n-1)
604 emit(square_name(p), v)
605 total+=v
606 end
607 emit("total", total)
608 end
609end
610function main()
611 e=Othello()
612 for _ in 1:parse(Int, readline())
613 execute!(e, readline())
614 end
615end
