96fn flips(board: *const Board, side: u8, p: usize) Squares {
97 var result: Squares = .{};
98 if (board[p] != '.') return result;
99 const r0: i32 = @intCast(p / 8);
100 const c0: i32 = @intCast(p % 8);
101 for (0..3) |ri| {
102 const dr: i32 = @as(i32, @intCast(ri)) - 1;
103 for (0..3) |ci| {
104 const dc: i32 = @as(i32, @intCast(ci)) - 1;
105 if (dr == 0 and dc == 0) continue;
106 var r = r0 + dr;
107 var c = c0 + dc;
108 var line: Squares = .{};
109 while (inside(r, c) and board[@intCast(8 * r + c)] == other(side)) {
110 line.add(@intCast(8 * r + c));
111 r += dr;
112 c += dc;
113 }
114 if (line.len > 0 and inside(r, c) and board[@intCast(8 * r + c)] == side) {
115 for (line.data[0..line.len]) |q| result.add(q);
116 }
117 }
118 }
119 for (0..result.len) |i| {
120 var j = i;
121 while (j > 0 and result.data[j] < result.data[j - 1]) : (j -= 1) {
122 std.mem.swap(usize, &result.data[j], &result.data[j - 1]);
123 }
124 }
125 return result;
126}

128fn legal(board: *const Board, side: u8) Squares {
129 var moves: Squares = .{};
130 for (0..64) |p| {
131 if (board[p] == '.' and flips(board, side, p).len > 0) moves.add(p);
132 }
133 return moves;

140 result[p] = side;
141 return result;

156fn gameOver(board: *const Board, out: *Builder) !void {
157 const black = count(board, 'B');
158 const white = count(board, 'W');
159 const result = if (black > white) "winner B" else if (black < white) "winner W" else "draw";
160 try out.line("game over B {d} W {d} {s}", .{ black, white, result });
161}

163fn evaluate(board: *const Board, side: u8) [4]i32 {
164 var w: i32 = 0;
165 for (board.*, 0..) |disc, p| {
166 if (disc == side) {
167 w += weights[p];
168 } else if (disc == other(side)) {
169 w -= weights[p];
170 }
171 }
172 const own: i32 = @intCast(legal(board, side).len);
173 const enemy: i32 = @intCast(legal(board, other(side)).len);
174 return .{ w + 10 * (own - enemy), w, own, enemy };
175}

183const Solver = struct {

187 return .{ .cache = std.AutoHashMap(SearchKey, i32).init(gpa) };

190 fn search(self: *Solver, board: Board, side: u8, depth: i32) !i32 {
191 const key = SearchKey{ .board = board, .side = side, .depth = depth };
192 if (self.cache.get(key)) |v| return v;
193 const moves = legal(&board, side);
194 const enemy = other(side);
195 const next_depth = if (depth < 0) @as(i32, -1) else depth - 1;
196 var result: i32 = 0;
197 if (moves.len == 0 and legal(&board, enemy).len == 0) {
198 const black: i32 = @intCast(count(&board, side));
199 const white: i32 = @intCast(count(&board, enemy));
200 result = (if (depth < 0) @as(i32, 1) else 1000) * (black - white);
201 } else if (depth == 0) {
202 result = evaluate(&board, side)[0];
203 } else if (moves.len == 0) {
204 result = -try self.search(board, enemy, next_depth);
205 } else {
206 result = -1000000;
207 for (moves.data[0..moves.len]) |p| {
208 const v = -try self.search(place(&board, side, p), enemy, next_depth);
209 result = @max(result, v);
210 }
211 }
212 if (self.cache.count() > 300000) self.cache.clearRetainingCapacity();
213 try self.cache.put(key, result);
214 return result;
215 }
216
217 fn ranked(self: *Solver, board: Board, side: u8, depth: i32) !Rankings {
218 var result: Rankings = .{};
219 const moves = legal(&board, side);
220 for (moves.data[0..moves.len]) |p| {
221 const value = -try self.search(place(&board, side, p), other(side), if (depth < 0) -1 else depth - 1);
222 result.data[result.len] = .{ .p = p, .value = value };
223 result.len += 1;
224 }
225 for (0..result.len) |i| {
226 var j = i;
227 while (j > 0 and (result.data[j].value > result.data[j - 1].value or
228 (result.data[j].value == result.data[j - 1].value and result.data[j].p < result.data[j - 1].p))) : (j -= 1)
229 {
230 std.mem.swap(Ranking, &result.data[j], &result.data[j - 1]);
231 }
232 }
233 return result;
234 }

236 fn variation(self: *Solver, start: Board, side: u8, depth: i32, gpa: Allocator) !Variation {
237 var board = start;
238 var turn = side;
239 var remaining = depth;
240 var path: std.ArrayList(u8) = .empty;
241 while (remaining != 0 and !terminal(&board)) {
242 if (path.items.len > 0) try path.append(gpa, ' ');
243 const choices = try self.ranked(board, turn, remaining);
244 if (choices.len == 0) {
245 try path.appendSlice(gpa, "pass");
246 } else {
247 const p = choices.data[0].p;
248 const sq = squareName(p);
249 try path.appendSlice(gpa, &sq);
250 board = place(&board, turn, p);
251 }
252 turn = other(turn);
253 if (remaining > 0) remaining -= 1;
254 }
255 return .{ .path = path.items, .board = board };
256 }

258 fn perft(self: *Solver, board: Board, side: u8, depth: u32) !u64 {
259 if (depth == 0) return 1;
260 const moves = legal(&board, side);
261 const enemy = other(side);
262 if (moves.len == 0) {
263 return if (legal(&board, enemy).len == 0) 1 else try self.perft(board, enemy, depth - 1);
264 }
265 var result: u64 = 0;
266 for (moves.data[0..moves.len]) |p| result += try self.perft(place(&board, side, p), enemy, depth - 1);
267 return result;
268 }

286fn stable(board: *const Board) [2]usize {
287 var known: [64]bool = @splat(false);
288 const axes = [_][2]i32{ .{ 0, 1 }, .{ 1, 0 }, .{ 1, 1 }, .{ 1, -1 } };
289 while (true) {
290 var found: Squares = .{};
291 for (0..64) |p| {
292 if (board[p] == '.' or known[p]) continue;
293 var valid = true;
294 for (axes) |axis| {
295 var full = true;
296 var anchor = false;
297 for ([_]i32{ -1, 1 }) |sign| {
298 var r: i32 = @as(i32, @intCast(p / 8)) + axis[0] * sign;
299 var c: i32 = @as(i32, @intCast(p % 8)) + axis[1] * sign;
300 if (!inside(r, c)) {
301 anchor = true;
302 } else if (known[@intCast(8 * r + c)] and board[@intCast(8 * r + c)] == board[p]) {
303 anchor = true;
304 }
305 while (inside(r, c)) {
306 if (board[@intCast(8 * r + c)] == '.') full = false;
307 r += axis[0] * sign;
308 c += axis[1] * sign;
309 }
310 }
311 if (!full and !anchor) {
312 valid = false;
313 break;
314 }
315 }
316 if (valid) found.add(p);
317 }
318 if (found.len == 0) break;
319 for (found.data[0..found.len]) |p| known[p] = true;
320 }
321 var result = [_]usize{ 0, 0 };
322 for (0..64) |p| {
323 if (known[p] and board[p] == 'B') result[0] += 1;
324 if (known[p] and board[p] == 'W') result[1] += 1;
325 }
326 return result;
327}

329fn transform(p: usize, operation: Text) usize {
330 const r = p / 8;
331 const c = p % 8;
332 if (eq(operation, "rot90")) return 8 * c + 7 - r;
333 if (eq(operation, "rot180")) return 63 - p;
334 if (eq(operation, "rot270")) return 8 * (7 - c) + r;
335 if (eq(operation, "mirror")) return 8 * r + 7 - c;
336 if (eq(operation, "flip")) return 8 * (7 - r) + c;
337 if (eq(operation, "diag")) return 8 * c + r;
338 return 8 * (7 - c) + 7 - r;
339}

341fn transformBoard(board: *const Board, operation: Text) Board {
342 var result: [64]u8 = @splat('.');
343 for (0..64) |p| result[transform(p, operation)] = board[p];
344 return result;
345}

388fn allowedNumber(cmd: Text, args: []const Text) bool {
389 const expected: usize = if (eq(cmd, "AUTOPLAY") or eq(cmd, "MC")) 2 else 1;
390 if (args.len != expected) return false;
391 const a = number(args[0]) orelse return false;
392 const b = if (expected == 2) (number(args[1]) orelse return false) else 0;
393 if (eq(cmd, "AUTOPLAY")) return a >= 1 and a <= 4 and b >= 1 and b <= 4;
394 if (includes(cmd, &.{ "HINT", "PV", "AI", "ANALYZE" })) return a >= 1 and a <= 4;
395 if (includes(cmd, &.{ "PERFT", "DIVIDE" })) return a >= 1 and a <= 8;
396 if (eq(cmd, "MC")) return a >= 1 and a <= 100 and b >= 1 and b <= 2147483646;
397 if (eq(cmd, "RANDOM")) return a >= 1 and a <= 2147483646;
398 return a >= 1;
399}

442 fn adjust(self: *Game, out: *Builder, quiet: bool) !bool {
443 if (legal(&self.pos.board, self.pos.side).len > 0) return false;
444 if (legal(&self.pos.board, other(self.pos.side)).len > 0) {
445 if (!quiet) try out.line("{c} passes", .{self.pos.side});
446 self.pos.side = other(self.pos.side);
447 return true;
448 }
449 self.pos.over = true;
450 if (!quiet) try gameOver(&self.pos.board, out);
451 return false;

454 fn play(self: *Game, p: usize, out: *Builder, quiet: bool, value: ?i32) !bool {
455 const before = self.pos;
456 const s = self.pos.side;
457 const flipped = flips(&self.pos.board, s, p).len;
458 self.pos.board = place(&self.pos.board, s, p);
459 if (!quiet) {
460 const sq = squareName(p);
461 if (value) |v| try out.line("{c} {s} flips {d} value {d}", .{ s, &sq, flipped, v }) else try out.line("{c} {s} flips {d}", .{ s, &sq, flipped });
462 }
463 self.pos.side = other(s);
464 const passed = try self.adjust(out, quiet);
465 self.history[self.nh] = .{ .side = s, .p = p, .flipped = flipped, .passed = passed, .before = before, .after = self.pos };
466 self.nh += 1;
467 self.nr = 0;
468 return passed;
469 }

478 fn record(self: *Game, text: Text, out: *Builder) !void {
479 var trial = Game.init(out.gpa);
480 var passes: usize = 0;
481 var squares: [60]usize = undefined;
482 for (0..text.len / 2) |i| {
483 const p = squareIndex(text[2 * i .. 2 * i + 2]);
484 if (trial.reason(p)) |why| {
485 const sq = squareName(p);
486 try out.line("ILLEGAL move {d} {s} {s}", .{ i + 1, &sq, why });
487 return;
488 }
489 squares[i] = p;
490 if (try trial.play(p, out, true, null)) passes += 1;
491 }
492 self.pos = trial.pos;
493 for (trial.history[0..trial.nh], 0..) |m, i| self.history[i] = m;
494 self.nh = trial.nh;
495 self.nr = 0;
496 const first = squareName(squares[0]);
497 const operation: Text = if (eq(&first, "e6")) "diag" else if (eq(&first, "d3")) "anti" else if (eq(&first, "c4")) "rot180" else "";
498 var normalized: [120]u8 = undefined;
499 for (squares[0 .. text.len / 2], 0..) |p, i| {
500 const sq = squareName(if (operation.len == 0) p else transform(p, operation));
501 normalized[2 * i] = sq[0];
502 normalized[2 * i + 1] = sq[1];
503 }
504 try out.line("record {d} moves {d} passes", .{ trial.nh, passes });
505 try out.line("normal {s}", .{normalized[0..text.len]});
506 try self.status(out);
507 }

509 fn sheet(self: *const Game, out: *Builder) !void {
510 const board = if (self.nh == 0) self.pos.board else self.history[0].before.board;
511 var fields: [64]Text = undefined;
512 for (0..64) |p| {
513 fields[p] = try std.fmt.allocPrint(out.gpa, "{c}", .{board[p]});
514 }
515 for (0..self.nh) |i| fields[self.history[i].p] = try std.fmt.allocPrint(out.gpa, "{d}", .{i + 1});
516 try out.line(" a b c d e f g h", .{});
517 for (0..8) |r| {
518 try out.print("{d}", .{r + 1});
519 for (fields[8 * r .. 8 * r + 8]) |item| try out.print("{s: >3}", .{item});
520 try out.add("\n");
521 }

524 fn stats(self: *const Game, out: *Builder) !void {
525 var flipped: usize = 0;
526 var passes: usize = 0;
527 for (self.history[0..self.nh]) |m| {
528 flipped += m.flipped;
529 if (m.passed) passes += 1;
530 }
531 if (self.nh == 0) try out.line("moves 0", .{}) else try out.line("moves {d} flips {d} avg {d}.{d:0>2} passes {d}", .{
532 self.nh, flipped, flipped * 100 / self.nh / 100, (flipped * 100 / self.nh) % 100, passes,
533 });
534 for ([_]u8{ 'B', 'W' }) |side| {
535 var n: usize = 0;
536 var total: usize = 0;
537 var highest: usize = 0;
538 var square: usize = 0;
539 var corners: usize = 0;
540 for (self.history[0..self.nh]) |m| {
541 if (m.side != side) continue;
542 n += 1;
543 total += m.flipped;
544 if (n == 1 or m.flipped > highest) {
545 highest = m.flipped;
546 square = m.p;
547 }
548 if (m.p == 0 or m.p == 7 or m.p == 56 or m.p == 63) corners += 1;
549 }
550 if (n == 0) {
551 try out.line("{c} moves 0", .{side});
552 } else {
553 const sq = squareName(square);
554 const scaled = total * 100 / n;
555 try out.line("{c} moves {d} flips {d} avg {d}.{d:0>2} max {d} {s} corners {d}", .{
556 side, n, total, scaled / 100, scaled % 100, highest, &sq, corners,
557 });
558 }
559 }
560 }

576 fn automatic(self: *Game, black: i32, white: i32, seed: ?u64, out: *Builder) !void {
577 var random = Random{ .state = seed orelse 0 };
578 var line: [120]u8 = undefined;
579 var moves: usize = 0;
580 var passes: usize = 0;
581 while (!self.pos.over) {
582 const p = if (seed != null) blk: {
583 const available = legal(&self.pos.board, self.pos.side);
584 break :blk available.data[@intCast(random.draw() % @as(u64, @intCast(available.len)))];
585 } else (try self.solver.ranked(self.pos.board, self.pos.side, if (self.pos.side == 'B') black else white)).data[0].p;
586 const sq = squareName(p);
587 line[2 * moves] = sq[0];
588 line[2 * moves + 1] = sq[1];
589 moves += 1;
590 if (try self.play(p, out, true, null)) passes += 1;
591 }
592 try out.line("{s} {d} moves {d} passes", .{ if (seed != null) "random" else "autoplay", moves, passes });
593 try out.line("moves {s}", .{line[0 .. 2 * moves]});
594 try gameOver(&self.pos.board, out);
595 }

597 fn monteCarlo(self: *const Game, trials: usize, seed: u64, out: *Builder) !void {
598 var random = Random{ .state = seed };
599 var best: i32 = -1;
600 var best_square: usize = 0;
601 const moves = legal(&self.pos.board, self.pos.side);
602 for (moves.data[0..moves.len]) |p| {
603 var wins: i32 = 0;
604 var draws: i32 = 0;
605 var losses: i32 = 0;
606 const first = place(&self.pos.board, self.pos.side, p);
607 for (0..trials) |_| {
608 const last = randomGame(first, other(self.pos.side), &random);
609 const x: i32 = @intCast(count(&last, self.pos.side));
610 const y: i32 = @intCast(count(&last, other(self.pos.side)));
611 if (x > y) wins += 1 else if (x == y) draws += 1 else losses += 1;
612 }
613 const sq = squareName(p);
614 try out.line("{s} {d} {d} {d}", .{ &sq, wins, draws, losses });
615 const points = 2 * wins + draws;
616 if (points > best) {
617 best = points;
618 best_square = p;
619 }
620 }
621 const sq = squareName(best_square);
622 try out.line("best {s} {d}", .{ &sq, best });
623 }
624
625 fn execute(self: *Game, cmdline: Text, out: *Builder) !void {
626 const words = try split(out.gpa, cmdline);
627 const cmd = words[0];
628 const args = words[1..];
629 const zero = [_]Text{ "NEW", "DUMP", "SHOW", "SCORE", "LEGAL", "HISTORY", "SHEET", "STATS", "EVAL", "STABLE", "SOLVE" };
630 const numeric = [_]Text{ "UNDO", "REDO", "HINT", "PV", "AI", "ANALYZE", "RANDOM", "PERFT", "DIVIDE" };
631 const additional = [_]Text{ "LOAD", "TRANSFORM", "AUTOPLAY", "MC", "FLIPS", "MOVE", "PLAY", "RECORD" };
632 const active = [_]Text{ "FLIPS", "MOVE", "PLAY", "HINT", "PV", "AI", "SOLVE", "AUTOPLAY", "RANDOM", "MC", "DIVIDE" };
633 if (!includes(cmd, &zero) and !includes(cmd, &numeric) and !includes(cmd, &additional)) {
634 try out.line("ERROR unknown command", .{});
635 return;
636 }
637 var error_message: ?Text = null;
638 if (eq(cmd, "LOAD")) {
639 if (args.len != 2 or args[0].len != 64) {
640 error_message = "bad board";
641 } else {
642 for (args[0]) |c| {
643 if (c != '.' and c != 'B' and c != 'W') error_message = "bad board";
644 }
645 if (!eq(args[1], "B") and !eq(args[1], "W")) error_message = "bad board";
646 }
647 } else if (eq(cmd, "MOVE") or eq(cmd, "FLIPS")) {
648 if (args.len != 1) error_message = "bad arguments" else if (!validSquare(args[0])) error_message = "bad square";
649 } else if (eq(cmd, "PLAY") or eq(cmd, "RECORD")) {
650 if (args.len != 1) error_message = "bad arguments" else if (!validRecord(args[0])) error_message = "bad record";
651 } else if (includes(cmd, &zero)) {
652 if (args.len != 0) error_message = "bad arguments";
653 } else if (eq(cmd, "TRANSFORM")) {
654 if (args.len != 1 or !includes(args[0], &.{ "rot90", "rot180", "rot270", "mirror", "flip", "diag", "anti" }))
655 error_message = "bad arguments";
656 } else if (!allowedNumber(cmd, args)) {
657 error_message = "bad arguments";
658 }
659 if (error_message) |msg| {
660 try out.line("ERROR {s}", .{msg});
661 return;
662 }
663 if (self.pos.over and includes(cmd, &active)) {
664 try out.line("ERROR game over", .{});
665 return;
666 }
667 const n: usize = if (includes(cmd, &numeric) or eq(cmd, "MC") or eq(cmd, "AUTOPLAY")) @intCast(number(args[0]).?) else 0;
668 if (eq(cmd, "NEW")) {
669 self.reset();
670 try out.line("OK", .{});
671 } else if (eq(cmd, "LOAD")) {
672 @memcpy(self.pos.board[0..], args[0]);
673 self.pos.side = args[1][0];
674 self.pos.over = false;
675 self.nh = 0;
676 self.nr = 0;
677 try out.line("OK", .{});
678 _ = try self.adjust(out, false);
679 } else if (eq(cmd, "DUMP")) {
680 try out.line("board {s} {c}", .{ &self.pos.board, self.pos.side });
681 } else if (eq(cmd, "SHOW")) {
682 try out.line(" a b c d e f g h", .{});
683 const moves = if (self.pos.over) Squares{} else legal(&self.pos.board, self.pos.side);
684 for (0..8) |r| {
685 try out.print("{d}", .{r + 1});
686 for (0..8) |c| {
687 const p = 8 * r + c;
688 try out.print(" {c}", .{if (moves.has(p)) @as(u8, '*') else self.pos.board[p]});
689 }
690 try out.add("\n");
691 }
692 try self.status(out);
693 } else if (eq(cmd, "SCORE")) {
694 try out.line("score B {d} W {d} empty {d}", .{
695 count(&self.pos.board, 'B'), count(&self.pos.board, 'W'), count(&self.pos.board, '.'),
696 });
697 } else if (eq(cmd, "LEGAL")) {
698 const moves = if (self.pos.over) Squares{} else legal(&self.pos.board, self.pos.side);
699 try out.print("legal {d}", .{moves.len});
700 for (moves.data[0..moves.len]) |p| {
701 const sq = squareName(p);
702 try out.print(" {s}", .{&sq});
703 }
704 try out.add("\n");
705 } else if (eq(cmd, "MOVE") or eq(cmd, "FLIPS")) {
706 const p = squareIndex(args[0]);
707 if (self.reason(p)) |why| try out.line("ILLEGAL {s}", .{why}) else if (eq(cmd, "MOVE")) _ = try self.play(p, out, false, null) else {
708 const flipped = flips(&self.pos.board, self.pos.side, p);
709 try out.print("flips {s} {d}", .{ args[0], flipped.len });
710 for (flipped.data[0..flipped.len]) |q| {
711 const sq = squareName(q);
712 try out.print(" {s}", .{&sq});
713 }
714 try out.add("\n");
715 }
716 } else if (eq(cmd, "PLAY")) {
717 for (0..args[0].len / 2) |i| {
718 const p = squareIndex(args[0][2 * i .. 2 * i + 2]);
719 if (self.reason(p)) |why| {
720 const sq = squareName(p);
721 try out.line("ILLEGAL move {d} {s} {s}", .{ i + 1, &sq, why });
722 break;
723 }
724 _ = try self.play(p, out, false, null);
725 }
726 } else if (eq(cmd, "RECORD")) {
727 try self.record(args[0], out);
728 } else if (eq(cmd, "UNDO")) {
729 if (n > self.nh) {
730 try out.line("ERROR cannot undo", .{});
731 } else {
732 const start = self.nh - n;
733 self.pos = self.history[start].before;
734 var i = self.nr;
735 while (i > 0) {
736 i -= 1;
737 self.redo[i + n] = self.redo[i];
738 }
739 for (0..n) |j| self.redo[j] = self.history[start + j];
740 self.nh = start;
741 self.nr += n;
742 try self.status(out);
743 }
744 } else if (eq(cmd, "REDO")) {
745 if (n > self.nr) {
746 try out.line("ERROR cannot redo", .{});
747 } else {
748 for (0..n) |j| self.history[self.nh + j] = self.redo[j];
749 self.nh += n;
750 self.pos = self.redo[n - 1].after;
751 self.nr -= n;
752 for (0..self.nr) |j| self.redo[j] = self.redo[n + j];
753 try self.status(out);
754 }
755 } else if (eq(cmd, "HISTORY")) {
756 try out.line("history {d}", .{self.nh});
757 for (self.history[0..self.nh], 0..) |m, i| {
758 const sq = squareName(m.p);
759 try out.line("{d} {c} {s} {d}{s}", .{ i + 1, m.side, &sq, m.flipped, if (m.passed) " pass" else "" });
760 }
761 } else if (eq(cmd, "SHEET")) {
762 try self.sheet(out);
763 } else if (eq(cmd, "STATS")) {
764 try self.stats(out);
765 } else if (eq(cmd, "TRANSFORM")) {
766 self.transformGame(args[0]);
767 try out.line("OK", .{});
768 } else if (eq(cmd, "EVAL")) {
769 const values = evaluate(&self.pos.board, self.pos.side);
770 try out.line("eval {d} weights {d} mobility {d} {d}", .{ values[0], values[1], values[2], values[3] });
771 } else if (eq(cmd, "STABLE")) {
772 const values = stable(&self.pos.board);
773 try out.line("stable B {d} W {d}", .{ values[0], values[1] });
774 } else if (eq(cmd, "HINT")) {
775 const choices = try self.solver.ranked(self.pos.board, self.pos.side, @intCast(n));
776 for (choices.data[0..choices.len]) |choice| {
777 const sq = squareName(choice.p);
778 try out.line("{s} {d}", .{ &sq, choice.value });
779 }
780 } else if (eq(cmd, "PV")) {
781 const value = try self.solver.search(self.pos.board, self.pos.side, @intCast(n));
782 const pv = try self.solver.variation(self.pos.board, self.pos.side, @intCast(n), out.gpa);
783 try out.line("pv {d} {s}", .{ value, pv.path });
784 } else if (eq(cmd, "AI")) {
785 const choices = try self.solver.ranked(self.pos.board, self.pos.side, @intCast(n));
786 _ = try self.play(choices.data[0].p, out, false, choices.data[0].value);
787 } else if (eq(cmd, "ANALYZE")) {
788 var losses = [_]i32{ 0, 0 };
789 for (self.history[0..self.nh], 0..) |m, i| {
790 const choices = try self.solver.ranked(m.before.board, m.side, @intCast(n));
791 const best = choices.data[0];
792 var played: i32 = best.value;
793 for (choices.data[0..choices.len]) |choice| {
794 if (choice.p == m.p) played = choice.value;
795 }
796 losses[if (m.side == 'B') 0 else 1] += best.value - played;
797 const sq = squareName(m.p);
798 const winner = squareName(best.p);
799 try out.line("{d} {c} {s} {d} best {s} {d}", .{ i + 1, m.side, &sq, played, &winner, best.value });
800 }
801 try out.line("loss B {d} W {d}", .{ losses[0], losses[1] });
802 } else if (eq(cmd, "SOLVE")) {
803 if (count(&self.pos.board, '.') >= 9) {
804 try out.line("ERROR too many empties", .{});
805 } else {
806 const choices = try self.solver.ranked(self.pos.board, self.pos.side, -1);
807 for (choices.data[0..choices.len]) |choice| {
808 const sq = squareName(choice.p);
809 try out.line("{s} {d}", .{ &sq, choice.value });
810 }
811 const pv = try self.solver.variation(self.pos.board, self.pos.side, -1, out.gpa);
812 try out.line("line {s}", .{pv.path});
813 try out.line("final B {d} W {d}", .{ count(&pv.board, 'B'), count(&pv.board, 'W') });
814 }
815 } else if (eq(cmd, "AUTOPLAY")) {
816 try self.automatic(@intCast(n), @intCast(number(args[1]).?), null, out);
817 } else if (eq(cmd, "RANDOM")) {
818 try self.automatic(0, 0, n, out);
819 } else if (eq(cmd, "MC")) {
820 try self.monteCarlo(n, number(args[1]).?, out);
821 } else if (eq(cmd, "PERFT")) {
822 try out.line("perft {d} {d}", .{ n, try self.solver.perft(self.pos.board, self.pos.side, @intCast(n)) });
823 } else if (eq(cmd, "DIVIDE")) {
824 const moves = legal(&self.pos.board, self.pos.side);
825 var total: u64 = 0;
826 for (moves.data[0..moves.len]) |p| {
827 const value = try self.solver.perft(place(&self.pos.board, self.pos.side, p), other(self.pos.side), @intCast(n - 1));
828 const sq = squareName(p);
829 try out.line("{s} {d}", .{ &sq, value });
830 total += value;
831 }
832 try out.line("total {d}", .{total});
833 }
834 }
835};

837pub fn main(init: std.process.Init) !void {
838 var arena = std.heap.ArenaAllocator.init(std.heap.page_allocator);
839 defer arena.deinit();
840 const gpa = arena.allocator();
841 var input_buffer: [8192]u8 = undefined;
842 var input = std.Io.File.stdin().readerStreaming(init.io, &input_buffer);
843 const text = try input.interface.allocRemaining(gpa, .unlimited);
844 var lines = std.mem.splitScalar(u8, text, '\n');
845 const commands = try std.fmt.parseInt(usize, std.mem.trim(u8, lines.next().?, "\r"), 10);
846 var out: Builder = .{ .gpa = gpa };
847 var game = Game.init(gpa);
848 for (0..commands) |_| {
849 try game.execute(std.mem.trimEnd(u8, lines.next().?, "\r"), &out);
850 }
851 var output_buffer: [8192]u8 = undefined;
852 var output = std.Io.File.stdout().writerStreaming(init.io, &output_buffer);
853 try output.interface.writeAll(out.bytes.items);
854 try output.interface.flush();
855}
